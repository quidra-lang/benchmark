35fn integer(n: usize) i64 {
36 return @intCast(n);
37}
38fn magnitude(n: i64) i64 {
39 return if (n < 0) -n else n;
40}
41fn gcd(a: i64, b: i64) i64 {
42 var x = magnitude(a);
43 var y = magnitude(b);
44 while (y != 0) {
45 const remainder = @mod(x, y);
46 x = y;
47 y = remainder;
48 }
49 return x;
50}
51fn isqrt(n: usize) usize {
52 var result: usize = 1;
53 while ((result + 1) * (result + 1) <= n) : (result += 1) {}
54 return result;
55}
56fn log2(n: usize) usize {
57 var result: usize = 0;
58 var power: usize = 1;
59 while (power * 2 <= n) {
60 result += 1;
61 power *= 2;
62 }
63 return result;
64}
65fn power2(n: usize) usize {
66 return @as(usize, 1) << @intCast(n);
67}
68fn lowbit(n: usize) usize {
69 return n & (~n + 1);
70}

82fn Owned(comptime T: type) type {
83 return struct {
84 arena: std.heap.ArenaAllocator,
85 value: T,
86 const Self = @This();
87 fn create(args: anytype) Allocator.Error!*Self {
88 const self = try std.heap.page_allocator.create(Self);
89 self.* = .{ .arena = .init(std.heap.page_allocator), .value = undefined };
90 errdefer self.destroy();
91 self.value = try @call(.auto, T.init, .{self.arena.allocator()} ++ args);
92 return self;
93 }
94 fn destroy(self: *Self) void {
95 self.arena.deinit();
96 std.heap.page_allocator.destroy(self);
97 }
98 };
99}
100fn replace(comptime T: type, slot: *?*Owned(T), args: anytype) Allocator.Error!void {
101 const replacement = try Owned(T).create(args);
102 if (slot.*) |old| old.destroy();
103 slot.* = replacement;
104}

106const Interval = struct {
107 left: usize,
108 right: usize,
109 fn leaf(range: Interval) bool {
110 return range.left == range.right;
111 }
112 fn middle(range: Interval) usize {
113 return (range.left + range.right) / 2;
114 }
115 fn length(range: Interval) usize {
116 return range.right - range.left + 1;
117 }
118 fn covered(range: Interval, left: usize, right: usize) bool {
119 return left <= range.left and range.right <= right;
120 }
121};
122const Layout = struct {
123 ranges: []?Interval,
124 count: usize,
125 max_node: usize = 0,
126 fn init(gpa: Allocator, n: usize) Allocator.Error!Layout {
127 const ranges = try gpa.alloc(?Interval, 4 * n);
128 @memset(ranges, null);
129 var shape: Layout = .{ .ranges = ranges, .count = 2 * n - 1 };
130 shape.build(1, 0, n - 1);
131 return shape;
132 }
133 fn build(shape: *Layout, node: usize, left: usize, right: usize) void {
134 shape.ranges[node] = .{ .left = left, .right = right };
135 shape.max_node = @max(shape.max_node, node);
136 if (left == right) return;
137 const middle = (left + right) / 2;
138 shape.build(2 * node, left, middle);
139 shape.build(2 * node + 1, middle + 1, right);
140 }
141 fn at(shape: Layout, node: usize) Interval {
142 return shape.ranges[node].?;
143 }

159const SegmentTree = struct {
160 op: Operation,
161 shape: Layout,
162 values: []i64,
163 fn init(gpa: Allocator, contents: []const i64, op: Operation) Allocator.Error!SegmentTree {
164 var tree: SegmentTree = .{ .op = op, .shape = try Layout.init(gpa, contents.len), .values = try gpa.alloc(i64, 4 * contents.len) };
165 tree.build(1, contents);
166 return tree;
167 }
168 fn build(tree: *SegmentTree, node: usize, contents: []const i64) void {
169 const range = tree.shape.at(node);
170 if (range.leaf()) {
171 tree.values[node] = if (tree.op == .GCD) magnitude(contents[range.left]) else contents[range.left];
172 } else {
173 tree.build(2 * node, contents);
174 tree.build(2 * node + 1, contents);
175 tree.pull(node);
176 }
177 }
178 fn pull(tree: *SegmentTree, node: usize) void {
179 tree.values[node] = tree.op.combine(tree.values[2 * node], tree.values[2 * node + 1]);
180 }
181 fn query(tree: *const SegmentTree, left: usize, right: usize, node: usize) Answer {
182 const range = tree.shape.at(node);
183 if (range.covered(left, right)) return .{ .value = tree.values[node], .visited = 1 };
184 if (right <= range.middle() or left > range.middle()) {
185 var child = tree.query(left, right, if (right <= range.middle()) 2 * node else 2 * node + 1);
186 child.visited += 1;
187 return child;
188 }
189 const a = tree.query(left, right, 2 * node);
190 const b = tree.query(left, right, 2 * node + 1);
191 return .{ .value = tree.op.combine(a.value, b.value), .visited = a.visited + b.visited + 1 };
192 }
193 fn update(tree: *SegmentTree, index: usize, value: i64, node: usize) usize {
194 const range = tree.shape.at(node);
195 if (range.leaf()) {
196 tree.values[node] = if (tree.op == .GCD) magnitude(value) else value;
197 return 1;
198 }
199 const visited = tree.update(index, value, if (index <= range.middle()) 2 * node else 2 * node + 1);
200 tree.pull(node);
201 return visited + 1;
202 }
203 fn describe(tree: *const SegmentTree, answer: Answer) SegmentAnswer {
204 return .{ .op = tree.op, .answer = answer };
205 }
206 fn dump(tree: *const SegmentTree, out: *Writer) Writer.Error!void {
207 try line(out, "SEG DUMP {s} nodes={d}", .{ @tagName(tree.op), tree.shape.count });
208 for (tree.shape.ranges, 0..) |maybe, node| if (maybe) |range| {
209 try line(out, "NODE {d} {d} {d} {d}", .{ node, range.left, range.right, tree.values[node] });
210 };
211 }

244const LazyTree = struct {
245 shape: Layout,
246 values: []Aggregate,
247 assign: []?i64,
248 add: []i64,
249 fn init(gpa: Allocator, contents: []const i64) Allocator.Error!LazyTree {
250 const size = 4 * contents.len;
251 var tree: LazyTree = .{ .shape = try Layout.init(gpa, contents.len), .values = try gpa.alloc(Aggregate, size), .assign = try gpa.alloc(?i64, size), .add = try gpa.alloc(i64, size) };
252 @memset(tree.assign, null);
253 @memset(tree.add, 0);
254 tree.build(1, contents);
255 return tree;
256 }
257 fn build(tree: *LazyTree, node: usize, contents: []const i64) void {
258 const range = tree.shape.at(node);
259 if (range.leaf()) {
260 const value = contents[range.left];
261 tree.values[node] = .{ .sum = value, .min = value, .max = value };
262 } else {
263 tree.build(2 * node, contents);
264 tree.build(2 * node + 1, contents);
265 tree.pull(node);
266 }
267 }
268 fn pull(tree: *LazyTree, node: usize) void {
269 tree.values[node] = Aggregate.merge(tree.values[2 * node], tree.values[2 * node + 1]);
270 }
271 fn apply(tree: *LazyTree, node: usize, value: i64, assignment: bool) void {
272 const range = tree.shape.at(node);
273 const old = tree.values[node];
274 tree.values[node] = if (assignment) .{ .sum = value * integer(range.length()), .min = value, .max = value } else .{ .sum = old.sum + value * integer(range.length()), .min = old.min + value, .max = old.max + value };
275 if (range.leaf()) return;
276 if (assignment) {
277 tree.assign[node] = value;
278 tree.add[node] = 0;
279 } else tree.add[node] += value;
280 }
281 fn push(tree: *LazyTree, node: usize) usize {
282 if (tree.assign[node] == null and tree.add[node] == 0) return 0;
283 for (2 * node..2 * node + 2) |child| {
284 if (tree.assign[node]) |value| tree.apply(child, value, true);
285 if (tree.add[node] != 0) tree.apply(child, tree.add[node], false);
286 }
287 tree.assign[node] = null;
288 tree.add[node] = 0;
289 return 1;
290 }
291 fn update(tree: *LazyTree, left: usize, right: usize, value: i64, assignment: bool, node: usize, work: *Work) void {
292 work.visited += 1;
293 const range = tree.shape.at(node);
294 if (range.covered(left, right)) {
295 work.covered += 1;
296 tree.apply(node, value, assignment);
297 return;
298 }
299 work.pushes += tree.push(node);
300 if (left <= range.middle()) tree.update(left, right, value, assignment, 2 * node, work);
301 if (right > range.middle()) tree.update(left, right, value, assignment, 2 * node + 1, work);
302 tree.pull(node);
303 }
304 fn query(tree: *LazyTree, left: usize, right: usize, node: usize) LazyAnswer {
305 const range = tree.shape.at(node);
306 if (range.covered(left, right)) return .{ .value = tree.values[node], .visited = 1, .pushes = 0 };
307 const pushes = tree.push(node);
308 if (right <= range.middle() or left > range.middle()) {
309 var child = tree.query(left, right, if (right <= range.middle()) 2 * node else 2 * node + 1);
310 child.visited += 1;
311 child.pushes += pushes;
312 return child;
313 }
314 const a = tree.query(left, right, 2 * node);
315 const b = tree.query(left, right, 2 * node + 1);
316 return .{ .value = Aggregate.merge(a.value, b.value), .visited = a.visited + b.visited + 1, .pushes = a.pushes + b.pushes + pushes };
317 }
318 fn dump(tree: *const LazyTree, out: *Writer) Writer.Error!void {
319 try line(out, "LAZY DUMP nodes={d}", .{tree.shape.count});
320 for (tree.shape.ranges, 0..) |maybe, node| if (maybe) |range| {
321 try out.print("NODE {d} {d} {d} {f} assign=", .{ node, range.left, range.right, tree.values[node] });
322 if (tree.assign[node]) |value| try out.print("{d}", .{value}) else try out.writeByte('-');
323 try line(out, " add={d}", .{tree.add[node]});
324 };
325 }

328const Fenwick = struct {
329 values: []i64,
330 fn init(gpa: Allocator, n: usize) Allocator.Error!Fenwick {
331 const values = try gpa.alloc(i64, n + 1);
332 @memset(values, 0);
333 return .{ .values = values };
334 }
335 fn add(tree: *Fenwick, position: usize, value: i64) usize {
336 var p = position;
337 var visited: usize = 0;
338 while (p < tree.values.len) : (p += lowbit(p)) {
339 tree.values[p] += value;
340 visited += 1;
341 }
342 return visited;
343 }
344 fn prefix(tree: *const Fenwick, position: usize) Answer {
345 var p = position;
346 var answer: Answer = .{ .value = 0, .visited = 0 };
347 while (p > 0) : (p -= lowbit(p)) {
348 answer.value += tree.values[p];
349 answer.visited += 1;
350 }
351 return answer;
352 }
353 fn dump(tree: *const Fenwick, out: *Writer, label: []const u8) Writer.Error!void {
354 try out.writeAll(label);
355 try writeNumbers(out, tree.values[1..]);
356 }

358const PointFenwick = struct {
359 contents: []i64,
360 bits: Fenwick,
361 fn init(gpa: Allocator, contents: []const i64) Allocator.Error!PointFenwick {
362 var tree: PointFenwick = .{ .contents = try gpa.dupe(i64, contents), .bits = try Fenwick.init(gpa, contents.len) };
363 for (contents, 0..) |value, i| _ = tree.bits.add(i + 1, value);
364 return tree;
365 }
366 fn add(tree: *PointFenwick, index: usize, value: i64) usize {
367 tree.contents[index] += value;
368 return tree.bits.add(index + 1, value);
369 }
370 fn query(tree: *const PointFenwick, left: usize, right: usize) Answer {
371 const a = tree.bits.prefix(right + 1);
372 const b = tree.bits.prefix(left);
373 return .{ .value = a.value - b.value, .visited = a.visited + b.visited };
374 }
375 fn lower(tree: *const PointFenwick, target: i64) struct { index: ?usize, visited: usize } {
376 var pos: usize = 0;
377 var rest = target;
378 var visited: usize = 0;
379 var step = power2(log2(tree.contents.len));
380 while (step >= 1) : (step /= 2) {
381 if (pos + step > tree.contents.len) continue;
382 visited += 1;
383 const value = tree.bits.values[pos + step];
384 if (value < rest) {
385 rest -= value;
386 pos += step;
387 }
388 }
389 return .{ .index = if (pos == tree.contents.len) null else pos, .visited = visited };
390 }

392const RangeFenwick = struct {
393 b1: Fenwick,
394 b2: Fenwick,
395 fn init(gpa: Allocator, contents: []const i64) Allocator.Error!RangeFenwick {
396 var tree: RangeFenwick = .{ .b1 = try Fenwick.init(gpa, contents.len), .b2 = try Fenwick.init(gpa, contents.len) };
397 for (contents, 0..) |value, i| {
398 const difference = value - (if (i == 0) @as(i64, 0) else contents[i - 1]);
399 _ = tree.b1.add(i + 1, difference);
400 _ = tree.b2.add(i + 1, integer(i) * difference);
401 }
402 return tree;
403 }
404 fn add(tree: *RangeFenwick, left: usize, right: usize, value: i64) usize {
405 var visited = tree.b1.add(left + 1, value);
406 visited += tree.b1.add(right + 2, -value);
407 visited += tree.b2.add(left + 1, value * integer(left));
408 visited += tree.b2.add(right + 2, -value * integer(right + 1));
409 return visited;
410 }
411 fn prefix(tree: *const RangeFenwick, position: usize) Answer {
412 const a = tree.b1.prefix(position);
413 const b = tree.b2.prefix(position);
414 return .{ .value = integer(position) * a.value - b.value, .visited = a.visited + b.visited };
415 }
416 fn query(tree: *const RangeFenwick, left: usize, right: usize) Answer {
417 const a = tree.prefix(right + 1);
418 const b = tree.prefix(left);
419 return .{ .value = a.value - b.value, .visited = a.visited + b.visited };
420 }

433const SparseTable = struct {
434 contents: []i64,
435 minimum: [][]usize,
436 maximum: [][]usize,
437 entries: usize,
438 fn init(gpa: Allocator, contents: []const i64) Allocator.Error!SparseTable {
439 const levels = log2(contents.len) + 1;
440 var tree: SparseTable = .{ .contents = try gpa.dupe(i64, contents), .minimum = try gpa.alloc([]usize, levels), .maximum = try gpa.alloc([]usize, levels), .entries = 0 };
441 for (0..levels) |level| {
442 const length = contents.len - power2(level) + 1;
443 tree.minimum[level] = try gpa.alloc(usize, length);
444 tree.maximum[level] = try gpa.alloc(usize, length);
445 tree.entries += length;
446 for (0..length) |i| {
447 if (level == 0) {
448 tree.minimum[level][i] = i;
449 tree.maximum[level][i] = i;
450 } else {
451 const half = power2(level - 1);
452 tree.minimum[level][i] = tree.select(tree.minimum[level - 1][i], tree.minimum[level - 1][i + half], true);
453 tree.maximum[level][i] = tree.select(tree.maximum[level - 1][i], tree.maximum[level - 1][i + half], false);
454 }
455 }
456 }
457 return tree;
458 }
459 fn select(tree: *const SparseTable, a: usize, b: usize, minimum: bool) usize {
460 return if (if (minimum) tree.contents[a] <= tree.contents[b] else tree.contents[a] >= tree.contents[b]) a else b;
461 }
462 fn query(tree: *const SparseTable, left: usize, right: usize) SparseAnswer {
463 const level = log2(right - left + 1);
464 const end = right + 1 - power2(level);
465 const a = tree.select(tree.minimum[level][left], tree.minimum[level][end], true);
466 const z = tree.select(tree.maximum[level][left], tree.maximum[level][end], false);
467 return .{ .min = tree.contents[a], .minpos = a, .max = tree.contents[z], .maxpos = z, .level = level };
468 }
469 fn dump(tree: *const SparseTable, out: *Writer) Writer.Error!void {
470 try line(out, "SPARSE DUMP levels={d}", .{tree.minimum.len});
471 for (tree.minimum, tree.maximum, 0..) |minima, maxima, level| {
472 try out.print("MIN {d}", .{level});
473 try writeNumbers(out, minima);
474 try out.print("MAX {d}", .{level});
475 try writeNumbers(out, maxima);
476 }
477 }

501const SqrtBlocks = struct {
502 block: usize,
503 raw: []i64,
504 tags: []i64,
505 sums: []i64,
506 fn init(gpa: Allocator, contents: []const i64, block: usize) Allocator.Error!SqrtBlocks {
507 const count = (contents.len + block - 1) / block;
508 const tree: SqrtBlocks = .{ .block = block, .raw = try gpa.dupe(i64, contents), .tags = try gpa.alloc(i64, count), .sums = try gpa.alloc(i64, count) };
509 @memset(tree.tags, 0);
510 @memset(tree.sums, 0);
511 for (contents, 0..) |value, i| tree.sums[i / block] += value;
512 return tree;
513 }
514 fn add(tree: *SqrtBlocks, left: usize, right: usize, value: i64) Split {
515 const split = Split.init(left, right, tree.block);
516 for (left..right + 1) |i| {
517 const block = i / tree.block;
518 if (block == split.first or block == split.last) {
519 tree.raw[i] += value;
520 tree.sums[block] += value;
521 }
522 }
523 if (split.whole > 0) for (split.first + 1..split.last) |block| {
524 tree.tags[block] += value;
525 tree.sums[block] += value * integer(tree.block);
526 };
527 return split;
528 }
529 fn query(tree: *const SqrtBlocks, left: usize, right: usize) SqrtAnswer {
530 const split = Split.init(left, right, tree.block);
531 var sum: i64 = 0;
532 for (left..right + 1) |i| {
533 const block = i / tree.block;
534 if (block == split.first or block == split.last) sum += tree.raw[i] + tree.tags[block];
535 }
536 if (split.whole > 0) for (split.first + 1..split.last) |block| {
537 sum += tree.sums[block];
538 };
539 return .{ .sum = sum, .split = split };
540 }
541 fn dump(tree: *const SqrtBlocks, out: *Writer) Writer.Error!void {
542 try line(out, "SQRT DUMP block={d} blocks={d}", .{ tree.block, tree.tags.len });
543 for (0..tree.tags.len) |block| {
544 const first = block * tree.block;
545 const end = @min((block + 1) * tree.block, tree.raw.len);
546 try out.print("BLOCK {d} {d} {d} add={d} sum={d}", .{ block, first, end - 1, tree.tags[block], tree.sums[block] });
547 try writeNumbers(out, tree.raw[first..end]);
548 }
549 }

562 fn init(gpa: Allocator, contents: []const i64) Allocator.Error!PersistentTree {
563 const values = try gpa.dupe(i64, contents);
564 std.sort.heap(i64, values, {}, less);
565 var distinct: usize = 0;
566 for (values) |value| {
567 if (distinct == 0 or value != values[distinct - 1]) {
568 values[distinct] = value;
569 distinct += 1;
570 }
571 }
572 var tree: PersistentTree = .{ .gpa = gpa, .n = contents.len, .values = values[0..distinct], .value_roots = try gpa.alloc(*const PersistentNode, contents.len + 1), .position_nodes = 2 * contents.len - 1, .value_nodes = 2 * distinct - 1 };
573 try tree.versions.append(gpa, try tree.build(contents, 0, contents.len - 1));
574 const zero = try gpa.alloc(i64, distinct);
575 @memset(zero, 0);
576 tree.value_roots[0] = try tree.build(zero, 0, distinct - 1);
577 for (contents, 0..) |value, i| {
578 var left: usize = 0;
579 var right = distinct;
580 while (left < right) {
581 const middle = (left + right) / 2;
582 if (tree.values[middle] < value) left = middle + 1 else right = middle;
583 }
584 const created = try tree.change(tree.value_roots[i], 0, distinct - 1, left, 1, true);
585 tree.value_roots[i + 1] = created.node;
586 tree.value_nodes += created.count;
587 }
588 return tree;
589 }
590 fn node(tree: *const PersistentTree, value: i64, left: ?*const PersistentNode, right: ?*const PersistentNode) Allocator.Error!*const PersistentNode {
591 const result = try tree.gpa.create(PersistentNode);
592 result.* = .{ .value = value, .left = left, .right = right };
593 return result;
594 }
595 fn build(tree: *const PersistentTree, contents: []const i64, left: usize, right: usize) Allocator.Error!*const PersistentNode {
596 if (left == right) return tree.node(contents[left], null, null);
597 const middle = (left + right) / 2;
598 const a = try tree.build(contents, left, middle);
599 const b = try tree.build(contents, middle + 1, right);
600 return tree.node(a.value + b.value, a, b);
601 }
602 fn change(tree: *const PersistentTree, old: *const PersistentNode, left: usize, right: usize, index: usize, value: i64, increment: bool) Allocator.Error!Created {
603 if (left == right) return .{ .node = try tree.node(if (increment) old.value + value else value, null, null), .count = 1 };
604 const middle = (left + right) / 2;
605 var a = old.left.?;
606 var b = old.right.?;
607 const changed = if (index <= middle) try tree.change(a, left, middle, index, value, increment) else try tree.change(b, middle + 1, right, index, value, increment);
608 if (index <= middle) a = changed.node else b = changed.node;
609 return .{ .node = try tree.node(a.value + b.value, a, b), .count = changed.count + 1 };
610 }
611 fn set(tree: *PersistentTree, version: usize, index: usize, value: i64) Allocator.Error!usize {
612 const changed = try tree.change(tree.versions.items[version], 0, tree.n - 1, index, value, false);
613 try tree.versions.append(tree.gpa, changed.node);
614 tree.position_nodes += changed.count;
615 return changed.count;
616 }
617 fn sum(current: *const PersistentNode, nl: usize, nr: usize, left: usize, right: usize) Answer {
618 if (left <= nl and nr <= right) return .{ .value = current.value, .visited = 1 };
619 const middle = (nl + nr) / 2;
620 if (right <= middle or left > middle) {
621 var child = if (right <= middle) sum(current.left.?, nl, middle, left, right) else sum(current.right.?, middle + 1, nr, left, right);
622 child.visited += 1;
623 return child;
624 }
625 const a = sum(current.left.?, nl, middle, left, right);
626 const b = sum(current.right.?, middle + 1, nr, left, right);
627 return .{ .value = a.value + b.value, .visited = a.visited + b.visited + 1 };
628 }
629 fn query(tree: *const PersistentTree, version: usize, left: usize, right: usize) Answer {
630 return sum(tree.versions.items[version], 0, tree.n - 1, left, right);
631 }
632 fn kth(tree: *const PersistentTree, left: usize, right: usize, rank: usize) Answer {
633 var u = tree.value_roots[right + 1];
634 var w = tree.value_roots[left];
635 var p: usize = 0;
636 var q = tree.values.len - 1;
637 var k = rank;
638 var visited: usize = 1;
639 while (p < q) {
640 const middle = (p + q) / 2;
641 const difference: usize = @intCast(u.left.?.value - w.left.?.value);
642 if (k <= difference) {
643 u = u.left.?;
644 w = w.left.?;
645 q = middle;
646 } else {
647 k -= difference;
648 u = u.right.?;
649 w = w.right.?;
650 p = middle + 1;
651 }
652 visited += 1;
653 }
654 return .{ .value = tree.values[p], .visited = visited };
655 }

658const GridFenwick = struct {
659 height: usize,
660 width: usize,
661 values: []i64,
662 fn init(gpa: Allocator, height: usize, width: usize, contents: []const i64) Allocator.Error!GridFenwick {
663 var tree: GridFenwick = .{ .height = height, .width = width, .values = try gpa.alloc(i64, (height + 1) * (width + 1)) };
664 @memset(tree.values, 0);
665 for (0..height) |i| for (0..width) |j| {
666 _ = tree.add(i, j, contents[i * width + j]);
667 };
668 return tree;
669 }
670 fn add(tree: *GridFenwick, row: usize, column: usize, value: i64) usize {
671 var visited: usize = 0;
672 var y = row + 1;
673 while (y <= tree.height) : (y += lowbit(y)) {
674 var z = column + 1;
675 while (z <= tree.width) : (z += lowbit(z)) {
676 tree.values[y * (tree.width + 1) + z] += value;
677 visited += 1;
678 }
679 }
680 return visited;
681 }
682 fn prefix(tree: *const GridFenwick, row: usize, column: usize) Answer {
683 var answer: Answer = .{ .value = 0, .visited = 0 };
684 var y = row;
685 while (y > 0) : (y -= lowbit(y)) {
686 var z = column;
687 while (z > 0) : (z -= lowbit(z)) {
688 answer.value += tree.values[y * (tree.width + 1) + z];
689 answer.visited += 1;
690 }
691 }
692 return answer;
693 }
694 fn query(tree: *const GridFenwick, top: usize, left: usize, bottom: usize, right: usize) Answer {
695 const a = tree.prefix(bottom + 1, right + 1);
696 const b = tree.prefix(top, right + 1);
697 const c = tree.prefix(bottom + 1, left);
698 const d = tree.prefix(top, left);
699 return .{ .value = a.value - b.value - c.value + d.value, .visited = a.visited + b.visited + c.visited + d.visited };
700 }
701 fn dump(tree: *const GridFenwick, out: *Writer) Writer.Error!void {
702 try line(out, "GRID DUMP rows={d} cols={d}", .{ tree.height, tree.width });
703 for (1..tree.height + 1) |row| {
704 try out.print("ROW {d}", .{row});
705 const first = row * (tree.width + 1) + 1;
706 try writeNumbers(out, tree.values[first .. first + tree.width]);
707 }
708 }

771const MoQuery = struct {
772 left: usize,
773 right: usize,
774 number: usize,
775 fn before(block: usize, a: MoQuery, b: MoQuery) bool {
776 const x = a.left / block;
777 const y = b.left / block;
778 if (x != y) return x < y;
779 if (a.right != b.right) return if (x % 2 == 0) a.right < b.right else a.right > b.right;
780 return a.number < b.number;
781 }

783const MoWindow = struct {
784 ranks: []usize,
785 frequency: []usize,
786 distinct: usize = 0,
787 moves: usize = 0,
788 fn add(window: *MoWindow, index: usize) void {
789 const rank = window.ranks[index];
790 if (window.frequency[rank] == 0) window.distinct += 1;
791 window.frequency[rank] += 1;
792 window.moves += 1;
793 }
794 fn remove(window: *MoWindow, index: usize) void {
795 const rank = window.ranks[index];
796 window.frequency[rank] -= 1;
797 if (window.frequency[rank] == 0) window.distinct -= 1;
798 window.moves += 1;
799 }

845 fn report(session: *Session, kind: Kind, matched: bool, result: anytype, ok: *usize, diff: *usize) Writer.Error!void {
846 if (matched) ok.* += 1 else diff.* += 1;
847 try line(session.out, "{s} {s} {f}", .{ @tagName(kind), if (matched) @as([]const u8, "OK") else "DIFF", result });

849 fn check(session: *Session, range: Interval) Writer.Error!void {
850 const state = session.state;
851 var direct: Aggregate = .{ .sum = 0, .min = state.values[range.left], .max = state.values[range.left] };
852 var divisor: i64 = 0;
853 for (state.values[range.left .. range.right + 1]) |value| {
854 direct.sum += value;
855 direct.min = @min(direct.min, value);
856 direct.max = @max(direct.max, value);
857 divisor = gcd(divisor, value);
858 }
859 try line(session.out, "CHECK {d} {d} {f} gcd={d}", .{ range.left, range.right, direct, divisor });
860 var ok: usize = 0;
861 var diff: usize = 0;
862 if (state.segment) |box| {
863 const answer = box.value.query(range.left, range.right, 1);
864 const expected = switch (box.value.op) {
865 .SUM => direct.sum,
866 .MIN => direct.min,
867 .MAX => direct.max,
868 .GCD => divisor,
869 };
870 try session.report(.SEG, answer.value == expected, box.value.describe(answer), &ok, &diff);
871 }
872 if (state.lazy) |box| {
873 const answer = box.value.query(range.left, range.right, 1);
874 try session.report(.LAZY, answer.value.eql(direct), answer, &ok, &diff);
875 }
876 if (state.fenwick) |box| {
877 const answer = box.value.query(range.left, range.right);
878 try session.report(.FEN, answer.value == direct.sum, answer, &ok, &diff);
879 }
880 if (state.range_fenwick) |box| {
881 const answer = box.value.query(range.left, range.right);
882 try session.report(.RFEN, answer.value == direct.sum, answer, &ok, &diff);
883 }
884 if (state.sparse) |box| {
885 const answer = box.value.query(range.left, range.right);
886 try session.report(.SPARSE, answer.min == direct.min and answer.max == direct.max, answer, &ok, &diff);
887 }
888 if (state.sqrt) |box| {
889 const answer = box.value.query(range.left, range.right);
890 try session.report(.SQRT, answer.sum == direct.sum, answer, &ok, &diff);
891 }
892 if (state.persistent) |box| {
893 const version = box.value.versions.items.len - 1;
894 const answer = box.value.query(version, range.left, range.right);
895 try session.report(.PERSIST, answer.value == direct.sum, VersionAnswer{ .version = version, .answer = answer }, &ok, &diff);
896 }
897 try line(session.out, "RESULT ok={d} diff={d}", .{ ok, diff });
898 }
899 fn mo(session: *Session, words: []const []const u8) Error!void {
900 const state = session.state;
901 const queries = try session.gpa.alloc(MoQuery, (words.len - 1) / 2);
902 for (queries, 0..) |*query, i| {
903 const range = try session.checkRange(arg(words, 2 * i + 1), arg(words, 2 * i + 2));
904 query.* = .{ .left = range.left, .right = range.right, .number = i + 1 };
905 }
906 const block = isqrt(state.n);
907 const order = try session.gpa.dupe(MoQuery, queries);
908 std.sort.heap(MoQuery, order, block, MoQuery.before);
909 const sorted = try session.gpa.dupe(i64, state.array());
910 std.sort.heap(i64, sorted, {}, less);
911 const ranks = try session.gpa.alloc(usize, state.n);
912 for (state.array(), ranks) |value, *rank| {
913 var left: usize = 0;
914 var right = sorted.len;
915 while (left < right) {
916 const middle = (left + right) / 2;
917 if (sorted[middle] < value) left = middle + 1 else right = middle;
918 }
919 rank.* = left;
920 }
921 var window: MoWindow = .{ .ranks = ranks, .frequency = try session.gpa.alloc(usize, state.n) };
922 @memset(window.frequency, 0);
923 const answers = try session.gpa.alloc(usize, queries.len);
924 var left: usize = 0;
925 var right: i64 = -1;
926 for (order) |query| {
927 while (left > query.left) {
928 left -= 1;
929 window.add(left);
930 }
931 while (right < integer(query.right)) {
932 right += 1;
933 window.add(@intCast(right));
934 }
935 while (left < query.left) {
936 window.remove(left);
937 left += 1;
938 }
939 while (right > integer(query.right)) {
940 window.remove(@intCast(right));
941 right -= 1;
942 }
943 answers[query.number - 1] = window.distinct;
944 }
945 try line(session.out, "MO queries={d} block={d} moves={d}", .{ queries.len, block, window.moves });
946 try session.out.writeAll("ORDER");
947 for (order) |query| try session.out.print(" {d}", .{query.number});
948 try session.out.writeByte('\n');
949 for (queries) |query| try line(session.out, "ANSWER {d} {d} {d} distinct={d}", .{ query.number, query.left, query.right, answers[query.number - 1] });
950 }
951 fn execute(session: *Session, text: []const u8) Error!void {
952 const words = try tokenize(session.gpa, text);
953 const command = words[0];
954 const state = session.state;
955 const out = session.out;
956 if (equal(command, "ARRAY")) {
957 const range = try session.checkRange(arg(words, 1), arg(words, 2));
958 try out.print("ARRAY {d} {d}", .{ range.left, range.right });
959 return writeNumbers(out, state.values[range.left .. range.right + 1]);
960 }
961 if (equal(command, "LOAD")) {
962 const removed = state.count();
963 state.load(try parseValues(session.gpa, words[1..]));
964 return line(out, "LOAD n={d} removed={d}", .{ state.n, removed });
965 }
966 if (equal(command, "RESET")) {
967 const rebuilt = state.count();
968 try state.reset();
969 return line(out, "RESET rebuilt={d}", .{rebuilt});
970 }
971 if (equal(command, "CHECK")) return session.check(try session.checkRange(arg(words, 1), arg(words, 2)));
972 if (equal(command, "MO")) return session.mo(words);
973 const kind = std.meta.stringToEnum(Kind, command).?;
974 const action = words[1];
975 switch (kind) {
976 .SEG => {
977 if (equal(action, "BUILD")) {
978 const op = std.meta.stringToEnum(Operation, words[2]) orelse return session.fail("BAD OP {s}", .{words[2]});
979 try replace(SegmentTree, &state.segment, .{ state.array(), op });
980 const tree = &state.segment.?.value;
981 return line(out, "SEG BUILD {s} nodes={d} maxnode={d} root={d}", .{ @tagName(op), tree.shape.count, tree.shape.max_node, tree.values[1] });
982 }
983 const tree = try session.need(SegmentTree, state.segment, kind);
984 if (equal(action, "DUMP")) return tree.dump(out);
985 if (equal(action, "UPDATE")) {
986 const index = try session.checkIndex(arg(words, 2));
987 const value = arg(words, 3);
988 const visited = tree.update(index, value, 1);
989 state.values[index] = value;
990 return line(out, "SEG UPDATE {d} {d} root={d} visited={d}", .{ index, value, tree.values[1], visited });
991 }
992 const range = try session.checkRange(arg(words, 2), arg(words, 3));
993 return line(out, "SEG QUERY {d} {d} {f}", .{ range.left, range.right, tree.describe(tree.query(range.left, range.right, 1)) });
994 },
995 .LAZY => {
996 if (equal(action, "BUILD")) {
997 try replace(LazyTree, &state.lazy, .{state.array()});
998 const tree = &state.lazy.?.value;
999 return line(out, "LAZY BUILD nodes={d} {f}", .{ tree.shape.count, tree.values[1] });
1000 }
1001 const tree = try session.need(LazyTree, state.lazy, kind);
1002 if (equal(action, "DUMP")) return tree.dump(out);
1003 const range = try session.checkRange(arg(words, 2), arg(words, 3));
1004 if (equal(action, "QUERY")) return line(out, "LAZY QUERY {d} {d} {f}", .{ range.left, range.right, tree.query(range.left, range.right, 1) });
1005 const value = arg(words, 4);
1006 const assignment = equal(action, "ASSIGN");
1007 var work: Work = .{};
1008 tree.update(range.left, range.right, value, assignment, 1, &work);
1009 state.change(range, value, assignment);
1010 return line(out, "LAZY {s} {d} {d} {d} {f}", .{ action, range.left, range.right, value, work });
1011 },
1012 .FEN => {
1013 if (equal(action, "BUILD")) {
1014 try replace(PointFenwick, &state.fenwick, .{state.array()});
1015 const tree = &state.fenwick.?.value;
1016 return line(out, "FEN BUILD n={d} total={d}", .{ state.n, tree.bits.prefix(state.n).value });
1017 }
1018 const tree = try session.need(PointFenwick, state.fenwick, kind);
1019 if (equal(action, "DUMP")) {
1020 try line(out, "FEN DUMP n={d}", .{state.n});
1021 return tree.bits.dump(out, "T");
1022 }
1023 if (equal(action, "LOWER")) {
1024 for (tree.contents, 0..) |value, i| if (value < 0) {
1025 return session.fail("NEGATIVE {d}", .{i});
1026 };
1027 const target = arg(words, 2);
1028 const answer = tree.lower(target);
1029 try out.print("FEN LOWER {d} index=", .{target});
1030 if (answer.index) |index| try out.print("{d}", .{index}) else try out.writeAll("none");
1031 return line(out, " visited={d}", .{answer.visited});
1032 }
1033 if (equal(action, "RANGE")) {
1034 const range = try session.checkRange(arg(words, 2), arg(words, 3));
1035 return line(out, "FEN RANGE {d} {d} {f}", .{ range.left, range.right, tree.query(range.left, range.right) });
1036 }
1037 const index = try session.checkIndex(arg(words, 2));
1038 if (equal(action, "PREFIX")) return line(out, "FEN PREFIX {d} {f}", .{ index, tree.bits.prefix(index + 1) });
1039 const value = arg(words, 3);
1040 const visited = tree.add(index, value);
1041 state.values[index] += value;
1042 return line(out, "FEN ADD {d} {d} visited={d}", .{ index, value, visited });
1043 },
1044 .RFEN => {
1045 if (equal(action, "BUILD")) {
1046 try replace(RangeFenwick, &state.range_fenwick, .{state.array()});
1047 const tree = &state.range_fenwick.?.value;
1048 return line(out, "RFEN BUILD n={d} total={d}", .{ state.n, tree.query(0, state.n - 1).value });
1049 }
1050 const tree = try session.need(RangeFenwick, state.range_fenwick, kind);
1051 if (equal(action, "DUMP")) {
1052 try line(out, "RFEN DUMP n={d}", .{state.n});
1053 try tree.b1.dump(out, "B1");
1054 return tree.b2.dump(out, "B2");
1055 }
1056 const range = try session.checkRange(arg(words, 2), arg(words, 3));
1057 if (equal(action, "SUM")) return line(out, "RFEN SUM {d} {d} {f}", .{ range.left, range.right, tree.query(range.left, range.right) });
1058 const value = arg(words, 4);
1059 const visited = tree.add(range.left, range.right, value);
1060 state.change(range, value, false);
1061 return line(out, "RFEN ADD {d} {d} {d} visited={d}", .{ range.left, range.right, value, visited });
1062 },
1063 .SPARSE => {
1064 if (equal(action, "BUILD")) {
1065 try replace(SparseTable, &state.sparse, .{state.array()});
1066 const tree = &state.sparse.?.value;
1067 return line(out, "SPARSE BUILD levels={d} entries={d}", .{ tree.minimum.len, tree.entries });
1068 }
1069 const tree = try session.need(SparseTable, state.sparse, kind);
1070 if (equal(action, "DUMP")) return tree.dump(out);
1071 const range = try session.checkRange(arg(words, 2), arg(words, 3));
1072 return line(out, "SPARSE QUERY {d} {d} {f}", .{ range.left, range.right, tree.query(range.left, range.right) });
1073 },
1074 .SQRT => {
1075 if (equal(action, "BUILD")) {
1076 const block: usize = if (words.len == 3) @intCast(arg(words, 2)) else isqrt(state.n);
1077 if (block > state.n) return session.fail("BAD BLOCK {d}", .{block});
1078 try replace(SqrtBlocks, &state.sqrt, .{ state.array(), block });
1079 return line(out, "SQRT BUILD block={d} blocks={d}", .{ block, state.sqrt.?.value.tags.len });
1080 }
1081 const tree = try session.need(SqrtBlocks, state.sqrt, kind);
1082 if (equal(action, "DUMP")) return tree.dump(out);
1083 const range = try session.checkRange(arg(words, 2), arg(words, 3));
1084 if (equal(action, "SUM")) return line(out, "SQRT SUM {d} {d} {f}", .{ range.left, range.right, tree.query(range.left, range.right) });
1085 const value = arg(words, 4);
1086 const split = tree.add(range.left, range.right, value);
1087 state.change(range, value, false);
1088 return line(out, "SQRT ADD {d} {d} {d} {f}", .{ range.left, range.right, value, split });
1089 },
1090 .PERSIST => {
1091 if (equal(action, "BUILD")) {
1092 try replace(PersistentTree, &state.persistent, .{state.array()});
1093 const tree = &state.persistent.?.value;
1094 return line(out, "PERSIST BUILD nodes={d} distinct={d} valuenodes={d}", .{ tree.position_nodes, tree.values.len, tree.value_nodes });
1095 }
1096 const tree = try session.need(PersistentTree, state.persistent, kind);
1097 if (equal(action, "NODES")) return line(out, "PERSIST NODES versions={d} position={d} value={d}", .{ tree.versions.items.len, tree.position_nodes, tree.value_nodes });
1098 if (equal(action, "KTH")) {
1099 const range = try session.checkRange(arg(words, 2), arg(words, 3));
1100 const k = arg(words, 4);
1101 if (k < 1 or k > integer(range.length())) return session.fail("BAD K {d}", .{k});
1102 const answer = tree.kth(range.left, range.right, @intCast(k));
1103 return line(out, "PERSIST KTH {d} {d} {d} value={d} visited={d}", .{ range.left, range.right, k, answer.value, answer.visited });
1104 }
1105 const version = try session.checkVersion(tree, arg(words, 2));
1106 if (equal(action, "SET")) {
1107 const index = try session.checkIndex(arg(words, 3));
1108 const value = arg(words, 4);
1109 const created = try tree.set(version, index, value);
1110 return line(out, "PERSIST SET {d} {d} {d} version={d} created={d}", .{ version, index, value, tree.versions.items.len - 1, created });
1111 }
1112 const range = try session.checkRange(arg(words, 3), arg(words, 4));
1113 return line(out, "PERSIST QUERY {d} {d} {d} {f}", .{ version, range.left, range.right, tree.query(version, range.left, range.right) });
1114 },
1115 .GRID => {
1116 if (equal(action, "BUILD")) {
1117 const height: usize = @intCast(arg(words, 2));
1118 const width: usize = @intCast(arg(words, 3));
1119 const count = words.len - 4;
1120 if (count != height * width) return session.fail("BAD COUNT {d} {d}", .{ count, height * width });
1121 const values = try parseValues(session.gpa, words[4..]);
1122 var total: i64 = 0;
1123 for (values) |value| total += value;
1124 try replace(GridFenwick, &state.grid, .{ height, width, values });
1125 return line(out, "GRID BUILD rows={d} cols={d} total={d}", .{ height, width, total });
1126 }
1127 const tree = try session.need(GridFenwick, state.grid, kind);
1128 if (equal(action, "DUMP")) return tree.dump(out);
1129 const row = arg(words, 2);
1130 const column = arg(words, 3);
1131 try session.checkCell(tree, row, column);
1132 if (equal(action, "ADD")) {
1133 const value = arg(words, 4);
1134 const visited = tree.add(@intCast(row), @intCast(column), value);
1135 return line(out, "GRID ADD {d} {d} {d} visited={d}", .{ row, column, value, visited });
1136 }
1137 const bottom = arg(words, 4);
1138 const right = arg(words, 5);
1139 try session.checkCell(tree, bottom, right);
1140 if (row > bottom or column > right) return session.fail("BAD RECT {d} {d} {d} {d}", .{ row, column, bottom, right });
1141 return line(out, "GRID SUM {d} {d} {d} {d} {f}", .{ row, column, bottom, right, tree.query(@intCast(row), @intCast(column), @intCast(bottom), @intCast(right)) });
1142 },
1143 }
1144 }

1147fn readLine(gpa: Allocator, input: *std.Io.Reader) ![]const u8 {
1148 var buffer: Writer.Allocating = .init(gpa);
1149 defer buffer.deinit();
1150 _ = try input.streamDelimiterEnding(&buffer.writer, '\n');
1151 if (input.seek < input.end) input.toss(1) else if (buffer.written().len == 0) return error.EndOfStream;
1152 const text = try buffer.toOwnedSlice();
1153 return if (text.len > 0 and text[text.len - 1] == '\r') text[0 .. text.len - 1] else text;
1154}
1155pub fn main(init: std.process.Init) !void {
1156 var header = std.heap.ArenaAllocator.init(std.heap.page_allocator);
1157 defer header.deinit();
1158 var state: State = .{};
1159 defer state.deinit();
1160 var input_buffer: [1024]u8 = undefined;
1161 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
1162 var output_buffer: [4096]u8 = undefined;
1163 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
1164 _ = try readLine(header.allocator(), &input.interface);
1165 const initial = try tokenize(header.allocator(), try readLine(header.allocator(), &input.interface));
1166 state.load(try parseValues(header.allocator(), initial));
1167 const count: usize = @intCast(number(try readLine(header.allocator(), &input.interface)));
1168 for (0..count) |_| {
1169 var scratch = std.heap.ArenaAllocator.init(std.heap.page_allocator);
1170 defer scratch.deinit();
1171 const gpa = scratch.allocator();
1172 const text = try readLine(gpa, &input.interface);
1173 var response: Writer.Allocating = .init(gpa);
1174 defer response.deinit();
1175 var session: Session = .{ .state = &state, .gpa = gpa, .out = &response.writer };
1176 session.execute(text) catch |err| {
1177 if (err != error.Invalid) return err;
1178 try line(&output.interface, "ERROR {s}", .{session.problem});
1179 try output.interface.flush();
1180 continue;
1181 };
1182 try output.interface.writeAll(response.written());
1183 try output.interface.flush();
1184 }
