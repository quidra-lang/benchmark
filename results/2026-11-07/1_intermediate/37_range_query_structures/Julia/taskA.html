47struct Layout
48 n::Int
49 lower::Vector{Int}
50 upper::Vector{Int}

53function Layout(n::Int)
54 lower, upper = fill(-1, 4n), zeros(Int, 4n)
55 function visit(node::Int, left::Int, right::Int)
56 lower[node], upper[node] = left, right
57 if left < right
58 middle = div(left + right, 2)
59 visit(2node, left, middle)
60 visit(2node + 1, middle + 1, right)
61 end
62 end
63 visit(1, 0, n - 1)
64 Layout(n, lower, upper, findall(x -> x >= 0, lower))
65end
66covered(shape::Layout, node::Int, left::Int, right::Int) = left <= shape.lower[node] && shape.upper[node] <= right
67middle(shape::Layout, node::Int) = div(shape.lower[node] + shape.upper[node], 2)

74function Segment(contents::Vector{Int}, op::Operation)
75 tree = Segment(Layout(length(contents)), op, zeros(Int, 4length(contents)))
76 function build(node::Int)
77 if tree.shape.lower[node] == tree.shape.upper[node]
78 tree.values[node] = leaf_value(op, contents[tree.shape.lower[node] + 1])
79 else
80 build(2node)
81 build(2node + 1)
82 segment_pull!(tree, node)
83 end
84 end
85 build(1)
86 tree
87end
88segment_pull!(tree::Segment, node::Int) = tree.values[node] = combine(tree.op, tree.values[2node], tree.values[2node + 1])
89function segment_query(tree::Segment, node::Int, left::Int, right::Int)
90 covered(tree.shape, node, left, right) && return Answer(tree.values[node], 1)
91 mid = middle(tree.shape, node)
92 if right <= mid
93 result = segment_query(tree, 2node, left, right)
94 elseif left > mid
95 result = segment_query(tree, 2node + 1, left, right)
96 else
97 a = segment_query(tree, 2node, left, right)
98 b = segment_query(tree, 2node + 1, left, right)
99 result = Answer(combine(tree.op, a.value, b.value), a.visited + b.visited)
100 end
101 Answer(result.value, result.visited + 1)
102end
103function query(tree::Segment, left::Int, right::Int)
104 result = segment_query(tree, 1, left, right)
105 Report("$(lowercase(string(tree.op)))=$(result.value) visited=$(result.visited)", [result.value])
106end
107function segment_update!(tree::Segment, node::Int, index::Int, value::Int)
108 if tree.shape.lower[node] == tree.shape.upper[node]
109 tree.values[node] = leaf_value(tree.op, value)
110 return 1
111 end
112 child = index <= middle(tree.shape, node) ? 2node : 2node + 1
113 visited = segment_update!(tree, child, index, value)
114 segment_pull!(tree, node)
115 visited + 1
116end

131function LazyTree(contents::Vector{Int})
132 n = length(contents)
133 tree = LazyTree(Layout(n), fill(Aggregate(0, 0, 0), 4n), Union{Nothing,Int}[nothing for _ in 1:4n], zeros(Int, 4n))
134 function build(node::Int)
135 if tree.shape.lower[node] == tree.shape.upper[node]
136 value = contents[tree.shape.lower[node] + 1]
137 tree.values[node] = Aggregate(value, value, value)
138 else
139 build(2node)
140 build(2node + 1)
141 lazy_pull!(tree, node)
142 end
143 end
144 build(1)
145 tree
146end
147lazy_pull!(tree::LazyTree, node::Int) = tree.values[node] = combine(tree.values[2node], tree.values[2node + 1])
148function lazy_apply!(tree::LazyTree, node::Int, value::Int, assign::Bool)
149 size = tree.shape.upper[node] - tree.shape.lower[node] + 1
150 previous = tree.values[node]
151 tree.values[node] = assign ? Aggregate(value * size, value, value) :
152 Aggregate(previous.total + value * size, previous.low + value, previous.high + value)
153 if size > 1
154 if assign
155 tree.assignments[node] = value
156 tree.additions[node] = 0
157 else
158 tree.additions[node] += value
159 end
160 end
161end
162function lazy_push!(tree::LazyTree, node::Int, counts::Counts)
163 assignment, addition = tree.assignments[node], tree.additions[node]
164 isnothing(assignment) && addition == 0 && return
165 for child in 2node:(2node + 1)
166 isnothing(assignment) || lazy_apply!(tree, child, assignment, true)
167 addition == 0 || lazy_apply!(tree, child, addition, false)
168 end
169 tree.assignments[node], tree.additions[node] = nothing, 0
170 counts.pushes += 1
171end
172function lazy_update!(tree::LazyTree, node::Int, left::Int, right::Int, value::Int, assign::Bool, counts::Counts)
173 counts.visited += 1
174 if covered(tree.shape, node, left, right)
175 counts.covered += 1
176 lazy_apply!(tree, node, value, assign)
177 return
178 end
179 lazy_push!(tree, node, counts)
180 mid = middle(tree.shape, node)
181 left <= mid && lazy_update!(tree, 2node, left, right, value, assign, counts)
182 right > mid && lazy_update!(tree, 2node + 1, left, right, value, assign, counts)
183 lazy_pull!(tree, node)
184end
185function lazy_query!(tree::LazyTree, node::Int, left::Int, right::Int, counts::Counts)
186 counts.visited += 1
187 covered(tree.shape, node, left, right) && return tree.values[node]
188 lazy_push!(tree, node, counts)
189 mid = middle(tree.shape, node)
190 right <= mid && return lazy_query!(tree, 2node, left, right, counts)
191 left > mid && return lazy_query!(tree, 2node + 1, left, right, counts)
192 a = lazy_query!(tree, 2node, left, right, counts)
193 b = lazy_query!(tree, 2node + 1, left, right, counts)
194 combine(a, b)
195end
196function query(tree::LazyTree, left::Int, right::Int)
197 counts = Counts()
198 value = lazy_query!(tree, 1, left, right, counts)
199 Report("$(describe(value)) visited=$(counts.visited) pushes=$(counts.pushes)", [value.total, value.low, value.high])
200end

214function bit_add!(tree::Vector{Int}, position::Int, value::Int)
215 visited = 0
216 while position <= length(tree)
217 tree[position] += value
218 visited += 1
219 position += lowbit(position)

223function build_bit(values::Vector{Int})
224 tree = zeros(Int, length(values))
225 for index in eachindex(values)
226 bit_add!(tree, index, values[index])
227 end

229end
230Fenwick(values::Vector{Int}) = Fenwick(copy(values), build_bit(values))
231function bit_prefix(tree::Vector{Int}, position::Int)
232 value, visited = 0, 0
233 while position > 0
234 value += tree[position]
235 visited += 1
236 position -= lowbit(position)
237 end
238 Answer(value, visited)
239end
240function query(tree::Fenwick, left::Int, right::Int)
241 a = bit_prefix(tree.tree, right + 1)
242 b = bit_prefix(tree.tree, left)
243 report(Answer(a.value - b.value, a.visited + b.visited))
244end
245function bit_lower(tree::Vector{Int}, rest::Int)
246 n = length(tree)
247 step, position, visited = 1, 0, 0
248 while 2step <= n
249 step *= 2
250 end
251 while step > 0
252 if position + step <= n
253 value = tree[position + step]
254 visited += 1
255 if value < rest
256 rest -= value
257 position += step
258 end
259 end
260 step = div(step, 2)
261 end
262 Answer(position < n ? position : -1, visited)
263end

265struct RangeFenwick

268end
269function RangeFenwick(values::Vector{Int})
270 differences = values - vcat(0, values[1:end-1])
271 RangeFenwick(build_bit(differences), build_bit(collect(0:(length(values) - 1)) .* differences))
272end
273function range_add!(tree::RangeFenwick, left::Int, right::Int, value::Int)
274 visited = bit_add!(tree.first, left + 1, value)
275 visited += bit_add!(tree.first, right + 2, -value)
276 visited += bit_add!(tree.second, left + 1, value * left)
277 visited += bit_add!(tree.second, right + 2, -value * (right + 1))
278 visited
279end
280function range_prefix(tree::RangeFenwick, position::Int)
281 a = bit_prefix(tree.first, position)
282 b = bit_prefix(tree.second, position)
283 Answer(position * a.value - b.value, a.visited + b.visited)
284end
285function query(tree::RangeFenwick, left::Int, right::Int)
286 a = range_prefix(tree, right + 1)
287 b = range_prefix(tree, left)
288 report(Answer(a.value - b.value, a.visited + b.visited))

292struct SparseTable
293 values::Vector{Int}
294 minima::Vector{Vector{Int}}
295 maxima::Vector{Vector{Int}}
296end
297function SparseTable(values::Vector{Int})
298 tree = SparseTable(copy(values), [collect(0:(length(values) - 1))], [collect(0:(length(values) - 1))])
299 width = 2
300 while width <= length(values)
301 previous_min, previous_max = tree.minima[end], tree.maxima[end]
302 low, high = Int[], Int[]
303 for index in 1:(length(values) - width + 1)
304 a, b = previous_min[index], previous_min[index + div(width, 2)]
305 push!(low, values[a + 1] <= values[b + 1] ? a : b)
306 a, b = previous_max[index], previous_max[index + div(width, 2)]
307 push!(high, values[a + 1] >= values[b + 1] ? a : b)
308 end
309 push!(tree.minima, low)
310 push!(tree.maxima, high)
311 width *= 2
312 end
313 tree
314end
315function query(tree::SparseTable, left::Int, right::Int)
316 level, width = 0, 1
317 while 2width <= right - left + 1
318 level += 1
319 width *= 2
320 end
321 ending = right - width + 1
322 a, b = tree.minima[level + 1][left + 1], tree.minima[level + 1][ending + 1]
323 low = tree.values[a + 1] <= tree.values[b + 1] ? a : b
324 a, b = tree.maxima[level + 1][left + 1], tree.maxima[level + 1][ending + 1]
325 high = tree.values[a + 1] >= tree.values[b + 1] ? a : b
326 Report("min=$(tree.values[low + 1]) minpos=$low max=$(tree.values[high + 1]) maxpos=$high level=$level", [tree.values[low + 1], tree.values[high + 1]])
327end
328function dump_structure(tree::SparseTable)

330 for level in eachindex(tree.minima)
331 push!(rows, "MIN $(level - 1) $(numbers(tree.minima[level]))", "MAX $(level - 1) $(numbers(tree.maxima[level]))")
332 end
333 rows
334end

346struct Partition
347 cells::Vector{Int}
348 blocks::UnitRange{Int}
349end
350describe(part::Partition) = "whole=$(length(part.blocks)) cells=$(length(part.cells))"
351function partition(tree::SqrtBlocks, left::Int, right::Int)
352 first, last = div(left, tree.block), div(right, tree.block)
353 first == last && return Partition(collect(left:right), 1:0)
354 cells = vcat(collect(left:((first + 1) * tree.block - 1)), collect((last * tree.block):right))
355 Partition(cells, (first + 1):(last - 1))
356end
357function blocks_add!(tree::SqrtBlocks, left::Int, right::Int, value::Int)
358 part = partition(tree, left, right)
359 for index in part.cells
360 tree.raw[index + 1] += value
361 tree.sums[div(index, tree.block) + 1] += value
362 end
363 for block in part.blocks
364 tree.tags[block + 1] += value
365 tree.sums[block + 1] += value * tree.block
366 end
367 part
368end
369function query(tree::SqrtBlocks, left::Int, right::Int)
370 part = partition(tree, left, right)
371 value = sum((tree.raw[index + 1] + tree.tags[div(index, tree.block) + 1] for index in part.cells); init = 0)
372 value += sum((tree.sums[block + 1] for block in part.blocks); init = 0)
373 Report("sum=$value $(describe(part))", [value])
374end
375function dump_structure(tree::SqrtBlocks)
376 rows = ["SQRT DUMP block=$(tree.block) blocks=$(length(tree.tags))"]
377 for block in eachindex(tree.tags)
378 first, ending = (block - 1) * tree.block, min(block * tree.block, length(tree.raw))
379 push!(rows, "BLOCK $(block - 1) $first $(ending - 1) add=$(tree.tags[block]) sum=$(tree.sums[block]) $(numbers(@view(tree.raw[(first + 1):ending])))")
380 end

384struct Node
385 total::Int
386 left::Int
387 right::Int
388end
389function append_node!(pool::Vector{Node}, value::Int, left::Int = 0, right::Int = 0)
390 push!(pool, Node(value, left, right))
391 length(pool)
392end
393function build_nodes!(pool::Vector{Node}, values::Vector{Int}, left::Int, right::Int)
394 left == right && return append_node!(pool, values[left + 1])
395 mid = div(left + right, 2)
396 a = build_nodes!(pool, values, left, mid)
397 b = build_nodes!(pool, values, mid + 1, right)
398 append_node!(pool, pool[a].total + pool[b].total, a, b)
399end
400function change_nodes!(pool::Vector{Node}, source::Int, left::Int, right::Int, index::Int, value::Int, increment::Bool)
401 node = pool[source]
402 left == right && return append_node!(pool, increment ? node.total + value : value)
403 mid = div(left + right, 2)
404 a, b = node.left, node.right
405 if index <= mid
406 a = change_nodes!(pool, a, left, mid, index, value, increment)
407 else
408 b = change_nodes!(pool, b, mid + 1, right, index, value, increment)
409 end
410 append_node!(pool, pool[a].total + pool[b].total, a, b)
411end
412function query_nodes(pool::Vector{Node}, root::Int, nl::Int, nr::Int, left::Int, right::Int)
413 node = pool[root]
414 left <= nl && nr <= right && return Answer(node.total, 1)
415 mid = div(nl + nr, 2)
416 if right <= mid
417 answer = query_nodes(pool, node.left, nl, mid, left, right)
418 elseif left > mid
419 answer = query_nodes(pool, node.right, mid + 1, nr, left, right)
420 else
421 a = query_nodes(pool, node.left, nl, mid, left, right)
422 b = query_nodes(pool, node.right, mid + 1, nr, left, right)
423 answer = Answer(a.value + b.value, a.visited + b.visited)
424 end
425 Answer(answer.value, answer.visited + 1)
426end
427struct Persistent
428 n::Int
429 position::Vector{Node}
430 value::Vector{Node}
431 roots::Vector{Int}
432 prefixes::Vector{Int}
433 domain::Vector{Int}
434end
435function Persistent(values::Vector{Int})
436 domain = sort(unique(values))
437 tree = Persistent(length(values), Node[], Node[], Int[], Int[], domain)
438 push!(tree.roots, build_nodes!(tree.position, values, 0, tree.n - 1))
439 push!(tree.prefixes, build_nodes!(tree.value, zeros(Int, length(domain)), 0, length(domain) - 1))
440 for value in values
441 rank = searchsortedfirst(domain, value) - 1
442 root = change_nodes!(tree.value, tree.prefixes[end], 0, length(domain) - 1, rank, 1, true)
443 push!(tree.prefixes, root)
444 end
445 tree
446end
447query(tree::Persistent, left::Int, right::Int, version::Int) =
448 report(query_nodes(tree.position, tree.roots[version + 1], 0, tree.n - 1, left, right))
449function persistent_set!(tree::Persistent, version::Int, index::Int, value::Int)
450 before, new_version = length(tree.position), length(tree.roots)
451 root = change_nodes!(tree.position, tree.roots[version + 1], 0, tree.n - 1, index, value, false)
452 push!(tree.roots, root)
453 new_version, length(tree.position) - before
454end
455function kth(tree::Persistent, left::Int, right::Int, k::Int)
456 upper, lower = tree.prefixes[right + 2], tree.prefixes[left + 1]
457 first, last, visited = 0, length(tree.domain) - 1, 1
458 while first < last
459 mid = div(first + last, 2)
460 a, b = tree.value[upper].left, tree.value[lower].left
461 count = tree.value[a].total - tree.value[b].total
462 if k <= count
463 upper, lower, last = a, b, mid
464 else
465 k -= count
466 upper, lower, first = tree.value[upper].right, tree.value[lower].right, mid + 1
467 end
468 visited += 1
469 end
470 Answer(tree.domain[first + 1], visited)
471end

501 x = column
502 while x > 0
503 value += grid.values[row, x]
504 visited += 1
505 x -= lowbit(x)
506 end
507 row -= lowbit(row)
508 end

599 direct = aggregate(values)
600 common = foldl(gcd, values; init = 0)
601 rows = ["CHECK $left $right $(describe(direct)) gcd=$common"]
602 ok, different = 0, 0
603 for kind in ARRAY_KINDS
604 tree = structure(lab, kind)
605 isnothing(tree) && continue
606 expected = if kind == SEG
607 [(direct.total, direct.low, direct.high, common)[Int(tree.op) + 1]]
608 elseif kind == LAZY
609 [direct.total, direct.low, direct.high]
610 elseif kind == SPARSE
611 [direct.low, direct.high]
612 else
613 [direct.total]
614 end
615 version = kind == PERSIST ? length(tree.roots) - 1 : 0
616 result = query_array(lab, kind, left, right; version)
617 same = result.values == expected
618 if same
619 ok += 1
620 else
621 different += 1
622 end
623 prefix = kind == PERSIST ? "version=$version " : ""
624 push!(rows, "$kind $(same ? "OK" : "DIFF") $prefix$(result.text)")
625 end
626 push!(rows, "RESULT ok=$ok diff=$different")
627 rows
628end
629function mo(values::Vector{Int}, queries::Vector{Span})
630 block = isqrt(length(values))
631 order = sort(collect(eachindex(queries)); by = index -> begin
632 query = queries[index]
633 group = div(query.left, block)
634 (group, iseven(group) ? query.right : -query.right, index)
635 end)
636 ranks_by_value = Dict(value => index for (index, value) in enumerate(unique(values)))
637 ranks = [ranks_by_value[value] for value in values]
638 frequencies = zeros(Int, length(ranks_by_value))
639 answers = zeros(Int, length(queries))
640 left, right, moves = 0, -1, 0
641 for index in order
642 target = queries[index]
643 moves += abs(target.left - left) + abs(target.right - right)
644 while left > target.left
645 left -= 1
646 frequencies[ranks[left + 1]] += 1
647 end
648 while right < target.right
649 right += 1
650 frequencies[ranks[right + 1]] += 1
651 end
652 while left < target.left
653 frequencies[ranks[left + 1]] -= 1
654 left += 1
655 end
656 while right > target.right
657 frequencies[ranks[right + 1]] -= 1
658 right -= 1
659 end
660 answers[index] = count(value -> value > 0, frequencies)
661 end
662 rows = ["MO queries=$(length(queries)) block=$block moves=$moves", "ORDER $(numbers(order))"]
663 for (index, query) in enumerate(queries)
664 push!(rows, "ANSWER $index $(query.left) $(query.right) distinct=$(answers[index])")
665 end
666 rows
667end

669function execute!(lab::Laboratory, line::String)
670 words = String.(split(line, ' '))
671 command, n = words[1], length(lab.values)
672 if command == "LOAD"
673 removed = count(kind -> has_structure(lab, kind), ARRAY_KINDS)
674 values = parse.(Int, words[2:end])
675 lab.values, lab.original = values, copy(values)
676 lab.segment = lab.lazy = lab.fenwick = lab.range_fenwick = nothing
677 lab.sparse = lab.blocks = lab.persistent = nothing
678 return ["LOAD n=$(length(values)) removed=$removed"]
679 elseif command == "RESET"
680 lab.values = copy(lab.original)
681 op = isnothing(lab.segment) ? SUM : lab.segment.op
682 block = isnothing(lab.blocks) ? 1 : lab.blocks.block
683 rebuilt = 0
684 for kind in ARRAY_KINDS
685 if has_structure(lab, kind)
686 build_array!(lab, kind; op, block)
687 rebuilt += 1
688 end
689 end
690 return ["RESET rebuilt=$rebuilt"]
691 elseif command in ("ARRAY", "CHECK")
692 left, right = parse.(Int, words[2:end])
693 check_range(n, left, right)
694 command == "CHECK" && return check_all!(lab, left, right)
695 return ["$line $(numbers(@view(lab.values[(left + 1):(right + 1)])))"]
696 elseif command == "MO"
697 args = parse.(Int, words[2:end])
698 queries = Span[]
699 for index in 1:2:length(args)
700 left, right = args[index], args[index + 1]
701 check_range(n, left, right)
702 push!(queries, Span(left, right))
703 end
704 return mo(lab.values, queries)
705 end
706 kind, action = KINDS[command], words[2]
707 if action == "BUILD"
708 op, block = SUM, isqrt(n)
709 if kind == SEG
710 haskey(OPERATIONS, words[3]) || fail("BAD OP $(words[3])")
711 op = OPERATIONS[words[3]]
712 elseif kind == SQRT
713 block = length(words) == 3 ? parse(Int, words[3]) : block
714 block <= n || fail("BAD BLOCK $block")
715 elseif kind == GRID
716 height, width = parse.(Int, words[3:4])
717 actual = length(words) - 4
718 actual == height * width || fail("BAD COUNT $actual $(height * width)")
719 values = parse.(Int, words[5:end])
720 lab.grid = GridTree(height, width, values)
721 return ["GRID BUILD rows=$height cols=$width total=$(sum(values))"]
722 end
723 return [build_array!(lab, kind; op, block)]
724 end
725 tree = structure(lab, kind)
726 isnothing(tree) && fail("NOT BUILT $kind")
727 action == "DUMP" && return dump_structure(tree)
728 args = parse.(Int, words[3:end])
729 if kind == GRID
730 row, column = args[1:2]
731 check_cell(tree, row, column)
732 if action == "ADD"
733 visited = grid_add!(tree, row, column, args[3])
734 return ["$line visited=$visited"]
735 end
736 bottom, right = args[3:4]
737 check_cell(tree, bottom, right)
738 row <= bottom && column <= right || fail("BAD RECT $row $column $bottom $right")
739 result = grid_query(tree, row, column, bottom, right)
740 return ["$line $(report(result).text)"]
741 elseif kind == PERSIST
742 action == "NODES" && return ["PERSIST NODES versions=$(length(tree.roots)) position=$(length(tree.position)) value=$(length(tree.value))"]
743 if action == "KTH"
744 left, right, k = args
745 check_range(n, left, right)
746 1 <= k <= right - left + 1 || fail("BAD K $k")
747 result = kth(tree, left, right, k)
748 return ["$line value=$(result.value) visited=$(result.visited)"]
749 end
750 version = args[1]
751 check_version(tree, version)
752 if action == "SET"
753 check_index(n, args[2])
754 new_version, created = persistent_set!(tree, version, args[2], args[3])
755 return ["$line version=$new_version created=$created"]
756 end
757 left, right = args[2:3]
758 check_range(n, left, right)
759 return ["$line $(query(tree, left, right, version).text)"]
760 elseif kind == FEN && action == "LOWER"
761 negative = findfirst(value -> value < 0, tree.values)
762 isnothing(negative) || fail("NEGATIVE $(negative - 1)")
763 result = bit_lower(tree.tree, args[1])
764 index = result.value < 0 ? "none" : string(result.value)
765 return ["$line index=$index visited=$(result.visited)"]
766 elseif kind == SEG && action == "UPDATE"
767 index, value = args
768 check_index(n, index)
769 visited = segment_update!(tree, 1, index, value)
770 lab.values[index + 1] = value
771 return ["$line root=$(tree.values[1]) visited=$visited"]
772 elseif kind == FEN && action in ("ADD", "PREFIX")
773 index = args[1]
774 check_index(n, index)
775 if action == "ADD"
776 value = args[2]
777 visited = bit_add!(tree.tree, index + 1, value)
778 tree.values[index + 1] += value
779 lab.values[index + 1] += value
780 return ["$line visited=$visited"]
781 end
782 return ["$line $(report(bit_prefix(tree.tree, index + 1)).text)"]
783 end
784 left, right = args[1:2]
785 check_range(n, left, right)
786 if action in ("ADD", "ASSIGN")
787 value, assign = args[3], action == "ASSIGN"
788 details = if kind == LAZY
789 counts = Counts()
790 lazy_update!(tree, 1, left, right, value, assign, counts)
791 "visited=$(counts.visited) covered=$(counts.covered) pushes=$(counts.pushes)"
792 elseif kind == RFEN
793 "visited=$(range_add!(tree, left, right, value))"
794 else
795 describe(blocks_add!(tree, left, right, value))
796 end
797 selected = @view lab.values[(left + 1):(right + 1)]
798 if assign
799 selected .= value
800 else
801 selected .+= value
802 end
803 return ["$line $details"]
804 end
805 ["$line $(query(tree, left, right).text)"]
806end

808function main()
809 eof(stdin) && return
810 parse(Int, readline())
811 lab = Laboratory(parse.(Int, split(readline(), ' ')))
812 command_count = parse(Int, readline())
813 for _ in 1:command_count
814 line = readline()
815 rows = try
816 execute!(lab, line)
817 catch error
818 error isa CommandError || rethrow()
819 ["ERROR $(error.message)"]
820 end
821 for row in rows
822 println(row)
823 end
824 end
825end
