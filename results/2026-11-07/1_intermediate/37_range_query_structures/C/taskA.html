52static int64_t minimum(int64_t a, int64_t b) { return a < b ? a : b; }
53static int64_t maximum(int64_t a, int64_t b) { return a > b ? a : b; }
54static int64_t gcd_value(int64_t a, int64_t b) {
55 if (a < 0) a = -a;
56 if (b < 0) b = -b;
57 while (b) { int64_t rest = a % b; a = b; b = rest; }
58 return a;
59}
60static int isqrt_value(int n) {
61 int result = 1;
62 while ((result + 1) * (result + 1) <= n) result++;
63 return result;
64}
65static int compare_values(const void *a, const void *b) {
66 int64_t x = *(const int64_t *)a, y = *(const int64_t *)b;
67 return (x > y) - (x < y);
68}
69static int unique_values(int64_t *values, int n) {
70 qsort(values, n, sizeof(*values), compare_values);
71 int size = 0;
72 for (int i = 0; i < n; i++) if (!size || values[i] != values[size - 1]) values[size++] = values[i];
73 return size;
74}
75static int rank_of(const int64_t *values, int n, int64_t target) {
76 int left = 0, right = n;
77 while (left < right) {
78 int mid = (left + right) / 2;
79 if (values[mid] < target) left = mid + 1; else right = mid;
80 }
81 return left;
82}
83static Aggregate combine(Aggregate a, Aggregate b) {
84 return (Aggregate){a.sum + b.sum, minimum(a.min, b.min), maximum(a.max, b.max)};
85}
86static int64_t operate(Operation op, int64_t a, int64_t b) {
87 switch (op) {
88 case SUM: return a + b;
89 case MIN: return minimum(a, b);
90 case MAX: return maximum(a, b);
91 case GCD: return gcd_value(a, b);
92 }
93 return 0;
94}
95static void shape(Lab *lab, int node, int left, int right) {
96 lab->bounds[node] = (Bounds){left, right, true};
97 if (node > lab->max_node) lab->max_node = node;
98 if (left == right) return;
99 int mid = (left + right) / 2;
100 shape(lab, 2 * node, left, mid);
101 shape(lab, 2 * node + 1, mid + 1, right);
102}
103static void reset_shape(Lab *lab) {
104 memset(lab->bounds, 0, sizeof(lab->bounds));
105 lab->max_node = 0;
106 shape(lab, 1, 0, lab->n - 1);
107}
108static void segment_build(Lab *lab, int node) {
109 Bounds b = lab->bounds[node];
110 if (b.left == b.right) {
111 int64_t value = lab->values[b.left];
112 lab->seg.values[node] = lab->seg.op == GCD ? gcd_value(value, 0) : value;
113 } else {
114 segment_build(lab, 2 * node);
115 segment_build(lab, 2 * node + 1);
116 lab->seg.values[node] = operate(lab->seg.op, lab->seg.values[2 * node], lab->seg.values[2 * node + 1]);
117 }
118}
119static int64_t segment_query(const Lab *lab, int node, int left, int right, int *visited) {
120 (*visited)++;
121 Bounds b = lab->bounds[node];
122 if (left <= b.left && b.right <= right) return lab->seg.values[node];
123 int mid = (b.left + b.right) / 2;
124 if (right <= mid) return segment_query(lab, 2 * node, left, right, visited);
125 if (left > mid) return segment_query(lab, 2 * node + 1, left, right, visited);
126 int64_t a = segment_query(lab, 2 * node, left, right, visited);
127 int64_t z = segment_query(lab, 2 * node + 1, left, right, visited);
128 return operate(lab->seg.op, a, z);
129}
130static int segment_update(Lab *lab, int node, int index, int64_t value) {
131 Bounds b = lab->bounds[node];
132 if (b.left == b.right) {
133 lab->seg.values[node] = lab->seg.op == GCD ? gcd_value(value, 0) : value;
134 return 1;
135 }
136 int child = 2 * node + (index > (b.left + b.right) / 2);
137 int visited = 1 + segment_update(lab, child, index, value);
138 lab->seg.values[node] = operate(lab->seg.op, lab->seg.values[2 * node], lab->seg.values[2 * node + 1]);
139 return visited;
140}
141static void lazy_build(Lab *lab, int node) {
142 Bounds b = lab->bounds[node];
143 LazyNode *current = &lab->lazy.nodes[node];
144 *current = (LazyNode){0};
145 if (b.left == b.right) {
146 int64_t value = lab->values[b.left];
147 current->value = (Aggregate){value, value, value};
148 } else {
149 lazy_build(lab, 2 * node);
150 lazy_build(lab, 2 * node + 1);
151 current->value = combine(lab->lazy.nodes[2 * node].value, lab->lazy.nodes[2 * node + 1].value);
152 }
153}
154static void lazy_apply(Lab *lab, int node, int64_t value, bool assign) {
155 Bounds b = lab->bounds[node];
156 LazyNode *current = &lab->lazy.nodes[node];
157 int length = b.right - b.left + 1;
158 if (assign) current->value = (Aggregate){value * length, value, value};
159 else {
160 current->value.sum += value * length;
161 current->value.min += value;
162 current->value.max += value;
163 }
164 if (b.left == b.right) return;
165 if (assign) { current->assigned = true; current->assign = value; current->add = 0; }
166 else current->add += value;
167}
168static void lazy_push(Lab *lab, int node, Counts *counts) {
169 LazyNode *current = &lab->lazy.nodes[node];
170 if (!current->assigned && current->add == 0) return;
171 counts->pushes++;
172 for (int child = 2 * node; child <= 2 * node + 1; child++) {
173 if (current->assigned) lazy_apply(lab, child, current->assign, true);
174 if (current->add) lazy_apply(lab, child, current->add, false);
175 }
176 current->assigned = false;
177 current->add = 0;
178}
179static void lazy_update(Lab *lab, int node, int left, int right, int64_t value, bool assign, Counts *counts) {
180 counts->visited++;
181 Bounds b = lab->bounds[node];
182 if (left <= b.left && b.right <= right) {
183 counts->covered++;
184 lazy_apply(lab, node, value, assign);
185 return;
186 }
187 lazy_push(lab, node, counts);
188 int mid = (b.left + b.right) / 2;
189 if (left <= mid) lazy_update(lab, 2 * node, left, right, value, assign, counts);
190 if (right > mid) lazy_update(lab, 2 * node + 1, left, right, value, assign, counts);
191 lab->lazy.nodes[node].value = combine(lab->lazy.nodes[2 * node].value, lab->lazy.nodes[2 * node + 1].value);
192}
193static Aggregate lazy_query(Lab *lab, int node, int left, int right, Counts *counts) {
194 counts->visited++;
195 Bounds b = lab->bounds[node];
196 if (left <= b.left && b.right <= right) return lab->lazy.nodes[node].value;
197 lazy_push(lab, node, counts);
198 int mid = (b.left + b.right) / 2;
199 if (right <= mid) return lazy_query(lab, 2 * node, left, right, counts);
200 if (left > mid) return lazy_query(lab, 2 * node + 1, left, right, counts);
201 Aggregate a = lazy_query(lab, 2 * node, left, right, counts);
202 Aggregate z = lazy_query(lab, 2 * node + 1, left, right, counts);
203 return combine(a, z);
204}
205static int fen_add(int64_t *tree, int n, int position, int64_t value) {
206 int visited = 0;
207 for (int p = position; p <= n; p += p & -p) { tree[p] += value; visited++; }
208 return visited;
209}
210static Answer fen_prefix(const int64_t *tree, int position) {
211 Answer answer = {0};
212 for (int p = position; p > 0; p -= p & -p) { answer.value += tree[p]; answer.visited++; }
213 return answer;
214}
215static Answer fen_range(const int64_t *tree, int left, int right) {
216 Answer a = fen_prefix(tree, right + 1), b = fen_prefix(tree, left);
217 return (Answer){a.value - b.value, a.visited + b.visited};
218}
219static Answer rfen_prefix(const RangeFenwick *tree, int position) {
220 Answer a = fen_prefix(tree->first, position), b = fen_prefix(tree->second, position);
221 return (Answer){position * a.value - b.value, a.visited + b.visited};
222}
223static Answer rfen_range(const RangeFenwick *tree, int left, int right) {
224 Answer a = rfen_prefix(tree, right + 1), b = rfen_prefix(tree, left);
225 return (Answer){a.value - b.value, a.visited + b.visited};
226}
227static void sparse_build(Lab *lab) {
228 Sparse *s = &lab->sparse;
229 memcpy(s->values, lab->values, lab->n * sizeof(int64_t));
230 s->levels = s->entries = 0;
231 for (int length = 1; length <= lab->n; length *= 2) {
232 int level = s->levels++;
233 for (int i = 0; i + length <= lab->n; i++) {
234 if (!level) { s->min[level][i] = s->max[level][i] = i; }
235 else {
236 int a = s->min[level - 1][i], b = s->min[level - 1][i + length / 2];
237 s->min[level][i] = s->values[a] <= s->values[b] ? a : b;
238 a = s->max[level - 1][i]; b = s->max[level - 1][i + length / 2];
239 s->max[level][i] = s->values[a] >= s->values[b] ? a : b;
240 }
241 s->entries++;
242 }
243 }
244}
245static SparseAnswer sparse_query(const Sparse *s, int left, int right) {
246 int level = 0, length = 1;
247 while (2 * length <= right - left + 1) { length *= 2; level++; }
248 int end = right - length + 1;
249 int a = s->min[level][left], b = s->min[level][end];
250 int low = s->values[a] <= s->values[b] ? a : b;
251 a = s->max[level][left]; b = s->max[level][end];
252 return (SparseAnswer){low, s->values[a] >= s->values[b] ? a : b, level};
253}
254static SqrtAnswer sqrt_process(Sqrt *s, int left, int right, int64_t value, bool update) {
255 SqrtAnswer answer = {0};
256 int first = left / s->block, last = right / s->block;
257 for (int i = left; i <= right; i++) {
258 int block = i / s->block;
259 if (block != first && block != last) continue;
260 if (update) { s->values[i] += value; s->sums[block] += value; }
261 else answer.sum += s->values[i] + s->tags[block];
262 answer.cells++;
263 }
264 for (int block = first + 1; block < last; block++) {
265 if (update) { s->tags[block] += value; s->sums[block] += value * s->block; }
266 else answer.sum += s->sums[block];
267 answer.whole++;
268 }
269 return answer;
270}
271static int pool_append(Pool *pool, PersistentNode node) {
272 if (pool->size == pool->capacity) {
273 int capacity = pool->capacity ? 2 * pool->capacity : 4096;
274 PersistentNode *nodes = realloc(pool->nodes, (size_t)capacity * sizeof(*nodes));
275 if (!nodes) exit(EXIT_FAILURE);
276 pool->nodes = nodes;
277 pool->capacity = capacity;
278 }
279 int index = pool->size++;
280 pool->nodes[index] = node;
281 return index;
282}
283static int persistent_build(Pool *pool, const int64_t *values, int left, int right) {
284 int index = pool_append(pool, (PersistentNode){-1, -1, 0});
285 if (left == right) pool->nodes[index].value = values ? values[left] : 0;
286 else {
287 int mid = (left + right) / 2;
288 int a = persistent_build(pool, values, left, mid);
289 int b = persistent_build(pool, values, mid + 1, right);
290 pool->nodes[index] = (PersistentNode){a, b, pool->nodes[a].value + pool->nodes[b].value};
291 }
292 return index;
293}
294static int persistent_set(Pool *pool, int source, int left, int right, int target, int64_t value, bool increment) {
295 PersistentNode node = pool->nodes[source];
296 int index = pool_append(pool, node);
297 if (left == right) pool->nodes[index].value = increment ? node.value + value : value;
298 else {
299 int mid = (left + right) / 2;
300 if (target <= mid) node.left = persistent_set(pool, node.left, left, mid, target, value, increment);
301 else node.right = persistent_set(pool, node.right, mid + 1, right, target, value, increment);
302 node.value = pool->nodes[node.left].value + pool->nodes[node.right].value;
303 pool->nodes[index] = node;
304 }
305 return index;
306}
307static int64_t persistent_query(const Pool *pool, int root, int nl, int nr, int left, int right, int *visited) {
308 (*visited)++;
309 PersistentNode node = pool->nodes[root];
310 if (left <= nl && nr <= right) return node.value;
311 int mid = (nl + nr) / 2;
312 if (right <= mid) return persistent_query(pool, node.left, nl, mid, left, right, visited);
313 if (left > mid) return persistent_query(pool, node.right, mid + 1, nr, left, right, visited);
314 int64_t a = persistent_query(pool, node.left, nl, mid, left, right, visited);
315 int64_t b = persistent_query(pool, node.right, mid + 1, nr, left, right, visited);
316 return a + b;
317}
318static void persist_clear(Persistent *p) {
319 free(p->position.nodes);
320 free(p->value.nodes);
321 *p = (Persistent){0};
322}
323static void persist_build(Lab *lab) {
324 Persistent *p = &lab->persist;
325 persist_clear(p);
326 p->roots[0] = persistent_build(&p->position, lab->values, 0, lab->n - 1);
327 p->versions = 1;
328 memcpy(p->values, lab->values, lab->n * sizeof(int64_t));
329 p->distinct = unique_values(p->values, lab->n);
330 p->prefixes[0] = persistent_build(&p->value, NULL, 0, p->distinct - 1);
331 for (int i = 0; i < lab->n; i++) {
332 int rank = rank_of(p->values, p->distinct, lab->values[i]);
333 p->prefixes[i + 1] = persistent_set(&p->value, p->prefixes[i], 0, p->distinct - 1, rank, 1, true);
334 }
335}
336static Answer persist_kth(const Persistent *p, int left, int right, int k) {
337 int u = p->prefixes[right + 1], w = p->prefixes[left];
338 int low = 0, high = p->distinct - 1, visited = 1;
339 while (low < high) {
340 PersistentNode a = p->value.nodes[u], b = p->value.nodes[w];
341 int64_t count = p->value.nodes[a.left].value - p->value.nodes[b.left].value;
342 int mid = (low + high) / 2;
343 if (k <= count) { u = a.left; w = b.left; high = mid; }
344 else { k -= (int)count; u = a.right; w = b.right; low = mid + 1; }
345 visited++;
346 }
347 return (Answer){p->values[low], visited};
348}
349static int grid_add(Grid *grid, int row, int col, int64_t value) {
350 int visited = 0;
351 for (int y = row + 1; y <= grid->height; y += y & -y) {
352 for (int x = col + 1; x <= grid->width; x += x & -x) { grid->tree[y][x] += value; visited++; }
353 }
354 return visited;
355}
356static Answer grid_prefix(const Grid *grid, int row, int col) {
357 Answer answer = {0};
358 for (int y = row; y > 0; y -= y & -y) {
359 for (int x = col; x > 0; x -= x & -x) { answer.value += grid->tree[y][x]; answer.visited++; }
360 }
361 return answer;
362}

364static void build_array(Lab *lab, Kind kind, int parameter, bool output) {
365 lab->built[kind] = true;
366 int64_t total = 0;
367 if (kind == SEG) {
368 lab->seg.op = (Operation)parameter;
369 segment_build(lab, 1);
370 if (output) printf("SEG BUILD %s nodes=%d maxnode=%d root=%" PRId64 "\n",
371 op_names[parameter], 2 * lab->n - 1, lab->max_node, lab->seg.values[1]);
372 } else if (kind == LAZY) {
373 lazy_build(lab, 1);
374 Aggregate root = lab->lazy.nodes[1].value;
375 if (output) printf("LAZY BUILD nodes=%d sum=%" PRId64 " min=%" PRId64 " max=%" PRId64 "\n",
376 2 * lab->n - 1, root.sum, root.min, root.max);
377 } else if (kind == FEN) {
378 memset(&lab->fen, 0, sizeof(lab->fen));
379 memcpy(lab->fen.values, lab->values, lab->n * sizeof(int64_t));
380 for (int i = 0; i < lab->n; i++) { fen_add(lab->fen.tree, lab->n, i + 1, lab->values[i]); total += lab->values[i]; }
381 if (output) printf("FEN BUILD n=%d total=%" PRId64 "\n", lab->n, total);
382 } else if (kind == RFEN) {
383 memset(&lab->rfen, 0, sizeof(lab->rfen));
384 for (int i = 0; i < lab->n; i++) {
385 int64_t difference = lab->values[i] - (i ? lab->values[i - 1] : 0);
386 fen_add(lab->rfen.first, lab->n, i + 1, difference);
387 fen_add(lab->rfen.second, lab->n, i + 1, i * difference);
388 total += lab->values[i];
389 }
390 if (output) printf("RFEN BUILD n=%d total=%" PRId64 "\n", lab->n, total);
391 } else if (kind == SPARSE) {
392 sparse_build(lab);
393 if (output) printf("SPARSE BUILD levels=%d entries=%d\n", lab->sparse.levels, lab->sparse.entries);
394 } else if (kind == SQRT) {
395 Sqrt *s = &lab->sqrt;
396 memset(s, 0, sizeof(*s));
397 s->block = parameter;
398 s->blocks = (lab->n + parameter - 1) / parameter;
399 memcpy(s->values, lab->values, lab->n * sizeof(int64_t));
400 for (int i = 0; i < lab->n; i++) s->sums[i / parameter] += lab->values[i];
401 if (output) printf("SQRT BUILD block=%d blocks=%d\n", parameter, s->blocks);
402 } else if (kind == PERSIST) {
403 persist_build(lab);
404 if (output) printf("PERSIST BUILD nodes=%d distinct=%d valuenodes=%d\n",
405 lab->persist.position.size, lab->persist.distinct, lab->persist.value.size);
406 }
407}
408static bool query_items(Lab *lab, Kind kind, int version, int left, int right, const Reference *expected, char *out) {
409 if (kind == SEG) {
410 int visited = 0;
411 int64_t answer = segment_query(lab, 1, left, right, &visited);
412 snprintf(out, 256, "%s=%" PRId64 " visited=%d", value_names[lab->seg.op], answer, visited);
413 if (!expected) return true;
414 int64_t values[] = {expected->values.sum, expected->values.min, expected->values.max, expected->gcd};
415 return answer == values[lab->seg.op];
416 }
417 if (kind == LAZY) {
418 Counts counts = {0};
419 Aggregate answer = lazy_query(lab, 1, left, right, &counts);
420 snprintf(out, 256, "sum=%" PRId64 " min=%" PRId64 " max=%" PRId64 " visited=%d pushes=%d",
421 answer.sum, answer.min, answer.max, counts.visited, counts.pushes);
422 return !expected || (answer.sum == expected->values.sum && answer.min == expected->values.min && answer.max == expected->values.max);
423 }
424 if (kind == SPARSE) {
425 SparseAnswer answer = sparse_query(&lab->sparse, left, right);
426 int64_t low = lab->sparse.values[answer.min], high = lab->sparse.values[answer.max];
427 snprintf(out, 256, "min=%" PRId64 " minpos=%d max=%" PRId64 " maxpos=%d level=%d", low, answer.min, high, answer.max, answer.level);
428 return !expected || (low == expected->values.min && high == expected->values.max);
429 }
430 if (kind == SQRT) {
431 SqrtAnswer answer = sqrt_process(&lab->sqrt, left, right, 0, false);
432 snprintf(out, 256, "sum=%" PRId64 " whole=%d cells=%d", answer.sum, answer.whole, answer.cells);
433 return !expected || answer.sum == expected->values.sum;
434 }
435 Answer answer = {0};
436 if (kind == FEN) answer = fen_range(lab->fen.tree, left, right);
437 else if (kind == RFEN) answer = rfen_range(&lab->rfen, left, right);
438 else answer.value = persistent_query(&lab->persist.position, lab->persist.roots[version], 0, lab->n - 1, left, right, &answer.visited);
439 snprintf(out, 256, "sum=%" PRId64 " visited=%d", answer.value, answer.visited);
440 return !expected || answer.value == expected->values.sum;
441}

468static void dump(const Lab *lab, Kind kind) {
469 if (kind == SEG || kind == LAZY) {
470 if (kind == SEG) printf("SEG DUMP %s nodes=%d\n", op_names[lab->seg.op], 2 * lab->n - 1);
471 else printf("LAZY DUMP nodes=%d\n", 2 * lab->n - 1);
472 for (int i = 1; i <= lab->max_node; i++) {
473 Bounds b = lab->bounds[i];
474 if (!b.present) continue;
475 printf("NODE %d %d %d ", i, b.left, b.right);
476 if (kind == SEG) printf("%" PRId64 "\n", lab->seg.values[i]);
477 else {
478 LazyNode node = lab->lazy.nodes[i];
479 printf("sum=%" PRId64 " min=%" PRId64 " max=%" PRId64 " assign=", node.value.sum, node.value.min, node.value.max);
480 if (b.left != b.right && node.assigned) printf("%" PRId64, node.assign); else putchar('-');
481 printf(" add=%" PRId64 "\n", b.left == b.right ? INT64_C(0) : node.add);
482 }
483 }
484 } else if (kind == FEN || kind == RFEN) {
485 printf("%s DUMP n=%d\n", kind_names[kind], lab->n);
486 if (kind == FEN) print_values("T", lab->fen.tree + 1, lab->n);
487 else { print_values("B1", lab->rfen.first + 1, lab->n); print_values("B2", lab->rfen.second + 1, lab->n); }
488 } else if (kind == SPARSE) {
489 printf("SPARSE DUMP levels=%d\n", lab->sparse.levels);
490 for (int level = 0; level < lab->sparse.levels; level++) {
491 printf("MIN %d", level);
492 for (int i = 0; i + (1 << level) <= lab->n; i++) printf(" %d", lab->sparse.min[level][i]);
493 printf("\nMAX %d", level);
494 for (int i = 0; i + (1 << level) <= lab->n; i++) printf(" %d", lab->sparse.max[level][i]);
495 putchar('\n');
496 }
497 } else if (kind == SQRT) {
498 const Sqrt *s = &lab->sqrt;
499 printf("SQRT DUMP block=%d blocks=%d\n", s->block, s->blocks);
500 for (int block = 0; block < s->blocks; block++) {
501 int first = block * s->block, end = (block + 1) * s->block;
502 if (end > lab->n) end = lab->n;
503 printf("BLOCK %d %d %d add=%" PRId64 " sum=%" PRId64, block, first, end - 1, s->tags[block], s->sums[block]);
504 for (int i = first; i < end; i++) printf(" %" PRId64, s->values[i]);
505 putchar('\n');
506 }
507 } else if (kind == GRID) {
508 printf("GRID DUMP rows=%d cols=%d\n", lab->grid.height, lab->grid.width);
509 for (int row = 1; row <= lab->grid.height; row++) {
510 printf("ROW %d", row);
511 for (int col = 1; col <= lab->grid.width; col++) printf(" %" PRId64, lab->grid.tree[row][col]);
512 putchar('\n');
513 }
514 }
515}
516static int compare_queries(const void *a, const void *b) {
517 const MoQuery *x = a, *y = b;
518 if (x->block != y->block) return x->block - y->block;
519 if (x->right != y->right) return x->block % 2 ? y->right - x->right : x->right - y->right;
520 return x->number - y->number;
521}
522static void mo(const Lab *lab, MoQuery *queries, int count) {
523 int64_t sorted[MAX_N];
524 int ranks[MAX_N], frequencies[MAX_N] = {0}, answers[100];
525 MoQuery original[100];
526 memcpy(original, queries, count * sizeof(*queries));
527 memcpy(sorted, lab->values, lab->n * sizeof(int64_t));
528 int distinct_values = unique_values(sorted, lab->n);
529 for (int i = 0; i < lab->n; i++) ranks[i] = rank_of(sorted, distinct_values, lab->values[i]);
530 qsort(queries, count, sizeof(*queries), compare_queries);
531 int left = 0, right = -1, moves = 0, distinct = 0;
532 for (int i = 0; i < count; i++) {
533 MoQuery q = queries[i];
534 moves += abs(q.left - left) + abs(q.right - right);
535 while (left > q.left) if (frequencies[ranks[--left]]++ == 0) distinct++;
536 while (right < q.right) if (frequencies[ranks[++right]]++ == 0) distinct++;
537 while (left < q.left) if (--frequencies[ranks[left++]] == 0) distinct--;
538 while (right > q.right) if (--frequencies[ranks[right--]] == 0) distinct--;
539 answers[q.number - 1] = distinct;
540 }
541 printf("MO queries=%d block=%d moves=%d\nORDER", count, isqrt_value(lab->n), moves);
542 for (int i = 0; i < count; i++) printf(" %d", queries[i].number);
543 putchar('\n');
544 for (int i = 0; i < count; i++) printf("ANSWER %d %d %d distinct=%d\n", i + 1, original[i].left, original[i].right, answers[i]);
545}
546static void check_all(Lab *lab, int left, int right) {
547 Reference reference = {{0, lab->values[left], lab->values[left]}, 0};
548 for (int i = left; i <= right; i++) {
549 reference.values.sum += lab->values[i];
550 reference.values.min = minimum(reference.values.min, lab->values[i]);
551 reference.values.max = maximum(reference.values.max, lab->values[i]);
552 reference.gcd = gcd_value(reference.gcd, lab->values[i]);
553 }
554 printf("CHECK %d %d sum=%" PRId64 " min=%" PRId64 " max=%" PRId64 " gcd=%" PRId64 "\n",
555 left, right, reference.values.sum, reference.values.min, reference.values.max, reference.gcd);
556 int ok = 0, diff = 0;
557 for (Kind kind = SEG; kind < GRID; kind++) {
558 if (!lab->built[kind]) continue;
559 char out[256];
560 int version = lab->persist.versions - 1;
561 bool same = query_items(lab, kind, version, left, right, &reference, out);
562 if (same) ok++; else diff++;
563 printf("%s %s ", kind_names[kind], same ? "OK" : "DIFF");
564 if (kind == PERSIST) printf("version=%d ", version);
565 puts(out);
566 }
567 printf("RESULT ok=%d diff=%d\n", ok, diff);
568}
569static int64_t number(char **words, int index) { return strtoll(words[index], NULL, 10); }
570static void execute(Lab *lab, char **words, int count) {
571 const char *command = words[0];
572 if (!strcmp(command, "LOAD")) {
573 int removed = 0;
574 for (Kind kind = SEG; kind < GRID; kind++) { removed += lab->built[kind]; lab->built[kind] = false; }
575 persist_clear(&lab->persist);
576 lab->n = count - 1;
577 for (int i = 0; i < lab->n; i++) lab->values[i] = lab->original[i] = number(words, i + 1);
578 reset_shape(lab);
579 printf("LOAD n=%d removed=%d\n", lab->n, removed);
580 return;
581 }
582 if (!strcmp(command, "RESET")) {
583 memcpy(lab->values, lab->original, lab->n * sizeof(int64_t));
584 int rebuilt = 0;
585 for (Kind kind = SEG; kind < GRID; kind++) if (lab->built[kind]) {
586 int parameter = kind == SEG ? (int)lab->seg.op : kind == SQRT ? lab->sqrt.block : 0;
587 build_array(lab, kind, parameter, false);
588 rebuilt++;
589 }
590 printf("RESET rebuilt=%d\n", rebuilt);
591 return;
592 }
593 if (!strcmp(command, "ARRAY") || !strcmp(command, "CHECK")) {
594 int left = (int)number(words, 1), right = (int)number(words, 2);
595 if (!check_range(lab, left, right)) return;
596 if (!strcmp(command, "CHECK")) check_all(lab, left, right);
597 else {
598 printf("ARRAY %d %d", left, right);
599 for (int i = left; i <= right; i++) printf(" %" PRId64, lab->values[i]);
600 putchar('\n');
601 }
602 return;
603 }
604 if (!strcmp(command, "MO")) {
605 MoQuery queries[100];
606 int total = (count - 1) / 2, block = isqrt_value(lab->n);
607 for (int i = 0; i < total; i++) {
608 int left = (int)number(words, 2 * i + 1), right = (int)number(words, 2 * i + 2);
609 if (!check_range(lab, left, right)) return;
610 queries[i] = (MoQuery){i + 1, left, right, left / block};
611 }
612 mo(lab, queries, total);
613 return;
614 }
615 Kind kind = SEG;
616 while (strcmp(command, kind_names[kind])) kind++;
617 const char *action = words[1];
618 if (!strcmp(action, "BUILD")) {
619 int parameter = 0;
620 if (kind == SEG) {
621 while (parameter < 4 && strcmp(words[2], op_names[parameter])) parameter++;
622 if (parameter == 4) { printf("ERROR BAD OP %s\n", words[2]); return; }
623 } else if (kind == SQRT) {
624 parameter = count == 3 ? (int)number(words, 2) : isqrt_value(lab->n);
625 if (parameter > lab->n) { printf("ERROR BAD BLOCK %d\n", parameter); return; }
626 } else if (kind == GRID) {
627 int height = (int)number(words, 2), width = (int)number(words, 3);
628 if (count - 4 != height * width) { printf("ERROR BAD COUNT %d %d\n", count - 4, height * width); return; }
629 lab->grid = (Grid){.height = height, .width = width};
630 int64_t total = 0;
631 for (int i = 0; i < height * width; i++) {
632 int64_t value = number(words, i + 4);
633 grid_add(&lab->grid, i / width, i % width, value);
634 total += value;
635 }
636 lab->built[GRID] = true;
637 printf("GRID BUILD rows=%d cols=%d total=%" PRId64 "\n", height, width, total);
638 return;
639 }
640 build_array(lab, kind, parameter, true);
641 return;
642 }
643 if (!lab->built[kind]) { printf("ERROR NOT BUILT %s\n", command); return; }
644 if (!strcmp(action, "DUMP")) { dump(lab, kind); return; }
645 if (kind == GRID) {
646 int row = (int)number(words, 2), col = (int)number(words, 3);
647 if (!check_cell(&lab->grid, row, col)) return;
648 if (!strcmp(action, "ADD")) {
649 int64_t value = number(words, 4);
650 int visited = grid_add(&lab->grid, row, col, value);
651 printf("GRID ADD %d %d %" PRId64 " visited=%d\n", row, col, value, visited);
652 } else {
653 int bottom = (int)number(words, 4), right = (int)number(words, 5);
654 if (!check_cell(&lab->grid, bottom, right)) return;
655 if (row > bottom || col > right) { printf("ERROR BAD RECT %d %d %d %d\n", row, col, bottom, right); return; }
656 Answer a = grid_prefix(&lab->grid, bottom + 1, right + 1), b = grid_prefix(&lab->grid, row, right + 1);
657 Answer c = grid_prefix(&lab->grid, bottom + 1, col), d = grid_prefix(&lab->grid, row, col);
658 printf("GRID SUM %d %d %d %d sum=%" PRId64 " visited=%d\n", row, col, bottom, right,
659 a.value - b.value - c.value + d.value, a.visited + b.visited + c.visited + d.visited);
660 }
661 return;
662 }
663 if (kind == PERSIST) {
664 Persistent *p = &lab->persist;
665 if (!strcmp(action, "NODES")) {
666 printf("PERSIST NODES versions=%d position=%d value=%d\n", p->versions, p->position.size, p->value.size);
667 return;
668 }
669 if (!strcmp(action, "KTH")) {
670 int left = (int)number(words, 2), right = (int)number(words, 3), k = (int)number(words, 4);
671 if (!check_range(lab, left, right)) return;
672 if (k < 1 || k > right - left + 1) { printf("ERROR BAD K %d\n", k); return; }
673 Answer answer = persist_kth(p, left, right, k);
674 printf("PERSIST KTH %d %d %d value=%" PRId64 " visited=%d\n", left, right, k, answer.value, answer.visited);
675 return;
676 }
677 int version = (int)number(words, 2), left = (int)number(words, 3);
678 if (!check_version(lab, version)) return;
679 if (!strcmp(action, "SET")) {
680 if (!check_index(lab, left)) return;
681 int64_t value = number(words, 4);
682 int old = p->position.size, next = p->versions;
683 p->roots[next] = persistent_set(&p->position, p->roots[version], 0, lab->n - 1, left, value, false);
684 p->versions++;
685 printf("PERSIST SET %d %d %" PRId64 " version=%d created=%d\n", version, left, value, next, p->position.size - old);
686 } else {
687 int right = (int)number(words, 4);
688 if (!check_range(lab, left, right)) return;
689 char out[256];
690 query_items(lab, PERSIST, version, left, right, NULL, out);
691 printf("PERSIST QUERY %d %d %d %s\n", version, left, right, out);
692 }
693 return;
694 }
695 if (kind == FEN && !strcmp(action, "LOWER")) {
696 for (int i = 0; i < lab->n; i++) if (lab->fen.values[i] < 0) { printf("ERROR NEGATIVE %d\n", i); return; }
697 int64_t k = number(words, 2), rest = k;
698 int step = 1, position = 0, visited = 0;
699 while (2 * step <= lab->n) step *= 2;
700 for (; step > 0; step /= 2) if (position + step <= lab->n) {
701 visited++;
702 if (lab->fen.tree[position + step] < rest) { rest -= lab->fen.tree[position + step]; position += step; }
703 }
704 printf("FEN LOWER %" PRId64 " index=", k);
705 if (position == lab->n) fputs("none", stdout); else printf("%d", position);
706 printf(" visited=%d\n", visited);
707 return;
708 }
709 int left = (int)number(words, 2);
710 if (kind == SEG && !strcmp(action, "UPDATE")) {
711 if (!check_index(lab, left)) return;
712 int64_t value = number(words, 3);
713 lab->values[left] = value;
714 int visited = segment_update(lab, 1, left, value);
715 printf("SEG UPDATE %d %" PRId64 " root=%" PRId64 " visited=%d\n", left, value, lab->seg.values[1], visited);
716 return;
717 }
718 if (kind == FEN && (!strcmp(action, "ADD") || !strcmp(action, "PREFIX"))) {
719 if (!check_index(lab, left)) return;
720 if (!strcmp(action, "ADD")) {
721 int64_t value = number(words, 3);
722 lab->values[left] += value;
723 lab->fen.values[left] += value;
724 int visited = fen_add(lab->fen.tree, lab->n, left + 1, value);
725 printf("FEN ADD %d %" PRId64 " visited=%d\n", left, value, visited);
726 } else {
727 Answer answer = fen_prefix(lab->fen.tree, left + 1);
728 printf("FEN PREFIX %d sum=%" PRId64 " visited=%d\n", left, answer.value, answer.visited);
729 }
730 return;
731 }
732 int right = (int)number(words, 3);
733 if (!check_range(lab, left, right)) return;
734 if (!strcmp(action, "ADD") || !strcmp(action, "ASSIGN")) {
735 int64_t value = number(words, 4);
736 bool assign = !strcmp(action, "ASSIGN");
737 for (int i = left; i <= right; i++) lab->values[i] = assign ? value : lab->values[i] + value;
738 if (kind == LAZY) {
739 Counts counts = {0};
740 lazy_update(lab, 1, left, right, value, assign, &counts);
741 printf("LAZY %s %d %d %" PRId64 " visited=%d covered=%d pushes=%d\n", action, left, right, value, counts.visited, counts.covered, counts.pushes);
742 } else if (kind == RFEN) {
743 int visited = fen_add(lab->rfen.first, lab->n, left + 1, value);
744 visited += fen_add(lab->rfen.first, lab->n, right + 2, -value);
745 visited += fen_add(lab->rfen.second, lab->n, left + 1, value * left);
746 visited += fen_add(lab->rfen.second, lab->n, right + 2, -value * (right + 1));
747 printf("RFEN ADD %d %d %" PRId64 " visited=%d\n", left, right, value, visited);
748 } else {
749 SqrtAnswer answer = sqrt_process(&lab->sqrt, left, right, value, true);
750 printf("SQRT ADD %d %d %" PRId64 " whole=%d cells=%d\n", left, right, value, answer.whole, answer.cells);
751 }
752 } else {
753 char out[256];
754 query_items(lab, kind, 0, left, right, NULL, out);
755 printf("%s %s %d %d %s\n", command, action, left, right, out);
756 }
757}
758static int tokenize(char *line, char **words) {
759 int count = 0;
760 for (char *word = strtok(line, " \r\n"); word; word = strtok(NULL, " \r\n")) words[count++] = word;
761 return count;
762}
763int main(void) {
764 static Lab lab;
765 char line[30002], *words[MAX_N + 2];
766 if (!fgets(line, sizeof(line), stdin)) return 0;
767 lab.n = atoi(line);
768 if (!fgets(line, sizeof(line), stdin)) return 0;
769 tokenize(line, words);
770 for (int i = 0; i < lab.n; i++) lab.values[i] = lab.original[i] = number(words, i);
771 reset_shape(&lab);
772 if (!fgets(line, sizeof(line), stdin)) return 0;
773 int commands = atoi(line);
774 for (int i = 0; i < commands; i++) {
775 if (!fgets(line, sizeof(line), stdin)) break;
776 int count = tokenize(line, words);
777 execute(&lab, words, count);
778 }
779 persist_clear(&lab.persist);
780 return 0;
781}
