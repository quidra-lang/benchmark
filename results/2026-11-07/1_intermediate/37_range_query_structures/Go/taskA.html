30type Operation int
31
32const (
33 SUM Operation = iota
34 MIN
35 MAX
36 GCD
37)
38
39var opNames = [...]string{"SUM", "MIN", "MAX", "GCD"}
40var valueNames = [...]string{"sum", "min", "max", "gcd"}
41
42func (op Operation) String() string { return opNames[op] }
43func (op Operation) combine(a, b int64) int64 {
44 switch op {
45 case SUM:
46 return a + b
47 case MIN:
48 return min(a, b)
49 case MAX:
50 return max(a, b)
51 default:
52 return gcd(a, b)
53 }
54}
55func (op Operation) leaf(value int64) int64 {
56 if op == GCD && value < 0 {
57 return -value
58 }
59 return value
60}
61func gcd(a, b int64) int64 {
62 if a < 0 {
63 a = -a
64 }
65 if b < 0 {
66 b = -b
67 }
68 for b != 0 {
69 a, b = b, a%b
70 }
71 return a
72}
73func isqrt(n int) int {
74 result := 1
75 for (result+1)*(result+1) <= n {
76 result++
77 }
78 return result
79}
80func joined[T ~int | ~int64](values []T) string {
81 words := make([]string, len(values))
82 for i, value := range values {
83 words[i] = strconv.FormatInt(int64(value), 10)
84 }
85 return strings.Join(words, " ")
86}
87func total(values []int64) int64 {
88 var sum int64
89 for _, value := range values {
90 sum += value
91 }
92 return sum

95type Span struct{ left, right int }
96
97func whole(n int) Span { return Span{0, n - 1} }
98func (s Span) leaf() bool { return s.left == s.right }
99func (s Span) length() int { return s.right - s.left + 1 }
100func (s Span) covers(other Span) bool { return s.left <= other.left && other.right <= s.right }
101func (s Span) split() (Span, Span) {
102 mid := (s.left + s.right) / 2
103 return Span{s.left, mid}, Span{mid + 1, s.right}
104}

106type NumberedSpan struct {
107 node int
108 span Span
109}
110
111func shape(n int) []NumberedSpan {
112 nodes := make([]NumberedSpan, 0, 2*n-1)
113 var visit func(int, Span)
114 visit = func(node int, span Span) {
115 nodes = append(nodes, NumberedSpan{node, span})
116 if !span.leaf() {
117 left, right := span.split()
118 visit(2*node, left)
119 visit(2*node+1, right)
120 }
121 }
122 visit(1, whole(n))
123 slices.SortFunc(nodes, func(a, b NumberedSpan) int { return cmp.Compare(a.node, b.node) })
124 return nodes
125}

129func (a Aggregate) combine(b Aggregate) Aggregate {

150type Segment struct {
151 op Operation
152 n, maxNode int
153 tree []int64
154}
155
156func newSegment(values []int64, op Operation) *Segment {
157 tree := &Segment{op: op, n: len(values), tree: make([]int64, 4*len(values))}
158 tree.build(1, whole(len(values)), values)
159 return tree
160}
161func (t *Segment) build(node int, span Span, values []int64) {
162 t.maxNode = max(t.maxNode, node)
163 if span.leaf() {
164 t.tree[node] = t.op.leaf(values[span.left])
165 return
166 }
167 left, right := span.split()
168 t.build(2*node, left, values)
169 t.build(2*node+1, right, values)
170 t.tree[node] = t.op.combine(t.tree[2*node], t.tree[2*node+1])
171}
172func (t *Segment) queryAt(node int, span, wanted Span) Answer {
173 if wanted.covers(span) {
174 return Answer{t.tree[node], 1}
175 }
176 left, right := span.split()
177 var answer Answer
178 if wanted.right <= left.right {
179 answer = t.queryAt(2*node, left, wanted)
180 } else if wanted.left >= right.left {
181 answer = t.queryAt(2*node+1, right, wanted)
182 } else {
183 a := t.queryAt(2*node, left, wanted)
184 b := t.queryAt(2*node+1, right, wanted)
185 answer = Answer{t.op.combine(a.value, b.value), a.visited + b.visited}
186 }
187 answer.visited++
188 return answer
189}
190func (t *Segment) query(span Span) Report {
191 a := t.queryAt(1, whole(t.n), span)
192 return Report{fmt.Sprintf("%s=%d visited=%d", valueNames[t.op], a.value, a.visited), []int64{a.value}}
193}
194func (t *Segment) updateAt(node int, span Span, index int, value int64) int {
195 if span.leaf() {
196 t.tree[node] = t.op.leaf(value)
197 return 1
198 }
199 left, right := span.split()
200 var visited int
201 if index <= left.right {
202 visited = t.updateAt(2*node, left, index, value)
203 } else {
204 visited = t.updateAt(2*node+1, right, index, value)
205 }
206 t.tree[node] = t.op.combine(t.tree[2*node], t.tree[2*node+1])
207 return visited + 1
208}
209func (t *Segment) dump() []string {
210 rows := []string{fmt.Sprintf("SEG DUMP %s nodes=%d", t.op, 2*t.n-1)}
211 for _, entry := range shape(t.n) {
212 rows = append(rows, fmt.Sprintf("NODE %d %d %d %d", entry.node, entry.span.left, entry.span.right, t.tree[entry.node]))
213 }
214 return rows
215}

217type LazyNode struct {
218 value Aggregate
219 assign, add int64
220 assigned bool
221}
222type Lazy struct {
223 n int
224 nodes []LazyNode
225}
226
227func newLazy(values []int64) *Lazy {
228 tree := &Lazy{len(values), make([]LazyNode, 4*len(values))}
229 tree.build(1, whole(len(values)), values)
230 return tree
231}
232func (t *Lazy) build(node int, span Span, values []int64) {
233 if span.leaf() {
234 x := values[span.left]
235 t.nodes[node].value = Aggregate{x, x, x}
236 return
237 }
238 left, right := span.split()
239 t.build(2*node, left, values)
240 t.build(2*node+1, right, values)
241 t.pull(node)
242}
243func (t *Lazy) pull(node int) {
244 t.nodes[node].value = t.nodes[2*node].value.combine(t.nodes[2*node+1].value)
245}
246func (t *Lazy) apply(node int, span Span, value int64, assign bool) {
247 current := &t.nodes[node]
248 if assign {
249 current.value = Aggregate{value * int64(span.length()), value, value}
250 } else {
251 current.value.sum += value * int64(span.length())
252 current.value.min += value
253 current.value.max += value
254 }
255 if !span.leaf() {
256 if assign {
257 current.assigned, current.assign, current.add = true, value, 0
258 } else {
259 current.add += value
260 }
261 }
262}
263func (t *Lazy) push(node int, span Span, counts *Counts) {
264 current := t.nodes[node]
265 if !current.assigned && current.add == 0 {
266 return
267 }
268 counts.pushes++
269 left, right := span.split()
270 for _, child := range []NumberedSpan{{2 * node, left}, {2*node + 1, right}} {
271 if current.assigned {
272 t.apply(child.node, child.span, current.assign, true)
273 }
274 if current.add != 0 {
275 t.apply(child.node, child.span, current.add, false)
276 }
277 }
278 t.nodes[node].assigned, t.nodes[node].add = false, 0
279}
280func (t *Lazy) updateAt(node int, span, wanted Span, value int64, assign bool, counts *Counts) {
281 counts.visited++
282 if wanted.covers(span) {
283 counts.covered++
284 t.apply(node, span, value, assign)
285 return
286 }
287 t.push(node, span, counts)
288 left, right := span.split()
289 if wanted.left <= left.right {
290 t.updateAt(2*node, left, wanted, value, assign, counts)
291 }
292 if wanted.right >= right.left {
293 t.updateAt(2*node+1, right, wanted, value, assign, counts)
294 }
295 t.pull(node)
296}
297func (t *Lazy) update(span Span, value int64, assign bool) Counts {
298 var counts Counts
299 t.updateAt(1, whole(t.n), span, value, assign, &counts)
300 return counts
301}
302func (t *Lazy) queryAt(node int, span, wanted Span, counts *Counts) Aggregate {
303 counts.visited++
304 if wanted.covers(span) {
305 return t.nodes[node].value
306 }
307 t.push(node, span, counts)
308 left, right := span.split()
309 if wanted.right <= left.right {
310 return t.queryAt(2*node, left, wanted, counts)
311 }
312 if wanted.left >= right.left {
313 return t.queryAt(2*node+1, right, wanted, counts)
314 }
315 a := t.queryAt(2*node, left, wanted, counts)
316 b := t.queryAt(2*node+1, right, wanted, counts)
317 return a.combine(b)
318}
319func (t *Lazy) query(span Span) Report {
320 var counts Counts
321 a := t.queryAt(1, whole(t.n), span, &counts)
322 return Report{fmt.Sprintf("%s visited=%d pushes=%d", a.text(), counts.visited, counts.pushes), []int64{a.sum, a.min, a.max}}
323}
324func (t *Lazy) dump() []string {
325 rows := []string{fmt.Sprintf("LAZY DUMP nodes=%d", 2*t.n-1)}
326 for _, entry := range shape(t.n) {
327 node := t.nodes[entry.node]
328 assign := "-"
329 if node.assigned {
330 assign = strconv.FormatInt(node.assign, 10)
331 }
332 rows = append(rows, fmt.Sprintf("NODE %d %d %d %s assign=%s add=%d", entry.node, entry.span.left, entry.span.right, node.value.text(), assign, node.add))
333 }
334 return rows
335}

337type Bit struct{ tree []int64 }
338
339func emptyBit(n int) Bit { return Bit{make([]int64, n+1)} }
340func newBit(values []int64) Bit {
341 bit := emptyBit(len(values))
342 for i, value := range values {
343 bit.add(i+1, value)
344 }
345 return bit
346}
347func (t *Bit) add(position int, value int64) int {
348 visited := 0
349 for ; position < len(t.tree); position += position & -position {
350 t.tree[position] += value
351 visited++
352 }
353 return visited
354}
355func (t *Bit) prefix(position int) Answer {
356 var answer Answer
357 for ; position > 0; position -= position & -position {
358 answer.value += t.tree[position]
359 answer.visited++
360 }
361 return answer
362}
363func (t *Bit) query(span Span) Answer {
364 a := t.prefix(span.right + 1)
365 b := t.prefix(span.left)
366 return Answer{a.value - b.value, a.visited + b.visited}
367}
368func (t *Bit) lower(rest int64) (int, int) {
369 n := len(t.tree) - 1
370 step, position, visited := 1, 0, 0
371 for 2*step <= n {
372 step *= 2
373 }
374 for ; step > 0; step /= 2 {
375 if position+step <= n {
376 visited++
377 if t.tree[position+step] < rest {
378 rest -= t.tree[position+step]
379 position += step
380 }
381 }
382 }
383 if position == n {
384 position = -1
385 }
386 return position, visited
387}

389type Fenwick struct {
390 values []int64
391 bit Bit
392}
393type RangeFenwick struct{ first, second Bit }
394
395func newRangeFenwick(values []int64) *RangeFenwick {
396 tree := &RangeFenwick{emptyBit(len(values)), emptyBit(len(values))}
397 for i, value := range values {
398 difference := value
399 if i > 0 {
400 difference -= values[i-1]
401 }
402 tree.first.add(i+1, difference)
403 tree.second.add(i+1, difference*int64(i))
404 }
405 return tree
406}
407func (t *RangeFenwick) add(span Span, value int64) int {
408 visited := t.first.add(span.left+1, value)
409 visited += t.first.add(span.right+2, -value)
410 visited += t.second.add(span.left+1, value*int64(span.left))
411 visited += t.second.add(span.right+2, -value*int64(span.right+1))
412 return visited
413}
414func (t *RangeFenwick) prefix(position int) Answer {
415 a := t.first.prefix(position)
416 b := t.second.prefix(position)
417 return Answer{int64(position)*a.value - b.value, a.visited + b.visited}
418}
419func (t *RangeFenwick) query(span Span) Answer {
420 a := t.prefix(span.right + 1)
421 b := t.prefix(span.left)
422 return Answer{a.value - b.value, a.visited + b.visited}
423}

425type Sparse struct {
426 values []int64
427 minimum, maximum [][]int
428 entries int
429}
430
431func newSparse(values []int64) *Sparse {
432 tree := &Sparse{values: slices.Clone(values)}
433 base := make([]int, len(values))
434 for i := range base {
435 base[i] = i
436 }
437 tree.minimum, tree.maximum = [][]int{base}, [][]int{slices.Clone(base)}
438 tree.entries = len(values)
439 for length := 2; length <= len(values); length *= 2 {
440 level := len(tree.minimum)
441 lows, highs := make([]int, len(values)-length+1), make([]int, len(values)-length+1)
442 for i := range lows {
443 a, b := tree.minimum[level-1][i], tree.minimum[level-1][i+length/2]
444 lows[i] = a
445 if values[a] > values[b] {
446 lows[i] = b
447 }
448 a, b = tree.maximum[level-1][i], tree.maximum[level-1][i+length/2]
449 highs[i] = a
450 if values[a] < values[b] {
451 highs[i] = b
452 }
453 }
454 tree.minimum = append(tree.minimum, lows)
455 tree.maximum = append(tree.maximum, highs)
456 tree.entries += len(lows)
457 }
458 return tree
459}
460func (t *Sparse) query(span Span) Report {
461 level, length := 0, 1
462 for 2*length <= span.length() {
463 level++
464 length *= 2
465 }
466 end := span.right + 1 - length
467 a, b := t.minimum[level][span.left], t.minimum[level][end]
468 low := a
469 if t.values[a] > t.values[b] {
470 low = b
471 }
472 a, b = t.maximum[level][span.left], t.maximum[level][end]
473 high := a
474 if t.values[a] < t.values[b] {
475 high = b
476 }
477 return Report{fmt.Sprintf("min=%d minpos=%d max=%d maxpos=%d level=%d", t.values[low], low, t.values[high], high, level), []int64{t.values[low], t.values[high]}}
478}
479func (t *Sparse) dump() []string {
480 rows := []string{fmt.Sprintf("SPARSE DUMP levels=%d", len(t.minimum))}
481 for level := range t.minimum {
482 rows = append(rows, fmt.Sprintf("MIN %d %s", level, joined(t.minimum[level])), fmt.Sprintf("MAX %d %s", level, joined(t.maximum[level])))
483 }
484 return rows
485}

487type Sqrt struct {
488 block int
489 values, tags, sums []int64
490}
491
492func newSqrt(values []int64, block int) *Sqrt {
493 blocks := (len(values) + block - 1) / block
494 tree := &Sqrt{block, slices.Clone(values), make([]int64, blocks), make([]int64, blocks)}
495 for i, value := range values {
496 tree.sums[i/block] += value
497 }
498 return tree
499}
500func (t *Sqrt) split(span Span) (cells, blocks []int) {
501 first, last := span.left/t.block, span.right/t.block
502 for i := span.left; i <= span.right; i++ {
503 if i/t.block == first || i/t.block == last {
504 cells = append(cells, i)
505 }
506 }
507 for block := first + 1; block < last; block++ {
508 blocks = append(blocks, block)
509 }
510 return
511}
512func (t *Sqrt) add(span Span, value int64) (int, int) {
513 cells, blocks := t.split(span)
514 for _, i := range cells {
515 t.values[i] += value
516 t.sums[i/t.block] += value
517 }
518 for _, block := range blocks {
519 t.tags[block] += value
520 t.sums[block] += value * int64(t.block)
521 }
522 return len(blocks), len(cells)
523}
524func (t *Sqrt) query(span Span) Report {
525 cells, blocks := t.split(span)
526 var sum int64
527 for _, i := range cells {
528 sum += t.values[i] + t.tags[i/t.block]
529 }
530 for _, block := range blocks {
531 sum += t.sums[block]
532 }
533 return Report{fmt.Sprintf("sum=%d whole=%d cells=%d", sum, len(blocks), len(cells)), []int64{sum}}
534}
535func (t *Sqrt) dump() []string {
536 rows := []string{fmt.Sprintf("SQRT DUMP block=%d blocks=%d", t.block, len(t.tags))}
537 for block := range t.tags {
538 first, end := block*t.block, min((block+1)*t.block, len(t.values))
539 rows = append(rows, fmt.Sprintf("BLOCK %d %d %d add=%d sum=%d %s", block, first, end-1, t.tags[block], t.sums[block], joined(t.values[first:end])))
540 }
541 return rows
542}

544type PersistentNode struct {
545 value int64
546 left, right *PersistentNode
547}
548
549func buildNode(values []int64, span Span, created *int) *PersistentNode {
550 *created += 1
551 if span.leaf() {
552 var value int64
553 if values != nil {
554 value = values[span.left]
555 }
556 return &PersistentNode{value: value}
557 }
558 leftSpan, rightSpan := span.split()
559 left := buildNode(values, leftSpan, created)
560 right := buildNode(values, rightSpan, created)
561 return &PersistentNode{left.value + right.value, left, right}
562}
563func setNode(source *PersistentNode, span Span, index int, value int64, increment bool, created *int) *PersistentNode {
564 *created += 1
565 node := *source
566 if span.leaf() {
567 if increment {
568 node.value += value
569 } else {
570 node.value = value
571 }
572 } else {
573 left, right := span.split()
574 if index <= left.right {
575 node.left = setNode(node.left, left, index, value, increment, created)
576 } else {
577 node.right = setNode(node.right, right, index, value, increment, created)
578 }
579 node.value = node.left.value + node.right.value
580 }
581 return &node
582}
583func (t *PersistentNode) query(span, wanted Span) Answer {
584 if wanted.covers(span) {
585 return Answer{t.value, 1}
586 }
587 left, right := span.split()
588 var answer Answer
589 if wanted.right <= left.right {
590 answer = t.left.query(left, wanted)
591 } else if wanted.left >= right.left {
592 answer = t.right.query(right, wanted)
593 } else {
594 a := t.left.query(left, wanted)
595 b := t.right.query(right, wanted)
596 answer = Answer{a.value + b.value, a.visited + b.visited}
597 }
598 answer.visited++
599 return answer
600}

602type Persistent struct {
603 n int
604 roots, prefixes []*PersistentNode
605 values []int64
606 positionNodes, valueNodes int
607}
608
609func newPersistent(values []int64) *Persistent {
610 tree := &Persistent{n: len(values), values: slices.Clone(values)}
611 tree.roots = []*PersistentNode{buildNode(values, whole(len(values)), &tree.positionNodes)}
612 slices.Sort(tree.values)
613 tree.values = slices.Compact(tree.values)
614 span := whole(len(tree.values))
615 tree.prefixes = []*PersistentNode{buildNode(nil, span, &tree.valueNodes)}
616 for _, value := range values {
617 rank, _ := slices.BinarySearch(tree.values, value)
618 root := setNode(tree.prefixes[len(tree.prefixes)-1], span, rank, 1, true, &tree.valueNodes)
619 tree.prefixes = append(tree.prefixes, root)
620 }
621 return tree
622}
623func (t *Persistent) set(version, index int, value int64) (int, int) {
624 before, next := t.positionNodes, len(t.roots)
625 root := setNode(t.roots[version], whole(t.n), index, value, false, &t.positionNodes)
626 t.roots = append(t.roots, root)
627 return next, t.positionNodes - before
628}
629func (t *Persistent) query(version int, span Span) Report {
630 return sumReport(t.roots[version].query(whole(t.n), span))
631}
632func (t *Persistent) kth(span Span, k int64) Answer {
633 u, w := t.prefixes[span.right+1], t.prefixes[span.left]
634 position, visited := whole(len(t.values)), 1
635 for !position.leaf() {
636 left, right := position.split()
637 count := u.left.value - w.left.value
638 if k <= count {
639 u, w, position = u.left, w.left, left
640 } else {
641 k -= count
642 u, w, position = u.right, w.right, right
643 }
644 visited++
645 }
646 return Answer{t.values[position.left], visited}
647}

649type Grid struct {
650 height, width int
651 tree [][]int64
652}
653
654func newGrid(height, width int, values []int64) *Grid {
655 grid := &Grid{height, width, make([][]int64, height+1)}
656 for i := range grid.tree {
657 grid.tree[i] = make([]int64, width+1)
658 }
659 for i, value := range values {
660 grid.add(i/width, i%width, value)
661 }
662 return grid
663}
664func (g *Grid) cell(row, col int) error {
665 if row < 0 || row >= g.height || col < 0 || col >= g.width {
666 return fmt.Errorf("BAD CELL %d %d", row, col)
667 }
668 return nil
669}
670func (g *Grid) add(row, col int, value int64) int {
671 visited := 0
672 for y := row + 1; y <= g.height; y += y & -y {
673 for x := col + 1; x <= g.width; x += x & -x {
674 g.tree[y][x] += value
675 visited++
676 }
677 }
678 return visited
679}
680func (g *Grid) prefix(row, col int) Answer {
681 var answer Answer
682 for y := row; y > 0; y -= y & -y {
683 for x := col; x > 0; x -= x & -x {
684 answer.value += g.tree[y][x]
685 answer.visited++
686 }
687 }
688 return answer
689}
690func (g *Grid) query(row, col, bottom, right int) Answer {
691 a := g.prefix(bottom+1, right+1)
692 b := g.prefix(row, right+1)
693 c := g.prefix(bottom+1, col)
694 d := g.prefix(row, col)
695 return Answer{a.value - b.value - c.value + d.value, a.visited + b.visited + c.visited + d.visited}
696}
697func (g *Grid) dump() []string {
698 rows := []string{fmt.Sprintf("GRID DUMP rows=%d cols=%d", g.height, g.width)}
699 for row := 1; row <= g.height; row++ {
700 rows = append(rows, fmt.Sprintf("ROW %d %s", row, joined(g.tree[row][1:])))
701 }
702 return rows
703}

747 if err := l.index(span.left); err != nil {
748 return err
749 }
750 if err := l.index(span.right); err != nil {
751 return err

753 if span.left > span.right {
754 return fmt.Errorf("BAD RANGE %d %d", span.left, span.right)
755 }
756 return nil

764func (l *Lab) build(kind Kind, op Operation, block int) string {
765 n := len(l.values)
766 switch kind {
767 case SEG:
768 l.seg = newSegment(l.values, op)
769 return fmt.Sprintf("SEG BUILD %s nodes=%d maxnode=%d root=%d", op, 2*n-1, l.seg.maxNode, l.seg.tree[1])
770 case LAZY:
771 l.lazy = newLazy(l.values)
772 return fmt.Sprintf("LAZY BUILD nodes=%d %s", 2*n-1, l.lazy.nodes[1].value.text())
773 case FEN:
774 l.fen = &Fenwick{slices.Clone(l.values), newBit(l.values)}
775 return fmt.Sprintf("FEN BUILD n=%d total=%d", n, total(l.values))
776 case RFEN:
777 l.rfen = newRangeFenwick(l.values)
778 return fmt.Sprintf("RFEN BUILD n=%d total=%d", n, total(l.values))
779 case SPARSE:
780 l.sparse = newSparse(l.values)
781 return fmt.Sprintf("SPARSE BUILD levels=%d entries=%d", len(l.sparse.minimum), l.sparse.entries)
782 case SQRT:
783 l.sqrt = newSqrt(l.values, block)
784 return fmt.Sprintf("SQRT BUILD block=%d blocks=%d", block, len(l.sqrt.tags))
785 case PERSIST:
786 l.persist = newPersistent(l.values)
787 return fmt.Sprintf("PERSIST BUILD nodes=%d distinct=%d valuenodes=%d", l.persist.positionNodes, len(l.persist.values), l.persist.valueNodes)
788 }
789 panic("unreachable array build")
790}

829func (l *Lab) check(span Span) []string {
830 aggregate := Aggregate{0, l.values[span.left], l.values[span.left]}
831 var common int64
832 for _, value := range l.values[span.left : span.right+1] {
833 aggregate.sum += value
834 aggregate.min = min(aggregate.min, value)
835 aggregate.max = max(aggregate.max, value)
836 common = gcd(common, value)
837 }
838 rows := []string{fmt.Sprintf("CHECK %d %d %s gcd=%d", span.left, span.right, aggregate.text(), common)}
839 ok, diff := 0, 0
840 for kind := SEG; kind < GRID; kind++ {
841 if !l.has(kind) {
842 continue
843 }
844 expected := []int64{aggregate.sum}
845 switch kind {
846 case SEG:
847 expected[0] = []int64{aggregate.sum, aggregate.min, aggregate.max, common}[l.seg.op]
848 case LAZY:
849 expected = []int64{aggregate.sum, aggregate.min, aggregate.max}
850 case SPARSE:
851 expected = []int64{aggregate.min, aggregate.max}
852 }
853 version := 0
854 if kind == PERSIST {
855 version = len(l.persist.roots) - 1
856 }
857 answer := l.query(kind, version, span)
858 verdict := "DIFF"
859 if slices.Equal(answer.values, expected) {
860 verdict = "OK"
861 ok++
862 } else {
863 diff++
864 }
865 prefix := ""
866 if kind == PERSIST {
867 prefix = fmt.Sprintf("version=%d ", version)
868 }
869 rows = append(rows, fmt.Sprintf("%s %s %s%s", kind, verdict, prefix, answer.text))
870 }
871 return append(rows, fmt.Sprintf("RESULT ok=%d diff=%d", ok, diff))
872}
873func (l *Lab) mo(queries []Span) []string {
874 blockSize := isqrt(len(l.values))
875 order := make([]int, len(queries))
876 for i := range order {
877 order[i] = i
878 }
879 slices.SortFunc(order, func(a, b int) int {
880 first, second := queries[a], queries[b]
881 block := first.left / blockSize
882 if compared := cmp.Compare(block, second.left/blockSize); compared != 0 {
883 return compared
884 }
885 compared := cmp.Compare(first.right, second.right)
886 if compared != 0 {
887 if block%2 == 1 {
888 return -compared
889 }
890 return compared
891 }
892 return cmp.Compare(a, b)
893 })
894 unique := slices.Clone(l.values)
895 slices.Sort(unique)
896 unique = slices.Compact(unique)
897 ranks := make([]int, len(l.values))
898 for i, value := range l.values {
899 ranks[i], _ = slices.BinarySearch(unique, value)
900 }
901 frequencies, answers := make([]int, len(unique)), make([]int, len(queries))
902 left, right, moves, distinct := 0, 0, 0, 0
903 for _, index := range order {
904 span := queries[index]
905 end := span.right + 1
906 moves += max(left, span.left) - min(left, span.left) + max(right, end) - min(right, end)
907 for left > span.left {
908 left--
909 rank := ranks[left]
910 if frequencies[rank] == 0 {
911 distinct++
912 }
913 frequencies[rank]++
914 }
915 for right < end {
916 rank := ranks[right]
917 if frequencies[rank] == 0 {
918 distinct++
919 }
920 frequencies[rank]++
921 right++
922 }
923 for left < span.left {
924 rank := ranks[left]
925 frequencies[rank]--
926 if frequencies[rank] == 0 {
927 distinct--
928 }
929 left++
930 }
931 for right > end {
932 right--
933 rank := ranks[right]
934 frequencies[rank]--
935 if frequencies[rank] == 0 {
936 distinct--
937 }
938 }
939 answers[index] = distinct
940 }
941 numbers := make([]int, len(order))
942 for i, index := range order {
943 numbers[i] = index + 1
944 }
945 rows := []string{fmt.Sprintf("MO queries=%d block=%d moves=%d", len(queries), blockSize, moves), "ORDER " + joined(numbers)}
946 for i, span := range queries {
947 rows = append(rows, fmt.Sprintf("ANSWER %d %d %d distinct=%d", i+1, span.left, span.right, answers[i]))
948 }
949 return rows
950}
951func (l *Lab) execute(line string) ([]string, error) {
952 words := strings.Fields(line)
953 number := func(index int) int64 { value, _ := strconv.ParseInt(words[index], 10, 64); return value }
954 index := func(i int) int { return int(number(i)) }
955 switch words[0] {
956 case "LOAD":
957 removed := 0
958 for kind := SEG; kind < GRID; kind++ {
959 if l.has(kind) {
960 removed++
961 }
962 }
963 values := make([]int64, len(words)-1)
964 for i := range values {
965 values[i] = number(i + 1)
966 }
967 grid := l.grid
968 *l = *newLab(values)
969 l.grid = grid
970 return []string{fmt.Sprintf("LOAD n=%d removed=%d", len(values), removed)}, nil
971 case "RESET":
972 l.values = slices.Clone(l.original)
973 op, block, rebuilt := SUM, 0, 0
974 if l.seg != nil {
975 op = l.seg.op
976 }
977 if l.sqrt != nil {
978 block = l.sqrt.block
979 }
980 for kind := SEG; kind < GRID; kind++ {
981 if l.has(kind) {
982 l.build(kind, op, block)
983 rebuilt++
984 }
985 }
986 return []string{fmt.Sprintf("RESET rebuilt=%d", rebuilt)}, nil
987 case "ARRAY", "CHECK":
988 span := Span{index(1), index(2)}
989 if err := l.checkRange(span); err != nil {
990 return nil, err
991 }
992 if words[0] == "CHECK" {
993 return l.check(span), nil
994 }
995 return []string{fmt.Sprintf("ARRAY %d %d %s", span.left, span.right, joined(l.values[span.left:span.right+1]))}, nil
996 case "MO":
997 queries := make([]Span, 0, (len(words)-1)/2)
998 for i := 1; i < len(words); i += 2 {
999 span := Span{index(i), index(i + 1)}
1000 if err := l.checkRange(span); err != nil {
1001 return nil, err
1002 }
1003 queries = append(queries, span)
1004 }
1005 return l.mo(queries), nil
1006 }
1007 kind := Kind(slices.Index(kindNames[:], words[0]))
1008 action := words[1]
1009 if action == "BUILD" {
1010 op, block := SUM, 0
1011 if kind == SEG {
1012 found := slices.Index(opNames[:], words[2])
1013 if found < 0 {
1014 return nil, fmt.Errorf("BAD OP %s", words[2])
1015 }
1016 op = Operation(found)
1017 }
1018 if kind == SQRT {
1019 block = isqrt(len(l.values))
1020 if len(words) == 3 {
1021 block = index(2)
1022 }
1023 if block > len(l.values) {
1024 return nil, fmt.Errorf("BAD BLOCK %d", block)
1025 }
1026 }
1027 if kind == GRID {
1028 height, width := index(2), index(3)
1029 if len(words)-4 != height*width {
1030 return nil, fmt.Errorf("BAD COUNT %d %d", len(words)-4, height*width)
1031 }
1032 values := make([]int64, len(words)-4)
1033 for i := range values {
1034 values[i] = number(i + 4)
1035 }
1036 l.grid = newGrid(height, width, values)
1037 return []string{fmt.Sprintf("GRID BUILD rows=%d cols=%d total=%d", height, width, total(values))}, nil
1038 }
1039 return []string{l.build(kind, op, block)}, nil
1040 }
1041 if !l.has(kind) {
1042 return nil, fmt.Errorf("NOT BUILT %s", kind)
1043 }
1044 if action == "DUMP" {
1045 return l.dump(kind), nil
1046 }
1047 if kind == GRID {
1048 row, col := index(2), index(3)
1049 if err := l.grid.cell(row, col); err != nil {
1050 return nil, err
1051 }
1052 if action == "ADD" {
1053 value := number(4)
1054 visited := l.grid.add(row, col, value)
1055 return []string{fmt.Sprintf("GRID ADD %d %d %d visited=%d", row, col, value, visited)}, nil
1056 }
1057 bottom, right := index(4), index(5)
1058 if err := l.grid.cell(bottom, right); err != nil {
1059 return nil, err
1060 }
1061 if row > bottom || col > right {
1062 return nil, fmt.Errorf("BAD RECT %d %d %d %d", row, col, bottom, right)
1063 }
1064 answer := l.grid.query(row, col, bottom, right)
1065 return []string{fmt.Sprintf("GRID SUM %d %d %d %d sum=%d visited=%d", row, col, bottom, right, answer.value, answer.visited)}, nil
1066 }
1067 if kind == PERSIST {
1068 if action == "NODES" {
1069 return []string{fmt.Sprintf("PERSIST NODES versions=%d position=%d value=%d", len(l.persist.roots), l.persist.positionNodes, l.persist.valueNodes)}, nil
1070 }
1071 if action == "KTH" {
1072 span, k := Span{index(2), index(3)}, number(4)
1073 if err := l.checkRange(span); err != nil {
1074 return nil, err
1075 }
1076 if k < 1 || k > int64(span.length()) {
1077 return nil, fmt.Errorf("BAD K %d", k)
1078 }
1079 answer := l.persist.kth(span, k)
1080 return []string{fmt.Sprintf("PERSIST KTH %d %d %d value=%d visited=%d", span.left, span.right, k, answer.value, answer.visited)}, nil
1081 }
1082 version := index(2)
1083 if err := l.version(version); err != nil {
1084 return nil, err
1085 }
1086 position := index(3)
1087 if action == "SET" {
1088 if err := l.index(position); err != nil {
1089 return nil, err
1090 }
1091 value := number(4)
1092 next, created := l.persist.set(version, position, value)
1093 return []string{fmt.Sprintf("PERSIST SET %d %d %d version=%d created=%d", version, position, value, next, created)}, nil
1094 }
1095 span := Span{position, index(4)}
1096 if err := l.checkRange(span); err != nil {
1097 return nil, err
1098 }
1099 return []string{fmt.Sprintf("PERSIST QUERY %d %d %d %s", version, span.left, span.right, l.persist.query(version, span).text)}, nil
1100 }
1101 if kind == FEN && action == "LOWER" {
1102 for i, value := range l.fen.values {
1103 if value < 0 {
1104 return nil, fmt.Errorf("NEGATIVE %d", i)
1105 }
1106 }
1107 k := number(2)
1108 position, visited := l.fen.bit.lower(k)
1109 answer := "none"
1110 if position >= 0 {
1111 answer = strconv.Itoa(position)
1112 }
1113 return []string{fmt.Sprintf("FEN LOWER %d index=%s visited=%d", k, answer, visited)}, nil
1114 }
1115 if kind == SEG && action == "UPDATE" {
1116 position := index(2)
1117 if err := l.index(position); err != nil {
1118 return nil, err
1119 }
1120 value := number(3)
1121 visited := l.seg.updateAt(1, whole(len(l.values)), position, value)
1122 l.values[position] = value
1123 return []string{fmt.Sprintf("SEG UPDATE %d %d root=%d visited=%d", position, value, l.seg.tree[1], visited)}, nil
1124 }
1125 if kind == FEN && (action == "ADD" || action == "PREFIX") {
1126 position := index(2)
1127 if err := l.index(position); err != nil {
1128 return nil, err
1129 }
1130 if action == "ADD" {
1131 value := number(3)
1132 l.values[position] += value
1133 l.fen.values[position] += value
1134 visited := l.fen.bit.add(position+1, value)
1135 return []string{fmt.Sprintf("FEN ADD %d %d visited=%d", position, value, visited)}, nil
1136 }
1137 answer := l.fen.bit.prefix(position + 1)
1138 return []string{fmt.Sprintf("FEN PREFIX %d sum=%d visited=%d", position, answer.value, answer.visited)}, nil
1139 }
1140 span := Span{index(2), index(3)}
1141 if err := l.checkRange(span); err != nil {
1142 return nil, err
1143 }
1144 if action == "ADD" || action == "ASSIGN" {
1145 value, assign := number(4), action == "ASSIGN"
1146 var details string
1147 switch kind {
1148 case LAZY:
1149 counts := l.lazy.update(span, value, assign)
1150 details = fmt.Sprintf("visited=%d covered=%d pushes=%d", counts.visited, counts.covered, counts.pushes)
1151 case RFEN:
1152 details = fmt.Sprintf("visited=%d", l.rfen.add(span, value))
1153 case SQRT:
1154 whole, cells := l.sqrt.add(span, value)
1155 details = fmt.Sprintf("whole=%d cells=%d", whole, cells)
1156 }
1157 for i := span.left; i <= span.right; i++ {
1158 if assign {
1159 l.values[i] = value
1160 } else {
1161 l.values[i] += value
1162 }
1163 }
1164 return []string{fmt.Sprintf("%s %s %d %d %d %s", kind, action, span.left, span.right, value, details)}, nil
1165 }
1166 answer := l.query(kind, 0, span)
1167 return []string{fmt.Sprintf("%s %s %d %d %s", kind, action, span.left, span.right, answer.text)}, nil
1168}
