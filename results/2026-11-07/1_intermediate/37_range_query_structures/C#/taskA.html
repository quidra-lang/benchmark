10static class Arithmetic
11{
12 public static long Gcd(long a, long b)
13 {
14 a = Math.Abs(a);
15 b = Math.Abs(b);
16 while (b != 0) (a, b) = (b, a % b);
17 return a;
18 }
19
20 public static int IntegerSqrt(int n)
21 {
22 int result = 1;
23 while ((result + 1) * (result + 1) <= n) result++;
24 return result;
25 }
26
27 public static int Log2(int n)
28 {
29 int result = 0;
30 while ((1 << (result + 1)) <= n) result++;
31 return result;
32 }
33
34 public static long Combine(Operation op, long a, long b) => op switch
35 {
36 Operation.SUM => a + b,
37 Operation.MIN => Math.Min(a, b),
38 Operation.MAX => Math.Max(a, b),
39 _ => Gcd(a, b)
40 };
41}

43readonly record struct Interval(int Left, int Right)
44{
45 public bool Leaf => Left == Right;
46 public int Middle => (Left + Right) / 2;
47 public int Length => Right - Left + 1;
48 public bool Covered(int left, int right) => left <= Left && Right <= right;
49}

51sealed class Layout
52{
53 readonly Interval?[] ranges;
54 public int Count { get; }
55 public int MaxNode { get; private set; }
56 public Interval this[int node] => ranges[node]!.Value;
57
58 public Layout(int n)
59 {
60 ranges = new Interval?[4 * n];
61 Count = 2 * n - 1;
62 Build(1, 0, n - 1);
63 }
64
65 void Build(int node, int left, int right)
66 {
67 ranges[node] = new(left, right);
68 MaxNode = Math.Max(MaxNode, node);
69 if (left == right) return;
70 int middle = (left + right) / 2;
71 Build(2 * node, left, middle);
72 Build(2 * node + 1, middle + 1, right);
73 }
74
75 public IEnumerable<(int Node, Interval Range)> Nodes()
76 {
77 for (int node = 1; node <= MaxNode; node++)
78 if (ranges[node] is Interval range) yield return (node, range);
79 }
80}

87sealed class SegmentTree
88{
89 public Operation Op { get; }
90 public Layout Shape { get; }
91 readonly long[] values;
92 public long Root => values[1];
93 public string Name => Op.ToString().ToLowerInvariant();
94
95 public SegmentTree(long[] contents, Operation op)
96 {
97 Op = op;
98 Shape = new(contents.Length);
99 values = new long[4 * contents.Length];
100 Build(1, contents);
101 }
102
103 void Build(int node, long[] contents)
104 {
105 Interval range = Shape[node];
106 if (range.Leaf) values[node] = Op == Operation.GCD ? Math.Abs(contents[range.Left]) : contents[range.Left];
107 else
108 {
109 Build(2 * node, contents);
110 Build(2 * node + 1, contents);
111 Pull(node);
112 }
113 }
114
115 void Pull(int node) => values[node] = Arithmetic.Combine(Op, values[2 * node], values[2 * node + 1]);
116
117 public SumAnswer Query(int left, int right, int node = 1)
118 {
119 Interval range = Shape[node];
120 if (range.Covered(left, right)) return new(values[node], 1);
121 if (right <= range.Middle)
122 {
123 SumAnswer child = Query(left, right, 2 * node);
124 return child with { Visited = child.Visited + 1 };
125 }
126 if (left > range.Middle)
127 {
128 SumAnswer child = Query(left, right, 2 * node + 1);
129 return child with { Visited = child.Visited + 1 };
130 }
131 SumAnswer a = Query(left, right, 2 * node);
132 SumAnswer b = Query(left, right, 2 * node + 1);
133 return new(Arithmetic.Combine(Op, a.Sum, b.Sum), a.Visited + b.Visited + 1);
134 }
135
136 public int Update(int index, long value, int node = 1)
137 {
138 Interval range = Shape[node];
139 if (range.Leaf)
140 {
141 values[node] = Op == Operation.GCD ? Math.Abs(value) : value;
142 return 1;
143 }
144 int visited = Update(index, value, index <= range.Middle ? 2 * node : 2 * node + 1);
145 Pull(node);
146 return visited + 1;
147 }
148
149 public string Describe(SumAnswer answer) => $"{Name}={answer.Sum} visited={answer.Visited}";
150 public IEnumerable<string> Dump()
151 {
152 yield return $"SEG DUMP {Op} nodes={Shape.Count}";
153 foreach (var (node, range) in Shape.Nodes())
154 yield return $"NODE {node} {range.Left} {range.Right} {values[node]}";
155 }
156}

175sealed class LazyTree
176{
177 public Layout Shape { get; }
178 readonly Aggregate[] values;
179 readonly long?[] assign;
180 readonly long[] add;
181 public Aggregate Root => values[1];
182
183 public LazyTree(long[] contents)
184 {
185 Shape = new(contents.Length);
186 values = new Aggregate[4 * contents.Length];
187 assign = new long?[values.Length];
188 add = new long[values.Length];
189 Build(1, contents);
190 }
191
192 void Build(int node, long[] contents)
193 {
194 Interval range = Shape[node];
195 if (range.Leaf) values[node] = new(contents[range.Left], contents[range.Left], contents[range.Left]);
196 else
197 {
198 Build(2 * node, contents);
199 Build(2 * node + 1, contents);
200 Pull(node);
201 }
202 }
203
204 void Pull(int node) => values[node] = Aggregate.Merge(values[2 * node], values[2 * node + 1]);
205
206 void Apply(int node, long value, bool assignment)
207 {
208 Interval range = Shape[node];
209 Aggregate old = values[node];
210 values[node] = assignment ? new(value * range.Length, value, value)
211 : new(old.Sum + value * range.Length, old.Min + value, old.Max + value);
212 if (range.Leaf) return;
213 if (assignment) { assign[node] = value; add[node] = 0; }
214 else add[node] += value;
215 }
216
217 int Push(int node)
218 {
219 if (assign[node] is null && add[node] == 0) return 0;
220 for (int child = 2 * node; child <= 2 * node + 1; child++)
221 {
222 if (assign[node] is long value) Apply(child, value, true);
223 if (add[node] != 0) Apply(child, add[node], false);
224 }
225 assign[node] = null;
226 add[node] = 0;
227 return 1;
228 }
229
230 public Work Update(int left, int right, long value, bool assignment)
231 {
232 Work work = new();
233 void Visit(int node)
234 {
235 work.Visited++;
236 Interval range = Shape[node];
237 if (range.Covered(left, right))
238 {
239 work.Covered++;
240 Apply(node, value, assignment);
241 return;
242 }
243 work.Pushes += Push(node);
244 if (left <= range.Middle) Visit(2 * node);
245 if (right > range.Middle) Visit(2 * node + 1);
246 Pull(node);
247 }
248 Visit(1);
249 return work;
250 }
251
252 public LazyAnswer Query(int left, int right, int node = 1)
253 {
254 Interval range = Shape[node];
255 if (range.Covered(left, right)) return new(values[node], 1, 0);
256 int pushes = Push(node);
257 if (right <= range.Middle || left > range.Middle)
258 {
259 LazyAnswer child = Query(left, right, right <= range.Middle ? 2 * node : 2 * node + 1);
260 return new(child.Value, child.Visited + 1, child.Pushes + pushes);
261 }
262 LazyAnswer a = Query(left, right, 2 * node);
263 LazyAnswer b = Query(left, right, 2 * node + 1);
264 return new(Aggregate.Merge(a.Value, b.Value), a.Visited + b.Visited + 1, a.Pushes + b.Pushes + pushes);
265 }
266
267 public IEnumerable<string> Dump()
268 {
269 yield return $"LAZY DUMP nodes={Shape.Count}";
270 foreach (var (node, range) in Shape.Nodes())
271 yield return $"NODE {node} {range.Left} {range.Right} {values[node].Text} assign={assign[node]?.ToString() ?? "-"} add={add[node]}";
272 }
273}

299sealed class PointFenwick
300{
301 public long[] Contents { get; }
302 public Fenwick Bits { get; }
303 public PointFenwick(long[] contents)
304 {
305 Contents = (long[])contents.Clone();
306 Bits = new(contents.Length);
307 for (int i = 0; i < contents.Length; i++) Bits.Add(i + 1, contents[i]);
308 }
309
310 public int Add(int index, long value)
311 {
312 Contents[index] += value;
313 return Bits.Add(index + 1, value);
314 }
315
316 public SumAnswer Query(int left, int right)
317 {
318 SumAnswer a = Bits.Prefix(right + 1), b = Bits.Prefix(left);
319 return new(a.Sum - b.Sum, a.Visited + b.Visited);
320 }
321
322 public (int? Index, int Visited) Lower(long rest)
323 {
324 int position = 0, visited = 0;
325 for (int step = 1 << Arithmetic.Log2(Contents.Length); step >= 1; step /= 2)
326 {
327 if (position + step > Contents.Length) continue;
328 visited++;
329 long value = Bits.T[position + step];
330 if (value < rest) { rest -= value; position += step; }
331 }
332 return (position == Contents.Length ? null : position, visited);
333 }

336sealed class RangeFenwick
337{
338 public Fenwick B1 { get; }
339 public Fenwick B2 { get; }
340 public RangeFenwick(long[] contents)
341 {
342 B1 = new(contents.Length);
343 B2 = new(contents.Length);
344 for (int i = 0; i < contents.Length; i++)
345 {
346 long difference = contents[i] - (i == 0 ? 0 : contents[i - 1]);
347 B1.Add(i + 1, difference);
348 B2.Add(i + 1, i * difference);
349 }
350 }
351
352 public int Add(int left, int right, long value)
353 {
354 int visited = B1.Add(left + 1, value);
355 visited += B1.Add(right + 2, -value);
356 visited += B2.Add(left + 1, value * left);
357 visited += B2.Add(right + 2, -value * (right + 1));
358 return visited;
359 }
360
361 SumAnswer Prefix(int position)
362 {
363 SumAnswer a = B1.Prefix(position), b = B2.Prefix(position);
364 return new(position * a.Sum - b.Sum, a.Visited + b.Visited);
365 }
366
367 public SumAnswer Query(int left, int right)
368 {
369 SumAnswer a = Prefix(right + 1), b = Prefix(left);
370 return new(a.Sum - b.Sum, a.Visited + b.Visited);
371 }

379sealed class SparseTable
380{
381 readonly long[] contents;
382 public List<int[]> Minimum { get; } = [];
383 public List<int[]> Maximum { get; } = [];
384 public int Entries => Minimum.Sum(row => row.Length);
385
386 public SparseTable(long[] values)
387 {
388 contents = (long[])values.Clone();
389 Minimum.Add(Enumerable.Range(0, values.Length).ToArray());
390 Maximum.Add(Enumerable.Range(0, values.Length).ToArray());
391 for (int level = 1; (1 << level) <= values.Length; level++)
392 {
393 int length = values.Length - (1 << level) + 1, half = 1 << (level - 1);
394 int[] minima = new int[length], maxima = new int[length];
395 for (int i = 0; i < length; i++)
396 {
397 minima[i] = Select(Minimum[level - 1][i], Minimum[level - 1][i + half], true);
398 maxima[i] = Select(Maximum[level - 1][i], Maximum[level - 1][i + half], false);
399 }
400 Minimum.Add(minima);
401 Maximum.Add(maxima);
402 }
403 }
404
405 int Select(int a, int b, bool minimum) => (minimum ? contents[a] <= contents[b] : contents[a] >= contents[b]) ? a : b;
406
407 public SparseAnswer Query(int left, int right)
408 {
409 int level = Arithmetic.Log2(right - left + 1), end = right - (1 << level) + 1;
410 int a = Select(Minimum[level][left], Minimum[level][end], true);
411 int z = Select(Maximum[level][left], Maximum[level][end], false);
412 return new(contents[a], a, contents[z], z, level);
413 }
414
415 public IEnumerable<string> Dump()
416 {
417 yield return $"SPARSE DUMP levels={Minimum.Count}";
418 for (int level = 0; level < Minimum.Count; level++)
419 {
420 yield return $"MIN {level} " + string.Join(" ", Minimum[level]);
421 yield return $"MAX {level} " + string.Join(" ", Maximum[level]);
422 }
423 }

436sealed class SqrtBlocks
437{
438 readonly long[] raw, tag, sums;
439 public int BlockSize { get; }
440 public int Count => tag.Length;
441
442 public SqrtBlocks(long[] contents, int blockSize)
443 {
444 BlockSize = blockSize;
445 raw = (long[])contents.Clone();
446 tag = new long[(contents.Length + blockSize - 1) / blockSize];
447 sums = new long[tag.Length];
448 for (int i = 0; i < raw.Length; i++) sums[i / blockSize] += raw[i];
449 }
450
451 Split Partition(int left, int right)
452 {
453 List<int> cells = [], whole = [];
454 int first = left / BlockSize, last = right / BlockSize;
455 if (first == last)
456 {
457 for (int i = left; i <= right; i++) cells.Add(i);
458 }
459 else
460 {
461 for (int i = left; i < (first + 1) * BlockSize; i++) cells.Add(i);
462 for (int i = last * BlockSize; i <= right; i++) cells.Add(i);
463 for (int block = first + 1; block < last; block++) whole.Add(block);
464 }
465 return new(cells, whole);
466 }
467
468 public Split Add(int left, int right, long value)
469 {
470 Split split = Partition(left, right);
471 foreach (int i in split.Cells) { raw[i] += value; sums[i / BlockSize] += value; }
472 foreach (int block in split.Whole) { tag[block] += value; sums[block] += value * BlockSize; }
473 return split;
474 }
475
476 public SqrtAnswer Query(int left, int right)
477 {
478 Split split = Partition(left, right);
479 long sum = 0;
480 foreach (int i in split.Cells) sum += raw[i] + tag[i / BlockSize];
481 foreach (int block in split.Whole) sum += sums[block];
482 return new(sum, split);
483 }
484
485 public IEnumerable<string> Dump()
486 {
487 yield return $"SQRT DUMP block={BlockSize} blocks={Count}";
488 for (int block = 0; block < Count; block++)
489 {
490 int first = block * BlockSize, end = Math.Min((block + 1) * BlockSize, raw.Length);
491 yield return $"BLOCK {block} {first} {end - 1} add={tag[block]} sum={sums[block]} " + string.Join(" ", raw[first..end]);
492 }
493 }
494}

498sealed class PersistentTree
499{
500 readonly int n;
501 readonly long[] values;
502 readonly List<PersistentNode> valueRoots = [];
503 public List<PersistentNode> Versions { get; } = [];
504 public int PositionNodes { get; private set; }
505 public int ValueNodes { get; private set; }
506 public int Distinct => values.Length;
507
508 public PersistentTree(long[] contents)
509 {
510 n = contents.Length;
511 Versions.Add(Build(contents, 0, n - 1));
512 PositionNodes = 2 * n - 1;
513 values = contents.Distinct().Order().ToArray();
514 valueRoots.Add(Build(new long[values.Length], 0, values.Length - 1));
515 ValueNodes = 2 * values.Length - 1;
516 foreach (long value in contents)
517 {
518 int created = 0;
519 int rank = Array.BinarySearch(values, value);
520 valueRoots.Add(Change(valueRoots[^1], 0, values.Length - 1, rank, 1, true, ref created));
521 ValueNodes += created;
522 }
523 }
524
525 static PersistentNode Build(long[] contents, int left, int right)
526 {
527 if (left == right) return new(contents[left]);
528 int middle = (left + right) / 2;
529 PersistentNode a = Build(contents, left, middle);
530 PersistentNode b = Build(contents, middle + 1, right);
531 return new(a.Value + b.Value, a, b);
532 }
533
534 static PersistentNode Change(PersistentNode node, int left, int right, int index, long value, bool increment, ref int created)
535 {
536 created++;
537 if (left == right) return new(increment ? node.Value + value : value);
538 int middle = (left + right) / 2;
539 PersistentNode a = node.Left!, b = node.Right!;
540 if (index <= middle) a = Change(a, left, middle, index, value, increment, ref created);
541 else b = Change(b, middle + 1, right, index, value, increment, ref created);
542 return new(a.Value + b.Value, a, b);
543 }
544
545 public int Set(int version, int index, long value)
546 {
547 int created = 0;
548 Versions.Add(Change(Versions[version], 0, n - 1, index, value, false, ref created));
549 PositionNodes += created;
550 return created;
551 }
552
553 static SumAnswer Sum(PersistentNode node, int nl, int nr, int left, int right)
554 {
555 if (left <= nl && nr <= right) return new(node.Value, 1);
556 int middle = (nl + nr) / 2;
557 if (right <= middle)
558 {
559 SumAnswer answer = Sum(node.Left!, nl, middle, left, right);
560 return answer with { Visited = answer.Visited + 1 };
561 }
562 if (left > middle)
563 {
564 SumAnswer answer = Sum(node.Right!, middle + 1, nr, left, right);
565 return answer with { Visited = answer.Visited + 1 };
566 }
567 SumAnswer a = Sum(node.Left!, nl, middle, left, right);
568 SumAnswer b = Sum(node.Right!, middle + 1, nr, left, right);
569 return new(a.Sum + b.Sum, a.Visited + b.Visited + 1);
570 }
571
572 public SumAnswer Query(int version, int left, int right) => Sum(Versions[version], 0, n - 1, left, right);
573
574 public SumAnswer Kth(int left, int right, int k)
575 {
576 PersistentNode u = valueRoots[right + 1], w = valueRoots[left];
577 int p = 0, q = values.Length - 1, visited = 1;
578 while (p < q)
579 {
580 int middle = (p + q) / 2;
581 long difference = u.Left!.Value - w.Left!.Value;
582 if (k <= difference) { u = u.Left; w = w.Left; q = middle; }
583 else { k -= (int)difference; u = u.Right!; w = w.Right!; p = middle + 1; }
584 visited++;
585 }
586 return new(values[p], visited);
587 }
588}

609 return visited;

613 {
614 long sum = 0;
615 int visited = 0;
616 for (int y = row; y > 0; y -= y & -y)
617 for (int z = column; z > 0; z -= z & -z) { sum += tree[y, z]; visited++; }
618 return new(sum, visited);
619 }

703 SumAnswer answer = segment.Query(left, right);
704 long expected = segment.Op switch { Operation.SUM => sum, Operation.MIN => minimum, Operation.MAX => maximum, _ => gcd };
705 Report("SEG", answer.Sum == expected, segment.Describe(answer));
706 }
707 if (lazy is not null)
708 {
709 LazyAnswer answer = lazy.Query(left, right);
710 Report("LAZY", answer.Value == new Aggregate(sum, minimum, maximum), answer.Text);
711 }
712 if (fenwick is not null)
713 {
714 SumAnswer answer = fenwick.Query(left, right);
715 Report("FEN", answer.Sum == sum, answer.Text);
716 }
717 if (rangeFenwick is not null)
718 {
719 SumAnswer answer = rangeFenwick.Query(left, right);
720 Report("RFEN", answer.Sum == sum, answer.Text);
721 }
722 if (sparse is not null)
723 {
724 SparseAnswer answer = sparse.Query(left, right);
725 Report("SPARSE", answer.Min == minimum && answer.Max == maximum, answer.Text);
726 }
727 if (sqrt is not null)
728 {
729 SqrtAnswer answer = sqrt.Query(left, right);
730 Report("SQRT", answer.Sum == sum, answer.Text);
731 }
732 if (persistent is not null)
733 {
734 int version = persistent.Versions.Count - 1;
735 SumAnswer answer = persistent.Query(version, left, right);
736 Report("PERSIST", answer.Sum == sum, $"version={version} {answer.Text}");
737 }
738 rows.Add($"RESULT ok={ok} diff={diff}");
739 return rows;
740 }

742 List<string> Mo(string[] words)
743 {
744 List<MoQuery> queries = [];
745 for (int i = 1; i < words.Length; i += 2)
746 {
747 int left = int.Parse(words[i], CultureInfo.InvariantCulture), right = int.Parse(words[i + 1], CultureInfo.InvariantCulture);
748 CheckRange(left, right);
749 queries.Add(new(left, right, queries.Count + 1));
750 }
751 int blockSize = Arithmetic.IntegerSqrt(array.Length);
752 MoQuery[] order = queries.OrderBy(q => q.Left / blockSize)
753 .ThenBy(q => q.Left / blockSize % 2 == 0 ? q.Right : -q.Right).ThenBy(q => q.Number).ToArray();
754 Dictionary<long, int> frequency = new();
755 int[] answers = new int[queries.Count];
756 int leftEdge = 0, rightEdge = -1, moves = 0;
757 void Add(int i) { frequency.TryGetValue(array[i], out int old); frequency[array[i]] = old + 1; moves++; }
758 void Remove(int i)
759 {
760 int count = frequency[array[i]] - 1;
761 if (count == 0) frequency.Remove(array[i]); else frequency[array[i]] = count;
762 moves++;
763 }
764 foreach (MoQuery query in order)
765 {
766 while (leftEdge > query.Left) Add(--leftEdge);
767 while (rightEdge < query.Right) Add(++rightEdge);
768 while (leftEdge < query.Left) Remove(leftEdge++);
769 while (rightEdge > query.Right) Remove(rightEdge--);
770 answers[query.Number - 1] = frequency.Count;
771 }
772 List<string> rows = [$"MO queries={queries.Count} block={blockSize} moves={moves}", "ORDER " + string.Join(" ", order.Select(q => q.Number))];
773 foreach (MoQuery query in queries) rows.Add($"ANSWER {query.Number} {query.Left} {query.Right} distinct={answers[query.Number - 1]}");
774 return rows;
775 }

777 public List<string> Execute(string line)
778 {
779 string[] words = line.Split(' ');
780 long Number(int index) => long.Parse(words[index], CultureInfo.InvariantCulture);
781 int Index(int index) => (int)Number(index);
782 string kind = words[0];
783 if (kind == "ARRAY")
784 {
785 int left = Index(1), right = Index(2);
786 CheckRange(left, right);
787 return [$"ARRAY {left} {right} " + string.Join(" ", array[left..(right + 1)])];
788 }
789 if (kind == "LOAD")
790 {
791 int removed = StructureCount;
792 original = words.Skip(1).Select(value => long.Parse(value, CultureInfo.InvariantCulture)).ToArray();
793 array = (long[])original.Clone();
794 segment = null; lazy = null; fenwick = null; rangeFenwick = null; sparse = null; sqrt = null; persistent = null;
795 return [$"LOAD n={array.Length} removed={removed}"];
796 }
797 if (kind == "RESET")
798 {
799 int rebuilt = StructureCount;
800 array = (long[])original.Clone();
801 if (segment is not null) segment = new(array, segment.Op);
802 if (lazy is not null) lazy = new(array);
803 if (fenwick is not null) fenwick = new(array);
804 if (rangeFenwick is not null) rangeFenwick = new(array);
805 if (sparse is not null) sparse = new(array);
806 if (sqrt is not null) sqrt = new(array, sqrt.BlockSize);
807 if (persistent is not null) persistent = new(array);
808 return [$"RESET rebuilt={rebuilt}"];
809 }
810 if (kind == "CHECK") return Check(Index(1), Index(2));
811 if (kind == "MO") return Mo(words);
812 string action = words[1];
813 switch (kind)
814 {
815 case "SEG":
816 {
817 if (action == "BUILD")
818 {
819 if (!Enum.TryParse(words[2], out Operation operation)) throw new CommandError("BAD OP " + words[2]);
820 segment = new(array, operation);
821 return [$"SEG BUILD {operation} nodes={segment.Shape.Count} maxnode={segment.Shape.MaxNode} root={segment.Root}"];
822 }
823 SegmentTree tree = Need(segment, kind);
824 if (action == "DUMP") return tree.Dump().ToList();
825 int left = Index(2);
826 if (action == "UPDATE")
827 {
828 CheckIndex(left);
829 long value = Number(3);
830 int visited = tree.Update(left, value);
831 array[left] = value;
832 return [$"SEG UPDATE {left} {value} root={tree.Root} visited={visited}"];
833 }
834 int right = Index(3);
835 CheckRange(left, right);
836 return [$"SEG QUERY {left} {right} {tree.Describe(tree.Query(left, right))}"];
837 }
838 case "LAZY":
839 {
840 if (action == "BUILD")
841 {
842 lazy = new(array);
843 return [$"LAZY BUILD nodes={lazy.Shape.Count} {lazy.Root.Text}"];
844 }
845 LazyTree tree = Need(lazy, kind);
846 if (action == "DUMP") return tree.Dump().ToList();
847 int left = Index(2), right = Index(3);
848 CheckRange(left, right);
849 if (action == "QUERY") return [$"LAZY QUERY {left} {right} {tree.Query(left, right).Text}"];
850 long value = Number(4);
851 bool assignment = action == "ASSIGN";
852 Work work = tree.Update(left, right, value, assignment);
853 ChangeArray(left, right, value, assignment);
854 return [$"LAZY {action} {left} {right} {value} {work}"];
855 }
856 case "FEN":
857 {
858 if (action == "BUILD")
859 {
860 fenwick = new(array);
861 return [$"FEN BUILD n={array.Length} total={fenwick.Bits.Prefix(array.Length).Sum}"];
862 }
863 PointFenwick tree = Need(fenwick, kind);
864 if (action == "DUMP") return [$"FEN DUMP n={array.Length}", tree.Bits.Dump("T")];
865 if (action == "LOWER")
866 {
867 for (int i = 0; i < array.Length; i++)
868 if (tree.Contents[i] < 0) throw new CommandError($"NEGATIVE {i}");
869 long target = Number(2);
870 var (index, visited) = tree.Lower(target);
871 return [$"FEN LOWER {target} index={index?.ToString() ?? "none"} visited={visited}"];
872 }
873 int left = Index(2);
874 if (action == "RANGE")
875 {
876 int right = Index(3);
877 CheckRange(left, right);
878 return [$"FEN RANGE {left} {right} {tree.Query(left, right).Text}"];
879 }
880 CheckIndex(left);
881 if (action == "PREFIX") return [$"FEN PREFIX {left} {tree.Bits.Prefix(left + 1).Text}"];
882 long value = Number(3);
883 int visits = tree.Add(left, value);
884 array[left] += value;
885 return [$"FEN ADD {left} {value} visited={visits}"];
886 }
887 case "RFEN":
888 {
889 if (action == "BUILD")
890 {
891 rangeFenwick = new(array);
892 return [$"RFEN BUILD n={array.Length} total={rangeFenwick.Query(0, array.Length - 1).Sum}"];
893 }
894 RangeFenwick tree = Need(rangeFenwick, kind);
895 if (action == "DUMP") return [$"RFEN DUMP n={array.Length}", tree.B1.Dump("B1"), tree.B2.Dump("B2")];
896 int left = Index(2), right = Index(3);
897 CheckRange(left, right);
898 if (action == "SUM") return [$"RFEN SUM {left} {right} {tree.Query(left, right).Text}"];
899 long value = Number(4);
900 int visited = tree.Add(left, right, value);
901 ChangeArray(left, right, value);
902 return [$"RFEN ADD {left} {right} {value} visited={visited}"];
903 }
904 case "SPARSE":
905 {
906 if (action == "BUILD")
907 {
908 sparse = new(array);
909 return [$"SPARSE BUILD levels={sparse.Minimum.Count} entries={sparse.Entries}"];
910 }
911 SparseTable tree = Need(sparse, kind);
912 if (action == "DUMP") return tree.Dump().ToList();
913 int left = Index(2), right = Index(3);
914 CheckRange(left, right);
915 return [$"SPARSE QUERY {left} {right} {tree.Query(left, right).Text}"];
916 }
917 case "SQRT":
918 {
919 if (action == "BUILD")
920 {
921 int size = words.Length == 3 ? Index(2) : Arithmetic.IntegerSqrt(array.Length);
922 if (size > array.Length) throw new CommandError($"BAD BLOCK {size}");
923 sqrt = new(array, size);
924 return [$"SQRT BUILD block={size} blocks={sqrt.Count}"];
925 }
926 SqrtBlocks tree = Need(sqrt, kind);
927 if (action == "DUMP") return tree.Dump().ToList();
928 int left = Index(2), right = Index(3);
929 CheckRange(left, right);
930 if (action == "SUM") return [$"SQRT SUM {left} {right} {tree.Query(left, right).Text}"];
931 long value = Number(4);
932 Split split = tree.Add(left, right, value);
933 ChangeArray(left, right, value);
934 return [$"SQRT ADD {left} {right} {value} {split.Text}"];
935 }
936 case "PERSIST":
937 {
938 if (action == "BUILD")
939 {
940 persistent = new(array);
941 return [$"PERSIST BUILD nodes={persistent.PositionNodes} distinct={persistent.Distinct} valuenodes={persistent.ValueNodes}"];
942 }
943 PersistentTree tree = Need(persistent, kind);
944 if (action == "NODES") return [$"PERSIST NODES versions={tree.Versions.Count} position={tree.PositionNodes} value={tree.ValueNodes}"];
945 if (action == "KTH")
946 {
947 int left = Index(2), right = Index(3), k = Index(4);
948 CheckRange(left, right);
949 if (k < 1 || k > right - left + 1) throw new CommandError($"BAD K {k}");
950 SumAnswer answer = tree.Kth(left, right, k);
951 return [$"PERSIST KTH {left} {right} {k} value={answer.Sum} visited={answer.Visited}"];
952 }
953 int version = Index(2);
954 CheckVersion(tree, version);
955 int index = Index(3);
956 if (action == "SET")
957 {
958 CheckIndex(index);
959 long value = Number(4);
960 int created = tree.Set(version, index, value);
961 return [$"PERSIST SET {version} {index} {value} version={tree.Versions.Count - 1} created={created}"];
962 }
963 int end = Index(4);
964 CheckRange(index, end);
965 return [$"PERSIST QUERY {version} {index} {end} {tree.Query(version, index, end).Text}"];
966 }
967 case "GRID":
968 {
969 if (action == "BUILD")
970 {
971 int height = Index(2), width = Index(3), count = words.Length - 4;
972 if (count != height * width) throw new CommandError($"BAD COUNT {count} {height * width}");
973 long[] values = words.Skip(4).Select(value => long.Parse(value, CultureInfo.InvariantCulture)).ToArray();
974 grid = new(height, width, values);
975 return [$"GRID BUILD rows={height} cols={width} total={values.Sum()}"];
976 }
977 GridFenwick tree = Need(grid, kind);
978 if (action == "DUMP") return tree.Dump().ToList();
979 int row = Index(2), column = Index(3);
980 CheckCell(tree, row, column);
981 if (action == "ADD")
982 {
983 long value = Number(4);
984 return [$"GRID ADD {row} {column} {value} visited={tree.Add(row, column, value)}"];
985 }
986 int bottom = Index(4), right = Index(5);
987 CheckCell(tree, bottom, right);
988 if (row > bottom || column > right) throw new CommandError($"BAD RECT {row} {column} {bottom} {right}");
989 return [$"GRID SUM {row} {column} {bottom} {right} {tree.Query(row, column, bottom, right).Text}"];
990 }
991 }
992 throw new InvalidOperationException("Unknown command");
993 }
994}
