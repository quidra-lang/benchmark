7def combine(op, a, b)
8 case op
9 when :SUM
10 a + b
11 when :MIN
12 [a, b].min
13 when :MAX
14 [a, b].max
15 when :GCD
16 a.gcd(b)
17 end

20Span = Data.define(:left, :right) do
21 def length = right - left + 1
22 def middle = (left + right) / 2
23 def leaf? = left == right
24 def covers?(other) = left <= other.left && other.right <= right
25 def left_child = Span.new(left, middle)
26 def right_child = Span.new(middle + 1, right)

34Aggregate = Data.define(:sum, :min, :max) do
35 def self.single(value) = new(value, value, value)
36 def merge(other) = Aggregate.new(sum + other.sum, [min, other.min].min, [max, other.max].max)

60class Segment
61 attr_reader :op, :shape, :tree
62 def initialize(values, op)
63 @op = op
64 @shape = Layout.new(values.length)
65 @tree = Array.new(4 * values.length, 0)
66 build(1, values)
67 end
68
69 def leaf_value(value) = @op == :GCD ? value.abs : value
70 def build(node, values)
71 span = @shape.spans[node]
72 if span.leaf?
73 @tree[node] = leaf_value(values[span.left])
74 else
75 build(2 * node, values)
76 build(2 * node + 1, values)
77 pull(node)
78 end
79 end
80
81 def pull(node) = @tree[node] = combine(@op, @tree[2 * node], @tree[2 * node + 1])
82
83 def query_at(node, wanted)
84 span = @shape.spans[node]
85 return Answer.new(@tree[node], 1) if wanted.covers?(span)
86 answer = if wanted.right <= span.middle
87 query_at(2 * node, wanted)
88 elsif wanted.left > span.middle
89 query_at(2 * node + 1, wanted)
90 else
91 left = query_at(2 * node, wanted)
92 right = query_at(2 * node + 1, wanted)
93 Answer.new(combine(@op, left.value, right.value), left.visited + right.visited)
94 end
95
96 answer.with(visited: answer.visited + 1)
97 end
98
99 def query(wanted)
100 answer = query_at(1, wanted)
101 Report.new("#{@op.downcase}=#{answer.value} visited=#{answer.visited}", [answer.value])
102 end
103
104 def update(index, value, node = 1)
105 span = @shape.spans[node]
106 if span.leaf?
107 @tree[node] = leaf_value(value)
108 return 1
109 end
110
111 visited = update(index, value, index <= span.middle ? 2 * node : 2 * node + 1)
112 pull(node)
113 visited + 1
114 end

116 def dump
117 rows = ["SEG DUMP #{@op} nodes=#{@shape.count}"]
118 @shape.spans.each_with_index do |span, node|
119 rows << "NODE #{node} #{span.left} #{span.right} #{@tree[node]}" if span
120 end
121
122 rows
123 end

133class Lazy
134 attr_reader :shape, :nodes
135 def initialize(values)
136 @shape = Layout.new(values.length)
137 @nodes = Array.new(4 * values.length)
138 build(1, values)
139 end
140
141 def build(node, values)
142 span = @shape.spans[node]
143 @nodes[node] = LazyNode.new(nil, nil, 0)
144 if span.leaf?
145 @nodes[node].value = Aggregate.single(values[span.left])
146 else
147 build(2 * node, values)
148 build(2 * node + 1, values)
149 pull(node)
150 end
151 end
152
153 def pull(node) = @nodes[node].value = @nodes[2 * node].value.merge(@nodes[2 * node + 1].value)
154 def apply(node, value, assignment)
155 span = @shape.spans[node]
156 target = @nodes[node]
157 target.value = assignment ? Aggregate.new(value * span.length, value, value) : target.value.add(value, span.length)
158 return if span.leaf?
159 if assignment
160 target.assign = value
161 target.add = 0
162 else
163 target.add += value
164 end
165 end
166
167 def push(node)
168 source = @nodes[node]
169 return 0 if source.assign.nil? && source.add.zero?
170 (2 * node..2 * node + 1).each do |child|
171 apply(child, source.assign, true) unless source.assign.nil?
172 apply(child, source.add, false) unless source.add.zero?
173 end
174
175 source.assign = nil
176 source.add = 0
177 1
178 end
179
180 def update_at(node, wanted, value, assignment, counts)
181 counts.visited += 1
182 span = @shape.spans[node]
183 if wanted.covers?(span)
184 counts.covered += 1
185 apply(node, value, assignment)
186 return
187 end
188
189 counts.pushes += push(node)
190 update_at(2 * node, wanted, value, assignment, counts) if wanted.left <= span.middle
191 update_at(2 * node + 1, wanted, value, assignment, counts) if wanted.right > span.middle
192 pull(node)
193 end
194
195 def update(wanted, value, assignment)
196 counts = Counts.new(0, 0, 0)
197 update_at(1, wanted, value, assignment, counts)
198 counts
199 end
200
201 def query_at(node, wanted)
202 span = @shape.spans[node]
203 return LazyAnswer.new(@nodes[node].value, 1, 0) if wanted.covers?(span)
204 pushes = push(node)
205 answer = if wanted.right <= span.middle
206 query_at(2 * node, wanted)
207 elsif wanted.left > span.middle
208 query_at(2 * node + 1, wanted)
209 else
210 left = query_at(2 * node, wanted)
211 right = query_at(2 * node + 1, wanted)
212 LazyAnswer.new(left.value.merge(right.value), left.visited + right.visited, left.pushes + right.pushes)
213 end
214
215 answer.with(visited: answer.visited + 1, pushes: answer.pushes + pushes)
216 end
217
218 def query(wanted)
219 answer = query_at(1, wanted)
220 value = answer.value
221 Report.new("#{value} visited=#{answer.visited} pushes=#{answer.pushes}", [value.sum, value.min, value.max])
222 end
223
224 def dump
225 rows = ["LAZY DUMP nodes=#{@shape.count}"]
226 @shape.spans.each_with_index do |span, node|
227 next unless span
228 item = @nodes[node]
229 rows <<
230 "NODE #{node} #{span.left} #{span.right} #{item.value} assign=#{item.assign.nil? ? "-" : item.assign} add=#{item.add}"
231 end
232
233 rows
234 end
235end

237class Bit
238 attr_reader :tree
239 def initialize(n)
240 @tree = Array.new(n + 1, 0)
241 end
242
243 def add(position, value)
244 visited = 0
245 while position < @tree.length
246 @tree[position] += value
247 visited += 1
248 position += position & -position
249 end
250
251 visited
252 end
253
254 def prefix(position)
255 sum = visited = 0
256 while position.positive?
257 sum += @tree[position]
258 visited += 1
259 position -= position & -position
260 end
261
262 Answer.new(sum, visited)
263 end
264
265 def query(span)
266 a = prefix(span.right + 1)
267 b = prefix(span.left)
268 Answer.new(a.value - b.value, a.visited + b.visited)
269 end
270
271 def lower(rest)
272 n = @tree.length - 1
273 position = visited = 0
274 step = 1 << (n.bit_length - 1)
275 while step.positive?
276 if position + step <= n
277 value = @tree[position + step]
278 visited += 1
279 if value < rest
280 rest -= value
281 position += step
282 end
283 end
284
285 step /= 2
286 end
287
288 Answer.new(position < n ? position : nil, visited)
289 end
290
291 def dump(name) = "#{name} #{@tree.drop(1).join(" ")}"
292end

303class RangeFenwick
304 attr_reader :first, :second
305 def initialize(values)
306 @first = Bit.new(values.length)
307 @second = Bit.new(values.length)
308 values.each_with_index do |value, i|
309 difference = value - (i.zero? ? 0 : values[i - 1])
310 @first.add(i + 1, difference)
311 @second.add(i + 1, difference * i)
312 end
313 end
314
315 def add(span, value)
316 visited = @first.add(span.left + 1, value)
317 visited += @first.add(span.right + 2, -value)
318 visited += @second.add(span.left + 1, value * span.left)
319 visited += @second.add(span.right + 2, -value * (span.right + 1))
320 visited
321 end
322
323 def prefix(position)
324 a = @first.prefix(position)
325 b = @second.prefix(position)
326 Answer.new(position * a.value - b.value, a.visited + b.visited)
327 end

329 def query(span)
330 a = prefix(span.right + 1)
331 b = prefix(span.left)
332 Answer.new(a.value - b.value, a.visited + b.visited)

336class Sparse
337 attr_reader :values, :minimum, :maximum, :entries
338 def initialize(values)
339 @values = values.dup.freeze
340 @minimum = [(0...values.length).to_a.freeze]
341 @maximum = [@minimum.first]
342 @entries = values.length
343 level = 1
344 while (1 << level) <= values.length
345 length = values.length - (1 << level) + 1
346 half = 1 << (level - 1)
347 @minimum << Array.new(length) { |i| select(@minimum[level - 1][i], @minimum[level - 1][i + half], true) }.freeze
348 @maximum << Array.new(length) { |i| select(@maximum[level - 1][i], @maximum[level - 1][i + half], false) }.freeze
349 @entries += length
350 level += 1
351 end
352
353 @minimum.freeze
354 @maximum.freeze
355 end
356
357 def select(a, b, minimum)
358 (minimum ? @values[a] <= @values[b] : @values[a] >= @values[b]) ? a : b
359 end
360
361 def query(span)
362 level = span.length.bit_length - 1
363 finish = span.right - (1 << level) + 1
364 low = select(@minimum[level][span.left], @minimum[level][finish], true)
365 high = select(@maximum[level][span.left], @maximum[level][finish], false)
366 Report.new(
367 "min=#{@values[low]} minpos=#{low} max=#{@values[high]} maxpos=#{high} level=#{level}",
368 [@values[low], @values[high]]
369 )
370 end
371
372 def dump
373 rows = ["SPARSE DUMP levels=#{@minimum.length}"]
374 @minimum.each_index do |level|
375 rows << "MIN #{level} #{@minimum[level].join(" ")}"
376 rows << "MAX #{level} #{@maximum[level].join(" ")}"
377 end
378
379 rows
380 end
381end

387class Sqrt
388 attr_reader :block, :raw, :tags, :sums
389 def initialize(values, block)
390 @block = block
391 @raw = values.dup
392 @tags = Array.new((values.length + block - 1) / block, 0)
393 @sums = Array.new(@tags.length, 0)
394 @raw.each_with_index { |value, i| @sums[i / block] += value }
395 end
396
397 def split(span)
398 first = span.left / @block
399 last = span.right / @block
400 cells = (span.left..span.right).select { |i| i / @block == first || i / @block == last }
401 Partition.new(cells, ((first + 1)...last).to_a)
402 end
403
404 def add(span, value)
405 part = split(span)
406 part.cells.each { |i|
407 @raw[i] += value
408 @sums[i / @block] += value
409 }
410 part.blocks.each { |b|
411 @tags[b] += value
412 @sums[b] += value * @block
413 }
414 part
415 end
416
417 def query(span)
418 part = split(span)
419 sum = part.cells.sum { |i| @raw[i] + @tags[i / @block] }
420 part.blocks.each { |b| sum += @sums[b] }
421 Report.new("sum=#{sum} #{part}", [sum])
422 end
423
424 def dump
425 rows = ["SQRT DUMP block=#{@block} blocks=#{@tags.length}"]
426 @tags.each_index do |b|
427 first = b * @block
428 finish = [(b + 1) * @block, @raw.length].min
429 rows << "BLOCK #{b} #{first} #{finish - 1} add=#{@tags[b]} sum=#{@sums[b]} #{@raw[first...finish].join(" ")}"
430 end
431
432 rows
433 end
434end

445def change_node(source, span, index, value, increment)
446 return ChangedNode.new(PersistentNode.new(increment ? source.value + value : value, nil, nil), 1) if span.leaf?
447 left = source.left
448 right = source.right
449 if index <= span.middle
450 changed = change_node(left, span.left_child, index, value, increment)
451 left = changed.node
452 else
453 changed = change_node(right, span.right_child, index, value, increment)
454 right = changed.node
455 end

458end

464 elsif wanted.left > span.middle
465 query_node(node.right, span.right_child, wanted)
466 else

475class Persistent
476 attr_reader :roots, :values, :position_nodes, :value_nodes
477 def initialize(contents)
478 @n = contents.length
479 @values = contents.uniq.sort.freeze
480 @roots = [build_node(contents, whole(@n))]
481 @prefixes = [build_node(nil, whole(@values.length))]
482 @position_nodes = 2 * @n - 1
483 @value_nodes = 2 * @values.length - 1
484 ranks = @values.each_with_index.to_h
485 contents.each do |value|
486 changed = change_node(@prefixes.last, whole(@values.length), ranks.fetch(value), 1, true)
487 @prefixes << changed.node
488 @value_nodes += changed.created
489 end
490
491 @prefixes.freeze
492 end
493
494 def set(version, index, value)
495 changed = change_node(@roots[version], whole(@n), index, value, false)
496 @roots << changed.node
497 @position_nodes += changed.created
498 changed.created
499 end
500
501 def query(version, span) = query_node(@roots[version], whole(@n), span).report
502 def kth(wanted, k)
503 u = @prefixes[wanted.right + 1]
504 w = @prefixes[wanted.left]
505 span = whole(@values.length)
506 visited = 1
507 until span.leaf?
508 count = u.left.value - w.left.value
509 if k <= count
510 u = u.left
511 w = w.left
512 span = span.left_child
513 else
514 k -= count
515 u = u.right
516 w = w.right
517 span = span.right_child
518 end
519
520 visited += 1
521 end
522
523 Answer.new(@values[span.left], visited)
524 end
525end

678 @values[span.left..span.right].each { |value|
679 expected = expected.merge(Aggregate.single(value))

713 def mo(queries)
714 block = Integer.sqrt(@values.length)
715 order = queries.each_index.sort_by do |i|
716 span = queries[i]
717 b = span.left / block
718 [b, b.even? ? span.right : -span.right, i]
719 end
720
721 frequencies = Hash.new(0)
722 answers = Array.new(queries.length)
723 left = right = moves = distinct = 0
724 add = lambda do |i|
725 value = @values[i]
726 distinct += 1 if frequencies[value].zero?
727 frequencies[value] += 1
728 end
729
730 remove = lambda do |i|
731 value = @values[i]
732 frequencies[value] -= 1
733 distinct -= 1 if frequencies[value].zero?
734 end
735
736 order.each do |index|
737 span = queries[index]
738 finish = span.right + 1
739 moves += (span.left - left).abs + (finish - right).abs
740 while left > span.left
741 left -= 1
742 add.call(left)
743 end
744
745 while right < finish
746 add.call(right)
747 right += 1
748 end
749
750 while left < span.left
751 remove.call(left)
752 left += 1
753 end
754
755 while right > finish
756 right -= 1
757 remove.call(right)
758 end
759
760 answers[index] = distinct
761 end
762
763 rows = [
764 "MO queries=#{queries.length} block=#{block} moves=#{moves}",
765 "ORDER #{order.map { |i| i + 1 }.join(" ")}"
766 ]
767 queries.each_with_index { |span, i| rows << "ANSWER #{i + 1} #{span.left} #{span.right} distinct=#{answers[i]}" }
768 rows
769 end

771 def execute(line)
772 words = line.split
773 case words[0]
774 when "LOAD"
775 removed = array_count
776 @values = words.drop(1).map(&:to_i)
777 @original = @values.dup.freeze
778 @structures.select! { |kind, _| kind == :GRID }
779 return ["LOAD n=#{@values.length} removed=#{removed}"]
780 when "RESET"
781 @values = @original.dup
782 op = @structures[:SEG]&.op || :SUM
783 block = @structures[:SQRT]&.block || 0
784 rebuilt = array_count
785 KINDS.each { |kind| build(kind, op, block) if kind != :GRID && @structures.key?(kind) }
786 return ["RESET rebuilt=#{rebuilt}"]
787 when "ARRAY", "CHECK"
788 span = Span.new(words[1].to_i, words[2].to_i)
789 check_range(span)
790 return check(span) if words[0] == "CHECK"
791 return ["ARRAY #{span.left} #{span.right} #{@values[span.left..span.right].join(" ")}"]
792 when "MO"
793 queries = words.drop(1).each_slice(2).map do |left, right|
794 span = Span.new(left.to_i, right.to_i)
795 check_range(span)
796 span
797 end
798
799 return mo(queries)
800 end
801
802 kind = words[0].to_sym
803 action = words[1]
804 if action == "BUILD"
805 op = :SUM
806 block = 0
807 if kind == :SEG
808 op = words[2].to_sym
809 raise CommandError, "BAD OP #{words[2]}" unless OPERATIONS.include?(op)
810 end
811
812 if kind == :SQRT
813 block = words.length == 3 ? words[2].to_i : Integer.sqrt(@values.length)
814 raise CommandError, "BAD BLOCK #{block}" if block > @values.length
815 end
816
817 if kind == :GRID
818 height = words[2].to_i
819 width = words[3].to_i
820 count = words.length - 4
821 raise CommandError, "BAD COUNT #{count} #{height * width}" unless count == height * width
822 values = words.drop(4).map(&:to_i)
823 @structures[kind] = Grid.new(height, width, values)
824 return ["GRID BUILD rows=#{height} cols=#{width} total=#{values.sum}"]
825 end
826
827 return [build(kind, op, block)]
828 end
829
830 tree = @structures[kind]
831 raise CommandError, "NOT BUILT #{kind}" unless tree
832 return dump(kind) if action == "DUMP"
833 if kind == :GRID
834 row = words[2].to_i
835 column = words[3].to_i
836 tree.check_cell(row, column)
837 if action == "ADD"
838 value = words[4].to_i
839 return ["GRID ADD #{row} #{column} #{value} visited=#{tree.add(row, column, value)}"]
840 end
841
842 bottom = words[4].to_i
843 right = words[5].to_i
844 tree.check_cell(bottom, right)
845 raise CommandError, "BAD RECT #{row} #{column} #{bottom} #{right}" if row > bottom || column > right
846 return ["GRID SUM #{row} #{column} #{bottom} #{right} #{tree.query(row, column, bottom, right).report.text}"]
847 end
848
849 if kind == :PERSIST
850 if action == "NODES"
851 return ["PERSIST NODES versions=#{tree.roots.length} position=#{tree.position_nodes} value=#{tree.value_nodes}"]
852 end
853
854 if action == "KTH"
855 span = Span.new(words[2].to_i, words[3].to_i)
856 check_range(span)
857 k = words[4].to_i
858 raise CommandError, "BAD K #{k}" if k < 1 || k > span.length
859 answer = tree.kth(span, k)
860 return ["PERSIST KTH #{span.left} #{span.right} #{k} value=#{answer.value} visited=#{answer.visited}"]
861 end
862
863 version = words[2].to_i
864 check_version(version)
865 index = words[3].to_i
866 if action == "SET"
867 check_index(index)
868 value = words[4].to_i
869 next_version = tree.roots.length
870 created = tree.set(version, index, value)
871 return ["PERSIST SET #{version} #{index} #{value} version=#{next_version} created=#{created}"]
872 end
873
874 span = Span.new(index, words[4].to_i)
875 check_range(span)
876 return ["PERSIST QUERY #{version} #{span.left} #{span.right} #{tree.query(version, span).text}"]
877 end
878
879 if kind == :FEN && action == "LOWER"
880 tree.values.each_with_index { |value, i| raise CommandError, "NEGATIVE #{i}" if value.negative? }
881 k = words[2].to_i
882 answer = tree.bit.lower(k)
883 return ["FEN LOWER #{k} index=#{answer.value || "none"} visited=#{answer.visited}"]
884 end
885
886 if kind == :SEG && action == "UPDATE"
887 index = words[2].to_i
888 check_index(index)
889 value = words[3].to_i
890 visited = tree.update(index, value)
891 @values[index] = value
892 return ["SEG UPDATE #{index} #{value} root=#{tree.tree[1]} visited=#{visited}"]
893 end
894
895 if kind == :FEN && %w[ADD PREFIX].include?(action)
896 index = words[2].to_i
897 check_index(index)
898 return ["FEN PREFIX #{index} #{tree.bit.prefix(index + 1).report.text}"] if action == "PREFIX"
899 value = words[3].to_i
900 @values[index] += value
901 tree.values[index] += value
902 return ["FEN ADD #{index} #{value} visited=#{tree.bit.add(index + 1, value)}"]
903 end
904
905 span = Span.new(words[2].to_i, words[3].to_i)
906 check_range(span)
907 if %w[ADD ASSIGN].include?(action)
908 value = words[4].to_i
909 assignment = action == "ASSIGN"
910 details = case kind
911 when :LAZY
912 tree.update(span, value, assignment).to_s
913 when :RFEN
914 "visited=#{tree.add(span, value)}"
915 when :SQRT
916 tree.add(span, value).to_s
917 end
918
919 (span.left..span.right).each { |i| @values[i] = assignment ? value : @values[i] + value }
920 return ["#{kind} #{action} #{span.left} #{span.right} #{value} #{details}"]
921 end
922
923 ["#{kind} #{action} #{span.left} #{span.right} #{query(kind, span).text}"]
924 end
925end
926
927exit unless STDIN.gets
928laboratory = Laboratory.new(STDIN.gets.split.map(&:to_i))
929commands = STDIN.gets.to_i
930commands.times do
931 begin
932 rows = laboratory.execute(STDIN.gets)
933 STDOUT.puts(rows)
934 rescue CommandError => error
935 STDOUT.puts("ERROR #{error.message}")
936 end
937
938 STDOUT.flush
939end
