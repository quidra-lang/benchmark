class Counts(visited := 0, covered := 0, pushes := 0)
class LazyNode(value := Aggregate(), int | none assignment := none, addition := 0)
Aggregate combine(const Aggregate a, const Aggregate b)
    return Aggregate(a.sum + b.sum, math.min(a.minimum, b.minimum), math.max(a.maximum, b.maximum))

bool covered(Span wanted, Span node)
    return wanted.left <= node.left and node.right <= wanted.right
void lazy_apply(LazyTree &tree, int node, Span span, int value, bool assign)
    length := span.right - span.left + 1
    if assign
        tree.nodes[node].value = Aggregate(value * length, value, value)
    else
        tree.nodes[node].value.sum += value * length
        tree.nodes[node].value.minimum += value
        tree.nodes[node].value.maximum += value
    if span.left < span.right
        if assign
            tree.nodes[node].assignment = value
            tree.nodes[node].addition = 0
        else
            tree.nodes[node].addition += value

void lazy_push(LazyTree &tree, int node, Span span, Counts &counts)
    saved := tree.nodes[node]
    assigned := false
    assignment := 0
    match saved.assignment
        int value
            assigned = true
            assignment = value
        none
            assigned = false
    if not assigned and saved.addition == 0
        return
    counts.pushes += 1
    middle := (span.left + span.right) / 2
    for: 0 <= side < 2
        child := if side == 0 then Span(span.left, middle) else Span(middle + 1, span.right)
        if assigned
            lazy_apply(&tree, 2 * node + side, child, assignment, true)
        if saved.addition != 0
            lazy_apply(&tree, 2 * node + side, child, saved.addition, false)
    tree.nodes[node].assignment = none
    tree.nodes[node].addition = 0
void lazy_update(LazyTree &tree, int node, Span span, Span wanted, int value, bool assign, Counts &counts)
    counts.visited += 1
    if covered(wanted, span)
        counts.covered += 1
        lazy_apply(&tree, node, span, value, assign)
        return
Aggregate lazy_query(LazyTree &tree, int node, Span span, Span wanted, Counts &counts)
    counts.visited += 1
    if covered(wanted, span)
        return tree.nodes[node].value
    lazy_push(&tree, node, span, &counts)
    middle := (span.left + span.right) / 2
    if wanted.right <= middle
        return lazy_query(&tree, 2 * node, Span(span.left, middle), wanted, &counts)
    if wanted.left > middle
        return lazy_query(&tree, 2 * node + 1, Span(middle + 1, span.right), wanted, &counts)
    first := lazy_query(&tree, 2 * node, Span(span.left, middle), wanted, &counts)
    second := lazy_query(&tree, 2 * node + 1, Span(middle + 1, span.right), wanted, &counts)
    return combine(first, second)
