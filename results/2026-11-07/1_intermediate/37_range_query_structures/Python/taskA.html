13def lowbit(position):
14 return position & -position

17def fenwick_add(tree, position, delta):
18 size = len(tree) - 1
19 visits = 0
20 while position <= size:
21 tree[position] += delta
22 visits += 1
23 position += lowbit(position)
24 return visits

27def fenwick_prefix(tree, position):
28 total, visits = 0, 0
29 while position >= 1:
30 total += tree[position]
31 visits += 1
32 position -= lowbit(position)
33 return total, visits

36def tree_layout(n):
37 low = [0] * (4 * n)
38 high = [0] * (4 * n)
39 numbers = []
40 pending = [(1, 0, n - 1)]
41 while pending:
42 node, first, last = pending.pop()
43 low[node] = first
44 high[node] = last
45 numbers.append(node)
46 if first < last:
47 middle = (first + last) // 2
48 pending.append((2 * node, first, middle))
49 pending.append((2 * node + 1, middle + 1, last))
50 numbers.sort()
51 return low, high, numbers

54class SegmentTree:
55 def __init__(self, values, op):
56 self.op = op
57 self.low, self.high, self.numbers = tree_layout(len(values))
58 self.value = [0] * len(self.low)
59 self.visits = 0
60 self.build(1, values)
61
62 def leaf(self, number):
63 return abs(number) if self.op == "GCD" else number
64
65 def combine(self, first, second):
66 if self.op == "SUM":
67 return first + second
68 if self.op == "MIN":
69 return min(first, second)
70 if self.op == "MAX":
71 return max(first, second)
72 return gcd(first, second)
73
74 def pull(self, node):
75 self.value[node] = self.combine(self.value[2 * node], self.value[2 * node + 1])
76
77 def build(self, node, values):
78 if self.low[node] == self.high[node]:
79 self.value[node] = self.leaf(values[self.low[node]])
80 return
81 self.build(2 * node, values)
82 self.build(2 * node + 1, values)
83 self.pull(node)
84
85 def query(self, left, right):
86 self.visits = 0
87 answer = self.query_node(1, left, right)
88 return answer, self.visits
89
90 def query_node(self, node, left, right):
91 self.visits += 1
92 low, high = self.low[node], self.high[node]
93 if left <= low and high <= right:
94 return self.value[node]
95 middle = (low + high) // 2
96 if right <= middle:
97 return self.query_node(2 * node, left, right)
98 if left > middle:
99 return self.query_node(2 * node + 1, left, right)
100 left_result = self.query_node(2 * node, left, right)
101 right_result = self.query_node(2 * node + 1, left, right)
102 return self.combine(left_result, right_result)
103
104 def update(self, index, number):
105 node = 1
106 while self.low[node] < self.high[node]:
107 middle = (self.low[node] + self.high[node]) // 2
108 node = 2 * node if index <= middle else 2 * node + 1
109 self.value[node] = self.leaf(number)
110 visited = 1
111 while node > 1:
112 node //= 2
113 self.pull(node)
114 visited += 1
115 return visited

118class LazyTree:
119 def __init__(self, values):
120 self.low, self.high, self.numbers = tree_layout(len(values))
121 size = len(self.low)
122 self.sums = [0] * size
123 self.mins = [0] * size
124 self.maxes = [0] * size
125 self.assigned = [None] * size
126 self.added = [0] * size
127 self.visits = self.covered = self.pushes = 0
128 self.build(1, values)
129
130 def is_leaf(self, node):
131 return self.low[node] == self.high[node]
132
133 def length(self, node):
134 return self.high[node] - self.low[node] + 1
135
136 def build(self, node, values):
137 if self.is_leaf(node):
138 value = values[self.low[node]]
139 self.sums[node] = self.mins[node] = self.maxes[node] = value
140 return
141 self.build(2 * node, values)
142 self.build(2 * node + 1, values)
143 self.pull(node)
144
145 def pull(self, node):
146 left, right = 2 * node, 2 * node + 1
147 self.sums[node] = self.sums[left] + self.sums[right]
148 self.mins[node] = min(self.mins[left], self.mins[right])
149 self.maxes[node] = max(self.maxes[left], self.maxes[right])
150
151 def apply_assign(self, node, value):
152 self.sums[node] = value * self.length(node)
153 self.mins[node] = self.maxes[node] = value
154 if not self.is_leaf(node):
155 self.assigned[node] = value
156 self.added[node] = 0
157
158 def apply_add(self, node, value):
159 self.sums[node] += value * self.length(node)
160 self.mins[node] += value
161 self.maxes[node] += value
162 if not self.is_leaf(node):
163 self.added[node] += value
164
165 def push(self, node):
166 assigned, added = self.assigned[node], self.added[node]
167 if assigned is None and added == 0:
168 return
169 self.pushes += 1
170 for child in (2 * node, 2 * node + 1):
171 if assigned is not None:
172 self.apply_assign(child, assigned)
173 if added != 0:
174 self.apply_add(child, added)
175 self.assigned[node] = None
176 self.added[node] = 0
177
178 def update(self, left, right, value, assign):
179 self.visits = self.covered = self.pushes = 0
180 self.update_node(1, left, right, value, assign)
181 return self.visits, self.covered, self.pushes
182
183 def update_node(self, node, left, right, value, assign):
184 self.visits += 1
185 low, high = self.low[node], self.high[node]
186 if left <= low and high <= right:
187 self.covered += 1
188 if assign:
189 self.apply_assign(node, value)
190 else:
191 self.apply_add(node, value)
192 return
193 self.push(node)
194 middle = (low + high) // 2
195 if left <= middle:
196 self.update_node(2 * node, left, right, value, assign)
197 if right > middle:
198 self.update_node(2 * node + 1, left, right, value, assign)
199 self.pull(node)
200
201 def query(self, left, right):
202 self.visits = self.pushes = 0
203 total, smallest, largest = self.query_node(1, left, right)
204 return total, smallest, largest, self.visits, self.pushes
205
206 def query_node(self, node, left, right):
207 self.visits += 1
208 low, high = self.low[node], self.high[node]
209 if left <= low and high <= right:
210 return self.sums[node], self.mins[node], self.maxes[node]
211 self.push(node)
212 middle = (low + high) // 2
213 if right <= middle:
214 return self.query_node(2 * node, left, right)
215 if left > middle:
216 return self.query_node(2 * node + 1, left, right)
217 left_sum, left_min, left_max = self.query_node(2 * node, left, right)
218 right_sum, right_min, right_max = self.query_node(2 * node + 1, left, right)
219 return left_sum + right_sum, min(left_min, right_min), max(left_max, right_max)

222class FenwickTree:
223 def __init__(self, values):
224 self.values = list(values)
225 self.tree = [0] * (len(values) + 1)
226 for index, value in enumerate(values):
227 fenwick_add(self.tree, index + 1, value)
228
229 def add(self, index, delta):
230 self.values[index] += delta
231 return fenwick_add(self.tree, index + 1, delta)
232
233 def range_sum(self, left, right):
234 upper, upper_visits = fenwick_prefix(self.tree, right + 1)
235 lower, lower_visits = fenwick_prefix(self.tree, left)
236 return upper - lower, upper_visits + lower_visits
237
238 def lower_bound(self, k):
239 n = len(self.values)
240 step = 1 << (n.bit_length() - 1)
241 position, rest, visits = 0, k, 0
242 while step >= 1:
243 if position + step <= n:
244 visits += 1
245 value = self.tree[position + step]
246 if value < rest:
247 rest -= value
248 position += step
249 step //= 2
250 return (position if position < n else None), visits

253class RangeFenwick:
254 def __init__(self, values):
255 n = len(values)
256 self.b1 = [0] * (n + 1)
257 self.b2 = [0] * (n + 1)
258 for position in range(1, n + 1):
259 previous = values[position - 2] if position > 1 else 0
260 difference = values[position - 1] - previous
261 fenwick_add(self.b1, position, difference)
262 fenwick_add(self.b2, position, (position - 1) * difference)
263
264 def add(self, left, right, delta):
265 visits = fenwick_add(self.b1, left + 1, delta)
266 visits += fenwick_add(self.b1, right + 2, -delta)
267 visits += fenwick_add(self.b2, left + 1, delta * left)
268 visits += fenwick_add(self.b2, right + 2, -delta * (right + 1))
269 return visits
270
271 def prefix(self, position):
272 first, first_visits = fenwick_prefix(self.b1, position)
273 second, second_visits = fenwick_prefix(self.b2, position)
274 return position * first - second, first_visits + second_visits
275
276 def range_sum(self, left, right):
277 upper, upper_visits = self.prefix(right + 1)
278 lower, lower_visits = self.prefix(left)
279 return upper - lower, upper_visits + lower_visits

282class SparseTable:
283 def __init__(self, values):
284 self.values = list(values)
285 n = len(values)
286 self.levels = n.bit_length()
287 self.mins = [list(range(n))]
288 self.maxes = [list(range(n))]
289 for level in range(1, self.levels):
290 half = 1 << (level - 1)
291 mins = []
292 maxes = []
293 for index in range(n - (1 << level) + 1):
294 mins.append(self.lower(self.mins[-1][index], self.mins[-1][index + half]))
295 maxes.append(self.higher(self.maxes[-1][index], self.maxes[-1][index + half]))
296 self.mins.append(mins)
297 self.maxes.append(maxes)
298
299 def lower(self, first, second):
300 return first if self.values[first] <= self.values[second] else second
301
302 def higher(self, first, second):
303 return first if self.values[first] >= self.values[second] else second
304
305 def query(self, left, right):
306 level = (right - left + 1).bit_length() - 1
307 end = right - (1 << level) + 1
308 minimum = self.lower(self.mins[level][left], self.mins[level][end])
309 maximum = self.higher(self.maxes[level][left], self.maxes[level][end])
310 return minimum, maximum, level

313class SqrtDecomposition:
314 def __init__(self, values, block):
315 self.n = len(values)
316 self.block = block
317 self.count = (self.n + block - 1) // block
318 self.raw = list(values)
319 self.tags = [0] * self.count
320 self.sums = [0] * self.count
321 for index, value in enumerate(values):
322 self.sums[index // block] += value
323
324 def split(self, left, right):
325 first, last = left // self.block, right // self.block
326 if first == last:
327 return list(range(left, right + 1)), []
328 cells = list(range(left, (first + 1) * self.block))
329 cells.extend(range(last * self.block, right + 1))
330 return cells, list(range(first + 1, last))
331
332 def add(self, left, right, delta):
333 cells, wholes = self.split(left, right)
334 for index in cells:
335 self.raw[index] += delta
336 self.sums[index // self.block] += delta
337 for number in wholes:
338 self.tags[number] += delta
339 self.sums[number] += delta * self.block
340 return len(wholes), len(cells)
341
342 def range_sum(self, left, right):
343 cells, wholes = self.split(left, right)
344 total = sum(self.raw[index] + self.tags[index // self.block] for index in cells)
345 total += sum(self.sums[number] for number in wholes)
346 return total, len(wholes), len(cells)

349class NodePool:
350 def __init__(self):
351 self.left = []
352 self.right = []
353 self.total = []
354
355 def create(self, left, right, total):
356 self.left.append(left)
357 self.right.append(right)
358 self.total.append(total)
359 return len(self.total) - 1
360
361 def build(self, values, low, high):
362 if low == high:
363 return self.create(-1, -1, values[low])
364 middle = (low + high) // 2
365 left = self.build(values, low, middle)
366 right = self.build(values, middle + 1, high)
367 return self.create(left, right, self.total[left] + self.total[right])
368
369 def assign(self, node, low, high, index, value):
370 if low == high:
371 return self.create(-1, -1, value)
372 middle = (low + high) // 2
373 left, right = self.left[node], self.right[node]
374 if index <= middle:
375 left = self.assign(left, low, middle, index, value)
376 else:
377 right = self.assign(right, middle + 1, high, index, value)
378 return self.create(left, right, self.total[left] + self.total[right])
379
380 def increment(self, node, low, high, index):
381 left, right = self.left[node], self.right[node]
382 if low < high:
383 middle = (low + high) // 2
384 if index <= middle:
385 left = self.increment(left, low, middle, index)
386 else:
387 right = self.increment(right, middle + 1, high, index)
388 return self.create(left, right, self.total[node] + 1)

391class PersistentTree:
392 def __init__(self, values):
393 self.n = len(values)
394 self.positions = NodePool()
395 self.versions = [self.positions.build(values, 0, self.n - 1)]
396 self.distinct = sorted(set(values))
397 ranks = {value: rank for rank, value in enumerate(self.distinct)}
398 last = len(self.distinct) - 1
399 self.counts = NodePool()
400 self.prefixes = [self.counts.build([0] * len(self.distinct), 0, last)]
401 for value in values:
402 self.prefixes.append(self.counts.increment(self.prefixes[-1], 0, last, ranks[value]))
403 self.visits = 0
404
405 def assign(self, version, index, value):
406 before = len(self.positions.total)
407 root = self.positions.assign(self.versions[version], 0, self.n - 1, index, value)
408 self.versions.append(root)
409 return len(self.versions) - 1, len(self.positions.total) - before
410
411 def query(self, version, left, right):
412 self.visits = 0
413 total = self.query_node(self.versions[version], 0, self.n - 1, left, right)
414 return total, self.visits
415
416 def query_node(self, node, low, high, left, right):
417 self.visits += 1
418 if left <= low and high <= right:
419 return self.positions.total[node]
420 middle = (low + high) // 2
421 if right <= middle:
422 return self.query_node(self.positions.left[node], low, middle, left, right)
423 if left > middle:
424 return self.query_node(self.positions.right[node], middle + 1, high, left, right)
425 left_total = self.query_node(self.positions.left[node], low, middle, left, right)
426 right_total = self.query_node(self.positions.right[node], middle + 1, high, left, right)
427 return left_total + right_total
428
429 def kth(self, left, right, k):
430 pool = self.counts
431 newer = self.prefixes[right + 1]
432 older = self.prefixes[left]
433 low, high = 0, len(self.distinct) - 1
434 visits = 1
435 while low < high:
436 middle = (low + high) // 2
437 difference = pool.total[pool.left[newer]] - pool.total[pool.left[older]]
438 if k <= difference:
439 newer, older = pool.left[newer], pool.left[older]
440 high = middle
441 else:
442 k -= difference
443 newer, older = pool.right[newer], pool.right[older]
444 low = middle + 1
445 visits += 1
446 return self.distinct[low], visits

449class Grid:
450 def __init__(self, rows, cols, values):
451 self.rows = rows
452 self.cols = cols
453 self.tree = [[0] * (cols + 1) for _ in range(rows + 1)]
454 for row in range(rows):
455 for col in range(cols):
456 self.add(row, col, values[row * cols + col])
457
458 def add(self, row, col, delta):
459 visits = 0
460 y = row + 1
461 while y <= self.rows:
462 z = col + 1
463 while z <= self.cols:
464 self.tree[y][z] += delta
465 visits += 1
466 z += lowbit(z)
467 y += lowbit(y)
468 return visits
469
470 def prefix(self, rows, cols):
471 total, visits = 0, 0
472 y = rows
473 while y >= 1:
474 z = cols
475 while z >= 1:
476 total += self.tree[y][z]
477 visits += 1
478 z -= lowbit(z)
479 y -= lowbit(y)
480 return total, visits
481
482 def rectangle(self, top, left, bottom, right):
483 whole, whole_visits = self.prefix(bottom + 1, right + 1)
484 above, above_visits = self.prefix(top, right + 1)
485 beside, beside_visits = self.prefix(bottom + 1, left)
486 corner, corner_visits = self.prefix(top, left)
487 visits = whole_visits + above_visits + beside_visits + corner_visits
488 return whole - above - beside + corner, visits

491class DistinctCounter:
492 def __init__(self, size):
493 self.counts = [0] * size
494 self.distinct = 0
495

507def mo_order(queries, block):
508 def key(number):

516def run_mo(values, queries, order):
517 ranks = {value: rank for rank, value in enumerate(sorted(set(values)))}
518 codes = [ranks[value] for value in values]
519 window = DistinctCounter(len(ranks))
520 answers = [0] * len(queries)
521 current_left, current_right = 0, -1
522 moves = 0
523 for number in order:
524 left, right = queries[number]
525 moves += abs(left - current_left) + abs(right - current_right)
526 while current_left > left:
527 current_left -= 1
528 window.add(codes[current_left])
529 while current_right < right:
530 current_right += 1
531 window.add(codes[current_right])
532 while current_left < left:
533 window.remove(codes[current_left])
534 current_left += 1
535 while current_right > right:
536 window.remove(codes[current_right])
537 current_right -= 1
538 answers[number] = window.distinct
539 return answers, moves

662 reports.append(("SEG", *self.seg_report(left, right)))
663 if self.lazy is not None:
664 reports.append(("LAZY", *self.lazy_report(left, right)))
665 if self.fen is not None:
666 reports.append(("FEN", *self.fen_report(left, right)))
667 if self.rfen is not None:
668 reports.append(("RFEN", *self.rfen_report(left, right)))
669 if self.sparse is not None:
670 reports.append(("SPARSE", *self.sparse_report(left, right)))
671 if self.sqrt is not None:
672 reports.append(("SQRT", *self.sqrt_report(left, right)))
673 if self.persist is not None:
674 version = len(self.persist.versions) - 1
675 items, pairs = self.persist_report(version, left, right)
676 reports.append(("PERSIST", f"version={version} {items}", pairs))
677 ok = 0
678 for kind, items, pairs in reports:
679 if all(value == expected[key] for value, key in pairs):
680 verdict = "OK"
681 ok += 1
682 else:
683 verdict = "DIFF"
684 self.emit(kind, verdict, items)
685 self.emit(f"RESULT ok={ok} diff={len(reports) - ok}")
