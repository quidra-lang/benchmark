9class CommandError extends Error {}

11function gcd(a, b) {
12 a = Math.abs(a); b = Math.abs(b);
13 while (b !== 0) [a, b] = [b, a % b];
14 return a;
15}
16function combine(operation, a, b) {
17 switch (operation) {
18 case 'SUM': return a + b;
19 case 'MIN': return Math.min(a, b);
20 case 'MAX': return Math.max(a, b);
21 case 'GCD': return gcd(a, b);
22 }
23}
24function merge(a, b) { return { sum: a.sum + b.sum, min: Math.min(a.min, b.min), max: Math.max(a.max, b.max) }; }

26class SegmentTree {
27 constructor(values, operation) {
28 this.n = values.length;
29 this.operation = operation;
30 this.nodes = [];
31 const build = (index, l, r) => {
32 let value;
33 if (l === r) value = operation === 'GCD' ? Math.abs(values[l]) : values[l];
34 else {
35 const middle = Math.floor((l + r) / 2);
36 value = combine(operation, build(2 * index, l, middle), build(2 * index + 1, middle + 1, r));
37 }
38 this.nodes[index] = { l, r, value };
39 return value;
40 };
41 build(1, 0, this.n - 1);
42 }
43 query(l, r) {
44 let visited = 0;
45 const walk = index => {
46 ++visited;
47 const node = this.nodes[index];
48 if (l <= node.l && node.r <= r) return node.value;
49 const middle = Math.floor((node.l + node.r) / 2);
50 if (r <= middle) return walk(2 * index);
51 if (l > middle) return walk(2 * index + 1);
52 return combine(this.operation, walk(2 * index), walk(2 * index + 1));
53 };
54 const value = walk(1);
55 return { value, visited };
56 }
57 update(position, value) {
58 let visited = 0;
59 const walk = index => {
60 ++visited;
61 const node = this.nodes[index];
62 if (node.l === node.r) node.value = this.operation === 'GCD' ? Math.abs(value) : value;
63 else {
64 const middle = Math.floor((node.l + node.r) / 2);
65 walk(position <= middle ? 2 * index : 2 * index + 1);
66 node.value = combine(this.operation, this.nodes[2 * index].value, this.nodes[2 * index + 1].value);
67 }
68 };
69 walk(1);
70 return visited;
71 }
72 buildLine() { return `SEG BUILD ${this.operation} nodes=${2 * this.n - 1} maxnode=${this.nodes.length - 1} root=${this.nodes[1].value}`; }
73 dump() {
74 const rows = [`SEG DUMP ${this.operation} nodes=${2 * this.n - 1}`];
75 this.nodes.forEach((node, index) => rows.push(`NODE ${index} ${node.l} ${node.r} ${node.value}`));
76 return rows;
77 }
78}
79class LazyTree {
80 constructor(values) {
81 this.n = values.length;
82 this.nodes = [];
83 const build = (index, l, r) => {
84 let valuesAtNode;
85 if (l === r) valuesAtNode = { sum: values[l], min: values[l], max: values[l] };
86 else {
87 const middle = Math.floor((l + r) / 2);
88 valuesAtNode = merge(build(2 * index, l, middle), build(2 * index + 1, middle + 1, r));
89 }
90 return this.nodes[index] = { l, r, ...valuesAtNode, assign: null, add: 0 };
91 };
92 build(1, 0, this.n - 1);
93 }
94 assign(index, value) {
95 const node = this.nodes[index];
96 node.sum = value * (node.r - node.l + 1);
97 node.min = node.max = value;
98 if (node.l < node.r) { node.assign = value; node.add = 0; }
99 }
100 add(index, value) {
101 const node = this.nodes[index];
102 node.sum += value * (node.r - node.l + 1);
103 node.min += value; node.max += value;
104 if (node.l < node.r) node.add += value;
105 }
106 push(index) {
107 const node = this.nodes[index];
108 if (node.assign === null && node.add === 0) return 0;
109 for (const child of [2 * index, 2 * index + 1]) {
110 if (node.assign !== null) this.assign(child, node.assign);
111 if (node.add !== 0) this.add(child, node.add);
112 }
113 node.assign = null; node.add = 0;
114 return 1;
115 }
116 update(l, r, value, assignment) {
117 let visited = 0, covered = 0, pushes = 0;
118 const walk = index => {
119 ++visited;
120 const node = this.nodes[index];
121 if (l <= node.l && node.r <= r) {
122 ++covered;
123 if (assignment) this.assign(index, value);
124 else this.add(index, value);
125 return;
126 }
127 pushes += this.push(index);
128 const middle = Math.floor((node.l + node.r) / 2);
129 if (l <= middle) walk(2 * index);
130 if (r > middle) walk(2 * index + 1);
131 Object.assign(node, merge(this.nodes[2 * index], this.nodes[2 * index + 1]));
132 };
133 walk(1);
134 return { visited, covered, pushes };
135 }
136 query(l, r) {
137 let visited = 0, pushes = 0;
138 const walk = index => {
139 ++visited;
140 const node = this.nodes[index];
141 if (l <= node.l && node.r <= r) return { sum: node.sum, min: node.min, max: node.max };
142 pushes += this.push(index);
143 const middle = Math.floor((node.l + node.r) / 2);
144 if (r <= middle) return walk(2 * index);
145 if (l > middle) return walk(2 * index + 1);
146 return merge(walk(2 * index), walk(2 * index + 1));
147 };
148 return { ...walk(1), visited, pushes };
149 }
150 buildLine() { return `LAZY BUILD nodes=${2 * this.n - 1} ${summary(this.nodes[1])}`; }
151 dump() {
152 const rows = [`LAZY DUMP nodes=${2 * this.n - 1}`];
153 this.nodes.forEach((node, index) => rows.push(`NODE ${index} ${node.l} ${node.r} ${summary(node)} assign=${node.assign ?? '-'} add=${node.add}`));
154 return rows;
155 }
156}
157class Fenwick {
158 constructor(values) {
159 this.n = values.length;
160 this.tree = [0, ...values];
161 for (let position = 1; position <= this.n; ++position) {
162 const parent = position + lowbit(position);
163 if (parent <= this.n) this.tree[parent] += this.tree[position];
164 }
165 }
166 add(position, value) {
167 let visited = 0;
168 for (; position <= this.n; position += lowbit(position)) { this.tree[position] += value; ++visited; }
169 return visited;
170 }
171 prefix(position) {
172 let value = 0, visited = 0;
173 for (; position >= 1; position -= lowbit(position)) { value += this.tree[position]; ++visited; }
174 return { value, visited };
175 }
176 range(l, r) {
177 const right = this.prefix(r + 1), left = this.prefix(l);
178 return { value: right.value - left.value, visited: right.visited + left.visited };
179 }
180 lower(value) {
181 let position = 0, visited = 0;
182 for (let step = 2 ** Math.floor(Math.log2(this.n)); step >= 1; step = Math.floor(step / 2)) {
183 if (position + step <= this.n) {
184 ++visited;
185 if (this.tree[position + step] < value) { value -= this.tree[position + step]; position += step; }
186 }
187 }
188 return { index: position < this.n ? position : null, visited };
189 }
190}
191class FenwickArray {
192 constructor(values) { this.contents = values.slice(); this.bit = new Fenwick(values); }
193 add(index, value) { this.contents[index] += value; return this.bit.add(index + 1, value); }
194 buildLine() { return `FEN BUILD n=${this.contents.length} total=${sum(this.contents)}`; }
195 dump() { return [`FEN DUMP n=${this.contents.length}`, 'T ' + this.bit.tree.slice(1).join(' ')]; }
196}
197class RangeFenwick {
198 constructor(values) {
199 this.n = values.length;
200 const differences = values.map((value, index) => value - (index === 0 ? 0 : values[index - 1]));
201 this.first = new Fenwick(differences);
202 this.second = new Fenwick(differences.map((value, index) => value * index));
203 }
204 add(l, r, value) {
205 return this.first.add(l + 1, value) + this.first.add(r + 2, -value)
206 + this.second.add(l + 1, value * l) + this.second.add(r + 2, -value * (r + 1));
207 }
208 prefix(position) {
209 const first = this.first.prefix(position), second = this.second.prefix(position);
210 return { value: position * first.value - second.value, visited: first.visited + second.visited };
211 }
212 query(l, r) {
213 const right = this.prefix(r + 1), left = this.prefix(l);
214 return { value: right.value - left.value, visited: right.visited + left.visited };
215 }
216 buildLine() { return `RFEN BUILD n=${this.n} total=${this.prefix(this.n).value}`; }
217 dump() { return [`RFEN DUMP n=${this.n}`, 'B1 ' + this.first.tree.slice(1).join(' '), 'B2 ' + this.second.tree.slice(1).join(' ')]; }
218}
219class SparseTable {
220 constructor(values) {
221 this.contents = values.slice();
222 this.minimum = [values.map((_, index) => index)];
223 this.maximum = [values.map((_, index) => index)];
224 for (let level = 1, width = 2; width <= values.length; ++level, width *= 2) {
225 const minima = [], maxima = [], half = width / 2;
226 for (let index = 0; index + width <= values.length; ++index) {
227 const a = this.minimum[level - 1][index], b = this.minimum[level - 1][index + half];
228 const c = this.maximum[level - 1][index], d = this.maximum[level - 1][index + half];
229 minima.push(values[a] <= values[b] ? a : b);
230 maxima.push(values[c] >= values[d] ? c : d);
231 }
232 this.minimum.push(minima); this.maximum.push(maxima);
233 }
234 }
235 query(l, r) {
236 const level = Math.floor(Math.log2(r - l + 1)), end = r - 2 ** level + 1;
237 const a = this.minimum[level][l], b = this.minimum[level][end], c = this.maximum[level][l], d = this.maximum[level][end];
238 const minpos = this.contents[a] <= this.contents[b] ? a : b, maxpos = this.contents[c] >= this.contents[d] ? c : d;
239 return { min: this.contents[minpos], minpos, max: this.contents[maxpos], maxpos, level };
240 }
241 buildLine() { return `SPARSE BUILD levels=${this.minimum.length} entries=${sum(this.minimum.map(row => row.length))}`; }
242 dump() {
243 const rows = [`SPARSE DUMP levels=${this.minimum.length}`];
244 this.minimum.forEach((row, level) => rows.push(`MIN ${level} ${row.join(' ')}`, `MAX ${level} ${this.maximum[level].join(' ')}`));
245 return rows;
246 }

248class SquareRoot {
249 constructor(values, block) {
250 this.n = values.length;
251 this.block = block;
252 this.raw = values.slice();
253 this.tags = Array(Math.ceil(this.n / block)).fill(0);
254 this.sums = this.tags.slice();
255 values.forEach((value, index) => this.sums[Math.floor(index / block)] += value);
256 }
257 split(l, r) {
258 const left = Math.floor(l / this.block), right = Math.floor(r / this.block), cells = [], whole = [];
259 if (left === right) {
260 for (let index = l; index <= r; ++index) cells.push(index);
261 } else {
262 for (let index = l; index < (left + 1) * this.block; ++index) cells.push(index);
263 for (let index = right * this.block; index <= r; ++index) cells.push(index);
264 for (let block = left + 1; block < right; ++block) whole.push(block);
265 }
266 return { cells, whole };
267 }
268 add(l, r, value) {
269 const { cells, whole } = this.split(l, r);
270 for (const index of cells) { this.raw[index] += value; this.sums[Math.floor(index / this.block)] += value; }
271 for (const block of whole) { this.tags[block] += value; this.sums[block] += value * this.block; }
272 return { cells: cells.length, whole: whole.length };
273 }
274 query(l, r) {
275 const { cells, whole } = this.split(l, r);
276 const value = sum(cells.map(index => this.raw[index] + this.tags[Math.floor(index / this.block)])) + sum(whole.map(block => this.sums[block]));
277 return { value, cells: cells.length, whole: whole.length };
278 }
279 buildLine() { return `SQRT BUILD block=${this.block} blocks=${this.tags.length}`; }
280 dump() {
281 return [`SQRT DUMP block=${this.block} blocks=${this.tags.length}`, ...this.tags.map((tag, block) => {
282 const first = block * this.block, last = Math.min(first + this.block, this.n) - 1;
283 return `BLOCK ${block} ${first} ${last} add=${tag} sum=${this.sums[block]} ${this.raw.slice(first, last + 1).join(' ')}`;
284 })];
285 }

287class PersistentTree {
288 constructor(values) {
289 this.n = values.length;
290 this.nodes = 0;
291 const build = (l, r) => {
292 ++this.nodes;
293 if (l === r) return { sum: values[l], left: null, right: null };
294 const middle = Math.floor((l + r) / 2), left = build(l, middle), right = build(middle + 1, r);
295 return { sum: left.sum + right.sum, left, right };
296 };
297 this.roots = [build(0, this.n - 1)];
298 }
299 fork(version, index, value, addition = false) {
300 const before = this.nodes;
301 const copy = (node, l, r) => {
302 ++this.nodes;
303 if (l === r) return { sum: addition ? node.sum + value : value, left: null, right: null };
304 const middle = Math.floor((l + r) / 2);
305 const left = index <= middle ? copy(node.left, l, middle) : node.left;
306 const right = index > middle ? copy(node.right, middle + 1, r) : node.right;
307 return { sum: left.sum + right.sum, left, right };
308 };
309 this.roots.push(copy(this.roots[version], 0, this.n - 1));
310 return this.nodes - before;
311 }
312 query(version, l, r) {
313 let visited = 0;
314 const walk = (node, nl, nr) => {
315 ++visited;
316 if (l <= nl && nr <= r) return node.sum;
317 const middle = Math.floor((nl + nr) / 2);
318 if (r <= middle) return walk(node.left, nl, middle);
319 if (l > middle) return walk(node.right, middle + 1, nr);
320 return walk(node.left, nl, middle) + walk(node.right, middle + 1, nr);
321 };
322 const value = walk(this.roots[version], 0, this.n - 1);
323 return { value, visited };
324 }
325}
326class Persistence {
327 constructor(values) {
328 this.positions = new PersistentTree(values);
329 this.values = [...new Set(values)].sort((a, b) => a - b);
330 this.prefixes = new PersistentTree(Array(this.values.length).fill(0));
331 const ranks = new Map(this.values.map((value, index) => [value, index]));
332 values.forEach((value, index) => this.prefixes.fork(index, ranks.get(value), 1, true));
333 }
334 kth(l, r, k) {
335 let upper = this.prefixes.roots[r + 1], lower = this.prefixes.roots[l];
336 let left = 0, right = this.values.length - 1, visited = 1;
337 while (left < right) {
338 const middle = Math.floor((left + right) / 2), count = upper.left.sum - lower.left.sum;
339 if (k <= count) { upper = upper.left; lower = lower.left; right = middle; }
340 else { k -= count; upper = upper.right; lower = lower.right; left = middle + 1; }
341 ++visited;
342 }
343 return { value: this.values[left], visited };
344 }
345 buildLine() { return `PERSIST BUILD nodes=${this.positions.nodes} distinct=${this.values.length} valuenodes=${this.prefixes.nodes}`; }
346 nodeLine() { return `PERSIST NODES versions=${this.positions.roots.length} position=${this.positions.nodes} value=${this.prefixes.nodes}`; }
347}
348class Grid {
349 constructor(height, width, values) {
350 this.h = height; this.w = width;
351 this.tree = Array.from({ length: height + 1 }, () => Array(width + 1).fill(0));
352 values.forEach((value, index) => this.add(Math.floor(index / width), index % width, value));
353 }
354 add(i, j, value) {
355 let visited = 0;
356 for (let y = i + 1; y <= this.h; y += lowbit(y)) {
357 for (let z = j + 1; z <= this.w; z += lowbit(z)) { this.tree[y][z] += value; ++visited; }
358 }
359 return visited;
360 }
361 prefix(i, j) {
362 let value = 0, visited = 0;
363 for (let y = i; y >= 1; y -= lowbit(y)) {
364 for (let z = j; z >= 1; z -= lowbit(z)) { value += this.tree[y][z]; ++visited; }
365 }
366 return { value, visited };
367 }
368 query(i1, j1, i2, j2) {
369 const a = this.prefix(i2 + 1, j2 + 1), b = this.prefix(i1, j2 + 1), c = this.prefix(i2 + 1, j1), d = this.prefix(i1, j1);
370 return { value: a.value - b.value - c.value + d.value, visited: a.visited + b.visited + c.visited + d.visited };
371 }
372 dump() { return [`GRID DUMP rows=${this.h} cols=${this.w}`, ...this.tree.slice(1).map((row, index) => `ROW ${index + 1} ${row.slice(1).join(' ')}`)]; }
373}
374function mo(values, ranges) {
375 const block = Math.floor(Math.sqrt(values.length));
376 const queries = ranges.map(([l, r], index) => ({ l, r, index, block: Math.floor(l / block) }));
377 const order = queries.slice().sort((a, b) => a.block - b.block || (a.block % 2 === 0 ? a.r - b.r : b.r - a.r) || a.index - b.index);
378 const counts = new Map(), answers = Array(ranges.length);
379 let left = 0, right = -1, moves = 0;
380 const add = index => { const value = values[index]; counts.set(value, (counts.get(value) ?? 0) + 1); ++moves; };
381 const remove = index => {
382 const value = values[index], count = counts.get(value) - 1;
383 if (count === 0) counts.delete(value);
384 else counts.set(value, count);
385 ++moves;
386 };
387 for (const query of order) {
388 while (left > query.l) add(--left);
389 while (right < query.r) add(++right);
390 while (left < query.l) remove(left++);
391 while (right > query.r) remove(right--);
392 answers[query.index] = counts.size;
393 }
394 return [`MO queries=${ranges.length} block=${block} moves=${moves}`, 'ORDER ' + order.map(query => query.index + 1).join(' '),
395 ...queries.map(query => `ANSWER ${query.index + 1} ${query.l} ${query.r} distinct=${answers[query.index]}`)];
396}
397class Laboratory {
398 constructor(values) {
399 this.array = values.slice();
400 this.original = values.slice();
401 this.structures = Object.fromEntries(kinds.map(kind => [kind, null]));
402 this.grid = null;
403 }
404 checkIndex(index) { if (index < 0 || index >= this.array.length) fail(`BAD INDEX ${index}`); }
405 checkRange(l, r) {
406 this.checkIndex(l); this.checkIndex(r);
407 if (l > r) fail(`BAD RANGE ${l} ${r}`);
408 }
409 checkVersion(state, version) {
410 if (version < 0 || version >= state.positions.roots.length) fail(`BAD VERSION ${version}`);
411 }
412 checkCell(i, j) {
413 if (i < 0 || i >= this.grid.h || j < 0 || j >= this.grid.w) fail(`BAD CELL ${i} ${j}`);
414 }
415 create(kind, option) {
416 switch (kind) {
417 case 'SEG': return new SegmentTree(this.array, option);
418 case 'LAZY': return new LazyTree(this.array);
419 case 'FEN': return new FenwickArray(this.array);
420 case 'RFEN': return new RangeFenwick(this.array);
421 case 'SPARSE': return new SparseTable(this.array);
422 case 'SQRT': return new SquareRoot(this.array, option);
423 case 'PERSIST': return new Persistence(this.array);
424 default: throw new Error(`Unknown structure ${kind}`);
425 }
426 }
427 describe(kind, l, r, version = null) {
428 const state = this.structures[kind];
429 switch (kind) {
430 case 'SEG': {
431 const result = state.query(l, r), key = state.operation.toLowerCase();
432 return { text: `${key}=${result.value} visited=${result.visited}`, values: { [key]: result.value } };
433 }
434 case 'LAZY': {
435 const result = state.query(l, r);
436 return { text: `${summary(result)} visited=${result.visited} pushes=${result.pushes}`,
437 values: { sum: result.sum, min: result.min, max: result.max } };
438 }
439 case 'FEN':
440 case 'RFEN': {
441 const result = kind === 'FEN' ? state.bit.range(l, r) : state.query(l, r);
442 return { text: `sum=${result.value} visited=${result.visited}`, values: { sum: result.value } };
443 }
444 case 'SPARSE': {
445 const result = state.query(l, r);
446 return { text: `min=${result.min} minpos=${result.minpos} max=${result.max} maxpos=${result.maxpos} level=${result.level}`,
447 values: { min: result.min, max: result.max } };
448 }
449 case 'SQRT': {
450 const result = state.query(l, r);
451 return { text: `sum=${result.value} whole=${result.whole} cells=${result.cells}`, values: { sum: result.value } };
452 }
453 case 'PERSIST': {
454 const selected = version ?? state.positions.roots.length - 1, result = state.positions.query(selected, l, r);
455 return { text: `sum=${result.value} visited=${result.visited}`, values: { sum: result.value }, version: selected };
456 }
457 }
458 }
459 check(l, r) {
460 const values = this.array.slice(l, r + 1), direct = { sum: sum(values), min: Math.min(...values), max: Math.max(...values), gcd: values.reduce(gcd, 0) };
461 const rows = [`CHECK ${l} ${r} ${summary(direct)} gcd=${direct.gcd}`];
462 let ok = 0, different = 0;
463 for (const kind of kinds) if (this.structures[kind] !== null) {
464 const result = this.describe(kind, l, r), matches = Object.entries(result.values).every(([key, value]) => value === direct[key]);
465 if (matches) ++ok;
466 else ++different;
467 const prefix = kind === 'PERSIST' ? `version=${result.version} ` : '';
468 rows.push(`${kind} ${matches ? 'OK' : 'DIFF'} ${prefix}${result.text}`);
469 }
470 rows.push(`RESULT ok=${ok} diff=${different}`);
471 return rows;
472 }
473 modifyArray(l, r, value, assignment = false) {
474 for (let index = l; index <= r; ++index) this.array[index] = assignment ? value : this.array[index] + value;
475 }
476 execute(line) {
477 const words = line.split(' '), kind = words[0], action = words[1];
478 if (kind === 'LOAD') {
479 const removed = kinds.filter(kind => this.structures[kind] !== null).length;
480 this.array = words.slice(1).map(Number);
481 this.original = this.array.slice();
482 this.structures = Object.fromEntries(kinds.map(kind => [kind, null]));
483 return [`LOAD n=${this.array.length} removed=${removed}`];
484 }
485 if (kind === 'RESET') {
486 this.array = this.original.slice();
487 let rebuilt = 0;
488 for (const kind of kinds) {
489 const previous = this.structures[kind];
490 if (previous === null) continue;
491 const option = kind === 'SEG' ? previous.operation : kind === 'SQRT' ? previous.block : undefined;
492 this.structures[kind] = this.create(kind, option);
493 ++rebuilt;
494 }
495 return [`RESET rebuilt=${rebuilt}`];
496 }
497 if (kind === 'ARRAY' || kind === 'CHECK') {
498 const [l, r] = words.slice(1).map(Number);
499 this.checkRange(l, r);
500 return kind === 'CHECK' ? this.check(l, r) : [`ARRAY ${l} ${r} ${this.array.slice(l, r + 1).join(' ')}`];
501 }
502 if (kind === 'MO') {
503 const numbers = words.slice(1).map(Number), ranges = [];
504 for (let index = 0; index < numbers.length; index += 2) {
505 const l = numbers[index], r = numbers[index + 1];
506 this.checkRange(l, r);
507 ranges.push([l, r]);
508 }
509 return mo(this.array, ranges);
510 }
511 const args = words.slice(2);
512 if (action === 'BUILD') {
513 let option;
514 if (kind === 'SEG') {
515 option = args[0];
516 if (!operations.includes(option)) fail(`BAD OP ${option}`);
517 } else if (kind === 'SQRT') {
518 option = args.length === 0 ? Math.floor(Math.sqrt(this.array.length)) : Number(args[0]);
519 if (option > this.array.length) fail(`BAD BLOCK ${option}`);
520 } else if (kind === 'GRID') {
521 const [height, width, ...values] = args.map(Number);
522 if (values.length !== height * width) fail(`BAD COUNT ${values.length} ${height * width}`);
523 this.grid = new Grid(height, width, values);
524 return [`GRID BUILD rows=${height} cols=${width} total=${sum(values)}`];
525 }
526 const state = this.create(kind, option);
527 this.structures[kind] = state;
528 return [state.buildLine()];
529 }
530 const state = kind === 'GRID' ? this.grid : this.structures[kind];
531 if (state === null) fail(`NOT BUILT ${kind}`);
532 if (action === 'DUMP') return state.dump();
533 const numbers = args.map(Number);
534 if (kind === 'SEG') {
535 if (action === 'UPDATE') {
536 const [index, value] = numbers;
537 this.checkIndex(index);
538 const visited = state.update(index, value);
539 this.array[index] = value;
540 return [`SEG UPDATE ${index} ${value} root=${state.nodes[1].value} visited=${visited}`];
541 }
542 const [l, r] = numbers;
543 this.checkRange(l, r);
544 return [`SEG QUERY ${l} ${r} ${this.describe(kind, l, r).text}`];
545 }
546 if (kind === 'LAZY') {
547 const [l, r, value] = numbers;
548 this.checkRange(l, r);
549 if (action === 'QUERY') return [`LAZY QUERY ${l} ${r} ${this.describe(kind, l, r).text}`];
550 const assignment = action === 'ASSIGN', result = state.update(l, r, value, assignment);
551 this.modifyArray(l, r, value, assignment);
552 return [`LAZY ${action} ${l} ${r} ${value} visited=${result.visited} covered=${result.covered} pushes=${result.pushes}`];
553 }
554 if (kind === 'FEN') {
555 if (action === 'LOWER') {
556 const negative = state.contents.findIndex(value => value < 0);
557 if (negative >= 0) fail(`NEGATIVE ${negative}`);
558 const result = state.bit.lower(numbers[0]);
559 return [`FEN LOWER ${numbers[0]} index=${result.index ?? 'none'} visited=${result.visited}`];
560 }
561 if (action === 'RANGE') {
562 const [l, r] = numbers;
563 this.checkRange(l, r);
564 return [`FEN RANGE ${l} ${r} ${this.describe(kind, l, r).text}`];
565 }
566 const index = numbers[0];
567 this.checkIndex(index);
568 if (action === 'ADD') {
569 const value = numbers[1], visited = state.add(index, value);
570 this.array[index] += value;
571 return [`FEN ADD ${index} ${value} visited=${visited}`];
572 }
573 if (action === 'PREFIX') {
574 const result = state.bit.prefix(index + 1);
575 return [`FEN PREFIX ${index} sum=${result.value} visited=${result.visited}`];
576 }
577 }
578 if (kind === 'RFEN' || kind === 'SQRT') {
579 const [l, r, value] = numbers;
580 this.checkRange(l, r);
581 if (action === 'SUM') return [`${kind} SUM ${l} ${r} ${this.describe(kind, l, r).text}`];
582 const result = state.add(l, r, value);
583 this.modifyArray(l, r, value);
584 const detail = kind === 'RFEN' ? `visited=${result}` : `whole=${result.whole} cells=${result.cells}`;
585 return [`${kind} ADD ${l} ${r} ${value} ${detail}`];
586 }
587 if (kind === 'SPARSE') {
588 const [l, r] = numbers;
589 this.checkRange(l, r);
590 return [`SPARSE QUERY ${l} ${r} ${this.describe(kind, l, r).text}`];
591 }
592 if (kind === 'PERSIST') {
593 if (action === 'NODES') return [state.nodeLine()];
594 if (action === 'KTH') {
595 const [l, r, k] = numbers;
596 this.checkRange(l, r);
597 if (k < 1 || k > r - l + 1) fail(`BAD K ${k}`);
598 const result = state.kth(l, r, k);
599 return [`PERSIST KTH ${l} ${r} ${k} value=${result.value} visited=${result.visited}`];
600 }
601 const version = numbers[0];
602 this.checkVersion(state, version);
603 if (action === 'SET') {
604 const [, index, value] = numbers;
605 this.checkIndex(index);
606 const newVersion = state.positions.roots.length, created = state.positions.fork(version, index, value);
607 return [`PERSIST SET ${version} ${index} ${value} version=${newVersion} created=${created}`];
608 }
609 const [, l, r] = numbers;
610 this.checkRange(l, r);
611 return [`PERSIST QUERY ${version} ${l} ${r} ${this.describe(kind, l, r, version).text}`];
612 }
613 if (kind === 'GRID') {
614 if (action === 'ADD') {
615 const [i, j, value] = numbers;
616 this.checkCell(i, j);
617 return [`GRID ADD ${i} ${j} ${value} visited=${state.add(i, j, value)}`];
618 }
619 const [i1, j1, i2, j2] = numbers;
620 this.checkCell(i1, j1); this.checkCell(i2, j2);
621 if (i1 > i2 || j1 > j2) fail(`BAD RECT ${i1} ${j1} ${i2} ${j2}`);
622 const result = state.query(i1, j1, i2, j2);
623 return [`GRID SUM ${i1} ${j1} ${i2} ${j2} sum=${result.value} visited=${result.visited}`];
624 }
625 throw new Error(`Unhandled command ${line}`);
626 }
627}
