41tree_shape <- function(n) {
42 tree <- record(n = n, left = rep.int(NA_integer_, 4L * n), right = integer(4L * n))
43 visit <- function(node, left, right) {
44 tree$left[node] <- left
45 tree$right[node] <- right
46 if (left < right) {
47 middle <- (left + right) %/% 2L
48 visit(2L * node, left, middle)
49 visit(2L * node + 1L, middle + 1L, right)
50 }
51 }
52 visit(1L, 0L, n - 1L)
53 tree$nodes <- which(!is.na(tree$left))
54 tree
55}
56new_segment <- function(values, op) {
57 tree <- tree_shape(length(values))
58 tree$op <- op
59 tree$value <- numeric(4L * tree$n)
60 build <- function(node) {
61 left <- tree$left[node]
62 right <- tree$right[node]
63 if (left == right) {
64 value <- values[left + 1L]
65 tree$value[node] <- if (op == "GCD") abs(value) else value
66 } else {
67 build(2L * node)
68 build(2L * node + 1L)
69 tree$value[node] <- operate(op, tree$value[c(2L * node, 2L * node + 1L)])
70 }
71 }
72 build(1L)
73 tree
74}
75segment_query <- function(tree, node, left, right, count) {
76 count$visited <- count$visited + 1L
77 nl <- tree$left[node]
78 nr <- tree$right[node]
79 if (left <= nl && nr <= right) return(tree$value[node])
80 middle <- (nl + nr) %/% 2L
81 if (right <= middle) return(segment_query(tree, 2L * node, left, right, count))
82 if (left > middle) return(segment_query(tree, 2L * node + 1L, left, right, count))
83 first <- segment_query(tree, 2L * node, left, right, count)
84 second <- segment_query(tree, 2L * node + 1L, left, right, count)
85 operate(tree$op, c(first, second))
86}
87segment_update <- function(tree, node, index, value) {
88 left <- tree$left[node]
89 right <- tree$right[node]
90 if (left == right) {
91 tree$value[node] <- if (tree$op == "GCD") abs(value) else value
92 return(1L)
93 }
94 child <- 2L * node + as.integer(index > (left + right) %/% 2L)
95 visited <- segment_update(tree, child, index, value) + 1L
96 tree$value[node] <- operate(tree$op, tree$value[c(2L * node, 2L * node + 1L)])
97 visited
98}
99lazy_pull <- function(tree, node) {
100 tree$value[node, ] <- combine(tree$value[2L * node, ], tree$value[2L * node + 1L, ])
101}
102new_lazy <- function(values) {
103 tree <- tree_shape(length(values))
104 tree$value <- matrix(0, 4L * tree$n, 3L)
105 tree$assign <- rep.int(NA_real_, 4L * tree$n)
106 tree$add <- numeric(4L * tree$n)
107 build <- function(node) {
108 if (tree$left[node] == tree$right[node]) {
109 tree$value[node, ] <- values[tree$left[node] + 1L]
110 } else {
111 build(2L * node)
112 build(2L * node + 1L)
113 lazy_pull(tree, node)
114 }
115 }
116 build(1L)
117 tree
118}
119lazy_apply <- function(tree, node, value, assignment) {
120 change <- c(value * (tree$right[node] - tree$left[node] + 1L), value, value)
121 tree$value[node, ] <- if (assignment) change else tree$value[node, ] + change
122 if (tree$left[node] < tree$right[node]) {
123 if (assignment) {
124 tree$assign[node] <- value
125 tree$add[node] <- 0
126 } else tree$add[node] <- tree$add[node] + value
127 }
128}
129lazy_push <- function(tree, node, count) {
130 assignment <- tree$assign[node]
131 addition <- tree$add[node]
132 if (is.na(assignment) && addition == 0) return(invisible(NULL))
133 count$pushes <- count$pushes + 1L
134 for (child in c(2L * node, 2L * node + 1L)) {
135 if (!is.na(assignment)) lazy_apply(tree, child, assignment, TRUE)
136 if (addition != 0) lazy_apply(tree, child, addition, FALSE)
137 }
138 tree$assign[node] <- NA_real_
139 tree$add[node] <- 0
140}
141lazy_update <- function(tree, node, left, right, value, assignment, count) {
142 count$visited <- count$visited + 1L
143 nl <- tree$left[node]
144 nr <- tree$right[node]
145 if (left <= nl && nr <= right) {
146 count$covered <- count$covered + 1L
147 lazy_apply(tree, node, value, assignment)
148 return(invisible(NULL))
149 }
150 lazy_push(tree, node, count)
151 middle <- (nl + nr) %/% 2L
152 if (left <= middle) lazy_update(tree, 2L * node, left, right, value, assignment, count)
153 if (right > middle) lazy_update(tree, 2L * node + 1L, left, right, value, assignment, count)
154 lazy_pull(tree, node)
155}
156lazy_query <- function(tree, node, left, right, count) {
157 count$visited <- count$visited + 1L
158 nl <- tree$left[node]
159 nr <- tree$right[node]
160 if (left <= nl && nr <= right) return(tree$value[node, ])
161 lazy_push(tree, node, count)
162 middle <- (nl + nr) %/% 2L
163 if (right <= middle) return(lazy_query(tree, 2L * node, left, right, count))
164 if (left > middle) return(lazy_query(tree, 2L * node + 1L, left, right, count))
165 first <- lazy_query(tree, 2L * node, left, right, count)
166 second <- lazy_query(tree, 2L * node + 1L, left, right, count)
167 combine(first, second)
168}

192bit_lower <- function(tree, remaining, count) {
193 n <- length(tree)
194 step <- 1L
195 while (2L * step <= n) step <- 2L * step
196 position <- 0L
197 while (step > 0L) {
198 if (position + step <= n) {
199 count$visited <- count$visited + 1L
200 if (tree[position + step] < remaining) {
201 remaining <- remaining - tree[position + step]
202 position <- position + step
203 }
204 }
205 step <- step %/% 2L
206 }
207 if (position == n) "none" else number(position)
208}
209rfen_prefix <- function(tree, position, count) {

211 second <- bit_prefix(tree$second, position, count)
212 position * first - second

214new_sparse <- function(values) {
215 n <- length(values)
216 tree <- record(values = values, minima = list(seq_len(n) - 1L), maxima = list(seq_len(n) - 1L))
217 width <- 2L
218 while (width <= n) {
219 level <- length(tree$minima)
220 positions <- seq_len(n - width + 1L)
221 a <- tree$minima[[level]][positions]
222 b <- tree$minima[[level]][positions + width %/% 2L]
223 tree$minima[[level + 1L]] <- ifelse(values[a + 1L] <= values[b + 1L], a, b)
224 a <- tree$maxima[[level]][positions]
225 b <- tree$maxima[[level]][positions + width %/% 2L]
226 tree$maxima[[level + 1L]] <- ifelse(values[a + 1L] >= values[b + 1L], a, b)
227 width <- 2L * width
228 }
229 tree
230}
231sparse_query <- function(tree, left, right) {
232 level <- 0L
233 width <- 1L
234 while (2L * width <= right - left + 1L) {
235 width <- 2L * width
236 level <- level + 1L
237 }
238 end <- right - width + 1L
239 pair <- tree$minima[[level + 1L]][c(left, end) + 1L]
240 low <- if (tree$values[pair[1L] + 1L] <= tree$values[pair[2L] + 1L]) pair[1L] else pair[2L]
241 pair <- tree$maxima[[level + 1L]][c(left, end) + 1L]
242 high <- if (tree$values[pair[1L] + 1L] >= tree$values[pair[2L] + 1L]) pair[1L] else pair[2L]
243 values <- tree$values[c(low, high) + 1L]
244 list(text = paste0("min=", number(values[1L]), " minpos=", number(low), " max=", number(values[2L]),
245 " maxpos=", number(high), " level=", level), values = values)
246}
247new_sqrt <- function(values, block) {
248 groups <- (seq_along(values) - 1L) %/% block
249 sums <- as.numeric(rowsum(values, groups, reorder = FALSE))
250 record(values = values, block = block, sums = sums, tags = numeric(length(sums)))
251}
252split_blocks <- function(tree, left, right) {
253 first <- left %/% tree$block
254 last <- right %/% tree$block
255 if (first == last) cells <- indices(left, right) else {
256 cells <- c(indices(left, (first + 1L) * tree$block - 1L), indices(last * tree$block, right))
257 }
258 list(cells = cells, blocks = indices(first + 1L, last - 1L))
259}
260sqrt_update <- function(tree, left, right, value) {
261 split <- split_blocks(tree, left, right)
262 for (index in split$cells) {
263 block <- index %/% tree$block + 1L
264 tree$values[index + 1L] <- tree$values[index + 1L] + value
265 tree$sums[block] <- tree$sums[block] + value
266 }
267 blocks <- split$blocks + 1L
268 tree$tags[blocks] <- tree$tags[blocks] + value
269 tree$sums[blocks] <- tree$sums[blocks] + value * tree$block
270 split
271}
272sqrt_query <- function(tree, left, right) {
273 split <- split_blocks(tree, left, right)
274 value <- sum(tree$values[split$cells + 1L] + tree$tags[split$cells %/% tree$block + 1L])
275 value <- value + sum(tree$sums[split$blocks + 1L])
276 list(text = paste0("sum=", number(value), " whole=", length(split$blocks), " cells=", length(split$cells)), values = value)
277}

280pool_append <- function(pool, left, right, value) {
281 pool$size <- pool$size + 1L
282 node <- pool$size
283 pool$left[node] <- left
284 pool$right[node] <- right
285 pool$values[node] <- value
286 node
287}
288pool_build <- function(pool, values, left, right) {
289 if (left == right) return(pool_append(pool, 0L, 0L, values[left + 1L]))
290 middle <- (left + right) %/% 2L
291 a <- pool_build(pool, values, left, middle)
292 b <- pool_build(pool, values, middle + 1L, right)
293 pool_append(pool, a, b, pool$values[a] + pool$values[b])
294}
295pool_fork <- function(pool, source, left, right, index, value, increment) {
296 a <- pool$left[source]
297 b <- pool$right[source]
298 if (left == right) {
299 if (increment) value <- pool$values[source] + value
300 return(pool_append(pool, 0L, 0L, value))
301 }
302 middle <- (left + right) %/% 2L
303 if (index <= middle) {
304 a <- pool_fork(pool, a, left, middle, index, value, increment)
305 } else b <- pool_fork(pool, b, middle + 1L, right, index, value, increment)
306 pool_append(pool, a, b, pool$values[a] + pool$values[b])
307}
308pool_query <- function(pool, root, nl, nr, left, right, count) {
309 count$visited <- count$visited + 1L
310 if (left <= nl && nr <= right) return(pool$values[root])
311 middle <- (nl + nr) %/% 2L
312 if (right <= middle) return(pool_query(pool, pool$left[root], nl, middle, left, right, count))
313 if (left > middle) return(pool_query(pool, pool$right[root], middle + 1L, nr, left, right, count))
314 a <- pool_query(pool, pool$left[root], nl, middle, left, right, count)
315 b <- pool_query(pool, pool$right[root], middle + 1L, nr, left, right, count)
316 a + b
317}
318new_persistent <- function(values) {
319 tree <- record(n = length(values), position = new_pool(), value = new_pool(), domain = sort(unique(values)))
320 tree$roots <- pool_build(tree$position, values, 0L, tree$n - 1L)
321 m <- length(tree$domain)
322 tree$prefixes <- pool_build(tree$value, numeric(m), 0L, m - 1L)
323 for (i in seq_along(values)) {
324 rank <- match(values[i], tree$domain) - 1L
325 tree$prefixes[i + 1L] <- pool_fork(tree$value, tree$prefixes[i], 0L, m - 1L, rank, 1, TRUE)
326 }
327 tree

329persistent_kth <- function(tree, left, right, k, count) {
330 upper <- tree$prefixes[right + 2L]
331 lower <- tree$prefixes[left + 1L]
332 left <- 0L
333 right <- length(tree$domain) - 1L
334 count$visited <- 1L
335 pool <- tree$value
336 while (left < right) {
337 middle <- (left + right) %/% 2L
338 a <- pool$left[upper]
339 b <- pool$left[lower]
340 size <- pool$values[a] - pool$values[b]
341 if (k <= size) {
342 upper <- a
343 lower <- b
344 right <- middle
345 } else {
346 k <- k - size
347 upper <- pool$right[upper]
348 lower <- pool$right[lower]
349 left <- middle + 1L
350 }
351 count$visited <- count$visited + 1L
352 }
353 tree$domain[left + 1L]
354}

404 RFEN = paste0("RFEN BUILD n=", n, " total=", number(sum(values))),
405 SPARSE = paste0("SPARSE BUILD levels=", length(tree$minima), " entries=", sum(lengths(tree$minima))),
406 SQRT = paste0("SQRT BUILD block=", number(tree$block), " blocks=", length(tree$tags)),

409query_array <- function(lab, kind, left, right, version = 0L) {
410 tree <- lab$structures[[kind]]
411 count <- counts()
412 if (kind == "SPARSE") return(sparse_query(tree, left, right))
413 if (kind == "SQRT") return(sqrt_query(tree, left, right))
414 if (kind == "SEG") {
415 value <- segment_query(tree, 1L, left, right, count)
416 return(list(text = paste0(tolower(tree$op), "=", number(value), " visited=", count$visited), values = value))
417 }
418 if (kind == "LAZY") {
419 value <- lazy_query(tree, 1L, left, right, count)
420 return(list(text = paste0(aggregate_text(value), " visited=", count$visited, " pushes=", count$pushes), values = value))
421 }
422 if (kind == "FEN") {
423 first <- bit_prefix(tree$tree, right + 1L, count)
424 second <- bit_prefix(tree$tree, left, count)
425 value <- first - second
426 } else if (kind == "RFEN") {
427 first <- rfen_prefix(tree, right + 1L, count)
428 second <- rfen_prefix(tree, left, count)
429 value <- first - second
430 } else value <- pool_query(tree$position, tree$roots[version + 1L], 0L, tree$n - 1L, left, right, count)
431 list(text = paste0("sum=", number(value), " visited=", count$visited), values = value)
432}

471check_all <- function(lab, left, right) {
472 values <- lab$values[indices(left, right) + 1L]
473 direct <- aggregate(values)
474 common <- Reduce(gcd, values, init = 0)
475 rows <- paste0("CHECK ", numbers(c(left, right)), " ", aggregate_text(direct), " gcd=", number(common))
476 ok <- different <- 0L
477 for (kind in kinds) {
478 tree <- lab$structures[[kind]]
479 if (is.null(tree)) next
480 expected <- direct[1L]
481 if (kind == "SEG") {
482 expected <- c(direct, common)[match(tree$op, operations)]
483 } else if (kind == "LAZY") expected <- direct else if (kind == "SPARSE") expected <- direct[2:3]
484 version <- if (kind == "PERSIST") length(tree$roots) - 1L else 0L
485 answer <- query_array(lab, kind, left, right, version)
486 same <- all(answer$values == expected)
487 if (same) ok <- ok + 1L else different <- different + 1L
488 prefix <- if (kind == "PERSIST") paste0("version=", version, " ") else ""
489 rows <- c(rows, paste(kind, if (same) "OK" else "DIFF", paste0(prefix, answer$text)))
490 }
491 c(rows, paste0("RESULT ok=", ok, " diff=", different))
492}
493mo <- function(values, queries) {
494 block <- floor(sqrt(length(values)))
495 groups <- queries[, 1L] %/% block
496 direction <- ifelse(groups %% 2L == 0L, queries[, 2L], -queries[, 2L])
497 order <- order(groups, direction, seq_len(nrow(queries)), method = "radix")
498 ranks <- match(values, unique(values))
499 frequency <- integer(max(ranks))
500 answers <- integer(nrow(queries))
501 left <- 0L
502 right <- -1L
503 moves <- 0L
504 for (index in order) {
505 target <- queries[index, ]
506 moves <- moves + abs(target[1L] - left) + abs(target[2L] - right)
507 while (left > target[1L]) {
508 left <- left - 1L
509 rank <- ranks[left + 1L]
510 frequency[rank] <- frequency[rank] + 1L
511 }
512 while (right < target[2L]) {
513 right <- right + 1L
514 rank <- ranks[right + 1L]
515 frequency[rank] <- frequency[rank] + 1L
516 }
517 while (left < target[1L]) {
518 rank <- ranks[left + 1L]
519 frequency[rank] <- frequency[rank] - 1L
520 left <- left + 1L
521 }
522 while (right > target[2L]) {
523 rank <- ranks[right + 1L]
524 frequency[rank] <- frequency[rank] - 1L
525 right <- right - 1L
526 }
527 answers[index] <- sum(frequency > 0L)
528 }
529 rows <- c(paste0("MO queries=", nrow(queries), " block=", number(block), " moves=", number(moves)), paste("ORDER", numbers(order)))
530 for (index in seq_len(nrow(queries))) rows <- c(rows, paste0("ANSWER ", index, " ", numbers(queries[index, ]), " distinct=", answers[index]))
531 rows
532}
533execute <- function(lab, line) {
534 words <- strsplit(line, " ", fixed = TRUE)[[1L]]
535 command <- words[1L]
536 n <- length(lab$values)
537 if (command == "LOAD") {
538 removed <- sum(!vapply(lab$structures, is.null, logical(1L)))
539 lab$values <- lab$original <- as.numeric(words[-1L])
540 lab$structures <- setNames(vector("list", length(kinds)), kinds)
541 return(paste0("LOAD n=", length(lab$values), " removed=", removed))
542 }
543 if (command == "RESET") {
544 lab$values <- lab$original
545 rebuilt <- 0L
546 for (kind in kinds) {
547 tree <- lab$structures[[kind]]
548 if (is.null(tree)) next
549 parameter <- if (kind == "SEG") tree$op else if (kind == "SQRT") tree$block else NULL
550 build_array(lab, kind, parameter)
551 rebuilt <- rebuilt + 1L
552 }
553 return(paste0("RESET rebuilt=", rebuilt))
554 }
555 if (command %in% c("ARRAY", "CHECK")) {
556 args <- as.numeric(words[-1L])
557 check_range(n, args[1L], args[2L])
558 if (command == "CHECK") return(check_all(lab, args[1L], args[2L]))
559 return(paste(line, numbers(lab$values[indices(args[1L], args[2L]) + 1L])))
560 }
561 if (command == "MO") {
562 queries <- matrix(as.numeric(words[-1L]), ncol = 2L, byrow = TRUE)
563 for (index in seq_len(nrow(queries))) check_range(n, queries[index, 1L], queries[index, 2L])
564 return(mo(lab$values, queries))
565 }
566 kind <- command
567 action <- words[2L]
568 if (action == "BUILD") {
569 parameter <- NULL
570 if (kind == "SEG") {
571 parameter <- words[3L]
572 if (!parameter %in% operations) fail(paste("BAD OP", parameter))
573 } else if (kind == "SQRT") {
574 parameter <- if (length(words) == 3L) as.numeric(words[3L]) else floor(sqrt(n))
575 if (parameter > n) fail(paste("BAD BLOCK", number(parameter)))
576 } else if (kind == "GRID") {
577 height <- as.numeric(words[3L])
578 width <- as.numeric(words[4L])
579 actual <- length(words) - 4L
580 if (actual != height * width) fail(paste("BAD COUNT", actual, number(height * width)))
581 values <- as.numeric(words[-seq_len(4L)])
582 lab$grid <- new_grid(height, width, values)
583 return(paste0("GRID BUILD rows=", number(height), " cols=", number(width), " total=", number(sum(values))))
584 }
585 return(build_array(lab, kind, parameter))
586 }
587 tree <- if (kind == "GRID") lab$grid else lab$structures[[kind]]
588 if (is.null(tree)) fail(paste("NOT BUILT", kind))
589 if (action == "DUMP") return(dump_structure(lab, kind))
590 args <- as.numeric(words[-c(1L, 2L)])
591 count <- counts()
592 if (kind == "GRID") {
593 check_cell(tree, args[1L], args[2L])
594 if (action == "ADD") {
595 grid_add(tree, args[1L], args[2L], args[3L], count)
596 return(paste0(line, " visited=", count$visited))
597 }
598 check_cell(tree, args[3L], args[4L])
599 if (args[1L] > args[3L] || args[2L] > args[4L]) fail(paste("BAD RECT", numbers(args)))
600 first <- grid_prefix(tree, args[3L] + 1L, args[4L] + 1L, count)
601 second <- grid_prefix(tree, args[1L], args[4L] + 1L, count)
602 third <- grid_prefix(tree, args[3L] + 1L, args[2L], count)
603 fourth <- grid_prefix(tree, args[1L], args[2L], count)
604 return(paste0(line, " sum=", number(first - second - third + fourth), " visited=", count$visited))
605 }
606 if (kind == "PERSIST") {
607 if (action == "NODES") return(paste0(line, " versions=", length(tree$roots), " position=", tree$position$size, " value=", tree$value$size))
608 if (action == "KTH") {
609 check_range(n, args[1L], args[2L])
610 if (args[3L] < 1 || args[3L] > args[2L] - args[1L] + 1L) fail(paste("BAD K", number(args[3L])))
611 value <- persistent_kth(tree, args[1L], args[2L], args[3L], count)
612 return(paste0(line, " value=", number(value), " visited=", count$visited))
613 }
614 version <- args[1L]
615 check_version(tree, version)
616 if (action == "SET") {
617 check_index(n, args[2L])
618 before <- tree$position$size
619 new_version <- length(tree$roots)
620 root <- pool_fork(tree$position, tree$roots[version + 1L], 0L, n - 1L, args[2L], args[3L], FALSE)
621 tree$roots <- c(tree$roots, root)
622 return(paste0(line, " version=", new_version, " created=", tree$position$size - before))
623 }
624 check_range(n, args[2L], args[3L])
625 return(paste(line, query_array(lab, kind, args[2L], args[3L], version)$text))
626 }
627 if (kind == "FEN" && action == "LOWER") {
628 negative <- which(tree$values < 0)
629 if (length(negative)) fail(paste("NEGATIVE", negative[1L] - 1L))
630 index <- bit_lower(tree$tree, args[1L], count)
631 return(paste0(line, " index=", index, " visited=", count$visited))
632 }
633 if (kind == "SEG" && action == "UPDATE") {
634 check_index(n, args[1L])
635 visited <- segment_update(tree, 1L, args[1L], args[2L])
636 lab$values[args[1L] + 1L] <- args[2L]
637 return(paste0(line, " root=", number(tree$value[1L]), " visited=", visited))
638 }
639 if (kind == "FEN" && action %in% c("ADD", "PREFIX")) {
640 check_index(n, args[1L])
641 index <- args[1L] + 1L
642 if (action == "ADD") {
643 tree$tree <- bit_add(tree$tree, index, args[2L], count)
644 tree$values[index] <- tree$values[index] + args[2L]
645 lab$values[index] <- lab$values[index] + args[2L]
646 return(paste0(line, " visited=", count$visited))
647 }
648 value <- bit_prefix(tree$tree, index, count)
649 return(paste0(line, " sum=", number(value), " visited=", count$visited))
650 }
651 left <- args[1L]
652 right <- args[2L]
653 check_range(n, left, right)
654 if (action %in% c("ADD", "ASSIGN")) {
655 value <- args[3L]
656 assignment <- action == "ASSIGN"
657 if (kind == "LAZY") {
658 lazy_update(tree, 1L, left, right, value, assignment, count)
659 details <- paste0("visited=", count$visited, " covered=", count$covered, " pushes=", count$pushes)
660 } else if (kind == "RFEN") {
661 tree$first <- bit_add(tree$first, left + 1L, value, count)
662 tree$first <- bit_add(tree$first, right + 2L, -value, count)
663 tree$second <- bit_add(tree$second, left + 1L, value * left, count)
664 tree$second <- bit_add(tree$second, right + 2L, -value * (right + 1L), count)
665 details <- paste0("visited=", count$visited)
666 } else {
667 split <- sqrt_update(tree, left, right, value)
668 details <- paste0("whole=", length(split$blocks), " cells=", length(split$cells))
669 }
670 selected <- indices(left, right) + 1L
671 lab$values[selected] <- if (assignment) value else lab$values[selected] + value
672 return(paste(line, details))
673 }
674 paste(line, query_array(lab, kind, left, right)$text)
675}

677input <- file("stdin", open = "r")
678initial_size <- as.integer(readLines(input, n = 1L, warn = FALSE))
679initial_values <- as.numeric(strsplit(readLines(input, n = 1L, warn = FALSE), " ", fixed = TRUE)[[1L]])

684 rows <- tryCatch(execute(lab, line), command_error = function(problem) paste("ERROR", problem$message))
685 cat(paste0(rows, "\n"), sep = "")
686}
687close(input)
