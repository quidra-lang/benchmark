5private fun displayWord(word: String) = word.ifEmpty {

8private data class Transition(val from: Int, val symbol: Char, val to: Int)

10private data class Automaton(val dfa: Boolean, val states: List<String>, val alphabet: List<Char>, val start: Int, val accepting: Set<Int>, val transitions: Set<Transition>) {
11 private val adjacency = transitions.groupBy {
12 it.from to it.symbol
13 }.mapValues {
14 (_, ts) -> ts.map {
15 it.to
16 }.toSet()
17 }
18 fun direct(p: Int, c: Char) = adjacency[p to c] ?: emptySet()
19
20 fun delta(p: Int, c: Char) = direct(p, c).first()
21
22 fun summary(name: String) = "${if (dfa) "DFA" else "NFA"} $name: alphabet=${alphabet.joinToString("")} states=${states.size} accepting=${accepting.size} transitions=${transitions.size}"

24 fun closure(set: Set<Int>): Set<Int> {
25 val result = set.toMutableSet()
26 val queue = ArrayDeque<Int>()
27 queue.addAll(set)
28 while (queue.isNotEmpty()) for (q in direct(queue.removeFirst(), 'e')) if (result.add(q)) queue.add(q)
29 return result
30 }
31 fun initial() = closure(setOf(start))
32
33 fun destination(set: Set<Int>, c: Char) = closure(set.flatMap {
34 direct(it, c)
35 }.toSet())
36
37 fun accepted(set: Set<Int>) = set.any {
38 it in accepting
39 }
40 fun setText(set: Set<Int>) = set.sorted().joinToString(",", "{", "}") {
41 states[it]
42 }
43 fun listText(set: Set<Int>) = set.sorted().joinToString(",") {
44 states[it]
45 }.ifEmpty {
46 "none"
47 }
48 fun determinize(): Pair<Automaton, List<String>> {
49 val sets = mutableListOf(initial())
50 val index = mutableMapOf(sets[0] to 0)
51 val transitions = mutableSetOf<Transition>()
52 var i = 0
53 while (i < sets.size) {
54 for (c in alphabet) {
55 val next = destination(sets[i], c)
56 val j = index[next] ?: sets.size.also {
57 index[next] = it
58 sets.add(next)
59 }
60 transitions.add(Transition(i, c, j))
61 }
62 i++
63 }
64 val result = generated(true, "D", sets.size, alphabet, 0, sets.indices.filter {
65 accepted(sets[it])
66 }.toSet(), transitions)
67 return result to sets.indices.map {
68 " D$it = ${setText(sets[it])}"
69 }
70 }
71 fun reachable(reverse: Boolean): Set<Int> {
72 val result = (if (reverse) accepting else setOf(start)).toMutableSet()
73 val queue = ArrayDeque<Int>()
74 queue.addAll(result)
75 while (queue.isNotEmpty()) {
76 val s = queue.removeFirst()
77 for (t in transitions) if ((if (reverse) t.to else t.from) == s) {
78 val next = if (reverse) t.from else t.to
79 if (result.add(next)) queue.add(next)
80 }
81 }
82 return result
83 }
84 fun minimize(): Pair<Automaton, List<String>> {
85 val reachable = reachable(false)
86 val details = mutableListOf(" unreachable: ${listText(states.indices.filter { it !in reachable }.toSet())}")
87 var blocks = listOf(true, false).map {
88 accept -> reachable.filter {
89 (it in accepting) == accept
90 }.toSet()
91 }.filter {
92 it.isNotEmpty()
93 }.sortedBy {
94 it.min()
95 }
96 fun partition(bs: List<Set<Int>>) = bs.joinToString(" ") {
97 setText(it)
98 }
99 details.add(" round 0: ${partition(blocks)}")
100 var round = 1
101 while (true) {
102 val group = IntArray(states.size)
103 for ((i, block) in blocks.withIndex()) for (s in block) group[s] = i
104 val split = reachable.sorted().groupBy {
105 s -> listOf(group[s]) + alphabet.map {
106 group[delta(s, it)]
107 }
108 }
109 val next = split.values.map {
110 it.toSet()
111 }.sortedBy {
112 it.min()
113 }
114 if (next.size == blocks.size) break
115 blocks = next
116 details.add(" round $round: ${partition(blocks)}")
117 round++
118 }
119 val group = IntArray(states.size)
120 for ((i, block) in blocks.withIndex()) for (s in block) group[s] = i
121 val sequence = mutableListOf(group[start])
122 val names = mutableMapOf(sequence[0] to 0)
123 val transitions = mutableSetOf<Transition>()
124 var i = 0
125 while (i < sequence.size) {
126 val s = blocks[sequence[i]].min()
127 for (c in alphabet) {
128 val b = group[delta(s, c)]
129 val j = names[b] ?: sequence.size.also {
130 names[b] = it
131 sequence.add(b)
132 }
133 transitions.add(Transition(i, c, j))
134 }
135 i++
136 }
137 val accepting = sequence.indices.filter {
138 accepted(blocks[sequence[it]])
139 }.toSet()
140 for (j in sequence.indices) details.add(" M$j = ${setText(blocks[sequence[j]])}")
141 return generated(true, "M", sequence.size, alphabet, 0, accepting, transitions) to details
142 }
143 fun product(other: Automaton, command: String): Pair<Automaton, List<String>> {
144 val pairs = mutableListOf(start to other.start)
145 val index = mutableMapOf(pairs[0] to 0)
146 val transitions = mutableSetOf<Transition>()
147 var i = 0
148 while (i < pairs.size) {
149 for (c in alphabet) {
150 val pair = delta(pairs[i].first, c) to other.delta(pairs[i].second, c)
151 val j = index[pair] ?: pairs.size.also {
152 index[pair] = it
153 pairs.add(pair)
154 }
155 transitions.add(Transition(i, c, j))
156 }
157 i++
158 }
159 val accepting = pairs.indices.filter {
160 j ->
161 val (p, q) = pairs[j]
162 val a = p in this.accepting
163 val b = q in other.accepting
164 when (command) {
165 "UNION" -> a || b
166 "INTERSECT" -> a && b
167 else -> a && !b
168 }
169 }.toSet()
170 return generated(true, "P", pairs.size, alphabet, 0, accepting, transitions) to pairs.indices.map {
171 j -> " P$j = (${states[pairs[j].first]},${other.states[pairs[j].second]})"
172 }
173 }
174 fun smallest(): String? {
175 val seen = mutableSetOf(start)
176 val queue = ArrayDeque<Pair<Int, String>>()
177 queue.add(start to "")
178 while (queue.isNotEmpty()) {
179 val (s, word) = queue.removeFirst()
180 if (s in accepting) return word
181 for (c in alphabet) {
182 val q = delta(s, c)
183 if (seen.add(q)) queue.add(q to (word + c))
184 }
185 }
186 return null
187 }
188 fun language(counting: Boolean = true): Long? {
189 val live = reachable(true)
190 val color = IntArray(states.size)
191 val counts = LongArray(states.size)
192
193 fun visit(s: Int): Long? {
194 if (s !in live) return 0
195 if (color[s] == 1) return null
196 if (color[s] == 2) return counts[s]
197 color[s] = 1
198 var count = if (s in accepting) 1L else 0L
199 for (c in alphabet) {
200 val n = visit(delta(s, c)) ?: return null
201 if (counting) count += n
202 }
203 color[s] = 2
204 counts[s] = if (counting) count else 1L
205 return counts[s]
206 }
207 return visit(start)
208 }
209 fun count(n: Int): Long {
210 var counts = LongArray(states.size)
211 counts[start] = 1
212 repeat(n) {
213 val next = LongArray(states.size)
214 for (s in states.indices) for (c in alphabet) next[delta(s, c)] += counts[s]
215 counts = next
216 }
217 return accepting.sumOf {
218 counts[it]
219 }
220 }
221 fun enumerate(limit: Int): String {
222 val queue = ArrayDeque<Pair<Int, String>>()
223 queue.add(start to "")
224 var count = 0
225 val first = mutableListOf<String>()
226 while (queue.isNotEmpty()) {
227 val (s, word) = queue.removeFirst()
228 if (s in accepting) {
229 count++
230 if (first.size < 50) first.add(displayWord(word))
231 }
232 if (word.length < limit) for (c in alphabet) queue.add(delta(s, c) to (word + c))
233 }
234 return "$count" + (if (first.isEmpty()) "" else " " + first.joinToString(" ")) + if (count > 50) " ..." else ""
235 }
236 fun synchronize(): String {
237 val initial = states.indices.toSet()
238 val seen = mutableSetOf(initial)
239 val queue = ArrayDeque<Pair<Set<Int>, String>>()
240 queue.add(initial to "")
241 while (queue.isNotEmpty()) {
242 val (set, word) = queue.removeFirst()
243 if (set.size == 1) return "SYNCHRONIZING ${displayWord(word)} -> ${states[set.first()]}"
244 for (c in alphabet) {
245 val next = set.map {
246 delta(it, c)
247 }.toSet()
248 if (seen.add(next)) queue.add(next to (word + c))
249 }
250 }
251 return "NOT SYNCHRONIZING"
252 }
253 fun table(): List<String> {
254 val symbols = alphabet + if (!dfa && transitions.any {
255 it.symbol == 'e'
256 })
257 listOf('e') else emptyList()
258 val rows = mutableListOf(listOf("") + symbols.map {
259 it.toString()
260 })
261 for (p in states.indices) {
262 val label = (if (p == start) "->" else " ") + (if (p in accepting) "*" else " ") + states[p]
263 rows.add(listOf(label) + symbols.map {
264 c -> val set = direct(p, c)
265 if (dfa) states[set.first()] else if (set.isEmpty()) "-" else setText(set)
266 })
267 }
268 val widths = rows[0].indices.map {
269 i -> rows.maxOf {
270 it[i].length
271 }
272 }
273 return rows.map {
274 row -> row.indices.joinToString(" ") {
275 row[it].padEnd(widths[it])
276 }.trimEnd()
277 }
278 }
279 fun show(name: String): String {
280 val words = mutableListOf(if (dfa) "DFA" else "NFA", name, states.joinToString(","), alphabet.joinToString(""), states[start], if (accepting.isEmpty()) "-" else listText(accepting))
281 for (p in states.indices) for (c in alphabet + 'e') for (q in direct(p, c).sorted()) words.add("${states[p]}:$c:${states[q]}")
282 return words.joinToString(" ")
283 }

290private class Regular(val kind: Char, val children: List<Regular> = emptyList(), val symbol: Char = 'e')
291
292private class RegexParser(val text: String) {
293 var at = 0
294
295 fun peek() = text.getOrNull(at) ?: '\u0000'
296
297 fun error(): Nothing = fail("syntax error at ${at + 1}")
298
299 fun atom(): Regular {
300 val c = peek()
301 if (c in 'a'..'z' || c in '0'..'9') {
302 at++
303 return Regular('s', symbol = c)
304 }
305 if (c == '(') {
306 at++
307 val r = alternation()
308 if (peek() != ')') error()
309 at++
310 return r
311 }
312 error()
313 }
314 fun factor(): Regular {
315 var r = atom()
316 while (peek() in "*+?") {
317 val c = peek()
318 at++
319 r = Regular(c, listOf(r))
320 }
321 return r
322 }
323 fun sequence(): Regular {
324 val list = mutableListOf(factor())
325 while (peek() in 'a'..'z' || peek() in '0'..'9' || peek() == '(') list.add(factor())
326 return if (list.size == 1) list[0] else Regular('.', list)
327 }
328 fun alternation(): Regular {
329 val list = mutableListOf(sequence())
330 while (peek() == '|') {
331 at++
332 list.add(sequence())
333 }
334 return if (list.size == 1) list[0] else Regular('|', list)
335 }
336}
337private fun regexAutomaton(alphabet: List<Char>, text: String): Automaton {
338 for (c in text) if ((c in 'a'..'z' || c in '0'..'9') && c != 'e' && c !in alphabet) fail("unknown symbol $c")
339 val parser = RegexParser(text)
340 val tree = parser.alternation()
341 if (parser.at != text.length) parser.error()
342 var size = 0
343 val transitions = mutableSetOf<Transition>()
344
345 fun state() = size++
346
347 fun build(r: Regular): Pair<Int, Int> {
348 if (r.kind == 's') {
349 val s = state()
350 val f = state()
351 transitions.add(Transition(s, r.symbol, f))
352 return s to f
353 }
354 if (r.kind == '.') {
355 val parts = r.children.map(::build)
356 for ((a, b) in parts.zipWithNext()) transitions.add(Transition(a.second, 'e', b.first))
357 return parts.first().first to parts.last().second
358 }
359 val s = state()
360 if (r.kind == '|') {
361 val parts = r.children.map(::build)
362 val f = state()
363 for ((a, b) in parts) {
364 transitions.add(Transition(s, 'e', a))
365 transitions.add(Transition(b, 'e', f))
366 }
367 return s to f
368 }
369 val (a, b) = build(r.children[0])
370 val f = state()
371 transitions.add(Transition(s, 'e', a))
372 transitions.add(Transition(b, 'e', f))
373 if (r.kind != '?') transitions.add(Transition(b, 'e', a))
374 if (r.kind != '+') transitions.add(Transition(s, 'e', f))
375 return s to f
376 }
377 val (s, f) = build(tree)
378 return generated(false, "N", size, alphabet, s, setOf(f), transitions)
379}
380private class Toolkit {
381 val automata = linkedMapOf<String, Automaton>()
382
383 fun get(name: String) = automata[name] ?: fail("unknown automaton $name")
384
385 fun fresh(name: String) {
386 if (name in automata) fail("$name already exists")
387 }
388 fun register(name: String, a: Automaton, details: List<String> = emptyList()): List<String> {
389 automata[name] = a
390 return listOf(a.summary(name)) + details
391 }
392 fun define(dfa: Boolean, args: List<String>): List<String> {
393 if (args.size < 5) fail("wrong number of arguments")
394 fresh(args[0])
395 val states = mutableListOf<String>()
396 for (name in args[1].split(',')) {
397 if (name in states) fail("duplicate state $name")
398 states.add(name)
399 }
400 fun state(name: String): Int {
401 val s = states.indexOf(name)
402 if (s < 0) fail("unknown state $name")
403 return s
404 }
405 val start = state(args[3])
406 val accepting = mutableSetOf<Int>()
407 if (args[4] != "-") for (name in args[4].split(',')) {
408 val s = state(name)
409 if (!accepting.add(s)) fail("duplicate state $name")
410 }
411 val alphabet = args[2].toList().sorted()
412 val transitions = mutableSetOf<Transition>()
413 for (word in args.drop(5)) {
414 val fields = word.split(':')
415 val p = state(fields[0])
416 val c = fields[1][0]
417 if (c == 'e' && dfa) fail("epsilon transition in DFA")
418 if (c != 'e' && c !in alphabet) fail("unknown symbol $c")
419 val q = state(fields[2])
420 val t = Transition(p, c, q)
421 if (t in transitions) fail("duplicate transition $word")
422 if (dfa && transitions.any {
423 it.from == p && it.symbol == c
424 })
425 fail("conflicting transition $word")
426 transitions.add(t)
427 }
428 if (dfa) for (p in states.indices) for (c in alphabet) if (transitions.none {
429 it.from == p && it.symbol == c
430 })
431 fail("missing transition ${states[p]}:$c")
432 return register(args[0], Automaton(dfa, states, alphabet, start, accepting, transitions))
433 }
434 fun command(line: String): List<String> {
435 val fields = line.split(' ')
436 val c = fields[0]
437 val args = fields.drop(1)
438 val count = when (c) {
439 "DFA", "NFA" -> return define(c == "DFA", args)
440 "REGEX" -> {
441 if (args.size != 3) fail("wrong number of arguments")
442 fresh(args[0])
443 return register(args[0], regexAutomaton(args[1].toList().sorted(), args[2]))
444 }
445 "ACCEPT", "CLOSURE", "TODFA", "EPSFREE", "MINIMIZE", "EQUIV", "COMPLEMENT", "REVERSE", "STAR", "ENUM", "COUNT" -> 2
446 "UNION", "INTERSECT", "DIFF", "CONCAT" -> 3
447 "EMPTY", "FINITE", "INFO", "TABLE", "SHOW", "SYNC", "DELETE" -> 1
448 "LIST" -> 0
449 else -> fail("unknown command $c")
450 }
451 if (args.size != count) fail("wrong number of arguments")
452 if (c == "LIST") return automata.map {
453 (name, a) -> a.summary(name)
454 }.ifEmpty {
455 listOf("(none)")
456 }
457 val a = get(args[0])
458 val two = c in setOf("EQUIV", "UNION", "INTERSECT", "DIFF", "CONCAT")
459 val b = if (two) get(args[1]) else null
460 val creates = c in setOf("TODFA", "EPSFREE", "MINIMIZE", "UNION", "INTERSECT", "DIFF", "COMPLEMENT", "REVERSE", "CONCAT", "STAR")
461 if (creates) fresh(args.last())
462 if (c in setOf("MINIMIZE", "COMPLEMENT", "UNION", "INTERSECT", "DIFF", "SYNC")) {
463 if (!a.dfa) fail("${args[0]} is not a DFA")
464 if (two && !b!!.dfa) fail("${args[1]} is not a DFA")
465 }
466 if (two && a.alphabet != b!!.alphabet) fail("alphabets differ")
467 when (c) {
468 "DELETE" -> {
469 automata.remove(args[0])
470 return listOf("deleted ${args[0]}")
471 }
472 "ACCEPT" -> {
473 val word = if (args[1] == "e") "" else args[1]
474 for (ch in word) if (ch !in a.alphabet) fail("unknown symbol $ch")
475 var set = a.initial()
476
477 fun text(s: Set<Int>) = if (a.dfa) a.states[s.first()] else a.setText(s)
478 var path = text(set)
479 for (ch in word) {
480 set = a.destination(set, ch)
481 path += " -$ch-> ${text(set)}"
482 }
483 return listOf("${if (a.accepted(set)) "ACCEPT" else "REJECT"} $path")
484 }
485 "CLOSURE" -> {
486 val set = mutableSetOf<Int>()
487 for (name in args[1].split(',')) {
488 val s = a.states.indexOf(name)
489 if (s < 0) fail("unknown state $name")
490 set.add(s)
491 }
492 return listOf(a.setText(a.closure(set)))
493 }
494 "TABLE" -> return a.table()
495 "SHOW" -> return listOf(a.show(args[0]))
496 "SYNC" -> return listOf(a.synchronize())
497 "EQUIV" -> {
498 val x = a.determinize().first
499 val z = b!!.determinize().first
500 val first = x.start to z.start
501 val seen = mutableSetOf(first)
502 val queue = ArrayDeque<Pair<Pair<Int, Int>, String>>()
503 queue.add(first to "")
504 while (queue.isNotEmpty()) {
505 val (pair, word) = queue.removeFirst()
506 val (p, q) = pair
507 val xp = p in x.accepting
508 val zq = q in z.accepting
509 if (xp != zq) return listOf("NOT EQUIVALENT: ${displayWord(word)} is accepted only by ${if (xp) args[0] else args[1]}")
510 for (ch in x.alphabet) {
511 val next = x.delta(p, ch) to z.delta(q, ch)
512 if (seen.add(next)) queue.add(next to (word + ch))
513 }
514 }
515 return listOf("EQUIVALENT")
516 }
517 }
518 if (c in setOf("ENUM", "COUNT", "EMPTY", "FINITE", "INFO")) {
519 val d = a.determinize().first
520 return when (c) {
521 "ENUM" -> listOf(d.enumerate(args[1].toInt()))
522 "COUNT" -> listOf(d.count(args[1].toInt()).toString())
523 "EMPTY" -> listOf(d.smallest()?.let {
524 "NOT EMPTY ${displayWord(it)}"
525 } ?: "EMPTY")
526 "FINITE" -> listOf(d.language()?.let {
527 "FINITE $it"
528 } ?: "INFINITE")
529 else -> {
530 val deterministic = a.transitions.none {
531 it.symbol == 'e'
532 } && a.states.indices.all {
533 s -> a.alphabet.all {
534 a.direct(s, it).size <= 1
535 }
536 }
537 val complete = a.states.indices.all {
538 s -> a.alphabet.all {
539 a.direct(s, it).isNotEmpty()
540 }
541 }
542 val language = when (d.language(false)) {
543 null -> "infinite"
544 0L -> "empty"
545 else -> "finite"
546 }
547 listOf(a.summary(args[0]), " epsilon transitions: ${a.transitions.count { it.symbol == 'e' }}", " reachable: ${a.listText(a.reachable(false))}", " live: ${a.listText(a.reachable(true))}", " deterministic: ${if (deterministic) "yes" else "no"}", " complete: ${if (complete) "yes" else "no"}", " language: $language", " minimal DFA states: ${d.minimize().first.states.size}")
548 }
549 }
550 }
551 val result: Automaton
552 var details = emptyList<String>()
553 when (c) {
554 "TODFA" -> {
555 val pair = a.determinize()
556 result = pair.first
557 details = pair.second
558 }
559 "MINIMIZE" -> {
560 val pair = a.minimize()
561 result = pair.first
562 details = pair.second
563 }
564 "UNION", "INTERSECT", "DIFF" -> {
565 val pair = a.product(b!!, c)
566 result = pair.first
567 details = pair.second
568 }
569 "COMPLEMENT" -> result = a.copy(accepting = a.states.indices.filter {
570 it !in a.accepting
571 }.toSet())
572 "EPSFREE" -> {
573 val accepting = mutableSetOf<Int>()
574 val transitions = mutableSetOf<Transition>()
575 for (p in a.states.indices) {
576 val closure = a.closure(setOf(p))
577 if (a.accepted(closure)) accepting.add(p)
578 for (ch in a.alphabet) for (q in a.destination(closure, ch)) transitions.add(Transition(p, ch, q))
579 }
580 result = a.copy(dfa = false, accepting = accepting, transitions = transitions)
581 }
582 "REVERSE", "STAR" -> {
583 val transitions = mutableSetOf<Transition>()
584 val accepting = mutableSetOf<Int>()
585 if (c == "REVERSE") {
586 for (t in a.transitions) transitions.add(Transition(t.to + 1, t.symbol, t.from + 1))
587 for (p in a.accepting) transitions.add(Transition(0, 'e', p + 1))
588 accepting.add(a.start + 1)
589 } else {
590 for (t in a.transitions) transitions.add(Transition(t.from + 1, t.symbol, t.to + 1))
591 transitions.add(Transition(0, 'e', a.start + 1))
592 accepting.add(0)
593 for (p in a.accepting) {
594 transitions.add(Transition(p + 1, 'e', a.start + 1))
595 accepting.add(p + 1)
596 }
597 }
598 result = generated(false, "N", a.states.size + 1, a.alphabet, 0, accepting, transitions)
599 }
600 "CONCAT" -> {
601 val z = b!!
602 val offset = a.states.size
603 val transitions = a.transitions.toMutableSet()
604 for (t in z.transitions) transitions.add(Transition(t.from + offset, t.symbol, t.to + offset))
605 for (p in a.accepting) transitions.add(Transition(p, 'e', z.start + offset))
606 result = generated(false, "N", offset + z.states.size, a.alphabet, a.start, z.accepting.map {
607 it + offset
608 }.toSet(), transitions)
609 } else -> error("unreachable command")
610 }
611 return register(args.last(), result, details)
612 }
613}
