61const Automaton = struct {
62 gpa: Allocator,
63 nfa: bool,
64 states: []const Text,
65 alphabet: Text,
66 start: usize,
67 final: []bool,
68 transitions: std.ArrayList(Transition) = .empty,
69 fn create(gpa: Allocator, nfa: bool, states: []const Text, alphabet: Text, start: usize) Error!*Automaton {
70 const result = try gpa.create(Automaton);
71 result.* = .{ .gpa = gpa, .nfa = nfa, .states = states, .alphabet = alphabet, .start = start, .final = try gpa.alloc(bool, states.len) };
72 @memset(result.final, false);
73 return result;
74 }
75 fn add(self: *Automaton, from: usize, symbol: u8, to: usize) Error!void {
76 for (self.transitions.items) |t| if (t.from == from and t.symbol == symbol and t.to == to) return;
77 try self.transitions.append(self.gpa, .{ .from = from, .symbol = symbol, .to = to });
78 }
79 fn delta(self: *const Automaton, state: usize, c: u8) usize {
80 for (self.transitions.items) |t| if (t.from == state and t.symbol == c) return t.to;
81 unreachable;
82 }
83 fn destinations(self: *const Automaton, gpa: Allocator, state: usize, c: u8) Error![]const usize {
84 const present = try gpa.alloc(bool, self.states.len);
85 @memset(present, false);
86 for (self.transitions.items) |t| if (t.from == state and t.symbol == c) {
87 present[t.to] = true;
88 };
89 return orderedSet(gpa, present);
90 }
91 fn closure(self: *const Automaton, gpa: Allocator, states: []const usize) Error![]const usize {
92 const present = try gpa.alloc(bool, self.states.len);
93 @memset(present, false);
94 var queue: std.ArrayList(usize) = .empty;
95 for (states) |state| if (!present[state]) {
96 present[state] = true;
97 try queue.append(gpa, state);
98 };
99 var i: usize = 0;
100 while (i < queue.items.len) : (i += 1) for (self.transitions.items) |t| {
101 if (t.from == queue.items[i] and t.symbol == 'e' and !present[t.to]) {
102 present[t.to] = true;
103 try queue.append(gpa, t.to);
104 }
105 };
106 return orderedSet(gpa, present);
107 }
108 fn step(self: *const Automaton, gpa: Allocator, states: []const usize, c: u8) Error![]const usize {
109 const present = try gpa.alloc(bool, self.states.len);
110 @memset(present, false);
111 for (states) |state| present[state] = true;
112 var destinationsList: std.ArrayList(usize) = .empty;
113 for (self.transitions.items) |t| if (present[t.from] and t.symbol == c) try destinationsList.append(gpa, t.to);
114 return self.closure(gpa, destinationsList.items);
115 }
116 fn accepts(self: *const Automaton, states: []const usize) bool {
117 for (states) |state| if (self.final[state]) return true;
118 return false;
119 }

147const Subsets = struct { automaton: *Automaton, sets: []const []const usize };
148fn determinize(gpa: Allocator, a: *const Automaton) Error!Subsets {
149 var sets: std.ArrayList([]const usize) = .empty;
150 try sets.append(gpa, try a.closure(gpa, &.{a.start}));
151 var indices: std.StringHashMapUnmanaged(usize) = .empty;
152 try indices.put(gpa, try setKey(gpa, sets.items[0]), 0);
153 var rows: std.ArrayList([]const usize) = .empty;
154 var i: usize = 0;
155 while (i < sets.items.len) : (i += 1) {
156 var row: std.ArrayList(usize) = .empty;
157 for (a.alphabet) |c| {
158 const next = try a.step(gpa, sets.items[i], c);
159 const key = try setKey(gpa, next);
160 const entry = try indices.getOrPut(gpa, key);
161 if (!entry.found_existing) {
162 entry.value_ptr.* = sets.items.len;
163 try sets.append(gpa, next);
164 }
165 try row.append(gpa, entry.value_ptr.*);
166 }
167 try rows.append(gpa, row.items);
168 }
169 const result = try Automaton.create(gpa, false, try series(gpa, "D", sets.items.len), a.alphabet, 0);
170 for (sets.items, rows.items, 0..) |set, row, state| {
171 result.final[state] = a.accepts(set);
172 for (a.alphabet, row) |c, next| try result.add(state, c, next);
173 }
174 return .{ .automaton = result, .sets = sets.items };
175}
176fn graphReach(gpa: Allocator, a: *const Automaton, starts: []const usize, reverse: bool) Error![]bool {
177 const present = try gpa.alloc(bool, a.states.len);
178 @memset(present, false);
179 var queue: std.ArrayList(usize) = .empty;
180 for (starts) |state| if (!present[state]) {
181 present[state] = true;
182 try queue.append(gpa, state);
183 };
184 var i: usize = 0;
185 while (i < queue.items.len) : (i += 1) for (a.transitions.items) |t| {
186 const from = if (reverse) t.to else t.from;
187 const to = if (reverse) t.from else t.to;
188 if (from == queue.items[i] and !present[to]) {
189 present[to] = true;
190 try queue.append(gpa, to);
191 }
192 };
193 return present;
194}
195fn partitionText(gpa: Allocator, a: *const Automaton, blocks: []const []const usize) Error!Text {
196 var out: Builder = .{ .gpa = gpa };
197 for (blocks, 0..) |block, i| {
198 if (i > 0) try out.add(" ");
199 try out.add(try a.setText(gpa, block));
200 }
201 return out.bytes.items;
202}
203const Minimized = struct { automaton: *Automaton, details: []const Text };
204fn minimize(gpa: Allocator, a: *const Automaton) Error!Minimized {
205 const reachable = try graphReach(gpa, a, &.{a.start}, false);
206 var missing: std.ArrayList(usize) = .empty;
207 var blocks: std.ArrayList(std.ArrayList(usize)) = .empty;
208 var byFinal = [_]?usize{ null, null };
209 for (a.states, 0..) |_, state| {
210 if (!reachable[state]) {
211 try missing.append(gpa, state);
212 continue;
213 }
214 const key: usize = if (a.final[state]) 1 else 0;
215 if (byFinal[key] == null) {
216 byFinal[key] = blocks.items.len;
217 try blocks.append(gpa, .empty);
218 }
219 try blocks.items[byFinal[key].?].append(gpa, state);
220 }
221 var details: std.ArrayList(Text) = .empty;
222 try details.append(gpa, try std.fmt.allocPrint(gpa, " unreachable: {s}", .{try a.listText(gpa, missing.items)}));
223 var printedBlocks: std.ArrayList([]const usize) = .empty;
224 for (blocks.items) |block| try printedBlocks.append(gpa, block.items);
225 try details.append(gpa, try std.fmt.allocPrint(gpa, " round 0: {s}", .{try partitionText(gpa, a, printedBlocks.items)}));
226 var round: usize = 1;
227 const membership = try gpa.alloc(usize, a.states.len);
228 while (true) : (round += 1) {
229 for (blocks.items, 0..) |block, index| for (block.items) |state| {
230 membership[state] = index;
231 };
232 var next: std.ArrayList(std.ArrayList(usize)) = .empty;
233 var signatures: std.StringHashMapUnmanaged(usize) = .empty;
234 for (a.states, 0..) |_, state| {
235 if (!reachable[state]) continue;
236 var signature: std.ArrayList(usize) = .empty;
237 try signature.append(gpa, membership[state]);
238 for (a.alphabet) |c| try signature.append(gpa, membership[a.delta(state, c)]);
239 const entry = try signatures.getOrPut(gpa, try setKey(gpa, signature.items));
240 if (!entry.found_existing) {
241 entry.value_ptr.* = next.items.len;
242 try next.append(gpa, .empty);
243 }
244 try next.items[entry.value_ptr.*].append(gpa, state);
245 }
246 if (next.items.len == blocks.items.len) break;
247 blocks = next;
248 printedBlocks.clearRetainingCapacity();
249 for (blocks.items) |block| try printedBlocks.append(gpa, block.items);
250 try details.append(gpa, try std.fmt.allocPrint(gpa, " round {d}: {s}", .{ round, try partitionText(gpa, a, printedBlocks.items) }));
251 }
252 for (blocks.items, 0..) |block, index| for (block.items) |state| {
253 membership[state] = index;
254 };
255 var order: std.ArrayList(usize) = .empty;
256 try order.append(gpa, membership[a.start]);
257 var indices: std.AutoHashMapUnmanaged(usize, usize) = .empty;
258 try indices.put(gpa, order.items[0], 0);
259 var rows: std.ArrayList([]const usize) = .empty;
260 var i: usize = 0;
261 while (i < order.items.len) : (i += 1) {
262 const representative = blocks.items[order.items[i]].items[0];
263 var row: std.ArrayList(usize) = .empty;
264 for (a.alphabet) |c| {
265 const group = membership[a.delta(representative, c)];
266 const entry = try indices.getOrPut(gpa, group);
267 if (!entry.found_existing) {
268 entry.value_ptr.* = order.items.len;
269 try order.append(gpa, group);
270 }
271 try row.append(gpa, entry.value_ptr.*);
272 }
273 try rows.append(gpa, row.items);
274 }
275 const result = try Automaton.create(gpa, false, try series(gpa, "M", order.items.len), a.alphabet, 0);
276 for (order.items, rows.items, 0..) |group, row, index| {
277 result.final[index] = a.final[blocks.items[group].items[0]];
278 for (a.alphabet, row) |c, next| try result.add(index, c, next);
279 try details.append(gpa, try std.fmt.allocPrint(gpa, " M{d} = {s}", .{ index, try a.setText(gpa, blocks.items[group].items) }));
280 }
281 return .{ .automaton = result, .details = details.items };
282}

284const Product = struct { automaton: *Automaton, details: []const Text };
285fn product(gpa: Allocator, a: *const Automaton, b: *const Automaton, command: Text) Error!Product {
286 var pairs: std.ArrayList(Pair) = .empty;
287 try pairs.append(gpa, .{ .left = a.start, .right = b.start });
288 var indices: std.AutoHashMapUnmanaged(Pair, usize) = .empty;
289 try indices.put(gpa, pairs.items[0], 0);
290 var rows: std.ArrayList([]const usize) = .empty;
291 var i: usize = 0;
292 while (i < pairs.items.len) : (i += 1) {
293 const pair = pairs.items[i];
294 var row: std.ArrayList(usize) = .empty;
295 for (a.alphabet) |c| {
296 const next: Pair = .{ .left = a.delta(pair.left, c), .right = b.delta(pair.right, c) };
297 const entry = try indices.getOrPut(gpa, next);
298 if (!entry.found_existing) {
299 entry.value_ptr.* = pairs.items.len;
300 try pairs.append(gpa, next);
301 }
302 try row.append(gpa, entry.value_ptr.*);
303 }
304 try rows.append(gpa, row.items);
305 }
306 const result = try Automaton.create(gpa, false, try series(gpa, "P", pairs.items.len), a.alphabet, 0);
307 var details: std.ArrayList(Text) = .empty;
308 for (pairs.items, rows.items, 0..) |pair, row, index| {
309 const x = a.final[pair.left];
310 const y = b.final[pair.right];
311 result.final[index] = eq(command, "UNION") and (x or y) or eq(command, "INTERSECT") and x and y or eq(command, "DIFF") and x and !y;
312 for (a.alphabet, row) |c, next| try result.add(index, c, next);
313 try details.append(gpa, try std.fmt.allocPrint(gpa, " P{d} = ({s},{s})", .{ index, a.states[pair.left], b.states[pair.right] }));
314 }
315 return .{ .automaton = result, .details = details.items };
316}
317fn witness(gpa: Allocator, a: *const Automaton) Error!?Text {
318 const Item = struct { state: usize, word: Text };
319 var queue: std.ArrayList(Item) = .empty;
320 try queue.append(gpa, .{ .state = a.start, .word = "" });
321 const seen = try gpa.alloc(bool, a.states.len);
322 @memset(seen, false);
323 seen[a.start] = true;
324 var i: usize = 0;
325 while (i < queue.items.len) : (i += 1) {
326 const item = queue.items[i];
327 if (a.final[item.state]) return item.word;
328 for (a.alphabet) |c| {
329 const next = a.delta(item.state, c);
330 if (!seen[next]) {
331 seen[next] = true;
332 try queue.append(gpa, .{ .state = next, .word = try std.fmt.allocPrint(gpa, "{s}{c}", .{ item.word, c }) });
333 }
334 }
335 }
336 return null;
337}

339fn equivalent(gpa: Allocator, a: *const Automaton, b: *const Automaton) Error!?Difference {
340 const Item = struct { pair: Pair, word: Text };
341 var queue: std.ArrayList(Item) = .empty;
342 const initial: Pair = .{ .left = a.start, .right = b.start };
343 try queue.append(gpa, .{ .pair = initial, .word = "" });
344 var seen: std.AutoHashMapUnmanaged(Pair, void) = .empty;
345 try seen.put(gpa, initial, {});
346 var i: usize = 0;
347 while (i < queue.items.len) : (i += 1) {
348 const item = queue.items[i];
349 const pair = item.pair;
350 if (a.final[pair.left] != b.final[pair.right]) return .{ .word = item.word, .left = a.final[pair.left] };
351 for (a.alphabet) |c| {
352 const next: Pair = .{ .left = a.delta(pair.left, c), .right = b.delta(pair.right, c) };
353 if (!seen.contains(next)) {
354 try seen.put(gpa, next, {});
355 try queue.append(gpa, .{ .pair = next, .word = try std.fmt.allocPrint(gpa, "{s}{c}", .{ item.word, c }) });
356 }
357 }
358 }
359 return null;

361const CycleSearch = struct {
362 automaton: *const Automaton,
363 live: []const bool,
364 colors: []u8,
365 fn visit(self: *CycleSearch, state: usize) bool {
366 self.colors[state] = 1;
367 for (self.automaton.alphabet) |c| {
368 const next = self.automaton.delta(state, c);
369 if (!self.live[next]) continue;
370 if (self.colors[next] == 1 or self.colors[next] == 0 and self.visit(next)) return true;
371 }
372 self.colors[state] = 2;
373 return false;
374 }

385const FiniteCounter = struct {
386 automaton: *const Automaton,
387 live: []const bool,
388 memo: []?i64,
389 fn count(self: *FiniteCounter, state: usize) i64 {
390 if (!self.live[state]) return 0;
391 if (self.memo[state]) |value| return value;
392 var total: i64 = if (self.automaton.final[state]) 1 else 0;
393 for (self.automaton.alphabet) |c| total += self.count(self.automaton.delta(state, c));
394 self.memo[state] = total;
395 return total;
396 }

404fn countLength(gpa: Allocator, a: *const Automaton, n: usize) Error!i64 {
405 var counts = try gpa.alloc(i64, a.states.len);
406 @memset(counts, 0);
407 counts[a.start] = 1;
408 for (0..n) |_| {
409 const next = try gpa.alloc(i64, a.states.len);
410 @memset(next, 0);
411 for (counts, 0..) |count, state| for (a.alphabet) |c| {
412 next[a.delta(state, c)] += count;
413 };
414 counts = next;
415 }
416 var total: i64 = 0;
417 for (counts, a.final) |count, accepting| if (accepting) {
418 total += count;
419 };
420 return total;
421}

424 const Item = struct { states: []const usize, word: Text };
425 const initial = try gpa.alloc(usize, a.states.len);
426 for (initial, 0..) |*state, i| state.* = i;
427 var queue: std.ArrayList(Item) = .empty;

431 var i: usize = 0;

436 const present = try gpa.alloc(bool, a.states.len);

441 if (!seen.contains(key)) {

444 }
445 }

449const Regex = struct { kind: u8, children: []const *Regex = &.{} };
450const RegexParser = struct {
451 gpa: Allocator,
452 source: Text,
453 position: usize = 0,
454 fn peek(self: *RegexParser) u8 {
455 return if (self.position < self.source.len) self.source[self.position] else 0;
456 }
457 fn invalid(self: *RegexParser) Error {
458 return bad(self.gpa, "syntax error at {d}", .{self.position + 1});
459 }
460 fn node(self: *RegexParser, kind: u8, children: []const *Regex) Error!*Regex {
461 const result = try self.gpa.create(Regex);
462 result.* = .{ .kind = kind, .children = children };
463 return result;
464 }
465 fn atom(self: *RegexParser) Error!*Regex {
466 const c = self.peek();
467 if (symbolCharacter(c)) {
468 self.position += 1;
469 return self.node(c, &.{});
470 }
471 if (c == '(') {
472 self.position += 1;
473 const result = try self.alternation();
474 if (self.peek() != ')') return self.invalid();
475 self.position += 1;
476 return result;
477 }
478 return self.invalid();
479 }
480 fn factor(self: *RegexParser) Error!*Regex {
481 var result = try self.atom();
482 while (std.mem.findScalar(u8, "*+?", self.peek()) != null and self.peek() != 0) {
483 const c = self.peek();
484 self.position += 1;
485 result = try self.node(c, try self.gpa.dupe(*Regex, &.{result}));
486 }
487 return result;
488 }
489 fn concatenation(self: *RegexParser) Error!*Regex {
490 var children: std.ArrayList(*Regex) = .empty;
491 try children.append(self.gpa, try self.factor());
492 while (symbolCharacter(self.peek()) or self.peek() == '(') try children.append(self.gpa, try self.factor());
493 return if (children.items.len == 1) children.items[0] else self.node('.', children.items);
494 }
495 fn alternation(self: *RegexParser) Error!*Regex {
496 var children: std.ArrayList(*Regex) = .empty;
497 try children.append(self.gpa, try self.concatenation());
498 while (self.peek() == '|') {
499 self.position += 1;
500 try children.append(self.gpa, try self.concatenation());
501 }
502 return if (children.items.len == 1) children.items[0] else self.node('|', children.items);
503 }

513 return index;
514 }
515 fn add(self: *Thompson, from: usize, symbol: u8, to: usize) Error!void {
516 try self.transitions.append(self.gpa, .{ .from = from, .symbol = symbol, .to = to });
517 }
518 fn build(self: *Thompson, tree: *const Regex) Error!Part {
519 if (tree.children.len == 0) {
520 const start = self.state();
521 const end = self.state();
522 try self.add(start, tree.kind, end);
523 return .{ .start = start, .end = end };
524 }
525 if (tree.kind == '.') {
526 var result = try self.build(tree.children[0]);
527 for (tree.children[1..]) |child| {
528 const part = try self.build(child);
529 try self.add(result.end, 'e', part.start);
530 result.end = part.end;
531 }
532 return result;
533 }
534 const start = self.state();
535 if (tree.kind == '|') {
536 var parts: std.ArrayList(Part) = .empty;
537 for (tree.children) |child| try parts.append(self.gpa, try self.build(child));
538 const end = self.state();
539 for (parts.items) |part| {
540 try self.add(start, 'e', part.start);
541 try self.add(part.end, 'e', end);
542 }
543 return .{ .start = start, .end = end };
544 }
545 const part = try self.build(tree.children[0]);
546 const end = self.state();
547 try self.add(start, 'e', part.start);
548 try self.add(part.end, 'e', end);
549 if (tree.kind != '?') try self.add(part.end, 'e', part.start);
550 if (tree.kind != '+') try self.add(start, 'e', end);
551 return .{ .start = start, .end = end };
552 }
553};
554fn regexAutomaton(gpa: Allocator, tree: *const Regex, alphabet: Text) Error!*Automaton {
555 var builder: Thompson = .{ .gpa = gpa };
556 const part = try builder.build(tree);
557 const result = try Automaton.create(gpa, true, try series(gpa, "N", builder.next), alphabet, part.start);
558 result.final[part.end] = true;
559 for (builder.transitions.items) |t| try result.add(t.from, t.symbol, t.to);
560 return result;
561}
562fn sortedAlphabet(gpa: Allocator, text: Text) Error!Text {
563 const result = try gpa.dupe(u8, text);
564 std.mem.sort(u8, result, {}, std.sort.asc(u8));
565 return result;
566}
567fn stateIndex(gpa: Allocator, a: *const Automaton, name: Text) Error!usize {
568 for (a.states, 0..) |state, i| if (eq(state, name)) return i;
569 return bad(gpa, "unknown state {s}", .{name});
570}
571const Entry = struct { name: Text, automaton: *Automaton };
572const countTable = [_]struct { name: Text, count: usize }{
573 .{ .name = "ACCEPT", .count = 2 }, .{ .name = "CLOSURE", .count = 2 }, .{ .name = "TODFA", .count = 2 }, .{ .name = "EPSFREE", .count = 2 }, .{ .name = "MINIMIZE", .count = 2 },
574 .{ .name = "EQUIV", .count = 2 }, .{ .name = "UNION", .count = 3 }, .{ .name = "INTERSECT", .count = 3 }, .{ .name = "DIFF", .count = 3 }, .{ .name = "COMPLEMENT", .count = 2 },
575 .{ .name = "REVERSE", .count = 2 }, .{ .name = "CONCAT", .count = 3 }, .{ .name = "STAR", .count = 2 }, .{ .name = "ENUM", .count = 2 }, .{ .name = "COUNT", .count = 2 },
576 .{ .name = "EMPTY", .count = 1 }, .{ .name = "FINITE", .count = 1 }, .{ .name = "INFO", .count = 1 }, .{ .name = "TABLE", .count = 1 }, .{ .name = "SHOW", .count = 1 },
577 .{ .name = "SYNC", .count = 1 }, .{ .name = "LIST", .count = 0 }, .{ .name = "DELETE", .count = 1 },
578};
579const Toolkit = struct {
580 gpa: Allocator,
581 scratch: Allocator,
582 out: *Builder,
583 entries: std.ArrayList(Entry) = .empty,
584 fn lookup(self: *const Toolkit, name: Text) ?*Automaton {
585 for (self.entries.items) |entry| if (eq(entry.name, name)) return entry.automaton;
586 return null;
587 }
588 fn register(self: *Toolkit, name: Text, a: *Automaton, details: []const Text) Error!void {
589 try self.entries.append(self.gpa, .{ .name = name, .automaton = a });
590 try a.summary(self.out, name);
591 for (details) |text| try self.out.line("{s}", .{text});
592 }
593 fn define(self: *Toolkit, command: Text, args: []const Text) Error!void {
594 if (args.len < 5) return bad(self.scratch, "wrong number of arguments", .{});
595 const name = args[0];
596 if (self.lookup(name) != null) return bad(self.scratch, "{s} already exists", .{name});
597 const states = try split(self.gpa, args[1], ',');
598 var seen: std.StringHashMapUnmanaged(void) = .empty;
599 for (states) |state| {
600 if (seen.contains(state)) return bad(self.scratch, "duplicate state {s}", .{state});
601 try seen.put(self.scratch, state, {});
602 }
603 const a = try Automaton.create(self.gpa, eq(command, "NFA"), states, try sortedAlphabet(self.gpa, args[2]), 0);
604 a.start = try stateIndex(self.scratch, a, args[3]);
605 if (!eq(args[4], "-")) for (try split(self.scratch, args[4], ',')) |state| {
606 const index = try stateIndex(self.scratch, a, state);
607 if (a.final[index]) return bad(self.scratch, "duplicate state {s}", .{state});
608 a.final[index] = true;
609 };
610 const TransitionPair = struct { state: usize, symbol: u8 };
611 var triples: std.StringHashMapUnmanaged(void) = .empty;
612 var pairs: std.AutoHashMapUnmanaged(TransitionPair, void) = .empty;
613 for (args[5..]) |word| {
614 const parts = try split(self.scratch, word, ':');
615 const p = try stateIndex(self.scratch, a, parts[0]);
616 const c = parts[1][0];
617 if (c == 'e') {
618 if (!a.nfa) return bad(self.scratch, "epsilon transition in DFA", .{});
619 } else if (std.mem.findScalar(u8, a.alphabet, c) == null) return bad(self.scratch, "unknown symbol {c}", .{c});
620 const q = try stateIndex(self.scratch, a, parts[2]);
621 if (triples.contains(word)) return bad(self.scratch, "duplicate transition {s}", .{word});
622 try triples.put(self.scratch, word, {});
623 const key: TransitionPair = .{ .state = p, .symbol = c };
624 if (!a.nfa and pairs.contains(key)) return bad(self.scratch, "conflicting transition {s}", .{word});
625 try pairs.put(self.scratch, key, {});
626 try a.add(p, c, q);
627 }
628 if (!a.nfa) for (states, 0..) |state, p| for (a.alphabet) |c| {
629 if ((try a.destinations(self.scratch, p, c)).len == 0) return bad(self.scratch, "missing transition {s}:{c}", .{ state, c });
630 };
631 try self.register(name, a, &.{});
632 }
633 fn executeLine(self: *Toolkit, words: []const Text) Error!void {
634 const command = words[0];
635 const args = words[1..];
636 if (eq(command, "DFA") or eq(command, "NFA")) return self.define(command, args);
637 if (eq(command, "REGEX")) {
638 if (args.len != 3) return bad(self.scratch, "wrong number of arguments", .{});
639 if (self.lookup(args[0]) != null) return bad(self.scratch, "{s} already exists", .{args[0]});
640 const alphabet = try sortedAlphabet(self.gpa, args[1]);
641 for (args[2]) |c| if (symbolCharacter(c) and c != 'e' and std.mem.findScalar(u8, alphabet, c) == null) return bad(self.scratch, "unknown symbol {c}", .{c});
642 var parser: RegexParser = .{ .gpa = self.scratch, .source = args[2] };
643 const tree = try parser.alternation();
644 if (parser.position != parser.source.len) return parser.invalid();
645 try self.register(args[0], try regexAutomaton(self.gpa, tree, alphabet), &.{});
646 return;
647 }
648 var expected: ?usize = null;
649 for (countTable) |item| if (eq(item.name, command)) {
650 expected = item.count;
651 break;
652 };
653 if (expected == null) return bad(self.scratch, "unknown command {s}", .{command});
654 if (args.len != expected.?) return bad(self.scratch, "wrong number of arguments", .{});
655 if (eq(command, "LIST")) {
656 if (self.entries.items.len == 0) try self.out.line("(none)", .{});
657 for (self.entries.items) |entry| try entry.automaton.summary(self.out, entry.name);
658 return;
659 }
660 const references: usize = if (includes(command, &.{ "EQUIV", "UNION", "INTERSECT", "DIFF", "CONCAT" })) 2 else 1;
661 var autos: [2]*Automaton = undefined;
662 for (args[0..references], 0..) |name, i| autos[i] = self.lookup(name) orelse return bad(self.scratch, "unknown automaton {s}", .{name});
663 const creates = includes(command, &.{ "TODFA", "EPSFREE", "MINIMIZE", "UNION", "INTERSECT", "DIFF", "COMPLEMENT", "REVERSE", "CONCAT", "STAR" });
664 if (creates and self.lookup(args[args.len - 1]) != null) return bad(self.scratch, "{s} already exists", .{args[args.len - 1]});
665 if (includes(command, &.{ "MINIMIZE", "UNION", "INTERSECT", "DIFF", "COMPLEMENT", "SYNC" })) for (autos[0..references], args[0..references]) |a, name| if (a.nfa) return bad(self.scratch, "{s} is not a DFA", .{name});
666 if (references == 2 and !eq(autos[0].alphabet, autos[1].alphabet)) return bad(self.scratch, "alphabets differ", .{});
667 const a = autos[0];
668 if (eq(command, "DELETE")) {
669 for (self.entries.items, 0..) |entry, i| if (eq(entry.name, args[0])) {
670 _ = self.entries.orderedRemove(i);
671 break;
672 };
673 try self.out.line("deleted {s}", .{args[0]});
674 return;
675 }
676 if (eq(command, "ACCEPT")) {
677 const word = if (eq(args[1], "e")) "" else args[1];
678 for (word) |c| if (std.mem.findScalar(u8, a.alphabet, c) == null) return bad(self.scratch, "unknown symbol {c}", .{c});
679 var states = try a.closure(self.scratch, &.{a.start});
680 var path: Builder = .{ .gpa = self.scratch };
681 try path.add(if (a.nfa) try a.setText(self.scratch, states) else a.states[states[0]]);
682 for (word) |c| {
683 states = try a.step(self.scratch, states, c);
684 try path.print(" -{c}-> {s}", .{ c, if (a.nfa) try a.setText(self.scratch, states) else a.states[states[0]] });
685 }
686 try self.out.line("{s} {s}", .{ if (a.accepts(states)) "ACCEPT" else "REJECT", path.bytes.items });
687 return;
688 }
689 if (eq(command, "CLOSURE")) {
690 var states: std.ArrayList(usize) = .empty;
691 for (try split(self.scratch, args[1], ',')) |name| try states.append(self.scratch, try stateIndex(self.scratch, a, name));
692 try self.out.line("{s}", .{try a.setText(self.scratch, try a.closure(self.scratch, states.items))});
693 return;
694 }
695 if (eq(command, "TODFA")) {
696 const result = try determinize(self.gpa, a);
697 var details: std.ArrayList(Text) = .empty;
698 for (result.sets, 0..) |set, i| try details.append(self.scratch, try std.fmt.allocPrint(self.scratch, " D{d} = {s}", .{ i, try a.setText(self.scratch, set) }));
699 try self.register(args[1], result.automaton, details.items);
700 return;
701 }
702 if (eq(command, "EPSFREE")) {
703 const result = try Automaton.create(self.gpa, true, a.states, a.alphabet, a.start);
704 for (a.states, 0..) |_, state| {
705 const closed = try a.closure(self.scratch, &.{state});
706 result.final[state] = a.accepts(closed);
707 for (a.alphabet) |c| for (try a.step(self.scratch, closed, c)) |next| try result.add(state, c, next);
708 }
709 try self.register(args[1], result, &.{});
710 return;
711 }
712 if (eq(command, "MINIMIZE")) {
713 const result = try minimize(self.gpa, a);
714 try self.register(args[1], result.automaton, result.details);
715 return;
716 }
717 if (includes(command, &.{ "UNION", "INTERSECT", "DIFF" })) {
718 const result = try product(self.gpa, a, autos[1], command);
719 try self.register(args[2], result.automaton, result.details);
720 return;
721 }
722 if (eq(command, "EQUIV")) {
723 const x = (try determinize(self.scratch, a)).automaton;
724 const z = (try determinize(self.scratch, autos[1])).automaton;
725 if (try equivalent(self.scratch, x, z)) |difference| {
726 try self.out.line("NOT EQUIVALENT: {s} is accepted only by {s}", .{ emptyWord(difference.word), if (difference.left) args[0] else args[1] });
727 } else try self.out.line("EQUIVALENT", .{});
728 return;
729 }
730 if (eq(command, "COMPLEMENT")) {
731 const result = try Automaton.create(self.gpa, false, a.states, a.alphabet, a.start);
732 for (a.final, result.final) |original, *accepting| accepting.* = !original;
733 for (a.transitions.items) |t| try result.add(t.from, t.symbol, t.to);
734 try self.register(args[1], result, &.{});
735 return;
736 }
737 if (includes(command, &.{ "REVERSE", "CONCAT", "STAR" })) {
738 const concat = eq(command, "CONCAT");
739 const total = if (concat) a.states.len + autos[1].states.len else a.states.len + 1;
740 const result = try Automaton.create(self.gpa, true, try series(self.gpa, "N", total), a.alphabet, if (concat) a.start else 0);
741 if (eq(command, "REVERSE")) {
742 for (a.transitions.items) |t| try result.add(t.to + 1, t.symbol, t.from + 1);
743 for (a.final, 0..) |accepting, state| if (accepting) try result.add(0, 'e', state + 1);
744 result.final[a.start + 1] = true;
745 } else if (eq(command, "STAR")) {
746 for (a.transitions.items) |t| try result.add(t.from + 1, t.symbol, t.to + 1);
747 result.final[0] = true;
748 try result.add(0, 'e', a.start + 1);
749 for (a.final, 0..) |accepting, state| if (accepting) {
750 result.final[state + 1] = true;
751 try result.add(state + 1, 'e', a.start + 1);
752 };
753 } else {
754 for (a.transitions.items) |t| try result.add(t.from, t.symbol, t.to);
755 const b = autos[1];
756 const offset = a.states.len;
757 for (b.transitions.items) |t| try result.add(t.from + offset, t.symbol, t.to + offset);
758 for (a.final, 0..) |accepting, state| if (accepting) try result.add(state, 'e', b.start + offset);
759 for (b.final, 0..) |accepting, state| result.final[state + offset] = accepting;
760 }
761 try self.register(args[args.len - 1], result, &.{});
762 return;
763 }
764 if (eq(command, "SYNC")) {
765 if (try synchronize(self.scratch, a)) |result| {
766 try self.out.line("SYNCHRONIZING {s} -> {s}", .{ emptyWord(result.word), a.states[result.state] });
767 } else try self.out.line("NOT SYNCHRONIZING", .{});
768 return;
769 }
770 if (eq(command, "COUNT")) {
771 const n = std.fmt.parseInt(usize, args[1], 10) catch unreachable;
772 const d = (try determinize(self.scratch, a)).automaton;
773 try self.out.line("{d}", .{try countLength(self.scratch, d, n)});
774 return;
775 }
776 if (eq(command, "ENUM")) {
777 const k = std.fmt.parseInt(usize, args[1], 10) catch unreachable;
778 const d = (try determinize(self.scratch, a)).automaton;
779 const Item = struct { state: usize, word: Text };
780 var queue: std.ArrayList(Item) = .empty;
781 try queue.append(self.scratch, .{ .state = d.start, .word = "" });
782 var total: usize = 0;
783 var accepted: std.ArrayList(Text) = .empty;
784 var i: usize = 0;
785 while (i < queue.items.len) : (i += 1) {
786 const item = queue.items[i];
787 if (d.final[item.state]) {
788 total += 1;
789 if (accepted.items.len < 50) try accepted.append(self.scratch, emptyWord(item.word));
790 }
791 if (item.word.len < k) for (d.alphabet) |c| try queue.append(self.scratch, .{ .state = d.delta(item.state, c), .word = try std.fmt.allocPrint(self.scratch, "{s}{c}", .{ item.word, c }) });
792 }
793 try self.out.print("{d}", .{total});
794 for (accepted.items) |word| try self.out.print(" {s}", .{word});
795 if (total > 50) try self.out.add(" ...");
796 try self.out.add("\n");
797 return;
798 }
799 if (eq(command, "EMPTY")) {
800 const d = (try determinize(self.scratch, a)).automaton;
801 if (try witness(self.scratch, d)) |word| {
802 try self.out.line("NOT EMPTY {s}", .{emptyWord(word)});
803 } else try self.out.line("EMPTY", .{});
804 return;
805 }
806 if (eq(command, "FINITE")) {
807 const d = (try determinize(self.scratch, a)).automaton;
808 const result = try language(self.scratch, d);
809 if (eq(result.classification, "infinite")) {
810 try self.out.line("INFINITE", .{});
811 } else try self.out.line("FINITE {d}", .{try finiteCount(self.scratch, d, result.live)});
812 return;
813 }
814 if (eq(command, "INFO")) {
815 var epsilon: usize = 0;
816 var deterministic = true;
817 var complete = true;
818 for (a.transitions.items) |t| if (t.symbol == 'e') {
819 epsilon += 1;
820 };
821 if (epsilon > 0) deterministic = false;
822 for (a.states, 0..) |_, state| for (a.alphabet) |c| {
823 const count = (try a.destinations(self.scratch, state, c)).len;
824 if (count > 1) deterministic = false;
825 if (count == 0) complete = false;
826 };
827 const d = (try determinize(self.scratch, a)).automaton;
828 const result = try language(self.scratch, d);
829 const minimal = (try minimize(self.scratch, d)).automaton;
830 try a.summary(self.out, args[0]);
831 try self.out.line(" epsilon transitions: {d}", .{epsilon});
832 try self.out.line(" reachable: {s}", .{try a.listText(self.scratch, try orderedSet(self.scratch, try graphReach(self.scratch, a, &.{a.start}, false)))});
833 try self.out.line(" live: {s}", .{try a.listText(self.scratch, try orderedSet(self.scratch, try graphReach(self.scratch, a, try orderedSet(self.scratch, a.final), true)))});
834 try self.out.line(" deterministic: {s}", .{yes(deterministic)});
835 try self.out.line(" complete: {s}", .{yes(complete)});
836 try self.out.line(" language: {s}", .{result.classification});
837 try self.out.line(" minimal DFA states: {d}", .{minimal.states.len});
838 return;
839 }
840 if (eq(command, "SHOW")) {
841 var parts: Builder = .{ .gpa = self.scratch };
842 try parts.print("{s} {s} ", .{ if (a.nfa) "NFA" else "DFA", args[0] });
843 for (a.states, 0..) |name, i| {
844 if (i > 0) try parts.add(",");
845 try parts.add(name);
846 }
847 try parts.print(" {s} {s} ", .{ a.alphabet, a.states[a.start] });
848 const finals = try orderedSet(self.scratch, a.final);
849 if (finals.len == 0) {
850 try parts.add("-");
851 } else for (finals, 0..) |state, i| {
852 if (i > 0) try parts.add(",");
853 try parts.add(a.states[state]);
854 }
855 const symbols = try std.fmt.allocPrint(self.scratch, "{s}e", .{a.alphabet});
856 for (a.states, 0..) |name, state| for (symbols) |c| for (try a.destinations(self.scratch, state, c)) |next| try parts.print(" {s}:{c}:{s}", .{ name, c, a.states[next] });
857 try self.out.line("{s}", .{parts.bytes.items});
858 return;
859 }
860 if (eq(command, "TABLE")) {
861 var epsilon = false;
862 if (a.nfa) for (a.transitions.items) |t| if (t.symbol == 'e') {
863 epsilon = true;
864 break;
865 };
866 const symbols = if (epsilon) try std.fmt.allocPrint(self.scratch, "{s}e", .{a.alphabet}) else a.alphabet;
867 const columns = symbols.len + 1;
868 var rows: std.ArrayList([]Text) = .empty;
869 const header = try self.scratch.alloc(Text, columns);
870 header[0] = "";
871 for (symbols, 1..) |_, i| header[i] = symbols[i - 1 .. i];
872 try rows.append(self.scratch, header);
873 for (a.states, 0..) |name, state| {
874 const row = try self.scratch.alloc(Text, columns);
875 row[0] = try std.fmt.allocPrint(self.scratch, "{s}{s}{s}", .{ if (state == a.start) "->" else " ", if (a.final[state]) "*" else " ", name });
876 for (symbols, 1..) |c, i| {
877 const destinations = try a.destinations(self.scratch, state, c);
878 row[i] = if (!a.nfa) a.states[destinations[0]] else if (destinations.len == 0) "-" else try a.setText(self.scratch, destinations);
879 }
880 try rows.append(self.scratch, row);
881 }
882 const widths = try self.scratch.alloc(usize, columns);
883 @memset(widths, 0);
884 for (rows.items) |row| for (row, 0..) |cell, i| {
885 widths[i] = @max(widths[i], cell.len);
886 };
887 for (rows.items) |row| {
888 var text: Builder = .{ .gpa = self.scratch };
889 for (row, 0..) |cell, i| {
890 if (i > 0) try text.add(" ");
891 try text.add(cell);
892 for (0..widths[i] - cell.len) |_| try text.add(" ");
893 }
894 try self.out.line("{s}", .{std.mem.trimEnd(u8, text.bytes.items, " ")});
895 }
896 return;
897 }
898 }
899};
900pub fn main(init: std.process.Init) !void {
901 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
902 defer arena.deinit();
903 const gpa = arena.allocator();
904 var inputBuffer: [8192]u8 = undefined;
905 var input = std.Io.File.stdin().readerStreaming(init.io, &inputBuffer);
906 const text = try input.interface.allocRemaining(gpa, .unlimited);
907 var lines = std.mem.splitScalar(u8, text, '\n');
908 const count = try std.fmt.parseInt(usize, std.mem.trim(u8, lines.next().?, "\r"), 10);
909 var outputBuffer: [8192]u8 = undefined;
910 var output = std.Io.File.stdout().writerStreaming(init.io, &outputBuffer);
911 var unused: Builder = .{ .gpa = gpa };
912 var toolkit: Toolkit = .{ .gpa = gpa, .scratch = gpa, .out = &unused };
913 for (0..count) |_| {
914 const source = std.mem.trimEnd(u8, lines.next().?, "\r");
915 if (source[0] == '#') continue;
916 var commandArena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
917 defer commandArena.deinit();
918 const scratch = commandArena.allocator();
919 var out: Builder = .{ .gpa = scratch };
920 toolkit.scratch = scratch;
921 toolkit.out = &out;
922 toolkit.executeLine(try split(scratch, source, ' ')) catch |err| {
923 if (err != error.Invalid) return err;
924 out.bytes.clearRetainingCapacity();
925 try out.line("error: {s}", .{failure});
926 };
927 try output.interface.writeAll(out.bytes.items);
928 }
929 try output.interface.flush();
