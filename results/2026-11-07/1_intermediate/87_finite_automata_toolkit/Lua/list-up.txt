local function fail(format, ...)
    error({reason = string.format(format, ...)}, 0)
end
local function contains(items, value)
    for _, item in ipairs(items) do
        if item == value then
            return true
        end
    end
    return false
end
local function series(prefix, n)
    local result = {}
    for i = 1, n do
        result[i] = prefix .. (i - 1)
    end
    return result
end
local function make_automaton(kind, states, alphabet, start)
    local result = {
        kind = kind,
        states = states,
        alphabet = alphabet,
        start = start,
        final = {},
        transitions = {}
    }
    for i = 1, #states do
        result.final[i] = false
        result.transitions[i] = {}
        for c = 1, #alphabet + 1 do
            result.transitions[i][c] = {}
        end
    end
    return result
end
local function add(a, p, c, q)
    local destinations = a.transitions[p][c]
    if not contains(destinations, q) then
        destinations[#destinations + 1] = q
        table.sort(destinations)
    end
end
local function ordered_set(set, n)
    local result = {}
    for i = 1, n do
        if set[i] then
            result[#result + 1] = i
        end
    end
    return result
end
local function closure(a, states)
    local seen, queue = {}, {}
    for _, state in ipairs(states) do
        if not seen[state] then
            seen[state] = true
            queue[#queue + 1] = state
        end
    end
    local i = 1
    while i <= #queue do
        for _, next_state in ipairs(a.transitions[queue[i]][#a.alphabet + 1]) do
            if not seen[next_state] then
                seen[next_state] = true
                queue[#queue + 1] = next_state
            end
        end
        i = i + 1
    end
    return ordered_set(seen, #a.states)
end
local function destination(a, states, c)
    local next_states = {}
    for _, state in ipairs(states) do
        for _, next_state in ipairs(a.transitions[state][c]) do
            next_states[#next_states + 1] = next_state
        end
    end
    return closure(a, next_states)
end
local function accepts(a, states)
    for _, state in ipairs(states) do
        if a.final[state] then
            return true
        end
    end
    return false
end
local function set_key(states)
    return table.concat(states, ",")
end
local function determinize(a)
    local sets = {closure(a, {a.start})}
    local indices = {[set_key(sets[1])] = 1}
    local rows = {}
    local i = 1
    while i <= #sets do
        local row = {}
        for c = 1, #a.alphabet do
            local next_states = destination(a, sets[i], c)
            local key = set_key(next_states)
            local index = indices[key]
            if not index then
                index = #sets + 1
                indices[key] = index
                sets[index] = next_states
            end
            row[c] = index
        end
        rows[i] = row
        i = i + 1
    end
    local result = make_automaton("DFA", series("D", #sets), a.alphabet, 1)
    for index, set in ipairs(sets) do
        result.final[index] = accepts(a, set)
        for c, next_state in ipairs(rows[index]) do
            add(result, index, c, next_state)
        end
    end
    return result, sets
local function count_length(a, n)
    local counts = {[a.start] = 1}
    for _ = 1, n do
        local next_counts = {}
        for state, count in pairs(counts) do
            for c = 1, #a.alphabet do
                local next_state = a.transitions[state][c][1]
                next_counts[next_state] = (next_counts[next_state] or 0) + count
            end
        end
        counts = next_counts
    end
    local total = 0
    for state, count in pairs(counts) do
        if a.final[state] then
            total = total + count
        end
    end
    return total
function Toolkit:find(name)
    for _, entry in ipairs(self.entries) do
        if entry.name == name then
            return entry.automaton
        end
    end
end
local counts = {
    COUNT = 2,
function Toolkit:command(words)
    local command = table.remove(words, 1)
    local args = words
    if counts[command] == nil then
        fail("unknown command %s", command)
    end
    if #args ~= counts[command] then
        fail("wrong number of arguments")
    end
    local reference_count =
        contains({"EQUIV", "UNION", "INTERSECT", "DIFF", "CONCAT"}, command) and 2 or 1
    local autos = {}
    for i = 1, reference_count do
        autos[i] = self:find(args[i])
        if not autos[i] then
            fail("unknown automaton %s", args[i])
        end
    end
    local a = autos[1]
    elseif command == "COUNT" then
        return {tostring(count_length(determinize(a), tonumber(args[2])))}
