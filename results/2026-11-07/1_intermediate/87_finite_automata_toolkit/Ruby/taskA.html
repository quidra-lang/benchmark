3class AutomatonError < StandardError
4end

6class Automaton
7 attr_reader :kind, :states, :alphabet, :start, :accepting, :edges
8
9 def initialize(kind, states, alphabet, start, accepting, triples)
10 @kind = kind
11 @states = states
12 @alphabet = alphabet.sort
13 @start = start
14 @accepting = accepting.to_set
15 @edges = Array.new(states.length) { {} }
16 triples.each do |p, c, q|
17 (@edges[p][c] ||= []) << q
18 end
19 @edges.each do |row|
20 row.each_value(&:sort!)
21 row.each_value(&:uniq!)
22 end
23 end
24
25 def triples
26 result = []
27 @edges.each_with_index do |row, p|
28 (@alphabet + ['e']).each do |c|
29 (row[c] || []).each { |q| result << [p, c, q] }
30 end
31 end
32 result
33 end
34
35 def summary(name)
36 "#{@kind} #{name}: alphabet=#{@alphabet.join} states=#{@states.length} accepting=#{@accepting.length} transitions=#{triples.length}"
37 end
38
39 def set_text(values)
40 '{' + values.sort.map { |i| @states[i] }.join(',') + '}'
41 end
42
43 def list_text(values)
44 values.empty? ? 'none' : values.sort.map { |i| @states[i] }.join(',')
45 end
46
47 def closure(values)
48 seen = values.to_set
49 queue = seen.to_a
50 i = 0
51 while i < queue.length
52 (@edges[queue[i]]['e'] || []).each do |q|
53 next if seen.include?(q)
54 seen.add(q)
55 queue << q
56 end
57 i += 1
58 end
59 seen.to_a.sort
60 end
61
62 def move(values, c)
63 closure(values.flat_map { |p| @edges[p][c] || [] }.uniq)
64 end
65
66 def accepts_set?(values)
67 values.any? { |p| @accepting.include?(p) }
68 end
69
70 def graph_reachable(seeds, reverse = false)
71 graph = Array.new(@states.length) { [] }
72 triples.each do |p, _, q|
73 reverse ? graph[q] << p : graph[p] << q
74 end
75 seen = seeds.to_set
76 queue = seen.to_a
77 i = 0
78 while i < queue.length
79 graph[queue[i]].each do |q|
80 next if seen.include?(q)
81 seen.add(q)
82 queue << q
83 end
84 i += 1
85 end
86 seen.to_a.sort
87 end
88
89 def determinize
90 subsets = [closure([@start])]
91 indices = { subsets[0] => 0 }
92 transitions = []
93 accepting = []
94 i = 0
95 while i < subsets.length
96 accepting << i if accepts_set?(subsets[i])
97 @alphabet.each do |c|
98 target = move(subsets[i], c)
99 unless indices.key?(target)
100 indices[target] = subsets.length
101 subsets << target
102 end
103 transitions << [i, c, indices[target]]
104 end
105 i += 1
106 end
107 result = Automaton.new('DFA', Array.new(subsets.length) { |j| "D#{j}" }, @alphabet, 0, accepting, transitions)
108 [result, subsets]
109 end
110
111 def minimize
112 reachable = graph_reachable([@start])
113 blocks = [reachable.select { |p| @accepting.include?(p) }, reachable.reject { |p| @accepting.include?(p) }]
114 blocks.reject!(&:empty?)
115 blocks.sort_by!(&:first)
116 rounds = [blocks]
117 loop do
118 membership = {}
119 blocks.each_with_index { |block, i| block.each { |p| membership[p] = i } }
120 groups = {}
121 reachable.each do |p|
122 signature = [membership[p]] + @alphabet.map { |c| membership[@edges[p][c][0]] }
123 (groups[signature] ||= []) << p
124 end
125 refined = groups.values.sort_by(&:first)
126 break if refined.length == blocks.length
127 blocks = refined
128 rounds << blocks
129 end
130 membership = {}
131 blocks.each_with_index { |block, i| block.each { |p| membership[p] = i } }
132 order = [membership[@start]]
133 named = { order[0] => 0 }
134 transitions = []
135 accepting = []
136 i = 0
137 while i < order.length
138 representative = blocks[order[i]][0]
139 accepting << i if @accepting.include?(representative)
140 @alphabet.each do |c|
141 target = membership[@edges[representative][c][0]]
142 unless named.key?(target)
143 named[target] = order.length
144 order << target
145 end
146 transitions << [i, c, named[target]]
147 end
148 i += 1
149 end
150 result = Automaton.new('DFA', Array.new(order.length) { |j| "M#{j}" }, @alphabet, 0, accepting, transitions)
151 unreachable = (0...@states.length).to_a - reachable
152 [result, unreachable, rounds, order.map { |j| blocks[j] }]
153 end
154
155 def shortest_word
156 queue = [[@start, '']]
157 seen = Set[@start]
158 i = 0
159 while i < queue.length
160 p, word = queue[i]
161 return word if @accepting.include?(p)
162 @alphabet.each do |c|
163 q = @edges[p][c][0]
164 next if seen.include?(q)
165 seen.add(q)
166 queue << [q, word + c]
167 end
168 i += 1
169 end
170 nil
171 end
172
173 def finite_size(count = true)
174 live = graph_reachable(@accepting.to_a, true).to_set
175 color = Array.new(@states.length, 0)
176 counts = Array.new(@states.length, 0)
177 visit = lambda do |p|
178 return 0 unless live.include?(p)
179 return nil if color[p] == 1
180 return counts[p] if color[p] == 2
181 color[p] = 1
182 total = @accepting.include?(p) ? 1 : 0
183 @alphabet.each do |c|
184 value = visit.call(@edges[p][c][0])
185 return nil if value.nil?
186 total = count ? total + value : (total.positive? || value.positive? ? 1 : 0)
187 end
188 color[p] = 2
189 counts[p] = total
190 end
191 visit.call(@start)
192 end
193end

195class RegexParser
196 def initialize(text)
197 @text = text
198 @position = 0
199 end
200
201 def peek
202 @text[@position]
203 end
204
205 def fail_parse
206 raise AutomatonError, "syntax error at #{@position + 1}"
207 end
208
209 def parse
210 result = alternation
211 fail_parse unless peek.nil?
212 result
213 end
214
215 def alternation
216 parts = [concatenation]
217 while peek == '|'
218 @position += 1
219 parts << concatenation
220 end
221 parts.length == 1 ? parts[0] : [:alternation, parts]
222 end
223
224 def concatenation
225 parts = [factor]
226 while peek && peek.match?(/[a-z0-9(]/)
227 parts << factor
228 end
229 parts.length == 1 ? parts[0] : [:concatenation, parts]
230 end
231
232 def factor
233 result = atom
234 while ['*', '+', '?'].include?(peek)
235 operator = peek
236 @position += 1
237 result = [operator, result]
238 end
239 result
240 end
241
242 def atom
243 c = peek
244 if c && c.match?(/[a-z0-9]/)
245 @position += 1
246 [:symbol, c]
247 elsif c == '('
248 @position += 1
249 result = alternation
250 fail_parse unless peek == ')'
251 @position += 1
252 result
253 else
254 fail_parse
255 end
256 end
257end

259class Toolkit
260 ROLES = {
261 'ACCEPT' => [:reference, :other], 'CLOSURE' => [:reference, :other],
262 'ENUM' => [:reference, :other], 'COUNT' => [:reference, :other],
263 'TODFA' => [:reference, :new], 'EPSFREE' => [:reference, :new],
264 'REVERSE' => [:reference, :new], 'STAR' => [:reference, :new],
265 'MINIMIZE' => [:dfa, :new], 'COMPLEMENT' => [:dfa, :new],
266 'EQUIV' => [:reference, :reference],
267 'UNION' => [:dfa, :dfa, :new], 'INTERSECT' => [:dfa, :dfa, :new],
268 'DIFF' => [:dfa, :dfa, :new], 'CONCAT' => [:reference, :reference, :new],
269 'EMPTY' => [:reference], 'FINITE' => [:reference], 'INFO' => [:reference],
270 'TABLE' => [:reference], 'SHOW' => [:reference], 'DELETE' => [:reference],
271 'SYNC' => [:dfa], 'LIST' => []
272 }.freeze

274 def initialize
275 @table = {}
276 end

278 def register(name, automaton, details = [])
279 @table[name] = automaton
280 [automaton.summary(name)] + details

283 def define(kind, args)
284 raise AutomatonError, 'wrong number of arguments' if args.length < 5
285 name, state_word, alphabet_word, start, accept_word, *transition_words = args
286 raise AutomatonError, "#{name} already exists" if @table.key?(name)
287 states = state_word.split(',')
288 indices = {}
289 states.each_with_index do |state, i|
290 raise AutomatonError, "duplicate state #{state}" if indices.key?(state)
291 indices[state] = i
292 end
293 raise AutomatonError, "unknown state #{start}" unless indices.key?(start)
294 accepting = []
295 unless accept_word == '-'
296 accept_word.split(',').each do |state|
297 raise AutomatonError, "unknown state #{state}" unless indices.key?(state)
298 raise AutomatonError, "duplicate state #{state}" if accepting.include?(indices[state])
299 accepting << indices[state]
300 end
301 end
302 alphabet = alphabet_word.chars.sort
303 triples = []
304 seen = Set.new
305 pairs = Set.new
306 transition_words.each do |word|
307 p, c, q = word.split(':')
308 raise AutomatonError, "unknown state #{p}" unless indices.key?(p)
309 raise AutomatonError, 'epsilon transition in DFA' if c == 'e' && kind == 'DFA'
310 raise AutomatonError, "unknown symbol #{c}" unless c == 'e' || alphabet.include?(c)
311 raise AutomatonError, "unknown state #{q}" unless indices.key?(q)
312 raise AutomatonError, "duplicate transition #{word}" if seen.include?(word)
313 pair = [p, c]
314 raise AutomatonError, "conflicting transition #{word}" if kind == 'DFA' && pairs.include?(pair)
315 seen.add(word)
316 pairs.add(pair)
317 triples << [indices[p], c, indices[q]]
318 end
319 if kind == 'DFA'
320 states.each do |p|
321 alphabet.each do |c|
322 raise AutomatonError, "missing transition #{p}:#{c}" unless pairs.include?([p, c])
323 end
324 end
325 end
326 register(name, Automaton.new(kind, states, alphabet, indices[start], accepting, triples))
327 end

329 def regex(args)
330 raise AutomatonError, 'wrong number of arguments' unless args.length == 3
331 name, alphabet, text = args
332 raise AutomatonError, "#{name} already exists" if @table.key?(name)
333 text.each_char do |c|
334 if c.match?(/[a-z0-9]/) && c != 'e' && !alphabet.include?(c)
335 raise AutomatonError, "unknown symbol #{c}"
336 end
337 end
338 tree = RegexParser.new(text).parse
339 states = []
340 triples = []
341 create = lambda do
342 index = states.length
343 states << "N#{index}"
344 index
345 end
346 build = lambda do |node|
347 case node[0]
348 when :symbol
349 s = create.call
350 f = create.call
351 triples << [s, node[1], f]
352 [s, f]
353 when :concatenation
354 parts = node[1].map { |part| build.call(part) }
355 parts.each_cons(2) { |a, b| triples << [a[1], 'e', b[0]] }
356 [parts[0][0], parts[-1][1]]
357 when :alternation
358 s = create.call
359 parts = node[1].map { |part| build.call(part) }
360 f = create.call
361 parts.each do |a, b|
362 triples << [s, 'e', a]
363 triples << [b, 'e', f]
364 end
365 [s, f]
366 else
367 s = create.call
368 a, b = build.call(node[1])
369 f = create.call
370 triples << [s, 'e', a]
371 triples << [b, 'e', f]
372 triples << [b, 'e', a] unless node[0] == '?'
373 triples << [s, 'e', f] unless node[0] == '+'
374 [s, f]
375 end
376 end
377 start, finish = build.call(tree)
378 register(name, Automaton.new('NFA', states, alphabet.chars, start, [finish], triples))
379 end

381 def product(x, z, operation)
382 pairs = [[x.start, z.start]]
383 indices = { pairs[0] => 0 }
384 triples = []
385 accepting = []
386 i = 0
387 while i < pairs.length
388 p, q = pairs[i]
389 a = x.accepting.include?(p)
390 b = z.accepting.include?(q)
391 good = operation == 'UNION' ? a || b : operation == 'INTERSECT' ? a && b : a && !b
392 accepting << i if good
393 x.alphabet.each do |c|
394 target = [x.edges[p][c][0], z.edges[q][c][0]]
395 unless indices.key?(target)
396 indices[target] = pairs.length
397 pairs << target
398 end
399 triples << [i, c, indices[target]]
400 end
401 i += 1
402 end
403 result = Automaton.new('DFA', Array.new(pairs.length) { |j| "P#{j}" }, x.alphabet, 0, accepting, triples)
404 [result, pairs]
405 end

407 def equivalent(x, z, names)
408 a = x.determinize[0]
409 b = z.determinize[0]
410 queue = [[a.start, b.start, '']]
411 seen = Set[[a.start, b.start]]
412 i = 0
413 while i < queue.length
414 p, q, word = queue[i]
415 if a.accepting.include?(p) != b.accepting.include?(q)
416 owner = a.accepting.include?(p) ? names[0] : names[1]
417 return ["NOT EQUIVALENT: #{word.empty? ? 'e' : word} is accepted only by #{owner}"]
418 end
419 a.alphabet.each do |c|
420 target = [a.edges[p][c][0], b.edges[q][c][0]]
421 next if seen.include?(target)
422 seen.add(target)
423 queue << [target[0], target[1], word + c]
424 end
425 i += 1
426 end
427 ['EQUIVALENT']
428 end

430 def synchronization(x)
431 initial = (0...x.states.length).to_a
432 queue = [[initial, '']]
433 seen = Set[initial]
434 i = 0
435 while i < queue.length
436 subset, word = queue[i]
437 return ["SYNCHRONIZING #{word.empty? ? 'e' : word} -> #{x.states[subset[0]]}"] if subset.length == 1
438 x.alphabet.each do |c|
439 target = subset.map { |p| x.edges[p][c][0] }.uniq.sort
440 next if seen.include?(target)
441 seen.add(target)
442 queue << [target, word + c]
443 end
444 i += 1
445 end
446 ['NOT SYNCHRONIZING']
447 end

449 def execute(command, args)
450 return define(command, args) if ['DFA', 'NFA'].include?(command)
451 return regex(args) if command == 'REGEX'
452 roles = ROLES[command]
453 raise AutomatonError, "unknown command #{command}" unless roles
454 raise AutomatonError, 'wrong number of arguments' unless roles.length == args.length
455 roles.each_with_index do |role, i|
456 if [:reference, :dfa].include?(role) && !@table.key?(args[i])
457 raise AutomatonError, "unknown automaton #{args[i]}"
458 end
459 end
460 roles.each_with_index do |role, i|
461 raise AutomatonError, "#{args[i]} already exists" if role == :new && @table.key?(args[i])
462 end
463 roles.each_with_index do |role, i|
464 raise AutomatonError, "#{args[i]} is not a DFA" if role == :dfa && @table[args[i]].kind != 'DFA'
465 end
466 references = roles.each_index.select { |i| [:reference, :dfa].include?(roles[i]) }
467 x = references.empty? ? nil : @table[args[0]]
468 z = references.length == 2 ? @table[args[1]] : nil
469 raise AutomatonError, 'alphabets differ' if z && x.alphabet != z.alphabet
470 case command
471 when 'LIST'
472 @table.empty? ? ['(none)'] : @table.map { |name, a| a.summary(name) }
473 when 'DELETE'
474 @table.delete(args[0])
475 ["deleted #{args[0]}"]
476 when 'ACCEPT'
477 word = args[1] == 'e' ? '' : args[1]
478 word.each_char do |c|
479 raise AutomatonError, "unknown symbol #{c}" unless x.alphabet.include?(c)
480 end
481 current = x.closure([x.start])
482 format_set = ->(s) { x.kind == 'DFA' ? x.states[s[0]] : x.set_text(s) }
483 trace = format_set.call(current)
484 word.each_char do |c|
485 current = x.move(current, c)
486 trace += " -#{c}-> #{format_set.call(current)}"
487 end
488 ["#{x.accepts_set?(current) ? 'ACCEPT' : 'REJECT'} #{trace}"]
489 when 'CLOSURE'
490 values = args[1].split(',').map do |name|
491 index = x.states.index(name)
492 raise AutomatonError, "unknown state #{name}" unless index
493 index
494 end
495 [x.set_text(x.closure(values))]
496 when 'TODFA'
497 result, subsets = x.determinize
498 register(args[1], result, subsets.each_with_index.map { |s, i| " D#{i} = #{x.set_text(s)}" })
499 when 'EPSFREE'
500 triples = []
501 accepting = []
502 x.states.each_index do |p|
503 initial = x.closure([p])
504 accepting << p if x.accepts_set?(initial)
505 x.alphabet.each do |c|
506 x.move(initial, c).each { |q| triples << [p, c, q] }
507 end
508 end
509 register(args[1], Automaton.new('NFA', x.states, x.alphabet, x.start, accepting, triples))
510 when 'MINIMIZE'
511 result, unreachable, rounds, blocks = x.minimize
512 details = [" unreachable: #{x.list_text(unreachable)}"]
513 rounds.each_with_index { |partition, i| details << " round #{i}: #{partition.map { |s| x.set_text(s) }.join(' ')}" }
514 blocks.each_with_index { |block, i| details << " M#{i} = #{x.set_text(block)}" }
515 register(args[1], result, details)
516 when 'EQUIV'
517 equivalent(x, z, args)
518 when 'UNION', 'INTERSECT', 'DIFF'
519 result, pairs = product(x, z, command)
520 details = pairs.each_with_index.map { |(p, q), i| " P#{i} = (#{x.states[p]},#{z.states[q]})" }
521 register(args[2], result, details)
522 when 'COMPLEMENT'
523 accepting = (0...x.states.length).to_a - x.accepting.to_a
524 register(args[1], Automaton.new('DFA', x.states, x.alphabet, x.start, accepting, x.triples))
525 when 'REVERSE'
526 triples = x.triples.map { |p, c, q| [q + 1, c, p + 1] }
527 x.accepting.each { |p| triples << [0, 'e', p + 1] }
528 register(args[1], Automaton.new('NFA', Array.new(x.states.length + 1) { |i| "N#{i}" }, x.alphabet, 0, [x.start + 1], triples))
529 when 'CONCAT'
530 offset = x.states.length
531 triples = x.triples + z.triples.map { |p, c, q| [p + offset, c, q + offset] }
532 x.accepting.each { |p| triples << [p, 'e', z.start + offset] }
533 accepting = z.accepting.map { |p| p + offset }
534 register(args[2], Automaton.new('NFA', Array.new(offset + z.states.length) { |i| "N#{i}" }, x.alphabet, x.start, accepting, triples))
535 when 'STAR'
536 triples = x.triples.map { |p, c, q| [p + 1, c, q + 1] }
537 triples << [0, 'e', x.start + 1]
538 x.accepting.each { |p| triples << [p + 1, 'e', x.start + 1] }
539 accepting = [0] + x.accepting.map { |p| p + 1 }
540 register(args[1], Automaton.new('NFA', Array.new(x.states.length + 1) { |i| "N#{i}" }, x.alphabet, 0, accepting, triples))
541 when 'ENUM'
542 dfa = x.determinize[0]
543 level = [[dfa.start, '']]
544 total = 0
545 examples = []
546 (0..args[1].to_i).each do |length|
547 level.each do |p, word|
548 next unless dfa.accepting.include?(p)
549 total += 1
550 examples << (word.empty? ? 'e' : word) if examples.length < 50
551 end
552 if length < args[1].to_i
553 level = level.flat_map { |p, word| dfa.alphabet.map { |c| [dfa.edges[p][c][0], word + c] } }
554 end
555 end
556 [[total.to_s, *examples, *(total > 50 ? ['...'] : [])].join(' ')]
557 when 'COUNT'
558 dfa = x.determinize[0]
559 counts = Array.new(dfa.states.length, 0)
560 counts[dfa.start] = 1
561 args[1].to_i.times do
562 next_counts = Array.new(counts.length, 0)
563 counts.each_with_index do |value, p|
564 dfa.alphabet.each { |c| next_counts[dfa.edges[p][c][0]] += value }
565 end
566 counts = next_counts
567 end
568 [dfa.accepting.sum { |p| counts[p] }.to_s]
569 when 'EMPTY'
570 word = x.determinize[0].shortest_word
571 [word.nil? ? 'EMPTY' : "NOT EMPTY #{word.empty? ? 'e' : word}"]
572 when 'FINITE'
573 size = x.determinize[0].finite_size
574 [size.nil? ? 'INFINITE' : "FINITE #{size}"]
575 when 'SYNC'
576 synchronization(x)
577 when 'INFO'
578 dfa = x.determinize[0]
579 size = dfa.finite_size(false)
580 epsilon = x.triples.count { |_, c, _| c == 'e' }
581 deterministic = epsilon.zero? && x.edges.all? { |row| x.alphabet.all? { |c| (row[c] || []).length <= 1 } }
582 complete = x.edges.all? { |row| x.alphabet.all? { |c| !(row[c] || []).empty? } }
583 language = size.nil? ? 'infinite' : size.zero? ? 'empty' : 'finite'
584 [x.summary(args[0]), " epsilon transitions: #{epsilon}",
585 " reachable: #{x.list_text(x.graph_reachable([x.start]))}",
586 " live: #{x.list_text(x.graph_reachable(x.accepting.to_a, true))}",
587 " deterministic: #{deterministic ? 'yes' : 'no'}", " complete: #{complete ? 'yes' : 'no'}",
588 " language: #{language}", " minimal DFA states: #{dfa.minimize[0].states.length}"]
589 when 'TABLE'
590 symbols = x.alphabet + (x.kind == 'NFA' && x.triples.any? { |_, c, _| c == 'e' } ? ['e'] : [])
591 rows = [[''] + symbols]
592 x.states.each_with_index do |name, p|
593 marker = (p == x.start ? '->' : ' ') + (x.accepting.include?(p) ? '*' : ' ') + name
594 row = [marker]
595 symbols.each do |c|
596 values = x.edges[p][c] || []
597 row << (x.kind == 'DFA' ? x.states[values[0]] : values.empty? ? '-' : x.set_text(values))
598 end
599 rows << row
600 end
601 widths = rows[0].each_index.map { |i| rows.map { |row| row[i].length }.max }
602 rows.map { |row| row.each_with_index.map { |cell, i| cell.ljust(widths[i]) }.join(' ').rstrip }
603 when 'SHOW'
604 accepting = x.accepting.empty? ? '-' : x.accepting.to_a.sort.map { |p| x.states[p] }.join(',')
605 words = [x.kind, args[0], x.states.join(','), x.alphabet.join, x.states[x.start], accepting]
606 x.triples.each { |p, c, q| words << "#{x.states[p]}:#{c}:#{x.states[q]}" }
607 [words.join(' ')]
608 end
609 end
610end
