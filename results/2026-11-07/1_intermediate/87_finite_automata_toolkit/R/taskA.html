79determinize <- function(a) {
80 subsets <- list(closure(a, a$start))
81 indices <- setNames(list(1L), state_key(subsets[[1]]))
82 triples <- list()
83 accepting <- integer()
84 i <- 1L
85 while (i <= length(subsets)) {
86 if (accepts_set(a, subsets[[i]])) accepting <- c(accepting, i)
87 for (symbol in a$alphabet) {
88 target <- move(a, subsets[[i]], symbol)
89 key <- state_key(target)
90 if (is.null(indices[[key]])) {
91 indices[[key]] <- length(subsets) + 1L
92 subsets[[length(subsets) + 1L]] <- target
93 }
94 triples[[length(triples) + 1L]] <- list(i, symbol, indices[[key]])
95 }
96 i <- i + 1L
97 }
98 result <- new_automaton("DFA", paste0("D", seq_along(subsets) - 1L), a$alphabet, 1L, accepting, triples)
99 list(automaton = result, subsets = subsets)
100}

102minimize <- function(a) {
103 reached <- reachable(a, a$start)
104 blocks <- list(reached[reached %in% a$accepting], reached[!(reached %in% a$accepting)])
105 blocks <- Filter(function(block) length(block) > 0L, blocks)
106 blocks <- blocks[order(vapply(blocks, function(block) block[[1]], integer(1)))]
107 rounds <- list(blocks)
108 repeat {
109 membership <- integer(length(a$states))
110 for (i in seq_along(blocks)) membership[blocks[[i]]] <- i
111 groups <- list()
112 for (p in reached) {
113 signature <- c(membership[[p]], vapply(a$alphabet, function(symbol) membership[[a$edges[[p]][[symbol]][[1]]]], integer(1)))
114 key <- state_key(signature)
115 groups[[key]] <- c(groups[[key]], p)
116 }
117 refined <- unname(groups)
118 refined <- refined[order(vapply(refined, function(block) block[[1]], integer(1)))]
119 if (length(refined) == length(blocks)) break
120 blocks <- refined
121 rounds[[length(rounds) + 1L]] <- blocks
122 }
123 membership <- integer(length(a$states))
124 for (i in seq_along(blocks)) membership[blocks[[i]]] <- i
125 block_order <- membership[[a$start]]
126 named <- setNames(list(1L), state_key(block_order))
127 triples <- list()
128 accepting <- integer()
129 i <- 1L
130 while (i <= length(block_order)) {
131 p <- blocks[[block_order[[i]]]][[1]]
132 if (p %in% a$accepting) accepting <- c(accepting, i)
133 for (symbol in a$alphabet) {
134 target <- membership[[a$edges[[p]][[symbol]][[1]]]]
135 key <- state_key(target)
136 if (is.null(named[[key]])) {
137 named[[key]] <- length(block_order) + 1L
138 block_order <- c(block_order, target)
139 }
140 triples[[length(triples) + 1L]] <- list(i, symbol, named[[key]])
141 }
142 i <- i + 1L
143 }
144 result <- new_automaton("DFA", paste0("M", seq_along(block_order) - 1L), a$alphabet, 1L, accepting, triples)
145 list(automaton = result, unreachable = setdiff(seq_along(a$states), reached), rounds = rounds, blocks = blocks[block_order])
146}

168finite_size <- function(a, count = TRUE) {
169 live <- reachable(a, a$accepting, TRUE)
170 color <- integer(length(a$states))
171 counts <- numeric(length(a$states))
172 visit <- function(p) {
173 if (!(p %in% live)) return(0)
174 if (color[[p]] == 1L) return(NULL)
175 if (color[[p]] == 2L) return(counts[[p]])
176 color[[p]] <<- 1L
177 total <- as.numeric(p %in% a$accepting)
178 for (symbol in a$alphabet) {
179 value <- visit(a$edges[[p]][[symbol]][[1]])
180 if (is.null(value)) return(NULL)
181 total <- if (count) total + value else as.numeric(total > 0 || value > 0)
182 }
183 color[[p]] <<- 2L
184 counts[[p]] <<- total
185 total
186 }
187 visit(a$start)
188}

190parse_regex <- function(text) {
191 position <- 1L
192 length <- nchar(text, type = "bytes")
193 peek <- function() if (position > length) "" else substr(text, position, position)
194 fail <- function() automaton_error(paste0("syntax error at ", position))
195 atom <- function() {
196 c <- peek()
197 if (grepl("^[a-z0-9]$", c)) {
198 position <<- position + 1L
199 return(list(kind = "symbol", value = c))
200 }
201 if (c == "(") {
202 position <<- position + 1L
203 result <- alternation()
204 if (peek() != ")") fail()
205 position <<- position + 1L
206 return(result)
207 }
208 fail()
209 }
210 factor <- function() {
211 result <- atom()
212 while (peek() %in% c("*", "+", "?")) {
213 operator <- peek()
214 position <<- position + 1L
215 result <- list(kind = operator, child = result)
216 }
217 result
218 }
219 concatenation <- function() {
220 parts <- list(factor())
221 while (grepl("^[a-z0-9(]$", peek())) parts[[length(parts) + 1L]] <- factor()
222 if (length(parts) == 1L) parts[[1]] else list(kind = "concatenation", parts = parts)
223 }
224 alternation <- function() {
225 parts <- list(concatenation())
226 while (peek() == "|") {
227 position <<- position + 1L
228 parts[[length(parts) + 1L]] <- concatenation()
229 }
230 if (length(parts) == 1L) parts[[1]] else list(kind = "alternation", parts = parts)
231 }
232 result <- alternation()
233 if (peek() != "") fail()
234 result
235}

259 seen <- c(seen, state)
260 }
261 start <- match(args[[4]], states)
262 if (is.na(start)) automaton_error(paste0("unknown state ", args[[4]]))
263 accepting <- integer()
264 if (args[[5]] != "-") {
265 for (state in strsplit(args[[5]], ",", fixed = TRUE)[[1]]) {
266 index <- match(state, states)
267 if (is.na(index)) automaton_error(paste0("unknown state ", state))
268 if (index %in% accepting) automaton_error(paste0("duplicate state ", state))
269 accepting <- c(accepting, index)

274 seen <- character()

281 if (is.na(p)) automaton_error(paste0("unknown state ", parts[[1]]))
282 if (symbol == "e" && kind == "DFA") automaton_error("epsilon transition in DFA")
283 if (symbol != "e" && !(symbol %in% alphabet)) automaton_error(paste0("unknown symbol ", symbol))
284 if (is.na(q)) automaton_error(paste0("unknown state ", parts[[3]]))

288 seen <- c(seen, word)

292 if (kind == "DFA") {

321 s <- create()

352 product <- function(x, z, operation) {
353 pairs <- list(c(x$start, z$start))
354 indices <- setNames(list(1L), state_key(pairs[[1]]))
355 triples <- list()
356 accepting <- integer()
357 i <- 1L
358 while (i <= length(pairs)) {
359 p <- pairs[[i]][[1]]
360 q <- pairs[[i]][[2]]
361 a <- p %in% x$accepting
362 b <- q %in% z$accepting
363 good <- if (operation == "UNION") a || b else if (operation == "INTERSECT") a && b else a && !b
364 if (good) accepting <- c(accepting, i)
365 for (symbol in x$alphabet) {
366 target <- c(x$edges[[p]][[symbol]][[1]], z$edges[[q]][[symbol]][[1]])
367 key <- state_key(target)
368 if (is.null(indices[[key]])) {
369 indices[[key]] <- length(pairs) + 1L
370 pairs[[length(pairs) + 1L]] <- target
371 }
372 triples[[length(triples) + 1L]] <- list(i, symbol, indices[[key]])
373 }
374 i <- i + 1L
375 }
376 result <- new_automaton("DFA", paste0("P", seq_along(pairs) - 1L), x$alphabet, 1L, accepting, triples)
377 list(automaton = result, pairs = pairs)
378 }
379 equivalent <- function(x, z, names) {
380 a <- determinize(x)$automaton
381 b <- determinize(z)$automaton
382 queue <- list(list(a$start, b$start, ""))
383 seen <- state_key(c(a$start, b$start))
384 i <- 1L
385 while (i <= length(queue)) {
386 p <- queue[[i]][[1]]
387 q <- queue[[i]][[2]]
388 word <- queue[[i]][[3]]
389 if ((p %in% a$accepting) != (q %in% b$accepting)) {
390 owner <- if (p %in% a$accepting) names[[1]] else names[[2]]
391 return(paste0("NOT EQUIVALENT: ", if (word == "") "e" else word, " is accepted only by ", owner))
392 }
393 for (symbol in a$alphabet) {
394 target <- c(a$edges[[p]][[symbol]][[1]], b$edges[[q]][[symbol]][[1]])
395 key <- state_key(target)
396 if (!(key %in% seen)) {
397 seen <- c(seen, key)
398 queue[[length(queue) + 1L]] <- list(target[[1]], target[[2]], paste0(word, symbol))
399 }
400 }
401 i <- i + 1L
402 }
403 "EQUIVALENT"
404 }
405 synchronization <- function(x) {
406 initial <- seq_along(x$states)
407 queue <- list(list(initial, ""))
408 seen <- state_key(initial)
409 i <- 1L
410 while (i <= length(queue)) {
411 subset <- queue[[i]][[1]]
412 word <- queue[[i]][[2]]
413 if (length(subset) == 1L) return(paste0("SYNCHRONIZING ", if (word == "") "e" else word, " -> ", x$states[[subset[[1]]]]))
414 for (symbol in x$alphabet) {
415 target <- sort(unique(vapply(subset, function(p) x$edges[[p]][[symbol]][[1]], integer(1))))
416 key <- state_key(target)
417 if (!(key %in% seen)) {
418 seen <- c(seen, key)
419 queue[[length(queue) + 1L]] <- list(target, paste0(word, symbol))
420 }
421 }
422 i <- i + 1L
423 }
424 "NOT SYNCHRONIZING"
425 }
426 execute <- function(command, args) {
427 if (command %in% c("DFA", "NFA")) return(define(command, args))
428 if (command == "REGEX") return(regex(args))
429 if (!(command %in% names(roles))) automaton_error(paste0("unknown command ", command))
430 command_roles <- roles[[command]]
431 if (length(command_roles) != length(args)) automaton_error("wrong number of arguments")
432 references <- which(command_roles %in% c("ref", "dfa"))
433 for (i in references) if (is.null(table[[args[[i]]]])) automaton_error(paste0("unknown automaton ", args[[i]]))
434 for (i in which(command_roles == "new")) if (!is.null(table[[args[[i]]]])) automaton_error(paste0(args[[i]], " already exists"))
435 for (i in which(command_roles == "dfa")) if (table[[args[[i]]]]$kind != "DFA") automaton_error(paste0(args[[i]], " is not a DFA"))
436 x <- if (length(references) == 0L) NULL else table[[args[[1]]]]
437 z <- if (length(references) == 2L) table[[args[[2]]]] else NULL
438 if (!is.null(z) && !identical(x$alphabet, z$alphabet)) automaton_error("alphabets differ")
439 if (command == "LIST") {
440 if (length(table) == 0L) return("(none)")
441 return(vapply(names(table), function(name) summary_line(table[[name]], name), character(1)))
442 }
443 if (command == "DELETE") {
444 table[[args[[1]]]] <<- NULL
445 return(paste0("deleted ", args[[1]]))
446 }
447 if (command == "ACCEPT") {
448 word <- if (args[[2]] == "e") "" else args[[2]]
449 for (symbol in chars(word)) if (!(symbol %in% x$alphabet)) automaton_error(paste0("unknown symbol ", symbol))
450 current <- closure(x, x$start)
451 format_set <- function(s) if (x$kind == "DFA") x$states[[s[[1]]]] else set_text(x, s)
452 trace <- format_set(current)
453 for (symbol in chars(word)) {
454 current <- move(x, current, symbol)
455 trace <- paste0(trace, " -", symbol, "-> ", format_set(current))
456 }
457 return(paste0(if (accepts_set(x, current)) "ACCEPT " else "REJECT ", trace))
458 }
459 if (command == "CLOSURE") {
460 values <- integer()
461 for (name in strsplit(args[[2]], ",", fixed = TRUE)[[1]]) {
462 index <- match(name, x$states)
463 if (is.na(index)) automaton_error(paste0("unknown state ", name))
464 values <- c(values, index)
465 }
466 return(set_text(x, closure(x, values)))
467 }
468 if (command == "TODFA") {
469 result <- determinize(x)
470 details <- vapply(seq_along(result$subsets), function(i) paste0(" D", i - 1L, " = ", set_text(x, result$subsets[[i]])), character(1))
471 return(register(args[[2]], result$automaton, details))
472 }
473 if (command == "EPSFREE") {
474 triples <- list()
475 accepting <- integer()
476 for (p in seq_along(x$states)) {
477 initial <- closure(x, p)
478 if (accepts_set(x, initial)) accepting <- c(accepting, p)
479 for (symbol in x$alphabet) {
480 for (q in move(x, initial, symbol)) triples[[length(triples) + 1L]] <- list(p, symbol, q)
481 }
482 }
483 return(register(args[[2]], new_automaton("NFA", x$states, x$alphabet, x$start, accepting, triples)))
484 }
485 if (command == "MINIMIZE") {
486 result <- minimize(x)
487 details <- paste0(" unreachable: ", list_text(x, result$unreachable))
488 for (i in seq_along(result$rounds)) details <- c(details, paste0(" round ", i - 1L, ": ", paste(vapply(result$rounds[[i]], function(block) set_text(x, block), character(1)), collapse = " ")))
489 for (i in seq_along(result$blocks)) details <- c(details, paste0(" M", i - 1L, " = ", set_text(x, result$blocks[[i]])))
490 return(register(args[[2]], result$automaton, details))
491 }
492 if (command == "EQUIV") return(equivalent(x, z, args))
493 if (command %in% c("UNION", "INTERSECT", "DIFF")) {
494 result <- product(x, z, command)
495 details <- vapply(seq_along(result$pairs), function(i) {
496 pair <- result$pairs[[i]]
497 paste0(" P", i - 1L, " = (", x$states[[pair[[1]]]], ",", z$states[[pair[[2]]]], ")")
498 }, character(1))
499 return(register(args[[3]], result$automaton, details))
500 }
501 if (command == "COMPLEMENT") return(register(args[[2]], new_automaton("DFA", x$states, x$alphabet, x$start, setdiff(seq_along(x$states), x$accepting), all_triples(x))))
502 if (command == "REVERSE") {
503 triples <- lapply(all_triples(x), function(t) list(t[[3]] + 1L, t[[2]], t[[1]] + 1L))
504 for (p in x$accepting) triples[[length(triples) + 1L]] <- list(1L, "e", p + 1L)
505 return(register(args[[2]], new_automaton("NFA", paste0("N", 0:length(x$states)), x$alphabet, 1L, x$start + 1L, triples)))
506 }
507 if (command == "CONCAT") {
508 offset <- length(x$states)
509 triples <- c(all_triples(x), lapply(all_triples(z), function(t) list(t[[1]] + offset, t[[2]], t[[3]] + offset)))
510 for (p in x$accepting) triples[[length(triples) + 1L]] <- list(p, "e", z$start + offset)
511 return(register(args[[3]], new_automaton("NFA", paste0("N", seq_len(offset + length(z$states)) - 1L), x$alphabet, x$start, z$accepting + offset, triples)))
512 }
513 if (command == "STAR") {
514 triples <- lapply(all_triples(x), function(t) list(t[[1]] + 1L, t[[2]], t[[3]] + 1L))
515 triples[[length(triples) + 1L]] <- list(1L, "e", x$start + 1L)
516 for (p in x$accepting) triples[[length(triples) + 1L]] <- list(p + 1L, "e", x$start + 1L)
517 return(register(args[[2]], new_automaton("NFA", paste0("N", 0:length(x$states)), x$alphabet, 1L, c(1L, x$accepting + 1L), triples)))
518 }
519 if (command == "ENUM") {
520 dfa <- determinize(x)$automaton
521 level <- list(list(dfa$start, ""))
522 total <- 0
523 examples <- character()
524 limit <- as.integer(args[[2]])
525 for (length in 0:limit) {
526 for (entry in level) {
527 if (entry[[1]] %in% dfa$accepting) {
528 total <- total + 1
529 if (length(examples) < 50L) examples <- c(examples, if (entry[[2]] == "") "e" else entry[[2]])
530 }
531 }
532 if (length < limit) {
533 next_level <- list()
534 for (entry in level) {
535 for (symbol in dfa$alphabet) next_level[[length(next_level) + 1L]] <- list(dfa$edges[[entry[[1]]]][[symbol]][[1]], paste0(entry[[2]], symbol))
536 }
537 level <- next_level
538 }
539 }
540 return(paste(c(integer_text(total), examples, if (total > 50) "..." else character()), collapse = " "))
541 }
542 if (command == "COUNT") {
543 dfa <- determinize(x)$automaton
544 counts <- numeric(length(dfa$states))
