20vector<string> split(const string& text, char separator) {
21 vector<string> result;
22 size_t start = 0;
23 for (;;) {
24 size_t end = text.find(separator, start);
25 result.push_back(text.substr(start, end == string::npos ? end : end - start));
26 if (end == string::npos) return result;
27 start = end + 1;
28 }
29}
30string join(const vector<string>& parts, const string& separator) {
31 string result;
32 for (const string& part : parts) {
33 if (!result.empty()) result += separator;
34 result += part;
35 }
36 return result;
37}
38string printedWord(const string& word) {
39 return word.empty() ? "e" : word;

41struct Automaton {
42 bool dfa = false;
43 string alphabet;
44 vector<string> names;
45 int start = 0;
46 States accepting;
47 vector<map<char, States>> transitions;
48 int addState(string name) {
49 int index = names.size();
50 names.push_back(move(name));
51 transitions.emplace_back();
52 return index;
53 }
54 void series(int count, char prefix) {
55 for (int i = 0; i < count; ++i) addState(string(1, prefix) + to_string(i));
56 }
57 void edge(int from, char symbol, int to) {
58 transitions[from][symbol].insert(to);
59 }
60 const States& destinations(int state, char symbol) const {
61 static const States empty;
62 auto found = transitions[state].find(symbol);
63 return found == transitions[state].end() ? empty : found->second;
64 }
65 int delta(int state, char symbol) const {
66 return *destinations(state, symbol).begin();
67 }
68 States closure(States states) const {
69 vector<int> pending(states.begin(), states.end());
70 for (size_t i = 0; i < pending.size(); ++i) for (int q : destinations(pending[i], 'e')) {
71 if (states.insert(q).second) pending.push_back(q);
72 }
73 return states;
74 }
75 States initial() const {
76 return closure({start});
77 }
78 States moveOn(const States& states, char symbol) const {
79 States result;
80 for (int p : states) {
81 const auto& qs = destinations(p, symbol);
82 result.insert(qs.begin(), qs.end());
83 }
84 return closure(move(result));
85 }
86 bool accepts(const States& states) const {
87 for (int p : states) if (accepting.count(p)) return true;
88 return false;
89 }
90 string list(const States& states) const {
91 vector<string> parts;
92 for (int p : states) parts.push_back(names[p]);
93 return parts.empty() ? "none" : join(parts, ",");
94 }
95 string setText(const States& states) const {
96 return states.empty() ? "{}" : "{" + list(states) + "}";
97 }
98 int transitionCount(char only = 0) const {
99 int result = 0;
100 for (const auto& row : transitions) for (const auto& edge : row) if (!only || edge.first == only) result += edge.second.size();
101 return result;
102 }
103 string summary(const string& name) const {
104 return string(dfa ? "DFA " : "NFA ") + name + ": alphabet=" + alphabet + " states=" + to_string(names.size())
105 + " accepting=" + to_string(accepting.size()) + " transitions=" + to_string(transitionCount());
106 }
107};
108pair<Automaton, vector<States>> determinize(const Automaton& source) {
109 Automaton result;
110 result.dfa = true;
111 result.alphabet = source.alphabet;
112 vector<States> subsets{source.initial()};
113 map<States, int> indices{{subsets[0], 0}};
114 result.addState("D0");
115 for (size_t i = 0; i < subsets.size(); ++i) {
116 if (source.accepts(subsets[i])) result.accepting.insert(i);
117 for (char c : source.alphabet) {
118 States destination = source.moveOn(subsets[i], c);
119 auto found = indices.find(destination);
120 int index;
121 if (found == indices.end()) {
122 index = subsets.size();
123 indices[destination] = index;
124 subsets.push_back(destination);
125 result.addState("D" + to_string(index));
126 } else index = found->second;
127 result.edge(i, c, index);
128 }
129 }
130 return {move(result), move(subsets)};
131}
132States graphReach(const Automaton& a, bool backwards) {
133 vector<States> adjacency(a.names.size());
134 for (size_t p = 0; p < a.names.size(); ++p) for (const auto& edge : a.transitions[p]) for (int q : edge.second) {
135 if (backwards) adjacency[q].insert(p);
136 else adjacency[p].insert(q);
137 }
138 States result = backwards ? a.accepting : States{a.start};
139 vector<int> pending(result.begin(), result.end());
140 for (size_t i = 0; i < pending.size(); ++i) for (int q : adjacency[pending[i]]) {
141 if (result.insert(q).second) pending.push_back(q);
142 }
143 return result;
144}
145pair<Automaton, vector<string>> minimize(const Automaton& a) {
146 States reachable = graphReach(a, false), unreachable, yes, no;
147 for (size_t p = 0; p < a.names.size(); ++p) {
148 if (!reachable.count(p)) unreachable.insert(p);
149 else (a.accepting.count(p) ? yes : no).insert(p);
150 }
151 vector<string> details{" unreachable: " + a.list(unreachable)};
152 vector<States> blocks;
153 if (!yes.empty()) blocks.push_back(yes);
154 if (!no.empty()) blocks.push_back(no);
155 auto orderBlocks = [](vector<States>& bs) {
156 sort(bs.begin(), bs.end(), [](const States& x, const States& y) {
157 return *x.begin() < *y.begin();
158 });
159 };
160 auto showBlocks = [&](const vector<States>& bs) {
161 vector<string> parts;
162 for (const States& block : bs) parts.push_back(a.setText(block));
163 return join(parts, " ");
164 };
165 orderBlocks(blocks);
166 details.push_back(" round 0: " + showBlocks(blocks));
167 vector<int> membership(a.names.size());
168 for (int round = 1;; ++round) {
169 for (size_t i = 0; i < blocks.size(); ++i) for (int p : blocks[i]) membership[p] = i;
170 vector<States> refined;
171 for (const States& block : blocks) {
172 map<vector<int>, States> groups;
173 for (int p : block) {
174 vector<int> signature;
175 for (char c : a.alphabet) signature.push_back(membership[a.delta(p, c)]);
176 groups[signature].insert(p);
177 }
178 for (const auto& group : groups) refined.push_back(group.second);
179 }
180 if (refined.size() == blocks.size()) break;
181 orderBlocks(refined);
182 blocks = move(refined);
183 details.push_back(" round " + to_string(round) + ": " + showBlocks(blocks));
184 }
185 for (size_t i = 0; i < blocks.size(); ++i) for (int p : blocks[i]) membership[p] = i;
186 Automaton result;
187 result.dfa = true;
188 result.alphabet = a.alphabet;
189 vector<int> sequence{membership[a.start]};
190 map<int, int> indices{{sequence[0], 0}};
191 result.addState("M0");
192 for (size_t i = 0; i < sequence.size(); ++i) {
193 const States block = blocks[sequence[i]];
194 int representative = *block.begin();
195 if (a.accepting.count(representative)) result.accepting.insert(i);
196 for (char c : a.alphabet) {
197 int destination = membership[a.delta(representative, c)];
198 auto found = indices.find(destination);
199 int index;
200 if (found == indices.end()) {
201 index = sequence.size();
202 indices[destination] = index;
203 sequence.push_back(destination);
204 result.addState("M" + to_string(index));
205 } else index = found->second;
206 result.edge(i, c, index);
207 }
208 details.push_back(" M" + to_string(i) + " = " + a.setText(block));
209 }
210 return {move(result), move(details)};
211}
212struct Language {
213 bool infinite = false;
214 Count count = 0;
215 optional<string> smallest;
216};
217Language inspectLanguage(const Automaton& input, bool needCount = false) {
218 Automaton a = determinize(input).first;
219 Language result;
220 vector<bool> seen(a.names.size());
221 vector<pair<int, string>> pending{{0, ""}};
222 seen[0] = true;
223 for (size_t i = 0; i < pending.size(); ++i) {
224 int p = pending[i].first;
225 string w = pending[i].second;
226 if (!result.smallest && a.accepting.count(p)) result.smallest = w;
227 for (char c : a.alphabet) {
228 int q = a.delta(p, c);
229 if (!seen[q]) {
230 seen[q] = true;
231 pending.emplace_back(q, w + c);
232 }
233 }
234 }
235 if (!result.smallest) return result;
236 States live = graphReach(a, true);
237 vector<int> color(a.names.size());
238 vector<Count> count(a.names.size());
239 function<Count(int)> visit = [&](int p) -> Count {
240 if (!live.count(p)) return 0;
241 if (color[p] == 1) {
242 result.infinite = true;
243 return 0;
244 }
245 if (color[p] == 2) return count[p];
246 color[p] = 1;
247 Count total = needCount ? a.accepting.count(p) : 0;
248 for (char c : a.alphabet) {
249 Count suffixes = visit(a.delta(p, c));
250 if (needCount && !result.infinite) total += suffixes;
251 }
252 color[p] = 2;
253 return count[p] = total;
254 };
255 result.count = visit(0);
256 return result;
257}
258struct RegexNode {
259 char op;
260 vector<shared_ptr<RegexNode>> children;
261};
262using Regex = shared_ptr<RegexNode>;
263class RegexParser {
264 const string& text;
265 size_t index = 0;
266 char next() const {
267 return index < text.size() ? text[index] : '\0';
268 }
269 [[noreturn]] void error() const {
270 throw runtime_error("syntax error at " + to_string(index + 1));
271 }
272 bool atomStart(char c) const {
273 return (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') || c == '(';
274 }
275 Regex atom() {
276 char c = next();
277 if (c == '(') {
278 ++index;
279 auto result = alternation();
280 if (next() != ')') error();
281 ++index;
282 return result;
283 }
284 if (!atomStart(c)) error();
285 ++index;
286 return make_shared<RegexNode>(RegexNode{c, {}});
287 }
288 Regex factor() {
289 auto result = atom();
290 while (next() == '*' || next() == '+' || next() == '?') {
291 char op = text[index++];
292 result = make_shared<RegexNode>(RegexNode{op, {result}});
293 }
294 return result;
295 }
296 Regex concatenation() {
297 vector<Regex> parts{factor()};
298 while (atomStart(next())) parts.push_back(factor());
299 return parts.size() == 1 ? parts[0] : make_shared<RegexNode>(RegexNode{'.', move(parts)});
300 }
301 Regex alternation() {
302 vector<Regex> parts{concatenation()};
303 while (next() == '|') {
304 ++index;
305 parts.push_back(concatenation());
306 }
307 return parts.size() == 1 ? parts[0] : make_shared<RegexNode>(RegexNode{'|', move(parts)});
308 }
309public:
310 explicit RegexParser(const string& s) : text(s) {
311 }
312 Regex parse() {
313 auto result = alternation();
314 if (index != text.size()) error();
315 return result;
316 }
317};
318Automaton thompson(const Regex& root, const string& alphabet) {
319 Automaton a;
320 a.alphabet = alphabet;
321 auto state = [&]() {
322 return a.addState("N" + to_string(a.names.size()));
323 };
324 function<pair<int, int>(const Regex&)> build = [&](const Regex& r) -> pair<int, int> {
325 if (r->children.empty()) {
326 int s = state(), f = state();
327 a.edge(s, r->op, f);
328 return {s, f};
329 }
330 if (r->op == '.') {
331 auto first = build(r->children[0]);
332 int last = first.second;
333 for (size_t i = 1; i < r->children.size(); ++i) {
334 auto next = build(r->children[i]);
335 a.edge(last, 'e', next.first);
336 last = next.second;
337 }
338 return {first.first, last};
339 }
340 int s = state();
341 vector<pair<int, int>> parts;
342 for (const Regex& child : r->children) parts.push_back(build(child));
343 int f = state();
344 for (auto part : parts) {
345 a.edge(s, 'e', part.first);
346 a.edge(part.second, 'e', f);
347 }
348 if (r->op != '|') {
349 if (r->op != '+') a.edge(s, 'e', f);
350 if (r->op != '?') a.edge(parts[0].second, 'e', parts[0].first);
351 }
352 return {s, f};
353 };
354 auto ends = build(root);
355 a.start = ends.first;
356 a.accepting.insert(ends.second);
357 return a;
358}
359class Toolkit {
360 vector<pair<string, Automaton>> table;
361 int indexOf(const string& name) const {
362 for (size_t i = 0; i < table.size(); ++i) if (table[i].first == name) return i;
363 return -1;
364 }
365 void add(const string& name, Automaton a, const vector<string>& details, ostream& out) {
366 out << a.summary(name) << '\n';
367 for (const string& detail : details) out << detail << '\n';
368 table.emplace_back(name, move(a));
369 }
370 int stateIndex(const Automaton& a, const string& name) const {
371 auto found = find(a.names.begin(), a.names.end(), name);
372 if (found == a.names.end()) throw runtime_error("unknown state " + name);
373 return found - a.names.begin();
374 }
375 void definition(const string& command, const vector<string>& args, ostream& out) {
376 if (args.size() < 5) throw runtime_error("wrong number of arguments");
377 if (indexOf(args[0]) >= 0) throw runtime_error(args[0] + " already exists");
378 Automaton a;
379 a.dfa = command == "DFA";
380 a.alphabet = args[2];
381 sort(a.alphabet.begin(), a.alphabet.end());
382 for (const string& name : split(args[1], ',')) {
383 if (find(a.names.begin(), a.names.end(), name) != a.names.end()) throw runtime_error("duplicate state " + name);
384 a.addState(name);
385 }
386 a.start = stateIndex(a, args[3]);
387 if (args[4] != "-") for (const string& name : split(args[4], ',')) {
388 int index = stateIndex(a, name);
389 if (!a.accepting.insert(index).second) throw runtime_error("duplicate state " + name);
390 }
391 for (size_t i = 5; i < args.size(); ++i) {
392 auto parts = split(args[i], ':');
393 int p = stateIndex(a, parts[0]);
394 char c = parts[1][0];
395 if (c == 'e' && a.dfa) throw runtime_error("epsilon transition in DFA");
396 if (c != 'e' && a.alphabet.find(c) == string::npos) throw runtime_error("unknown symbol " + string(1, c));
397 int q = stateIndex(a, parts[2]);
398 const auto& previous = a.destinations(p, c);
399 if (previous.count(q)) throw runtime_error("duplicate transition " + args[i]);
400 if (a.dfa && !previous.empty()) throw runtime_error("conflicting transition " + args[i]);
401 a.edge(p, c, q);
402 }
403 if (a.dfa) for (size_t p = 0; p < a.names.size(); ++p) for (char c : a.alphabet) {
404 if (a.destinations(p, c).empty()) throw runtime_error("missing transition " + a.names[p] + ":" + c);
405 }
406 add(args[0], move(a), {}, out);
407 }
408public:
409 void execute(const string& line, ostream& out) {
410 if (line[0] == '#') return;
411 auto tokens = split(line, ' ');
412 string command = tokens[0];
413 vector<string> args(tokens.begin() + 1, tokens.end());
414 const map<string, vector<char>> roles = {
415 {"ACCEPT", {'r','o'}}, {"CLOSURE", {'r','o'}}, {"ENUM", {'r','o'}}, {"COUNT", {'r','o'}},
416 {"TODFA", {'r','n'}}, {"EPSFREE", {'r','n'}}, {"REVERSE", {'r','n'}}, {"STAR", {'r','n'}},
417 {"MINIMIZE", {'d','n'}}, {"COMPLEMENT", {'d','n'}}, {"EQUIV", {'r','r'}},
418 {"UNION", {'d','d','n'}}, {"INTERSECT", {'d','d','n'}}, {"DIFF", {'d','d','n'}}, {"CONCAT", {'r','r','n'}},
419 {"EMPTY", {'r'}}, {"FINITE", {'r'}}, {"INFO", {'r'}}, {"TABLE", {'r'}}, {"SHOW", {'r'}},
420 {"DELETE", {'r'}}, {"SYNC", {'d'}}, {"LIST", {}}
421 };
422 if (command == "DFA" || command == "NFA") {
423 definition(command, args, out);
424 return;
425 }
426 if (command == "REGEX") {
427 if (args.size() != 3) throw runtime_error("wrong number of arguments");
428 if (indexOf(args[0]) >= 0) throw runtime_error(args[0] + " already exists");
429 string alphabet = args[1];
430 sort(alphabet.begin(), alphabet.end());
431 for (char c : args[2]) if (((c >= 'a' && c <= 'z') || (c >= '0' && c <= '9')) && c != 'e' && alphabet.find(c) == string::npos) {
432 throw runtime_error("unknown symbol " + string(1, c));
433 }
434 auto expression = RegexParser(args[2]).parse();
435 add(args[0], thompson(expression, alphabet), {}, out);
436 return;
437 }
438 auto found = roles.find(command);
439 if (found == roles.end()) throw runtime_error("unknown command " + command);
440 if (args.size() != found->second.size()) throw runtime_error("wrong number of arguments");
441 const auto& rs = found->second;
442 for (size_t i = 0; i < rs.size(); ++i) if ((rs[i] == 'r' || rs[i] == 'd') && indexOf(args[i]) < 0) {
443 throw runtime_error("unknown automaton " + args[i]);
444 }
445 for (size_t i = 0; i < rs.size(); ++i) if (rs[i] == 'n' && indexOf(args[i]) >= 0) throw runtime_error(args[i] + " already exists");
446 for (size_t i = 0; i < rs.size(); ++i) if (rs[i] == 'd' && !table[indexOf(args[i])].second.dfa) throw runtime_error(args[i] + " is not a DFA");
447 if (command == "LIST") {
448 if (table.empty()) out << "(none)\n";
449 for (const auto& entry : table) out << entry.second.summary(entry.first) << '\n';
450 return;
451 }
452 Automaton a = table[indexOf(args[0])].second;
453 Automaton b;
454 bool binary = command == "EQUIV" || command == "UNION" || command == "INTERSECT" || command == "DIFF" || command == "CONCAT";
455 if (binary) {
456 b = table[indexOf(args[1])].second;
457 if (a.alphabet != b.alphabet) throw runtime_error("alphabets differ");
458 }
459 if (command == "ACCEPT") {
460 string w = args[1] == "e" ? "" : args[1];
461 for (char c : w) if (a.alphabet.find(c) == string::npos) throw runtime_error("unknown symbol " + string(1, c));
462 States current = a.initial();
463 auto text = [&](const States& s) {
464 return a.dfa ? a.names[*s.begin()] : a.setText(s);
465 };
466 string trace = text(current);
467 for (char c : w) {
468 current = a.moveOn(current, c);
469 trace += " -" + string(1, c) + "-> " + text(current);
470 }
471 out << (a.accepts(current) ? "ACCEPT " : "REJECT ") << trace << '\n';
472 } else if (command == "CLOSURE") {
473 States states;
474 for (const string& name : split(args[1], ',')) states.insert(stateIndex(a, name));
475 out << a.setText(a.closure(move(states))) << '\n';
476 } else if (command == "TODFA") {
477 auto converted = determinize(a);
478 vector<string> details;
479 for (size_t i = 0; i < converted.second.size(); ++i) details.push_back(" D" + to_string(i) + " = " + a.setText(converted.second[i]));
480 add(args[1], move(converted.first), details, out);
481 } else if (command == "MINIMIZE") {
482 auto minimized = minimize(a);
483 add(args[1], move(minimized.first), minimized.second, out);
484 } else if (command == "EPSFREE") {
485 Automaton result;
486 result.alphabet = a.alphabet;
487 for (const string& name : a.names) result.addState(name);
488 result.start = a.start;
489 for (size_t p = 0; p < a.names.size(); ++p) {
490 auto closed = a.closure({static_cast<int>(p)});
491 if (a.accepts(closed)) result.accepting.insert(p);
492 for (char c : a.alphabet) for (int q : a.moveOn(closed, c)) result.edge(p, c, q);
493 }
494 add(args[1], move(result), {}, out);
495 } else if (command == "COMPLEMENT") {
496 States accepting;
497 for (size_t p = 0; p < a.names.size(); ++p) if (!a.accepting.count(p)) accepting.insert(p);
498 a.accepting = move(accepting);
499 add(args[1], move(a), {}, out);
500 } else if (command == "UNION" || command == "INTERSECT" || command == "DIFF") {
501 Automaton result;
502 result.dfa = true;
503 result.alphabet = a.alphabet;
504 vector<pair<int, int>> pairs{{a.start, b.start}};
505 map<pair<int, int>, int> indices{{pairs[0], 0}};
506 result.addState("P0");
507 vector<string> details;
508 for (size_t i = 0; i < pairs.size(); ++i) {
509 auto [p, q] = pairs[i];
510 bool ap = a.accepting.count(p), bq = b.accepting.count(q);
511 if (command == "UNION" ? ap || bq : command == "INTERSECT" ? ap && bq : ap && !bq) result.accepting.insert(i);
512 for (char c : a.alphabet) {
513 auto destination = make_pair(a.delta(p, c), b.delta(q, c));
514 auto present = indices.find(destination);
515 int index;
516 if (present == indices.end()) {
517 index = pairs.size();
518 indices[destination] = index;
519 pairs.push_back(destination);
520 result.addState("P" + to_string(index));
521 } else index = present->second;
522 result.edge(i, c, index);
523 }
524 details.push_back(" P" + to_string(i) + " = (" + a.names[p] + "," + b.names[q] + ")");
525 }
526 add(args[2], move(result), details, out);
527 } else if (command == "EQUIV") {
528 a = determinize(a).first;
529 b = determinize(b).first;
530 vector<tuple<int, int, string>> pending{{0, 0, ""}};
531 set<pair<int, int>> seen{{0, 0}};
532 bool equivalent = true;
533 for (size_t i = 0; i < pending.size(); ++i) {
534 auto [p, q, w] = pending[i];
535 if (a.accepting.count(p) != b.accepting.count(q)) {
536 out << "NOT EQUIVALENT: " << printedWord(w) << " is accepted only by " << (a.accepting.count(p) ? args[0] : args[1]) << '\n';
537 equivalent = false;
538 break;
539 }
540 for (char c : a.alphabet) {
541 auto destination = make_pair(a.delta(p, c), b.delta(q, c));
542 if (seen.insert(destination).second) pending.emplace_back(destination.first, destination.second, w + c);
543 }
544 }
545 if (equivalent) out << "EQUIVALENT\n";
546 } else if (command == "REVERSE" || command == "CONCAT" || command == "STAR") {
547 Automaton result;
548 result.alphabet = a.alphabet;
549 int offset = command == "CONCAT" ? 0 : 1;
550 result.series(a.names.size() + (command == "CONCAT" ? b.names.size() : 1), 'N');
551 for (size_t p = 0; p < a.names.size(); ++p) for (const auto& edge : a.transitions[p]) for (int q : edge.second) {
552 if (command == "REVERSE") result.edge(q + 1, edge.first, p + 1);
553 else result.edge(p + offset, edge.first, q + offset);
554 }
555 if (command == "REVERSE") {
556 for (int p : a.accepting) result.edge(0, 'e', p + 1);
557 result.accepting.insert(a.start + 1);
558 } else if (command == "STAR") {
559 result.edge(0, 'e', a.start + 1);
560 result.accepting.insert(0);
561 for (int p : a.accepting) {
562 result.edge(p + 1, 'e', a.start + 1);
563 result.accepting.insert(p + 1);
564 }
565 } else {
566 int m = a.names.size();
567 result.start = a.start;
568 for (size_t p = 0; p < b.names.size(); ++p) for (const auto& edge : b.transitions[p]) for (int q : edge.second) result.edge(p + m, edge.first, q + m);
569 for (int p : a.accepting) result.edge(p, 'e', b.start + m);
570 for (int q : b.accepting) result.accepting.insert(q + m);
571 }
572 add(args.back(), move(result), {}, out);
573 } else if (command == "ENUM" || command == "COUNT") {
574 int length = stoi(args[1]);
575 a = determinize(a).first;
576 if (command == "COUNT") {
577 vector<Count> counts(a.names.size());
578 counts[0] = 1;
579 for (int i = 0; i < length; ++i) {
580 vector<Count> next(a.names.size());
581 for (size_t p = 0; p < counts.size(); ++p) for (char c : a.alphabet) next[a.delta(p, c)] += counts[p];
582 counts = move(next);
583 }
584 Count total = 0;
585 for (int p : a.accepting) total += counts[p];
586 out << total << '\n';
587 } else {
588 vector<pair<int, string>> pending{{0, ""}};
589 vector<string> first;
590 Count total = 0;
591 for (size_t i = 0; i < pending.size(); ++i) {
592 auto [p, w] = pending[i];
593 if (a.accepting.count(p)) {
594 ++total;
595 if (first.size() < 50) first.push_back(printedWord(w));
596 }
597 if (static_cast<int>(w.size()) < length) for (char c : a.alphabet) pending.emplace_back(a.delta(p, c), w + c);
598 }
599 out << total;
600 for (const string& w : first) out << ' ' << w;
601 if (total > 50) out << " ...";
602 out << '\n';
603 }
604 } else if (command == "EMPTY" || command == "FINITE") {
605 auto language = inspectLanguage(a, command == "FINITE");
606 if (command == "EMPTY") out << (language.smallest ? "NOT EMPTY " + printedWord(*language.smallest) : "EMPTY") << '\n';
607 else if (language.infinite) out << "INFINITE\n";
608 else out << "FINITE " << language.count << '\n';
609 } else if (command == "SYNC") {
610 States all;
611 for (size_t p = 0; p < a.names.size(); ++p) all.insert(p);
612 vector<pair<States, string>> pending{{all, ""}};
613 set<States> seen{all};
614 bool foundWord = false;
615 for (size_t i = 0; i < pending.size(); ++i) {
616 auto [states, w] = pending[i];
617 if (states.size() == 1) {
618 out << "SYNCHRONIZING " << printedWord(w) << " -> " << a.names[*states.begin()] << '\n';
619 foundWord = true;
620 break;
621 }
622 for (char c : a.alphabet) {
623 States destination;
624 for (int p : states) destination.insert(a.delta(p, c));
625 if (seen.insert(destination).second) pending.emplace_back(destination, w + c);
626 }
627 }
628 if (!foundWord) out << "NOT SYNCHRONIZING\n";
629 } else if (command == "INFO") {
630 auto language = inspectLanguage(a);
631 bool deterministic = !a.transitionCount('e'), complete = true;
632 for (size_t p = 0; p < a.names.size(); ++p) for (char c : a.alphabet) {
633 if (a.destinations(p, c).size() > 1) deterministic = false;
634 if (a.destinations(p, c).empty()) complete = false;
635 }
636 auto dfa = determinize(a).first;
637 auto minimal = minimize(dfa).first;
638 out << a.summary(args[0]) << "\n epsilon transitions: " << a.transitionCount('e')
639 << "\n reachable: " << a.list(graphReach(a, false)) << "\n live: " << a.list(graphReach(a, true))
640 << "\n deterministic: " << (deterministic ? "yes" : "no") << "\n complete: " << (complete ? "yes" : "no")
641 << "\n language: " << (!language.smallest ? "empty" : language.infinite ? "infinite" : "finite")
642 << "\n minimal DFA states: " << minimal.names.size() << '\n';
643 } else if (command == "SHOW") {
644 out << (a.dfa ? "DFA " : "NFA ") << args[0] << ' ' << join(a.names, ",") << ' ' << a.alphabet << ' ' << a.names[a.start] << ' '
645 << (a.accepting.empty() ? "-" : a.list(a.accepting));
646 string symbols = a.alphabet + 'e';
647 for (size_t p = 0; p < a.names.size(); ++p) for (char c : symbols) for (int q : a.destinations(p, c)) {
648 out << ' ' << a.names[p] << ':' << c << ':' << a.names[q];
649 }
650 out << '\n';
651 } else if (command == "TABLE") {
652 string symbols = a.alphabet;
653 if (!a.dfa && a.transitionCount('e')) symbols += 'e';
654 vector<vector<string>> rows(1, vector<string>{""});
655 for (char c : symbols) rows[0].push_back(string(1, c));
656 for (size_t p = 0; p < a.names.size(); ++p) {
657 vector<string> row{string(static_cast<int>(p) == a.start ? "->" : " ") + (a.accepting.count(p) ? "*" : " ") + a.names[p]};
658 for (char c : symbols) {
659 const States& destinations = a.destinations(p, c);
660 row.push_back(a.dfa ? a.names[*destinations.begin()] : destinations.empty() ? "-" : a.setText(destinations));
661 }
662 rows.push_back(move(row));
663 }
664 vector<size_t> widths(symbols.size() + 1);
665 for (const auto& row : rows) for (size_t i = 0; i < row.size(); ++i) widths[i] = max(widths[i], row[i].size());
666 for (const auto& row : rows) {
667 string text;
668 for (size_t i = 0; i < row.size(); ++i) {
669 if (i) text += " ";
670 text += row[i] + string(widths[i] - row[i].size(), ' ');
671 }
672 while (!text.empty() && text.back() == ' ') text.pop_back();
673 out << text << '\n';
674 }
675 } else if (command == "DELETE") {
676 table.erase(table.begin() + indexOf(args[0]));
677 out << "deleted " << args[0] << '\n';
678 }
679 }
