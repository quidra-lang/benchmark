32 static class Automaton {
33 String kind, alphabet;
34 List<String> states = new ArrayList<>();
35 List<BitSet[]> transitions = new ArrayList<>();
36 BitSet accepting = new BitSet();
37 int start;
38 Automaton(String kind, String alphabet) {
39 this.kind = kind;
40 this.alphabet = alphabet;
41 }
42 int addState(String name) {
43 int id = states.size();
44 states.add(name);
45 BitSet[] row = new BitSet[alphabet.length() + 1];
46 for (int i = 0; i < row.length; i++) row[i] = new BitSet();
47 transitions.add(row);
48 return id;
49 }
50 void names(String prefix, int count) {
51 for (int i = 0; i < count; i++) addState(prefix + i);
52 }
53 void edge(int from, int symbol, int to) {
54 transitions.get(from)[symbol].set(to);
55 }
56 int delta(int from, int symbol) {
57 return transitions.get(from)[symbol].nextSetBit(0);
58 }
59 int transitionCount() {
60 int count = 0;
61 for (BitSet[] row : transitions) for (BitSet set : row) count += set.cardinality();
62 return count;
63 }
64 int epsilonCount() {
65 int count = 0;
66 for (BitSet[] row : transitions) count += row[alphabet.length()].cardinality();
67 return count;
68 }
69 String summary(String name) {
70 return kind + " " + name + ": alphabet=" + alphabet + " states=" + states.size() + " accepting=" + accepting.cardinality() + " transitions=" + transitionCount();
71 }
72 String set(BitSet states) {
73 List<String> names = new ArrayList<>();
74 for (int i = states.nextSetBit(0); i >= 0; i = states.nextSetBit(i + 1)) names.add(this.states.get(i));
75 return "{" + String.join(",", names) + "}";
76 }
77 String list(BitSet states) {
78 String text = set(states);
79 return states.isEmpty() ? "none" : text.substring(1, text.length() - 1);
80 }
81 BitSet close(BitSet input) {
82 BitSet out = (BitSet)input.clone();
83 ArrayDeque<Integer> queue = new ArrayDeque<>();
84 for (int p = out.nextSetBit(0); p >= 0; p = out.nextSetBit(p + 1)) queue.add(p);
85 while (!queue.isEmpty()) {
86 BitSet edges = transitions.get(queue.remove())[alphabet.length()];
87 for (int q = edges.nextSetBit(0); q >= 0; q = edges.nextSetBit(q + 1)) if (!out.get(q)) {
88 out.set(q);
89 queue.add(q);
90 }
91 }
92 return out;
93 }
94 BitSet initial() {
95 return close(singleton(start));
96 }
97 BitSet move(BitSet from, int symbol) {
98 BitSet out = new BitSet();
99 for (int p = from.nextSetBit(0); p >= 0; p = from.nextSetBit(p + 1)) out.or(transitions.get(p)[symbol]);
100 return close(out);
101 }
102 boolean accepts(BitSet set) {
103 return set.intersects(accepting);
104 }
105 BitSet graphReach(boolean reverse) {
106 BitSet reached = reverse ? (BitSet)accepting.clone() : singleton(start);
107 boolean changed;
108 do {
109 int old = reached.cardinality();
110 for (int p = 0; p < states.size(); p++) for (BitSet edges : transitions.get(p)) {
111 if (reverse) {
112 if (edges.intersects(reached)) reached.set(p);
113 } else if (reached.get(p)) reached.or(edges);
114 }
115 changed = old != reached.cardinality();
116 } while (changed);
117 return reached;
118 }
119 }

131 static Created determinize(Automaton a) {
132 Automaton d = new Automaton("DFA", a.alphabet);
133 List<BitSet> subsets = new ArrayList<>();
134 Map<BitSet, Integer> ids = new HashMap<>();
135 List<String> details = new ArrayList<>();
136 BitSet first = a.initial();
137 subsets.add(first);
138 ids.put(first, 0);
139 d.addState("D0");
140 for (int i = 0; i < subsets.size(); i++) {
141 BitSet set = subsets.get(i);
142 if (a.accepts(set)) d.accepting.set(i);
143 details.add(" D" + i + " = " + a.set(set));
144 for (int c = 0; c < a.alphabet.length(); c++) {
145 BitSet next = a.move(set, c);
146 Integer id = ids.get(next);
147 if (id == null) {
148 id = subsets.size();
149 ids.put(next, id);
150 subsets.add(next);
151 d.addState("D" + id);
152 }
153 d.edge(i, c, id);
154 }
155 }
156 return new Created(d, details);
157 }

176 static Created minimize(Automaton a) {
177 BitSet reachable = a.graphReach(false), missing = new BitSet();
178 missing.set(0, a.states.size());
179 missing.andNot(reachable);
180 List<String> details = new ArrayList<>();
181 details.add(" unreachable: " + a.list(missing));
182 List<List<Integer>> blocks = new ArrayList<>();
183 List<Integer> yes = new ArrayList<>(), no = new ArrayList<>();
184 for (int p = 0; p < a.states.size(); p++) if (reachable.get(p)) (a.accepting.get(p) ? yes : no).add(p);
185 if (!yes.isEmpty()) blocks.add(yes);
186 if (!no.isEmpty()) blocks.add(no);
187 blocks.sort((x, y) -> Integer.compare(x.getFirst(), y.getFirst()));
188 details.add(" round 0: " + partition(a, blocks));
189 int round = 0;
190 while (true) {
191 int[] index = blockIndices(blocks, a.states.size());
192 List<List<Integer>> next = new ArrayList<>();
193 for (List<Integer> block : blocks) {
194 Map<List<Integer>, List<Integer>> groups = new LinkedHashMap<>();
195 for (int p : block) {
196 List<Integer> signature = new ArrayList<>();
197 for (int c = 0; c < a.alphabet.length(); c++) signature.add(index[a.delta(p, c)]);
198 groups.computeIfAbsent(signature, s -> new ArrayList<>()).add(p);
199 }
200 next.addAll(groups.values());
201 }
202 next.sort((x, y) -> Integer.compare(x.getFirst(), y.getFirst()));
203 if (next.size() == blocks.size()) break;
204 blocks = next;
205 details.add(" round " + (++round) + ": " + partition(a, blocks));
206 }
207 int[] index = blockIndices(blocks, a.states.size());
208 List<Integer> named = new ArrayList<>();
209 Map<Integer, Integer> ids = new HashMap<>();
210 named.add(index[a.start]);
211 ids.put(index[a.start], 0);
212 Automaton m = new Automaton("DFA", a.alphabet);
213 m.addState("M0");
214 for (int i = 0; i < named.size(); i++) {
215 List<Integer> block = blocks.get(named.get(i));
216 int p = block.getFirst();
217 if (a.accepting.get(p)) m.accepting.set(i);
218 BitSet set = new BitSet();
219 for (int state : block) set.set(state);
220 details.add(" M" + i + " = " + a.set(set));
221 for (int c = 0; c < a.alphabet.length(); c++) {
222 int target = index[a.delta(p, c)];
223 Integer id = ids.get(target);
224 if (id == null) {
225 id = named.size();
226 named.add(target);
227 ids.put(target, id);
228 m.addState("M" + id);
229 }
230 m.edge(i, c, id);
231 }
232 }
233 return new Created(m, details);
234 }

236 static void copyEdges(Automaton source, Automaton target, int offset, boolean reverse) {
237 for (int p = 0; p < source.states.size(); p++) for (int c = 0; c <= source.alphabet.length(); c++) {
238 BitSet edges = source.transitions.get(p)[c];
239 for (int q = edges.nextSetBit(0); q >= 0; q = edges.nextSetBit(q + 1)) target.edge((reverse ? q : p) + offset, c, (reverse ? p : q) + offset);
240 }
241 }

243 static Automaton epsilonFree(Automaton a) {
244 Automaton n = new Automaton("NFA", a.alphabet);
245 for (String state : a.states) n.addState(state);
246 n.start = a.start;
247 for (int p = 0; p < a.states.size(); p++) {
248 BitSet closure = a.close(singleton(p));
249 if (a.accepts(closure)) n.accepting.set(p);
250 for (int c = 0; c < a.alphabet.length(); c++) n.transitions.get(p)[c].or(a.move(closure, c));
251 }
252 return n;
253 }

255 static Automaton unaryConstruction(Automaton a, String command) {
256 if (command.equals("COMPLEMENT")) {
257 Automaton n = new Automaton("DFA", a.alphabet);
258 for (String state : a.states) n.addState(state);
259 n.start = a.start;
260 copyEdges(a, n, 0, false);
261 n.accepting.set(0, a.states.size());
262 n.accepting.xor(a.accepting);
263 return n;
264 }
265 Automaton n = new Automaton("NFA", a.alphabet);
266 n.names("N", a.states.size() + 1);
267 copyEdges(a, n, 1, command.equals("REVERSE"));
268 if (command.equals("REVERSE")) {
269 n.accepting.set(a.start + 1);
270 for (int p = a.accepting.nextSetBit(0); p >= 0; p = a.accepting.nextSetBit(p + 1)) n.edge(0, a.alphabet.length(), p + 1);
271 } else {
272 n.accepting.set(0);
273 n.edge(0, a.alphabet.length(), a.start + 1);
274 for (int p = a.accepting.nextSetBit(0); p >= 0; p = a.accepting.nextSetBit(p + 1)) {
275 n.accepting.set(p + 1);
276 n.edge(p + 1, a.alphabet.length(), a.start + 1);
277 }
278 }
279 return n;
280 }

282 static Automaton concatenate(Automaton a, Automaton b) {
283 Automaton n = new Automaton("NFA", a.alphabet);
284 int offset = a.states.size();
285 n.names("N", offset + b.states.size());
286 n.start = a.start;
287 copyEdges(a, n, 0, false);
288 copyEdges(b, n, offset, false);
289 for (int p = a.accepting.nextSetBit(0); p >= 0; p = a.accepting.nextSetBit(p + 1)) n.edge(p, a.alphabet.length(), offset + b.start);
290 for (int p = b.accepting.nextSetBit(0); p >= 0; p = b.accepting.nextSetBit(p + 1)) n.accepting.set(offset + p);
291 return n;
292 }

296 static Created product(Automaton a, Automaton b, String command) {
297 Automaton p = new Automaton("DFA", a.alphabet);
298 List<Pair> pairs = new ArrayList<>();
299 Map<Pair, Integer> ids = new HashMap<>();
300 List<String> details = new ArrayList<>();
301 Pair first = new Pair(a.start, b.start);
302 pairs.add(first);
303 ids.put(first, 0);
304 p.addState("P0");
305 for (int i = 0; i < pairs.size(); i++) {
306 Pair pair = pairs.get(i);
307 boolean x = a.accepting.get(pair.a), y = b.accepting.get(pair.b);
308 if (command.equals("UNION") ? x || y : command.equals("INTERSECT") ? x && y : x && !y) p.accepting.set(i);
309 details.add(" P" + i + " = (" + a.states.get(pair.a) + "," + b.states.get(pair.b) + ")");
310 for (int c = 0; c < a.alphabet.length(); c++) {
311 Pair next = new Pair(a.delta(pair.a, c), b.delta(pair.b, c));
312 Integer id = ids.get(next);
313 if (id == null) {
314 id = pairs.size();
315 pairs.add(next);
316 ids.put(next, id);
317 p.addState("P" + id);
318 }
319 p.edge(i, c, id);
320 }
321 }
322 return new Created(p, details);
323 }

338 static String equivalent(Automaton a, Automaton b, String x, String z) {
339 a = determinize(a).automaton;
340 b = determinize(b).automaton;
341 List<Pair> pairs = new ArrayList<>();
342 Map<Pair, Integer> ids = new HashMap<>();
343 List<Integer> parents = new ArrayList<>();
344 List<Character> incoming = new ArrayList<>();
345 Pair first = new Pair(a.start, b.start);
346 pairs.add(first);
347 ids.put(first, 0);
348 parents.add(-1);
349 incoming.add(' ');
350 for (int i = 0; i < pairs.size(); i++) {
351 Pair pair = pairs.get(i);
352 boolean accepted = a.accepting.get(pair.a);
353 if (accepted != b.accepting.get(pair.b)) return "NOT EQUIVALENT: " + visible(word(parents, incoming, i)) + " is accepted only by " + (accepted ? x : z);
354 for (int c = 0; c < a.alphabet.length(); c++) {
355 Pair next = new Pair(a.delta(pair.a, c), b.delta(pair.b, c));
356 if (!ids.containsKey(next)) {
357 ids.put(next, pairs.size());
358 pairs.add(next);
359 parents.add(i);
360 incoming.add(a.alphabet.charAt(c));
361 }
362 }
363 }
364 return "EQUIVALENT";
365 }

393 static Language language(Automaton d, boolean countWords) {
394 BitSet reachable = d.graphReach(false), live = d.graphReach(true);
395 live.and(reachable);
396 if (!live.get(d.start)) return new Language("empty", 0);
397 int[] indegree = new int[d.states.size()];
398 for (int p = live.nextSetBit(0); p >= 0; p = live.nextSetBit(p + 1)) for (int c = 0; c < d.alphabet.length(); c++) {
399 int q = d.delta(p, c);
400 if (live.get(q)) indegree[q]++;
401 }
402 List<Integer> order = new ArrayList<>();
403 for (int p = live.nextSetBit(0); p >= 0; p = live.nextSetBit(p + 1)) if (indegree[p] == 0) order.add(p);
404 for (int i = 0; i < order.size(); i++) for (int c = 0; c < d.alphabet.length(); c++) {
405 int q = d.delta(order.get(i), c);
406 if (live.get(q) && --indegree[q] == 0) order.add(q);
407 }
408 if (order.size() != live.cardinality()) return new Language("infinite", 0);
409 if (!countWords) return new Language("finite", 0);
410 long[] counts = new long[d.states.size()];
411 for (int i = order.size() - 1; i >= 0; i--) {
412 int p = order.get(i);
413 counts[p] = d.accepting.get(p) ? 1 : 0;
414 for (int c = 0; c < d.alphabet.length(); c++) counts[p] += counts[d.delta(p, c)];
415 }
416 return new Language("finite", counts[d.start]);
417 }

419 static long count(Automaton d, int length) {
420 long[] values = new long[d.states.size()];
421 values[d.start] = 1;
422 for (int i = 0; i < length; i++) {
423 long[] next = new long[values.length];
424 for (int p = 0; p < values.length; p++) for (int c = 0; c < d.alphabet.length(); c++) next[d.delta(p, c)] += values[p];
425 values = next;
426 }
427 long total = 0;
428 for (int p = d.accepting.nextSetBit(0); p >= 0; p = d.accepting.nextSetBit(p + 1)) total += values[p];
429 return total;

452 static String synchronize(Automaton d) {
453 int start = (1 << d.states.size()) - 1;
454 List<Integer> sets = new ArrayList<>(), parents = new ArrayList<>();
455 List<Character> incoming = new ArrayList<>();
456 Map<Integer, Integer> ids = new HashMap<>();
457 sets.add(start);
458 parents.add(-1);
459 incoming.add(' ');
460 ids.put(start, 0);
461 for (int i = 0; i < sets.size(); i++) {
462 int set = sets.get(i);
463 if (Integer.bitCount(set) == 1) return "SYNCHRONIZING " + visible(word(parents, incoming, i)) + " -> " + d.states.get(Integer.numberOfTrailingZeros(set));
464 for (int c = 0; c < d.alphabet.length(); c++) {
465 int next = 0;
466 for (int p = 0; p < d.states.size(); p++) if ((set & 1 << p) != 0) next |= 1 << d.delta(p, c);
467 if (!ids.containsKey(next)) {
468 ids.put(next, sets.size());
469 sets.add(next);
470 parents.add(i);
471 incoming.add(d.alphabet.charAt(c));
472 }
473 }
474 }
475 return "NOT SYNCHRONIZING";
476 }

480 static class RegexParser {
481 String text;
482 int at;
483 RegexParser(String text) {
484 this.text = text;
485 }
486 char peek() {
487 return at == text.length() ? '\0' : text.charAt(at);
488 }
489 void error() {
490 fail("syntax error at " + (at + 1));
491 }
492 static boolean symbol(char c) {
493 return c >= 'a' && c <= 'z' || c >= '0' && c <= '9';
494 }
495 Regex alternation() {
496 List<Regex> list = new ArrayList<>();
497 list.add(concatenation());
498 while (peek() == '|') {
499 at++;
500 list.add(concatenation());
501 }
502 return list.size() == 1 ? list.getFirst() : new Regex('|', list);
503 }
504 Regex concatenation() {
505 List<Regex> list = new ArrayList<>();
506 list.add(factor());
507 while (symbol(peek()) || peek() == '(') list.add(factor());
508 return list.size() == 1 ? list.getFirst() : new Regex('.', list);
509 }
510 Regex factor() {
511 Regex r = atom();
512 while (peek() == '*' || peek() == '+' || peek() == '?') r = new Regex(text.charAt(at++), List.of(r));
513 return r;
514 }
515 Regex atom() {
516 char c = peek();
517 if (symbol(c)) {
518 at++;
519 return new Regex(c, List.of());
520 }
521 if (c == '(') {
522 at++;
523 Regex r = alternation();
524 if (peek() != ')') error();
525 at++;
526 return r;
527 }
528 error();
529 return null;
530 }
531 Regex parse() {
532 Regex r = alternation();
533 if (at != text.length()) error();
534 return r;
535 }
536 }

538 record Fragment(int entry, int exit) { }
539
540 static Fragment build(Regex regex, Automaton a) {
541 if (regex.kind == '.') {
542 Fragment first = null, last = null;
543 for (Regex child : regex.children) {
544 Fragment next = build(child, a);
545 if (last != null) a.edge(last.exit, a.alphabet.length(), next.entry);
546 else first = next;
547 last = next;
548 }
549 return new Fragment(first.entry, last.exit);
550 }
551 int entry = a.addState("N" + a.states.size());
552 if (regex.children.isEmpty()) {
553 int exit = a.addState("N" + a.states.size());
554 a.edge(entry, regex.kind == 'e' ? a.alphabet.length() : a.alphabet.indexOf(regex.kind), exit);
555 return new Fragment(entry, exit);
556 }
557 List<Fragment> parts = new ArrayList<>();
558 for (Regex child : regex.children) parts.add(build(child, a));
559 int exit = a.addState("N" + a.states.size());
560 for (Fragment f : parts) {
561 a.edge(entry, a.alphabet.length(), f.entry);
562 a.edge(f.exit, a.alphabet.length(), exit);
563 }
564 if (regex.kind != '|') {
565 Fragment f = parts.getFirst();
566 if (regex.kind != '?') a.edge(f.exit, a.alphabet.length(), f.entry);
567 if (regex.kind != '+') a.edge(entry, a.alphabet.length(), exit);
568 }
569 return new Fragment(entry, exit);
570 }

572 Automaton explicit(String[] w) {
573 if (w.length < 6) fail("wrong number of arguments");
574 if (automata.containsKey(w[1])) fail(w[1] + " already exists");
575 Automaton a = new Automaton(w[0], alphabet(w[3]));
576 for (String state : w[2].split(",")) {
577 if (a.states.contains(state)) fail("duplicate state " + state);
578 a.addState(state);
579 }
580 a.start = a.states.indexOf(w[4]);
581 if (a.start < 0) fail("unknown state " + w[4]);
582 if (!w[5].equals("-")) for (String state : w[5].split(",")) {
583 int p = a.states.indexOf(state);
584 if (p < 0) fail("unknown state " + state);
585 if (a.accepting.get(p)) fail("duplicate state " + state);
586 a.accepting.set(p);
587 }
588 for (int i = 6; i < w.length; i++) {
589 String[] t = w[i].split(":");
590 int p = a.states.indexOf(t[0]);
591 char c = t[1].charAt(0);
592 if (p < 0) fail("unknown state " + t[0]);
593 if (c == 'e' && a.kind.equals("DFA")) fail("epsilon transition in DFA");
594 int symbol = c == 'e' ? a.alphabet.length() : a.alphabet.indexOf(c);
595 if (symbol < 0) fail("unknown symbol " + c);
596 int q = a.states.indexOf(t[2]);
597 if (q < 0) fail("unknown state " + t[2]);
598 BitSet edges = a.transitions.get(p)[symbol];
599 if (edges.get(q)) fail("duplicate transition " + w[i]);
600 if (a.kind.equals("DFA") && !edges.isEmpty()) fail("conflicting transition " + w[i]);
601 a.edge(p, symbol, q);
602 }
603 if (a.kind.equals("DFA")) for (int p = 0; p < a.states.size(); p++) for (int c = 0; c < a.alphabet.length(); c++) if (a.transitions.get(p)[c].isEmpty()) fail("missing transition " + a.states.get(p) + ":" + a.alphabet.charAt(c));
604 return a;
605 }

607 static List<String> table(Automaton a) {
608 int columns = a.alphabet.length() + 1 + (a.kind.equals("NFA") && a.epsilonCount() > 0 ? 1 : 0);
609 String[][] cells = new String[a.states.size() + 1][columns];
610 cells[0][0] = "";
611 for (int c = 1; c < columns; c++) cells[0][c] = c <= a.alphabet.length() ? Character.toString(a.alphabet.charAt(c - 1)) : "e";
612 for (int p = 0; p < a.states.size(); p++) {
613 cells[p + 1][0] = (p == a.start ? "->" : " ") + (a.accepting.get(p) ? "*" : " ") + a.states.get(p);
614 for (int c = 1; c < columns; c++) {
615 BitSet edges = a.transitions.get(p)[c - 1];
616 cells[p + 1][c] = a.kind.equals("DFA") ? a.states.get(edges.nextSetBit(0)) : edges.isEmpty() ? "-" : a.set(edges);
617 }
618 }
619 int[] widths = new int[columns];
620 for (String[] row : cells) for (int c = 0; c < columns; c++) widths[c] = Math.max(widths[c], row[c].length());
621 List<String> out = new ArrayList<>();
622 for (String[] row : cells) {
623 StringBuilder text = new StringBuilder();
624 for (int c = 0; c < columns; c++) {
625 if (c > 0) text.append(" ");
626 text.append(row[c]).append(" ".repeat(widths[c] - row[c].length()));
627 }
628 while (!text.isEmpty() && text.charAt(text.length() - 1) == ' ') text.setLength(text.length() - 1);
629 out.add(text.toString());
630 }
631 return out;
632 }

634 static String show(Automaton a, String name) {
635 StringBuilder text = new StringBuilder(a.kind + " " + name + " " + String.join(",", a.states) + " " + a.alphabet + " " + a.states.get(a.start) + " " + (a.accepting.isEmpty() ? "-" : a.list(a.accepting)));
636 for (int p = 0; p < a.states.size(); p++) for (int c = 0; c <= a.alphabet.length(); c++) {
637 BitSet edges = a.transitions.get(p)[c];
638 for (int q = edges.nextSetBit(0); q >= 0; q = edges.nextSetBit(q + 1)) text.append(' ').append(a.states.get(p)).append(':').append(c == a.alphabet.length() ? 'e' : a.alphabet.charAt(c)).append(':').append(a.states.get(q));
639 }
640 return text.toString();
641 }

643 List<String> register(String name, Created created) {
644 automata.put(name, created.automaton);
645 List<String> out = new ArrayList<>();
646 out.add(created.automaton.summary(name));
647 out.addAll(created.details);
648 return out;
649 }
650 static final Map<String, String> ROLES = Map.ofEntries(
651 Map.entry("ACCEPT", "RO"),
652 Map.entry("CLOSURE", "RO"),
653 Map.entry("ENUM", "RO"),
654 Map.entry("COUNT", "RO"),
655 Map.entry("TODFA", "RN"),
656 Map.entry("EPSFREE", "RN"),
657 Map.entry("REVERSE", "RN"),
658 Map.entry("STAR", "RN"),
659 Map.entry("MINIMIZE", "DN"),
660 Map.entry("COMPLEMENT", "DN"),
661 Map.entry("EQUIV", "RR"),
662 Map.entry("UNION", "DDN"),
663 Map.entry("INTERSECT", "DDN"),
664 Map.entry("DIFF", "DDN"),
665 Map.entry("CONCAT", "RRN"),
666 Map.entry("EMPTY", "R"),
667 Map.entry("FINITE", "R"),
668 Map.entry("INFO", "R"),
669 Map.entry("TABLE", "R"),
670 Map.entry("SHOW", "R"),
671 Map.entry("DELETE", "R"),
672 Map.entry("SYNC", "D"),
673 Map.entry("LIST", "")
674 );
675
676 List<String> command(String line) {
677 String[] w = line.split(" ");
678 String c = w[0];
679 if (!ROLES.containsKey(c) && !c.equals("DFA") && !c.equals("NFA") && !c.equals("REGEX")) fail("unknown command " + c);
680 if (c.equals("DFA") || c.equals("NFA")) {
681 Automaton a = explicit(w);
682 return register(w[1], new Created(a, List.of()));
683 }
684 if (c.equals("REGEX")) {
685 if (w.length != 4) fail("wrong number of arguments");
686 if (automata.containsKey(w[1])) fail(w[1] + " already exists");
687 for (char ch : w[3].toCharArray()) if (RegexParser.symbol(ch) && ch != 'e' && w[2].indexOf(ch) < 0) fail("unknown symbol " + ch);
688 Regex regex = new RegexParser(w[3]).parse();
689 Automaton a = new Automaton("NFA", alphabet(w[2]));
690 Fragment f = build(regex, a);
691 a.start = f.entry;
692 a.accepting.set(f.exit);
693 return register(w[1], new Created(a, List.of()));
694 }
695 String roles = ROLES.get(c);
696 if (w.length != roles.length() + 1) fail("wrong number of arguments");
697 for (int i = 0; i < roles.length(); i++) if (roles.charAt(i) == 'R' || roles.charAt(i) == 'D') get(w[i + 1]);
698 for (int i = 0; i < roles.length(); i++) if (roles.charAt(i) == 'N' && automata.containsKey(w[i + 1])) fail(w[i + 1] + " already exists");
699 for (int i = 0; i < roles.length(); i++) if (roles.charAt(i) == 'D' && get(w[i + 1]).kind.equals("NFA")) fail(w[i + 1] + " is not a DFA");
700 if (List.of("EQUIV", "UNION", "INTERSECT", "DIFF", "CONCAT").contains(c) && !get(w[1]).alphabet.equals(get(w[2]).alphabet)) fail("alphabets differ");
701 if (c.equals("LIST")) {
702 List<String> out = new ArrayList<>();
703 for (var e : automata.entrySet()) out.add(e.getValue().summary(e.getKey()));
704 return out.isEmpty() ? List.of("(none)") : out;
705 }
706 Automaton a = get(w[1]);
707 switch (c) {
708 case "DELETE" -> {
709 automata.remove(w[1]);
710 return List.of("deleted " + w[1]);
711 }
712 case "ACCEPT" -> {
713 String word = w[2].equals("e") ? "" : w[2];
714 for (char ch : word.toCharArray()) if (a.alphabet.indexOf(ch) < 0) fail("unknown symbol " + ch);
715 BitSet set = a.initial();
716 StringBuilder path = new StringBuilder(a.kind.equals("DFA") ? a.states.get(set.nextSetBit(0)) : a.set(set));
717 for (char ch : word.toCharArray()) {
718 set = a.move(set, a.alphabet.indexOf(ch));
719 path.append(" -").append(ch).append("-> ").append(a.kind.equals("DFA") ? a.states.get(set.nextSetBit(0)) : a.set(set));
720 }
721 return List.of((a.accepts(set) ? "ACCEPT " : "REJECT ") + path);
722 }
723 case "CLOSURE" -> {
724 BitSet set = new BitSet();
725 for (String state : w[2].split(",")) {
726 int p = a.states.indexOf(state);
727 if (p < 0) fail("unknown state " + state);
728 set.set(p);
729 }
730 return List.of(a.set(a.close(set)));
731 }
732 case "TODFA" -> {
733 return register(w[2], determinize(a));
734 }
735 case "EPSFREE" -> {
736 return register(w[2], new Created(epsilonFree(a), List.of()));
737 }
738 case "MINIMIZE" -> {
739 return register(w[2], minimize(a));
740 }
741 case "COMPLEMENT", "REVERSE", "STAR" -> {
742 return register(w[2], new Created(unaryConstruction(a, c), List.of()));
743 }
744 case "CONCAT" -> {
745 return register(w[3], new Created(concatenate(a, get(w[2])), List.of()));
746 }
747 case "UNION", "INTERSECT", "DIFF" -> {
748 return register(w[3], product(a, get(w[2]), c));
749 }
750 case "EQUIV" -> {
751 return List.of(equivalent(a, get(w[2]), w[1], w[2]));
752 }
753 case "TABLE" -> {
754 return table(a);
755 }
756 case "SHOW" -> {
757 return List.of(show(a, w[1]));
758 }
759 case "SYNC" -> {
760 return List.of(synchronize(a));
761 }
762 }
763 Automaton d = determinize(a).automaton;
764 switch (c) {
765 case "COUNT" -> {
766 return List.of(Long.toString(count(d, Integer.parseInt(w[2]))));
767 }
768 case "ENUM" -> {
769 return List.of(enumerate(d, Integer.parseInt(w[2])));
770 }
771 case "EMPTY" -> {
772 String smallest = smallest(d);
773 return List.of(smallest == null ? "EMPTY" : "NOT EMPTY " + visible(smallest));
774 }
775 case "FINITE" -> {
776 Language l = language(d, true);
777 return List.of(l.kind.equals("infinite") ? "INFINITE" : "FINITE " + l.count);
778 }
779 case "INFO" -> {
780 boolean deterministic = a.epsilonCount() == 0, complete = true;
781 for (BitSet[] row : a.transitions) for (int symbol = 0; symbol < a.alphabet.length(); symbol++) {
782 deterministic &= row[symbol].cardinality() <= 1;
783 complete &= !row[symbol].isEmpty();
784 }
785 return List.of(a.summary(w[1]), " epsilon transitions: " + a.epsilonCount(), " reachable: " + a.list(a.graphReach(false)), " live: " + a.list(a.graphReach(true)), " deterministic: " + (deterministic ? "yes" : "no"), " complete: " + (complete ? "yes" : "no"), " language: " + language(d, false).kind, " minimal DFA states: " + minimize(d).automaton.states.size());
786 }
787 default -> throw new IllegalStateException();
788 }
789 }

792 BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
793 int count = Integer.parseInt(reader.readLine());
794 Main toolkit = new Main();
795 for (int i = 0; i < count; i++) {
796 String line = reader.readLine();
797 if (line.startsWith("#")) continue;
798 try {
799 for (String output : toolkit.command(line)) System.out.println(output);
800 } catch (Problem error) {
801 System.out.println(error.getMessage());
802 }
803 }
