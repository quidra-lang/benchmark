20 class Automaton(string kind, string alphabet)
21 {
22 public string Kind = kind, Alphabet = alphabet;
23 public List<string> States = new();
24 public List<HashSet<int>[]> Transitions = new();
25 public HashSet<int> Accepting = new();
26 public int Start;
27 public int AddState(string name)
28 {
29 int id = States.Count;
30 States.Add(name);
31 Transitions.Add(Enumerable.Range(0, Alphabet.Length + 1).Select(_ => new HashSet<int>()).ToArray());
32 return id;
33 }
34 public void Names(string prefix, int count)
35 {
36 for (int i = 0; i < count; i++) AddState(prefix + i);
37 }
38 public void Edge(int from, int symbol, int to) => Transitions[from][symbol].Add(to);
39 public int Delta(int from, int symbol) => Transitions[from][symbol].First();
40 public int TransitionCount => Transitions.Sum(row => row.Sum(set => set.Count));
41 public int EpsilonCount => Transitions.Sum(row => row[Alphabet.Length].Count);
42 public string Summary(string name) => $"{Kind} {name}: alphabet={Alphabet} states={States.Count} accepting={Accepting.Count} transitions={TransitionCount}";
43 public string Set(HashSet<int> set) => "{" + string.Join(",", set.Order().Select(p => States[p])) + "}";
44 public string List(HashSet<int> set) => set.Count == 0 ? "none" : string.Join(",", set.Order().Select(p => States[p]));
45 public HashSet<int> Close(HashSet<int> input)
46 {
47 HashSet<int> output = new(input);
48 Queue<int> queue = new(input);
49 while (queue.Count > 0) foreach (int q in Transitions[queue.Dequeue()][Alphabet.Length]) if (output.Add(q)) queue.Enqueue(q);
50 return output;
51 }
52 public HashSet<int> Initial() => Close(Singleton(Start));
53 public HashSet<int> Move(HashSet<int> from, int symbol)
54 {
55 HashSet<int> output = new();
56 foreach (int p in from) output.UnionWith(Transitions[p][symbol]);
57 return Close(output);
58 }
59 public bool Accepts(HashSet<int> set) => set.Overlaps(Accepting);
60 public HashSet<int> GraphReach(bool reverse)
61 {
62 HashSet<int> reached = reverse ? new(Accepting) : Singleton(Start);
63 bool changed;
64 do
65 {
66 int old = reached.Count;
67 for (int p = 0; p < States.Count; p++) foreach (var edges in Transitions[p])
68 {
69 if (reverse)
70 {
71 if (edges.Overlaps(reached)) reached.Add(p);
72 }
73 else if (reached.Contains(p)) reached.UnionWith(edges);
74 }
75 changed = old != reached.Count;
76 } while (changed);
77 return reached;
78 }
79 }

87 static Created Determinize(Automaton a)
88 {
89 Automaton d = new("DFA", a.Alphabet);
90 List<HashSet<int>> subsets = new();
91 Dictionary<string, int> ids = new(StringComparer.Ordinal);
92 List<string> details = new();
93 HashSet<int> first = a.Initial();
94 subsets.Add(first);
95 ids[SetKey(first)] = 0;
96 d.AddState("D0");
97 for (int i = 0; i < subsets.Count; i++)
98 {
99 var set = subsets[i];
100 if (a.Accepts(set)) d.Accepting.Add(i);
101 details.Add($" D{i} = {a.Set(set)}");
102 for (int c = 0; c < a.Alphabet.Length; c++)
103 {
104 var next = a.Move(set, c);
105 string key = SetKey(next);
106 if (!ids.TryGetValue(key, out int id))
107 {
108 id = subsets.Count;
109 ids[key] = id;
110 subsets.Add(next);
111 d.AddState("D" + id);
112 }
113 d.Edge(i, c, id);
114 }
115 }
116 return new(d, details);
117 }

128 static Created Minimize(Automaton a)
129 {
130 var reachable = a.GraphReach(false);
131 HashSet<int> missing = new(Enumerable.Range(0, a.States.Count));
132 missing.ExceptWith(reachable);
133 List<string> details = new() { " unreachable: " + a.List(missing) };
134 List<List<int>> blocks = new();
135 List<int> yes = new(), no = new();
136 for (int p = 0; p < a.States.Count; p++) if (reachable.Contains(p)) (a.Accepting.Contains(p) ? yes : no).Add(p);
137 if (yes.Count > 0) blocks.Add(yes);
138 if (no.Count > 0) blocks.Add(no);
139 blocks.Sort((x, y) => x[0].CompareTo(y[0]));
140 details.Add(" round 0: " + Partition(a, blocks));
141 int round = 0;
142 while (true)
143 {
144 int[] index = BlockIndices(blocks, a.States.Count);
145 List<List<int>> next = new();
146 foreach (List<int> block in blocks)
147 {
148 Dictionary<string, List<int>> groups = new(StringComparer.Ordinal);
149 foreach (int p in block)
150 {
151 string signature = string.Join(",", Enumerable.Range(0, a.Alphabet.Length).Select(c => index[a.Delta(p, c)]));
152 if (!groups.TryGetValue(signature, out var group))
153 {
154 group = new();
155 groups[signature] = group;
156 next.Add(group);
157 }
158 group.Add(p);
159 }
160 }
161 next.Sort((x, y) => x[0].CompareTo(y[0]));
162 if (next.Count == blocks.Count) break;
163 blocks = next;
164 details.Add($" round {++round}: {Partition(a, blocks)}");
165 }
166 int[] finalIndex = BlockIndices(blocks, a.States.Count);
167 List<int> named = new() { finalIndex[a.Start] };
168 Dictionary<int, int> ids = new() { [finalIndex[a.Start]] = 0 };
169 Automaton m = new("DFA", a.Alphabet);
170 m.AddState("M0");
171 for (int i = 0; i < named.Count; i++)
172 {
173 var block = blocks[named[i]];
174 int p = block[0];
175 if (a.Accepting.Contains(p)) m.Accepting.Add(i);
176 details.Add($" M{i} = {a.Set(new(block))}");
177 for (int c = 0; c < a.Alphabet.Length; c++)
178 {
179 int target = finalIndex[a.Delta(p, c)];
180 if (!ids.TryGetValue(target, out int id))
181 {
182 id = named.Count;
183 named.Add(target);
184 ids[target] = id;
185 m.AddState("M" + id);
186 }
187 m.Edge(i, c, id);
188 }
189 }
190 return new(m, details);
191 }

193 static void CopyEdges(Automaton source, Automaton target, int offset, bool reverse)
194 {
195 for (int p = 0; p < source.States.Count; p++) for (int c = 0; c <= source.Alphabet.Length; c++) foreach (int q in source.Transitions[p][c]) target.Edge((reverse ? q : p) + offset, c, (reverse ? p : q) + offset);
196 }

198 static Automaton EpsilonFree(Automaton a)
199 {
200 Automaton n = new("NFA", a.Alphabet);
201 foreach (string state in a.States) n.AddState(state);
202 n.Start = a.Start;
203 for (int p = 0; p < a.States.Count; p++)
204 {
205 var closure = a.Close(Singleton(p));
206 if (a.Accepts(closure)) n.Accepting.Add(p);
207 for (int c = 0; c < a.Alphabet.Length; c++) n.Transitions[p][c].UnionWith(a.Move(closure, c));
208 }
209 return n;
210 }

212 static Automaton UnaryConstruction(Automaton a, string command)
213 {
214 if (command == "COMPLEMENT")
215 {
216 Automaton n = new("DFA", a.Alphabet);
217 foreach (string state in a.States) n.AddState(state);
218 n.Start = a.Start;
219 CopyEdges(a, n, 0, false);
220 n.Accepting.UnionWith(Enumerable.Range(0, a.States.Count));
221 n.Accepting.ExceptWith(a.Accepting);
222 return n;
223 }
224 Automaton result = new("NFA", a.Alphabet);
225 result.Names("N", a.States.Count + 1);
226 CopyEdges(a, result, 1, command == "REVERSE");
227 if (command == "REVERSE")
228 {
229 result.Accepting.Add(a.Start + 1);
230 foreach (int p in a.Accepting) result.Edge(0, a.Alphabet.Length, p + 1);
231 }
232 else
233 {
234 result.Accepting.Add(0);
235 result.Edge(0, a.Alphabet.Length, a.Start + 1);
236 foreach (int p in a.Accepting)
237 {
238 result.Accepting.Add(p + 1);
239 result.Edge(p + 1, a.Alphabet.Length, a.Start + 1);
240 }
241 }
242 return result;
243 }

245 static Automaton Concatenate(Automaton a, Automaton b)
246 {
247 Automaton n = new("NFA", a.Alphabet);
248 int offset = a.States.Count;
249 n.Names("N", offset + b.States.Count);
250 n.Start = a.Start;
251 CopyEdges(a, n, 0, false);
252 CopyEdges(b, n, offset, false);
253 foreach (int p in a.Accepting) n.Edge(p, a.Alphabet.Length, offset + b.Start);
254 foreach (int p in b.Accepting) n.Accepting.Add(offset + p);
255 return n;

259 static Created Product(Automaton a, Automaton b, string command)
260 {
261 Automaton p = new("DFA", a.Alphabet);
262 Pair first = new(a.Start, b.Start);
263 List<Pair> pairs = new() { first };
264 Dictionary<Pair, int> ids = new() { [first] = 0 };
265 List<string> details = new();
266 p.AddState("P0");
267 for (int i = 0; i < pairs.Count; i++)
268 {
269 Pair pair = pairs[i];
270 bool x = a.Accepting.Contains(pair.A), y = b.Accepting.Contains(pair.B);
271 if (command == "UNION" ? x || y : command == "INTERSECT" ? x && y : x && !y) p.Accepting.Add(i);
272 details.Add($" P{i} = ({a.States[pair.A]},{b.States[pair.B]})");
273 for (int c = 0; c < a.Alphabet.Length; c++)
274 {
275 Pair next = new(a.Delta(pair.A, c), b.Delta(pair.B, c));
276 if (!ids.TryGetValue(next, out int id))
277 {
278 id = pairs.Count;
279 pairs.Add(next);
280 ids[next] = id;
281 p.AddState("P" + id);
282 }
283 p.Edge(i, c, id);
284 }
285 }
286 return new(p, details);
287 }

302 static string Equivalent(Automaton a, Automaton b, string x, string z)
303 {
304 a = Determinize(a).Automaton;
305 b = Determinize(b).Automaton;
306 Pair first = new(a.Start, b.Start);
307 List<Pair> pairs = new() { first };
308 Dictionary<Pair, int> ids = new() { [first] = 0 };
309 List<int> parents = new() { -1 };
310 List<char> incoming = new() { ' ' };
311 for (int i = 0; i < pairs.Count; i++)
312 {
313 Pair pair = pairs[i];
314 bool accepted = a.Accepting.Contains(pair.A);
315 if (accepted != b.Accepting.Contains(pair.B)) return $"NOT EQUIVALENT: {Visible(Word(parents, incoming, i))} is accepted only by {(accepted ? x : z)}";
316 for (int c = 0; c < a.Alphabet.Length; c++)
317 {
318 Pair next = new(a.Delta(pair.A, c), b.Delta(pair.B, c));
319 if (!ids.ContainsKey(next))
320 {
321 ids[next] = pairs.Count;
322 pairs.Add(next);
323 parents.Add(i);
324 incoming.Add(a.Alphabet[c]);
325 }
326 }
327 }
328 return "EQUIVALENT";
329 }

331 static string? Smallest(Automaton d)
332 {
333 List<int> states = new() { d.Start }, parents = new() { -1 };
334 List<char> incoming = new() { ' ' };
335 bool[] seen = new bool[d.States.Count];
336 seen[d.Start] = true;
337 for (int i = 0; i < states.Count; i++)
338 {
339 int p = states[i];
340 if (d.Accepting.Contains(p)) return Word(parents, incoming, i);
341 for (int c = 0; c < d.Alphabet.Length; c++)
342 {
343 int q = d.Delta(p, c);
344 if (!seen[q])

346 seen[q] = true;
347 states.Add(q);
348 parents.Add(i);
349 incoming.Add(d.Alphabet[c]);
350 }
351 }
352 }
353 return null;

358 static Language Classify(Automaton d, bool countWords)
359 {
360 var reachable = d.GraphReach(false);
361 var live = d.GraphReach(true);
362 live.IntersectWith(reachable);
363 if (!live.Contains(d.Start)) return new("empty", 0);
364 int[] indegree = new int[d.States.Count];
365 foreach (int p in live) for (int c = 0; c < d.Alphabet.Length; c++)
366 {
367 int q = d.Delta(p, c);
368 if (live.Contains(q)) indegree[q]++;
369 }
370 List<int> order = live.Where(p => indegree[p] == 0).Order().ToList();
371 for (int i = 0; i < order.Count; i++) for (int c = 0; c < d.Alphabet.Length; c++)
372 {
373 int q = d.Delta(order[i], c);
374 if (live.Contains(q) && --indegree[q] == 0) order.Add(q);
375 }
376 if (order.Count != live.Count) return new("infinite", 0);
377 if (!countWords) return new("finite", 0);
378 long[] counts = new long[d.States.Count];
379 for (int i = order.Count - 1; i >= 0; i--)
380 {
381 int p = order[i];
382 counts[p] = d.Accepting.Contains(p) ? 1 : 0;
383 for (int c = 0; c < d.Alphabet.Length; c++) counts[p] += counts[d.Delta(p, c)];
384 }
385 return new("finite", counts[d.Start]);
386 }

388 static long Count(Automaton d, int length)
389 {
390 long[] values = new long[d.States.Count];
391 values[d.Start] = 1;
392 for (int i = 0; i < length; i++)
393 {
394 long[] next = new long[values.Length];
395 for (int p = 0; p < values.Length; p++) for (int c = 0; c < d.Alphabet.Length; c++) next[d.Delta(p, c)] += values[p];
396 values = next;
397 }
398 return d.Accepting.Sum(p => values[p]);
399 }

425 static string Synchronize(Automaton d)
426 {
427 int start = (1 << d.States.Count) - 1;
428 List<int> sets = new() { start }, parents = new() { -1 };
429 List<char> incoming = new() { ' ' };
430 Dictionary<int, int> ids = new() { [start] = 0 };
431 for (int i = 0; i < sets.Count; i++)
432 {
433 int set = sets[i];
434 if (BitOperations.PopCount((uint)set) == 1) return $"SYNCHRONIZING {Visible(Word(parents, incoming, i))} -> {d.States[BitOperations.TrailingZeroCount((uint)set)]}";
435 for (int c = 0; c < d.Alphabet.Length; c++)
436 {
437 int next = 0;
438 for (int p = 0; p < d.States.Count; p++) if ((set & 1 << p) != 0) next |= 1 << d.Delta(p, c);
439 if (!ids.ContainsKey(next))
440 {
441 ids[next] = sets.Count;
442 sets.Add(next);
443 parents.Add(i);
444 incoming.Add(d.Alphabet[c]);
445 }
446 }
447 }
448 return "NOT SYNCHRONIZING";
449 }

453 class RegexParser(string text)
454 {
455 int at;
456 char Peek() => at == text.Length ? '\0' : text[at];
457 void Error() => Fail("syntax error at " + (at + 1));
458 public static bool Symbol(char c) => c is >= 'a' and <= 'z' or >= '0' and <= '9';
459 Regex Alternation()
460 {
461 List<Regex> list = new() { Concatenation() };
462 while (Peek() == '|')
463 {
464 at++;
465 list.Add(Concatenation());
466 }
467 return list.Count == 1 ? list[0] : new('|', list);
468 }
469 Regex Concatenation()
470 {
471 List<Regex> list = new() { Factor() };
472 while (Symbol(Peek()) || Peek() == '(') list.Add(Factor());
473 return list.Count == 1 ? list[0] : new('.', list);
474 }
475 Regex Factor()
476 {
477 Regex r = Atom();
478 while (Peek() is '*' or '+' or '?') r = new(text[at++], new() { r });
479 return r;
480 }
481 Regex Atom()
482 {
483 char c = Peek();
484 if (Symbol(c))
485 {
486 at++;
487 return new(c, new());
488 }
489 if (c == '(')
490 {
491 at++;
492 Regex r = Alternation();
493 if (Peek() != ')') Error();
494 at++;
495 return r;
496 }
497 Error();
498 return null!;
499 }
500 public Regex Parse()
501 {
502 Regex r = Alternation();
503 if (at != text.Length) Error();
504 return r;
505 }
506 }
507 readonly record struct Fragment(int Entry, int Exit);
508
509 static Fragment Build(Regex regex, Automaton a)
510 {
511 if (regex.Kind == '.')
512 {
513 Fragment? first = null, last = null;
514 foreach (Regex child in regex.Children)
515 {
516 Fragment next = Build(child, a);
517 if (last.HasValue) a.Edge(last.Value.Exit, a.Alphabet.Length, next.Entry);
518 else first = next;
519 last = next;
520 }
521 return new(first!.Value.Entry, last!.Value.Exit);
522 }
523 int entry = a.AddState("N" + a.States.Count);
524 if (regex.Children.Count == 0)
525 {
526 int exit = a.AddState("N" + a.States.Count);
527 a.Edge(entry, regex.Kind == 'e' ? a.Alphabet.Length : a.Alphabet.IndexOf(regex.Kind), exit);
528 return new(entry, exit);
529 }
530 List<Fragment> parts = regex.Children.Select(child => Build(child, a)).ToList();
531 int final = a.AddState("N" + a.States.Count);
532 foreach (Fragment f in parts)
533 {
534 a.Edge(entry, a.Alphabet.Length, f.Entry);
535 a.Edge(f.Exit, a.Alphabet.Length, final);
536 }
537 if (regex.Kind != '|')
538 {
539 Fragment f = parts[0];
540 if (regex.Kind != '?') a.Edge(f.Exit, a.Alphabet.Length, f.Entry);
541 if (regex.Kind != '+') a.Edge(entry, a.Alphabet.Length, final);
542 }
543 return new(entry, final);
544 }

546 Automaton Explicit(string[] w)
547 {
548 if (w.Length < 6) Fail("wrong number of arguments");
549 if (automata.ContainsKey(w[1])) Fail(w[1] + " already exists");
550 Automaton a = new(w[0], Alphabet(w[3]));
551 foreach (string state in w[2].Split(','))
552 {
553 if (a.States.Contains(state)) Fail("duplicate state " + state);
554 a.AddState(state);
555 }
556 a.Start = a.States.IndexOf(w[4]);
557 if (a.Start < 0) Fail("unknown state " + w[4]);
558 if (w[5] != "-") foreach (string state in w[5].Split(','))
559 {
560 int p = a.States.IndexOf(state);
561 if (p < 0) Fail("unknown state " + state);
562 if (!a.Accepting.Add(p)) Fail("duplicate state " + state);
563 }
564 for (int i = 6; i < w.Length; i++)
565 {
566 string[] t = w[i].Split(':');
567 int p = a.States.IndexOf(t[0]);
568 char c = t[1][0];
569 if (p < 0) Fail("unknown state " + t[0]);
570 if (c == 'e' && a.Kind == "DFA") Fail("epsilon transition in DFA");
571 int symbol = c == 'e' ? a.Alphabet.Length : a.Alphabet.IndexOf(c);
572 if (symbol < 0) Fail("unknown symbol " + c);
573 int q = a.States.IndexOf(t[2]);
574 if (q < 0) Fail("unknown state " + t[2]);
575 var edges = a.Transitions[p][symbol];
576 if (edges.Contains(q)) Fail("duplicate transition " + w[i]);
577 if (a.Kind == "DFA" && edges.Count > 0) Fail("conflicting transition " + w[i]);
578 a.Edge(p, symbol, q);
579 }
580 if (a.Kind == "DFA") for (int p = 0; p < a.States.Count; p++) for (int c = 0; c < a.Alphabet.Length; c++) if (a.Transitions[p][c].Count == 0) Fail($"missing transition {a.States[p]}:{a.Alphabet[c]}");
581 return a;
582 }

584 static List<string> Table(Automaton a)
585 {
586 int columns = a.Alphabet.Length + 1 + (a.Kind == "NFA" && a.EpsilonCount > 0 ? 1 : 0);
587 string[][] cells = Enumerable.Range(0, a.States.Count + 1).Select(_ => new string[columns]).ToArray();
588 cells[0][0] = "";
589 for (int c = 1; c < columns; c++) cells[0][c] = c <= a.Alphabet.Length ? a.Alphabet[c - 1].ToString() : "e";
590 for (int p = 0; p < a.States.Count; p++)
591 {
592 cells[p + 1][0] = (p == a.Start ? "->" : " ") + (a.Accepting.Contains(p) ? "*" : " ") + a.States[p];
593 for (int c = 1; c < columns; c++)
594 {
595 var edges = a.Transitions[p][c - 1];
596 cells[p + 1][c] = a.Kind == "DFA" ? a.States[edges.First()] : edges.Count == 0 ? "-" : a.Set(edges);
597 }
598 }
599 int[] widths = Enumerable.Range(0, columns).Select(c => cells.Max(row => row[c].Length)).ToArray();
600 return cells.Select(row => string.Join(" ", row.Select((cell, c) => cell.PadRight(widths[c]))).TrimEnd(' ')).ToList();
601 }

621 ["CLOSURE"] = "RO",
622 ["ENUM"] = "RO",

624 ["TODFA"] = "RN",

641 ["SYNC"] = "D",
642 ["LIST"] = ""

645 List<string> Command(string line)
646 {
647 string[] w = line.Split(' ');
648 string c = w[0];
649 if (!Roles.ContainsKey(c) && c is not ("DFA" or "NFA" or "REGEX")) Fail("unknown command " + c);
650 if (c is "DFA" or "NFA")
651 {
652 Automaton created = Explicit(w);
653 return Register(w[1], new(created, new()));
654 }
655 if (c == "REGEX")
656 {
657 if (w.Length != 4) Fail("wrong number of arguments");
658 if (automata.ContainsKey(w[1])) Fail(w[1] + " already exists");
659 foreach (char ch in w[3]) if (RegexParser.Symbol(ch) && ch != 'e' && !w[2].Contains(ch)) Fail("unknown symbol " + ch);
660 Regex regex = new RegexParser(w[3]).Parse();
661 Automaton created = new("NFA", Alphabet(w[2]));
662 Fragment fragment = Build(regex, created);
663 created.Start = fragment.Entry;
664 created.Accepting.Add(fragment.Exit);
665 return Register(w[1], new(created, new()));
666 }
667 string roles = Roles[c];
668 if (w.Length != roles.Length + 1) Fail("wrong number of arguments");
669 for (int i = 0; i < roles.Length; i++) if (roles[i] is 'R' or 'D') Get(w[i + 1]);
670 for (int i = 0; i < roles.Length; i++) if (roles[i] == 'N' && automata.ContainsKey(w[i + 1])) Fail(w[i + 1] + " already exists");
671 for (int i = 0; i < roles.Length; i++) if (roles[i] == 'D' && Get(w[i + 1]).Kind == "NFA") Fail(w[i + 1] + " is not a DFA");
672 if ((c is "EQUIV" or "UNION" or "INTERSECT" or "DIFF" or "CONCAT") && Get(w[1]).Alphabet != Get(w[2]).Alphabet) Fail("alphabets differ");
673 if (c == "LIST") return registration.Count == 0 ? new() { "(none)" } : registration.Select(name => automata[name].Summary(name)).ToList();
674 Automaton a = Get(w[1]);
675 switch (c)
676 {
677 case "DELETE":
678 automata.Remove(w[1]);
679 registration.Remove(w[1]);
680 return new() { "deleted " + w[1] };
681 case "ACCEPT":
682 {
683 string word = w[2] == "e" ? "" : w[2];
684 foreach (char ch in word) if (!a.Alphabet.Contains(ch)) Fail("unknown symbol " + ch);
685 var set = a.Initial();
686 StringBuilder path = new(a.Kind == "DFA" ? a.States[set.Single()] : a.Set(set));
687 foreach (char ch in word)
688 {
689 set = a.Move(set, a.Alphabet.IndexOf(ch));
690 path.Append(" -").Append(ch).Append("-> ").Append(a.Kind == "DFA" ? a.States[set.Single()] : a.Set(set));
691 }
692 return new() { (a.Accepts(set) ? "ACCEPT " : "REJECT ") + path };
693 }
694 case "CLOSURE":
695 {
696 HashSet<int> set = new();
697 foreach (string state in w[2].Split(','))
698 {
699 int p = a.States.IndexOf(state);
700 if (p < 0) Fail("unknown state " + state);
701 set.Add(p);
702 }
703 return new() { a.Set(a.Close(set)) };
704 }
705 case "TODFA":
706 return Register(w[2], Determinize(a));
707 case "EPSFREE":
708 return Register(w[2], new(EpsilonFree(a), new()));
709 case "MINIMIZE":
710 return Register(w[2], Minimize(a));
711 case "COMPLEMENT":
712 case "REVERSE":
713 case "STAR":
714 return Register(w[2], new(UnaryConstruction(a, c), new()));
715 case "CONCAT":
716 return Register(w[3], new(Concatenate(a, Get(w[2])), new()));
717 case "UNION":
718 case "INTERSECT":
719 case "DIFF":
720 return Register(w[3], Product(a, Get(w[2]), c));
721 case "EQUIV":
722 return new() { Equivalent(a, Get(w[2]), w[1], w[2]) };
723 case "TABLE":
724 return Table(a);
725 case "SHOW":
726 return new() { Show(a, w[1]) };
727 case "SYNC":
728 return new() { Synchronize(a) };
729 }
730 Automaton d = Determinize(a).Automaton;
731 switch (c)
732 {
733 case "COUNT":
734 return new() { Count(d, int.Parse(w[2])).ToString() };
735 case "ENUM":
736 return new() { Enumerate(d, int.Parse(w[2])) };
737 case "EMPTY":
738 string? smallest = Smallest(d);
739 return new() { smallest == null ? "EMPTY" : "NOT EMPTY " + Visible(smallest) };
740 case "FINITE":
741 Language language = Classify(d, true);
742 return new() { language.Kind == "infinite" ? "INFINITE" : "FINITE " + language.Count };
743 case "INFO":
744 bool deterministic = a.EpsilonCount == 0, complete = true;
745 foreach (var row in a.Transitions) for (int symbol = 0; symbol < a.Alphabet.Length; symbol++)
746 {
747 deterministic &= row[symbol].Count <= 1;
748 complete &= row[symbol].Count > 0;
749 }
750 return new() { a.Summary(w[1]), " epsilon transitions: " + a.EpsilonCount, " reachable: " + a.List(a.GraphReach(false)), " live: " + a.List(a.GraphReach(true)), " deterministic: " + (deterministic ? "yes" : "no"), " complete: " + (complete ? "yes" : "no"), " language: " + Classify(d, false).Kind, " minimal DFA states: " + Minimize(d).Automaton.States.Count };
751 }
752 throw new InvalidOperationException();
753 }

755 static void Main()
756 {
757 CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
758 int count = int.Parse(Console.ReadLine()!);
759 Program toolkit = new();
760 for (int i = 0; i < count; i++)
761 {
762 string line = Console.ReadLine()!;
763 if (line.StartsWith('#')) continue;
764 try
765 {
766 foreach (string output in toolkit.Command(line)) Console.WriteLine(output);
767 }
768 catch (Problem problem)
769 {
770 Console.WriteLine(problem.Message);
771 }
772 }
773 }
