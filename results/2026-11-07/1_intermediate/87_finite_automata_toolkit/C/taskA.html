17static void *take(Allocation **arena, size_t size) {
18 Allocation *a = malloc(sizeof *a);
19 void *p = calloc(1, size ? size : 1);
20 if (!a || !p) exit(EXIT_FAILURE);
21 *a = (Allocation){p, *arena};
22 *arena = a;
23 return p;
24}

54 size_t length = strlen(s);
55 for (const char *p = list; *p;) {
56 const char *end = strchr(p, ' ');
57 size_t n = end ? (size_t)(end - p) : strlen(p);

75typedef struct {

80static void append(Buffer *b, const char *text) {
81 size_t n = strlen(text);
82 if (b->length + n + 1 > b->capacity) {
83 size_t capacity = b->capacity ? b->capacity : 64;
84 while (capacity < b->length + n + 1) capacity *= 2;
85 char *p = realloc(b->text, capacity);
86 if (!p) exit(EXIT_FAILURE);
87 b->text = p;
88 b->capacity = capacity;
89 }
90 memcpy(b->text + b->length, text, n + 1);
91 b->length += n;
92}

131typedef struct {
132 uint64_t words[SET_WORDS];

135static bool has(Set s, int i) {
136 return !!(s.words[i / 64] & (UINT64_C(1) << (i % 64)));
137}
138
139static bool insert(Set *s, int i) {
140 bool old = has(*s, i);
141 s->words[i / 64] |= UINT64_C(1) << (i % 64);
142 return !old;
143}

145static Set singleton(int i) {

147 {0}

153static void unite(Set *a, Set b) {
154 for (int i = 0; i < SET_WORDS; i++) a->words[i] |= b.words[i];
155}
156
157static bool intersects(Set a, Set b) {
158 for (int i = 0; i < SET_WORDS; i++) if (a.words[i] & b.words[i]) return true;
159 return false;
160}

162static bool same_set(Set a, Set b) {
163 for (int i = 0; i < SET_WORDS; i++) if (a.words[i] != b.words[i]) return false;
164 return true;
165}

167static int set_count(Set s) {
168 int count = 0;
169 for (int i = 0; i < SET_WORDS; i++) {
170 uint64_t v = s.words[i];
171 while (v) {
172 v &= v - 1;
173 count++;
174 }
175 }
176 return count;
177}

270static Set closure(const Automaton *a, Set s) {
271 int queue[MAX_STATES], count = 0;
272 for (int i = 0; i < a->count; i++) if (has(s, i)) queue[count++] = i;
273 for (int at = 0; at < count; at++) {
274 Set edges = a->transitions[queue[at]][a->symbols];
275 for (int q = 0; q < a->count; q++) if (has(edges, q) && insert(&s, q)) queue[count++] = q;
276 }
277 return s;
278}

284static Set move(const Automaton *a, Set s, int c) {
285 Set out = {
286 {0}
287 };
288 for (int p = 0; p < a->count; p++) if (has(s, p)) unite(&out, a->transitions[p][c]);
289 return closure(a, out);
290}

292static Set graph_reach(const Automaton *a, bool reverse) {
293 Set out = reverse ? a->accepting : singleton(a->start);
294 bool changed;
295 do {
296 int before = set_count(out);
297 for (int p = 0; p < a->count; p++) for (int c = 0; c <= a->symbols; c++) {
298 if (reverse) {
299 if (intersects(a->transitions[p][c], out)) insert(&out, p);
300 } else if (has(out, p)) unite(&out, a->transitions[p][c]);
301 }
302 changed = before != set_count(out);
303 } while (changed);
304 return out;
305}

341static Automaton *determinize(const Automaton *a, Lines *details) {
342 Automaton *d = automaton(true, a->alphabet);
343 Set sets[MAX_STATES];
344 int count = 1;
345 sets[0] = initial(a);
346 add_state(d, "D0");
347 for (int i = 0; i < count; i++) {
348 if (intersects(sets[i], a->accepting)) insert(&d->accepting, i);
349 if (details) add_line(details, format(" D%d = %s", i, set_text(a, sets[i])));
350 for (int c = 0; c < a->symbols; c++) {
351 Set next = move(a, sets[i], c);
352 int id = -1;
353 for (int j = 0; j < count; j++) if (same_set(sets[j], next)) {
354 id = j;
355 break;
356 }
357 if (id < 0) {
358 id = count;
359 sets[count++] = next;
360 add_state(d, format("D%d", id));
361 }
362 edge(d, i, c, id);
363 }
364 }
365 return d;
366}

386static Automaton *minimize(const Automaton *a, Lines *details) {
387 Set reached = graph_reach(a, false), missing = {
388 {0}
389 };
390 for (int p = 0; p < a->count; p++) if (!has(reached, p)) insert(&missing, p);
391 if (details) add_line(details, format(" unreachable: %s", list_text(a, missing)));
392 Partition current = {0};
393 for (int p = 0; p < a->count; p++) if (has(reached, p)) {
394 int id = -1;
395 for (int i = 0; i < current.count; i++) if (has(a->accepting, first(current.blocks[i])) == has(a->accepting, p)) {
396 id = i;
397 break;
398 }
399 if (id < 0) id = current.count++;
400 insert(&current.blocks[id], p);
401 }
402 index_partition(&current, a->count);
403 int round = 0;
404 if (details) add_line(details, format(" round 0: %s", partition_text(a, &current)));
405 while (true) {
406 Partition next = {0};
407 for (int p = 0; p < a->count; p++) if (has(reached, p)) {
408 int id = -1;
409 for (int i = 0; i < next.count; i++) {
410 int q = first(next.blocks[i]);
411 if (current.which[p] != current.which[q]) continue;
412 bool same = true;
413 for (int c = 0; c < a->symbols; c++) if (current.which[delta(a, p, c)] != current.which[delta(a, q, c)]) {
414 same = false;
415 break;
416 }
417 if (same) {
418 id = i;
419 break;
420 }
421 }
422 if (id < 0) id = next.count++;
423 insert(&next.blocks[id], p);
424 }
425 if (next.count == current.count) break;
426 current = next;
427 index_partition(&current, a->count);
428 if (details) add_line(details, format(" round %d: %s", ++round, partition_text(a, &current)));
429 }
430 Automaton *m = automaton(true, a->alphabet);
431 int ids[MAX_STATES], queue[MAX_STATES], count = 1;
432 for (int i = 0; i < current.count; i++) ids[i] = -1;
433 queue[0] = current.which[a->start];
434 ids[queue[0]] = 0;
435 add_state(m, "M0");
436 for (int i = 0; i < count; i++) {
437 Set block = current.blocks[queue[i]];
438 int p = first(block);
439 if (has(a->accepting, p)) insert(&m->accepting, i);
440 if (details) add_line(details, format(" M%d = %s", i, set_text(a, block)));
441 for (int c = 0; c < a->symbols; c++) {
442 int b = current.which[delta(a, p, c)];
443 if (ids[b] < 0) {
444 ids[b] = count;
445 queue[count++] = b;
446 add_state(m, format("M%d", ids[b]));
447 }
448 edge(m, i, c, ids[b]);
449 }
450 }
451 return m;
452}

454static void copy_edges(const Automaton *source, Automaton *target, int offset, bool reverse) {
455 for (int p = 0; p < source->count; p++) for (int c = 0; c <= source->symbols; c++) for (int q = 0; q < source->count; q++) if (has(source->transitions[p][c], q)) edge(target, (reverse ? q : p) + offset, c, (reverse ? p : q) + offset);
456}

458static Automaton *epsilon_free(const Automaton *a) {
459 Automaton *n = automaton(false, a->alphabet);
460 for (int p = 0; p < a->count; p++) add_state(n, a->states[p]);
461 n->start = a->start;
462 for (int p = 0; p < a->count; p++) {
463 Set set = closure(a, singleton(p));
464 if (intersects(set, a->accepting)) insert(&n->accepting, p);
465 for (int c = 0; c < a->symbols; c++) n->transitions[p][c] = move(a, set, c);
466 }
467 return n;

470static Automaton *unary_construction(const Automaton *a, const char *command) {
471 if (equal(command, "COMPLEMENT")) {
472 Automaton *n = allocate(sizeof *n);
473 *n = *a;
474 n->accepting = (Set) {
475 {0}
476 };
477 for (int p = 0; p < a->count; p++) if (!has(a->accepting, p)) insert(&n->accepting, p);
478 return n;
479 }
480 Automaton *n = automaton(false, a->alphabet);
481 series(n, 'N', a->count + 1);
482 copy_edges(a, n, 1, equal(command, "REVERSE"));
483 if (equal(command, "REVERSE")) {
484 insert(&n->accepting, a->start + 1);
485 for (int p = 0; p < a->count; p++) if (has(a->accepting, p)) edge(n, 0, a->symbols, p + 1);
486 } else {
487 insert(&n->accepting, 0);
488 edge(n, 0, a->symbols, a->start + 1);
489 for (int p = 0; p < a->count; p++) if (has(a->accepting, p)) {
490 insert(&n->accepting, p + 1);
491 edge(n, p + 1, a->symbols, a->start + 1);
492 }
493 }
494 return n;
495}

497static Automaton *concatenate(const Automaton *a, const Automaton *b) {
498 Automaton *n = automaton(false, a->alphabet);
499 int offset = a->count;
500 series(n, 'N', offset + b->count);
501 n->start = a->start;
502 copy_edges(a, n, 0, false);
503 copy_edges(b, n, offset, false);
504 for (int p = 0; p < a->count; p++) if (has(a->accepting, p)) edge(n, p, a->symbols, offset + b->start);
505 for (int p = 0; p < b->count; p++) if (has(b->accepting, p)) insert(&n->accepting, offset + p);
506 return n;
507}

512static Automaton *product(const Automaton *a, const Automaton *b, const char *command, Lines *details) {
513 Automaton *p = automaton(true, a->alphabet);
514 Pair pairs[MAX_STATES] = {
515 {a->start, b->start}
516 };
517 int count = 1;
518 add_state(p, "P0");
519 for (int i = 0; i < count; i++) {
520 Pair pair = pairs[i];
521 bool x = has(a->accepting, pair.a), y = has(b->accepting, pair.b);
522 if (equal(command, "UNION") ? x || y : equal(command, "INTERSECT") ? x && y : x && !y) insert(&p->accepting, i);
523 add_line(details, format(" P%d = (%s,%s)", i, a->states[pair.a], b->states[pair.b]));
524 for (int c = 0; c < a->symbols; c++) {
525 Pair next = {delta(a, pair.a, c), delta(b, pair.b, c)};
526 int id = -1;
527 for (int j = 0; j < count; j++) if (pairs[j].a == next.a && pairs[j].b == next.b) {
528 id = j;
529 break;
530 }
531 if (id < 0) {
532 id = count;
533 pairs[count++] = next;
534 add_state(p, format("P%d", id));
535 }
536 edge(p, i, c, id);
537 }
538 }
539 return p;
540}

561static char *equivalent(const Automaton *x, const Automaton *z, const char *xname, const char *zname) {
562 Automaton *a = determinize(x, NULL), *b = determinize(z, NULL);
563 int maximum = a->count * b->count;
564 Search *queue = allocate((size_t)maximum * sizeof(Search));
565 int *ids = allocate((size_t)maximum * sizeof(int));
566 for (int i = 0; i < maximum; i++) ids[i] = -1;
567 queue[0] = (Search){a->start, b->start, -1, ' '};
568 ids[a->start * b->count + b->start] = 0;
569 int count = 1;
570 for (int i = 0; i < count; i++) {
571 Search pair = queue[i];
572 bool accepted = has(a->accepting, pair.a);
573 if (accepted != has(b->accepting, pair.b)) return format("NOT EQUIVALENT: %s is accepted only by %s", visible(word(queue, i)), accepted ? xname : zname);
574 for (int c = 0; c < a->symbols; c++) {
575 int p = delta(a, pair.a, c), q = delta(b, pair.b, c), key = p * b->count + q;
576 if (ids[key] < 0) {
577 ids[key] = count;
578 queue[count++] = (Search){p, q, i, a->alphabet[c]};
579 }
580 }
581 }
582 return "EQUIVALENT";
583}

585static char *smallest(const Automaton *d) {
586 Search queue[MAX_STATES] = {
587 {d->start, 0, -1, ' '}
588 };
589 bool seen[MAX_STATES] = {false};
590 seen[d->start] = true;
591 int count = 1;
592 for (int i = 0; i < count; i++) {
593 int p = queue[i].a;
594 if (has(d->accepting, p)) return word(queue, i);
595 for (int c = 0; c < d->symbols; c++) {
596 int q = delta(d, p, c);
597 if (!seen[q]) {
598 seen[q] = true;
599 queue[count++] = (Search){q, 0, i, d->alphabet[c]};
600 }
601 }
602 }
603 return NULL;
604}
605typedef enum { EMPTY_LANGUAGE, FINITE_LANGUAGE, INFINITE_LANGUAGE } Language;
606
607static Language classify(const Automaton *d, long long *total) {
608 Set reached = graph_reach(d, false), live = graph_reach(d, true);
609 for (int i = 0; i < SET_WORDS; i++) live.words[i] &= reached.words[i];
610 if (!has(live, d->start)) {
611 if (total) *total = 0;
612 return EMPTY_LANGUAGE;
613 }
614 int indegree[MAX_STATES] = {0}, order[MAX_STATES], count = 0;
615 for (int p = 0; p < d->count; p++) if (has(live, p)) for (int c = 0; c < d->symbols; c++) {
616 int q = delta(d, p, c);
617 if (has(live, q)) indegree[q]++;
618 }
619 for (int p = 0; p < d->count; p++) if (has(live, p) && !indegree[p]) order[count++] = p;
620 for (int i = 0; i < count; i++) for (int c = 0; c < d->symbols; c++) {
621 int q = delta(d, order[i], c);
622 if (has(live, q) && --indegree[q] == 0) order[count++] = q;
623 }
624 if (count != set_count(live)) return INFINITE_LANGUAGE;
625 if (total) {
626 long long values[MAX_STATES] = {0};
627 for (int i = count - 1; i >= 0; i--) {
628 int p = order[i];
629 values[p] = has(d->accepting, p);
630 for (int c = 0; c < d->symbols; c++) values[p] += values[delta(d, p, c)];
631 }
632 *total = values[d->start];
633 }
634 return FINITE_LANGUAGE;
635}

637static long long count_words(const Automaton *d, int length) {
638 long long values[MAX_STATES] = {0};
639 values[d->start] = 1;
640 for (int i = 0; i < length; i++) {
641 long long next[MAX_STATES] = {0};
642 for (int p = 0; p < d->count; p++) for (int c = 0; c < d->symbols; c++) next[delta(d, p, c)] += values[p];
643 memcpy(values, next, sizeof values);
644 }
645 long long total = 0;
646 for (int p = 0; p < d->count; p++) if (has(d->accepting, p)) total += values[p];
647 return total;
648}

651 int state;

654static char *enumerate(const Automaton *d, int maximum) {
655 Enumeration *level = allocate(sizeof(Enumeration));
656 level[0].state = d->start;
657 int count = 1, first_count = 0;
658 const char *first_words[50];
659 long long total = 0;
660 for (int length = 0; length <= maximum; length++) {
661 Enumeration *next = length < maximum ? allocate((size_t)count * (size_t)d->symbols * sizeof(Enumeration)) : NULL;
662 int next_count = 0;
663 for (int i = 0; i < count; i++) {
664 Enumeration item = level[i];
665 if (has(d->accepting, item.state)) {
666 total++;
667 if (first_count < 50) first_words[first_count++] = *item.text ? part(item.text, (size_t)length) : "e";
668 }
669 if (length < maximum) for (int c = 0; c < d->symbols; c++) {
670 Enumeration *entry = &next[next_count++];
671 memcpy(entry->text, item.text, (size_t)length);
672 entry->text[length] = d->alphabet[c];
673 entry->state = delta(d, item.state, c);
674 }
675 }
676 level = next;
677 count = next_count;
678 }
679 Buffer out = {0};
680 append(&out, format("%lld", total));
681 for (int i = 0; i < first_count; i++) {
682 append(&out, " ");
683 append(&out, first_words[i]);
684 }
685 if (total > 50) append(&out, " ...");
686 return finish(&out);
687}

689static char *synchronize(const Automaton *d) {
690 unsigned start = (1u << d->count) - 1;
691 int ids[1024], count = 1;
692 for (int i = 0; i < 1024; i++) ids[i] = -1;
693 Search queue[1024] = {
694 {(int)start, 0, -1, ' '}
695 };
696 ids[start] = 0;
697 for (int i = 0; i < count; i++) {
698 unsigned set = (unsigned)queue[i].a;
699 if (!(set & (set - 1))) {
700 int p = 0;
701 while (!(set & 1u << p)) p++;
702 return format("SYNCHRONIZING %s -> %s", visible(word(queue, i)), d->states[p]);
703 }
704 for (int c = 0; c < d->symbols; c++) {
705 unsigned next = 0;
706 for (int p = 0; p < d->count; p++) if (set & 1u << p) next |= 1u << delta(d, p, c);
707 if (ids[next] < 0) {
708 ids[next] = count;
709 queue[count++] = (Search){(int)next, 0, i, d->alphabet[c]};
710 }
711 }
712 }
713 return "NOT SYNCHRONIZING";
714}
715typedef struct Regex Regex;
716struct Regex {
717 char kind;
718 Regex **children;
719 int count;
720};
721typedef struct {
722 const char *text;
723 int at;
724} RegexParser;
725
726static char regex_peek(const RegexParser *p) {
727 return p->text[p->at];
728}
729
730static bool regex_symbol(char c) {
731 return (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9');
732}
733
734static Regex *regex_node(char kind, Regex **children, int count) {
735 Regex *r = allocate(sizeof *r);
736 r->kind = kind;
737 r->count = count;
738 r->children = allocate((size_t)count * sizeof(Regex *));
739 if (count) memcpy(r->children, children, (size_t)count * sizeof(Regex *));
740 return r;
741}
742static Regex *alternation(RegexParser *p);
743
744static Regex *regex_atom(RegexParser *p) {
745 char c = regex_peek(p);
746 if (regex_symbol(c)) {
747 p->at++;
748 return regex_node(c, NULL, 0);
749 }
750 if (c == '(') {
751 p->at++;
752 Regex *r = alternation(p);
753 if (regex_peek(p) != ')') fail("syntax error at %d", p->at + 1);
754 p->at++;
755 return r;
756 }
757 fail("syntax error at %d", p->at + 1);
758 return NULL;
759}
760
761static Regex *regex_factor(RegexParser *p) {
762 Regex *r = regex_atom(p);
763 while (regex_peek(p) && strchr("*+?", regex_peek(p))) {
764 char kind = p->text[p->at++];
765 Regex *child[] = {r};
766 r = regex_node(kind, child, 1);
767 }
768 return r;
769}
770
771static Regex *regex_concatenation(RegexParser *p) {
772 Regex *children[41];
773 int count = 0;
774 children[count++] = regex_factor(p);
775 while (regex_symbol(regex_peek(p)) || regex_peek(p) == '(') children[count++] = regex_factor(p);
776 return count == 1 ? children[0] : regex_node('.', children, count);
777}
778
779static Regex *alternation(RegexParser *p) {
780 Regex *children[41];
781 int count = 0;
782 children[count++] = regex_concatenation(p);
783 while (regex_peek(p) == '|') {
784 p->at++;
785 children[count++] = regex_concatenation(p);
786 }
787 return count == 1 ? children[0] : regex_node('|', children, count);
788}
789typedef struct {
790 int entry, exit;
791} Fragment;
792
793static int add_regex_state(Automaton *a) {
794 char name[9];
795 snprintf(name, sizeof name, "N%d", a->count);
796 return add_state(a, name);
797}

799static Fragment build(Regex *r, Automaton *a) {
800 if (r->kind == '.') {
801 Fragment first_part = {0}, last = {0};
802 for (int i = 0; i < r->count; i++) {
803 Fragment next = build(r->children[i], a);
804 if (i) edge(a, last.exit, a->symbols, next.entry);
805 else first_part = next;
806 last = next;
807 }
808 return (Fragment){first_part.entry, last.exit};
809 }
810 int entry = add_regex_state(a);
811 if (!r->count) {
812 int final = add_regex_state(a);
813 edge(a, entry, r->kind == 'e' ? a->symbols : symbol_index(a, r->kind), final);
814 return (Fragment){entry, final};
815 }
816 Fragment parts[41];
817 for (int i = 0; i < r->count; i++) parts[i] = build(r->children[i], a);
818 int final = add_regex_state(a);
819 for (int i = 0; i < r->count; i++) {
820 edge(a, entry, a->symbols, parts[i].entry);
821 edge(a, parts[i].exit, a->symbols, final);
822 }
823 if (r->kind != '|') {
824 if (r->kind != '?') edge(a, parts[0].exit, a->symbols, parts[0].entry);
825 if (r->kind != '+') edge(a, entry, a->symbols, final);
826 }
827 return (Fragment){entry, final};
828}
829typedef struct {
830 const char *items[1001];
831 int count;

834static Words split_words(const char *text, char separator) {
835 Words out = {0};
836 const char *p = text;
837 while (*p) {
838 const char *first_word = p;
839 while (*p && *p != separator) p++;
840 out.items[out.count++] = part(first_word, (size_t)(p - first_word));
841 if (*p) p++;
842 }
843 return out;
844}

846static Automaton *explicit_automaton(Words words) {
847 const char **w = words.items;
848 if (words.count < 6) fail("wrong number of arguments");
849 if (find(w[1]) >= 0) fail("%s already exists", w[1]);
850 Automaton *a = automaton(equal(w[0], "DFA"), w[3]);
851 Words states = split_words(w[2], ',');
852 for (int i = 0; i < states.count; i++) {
853 if (state_index(a, states.items[i]) >= 0) fail("duplicate state %s", states.items[i]);
854 add_state(a, states.items[i]);
855 }
856 a->start = state_index(a, w[4]);
857 if (a->start < 0) fail("unknown state %s", w[4]);
858 if (!equal(w[5], "-")) {
859 Words accepting = split_words(w[5], ',');
860 for (int i = 0; i < accepting.count; i++) {
861 int p = state_index(a, accepting.items[i]);
862 if (p < 0) fail("unknown state %s", accepting.items[i]);
863 if (!insert(&a->accepting, p)) fail("duplicate state %s", accepting.items[i]);
864 }
865 }
866 for (int i = 6; i < words.count; i++) {
867 Words t = split_words(w[i], ':');
868 int p = state_index(a, t.items[0]);
869 char c = *t.items[1];
870 if (p < 0) fail("unknown state %s", t.items[0]);
871 if (c == 'e' && a->dfa) fail("epsilon transition in DFA");
872 int symbol = c == 'e' ? a->symbols : symbol_index(a, c);
873 if (symbol < 0) fail("unknown symbol %c", c);
874 int q = state_index(a, t.items[2]);
875 if (q < 0) fail("unknown state %s", t.items[2]);
876 Set existing = a->transitions[p][symbol];
877 if (has(existing, q)) fail("duplicate transition %s", w[i]);
878 if (a->dfa && set_count(existing)) fail("conflicting transition %s", w[i]);
879 edge(a, p, symbol, q);
880 }
881 if (a->dfa) for (int p = 0; p < a->count; p++) for (int c = 0; c < a->symbols; c++) if (!set_count(a->transitions[p][c])) fail("missing transition %s:%c", a->states[p], a->alphabet[c]);
882 return a;
883}

885static void table(const Automaton *a) {
886 int columns = a->symbols + 1 + (!a->dfa && epsilon_count(a) ? 1 : 0);
887 char *(*cells)[12] = allocate((size_t)(a->count + 1) * sizeof *cells);
888 int widths[12] = {0};
889 cells[0][0] = "";
890 for (int c = 1; c < columns; c++) cells[0][c] = format("%c", c <= a->symbols ? a->alphabet[c - 1] : 'e');
891 for (int p = 0; p < a->count; p++) {
892 cells[p + 1][0] = format("%s%c%s", p == a->start ? "->" : " ", has(a->accepting, p) ? '*' : ' ', a->states[p]);
893 for (int c = 1; c < columns; c++) {
894 Set edges = a->transitions[p][c - 1];
895 cells[p + 1][c] = a->dfa ? (char *)a->states[first(edges)] : !set_count(edges) ? "-" : set_text(a, edges);
896 }
897 }
898 for (int r = 0; r <= a->count; r++) for (int c = 0; c < columns; c++) {
899 int length = (int)strlen(cells[r][c]);
900 if (length > widths[c]) widths[c] = length;
901 }
902 for (int r = 0; r <= a->count; r++) {
903 Buffer line = {0};
904 for (int c = 0; c < columns; c++) {
905 if (c) append(&line, " ");
906 append(&line, cells[r][c]);
907 for (int j = (int)strlen(cells[r][c]); j < widths[c]; j++) append(&line, " ");
908 }
909 while (line.length && line.text[line.length - 1] == ' ') line.text[--line.length] = '\0';
910 puts(line.text);
911 free(line.text);
912 }
913}

915static char *show(const Automaton *a, const char *name) {
916 Buffer b = {0};
917 append(&b, a->dfa ? "DFA " : "NFA ");
918 append(&b, name);
919 append(&b, " ");
920 for (int p = 0; p < a->count; p++) {
921 if (p) append(&b, ",");
922 append(&b, a->states[p]);
923 }
924 append(&b, " ");
925 append(&b, a->alphabet);
926 append(&b, " ");
927 append(&b, a->states[a->start]);
928 append(&b, " ");
929 append(&b, set_count(a->accepting) ? list_text(a, a->accepting) : "-");
930 for (int p = 0; p < a->count; p++) for (int c = 0; c <= a->symbols; c++) for (int q = 0; q < a->count; q++) if (has(a->transitions[p][c], q)) append(&b, format(" %s:%c:%s", a->states[p], c == a->symbols ? 'e' : a->alphabet[c], a->states[q]));
931 return finish(&b);

962static void command(const char *line) {
963 Words words = split_words(line, ' ');
964 const char **w = words.items, *c = w[0], *roles = NULL;
965 for (size_t i = 0; i < sizeof commands / sizeof commands[0]; i++) if (equal(commands[i].name, c)) roles = commands[i].roles;
966 if (!roles && !listed(c, "DFA NFA REGEX")) fail("unknown command %s", c);
967 Lines details = {0};
968 if (equal(c, "DFA") || equal(c, "NFA")) {
969 Automaton *created = explicit_automaton(words);
970 register_automaton(w[1], created, details);
971 return;
972 }
973 if (equal(c, "REGEX")) {
974 if (words.count != 4) fail("wrong number of arguments");
975 if (find(w[1]) >= 0) fail("%s already exists", w[1]);
976 for (const char *p = w[3]; *p; p++) if (regex_symbol(*p) && *p != 'e' && !strchr(w[2], *p)) fail("unknown symbol %c", *p);
977 RegexParser parser = {w[3], 0};
978 Regex *r = alternation(&parser);
979 if (regex_peek(&parser)) fail("syntax error at %d", parser.at + 1);
980 Automaton *created = automaton(false, w[2]);
981 Fragment fragment = build(r, created);
982 created->start = fragment.entry;
983 insert(&created->accepting, fragment.exit);
984 register_automaton(w[1], created, details);
985 return;
986 }
987 int count = (int)strlen(roles);
988 if (words.count != count + 1) fail("wrong number of arguments");
989 for (int i = 0; i < count; i++) if (roles[i] == 'R' || roles[i] == 'D') get(w[i + 1]);
990 for (int i = 0; i < count; i++) if (roles[i] == 'N' && find(w[i + 1]) >= 0) fail("%s already exists", w[i + 1]);
991 for (int i = 0; i < count; i++) if (roles[i] == 'D' && !get(w[i + 1])->dfa) fail("%s is not a DFA", w[i + 1]);
992 if (listed(c, "EQUIV UNION INTERSECT DIFF CONCAT") && !equal(get(w[1])->alphabet, get(w[2])->alphabet)) fail("alphabets differ");
993 if (equal(c, "LIST")) {
994 if (!entry_count) puts("(none)");
995 for (int i = 0; i < entry_count; i++) puts(summary(entries[i].automaton, entries[i].name));
996 return;
997 }
998 Automaton *a = get(w[1]);
999 if (equal(c, "DELETE")) {
1000 int at = find(w[1]);
1001 memmove(entries + at, entries + at + 1, (size_t)(entry_count - at - 1) * sizeof(Entry));
1002 entry_count--;
1003 printf("deleted %s\n", w[1]);
1004 return;
1005 }
1006 if (equal(c, "ACCEPT")) {
1007 const char *text = equal(w[2], "e") ? "" : w[2];
1008 for (const char *p = text; *p; p++) if (symbol_index(a, *p) < 0) fail("unknown symbol %c", *p);
1009 Set set = initial(a);
1010 Buffer path = {0};
1011 append(&path, a->dfa ? a->states[first(set)] : set_text(a, set));
1012 for (const char *p = text; *p; p++) {
1013 set = move(a, set, symbol_index(a, *p));
1014 append(&path, format(" -%c-> %s", *p, a->dfa ? a->states[first(set)] : set_text(a, set)));
1015 }
1016 printf("%s %s\n", intersects(set, a->accepting) ? "ACCEPT" : "REJECT", path.text);
1017 free(path.text);
1018 return;
1019 }
1020 if (equal(c, "CLOSURE")) {
1021 Words states = split_words(w[2], ',');
1022 Set set = {
1023 {0}
1024 };
1025 for (int i = 0; i < states.count; i++) {
1026 int p = state_index(a, states.items[i]);
1027 if (p < 0) fail("unknown state %s", states.items[i]);
1028 insert(&set, p);
1029 }
1030 puts(set_text(a, closure(a, set)));
1031 return;
1032 }
1033 if (equal(c, "TODFA")) {
1034 Automaton *created = determinize(a, &details);
1035 register_automaton(w[2], created, details);
1036 return;
1037 }
1038 if (equal(c, "EPSFREE")) {
1039 register_automaton(w[2], epsilon_free(a), details);
1040 return;
1041 }
1042 if (equal(c, "MINIMIZE")) {
1043 Automaton *created = minimize(a, &details);
1044 register_automaton(w[2], created, details);
1045 return;
1046 }
1047 if (listed(c, "COMPLEMENT REVERSE STAR")) {
1048 register_automaton(w[2], unary_construction(a, c), details);
1049 return;
1050 }
1051 if (equal(c, "CONCAT")) {
1052 register_automaton(w[3], concatenate(a, get(w[2])), details);
1053 return;
1054 }
1055 if (listed(c, "UNION INTERSECT DIFF")) {
1056 Automaton *created = product(a, get(w[2]), c, &details);
1057 register_automaton(w[3], created, details);
1058 return;
1059 }
1060 if (equal(c, "EQUIV")) {
1061 puts(equivalent(a, get(w[2]), w[1], w[2]));
1062 return;
1063 }
1064 if (equal(c, "TABLE")) {
1065 table(a);
1066 return;
1067 }
1068 if (equal(c, "SHOW")) {
1069 puts(show(a, w[1]));
1070 return;
1071 }
1072 if (equal(c, "SYNC")) {
1073 puts(synchronize(a));
1074 return;
1075 }
1076 Automaton *d = determinize(a, NULL);
1077 if (equal(c, "COUNT")) printf("%lld\n", count_words(d, atoi(w[2])));
1078 else if (equal(c, "ENUM")) puts(enumerate(d, atoi(w[2])));
1079 else if (equal(c, "EMPTY")) {
1080 const char *text = smallest(d);
1081 if (text) printf("NOT EMPTY %s\n", visible(text));
1082 else puts("EMPTY");
1083 } else if (equal(c, "FINITE")) {
1084 long long total = 0;
1085 Language language = classify(d, &total);
1086 if (language == INFINITE_LANGUAGE) puts("INFINITE");
1087 else printf("FINITE %lld\n", total);
1088 } else if (equal(c, "INFO")) {
1089 bool deterministic = !epsilon_count(a), complete = true;
1090 for (int p = 0; p < a->count; p++) for (int c = 0; c < a->symbols; c++) {
1091 int n = set_count(a->transitions[p][c]);
1092 deterministic = deterministic && n <= 1;
1093 complete = complete && n > 0;
1094 }
1095 Language language = classify(d, NULL);
1096 puts(summary(a, w[1]));
1097 printf(" epsilon transitions: %d\n", epsilon_count(a));
1098 printf(" reachable: %s\n", list_text(a, graph_reach(a, false)));
1099 printf(" live: %s\n", list_text(a, graph_reach(a, true)));
1100 printf(" deterministic: %s\n complete: %s\n", deterministic ? "yes" : "no", complete ? "yes" : "no");
1101 printf(" language: %s\n", language == EMPTY_LANGUAGE ? "empty" : language == FINITE_LANGUAGE ? "finite" : "infinite");
1102 printf(" minimal DFA states: %d\n", minimize(d, NULL)->count);
1103 }
1104}
