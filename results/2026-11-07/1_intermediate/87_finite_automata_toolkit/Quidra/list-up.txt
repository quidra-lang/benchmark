bool includes(string[] items, string text)
    for item := items[]
        if item == text
            return true
    return false

nat locate(string[] items, string text)
    for: i < #items
        if items[i] == text
            return i
    return #items

nat[] ordered_set(nat[] items)
    nat[] result := []
    for item := items.sorted()[]
        if #result == 0 or result[-1] != item
            result = result.append(item)
    return result

string index_key(nat[] items)
    return [for item := items[] then string(item)].join(",")
class Edge(nat source, nat column, nat target)
class Automaton(string kind, string[] states, string alphabet, nat start, nat[] accepting, nat[][][] cells)
    bool accepts(nat[] indices)
        for i := indices[]
            for member := accepting[]
                if i == member
                    return true
        return false

    nat[] closure(nat[] seeds)
        seen := [for i := (nat(0):<#states)[] then false]
        nat[] pending := []
        for item := seeds[]
            if not seen[item]
                seen[item] = true
                pending = pending.append(item)
        nat cursor := 0
        while cursor < #pending
            current := pending[cursor]
            cursor += 1
            for target := cells[current][#alphabet][]
                if not seen[target]
                    seen[target] = true
                    pending = pending.append(target)
        return [for i := (nat(0):<#states)[] if seen[i] then i]

    nat[] destination(nat[] source_set, nat symbol_index)
        nat[] targets := []
        for source := source_set[]
            targets = targets.concat(cells[source][symbol_index])
        return closure(ordered_set(targets))

    nat delta(nat source, nat symbol_index)
        return cells[source][symbol_index][0]
Automaton make_automaton(string kind, string[] names, string alphabet, nat start, nat[] final_states, Edge[] edges)
    nat[][][] cells := [for i := (nat(0):<#names)[] then [for j := (nat(0):<(nat(#alphabet) + 1))[] then []]]
    for edge := edges[]
        cells[edge.source][edge.column] = cells[edge.source][edge.column].append(edge.target)
    for: i < #names
        for: j <= #alphabet
            cells[i][j] = ordered_set(cells[i][j])
    return Automaton(kind, names, alphabet, start, ordered_set(final_states), cells)
class Conversion(Automaton machine, string[] details)
Conversion determinize(Automaton machine)
    subsets := [machine.closure([machine.start])]
    map<string, nat> indices
    .indices.set(index_key(subsets[0]), 0)
    Edge[] edges := []
    nat[] final_states := []
    nat cursor := 0
    while cursor < #subsets
        current := subsets[cursor]
        if machine.accepts(current)
            final_states = final_states.append(cursor)
        for: c < #machine.alphabet
            target := machine.destination(current, c)
            key := index_key(target)
            target_index := #subsets
            match indices.get(key)
                nat found
                    target_index = found
                none
                    .indices.set(key, target_index)
                    subsets = subsets.append(target)
            edges = edges.append(Edge(cursor, c, target_index))
        cursor += 1
    names := series("D", #subsets)
    return Conversion(make_automaton("DFA", names, machine.alphabet, 0, final_states, edges), details)
nat[] next_counts(Automaton machine, nat[] counts)
    nat[] result := [for i := (nat(0):<#machine.states)[] then 0]
    for: source < #machine.states
        for: c < #machine.alphabet
            result[machine.delta(source, c)] += counts[source]
    return result

nat accepted_count(Automaton machine, nat[] counts)
    nat result := 0
    for item := machine.accepting[]
        result += counts[item]
