20class Automaton {
21 final String kind;
22 final List<String> names, alphabet;
23 final int start;
24 final Set<int> accepting;
25 final List<Map<String, Set<int>>> rows;
26 List<Set<int>>? closures;
27 Determinization? determinized;
28 Automaton(this.kind, List<String> names, Iterable<String> alphabet, this.start, [Iterable<int> accepting = const []])
29 : names = List.of(names), alphabet = alphabet.toList()..sort(), accepting = accepting.toSet(),
30 rows = List.generate(names.length, (_) => <String, Set<int>>{});
31 void add(int from, String symbol, int to) => rows[from].putIfAbsent(symbol, () => <int>{}).add(to);
32 Set<int> targets(int from, String symbol) => rows[from][symbol] ?? <int>{};
33 int delta(int from, String symbol) => targets(from, symbol).first;
34 List<Triple> triples() {
35 final result = <Triple>[];
36 for (int p = 0; p < names.length; p++) {
37 for (final symbol in [...alphabet, 'e']) {
38 for (final q in ordered(targets(p, symbol))) { result.add((p, symbol, q)); }
39 }
40 }
41 return result;
42 }
43 String summary(String name) {
44 int transitions = 0;
45 for (final row in rows) { for (final values in row.values) { transitions += values.length; } }
46 return '$kind $name: alphabet=${alphabet.join()} states=${names.length} accepting=${accepting.length} transitions=$transitions';
47 }
48 String notation(Iterable<int> states) => '{${ordered(states).map((i) => names[i]).join(',')}}';
49 String list(Iterable<int> states) {
50 final text = ordered(states).map((i) => names[i]).join(',');
51 return text.isEmpty ? 'none' : text;
52 }
53 Set<int> closure(Iterable<int> states) {
54 closures ??= List.generate(names.length, (start) {
55 final seen = <int>{start}, queue = <int>[start];
56 for (int i = 0; i < queue.length; i++) {
57 for (final q in targets(queue[i], 'e')) { if (seen.add(q)) queue.add(q); }
58 }
59 return seen;
60 });
61 final result = <int>{};
62 for (final p in states) { result.addAll(closures![p]); }
63 return result;
64 }
65 Set<int> destination(Iterable<int> states, String symbol) {
66 final moved = <int>{};
67 for (final p in states) { moved.addAll(targets(p, symbol)); }
68 return closure(moved);
69 }
70 bool accepts(Iterable<int> states) => states.any(accepting.contains);
71}

91Determinization determinize(Automaton source) {
92 final cached = source.determinized;
93 if (cached != null) return cached;
94 final sets = <Set<int>>[source.closure([source.start])], transitions = <Triple>[];
95 final indices = <String, int>{subsetKey(sets[0]): 0};
96 for (int i = 0; i < sets.length; i++) {
97 for (final symbol in source.alphabet) {
98 final next = source.destination(sets[i], symbol), key = subsetKey(next);
99 if (!indices.containsKey(key)) { indices[key] = sets.length; sets.add(next); }
100 transitions.add((i, symbol, indices[key]!));
101 }
102 }
103 final accepting = <int>[];
104 for (int i = 0; i < sets.length; i++) { if (source.accepts(sets[i])) accepting.add(i); }
105 final result = Automaton('DFA', sequence('D', sets.length), source.alphabet, 0, accepting);
106 for (final (p, symbol, q) in transitions) { result.add(p, symbol, q); }
107 return source.determinized = Determinization(result, sets);
108}

110Set<int> graphReach(Automaton source, [bool reverse = false]) {
111 final edges = List.generate(source.names.length, (_) => <int>{});
112 for (final (p, _, q) in source.triples()) { edges[reverse ? q : p].add(reverse ? p : q); }
113 final seen = reverse ? Set<int>.of(source.accepting) : <int>{source.start}, queue = <int>[];
114 queue.addAll(seen);
115 for (int i = 0; i < queue.length; i++) { for (final q in edges[queue[i]]) { if (seen.add(q)) queue.add(q); } }
116 return seen;
117}

119Minimization minimize(Automaton source) {
120 final reachable = graphReach(source), states = ordered(reachable);
121 var blocks = <List<int>>[states.where(source.accepting.contains).toList(), states.where((p) => !source.accepting.contains(p)).toList()]
122 .where((block) => block.isNotEmpty).toList()..sort((a, b) => a.first.compareTo(b.first));
123 final rounds = <List<List<int>>>[blocks];
124 while (true) {
125 final membership = <int, int>{};
126 for (int i = 0; i < blocks.length; i++) { for (final p in blocks[i]) { membership[p] = i; } }
127 final groups = <String, List<int>>{};
128 for (final p in states) {
129 final key = [membership[p]!, ...source.alphabet.map((symbol) => membership[source.delta(p, symbol)]!)].join(',');
130 groups.putIfAbsent(key, () => <int>[]).add(p);
131 }
132 final next = groups.values.toList();
133 if (next.length == blocks.length) break;
134 blocks = next;
135 rounds.add(blocks);
136 }
137 final membership = <int, int>{};
138 for (int i = 0; i < blocks.length; i++) { for (final p in blocks[i]) { membership[p] = i; } }
139 final order = <int>[membership[source.start]!], transitions = <Triple>[];
140 final indices = <int, int>{order[0]: 0};
141 for (int i = 0; i < order.length; i++) {
142 for (final symbol in source.alphabet) {
143 final next = membership[source.delta(blocks[order[i]][0], symbol)]!;
144 if (!indices.containsKey(next)) { indices[next] = order.length; order.add(next); }
145 transitions.add((i, symbol, indices[next]!));
146 }
147 }
148 final accepting = <int>[];
149 for (int i = 0; i < order.length; i++) { if (source.accepting.contains(blocks[order[i]][0])) accepting.add(i); }
150 final result = Automaton('DFA', sequence('M', order.length), source.alphabet, 0, accepting);
151 for (final (p, symbol, q) in transitions) { result.add(p, symbol, q); }
152 return Minimization(result, rounds, order.map((i) => blocks[i]).toList(),
153 List.generate(source.names.length, (i) => i).where((i) => !reachable.contains(i)).toList());
154}

156({String kind, int count}) language(Automaton source, [bool countWords = false]) {
157 final dfa = determinize(source).automaton, live = graphReach(dfa, true);
158 if (!live.contains(dfa.start)) return (kind: 'empty', count: 0);
159 final color = List.filled(dfa.names.length, 0), postorder = <int>[];
160 bool infinite = false;
161 void visit(int p) {
162 if (!live.contains(p)) return;
163 if (color[p] == 1) { infinite = true; return; }
164 if (color[p] == 2) return;
165 color[p] = 1;
166 for (final symbol in dfa.alphabet) { visit(dfa.delta(p, symbol)); }
167 color[p] = 2;
168 postorder.add(p);
169 }
170 visit(dfa.start);
171 if (infinite) return (kind: 'infinite', count: 0);
172 if (!countWords) return (kind: 'finite', count: 0);
173 final totals = List.filled(dfa.names.length, 0);
174 for (final p in postorder) {
175 totals[p] = dfa.accepting.contains(p) ? 1 : 0;
176 for (final symbol in dfa.alphabet) { totals[p] += totals[dfa.delta(p, symbol)]; }
177 }
178 return (kind: 'finite', count: totals[dfa.start]);
179}

181String? smallest(Automaton source) {
182 final dfa = determinize(source).automaton, queue = <(int, String)>[(dfa.start, '')], seen = <int>{dfa.start};
183 for (int i = 0; i < queue.length; i++) {
184 final (state, word) = queue[i];
185 if (dfa.accepting.contains(state)) return word;
186 for (final symbol in dfa.alphabet) {
187 final next = dfa.delta(state, symbol);
188 if (seen.add(next)) queue.add((next, word + symbol));
189 }
190 }
191 return null;
192}

209class RegexParser {
210 final String text;
211 int position = 0;
212 RegexParser(this.text);
213 String get peek => position < text.length ? text[position] : '';
214 bool atomStart() => peek.isNotEmpty && RegExp(r'[a-z0-9(]').hasMatch(peek);
215 Never error() => fail('syntax error at ${position + 1}');
216 RegexTree atom() {
217 final symbol = peek;
218 if (symbol.isNotEmpty && RegExp(r'[a-z0-9]').hasMatch(symbol)) { position++; return RegexSymbol(symbol); }
219 if (symbol == '(') {
220 position++;
221 final result = alternation();
222 if (peek != ')') error();
223 position++;
224 return result;
225 }
226 error();
227 }
228 RegexTree factor() {
229 var result = atom();
230 while (peek.isNotEmpty && '*+?'.contains(peek)) {
231 final op = peek;
232 position++;
233 result = RegexPostfix(op, result);
234 }
235 return result;
236 }
237 RegexTree concatenation() {
238 final children = <RegexTree>[factor()];
239 while (atomStart()) { children.add(factor()); }
240 return children.length == 1 ? children[0] : RegexMany('concat', children);
241 }
242 RegexTree alternation() {
243 final children = <RegexTree>[concatenation()];
244 while (peek == '|') { position++; children.add(concatenation()); }
245 return children.length == 1 ? children[0] : RegexMany('alternate', children);
246 }
247 RegexTree parse() {
248 final result = alternation();
249 if (position != text.length) error();
250 return result;
251 }
252}

254Automaton fromRegex(Iterable<String> alphabet, RegexTree tree) {
255 int count = 0;
256 final transitions = <Triple>[];
257 Pair build(RegexTree node) {
258 if (node is RegexSymbol) {
259 final start = count++, finish = count++;
260 transitions.add((start, node.symbol, finish));
261 return (start, finish);
262 }
263 if (node is RegexMany && node.kind == 'concat') {
264 final parts = node.children.map(build).toList();
265 for (int i = 1; i < parts.length; i++) { transitions.add((parts[i - 1].$2, 'e', parts[i].$1)); }
266 return (parts.first.$1, parts.last.$2);
267 }
268 final start = count++;
269 if (node is RegexMany) {
270 final parts = node.children.map(build).toList(), finish = count++;
271 for (final (s, f) in parts) { transitions.addAll([(start, 'e', s), (f, 'e', finish)]); }
272 return (start, finish);
273 }
274 final postfix = node as RegexPostfix, part = build(postfix.child), finish = count++;
275 transitions.addAll([(start, 'e', part.$1), (part.$2, 'e', finish)]);
276 if (postfix.op != '?') transitions.add((part.$2, 'e', part.$1));
277 if (postfix.op != '+') transitions.add((start, 'e', finish));
278 return (start, finish);
279 }
280 final part = build(tree), result = Automaton('NFA', sequence('N', count), alphabet, part.$1, [part.$2]);
281 for (final (p, symbol, q) in transitions) { result.add(p, symbol, q); }
282 return result;
283}

285Product product(Automaton left, Automaton right, String operation) {
286 final pairs = <Pair>[(left.start, right.start)], transitions = <Triple>[];
287 final indices = <Pair, int>{pairs[0]: 0};
288 for (int i = 0; i < pairs.length; i++) {
289 for (final symbol in left.alphabet) {
290 final next = (left.delta(pairs[i].$1, symbol), right.delta(pairs[i].$2, symbol));
291 if (!indices.containsKey(next)) { indices[next] = pairs.length; pairs.add(next); }
292 transitions.add((i, symbol, indices[next]!));
293 }
294 }
295 final accepting = <int>[];
296 for (int i = 0; i < pairs.length; i++) {
297 final a = left.accepting.contains(pairs[i].$1), b = right.accepting.contains(pairs[i].$2);
298 if (operation == 'UNION' ? a || b : operation == 'INTERSECT' ? a && b : a && !b) accepting.add(i);
299 }
300 final result = Automaton('DFA', sequence('P', pairs.length), left.alphabet, 0, accepting);
301 for (final (p, symbol, q) in transitions) { result.add(p, symbol, q); }
302 return Product(result, pairs);
303}

305({String word, bool left})? distinguishing(Automaton left, Automaton right) {
306 final a = determinize(left).automaton, b = determinize(right).automaton;
307 final queue = <(int, int, String)>[(a.start, b.start, '')], seen = <Pair>{(a.start, b.start)};
308 for (int i = 0; i < queue.length; i++) {
309 final (p, q, word) = queue[i];
310 if (a.accepting.contains(p) != b.accepting.contains(q)) return (word: word, left: a.accepting.contains(p));
311 for (final symbol in a.alphabet) {
312 final next = (a.delta(p, symbol), b.delta(q, symbol));
313 if (seen.add(next)) queue.add((next.$1, next.$2, word + symbol));
314 }
315 }
316 return null;
317}

319Automaton combine(Automaton source, Automaton? other, String operation) {
320 final count = source.names.length, extra = operation == 'CONCAT' ? other!.names.length : 1;
321 final offset = operation == 'CONCAT' ? 0 : 1, start = operation == 'CONCAT' ? source.start : 0;
322 final accepting = operation == 'REVERSE' ? [source.start + 1] : operation == 'STAR'
323 ? [0, ...source.accepting.map((p) => p + 1)] : other!.accepting.map((p) => p + count);
324 final result = Automaton('NFA', sequence('N', count + extra), source.alphabet, start, accepting);
325 for (final (p, symbol, q) in source.triples()) {
326 if (operation == 'REVERSE') { result.add(q + 1, symbol, p + 1); }
327 else { result.add(p + offset, symbol, q + offset); }
328 }
329 if (operation == 'REVERSE') {
330 for (final p in source.accepting) { result.add(0, 'e', p + 1); }
331 } else if (operation == 'STAR') {
332 result.add(0, 'e', source.start + 1);
333 for (final p in source.accepting) { result.add(p + 1, 'e', source.start + 1); }
334 } else {
335 final right = other!;
336 for (final (p, symbol, q) in right.triples()) { result.add(p + count, symbol, q + count); }
337 for (final p in source.accepting) { result.add(p, 'e', right.start + count); }
338 }
339 return result;
340}

349class Toolkit {
350 final table = <String, Automaton>{};
351 List<String> register(String name, Automaton source, [List<String> details = const []]) {
352 table[name] = source;
353 return [source.summary(name), ...details];
354 }
355 List<String> define(String kind, List<String> args) {
356 if (args.length < 5) fail('wrong number of arguments');
357 final name = args[0], names = args[1].split(','), alphabet = args[2], startName = args[3], acceptingText = args[4];
358 if (table.containsKey(name)) fail('$name already exists');
359 final index = <String, int>{};
360 for (int i = 0; i < names.length; i++) {
361 if (index.containsKey(names[i])) fail('duplicate state ${names[i]}');
362 index[names[i]] = i;
363 }
364 if (!index.containsKey(startName)) fail('unknown state $startName');
365 final accepting = <int>{};
366 if (acceptingText != '-') {
367 for (final state in acceptingText.split(',')) {
368 if (!index.containsKey(state)) fail('unknown state $state');
369 if (!accepting.add(index[state]!)) fail('duplicate state $state');
370 }
371 }
372 final result = Automaton(kind, names, alphabet.split(''), index[startName]!, accepting), seen = <String>{};
373 for (final word in args.skip(5)) {
374 final parts = word.split(':'), p = parts[0], symbol = parts[1], q = parts[2];
375 if (!index.containsKey(p)) fail('unknown state $p');
376 if (symbol == 'e' && kind == 'DFA') fail('epsilon transition in DFA');
377 if (symbol != 'e' && !result.alphabet.contains(symbol)) fail('unknown symbol $symbol');
378 if (!index.containsKey(q)) fail('unknown state $q');
379 if (seen.contains(word)) fail('duplicate transition $word');
380 if (kind == 'DFA' && result.targets(index[p]!, symbol).isNotEmpty) fail('conflicting transition $word');
381 seen.add(word);
382 result.add(index[p]!, symbol, index[q]!);
383 }
384 if (kind == 'DFA') {
385 for (int p = 0; p < names.length; p++) {
386 for (final symbol in result.alphabet) { if (result.targets(p, symbol).isEmpty) fail('missing transition ${names[p]}:$symbol'); }
387 }
388 }
389 return register(name, result);
390 }
391 List<String> command(String line) {
392 final words = line.split(' '), command = words[0], args = words.sublist(1);
393 if (command == 'DFA' || command == 'NFA') return define(command, args);
394 if (command == 'REGEX') {
395 if (args.length != 3) fail('wrong number of arguments');
396 final name = args[0], alphabet = args[1], text = args[2];
397 if (table.containsKey(name)) fail('$name already exists');
398 for (final symbol in text.split('')) {
399 if (RegExp(r'[a-z0-9]').hasMatch(symbol) && symbol != 'e' && !alphabet.contains(symbol)) fail('unknown symbol $symbol');
400 }
401 return register(name, fromRegex(alphabet.split(''), RegexParser(text).parse()));
402 }
403 if (!roles.containsKey(command)) fail('unknown command $command');
404 final kinds = roles[command]!;
405 if (args.length != kinds.length) fail('wrong number of arguments');
406 for (int i = 0; i < kinds.length; i++) {
407 if ('RD'.contains(kinds[i]) && !table.containsKey(args[i])) fail('unknown automaton ${args[i]}');
408 }
409 final newIndex = kinds.indexOf('N');
410 if (newIndex >= 0 && table.containsKey(args[newIndex])) fail('${args[newIndex]} already exists');
411 for (int i = 0; i < kinds.length; i++) {
412 if (kinds[i] == 'D' && table[args[i]]!.kind != 'DFA') fail('${args[i]} is not a DFA');
413 }
414 if (command == 'LIST') return table.isEmpty ? ['(none)'] : table.entries.map((item) => item.value.summary(item.key)).toList();
415 final source = table[args[0]]!;
416 final other = kinds.length > 1 && 'RD'.contains(kinds[1]) ? table[args[1]]! : null;
417 if (other != null && source.alphabet.join() != other.alphabet.join()) fail('alphabets differ');
418 switch (command) {
419 case 'DELETE': table.remove(args[0]); return ['deleted ${args[0]}'];
420 case 'ACCEPT':
421 final word = args[1] == 'e' ? '' : args[1];
422 for (final symbol in word.split('')) { if (!source.alphabet.contains(symbol)) fail('unknown symbol $symbol'); }
423 var states = source.closure([source.start]);
424 String notation(Set<int> set) => source.kind == 'DFA' ? source.names[set.first] : source.notation(set);
425 var path = notation(states);
426 for (final symbol in word.split('')) { states = source.destination(states, symbol); path += ' -$symbol-> ${notation(states)}'; }
427 return ['${source.accepts(states) ? 'ACCEPT' : 'REJECT'} $path'];
428 case 'CLOSURE':
429 final states = <int>[];
430 for (final name in args[1].split(',')) {
431 final i = source.names.indexOf(name);
432 if (i < 0) fail('unknown state $name');
433 states.add(i);
434 }
435 return [source.notation(source.closure(states))];
436 case 'TODFA':
437 final result = determinize(source), details = <String>[];
438 for (int i = 0; i < result.sets.length; i++) { details.add(' D$i = ${source.notation(result.sets[i])}'); }
439 return register(args[1], result.automaton, details);
440 case 'EPSFREE':
441 final result = Automaton('NFA', source.names, source.alphabet, source.start);
442 for (int p = 0; p < source.names.length; p++) {
443 final set = source.closure([p]);
444 if (source.accepts(set)) result.accepting.add(p);
445 for (final symbol in source.alphabet) { for (final q in source.destination(set, symbol)) { result.add(p, symbol, q); } }
446 }
447 return register(args[1], result);
448 case 'MINIMIZE':
449 final result = minimize(source), details = [' unreachable: ${source.list(result.unreachable)}'];
450 for (int i = 0; i < result.rounds.length; i++) { details.add(' round $i: ${result.rounds[i].map(source.notation).join(' ')}'); }
451 for (int i = 0; i < result.blocks.length; i++) { details.add(' M$i = ${source.notation(result.blocks[i])}'); }
452 return register(args[1], result.automaton, details);
453 case 'EQUIV':
454 final result = distinguishing(source, other!);
455 return [result == null ? 'EQUIVALENT' : 'NOT EQUIVALENT: ${result.word.isEmpty ? 'e' : result.word} is accepted only by ${args[result.left ? 0 : 1]}'];
456 case 'UNION': case 'INTERSECT': case 'DIFF':
457 final right = other!, result = product(source, right, command), details = <String>[];
458 for (int i = 0; i < result.pairs.length; i++) {
459 final (p, q) = result.pairs[i];
460 details.add(' P$i = (${source.names[p]},${right.names[q]})');
461 }
462 return register(args[2], result.automaton, details);
463 case 'COMPLEMENT':
464 final result = Automaton('DFA', source.names, source.alphabet, source.start,
465 List.generate(source.names.length, (i) => i).where((i) => !source.accepting.contains(i)));
466 for (final (p, symbol, q) in source.triples()) { result.add(p, symbol, q); }
467 return register(args[1], result);
468 case 'REVERSE': case 'STAR': case 'CONCAT': return register(args[newIndex], combine(source, other, command));
469 case 'COUNT':
470 final dfa = determinize(source).automaton, length = int.parse(args[1]);
471 var counts = List.filled(dfa.names.length, 0);
472 counts[dfa.start] = 1;
473 for (int i = 0; i < length; i++) {
474 final next = List.filled(dfa.names.length, 0);
475 for (int p = 0; p < counts.length; p++) { for (final symbol in dfa.alphabet) { next[dfa.delta(p, symbol)] += counts[p]; } }
476 counts = next;
477 }
478 return ['${dfa.accepting.fold<int>(0, (sum, p) => sum + counts[p])}'];
479 case 'ENUM':
480 final dfa = determinize(source).automaton, accepted = <String>[], maximum = int.parse(args[1]);
481 var level = <(int, String)>[(dfa.start, '')];
482 int total = 0;
483 for (int length = 0; length <= maximum; length++) {
484 for (final (state, word) in level) {
485 if (dfa.accepting.contains(state)) { total++; if (accepted.length < 50) accepted.add(word.isEmpty ? 'e' : word); }
486 }
487 if (length < maximum) {
488 level = [for (final (state, word) in level) for (final symbol in dfa.alphabet) (dfa.delta(state, symbol), word + symbol)];
489 }
490 }
491 return ['$total${accepted.isEmpty ? '' : ' ${accepted.join(' ')}'}${total > 50 ? ' ...' : ''}'];
492 case 'EMPTY':
493 final word = smallest(source);
494 return [word == null ? 'EMPTY' : 'NOT EMPTY ${word.isEmpty ? 'e' : word}'];
495 case 'FINITE':
496 final result = language(source, true);
497 return [result.kind == 'infinite' ? 'INFINITE' : 'FINITE ${result.count}'];
498 case 'SYNC':
499 final initial = (1 << source.names.length) - 1, seen = <int>{initial}, queue = <(int, String)>[(initial, '')];
500 for (int i = 0; i < queue.length; i++) {
501 final (set, word) = queue[i];
502 if ((set & (set - 1)) == 0) return ['SYNCHRONIZING ${word.isEmpty ? 'e' : word} -> ${source.names[set.bitLength - 1]}'];
503 for (final symbol in source.alphabet) {
504 int next = 0;
505 for (int p = 0; p < source.names.length; p++) { if ((set & (1 << p)) != 0) next |= 1 << source.delta(p, symbol); }
506 if (seen.add(next)) queue.add((next, word + symbol));
507 }
508 }
509 return ['NOT SYNCHRONIZING'];
510 case 'INFO':
511 int epsilon = 0;
512 for (int p = 0; p < source.names.length; p++) { epsilon += source.targets(p, 'e').length; }
513 bool deterministic = epsilon == 0, complete = true;
514 for (int p = 0; p < source.names.length; p++) {
515 for (final symbol in source.alphabet) {
516 final size = source.targets(p, symbol).length;
517 if (size > 1) deterministic = false;
518 if (size == 0) complete = false;
519 }
520 }
521 return [source.summary(args[0]), ' epsilon transitions: $epsilon', ' reachable: ${source.list(graphReach(source))}',
522 ' live: ${source.list(graphReach(source, true))}', ' deterministic: ${deterministic ? 'yes' : 'no'}',
523 ' complete: ${complete ? 'yes' : 'no'}', ' language: ${language(source).kind}',
524 ' minimal DFA states: ${minimize(determinize(source).automaton).automaton.names.length}'];
525 case 'TABLE':
526 final columns = List<String>.of(source.alphabet);
527 if (source.kind == 'NFA' && source.rows.asMap().keys.any((p) => source.targets(p, 'e').isNotEmpty)) columns.add('e');
528 final rows = <List<String>>[['', ...columns]];
529 for (int p = 0; p < source.names.length; p++) {
530 rows.add(['${p == source.start ? '->' : ' '}${source.accepting.contains(p) ? '*' : ' '}${source.names[p]}',
531 ...columns.map((symbol) => source.kind == 'DFA' ? source.names[source.delta(p, symbol)]
532 : source.targets(p, symbol).isEmpty ? '-' : source.notation(source.targets(p, symbol)))]);
533 }
534 final widths = List.filled(rows[0].length, 0);
535 for (final row in rows) { for (int i = 0; i < row.length; i++) { if (row[i].length > widths[i]) widths[i] = row[i].length; } }
536 return rows.map((row) => stripRight([for (int i = 0; i < row.length; i++) row[i].padRight(widths[i])].join(' '))).toList();
537 case 'SHOW':
538 final accepting = ordered(source.accepting).map((p) => source.names[p]).join(',');
539 return [[source.kind, args[0], source.names.join(','), source.alphabet.join(), source.names[source.start], accepting.isEmpty ? '-' : accepting,
540 ...source.triples().map((t) => '${source.names[t.$1]}:${t.$2}:${source.names[t.$3]}')].join(' ')];
541 }
542 throw StateError('unhandled command');
543 }
544}

547 final count = int.parse(stdin.readLineSync()!), toolkit = Toolkit(), output = <String>[];
548 for (int i = 0; i < count; i++) {
549 final line = stdin.readLineSync()!;
550 if (line.startsWith('#')) continue;
551 try { output.addAll(toolkit.command(line)); }
552 on CommandError catch (error) { output.add('error: ${error.reason}'); }
553 }
554 if (output.isNotEmpty) stdout.write('${output.join('\n')}\n');
555}
