60def reachable_from(sources, successors):
61 seen = set(sources)
62 pending = list(seen)
63 while pending:
64 for target in successors(pending.pop()):
65 if target not in seen:
66 seen.add(target)
67 pending.append(target)
68 return seen

71def explore(origin, alphabet, step):
72 nodes = [origin]
73 words = [""]
74 index_of = {origin: 0}
75 edges = []
76 current = 0
77 while current < len(nodes):
78 for symbol in alphabet:
79 target = step(nodes[current], symbol)
80 if target not in index_of:
81 index_of[target] = len(nodes)
82 nodes.append(target)
83 words.append(words[current] + symbol)
84 edges.append((current, symbol, index_of[target]))
85 current += 1
86 return nodes, words, edges

89class Automaton:
90 def __init__(self, kind, states, alphabet, start, accepting, transitions):
91 self.kind = kind
92 self.states = states
93 self.index_of = {state: index for index, state in enumerate(states)}
94 self.alphabet = "".join(sorted(alphabet))
95 self.start = start
96 self.accepting = frozenset(accepting)
97 self.transitions = frozenset(transitions)
98 self.targets = {}
99 for source, symbol, target in self.transitions:
100 self.targets.setdefault((source, symbol), set()).add(target)
101
102 def summary(self, name):
103 return (
104 f"{self.kind} {name}: alphabet={self.alphabet} states={len(self.states)} "
105 f"accepting={len(self.accepting)} transitions={len(self.transitions)}"
106 )
107
108 def join_names(self, states):
109 return ",".join(self.states[state] for state in sorted(states))
110
111 def format_set(self, states):
112 return "{" + self.join_names(states) + "}"
113
114 def format_list(self, states):
115 return self.join_names(states) or "none"
116
117 def successors(self, state, symbol):
118 return self.targets.get((state, symbol), set())
119
120 def delta(self, state, symbol):
121 (target,) = self.targets[state, symbol]
122 return target
123
124 def closure(self, states):
125 return frozenset(reachable_from(states, lambda state: self.successors(state, EPSILON)))
126
127 def initial_set(self):
128 return self.closure([self.start])
129
130 def destination(self, states, symbol):
131 return self.closure(
132 {target for state in states for target in self.successors(state, symbol)}
133 )
134
135 def contains_accepting(self, states):
136 return not self.accepting.isdisjoint(states)
137
138 def epsilon_count(self):
139 return sum(1 for _, symbol, _ in self.transitions if symbol == EPSILON)
140
141 def neighbors(self, state):
142 return [
143 target
144 for symbol in self.alphabet + EPSILON
145 for target in self.successors(state, symbol)
146 ]
147
148 def reachable_states(self):
149 return reachable_from([self.start], self.neighbors)
150
151 def live_states(self):
152 predecessors = defaultdict(set)
153 for source, _, target in self.transitions:
154 predecessors[target].add(source)
155 return reachable_from(self.accepting, lambda state: predecessors[state])

158class RegexParser:
159 def __init__(self, expression):
160 self.expression = expression
161 self.position = 0
162
163 def peek(self):
164 if self.position < len(self.expression):
165 return self.expression[self.position]
166 return ""
167
168 def error(self):
169 return CommandError(f"syntax error at {self.position + 1}")
170
171 def parse(self):
172 node = self.alternation()
173 if self.peek():
174 raise self.error()
175 return node
176
177 def alternation(self):
178 terms = [self.concatenation()]
179 while self.peek() == "|":
180 self.position += 1
181 terms.append(self.concatenation())
182 return terms[0] if len(terms) == 1 else ("alternation", terms)
183
184 def concatenation(self):
185 factors = [self.factor()]
186 while is_symbol_char(self.peek()) or self.peek() == "(":
187 factors.append(self.factor())
188 return factors[0] if len(factors) == 1 else ("concatenation", factors)
189
190 def factor(self):
191 node = self.atom()
192 while self.peek() in POSTFIX_OPERATORS:
193 node = (self.peek(), node)
194 self.position += 1
195 return node
196
197 def atom(self):
198 char = self.peek()
199 if is_symbol_char(char):
200 self.position += 1
201 return ("symbol", char)
202 if char != "(":
203 raise self.error()
204 self.position += 1
205 node = self.alternation()
206 if self.peek() != ")":
207 raise self.error()
208 self.position += 1
209 return node

212class ThompsonBuilder:
213 def __init__(self):
214 self.state_count = 0
215 self.transitions = set()
216
217 def new_state(self):
218 self.state_count += 1
219 return self.state_count - 1
220
221 def link(self, source, target):
222 self.transitions.add((source, EPSILON, target))
223
224 def build(self, node):
225 kind, content = node
226 if kind == "concatenation":
227 parts = [self.build(factor) for factor in content]
228 for (_, previous_exit), (next_entry, _) in zip(parts, parts[1:]):
229 self.link(previous_exit, next_entry)
230 return parts[0][0], parts[-1][1]
231 entry = self.new_state()
232 if kind == "symbol":
233 exit_state = self.new_state()
234 self.transitions.add((entry, content, exit_state))
235 return entry, exit_state
236 if kind == "alternation":
237 parts = [self.build(term) for term in content]
238 exit_state = self.new_state()
239 for part_entry, part_exit in parts:
240 self.link(entry, part_entry)
241 self.link(part_exit, exit_state)
242 return entry, exit_state
243 inner_entry, inner_exit = self.build(content)
244 exit_state = self.new_state()
245 self.link(entry, inner_entry)
246 self.link(inner_exit, exit_state)
247 if kind != "?":
248 self.link(inner_exit, inner_entry)
249 if kind != "+":
250 self.link(entry, exit_state)
251 return entry, exit_state

254def determinize(automaton):
255 subsets, _, edges = explore(automaton.initial_set(), automaton.alphabet, automaton.destination)
256 accepting = {
257 index for index, subset in enumerate(subsets) if automaton.contains_accepting(subset)
258 }
259 dfa = Automaton("DFA", series("D", len(subsets)), automaton.alphabet, 0, accepting, edges)
260 return dfa, subsets

263def remove_epsilon(automaton):
264 accepting = set()
265 transitions = set()
266 for state in range(len(automaton.states)):
267 reach = automaton.closure([state])
268 if automaton.contains_accepting(reach):
269 accepting.add(state)
270 for symbol in automaton.alphabet:
271 transitions.update(
272 (state, symbol, target) for target in automaton.destination(reach, symbol)
273 )
274 return Automaton(
275 "NFA", automaton.states, automaton.alphabet, automaton.start, accepting, transitions

291def minimize(dfa):
292 reachable = sorted(dfa.reachable_states())
293 unreachable = set(range(len(dfa.states))) - set(reachable)
294 details = [f" unreachable: {dfa.format_list(unreachable)}"]
295 blocks = number_blocks({state: state in dfa.accepting for state in reachable})
296 groups = group_blocks(blocks)
297 round_number = 0
298 while True:
299 partition = " ".join(dfa.format_set(group) for group in groups)
300 details.append(f" round {round_number}: {partition}")
301 refined = number_blocks(
302 {
303 state: (
304 blocks[state],
305 tuple(blocks[dfa.delta(state, symbol)] for symbol in dfa.alphabet),
306 )
307 for state in reachable
308 }
309 )
310 refined_groups = group_blocks(refined)
311 if len(refined_groups) == len(groups):
312 break
313 blocks, groups = refined, refined_groups
314 round_number += 1
315 named, _, edges = explore(
316 blocks[dfa.start],
317 dfa.alphabet,
318 lambda block, symbol: blocks[dfa.delta(groups[block][0], symbol)],
319 )
320 accepting = {index for index, block in enumerate(named) if groups[block][0] in dfa.accepting}
321 details += [
322 f" M{index} = {dfa.format_set(groups[block])}" for index, block in enumerate(named)
323 ]
324 minimal = Automaton("DFA", series("M", len(named)), dfa.alphabet, 0, accepting, edges)
325 return minimal, details

328def product_pairs(left, right):
329 return explore(
330 (left.start, right.start),
331 left.alphabet,
332 lambda pair, symbol: (left.delta(pair[0], symbol), right.delta(pair[1], symbol)),

336def product_dfa(left, right, accepts):
337 pairs, _, edges = product_pairs(left, right)
338 accepting = {
339 index
340 for index, (first, second) in enumerate(pairs)
341 if accepts(first in left.accepting, second in right.accepting)
342 }
343 dfa = Automaton("DFA", series("P", len(pairs)), left.alphabet, 0, accepting, edges)
344 return dfa, pairs

347def find_difference(first, second):
348 left, _ = determinize(first)
349 right, _ = determinize(second)
350 pairs, words, _ = product_pairs(left, right)
351 for (p, q), word in zip(pairs, words):
352 if (p in left.accepting) != (q in right.accepting):
353 return word, p in left.accepting
354 return None

362def reverse(automaton):
363 transitions = {
364 (target + 1, symbol, source + 1) for source, symbol, target in automaton.transitions
365 }
366 transitions |= {(0, EPSILON, state + 1) for state in automaton.accepting}
367 states = series("N", len(automaton.states) + 1)
368 return Automaton("NFA", states, automaton.alphabet, 0, {automaton.start + 1}, transitions)

371def concatenate(first, second):
372 offset = len(first.states)
373 second_start = second.start + offset
374 transitions = set(first.transitions)
375 transitions |= {
376 (source + offset, symbol, target + offset) for source, symbol, target in second.transitions
377 }
378 transitions |= {(state, EPSILON, second_start) for state in first.accepting}
379 accepting = {state + offset for state in second.accepting}
380 states = series("N", offset + len(second.states))
381 return Automaton("NFA", states, first.alphabet, first.start, accepting, transitions)

384def star(automaton):
385 start = automaton.start + 1
386 transitions = {
387 (source + 1, symbol, target + 1) for source, symbol, target in automaton.transitions
388 }
389 transitions.add((0, EPSILON, start))
390 transitions |= {(state + 1, EPSILON, start) for state in automaton.accepting}
391 accepting = {0} | {state + 1 for state in automaton.accepting}
392 states = series("N", len(automaton.states) + 1)
393 return Automaton("NFA", states, automaton.alphabet, 0, accepting, transitions)

398 level = [("", dfa.start)]
399 for length in range(max_length + 1):
400 accepted += [word for word, state in level if state in dfa.accepting]

405 for symbol in dfa.alphabet

410def count_by_length(dfa, max_length):
411 counts = []
412 weights = {dfa.start: 1}
413 for length in range(max_length + 1):
414 counts.append(sum(weight for state, weight in weights.items() if state in dfa.accepting))
415 if length < max_length:
416 following = defaultdict(int)
417 for state, weight in weights.items():
418 for symbol in dfa.alphabet:
419 following[dfa.delta(state, symbol)] += weight
420 weights = following
421 return counts

454def transition_table(automaton):
455 columns = list(automaton.alphabet)
456 if automaton.kind == "NFA" and automaton.epsilon_count() > 0:
457 columns.append(EPSILON)
458 rows = [[""] + columns]
459 for state, name in enumerate(automaton.states):
460 start_mark = "->" if state == automaton.start else " "
461 accepting_mark = "*" if state in automaton.accepting else " "
462 row = [start_mark + accepting_mark + name]
463 for symbol in columns:
464 if automaton.kind == "DFA":
465 row.append(automaton.states[automaton.delta(state, symbol)])
466 else:
467 targets = automaton.successors(state, symbol)
468 row.append(automaton.format_set(targets) if targets else "-")
469 rows.append(row)
470 widths = [max(len(cell) for cell in column) for column in zip(*rows)]
471 return [
472 " ".join(cell.ljust(width) for cell, width in zip(row, widths)).rstrip() for row in rows
473 ]

497class Toolkit:
498 def __init__(self):
499 self.table = {}
500
501 def execute(self, words):
502 command, args = words[0], words[1:]
503 try:
504 if command in ("DFA", "NFA"):
505 return [self.define(command, args)]
506 if command == "REGEX":
507 return [self.define_regex(args)]
508 if command not in COMMAND_ROLES:
509 raise CommandError(f"unknown command {command}")
510 self.check_arguments(COMMAND_ROLES[command], args)
511 return getattr(self, "do_" + command.lower())(*args)
512 except CommandError as error:
513 return [f"error: {error}"]
514
515 def check_arguments(self, roles, args):
516 if len(args) != len(roles):
517 raise CommandError("wrong number of arguments")
518 arguments = list(zip(roles, args))
519 references = [name for role, name in arguments if role in (REFERENCE, DFA_REFERENCE)]
520 for name in references:
521 if name not in self.table:
522 raise CommandError(f"unknown automaton {name}")
523 for role, name in arguments:
524 if role == NEW_NAME and name in self.table:
525 raise CommandError(f"{name} already exists")
526 for role, name in arguments:
527 if role == DFA_REFERENCE and self.table[name].kind != "DFA":
528 raise CommandError(f"{name} is not a DFA")
529 if len(references) == 2:
530 first, second = (self.table[name] for name in references)
531 if first.alphabet != second.alphabet:
532 raise CommandError("alphabets differ")
533
534 def register(self, name, automaton):
535 self.table[name] = automaton
536 return automaton.summary(name)
537
538 def define(self, kind, args):
539 if len(args) < 5:
540 raise CommandError("wrong number of arguments")
541 name, state_list, alphabet, start, accepting_list, *transition_words = args
542 if name in self.table:
543 raise CommandError(f"{name} already exists")
544 states = state_list.split(",")
545 index_of = {}
546 for state in states:
547 if state in index_of:
548 raise CommandError(f"duplicate state {state}")
549 index_of[state] = len(index_of)
550 if start not in index_of:
551 raise CommandError(f"unknown state {start}")
552 accepting = set()
553 if accepting_list != "-":
554 for state in accepting_list.split(","):
555 if state not in index_of:
556 raise CommandError(f"unknown state {state}")
557 if index_of[state] in accepting:
558 raise CommandError(f"duplicate state {state}")
559 accepting.add(index_of[state])
560 transitions = set()
561 defined = set()
562 for word in transition_words:
563 source, symbol, target = word.split(":")
564 if source not in index_of:
565 raise CommandError(f"unknown state {source}")
566 if symbol == EPSILON and kind == "DFA":
567 raise CommandError("epsilon transition in DFA")
568 if symbol != EPSILON and symbol not in alphabet:
569 raise CommandError(f"unknown symbol {symbol}")
570 if target not in index_of:
571 raise CommandError(f"unknown state {target}")
572 transition = (index_of[source], symbol, index_of[target])
573 if transition in transitions:
574 raise CommandError(f"duplicate transition {word}")
575 if kind == "DFA" and transition[:2] in defined:
576 raise CommandError(f"conflicting transition {word}")
577 transitions.add(transition)
578 defined.add(transition[:2])
579 if kind == "DFA":
580 for state in states:
581 for symbol in sorted(alphabet):
582 if (index_of[state], symbol) not in defined:
583 raise CommandError(f"missing transition {state}:{symbol}")
584 automaton = Automaton(kind, states, alphabet, index_of[start], accepting, transitions)
585 return self.register(name, automaton)

587 def define_regex(self, args):
588 if len(args) != 3:
589 raise CommandError("wrong number of arguments")
590 name, alphabet, expression = args
591 if name in self.table:
592 raise CommandError(f"{name} already exists")
593 for char in expression:
594 if is_symbol_char(char) and char != EPSILON and char not in alphabet:
595 raise CommandError(f"unknown symbol {char}")
596 tree = RegexParser(expression).parse()
597 builder = ThompsonBuilder()
598 entry, exit_state = builder.build(tree)
599 states = series("N", builder.state_count)
600 automaton = Automaton("NFA", states, alphabet, entry, {exit_state}, builder.transitions)
601 return self.register(name, automaton)

700 def do_empty(self, name):
701 word = smallest_accepted(self.table[name])
702 if word is None:
703 return ["EMPTY"]
704 return [f"NOT EMPTY {show_word(word)}"]
705
706 def do_finite(self, name):
707 automaton = self.table[name]
708 dfa, _ = determinize(automaton)
709 if is_infinite(dfa):
710 return ["INFINITE"]
711 total = sum(count_by_length(dfa, len(automaton.states) - 1))
712 return [f"FINITE {total}"]
713
714 def do_sync(self, name):
715 dfa = self.table[name]
716 result = synchronizing_word(dfa)
717 if result is None:
718 return ["NOT SYNCHRONIZING"]
719 word, state = result
720 return [f"SYNCHRONIZING {show_word(word)} -> {dfa.states[state]}"]
721
722 def do_info(self, name):
723 automaton = self.table[name]
724 epsilon_count = automaton.epsilon_count()
725 deterministic = epsilon_count == 0 and all(
726 len(targets) == 1 for targets in automaton.targets.values()
727 )
728 complete = all(
729 (state, symbol) in automaton.targets
730 for state in range(len(automaton.states))
731 for symbol in automaton.alphabet
732 )
733 dfa, _ = determinize(automaton)
734 if not dfa.accepting:
735 language = "empty"
736 elif is_infinite(dfa):
737 language = "infinite"
738 else:
739 language = "finite"
740 minimal, _ = minimize(dfa)
741 return [
742 automaton.summary(name),
743 f" epsilon transitions: {epsilon_count}",
744 f" reachable: {automaton.format_list(automaton.reachable_states())}",
745 f" live: {automaton.format_list(automaton.live_states())}",
746 f" deterministic: {yes_no(deterministic)}",
747 f" complete: {yes_no(complete)}",
748 f" language: {language}",
749 f" minimal DFA states: {len(minimal.states)}",
750 ]
751
752 def do_table(self, name):
753 return transition_table(self.table[name])
754
755 def do_show(self, name):
756 return [definition_line(name, self.table[name])]
757
758 def do_list(self):
759 return [automaton.summary(name) for name, automaton in self.table.items()] or ["(none)"]

762 del self.table[name]
763 return [f"deleted {name}"]
