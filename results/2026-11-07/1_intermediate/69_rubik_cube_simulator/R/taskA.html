119parse_sequence <- function(text, base, line_length) {
120 position <- 1
121 char <- function() if (position > nchar(text)) "" else substr(text, position, position)
122 syntax <- function() fail(paste0("syntax at ", min(base + position, line_length + 1)))
123 spaces <- function() while (char() == " ") position <<- position + 1
124 power <- function(sequence) {
125 count <- 1
126 if (grepl("^[1-9]$", char())) {
127 count <- as.integer(char())
128 position <<- position + 1
129 if (grepl("^[0-9]$", char())) {
130 count <- 10 * count + as.integer(char())
131 position <<- position + 1
132 }
133 }
134 sequence <- rep(sequence, count)
135 if (char() == "'") {
136 position <<- position + 1
137 sequence <- inverse(sequence)
138 }
139 sequence
140 }
141 read <- function() {
142 output <- integer()
143 repeat {
144 spaces()
145 c <- char()
146 if (c %in% move_letters) {
147 position <<- position + 1
148 amount <- 1L
149 if (char() %in% c("2", "'")) {
150 amount <- if (char() == "2") 2L else 3L
151 position <<- position + 1
152 }
153 output <- c(output, move_code(c, amount))
154 } else if (c == "(") {
155 position <<- position + 1
156 sequence <- read()
157 spaces()
158 if (char() != ")") syntax()
159 position <<- position + 1
160 output <- c(output, power(sequence))
161 } else if (c == "[") {
162 position <<- position + 1
163 a <- read()
164 spaces()
165 delimiter <- char()
166 if (!(delimiter %in% c(",", ":"))) syntax()
167 position <<- position + 1
168 b <- read()
169 spaces()
170 if (char() != "]") syntax()
171 position <<- position + 1
172 sequence <- c(a, b, inverse(a))
173 if (delimiter == ",") sequence <- c(sequence, inverse(b))
174 output <- c(output, power(sequence))
175 } else if (c == "{") {
176 position <<- position + 1
177 start <- position
178 while (grepl("^[a-z]$", char())) position <<- position + 1
179 name <- if (position > start) substr(text, start, position - 1) else ""
180 if (name == "" || char() != "}") syntax()
181 if (!(name %in% names(macros))) fail(paste0("undefined ", name))
182 position <<- position + 1
183 output <- c(output, power(macros[[name]]))
184 } else break
185 }
186 output
187 }
188 sequence <- read()
189 spaces()
190 if (position <= nchar(text)) syntax()
191 if (length(sequence) > 500) fail("too long")
192 sequence
193}

200simplify <- function(sequence) {
201 repeat {
202 output <- integer()
203 position <- 1
204 while (position <= length(sequence)) {
205 current_axis <- axis(move_letter(sequence[position]))
206 names <- character()
207 amounts <- numeric()
208 while (position <= length(sequence) && axis(move_letter(sequence[position])) == current_axis) {
209 letter <- move_letter(sequence[position])
210 if (!(letter %in% names)) {
211 names <- c(names, letter)
212 amounts[letter] <- 0
213 }
214 amounts[letter] <- (amounts[letter] + move_amount(sequence[position])) %% 4
215 position <- position + 1
216 }
217 for (letter in names) if (amounts[letter] != 0) output <- c(output, move_code(letter, amounts[letter]))
218 }
219 if (identical(output, sequence)) return(output)
220 sequence <- output
221 }

264 total <- 0

270cycles <- function(state, kind) {
271 pieces <- identify(state, kind)
272 destinations <- integer(length(pieces))
273 orientations <- integer(length(pieces))
274 for (i in seq_along(pieces)) {
275 destinations[pieces[[i]][1]] <- i
276 orientations[pieces[[i]][1]] <- pieces[[i]][2]
277 }
278 seen <- logical(length(pieces))
279 output <- character()
280 for (start in seq_along(pieces)) {
281 if (seen[start]) next
282 cycle <- integer()
283 current <- start
284 repeat {
285 cycle <- c(cycle, current)
286 seen[current] <- TRUE
287 current <- destinations[current]
288 if (current == start) break
289 }
290 sum <- sum(orientations[cycle])
291 mark <- if (kind == 1) c("", "+", "-")[sum %% 3 + 1] else if (kind == 2 && sum %% 2 == 1) "+" else ""
292 if (length(cycle) >= 2 || mark != "") output <- c(output, paste0("(", paste(piece_names[[kind]][cycle], collapse = " "), ")", mark))
293 }
294 if (length(output)) paste(output, collapse = " ") else "none"
295}
296sequence_order <- function(permutation) {
297 seen <- logical(54)
298 result <- 1
299 for (start in seq_along(permutation)) {
300 if (seen[start]) next
301 current <- start
302 length <- 0
303 repeat {
304 seen[current] <- TRUE
305 length <- length + 1
306 current <- permutation[current]
307 if (current == start) break
308 }
309 a <- result
310 b <- length
311 while (b != 0) {
312 remainder <- a %% b
313 a <- b
314 b <- remainder
315 }
316 result <- result / a * length
317 }
318 result

327unrotate <- function(sequence) {
328 replacements <- list(M = c(move_code("R"), move_code("L", 3), move_code("x", 3)), E = c(move_code("U"), move_code("D", 3), move_code("y", 3)),
329 S = c(move_code("F", 3), move_code("B"), move_code("z")), r = c(move_code("L"), move_code("x")), l = c(move_code("R"), move_code("x", 3)),
330 u = c(move_code("D"), move_code("y")), d = c(move_code("U"), move_code("y", 3)), f = c(move_code("B"), move_code("z")), b = c(move_code("F"), move_code("z", 3)))
331 expanded <- integer()
332 for (code in sequence) {
333 letter <- move_letter(code)
334 if (letter %in% names(replacements)) {
335 original <- replacements[[letter]]
336 expanded <- c(expanded, move_code(move_letter(original), (move_amount(original) * move_amount(code)) %% 4))
337 } else expanded <- c(expanded, code)
338 }
339 rotation <- identity
340 output <- integer()
341 for (code in expanded) {
342 if (move_letter(code) %in% c("x", "y", "z")) rotation <- compose(rotation, moves[[code]])
343 else output <- c(output, converted(code, rotation))
344 }
345 for (i in seq_along(rotation_effects)) if (identical(rotation_effects[[i]], rotation)) return(c(output, rotations[[i]]))
346 stop("No matching rotation")
347}
348search <- function(depth) {
349 candidates <- unlist(lapply(faces, function(face) move_code(face, 1:3)), use.names = FALSE)
350 visit <- function(state, left, previous) {
351 if (is_solved(state)) return(integer())
352 if (left == 0) return(NULL)
353 for (code in candidates) {
354 face <- move_letter(code)
355 if ((!is.null(previous) && face == previous) || (!is.null(previous) && paste0(previous, face) %in% c("DU", "LR", "BF"))) next
356 tail <- visit(permute(state, moves[[code]]), left - 1, face)
357 if (!is.null(tail)) return(c(code, tail))
358 }
359 NULL
360 }
361 for (limit in seq.int(0, depth)) {
362 found <- visit(state, limit, NULL)
363 if (!is.null(found)) return(found)
364 }
365 NULL
366}
367command <- function(line) {
368 at <- regexpr(" ", line, fixed = TRUE)[1]
369 name <- if (at < 0) line else substr(line, 1, at - 1)
370 argument <- if (at < 0) NULL else substring(line, at + 1)
371 tryCatch({
372 if (!(name %in% c(no_arguments, names(word_commands), sequence_commands, head_commands))) fail(paste0("unknown command ", name))
373 words <- if (is.null(argument)) character() else strsplit(argument, " ", fixed = TRUE)[[1]]
374 if (name %in% no_arguments && !is.null(argument)) fail("bad arguments")
375 if (name %in% names(word_commands) && length(words) != word_commands[name]) fail("bad arguments")
376 if (name %in% c("UNDO", "REDO", "SEARCH", "SCRAMBLE") && !all(grepl("^(0|[1-9][0-9]{0,9})$", words))) fail("bad arguments")
377 if (name %in% c("SAVE", "RESTORE") && !grepl("^[a-z]{1,8}$", words[1])) fail("bad arguments")
378 sequence <- integer()
379 if (name %in% head_commands) {
380 if (is.null(argument)) fail("bad arguments")
381 at <- regexpr(" ", argument, fixed = TRUE)[1]
382 head <- if (at < 0) argument else substr(argument, 1, at - 1)
383 text <- if (at < 0) "" else substring(argument, at + 1)
384 valid <- switch(name, DEFINE = grepl("^[a-z]{1,8}$", head), MIRROR = head %in% c("LR", "UD", "FB"), ROTATE = grepl("^[xyz](2|')?$", head))
385 if (!valid) fail("bad arguments")
386 sequence <- parse_sequence(text, nchar(name) + nchar(head) + 2, nchar(line))
387 } else if (name %in% sequence_commands) sequence <- parse_sequence(if (is.null(argument)) "" else argument, nchar(name) + 1, nchar(line))
388 if (name %in% c("UNDO", "REDO") && as.numeric(words[1]) < 1) fail("bad value")
389 if (name == "SEARCH" && as.numeric(words[1]) > 4) fail("bad value")
390 if (name == "SCRAMBLE" && (as.numeric(words[1]) < 1 || as.numeric(words[1]) > 50 || as.numeric(words[2]) < 1 || as.numeric(words[2]) > 2147483646)) fail("bad value")
391 if (name == "WHERE" && !(words[1] %in% c(corner_names, edge_names))) fail(paste0("unknown piece ", words[1]))
392 if (name == "RESET") {
393 state <<- solved_state
394 history <<- integer()
395 redo <<- integer()
396 emit("reset")
397 } else if (name == "SHOW") {
398 for (side in c("U", "LFRB", "D")) for (row in 1:3) {
399 stickers <- unlist(lapply(strsplit(side, "", fixed = TRUE)[[1]], function(face) state[vapply(1:3, function(col) index(face, row, col), integer(1))]), use.names = FALSE)
400 emit(if (nchar(side) == 1) " " else "", paste(stickers, collapse = " "))
401 }
402 } else if (name == "STATE") emit("state ", paste(state, collapse = ""))
403 else if (name == "ISSOLVED") {
404 count <- sum(state != rep(state[seq(5, 50, 9)], each = 9))
405 emit(if (count == 0) "solved" else paste0("unsolved ", count))
406 } else if (name == "HISTORY") emit("history ", notation(history))
407 else if (name == "SOLUTION") emit("solution ", notation(simplify(inverse(history))))
408 else if (name == "ANALYZE") {
409 normalized <- normalize(state)
410 emit("rotation ", if (normalized$rotation == 1) "none" else move_words(rotations[[normalized$rotation]]))
411 emit("corners ", cycles(normalized$state, 1))
412 emit("edges ", cycles(normalized$state, 2))
413 } else if (name == "STAGE") {
414 normalized <- normalize(state)$state
415 corners <- identify(normalized, 1)
416 edges <- identify(normalized, 2)
417 cross <- sum(vapply(5:8, function(i) all(edges[[i]] == c(i, 0)), logical(1)))
418 slots <- sum(vapply(5:8, function(i) all(corners[[i]] == c(i, 0)) && all(edges[[i + 4]] == c(i + 4, 0)), logical(1)))
419 top <- sum(normalized[1:9] == "W")
420 stage <- if (is_solved(normalized)) "solved" else if (cross == 4 && slots == 4 && top == 9) "oll" else if (cross == 4 && slots == 4) "f2l" else if (cross == 4) "cross" else "none"
421 emit("stage ", stage, " cross ", cross, " slots ", slots, " top ", top)
422 } else if (name == "MACROS") {
423 emit("macros ", length(macros))
424 for (key in sort(names(macros), method = "radix")) emit("macro ", key, " ", notation(macros[[key]]))
425 } else if (name %in% c("CHECK", "LOAD")) {
426 new_state <- strsplit(words[1], "", fixed = TRUE)[[1]]
427 reason <- invalid(new_state)
428 if (name == "CHECK") emit(if (is.null(reason)) "valid" else paste0("invalid ", reason))
429 else {
430 if (!is.null(reason)) fail(paste0("invalid ", reason))
431 state <<- new_state
432 history <<- integer()
433 redo <<- integer()
434 emit("loaded")
435 }
436 } else if (name %in% c("UNDO", "REDO")) {
437 source <- if (name == "UNDO") history else redo
438 if (!length(source)) fail(if (name == "UNDO") "nothing to undo" else "nothing to redo")
439 count <- min(as.numeric(words[1]), length(source))
440 for (i in seq_len(count)) {
441 if (name == "UNDO") {
442 code <- tail(history, 1)
443 history <<- head(history, -1)
444 state <<- permute(state, moves[[inverse(code)]])
445 redo <<- c(redo, code)
446 } else {
447 code <- tail(redo, 1)
448 redo <<- head(redo, -1)
449 state <<- permute(state, moves[[code]])
450 history <<- c(history, code)
451 }
452 }
453 emit(if (name == "UNDO") "undone " else "redone ", count)
454 } else if (name == "SEARCH") {
455 found <- search(as.numeric(words[1]))
456 emit(if (is.null(found)) "not found" else paste0("found ", notation(found)))
457 } else if (name == "WHERE") {
458 normalized <- normalize(state)$state
459 kind <- if (words[1] %in% corner_names) 1 else 2
460 wanted <- match(words[1], piece_names[[kind]])
461 pieces <- identify(normalized, kind)
462 position <- which(vapply(pieces, function(piece) piece[1] == wanted, logical(1)))[1]
463 emit(words[1], " at ", piece_names[[kind]][position], if (kind == 1) " twist " else " flip ", pieces[[position]][2])
464 } else if (name == "SAVE") {
465 saves[[words[1]]] <<- list(state = state, history = history)
466 emit("saved ", words[1])
467 } else if (name == "RESTORE") {
468 if (!(words[1] %in% names(saves))) fail(paste0("no save ", words[1]))
469 state <<- saves[[words[1]]]$state
470 history <<- saves[[words[1]]]$history
471 redo <<- integer()
472 emit("restored ", words[1])
473 } else if (name %in% c("SCRAMBLE", "APPLY")) {
474 if (name == "SCRAMBLE") {
