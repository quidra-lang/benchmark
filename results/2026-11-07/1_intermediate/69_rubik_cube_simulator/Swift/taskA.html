434 func cycles(_ positions:[(Int,Int)?],_ names:[String],_ modulus:Int) -> String {
435 var destination=Array(repeating:0,count:names.count), orientation=destination, seen=Array(repeating:false,count:names.count), out:[String]=[]
436 for (q,entry) in positions.enumerated() {
437 let (piece,t)=entry!
438 destination[piece]=q
439 orientation[piece]=t
440 }
441 for start in names.indices {
442 if seen[start] {
443 continue
444 }
445 var cycle:[String]=[], sum=0, p=start
446 while !seen[p] {
447 seen[p]=true
448 cycle.append(names[p])
449 sum += orientation[p]
450 p=destination[p]
451 }
452 let mark=modulus==1 || sum%modulus==0 ? "" : (modulus==3 && sum%3==2 ? "-" : "+")
453 if cycle.count>=2 || !mark.isEmpty {
454 out.append("("+cycle.joined(separator:" ")+")"+mark)
455 }
456 }
457 return out.isEmpty ? "none" : out.joined(separator:" ")
458 }
