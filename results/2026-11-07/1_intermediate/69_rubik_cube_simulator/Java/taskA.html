153 static class SequenceParser {
154 String line;
155 int at;
156 SequenceParser(String line, int start) {
157 this.line = line;
158 at = start;
159 }
160 char peek() {
161 return at >= line.length() ? '\0' : line.charAt(at);
162 }
163 void spaces() {
164 while (peek() == ' ') at++;
165 }
166 void syntax() {
167 throw new Problem("error: syntax at " + (Math.min(at, line.length()) + 1));
168 }
169 List<Move> power(List<Move> moves) {
170 int count = 1;
171 if (peek() >= '1' && peek() <= '9') {
172 count = peek() - '0';
173 at++;
174 if (peek() >= '0' && peek() <= '9') {
175 count = 10 * count + peek() - '0';
176 at++;
177 }
178 }
179 boolean reverse = peek() == '\'';
180 if (reverse) at++;
181 List<Move> out = new ArrayList<>();
182 for (int i = 0; i < count; i++) out.addAll(moves);
183 return reverse ? invert(out) : out;
184 }
185 List<Move> sequence() {
186 List<Move> out = new ArrayList<>();
187 while (true) {
188 spaces();
189 char c = peek();
190 if (LETTERS.indexOf(c) >= 0 && c != '\0') {
191 at++;
192 int amount = peek() == '2' ? 2 : peek() == '\'' ? 3 : 1;
193 if (amount != 1) at++;
194 out.add(new Move(c, amount));
195 } else if (c == '(') {
196 at++;
197 List<Move> group = sequence();
198 spaces();
199 if (peek() != ')') syntax();
200 at++;
201 out.addAll(power(group));
202 } else if (c == '[') {
203 at++;
204 List<Move> first = sequence();
205 spaces();
206 char separator = peek();
207 if (separator != ',' && separator != ':') syntax();
208 at++;
209 List<Move> second = sequence();
210 spaces();
211 if (peek() != ']') syntax();
212 at++;
213 List<Move> group = new ArrayList<>(first);
214 group.addAll(second);
215 group.addAll(invert(first));
216 if (separator == ',') group.addAll(invert(second));
217 out.addAll(power(group));
218 } else if (c == '{') {
219 at++;
220 int start = at;
221 while (peek() >= 'a' && peek() <= 'z') at++;
222 String name = line.substring(start, at);
223 if (name.isEmpty() || peek() != '}') syntax();
224 if (!macros.containsKey(name)) throw new Problem("error: undefined " + name);
225 at++;
226 out.addAll(power(macros.get(name)));
227 } else return out;
228 }
229 }
230 List<Move> whole() {
231 List<Move> out = sequence();
232 spaces();
233 if (at < line.length()) syntax();
234 if (out.size() > 500) throw new Problem("error: too long");
235 return out;
236 }
237 }

304 static String cycles(Pieces pieces, String[] names, int base) {
305 int[] destination = new int[names.length], orientation = new int[names.length];
306 for (int q = 0; q < names.length; q++) {
307 destination[pieces.piece()[q]] = q;
308 orientation[pieces.piece()[q]] = pieces.orientation()[q];
309 }
310 boolean[] seen = new boolean[names.length];
311 List<String> out = new ArrayList<>();
312 for (int start = 0; start < names.length; start++) if (!seen[start]) {
313 List<String> cycle = new ArrayList<>();
314 int p = start, sum = 0;
315 do {
316 seen[p] = true;
317 cycle.add(names[p]);
318 sum += orientation[p];
319 p = destination[p];
320 }
321 while (p != start);
322 String mark = base == 3 ? sum % 3 == 1 ? "+" : sum % 3 == 2 ? "-" : "" : base == 2 && sum % 2 == 1 ? "+" : "";
323 if (cycle.size() >= 2 || !mark.isEmpty()) out.add("(" + String.join(" ", cycle) + ")" + mark);
324 }
325 return out.isEmpty() ? "none" : String.join(" ", out);
326 }

390 static long order(int[] permutation) {
391 boolean[] seen = new boolean[54];
392 long order = 1;
393 for (int i = 0; i < 54; i++) if (!seen[i]) {
394 int count = 0, p = i;
395 do {
396 seen[p] = true;
397 count++;
398 p = permutation[p];

400 while (p != i);
401 order = order / gcd(order, count) * count;
402 }
403 return order;

544 boolean found = false;
545 for (int depth = 0; depth <= n; depth++) {
546 solution.clear();
547 if (search(state, depth, -1, solution)) {
548 found = true;
549 break;
550 }
551 }
552 System.out.println(found ? "found " + sequenceText(solution) : "not found");
553 break;
554 }
555 case "WHERE":
556 {
557 char[] normalized = normalize(state).state();
558 boolean corner = Arrays.asList(CORNERS).contains(words[0]);
559 String[] names = corner ? CORNERS : EDGES;
560 Pieces pieces = identify(normalized, corner ? CORNER_STICKERS : EDGE_STICKERS, names);
561 int piece = Arrays.asList(names).indexOf(words[0]);
562 for (int q = 0; q < names.length; q++) if (pieces.piece()[q] == piece) System.out.println(words[0] + " at " + names[q] + (corner ? " twist " : " flip ") + pieces.orientation()[q]);
563 break;
564 }
565 case "SAVE":
566 saves.put(words[0], new Save(state.clone(), new ArrayList<>(history)));
567 System.out.println("saved " + words[0]);
568 break;
569 case "RESTORE":
570 {
571 Save save = saves.get(words[0]);
572 require(save != null, "error: no save " + words[0]);
573 state = save.state().clone();
574 history.clear();
575 history.addAll(save.history());
576 redo.clear();
577 System.out.println("restored " + words[0]);
578 break;
579 }
580 case "SCRAMBLE":
581 {
582 List<Move> scramble = new ArrayList<>();
583 long random = seed;
584 for (int i = 0; i < n; i++) {
585 List<Character> usable = new ArrayList<>();
586 for (char face : FACES.toCharArray()) {
587 if (i >= 1 && scramble.get(i - 1).letter() == face) continue;
588 if (i >= 2 && axis(scramble.get(i - 1).letter()) == axis(scramble.get(i - 2).letter()) && axis(face) == axis(scramble.get(i - 1).letter())) continue;
589 usable.add(face);
590 }
591 random = 48271 * random % 2147483647;
592 char face = usable.get((int)(random % usable.size()));
593 random = 48271 * random % 2147483647;
594 int amount = new int[]{1,3,2}[(int)(random % 3)];
595 scramble.add(new Move(face, amount));
596 }
597 applyCurrent(scramble);
598 System.out.println("scramble " + moveText(scramble));
599 break;
600 }
601 case "APPLY":
602 applyCurrent(moves);
603 System.out.println("applied " + moves.size());
604 break;
605 case "INVERT":
606 System.out.println("inverse " + sequenceText(invert(moves)));
607 break;
608 case "SIMPLIFY":
609 System.out.println("simplified " + sequenceText(simplify(moves)));
610 break;
611 case "METRICS":
612 {
613 int htm = 0, qtm = 0, stm = 0;
614 for (Move move : moves) if ("xyz".indexOf(move.letter()) < 0) {
615 int weight = "MES".indexOf(move.letter()) >= 0 ? 2 : 1;
616 htm += weight;
617 qtm += weight * (move.amount() == 2 ? 2 : 1);
618 stm++;
619 }
620 System.out.println("metrics htm " + htm + " qtm " + qtm + " stm " + stm + " etm " + moves.size());
621 break;
622 }
623 case "ORDER":
624 System.out.println("order " + order(effect(moves)));
625 break;
626 case "CYCLES":
627 cycles(apply(solvedState(), moves), true);
628 break;
629 case "UNROTATE":
630 System.out.println("unrotated " + sequenceText(unrotate(moves)));
631 break;
632 case "UNTIL":
633 {
634 int[] permutation = effect(moves);
635 long order = order(permutation), found = 0;
636 char[] candidate = state;
637 for (long k = 1; k <= order; k++) {
638 candidate = apply(candidate, permutation);
639 if (solved(candidate)) {
640 found = k;
641 break;
642 }
643 }
644 System.out.println(found == 0 ? "until never" : "until " + found);
645 break;
646 }
647 case "DEFINE":
648 macros.put(head, new ArrayList<>(moves));
649 System.out.println("defined " + head + " " + moves.size());
650 break;
651 case "MIRROR":
652 {
653 int mirrorAxis = head.equals("LR") ? 0 : head.equals("UD") ? 1 : 2;
654 String opposites = "DLBURFdlburf", originals = "URFDLBurfdlb";
655 List<Move> out = new ArrayList<>();
656 for (Move move : moves) {
657 char letter = move.letter();
658 int amount = move.amount();
659 if (axis(letter) == mirrorAxis) {
660 int i = originals.indexOf(letter);
661 if (i >= 0) {
662 letter = opposites.charAt(i);
663 amount = 4 - amount;
664 }
665 } else amount = 4 - amount;
666 out.add(new Move(letter, amount));
667 }
668 System.out.println("mirror " + sequenceText(out));
669 break;
670 }
671 case "ROTATE":
672 {
673 Move rotation = new Move(head.charAt(0), head.length() == 1 ? 1 : head.charAt(1) == '2' ? 2 : 3);
674 List<Move> out = new ArrayList<>();
675 for (Move move : moves) out.add(conjugate(permutations.get(rotation), move));
676 System.out.println("rotated " + sequenceText(out));
677 }
678 }
