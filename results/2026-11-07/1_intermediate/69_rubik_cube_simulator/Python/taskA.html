159def sticker_number(label):
160 return 9 * FACES.index(label[0]) + int(label[1]) - 1

163def sticker_places():
164 places = []
165 for normal, column_step, row_step in FACE_FRAMES:
166 for row in (-1, 0, 1):
167 for column in (-1, 0, 1):
168 cubie = tuple(
169 n + column * c + row * r for n, c, r in zip(normal, column_step, row_step)
170 )
171 places.append((cubie, normal))
172 return places

175def quarter_turn(vector, axis):
176 turned = list(vector)
177 first, second = (axis + 1) % 3, (axis + 2) % 3
178 turned[first], turned[second] = vector[second], -vector[first]
179 return tuple(turned)

182def permute(state, table):
183 return [state[source] for source in table]

186def build_move_tables():
187 places = sticker_places()
188 numbers = {place: number for number, place in enumerate(places)}
189 tables = {}
190 for letter, (axis, turns, layers) in TURNS.items():
191 quarter = list(range(54))
192 for number, (cubie, normal) in enumerate(places):
193 if cubie[axis] in layers:
194 for _ in range(turns):
195 cubie = quarter_turn(cubie, axis)
196 normal = quarter_turn(normal, axis)
197 quarter[numbers[cubie, normal]] = number
198 table = list(range(54))
199 for amount in (1, 2, 3):
200 table = permute(table, quarter)
201 tables[letter, amount] = table
202 return tables

205def apply_moves(state, moves):
206 for move in moves:
207 state = permute(state, MOVE_TABLES[move])

211def effect(moves):
212 return tuple(apply_moves(list(range(54)), moves))

226class PieceKind:
227 def __init__(self, layout):
228 self.names = list(layout)
229 self.stickers = [
230 [sticker_number(label) for label in labels.split()] for labels in layout.values()
231 ]
232 self.colors = [[FACE_COLORS[face] for face in name] for name in self.names]
233
234 def identify(self, colors):
235 for piece, piece_colors in enumerate(self.colors):
236 for orientation in range(len(colors)):
237 if colors[orientation:] + colors[:orientation] == piece_colors:
238 return piece, orientation
239 return None
240
241 def contents(self, state):
242 return [self.identify([state[number] for number in group]) for group in self.stickers]
243
244 def placed(self, state):
245 return {
246 self.names[position]
247 for position, content in enumerate(self.contents(state))
248 if content == (position, 0)
249 }
250
251 def locate(self, state, name):
252 piece = self.names.index(name)
253 contents = self.contents(state)
254 position = next(index for index, content in enumerate(contents) if content[0] == piece)
255 return self.names[position], contents[position][1]
256
257 def cycle_notation(self, state):
258 size = len(self.names)
259 location = [0] * size
260 twist = [0] * size
261 for position, (piece, orientation) in enumerate(self.contents(state)):
262 location[piece] = position
263 twist[piece] = orientation
264 visited = set()
265 cycles = []
266 for start in range(size):
267 if start in visited:
268 continue
269 cycle = [start]
270 position = location[start]
271 while position != start:
272 cycle.append(position)
273 position = location[position]
274 visited.update(cycle)
275 mark = MARKS[sum(twist[piece] for piece in cycle) % len(self.stickers[0])]
276 if len(cycle) >= 2 or mark:
277 cycles.append("(" + " ".join(self.names[piece] for piece in cycle) + ")" + mark)
278 return " ".join(cycles) or "none"

311def permutation_parity(contents):
312 pieces = [piece for piece, _ in contents]
313 return sum(first > second for first, second in combinations(pieces, 2)) % 2

316def invalid_reason(text):
317 if len(text) != 54:
318 return f"length {len(text)}"
319 for position, color in enumerate(text, 1):
320 if color not in COLORS:
321 return f"char {position}"
322 for color in COLORS:
323 if text.count(color) != 9:
324 return f"count {color} {text.count(color)}"
325 state = list(text)
326 rotation = normalizing_rotation(state)
327 if rotation is None:
328 return "centers"
329 kinds = ((CORNERS, CORNERS.contents(state), "corner"), (EDGES, EDGES.contents(state), "edge"))
330 for kind, contents, label in kinds:
331 if None in contents:
332 return f"{label} {kind.names[contents.index(None)]}"
333 for kind, contents, _ in kinds:
334 present = {piece for piece, _ in contents}
335 for piece, name in enumerate(kind.names):
336 if piece not in present:
337 return f"missing {name}"
338 normal = apply_moves(state, rotation)
339 corners = CORNERS.contents(normal)
340 edges = EDGES.contents(normal)
341 if sum(orientation for _, orientation in corners) % 3:
342 return "twist"
343 if sum(orientation for _, orientation in edges) % 2:
344 return "flip"
345 if permutation_parity(corners) != permutation_parity(edges):
346 return "parity"
347 return None

350def simplify(moves):
351 while True:
352 result = []
353 for _, run in groupby(moves, key=lambda move: AXIS[move[0]]):
354 totals = {}
355 for letter, amount in run:
356 totals[letter] = (totals.get(letter, 0) + amount) % 4
357 result.extend((letter, total) for letter, total in totals.items() if total)
358 if result == moves:
359 return result
360 moves = result

363def mirror(moves, plane):
364 mirror_axis = MIRROR_AXES[plane]
365 result = []
366 for letter, amount in moves:
367 if AXIS[letter] != mirror_axis:
368 result.append((letter, 4 - amount))
369 elif letter in OPPOSITES:
370 result.append((OPPOSITES[letter], 4 - amount))
371 else:
372 result.append((letter, amount))
373 return result

376def rotate(moves, rotation):
377 undo = inverse([rotation])
378 return [MOVES_BY_EFFECT[effect([rotation, move] + undo)] for move in moves]

381def unrotate(moves):
382 expanded = []
383 for letter, amount in moves:
384 if letter in REPLACEMENTS:
385 expanded.extend(
386 (part, part_amount * amount % 4) for part, part_amount in REPLACEMENTS[letter]
387 )
388 else:
389 expanded.append((letter, amount))
390 held = []
391 result = []
392 for move in expanded:
393 if move[0] in ROTATION_LETTERS:
394 held = ROTATIONS_BY_EFFECT[effect(held + [move])]
395 else:
396 result.append(MOVES_BY_EFFECT[effect(held + [move] + inverse(held))])
397 return result + held

400def metrics(moves):
401 htm = qtm = stm = 0
402 for letter, amount in moves:
403 if letter in ROTATION_LETTERS:
404 continue
405 weight = 2 if letter in SLICE_LETTERS else 1
406 htm += weight
407 qtm += weight * (2 if amount == 2 else 1)
408 stm += 1
409 return htm, qtm, stm, len(moves)

412def order(table):
413 result = 1
414 seen = set()
415 for start in range(54):
416 length = 0
417 sticker = start
418 while sticker not in seen:
419 seen.add(sticker)
420 sticker = table[sticker]
421 length += 1
422 if length:
423 result = math.lcm(result, length)
424 return result

427def minstd(seed):
428 value = seed
429 while True:
430 value = MULTIPLIER * value % MODULUS
431 yield value

434def scramble(count, seed):
435 numbers = minstd(seed)
436 moves = []
437 for _ in range(count):
438 faces = list(FACES)
439 if moves:
440 last = moves[-1][0]
441 faces.remove(last)
442 if len(moves) >= 2 and AXIS[last] == AXIS[moves[-2][0]]:
443 faces = [face for face in faces if AXIS[face] != AXIS[last]]
444 face = faces[next(numbers) % len(faces)]
445 amount = (1, 3, 2)[next(numbers) % 3]
446 moves.append((face, amount))
447 return moves

450def search(state, limit):
451 for depth in range(limit + 1):
452 found = search_depth(state, depth, None)
453 if found is not None:
454 return found
455 return None
456
457
458def search_depth(state, depth, previous):
459 if depth == 0:
460 return [] if is_solved(state) else None
461 for move in SEARCH_MOVES:
462 face = move[0]
463 if face == previous or BLOCKED_AFTER.get(previous) == face:
464 continue
465 found = search_depth(permute(state, MOVE_TABLES[move]), depth - 1, face)
466 if found is not None:
467 return [move] + found
468 return None

471class SequenceParser:
472 def __init__(self, line, start, macros):
473 self.line = line
474 self.position = start
475 self.macros = macros
476
477 def current(self):
478 if self.position < len(self.line):
479 return self.line[self.position]
480 return None
481
482 def read(self):
483 character = self.current()
484 self.position += 1
485 return character
486
487 def accept(self, character):
488 if self.current() != character:
489 return False
490 self.position += 1
491 return True
492
493 def fail(self):
494 raise CommandError(f"syntax at {self.position + 1}")
495
496 def expect(self, character):
497 if not self.accept(character):
498 self.fail()
499
500 def skip_spaces(self):
501 while self.current() == " ":
502 self.position += 1
503
504 def parse(self):
505 moves = self.sequence()
506 self.skip_spaces()
507 if self.current() is not None:
508 self.fail()
509 require(len(moves) <= MAX_MOVES, "too long")
510 return moves
511
512 def sequence(self):
513 moves = []
514 while True:
515 self.skip_spaces()
516 character = self.current()
517 if character in TURNS:
518 moves.append(self.move())
519 elif character == "(":
520 moves.extend(self.group())
521 elif character == "[":
522 moves.extend(self.bracket())
523 elif character == "{":
524 moves.extend(self.macro())
525 else:
526 return moves
527
528 def move(self):
529 letter = self.read()
530 if self.accept("2"):
531 return letter, 2
532 if self.accept("'"):
533 return letter, 3
534 return letter, 1
535
536 def group(self):
537 self.read()
538 moves = self.sequence()
539 self.skip_spaces()
540 self.expect(")")
541 return self.power(moves)
542
543 def bracket(self):
544 self.read()
545 first = self.sequence()
546 self.skip_spaces()
547 separator = self.current()
548 if separator not in (",", ":"):
549 self.fail()
550 self.read()
551 second = self.sequence()
552 self.skip_spaces()
553 self.expect("]")
554 moves = first + second + inverse(first)
555 if separator == ",":
556 moves += inverse(second)
557 return self.power(moves)
558
559 def macro(self):
560 self.read()
561 start = self.position
562 while self.current() in LOWERCASE:
563 self.position += 1
564 name = self.line[start:self.position]
565 if not name or self.current() != "}":
566 self.fail()
567 if name not in self.macros:
568 raise CommandError(f"undefined {name}")
569 self.read()
570 return self.power(self.macros[name])
571
572 def power(self, moves):
573 count = 1
574 if self.current() in NONZERO_DIGITS:
575 count = int(self.read())
576 if self.current() in DIGITS:
577 count = 10 * count + int(self.read())
578 moves = moves * count
579 return inverse(moves) if self.accept("'") else moves

631 def do_stage(self):
632 normal = normalized(self.state)
633 corners = CORNERS.placed(normal)
634 edges = EDGES.placed(normal)
635 cross = sum(edge in edges for edge in CROSS_EDGES)
636 slots = sum(corner in corners and edge in edges for corner, edge in F2L_SLOTS)
637 top = normal[:9].count("W")
638 if is_solved(normal):
639 stage = "solved"
640 elif cross == 4 and slots == 4 and top == 9:
641 stage = "oll"
642 elif cross == 4 and slots == 4:
643 stage = "f2l"
644 elif cross == 4:
645 stage = "cross"
646 else:
647 stage = "none"
648 print("stage", stage, "cross", cross, "slots", slots, "top", top)

668 def do_undo(self, count):

670 require(self.history, "nothing to undo")
671 steps = min(count, len(self.history))
672 for _ in range(steps):
673 move = self.history.pop()
674 self.state = apply_moves(self.state, inverse([move]))

683 move = self.redo.pop()
684 self.state = apply_moves(self.state, [move])
685 self.history.append(move)
686 print("redone", steps)

770 def run(self, line):
771 name, separator, argument = line.partition(" ")
772 if name in NO_ARGUMENT_COMMANDS:
773 require(not separator)
774 arguments = []
775 elif name in WORD_COMMANDS:
776 words = argument.split(" ") if separator else []
777 patterns = WORD_COMMANDS[name]
778 require(len(words) == len(patterns))
779 require(all(pattern.fullmatch(word) for pattern, word in zip(patterns, words)))
780 arguments = [
781 int(word) if pattern is NUMBER else word for pattern, word in zip(patterns, words)
782 ]
783 elif name in SEQUENCE_COMMANDS:
784 arguments = [self.parse(line, len(name) + 1)]
785 elif name in HEAD_WORD_COMMANDS:
786 require(separator)
787 head = argument.partition(" ")[0]
788 require(HEAD_WORD_COMMANDS[name].fullmatch(head))
789 arguments = [head, self.parse(line, len(name) + len(head) + 2)]
790 else:
791 raise CommandError(f"unknown command {name}")
792 getattr(self, f"do_{name.lower()}")(*arguments)
793
794 def execute(self, line):
795 try:
796 self.run(line)
797 except CommandError as error:
798 print(f"error: {error}")
