50function quarter(v, axis)
51 a=mod1(axis+1, 3)
52 b=mod1(axis+2, 3)
53 ntuple(i->i==a ? v[b] : i==b ? -v[a] : v[i], 3)
54end
55function tables()
56 places=[
57 (ntuple(i->normal[i]+c*column[i]+r*row[i], 3), normal) for (normal, column, row) in FRAMES
58 for r in -1:1 for c in -1:1
59 ]
60 numbers=Dict(place=>i for (i, place) in enumerate(places))
61 result=Dict{Move,Vector{Int}}()
62 for (letter, (axis, turns, layers)) in TURNS
63 perm=collect(1:54)
64 for (i, (position, normal)) in enumerate(places)
65 if position[axis] in layers
66 for _ in 1:turns
67 position=quarter(position, axis)
68 normal=quarter(normal, axis)
69 end
70 perm[numbers[(position, normal)]]=i
71 end
72 end
73 power=collect(1:54)
74 for a in 1:3
75 power=power[perm]
76 result[(letter, a)]=power
77 end
78 end
79 result
80end

151function contents(kind, state)
152 result=Tuple{Int,Int}[]
153 for stickers in kind.stickers
154 colors=state[stickers]
155 found=(0, 0)
156 for (p, target) in enumerate(kind.colors), t in 0:(length(colors)-1)
157 if [colors[mod1(i+t, length(colors))] for i in eachindex(colors)]==target
158 found=(p, t)
159 break
160 end
161 end
162 push!(result, found)
163 end
164 result
165end
166function cycles(kind, state)
167 pieces=contents(kind, state)
168 locations=zeros(Int, length(pieces))
169 turns=zeros(Int, length(pieces))
170 seen=falses(length(pieces))
171 result=String[]
172 for (position, (piece, t)) in enumerate(pieces)
173 locations[piece]=position
174 turns[piece]=t
175 end
176 for start in eachindex(pieces)
177 seen[start] && continue
178 cycle=Int[]
179 p=start
180 while !seen[p]
181 seen[p]=true
182 push!(cycle, p)
183 p=locations[p]
184 end
185 remainder=mod(sum(turns[cycle]), length(kind.stickers[1]))
186 mark=("", "+", "-")[remainder+1]
187 if length(cycle)>=2 || !isempty(mark)
188 push!(result, "("*join(kind.names[cycle], " ")*")"*mark)
189 end
190 end
191 isempty(result) ? "none" : join(result, " ")
192end
193solved(state) = all(i->all(==(state[i+4]), state[i:(i+8)]), 1:9:54)
194function normalization(state)
195 for rotation in ROTATIONS
196 moved=applymoves(state, rotation)
197 all(i->moved[9*(i-1)+5]==COLORS[i], 1:6) && return rotation
198 end
199 nothing
200end
201normalized(state) = applymoves(state, normalization(state))
202parity(pieces) = mod(
203 sum(
204 pieces[i][1]>pieces[j][1] for i in eachindex(pieces) for j in (i+1):length(pieces);
205 init = 0,
206 ),
207 2,
208)
209function invalid(s)
210 length(s)==54 || return "length $(length(s))"
211 for (i, c) in enumerate(s)
212 c in COLORS || return "char $i"
213 end
214 for c in COLORS
215 n=count(==(c), s)
216 n==9 || return "count $c $n"
217 end
218 state=collect(s)
219 rotation=normalization(state)
220 rotation===nothing && return "centers"
221 c=contents(CORNERS, state)
222 e=contents(EDGES, state)
223 for (kind, pieces, label) in ((CORNERS, c, "corner"), (EDGES, e, "edge"))
224 missing=findfirst(x->x[1]==0, pieces)
225 missing!==nothing && return "$label $(kind.names[missing])"
226 end
227 for (kind, pieces) in ((CORNERS, c), (EDGES, e)), (i, name) in enumerate(kind.names)
228 any(x->x[1]==i, pieces) || return "missing $name"
229 end
230 normal=applymoves(state, rotation)
231 c=contents(CORNERS, normal)
232 e=contents(EDGES, normal)
233 mod(sum(last, c), 3)==0 || return "twist"
234 mod(sum(last, e), 2)==0 || return "flip"
235 parity(c)==parity(e) || return "parity"
236 nothing
237end
238function simplify(moves)
239 while true
240 result=Move[]
241 at=1
242 while at<=length(moves)
243 axis=AXIS[moves[at][1]]
244 letters=Char[]
245 totals=Dict{Char,Int}()
246 while at<=length(moves) && AXIS[moves[at][1]]==axis
247 c, a=moves[at]
248 !haskey(totals, c) && push!(letters, c)
249 totals[c]=mod(get(totals, c, 0)+a, 4)
250 at+=1
251 end
252 append!(result, [(c, totals[c]) for c in letters if totals[c]!=0])
253 end
254 result==moves && return result
255 moves=result
256 end
257end
258function unrotate(moves)
259 expanded=Move[]
260 for (c, a) in moves
261 append!(
262 expanded,
263 haskey(REPLACEMENTS, c) ? [(d, mod(b*a, 4)) for (d, b) in REPLACEMENTS[c]] : [(c, a)],
264 )
265 end
266 held=Move[]
267 result=Move[]
268 for move in expanded
269 if move[1] in "xyz"
270 held=ROTATIONEFFECT[Tuple(effect([held; [move]]))]
271 else
272 push!(result, BYEFFECT[Tuple(effect([held; [move]; inverse(held)]))])
273 end
274 end
275 [result; held]
276end
277function permutationorder(table)
278 seen=falses(54)
279 result=1
280 for start in 1:54
281 n=0
282 p=start
283 while !seen[p]
284 seen[p]=true
285 n+=1
286 p=table[p]
287 end
288 n>0 && (result=lcm(result, n))
289 end
290 result
291end
292function scramble(n, seed)
293 result=Move[]
294 for _ in 1:n
295 faces=[c for c in FACES if isempty(result) || c!=last(result)[1]]
296 if length(result)>=2 && AXIS[result[end][1]]==AXIS[result[end-1][1]]
297 filter!(c->AXIS[c]!=AXIS[result[end][1]], faces)
298 end
299 seed=mod(48271*seed, 2147483647)
300 c=faces[mod(seed, length(faces))+1]
301 seed=mod(48271*seed, 2147483647)
302 push!(result, (c, (1, 3, 2)[mod(seed, 3)+1]))
303 end
304 result
305end
306function searchdepth(state, depth, previous = ' ')
307 depth==0 && return solved(state) ? Move[] : nothing
308 for c in FACES, a in 1:3
309 (
310 c==previous ||
311 (previous=='D' && c=='U') ||
312 (previous=='L' && c=='R') ||
313 (previous=='B' && c=='F')
314 ) && continue
315 found=searchdepth(state[TABLES[(c, a)]], depth-1, c)
316 found!==nothing && return [[(c, a)]; found]
317 end
318 nothing
319end
320mutable struct Parser
321 line::String
322 at::Int
323 macros::Dict{String,Vector{Move}}
324end
325current(p) = p.at<=length(p.line) ? p.line[p.at] : '\0'
326function accept!(p, c)
327 current(p)==c || return false
328 p.at+=1
329 true
330end
331syntax(p) = fail("syntax at $(min(p.at,length(p.line)+1))")
332skip!(p) = begin
333 while current(p)==' '
334 p.at+=1
335 end
336end
337function power!(p, moves)
338 n=1
339 if '1'<=current(p)<='9'
340 n=Int(current(p)-'0')
341 p.at+=1
342 if '0'<=current(p)<='9'
343 n=10n+Int(current(p)-'0')
344 p.at+=1
345 end
346 end
347 result=repeat(moves, n)
348 accept!(p, '\'') ? inverse(result) : result
349end
350function readsequence!(p)
351 result=Move[]
352 while true
353 skip!(p)
354 c=current(p)
355 if c in LETTERS
356 p.at+=1
357 a=accept!(p, '2') ? 2 : accept!(p, '\'') ? 3 : 1
358 push!(result, (c, a))
359 elseif c=='('
360 p.at+=1
361 moves=readsequence!(p)
362 skip!(p)
363 accept!(p, ')') || syntax(p)
364 append!(result, power!(p, moves))
365 elseif c=='['
366 p.at+=1
367 first=readsequence!(p)
368 skip!(p)
369 separator=current(p)
370 separator in (',', ':') || syntax(p)
371 p.at+=1
372 second=readsequence!(p)
373 skip!(p)
374 accept!(p, ']') || syntax(p)
375 append!(
376 result,
377 power!(
378 p,
379 [first; second; inverse(first); separator==',' ? inverse(second) : Move[]],
380 ),
381 )
382 elseif c=='{'
383 p.at+=1
384 start=p.at
385 while 'a'<=current(p)<='z'
386 p.at+=1
387 end
388 name=part(p.line, start, p.at-1)
389 !isempty(name) && current(p)=='}' || syntax(p)
390 haskey(p.macros, name) || fail("undefined $name")
391 p.at+=1
392 append!(result, power!(p, p.macros[name]))
393 else
394 return result
395 end
396 end
397end
398function parse_sequence(line, start, macros)
399 p=Parser(line, start, macros)
400 result=readsequence!(p)
401 skip!(p)
402 current(p)=='\0' || syntax(p)
403 require(length(result)<=500, "too long")
404 result
405end
406mutable struct Simulator
407 state::Vector{Char}
408 history::Vector{Move}
409 redo::Vector{Move}
410 macros::Dict{String,Vector{Move}}
411 saves::Dict{String,Tuple{Vector{Char},Vector{Move}}}
412end
413Simulator() = Simulator(
414 copy(SOLVED),
415 Move[],
416 Move[],
417 Dict{String,Vector{Move}}(),
418 Dict{String,Tuple{Vector{Char},Vector{Move}}}(),
419)
420function record!(s, moves)
421 s.state=applymoves(s.state, moves)
422 append!(s.history, moves)
423 empty!(s.redo)
424end
425function execute!(s, line)
426 splitat=findfirst(' ', line)
427 cmd=splitat===nothing ? line : part(line, 1, splitat-1)
428 argument=splitat===nothing ? "" : part(line, splitat+1)
429 noargs=(
430 "RESET",
431 "SHOW",
432 "STATE",
433 "ISSOLVED",
434 "HISTORY",
435 "SOLUTION",
436 "ANALYZE",
437 "STAGE",
438 "MACROS",
439 )
440 wordcounts=Dict(
441 "CHECK"=>1,
442 "LOAD"=>1,
443 "UNDO"=>1,
444 "REDO"=>1,
445 "SEARCH"=>1,
446 "WHERE"=>1,
447 "SAVE"=>1,
448 "RESTORE"=>1,
449 "SCRAMBLE"=>2,
450 )
451 seqcmds=("APPLY", "INVERT", "SIMPLIFY", "METRICS", "ORDER", "CYCLES", "UNROTATE", "UNTIL")
452 words=String[]
453 head=""
454 moves=Move[]
455 if cmd in noargs
456 require(splitat===nothing)
457 elseif haskey(wordcounts, cmd)
458 words=splitat===nothing ? String[] : String.(split(argument, ' '; keepempty = true))
459 require(length(words)==wordcounts[cmd])
460 if cmd in ("UNDO", "REDO", "SEARCH", "SCRAMBLE")
461 require(all(w->occursin(r"^(?:0|[1-9][0-9]{0,9})$", w), words))
462 elseif cmd in ("SAVE", "RESTORE")
463 require(occursin(r"^[a-z]{1,8}$", words[1]))
464 end
465 elseif cmd in seqcmds
466 moves=parse_sequence(line, length(cmd)+2, s.macros)
467 elseif cmd in ("DEFINE", "MIRROR", "ROTATE")
468 require(splitat!==nothing)
469 at=findfirst(' ', argument)
470 head=at===nothing ? argument : part(argument, 1, at-1)
471 require(
472 cmd=="DEFINE" ? occursin(r"^[a-z]{1,8}$", head) :
473 cmd=="MIRROR" ? head in ("LR", "UD", "FB") : occursin(r"^[xyz][2']?$", head),
474 )
475 moves=parse_sequence(line, length(cmd)+length(head)+3, s.macros)
476 else
477 fail("unknown command $cmd")
478 end
479 if cmd=="RESET"
480 s.state=copy(SOLVED)
481 empty!(s.history)
482 empty!(s.redo)
483 println("reset")
484 elseif cmd=="SHOW"
485 row(face, r) =
486 s.state[(9*(findfirst(==(face), FACES)-1)+3r+1):(9*(findfirst(==(face), FACES)-1)+3r+3)]
487 for r in 0:2
488 println(" ", join(row('U', r), ' '))
489 end
490 for r in 0:2
491 println(join([row('L', r); row('F', r); row('R', r); row('B', r)], ' '))
492 end
493 for r in 0:2
494 println(" ", join(row('D', r), ' '))
495 end
496 elseif cmd=="STATE"
497 println("state ", join(s.state))
498 elseif cmd=="ISSOLVED"
499 println(
500 solved(s.state) ? "solved" :
501 "unsolved $(count(i->s.state[i]!=s.state[9*fld(i-1,9)+5],1:54))",
502 )
503 elseif cmd=="HISTORY"
504 println("history ", sequence(s.history))
505 elseif cmd=="SOLUTION"
506 println("solution ", sequence(simplify(inverse(s.history))))
507 elseif cmd=="MACROS"
508 println("macros $(length(s.macros))")
509 for name in sort!(collect(keys(s.macros)))
510 println("macro $name $(sequence(s.macros[name]))")
511 end
512 elseif cmd=="ANALYZE"
513 rotation=normalization(s.state)
514 normal=applymoves(s.state, rotation)
515 println("rotation ", isempty(rotation) ? "none" : movestext(rotation))
516 println("corners ", cycles(CORNERS, normal))
517 println("edges ", cycles(EDGES, normal))
518 elseif cmd=="STAGE"
519 normal=normalized(s.state)
520 corners=contents(CORNERS, normal)
521 edges=contents(EDGES, normal)
522 cross=count(i->edges[i]==(i, 0), 5:8)
523 slots=count(i->corners[i+4]==(i+4, 0) && edges[i+8]==(i+8, 0), 1:4)
524 top=count(==('W'), normal[1:9])
525 stage=solved(normal) ? "solved" :
526 cross==4 && slots==4 && top==9 ? "oll" :
527 cross==4 && slots==4 ? "f2l" : cross==4 ? "cross" : "none"
528 println("stage $stage cross $cross slots $slots top $top")
529 elseif cmd in ("CHECK", "LOAD")
530 reason=invalid(words[1])
531 if cmd=="CHECK"
532 println(reason===nothing ? "valid" : "invalid $reason")
533 else
534 reason===nothing || fail("invalid $reason")
535 s.state=collect(words[1])
536 empty!(s.history)
537 empty!(s.redo)
538 println("loaded")
539 end
540 elseif cmd in ("UNDO", "REDO")
541 n=parse(Int, words[1])
542 require(n>=1, "bad value")
543 source=cmd=="UNDO" ? s.history : s.redo
544 target=cmd=="UNDO" ? s.redo : s.history
545 require(!isempty(source), cmd=="UNDO" ? "nothing to undo" : "nothing to redo")
546 n=min(n, length(source))
547 for _ in 1:n
548 move=pop!(source)
549 s.state=applymoves(s.state, cmd=="UNDO" ? inverse([move]) : [move])
550 push!(target, move)
551 end
552 println(cmd=="UNDO" ? "undone $n" : "redone $n")
553 elseif cmd=="SEARCH"
554 n=parse(Int, words[1])
555 require(n<=4, "bad value")
556 found=nothing
557 for depth in 0:n
558 found=searchdepth(s.state, depth)
559 found!==nothing && break
560 end
561 println(found===nothing ? "not found" : "found $(sequence(found))")
562 elseif cmd=="WHERE"
563 name=words[1]
564 kind=name in CORNERS.names ? CORNERS : name in EDGES.names ? EDGES : nothing
565 kind===nothing && fail("unknown piece $name")
566 piece=findfirst(==(name), kind.names)
567 pieces=contents(kind, normalized(s.state))
568 position=findfirst(x->x[1]==piece, pieces)
569 println(
570 "$name at $(kind.names[position]) $(kind===CORNERS ? "twist" : "flip") $(pieces[position][2])",
571 )
572 elseif cmd=="SAVE"
573 name=words[1]
574 s.saves[name]=(copy(s.state), copy(s.history))
575 println("saved $name")
576 elseif cmd=="RESTORE"
577 name=words[1]
578 require(haskey(s.saves, name), "no save $name")
579 state, history=s.saves[name]
580 s.state=copy(state)
581 s.history=copy(history)
582 empty!(s.redo)
583 println("restored $name")
584 elseif cmd=="SCRAMBLE"
585 n, seed=parse.(Int, words)
586 require(1<=n<=50 && 1<=seed<=2147483646, "bad value")
587 moves=scramble(n, seed)
588 record!(s, moves)
589 println("scramble ", movestext(moves))
590 elseif cmd=="APPLY"
591 record!(s, moves)
592 println("applied $(length(moves))")
593 elseif cmd=="INVERT"
594 println("inverse ", sequence(inverse(moves)))
595 elseif cmd=="SIMPLIFY"
596 println("simplified ", sequence(simplify(moves)))
597 elseif cmd=="METRICS"
598 htm=0
599 qtm=0
600 stm=0
601 for (c, a) in moves
602 c in "xyz" && continue
603 w=c in "MES" ? 2 : 1
604 htm+=w
605 qtm+=w*(a==2 ? 2 : 1)
606 stm+=1
607 end
608 println("metrics htm $htm qtm $qtm stm $stm etm $(length(moves))")
609 elseif cmd=="ORDER"
610 println("order ", permutationorder(effect(moves)))
611 elseif cmd=="CYCLES"
612 state=applymoves(SOLVED, moves)
613 println("corners ", cycles(CORNERS, state))
614 println("edges ", cycles(EDGES, state))
615 println("centers ", cycles(CENTERS, state))
616 elseif cmd=="UNROTATE"
617 println("unrotated ", sequence(unrotate(moves)))
618 elseif cmd=="UNTIL"
619 table=effect(moves)
620 state=s.state
621 found=0
622 for i in 1:permutationorder(table)
623 state=state[table]
624 if solved(state)
625 found=i
626 break
627 end
628 end
629 println(found==0 ? "until never" : "until $found")
630 elseif cmd=="DEFINE"
631 s.macros[head]=copy(moves)
632 println("defined $head $(length(moves))")
633 elseif cmd=="MIRROR"
634 axis=head=="LR" ? 1 : head=="UD" ? 2 : 3
635 result=Move[
636 AXIS[c]!=axis ? (c, 4-a) : haskey(OPPOSITE, c) ? (OPPOSITE[c], 4-a) : (c, a) for
637 (c, a) in moves
638 ]
639 println("mirror ", sequence(result))
640 elseif cmd=="ROTATE"
641 rotation=plainmoves(head)
642 result=Move[
643 BYEFFECT[Tuple(effect([rotation; [move]; inverse(rotation)]))] for move in moves
644 ]
645 println("rotated ", sequence(result))
646 end
647end
648function main()
649 s=Simulator()
650 n=parse(Int, readline())
651 for _ in 1:n
652 line=readline()
653 try
654 execute!(s, line)
655 catch e
656 e isa CommandError || rethrow()
657 println("error: $(e.message)")
658 end
659 end
660end
