const FACES="URFDLB"
const AXIS=Dict(c=>i for (i, group) in enumerate(("RLMrlx", "UDEudy", "FBSfbz")) for c in group)
movetext(move) = string(move[1])*(move[2]==1 ? "" : move[2]==2 ? "2" : "'")
movestext(moves) = join(movetext.(moves), " ")
function applymoves(state, moves)
    for move in moves
        state=state[TABLES[move]]
    end
    state
end
function scramble(n, seed)
    result=Move[]
    for _ in 1:n
        faces=[c for c in FACES if isempty(result) || c!=last(result)[1]]
        if length(result)>=2 && AXIS[result[end][1]]==AXIS[result[end-1][1]]
            filter!(c->AXIS[c]!=AXIS[result[end][1]], faces)
        end
        seed=mod(48271*seed, 2147483647)
        c=faces[mod(seed, length(faces))+1]
        seed=mod(48271*seed, 2147483647)
        push!(result, (c, (1, 3, 2)[mod(seed, 3)+1]))
    end
    result
end
function record!(s, moves)
    s.state=applymoves(s.state, moves)
    append!(s.history, moves)
    empty!(s.redo)
