228static Vector rotate_vector(Vector v, int axis) {
229 if (axis == 0) return (Vector){{v.v[0], -v.v[2], v.v[1]}};
230 if (axis == 1) return (Vector){{v.v[2], v.v[1], -v.v[0]}};

238static void initialize(void) {
239 Vector positions[54], normals[54], face_normals[6] = {
240 {{0, 1, 0}}, {{1, 0, 0}}, {{0, 0, 1}},
241 {{0, -1, 0}}, {{-1, 0, 0}}, {{0, 0, -1}}
242 };
243 for (int face = 0; face < 6; face++) for (int row = 0; row < 3; row++) for (int col = 0; col < 3; col++) {
244 int i = face * 9 + row * 3 + col;
245 normals[i] = face_normals[face];
246 switch (face) {
247 case 0:
248 positions[i] = (Vector) {
249 {col-1,1,row-1}
250 };
251 break;
252 case 1:
253 positions[i] = (Vector) {
254 {1,1-row,1-col}
255 };
256 break;
257 case 2:
258 positions[i] = (Vector) {
259 {col-1,1-row,1}
260 };
261 break;
262 case 3:
263 positions[i] = (Vector) {
264 {col-1,-1,1-row}
265 };
266 break;
267 case 4:
268 positions[i] = (Vector) {
269 {-1,1-row,col-1}
270 };
271 break;
272 default:
273 positions[i] = (Vector) {
274 {1-col,1-row,-1}
275 };
276 }
277 }
278 for (int m = 0; m < 18; m++) {
279 char letter = letters[m];
280 int ax = axis(letter), direction = in_letters("URFSurfxyz", letter) ? -1 : 1, layer = in_letters("MES", letter) ? 0 : in_letters("URFurf", letter) ? 1 : -1;
281 bool all = in_letters("xyz", letter), wide = in_letters("urfdlb", letter);
282 Permutation p = identity();
283 for (int i = 0; i < 54; i++) {
284 if (!(all || (wide ? positions[i].v[ax] * layer >= 0 : positions[i].v[ax] == layer))) continue;
285 Vector pos = positions[i], normal = normals[i];
286 for (int n = 0; n < (direction == 1 ? 1 : 3); n++) {
287 pos = rotate_vector(pos, ax);
288 normal = rotate_vector(normal, ax);
289 }
290 for (int j = 0; j < 54; j++) if (same_vector(pos, positions[j]) && same_vector(normal, normals[j])) {
291 p.to[i] = j;
292 break;
293 }
294 }
295 Permutation total = identity();
296 for (int amount = 1; amount <= 3; amount++) {
297 total = compose(total, p);
298 permutations[m][amount] = total;
299 }
300 }
301 for (int base = 0; base < 6; base++) for (int y = 0; y < 4; y++) {
302 Sequence *rotation = rotations + base * 4 + y;
303 if (base) APPEND(rotation->items, rotation->count, ((Move){base < 4 ? 'x' : 'z', base < 4 ? base : base == 4 ? 1 : 3}));
304 if (y) APPEND(rotation->items, rotation->count, ((Move){'y', y}));
305 }
306 state = solved_state();
307}

327static Sequence power(Parser *p, Sequence sequence) {
328 int count = 1;
329 if (peek(p) >= '1' && peek(p) <= '9') {
330 count = peek(p) - '0';
331 p->position++;
332 if (peek(p) >= '0' && peek(p) <= '9') {
333 count = 10 * count + peek(p) - '0';
334 p->position++;
335 }
336 }
337 bool reverse = peek(p) == '\'';
338 if (reverse) p->position++;
339 Sequence out = {0};
340 for (int i = 0; i < count; i++) extend(&out, sequence);
341 return reverse ? invert(out) : out;
342}

344static Sequence parse_sequence(Parser *p) {
345 Sequence out = {0};
346 while (true) {
347 spaces(p);
348 char c = peek(p);
349 if (in_letters(letters, c)) {
350 p->position++;
351 int amount = peek(p) == '2' ? 2 : peek(p) == '\'' ? 3 : 1;
352 if (amount != 1) p->position++;
353 APPEND(out.items, out.count, ((Move){c, amount}));
354 } else if (c == '(') {
355 p->position++;
356 Sequence group = parse_sequence(p);
357 spaces(p);
358 if (peek(p) != ')') syntax(p);
359 p->position++;
360 extend(&out, power(p, group));
361 } else if (c == '[') {
362 p->position++;
363 Sequence first = parse_sequence(p);
364 spaces(p);
365 char separator = peek(p);
366 if (separator != ',' && separator != ':') syntax(p);
367 p->position++;
368 Sequence second = parse_sequence(p);
369 spaces(p);
370 if (peek(p) != ']') syntax(p);
371 p->position++;
372 Sequence group = clone_sequence(first);
373 extend(&group, second);
374 extend(&group, invert(first));
375 if (separator == ',') extend(&group, invert(second));
376 extend(&out, power(p, group));
377 } else if (c == '{') {
378 p->position++;
379 size_t start = p->position;
380 while (peek(p) >= 'a' && peek(p) <= 'z') p->position++;
381 char *name = slice(p->line + start, p->position - start);
382 if (!*name || peek(p) != '}') syntax(p);
383 Macro *macro = find_macro(name);
384 if (!macro) error("error: undefined %s", name);
385 p->position++;
386 extend(&out, power(p, macro->sequence));
387 } else return out;
388 }
389}

446static char *check(const char *text) {
447 size_t length = strlen(text);
448 if (length != 54) return format("length %zu", length);
449 for (int i = 0; i < 54; i++) if (!strchr(colors, text[i])) return format("char %d", i + 1);
450 for (int c = 0; c < 6; c++) {
451 int count = 0;
452 for (int i = 0; i < 54; i++) if (text[i] == colors[c]) count++;
453 if (count != 9) return format("count %c %d", colors[c], count);
454 }
455 Cube input, normalized;
456 memcpy(input.stickers, text, 55);
457 if (normalize(input, &normalized) < 0) return "centers";
458 Pieces c = corners(input), e = edges(input);
459 for (int i = 0; i < 8; i++) if (c.piece[i] < 0) return format("corner %s", corner_names[i]);
460 for (int i = 0; i < 12; i++) if (e.piece[i] < 0) return format("edge %s", edge_names[i]);
461 for (int i = 0; i < 8; i++) {
462 bool found = false;
463 for (int q = 0; q < 8; q++) if (c.piece[q] == i) found = true;
464 if (!found) return format("missing %s", corner_names[i]);
465 }
466 for (int i = 0; i < 12; i++) {
467 bool found = false;
468 for (int q = 0; q < 12; q++) if (e.piece[q] == i) found = true;
469 if (!found) return format("missing %s", edge_names[i]);
470 }
471 c = corners(normalized);
472 e = edges(normalized);
473 int twist = 0, flip = 0;
474 for (int i = 0; i < 8; i++) twist += c.orientation[i];
475 for (int i = 0; i < 12; i++) flip += e.orientation[i];
476 if (twist % 3) return "twist";
477 if (flip % 2) return "flip";
478 return parity(c) != parity(e) ? "parity" : NULL;
479}

481static char *cycle_text(Pieces pieces, const char **names, int base) {
482 int destination[12] = {0}, orientation[12] = {0};
483 bool seen[12] = {0};
484 for (int q = 0; q < pieces.count; q++) {
485 destination[pieces.piece[q]] = q;
486 orientation[pieces.piece[q]] = pieces.orientation[q];
487 }
488 Buffer out = {0};
489 for (int start = 0; start < pieces.count; start++) if (!seen[start]) {
490 Buffer cycle = {0};
491 int p = start, count = 0, sum = 0;
492 do {
493 seen[p] = true;
494 if (count++) append(&cycle, " ");
495 append(&cycle, names[p]);
496 sum += orientation[p];
497 p = destination[p];
498 }
499 while (p != start);
500 char *mark = base == 3 ? sum % 3 == 1 ? "+" : sum % 3 == 2 ? "-" : "" : base == 2 && sum % 2 ? "+" : "";
501 if (count >= 2 || *mark) {
502 if (out.length) append(&out, " ");
503 append(&out, format("(%s)%s", finish(&cycle), mark));
504 }
505 }
506 return out.length ? finish(&out) : "none";
507}

525static Sequence simplify(Sequence input) {
526 Sequence moves = input;
527 while (true) {
528 Sequence out = {0};
529 for (size_t i = 0; i < moves.count;) {
530 int ax = axis(moves.items[i].letter), totals[18] = {0};
531 char order[18];
532 size_t count = 0;
533 while (i < moves.count && axis(moves.items[i].letter) == ax) {
534 Move move = moves.items[i++];
535 int index = letter_index(move.letter);
536 bool present = false;
537 for (size_t j = 0; j < count; j++) if (order[j] == move.letter) present = true;
538 if (!present) order[count++] = move.letter;
539 totals[index] += move.amount;
540 }
541 for (size_t j = 0; j < count; j++) {
542 int amount = totals[letter_index(order[j])] % 4;
543 if (amount) APPEND(out.items, out.count, ((Move){order[j], amount}));
544 }
545 }
546 if (same_sequence(out, moves)) return out;
547 moves = out;
548 }
549}

552 Permutation target = compose(compose(rotation, move_permutation(move)), inverse(rotation));
553 for (int i = 0; i < 18; i++) for (int amount = 1; amount <= 3; amount++) if (same_permutation(target, permutations[i][amount])) return (Move){letters[i], amount};
554 exit(EXIT_FAILURE);
555}

557static Sequence replacement(Move move) {
558 const char *text = NULL;
559 int amounts[3] = {1,1,1};
560 switch (move.letter) {
561 case 'M':
562 text = "RLx";
563 amounts[1] = amounts[2] = 3;
564 break;
565 case 'E':
566 text = "UDy";
567 amounts[1] = amounts[2] = 3;
568 break;
569 case 'S':
570 text = "FBz";
571 amounts[0] = 3;
572 break;
573 case 'r':
574 text = "Lx";
575 break;
576 case 'l':
577 text = "Rx";
578 amounts[1] = 3;
579 break;
580 case 'u':
581 text = "Dy";
582 break;
583 case 'd':
584 text = "Uy";
585 amounts[1] = 3;
586 break;
587 case 'f':
588 text = "Bz";
589 break;
590 case 'b':
591 text = "Fz";
592 amounts[1] = 3;
593 break;
594 default:
595 break;
596 }
597 Sequence out = {0};
598 if (!text) APPEND(out.items, out.count, move);
599 else for (size_t i = 0; text[i]; i++) APPEND(out.items, out.count, ((Move){text[i], amounts[i] * move.amount % 4}));
600 return out;
601}

603static Sequence unrotate(Sequence input) {
604 Permutation rotation = identity();
605 Sequence out = {0};
606 for (size_t i = 0; i < input.count; i++) {
607 Sequence expanded = replacement(input.items[i]);
608 for (size_t j = 0; j < expanded.count; j++) {
609 Move move = expanded.items[j];
610 if (in_letters("xyz", move.letter)) rotation = compose(rotation, move_permutation(move));
611 else APPEND(out.items, out.count, conjugate(rotation, move));
612 }
613 }
614 for (int i = 0; i < 24; i++) if (same_permutation(effect(rotations[i]), rotation)) {
615 extend(&out, rotations[i]);
616 break;
617 }
618 return out;
619}

621static uint64_t gcd(uint64_t a, uint64_t b) {
622 while (b) {
623 uint64_t remainder = a % b;
624 a = b;
625 b = remainder;
626 }
627 return a;
628}

630static uint64_t order(Permutation p) {
631 bool seen[54] = {0};
632 uint64_t result = 1;
633 for (int i = 0; i < 54; i++) if (!seen[i]) {
634 int count = 0, at = i;
635 do {
636 seen[at] = true;
637 count++;
638 at = p.to[at];
639 }
640 while (at != i);
641 result = result / gcd(result, (uint64_t)count) * (uint64_t)count;
642 }
643 return result;
644}

646static bool search(Cube input, int left, int previous, Sequence *path) {
647 if (solved(input)) return true;
648 if (!left) return false;
649 for (int face = 0; face < 6; face++) {
650 if (face == previous || (previous == 3 && face == 0) || (previous == 4 && face == 1) || (previous == 5 && face == 2)) continue;
651 for (int amount = 1; amount <= 3; amount++) {
652 Move move = {faces[face], amount};
653 APPEND(path->items, path->count, move);
654 if (search(apply_permutation(input, move_permutation(move)), left - 1, face, path)) return true;
655 path->count--;
656 }
657 }
658 return false;

671static uint64_t number(const char *word) {
672 size_t n = strlen(word);
673 if (!n || n > 10 || (n > 1 && *word == '0')) error("error: bad arguments");
674 uint64_t value = 0;
675 for (size_t i = 0; i < n; i++) {
676 if (word[i] < '0' || word[i] > '9') error("error: bad arguments");
677 value = value * 10 + (unsigned)(word[i] - '0');
678 }
679 return value;
680}

683 size_t n = strlen(name);
684 if (!n || n > 8) return false;
685 for (size_t i = 0; i < n; i++) if (name[i] < 'a' || name[i] > 'z') return false;
686 return true;

698static char **words(const char *s, size_t *count) {
699 char **out = NULL;
700 *count = 0;
701 if (!s) return out;
702 size_t start = 0;
703 for (size_t i = 0;; i++) if (!s[i] || s[i] == ' ') {
704 APPEND(out, *count, slice(s + start, i - start));
705 start = i + 1;
706 if (!s[i]) break;
707 }
708 return out;
709}

711static void command(const char *line) {
712 const char *space = strchr(line, ' '), *argument = space ? space + 1 : NULL;
713 char *command = space ? slice(line, (size_t)(space - line)) : copy_text(line);
714 bool no_args = one_of(command, " RESET SHOW STATE ISSOLVED HISTORY SOLUTION ANALYZE STAGE MACROS "), word_command = one_of(command, " CHECK LOAD UNDO REDO SEARCH WHERE SAVE RESTORE SCRAMBLE "), sequence_command = one_of(command, " APPLY INVERT SIMPLIFY METRICS ORDER CYCLES UNROTATE UNTIL "), head_command = one_of(command, " DEFINE MIRROR ROTATE ");
715 if (!no_args && !word_command && !sequence_command && !head_command) error("error: unknown command %s", command);
716 size_t argc;
717 char **args = words(argument, &argc);
718 char *head = NULL;
719 Sequence moves = {0};
720 uint64_t n = 0, seed = 0;
721 if (no_args && argument) error("error: bad arguments");
722 if (word_command) {
723 if (argc != (size_t)(!strcmp(command, "SCRAMBLE") ? 2 : 1)) error("error: bad arguments");
724 if (one_of(command, " UNDO REDO SEARCH SCRAMBLE ")) n = number(args[0]);
725 if (!strcmp(command, "SCRAMBLE")) seed = number(args[1]);
726 if (one_of(command, " SAVE RESTORE ") && !valid_name(args[0])) error("error: bad arguments");
727 }
728 if (head_command) {
729 if (!argument) error("error: bad arguments");
730 head = args[0];
731 bool valid = !strcmp(command, "DEFINE") ? valid_name(head) : !strcmp(command, "MIRROR") ? one_of(head, " LR UD FB ") : (strlen(head) == 1 || (strlen(head) == 2 && (head[1] == '2' || head[1] == '\''))) && in_letters("xyz", head[0]);
732 if (!valid) error("error: bad arguments");
733 moves = parse(line, strlen(command) + strlen(head) + 2);
734 } else if (sequence_command) moves = parse(line, strlen(command) + 1);
735 if (one_of(command, " UNDO REDO ") && n < 1) error("error: bad value");
736 if (!strcmp(command, "SEARCH") && n > 4) error("error: bad value");
737 if (!strcmp(command, "SCRAMBLE") && (n < 1 || n > 50 || seed < 1 || seed > 2147483646ULL)) error("error: bad value");
738 if (!strcmp(command, "WHERE") && piece_index(args[0], corner_names, 8) < 0 && piece_index(args[0], edge_names, 12) < 0) error("error: unknown piece %s", args[0]);
739 if (!strcmp(command, "RESET")) {
740 state = solved_state();
741 history.count = redo.count = 0;
742 puts("reset");
743 } else if (!strcmp(command, "SHOW")) {
744 for (int row = 0; row < 3; row++) printf(" %c %c %c\n", state.stickers[row * 3], state.stickers[row * 3 + 1], state.stickers[row * 3 + 2]);
745 int net[] = {4,2,1,5};
746 for (int row = 0; row < 3; row++) {
747 bool first = true;
748 for (int f = 0; f < 4; f++) for (int col = 0; col < 3; col++) {
749 if (!first) putchar(' ');
750 putchar(state.stickers[net[f] * 9 + row * 3 + col]);
751 first = false;
752 }
753 putchar('\n');
754 }
755 for (int row = 0; row < 3; row++) printf(" %c %c %c\n", state.stickers[27 + row * 3], state.stickers[28 + row * 3], state.stickers[29 + row * 3]);
756 } else if (!strcmp(command, "STATE")) printf("state %s\n", state.stickers);
757 else if (!strcmp(command, "ISSOLVED")) {
758 int count = 0;
759 for (int i = 0; i < 54; i++) if (state.stickers[i] != state.stickers[(i / 9) * 9 + 4]) count++;
760 if (!count) puts("solved");
761 else printf("unsolved %d\n", count);
762 } else if (!strcmp(command, "HISTORY")) printf("history %s\n", sequence_text(history));
763 else if (!strcmp(command, "SOLUTION")) printf("solution %s\n", sequence_text(simplify(invert(history))));
764 else if (!strcmp(command, "ANALYZE")) {
765 Cube normalized;
766 int rotation = normalize(state, &normalized);
767 printf("rotation %s\n", rotations[rotation].count ? moves_text(rotations[rotation]) : "none");
768 print_cycles(normalized, false);
769 } else if (!strcmp(command, "STAGE")) {
770 Cube normalized;
771 normalize(state, &normalized);
772 Pieces c = corners(normalized), e = edges(normalized);
773 int cross = 0, slots = 0, top = 0;
774 for (int i = 4; i < 8; i++) if (e.piece[i] == i && !e.orientation[i]) cross++;
775 for (int i = 0; i < 4; i++) if (c.piece[i + 4] == i + 4 && !c.orientation[i + 4] && e.piece[i + 8] == i + 8 && !e.orientation[i + 8]) slots++;
776 for (int i = 0; i < 9; i++) if (normalized.stickers[i] == 'W') top++;
777 char *stage = solved(normalized) ? "solved" : cross == 4 && slots == 4 && top == 9 ? "oll" : cross == 4 && slots == 4 ? "f2l" : cross == 4 ? "cross" : "none";
778 printf("stage %s cross %d slots %d top %d\n", stage, cross, slots, top);
779 } else if (!strcmp(command, "MACROS")) {
780 printf("macros %zu\n", macro_count);
781 Macro **ordered = allocate(macro_count * sizeof(*ordered));
782 for (size_t i = 0; i < macro_count; i++) ordered[i] = macros + i;
783 qsort(ordered, macro_count, sizeof(*ordered), macro_compare);
784 for (size_t i = 0; i < macro_count; i++) printf("macro %s %s\n", ordered[i]->name, sequence_text(ordered[i]->sequence));
785 } else if (!strcmp(command, "CHECK") || !strcmp(command, "LOAD")) {
786 char *reason = check(args[0]);
787 if (!strcmp(command, "CHECK")) {
788 if (reason) printf("invalid %s\n", reason);
789 else puts("valid");
790 } else {
791 if (reason) error("error: invalid %s", reason);
792 memcpy(state.stickers, args[0], 55);
793 history.count = redo.count = 0;
794 puts("loaded");
795 }
796 } else if (!strcmp(command, "UNDO") || !strcmp(command, "REDO")) {
797 bool undo = !strcmp(command, "UNDO");
798 Sequence *source = undo ? &history : &redo, *target = undo ? &redo : &history;
799 if (!source->count) error("error: nothing to %s", undo ? "undo" : "redo");
800 size_t count = n < source->count ? (size_t)n : source->count;
801 for (size_t i = 0; i < count; i++) {
802 Move move = source->items[--source->count];
803 state = apply_permutation(state, move_permutation(undo ? inverse_move(move) : move));
804 APPEND(target->items, target->count, move);
805 }
806 printf("%s %zu\n", undo ? "undone" : "redone", count);
807 } else if (!strcmp(command, "SEARCH")) {
808 Sequence solution = {0};
809 bool found = false;
810 for (int depth = 0; depth <= (int)n; depth++) {
811 solution.count = 0;
812 if (search(state, depth, -1, &solution)) {
813 found = true;
814 break;
815 }
816 }
817 if (found) printf("found %s\n", sequence_text(solution));
818 else puts("not found");
819 } else if (!strcmp(command, "WHERE")) {
820 Cube normalized;
821 normalize(state, &normalized);
822 int piece = piece_index(args[0], corner_names, 8);
823 bool corner = piece >= 0;
824 if (!corner) piece = piece_index(args[0], edge_names, 12);
825 Pieces pieces = corner ? corners(normalized) : edges(normalized);
826 const char **names = corner ? corner_names : edge_names;
827 for (int q = 0; q < pieces.count; q++) if (pieces.piece[q] == piece) printf("%s at %s %s %d\n", args[0], names[q], corner ? "twist" : "flip", pieces.orientation[q]);
828 } else if (!strcmp(command, "SAVE")) {
829 Save *save = NULL;
830 for (size_t i = 0; i < save_count; i++) if (!strcmp(saves[i].name, args[0])) save = saves + i;
831 if (save) *save = (Save){args[0], state, clone_sequence(history)};
832 else APPEND(saves, save_count, ((Save){args[0], state, clone_sequence(history)}));
833 printf("saved %s\n", args[0]);
834 } else if (!strcmp(command, "RESTORE")) {
835 Save *save = NULL;
836 for (size_t i = 0; i < save_count; i++) if (!strcmp(saves[i].name, args[0])) save = saves + i;
837 if (!save) error("error: no save %s", args[0]);
838 state = save->cube;
839 history = clone_sequence(save->history);
840 redo.count = 0;
841 printf("restored %s\n", args[0]);
842 } else if (!strcmp(command, "SCRAMBLE")) {
843 Sequence scramble = {0};
844 uint64_t random = seed;
845 for (size_t i = 0; i < n; i++) {
846 char usable[6];
847 size_t count = 0;
848 for (int f = 0; f < 6; f++) {
849 char face = faces[f];
850 if (i >= 1 && scramble.items[i - 1].letter == face) continue;
851 if (i >= 2 && axis(scramble.items[i - 1].letter) == axis(scramble.items[i - 2].letter) && axis(face) == axis(scramble.items[i - 1].letter)) continue;
852 usable[count++] = face;
853 }
854 random = 48271 * random % 2147483647;
855 char face = usable[random % count];
856 random = 48271 * random % 2147483647;
857 int amounts[] = {1,3,2};
858 APPEND(scramble.items, scramble.count, ((Move){face, amounts[random % 3]}));
859 }
860 apply_current(scramble);
861 printf("scramble %s\n", moves_text(scramble));
862 } else if (!strcmp(command, "APPLY")) {
863 apply_current(moves);
864 printf("applied %zu\n", moves.count);
865 } else if (!strcmp(command, "INVERT")) printf("inverse %s\n", sequence_text(invert(moves)));
866 else if (!strcmp(command, "SIMPLIFY")) printf("simplified %s\n", sequence_text(simplify(moves)));
867 else if (!strcmp(command, "METRICS")) {
868 int htm = 0, qtm = 0, stm = 0;
869 for (size_t i = 0; i < moves.count; i++) {
870 Move move = moves.items[i];
871 if (in_letters("xyz", move.letter)) continue;
872 int weight = in_letters("MES", move.letter) ? 2 : 1;
873 htm += weight;
874 qtm += weight * (move.amount == 2 ? 2 : 1);
875 stm++;
876 }
877 printf("metrics htm %d qtm %d stm %d etm %zu\n", htm, qtm, stm, moves.count);
878 } else if (!strcmp(command, "ORDER")) printf("order %llu\n", (unsigned long long)order(effect(moves)));
879 else if (!strcmp(command, "CYCLES")) print_cycles(apply(solved_state(), moves), true);
880 else if (!strcmp(command, "UNROTATE")) printf("unrotated %s\n", sequence_text(unrotate(moves)));
881 else if (!strcmp(command, "UNTIL")) {
882 Permutation p = effect(moves);
883 uint64_t count = order(p), found = 0;
884 Cube candidate = state;
885 for (uint64_t k = 1; k <= count; k++) {
886 candidate = apply_permutation(candidate, p);
887 if (solved(candidate)) {
888 found = k;
889 break;
890 }
891 }
892 if (found) printf("until %llu\n", (unsigned long long)found);
893 else puts("until never");
894 } else if (!strcmp(command, "DEFINE")) {
895 Macro *macro = find_macro(head);
896 if (macro) macro->sequence = clone_sequence(moves);
897 else APPEND(macros, macro_count, ((Macro){head, clone_sequence(moves)}));
898 printf("defined %s %zu\n", head, moves.count);
899 } else if (!strcmp(command, "MIRROR")) {
900 int mirror_axis = !strcmp(head, "LR") ? 0 : !strcmp(head, "UD") ? 1 : 2;
901 const char *originals = "URFDLBurfdlb", *opposites = "DLBURFdlburf";
902 Sequence out = {0};
903 for (size_t i = 0; i < moves.count; i++) {
904 Move move = moves.items[i];
905 if (axis(move.letter) == mirror_axis) {
906 const char *position = strchr(originals, move.letter);
907 if (position) {
908 move.letter = opposites[position - originals];
909 move.amount = 4 - move.amount;
910 }
911 } else move.amount = 4 - move.amount;
912 APPEND(out.items, out.count, move);
913 }
914 printf("mirror %s\n", sequence_text(out));
915 } else {
916 Move rotation = {head[0], !head[1] ? 1 : head[1] == '2' ? 2 : 3};
917 Sequence out = {0};
918 for (size_t i = 0; i < moves.count; i++) APPEND(out.items, out.count, conjugate(move_permutation(rotation), moves.items[i]));
919 printf("rotated %s\n", sequence_text(out));
920 }
921}

924 Buffer b = {0};
925 int c;
926 while ((c = getchar()) != EOF && c != '\n') append(&b, (char[]){(char)c, 0});
927 if (c == EOF && !b.length) return NULL;
928 return finish(&b);
929}
