97function identify(state: string[], names: string[], positions: number[][]): PieceState {
98 const pieces: number[] = [];
99 const orientations: number[] = [];
100 for (const position of positions) {
101 const current = position.map(index => state[index]);
102 let found = -1;
103 let twist = 0;
104 for (let piece = 0; piece < names.length && found < 0; piece++) {
105 const expected = [...names[piece]].map(face => colors[faces.indexOf(face)]);
106 for (let orientation = 0; orientation < current.length; orientation++) {
107 if (expected.every((color, index) => current[(orientation + index) % current.length] === color)) {
108 found = piece;
109 twist = orientation;
110 break;
111 }
112 }
113 }
114 pieces.push(found);
115 orientations.push(twist);
116 }
117 return { pieces, orientations };
118}
119function cycles(state: string[], names: string[], positions: number[][]): string {
120 const identified = identify(state, names, positions);
121 const destination: number[] = [];
122 identified.pieces.forEach((piece, position) => destination[piece] = position);
123 const seen = new Set<number>();
124 const result: string[] = [];
125 for (let start = 0; start < names.length; start++) {
126 if (seen.has(start)) continue;
127 const cycle: string[] = [];
128 let current = start;
129 let orientation = 0;
130 do {
131 seen.add(current);
132 cycle.push(names[current]);
133 current = destination[current];
134 orientation += identified.orientations[current];
135 } while (current !== start);
136 const remainder = orientation % positions[0].length;
137 const mark = remainder === 0 ? '' : remainder === 1 ? '+' : '-';
138 if (cycle.length > 1 || mark !== '') result.push('(' + cycle.join(' ') + ')' + mark);
139 }
140 return result.length === 0 ? 'none' : result.join(' ');
141}
142function parity(pieces: number[]): number {
143 let count = 0;
144 for (let i = 0; i < pieces.length; i++) for (let j = i + 1; j < pieces.length; j++) if (pieces[i] > pieces[j]) count++;
145 return count % 2;
146}
147function invalidState(text: string): string | null {
148 if (text.length !== 54) return `length ${text.length}`;
149 const state = [...text];
150 const bad = state.findIndex(color => !colors.includes(color));
151 if (bad >= 0) return `char ${bad + 1}`;
152 for (const color of colors) {
153 const count = state.filter(value => value === color).length;
154 if (count !== 9) return `count ${color} ${count}`;
155 }
156 const normalized = normalize(state);
157 if (normalized === null) return 'centers';
158 const corners = identify(state, cornerNames, cornerIndices);
159 const edges = identify(state, edgeNames, edgeIndices);
160 if (corners.pieces.includes(-1)) return `corner ${cornerNames[corners.pieces.indexOf(-1)]}`;
161 if (edges.pieces.includes(-1)) return `edge ${edgeNames[edges.pieces.indexOf(-1)]}`;
162 for (let piece = 0; piece < 8; piece++) if (!corners.pieces.includes(piece)) return `missing ${cornerNames[piece]}`;
163 for (let piece = 0; piece < 12; piece++) if (!edges.pieces.includes(piece)) return `missing ${edgeNames[piece]}`;
164 const normalizedCorners = identify(normalized.state, cornerNames, cornerIndices);
165 const normalizedEdges = identify(normalized.state, edgeNames, edgeIndices);
166 if (normalizedCorners.orientations.reduce((a, b) => a + b, 0) % 3 !== 0) return 'twist';
167 if (normalizedEdges.orientations.reduce((a, b) => a + b, 0) % 2 !== 0) return 'flip';
168 return parity(normalizedCorners.pieces) === parity(normalizedEdges.pieces) ? null : 'parity';
169}

173function simplify(moves: Move[]): Move[] {
174 let current = moves;
175 while (true) {
176 const result: Move[] = [];
177 for (let start = 0; start < current.length;) {
178 let end = start;
179 const sums = new Map<string, number>();
180 while (end < current.length && axis(current[end].letter) === axis(current[start].letter)) {
181 const move = current[end++];
182 sums.set(move.letter, (sums.get(move.letter) ?? 0) + move.amount);
183 }
184 for (const [letter, sum] of sums) if (sum % 4 !== 0) result.push({ letter, amount: sum % 4 });
185 start = end;
186 }
187 if (sequenceText(result) === sequenceText(current)) return result;
188 current = result;
189 }

191function matchingMove(permutation: Permutation, outerOnly = false): Move {
192 for (const letter of outerOnly ? faces : letters) for (let amount = 1; amount <= 3; amount++) {
193 const candidate = movePermutations.get(moveText({ letter, amount }))!;
194 if (candidate.every((destination, index) => destination === permutation[index])) return { letter, amount };
195 }
196 throw new Error('move effect unavailable');

198function conjugate(move: Move, rotation: Permutation, outerOnly = false): Move {
199 return matchingMove(compose(compose(rotation, movePermutations.get(moveText(move))!), inversePermutation(rotation)), outerOnly);

201function unrotate(moves: Move[]): Move[] {
202 const replacements: Record<string, string> = { M: "R L' x'", E: "U D' y'", S: "F' B z", r: 'L x', l: "R x'", u: 'D y', d: "U y'", f: 'B z', b: "F z'" };
203 const result: Move[] = [];
204 let rotation = identity();
205 for (const move of moves) {
206 const expanded = replacements[move.letter] === undefined ? [move] : basicSequence(replacements[move.letter]).map(part => ({ letter: part.letter, amount: part.amount * move.amount % 4 }));
207 for (const part of expanded) {
208 if ('xyz'.includes(part.letter)) rotation = compose(rotation, movePermutations.get(moveText(part))!);
209 else result.push(conjugate(part, rotation, true));
210 }
211 }
212 const index = rotationEffects.findIndex(candidate => candidate.every((destination, i) => destination === rotation[i]));
213 return [...result, ...rotations[index]];
214}
215function gcd(a: number, b: number): number {
216 while (b !== 0) [a, b] = [b, a % b];
217 return a;
218}
219function order(permutation: Permutation): number {
220 const seen = new Set<number>();
221 let result = 1;
222 for (let start = 0; start < 54; start++) {
223 if (seen.has(start)) continue;
224 let current = start;
225 let length = 0;
226 do { seen.add(current); current = permutation[current]; length++; } while (current !== start);
227 result = result / gcd(result, length) * length;
228 }
229 return result;
230}

232class SequenceParser {
233 constructor(private line: string, private position: number, private macros: Map<string, Move[]>) {}
234 current(): string { return this.line[this.position] ?? ''; }
235 spaces(): void { while (this.current() === ' ') this.position++; }
236 syntax(): never { throw new Error(`syntax at ${Math.min(this.position, this.line.length) + 1}`); }
237 power(moves: Move[]): Move[] {
238 let count = 1;
239 if (/^[1-9]$/.test(this.current())) {
240 count = Number(this.current()); this.position++;
241 if (/^[0-9]$/.test(this.current())) { count = count * 10 + Number(this.current()); this.position++; }
242 }
243 const invert = this.current() === "'";
244 if (invert) this.position++;
245 const result: Move[] = [];
246 for (let i = 0; i < count; i++) result.push(...moves);
247 return invert ? inverse(result) : result;
248 }
249 sequence(): Move[] {
250 const result: Move[] = [];
251 while (true) {
252 this.spaces();
253 const char = this.current();
254 if (char !== '' && letters.includes(char)) {
255 this.position++;
256 let amount = 1;
257 if (this.current() === '2') { amount = 2; this.position++; }
258 else if (this.current() === "'") { amount = 3; this.position++; }
259 result.push({ letter: char, amount });
260 } else if (char === '(') {
261 this.position++;
262 const group = this.sequence(); this.spaces();
263 if (this.current() !== ')') this.syntax();
264 this.position++; result.push(...this.power(group));
265 } else if (char === '[') {
266 this.position++;
267 const left = this.sequence(); this.spaces();
268 const separator = this.current();
269 if (separator !== ',' && separator !== ':') this.syntax();
270 this.position++;
271 const right = this.sequence(); this.spaces();
272 if (this.current() !== ']') this.syntax();
273 this.position++;
274 const group = [...left, ...right, ...inverse(left), ...(separator === ',' ? inverse(right) : [])];
275 result.push(...this.power(group));
276 } else if (char === '{') {
277 this.position++;
278 const start = this.position;
279 while (/^[a-z]$/.test(this.current())) this.position++;
280 const name = this.line.slice(start, this.position);
281 if (name === '' || this.current() !== '}') this.syntax();
282 const macro = this.macros.get(name);
283 if (macro === undefined) throw new Error(`undefined ${name}`);
284 this.position++; result.push(...this.power(macro));
285 } else return result;
286 }
287 }
288 parse(): Move[] {
289 const result = this.sequence(); this.spaces();
290 if (this.current() !== '') this.syntax();
291 if (result.length > 500) throw new Error('too long');
292 return result;
293 }
294}

296class Cube {
297 state = solvedState();
298 history: Move[] = [];
299 redo: Move[] = [];

304 search(depth: number): Move[] | null {
305 const visit = (state: string[], remaining: number, previous: string): Move[] | null => {
306 if (remaining === 0) return mismatch(state) === 0 ? [] : null;
307 for (const face of faces) {
308 if (face === previous || previous === 'D' && face === 'U' || previous === 'L' && face === 'R' || previous === 'B' && face === 'F') continue;
309 for (const amount of [1, 2, 3]) {
310 const move = { letter: face, amount };
311 const tail = visit(applyPermutation(state, movePermutations.get(moveText(move))!), remaining - 1, face);
312 if (tail !== null) return [move, ...tail];
313 }
314 }
315 return null;
316 };
317 for (let limit = 0; limit <= depth; limit++) {
318 const result = visit(this.state, limit, '');
319 if (result !== null) return result;
320 }
321 return null;
322 }
323 apply(moves: Move[]): void {
324 this.state = applyPermutation(this.state, effect(moves));
325 this.history.push(...moves);
326 this.redo = [];

328 execute(line: string): void {
329 const space = line.indexOf(' ');
330 const command = space < 0 ? line : line.slice(0, space);
331 const argument = space < 0 ? null : line.slice(space + 1);
332 const noArguments = new Set(['RESET', 'SHOW', 'STATE', 'ISSOLVED', 'HISTORY', 'SOLUTION', 'ANALYZE', 'STAGE', 'MACROS']);
333 const wordCommands: Record<string, number> = { CHECK: 1, LOAD: 1, UNDO: 1, REDO: 1, SEARCH: 1, WHERE: 1, SAVE: 1, RESTORE: 1, SCRAMBLE: 2 };
334 const sequenceCommands = new Set(['APPLY', 'INVERT', 'SIMPLIFY', 'METRICS', 'ORDER', 'CYCLES', 'UNROTATE', 'UNTIL']);
335 const headCommands = new Set(['DEFINE', 'MIRROR', 'ROTATE']);
336 if (!noArguments.has(command) && !Object.hasOwn(wordCommands, command) && !sequenceCommands.has(command) && !headCommands.has(command)) throw new Error(`unknown command ${command}`);
337 const words = argument === null ? [] : argument.split(' ');
338 const numberWord = /^(?:0|[1-9][0-9]{0,9})$/;
339 if (noArguments.has(command) && argument !== null || Object.hasOwn(wordCommands, command) && words.length !== wordCommands[command]) throw new Error('bad arguments');
340 if (['UNDO', 'REDO', 'SEARCH', 'SCRAMBLE'].includes(command) && !words.every(word => numberWord.test(word))) throw new Error('bad arguments');
341 if (['SAVE', 'RESTORE'].includes(command) && !/^[a-z]{1,8}$/.test(words[0])) throw new Error('bad arguments');
342 if (headCommands.has(command) && (argument === null || command === 'DEFINE' && !/^[a-z]{1,8}$/.test(words[0]) || command === 'MIRROR' && !['LR', 'UD', 'FB'].includes(words[0]) || command === 'ROTATE' && !/^[xyz](?:2|')?$/.test(words[0]))) throw new Error('bad arguments');
343 let moves: Move[] = [];
344 if (sequenceCommands.has(command)) moves = new SequenceParser(line, space < 0 ? line.length : space + 1, this.macros).parse();
345 else if (headCommands.has(command)) moves = new SequenceParser(line, (space + 1) + words[0].length + 1, this.macros).parse();
346 if (['UNDO', 'REDO'].includes(command) && Number(words[0]) < 1 || command === 'SEARCH' && Number(words[0]) > 4 || command === 'SCRAMBLE' && (Number(words[0]) < 1 || Number(words[0]) > 50 || Number(words[1]) < 1 || Number(words[1]) > 2147483646)) throw new Error('bad value');
347 if (command === 'WHERE' && !cornerNames.includes(words[0]) && !edgeNames.includes(words[0])) throw new Error(`unknown piece ${words[0]}`);
348 switch (command) {
349 case 'RESET': this.state = solvedState(); this.history = []; this.redo = []; this.output.push('reset'); break;
350 case 'STATE': this.output.push('state ' + this.state.join('')); break;
351 case 'ISSOLVED': { const count = mismatch(this.state); this.output.push(count === 0 ? 'solved' : `unsolved ${count}`); break; }
352 case 'HISTORY': this.output.push('history ' + sequenceText(this.history)); break;
353 case 'SOLUTION': this.output.push('solution ' + sequenceText(simplify(inverse(this.history)))); break;
354 case 'SHOW': {
355 const row = (face: string, index: number): string => this.state.slice(faces.indexOf(face) * 9 + index * 3, faces.indexOf(face) * 9 + index * 3 + 3).join(' ');
356 for (let i = 0; i < 3; i++) this.output.push(' ' + row('U', i));
357 for (let i = 0; i < 3; i++) this.output.push(['L', 'F', 'R', 'B'].map(face => row(face, i)).join(' '));
358 for (let i = 0; i < 3; i++) this.output.push(' ' + row('D', i));
359 break;
360 }
361 case 'MACROS': this.output.push(`macros ${this.macros.size}`); for (const name of [...this.macros.keys()].sort()) this.output.push(`macro ${name} ${sequenceText(this.macros.get(name)!)}`); break;
362 case 'CHECK': case 'LOAD': {
363 const reason = invalidState(words[0]);
364 if (command === 'CHECK') this.output.push(reason === null ? 'valid' : 'invalid ' + reason);
365 else { if (reason !== null) throw new Error('invalid ' + reason); this.state = [...words[0]]; this.history = []; this.redo = []; this.output.push('loaded'); }
366 break;
367 }
368 case 'UNDO': case 'REDO': {
369 const source = command === 'UNDO' ? this.history : this.redo;
370 const target = command === 'UNDO' ? this.redo : this.history;
371 if (source.length === 0) throw new Error(command === 'UNDO' ? 'nothing to undo' : 'nothing to redo');
372 const count = Math.min(Number(words[0]), source.length);
373 for (let i = 0; i < count; i++) { const move = source.pop()!; this.state = applyPermutation(this.state, effect(command === 'UNDO' ? inverse([move]) : [move])); target.push(move); }
374 this.output.push(`${command === 'UNDO' ? 'undone' : 'redone'} ${count}`); break;
375 }
376 case 'SAVE': this.saves.set(words[0], { state: [...this.state], history: [...this.history] }); this.output.push('saved ' + words[0]); break;
377 case 'RESTORE': { const save = this.saves.get(words[0]); if (save === undefined) throw new Error('no save ' + words[0]); this.state = [...save.state]; this.history = [...save.history]; this.redo = []; this.output.push('restored ' + words[0]); break; }
378 case 'SEARCH': { const solution = this.search(Number(words[0])); this.output.push(solution === null ? 'not found' : 'found ' + sequenceText(solution)); break; }
379 case 'ANALYZE': case 'WHERE': case 'STAGE': {
380 const normalized = normalize(this.state)!;
381 const corners = identify(normalized.state, cornerNames, cornerIndices);
382 const edges = identify(normalized.state, edgeNames, edgeIndices);
383 if (command === 'ANALYZE') {
384 const rotation = rotations[normalized.rotation];
385 this.output.push('rotation ' + (rotation.length === 0 ? 'none' : rotation.map(moveText).join(' ')));
386 this.output.push('corners ' + cycles(normalized.state, cornerNames, cornerIndices));
387 this.output.push('edges ' + cycles(normalized.state, edgeNames, edgeIndices));
388 } else if (command === 'WHERE') {
389 const corner = cornerNames.includes(words[0]);
390 const names = corner ? cornerNames : edgeNames;
391 const data = corner ? corners : edges;
392 const at = data.pieces.indexOf(names.indexOf(words[0]));
393 this.output.push(`${words[0]} at ${names[at]} ${corner ? 'twist' : 'flip'} ${data.orientations[at]}`);
394 } else {
395 const correct = (data: PieceState, index: number): boolean => data.pieces[index] === index && data.orientations[index] === 0;
396 const cross = [4, 5, 6, 7].filter(index => correct(edges, index)).length;
397 const slots = [4, 5, 6, 7].filter(index => correct(corners, index) && correct(edges, index + 4)).length;
398 const top = normalized.state.slice(0, 9).filter(color => color === 'W').length;
399 const stage = mismatch(normalized.state) === 0 ? 'solved'
400 : cross === 4 && slots === 4 ? top === 9 ? 'oll' : 'f2l'
401 : cross === 4 ? 'cross' : 'none';
402 this.output.push(`stage ${stage} cross ${cross} slots ${slots} top ${top}`);
403 }
404 break;
405 }
406 case 'SCRAMBLE': {
407 let random = Number(words[1]);
408 const next = (): number => random = random * 48271 % 2147483647;
409 const scramble: Move[] = [];
410 for (let i = 0; i < Number(words[0]); i++) {
411 const previous = scramble[i - 1]?.letter;
412 const before = scramble[i - 2]?.letter;
413 const usable = [...faces].filter(face => face !== previous
414 && !(before !== undefined && previous !== undefined
415 && axis(before) === axis(previous) && axis(face) === axis(previous)));
416 const letter = usable[next() % usable.length];
417 const amount = [1, 3, 2][next() % 3];
418 scramble.push({ letter, amount });
419 }
420 this.apply(scramble);
421 this.output.push('scramble ' + scramble.map(moveText).join(' '));
422 break;
423 }
424 case 'APPLY': this.apply(moves); this.output.push(`applied ${moves.length}`); break;
425 case 'DEFINE': this.macros.set(words[0], [...moves]); this.output.push(`defined ${words[0]} ${moves.length}`); break;
426 case 'INVERT': this.output.push('inverse ' + sequenceText(inverse(moves))); break;
427 case 'SIMPLIFY': this.output.push('simplified ' + sequenceText(simplify(moves))); break;
428 case 'UNROTATE': this.output.push('unrotated ' + sequenceText(unrotate(moves))); break;
429 case 'ROTATE': { const rotation = effect(basicSequence(words[0])); this.output.push('rotated ' + sequenceText(moves.map(move => conjugate(move, rotation)))); break; }
430 case 'MIRROR': {
431 const mirrorAxis = words[0] === 'LR' ? 'x' : words[0] === 'UD' ? 'y' : 'z';
432 const opposites: Record<string, string> = {
433 R: 'L', L: 'R', r: 'l', l: 'r', U: 'D', D: 'U', u: 'd', d: 'u',
434 F: 'B', B: 'F', f: 'b', b: 'f',
435 };
436 const mirrored = moves.map(move => axis(move.letter) !== mirrorAxis
437 ? { letter: move.letter, amount: 4 - move.amount }
438 : Object.hasOwn(opposites, move.letter)
439 ? { letter: opposites[move.letter], amount: 4 - move.amount } : move);
440 this.output.push('mirror ' + sequenceText(mirrored));
441 break;
442 }
443 case 'METRICS': {
444 let htm = 0; let qtm = 0; let stm = 0;
445 for (const move of moves) {
446 if ('xyz'.includes(move.letter)) continue;
447 const weight = 'MES'.includes(move.letter) ? 2 : 1;
448 htm += weight; qtm += weight * (move.amount === 2 ? 2 : 1); stm++;
449 }
450 this.output.push(`metrics htm ${htm} qtm ${qtm} stm ${stm} etm ${moves.length}`);
451 break;
452 }
453 case 'ORDER': this.output.push(`order ${order(effect(moves))}`); break;
454 case 'CYCLES': { const state = applyPermutation(solvedState(), effect(moves)); this.output.push('corners ' + cycles(state, cornerNames, cornerIndices), 'edges ' + cycles(state, edgeNames, edgeIndices), 'centers ' + cycles(state, [...faces], centerIndices)); break; }
455 case 'UNTIL': { const permutation = effect(moves); let state = this.state; let found = 0; for (let i = 1; i <= order(permutation); i++) { state = applyPermutation(state, permutation); if (mismatch(state) === 0) { found = i; break; } } this.output.push(found === 0 ? 'until never' : `until ${found}`); break; }
456 }
457 }
458 run(lines: string[]): void {
459 for (const line of lines.slice(1, Number(lines[0]) + 1)) {
460 try { this.execute(line); } catch (error) { this.output.push(`error: ${(error as Error).message}`); }
461 }
462 }
463}
