60 {
61 int[][] positions = new int[54][], normals = new int[54][];
62 int[][] faceNormals = [[0,1,0],[1,0,0],[0,0,1],[0,-1,0],[-1,0,0],[0,0,-1]];
63 for (int face = 0; face < 6; face++) for (int row = 0; row < 3; row++) for (int column = 0; column < 3; column++)
64 {
65 int index = face * 9 + row * 3 + column;
66 positions[index] = face switch { 0 => [column-1,1,row-1], 1 => [1,1-row,1-column], 2 => [column-1,1-row,1], 3 => [column-1,-1,1-row], 4 => [-1,1-row,column-1], _ => [1-column,1-row,-1] };

82 normal = Rotate(normal, axis);

85 {
86 permutation[i] = j;

92 {
93 effect = Compose(effect, permutation);

133 class SequenceParser(string line, int start)
134 {
135 int position = start;
136 char Peek => position >= line.Length ? '\0' : line[position];
137 void Spaces()
138 {
139 while (Peek == ' ') position++;
140 }
141 void Syntax() => throw new Problem("error: syntax at " + (Math.Min(position, line.Length) + 1));
142 List<Move> Power(List<Move> moves)
143 {
144 int count = 1;
145 if (Peek is >= '1' and <= '9')
146 {
147 count = Peek - '0';
148 position++;
149 if (Peek is >= '0' and <= '9')
150 {
151 count = 10 * count + Peek - '0';
152 position++;
153 }
154 }
155 bool reverse = Peek == '\'';
156 if (reverse) position++;
157 List<Move> result = [];
158 for (int i = 0; i < count; i++) result.AddRange(moves);
159 return reverse ? Invert(result) : result;
160 }
161 List<Move> Sequence()
162 {
163 List<Move> result = [];
164 while (true)
165 {
166 Spaces();
167 char c = Peek;
168 if (Letters.Contains(c))
169 {
170 position++;
171 int amount = Peek == '2' ? 2 : Peek == '\'' ? 3 : 1;
172 if (amount != 1) position++;
173 result.Add(new(c, amount));
174 }
175 else if (c == '(')
176 {
177 position++;
178 var group = Sequence();
179 Spaces();
180 if (Peek != ')') Syntax();
181 position++;
182 result.AddRange(Power(group));
183 }
184 else if (c == '[')
185 {
186 position++;
187 var first = Sequence();
188 Spaces();
189 char separator = Peek;
190 if (separator is not (',' or ':')) Syntax();
191 position++;
192 var second = Sequence();
193 Spaces();
194 if (Peek != ']') Syntax();
195 position++;
196 List<Move> group = [.. first, .. second, .. Invert(first)];
197 if (separator == ',') group.AddRange(Invert(second));
198 result.AddRange(Power(group));
199 }
200 else if (c == '{')
201 {
202 position++;
203 int begin = position;
204 while (Peek is >= 'a' and <= 'z') position++;
205 string name = line[begin..position];
206 if (name.Length == 0 || Peek != '}') Syntax();
207 if (!Macros.TryGetValue(name, out List<Move>? macro)) throw new Problem("error: undefined " + name);
208 position++;
209 result.AddRange(Power(macro));
210 }
211 else return result;
212 }
213 }
214 public List<Move> Whole()
215 {
216 var result = Sequence();
217 Spaces();
218 if (position < line.Length) Syntax();
219 if (result.Count > 500) throw new Problem("error: too long");
220 return result;
221 }
222 }

240 static Pieces Identify(char[] input, int[][] stickers, string[] names)
241 {
242 int[] piece = Enumerable.Repeat(-1, names.Length).ToArray(), orientation = new int[names.Length];
243 for (int q = 0; q < names.Length; q++) for (int p = 0; p < names.Length; p++) for (int t = 0; t < stickers[q].Length; t++)
244 {
245 bool good = true;
246 for (int i = 0; i < stickers[q].Length; i++) if (input[stickers[q][(t + i) % stickers[q].Length]] != Colors[Faces.IndexOf(names[p][i])]) good = false;
247 if (good)
248 {
249 piece[q] = p;
250 orientation[q] = t;
251 }
252 }
253 return new(piece, orientation);
254 }

257 {
258 int count = 0;
259 for (int i = 0; i < permutation.Length; i++) for (int j = i + 1; j < permutation.Length; j++) if (permutation[i] > permutation[j]) count++;

263 static string? Check(string text)
264 {
265 if (text.Length != 54) return "length " + text.Length;
266 for (int i = 0; i < 54; i++) if (!Colors.Contains(text[i])) return "char " + (i + 1);
267 foreach (char color in Colors)
268 {
269 int count = text.Count(c => c == color);
270 if (count != 9) return $"count {color} {count}";
271 }
272 char[] input = text.ToCharArray();
273 Normalized? normalized = Normalize(input);
274 if (normalized is null) return "centers";
275 Pieces corners = Identify(input, CornerStickers, Corners), edges = Identify(input, EdgeStickers, Edges);
276 for (int i = 0; i < 8; i++) if (corners.Piece[i] < 0) return "corner " + Corners[i];
277 for (int i = 0; i < 12; i++) if (edges.Piece[i] < 0) return "edge " + Edges[i];
278 for (int i = 0; i < 8; i++) if (!corners.Piece.Contains(i)) return "missing " + Corners[i];
279 for (int i = 0; i < 12; i++) if (!edges.Piece.Contains(i)) return "missing " + Edges[i];
280 corners = Identify(normalized.State, CornerStickers, Corners);
281 edges = Identify(normalized.State, EdgeStickers, Edges);
282 if (corners.Orientation.Sum() % 3 != 0) return "twist";
283 if (edges.Orientation.Sum() % 2 != 0) return "flip";
284 return Parity(corners.Piece) != Parity(edges.Piece) ? "parity" : null;
285 }

287 static string Cycles(Pieces pieces, string[] names, int modulus)
288 {
289 int[] destination = new int[names.Length], orientation = new int[names.Length];
290 for (int q = 0; q < names.Length; q++)
291 {
292 destination[pieces.Piece[q]] = q;
293 orientation[pieces.Piece[q]] = pieces.Orientation[q];
294 }
295 bool[] seen = new bool[names.Length];
296 List<string> output = [];
297 for (int start = 0; start < names.Length; start++) if (!seen[start])
298 {
299 List<string> cycle = [];
300 int p = start, sum = 0;
301 do
302 {
303 seen[p] = true;
304 cycle.Add(names[p]);
305 sum += orientation[p];
306 p = destination[p];
307 }
308 while (p != start);
309 string mark = modulus == 3 ? sum % 3 == 1 ? "+" : sum % 3 == 2 ? "-" : "" : modulus == 2 && sum % 2 == 1 ? "+" : "";
310 if (cycle.Count >= 2 || mark.Length > 0) output.Add("(" + string.Join(" ", cycle) + ")" + mark);
311 }
312 return output.Count == 0 ? "none" : string.Join(" ", output);
313 }

322 static List<Move> Simplify(List<Move> input)
323 {
324 List<Move> moves = [.. input];
325 while (true)
326 {
327 List<Move> result = [];
328 for (int i = 0; i < moves.Count;)
329 {
330 int axis = Axis(moves[i].Letter);
331 List<char> order = [];
332 Dictionary<char, int> amounts = [];
333 while (i < moves.Count && Axis(moves[i].Letter) == axis)
334 {
335 Move move = moves[i++];
336 if (!amounts.ContainsKey(move.Letter)) order.Add(move.Letter);
337 amounts[move.Letter] = amounts.GetValueOrDefault(move.Letter) + move.Amount;
338 }
339 foreach (char letter in order) if (amounts[letter] % 4 != 0) result.Add(new(letter, amounts[letter] % 4));
340 }
341 if (result.SequenceEqual(moves)) return result;
342 moves = result;
343 }
344 }

346 static Move Conjugate(int[] rotation, Move move)
347 {
348 int[] target = Compose(Compose(rotation, Permutations[move]), Inverse(rotation));
349 foreach (char letter in Letters) for (int amount = 1; amount <= 3; amount++)
350 {
351 Move candidate = new(letter, amount);
352 if (target.SequenceEqual(Permutations[candidate])) return candidate;
353 }
354 throw new InvalidOperationException();
355 }

357 static List<Move> Unrotate(List<Move> moves)
358 {
359 Dictionary<char, List<Move>> replacements = new() { ['M'] = [new('R',1),new('L',3),new('x',3)], ['E'] = [new('U',1),new('D',3),new('y',3)], ['S'] = [new('F',3),new('B',1),new('z',1)], ['r'] = [new('L',1),new('x',1)], ['l'] = [new('R',1),new('x',3)], ['u'] = [new('D',1),new('y',1)], ['d'] = [new('U',1),new('y',3)], ['f'] = [new('B',1),new('z',1)], ['b'] = [new('F',1),new('z',3)] };
360 int[] rotation = Identity();
361 List<Move> output = [];
362 foreach (Move original in moves)
363 {
364 IEnumerable<Move> expanded = replacements.TryGetValue(original.Letter, out var replacement) ? replacement.Select(part => new Move(part.Letter, part.Amount * original.Amount % 4)) : [original];
365 foreach (Move move in expanded) if ("xyz".Contains(move.Letter)) rotation = Compose(rotation, Permutations[move]);
366 else output.Add(Conjugate(rotation, move));
367 }
368 foreach (List<Move> candidate in Rotations) if (Effect(candidate).SequenceEqual(rotation))
369 {
370 output.AddRange(candidate);
371 break;
372 }
373 return output;
374 }

376 static long Gcd(long left, long right)
377 {
378 while (right != 0)
379 {
380 long remainder = left % right;
381 left = right;
382 right = remainder;
383 }
384 return left;
385 }

387 static long Order(int[] permutation)
388 {
389 bool[] seen = new bool[54];
390 long order = 1;
391 for (int i = 0; i < 54; i++) if (!seen[i])
392 {
393 int count = 0, p = i;
394 do
395 {
396 seen[p] = true;
397 count++;
398 p = permutation[p];
399 }
400 while (p != i);
401 order = order / Gcd(order, count) * count;
402 }
403 return order;
404 }

406 static bool Search(char[] input, int left, int previous, List<Move> path)
407 {
408 if (Solved(input)) return true;
409 if (left == 0) return false;
410 for (int face = 0; face < 6; face++)
411 {
412 if (face == previous || previous == 3 && face == 0 || previous == 4 && face == 1 || previous == 5 && face == 2) continue;
413 for (int amount = 1; amount <= 3; amount++)
414 {
415 Move move = new(Faces[face], amount);
416 path.Add(move);
417 if (Search(Apply(input, Permutations[move]), left - 1, face, path)) return true;
418 path.RemoveAt(path.Count - 1);
419 }
420 }
421 return false;
422 }

444 static void Command(string line)
445 {
446 int space = line.IndexOf(' ');
447 string command = space < 0 ? line : line[..space];
448 string? argument = space < 0 ? null : line[(space + 1)..];
449 Require(NoArguments.Contains(command) || WordCommands.Contains(command) || SequenceCommands.Contains(command) || HeadCommands.Contains(command), "error: unknown command " + command);
450 string[] words = argument is null ? [] : argument.Split(' ');
451 string? head = null;
452 List<Move> moves = [];
453 long n = 0, seed = 0;
454 if (NoArguments.Contains(command)) Require(argument is null, "error: bad arguments");
455 if (WordCommands.Contains(command))
456 {
457 Require(words.Length == (command == "SCRAMBLE" ? 2 : 1), "error: bad arguments");
458 if (command is "UNDO" or "REDO" or "SEARCH" or "SCRAMBLE") n = Number(words[0]);
459 if (command == "SCRAMBLE") seed = Number(words[1]);
460 if (command is "SAVE" or "RESTORE") Require(Matches(words[0], "[a-z]{1,8}"), "error: bad arguments");
461 }
462 if (HeadCommands.Contains(command))
463 {
464 Require(argument is not null, "error: bad arguments");
465 head = words[0];
466 Require(command == "DEFINE" ? Matches(head, "[a-z]{1,8}") : command == "MIRROR" ? head is "LR" or "UD" or "FB" : Matches(head, "[xyz](?:'|2)?"), "error: bad arguments");
467 moves = new SequenceParser(line, command.Length + head.Length + 2).Whole();
468 }
469 else if (SequenceCommands.Contains(command)) moves = new SequenceParser(line, command.Length + 1).Whole();
470 if (command is "UNDO" or "REDO") Require(n >= 1, "error: bad value");
471 if (command == "SEARCH") Require(n <= 4, "error: bad value");
472 if (command == "SCRAMBLE") Require(n is >= 1 and <= 50 && seed is >= 1 and <= 2147483646L, "error: bad value");
473 if (command == "WHERE") Require(Corners.Contains(words[0]) || Edges.Contains(words[0]), "error: unknown piece " + words[0]);
474 switch (command)
475 {
476 case "RESET":
477 state = SolvedState();
478 History.Clear();
479 Redo.Clear();
480 Console.WriteLine("reset");
481 break;
482 case "SHOW":
483 for (int row = 0; row < 3; row++) Console.WriteLine(" " + string.Join(" ", state.Skip(row * 3).Take(3)));
484 for (int row = 0; row < 3; row++) Console.WriteLine(string.Join(" ", new[] {4,2,1,5}.SelectMany(face => state.Skip(face * 9 + row * 3).Take(3))));
485 for (int row = 0; row < 3; row++) Console.WriteLine(" " + string.Join(" ", state.Skip(27 + row * 3).Take(3)));
486 break;
487 case "STATE":
488 Console.WriteLine("state " + new string(state));
489 break;
490 case "ISSOLVED": int count = Enumerable.Range(0, 54).Count(index => state[index] != state[(index / 9) * 9 + 4]);
491 Console.WriteLine(count == 0 ? "solved" : "unsolved " + count);
492 break;
493 case "HISTORY":
494 Console.WriteLine("history " + SequenceText(History));
495 break;
496 case "SOLUTION":
497 Console.WriteLine("solution " + SequenceText(Simplify(Invert(History))));
498 break;
499 case "ANALYZE":
500 {
501 var normalized = Normalize(state)!;
502 Console.WriteLine("rotation " + (normalized.Rotation.Count == 0 ? "none" : MoveText(normalized.Rotation)));
503 Cycles(normalized.State, false);
504 break;
505 }
506 case "STAGE":
507 {
508 char[] normalized = Normalize(state)!.State;
509 Pieces corners = Identify(normalized, CornerStickers, Corners), edges = Identify(normalized, EdgeStickers, Edges);
510 int cross = 0, slots = 0, top = normalized.Take(9).Count(color => color == 'W');
511 for (int i = 4; i < 8; i++) if (edges.Piece[i] == i && edges.Orientation[i] == 0) cross++;
512 for (int i = 0; i < 4; i++) if (corners.Piece[i + 4] == i + 4 && corners.Orientation[i + 4] == 0 && edges.Piece[i + 8] == i + 8 && edges.Orientation[i + 8] == 0) slots++;
513 string stage = Solved(normalized) ? "solved" : cross == 4 && slots == 4 && top == 9 ? "oll" : cross == 4 && slots == 4 ? "f2l" : cross == 4 ? "cross" : "none";
514 Console.WriteLine($"stage {stage} cross {cross} slots {slots} top {top}");
515 break;
516 }
517 case "MACROS":
518 Console.WriteLine($"macros {Macros.Count}");
519 foreach (var entry in Macros) Console.WriteLine($"macro {entry.Key} {SequenceText(entry.Value)}");
520 break;
521 case "CHECK":
522 case "LOAD":
523 {
524 string? reason = Check(words[0]);
525 if (command == "CHECK") Console.WriteLine(reason is null ? "valid" : "invalid " + reason);
526 else
527 {
528 Require(reason is null, "error: invalid " + reason);
529 state = words[0].ToCharArray();
530 History.Clear();
531 Redo.Clear();
532 Console.WriteLine("loaded");
533 }
534 break;
535 }
536 case "UNDO":
537 case "REDO":
538 {
539 bool undo = command == "UNDO";
540 var source = undo ? History : Redo;
541 var target = undo ? Redo : History;
542 Require(source.Count > 0, undo ? "error: nothing to undo" : "error: nothing to redo");
543 int countMoves = (int)Math.Min(n, source.Count);
544 for (int i = 0; i < countMoves; i++)
545 {
546 Move move = source[^1];
547 source.RemoveAt(source.Count - 1);
548 state = Apply(state, Permutations[undo ? move.Inverse : move]);
549 target.Add(move);
550 }
551 Console.WriteLine((undo ? "undone " : "redone ") + countMoves);
552 break;
553 }
554 case "SEARCH":
555 {
556 List<Move> solution = [];
557 bool found = false;
558 for (int depth = 0; depth <= n; depth++)
559 {
560 solution.Clear();
561 if (Search(state, depth, -1, solution))
562 {
563 found = true;
564 break;
565 }
566 }
567 Console.WriteLine(found ? "found " + SequenceText(solution) : "not found");
568 break;
569 }
570 case "WHERE":
571 {
572 char[] normalized = Normalize(state)!.State;
573 bool corner = Corners.Contains(words[0]);
574 string[] names = corner ? Corners : Edges;
575 Pieces pieces = Identify(normalized, corner ? CornerStickers : EdgeStickers, names);
576 int piece = Array.IndexOf(names, words[0]);
577 for (int q = 0; q < names.Length; q++) if (pieces.Piece[q] == piece) Console.WriteLine(words[0] + " at " + names[q] + (corner ? " twist " : " flip ") + pieces.Orientation[q]);
578 break;
579 }
580 case "SAVE":
581 Saves[words[0]] = new((char[])state.Clone(), [.. History]);
582 Console.WriteLine("saved " + words[0]);
583 break;
584 case "RESTORE":
585 {
586 Require(Saves.ContainsKey(words[0]), "error: no save " + words[0]);
587 Save save = Saves[words[0]];
588 state = (char[])save.State.Clone();
589 History.Clear();
590 History.AddRange(save.History);
591 Redo.Clear();
592 Console.WriteLine("restored " + words[0]);
593 break;
594 }
595 case "SCRAMBLE":
596 {
597 List<Move> scramble = [];
598 long random = seed;
599 for (int i = 0; i < n; i++)
600 {
601 List<char> usable = [];
602 foreach (char face in Faces)
603 {
604 if (i >= 1 && scramble[i - 1].Letter == face) continue;
605 if (i >= 2 && Axis(scramble[i - 1].Letter) == Axis(scramble[i - 2].Letter) && Axis(face) == Axis(scramble[i - 1].Letter)) continue;
606 usable.Add(face);
607 }
608 random = 48271 * random % 2147483647;
609 char selected = usable[(int)(random % usable.Count)];
610 random = 48271 * random % 2147483647;
611 int[] amounts = [1,3,2];
612 scramble.Add(new(selected, amounts[(int)(random % 3)]));
613 }
614 ApplyCurrent(scramble);
615 Console.WriteLine("scramble " + MoveText(scramble));
616 break;
617 }
618 case "APPLY":
619 ApplyCurrent(moves);
620 Console.WriteLine($"applied {moves.Count}");
621 break;
622 case "INVERT":
623 Console.WriteLine("inverse " + SequenceText(Invert(moves)));
624 break;
625 case "SIMPLIFY":
626 Console.WriteLine("simplified " + SequenceText(Simplify(moves)));
627 break;
628 case "METRICS":
629 {
630 int htm = 0, qtm = 0, stm = 0;
631 foreach (Move move in moves) if (!"xyz".Contains(move.Letter))
632 {
633 int weight = "MES".Contains(move.Letter) ? 2 : 1;
634 htm += weight;
635 qtm += weight * (move.Amount == 2 ? 2 : 1);
636 stm++;
637 }
638 Console.WriteLine($"metrics htm {htm} qtm {qtm} stm {stm} etm {moves.Count}");
639 break;
640 }
641 case "ORDER":
642 Console.WriteLine("order " + Order(Effect(moves)));
643 break;
644 case "CYCLES":
645 Cycles(Apply(SolvedState(), moves), true);
646 break;
647 case "UNROTATE":
648 Console.WriteLine("unrotated " + SequenceText(Unrotate(moves)));
649 break;
650 case "UNTIL":
651 {
652 int[] permutation = Effect(moves);
653 long order = Order(permutation), found = 0;
654 char[] candidate = state;
655 for (long k = 1; k <= order; k++)
656 {
657 candidate = Apply(candidate, permutation);
658 if (Solved(candidate))
659 {
660 found = k;
661 break;
662 }
663 }
664 Console.WriteLine(found == 0 ? "until never" : "until " + found);
665 break;
666 }
667 case "DEFINE":
668 Macros[head!] = [.. moves];
669 Console.WriteLine($"defined {head} {moves.Count}");
670 break;
671 case "MIRROR":
672 {
673 int mirrorAxis = head == "LR" ? 0 : head == "UD" ? 1 : 2;
674 const string originals = "URFDLBurfdlb", opposites = "DLBURFdlburf";
675 List<Move> result = [];
676 foreach (Move move in moves)
677 {
678 char letter = move.Letter;
679 int amount = move.Amount;
680 if (Axis(letter) == mirrorAxis)
681 {
682 int index = originals.IndexOf(letter);
683 if (index >= 0)
684 {
685 letter = opposites[index];
686 amount = 4 - amount;
687 }
688 }
689 else amount = 4 - amount;
690 result.Add(new(letter, amount));
691 }
692 Console.WriteLine("mirror " + SequenceText(result));
693 break;
694 }
695 case "ROTATE":
696 {
697 Move rotation = new(head![0], head.Length == 1 ? 1 : head[1] == '2' ? 2 : 3);
698 List<Move> result = [];
699 foreach (Move move in moves) result.Add(Conjugate(Permutations[rotation], move));
700 Console.WriteLine("rotated " + SequenceText(result));
701 break;
702 }
703 }
