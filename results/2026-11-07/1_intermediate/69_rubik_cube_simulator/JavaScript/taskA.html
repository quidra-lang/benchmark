93function identify(state, names, positions) {
94 const pieces = [];
95 const orientations = [];
96 for (const position of positions) {
97 const current = position.map(index => state[index]);
98 let found = -1;
99 let twist = 0;
100 for (let piece = 0; piece < names.length && found < 0; piece++) {
101 const expected = [...names[piece]].map(face => colors[faces.indexOf(face)]);
102 for (let orientation = 0; orientation < current.length; orientation++) {
103 if (expected.every((color, index) => current[(orientation + index) % current.length] === color)) {
104 found = piece;
105 twist = orientation;
106 break;
107 }
108 }
109 }
110 pieces.push(found);
111 orientations.push(twist);

115function cycles(state, names, positions) {
116 const identified = identify(state, names, positions);
117 const destination = [];
118 identified.pieces.forEach((piece, position) => destination[piece] = position);
119 const seen = new Set();
120 const result = [];
121 for (let start = 0; start < names.length; start++) {
122 if (seen.has(start)) continue;
123 const cycle = [];
124 let current = start;
125 let orientation = 0;
126 do {
127 seen.add(current);
128 cycle.push(names[current]);
129 current = destination[current];
130 orientation += identified.orientations[current];
131 } while (current !== start);
132 const remainder = orientation % positions[0].length;
133 const mark = remainder === 0 ? '' : remainder === 1 ? '+' : '-';
134 if (cycle.length > 1 || mark !== '') result.push('(' + cycle.join(' ') + ')' + mark);
135 }
136 return result.length === 0 ? 'none' : result.join(' ');

138function parity(pieces) {
139 let count = 0;
140 for (let i = 0; i < pieces.length; i++) for (let j = i + 1; j < pieces.length; j++) if (pieces[i] > pieces[j]) count++;
141 return count % 2;
142}
143function invalidState(text) {
144 if (text.length !== 54) return `length ${text.length}`;
145 const state = [...text];
146 const bad = state.findIndex(color => !colors.includes(color));
147 if (bad >= 0) return `char ${bad + 1}`;
148 for (const color of colors) {
149 const count = state.filter(value => value === color).length;
150 if (count !== 9) return `count ${color} ${count}`;
151 }
152 const normalized = normalize(state);
153 if (normalized === null) return 'centers';
154 const corners = identify(state, cornerNames, cornerIndices);
155 const edges = identify(state, edgeNames, edgeIndices);
156 if (corners.pieces.includes(-1)) return `corner ${cornerNames[corners.pieces.indexOf(-1)]}`;
157 if (edges.pieces.includes(-1)) return `edge ${edgeNames[edges.pieces.indexOf(-1)]}`;
158 for (let piece = 0; piece < 8; piece++) if (!corners.pieces.includes(piece)) return `missing ${cornerNames[piece]}`;
159 for (let piece = 0; piece < 12; piece++) if (!edges.pieces.includes(piece)) return `missing ${edgeNames[piece]}`;
160 const normalizedCorners = identify(normalized.state, cornerNames, cornerIndices);
161 const normalizedEdges = identify(normalized.state, edgeNames, edgeIndices);
162 if (normalizedCorners.orientations.reduce((a, b) => a + b, 0) % 3 !== 0) return 'twist';
163 if (normalizedEdges.orientations.reduce((a, b) => a + b, 0) % 2 !== 0) return 'flip';
164 return parity(normalizedCorners.pieces) === parity(normalizedEdges.pieces) ? null : 'parity';
165}

169function simplify(moves) {
170 let current = moves;
171 while (true) {
172 const result = [];
173 for (let start = 0; start < current.length;) {
174 let end = start;
175 const sums = new Map();
176 while (end < current.length && axis(current[end].letter) === axis(current[start].letter)) {
177 const move = current[end++];
178 sums.set(move.letter, (sums.get(move.letter) ?? 0) + move.amount);
179 }
180 for (const [letter, sum] of sums) if (sum % 4 !== 0) result.push({ letter, amount: sum % 4 });
181 start = end;
182 }
183 if (sequenceText(result) === sequenceText(current)) return result;
184 current = result;
185 }

197function unrotate(moves) {
198 const replacements = { M: "R L' x'", E: "U D' y'", S: "F' B z", r: 'L x', l: "R x'", u: 'D y', d: "U y'", f: 'B z', b: "F z'" };

200 let rotation = identity();
201 for (const move of moves) {
202 const expanded = replacements[move.letter] === undefined ? [move] : basicSequence(replacements[move.letter]).map(part => ({ letter: part.letter, amount: part.amount * move.amount % 4 }));
203 for (const part of expanded) {
204 if ('xyz'.includes(part.letter)) rotation = compose(rotation, movePermutations.get(moveText(part)));
205 else result.push(conjugate(part, rotation, true));
206 }
207 }
208 const index = rotationEffects.findIndex(candidate => candidate.every((destination, i) => destination === rotation[i]));
209 return [...result, ...rotations[index]];

212 while (b !== 0) [a, b] = [b, a % b];

215function order(permutation) {
216 const seen = new Set();
217 let result = 1;
218 for (let start = 0; start < 54; start++) {
219 if (seen.has(start)) continue;
220 let current = start;
221 let length = 0;
222 do { seen.add(current); current = permutation[current]; length++; } while (current !== start);
223 result = result / gcd(result, length) * length;
224 }
225 return result;

228class SequenceParser {
229 constructor(line, position, macros) {
230 this.line = line;
231 this.position = position;
232 this.macros = macros;
233 }
234 current() { return this.line[this.position] ?? ''; }
235 spaces() { while (this.current() === ' ') this.position++; }
236 syntax() { throw new Error(`syntax at ${Math.min(this.position, this.line.length) + 1}`); }
237 power(moves) {
238 let count = 1;
239 if (/^[1-9]$/.test(this.current())) {
240 count = Number(this.current()); this.position++;
241 if (/^[0-9]$/.test(this.current())) { count = count * 10 + Number(this.current()); this.position++; }
242 }
243 const invert = this.current() === "'";
244 if (invert) this.position++;
245 const result = [];
246 for (let i = 0; i < count; i++) result.push(...moves);
247 return invert ? inverse(result) : result;
248 }
249 sequence() {
250 const result = [];
251 while (true) {
252 this.spaces();
253 const char = this.current();
254 if (char !== '' && letters.includes(char)) {
255 this.position++;
256 let amount = 1;
257 if (this.current() === '2') { amount = 2; this.position++; }
258 else if (this.current() === "'") { amount = 3; this.position++; }
259 result.push({ letter: char, amount });
260 } else if (char === '(') {
261 this.position++;
262 const group = this.sequence(); this.spaces();
263 if (this.current() !== ')') this.syntax();
264 this.position++; result.push(...this.power(group));
265 } else if (char === '[') {
266 this.position++;
267 const left = this.sequence(); this.spaces();
268 const separator = this.current();
269 if (separator !== ',' && separator !== ':') this.syntax();
270 this.position++;
271 const right = this.sequence(); this.spaces();
272 if (this.current() !== ']') this.syntax();
273 this.position++;
274 const group = [...left, ...right, ...inverse(left), ...(separator === ',' ? inverse(right) : [])];
275 result.push(...this.power(group));
276 } else if (char === '{') {
277 this.position++;
278 const start = this.position;
279 while (/^[a-z]$/.test(this.current())) this.position++;
280 const name = this.line.slice(start, this.position);
281 if (name === '' || this.current() !== '}') this.syntax();
282 const macro = this.macros.get(name);
283 if (macro === undefined) throw new Error(`undefined ${name}`);
284 this.position++; result.push(...this.power(macro));
285 } else return result;
286 }
287 }
288 parse() {
289 const result = this.sequence(); this.spaces();
290 if (this.current() !== '') this.syntax();
291 if (result.length > 500) throw new Error('too long');
292 return result;
293 }
294}

304 search(depth) {

330 const command = space < 0 ? line : line.slice(0, space);
331 const argument = space < 0 ? null : line.slice(space + 1);

335 const headCommands = new Set(['DEFINE', 'MIRROR', 'ROTATE']);

341 if (['SAVE', 'RESTORE'].includes(command) && !/^[a-z]{1,8}$/.test(words[0])) throw new Error('bad arguments');

360 }

362 case 'CHECK': case 'LOAD': {

364 if (command === 'CHECK') this.output.push(reason === null ? 'valid' : 'invalid ' + reason);

367 }

372 const count = Math.min(Number(words[0]), source.length);

379 case 'ANALYZE': case 'WHERE': case 'STAGE': {
380 const normalized = normalize(this.state);
381 const corners = identify(normalized.state, cornerNames, cornerIndices);
382 const edges = identify(normalized.state, edgeNames, edgeIndices);
383 if (command === 'ANALYZE') {
384 const rotation = rotations[normalized.rotation];
385 this.output.push('rotation ' + (rotation.length === 0 ? 'none' : rotation.map(moveText).join(' ')));
386 this.output.push('corners ' + cycles(normalized.state, cornerNames, cornerIndices));
387 this.output.push('edges ' + cycles(normalized.state, edgeNames, edgeIndices));
388 } else if (command === 'WHERE') {
389 const corner = cornerNames.includes(words[0]);
390 const names = corner ? cornerNames : edgeNames;
391 const data = corner ? corners : edges;
392 const at = data.pieces.indexOf(names.indexOf(words[0]));

412 const before = scramble[i - 2]?.letter;
413 const usable = [...faces].filter(face => face !== previous
414 && !(before !== undefined && previous !== undefined
415 && axis(before) === axis(previous) && axis(face) === axis(previous)));
416 const letter = usable[next() % usable.length];
417 const amount = [1, 3, 2][next() % 3];

426 case 'INVERT': this.output.push('inverse ' + sequenceText(inverse(moves))); break;
427 case 'SIMPLIFY': this.output.push('simplified ' + sequenceText(simplify(moves))); break;
428 case 'UNROTATE': this.output.push('unrotated ' + sequenceText(unrotate(moves))); break;
429 case 'ROTATE': { const rotation = effect(basicSequence(words[0])); this.output.push('rotated ' + sequenceText(moves.map(move => conjugate(move, rotation)))); break; }
430 case 'MIRROR': {
431 const mirrorAxis = words[0] === 'LR' ? 'x' : words[0] === 'UD' ? 'y' : 'z';
432 const opposites = {
433 R: 'L', L: 'R', r: 'l', l: 'r', U: 'D', D: 'U', u: 'd', d: 'u',
434 F: 'B', B: 'F', f: 'b', b: 'f',
435 };
436 const mirrored = moves.map(move => axis(move.letter) !== mirrorAxis
437 ? { letter: move.letter, amount: 4 - move.amount }
438 : Object.hasOwn(opposites, move.letter)
439 ? { letter: opposites[move.letter], amount: 4 - move.amount } : move);
440 this.output.push('mirror ' + sequenceText(mirrored));
441 break;
442 }
443 case 'METRICS': {
444 let htm = 0; let qtm = 0; let stm = 0;
445 for (const move of moves) {
446 if ('xyz'.includes(move.letter)) continue;
447 const weight = 'MES'.includes(move.letter) ? 2 : 1;
448 htm += weight; qtm += weight * (move.amount === 2 ? 2 : 1); stm++;
449 }
450 this.output.push(`metrics htm ${htm} qtm ${qtm} stm ${stm} etm ${moves.length}`);
451 break;
452 }
453 case 'ORDER': this.output.push(`order ${order(effect(moves))}`); break;
454 case 'CYCLES': { const state = applyPermutation(solvedState(), effect(moves)); this.output.push('corners ' + cycles(state, cornerNames, cornerIndices), 'edges ' + cycles(state, edgeNames, edgeIndices), 'centers ' + cycles(state, [...faces], centerIndices)); break; }
455 case 'UNTIL': { const permutation = effect(moves); let state = this.state; let found = 0; for (let i = 1; i <= order(permutation); i++) { state = applyPermutation(state, permutation); if (mismatch(state) === 0) { found = i; break; } } this.output.push(found === 0 ? 'until never' : `until ${found}`); break; }
456 }
457 }
458 run(lines) {
459 for (const line of lines.slice(1, Number(lines[0]) + 1)) {
460 try { this.execute(line); } catch (error) { this.output.push(`error: ${(error).message}`); }
461 }
462 }
463}
