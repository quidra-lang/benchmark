216local function identify(cube, group)
217 local result = {}
218 for position, stickers in ipairs(group.stickers) do
219 result[position] = {id = 0, orientation = 0}
220 local found = false
221 for id, name in ipairs(group.names) do
222 for orientation = 0, #stickers - 1 do
223 local same = true
224 for i = 1, #stickers do
225 local face = faces:find(name:sub(i, i), 1, true)

227 if cube:sub(at, at) ~= colors:sub(face, face) then
228 same = false
229 break
230 end
231 end
232 if same then
233 result[position] = {id = id, orientation = orientation}
234 found = true
235 break
236 end
237 end
238 if found then
239 break
240 end
241 end
242 end

245local function cycles(cube, group)
246 local where, orientations, seen, written = {}, {}, {}, {}
247 for position, piece in ipairs(identify(cube, group)) do
248 where[piece.id], orientations[piece.id] = position, piece.orientation
249 end
250 for first = 1, #group.names do

252 local names, sum, at = {}, 0, first
253 while not seen[at] do
254 seen[at] = true

256 sum, at = sum + orientations[at], where[at]
257 end

261 end

263 written[#written + 1] = "(" .. table.concat(names, " ") .. ")" .. mark
264 end
265 end
266 end

268end

283 end

285 for i = 1, #word do
286 local at = colors:find(word:sub(i, i), 1, true)

332 flip = flip + piece.orientation

339 end

356 if #a ~= #b then

366local function simplify(seq)
367 while true do
368 local result, start = {}, 1
369 while start <= #seq do
370 local finish = start + 1
371 while finish <= #seq and turns[seq[finish][1]][1] == turns[seq[start][1]][1] do
372 finish = finish + 1
373 end
374 local order, amounts = {}, {}
375 for i = start, finish - 1 do
376 local letter, amount = seq[i][1], seq[i][2]
377 if amounts[letter] == nil then
378 order[#order + 1] = letter
379 amounts[letter] = 0
380 end
381 amounts[letter] = amounts[letter] + amount
382 end
383 for _, letter in ipairs(order) do
384 if amounts[letter] % 4 ~= 0 then
385 result[#result + 1] = {letter, amounts[letter] % 4}
386 end
387 end
388 start = finish
389 end
390 if same_moves(seq, result) then
391 return result
392 end
393 seq = result
394 end
395end
396local function gcd(a, b)
397 while b ~= 0 do
398 a, b = b, a % b
399 end
400 return a
401end
402local function order(p)
403 local seen, result = {}, 1
404 for first = 1, 54 do
405 if not seen[first] then
406 local length, at = 0, first
407 while not seen[at] do
408 seen[at] = true
409 length = length + 1
410 at = p[at]
411 end
412 result = (result // gcd(result, length)) * length
413 end
414 end
415 return result
416end

424local function unrotate(seq)
425 local result, rotation = {}, identity()
426 for _, move in ipairs(seq) do
427 local expanded
428 if replacements[move[1]] then
429 expanded = copy(replacements[move[1]])
430 for _, part in ipairs(expanded) do
431 part[2] = part[2] * move[2] % 4
432 end
433 else
434 expanded = {move}
435 end
436 for _, part in ipairs(expanded) do
437 if ("xyz"):find(part[1], 1, true) then
438 rotation = compose(rotation, tables[part[1] .. part[2]])
439 else
440 result[#result + 1] = conjugate(rotation, part)
441 end
442 end
443 end

451local function search(cube, depth, previous)
452 if depth == 0 then
453 return misplaced(cube) == 0 and {} or nil
454 end
455 for i = 1, #faces do
456 local face = faces:sub(i, i)
457 if face ~= previous and not (previous == "D" and face == "U" or previous == "L" and face == "R" or previous == "B" and face == "F") then
458 for amount = 1, 3 do
459 local move = {face, amount}
460 local rest = search(permute(cube, tables[face .. amount]), depth - 1, face)
461 if rest then
462 local result = {move}

464 return result
465 end
466 end
467 end
468 end
469 return nil
470end
471local Parser = {}
472Parser.__index = Parser
473function Parser:peek()
474 return self.line:sub(self.pos, self.pos)
475end
476function Parser:skip()
477 while self:peek() == " " do
478 self.pos = self.pos + 1
479 end
480end
481function Parser:syntax()
482 bad("syntax at %d", math.min(self.pos, #self.line + 1))
483end
484function Parser:power(seq)
485 local count, current = 1, self:peek()
486 if current ~= "" and current >= "1" and current <= "9" then
487 count, self.pos = tonumber(current), self.pos + 1
488 current = self:peek()
489 if current ~= "" and current >= "0" and current <= "9" then
490 count, self.pos = 10 * count + tonumber(current), self.pos + 1
491 end
492 end
493 local invert = self:peek() == "'"
494 if invert then
495 self.pos = self.pos + 1
496 end
497 local result = {}
498 for _ = 1, count do
499 append(result, seq)
500 end
501 return invert and inverse(result) or result
502end
503function Parser:sequence()
504 local result = {}
505 while true do
506 self:skip()
507 local current, part = self:peek(), nil
508 if turns[current] then
509 self.pos = self.pos + 1
510 local amount = 1
511 if self:peek() == "2" then
512 amount, self.pos = 2, self.pos + 1
513 elseif self:peek() == "'" then
514 amount, self.pos = 3, self.pos + 1
515 end
516 part = {{current, amount}}
517 elseif current == "(" then
518 self.pos = self.pos + 1
519 local nested = self:sequence()
520 self:skip()
521 if self:peek() ~= ")" then
522 self:syntax()
523 end
524 self.pos = self.pos + 1
525 part = self:power(nested)
526 elseif current == "[" then
527 self.pos = self.pos + 1
528 local a = self:sequence()
529 self:skip()
530 local separator = self:peek()
531 if separator ~= "," and separator ~= ":" then
532 self:syntax()
533 end
534 self.pos = self.pos + 1
535 local b = self:sequence()
536 self:skip()
537 if self:peek() ~= "]" then
538 self:syntax()
539 end
540 self.pos = self.pos + 1
541 local nested = copy(a)
542 append(nested, b)
543 append(nested, inverse(a))
544 if separator == "," then
545 append(nested, inverse(b))
546 end
547 part = self:power(nested)
548 elseif current == "{" then
549 self.pos = self.pos + 1
550 local start = self.pos
551 while self:peek() ~= "" and self:peek() >= "a" and self:peek() <= "z" do
552 self.pos = self.pos + 1
553 end
554 local name = self.line:sub(start, self.pos - 1)
555 if name == "" or self:peek() ~= "}" then
556 self:syntax()
557 end
558 if not self.macros[name] then
559 bad("undefined %s", name)
560 end
561 self.pos = self.pos + 1
562 part = self:power(self.macros[name])
563 else
564 return result
565 end
566 append(result, part)
567 end
568end
569function Parser:whole()
570 local result = self:sequence()
571 self:skip()
572 if self:peek() ~= "" then
573 self:syntax()
574 end
575 if #result > 500 then
576 bad("too long")
577 end
578 return result
579end

608function simulator:command(line)
609 local at = line:find(" ", 1, true)
610 local command, arguments = at and line:sub(1, at - 1) or line, at and line:sub(at + 1) or ""
611 local no, count, sequence_command, head_command = no_arguments[command], word_counts[command], sequence_commands[command], head_commands[command]
612 if not no and not count and not sequence_command and not head_command then
613 bad("unknown command %s", command)
614 end
615 local arguments_words, head, seq_start = words(arguments), "", #command + 2
616 local invalid = no and at ~= nil or count and #arguments_words ~= count or head_command and not at
617 if not invalid and count then
618 if command == "UNDO" or command == "REDO" or command == "SEARCH" or command == "SCRAMBLE" then
619 for _, word in ipairs(arguments_words) do
620 if not number_word(word) then
621 invalid = true
622 end
623 end
624 end
625 if (command == "SAVE" or command == "RESTORE") and not name_word(arguments_words[1]) then
626 invalid = true
627 end
628 end
629 if not invalid and head_command then
630 head = arguments_words[1] or ""
631 seq_start = #command + #head + 3
632 if command == "DEFINE" then
633 invalid = not name_word(head)
634 elseif command == "MIRROR" then
635 invalid = not contains({"LR", "UD", "FB"}, head)
636 else
637 invalid = not contains({"x", "x2", "x'", "y", "y2", "y'", "z", "z2", "z'"}, head)
638 end
639 end
640 if invalid then
641 bad("bad arguments")
642 end
643 local seq = {}
644 if sequence_command or head_command then
645 seq = setmetatable({line = line, pos = seq_start, macros = self.macros}, Parser):whole()
646 end
647 local w = arguments_words
648 if (command == "UNDO" or command == "REDO") and tonumber(w[1]) < 1
649 or command == "SEARCH" and tonumber(w[1]) > 4
650 or command == "SCRAMBLE" and (tonumber(w[1]) < 1 or tonumber(w[1]) > 50 or tonumber(w[2]) < 1 or tonumber(w[2]) > 2147483646) then
651 bad("bad value")
652 end
653 if command == "WHERE" and not contains(corners.names, w[1]) and not contains(edges.names, w[1]) then
654 bad("unknown piece %s", w[1])
655 end
656 if command == "RESET" then
657 self.cube, self.history, self.redo = solved, {}, {}
658 emit("reset")
659 elseif command == "SHOW" then
660 for row = 1, 9 do
661 local cells, prefix = {}, ""
662 if row <= 3 or row >= 7 then
663 local face, r = 0, row - 1
664 if row >= 7 then
665 face, r = 3, row - 7
666 end
667 prefix = " "
668 for column = 1, 3 do
669 local index = face * 9 + r * 3 + column
670 cells[#cells + 1] = self.cube:sub(index, index)
671 end
672 else
673 for _, face in ipairs({4, 2, 1, 5}) do
674 for column = 1, 3 do
675 local index = face * 9 + (row - 4) * 3 + column
676 cells[#cells + 1] = self.cube:sub(index, index)
677 end
678 end
679 end
680 emit("%s%s", prefix, table.concat(cells, " "))
681 end
682 elseif command == "STATE" then
683 emit("state %s", self.cube)
684 elseif command == "ISSOLVED" then
685 local n = misplaced(self.cube)
686 if n == 0 then
687 emit("solved")
688 else
689 emit("unsolved %d", n)
690 end
691 elseif command == "HISTORY" then
692 emit("history %s", sequence_text(self.history))
693 elseif command == "SOLUTION" then
694 emit("solution %s", sequence_text(simplify(inverse(self.history))))
695 elseif command == "ANALYZE" or command == "STAGE" or command == "WHERE" then
696 local cube, rotation = normalize(self.cube)
697 if command == "ANALYZE" then
698 emit("rotation %s", #rotations[rotation] == 0 and "none" or moves_text(rotations[rotation]))
699 emit("corners %s", cycles(cube, corners))
700 emit("edges %s", cycles(cube, edges))
701 elseif command == "WHERE" then
702 local group, orientation = edges, "flip"
703 if contains(corners.names, w[1]) then
704 group, orientation = corners, "twist"
705 end
706 for position, piece in ipairs(identify(cube, group)) do
707 if group.names[piece.id] == w[1] then
708 emit("%s at %s %s %d", w[1], group.names[position], orientation, piece.orientation)
709 break
710 end
711 end
712 else
713 local c, e = identify(cube, corners), identify(cube, edges)
714 local cross, slots, top = 0, 0, 0
715 for i = 5, 8 do
716 if e[i].id == i and e[i].orientation == 0 then
717 cross = cross + 1
718 end
719 if c[i].id == i and c[i].orientation == 0 and e[i + 4].id == i + 4 and e[i + 4].orientation == 0 then
720 slots = slots + 1
721 end
722 end
723 for i = 1, 9 do
724 if cube:sub(i, i) == "W" then
725 top = top + 1
726 end
727 end
728 local stage = "none"
729 if misplaced(cube) == 0 then
730 stage = "solved"
731 elseif cross == 4 and slots == 4 and top == 9 then
732 stage = "oll"
733 elseif cross == 4 and slots == 4 then
734 stage = "f2l"
735 elseif cross == 4 then
736 stage = "cross"
737 end
738 emit("stage %s cross %d slots %d top %d", stage, cross, slots, top)
739 end
740 elseif command == "MACROS" then
741 local names = {}
742 for name in pairs(self.macros) do
743 names[#names + 1] = name
744 end
745 table.sort(names)
746 emit("macros %d", #names)
747 for _, name in ipairs(names) do
748 emit("macro %s %s", name, sequence_text(self.macros[name]))
749 end
750 elseif command == "CHECK" or command == "LOAD" then
751 local invalidity = check(w[1])
752 if command == "CHECK" then
753 if invalidity then
754 emit("invalid %s", invalidity)
755 else
756 emit("valid")
757 end
758 else
759 if invalidity then
760 bad("invalid %s", invalidity)
761 end
762 self.cube, self.history, self.redo = w[1], {}, {}
763 emit("loaded")
764 end
765 elseif command == "UNDO" or command == "REDO" then
766 local source = command == "UNDO" and self.history or self.redo
767 if #source == 0 then
768 bad("nothing to %s", command == "UNDO" and "undo" or "redo")
769 end
770 local count_moves = math.min(tonumber(w[1]), #source)
771 for _ = 1, count_moves do
772 local move = table.remove(source)
773 if command == "UNDO" then
774 self.cube = permute(self.cube, tables[move[1] .. (4 - move[2])])
775 self.redo[#self.redo + 1] = move
776 else
777 self.cube = permute(self.cube, tables[move[1] .. move[2]])
778 self.history[#self.history + 1] = move
779 end
780 end
781 emit("%s %d", command == "UNDO" and "undone" or "redone", count_moves)
782 elseif command == "SEARCH" then
783 for depth = 0, tonumber(w[1]) do
784 local found = search(self.cube, depth, nil)
785 if found then
786 emit("found %s", sequence_text(found))
787 return
788 end
789 end
790 emit("not found")
791 elseif command == "SAVE" then
792 self.saves[w[1]] = {cube = self.cube, history = copy(self.history)}
793 emit("saved %s", w[1])
794 elseif command == "RESTORE" then
795 local save = self.saves[w[1]]
796 if not save then
797 bad("no save %s", w[1])
798 end
799 self.cube, self.history, self.redo = save.cube, copy(save.history), {}
800 emit("restored %s", w[1])
801 elseif command == "SCRAMBLE" then
802 local scramble, seed = {}, tonumber(w[2])
803 for i = 1, tonumber(w[1]) do
804 local usable = {}
805 for j = 1, #faces do
806 local face = faces:sub(j, j)
807 if not (i >= 2 and face == scramble[i - 1][1]) and not (i >= 3 and turns[scramble[i - 1][1]][1] == turns[scramble[i - 2][1]][1] and turns[face][1] == turns[scramble[i - 1][1]][1]) then
808 usable[#usable + 1] = face
809 end
810 end
811 seed = 48271 * seed % 2147483647
812 local face = usable[seed % #usable + 1]
813 seed = 48271 * seed % 2147483647
814 scramble[#scramble + 1] = {face, ({1, 3, 2})[seed % 3 + 1]}
815 end
816 self:record(scramble)
817 emit("scramble %s", moves_text(scramble))
818 elseif command == "APPLY" then
819 self:record(seq)
820 emit("applied %d", #seq)
821 elseif command == "INVERT" then
822 emit("inverse %s", sequence_text(inverse(seq)))
823 elseif command == "SIMPLIFY" then
824 emit("simplified %s", sequence_text(simplify(seq)))
825 elseif command == "METRICS" then
826 local htm, qtm, stm = 0, 0, 0
827 for _, move in ipairs(seq) do
828 if not ("xyz"):find(move[1], 1, true) then
829 local h, q = ("MES"):find(move[1], 1, true) and 2 or 1, move[2] == 2 and 2 or 1
830 htm, qtm, stm = htm + h, qtm + h * q, stm + 1
831 end
832 end
833 emit("metrics htm %d qtm %d stm %d etm %d", htm, qtm, stm, #seq)
834 elseif command == "ORDER" then
835 emit("order %d", order(effect(seq)))
836 elseif command == "CYCLES" then
837 local cube = apply(solved, seq)
838 emit("corners %s", cycles(cube, corners))
839 emit("edges %s", cycles(cube, edges))
840 emit("centers %s", cycles(cube, centers))
841 elseif command == "UNROTATE" then
842 emit("unrotated %s", sequence_text(unrotate(seq)))
843 elseif command == "UNTIL" then
844 local p, cube = effect(seq), self.cube
845 for k = 1, order(p) do
846 cube = permute(cube, p)
847 if misplaced(cube) == 0 then
848 emit("until %d", k)
849 return
850 end
851 end
852 emit("until never")
853 elseif command == "DEFINE" then
854 self.macros[head] = copy(seq)
855 emit("defined %s %d", head, #seq)
856 elseif command == "MIRROR" then
857 local axis = ({LR = 1, UD = 2, FB = 3})[head]
858 local opposites = {R = "L", L = "R", U = "D", D = "U", F = "B", B = "F", r = "l", l = "r", u = "d", d = "u", f = "b", b = "f"}
859 local mirrored = copy(seq)
860 for _, move in ipairs(mirrored) do
861 if turns[move[1]][1] ~= axis then
862 move[2] = 4 - move[2]
863 elseif opposites[move[1]] then
864 move[1], move[2] = opposites[move[1]], 4 - move[2]
865 end
866 end
867 emit("mirror %s", sequence_text(mirrored))
868 elseif command == "ROTATE" then
869 local rotation, rotated = effect(literal_moves(head)), {}
870 for i, move in ipairs(seq) do
871 rotated[i] = conjugate(rotation, move)
872 end
873 emit("rotated %s", sequence_text(rotated))
874 end
875end
876local n = tonumber(io.read("*l"))
877for _ = 1, n do
878 local line = io.read("*l")
879 local ok, message = pcall(function()
880 simulator:command(line)
881 end)
882 if not ok then
883 emit("error: %s", message)
884 end
885end
