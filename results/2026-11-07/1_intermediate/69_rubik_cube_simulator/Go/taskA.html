57func identity() permutation {
58 var p permutation
59 for i := range p {
60 p[i] = i
61 }
62 return p
63}
64func compose(a, b permutation) permutation {
65 var p permutation
66 for i := range p {
67 p[i] = a[b[i]]
68 }
69 return p
70}
71func inversePermutation(p permutation) permutation {
72 var result permutation
73 for i, source := range p {
74 result[source] = i
75 }
76 return result
77}
78func applyPermutation(cube state, p permutation) state {
79 var result state
80 for i := range result {
81 result[i] = cube[p[i]]
82 }
83 return result
84}
85func apply(cube state, seq []move) state {
86 for _, m := range seq {
87 cube = applyPermutation(cube, tables[m])

89 return cube

91func effect(seq []move) permutation {
92 p := identity()
93 for _, m := range seq {
94 p = compose(p, tables[m])

96 return p

98func inverse(seq []move) []move {
99 result := make([]move, len(seq))
100 for i, m := range seq {
101 result[len(seq)-1-i] = move{m.letter, 4 - m.amount}

103 return result

105func copyMoves(seq []move) []move {

108func quarter(v vector, axis int) vector {
109 a, b := (axis+1)%3, (axis+2)%3
110 v[a], v[b] = v[b], -v[a]
111 return v
112}
113func literalMoves(text string) []move {
114 result := []move{}
115 for _, word := range strings.Fields(text) {
116 amount := 1
117 if len(word) > 1 {
118 if word[1] == '2' {
119 amount = 2
120 } else {
121 amount = 3
122 }
123 }
124 result = append(result, move{word[0], amount})
125 }
126 return result
127}
128func makeGroup(names, layout string) group {
129 g := group{names: strings.Fields(names)}
130 for _, row := range strings.Split(layout, ";") {
131 positions := []int{}
132 for _, word := range strings.Fields(row) {
133 positions = append(positions, strings.IndexByte(faces, word[0])*9+int(word[1]-'1'))
134 }
135 g.stickers = append(g.stickers, positions)
136 }
137 return g
138}
139func initialize() {
140 frames := [6][3]vector{
141 {{0, 1, 0}, {1, 0, 0}, {0, 0, 1}},
142 {{1, 0, 0}, {0, 0, -1}, {0, -1, 0}},
143 {{0, 0, 1}, {1, 0, 0}, {0, -1, 0}},
144 {{0, -1, 0}, {1, 0, 0}, {0, 0, -1}},
145 {{-1, 0, 0}, {0, 0, 1}, {0, -1, 0}},
146 {{0, 0, -1}, {-1, 0, 0}, {0, -1, 0}},
147 }
148 places := []place{}
149 numbers := map[place]int{}
150 for _, frame := range frames {
151 for row := -1; row <= 1; row++ {
152 for column := -1; column <= 1; column++ {
153 cubie := frame[0]
154 for axis := 0; axis < 3; axis++ {
155 cubie[axis] += column*frame[1][axis] + row*frame[2][axis]
156 }
157 p := place{cubie, frame[0]}
158 numbers[p] = len(places)
159 places = append(places, p)
160 }
161 }
162 }
163 tables, byEffect = map[move]permutation{}, map[permutation]move{}
164 for letter, t := range turns {
165 p := identity()
166 for source, original := range places {
167 selected := false
168 for _, layer := range t.layers {
169 if original.cubie[t.axis] == layer {
170 selected = true
171 }
172 }
173 if selected {
174 destination := original
175 for i := 0; i < t.count; i++ {
176 destination.cubie = quarter(destination.cubie, t.axis)
177 destination.normal = quarter(destination.normal, t.axis)
178 }
179 p[numbers[destination]] = source
180 }
181 }
182 current := identity()
183 for amount := 1; amount <= 3; amount++ {
184 current = compose(current, p)
185 m := move{letter, amount}
186 tables[m], byEffect[current] = current, m
187 }
188 }
189 for _, first := range []string{"", "x", "x2", "x'", "z", "z'"} {
190 for _, second := range []string{"", "y", "y2", "y'"} {
191 seq := literalMoves(first + " " + second)
192 rotations = append(rotations, seq)
193 rotationEffects = append(rotationEffects, effect(seq))
194 }
195 }
196 for i := range solved {
197 solved[i] = colors[i/9]
198 }
199 corners = makeGroup("URF UFL ULB UBR DFR DLF DBL DRB", "U9 R1 F3;U7 F1 L3;U1 L1 B3;U3 B1 R3;D3 F9 R7;D1 L9 F7;D7 B9 L7;D9 R9 B7")
200 edges = makeGroup("UR UF UL UB DR DF DL DB FR FL BL BR", "U6 R2;U8 F2;U4 L2;U2 B2;D6 R8;D2 F8;D4 L8;D8 B8;F6 R4;F4 L6;B6 L4;B4 R6")
201 centers = makeGroup("U R F D L B", "U5;R5;F5;D5;L5;B5")
202}

225func normalize(cube state) (state, int, bool) {
226 for i, p := range rotationEffects {
227 candidate := applyPermutation(cube, p)
228 good := true
229 for face := 0; face < 6; face++ {
230 if candidate[9*face+4] != colors[face] {
231 good = false
232 break
233 }
234 }
235 if good {
236 return candidate, i, true
237 }
238 }
239 return cube, 0, false
240}
241func identify(cube state, g group) []piece {
242 result := make([]piece, len(g.names))
243 for position, stickers := range g.stickers {
244 result[position] = piece{-1, 0}
245 found := false
246 for id, name := range g.names {
247 for orientation := 0; orientation < len(stickers); orientation++ {
248 same := true
249 for i := range stickers {
250 color := colors[strings.IndexByte(faces, name[i])]
251 if cube[stickers[(orientation+i)%len(stickers)]] != color {
252 same = false
253 break
254 }
255 }
256 if same {
257 result[position] = piece{id, orientation}
258 found = true
259 break
260 }
261 }
262 if found {
263 break
264 }
265 }
266 }
267 return result
268}
269func cycles(cube state, g group) string {
270 identified := identify(cube, g)
271 where, orientation := make([]int, len(g.names)), make([]int, len(g.names))
272 for position, p := range identified {
273 where[p.id], orientation[p.id] = position, p.orientation
274 }
275 seen := make([]bool, len(g.names))
276 written := []string{}
277 for first := range g.names {
278 if seen[first] {
279 continue
280 }
281 names, sum, at := []string{}, 0, first
282 for !seen[at] {
283 seen[at] = true
284 names = append(names, g.names[at])
285 sum += orientation[at]
286 at = where[at]
287 }
288 mark := ""
289 if width := len(g.stickers[0]); width > 1 {
290 mark = []string{"", "+", "-"}[sum%width]
291 }
292 if len(names) > 1 || mark != "" {
293 written = append(written, "("+strings.Join(names, " ")+")"+mark)
294 }
295 }
296 if len(written) == 0 {
297 return "none"
298 }
299 return strings.Join(written, " ")
300}
301func parity(pieces []piece) int {
302 result := 0
303 for i, a := range pieces {
304 for _, b := range pieces[i+1:] {
305 if a.id > b.id {
306 result ^= 1
307 }
308 }
309 }
310 return result
311}
312func check(word string) string {
313 if len(word) != 54 {
314 return fmt.Sprintf("length %d", len(word))
315 }
316 counts := [6]int{}
317 for i := range word {
318 at := strings.IndexByte(colors, word[i])
319 if at < 0 {
320 return fmt.Sprintf("char %d", i+1)
321 }
322 counts[at]++
323 }
324 for i, count := range counts {
325 if count != 9 {
326 return fmt.Sprintf("count %c %d", colors[i], count)
327 }
328 }
329 var cube state
330 copy(cube[:], word)
331 normalized, _, ok := normalize(cube)
332 if !ok {
333 return "centers"
334 }
335 c, e := identify(cube, corners), identify(cube, edges)
336 for i, p := range c {
337 if p.id < 0 {
338 return "corner " + corners.names[i]
339 }
340 }
341 for i, p := range e {
342 if p.id < 0 {
343 return "edge " + edges.names[i]
344 }
345 }
346 for _, pair := range []struct {
347 g group
348 pieces []piece
349 }{{corners, c}, {edges, e}} {
350 for id, name := range pair.g.names {
351 found := false
352 for _, p := range pair.pieces {
353 if p.id == id {
354 found = true
355 break
356 }
357 }
358 if !found {
359 return "missing " + name
360 }
361 }
362 }
363 c, e = identify(normalized, corners), identify(normalized, edges)
364 twist, flip := 0, 0
365 for _, p := range c {
366 twist += p.orientation
367 }
368 for _, p := range e {
369 flip += p.orientation
370 }
371 if twist%3 != 0 {
372 return "twist"
373 }
374 if flip%2 != 0 {
375 return "flip"
376 }
377 if parity(c) != parity(e) {
378 return "parity"
379 }
380 return ""
381}

402func simplify(seq []move) []move {
403 for {
404 result := []move{}
405 for start := 0; start < len(seq); {
406 end := start + 1
407 for end < len(seq) && turns[seq[end].letter].axis == turns[seq[start].letter].axis {
408 end++
409 }
410 order := []byte{}
411 amounts := map[byte]int{}
412 for _, m := range seq[start:end] {
413 if _, exists := amounts[m.letter]; !exists {
414 order = append(order, m.letter)
415 }
416 amounts[m.letter] += m.amount
417 }
418 for _, letter := range order {
419 if amount := amounts[letter] % 4; amount != 0 {
420 result = append(result, move{letter, amount})
421 }
422 }
423 start = end
424 }
425 if sameMoves(seq, result) {
426 return result
427 }
428 seq = result
429 }
430}
431func gcd(a, b int64) int64 {
432 for b != 0 {
433 a, b = b, a%b
434 }
435 return a
436}
437func order(p permutation) int64 {
438 seen := [54]bool{}
439 result := int64(1)
440 for first := range p {
441 if seen[first] {
442 continue
443 }
444 length, at := int64(0), first
445 for !seen[at] {
446 seen[at] = true
447 length++
448 at = p[at]
449 }
450 result = result / gcd(result, length) * length
451 }
452 return result
453}
454func conjugate(rotation permutation, m move) move {
455 return byEffect[compose(compose(rotation, tables[m]), inversePermutation(rotation))]
456}
457func unrotate(seq []move) []move {
458 replacements := map[byte][]move{
459 'M': literalMoves("R L' x'"), 'E': literalMoves("U D' y'"), 'S': literalMoves("F' B z"),
460 'r': literalMoves("L x"), 'l': literalMoves("R x'"), 'u': literalMoves("D y"),
461 'd': literalMoves("U y'"), 'f': literalMoves("B z"), 'b': literalMoves("F z'"),
462 }
463 result, rotation := []move{}, identity()
464 for _, m := range seq {
465 expanded, exists := replacements[m.letter]
466 if exists {
467 expanded = copyMoves(expanded)
468 for i := range expanded {
469 expanded[i].amount = expanded[i].amount * m.amount % 4
470 }
471 } else {
472 expanded = []move{m}
473 }
474 for _, part := range expanded {
475 if strings.ContainsRune("xyz", rune(part.letter)) {
476 rotation = compose(rotation, tables[part])
477 } else {
478 result = append(result, conjugate(rotation, part))
479 }
480 }
481 }
482 for i, p := range rotationEffects {
483 if p == rotation {
484 return append(result, rotations[i]...)
485 }
486 }
487 return result
488}
489func search(cube state, depth int, previous byte) ([]move, bool) {
490 if depth == 0 {
491 return []move{}, misplaced(cube) == 0
492 }
493 for i := range faces {
494 face := faces[i]
495 if face == previous || previous == 'D' && face == 'U' || previous == 'L' && face == 'R' || previous == 'B' && face == 'F' {
496 continue
497 }
498 for amount := 1; amount <= 3; amount++ {
499 m := move{face, amount}
500 if rest, found := search(applyPermutation(cube, tables[m]), depth-1, face); found {
501 return append([]move{m}, rest...), true
502 }
503 }
504 }
505 return nil, false
506}

508type parser struct {
509 line string
510 pos int
511 macros map[string][]move
512}
513
514func (p *parser) peek() byte {
515 if p.pos < len(p.line) {
516 return p.line[p.pos]
517 }
518 return 0
519}
520func (p *parser) skip() {
521 for p.peek() == ' ' {
522 p.pos++
523 }
524}
525func (p *parser) syntax() error {
526 position := p.pos + 1
527 if position > len(p.line)+1 {
528 position = len(p.line) + 1
529 }
530 return fmt.Errorf("syntax at %d", position)
531}
532func (p *parser) power(seq []move) []move {
533 count := 1
534 if p.peek() >= '1' && p.peek() <= '9' {
535 count = int(p.peek() - '0')
536 p.pos++
537 if p.peek() >= '0' && p.peek() <= '9' {
538 count = count*10 + int(p.peek()-'0')
539 p.pos++
540 }
541 }
542 invert := p.peek() == '\''
543 if invert {
544 p.pos++
545 }
546 result := []move{}
547 for i := 0; i < count; i++ {
548 result = append(result, seq...)
549 }
550 if invert {
551 return inverse(result)
552 }
553 return result
554}
555func (p *parser) sequence() ([]move, error) {
556 result := []move{}
557 for {
558 p.skip()
559 c := p.peek()
560 var part []move
561 if _, exists := turns[c]; exists {
562 p.pos++
563 amount := 1
564 if p.peek() == '2' {
565 amount = 2
566 p.pos++
567 } else if p.peek() == '\'' {
568 amount = 3
569 p.pos++
570 }
571 part = []move{{c, amount}}
572 } else if c == '(' {
573 p.pos++
574 nested, err := p.sequence()
575 if err != nil {
576 return nil, err
577 }
578 p.skip()
579 if p.peek() != ')' {
580 return nil, p.syntax()
581 }
582 p.pos++
583 part = p.power(nested)
584 } else if c == '[' {
585 p.pos++
586 a, err := p.sequence()
587 if err != nil {
588 return nil, err
589 }
590 p.skip()
591 separator := p.peek()
592 if separator != ',' && separator != ':' {
593 return nil, p.syntax()
594 }
595 p.pos++
596 b, err := p.sequence()
597 if err != nil {
598 return nil, err
599 }
600 p.skip()
601 if p.peek() != ']' {
602 return nil, p.syntax()
603 }
604 p.pos++
605 nested := append(copyMoves(a), b...)
606 nested = append(nested, inverse(a)...)
607 if separator == ',' {
608 nested = append(nested, inverse(b)...)
609 }
610 part = p.power(nested)
611 } else if c == '{' {
612 p.pos++
613 start := p.pos
614 for p.peek() >= 'a' && p.peek() <= 'z' {
615 p.pos++
616 }
617 name := p.line[start:p.pos]
618 if name == "" || p.peek() != '}' {
619 return nil, p.syntax()
620 }
621 nested, exists := p.macros[name]
622 if !exists {
623 return nil, fmt.Errorf("undefined %s", name)
624 }
625 p.pos++
626 part = p.power(nested)
627 } else {
628 return result, nil
629 }
630 result = append(result, part...)
631 }
632}
633func (p *parser) whole() ([]move, error) {
634 result, err := p.sequence()
635 if err != nil {
636 return nil, err
637 }
638 p.skip()
639 if p.peek() != 0 {
640 return nil, p.syntax()
641 }
642 if len(result) > 500 {
643 return nil, fmt.Errorf("too long")
644 }
645 return result, nil
646}
647func nameWord(word string) bool {
648 if len(word) == 0 || len(word) > 8 {
649 return false
650 }
651 for i := range word {
652 if word[i] < 'a' || word[i] > 'z' {
653 return false
654 }
655 }
656 return true
657}
658func numberWord(word string) bool {
659 if len(word) == 0 || len(word) > 10 || len(word) > 1 && word[0] == '0' {
660 return false
661 }
662 for i := range word {
663 if word[i] < '0' || word[i] > '9' {
664 return false
665 }
666 }
667 return true
668}
669func numeric(word string) int64 {
670 n, _ := strconv.ParseInt(word, 10, 64)
671 return n
672}

674 for _, s := range words {
675 if s == word {
676 return true
677 }

695func (s *simulator) command(line string) error {
696 command, arguments, hasArguments := strings.Cut(line, " ")
697 noArguments := contains([]string{"RESET", "SHOW", "STATE", "ISSOLVED", "HISTORY", "SOLUTION", "ANALYZE", "STAGE", "MACROS"}, command)
698 wordCounts := map[string]int{"CHECK": 1, "LOAD": 1, "UNDO": 1, "REDO": 1, "SEARCH": 1, "WHERE": 1, "SAVE": 1, "RESTORE": 1, "SCRAMBLE": 2}
699 count, wordCommand := wordCounts[command]
700 sequenceCommand := contains([]string{"APPLY", "INVERT", "SIMPLIFY", "METRICS", "ORDER", "CYCLES", "UNROTATE", "UNTIL"}, command)
701 headCommand := contains([]string{"DEFINE", "MIRROR", "ROTATE"}, command)
702 if !noArguments && !wordCommand && !sequenceCommand && !headCommand {
703 return fmt.Errorf("unknown command %s", command)
704 }
705 words, head := strings.Fields(arguments), ""
706 seqStart := len(command) + 1
707 badArguments := noArguments && hasArguments || wordCommand && len(words) != count || headCommand && !hasArguments
708 if !badArguments && wordCommand {
709 if contains([]string{"UNDO", "REDO", "SEARCH", "SCRAMBLE"}, command) {
710 for _, word := range words {
711 if !numberWord(word) {
712 badArguments = true
713 }
714 }
715 }
716 if (command == "SAVE" || command == "RESTORE") && !nameWord(words[0]) {
717 badArguments = true
718 }
719 }
720 if !badArguments && headCommand {
721 head, _, _ = strings.Cut(arguments, " ")
722 seqStart = len(command) + len(head) + 2
723 switch command {
724 case "DEFINE":
725 badArguments = !nameWord(head)
726 case "MIRROR":
727 badArguments = !contains([]string{"LR", "UD", "FB"}, head)
728 case "ROTATE":
729 badArguments = !contains([]string{"x", "x2", "x'", "y", "y2", "y'", "z", "z2", "z'"}, head)
730 }
731 }
732 if badArguments {
733 return fmt.Errorf("bad arguments")
734 }
735 var seq []move
736 if sequenceCommand || headCommand {
737 p := parser{line: line, pos: seqStart, macros: s.macros}
738 var err error
739 seq, err = p.whole()
740 if err != nil {
741 return err
742 }
743 }
744 if (command == "UNDO" || command == "REDO") && numeric(words[0]) < 1 || command == "SEARCH" && numeric(words[0]) > 4 || command == "SCRAMBLE" && (numeric(words[0]) < 1 || numeric(words[0]) > 50 || numeric(words[1]) < 1 || numeric(words[1]) > 2147483646) {
745 return fmt.Errorf("bad value")
746 }
747 if command == "WHERE" && !contains(corners.names, words[0]) && !contains(edges.names, words[0]) {
748 return fmt.Errorf("unknown piece %s", words[0])
749 }
750 switch command {
751 case "RESET":
752 s.cube, s.history, s.redo = solved, nil, nil
753 fmt.Fprintln(s.out, "reset")
754 case "SHOW":
755 for row := 0; row < 9; row++ {
756 indices, prefix := []int{}, ""
757 if row < 3 || row >= 6 {
758 face, r := 0, row
759 if row >= 6 {
760 face, r = 3, row-6
761 }
762 prefix = " "
763 for column := 0; column < 3; column++ {
764 indices = append(indices, face*9+r*3+column)
765 }
766 } else {
767 for _, face := range []int{4, 2, 1, 5} {
768 for column := 0; column < 3; column++ {
769 indices = append(indices, face*9+(row-3)*3+column)
770 }
771 }
772 }
773 cells := []string{}
774 for _, index := range indices {
775 cells = append(cells, string(s.cube[index]))
776 }
777 fmt.Fprintln(s.out, prefix+strings.Join(cells, " "))
778 }
779 case "STATE":
780 fmt.Fprintln(s.out, "state "+string(s.cube[:]))
781 case "ISSOLVED":
782 if n := misplaced(s.cube); n == 0 {
783 fmt.Fprintln(s.out, "solved")
784 } else {
785 fmt.Fprintf(s.out, "unsolved %d\n", n)
786 }
787 case "HISTORY":
788 fmt.Fprintln(s.out, "history "+sequenceText(s.history))
789 case "SOLUTION":
790 fmt.Fprintln(s.out, "solution "+sequenceText(simplify(inverse(s.history))))
791 case "ANALYZE", "STAGE", "WHERE":
792 cube, rotation, _ := normalize(s.cube)
793 if command == "ANALYZE" {
794 text := movesText(rotations[rotation])
795 if text == "" {
796 text = "none"
797 }
798 fmt.Fprintln(s.out, "rotation "+text)
799 fmt.Fprintln(s.out, "corners "+cycles(cube, corners))
800 fmt.Fprintln(s.out, "edges "+cycles(cube, edges))
801 } else if command == "WHERE" {
802 g, orientation := edges, "flip"
803 if contains(corners.names, words[0]) {
804 g, orientation = corners, "twist"
805 }
806 for position, p := range identify(cube, g) {
807 if g.names[p.id] == words[0] {
808 fmt.Fprintf(s.out, "%s at %s %s %d\n", words[0], g.names[position], orientation, p.orientation)
809 break
810 }
811 }
812 } else {
813 c, e := identify(cube, corners), identify(cube, edges)
814 cross, slots, top := 0, 0, 0
815 for i := 4; i < 8; i++ {
816 if e[i].id == i && e[i].orientation == 0 {
817 cross++
818 }
819 if c[i].id == i && c[i].orientation == 0 && e[i+4].id == i+4 && e[i+4].orientation == 0 {
820 slots++
821 }
822 }
823 for _, color := range cube[:9] {
824 if color == 'W' {
825 top++
826 }
827 }
828 stage := "none"
829 if misplaced(cube) == 0 {
830 stage = "solved"
831 } else if cross == 4 && slots == 4 && top == 9 {
832 stage = "oll"
833 } else if cross == 4 && slots == 4 {
834 stage = "f2l"
835 } else if cross == 4 {
836 stage = "cross"
837 }
838 fmt.Fprintf(s.out, "stage %s cross %d slots %d top %d\n", stage, cross, slots, top)
839 }
840 case "MACROS":
841 names := []string{}
842 for name := range s.macros {
843 names = append(names, name)
844 }
845 sort.Strings(names)
846 fmt.Fprintf(s.out, "macros %d\n", len(names))
847 for _, name := range names {
848 fmt.Fprintf(s.out, "macro %s %s\n", name, sequenceText(s.macros[name]))
849 }
850 case "CHECK", "LOAD":
851 invalid := check(words[0])
852 if command == "CHECK" {
853 if invalid == "" {
854 fmt.Fprintln(s.out, "valid")
855 } else {
856 fmt.Fprintln(s.out, "invalid "+invalid)
857 }
858 } else {
859 if invalid != "" {
860 return fmt.Errorf("invalid %s", invalid)
861 }
862 copy(s.cube[:], words[0])
863 s.history, s.redo = nil, nil
864 fmt.Fprintln(s.out, "loaded")
865 }
866 case "UNDO", "REDO":
867 source := &s.history
868 message := "undo"
869 if command == "REDO" {
870 source, message = &s.redo, "redo"
871 }
872 if len(*source) == 0 {
873 return fmt.Errorf("nothing to %s", message)
874 }
875 count := numeric(words[0])
876 if count > int64(len(*source)) {
877 count = int64(len(*source))
878 }
879 for i := int64(0); i < count; i++ {
880 m := (*source)[len(*source)-1]
881 *source = (*source)[:len(*source)-1]
882 if command == "UNDO" {
883 s.cube = applyPermutation(s.cube, tables[move{m.letter, 4 - m.amount}])
884 s.redo = append(s.redo, m)
885 } else {
886 s.cube = applyPermutation(s.cube, tables[m])
887 s.history = append(s.history, m)
888 }
889 }
890 if command == "UNDO" {
891 fmt.Fprintf(s.out, "undone %d\n", count)
892 } else {
893 fmt.Fprintf(s.out, "redone %d\n", count)
894 }
895 case "SEARCH":
896 for depth := 0; depth <= int(numeric(words[0])); depth++ {
897 if found, ok := search(s.cube, depth, 0); ok {
898 fmt.Fprintln(s.out, "found "+sequenceText(found))
899 return nil
900 }
901 }
902 fmt.Fprintln(s.out, "not found")
903 case "SAVE":
904 s.saves[words[0]] = saved{s.cube, copyMoves(s.history)}
905 fmt.Fprintln(s.out, "saved "+words[0])
906 case "RESTORE":
907 save, exists := s.saves[words[0]]
908 if !exists {
909 return fmt.Errorf("no save %s", words[0])
910 }
911 s.cube, s.history, s.redo = save.cube, copyMoves(save.history), nil
912 fmt.Fprintln(s.out, "restored "+words[0])
913 case "SCRAMBLE":
914 scramble, seed := []move{}, numeric(words[1])
915 for i := 0; i < int(numeric(words[0])); i++ {
916 usable := []byte{}
917 for j := range faces {
918 face := faces[j]
919 if i > 0 && face == scramble[i-1].letter {
920 continue
921 }
922 if i > 1 && turns[scramble[i-1].letter].axis == turns[scramble[i-2].letter].axis && turns[face].axis == turns[scramble[i-1].letter].axis {
923 continue
924 }
925 usable = append(usable, face)
926 }
927 seed = 48271 * seed % 2147483647
928 face := usable[seed%int64(len(usable))]
929 seed = 48271 * seed % 2147483647
930 scramble = append(scramble, move{face, []int{1, 3, 2}[seed%3]})
931 }
932 s.record(scramble)
933 fmt.Fprintln(s.out, "scramble "+movesText(scramble))
934 case "APPLY":
935 s.record(seq)
936 fmt.Fprintf(s.out, "applied %d\n", len(seq))
937 case "INVERT":
938 fmt.Fprintln(s.out, "inverse "+sequenceText(inverse(seq)))
939 case "SIMPLIFY":
940 fmt.Fprintln(s.out, "simplified "+sequenceText(simplify(seq)))
941 case "METRICS":
942 htm, qtm, stm := 0, 0, 0
943 for _, m := range seq {
944 if strings.ContainsRune("xyz", rune(m.letter)) {
945 continue
946 }
947 h, q := 1, 1
948 if strings.ContainsRune("MES", rune(m.letter)) {
949 h = 2
950 }
951 if m.amount == 2 {
952 q = 2
953 }
954 htm, qtm, stm = htm+h, qtm+h*q, stm+1
955 }
956 fmt.Fprintf(s.out, "metrics htm %d qtm %d stm %d etm %d\n", htm, qtm, stm, len(seq))
957 case "ORDER":
958 fmt.Fprintf(s.out, "order %d\n", order(effect(seq)))
959 case "CYCLES":
960 cube := apply(solved, seq)
961 fmt.Fprintln(s.out, "corners "+cycles(cube, corners))
962 fmt.Fprintln(s.out, "edges "+cycles(cube, edges))
963 fmt.Fprintln(s.out, "centers "+cycles(cube, centers))
964 case "UNROTATE":
965 fmt.Fprintln(s.out, "unrotated "+sequenceText(unrotate(seq)))
966 case "UNTIL":
967 p, cube := effect(seq), s.cube
968 for k, limit := int64(1), order(p); k <= limit; k++ {
969 cube = applyPermutation(cube, p)
970 if misplaced(cube) == 0 {
971 fmt.Fprintf(s.out, "until %d\n", k)
972 return nil
973 }
974 }
975 fmt.Fprintln(s.out, "until never")
976 case "DEFINE":
977 s.macros[head] = copyMoves(seq)
978 fmt.Fprintf(s.out, "defined %s %d\n", head, len(seq))
979 case "MIRROR":
980 axis := map[string]int{"LR": 0, "UD": 1, "FB": 2}[head]
981 opposites := map[byte]byte{'R': 'L', 'L': 'R', 'U': 'D', 'D': 'U', 'F': 'B', 'B': 'F', 'r': 'l', 'l': 'r', 'u': 'd', 'd': 'u', 'f': 'b', 'b': 'f'}
982 mirrored := copyMoves(seq)
983 for i, m := range mirrored {
984 if turns[m.letter].axis != axis {
985 mirrored[i].amount = 4 - m.amount
986 } else if opposite, exists := opposites[m.letter]; exists {
987 mirrored[i] = move{opposite, 4 - m.amount}
988 }
989 }
990 fmt.Fprintln(s.out, "mirror "+sequenceText(mirrored))
991 case "ROTATE":
992 rotation, rotated := effect(literalMoves(head)), copyMoves(seq)
993 for i, m := range seq {
994 rotated[i] = conjugate(rotation, m)
995 }
996 fmt.Fprintln(s.out, "rotated "+sequenceText(rotated))
997 }
998 return nil
999}
