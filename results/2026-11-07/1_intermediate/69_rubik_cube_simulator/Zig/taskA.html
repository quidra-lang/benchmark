60fn split(gpa: Allocator, text: Text, separator: u8) Allocator.Error![]Text {
61 var result: std.ArrayList(Text) = .empty;
62 var start: usize = 0;
63 for (text, 0..) |c, i| if (c == separator) {
64 try result.append(gpa, text[start..i]);
65 start = i + 1;
66 };
67 try result.append(gpa, text[start..]);
68 return result.items;
69}
70fn join(gpa: Allocator, parts: []const Text, separator: Text) Allocator.Error!Text {
71 var out: Builder = .{ .gpa = gpa };
72 for (parts, 0..) |part, i| {
73 if (i > 0) try out.add(separator);
74 try out.add(part);
75 }
76 return out.result();

103fn quarter(vector: Vector, axis: usize) Vector {
104 var result = vector;
105 const a = (axis + 1) % 3;
106 const b = (axis + 2) % 3;
107 result[a] = vector[b];
108 result[b] = -vector[a];
109 return result;

167fn identify(cube: State, group: Group) Pieces {
168 var result: Pieces = .{ .count = group.names.len };
169 for (group.stickers, 0..) |stickers, position| {
170 result.values[position] = .{};

172 for (0..stickers.len) |orientation| {
173 var same = true;
174 for (stickers, 0..) |_, i| {
175 const face = std.mem.findScalar(u8, faces, name[i]).?;

177 same = false;
178 break;
179 }

182 result.values[position] = .{ .id = id, .orientation = orientation };

184 }
185 }
186 }

190fn cycles(gpa: Allocator, cube: State, group: Group) Error!Text {
191 const identified = identify(cube, group);
192 var where: [12]usize = undefined;
193 var orientations: [12]usize = undefined;
194 var seen: [12]bool = @splat(false);
195 for (identified.values[0..identified.count], 0..) |piece, position| {
196 where[piece.id.?] = position;
197 orientations[piece.id.?] = piece.orientation;
198 }
199 var written: std.ArrayList(Text) = .empty;
200 for (0..group.names.len) |first| {
201 if (seen[first]) continue;
202 var names: std.ArrayList(Text) = .empty;
203 var sum: usize = 0;
204 var at = first;
205 while (!seen[at]) {
206 seen[at] = true;
207 try names.append(gpa, group.names[at]);
208 sum += orientations[at];
209 at = where[at];
210 }
211 const width = group.stickers[0].len;
212 const mark: Text = if (width == 1 or sum % width == 0) "" else if (sum % width == 1) "+" else "-";
213 if (names.items.len > 1 or mark.len != 0) try written.append(gpa, try std.fmt.allocPrint(gpa, "({s}){s}", .{ try join(gpa, names.items, " "), mark }));
214 }
215 return if (written.items.len == 0) "none" else join(gpa, written.items, " ");
216}
217fn parity(pieces: Pieces) u1 {
218 var result: u1 = 0;
219 for (pieces.values[0..pieces.count], 0..) |a, i| {
220 for (pieces.values[i + 1 .. pieces.count]) |b| if (a.id.? > b.id.?) {
221 result ^= 1;
222 };
223 }
224 return result;
225}
226fn sameMoves(a: []const Move, b: []const Move) bool {
227 if (a.len != b.len) return false;
228 for (a, b) |x, y| if (x.letter != y.letter or x.amount != y.amount) return false;
229 return true;
230}
231fn simplify(gpa: Allocator, sequence: []const Move) Error![]const Move {
232 var current = sequence;
233 while (true) {
234 var result: std.ArrayList(Move) = .empty;
235 var start: usize = 0;
236 while (start < current.len) {
237 var end = start + 1;
238 while (end < current.len and turn(current[end].letter).axis == turn(current[start].letter).axis) end += 1;
239 var letter_order: std.ArrayList(u8) = .empty;
240 var amounts: [256]usize = @splat(0);
241 var seen: [256]bool = @splat(false);
242 for (current[start..end]) |move| {
243 if (!seen[move.letter]) {
244 seen[move.letter] = true;
245 try letter_order.append(gpa, move.letter);
246 }
247 amounts[move.letter] += move.amount;
248 }
249 for (letter_order.items) |letter| if (amounts[letter] % 4 != 0) {
250 try result.append(gpa, .{ .letter = letter, .amount = @intCast(amounts[letter] % 4) });
251 };
252 start = end;
253 }
254 if (sameMoves(current, result.items)) return result.items;
255 current = result.items;
256 }
257}
258fn gcd(first: u64, second: u64) u64 {
259 var a = first;
260 var b = second;
261 while (b != 0) {
262 const remainder = a % b;
263 a = b;
264 b = remainder;
265 }
266 return a;
267}
268fn order(p: Permutation) u64 {
269 var seen: [54]bool = @splat(false);
270 var result: u64 = 1;
271 for (0..54) |first| {
272 if (seen[first]) continue;
273 var length: u64 = 0;
274 var at = first;
275 while (!seen[at]) {
276 seen[at] = true;
277 length += 1;
278 at = p[at];
279 }
280 result = result / gcd(result, length) * length;
281 }
282 return result;
283}
284const Model = struct {
285 gpa: Allocator,
286 tables: [18][3]Permutation = undefined,
287 rotations: [24]Rotation = undefined,
288 solved: State = undefined,
289 fn table(self: *const Model, move: Move) Permutation {
290 return self.tables[std.mem.findScalar(u8, letters, move.letter).?][move.amount - 1];
291 }
292 fn effect(self: *const Model, moves: []const Move) Permutation {
293 var result = identity();
294 for (moves) |move| result = compose(result, self.table(move));
295 return result;
296 }
297 fn apply(self: *const Model, cube: State, moves: []const Move) State {
298 var result = cube;
299 for (moves) |move| result = permute(result, self.table(move));
300 return result;
301 }
302 fn findMove(self: *const Model, p: Permutation) Move {
303 for (letters, 0..) |letter, i| for (self.tables[i], 0..) |candidate, j| {
304 if (std.mem.eql(u8, &p, &candidate)) return .{ .letter = letter, .amount = @intCast(j + 1) };
305 };
306 unreachable;
307 }
308 fn conjugate(self: *const Model, rotation: Permutation, move: Move) Move {
309 return self.findMove(compose(compose(rotation, self.table(move)), inversePermutation(rotation)));
310 }
311 fn init(self: *Model) Error!void {
312 const frames = [6][3]Vector{
313 .{ .{ 0, 1, 0 }, .{ 1, 0, 0 }, .{ 0, 0, 1 } },
314 .{ .{ 1, 0, 0 }, .{ 0, 0, -1 }, .{ 0, -1, 0 } },
315 .{ .{ 0, 0, 1 }, .{ 1, 0, 0 }, .{ 0, -1, 0 } },
316 .{ .{ 0, -1, 0 }, .{ 1, 0, 0 }, .{ 0, 0, -1 } },
317 .{ .{ -1, 0, 0 }, .{ 0, 0, 1 }, .{ 0, -1, 0 } },
318 .{ .{ 0, 0, -1 }, .{ -1, 0, 0 }, .{ 0, -1, 0 } },
319 };
320 var places: [54]Place = undefined;
321 var next: usize = 0;
322 for (frames) |frame| {
323 for (0..3) |row| for (0..3) |column| {
324 var cubie = frame[0];
325 for (0..3) |axis| cubie[axis] += (@as(i8, @intCast(column)) - 1) * frame[1][axis] + (@as(i8, @intCast(row)) - 1) * frame[2][axis];
326 places[next] = .{ .cubie = cubie, .normal = frame[0] };
327 next += 1;
328 };
329 }
330 for (letters, 0..) |letter, i| {
331 const t = turn(letter);
332 var p = identity();
333 for (places, 0..) |original, source| {
334 if (std.mem.findScalar(i8, t.layers, original.cubie[t.axis]) == null) continue;
335 var destination = original;
336 for (0..t.count) |_| {
337 destination.cubie = quarter(destination.cubie, t.axis);
338 destination.normal = quarter(destination.normal, t.axis);
339 }
340 for (places, 0..) |candidate, j| {
341 if (std.mem.eql(i8, &destination.cubie, &candidate.cubie) and std.mem.eql(i8, &destination.normal, &candidate.normal)) {
342 p[j] = @intCast(source);
343 break;
344 }
345 }
346 }
347 var current = identity();
348 for (0..3) |amount| {
349 current = compose(current, p);
350 self.tables[i][amount] = current;
351 }
352 }
353 next = 0;
354 for ([_]Text{ "", "x", "x2", "x'", "z", "z'" }) |first| {
355 for ([_]Text{ "", "y", "y2", "y'" }) |second| {
356 const moves = try literalMoves(self.gpa, try std.fmt.allocPrint(self.gpa, "{s} {s}", .{ first, second }));
357 self.rotations[next] = .{ .moves = moves, .effect = self.effect(moves) };
358 next += 1;
359 }
360 }
361 for (&self.solved, 0..) |*color, i| color.* = colors[i / 9];
362 }
363 fn normalize(self: *const Model, cube: State) ?Normalized {
364 for (self.rotations, 0..) |rotation, i| {
365 const candidate = permute(cube, rotation.effect);
366 var good = true;
367 for (0..6) |face| if (candidate[face * 9 + 4] != colors[face]) {
368 good = false;
369 break;
370 };
371 if (good) return .{ .cube = candidate, .rotation = i };
372 }
373 return null;
374 }
375 fn check(self: *const Model, word: Text) Error!?Text {
376 const gpa = self.gpa;
377 if (word.len != 54) return try std.fmt.allocPrint(gpa, "length {d}", .{word.len});
378 var counts: [6]usize = @splat(0);
379 for (word, 0..) |color, i| {
380 const at = std.mem.findScalar(u8, colors, color) orelse return try std.fmt.allocPrint(gpa, "char {d}", .{i + 1});
381 counts[at] += 1;
382 }
383 for (counts, 0..) |count, i| if (count != 9) return try std.fmt.allocPrint(gpa, "count {c} {d}", .{ colors[i], count });
384 var cube: State = undefined;
385 @memcpy(&cube, word);
386 const normalized = self.normalize(cube) orelse return "centers";
387 var c = identify(cube, corner_group);
388 var e = identify(cube, edge_group);
389 for (c.values[0..c.count], 0..) |piece, i| if (piece.id == null) return try std.fmt.allocPrint(gpa, "corner {s}", .{corner_group.names[i]});
390 for (e.values[0..e.count], 0..) |piece, i| if (piece.id == null) return try std.fmt.allocPrint(gpa, "edge {s}", .{edge_group.names[i]});
391 for ([_]Group{ corner_group, edge_group }, [_]Pieces{ c, e }) |group, pieces| {
392 for (group.names, 0..) |name, id| {
393 var found = false;
394 for (pieces.values[0..pieces.count]) |piece| if (piece.id.? == id) {
395 found = true;
396 break;
397 };
398 if (!found) return try std.fmt.allocPrint(gpa, "missing {s}", .{name});
399 }
400 }
401 c = identify(normalized.cube, corner_group);
402 e = identify(normalized.cube, edge_group);
403 var twist: usize = 0;
404 var flip: usize = 0;
405 for (c.values[0..c.count]) |piece| twist += piece.orientation;
406 for (e.values[0..e.count]) |piece| flip += piece.orientation;
407 if (twist % 3 != 0) return "twist";
408 if (flip % 2 != 0) return "flip";
409 if (parity(c) != parity(e)) return "parity";
410 return null;
411 }
412 fn unrotate(self: *const Model, moves: []const Move) Error![]Move {
413 var result: std.ArrayList(Move) = .empty;
414 var rotation = identity();
415 for (moves) |move| {
416 const replacement: ?Text = switch (move.letter) {
417 'M' => "R L' x'",
418 'E' => "U D' y'",
419 'S' => "F' B z",
420 'r' => "L x",
421 'l' => "R x'",
422 'u' => "D y",
423 'd' => "U y'",
424 'f' => "B z",
425 'b' => "F z'",
426 else => null,
427 };
428 var expanded: []const Move = undefined;
429 if (replacement) |text| {
430 const parts = try literalMoves(self.gpa, text);
431 for (parts) |*part| part.amount = part.amount * move.amount % 4;
432 expanded = parts;
433 } else {
434 const single = try self.gpa.alloc(Move, 1);
435 single[0] = move;
436 expanded = single;
437 }
438 for (expanded) |part| {
439 if (std.mem.findScalar(u8, "xyz", part.letter) != null) rotation = compose(rotation, self.table(part)) else try result.append(self.gpa, self.conjugate(rotation, part));
440 }
441 }
442 for (self.rotations) |candidate| {
443 if (std.mem.eql(u8, &rotation, &candidate.effect)) {
444 try result.appendSlice(self.gpa, candidate.moves);
445 return result.items;
446 }
447 }
448 unreachable;
449 }
450 fn search(self: *const Model, cube: State, depth: usize, previous: u8) Error!?[]Move {
451 if (depth == 0) return if (misplaced(cube) == 0) try self.gpa.alloc(Move, 0) else null;
452 for (faces) |face| {
453 if (face == previous or (previous == 'D' and face == 'U') or (previous == 'L' and face == 'R') or (previous == 'B' and face == 'F')) continue;
454 for (1..4) |amount| {
455 const move: Move = .{ .letter = face, .amount = @intCast(amount) };
456 if (try self.search(permute(cube, self.table(move)), depth - 1, face)) |rest| {
457 const result = try self.gpa.alloc(Move, rest.len + 1);
458 result[0] = move;
459 @memcpy(result[1..], rest);
460 return result;
461 }
462 }
463 }
464 return null;
465 }

467const Parser = struct {
468 gpa: Allocator,
469 line: Text,
470 pos: usize,
471 macros: *const Macros,
472 fn peek(self: *const Parser) u8 {
473 return if (self.pos < self.line.len) self.line[self.pos] else 0;
474 }
475 fn skip(self: *Parser) void {
476 while (self.peek() == ' ') self.pos += 1;
477 }
478 fn syntax(self: *const Parser) Error {
479 return bad(self.gpa, "syntax at {d}", .{@min(self.pos + 1, self.line.len + 1)});
480 }
481 fn power(self: *Parser, moves: []const Move) Error![]Move {
482 var count: usize = 1;
483 if (self.peek() >= '1' and self.peek() <= '9') {
484 count = self.peek() - '0';
485 self.pos += 1;
486 if (self.peek() >= '0' and self.peek() <= '9') {
487 count = count * 10 + self.peek() - '0';
488 self.pos += 1;
489 }
490 }
491 const invert = self.peek() == '\'';
492 if (invert) self.pos += 1;
493 var result: std.ArrayList(Move) = .empty;
494 for (0..count) |_| try result.appendSlice(self.gpa, moves);
495 return if (invert) inverse(self.gpa, result.items) else result.items;
496 }
497 fn sequence(self: *Parser) Error![]Move {
498 var result: std.ArrayList(Move) = .empty;
499 while (true) {
500 self.skip();
501 const current = self.peek();
502 var part: []const Move = undefined;
503 if (std.mem.findScalar(u8, letters, current) != null) {
504 self.pos += 1;
505 var amount: u8 = 1;
506 if (self.peek() == '2') {
507 amount = 2;
508 self.pos += 1;
509 } else if (self.peek() == '\'') {
510 amount = 3;
511 self.pos += 1;
512 }
513 const single = try self.gpa.alloc(Move, 1);
514 single[0] = .{ .letter = current, .amount = amount };
515 part = single;
516 } else if (current == '(') {
517 self.pos += 1;
518 const nested = try self.sequence();
519 self.skip();
520 if (self.peek() != ')') return self.syntax();
521 self.pos += 1;
522 part = try self.power(nested);
523 } else if (current == '[') {
524 self.pos += 1;
525 const a = try self.sequence();
526 self.skip();
527 const separator = self.peek();
528 if (separator != ',' and separator != ':') return self.syntax();
529 self.pos += 1;
530 const b = try self.sequence();
531 self.skip();
532 if (self.peek() != ']') return self.syntax();
533 self.pos += 1;
534 var nested: std.ArrayList(Move) = .empty;
535 try nested.appendSlice(self.gpa, a);
536 try nested.appendSlice(self.gpa, b);
537 try nested.appendSlice(self.gpa, try inverse(self.gpa, a));
538 if (separator == ',') try nested.appendSlice(self.gpa, try inverse(self.gpa, b));
539 part = try self.power(nested.items);
540 } else if (current == '{') {
541 self.pos += 1;
542 const start = self.pos;
543 while (self.peek() >= 'a' and self.peek() <= 'z') self.pos += 1;
544 const name = self.line[start..self.pos];
545 if (name.len == 0 or self.peek() != '}') return self.syntax();
546 const nested = self.macros.get(name) orelse return bad(self.gpa, "undefined {s}", .{name});
547 self.pos += 1;
548 part = try self.power(nested);
549 } else return result.items;
550 try result.appendSlice(self.gpa, part);
551 }
552 }
553 fn whole(self: *Parser) Error![]Move {
554 const result = try self.sequence();
555 self.skip();
556 if (self.peek() != 0) return self.syntax();
557 if (result.len > 500) return bad(self.gpa, "too long", .{});
558 return result;
559 }
560};
561fn nameWord(word: Text) bool {
562 if (word.len == 0 or word.len > 8) return false;
563 for (word) |c| if (c < 'a' or c > 'z') return false;
564 return true;
565}
566fn numberWord(word: Text) bool {
567 if (word.len == 0 or word.len > 10 or (word.len > 1 and word[0] == '0')) return false;
568 for (word) |c| if (!std.ascii.isDigit(c)) return false;
569 return true;
570}
571fn number(word: Text) u64 {
572 return std.fmt.parseInt(u64, word, 10) catch unreachable;
573}
574const command_names = [_]Text{ "RESET", "SHOW", "STATE", "ISSOLVED", "HISTORY", "SOLUTION", "ANALYZE", "STAGE", "MACROS", "CHECK", "LOAD", "UNDO", "REDO", "SEARCH", "WHERE", "SAVE", "RESTORE", "SCRAMBLE", "APPLY", "INVERT", "SIMPLIFY", "METRICS", "ORDER", "CYCLES", "UNROTATE", "UNTIL", "DEFINE", "MIRROR", "ROTATE" };
575const Command = enum { reset, show, state, issolved, history, solution, analyze, stage, macros, check, load, undo, redo, search, where, save, restore, scramble, apply, invert, simplify, metrics, order, cycles, unrotate, until, define, mirror, rotate };
576const Simulator = struct {
577 gpa: Allocator,
578 model: *const Model,
579 cube: State,
580 out: Builder,
581 history: std.ArrayList(Move) = .empty,
582 redo: std.ArrayList(Move) = .empty,
583 macros: Macros = .empty,
584 saves: std.StringHashMapUnmanaged(Saved) = .empty,
585 fn record(self: *Simulator, moves: []const Move) Error!void {
586 self.cube = self.model.apply(self.cube, moves);
587 try self.history.appendSlice(self.gpa, moves);
588 self.redo = .empty;
589 }
590 fn command(self: *Simulator, line: Text) Error!void {
591 const gpa = self.gpa;
592 const at = std.mem.findScalar(u8, line, ' ');
593 const name = if (at) |i| line[0..i] else line;
594 const arguments = if (at) |i| line[i + 1 ..] else "";
595 var found: ?Command = null;
596 for (command_names, 0..) |candidate, i| if (equal(candidate, name)) {
597 found = @fromBackingInt(@intCast(i));
598 break;
599 };
600 const cmd = found orelse return bad(gpa, "unknown command {s}", .{name});
601 const no_arguments = @backingInt(cmd) <= @backingInt(Command.macros);
602 const word_command = @backingInt(cmd) >= @backingInt(Command.check) and @backingInt(cmd) <= @backingInt(Command.scramble);
603 const sequence_command = @backingInt(cmd) >= @backingInt(Command.apply) and @backingInt(cmd) <= @backingInt(Command.until);
604 const head_command = @backingInt(cmd) >= @backingInt(Command.define);
605 const words: []const Text = if (arguments.len == 0) &.{} else try split(gpa, arguments, ' ');
606 const count: usize = if (cmd == .scramble) 2 else 1;
607 var invalid = (no_arguments and at != null) or (word_command and words.len != count) or (head_command and at == null);
608 if (!invalid and word_command) {
609 if (cmd == .undo or cmd == .redo or cmd == .search or cmd == .scramble) {
610 for (words) |word| if (!numberWord(word)) {
611 invalid = true;
612 };
613 }
614 if ((cmd == .save or cmd == .restore) and !nameWord(words[0])) invalid = true;
615 }
616 var head: Text = "";
617 var seq_start = name.len + 1;
618 if (!invalid and head_command) {
619 head = words[0];
620 seq_start = name.len + head.len + 2;
621 invalid = switch (cmd) {
622 .define => !nameWord(head),
623 .mirror => !contains(&.{ "LR", "UD", "FB" }, head),
624 .rotate => !contains(&.{ "x", "x2", "x'", "y", "y2", "y'", "z", "z2", "z'" }, head),
625 else => unreachable,
626 };
627 }
628 if (invalid) return bad(gpa, "bad arguments", .{});
629 var moves: []const Move = &.{};
630 if (sequence_command or head_command) {
631 var parser: Parser = .{ .gpa = gpa, .line = line, .pos = seq_start, .macros = &self.macros };
632 moves = try parser.whole();
633 }
634 if (((cmd == .undo or cmd == .redo) and number(words[0]) < 1) or (cmd == .search and number(words[0]) > 4) or (cmd == .scramble and (number(words[0]) < 1 or number(words[0]) > 50 or number(words[1]) < 1 or number(words[1]) > 2147483646))) return bad(gpa, "bad value", .{});
635 if (cmd == .where and !contains(corner_group.names, words[0]) and !contains(edge_group.names, words[0])) return bad(gpa, "unknown piece {s}", .{words[0]});
636 switch (cmd) {
637 .reset => {
638 self.cube = self.model.solved;
639 self.history = .empty;
640 self.redo = .empty;
641 try self.out.add("reset\n");
642 },
643 .show => {
644 for (0..9) |row| {
645 var cells: Builder = .{ .gpa = gpa };
646 var first = true;
647 if (row < 3 or row >= 6) {
648 try cells.add(" ");
649 const face: usize = if (row < 3) 0 else 3;
650 const r = if (row < 3) row else row - 6;
651 for (0..3) |column| {
652 if (!first) try cells.byte(' ');
653 first = false;
654 try cells.byte(self.cube[face * 9 + r * 3 + column]);
655 }
656 } else {
657 for ([_]usize{ 4, 2, 1, 5 }) |face| for (0..3) |column| {
658 if (!first) try cells.byte(' ');
659 first = false;
660 try cells.byte(self.cube[face * 9 + (row - 3) * 3 + column]);
661 };
662 }
663 try self.out.line("{s}", .{cells.result()});
664 }
665 },
666 .state => try self.out.line("state {s}", .{&self.cube}),
667 .issolved => {
668 const n = misplaced(self.cube);
669 if (n == 0) try self.out.add("solved\n") else try self.out.line("unsolved {d}", .{n});
670 },
671 .history => try self.out.line("history {s}", .{try sequenceText(gpa, self.history.items)}),
672 .solution => try self.out.line("solution {s}", .{try sequenceText(gpa, try simplify(gpa, try inverse(gpa, self.history.items)))}),
673 .analyze, .stage, .where => {
674 const normalized = self.model.normalize(self.cube).?;
675 const cube = normalized.cube;
676 if (cmd == .analyze) {
677 const rotation = self.model.rotations[normalized.rotation].moves;
678 try self.out.line("rotation {s}", .{if (rotation.len == 0) "none" else try movesText(gpa, rotation)});
679 try self.out.line("corners {s}", .{try cycles(gpa, cube, corner_group)});
680 try self.out.line("edges {s}", .{try cycles(gpa, cube, edge_group)});
681 } else if (cmd == .where) {
682 const group = if (contains(corner_group.names, words[0])) corner_group else edge_group;
683 const pieces = identify(cube, group);
684 for (pieces.values[0..pieces.count], 0..) |piece, position| {
685 if (equal(group.names[piece.id.?], words[0])) {
686 try self.out.line("{s} at {s} {s} {d}", .{ words[0], group.names[position], if (group.names.len == 8) "twist" else "flip", piece.orientation });
687 break;
688 }
689 }
690 } else {
691 const c = identify(cube, corner_group);
692 const e = identify(cube, edge_group);
693 var cross: usize = 0;
694 var slots: usize = 0;
695 var top: usize = 0;
696 for (4..8) |i| {
697 if (e.values[i].id.? == i and e.values[i].orientation == 0) cross += 1;
698 if (c.values[i].id.? == i and c.values[i].orientation == 0 and e.values[i + 4].id.? == i + 4 and e.values[i + 4].orientation == 0) slots += 1;
699 }
700 for (cube[0..9]) |color| if (color == 'W') {
701 top += 1;
702 };
703 const stage: Text = if (misplaced(cube) == 0) "solved" else if (cross == 4 and slots == 4 and top == 9) "oll" else if (cross == 4 and slots == 4) "f2l" else if (cross == 4) "cross" else "none";
704 try self.out.line("stage {s} cross {d} slots {d} top {d}", .{ stage, cross, slots, top });
705 }
706 },
707 .macros => {
708 var names: std.ArrayList(Text) = .empty;
709 var iterator = self.macros.keyIterator();
710 while (iterator.next()) |name_ptr| try names.append(gpa, name_ptr.*);
711 std.mem.sort(Text, names.items, {}, textLess);
712 try self.out.line("macros {d}", .{names.items.len});
713 for (names.items) |macro_name| try self.out.line("macro {s} {s}", .{ macro_name, try sequenceText(gpa, self.macros.get(macro_name).?) });
714 },
715 .check, .load => {
716 const invalidity = try self.model.check(words[0]);
717 if (cmd == .check) {
718 if (invalidity) |text| try self.out.line("invalid {s}", .{text}) else try self.out.add("valid\n");
719 } else {
720 if (invalidity) |text| return bad(gpa, "invalid {s}", .{text});
721 @memcpy(&self.cube, words[0]);
722 self.history = .empty;
723 self.redo = .empty;
724 try self.out.add("loaded\n");
725 }
726 },
727 .undo, .redo => {
728 const source = if (cmd == .undo) &self.history else &self.redo;
729 if (source.items.len == 0) return bad(gpa, "nothing to {s}", .{if (cmd == .undo) "undo" else "redo"});
730 const n: usize = @intCast(@min(number(words[0]), source.items.len));
731 for (0..n) |_| {
732 const move = source.items[source.items.len - 1];
733 source.items.len -= 1;
734 if (cmd == .undo) {
735 self.cube = permute(self.cube, self.model.table(.{ .letter = move.letter, .amount = 4 - move.amount }));
736 try self.redo.append(gpa, move);
737 } else {
738 self.cube = permute(self.cube, self.model.table(move));
739 try self.history.append(gpa, move);
740 }
741 }
742 try self.out.line("{s} {d}", .{ if (cmd == .undo) "undone" else "redone", n });
743 },
744 .search => {
745 const limit: usize = @intCast(number(words[0]));
746 for (0..limit + 1) |depth| {
747 if (try self.model.search(self.cube, depth, 0)) |solution| {
748 try self.out.line("found {s}", .{try sequenceText(gpa, solution)});
749 return;
750 }
751 }
752 try self.out.add("not found\n");
753 },
754 .save => {
755 try self.saves.put(gpa, words[0], .{ .cube = self.cube, .history = try gpa.dupe(Move, self.history.items) });
756 try self.out.line("saved {s}", .{words[0]});
757 },
758 .restore => {
759 const save = self.saves.get(words[0]) orelse return bad(gpa, "no save {s}", .{words[0]});
760 self.cube = save.cube;
761 self.history = .empty;
762 try self.history.appendSlice(gpa, save.history);
763 self.redo = .empty;
764 try self.out.line("restored {s}", .{words[0]});
765 },
766 .scramble => {
767 var scramble: std.ArrayList(Move) = .empty;
768 var seed = number(words[1]);
769 const n: usize = @intCast(number(words[0]));
770 for (0..n) |i| {
771 var usable: std.ArrayList(u8) = .empty;
772 for (faces) |face| {
773 if (i > 0 and face == scramble.items[i - 1].letter) continue;
774 if (i > 1 and turn(scramble.items[i - 1].letter).axis == turn(scramble.items[i - 2].letter).axis and turn(face).axis == turn(scramble.items[i - 1].letter).axis) continue;
775 try usable.append(gpa, face);
776 }
777 seed = 48271 * seed % 2147483647;
778 const face = usable.items[@intCast(seed % usable.items.len)];
779 seed = 48271 * seed % 2147483647;
780 const amounts = [_]u8{ 1, 3, 2 };
781 try scramble.append(gpa, .{ .letter = face, .amount = amounts[@intCast(seed % 3)] });
782 }
783 try self.record(scramble.items);
784 try self.out.line("scramble {s}", .{try movesText(gpa, scramble.items)});
785 },
786 .apply => {
787 try self.record(moves);
788 try self.out.line("applied {d}", .{moves.len});
789 },
790 .invert => try self.out.line("inverse {s}", .{try sequenceText(gpa, try inverse(gpa, moves))}),
791 .simplify => try self.out.line("simplified {s}", .{try sequenceText(gpa, try simplify(gpa, moves))}),
792 .metrics => {
793 var htm: usize = 0;
794 var qtm: usize = 0;
795 var stm: usize = 0;
796 for (moves) |move| {
797 if (std.mem.findScalar(u8, "xyz", move.letter) != null) continue;
798 const h: usize = if (std.mem.findScalar(u8, "MES", move.letter) != null) 2 else 1;
799 const q: usize = if (move.amount == 2) 2 else 1;
800 htm += h;
801 qtm += h * q;
802 stm += 1;
803 }
804 try self.out.line("metrics htm {d} qtm {d} stm {d} etm {d}", .{ htm, qtm, stm, moves.len });
805 },
806 .order => try self.out.line("order {d}", .{order(self.model.effect(moves))}),
807 .cycles => {
808 const cube = self.model.apply(self.model.solved, moves);
809 try self.out.line("corners {s}", .{try cycles(gpa, cube, corner_group)});
810 try self.out.line("edges {s}", .{try cycles(gpa, cube, edge_group)});
811 try self.out.line("centers {s}", .{try cycles(gpa, cube, center_group)});
812 },
813 .unrotate => try self.out.line("unrotated {s}", .{try sequenceText(gpa, try self.model.unrotate(moves))}),
814 .until => {
815 const p = self.model.effect(moves);
816 const limit = order(p);
817 var cube = self.cube;
818 var k: u64 = 1;
819 while (k <= limit) : (k += 1) {
820 cube = permute(cube, p);
821 if (misplaced(cube) == 0) {
822 try self.out.line("until {d}", .{k});
823 return;
824 }
825 }
826 try self.out.add("until never\n");
827 },
828 .define => {
829 try self.macros.put(gpa, head, try gpa.dupe(Move, moves));
830 try self.out.line("defined {s} {d}", .{ head, moves.len });
831 },
832 .mirror => {
833 const axis: usize = if (equal(head, "LR")) 0 else if (equal(head, "UD")) 1 else 2;
834 const mirrored = try gpa.dupe(Move, moves);
835 for (mirrored) |*move| {
836 const opposite: ?u8 = switch (move.letter) {
837 'R' => 'L',
838 'L' => 'R',
839 'U' => 'D',
840 'D' => 'U',
841 'F' => 'B',
842 'B' => 'F',
843 'r' => 'l',
844 'l' => 'r',
845 'u' => 'd',
846 'd' => 'u',
847 'f' => 'b',
848 'b' => 'f',
849 else => null,
850 };
851 if (turn(move.letter).axis != axis) move.amount = 4 - move.amount else if (opposite) |letter| {
852 move.letter = letter;
853 move.amount = 4 - move.amount;
854 }
855 }
856 try self.out.line("mirror {s}", .{try sequenceText(gpa, mirrored)});
857 },
858 .rotate => {
859 const rotation = self.model.effect(try literalMoves(gpa, head));
860 const rotated = try gpa.alloc(Move, moves.len);
861 for (moves, 0..) |move, i| rotated[i] = self.model.conjugate(rotation, move);
862 try self.out.line("rotated {s}", .{try sequenceText(gpa, rotated)});
863 },
864 }
865 }
866};
867fn textLess(_: void, a: Text, b: Text) bool {
868 return std.mem.order(u8, a, b) == .lt;
869}
870pub fn main(init: std.process.Init) !void {
871 var storage = std.heap.ArenaAllocator.init(std.heap.page_allocator);
872 defer storage.deinit();
873 const gpa = storage.allocator();
874 var input_buffer: [4096]u8 = undefined;
875 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
876 var output_buffer: [4096]u8 = undefined;
877 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
878 const text = try input.interface.allocRemaining(gpa, .unlimited);
879 const lines = try split(gpa, text, '\n');
880 var model: Model = .{ .gpa = gpa };
881 try model.init();
882 var simulator: Simulator = .{ .gpa = gpa, .model = &model, .cube = model.solved, .out = .{ .gpa = gpa } };
883 const count = std.fmt.parseInt(usize, lines[0], 10) catch unreachable;
884 for (lines[1 .. count + 1]) |line| {
885 simulator.command(line) catch |err| {
886 if (err != error.Invalid) return err;
887 try simulator.out.line("error: {s}", .{reason});
888 };
889 }
890 try output.interface.writeAll(simulator.out.result());
891 try output.interface.flush();
892}
