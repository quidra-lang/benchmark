245 private function parse(string $sequence, int $base, int $lineLength): array
246 {
247 $position = 0;
248 $syntax = function () use (&$position, $base, $lineLength) {

251 $spaces = function () use (&$position, $sequence) {
252 while (($sequence[$position] ?? "") === " ") {
253 $position++;
254 }

256 $power = function ($moves) use (&$position, $sequence) {
257 $count = 1;
258 if (preg_match("/[1-9]/", $sequence[$position] ?? "")) {
259 $count = (int) $sequence[$position++];

261 $count = 10 * $count + (int) $sequence[$position++];
262 }
263 }
264 $output = [];

266 array_push($output, ...$moves);
267 }
268 if (($sequence[$position] ?? "") === "'") {
269 $position++;

271 }
272 return $output;
273 };
274 $read = null;
275 $read = function () use (&$read, &$position, $sequence, $spaces, $syntax, $power) {
276 $output = [];
277 while (true) {
278 $spaces();
279 $char = $sequence[$position] ?? "";
280 if ($char !== "" && str_contains(self::LETTERS, $char)) {
281 $position++;
282 $amount = 1;
283 if (in_array($sequence[$position] ?? "", ["2", "'"], true)) {
284 $amount = $sequence[$position++] === "2" ? 2 : 3;
285 }
286 $output[] = [$char, $amount];
287 } elseif ($char === "(") {
288 $position++;
289 $moves = $read();
290 $spaces();
291 if (($sequence[$position] ?? "") !== ")") {
292 $syntax();
293 }
294 $position++;
295 array_push($output, ...$power($moves));
296 } elseif ($char === "[") {
297 $position++;
298 $a = $read();
299 $spaces();
300 $delimiter = $sequence[$position] ?? "";
301 if (!in_array($delimiter, [",", ":"], true)) {
302 $syntax();
303 }
304 $position++;
305 $b = $read();
306 $spaces();
307 if (($sequence[$position] ?? "") !== "]") {
308 $syntax();
309 }
310 $position++;
311 $moves = [...$a, ...$b, ...$this->inverse($a)];
312 if ($delimiter === ",") {
313 array_push($moves, ...$this->inverse($b));
314 }
315 array_push($output, ...$power($moves));
316 } elseif ($char === "{") {
317 $position++;
318 $start = $position;
319 while (preg_match("/[a-z]/", $sequence[$position] ?? "")) {
320 $position++;
321 }
322 $name = substr($sequence, $start, $position - $start);
323 if ($name === "" || ($sequence[$position] ?? "") !== "}") {
324 $syntax();
325 }
326 if (!array_key_exists($name, $this->macros)) {
327 $this->fail("undefined $name");
328 }
329 $position++;
330 array_push($output, ...$power($this->macros[$name]));
331 } else {
332 break;
333 }
334 }
335 return $output;
336 };
337 $moves = $read();
338 $spaces();
339 if ($position < strlen($sequence)) {
340 $syntax();
341 }
342 if (count($moves) > 500) {
343 $this->fail("too long");
344 }
345 return $moves;
346 }

500 private function cycles(array $state, int $kind): string
501 {
502 $pieces = $this->identify($state, $kind);
503 $destinations = [];
504 $orientations = [];
505 foreach ($pieces as $index => [$piece, $twist]) {
506 $destinations[$piece] = $index;
507 $orientations[$piece] = $twist;

510 $output = [];

513 continue;

517 do {
518 $cycle[] = $current;

525 if (count($cycle) >= 2 || $mark !== "") {
526 $output[] =
527 "(" .
528 implode(" ", array_map(fn($piece) => $this->names[$kind][$piece], $cycle)) .
529 ")" .

531 }

535 private function order(array $permutation): int
536 {
537 $seen = [];
538 $result = 1;
539 foreach ($permutation as $start => $destination) {
540 if (isset($seen[$start])) {
541 continue;
542 }
543 $current = $start;
544 $length = 0;
545 do {
546 $seen[$current] = true;
547 $length++;
548 $current = $permutation[$current];
549 } while ($current !== $start);
550 $a = $result;
551 $b = $length;
552 while ($b !== 0) {
553 $remainder = $a % $b;
554 $a = $b;
555 $b = $remainder;
556 }
557 $result = intdiv($result, $a) * $length;
558 }
559 return $result;

619 private function search(int $depth): ?array
620 {
621 $candidates = [];
622 foreach (self::FACES as $face) {
623 for ($amount = 1; $amount <= 3; $amount++) {
624 $candidates[] = [$face, $amount];
625 };
626 }
627 $visit = null;
628 $visit = function ($state, $left, $previous) use (&$visit, $candidates) {
629 if ($this->solved($state)) {
630 return [];
631 }
632 if ($left === 0) {
633 return null;
634 }
635 foreach ($candidates as $move) {
636 $face = $move[0];
637 if (
638 $face === $previous ||
639 in_array([$previous, $face], [["D", "U"], ["L", "R"], ["B", "F"]], true)
640 ) {
641 continue;
642 }
643 $tail = $visit(
644 $this->permute($state, $this->moves[$face][$move[1]]),
645 $left - 1,
646 $face,
647 );
648 if ($tail !== null) {
649 return [$move, ...$tail];
650 }
651 }
652 return null;
653 };
654 for ($limit = 0; $limit <= $depth; $limit++) {
655 $found = $visit($this->state, $limit, null);
656 if ($found !== null) {
657 return $found;
658 }
659 }
660 return null;

702 if ($argument === null) {
703 $this->fail("bad arguments");

713 }
714 $moves = $this->parse(
715 $sequence,
716 strlen($command) + strlen($head) + 2,
717 strlen($line),
718 );
719 } elseif (in_array($command, self::SEQUENCES, true)) {
720 $moves = $this->parse($argument ?? "", strlen($command) + 1, strlen($line));
721 }
722 if (in_array($command, ["UNDO", "REDO"], true) && (int) $words[0] < 1) {
723 $this->fail("bad value");
724 }
725 if ($command === "SEARCH" && (int) $words[0] > 4) {
726 $this->fail("bad value");
727 }
728 if (
729 $command === "SCRAMBLE" &&
730 ((int) $words[0] < 1 ||
731 (int) $words[0] > 50 ||
732 (int) $words[1] < 1 ||
733 (int) $words[1] > 2147483646)
734 ) {
735 $this->fail("bad value");
736 }
737 if (
738 $command === "WHERE" &&
739 !in_array($words[0], [...self::CORNERS, ...self::EDGES], true)
740 ) {
741 $this->fail("unknown piece $words[0]");
742 }
743 switch ($command) {
744 case "RESET":
745 $this->state = $this->solved;
746 $this->history = [];
747 $this->redo = [];
748 echo "reset\n";
749 break;
750 case "SHOW":
751 foreach (["U", "LFRB", "D"] as $faces) {
752 for ($row = 1; $row <= 3; $row++) {
753 $colors = [];
754 foreach (str_split($faces) as $face) {
755 for ($col = 1; $col <= 3; $col++) {
756 $colors[] = $this->state[$this->index($face, $row, $col)];
757 };
758 }
759 echo (strlen($faces) === 1 ? " " : "") .
760 implode(" ", $colors) .
761 "\n";
762 };
763 }
764 break;
765 case "STATE":
766 echo "state " . implode("", $this->state) . "\n";
767 break;
768 case "ISSOLVED":
769 $count = 0;
770 foreach ($this->state as $i => $color) {
771 if ($color !== $this->state[intdiv($i, 9) * 9 + 4]) {
772 $count++;
773 }
774 }
775 echo $count === 0 ? "solved\n" : "unsolved $count\n";
776 break;
777 case "HISTORY":
778 echo "history " . $this->notation($this->history) . "\n";
779 break;
780 case "SOLUTION":
781 echo "solution " .
782 $this->notation($this->simplify($this->inverse($this->history))) .
783 "\n";
784 break;
785 case "ANALYZE":
786 [$rotation, $state] = $this->normalize($this->state);
787 echo "rotation " .
788 ($rotation === 0 ? "none" : $this->moveWords($this->rotations[$rotation])) .
789 "\n";
790 echo "corners " .
791 $this->cycles($state, 0) .
792 "\nedges " .
793 $this->cycles($state, 1) .
794 "\n";
795 break;
796 case "STAGE":
797 [, $state] = $this->normalize($this->state);
798 $corners = $this->identify($state, 0);
799 $edges = $this->identify($state, 1);
800 $cross = 0;
801 $slots = 0;
802 for ($i = 4; $i <= 7; $i++) {
803 if ($edges[$i] === [$i, 0]) {
804 $cross++;
805 }
806 if ($corners[$i] === [$i, 0] && $edges[$i + 4] === [$i + 4, 0]) {
807 $slots++;
808 }
809 }
810 $top = count(
811 array_filter(array_slice($state, 0, 9), fn($color) => $color === "W"),
812 );
813 $stage = $this->solved($state)
814 ? "solved"
815 : ($cross === 4 && $slots === 4 && $top === 9
816 ? "oll"
817 : ($cross === 4 && $slots === 4
818 ? "f2l"
819 : ($cross === 4
820 ? "cross"
821 : "none")));
822 echo "stage $stage cross $cross slots $slots top $top\n";
823 break;
824 case "MACROS":
825 echo "macros " . count($this->macros) . "\n";
826 $macros = $this->macros;
827 ksort($macros, SORT_STRING);
828 foreach ($macros as $name => $sequence) {
829 echo "macro $name " . $this->notation($sequence) . "\n";
830 }
831 break;
832 case "CHECK":
833 case "LOAD":
834 $state = str_split($words[0]);
835 $reason = $this->invalid($state);
836 if ($command === "CHECK") {
837 echo $reason === null ? "valid\n" : "invalid $reason\n";
838 } else {
839 if ($reason !== null) {
840 $this->fail("invalid $reason");
841 }
842 $this->state = $state;
843 $this->history = [];
844 $this->redo = [];
845 echo "loaded\n";
846 }
847 break;
848 case "UNDO":
849 case "REDO":
850 $source = $command === "UNDO" ? $this->history : $this->redo;
851 if ($source === []) {
852 $this->fail($command === "UNDO" ? "nothing to undo" : "nothing to redo");
853 }
854 $count = min((int) $words[0], count($source));
855 for ($i = 0; $i < $count; $i++) {
856 $move =
857 $command === "UNDO"
858 ? array_pop($this->history)
859 : array_pop($this->redo);
860 $this->state = $this->permute(
861 $this->state,
862 $this->moves[$move[0]][$command === "UNDO" ? 4 - $move[1] : $move[1]],
863 );
864 if ($command === "UNDO") {
865 $this->redo[] = $move;
866 } else {
867 $this->history[] = $move;
868 }
869 }
870 echo ($command === "UNDO" ? "undone " : "redone ") . $count . "\n";
871 break;
872 case "SEARCH":
873 $found = $this->search((int) $words[0]);
874 echo $found === null
875 ? "not found\n"
876 : "found " . $this->notation($found) . "\n";
877 break;
878 case "WHERE":
879 [, $state] = $this->normalize($this->state);
880 $kind = in_array($words[0], self::CORNERS, true) ? 0 : 1;
881 $wanted = array_search($words[0], $this->names[$kind], true);
882 foreach ($this->identify($state, $kind) as $position => [$piece, $twist]) {
883 if ($piece === $wanted) {
884 echo "$words[0] at " .
885 $this->names[$kind][$position] .
886 ($kind === 0 ? " twist " : " flip ") .
887 $twist .
888 "\n";
889 }
890 }
891 break;
892 case "SAVE":
893 $this->saves[$words[0]] = [$this->state, $this->history];
894 echo "saved $words[0]\n";
895 break;
896 case "RESTORE":
897 if (!isset($this->saves[$words[0]])) {
898 $this->fail("no save $words[0]");
899 }
900 [$this->state, $this->history] = $this->saves[$words[0]];
901 $this->redo = [];
902 echo "restored $words[0]\n";
903 break;
904 case "SCRAMBLE":
905 case "APPLY":
906 if ($command === "SCRAMBLE") {
907 $moves = [];
908 $random = (int) $words[1];
909 for ($i = 0; $i < (int) $words[0]; $i++) {
910 $usable = array_values(
911 array_filter(self::FACES, function ($face) use ($moves) {
912 $n = count($moves);
913 return !($n >= 1 && $face === $moves[$n - 1][0]) &&
914 !(
915 $n >= 2 &&
916 $this->axis($moves[$n - 1][0]) ===
917 $this->axis($moves[$n - 2][0]) &&
918 $this->axis($face) === $this->axis($moves[$n - 1][0])
919 );
920 }),
921 );
922 $random = ($random * 48271) % 2147483647;
923 $face = $usable[$random % count($usable)];
924 $random = ($random * 48271) % 2147483647;
925 $moves[] = [$face, [1, 3, 2][$random % 3]];
926 }
927 }
928 $this->state = $this->permute($this->state, $this->effect($moves));
929 array_push($this->history, ...$moves);
930 $this->redo = [];
931 echo $command === "APPLY"
932 ? "applied " . count($moves) . "\n"
933 : "scramble " . $this->moveWords($moves) . "\n";
934 break;
935 case "INVERT":
936 echo "inverse " . $this->notation($this->inverse($moves)) . "\n";
937 break;
938 case "SIMPLIFY":
939 echo "simplified " . $this->notation($this->simplify($moves)) . "\n";
940 break;
941 case "METRICS":
942 $htm = $qtm = $stm = 0;
943 foreach ($moves as [$letter, $amount]) {
944 if (str_contains("xyz", $letter)) {
945 continue;
946 }
947 $weight = str_contains("MES", $letter) ? 2 : 1;
948 $htm += $weight;
949 $qtm += $weight * ($amount === 2 ? 2 : 1);
950 $stm++;
951 }
952 echo "metrics htm $htm qtm $qtm stm $stm etm " . count($moves) . "\n";
953 break;
954 case "ORDER":
955 echo "order " . $this->order($this->effect($moves)) . "\n";
956 break;
957 case "CYCLES":
958 $state = $this->permute($this->solved, $this->effect($moves));
959 foreach (["corners", "edges", "centers"] as $kind => $name) {
960 echo $name . " " . $this->cycles($state, $kind) . "\n";
961 }
962 break;
963 case "UNROTATE":
964 echo "unrotated " . $this->notation($this->unrotate($moves)) . "\n";
965 break;
966 case "UNTIL":
967 $permutation = $this->effect($moves);
968 $state = $this->state;
969 $found = null;
970 $order = $this->order($permutation);
971 for ($count = 1; $count <= $order; $count++) {
972 $state = $this->permute($state, $permutation);
973 if ($this->solved($state)) {
974 $found = $count;
975 break;
976 }
977 }
978 echo "until " . ($found ?? "never") . "\n";
979 break;
980 case "DEFINE":
981 $this->macros[$head] = $moves;
982 echo "defined $head " . count($moves) . "\n";
983 break;
984 case "MIRROR":
985 $mirrorAxis = ["LR" => "x", "UD" => "y", "FB" => "z"][$head];
986 $transformed = [];
987 foreach ($moves as [$letter, $amount]) {
988 if ($this->axis($letter) !== $mirrorAxis) {
989 $transformed[] = [$letter, 4 - $amount];
990 } elseif (str_contains("MESxyz", $letter)) {
991 $transformed[] = [$letter, $amount];
992 } else {
993 $transformed[] = [
994 strtr($letter, "RLUDFBrludfb", "LRDUBFlrdubf"),
995 4 - $amount,
996 ];
997 }
998 }
999 echo "mirror " . $this->notation($transformed) . "\n";
1000 break;
1001 case "ROTATE":
1002 $amount = strlen($head) === 1 ? 1 : ($head[1] === "2" ? 2 : 3);
1003 $rotation = $this->moves[$head[0]][$amount];
1004 echo "rotated " .
1005 $this->notation(
1006 array_map(fn($move) => $this->converted($move, $rotation), $moves),
1007 ) .
1008 "\n";
1009 break;
1010 }
1011 } catch (CubeFault $error) {
1012 echo $error->getMessage() . "\n";
1013 }
1014 }
1015 public function run(): void
1016 {
1017 $count = (int) fgets(STDIN);
1018 for ($i = 0; $i < $count; $i++) {
1019 $this->command(rtrim(fgets(STDIN), "\r\n"));
1020 }
1021 }
1022}
