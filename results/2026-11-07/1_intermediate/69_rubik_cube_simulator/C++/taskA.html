55Sequence simplify(Sequence sequence) {
56 while (true) {
57 Sequence result;
58 std::size_t position = 0;
59 while (position < sequence.size()) {
60 char current_axis = axis(sequence[position].letter);
61 std::vector<char> order;
62 std::map<char, int> amounts;
63 while (position < sequence.size() && axis(sequence[position].letter) == current_axis) {
64 const Move& move = sequence[position++];
65 if (!amounts.count(move.letter)) order.push_back(move.letter);
66 amounts[move.letter] = (amounts[move.letter] + move.amount) % 4;
67 }
68 for (char letter : order) if (amounts[letter]) result.push_back({letter, amounts[letter]});
69 }
70 if (result == sequence) return result;
71 sequence = std::move(result);
72 }
73}

211struct Placement { int piece = -1, orientation = 0; };
212std::vector<Placement> identify(const State& state, const std::vector<Piece>& pieces) {
213 std::vector<Placement> result(pieces.size());
214 for (std::size_t q = 0; q < pieces.size(); ++q) {
215 std::size_t width = pieces[q].stickers.size();
216 for (std::size_t p = 0; p < pieces.size(); ++p) for (std::size_t t = 0; t < width; ++t) {
217 bool match = true;
218 for (std::size_t k = 0; k < width; ++k)
219 if (state[pieces[q].stickers[(t + k) % width]] != pieces[p].colors[k]) match = false;
220 if (match) result[q] = {static_cast<int>(p), static_cast<int>(t)};
221 }
222 }
223 return result;
224}

226std::string cycles(const State& state, const std::vector<Piece>& pieces) {
227 std::vector<Placement> placements = identify(state, pieces);
228 std::vector<int> destination(pieces.size()), orientation(pieces.size());
229 for (std::size_t q = 0; q < pieces.size(); ++q) {
230 destination[placements[q].piece] = static_cast<int>(q);
231 orientation[placements[q].piece] = placements[q].orientation;
232 }
233 std::vector<bool> seen(pieces.size());
234 std::vector<std::string> result;
235 for (std::size_t start = 0; start < pieces.size(); ++start) {
236 if (seen[start]) continue;
237 std::vector<std::string> names;
238 int sum = 0, position = static_cast<int>(start);
239 do {
240 seen[position] = true;
241 names.push_back(pieces[position].name);
242 sum += orientation[position];
243 position = destination[position];
244 } while (position != static_cast<int>(start));
245 int remainder = sum % static_cast<int>(pieces.front().stickers.size());
246 std::string mark = remainder == 0 ? "" : remainder == 1 ? "+" : "-";
247 if (names.size() < 2 && mark.empty()) continue;
248 std::string text = "(";
249 for (std::size_t i = 0; i < names.size(); ++i) {
250 if (i) text += ' ';
251 text += names[i];
252 }
253 result.push_back(text + ")" + mark);
254 }
255 if (result.empty()) return "none";
256 std::string text;
257 for (std::size_t i = 0; i < result.size(); ++i) {
258 if (i) text += ' ';
259 text += result[i];
260 }
261 return text;
262}

272std::string invalidity(const State& state, const Geometry& geometry) {
273 if (state.size() != 54) return "length " + std::to_string(state.size());
274 for (std::size_t i = 0; i < state.size(); ++i)
275 if (colors.find(state[i]) == std::string::npos) return "char " + std::to_string(i + 1);
276 for (char color : colors) {
277 int count = static_cast<int>(std::count(state.begin(), state.end(), color));
278 if (count != 9) return "count " + std::string(1, color) + " " + std::to_string(count);
279 }
280 std::optional<std::pair<State, Sequence>> normalized = geometry.normalize(state);
281 if (!normalized) return "centers";
282 std::vector<Placement> cp = identify(state, corners), ep = identify(state, edges);
283 for (std::size_t i = 0; i < cp.size(); ++i) if (cp[i].piece < 0) return "corner " + corners[i].name;
284 for (std::size_t i = 0; i < ep.size(); ++i) if (ep[i].piece < 0) return "edge " + edges[i].name;
285 for (std::size_t i = 0; i < corners.size(); ++i)
286 if (std::none_of(cp.begin(), cp.end(), [i](Placement p) { return p.piece == static_cast<int>(i); })) return "missing " + corners[i].name;
287 for (std::size_t i = 0; i < edges.size(); ++i)
288 if (std::none_of(ep.begin(), ep.end(), [i](Placement p) { return p.piece == static_cast<int>(i); })) return "missing " + edges[i].name;
289 cp = identify(normalized->first, corners);
290 ep = identify(normalized->first, edges);
291 int twists = 0, flips = 0;
292 for (Placement p : cp) twists += p.orientation;
293 for (Placement p : ep) flips += p.orientation;
294 if (twists % 3) return "twist";
295 if (flips % 2) return "flip";
296 if (parity(cp) != parity(ep)) return "parity";
297 return "";
298}

300class SequenceReader {
301 const std::string& line;
302 std::size_t position;
303 const std::map<std::string, Sequence>& macros;
304 char peek() const { return position < line.size() ? line[position] : '\0'; }
305 void spaces() { while (peek() == ' ') ++position; }
306 [[noreturn]] void syntax() const { throw std::runtime_error("syntax at " + std::to_string(std::min(position + 1, line.size() + 1))); }
307 Sequence power(const Sequence& sequence) {
308 int count = 1;
309 if (peek() >= '1' && peek() <= '9') {
310 count = peek() - '0';
311 ++position;
312 if (peek() >= '0' && peek() <= '9') {
313 count = 10 * count + peek() - '0';
314 ++position;
315 }
316 }
317 bool invert = peek() == '\'';
318 if (invert) ++position;
319 Sequence result;
320 for (int i = 0; i < count; ++i) append(result, sequence);
321 return invert ? inverse(std::move(result)) : result;
322 }
323 Sequence sequence() {
324 Sequence result;
325 while (true) {
326 spaces();
327 char c = peek();
328 Sequence next;
329 if (c != '\0' && letters.find(c) != std::string::npos) {
330 ++position;
331 int amount = 1;
332 if (peek() == '2' || peek() == '\'') {
333 amount = peek() == '2' ? 2 : 3;
334 ++position;
335 }
336 next.push_back({c, amount});
337 } else if (c == '(') {
338 ++position;
339 Sequence group = sequence();
340 spaces();
341 if (peek() != ')') syntax();
342 ++position;
343 next = power(group);
344 } else if (c == '[') {
345 ++position;
346 Sequence a = sequence();
347 spaces();
348 char separator = peek();
349 if (separator != ',' && separator != ':') syntax();
350 ++position;
351 Sequence b = sequence();
352 spaces();
353 if (peek() != ']') syntax();
354 ++position;
355 Sequence group = a;
356 append(group, b);
357 append(group, inverse(a));
358 if (separator == ',') append(group, inverse(b));
359 next = power(group);
360 } else if (c == '{') {
361 ++position;
362 std::size_t start = position;
363 while (peek() >= 'a' && peek() <= 'z') ++position;
364 std::string name = line.substr(start, position - start);
365 if (name.empty() || peek() != '}') syntax();
366 auto found = macros.find(name);
367 if (found == macros.end()) throw std::runtime_error("undefined " + name);
368 ++position;
369 next = power(found->second);
370 } else break;
371 append(result, next);
372 }
373 return result;
374 }
375public:
376 SequenceReader(const std::string& line, std::size_t position, const std::map<std::string, Sequence>& macros)
377 : line(line), position(position), macros(macros) {}
378 Sequence parse() {
379 Sequence result = sequence();
380 spaces();
381 if (peek() != '\0') syntax();
382 if (result.size() > 500) throw std::runtime_error("too long");
383 return result;
384 }
385};

387std::int64_t order(const Permutation& permutation) {
388 std::array<bool, 54> seen{};
389 std::int64_t result = 1;
390 for (int start = 0; start < 54; ++start) {
391 if (seen[start]) continue;
392 int position = start, count = 0;
393 do {
394 seen[position] = true;
395 ++count;
396 position = permutation[position];
397 } while (position != start);
398 result = std::lcm(result, static_cast<std::int64_t>(count));
399 }
400 return result;
401}

403Sequence unrotate(const Sequence& sequence, const Geometry& geometry) {
404 const std::map<char, Sequence> replacements{
405 {'M', {{'R', 1}, {'L', 3}, {'x', 3}}}, {'E', {{'U', 1}, {'D', 3}, {'y', 3}}},
406 {'S', {{'F', 3}, {'B', 1}, {'z', 1}}}, {'r', {{'L', 1}, {'x', 1}}},
407 {'l', {{'R', 1}, {'x', 3}}}, {'u', {{'D', 1}, {'y', 1}}},
408 {'d', {{'U', 1}, {'y', 3}}}, {'f', {{'B', 1}, {'z', 1}}}, {'b', {{'F', 1}, {'z', 3}}}
409 };
410 Sequence expanded;
411 for (Move move : sequence) {
412 auto replacement = replacements.find(move.letter);
413 if (replacement == replacements.end()) expanded.push_back(move);
414 else for (Move item : replacement->second) {
415 item.amount = item.amount * move.amount % 4;
416 if (item.amount) expanded.push_back(item);
417 }
418 }
419 Permutation orientation = identity();
420 Sequence result;
421 for (Move move : expanded) {
422 if (rotation(move.letter)) orientation = compose(orientation, geometry.effect(move));
423 else result.push_back(geometry.matching(compose(compose(orientation, geometry.effect(move)), inverse_effect(orientation)), true));
424 }
425 for (const Sequence& rotation : geometry.rotations) if (geometry.effect(rotation) == orientation) {
426 append(result, rotation);
427 break;
428 }
429 return result;
430}

463 bool search(const State& current, int remaining, char previous, Sequence& path) const {
464 if (!remaining) return mismatches(current) == 0;

466 if (face == previous || (previous == 'D' && face == 'U') || (previous == 'L' && face == 'R') ||

478 Sequence scramble(int count, std::uint64_t seed) const {
479 Sequence sequence;
480 auto random = [&]() {
481 seed = (48271 * seed) % 2147483647;
482 return seed;
483 };
484 for (int i = 0; i < count; ++i) {
485 std::string usable;
486 for (char face : faces) {
487 if (!sequence.empty() && face == sequence.back().letter) continue;
488 if (sequence.size() >= 2 && axis(sequence.back().letter) == axis(sequence[sequence.size() - 2].letter) &&
489 axis(face) == axis(sequence.back().letter)) continue;
490 usable += face;
491 }
492 char face = usable[random() % usable.size()];
493 std::uint64_t value = random() % 3;
494 sequence.push_back({face, value == 0 ? 1 : value == 1 ? 3 : 2});
495 }

540 "APPLY", "INVERT", "SIMPLIFY", "METRICS", "ORDER", "CYCLES", "UNROTATE", "UNTIL"
541 };
542 bool headed = command == "DEFINE" || command == "MIRROR" || command == "ROTATE";
543 if (!no_arguments.count(command) && !word_commands.count(command) && !sequence_commands.count(command) && !headed)
544 throw std::runtime_error("unknown command " + command);
545 if (no_arguments.count(command) && argument) throw std::runtime_error("bad arguments");
546 std::vector<std::string> words;
547 if (word_commands.count(command)) {
548 words = split_words(argument);
549 if (words.size() != word_commands.at(command)) throw std::runtime_error("bad arguments");
550 if ((command == "UNDO" || command == "REDO" || command == "SEARCH") && !number_word(words[0]))
551 throw std::runtime_error("bad arguments");
552 if (command == "SCRAMBLE" && (!number_word(words[0]) || !number_word(words[1])))
553 throw std::runtime_error("bad arguments");
554 if ((command == "SAVE" || command == "RESTORE") && !name_word(words[0])) throw std::runtime_error("bad arguments");
555 }
556 std::string head;
557 std::size_t sequence_start = argument ? space + 1 : line.size();
558 if (headed) {
559 if (!argument) throw std::runtime_error("bad arguments");
560 std::size_t separator = argument->find(' ');
561 head = argument->substr(0, separator);
562 sequence_start = separator == std::string::npos ? line.size() : space + separator + 2;
563 if (command == "DEFINE" && !name_word(head)) throw std::runtime_error("bad arguments");
564 if (command == "MIRROR" && head != "LR" && head != "UD" && head != "FB") throw std::runtime_error("bad arguments");
565 if (command == "ROTATE" && (head.empty() || !rotation(head[0]) ||
566 (head.size() != 1 && (head.size() != 2 || (head[1] != '\'' && head[1] != '2')))))
567 throw std::runtime_error("bad arguments");
568 }
569 Sequence sequence;
570 if (headed || sequence_commands.count(command)) sequence = SequenceReader(line, sequence_start, macros).parse();
571 std::uint64_t number = 0;
572 if (command == "UNDO" || command == "REDO" || command == "SEARCH" || command == "SCRAMBLE") {
573 number = std::stoull(words[0]);
574 if ((command == "UNDO" || command == "REDO") && number == 0) throw std::runtime_error("bad value");
575 if (command == "SEARCH" && number > 4) throw std::runtime_error("bad value");
576 if (command == "SCRAMBLE" && (number < 1 || number > 50 || std::stoull(words[1]) < 1 ||
577 std::stoull(words[1]) > 2147483646)) throw std::runtime_error("bad value");
578 }
579 if (command == "WHERE") {
580 bool found = false;
581 for (const Piece& p : corners) if (p.name == words[0]) found = true;
582 for (const Piece& p : edges) if (p.name == words[0]) found = true;
583 if (!found) throw std::runtime_error("unknown piece " + words[0]);
584 }
585 if (command == "RESET") {
586 state = solved_state();
587 history.clear();
588 redo.clear();
589 std::cout << "reset\n";
590 } else if (command == "SHOW") show();
591 else if (command == "STATE") std::cout << "state " << state << '\n';
592 else if (command == "ISSOLVED") {
593 int count = mismatches(state);
594 if (count == 0) std::cout << "solved\n";
595 else std::cout << "unsolved " << count << '\n';
596 } else if (command == "HISTORY") std::cout << "history " << sequence_text(history) << '\n';
597 else if (command == "SOLUTION") std::cout << "solution " << sequence_text(simplify(inverse(history))) << '\n';
598 else if (command == "ANALYZE" || command == "STAGE" || command == "WHERE") {
599 std::pair<State, Sequence> normalized = *geometry.normalize(state);
600 if (command == "ANALYZE") {
601 std::cout << "rotation" << (normalized.second.empty() ? " none" : moves_text(normalized.second)) << '\n';
602 std::cout << "corners " << cycles(normalized.first, corners) << '\n';
603 std::cout << "edges " << cycles(normalized.first, edges) << '\n';
604 } else if (command == "STAGE") stage(normalized.first);
605 else {
606 bool corner = std::any_of(corners.begin(), corners.end(), [&](const Piece& p) { return p.name == words[0]; });
607 const std::vector<Piece>& pieces = corner ? corners : edges;
608 std::vector<Placement> placements = identify(normalized.first, pieces);
609 for (std::size_t i = 0; i < placements.size(); ++i) if (pieces[placements[i].piece].name == words[0])
610 std::cout << words[0] << " at " << pieces[i].name << (corner ? " twist " : " flip ") << placements[i].orientation << '\n';
611 }
612 } else if (command == "MACROS") {
613 std::cout << "macros " << macros.size() << '\n';
614 for (const auto& macro : macros) std::cout << "macro " << macro.first << ' ' << sequence_text(macro.second) << '\n';
615 } else if (command == "CHECK" || command == "LOAD") {
616 std::string reason = invalidity(words[0], geometry);
617 if (command == "CHECK") std::cout << (reason.empty() ? "valid" : "invalid " + reason) << '\n';
618 else {
619 if (!reason.empty()) throw std::runtime_error("invalid " + reason);
620 state = words[0];
621 history.clear();
622 redo.clear();
623 std::cout << "loaded\n";
624 }
625 } else if (command == "UNDO" || command == "REDO") {
626 Sequence& source = command == "UNDO" ? history : redo;
627 Sequence& destination = command == "UNDO" ? redo : history;
628 if (source.empty()) throw std::runtime_error(command == "UNDO" ? "nothing to undo" : "nothing to redo");
629 std::size_t count = static_cast<std::size_t>(std::min(number, static_cast<std::uint64_t>(source.size())));
630 for (std::size_t i = 0; i < count; ++i) {
631 Move move = source.back();
632 source.pop_back();
633 destination.push_back(move);
634 if (command == "UNDO") move.amount = 4 - move.amount;
635 state = apply_effect(state, geometry.effect(move));
636 }
637 std::cout << (command == "UNDO" ? "undone " : "redone ") << count << '\n';
638 } else if (command == "SEARCH") {
639 for (int depth = 0; depth <= static_cast<int>(number); ++depth) {
640 Sequence result;
641 if (search(state, depth, '\0', result)) {
642 std::cout << "found " << sequence_text(result) << '\n';
643 return;
644 }
645 }
646 std::cout << "not found\n";
647 } else if (command == "SAVE") {
648 saves[words[0]] = {state, history};
649 std::cout << "saved " << words[0] << '\n';
650 } else if (command == "RESTORE") {
651 auto found = saves.find(words[0]);
652 if (found == saves.end()) throw std::runtime_error("no save " + words[0]);
653 state = found->second.first;
654 history = found->second.second;
655 redo.clear();
656 std::cout << "restored " << words[0] << '\n';
657 } else if (command == "SCRAMBLE") {
658 Sequence result = scramble(static_cast<int>(number), std::stoull(words[1]));
659 apply(result);
660 std::cout << "scramble" << moves_text(result) << '\n';
661 } else if (command == "APPLY") {
662 apply(sequence);
663 std::cout << "applied " << sequence.size() << '\n';
664 } else if (command == "INVERT") std::cout << "inverse " << sequence_text(inverse(sequence)) << '\n';
665 else if (command == "SIMPLIFY") std::cout << "simplified " << sequence_text(simplify(sequence)) << '\n';
666 else if (command == "METRICS") {
667 int htm = 0, qtm = 0, stm = 0;
668 for (Move move : sequence) if (!rotation(move.letter)) {
669 int weight = outer(move.letter) || wide(move.letter) ? 1 : 2;
670 htm += weight;
671 qtm += weight * (move.amount == 2 ? 2 : 1);
672 ++stm;
673 }
674 std::cout << "metrics htm " << htm << " qtm " << qtm << " stm " << stm << " etm " << sequence.size() << '\n';
675 } else if (command == "ORDER") std::cout << "order " << order(geometry.effect(sequence)) << '\n';
676 else if (command == "CYCLES") {
677 State result = geometry.apply(solved_state(), sequence);
678 std::cout << "corners " << cycles(result, corners) << '\n';
679 std::cout << "edges " << cycles(result, edges) << '\n';
680 std::cout << "centers " << cycles(result, centers) << '\n';
681 } else if (command == "UNROTATE") std::cout << "unrotated " << sequence_text(unrotate(sequence, geometry)) << '\n';
682 else if (command == "UNTIL") {
683 Permutation permutation = geometry.effect(sequence);
684 std::int64_t period = order(permutation);
685 State candidate = state;
686 for (std::int64_t i = 1; i <= period; ++i) {
687 candidate = apply_effect(candidate, permutation);
688 if (mismatches(candidate) == 0) {
689 std::cout << "until " << i << '\n';
690 return;
691 }
692 }
693 std::cout << "until never\n";
694 } else if (command == "DEFINE") {
695 macros[head] = sequence;
696 std::cout << "defined " << head << ' ' << sequence.size() << '\n';
697 } else if (command == "MIRROR") {
698 char mirror_axis = head == "LR" ? 'x' : head == "UD" ? 'y' : 'z';
699 static const std::string from = "RLrlUDudFBfb", to = "LRlrDUduBFbf";
700 for (Move& move : sequence) {
701 if (axis(move.letter) == mirror_axis) {
702 if (outer(move.letter) || wide(move.letter)) {
703 move.letter = to[from.find(move.letter)];
704 move.amount = 4 - move.amount;
705 }
706 } else move.amount = 4 - move.amount;
707 }
708 std::cout << "mirror " << sequence_text(sequence) << '\n';
709 } else if (command == "ROTATE") {
710 Move move{head[0], head.size() == 1 ? 1 : head[1] == '2' ? 2 : 3};
711 const Permutation& rho = geometry.effect(move);
712 for (Move& item : sequence)
713 item = geometry.matching(compose(compose(rho, geometry.effect(item)), inverse_effect(rho)));
714 std::cout << "rotated " << sequence_text(sequence) << '\n';
715 }
716 }
717public:
718 void command(const std::string& line) {
719 try { execute(line); }
720 catch (const std::runtime_error& error) { std::cout << "error: " << error.what() << '\n'; }
721 }
722};

728 std::getline(std::cin, line);
729 Cube cube;
730 for (int i = 0; i < count; ++i) {
731 std::getline(std::cin, line);
732 cube.command(line);
733 }
