1class CubeError < StandardError
2end
3
4class Cube
5 FACES = %w[U R F D L B].freeze
6 COLORS = %w[W R G Y O B].freeze
7 LETTERS = 'UDFBLRMESudfblrxyz'
8 CORNERS = %w[URF UFL ULB UBR DFR DLF DBL DRB].freeze
9 EDGES = %w[UR UF UL UB DR DF DL DB FR FL BL BR].freeze
10 CORNER_STICKERS = %w[U9,R1,F3 U7,F1,L3 U1,L1,B3 U3,B1,R3 D3,F9,R7 D1,L9,F7 D7,B9,L7 D9,R9,B7].freeze
11 EDGE_STICKERS = %w[U6,R2 U8,F2 U4,L2 U2,B2 D6,R8 D2,F8 D4,L8 D8,B8 F6,R4 F4,L6 B6,L4 B4,R6].freeze
12 NO_ARGS = %w[RESET SHOW STATE ISSOLVED HISTORY SOLUTION ANALYZE STAGE MACROS].freeze
13 WORDS = { 'CHECK' => 1, 'LOAD' => 1, 'UNDO' => 1, 'REDO' => 1, 'SEARCH' => 1, 'WHERE' => 1, 'SAVE' => 1, 'RESTORE' => 1, 'SCRAMBLE' => 2 }.freeze
14 SEQUENCES = %w[APPLY INVERT SIMPLIFY METRICS ORDER CYCLES UNROTATE UNTIL].freeze
15 HEADS = %w[DEFINE MIRROR ROTATE].freeze
16 def initialize
17 @identity = (0...54).to_a
18 @solved = COLORS.flat_map { |color| [color] * 9 }
19 @state = @solved.dup
20 @history = []
21 @redo = []
22 @macros = {}
23 @saves = {}
24 @moves = {}
25 build_moves
26 @rotations = []
27 [[], [['x', 1]], [['x', 2]], [['x', 3]], [['z', 1]], [['z', 3]]].each do |prefix|
28 [[], [['y', 1]], [['y', 2]], [['y', 3]]].each { |suffix| @rotations << prefix + suffix }
29 end
30 @rotation_effects = @rotations.map { |sequence| effect(sequence) }
31 @positions = [CORNER_STICKERS.map { |group| group.split(',').map { |sticker| sticker_index(sticker) } },
32 EDGE_STICKERS.map { |group| group.split(',').map { |sticker| sticker_index(sticker) } },
33 FACES.map { |face| [index(face, 2, 2)] }]
34 @names = [CORNERS, EDGES, FACES]
35 @piece_colors = @positions.map { |positions| positions.map { |stickers| stickers.map { |i| @solved[i] } } }
36 end
37 def error(message)
38 raise CubeError, "error: #{message}"
39 end
40 def index(face, row, col)
41 FACES.index(face) * 9 + (row - 1) * 3 + col - 1
42 end
43 def sticker_index(word)
44 FACES.index(word[0]) * 9 + word[1].to_i - 1
45 end
46 def compose(first, second)
47 first.map { |destination| second[destination] }
48 end
49 def inverse(sequence)
50 sequence.reverse.map { |letter, amount| [letter, 4 - amount] }
51 end
52 def permute(state, permutation)
53 result = Array.new(54)
54 permutation.each_with_index { |destination, source| result[destination] = state[source] }
55 result
56 end
57 def effect(sequence)
58 sequence.reduce(@identity) { |permutation, move| compose(permutation, @moves[move]) }
59 end
60 def build_moves
61 %w[U R F D L B M E S].each do |letter|
62 p = @identity.dup
63 if FACES.include?(letter)
64 (1..3).each { |i| (1..3).each { |j| p[index(letter, i, j)] = index(letter, j, 4 - i) } }
65 end
66 put = ->(a, i, j, b, k, l) { p[index(a, i, j)] = index(b, k, l) }
67 (1..3).each do |i|
68 case letter
69 when 'U'
70 %w[F L B R].each_with_index { |face, j| put.call(face, 1, i, %w[L B R F][j], 1, i) }
71 when 'D'
72 %w[F R B L].each_with_index { |face, j| put.call(face, 3, i, %w[R B L F][j], 3, i) }
73 when 'R'
74 put.call('F', i, 3, 'U', i, 3)
75 put.call('U', i, 3, 'B', 4 - i, 1)
76 put.call('B', i, 1, 'D', 4 - i, 3)
77 put.call('D', i, 3, 'F', i, 3)
78 when 'L'
79 put.call('U', i, 1, 'F', i, 1)
80 put.call('F', i, 1, 'D', i, 1)
81 put.call('D', i, 1, 'B', 4 - i, 3)
82 put.call('B', i, 3, 'U', 4 - i, 1)
83 when 'F'
84 put.call('U', 3, i, 'R', i, 1)
85 put.call('R', i, 1, 'D', 1, 4 - i)
86 put.call('D', 1, i, 'L', i, 3)
87 put.call('L', i, 3, 'U', 3, 4 - i)
88 when 'B'
89 put.call('U', 1, i, 'L', 4 - i, 1)
90 put.call('L', i, 1, 'D', 3, i)
91 put.call('D', 3, i, 'R', 4 - i, 3)
92 put.call('R', i, 3, 'U', 1, i)
93 when 'M'
94 put.call('U', i, 2, 'F', i, 2)
95 put.call('F', i, 2, 'D', i, 2)
96 put.call('D', i, 2, 'B', 4 - i, 2)
97 put.call('B', i, 2, 'U', 4 - i, 2)
98 when 'E'
99 %w[F R B L].each_with_index { |face, j| put.call(face, 2, i, %w[R B L F][j], 2, i) }
100 when 'S'
101 put.call('U', 2, i, 'R', i, 2)
102 put.call('R', i, 2, 'D', 2, 4 - i)
103 put.call('D', 2, i, 'L', i, 2)
104 put.call('L', i, 2, 'U', 2, 4 - i)
105 end
106 end
107 (1..3).each { |amount| @moves[[letter, amount]] = amount.times.reduce(@identity) { |current, _| compose(current, p) } }
108 end
109 { 'r' => [['R', 1], ['M', 3]], 'l' => [['L', 1], ['M', 1]], 'u' => [['U', 1], ['E', 3]],
110 'd' => [['D', 1], ['E', 1]], 'f' => [['F', 1], ['S', 1]], 'b' => [['B', 1], ['S', 3]],
111 'x' => [['R', 1], ['M', 3], ['L', 3]], 'y' => [['U', 1], ['E', 3], ['D', 3]], 'z' => [['F', 1], ['S', 1], ['B', 3]] }.each do |letter, sequence|
112 p = effect(sequence)
113 (1..3).each { |amount| @moves[[letter, amount]] = amount.times.reduce(@identity) { |current, _| compose(current, p) } }
114 end
115 end
116 def parse(sequence, base, line_length)
117 position = 0
118 syntax = -> { error("syntax at #{[base + position + 1, line_length + 1].min}") }
119 spaces = -> { position += 1 while sequence[position] == ' ' }
120 power = lambda do |moves|
121 count = 1
122 if sequence[position]&.match?(/[1-9]/)
123 count = sequence[position].to_i
124 position += 1
125 if sequence[position]&.match?(/[0-9]/)
126 count = 10 * count + sequence[position].to_i
127 position += 1
128 end
129 end
130 moves *= count
131 if sequence[position] == "'"
132 position += 1
133 moves = inverse(moves)
134 end
135 moves
136 end
137 read = nil
138 read = lambda do
139 output = []
140 loop do
141 spaces.call
142 char = sequence[position]
143 if char && LETTERS.include?(char)
144 position += 1
145 amount = 1
146 if ['2', "'"].include?(sequence[position])
147 amount = sequence[position] == '2' ? 2 : 3
148 position += 1
149 end
150 output << [char, amount]
151 elsif char == '('
152 position += 1
153 moves = read.call
154 spaces.call
155 syntax.call unless sequence[position] == ')'
156 position += 1
157 output.concat(power.call(moves))
158 elsif char == '['
159 position += 1
160 a = read.call
161 spaces.call
162 delimiter = sequence[position]
163 syntax.call unless [',', ':'].include?(delimiter)
164 position += 1
165 b = read.call
166 spaces.call
167 syntax.call unless sequence[position] == ']'
168 position += 1
169 moves = a + b + inverse(a)
170 moves += inverse(b) if delimiter == ','
171 output.concat(power.call(moves))
172 elsif char == '{'
173 position += 1
174 start = position
175 position += 1 while sequence[position]&.match?(/[a-z]/)
176 name = sequence[start...position]
177 syntax.call if name.empty? || sequence[position] != '}'
178 error("undefined #{name}") unless @macros.key?(name)
179 position += 1
180 output.concat(power.call(@macros[name].dup))
181 else
182 break
183 end
184 end
185 output
186 end
187 moves = read.call
188 spaces.call
189 syntax.call if position < sequence.length
190 error('too long') if moves.length > 500
191 moves
192 end
193 def notation(sequence)
194 ([sequence.length.to_s] + sequence.map { |letter, amount| letter + (amount == 1 ? '' : amount == 2 ? '2' : "'") }).join(' ')
195 end
196 def move_words(sequence)
197 sequence.map { |letter, amount| letter + (amount == 1 ? '' : amount == 2 ? '2' : "'") }.join(' ')
198 end
199 def axis(letter)
200 return 'x' if 'RLMrlx'.include?(letter)
201 return 'y' if 'UDEudy'.include?(letter)
202 'z'
203 end
204 def simplify(sequence)
205 loop do
206 output = []
207 position = 0
208 while position < sequence.length
209 current_axis = axis(sequence[position][0])
210 amounts = {}
211 while position < sequence.length && axis(sequence[position][0]) == current_axis
212 letter, amount = sequence[position]
213 amounts[letter] = (amounts.fetch(letter, 0) + amount) % 4
214 position += 1
215 end
216 amounts.each { |letter, amount| output << [letter, amount] unless amount == 0 }
217 end
218 return output if output == sequence
219 sequence = output
220 end
221 end
222 def solved?(state)
223 (0...6).all? { |face| state[face * 9, 9].all? { |color| color == state[face * 9 + 4] } }
224 end
225 def normalize(state)
226 @rotation_effects.each_with_index do |permutation, index|
227 rotated = permute(state, permutation)
228 return [index, rotated] if (0...6).all? { |face| rotated[face * 9 + 4] == COLORS[face] }
229 end
230 nil
231 end
232 def identify(state, kind)
233 @positions[kind].map do |stickers|
234 colors = stickers.map { |i| state[i] }
235 found = nil
236 @piece_colors[kind].each_with_index do |piece, index|
237 (0...piece.length).each { |twist| found = [index, twist] if colors.rotate(twist) == piece }
238 end
239 found
240 end
241 end
242 def invalid(state)
243 return "length #{state.length}" unless state.length == 54
244 bad = state.index { |color| !COLORS.include?(color) }
245 return "char #{bad + 1}" if bad
246 COLORS.each { |color| return "count #{color} #{state.count(color)}" unless state.count(color) == 9 }
247 normalized = normalize(state)
248 return 'centers' unless normalized
249 original = [identify(state, 0), identify(state, 1)]
250 original.each_with_index do |pieces, kind|
251 bad = pieces.index(nil)
252 return "#{kind == 0 ? 'corner' : 'edge'} #{@names[kind][bad]}" if bad
253 end
254 original.each_with_index do |pieces, kind|
255 @names[kind].each_with_index { |name, index| return "missing #{name}" unless pieces.any? { |piece| piece[0] == index } }
256 end
257 corners = identify(normalized[1], 0)
258 edges = identify(normalized[1], 1)
259 return 'twist' unless corners.sum { |piece| piece[1] } % 3 == 0
260 return 'flip' unless edges.sum { |piece| piece[1] } % 2 == 0
261 parity = ->(pieces) { pieces.each_index.sum { |i| ((i + 1)...pieces.length).count { |j| pieces[i][0] > pieces[j][0] } } % 2 }
262 return 'parity' unless parity.call(corners) == parity.call(edges)
263 nil
264 end
265 def cycles(state, kind)
266 pieces = identify(state, kind)
267 destination = Array.new(pieces.length)
268 orientation = Array.new(pieces.length)
269 pieces.each_with_index do |(piece, twist), index|
270 destination[piece] = index
271 orientation[piece] = twist
272 end
273 seen = []
274 output = []
275 pieces.each_index do |start|
276 next if seen.include?(start)
277 cycle = []
278 current = start
279 loop do
280 cycle << current
281 seen << current
282 current = destination[current]
283 break if current == start
284 end
285 sum = cycle.sum { |piece| orientation[piece] }
286 mark = kind == 0 ? ['', '+', '-'][sum % 3] : kind == 1 && sum.odd? ? '+' : ''
287 output << '(' + cycle.map { |piece| @names[kind][piece] }.join(' ') + ')' + mark if cycle.length >= 2 || !mark.empty?
288 end
289 output.empty? ? 'none' : output.join(' ')
290 end
291 def order(permutation)
292 seen = []
293 result = 1
294 permutation.each_index do |start|
295 next if seen.include?(start)
296 current = start
297 length = 0
298 loop do
299 seen << current
300 length += 1
301 current = permutation[current]
302 break if current == start
303 end
304 result = result.lcm(length)
305 end
306 result
307 end
308 def converted(move, rotation)
309 inverse_rotation = Array.new(54)
310 rotation.each_with_index { |destination, source| inverse_rotation[destination] = source }
311 target = compose(compose(rotation, @moves[move]), inverse_rotation)
312 @moves.find { |_, permutation| permutation == target }[0]
313 end
314 def unrotate(sequence)
315 replacements = { 'M' => [['R', 1], ['L', 3], ['x', 3]], 'E' => [['U', 1], ['D', 3], ['y', 3]], 'S' => [['F', 3], ['B', 1], ['z', 1]],
316 'r' => [['L', 1], ['x', 1]], 'l' => [['R', 1], ['x', 3]], 'u' => [['D', 1], ['y', 1]], 'd' => [['U', 1], ['y', 3]],
317 'f' => [['B', 1], ['z', 1]], 'b' => [['F', 1], ['z', 3]] }
318 expanded = sequence.flat_map do |letter, amount|
319 replacements.key?(letter) ? replacements[letter].map { |name, count| [name, count * amount % 4] } : [[letter, amount]]
320 end
321 rotation = @identity
322 output = []
323 expanded.each do |move|
324 if 'xyz'.include?(move[0])
325 rotation = compose(rotation, @moves[move])
326 else
327 output << converted(move, rotation)
328 end
329 end
330 output + @rotations[@rotation_effects.index(rotation)]
331 end
332 def search(depth)
333 candidates = FACES.flat_map { |face| (1..3).map { |amount| [face, amount] } }
334 visit = nil
335 visit = lambda do |state, left, previous|
336 return [] if solved?(state)
337 return nil if left == 0
338 candidates.each do |move|
339 face = move[0]
340 next if face == previous || [['D', 'U'], ['L', 'R'], ['B', 'F']].include?([previous, face])
341 tail = visit.call(permute(state, @moves[move]), left - 1, face)
342 return [move] + tail if tail
343 end
344 nil
345 end
346 (0..depth).each do |limit|
347 found = visit.call(@state, limit, nil)
348 return found if found
349 end
350 nil
351 end
352 def command(line)
353 command, argument = line.split(' ', 2)
354 error("unknown command #{command}") unless (NO_ARGS + WORDS.keys + SEQUENCES + HEADS).include?(command)
355 words = argument ? argument.split(' ') : []
356 error('bad arguments') if NO_ARGS.include?(command) && argument
357 error('bad arguments') if WORDS.key?(command) && words.length != WORDS[command]
358 if %w[UNDO REDO SEARCH SCRAMBLE].include?(command)
359 error('bad arguments') unless words.all? { |word| word.match?(/\A(?:0|[1-9][0-9]{0,9})\z/) }
360 end
361 if %w[SAVE RESTORE].include?(command)
362 error('bad arguments') unless words[0].match?(/\A[a-z]{1,8}\z/)
363 end
364 if HEADS.include?(command)
365 error('bad arguments') unless argument
366 head, sequence = argument.split(' ', 2)
367 valid = command == 'DEFINE' ? head.match?(/\A[a-z]{1,8}\z/) : command == 'MIRROR' ? %w[LR UD FB].include?(head) : head.match?(/\A[xyz](?:2|')?\z/)
368 error('bad arguments') unless valid
369 moves = parse(sequence || '', command.length + head.length + 2, line.length)
370 elsif SEQUENCES.include?(command)
371 moves = parse(argument || '', command.length + 1, line.length)
372 end
373 error('bad value') if %w[UNDO REDO].include?(command) && words[0].to_i < 1
374 error('bad value') if command == 'SEARCH' && words[0].to_i > 4
375 error('bad value') if command == 'SCRAMBLE' && (!words[0].to_i.between?(1, 50) || !words[1].to_i.between?(1, 2147483646))
376 error("unknown piece #{words[0]}") if command == 'WHERE' && !(CORNERS + EDGES).include?(words[0])
377 case command
378 when 'RESET'
379 @state = @solved.dup
380 @history = []
381 @redo = []
382 puts 'reset'
383 when 'SHOW'
384 %w[U LFRB D].each do |faces|
385 (1..3).each { |row| puts (faces.length == 1 ? ' ' : '') + faces.chars.flat_map { |face| (1..3).map { |col| @state[index(face, row, col)] } }.join(' ') }
386 end
387 when 'STATE' then puts 'state ' + @state.join
388 when 'ISSOLVED'
389 count = @state.each_index.count { |i| @state[i] != @state[i / 9 * 9 + 4] }
390 puts(count == 0 ? 'solved' : "unsolved #{count}")
391 when 'HISTORY' then puts 'history ' + notation(@history)
392 when 'SOLUTION' then puts 'solution ' + notation(simplify(inverse(@history)))
393 when 'ANALYZE'
394 rotation, state = normalize(@state)
395 puts 'rotation ' + (rotation == 0 ? 'none' : move_words(@rotations[rotation]))
396 puts 'corners ' + cycles(state, 0)
397 puts 'edges ' + cycles(state, 1)
398 when 'STAGE'
399 _, state = normalize(@state)
400 corners = identify(state, 0)
401 edges = identify(state, 1)
402 cross = (4..7).count { |i| edges[i] == [i, 0] }
403 slots = (4..7).count { |i| corners[i] == [i, 0] && edges[i + 4] == [i + 4, 0] }
404 top = state[0, 9].count('W')
405 stage = solved?(state) ? 'solved' : cross == 4 && slots == 4 && top == 9 ? 'oll' : cross == 4 && slots == 4 ? 'f2l' : cross == 4 ? 'cross' : 'none'
406 puts "stage #{stage} cross #{cross} slots #{slots} top #{top}"
407 when 'MACROS'
408 puts "macros #{@macros.length}"
409 @macros.keys.sort.each { |name| puts "macro #{name} #{notation(@macros[name])}" }
410 when 'CHECK', 'LOAD'
411 state = words[0].chars
412 reason = invalid(state)
413 if command == 'CHECK'
414 puts(reason ? 'invalid ' + reason : 'valid')
415 else
416 error('invalid ' + reason) if reason
417 @state = state
418 @history = []
419 @redo = []
420 puts 'loaded'
421 end
422 when 'UNDO', 'REDO'
423 source = command == 'UNDO' ? @history : @redo
424 error(command == 'UNDO' ? 'nothing to undo' : 'nothing to redo') if source.empty?
425 count = [words[0].to_i, source.length].min
426 count.times do
427 move = source.pop
428 @state = permute(@state, @moves[command == 'UNDO' ? [move[0], 4 - move[1]] : move])
429 (command == 'UNDO' ? @redo : @history) << move
430 end
431 puts "#{command == 'UNDO' ? 'undone' : 'redone'} #{count}"
432 when 'SEARCH'
433 found = search(words[0].to_i)
434 puts(found ? 'found ' + notation(found) : 'not found')
435 when 'WHERE'
436 _, state = normalize(@state)
437 kind = CORNERS.include?(words[0]) ? 0 : 1
438 wanted = @names[kind].index(words[0])
439 pieces = identify(state, kind)
440 position = pieces.index { |piece| piece[0] == wanted }
441 puts "#{words[0]} at #{@names[kind][position]} #{kind == 0 ? 'twist' : 'flip'} #{pieces[position][1]}"
442 when 'SAVE'
443 @saves[words[0]] = [@state.dup, @history.map(&:dup)]
444 puts "saved #{words[0]}"
445 when 'RESTORE'
446 error("no save #{words[0]}") unless @saves.key?(words[0])
447 @state, @history = @saves[words[0]].map { |part| part.map { |item| item.is_a?(Array) ? item.dup : item } }
448 @redo = []
449 puts "restored #{words[0]}"
450 when 'SCRAMBLE', 'APPLY'
451 if command == 'SCRAMBLE'
452 moves = []
453 random = words[1].to_i
454 words[0].to_i.times do
455 usable = FACES.reject do |face|
456 (!moves.empty? && face == moves[-1][0]) || (moves.length >= 2 && axis(moves[-1][0]) == axis(moves[-2][0]) && axis(face) == axis(moves[-1][0]))
457 end
458 random = random * 48271 % 2147483647
459 face = usable[random % usable.length]
460 random = random * 48271 % 2147483647
461 moves << [face, [1, 3, 2][random % 3]]
462 end
463 end
464 @state = permute(@state, effect(moves))
465 @history.concat(moves)
466 @redo = []
467 puts(command == 'APPLY' ? "applied #{moves.length}" : 'scramble ' + move_words(moves))
468 when 'INVERT' then puts 'inverse ' + notation(inverse(moves))
469 when 'SIMPLIFY' then puts 'simplified ' + notation(simplify(moves))
470 when 'METRICS'
471 htm = qtm = stm = 0
472 moves.each do |letter, amount|
473 next if 'xyz'.include?(letter)
474 weight = 'MES'.include?(letter) ? 2 : 1
475 htm += weight
476 qtm += weight * (amount == 2 ? 2 : 1)
477 stm += 1
478 end
479 puts "metrics htm #{htm} qtm #{qtm} stm #{stm} etm #{moves.length}"
480 when 'ORDER' then puts "order #{order(effect(moves))}"
481 when 'CYCLES'
482 state = permute(@solved, effect(moves))
483 %w[corners edges centers].each_with_index { |kind, i| puts kind + ' ' + cycles(state, i) }
484 when 'UNROTATE' then puts 'unrotated ' + notation(unrotate(moves))
485 when 'UNTIL'
486 permutation = effect(moves)
487 state = @state
488 found = nil
489 (1..order(permutation)).each do |count|
490 state = permute(state, permutation)
491 if solved?(state)
492 found = count
493 break
494 end
495 end
496 puts "until #{found || 'never'}"
497 when 'DEFINE'
498 @macros[head] = moves
499 puts "defined #{head} #{moves.length}"
500 when 'MIRROR'
501 mirror_axis = { 'LR' => 'x', 'UD' => 'y', 'FB' => 'z' }[head]
502 opposites = 'RLUDFBrludfb'.chars.zip('LRDUBFlrdubf'.chars).to_h
503 transformed = moves.map do |letter, amount|
504 if axis(letter) != mirror_axis
505 [letter, 4 - amount]
506 elsif 'MESxyz'.include?(letter)
507 [letter, amount]
508 else
509 [opposites[letter], 4 - amount]
510 end
511 end
512 puts 'mirror ' + notation(transformed)
513 when 'ROTATE'
514 amount = head.length == 1 ? 1 : head[1] == '2' ? 2 : 3
515 rotation = @moves[[head[0], amount]]
516 puts 'rotated ' + notation(moves.map { |move| converted(move, rotation) })
517 end
518 rescue CubeError => exception
519 puts exception.message
520 end
521 def run
522 count = STDIN.readline.to_i
523 count.times { command(STDIN.readline.chomp("\n").delete_suffix("\r")) }
524 end
525end
