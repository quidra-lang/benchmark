50private fun simplify(input: List<Move>): List<Move> {
51 var sequence = input
52 while (true) {
53 val out = mutableListOf<Move>()
54 var at = 0
55 while (at < sequence.size) {
56 val a = axis(sequence[at].letter)
57 val run = mutableListOf<Move>()
58 while (at < sequence.size && axis(sequence[at].letter) == a) {
59 val m = sequence[at++]
60 val i = run.indexOfFirst {
61 it.letter == m.letter
62 }
63 if (i < 0) run.add(m) else run[i] = run[i].copy(amount = (run[i].amount+m.amount)%4)
64 }
65 out.addAll(run.filter {
66 it.amount != 0
67 })
68 }
69 if (out == sequence) return out
70 sequence = out
71 }

79private fun simple(s: String): List<Move> = buildList {
80 var at = 0

82 if (s[at] == ' ') {
83 at++

88 '2' -> {

101private class SequenceParser(private val line: String, private var at: Int, private val macros: Map<String, List<Move>>) {
102 private fun current() = line.getOrNull(at)
103 private fun skip() {
104 while (current() == ' ') at++
105 }
106 private fun syntax(): Nothing = throw Problem("error: syntax at ${minOf(at+1, line.length+1)}")
107 private fun power(sequence: List<Move>): List<Move> {
108 var count = 1
109 if (current()?.let {
110 it in '1'..'9'
111 } == true) {
112 count = line[at++]-'0'
113 if (current()?.let {
114 it in '0'..'9'
115 } == true) count = count*10+(line[at++]-'0')
116 }
117 val invert = current() == '\''
118 if (invert) at++
119 val repeated = buildList {
120 repeat(count) {
121 addAll(sequence)
122 }
123 }
124 return if (invert) inverse(repeated) else repeated
125 }
126 private fun sequence(): List<Move> {
127 val out = mutableListOf<Move>()
128 while (true) {
129 skip()
130 val c = current() ?: break
131 val next = when {
132 c in LETTERS -> {
133 at++
134 val amount = when (current()) {
135 '2' -> {
136 at++
137 2
138 }
139 '\'' -> {
140 at++
141 3
142 }
143 else -> 1
144 }
145 listOf(Move(c, amount))
146 }
147 c == '(' -> {
148 at++
149 val sequence = sequence()
150 skip()
151 if (current() != ')') syntax()
152 at++
153 power(sequence)
154 }
155 c == '[' -> {
156 at++
157 val a = sequence()
158 skip()
159 val separator = current()
160 if (separator != ',' && separator != ':') syntax()
161 at++
162 val b = sequence()
163 skip()
164 if (current() != ']') syntax()
165 at++
166 power(a+b+inverse(a)+(if (separator == ',') inverse(b) else emptyList()))
167 }
168 c == '{' -> {
169 at++
170 val start = at
171 while (current()?.let {
172 it in 'a'..'z'
173 } == true) at++
174 if (at == start || current() != '}') syntax()
175 val name = line.substring(start, at)
176 val sequence = macros[name] ?: throw Problem("error: undefined $name")
177 at++
178 power(sequence)
179 }
180 else -> break
181 }
182 out.addAll(next)
183 }
184 return out
185 }
186 fun parse(): List<Move> {
187 val sequence = sequence()
188 skip()
189 if (current() != null) syntax()
190 if (sequence.size > 500) throw Problem("error: too long")
191 return sequence
192 }
193}

372 fun cycles(positions: List<Pair<Int, Int>?>, names: List<String>, modulus: Int): String {
373 val destination = IntArray(names.size)
374 val orientation = IntArray(names.size)
375 for ((q, entry) in positions.withIndex()) {
376 val (piece, t) = entry!!
377 destination[piece] = q
378 orientation[piece] = t
379 }
380 val seen = BooleanArray(names.size)
381 val out = mutableListOf<String>()
382 for (start in names.indices) {
383 if (seen[start]) continue
384 val cycle = mutableListOf<String>()
385 var sum = 0
386 var p = start
387 while (!seen[p]) {
388 seen[p] = true
389 cycle.add(names[p])
390 sum += orientation[p]
391 p = destination[p]
392 }
393 val mark = if (modulus == 1 || sum%modulus == 0) "" else if (modulus == 3 && sum%3 == 2) "-" else "+"
394 if (cycle.size >= 2 || mark.isNotEmpty()) out.add("(${cycle.joinToString(" ")})$mark")
395 }
396 return out.ifEmpty {
397 listOf("none")
398 }.joinToString(" ")
399 }

402 var x = a
403 var y = b
404 while (y != 0L) {
405 val t = x%y
406 x = y
407 y = t
408 }
409 return x
410 }
411 val p = effect(sequence)
412 val seen = BooleanArray(54)
413 var result = 1L
414 for (start in 0 until 54) {
415 var i = start
416 var length = 0L
417 while (!seen[i]) {
418 seen[i] = true

421 }

423 }

426 fun search(state: State, left: Int, previous: Char?, path: MutableList<Move>): Boolean {
427 if (left == 0) return solved(state)
428 for (face in FACES) {
429 if (previous == face || (previous == 'D' && face == 'U') || (previous == 'L' && face == 'R') || (previous == 'B' && face == 'F')) continue
430 for (amount in 1..3) {
431 val m = Move(face,amount)
432 path.add(m)
433 if (search(apply(state,permutation(m)),left-1,face,path)) return true
434 path.removeAt(path.lastIndex)
435 }
436 }
437 return false
438 }
439 fun command(line: String): List<String> {
440 val command = line.substringBefore(' ')
441 val arg = if (' ' in line) line.substringAfter(' ') else null
442 val noArgs = setOf("RESET", "SHOW", "STATE", "ISSOLVED", "HISTORY", "SOLUTION", "ANALYZE", "STAGE", "MACROS")
443 val word = setOf("CHECK", "LOAD", "UNDO", "REDO", "SEARCH", "WHERE", "SAVE", "RESTORE", "SCRAMBLE")
444 val sequences = setOf("APPLY", "INVERT", "SIMPLIFY", "METRICS", "ORDER", "CYCLES", "UNROTATE", "UNTIL")
445 val heads = setOf("DEFINE", "MIRROR", "ROTATE")
446 if (command !in noArgs && command !in word && command !in sequences && command !in heads) throw Problem("error: unknown command $command")
447 val words = arg?.split(' ') ?: emptyList()
448 val head = words.firstOrNull() ?: ""
449 fun bad(): Nothing = throw Problem("error: bad arguments")
450 if (command in noArgs && arg != null) bad()
451 if (command in word) {
452 if (words.size != if (command == "SCRAMBLE") 2 else 1) bad()
453 if (command in setOf("UNDO", "REDO", "SEARCH", "SCRAMBLE") && words.any {
454 number(it) == null
455 }) bad()
456 if (command in setOf("SAVE", "RESTORE") && !name(head)) bad()
457 }
458 if (command in heads && (arg == null || when (command) {
459 "DEFINE" -> !name(head)
460 "MIRROR" -> head !in setOf("LR", "UD", "FB")
461 else -> head !in setOf("x", "x'", "x2", "y", "y'", "y2", "z", "z'", "z2")
462 })) bad()
463 val sequence = if (command in sequences || command in heads) SequenceParser(line, minOf(line.length, command.length+(if (command in heads) head.length+2 else 1)), macros).parse() else emptyList()
464 val n = if (command in setOf("UNDO", "REDO", "SEARCH", "SCRAMBLE")) number(head)!! else 0L
465 if ((command in setOf("UNDO", "REDO") && n == 0L) || (command == "SEARCH" && n > 4) || (command == "SCRAMBLE" && (n !in 1L..50L || number(words[1])!! !in 1L..2147483646L))) throw Problem("error: bad value")
466 if (command == "WHERE" && head !in corners && head !in edges) throw Problem("error: unknown piece $head")
467 val text = when (command) {
468 "RESET" -> {
469 state = solvedState()
470 history.clear()
471 redo.clear()
472 "reset"
473 }
474 "SHOW" -> return buildList {

590 }

621 }

628 }

630 val letter = faces[(x%faces.length).toInt()]
631 x = 48271*x%2147483647
632 out.add(Move(letter, listOf(1,3,2)[(x%3).toInt()]))
633 }

635 history.addAll(out)
636 redo.clear()
637 "scramble ${out.joinToString(" ") { it.text() }}"
638 }
639 "ANALYZE", "STAGE", "WHERE" -> {
640 val (rotation, state) = normalized(this.state)!!
641 val cp = pieces(state,cornerStickers)
642 val ep = pieces(state,edgeStickers)
643 if (command == "ANALYZE") return listOf("rotation ${if (rotation.isEmpty()) "none" else rotation.joinToString(" ") { it.text() }}", "corners ${cycles(cp,corners,3)}", "edges ${cycles(ep,edges,2)}")
644 if (command == "WHERE") {
645 val names = if (head in corners) corners else edges
646 val positions = if (head in corners) cp else ep
647 val piece = names.indexOf(head)
648 val q = positions.indexOfFirst {
649 it!!.first == piece
650 }
651 "$head at ${names[q]} ${if (head in corners) "twist" else "flip"} ${positions[q]!!.second}"
652 } else {
653 val cross = (4..7).count {
654 ep[it] == (it to 0)
655 }
656 val slots = (4..7).count {
657 cp[it] == (it to 0) && ep[it+4] == (it+4 to 0)
658 }
659 val top = state.take(9).count {
660 it == 'W'
661 }
662 val stage = when {
663 solved(state) -> "solved"
664 cross == 4 && slots == 4 && top == 9 -> "oll"
665 cross == 4 && slots == 4 -> "f2l"
666 cross == 4 -> "cross"
667 else -> "none"
668 }
669 "stage $stage cross $cross slots $slots top $top"
670 }
671 }
672 else -> {
673 val state = applySequence(solvedState(),sequence)
674 val centers = (0..5).map {
675 COLORS.indexOf(state[it*9+4]) to 0
676 }
677 return listOf("corners ${cycles(pieces(state,cornerStickers),corners,3)}", "edges ${cycles(pieces(state,edgeStickers),edges,2)}", "centers ${cycles(centers,FACES.map(Char::toString),1)}")
678 }
679 }
680 return listOf(text)
681 }
