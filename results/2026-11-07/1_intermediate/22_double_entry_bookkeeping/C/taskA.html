117static Text money(Money cents) {
118 Text result;
119 Money value = absolute(cents);
120 snprintf(result.text, sizeof result.text, "%s%" PRId64 ".%02" PRId64,

129static bool leap(int year) { return year % 4 == 0 && (year % 100 != 0 || year % 400 == 0); }
130static int month_days(int year, int month) {
131 static const int days[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
132 return days[month - 1] + (month == 2 && leap(year));
133}

135static int day_number(int year, int month, int day) {
136 int result = day - 1;
137 for (int y = 2000; y < year; y++) result += 365 + leap(y);
138 for (int m = 1; m < month; m++) result += month_days(year, m);
139 return result;
140}

142static void date_parts(int number, int *year, int *month, int *day) {
143 *year = 2000;
144 while (number >= 365 + leap(*year)) {
145 number -= 365 + leap(*year);
146 (*year)++;
147 }
148 *month = 1;
149 while (number >= month_days(*year, *month)) {
150 number -= month_days(*year, *month);
151 (*month)++;
152 }
153 *day = number + 1;
154}

156static Text date_text(int number) {
157 Text result;
158 int year, month, day;
159 date_parts(number, &year, &month, &day);
160 snprintf(result.text, sizeof result.text, "%04d-%02d-%02d", year, month, day);
161 return result;
162}

164static bool check_date(const char *text, int *number) {
165 int year = atoi(text), month = atoi(text + 5), day = atoi(text + 8);
166 if (year < 2000 || year > 2099 || month < 1 || month > 12 || day < 1 || day > month_days(year, month)) {
167 return fail("invalid date %s", text);
168 }
169 *number = day_number(year, month, day);
170 return true;
171}

173static bool check_range(const char *first, const char *last, int *start, int *end) {
174 if (!check_date(first, start) || !check_date(last, end)) return false;
175 return *start <= *end || fail("invalid range %s %s", first, last);
176}

178static int months_after(int date, int months) {
179 int year, month, day;
180 date_parts(date, &year, &month, &day);
181 int total = month - 1 + months;
182 year += total / 12;
183 month = total % 12 + 1;
184 if (day > month_days(year, month)) day = month_days(year, month);
185 return day_number(year, month, day);
186}

199static Account *find_account(const Book *book, const char *code) {
200 int low = 0, high = book->account_count;
201 while (low < high) {
202 int middle = (low + high) / 2;
203 int comparison = strcmp(book->account_order[middle]->code, code);
204 if (comparison < 0) low = middle + 1;
205 else if (comparison > 0) high = middle;
206 else return book->account_order[middle];
207 }
208 return NULL;
209}

259static void add_account_total(Lines *lines, const Line *line) {

267static void append_line(Lines *lines, Line line) {
268 if (lines->count == lines->capacity) {
269 lines->capacity = lines->capacity ? lines->capacity * 2 : 4;
270 lines->items = resize(lines->items, lines->capacity * sizeof *lines->items);
271 }
272 lines->items[lines->count++] = line;
273 if (line.debit) lines->amount.debit += line.amount;
274 else lines->amount.credit += line.amount;
275 if (line.account) add_account_total(lines, &line);
276}

284static bool validate_lines(const Book *book, Draft *draft) {
285 Lines *lines = draft->lines;
286 if (lines->count < 2) return fail("too few lines %s", draft->key);
287 for (size_t i = 0; i < lines->count; i++) {
288 Line *line = &lines->items[i];
289 Account *account = require_account(book, line->code);
290 if (!account) return false;
291 if (draft->journal && !require_active(account)) return false;
292 if (line->amount == 0) return fail("zero amount %s", line->code);
293 line->account = account;
294 add_account_total(lines, line);
295 }
296 if (lines->amount.debit != lines->amount.credit) {
297 return fail("unbalanced %s debit %s credit %s", draft->key,
298 money(lines->amount.debit).text, money(lines->amount.credit).text);
299 }
300 return true;
301}

399static Money daily_sum(const Book *book, const Account *account, int start, int end) {
400 int days = end - start + 1;
401 Money *changes = allocate((size_t) days, sizeof *changes);
402 Money running = 0, total = 0;
403 int sign = normal_sign(account);
404 for (size_t i = 0; i < book->entry_count; i++) {
405 const Entry *entry = book->entries[i];
406 if (entry->date > end) continue;
407 Totals amount = entry->lines->accounts[account->index];
408 Money change = sign * (amount.debit - amount.credit);
409 if (entry->date < start) running += change;
410 else changes[entry->date - start] += change;
411 }
412 for (int day = 0; day < days; day++) {
413 running += changes[day];
414 total += running;
415 }
416 free(changes);
417 return total;
418}

425static Partners account_partners(const Book *book, const Account *account, int date) {
426 bool seen[1000] = {false};
427 for (size_t i = 0; i < book->entry_count; i++) {
428 const Entry *entry = book->entries[i];
429 if (entry->date > date) continue;
430 for (size_t j = 0; j < entry->lines->count; j++) {
431 const Line *line = &entry->lines->items[j];
432 if (line->account == account && line->partner >= 0) seen[line->partner] = true;
433 }
434 }
435 Partners result = {.count = 0};
436 for (int i = 0; i < book->partner_count; i++) {
437 if (seen[i]) result.items[result.count++] = (Partner) {i, book->partners[i]};
438 }
439 qsort(result.items, (size_t) result.count, sizeof *result.items, compare_partners);
440 return result;
441}

443static void append_item(OpenItems *items, OpenItem item) {
444 if (items->count == items->capacity) {
445 if (items->head) {
446 items->count -= items->head;
447 memmove(items->items, items->items + items->head, items->count * sizeof *items->items);
448 items->head = 0;
449 } else {
450 items->capacity = items->capacity ? items->capacity * 2 : 8;
451 items->items = resize(items->items, items->capacity * sizeof *items->items);
452 }
453 }
454 items->items[items->count++] = item;
455}

457static OpenItems open_items(Book *book, const Account *account, int partner, int date) {
458 OpenItems result = {0};
459 Entry **ordered = posting_order(book);
460 for (size_t i = 0; i < book->entry_count; i++) {
461 Entry *entry = ordered[i];
462 if (entry->date > date) break;
463 Totals total = entry->lines->accounts[account->index];
464 if (total.debit == 0 && total.credit == 0) continue;
465 for (size_t j = 0; j < entry->lines->count; j++) {
466 const Line *line = &entry->lines->items[j];
467 if (line->account != account || line->partner != partner) continue;
468 Money value = normal_sign(account) * (line->debit ? line->amount : -line->amount);
469 if (value > 0) {
470 Money applied = minimum(value, result.unapplied);
471 result.unapplied -= applied;
472 if (value > applied) append_item(&result, (OpenItem) {entry, value, value - applied});
473 } else {
474 Money payment = -value;
475 while (payment > 0 && result.head < result.count) {
476 OpenItem *item = &result.items[result.head];
477 Money applied = minimum(payment, item->remaining);
478 payment -= applied;
479 item->remaining -= applied;
480 if (item->remaining == 0) result.head++;
481 }
482 result.unapplied += payment;
483 }
484 }
485 }
486 return result;
487}

489static Money section(const Book *book, const char *heading, unsigned types, int sign,
490 const Totals totals[200], bool show_earnings, Money earnings) {
491 Money amounts[200] = {0}, total = show_earnings ? earnings : 0;
492 for (int i = 0; i < book->account_count; i++) {
493 const Account *account = &book->accounts[i];
494 if (types & (1u << account->type)) {
495 amounts[i] = sign * (totals[i].debit - totals[i].credit);
496 total += amounts[i];
497 }
498 }
499 printf(" %s %s\n", heading, money(total).text);
500 for (int i = 0; i < book->account_count; i++) {
501 const Account *top = book->account_order[i];
502 if (parent_account(book, top)) continue;
503 Money subtotal = 0;
504 bool visible = false;
505 for (int j = 0; j < book->account_count; j++) {
506 const Account *account = book->account_order[j];
507 if (!prefix(top->code, account->code)) continue;
508 subtotal += amounts[account->index];
509 visible |= amounts[account->index] != 0;
510 }
511 if (!visible) continue;
512 printf(" %s %s %s\n", top->code, top->name, money(subtotal).text);
513 for (int j = 0; j < book->account_count; j++) {
514 const Account *account = book->account_order[j];
515 if (prefix(top->code, account->code) && amounts[account->index] != 0) {
516 printf(" %s %s %s\n", account->code, account->name, money(amounts[account->index]).text);
517 }
518 }
519 }
520 if (show_earnings) printf(" current earnings %s\n", money(earnings).text);
521 return total;
522}

536static void execute(Book *book, char **words, int count) {
537 const char *command = words[0];
538 if (strcmp(command, "ACCOUNT") == 0) {
539 if (find_account(book, words[1])) { fail("duplicate account %s", words[1]); return; }
540 AccountType type = ASSET;
541 while (strcmp(TYPE_NAMES[type], words[2]) != 0) type++;
542 for (int i = 0; i < book->account_count; i++) {
543 const Account *other = book->account_order[i];
544 if ((prefix(other->code, words[1]) || prefix(words[1], other->code)) && type_class(type) != type_class(other->type)) {
545 fail("class mismatch %s %s", words[1], other->code);
546 return;
547 }
548 }
549 Account *account = &book->accounts[book->account_count];
550 strcpy(account->code, words[1]);
551 strcpy(account->name, words[3]);
552 account->type = type;
553 account->index = book->account_count;
554 account->active = true;
555 int position = book->account_count++;
556 while (position > 0 && strcmp(book->account_order[position - 1]->code, account->code) > 0) {
557 book->account_order[position] = book->account_order[position - 1];
558 position--;
559 }
560 book->account_order[position] = account;
561 printf("ACCOUNT %s %s %s parent %s\n", account->code, TYPE_NAMES[type], account->name, parent_code(book, account));
562 } else if (strcmp(command, "DEACTIVATE") == 0) {
563 Account *account = require_account(book, words[1]);
564 if (!account) return;
565 if (!account->active) { fail("already inactive %s", words[1]); return; }
566 Money current = balance(book, account, 36600);
567 if (current != 0) { fail("nonzero balance %s %s", words[1], money(current).text); return; }
568 account->active = false;
569 printf("DEACTIVATE %s %s\n", account->code, account->name);
570 } else if (strcmp(command, "JOURNAL") == 0 || strcmp(command, "TEMPLATE") == 0) {
571 if (book->has_draft) printf("DISCARD %s lines %zu\n", book->draft.key, book->draft.lines->count);
572 book->draft.journal = strcmp(command, "JOURNAL") == 0;
573 strcpy(book->draft.key, words[1]);
574 strcpy(book->draft.memo, words[book->draft.journal ? 3 : 2]);
575 if (book->draft.journal) strcpy(book->draft.date, words[2]);
576 book->draft.lines = new_lines(book);
577 book->has_draft = true;
578 } else if (strcmp(command, "DR") == 0 || strcmp(command, "CR") == 0) {
579 if (!book->has_draft) { fail("no open entry"); return; }
580 Line line = {.amount = parse_money(words[2]), .partner = count == 4 ? partner_index(book, words[3]) : -1,
581 .debit = strcmp(command, "DR") == 0};
582 strcpy(line.code, words[1]);
583 append_line(book->draft.lines, line);
584 } else if (strcmp(command, "POST") == 0) {
585 if (!book->has_draft) { fail("no open entry"); return; }
586 Draft draft = book->draft;
587 book->has_draft = false;
588 if (draft.journal) {
589 if (find_entry(book, draft.key)) { fail("duplicate entry %s", draft.key); return; }
590 int date;
591 if (!check_date(draft.date, &date) || !check_open(book, date) || !validate_lines(book, &draft)) return;
592 Entry *entry = post(book, draft.key, date, JOURNAL, draft.memo, draft.lines);
593 printf("POST %s %s lines %zu amount %s\n", entry->id, date_text(date).text, entry->lines->count, money(entry->lines->amount.debit).text);
594 } else {
595 if (find_template(book, draft.key)) { fail("duplicate template %s", draft.key); return; }
596 if (!validate_lines(book, &draft)) return;
597 Template *pattern = &book->templates[book->template_count++];
598 strcpy(pattern->name, draft.key);
599 strcpy(pattern->memo, draft.memo);
600 pattern->lines = draft.lines;
601 pattern->uses = 0;
602 printf("TEMPLATE %s lines %zu amount %s\n", pattern->name, pattern->lines->count, money(pattern->lines->amount.debit).text);
603 }
604 } else if (strcmp(command, "DRAFT") == 0) {
605 if (!book->has_draft) { fail("no open entry"); return; }
606 Totals totals = book->draft.lines->amount;
607 printf("DRAFT %s %s lines %zu debit %s credit %s difference %s\n", book->draft.journal ? "journal" : "template",
608 book->draft.key, book->draft.lines->count, money(totals.debit).text, money(totals.credit).text, money(totals.debit - totals.credit).text);
609 } else if (strcmp(command, "REVERSE") == 0) {
610 Entry *entry = find_entry(book, words[1]);
611 if (!entry) { fail("unknown entry %s", words[1]); return; }
612 if (entry->category == REVERSAL || entry->category == CLOSING || entry->category == REOPEN) {
613 fail("cannot reverse %s", words[1]); return;
614 }
615 if (entry->reversed_by) { fail("already reversed %s", words[1]); return; }
616 int date;
617 if (!check_date(words[2], &date)) return;
618 if (date < entry->date) { fail("reverse before entry %s %s", words[1], date_text(entry->date).text); return; }
619 if (!check_open(book, date) || !active_lines(entry->lines)) return;
620 Entry *reversal = reverse_entry(book, entry, date, REVERSAL, entry->memo);
621 printf("REVERSE %s entry %s %s amount %s\n", entry->id, reversal->id, date_text(date).text, money(reversal->lines->amount.debit).text);
622 } else if (strcmp(command, "RECUR") == 0) {
623 Template *pattern = find_template(book, words[1]);
624 if (!pattern) { fail("unknown template %s", words[1]); return; }
625 int date, repetitions = atoi(words[3]);
626 if (!check_date(words[2], &date)) return;
627 if (repetitions == 0) { fail("invalid count"); return; }
628 if (!check_open(book, date) || !active_lines(pattern->lines)) return;

632 snprintf(id, sizeof id, "%s-%d", pattern->name, ++pattern->uses);

634 printf(" POST %s %s lines %zu amount %s\n", entry->id, date_text(entry->date).text, entry->lines->count, money(entry->lines->amount.debit).text);
635 }
636 } else if (strcmp(command, "DEPRECIATE") == 0) {
637 int date;
638 if (!check_date(words[1], &date) || !check_open(book, date)) return;
639 Account *asset = require_type(book, words[2], ASSET);
640 if (!asset) return;
641 Account *contra = require_type(book, words[3], CONTRA_ASSET);
642 if (!contra) return;
643 Account *expense = require_type(book, words[4], EXPENSE);
644 if (!expense || !require_active(contra) || !require_active(expense)) return;
645 Money cost = balance(book, asset, date), accumulated = balance(book, contra, date);
646 Money amount = cost > 0 ? minimum(cost / atoi(words[5]), cost - accumulated) : 0;
647 if (amount <= 0) { fail("nothing to depreciate %s", asset->code); return; }
648 Lines *lines = new_lines(book);
649 generated_line(lines, true, expense, amount, -1);
650 generated_line(lines, false, contra, amount, -1);
651 char id[21];
652 snprintf(id, sizeof id, "DEP-%d", ++book->depreciation_counter);
653 post(book, id, date, DEPRECIATION, "depreciation", lines);
654 printf("DEPRECIATE %s entry %s %s amount %s accumulated %s book %s\n", asset->code, id, date_text(date).text,
655 money(amount).text, money(accumulated + amount).text, money(cost - accumulated - amount).text);
656 } else if (strcmp(command, "INTEREST") == 0) {
657 Account *account = require_account(book, words[1]);
658 if (!account) return;
659 if (account->type != ASSET && account->type != LIABILITY) { fail("not interest bearing %s", words[1]); return; }
660 Account *offset = require_type(book, words[2], account->type == ASSET ? REVENUE : EXPENSE);
661 if (!offset) return;
662 int start, end;
663 if (!check_range(words[3], words[4], &start, &end) || !check_open(book, end) ||
664 !require_active(account) || !require_active(offset)) return;
665 Money sum = daily_sum(book, account, start, end);
666 Money amount = sum * atoi(words[5]) / 3650000;
667 if (amount <= 0) { fail("no interest %s", account->code); return; }
668 Lines *lines = new_lines(book);
669 generated_line(lines, true, account->type == ASSET ? account : offset, amount, -1);
670 generated_line(lines, false, account->type == ASSET ? offset : account, amount, -1);
671 char id[21];
672 snprintf(id, sizeof id, "INT-%d", ++book->interest_counter);
673 post(book, id, end, INTEREST, "interest", lines);
674 printf("INTEREST %s entry %s %s days %d average %s amount %s\n", account->code, id, date_text(end).text,
675 end - start + 1, money(sum / (end - start + 1)).text, money(amount).text);
676 } else if (strcmp(command, "WRITEOFF") == 0) {
677 int date;
678 if (!check_date(words[1], &date) || !check_open(book, date)) return;
679 Account *account = require_type(book, words[2], ASSET);
680 if (!account) return;
681 Account *expense = require_type(book, words[3], EXPENSE);
682 if (!expense || !require_active(account) || !require_active(expense)) return;
683 int age = atoi(words[4]), targets = 0;
684 Partners partners = account_partners(book, account, date);
685 Money amounts[1000] = {0}, total = 0;
686 for (int i = 0; i < partners.count; i++) {
687 OpenItems items = open_items(book, account, partners.items[i].index, date);
688 for (size_t j = items.head; j < items.count; j++) {
689 if (date - items.items[j].entry->date > age) amounts[i] += items.items[j].remaining;
690 }
691 free(items.items);
692 if (amounts[i] > 0) { targets++; total += amounts[i]; }
693 }
694 if (total == 0) { fail("nothing to write off %s", account->code); return; }
695 Lines *lines = new_lines(book);
696 generated_line(lines, true, expense, total, -1);
697 for (int i = 0; i < partners.count; i++) {
698 if (amounts[i] > 0) generated_line(lines, false, account, amounts[i], partners.items[i].index);
699 }
700 char id[21];
701 snprintf(id, sizeof id, "WO-%d", ++book->writeoff_counter);
702 post(book, id, date, WRITEOFF, "writeoff", lines);
703 printf("WRITEOFF %s entry %s %s partners %d amount %s\n", account->code, id, date_text(date).text, targets, money(total).text);
704 for (int i = 0; i < partners.count; i++) if (amounts[i] > 0) printf(" %s %s\n", partners.items[i].name, money(amounts[i]).text);
705 } else if (strcmp(command, "CLOSE") == 0) {
706 int date;
707 if (!check_date(words[1], &date) || !check_open(book, date)) return;
708 Account *equity = require_type(book, words[2], EQUITY);
709 if (!equity || !require_active(equity)) return;
710 Totals totals[200];
711 all_totals(book, 0, date, false, totals);
712 for (int i = 0; i < book->account_count; i++) {
713 Account *account = book->account_order[i];
714 Totals total = totals[account->index];
715 if ((account->type == REVENUE || account->type == EXPENSE) && total.debit != total.credit && !require_active(account)) return;
716 }
717 Lines *lines = new_lines(book);
718 Money income = 0;
719 for (int i = 0; i < book->account_count; i++) {
720 Account *account = book->account_order[i];
721 if (account->type != REVENUE && account->type != EXPENSE) continue;
722 Money difference = totals[account->index].debit - totals[account->index].credit;
723 if (difference != 0) generated_line(lines, difference < 0, account, absolute(difference), -1);
724 income -= difference;
725 }
726 if (income != 0) generated_line(lines, income < 0, equity, absolute(income), -1);
727 char id[21];
728 snprintf(id, sizeof id, "CLOSE-%d", ++book->close_counter);
729 Entry *entry = lines->count ? post(book, id, date, CLOSING, "closing", lines) : NULL;
730 book->closes[book->close_count++] = (CloseRecord) {date, entry};
731 if (entry) printf("CLOSE %s entry %s lines %zu net income %s\n", date_text(date).text, id, lines->count, money(income).text);
732 else printf("CLOSE %s no entry\n", date_text(date).text);
733 } else if (strcmp(command, "REOPEN") == 0) {
734 if (book->close_count == 0) { fail("nothing to reopen"); return; }
735 CloseRecord record = book->closes[book->close_count - 1];
736 if (record.entry && !active_lines(record.entry->lines)) return;
737 Entry *entry = record.entry ? reverse_entry(book, record.entry, record.date, REOPEN, "reopen") : NULL;
738 book->close_count--;
739 printf("REOPEN %s ", date_text(record.date).text);
740 if (entry) printf("entry %s", entry->id);
741 else fputs("no entry", stdout);
742 printf(" closed %s\n", closed_text(book).text);
743 } else if (strcmp(command, "LEDGER") == 0) {
744 Account *account = require_account(book, words[1]);
745 if (!account) return;
746 int start, end;
747 if (!check_range(words[2], words[3], &start, &end)) return;
748 Money running = balance(book, account, start - 1);
749 Totals total = {0};
750 printf("LEDGER %s %s %s %s\n opening %s\n", account->code, account->name, words[2], words[3], money(running).text);
751 Entry **ordered = posting_order(book);
752 for (size_t i = 0; i < book->entry_count; i++) {
753 const Entry *entry = ordered[i];
754 if (entry->date < start || entry->date > end) continue;
755 for (size_t j = 0; j < entry->lines->count; j++) {
756 const Line *line = &entry->lines->items[j];
757 if (line->account != account) continue;
758 if (line->debit) total.debit += line->amount;
759 else total.credit += line->amount;
760 running += normal_sign(account) * (line->debit ? line->amount : -line->amount);
761 printf(" %s %s %s %s balance %s %s\n", date_text(entry->date).text, entry->id, line->debit ? "DR" : "CR",
762 money(line->amount).text, money(running).text, entry->memo);
763 }
764 }
765 int days = end - start + 1;
766 printf(" ending %s debits %s credits %s days %d average %s\n", money(running).text, money(total.debit).text,
767 money(total.credit).text, days, money(daily_sum(book, account, start, end) / days).text);
768 } else if (strcmp(command, "ACTIVITY") == 0) {
769 Account *account = require_account(book, words[1]);
770 if (!account) return;
771 int year = atoi(words[2]);
772 Totals total = {0};
773 printf("ACTIVITY %s %s %s\n", account->code, account->name, words[2]);
774 for (int month = 1; month <= 12; month++) {
775 int start = day_number(year, month, 1), end = start + month_days(year, month) - 1;
776 Totals current = account_totals(book, account, start, end);
777 total.debit += current.debit;
778 total.credit += current.credit;
779 printf(" %04d-%02d debit %s credit %s balance %s\n", year, month, money(current.debit).text,
780 money(current.credit).text, money(balance(book, account, end)).text);

782 printf(" total debit %s credit %s\n", money(total.debit).text, money(total.credit).text);
783 } else if (strcmp(command, "TRIAL") == 0) {
784 int date;
785 if (!check_date(words[1], &date)) return;
786 Totals totals[200], total = {0};
787 all_totals(book, 0, date, false, totals);
788 printf("TRIAL %s\n", words[1]);
789 for (int i = 0; i < book->account_count; i++) {
790 const Account *account = book->account_order[i];
791 Money difference = totals[account->index].debit - totals[account->index].credit;
792 if (difference == 0) continue;
793 printf(" %s %s %s %s\n", account->code, account->name, difference > 0 ? "debit" : "credit", money(absolute(difference)).text);
794 if (difference > 0) total.debit += difference;
795 else total.credit -= difference;
796 }
797 printf(" total debit %s credit %s\n", money(total.debit).text, money(total.credit).text);
798 } else if (strcmp(command, "INCOME") == 0) {
799 int start, end;
800 if (!check_range(words[1], words[2], &start, &end)) return;
801 Totals totals[200];
802 all_totals(book, start, end, true, totals);
803 printf("INCOME %s %s\n", words[1], words[2]);
804 Money revenue = section(book, "REVENUE", 1u << REVENUE, -1, totals, false, 0);
805 Money expense = section(book, "EXPENSE", 1u << EXPENSE, 1, totals, false, 0);
806 printf(" NET INCOME %s\n", money(revenue - expense).text);
807 } else if (strcmp(command, "BALANCE") == 0) {
808 int date;

810 Totals totals[200];
811 all_totals(book, 0, date, false, totals);
812 Money earnings = 0;
813 for (int i = 0; i < book->account_count; i++) {
814 if (book->accounts[i].type == REVENUE || book->accounts[i].type == EXPENSE) earnings -= totals[i].debit - totals[i].credit;
815 }
816 printf("BALANCE %s\n", words[1]);
817 Money assets = section(book, "ASSETS", (1u << ASSET) | (1u << CONTRA_ASSET), 1, totals, false, 0);
818 Money liabilities = section(book, "LIABILITIES", 1u << LIABILITY, -1, totals, false, 0);
819 Money equity = section(book, "EQUITY", 1u << EQUITY, -1, totals, true, earnings);
820 if (assets == liabilities + equity) puts(" check ok");
821 else printf(" check mismatch %s\n", money(assets - liabilities - equity).text);
822 } else if (strcmp(command, "ROLLUP") == 0) {
823 int date;
824 if (!check_date(words[2], &date)) return;
825 Totals totals[200], postings = {0};
826 all_totals(book, 0, date, false, totals);
827 printf("ROLLUP %s %s\n", words[1], words[2]);
828 bool any = false;
829 for (int i = 0; i < book->account_count; i++) {
830 const Account *account = book->account_order[i];
831 if (!prefix(words[1], account->code)) continue;

833 int depth = 0;

836 const Account *other = book->account_order[j];
837 if (other != account && prefix(words[1], other->code) && prefix(other->code, account->code)) depth++;
838 if (prefix(account->code, other->code)) subtotal += totals[other->index].debit - totals[other->index].credit;
839 }
840 Totals own = totals[account->index];
841 printf("%*s%s %s own %s total %s\n", 2 * (1 + depth), "", account->code, account->name,
842 money(normal_sign(account) * (own.debit - own.credit)).text, money(normal_sign(account) * subtotal).text);
843 postings.debit += own.debit;
844 postings.credit += own.credit;
845 }
846 if (any) printf(" postings debit %s credit %s\n", money(postings.debit).text, money(postings.credit).text);
847 else puts(" (none)");
848 } else if (strcmp(command, "AGING") == 0 || strcmp(command, "ITEMS") == 0) {
849 Account *account = require_account(book, words[1]);
850 if (!account) return;
851 bool aging = strcmp(command, "AGING") == 0;
852 int date;
853 if (!check_date(words[aging ? 2 : 3], &date)) return;
854 if (aging) {
855 printf("AGING %s %s %s\n", account->code, account->name, words[2]);
856 Partners partners = account_partners(book, account, date);
857 Money totals[5] = {0};
858 for (int i = 0; i < partners.count; i++) {

862 int age = date - items.items[j].entry->date;
863 int bucket = age <= 30 ? 0 : age <= 60 ? 1 : age <= 90 ? 2 : 3;
864 values[bucket] += items.items[j].remaining;
865 }
866 if (items.head < items.count || items.unapplied > 0) {
867 aging_row(partners.items[i].name, values);
868 for (int j = 0; j < 5; j++) totals[j] += values[j];
869 }
870 free(items.items);
871 }
872 aging_row("total", totals);
873 } else {
874 printf("ITEMS %s %s %s %s\n", account->code, account->name, words[2], words[3]);

876 for (int i = 0; i < book->partner_count; i++) if (strcmp(book->partners[i], words[2]) == 0) partner = i;
877 OpenItems items = open_items(book, account, partner, date);
878 Money remaining = 0;
879 for (size_t i = items.head; i < items.count; i++) {
880 OpenItem item = items.items[i];
881 remaining += item.remaining;
882 printf(" %s %s amount %s open %s age %d\n", date_text(item.entry->date).text, item.entry->id,
883 money(item.original).text, money(item.remaining).text, date - item.entry->date);
884 }
885 printf(" unapplied %s balance %s\n", money(items.unapplied).text, money(remaining - items.unapplied).text);
886 free(items.items);
887 }
888 } else if (strcmp(command, "ENTRY") == 0) {
889 Entry *entry = find_entry(book, words[1]);
890 if (!entry) { fail("unknown entry %s", words[1]); return; }
891 printf("ENTRY %s %s %s %s\n", entry->id, date_text(entry->date).text, CATEGORY_NAMES[entry->category], entry->memo);
892 for (size_t i = 0; i < entry->lines->count; i++) {
893 const Line *line = &entry->lines->items[i];
894 printf(" %s %s %s %s", line->debit ? "DR" : "CR", line->account->code, line->account->name, money(line->amount).text);
895 if (line->partner >= 0) printf(" %s", book->partners[line->partner]);
896 putchar('\n');
897 }
898 printf(" total %s\n", money(entry->lines->amount.debit).text);

900 if (entry->reversed_by) printf(" reversed by %s\n", entry->reversed_by->id);

909 const Entry *entry = ordered[i];
910 if (entry->date < start || entry->date > end) continue;
911 entries++;

926 if (book->account_count == 0) puts(" (none)");

941 inactive, book->entry_count, book->template_count, closed_text(book).text, book->has_draft ? book->draft.key : "none");

945int main(void) {

947 char line[256];

954 execute(book, words, count);
955 }
956 for (size_t i = 0; i < book->entry_count; i++) free(book->entries[i]);
957 free(book->entries);
958 free(book->ordered);
959 while (book->line_storage) {

961 book->line_storage = lines->next;
962 free(lines->items);
963 free(lines);
964 }
965 free(book);
