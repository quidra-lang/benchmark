1local LAST_DAY = 36524
2
3local function fail(message) error({message = message}, 0) end
4local function money(cents)
5 local absolute = math.abs(cents)
6 return (cents < 0 and "-" or "") .. string.format("%d.%02d", absolute // 100, absolute % 100)
7end
8local function parse_money(text)
9 local whole, fraction = text:match("^(%d+)%.(%d%d)$")
10 return tonumber(whole) * 100 + tonumber(fraction)
11end
12local function trunc_div(value, divisor)
13 return value < 0 and -((-value) // divisor) or value // divisor
14end
15local function starts_with(text, prefix) return text:sub(1, #prefix) == prefix end
16local function ascii_less(a, b)
17 for i = 1, math.min(#a, #b) do
18 local first, second = a:byte(i), b:byte(i)
19 if first ~= second then return first < second end
20 end
21 return #a < #b
22end
23local function sorted_keys(values)
24 local result = {}
25 for key in pairs(values) do result[#result + 1] = key end
26 table.sort(result, ascii_less)
27 return result
28end
29local function normal(account) return (account.kind == "ASSET" or account.kind == "EXPENSE") and 1 or -1 end
30local function type_class(kind) return kind == "CONTRA_ASSET" and "ASSET" or kind end
31local function difference(line) return line.debit and line.amount or -line.amount end
32local function new_line(debit, code, amount, partner)
33 return {debit = debit, code = code, amount = amount, partner = partner}
34end
35local function line_totals(lines)
36 local debit, credit = 0, 0
37 for _, line in ipairs(lines) do
38 if line.debit then debit = debit + line.amount else credit = credit + line.amount end
39 end
40 return debit, credit
41end

43local month_lengths = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31}
44local function month_days(year, month)
45 local leap = year % 4 == 0 and (year % 100 ~= 0 or year % 400 == 0)
46 return month_lengths[month] + (month == 2 and leap and 1 or 0)
47end
48local function new_date(year, month, day)
49 local prior = year - 1
50 local number = 365 * (year - 2000) + prior // 4 - 1999 // 4 - prior // 100 + 1999 // 100 + prior // 400 - 1999 // 400
51 for m = 1, month - 1 do number = number + month_days(year, m) end
52 return {year = year, month = month, day = day, number = number + day - 1, text = string.format("%04d-%02d-%02d", year, month, day)}
53end
54local function parse_date(text)
55 local year, month, day = tonumber(text:sub(1, 4)), tonumber(text:sub(6, 7)), tonumber(text:sub(9, 10))
56 if year < 2000 or year > 2099 or month < 1 or month > 12 or day < 1 or day > month_days(year, month) then
57 fail("invalid date " .. text)
58 end
59 return new_date(year, month, day)
60end
61local function date_range(first, last)
62 local a = parse_date(first)
63 local b = parse_date(last)
64 if a.number > b.number then fail("invalid range " .. first .. " " .. last) end
65 return a, b
66end
67local function add_months(date, count)
68 local offset = date.month - 1 + count
69 local year, month = date.year + offset // 12, offset % 12 + 1
70 return new_date(year, month, math.min(date.day, month_days(year, month)))
71end

94function Books:typed_account(code, kind)
95 local account = self:account(code)
96 if account.kind ~= kind then fail("wrong type " .. code .. " " .. kind) end
97 return account

99function Books:active(accounts)
100 for _, account in ipairs(accounts) do if not account.active then fail("inactive account " .. account.code) end end
101end
102function Books:active_lines(lines)
103 for _, line in ipairs(lines) do self:active({self.accounts[line.code]}) end
104end
105function Books:closed() return #self.closes == 0 and "none" or self.closes[#self.closes].date.text end
106function Books:check_closed(date)
107 if #self.closes > 0 and date.number <= self.closes[#self.closes].date.number then fail("period closed " .. date.text) end
108end
109function Books:ordered_entries(first, last)
110 first, last = first or 0, last or LAST_DAY
111 local selected = {}
112 for _, entry in ipairs(self.entries) do
113 if entry.date.number >= first and entry.date.number <= last then selected[#selected + 1] = entry end
114 end
115 table.sort(selected, function(a, b)
116 if a.date.number == b.date.number then return a.sequence < b.sequence end
117 return a.date.number < b.date.number
118 end)
119 return selected
120end
121function Books:aggregate(first, last, operating)
122 first, last = first or 0, last or LAST_DAY
123 local result = {}
124 for _, entry in ipairs(self.entries) do
125 if entry.date.number >= first and entry.date.number <= last and
126 not (operating and (entry.kind == "closing" or entry.kind == "reopen")) then
127 for _, line in ipairs(entry.lines) do
128 local total = result[line.code] or {debit = 0, credit = 0, difference = 0}
129 result[line.code] = total
130 if line.debit then total.debit = total.debit + line.amount else total.credit = total.credit + line.amount end
131 total.difference = total.difference + difference(line)
132 end
133 end
134 end
135 return result
136end
137local function totals_for(totals, code) return totals[code] or {debit = 0, credit = 0, difference = 0} end
138function Books:balance(account, last)
139 last = last or LAST_DAY
140 local value = 0
141 for _, entry in ipairs(self.entries) do
142 if entry.date.number <= last then
143 for _, line in ipairs(entry.lines) do if line.code == account.code then value = value + difference(line) end end
144 end
145 end
146 return value * normal(account)
147end
148function Books:daily_sum(account, first, last)
149 local total = 0
150 for _, entry in ipairs(self.entries) do
151 if entry.date.number <= last then
152 local change = 0
153 for _, line in ipairs(entry.lines) do if line.code == account.code then change = change + difference(line) end end
154 total = total + change * normal(account) * (last - math.max(first, entry.date.number) + 1)
155 end
156 end
157 return total
158end
159function Books:parent(account)
160 local parent
161 for _, other in ipairs(self.account_order) do
162 if other.code ~= account.code and starts_with(account.code, other.code) and
163 (not parent or #other.code > #parent.code) then parent = other end
164 end
165 return parent
166end
167function Books:top(account)
168 for _, other in ipairs(self.account_order) do if starts_with(account.code, other.code) then return other end end
169end
170function Books:post(id, date, kind, memo, lines)
171 local saved = {}
172 for i, line in ipairs(lines) do saved[i] = line end
173 local amount = line_totals(saved)
174 local entry = {id = id, date = date, kind = kind, memo = memo, lines = saved, sequence = #self.entries + 1, amount = amount}
175 self.entries[#self.entries + 1] = entry
176 self.entry_ids[id] = entry
177 return entry
178end
179function Books:reverse(original, date, kind, memo)
180 local lines = {}
181 for i, line in ipairs(original.lines) do lines[i] = new_line(not line.debit, line.code, line.amount, line.partner) end
182 local entry = self:post(original.id .. "-R", date, kind, memo, lines)
183 original.reversed_by, entry.reverses = entry.id, original.id
184 return entry
185end
186function Books:settlements(account, day)
187 local result = {}
188 for _, entry in ipairs(self:ordered_entries(0, day)) do
189 for _, line in ipairs(entry.lines) do
190 if line.code == account.code and line.partner ~= nil then
191 local settlement = result[line.partner] or {items = {}, head = 1, unapplied = 0}
192 result[line.partner] = settlement
193 local signed = difference(line) * normal(account)
194 if signed > 0 then
195 local applied = math.min(signed, settlement.unapplied)
196 settlement.unapplied = settlement.unapplied - applied
197 if signed > applied then
198 settlement.items[#settlement.items + 1] = {entry = entry, original = signed, remaining = signed - applied}
199 end
200 else
201 local remaining = -signed
202 while remaining > 0 and settlement.head <= #settlement.items do
203 local item = settlement.items[settlement.head]
204 local applied = math.min(remaining, item.remaining)
205 item.remaining, remaining = item.remaining - applied, remaining - applied
206 if item.remaining == 0 then settlement.head = settlement.head + 1 end
207 end
208 settlement.unapplied = settlement.unapplied + remaining
209 end
210 end
211 end
212 end
213 return result
214end
215function Books:section(heading, kinds, sign, totals, earnings)
216 local groups, total = {}, earnings or 0
217 for _, account in ipairs(self.account_order) do
218 if kinds[account.kind] then
219 local root = self:top(account)
220 local group = groups[root.code] or {root = root, members = {}, total = 0, visible = false}
221 groups[root.code] = group
222 local amount = totals_for(totals, account.code).difference * sign
223 group.members[#group.members + 1] = {account = account, amount = amount}
224 group.total, group.visible = group.total + amount, group.visible or amount ~= 0
225 total = total + amount
226 end
227 end
228 self:emit(" %s %s", heading, money(total))
229 for _, code in ipairs(sorted_keys(groups)) do
230 local group = groups[code]
231 if group.visible then
232 self:emit(" %s %s %s", code, group.root.name, money(group.total))
233 for _, member in ipairs(group.members) do
234 if member.amount ~= 0 then self:emit(" %s %s %s", member.account.code, member.account.name, money(member.amount)) end
235 end
236 end
237 end
238 if earnings ~= nil then self:emit(" current earnings %s", money(earnings)) end
239 return total
240end
241function Books:post_draft()
242 local block = self.draft
243 if not block then fail("no open entry") end
244 self.draft = nil
245 local date
246 if block.kind == "journal" then
247 if self.entry_ids[block.key] then fail("duplicate entry " .. block.key) end
248 date = parse_date(block.date)
249 self:check_closed(date)
250 elseif self.templates[block.key] then fail("duplicate template " .. block.key) end
251 if #block.lines < 2 then fail("too few lines " .. block.key) end
252 for _, line in ipairs(block.lines) do
253 local account = self:account(line.code)
254 if block.kind == "journal" and not account.active then fail("inactive account " .. line.code) end
255 if line.amount == 0 then fail("zero amount " .. line.code) end
256 end
257 local debit, credit = line_totals(block.lines)
258 if debit ~= credit then fail(string.format("unbalanced %s debit %s credit %s", block.key, money(debit), money(credit))) end
259 if block.kind == "journal" then
260 local entry = self:post(block.key, date, "journal", block.memo, block.lines)
261 self:emit("POST %s %s lines %d amount %s", entry.id, date.text, #entry.lines, money(entry.amount))
262 else
263 local template = {name = block.key, memo = block.memo, lines = block.lines, uses = 0}
264 self.templates[block.key] = template
265 self.template_order[#self.template_order + 1] = template
266 table.sort(self.template_order, function(a, b) return ascii_less(a.name, b.name) end)
267 self:emit("TEMPLATE %s lines %d amount %s", block.key, #block.lines, money(debit))
268 end
269end
270function Books:close(date, code)
271 self:check_closed(date)
272 local equity = self:typed_account(code, "EQUITY")
273 self:active({equity})
274 local totals, targets = self:aggregate(0, date.number), {}
275 for _, account in ipairs(self.account_order) do
276 if (account.kind == "REVENUE" or account.kind == "EXPENSE") and totals_for(totals, account.code).difference ~= 0 then
277 targets[#targets + 1] = account
278 end
279 end
280 self:active(targets)
281 local lines, income = {}, 0
282 for _, account in ipairs(targets) do
283 local value = totals[account.code].difference
284 lines[#lines + 1] = new_line(value < 0, account.code, math.abs(value))
285 income = income - value
286 end
287 if income ~= 0 then lines[#lines + 1] = new_line(income < 0, code, math.abs(income)) end
288 self.close_count = self.close_count + 1
289 local entry
290 if #lines > 0 then entry = self:post("CLOSE-" .. self.close_count, date, "closing", "closing", lines) end
291 self.closes[#self.closes + 1] = {date = date, entry = entry}
292 if entry then self:emit("CLOSE %s entry %s lines %d net income %s", date.text, entry.id, #lines, money(income))
293 else self:emit("CLOSE %s no entry", date.text) end
294end
295function Books:ledger(account, first, last)
296 local running, debit, credit = self:balance(account, first.number - 1), 0, 0
297 self:emit("LEDGER %s %s %s %s", account.code, account.name, first.text, last.text)
298 self:emit(" opening %s", money(running))
299 for _, entry in ipairs(self:ordered_entries(first.number, last.number)) do
300 for _, line in ipairs(entry.lines) do
301 if line.code == account.code then
302 running = running + difference(line) * normal(account)
303 if line.debit then debit = debit + line.amount else credit = credit + line.amount end
304 self:emit(" %s %s %s %s balance %s %s", entry.date.text, entry.id, line.debit and "DR" or "CR", money(line.amount), money(running), entry.memo)
305 end
306 end
307 end
308 local days = last.number - first.number + 1
309 self:emit(" ending %s debits %s credits %s days %d average %s", money(running), money(debit), money(credit), days, money(trunc_div(self:daily_sum(account, first.number, last.number), days)))
310end
311function Books:aging_line(name, values)
312 local balance = values[1] + values[2] + values[3] + values[4] - values[5]
313 self:emit(" %s 0-30 %s 31-60 %s 61-90 %s 91+ %s unapplied %s balance %s", name,
314 money(values[1]), money(values[2]), money(values[3]), money(values[4]), money(values[5]), money(balance))
315end

317function Books:execute(words)
318 local command = words[1]
319 if command == "ACCOUNT" then
320 local code, kind, name = words[2], words[3], words[4]
321 if self.accounts[code] then fail("duplicate account " .. code) end
322 for _, other in ipairs(self.account_order) do
323 if (starts_with(code, other.code) or starts_with(other.code, code)) and type_class(kind) ~= type_class(other.kind) then
324 fail("class mismatch " .. code .. " " .. other.code)
325 end
326 end
327 local account = {code = code, kind = kind, name = name, active = true}
328 self.accounts[code], self.account_order[#self.account_order + 1] = account, account
329 table.sort(self.account_order, function(a, b) return ascii_less(a.code, b.code) end)
330 local parent = self:parent(account)
331 self:emit("ACCOUNT %s %s %s parent %s", code, kind, name, parent and parent.code or "none")
332 elseif command == "DEACTIVATE" then
333 local account = self:account(words[2])
334 if not account.active then fail("already inactive " .. account.code) end
335 local balance = self:balance(account)
336 if balance ~= 0 then fail("nonzero balance " .. account.code .. " " .. money(balance)) end
337 account.active = false
338 self:emit("DEACTIVATE %s %s", account.code, account.name)
339 elseif command == "JOURNAL" or command == "TEMPLATE" then
340 if self.draft then self:emit("DISCARD %s lines %d", self.draft.key, #self.draft.lines) end
341 self.draft = {kind = command == "JOURNAL" and "journal" or "template", key = words[2],
342 date = command == "JOURNAL" and words[3] or nil, memo = command == "JOURNAL" and words[4] or words[3], lines = {}}
343 elseif command == "DR" or command == "CR" then
344 if not self.draft then fail("no open entry") end
345 self.draft.lines[#self.draft.lines + 1] = new_line(command == "DR", words[2], parse_money(words[3]), words[4])
346 elseif command == "POST" then
347 self:post_draft()
348 elseif command == "DRAFT" then
349 local block = self.draft
350 if not block then fail("no open entry") end
351 local debit, credit = line_totals(block.lines)
352 self:emit("DRAFT %s %s lines %d debit %s credit %s difference %s", block.kind, block.key, #block.lines, money(debit), money(credit), money(debit - credit))
353 elseif command == "REVERSE" then
354 local original = self:entry(words[2])
355 if original.kind == "reversal" or original.kind == "closing" or original.kind == "reopen" then fail("cannot reverse " .. original.id) end
356 if original.reversed_by then fail("already reversed " .. original.id) end
357 local date = parse_date(words[3])
358 if date.number < original.date.number then fail("reverse before entry " .. original.id .. " " .. original.date.text) end
359 self:check_closed(date)
360 self:active_lines(original.lines)
361 local entry = self:reverse(original, date, "reversal", original.memo)
362 self:emit("REVERSE %s entry %s %s amount %s", original.id, entry.id, date.text, money(entry.amount))
363 elseif command == "RECUR" then
364 local template = self.templates[words[2]]
365 if not template then fail("unknown template " .. words[2]) end
366 local date = parse_date(words[3])
367 local count = tonumber(words[4])
368 if count == 0 then fail("invalid count") end
369 self:check_closed(date)
370 self:active_lines(template.lines)
371 self:emit("RECUR %s count %d", template.name, count)
372 for month = 0, count - 1 do
373 template.uses = template.uses + 1
374 local entry = self:post(template.name .. "-" .. template.uses, add_months(date, month), "recurring", template.memo, template.lines)
375 self:emit(" POST %s %s lines %d amount %s", entry.id, entry.date.text, #entry.lines, money(entry.amount))
376 end
377 elseif command == "DEPRECIATE" then
378 local date = parse_date(words[2])
379 self:check_closed(date)
380 local asset = self:typed_account(words[3], "ASSET")
381 local contra = self:typed_account(words[4], "CONTRA_ASSET")
382 local expense = self:typed_account(words[5], "EXPENSE")
383 self:active({contra, expense})
384 local cost, accumulated = self:balance(asset, date.number), self:balance(contra, date.number)
385 local amount = cost > 0 and math.min(cost // tonumber(words[6]), cost - accumulated) or 0
386 if amount <= 0 then fail("nothing to depreciate " .. asset.code) end
387 self.depreciation_count = self.depreciation_count + 1
388 local entry = self:post("DEP-" .. self.depreciation_count, date, "depreciation", "depreciation", {
389 new_line(true, expense.code, amount), new_line(false, contra.code, amount),
390 })
391 self:emit("DEPRECIATE %s entry %s %s amount %s accumulated %s book %s", asset.code, entry.id, date.text, money(amount), money(accumulated + amount), money(cost - accumulated - amount))
392 elseif command == "INTEREST" then
393 local account = self:account(words[2])
394 if account.kind ~= "ASSET" and account.kind ~= "LIABILITY" then fail("not interest bearing " .. account.code) end
395 local offset = self:typed_account(words[3], account.kind == "ASSET" and "REVENUE" or "EXPENSE")
396 local first, last = date_range(words[4], words[5])
397 self:check_closed(last)
398 self:active({account, offset})
399 local sum = self:daily_sum(account, first.number, last.number)
400 local amount = trunc_div(sum * tonumber(words[6]), 3650000)
401 if amount <= 0 then fail("no interest " .. account.code) end
402 local debit = account.kind == "ASSET" and account.code or offset.code
403 local credit = account.kind == "ASSET" and offset.code or account.code
404 self.interest_count = self.interest_count + 1
405 local entry = self:post("INT-" .. self.interest_count, last, "interest", "interest", {new_line(true, debit, amount), new_line(false, credit, amount)})
406 local days = last.number - first.number + 1
407 self:emit("INTEREST %s entry %s %s days %d average %s amount %s", account.code, entry.id, last.text, days, money(trunc_div(sum, days)), money(amount))
408 elseif command == "WRITEOFF" then
409 local date = parse_date(words[2])
410 self:check_closed(date)
411 local account = self:typed_account(words[3], "ASSET")
412 local expense = self:typed_account(words[4], "EXPENSE")
413 self:active({account, expense})
414 local settlements = self:settlements(account, date.number)
415 local targets, total = {}, 0
416 for _, partner in ipairs(sorted_keys(settlements)) do
417 local settlement, amount = settlements[partner], 0
418 for i = settlement.head, #settlement.items do
419 local item = settlement.items[i]
420 if date.number - item.entry.date.number > tonumber(words[5]) then amount = amount + item.remaining end
421 end
422 if amount > 0 then targets[#targets + 1] = {partner = partner, amount = amount}; total = total + amount end
423 end
424 if total == 0 then fail("nothing to write off " .. account.code) end
425 local lines = {new_line(true, expense.code, total)}
426 for _, target in ipairs(targets) do lines[#lines + 1] = new_line(false, account.code, target.amount, target.partner) end
427 self.writeoff_count = self.writeoff_count + 1
428 local entry = self:post("WO-" .. self.writeoff_count, date, "writeoff", "writeoff", lines)
429 self:emit("WRITEOFF %s entry %s %s partners %d amount %s", account.code, entry.id, date.text, #targets, money(total))
430 for _, target in ipairs(targets) do self:emit(" %s %s", target.partner, money(target.amount)) end
431 elseif command == "CLOSE" then
432 self:close(parse_date(words[2]), words[3])
433 elseif command == "REOPEN" then
434 if #self.closes == 0 then fail("nothing to reopen") end
435 local record, detail = self.closes[#self.closes], "no entry"
436 if record.entry then
437 self:active_lines(record.entry.lines)
438 detail = "entry " .. self:reverse(record.entry, record.date, "reopen", "reopen").id
439 end
440 table.remove(self.closes)
441 self:emit("REOPEN %s %s closed %s", record.date.text, detail, self:closed())
442 elseif command == "LEDGER" then
443 local account = self:account(words[2])
444 local first, last = date_range(words[3], words[4])
445 self:ledger(account, first, last)
446 elseif command == "ACTIVITY" then
447 local account = self:account(words[2])
448 local year = tonumber(words[3])
449 local debit, credit = 0, 0
450 self:emit("ACTIVITY %s %s %d", account.code, account.name, year)
451 for month = 1, 12 do
452 local first, last = new_date(year, month, 1).number, new_date(year, month, month_days(year, month)).number
453 local totals = totals_for(self:aggregate(first, last), account.code)
454 debit, credit = debit + totals.debit, credit + totals.credit
455 self:emit(" %04d-%02d debit %s credit %s balance %s", year, month, money(totals.debit), money(totals.credit), money(self:balance(account, last)))
456 end
457 self:emit(" total debit %s credit %s", money(debit), money(credit))
458 elseif command == "TRIAL" then
459 local date = parse_date(words[2])
460 local totals = self:aggregate(0, date.number)
461 local debit, credit = 0, 0
462 self:emit("TRIAL %s", date.text)
463 for _, account in ipairs(self.account_order) do
464 local value = totals_for(totals, account.code).difference
465 if value ~= 0 then
466 if value > 0 then debit = debit + value else credit = credit - value end
467 self:emit(" %s %s %s %s", account.code, account.name, value > 0 and "debit" or "credit", money(math.abs(value)))
468 end
469 end
470 self:emit(" total debit %s credit %s", money(debit), money(credit))
471 elseif command == "INCOME" then
472 local first, last = date_range(words[2], words[3])
473 local totals = self:aggregate(first.number, last.number, true)
474 self:emit("INCOME %s %s", first.text, last.text)
475 local revenue = self:section("REVENUE", {REVENUE = true}, -1, totals)
476 local expense = self:section("EXPENSE", {EXPENSE = true}, 1, totals)
477 self:emit(" NET INCOME %s", money(revenue - expense))
478 elseif command == "BALANCE" then
479 local date = parse_date(words[2])
480 local totals, earnings = self:aggregate(0, date.number), 0
481 for _, account in ipairs(self.account_order) do
482 if account.kind == "REVENUE" or account.kind == "EXPENSE" then earnings = earnings - totals_for(totals, account.code).difference end
483 end
484 self:emit("BALANCE %s", date.text)
485 local assets = self:section("ASSETS", {ASSET = true, CONTRA_ASSET = true}, 1, totals)
486 local liabilities = self:section("LIABILITIES", {LIABILITY = true}, -1, totals)
487 local equity = self:section("EQUITY", {EQUITY = true}, -1, totals, earnings)
488 if assets == liabilities + equity then self:emit(" check ok")
489 else self:emit(" check mismatch %s", money(assets - liabilities - equity)) end
490 elseif command == "ROLLUP" then
491 local date = parse_date(words[3])
492 local targets = {}
493 for _, account in ipairs(self.account_order) do if starts_with(account.code, words[2]) then targets[#targets + 1] = account end end
494 self:emit("ROLLUP %s %s", words[2], date.text)
495 if #targets == 0 then self:emit(" (none)"); return end
496 local totals = self:aggregate(0, date.number)
497 local debit, credit = 0, 0
498 for _, account in ipairs(targets) do
499 local depth, subtotal = 0, 0
500 for _, other in ipairs(targets) do
501 if other.code ~= account.code and starts_with(account.code, other.code) then depth = depth + 1 end
502 if starts_with(other.code, account.code) then subtotal = subtotal + totals_for(totals, other.code).difference end
503 end
504 local own = totals_for(totals, account.code)
505 debit, credit = debit + own.debit, credit + own.credit
506 self:emit("%s%s %s own %s total %s", string.rep(" ", 2 * (1 + depth)), account.code, account.name, money(own.difference * normal(account)), money(subtotal * normal(account)))
507 end
508 self:emit(" postings debit %s credit %s", money(debit), money(credit))
509 elseif command == "AGING" or command == "ITEMS" then
510 local account = self:account(words[2])
511 local date = parse_date(words[#words])
512 local settlements = self:settlements(account, date.number)
513 if command == "ITEMS" then
514 local partner = words[3]
515 local settlement = settlements[partner] or {items = {}, head = 1, unapplied = 0}
516 self:emit("ITEMS %s %s %s %s", account.code, account.name, partner, date.text)
517 local remaining = 0
518 for i = settlement.head, #settlement.items do
519 local item = settlement.items[i]
520 remaining = remaining + item.remaining
521 self:emit(" %s %s amount %s open %s age %d", item.entry.date.text, item.entry.id, money(item.original), money(item.remaining), date.number - item.entry.date.number)
522 end
523 self:emit(" unapplied %s balance %s", money(settlement.unapplied), money(remaining - settlement.unapplied))
524 else
525 self:emit("AGING %s %s %s", account.code, account.name, date.text)
526 local total = {0, 0, 0, 0, 0}
527 for _, partner in ipairs(sorted_keys(settlements)) do
528 local settlement = settlements[partner]
529 if settlement.head <= #settlement.items or settlement.unapplied > 0 then
530 local values = {0, 0, 0, 0, settlement.unapplied}
531 for i = settlement.head, #settlement.items do
532 local item = settlement.items[i]
533 local age = date.number - item.entry.date.number
534 local bucket = age <= 30 and 1 or age <= 60 and 2 or age <= 90 and 3 or 4
535 values[bucket] = values[bucket] + item.remaining
536 end
537 for i = 1, 5 do total[i] = total[i] + values[i] end
538 self:aging_line(partner, values)
539 end
540 end
541 self:aging_line("total", total)
542 end
543 elseif command == "ENTRY" then
544 local entry = self:entry(words[2])
545 self:emit("ENTRY %s %s %s %s", entry.id, entry.date.text, entry.kind, entry.memo)
546 for _, line in ipairs(entry.lines) do
547 self:emit(" %s %s %s %s%s", line.debit and "DR" or "CR", line.code, self.accounts[line.code].name,
548 money(line.amount), line.partner and (" " .. line.partner) or "")
549 end
550 self:emit(" total %s", money(entry.amount))
551 if entry.reverses then self:emit(" reverses %s", entry.reverses) end
552 if entry.reversed_by then self:emit(" reversed by %s", entry.reversed_by) end
553 elseif command == "DAYBOOK" then
554 local first, last = date_range(words[2], words[3])
555 local selected, total = self:ordered_entries(first.number, last.number), 0
556 self:emit("DAYBOOK %s %s", first.text, last.text)
557 for _, entry in ipairs(selected) do
558 total = total + entry.amount
559 self:emit(" %s %s %s %s %s", entry.date.text, entry.id, entry.kind, money(entry.amount), entry.memo)
560 end
561 self:emit(" total entries %d amount %s", #selected, money(total))
562 elseif command == "ACCOUNTS" then
563 local totals = self:aggregate()
564 self:emit("ACCOUNTS")
565 for _, account in ipairs(self.account_order) do
566 local parent = self:parent(account)
567 self:emit(" %s %s %s parent %s balance %s %s", account.code, account.kind, account.name, parent and parent.code or "none",
568 money(totals_for(totals, account.code).difference * normal(account)), account.active and "active" or "inactive")
569 end
570 if #self.account_order == 0 then self:emit(" (none)") end
571 elseif command == "TEMPLATES" then
572 self:emit("TEMPLATES")
573 for _, template in ipairs(self.template_order) do
574 local amount = line_totals(template.lines)
575 self:emit(" %s %s lines %d amount %s used %d", template.name, template.memo, #template.lines, money(amount), template.uses)
576 end
577 if #self.template_order == 0 then self:emit(" (none)") end
578 elseif command == "STATUS" then
579 local inactive = 0
580 for _, account in ipairs(self.account_order) do if not account.active then inactive = inactive + 1 end end
581 self:emit("STATUS accounts %d inactive %d entries %d templates %d closed %s open %s", #self.account_order, inactive, #self.entries,
582 #self.template_order, self:closed(), self.draft and self.draft.key or "none")
583 end
584end
