31month_length <- function(year, month) {
32 lengths <- c(31L, 28L, 31L, 30L, 31L, 30L, 31L, 31L, 30L, 31L, 30L, 31L)
33 lengths[month] + as.integer(month == 2L && leap(year))

36day_number <- function(year, month, day) {
37 prior <- year - 1L
38 result <- 365L * prior + prior %/% 4L - prior %/% 100L + prior %/% 400L + day - 730120L
39 for (m in seq_len(month - 1L)) result <- result + month_length(year, m)
40 result
41}

43date_parts <- function(number) {
44 year <- 2000L
45 while (number >= 365L + as.integer(leap(year))) {
46 number <- number - 365L - as.integer(leap(year))
47 year <- year + 1L
48 }
49 month <- 1L
50 while (number >= month_length(year, month)) {
51 number <- number - month_length(year, month)
52 month <- month + 1L
53 }
54 c(year, month, number + 1L)
55}

62checked_date <- function(text) {
63 parts <- as.integer(strsplit(text, "-", fixed = TRUE)[[1]])
64 year <- parts[1]
65 month <- parts[2]
66 day <- parts[3]
67 if (year < 2000L || year > 2099L || month < 1L || month > 12L || day < 1L) {
68 fail(paste("invalid date", text))
69 }
70 if (day > month_length(year, month)) fail(paste("invalid date", text))
71 day_number(year, month, day)
72}

81months_after <- function(number, count) {
82 parts <- date_parts(number)
83 offset <- parts[2] - 1L + count
84 year <- parts[1] + offset %/% 12L
85 month <- offset %% 12L + 1L
86 day_number(year, month, min(parts[3], month_length(year, month)))
87}

179aggregate_entries <- function(books, first = 0L, last = 36524L, operating = FALSE) {
180 totals <- list()
181 for (entry in books$entries) {
182 if (entry$day < first || entry$day > last ||
183 (operating && entry$category %in% c("closing", "reopen"))) next
184 for (line in entry$lines) {
185 values <- total_pair(totals, line$code)
186 side <- if (line$debit) 1L else 2L
187 values[side] <- values[side] + line$amount
188 totals[[line$code]] <- values
189 }
190 }
191 totals
192}

198daily_sum <- function(books, account, first, last) {
199 changes <- rep(zero, last - first + 1L)
200 opening <- zero
201 for (entry in books$entries) {
202 if (entry$day > last) next
203 selected <- Filter(function(line) line$code == account$code, entry$lines)
204 change <- account$sign * sum_big(lapply(selected, line_difference))
205 if (entry$day < first) {
206 opening <- opening + change
207 } else {
208 index <- entry$day - first + 1L
209 changes[index] <- changes[index] + change
210 }
211 }
212 sum(cumsum(changes) + opening)
213}

248settlements_for <- function(books, account, day) {
249 settlements <- list()
250 for (index in posting_order(books)) {
251 entry <- books$entries[[index]]
252 if (entry$day > day) break
253 for (line in entry$lines) {
254 if (line$code != account$code || is.na(line$partner)) next
255 group <- settlements[[line$partner]]
256 if (is.null(group)) group <- new_settlement()
257 value <- account$sign * line_difference(line)
258 if (value > 0L) {
259 used <- min(value, group$unapplied)
260 group$unapplied <- group$unapplied - used
261 if (value > used) {
262 group$items[[length(group$items) + 1L]] <- list(entry = index, original = value, remaining = value - used)
263 }
264 } else {
265 remaining <- -value
266 while (remaining > 0L && group$head <= length(group$items)) {
267 used <- min(remaining, group$items[[group$head]]$remaining)
268 group$items[[group$head]]$remaining <- group$items[[group$head]]$remaining - used
269 remaining <- remaining - used
270 if (group$items[[group$head]]$remaining == 0L) group$head <- group$head + 1L
271 }
272 group$unapplied <- group$unapplied + remaining
273 }
274 settlements[[line$partner]] <- group
275 }
276 }
277 settlements
278}

280section <- function(books, heading, kinds, sign, totals, earnings = NULL) {
281 selected <- Filter(function(code) books$accounts[[code]]$kind %in% kinds, sorted_names(books$accounts))
282 total <- sign * sum_big(lapply(selected, function(code) difference(totals, code)))
283 if (!is.null(earnings)) total <- total + earnings
284 emit(paste0(" ", heading), money(total))
285 roots <- vapply(selected, function(code) top_account(books, code), character(1))
286 for (root in sort(unique(roots), method = "radix")) {
287 group <- selected[roots == root]
288 values <- c_bigz(lapply(group, function(code) sign * difference(totals, code)))
289 if (!any(values != 0L)) next
290 emit(paste0(" ", root), books$accounts[[root]]$name, money(sum(values)))
291 for (i in which(values != 0L)) {
292 code <- group[i]
293 emit(paste0(" ", code), books$accounts[[code]]$name, money(values[i]))
294 }
295 }
296 if (!is.null(earnings)) emit(" current earnings", money(earnings))
297 total
298}

305post_draft <- function(books) {
306 block <- books$draft
307 if (is.null(block)) fail("no open entry")
308 books$draft <- NULL
309 journal <- block$kind == "journal"
310 day <- 0L
311 if (journal) {
312 if (!is.null(books$entry_positions[[block$key]])) fail(paste("duplicate entry", block$key))
313 day <- checked_date(block$date_token)
314 check_closed(books, day)
315 } else if (!is.null(books$templates[[block$key]])) {
316 fail(paste("duplicate template", block$key))
317 }
318 if (length(block$lines) < 2L) fail(paste("too few lines", block$key))
319 for (line in block$lines) {
320 account <- get_account(books, line$code)
321 if (journal && !account$active) fail(paste("inactive account", line$code))
322 if (line$amount == 0L) fail(paste("zero amount", line$code))
323 }
324 values <- line_totals(block$lines)
325 if (values[1] != values[2]) {
326 fail(paste("unbalanced", block$key, "debit", money(values[1]), "credit", money(values[2])))
327 }
328 if (journal) {
329 index <- post_entry(books, block$key, day, "journal", block$memo, block$lines)
330 emit(posted_text(books$entries[[index]]))
331 } else {
332 books$templates[[block$key]] <- list(memo = block$memo, lines = block$lines, uses = 0L)
333 emit("TEMPLATE", block$key, "lines", length(block$lines), "amount", money(values[1]))
334 }
335}

337command <- function(books, words) {
338 op <- words[1]
339 if (op == "ACCOUNT") {
340 code <- words[2]
341 if (!is.null(books$accounts[[code]])) fail(paste("duplicate account", code))
342 kind <- words[3]
343 family <- if (kind == "CONTRA_ASSET") "ASSET" else kind
344 for (other in sorted_names(books$accounts)) {
345 if ((startsWith(code, other) || startsWith(other, code)) && books$accounts[[other]]$family != family) {
346 fail(paste("class mismatch", code, other))
347 }
348 }
349 books$accounts[[code]] <- list(code = code, kind = kind, name = words[4], active = TRUE,
350 sign = if (kind %in% c("ASSET", "EXPENSE")) 1L else -1L, family = family)
351 emit("ACCOUNT", code, kind, words[4], "parent", parent_code(books, code))
352 } else if (op == "DEACTIVATE") {
353 account <- get_account(books, words[2])
354 if (!account$active) fail(paste("already inactive", account$code))
355 balance <- account_balance(books, account)
356 if (balance != 0L) fail(paste("nonzero balance", account$code, money(balance)))
357 books$accounts[[account$code]]$active <- FALSE
358 emit("DEACTIVATE", account$code, account$name)
359 } else if (op %in% c("JOURNAL", "TEMPLATE")) {
360 if (!is.null(books$draft)) emit("DISCARD", books$draft$key, "lines", length(books$draft$lines))
361 journal <- op == "JOURNAL"
362 books$draft <- list(kind = if (journal) "journal" else "template", key = words[2],
363 date_token = if (journal) words[3] else NULL,
364 memo = if (journal) words[4] else words[3], lines = list())
365 } else if (op %in% c("DR", "CR")) {
366 if (is.null(books$draft)) fail("no open entry")
367 partner <- if (length(words) == 4L) words[4] else NA_character_
368 books$draft$lines[[length(books$draft$lines) + 1L]] <- posting_line(op == "DR", words[2], parse_cents(words[3]), partner)
369 } else if (op == "POST") {
370 post_draft(books)
371 } else if (op == "DRAFT") {
372 block <- books$draft
373 if (is.null(block)) fail("no open entry")
374 values <- line_totals(block$lines)
375 emit("DRAFT", block$kind, block$key, "lines", length(block$lines), "debit", money(values[1]),
376 "credit", money(values[2]), "difference", money(values[1] - values[2]))
377 } else if (op == "REVERSE") {
378 index <- get_entry(books, words[2])
379 original <- books$entries[[index]]
380 if (original$category %in% c("reversal", "closing", "reopen")) fail(paste("cannot reverse", original$id))
381 if (!is.null(original$reversed_by)) fail(paste("already reversed", original$id))
382 day <- checked_date(words[3])
383 if (day < original$day) fail(paste("reverse before entry", original$id, date_text(original$day)))
384 check_closed(books, day)
385 active_check(books, line_codes(original$lines))
386 created <- reverse_entry(books, index, day, "reversal", original$memo)
387 posted <- books$entries[[created]]
388 emit("REVERSE", original$id, "entry", posted$id, date_text(day), "amount", money(posted$amount))
389 } else if (op == "RECUR") {
390 pattern <- get_template(books, words[2])
391 day <- checked_date(words[3])
392 count <- as.integer(words[4])
393 if (count == 0L) fail("invalid count")
394 check_closed(books, day)
395 active_check(books, line_codes(pattern$lines))
396 emit("RECUR", words[2], "count", count)
397 for (offset in seq_len(count) - 1L) {
398 pattern$uses <- pattern$uses + 1L
399 id <- paste0(words[2], "-", pattern$uses)
400 index <- post_entry(books, id, months_after(day, offset), "recurring", pattern$memo, pattern$lines)
401 emit(paste0(" ", posted_text(books$entries[[index]])))
402 }
403 books$templates[[words[2]]] <- pattern
404 } else if (op == "DEPRECIATE") {
405 day <- checked_date(words[2])
406 check_closed(books, day)
407 asset <- get_account(books, words[3], "ASSET")
408 contra <- get_account(books, words[4], "CONTRA_ASSET")
409 expense <- get_account(books, words[5], "EXPENSE")
410 active_check(books, c(contra$code, expense$code))
411 cost <- account_balance(books, asset, day)
412 accumulated <- account_balance(books, contra, day)
413 amount <- if (cost > 0L) min(cost %/% as.integer(words[6]), cost - accumulated) else zero
414 if (amount <= 0L) fail(paste("nothing to depreciate", asset$code))
415 books$depreciation_count <- books$depreciation_count + 1L
416 id <- paste0("DEP-", books$depreciation_count)
417 post_entry(books, id, day, "depreciation", "depreciation",
418 list(posting_line(TRUE, expense$code, amount), posting_line(FALSE, contra$code, amount)))
419 emit("DEPRECIATE", asset$code, "entry", id, date_text(day), "amount", money(amount),
420 "accumulated", money(accumulated + amount), "book", money(cost - accumulated - amount))
421 } else if (op == "INTEREST") {
422 account <- get_account(books, words[2])
423 if (!account$kind %in% c("ASSET", "LIABILITY")) fail(paste("not interest bearing", account$code))
424 offset <- get_account(books, words[3], if (account$kind == "ASSET") "REVENUE" else "EXPENSE")
425 period <- checked_range(words[4], words[5])
426 check_closed(books, period[2])
427 active_check(books, c(account$code, offset$code))
428 total <- daily_sum(books, account, period[1], period[2])
429 amount <- (total * as.integer(words[6])) %/% 3650000L
430 if (amount <= 0L) fail(paste("no interest", account$code))
431 books$interest_count <- books$interest_count + 1L
432 id <- paste0("INT-", books$interest_count)
433 debit <- if (account$kind == "ASSET") account$code else offset$code
434 credit <- if (account$kind == "ASSET") offset$code else account$code
435 post_entry(books, id, period[2], "interest", "interest",
436 list(posting_line(TRUE, debit, amount), posting_line(FALSE, credit, amount)))
437 days <- period[2] - period[1] + 1L
438 emit("INTEREST", account$code, "entry", id, date_text(period[2]), "days", days,
439 "average", money(total %/% days), "amount", money(amount))
440 } else if (op == "WRITEOFF") {
441 day <- checked_date(words[2])
442 check_closed(books, day)
443 account <- get_account(books, words[3], "ASSET")
444 expense <- get_account(books, words[4], "EXPENSE")
445 active_check(books, c(account$code, expense$code))
446 age <- as.integer(words[5])
447 settlements <- settlements_for(books, account, day)
448 amounts <- list()
449 for (partner in sorted_names(settlements)) {
450 old <- Filter(function(item) day - books$entries[[item$entry]]$day > age, open_items(settlements[[partner]]))
451 amount <- sum_big(lapply(old, function(item) item$remaining))
452 if (amount > 0L) amounts[[partner]] <- amount
453 }
454 total <- sum_big(amounts)
455 if (total == 0L) fail(paste("nothing to write off", account$code))
456 lines <- list(posting_line(TRUE, expense$code, total))
457 for (partner in names(amounts)) {
458 lines[[length(lines) + 1L]] <- posting_line(FALSE, account$code, amounts[[partner]], partner)
459 }
460 books$writeoff_count <- books$writeoff_count + 1L
461 id <- paste0("WO-", books$writeoff_count)
462 post_entry(books, id, day, "writeoff", "writeoff", lines)
463 emit("WRITEOFF", account$code, "entry", id, date_text(day), "partners", length(amounts), "amount", money(total))
464 for (partner in names(amounts)) emit(paste0(" ", partner), money(amounts[[partner]]))
465 } else if (op == "CLOSE") {
466 day <- checked_date(words[2])
467 check_closed(books, day)
468 equity <- get_account(books, words[3], "EQUITY")
469 active_check(books, equity$code)
470 totals <- aggregate_entries(books, last = day)
471 selected <- Filter(function(code) books$accounts[[code]]$kind %in% c("REVENUE", "EXPENSE") &&
472 difference(totals, code) != 0L, sorted_names(books$accounts))
473 active_check(books, selected)
474 lines <- lapply(selected, function(code) {
475 delta <- difference(totals, code)
476 posting_line(delta < 0L, code, abs(delta))
477 })
478 income <- -sum_big(lapply(selected, function(code) difference(totals, code)))
479 if (income != 0L) lines[[length(lines) + 1L]] <- posting_line(income < 0L, equity$code, abs(income))
480 books$close_count <- books$close_count + 1L
481 index <- NULL
482 if (length(lines)) {
483 id <- paste0("CLOSE-", books$close_count)
484 index <- post_entry(books, id, day, "closing", "closing", lines)
485 emit("CLOSE", date_text(day), "entry", id, "lines", length(lines), "net income", money(income))
486 } else {
487 emit("CLOSE", date_text(day), "no entry")
488 }
489 books$closes[[length(books$closes) + 1L]] <- list(day = day, entry = index)
490 } else if (op == "REOPEN") {
491 if (!length(books$closes)) fail("nothing to reopen")
492 record <- tail(books$closes, 1L)[[1]]
493 detail <- "no entry"
494 if (!is.null(record$entry)) {
495 active_check(books, line_codes(books$entries[[record$entry]]$lines))
496 created <- reverse_entry(books, record$entry, record$day, "reopen", "reopen")
497 detail <- paste("entry", books$entries[[created]]$id)
498 }
499 books$closes <- head(books$closes, -1L)
500 emit("REOPEN", date_text(record$day), detail, "closed", closed_text(books))
501 } else if (op == "LEDGER") {
502 account <- get_account(books, words[2])
503 period <- checked_range(words[3], words[4])
504 running <- account_balance(books, account, period[1] - 1L)
505 debit <- credit <- zero
506 emit("LEDGER", account$code, account$name, date_text(period[1]), date_text(period[2]))
507 emit(" opening", money(running))
508 for (index in posting_order(books)) {
509 entry <- books$entries[[index]]
510 if (entry$day < period[1] || entry$day > period[2]) next
511 for (line in entry$lines) {
512 if (line$code != account$code) next
513 running <- running + account$sign * line_difference(line)
514 if (line$debit) debit <- debit + line$amount else credit <- credit + line$amount
515 emit(paste0(" ", date_text(entry$day)), entry$id, if (line$debit) "DR" else "CR",
516 money(line$amount), "balance", money(running), entry$memo)
517 }
518 }
519 days <- period[2] - period[1] + 1L
520 average <- daily_sum(books, account, period[1], period[2]) %/% days
521 emit(" ending", money(running), "debits", money(debit), "credits", money(credit), "days", days, "average", money(average))
522 } else if (op == "ACTIVITY") {
523 account <- get_account(books, words[2])
524 year <- as.integer(words[3])
525 total <- rep(zero, 2L)
526 emit("ACTIVITY", account$code, account$name, year)
527 for (month in seq_len(12L)) {
528 first <- day_number(year, month, 1L)
529 last <- first + month_length(year, month) - 1L
530 values <- total_pair(aggregate_entries(books, first, last), account$code)
531 total <- total + values
532 emit(paste0(" ", sprintf("%04d-%02d", year, month)), "debit", money(values[1]),
533 "credit", money(values[2]), "balance", money(account_balance(books, account, last)))
534 }
535 emit(" total debit", money(total[1]), "credit", money(total[2]))
536 } else if (op == "TRIAL") {
537 day <- checked_date(words[2])
538 totals <- aggregate_entries(books, last = day)
539 debit <- credit <- zero
540 emit("TRIAL", date_text(day))
541 for (code in sorted_names(books$accounts)) {
542 delta <- difference(totals, code)
543 if (delta == 0L) next
544 emit(paste0(" ", code), books$accounts[[code]]$name, if (delta > 0L) "debit" else "credit", money(abs(delta)))
545 if (delta > 0L) debit <- debit + delta else credit <- credit - delta
546 }
547 emit(" total debit", money(debit), "credit", money(credit))
548 } else if (op == "INCOME") {
549 period <- checked_range(words[2], words[3])
550 totals <- aggregate_entries(books, period[1], period[2], operating = TRUE)
551 emit("INCOME", date_text(period[1]), date_text(period[2]))
552 revenue <- section(books, "REVENUE", "REVENUE", -1L, totals)
553 expense <- section(books, "EXPENSE", "EXPENSE", 1L, totals)
554 emit(" NET INCOME", money(revenue - expense))
555 } else if (op == "BALANCE") {
556 day <- checked_date(words[2])
557 totals <- aggregate_entries(books, last = day)
558 selected <- Filter(function(code) books$accounts[[code]]$kind %in% c("REVENUE", "EXPENSE"), names(books$accounts))
559 earnings <- -sum_big(lapply(selected, function(code) difference(totals, code)))
560 emit("BALANCE", date_text(day))
561 assets <- section(books, "ASSETS", c("ASSET", "CONTRA_ASSET"), 1L, totals)
562 liabilities <- section(books, "LIABILITIES", "LIABILITY", -1L, totals)
563 equity <- section(books, "EQUITY", "EQUITY", -1L, totals, earnings)
564 if (assets == liabilities + equity) emit(" check ok") else emit(" check mismatch", money(assets - liabilities - equity))
565 } else if (op == "ROLLUP") {
566 day <- checked_date(words[3])
567 selected <- sorted_names(books$accounts)
568 selected <- selected[startsWith(selected, words[2])]
569 emit("ROLLUP", words[2], date_text(day))
570 if (!length(selected)) {
571 emit(" (none)")
572 return(invisible(NULL))
573 }
574 totals <- aggregate_entries(books, last = day)
575 postings <- rep(zero, 2L)
576 for (code in selected) {
577 account <- books$accounts[[code]]
578 descendants <- selected[startsWith(selected, code)]
579 depth <- sum(selected != code & startsWith(code, selected))
580 subtotal <- account$sign * sum_big(lapply(descendants, function(other) difference(totals, other)))
581 emit(paste0(strrep(" ", 2L * (1L + depth)), code), account$name,
582 "own", money(account$sign * difference(totals, code)), "total", money(subtotal))
583 postings <- postings + total_pair(totals, code)
584 }
585 emit(" postings debit", money(postings[1]), "credit", money(postings[2]))
586 } else if (op %in% c("AGING", "ITEMS")) {
587 account <- get_account(books, words[2])
588 day <- checked_date(tail(words, 1L))
589 settlements <- settlements_for(books, account, day)
590 if (op == "ITEMS") {
591 group <- settlements[[words[3]]]
592 if (is.null(group)) group <- new_settlement()
593 emit("ITEMS", account$code, account$name, words[3], date_text(day))
594 items <- open_items(group)
595 for (item in items) {
596 entry <- books$entries[[item$entry]]
597 emit(paste0(" ", date_text(entry$day)), entry$id, "amount", money(item$original),
598 "open", money(item$remaining), "age", day - entry$day)
599 }
600 balance <- sum_big(lapply(items, function(item) item$remaining)) - group$unapplied
601 emit(" unapplied", money(group$unapplied), "balance", money(balance))
602 } else {
603 emit("AGING", account$code, account$name, date_text(day))
604 totals <- rep(zero, 5L)
605 for (partner in sorted_names(settlements)) {
606 group <- settlements[[partner]]
607 items <- open_items(group)
608 if (!length(items) && group$unapplied == 0L) next
609 values <- rep(zero, 5L)
610 for (item in items) {
611 age <- day - books$entries[[item$entry]]$day
612 bucket <- if (age <= 30L) 1L else if (age <= 60L) 2L else if (age <= 90L) 3L else 4L
613 values[bucket] <- values[bucket] + item$remaining
614 }
615 values[5] <- group$unapplied
616 totals <- totals + values
617 emit(paste0(" ", partner), aging_columns(values))
618 }
619 emit(" total", aging_columns(totals))
620 }
621 } else if (op == "ENTRY") {
622 entry <- books$entries[[get_entry(books, words[2])]]
623 emit("ENTRY", entry$id, date_text(entry$day), entry$category, entry$memo)
624 for (line in entry$lines) {
625 detail <- paste(if (line$debit) "DR" else "CR", line$code, books$accounts[[line$code]]$name, money(line$amount))
626 if (!is.na(line$partner)) detail <- paste(detail, line$partner)
627 emit(paste0(" ", detail))
628 }
629 emit(" total", money(entry$amount))
630 if (!is.null(entry$reverses)) emit(" reverses", books$entries[[entry$reverses]]$id)
631 if (!is.null(entry$reversed_by)) emit(" reversed by", books$entries[[entry$reversed_by]]$id)
632 } else if (op == "DAYBOOK") {
633 period <- checked_range(words[2], words[3])
634 count <- 0L
635 total <- zero
636 emit("DAYBOOK", date_text(period[1]), date_text(period[2]))
637 for (index in posting_order(books)) {
638 entry <- books$entries[[index]]
639 if (entry$day < period[1] || entry$day > period[2]) next
640 count <- count + 1L
641 total <- total + entry$amount
642 emit(paste0(" ", date_text(entry$day)), entry$id, entry$category, money(entry$amount), entry$memo)
643 }
644 emit(" total entries", count, "amount", money(total))
645 } else if (op == "ACCOUNTS") {
646 emit("ACCOUNTS")
647 codes <- sorted_names(books$accounts)
648 if (!length(codes)) emit(" (none)")
649 totals <- aggregate_entries(books)
650 for (code in codes) {
651 account <- books$accounts[[code]]
652 emit(paste0(" ", code), account$kind, account$name, "parent", parent_code(books, code),
653 "balance", money(account$sign * difference(totals, code)), if (account$active) "active" else "inactive")
654 }
655 } else if (op == "TEMPLATES") {
656 emit("TEMPLATES")
657 template_names <- sorted_names(books$templates)
658 if (!length(template_names)) emit(" (none)")
659 for (name in template_names) {
660 pattern <- books$templates[[name]]
661 emit(paste0(" ", name), pattern$memo, "lines", length(pattern$lines),
662 "amount", money(line_totals(pattern$lines)[1]), "used", pattern$uses)
663 }
664 } else if (op == "STATUS") {
665 inactive <- sum(vapply(books$accounts, function(account) !account$active, logical(1)))
666 open <- if (is.null(books$draft)) "none" else books$draft$key
667 emit("STATUS accounts", length(books$accounts), "inactive", inactive, "entries", length(books$entries),
668 "templates", length(books$templates), "closed", closed_text(books), "open", open)
669 }
670 invisible(NULL)
671}

676for (i in seq_len(count)) {

678 tryCatch(command(books, words), bookkeeping_error = function(problem) emit("ERROR", conditionMessage(problem)))
679}
