16fn magnitude(value: i64) i64 {

19fn cents(text: []const u8) i64 {

32const Date = struct {
33 year: i64,
34 month: i64,
35 day: i64,
36 fn monthDays(year: i64, month: i64) i64 {
37 const lengths = [_]i64{ 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31 };
38 const leap = @mod(year, 4) == 0 and (@mod(year, 100) != 0 or @mod(year, 400) == 0);
39 return lengths[@intCast(month - 1)] + @as(i64, if (month == 2 and leap) 1 else 0);
40 }
41 fn parse(text: []const u8) ?Date {
42 const result: Date = .{ .year = integer(text[0..4]), .month = integer(text[5..7]), .day = integer(text[8..]) };
43 if (result.year < 2000 or result.year > 2099 or result.month < 1 or result.month > 12 or result.day < 1) return null;
44 if (result.day > monthDays(result.year, result.month)) return null;
45 return result;
46 }
47 fn number(date: Date) i64 {
48 const prior = date.year - 1;
49 var result = 365 * (date.year - 2000) + @divTrunc(prior, 4) - @divTrunc(1999, 4) - @divTrunc(prior, 100) + @divTrunc(1999, 100) + @divTrunc(prior, 400) - @divTrunc(1999, 400);
50 var month: i64 = 1;
51 while (month < date.month) : (month += 1) result += monthDays(date.year, month);
52 return result + date.day - 1;
53 }
54 fn addMonths(date: Date, count: usize) Date {
55 const offset = date.month - 1 + @as(i64, @intCast(count));
56 const year = date.year + @divTrunc(offset, 12);
57 const month = @mod(offset, 12) + 1;
58 return .{ .year = year, .month = month, .day = @min(date.day, monthDays(year, month)) };
59 }
60 pub fn format(date: Date, out: *std.Io.Writer) std.Io.Writer.Error!void {
61 try out.print("{d:0>4}-{d:0>2}-{d:0>2}", .{ date.year, date.month, date.day });
62 }

248 fn aggregate(books: *const Books, first: i64, last: i64, operating: bool) AccountTotals {
249 var result: AccountTotals = @splat(.{});
250 for (books.entries.items) |posted| {
251 if (posted.date.number() < first or posted.date.number() > last or (operating and (posted.kind == .closing or posted.kind == .reopen))) continue;
252 for (posted.lines) |line| result[line.account].add(line);
253 }
254 return result;

256 fn balance(books: *const Books, selected: usize, last: i64) i64 {
257 var difference: i64 = 0;
258 for (books.entries.items) |posted| {
259 if (posted.date.number() > last) continue;
260 for (posted.lines) |line| if (line.account == selected) {
261 difference += line.difference();
262 };
263 }
264 return difference * books.accounts[selected].kind.sign();
265 }
266 fn dailySum(books: *const Books, selected: usize, first: i64, last: i64) i64 {
267 var total: i64 = 0;
268 for (books.entries.items) |posted| {
269 const day = posted.date.number();
270 if (day > last) continue;
271 var change: i64 = 0;
272 for (posted.lines) |line| if (line.account == selected) {
273 change += line.difference();
274 };
275 total += change * books.accounts[selected].kind.sign() * (last - @max(first, day) + 1);
276 }
277 return total;
278 }

321 fn settlements(books: *Books, selected: usize, day: i64) Allocator.Error![]Settlement {
322 var result: std.ArrayList(Settlement) = .empty;
323 var partners: std.StringHashMapUnmanaged(usize) = .empty;
324 for (try books.orderedEntries()) |posted| {
325 if (posted.date.number() > day) continue;
326 for (posted.lines) |line| {
327 if (line.account != selected) continue;
328 const partner = line.partner orelse continue;
329 const index = partners.get(partner) orelse blk: {
330 const position = result.items.len;
331 try result.append(books.gpa, .{ .partner = partner });
332 try partners.put(books.gpa, partner, position);
333 break :blk position;
334 };
335 const settlement = &result.items[index];
336 const signed = line.difference() * books.accounts[selected].kind.sign();
337 if (signed > 0) {
338 const applied = @min(signed, settlement.unapplied);
339 settlement.unapplied -= applied;
340 if (signed > applied) try settlement.items.append(books.gpa, .{ .entry = posted, .original = signed, .remaining = signed - applied });
341 } else {
342 var remaining = -signed;
343 while (remaining > 0 and settlement.head < settlement.items.items.len) {
344 const item = &settlement.items.items[settlement.head];
345 const applied = @min(remaining, item.remaining);
346 item.remaining -= applied;
347 remaining -= applied;
348 if (item.remaining == 0) settlement.head += 1;
349 }
350 settlement.unapplied += remaining;
351 }
352 }
353 }
354 std.mem.sort(Settlement, result.items, {}, settlementLess);
355 return try result.toOwnedSlice(books.gpa);
356 }
357 fn section(books: *Books, heading: Section, totals: *const AccountTotals, earnings: ?i64) std.Io.Writer.Error!i64 {
358 const order = books.orderedAccounts();
359 var roots: [max_accounts]usize = undefined;
360 for (order.slice()) |i| roots[i] = books.top(i);
361 var total = earnings orelse 0;
362 for (order.slice()) |i| if (heading.includes(books.accounts[i].kind)) {
363 total += totals[i].difference() * heading.sign();
364 };
365 try books.out.print(" {s} {f}\n", .{ @tagName(heading), money(total) });
366 for (order.slice()) |root| {
367 if (roots[root] != root) continue;
368 var group: i64 = 0;
369 var visible = false;
370 for (order.slice()) |i| {
371 if (!heading.includes(books.accounts[i].kind) or roots[i] != root) continue;
372 const value = totals[i].difference() * heading.sign();
373 group += value;
374 visible = visible or value != 0;
375 }
376 if (!visible) continue;
377 try books.out.print(" {s} {s} {f}\n", .{ books.accounts[root].code, books.accounts[root].name, money(group) });
378 for (order.slice()) |i| {
379 if (!heading.includes(books.accounts[i].kind) or roots[i] != root) continue;
380 const value = totals[i].difference() * heading.sign();
381 if (value != 0) try books.out.print(" {s} {s} {f}\n", .{ books.accounts[i].code, books.accounts[i].name, money(value) });
382 }
383 }
384 if (earnings) |value| try books.out.print(" current earnings {f}\n", .{money(value)});
385 return total;
386 }
387 fn postDraft(books: *Books) Error!void {
388 const block = books.draft orelse return books.fail("no open entry", .{});
389 books.draft = null;
390 var value: Date = undefined;
391 if (block.journal) {
392 if (books.entry_ids.contains(block.key)) return books.fail("duplicate entry {s}", .{block.key});
393 value = try books.date(block.date.?);
394 try books.checkClosed(value);
395 } else if (books.templates.contains(block.key)) return books.fail("duplicate template {s}", .{block.key});
396 if (block.lines.items.len < 2) return books.fail("too few lines {s}", .{block.key});
397 const lines = try books.gpa.alloc(Line, block.lines.items.len);
398 var totals: Totals = .{};
399 for (block.lines.items, lines) |source, *line| {
400 const selected = try books.account(source.code);
401 if (block.journal) try books.active(&.{selected});
402 if (source.amount == 0) return books.fail("zero amount {s}", .{source.code});
403 line.* = .{ .debit = source.debit, .account = selected, .amount = source.amount, .partner = source.partner };
404 totals.add(line.*);
405 }
406 if (totals.debit != totals.credit) return books.fail("unbalanced {s} debit {f} credit {f}", .{ block.key, money(totals.debit), money(totals.credit) });
407 if (block.journal) {
408 const posted = try books.post(block.key, value, .journal, block.memo, lines);
409 try books.postText(posted, 0);
410 } else {
411 const template = try books.gpa.create(Template);
412 template.* = .{ .name = block.key, .memo = block.memo, .lines = lines };
413 try books.templates.put(books.gpa, block.key, template);
414 try books.out.print("TEMPLATE {s} lines {d} amount {f}\n", .{ block.key, lines.len, money(totals.debit) });
415 }
416 }
417 fn close(books: *Books, value: Date, code: []const u8) Error!void {
418 try books.checkClosed(value);
419 const equity = try books.typedAccount(code, .EQUITY);
420 try books.active(&.{equity});
421 const totals = books.aggregate(0, value.number(), false);
422 const order = books.orderedAccounts();
423 var targets: AccountOrder = .{};
424 for (order.slice()) |i| if ((books.accounts[i].kind == .REVENUE or books.accounts[i].kind == .EXPENSE) and totals[i].difference() != 0) {
425 targets.append(i);
426 };
427 try books.active(targets.slice());
428 var lines: std.ArrayList(Line) = .empty;
429 var income: i64 = 0;
430 for (targets.slice()) |i| {
431 const difference = totals[i].difference();
432 try lines.append(books.gpa, .{ .debit = difference < 0, .account = i, .amount = magnitude(difference) });
433 income -= difference;
434 }
435 if (income != 0) try lines.append(books.gpa, .{ .debit = income < 0, .account = equity, .amount = magnitude(income) });
436 books.close_count += 1;
437 var posted: ?*Entry = null;
438 if (lines.items.len > 0) {
439 const id = try std.fmt.allocPrint(books.gpa, "CLOSE-{d}", .{books.close_count});
440 posted = try books.post(id, value, .closing, "closing", lines.items);
441 try books.out.print("CLOSE {f} entry {s} lines {d} net income {f}\n", .{ value, id, lines.items.len, money(income) });
442 } else try books.out.print("CLOSE {f} no entry\n", .{value});
443 try books.closes.append(books.gpa, .{ .date = value, .entry = posted });
444 }
445 fn ledger(books: *Books, selected: usize, first: Date, last: Date) Error!void {
446 const found = books.accounts[selected];
447 var running = books.balance(selected, first.number() - 1);
448 var totals: Totals = .{};
449 try books.out.print("LEDGER {s} {s} {f} {f}\n opening {f}\n", .{ found.code, found.name, first, last, money(running) });
450 for (try books.orderedEntries()) |posted| {
451 if (posted.date.number() < first.number() or posted.date.number() > last.number()) continue;
452 for (posted.lines) |line| {
453 if (line.account != selected) continue;
454 running += line.difference() * found.kind.sign();
455 totals.add(line);
456 try books.out.print(" {f} {s} {s} {f} balance {f} {s}\n", .{ posted.date, posted.id, line.side(), money(line.amount), money(running), posted.memo });
457 }
458 }
459 const days = last.number() - first.number() + 1;
460 try books.out.print(" ending {f} debits {f} credits {f} days {d} average {f}\n", .{ money(running), money(totals.debit), money(totals.credit), days, money(@divTrunc(books.dailySum(selected, first.number(), last.number()), days)) });
461 }

499 fn execute(books: *Books, words: []const []const u8) Error!void {
500 const command = std.meta.stringToEnum(Command, words[0]).?;
501 switch (command) {
502 .ACCOUNT => {
503 const code = words[1];
504 if (books.account_codes.contains(code)) return books.fail("duplicate account {s}", .{code});
505 const kind = std.meta.stringToEnum(AccountType, words[2]).?;
506 const order = books.orderedAccounts();
507 for (order.slice()) |i| {
508 const other = books.accounts[i];
509 if ((prefix(code, other.code) or prefix(other.code, code)) and kind.class() != other.kind.class())
510 return books.fail("class mismatch {s} {s}", .{ code, other.code });
511 }
512 const selected = books.account_count;
513 books.accounts[selected] = .{ .code = code, .kind = kind, .name = words[3] };
514 books.account_count += 1;
515 try books.account_codes.put(books.gpa, code, selected);
516 try books.out.print("ACCOUNT {s} {s} {s} parent {s}\n", .{ code, @tagName(kind), words[3], books.parentCode(selected) });
517 },
518 .DEACTIVATE => {
519 const selected = try books.account(words[1]);
520 const found = &books.accounts[selected];
521 if (!found.active) return books.fail("already inactive {s}", .{found.code});
522 const current = books.balance(selected, last_day);
523 if (current != 0) return books.fail("nonzero balance {s} {f}", .{ found.code, money(current) });
524 found.active = false;
525 try books.out.print("DEACTIVATE {s} {s}\n", .{ found.code, found.name });
526 },
527 .JOURNAL, .TEMPLATE => {
528 if (books.draft) |previous| try books.out.print("DISCARD {s} lines {d}\n", .{ previous.key, previous.lines.items.len });
529 const block = try books.gpa.create(Draft);
530 block.* = .{ .journal = command == .JOURNAL, .key = words[1], .date = if (command == .JOURNAL) words[2] else null, .memo = if (command == .JOURNAL) words[3] else words[2] };
531 books.draft = block;
532 },
533 .DR, .CR => {
534 const block = books.draft orelse return books.fail("no open entry", .{});
535 try block.lines.append(books.gpa, .{ .debit = command == .DR, .code = words[1], .amount = cents(words[2]), .partner = if (words.len == 4) words[3] else null });
536 },
537 .POST => try books.postDraft(),
538 .DRAFT => {
539 const block = books.draft orelse return books.fail("no open entry", .{});
540 var totals: Totals = .{};
541 for (block.lines.items) |line| if (line.debit) {
542 totals.debit += line.amount;
543 } else {
544 totals.credit += line.amount;
545 };
546 try books.out.print("DRAFT {s} {s} lines {d} debit {f} credit {f} difference {f}\n", .{ block.kind(), block.key, block.lines.items.len, money(totals.debit), money(totals.credit), money(totals.difference()) });
547 },
548 .REVERSE => {
549 const original = try books.entry(words[1]);
550 if (original.kind == .reversal or original.kind == .closing or original.kind == .reopen) return books.fail("cannot reverse {s}", .{original.id});
551 if (original.reversed_by != null) return books.fail("already reversed {s}", .{original.id});
552 const value = try books.date(words[2]);
553 if (value.number() < original.date.number()) return books.fail("reverse before entry {s} {f}", .{ original.id, original.date });
554 try books.checkClosed(value);
555 try books.activeLines(original.lines);
556 const posted = try books.reverse(original, value, .reversal, original.memo);
557 try books.out.print("REVERSE {s} entry {s} {f} amount {f}\n", .{ original.id, posted.id, value, money(posted.amount()) });
558 },
559 .RECUR => {
560 const template = books.templates.get(words[1]) orelse return books.fail("unknown template {s}", .{words[1]});
561 const value = try books.date(words[2]);
562 const count: usize = @intCast(integer(words[3]));
563 if (count == 0) return books.fail("invalid count", .{});
564 try books.checkClosed(value);
565 try books.activeLines(template.lines);
566 try books.out.print("RECUR {s} count {d}\n", .{ template.name, count });
567 for (0..count) |month| {
568 template.uses += 1;
569 const id = try std.fmt.allocPrint(books.gpa, "{s}-{d}", .{ template.name, template.uses });
570 const posted = try books.post(id, value.addMonths(month), .recurring, template.memo, template.lines);
571 try books.postText(posted, 2);
572 }
573 },
574 .DEPRECIATE => {
575 const value = try books.date(words[1]);
576 try books.checkClosed(value);
577 const asset = try books.typedAccount(words[2], .ASSET);
578 const contra = try books.typedAccount(words[3], .CONTRA_ASSET);
579 const expense = try books.typedAccount(words[4], .EXPENSE);
580 try books.active(&.{ contra, expense });
581 const cost = books.balance(asset, value.number());
582 const accumulated = books.balance(contra, value.number());
583 const amount = if (cost > 0) @min(@divTrunc(cost, integer(words[5])), cost - accumulated) else 0;
584 if (amount <= 0) return books.fail("nothing to depreciate {s}", .{words[2]});
585 books.depreciation_count += 1;
586 const id = try std.fmt.allocPrint(books.gpa, "DEP-{d}", .{books.depreciation_count});
587 _ = try books.post(id, value, .depreciation, "depreciation", &.{ .{ .debit = true, .account = expense, .amount = amount }, .{ .debit = false, .account = contra, .amount = amount } });
588 try books.out.print("DEPRECIATE {s} entry {s} {f} amount {f} accumulated {f} book {f}\n", .{ words[2], id, value, money(amount), money(accumulated + amount), money(cost - accumulated - amount) });
589 },
590 .INTEREST => {
591 const selected = try books.account(words[1]);
592 const kind = books.accounts[selected].kind;
593 if (kind != .ASSET and kind != .LIABILITY) return books.fail("not interest bearing {s}", .{words[1]});
594 const offset = try books.typedAccount(words[2], if (kind == .ASSET) .REVENUE else .EXPENSE);
595 const period = try books.range(words[3], words[4]);
596 try books.checkClosed(period.last);
597 try books.active(&.{ selected, offset });
598 const sum = books.dailySum(selected, period.first.number(), period.last.number());
599 const amount = @divTrunc(sum * integer(words[5]), 3650000);
600 if (amount <= 0) return books.fail("no interest {s}", .{words[1]});
601 const debit = if (kind == .ASSET) selected else offset;
602 const credit = if (kind == .ASSET) offset else selected;
603 books.interest_count += 1;
604 const id = try std.fmt.allocPrint(books.gpa, "INT-{d}", .{books.interest_count});
605 _ = try books.post(id, period.last, .interest, "interest", &.{ .{ .debit = true, .account = debit, .amount = amount }, .{ .debit = false, .account = credit, .amount = amount } });
606 const days = period.last.number() - period.first.number() + 1;
607 try books.out.print("INTEREST {s} entry {s} {f} days {d} average {f} amount {f}\n", .{ words[1], id, period.last, days, money(@divTrunc(sum, days)), money(amount) });
608 },
609 .WRITEOFF => {
610 const value = try books.date(words[1]);
611 try books.checkClosed(value);
612 const selected = try books.typedAccount(words[2], .ASSET);
613 const expense = try books.typedAccount(words[3], .EXPENSE);
614 try books.active(&.{ selected, expense });
615 const age = integer(words[4]);
616 var targets: std.ArrayList(struct { partner: []const u8, amount: i64 }) = .empty;
617 var total: i64 = 0;
618 for (try books.settlements(selected, value.number())) |settlement| {
619 var amount: i64 = 0;
620 for (settlement.open()) |item| if (value.number() - item.entry.date.number() > age) {
621 amount += item.remaining;
622 };
623 if (amount > 0) {
624 try targets.append(books.gpa, .{ .partner = settlement.partner, .amount = amount });
625 total += amount;
626 }
627 }
628 if (total == 0) return books.fail("nothing to write off {s}", .{words[2]});
629 var lines: std.ArrayList(Line) = .empty;
630 try lines.append(books.gpa, .{ .debit = true, .account = expense, .amount = total });
631 for (targets.items) |target| try lines.append(books.gpa, .{ .debit = false, .account = selected, .amount = target.amount, .partner = target.partner });
632 books.writeoff_count += 1;
633 const id = try std.fmt.allocPrint(books.gpa, "WO-{d}", .{books.writeoff_count});
634 _ = try books.post(id, value, .writeoff, "writeoff", lines.items);
635 try books.out.print("WRITEOFF {s} entry {s} {f} partners {d} amount {f}\n", .{ words[2], id, value, targets.items.len, money(total) });
636 for (targets.items) |target| try books.out.print(" {s} {f}\n", .{ target.partner, money(target.amount) });
637 },
638 .CLOSE => try books.close(try books.date(words[1]), words[2]),
639 .REOPEN => {
640 if (books.closes.items.len == 0) return books.fail("nothing to reopen", .{});
641 const record = books.closes.items[books.closes.items.len - 1];
642 var posted: ?*Entry = null;
643 if (record.entry) |original| {
644 try books.activeLines(original.lines);
645 posted = try books.reverse(original, record.date, .reopen, "reopen");
646 }
647 books.closes.items.len -= 1;
648 try books.out.print("REOPEN {f} ", .{record.date});
649 if (posted) |created| try books.out.print("entry {s}", .{created.id}) else try books.out.writeAll("no entry");
650 try books.out.print(" closed {s}\n", .{try books.closedText()});
651 },
652 .LEDGER => {
653 const selected = try books.account(words[1]);
654 const period = try books.range(words[2], words[3]);
655 try books.ledger(selected, period.first, period.last);
656 },
657 .ACTIVITY => {
658 const selected = try books.account(words[1]);
659 const year = integer(words[2]);
660 const found = books.accounts[selected];
661 var annual: Totals = .{};
662 try books.out.print("ACTIVITY {s} {s} {d}\n", .{ found.code, found.name, year });
663 var month: i64 = 1;
664 while (month <= 12) : (month += 1) {
665 const first: Date = .{ .year = year, .month = month, .day = 1 };
666 const last: Date = .{ .year = year, .month = month, .day = Date.monthDays(year, month) };
667 const totals = books.aggregate(first.number(), last.number(), false)[selected];
668 annual.debit += totals.debit;
669 annual.credit += totals.credit;
670 try books.out.print(" {d:0>4}-{d:0>2} debit {f} credit {f} balance {f}\n", .{ year, month, money(totals.debit), money(totals.credit), money(books.balance(selected, last.number())) });
671 }
672 try books.out.print(" total debit {f} credit {f}\n", .{ money(annual.debit), money(annual.credit) });
673 },
674 .TRIAL => {
675 const value = try books.date(words[1]);
676 const totals = books.aggregate(0, value.number(), false);
677 const order = books.orderedAccounts();
678 var debit: i64 = 0;
679 var credit: i64 = 0;
680 try books.out.print("TRIAL {f}\n", .{value});
681 for (order.slice()) |i| {
682 const difference = totals[i].difference();
683 if (difference == 0) continue;
684 if (difference > 0) debit += difference else credit -= difference;
685 try books.out.print(" {s} {s} {s} {f}\n", .{ books.accounts[i].code, books.accounts[i].name, if (difference > 0) "debit" else "credit", money(magnitude(difference)) });
686 }
687 try books.out.print(" total debit {f} credit {f}\n", .{ money(debit), money(credit) });
688 },
689 .INCOME => {
690 const period = try books.range(words[1], words[2]);
691 const totals = books.aggregate(period.first.number(), period.last.number(), true);
692 try books.out.print("INCOME {f} {f}\n", .{ period.first, period.last });
693 const revenue = try books.section(.REVENUE, &totals, null);
694 const expense = try books.section(.EXPENSE, &totals, null);
695 try books.out.print(" NET INCOME {f}\n", .{money(revenue - expense)});
696 },
697 .BALANCE => {
698 const value = try books.date(words[1]);
699 const totals = books.aggregate(0, value.number(), false);
700 var earnings: i64 = 0;
701 for (books.accounts[0..books.account_count], 0..) |found, i|
702 if (found.kind == .REVENUE or found.kind == .EXPENSE) {
703 earnings -= totals[i].difference();
704 };
705 try books.out.print("BALANCE {f}\n", .{value});
706 const assets = try books.section(.ASSETS, &totals, null);
707 const liabilities = try books.section(.LIABILITIES, &totals, null);
708 const equity = try books.section(.EQUITY, &totals, earnings);
709 if (assets == liabilities + equity) try books.out.writeAll(" check ok\n") else try books.out.print(" check mismatch {f}\n", .{money(assets - liabilities - equity)});
710 },
711 .ROLLUP => {
712 const value = try books.date(words[2]);
713 const order = books.orderedAccounts();
714 var targets: AccountOrder = .{};
715 for (order.slice()) |i| if (prefix(words[1], books.accounts[i].code)) {
716 targets.append(i);
717 };
718 try books.out.print("ROLLUP {s} {f}\n", .{ words[1], value });
719 if (targets.len == 0) {
720 try books.out.writeAll(" (none)\n");
721 return;
722 }
723 const totals = books.aggregate(0, value.number(), false);
724 var debit: i64 = 0;
725 var credit: i64 = 0;
726 for (targets.slice()) |i| {
727 const found = books.accounts[i];
728 var depth: usize = 0;
729 var subtotal: i64 = 0;
730 for (targets.slice()) |other| {
731 if (i != other and prefix(books.accounts[other].code, found.code)) depth += 1;
732 if (prefix(found.code, books.accounts[other].code)) subtotal += totals[other].difference();
733 }
734 debit += totals[i].debit;
735 credit += totals[i].credit;
736 try books.out.splatByteAll(' ', 2 * (1 + depth));
737 try books.out.print("{s} {s} own {f} total {f}\n", .{ found.code, found.name, money(totals[i].difference() * found.kind.sign()), money(subtotal * found.kind.sign()) });
738 }
739 try books.out.print(" postings debit {f} credit {f}\n", .{ money(debit), money(credit) });
740 },
741 .AGING, .ITEMS => {
742 const selected = try books.account(words[1]);
743 const value = try books.date(words[words.len - 1]);
744 const found = books.accounts[selected];
745 const settlements_list = try books.settlements(selected, value.number());
746 if (command == .ITEMS) {
747 try books.out.print("ITEMS {s} {s} {s} {f}\n", .{ found.code, found.name, words[2], value });
748 var settlement: Settlement = .{ .partner = words[2] };
749 for (settlements_list) |candidate| if (equal(candidate.partner, words[2])) {
750 settlement = candidate;
751 break;
752 };
753 var remaining: i64 = 0;
754 for (settlement.open()) |item| {
755 remaining += item.remaining;
756 try books.out.print(" {f} {s} amount {f} open {f} age {d}\n", .{ item.entry.date, item.entry.id, money(item.original), money(item.remaining), value.number() - item.entry.date.number() });
757 }
758 try books.out.print(" unapplied {f} balance {f}\n", .{ money(settlement.unapplied), money(remaining - settlement.unapplied) });
759 } else {
760 try books.out.print("AGING {s} {s} {f}\n", .{ found.code, found.name, value });
761 var total: [5]i64 = @splat(0);
762 for (settlements_list) |settlement| {
763 if (settlement.open().len == 0 and settlement.unapplied == 0) continue;
764 var values: [5]i64 = @splat(0);
765 for (settlement.open()) |item| {
766 const age = value.number() - item.entry.date.number();
767 const bucket: usize = if (age <= 30) 0 else if (age <= 60) 1 else if (age <= 90) 2 else 3;
768 values[bucket] += item.remaining;
769 }
770 values[4] = settlement.unapplied;
771 for (0..5) |i| total[i] += values[i];
772 try books.agingLine(settlement.partner, values);
773 }
774 try books.agingLine("total", total);
775 }
776 },
777 .ENTRY => {
778 const posted = try books.entry(words[1]);
779 try books.out.print("ENTRY {s} {f} {s} {s}\n", .{ posted.id, posted.date, @tagName(posted.kind), posted.memo });
780 for (posted.lines) |line| {
781 const found = books.accounts[line.account];
782 try books.out.print(" {s} {s} {s} {f}", .{ line.side(), found.code, found.name, money(line.amount) });
783 if (line.partner) |partner| try books.out.print(" {s}", .{partner});
784 try books.out.writeAll("\n");
785 }
786 try books.out.print(" total {f}\n", .{money(posted.amount())});
787 if (posted.reverses) |id| try books.out.print(" reverses {s}\n", .{id});
788 if (posted.reversed_by) |id| try books.out.print(" reversed by {s}\n", .{id});
789 },
790 .DAYBOOK => {
791 const period = try books.range(words[1], words[2]);
792 try books.out.print("DAYBOOK {f} {f}\n", .{ period.first, period.last });
793 var count: usize = 0;
794 var amount: i64 = 0;
795 for (try books.orderedEntries()) |posted| {
796 if (posted.date.number() < period.first.number() or posted.date.number() > period.last.number()) continue;
797 count += 1;
798 amount += posted.amount();
799 try books.out.print(" {f} {s} {s} {f} {s}\n", .{ posted.date, posted.id, @tagName(posted.kind), money(posted.amount()), posted.memo });
800 }
801 try books.out.print(" total entries {d} amount {f}\n", .{ count, money(amount) });
802 },
803 .ACCOUNTS => {
804 const totals = books.aggregate(0, last_day, false);
805 const order = books.orderedAccounts();
806 try books.out.writeAll("ACCOUNTS\n");
807 for (order.slice()) |i| {
808 const found = books.accounts[i];
809 try books.out.print(" {s} {s} {s} parent {s} balance {f} {s}\n", .{ found.code, @tagName(found.kind), found.name, books.parentCode(i), money(totals[i].difference() * found.kind.sign()), if (found.active) "active" else "inactive" });
810 }
811 if (order.len == 0) try books.out.writeAll(" (none)\n");
812 },
813 .TEMPLATES => {
814 var selected: std.ArrayList(*Template) = .empty;
815 var iterator = books.templates.valueIterator();
816 while (iterator.next()) |value| try selected.append(books.gpa, value.*);
817 std.mem.sort(*Template, selected.items, {}, templateLess);
818 try books.out.writeAll("TEMPLATES\n");
819 for (selected.items) |template| {
820 var totals: Totals = .{};
821 for (template.lines) |line| totals.add(line);
822 try books.out.print(" {s} {s} lines {d} amount {f} used {d}\n", .{ template.name, template.memo, template.lines.len, money(totals.debit), template.uses });
823 }
824 if (selected.items.len == 0) try books.out.writeAll(" (none)\n");
825 },
826 .STATUS => {
827 var inactive: usize = 0;
828 for (books.accounts[0..books.account_count]) |found| if (!found.active) {
829 inactive += 1;
830 };
831 try books.out.print("STATUS accounts {d} inactive {d} entries {d} templates {d} closed {s} open {s}\n", .{ books.account_count, inactive, books.entries.items.len, books.templates.count(), try books.closedText(), if (books.draft) |block| block.key else "none" });
832 },
833 }
834 }
835};

837fn readWords(gpa: Allocator, input: *std.Io.Reader) ![]const []const u8 {
838 var buffer: std.Io.Writer.Allocating = .init(gpa);
839 defer buffer.deinit();
840 _ = try input.streamDelimiterEnding(&buffer.writer, '\n');
841 if (input.seek < input.end) input.toss(1) else if (buffer.written().len == 0) return error.EndOfStream;
842 const line = try buffer.toOwnedSlice();
843 var words: std.ArrayList([]const u8) = .empty;
844 var parts = std.mem.tokenizeAny(u8, line, " \r");
845 while (parts.next()) |part| try words.append(gpa, part);
846 return try words.toOwnedSlice(gpa);
847}

849pub fn main(init: std.process.Init) !void {
850 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
851 defer arena.deinit();
852 const gpa = arena.allocator();
853 var input_buffer: [4096]u8 = undefined;
854 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
855 var output_buffer: [4096]u8 = undefined;
856 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
857 const count: usize = @intCast(integer((try readWords(gpa, &input.interface))[0]));
858 var books: Books = .{ .gpa = gpa, .out = &output.interface };
859 for (0..count) |_| {
860 const words = try readWords(gpa, &input.interface);
861 var response: std.Io.Writer.Allocating = .init(gpa);
862 defer response.deinit();
863 books.out = &response.writer;
864 books.execute(words) catch |err| {
865 if (err != error.BadCommand) return err;
866 try output.interface.print("ERROR {s}\n", .{books.error_message});
867 continue;
868 };
869 try output.interface.writeAll(response.written());
870 }
871 try output.interface.flush();
872}
