34func (kind AccountType) class() AccountType {
35 if kind == contraAsset { return asset }
36 return kind
37}

39func (kind AccountType) sign() Cents {
40 if kind == asset || kind == expense { return 1 }

46func integer(text string) int {
47 value, _ := strconv.Atoi(text)

51func parseMoney(text string) Cents {
52 parts := strings.Split(text, ".")
53 return Cents(integer(parts[0]))*100 + Cents(integer(parts[1]))
54}

56func money(value Cents) string {
57 sign := ""
58 if value < 0 { sign, value = "-", -value }
59 return fmt.Sprintf("%s%d.%02d", sign, value/100, value%100)
60}

62func leap(year int) bool { return year%4 == 0 && (year%100 != 0 || year%400 == 0) }
63
64func monthLength(year, month int) int {
65 lengths := [...]int{31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31}
66 if month == 2 && leap(year) { return 29 }
67 return lengths[month-1]
68}
69
70func yearLength(year int) int {
71 if leap(year) { return 366 }
72 return 365
73}
74
75func dayNumber(year, month, day int) int {
76 number := day - 1
77 for y := 2000; y < year; y++ { number += yearLength(y) }
78 for m := 1; m < month; m++ { number += monthLength(year, m) }
79 return number
80}
81
82func calendar(number int) (int, int, int) {
83 year, month := 2000, 1
84 for number >= yearLength(year) { number -= yearLength(year); year++ }
85 for number >= monthLength(year, month) { number -= monthLength(year, month); month++ }
86 return year, month, number + 1
87}
88
89func dateText(number int) string {
90 year, month, day := calendar(number)
91 return fmt.Sprintf("%04d-%02d-%02d", year, month, day)
92}
93
94func parseDate(text string) (int, error) {
95 year, month, day := integer(text[:4]), integer(text[5:7]), integer(text[8:])
96 if year < 2000 || year > 2099 || month < 1 || month > 12 || day < 1 || day > monthLength(year, month) {
97 return 0, fmt.Errorf("invalid date %s", text)
98 }
99 return dayNumber(year, month, day), nil
100}
101
102func dateRange(first, last string) (int, int, error) {
103 start, err := parseDate(first)
104 if err != nil { return 0, 0, err }
105 end, err := parseDate(last)
106 if err != nil { return 0, 0, err }
107 if start > end { return 0, 0, fmt.Errorf("invalid range %s %s", first, last) }
108 return start, end, nil
109}
110
111func monthsAfter(number, offset int) int {
112 year, month, day := calendar(number)
113 total := month - 1 + offset
114 year, month = year + total/12, total%12 + 1
115 return dayNumber(year, month, min(day, monthLength(year, month)))
116}

118func sortedKeys[V any](values map[string]V) []string {
119 keys := make([]string, 0, len(values))
120 for key := range values { keys = append(keys, key) }
121 sort.Strings(keys)
122 return keys
123}

135func (totals *Totals) add(debit bool, amount Cents) {

140 totals.debit += other.debit

167func newLines(items []Line) *Lines {
168 lines := &Lines{items: items, accounts: make(map[*Account]Totals)}
169 for _, line := range items {
170 totals := lines.accounts[line.account]
171 totals.add(line.debit, line.amount)
172 lines.accounts[line.account] = totals
173 if line.debit { lines.amount += line.amount }
174 }
175 return lines

297func (book *Book) ordered() []*Entry {
298 entries := append([]*Entry(nil), book.entries...)
299 sort.Slice(entries, func(i, j int) bool {
300 if entries[i].day != entries[j].day { return entries[i].day < entries[j].day }
301 return entries[i].sequence < entries[j].sequence
302 })
303 return entries
304}

306func (book *Book) totals(first, last int, operating bool) map[*Account]Totals {
307 totals := make(map[*Account]Totals)
308 for _, entry := range book.entries {
309 if entry.day < first || entry.day > last || operating && (entry.category == closing || entry.category == reopen) { continue }
310 for account, amount := range entry.lines.accounts {
311 current := totals[account]
312 current.merge(amount)
313 totals[account] = current
314 }
315 }
316 return totals
317}

319func (book *Book) accountTotals(account *Account, first, last int) Totals {
320 var totals Totals
321 for _, entry := range book.entries {
322 if first <= entry.day && entry.day <= last { totals.merge(entry.lines.accounts[account]) }
323 }
324 return totals
325}

327func (book *Book) balance(account *Account, day int) Cents {
328 return account.kind.sign() * book.accountTotals(account, 0, day).difference()
329}

331func (book *Book) dailySum(account *Account, first, last int) Cents {
332 changes := make([]Cents, last-first+1)
333 var running, sum Cents
334 for _, entry := range book.entries {
335 if entry.day > last { continue }
336 value := account.kind.sign() * entry.lines.accounts[account].difference()
337 if entry.day < first { running += value } else { changes[entry.day-first] += value }
338 }
339 for _, change := range changes { running += change; sum += running }
340 return sum
341}

350func (book *Book) reverse(original *Entry, day int, category Category, memo string) *Entry {
351 items := append([]Line(nil), original.lines.items...)
352 for i := range items { items[i].debit = !items[i].debit }
353 entry := book.post(original.id+"-R", day, category, memo, newLines(items))
354 original.reversedBy, entry.reverses = entry, original
355 return entry

358func (book *Book) settlements(account *Account, day int) map[string]*Settlement {
359 result := make(map[string]*Settlement)
360 for _, entry := range book.ordered() {
361 if entry.day > day { continue }
362 for _, line := range entry.lines.items {
363 if line.account != account || line.partner == nil { continue }
364 partner := *line.partner
365 settled := result[partner]
366 if settled == nil { settled = &Settlement{}; result[partner] = settled }
367 value := account.kind.sign() * line.difference()
368 if value > 0 {
369 applied := min(value, settled.unapplied)
370 settled.unapplied -= applied
371 if value > applied { settled.items = append(settled.items, OpenItem{entry, value, value-applied}) }
372 } else {
373 payment := -value
374 for payment > 0 && len(settled.items) > 0 {
375 item := &settled.items[0]
376 applied := min(payment, item.remaining)
377 payment -= applied
378 item.remaining -= applied
379 if item.remaining == 0 { settled.items = settled.items[1:] }
380 }
381 settled.unapplied += payment
382 if len(settled.items) == 0 { settled.items = nil }
383 }
384 }
385 }
386 return result
387}

389func (book *Book) postDraft() error {
390 if book.draft == nil { return fmt.Errorf("no open entry") }
391 draft := book.draft
392 book.draft = nil
393 day := 0
394 if draft.journal {
395 if _, ok := book.entryIDs[draft.key]; ok { return fmt.Errorf("duplicate entry %s", draft.key) }
396 var err error
397 day, err = parseDate(draft.date)
398 if err != nil { return err }
399 if err := book.checkOpen(day); err != nil { return err }
400 } else if _, ok := book.templates[draft.key]; ok { return fmt.Errorf("duplicate template %s", draft.key) }
401 if len(draft.lines) < 2 { return fmt.Errorf("too few lines %s", draft.key) }
402 items := make([]Line, 0, len(draft.lines))
403 for _, line := range draft.lines {
404 account, err := book.account(line.code)
405 if err != nil { return err }
406 if draft.journal { if err := checkActive(account); err != nil { return err } }
407 if line.amount == 0 { return fmt.Errorf("zero amount %s", line.code) }
408 items = append(items, Line{line.debit, account, line.amount, line.partner})
409 }
410 totals := draft.totals()
411 if totals.debit != totals.credit {
412 return fmt.Errorf("unbalanced %s debit %s credit %s", draft.key, money(totals.debit), money(totals.credit))
413 }
414 lines := newLines(items)
415 if draft.journal {
416 book.post(draft.key, day, journal, draft.memo, lines)
417 book.emit("POST %s %s lines %d amount %s", draft.key, dateText(day), len(items), money(lines.amount))
418 } else {
419 book.templates[draft.key] = &Template{memo: draft.memo, lines: lines}
420 book.emit("TEMPLATE %s lines %d amount %s", draft.key, len(items), money(lines.amount))
421 }
422 return nil
423}

427func (book *Book) section(heading string, kinds []AccountType, sign Cents, totals map[*Account]Totals, earnings *Cents) Cents {
428 groups := make(map[string][]Displayed)
429 var total Cents
430 if earnings != nil { total = *earnings }
431 for _, code := range sortedKeys(book.accounts) {
432 account := book.accounts[code]
433 selected := false
434 for _, kind := range kinds { selected = selected || account.kind == kind }
435 if !selected { continue }
436 value := sign * totals[account].difference()
437 root := book.top(code)
438 groups[root] = append(groups[root], Displayed{account, value})
439 total += value
440 }
441 book.emit(" %s %s", heading, money(total))
442 for _, root := range sortedKeys(groups) {
443 var subtotal Cents
444 nonzero := false
445 for _, item := range groups[root] { subtotal += item.amount; nonzero = nonzero || item.amount != 0 }
446 if !nonzero { continue }
447 book.emit(" %s %s %s", root, book.accounts[root].name, money(subtotal))
448 for _, item := range groups[root] {
449 if item.amount != 0 { book.emit(" %s %s %s", item.account.code, item.account.name, money(item.amount)) }
450 }
451 }
452 if earnings != nil { book.emit(" current earnings %s", money(*earnings)) }
453 return total
454}

462func (book *Book) execute(words []string) error {
463 switch words[0] {
464 case "ACCOUNT":
465 code, kind := words[1], AccountType(words[2])
466 if _, ok := book.accounts[code]; ok { return fmt.Errorf("duplicate account %s", code) }
467 for _, other := range sortedKeys(book.accounts) {
468 if (strings.HasPrefix(code, other) || strings.HasPrefix(other, code)) && kind.class() != book.accounts[other].kind.class() {
469 return fmt.Errorf("class mismatch %s %s", code, other)
470 }
471 }
472 book.accounts[code] = &Account{code, words[3], kind, true}
473 book.emit("ACCOUNT %s %s %s parent %s", code, kind, words[3], book.parent(code))
474 case "DEACTIVATE":
475 account, err := book.account(words[1])
476 if err != nil { return err }
477 if !account.active { return fmt.Errorf("already inactive %s", account.code) }
478 balance := book.balance(account, lastDay)
479 if balance != 0 { return fmt.Errorf("nonzero balance %s %s", account.code, money(balance)) }
480 account.active = false
481 book.emit("DEACTIVATE %s %s", account.code, account.name)
482 case "JOURNAL", "TEMPLATE":
483 if book.draft != nil { book.emit("DISCARD %s lines %d", book.draft.key, len(book.draft.lines)) }
484 draft := &Draft{journal: words[0] == "JOURNAL", key: words[1]}
485 if draft.journal { draft.date, draft.memo = words[2], words[3] } else { draft.memo = words[2] }
486 book.draft = draft
487 case "DR", "CR":
488 if book.draft == nil { return fmt.Errorf("no open entry") }
489 var partner *string
490 if len(words) == 4 { name := words[3]; partner = &name }
491 book.draft.lines = append(book.draft.lines, DraftLine{words[0] == "DR", words[1], parseMoney(words[2]), partner})
492 case "POST":
493 return book.postDraft()
494 case "DRAFT":
495 if book.draft == nil { return fmt.Errorf("no open entry") }
496 kind := "template"
497 if book.draft.journal { kind = "journal" }
498 totals := book.draft.totals()
499 book.emit("DRAFT %s %s lines %d debit %s credit %s difference %s", kind, book.draft.key, len(book.draft.lines),
500 money(totals.debit), money(totals.credit), money(totals.difference()))
501 case "REVERSE":
502 original, ok := book.entryIDs[words[1]]
503 if !ok { return fmt.Errorf("unknown entry %s", words[1]) }
504 if original.category == reversal || original.category == closing || original.category == reopen { return fmt.Errorf("cannot reverse %s", original.id) }
505 if original.reversedBy != nil { return fmt.Errorf("already reversed %s", original.id) }
506 day, err := parseDate(words[2])
507 if err != nil { return err }
508 if day < original.day { return fmt.Errorf("reverse before entry %s %s", original.id, dateText(original.day)) }
509 if err := book.checkOpen(day); err != nil { return err }
510 if err := activeLines(original.lines); err != nil { return err }
511 entry := book.reverse(original, day, reversal, original.memo)
512 book.emit("REVERSE %s entry %s %s amount %s", original.id, entry.id, dateText(day), money(entry.lines.amount))
513 case "RECUR":
514 model, ok := book.templates[words[1]]
515 if !ok { return fmt.Errorf("unknown template %s", words[1]) }
516 day, err := parseDate(words[2])
517 if err != nil { return err }
518 count := integer(words[3])
519 if count == 0 { return fmt.Errorf("invalid count") }
520 if err := book.checkOpen(day); err != nil { return err }
521 if err := activeLines(model.lines); err != nil { return err }
522 book.emit("RECUR %s count %d", words[1], count)
523 for offset := 0; offset < count; offset++ {
524 model.uses++
525 entry := book.post(words[1]+"-"+strconv.Itoa(model.uses), monthsAfter(day, offset), recurring, model.memo, model.lines)
526 book.emit(" POST %s %s lines %d amount %s", entry.id, dateText(entry.day), len(entry.lines.items), money(entry.lines.amount))
527 }
528 case "DEPRECIATE":
529 day, err := parseDate(words[1])
530 if err != nil { return err }
531 if err := book.checkOpen(day); err != nil { return err }
532 acquisition, err := book.requireType(words[2], asset)
533 if err != nil { return err }
534 contra, err := book.requireType(words[3], contraAsset)
535 if err != nil { return err }
536 costAccount, err := book.requireType(words[4], expense)
537 if err != nil { return err }
538 if err := checkActive(contra, costAccount); err != nil { return err }
539 cost, accumulated := book.balance(acquisition, day), book.balance(contra, day)
540 var amount Cents
541 if cost > 0 { amount = min(cost/Cents(integer(words[5])), cost-accumulated) }
542 if amount <= 0 { return fmt.Errorf("nothing to depreciate %s", acquisition.code) }
543 book.depreciationCount++
544 entry := book.post("DEP-"+strconv.Itoa(book.depreciationCount), day, depreciation, "depreciation",
545 newLines([]Line{{true, costAccount, amount, nil}, {false, contra, amount, nil}}))
546 book.emit("DEPRECIATE %s entry %s %s amount %s accumulated %s book %s", acquisition.code, entry.id, dateText(day),
547 money(amount), money(accumulated+amount), money(cost-accumulated-amount))
548 case "INTEREST":
549 account, err := book.account(words[1])
550 if err != nil { return err }
551 if account.kind != asset && account.kind != liability { return fmt.Errorf("not interest bearing %s", account.code) }
552 offsetType := revenue
553 if account.kind == liability { offsetType = expense }
554 offset, err := book.requireType(words[2], offsetType)
555 if err != nil { return err }
556 first, last, err := dateRange(words[3], words[4])
557 if err != nil { return err }
558 if err := book.checkOpen(last); err != nil { return err }
559 if err := checkActive(account, offset); err != nil { return err }
560 sum := book.dailySum(account, first, last)
561 amount := sum * Cents(integer(words[5])) / 3650000
562 if amount <= 0 { return fmt.Errorf("no interest %s", account.code) }
563 debit, credit := account, offset
564 if account.kind == liability { debit, credit = offset, account }
565 book.interestCount++
566 entry := book.post("INT-"+strconv.Itoa(book.interestCount), last, interest, "interest",
567 newLines([]Line{{true, debit, amount, nil}, {false, credit, amount, nil}}))
568 days := last-first+1
569 book.emit("INTEREST %s entry %s %s days %d average %s amount %s", account.code, entry.id, dateText(last), days,
570 money(sum/Cents(days)), money(amount))
571 case "WRITEOFF":
572 day, err := parseDate(words[1])
573 if err != nil { return err }
574 if err := book.checkOpen(day); err != nil { return err }
575 account, err := book.requireType(words[2], asset)
576 if err != nil { return err }
577 cost, err := book.requireType(words[3], expense)
578 if err != nil { return err }
579 if err := checkActive(account, cost); err != nil { return err }
580 age := integer(words[4])
581 settlements := book.settlements(account, day)
582 targets := make(map[string]Cents)
583 var total Cents
584 for _, partner := range sortedKeys(settlements) {
585 var amount Cents
586 for _, item := range settlements[partner].items { if day-item.entry.day > age { amount += item.remaining } }
587 if amount > 0 { targets[partner] = amount; total += amount }
588 }
589 if total == 0 { return fmt.Errorf("nothing to write off %s", account.code) }
590 items := []Line{{true, cost, total, nil}}
591 for _, partner := range sortedKeys(targets) {
592 name := partner
593 items = append(items, Line{false, account, targets[partner], &name})
594 }
595 book.writeoffCount++
596 entry := book.post("WO-"+strconv.Itoa(book.writeoffCount), day, writeoff, "writeoff", newLines(items))
597 book.emit("WRITEOFF %s entry %s %s partners %d amount %s", account.code, entry.id, dateText(day), len(targets), money(total))
598 for _, partner := range sortedKeys(targets) { book.emit(" %s %s", partner, money(targets[partner])) }
599 case "CLOSE":
600 day, err := parseDate(words[1])
601 if err != nil { return err }
602 if err := book.checkOpen(day); err != nil { return err }
603 capital, err := book.requireType(words[2], equity)
604 if err != nil { return err }
605 if err := checkActive(capital); err != nil { return err }
606 totals := book.totals(0, day, false)
607 var targets []*Account
608 for _, code := range sortedKeys(book.accounts) {
609 account := book.accounts[code]
610 if account.kind.temporary() && totals[account].difference() != 0 { targets = append(targets, account) }
611 }
612 if err := checkActive(targets...); err != nil { return err }
613 var items []Line
614 var income Cents
615 for _, account := range targets {
616 value := totals[account].difference()
617 amount := value
618 if amount < 0 { amount = -amount }
619 items = append(items, Line{value < 0, account, amount, nil})
620 income -= value
621 }
622 if income != 0 {
623 amount := income
624 if amount < 0 { amount = -amount }
625 items = append(items, Line{income < 0, capital, amount, nil})
626 }
627 book.closeCount++
628 var entry *Entry
629 if len(items) > 0 { entry = book.post("CLOSE-"+strconv.Itoa(book.closeCount), day, closing, "closing", newLines(items)) }
630 book.closes = append(book.closes, CloseRecord{day, entry})
631 if entry == nil { book.emit("CLOSE %s no entry", dateText(day)) } else {
632 book.emit("CLOSE %s entry %s lines %d net income %s", dateText(day), entry.id, len(items), money(income))
633 }
634 case "REOPEN":
635 if len(book.closes) == 0 { return fmt.Errorf("nothing to reopen") }
636 record := book.closes[len(book.closes)-1]
637 detail := "no entry"
638 if record.entry != nil {
639 if err := activeLines(record.entry.lines); err != nil { return err }
640 detail = "entry " + book.reverse(record.entry, record.day, reopen, "reopen").id
641 }
642 book.closes = book.closes[:len(book.closes)-1]
643 book.emit("REOPEN %s %s closed %s", dateText(record.day), detail, book.closedText())
644 case "LEDGER":
645 account, err := book.account(words[1])
646 if err != nil { return err }
647 first, last, err := dateRange(words[2], words[3])
648 if err != nil { return err }
649 running := book.balance(account, first-1)
650 var totals Totals
651 book.emit("LEDGER %s %s %s %s", account.code, account.name, words[2], words[3])
652 book.emit(" opening %s", money(running))
653 for _, entry := range book.ordered() {
654 if entry.day < first || entry.day > last { continue }
655 for _, line := range entry.lines.items {
656 if line.account != account { continue }
657 totals.add(line.debit, line.amount)
658 running += account.kind.sign()*line.difference()
659 book.emit(" %s %s %s %s balance %s %s", dateText(entry.day), entry.id, line.side(), money(line.amount), money(running), entry.memo)
660 }
661 }
662 days := last-first+1
663 book.emit(" ending %s debits %s credits %s days %d average %s", money(running), money(totals.debit), money(totals.credit),
664 days, money(book.dailySum(account, first, last)/Cents(days)))
665 case "ACTIVITY":
666 account, err := book.account(words[1])
667 if err != nil { return err }
668 year := integer(words[2])
669 var totals Totals
670 book.emit("ACTIVITY %s %s %d", account.code, account.name, year)
671 for month := 1; month <= 12; month++ {
672 first, last := dayNumber(year, month, 1), dayNumber(year, month, monthLength(year, month))
673 current := book.accountTotals(account, first, last)
674 totals.merge(current)
675 book.emit(" %04d-%02d debit %s credit %s balance %s", year, month, money(current.debit), money(current.credit), money(book.balance(account, last)))
676 }
677 book.emit(" total debit %s credit %s", money(totals.debit), money(totals.credit))
678 case "TRIAL":
679 day, err := parseDate(words[1])
680 if err != nil { return err }
681 totals := book.totals(0, day, false)
682 var grand Totals
683 book.emit("TRIAL %s", words[1])
684 for _, code := range sortedKeys(book.accounts) {
685 value := totals[book.accounts[code]].difference()
686 if value == 0 { continue }
687 side, amount := "debit", value
688 if value > 0 { grand.debit += value } else { side, amount = "credit", -value; grand.credit -= value }
689 book.emit(" %s %s %s %s", code, book.accounts[code].name, side, money(amount))
690 }
691 book.emit(" total debit %s credit %s", money(grand.debit), money(grand.credit))
692 case "INCOME":
693 first, last, err := dateRange(words[1], words[2])
694 if err != nil { return err }
695 totals := book.totals(first, last, true)
696 book.emit("INCOME %s %s", words[1], words[2])
697 sales := book.section("REVENUE", []AccountType{revenue}, -1, totals, nil)
698 cost := book.section("EXPENSE", []AccountType{expense}, 1, totals, nil)
699 book.emit(" NET INCOME %s", money(sales-cost))
700 case "BALANCE":
701 day, err := parseDate(words[1])
702 if err != nil { return err }
703 totals := book.totals(0, day, false)
704 var earnings Cents
705 for _, code := range sortedKeys(book.accounts) {
706 account := book.accounts[code]
707 if account.kind.temporary() { earnings -= totals[account].difference() }
708 }
709 book.emit("BALANCE %s", words[1])
710 assets := book.section("ASSETS", []AccountType{asset, contraAsset}, 1, totals, nil)
711 liabilities := book.section("LIABILITIES", []AccountType{liability}, -1, totals, nil)
712 capital := book.section("EQUITY", []AccountType{equity}, -1, totals, &earnings)
713 if assets == liabilities+capital { book.emit(" check ok") } else { book.emit(" check mismatch %s", money(assets-liabilities-capital)) }
714 case "ROLLUP":
715 day, err := parseDate(words[2])
716 if err != nil { return err }
717 var targets []string
718 for _, code := range sortedKeys(book.accounts) { if strings.HasPrefix(code, words[1]) { targets = append(targets, code) } }
719 book.emit("ROLLUP %s %s", words[1], words[2])
720 if len(targets) == 0 { book.emit(" (none)"); return nil }
721 totals := book.totals(0, day, false)
722 var postings Totals
723 for _, code := range targets {
724 account := book.accounts[code]
725 depth := 0
726 var subtotal Cents
727 for _, other := range targets {
728 if other != code && strings.HasPrefix(code, other) { depth++ }
729 if strings.HasPrefix(other, code) { subtotal += totals[book.accounts[other]].difference() }
730 }
731 own := totals[account]
732 book.emit("%s%s %s own %s total %s", strings.Repeat(" ", 2*(1+depth)), code, account.name,
733 money(account.kind.sign()*own.difference()), money(account.kind.sign()*subtotal))
734 postings.merge(own)
735 }
736 book.emit(" postings debit %s credit %s", money(postings.debit), money(postings.credit))
737 case "AGING", "ITEMS":
738 account, err := book.account(words[1])
739 if err != nil { return err }
740 aging := words[0] == "AGING"
741 dateIndex := 3
742 if aging { dateIndex = 2 }
743 day, err := parseDate(words[dateIndex])
744 if err != nil { return err }
745 settlements := book.settlements(account, day)
746 if aging {
747 book.emit("AGING %s %s %s", account.code, account.name, words[2])
748 var totals [5]Cents
749 for _, partner := range sortedKeys(settlements) {
750 settled := settlements[partner]
751 if len(settled.items) == 0 && settled.unapplied == 0 { continue }
752 values := [5]Cents{0, 0, 0, 0, settled.unapplied}
753 for _, item := range settled.items {
754 age, bucket := day-item.entry.day, 3
755 if age <= 30 { bucket = 0 } else if age <= 60 { bucket = 1 } else if age <= 90 { bucket = 2 }
756 values[bucket] += item.remaining
757 }
758 book.agingRow(partner, values)
759 for i, amount := range values { totals[i] += amount }
760 }
761 book.agingRow("total", totals)
762 } else {
763 settled := settlements[words[2]]
764 if settled == nil { settled = &Settlement{} }
765 book.emit("ITEMS %s %s %s %s", account.code, account.name, words[2], words[3])
766 var remaining Cents
767 for _, item := range settled.items {
768 book.emit(" %s %s amount %s open %s age %d", dateText(item.entry.day), item.entry.id,
769 money(item.original), money(item.remaining), day-item.entry.day)
770 remaining += item.remaining
771 }
772 book.emit(" unapplied %s balance %s", money(settled.unapplied), money(remaining-settled.unapplied))
773 }
774 case "ENTRY":
775 entry, ok := book.entryIDs[words[1]]
776 if !ok { return fmt.Errorf("unknown entry %s", words[1]) }
777 book.emit("ENTRY %s %s %s %s", entry.id, dateText(entry.day), entry.category, entry.memo)
778 for _, line := range entry.lines.items {
779 partner := ""
780 if line.partner != nil { partner = " " + *line.partner }
781 book.emit(" %s %s %s %s%s", line.side(), line.account.code, line.account.name, money(line.amount), partner)
782 }
783 book.emit(" total %s", money(entry.lines.amount))
784 if entry.reverses != nil { book.emit(" reverses %s", entry.reverses.id) }
785 if entry.reversedBy != nil { book.emit(" reversed by %s", entry.reversedBy.id) }
786 case "DAYBOOK":
787 first, last, err := dateRange(words[1], words[2])
788 if err != nil { return err }
789 count := 0
790 var total Cents
791 book.emit("DAYBOOK %s %s", words[1], words[2])
792 for _, entry := range book.ordered() {
793 if entry.day < first || entry.day > last { continue }
794 count++
795 total += entry.lines.amount
796 book.emit(" %s %s %s %s %s", dateText(entry.day), entry.id, entry.category, money(entry.lines.amount), entry.memo)
797 }
798 book.emit(" total entries %d amount %s", count, money(total))
799 case "ACCOUNTS":
800 totals := book.totals(0, lastDay, false)
801 book.emit("ACCOUNTS")
802 for _, code := range sortedKeys(book.accounts) {
803 account := book.accounts[code]
804 status := "inactive"
805 if account.active { status = "active" }
806 book.emit(" %s %s %s parent %s balance %s %s", code, account.kind, account.name, book.parent(code),
807 money(account.kind.sign()*totals[account].difference()), status)
808 }
809 if len(book.accounts) == 0 { book.emit(" (none)") }
810 case "TEMPLATES":
811 book.emit("TEMPLATES")
812 for _, name := range sortedKeys(book.templates) {
813 model := book.templates[name]
814 book.emit(" %s %s lines %d amount %s used %d", name, model.memo, len(model.lines.items), money(model.lines.amount), model.uses)
815 }
816 if len(book.templates) == 0 { book.emit(" (none)") }
817 case "STATUS":
818 inactive := 0
819 for _, account := range book.accounts { if !account.active { inactive++ } }
820 open := "none"
821 if book.draft != nil { open = book.draft.key }
822 book.emit("STATUS accounts %d inactive %d entries %d templates %d closed %s open %s", len(book.accounts), inactive,
823 len(book.entries), len(book.templates), book.closedText(), open)
824 }
825 return nil
826}
