5type Money = i64;
6type Day = i32;

8fn emit(output: &mut impl Write, text: impl AsRef<str>) {
9 writeln!(output, "{}", text.as_ref()).unwrap();
10}

12fn money(cents: Money) -> String {
13 let value = cents.abs();
14 format!("{}{}.{:02}", if cents < 0 { "-" } else { "" }, value / 100, value % 100)
15}
16
17fn parse_money(text: &str) -> Money {
18 let (whole, fraction) = text.split_once('.').unwrap();
19 whole.parse::<Money>().unwrap() * 100 + fraction.parse::<Money>().unwrap()
20}
21
22fn leap(year: i32) -> bool { year % 4 == 0 && (year % 100 != 0 || year % 400 == 0) }
23
24fn month_days(year: i32, month: usize) -> i32 {
25 let days = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
26 days[month - 1] + i32::from(month == 2 && leap(year))
27}
28
29fn day_number(year: i32, month: usize, day: i32) -> Day {
30 (2000..year).map(|y| 365 + i32::from(leap(y))).sum::<i32>()
31 + (1..month).map(|m| month_days(year, m)).sum::<i32>() + day - 1
32}
33
34fn date_parts(mut day: Day) -> (i32, usize, i32) {
35 let mut year = 2000;
36 while day >= 365 + i32::from(leap(year)) {
37 day -= 365 + i32::from(leap(year));
38 year += 1;
39 }
40 let mut month = 1;
41 while day >= month_days(year, month) {
42 day -= month_days(year, month);
43 month += 1;
44 }
45 (year, month, day + 1)
46}
47
48fn date_text(day: Day) -> String {
49 let (year, month, day) = date_parts(day);
50 format!("{year:04}-{month:02}-{day:02}")
51}
52
53fn parse_date(text: &str) -> Result<Day, String> {
54 let year: i32 = text[..4].parse().unwrap();
55 let month: usize = text[5..7].parse().unwrap();
56 let day: i32 = text[8..].parse().unwrap();
57 if !(2000..=2099).contains(&year) || !(1..=12).contains(&month)
58 || day < 1 || day > month_days(year, month) {
59 return Err(format!("invalid date {text}"));
60 }
61 Ok(day_number(year, month, day))
62}
63
64fn date_range(first: &str, last: &str) -> Result<(Day, Day), String> {
65 let start = parse_date(first)?;
66 let end = parse_date(last)?;
67 if start > end { return Err(format!("invalid range {first} {last}")); }
68 Ok((start, end))
69}
70
71fn months_after(date: Day, months: usize) -> Day {
72 let (mut year, month, day) = date_parts(date);
73 let total = month - 1 + months;
74 year += (total / 12) as i32;
75 let month = total % 12 + 1;
76 day_number(year, month, day.min(month_days(year, month)))
77}

79#[derive(Clone, Copy, PartialEq, Eq)]
80enum AccountType { Asset, Liability, Equity, Revenue, Expense, ContraAsset }
81
82impl AccountType {
83 fn parse(text: &str) -> Self {
84 match text {
85 "ASSET" => Self::Asset,
86 "LIABILITY" => Self::Liability,
87 "EQUITY" => Self::Equity,
88 "REVENUE" => Self::Revenue,
89 "EXPENSE" => Self::Expense,
90 "CONTRA_ASSET" => Self::ContraAsset,
91 _ => unreachable!(),
92 }
93 }
94
95 fn text(self) -> &'static str {
96 match self {
97 Self::Asset => "ASSET", Self::Liability => "LIABILITY", Self::Equity => "EQUITY",
98 Self::Revenue => "REVENUE", Self::Expense => "EXPENSE", Self::ContraAsset => "CONTRA_ASSET",
99 }
100 }
101
102 fn class(self) -> Self { if self == Self::ContraAsset { Self::Asset } else { self } }
103 fn sign(self) -> Money { if matches!(self, Self::Asset | Self::Expense) { 1 } else { -1 } }
104 fn temporary(self) -> bool { matches!(self, Self::Revenue | Self::Expense) }
105}

107#[derive(Clone, Copy, PartialEq, Eq)]
108enum Category { Journal, Recurring, Reversal, Closing, Reopen, Depreciation, Interest, Writeoff }
109
110impl Category {
111 fn text(self) -> &'static str {
112 match self {
113 Self::Journal => "journal", Self::Recurring => "recurring", Self::Reversal => "reversal",
114 Self::Closing => "closing", Self::Reopen => "reopen", Self::Depreciation => "depreciation",
115 Self::Interest => "interest", Self::Writeoff => "writeoff",
116 }
117 }
118}

120#[derive(Clone, Copy, PartialEq, Eq)]
121enum Side { Debit, Credit }
122
123impl Side {
124 fn text(self) -> &'static str { if self == Self::Debit { "DR" } else { "CR" } }
125 fn sign(self) -> Money { if self == Self::Debit { 1 } else { -1 } }
126 fn reversed(self) -> Self { if self == Self::Debit { Self::Credit } else { Self::Debit } }
127}

136#[derive(Clone, Copy, Default)]
137struct Totals { debit: Money, credit: Money }
138
139impl Totals {
140 fn difference(self) -> Money { self.debit - self.credit }
141 fn add(&mut self, side: Side, amount: Money) {
142 match side {
143 Side::Debit => self.debit += amount,
144 Side::Credit => self.credit += amount,
145 }
146 }
147 fn merge(&mut self, other: Self) {
148 self.debit += other.debit;
149 self.credit += other.credit;
150 }
151}

157 amount: Money,
158 partner: Option<String>,
159}

161impl Line {

167struct Lines {
168 items: Vec<Line>,
169 amount: Totals,
170 accounts: BTreeMap<usize, Totals>,
171}
172
173impl Lines {
174 fn new(items: Vec<Line>) -> Rc<Self> {
175 let mut amount = Totals::default();
176 let mut accounts: BTreeMap<usize, Totals> = BTreeMap::new();
177 for line in &items {
178 amount.add(line.side, line.amount);
179 accounts.entry(line.account).or_default().add(line.side, line.amount);
180 }
181 Rc::new(Self { items, amount, accounts })
182 }
183
184 fn account_total(&self, account: usize) -> Totals {
185 self.accounts.get(&account).copied().unwrap_or_default()
186 }

239impl Book {
240 fn account(&self, code: &str) -> Result<usize, String> {
241 self.account_ids.get(code).copied().ok_or_else(|| format!("unknown account {code}"))
242 }
243
244 fn require_type(&self, code: &str, kind: AccountType) -> Result<usize, String> {
245 let index = self.account(code)?;
246 if self.accounts[index].kind != kind { return Err(format!("wrong type {code} {}", kind.text())); }
247 Ok(index)
248 }
249
250 fn active(&self, index: usize) -> Result<(), String> {
251 if !self.accounts[index].active { return Err(format!("inactive account {}", self.accounts[index].code)); }
252 Ok(())
253 }
254
255 fn active_lines(&self, lines: &Lines) -> Result<(), String> {
256 for line in &lines.items { self.active(line.account)?; }
257 Ok(())
258 }
259
260 fn check_open(&self, date: Day) -> Result<(), String> {
261 if self.closes.last().is_some_and(|record| date <= record.date) {
262 return Err(format!("period closed {}", date_text(date)));
263 }
264 Ok(())
265 }
266
267 fn closed_text(&self) -> String {
268 self.closes.last().map(|record| date_text(record.date)).unwrap_or_else(|| "none".into())
269 }
270
271 fn parent(&self, index: usize) -> Option<usize> {
272 let code = &self.accounts[index].code;
273 self.accounts.iter().enumerate().filter(|(i, account)| *i != index && code.starts_with(&account.code))
274 .max_by_key(|(_, account)| account.code.len()).map(|(i, _)| i)
275 }
276
277 fn parent_code(&self, index: usize) -> &str {
278 self.parent(index).map(|i| self.accounts[i].code.as_str()).unwrap_or("none")
279 }
280
281 fn ordered(&self) -> Vec<usize> {
282 let mut indices: Vec<usize> = (0..self.entries.len()).collect();
283 indices.sort_by_key(|&i| (self.entries[i].date, i));
284 indices
285 }

287 fn post(&mut self, id: String, date: Day, category: Category, memo: String, lines: Rc<Lines>) -> usize {
288 let index = self.entries.len();
289 self.entry_ids.insert(id.clone(), index);
290 self.entries.push(Entry { id, date, category, memo, lines, reverses: None, reversed_by: None });
291 index
292 }
293
294 fn reverse(&mut self, original: usize, date: Day, category: Category, memo: String) -> usize {
295 let source = &self.entries[original];
296 let lines = source.lines.items.iter().map(|line| Line { side: line.side.reversed(), ..line.clone() }).collect();
297 let id = format!("{}-R", source.id);
298 let index = self.post(id, date, category, memo, Lines::new(lines));
299 self.entries[original].reversed_by = Some(index);
300 self.entries[index].reverses = Some(original);
301 index
302 }
303
304 fn validate_draft(&self, draft: &Draft) -> Result<Rc<Lines>, String> {
305 if draft.lines.len() < 2 { return Err(format!("too few lines {}", draft.key)); }
306 let mut items = Vec::new();
307 for line in &draft.lines {
308 let account = self.account(&line.code)?;
309 if draft.date.is_some() { self.active(account)?; }
310 if line.amount == 0 { return Err(format!("zero amount {}", line.code)); }
311 items.push(Line { side: line.side, account, amount: line.amount, partner: line.partner.clone() });
312 }
313 let lines = Lines::new(items);
314 if lines.amount.debit != lines.amount.credit {
315 return Err(format!("unbalanced {} debit {} credit {}", draft.key,
316 money(lines.amount.debit), money(lines.amount.credit)));
317 }
318 Ok(lines)
319 }
320
321 fn account_totals(&self, account: usize, start: Day, end: Day) -> Totals {
322 let mut total = Totals::default();
323 for entry in &self.entries {
324 if (start..=end).contains(&entry.date) { total.merge(entry.lines.account_total(account)); }
325 }
326 total
327 }
328
329 fn balance(&self, account: usize, end: Day) -> Money {
330 self.accounts[account].kind.sign() * self.account_totals(account, 0, end).difference()
331 }
332
333 fn totals(&self, start: Day, end: Day, income: bool) -> Vec<Totals> {
334 let mut totals = vec![Totals::default(); self.accounts.len()];
335 for entry in &self.entries {
336 if !(start..=end).contains(&entry.date) || income && matches!(entry.category, Category::Closing | Category::Reopen) {
337 continue;
338 }
339 for (&account, &amount) in &entry.lines.accounts { totals[account].merge(amount); }
340 }
341 totals
342 }
343
344 fn daily_sum(&self, account: usize, start: Day, end: Day) -> Money {
345 let mut changes = vec![0; (end - start + 1) as usize];
346 let mut running = 0;
347 let sign = self.accounts[account].kind.sign();
348 for entry in &self.entries {
349 if entry.date > end { continue; }
350 let change = sign * entry.lines.account_total(account).difference();
351 if entry.date < start { running += change; }
352 else { changes[(entry.date - start) as usize] += change; }
353 }
354 let mut total = 0;
355 for change in changes {
356 running += change;
357 total += running;
358 }
359 total
360 }
361
362 fn settlements(&self, account: usize, date: Day) -> BTreeMap<String, Settlement> {
363 let mut result: BTreeMap<String, Settlement> = BTreeMap::new();
364 let sign = self.accounts[account].kind.sign();
365 for index in self.ordered() {
366 let entry = &self.entries[index];
367 if entry.date > date { break; }
368 if !entry.lines.accounts.contains_key(&account) { continue; }
369 for line in &entry.lines.items {
370 if line.account != account { continue; }
371 let Some(partner) = &line.partner else { continue };
372 let settlement = result.entry(partner.clone()).or_default();
373 let value = sign * line.side.sign() * line.amount;
374 if value > 0 {
375 let applied = value.min(settlement.unapplied);
376 settlement.unapplied -= applied;
377 if value > applied {
378 settlement.items.push_back(OpenItem { entry: index, original: value, remaining: value - applied });
379 }
380 } else {
381 let mut payment = -value;
382 while payment > 0 {
383 let Some(item) = settlement.items.front_mut() else { break };
384 let applied = payment.min(item.remaining);
385 payment -= applied;
386 item.remaining -= applied;
387 if item.remaining == 0 { settlement.items.pop_front(); }
388 }
389 settlement.unapplied += payment;
390 }
391 }
392 }
393 result
394 }

396 fn section(&self, heading: &str, kinds: &[AccountType], sign: Money, totals: &[Totals],
397 earnings: Option<Money>, output: &mut impl Write) -> Money {
398 let amounts: Vec<Money> = self.accounts.iter().enumerate().map(|(i, account)| {
399 if kinds.contains(&account.kind) { sign * totals[i].difference() } else { 0 }
400 }).collect();
401 let total = amounts.iter().sum::<Money>() + earnings.unwrap_or(0);
402 emit(output, format!(" {heading} {}", money(total)));
403 for &top in self.account_ids.values() {
404 if self.parent(top).is_some() { continue; }
405 let code = &self.accounts[top].code;
406 let members: Vec<usize> = self.account_ids.values().copied()
407 .filter(|&i| self.accounts[i].code.starts_with(code)).collect();
408 if members.iter().all(|&i| amounts[i] == 0) { continue; }
409 let subtotal: Money = members.iter().map(|&i| amounts[i]).sum();
410 emit(output, format!(" {code} {} {}", self.accounts[top].name, money(subtotal)));
411 for index in members {
412 if amounts[index] != 0 {
413 emit(output, format!(" {} {} {}", self.accounts[index].code, self.accounts[index].name, money(amounts[index])));
414 }
415 }
416 }
417 if let Some(value) = earnings { emit(output, format!(" current earnings {}", money(value))); }
418 total
419 }

421 fn execute(&mut self, words: &[&str], output: &mut impl Write) -> Result<(), String> {
422 match words[0] {
423 "ACCOUNT" => {
424 let code = words[1];
425 if self.account_ids.contains_key(code) { return Err(format!("duplicate account {code}")); }
426 let kind = AccountType::parse(words[2]);
427 for (other, &index) in &self.account_ids {
428 if (other.starts_with(code) || code.starts_with(other)) && kind.class() != self.accounts[index].kind.class() {
429 return Err(format!("class mismatch {code} {other}"));
430 }
431 }
432 let index = self.accounts.len();
433 self.accounts.push(Account { code: code.into(), kind, name: words[3].into(), active: true });
434 self.account_ids.insert(code.into(), index);
435 emit(output, format!("ACCOUNT {code} {} {} parent {}", kind.text(), words[3], self.parent_code(index)));
436 }
437 "DEACTIVATE" => {
438 let index = self.account(words[1])?;
439 if !self.accounts[index].active { return Err(format!("already inactive {}", words[1])); }
440 let balance = self.balance(index, Day::MAX);
441 if balance != 0 { return Err(format!("nonzero balance {} {}", words[1], money(balance))); }
442 self.accounts[index].active = false;
443 emit(output, format!("DEACTIVATE {} {}", words[1], self.accounts[index].name));
444 }
445 "JOURNAL" | "TEMPLATE" => {
446 if let Some(draft) = &self.draft { emit(output, format!("DISCARD {} lines {}", draft.key, draft.lines.len())); }
447 let journal = words[0] == "JOURNAL";
448 self.draft = Some(Draft {
449 key: words[1].into(), date: journal.then(|| words[2].into()),
450 memo: words[if journal { 3 } else { 2 }].into(), lines: Vec::new(),
451 });
452 }
453 "DR" | "CR" => {
454 let draft = self.draft.as_mut().ok_or("no open entry")?;
455 draft.lines.push(DraftLine {
456 side: if words[0] == "DR" { Side::Debit } else { Side::Credit },
457 code: words[1].into(), amount: parse_money(words[2]), partner: words.get(3).map(|name| (*name).into()),
458 });
459 }
460 "POST" => {
461 let draft = self.draft.take().ok_or("no open entry")?;
462 if let Some(token) = &draft.date {
463 if self.entry_ids.contains_key(&draft.key) { return Err(format!("duplicate entry {}", draft.key)); }
464 let date = parse_date(token)?;
465 self.check_open(date)?;
466 let lines = self.validate_draft(&draft)?;
467 let index = self.post(draft.key, date, Category::Journal, draft.memo, lines);
468 let entry = &self.entries[index];
469 emit(output, format!("POST {} {} lines {} amount {}", entry.id, date_text(date),
470 entry.lines.items.len(), money(entry.lines.amount.debit)));
471 } else {
472 if self.templates.contains_key(&draft.key) { return Err(format!("duplicate template {}", draft.key)); }
473 let lines = self.validate_draft(&draft)?;
474 emit(output, format!("TEMPLATE {} lines {} amount {}", draft.key, lines.items.len(), money(lines.amount.debit)));
475 self.templates.insert(draft.key, Template { memo: draft.memo, lines, uses: 0 });
476 }
477 }
478 "DRAFT" => {
479 let draft = self.draft.as_ref().ok_or("no open entry")?;
480 let mut total = Totals::default();
481 for line in &draft.lines { total.add(line.side, line.amount); }
482 let kind = if draft.date.is_some() { "journal" } else { "template" };
483 emit(output, format!("DRAFT {kind} {} lines {} debit {} credit {} difference {}", draft.key,
484 draft.lines.len(), money(total.debit), money(total.credit), money(total.difference())));
485 }
486 "REVERSE" => {
487 let index = *self.entry_ids.get(words[1]).ok_or_else(|| format!("unknown entry {}", words[1]))?;
488 let entry = &self.entries[index];
489 if matches!(entry.category, Category::Reversal | Category::Closing | Category::Reopen) {
490 return Err(format!("cannot reverse {}", words[1]));
491 }
492 if entry.reversed_by.is_some() { return Err(format!("already reversed {}", words[1])); }
493 let date = parse_date(words[2])?;
494 if date < entry.date { return Err(format!("reverse before entry {} {}", words[1], date_text(entry.date))); }
495 self.check_open(date)?;
496 self.active_lines(&entry.lines)?;
497 let memo = entry.memo.clone();
498 let reversed = self.reverse(index, date, Category::Reversal, memo);
499 let entry = &self.entries[reversed];
500 emit(output, format!("REVERSE {} entry {} {} amount {}", words[1], entry.id, date_text(date), money(entry.lines.amount.debit)));
501 }
502 "RECUR" => {
503 let name = words[1];
504 let template = self.templates.get(name).cloned().ok_or_else(|| format!("unknown template {name}"))?;
505 let date = parse_date(words[2])?;
506 let count: usize = words[3].parse().unwrap();
507 if count == 0 { return Err("invalid count".into()); }
508 self.check_open(date)?;
509 self.active_lines(&template.lines)?;
510 emit(output, format!("RECUR {name} count {count}"));
511 for month in 0..count {
512 let pattern = self.templates.get_mut(name).unwrap();
513 pattern.uses += 1;
514 let id = format!("{name}-{}", pattern.uses);
515 let index = self.post(id, months_after(date, month), Category::Recurring, template.memo.clone(), Rc::clone(&template.lines));
516 let entry = &self.entries[index];
517 emit(output, format!(" POST {} {} lines {} amount {}", entry.id, date_text(entry.date),
518 entry.lines.items.len(), money(entry.lines.amount.debit)));
519 }
520 }
521 "DEPRECIATE" => {
522 let date = parse_date(words[1])?;
523 self.check_open(date)?;
524 let asset = self.require_type(words[2], AccountType::Asset)?;
525 let contra = self.require_type(words[3], AccountType::ContraAsset)?;
526 let expense = self.require_type(words[4], AccountType::Expense)?;
527 self.active(contra)?;
528 self.active(expense)?;
529 let cost = self.balance(asset, date);
530 let accumulated = self.balance(contra, date);
531 let life: Money = words[5].parse().unwrap();
532 let amount = if cost > 0 { (cost / life).min(cost - accumulated) } else { 0 };
533 if amount <= 0 { return Err(format!("nothing to depreciate {}", words[2])); }
534 self.depreciation_counter += 1;
535 let id = format!("DEP-{}", self.depreciation_counter);
536 let lines = Lines::new(vec![Line::new(Side::Debit, expense, amount), Line::new(Side::Credit, contra, amount)]);
537 self.post(id.clone(), date, Category::Depreciation, "depreciation".into(), lines);
538 emit(output, format!("DEPRECIATE {} entry {id} {} amount {} accumulated {} book {}", words[2], date_text(date),
539 money(amount), money(accumulated + amount), money(cost - accumulated - amount)));
540 }
541 "INTEREST" => {
542 let account = self.account(words[1])?;
543 let kind = self.accounts[account].kind;
544 if !matches!(kind, AccountType::Asset | AccountType::Liability) {
545 return Err(format!("not interest bearing {}", words[1]));
546 }
547 let offset = self.require_type(words[2], if kind == AccountType::Asset { AccountType::Revenue } else { AccountType::Expense })?;
548 let (start, end) = date_range(words[3], words[4])?;
549 self.check_open(end)?;
550 self.active(account)?;
551 self.active(offset)?;
552 let total = self.daily_sum(account, start, end);
553 let rate: Money = words[5].parse().unwrap();
554 let amount = total * rate / 3_650_000;
555 if amount <= 0 { return Err(format!("no interest {}", words[1])); }
556 let (debit, credit) = if kind == AccountType::Asset { (account, offset) } else { (offset, account) };
557 let lines = Lines::new(vec![Line::new(Side::Debit, debit, amount), Line::new(Side::Credit, credit, amount)]);
558 self.interest_counter += 1;
559 let id = format!("INT-{}", self.interest_counter);
560 self.post(id.clone(), end, Category::Interest, "interest".into(), lines);
561 let days = end - start + 1;
562 emit(output, format!("INTEREST {} entry {id} {} days {days} average {} amount {}", words[1], date_text(end),
563 money(total / Money::from(days)), money(amount)));
564 }
565 "WRITEOFF" => {
566 let date = parse_date(words[1])?;
567 self.check_open(date)?;
568 let account = self.require_type(words[2], AccountType::Asset)?;
569 let expense = self.require_type(words[3], AccountType::Expense)?;
570 self.active(account)?;
571 self.active(expense)?;
572 let age: Day = words[4].parse().unwrap();
573 let targets: Vec<(String, Money)> = self.settlements(account, date).into_iter().filter_map(|(partner, settlement)| {
574 let amount: Money = settlement.items.iter().filter(|item| date - self.entries[item.entry].date > age)
575 .map(|item| item.remaining).sum();
576 (amount > 0).then_some((partner, amount))
577 }).collect();
578 let total: Money = targets.iter().map(|(_, amount)| amount).sum();
579 if total == 0 { return Err(format!("nothing to write off {}", words[2])); }
580 let mut lines = vec![Line::new(Side::Debit, expense, total)];
581 for (partner, amount) in &targets {
582 lines.push(Line { side: Side::Credit, account, amount: *amount, partner: Some(partner.clone()) });
583 }
584 self.writeoff_counter += 1;
585 let id = format!("WO-{}", self.writeoff_counter);
586 self.post(id.clone(), date, Category::Writeoff, "writeoff".into(), Lines::new(lines));
587 emit(output, format!("WRITEOFF {} entry {id} {} partners {} amount {}", words[2], date_text(date), targets.len(), money(total)));
588 for (partner, amount) in targets { emit(output, format!(" {partner} {}", money(amount))); }
589 }
590 "CLOSE" => {
591 let date = parse_date(words[1])?;
592 self.check_open(date)?;
593 let equity = self.require_type(words[2], AccountType::Equity)?;
594 self.active(equity)?;
595 let totals = self.totals(0, date, false);
596 let targets: Vec<usize> = self.account_ids.values().copied()
597 .filter(|&i| self.accounts[i].kind.temporary() && totals[i].difference() != 0).collect();
598 for &index in &targets { self.active(index)?; }
599 let mut lines = Vec::new();
600 let mut income = 0;
601 for index in targets {
602 let difference = totals[index].difference();
603 lines.push(Line::new(if difference > 0 { Side::Credit } else { Side::Debit }, index, difference.abs()));
604 income -= difference;
605 }
606 if income != 0 { lines.push(Line::new(if income > 0 { Side::Credit } else { Side::Debit }, equity, income.abs())); }
607 self.close_counter += 1;
608 let entry = if lines.is_empty() { None } else {
609 Some(self.post(format!("CLOSE-{}", self.close_counter), date, Category::Closing, "closing".into(), Lines::new(lines)))
610 };
611 self.closes.push(CloseRecord { date, entry });
612 if let Some(index) = entry {
613 let entry = &self.entries[index];
614 emit(output, format!("CLOSE {} entry {} lines {} net income {}", date_text(date), entry.id, entry.lines.items.len(), money(income)));
615 } else { emit(output, format!("CLOSE {} no entry", date_text(date))); }
616 }
617 "REOPEN" => {
618 let record = self.closes.last().copied().ok_or("nothing to reopen")?;
619 if let Some(index) = record.entry { self.active_lines(&self.entries[index].lines)?; }
620 let detail = if let Some(index) = record.entry {
621 let reversed = self.reverse(index, record.date, Category::Reopen, "reopen".into());
622 format!("entry {}", self.entries[reversed].id)
623 } else { "no entry".into() };
624 self.closes.pop();
625 emit(output, format!("REOPEN {} {detail} closed {}", date_text(record.date), self.closed_text()));
626 }
627 "LEDGER" => {
628 let index = self.account(words[1])?;
629 let (start, end) = date_range(words[2], words[3])?;
630 let account = &self.accounts[index];
631 let mut running = self.balance(index, start - 1);
632 let mut totals = Totals::default();
633 emit(output, format!("LEDGER {} {} {} {}", account.code, account.name, words[2], words[3]));
634 emit(output, format!(" opening {}", money(running)));
635 for entry_index in self.ordered() {
636 let entry = &self.entries[entry_index];
637 if !(start..=end).contains(&entry.date) { continue; }
638 for line in &entry.lines.items {
639 if line.account != index { continue; }
640 totals.add(line.side, line.amount);
641 running += account.kind.sign() * line.side.sign() * line.amount;
642 emit(output, format!(" {} {} {} {} balance {} {}", date_text(entry.date), entry.id, line.side.text(),
643 money(line.amount), money(running), entry.memo));
644 }
645 }
646 let days = end - start + 1;
647 emit(output, format!(" ending {} debits {} credits {} days {days} average {}", money(running),
648 money(totals.debit), money(totals.credit), money(self.daily_sum(index, start, end) / Money::from(days))));
649 }
650 "ACTIVITY" => {
651 let index = self.account(words[1])?;
652 let year: i32 = words[2].parse().unwrap();
653 emit(output, format!("ACTIVITY {} {} {year}", words[1], self.accounts[index].name));
654 let mut totals = Totals::default();
655 for month in 1..=12 {
656 let start = day_number(year, month, 1);
657 let end = start + month_days(year, month) - 1;
658 let current = self.account_totals(index, start, end);
659 totals.merge(current);
660 emit(output, format!(" {year:04}-{month:02} debit {} credit {} balance {}", money(current.debit),
661 money(current.credit), money(self.balance(index, end))));
662 }
663 emit(output, format!(" total debit {} credit {}", money(totals.debit), money(totals.credit)));
664 }
665 "TRIAL" => {
666 let date = parse_date(words[1])?;
667 let totals = self.totals(0, date, false);
668 let mut total = Totals::default();
669 emit(output, format!("TRIAL {}", words[1]));
670 for &index in self.account_ids.values() {
671 let difference = totals[index].difference();
672 if difference == 0 { continue; }
673 let side = if difference > 0 { "debit" } else { "credit" };
674 let account = &self.accounts[index];
675 emit(output, format!(" {} {} {side} {}", account.code, account.name, money(difference.abs())));
676 if difference > 0 { total.debit += difference; } else { total.credit -= difference; }
677 }
678 emit(output, format!(" total debit {} credit {}", money(total.debit), money(total.credit)));
679 }
680 "INCOME" => {
681 let (start, end) = date_range(words[1], words[2])?;
682 let totals = self.totals(start, end, true);
683 emit(output, format!("INCOME {} {}", words[1], words[2]));
684 let revenue = self.section("REVENUE", &[AccountType::Revenue], -1, &totals, None, output);
685 let expense = self.section("EXPENSE", &[AccountType::Expense], 1, &totals, None, output);
686 emit(output, format!(" NET INCOME {}", money(revenue - expense)));
687 }
688 "BALANCE" => {
689 let date = parse_date(words[1])?;
690 let totals = self.totals(0, date, false);
691 let earnings = -self.accounts.iter().enumerate().filter(|(_, account)| account.kind.temporary())
692 .map(|(i, _)| totals[i].difference()).sum::<Money>();
693 emit(output, format!("BALANCE {}", words[1]));
694 let assets = self.section("ASSETS", &[AccountType::Asset, AccountType::ContraAsset], 1, &totals, None, output);
695 let liabilities = self.section("LIABILITIES", &[AccountType::Liability], -1, &totals, None, output);
696 let equity = self.section("EQUITY", &[AccountType::Equity], -1, &totals, Some(earnings), output);
697 if assets == liabilities + equity { emit(output, " check ok"); }
698 else { emit(output, format!(" check mismatch {}", money(assets - liabilities - equity))); }
699 }
700 "ROLLUP" => {
701 let date = parse_date(words[2])?;
702 let selected: Vec<usize> = self.account_ids.iter().filter(|(code, _)| code.starts_with(words[1])).map(|(_, &i)| i).collect();
703 emit(output, format!("ROLLUP {} {}", words[1], words[2]));
704 if selected.is_empty() { emit(output, " (none)"); return Ok(()); }
705 let totals = self.totals(0, date, false);
706 let mut postings = Totals::default();
707 for &index in &selected {
708 let account = &self.accounts[index];
709 let depth = selected.iter().filter(|&&i| i != index && account.code.starts_with(&self.accounts[i].code)).count();
710 let subtotal: Money = selected.iter().filter(|&&i| self.accounts[i].code.starts_with(&account.code))
711 .map(|&i| totals[i].difference()).sum();
712 emit(output, format!("{}{} {} own {} total {}", " ".repeat(2 * (1 + depth)), account.code, account.name,
713 money(account.kind.sign() * totals[index].difference()), money(account.kind.sign() * subtotal)));
714 postings.merge(totals[index]);
715 }
716 emit(output, format!(" postings debit {} credit {}", money(postings.debit), money(postings.credit)));
717 }
718 "AGING" | "ITEMS" => {
719 let index = self.account(words[1])?;
720 let aging = words[0] == "AGING";
721 let date = parse_date(words[if aging { 2 } else { 3 }])?;
722 let account = &self.accounts[index];
723 let mut settlements = self.settlements(index, date);
724 if aging {
725 emit(output, format!("AGING {} {} {}", account.code, account.name, words[2]));
726 let mut totals = [0; 5];
727 for (partner, settlement) in settlements {
728 if settlement.items.is_empty() && settlement.unapplied == 0 { continue; }
729 let mut values = [0, 0, 0, 0, settlement.unapplied];
730 for item in settlement.items {
731 let age = date - self.entries[item.entry].date;
732 let bucket = if age <= 30 { 0 } else if age <= 60 { 1 } else if age <= 90 { 2 } else { 3 };
733 values[bucket] += item.remaining;
734 }
735 emit(output, aging_row(&partner, &values));
736 for i in 0..5 { totals[i] += values[i]; }
737 }
738 emit(output, aging_row("total", &totals));
739 } else {
740 emit(output, format!("ITEMS {} {} {} {}", account.code, account.name, words[2], words[3]));
741 let settlement = settlements.remove(words[2]).unwrap_or_default();
742 let mut remaining = 0;
743 for item in settlement.items {
744 let entry = &self.entries[item.entry];
745 remaining += item.remaining;
746 emit(output, format!(" {} {} amount {} open {} age {}", date_text(entry.date), entry.id,
747 money(item.original), money(item.remaining), date - entry.date));
748 }
749 emit(output, format!(" unapplied {} balance {}", money(settlement.unapplied), money(remaining - settlement.unapplied)));
750 }
751 }
752 "ENTRY" => {
753 let index = *self.entry_ids.get(words[1]).ok_or_else(|| format!("unknown entry {}", words[1]))?;
754 let entry = &self.entries[index];
755 emit(output, format!("ENTRY {} {} {} {}", entry.id, date_text(entry.date), entry.category.text(), entry.memo));
756 for line in &entry.lines.items {
757 let account = &self.accounts[line.account];
758 let partner = line.partner.as_ref().map(|name| format!(" {name}")).unwrap_or_default();
759 emit(output, format!(" {} {} {} {}{partner}", line.side.text(), account.code, account.name, money(line.amount)));
760 }
761 emit(output, format!(" total {}", money(entry.lines.amount.debit)));
762 if let Some(original) = entry.reverses { emit(output, format!(" reverses {}", self.entries[original].id)); }
763 if let Some(reversal) = entry.reversed_by { emit(output, format!(" reversed by {}", self.entries[reversal].id)); }
764 }
765 "DAYBOOK" => {
766 let (start, end) = date_range(words[1], words[2])?;
767 emit(output, format!("DAYBOOK {} {}", words[1], words[2]));
768 let mut count = 0;
769 let mut total = 0;
770 for index in self.ordered() {
771 let entry = &self.entries[index];
772 if !(start..=end).contains(&entry.date) { continue; }
773 count += 1;
774 total += entry.lines.amount.debit;
775 emit(output, format!(" {} {} {} {} {}", date_text(entry.date), entry.id, entry.category.text(), money(entry.lines.amount.debit), entry.memo));
776 }
777 emit(output, format!(" total entries {count} amount {}", money(total)));
778 }
779 "ACCOUNTS" => {
780 emit(output, "ACCOUNTS");
781 let totals = self.totals(0, Day::MAX, false);
782 for &index in self.account_ids.values() {
783 let account = &self.accounts[index];
784 let status = if account.active { "active" } else { "inactive" };
785 emit(output, format!(" {} {} {} parent {} balance {} {status}", account.code, account.kind.text(), account.name,
786 self.parent_code(index), money(account.kind.sign() * totals[index].difference())));
787 }
788 if self.accounts.is_empty() { emit(output, " (none)"); }
789 }
790 "TEMPLATES" => {
791 emit(output, "TEMPLATES");
792 for (name, template) in &self.templates {
793 emit(output, format!(" {name} {} lines {} amount {} used {}", template.memo, template.lines.items.len(),
794 money(template.lines.amount.debit), template.uses));
795 }
796 if self.templates.is_empty() { emit(output, " (none)"); }
797 }
798 "STATUS" => {
799 let inactive = self.accounts.iter().filter(|account| !account.active).count();
800 let open = self.draft.as_ref().map(|draft| draft.key.as_str()).unwrap_or("none");
801 emit(output, format!("STATUS accounts {} inactive {inactive} entries {} templates {} closed {} open {open}",
802 self.accounts.len(), self.entries.len(), self.templates.len(), self.closed_text()));
803 }
804 _ => unreachable!(),
805 }
806 Ok(())
807 }
808}
809
810fn aging_row(name: &str, values: &[Money; 5]) -> String {
811 let balance = values[..4].iter().sum::<Money>() - values[4];
812 format!(" {name} 0-30 {} 31-60 {} 61-90 {} 91+ {} unapplied {} balance {}",
813 money(values[0]), money(values[1]), money(values[2]), money(values[3]), money(values[4]), money(balance))
814}
815
816fn main() {
817 let stdin = io::stdin();
818 let mut input = stdin.lock().lines();
819 let count: usize = input.next().unwrap().unwrap().parse().unwrap();
820 let stdout = io::stdout();
821 let mut output = io::BufWriter::new(stdout.lock());
822 let mut book = Book::default();
823 for line in input.take(count) {
824 let line = line.unwrap();
825 let words: Vec<&str> = line.split_whitespace().collect();
826 if let Err(error) = book.execute(&words, &mut output) { emit(&mut output, format!("ERROR {error}")); }
827 }
828}
