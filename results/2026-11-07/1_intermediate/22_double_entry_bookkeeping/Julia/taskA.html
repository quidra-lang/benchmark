10@enum AccountType ASSET LIABILITY EQUITY REVENUE EXPENSE CONTRA_ASSET
11@enum EntryCategory Journal Recurring Reversal Closing Reopen Depreciation Interest Writeoff
12const ACCOUNT_TYPES = Dict(string(kind) => kind for kind in instances(AccountType))
13normal_sign(kind::AccountType)::Cents = kind in (ASSET, EXPENSE) ? 1 : -1
14type_family(kind::AccountType) = kind == CONTRA_ASSET ? ASSET : kind
15category_text(category::EntryCategory) = lowercase(string(category))

17function money(amount::Cents)
18 whole, remainder = divrem(abs(amount), 100)
19 return "$(amount < 0 ? "-" : "")$whole.$(lpad(string(remainder), 2, '0'))"
20end
21
22function parse_cents(text::String)
23 parts = split(text, '.')
24 return parse(Cents, parts[1]) * 100 + parse(Cents, parts[2])
25end
26
27leap(year::Int) = year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
28
29function month_length(year::Int, month::Int)
30 lengths = (31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31)
31 return lengths[month] + (month == 2 && leap(year))
32end
33
34function day_number(year::Int, month::Int, day::Int)
35 prior = year - 1
36 result = 365 * prior + div(prior, 4) - div(prior, 100) + div(prior, 400) + day - 730120
37 for m in 1:month-1
38 result += month_length(year, m)
39 end
40 return result
41end
42
43function date_parts(number::Int)
44 year = 2000
45 while number >= 365 + leap(year)
46 number -= 365 + leap(year)
47 year += 1
48 end
49 month = 1
50 while number >= month_length(year, month)
51 number -= month_length(year, month)
52 month += 1
53 end
54 return year, month, number + 1
55end
56
57function date_text(number::Int)
58 year, month, day = date_parts(number)
59 return "$(lpad(string(year), 4, '0'))-$(lpad(string(month), 2, '0'))-$(lpad(string(day), 2, '0'))"
60end
61
62function checked_date(text::String)
63 year, month, day = parse.(Int, split(text, '-'))
64 if !(2000 <= year <= 2099 && 1 <= month <= 12 && day >= 1)
65 fail("invalid date $text")
66 end
67 day <= month_length(year, month) || fail("invalid date $text")
68 return day_number(year, month, day)
69end
70
71function checked_range(first::String, last::String)
72 start = checked_date(first)
73 stop = checked_date(last)
74 start <= stop || fail("invalid range $first $last")
75 return start, stop
76end
77
78function months_after(number::Int, count::Int)
79 year, month, day = date_parts(number)
80 offset = month - 1 + count
81 year += div(offset, 12)
82 month = offset % 12 + 1
83 return day_number(year, month, min(day, month_length(year, month)))
84end

103mutable struct Totals

105 credit::Cents
106end

108
109function line_totals(lines::Vector{PostingLine})
110 debit = sum((line.amount for line in lines if line.debit); init=Cents(0))
111 credit = sum((line.amount for line in lines if !line.debit); init=Cents(0))
112 return Totals(debit, credit)
113end

213function difference(totals::Dict{String,Totals}, code::String)
214 values = total_pair(totals, code)

218function aggregate_entries(books::Books, first::Int=0, last::Int=36524; operating::Bool=false)
219 totals = Dict{String,Totals}()
220 for entry in books.entries
221 if entry.day < first || entry.day > last || (operating && entry.category in (Closing, Reopen))
222 continue
223 end
224 for line in entry.lines
225 values = get!(totals, line.code, Totals())
226 if line.debit
227 values.debit += line.amount
228 else
229 values.credit += line.amount
230 end
231 end
232 end
233 return totals
234end

236function account_balance(books::Books, account::Account, day::Int=36524)
237 return normal_sign(account.kind) * difference(aggregate_entries(books, 0, day), account.code)
238end

240function daily_sum(books::Books, account::Account, first::Int, last::Int)
241 changes = zeros(Cents, last - first + 1)
242 opening = Cents(0)
243 for entry in books.entries
244 entry.day > last && continue
245 change = normal_sign(account.kind) * sum((line_difference(line) for line in entry.lines
246 if line.code == account.code); init=Cents(0))
247 if entry.day < first
248 opening += change
249 else
250 changes[entry.day - first + 1] += change
251 end
252 end
253 return sum(cumsum(changes) .+ opening)
254end

258function post_entry!(books::Books, id::String, day::Int, category::EntryCategory, memo::String, lines::Vector{PostingLine})
259 index = length(books.entries) + 1
260 push!(books.entries, Entry(id, day, category, memo, copy(lines), line_totals(lines).debit, nothing, nothing))
261 books.entry_positions[id] = index
262 return index
263end

265function reverse_entry!(books::Books, index::Int, day::Int, category::EntryCategory, memo::String)
266 original = books.entries[index]
267 lines = [PostingLine(!line.debit, line.code, line.amount, line.partner) for line in original.lines]
268 created = post_entry!(books, original.id * "-R", day, category, memo, lines)
269 original.reversed_by = created
270 books.entries[created].reverses = index
271 return created
272end

278 original::Cents
279 remaining::Cents
280end

282mutable struct Settlement
283 items::Vector{OpenItem}
284 head::Int
285 unapplied::Cents
286end
287Settlement() = Settlement(OpenItem[], 1, 0)
288open_items(group::Settlement) = @view group.items[group.head:end]
289
290function settlements_for(books::Books, account::Account, day::Int)
291 settlements = Dict{String,Settlement}()
292 for index in posting_order(books)
293 entry = books.entries[index]
294 entry.day > day && break
295 for line in entry.lines
296 (line.code != account.code || line.partner === nothing) && continue
297 group = get!(settlements, line.partner, Settlement())
298 value = normal_sign(account.kind) * line_difference(line)
299 if value > 0
300 used = min(value, group.unapplied)
301 group.unapplied -= used
302 value > used && push!(group.items, OpenItem(index, value, value - used))
303 else
304 remaining = -value
305 while remaining > 0 && group.head <= length(group.items)
306 item = group.items[group.head]
307 used = min(remaining, item.remaining)
308 item.remaining -= used
309 remaining -= used
310 item.remaining == 0 && (group.head += 1)
311 end
312 group.unapplied += remaining
313 end
314 end
315 end
316 return settlements
317end

319function section(books::Books, heading::String, kinds, sign::Cents, totals::Dict{String,Totals}; earnings::Union{Nothing,Cents}=nothing)
320 selected = [code for code in sorted_codes(books) if books.accounts[code].kind in kinds]
321 total = sign * sum((difference(totals, code) for code in selected); init=Cents(0))
322 earnings !== nothing && (total += earnings)
323 emit(" $heading", money(total))
324 roots = sort!(unique([top_account(books, code) for code in selected]))
325 for root in roots
326 group = [code for code in selected if top_account(books, code) == root]
327 values = [sign * difference(totals, code) for code in group]
328 any(value -> value != 0, values) || continue
329 emit(" $root", books.accounts[root].name, money(sum(values)))
330 for (code, value) in zip(group, values)
331 value != 0 && emit(" $code", books.accounts[code].name, money(value))
332 end
333 end
334 earnings !== nothing && emit(" current earnings", money(earnings))
335 return total
336end

342function post_draft!(books::Books)
343 block = books.draft
344 block === nothing && fail("no open entry")
345 books.draft = nothing
346 day = 0
347 if block.journal
348 haskey(books.entry_positions, block.key) && fail("duplicate entry $(block.key)")
349 day = checked_date(block.date_token)
350 check_closed(books, day)
351 elseif haskey(books.templates, block.key)
352 fail("duplicate template $(block.key)")
353 end
354 length(block.lines) >= 2 || fail("too few lines $(block.key)")
355 for line in block.lines
356 account = get_account(books, line.code)
357 block.journal && !account.active && fail("inactive account $(line.code)")
358 line.amount == 0 && fail("zero amount $(line.code)")
359 end
360 values = line_totals(block.lines)
361 if values.debit != values.credit
362 fail("unbalanced $(block.key) debit $(money(values.debit)) credit $(money(values.credit))")
363 end
364 if block.journal
365 index = post_entry!(books, block.key, day, Journal, block.memo, block.lines)
366 emit(posted_text(books.entries[index]))
367 else
368 books.templates[block.key] = Template(block.memo, copy(block.lines), 0)
369 emit("TEMPLATE", block.key, "lines", length(block.lines), "amount", money(values.debit))
370 end
371end

373function command!(books::Books, words::Vector{String})
374 op = words[1]
375 if op == "ACCOUNT"
376 code = words[2]
377 haskey(books.accounts, code) && fail("duplicate account $code")
378 kind = ACCOUNT_TYPES[words[3]]
379 for other in sorted_codes(books)
380 if (startswith(code, other) || startswith(other, code)) && type_family(kind) != type_family(books.accounts[other].kind)
381 fail("class mismatch $code $other")
382 end
383 end
384 books.accounts[code] = Account(code, kind, words[4], true)
385 emit("ACCOUNT", code, kind, words[4], "parent", parent_code(books, code))
386 elseif op == "DEACTIVATE"
387 account = get_account(books, words[2])
388 account.active || fail("already inactive $(account.code)")
389 balance = account_balance(books, account)
390 balance == 0 || fail("nonzero balance $(account.code) $(money(balance))")
391 account.active = false
392 emit("DEACTIVATE", account.code, account.name)
393 elseif op in ("JOURNAL", "TEMPLATE")
394 if books.draft !== nothing
395 emit("DISCARD", books.draft.key, "lines", length(books.draft.lines))
396 end
397 journal = op == "JOURNAL"
398 books.draft = Draft(journal, words[2], journal ? words[3] : nothing,
399 journal ? words[4] : words[3], PostingLine[])
400 elseif op in ("DR", "CR")
401 books.draft === nothing && fail("no open entry")
402 partner = length(words) == 4 ? words[4] : nothing
403 push!(books.draft.lines, PostingLine(op == "DR", words[2], parse_cents(words[3]), partner))
404 elseif op == "POST"
405 post_draft!(books)
406 elseif op == "DRAFT"
407 block = books.draft
408 block === nothing && fail("no open entry")
409 amounts = line_totals(block.lines)
410 emit("DRAFT", block.journal ? "journal" : "template", block.key, "lines", length(block.lines),
411 "debit", money(amounts.debit), "credit", money(amounts.credit), "difference", money(amounts.debit - amounts.credit))
412 elseif op == "REVERSE"
413 index = get_entry(books, words[2])
414 original = books.entries[index]
415 original.category in (Reversal, Closing, Reopen) && fail("cannot reverse $(original.id)")
416 original.reversed_by !== nothing && fail("already reversed $(original.id)")
417 day = checked_date(words[3])
418 day < original.day && fail("reverse before entry $(original.id) $(date_text(original.day))")
419 check_closed(books, day)
420 active_check(books, line_codes(original.lines))
421 created = reverse_entry!(books, index, day, Reversal, original.memo)
422 posted = books.entries[created]
423 emit("REVERSE", original.id, "entry", posted.id, date_text(day), "amount", money(posted.amount))
424 elseif op == "RECUR"
425 pattern = get_template(books, words[2])
426 day = checked_date(words[3])
427 repeat_count = parse(Int, words[4])
428 repeat_count == 0 && fail("invalid count")
429 check_closed(books, day)
430 active_check(books, line_codes(pattern.lines))
431 emit("RECUR", words[2], "count", repeat_count)
432 for offset in 0:repeat_count-1
433 pattern.uses += 1
434 id = "$(words[2])-$(pattern.uses)"
435 index = post_entry!(books, id, months_after(day, offset), Recurring, pattern.memo, pattern.lines)
436 emit(" " * posted_text(books.entries[index]))
437 end
438 elseif op == "DEPRECIATE"
439 day = checked_date(words[2])
440 check_closed(books, day)
441 asset = get_account(books, words[3], ASSET)
442 contra = get_account(books, words[4], CONTRA_ASSET)
443 expense = get_account(books, words[5], EXPENSE)
444 active_check(books, [contra.code, expense.code])
445 cost = account_balance(books, asset, day)
446 accumulated = account_balance(books, contra, day)
447 amount = cost > 0 ? min(div(cost, parse(Int, words[6])), cost - accumulated) : Cents(0)
448 amount <= 0 && fail("nothing to depreciate $(asset.code)")
449 books.depreciation_count += 1
450 id = "DEP-$(books.depreciation_count)"
451 post_entry!(books, id, day, Depreciation, "depreciation",
452 [PostingLine(true, expense.code, amount), PostingLine(false, contra.code, amount)])
453 emit("DEPRECIATE", asset.code, "entry", id, date_text(day), "amount", money(amount),
454 "accumulated", money(accumulated + amount), "book", money(cost - accumulated - amount))
455 elseif op == "INTEREST"
456 account = get_account(books, words[2])
457 account.kind in (ASSET, LIABILITY) || fail("not interest bearing $(account.code)")
458 offset = get_account(books, words[3], account.kind == ASSET ? REVENUE : EXPENSE)
459 first, last = checked_range(words[4], words[5])
460 check_closed(books, last)
461 active_check(books, [account.code, offset.code])
462 total = daily_sum(books, account, first, last)
463 amount = div(total * parse(Cents, words[6]), 3650000)
464 amount <= 0 && fail("no interest $(account.code)")
465 books.interest_count += 1
466 id = "INT-$(books.interest_count)"
467 debit = account.kind == ASSET ? account.code : offset.code
468 credit = account.kind == ASSET ? offset.code : account.code
469 post_entry!(books, id, last, Interest, "interest", [PostingLine(true, debit, amount), PostingLine(false, credit, amount)])
470 days = last - first + 1
471 emit("INTEREST", account.code, "entry", id, date_text(last), "days", days,
472 "average", money(div(total, days)), "amount", money(amount))
473 elseif op == "WRITEOFF"
474 day = checked_date(words[2])
475 check_closed(books, day)
476 account = get_account(books, words[3], ASSET)
477 expense = get_account(books, words[4], EXPENSE)
478 active_check(books, [account.code, expense.code])
479 age = parse(Int, words[5])
480 settlements = settlements_for(books, account, day)
481 targets = Tuple{String,Cents}[]
482 for partner in sort!(collect(keys(settlements)))
483 amount = sum((item.remaining for item in open_items(settlements[partner])
484 if day - books.entries[item.entry].day > age); init=Cents(0))
485 amount > 0 && push!(targets, (partner, amount))
486 end
487 total = sum((amount for (_, amount) in targets); init=Cents(0))
488 total == 0 && fail("nothing to write off $(account.code)")
489 lines = [PostingLine(true, expense.code, total)]
490 append!(lines, [PostingLine(false, account.code, amount, partner) for (partner, amount) in targets])
491 books.writeoff_count += 1
492 id = "WO-$(books.writeoff_count)"
493 post_entry!(books, id, day, Writeoff, "writeoff", lines)
494 emit("WRITEOFF", account.code, "entry", id, date_text(day), "partners", length(targets), "amount", money(total))
495 for (partner, amount) in targets
496 emit(" $partner", money(amount))
497 end
498 elseif op == "CLOSE"
499 day = checked_date(words[2])
500 check_closed(books, day)
501 equity = get_account(books, words[3], EQUITY)
502 active_check(books, [equity.code])
503 totals = aggregate_entries(books, 0, day)
504 selected = [code for code in sorted_codes(books)
505 if books.accounts[code].kind in (REVENUE, EXPENSE) && difference(totals, code) != 0]
506 active_check(books, selected)
507 lines = PostingLine[]
508 for code in selected
509 delta = difference(totals, code)
510 push!(lines, PostingLine(delta < 0, code, abs(delta)))
511 end
512 income = -sum((difference(totals, code) for code in selected); init=Cents(0))
513 income != 0 && push!(lines, PostingLine(income < 0, equity.code, abs(income)))
514 books.close_count += 1
515 index = nothing
516 if !isempty(lines)
517 id = "CLOSE-$(books.close_count)"
518 index = post_entry!(books, id, day, Closing, "closing", lines)
519 emit("CLOSE", date_text(day), "entry", id, "lines", length(lines), "net income", money(income))
520 else
521 emit("CLOSE", date_text(day), "no entry")
522 end
523 push!(books.closes, CloseRecord(day, index))
524 elseif op == "REOPEN"
525 isempty(books.closes) && fail("nothing to reopen")
526 record = books.closes[end]
527 detail = "no entry"
528 if record.entry !== nothing
529 active_check(books, line_codes(books.entries[record.entry].lines))
530 created = reverse_entry!(books, record.entry, record.day, Reopen, "reopen")
531 detail = "entry " * books.entries[created].id
532 end
533 pop!(books.closes)
534 emit("REOPEN", date_text(record.day), detail, "closed", closed_text(books))
535 elseif op == "LEDGER"
536 account = get_account(books, words[2])
537 first, last = checked_range(words[3], words[4])
538 running = account_balance(books, account, first - 1)
539 debit = credit = Cents(0)
540 emit("LEDGER", account.code, account.name, date_text(first), date_text(last))
541 emit(" opening", money(running))
542 for index in posting_order(books)
543 entry = books.entries[index]
544 first <= entry.day <= last || continue
545 for line in entry.lines
546 line.code == account.code || continue
547 running += normal_sign(account.kind) * line_difference(line)
548 if line.debit
549 debit += line.amount
550 else
551 credit += line.amount
552 end
553 emit(" $(date_text(entry.day))", entry.id, line.debit ? "DR" : "CR", money(line.amount), "balance", money(running), entry.memo)
554 end
555 end
556 days = last - first + 1
557 average = div(daily_sum(books, account, first, last), days)
558 emit(" ending", money(running), "debits", money(debit), "credits", money(credit), "days", days, "average", money(average))
559 elseif op == "ACTIVITY"
560 account = get_account(books, words[2])
561 year = parse(Int, words[3])
562 debit = credit = Cents(0)
563 emit("ACTIVITY", account.code, account.name, year)
564 for month in 1:12
565 first = day_number(year, month, 1)
566 last = first + month_length(year, month) - 1
567 amounts = total_pair(aggregate_entries(books, first, last), account.code)
568 debit += amounts.debit
569 credit += amounts.credit
570 emit(" $year-$(lpad(string(month), 2, '0'))", "debit", money(amounts.debit), "credit", money(amounts.credit),
571 "balance", money(account_balance(books, account, last)))
572 end
573 emit(" total debit", money(debit), "credit", money(credit))
574 elseif op == "TRIAL"
575 day = checked_date(words[2])
576 totals = aggregate_entries(books, 0, day)
577 debit = credit = Cents(0)
578 emit("TRIAL", date_text(day))
579 for code in sorted_codes(books)
580 delta = difference(totals, code)
581 delta == 0 && continue
582 emit(" $code", books.accounts[code].name, delta > 0 ? "debit" : "credit", money(abs(delta)))
583 if delta > 0
584 debit += delta
585 else
586 credit -= delta
587 end
588 end
589 emit(" total debit", money(debit), "credit", money(credit))
590 elseif op == "INCOME"
591 first, last = checked_range(words[2], words[3])
592 totals = aggregate_entries(books, first, last; operating=true)
593 emit("INCOME", date_text(first), date_text(last))
594 revenue = section(books, "REVENUE", (REVENUE,), Cents(-1), totals)
595 expense = section(books, "EXPENSE", (EXPENSE,), Cents(1), totals)
596 emit(" NET INCOME", money(revenue - expense))
597 elseif op == "BALANCE"
598 day = checked_date(words[2])
599 totals = aggregate_entries(books, 0, day)
600 earnings = -sum((difference(totals, code) for code in keys(books.accounts)
601 if books.accounts[code].kind in (REVENUE, EXPENSE)); init=Cents(0))
602 emit("BALANCE", date_text(day))
603 assets = section(books, "ASSETS", (ASSET, CONTRA_ASSET), Cents(1), totals)
604 liabilities = section(books, "LIABILITIES", (LIABILITY,), Cents(-1), totals)
605 equity = section(books, "EQUITY", (EQUITY,), Cents(-1), totals; earnings=earnings)
606 if assets == liabilities + equity
607 emit(" check ok")
608 else
609 emit(" check mismatch", money(assets - liabilities - equity))
610 end
611 elseif op == "ROLLUP"
612 day = checked_date(words[3])
613 selected = [code for code in sorted_codes(books) if startswith(code, words[2])]
614 emit("ROLLUP", words[2], date_text(day))
615 if isempty(selected)
616 emit(" (none)")
617 return nothing
618 end
619 totals = aggregate_entries(books, 0, day)
620 debit = credit = Cents(0)
621 for code in selected
622 account = books.accounts[code]
623 depth = count(other -> other != code && startswith(code, other), selected)
624 subtotal = normal_sign(account.kind) * sum((difference(totals, other) for other in selected
625 if startswith(other, code)); init=Cents(0))
626 padding = repeat(" ", 2 * (1 + depth))
627 own = total_pair(totals, code)
628 emit("$padding$code", account.name, "own", money(normal_sign(account.kind) * (own.debit - own.credit)), "total", money(subtotal))
629 debit += own.debit
630 credit += own.credit
631 end
632 emit(" postings debit", money(debit), "credit", money(credit))
633 elseif op in ("AGING", "ITEMS")
634 account = get_account(books, words[2])
635 day = checked_date(words[end])
636 settlements = settlements_for(books, account, day)
637 if op == "ITEMS"
638 group = get(settlements, words[3], Settlement())
639 emit("ITEMS", account.code, account.name, words[3], date_text(day))
640 items = open_items(group)
641 for item in items
642 entry = books.entries[item.entry]
643 emit(" $(date_text(entry.day))", entry.id, "amount", money(item.original), "open", money(item.remaining), "age", day - entry.day)
644 end
645 balance = sum((item.remaining for item in items); init=Cents(0)) - group.unapplied
646 emit(" unapplied", money(group.unapplied), "balance", money(balance))
647 else
648 emit("AGING", account.code, account.name, date_text(day))
649 totals = zeros(Cents, 5)
650 for partner in sort!(collect(keys(settlements)))
651 group = settlements[partner]
652 items = open_items(group)
653 isempty(items) && group.unapplied == 0 && continue
654 columns = zeros(Cents, 5)
655 for item in items
656 age = day - books.entries[item.entry].day
657 bucket = age <= 30 ? 1 : age <= 60 ? 2 : age <= 90 ? 3 : 4
658 columns[bucket] += item.remaining
659 end
660 columns[5] = group.unapplied
661 totals .+= columns
662 emit(" $partner", aging_columns(columns))
663 end
664 emit(" total", aging_columns(totals))
665 end
666 elseif op == "ENTRY"
667 entry = books.entries[get_entry(books, words[2])]
668 emit("ENTRY", entry.id, date_text(entry.day), category_text(entry.category), entry.memo)
669 for line in entry.lines
670 detail = "$(line.debit ? "DR" : "CR") $(line.code) $(books.accounts[line.code].name) $(money(line.amount))"
671 line.partner !== nothing && (detail *= " " * line.partner)
672 emit(" " * detail)
673 end
674 emit(" total", money(entry.amount))
675 entry.reverses !== nothing && emit(" reverses", books.entries[entry.reverses].id)
676 entry.reversed_by !== nothing && emit(" reversed by", books.entries[entry.reversed_by].id)
677 elseif op == "DAYBOOK"
678 first, last = checked_range(words[2], words[3])
679 entry_count = 0
680 total = Cents(0)
681 emit("DAYBOOK", date_text(first), date_text(last))
682 for index in posting_order(books)
683 entry = books.entries[index]
684 first <= entry.day <= last || continue
685 entry_count += 1
686 total += entry.amount
687 emit(" $(date_text(entry.day))", entry.id, category_text(entry.category), money(entry.amount), entry.memo)
688 end
689 emit(" total entries", entry_count, "amount", money(total))
690 elseif op == "ACCOUNTS"
691 emit("ACCOUNTS")
692 isempty(books.accounts) && emit(" (none)")
693 totals = aggregate_entries(books)
694 for code in sorted_codes(books)
695 account = books.accounts[code]
696 emit(" $code", account.kind, account.name, "parent", parent_code(books, code),
697 "balance", money(normal_sign(account.kind) * difference(totals, code)), account.active ? "active" : "inactive")
698 end
699 elseif op == "TEMPLATES"
700 emit("TEMPLATES")
701 isempty(books.templates) && emit(" (none)")
702 for name in sort!(collect(keys(books.templates)))
703 pattern = books.templates[name]
704 emit(" $name", pattern.memo, "lines", length(pattern.lines), "amount", money(line_totals(pattern.lines).debit), "used", pattern.uses)
705 end
706 elseif op == "STATUS"
707 inactive = count(account -> !account.active, values(books.accounts))
708 open = books.draft === nothing ? "none" : books.draft.key
709 emit("STATUS accounts", length(books.accounts), "inactive", inactive, "entries", length(books.entries),
710 "templates", length(books.templates), "closed", closed_text(books), "open", open)
711 end
712 return nothing
713end
