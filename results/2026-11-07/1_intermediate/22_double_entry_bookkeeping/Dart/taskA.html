10enum AccountType {

13 equity('EQUITY'),
14 revenue('REVENUE'),

26int sum(Iterable<int> values) => values.fold(0, (total, value) => total + value);
27String pad(int value, int width) => value.toString().padLeft(width, '0');
28String money(int cents) => '${cents < 0 ? '-' : ''}${cents.abs() ~/ 100}.${pad(cents.abs() % 100, 2)}';
29int parseMoney(String text) => int.parse(text.substring(0, text.length - 3)) * 100 + int.parse(text.substring(text.length - 2));

31class BookDate {
32 static const lastNumber = 36524;
33 static const monthLengths = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
34 final int year, month, day;
35 const BookDate(this.year, this.month, this.day);
36
37 static int daysInMonth(int year, int month) => monthLengths[month - 1] +
38 (month == 2 && year % 4 == 0 && (year % 100 != 0 || year % 400 == 0) ? 1 : 0);
39
40 factory BookDate.parse(String text) {
41 final year = int.parse(text.substring(0, 4));
42 final month = int.parse(text.substring(5, 7));
43 final day = int.parse(text.substring(8));
44 if (year < 2000 || year > 2099 || month < 1 || month > 12 || day < 1 || day > daysInMonth(year, month)) {
45 throw CommandError('invalid date $text');
46 }
47 return BookDate(year, month, day);
48 }
49
50 int get number {
51 final previous = year - 1;
52 var days = 365 * (year - 2000) + previous ~/ 4 - 1999 ~/ 4 - previous ~/ 100 + 1999 ~/ 100 + previous ~/ 400 - 1999 ~/ 400;
53 for (var m = 1; m < month; m++) {
54 days += daysInMonth(year, m);
55 }
56 return days + day - 1;
57 }
58
59 BookDate addMonths(int count) {
60 final offset = month - 1 + count;
61 final nextYear = year + offset ~/ 12;
62 final nextMonth = offset % 12 + 1;
63 return BookDate(nextYear, nextMonth, min(day, daysInMonth(nextYear, nextMonth)));
64 }
65
66 @override
67 String toString() => '${pad(year, 4)}-${pad(month, 2)}-${pad(day, 2)}';
68}

174 void checkClosed(BookDate date) {
175 if (closes.isNotEmpty && date.number <= closes.last.date.number) throw CommandError('period closed $date');

178 (BookDate, BookDate) range(String start, String end) {
179 final a = BookDate.parse(start);
180 final b = BookDate.parse(end);
181 if (a.number > b.number) throw CommandError('invalid range $start $end');
182 return (a, b);
183 }
184
185 List<Entry> ordered({int start = 0, int end = BookDate.lastNumber}) => entries
186 .where((entry) => entry.date.number >= start && entry.date.number <= end).toList()
187 ..sort((a, b) {
188 final date = a.date.number.compareTo(b.date.number);
189 return date == 0 ? a.sequence.compareTo(b.sequence) : date;
190 });
191
192 Map<String, Totals> aggregate({int start = 0, int end = BookDate.lastNumber, bool operating = false}) {
193 final result = <String, Totals>{};
194 for (final entry in entries) {
195 if (entry.date.number < start || entry.date.number > end ||
196 operating && (entry.kind == EntryKind.closing || entry.kind == EntryKind.reopen)) continue;
197 for (final line in entry.lines) {
198 result.putIfAbsent(line.code, Totals.new).add(line);
199 }
200 }
201 return result;
202 }
203
204 int balance(Account account, [int end = BookDate.lastNumber]) => sum(entries
205 .where((entry) => entry.date.number <= end)
206 .map((entry) => sum(entry.lines.where((line) => line.code == account.code).map((line) => line.difference)))) * account.sign;
207
208 int dailySum(Account account, int start, int end) {
209 var total = 0;
210 for (final entry in entries) {
211 if (entry.date.number > end) continue;
212 final change = sum(entry.lines.where((line) => line.code == account.code).map((line) => line.difference)) * account.sign;
213 total += change * (end - max(start, entry.date.number) + 1);
214 }
215 return total;
216 }

218 String parent(String code) {
219 var result = '';
220 for (final other in accounts.keys) {
221 if (other != code && code.startsWith(other) && other.length > result.length) result = other;
222 }
223 return result.isEmpty ? 'none' : result;
224 }

226 String top(String code) => accounts.keys.firstWhere((other) => code.startsWith(other));

228 Entry post(String id, BookDate date, EntryKind kind, String memo, Iterable<Line> lines) {
229 final posted = Entry(id, date, kind, memo, lines, entries.length + 1);
230 entries.add(posted);
231 entryIds[id] = posted;
232 return posted;
233 }

235 Entry reverse(Entry original, BookDate date, EntryKind kind, String memo) {
236 final posted = post('${original.id}-R', date, kind, memo, original.lines.map((line) => line.reversed));
237 original.reversedBy = posted.id;
238 posted.reverses = original.id;
239 return posted;
240 }

250 SplayTreeMap<String, Settlement> settlements(Account account, int date) {
251 final result = SplayTreeMap<String, Settlement>();
252 for (final entry in ordered(end: date)) {
253 for (final line in entry.lines) {
254 final partner = line.partner;
255 if (line.code != account.code || partner == null) continue;
256 final settlement = result.putIfAbsent(partner, Settlement.new);
257 final signed = line.difference * account.sign;
258 if (signed > 0) {
259 final applied = min(signed, settlement.unapplied);
260 settlement.unapplied -= applied;
261 if (signed > applied) settlement.items.add(OpenItem(entry, signed, signed - applied));
262 } else {
263 var remaining = -signed;
264 while (remaining > 0 && settlement.items.isNotEmpty) {
265 final item = settlement.items.first;
266 final applied = min(remaining, item.remaining);
267 item.remaining -= applied;
268 remaining -= applied;
269 if (item.remaining == 0) settlement.items.removeFirst();
270 }
271 settlement.unapplied += remaining;
272 }
273 }
274 }
275 return result;
276 }

278 int section(String heading, Set<AccountType> types, int sign, Map<String, Totals> totals, {int? earnings}) {
279 final groups = SplayTreeMap<String, List<Account>>();
280 int display(Account account) => (totals[account.code]?.difference ?? 0) * sign;
281 var total = earnings ?? 0;
282 for (final account in accounts.values.where((account) => types.contains(account.type))) {
283 groups.putIfAbsent(top(account.code), () => []).add(account);
284 total += display(account);
285 }
286 emit(' $heading ${money(total)}');
287 for (final group in groups.entries) {
288 if (!group.value.any((account) => display(account) != 0)) continue;
289 emit(' ${group.key} ${accounts[group.key]!.name} ${money(sum(group.value.map(display)))}');
290 for (final account in group.value) {
291 if (display(account) != 0) emit(' ${account.code} ${account.name} ${money(display(account))}');
292 }
293 }
294 if (earnings != null) emit(' current earnings ${money(earnings)}');
295 return total;
296 }

298 void postDraft() {
299 final block = draft ?? (throw const CommandError('no open entry'));
300 draft = null;
301 BookDate? date;
302 if (block.journal) {
303 if (entryIds.containsKey(block.key)) throw CommandError('duplicate entry ${block.key}');
304 date = BookDate.parse(block.date!);
305 checkClosed(date);
306 } else if (templates.containsKey(block.key)) {
307 throw CommandError('duplicate template ${block.key}');
308 }
309 if (block.lines.length < 2) throw CommandError('too few lines ${block.key}');
310 for (final line in block.lines) {
311 final found = account(line.code);
312 if (block.journal && !found.active) throw CommandError('inactive account ${line.code}');
313 if (line.amount == 0) throw CommandError('zero amount ${line.code}');
314 }
315 final total = lineTotals(block.lines);
316 if (total.debit != total.credit) {
317 throw CommandError('unbalanced ${block.key} debit ${money(total.debit)} credit ${money(total.credit)}');
318 }
319 if (block.journal) {
320 final entry = post(block.key, date!, EntryKind.journal, block.memo, block.lines);
321 emit('POST ${entry.id} $date lines ${entry.lines.length} amount ${money(entry.amount)}');
322 } else {
323 templates[block.key] = Template(block.memo, block.lines);
324 emit('TEMPLATE ${block.key} lines ${block.lines.length} amount ${money(total.debit)}');
325 }
326 }

328 void close(BookDate date, String code) {
329 checkClosed(date);
330 final equity = requireType(code, AccountType.equity);
331 active([equity]);
332 final totals = aggregate(end: date.number);
333 final targets = accounts.values.where((account) =>
334 (account.type == AccountType.revenue || account.type == AccountType.expense) &&
335 (totals[account.code]?.difference ?? 0) != 0).toList();
336 active(targets);
337 final lines = <Line>[];
338 var income = 0;
339 for (final account in targets) {
340 final difference = totals[account.code]!.difference;
341 lines.add(Line(difference < 0, account.code, difference.abs()));
342 income -= difference;
343 }
344 if (income != 0) lines.add(Line(income < 0, code, income.abs()));
345 closeCount++;
346 final entry = lines.isEmpty ? null : post('CLOSE-$closeCount', date, EntryKind.closing, 'closing', lines);
347 closes.add(CloseRecord(date, entry));
348 emit(entry == null ? 'CLOSE $date no entry' : 'CLOSE $date entry ${entry.id} lines ${lines.length} net income ${money(income)}');
349 }

351 void ledger(Account account, BookDate start, BookDate end) {
352 var running = balance(account, start.number - 1), debit = 0, credit = 0;
353 emit('LEDGER ${account.code} ${account.name} $start $end');
354 emit(' opening ${money(running)}');
355 for (final entry in ordered(start: start.number, end: end.number)) {
356 for (final line in entry.lines) {
357 if (line.code != account.code) continue;
358 running += line.difference * account.sign;
359 if (line.debit) {
360 debit += line.amount;
361 } else {
362 credit += line.amount;
363 }
364 emit(' ${entry.date} ${entry.id} ${line.side} ${money(line.amount)} balance ${money(running)} ${entry.memo}');
365 }
366 }
367 final days = end.number - start.number + 1;
368 emit(' ending ${money(running)} debits ${money(debit)} credits ${money(credit)} days $days average ${money(dailySum(account, start.number, end.number) ~/ days)}');
369 }

374 void execute(String text) {
375 final words = text.split(' ');
376 switch (words[0]) {
377 case 'ACCOUNT':
378 final code = words[1];
379 if (accounts.containsKey(code)) throw CommandError('duplicate account $code');
380 final type = AccountType.values.firstWhere((type) => type.label == words[2]);
381 for (final other in accounts.values) {
382 if ((code.startsWith(other.code) || other.code.startsWith(code)) && type.group != other.type.group) {
383 throw CommandError('class mismatch $code ${other.code}');
384 }
385 }
386 accounts[code] = Account(code, type, words[3]);
387 emit('ACCOUNT $code ${type.label} ${words[3]} parent ${parent(code)}');
388 return;
389 case 'DEACTIVATE':
390 final found = account(words[1]);
391 if (!found.active) throw CommandError('already inactive ${found.code}');
392 final current = balance(found);
393 if (current != 0) throw CommandError('nonzero balance ${found.code} ${money(current)}');
394 found.active = false;
395 emit('DEACTIVATE ${found.code} ${found.name}');
396 return;
397 case 'JOURNAL':
398 case 'TEMPLATE':
399 final previous = draft;
400 if (previous != null) emit('DISCARD ${previous.key} lines ${previous.lines.length}');
401 draft = words[0] == 'JOURNAL' ? Draft(true, words[1], words[2], words[3]) : Draft(false, words[1], null, words[2]);
402 return;
403 case 'DR':
404 case 'CR':
405 final block = draft ?? (throw const CommandError('no open entry'));
406 block.lines.add(Line(words[0] == 'DR', words[1], parseMoney(words[2]), words.length == 4 ? words[3] : null));
407 return;
408 case 'POST':
409 postDraft();
410 return;
411 case 'DRAFT':
412 final block = draft ?? (throw const CommandError('no open entry'));
413 final total = lineTotals(block.lines);
414 emit('DRAFT ${block.kind} ${block.key} lines ${block.lines.length} debit ${money(total.debit)} credit ${money(total.credit)} difference ${money(total.difference)}');
415 return;
416 case 'REVERSE':
417 final original = entry(words[1]);
418 if ({EntryKind.reversal, EntryKind.closing, EntryKind.reopen}.contains(original.kind)) throw CommandError('cannot reverse ${original.id}');
419 if (original.reversedBy != null) throw CommandError('already reversed ${original.id}');
420 final date = BookDate.parse(words[2]);
421 if (date.number < original.date.number) throw CommandError('reverse before entry ${original.id} ${original.date}');
422 checkClosed(date);
423 active(original.lines.map((line) => accounts[line.code]!));
424 final posted = reverse(original, date, EntryKind.reversal, original.memo);
425 emit('REVERSE ${original.id} entry ${posted.id} $date amount ${money(posted.amount)}');
426 return;
427 case 'RECUR':
428 final template = templates[words[1]] ?? (throw CommandError('unknown template ${words[1]}'));
429 final date = BookDate.parse(words[2]);
430 final count = int.parse(words[3]);
431 if (count == 0) throw const CommandError('invalid count');
432 checkClosed(date);
433 active(template.lines.map((line) => accounts[line.code]!));
434 emit('RECUR ${words[1]} count $count');
435 for (var month = 0; month < count; month++) {
436 final posted = post('${words[1]}-${++template.uses}', date.addMonths(month), EntryKind.recurring, template.memo, template.lines);
437 emit(' POST ${posted.id} ${posted.date} lines ${posted.lines.length} amount ${money(posted.amount)}');
438 }
439 return;
440 case 'DEPRECIATE':
441 final date = BookDate.parse(words[1]);
442 checkClosed(date);
443 final asset = requireType(words[2], AccountType.asset);
444 final contra = requireType(words[3], AccountType.contraAsset);
445 final expense = requireType(words[4], AccountType.expense);
446 active([contra, expense]);
447 final cost = balance(asset, date.number), accumulated = balance(contra, date.number);
448 final amount = cost > 0 ? min(cost ~/ int.parse(words[5]), cost - accumulated) : 0;
449 if (amount <= 0) throw CommandError('nothing to depreciate ${asset.code}');
450 final posted = post('DEP-${++depreciationCount}', date, EntryKind.depreciation, 'depreciation',
451 [Line(true, expense.code, amount), Line(false, contra.code, amount)]);
452 emit('DEPRECIATE ${asset.code} entry ${posted.id} $date amount ${money(amount)} accumulated ${money(accumulated + amount)} book ${money(cost - accumulated - amount)}');
453 return;
454 case 'INTEREST':
455 final found = account(words[1]);
456 if (found.type != AccountType.asset && found.type != AccountType.liability) throw CommandError('not interest bearing ${found.code}');
457 final offset = requireType(words[2], found.type == AccountType.asset ? AccountType.revenue : AccountType.expense);
458 final (start, end) = range(words[3], words[4]);
459 checkClosed(end);
460 active([found, offset]);
461 final daily = dailySum(found, start.number, end.number);
462 final amount = daily * int.parse(words[5]) ~/ 3650000;
463 if (amount <= 0) throw CommandError('no interest ${found.code}');
464 final debit = found.type == AccountType.asset ? found.code : offset.code;
465 final credit = found.type == AccountType.asset ? offset.code : found.code;
466 final posted = post('INT-${++interestCount}', end, EntryKind.interest, 'interest', [Line(true, debit, amount), Line(false, credit, amount)]);
467 final days = end.number - start.number + 1;
468 emit('INTEREST ${found.code} entry ${posted.id} $end days $days average ${money(daily ~/ days)} amount ${money(amount)}');
469 return;
470 case 'WRITEOFF':
471 final date = BookDate.parse(words[1]);
472 checkClosed(date);
473 final found = requireType(words[2], AccountType.asset);
474 final expense = requireType(words[3], AccountType.expense);
475 active([found, expense]);
476 final age = int.parse(words[4]);
477 final targets = <(String, int)>[];
478 for (final pair in settlements(found, date.number).entries) {
479 final amount = sum(pair.value.items.where((item) => date.number - item.entry.date.number > age).map((item) => item.remaining));
480 if (amount > 0) targets.add((pair.key, amount));
481 }
482 final total = sum(targets.map((target) => target.$2));
483 if (total == 0) throw CommandError('nothing to write off ${found.code}');
484 final lines = [Line(true, expense.code, total), for (final (partner, amount) in targets) Line(false, found.code, amount, partner)];
485 final posted = post('WO-${++writeoffCount}', date, EntryKind.writeoff, 'writeoff', lines);
486 emit('WRITEOFF ${found.code} entry ${posted.id} $date partners ${targets.length} amount ${money(total)}');
487 for (final (partner, amount) in targets) {
488 emit(' $partner ${money(amount)}');
489 }
490 return;
491 case 'CLOSE':
492 close(BookDate.parse(words[1]), words[2]);
493 return;
494 case 'REOPEN':
495 if (closes.isEmpty) throw const CommandError('nothing to reopen');
496 final record = closes.last;
497 final original = record.entry;
498 var result = 'no entry';
499 if (original != null) {
500 active(original.lines.map((line) => accounts[line.code]!));
501 result = 'entry ${reverse(original, record.date, EntryKind.reopen, 'reopen').id}';
502 }
503 closes.removeLast();
504 emit('REOPEN ${record.date} $result closed $closed');
505 return;
506 case 'LEDGER':
507 final found = account(words[1]);
508 final (start, end) = range(words[2], words[3]);
509 ledger(found, start, end);
510 return;
511 case 'ACTIVITY':
512 final found = account(words[1]);
513 final year = int.parse(words[2]);
514 var debit = 0, credit = 0;
515 emit('ACTIVITY ${found.code} ${found.name} $year');
516 for (var month = 1; month <= 12; month++) {
517 final start = BookDate(year, month, 1).number;
518 final end = BookDate(year, month, BookDate.daysInMonth(year, month)).number;
519 final total = aggregate(start: start, end: end)[found.code] ?? Totals();
520 debit += total.debit;
521 credit += total.credit;
522 emit(' ${pad(year, 4)}-${pad(month, 2)} debit ${money(total.debit)} credit ${money(total.credit)} balance ${money(balance(found, end))}');
523 }
524 emit(' total debit ${money(debit)} credit ${money(credit)}');
525 return;
526 case 'TRIAL':
527 final date = BookDate.parse(words[1]);
528 final totals = aggregate(end: date.number);
529 var debit = 0, credit = 0;
530 emit('TRIAL $date');
531 for (final account in accounts.values) {
532 final difference = totals[account.code]?.difference ?? 0;
533 if (difference == 0) continue;
534 if (difference > 0) { debit += difference; } else { credit -= difference; }
535 emit(' ${account.code} ${account.name} ${difference > 0 ? 'debit' : 'credit'} ${money(difference.abs())}');
536 }
537 emit(' total debit ${money(debit)} credit ${money(credit)}');
538 return;
539 case 'INCOME':
540 final (start, end) = range(words[1], words[2]);
541 final totals = aggregate(start: start.number, end: end.number, operating: true);
542 emit('INCOME $start $end');
543 final revenue = section('REVENUE', {AccountType.revenue}, -1, totals);
544 final expense = section('EXPENSE', {AccountType.expense}, 1, totals);
545 emit(' NET INCOME ${money(revenue - expense)}');
546 return;
547 case 'BALANCE':
548 final date = BookDate.parse(words[1]);
549 final totals = aggregate(end: date.number);
550 final earnings = -sum(accounts.values.where((account) => account.type == AccountType.revenue || account.type == AccountType.expense)
551 .map((account) => totals[account.code]?.difference ?? 0));
552 emit('BALANCE $date');
553 final assets = section('ASSETS', {AccountType.asset, AccountType.contraAsset}, 1, totals);
554 final liabilities = section('LIABILITIES', {AccountType.liability}, -1, totals);
555 final equity = section('EQUITY', {AccountType.equity}, -1, totals, earnings: earnings);
556 emit(assets == liabilities + equity ? ' check ok' : ' check mismatch ${money(assets - liabilities - equity)}');
557 return;
558 case 'ROLLUP':
559 final date = BookDate.parse(words[2]);
560 final targets = accounts.values.where((account) => account.code.startsWith(words[1])).toList();
561 emit('ROLLUP ${words[1]} $date');
562 if (targets.isEmpty) { emit(' (none)'); return; }
563 final totals = aggregate(end: date.number);
564 for (final account in targets) {
565 final depth = targets.where((other) => other.code != account.code && account.code.startsWith(other.code)).length;
566 final subtotal = sum(targets.where((other) => other.code.startsWith(account.code))
567 .map((other) => totals[other.code]?.difference ?? 0)) * account.sign;
568 emit('${' ' * (2 * (1 + depth))}${account.code} ${account.name} own ${money((totals[account.code]?.difference ?? 0) * account.sign)} total ${money(subtotal)}');
569 }
570 emit(' postings debit ${money(sum(targets.map((account) => totals[account.code]?.debit ?? 0)))} credit ${money(sum(targets.map((account) => totals[account.code]?.credit ?? 0)))}');
571 return;
572 case 'AGING':
573 final found = account(words[1]);
574 final date = BookDate.parse(words[2]);
575 final total = List.filled(5, 0);
576 emit('AGING ${found.code} ${found.name} $date');
577 for (final pair in settlements(found, date.number).entries) {
578 final settlement = pair.value;
579 if (settlement.items.isEmpty && settlement.unapplied == 0) continue;
580 final values = List.filled(5, 0);
581 for (final item in settlement.items) {
582 final age = date.number - item.entry.date.number;
583 values[age <= 30 ? 0 : age <= 60 ? 1 : age <= 90 ? 2 : 3] += item.remaining;
584 }
585 values[4] = settlement.unapplied;
586 for (var i = 0; i < 5; i++) { total[i] += values[i]; }
587 emit(' ${pair.key} ${ageColumns(values)}');
588 }
589 emit(' total ${ageColumns(total)}');
590 return;
591 case 'ITEMS':
592 final found = account(words[1]);
593 final date = BookDate.parse(words[3]);
594 final settlement = settlements(found, date.number)[words[2]] ?? Settlement();
595 emit('ITEMS ${found.code} ${found.name} ${words[2]} $date');
596 for (final item in settlement.items) {
597 emit(' ${item.entry.date} ${item.entry.id} amount ${money(item.original)} open ${money(item.remaining)} age ${date.number - item.entry.date.number}');
598 }
599 emit(' unapplied ${money(settlement.unapplied)} balance ${money(sum(settlement.items.map((item) => item.remaining)) - settlement.unapplied)}');
600 return;
601 case 'ENTRY':
602 final found = entry(words[1]);
603 emit('ENTRY ${found.id} ${found.date} ${found.kind.name} ${found.memo}');
604 for (final line in found.lines) {
605 emit(' ${line.side} ${line.code} ${accounts[line.code]!.name} ${money(line.amount)}${line.partner == null ? '' : ' ${line.partner}'}');
606 }
607 emit(' total ${money(found.amount)}');
608 if (found.reverses != null) emit(' reverses ${found.reverses}');
609 if (found.reversedBy != null) emit(' reversed by ${found.reversedBy}');
610 return;
611 case 'DAYBOOK':
612 final (start, end) = range(words[1], words[2]);
613 final selected = ordered(start: start.number, end: end.number);
614 emit('DAYBOOK $start $end');
615 for (final entry in selected) { emit(' ${entry.date} ${entry.id} ${entry.kind.name} ${money(entry.amount)} ${entry.memo}'); }
616 emit(' total entries ${selected.length} amount ${money(sum(selected.map((entry) => entry.amount)))}');
617 return;
618 case 'ACCOUNTS':
619 final totals = aggregate();
620 emit('ACCOUNTS');
621 for (final account in accounts.values) {
622 emit(' ${account.code} ${account.type.label} ${account.name} parent ${parent(account.code)} balance ${money((totals[account.code]?.difference ?? 0) * account.sign)} ${account.active ? 'active' : 'inactive'}');
623 }
624 if (accounts.isEmpty) emit(' (none)');
625 return;
626 case 'TEMPLATES':
627 emit('TEMPLATES');
628 for (final pair in templates.entries) {
629 final template = pair.value;
630 emit(' ${pair.key} ${template.memo} lines ${template.lines.length} amount ${money(lineTotals(template.lines).debit)} used ${template.uses}');
631 }
632 if (templates.isEmpty) emit(' (none)');
633 return;
634 case 'STATUS':
635 emit('STATUS accounts ${accounts.length} inactive ${accounts.values.where((account) => !account.active).length} entries ${entries.length} templates ${templates.length} closed $closed open ${draft?.key ?? 'none'}');
636 return;
637 }
638 }
