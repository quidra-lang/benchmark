39final readonly class BookDate
40{
41 public const LAST_NUMBER = 36524;
42 private const MONTH_LENGTHS = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
43 public function __construct(public int $year, public int $month, public int $day) {}
44
45 public static function daysInMonth(int $year, int $month): int
46 {
47 $leap = $year % 4 === 0 && ($year % 100 !== 0 || $year % 400 === 0);
48 return self::MONTH_LENGTHS[$month - 1] + ($month === 2 && $leap ? 1 : 0);
49 }
50
51 public static function parse(string $text): self
52 {
53 $year = (int)substr($text, 0, 4);
54 $month = (int)substr($text, 5, 2);
55 $day = (int)substr($text, 8, 2);
56 if ($year < 2000 || $year > 2099 || $month < 1 || $month > 12 || $day < 1 || $day > self::daysInMonth($year, $month)) {
57 throw new CommandError("invalid date $text");
58 }
59 return new self($year, $month, $day);
60 }
61
62 public function number(): int
63 {
64 $prior = $this->year - 1;
65 $result = 365 * ($this->year - 2000) + intdiv($prior, 4) - intdiv(1999, 4)
66 - intdiv($prior, 100) + intdiv(1999, 100) + intdiv($prior, 400) - intdiv(1999, 400);
67 for ($month = 1; $month < $this->month; $month++) $result += self::daysInMonth($this->year, $month);
68 return $result + $this->day - 1;
69 }
70
71 public function addMonths(int $count): self
72 {
73 $offset = $this->month - 1 + $count;
74 $year = $this->year + intdiv($offset, 12);
75 $month = $offset % 12 + 1;
76 return new self($year, $month, min($this->day, self::daysInMonth($year, $month)));
77 }
78
79 public function __toString(): string { return sprintf('%04d-%02d-%02d', $this->year, $this->month, $this->day); }
80}

242 {
243 $difference = 0;
244 foreach ($this->entries as $entry) {
245 if ($entry->date->number() > $end) continue;
246 foreach ($entry->lines as $line) if ($line->code === $account->code) $difference += $line->difference();

251 private function dailySum(Account $account, int $start, int $end): int
252 {
253 $total = 0;
254 foreach ($this->entries as $entry) {
255 if ($entry->date->number() > $end) continue;
256 $change = 0;
257 foreach ($entry->lines as $line) if ($line->code === $account->code) $change += $line->difference();
258 $total += $change * $account->type->sign() * ($end - max($start, $entry->date->number()) + 1);
259 }
260 return $total;

305 private function settlements(Account $account, int $date): array
306 {
307 $result = [];
308 foreach ($this->orderedEntries(end: $date) as $entry) {
309 foreach ($entry->lines as $line) {
310 if ($line->code !== $account->code || $line->partner === null) continue;
311 $settlement = $result[$line->partner] ??= new Settlement();
312 $signed = $line->difference() * $account->type->sign();
313 if ($signed > 0) {
314 $applied = min($signed, $settlement->unapplied);
315 $settlement->unapplied -= $applied;
316 if ($signed > $applied) $settlement->items[] = new OpenItem($entry, $signed, $signed - $applied);
317 } else {
318 $remaining = -$signed;
319 while ($remaining > 0 && $settlement->head < count($settlement->items)) {
320 $item = $settlement->items[$settlement->head];
321 $applied = min($remaining, $item->remaining);
322 $item->remaining -= $applied;
323 $remaining -= $applied;
324 if ($item->remaining === 0) $settlement->head++;
325 }
326 $settlement->unapplied += $remaining;
327 }
328 }
329 }
330 ksort($result, SORT_STRING);
331 return $result;
332 }

334 private function section(string $heading, array $types, int $sign, array $totals, ?int $earnings = null): int
335 {
336 $groups = [];
337 $total = $earnings ?? 0;
338 foreach ($this->orderedAccounts() as $account) {
339 if (!in_array($account->type, $types, true)) continue;
340 $root = $this->top($account);
341 $groups[$root->code] ??= ['root' => $root, 'members' => [], 'total' => 0, 'visible' => false];
342 $amount = ($totals[$account->code] ?? new Totals())->difference() * $sign;
343 $groups[$root->code]['members'][] = [$account, $amount];
344 $groups[$root->code]['total'] += $amount;
345 $groups[$root->code]['visible'] = $groups[$root->code]['visible'] || $amount !== 0;
346 $total += $amount;
347 }
348 ksort($groups, SORT_STRING);
349 $this->emit(' %s %s', $heading, money($total));
350 foreach ($groups as $group) {
351 if (!$group['visible']) continue;
352 $root = $group['root'];
353 $this->emit(' %s %s %s', $root->code, $root->name, money($group['total']));
354 foreach ($group['members'] as [$account, $amount]) {
355 if ($amount !== 0) $this->emit(' %s %s %s', $account->code, $account->name, money($amount));
356 }
357 }
358 if ($earnings !== null) $this->emit(' current earnings %s', money($earnings));
359 return $total;
360 }

362 private function postDraft(): void
363 {
364 $block = $this->draft ?? throw new CommandError('no open entry');
365 $this->draft = null;
366 $date = null;
367 if ($block->journal) {
368 if (isset($this->entryIds[$block->key])) throw new CommandError("duplicate entry {$block->key}");
369 $date = BookDate::parse($block->date);
370 $this->checkClosed($date);
371 } elseif (isset($this->templates[$block->key])) {
372 throw new CommandError("duplicate template {$block->key}");
373 }
374 if (count($block->lines) < 2) throw new CommandError("too few lines {$block->key}");
375 foreach ($block->lines as $line) {
376 $account = $this->account($line->code);
377 if ($block->journal && !$account->active) throw new CommandError("inactive account {$line->code}");
378 if ($line->amount === 0) throw new CommandError("zero amount {$line->code}");
379 }
380 $totals = $this->lineTotals($block->lines);
381 if ($totals->debit !== $totals->credit) {
382 throw new CommandError(sprintf('unbalanced %s debit %s credit %s', $block->key, money($totals->debit), money($totals->credit)));
383 }
384 if ($block->journal) {
385 $entry = $this->post($block->key, $date, EntryKind::Journal, $block->memo, $block->lines);
386 $this->emit('POST %s %s lines %d amount %s', $entry->id, $date, count($entry->lines), money($entry->amount()));
387 } else {
388 $this->templates[$block->key] = new Template($block->key, $block->memo, $block->lines);
389 $this->emit('TEMPLATE %s lines %d amount %s', $block->key, count($block->lines), money($totals->debit));
390 }
391 }

393 private function close(BookDate $date, string $code): void
394 {
395 $this->checkClosed($date);
396 $equity = $this->requireType($code, AccountType::Equity);
397 $this->active([$equity]);
398 $totals = $this->aggregate(end: $date->number());
399 $targets = array_filter($this->orderedAccounts(), static fn(Account $account): bool =>
400 in_array($account->type, [AccountType::Revenue, AccountType::Expense], true) && ($totals[$account->code] ?? new Totals())->difference() !== 0);
401 $this->active($targets);
402 $lines = [];
403 $income = 0;
404 foreach ($targets as $account) {
405 $difference = $totals[$account->code]->difference();
406 $lines[] = new Line($difference < 0, $account->code, abs($difference));
407 $income -= $difference;
408 }
409 if ($income !== 0) $lines[] = new Line($income < 0, $code, abs($income));
410 $this->closeCount++;
411 $entry = $lines === [] ? null : $this->post('CLOSE-' . $this->closeCount, $date, EntryKind::Closing, 'closing', $lines);
412 $this->closes[] = new CloseRecord($date, $entry);
413 if ($entry === null) $this->emit('CLOSE %s no entry', $date);
414 else $this->emit('CLOSE %s entry %s lines %d net income %s', $date, $entry->id, count($lines), money($income));
415 }

417 private function ledger(Account $account, BookDate $start, BookDate $end): void
418 {
419 $running = $this->balance($account, $start->number() - 1);
420 $totals = new Totals();
421 $this->emit('LEDGER %s %s %s %s', $account->code, $account->name, $start, $end);
422 $this->emit(' opening %s', money($running));
423 foreach ($this->orderedEntries($start->number(), $end->number()) as $entry) {
424 foreach ($entry->lines as $line) {
425 if ($line->code !== $account->code) continue;
426 $running += $line->difference() * $account->type->sign();
427 $totals->add($line);
428 $this->emit(' %s %s %s %s balance %s %s', $entry->date, $entry->id, $line->side(), money($line->amount), money($running), $entry->memo);
429 }
430 }
431 $days = $end->number() - $start->number() + 1;
432 $this->emit(' ending %s debits %s credits %s days %d average %s', money($running), money($totals->debit), money($totals->credit),
433 $days, money(intdiv($this->dailySum($account, $start->number(), $end->number()), $days)));
434 }

443 public function execute(string $text): void
444 {
445 $words = explode(' ', $text);
446 switch ($words[0]) {
447 case 'ACCOUNT':
448 $code = $words[1];
449 if (isset($this->accounts[$code])) throw new CommandError("duplicate account $code");
450 $type = AccountType::from($words[2]);
451 foreach ($this->orderedAccounts() as $other) {
452 if ((str_starts_with($code, $other->code) || str_starts_with($other->code, $code)) && $type->group() !== $other->type->group()) {
453 throw new CommandError("class mismatch $code {$other->code}");
454 }
455 }
456 $account = new Account($code, $type, $words[3]);
457 $this->accounts[$code] = $account;
458 $this->emit('ACCOUNT %s %s %s parent %s', $code, $type->value, $account->name, $this->parent($account)?->code ?? 'none');
459 return;
460 case 'DEACTIVATE':
461 $account = $this->account($words[1]);
462 if (!$account->active) throw new CommandError("already inactive {$account->code}");
463 $balance = $this->balance($account);
464 if ($balance !== 0) throw new CommandError("nonzero balance {$account->code} " . money($balance));
465 $account->active = false;
466 $this->emit('DEACTIVATE %s %s', $account->code, $account->name);
467 return;
468 case 'JOURNAL':
469 case 'TEMPLATE':
470 if ($this->draft !== null) $this->emit('DISCARD %s lines %d', $this->draft->key, count($this->draft->lines));
471 $this->draft = $words[0] === 'JOURNAL' ? new Draft(true, $words[1], $words[2], $words[3]) : new Draft(false, $words[1], null, $words[2]);
472 return;
473 case 'DR':
474 case 'CR':
475 $block = $this->draft ?? throw new CommandError('no open entry');
476 $block->lines[] = new Line($words[0] === 'DR', $words[1], parseMoney($words[2]), $words[3] ?? null);
477 return;
478 case 'POST':
479 $this->postDraft();
480 return;
481 case 'DRAFT':
482 $block = $this->draft ?? throw new CommandError('no open entry');
483 $totals = $this->lineTotals($block->lines);
484 $this->emit('DRAFT %s %s lines %d debit %s credit %s difference %s', $block->kind(), $block->key, count($block->lines), money($totals->debit), money($totals->credit), money($totals->difference()));
485 return;
486 case 'REVERSE':
487 $original = $this->entry($words[1]);
488 if (in_array($original->kind, [EntryKind::Reversal, EntryKind::Closing, EntryKind::Reopen], true)) throw new CommandError("cannot reverse {$original->id}");
489 if ($original->reversedBy !== null) throw new CommandError("already reversed {$original->id}");
490 $date = BookDate::parse($words[2]);
491 if ($date->number() < $original->date->number()) throw new CommandError("reverse before entry {$original->id} {$original->date}");
492 $this->checkClosed($date);
493 $this->activeLines($original->lines);
494 $entry = $this->reverse($original, $date, EntryKind::Reversal, $original->memo);
495 $this->emit('REVERSE %s entry %s %s amount %s', $original->id, $entry->id, $date, money($entry->amount()));
496 return;
497 case 'RECUR':
498 $template = $this->templates[$words[1]] ?? throw new CommandError("unknown template {$words[1]}");
499 $date = BookDate::parse($words[2]);
500 $count = (int)$words[3];
501 if ($count === 0) throw new CommandError('invalid count');
502 $this->checkClosed($date);
503 $this->activeLines($template->lines);
504 $this->emit('RECUR %s count %d', $template->name, $count);
505 for ($month = 0; $month < $count; $month++) {
506 $entry = $this->post($template->name . '-' . ++$template->uses, $date->addMonths($month), EntryKind::Recurring, $template->memo, $template->lines);
507 $this->emit(' POST %s %s lines %d amount %s', $entry->id, $entry->date, count($entry->lines), money($entry->amount()));
508 }
509 return;
510 case 'DEPRECIATE':
511 $date = BookDate::parse($words[1]);
512 $this->checkClosed($date);
513 $asset = $this->requireType($words[2], AccountType::Asset);
514 $contra = $this->requireType($words[3], AccountType::ContraAsset);
515 $expense = $this->requireType($words[4], AccountType::Expense);
516 $this->active([$contra, $expense]);
517 $cost = $this->balance($asset, $date->number());
518 $accumulated = $this->balance($contra, $date->number());
519 $amount = $cost > 0 ? min(intdiv($cost, (int)$words[5]), $cost - $accumulated) : 0;
520 if ($amount <= 0) throw new CommandError("nothing to depreciate {$asset->code}");
521 $entry = $this->post('DEP-' . ++$this->depreciationCount, $date, EntryKind::Depreciation, 'depreciation',
522 [new Line(true, $expense->code, $amount), new Line(false, $contra->code, $amount)]);
523 $this->emit('DEPRECIATE %s entry %s %s amount %s accumulated %s book %s', $asset->code, $entry->id, $date, money($amount), money($accumulated + $amount), money($cost - $accumulated - $amount));
524 return;
525 case 'INTEREST':
526 $account = $this->account($words[1]);
527 if (!in_array($account->type, [AccountType::Asset, AccountType::Liability], true)) throw new CommandError("not interest bearing {$account->code}");
528 $offset = $this->requireType($words[2], $account->type === AccountType::Asset ? AccountType::Revenue : AccountType::Expense);
529 [$start, $end] = $this->range($words[3], $words[4]);
530 $this->checkClosed($end);
531 $this->active([$account, $offset]);
532 $sum = $this->dailySum($account, $start->number(), $end->number());
533 $amount = intdiv($sum * (int)$words[5], 3650000);
534 if ($amount <= 0) throw new CommandError("no interest {$account->code}");
535 $debit = $account->type === AccountType::Asset ? $account->code : $offset->code;
536 $credit = $account->type === AccountType::Asset ? $offset->code : $account->code;
537 $entry = $this->post('INT-' . ++$this->interestCount, $end, EntryKind::Interest, 'interest', [new Line(true, $debit, $amount), new Line(false, $credit, $amount)]);
538 $days = $end->number() - $start->number() + 1;
539 $this->emit('INTEREST %s entry %s %s days %d average %s amount %s', $account->code, $entry->id, $end, $days, money(intdiv($sum, $days)), money($amount));
540 return;
541 case 'WRITEOFF':
542 $date = BookDate::parse($words[1]);
543 $this->checkClosed($date);
544 $account = $this->requireType($words[2], AccountType::Asset);
545 $expense = $this->requireType($words[3], AccountType::Expense);
546 $this->active([$account, $expense]);
547 $targets = [];
548 $total = 0;
549 foreach ($this->settlements($account, $date->number()) as $partner => $settlement) {
550 $amount = 0;
551 foreach ($settlement->open() as $item) if ($date->number() - $item->entry->date->number() > (int)$words[4]) $amount += $item->remaining;
552 if ($amount > 0) { $targets[$partner] = $amount; $total += $amount; }
553 }
554 if ($total === 0) throw new CommandError("nothing to write off {$account->code}");
555 $lines = [new Line(true, $expense->code, $total)];
556 foreach ($targets as $partner => $amount) $lines[] = new Line(false, $account->code, $amount, $partner);
557 $entry = $this->post('WO-' . ++$this->writeoffCount, $date, EntryKind::Writeoff, 'writeoff', $lines);
558 $this->emit('WRITEOFF %s entry %s %s partners %d amount %s', $account->code, $entry->id, $date, count($targets), money($total));
559 foreach ($targets as $partner => $amount) $this->emit(' %s %s', $partner, money($amount));
560 return;
561 case 'CLOSE':
562 $this->close(BookDate::parse($words[1]), $words[2]);
563 return;
564 case 'REOPEN':
565 if ($this->closes === []) throw new CommandError('nothing to reopen');
566 $record = $this->closes[count($this->closes) - 1];
567 $detail = 'no entry';
568 if ($record->entry !== null) {
569 $this->activeLines($record->entry->lines);
570 $detail = 'entry ' . $this->reverse($record->entry, $record->date, EntryKind::Reopen, 'reopen')->id;
571 }
572 array_pop($this->closes);
573 $this->emit('REOPEN %s %s closed %s', $record->date, $detail, $this->closed());
574 return;
575 case 'LEDGER':
576 $account = $this->account($words[1]);
577 [$start, $end] = $this->range($words[2], $words[3]);
578 $this->ledger($account, $start, $end);
579 return;
580 case 'ACTIVITY':
581 $account = $this->account($words[1]);
582 $year = (int)$words[2];
583 $debit = $credit = 0;
584 $this->emit('ACTIVITY %s %s %d', $account->code, $account->name, $year);
585 for ($month = 1; $month <= 12; $month++) {
586 $start = (new BookDate($year, $month, 1))->number();
587 $end = (new BookDate($year, $month, BookDate::daysInMonth($year, $month)))->number();
588 $totals = $this->aggregate($start, $end)[$account->code] ?? new Totals();
589 $debit += $totals->debit;
590 $credit += $totals->credit;
591 $this->emit(' %04d-%02d debit %s credit %s balance %s', $year, $month, money($totals->debit), money($totals->credit), money($this->balance($account, $end)));
592 }
593 $this->emit(' total debit %s credit %s', money($debit), money($credit));
594 return;
595 case 'TRIAL':
596 $date = BookDate::parse($words[1]);
597 $totals = $this->aggregate(end: $date->number());
598 $debit = $credit = 0;
599 $this->emit('TRIAL %s', $date);
600 foreach ($this->orderedAccounts() as $account) {
601 $difference = ($totals[$account->code] ?? new Totals())->difference();
602 if ($difference === 0) continue;
603 if ($difference > 0) $debit += $difference; else $credit -= $difference;
604 $this->emit(' %s %s %s %s', $account->code, $account->name, $difference > 0 ? 'debit' : 'credit', money(abs($difference)));
605 }
606 $this->emit(' total debit %s credit %s', money($debit), money($credit));
607 return;
608 case 'INCOME':
609 [$start, $end] = $this->range($words[1], $words[2]);
610 $totals = $this->aggregate($start->number(), $end->number(), operating: true);
611 $this->emit('INCOME %s %s', $start, $end);
612 $revenue = $this->section('REVENUE', [AccountType::Revenue], -1, $totals);
613 $expense = $this->section('EXPENSE', [AccountType::Expense], 1, $totals);
614 $this->emit(' NET INCOME %s', money($revenue - $expense));
615 return;
616 case 'BALANCE':
617 $date = BookDate::parse($words[1]);
618 $totals = $this->aggregate(end: $date->number());
619 $earnings = 0;
620 foreach ($this->accounts as $account) {
621 if (in_array($account->type, [AccountType::Revenue, AccountType::Expense], true)) $earnings -= ($totals[$account->code] ?? new Totals())->difference();
622 }
623 $this->emit('BALANCE %s', $date);
624 $assets = $this->section('ASSETS', [AccountType::Asset, AccountType::ContraAsset], 1, $totals);
625 $liabilities = $this->section('LIABILITIES', [AccountType::Liability], -1, $totals);
626 $equity = $this->section('EQUITY', [AccountType::Equity], -1, $totals, $earnings);
627 if ($assets === $liabilities + $equity) $this->emit(' check ok');
628 else $this->emit(' check mismatch %s', money($assets - $liabilities - $equity));
629 return;
630 case 'ROLLUP':
631 $date = BookDate::parse($words[2]);
632 $targets = array_values(array_filter($this->orderedAccounts(), static fn(Account $account): bool => str_starts_with($account->code, $words[1])));
633 $this->emit('ROLLUP %s %s', $words[1], $date);
634 if ($targets === []) { $this->emit(' (none)'); return; }
635 $totals = $this->aggregate(end: $date->number());
636 $debit = $credit = 0;
637 foreach ($targets as $account) {
638 $depth = $subtotal = 0;
639 foreach ($targets as $other) {
640 if ($other->code !== $account->code && str_starts_with($account->code, $other->code)) $depth++;
641 if (str_starts_with($other->code, $account->code)) $subtotal += ($totals[$other->code] ?? new Totals())->difference();
642 }
643 $own = $totals[$account->code] ?? new Totals();
644 $debit += $own->debit;
645 $credit += $own->credit;
646 $this->emit('%s%s %s own %s total %s', str_repeat(' ', 2 * (1 + $depth)), $account->code, $account->name, money($own->difference() * $account->type->sign()), money($subtotal * $account->type->sign()));
647 }
648 $this->emit(' postings debit %s credit %s', money($debit), money($credit));
649 return;
650 case 'AGING':
651 $account = $this->account($words[1]);
652 $date = BookDate::parse($words[2]);
653 $total = array_fill(0, 5, 0);
654 $this->emit('AGING %s %s %s', $account->code, $account->name, $date);
655 foreach ($this->settlements($account, $date->number()) as $partner => $settlement) {
656 $items = $settlement->open();
657 if ($items === [] && $settlement->unapplied === 0) continue;
658 $values = array_fill(0, 5, 0);
659 foreach ($items as $item) {
660 $age = $date->number() - $item->entry->date->number();
661 $values[$age <= 30 ? 0 : ($age <= 60 ? 1 : ($age <= 90 ? 2 : 3))] += $item->remaining;
662 }
663 $values[4] = $settlement->unapplied;
664 for ($i = 0; $i < 5; $i++) $total[$i] += $values[$i];
665 $this->agingLine($partner, $values);
666 }
667 $this->agingLine('total', $total);
668 return;
669 case 'ITEMS':
670 $account = $this->account($words[1]);
671 $date = BookDate::parse($words[3]);
672 $settlement = $this->settlements($account, $date->number())[$words[2]] ?? new Settlement();
673 $this->emit('ITEMS %s %s %s %s', $account->code, $account->name, $words[2], $date);
674 $remaining = 0;
675 foreach ($settlement->open() as $item) {
676 $remaining += $item->remaining;
677 $this->emit(' %s %s amount %s open %s age %d', $item->entry->date, $item->entry->id, money($item->original), money($item->remaining), $date->number() - $item->entry->date->number());
678 }
679 $this->emit(' unapplied %s balance %s', money($settlement->unapplied), money($remaining - $settlement->unapplied));
680 return;
681 case 'ENTRY':
682 $entry = $this->entry($words[1]);
683 $this->emit('ENTRY %s %s %s %s', $entry->id, $entry->date, $entry->kind->value, $entry->memo);
684 foreach ($entry->lines as $line) {
685 $this->emit(' %s %s %s %s%s', $line->side(), $line->code, $this->accounts[$line->code]->name, money($line->amount), $line->partner === null ? '' : ' ' . $line->partner);
686 }
687 $this->emit(' total %s', money($entry->amount()));
688 if ($entry->reverses !== null) $this->emit(' reverses %s', $entry->reverses);
689 if ($entry->reversedBy !== null) $this->emit(' reversed by %s', $entry->reversedBy);
690 return;
691 case 'DAYBOOK':
692 [$start, $end] = $this->range($words[1], $words[2]);
693 $selected = $this->orderedEntries($start->number(), $end->number());
694 $this->emit('DAYBOOK %s %s', $start, $end);
695 $total = 0;
696 foreach ($selected as $entry) {
697 $total += $entry->amount();
698 $this->emit(' %s %s %s %s %s', $entry->date, $entry->id, $entry->kind->value, money($entry->amount()), $entry->memo);
699 }
700 $this->emit(' total entries %d amount %s', count($selected), money($total));
701 return;
702 case 'ACCOUNTS':
703 $totals = $this->aggregate();
704 $this->emit('ACCOUNTS');
705 foreach ($this->orderedAccounts() as $account) {
706 $this->emit(' %s %s %s parent %s balance %s %s', $account->code, $account->type->value, $account->name,
707 $this->parent($account)?->code ?? 'none', money(($totals[$account->code] ?? new Totals())->difference() * $account->type->sign()), $account->active ? 'active' : 'inactive');
708 }
709 if ($this->accounts === []) $this->emit(' (none)');
710 return;
711 case 'TEMPLATES':
712 $selected = $this->templates;
713 ksort($selected, SORT_STRING);
714 $this->emit('TEMPLATES');
715 foreach ($selected as $template) $this->emit(' %s %s lines %d amount %s used %d', $template->name, $template->memo, count($template->lines), money($this->lineTotals($template->lines)->debit), $template->uses);
716 if ($selected === []) $this->emit(' (none)');
717 return;
718 case 'STATUS':
719 $inactive = count(array_filter($this->accounts, static fn(Account $account): bool => !$account->active));
720 $this->emit('STATUS accounts %d inactive %d entries %d templates %d closed %s open %s', count($this->accounts), $inactive, count($this->entries), count($this->templates), $this->closed(), $this->draft?->key ?? 'none');
721 return;
722 }
723 }

726 {
727 try { $this->execute($line); }
728 catch (CommandError $error) { $this->emit('ERROR %s', $error->getMessage()); }
729 }
