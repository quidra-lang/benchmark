36 static boolean leap(int year) { return year % 4 == 0 && (year % 100 != 0 || year % 400 == 0); }
37 static int yearLength(int year) { return leap(year) ? 366 : 365; }
38
39 static int monthLength(int year, int month) {
40 int[] lengths = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
41 return month == 2 && leap(year) ? 29 : lengths[month - 1];
42 }
43
44 static int dayNumber(int year, int month, int day) {
45 int number = day - 1;
46 for (int y = 2000; y < year; y++) number += yearLength(y);
47 for (int m = 1; m < month; m++) number += monthLength(year, m);
48 return number;
49 }

51 record CalendarDate(int year, int month, int day) {}
52 record DayRange(int first, int last) {}

54 static CalendarDate calendar(int number) {
55 int year = 2000, month = 1;
56 while (number >= yearLength(year)) { number -= yearLength(year); year++; }
57 while (number >= monthLength(year, month)) { number -= monthLength(year, month); month++; }
58 return new CalendarDate(year, month, number + 1);
59 }
60
61 static String dateText(int number) {
62 CalendarDate date = calendar(number);
63 return String.format(Locale.ROOT, "%04d-%02d-%02d", date.year, date.month, date.day);
64 }

66 static int parseDate(String text) {
67 int year = Integer.parseInt(text.substring(0, 4));
68 int month = Integer.parseInt(text.substring(5, 7)), day = Integer.parseInt(text.substring(8));
69 if (year < 2000 || year > 2099 || month < 1 || month > 12 || day < 1 || day > monthLength(year, month)) {
70 throw new CommandError("invalid date " + text);
71 }
72 return dayNumber(year, month, day);
73 }
74
75 static DayRange dateRange(String first, String last) {
76 int start = parseDate(first), end = parseDate(last);
77 if (start > end) throw new CommandError("invalid range " + first + " " + last);
78 return new DayRange(start, end);
79 }

81 static int monthsAfter(int number, int offset) {
82 CalendarDate date = calendar(number);
83 int total = date.month - 1 + offset;
84 int year = date.year + total / 12, month = total % 12 + 1;
85 return dayNumber(year, month, Math.min(date.day, monthLength(year, month)));
86 }

100 record Totals(long debit, long credit) {
101 static final Totals ZERO = new Totals(0, 0);
102 long difference() { return debit - credit; }
103 Totals plus(Totals other) { return new Totals(debit + other.debit, credit + other.credit); }
104 Totals add(boolean isDebit, long amount) {
105 return isDebit ? new Totals(debit + amount, credit) : new Totals(debit, credit + amount);
106 }
107 }

263 Map<Account, Totals> totals(int first, int last, boolean operating) {
264 Map<Account, Totals> totals = new HashMap<>();
265 for (Entry entry : entries) {
266 if (entry.day < first || entry.day > last || operating && (entry.category == Category.CLOSING || entry.category == Category.REOPEN)) continue;

280 long dailySum(Account account, int first, int last) {
281 long[] changes = new long[last - first + 1];
282 long running = 0, sum = 0;
283 for (Entry entry : entries) {
284 if (entry.day > last) continue;
285 long value = account.type.sign() * entry.lines.totals(account).difference();
286 if (entry.day < first) running += value;
287 else changes[entry.day - first] += value;
288 }
289 for (long change : changes) { running += change; sum += running; }
290 return sum;
291 }

309 TreeMap<String, Settlement> settlements(Account account, int day) {
310 TreeMap<String, Settlement> settlements = new TreeMap<>();
311 for (Entry entry : ordered()) {
312 if (entry.day > day) continue;
313 for (Line line : entry.lines.items) {
314 if (line.account != account || line.partner == null) continue;
315 Settlement settled = settlements.computeIfAbsent(line.partner, ignored -> new Settlement());
316 long value = account.type.sign() * line.difference();
317 if (value > 0) {
318 long applied = Math.min(value, settled.unapplied);
319 settled.unapplied -= applied;
320 if (value > applied) settled.items.addLast(new OpenItem(entry, value, value - applied));
321 } else {
322 long payment = -value;
323 while (payment > 0 && !settled.items.isEmpty()) {
324 OpenItem item = settled.items.getFirst();
325 long applied = Math.min(payment, item.remaining);
326 payment -= applied;
327 item.remaining -= applied;
328 if (item.remaining == 0) settled.items.removeFirst();
329 }
330 settled.unapplied += payment;
331 }
332 }
333 }
334 return settlements;
335 }

337 void postDraft() {
338 if (draft == null) throw new CommandError("no open entry");
339 Draft block = draft;
340 draft = null;
341 int day = 0;
342 if (block.journal) {
343 if (entryIds.containsKey(block.key)) throw new CommandError("duplicate entry " + block.key);
344 day = parseDate(block.date);
345 checkOpen(day);
346 } else if (templates.containsKey(block.key)) throw new CommandError("duplicate template " + block.key);
347 if (block.lines.size() < 2) throw new CommandError("too few lines " + block.key);
348 List<Line> items = new ArrayList<>();
349 for (DraftLine line : block.lines) {
350 Account account = account(line.code);
351 if (block.journal) active(account);
352 if (line.amount == 0) throw new CommandError("zero amount " + line.code);
353 items.add(new Line(line.debit, account, line.amount, line.partner));
354 }
355 Totals totals = block.totals();
356 if (totals.debit != totals.credit) throw new CommandError("unbalanced " + block.key + " debit " + money(totals.debit) + " credit " + money(totals.credit));
357 Lines lines = new Lines(items);
358 if (block.journal) {
359 post(block.key, day, Category.JOURNAL, block.memo, lines);
360 emit("POST %s %s lines %d amount %s", block.key, dateText(day), items.size(), money(lines.amount));
361 } else {
362 templates.put(block.key, new Template(block.memo, lines));
363 emit("TEMPLATE %s lines %d amount %s", block.key, items.size(), money(lines.amount));
364 }
365 }

367 long section(String heading, Set<AccountType> types, int sign, Map<Account, Totals> totals, Long earnings) {
368 TreeMap<String, List<Displayed>> groups = new TreeMap<>();
369 long total = earnings == null ? 0 : earnings;
370 for (Account account : accounts.values()) {
371 if (!types.contains(account.type)) continue;
372 long value = sign * totals.getOrDefault(account, Totals.ZERO).difference();
373 groups.computeIfAbsent(top(account.code), ignored -> new ArrayList<>()).add(new Displayed(account, value));
374 total += value;
375 }
376 emit(" %s %s", heading, money(total));
377 for (var group : groups.entrySet()) {
378 long subtotal = 0;
379 boolean nonzero = false;
380 for (Displayed item : group.getValue()) { subtotal += item.amount; nonzero |= item.amount != 0; }
381 if (!nonzero) continue;
382 Account root = accounts.get(group.getKey());
383 emit(" %s %s %s", root.code, root.name, money(subtotal));
384 for (Displayed item : group.getValue()) {
385 if (item.amount != 0) emit(" %s %s %s", item.account.code, item.account.name, money(item.amount));
386 }
387 }
388 if (earnings != null) emit(" current earnings %s", money(earnings));
389 return total;
390 }

398 void execute(String[] words) {
399 switch (words[0]) {
400 case "ACCOUNT" -> {
401 String code = words[1];
402 if (accounts.containsKey(code)) throw new CommandError("duplicate account " + code);
403 AccountType type = AccountType.valueOf(words[2]);
404 for (Account other : accounts.values()) {
405 if ((code.startsWith(other.code) || other.code.startsWith(code)) && type.typeClass() != other.type.typeClass()) {
406 throw new CommandError("class mismatch " + code + " " + other.code);
407 }
408 }
409 accounts.put(code, new Account(code, type, words[3]));
410 emit("ACCOUNT %s %s %s parent %s", code, type, words[3], parent(code));
411 }
412 case "DEACTIVATE" -> {
413 Account account = account(words[1]);
414 if (!account.active) throw new CommandError("already inactive " + account.code);
415 long value = balance(account, LAST_DAY);
416 if (value != 0) throw new CommandError("nonzero balance " + account.code + " " + money(value));
417 account.active = false;
418 emit("DEACTIVATE %s %s", account.code, account.name);
419 }
420 case "JOURNAL", "TEMPLATE" -> {
421 if (draft != null) emit("DISCARD %s lines %d", draft.key, draft.lines.size());
422 boolean journal = words[0].equals("JOURNAL");
423 draft = new Draft(journal, words[1], journal ? words[2] : null, words[journal ? 3 : 2]);
424 }
425 case "DR", "CR" -> {
426 if (draft == null) throw new CommandError("no open entry");
427 draft.lines.add(new DraftLine(words[0].equals("DR"), words[1], parseMoney(words[2]), words.length == 4 ? words[3] : null));
428 }
429 case "POST" -> postDraft();
430 case "DRAFT" -> {
431 if (draft == null) throw new CommandError("no open entry");
432 Totals totals = draft.totals();
433 emit("DRAFT %s %s lines %d debit %s credit %s difference %s", draft.journal ? "journal" : "template", draft.key,
434 draft.lines.size(), money(totals.debit), money(totals.credit), money(totals.difference()));
435 }
436 case "REVERSE" -> {
437 Entry original = entryIds.get(words[1]);
438 if (original == null) throw new CommandError("unknown entry " + words[1]);
439 if (original.category == Category.REVERSAL || original.category == Category.CLOSING || original.category == Category.REOPEN) {
440 throw new CommandError("cannot reverse " + original.id);
441 }
442 if (original.reversedBy != null) throw new CommandError("already reversed " + original.id);
443 int day = parseDate(words[2]);
444 if (day < original.day) throw new CommandError("reverse before entry " + original.id + " " + dateText(original.day));
445 checkOpen(day);
446 activeLines(original.lines);
447 Entry entry = reverse(original, day, Category.REVERSAL, original.memo);
448 emit("REVERSE %s entry %s %s amount %s", original.id, entry.id, dateText(day), money(entry.lines.amount));
449 }
450 case "RECUR" -> {
451 Template model = templates.get(words[1]);
452 if (model == null) throw new CommandError("unknown template " + words[1]);
453 int day = parseDate(words[2]), count = Integer.parseInt(words[3]);
454 if (count == 0) throw new CommandError("invalid count");
455 checkOpen(day);
456 activeLines(model.lines);
457 emit("RECUR %s count %d", words[1], count);
458 for (int offset = 0; offset < count; offset++) {
459 Entry entry = post(words[1] + "-" + ++model.uses, monthsAfter(day, offset), Category.RECURRING, model.memo, model.lines);
460 emit(" POST %s %s lines %d amount %s", entry.id, dateText(entry.day), entry.lines.items.size(), money(entry.lines.amount));
461 }
462 }
463 case "DEPRECIATE" -> {
464 int day = parseDate(words[1]);
465 checkOpen(day);
466 Account asset = requireType(words[2], AccountType.ASSET);
467 Account contra = requireType(words[3], AccountType.CONTRA_ASSET);
468 Account expense = requireType(words[4], AccountType.EXPENSE);
469 active(contra, expense);
470 long cost = balance(asset, day), accumulated = balance(contra, day);
471 long amount = cost > 0 ? Math.min(cost / Integer.parseInt(words[5]), cost - accumulated) : 0;
472 if (amount <= 0) throw new CommandError("nothing to depreciate " + asset.code);
473 Lines lines = new Lines(List.of(new Line(true, expense, amount), new Line(false, contra, amount)));
474 Entry entry = post("DEP-" + ++depreciationCount, day, Category.DEPRECIATION, "depreciation", lines);
475 emit("DEPRECIATE %s entry %s %s amount %s accumulated %s book %s", asset.code, entry.id, dateText(day),
476 money(amount), money(accumulated + amount), money(cost - accumulated - amount));
477 }
478 case "INTEREST" -> {
479 Account account = account(words[1]);
480 if (account.type != AccountType.ASSET && account.type != AccountType.LIABILITY) throw new CommandError("not interest bearing " + account.code);
481 Account offset = requireType(words[2], account.type == AccountType.ASSET ? AccountType.REVENUE : AccountType.EXPENSE);
482 DayRange range = dateRange(words[3], words[4]);
483 checkOpen(range.last);
484 active(account, offset);
485 long sum = dailySum(account, range.first, range.last), amount = sum * Integer.parseInt(words[5]) / 3650000;
486 if (amount <= 0) throw new CommandError("no interest " + account.code);
487 boolean asset = account.type == AccountType.ASSET;
488 Lines lines = new Lines(List.of(new Line(true, asset ? account : offset, amount), new Line(false, asset ? offset : account, amount)));
489 Entry entry = post("INT-" + ++interestCount, range.last, Category.INTEREST, "interest", lines);
490 int days = range.last - range.first + 1;
491 emit("INTEREST %s entry %s %s days %d average %s amount %s", account.code, entry.id, dateText(range.last), days, money(sum / days), money(amount));
492 }
493 case "WRITEOFF" -> {
494 int day = parseDate(words[1]);
495 checkOpen(day);
496 Account account = requireType(words[2], AccountType.ASSET), expense = requireType(words[3], AccountType.EXPENSE);
497 active(account, expense);
498 int age = Integer.parseInt(words[4]);
499 TreeMap<String, Long> targets = new TreeMap<>();
500 long total = 0;
501 for (var group : settlements(account, day).entrySet()) {
502 long amount = 0;
503 for (OpenItem item : group.getValue().items) if (day - item.entry.day > age) amount += item.remaining;
504 if (amount > 0) { targets.put(group.getKey(), amount); total += amount; }
505 }
506 if (total == 0) throw new CommandError("nothing to write off " + account.code);
507 List<Line> items = new ArrayList<>();
508 items.add(new Line(true, expense, total));
509 for (var target : targets.entrySet()) items.add(new Line(false, account, target.getValue(), target.getKey()));
510 Entry entry = post("WO-" + ++writeoffCount, day, Category.WRITEOFF, "writeoff", new Lines(items));
511 emit("WRITEOFF %s entry %s %s partners %d amount %s", account.code, entry.id, dateText(day), targets.size(), money(total));
512 for (var target : targets.entrySet()) emit(" %s %s", target.getKey(), money(target.getValue()));
513 }
514 case "CLOSE" -> {
515 int day = parseDate(words[1]);
516 checkOpen(day);
517 Account equity = requireType(words[2], AccountType.EQUITY);
518 active(equity);
519 Map<Account, Totals> totals = totals(0, day, false);
520 List<Account> targets = new ArrayList<>();
521 for (Account account : accounts.values()) {
522 if (account.type.temporary() && totals.getOrDefault(account, Totals.ZERO).difference() != 0) targets.add(account);
523 }
524 for (Account account : targets) active(account);
525 List<Line> items = new ArrayList<>();
526 long income = 0;
527 for (Account account : targets) {
528 long value = totals.get(account).difference();
529 items.add(new Line(value < 0, account, Math.abs(value)));
530 income -= value;
531 }
532 if (income != 0) items.add(new Line(income < 0, equity, Math.abs(income)));
533 closeCount++;
534 Entry entry = items.isEmpty() ? null : post("CLOSE-" + closeCount, day, Category.CLOSING, "closing", new Lines(items));
535 closes.addLast(new CloseRecord(day, entry));
536 if (entry == null) emit("CLOSE %s no entry", dateText(day));
537 else emit("CLOSE %s entry %s lines %d net income %s", dateText(day), entry.id, items.size(), money(income));
538 }
539 case "REOPEN" -> {
540 if (closes.isEmpty()) throw new CommandError("nothing to reopen");
541 CloseRecord record = closes.getLast();
542 String detail = "no entry";
543 if (record.entry != null) {
544 activeLines(record.entry.lines);
545 detail = "entry " + reverse(record.entry, record.day, Category.REOPEN, "reopen").id;
546 }
547 closes.removeLast();
548 emit("REOPEN %s %s closed %s", dateText(record.day), detail, closedText());
549 }
550 case "LEDGER" -> {
551 Account account = account(words[1]);
552 DayRange range = dateRange(words[2], words[3]);
553 long running = balance(account, range.first - 1);
554 Totals totals = Totals.ZERO;
555 emit("LEDGER %s %s %s %s", account.code, account.name, words[2], words[3]);
556 emit(" opening %s", money(running));
557 for (Entry entry : ordered()) {
558 if (entry.day < range.first || entry.day > range.last) continue;
559 for (Line line : entry.lines.items) {
560 if (line.account != account) continue;
561 totals = totals.add(line.debit, line.amount);
562 running += account.type.sign() * line.difference();
563 emit(" %s %s %s %s balance %s %s", dateText(entry.day), entry.id, line.side(), money(line.amount), money(running), entry.memo);
564 }
565 }
566 int days = range.last - range.first + 1;
567 emit(" ending %s debits %s credits %s days %d average %s", money(running), money(totals.debit), money(totals.credit),
568 days, money(dailySum(account, range.first, range.last) / days));
569 }
570 case "ACTIVITY" -> {
571 Account account = account(words[1]);
572 int year = Integer.parseInt(words[2]);
573 Totals totals = Totals.ZERO;
574 emit("ACTIVITY %s %s %d", account.code, account.name, year);
575 for (int month = 1; month <= 12; month++) {
576 int first = dayNumber(year, month, 1), last = dayNumber(year, month, monthLength(year, month));
577 Totals current = accountTotals(account, first, last);
578 totals = totals.plus(current);
579 emit(" %04d-%02d debit %s credit %s balance %s", year, month, money(current.debit), money(current.credit), money(balance(account, last)));
580 }
581 emit(" total debit %s credit %s", money(totals.debit), money(totals.credit));
582 }
583 case "TRIAL" -> {
584 int day = parseDate(words[1]);
585 Map<Account, Totals> totals = totals(0, day, false);
586 long debit = 0, credit = 0;
587 emit("TRIAL %s", words[1]);
588 for (Account account : accounts.values()) {
589 long value = totals.getOrDefault(account, Totals.ZERO).difference();
590 if (value == 0) continue;
591 if (value > 0) debit += value; else credit -= value;
592 emit(" %s %s %s %s", account.code, account.name, value > 0 ? "debit" : "credit", money(Math.abs(value)));
593 }
594 emit(" total debit %s credit %s", money(debit), money(credit));
595 }
596 case "INCOME" -> {
597 DayRange range = dateRange(words[1], words[2]);
598 Map<Account, Totals> totals = totals(range.first, range.last, true);
599 emit("INCOME %s %s", words[1], words[2]);
600 long revenue = section("REVENUE", Set.of(AccountType.REVENUE), -1, totals, null);
601 long expense = section("EXPENSE", Set.of(AccountType.EXPENSE), 1, totals, null);
602 emit(" NET INCOME %s", money(revenue - expense));
603 }
604 case "BALANCE" -> {
605 int day = parseDate(words[1]);
606 Map<Account, Totals> totals = totals(0, day, false);
607 long earnings = 0;
608 for (Account account : accounts.values()) if (account.type.temporary()) earnings -= totals.getOrDefault(account, Totals.ZERO).difference();
609 emit("BALANCE %s", words[1]);
610 long assets = section("ASSETS", Set.of(AccountType.ASSET, AccountType.CONTRA_ASSET), 1, totals, null);
611 long liabilities = section("LIABILITIES", Set.of(AccountType.LIABILITY), -1, totals, null);
612 long equity = section("EQUITY", Set.of(AccountType.EQUITY), -1, totals, earnings);
613 if (assets == liabilities + equity) emit(" check ok");
614 else emit(" check mismatch %s", money(assets - liabilities - equity));
615 }
616 case "ROLLUP" -> {
617 int day = parseDate(words[2]);
618 List<Account> targets = accounts.values().stream().filter(account -> account.code.startsWith(words[1])).toList();
619 emit("ROLLUP %s %s", words[1], words[2]);
620 if (targets.isEmpty()) { emit(" (none)"); return; }
621 Map<Account, Totals> totals = totals(0, day, false);
622 Totals postings = Totals.ZERO;
623 for (Account account : targets) {
624 int depth = 0;
625 long subtotal = 0;
626 for (Account other : targets) {
627 if (other != account && account.code.startsWith(other.code)) depth++;
628 if (other.code.startsWith(account.code)) subtotal += totals.getOrDefault(other, Totals.ZERO).difference();
629 }
630 Totals own = totals.getOrDefault(account, Totals.ZERO);
631 emit("%s%s %s own %s total %s", " ".repeat(2 * (1 + depth)), account.code, account.name,
632 money(account.type.sign() * own.difference()), money(account.type.sign() * subtotal));
633 postings = postings.plus(own);
634 }
635 emit(" postings debit %s credit %s", money(postings.debit), money(postings.credit));
636 }
637 case "AGING", "ITEMS" -> {
638 Account account = account(words[1]);
639 boolean aging = words[0].equals("AGING");
640 int day = parseDate(words[aging ? 2 : 3]);
641 TreeMap<String, Settlement> settlements = settlements(account, day);
642 if (aging) {
643 emit("AGING %s %s %s", account.code, account.name, words[2]);
644 long[] totals = new long[5];
645 for (var group : settlements.entrySet()) {
646 Settlement settled = group.getValue();
647 if (settled.items.isEmpty() && settled.unapplied == 0) continue;
648 long[] values = {0, 0, 0, 0, settled.unapplied};
649 for (OpenItem item : settled.items) {
650 int age = day - item.entry.day, bucket = age <= 30 ? 0 : age <= 60 ? 1 : age <= 90 ? 2 : 3;
651 values[bucket] += item.remaining;
652 }
653 agingRow(group.getKey(), values);
654 for (int i = 0; i < 5; i++) totals[i] += values[i];
655 }
656 agingRow("total", totals);
657 } else {
658 Settlement settled = settlements.getOrDefault(words[2], new Settlement());
659 emit("ITEMS %s %s %s %s", account.code, account.name, words[2], words[3]);
660 long remaining = 0;
661 for (OpenItem item : settled.items) {
662 emit(" %s %s amount %s open %s age %d", dateText(item.entry.day), item.entry.id, money(item.original), money(item.remaining), day - item.entry.day);
663 remaining += item.remaining;
664 }
665 emit(" unapplied %s balance %s", money(settled.unapplied), money(remaining - settled.unapplied));
666 }
667 }
668 case "ENTRY" -> {
669 Entry entry = entryIds.get(words[1]);
670 if (entry == null) throw new CommandError("unknown entry " + words[1]);
671 emit("ENTRY %s %s %s %s", entry.id, dateText(entry.day), entry.category.text(), entry.memo);
672 for (Line line : entry.lines.items) {
673 String partner = line.partner == null ? "" : " " + line.partner;
674 emit(" %s %s %s %s%s", line.side(), line.account.code, line.account.name, money(line.amount), partner);
675 }
676 emit(" total %s", money(entry.lines.amount));
677 if (entry.reverses != null) emit(" reverses %s", entry.reverses.id);
678 if (entry.reversedBy != null) emit(" reversed by %s", entry.reversedBy.id);
679 }
680 case "DAYBOOK" -> {
681 DayRange range = dateRange(words[1], words[2]);
682 long total = 0;
683 int count = 0;
684 emit("DAYBOOK %s %s", words[1], words[2]);
685 for (Entry entry : ordered()) {
686 if (entry.day < range.first || entry.day > range.last) continue;
687 total += entry.lines.amount;
688 count++;
689 emit(" %s %s %s %s %s", dateText(entry.day), entry.id, entry.category.text(), money(entry.lines.amount), entry.memo);
690 }
691 emit(" total entries %d amount %s", count, money(total));
692 }
693 case "ACCOUNTS" -> {
694 Map<Account, Totals> totals = totals(0, LAST_DAY, false);
695 emit("ACCOUNTS");
696 for (Account account : accounts.values()) {
697 emit(" %s %s %s parent %s balance %s %s", account.code, account.type, account.name, parent(account.code),
698 money(account.type.sign() * totals.getOrDefault(account, Totals.ZERO).difference()), account.active ? "active" : "inactive");
699 }
700 if (accounts.isEmpty()) emit(" (none)");
701 }
702 case "TEMPLATES" -> {
703 emit("TEMPLATES");
704 for (var item : templates.entrySet()) {
705 Template model = item.getValue();
706 emit(" %s %s lines %d amount %s used %d", item.getKey(), model.memo, model.lines.items.size(), money(model.lines.amount), model.uses);
707 }
708 if (templates.isEmpty()) emit(" (none)");
709 }
710 case "STATUS" -> {
711 long inactive = accounts.values().stream().filter(account -> !account.active).count();
712 emit("STATUS accounts %d inactive %d entries %d templates %d closed %s open %s", accounts.size(), inactive, entries.size(),
713 templates.size(), closedText(), draft == null ? "none" : draft.key);
714 }
715 }
716 }
717 }

727 }
