6def parse_money(text)
7 whole, fraction = text.split('.').map(&:to_i)
8 whole * 100 + fraction
9end
10
11def money(value)
12 magnitude = value.abs
13 format('%s%d.%02d', value.negative? ? '-' : '', magnitude / 100, magnitude % 100)
14end
15
16def trunc(numerator, denominator)
17 numerator.negative? ? -(numerator.abs / denominator) : numerator / denominator
18end

20def leap?(year)
21 (year % 4).zero? && (!(year % 100).zero? || (year % 400).zero?)
22end
23
24def year_length(year) = leap?(year) ? 366 : 365
25
26def month_length(year, month)
27 return 29 if month == 2 && leap?(year)
28
29 [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31][month - 1]
30end

32def day_number(year, month, day)
33 number = day - 1
34 (2000...year).each { |y| number += year_length(y) }
35 (1...month).each { |m| number += month_length(year, m) }
36 number
37end
38
39def calendar(number)
40 year = 2000
41 month = 1
42 while number >= year_length(year)
43 number -= year_length(year)
44 year += 1
45 end
46 while number >= month_length(year, month)
47 number -= month_length(year, month)
48 month += 1
49 end
50 [year, month, number + 1]
51end
52
53def date_text(number) = format('%04d-%02d-%02d', *calendar(number))
54
55def parse_date(text)
56 year, month, day = text.split('-').map(&:to_i)
57 unless (2000..2099).cover?(year) && (1..12).cover?(month) && day >= 1 && day <= month_length(year, month)
58 raise CommandError, "invalid date #{text}"
59 end
60 day_number(year, month, day)
61end
62
63def date_range(first, last)
64 start_day = parse_date(first)
65 end_day = parse_date(last)
66 raise CommandError, "invalid range #{first} #{last}" if start_day > end_day
67
68 start_day..end_day
69end
70
71def months_after(number, offset)
72 year, month, day = calendar(number)
73 total = month - 1 + offset
74 year += total / 12
75 month = total % 12 + 1
76 day_number(year, month, [day, month_length(year, month)].min)
77end

94 def difference = debit - credit
95 def +(other) = Totals.new(debit + other.debit, credit + other.credit)

164 @entries = []

194 def check_open(day)
195 raise CommandError, "period closed #{date_text(day)}" if !@closes.empty? && day <= @closes.last.day
196 end
197
198 def closed_text = @closes.empty? ? 'none' : date_text(@closes.last.day)
199 def ordered_accounts = @accounts.keys.sort.map { |code| @accounts[code] }
200 def ordered_entries = @entries.sort_by { |entry| [entry.day, entry.sequence] }

202 def parent(code)
203 (code.length - 1).downto(1) { |length| return code[0, length] if @accounts.key?(code[0, length]) }
204 'none'
205 end

207 def top(code)
208 (1...code.length).each { |length| return code[0, length] if @accounts.key?(code[0, length]) }
209 code
210 end

212 def totals(first, last, operating: false)
213 result = Hash.new(ZERO_TOTALS)
214 @entries.each do |entry|
215 next unless (first..last).cover?(entry.day)
216 next if operating && %i[closing reopen].include?(entry.category)
217
218 entry.lines.accounts.each { |found, amount| result[found] = result[found] + amount }
219 end
220 result
221 end

223 def account_totals(found, first, last)
224 result = ZERO_TOTALS
225 @entries.each { |entry| result += entry.lines.accounts[found] if (first..last).cover?(entry.day) }
226 result
227 end
228
229 def balance(found, day) = found.sign * account_totals(found, 0, day).difference
230
231 def daily_sum(found, first, last)
232 changes = Array.new(last - first + 1, 0)
233 running = 0
234 @entries.each do |entry|
235 next if entry.day > last
236
237 value = found.sign * entry.lines.accounts[found].difference
238 if entry.day < first
239 running += value
240 else
241 changes[entry.day - first] += value
242 end
243 end
244 sum = 0
245 changes.each { |change| running += change; sum += running }
246 sum
247 end

256 def reverse(original, day, category, memo)
257 lines = Lines.new(original.lines.items.map { |line| Line.new(!line.debit, line.account, line.amount, line.partner) })
258 entry = post("#{original.id}-R", day, category, memo, lines)
259 original.reversed_by = entry
260 entry.reverses = original
261 entry
262 end

264 def settlements(found, day)
265 result = {}
266 ordered_entries.each do |entry|
267 next if entry.day > day
268
269 entry.lines.items.each do |line|
270 next unless line.account.equal?(found) && !line.partner.nil?
271
272 settled = result[line.partner] ||= Settlement.new
273 value = found.sign * line.difference
274 if value.positive?
275 applied = [value, settled.unapplied].min
276 settled.unapplied -= applied
277 settled.items << OpenItem.new(entry, value, value - applied) if value > applied
278 else
279 payment = -value
280 while payment.positive? && !settled.items.empty?
281 item = settled.items.first
282 applied = [payment, item.remaining].min
283 payment -= applied
284 item.remaining -= applied
285 settled.items.shift if item.remaining.zero?
286 end
287 settled.unapplied += payment
288 end
289 end
290 end
291 result
292 end

294 def post_draft
295 raise CommandError, 'no open entry' unless @draft
296
297 block = @draft
298 @draft = nil
299 day = 0
300 if block.kind == :journal
301 raise CommandError, "duplicate entry #{block.key}" if @entry_ids.key?(block.key)
302
303 day = parse_date(block.date)
304 check_open(day)
305 elsif @templates.key?(block.key)
306 raise CommandError, "duplicate template #{block.key}"
307 end
308 raise CommandError, "too few lines #{block.key}" if block.lines.length < 2
309
310 items = block.lines.map do |line|
311 found = account(line.code)
312 active(found) if block.kind == :journal
313 raise CommandError, "zero amount #{line.code}" if line.amount.zero?
314
315 Line.new(line.debit, found, line.amount, line.partner)
316 end
317 amount = block.totals
318 unless amount.debit == amount.credit
319 raise CommandError, "unbalanced #{block.key} debit #{money(amount.debit)} credit #{money(amount.credit)}"
320 end
321 lines = Lines.new(items)
322 if block.kind == :journal
323 post(block.key, day, :journal, block.memo, lines)
324 emit("POST #{block.key} #{date_text(day)} lines #{items.length} amount #{money(lines.amount)}")
325 else
326 @templates[block.key] = Template.new(block.memo, lines)
327 emit("TEMPLATE #{block.key} lines #{items.length} amount #{money(lines.amount)}")
328 end
329 end

331 def section(heading, types, sign, amounts, earnings: nil)
332 groups = Hash.new { |hash, key| hash[key] = [] }
333 total = earnings || 0
334 ordered_accounts.each do |found|
335 next unless types.include?(found.type)
336
337 value = sign * amounts[found].difference
338 groups[top(found.code)] << [found, value]
339 total += value
340 end
341 emit(" #{heading} #{money(total)}")
342 groups.keys.sort.each do |root|
343 group = groups[root]
344 next unless group.any? { |_, amount| !amount.zero? }
345
346 emit(" #{root} #{@accounts[root].name} #{money(group.sum { |_, amount| amount })}")
347 group.each { |found, amount| emit(" #{found.code} #{found.name} #{money(amount)}") unless amount.zero? }
348 end
349 emit(" current earnings #{money(earnings)}") unless earnings.nil?
350 total
351 end

353 def aging_row(partner, values)
354 balance = values[0, 4].sum - values[4]
355 emit(" #{partner} 0-30 #{money(values[0])} 31-60 #{money(values[1])} 61-90 #{money(values[2])} " \
356 "91+ #{money(values[3])} unapplied #{money(values[4])} balance #{money(balance)}")
357 end

359 def execute(words)
360 case words[0]
361 when 'ACCOUNT'
362 code, type, name = words[1, 3]
363 raise CommandError, "duplicate account #{code}" if @accounts.key?(code)
364
365 type_class = type == 'CONTRA_ASSET' ? 'ASSET' : type
366 ordered_accounts.each do |other|
367 if (code.start_with?(other.code) || other.code.start_with?(code)) && type_class != other.type_class
368 raise CommandError, "class mismatch #{code} #{other.code}"
369 end
370 end
371 @accounts[code] = Account.new(code, type, name)
372 emit("ACCOUNT #{code} #{type} #{name} parent #{parent(code)}")
373 when 'DEACTIVATE'
374 found = account(words[1])
375 raise CommandError, "already inactive #{found.code}" unless found.active
376
377 value = balance(found, LAST_DAY)
378 raise CommandError, "nonzero balance #{found.code} #{money(value)}" unless value.zero?
379
380 found.active = false
381 emit("DEACTIVATE #{found.code} #{found.name}")
382 when 'JOURNAL', 'TEMPLATE'
383 emit("DISCARD #{@draft.key} lines #{@draft.lines.length}") if @draft
384 journal = words[0] == 'JOURNAL'
385 @draft = Draft.new(journal ? :journal : :template, words[1], journal ? words[2] : nil, words[journal ? 3 : 2], [])
386 when 'DR', 'CR'
387 raise CommandError, 'no open entry' unless @draft
388
389 @draft.lines << DraftLine.new(words[0] == 'DR', words[1], parse_money(words[2]), words[3])
390 when 'POST'
391 post_draft
392 when 'DRAFT'
393 raise CommandError, 'no open entry' unless @draft
394
395 amount = @draft.totals
396 emit("DRAFT #{@draft.kind} #{@draft.key} lines #{@draft.lines.length} debit #{money(amount.debit)} " \
397 "credit #{money(amount.credit)} difference #{money(amount.difference)}")
398 when 'REVERSE'
399 original = @entry_ids.fetch(words[1]) { raise CommandError, "unknown entry #{words[1]}" }
400 raise CommandError, "cannot reverse #{original.id}" if %i[reversal closing reopen].include?(original.category)
401 raise CommandError, "already reversed #{original.id}" if original.reversed_by
402
403 day = parse_date(words[2])
404 raise CommandError, "reverse before entry #{original.id} #{date_text(original.day)}" if day < original.day
405
406 check_open(day)
407 active_lines(original.lines)
408 entry = reverse(original, day, :reversal, original.memo)
409 emit("REVERSE #{original.id} entry #{entry.id} #{date_text(day)} amount #{money(entry.lines.amount)}")
410 when 'RECUR'
411 model = @templates.fetch(words[1]) { raise CommandError, "unknown template #{words[1]}" }
412 day = parse_date(words[2])
413 count = words[3].to_i
414 raise CommandError, 'invalid count' if count.zero?
415
416 check_open(day)
417 active_lines(model.lines)
418 emit("RECUR #{words[1]} count #{count}")
419 count.times do |offset|
420 model.uses += 1
421 entry = post("#{words[1]}-#{model.uses}", months_after(day, offset), :recurring, model.memo, model.lines)
422 emit(" POST #{entry.id} #{date_text(entry.day)} lines #{entry.lines.items.length} amount #{money(entry.lines.amount)}")
423 end
424 when 'DEPRECIATE'
425 day = parse_date(words[1])
426 check_open(day)
427 asset = require_type(words[2], 'ASSET')
428 contra = require_type(words[3], 'CONTRA_ASSET')
429 expense = require_type(words[4], 'EXPENSE')
430 active(contra, expense)
431 cost = balance(asset, day)
432 accumulated = balance(contra, day)
433 amount = cost.positive? ? [cost / words[5].to_i, cost - accumulated].min : 0
434 raise CommandError, "nothing to depreciate #{asset.code}" unless amount.positive?
435
436 lines = Lines.new([Line.new(true, expense, amount, nil), Line.new(false, contra, amount, nil)])
437 @depreciation_count += 1
438 entry = post("DEP-#{@depreciation_count}", day, :depreciation, 'depreciation', lines)
439 emit("DEPRECIATE #{asset.code} entry #{entry.id} #{date_text(day)} amount #{money(amount)} " \
440 "accumulated #{money(accumulated + amount)} book #{money(cost - accumulated - amount)}")
441 when 'INTEREST'
442 found = account(words[1])
443 raise CommandError, "not interest bearing #{found.code}" unless %w[ASSET LIABILITY].include?(found.type)
444
445 offset = require_type(words[2], found.type == 'ASSET' ? 'REVENUE' : 'EXPENSE')
446 range = date_range(words[3], words[4])
447 check_open(range.end)
448 active(found, offset)
449 sum = daily_sum(found, range.begin, range.end)
450 amount = trunc(sum * words[5].to_i, 3_650_000)
451 raise CommandError, "no interest #{found.code}" unless amount.positive?
452
453 debit, credit = found.type == 'ASSET' ? [found, offset] : [offset, found]
454 lines = Lines.new([Line.new(true, debit, amount, nil), Line.new(false, credit, amount, nil)])
455 @interest_count += 1
456 entry = post("INT-#{@interest_count}", range.end, :interest, 'interest', lines)
457 days = range.end - range.begin + 1
458 emit("INTEREST #{found.code} entry #{entry.id} #{date_text(range.end)} days #{days} average #{money(trunc(sum, days))} amount #{money(amount)}")
459 when 'WRITEOFF'
460 day = parse_date(words[1])
461 check_open(day)
462 found = require_type(words[2], 'ASSET')
463 expense = require_type(words[3], 'EXPENSE')
464 active(found, expense)
465 age = words[4].to_i
466 settled = settlements(found, day)
467 targets = {}
468 settled.keys.sort.each do |partner|
469 amount = settled[partner].items.sum { |item| day - item.entry.day > age ? item.remaining : 0 }
470 targets[partner] = amount if amount.positive?
471 end
472 total = targets.values.sum
473 raise CommandError, "nothing to write off #{found.code}" if total.zero?
474
475 items = [Line.new(true, expense, total, nil)]
476 targets.each { |partner, amount| items << Line.new(false, found, amount, partner) }
477 @writeoff_count += 1
478 entry = post("WO-#{@writeoff_count}", day, :writeoff, 'writeoff', Lines.new(items))
479 emit("WRITEOFF #{found.code} entry #{entry.id} #{date_text(day)} partners #{targets.length} amount #{money(total)}")
480 targets.each { |partner, amount| emit(" #{partner} #{money(amount)}") }
481 when 'CLOSE'
482 day = parse_date(words[1])
483 check_open(day)
484 equity = require_type(words[2], 'EQUITY')
485 active(equity)
486 amounts = totals(0, day)
487 targets = ordered_accounts.select { |found| found.temporary? && !amounts[found].difference.zero? }
488 targets.each { |found| active(found) }
489 income = 0
490 items = targets.map do |found|
491 value = amounts[found].difference
492 income -= value
493 Line.new(value.negative?, found, value.abs, nil)
494 end
495 items << Line.new(income.negative?, equity, income.abs, nil) unless income.zero?
496 @close_count += 1
497 entry = items.empty? ? nil : post("CLOSE-#{@close_count}", day, :closing, 'closing', Lines.new(items))
498 @closes << CloseRecord.new(day, entry)
499 if entry
500 emit("CLOSE #{date_text(day)} entry #{entry.id} lines #{items.length} net income #{money(income)}")
501 else
502 emit("CLOSE #{date_text(day)} no entry")
503 end
504 when 'REOPEN'
505 raise CommandError, 'nothing to reopen' if @closes.empty?
506
507 record = @closes.last
508 detail = 'no entry'
509 if record.entry
510 active_lines(record.entry.lines)
511 detail = "entry #{reverse(record.entry, record.day, :reopen, 'reopen').id}"
512 end
513 @closes.pop
514 emit("REOPEN #{date_text(record.day)} #{detail} closed #{closed_text}")
515 when 'LEDGER'
516 found = account(words[1])
517 range = date_range(words[2], words[3])
518 running = balance(found, range.begin - 1)
519 amounts = ZERO_TOTALS
520 emit("LEDGER #{found.code} #{found.name} #{words[2]} #{words[3]}")
521 emit(" opening #{money(running)}")
522 ordered_entries.each do |entry|
523 next unless range.cover?(entry.day)
524
525 entry.lines.items.each do |line|
526 next unless line.account.equal?(found)
527
528 amounts = amounts.add(line.debit, line.amount)
529 running += found.sign * line.difference
530 emit(" #{date_text(entry.day)} #{entry.id} #{line.side} #{money(line.amount)} balance #{money(running)} #{entry.memo}")
531 end
532 end
533 days = range.end - range.begin + 1
534 emit(" ending #{money(running)} debits #{money(amounts.debit)} credits #{money(amounts.credit)} days #{days} " \
535 "average #{money(trunc(daily_sum(found, range.begin, range.end), days))}")
536 when 'ACTIVITY'
537 found = account(words[1])
538 year = words[2].to_i
539 amounts = ZERO_TOTALS
540 emit("ACTIVITY #{found.code} #{found.name} #{year}")
541 (1..12).each do |month|
542 first = day_number(year, month, 1)
543 last = day_number(year, month, month_length(year, month))
544 current = account_totals(found, first, last)
545 amounts += current
546 emit(" #{format('%04d-%02d', year, month)} debit #{money(current.debit)} credit #{money(current.credit)} balance #{money(balance(found, last))}")
547 end
548 emit(" total debit #{money(amounts.debit)} credit #{money(amounts.credit)}")
549 when 'TRIAL'
550 day = parse_date(words[1])
551 amounts = totals(0, day)
552 debit = credit = 0
553 emit("TRIAL #{words[1]}")
554 ordered_accounts.each do |found|
555 value = amounts[found].difference
556 next if value.zero?
557
558 if value.positive? then debit += value else credit -= value end
559 emit(" #{found.code} #{found.name} #{value.positive? ? 'debit' : 'credit'} #{money(value.abs)}")
560 end
561 emit(" total debit #{money(debit)} credit #{money(credit)}")
562 when 'INCOME'
563 range = date_range(words[1], words[2])
564 amounts = totals(range.begin, range.end, operating: true)
565 emit("INCOME #{words[1]} #{words[2]}")
566 revenue = section('REVENUE', ['REVENUE'], -1, amounts)
567 expense = section('EXPENSE', ['EXPENSE'], 1, amounts)
568 emit(" NET INCOME #{money(revenue - expense)}")
569 when 'BALANCE'
570 day = parse_date(words[1])
571 amounts = totals(0, day)
572 earnings = -ordered_accounts.sum { |found| found.temporary? ? amounts[found].difference : 0 }
573 emit("BALANCE #{words[1]}")
574 assets = section('ASSETS', %w[ASSET CONTRA_ASSET], 1, amounts)
575 liabilities = section('LIABILITIES', ['LIABILITY'], -1, amounts)
576 equity = section('EQUITY', ['EQUITY'], -1, amounts, earnings: earnings)
577 emit(assets == liabilities + equity ? ' check ok' : " check mismatch #{money(assets - liabilities - equity)}")
578 when 'ROLLUP'
579 day = parse_date(words[2])
580 targets = ordered_accounts.select { |found| found.code.start_with?(words[1]) }
581 emit("ROLLUP #{words[1]} #{words[2]}")
582 if targets.empty?
583 emit(' (none)')
584 return
585 end
586 amounts = totals(0, day)
587 postings = ZERO_TOTALS
588 targets.each do |found|
589 depth = targets.count { |other| !other.equal?(found) && found.code.start_with?(other.code) }
590 subtotal = targets.sum { |other| other.code.start_with?(found.code) ? amounts[other].difference : 0 }
591 own = amounts[found]
592 emit("#{' ' * (2 * (1 + depth))}#{found.code} #{found.name} own #{money(found.sign * own.difference)} total #{money(found.sign * subtotal)}")
593 postings += own
594 end
595 emit(" postings debit #{money(postings.debit)} credit #{money(postings.credit)}")
596 when 'AGING', 'ITEMS'
597 found = account(words[1])
598 aging = words[0] == 'AGING'
599 day = parse_date(words[aging ? 2 : 3])
600 settled = settlements(found, day)
601 if aging
602 emit("AGING #{found.code} #{found.name} #{words[2]}")
603 columns = Array.new(5, 0)
604 settled.keys.sort.each do |partner|
605 group = settled[partner]
606 next if group.items.empty? && group.unapplied.zero?
607
608 values = [0, 0, 0, 0, group.unapplied]
609 group.items.each do |item|
610 age = day - item.entry.day
611 bucket = if age <= 30 then 0 elsif age <= 60 then 1 elsif age <= 90 then 2 else 3 end
612 values[bucket] += item.remaining
613 end
614 aging_row(partner, values)
615 5.times { |i| columns[i] += values[i] }
616 end
617 aging_row('total', columns)
618 else
619 group = settled[words[2]] || Settlement.new
620 emit("ITEMS #{found.code} #{found.name} #{words[2]} #{words[3]}")
621 remaining = 0
622 group.items.each do |item|
623 emit(" #{date_text(item.entry.day)} #{item.entry.id} amount #{money(item.original)} open #{money(item.remaining)} age #{day - item.entry.day}")
624 remaining += item.remaining
625 end
626 emit(" unapplied #{money(group.unapplied)} balance #{money(remaining - group.unapplied)}")
627 end
628 when 'ENTRY'
629 entry = @entry_ids.fetch(words[1]) { raise CommandError, "unknown entry #{words[1]}" }
630 emit("ENTRY #{entry.id} #{date_text(entry.day)} #{entry.category} #{entry.memo}")
631 entry.lines.items.each do |line|
632 partner = line.partner.nil? ? '' : " #{line.partner}"
633 emit(" #{line.side} #{line.account.code} #{line.account.name} #{money(line.amount)}#{partner}")
634 end
635 emit(" total #{money(entry.lines.amount)}")
636 emit(" reverses #{entry.reverses.id}") if entry.reverses
637 emit(" reversed by #{entry.reversed_by.id}") if entry.reversed_by
638 when 'DAYBOOK'
639 range = date_range(words[1], words[2])
640 total = count = 0
641 emit("DAYBOOK #{words[1]} #{words[2]}")
642 ordered_entries.each do |entry|
643 next unless range.cover?(entry.day)
644
645 total += entry.lines.amount
646 count += 1
647 emit(" #{date_text(entry.day)} #{entry.id} #{entry.category} #{money(entry.lines.amount)} #{entry.memo}")
648 end
649 emit(" total entries #{count} amount #{money(total)}")
650 when 'ACCOUNTS'
651 amounts = totals(0, LAST_DAY)
652 emit('ACCOUNTS')
653 ordered_accounts.each do |found|
654 emit(" #{found.code} #{found.type} #{found.name} parent #{parent(found.code)} " \
655 "balance #{money(found.sign * amounts[found].difference)} #{found.active ? 'active' : 'inactive'}")
656 end
657 emit(' (none)') if @accounts.empty?
658 when 'TEMPLATES'
659 emit('TEMPLATES')
660 @templates.keys.sort.each do |name|
661 model = @templates[name]
662 emit(" #{name} #{model.memo} lines #{model.lines.items.length} amount #{money(model.lines.amount)} used #{model.uses}")
663 end
664 emit(' (none)') if @templates.empty?
665 when 'STATUS'
666 inactive = @accounts.values.count { |found| !found.active }
667 emit("STATUS accounts #{@accounts.length} inactive #{inactive} entries #{@entries.length} templates #{@templates.length} " \
668 "closed #{closed_text} open #{@draft ? @draft.key : 'none'}")
669 end
670 end
671end
