33int day_number(const CalendarDate date)
34 prior := date.year - 1
35 result := 365 * prior + prior / 4 - prior / 100 + prior / 400 + date.day - 730120
36 for: 1 <= month < date.month
37 result += month_length(date.year, month)

40CalendarDate calendar_date(int number)
41 year := 2000
42 while number >= 365 + (if leap(year) then 1 else 0)
43 number -= 365 + (if leap(year) then 1 else 0)
44 year += 1
45 month := 1
46 while number >= month_length(year, month)
47 number -= month_length(year, month)
48 month += 1
49 return CalendarDate(year, month, number + 1)

240map<string, Settlement> settlements_for(const Books books, const Account account, int day)
241 map<string, Settlement> settlements
242 for index := posting_order(books)[]
243 posted := books.entries[index]
244 if posted.day > day
245 break
246 for line := posted.lines[]
247 if line.code != account.code
248 continue
249 string partner
250 match line.partner
251 string supplied
252 partner = supplied
253 none
254 continue
255 group := get_settlement(settlements, partner)
256 signed := account.sign * line_difference(line)
257 if signed > 0
258 used := math.min(signed, group.unapplied)
259 group.unapplied -= used
260 if signed > used
261 group.items = group.items.append(OpenItem(index, signed, signed - used))
262 else
263 remaining := -signed
264 while remaining > 0 and group.head < int(#group.items)
265 used := math.min(remaining, group.items[group.head].remaining)
266 group.items[group.head].remaining -= used
267 remaining -= used
268 if group.items[group.head].remaining == 0
269 group.head += 1
270 group.unapplied += remaining
271 .settlements.set(partner, group)

337void | error command(Books &books, const string[] words)
338 op := words[0]
339 if op == "ACCOUNT"
340 code := words[1]
341 if books.accounts.has(code)
342 return error("duplicate account " + code)
343 account := Account(code, words[2], words[3])
344 for other := books.accounts.keys().sorted()[]
345 existing := get_account(books, other)
346 if (code.starts_with(other) or other.starts_with(code)) and account.family != existing.family
347 return error("class mismatch " + code + " " + other)
348 .books.accounts.set(code, account)
349 emit("ACCOUNT {code} {account.kind} {account.name} parent {parent_code(books, code)}")
350 elif op == "DEACTIVATE"
351 account := try get_account(books, words[1])
352 if not account.active
353 return error("already inactive " + account.code)
354 balance := account_balance(books, account)
355 if balance != 0
356 return error("nonzero balance {account.code} {money(balance)}")
357 account.active = false
358 .books.accounts.set(account.code, account)
359 emit("DEACTIVATE {account.code} {account.name}")
360 elif op == "JOURNAL" or op == "TEMPLATE"
361 match books.draft
362 Draft discarded
363 emit("DISCARD {discarded.key} lines {#discarded.lines}")
364 none
365 books.draft = none
366 books.draft = if op == "JOURNAL" then Draft("journal", words[1], words[2], words[3]) else Draft("template", words[1], "", words[2])
367 elif op == "DR" or op == "CR"
368 Draft block
369 match books.draft
370 Draft found
371 block = found
372 none
373 return error("no open entry")
374 string | none partner := none
375 if #words == 4
376 partner = words[3]
377 block.lines = block.lines.append(PostingLine(op == "DR", words[1], parse_cents(words[2]), partner))
378 books.draft = block
379 elif op == "POST"
380 return post_draft(&books)
381 elif op == "DRAFT"
382 match books.draft
383 Draft block
384 values := line_totals(block.lines)
385 emit("DRAFT {block.kind} {block.key} lines {#block.lines} debit {money(values[0])} credit {money(values[1])} difference {money(values[0] - values[1])}")
386 none
387 return error("no open entry")
388 elif op == "REVERSE"
389 index := try get_entry(books, words[1])
390 original := books.entries[index]
391 if includes(["reversal", "closing", "reopen"], original.category)
392 return error("cannot reverse " + original.id)
393 match original.reversed_by
394 int prior
395 return error("already reversed " + original.id)
396 none
397 original.reversed_by = none
398 day := try checked_date(words[2])
399 if day < original.day
400 return error("reverse before entry {original.id} {date_text(original.day)}")
401 try check_closed(books, day)
402 try active_check(books, [for line := original.lines[] then line.code])
403 created := reverse_entry(&books, index, day, "reversal", original.memo)
404 posted := books.entries[created]
405 emit("REVERSE {original.id} entry {posted.id} {date_text(day)} amount {money(posted.amount)}")
406 elif op == "RECUR"
407 pattern := try get_template(books, words[1])
408 day := try checked_date(words[2])
409 count := parse_integer(words[3])
410 if count == 0
411 return error("invalid count")
412 try check_closed(books, day)
413 try active_check(books, [for line := pattern.lines[] then line.code])
414 emit("RECUR {words[1]} count {count}")
415 for: 0 <= offset < count
416 pattern.uses += 1
417 index := post_entry(&books, "{words[1]}-{pattern.uses}", months_after(day, offset), "recurring", pattern.memo, pattern.lines)
418 emit(" " + posted_text(books.entries[index]))
419 .books.templates.set(words[1], pattern)
420 elif op == "DEPRECIATE"
421 day := try checked_date(words[1])
422 try check_closed(books, day)
423 asset := try get_account(books, words[2], "ASSET")
424 contra := try get_account(books, words[3], "CONTRA_ASSET")
425 expense := try get_account(books, words[4], "EXPENSE")
426 try active_check(books, [contra.code, expense.code])
427 cost := account_balance(books, asset, day)
428 accumulated := account_balance(books, contra, day)
429 amount := if cost > 0 then math.min(cost / parse_integer(words[5]), cost - accumulated) else 0
430 if amount <= 0
431 return error("nothing to depreciate " + asset.code)
432 books.depreciation_count += 1
433 id := "DEP-{books.depreciation_count}"
434 post_entry(&books, id, day, "depreciation", "depreciation", [PostingLine(true, expense.code, amount), PostingLine(false, contra.code, amount)])
435 emit("DEPRECIATE {asset.code} entry {id} {date_text(day)} amount {money(amount)} accumulated {money(accumulated + amount)} book {money(cost - accumulated - amount)}")
436 elif op == "INTEREST"
437 account := try get_account(books, words[1])
438 if account.kind != "ASSET" and account.kind != "LIABILITY"
439 return error("not interest bearing " + account.code)
440 offset := try get_account(books, words[2], if account.kind == "ASSET" then "REVENUE" else "EXPENSE")
441 period := try checked_range(words[3], words[4])
442 try check_closed(books, period.last)
443 try active_check(books, [account.code, offset.code])
444 sum := daily_sum(books, account, period.first, period.last)
445 amount := sum * parse_integer(words[5]) / 3650000
446 if amount <= 0
447 return error("no interest " + account.code)
448 books.interest_count += 1
449 id := "INT-{books.interest_count}"
450 debit := if account.kind == "ASSET" then account.code else offset.code
451 credit := if account.kind == "ASSET" then offset.code else account.code

455 elif op == "WRITEOFF"
