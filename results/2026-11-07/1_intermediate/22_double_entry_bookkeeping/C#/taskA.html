13 readonly record struct BookDate(int Year, int Month, int Day)
14 {
15 static readonly int[] MonthLengths = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
16 public static int DaysInMonth(int year, int month) =>
17 MonthLengths[month - 1] + (month == 2 && year % 4 == 0 && (year % 100 != 0 || year % 400 == 0) ? 1 : 0);
18 public int Number
19 {
20 get
21 {
22 int previous = Year - 1;
23 int days = 365 * (Year - 2000) + previous / 4 - 1999 / 4
24 - previous / 100 + 1999 / 100 + previous / 400 - 1999 / 400;
25 for (int month = 1; month < Month; month++) days += DaysInMonth(Year, month);
26 return days + Day - 1;
27 }
28 }
29 public static BookDate Parse(string text)
30 {
31 int year = int.Parse(text[..4]), month = int.Parse(text[5..7]), day = int.Parse(text[8..]);
32 if (year < 2000 || year > 2099 || month < 1 || month > 12 || day < 1 || day > DaysInMonth(year, month))
33 throw new CommandError($"invalid date {text}");
34 return new(year, month, day);
35 }
36 public BookDate AddMonths(int count)
37 {
38 int offset = Month - 1 + count, year = Year + offset / 12, month = offset % 12 + 1;
39 return new(year, month, Math.Min(Day, DaysInMonth(year, month)));
40 }
41 public override string ToString() => $"{Year:D4}-{Month:D2}-{Day:D2}";
42 }

104 static string Money(long cents)
105 {
106 long absolute = Math.Abs(cents);
107 return $"{(cents < 0 ? "-" : "")}{absolute / 100}.{absolute % 100:D2}";
108 }
109 static long ParseMoney(string text) => long.Parse(text[..^3]) * 100 + int.Parse(text[^2..]);
110 static string TypeName(AccountType type) => type == AccountType.ContraAsset ? "CONTRA_ASSET" : type.ToString().ToUpperInvariant();
111 static AccountType ParseType(string text) => text switch
112 {
113 "ASSET" => AccountType.Asset, "LIABILITY" => AccountType.Liability, "EQUITY" => AccountType.Equity,
114 "REVENUE" => AccountType.Revenue, "EXPENSE" => AccountType.Expense, _ => AccountType.ContraAsset
115 };
116 static bool Prefix(string prefix, string code) => code.StartsWith(prefix, StringComparison.Ordinal);

118 sealed class Books

120 readonly SortedDictionary<string, Account> accounts = new(StringComparer.Ordinal);
121 readonly List<Entry> entries = [];
122 readonly Dictionary<string, Entry> entryIds = new(StringComparer.Ordinal);
123 readonly SortedDictionary<string, Template> templates = new(StringComparer.Ordinal);

125 readonly StringBuilder output = new();

129 void Emit(string text) => output.Append(text).Append('\n');
130 string Closed => closes.Count == 0 ? "none" : closes.Peek().Date.ToString();
131 Account GetAccount(string code) => accounts.TryGetValue(code, out var account) ? account : throw new CommandError($"unknown account {code}");
132 Entry GetEntry(string id) => entryIds.TryGetValue(id, out var entry) ? entry : throw new CommandError($"unknown entry {id}");
133 Account RequireType(string code, AccountType type)
134 {
135 var account = GetAccount(code);
136 if (account.Type != type) throw new CommandError($"wrong type {code} {TypeName(type)}");
137 return account;

139 static void Active(IEnumerable<Account> sequence)
140 {
141 foreach (var account in sequence)
142 if (!account.Active) throw new CommandError($"inactive account {account.Code}");
143 }
144 void CheckClosed(BookDate date)
145 {
146 if (closes.Count > 0 && date.Number <= closes.Peek().Date.Number) throw new CommandError($"period closed {date}");
147 }
148 static (BookDate Start, BookDate End) Range(string start, string end)
149 {
150 var a = BookDate.Parse(start);
151 var b = BookDate.Parse(end);
152 if (a.Number > b.Number) throw new CommandError($"invalid range {start} {end}");
153 return (a, b);
154 }
155 IEnumerable<Entry> Ordered(int start = 0, int end = int.MaxValue) => entries
156 .Where(entry => entry.Date.Number >= start && entry.Date.Number <= end)
157 .OrderBy(entry => entry.Date.Number).ThenBy(entry => entry.Sequence);
158 Dictionary<string, Totals> Aggregate(int start = 0, int end = int.MaxValue, bool operating = false)
159 {
160 var result = new Dictionary<string, Totals>(StringComparer.Ordinal);
161 foreach (var entry in entries)
162 {
163 if (entry.Date.Number < start || entry.Date.Number > end || operating && (entry.Kind is EntryKind.Closing or EntryKind.Reopen)) continue;
164 foreach (var line in entry.Lines) result[line.Code] = result.GetValueOrDefault(line.Code).Add(line);
165 }
166 return result;
167 }
168 long Balance(Account account, int end = int.MaxValue) =>
169 entries.Where(entry => entry.Date.Number <= end).Sum(entry => entry.Lines.Where(line => line.Code == account.Code).Sum(line => line.Difference)) * account.Sign;
170 long DailySum(Account account, int start, int end)
171 {
172 long sum = 0;
173 foreach (var entry in entries)
174 {
175 if (entry.Date.Number > end) continue;
176 long change = entry.Lines.Where(line => line.Code == account.Code).Sum(line => line.Difference) * account.Sign;
177 sum += change * (end - Math.Max(start, entry.Date.Number) + 1);
178 }
179 return sum;
180 }
181 string Parent(string code) => accounts.Keys.Where(other => other != code && Prefix(other, code)).OrderByDescending(other => other.Length).FirstOrDefault() ?? "none";
182 string Top(string code) => accounts.Keys.First(other => Prefix(other, code));
183 Entry Post(string id, BookDate date, EntryKind kind, string memo, IEnumerable<Line> lines)
184 {
185 var entry = new Entry(id, date, kind, memo, lines.ToArray(), entries.Count + 1);
186 entries.Add(entry);
187 entryIds.Add(id, entry);
188 return entry;
189 }
190 Entry Reverse(Entry original, BookDate date, EntryKind kind, string memo)
191 {
192 var entry = Post(original.Id + "-R", date, kind, memo, original.Lines.Select(line => line with { Debit = !line.Debit }));
193 original.ReversedBy = entry.Id;
194 entry.Reverses = original.Id;
195 return entry;
196 }
197 static Totals LineTotals(IEnumerable<Line> lines)
198 {
199 var total = new Totals();
200 foreach (var line in lines) total = total.Add(line);
201 return total;
202 }
203 SortedDictionary<string, Settlement> Settlements(Account account, int date)
204 {
205 var result = new SortedDictionary<string, Settlement>(StringComparer.Ordinal);
206 foreach (var entry in Ordered(end: date))
207 foreach (var line in entry.Lines)
208 {
209 if (line.Code != account.Code || line.Partner is null) continue;
210 if (!result.TryGetValue(line.Partner, out var settlement)) result.Add(line.Partner, settlement = new());
211 long signed = line.Difference * account.Sign;
212 if (signed > 0)
213 {
214 long applied = Math.Min(signed, settlement.Unapplied);
215 settlement.Unapplied -= applied;
216 if (signed > applied) settlement.Items.Enqueue(new(entry, signed, signed - applied));
217 }
218 else
219 {
220 long remaining = -signed;
221 while (remaining > 0 && settlement.Items.Count > 0)
222 {
223 var item = settlement.Items.Peek();
224 long applied = Math.Min(remaining, item.Remaining);
225 item.Remaining -= applied;
226 remaining -= applied;
227 if (item.Remaining == 0) settlement.Items.Dequeue();
228 }
229 settlement.Unapplied += remaining;
230 }
231 }
232 return result;
233 }
234 long Section(string heading, AccountType[] types, int sign, Dictionary<string, Totals> totals, long? earnings = null)
235 {
236 var targets = accounts.Values.Where(account => types.Contains(account.Type)).ToArray();
237 long Display(Account account) => totals.GetValueOrDefault(account.Code).Difference * sign;
238 long total = targets.Sum(Display) + (earnings ?? 0);
239 Emit($" {heading} {Money(total)}");
240 foreach (var group in targets.GroupBy(account => Top(account.Code)).OrderBy(group => group.Key, StringComparer.Ordinal))
241 {
242 if (!group.Any(account => Display(account) != 0)) continue;
243 Emit($" {group.Key} {accounts[group.Key].Name} {Money(group.Sum(Display))}");
244 foreach (var account in group)
245 if (Display(account) != 0) Emit($" {account.Code} {account.Name} {Money(Display(account))}");
246 }
247 if (earnings is not null) Emit($" current earnings {Money(earnings.Value)}");
248 return total;
249 }

251 void PostDraft()
252 {
253 var block = draft ?? throw new CommandError("no open entry");
254 draft = null;
255 BookDate date = default;
256 if (block.Journal)
257 {
258 if (entryIds.ContainsKey(block.Key)) throw new CommandError($"duplicate entry {block.Key}");
259 date = BookDate.Parse(block.Date!);
260 CheckClosed(date);
261 }
262 else if (templates.ContainsKey(block.Key)) throw new CommandError($"duplicate template {block.Key}");
263 if (block.Lines.Count < 2) throw new CommandError($"too few lines {block.Key}");
264 foreach (var line in block.Lines)
265 {
266 var account = GetAccount(line.Code);
267 if (block.Journal && !account.Active) throw new CommandError($"inactive account {line.Code}");
268 if (line.Amount == 0) throw new CommandError($"zero amount {line.Code}");
269 }
270 var total = LineTotals(block.Lines);
271 if (total.Debit != total.Credit) throw new CommandError($"unbalanced {block.Key} debit {Money(total.Debit)} credit {Money(total.Credit)}");
272 if (block.Journal)
273 {
274 var entry = Post(block.Key, date, EntryKind.Journal, block.Memo, block.Lines);
275 Emit($"POST {entry.Id} {date} lines {entry.Lines.Length} amount {Money(entry.Amount)}");
276 }
277 else
278 {
279 templates.Add(block.Key, new(block.Memo, block.Lines.ToArray()));
280 Emit($"TEMPLATE {block.Key} lines {block.Lines.Count} amount {Money(total.Debit)}");
281 }
282 }
283 void Close(BookDate date, string code)
284 {
285 CheckClosed(date);
286 var equity = RequireType(code, AccountType.Equity);
287 Active([equity]);
288 var totals = Aggregate(end: date.Number);
289 var targets = accounts.Values.Where(account => (account.Type is AccountType.Revenue or AccountType.Expense)
290 && totals.GetValueOrDefault(account.Code).Difference != 0).ToArray();
291 Active(targets);
292 var lines = new List<Line>();
293 long income = 0;
294 foreach (var account in targets)
295 {
296 long difference = totals.GetValueOrDefault(account.Code).Difference;
297 lines.Add(new(difference < 0, account.Code, Math.Abs(difference)));
298 income -= difference;
299 }
300 if (income != 0) lines.Add(new(income < 0, code, Math.Abs(income)));
301 closeCount++;
302 Entry? entry = lines.Count == 0 ? null : Post($"CLOSE-{closeCount}", date, EntryKind.Closing, "closing", lines);
303 closes.Push(new(date, entry));
304 Emit(entry is null ? $"CLOSE {date} no entry" : $"CLOSE {date} entry {entry.Id} lines {lines.Count} net income {Money(income)}");
305 }
306 void Ledger(Account account, BookDate start, BookDate end)
307 {
308 long running = Balance(account, start.Number - 1), debit = 0, credit = 0;
309 Emit($"LEDGER {account.Code} {account.Name} {start} {end}");
310 Emit($" opening {Money(running)}");
311 foreach (var entry in Ordered(start.Number, end.Number))
312 foreach (var line in entry.Lines)
313 {
314 if (line.Code != account.Code) continue;
315 running += line.Difference * account.Sign;
316 if (line.Debit) debit += line.Amount; else credit += line.Amount;
317 Emit($" {entry.Date} {entry.Id} {line.Side} {Money(line.Amount)} balance {Money(running)} {entry.Memo}");
318 }
319 int days = end.Number - start.Number + 1;
320 Emit($" ending {Money(running)} debits {Money(debit)} credits {Money(credit)} days {days} average {Money(DailySum(account, start.Number, end.Number) / days)}");
321 }
322 static string AgeColumns(long[] values) =>
323 $"0-30 {Money(values[0])} 31-60 {Money(values[1])} 61-90 {Money(values[2])} 91+ {Money(values[3])} unapplied {Money(values[4])} balance {Money(values.Take(4).Sum() - values[4])}";

325 public void Execute(string text)
326 {
327 var words = text.Split(' ');
328 switch (words[0])
329 {
330 case "ACCOUNT":
331 {
332 string code = words[1];
333 if (accounts.ContainsKey(code)) throw new CommandError($"duplicate account {code}");
334 var account = new Account(code, ParseType(words[2]), words[3]);
335 foreach (var other in accounts.Values)
336 if ((Prefix(code, other.Code) || Prefix(other.Code, code)) && account.Class != other.Class)
337 throw new CommandError($"class mismatch {code} {other.Code}");
338 accounts.Add(code, account);
339 Emit($"ACCOUNT {code} {words[2]} {account.Name} parent {Parent(code)}");
340 break;
341 }
342 case "DEACTIVATE":
343 {
344 var account = GetAccount(words[1]);
345 if (!account.Active) throw new CommandError($"already inactive {account.Code}");
346 long balance = Balance(account);
347 if (balance != 0) throw new CommandError($"nonzero balance {account.Code} {Money(balance)}");
348 account.Active = false;
349 Emit($"DEACTIVATE {account.Code} {account.Name}");
350 break;
351 }
352 case "JOURNAL":
353 case "TEMPLATE":
354 if (draft is not null) Emit($"DISCARD {draft.Key} lines {draft.Lines.Count}");
355 draft = words[0] == "JOURNAL" ? new(true, words[1], words[2], words[3]) : new(false, words[1], null, words[2]);
356 break;
357 case "DR":
358 case "CR":
359 if (draft is null) throw new CommandError("no open entry");
360 draft.Lines.Add(new(words[0] == "DR", words[1], ParseMoney(words[2]), words.Length == 4 ? words[3] : null));
361 break;
362 case "POST": PostDraft(); break;
363 case "DRAFT":
364 {
365 var block = draft ?? throw new CommandError("no open entry");
366 var total = LineTotals(block.Lines);
367 Emit($"DRAFT {block.Kind} {block.Key} lines {block.Lines.Count} debit {Money(total.Debit)} credit {Money(total.Credit)} difference {Money(total.Difference)}");
368 break;
369 }
370 case "REVERSE":
371 {
372 var original = GetEntry(words[1]);
373 if (original.Kind is EntryKind.Reversal or EntryKind.Closing or EntryKind.Reopen) throw new CommandError($"cannot reverse {original.Id}");
374 if (original.ReversedBy is not null) throw new CommandError($"already reversed {original.Id}");
375 var date = BookDate.Parse(words[2]);
376 if (date.Number < original.Date.Number) throw new CommandError($"reverse before entry {original.Id} {original.Date}");
377 CheckClosed(date);
378 Active(original.Lines.Select(line => accounts[line.Code]));
379 var entry = Reverse(original, date, EntryKind.Reversal, original.Memo);
380 Emit($"REVERSE {original.Id} entry {entry.Id} {date} amount {Money(entry.Amount)}");
381 break;
382 }
383 case "RECUR":
384 {
385 if (!templates.TryGetValue(words[1], out var template)) throw new CommandError($"unknown template {words[1]}");
386 var date = BookDate.Parse(words[2]);
387 int count = int.Parse(words[3]);
388 if (count == 0) throw new CommandError("invalid count");
389 CheckClosed(date);
390 Active(template.Lines.Select(line => accounts[line.Code]));
391 Emit($"RECUR {words[1]} count {count}");
392 for (int month = 0; month < count; month++)
393 {
394 var entry = Post($"{words[1]}-{++template.Uses}", date.AddMonths(month), EntryKind.Recurring, template.Memo, template.Lines);
395 Emit($" POST {entry.Id} {entry.Date} lines {entry.Lines.Length} amount {Money(entry.Amount)}");
396 }
397 break;
398 }
399 case "DEPRECIATE":
400 {
401 var date = BookDate.Parse(words[1]);
402 CheckClosed(date);
403 var asset = RequireType(words[2], AccountType.Asset);
404 var contra = RequireType(words[3], AccountType.ContraAsset);
405 var expense = RequireType(words[4], AccountType.Expense);
406 Active([contra, expense]);
407 long cost = Balance(asset, date.Number), accumulated = Balance(contra, date.Number);
408 long amount = cost > 0 ? Math.Min(cost / int.Parse(words[5]), cost - accumulated) : 0;
409 if (amount <= 0) throw new CommandError($"nothing to depreciate {asset.Code}");
410 var entry = Post($"DEP-{++depreciationCount}", date, EntryKind.Depreciation, "depreciation", [new(true, expense.Code, amount), new(false, contra.Code, amount)]);
411 Emit($"DEPRECIATE {asset.Code} entry {entry.Id} {date} amount {Money(amount)} accumulated {Money(accumulated + amount)} book {Money(cost - accumulated - amount)}");
412 break;
413 }
414 case "INTEREST":
415 {
416 var account = GetAccount(words[1]);
417 if (account.Type is not (AccountType.Asset or AccountType.Liability)) throw new CommandError($"not interest bearing {account.Code}");
418 var offset = RequireType(words[2], account.Type == AccountType.Asset ? AccountType.Revenue : AccountType.Expense);
419 var (start, end) = Range(words[3], words[4]);
420 CheckClosed(end);
421 Active([account, offset]);
422 long sum = DailySum(account, start.Number, end.Number), amount = sum * int.Parse(words[5]) / 3650000;
423 if (amount <= 0) throw new CommandError($"no interest {account.Code}");
424 string debit = account.Type == AccountType.Asset ? account.Code : offset.Code;
425 string credit = account.Type == AccountType.Asset ? offset.Code : account.Code;
426 var entry = Post($"INT-{++interestCount}", end, EntryKind.Interest, "interest", [new(true, debit, amount), new(false, credit, amount)]);
427 int days = end.Number - start.Number + 1;
428 Emit($"INTEREST {account.Code} entry {entry.Id} {end} days {days} average {Money(sum / days)} amount {Money(amount)}");
429 break;
430 }
431 case "WRITEOFF":
432 {
433 var date = BookDate.Parse(words[1]);
434 CheckClosed(date);
435 var account = RequireType(words[2], AccountType.Asset);
436 var expense = RequireType(words[3], AccountType.Expense);
437 Active([account, expense]);
438 int age = int.Parse(words[4]);
439 var targets = Settlements(account, date.Number).Select(pair => (Partner: pair.Key,
440 Amount: pair.Value.Items.Where(item => date.Number - item.Entry.Date.Number > age).Sum(item => item.Remaining)))
441 .Where(target => target.Amount > 0).ToArray();
442 long total = targets.Sum(target => target.Amount);
443 if (total == 0) throw new CommandError($"nothing to write off {account.Code}");
444 var lines = new List<Line> { new(true, expense.Code, total) };
445 lines.AddRange(targets.Select(target => new Line(false, account.Code, target.Amount, target.Partner)));
446 var entry = Post($"WO-{++writeoffCount}", date, EntryKind.Writeoff, "writeoff", lines);
447 Emit($"WRITEOFF {account.Code} entry {entry.Id} {date} partners {targets.Length} amount {Money(total)}");
448 foreach (var target in targets) Emit($" {target.Partner} {Money(target.Amount)}");
449 break;
450 }
451 case "CLOSE": Close(BookDate.Parse(words[1]), words[2]); break;
452 case "REOPEN":
453 {
454 if (closes.Count == 0) throw new CommandError("nothing to reopen");
455 var record = closes.Peek();
456 string result = "no entry";
457 if (record.Entry is not null)
458 {
459 Active(record.Entry.Lines.Select(line => accounts[line.Code]));
460 result = "entry " + Reverse(record.Entry, record.Date, EntryKind.Reopen, "reopen").Id;
461 }
462 closes.Pop();
463 Emit($"REOPEN {record.Date} {result} closed {Closed}");
464 break;
465 }
466 case "LEDGER":
467 {
468 var account = GetAccount(words[1]);
469 var (start, end) = Range(words[2], words[3]);
470 Ledger(account, start, end);
471 break;
472 }
473 case "ACTIVITY":
474 {
475 var account = GetAccount(words[1]);
476 int year = int.Parse(words[2]);
477 long debit = 0, credit = 0;
478 Emit($"ACTIVITY {account.Code} {account.Name} {year}");
479 for (int month = 1; month <= 12; month++)
480 {
481 int start = new BookDate(year, month, 1).Number, end = new BookDate(year, month, BookDate.DaysInMonth(year, month)).Number;
482 var totals = Aggregate(start, end).GetValueOrDefault(account.Code);
483 debit += totals.Debit;
484 credit += totals.Credit;
485 Emit($" {year:D4}-{month:D2} debit {Money(totals.Debit)} credit {Money(totals.Credit)} balance {Money(Balance(account, end))}");
486 }
487 Emit($" total debit {Money(debit)} credit {Money(credit)}");
488 break;
489 }
490 case "TRIAL":
491 {
492 var date = BookDate.Parse(words[1]);
493 var totals = Aggregate(end: date.Number);
494 long debit = 0, credit = 0;
495 Emit($"TRIAL {date}");
496 foreach (var account in accounts.Values)
497 {
498 long difference = totals.GetValueOrDefault(account.Code).Difference;
499 if (difference == 0) continue;
500 if (difference > 0) debit += difference; else credit -= difference;
501 Emit($" {account.Code} {account.Name} {(difference > 0 ? "debit" : "credit")} {Money(Math.Abs(difference))}");
502 }
503 Emit($" total debit {Money(debit)} credit {Money(credit)}");
504 break;
505 }
506 case "INCOME":
507 {
508 var (start, end) = Range(words[1], words[2]);
509 var totals = Aggregate(start.Number, end.Number, operating: true);
510 Emit($"INCOME {start} {end}");
511 long revenue = Section("REVENUE", [AccountType.Revenue], -1, totals);
512 long expense = Section("EXPENSE", [AccountType.Expense], 1, totals);
513 Emit($" NET INCOME {Money(revenue - expense)}");
514 break;
515 }
516 case "BALANCE":
517 {
518 var date = BookDate.Parse(words[1]);
519 var totals = Aggregate(end: date.Number);
520 long earnings = -accounts.Values.Where(account => account.Type is AccountType.Revenue or AccountType.Expense)
521 .Sum(account => totals.GetValueOrDefault(account.Code).Difference);
522 Emit($"BALANCE {date}");
523 long assets = Section("ASSETS", [AccountType.Asset, AccountType.ContraAsset], 1, totals);
524 long liabilities = Section("LIABILITIES", [AccountType.Liability], -1, totals);
525 long equity = Section("EQUITY", [AccountType.Equity], -1, totals, earnings);
526 Emit(assets == liabilities + equity ? " check ok" : $" check mismatch {Money(assets - liabilities - equity)}");
527 break;
528 }
529 case "ROLLUP":
530 {
531 var date = BookDate.Parse(words[2]);
532 var targets = accounts.Values.Where(account => Prefix(words[1], account.Code)).ToArray();
533 Emit($"ROLLUP {words[1]} {date}");
534 if (targets.Length == 0) { Emit(" (none)"); break; }
535 var totals = Aggregate(end: date.Number);
536 foreach (var account in targets)
537 {
538 int depth = targets.Count(other => other.Code != account.Code && Prefix(other.Code, account.Code));
539 long subtotal = targets.Where(other => Prefix(account.Code, other.Code)).Sum(other => totals.GetValueOrDefault(other.Code).Difference) * account.Sign;
540 Emit($"{new string(' ', 2 * (1 + depth))}{account.Code} {account.Name} own {Money(totals.GetValueOrDefault(account.Code).Difference * account.Sign)} total {Money(subtotal)}");
541 }
542 Emit($" postings debit {Money(targets.Sum(account => totals.GetValueOrDefault(account.Code).Debit))} credit {Money(targets.Sum(account => totals.GetValueOrDefault(account.Code).Credit))}");
543 break;
544 }
545 case "AGING":
546 {
547 var account = GetAccount(words[1]);
548 var date = BookDate.Parse(words[2]);
549 Emit($"AGING {account.Code} {account.Name} {date}");
550 var total = new long[5];
551 foreach (var (partner, settlement) in Settlements(account, date.Number))
552 {
553 if (settlement.Items.Count == 0 && settlement.Unapplied == 0) continue;
554 var values = new long[5];
555 foreach (var item in settlement.Items)
556 {
557 int age = date.Number - item.Entry.Date.Number;
558 values[age <= 30 ? 0 : age <= 60 ? 1 : age <= 90 ? 2 : 3] += item.Remaining;
559 }
560 values[4] = settlement.Unapplied;
561 for (int i = 0; i < 5; i++) total[i] += values[i];
562 Emit($" {partner} {AgeColumns(values)}");
563 }
564 Emit($" total {AgeColumns(total)}");
565 break;
566 }
567 case "ITEMS":
568 {
569 var account = GetAccount(words[1]);
570 var date = BookDate.Parse(words[3]);
571 var settlements = Settlements(account, date.Number);
572 var settlement = settlements.TryGetValue(words[2], out var found) ? found : new Settlement();
573 Emit($"ITEMS {account.Code} {account.Name} {words[2]} {date}");
574 foreach (var item in settlement.Items)
575 Emit($" {item.Entry.Date} {item.Entry.Id} amount {Money(item.Original)} open {Money(item.Remaining)} age {date.Number - item.Entry.Date.Number}");
576 Emit($" unapplied {Money(settlement.Unapplied)} balance {Money(settlement.Items.Sum(item => item.Remaining) - settlement.Unapplied)}");
577 break;
578 }
579 case "ENTRY":
580 {
581 var entry = GetEntry(words[1]);
582 Emit($"ENTRY {entry.Id} {entry.Date} {entry.Category} {entry.Memo}");
583 foreach (var line in entry.Lines)
584 Emit($" {line.Side} {line.Code} {accounts[line.Code].Name} {Money(line.Amount)}{(line.Partner is null ? "" : " " + line.Partner)}");
585 Emit($" total {Money(entry.Amount)}");
586 if (entry.Reverses is not null) Emit($" reverses {entry.Reverses}");
587 if (entry.ReversedBy is not null) Emit($" reversed by {entry.ReversedBy}");
588 break;
589 }
590 case "DAYBOOK":
591 {
592 var (start, end) = Range(words[1], words[2]);
593 var selected = Ordered(start.Number, end.Number).ToArray();
594 Emit($"DAYBOOK {start} {end}");
595 foreach (var entry in selected) Emit($" {entry.Date} {entry.Id} {entry.Category} {Money(entry.Amount)} {entry.Memo}");
596 Emit($" total entries {selected.Length} amount {Money(selected.Sum(entry => entry.Amount))}");
597 break;
598 }
599 case "ACCOUNTS":
600 {
601 var totals = Aggregate();
602 Emit("ACCOUNTS");
603 foreach (var account in accounts.Values)
604 Emit($" {account.Code} {TypeName(account.Type)} {account.Name} parent {Parent(account.Code)} balance {Money(totals.GetValueOrDefault(account.Code).Difference * account.Sign)} {(account.Active ? "active" : "inactive")}");
605 if (accounts.Count == 0) Emit(" (none)");
606 break;
607 }
608 case "TEMPLATES":
609 Emit("TEMPLATES");
610 foreach (var (name, template) in templates)
611 Emit($" {name} {template.Memo} lines {template.Lines.Length} amount {Money(LineTotals(template.Lines).Debit)} used {template.Uses}");
612 if (templates.Count == 0) Emit(" (none)");
613 break;
614 case "STATUS":
615 Emit($"STATUS accounts {accounts.Count} inactive {accounts.Values.Count(account => !account.Active)} entries {entries.Count} templates {templates.Count} closed {Closed} open {draft?.Key ?? "none"}");
616 break;
617 }
618 }
619 public void Process(string text)
620 {
621 try { Execute(text); }
622 catch (CommandError error) { Emit("ERROR " + error.Message); }
623 }
624 public override string ToString() => output.ToString();
625 }
