10enum AccountType: String, CaseIterable, Sendable {
11 case asset = "ASSET", liability = "LIABILITY", equity = "EQUITY"
12 case revenue = "REVENUE", expense = "EXPENSE", contraAsset = "CONTRA_ASSET"
13 var sign: Cents {
14 self == .asset || self == .expense ? 1 : -1
15 }
16 var group: AccountType {
17 self == .contraAsset ? .asset : self
18 }
19 var isIncome: Bool {
20 self == .revenue || self == .expense
21 }
22}

35func padded(_ value: Int, _ width: Int = 2) -> String {
36 let text = String(value)
37 return String(repeating: "0", count: max(0, width - text.count)) + text
38}

40func money(_ value: Cents) -> String {
41 let magnitude = value < 0 ? -value : value
42 return "\(value < 0 ? "-" : "")\(magnitude / 100).\(padded(Int(magnitude % 100)))"
43}

45func parseMoney(_ text: String) -> Cents {
46 let parts = text.split(separator: ".")
47 return Cents(parts[0])! * 100 + Cents(parts[1])!
48}

50struct BookDate: Sendable {
51 static let lastNumber = 36524
52 static let monthLengths = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]
53 let year: Int
54 let month: Int
55 let day: Int
56
57 init(_ year: Int, _ month: Int, _ day: Int) {
58 self.year = year
59 self.month = month
60 self.day = day
61 }
62
63 static func daysInMonth(_ year: Int, _ month: Int) -> Int {
64 let leap = year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
65 return monthLengths[month - 1] + (month == 2 && leap ? 1 : 0)
66 }
67
68 static func parse(_ text: String) throws -> BookDate {
69 let parts = text.split(separator: "-").map { Int($0)! }
70 let year = parts[0], month = parts[1], day = parts[2]
71 guard (2000...2099).contains(year), (1...12).contains(month),
72 day >= 1, day <= daysInMonth(year, month)
73 else {
74 throw CommandError("invalid date \(text)")
75 }
76 return BookDate(year, month, day)
77 }
78
79 var number: Int {
80 let previous = year - 1
81 var result = 365 * (year - 2000) + previous / 4 - 1999 / 4
82 - previous / 100 + 1999 / 100 + previous / 400 - 1999 / 400
83 for month in 1..<self.month {
84 result += Self.daysInMonth(year, month)
85 }
86 return result + day - 1
87 }
88
89 var text: String {
90 "\(padded(year, 4))-\(padded(month))-\(padded(day))"
91 }
92
93 func addingMonths(_ count: Int) -> BookDate {
94 let offset = month - 1 + count
95 let year = self.year + offset / 12, month = offset % 12 + 1
96 return BookDate(year, month, min(day, Self.daysInMonth(year, month)))
97 }
98}

100func checkedRange(_ first: String, _ last: String) throws -> (BookDate, BookDate) {
101 let start = try BookDate.parse(first)
102 let end = try BookDate.parse(last)
103 if start.number > end.number {
104 throw CommandError("invalid range \(first) \(last)")
105 }
106 return (start, end)
107}

309 func ordered(_ first: Int = 0, _ last: Int = BookDate.lastNumber) -> [Entry] {
310 entries.filter { first <= $0.date.number && $0.date.number <= last }.sorted {
311 $0.date.number == $1.date.number ? $0.sequence < $1.sequence : $0.date.number < $1.date.number

315 func aggregate(_ first: Int = 0, _ last: Int = BookDate.lastNumber, operating: Bool = false) -> [String: Totals] {
316 var totals: [String: Totals] = [:]
317 for entry in entries {
318 if entry.date.number < first || entry.date.number > last ||
319 operating && (entry.category == .closing || entry.category == .reopen)
320 {
321 continue
322 }
323 for line in entry.lines {
324 totals[line.code, default: Totals()].add(line)
325 }
326 }
327 return totals
328 }

330 func balance(_ code: String, _ last: Int = BookDate.lastNumber) -> Cents {
331 var difference: Cents = 0
332 for entry in entries where entry.date.number <= last {
333 for line in entry.lines where line.code == code {
334 difference += line.difference
335 }
336 }
337 return difference * accounts[code]!.type.sign
338 }

340 func dailySum(_ code: String, _ first: Int, _ last: Int) -> Cents {
341 var changes = [Cents](repeating: 0, count: last - first + 1)
342 var running: Cents = 0, sum: Cents = 0
343 let sign = accounts[code]!.type.sign
344 for entry in entries where entry.date.number <= last {
345 for line in entry.lines where line.code == code {
346 let change = sign * line.difference
347 if entry.date.number < first {
348 running += change
349 } else {
350 changes[entry.date.number - first] += change
351 }
352 }
353 }
354 for change in changes {
355 running += change
356 sum += running
357 }
358 return sum
359 }

376 func settlements(_ code: String, _ last: Int) -> [String: Settlement] {
377 var result: [String: Settlement] = [:]
378 let sign = accounts[code]!.type.sign
379 for entry in ordered(0, last) {
380 for line in entry.lines where line.code == code {
381 guard let partner = line.partner else {
382 continue
383 }
384 let settled = result[partner] ?? Settlement()
385 result[partner] = settled
386 let value = sign * line.difference
387 if value > 0 {
388 let applied = min(value, settled.unapplied)
389 settled.unapplied -= applied
390 if value > applied {
391 settled.items.append(OpenItem(entry, value, value - applied))
392 }
393 } else {
394 var remaining = -value
395 while remaining > 0 && settled.head < settled.items.count {
396 let item = settled.items[settled.head]
397 let applied = min(remaining, item.remaining)
398 remaining -= applied
399 item.remaining -= applied
400 if item.remaining == 0 {
401 settled.head += 1
402 }
403 }
404 settled.unapplied += remaining
405 }
406 }
407 }
408 return result
409 }

418 func postDraft() throws {
419 guard let block = draft else {
420 throw CommandError("no open entry")
421 }
422 draft = nil
423 var date: BookDate?
424 switch block.kind {
425 case .journal(let token):
426 if entryIds[block.key] != nil {
427 throw CommandError("duplicate entry \(block.key)")
428 }
429 date = try BookDate.parse(token)
430 try checkClosed(date!)
431 case .template:
432 if templates[block.key] != nil {
433 throw CommandError("duplicate template \(block.key)")
434 }
435 }
436 if block.lines.count < 2 {
437 throw CommandError("too few lines \(block.key)")
438 }
439 for line in block.lines {
440 let found = try account(line.code)
441 if block.kind.isJournal && !found.active {
442 throw CommandError("inactive account \(line.code)")
443 }
444 if line.amount == 0 {
445 throw CommandError("zero amount \(line.code)")
446 }
447 }
448 let debit = lineTotal(block.lines), credit = lineTotal(block.lines, .credit)
449 if debit !=
450 credit
451 {
452 throw CommandError("unbalanced \(block.key) debit \(money(debit)) credit \(money(credit))")
453 }
454 if block.kind.isJournal {
455 let posted = post(block.key, date!, .journal, block.memo, block.lines)
456 emit("POST \(posted.id) \(posted.date.text) lines \(posted.lines.count) amount \(money(debit))")
457 } else {
458 templates[block.key] = Template(block.memo, block.lines)
459 emit("TEMPLATE \(block.key) lines \(block.lines.count) amount \(money(debit))")
460 }
461 }

463 func close(_ date: BookDate, _ equity: String) throws {
464 try checkClosed(date)
465 _ = try account(equity, .equity)
466 try active([equity])
467 let totals = aggregate(0, date.number)
468 let targets = codes.filter { accounts[$0]!.type.isIncome && (totals[$0]?.difference ?? 0) != 0 }
469 try active(targets)
470 var lines: [Line] = []
471 var income: Cents = 0
472 for code in targets {
473 let value = totals[code]!.difference
474 lines.append(Line(value < 0 ? .debit : .credit, code, abs(value)))
475 income -= value
476 }
477 if income != 0 {
478 lines.append(Line(income < 0 ? .debit : .credit, equity, abs(income)))
479 }
480 closeCount += 1
481 let entry = lines.isEmpty ? nil : post("CLOSE-\(closeCount)", date, .closing, "closing", lines)
482 closes.append(CloseRecord(date: date, entry: entry))
483 if let entry {
484 emit("CLOSE \(date.text) entry \(entry.id) lines \(lines.count) net income \(money(income))")
485 } else {
486 emit("CLOSE \(date.text) no entry")
487 }
488 }

490 func section(
491 _ heading: String,
492 _ types: Set<AccountType>,
493 _ sign: Cents,
494 _ totals: [String: Totals],
495 earnings: Cents? = nil
496 ) -> Cents {
497 var groups: [String: [(String, Cents)]] = [:]
498 var total = earnings ?? 0
499 for code in codes where types.contains(accounts[code]!.type) {
500 let value = sign * (totals[code]?.difference ?? 0)
501 groups[top(code), default: []].append((code, value))
502 total += value
503 }
504 emit(" \(heading) \(money(total))")
505 for root in groups.keys.sorted() {
506 let group = groups[root]!
507 if !group.contains(where: { $0.1 != 0 }) {
508 continue
509 }
510 let subtotal = group.reduce(0) { $0 + $1.1 }
511 emit(" \(root) \(accounts[root]!.name) \(money(subtotal))")
512 for (code, value) in group
513 where value != 0
514 {
515 emit(" \(code) \(accounts[code]!.name) \(money(value))")
516 }
517 }
518 if let earnings {
519 emit(" current earnings \(money(earnings))")
520 }
521 return total
522 }

524 func ledger(_ code: String, _ first: BookDate, _ last: BookDate) {
525 let found = accounts[code]!
526 var running = balance(code, first.number - 1)
527 var debit: Cents = 0, credit: Cents = 0
528 emit("LEDGER \(code) \(found.name) \(first.text) \(last.text)")
529 emit(" opening \(money(running))")
530 for entry in ordered(first.number, last.number) {
531 for line in entry.lines where line.code == code {
532 running += found.type.sign * line.difference
533 if line.side == .debit {
534 debit += line.amount
535 } else {
536 credit += line.amount
537 }
538 emit(
539 " \(entry.date.text) \(entry.id) \(line.side.rawValue) \(money(line.amount)) balance \(money(running)) \(entry.memo)"
540 )
541 }
542 }
543 let days = last.number - first.number + 1
544 emit(
545 " ending \(money(running)) debits \(money(debit)) credits \(money(credit)) days \(days) average \(money(dailySum(code, first.number, last.number) / Cents(days)))"
546 )
547 }

556 func execute(_ words: [String]) throws {
557 switch words[0] {
558 case "ACCOUNT":
559 let code = words[1]
560 if accounts[code] != nil {
561 throw CommandError("duplicate account \(code)")
562 }
563 let type = AccountType(rawValue: words[2])!
564 for other in codes
565 where (code.hasPrefix(other) || other.hasPrefix(code)) && type.group != accounts[other]!.type.group
566 {
567 throw CommandError("class mismatch \(code) \(other)")
568 }
569 accounts[code] = Account(code, type, words[3])
570 emit("ACCOUNT \(code) \(type.rawValue) \(words[3]) parent \(parent(code))")
571 case "DEACTIVATE":
572 let found = try account(words[1])
573 if !found.active {
574 throw CommandError("already inactive \(found.code)")
575 }
576 let value = balance(found.code)
577 if value != 0 {
578 throw CommandError("nonzero balance \(found.code) \(money(value))")
579 }
580 found.active = false
581 emit("DEACTIVATE \(found.code) \(found.name)")
582 case "JOURNAL":
583 open(.journal(words[2]), words[1], words[3])
584 case "TEMPLATE":
585 open(.template, words[1], words[2])
586 case "DR", "CR":
587 guard let draft else {
588 throw CommandError("no open entry")
589 }
590 draft.lines.append(Line(
591 Side(rawValue: words[0])!,
592 words[1],
593 parseMoney(words[2]),
594 words.count == 4 ? words[3] : nil
595 ))
596 case "POST":
597 try postDraft()
598 case "DRAFT":
599 guard let draft else {
600 throw CommandError("no open entry")
601 }
602 let debit = lineTotal(draft.lines), credit = lineTotal(draft.lines, .credit)
603 emit(
604 "DRAFT \(draft.kind.name) \(draft.key) lines \(draft.lines.count) debit \(money(debit)) credit \(money(credit)) difference \(money(debit - credit))"
605 )
606 case "REVERSE":
607 let original = try entry(words[1])
608 if [.reversal, .closing, .reopen]
609 .contains(original.category)
610 {
611 throw CommandError("cannot reverse \(original.id)")
612 }
613 if original.reversedBy != nil {
614 throw CommandError("already reversed \(original.id)")
615 }
616 let date = try BookDate.parse(words[2])
617 if date.number < original.date
618 .number
619 {
620 throw CommandError("reverse before entry \(original.id) \(original.date.text)")
621 }
622 try checkClosed(date)
623 try active(original.lines.map(\.code))
624 let posted = reverse(original, date, .reversal, original.memo)
625 emit("REVERSE \(original.id) entry \(posted.id) \(date.text) amount \(money(lineTotal(posted.lines)))")
626 case "RECUR":
627 guard let template = templates[words[1]] else {
628 throw CommandError("unknown template \(words[1])")
629 }
630 let date = try BookDate.parse(words[2])
631 let count = Int(words[3])!
632 if count == 0 {
633 throw CommandError("invalid count")
634 }
635 try checkClosed(date)
636 try active(template.lines.map(\.code))
637 emit("RECUR \(words[1]) count \(count)")
638 for month in 0..<count {
639 template.uses += 1
640 let posted = post(
641 "\(words[1])-\(template.uses)",
642 date.addingMonths(month),
643 .recurring,
644 template.memo,
645 template.lines
646 )
647 emit(
648 " POST \(posted.id) \(posted.date.text) lines \(posted.lines.count) amount \(money(lineTotal(posted.lines)))"
649 )
650 }
651 case "DEPRECIATE":
652 let date = try BookDate.parse(words[1])
653 try checkClosed(date)
654 _ = try account(words[2], .asset)
655 _ = try account(words[3], .contraAsset)
656 _ = try account(words[4], .expense)
657 try active([words[3], words[4]])
658 let cost = balance(words[2], date.number), accumulated = balance(words[3], date.number)
659 let amount = cost > 0 ? min(cost / Cents(words[5])!, cost - accumulated) : 0
660 if amount <= 0 {
661 throw CommandError("nothing to depreciate \(words[2])")
662 }
663 depreciationCount += 1
664 let posted = post(
665 "DEP-\(depreciationCount)",
666 date,
667 .depreciation,
668 "depreciation",
669 [Line(.debit, words[4], amount), Line(.credit, words[3], amount)]
670 )
671 emit(
672 "DEPRECIATE \(words[2]) entry \(posted.id) \(date.text) amount \(money(amount)) accumulated \(money(accumulated + amount)) book \(money(cost - accumulated - amount))"
673 )
674 case "INTEREST":
675 let found = try account(words[1])
676 if found.type != .asset && found
677 .type != .liability
678 {
679 throw CommandError("not interest bearing \(found.code)")
680 }
681 _ = try account(words[2], found.type == .asset ? .revenue : .expense)
682 let (first, last) = try checkedRange(words[3], words[4])
683 try checkClosed(last)
684 try active([words[1], words[2]])
685 let sum = dailySum(found.code, first.number, last.number)
686 let amount = sum * Cents(words[5])! / 3650000
687 if amount <= 0 {
688 throw CommandError("no interest \(found.code)")
689 }
690 interestCount += 1
691 let asset = found.type == .asset
692 let posted = post("INT-\(interestCount)", last, .interest, "interest",
693 [
694 Line(.debit, asset ? words[1] : words[2], amount),
695 Line(.credit, asset ? words[2] : words[1], amount)
696 ])
697 let days = last.number - first.number + 1
698 emit(
699 "INTEREST \(found.code) entry \(posted.id) \(last.text) days \(days) average \(money(sum / Cents(days))) amount \(money(amount))"
700 )
701 case "WRITEOFF":
702 let date = try BookDate.parse(words[1])
703 try checkClosed(date)
704 _ = try account(words[2], .asset)
705 _ = try account(words[3], .expense)
706 try active([words[2], words[3]])
707 let age = Int(words[4])!
708 let groups = settlements(words[2], date.number)
709 var targets: [(String, Cents)] = []
710 var total: Cents = 0
711 for partner in groups.keys.sorted() {
712 let amount = groups[partner]!.openItems.filter { date.number - $0.entry.date.number > age }
713 .reduce(0) { $0 + $1.remaining }
714 if amount > 0 {
715 targets.append((partner, amount))
716 total += amount
717 }
718 }
719 if total == 0 {
720 throw CommandError("nothing to write off \(words[2])")
721 }
722 let lines = [Line(.debit, words[3], total)] + targets.map { Line(.credit, words[2], $0.1, $0.0) }
723 writeoffCount += 1
724 let posted = post("WO-\(writeoffCount)", date, .writeoff, "writeoff", lines)
725 emit(
726 "WRITEOFF \(words[2]) entry \(posted.id) \(date.text) partners \(targets.count) amount \(money(total))"
727 )
728 for (partner, amount) in targets {
729 emit(" \(partner) \(money(amount))")
730 }
731 case "CLOSE":
732 try close(BookDate.parse(words[1]), words[2])
733 case "REOPEN":
734 guard let record = closes.last else {
735 throw CommandError("nothing to reopen")
736 }
737 var result = "no entry"
738 if let original = record.entry {
739 try active(original.lines.map(\.code))
740 result = "entry " + reverse(original, record.date, .reopen, "reopen").id
741 }
742 closes.removeLast()
743 emit("REOPEN \(record.date.text) \(result) closed \(closed)")
744 case "LEDGER":
745 _ = try account(words[1])
746 let (first, last) = try checkedRange(words[2], words[3])
747 ledger(words[1], first, last)
748 case "ACTIVITY":
749 let found = try account(words[1])
750 let year = Int(words[2])!
751 var debit: Cents = 0, credit: Cents = 0
752 emit("ACTIVITY \(found.code) \(found.name) \(year)")
753 for month in 1...12 {
754 let first = BookDate(year, month, 1).number
755 let last = BookDate(year, month, BookDate.daysInMonth(year, month)).number
756 let total = aggregate(first, last)[found.code] ?? Totals()
757 debit += total.debit
758 credit += total.credit
759 emit(
760 " \(padded(year, 4))-\(padded(month)) debit \(money(total.debit)) credit \(money(total.credit)) balance \(money(balance(found.code, last)))"
761 )
762 }
763 emit(" total debit \(money(debit)) credit \(money(credit))")
764 case "TRIAL":
765 let date = try BookDate.parse(words[1])
766 let totals = aggregate(0, date.number)
767 var debit: Cents = 0, credit: Cents = 0
768 emit("TRIAL \(date.text)")
769 for code in codes {
770 let value = totals[code]?.difference ?? 0
771 if value == 0 {
772 continue
773 }
774 if value > 0 {
775 debit += value
776 } else {
777 credit -= value
778 }
779 emit(" \(code) \(accounts[code]!.name) \(value > 0 ? "debit" : "credit") \(money(abs(value)))")
780 }
781 emit(" total debit \(money(debit)) credit \(money(credit))")
782 case "INCOME":
783 let (first, last) = try checkedRange(words[1], words[2])
784 let totals = aggregate(first.number, last.number, operating: true)
785 emit("INCOME \(first.text) \(last.text)")
786 let revenue = section("REVENUE", [.revenue], -1, totals)
787 let expense = section("EXPENSE", [.expense], 1, totals)
788 emit(" NET INCOME \(money(revenue - expense))")
789 case "BALANCE":
790 let date = try BookDate.parse(words[1])
791 let totals = aggregate(0, date.number)
792 let earnings = -codes.filter { accounts[$0]!.type.isIncome }
793 .reduce(Cents(0)) { $0 + (totals[$1]?.difference ?? 0) }
794 emit("BALANCE \(date.text)")
795 let assets = section("ASSETS", [.asset, .contraAsset], 1, totals)
796 let liabilities = section("LIABILITIES", [.liability], -1, totals)
797 let equity = section("EQUITY", [.equity], -1, totals, earnings: earnings)
798 emit(assets == liabilities + equity ? " check ok" :
799 " check mismatch \(money(assets - liabilities - equity))")
800 case "ROLLUP":
801 let date = try BookDate.parse(words[2])
802 let targets = codes.filter { $0.hasPrefix(words[1]) }
803 emit("ROLLUP \(words[1]) \(date.text)")
804 if targets.isEmpty {
805 emit(" (none)")
806 return
807 }
808 let totals = aggregate(0, date.number)
809 var debit: Cents = 0, credit: Cents = 0
810 for code in targets {
811 let depth = targets.filter { $0 != code && code.hasPrefix($0) }.count
812 let subtotal = targets.filter { $0.hasPrefix(code) }
813 .reduce(Cents(0)) { $0 + (totals[$1]?.difference ?? 0) }
814 let own = totals[code] ?? Totals()
815 let sign = accounts[code]!.type.sign
816 let indent = String(repeating: " ", count: 2 * (1 + depth))
817 emit(
818 "\(indent)\(code) \(accounts[code]!.name) own \(money(sign * own.difference)) total \(money(sign * subtotal))"
819 )
820 debit += own.debit
821 credit += own.credit
822 }
823 emit(" postings debit \(money(debit)) credit \(money(credit))")
824 case "AGING":
825 let found = try account(words[1])
826 let date = try BookDate.parse(words[2])
827 let groups = settlements(found.code, date.number)
828 var total = [Cents](repeating: 0, count: 5)
829 emit("AGING \(found.code) \(found.name) \(date.text)")
830 for partner in groups.keys.sorted() {
831 let settled = groups[partner]!
832 if settled.openItems.isEmpty && settled.unapplied == 0 {
833 continue
834 }
835 var amounts = [Cents](repeating: 0, count: 5)
836 for item in settled.openItems {
837 let age = date.number - item.entry.date.number
838 amounts[age <= 30 ? 0 : age <= 60 ? 1 : age <= 90 ? 2 : 3] += item.remaining
839 }
840 amounts[4] = settled.unapplied
841 for i in 0..<5 {
842 total[i] += amounts[i]
843 }
844 agingLine(partner, amounts)
845 }
846 agingLine("total", total)
847 case "ITEMS":
848 let found = try account(words[1])
849 let date = try BookDate.parse(words[3])
850 let settled = settlements(found.code, date.number)[words[2]] ?? Settlement()
851 var balance = -settled.unapplied
852 emit("ITEMS \(found.code) \(found.name) \(words[2]) \(date.text)")
853 for item in settled.openItems {
854 emit(
855 " \(item.entry.date.text) \(item.entry.id) amount \(money(item.original)) open \(money(item.remaining)) age \(date.number - item.entry.date.number)"
856 )
857 balance += item.remaining
858 }
859 emit(" unapplied \(money(settled.unapplied)) balance \(money(balance))")
860 case "ENTRY":
861 let found = try entry(words[1])
862 emit("ENTRY \(found.id) \(found.date.text) \(found.category.rawValue) \(found.memo)")
863 for line in found.lines {
864 let suffix = line.partner.map { " " + $0 } ?? ""
865 emit(" \(line.side.rawValue) \(line.code) \(accounts[line.code]!.name) \(money(line.amount))\(suffix)")
866 }
867 emit(" total \(money(lineTotal(found.lines)))")
868 if let reverses = found.reverses {
869 emit(" reverses \(reverses)")
870 }
871 if let reversedBy = found.reversedBy {
872 emit(" reversed by \(reversedBy)")
873 }
874 case "DAYBOOK":
875 let (first, last) = try checkedRange(words[1], words[2])
876 let selected = ordered(first.number, last.number)
877 var total: Cents = 0
878 emit("DAYBOOK \(first.text) \(last.text)")
879 for entry in selected {
880 let amount = lineTotal(entry.lines)
881 emit(" \(entry.date.text) \(entry.id) \(entry.category.rawValue) \(money(amount)) \(entry.memo)")
882 total += amount
883 }
884 emit(" total entries \(selected.count) amount \(money(total))")
885 case "ACCOUNTS":
886 let totals = aggregate()
887 emit("ACCOUNTS")
888 for code in codes {
889 let found = accounts[code]!
890 emit(
891 " \(code) \(found.type.rawValue) \(found.name) parent \(parent(code)) balance \(money(found.type.sign * (totals[code]?.difference ?? 0))) \(found.active ? "active" : "inactive")"
892 )
893 }
894 if accounts.isEmpty {
895 emit(" (none)")
896 }
897 case "TEMPLATES":
898 emit("TEMPLATES")
899 for name in templates.keys.sorted() {
900 let template = templates[name]!
901 emit(
902 " \(name) \(template.memo) lines \(template.lines.count) amount \(money(lineTotal(template.lines))) used \(template.uses)"
903 )
904 }
905 if templates.isEmpty {
906 emit(" (none)")
907 }
908 case "STATUS":
909 let inactive = accounts.values.filter { !$0.active }.count
910 emit(
911 "STATUS accounts \(accounts.count) inactive \(inactive) entries \(entries.count) templates \(templates.count) closed \(closed) open \(draft?.key ?? "none")"
912 )
913 default:
914 preconditionFailure("unexpected command")
915 }
916 }
917}
