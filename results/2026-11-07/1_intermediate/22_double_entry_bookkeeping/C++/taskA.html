35AccountType parseType(const std::string& text) {
36 return static_cast<AccountType>(std::find(typeNames.begin(), typeNames.end(), text) - typeNames.begin());
37}

50std::string padded(int value, std::size_t width = 2) {
51 std::string text = std::to_string(value);
52 return std::string(width > text.size() ? width - text.size() : 0, '0') + text;
53}
54
55std::string money(Cents value) {
56 Cents magnitude = value < 0 ? -value : value;
57 return (value < 0 ? "-" : "") + std::to_string(magnitude / 100) + "." + padded(static_cast<int>(magnitude % 100));
58}
59
60Cents parseMoney(const std::string& text) {
61 auto dot = text.find('.');
62 return std::stoll(text.substr(0, dot)) * 100 + std::stoll(text.substr(dot + 1));
63}

65bool leap(int year) { return year % 4 == 0 && (year % 100 != 0 || year % 400 == 0); }
66
67int monthLength(int year, int month) {
68 constexpr std::array<int, 12> lengths = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
69 return lengths[month - 1] + (month == 2 && leap(year) ? 1 : 0);
70}
71
72int dayNumber(int year, int month, int day) {
73 int result = day - 1;
74 for (int y = 2000; y < year; ++y) result += leap(y) ? 366 : 365;
75 for (int m = 1; m < month; ++m) result += monthLength(year, m);
76 return result;
77}
78
79struct CalendarDate { int year, month, day; };
80
81CalendarDate calendar(int number) {
82 int year = 2000, month = 1;
83 while (number >= (leap(year) ? 366 : 365)) {
84 number -= leap(year) ? 366 : 365;
85 ++year;
86 }
87 while (number >= monthLength(year, month)) {
88 number -= monthLength(year, month);
89 ++month;
90 }
91 return {year, month, number + 1};
92}
93
94std::string dateText(int number) {
95 auto date = calendar(number);
96 return padded(date.year, 4) + "-" + padded(date.month) + "-" + padded(date.day);
97}
98
99int checkedDate(const std::string& text) {
100 int year = std::stoi(text.substr(0, 4)), month = std::stoi(text.substr(5, 2)), day = std::stoi(text.substr(8, 2));
101 if (year < 2000 || year > 2099 || month < 1 || month > 12 || day < 1 || day > monthLength(year, month)) {
102 throw CommandError("invalid date " + text);
103 }
104 return dayNumber(year, month, day);
105}
106
107std::pair<int, int> checkedRange(const std::string& first, const std::string& last) {
108 int start = checkedDate(first), end = checkedDate(last);
109 if (start > end) throw CommandError("invalid range " + first + " " + last);
110 return {start, end};
111}
112
113int monthsAfter(int number, int count) {
114 auto original = calendar(number);
115 int offset = original.month - 1 + count;
116 int year = original.year + offset / 12, month = offset % 12 + 1;
117 return dayNumber(year, month, std::min(original.day, monthLength(year, month)));
118}

198 Account& account(const std::string& code, std::optional<AccountType> type = std::nullopt) {
199 auto found = accounts.find(code);
200 if (found == accounts.end()) throw CommandError("unknown account " + code);
201 if (type && found->second.type != *type) throw CommandError("wrong type " + code + " " + label(*type));
202 return found->second;

225 std::vector<std::string> ancestors(const std::string& code) const {
226 std::vector<std::string> result;

228 auto prefix = code.substr(0, length);

244 std::vector<const Entry*> ordered(int first = 0, int last = lastDay) const {
245 std::vector<const Entry*> result;
246 for (const auto& entry : entries) if (first <= entry.day && entry.day <= last) result.push_back(&entry);
247 std::sort(result.begin(), result.end(), [](const Entry* a, const Entry* b) {
248 return a->day == b->day ? a->sequence < b->sequence : a->day < b->day;
249 });
250 return result;
251 }
252
253 TotalsByCode aggregate(int first = 0, int last = lastDay, bool operating = false) const {
254 TotalsByCode result;
255 for (const auto& entry : entries) {
256 if (entry.day < first || entry.day > last ||
257 (operating && (entry.category == Category::Closing || entry.category == Category::Reopen))) continue;
258 for (const auto& line : entry.lines) {
259 auto& total = result[line.code];
260 (line.debit ? total.debit : total.credit) += line.amount;
261 }
262 }
263 return result;
264 }
265
266 Cents balance(const std::string& code, int day = lastDay) const {
267 Cents difference = 0;
268 for (const auto& entry : entries) {
269 if (entry.day > day) continue;
270 for (const auto& line : entry.lines) if (line.code == code) difference += line.difference();
271 }
272 return difference * normalSign(accounts.at(code).type);
273 }
274
275 Cents dailySum(const std::string& code, int first, int last) const {
276 std::vector<Cents> changes(last - first + 1);
277 Cents running = 0, sum = 0;
278 int sign = normalSign(accounts.at(code).type);
279 for (const auto& entry : entries) {
280 if (entry.day > last) continue;
281 for (const auto& line : entry.lines) {
282 if (line.code != code) continue;
283 Cents value = sign * line.difference();
284 if (entry.day < first) running += value;
285 else changes[entry.day - first] += value;
286 }
287 }
288 for (Cents change : changes) { running += change; sum += running; }
289 return sum;
290 }

292 Entry& post(const std::string& id, int day, Category category, const std::string& memo, const std::vector<Line>& lines) {
293 entries.push_back({id, day, category, memo, lines, entries.size() + 1, std::nullopt, std::nullopt});
294 Entry& posted = entries.back();
295 entryIds[id] = &posted;
296 return posted;
297 }

299 Entry& reverseEntry(Entry& original, int day, Category category, const std::string& memo) {
300 auto lines = original.lines;
301 for (auto& line : lines) line.debit = !line.debit;
302 Entry& posted = post(original.id + "-R", day, category, memo, lines);
303 original.reversedBy = posted.id;
304 posted.reverses = original.id;
305 return posted;
306 }

308 std::map<std::string, Settlement> settlements(const std::string& code, int day) const {
309 std::map<std::string, Settlement> result;
310 int sign = normalSign(accounts.at(code).type);
311 for (const Entry* entry : ordered(0, day)) {
312 for (const auto& line : entry->lines) {
313 if (line.code != code || !line.partner) continue;
314 auto& settled = result[*line.partner];
315 Cents value = sign * line.difference();
316 if (value > 0) {
317 Cents applied = std::min(value, settled.unapplied);
318 settled.unapplied -= applied;
319 if (value > applied) settled.items.push_back({entry, value, value - applied});
320 } else {
321 Cents remaining = -value;
322 while (remaining > 0 && !settled.items.empty()) {
323 auto& item = settled.items.front();
324 Cents applied = std::min(remaining, item.remaining);
325 remaining -= applied;
326 item.remaining -= applied;
327 if (item.remaining == 0) settled.items.pop_front();
328 }
329 settled.unapplied += remaining;
330 }
331 }
332 }
333 return result;
334 }

336 void openDraft(bool journal, const std::string& key, const std::string& date, const std::string& memo) {
337 if (draft) emit("DISCARD", draft->key, "lines", draft->lines.size());
338 draft = Draft{journal, key, date, memo, {}};
339 }
340
341 void postDraft() {
342 if (!draft) throw CommandError("no open entry");
343 Draft block = std::move(*draft);
344 draft.reset();
345 int day = 0;
346 if (block.journal) {
347 if (entryIds.count(block.key)) throw CommandError("duplicate entry " + block.key);
348 day = checkedDate(block.date);
349 checkClosed(day);
350 } else if (templates.count(block.key)) throw CommandError("duplicate template " + block.key);
351 if (block.lines.size() < 2) throw CommandError("too few lines " + block.key);
352 for (const auto& line : block.lines) {
353 auto& found = account(line.code);
354 if (block.journal && !found.active) throw CommandError("inactive account " + line.code);
355 if (line.amount == 0) throw CommandError("zero amount " + line.code);
356 }
357 Cents debit = lineTotal(block.lines), credit = lineTotal(block.lines, false);
358 if (debit != credit) throw CommandError("unbalanced " + block.key + " debit " + money(debit) + " credit " + money(credit));
359 if (block.journal) {
360 auto& posted = post(block.key, day, Category::Journal, block.memo, block.lines);
361 emit("POST", posted.id, dateText(day), "lines", posted.lines.size(), "amount", money(debit));
362 } else {
363 templates.emplace(block.key, Template{block.memo, block.lines});
364 emit("TEMPLATE", block.key, "lines", block.lines.size(), "amount", money(debit));
365 }
366 }

368 void close(int day, const std::string& equity) {
369 checkClosed(day);
370 account(equity, AccountType::Equity);
371 active({equity});
372 auto totals = aggregate(0, day);
373 std::vector<std::string> targets;
374 for (const auto& [code, found] : accounts) {
375 if (incomeType(found.type) && getTotals(totals, code).difference() != 0) targets.push_back(code);
376 }
377 active(targets);
378 std::vector<Line> lines;
379 Cents income = 0;
380 for (const auto& code : targets) {
381 Cents value = getTotals(totals, code).difference();
382 lines.push_back({value < 0, code, value < 0 ? -value : value, std::nullopt});
383 income -= value;
384 }
385 if (income != 0) lines.push_back({income < 0, equity, income < 0 ? -income : income, std::nullopt});
386 ++closeCount;
387 Entry* posted = nullptr;
388 if (!lines.empty()) posted = &post("CLOSE-" + std::to_string(closeCount), day, Category::Closing, "closing", lines);
389 closes.push_back({day, posted});
390 if (posted) emit("CLOSE", dateText(day), "entry", posted->id, "lines", lines.size(), "net income", money(income));
391 else emit("CLOSE", dateText(day), "no entry");
392 }

394 Cents section(const std::string& heading, const std::set<AccountType>& types, int sign,
395 const TotalsByCode& totals, std::optional<Cents> earnings = std::nullopt) const {
396 std::map<std::string, std::vector<std::pair<std::string, Cents>>> groups;
397 Cents total = earnings.value_or(0);
398 for (const auto& [code, found] : accounts) {
399 if (!types.count(found.type)) continue;
400 Cents value = sign * getTotals(totals, code).difference();
401 groups[top(code)].push_back({code, value});
402 total += value;
403 }
404 indented(2, heading, money(total));
405 for (const auto& [root, group] : groups) {
406 bool nonzero = false;
407 Cents subtotal = 0;
408 for (const auto& item : group) { nonzero = nonzero || item.second != 0; subtotal += item.second; }
409 if (!nonzero) continue;
410 indented(4, root, accounts.at(root).name, money(subtotal));
411 for (const auto& [code, value] : group) if (value != 0) indented(6, code, accounts.at(code).name, money(value));
412 }
413 if (earnings) indented(4, "current earnings", money(*earnings));
414 return total;
415 }

417 void ledger(const std::string& code, int first, int last) const {
418 const auto& found = accounts.at(code);
419 Cents running = balance(code, first - 1), debit = 0, credit = 0;
420 emit("LEDGER", code, found.name, dateText(first), dateText(last));
421 indented(2, "opening", money(running));
422 for (const Entry* entry : ordered(first, last)) {
423 for (const auto& line : entry->lines) {
424 if (line.code != code) continue;
425 running += normalSign(found.type) * line.difference();
426 (line.debit ? debit : credit) += line.amount;
427 indented(2, dateText(entry->day), entry->id, line.side(), money(line.amount), "balance", money(running), entry->memo);
428 }
429 }
430 int days = last - first + 1;
431 indented(2, "ending", money(running), "debits", money(debit), "credits", money(credit),
432 "days", days, "average", money(dailySum(code, first, last) / days));
433 }

435 void agingLine(const std::string& partner, const std::array<Cents, 5>& amounts) const {
436 Cents balance = amounts[0] + amounts[1] + amounts[2] + amounts[3] - amounts[4];
437 indented(2, partner, "0-30", money(amounts[0]), "31-60", money(amounts[1]), "61-90", money(amounts[2]),
438 "91+", money(amounts[3]), "unapplied", money(amounts[4]), "balance", money(balance));
439 }

441public:
442 void execute(const std::vector<std::string>& words) {
443 const auto& command = words[0];
444 if (command == "ACCOUNT") {
445 const auto& code = words[1];
446 if (accounts.count(code)) throw CommandError("duplicate account " + code);
447 auto type = parseType(words[2]);
448 for (const auto& [other, found] : accounts) {
449 if ((startsWith(code, other) || startsWith(other, code)) && accountClass(type) != accountClass(found.type)) {
450 throw CommandError("class mismatch " + code + " " + other);
451 }
452 }
453 accounts.emplace(code, Account{code, words[3], type});
454 emit("ACCOUNT", code, label(type), words[3], "parent", parent(code));
455 } else if (command == "DEACTIVATE") {
456 auto& found = account(words[1]);
457 if (!found.active) throw CommandError("already inactive " + found.code);
458 Cents value = balance(found.code);
459 if (value != 0) throw CommandError("nonzero balance " + found.code + " " + money(value));
460 found.active = false;
461 emit("DEACTIVATE", found.code, found.name);
462 } else if (command == "JOURNAL") {
463 openDraft(true, words[1], words[2], words[3]);
464 } else if (command == "TEMPLATE") {
465 openDraft(false, words[1], "", words[2]);
466 } else if (command == "DR" || command == "CR") {
467 if (!draft) throw CommandError("no open entry");
468 std::optional<std::string> partner = words.size() == 4 ? std::optional<std::string>(words[3]) : std::nullopt;
469 draft->lines.push_back({command == "DR", words[1], parseMoney(words[2]), partner});
470 } else if (command == "POST") {
471 postDraft();
472 } else if (command == "DRAFT") {
473 if (!draft) throw CommandError("no open entry");
474 Cents debit = lineTotal(draft->lines), credit = lineTotal(draft->lines, false);
475 emit("DRAFT", draft->journal ? "journal" : "template", draft->key, "lines", draft->lines.size(),
476 "debit", money(debit), "credit", money(credit), "difference", money(debit - credit));
477 } else if (command == "REVERSE") {
478 auto& original = entry(words[1]);
479 if (original.category == Category::Reversal || original.category == Category::Closing || original.category == Category::Reopen) {
480 throw CommandError("cannot reverse " + original.id);
481 }
482 if (original.reversedBy) throw CommandError("already reversed " + original.id);
483 int day = checkedDate(words[2]);
484 if (day < original.day) throw CommandError("reverse before entry " + original.id + " " + dateText(original.day));
485 checkClosed(day);
486 activeLines(original.lines);
487 auto& posted = reverseEntry(original, day, Category::Reversal, original.memo);
488 emit("REVERSE", original.id, "entry", posted.id, dateText(day), "amount", money(lineTotal(posted.lines)));
489 } else if (command == "RECUR") {
490 auto found = templates.find(words[1]);
491 if (found == templates.end()) throw CommandError("unknown template " + words[1]);
492 int day = checkedDate(words[2]), count = std::stoi(words[3]);
493 if (count == 0) throw CommandError("invalid count");
494 checkClosed(day);
495 auto& model = found->second;
496 activeLines(model.lines);
497 emit("RECUR", words[1], "count", count);
498 for (int month = 0; month < count; ++month) {
499 auto& posted = post(words[1] + "-" + std::to_string(++model.uses), monthsAfter(day, month), Category::Recurring, model.memo, model.lines);
500 indented(2, "POST", posted.id, dateText(posted.day), "lines", posted.lines.size(), "amount", money(lineTotal(posted.lines)));
501 }
502 } else if (command == "DEPRECIATE") {
503 int day = checkedDate(words[1]);
504 checkClosed(day);
505 account(words[2], AccountType::Asset);
506 account(words[3], AccountType::ContraAsset);
507 account(words[4], AccountType::Expense);
508 active({words[3], words[4]});
509 Cents cost = balance(words[2], day), accumulated = balance(words[3], day);
510 Cents amount = cost > 0 ? std::min(cost / std::stoi(words[5]), cost - accumulated) : 0;
511 if (amount <= 0) throw CommandError("nothing to depreciate " + words[2]);
512 auto& posted = post("DEP-" + std::to_string(++depreciationCount), day, Category::Depreciation, "depreciation",
513 {{true, words[4], amount, std::nullopt}, {false, words[3], amount, std::nullopt}});
514 emit("DEPRECIATE", words[2], "entry", posted.id, dateText(day), "amount", money(amount),
515 "accumulated", money(accumulated + amount), "book", money(cost - accumulated - amount));
516 } else if (command == "INTEREST") {
517 auto& found = account(words[1]);
518 if (found.type != AccountType::Asset && found.type != AccountType::Liability) throw CommandError("not interest bearing " + found.code);
519 account(words[2], found.type == AccountType::Asset ? AccountType::Revenue : AccountType::Expense);
520 auto [first, last] = checkedRange(words[3], words[4]);
521 checkClosed(last);
522 active({words[1], words[2]});
523 Cents sum = dailySum(found.code, first, last), amount = sum * std::stoi(words[5]) / 3650000;
524 if (amount <= 0) throw CommandError("no interest " + found.code);
525 bool asset = found.type == AccountType::Asset;
526 auto& posted = post("INT-" + std::to_string(++interestCount), last, Category::Interest, "interest",
527 {{true, asset ? words[1] : words[2], amount, std::nullopt},
528 {false, asset ? words[2] : words[1], amount, std::nullopt}});
529 int days = last - first + 1;
530 emit("INTEREST", found.code, "entry", posted.id, dateText(last), "days", days, "average", money(sum / days), "amount", money(amount));
531 } else if (command == "WRITEOFF") {
532 int day = checkedDate(words[1]);
533 checkClosed(day);
534 account(words[2], AccountType::Asset);
535 account(words[3], AccountType::Expense);
536 active({words[2], words[3]});
537 int age = std::stoi(words[4]);
538 std::map<std::string, Cents> targets;
539 Cents total = 0;
540 for (const auto& [partner, settled] : settlements(words[2], day)) {
541 Cents amount = 0;
542 for (const auto& item : settled.items) if (day - item.entry->day > age) amount += item.remaining;
543 if (amount > 0) { targets[partner] = amount; total += amount; }
544 }
545 if (total == 0) throw CommandError("nothing to write off " + words[2]);
546 std::vector<Line> lines = {{true, words[3], total, std::nullopt}};
547 for (const auto& [partner, amount] : targets) lines.push_back({false, words[2], amount, partner});
548 auto& posted = post("WO-" + std::to_string(++writeoffCount), day, Category::Writeoff, "writeoff", lines);
549 emit("WRITEOFF", words[2], "entry", posted.id, dateText(day), "partners", targets.size(), "amount", money(total));
550 for (const auto& [partner, amount] : targets) indented(2, partner, money(amount));
551 } else if (command == "CLOSE") {
552 close(checkedDate(words[1]), words[2]);
553 } else if (command == "REOPEN") {
554 if (closes.empty()) throw CommandError("nothing to reopen");
555 auto record = closes.back();
556 std::string result = "no entry";
557 if (record.entry) {
558 activeLines(record.entry->lines);
559 result = "entry " + reverseEntry(*record.entry, record.day, Category::Reopen, "reopen").id;
560 }
561 closes.pop_back();
562 emit("REOPEN", dateText(record.day), result, "closed", closedText());
563 } else if (command == "LEDGER") {
564 account(words[1]);
565 auto [first, last] = checkedRange(words[2], words[3]);
566 ledger(words[1], first, last);
567 } else if (command == "ACTIVITY") {
568 auto& found = account(words[1]);
569 int year = std::stoi(words[2]);
570 Cents debit = 0, credit = 0;
571 emit("ACTIVITY", found.code, found.name, year);
572 for (int month = 1; month <= 12; ++month) {
573 int first = dayNumber(year, month, 1), last = dayNumber(year, month, monthLength(year, month));
574 auto total = getTotals(aggregate(first, last), found.code);
575 debit += total.debit;
576 credit += total.credit;
577 indented(2, padded(year, 4) + "-" + padded(month), "debit", money(total.debit), "credit", money(total.credit), "balance", money(balance(found.code, last)));
578 }
579 indented(2, "total debit", money(debit), "credit", money(credit));
580 } else if (command == "TRIAL") {
581 int day = checkedDate(words[1]);
582 auto totals = aggregate(0, day);
583 Cents debit = 0, credit = 0;
584 emit("TRIAL", dateText(day));
585 for (const auto& [code, found] : accounts) {
586 Cents value = getTotals(totals, code).difference();
587 if (value == 0) continue;
588 if (value > 0) debit += value; else credit -= value;
589 indented(2, code, found.name, value > 0 ? "debit" : "credit", money(value > 0 ? value : -value));
590 }
591 indented(2, "total debit", money(debit), "credit", money(credit));
592 } else if (command == "INCOME") {
593 auto [first, last] = checkedRange(words[1], words[2]);
594 auto totals = aggregate(first, last, true);
595 emit("INCOME", dateText(first), dateText(last));
596 Cents revenue = section("REVENUE", {AccountType::Revenue}, -1, totals);
597 Cents expense = section("EXPENSE", {AccountType::Expense}, 1, totals);
598 indented(2, "NET INCOME", money(revenue - expense));
599 } else if (command == "BALANCE") {
600 int day = checkedDate(words[1]);
601 auto totals = aggregate(0, day);
602 Cents earnings = 0;
603 for (const auto& [code, found] : accounts) if (incomeType(found.type)) earnings -= getTotals(totals, code).difference();
604 emit("BALANCE", dateText(day));
605 Cents assets = section("ASSETS", {AccountType::Asset, AccountType::ContraAsset}, 1, totals);
606 Cents liabilities = section("LIABILITIES", {AccountType::Liability}, -1, totals);
607 Cents equity = section("EQUITY", {AccountType::Equity}, -1, totals, earnings);
608 if (assets == liabilities + equity) indented(2, "check ok");
609 else indented(2, "check mismatch", money(assets - liabilities - equity));
610 } else if (command == "ROLLUP") {
611 int day = checkedDate(words[2]);
612 std::vector<std::string> targets;
613 for (const auto& [code, found] : accounts) if (startsWith(code, words[1])) targets.push_back(code);
614 emit("ROLLUP", words[1], dateText(day));
615 if (targets.empty()) { indented(2, "(none)"); return; }
616 auto totals = aggregate(0, day);
617 Cents debit = 0, credit = 0;
618 for (const auto& code : targets) {
619 int depth = 0;
620 Cents subtotal = 0;
621 for (const auto& other : targets) {
622 if (other != code && startsWith(code, other)) ++depth;
623 if (startsWith(other, code)) subtotal += getTotals(totals, other).difference();
624 }
625 auto own = getTotals(totals, code);
626 int sign = normalSign(accounts.at(code).type);
627 indented(2 * (1 + depth), code, accounts.at(code).name, "own", money(sign * own.difference()), "total", money(sign * subtotal));
628 debit += own.debit;
629 credit += own.credit;
630 }
631 indented(2, "postings debit", money(debit), "credit", money(credit));
632 } else if (command == "AGING") {
633 auto& found = account(words[1]);
634 int day = checkedDate(words[2]);
635 emit("AGING", found.code, found.name, dateText(day));
636 std::array<Cents, 5> totals{};
637 for (const auto& [partner, settled] : settlements(found.code, day)) {
638 if (settled.items.empty() && settled.unapplied == 0) continue;
639 std::array<Cents, 5> amounts{};
640 amounts[4] = settled.unapplied;
641 for (const auto& item : settled.items) {
642 int age = day - item.entry->day;
643 amounts[age <= 30 ? 0 : age <= 60 ? 1 : age <= 90 ? 2 : 3] += item.remaining;
644 }
645 agingLine(partner, amounts);
646 for (std::size_t i = 0; i < amounts.size(); ++i) totals[i] += amounts[i];
647 }
648 agingLine("total", totals);
649 } else if (command == "ITEMS") {
650 auto& found = account(words[1]);
651 int day = checkedDate(words[3]);
652 auto groups = settlements(found.code, day);
653 const auto& settled = groups[words[2]];
654 emit("ITEMS", found.code, found.name, words[2], dateText(day));
655 Cents balance = -settled.unapplied;
656 for (const auto& item : settled.items) {
657 indented(2, dateText(item.entry->day), item.entry->id, "amount", money(item.original), "open", money(item.remaining), "age", day - item.entry->day);
658 balance += item.remaining;
659 }
660 indented(2, "unapplied", money(settled.unapplied), "balance", money(balance));
661 } else if (command == "ENTRY") {
662 auto& found = entry(words[1]);
663 emit("ENTRY", found.id, dateText(found.day), label(found.category), found.memo);
664 for (const auto& line : found.lines) {
665 std::string suffix = line.partner ? " " + *line.partner : "";
666 indented(2, line.side(), line.code, accounts.at(line.code).name, money(line.amount) + suffix);
667 }
668 indented(2, "total", money(lineTotal(found.lines)));
669 if (found.reverses) indented(2, "reverses", *found.reverses);
670 if (found.reversedBy) indented(2, "reversed by", *found.reversedBy);
671 } else if (command == "DAYBOOK") {
672 auto [first, last] = checkedRange(words[1], words[2]);
673 auto selected = ordered(first, last);
674 Cents total = 0;
675 emit("DAYBOOK", dateText(first), dateText(last));
676 for (const Entry* entry : selected) {
677 Cents amount = lineTotal(entry->lines);
678 indented(2, dateText(entry->day), entry->id, label(entry->category), money(amount), entry->memo);
679 total += amount;
680 }
681 indented(2, "total entries", selected.size(), "amount", money(total));
682 } else if (command == "ACCOUNTS") {
683 auto totals = aggregate();
684 emit("ACCOUNTS");
685 for (const auto& [code, found] : accounts) {
686 indented(2, code, label(found.type), found.name, "parent", parent(code), "balance",
687 money(normalSign(found.type) * getTotals(totals, code).difference()), found.active ? "active" : "inactive");
688 }
689 if (accounts.empty()) indented(2, "(none)");
690 } else if (command == "TEMPLATES") {
691 emit("TEMPLATES");
692 for (const auto& [name, model] : templates) {
693 indented(2, name, model.memo, "lines", model.lines.size(), "amount", money(lineTotal(model.lines)), "used", model.uses);
694 }
695 if (templates.empty()) indented(2, "(none)");
696 } else if (command == "STATUS") {
697 auto inactive = std::count_if(accounts.begin(), accounts.end(), [](const auto& pair) { return !pair.second.active; });
698 emit("STATUS accounts", accounts.size(), "inactive", inactive, "entries", entries.size(),
699 "templates", templates.size(), "closed", closedText(), "open", draft ? draft->key : "none");
700 }
701 }
