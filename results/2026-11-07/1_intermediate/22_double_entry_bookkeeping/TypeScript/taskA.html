37function leap(year: number): boolean {
38 return year % 4 === 0 && (year % 100 !== 0 || year % 400 === 0);
39}

41function monthLength(year: number, month: number): number {
42 return month === 2 && leap(year) ? 29 : MONTH_LENGTHS[month - 1];
43}

45function dayNumber(year: number, month: number, day: number): number {
46 let result = day - 1;
47 for (let y = 2000; y < year; y++) result += leap(y) ? 366 : 365;
48 for (let m = 1; m < month; m++) result += monthLength(year, m);
49 return result;
50}

52function calendar(number: number): CalendarDate {
53 let year = 2000;
54 while (number >= (leap(year) ? 366 : 365)) {
55 number -= leap(year) ? 366 : 365;
56 year++;
57 }
58 let month = 1;
59 while (number >= monthLength(year, month)) {
60 number -= monthLength(year, month);
61 month++;
62 }
63 return {year, month, day: number + 1};
64}
