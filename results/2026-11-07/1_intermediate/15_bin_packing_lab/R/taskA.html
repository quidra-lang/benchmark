45pack <- function(items, capacity, method, parameter = 0) {
46 result <- list(method = method, parameter = parameter, bins = list(),
47 steps = list(), inspections = 0, closes = 0)
48 order_items <- if (method %in% c("FFD", "BFD", "WFD"))
49 descending(items) else items
50 active <- integer()
51 current <- list()
52 for (item in order_items) {
53 size <- item[2]
54 inspected_before <- result$inspections
55 chosen <- NA_integer_
56 closed <- NA_integer_
57 category <- NA_integer_
58 count <- length(result$bins)
59 if (method == "NEXTFIT") {
60 if (count != 0) {
61 result$inspections <- result$inspections + 1
62 if (result$bins[[count]]$load + size <= capacity) chosen <- count
63 }
64 } else if (method %in% c("FIRSTFIT", "FFD", "KBOUNDED")) {
65 candidates <- if (method == "KBOUNDED") active else seq_len(count)
66 for (index in candidates) {
67 result$inspections <- result$inspections + 1
68 if (result$bins[[index]]$load + size <= capacity) {
69 chosen <- index
70 break
71 }
72 }
73 } else if (method == "HARMONIC") {
74 category <- min(floor(capacity / size), parameter)
75 old <- current[[as.character(category)]]
76 if (!is.null(old)) {
77 result$inspections <- result$inspections + 1
78 if (result$bins[[old]]$load + size <= capacity) chosen <- old
79 }
80 } else {
81 result$inspections <- result$inspections + count
82 fitting <- which(vapply(result$bins, function(bin)
83 bin$load + size <= capacity, logical(1)))
84 if (length(fitting)) {
85 loads <- vapply(result$bins[fitting], function(bin) bin$load, numeric(1))
86 if (method %in% c("BESTFIT", "BFD")) {
87 chosen <- fitting[which.max(loads)]
88 } else {
89 sorted <- fitting[order(loads, fitting)]
90 chosen <- if (method == "ALMOSTWORST" && length(sorted) >= 2)
91 sorted[2] else sorted[1]
92 }
93 }
94 }
95 opened <- is.na(chosen)
96 if (opened) {
97 if (method == "KBOUNDED" && length(active) == parameter) {
98 loads <- vapply(result$bins[active], function(bin) bin$load, numeric(1))
99 closed <- active[which.max(loads)]
100 active <- active[active != closed]
101 result$closes <- result$closes + 1
102 }
103 chosen <- length(result$bins) + 1
104 result$bins[[chosen]] <- new_bin(if (!is.na(category)) category else NULL)
105 if (method == "KBOUNDED") active <- c(active, chosen)
106 if (method == "HARMONIC") current[[as.character(category)]] <- chosen
107 }
108 result$bins[[chosen]] <- put(result$bins[[chosen]], item)
109 result$steps[[length(result$steps) + 1]] <- list(
110 item = item, bin = chosen, inspected = result$inspections - inspected_before,
111 opened = opened, closed = closed, category = category
112 )
113 }
114 result
115}

117bounds <- function(items, capacity) {
118 sizes <- sizes_of(items)
119 thresholds <- sort(unique(c(0, sizes[2 * sizes <= capacity])))
120 rows <- lapply(thresholds, function(alpha) {
121 first <- sizes[sizes > capacity - alpha]
122 second <- sizes[2 * sizes > capacity & sizes <= capacity - alpha]
123 third <- sizes[2 * sizes <= capacity & sizes >= alpha]
124 excess <- sum(third) - (length(second) * capacity - sum(second))
125 bound <- length(first) + length(second) +
126 if (excess > 0) ceiling(excess, capacity) else 0
127 c(alpha, length(first), length(second), length(third), bound)
128 })
129 list(l1 = ceiling(sum(sizes), capacity),
130 l2 = max(vapply(rows, function(row) row[5], numeric(1))), rows = rows)
131}

133exact_search <- function(items, capacity, limit, lower, incumbent, emit) {
134 search <- new.env(parent = emptyenv())
135 search$items <- descending(items)
136 search$capacity <- capacity
137 search$limit <- limit
138 search$lower <- lower
139 search$best <- incumbent
140 search$live <- list()
141 search$nodes <- 0
142 search$status <- ""
143 k <- length(items)
144 remaining <- numeric(k + 1)
145 remaining[k + 1] <- 0
146 for (i in k:1) {
147 remaining[i] <- remaining[i + 1] + search$items[[i]][2]
148 }
149 search$remaining <- remaining
150 visit <- function(index) {
151 if (search$nodes == search$limit) {
152 search$status <- "limit"
153 return(invisible(NULL))
154 }
155 search$nodes <- search$nodes + 1
156 count <- length(search$live)
157 used <- remaining[1] - remaining[index]
158 free <- count * capacity - used
159 excess <- remaining[index] - free
160 minimum <- count + if (excess > 0) ceiling(excess, capacity) else 0
161 if (minimum >= length(search$best)) return(invisible(NULL))
162 if (index == k + 1) {
163 search$best <- search$live
164 emit(paste0("IMPROVE bins=", count, " nodes=", search$nodes))
165 if (count == lower) search$status <- "bound"
166 return(invisible(NULL))
167 }
168 item <- search$items[[index]]
169 seen_loads <- numeric()
170 for (i in seq_len(count)) {
171 old <- search$live[[i]]
172 if (old$load %in% seen_loads) next
173 seen_loads <- c(seen_loads, old$load)
174 if (old$load + item[2] > capacity) next
175 search$live[[i]] <- put(old, item)
176 visit(index + 1)
177 search$live[[i]] <- old
178 if (search$status != "") return(invisible(NULL))
179 }
180 if (count + 1 < length(search$best)) {
181 search$live[[count + 1]] <- put(new_bin(), item)
182 visit(index + 1)
183 search$live <- search$live[seq_len(count)]
184 }
185 invisible(NULL)
186 }
187 if (length(incumbent) == lower) {
188 search$status <- "bound"
189 } else {
190 visit(1)
191 }
192 search
193}

281cmd_reduce <- function(arguments) {
282 method <- arguments[1]
283 result <- method_result(method, arguments[-1])
284 bins <- result$bins
285 before <- length(bins)
286 events <- character()
287 repeat {
288 eliminated <- FALSE
289 loads <- vapply(bins, function(bin) bin$load, numeric(1))
290 for (source in order(loads, seq_along(bins))) {
291 virtual <- loads
292 moves <- list()
293 source_items <- descending(bins[[source]]$items)
294 success <- TRUE
295 for (item in source_items) {
296 fitting <- which(seq_along(bins) != source & virtual + item[2] <= lab$capacity)
297 if (!length(fitting)) {
298 success <- FALSE
299 break
300 }
301 destination <- fitting[which.max(virtual[fitting])]
302 virtual[destination] <- virtual[destination] + item[2]
303 moves[[length(moves) + 1]] <- list(item = item, destination = destination)
304 }
305 if (!success) next
306 descriptions <- vapply(moves, function(move)
307 paste0(item_text(move$item), "->", move$destination), character(1))
308 events <- c(events, paste0("ELIMINATE bin=", source,
309 " load=", bins[[source]]$load,
310 " moves=", paste(descriptions, collapse = " ")))
311 for (move in moves) {
312 destination <- move$destination
313 bins[[destination]] <- put(bins[[destination]], move$item)
314 }
315 bins <- bins[-source]
316 eliminated <- TRUE
317 break
318 }
319 if (!eliminated) break
320 }
321 emit(paste0("REDUCE ", packing_name(result), " before=", before,
322 " after=", length(bins), " eliminated=", before - length(bins)))
323 for (event in events) emit(event)
324 bin_lines(bins, method == "HARMONIC")
325}

427cmd_multifit <- function(arguments) {
428 count <- as.numeric(arguments[1])
429 if (count == 0) fail("BAD PARAMETER")
430 total <- total_size()
431 largest <- max(sizes_of(lab$items))
432 lower <- max(ceiling(total, count), largest)
433 upper <- max(ceiling(2 * total, count), largest)
434 emit(paste0("MULTIFIT m=", count, " lower=", lower, " upper=", upper))
435 while (lower < upper) {
436 middle <- floor((lower + upper) / 2)
437 n <- length(pack(lab$items, middle, "FFD")$bins)
438 success <- n <= count
439 emit(paste0("TRY cap=", middle, " bins=", n,
440 " ", if (success) "ok" else "fail"))
441 if (success) upper <- middle else lower <- middle + 1
442 }
443 machines <- pack(lab$items, upper, "FFD")$bins
444 while (length(machines) < count) machines[[length(machines) + 1]] <- new_bin()
445 emit(paste0("RESULT cap=", upper, " makespan=",
446 max(vapply(machines, function(bin) bin$load, numeric(1)))))
447 machine_lines(machines)
448 lpt <- replicate(count, new_bin(), simplify = FALSE)
449 for (item in descending(lab$items)) {

452 lpt[[index]] <- put(lpt[[index]], item)

458maximal_patterns <- function(length_bar, pieces) {
459 patterns <- list()
460 visit <- function(index, remaining, prefix) {
461 size <- pieces[[index]][1]
462 maximum <- floor(remaining / size)
463 if (index == length(pieces)) {
464 patterns[[length(patterns) + 1]] <<-
465 list(counts = c(prefix, maximum), trim = remaining - maximum * size)
466 return(invisible(NULL))
467 }
468 for (quantity in seq.int(maximum, 0)) {
469 visit(index + 1, remaining - quantity * size, c(prefix, quantity))
470 }
471 invisible(NULL)
472 }
473 visit(1, length_bar, numeric())
474 patterns
475}
476cmd_cutstock <- function(arguments) {
477 length_bar <- as.numeric(arguments[1])
478 if (length_bar == 0) fail("BAD CAPACITY")
479 tokens <- arguments[-1]
480 pieces <- list()
481 seen <- numeric()
482 for (i in seq_along(tokens)) {
483 pair <- as.numeric(strsplit(tokens[i], ":", fixed = TRUE)[[1]])
484 size <- pair[1]
485 demand <- pair[2]
486 if (size == 0 || size > length_bar || demand == 0)
487 fail(paste("BAD PIECE", i))
488 if (size %in% seen) fail(paste("DUPLICATE PIECE", i))
489 seen <- c(seen, size)
490 pieces[[length(pieces) + 1]] <- pair
491 }
492 sizes <- vapply(pieces, function(x) x[1], numeric(1))
493 pieces <- pieces[order(-sizes)]
494 patterns <- maximal_patterns(length_bar, pieces)
495 remaining <- vapply(pieces, function(p) p[2], numeric(1))
496 uses <- numeric(length(patterns))
497 while (any(remaining > 0)) {
498 useful <- vapply(patterns, function(p)
499 sum(pmin(p$counts, remaining)), numeric(1))
500 chosen <- which.max(useful)
501 uses[chosen] <- uses[chosen] + 1
502 remaining <- pmax(0, remaining - patterns[[chosen]]$counts)
503 }
504 cut <- vapply(seq_along(pieces), function(i)
505 sum(vapply(seq_along(patterns), function(j)
506 uses[j] * patterns[[j]]$counts[i], numeric(1))), numeric(1))
507 quantity <- sum(vapply(pieces, function(x) prod(x), numeric(1)))
508 bars <- sum(uses)
509 emit(paste0("CUTSTOCK L=", length_bar, " pieces=", length(pieces),
510 " patterns=", length(patterns)))
511 for (i in seq_along(patterns)) {
512 pattern <- patterns[[i]]
513 breakdown <- character()
514 for (j in seq_along(pieces)) {
515 count <- pattern$counts[j]
516 if (count != 0)
517 breakdown <- c(breakdown, paste0(count, "x", pieces[[j]][1]))
518 }
519 emit(paste0("PATTERN ", i, " ", paste(breakdown, collapse = "+"),
520 " rest=", pattern$trim))
521 }
522 for (i in seq_along(uses)) {
523 if (uses[i]) emit(paste0("USE ", i, " bars=", uses[i]))
524 }
525 for (i in seq_along(pieces)) {
526 emit(paste0("PIECE ", pieces[[i]][1], " demand=", pieces[[i]][2],
527 " cut=", cut[i]))
528 }
529 trim <- sum(vapply(seq_along(patterns), function(i)
530 uses[i] * patterns[[i]]$trim, numeric(1)))
531 surplus <- sum(cut) - sum(vapply(pieces, function(x) x[2], numeric(1)))
532 emit(paste0("TOTAL bars=", bars, " trim=", trim, " surplus=", surplus,
533 " waste=", bars * length_bar - quantity))
534 all_sizes <- unlist(lapply(pieces, function(p) rep(p[1], p[2])))
535 comparison <- pack(lapply(seq_along(all_sizes), function(i)
536 c(i, all_sizes[i])), length_bar, "FIRSTFIT")
537 n <- length(comparison$bins)
538 emit(paste0("FFD bars=", n, " waste=", n * length_bar - quantity))
539 emit(paste0("LOWER bars=", ceiling(quantity, length_bar)))
540}

557shelf_method <- function(method, width, rects, area) {
558 shelves <- list()
559 placements <- list()
560 for (rectangle in rects) {
561 number <- rectangle[1]
562 w <- rectangle[2]
563 h <- rectangle[3]
564 chosen <- NA_integer_
565 if (method == "NFDH") {
566 last <- length(shelves)
567 if (last && shelves[[last]]$used + w <= width) chosen <- last
568 } else {
569 fitting <- which(vapply(shelves, function(shelf)
570 shelf$used + w <= width, logical(1)))
571 if (length(fitting)) {
572 chosen <- if (method == "FFDH") fitting[1] else
573 fitting[which.max(vapply(shelves[fitting], function(s) s$used, numeric(1)))]
574 }
575 }
576 if (is.na(chosen)) {
577 last <- length(shelves)
578 bottom <- if (last) shelves[[last]]$bottom + shelves[[last]]$height else 0
579 chosen <- last + 1
580 shelves[[chosen]] <- new_shelf(bottom, h)
581 }
582 shelf <- shelves[[chosen]]
583 shelf$items[[length(shelf$items) + 1]] <- c(number, shelf$used)
584 placements[[length(placements) + 1]] <-
585 c(number, shelf$used, shelf$bottom, w, h, 0)
586 shelf$used <- shelf$used + w
587 shelves[[chosen]] <- shelf
588 }
589 last <- shelves[[length(shelves)]]
590 height <- last$bottom + last$height
591 emit(paste0(method, " height=", height, " shelves=", length(shelves),
592 " util=", floor(area * 100 / (width * height))))
593 for (i in seq_along(shelves)) {
594 shelf <- shelves[[i]]
595 words <- vapply(shelf$items, function(item)
596 paste0(rectangle_text(item[1]), "@", item[2]), character(1))
597 emit(paste0("LEVEL ", i, " y=", shelf$bottom, " h=", shelf$height,
598 " used=", shelf$used, " items=", paste(words, collapse = " ")))
599 }
600 drawing(width, height, placements)
601}
602cmd_shelf <- function(arguments) {
603 width <- as.numeric(arguments[1])
604 if (width == 0) fail("BAD CAPACITY")
605 rects <- rectangles(arguments[-1])
606 for (rect in rects) {
607 if (rect[2] == 0 || rect[3] == 0 || rect[2] > width)
608 fail(paste("BAD RECT", rect[1]))
609 }
610 rects <- rects[order(
611 -vapply(rects, function(r) r[3], numeric(1)),
612 -vapply(rects, function(r) r[2], numeric(1)),
613 vapply(rects, function(r) r[1], numeric(1))
614 )]
615 area <- sum(vapply(rects, function(r) r[2] * r[3], numeric(1)))
616 emit(paste0("SHELF W=", width, " rects=", length(rects), " area=", area))
617 for (method in c("NFDH", "FFDH", "BFDH"))
618 shelf_method(method, width, rects, area)
619 heights <- numeric(width)
620 placements <- list()
621 for (rect in rects) {
622 number <- rect[1]; w <- rect[2]; h <- rect[3]
623 x <- 0
624 best <- Inf
625 for (column in 0:(width - w)) {
626 y <- max(heights[seq.int(column + 1, column + w)])
627 if (y < best) {
628 best <- y
629 x <- column
630 }
631 }
632 heights[seq.int(x + 1, x + w)] <- best + h
633 placements[[length(placements) + 1]] <- c(number, x, best, w, h, 0)
634 }
635 height <- max(heights)
636 emit(paste0("SKYLINE height=", height,
637 " util=", floor(area * 100 / (width * height))))
638 for (p in placements) {
639 emit(paste0("PLACE ", rectangle_text(p[1]),
640 " x=", p[2], " y=", p[3], " w=", p[4], " h=", p[5]))
641 }
642 drawing(width, height, placements)
643}
644free_rectangle <- function(sheets, width, height, rotate, rule) {
645 chosen <- NULL
646 best <- NULL
647 for (sheet_number in seq_along(sheets)) {
648 sheet <- sheets[[sheet_number]]
649 for (i in seq_along(sheet$free)) {
650 rect <- sheet$free[[i]]
651 fits <- width <= rect[3] && height <= rect[4]
652 fits_rotated <- rotate && height <= rect[3] && width <= rect[4]
653 if (!(fits || fits_rotated)) next
654 if (rule == "FIRST") return(c(sheet_number, i))
655 key <- c(rect[3] * rect[4], sheet_number, i)
656 if (is.null(best) ||
657 any(key != best) && key[which(key != best)[1]] < best[which(key != best)[1]]) {
658 best <- key
659 chosen <- c(sheet_number, i)
660 }
661 }
662 }
663 chosen
664}
665cmd_guillotine <- function(arguments) {
666 width <- as.numeric(arguments[1])
667 height <- as.numeric(arguments[2])
668 mode <- arguments[3]
669 rule <- arguments[4]
670 if (width == 0 || height == 0) fail("BAD CAPACITY")
671 rects <- rectangles(arguments[-(1:4)])
672 rotate <- mode == "ROTATE"
673 for (rect in rects) {
674 w <- rect[2]; h <- rect[3]
675 fits <- w <= width && h <= height
676 fits_rotated <- rotate && h <= width && w <= height
677 if (w == 0 || h == 0 || !(fits || fits_rotated))
678 fail(paste("BAD RECT", rect[1]))
679 }
680 rects <- rects[order(
681 -vapply(rects, function(r) r[2] * r[3], numeric(1)),
682 vapply(rects, function(r) r[1], numeric(1))
683 )]
684 sheets <- list()
685 rotated_count <- 0
686 for (rect in rects) {
687 number <- rect[1]; w <- rect[2]; h <- rect[3]
688 chosen <- free_rectangle(sheets, w, h, rotate, rule)
689 if (is.null(chosen)) {
690 sheets[[length(sheets) + 1]] <- new_sheet(width, height)
691 chosen <- c(length(sheets), 1)
692 }
693 sheet_number <- chosen[1]
694 i <- chosen[2]
695 sheet <- sheets[[sheet_number]]
696 region <- sheet$free[[i]]
697 sheet$free <- sheet$free[-i]
698 x <- region[1]; y <- region[2]
699 free_width <- region[3]; free_height <- region[4]
700 rotated <- !(w <= free_width && h <= free_height)
701 pw <- if (rotated) h else w
702 ph <- if (rotated) w else h
703 rotated_count <- rotated_count + as.integer(rotated)
704 sheet$placements[[length(sheet$placements) + 1]] <-
705 c(number, x, y, pw, ph, as.integer(rotated))
706 sheet$area <- sheet$area + w * h
707 right_width <- free_width - pw
708 top_height <- free_height - ph
709 if (right_width <= top_height) {
710 children <- list(c(x, y + ph, free_width, top_height),
711 c(x + pw, y, right_width, ph))
712 } else {
713 children <- list(c(x, y + ph, pw, top_height),
714 c(x + pw, y, right_width, free_height))
715 }
716 for (child in children) {
717 if (child[3] > 0 && child[4] > 0)
718 sheet$free[[length(sheet$free) + 1]] <- child
719 }
720 sheets[[sheet_number]] <- sheet
721 }
722 area <- sum(vapply(sheets, function(s) s$area, numeric(1)))
723 emit(paste0("GUILLOTINE W=", width, " H=", height, " mode=", mode,
724 " rule=", rule, " rects=", length(rects),
725 " sheets=", length(sheets), " rotated=", rotated_count,
726 " util=", floor(area * 100 / (length(sheets) * width * height))))
727 for (i in seq_along(sheets)) {
728 sheet <- sheets[[i]]
729 emit(paste0("SHEET ", i, " placed=", length(sheet$placements),
730 " area=", sheet$area))
731 for (p in sheet$placements) {
732 text <- paste0("PLACE ", rectangle_text(p[1]),
733 " x=", p[2], " y=", p[3], " w=", p[4], " h=", p[5])
734 if (p[6]) text <- paste0(text, " rotated")
735 emit(text)
736 }
737 free <- if (length(sheet$free)) {
738 paste(vapply(sheet$free, function(r)
739 paste0(r[1], ",", r[2], ",", r[3], "x", r[4]), character(1)), collapse = " ")
740 } else "none"
741 emit(paste("FREE", free))
742 drawing(width, height, sheet$placements)
743 }
744}

751 is.null(lab$items)) fail("NO ITEMS")
752 if (command %in% methods) {
753 run_method(command, arguments)
754 } else if (command == "ITEMS") {
755 cmd_items(arguments)
756 } else if (command == "RANDOM") {
