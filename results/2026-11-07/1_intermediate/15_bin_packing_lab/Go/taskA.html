33func descending(items []item) []item {
34 result := append([]item(nil), items...)
35 sort.Slice(result, func(i, j int) bool {
36 if result[i].size != result[j].size {
37 return result[i].size > result[j].size
38 }
39 return result[i].id < result[j].id
40 })
41 return result
42}

93func pack(items []item, capacity int, method string, parameter int) *packing {
94 result := &packing{method: method, parameter: parameter}
95 ordered := items
96 if method == "FFD" || method == "BFD" || method == "WFD" {
97 ordered = descending(items)
98 }
99 active := []int{}
100 current := make(map[int]int)
101 for _, it := range ordered {
102 chosen, closed, class := -1, -1, 0
103 before := result.inspections
104 switch method {
105 case "NEXTFIT":
106 if len(result.bins) > 0 {
107 result.inspections++
108 last := len(result.bins)-1
109 if result.bins[last].load+it.size <= capacity {
110 chosen = last
111 }
112 }
113 case "FIRSTFIT", "FFD", "KBOUNDED":
114 candidates := active
115 if method != "KBOUNDED" {
116 candidates = make([]int, len(result.bins))
117 for i := range candidates { candidates[i] = i }
118 }
119 for _, i := range candidates {
120 result.inspections++
121 if result.bins[i].load+it.size <= capacity {
122 chosen = i
123 break
124 }
125 }
126 case "HARMONIC":
127 class = capacity / it.size
128 if class > parameter { class = parameter }
129 if index, ok := current[class]; ok {
130 result.inspections++
131 if result.bins[index].load+it.size <= capacity {
132 chosen = index
133 }
134 }
135 default:
136 result.inspections += len(result.bins)
137 fitting := []int{}
138 for i, b := range result.bins {
139 if b.load+it.size <= capacity {
140 fitting = append(fitting, i)
141 }
142 }
143 if len(fitting) > 0 {
144 if method == "BESTFIT" || method == "BFD" {
145 chosen = fitting[0]
146 for _, i := range fitting {
147 if result.bins[i].load > result.bins[chosen].load {
148 chosen = i
149 }
150 }
151 } else {
152 sort.Slice(fitting, func(i, j int) bool {
153 a, b := fitting[i], fitting[j]
154 if result.bins[a].load != result.bins[b].load {
155 return result.bins[a].load < result.bins[b].load
156 }
157 return a < b
158 })
159 chosen = fitting[0]
160 if method == "ALMOSTWORST" && len(fitting) > 1 {
161 chosen = fitting[1]
162 }
163 }
164 }
165 }
166 opened := chosen < 0
167 if opened {
168 if method == "KBOUNDED" && len(active) == parameter {
169 closed = active[0]
170 for _, i := range active {
171 if result.bins[i].load > result.bins[closed].load {
172 closed = i
173 }
174 }
175 for j, i := range active {
176 if i == closed {
177 active = append(active[:j], active[j+1:]...)
178 break
179 }
180 }
181 result.closes++
182 }
183 chosen = len(result.bins)
184 result.bins = append(result.bins, &bin{category: class})
185 if method == "KBOUNDED" { active = append(active, chosen) }
186 if method == "HARMONIC" { current[class] = chosen }
187 }
188 result.bins[chosen].put(it)
189 result.steps = append(result.steps, step{item: it, bin: chosen,
190 inspected: result.inspections-before, opened: opened, closed: closed, category: class})
191 }
192 return result
193}

195type boundRow struct { alpha, j1, j2, j3, bound int }
196type boundsResult struct { l1, l2 int; rows []boundRow }
197
198func bounds(items []item, capacity int) boundsResult {
199 total := 0
200 thresholds := map[int]bool{0:true}
201 for _, it := range items {
202 total += it.size
203 if 2*it.size <= capacity { thresholds[it.size] = true }
204 }
205 values := make([]int, 0, len(thresholds))
206 for alpha := range thresholds { values = append(values, alpha) }
207 sort.Ints(values)
208 answer := boundsResult{l1: ceiling(total, capacity)}
209 for _, alpha := range values {
210 j1, j2, j3, sum2, sum3 := 0, 0, 0, 0, 0
211 for _, it := range items {
212 s := it.size
213 switch {
214 case s > capacity-alpha:
215 j1++
216 case 2*s > capacity:
217 j2++
218 sum2 += s
219 case s >= alpha:
220 j3++
221 sum3 += s
222 }
223 }
224 excess := sum3 - (j2*capacity-sum2)
225 bound := j1+j2
226 if excess > 0 { bound += ceiling(excess, capacity) }
227 answer.rows = append(answer.rows, boundRow{alpha,j1,j2,j3,bound})
228 if bound > answer.l2 { answer.l2 = bound }
229 }
230 return answer
231}

243func newExact(items []item, capacity, limit, lower int, incumbent []*bin, lab *lab) *exactSearch {
244 order := descending(items)
245 remaining := make([]int, len(order)+1)
246 for i := len(order)-1; i >= 0; i-- {
247 remaining[i] = remaining[i+1]+order[i].size
248 }
249 return &exactSearch{order:order, capacity:capacity, limit:limit,
250 lower:lower, remaining:remaining, best:copyBins(incumbent), lab:lab}
251}

253func (e *exactSearch) visit(index int) {
254 if e.nodes == e.limit {
255 e.status = "limit"
256 return
257 }
258 e.nodes++
259 count := len(e.live)
260 used := e.remaining[0]-e.remaining[index]
261 free := count*e.capacity-used
262 excess := e.remaining[index]-free
263 minimum := count
264 if excess > 0 { minimum += ceiling(excess,e.capacity) }
265 if minimum >= len(e.best) { return }
266 if index == len(e.order) {
267 e.best = copyBins(e.live)
268 e.lab.emit(fmt.Sprintf("IMPROVE bins=%d nodes=%d", count, e.nodes))
269 if count == e.lower { e.status = "bound" }
270 return
271 }
272 it := e.order[index]
273 seen := map[int]bool{}
274 for _, b := range e.live {
275 old := b.load
276 if seen[old] || old+it.size > e.capacity { continue }
277 seen[old] = true
278 b.put(it)
279 e.visit(index+1)
280 b.items = b.items[:len(b.items)-1]
281 b.load = old
282 if e.status != "" { return }
283 }
284 if count+1 < len(e.best) {
285 b := &bin{}
286 b.put(it)
287 e.live = append(e.live,b)
288 e.visit(index+1)
289 e.live = e.live[:len(e.live)-1]
290 }
291}

404func (l *lab) reduce(method string,k int) {
405 p := pack(l.items,l.capacity,method,k)
406 bins := p.bins
407 before := len(bins)
408 events := []string{}
409 for {
410 sources := make([]int,len(bins))
411 for i := range sources { sources[i] = i }
412 sort.Slice(sources,func(i,j int)bool{
413 a,b := sources[i],sources[j]
414 if bins[a].load != bins[b].load { return bins[a].load < bins[b].load }
415 return a < b
416 })
417 eliminated := false
418 for _,source := range sources {
419 loads := make([]int,len(bins))
420 for i,b := range bins { loads[i] = b.load }
421 moved, destinations := []item{},[]int{}
422 possible := true
423 for _,it := range descending(bins[source].items) {
424 target := -1
425 for i := range bins {
426 if i==source || loads[i]+it.size>l.capacity { continue }
427 if target<0 || loads[i]>loads[target] { target=i }
428 }
429 if target<0 { possible=false; break }
430 loads[target] += it.size
431 moved=append(moved,it)
432 destinations=append(destinations,target)
433 }
434 if !possible { continue }
435 moves := make([]string,len(moved))
436 for i,it := range moved {
437 target:=destinations[i]
438 moves[i]=fmt.Sprintf("%s->%d",it.text(),target+1)
439 bins[target].put(it)
440 }
441 events=append(events,fmt.Sprintf("ELIMINATE bin=%d load=%d moves=%s",
442 source+1,bins[source].load,strings.Join(moves," ")))
443 bins=append(bins[:source],bins[source+1:]...)
444 eliminated=true
445 break
446 }
447 if !eliminated { break }
448 }
449 l.emit(fmt.Sprintf("REDUCE %s before=%d after=%d eliminated=%d",
450 p.name(),before,len(bins),before-len(bins)))
451 for _,event := range events { l.emit(event) }
452 l.binLines(bins,method=="HARMONIC")
453}

464func (l *lab) compare(k int) {
465 if k==0 { fail("BAD PARAMETER") }
466 b:=bounds(l.items,l.capacity)
467 results:=make([]*packing,0,len(methods))
468 for _,method:=range methods { results=append(results,pack(l.items,l.capacity,method,k)) }
469 sort.Slice(results,func(i,j int)bool{
470 if len(results[i].bins)!=len(results[j].bins) { return len(results[i].bins)<len(results[j].bins) }
471 return results[i].method<results[j].method
472 })
473 l.emit(fmt.Sprintf("COMPARE k=%d L1=%d L2=%d",k,b.l1,b.l2))
474 for _,p:=range results {
475 count:=len(p.bins)
476 l.emit(fmt.Sprintf("%s bins=%d gap=%d waste=%d inspections=%d",
477 p.method,count,count-b.l2,count*l.capacity-l.total(),p.inspections))
478 }

481func (l *lab) exact(limit int) {
482 if limit==0 { fail("BAD PARAMETER") }
483 if len(l.items)>=21 { fail("TOO MANY ITEMS") }
484 incumbent:=pack(l.items,l.capacity,"FFD",0).bins
485 lower:=bounds(l.items,l.capacity).l2
486 l.emit(fmt.Sprintf("EXACT ffd=%d L2=%d",len(incumbent),lower))
487 search:=newExact(l.items,l.capacity,limit,lower,incumbent,l)
488 if len(incumbent)==lower { search.status="bound" } else { search.visit(0) }
489 if search.status=="" { search.status="complete" }
490 l.emit(fmt.Sprintf("RESULT bins=%d nodes=%d status=%s",len(search.best),search.nodes,search.status))
491 l.binLines(search.best,false)
492}

495 if len(words)-1!=len(l.items) { fail("BAD ASSIGNMENT") }
496 bins:=map[int]*bin{}
497 for i,it:=range l.items{
498 label:=number(words[i+1])

500 if bins[label]==nil { bins[label]=&bin{} }
501 bins[label].put(it)

505 for label,b:=range bins {
506 labels=append(labels,label)
507 if b.load>l.capacity { over++ }
508 }
509 sort.Ints(labels)
510 status:="valid"
511 if over>0 { status="invalid" }
512 l.emit(fmt.Sprintf("CHECK bins=%d over=%d L2=%d status=%s",
513 len(bins),over,bounds(l.items,l.capacity).l2,status))
514 for _,label:=range labels {
515 l.binLine(label,bins[label],l.capacity,false,bins[label].load>l.capacity)

536func (l *lab) cover() {
537 l.emit(fmt.Sprintf("COVER total=%d upper=%d",l.total(),l.total()/l.capacity))
538 sequential:=[]*bin{}
539 for _,it:=range l.items {
540 if len(sequential)==0 || sequential[len(sequential)-1].load>=l.capacity {
541 sequential=append(sequential,&bin{})
542 }
543 sequential[len(sequential)-1].put(it)
544 }
545 l.coverLines("DNF",sequential)
546 order:=descending(l.items)
547 paired:=[]*bin{}
548 for first,last:=0,len(order)-1; first<=last; {
549 b:=&bin{}
550 b.put(order[first]);first++
551 for b.load<l.capacity && first<=last {
552 b.put(order[last]);last--
553 }
554 paired=append(paired,b)
555 }
556 l.coverLines("BIGSMALL",paired)
557}

569func (l *lab) multifit(m int) {
570 if m==0 { fail("BAD PARAMETER") }
571 largest:=0
572 for _,it:=range l.items { if it.size>largest {largest=it.size} }
573 lo:=ceiling(l.total(),m);hi:=ceiling(2*l.total(),m)
574 if largest>lo {lo=largest}
575 if largest>hi {hi=largest}
576 l.emit(fmt.Sprintf("MULTIFIT m=%d lower=%d upper=%d",m,lo,hi))
577 for lo<hi {
578 mid:=lo+(hi-lo)/2
579 count:=len(pack(l.items,mid,"FFD",0).bins)
580 ok:=count<=m
581 text:="fail"
582 if ok { text="ok"; hi=mid } else {lo=mid+1}
583 l.emit(fmt.Sprintf("TRY cap=%d bins=%d %s",mid,count,text))
584 }
585 bins:=pack(l.items,hi,"FFD",0).bins
586 for len(bins)<m { bins=append(bins,&bin{}) }
587 makespan:=0
588 for _,b:=range bins { if b.load>makespan {makespan=b.load} }
589 l.emit(fmt.Sprintf("RESULT cap=%d makespan=%d",hi,makespan))
590 l.machines(bins)
591 lpt:=make([]*bin,m)
592 for i:=range lpt {lpt[i]=&bin{}}
593 for _,it:=range descending(l.items) {
594 best:=0
595 for i:=1;i<m;i++ {if lpt[i].load<lpt[best].load {best=i}}
596 lpt[best].put(it)
597 }
598 makespan=0
599 for _,b:=range lpt {if b.load>makespan {makespan=b.load}}
600 l.emit(fmt.Sprintf("LPT makespan=%d",makespan))
601 l.machines(lpt)
602}

604func (l *lab) cutstock(words []string) {
605 length:=number(words[1])
606 if length==0 {fail("BAD CAPACITY")}
607 pieces:=make([]piece,0,len(words)-2)
608 seen:=map[int]bool{}
609 for i:=2;i<len(words);i++{
610 parts:=strings.Split(words[i],":")
611 size,demand:=number(parts[0]),number(parts[1])
612 if size==0 || size>length || demand==0 {fail(fmt.Sprintf("BAD PIECE %d",i-1))}
613 if seen[size] {fail(fmt.Sprintf("DUPLICATE PIECE %d",i-1))}
614 seen[size]=true
615 pieces=append(pieces,piece{size,demand})
616 }
617 sort.Slice(pieces,func(i,j int)bool{return pieces[i].length>pieces[j].length})
618 patterns:=[]pattern{}
619 counts:=make([]int,len(pieces))
620 var visit func(int,int)
621 visit=func(index,remaining int){
622 max:=remaining/pieces[index].length
623 if index==len(pieces)-1 {
624 counts[index]=max
625 patterns=append(patterns,pattern{
626 counts:append([]int(nil),counts...),
627 trim:remaining-max*pieces[index].length,
628 })
629 return
630 }
631 for c:=max;c>=0;c-- {
632 counts[index]=c
633 visit(index+1,remaining-c*pieces[index].length)
634 }
635 }
636 visit(0,length)
637 l.emit(fmt.Sprintf("CUTSTOCK L=%d pieces=%d patterns=%d",length,len(pieces),len(patterns)))
638 for i,p:=range patterns {
639 breakdown:=[]string{}
640 for j,c:=range p.counts {
641 if c>0 {breakdown=append(breakdown,fmt.Sprintf("%dx%d",c,pieces[j].length))}
642 }
643 l.emit(fmt.Sprintf("PATTERN %d %s rest=%d",i+1,strings.Join(breakdown,"+"),p.trim))
644 }
645 remaining:=make([]int,len(pieces))
646 cut:=make([]int,len(pieces))
647 used:=make([]int,len(patterns))
648 for i,p:=range pieces {remaining[i]=p.demand}
649 for {
650 needed:=0
651 for _,value:=range remaining { needed+=value }
652 if needed==0 {break}
653 chosen,highest:=-1,-1
654 for j,p:=range patterns {
655 useful:=0
656 for i,c:=range p.counts {
657 if c<remaining[i] {useful+=c} else {useful+=remaining[i]}
658 }
659 if useful>highest {chosen,highest=j,useful}
660 }
661 used[chosen]++
662 for i,c:=range patterns[chosen].counts {
663 cut[i]+=c
664 if c>remaining[i] {remaining[i]=0} else {remaining[i]-=c}
665 }
666 }
667 bars,trim,total,demandTotal,cutTotal:=0,0,0,0,0
668 stock:=[]item{}
669 for i,p:=range patterns {
670 if used[i]>0 {
671 l.emit(fmt.Sprintf("USE %d bars=%d",i+1,used[i]))
672 bars+=used[i];trim+=p.trim*used[i]
673 }
674 }
675 for _,p:=range pieces {
676 total+=p.length*p.demand
677 demandTotal+=p.demand
678 for j:=0;j<p.demand;j++ {stock=append(stock,item{len(stock)+1,p.length})}
679 }
680 for i,p:=range pieces {
681 l.emit(fmt.Sprintf("PIECE %d demand=%d cut=%d",p.length,p.demand,cut[i]))
682 cutTotal+=cut[i]
683 }
684 l.emit(fmt.Sprintf("TOTAL bars=%d trim=%d surplus=%d waste=%d",
685 bars,trim,cutTotal-demandTotal,bars*length-total))
686 ffd:=len(pack(stock,length,"FIRSTFIT",0).bins)
687 l.emit(fmt.Sprintf("FFD bars=%d waste=%d",ffd,ffd*length-total))
688 l.emit(fmt.Sprintf("LOWER bars=%d",ceiling(total,length)))
689}

691func rectangles(words []string,start,width,height int,rotate bool) []rect {
692 result:=make([]rect,0,len(words)-start)
693 for i:=start;i<len(words);i++ {
694 parts:=strings.Split(words[i],"x")
695 w,h:=number(parts[0]),number(parts[1])
696 fits:=w<=width
697 if height>0 {fits=(w<=width && h<=height) || (rotate && h<=width && w<=height)}
698 if w==0 || h==0 || !fits {fail(fmt.Sprintf("BAD RECT %d",i-start+1))}
699 result=append(result,rect{id:i-start+1,w:w,h:h})
700 }
701 return result
702}

720func (l *lab) shelfCommand(words []string) {
721 width:=number(words[1])
722 if width==0 {fail("BAD CAPACITY")}
723 source:=rectangles(words,2,width,0,false)
724 area:=0
725 for _,r:=range source {area+=r.area()}
726 l.emit(fmt.Sprintf("SHELF W=%d rects=%d area=%d",width,len(source),area))
727 order:=append([]rect(nil),source...)
728 sort.Slice(order,func(i,j int)bool {
729 a,b:=order[i],order[j]
730 if a.h!=b.h {return a.h>b.h}
731 if a.w!=b.w {return a.w>b.w}
732 return a.id<b.id
733 })
734 for _,method:=range []string{"NFDH","FFDH","BFDH"} {
735 shelves:=[]*shelf{}
736 placed:=[]placement{}
737 for _,r:=range order {
738 var selected *shelf
739 if method=="NFDH" {
740 if len(shelves)>0 && shelves[len(shelves)-1].used+r.w<=width {
741 selected=shelves[len(shelves)-1]
742 }
743 } else {
744 for _,candidate:=range shelves {
745 if candidate.used+r.w>width {continue}
746 if selected==nil || (method=="BFDH" && candidate.used>selected.used) {
747 selected=candidate
748 }
749 if method=="FFDH" {break}
750 }
751 }
752 if selected==nil {
753 y:=0
754 if len(shelves)>0 {
755 last:=shelves[len(shelves)-1]
756 y=last.y+last.h
757 }
758 selected=&shelf{y:y,h:r.h}
759 shelves=append(shelves,selected)
760 }
761 p:=placement{rect:r,x:selected.used,y:selected.y,w:r.w,h:r.h}
762 selected.placed=append(selected.placed,p)
763 placed=append(placed,p)
764 selected.used+=r.w
765 }
766 last:=shelves[len(shelves)-1]
767 height:=last.y+last.h
768 l.emit(fmt.Sprintf("%s height=%d shelves=%d util=%d",method,height,len(shelves),
769 area*100/(width*height)))
770 for i,s:=range shelves {
771 names:=make([]string,len(s.placed))
772 for j,p:=range s.placed {names[j]=fmt.Sprintf("%c@%d",p.rect.letter(),p.x)}
773 l.emit(fmt.Sprintf("LEVEL %d y=%d h=%d used=%d items=%s",
774 i+1,s.y,s.h,s.used,strings.Join(names," ")))
775 }
776 l.draw(width,height,placed)
777 }
778 columns:=make([]int,width)
779 placed:=[]placement{}
780 for _,r:=range order {
781 bestX,bestY:=0,int(^uint(0)>>1)
782 for x:=0;x<=width-r.w;x++{
783 y:=0
784 for j:=x;j<x+r.w;j++ {if columns[j]>y {y=columns[j]}}
785 if y<bestY {bestX,bestY=x,y}
786 }
787 p:=placement{rect:r,x:bestX,y:bestY,w:r.w,h:r.h}
788 placed=append(placed,p)
789 for j:=bestX;j<bestX+r.w;j++ {columns[j]=bestY+r.h}
790 }
791 height:=0
792 for _,h:=range columns {if h>height {height=h}}
793 l.emit(fmt.Sprintf("SKYLINE height=%d util=%d",height,area*100/(width*height)))
794 for _,p:=range placed {
795 l.emit(fmt.Sprintf("PLACE %c x=%d y=%d w=%d h=%d",p.rect.letter(),p.x,p.y,p.w,p.h))
796 }
797 l.draw(width,height,placed)
798}

800func fits(r rect,f freeRect,rotate bool) bool {
801 return (r.w<=f.w && r.h<=f.h) || (rotate && r.h<=f.w && r.w<=f.h)
802}

804func (l *lab) guillotine(words []string) {
805 width,height:=number(words[1]),number(words[2])
806 if width==0 || height==0 {fail("BAD CAPACITY")}
807 rotate:=words[3]=="ROTATE"
808 best:=words[4]=="BEST"
809 order:=rectangles(words,5,width,height,rotate)
810 sort.Slice(order,func(i,j int)bool{
811 if order[i].area()!=order[j].area() {return order[i].area()>order[j].area()}
812 return order[i].id<order[j].id
813 })
814 area:=0
815 for _,r:=range order {area+=r.area()}
816 sheets:=[]*sheet{}
817 rotated:=0
818 for _,r:=range order {
819 var selected *sheet
820 selectedIndex,selectedArea:=-1,int(^uint(0)>>1)
821 for _,s:=range sheets {
822 for i,f:=range s.free {
823 if !fits(r,f,rotate) {continue}
824 if selected==nil || (best && f.area()<selectedArea) {
825 selected=s
826 selectedIndex=i
827 selectedArea=f.area()
828 }
829 if !best {break}
830 }
831 if selected!=nil && !best {break}
832 }
833 if selected==nil {
834 selected=&sheet{free:[]freeRect{{0,0,width,height}}}
835 sheets=append(sheets,selected)
836 selectedIndex=0
837 }
838 f:=selected.free[selectedIndex]
839 selected.free=append(selected.free[:selectedIndex],selected.free[selectedIndex+1:]...)
840 turned:=r.w>f.w || r.h>f.h
841 w,h:=r.w,r.h
842 if turned {w,h=h,w;rotated++}
843 p:=placement{rect:r,x:f.x,y:f.y,w:w,h:h,rotated:turned}
844 selected.placed=append(selected.placed,p)
845 selected.area+=r.area()
846 rw,rh:=f.w-w,f.h-h
847 var top,right freeRect
848 if rw<=rh {
849 top=freeRect{f.x,f.y+h,f.w,rh}
850 right=freeRect{f.x+w,f.y,rw,h}
851 } else {
852 top=freeRect{f.x,f.y+h,w,rh}
853 right=freeRect{f.x+w,f.y,rw,f.h}
854 }
855 if top.w>0 && top.h>0 {selected.free=append(selected.free,top)}
856 if right.w>0 && right.h>0 {selected.free=append(selected.free,right)}
857 }
858 l.emit(fmt.Sprintf("GUILLOTINE W=%d H=%d mode=%s rule=%s rects=%d sheets=%d rotated=%d util=%d",
859 width,height,words[3],words[4],len(order),len(sheets),rotated,area*100/(len(sheets)*width*height)))
860 for i,s:=range sheets {
861 l.emit(fmt.Sprintf("SHEET %d placed=%d area=%d",i+1,len(s.placed),s.area))
862 for _,p:=range s.placed {
863 line:=fmt.Sprintf("PLACE %c x=%d y=%d w=%d h=%d",p.rect.letter(),p.x,p.y,p.w,p.h)
864 if p.rotated {line+=" rotated"}
865 l.emit(line)
866 }
867 if len(s.free)==0 {l.emit("FREE none")} else {
868 parts:=make([]string,len(s.free))
869 for j,f:=range s.free {parts[j]=f.text()}
870 l.emit("FREE "+strings.Join(parts," "))
871 }
872 l.draw(width,height,s.placed)
873 }
874}

876func (l *lab) process(words []string) {
877 defer func(){
878 if x:=recover();x!=nil {
879 if err,ok:=x.(commandError);ok {l.emit("ERROR "+string(err));return}
880 panic(x)
881 }
882 }()
883 cmd:=words[0]
884 switch cmd {
885 case "ITEMS": l.itemsCommand(words)
886 case "RANDOM": l.randomCommand(words)
887 case "CUTSTOCK": l.cutstock(words)
888 case "SHELF": l.shelfCommand(words)
889 case "GUILLOTINE": l.guillotine(words)
890 default:
891 if l.items==nil {fail("NO ITEMS")}
892 switch cmd {
893 case "NEXTFIT","FIRSTFIT","BESTFIT","WORSTFIT","ALMOSTWORST",
894 "HARMONIC","KBOUNDED","FFD","BFD","WFD":
895 l.methodCommand(cmd,parameter(cmd,words,1))
896 case "TRACE": l.trace(words[1],parameter(words[1],words,2))
897 case "REDUCE": l.reduce(words[1],parameter(words[1],words,2))
898 case "BOUNDS": l.boundsCommand()
899 case "COMPARE": l.compare(number(words[1]))
900 case "EXACT": l.exact(number(words[1]))
901 case "CHECK": l.check(words)
902 case "COVER": l.cover()
903 case "MULTIFIT": l.multifit(number(words[1]))
904 }
905 }
906}

913 for i:=0;i<n && scanner.Scan();i++ {

915 }
