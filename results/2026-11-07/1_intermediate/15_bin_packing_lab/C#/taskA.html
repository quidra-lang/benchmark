66static class PackingAlgorithms
67{
68 public static long Ceil(long a, long b) => (a + b - 1) / b;
69
70 public static List<Item> Descending(IEnumerable<Item> items) =>
71 items.OrderByDescending(item => item.Size).ThenBy(item => item.Number).ToList();
72
73 public static Packing Pack(List<Item> items, int capacity, string method, int parameter = 0)
74 {
75 var result = new Packing(method, parameter);
76 var order = method is "FFD" or "BFD" or "WFD" ? Descending(items) : items;
77 var active = new List<int>();
78 var current = new Dictionary<int, int>();
79 foreach (Item item in order)
80 {
81 int before = result.Inspections;
82 int chosen = -1;
83 int closed = -1;
84 int category = 0;
85 if (method == "NEXTFIT")
86 {
87 if (result.Bins.Count > 0)
88 {
89 result.Inspections++;
90 int last = result.Bins.Count - 1;
91 if (result.Bins[last].Load + item.Size <= capacity) chosen = last;
92 }
93 }
94 else if (method is "FIRSTFIT" or "FFD" or "KBOUNDED")
95 {
96 IEnumerable<int> candidates = method == "KBOUNDED"
97 ? active : Enumerable.Range(0, result.Bins.Count);
98 foreach (int index in candidates)
99 {
100 result.Inspections++;
101 if (result.Bins[index].Load + item.Size <= capacity)
102 {
103 chosen = index;
104 break;
105 }
106 }
107 }
108 else if (method == "HARMONIC")
109 {
110 category = Math.Min(capacity / item.Size, parameter);
111 if (current.TryGetValue(category, out int index))
112 {
113 result.Inspections++;
114 if (result.Bins[index].Load + item.Size <= capacity) chosen = index;
115 }
116 }
117 else
118 {
119 result.Inspections += result.Bins.Count;
120 var fitting = Enumerable.Range(0, result.Bins.Count)
121 .Where(index => result.Bins[index].Load + item.Size <= capacity).ToList();
122 if (fitting.Count > 0)
123 {
124 if (method is "BESTFIT" or "BFD")
125 chosen = fitting.OrderByDescending(index => result.Bins[index].Load)
126 .ThenBy(index => index).First();
127 else
128 {
129 fitting = fitting.OrderBy(index => result.Bins[index].Load)
130 .ThenBy(index => index).ToList();
131 chosen = method == "ALMOSTWORST" && fitting.Count >= 2
132 ? fitting[1] : fitting[0];
133 }
134 }
135 }
136
137 bool opened = chosen < 0;
138 if (opened)
139 {
140 if (method == "KBOUNDED" && active.Count == parameter)
141 {
142 closed = active.OrderByDescending(index => result.Bins[index].Load)
143 .ThenBy(index => index).First();
144 active.Remove(closed);
145 result.Closes++;
146 }
147 chosen = result.Bins.Count;
148 result.Bins.Add(new Bin(category));
149 if (method == "KBOUNDED") active.Add(chosen);
150 if (method == "HARMONIC") current[category] = chosen;
151 }
152 result.Bins[chosen].Put(item);
153 result.Steps.Add(new Step(item, chosen, result.Inspections - before,
154 opened, closed, category));
155 }
156 return result;
157 }
158
159 public static BoundResult Bounds(List<Item> items, int capacity)
160 {
161 long total = items.Sum(item => (long) item.Size);
162 var thresholds = new SortedSet<int> { 0 };
163 foreach (Item item in items)
164 if (2 * item.Size <= capacity) thresholds.Add(item.Size);
165 int maximum = 0;
166 var rows = new List<BoundRow>();
167 foreach (int alpha in thresholds)
168 {
169 int j1 = 0, j2 = 0, j3 = 0;
170 long sum2 = 0, sum3 = 0;
171 foreach (Item item in items)
172 {
173 int size = item.Size;
174 if (size > capacity - alpha) j1++;
175 else if (2 * size > capacity) { j2++; sum2 += size; }
176 else if (size >= alpha) { j3++; sum3 += size; }
177 }
178 long excess = sum3 - (j2 * (long) capacity - sum2);
179 int bound = j1 + j2 + (excess > 0 ? (int) Ceil(excess, capacity) : 0);
180 maximum = Math.Max(maximum, bound);
181 rows.Add(new BoundRow(alpha, j1, j2, j3, bound));
182 }
183 return new BoundResult(Ceil(total, capacity), maximum, rows);
184 }
185}

187sealed class ExactSearch
188{
189 private readonly List<Item> order;
190 private readonly int capacity, limit, lower;
191 private readonly long[] remaining;
192 private readonly List<Bin> live = new();
193 private readonly Action<string> emit;
194 public List<Bin> Best { get; private set; }
195 public int Nodes { get; private set; }
196 public string? Status { get; set; }
197
198 public ExactSearch(List<Item> items, int capacity, int limit, int lower,
199 List<Bin> incumbent, Action<string> emit)
200 {
201 order = PackingAlgorithms.Descending(items);
202 this.capacity = capacity;
203 this.limit = limit;
204 this.lower = lower;
205 this.emit = emit;
206 Best = incumbent.Select(bin => bin.Copy()).ToList();
207 remaining = new long[order.Count + 1];
208 for (int i = order.Count - 1; i >= 0; i--)
209 remaining[i] = remaining[i + 1] + order[i].Size;
210 }
211
212 public void Visit(int index)
213 {
214 if (Nodes == limit) { Status = "limit"; return; }
215 Nodes++;
216 int count = live.Count;
217 long used = remaining[0] - remaining[index];
218 long free = count * (long) capacity - used;
219 long excess = remaining[index] - free;
220 long minimum = count + (excess > 0 ? PackingAlgorithms.Ceil(excess, capacity) : 0);
221 if (minimum >= Best.Count) return;
222 if (index == order.Count)
223 {
224 Best = live.Select(bin => bin.Copy()).ToList();
225 emit($"IMPROVE bins={count} nodes={Nodes}");
226 if (count == lower) Status = "bound";
227 return;
228 }
229
230 Item item = order[index];
231 var seenLoads = new HashSet<int>();
232 foreach (Bin bin in live)
233 {
234 int oldLoad = bin.Load;
235 if (!seenLoads.Add(oldLoad) || oldLoad + item.Size > capacity) continue;
236 bin.Put(item);
237 Visit(index + 1);
238 bin.Items.RemoveAt(bin.Items.Count - 1);
239 bin.Load = oldLoad;
240 if (Status != null) return;
241 }
242 if (count + 1 < Best.Count)
243 {
244 var bin = new Bin();
245 bin.Put(item);
246 live.Add(bin);
247 Visit(index + 1);
248 live.RemoveAt(live.Count - 1);
249 }
250 }
251}

393 private void Reduce(string method, int parameter)
394 {
395 Packing result = PackingAlgorithms.Pack(items!, capacity, method, parameter);
396 List<Bin> bins = result.Bins;
397 int before = bins.Count;
398 var events = new List<string>();
399 while (true)
400 {
401 bool eliminated = false;
402 var sources = Enumerable.Range(0, bins.Count)
403 .OrderBy(index => bins[index].Load).ThenBy(index => index).ToList();
404 foreach (int source in sources)
405 {
406 int[] loads = bins.Select(bin => bin.Load).ToArray();
407 var moves = new List<(Item Item, int Destination)>();
408 bool possible = true;
409 foreach (Item item in PackingAlgorithms.Descending(bins[source].Items))
410 {
411 int destination = Enumerable.Range(0, bins.Count)
412 .Where(index => index != source && loads[index] + item.Size <= capacity)
413 .OrderByDescending(index => loads[index]).ThenBy(index => index)
414 .DefaultIfEmpty(-1).First();
415 if (destination < 0) { possible = false; break; }
416 loads[destination] += item.Size;
417 moves.Add((item, destination));
418 }
419 if (!possible) continue;
420 string descriptions = string.Join(" ", moves.Select(move =>
421 $"{move.Item}->{move.Destination + 1}"));
422 events.Add($"ELIMINATE bin={source + 1} load={bins[source].Load} moves={descriptions}");
423 foreach (var move in moves) bins[move.Destination].Put(move.Item);
424 bins.RemoveAt(source);
425 eliminated = true;
426 break;
427 }
428 if (!eliminated) break;
429 }
430 Emit($"REDUCE {result.Name} before={before} after={bins.Count} eliminated={before - bins.Count}");
431 foreach (string line in events) Emit(line);
432 BinLines(bins, method == "HARMONIC");
433 }

542 {
543 if (count == 0) Fail("BAD PARAMETER");
544 int largest = items!.Max(item => item.Size);
545 int low = (int) Math.Max(PackingAlgorithms.Ceil(Total, count), largest);
546 int high = (int) Math.Max(PackingAlgorithms.Ceil(2 * Total, count), largest);
547 Emit($"MULTIFIT m={count} lower={low} upper={high}");
548 while (low < high)
549 {
550 int middle = low + (high - low) / 2;
551 int bins = PackingAlgorithms.Pack(items, middle, "FFD").Bins.Count;
552 bool fits = bins <= count;
553 Emit($"TRY cap={middle} bins={bins} {(fits ? "ok" : "fail")}");
554 if (fits) high = middle;
555 else low = middle + 1;
556 }
557 var machines = PackingAlgorithms.Pack(items, high, "FFD").Bins;
558 while (machines.Count < count) machines.Add(new Bin());
559 Emit($"RESULT cap={high} makespan={machines.Max(bin => bin.Load)}");
560 MachineLines(machines);
561
562 var lpt = Enumerable.Range(0, count).Select(_ => new Bin()).ToList();
563 foreach (Item item in PackingAlgorithms.Descending(items))
564 {
565 Bin chosen = lpt.OrderBy(bin => bin.Load).First();
566 chosen.Put(item);
567 }
568 Emit($"LPT makespan={lpt.Max(bin => bin.Load)}");
569 MachineLines(lpt);

572 private void Cutstock(string[] words)
573 {
574 int length = Num(words[1]);
575 if (length == 0) Fail("BAD CAPACITY");
576 var pieces = new List<Piece>();
577 var seen = new HashSet<int>();
578 for (int i = 2; i < words.Length; i++)
579 {
580 string[] parts = words[i].Split(':');
581 int size = Num(parts[0]), demand = Num(parts[1]);
582 if (size == 0 || size > length || demand == 0) Fail($"BAD PIECE {i - 1}");
583 if (!seen.Add(size)) Fail($"DUPLICATE PIECE {i - 1}");
584 pieces.Add(new Piece(size, demand));
585 }
586 pieces = pieces.OrderByDescending(piece => piece.Length).ToList();
587 var patterns = new List<Pattern>();
588 var counts = new int[pieces.Count];
589
590 void Enumerate(int index, int remaining)
591 {
592 int maximum = remaining / pieces[index].Length;
593 if (index == pieces.Count - 1)
594 {
595 counts[index] = maximum;
596 patterns.Add(new Pattern((int[]) counts.Clone(),
597 remaining - maximum * pieces[index].Length));
598 return;
599 }
600 for (int i = maximum; i >= 0; i--)
601 {
602 counts[index] = i;
603 Enumerate(index + 1, remaining - i * pieces[index].Length);
604 }
605 }
606
607 Enumerate(0, length);
608 Emit($"CUTSTOCK L={length} pieces={pieces.Count} patterns={patterns.Count}");
609 for (int i = 0; i < patterns.Count; i++)
610 {
611 Pattern pattern = patterns[i];
612 var breakdown = Enumerable.Range(0, pieces.Count)
613 .Where(index => pattern.Counts[index] > 0)
614 .Select(index => $"{pattern.Counts[index]}x{pieces[index].Length}");
615 Emit($"PATTERN {i + 1} {string.Join("+", breakdown)} rest={pattern.Trim}");
616 }
617
618 int[] remainingDemand = pieces.Select(piece => piece.Demand).ToArray();
619 int[] cut = new int[pieces.Count];
620 int[] used = new int[patterns.Count];
621 while (remainingDemand.Any(value => value > 0))
622 {
623 int chosen = -1, maximum = -1;
624 for (int index = 0; index < patterns.Count; index++)
625 {
626 int useful = Enumerable.Range(0, pieces.Count)
627 .Sum(i => Math.Min(patterns[index].Counts[i], remainingDemand[i]));
628 if (useful > maximum) { maximum = useful; chosen = index; }
629 }
630 used[chosen]++;
631 for (int i = 0; i < pieces.Count; i++)
632 {
633 cut[i] += patterns[chosen].Counts[i];
634 remainingDemand[i] = Math.Max(0, remainingDemand[i] - patterns[chosen].Counts[i]);
635 }
636 }
637 int bars = 0, trim = 0;
638 for (int i = 0; i < used.Length; i++)
639 {
640 if (used[i] == 0) continue;
641 Emit($"USE {i + 1} bars={used[i]}");
642 bars += used[i];
643 trim += used[i] * patterns[i].Trim;
644 }
645 long total = pieces.Sum(piece => (long) piece.Length * piece.Demand);
646 int demandTotal = pieces.Sum(piece => piece.Demand);
647 int cutTotal = 0;
648 var stock = new List<Item>();
649 for (int i = 0; i < pieces.Count; i++)
650 {
651 Piece piece = pieces[i];
652 Emit($"PIECE {piece.Length} demand={piece.Demand} cut={cut[i]}");
653 cutTotal += cut[i];
654 for (int j = 0; j < piece.Demand; j++)
655 stock.Add(new Item(stock.Count + 1, piece.Length));
656 }
657 Emit($"TOTAL bars={bars} trim={trim} surplus={cutTotal - demandTotal} "
658 + $"waste={bars * (long) length - total}");
659 int ffd = PackingAlgorithms.Pack(stock, length, "FIRSTFIT").Bins.Count;
660 Emit($"FFD bars={ffd} waste={ffd * (long) length - total}");
661 Emit($"LOWER bars={PackingAlgorithms.Ceil(total, length)}");
662 }

664 private static List<Rect> Rectangles(string[] words, int start,
665 int width, int height, bool rotate)
666 {
667 var rectangles = new List<Rect>();
668 for (int i = start; i < words.Length; i++)
669 {
670 string[] sizes = words[i].Split('x');
671 int w = Num(sizes[0]), h = Num(sizes[1]);
672 bool fits = height == 0 ? w <= width
673 : (w <= width && h <= height) || (rotate && h <= width && w <= height);
674 if (w == 0 || h == 0 || !fits) Fail($"BAD RECT {i - start + 1}");
675 rectangles.Add(new Rect(i - start + 1, w, h));
676 }
677 return rectangles;
678 }

680 private void Draw(int width, int height, IEnumerable<Placement> placements)
681 {
682 char[][] grid = Enumerable.Range(0, height)
683 .Select(_ => Enumerable.Repeat('.', width).ToArray()).ToArray();
684 foreach (Placement placement in placements)
685 {
686 for (int y = placement.Y; y < placement.Y + placement.Height; y++)
687 for (int x = placement.X; x < placement.X + placement.Width; x++)
688 grid[y][x] = placement.Rect.Letter;
689 }
690 for (int y = height - 1; y >= 0; y--) Emit(new string(grid[y]));
691 }

693 private void ShelfCommand(string[] words)
694 {
695 int width = Num(words[1]);
696 if (width == 0) Fail("BAD CAPACITY");
697 var rectangles = Rectangles(words, 2, width, 0, false);
698 int area = rectangles.Sum(rect => rect.Area);
699 Emit($"SHELF W={width} rects={rectangles.Count} area={area}");
700 var ordered = rectangles.OrderByDescending(rect => rect.Height)
701 .ThenByDescending(rect => rect.Width).ThenBy(rect => rect.Number).ToList();
702 foreach (string method in new[] { "NFDH", "FFDH", "BFDH" })
703 {
704 var shelves = new List<Shelf>();
705 var placements = new List<Placement>();
706 foreach (Rect rect in ordered)
707 {
708 Shelf? chosen = null;
709 if (method == "NFDH")
710 {
711 if (shelves.Count > 0 && shelves[^1].Used + rect.Width <= width)
712 chosen = shelves[^1];
713 }
714 else
715 {
716 var fitting = shelves.Where(shelf => shelf.Used + rect.Width <= width);
717 chosen = method == "FFDH" ? fitting.FirstOrDefault()
718 : fitting.OrderByDescending(shelf => shelf.Used).FirstOrDefault();
719 }
720 if (chosen == null)
721 {
722 int bottom = shelves.Count == 0 ? 0 : shelves[^1].Bottom + shelves[^1].Height;
723 chosen = new Shelf(bottom, rect.Height);
724 shelves.Add(chosen);
725 }
726 var placement = new Placement(rect, chosen.Used, chosen.Bottom, rect.Width, rect.Height);
727 chosen.Placements.Add(placement);
728 placements.Add(placement);
729 chosen.Used += rect.Width;
730 }
731 int height = shelves[^1].Bottom + shelves[^1].Height;
732 Emit($"{method} height={height} shelves={shelves.Count} util={area * 100 / (width * height)}");
733 for (int i = 0; i < shelves.Count; i++)
734 {
735 Shelf shelf = shelves[i];
736 string contents = string.Join(" ", shelf.Placements.Select(p => $"{p.Rect.Letter}@{p.X}"));
737 Emit($"LEVEL {i + 1} y={shelf.Bottom} h={shelf.Height} used={shelf.Used} items={contents}");
738 }
739 Draw(width, height, placements);
740 }
741 int[] columns = new int[width];
742 var skyline = new List<Placement>();
743 foreach (Rect rect in ordered)
744 {
745 int chosenX = 0, chosenY = int.MaxValue;
746 for (int x = 0; x <= width - rect.Width; x++)
747 {
748 int top = columns.Skip(x).Take(rect.Width).Max();
749 if (top < chosenY) { chosenX = x; chosenY = top; }
750 }
751 skyline.Add(new Placement(rect, chosenX, chosenY, rect.Width, rect.Height));
752 for (int x = chosenX; x < chosenX + rect.Width; x++)
753 columns[x] = chosenY + rect.Height;
754 }
755 int skylineHeight = columns.Max();
756 Emit($"SKYLINE height={skylineHeight} util={area * 100 / (width * skylineHeight)}");
757 foreach (Placement p in skyline)
758 Emit($"PLACE {p.Rect.Letter} x={p.X} y={p.Y} w={p.Width} h={p.Height}");
759 Draw(width, skylineHeight, skyline);
760 }

762 private static bool Fits(Rect rect, FreeRect free, bool rotate) =>
763 (rect.Width <= free.Width && rect.Height <= free.Height)
764 || (rotate && rect.Height <= free.Width && rect.Width <= free.Height);

766 private void Guillotine(string[] words)
767 {
768 int width = Num(words[1]), height = Num(words[2]);
769 if (width == 0 || height == 0) Fail("BAD CAPACITY");
770 bool rotate = words[3] == "ROTATE";
771 bool best = words[4] == "BEST";
772 var ordered = Rectangles(words, 5, width, height, rotate)
773 .OrderByDescending(rect => rect.Area).ThenBy(rect => rect.Number).ToList();
774 var sheets = new List<Sheet>();
775 int rotated = 0, totalArea = ordered.Sum(rect => rect.Area);
776 foreach (Rect rect in ordered)
777 {
778 Sheet? selected = null;
779 int chosenIndex = -1, chosenArea = int.MaxValue;
780 foreach (Sheet sheet in sheets)
781 {
782 for (int i = 0; i < sheet.Free.Count; i++)
783 {
784 FreeRect free = sheet.Free[i];
785 if (!Fits(rect, free, rotate)) continue;
786 if (selected == null || (best && free.Area < chosenArea))
787 {
788 selected = sheet;
789 chosenIndex = i;
790 chosenArea = free.Area;
791 }
792 if (!best) break;
793 }
794 if (selected != null && !best) break;
795 }
796 if (selected == null)
797 {
798 selected = new Sheet(width, height);
799 sheets.Add(selected);
800 chosenIndex = 0;
801 }
802 FreeRect target = selected.Free[chosenIndex];
803 selected.Free.RemoveAt(chosenIndex);
804 bool turned = rect.Width > target.Width || rect.Height > target.Height;
805 int w = turned ? rect.Height : rect.Width;
806 int h = turned ? rect.Width : rect.Height;
807 if (turned) rotated++;
808 selected.Placements.Add(new Placement(rect, target.X, target.Y, w, h, turned));
809 selected.Area += rect.Area;
810 int rw = target.Width - w, rh = target.Height - h;
811 FreeRect top = rw <= rh
812 ? new FreeRect(target.X, target.Y + h, target.Width, rh)
813 : new FreeRect(target.X, target.Y + h, w, rh);
814 FreeRect right = rw <= rh
815 ? new FreeRect(target.X + w, target.Y, rw, h)
816 : new FreeRect(target.X + w, target.Y, rw, target.Height);
817 if (top.Width > 0 && top.Height > 0) selected.Free.Add(top);
818 if (right.Width > 0 && right.Height > 0) selected.Free.Add(right);
819 }
820 int count = sheets.Count;
821 Emit($"GUILLOTINE W={width} H={height} mode={words[3]} rule={words[4]} "
822 + $"rects={ordered.Count} sheets={count} rotated={rotated} "
823 + $"util={totalArea * 100 / (count * width * height)}");
824 for (int i = 0; i < sheets.Count; i++)
825 {
826 Sheet sheet = sheets[i];
827 Emit($"SHEET {i + 1} placed={sheet.Placements.Count} area={sheet.Area}");
828 foreach (Placement p in sheet.Placements)
829 Emit($"PLACE {p.Rect.Letter} x={p.X} y={p.Y} w={p.Width} h={p.Height}"
830 + (p.Rotated ? " rotated" : ""));
831 Emit(sheet.Free.Count == 0 ? "FREE none" : "FREE " + string.Join(" ", sheet.Free));
832 Draw(width, height, sheet.Placements);
833 }
834 }

836 public void Process(string[] words)
837 {
838 try
839 {
840 string command = words[0];
841 switch (command)
842 {
843 case "ITEMS": ItemsCommand(words); return;
844 case "RANDOM": RandomCommand(words); return;
845 case "CUTSTOCK": Cutstock(words); return;
846 case "SHELF": ShelfCommand(words); return;
847 case "GUILLOTINE": Guillotine(words); return;
848 }
849 if (items == null) Fail("NO ITEMS");
850 switch (command)
851 {
852 case "NEXTFIT":
853 case "FIRSTFIT":
854 case "BESTFIT":
855 case "WORSTFIT":
856 case "ALMOSTWORST":
857 case "HARMONIC":
858 case "KBOUNDED":
859 case "FFD":
860 case "BFD":
861 case "WFD":
862 MethodCommand(command, Parameter(command, words, 1));
863 break;
864 case "TRACE":
865 Trace(words[1], Parameter(words[1], words, 2));
866 break;
867 case "REDUCE":
868 Reduce(words[1], Parameter(words[1], words, 2));
869 break;
870 case "BOUNDS": BoundsCommand(); break;
871 case "COMPARE": Compare(Num(words[1])); break;
872 case "EXACT": Exact(Num(words[1])); break;
873 case "CHECK": Check(words); break;
874 case "COVER": Cover(); break;
875 case "MULTIFIT": Multifit(Num(words[1])); break;
876 }
877 }
878 catch (CommandError error)
879 {
880 Emit("ERROR " + error.Message);
881 }
882 }
883}
