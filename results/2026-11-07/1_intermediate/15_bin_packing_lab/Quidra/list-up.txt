class Item(int number, int size)
class Bin(Item[] items, int load, int category)
class Step(Item item, int container, int inspected, bool opened, int closed, int category)
class Packing(string method, int parameter, Bin[] bins, Step[] steps, int inspections, int closes)
class Lab()
    Item[] items := []
    capacity := 0
    ready := false

void emit(string text)
    println(text)
int integer(string text)
    return int(text)
string text(Item item)
    return string(item.number) + ":" + string(item.size)

string method_name(string name, int k)
    return name + (if name == "HARMONIC" or name == "KBOUNDED" then " k=" + string(k) else "")

bool parameterized(string name)
    return name == "HARMONIC" or name == "KBOUNDED"

Bin new_bin(category := 0)
    return Bin([], 0, category)
void put(Bin &container, Item item)
    container.items = container.items.append(item)
    container.load += item.size

Packing pack(Item[] items, int capacity, string method, parameter := 0)
    result := Packing(method, parameter, [], [], 0, 0)
    int[] order := []
    if method == "FFD" or method == "BFD" or method == "WFD"
    else
        for: 0 <= i < int(#items)
            order = order.append(i)
    int[] active := []
    for index := order[]
        item := items[index]
        chosen := -1
        closed := -1
        category := 0
        elif method == "FIRSTFIT" or method == "FFD" or method == "KBOUNDED"
            int[] candidates := []
            if method == "KBOUNDED"
                candidates = active
            else
            for i := candidates[]
                result.inspections += 1
                if result.bins[i].load + item.size <= capacity
                    chosen = i
                    break
        opened := chosen < 0
        if opened
            if method == "KBOUNDED" and #active == parameter
                closed = active[0]
                for i := active[]
                    if result.bins[i].load > result.bins[closed].load
                        closed = i
                int[] next_active := []
                for i := active[]
                    if i != closed
                        next_active = next_active.append(i)
                active = next_active
                result.closes += 1
            chosen = int(#result.bins)
            result.bins = result.bins.append(new_bin(category))
            if method == "KBOUNDED"
                active = active.append(chosen)
        put(&result.bins[chosen], item)
    return result
string header(Packing result, int capacity, int total)
    line := method_name(result.method, result.parameter) + " bins=" + string(#result.bins) + " inspections=" + string(result.inspections) + " waste=" + string(int(#result.bins) * capacity - total)
    if result.method == "KBOUNDED"
        line += " closes=" + string(result.closes)
    return line

void bin_line(int number, Bin container, int capacity, classes := false, over := false)
    line := "BIN " + string(number) + " load=" + string(container.load) + "/" + string(capacity) + " items=" + [for item := container.items[] then text(item)].join(" ")
    emit(line)
void bin_lines(Bin[] bins, int capacity, classes := false)
    for: 0 <= i < int(#bins)
        bin_line(i + 1, bins[i], capacity, classes)
void command_items(Lab &lab, string[] words)
    capacity := integer(words[1])
    if capacity == 0
        fail("BAD CAPACITY")
        return
    Item[] items := []
    for: 2 <= i < int(#words)
        size := integer(words[i])
        if size == 0 or size > capacity
            fail("BAD ITEM " + string(i - 1))
            return
        items = items.append(Item(i - 1, size))
    lab.items = items
    lab.capacity = capacity
    lab.ready = true
    emit("ITEMS count=" + string(#items) + " capacity=" + string(capacity) + " total=" + string([for item := items[] then item.size].sum()))
void command_method(Lab lab, string method, int k)
    result := pack(lab.items, lab.capacity, method, k)
    emit(header(result, lab.capacity, [for item := lab.items[] then item.size].sum()))
    bin_lines(result.bins, lab.capacity, method == "HARMONIC")
void command(Lab &lab, string line)
    words := line.split(" ")
    action := words[0]
    if action == "ITEMS"
        command_items(&lab, words)
    else
        if not lab.ready
            fail("NO ITEMS")
            return
        else
            k := if parameterized(action) then integer(words[1]) else 0
            if parameterized(action) and k == 0
                fail("BAD PARAMETER")
            else
                command_method(lab, action, k)
