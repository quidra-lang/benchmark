51private fun pack(items: List<Item>, capacity: Int, method: String, parameter: Int = 0): Packing {
52 val result = Packing(method, parameter)
53 val order = if (method in listOf("FFD", "BFD", "WFD")) descending(items) else items
54 val active = mutableListOf<Int>()
55 val current = mutableMapOf<Int, Int>()
56 for (item in order) {
57 val before = result.inspections
58 var chosen = -1
59 var closed = -1
60 var category = 0
61 when (method) {
62 "NEXTFIT" -> {
63 if (result.bins.isNotEmpty()) {
64 result.inspections++
65 val last = result.bins.lastIndex
66 if (result.bins[last].load + item.size <= capacity) chosen = last
67 }
68 }
69 "FIRSTFIT", "FFD", "KBOUNDED" -> {
70 val candidates = if (method == "KBOUNDED") active else result.bins.indices.toList()
71 for (index in candidates) {
72 result.inspections++
73 if (result.bins[index].load + item.size <= capacity) {
74 chosen = index
75 break
76 }
77 }
78 }
79 "HARMONIC" -> {
80 category = minOf(capacity / item.size, parameter)
81 val index = current[category]
82 if (index != null) {
83 result.inspections++
84 if (result.bins[index].load + item.size <= capacity) chosen = index
85 }
86 }
87 else -> {
88 result.inspections += result.bins.size
89 val fitting = result.bins.indices
90 .filter { result.bins[it].load + item.size <= capacity }
91 if (fitting.isNotEmpty()) {
92 chosen = if (method == "BESTFIT" || method == "BFD")
93 fitting.sortedWith(compareByDescending<Int> { result.bins[it].load }
94 .thenBy { it }).first()
95 else {
96 val ordered = fitting.sortedWith(compareBy<Int> { result.bins[it].load }
97 .thenBy { it })
98 if (method == "ALMOSTWORST" && ordered.size >= 2) ordered[1] else ordered[0]
99 }
100 }
101 }
102 }
103 val opened = chosen < 0
104 if (opened) {
105 if (method == "KBOUNDED" && active.size == parameter) {
106 closed = active.sortedWith(compareByDescending<Int> { result.bins[it].load }
107 .thenBy { it }).first()
108 active.remove(closed)
109 result.closes++
110 }
111 chosen = result.bins.size
112 result.bins.add(Bin(category))
113 if (method == "KBOUNDED") active.add(chosen)
114 if (method == "HARMONIC") current[category] = chosen
115 }
116 result.bins[chosen].put(item)
117 result.steps.add(Step(item, chosen, result.inspections - before,
118 opened, closed, category))
119 }
120 return result
121}

123private data class BoundRow(val alpha: Int, val j1: Int, val j2: Int, val j3: Int, val bound: Int)
124private data class Bounds(val l1: Long, val l2: Int, val rows: List<BoundRow>)

126private fun bounds(items: List<Item>, capacity: Int): Bounds {
127 val total = items.sumOf { it.size.toLong() }
128 val thresholds = sortedSetOf(0)
129 for (item in items) if (2 * item.size <= capacity) thresholds.add(item.size)
130 val rows = mutableListOf<BoundRow>()
131 var lower = 0
132 for (alpha in thresholds) {
133 var j1 = 0
134 var j2 = 0
135 var j3 = 0
136 var sum2 = 0L
137 var sum3 = 0L
138 for (item in items) {
139 when {
140 item.size > capacity - alpha -> j1++
141 2 * item.size > capacity -> { j2++; sum2 += item.size }
142 item.size >= alpha -> { j3++; sum3 += item.size }
143 }
144 }
145 val excess = sum3 - (j2.toLong() * capacity - sum2)
146 val bound = j1 + j2 + if (excess > 0) ceil(excess, capacity.toLong()).toInt() else 0
147 lower = maxOf(lower, bound)
148 rows.add(BoundRow(alpha, j1, j2, j3, bound))
149 }
150 return Bounds(ceil(total, capacity.toLong()), lower, rows)
151}

153private class ExactSearch(
154 items: List<Item>, val capacity: Int, val limit: Int, val lower: Int,
155 incumbent: List<Bin>, val emit: (String) -> Unit
156) {
157 private val order = descending(items)
158 private val live = mutableListOf<Bin>()
159 private val remaining = LongArray(order.size + 1)
160 var best = incumbent.map { it.copy() }
161 var nodes = 0
162 var status: String? = null
163
164 init {
165 for (i in order.lastIndex downTo 0)
166 remaining[i] = remaining[i + 1] + order[i].size
167 }
168
169 fun visit(index: Int) {
170 if (nodes == limit) { status = "limit"; return }
171 nodes++
172 val count = live.size
173 val used = remaining[0] - remaining[index]
174 val free = count.toLong() * capacity - used
175 val excess = remaining[index] - free
176 val minimum = count.toLong() +
177 (if (excess > 0) ceil(excess, capacity.toLong()) else 0L)
178 if (minimum >= best.size) return
179 if (index == order.size) {
180 best = live.map { it.copy() }
181 emit("IMPROVE bins=$count nodes=$nodes")
182 if (count == lower) status = "bound"
183 return
184 }
185 val item = order[index]
186 val seen = mutableSetOf<Int>()
187 for (bin in live) {
188 val oldLoad = bin.load
189 if (!seen.add(oldLoad) || oldLoad + item.size > capacity) continue
190 bin.put(item)
191 visit(index + 1)
192 bin.items.removeAt(bin.items.lastIndex)
193 bin.load = oldLoad
194 if (status != null) return
195 }
196 if (count + 1 < best.size) {
197 val bin = Bin()
198 bin.put(item)
199 live.add(bin)
200 visit(index + 1)
201 live.removeAt(live.lastIndex)
202 }
203 }
204}

308 private fun reduce(method: String, k: Int) {
309 val result = pack(items!!, capacity, method, k)
310 val bins = result.bins
311 val before = bins.size
312 val events = mutableListOf<String>()
313 while (true) {
314 var eliminated = false
315 val sources = bins.indices.sortedWith(compareBy<Int> { bins[it].load }.thenBy { it })
316 for (source in sources) {
317 val loads = bins.map { it.load }.toMutableList()
318 val moves = mutableListOf<Pair<Item, Int>>()
319 var possible = true
320 for (item in descending(bins[source].items)) {
321 val target = bins.indices.filter {
322 it != source && loads[it] + item.size <= capacity
323 }.sortedWith(compareByDescending<Int> { loads[it] }.thenBy { it }).firstOrNull()
324 if (target == null) { possible = false; break }
325 loads[target] += item.size
326 moves.add(item to target)
327 }
328 if (!possible) continue
329 val descriptions = moves.joinToString(" ") { "${it.first}->${it.second + 1}" }
330 events.add("ELIMINATE bin=${source + 1} load=${bins[source].load} moves=$descriptions")
331 for ((item, target) in moves) bins[target].put(item)
332 bins.removeAt(source)
333 eliminated = true
334 break
335 }
336 if (!eliminated) break
337 }
338 emit("REDUCE ${result.name()} before=$before after=${bins.size} eliminated=${before - bins.size}")
339 for (event in events) emit(event)
340 binLines(bins, method == "HARMONIC")
341 }

351 if (k == 0) fail("BAD PARAMETER")
352 val result = bounds(items!!, capacity)
353 val methods = listOf("NEXTFIT", "FIRSTFIT", "BESTFIT", "WORSTFIT", "ALMOSTWORST",
354 "HARMONIC", "KBOUNDED", "FFD", "BFD", "WFD")
355 val packed = methods.map { pack(items!!, capacity, it, k) }
356 .sortedWith(compareBy<Packing> { it.bins.size }.thenBy { it.method })
357 emit("COMPARE k=$k L1=${result.l1} L2=${result.l2}")
358 for (p in packed) {
359 val count = p.bins.size
360 emit("${p.method} bins=$count gap=${count - result.l2} " +
361 "waste=${count.toLong() * capacity - total()} inspections=${p.inspections}")
362 }

401 emit("COVER total=${total()} upper=${total() / capacity}")
402 val sequential = mutableListOf<Bin>()
403 for (item in items!!) {
404 if (sequential.isEmpty() || sequential.last().load >= capacity)
405 sequential.add(Bin())
406 sequential.last().put(item)
407 }
408 coverLines("DNF", sequential)

410 val paired = mutableListOf<Bin>()
411 var first = 0
412 var last = ordered.lastIndex
413 while (first <= last) {
414 val bin = Bin()
415 bin.put(ordered[first++])
416 while (bin.load < capacity && first <= last) bin.put(ordered[last--])
417 paired.add(bin)
418 }
419 coverLines("BIGSMALL", paired)
420 }

429 private fun multifit(m: Int) {
430 if (m == 0) fail("BAD PARAMETER")
431 val largest = items!!.maxOf { it.size }
432 var low = maxOf(ceil(total(), m.toLong()).toInt(), largest)
433 var high = maxOf(ceil(2 * total(), m.toLong()).toInt(), largest)
434 emit("MULTIFIT m=$m lower=$low upper=$high")
435 while (low < high) {
436 val mid = low + (high - low) / 2
437 val count = pack(items!!, mid, "FFD").bins.size
438 val ok = count <= m
439 emit("TRY cap=$mid bins=$count ${if (ok) "ok" else "fail"}")
440 if (ok) high = mid else low = mid + 1
441 }
442 val machines = pack(items!!, high, "FFD").bins
443 while (machines.size < m) machines.add(Bin())
444 emit("RESULT cap=$high makespan=${machines.maxOf { it.load }}")
445 machineLines(machines)
446 val lpt = MutableList(m) { Bin() }
447 for (item in descending(items!!))
448 lpt.minBy { it.load }.put(item)
449 emit("LPT makespan=${lpt.maxOf { it.load }}")
450 machineLines(lpt)
451 }

453 private fun cutstock(words: List<String>) {
454 val length = words[1].toInt()
455 if (length == 0) fail("BAD CAPACITY")
456 val pieces = mutableListOf<Piece>()
457 val seen = mutableSetOf<Int>()
458 for ((index, word) in words.drop(2).withIndex()) {
459 val (sizeText, demandText) = word.split(":")
460 val size = sizeText.toInt()
461 val demand = demandText.toInt()
462 if (size == 0 || size > length || demand == 0)
463 fail("BAD PIECE ${index + 1}")
464 if (!seen.add(size)) fail("DUPLICATE PIECE ${index + 1}")
465 pieces.add(Piece(size, demand))
466 }
467 pieces.sortByDescending { it.length }
468 val patterns = mutableListOf<Pattern>()
469 val counts = IntArray(pieces.size)
470 fun enumerate(index: Int, remaining: Int) {
471 val maximum = remaining / pieces[index].length
472 if (index == pieces.lastIndex) {
473 counts[index] = maximum
474 patterns.add(Pattern(counts.toList(), remaining - maximum * pieces[index].length))
475 return
476 }
477 for (c in maximum downTo 0) {
478 counts[index] = c
479 enumerate(index + 1, remaining - c * pieces[index].length)
480 }
481 }
482 enumerate(0, length)
483 emit("CUTSTOCK L=$length pieces=${pieces.size} patterns=${patterns.size}")
484 for ((index, pattern) in patterns.withIndex()) {
485 val breakdown = pattern.counts.mapIndexedNotNull { i, c ->
486 if (c > 0) "${c}x${pieces[i].length}" else null
487 }.joinToString("+")
488 emit("PATTERN ${index + 1} $breakdown rest=${pattern.trim}")
489 }
490 val remaining = pieces.map { it.demand }.toMutableList()
491 val cut = IntArray(pieces.size)
492 val used = IntArray(patterns.size)
493 while (remaining.any { it > 0 }) {
494 val chosen = patterns.indices.maxBy { index ->
495 patterns[index].counts.indices.sumOf { i ->
496 minOf(patterns[index].counts[i], remaining[i])
497 }
498 }
499 used[chosen]++
500 for (i in pieces.indices) {
501 val count = patterns[chosen].counts[i]
502 cut[i] += count
503 remaining[i] = maxOf(0, remaining[i] - count)
504 }
505 }
506 var bars = 0
507 var trim = 0
508 for (i in patterns.indices) {
509 if (used[i] == 0) continue
510 emit("USE ${i + 1} bars=${used[i]}")
511 bars += used[i]
512 trim += used[i] * patterns[i].trim
513 }
514 val quantity = pieces.sumOf { it.length.toLong() * it.demand }
515 val stock = mutableListOf<Item>()
516 for ((index, piece) in pieces.withIndex()) {
517 emit("PIECE ${piece.length} demand=${piece.demand} cut=${cut[index]}")
518 repeat(piece.demand) { stock.add(Item(stock.size + 1, piece.length)) }
519 }
520 emit("TOTAL bars=$bars trim=$trim surplus=${cut.sum() - pieces.sumOf { it.demand }} " +
521 "waste=${bars.toLong() * length - quantity}")
522 val ffd = pack(stock, length, "FIRSTFIT").bins.size
523 emit("FFD bars=$ffd waste=${ffd.toLong() * length - quantity}")
524 emit("LOWER bars=${ceil(quantity, length.toLong())}")
525 }

549 private fun shelf(words: List<String>) {
550 val width = words[1].toInt()
551 if (width == 0) fail("BAD CAPACITY")
552 val source = rectangles(words, 2, width, 0, false)
553 val area = source.sumOf { it.area }
554 emit("SHELF W=$width rects=${source.size} area=$area")
555 val order = source.sortedWith(compareByDescending<Rect> { it.height }
556 .thenByDescending { it.width }.thenBy { it.number })
557 for (method in listOf("NFDH", "FFDH", "BFDH")) {
558 val shelves = mutableListOf<Shelf>()
559 val placements = mutableListOf<Placement>()
560 for (rect in order) {
561 var chosen: Shelf? = null
562 if (method == "NFDH") {
563 if (shelves.isNotEmpty() && shelves.last().used + rect.width <= width)
564 chosen = shelves.last()
565 } else {
566 val fitting = shelves.filter { it.used + rect.width <= width }
567 chosen = if (method == "FFDH") fitting.firstOrNull()
568 else fitting.maxByOrNull { it.used }
569 }
570 if (chosen == null) {
571 val bottom = if (shelves.isEmpty()) 0 else shelves.last().bottom + shelves.last().height
572 chosen = Shelf(bottom, rect.height)
573 shelves.add(chosen)
574 }
575 val p = Placement(rect, chosen.used, chosen.bottom, rect.width, rect.height)
576 chosen.placements.add(p)
577 placements.add(p)
578 chosen.used += rect.width
579 }
580 val height = shelves.last().bottom + shelves.last().height
581 emit("$method height=$height shelves=${shelves.size} util=${area.toLong() * 100 / (width * height)}")
582 for ((i, shelf) in shelves.withIndex()) {
583 val content = shelf.placements.joinToString(" ") { "${it.rect.letter}@${it.x}" }
584 emit("LEVEL ${i + 1} y=${shelf.bottom} h=${shelf.height} used=${shelf.used} items=$content")
585 }
586 draw(width, height, placements)
587 }
588 val columns = IntArray(width)
589 val skyline = mutableListOf<Placement>()
590 for (rect in order) {
591 var bestX = 0
592 var bestY = Int.MAX_VALUE
593 for (x in 0..width - rect.width) {
594 val top = (x until x + rect.width).maxOf { columns[it] }
595 if (top < bestY) { bestX = x; bestY = top }
596 }
597 skyline.add(Placement(rect, bestX, bestY, rect.width, rect.height))
598 for (x in bestX until bestX + rect.width) columns[x] = bestY + rect.height
599 }
600 val height = columns.max()
601 emit("SKYLINE height=$height util=${area.toLong() * 100 / (width * height)}")
602 for (p in skyline)
603 emit("PLACE ${p.rect.letter} x=${p.x} y=${p.y} w=${p.width} h=${p.height}")
604 draw(width, height, skyline)
605 }

611 private fun guillotine(words: List<String>) {
612 val width = words[1].toInt()
613 val height = words[2].toInt()
614 if (width == 0 || height == 0) fail("BAD CAPACITY")
615 val rotate = words[3] == "ROTATE"
616 val best = words[4] == "BEST"
617 val order = rectangles(words, 5, width, height, rotate)
618 .sortedWith(compareByDescending<Rect> { it.area }.thenBy { it.number })
619 val area = order.sumOf { it.area }
620 val sheets = mutableListOf<Sheet>()
621 var rotated = 0
622 for (rect in order) {
623 var chosenSheet: Sheet? = null
624 var chosenIndex = -1
625 var chosenArea = Int.MAX_VALUE
626 for (sheet in sheets) {
627 for ((index, free) in sheet.free.withIndex()) {
628 if (!fits(rect, free, rotate)) continue
629 if (chosenSheet == null || (best && free.area < chosenArea)) {
630 chosenSheet = sheet
631 chosenIndex = index
632 chosenArea = free.area
633 }
634 if (!best) break
635 }
636 if (chosenSheet != null && !best) break
637 }
638 if (chosenSheet == null) {
639 chosenSheet = Sheet(width, height)
640 sheets.add(chosenSheet)
641 chosenIndex = 0
642 }
643 val sheet = chosenSheet
644 val free = sheet.free.removeAt(chosenIndex)
645 val turned = rect.width > free.width || rect.height > free.height
646 val w = if (turned) rect.height else rect.width
647 val h = if (turned) rect.width else rect.height
648 if (turned) rotated++
649 sheet.placed.add(Placement(rect, free.x, free.y, w, h, turned))
650 sheet.area += rect.area
651 val rw = free.width - w
652 val rh = free.height - h
653 val top = if (rw <= rh) FreeRect(free.x, free.y + h, free.width, rh)
654 else FreeRect(free.x, free.y + h, w, rh)
655 val right = if (rw <= rh) FreeRect(free.x + w, free.y, rw, h)
656 else FreeRect(free.x + w, free.y, rw, free.height)
657 if (top.width > 0 && top.height > 0) sheet.free.add(top)
658 if (right.width > 0 && right.height > 0) sheet.free.add(right)
659 }
660 emit("GUILLOTINE W=$width H=$height mode=${words[3]} rule=${words[4]} " +
661 "rects=${order.size} sheets=${sheets.size} rotated=$rotated " +
662 "util=${area.toLong() * 100 / (sheets.size * width * height)}")
663 for ((index, sheet) in sheets.withIndex()) {
664 emit("SHEET ${index + 1} placed=${sheet.placed.size} area=${sheet.area}")
665 for (p in sheet.placed) {
666 val rotation = if (p.rotated) " rotated" else ""
667 emit("PLACE ${p.rect.letter} x=${p.x} y=${p.y} w=${p.width} h=${p.height}$rotation")
668 }
669 emit(if (sheet.free.isEmpty()) "FREE none" else "FREE ${sheet.free.joinToString(" ")}")
670 draw(width, height, sheet.placed)
671 }
672 }

674 fun process(words: List<String>) {
675 try {
676 when (val command = words[0]) {
677 "ITEMS" -> itemsCommand(words)
678 "RANDOM" -> random(words)
679 "CUTSTOCK" -> cutstock(words)
680 "SHELF" -> shelf(words)
681 "GUILLOTINE" -> guillotine(words)
682 else -> {
683 if (items == null) fail("NO ITEMS")
684 when (command) {
685 "NEXTFIT", "FIRSTFIT", "BESTFIT", "WORSTFIT", "ALMOSTWORST",
686 "HARMONIC", "KBOUNDED", "FFD", "BFD", "WFD" ->
687 runMethod(command, parameter(command, words, 1))
688 "TRACE" -> trace(words[1], parameter(words[1], words, 2))
689 "REDUCE" -> reduce(words[1], parameter(words[1], words, 2))
690 "BOUNDS" -> boundsCommand()
691 "COMPARE" -> compare(words[1].toInt())
692 "EXACT" -> exact(words[1].toInt())
693 "CHECK" -> check(words)
694 "COVER" -> cover()
695 "MULTIFIT" -> multifit(words[1].toInt())
696 }
697 }
698 }
699 } catch (error: CommandError) {
700 emit("ERROR ${error.message}")
701 }
702 }
703}
