70Packing pack(List<Item> items, int capacity, String method, [int k = 0]) {
71 final result = Packing(method, k);
72 final order = {'FFD', 'BFD', 'WFD'}.contains(method) ? descending(items) : items;
73 final active = <int>[];
74 final current = <int, int>{};
75 for (final item in order) {
76 final before = result.inspections;
77 int? chosen, closed, category;
78 if (method == 'NEXTFIT') {
79 if (result.bins.isNotEmpty) {
80 result.inspections++;
81 final last = result.bins.length - 1;
82 if (result.bins[last].load + item.size <= capacity) chosen = last;
83 }
84 } else if ({'FIRSTFIT', 'FFD', 'KBOUNDED'}.contains(method)) {
85 final candidates = method == 'KBOUNDED'
86 ? active : List.generate(result.bins.length, (i) => i);
87 for (final index in candidates) {
88 result.inspections++;
89 if (result.bins[index].load + item.size <= capacity) {
90 chosen = index;
91 break;
92 }
93 }
94 } else if (method == 'HARMONIC') {
95 category = math.min(capacity ~/ item.size, k);
96 final index = current[category];
97 if (index != null) {
98 result.inspections++;
99 if (result.bins[index].load + item.size <= capacity) chosen = index;
100 }
101 } else {
102 result.inspections += result.bins.length;
103 final fitting = <int>[];
104 for (var index = 0; index < result.bins.length; index++) {
105 if (result.bins[index].load + item.size <= capacity) fitting.add(index);
106 }
107 fitting.sort((a, b) {
108 final compare = result.bins[a].load.compareTo(result.bins[b].load);
109 if (method == 'BESTFIT' || method == 'BFD') {
110 return compare != 0 ? -compare : a.compareTo(b);
111 }
112 return compare != 0 ? compare : a.compareTo(b);
113 });
114 if (fitting.isNotEmpty) {
115 chosen = method == 'ALMOSTWORST' && fitting.length >= 2
116 ? fitting[1] : fitting[0];
117 }
118 }
119 final opened = chosen == null;
120 if (opened) {
121 if (method == 'KBOUNDED' && active.length == k) {
122 closed = active[0];
123 for (final index in active.skip(1)) {
124 if (result.bins[index].load > result.bins[closed!].load ||
125 (result.bins[index].load == result.bins[closed!].load && index < closed!)) {
126 closed = index;
127 }
128 }
129 active.remove(closed);
130 result.closes++;
131 }
132 chosen = result.bins.length;
133 result.bins.add(Bin(category));
134 if (method == 'KBOUNDED') active.add(chosen!);
135 if (method == 'HARMONIC') current[category!] = chosen!;
136 }
137 result.bins[chosen!].put(item);
138 result.steps.add(Step(item, chosen, result.inspections - before, opened, closed, category));
139 }
140 return result;
141}

154Bounds bounds(List<Item> items, int capacity) {
155 final total = items.fold<int>(0, (sum, item) => sum + item.size);
156 final thresholds = <int>{0};
157 for (final item in items) {
158 if (2 * item.size <= capacity) thresholds.add(item.size);
159 }
160 final rows = <BoundRow>[];
161 var best = 0;
162 for (final alpha in thresholds.toList()..sort()) {
163 var j1 = 0, j2 = 0, j3 = 0, secondSum = 0, thirdSum = 0;
164 for (final item in items) {
165 final size = item.size;
166 if (size > capacity - alpha) {
167 j1++;
168 } else if (2 * size > capacity) {
169 j2++;
170 secondSum += size;
171 }
172 if (2 * size <= capacity && size >= alpha) {
173 j3++;
174 thirdSum += size;
175 }
176 }
177 final excess = thirdSum - (j2 * capacity - secondSum);
178 final bound = j1 + j2 + (excess > 0 ? ceilDiv(excess, capacity) : 0);
179 best = math.max(best, bound);
180 rows.add(BoundRow(alpha, j1, j2, j3, bound));
181 }
182 return Bounds(ceilDiv(total, capacity), best, rows);
183}

185class ExactSearch {
186 final List<Item> items;
187 final int capacity, limit, lower;
188 final List<int> remaining;
189 final List<Bin> live = [];
190 final void Function(String) emit;
191 late List<Bin> best;
192 int nodes = 0;
193 String? status;
194 ExactSearch(List<Item> source, this.capacity, this.limit, this.lower,
195 List<Bin> incumbent, this.emit)
196 : items = descending(source), remaining = List.filled(source.length + 1, 0) {
197 best = incumbent.map((bin) => bin.copy()).toList();
198 for (var i = items.length - 1; i >= 0; --i) {
199 remaining[i] = remaining[i + 1] + items[i].size;
200 }
201 }
202 void visit(int index) {
203 if (status != null) return;
204 if (nodes == limit) {
205 status = 'limit';
206 return;
207 }
208 nodes++;
209 final count = live.length;
210 final free = count * capacity - (remaining[0] - remaining[index]);
211 final excess = remaining[index] - free;
212 final minimum = count + (excess > 0 ? ceilDiv(excess, capacity) : 0);
213 if (minimum >= best.length) return;
214 if (index == items.length) {
215 best = live.map((bin) => bin.copy()).toList();
216 emit('IMPROVE bins=$count nodes=$nodes');
217 if (count == lower) status = 'bound';
218 return;
219 }
220 final item = items[index];
221 final seen = <int>{};
222 for (final bin in live) {
223 final old = bin.load;
224 if (!seen.add(old) || old + item.size > capacity) continue;
225 bin.put(item);
226 visit(index + 1);
227 bin.items.removeLast();
228 bin.load = old;
229 if (status != null) return;
230 }
231 if (count + 1 < best.length) {
232 live.add(Bin()..put(item));
233 visit(index + 1);
234 live.removeLast();
235 }
236 }
237}

352 void cmdReduce(List<String> args) {
353 final name = args.first;
354 final params = args.sublist(1);
355 final bins = method(name, params).bins;
356 final before = bins.length;
357 final events = <String>[];
358 while (true) {
359 var eliminated = false;
360 final order = List.generate(bins.length, (i) => i)..sort((a, b) {
361 final byLoad = bins[a].load.compareTo(bins[b].load);
362 return byLoad != 0 ? byLoad : a.compareTo(b);
363 });
364 for (final source in order) {
365 final loads = bins.map((bin) => bin.load).toList();
366 final moves = <(Item, int)>[];
367 var valid = true;
368 for (final item in descending(bins[source].items)) {
369 int? chosen;
370 for (var i = 0; i < bins.length; i++) {
371 if (i == source || loads[i] + item.size > capacity!) continue;
372 if (chosen == null || loads[i] > loads[chosen] ||
373 (loads[i] == loads[chosen] && i < chosen)) {
374 chosen = i;
375 }
376 }
377 if (chosen == null) {
378 valid = false;
379 break;
380 }
381 loads[chosen] += item.size;
382 moves.add((item, chosen));
383 }
384 if (!valid) continue;
385 final text = moves.map((m) => '${m.$1}->${m.$2 + 1}').join(' ');
386 events.add('ELIMINATE bin=${source + 1} load=${bins[source].load} moves=$text');
387 for (final (item, destination) in moves) {
388 bins[destination].put(item);
389 }
390 bins.removeAt(source);
391 eliminated = true;
392 break;
393 }
394 if (!eliminated) break;
395 }
396 final suffix = parameterized.contains(name) ? ' k=${int.parse(params[0])}' : '';
397 emit('REDUCE $name$suffix before=$before after=${bins.length} '
398 'eliminated=${before - bins.length}');
399 for (final event in events) emit(event);
400 binLines(bins, classes: name == 'HARMONIC');
401 }

509 final count = int.parse(args[0]);

511 final largest = items!.fold<int>(0, (m, item) => math.max(m, item.size));
512 var lo = math.max(ceilDiv(total, count), largest);

522 final machines = pack(items!, hi, 'FFD').bins;

527 final lpt = List.generate(count, (_) => Bin());
528 for (final item in descending(items!)) {

530 for (var i = 1; i < count; i++) {

535 emit('LPT makespan=${lpt.fold<int>(0, (m, bin) => math.max(m, bin.load))}');

539 List<(List<int>, int)> maximalPatterns(int cap, List<(int, int)> pieces) {
540 final patterns = <(List<int>, int)>[];
541 void enumerate(int index, int remaining, List<int> prefix) {
542 final size = pieces[index].$1;
543 final maxCount = remaining ~/ size;
544 if (index == pieces.length - 1) {
545 patterns.add(([...prefix, maxCount], remaining - maxCount * size));
546 return;
547 }
548 for (var count = maxCount; count >= 0; count--) {
549 enumerate(index + 1, remaining - count * size, [...prefix, count]);
550 }
551 }
552 enumerate(0, cap, []);
553 return patterns;
554 }

556 void cmdCutstock(List<String> args) {
557 final cap = int.parse(args[0]);
558 if (cap == 0) throw CommandError('BAD CAPACITY');
559 final pieces = <(int, int)>[];
560 final seen = <int>{};
561 for (var i = 1; i < args.length; i++) {
562 final values = args[i].split(':').map(int.parse).toList();
563 final size = values[0], demand = values[1];
564 if (size == 0 || size > cap || demand == 0) {
565 throw CommandError('BAD PIECE $i');
566 }
567 if (!seen.add(size)) throw CommandError('DUPLICATE PIECE $i');
568 pieces.add((size, demand));
569 }
570 pieces.sort((a, b) => b.$1.compareTo(a.$1));
571 final patterns = maximalPatterns(cap, pieces);
572 final remaining = pieces.map((p) => p.$2).toList();
573 final uses = List.filled(patterns.length, 0);
574 while (remaining.any((n) => n > 0)) {
575 var bestIndex = 0, bestUseful = -1;
576 for (var i = 0; i < patterns.length; i++) {
577 var useful = 0;
578 for (var j = 0; j < pieces.length; j++) {
579 useful += math.min(patterns[i].$1[j], remaining[j]);
580 }
581 if (useful > bestUseful) {
582 bestUseful = useful;
583 bestIndex = i;
584 }
585 }
586 uses[bestIndex]++;
587 for (var j = 0; j < remaining.length; j++) {
588 remaining[j] = math.max(0, remaining[j] - patterns[bestIndex].$1[j]);
589 }
590 }
591 final cut = List.filled(pieces.length, 0);
592 for (var i = 0; i < patterns.length; i++) {
593 for (var j = 0; j < pieces.length; j++) {
594 cut[j] += uses[i] * patterns[i].$1[j];
595 }
596 }
597 var quantity = 0, bars = 0, trim = 0, surplus = 0;
598 for (final piece in pieces) quantity += piece.$1 * piece.$2;
599 for (var i = 0; i < patterns.length; i++) {
600 bars += uses[i];
601 trim += uses[i] * patterns[i].$2;
602 }
603 for (var i = 0; i < pieces.length; i++) surplus += cut[i] - pieces[i].$2;
604 emit('CUTSTOCK L=$cap pieces=${pieces.length} patterns=${patterns.length}');
605 for (var i = 0; i < patterns.length; i++) {
606 final breakdown = <String>[];
607 for (var j = 0; j < pieces.length; j++) {
608 final count = patterns[i].$1[j];
609 if (count != 0) breakdown.add('${count}x${pieces[j].$1}');
610 }
611 emit('PATTERN ${i + 1} ${breakdown.join('+')} rest=${patterns[i].$2}');
612 }
613 for (var i = 0; i < uses.length; i++) {
614 if (uses[i] > 0) emit('USE ${i + 1} bars=${uses[i]}');
615 }
616 for (var i = 0; i < pieces.length; i++) {
617 emit('PIECE ${pieces[i].$1} demand=${pieces[i].$2} cut=${cut[i]}');
618 }
619 emit('TOTAL bars=$bars trim=$trim surplus=$surplus waste=${bars * cap - quantity}');
620 final comparison = <Item>[];
621 for (final piece in pieces) {
622 for (var j = 0; j < piece.$2; j++) {
623 comparison.add(Item(comparison.length + 1, piece.$1));
624 }
625 }
626 final number = pack(comparison, cap, 'FIRSTFIT').bins.length;
627 emit('FFD bars=$number waste=${number * cap - quantity}');
628 emit('LOWER bars=${ceilDiv(quantity, cap)}');
629 }

652 void shelfMethod(String name, int width, List<Rectangle> rects, int area) {
653 final shelves = <Shelf>[];
654 final placements = <Placement>[];
655 for (final rect in rects) {
656 int? chosen;
657 if (name == 'NFDH') {
658 if (shelves.isNotEmpty && shelves.last.used + rect.width <= width) {
659 chosen = shelves.length - 1;
660 }
661 } else {
662 for (var i = 0; i < shelves.length; i++) {
663 if (shelves[i].used + rect.width > width) continue;
664 if (chosen == null || (name == 'BFDH' &&
665 shelves[i].used > shelves[chosen].used)) chosen = i;
666 }
667 }
668 if (chosen == null) {
669 final bottom = shelves.isEmpty ? 0 : shelves.last.bottom + shelves.last.height;
670 chosen = shelves.length;
671 shelves.add(Shelf(bottom, rect.height));
672 }
673 final shelf = shelves[chosen];
674 shelf.items.add((rect, shelf.used));
675 placements.add(Placement(rect, shelf.used, shelf.bottom, rect.width, rect.height));
676 shelf.used += rect.width;
677 }
678 final height = shelves.last.bottom + shelves.last.height;
679 emit('$name height=$height shelves=${shelves.length} '
680 'util=${area * 100 ~/ (width * height)}');
681 for (var i = 0; i < shelves.length; i++) {
682 final shelf = shelves[i];
683 emit('LEVEL ${i + 1} y=${shelf.bottom} h=${shelf.height} '
684 'used=${shelf.used} items=${shelf.items.map((p) => '${p.$1.letter}@${p.$2}').join(' ')}');
685 }
686 drawing(width, height, placements);
687 }

689 void cmdShelf(List<String> args) {
690 final width = int.parse(args[0]);
691 if (width == 0) throw CommandError('BAD CAPACITY');
692 final rects = rectangles(args.sublist(1));
693 for (final rect in rects) {
694 if (rect.width == 0 || rect.height == 0 || rect.width > width) {
695 throw CommandError('BAD RECT ${rect.id}');
696 }
697 }
698 rects.sort((a, b) {
699 final byHeight = b.height.compareTo(a.height);
700 if (byHeight != 0) return byHeight;
701 final byWidth = b.width.compareTo(a.width);
702 return byWidth != 0 ? byWidth : a.id.compareTo(b.id);
703 });
704 final area = rects.fold<int>(0, (sum, rect) => sum + rect.area);
705 emit('SHELF W=$width rects=${rects.length} area=$area');
706 for (final name in ['NFDH', 'FFDH', 'BFDH']) {
707 shelfMethod(name, width, rects, area);
708 }
709 final heights = List.filled(width, 0);
710 final placements = <Placement>[];
711 for (final rect in rects) {
712 var chosenX = 0, chosenY = 1 << 60;
713 for (var x = 0; x + rect.width <= width; x++) {
714 var y = 0;
715 for (var column = x; column < x + rect.width; column++) {
716 y = math.max(y, heights[column]);
717 }
718 if (y < chosenY) { chosenX = x; chosenY = y; }
719 }
720 for (var column = chosenX; column < chosenX + rect.width; column++) {
721 heights[column] = chosenY + rect.height;
722 }
723 placements.add(Placement(rect, chosenX, chosenY, rect.width, rect.height));
724 }
725 final height = heights.reduce(math.max);
726 emit('SKYLINE height=$height util=${area * 100 ~/ (width * height)}');
727 for (final place in placements) {
728 emit('PLACE ${place.rectangle.letter} x=${place.x} y=${place.y} '
729 'w=${place.width} h=${place.height}');
730 }
731 drawing(width, height, placements);
732 }

734 (int, int)? freeRectangle(List<Sheet> sheets, int w, int h, bool rotate, String rule) {
735 (int, int)? chosen;
736 int? bestArea;
737 for (var s = 0; s < sheets.length; s++) {
738 for (var f = 0; f < sheets[s].free.length; f++) {
739 final free = sheets[s].free[f];
740 if (!(w <= free.width && h <= free.height) &&
741 !(rotate && h <= free.width && w <= free.height)) continue;
742 if (rule == 'FIRST') return (s, f);
743 if (bestArea == null || free.area < bestArea) {
744 bestArea = free.area;
745 chosen = (s, f);
746 }
747 }
748 }
749 return chosen;
750 }

752 void cmdGuillotine(List<String> args) {
753 final width = int.parse(args[0]), height = int.parse(args[1]);
754 final mode = args[2], rule = args[3];
755 if (width == 0 || height == 0) throw CommandError('BAD CAPACITY');
756 final rects = rectangles(args.sublist(4));
757 final rotate = mode == 'ROTATE';
758 for (final rect in rects) {
759 final fits = rect.width <= width && rect.height <= height;
760 final rotated = rotate && rect.height <= width && rect.width <= height;
761 if (rect.width == 0 || rect.height == 0 || (!fits && !rotated)) {
762 throw CommandError('BAD RECT ${rect.id}');
763 }
764 }
765 rects.sort((a, b) {
766 final byArea = b.area.compareTo(a.area);
767 return byArea != 0 ? byArea : a.id.compareTo(b.id);
768 });
769 final sheets = <Sheet>[];
770 var rotatedCount = 0;
771 for (final rect in rects) {
772 var chosen = freeRectangle(sheets, rect.width, rect.height, rotate, rule);
773 if (chosen == null) {
774 sheets.add(Sheet(width, height));
775 chosen = (sheets.length - 1, 0);
776 }
777 final sheet = sheets[chosen.$1];
778 final free = sheet.free.removeAt(chosen.$2);
779 final rotated = rect.width > free.width || rect.height > free.height;
780 final pw = rotated ? rect.height : rect.width;
781 final ph = rotated ? rect.width : rect.height;
782 if (rotated) rotatedCount++;
783 sheet.placements.add(Placement(rect, free.x, free.y, pw, ph, rotated));
784 sheet.area += rect.area;
785 final rw = free.width - pw;
786 final rh = free.height - ph;
787 final top = rw <= rh
788 ? FreeRect(free.x, free.y + ph, free.width, rh)
789 : FreeRect(free.x, free.y + ph, pw, rh);
790 final right = rw <= rh
791 ? FreeRect(free.x + pw, free.y, rw, ph)
792 : FreeRect(free.x + pw, free.y, rw, free.height);
793 if (top.width > 0 && top.height > 0) sheet.free.add(top);
794 if (right.width > 0 && right.height > 0) sheet.free.add(right);
795 }
796 final area = sheets.fold<int>(0, (sum, sheet) => sum + sheet.area);
797 emit('GUILLOTINE W=$width H=$height mode=$mode rule=$rule '
798 'rects=${rects.length} sheets=${sheets.length} '
799 'rotated=$rotatedCount util=${area * 100 ~/ (sheets.length * width * height)}');
800 for (var i = 0; i < sheets.length; i++) {
801 final sheet = sheets[i];
802 emit('SHEET ${i + 1} placed=${sheet.placements.length} area=${sheet.area}');
803 for (final place in sheet.placements) {
804 var line = 'PLACE ${place.rectangle.letter} x=${place.x} y=${place.y} '
805 'w=${place.width} h=${place.height}';
806 if (place.rotated) line += ' rotated';
807 emit(line);
808 }
809 emit('FREE ${sheet.free.isEmpty ? 'none' : sheet.free.join(' ')}');
810 drawing(width, height, sheet.placements);
811 }
812 }

814 void execute(String line) {
815 final tokens = line.split(' ');
816 final command = tokens.first;
817 final args = tokens.sublist(1);
818 try {
819 if (!{'ITEMS', 'RANDOM', 'CUTSTOCK', 'SHELF', 'GUILLOTINE'}.contains(command)
820 && items == null) throw CommandError('NO ITEMS');
821 if (methods.contains(command)) {
822 cmdMethod(command, args);
823 } else if (command == 'ITEMS') {
824 cmdItems(args);
825 } else if (command == 'RANDOM') {
826 cmdRandom(args);
827 } else if (command == 'TRACE') {
828 cmdTrace(args);
829 } else if (command == 'REDUCE') {
830 cmdReduce(args);
831 } else if (command == 'BOUNDS') {
832 cmdBounds();
833 } else if (command == 'COMPARE') {
834 cmdCompare(args);
835 } else if (command == 'EXACT') {
836 cmdExact(args);
837 } else if (command == 'CHECK') {
838 cmdCheck(args);
839 } else if (command == 'COVER') {
840 cmdCover();
841 } else if (command == 'MULTIFIT') {
842 cmdMultifit(args);
843 } else if (command == 'CUTSTOCK') {
844 cmdCutstock(args);
845 } else if (command == 'SHELF') {
846 cmdShelf(args);
847 } else if (command == 'GUILLOTINE') {
848 cmdGuillotine(args);
849 }
850 } on CommandError catch (error) {
851 emit('ERROR ${error.reason}');
852 }
853 }
854 String result() => output.isEmpty ? '' : '${output.join('\n')}\n';
855}
