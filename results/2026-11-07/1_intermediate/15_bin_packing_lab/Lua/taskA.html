30local function descending(list)
31 local sorted = {}
32 for _, item in ipairs(list) do sorted[#sorted + 1] = item end
33 table.sort(sorted, function(a, b)
34 if a[2] ~= b[2] then return a[2] > b[2] end
35 return a[1] < b[1]
36 end)
37 return sorted

85local function pack(current_items, current_capacity, method, parameter)
86 parameter = parameter or 0
87 local result = {
88 method = method, parameter = parameter, bins = {},
89 steps = {}, inspections = 0, closes = 0
90 }
91 local order = (method == "FFD" or method == "BFD" or method == "WFD")
92 and descending(current_items) or current_items
93 local active, current = {}, {}
94 for _, item in ipairs(order) do
95 local size = item[2]
96 local before = result.inspections
97 local chosen, closed, category = nil, nil, nil
98 if method == "NEXTFIT" then
99 if #result.bins > 0 then
100 result.inspections = result.inspections + 1
101 local index = #result.bins
102 if result.bins[index].load + size <= current_capacity then
103 chosen = index
104 end
105 end
106 elseif method == "FIRSTFIT" or method == "FFD" or method == "KBOUNDED" then
107 local candidates = {}
108 if method == "KBOUNDED" then
109 candidates = active
110 else
111 for index = 1, #result.bins do candidates[#candidates + 1] = index end
112 end
113 for _, index in ipairs(candidates) do
114 result.inspections = result.inspections + 1
115 if result.bins[index].load + size <= current_capacity then
116 chosen = index
117 break
118 end
119 end
120 elseif method == "HARMONIC" then
121 category = math.min(math.floor(current_capacity / size), parameter)
122 local index = current[category]
123 if index then
124 result.inspections = result.inspections + 1
125 if result.bins[index].load + size <= current_capacity then
126 chosen = index
127 end
128 end
129 else
130 result.inspections = result.inspections + #result.bins
131 local fitting = {}
132 for index, bin in ipairs(result.bins) do
133 if bin.load + size <= current_capacity then
134 fitting[#fitting + 1] = index
135 end
136 end
137 if #fitting > 0 then
138 if method == "BESTFIT" or method == "BFD" then
139 chosen = fitting[1]
140 for i = 2, #fitting do
141 local candidate = fitting[i]
142 local candidate_load = result.bins[candidate].load
143 if candidate_load > result.bins[chosen].load then
144 chosen = candidate
145 end
146 end
147 else
148 table.sort(fitting, function(a, b)
149 if result.bins[a].load ~= result.bins[b].load then
150 return result.bins[a].load < result.bins[b].load
151 end
152 return a < b
153 end)
154 chosen = method == "ALMOSTWORST" and fitting[2] or fitting[1]
155 if not chosen then chosen = fitting[1] end
156 end
157 end
158 end
159 local opened = chosen == nil
160 if opened then
161 if method == "KBOUNDED" and #active == parameter then
162 local index = 1
163 for i = 2, #active do
164 if result.bins[active[i]].load > result.bins[active[index]].load then
165 index = i
166 end
167 end
168 closed = active[index]
169 table.remove(active, index)
170 result.closes = result.closes + 1
171 end
172 result.bins[#result.bins + 1] = new_bin(category)
173 chosen = #result.bins
174 if method == "KBOUNDED" then active[#active + 1] = chosen end
175 if method == "HARMONIC" then current[category] = chosen end
176 end
177 put(result.bins[chosen], item)
178 result.steps[#result.steps + 1] = {
179 item = item, bin = chosen, inspected = result.inspections - before,
180 opened = opened, closed = closed, category = category
181 }
182 end
183 return result
184end

186local function bounds(current_items, current_capacity)
187 local total, thresholds = 0, {[0] = true}
188 for _, item in ipairs(current_items) do
189 total = total + item[2]
190 if item[2] * 2 <= current_capacity then thresholds[item[2]] = true end
191 end
192 local ordered = {}
193 for alpha in pairs(thresholds) do ordered[#ordered + 1] = alpha end
194 table.sort(ordered)
195 local rows, lower = {}, 0
196 for _, alpha in ipairs(ordered) do
197 local count1, count2, count3, sum2, sum3 = 0, 0, 0, 0, 0
198 for _, item in ipairs(current_items) do
199 local size = item[2]
200 if size > current_capacity - alpha then
201 count1 = count1 + 1
202 elseif 2 * size > current_capacity then
203 count2 = count2 + 1
204 sum2 = sum2 + size
205 elseif size >= alpha then
206 count3 = count3 + 1
207 sum3 = sum3 + size
208 end
209 end
210 local excess = sum3 - (count2 * current_capacity - sum2)
211 local bound = count1 + count2 +
212 (excess > 0 and ceiling(excess, current_capacity) or 0)
213 lower = math.max(lower, bound)
214 rows[#rows + 1] = {alpha, count1, count2, count3, bound}
215 end
216 return ceiling(total, current_capacity), lower, rows
217end

219local function exact_search(current_items, current_capacity, limit, lower, incumbent)
220 local search = {
221 items = descending(current_items),
222 capacity = current_capacity, limit = limit, lower = lower,
223 best = copy_bins(incumbent), live = {}, nodes = 0,
224 status = nil, events = {}
225 }
226 local remaining = {}
227 remaining[#search.items + 1] = 0
228 for i = #search.items, 1, -1 do
229 remaining[i] = remaining[i + 1] + search.items[i][2]
230 end
231
232 local function visit(index)
233 if search.nodes == search.limit then
234 search.status = "limit"
235 return
236 end
237 search.nodes = search.nodes + 1
238 local count = #search.live
239 local used = remaining[1] - remaining[index]
240 local available = count * current_capacity - used
241 local excess = remaining[index] - available
242 local bound = count + (excess > 0 and ceiling(excess, current_capacity) or 0)
243 if bound >= #search.best then return end
244 if index == #search.items + 1 then
245 search.best = copy_bins(search.live)
246 search.events[#search.events + 1] = "IMPROVE bins=" .. count ..
247 " nodes=" .. search.nodes
248 if count == lower then search.status = "bound" end
249 return
250 end
251 local item = search.items[index]
252 local seen = {}
253 for _, bin in ipairs(search.live) do
254 local load = bin.load
255 if not seen[load] then
256 seen[load] = true
257 if load + item[2] <= current_capacity then
258 put(bin, item)
259 visit(index + 1)
260 table.remove(bin.items)
261 bin.load = load
262 if search.status then return end
263 end
264 end
265 end
266 if count + 1 < #search.best then
267 local bin = new_bin()
268 put(bin, item)
269 search.live[#search.live + 1] = bin
270 visit(index + 1)
271 table.remove(search.live)
272 end
273 end
274 search.visit = visit
275 return search
276end

357local function cmd_reduce(arguments)
358 local method = arguments[1]
359 local params = {}
360 for i = 2, #arguments do params[#params + 1] = arguments[i] end
361 local result = run_method(method, params)
362 local bins = result.bins
363 local before = #bins
364 local events = {}
365 while true do
366 local eliminated = false
367 local sources = {}
368 for i = 1, #bins do sources[i] = i end
369 table.sort(sources, function(a, b)
370 if bins[a].load ~= bins[b].load then return bins[a].load < bins[b].load end
371 return a < b
372 end)
373 for _, source in ipairs(sources) do
374 local loads = {}
375 for i, bin in ipairs(bins) do loads[i] = bin.load end
376 local moves = {}
377 local success = true
378 for _, item in ipairs(descending(bins[source].items)) do
379 local destination = nil
380 for index = 1, #bins do
381 if index ~= source and loads[index] + item[2] <= capacity then
382 if not destination or loads[index] > loads[destination] then
383 destination = index
384 end
385 end
386 end
387 if not destination then
388 success = false
389 break
390 end
391 loads[destination] = loads[destination] + item[2]
392 moves[#moves + 1] = {item, destination}
393 end
394 if success then
395 local descriptions = {}
396 for _, move in ipairs(moves) do
397 descriptions[#descriptions + 1] = item_text(move[1]) .. "->" .. move[2]
398 end
399 events[#events + 1] = "ELIMINATE bin=" .. source ..
400 " load=" .. bins[source].load .. " moves=" .. table.concat(descriptions, " ")
401 for _, move in ipairs(moves) do put(bins[move[2]], move[1]) end
402 table.remove(bins, source)
403 eliminated = true
404 break
405 end
406 end
407 if not eliminated then break end
408 end
409 emit("REDUCE " .. method_name(result) .. " before=" .. before ..
410 " after=" .. #bins .. " eliminated=" .. (before - #bins))
411 for _, event in ipairs(events) do emit(event) end
412 bin_lines(bins, method == "HARMONIC")
413end

544local function cmd_multifit(arguments)
545 local count = tonumber(arguments[1])
546 if count == 0 then fail("BAD PARAMETER") end
547 local total = total_size()
548 local largest = 0
549 for _, item in ipairs(items) do largest = math.max(largest, item[2]) end
550 local lower = math.max(ceiling(total, count), largest)
551 local upper = math.max(ceiling(2 * total, count), largest)
552 emit("MULTIFIT m=" .. count .. " lower=" .. lower .. " upper=" .. upper)
553 while lower < upper do
554 local middle = math.floor((lower + upper) / 2)
555 local bins = #pack(items, middle, "FFD").bins
556 local success = bins <= count
557 emit("TRY cap=" .. middle .. " bins=" .. bins .. (success and " ok" or " fail"))
558 if success then
559 upper = middle
560 else
561 lower = middle + 1
562 end
563 end
564 local machines = pack(items, upper, "FFD").bins
565 while #machines < count do machines[#machines + 1] = new_bin() end
566 local makespan = 0
567 for _, bin in ipairs(machines) do makespan = math.max(makespan, bin.load) end
568 emit("RESULT cap=" .. upper .. " makespan=" .. makespan)
569 machine_lines(machines)
570 local lpt = {}
571 for i = 1, count do lpt[i] = new_bin() end
572 for _, item in ipairs(descending(items)) do
573 local chosen = 1
574 for i = 2, count do
575 if lpt[i].load < lpt[chosen].load then chosen = i end
576 end
577 put(lpt[chosen], item)
578 end
579 makespan = 0
580 for _, bin in ipairs(lpt) do makespan = math.max(makespan, bin.load) end
581 emit("LPT makespan=" .. makespan)
582 machine_lines(lpt)
583end

585local function maximal_patterns(bar_length, pieces)
586 local patterns = {}
587 local function enumerate_prefix(index, remaining, prefix)
588 local size = pieces[index][1]
589 local maximum = math.floor(remaining / size)
590 if index == #pieces then
591 local pattern = {}
592 for _, value in ipairs(prefix) do pattern[#pattern + 1] = value end
593 pattern[#pattern + 1] = maximum
594 patterns[#patterns + 1] = {
595 counts = pattern, trim = remaining - maximum * size
596 }
597 return
598 end
599 for count = maximum, 0, -1 do
600 prefix[#prefix + 1] = count
601 enumerate_prefix(index + 1, remaining - count * size, prefix)
602 prefix[#prefix] = nil
603 end
604 end
605 enumerate_prefix(1, bar_length, {})
606 return patterns
607end

609local function cmd_cutstock(arguments)
610 local bar_length = tonumber(arguments[1])
611 if bar_length == 0 then fail("BAD CAPACITY") end
612 local pieces, seen = {}, {}
613 for index = 2, #arguments do
614 local token = arguments[index]
615 local size_text, demand_text = token:match("^(%d+):(%d+)$")
616 local size, demand = tonumber(size_text), tonumber(demand_text)
617 if size == 0 or size > bar_length or demand == 0 then
618 fail("BAD PIECE " .. (index - 1))
619 end
620 if seen[size] then fail("DUPLICATE PIECE " .. (index - 1)) end
621 seen[size] = true
622 pieces[#pieces + 1] = {size, demand}
623 end
624 table.sort(pieces, function(a, b) return a[1] > b[1] end)
625 local patterns = maximal_patterns(bar_length, pieces)
626 local remaining, uses, cut = {}, {}, {}
627 for i, piece in ipairs(pieces) do
628 remaining[i] = piece[2]
629 cut[i] = 0
630 end
631 for index = 1, #patterns do uses[index] = 0 end
632
633 while true do
634 local unfinished = false
635 for _, amount in ipairs(remaining) do
636 if amount > 0 then unfinished = true; break end
637 end
638 if not unfinished then break end
639 local chosen, best_useful = nil, -1
640 for index, pattern in ipairs(patterns) do
641 local useful = 0
642 for j, amount in ipairs(remaining) do
643 useful = useful + math.min(pattern.counts[j], amount)
644 end
645 if useful > best_useful then
646 chosen, best_useful = index, useful
647 end
648 end
649 uses[chosen] = uses[chosen] + 1
650 for j, amount in ipairs(remaining) do
651 remaining[j] = math.max(0, amount - patterns[chosen].counts[j])
652 end
653 end
654
655 local quantity, bars, trim, total_cut, demanded = 0, 0, 0, 0, 0
656 for _, piece in ipairs(pieces) do
657 quantity = quantity + piece[1] * piece[2]
658 demanded = demanded + piece[2]
659 end
660 for index, use in ipairs(uses) do
661 bars = bars + use
662 trim = trim + use * patterns[index].trim
663 for j, count in ipairs(patterns[index].counts) do
664 cut[j] = cut[j] + use * count
665 end
666 end
667 for _, count in ipairs(cut) do total_cut = total_cut + count end
668
669 emit("CUTSTOCK L=" .. bar_length .. " pieces=" .. #pieces ..
670 " patterns=" .. #patterns)
671 for index, pattern in ipairs(patterns) do
672 local breakdown = {}
673 for j, count in ipairs(pattern.counts) do
674 if count > 0 then
675 breakdown[#breakdown + 1] = count .. "x" .. pieces[j][1]
676 end
677 end
678 emit("PATTERN " .. index .. " " .. table.concat(breakdown, "+") ..
679 " rest=" .. pattern.trim)
680 end
681 for index, use in ipairs(uses) do
682 if use > 0 then emit("USE " .. index .. " bars=" .. use) end
683 end
684 for index, piece in ipairs(pieces) do
685 emit("PIECE " .. piece[1] .. " demand=" .. piece[2] ..
686 " cut=" .. cut[index])
687 end
688 emit("TOTAL bars=" .. bars .. " trim=" .. trim ..
689 " surplus=" .. (total_cut - demanded) ..
690 " waste=" .. (bars * bar_length - quantity))
691
692 local comparison_items = {}
693 for _, piece in ipairs(pieces) do
694 for _ = 1, piece[2] do
695 comparison_items[#comparison_items + 1] = {
696 #comparison_items + 1, piece[1]
697 }
698 end
699 end
700 local comparison = pack(comparison_items, bar_length, "FIRSTFIT")
701 local comparison_bars = #comparison.bins
702 emit("FFD bars=" .. comparison_bars ..
703 " waste=" .. (comparison_bars * bar_length - quantity))
704 emit("LOWER bars=" .. ceiling(quantity, bar_length))
705end

734local function shelf_method(method, strip_width, rects, area)
735 local shelves, placements = {}, {}
736 for _, rect in ipairs(rects) do
737 local number, w, h = table.unpack(rect)
738 local chosen = nil
739 if method == "NFDH" then
740 if #shelves > 0 and shelves[#shelves].used + w <= strip_width then
741 chosen = #shelves
742 end
743 else
744 for index, shelf in ipairs(shelves) do
745 if shelf.used + w <= strip_width then
746 if method == "FFDH" then
747 chosen = index
748 break
749 end
750 if not chosen or shelf.used > shelves[chosen].used then
751 chosen = index
752 end
753 end
754 end
755 end
756 if not chosen then
757 local bottom = #shelves == 0 and 0 or
758 shelves[#shelves].bottom + shelves[#shelves].height
759 shelves[#shelves + 1] = {
760 bottom = bottom, height = h, used = 0, items = {}
761 }
762 chosen = #shelves
763 end
764 local shelf = shelves[chosen]
765 shelf.items[#shelf.items + 1] = {number, shelf.used}
766 placements[#placements + 1] = {
767 number, shelf.used, shelf.bottom, w, h, false
768 }
769 shelf.used = shelf.used + w
770 end
771 local height = shelves[#shelves].bottom + shelves[#shelves].height
772 emit(method .. " height=" .. height .. " shelves=" .. #shelves ..
773 " util=" .. math.floor(area * 100 / (strip_width * height)))
774 for index, shelf in ipairs(shelves) do
775 local parts = {}
776 for _, item in ipairs(shelf.items) do
777 parts[#parts + 1] = string.char(64 + item[1]) .. "@" .. item[2]
778 end
779 emit("LEVEL " .. index .. " y=" .. shelf.bottom ..
780 " h=" .. shelf.height .. " used=" .. shelf.used ..
781 " items=" .. table.concat(parts, " "))
782 end
783 drawing(strip_width, height, placements)
784end

786local function cmd_shelf(arguments)
787 local strip_width = tonumber(arguments[1])
788 if strip_width == 0 then fail("BAD CAPACITY") end
789 local tokens = {}
790 for i = 2, #arguments do tokens[#tokens + 1] = arguments[i] end
791 local rects = rectangles(tokens)
792 for _, rect in ipairs(rects) do
793 if rect[2] == 0 or rect[3] == 0 or rect[2] > strip_width then
794 fail("BAD RECT " .. rect[1])
795 end
796 end
797 table.sort(rects, function(a, b)
798 if a[3] ~= b[3] then return a[3] > b[3] end
799 if a[2] ~= b[2] then return a[2] > b[2] end
800 return a[1] < b[1]
801 end)
802 local area = 0
803 for _, rect in ipairs(rects) do area = area + rect[2] * rect[3] end
804 emit("SHELF W=" .. strip_width .. " rects=" .. #rects .. " area=" .. area)
805 for _, method in ipairs({"NFDH", "FFDH", "BFDH"}) do
806 shelf_method(method, strip_width, rects, area)
807 end
808 local heights, placements = {}, {}
809 for column = 1, strip_width do heights[column] = 0 end
810 for _, rect in ipairs(rects) do
811 local number, w, h = table.unpack(rect)
812 local best_x, best_y = nil, nil
813 for x = 0, strip_width - w do
814 local y = 0
815 for column = x + 1, x + w do
816 y = math.max(y, heights[column])
817 end
818 if not best_y or y < best_y then
819 best_x, best_y = x, y
820 end
821 end
822 placements[#placements + 1] = {
823 number, best_x, best_y, w, h, false
824 }
825 for column = best_x + 1, best_x + w do
826 heights[column] = best_y + h
827 end
828 end
829 local height = 0
830 for _, col_height in ipairs(heights) do
831 height = math.max(height, col_height)
832 end
833 emit("SKYLINE height=" .. height ..
834 " util=" .. math.floor(area * 100 / (strip_width * height)))
835 for _, rect in ipairs(placements) do
836 local number, x, y, w, h = table.unpack(rect)
837 emit("PLACE " .. string.char(64 + number) ..
838 " x=" .. x .. " y=" .. y .. " w=" .. w .. " h=" .. h)
839 end
840 drawing(strip_width, height, placements)
841end

847 area = 0

851local function free_rectangle(sheets, width, height, rotate, rule)
852 local chosen_sheet, chosen_index, best_area = nil, nil, nil
853 for number, sheet in ipairs(sheets) do
854 for index, free in ipairs(sheet.free) do
855 local fw, fh = free[3], free[4]
856 local fits = width <= fw and height <= fh
857 local swapped = rotate and height <= fw and width <= fh
858 if fits or swapped then
859 if rule == "FIRST" then return number, index end
860 local area = fw * fh
861 if not best_area or area < best_area then
862 chosen_sheet, chosen_index, best_area = number, index, area
863 end
864 end
865 end
866 end
867 return chosen_sheet, chosen_index
868end

870local function cmd_guillotine(arguments)
871 local sheet_width, sheet_height = tonumber(arguments[1]), tonumber(arguments[2])
872 if sheet_width == 0 or sheet_height == 0 then fail("BAD CAPACITY") end
873 local mode, rule = arguments[3], arguments[4]
874 local rotate = mode == "ROTATE"
875 local tokens = {}
876 for i = 5, #arguments do tokens[#tokens + 1] = arguments[i] end
877 local rects = rectangles(tokens)
878 for _, rect in ipairs(rects) do
879 local number, w, h = table.unpack(rect)
880 local fits = w <= sheet_width and h <= sheet_height
881 local swapped = rotate and h <= sheet_width and w <= sheet_height
882 if w == 0 or h == 0 or not (fits or swapped) then
883 fail("BAD RECT " .. number)
884 end
885 end
886 table.sort(rects, function(a, b)
887 local left, right = a[2] * a[3], b[2] * b[3]
888 if left ~= right then return left > right end
889 return a[1] < b[1]
890 end)
891 local sheets, rotated_count = {}, 0
892 for _, rect in ipairs(rects) do
893 local number, w, h = table.unpack(rect)
894 local selected_sheet, selected_index =
895 free_rectangle(sheets, w, h, rotate, rule)
896 if not selected_sheet then
897 sheets[#sheets + 1] = new_sheet(sheet_width, sheet_height)
898 selected_sheet, selected_index = #sheets, 1
899 end
900 local sheet = sheets[selected_sheet]
901 local free = table.remove(sheet.free, selected_index)
902 local x, y, fw, fh = table.unpack(free)
903 local rotated = not (w <= fw and h <= fh)
904 local pw, ph = w, h
905 if rotated then pw, ph = h, w end
906 if rotated then rotated_count = rotated_count + 1 end
907 sheet.placements[#sheet.placements + 1] = {
908 number, x, y, pw, ph, rotated
909 }
910 sheet.area = sheet.area + w * h
911
912 local rw, rh = fw - pw, fh - ph
913 local top, right
914 if rw <= rh then
915 top = {x, y + ph, fw, rh}
916 right = {x + pw, y, rw, ph}
917 else
918 top = {x, y + ph, pw, rh}
919 right = {x + pw, y, rw, fh}
920 end
921 if top[3] > 0 and top[4] > 0 then
922 sheet.free[#sheet.free + 1] = top
923 end
924 if right[3] > 0 and right[4] > 0 then
925 sheet.free[#sheet.free + 1] = right
926 end
927 end
928
929 local area = 0
930 for _, sheet in ipairs(sheets) do area = area + sheet.area end
931 local util = math.floor(area * 100 / (#sheets * sheet_width * sheet_height))
932 emit("GUILLOTINE W=" .. sheet_width .. " H=" .. sheet_height ..
933 " mode=" .. mode .. " rule=" .. rule ..
934 " rects=" .. #rects .. " sheets=" .. #sheets ..
935 " rotated=" .. rotated_count .. " util=" .. util)
936 for number, sheet in ipairs(sheets) do
937 emit("SHEET " .. number .. " placed=" .. #sheet.placements ..
938 " area=" .. sheet.area)
939 for _, placement in ipairs(sheet.placements) do
940 local id, x, y, w, h, rotated = table.unpack(placement)
941 local text = "PLACE " .. string.char(64 + id) .. " x=" .. x ..
942 " y=" .. y .. " w=" .. w .. " h=" .. h
943 if rotated then text = text .. " rotated" end
944 emit(text)
945 end
946 local free = {}
947 for _, region in ipairs(sheet.free) do
948 local x, y, w, h = table.unpack(region)
949 free[#free + 1] = x .. "," .. y .. "," .. w .. "x" .. h
950 end
951 emit("FREE " .. (#free > 0 and table.concat(free, " ") or "none"))
952 drawing(sheet_width, sheet_height, sheet.placements)
953 end
954end

956local function execute(line)
957 local words = {}
958 for token in line:gmatch("%S+") do
959 words[#words + 1] = token
960 end
961 local command = words[1]
962 local arguments = {}
963 for i = 2, #words do arguments[#arguments + 1] = words[i] end
964 local ok, reason = pcall(function()
965 if command ~= "ITEMS" and command ~= "RANDOM" and
966 command ~= "CUTSTOCK" and command ~= "SHELF" and
967 command ~= "GUILLOTINE" and not items then
968 fail("NO ITEMS")
969 end
970 if method_set[command] then
971 show_method(command, arguments)
972 elseif command == "ITEMS" then
973 cmd_items(arguments)
974 elseif command == "RANDOM" then
975 cmd_random(arguments)
976 elseif command == "TRACE" then
977 cmd_trace(arguments)
978 elseif command == "REDUCE" then
979 cmd_reduce(arguments)
980 elseif command == "BOUNDS" then
981 cmd_bounds()
982 elseif command == "COMPARE" then
983 cmd_compare(arguments)
984 elseif command == "EXACT" then
985 cmd_exact(arguments)
986 elseif command == "CHECK" then
987 cmd_check(arguments)
988 elseif command == "COVER" then
989 cmd_cover()
990 elseif command == "MULTIFIT" then
991 cmd_multifit(arguments)
992 elseif command == "CUTSTOCK" then
993 cmd_cutstock(arguments)
994 elseif command == "SHELF" then
995 cmd_shelf(arguments)
996 elseif command == "GUILLOTINE" then
997 cmd_guillotine(arguments)
998 end
999 end)
1000 if not ok then
1001 if type(reason) == "table" and reason.reason then
1002 emit("ERROR " .. reason.reason)
1003 else
1004 error(reason)
1005 end
1006 end
1007end
