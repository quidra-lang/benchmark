33def pack(items, capacity, method, k = 0)
34 bins = []
35 steps = []
36 current = {}
37 active = []
38 inspections = 0
39 closes = 0
40 order = %w[FFD BFD WFD].include?(method) ? descending(items) : items
41 order.each do |item|
42 before = inspections
43 destination = nil
44 closed = nil
45 klass = nil
46 if method == 'NEXTFIT'
47 unless bins.empty?
48 inspections += 1
49 destination = bins.length - 1 if bins.last.load + item.size <= capacity
50 end
51 elsif method == 'HARMONIC'
52 klass = [capacity / item.size, k].min
53 index = current[klass]
54 unless index.nil?
55 inspections += 1
56 destination = index if bins[index].load + item.size <= capacity
57 end
58 elsif %w[FIRSTFIT FFD KBOUNDED].include?(method)
59 candidates = method == 'KBOUNDED' ? active : bins.each_index.to_a
60 candidates.each do |index|
61 inspections += 1
62 if bins[index].load + item.size <= capacity
63 destination = index
64 break
65 end
66 end
67 else
68 inspections += bins.length
69 candidates = bins.each_index.select { |i| bins[i].load + item.size <= capacity }
70 largest = %w[BESTFIT BFD].include?(method)
71 candidates.sort_by! { |i| [largest ? -bins[i].load : bins[i].load, i] }
72 destination = candidates[method == 'ALMOSTWORST' && candidates.length > 1 ? 1 : 0]
73 end
74 fresh = destination.nil?
75 if fresh
76 if method == 'KBOUNDED' && active.length == k
77 index = active.min_by { |i| [-bins[i].load, i] }
78 active.delete(index)
79 closed = index + 1
80 closes += 1
81 end
82 destination = bins.length
83 bins << Bin.new(0, [], klass)
84 current[klass] = destination if method == 'HARMONIC'
85 active << destination if method == 'KBOUNDED'
86 end
87 bins[destination].items << item
88 bins[destination].load += item.size
89 steps << {item: item, bin: destination + 1, inspected: inspections - before,
90 fresh: fresh, closed: closed, klass: klass}
91 end
92 {bins: bins, steps: steps, inspections: inspections, closes: closes}
93end

95def bounds(items, capacity)
96 thresholds = [0, *items.select { |i| 2 * i.size <= capacity }.map(&:size)].uniq.sort
97 rows = thresholds.map do |alpha|
98 j1 = items.select { |i| i.size > capacity - alpha }
99 j2 = items.select { |i| 2 * i.size > capacity && i.size <= capacity - alpha }
100 j3 = items.select { |i| 2 * i.size <= capacity && i.size >= alpha }
101 extra = total(j3) - (j2.length * capacity - total(j2))
102 [alpha, j1.length, j2.length, j3.length, j1.length + j2.length + (extra.positive? ? ceil(extra, capacity) : 0)]
103 end
104 [ceil(total(items), capacity), rows.map(&:last).max, rows]
105end

120 def bin_line(bin, number, capacity, harmonic = false, over = false)
121 parts = ['BIN', number, "load=#{bin.load}/#{capacity}", 'items=' + bin.items.map { |i| notation(i) }.join(' ')]

123 parts << 'OVER' if over

127 def header(method, k, result)
128 parameter = %w[HARMONIC KBOUNDED].include?(method) ? " k=#{k}" : ''
129 method + parameter + " bins=#{result[:bins].length} inspections=#{result[:inspections]}" +
130 " waste=#{result[:bins].length * @capacity - total(@items)}" + (method == 'KBOUNDED' ? " closes=#{result[:closes]}" : '')

133 def reduce(method, k)
134 bins = pack(@items, @capacity, method, k)[:bins]
135 before = bins.length
136 eliminations = []
137 loop do
138 succeeded = false
139 bins.each_index.sort_by { |i| [bins[i].load, i] }.each do |source|
140 loads = bins.map(&:load)
141 moves = []
142 possible = true
143 descending(bins[source].items).each do |item|
144 candidates = bins.each_index.select { |i| i != source && loads[i] + item.size <= @capacity }
145 destination = candidates.min_by { |i| [-loads[i], i] }
146 if destination.nil?
147 possible = false
148 break
149 end
150 loads[destination] += item.size
151 moves << [item, destination]
152 end
153 next unless possible
154 text = moves.map { |item, destination| "#{notation(item)}->#{destination + 1}" }.join(' ')
155 eliminations << "ELIMINATE bin=#{source + 1} load=#{bins[source].load} moves=#{text}"
156 moves.each do |item, destination|
157 bins[destination].items << item
158 bins[destination].load += item.size
159 end
160 bins.delete_at(source)
161 succeeded = true
162 break
163 end
164 break unless succeeded
165 end
166 emit('REDUCE', method, *(%w[HARMONIC KBOUNDED].include?(method) ? ["k=#{k}"] : []),
167 "before=#{before}", "after=#{bins.length}", "eliminated=#{before - bins.length}")
168 eliminations.each { |line| emit(line) }
169 bins.each_with_index { |b, i| bin_line(b, i + 1, @capacity, method == 'HARMONIC') }
170 end

172 def exact(limit)
173 raise CommandError, 'BAD PARAMETER' if limit.zero?
174 raise CommandError, 'TOO MANY ITEMS' if @items.length > 20
175 lower = bounds(@items, @capacity)[1]
176 best = copy_bins(pack(@items, @capacity, 'FFD')[:bins])
177 emit('EXACT', "ffd=#{best.length}", "L2=#{lower}")
178 order = descending(@items)
179 remaining = Array.new(order.length + 1, 0)
180 (order.length - 1).downto(0) { |i| remaining[i] = remaining[i + 1] + order[i].size }
181 bins = []
182 nodes = 0
183 status = best.length == lower ? 'bound' : 'complete'
184 stopped = status == 'bound'
185 visit = lambda do |index, used|
186 if nodes == limit
187 status = 'limit'
188 stopped = true
189 next
190 end
191 nodes += 1
192 free = bins.length * @capacity - used
193 extra = remaining[index] - free
194 bound = bins.length + (extra.positive? ? ceil(extra, @capacity) : 0)
195 next if bound >= best.length
196 if index == order.length
197 best = copy_bins(bins)
198 emit('IMPROVE', "bins=#{best.length}", "nodes=#{nodes}")
199 if best.length == lower
200 status = 'bound'
201 stopped = true
202 end
203 next
204 end
205 item = order[index]
206 seen = {}
207 bins.each do |bin|
208 next if seen.key?(bin.load)
209 seen[bin.load] = true
210 next if bin.load + item.size > @capacity
211 bin.load += item.size
212 bin.items << item
213 visit.call(index + 1, used + item.size)
214 bin.items.pop
215 bin.load -= item.size
216 break if stopped
217 end
218 next if stopped
219 if bins.length + 1 < best.length
220 bins << Bin.new(item.size, [item], nil)
221 visit.call(index + 1, used + item.size)
222 bins.pop
223 end
224 end
225 visit.call(0, 0) unless stopped
226 emit('RESULT', "bins=#{best.length}", "nodes=#{nodes}", "status=#{status}")
227 best.each_with_index { |b, i| bin_line(b, i + 1, @capacity) }
228 end

230 def cover
231 emit('COVER', "total=#{total(@items)}", "upper=#{total(@items) / @capacity}")
232 %w[DNF BIGSMALL].each do |method|
233 remaining = method == 'DNF' ? @items.dup : descending(@items)
234 bins = []
235 until remaining.empty?
236 first = remaining.shift
237 bin = Bin.new(first.size, [first], nil)
238 while bin.load < @capacity && !remaining.empty?
239 item = method == 'DNF' ? remaining.shift : remaining.pop
240 bin.items << item
241 bin.load += item.size
242 end
243 bins << bin
244 end
245 covered = bins.select { |b| b.load >= @capacity }
246 rest = bins.find { |b| b.load < @capacity }
247 emit(method, "covered=#{covered.length}", "rest=#{rest ? rest.load : 0}")
248 covered.each_with_index { |b, i| bin_line(b, i + 1, @capacity) }
249 emit('REST', 'items=' + rest.items.map { |i| notation(i) }.join(' ')) if rest
250 end
251 end

254 count.times do |i|
255 bin = bins[i]
256 items = bin && !bin.items.empty? ? bin.items.map { |item| notation(item) }.join(' ') : '-'
257 emit('MACHINE', i + 1, "load=#{bin ? bin.load : 0}", "items=#{items}")
258 end

261 def multifit(count)
262 raise CommandError, 'BAD PARAMETER' if count.zero?
263 size = total(@items)
264 largest = @items.map(&:size).max
265 lo = [ceil(size, count), largest].max
266 hi = [ceil(2 * size, count), largest].max
267 emit('MULTIFIT', "m=#{count}", "lower=#{lo}", "upper=#{hi}")
268 while lo < hi
269 mid = (lo + hi) / 2
270 bins = pack(@items, mid, 'FFD')[:bins]
271 ok = bins.length <= count
272 emit('TRY', "cap=#{mid}", "bins=#{bins.length}", ok ? 'ok' : 'fail')
273 if ok
274 hi = mid
275 else
276 lo = mid + 1
277 end
278 end
279 bins = pack(@items, hi, 'FFD')[:bins]
280 emit('RESULT', "cap=#{hi}", "makespan=#{bins.map(&:load).max}")
281 machines(bins, count)
282 lpt = Array.new(count) { Bin.new(0, [], nil) }
283 descending(@items).each do |item|
284 chosen = lpt.each_index.min_by { |i| [lpt[i].load, i] }
285 lpt[chosen].items << item
286 lpt[chosen].load += item.size
287 end
288 emit('LPT', "makespan=#{lpt.map(&:load).max}")
289 machines(lpt, count)
290 end

292 def cutstock(capacity, tokens)
293 raise CommandError, 'BAD CAPACITY' if capacity.zero?
294 pieces = tokens.map { |t| t.split(':').map(&:to_i) }
295 seen = {}
296 pieces.each_with_index do |(length, demand), i|
297 raise CommandError, "BAD PIECE #{i + 1}" if length.zero? || length > capacity || demand.zero?
298 raise CommandError, "DUPLICATE PIECE #{i + 1}" if seen.key?(length)
299 seen[length] = true
300 end
301 pieces.sort_by! { |length, _| -length }
302 patterns = []
303 enumerate = lambda do |index, rest, prefix|
304 maximum = rest / pieces[index][0]
305 if index == pieces.length - 1
306 patterns << [*prefix, maximum]
307 else
308 maximum.downto(0) do |count|
309 enumerate.call(index + 1, rest - count * pieces[index][0], [*prefix, count])
310 end
311 end
312 end
313 enumerate.call(0, capacity, [])
314 remaining = pieces.map(&:last)
315 used = Array.new(patterns.length, 0)
316 while remaining.any?(&:positive?)
317 chosen = patterns.each_index.max_by do |p|
318 [patterns[p].each_with_index.sum { |a, i| [a, remaining[i]].min }, -p]
319 end
320 used[chosen] += 1
321 pieces.each_index { |i| remaining[i] = [0, remaining[i] - patterns[chosen][i]].max }
322 end
323 trims = patterns.map { |p| capacity - p.each_with_index.sum { |a, i| a * pieces[i][0] } }
324 cut = pieces.each_index.map { |i| patterns.each_with_index.sum { |p, j| p[i] * used[j] } }
325 bars = used.sum
326 size = pieces.sum { |length, demand| length * demand }
327 emit('CUTSTOCK', "L=#{capacity}", "pieces=#{pieces.length}", "patterns=#{patterns.length}")
328 patterns.each_with_index do |p, i|
329 breakdown = p.each_with_index.filter_map { |a, j| "#{a}x#{pieces[j][0]}" if a.positive? }.join('+')
330 emit('PATTERN', i + 1, breakdown, "rest=#{trims[i]}")
331 end
332 used.each_with_index { |n, i| emit('USE', i + 1, "bars=#{n}") if n.positive? }
333 pieces.each_with_index { |(length, demand), i| emit('PIECE', length, "demand=#{demand}", "cut=#{cut[i]}") }
334 emit('TOTAL', "bars=#{bars}", "trim=#{used.each_with_index.sum { |n, i| n * trims[i] }}",
335 "surplus=#{cut.each_with_index.sum { |n, i| n - pieces[i][1] }}", "waste=#{bars * capacity - size}")
336 items = []
337 pieces.each { |length, demand| demand.times { items << Item.new(items.length + 1, length) } }
338 ffd = pack(items, capacity, 'FIRSTFIT')[:bins].length
339 emit('FFD', "bars=#{ffd}", "waste=#{ffd * capacity - size}")
340 emit('LOWER', "bars=#{ceil(size, capacity)}")
341 end

360 def shelf(width, tokens)
361 raise CommandError, 'BAD CAPACITY' if width.zero?
362 rectangles = rectangles(tokens)
363 rectangles.each do |r|
364 raise CommandError, "BAD RECT #{r.id}" if r.w.zero? || r.h.zero? || r.w > width
365 end
366 order = rectangles.sort_by { |r| [-r.h, -r.w, r.id] }
367 area = rectangles.sum { |r| r.w * r.h }
368 emit('SHELF', "W=#{width}", "rects=#{rectangles.length}", "area=#{area}")
369 %w[NFDH FFDH BFDH].each do |method|
370 shelves = []
371 order.each do |r|
372 candidates = shelves.each_index.select { |i| shelves[i][:used] + r.w <= width }
373 candidates.select! { |i| i == shelves.length - 1 } if method == 'NFDH'
374 candidates.sort_by! { |i| [-shelves[i][:used], i] } if method == 'BFDH'
375 index = candidates.first
376 if index.nil?
377 index = shelves.length
378 last = shelves.last
379 shelves << {y: last ? last[:y] + last[:h] : 0, h: r.h, used: 0, placements: []}
380 end
381 s = shelves[index]
382 s[:placements] << Placement.new(r.id, r.w, r.h, s[:used], s[:y], false)
383 s[:used] += r.w
384 end
385 height = shelves.last[:y] + shelves.last[:h]
386 emit(method, "height=#{height}", "shelves=#{shelves.length}", "util=#{area * 100 / (width * height)}")
387 shelves.each_with_index do |s, i|
388 items = s[:placements].map { |p| "#{letter(p.id)}@#{p.x}" }.join(' ')
389 emit('LEVEL', i + 1, "y=#{s[:y]}", "h=#{s[:h]}", "used=#{s[:used]}", "items=#{items}")
390 end
391 drawing(width, height, shelves.flat_map { |s| s[:placements] })
392 end
393 heights = Array.new(width, 0)
394 placements = []
395 order.each do |r|
396 x = (0..(width - r.w)).min_by { |candidate| [heights[candidate, r.w].max, candidate] }
397 y = heights[x, r.w].max
398 placements << Placement.new(r.id, r.w, r.h, x, y, false)
399 heights.fill(y + r.h, x, r.w)
400 end
401 height = heights.max
402 emit('SKYLINE', "height=#{height}", "util=#{area * 100 / (width * height)}")
403 placements.each { |p| emit('PLACE', letter(p.id), "x=#{p.x}", "y=#{p.y}", "w=#{p.w}", "h=#{p.h}") }
404 drawing(width, height, placements)
405 end

407 def guillotine(width, height, mode, rule, tokens)
408 raise CommandError, 'BAD CAPACITY' if width.zero? || height.zero?
409 rectangles = rectangles(tokens)
410 fits = ->(r, f) { (r.w <= f.w && r.h <= f.h) || (mode == 'ROTATE' && r.h <= f.w && r.w <= f.h) }
411 rectangles.each do |r|
412 raise CommandError, "BAD RECT #{r.id}" if r.w.zero? || r.h.zero? || !fits.call(r, FreeRect.new(0, 0, width, height))
413 end
414 sheets = []
415 rotated_count = 0
416 rectangles.sort_by { |r| [-r.w * r.h, r.id] }.each do |r|
417 chosen_sheet = nil
418 chosen_free = nil
419 best_area = nil
420 sheets.each_with_index do |sheet, i|
421 sheet[:free].each_with_index do |f, j|
422 next unless fits.call(r, f)
423 if chosen_sheet.nil? || (rule == 'BEST' && f.w * f.h < best_area)
424 chosen_sheet = i
425 chosen_free = j
426 best_area = f.w * f.h
427 end
428 end
429 end
430 if chosen_sheet.nil?
431 chosen_sheet = sheets.length
432 chosen_free = 0
433 sheets << {free: [FreeRect.new(0, 0, width, height)], placements: []}
434 end
435 sheet = sheets[chosen_sheet]
436 f = sheet[:free].delete_at(chosen_free)
437 rotated = !(r.w <= f.w && r.h <= f.h)
438 w, h = rotated ? [r.h, r.w] : [r.w, r.h]
439 rotated_count += 1 if rotated
440 sheet[:placements] << Placement.new(r.id, w, h, f.x, f.y, rotated)
441 rw = f.w - w
442 rh = f.h - h
443 top = FreeRect.new(f.x, f.y + h, rw <= rh ? f.w : w, rh)
444 right = FreeRect.new(f.x + w, f.y, rw, rw <= rh ? h : f.h)
445 [top, right].each { |free| sheet[:free] << free if free.w.positive? && free.h.positive? }
446 end
447 area = rectangles.sum { |r| r.w * r.h }
448 emit('GUILLOTINE', "W=#{width}", "H=#{height}", "mode=#{mode}", "rule=#{rule}", "rects=#{rectangles.length}",
449 "sheets=#{sheets.length}", "rotated=#{rotated_count}", "util=#{area * 100 / (sheets.length * width * height)}")
450 sheets.each_with_index do |sheet, i|
451 emit('SHEET', i + 1, "placed=#{sheet[:placements].length}", "area=#{sheet[:placements].sum { |p| p.w * p.h }}")
452 sheet[:placements].each do |p|
453 emit('PLACE', letter(p.id), "x=#{p.x}", "y=#{p.y}", "w=#{p.w}", "h=#{p.h}", *(p.rotated ? ['rotated'] : []))
454 end
455 emit('FREE', *(sheet[:free].empty? ? ['none'] : sheet[:free].map { |f| "#{f.x},#{f.y},#{f.w}x#{f.h}" }))
456 drawing(width, height, sheet[:placements])
457 end
458 end

460 def command(cmd, a)
461 if %w[ITEMS RANDOM].include?(cmd)
462 if cmd == 'ITEMS'
463 capacity = a[0].to_i
464 raise CommandError, 'BAD CAPACITY' if capacity.zero?
465 sizes = a.drop(1).map(&:to_i)
466 bad = sizes.index { |s| s.zero? || s > capacity }
467 raise CommandError, "BAD ITEM #{bad + 1}" if bad
468 else
469 count = a[0].to_i
470 raise CommandError, 'BAD COUNT' if count.zero? || count > 40
471 capacity = a[1].to_i
472 raise CommandError, 'BAD CAPACITY' if capacity.zero?
473 seed = a[2].to_i
474 raise CommandError, 'BAD SEED' if seed.zero? || seed >= 2147483647
475 sizes = Array.new(count) do
476 seed = 48271 * seed % 2147483647
477 seed % capacity + 1
478 end
479 end
480 @capacity = capacity
481 @items = sizes.each_with_index.map { |size, i| Item.new(i + 1, size) }
482 emit(cmd, "count=#{sizes.length}", "capacity=#{capacity}", "total=#{total(@items)}")
483 emit('SIZES', *sizes) if cmd == 'RANDOM'
484 return
485 end
486 if cmd == 'CUTSTOCK'
487 cutstock(a[0].to_i, a.drop(1))
488 return
489 end
490 if cmd == 'SHELF'
491 shelf(a[0].to_i, a.drop(1))
492 return
493 end
494 if cmd == 'GUILLOTINE'
495 guillotine(a[0].to_i, a[1].to_i, a[2], a[3], a.drop(4))
496 return
497 end
498 raise CommandError, 'NO ITEMS' if @capacity.nil?
499 if METHODS.include?(cmd) || %w[TRACE REDUCE].include?(cmd)
500 specification = %w[TRACE REDUCE].include?(cmd)
501 method = specification ? a[0] : cmd
502 k = (a[specification ? 1 : 0] || '0').to_i
503 raise CommandError, 'BAD PARAMETER' if %w[HARMONIC KBOUNDED].include?(method) && k.zero?
504 if cmd == 'REDUCE'
505 reduce(method, k)
506 return
507 end
508 result = pack(@items, @capacity, method, k)
509 emit((cmd == 'TRACE' ? 'TRACE ' : '') + header(method, k, result))
510 if cmd == 'TRACE'
511 result[:steps].each_with_index do |s, i|
512 parts = ['STEP', i + 1, "item=#{notation(s[:item])}", "bin=#{s[:bin]}", "inspected=#{s[:inspected]}"]
513 parts << 'new' if s[:fresh]
514 parts << "closed=#{s[:closed]}" if s[:closed]
515 parts << "class=#{s[:klass]}" if s[:klass]
516 emit(*parts)
517 end
518 else
519 result[:bins].each_with_index { |b, i| bin_line(b, i + 1, @capacity, method == 'HARMONIC') }
520 end
521 elsif cmd == 'BOUNDS'
522 l1, l2, rows = bounds(@items, @capacity)
523 emit('BOUNDS', "total=#{total(@items)}", "L1=#{l1}", "L2=#{l2}")
524 rows.each { |alpha, j1, j2, j3, bound| emit('ALPHA', alpha, "j1=#{j1}", "j2=#{j2}", "j3=#{j3}", "bound=#{bound}") }
525 elsif cmd == 'COMPARE'
526 k = a[0].to_i
527 raise CommandError, 'BAD PARAMETER' if k.zero?
528 l1, l2, = bounds(@items, @capacity)
529 emit('COMPARE', "k=#{k}", "L1=#{l1}", "L2=#{l2}")
530 rows = METHODS.map { |method| [method, pack(@items, @capacity, method, k)] }
531 rows.sort_by { |method, r| [r[:bins].length, method] }.each do |method, r|
532 emit(method, "bins=#{r[:bins].length}", "gap=#{r[:bins].length - l2}",
533 "waste=#{r[:bins].length * @capacity - total(@items)}", "inspections=#{r[:inspections]}")
534 end
535 elsif cmd == 'EXACT'
536 exact(a[0].to_i)
537 elsif cmd == 'CHECK'
538 labels = a.map(&:to_i)
539 raise CommandError, 'BAD ASSIGNMENT' if labels.length != @items.length || labels.any?(&:zero?)
540 bins = {}
541 @items.each_with_index do |item, i|
542 bin = (bins[labels[i]] ||= Bin.new(0, [], nil))
543 bin.load += item.size
544 bin.items << item
545 end
546 over = bins.values.count { |b| b.load > @capacity }
547 emit('CHECK', "bins=#{bins.length}", "over=#{over}", "L2=#{bounds(@items, @capacity)[1]}", "status=#{over.zero? ? 'valid' : 'invalid'}")
548 bins.keys.sort.each { |label| bin_line(bins[label], label, @capacity, false, bins[label].load > @capacity) }
549 elsif cmd == 'COVER'
550 cover
551 elsif cmd == 'MULTIFIT'
552 multifit(a[0].to_i)
553 end
554 end
