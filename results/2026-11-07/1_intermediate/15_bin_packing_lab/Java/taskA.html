85 private static Packing pack(List<Item> items, int capacity, String method, int parameter) {
86 Packing result = new Packing(method, parameter);
87 List<Item> order = method.equals("FFD") || method.equals("BFD") || method.equals("WFD")
88 ? descending(items) : items;
89 List<Integer> active = new ArrayList<>();
90 Map<Integer, Integer> current = new HashMap<>();
91 for (Item item : order) {
92 int before = result.inspections;
93 int chosen = -1;
94 int closed = -1;
95 int category = 0;
96 int size = item.size();
97
98 if (method.equals("NEXTFIT")) {
99 if (!result.bins.isEmpty()) {
100 result.inspections++;
101 int last = result.bins.size() - 1;
102 if (result.bins.get(last).load + size <= capacity) chosen = last;
103 }
104 } else if (method.equals("FIRSTFIT") || method.equals("FFD") || method.equals("KBOUNDED")) {
105 List<Integer> candidates = new ArrayList<>();
106 if (method.equals("KBOUNDED")) candidates.addAll(active);
107 else for (int i = 0; i < result.bins.size(); i++) candidates.add(i);
108 for (int i : candidates) {
109 result.inspections++;
110 if (result.bins.get(i).load + size <= capacity) {
111 chosen = i;
112 break;
113 }
114 }
115 } else if (method.equals("HARMONIC")) {
116 category = Math.min(capacity / size, parameter);
117 Integer index = current.get(category);
118 if (index != null) {
119 result.inspections++;
120 if (result.bins.get(index).load + size <= capacity) chosen = index;
121 }
122 } else {
123 List<Integer> fitting = new ArrayList<>();
124 result.inspections += result.bins.size();
125 for (int i = 0; i < result.bins.size(); i++) {
126 if (result.bins.get(i).load + size <= capacity) fitting.add(i);
127 }
128 if (!fitting.isEmpty()) {
129 if (method.equals("BESTFIT") || method.equals("BFD")) {
130 chosen = fitting.get(0);
131 for (int i : fitting) {
132 if (result.bins.get(i).load > result.bins.get(chosen).load) chosen = i;
133 }
134 } else {
135 fitting.sort(Comparator.comparingInt((Integer i) -> result.bins.get(i).load)
136 .thenComparingInt(i -> i));
137 chosen = method.equals("ALMOSTWORST") && fitting.size() > 1
138 ? fitting.get(1) : fitting.get(0);
139 }
140 }
141 }
142 boolean opened = chosen < 0;
143 if (opened) {
144 if (method.equals("KBOUNDED") && active.size() == parameter) {
145 closed = active.get(0);
146 for (int i : active) {
147 if (result.bins.get(i).load > result.bins.get(closed).load) closed = i;
148 }
149 active.remove(Integer.valueOf(closed));
150 result.closes++;
151 }
152 chosen = result.bins.size();
153 result.bins.add(new Bin(category));
154 if (method.equals("KBOUNDED")) active.add(chosen);
155 if (method.equals("HARMONIC")) current.put(category, chosen);
156 }
157 result.bins.get(chosen).put(item);
158 result.steps.add(new Step(item, chosen, result.inspections - before, opened, closed, category));
159 }
160 return result;
161 }

163 private record BoundRow(int alpha, int j1, int j2, int j3, int bound) {}
164 private record Bounds(long l1, int l2, List<BoundRow> rows) {}

166 private static Bounds bounds(List<Item> items, int capacity) {
167 long total = 0;
168 Set<Integer> thresholds = new TreeSet<>();
169 thresholds.add(0);
170 for (Item item : items) {
171 total += item.size();
172 if (item.size() * 2 <= capacity) thresholds.add(item.size());
173 }
174 List<BoundRow> rows = new ArrayList<>();
175 int lower = 0;
176 for (int alpha : thresholds) {
177 int j1 = 0, j2 = 0, j3 = 0;
178 long sum2 = 0, sum3 = 0;
179 for (Item item : items) {
180 int s = item.size();
181 if (s > capacity - alpha) j1++;
182 else if (2 * s > capacity) { j2++; sum2 += s; }
183 else if (s >= alpha) { j3++; sum3 += s; }
184 }
185 long excess = sum3 - (j2 * (long) capacity - sum2);
186 int bound = j1 + j2 + (excess > 0 ? (int) ceil(excess, capacity) : 0);
187 lower = Math.max(lower, bound);
188 rows.add(new BoundRow(alpha, j1, j2, j3, bound));
189 }
190 return new Bounds(ceil(total, capacity), lower, rows);
191 }

193 private static class ExactSearch {
194 final List<Item> order;
195 final int capacity, limit, lower;
196 final long[] remaining;
197 final List<Bin> live = new ArrayList<>();
198 List<Bin> best;
199 int nodes;
200 String status;
201 final Lab lab;
202
203 ExactSearch(List<Item> items, int capacity, int limit, int lower, List<Bin> incumbent, Lab lab) {
204 this.order = descending(items);
205 this.capacity = capacity;
206 this.limit = limit;
207 this.lower = lower;
208 this.best = copied(incumbent);
209 this.lab = lab;
210 remaining = new long[order.size() + 1];
211 for (int i = order.size() - 1; i >= 0; i--) {
212 remaining[i] = remaining[i + 1] + order.get(i).size();
213 }
214 }
215
216 void visit(int index) {
217 if (nodes == limit) { status = "limit"; return; }
218 nodes++;
219 int count = live.size();
220 long used = remaining[0] - remaining[index];
221 long free = count * (long) capacity - used;
222 long excess = remaining[index] - free;
223 long minimum = count + (excess > 0 ? ceil(excess, capacity) : 0);
224 if (minimum >= best.size()) return;
225 if (index == order.size()) {
226 best = copied(live);
227 lab.emit("IMPROVE bins=" + count + " nodes=" + nodes);
228 if (count == lower) status = "bound";
229 return;
230 }
231 Item item = order.get(index);
232 Set<Integer> seen = new HashSet<>();
233 for (Bin bin : live) {
234 int old = bin.load;
235 if (!seen.add(old) || old + item.size() > capacity) continue;
236 bin.put(item);
237 visit(index + 1);
238 bin.items.remove(bin.items.size() - 1);
239 bin.load = old;
240 if (status != null) return;
241 }
242 if (count + 1 < best.size()) {
243 Bin bin = new Bin();
244 bin.put(item);
245 live.add(bin);
246 visit(index + 1);
247 live.remove(live.size() - 1);
248 }
249 }
250 }

567 void cmdCutstock(String[] args) {

569 if (length == 0) fail("BAD CAPACITY");
570 List<Piece> pieces = new ArrayList<>();
571 Set<Integer> seen = new HashSet<>();
572 for (int i = 2; i < args.length; i++) {
573 String[] pair = args[i].split(":");
574 int size = Integer.parseInt(pair[0]), demand = Integer.parseInt(pair[1]);
575 if (size == 0 || size > length || demand == 0) fail("BAD PIECE " + (i - 1));
576 if (!seen.add(size)) fail("DUPLICATE PIECE " + (i - 1));
577 pieces.add(new Piece(size, demand));
578 }
579 pieces.sort(Comparator.comparingInt(Piece::length).reversed());
580 List<Pattern> patterns = new ArrayList<>();
581 patterns(pieces, length, 0, length, new int[pieces.size()], patterns);
582 emit("CUTSTOCK L=" + length + " pieces=" + pieces.size() + " patterns=" + patterns.size());
583 for (int p = 0; p < patterns.size(); p++) {
584 Pattern pattern = patterns.get(p);
585 StringBuilder breakdown = new StringBuilder();
586 for (int i = 0; i < pieces.size(); i++) {
587 int count = pattern.counts[i];
588 if (count == 0) continue;
589 if (breakdown.length() > 0) breakdown.append('+');
590 breakdown.append(count).append('x').append(pieces.get(i).length());
591 }
592 emit("PATTERN " + (p + 1) + " " + breakdown + " rest=" + pattern.trim);
593 }
594 int[] remaining = new int[pieces.size()];
595 int[] cut = new int[pieces.size()];
596 int[] uses = new int[patterns.size()];
597 for (int i = 0; i < pieces.size(); i++) remaining[i] = pieces.get(i).demand();
598 while (true) {
599 int needed = 0;
600 for (int r : remaining) needed += r;
601 if (needed == 0) break;
602 int best = -1, useful = -1;
603 for (int p = 0; p < patterns.size(); p++) {
604 int score = 0;
605 for (int i = 0; i < pieces.size(); i++)
606 score += Math.min(patterns.get(p).counts[i], remaining[i]);
607 if (score > useful) { useful = score; best = p; }
608 }
609 uses[best]++;
610 for (int i = 0; i < pieces.size(); i++) {
611 int amount = patterns.get(best).counts[i];
612 cut[i] += amount;
613 remaining[i] = Math.max(0, remaining[i] - amount);
614 }
615 }
616 long quantity = 0, trim = 0, demandCount = 0, cutCount = 0, bars = 0;
617 List<Item> stock = new ArrayList<>();
618 for (int i = 0; i < patterns.size(); i++) {
619 if (uses[i] > 0) {
620 emit("USE " + (i + 1) + " bars=" + uses[i]);
621 bars += uses[i];
622 trim += uses[i] * (long) patterns.get(i).trim;
623 }
624 }
625 for (Piece piece : pieces) {
626 quantity += piece.demand() * (long) piece.length();
627 demandCount += piece.demand();
628 for (int j = 0; j < piece.demand(); j++)
629 stock.add(new Item(stock.size() + 1, piece.length()));
630 }
631 for (int i = 0; i < pieces.size(); i++) {
632 emit("PIECE " + pieces.get(i).length() + " demand=" + pieces.get(i).demand()
633 + " cut=" + cut[i]);
634 cutCount += cut[i];
635 }
636 emit("TOTAL bars=" + bars + " trim=" + trim + " surplus=" + (cutCount - demandCount)
637 + " waste=" + (bars * length - quantity));
638 int ffd = pack(stock, length, "FIRSTFIT", 0).bins.size();
639 emit("FFD bars=" + ffd + " waste=" + (ffd * (long) length - quantity));
640 emit("LOWER bars=" + ceil(quantity, length));
641 }
642 List<Rect> rectangles(String[] args, int start, int width, int height, boolean rotate) {
643 List<Rect> result = new ArrayList<>();
644 for (int i = start; i < args.length; i++) {
645 String[] parts = args[i].split("x");
646 int w = Integer.parseInt(parts[0]), h = Integer.parseInt(parts[1]);
647 boolean fits = height == 0 ? w <= width
648 : (w <= width && h <= height) || (rotate && h <= width && w <= height);
649 if (w == 0 || h == 0 || !fits) fail("BAD RECT " + (i - start + 1));
650 result.add(new Rect(i - start + 1, w, h));
651 }
652 return result;
653 }
654 List<String> drawing(int width, int height, List<Placement> placements) {
655 char[][] cells = new char[height][width];
656 for (char[] row : cells) Arrays.fill(row, '.');
657 for (Placement placement : placements) {
658 for (int y = placement.y; y < placement.y + placement.height; y++)
659 for (int x = placement.x; x < placement.x + placement.width; x++)
660 cells[y][x] = placement.rect.letter();
661 }
662 List<String> lines = new ArrayList<>();
663 for (int y = height - 1; y >= 0; y--) lines.add(new String(cells[y]));
664 return lines;

669 void cmdShelf(String[] args) {
670 int width = Integer.parseInt(args[1]);
671 if (width == 0) fail("BAD CAPACITY");
672 List<Rect> source = rectangles(args, 2, width, 0, false);
673 long area = 0;
674 for (Rect rect : source) area += rect.area();
675 emit("SHELF W=" + width + " rects=" + source.size() + " area=" + area);
676 List<Rect> order = new ArrayList<>(source);
677 order.sort(Comparator.comparingInt(Rect::height).reversed()
678 .thenComparing(Comparator.comparingInt(Rect::width).reversed())
679 .thenComparingInt(Rect::number));
680 for (String method : new String[]{"NFDH", "FFDH", "BFDH"}) {
681 List<Shelf> shelves = new ArrayList<>();
682 List<Placement> placements = new ArrayList<>();
683 for (Rect rect : order) {
684 Shelf selected = null;
685 if (method.equals("NFDH")) {
686 if (!shelves.isEmpty()) {
687 Shelf last = shelves.get(shelves.size() - 1);
688 if (last.used + rect.width() <= width) selected = last;
689 }
690 } else {
691 for (Shelf shelf : shelves) {
692 if (shelf.used + rect.width() > width) continue;
693 if (selected == null || (method.equals("BFDH") && shelf.used > selected.used))
694 selected = shelf;
695 if (method.equals("FFDH")) break;
696 }
697 }
698 if (selected == null) {
699 int bottom = shelves.isEmpty() ? 0
700 : shelves.get(shelves.size() - 1).bottom + shelves.get(shelves.size() - 1).height;
701 selected = new Shelf(bottom, rect.height());
702 shelves.add(selected);
703 }
704 Placement placement = new Placement(rect, selected.used, selected.bottom,
705 rect.width(), rect.height(), false);
706 selected.placements.add(placement);
707 placements.add(placement);
708 selected.used += rect.width();
709 }
710 int height = shelves.get(shelves.size() - 1).bottom + shelves.get(shelves.size() - 1).height;
711 emit(method + " height=" + height + " shelves=" + shelves.size()
712 + " util=" + (area * 100 / (width * (long) height)));
713 for (int i = 0; i < shelves.size(); i++) {
714 Shelf shelf = shelves.get(i);
715 StringBuilder row = new StringBuilder("LEVEL " + (i + 1) + " y=" + shelf.bottom
716 + " h=" + shelf.height + " used=" + shelf.used + " items=");
717 for (int j = 0; j < shelf.placements.size(); j++) {
718 if (j > 0) row.append(' ');
719 Placement p = shelf.placements.get(j);
720 row.append(p.rect.letter()).append('@').append(p.x);
721 }
722 emit(row.toString());
723 }
724 draw(width, height, placements);
725 }
726 int[] columns = new int[width];
727 List<Placement> placements = new ArrayList<>();
728 for (Rect rect : order) {
729 int xBest = 0, yBest = Integer.MAX_VALUE;
730 for (int x = 0; x <= width - rect.width(); x++) {
731 int y = 0;
732 for (int j = x; j < x + rect.width(); j++) y = Math.max(y, columns[j]);
733 if (y < yBest) { xBest = x; yBest = y; }
734 }
735 Placement p = new Placement(rect, xBest, yBest, rect.width(), rect.height(), false);
736 placements.add(p);
737 for (int j = xBest; j < xBest + rect.width(); j++) columns[j] = yBest + rect.height();
738 }
739 int height = 0;
740 for (int column : columns) height = Math.max(height, column);
741 emit("SKYLINE height=" + height + " util=" + (area * 100 / (width * (long) height)));
742 for (Placement p : placements) {
743 emit("PLACE " + p.rect.letter() + " x=" + p.x + " y=" + p.y
744 + " w=" + p.width + " h=" + p.height);
745 }
746 draw(width, height, placements);
747 }
748 boolean fits(Rect rect, FreeRect free, boolean rotate) {
749 return (rect.width() <= free.width && rect.height() <= free.height)
750 || (rotate && rect.height() <= free.width && rect.width() <= free.height);
751 }
752 void cmdGuillotine(String[] args) {
753 int width = Integer.parseInt(args[1]), height = Integer.parseInt(args[2]);
754 if (width == 0 || height == 0) fail("BAD CAPACITY");
755 boolean rotate = args[3].equals("ROTATE");
756 boolean best = args[4].equals("BEST");
757 List<Rect> order = rectangles(args, 5, width, height, rotate);
758 order.sort(Comparator.comparingLong(Rect::area).reversed().thenComparingInt(Rect::number));
759 long totalArea = 0;
760 for (Rect rect : order) totalArea += rect.area();
761 List<Sheet> sheets = new ArrayList<>();
762 int rotated = 0;
763 for (Rect rect : order) {
764 Sheet selected = null;
765 int selectedIndex = -1, selectedArea = Integer.MAX_VALUE;
766 for (Sheet sheet : sheets) {
767 for (int i = 0; i < sheet.free.size(); i++) {
768 FreeRect free = sheet.free.get(i);
769 if (!fits(rect, free, rotate)) continue;
770 if (selected == null || (best && free.area() < selectedArea)) {
771 selected = sheet;
772 selectedIndex = i;
773 selectedArea = free.area();
774 }
775 if (!best) break;
776 }
777 if (selected != null && !best) break;
778 }
779 if (selected == null) {
780 selected = new Sheet(width, height);
781 sheets.add(selected);
782 selectedIndex = 0;
783 }
784 FreeRect free = selected.free.remove(selectedIndex);
785 boolean turned = rect.width() > free.width || rect.height() > free.height;
786 int w = turned ? rect.height() : rect.width();
787 int h = turned ? rect.width() : rect.height();
788 if (turned) rotated++;
789 Placement placed = new Placement(rect, free.x, free.y, w, h, turned);
790 selected.placed.add(placed);
791 selected.area += rect.area();
792 int rw = free.width - w, rh = free.height - h;
793 FreeRect top = rw <= rh
794 ? new FreeRect(free.x, free.y + h, free.width, rh)
795 : new FreeRect(free.x, free.y + h, w, rh);
796 FreeRect right = rw <= rh
797 ? new FreeRect(free.x + w, free.y, rw, h)
798 : new FreeRect(free.x + w, free.y, rw, free.height);
799 if (top.width > 0 && top.height > 0) selected.free.add(top);
800 if (right.width > 0 && right.height > 0) selected.free.add(right);
801 }
802 int count = sheets.size();
803 emit("GUILLOTINE W=" + width + " H=" + height + " mode=" + args[3] + " rule=" + args[4]
804 + " rects=" + order.size() + " sheets=" + count + " rotated=" + rotated
805 + " util=" + (totalArea * 100 / (count * (long) width * height)));
806 for (int i = 0; i < sheets.size(); i++) {
807 Sheet sheet = sheets.get(i);
808 emit("SHEET " + (i + 1) + " placed=" + sheet.placed.size() + " area=" + sheet.area);
809 for (Placement p : sheet.placed) {
810 emit("PLACE " + p.rect.letter() + " x=" + p.x + " y=" + p.y
811 + " w=" + p.width + " h=" + p.height + (p.rotated ? " rotated" : ""));
812 }
813 StringBuilder free = new StringBuilder("FREE");
814 if (sheet.free.isEmpty()) free.append(" none");
815 else for (FreeRect item : sheet.free) free.append(' ').append(item.text());
816 emit(free.toString());
817 draw(width, height, sheet.placed);
818 }

820 void process(String[] args) {
821 String command = args[0];
822 try {
823 switch (command) {
824 case "ITEMS" -> cmdItems(args);
825 case "RANDOM" -> cmdRandom(args);
826 case "CUTSTOCK" -> cmdCutstock(args);
827 case "SHELF" -> cmdShelf(args);
828 case "GUILLOTINE" -> cmdGuillotine(args);
829 default -> {
830 if (items == null) fail("NO ITEMS");
831 switch (command) {
832 case "NEXTFIT", "FIRSTFIT", "BESTFIT", "WORSTFIT", "ALMOSTWORST",
833 "HARMONIC", "KBOUNDED", "FFD", "BFD", "WFD" ->
834 cmdMethod(command, parameter(command, args, 1));
835 case "TRACE" -> cmdTrace(args[1], parameter(args[1], args, 2));
836 case "REDUCE" -> cmdReduce(args[1], parameter(args[1], args, 2));
837 case "BOUNDS" -> cmdBounds();
838 case "COMPARE" -> cmdCompare(Integer.parseInt(args[1]));
839 case "EXACT" -> cmdExact(Integer.parseInt(args[1]));
840 case "CHECK" -> cmdCheck(args);
841 case "COVER" -> cmdCover();
842 case "MULTIFIT" -> cmdMultifit(Integer.parseInt(args[1]));
843 default -> throw new IllegalArgumentException("Unexpected command " + command);
844 }
845 }
846 }
847 } catch (CommandError error) {
848 emit("ERROR " + error.getMessage());
849 }
850 }
