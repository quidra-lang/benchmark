4const METHODS = ["NEXTFIT", "FIRSTFIT", "BESTFIT", "WORSTFIT", "ALMOSTWORST", "HARMONIC", "KBOUNDED", "FFD", "BFD", "WFD"];
5const total = (items) => items.reduce((v, i) => v + i.size, 0);
6const descending = (items) => items.toSorted((a, b) => b.size - a.size || a.id - b.id);
7const notation = (i) => i.id + ":" + i.size;
8const letter = (id) => String.fromCharCode(64 + id);
9const ceil = (a, b) => Math.floor((a + b - 1) / b);
10const copyBins = (bins) => bins.map(b => ({load: b.load, items: [...b.items], cls: b.cls}));

13function pack(items, capacity, method, k = 0) {
14 const bins = [];
15 const steps = [];
16 const current = new Map();
17 let active = [];
18 let inspections = 0;
19 let closes = 0;
20 const order = ["FFD", "BFD", "WFD"].includes(method) ? descending(items) : items;
21 for (const item of order) {
22 const before = inspections;
23 let destination = -1;
24 let closed = null;
25 let cls = null;
26 if (method === "NEXTFIT") {
27 if (bins.length) {
28 inspections++;
29 if (bins[bins.length - 1].load + item.size <= capacity) destination = bins.length - 1;
30 }
31 } else if (method === "HARMONIC") {
32 cls = Math.min(Math.floor(capacity / item.size), k);
33 const index = current.get(cls);
34 if (index !== undefined) {
35 inspections++;
36 if (bins[index].load + item.size <= capacity) destination = index;
37 }
38 } else if (["FIRSTFIT", "FFD", "KBOUNDED"].includes(method)) {
39 const candidates = method === "KBOUNDED" ? active : bins.map((_, i) => i);
40 for (const index of candidates) {
41 inspections++;
42 if (bins[index].load + item.size <= capacity) {
43 destination = index;
44 break;
45 }
46 }
47 } else {
48 inspections += bins.length;
49 const candidates = bins.map((_, i) => i).filter(i => bins[i].load + item.size <= capacity);
50 const largest = method === "BESTFIT" || method === "BFD";
51 candidates.sort((a, b) => (largest ? bins[b].load - bins[a].load : bins[a].load - bins[b].load) || a - b);
52 if (candidates.length) destination = candidates[method === "ALMOSTWORST" && candidates.length > 1 ? 1 : 0];
53 }
54 const fresh = destination < 0;
55 if (fresh) {
56 if (method === "KBOUNDED" && active.length === k) {
57 const index = active.toSorted((a, b) => bins[b].load - bins[a].load || a - b)[0];
58 active = active.filter(i => i !== index);
59 closed = index + 1;
60 closes++;
61 }
62 destination = bins.length;
63 bins.push({load: 0, items: [], cls});
64 if (method === "HARMONIC") current.set(cls, destination);
65 if (method === "KBOUNDED") active.push(destination);
66 }
67 bins[destination].items.push(item);
68 bins[destination].load += item.size;
69 steps.push({item, bin: destination + 1, inspected: inspections - before, fresh, closed, cls});
70 }
71 return {bins, steps, inspections, closes};
72}

74function bounds(items, capacity) {
75 const thresholds = [...new Set([0, ...items.filter(i => 2 * i.size <= capacity).map(i => i.size)])].sort((a, b) => a - b);
76 const rows = thresholds.map(alpha => {
77 const j1 = items.filter(i => i.size > capacity - alpha);
78 const j2 = items.filter(i => 2 * i.size > capacity && i.size <= capacity - alpha);
79 const j3 = items.filter(i => 2 * i.size <= capacity && i.size >= alpha);
80 const extra = total(j3) - (j2.length * capacity - total(j2));
81 return {alpha, j1: j1.length, j2: j2.length, j3: j3.length, bound: j1.length + j2.length + (extra > 0 ? ceil(extra, capacity) : 0)};
82 });
83 return {l1: ceil(total(items), capacity), l2: Math.max(...rows.map(r => r.bound)), rows};
84}

106 reduce(method, k) {
107 const capacity = this.capacity;
108 const bins = pack(this.items, capacity, method, k).bins;
109 const before = bins.length;
110 const eliminations = [];
111 while (true) {
112 let succeeded = false;
113 const order = bins.map((_, i) => i).sort((a, b) => bins[a].load - bins[b].load || a - b);
114 for (const source of order) {
115 const loads = bins.map(b => b.load);
116 const moves = [];
117 let possible = true;
118 for (const item of descending(bins[source].items)) {
119 const candidates = bins.map((_, i) => i).filter(i => i !== source && loads[i] + item.size <= capacity);
120 candidates.sort((a, b) => loads[b] - loads[a] || a - b);
121 if (!candidates.length) {
122 possible = false;
123 break;
124 }
125 const destination = candidates[0];
126 loads[destination] += item.size;
127 moves.push({item, destination});
128 }
129 if (!possible) continue;
130 eliminations.push("ELIMINATE bin=" + (source + 1) + " load=" + bins[source].load +
131 " moves=" + moves.map(m => notation(m.item) + "->" + (m.destination + 1)).join(" "));
132 for (const m of moves) {
133 bins[m.destination].items.push(m.item);
134 bins[m.destination].load += m.item.size;
135 }
136 bins.splice(source, 1);
137 succeeded = true;
138 break;
139 }
140 if (!succeeded) break;
141 }
142 this.emit("REDUCE", method, ...(["HARMONIC", "KBOUNDED"].includes(method) ? ["k=" + k] : []),
143 "before=" + before, "after=" + bins.length, "eliminated=" + (before - bins.length));
144 for (const line of eliminations) this.emit(line);
145 bins.forEach((b, i) => this.binLine(b, i + 1, capacity, method === "HARMONIC"));
146 }

148 exact(limit) {
149 if (limit === 0) throw new CommandError("BAD PARAMETER");
150 if (this.items.length > 20) throw new CommandError("TOO MANY ITEMS");
151 const capacity = this.capacity;
152 const lower = bounds(this.items, capacity).l2;
153 let best = copyBins(pack(this.items, capacity, "FFD").bins);
154 this.emit("EXACT", "ffd=" + best.length, "L2=" + lower);
155 const order = descending(this.items);
156 const remaining = Array(order.length + 1).fill(0);
157 for (let i = order.length - 1; i >= 0; i--) remaining[i] = remaining[i + 1] + order[i].size;
158 const bins = [];
159 let nodes = 0;
160 let status = best.length === lower ? "bound" : "complete";
161 let stopped = status === "bound";
162 const visit = (index, used) => {
163 if (nodes === limit) {
164 status = "limit";
165 stopped = true;
166 return;
167 }
168 nodes++;
169 const free = bins.length * capacity - used;
170 const extra = remaining[index] - free;
171 const bound = bins.length + (extra > 0 ? ceil(extra, capacity) : 0);
172 if (bound >= best.length) return;
173 if (index === order.length) {
174 best = copyBins(bins);
175 this.emit("IMPROVE", "bins=" + best.length, "nodes=" + nodes);
176 if (best.length === lower) {
177 status = "bound";
178 stopped = true;
179 }
180 return;
181 }
182 const item = order[index];
183 const seen = new Set();
184 for (const bin of bins) {
185 if (seen.has(bin.load)) continue;
186 seen.add(bin.load);
187 if (bin.load + item.size > capacity) continue;
188 bin.load += item.size;
189 bin.items.push(item);
190 visit(index + 1, used + item.size);
191 bin.items.pop();
192 bin.load -= item.size;
193 if (stopped) return;
194 }
195 if (bins.length + 1 < best.length) {
196 bins.push({load: item.size, items: [item], cls: null});
197 visit(index + 1, used + item.size);
198 bins.pop();
199 }
200 };
201 if (!stopped) visit(0, 0);
202 this.emit("RESULT", "bins=" + best.length, "nodes=" + nodes, "status=" + status);
203 best.forEach((b, i) => this.binLine(b, i + 1, capacity));
204 }

206 cover() {
207 const capacity = this.capacity;
208 this.emit("COVER", "total=" + total(this.items), "upper=" + Math.floor(total(this.items) / capacity));
209 for (const method of ["DNF", "BIGSMALL"]) {
210 const remaining = method === "DNF" ? [...this.items] : descending(this.items);
211 const bins = [];
212 while (remaining.length) {
213 const first = remaining.shift();
214 const bin = {load: first.size, items: [first], cls: null};
215 while (bin.load < capacity && remaining.length) {
216 const item = method === "DNF" ? remaining.shift() : remaining.pop();
217 bin.items.push(item);
218 bin.load += item.size;
219 }
220 bins.push(bin);
221 }
222 const covered = bins.filter(b => b.load >= capacity);
223 const rest = bins.find(b => b.load < capacity);
224 this.emit(method, "covered=" + covered.length, "rest=" + (rest?.load ?? 0));
225 covered.forEach((b, i) => this.binLine(b, i + 1, capacity));
226 if (rest) this.emit("REST", "items=" + rest.items.map(notation).join(" "));
227 }

237 multifit(count) {
238 if (count === 0) throw new CommandError("BAD PARAMETER");
239 const size = total(this.items);
240 const largest = Math.max(...this.items.map(i => i.size));
241 let lo = Math.max(ceil(size, count), largest);
242 let hi = Math.max(ceil(2 * size, count), largest);
243 this.emit("MULTIFIT", "m=" + count, "lower=" + lo, "upper=" + hi);
244 while (lo < hi) {
245 const mid = Math.floor((lo + hi) / 2);
246 const bins = pack(this.items, mid, "FFD").bins;
247 const ok = bins.length <= count;
248 this.emit("TRY", "cap=" + mid, "bins=" + bins.length, ok ? "ok" : "fail");
249 if (ok) hi = mid;
250 else lo = mid + 1;
251 }
252 const bins = pack(this.items, hi, "FFD").bins;
253 this.emit("RESULT", "cap=" + hi, "makespan=" + Math.max(...bins.map(b => b.load)));
254 this.machines(bins, count);
255 const lpt = Array.from({length: count}, () => ({load: 0, items: [], cls: null}));
256 for (const item of descending(this.items)) {
257 let chosen = 0;
258 for (let i = 1; i < count; i++) {
259 if (lpt[i].load < lpt[chosen].load) chosen = i;

271 cutstock(capacity, tokens) {
272 if (capacity === 0) throw new CommandError("BAD CAPACITY");
273 const pieces = tokens.map(t => t.split(":").map(Number));
274 const seen = new Set();
275 for (let i = 0; i < pieces.length; i++) {
276 const [length, demand] = pieces[i];
277 if (length === 0 || length > capacity || demand === 0) throw new CommandError("BAD PIECE " + (i + 1));
278 if (seen.has(length)) throw new CommandError("DUPLICATE PIECE " + (i + 1));
279 seen.add(length);
280 }
281 pieces.sort((a, b) => b[0] - a[0]);
282 const patterns = [];
283 const enumerate = (index, rest, prefix) => {
284 const maximum = Math.floor(rest / pieces[index][0]);
285 if (index === pieces.length - 1) {
286 patterns.push([...prefix, maximum]);
287 return;
288 }
289 for (let count = maximum; count >= 0; count--) enumerate(index + 1, rest - count * pieces[index][0], [...prefix, count]);
290 };
291 enumerate(0, capacity, []);
292 const remaining = pieces.map(p => p[1]);
293 const used = patterns.map(() => 0);
294 while (remaining.some(d => d > 0)) {
295 let chosen = 0;
296 let best = -1;
297 for (let p = 0; p < patterns.length; p++) {
298 const useful = patterns[p].reduce((v, a, i) => v + Math.min(a, remaining[i]), 0);
299 if (useful > best) {
300 best = useful;
301 chosen = p;
302 }
303 }
304 used[chosen]++;
305 for (let i = 0; i < pieces.length; i++) remaining[i] = Math.max(0, remaining[i] - patterns[chosen][i]);
306 }
307 const trims = patterns.map(p => capacity - p.reduce((v, a, i) => v + a * pieces[i][0], 0));
308 const cut = pieces.map((_, i) => patterns.reduce((v, p, j) => v + p[i] * used[j], 0));
309 const bars = used.reduce((a, b) => a + b, 0);
310 const size = pieces.reduce((v, p) => v + p[0] * p[1], 0);
311 this.emit("CUTSTOCK", "L=" + capacity, "pieces=" + pieces.length, "patterns=" + patterns.length);
312 patterns.forEach((p, i) => this.emit("PATTERN", i + 1, p.flatMap((a, j) => a > 0 ? [a + "x" + pieces[j][0]] : []).join("+"), "rest=" + trims[i]));
313 used.forEach((n, i) => {
314 if (n) this.emit("USE", i + 1, "bars=" + n);
315 });
316 pieces.forEach((p, i) => this.emit("PIECE", p[0], "demand=" + p[1], "cut=" + cut[i]));
317 this.emit("TOTAL", "bars=" + bars, "trim=" + used.reduce((v, n, i) => v + n * trims[i], 0),
318 "surplus=" + cut.reduce((v, n, i) => v + n - pieces[i][1], 0), "waste=" + (bars * capacity - size));
319 const items = [];
320 for (const [length, demand] of pieces) {
321 for (let i = 0; i < demand; i++) items.push({id: items.length + 1, size: length});
322 }
323 const ffd = pack(items, capacity, "FIRSTFIT").bins.length;
324 this.emit("FFD", "bars=" + ffd, "waste=" + (ffd * capacity - size));
325 this.emit("LOWER", "bars=" + ceil(size, capacity));
326 }

345 shelf(width, tokens) {
346 if (width === 0) throw new CommandError("BAD CAPACITY");
347 const rectangles = this.rectangles(tokens);
348 for (const r of rectangles) {
349 if (r.w === 0 || r.h === 0 || r.w > width) throw new CommandError("BAD RECT " + r.id);
350 }
351 const order = rectangles.toSorted((a, b) => b.h - a.h || b.w - a.w || a.id - b.id);
352 const area = rectangles.reduce((v, r) => v + r.w * r.h, 0);
353 this.emit("SHELF", "W=" + width, "rects=" + rectangles.length, "area=" + area);
354 for (const method of ["NFDH", "FFDH", "BFDH"]) {
355 const shelves = [];
356 for (const r of order) {
357 let candidates = shelves.map((_, i) => i).filter(i => shelves[i].used + r.w <= width);
358 if (method === "NFDH") candidates = candidates.filter(i => i === shelves.length - 1);
359 if (method === "BFDH") candidates.sort((a, b) => shelves[b].used - shelves[a].used || a - b);
360 let index = candidates[0];
361 if (index === undefined) {
362 index = shelves.length;
363 const last = shelves[shelves.length - 1];
364 shelves.push({y: last ? last.y + last.h : 0, h: r.h, used: 0, placements: []});
365 }
366 const shelf = shelves[index];
367 shelf.placements.push({...r, x: shelf.used, y: shelf.y, rotated: false});
368 shelf.used += r.w;
369 }
370 const last = shelves[shelves.length - 1];
371 const height = last.y + last.h;
372 this.emit(method, "height=" + height, "shelves=" + shelves.length, "util=" + Math.floor(area * 100 / (width * height)));
373 shelves.forEach((s, i) => this.emit("LEVEL", i + 1, "y=" + s.y, "h=" + s.h, "used=" + s.used,
374 "items=" + s.placements.map(p => letter(p.id) + "@" + p.x).join(" ")));
375 this.drawing(width, height, shelves.flatMap(s => s.placements));
376 }
377 const heights = Array(width).fill(0);
378 const placements = [];
379 for (const r of order) {
380 let x = 0;
381 let y = Infinity;
382 for (let candidate = 0; candidate + r.w <= width; candidate++) {
383 const level = Math.max(...heights.slice(candidate, candidate + r.w));
384 if (level < y) {
385 x = candidate;
386 y = level;
387 }
388 }
389 placements.push({...r, x, y, rotated: false});
390 heights.fill(y + r.h, x, x + r.w);
391 }
392 const height = Math.max(...heights);
393 this.emit("SKYLINE", "height=" + height, "util=" + Math.floor(area * 100 / (width * height)));
394 for (const p of placements) this.emit("PLACE", letter(p.id), "x=" + p.x, "y=" + p.y, "w=" + p.w, "h=" + p.h);
395 this.drawing(width, height, placements);
396 }

398 guillotine(width, height, mode, rule, tokens) {
399 if (width === 0 || height === 0) throw new CommandError("BAD CAPACITY");
400 const rectangles = this.rectangles(tokens);
401 const fits = (r, f) => (r.w <= f.w && r.h <= f.h) || (mode === "ROTATE" && r.h <= f.w && r.w <= f.h);
402 for (const r of rectangles) {
403 if (r.w === 0 || r.h === 0 || !fits(r, {x: 0, y: 0, w: width, h: height})) throw new CommandError("BAD RECT " + r.id);
404 }
405 const sheets = [];
406 let rotatedCount = 0;
407 for (const r of rectangles.toSorted((a, b) => b.w * b.h - a.w * a.h || a.id - b.id)) {
408 let chosenSheet = -1;
409 let chosenFree = -1;
410 let bestArea = Infinity;
411 for (let i = 0; i < sheets.length; i++) {
412 for (let j = 0; j < sheets[i].free.length; j++) {
413 const f = sheets[i].free[j];
414 if (!fits(r, f)) continue;
415 if (chosenSheet < 0 || (rule === "BEST" && f.w * f.h < bestArea)) {
416 chosenSheet = i;
417 chosenFree = j;
418 bestArea = f.w * f.h;
419 }
420 }
421 }
422 if (chosenSheet < 0) {
423 chosenSheet = sheets.length;
424 chosenFree = 0;
425 sheets.push({free: [{x: 0, y: 0, w: width, h: height}], placements: []});
426 }
427 const sheet = sheets[chosenSheet];
428 const [f] = sheet.free.splice(chosenFree, 1);
429 const rotated = !(r.w <= f.w && r.h <= f.h);
430 const w = rotated ? r.h : r.w;
431 const h = rotated ? r.w : r.h;
432 if (rotated) rotatedCount++;
433 sheet.placements.push({id: r.id, w, h, x: f.x, y: f.y, rotated});
434 const rw = f.w - w;
435 const rh = f.h - h;
436 const top = {x: f.x, y: f.y + h, w: rw <= rh ? f.w : w, h: rh};
437 const right = {x: f.x + w, y: f.y, w: rw, h: rw <= rh ? h : f.h};
438 for (const free of [top, right]) {
439 if (free.w > 0 && free.h > 0) sheet.free.push(free);
440 }
441 }
442 const area = rectangles.reduce((v, r) => v + r.w * r.h, 0);
443 this.emit("GUILLOTINE", "W=" + width, "H=" + height, "mode=" + mode, "rule=" + rule, "rects=" + rectangles.length,
444 "sheets=" + sheets.length, "rotated=" + rotatedCount, "util=" + Math.floor(area * 100 / (sheets.length * width * height)));
445 sheets.forEach((sheet, i) => {
446 this.emit("SHEET", i + 1, "placed=" + sheet.placements.length, "area=" + sheet.placements.reduce((v, p) => v + p.w * p.h, 0));
447 for (const p of sheet.placements) this.emit("PLACE", letter(p.id), "x=" + p.x, "y=" + p.y, "w=" + p.w, "h=" + p.h, ...(p.rotated ? ["rotated"] : []));
448 this.emit("FREE", ...(sheet.free.length ? sheet.free.map(f => f.x + "," + f.y + "," + f.w + "x" + f.h) : ["none"]));
449 this.drawing(width, height, sheet.placements);
450 });
451 }

453 command(cmd, a) {
454 if (cmd === "ITEMS" || cmd === "RANDOM") {
455 let capacity;
456 let sizes;
457 if (cmd === "ITEMS") {
458 capacity = Number(a[0]);
459 if (capacity === 0) throw new CommandError("BAD CAPACITY");
460 sizes = a.slice(1).map(Number);
461 const bad = sizes.findIndex(s => s === 0 || s > capacity);
462 if (bad >= 0) throw new CommandError("BAD ITEM " + (bad + 1));
463 } else {
464 const count = Number(a[0]);
465 if (count === 0 || count > 40) throw new CommandError("BAD COUNT");
466 capacity = Number(a[1]);
467 if (capacity === 0) throw new CommandError("BAD CAPACITY");
468 let seed = Number(a[2]);
469 if (seed === 0 || seed >= 2147483647) throw new CommandError("BAD SEED");
470 sizes = [];
471 for (let i = 0; i < count; i++) {
472 seed = 48271 * seed % 2147483647;
473 sizes.push(seed % capacity + 1);
474 }
475 }
476 this.capacity = capacity;
477 this.items = sizes.map((size, i) => ({id: i + 1, size}));
478 this.emit(cmd, "count=" + sizes.length, "capacity=" + capacity, "total=" + total(this.items));
479 if (cmd === "RANDOM") this.emit("SIZES", ...sizes);
480 return;
481 }
482 if (cmd === "CUTSTOCK") {
483 this.cutstock(Number(a[0]), a.slice(1));
484 return;
485 }
486 if (cmd === "SHELF") {
487 this.shelf(Number(a[0]), a.slice(1));
488 return;
489 }
490 if (cmd === "GUILLOTINE") {
491 this.guillotine(Number(a[0]), Number(a[1]), a[2], a[3], a.slice(4));
492 return;
493 }
494 if (this.capacity === null) throw new CommandError("NO ITEMS");
495 const capacity = this.capacity;
496 if (METHODS.includes(cmd) || cmd === "TRACE" || cmd === "REDUCE") {
497 const method = cmd === "TRACE" || cmd === "REDUCE" ? a[0] : cmd;
498 const k = Number(a[cmd === "TRACE" || cmd === "REDUCE" ? 1 : 0] ?? 0);
499 if (["HARMONIC", "KBOUNDED"].includes(method) && k === 0) throw new CommandError("BAD PARAMETER");
500 if (cmd === "REDUCE") {
501 this.reduce(method, k);
502 return;
503 }
504 const result = pack(this.items, capacity, method, k);
505 this.emit((cmd === "TRACE" ? "TRACE " : "") + this.header(method, k, result));
506 if (cmd === "TRACE") {
507 result.steps.forEach((s, i) => this.emit("STEP", i + 1, "item=" + notation(s.item), "bin=" + s.bin, "inspected=" + s.inspected,
508 ...(s.fresh ? ["new"] : []), ...(s.closed === null ? [] : ["closed=" + s.closed]), ...(s.cls === null ? [] : ["class=" + s.cls])));
509 } else result.bins.forEach((b, i) => this.binLine(b, i + 1, capacity, method === "HARMONIC"));
510 } else if (cmd === "BOUNDS") {
511 const b = bounds(this.items, capacity);
512 this.emit("BOUNDS", "total=" + total(this.items), "L1=" + b.l1, "L2=" + b.l2);
513 for (const r of b.rows) this.emit("ALPHA", r.alpha, "j1=" + r.j1, "j2=" + r.j2, "j3=" + r.j3, "bound=" + r.bound);
514 } else if (cmd === "COMPARE") {
515 const k = Number(a[0]);
516 if (k === 0) throw new CommandError("BAD PARAMETER");
517 const b = bounds(this.items, capacity);
518 this.emit("COMPARE", "k=" + k, "L1=" + b.l1, "L2=" + b.l2);
519 const rows = METHODS.map(method => ({method, result: pack(this.items, capacity, method, k)}));
520 rows.sort((a, b) => a.result.bins.length - b.result.bins.length || (a.method < b.method ? -1 : a.method > b.method ? 1 : 0));
521 for (const r of rows) this.emit(r.method, "bins=" + r.result.bins.length, "gap=" + (r.result.bins.length - b.l2),
522 "waste=" + (r.result.bins.length * capacity - total(this.items)), "inspections=" + r.result.inspections);
523 } else if (cmd === "EXACT") {
524 this.exact(Number(a[0]));
525 } else if (cmd === "CHECK") {
526 const labels = a.map(Number);
527 if (labels.length !== this.items.length || labels.some(n => n === 0)) throw new CommandError("BAD ASSIGNMENT");
528 const bins = new Map();
529 this.items.forEach((item, i) => {
530 if (!bins.has(labels[i])) bins.set(labels[i], {load: 0, items: [], cls: null});
531 const bin = bins.get(labels[i]);
532 bin.load += item.size;
533 bin.items.push(item);
534 });
535 const over = [...bins.values()].filter(b => b.load > capacity).length;
536 this.emit("CHECK", "bins=" + bins.size, "over=" + over, "L2=" + bounds(this.items, capacity).l2, "status=" + (over ? "invalid" : "valid"));
537 for (const label of [...bins.keys()].sort((a, b) => a - b)) this.binLine(bins.get(label), label, capacity, false, bins.get(label).load > capacity);
538 } else if (cmd === "COVER") {
539 this.cover();
540 } else if (cmd === "MULTIFIT") {
541 this.multifit(Number(a[0]));
542 }
543 }

547 try {
548 this.command(cmd, args);
549 } catch (error) {
550 if (!(error instanceof CommandError)) throw error;
551 this.emit("ERROR", error.message);
552 }
553 }
