103fn pack(items: &[Item], capacity: i64, method: &str, parameter: usize) -> Packing {
104 let mut result = Packing::new(method, parameter);
105 let ordered = if matches!(method, "FFD" | "BFD" | "WFD") {
106 descending(items)
107 } else {
108 items.to_vec()
109 };
110 let mut active = Vec::<usize>::new();
111 let mut current = BTreeMap::<usize, usize>::new();
112
113 for item in ordered {
114 let before = result.inspections;
115 let mut chosen = None;
116 let mut closed = None;
117 let mut category = 0;
118 match method {
119 "NEXTFIT" => {
120 if !result.bins.is_empty() {
121 result.inspections += 1;
122 let last = result.bins.len() - 1;
123 if result.bins[last].load + item.size <= capacity {
124 chosen = Some(last);
125 }
126 }
127 }
128 "FIRSTFIT" | "FFD" | "KBOUNDED" => {
129 let candidates: Vec<usize> = if method == "KBOUNDED" {
130 active.clone()
131 } else {
132 (0..result.bins.len()).collect()
133 };
134 for index in candidates {
135 result.inspections += 1;
136 if result.bins[index].load + item.size <= capacity {
137 chosen = Some(index);
138 break;
139 }
140 }
141 }
142 "HARMONIC" => {
143 category = ((capacity / item.size) as usize).min(parameter);
144 if let Some(&index) = current.get(&category) {
145 result.inspections += 1;
146 if result.bins[index].load + item.size <= capacity {
147 chosen = Some(index);
148 }
149 }
150 }
151 _ => {
152 result.inspections += result.bins.len();
153 let mut fitting: Vec<usize> = (0..result.bins.len())
154 .filter(|&i| result.bins[i].load + item.size <= capacity)
155 .collect();
156 if !fitting.is_empty() {
157 if matches!(method, "BESTFIT" | "BFD") {
158 chosen = fitting.into_iter()
159 .max_by_key(|&i| (result.bins[i].load, -(i as i64)));
160 } else {
161 fitting.sort_by_key(|&i| (result.bins[i].load, i));
162 chosen = Some(
163 if method == "ALMOSTWORST" && fitting.len() >= 2 {
164 fitting[1]
165 } else {
166 fitting[0]
167 },
168 );
169 }
170 }
171 }
172 }
173
174 let opened = chosen.is_none();
175 let index = if let Some(index) = chosen {
176 index
177 } else {
178 if method == "KBOUNDED" && active.len() == parameter {
179 let remove = *active.iter().max_by_key(|&&i| {
180 (result.bins[i].load, -(i as i64))
181 }).unwrap();
182 active.retain(|&i| i != remove);
183 result.closes += 1;
184 closed = Some(remove);
185 }
186 let index = result.bins.len();
187 result.bins.push(Bin::new(category));
188 if method == "KBOUNDED" {
189 active.push(index);
190 } else if method == "HARMONIC" {
191 current.insert(category, index);
192 }
193 index
194 };
195 result.bins[index].put(item);
196 result.steps.push(Step {
197 item,
198 bin: index,
199 inspected: result.inspections - before,
200 opened,
201 closed,
202 category,
203 });
204 }
205 result
206}

208struct BoundRow {
209 alpha: i64,
210 j1: usize,
211 j2: usize,
212 j3: usize,
213 bound: i64,
214}
215
216struct Bounds {
217 l1: i64,
218 l2: i64,
219 rows: Vec<BoundRow>,
220}
221
222fn lower_bounds(items: &[Item], capacity: i64) -> Bounds {
223 let total: i64 = items.iter().map(|item| item.size).sum();
224 let mut thresholds = BTreeSet::from([0]);
225 for item in items {
226 if 2 * item.size <= capacity {
227 thresholds.insert(item.size);
228 }
229 }
230 let mut lower = 0;
231 let mut rows = Vec::new();
232 for alpha in thresholds {
233 let (mut j1, mut j2, mut j3) = (0, 0, 0);
234 let (mut sum2, mut sum3) = (0, 0);
235 for item in items {
236 let size = item.size;
237 if size > capacity - alpha {
238 j1 += 1;
239 } else if 2 * size > capacity {
240 j2 += 1;
241 sum2 += size;
242 } else if size >= alpha {
243 j3 += 1;
244 sum3 += size;
245 }
246 }
247 let excess = sum3 - (j2 as i64 * capacity - sum2);
248 let bound = j1 as i64 + j2 as i64 +
249 if excess > 0 { ceiling(excess, capacity) } else { 0 };
250 lower = lower.max(bound);
251 rows.push(BoundRow { alpha, j1, j2, j3, bound });
252 }
253 Bounds { l1: ceiling(total, capacity), l2: lower, rows }
254}

256struct ExactSearch<'a> {
257 items: Vec<Item>,
258 capacity: i64,
259 limit: usize,
260 lower: i64,
261 remaining: Vec<i64>,
262 best: Vec<Bin>,
263 live: Vec<Bin>,
264 nodes: usize,
265 status: Option<&'static str>,
266 events: Vec<String>,
267 marker: std::marker::PhantomData<&'a ()>,
268}
269
270impl ExactSearch<'_> {
271 fn new(items: &[Item], capacity: i64, limit: usize, lower: i64,
272 incumbent: &[Bin]) -> Self {
273 let ordered = descending(items);
274 let mut remaining = vec![0; ordered.len() + 1];
275 for i in (0..ordered.len()).rev() {
276 remaining[i] = remaining[i + 1] + ordered[i].size;
277 }
278 Self {
279 items: ordered,
280 capacity,
281 limit,
282 lower,
283 remaining,
284 best: incumbent.to_vec(),
285 live: Vec::new(),
286 nodes: 0,
287 status: None,
288 events: Vec::new(),
289 marker: std::marker::PhantomData,
290 }
291 }
292
293 fn visit(&mut self, index: usize) {
294 if self.nodes == self.limit {
295 self.status = Some("limit");
296 return;
297 }
298 self.nodes += 1;
299 let count = self.live.len();
300 let used = self.remaining[0] - self.remaining[index];
301 let free = count as i64 * self.capacity - used;
302 let excess = self.remaining[index] - free;
303 let minimum = count as i64 +
304 if excess > 0 { ceiling(excess, self.capacity) } else { 0 };
305 if minimum >= self.best.len() as i64 {
306 return;
307 }
308 if index == self.items.len() {
309 self.best = self.live.clone();
310 self.events.push(format!("IMPROVE bins={} nodes={}", count, self.nodes));
311 if count as i64 == self.lower {
312 self.status = Some("bound");
313 }
314 return;
315 }
316 let item = self.items[index];
317 let mut seen = BTreeSet::new();
318 for i in 0..self.live.len() {
319 let old = self.live[i].load;
320 if !seen.insert(old) || old + item.size > self.capacity {
321 continue;
322 }
323 self.live[i].put(item);
324 self.visit(index + 1);
325 self.live[i].items.pop();
326 self.live[i].load = old;
327 if self.status.is_some() {
328 return;
329 }
330 }
331 if count + 1 < self.best.len() {
332 let mut bin = Bin::default();
333 bin.put(item);
334 self.live.push(bin);
335 self.visit(index + 1);
336 self.live.pop();
337 }
338 }
339}

554 fn cmd_reduce(&mut self, method: &str, parameter: usize) {
555 let result = pack(self.items(), self.capacity, method, parameter);
556 let mut bins = result.bins;
557 let before = bins.len();
558 let mut events = Vec::new();
559 loop {
560 let mut sources: Vec<usize> = (0..bins.len()).collect();
561 sources.sort_by_key(|&i| (bins[i].load, i));
562 let mut eliminated = false;
563 for source in sources {
564 let mut loads = bins.iter().map(|bin| bin.load).collect::<Vec<_>>();
565 let mut moves = Vec::<(Item, usize)>::new();
566 let mut possible = true;
567 for item in descending(&bins[source].items) {
568 let mut destination = None;
569 for i in 0..bins.len() {
570 if i == source || loads[i] + item.size > self.capacity {
571 continue;
572 }
573 if destination.map_or(true, |j| loads[i] > loads[j]) {
574 destination = Some(i);
575 }
576 }
577 if let Some(i) = destination {
578 loads[i] += item.size;
579 moves.push((item, i));
580 } else {
581 possible = false;
582 break;
583 }
584 }
585 if !possible {
586 continue;
587 }
588 let descriptions = moves.iter().map(|(item, i)| {
589 format!("{}->{}", item.text(), i + 1)
590 }).collect::<Vec<_>>().join(" ");
591 events.push(format!(
592 "ELIMINATE bin={} load={} moves={}",
593 source + 1, bins[source].load, descriptions
594 ));
595 for (item, destination) in moves {
596 bins[destination].put(item);
597 }
598 bins.remove(source);
599 eliminated = true;
600 break;
601 }
602 if !eliminated {
603 break;
604 }
605 }
606 self.emit(format!(
607 "REDUCE {} before={} after={} eliminated={}",
608 result.name(), before, bins.len(), before - bins.len()
609 ));
610 for event in events {
611 self.emit(event);
612 }
613 self.bin_lines(&bins, method == "HARMONIC");
614 }

735 sequential.push(Bin::default());

750 last -= 1;
751 bin.put(ordered[last]);
752 }
753 paired.push(bin);
754 }
755 self.cover_lines("BIGSMALL", &paired);

772 fn cmd_multifit(&mut self, machines: usize) -> Result<(), String> {
773 if machines == 0 {
774 return Err("BAD PARAMETER".into());
775 }
776 let largest = self.items().iter().map(|item| item.size).max().unwrap();
777 let mut low = ceiling(self.total(), machines as i64).max(largest);
778 let mut high = ceiling(2 * self.total(), machines as i64).max(largest);
779 self.emit(format!(
780 "MULTIFIT m={} lower={} upper={}", machines, low, high
781 ));
782 while low < high {
783 let middle = low + (high - low) / 2;
784 let count = pack(self.items(), middle, "FFD", 0).bins.len();
785 let ok = count <= machines;
786 self.emit(format!(
787 "TRY cap={} bins={} {}", middle, count, if ok { "ok" } else { "fail" }
788 ));
789 if ok { high = middle; } else { low = middle + 1; }
790 }
791 let mut result = pack(self.items(), high, "FFD", 0).bins;
792 while result.len() < machines {
793 result.push(Bin::default());
794 }
795 let makespan = result.iter().map(|bin| bin.load).max().unwrap();
796 self.emit(format!("RESULT cap={} makespan={}", high, makespan));
797 self.machine_lines(&result);
798
799 let mut lpt = vec![Bin::default(); machines];
800 for item in descending(self.items()) {
801 let chosen = (0..machines).min_by_key(|&i| (lpt[i].load, i)).unwrap();
802 lpt[chosen].put(item);
803 }
804 let makespan = lpt.iter().map(|bin| bin.load).max().unwrap();
805 self.emit(format!("LPT makespan={}", makespan));
806 self.machine_lines(&lpt);
807 Ok(())
808 }

810 fn enumerate_patterns(index: usize, remaining: i64, pieces: &[Piece],
811 counts: &mut Vec<usize>, patterns: &mut Vec<Pattern>) {
812 let maximum = (remaining / pieces[index].length) as usize;
813 if index == pieces.len() - 1 {
814 counts[index] = maximum;
815 patterns.push(Pattern {
816 counts: counts.clone(),
817 trim: remaining - maximum as i64 * pieces[index].length,
818 });
819 return;
820 }
821 for count in (0..=maximum).rev() {
822 counts[index] = count;
823 Self::enumerate_patterns(
824 index + 1, remaining - count as i64 * pieces[index].length,
825 pieces, counts, patterns
826 );
827 }
828 }

830 fn cmd_cutstock(&mut self, words: &[&str]) -> Result<(), String> {
831 let length: i64 = words[1].parse().unwrap();
832 if length == 0 {
833 return Err("BAD CAPACITY".into());
834 }
835 let mut pieces = Vec::new();
836 let mut seen = BTreeSet::new();
837 for (index, word) in words[2..].iter().enumerate() {
838 let (size, demand) = word.split_once(':').unwrap();
839 let size: i64 = size.parse().unwrap();
840 let demand: usize = demand.parse().unwrap();
841 if size == 0 || size > length || demand == 0 {
842 return Err(format!("BAD PIECE {}", index + 1));
843 }
844 if !seen.insert(size) {
845 return Err(format!("DUPLICATE PIECE {}", index + 1));
846 }
847 pieces.push(Piece { length: size, demand });
848 }
849 pieces.sort_by_key(|piece| -piece.length);
850 let mut patterns = Vec::new();
851 let mut counts = vec![0; pieces.len()];
852 Self::enumerate_patterns(0, length, &pieces, &mut counts, &mut patterns);
853
854 self.emit(format!(
855 "CUTSTOCK L={} pieces={} patterns={}",
856 length, pieces.len(), patterns.len()
857 ));
858 for (index, pattern) in patterns.iter().enumerate() {
859 let breakdown = pattern.counts.iter().enumerate()
860 .filter(|(_, count)| **count > 0)
861 .map(|(i, count)| format!("{}x{}", count, pieces[i].length))
862 .collect::<Vec<_>>().join("+");
863 self.emit(format!(
864 "PATTERN {} {} rest={}", index + 1, breakdown, pattern.trim
865 ));
866 }
867 let mut remaining = pieces.iter().map(|piece| piece.demand)
868 .collect::<Vec<_>>();
869 let mut cut = vec![0; pieces.len()];
870 let mut used = vec![0; patterns.len()];
871 while remaining.iter().any(|&count| count > 0) {
872 let mut chosen = 0;
873 let mut highest = 0;
874 for (index, pattern) in patterns.iter().enumerate() {
875 let useful: usize = pattern.counts.iter().zip(&remaining)
876 .map(|(&count, &need)| count.min(need)).sum();
877 if index == 0 || useful > highest {
878 chosen = index;
879 highest = useful;
880 }
881 }
882 used[chosen] += 1;
883 for i in 0..pieces.len() {
884 cut[i] += patterns[chosen].counts[i];
885 remaining[i] = remaining[i].saturating_sub(patterns[chosen].counts[i]);
886 }
887 }
888 let mut bars = 0;
889 let mut trim = 0;
890 for (i, &times) in used.iter().enumerate() {
891 if times > 0 {
892 self.emit(format!("USE {} bars={}", i + 1, times));
893 bars += times;
894 trim += times as i64 * patterns[i].trim;
895 }
896 }
897 let quantity: i64 = pieces.iter()
898 .map(|piece| piece.length * piece.demand as i64).sum();
899 let mut stock = Vec::new();
900 for (index, piece) in pieces.iter().enumerate() {
901 self.emit(format!(
902 "PIECE {} demand={} cut={}",
903 piece.length, piece.demand, cut[index]
904 ));
905 for _ in 0..piece.demand {
906 stock.push(Item { number: stock.len() + 1, size: piece.length });
907 }
908 }
909 let demand_total: usize = pieces.iter().map(|piece| piece.demand).sum();
910 let cut_total: usize = cut.iter().sum();
911 self.emit(format!(
912 "TOTAL bars={} trim={} surplus={} waste={}",
913 bars, trim, cut_total - demand_total, bars as i64 * length - quantity
914 ));
915 let ffd = pack(&stock, length, "FIRSTFIT", 0).bins.len();
916 self.emit(format!(
917 "FFD bars={} waste={}", ffd, ffd as i64 * length - quantity
918 ));
919 self.emit(format!("LOWER bars={}", ceiling(quantity, length)));
920 Ok(())
921 }

923 fn rectangles(words: &[&str], start: usize, width: usize, height: usize,
924 rotate: bool) -> Result<Vec<Rect>, String> {
925 let mut result = Vec::new();
926 for (index, word) in words[start..].iter().enumerate() {
927 let (w, h) = word.split_once('x').unwrap();
928 let w: usize = w.parse().unwrap();
929 let h: usize = h.parse().unwrap();
930 let fits = if height == 0 {
931 w <= width
932 } else {
933 (w <= width && h <= height) || (rotate && h <= width && w <= height)
934 };
935 if w == 0 || h == 0 || !fits {
936 return Err(format!("BAD RECT {}", index + 1));
937 }
938 result.push(Rect { number: index + 1, width: w, height: h });
939 }
940 Ok(result)
941 }

957 fn cmd_shelf(&mut self, words: &[&str]) -> Result<(), String> {
958 let width: usize = words[1].parse().unwrap();
959 if width == 0 {
960 return Err("BAD CAPACITY".into());
961 }
962 let mut rects = Self::rectangles(words, 2, width, 0, false)?;
963 let area: usize = rects.iter().map(|rect| rect.area()).sum();
964 self.emit(format!(
965 "SHELF W={} rects={} area={}", width, rects.len(), area
966 ));
967 rects.sort_by_key(|rect| (
968 -(rect.height as i64), -(rect.width as i64), rect.number
969 ));
970 for method in ["NFDH", "FFDH", "BFDH"] {
971 let mut shelves: Vec<Shelf> = Vec::new();
972 let mut placements = Vec::new();
973 for &rect in &rects {
974 let mut chosen = None;
975 if method == "NFDH" {
976 if let Some(last) = shelves.last() {
977 if last.used + rect.width <= width {
978 chosen = Some(shelves.len() - 1);
979 }
980 }
981 } else {
982 for (index, shelf) in shelves.iter().enumerate() {
983 if shelf.used + rect.width > width {
984 continue;
985 }
986 if chosen.map_or(true, |i| {
987 method == "BFDH" && shelf.used > shelves[i].used
988 }) {
989 chosen = Some(index);
990 }
991 if method == "FFDH" {
992 break;
993 }
994 }
995 }
996 let index = if let Some(index) = chosen {
997 index
998 } else {
999 let bottom = shelves.last().map_or(0, |last| last.bottom + last.height);
1000 shelves.push(Shelf {
1001 bottom,
1002 height: rect.height,
1003 used: 0,
1004 placements: Vec::new(),
1005 });
1006 shelves.len() - 1
1007 };
1008 let shelf = &mut shelves[index];
1009 let p = Placement {
1010 rect,
1011 x: shelf.used,
1012 y: shelf.bottom,
1013 width: rect.width,
1014 height: rect.height,
1015 rotated: false,
1016 };
1017 shelf.placements.push(p.clone());
1018 placements.push(p);
1019 shelf.used += rect.width;
1020 }
1021 let last = shelves.last().unwrap();
1022 let height = last.bottom + last.height;
1023 self.emit(format!(
1024 "{} height={} shelves={} util={}",
1025 method, height, shelves.len(), area * 100 / (width * height)
1026 ));
1027 for (index, shelf) in shelves.iter().enumerate() {
1028 let content = shelf.placements.iter()
1029 .map(|p| format!("{}@{}", p.rect.letter(), p.x))
1030 .collect::<Vec<_>>().join(" ");
1031 self.emit(format!(
1032 "LEVEL {} y={} h={} used={} items={}",
1033 index + 1, shelf.bottom, shelf.height, shelf.used, content
1034 ));
1035 }
1036 self.draw(width, height, &placements);
1037 }
1038
1039 let mut columns = vec![0; width];
1040 let mut skyline = Vec::new();
1041 for &rect in &rects {
1042 let mut best = (usize::MAX, 0);
1043 for x in 0..=width - rect.width {
1044 let y = *columns[x..x + rect.width].iter().max().unwrap();
1045 if y < best.0 {
1046 best = (y, x);
1047 }
1048 }
1049 let (y, x) = best;
1050 skyline.push(Placement {
1051 rect,
1052 x,
1053 y,
1054 width: rect.width,
1055 height: rect.height,
1056 rotated: false,
1057 });
1058 for column in &mut columns[x..x + rect.width] {
1059 *column = y + rect.height;
1060 }
1061 }
1062 let height = *columns.iter().max().unwrap();
1063 self.emit(format!(
1064 "SKYLINE height={} util={}", height, area * 100 / (width * height)
1065 ));
1066 for p in &skyline {
1067 self.emit(format!(
1068 "PLACE {} x={} y={} w={} h={}",
1069 p.rect.letter(), p.x, p.y, p.width, p.height
1070 ));
1071 }
1072 self.draw(width, height, &skyline);
1073 Ok(())
1074 }

1077 (rect.width <= free.width && rect.height <= free.height) ||
1078 (rotate && rect.height <= free.width && rect.width <= free.height)

1081 fn cmd_guillotine(&mut self, words: &[&str]) -> Result<(), String> {
1082 let width: usize = words[1].parse().unwrap();
1083 let height: usize = words[2].parse().unwrap();
1084 if width == 0 || height == 0 {
1085 return Err("BAD CAPACITY".into());
1086 }
1087 let rotate = words[3] == "ROTATE";
1088 let best = words[4] == "BEST";
1089 let mut rects = Self::rectangles(words, 5, width, height, rotate)?;
1090 rects.sort_by_key(|rect| (-(rect.area() as i64), rect.number));
1091 let area: usize = rects.iter().map(|rect| rect.area()).sum();
1092 let mut sheets: Vec<Sheet> = Vec::new();
1093 let mut rotated = 0;
1094
1095 for rect in &rects {
1096 let mut selected = None;
1097 let mut free_index = 0;
1098 let mut selected_area = usize::MAX;
1099 for (s, sheet) in sheets.iter().enumerate() {
1100 for (i, free) in sheet.free.iter().enumerate() {
1101 if !Self::fits(*rect, free, rotate) {
1102 continue;
1103 }
1104 if selected.is_none() || (best && free.area() < selected_area) {
1105 selected = Some(s);
1106 free_index = i;
1107 selected_area = free.area();
1108 }
1109 if !best {
1110 break;
1111 }
1112 }
1113 if selected.is_some() && !best {
1114 break;
1115 }
1116 }
1117 let sheet_index = if let Some(index) = selected {
1118 index
1119 } else {
1120 sheets.push(Sheet::new(width, height));
1121 free_index = 0;
1122 sheets.len() - 1
1123 };
1124 let sheet = &mut sheets[sheet_index];
1125 let free = sheet.free.remove(free_index);
1126 let turned = rect.width > free.width || rect.height > free.height;
1127 let (w, h) = if turned {
1128 (rect.height, rect.width)
1129 } else {
1130 (rect.width, rect.height)
1131 };
1132 if turned {
1133 rotated += 1;
1134 }
1135 sheet.placements.push(Placement {
1136 rect: *rect,
1137 x: free.x,
1138 y: free.y,
1139 width: w,
1140 height: h,
1141 rotated: turned,
1142 });
1143 sheet.area += rect.area();
1144 let rw = free.width - w;
1145 let rh = free.height - h;
1146 let (top, right) = if rw <= rh {
1147 (
1148 FreeRect { x: free.x, y: free.y + h, width: free.width, height: rh },
1149 FreeRect { x: free.x + w, y: free.y, width: rw, height: h },
1150 )
1151 } else {
1152 (
1153 FreeRect { x: free.x, y: free.y + h, width: w, height: rh },
1154 FreeRect { x: free.x + w, y: free.y, width: rw, height: free.height },
1155 )
1156 };
1157 if top.width > 0 && top.height > 0 {
1158 sheet.free.push(top);
1159 }
1160 if right.width > 0 && right.height > 0 {
1161 sheet.free.push(right);
1162 }
1163 }
1164 self.emit(format!(
1165 "GUILLOTINE W={} H={} mode={} rule={} rects={} sheets={} rotated={} util={}",
1166 width, height, words[3], words[4], rects.len(), sheets.len(), rotated,
1167 area * 100 / (sheets.len() * width * height)
1168 ));
1169 for (index, sheet) in sheets.iter().enumerate() {
1170 self.emit(format!(
1171 "SHEET {} placed={} area={}",
1172 index + 1, sheet.placements.len(), sheet.area
1173 ));
1174 for p in &sheet.placements {
1175 self.emit(format!(
1176 "PLACE {} x={} y={} w={} h={}{}",
1177 p.rect.letter(), p.x, p.y, p.width, p.height,
1178 if p.rotated { " rotated" } else { "" }
1179 ));
1180 }
1181 let free = if sheet.free.is_empty() {
1182 "FREE none".to_string()
1183 } else {
1184 format!("FREE {}", sheet.free.iter().map(|f| f.text())
1185 .collect::<Vec<_>>().join(" "))
1186 };
1187 self.emit(free);
1188 self.draw(width, height, &sheet.placements);
1189 }
1190 Ok(())
1191 }

1193 fn process(&mut self, words: &[&str]) -> Result<(), String> {
1194 let method = words[0];
1195 match method {
1196 "ITEMS" => self.cmd_items(words),
1197 "RANDOM" => self.cmd_random(words),
1198 "CUTSTOCK" => self.cmd_cutstock(words),
1199 "SHELF" => self.cmd_shelf(words),
1200 "GUILLOTINE" => self.cmd_guillotine(words),
1201 _ => {
1202 if self.items.is_none() {
1203 return Err("NO ITEMS".into());
1204 }
1205 match method {
1206 "NEXTFIT" | "FIRSTFIT" | "BESTFIT" | "WORSTFIT" |
1207 "ALMOSTWORST" | "HARMONIC" | "KBOUNDED" | "FFD" | "BFD" | "WFD" => {
1208 let k = Self::parameter(method, words, 1)?;
1209 self.cmd_method(method, k);
1210 }
1211 "TRACE" | "REDUCE" => {
1212 let name = words[1];
1213 let k = Self::parameter(name, words, 2)?;
1214 if method == "TRACE" {
1215 self.cmd_trace(name, k);
1216 } else {
1217 self.cmd_reduce(name, k);
1218 }
1219 }
1220 "BOUNDS" => self.cmd_bounds(),
1221 "COMPARE" => self.cmd_compare(words[1].parse().unwrap())?,
1222 "EXACT" => self.cmd_exact(words[1].parse().unwrap())?,
1223 "CHECK" => self.cmd_check(words)?,
1224 "COVER" => self.cmd_cover(),
1225 "MULTIFIT" => self.cmd_multifit(words[1].parse().unwrap())?,
1226 _ => unreachable!(),
1227 }
1228 Ok(())
1229 }
1230 }
1231 }
