60def pack(items, capacity, method, parameter=0):
61 result = Packing(method, parameter)
62 order = descending(items) if method in ("FFD", "BFD", "WFD") else items
63 active = []
64 current = {}
65 for item in order:
66 size = item[1]
67 before = result.inspections
68 chosen = None
69 closed = None
70 category = None
71 if method == "NEXTFIT":
72 if result.bins:
73 result.inspections += 1
74 if result.bins[-1].load + size <= capacity:
75 chosen = len(result.bins) - 1
76 elif method in ("FIRSTFIT", "FFD", "KBOUNDED"):
77 candidates = active if method == "KBOUNDED" else range(len(result.bins))
78 for index in candidates:
79 result.inspections += 1
80 if result.bins[index].load + size <= capacity:
81 chosen = index
82 break
83 elif method == "HARMONIC":
84 category = min(capacity // size, parameter)
85 if category in current:
86 result.inspections += 1
87 index = current[category]
88 if result.bins[index].load + size <= capacity:
89 chosen = index
90 else:
91 result.inspections += len(result.bins)
92 fitting = [index for index, bin in enumerate(result.bins) if bin.load + size <= capacity]
93 if fitting:
94 if method in ("BESTFIT", "BFD"):
95 chosen = max(fitting, key=lambda index: (result.bins[index].load, -index))
96 else:
97 fitting.sort(key=lambda index: (result.bins[index].load, index))
98 chosen = fitting[1] if method == "ALMOSTWORST" and len(fitting) >= 2 else fitting[0]
99 opened = chosen is None
100 if opened:
101 if method == "KBOUNDED" and len(active) == parameter:
102 closed = max(active, key=lambda index: (result.bins[index].load, -index))
103 active.remove(closed)
104 result.closes += 1
105 chosen = len(result.bins)
106 result.bins.append(Bin(category))
107 if method == "KBOUNDED":
108 active.append(chosen)
109 if method == "HARMONIC":
110 current[category] = chosen
111 result.bins[chosen].put(item)
112 result.steps.append((item, chosen, result.inspections - before, opened, closed, category))
113 return result

116def bounds(items, capacity):
117 total = sum(size for _, size in items)
118 thresholds = sorted({0} | {size for _, size in items if 2 * size <= capacity})
119 rows = []
120 for alpha in thresholds:
121 first = [size for _, size in items if size > capacity - alpha]
122 second = [size for _, size in items if 2 * size > capacity and size <= capacity - alpha]
123 third = [size for _, size in items if 2 * size <= capacity and size >= alpha]
124 excess = sum(third) - (len(second) * capacity - sum(second))
125 bound = len(first) + len(second) + (ceiling(excess, capacity) if excess > 0 else 0)
126 rows.append((alpha, len(first), len(second), len(third), bound))
127 return ceiling(total, capacity), max(row[4] for row in rows), rows

130class ExactSearch:
131 def __init__(self, items, capacity, limit, lower, incumbent, emit):
132 self.items = descending(items)
133 self.capacity = capacity
134 self.limit = limit
135 self.lower = lower
136 self.best = [bin.copy() for bin in incumbent]
137 self.live = []
138 self.nodes = 0
139 self.status = None
140 self.emit = emit
141 self.remaining = [0] * (len(items) + 1)
142 for index in range(len(items) - 1, -1, -1):
143 self.remaining[index] = self.remaining[index + 1] + self.items[index][1]
144
145 def visit(self, index):
146 if self.nodes == self.limit:
147 self.status = "limit"
148 return
149 self.nodes += 1
150 count = len(self.live)
151 used = self.remaining[0] - self.remaining[index]
152 free = count * self.capacity - used
153 excess = self.remaining[index] - free
154 minimum = count + (ceiling(excess, self.capacity) if excess > 0 else 0)
155 if minimum >= len(self.best):
156 return
157 if index == len(self.items):
158 self.best = [bin.copy() for bin in self.live]
159 self.emit(f"IMPROVE bins={count} nodes={self.nodes}")
160 if count == self.lower:
161 self.status = "bound"
162 return
163 item = self.items[index]
164 seen_loads = set()
165 for bin in self.live:
166 old_load = bin.load
167 if old_load in seen_loads:
168 continue
169 seen_loads.add(old_load)
170 if old_load + item[1] > self.capacity:
171 continue
172 bin.put(item)
173 self.visit(index + 1)
174 bin.items.pop()
175 bin.load = old_load
176 if self.status is not None:
177 return
178 if count + 1 < len(self.best):
179 bin = Bin()
180 bin.put(item)
181 self.live.append(bin)
182 self.visit(index + 1)
183 self.live.pop()

277 result = self.method(method, arguments)
278 bins = result.bins
279 before = len(bins)
280 events = []
281 while True:
282 eliminated = False
283 for source in sorted(range(len(bins)), key=lambda index: (bins[index].load, index)):
284 loads = [bin.load for bin in bins]
285 moves = []

389 def cmd_multifit(self, count):
390 count = int(count)
391 if count == 0:
392 raise CommandError("BAD PARAMETER")
393 total = self.total()
394 largest = max(size for _, size in self.items)
395 lower = max(ceiling(total, count), largest)
396 upper = max(ceiling(2 * total, count), largest)
397 self.emit(f"MULTIFIT m={count} lower={lower} upper={upper}")
398 while lower < upper:
399 middle = (lower + upper) // 2
400 number = len(pack(self.items, middle, "FFD").bins)
401 success = number <= count
402 self.emit(f"TRY cap={middle} bins={number} {'ok' if success else 'fail'}")
403 if success:
404 upper = middle
405 else:
406 lower = middle + 1
407 machines = pack(self.items, upper, "FFD").bins
408 machines.extend(Bin() for _ in range(count - len(machines)))
409 self.emit(f"RESULT cap={upper} makespan={max(bin.load for bin in machines)}")
410 self.machine_lines(machines)
411 lpt = [Bin() for _ in range(count)]
412 for item in descending(self.items):
413 chosen = min(range(count), key=lambda index: (lpt[index].load, index))
414 lpt[chosen].put(item)
415 self.emit(f"LPT makespan={max(bin.load for bin in lpt)}")

433 def cmd_cutstock(self, length, *tokens):
434 length = int(length)
435 if length == 0:
436 raise CommandError("BAD CAPACITY")
437 pieces = []
438 seen = set()
439 for number, token in enumerate(tokens, 1):
440 size, demand = (int(value) for value in token.split(":"))
441 if size == 0 or size > length or demand == 0:
442 raise CommandError(f"BAD PIECE {number}")
443 if size in seen:
444 raise CommandError(f"DUPLICATE PIECE {number}")
445 seen.add(size)
446 pieces.append((size, demand))
447 pieces.sort(reverse=True)
448 patterns = self.maximal_patterns(length, pieces)
449 remaining = [demand for _, demand in pieces]
450 uses = [0] * len(patterns)
451 while any(remaining):
452 chosen = max(
453 range(len(patterns)),
454 key=lambda index: (sum(min(count, needed) for count, needed in zip(patterns[index][0], remaining)), -index),
455 )
456 uses[chosen] += 1
457 remaining = [max(0, needed - count) for needed, count in zip(remaining, patterns[chosen][0])]
458 cut = [sum(use * pattern[0][index] for use, pattern in zip(uses, patterns)) for index in range(len(pieces))]
459 quantity = sum(size * demand for size, demand in pieces)
460 bars = sum(uses)
461 self.emit(f"CUTSTOCK L={length} pieces={len(pieces)} patterns={len(patterns)}")
462 for number, (counts, trim) in enumerate(patterns, 1):
463 breakdown = "+".join(f"{count}x{size}" for count, (size, _) in zip(counts, pieces) if count)
464 self.emit(f"PATTERN {number} {breakdown} rest={trim}")
465 for number, use in enumerate(uses, 1):
466 if use:
467 self.emit(f"USE {number} bars={use}")
468 for (size, demand), count in zip(pieces, cut):
469 self.emit(f"PIECE {size} demand={demand} cut={count}")
470 trim = sum(use * pattern[1] for use, pattern in zip(uses, patterns))
471 surplus = sum(cut) - sum(demand for _, demand in pieces)
472 self.emit(f"TOTAL bars={bars} trim={trim} surplus={surplus} waste={bars * length - quantity}")
473 sizes = [size for size, demand in pieces for _ in range(demand)]
474 comparison = pack(list(enumerate(sizes, 1)), length, "FIRSTFIT")
475 count = len(comparison.bins)
476 self.emit(f"FFD bars={count} waste={count * length - quantity}")
477 self.emit(f"LOWER bars={ceiling(quantity, length)}")

561 def cmd_guillotine(self, width, height, mode, rule, *tokens):
562 width, height = int(width), int(height)
563 if width == 0 or height == 0:
564 raise CommandError("BAD CAPACITY")
565 rectangles = self.rectangles(tokens)
566 rotate = mode == "ROTATE"
567 for number, rectangle_width, rectangle_height in rectangles:
568 fits = rectangle_width <= width and rectangle_height <= height
569 fits_rotated = rotate and rectangle_height <= width and rectangle_width <= height
570 if rectangle_width == 0 or rectangle_height == 0 or not (fits or fits_rotated):
571 raise CommandError(f"BAD RECT {number}")
572 rectangles.sort(key=lambda rectangle: (-rectangle[1] * rectangle[2], rectangle[0]))
573 sheets = []
574 rotated_count = 0
575 for number, rectangle_width, rectangle_height in rectangles:
576 chosen = self.free_rectangle(sheets, rectangle_width, rectangle_height, rotate, rule)
577 if chosen is None:
578 chosen = len(sheets), 0
579 sheets.append(Sheet(width, height))
580 sheet_number, index = chosen
581 sheet = sheets[sheet_number]
582 x, y, free_width, free_height = sheet.free.pop(index)
583 rotated = not (rectangle_width <= free_width and rectangle_height <= free_height)
584 placed_width, placed_height = (rectangle_height, rectangle_width) if rotated else (rectangle_width, rectangle_height)
585 rotated_count += rotated
586 sheet.placements.append((number, x, y, placed_width, placed_height, rotated))
587 sheet.area += rectangle_width * rectangle_height
588 right_width = free_width - placed_width
589 top_height = free_height - placed_height
590 if right_width <= top_height:
591 children = (
592 (x, y + placed_height, free_width, top_height),
593 (x + placed_width, y, right_width, placed_height),
594 )
595 else:
596 children = (
597 (x, y + placed_height, placed_width, top_height),
598 (x + placed_width, y, right_width, free_height),
599 )
600 sheet.free.extend(rectangle for rectangle in children if rectangle[2] and rectangle[3])
601 area = sum(sheet.area for sheet in sheets)
602 self.emit(
603 f"GUILLOTINE W={width} H={height} mode={mode} rule={rule} rects={len(rectangles)}",
604 f"sheets={len(sheets)} rotated={rotated_count} util={area * 100 // (len(sheets) * width * height)}",
605 )
606 for number, sheet in enumerate(sheets, 1):
607 self.emit(f"SHEET {number} placed={len(sheet.placements)} area={sheet.area}")
608 for rectangle, x, y, placed_width, placed_height, rotated in sheet.placements:
609 text = f"PLACE {chr(64 + rectangle)} x={x} y={y} w={placed_width} h={placed_height}"
610 self.emit(text + (" rotated" if rotated else ""))
611 free = " ".join(f"{x},{y},{free_width}x{free_height}" for x, y, free_width, free_height in sheet.free) or "none"
612 self.emit("FREE", free)
613 self.drawing(width, height, sheet.placements)

615 def execute(self, line):
616 command, *arguments = line.split()
617 try:
618 if command not in ("ITEMS", "RANDOM", "CUTSTOCK", "SHELF", "GUILLOTINE") and self.items is None:
619 raise CommandError("NO ITEMS")
620 if command in METHODS:
621 self.run_method(command, arguments)
622 else:
623 getattr(self, "cmd_" + command.lower())(*arguments)
624 except CommandError as error:
625 self.emit("ERROR", str(error))
