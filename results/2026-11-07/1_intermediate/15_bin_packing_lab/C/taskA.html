92static int compare_items(const void *left, const void *right) {
93 const Item *a = left, *b = right;
94 if (a->size != b->size) return a->size > b->size ? -1 : 1;
95 return (a->number > b->number) - (a->number < b->number);
96}

98static void descending(Item *target, const Item *source, int count) {
99 memcpy(target, source, sizeof(Item) * count);
100 qsort(target, count, sizeof(Item), compare_items);
101}

129static Packing pack(const Item *list, int count, int cap, int method, int k) {
130 Packing result = {0};
131 result.method = method;
132 result.parameter = k;
133 result.bins = calloc(MAX_BINS, sizeof(Bin));
134 result.steps = calloc(MAX_ITEMS, sizeof(Step));
135 Item order[MAX_ITEMS];
136 if (method == FFD || method == BFD || method == WFD) {
137 descending(order, list, count);
138 } else {
139 memcpy(order, list, sizeof(Item) * count);
140 }
141 int active[51] = {0}, active_count = 0;
142 int current[51] = {0};
143 for (int i = 0; i <= 50; ++i) current[i] = -1;
144
145 for (int i = 0; i < count; ++i) {
146 Item item = order[i];
147 int before = result.inspections, chosen = -1, closed = -1, category = -1;
148
149 if (method == NEXTFIT) {
150 if (result.count) {
151 result.inspections++;
152 int last = result.count - 1;
153 if (result.bins[last].load + item.size <= cap) chosen = last;
154 }
155 } else if (method == FIRSTFIT || method == FFD || method == KBOUNDED) {
156 int candidates = method == KBOUNDED ? active_count : result.count;
157 for (int b = 0; b < candidates; ++b) {
158 int index = method == KBOUNDED ? active[b] : b;
159 result.inspections++;
160 if (result.bins[index].load + item.size <= cap) {
161 chosen = index;
162 break;
163 }
164 }
165 } else if (method == HARMONIC) {
166 category = cap / item.size < k ? cap / item.size : k;
167 int index = current[category];
168 if (index >= 0) {
169 result.inspections++;
170 if (result.bins[index].load + item.size <= cap) chosen = index;
171 }
172 } else {
173 result.inspections += result.count;
174 int fitting[MAX_BINS], fits = 0;
175 for (int b = 0; b < result.count; ++b)
176 if (result.bins[b].load + item.size <= cap) fitting[fits++] = b;
177 if (fits) {
178 chosen = fitting[0];
179 if (method == BESTFIT || method == BFD) {
180 for (int b = 1; b < fits; ++b)
181 if (result.bins[fitting[b]].load > result.bins[chosen].load)
182 chosen = fitting[b];
183 } else {
184 for (int b = 0; b < fits; ++b) {
185 for (int j = b + 1; j < fits; ++j) {
186 int a = fitting[b], c = fitting[j];
187 if (result.bins[c].load < result.bins[a].load ||
188 (result.bins[c].load == result.bins[a].load && c < a)) {
189 fitting[b] = c;
190 fitting[j] = a;
191 }
192 }
193 }
194 chosen = method == ALMOSTWORST && fits >= 2 ? fitting[1] : fitting[0];
195 }
196 }
197 }
198 int opened = chosen < 0;
199 if (opened) {
200 if (method == KBOUNDED && active_count == k) {
201 int position = 0;
202 for (int j = 1; j < active_count; ++j) {
203 int a = active[j], b = active[position];
204 if (result.bins[a].load > result.bins[b].load ||
205 (result.bins[a].load == result.bins[b].load && a < b)) position = j;
206 }
207 closed = active[position];
208 memmove(&active[position], &active[position + 1],
209 (active_count - position - 1) * sizeof(int));
210 active_count--;
211 result.closes++;
212 }
213 chosen = result.count++;
214 result.bins[chosen].category = category;
215 if (method == KBOUNDED) active[active_count++] = chosen;
216 if (method == HARMONIC) current[category] = chosen;
217 }
218 put(&result.bins[chosen], item);
219 result.steps[result.steps_count++] =
220 (Step){item, chosen, result.inspections - before, opened, closed, category};
221 }
222 return result;
223}

243static Bounds calculate_bounds(const Item *list, int count, int cap) {
244 Bounds result = {0};
245 int alpha[41], alpha_count = 1;
246 alpha[0] = 0;
247 for (int i = 0; i < count; ++i) {
248 if (2 * list[i].size > cap) continue;
249 int exists = 0;
250 for (int j = 0; j < alpha_count; ++j)
251 if (alpha[j] == list[i].size) exists = 1;
252 if (!exists) alpha[alpha_count++] = list[i].size;
253 }
254 for (int i = 0; i < alpha_count; ++i)
255 for (int j = i + 1; j < alpha_count; ++j)
256 if (alpha[j] < alpha[i]) {
257 int tmp = alpha[i]; alpha[i] = alpha[j]; alpha[j] = tmp;
258 }
259 result.first = ceiling(total_size(list, count), cap);
260 for (int i = 0; i < alpha_count; ++i) {
261 int j1 = 0, j2 = 0, j3 = 0, second = 0, third = 0;
262 for (int j = 0; j < count; ++j) {
263 int size = list[j].size;
264 if (size > cap - alpha[i]) j1++;
265 else if (2 * size > cap) { j2++; second += size; }
266 if (2 * size <= cap && size >= alpha[i]) { j3++; third += size; }
267 }
268 int excess = third - (j2 * cap - second);
269 int bound = j1 + j2 + (excess > 0 ? ceiling(excess, cap) : 0);
270 result.rows[result.count++] = (BoundRow){alpha[i], j1, j2, j3, bound};
271 if (bound > result.second) result.second = bound;
272 }
273 return result;
274}

284static void visit(Exact *search, int index) {
285 if (search->status) return;
286 if (search->nodes == search->limit) {
287 search->status = 2;
288 return;
289 }
290 search->nodes++;
291 int count = search->live_count;
292 int used = search->total - search->remaining[index];
293 int free_space = count * search->capacity - used;
294 int excess = search->remaining[index] - free_space;
295 int minimum = count +
296 (excess > 0 ? ceiling(excess, search->capacity) : 0);
297 if (minimum >= search->best_count) return;
298 if (index == search->size) {
299 search->best_count = count;
300 memcpy(search->best, search->live, count * sizeof(Bin));
301 printf("IMPROVE bins=%d nodes=%d\n", count, search->nodes);
302 if (count == search->lower) search->status = 1;
303 return;
304 }
305 Item item = search->sorted[index];
306 int seen[21], seen_count = 0;
307 for (int b = 0; b < search->live_count; ++b) {
308 Bin *bin = &search->live[b];
309 int old = bin->load, repeated = 0;
310 for (int j = 0; j < seen_count; ++j)
311 if (seen[j] == old) repeated = 1;
312 if (repeated) continue;
313 seen[seen_count++] = old;
314 if (old + item.size > search->capacity) continue;
315 put(bin, item);
316 visit(search, index + 1);
317 bin->count--;
318 bin->load = old;
319 if (search->status) return;
320 }
321 if (count + 1 < search->best_count) {
322 Bin *bin = &search->live[search->live_count++];
323 memset(bin, 0, sizeof(Bin));
324 put(bin, item);
325 visit(search, index + 1);
326 search->live_count--;
327 }
328}

330static void run_exact(int limit) {
331 if (limit == 0) { puts("ERROR BAD PARAMETER"); return; }
332 if (item_count >= 21) { puts("ERROR TOO MANY ITEMS"); return; }
333 Packing ffd = pack(items, item_count, capacity, FFD, 0);
334 Bounds lower = calculate_bounds(items, item_count, capacity);
335 printf("EXACT ffd=%d L2=%d\n", ffd.count, lower.second);
336 Exact *search = calloc(1, sizeof(Exact));
337 search->size = item_count;
338 search->capacity = capacity;
339 search->limit = limit;
340 search->lower = lower.second;
341 search->best_count = ffd.count;
342 search->total = total_size(items, item_count);
343 memcpy(search->best, ffd.bins, ffd.count * sizeof(Bin));
344 descending(search->sorted, items, item_count);
345 for (int i = item_count - 1; i >= 0; --i)
346 search->remaining[i] = search->remaining[i + 1] + search->sorted[i].size;
347 if (ffd.count == lower.second) search->status = 1;
348 else visit(search, 0);
349 printf("RESULT bins=%d nodes=%d status=%s\n", search->best_count,
350 search->nodes, search->status == 1 ? "bound" :
351 search->status == 2 ? "limit" : "complete");
352 print_bins(search->best, search->best_count, capacity, 0);
353 free(search);
354 dispose(&ffd);
355}

388static void run_reduce(int method, int k) {
389 if (is_parameterized(method) && !k) { puts("ERROR BAD PARAMETER"); return; }
390 Packing result = pack(items, item_count, capacity, method, k);
391 int before = result.count, event_count = 0;
392 Elimination events[40];
393 for (;;) {
394 int eliminated = 0;
395 int order[40];
396 for (int i = 0; i < result.count; ++i) order[i] = i;
397 for (int i = 0; i < result.count; ++i)
398 for (int j = i + 1; j < result.count; ++j) {
399 int a = order[i], b = order[j];
400 if (result.bins[b].load < result.bins[a].load ||
401 (result.bins[b].load == result.bins[a].load && b < a)) {
402 order[i] = b; order[j] = a;
403 }
404 }
405 for (int index = 0; index < result.count; ++index) {
406 int source = order[index], loads[40], count = 0, valid = 1;
407 Item sorted[40];
408 int destinations[40];
409 for (int b = 0; b < result.count; ++b) loads[b] = result.bins[b].load;
410 descending(sorted, result.bins[source].items, result.bins[source].count);
411 for (int j = 0; j < result.bins[source].count; ++j) {
412 int dest = -1;
413 for (int b = 0; b < result.count; ++b) {
414 if (b == source || loads[b] + sorted[j].size > capacity) continue;
415 if (dest < 0 || loads[b] > loads[dest] ||
416 (loads[b] == loads[dest] && b < dest)) dest = b;
417 }
418 if (dest < 0) { valid = 0; break; }
419 loads[dest] += sorted[j].size;
420 destinations[count++] = dest;
421 }
422 if (!valid) continue;
423 Elimination *event = &events[event_count++];
424 event->source = source;
425 event->load = result.bins[source].load;
426 event->count = count;
427 for (int j = 0; j < count; ++j) {
428 event->item[j] = sorted[j];
429 event->destination[j] = destinations[j];
430 put(&result.bins[destinations[j]], sorted[j]);
431 }
432 memmove(&result.bins[source], &result.bins[source + 1],
433 (result.count - source - 1) * sizeof(Bin));
434 result.count--;
435 eliminated = 1;
436 break;
437 }
438 if (!eliminated) break;
439 }
440 printf("REDUCE ");
441 print_method_name(method, k);
442 printf(" before=%d after=%d eliminated=%d\n",
443 before, result.count, before - result.count);
444 for (int i = 0; i < event_count; ++i) {
445 Elimination *event = &events[i];
446 printf("ELIMINATE bin=%d load=%d moves=", event->source + 1, event->load);
447 for (int j = 0; j < event->count; ++j) {
448 if (j) putchar(' ');
449 printf("%d:%d->%d", event->item[j].number,
450 event->item[j].size, event->destination[j] + 1);
451 }
452 putchar('\n');
453 }
454 print_bins(result.bins, result.count, capacity, method == HARMONIC);
455 dispose(&result);
456}

473static void run_compare(int k) {
474 if (!k) { puts("ERROR BAD PARAMETER"); return; }
475 Bounds bound = calculate_bounds(items, item_count, capacity);
476 Comparison rows[10];
477 for (int method = 0; method < 10; ++method) {
478 Packing result = pack(items, item_count, capacity, method, k);
479 rows[method] = (Comparison){method, result.count, result.inspections};
480 dispose(&result);
481 }
482 for (int i = 0; i < 10; ++i)
483 for (int j = i + 1; j < 10; ++j)
484 if (rows[j].bins < rows[i].bins ||
485 (rows[j].bins == rows[i].bins &&
486 strcmp(METHODS[rows[j].method], METHODS[rows[i].method]) < 0)) {
487 Comparison temp = rows[i]; rows[i] = rows[j]; rows[j] = temp;
488 }
489 printf("COMPARE k=%d L1=%d L2=%d\n", k, bound.first, bound.second);
490 int total = total_size(items, item_count);
491 for (int i = 0; i < 10; ++i)
492 printf("%s bins=%d gap=%d waste=%d inspections=%d\n",
493 METHODS[rows[i].method], rows[i].bins, rows[i].bins - bound.second,
494 rows[i].bins * capacity - total, rows[i].inspections);
495}

497static void run_check(char **args, int count) {
498 if (count != item_count) { puts("ERROR BAD ASSIGNMENT"); return; }
499 int used[41] = {0};
500 for (int i = 0; i < count; ++i) {
501 int label = atoi(args[i]);
502 if (label == 0) { puts("ERROR BAD ASSIGNMENT"); return; }
503 used[label] = 1;
504 }
505 Bin *bins = calloc(41, sizeof(Bin));
506 int bin_count = 0, over = 0;
507 for (int i = 0; i < count; ++i)
508 put(&bins[atoi(args[i])], items[i]);
509 for (int i = 1; i <= 40; ++i) {
510 if (!used[i]) continue;
511 bin_count++;
512 if (bins[i].load > capacity) over++;
513 }
514 Bounds bound = calculate_bounds(items, item_count, capacity);
515 printf("CHECK bins=%d over=%d L2=%d status=%s\n",
516 bin_count, over, bound.second, over ? "invalid" : "valid");
517 for (int i = 1; i <= 40; ++i)
518 if (used[i]) print_bin(i, &bins[i], capacity, 0, bins[i].load > capacity);
519 free(bins);
520}

545 printf("COVER total=%d upper=%d\n",
546 total_size(items, item_count), total_size(items, item_count) / capacity);
547 Bin *bins = calloc(40, sizeof(Bin));
548 int count = 0;
549 for (int i = 0; i < item_count; ++i) {

551 put(&bins[count - 1], items[i]);

554 memset(bins, 0, 40 * sizeof(Bin));
555 Item order[40];
556 descending(order, items, item_count);

562 while (bin->load < capacity && left <= right) put(bin, order[right--]);
563 }
564 print_cover("BIGSMALL", bins, count);
565 free(bins);
566}

580static void run_multifit(int count) {
581 if (!count) { puts("ERROR BAD PARAMETER"); return; }
582 int total = total_size(items, item_count), largest = 0;
583 for (int i = 0; i < item_count; ++i)
584 if (items[i].size > largest) largest = items[i].size;
585 int lo = ceiling(total, count), hi = ceiling(2 * total, count);
586 if (lo < largest) lo = largest;
587 if (hi < largest) hi = largest;
588 printf("MULTIFIT m=%d lower=%d upper=%d\n", count, lo, hi);
589 while (lo < hi) {
590 int mid = (lo + hi) / 2;
591 Packing result = pack(items, item_count, mid, FFD, 0);
592 int bins = result.count, ok = bins <= count;
593 printf("TRY cap=%d bins=%d %s\n", mid, bins, ok ? "ok" : "fail");
594 dispose(&result);
595 if (ok) hi = mid;
596 else lo = mid + 1;
597 }
598 Packing result = pack(items, item_count, hi, FFD, 0);
599 int makespan = 0;
600 Bin *machines = calloc(count, sizeof(Bin));
601 for (int i = 0; i < result.count; ++i)
602 machines[i] = result.bins[i];
603 for (int i = 0; i < count; ++i)
604 if (machines[i].load > makespan) makespan = machines[i].load;
605 printf("RESULT cap=%d makespan=%d\n", hi, makespan);
606 print_machines(machines, count);
607 memset(machines, 0, count * sizeof(Bin));
608 Item sorted[40];
609 descending(sorted, items, item_count);
610 for (int i = 0; i < item_count; ++i) {
611 int chosen = 0;
612 for (int j = 1; j < count; ++j)
613 if (machines[j].load < machines[chosen].load) chosen = j;
614 put(&machines[chosen], sorted[i]);
615 }
616 makespan = 0;
617 for (int i = 0; i < count; ++i)
618 if (machines[i].load > makespan) makespan = machines[i].load;
619 printf("LPT makespan=%d\n", makespan);
620 print_machines(machines, count);
621 free(machines);
622 dispose(&result);
623}

628static Piece pieces[6];
629static Pattern patterns[MAX_PATTERNS];
630static int piece_count, pattern_count;
631
632static void enumerate_patterns(int index, int remaining, int *prefix) {
633 int maximum = remaining / pieces[index].size;
634 if (index == piece_count - 1) {
635 Pattern *pattern = &patterns[pattern_count++];
636 memcpy(pattern->counts, prefix, index * sizeof(int));
637 pattern->counts[index] = maximum;
638 pattern->trim = remaining - maximum * pieces[index].size;
639 return;
640 }
641 for (int count = maximum; count >= 0; --count) {
642 prefix[index] = count;
643 enumerate_patterns(index + 1, remaining - count * pieces[index].size, prefix);
644 }
645}

647static int compare_pieces(const void *left, const void *right) {
648 const Piece *a = left, *b = right;
649 return (a->size < b->size) - (a->size > b->size);
650}

652static void run_cutstock(char **args, int count) {
653 int cap = atoi(args[0]);
654 if (!cap) { puts("ERROR BAD CAPACITY"); return; }
655 piece_count = count - 1;
656 for (int i = 0; i < piece_count; ++i) {
657 int size = 0, demand = 0;
658 sscanf(args[i + 1], "%d:%d", &size, &demand);
659 if (!size || size > cap || !demand) {
660 printf("ERROR BAD PIECE %d\n", i + 1); return;
661 }
662 for (int j = 0; j < i; ++j)
663 if (pieces[j].size == size) {
664 printf("ERROR DUPLICATE PIECE %d\n", i + 1); return;
665 }
666 pieces[i] = (Piece){size, demand};
667 }
668 qsort(pieces, piece_count, sizeof(Piece), compare_pieces);
669 pattern_count = 0;
670 int prefix[6] = {0};
671 enumerate_patterns(0, cap, prefix);
672 int remaining[6], uses[MAX_PATTERNS] = {0}, cut[6] = {0};
673 int quantity = 0, bars = 0, trim = 0, surplus = 0;
674 for (int i = 0; i < piece_count; ++i) {
675 remaining[i] = pieces[i].demand;
676 quantity += pieces[i].size * pieces[i].demand;
677 }
678 for (;;) {
679 int needed = 0, chosen = 0, best = -1;
680 for (int j = 0; j < piece_count; ++j)
681 if (remaining[j]) needed = 1;
682 if (!needed) break;
683 for (int p = 0; p < pattern_count; ++p) {
684 int useful = 0;
685 for (int j = 0; j < piece_count; ++j) {
686 int value = patterns[p].counts[j];
687 useful += value < remaining[j] ? value : remaining[j];
688 }
689 if (useful > best) { best = useful; chosen = p; }
690 }
691 uses[chosen]++;
692 for (int j = 0; j < piece_count; ++j) {
693 remaining[j] -= patterns[chosen].counts[j];
694 if (remaining[j] < 0) remaining[j] = 0;
695 }
696 }
697 printf("CUTSTOCK L=%d pieces=%d patterns=%d\n", cap, piece_count, pattern_count);
698 for (int p = 0; p < pattern_count; ++p) {
699 printf("PATTERN %d ", p + 1);
700 int printed = 0;
701 for (int j = 0; j < piece_count; ++j) {
702 int n = patterns[p].counts[j];
703 if (n) {
704 if (printed++) putchar('+');
705 printf("%dx%d", n, pieces[j].size);
706 }
707 cut[j] += uses[p] * n;
708 }
709 printf(" rest=%d\n", patterns[p].trim);
710 if (uses[p]) { bars += uses[p]; trim += uses[p] * patterns[p].trim; }
711 }
712 for (int p = 0; p < pattern_count; ++p)
713 if (uses[p]) printf("USE %d bars=%d\n", p + 1, uses[p]);
714 for (int j = 0; j < piece_count; ++j) {
715 printf("PIECE %d demand=%d cut=%d\n",
716 pieces[j].size, pieces[j].demand, cut[j]);
717 surplus += cut[j] - pieces[j].demand;
718 }
719 printf("TOTAL bars=%d trim=%d surplus=%d waste=%d\n",
720 bars, trim, surplus, bars * cap - quantity);
721 Item comparison[MAX_ITEMS];
722 int k = 0;
723 for (int j = 0; j < piece_count; ++j)
724 for (int n = 0; n < pieces[j].demand; ++n)
725 {
726 comparison[k] = (Item){k + 1, pieces[j].size};
727 k++;
728 }
729 Packing firstfit = pack(comparison, k, cap, FIRSTFIT, 0);
730 printf("FFD bars=%d waste=%d\n", firstfit.count, firstfit.count * cap - quantity);
731 printf("LOWER bars=%d\n", ceiling(quantity, cap));
732 dispose(&firstfit);
733}

735static int compare_shelf_rects(const void *left, const void *right) {
736 const Rectangle *a = left, *b = right;
737 if (a->height != b->height) return a->height > b->height ? -1 : 1;
738 if (a->width != b->width) return a->width > b->width ? -1 : 1;
739 return a->number - b->number;
740}
741
742static int compare_area_rects(const void *left, const void *right) {
743 const Rectangle *a = left, *b = right;
744 int area_a = a->width * a->height, area_b = b->width * b->height;
745 if (area_a != area_b) return area_a > area_b ? -1 : 1;
746 return a->number - b->number;
747}

749static void draw(int width, int height, const Placement *places, int count) {
750 char grid[400][41];
751 for (int y = 0; y < height; ++y) {
752 memset(grid[y], '.', width);
753 grid[y][width] = '\0';
754 }
755 for (int i = 0; i < count; ++i) {
756 Placement p = places[i];
757 for (int y = p.y; y < p.y + p.height; ++y)
758 for (int x = p.x; x < p.x + p.width; ++x)
759 grid[y][x] = 'A' + p.rectangle.number - 1;
760 }
761 for (int y = height - 1; y >= 0; --y) puts(grid[y]);
762}

764static void run_shelf_method(int method, int width, Rectangle *rects, int count, int area) {
765 const char *names[] = {"NFDH", "FFDH", "BFDH"};
766 Shelf shelves[MAX_RECTS] = {0};
767 Placement placed[MAX_RECTS];
768 int shelf_count = 0;
769 for (int i = 0; i < count; ++i) {
770 Rectangle rect = rects[i];
771 int chosen = -1;
772 if (method == 0) {
773 if (shelf_count && shelves[shelf_count - 1].used + rect.width <= width)
774 chosen = shelf_count - 1;
775 } else {
776 for (int j = 0; j < shelf_count; ++j) {
777 if (shelves[j].used + rect.width > width) continue;
778 if (chosen < 0 || (method == 2 && shelves[j].used > shelves[chosen].used))
779 chosen = j;
780 }
781 }
782 if (chosen < 0) {
783 chosen = shelf_count++;
784 shelves[chosen].bottom = chosen ?
785 shelves[chosen - 1].bottom + shelves[chosen - 1].height : 0;
786 shelves[chosen].height = rect.height;
787 }
788 Shelf *shelf = &shelves[chosen];
789 shelf->item_numbers[shelf->count] = rect.number;
790 shelf->item_x[shelf->count++] = shelf->used;
791 placed[i] = (Placement){rect, shelf->used, shelf->bottom,
792 rect.width, rect.height, 0};
793 shelf->used += rect.width;
794 }
795 int height = shelves[shelf_count - 1].bottom + shelves[shelf_count - 1].height;
796 printf("%s height=%d shelves=%d util=%d\n", names[method],
797 height, shelf_count, area * 100 / (width * height));
798 for (int s = 0; s < shelf_count; ++s) {
799 Shelf *shelf = &shelves[s];
800 printf("LEVEL %d y=%d h=%d used=%d items=",
801 s + 1, shelf->bottom, shelf->height, shelf->used);
802 for (int j = 0; j < shelf->count; ++j) {
803 if (j) putchar(' ');
804 printf("%c@%d", 'A' + shelf->item_numbers[j] - 1, shelf->item_x[j]);
805 }
806 putchar('\n');
807 }
808 draw(width, height, placed, count);
809}

812 for (int i = 0; i < count; ++i) {
813 int width, height;
814 sscanf(args[i], "%dx%d", &width, &height);
815 rects[i] = (Rectangle){i + 1, width, height};
816 }
817 return count;

820static void run_shelf(char **args, int count) {
821 int width = atoi(args[0]);
822 if (!width) { puts("ERROR BAD CAPACITY"); return; }
823 Rectangle rects[MAX_RECTS];
824 int n = read_rectangles(args + 1, count - 1, rects);
825 for (int i = 0; i < n; ++i)
826 if (!rects[i].width || !rects[i].height || rects[i].width > width) {
827 printf("ERROR BAD RECT %d\n", i + 1); return;
828 }
829 qsort(rects, n, sizeof(Rectangle), compare_shelf_rects);
830 int area = 0;
831 for (int i = 0; i < n; ++i) area += rects[i].width * rects[i].height;
832 printf("SHELF W=%d rects=%d area=%d\n", width, n, area);
833 for (int method = 0; method < 3; ++method)
834 run_shelf_method(method, width, rects, n, area);
835
836 int heights[40] = {0}, height = 0;
837 Placement placed[MAX_RECTS];
838 for (int i = 0; i < n; ++i) {
839 Rectangle rect = rects[i];
840 int chosen_x = 0, chosen_y = 2147483647;
841 for (int x = 0; x + rect.width <= width; ++x) {
842 int y = 0;
843 for (int j = x; j < x + rect.width; ++j)
844 if (heights[j] > y) y = heights[j];
845 if (y < chosen_y) { chosen_x = x; chosen_y = y; }
846 }
847 for (int j = chosen_x; j < chosen_x + rect.width; ++j)
848 heights[j] = chosen_y + rect.height;
849 placed[i] = (Placement){rect, chosen_x, chosen_y, rect.width, rect.height, 0};
850 }
851 for (int x = 0; x < width; ++x)
852 if (heights[x] > height) height = heights[x];
853 printf("SKYLINE height=%d util=%d\n", height, area * 100 / (width * height));
854 for (int i = 0; i < n; ++i) {
855 Placement p = placed[i];
856 printf("PLACE %c x=%d y=%d w=%d h=%d\n",
857 'A' + p.rectangle.number - 1, p.x, p.y, p.width, p.height);
858 }
859 draw(width, height, placed, n);
860}

862static int free_rectangle(Sheet *sheets, int count, int width, int height,
863 int rotate, int rule, int *sheet_index, int *free_index) {
864 int best = -1;
865 for (int s = 0; s < count; ++s) {
866 for (int i = 0; i < sheets[s].free_count; ++i) {
867 FreeRect rect = sheets[s].free[i];
868 int fits = width <= rect.width && height <= rect.height;
869 int rotated = rotate && height <= rect.width && width <= rect.height;
870 if (!fits && !rotated) continue;
871 if (rule == 0) {
872 *sheet_index = s; *free_index = i; return 1;
873 }
874 int area = rect.width * rect.height;
875 if (best < 0 || area < best) {
876 best = area;
877 *sheet_index = s; *free_index = i;
878 }
879 }
880 }
881 return best >= 0;
882}

884static void run_guillotine(char **args, int count) {
885 int width = atoi(args[0]), height = atoi(args[1]);
886 if (!width || !height) { puts("ERROR BAD CAPACITY"); return; }
887 int rotate = strcmp(args[2], "ROTATE") == 0;
888 int rule = strcmp(args[3], "FIRST") == 0 ? 0 : 1;
889 Rectangle rects[MAX_RECTS];
890 int n = read_rectangles(args + 4, count - 4, rects);
891 for (int i = 0; i < n; ++i) {
892 Rectangle rect = rects[i];
893 int fits = rect.width <= width && rect.height <= height;
894 int rotated = rotate && rect.height <= width && rect.width <= height;
895 if (!rect.width || !rect.height || (!fits && !rotated)) {
896 printf("ERROR BAD RECT %d\n", i + 1); return;
897 }
898 }
899 qsort(rects, n, sizeof(Rectangle), compare_area_rects);
900 Sheet sheets[MAX_RECTS] = {0};
901 int sheet_count = 0, rotated_count = 0, area = 0;
902 for (int i = 0; i < n; ++i) {
903 Rectangle rect = rects[i];
904 int sheet_index = 0, free_index = 0;
905 if (!free_rectangle(sheets, sheet_count, rect.width, rect.height,
906 rotate, rule, &sheet_index, &free_index)) {
907 sheet_index = sheet_count++;
908 sheets[sheet_index].free[sheets[sheet_index].free_count++] =
909 (FreeRect){0, 0, width, height};
910 }
911 Sheet *sheet = &sheets[sheet_index];
912 FreeRect free_rect = sheet->free[free_index];
913 memmove(&sheet->free[free_index], &sheet->free[free_index + 1],
914 (sheet->free_count - free_index - 1) * sizeof(FreeRect));
915 sheet->free_count--;
916 int rotated = !(rect.width <= free_rect.width && rect.height <= free_rect.height);
917 int pw = rotated ? rect.height : rect.width;
918 int ph = rotated ? rect.width : rect.height;
919 rotated_count += rotated;
920 sheet->placed[sheet->placed_count++] =
921 (Placement){rect, free_rect.x, free_rect.y, pw, ph, rotated};
922 sheet->area += rect.width * rect.height;
923 area += rect.width * rect.height;
924 int rw = free_rect.width - pw, rh = free_rect.height - ph;
925 FreeRect top, right;
926 if (rw <= rh) {
927 top = (FreeRect){free_rect.x, free_rect.y + ph, free_rect.width, rh};
928 right = (FreeRect){free_rect.x + pw, free_rect.y, rw, ph};
929 } else {
930 top = (FreeRect){free_rect.x, free_rect.y + ph, pw, rh};
931 right = (FreeRect){free_rect.x + pw, free_rect.y, rw, free_rect.height};
932 }
933 if (top.width && top.height) sheet->free[sheet->free_count++] = top;
934 if (right.width && right.height) sheet->free[sheet->free_count++] = right;
935 }
936 printf("GUILLOTINE W=%d H=%d mode=%s rule=%s rects=%d sheets=%d "
937 "rotated=%d util=%d\n", width, height, args[2], args[3],
938 n, sheet_count, rotated_count, area * 100 / (sheet_count * width * height));
939 for (int s = 0; s < sheet_count; ++s) {
940 Sheet *sheet = &sheets[s];
941 printf("SHEET %d placed=%d area=%d\n", s + 1, sheet->placed_count, sheet->area);
942 for (int j = 0; j < sheet->placed_count; ++j) {
943 Placement p = sheet->placed[j];
944 printf("PLACE %c x=%d y=%d w=%d h=%d",
945 'A' + p.rectangle.number - 1, p.x, p.y, p.width, p.height);
946 if (p.rotated) printf(" rotated");
947 putchar('\n');
948 }
949 printf("FREE");
950 if (!sheet->free_count) printf(" none");
951 for (int j = 0; j < sheet->free_count; ++j) {
952 FreeRect f = sheet->free[j];
953 printf(" %d,%d,%dx%d", f.x, f.y, f.width, f.height);
954 }
955 putchar('\n');
956 draw(width, height, sheet->placed, sheet->placed_count);
957 }
958}

960static void handle(const char *command, char **args, int count) {
961 if (strcmp(command, "ITEMS") == 0) {
962 int cap = atoi(args[0]);
963 if (!cap) { puts("ERROR BAD CAPACITY"); return; }
964 Item next[40];
965 for (int i = 1; i < count; ++i) {
966 int size = atoi(args[i]);
967 if (!size || size > cap) {
968 printf("ERROR BAD ITEM %d\n", i); return;
969 }
970 next[i - 1] = (Item){i, size};
971 }
972 capacity = cap;
973 item_count = count - 1;
974 memcpy(items, next, item_count * sizeof(Item));
975 printf("ITEMS count=%d capacity=%d total=%d\n",
976 item_count, capacity, total_size(items, item_count));
977 return;
978 }
979 if (strcmp(command, "RANDOM") == 0) {
980 int n = atoi(args[0]), cap = atoi(args[1]);
981 int64_t state = strtoll(args[2], NULL, 10);
982 if (!n || n > 40) { puts("ERROR BAD COUNT"); return; }
983 if (!cap) { puts("ERROR BAD CAPACITY"); return; }
984 if (!state || state >= 2147483647) { puts("ERROR BAD SEED"); return; }
985 Item next[40];
986 for (int i = 0; i < n; ++i) {
987 state = 48271 * state % 2147483647;
988 next[i] = (Item){i + 1, state % cap + 1};
989 }
990 capacity = cap;
991 item_count = n;
992 memcpy(items, next, n * sizeof(Item));
993 printf("RANDOM count=%d capacity=%d total=%d\n", n, cap, total_size(items, n));
994 printf("SIZES");
995 for (int i = 0; i < n; ++i) printf(" %d", items[i].size);
996 putchar('\n');
997 return;
998 }
999 if (strcmp(command, "CUTSTOCK") == 0) { run_cutstock(args, count); return; }
1000 if (strcmp(command, "SHELF") == 0) { run_shelf(args, count); return; }
1001 if (strcmp(command, "GUILLOTINE") == 0) { run_guillotine(args, count); return; }
1002
1003 if (item_count < 0) { puts("ERROR NO ITEMS"); return; }
1004 int method = method_id(command);
1005 if (method >= 0) {
1006 run_method(method, is_parameterized(method) ? atoi(args[0]) : 0);
1007 } else if (strcmp(command, "TRACE") == 0) {
1008 method = method_id(args[0]);
1009 run_trace(method, is_parameterized(method) ? atoi(args[1]) : 0);
1010 } else if (strcmp(command, "REDUCE") == 0) {
1011 method = method_id(args[0]);
1012 run_reduce(method, is_parameterized(method) ? atoi(args[1]) : 0);
1013 } else if (strcmp(command, "BOUNDS") == 0) {
1014 run_bounds();
1015 } else if (strcmp(command, "COMPARE") == 0) {
1016 run_compare(atoi(args[0]));
1017 } else if (strcmp(command, "EXACT") == 0) {
1018 run_exact(atoi(args[0]));
1019 } else if (strcmp(command, "CHECK") == 0) {
1020 run_check(args, count);
1021 } else if (strcmp(command, "COVER") == 0) {
1022 run_cover();
1023 } else if (strcmp(command, "MULTIFIT") == 0) {
1024 run_multifit(atoi(args[0]));
1025 }
1026}

1028int main(void) {
1029 int commands;
1030 if (scanf("%d", &commands) != 1) return 0;
1031 int c;
1032 while ((c = getchar()) != '\n' && c != EOF) {}
1033 char line[8192];
1034 for (int k = 0; k < commands && fgets(line, sizeof(line), stdin); ++k) {
1035 char *args[320];
1036 int count = 0;
1037 char *command = strtok(line, " \r\n");
1038 char *token;
1039 while ((token = strtok(NULL, " \r\n")) != NULL) args[count++] = token;
1040 handle(command, args, count);
1041 }
1042 return 0;
