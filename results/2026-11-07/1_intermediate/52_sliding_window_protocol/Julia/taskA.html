1using Printf
2const KEYS = ["PROTOCOL", "FRAMES", "WINDOW", "SEQBITS", "TIMEOUT", "DELAY", "INTERVAL", "MAXTICKS", "DUPACK", "NAK", "BACKOFF", "CONGESTION", "UNSAFE", "TRACE", "RANDOM", "LOSEDATA", "LOSECTRL", "CORRUPTDATA", "CORRUPTCTRL", "DELAYDATA", "DELAYCTRL", "SNAPSHOT"]
3const LISTKEYS = KEYS[16:end]
4struct SettingsError <: Exception
5 text::String
6end
7bad(s) = throw(SettingsError(s))
8function configure(lines)
9 raw = Dict{String,Vector{String}}()
10 for w in lines
11 key, values = w[1], w[2:end]
12 key in KEYS || bad("unknown key $key")
13 haskey(raw, key) && bad("duplicate key $key")
14 count = key == "RANDOM" ? length(values) == 3 : key in LISTKEYS ? !isempty(values) : length(values) == 1
15 count || bad("wrong number of values for $key")
16 raw[key] = values
17 end
18 for key in KEYS[1:6]
19 haskey(raw, key) || bad("missing key $key")
20 end
21 ranges = Dict("FRAMES"=>(1,60), "WINDOW"=>(1,32), "SEQBITS"=>(1,6), "TIMEOUT"=>(1,50), "DELAY"=>(1,10), "INTERVAL"=>(0,10), "MAXTICKS"=>(1,2000), "DUPACK"=>(1,10))
22 values = Dict{String,Union{Int,String,Vector{Int},Dict{Int,Int}}}()
23 for key in KEYS
24 haskey(raw, key) || continue
25 v = raw[key]
26 valid = true
27 if haskey(ranges, key)
28 n = parse(Int, v[1])
29 lo, hi = ranges[key]
30 valid = lo <= n <= hi
31 values[key] = n
32 elseif key == "PROTOCOL"
33 valid = v[1] in ("SW", "GBN", "SR")
34 values[key] = v[1]
35 elseif key == "TRACE"
36 valid = v[1] in ("ALL", "SENDER", "RECEIVER", "NONE")
37 values[key] = v[1]
38 elseif key in ("NAK", "BACKOFF", "CONGESTION", "UNSAFE")
39 valid = v[1] in ("ON", "OFF")
40 values[key] = v[1]
41 elseif key == "RANDOM"
42 ns = parse.(Int, v)
43 valid = 1 <= ns[1] <= 2147483646 && ns[2] >= 0 && ns[3] >= 0 && ns[2] + ns[3] <= 100
44 values[key] = ns
45 elseif key in ("DELAYDATA", "DELAYCTRL")
46 delays = Dict{Int,Int}()
47 for text in v
48 i, e = parse.(Int, split(text, ':'))
49 if !(1 <= i <= 9999 && 1 <= e <= 50) || haskey(delays, i)
50 valid = false
51 end
52 delays[i] = e
53 end
54 values[key] = delays
55 else
56 ns = parse.(Int, v)
57 lo, hi = key == "SNAPSHOT" ? (0,2000) : (1,9999)
58 valid = all(n -> lo <= n <= hi, ns) && length(unique(ns)) == length(ns)
59 values[key] = ns
60 end
61 valid || bad("bad value for $key")
62 end
63 p, w, k = values["PROTOCOL"], values["WINDOW"], values["SEQBITS"]
64 p == "SR" && haskey(raw, "DUPACK") && bad("DUPACK not supported for SR")
65 limit = p == "SW" ? 1 : p == "GBN" ? 2^k - 1 : 2^(k - 1)
66 unsafe = get(values, "UNSAFE", "OFF") == "ON" && p != "SW"
67 w > limit && !unsafe && bad("window $w exceeds limit $limit for $p")
68 for (lost, corrupt, kind) in (("LOSEDATA", "CORRUPTDATA", "data"), ("LOSECTRL", "CORRUPTCTRL", "control"))
69 common = intersect(get(values, lost, Int[]), get(values, corrupt, Int[]))
70 isempty(common) || bad("$kind #$(minimum(common)) both lost and corrupted")
71 end
72 defaults = Dict{String,Union{Int,String}}("INTERVAL"=>0, "MAXTICKS"=>1000, "NAK"=>"OFF", "BACKOFF"=>"OFF", "CONGESTION"=>"OFF", "UNSAFE"=>"OFF", "TRACE"=>"ALL")
73 for (key, value) in defaults
74 get!(values, key, value)
75 end
76 for key in ("LOSEDATA", "LOSECTRL", "CORRUPTDATA", "CORRUPTCTRL", "SNAPSHOT")
77 get!(values, key, Int[])
78 end
79 for key in ("DELAYDATA", "DELAYCTRL")
80 get!(values, key, Dict{Int,Int}())
81 end
82 values
83end
84function decimal(x::Int, y::Int, digits::Int)
85 scale = 10^digits
86 value = div(scale * x, y)
87 string(div(value, scale), ".", lpad(string(mod(value, scale)), digits, '0'))
88end
89struct Transit
90 number::Int
91 value::Int
92 kind::String
93 outcome::String
94end
95struct Result
96 name::String
97 protocol::String
98 status::String
99 delivered::Int
100 completion::Int
101 data::Int
102end
103function simulate(name, c)
104 p, n, w, k, T, d = (c[x] for x in KEYS[1:6])
105 M = 2^k
106 g, L = c["INTERVAL"], c["MAXTICKS"]
107 sr = p == "SR"
108 congestion = c["CONGESTION"] == "ON"
109 trace = c["TRACE"]
110 println("config: $p n=$n w=$w k=$k T=$T d=$d")
111 randomtext = haskey(c, "RANDOM") ? join(c["RANDOM"], "/") : "off"
112 duptext = haskey(c, "DUPACK") ? string(c["DUPACK"]) : "off"
113 println("options: interval=$g limit=$L dupack=$duptext nak=$(lowercase(c["NAK"])) backoff=$(lowercase(c["BACKOFF"])) congestion=$(lowercase(c["CONGESTION"])) unsafe=$(lowercase(c["UNSAFE"])) trace=$(lowercase(trace)) random=$randomtext")
114 base = H = next = D = r = A = 0
115 C = T
116 timer = -1
117 cwnd, threshold = 1, w
118 nakmark = false
119 marks = falses(n)
120 timers = fill(-1, n)
121 queue = Int[]
122 buffer = Dict{Int,Int}()
123 tx = zeros(Int, n)
124 first = fill(-1, n)
125 delivered = fill(-1, n)
126 acked = fill(-1, n)
127 dataevents = [Transit[] for _ in 0:(L + 60)]
128 controlevents = [Transit[] for _ in 0:(L + 60)]
129 stats = Dict(key=>0 for key in ("data", "resent", "control", "nak", "timeouts", "fastretx", "lost_data", "lost_ctrl", "corrupt_data", "corrupt_ctrl", "delivered", "errors", "discarded", "max_buffer"))
130 x = haskey(c, "RANDOM") ? c["RANDOM"][1] : 1
131 latency = 0
132 completion = -1
133 tend = L
134 tick = 0
135 function emit(side, text)
136 (side == "ALWAYS" || trace == "ALL" || trace == side) && println("$tick $text")
137 end
138 function outcome(kind, number)
139 result = "normal"
140 if haskey(c, "RANDOM")
141 x = mod(48271x, 2147483647)
142 v = mod(x, 100)
143 loss, corrupt = c["RANDOM"][2:3]
144 result = v < loss ? "LOST" : v < loss + corrupt ? "CORRUPT" : "normal"
145 end
146 key = kind == "data" ? "DATA" : "CTRL"
147 number in c["LOSE" * key] && (result = "LOST")
148 number in c["CORRUPT" * key] && (result = "CORRUPT")
149 result
150 end
151 function control(kind, value)
152 stats["control"] += 1
153 j = stats["control"]
154 kind == "NAK" && (stats["nak"] += 1)
155 emit("RECEIVER", "$kind a=$value #$j")
156 arrival = tick + d + get(c["DELAYCTRL"], j, 0)
157 push!(controlevents[arrival + 1], Transit(j, value, kind, outcome(kind, j)))
158 end
159 function deliver(payload)
160 if payload == r
161 emit("RECEIVER", "DELIVER f=$r")
162 stats["delivered"] += 1
163 delivered[r + 1] = tick
164 latency += tick - first[payload + 1]
165 else
166 emit("RECEIVER", "DELIVER f=$r wrong=$payload")
167 stats["errors"] += 1
168 end
169 r += 1
170 nakmark = false
171 end
172 function discard(seq)
173 stats["discarded"] += 1
174 emit("RECEIVER", "DISCARD seq=$seq")
175 end
176 function goback()
177 next, timer, D = base, -1, 0
178 end
179 function increase(count)
180 congestion || return
181 for _ in 1:count
182 cwnd >= w && continue
183 if cwnd < threshold
184 cwnd += 1
185 else
186 A += 1
187 if A == cwnd
188 cwnd += 1
189 A = 0
190 end
191 end
192 end
193 end
194 function decrease(timeout)
195 congestion || return
196 threshold = max(div(cwnd, 2), 1)
197 cwnd = timeout ? 1 : threshold
198 A = 0
199 end
200 function cwndline(old)
201 congestion && old != (cwnd, threshold) && emit("SENDER", "CWND $cwnd ssthresh=$threshold")
202 end
203 for t in 0:L
204 tick = t
205 for e in sort(dataevents[t + 1]; by=e -> e.number)
206 if e.outcome != "normal"
207 stats[e.outcome == "LOST" ? "lost_data" : "corrupt_data"] += 1
208 emit("RECEIVER", "$(e.outcome) data #$(e.number)")
209 continue
210 end
211 seq = mod(e.value, M)
212 emit("RECEIVER", "RECV seq=$seq #$(e.number)")
213 if !sr
214 if seq == mod(r, M)
215 deliver(e.value)
216 control("ACK", seq)
217 else
218 discard(seq)
219 if c["NAK"] == "ON" && !nakmark
220 control("NAK", mod(r, M))
221 nakmark = true
222 elseif r > 0
223 control("ACK", mod(r - 1, M))
224 end
225 end
226 else
227 o = mod(seq - r, M)
228 if o < w
229 pos = r + o
230 if haskey(buffer, pos)
231 discard(seq)
232 control("ACK", seq)
233 elseif o == 0
234 deliver(e.value)
235 while haskey(buffer, r)
236 deliver(pop!(buffer, r))
237 end
238 control("ACK", seq)
239 else
240 buffer[pos] = e.value
241 stats["max_buffer"] = max(stats["max_buffer"], length(buffer))
242 emit("RECEIVER", "BUFFER f=$pos")
243 control("ACK", seq)
244 if c["NAK"] == "ON" && !nakmark
245 control("NAK", mod(r, M))
246 nakmark = true
247 end
248 end
249 else
250 discard(seq)
251 mod(r - seq, M) <= w && control("ACK", seq)
252 end
253 end
254 end
255 for e in sort(controlevents[t + 1]; by=e -> e.number)
256 kind, value, j = e.kind, e.value, e.number
257 if e.outcome != "normal"
258 stats[e.outcome == "LOST" ? "lost_ctrl" : "corrupt_ctrl"] += 1
259 emit("SENDER", "$(e.outcome) $(lowercase(kind)) #$j")
260 continue
261 end
262 old = (cwnd, threshold)
263 oldbase = base
264 f = base + mod(value - base, M)
265 accepted = false
266 if kind == "ACK"
267 if !sr && f < H
268 accepted = true
269 base = f + 1
270 next = max(next, base)
271 elseif sr && f < H && !marks[f + 1]
272 accepted = true
273 marks[f + 1] = true
274 acked[f + 1] < 0 && (acked[f + 1] = t)
275 timers[f + 1] = -1
276 filter!(!=(f), queue)
277 while base < H && marks[base + 1]
278 base += 1
279 end
280 end
281 elseif !sr && f <= H
282 accepted = true
283 base = f
284 goback()
285 elseif sr && f < H && !marks[f + 1]
286 accepted = true
287 timers[f + 1] = -1
288 f in queue || push!(queue, f)
289 end
290 if base > oldbase
291 C = T
292 !sr && (D = 0)
293 if !sr
294 for a in oldbase:(base - 1)
295 acked[a + 1] < 0 && (acked[a + 1] = t)
296 end
297 end
298 end
299 if accepted
300 newly = sr ? (kind == "ACK" ? 1 : 0) : base - oldbase
301 increase(newly)
302 if !sr && kind == "ACK"
303 timer = base == next ? -1 : t + C
304 end
305 kind == "NAK" && decrease(false)
306 emit("SENDER", "GOT$kind a=$value #$j base=$base")
307 else
308 emit("SENDER", "OLD$kind a=$value #$j")
309 if !sr && kind == "ACK" && haskey(c, "DUPACK") && base >= 1 && value == mod(base - 1, M)
310 D += 1
311 if D == c["DUPACK"]
312 stats["fastretx"] += 1
313 emit("SENDER", "FASTRETX f=$base")
314 goback()
315 decrease(false)
316 end
317 end
318 end
319 cwndline(old)
320 end
321 done = base == n
322 if !done
323 expired = sr ? [f for f in base:(H - 1) if timers[f + 1] == t] : timer == t ? [base] : Int[]
324 for f in expired
325 old = (cwnd, threshold)
326 stats["timeouts"] += 1
327 emit("SENDER", "TIMEOUT f=$f")
328 if sr
329 timers[f + 1] = -1
330 push!(queue, f)
331 else
332 goback()
333 end
334 c["BACKOFF"] == "ON" && (C = min(2C, 8T))
335 decrease(true)
336 cwndline(old)
337 end
338 E = congestion ? min(w, cwnd) : w
339 send = -1
340 if sr
341 if !isempty(queue)
342 send = popfirst!(queue)
343 elseif H < n && H < base + E && H * g <= t
344 send = H
345 H += 1
346 end
347 elseif next < n && next < base + E && (next < H || next * g <= t)
348 send = next
349 H = max(H, next + 1)
350 next += 1
351 end
352 if send >= 0
353 stats["data"] += 1
354 i = stats["data"]
355 word = tx[send + 1] == 0 ? "SEND" : "RESEND"
356 if word == "SEND"
357 first[send + 1] = t
358 else
359 stats["resent"] += 1
360 end
361 tx[send + 1] += 1
362 emit("SENDER", "$word f=$send seq=$(mod(send, M)) #$i")
363 arrival = t + d + get(c["DELAYDATA"], i, 0)
364 push!(dataevents[arrival + 1], Transit(i, send, "data", outcome("data", i)))
365 if sr
366 timers[send + 1] = t + C
367 elseif timer < 0
368 timer = t + C
369 end
370 end
371 end
372 if t in c["SNAPSHOT"]
373 E = congestion ? min(w, cwnd) : w
374 sx, rx = Char[], Char[]
375 for f in 0:(n - 1)
376 acknowledged = sr ? marks[f + 1] : f < base
377 waiting = sr ? f in queue : next <= f < H
378 sent = sr ? f < H : f < next
379 push!(sx, acknowledged ? 'A' : waiting ? 'R' : sent ? 'O' : f < base + E ? 'W' : '.')
380 push!(rx, f < r ? 'D' : sr && haskey(buffer, f) ? 'B' : (sr ? f < r + w : f == r) ? 'W' : '.')
381 end
382 emit("ALWAYS", "SNAPSHOT sender=$(join(sx)) receiver=$(join(rx))")
383 end
384 if done || t == L
385 emit("ALWAYS", done ? "DONE" : "ABORT")
386 completion = done ? t : -1
387 tend = t
388 break
389 end
390 end
391 text(v) = v < 0 ? "-" : string(v)
392 for f in 0:(n - 1)
393 println("frame $f: seq=$(mod(f, M)) tx=$(tx[f + 1]) first=$(text(first[f + 1])) delivered=$(text(delivered[f + 1])) acked=$(text(acked[f + 1]))")
394 end
395 for (prefix, keys) in (("totals", ["data", "resent", "control", "nak", "timeouts", "fastretx"]), ("channel", ["lost_data", "lost_ctrl", "corrupt_data", "corrupt_ctrl"]), ("receiver", ["delivered", "errors", "discarded", "max_buffer"]))
396 println("$prefix: ", join(("$key=$(stats[key])" for key in keys), " "))
397 end
398 Dgood, X = stats["delivered"], stats["data"]
399 goodput = completion < 0 ? "-" : decimal(Dgood, completion, 2)
400 lag = Dgood == 0 ? "-" : decimal(latency, Dgood, 2)
401 println("result: completion=$(text(completion)) goodput=$goodput efficiency=$(decimal(100Dgood, X, 1))% utilization=$(decimal(100X, tend + 1, 1))% latency=$lag")
402 Result(name, p, completion < 0 ? "aborted" : "done", Dgood, completion, X)
403end
404function main()
405 q = parse(Int, readline())
406 results = Result[]
407 for _ in 1:q
408 name = String(split(readline())[2])
409 lines = Vector{String}[]
410 while true
411 line = readline()
412 line == "END" && break
413 push!(lines, String.(split(line)))
414 end
415 println("== $name ==")
416 try
417 c = configure(lines)
418 push!(results, simulate(name, c))
419 catch e
420 e isa SettingsError || rethrow()
421 println("error: ", e.text)
422 push!(results, Result(name, "", "invalid", 0, -1, 0))
423 end
424 end
425 println("== ranking ==")
426 completed = filter(r -> r.status == "done", results)
427 function better(a, b)
428 left, right = a.delivered * b.completion, b.delivered * a.completion
429 left != right && return left > right
430 a.delivered * b.data > b.delivered * a.data
431 end
432 sort!(completed; lt=better, alg=MergeSort)
433 for (i, r) in enumerate(completed)
434 println("$i. $(r.name) $(r.protocol) goodput=$(decimal(r.delivered, r.completion, 2)) efficiency=$(decimal(100r.delivered, r.data, 1))%")
435 end
436 for r in results
437 r.status != "done" && println("- $(r.name) $(r.status)")
438 end
439 println("== protocols ==")
440 for p in ("SW", "GBN", "SR")
441 runs = filter(r -> r.protocol == p, completed)
442 isempty(runs) && continue
443 D, T, X = sum(r.delivered for r in runs), sum(r.completion for r in runs), sum(r.data for r in runs)
444 println("$p: runs=$(length(runs)) goodput=$(decimal(D, T, 2)) efficiency=$(decimal(100D, X, 1))%")
445 end
446end
447main()
