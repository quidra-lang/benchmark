3emit <- function(...) cat(paste0(...), "\n", sep = "")
4bad <- function(text) stop(structure(list(message = text), class = c("settings_error", "error", "condition")))
5configure <- function(lines) {
6 raw <- list()
7 for (words in lines) {
8 key <- words[1]
9 values <- words[-1]
10 if (!(key %in% keys)) bad(paste("unknown key", key))
11 if (key %in% names(raw)) bad(paste("duplicate key", key))
12 count <- if (key == "RANDOM") length(values) == 3 else if (key %in% listkeys) length(values) >= 1 else length(values) == 1
13 if (!count) bad(paste("wrong number of values for", key))
14 raw[[key]] <- values
15 }
16 for (key in keys[1:6]) if (!(key %in% names(raw))) bad(paste("missing key", key))
17 ranges <- list(FRAMES = c(1, 60), WINDOW = c(1, 32), SEQBITS = c(1, 6), TIMEOUT = c(1, 50), DELAY = c(1, 10), INTERVAL = c(0, 10), MAXTICKS = c(1, 2000), DUPACK = c(1, 10))
18 values <- list()
19 for (key in keys) {
20 if (!(key %in% names(raw))) next
21 v <- raw[[key]]
22 valid <- TRUE
23 if (key %in% names(ranges)) {
24 n <- as.numeric(v[1])
25 valid <- n >= ranges[[key]][1] && n <= ranges[[key]][2]
26 values[[key]] <- n
27 } else if (key == "PROTOCOL") {
28 valid <- v[1] %in% c("SW", "GBN", "SR")
29 values[[key]] <- v[1]
30 } else if (key == "TRACE") {
31 valid <- v[1] %in% c("ALL", "SENDER", "RECEIVER", "NONE")
32 values[[key]] <- v[1]
33 } else if (key %in% c("NAK", "BACKOFF", "CONGESTION", "UNSAFE")) {
34 valid <- v[1] %in% c("ON", "OFF")
35 values[[key]] <- v[1]
36 } else if (key == "RANDOM") {
37 ns <- as.numeric(v)
38 valid <- ns[1] >= 1 && ns[1] <= 2147483646 && ns[2] >= 0 && ns[3] >= 0 && ns[2] + ns[3] <= 100
39 values[[key]] <- ns
40 } else if (key %in% c("DELAYDATA", "DELAYCTRL")) {
41 pairs <- lapply(strsplit(v, ":", fixed = TRUE), as.numeric)
42 ids <- vapply(pairs, function(pair) pair[1], numeric(1))
43 delays <- vapply(pairs, function(pair) pair[2], numeric(1))
44 valid <- all(ids >= 1 & ids <= 9999 & delays >= 1 & delays <= 50) && !anyDuplicated(ids)
45 values[[key]] <- setNames(delays, as.character(ids))
46 } else {
47 ns <- as.numeric(v)
48 lo <- if (key == "SNAPSHOT") 0 else 1
49 hi <- if (key == "SNAPSHOT") 2000 else 9999
50 valid <- all(ns >= lo & ns <= hi) && !anyDuplicated(ns)
51 values[[key]] <- ns
52 }
53 if (!valid) bad(paste("bad value for", key))
54 }
55 p <- values$PROTOCOL
56 w <- values$WINDOW
57 k <- values$SEQBITS
58 if (p == "SR" && "DUPACK" %in% names(raw)) bad("DUPACK not supported for SR")
59 limit <- if (p == "SW") 1 else if (p == "GBN") 2^k - 1 else 2^(k - 1)
60 unsafe <- identical(values$UNSAFE, "ON") && p != "SW"
61 if (w > limit && !unsafe) bad(paste("window", w, "exceeds limit", limit, "for", p))
62 for (kind in c("data", "control")) {
63 lost <- if (kind == "data") "LOSEDATA" else "LOSECTRL"
64 corrupt <- if (kind == "data") "CORRUPTDATA" else "CORRUPTCTRL"
65 common <- intersect(values[[lost]], values[[corrupt]])
66 if (length(common)) bad(paste0(kind, " #", min(common), " both lost and corrupted"))
67 }
68 defaults <- list(INTERVAL = 0, MAXTICKS = 1000, NAK = "OFF", BACKOFF = "OFF", CONGESTION = "OFF", UNSAFE = "OFF", TRACE = "ALL")
69 for (key in names(defaults)) if (!(key %in% names(values))) values[[key]] <- defaults[[key]]
70 for (key in listkeys) if (!(key %in% names(values))) values[[key]] <- numeric()
71 values
72}
73decimal <- function(x, y, digits) {
74 scale <- 10^digits
75 value <- floor(scale * x / y)
76 paste0(format(value %/% scale, scientific = FALSE, trim = TRUE), ".", sprintf(paste0("%0", digits, "d"), as.integer(value %% scale)))
77}
78simulate <- function(name, c) {
79 p <- c$PROTOCOL
80 n <- c$FRAMES
81 w <- c$WINDOW
82 k <- c$SEQBITS
83 T <- c$TIMEOUT
84 d <- c$DELAY
85 g <- c$INTERVAL
86 L <- c$MAXTICKS
87 M <- 2^k
88 sr <- p == "SR"
89 congestion <- c$CONGESTION == "ON"
90 emit("config: ", p, " n=", n, " w=", w, " k=", k, " T=", T, " d=", d)
91 randomtext <- if (is.null(c$RANDOM)) "off" else paste(c$RANDOM, collapse = "/")
92 duptext <- if (is.null(c$DUPACK)) "off" else as.character(c$DUPACK)
93 emit("options: interval=", g, " limit=", L, " dupack=", duptext, " nak=", tolower(c$NAK), " backoff=", tolower(c$BACKOFF),
94 " congestion=", tolower(c$CONGESTION), " unsafe=", tolower(c$UNSAFE), " trace=", tolower(c$TRACE), " random=", randomtext)
95 s <- new.env(parent = emptyenv())
96 s$base <- s$H <- s$nextframe <- s$D <- s$r <- s$A <- 0
97 s$C <- T
98 s$timer <- -1
99 s$cwnd <- 1
100 s$threshold <- w
101 s$nakmark <- FALSE
102 s$marks <- rep(FALSE, n)
103 s$timers <- rep(-1, n)
104 s$queue <- numeric()
105 s$buffer <- list()
106 s$tx <- rep(0, n)
107 s$first <- s$delivered <- s$acked <- rep(-1, n)
108 s$dataevents <- s$controlevents <- vector("list", L + 61)
109 s$stats <- setNames(rep(0, 14), c("data", "resent", "control", "nak", "timeouts", "fastretx", "lost_data", "lost_ctrl", "corrupt_data", "corrupt_ctrl", "delivered", "errors", "discarded", "max_buffer"))
110 s$x <- if (is.null(c$RANDOM)) 1 else c$RANDOM[1]
111 s$latency <- 0
112 s$tick <- 0
113 completion <- -1
114 tend <- L
115 line <- function(side, text) {
116 if (side == "ALWAYS" || c$TRACE == "ALL" || c$TRACE == side) emit(s$tick, " ", text)
117 }
118 outcome <- function(kind, number) {
119 result <- "normal"
120 if (!is.null(c$RANDOM)) {
121 s$x <- (48271 * s$x) %% 2147483647
122 v <- s$x %% 100
123 loss <- c$RANDOM[2]
124 corrupt <- c$RANDOM[3]
125 result <- if (v < loss) "LOST" else if (v < loss + corrupt) "CORRUPT" else "normal"
126 }
127 key <- if (kind == "data") "DATA" else "CTRL"
128 if (number %in% c[[paste0("LOSE", key)]]) result <- "LOST"
129 if (number %in% c[[paste0("CORRUPT", key)]]) result <- "CORRUPT"
130 result
131 }
132 extra_delay <- function(key, number) {
133 value <- c[[key]][as.character(number)]
134 if (!length(value) || is.na(value)) 0 else unname(value)
135 }
136 control <- function(kind, value) {
137 s$stats["control"] <- s$stats["control"] + 1
138 j <- unname(s$stats["control"])
139 if (kind == "NAK") s$stats["nak"] <- s$stats["nak"] + 1
140 line("RECEIVER", paste0(kind, " a=", value, " #", j))
141 arrival <- s$tick + d + extra_delay("DELAYCTRL", j)
142 event <- list(number = j, value = value, kind = kind, outcome = outcome(kind, j))
143 s$controlevents[[arrival + 1]] <- c(s$controlevents[[arrival + 1]], list(event))
144 }
145 deliver <- function(payload) {
146 if (payload == s$r) {
147 line("RECEIVER", paste0("DELIVER f=", s$r))
148 s$stats["delivered"] <- s$stats["delivered"] + 1
149 s$delivered[s$r + 1] <- s$tick
150 s$latency <- s$latency + s$tick - s$first[payload + 1]
151 } else {
152 line("RECEIVER", paste0("DELIVER f=", s$r, " wrong=", payload))
153 s$stats["errors"] <- s$stats["errors"] + 1
154 }
155 s$r <- s$r + 1
156 s$nakmark <- FALSE
157 }
158 discard <- function(seq) {
159 s$stats["discarded"] <- s$stats["discarded"] + 1
160 line("RECEIVER", paste0("DISCARD seq=", seq))
161 }
162 goback <- function() {
163 s$nextframe <- s$base
164 s$timer <- -1
165 s$D <- 0
166 }
167 increase <- function(count) {
168 if (!congestion) return(invisible(NULL))
169 for (i in seq_len(count)) {
170 if (s$cwnd >= w) next
171 if (s$cwnd < s$threshold) {
172 s$cwnd <- s$cwnd + 1
173 } else {
174 s$A <- s$A + 1
175 if (s$A == s$cwnd) {
176 s$cwnd <- s$cwnd + 1
177 s$A <- 0
178 }
179 }
180 }
181 }
182 decrease <- function(timeout) {
183 if (!congestion) return(invisible(NULL))
184 s$threshold <- max(floor(s$cwnd / 2), 1)
185 s$cwnd <- if (timeout) 1 else s$threshold
186 s$A <- 0
187 }
188 cwndline <- function(old) {
189 if (congestion && !identical(old, c(s$cwnd, s$threshold))) line("SENDER", paste0("CWND ", s$cwnd, " ssthresh=", s$threshold))
190 }
191 for (t in 0:L) {
192 s$tick <- t
193 data <- s$dataevents[[t + 1]]
194 if (length(data)) data <- data[order(vapply(data, function(e) e$number, numeric(1)))]
195 for (e in data) {
196 if (e$outcome != "normal") {
197 key <- if (e$outcome == "LOST") "lost_data" else "corrupt_data"
198 s$stats[key] <- s$stats[key] + 1
199 line("RECEIVER", paste0(e$outcome, " data #", e$number))
200 next
201 }
202 seq <- e$value %% M
203 line("RECEIVER", paste0("RECV seq=", seq, " #", e$number))
204 if (!sr) {
205 if (seq == s$r %% M) {
206 deliver(e$value)
207 control("ACK", seq)
208 } else {
209 discard(seq)
210 if (c$NAK == "ON" && !s$nakmark) {
211 control("NAK", s$r %% M)
212 s$nakmark <- TRUE
213 } else if (s$r > 0) {
214 control("ACK", (s$r - 1) %% M)
215 }
216 }
217 } else {
218 o <- (seq - s$r) %% M
219 if (o < w) {
220 pos <- s$r + o
221 key <- as.character(pos)
222 if (!is.null(s$buffer[[key]])) {
223 discard(seq)
224 control("ACK", seq)
225 } else if (o == 0) {
226 deliver(e$value)
227 while (!is.null(s$buffer[[as.character(s$r)]])) {
228 key <- as.character(s$r)
229 payload <- s$buffer[[key]]
230 s$buffer[[key]] <- NULL
231 deliver(payload)
232 }
233 control("ACK", seq)
234 } else {
235 s$buffer[[key]] <- e$value
236 s$stats["max_buffer"] <- max(s$stats["max_buffer"], length(s$buffer))
237 line("RECEIVER", paste0("BUFFER f=", pos))
238 control("ACK", seq)
239 if (c$NAK == "ON" && !s$nakmark) {
240 control("NAK", s$r %% M)
241 s$nakmark <- TRUE
242 }
243 }
244 } else {
245 discard(seq)
246 if ((s$r - seq) %% M <= w) control("ACK", seq)
247 }
248 }
249 }
250 controls <- s$controlevents[[t + 1]]
251 if (length(controls)) controls <- controls[order(vapply(controls, function(e) e$number, numeric(1)))]
252 for (e in controls) {
253 kind <- e$kind
254 value <- e$value
255 j <- e$number
256 if (e$outcome != "normal") {
257 key <- if (e$outcome == "LOST") "lost_ctrl" else "corrupt_ctrl"
258 s$stats[key] <- s$stats[key] + 1
259 line("SENDER", paste0(e$outcome, " ", tolower(kind), " #", j))
260 next
261 }
262 old <- c(s$cwnd, s$threshold)
263 oldbase <- s$base
264 f <- s$base + (value - s$base) %% M
265 accepted <- FALSE
266 if (kind == "ACK") {
267 if (!sr && f < s$H) {
268 accepted <- TRUE
269 s$base <- f + 1
270 s$nextframe <- max(s$nextframe, s$base)
271 } else if (sr && f < s$H && !s$marks[f + 1]) {
272 accepted <- TRUE
273 s$marks[f + 1] <- TRUE
274 if (s$acked[f + 1] < 0) s$acked[f + 1] <- t
275 s$timers[f + 1] <- -1
276 s$queue <- s$queue[s$queue != f]
277 while (s$base < s$H && s$marks[s$base + 1]) s$base <- s$base + 1
278 }
279 } else if (!sr && f <= s$H) {
280 accepted <- TRUE
281 s$base <- f
282 goback()
283 } else if (sr && f < s$H && !s$marks[f + 1]) {
284 accepted <- TRUE
285 s$timers[f + 1] <- -1
286 if (!(f %in% s$queue)) s$queue <- c(s$queue, f)
287 }
288 if (s$base > oldbase) {
289 s$C <- T
290 if (!sr) {
291 s$D <- 0
292 for (a in seq.int(oldbase, s$base - 1)) if (s$acked[a + 1] < 0) s$acked[a + 1] <- t
293 }
294 }
295 if (accepted) {
296 newly <- if (sr) as.numeric(kind == "ACK") else s$base - oldbase
297 increase(newly)
298 if (!sr && kind == "ACK") s$timer <- if (s$base == s$nextframe) -1 else t + s$C
299 if (kind == "NAK") decrease(FALSE)
300 line("SENDER", paste0("GOT", kind, " a=", value, " #", j, " base=", s$base))
301 } else {
302 line("SENDER", paste0("OLD", kind, " a=", value, " #", j))
303 if (!sr && kind == "ACK" && !is.null(c$DUPACK) && s$base >= 1 && value == (s$base - 1) %% M) {
304 s$D <- s$D + 1
305 if (s$D == c$DUPACK) {
306 s$stats["fastretx"] <- s$stats["fastretx"] + 1
307 line("SENDER", paste0("FASTRETX f=", s$base))
308 goback()
309 decrease(FALSE)
310 }
311 }
312 }
313 cwndline(old)
314 }
315 done <- s$base == n
316 if (!done) {
317 expired <- numeric()
318 if (sr && s$base < s$H) {
319 frames <- seq.int(s$base, s$H - 1)
320 expired <- frames[s$timers[frames + 1] == t]
321 } else if (!sr && s$timer == t) expired <- s$base
322 for (f in expired) {
323 old <- c(s$cwnd, s$threshold)
324 s$stats["timeouts"] <- s$stats["timeouts"] + 1
325 line("SENDER", paste0("TIMEOUT f=", f))
326 if (sr) {
327 s$timers[f + 1] <- -1
328 s$queue <- c(s$queue, f)
329 } else {
330 goback()
331 }
332 if (c$BACKOFF == "ON") s$C <- min(2 * s$C, 8 * T)
333 decrease(TRUE)
334 cwndline(old)
335 }
336 E <- if (congestion) min(w, s$cwnd) else w
337 send <- -1
338 if (sr) {
339 if (length(s$queue)) {
340 send <- s$queue[1]
341 s$queue <- s$queue[-1]
342 } else if (s$H < n && s$H < s$base + E && s$H * g <= t) {
343 send <- s$H
344 s$H <- s$H + 1
345 }
346 } else if (s$nextframe < n && s$nextframe < s$base + E && (s$nextframe < s$H || s$nextframe * g <= t)) {
347 send <- s$nextframe
348 s$H <- max(s$H, s$nextframe + 1)
349 s$nextframe <- s$nextframe + 1
350 }
351 if (send >= 0) {
352 s$stats["data"] <- s$stats["data"] + 1
353 i <- unname(s$stats["data"])
354 word <- if (s$tx[send + 1] == 0) "SEND" else "RESEND"
355 if (word == "SEND") s$first[send + 1] <- t else s$stats["resent"] <- s$stats["resent"] + 1
356 s$tx[send + 1] <- s$tx[send + 1] + 1
357 line("SENDER", paste0(word, " f=", send, " seq=", send %% M, " #", i))
358 arrival <- t + d + extra_delay("DELAYDATA", i)
359 event <- list(number = i, value = send, kind = "data", outcome = outcome("data", i))
360 s$dataevents[[arrival + 1]] <- c(s$dataevents[[arrival + 1]], list(event))
361 if (sr) s$timers[send + 1] <- t + s$C else if (s$timer < 0) s$timer <- t + s$C
362 }
363 }
364 if (t %in% c$SNAPSHOT) {
365 E <- if (congestion) min(w, s$cwnd) else w
366 sx <- rx <- character(n)
367 for (f in 0:(n - 1)) {
368 acknowledged <- if (sr) s$marks[f + 1] else f < s$base
369 waiting <- if (sr) f %in% s$queue else s$nextframe <= f && f < s$H
370 sent <- if (sr) f < s$H else f < s$nextframe
371 sx[f + 1] <- if (acknowledged) "A" else if (waiting) "R" else if (sent) "O" else if (f < s$base + E) "W" else "."
372 rx[f + 1] <- if (f < s$r) "D" else if (sr && !is.null(s$buffer[[as.character(f)]])) "B" else if (if (sr) f < s$r + w else f == s$r) "W" else "."
373 }
374 line("ALWAYS", paste0("SNAPSHOT sender=", paste(sx, collapse = ""), " receiver=", paste(rx, collapse = "")))
375 }
376 if (done || t == L) {
377 line("ALWAYS", if (done) "DONE" else "ABORT")
378 completion <- if (done) t else -1
379 tend <- t
380 break
381 }
382 }
383 text <- function(v) if (v < 0) "-" else as.character(v)
384 for (f in 0:(n - 1)) emit("frame ", f, ": seq=", f %% M, " tx=", s$tx[f + 1], " first=", text(s$first[f + 1]), " delivered=", text(s$delivered[f + 1]), " acked=", text(s$acked[f + 1]))
385 groups <- list(totals = c("data", "resent", "control", "nak", "timeouts", "fastretx"), channel = c("lost_data", "lost_ctrl", "corrupt_data", "corrupt_ctrl"), receiver = c("delivered", "errors", "discarded", "max_buffer"))
386 for (prefix in names(groups)) {
387 ks <- groups[[prefix]]
388 emit(prefix, ": ", paste(paste0(ks, "=", s$stats[ks]), collapse = " "))
389 }
390 D <- unname(s$stats["delivered"])
391 X <- unname(s$stats["data"])
392 goodput <- if (completion < 0) "-" else decimal(D, completion, 2)
393 lag <- if (D == 0) "-" else decimal(s$latency, D, 2)
394 emit("result: completion=", text(completion), " goodput=", goodput, " efficiency=", decimal(100 * D, X, 1), "% utilization=", decimal(100 * X, tend + 1, 1), "% latency=", lag)
395 list(name = name, protocol = p, status = if (completion < 0) "aborted" else "done", delivered = D, completion = completion, data = X)
396}
397main <- function() {
398 options(scipen = 999)
399 input <- readLines(file("stdin"), warn = FALSE)
400 q <- as.integer(input[1])
401 position <- 2
402 results <- list()
403 for (scenario in seq_len(q)) {
404 name <- strsplit(input[position], " ", fixed = TRUE)[[1]][2]
405 position <- position + 1
406 lines <- list()
407 while (input[position] != "END") {
408 lines[[length(lines) + 1]] <- strsplit(input[position], " ", fixed = TRUE)[[1]]
409 position <- position + 1
410 }
411 position <- position + 1
412 emit("== ", name, " ==")
413 result <- tryCatch(simulate(name, configure(lines)), settings_error = function(e) {
414 emit("error: ", e$message)
415 list(name = name, protocol = "", status = "invalid", delivered = 0, completion = -1, data = 0)
416 })
417 results[[length(results) + 1]] <- result
418 }
419 emit("== ranking ==")
420 completed <- Filter(function(r) r$status == "done", results)
421 better <- function(a, b) {
422 left <- a$delivered * b$completion
423 right <- b$delivered * a$completion
424 if (left != right) return(left > right)
425 a$delivered * b$data > b$delivered * a$data
426 }
427 sorted <- list()
428 for (r in completed) {
429 i <- 1
430 while (i <= length(sorted) && !better(r, sorted[[i]])) i <- i + 1
431 sorted <- append(sorted, list(r), after = i - 1)
432 }
433 for (i in seq_along(sorted)) {
434 r <- sorted[[i]]
435 emit(i, ". ", r$name, " ", r$protocol, " goodput=", decimal(r$delivered, r$completion, 2), " efficiency=", decimal(100 * r$delivered, r$data, 1), "%")
436 }
437 for (r in results) if (r$status != "done") emit("- ", r$name, " ", r$status)
438 emit("== protocols ==")
439 for (p in c("SW", "GBN", "SR")) {
440 runs <- Filter(function(r) r$protocol == p, completed)
441 if (!length(runs)) next
442 D <- sum(vapply(runs, function(r) r$delivered, numeric(1)))
443 T <- sum(vapply(runs, function(r) r$completion, numeric(1)))
444 X <- sum(vapply(runs, function(r) r$data, numeric(1)))
445 emit(p, ": runs=", length(runs), " goodput=", decimal(D, T, 2), " efficiency=", decimal(100 * D, X, 1), "%")
446 }
447}
