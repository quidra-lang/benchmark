29String fixed(int n, int d, int digits) {
30 int scale = digits == 2 ? 100 : 10;
31 int v = d == 0 ? 0 : n * scale ~/ d;
32 return '${v ~/ scale}.${(v % scale).toString().padLeft(digits, '0')}';

35String tick(int t) => t < 0 ? '-' : '$t';
36String onOff(bool value) => value ? 'on' : 'off';

38class Scenario {
39 final String name;
40 String protocol = '', trace = 'ALL';
41 String? error;
42 final Map<String, List<String>> values = {};
43 final Map<String, Set<int>> lists = {};
44 final Map<String, Map<int, int>> delays = {};
45 int n = 0,
46 w = 0,
47 k = 0,
48 timeout = 0,
49 delay = 0,
50 interval = 0,
51 limit = 1000,
52 dupack = 0;
53 bool nak = false,
54 backoff = false,
55 congestion = false,
56 unsafe = false,
57 random = false;
58 int seed = 0, loss = 0, corruption = 0;
59 int completion = -1, end = 0, delivered = 0, data = 0;
60 Scenario(this.name);
61 int number(String key) => int.parse(values[key]![0]);
62 bool on(String key) => values[key]?[0] == 'ON';
63 void line(List<String> words) {
64 if (error != null) return;
65 String key = words[0];
66 int pos = keys.indexOf(key);
67 if (pos < 0) {
68 error = 'unknown key $key';
69 return;
70 }
71 if (values.containsKey(key)) {
72 error = 'duplicate key $key';
73 return;
74 }
75 int count = words.length - 1;
76 bool bad = key == 'RANDOM'
77 ? count != 3
78 : pos >= 15
79 ? count < 1
80 : count != 1;
81 if (bad) {
82 error = 'wrong number of values for $key';
83 return;
84 }
85 values[key] = words.sublist(1);
86 }
87
88 bool valid(String key) {
89 List<String> v = values[key]!;
90 if (key == 'PROTOCOL') return ['SW', 'GBN', 'SR'].contains(v[0]);
91 if (['NAK', 'BACKOFF', 'CONGESTION', 'UNSAFE'].contains(key))
92 return ['ON', 'OFF'].contains(v[0]);
93 if (key == 'TRACE')
94 return ['ALL', 'SENDER', 'RECEIVER', 'NONE'].contains(v[0]);
95 if (key == 'RANDOM') {
96 int s = int.parse(v[0]), p = int.parse(v[1]), c = int.parse(v[2]);
97 return s >= 1 && s <= 2147483646 && p >= 0 && c >= 0 && p + c <= 100;
98 }
99 if (keys.indexOf(key) >= 15) {
100 Set<int> found = {};
101 Map<int, int> extra = {};
102 bool delayed = key.startsWith('DELAY');
103 for (String item in v) {
104 List<String> pair = item.split(':');
105 int a = int.parse(pair[0]);
106 bool snapshot = key == 'SNAPSHOT';
107 if (a < (snapshot ? 0 : 1) ||
108 a > (snapshot ? 2000 : 9999) ||
109 !found.add(a))
110 return false;
111 if (delayed) {
112 int e = int.parse(pair[1]);
113 if (e < 1 || e > 50) return false;
114 extra[a] = e;
115 }
116 }
117 lists[key] = found;
118 if (delayed) delays[key] = extra;
119 return true;
120 }
121 int a = number(key), lo = 1, hi = 0;
122 switch (key) {
123 case 'FRAMES':
124 {
125 hi = 60;
126 }
127 case 'WINDOW':
128 {
129 hi = 32;
130 }
131 case 'SEQBITS':
132 {
133 hi = 6;
134 }
135 case 'TIMEOUT':
136 {
137 hi = 50;
138 }
139 case 'DELAY':
140 case 'DUPACK':
141 {
142 hi = 10;
143 }
144 case 'INTERVAL':
145 {
146 lo = 0;
147 hi = 10;
148 }
149 case 'MAXTICKS':
150 {
151 hi = 2000;
152 }
153 }
154 return a >= lo && a <= hi;
155 }

157 void check() {
158 if (error != null) return;
159 for (String key in keys.take(6)) {
160 if (!values.containsKey(key)) {
161 error = 'missing key $key';
162 return;
163 }
164 }
165 for (String key in keys) {
166 if (values.containsKey(key) && !valid(key)) {
167 error = 'bad value for $key';
168 return;
169 }
170 }
171 protocol = values['PROTOCOL']![0];
172 n = number('FRAMES');
173 w = number('WINDOW');
174 k = number('SEQBITS');
175 timeout = number('TIMEOUT');
176 delay = number('DELAY');
177 if (values.containsKey('INTERVAL')) interval = number('INTERVAL');
178 if (values.containsKey('MAXTICKS')) limit = number('MAXTICKS');
179 if (values.containsKey('DUPACK')) dupack = number('DUPACK');
180 nak = on('NAK');
181 backoff = on('BACKOFF');
182 congestion = on('CONGESTION');
183 unsafe = on('UNSAFE');
184 if (values.containsKey('TRACE')) trace = values['TRACE']![0];
185 random = values.containsKey('RANDOM');
186 if (random) {
187 List<String> v = values['RANDOM']!;
188 seed = int.parse(v[0]);
189 loss = int.parse(v[1]);
190 corruption = int.parse(v[2]);
191 }
192 if (protocol == 'SR' && dupack > 0) {
193 error = 'DUPACK not supported for SR';
194 return;
195 }
196 int upper = protocol == 'SW'
197 ? 1
198 : protocol == 'GBN'
199 ? (1 << k) - 1
200 : 1 << (k - 1);
201 if (w > upper && !(unsafe && protocol != 'SW')) {
202 error = 'window $w exceeds limit $upper for $protocol';
203 return;
204 }
205 for (String kind in ['DATA', 'CTRL']) {
206 List<int> both =
207 (lists['LOSE$kind'] ?? <int>{})
208 .intersection(lists['CORRUPT$kind'] ?? {})
209 .toList()
210 ..sort();
211 if (both.isNotEmpty) {
212 error =
213 '${kind == 'DATA' ? 'data' : 'control'} #${both.first} both lost and corrupted';
214 return;
215 }
216 }
217 }
218
219 bool contains(String key, int i) => lists[key]?.contains(i) ?? false;
220 int extra(String key, int i) => delays[key]?[i] ?? 0;
221}

223class Event {
224 final int arrival, number, value, outcome;
225 final String kind;
226 Event(this.arrival, this.number, this.value, this.outcome, this.kind);
227}

229class Simulation {
230 final Scenario s;
231 late final int modulus;
232 late final bool sr;
233 int t = 0,
234 base = 0,
235 high = 0,
236 next = 0,
237 timer = -1,
238 length = 0,
239 duplicates = 0,
240 r = 0;
241 bool nakSent = false;
242 int cwnd = 1, threshold = 0, growth = 0;
243 int controlCount = 0, naks = 0, resent = 0, timeouts = 0, fast = 0;
244 int lostData = 0, lostControl = 0, corruptData = 0, corruptControl = 0;
245 int errors = 0, discarded = 0, maxBuffer = 0, latency = 0, random = 0;
246 late final List<int> tx, first, delivery, acked, timers;
247 late final List<bool> marks;
248 final List<int> queue = [];
249 final Map<int, int> buffer = {};
250 final List<Event> dataEvents = [], controlEvents = [];
251 Simulation(this.s) {
252 modulus = 1 << s.k;
253 sr = s.protocol == 'SR';
254 length = s.timeout;
255 threshold = s.w;
256 random = s.seed;
257 tx = List.filled(s.n, 0);
258 first = List.filled(s.n, -1);
259 delivery = List.filled(s.n, -1);
260 acked = List.filled(s.n, -1);
261 timers = List.filled(s.n, -1);
262 marks = List.filled(s.n, false);
263 }
264 int mod(int a) => a % modulus;
265 void emit(String side, String text) {
266 if (side == 'ALWAYS' || s.trace == 'ALL' || s.trace == side)
267 output.add('$t $text');
268 }
269
270 int effective() => s.congestion ? math.min(s.w, cwnd) : s.w;
271 int outcome(String kind, int number) {
272 int result = 0;
273 if (s.random) {
274 random = random * 48271 % 2147483647;
275 int v = random % 100;
276 result = v < s.loss
277 ? 1
278 : v < s.loss + s.corruption
279 ? 2
280 : 0;
281 }
282 String suffix = kind == 'data' ? 'DATA' : 'CTRL';
283 if (s.contains('LOSE$suffix', number)) result = 1;
284 if (s.contains('CORRUPT$suffix', number)) result = 2;
285 return result;
286 }
287
288 void control(bool nak, int a) {
289 int id = ++controlCount;
290 if (nak) naks++;
291 String kind = nak ? 'nak' : 'ack';
292 emit('RECEIVER', '${kind.toUpperCase()} a=$a #$id');
293 controlEvents.add(
294 Event(
295 t + s.delay + s.extra('DELAYCTRL', id),
296 id,
297 a,
298 outcome(kind, id),
299 kind,
300 ),
301 );
302 }
303
304 void deliver(int payload) {
305 if (payload == r) {
306 emit('RECEIVER', 'DELIVER f=$r');
307 if (delivery[payload] < 0) delivery[payload] = t;
308 s.delivered++;
309 latency += t - first[payload];
310 } else {
311 emit('RECEIVER', 'DELIVER f=$r wrong=$payload');
312 errors++;
313 }
314 r++;
315 nakSent = false;
316 }
317
318 void discard(int seq) {
319 discarded++;
320 emit('RECEIVER', 'DISCARD seq=$seq');
321 }
322
323 void receive(Event e) {
324 int seq = mod(e.value);
325 emit('RECEIVER', 'RECV seq=$seq #${e.number}');
326 if (!sr) {
327 if (seq == mod(r)) {
328 deliver(e.value);
329 control(false, seq);
330 } else {
331 discard(seq);
332 if (s.nak && !nakSent) {
333 control(true, mod(r));
334 nakSent = true;
335 } else if (r > 0) {
336 control(false, mod(r - 1));
337 }
338 }
339 } else {
340 int offset = mod(seq - r);
341 if (offset < s.w) {
342 int position = r + offset;
343 if (buffer.containsKey(position)) {
344 discard(seq);
345 control(false, seq);
346 } else if (offset == 0) {
347 deliver(e.value);
348 while (buffer.containsKey(r)) {
349 deliver(buffer.remove(r)!);
350 }
351 control(false, seq);
352 } else {
353 buffer[position] = e.value;
354 maxBuffer = math.max(maxBuffer, buffer.length);
355 emit('RECEIVER', 'BUFFER f=$position');
356 control(false, seq);
357 if (s.nak && !nakSent) {
358 control(true, mod(r));
359 nakSent = true;
360 }
361 }
362 } else {
363 discard(seq);
364 if (mod(r - seq) <= s.w) control(false, seq);
365 }
366 }
367 }
368
369 void goBack() {
370 next = base;
371 timer = -1;
372 duplicates = 0;
373 }
374
375 void increase(int count) {
376 if (!s.congestion) return;
377 for (int i = 0; i < count && cwnd < s.w; i++) {
378 if (cwnd < threshold) {
379 cwnd++;
380 } else if (++growth == cwnd) {
381 cwnd++;
382 growth = 0;
383 }
384 }
385 }
386
387 void decrease(bool timeout) {
388 if (!s.congestion) return;
389 threshold = math.max(cwnd ~/ 2, 1);
390 cwnd = timeout ? 1 : threshold;
391 growth = 0;
392 }
393
394 void cwndLine(int c, int h) {
395 if (s.congestion && (c != cwnd || h != threshold))
396 emit('SENDER', 'CWND $cwnd ssthresh=$threshold');
397 }
398
399 void acknowledge(Event e) {
400 int oldC = cwnd, oldS = threshold, oldBase = base;
401 int a = e.value, f = base + mod(a - base);
402 bool nak = e.kind == 'nak', accepted = false;
403 if (sr) {
404 if (f < high && !marks[f]) {
405 accepted = true;
406 timers[f] = -1;
407 if (nak) {
408 if (!queue.contains(f)) queue.add(f);
409 } else {
410 marks[f] = true;
411 if (acked[f] < 0) acked[f] = t;
412 queue.remove(f);
413 while (base < high && marks[base]) {
414 base++;
415 }
416 increase(1);
417 }
418 }
419 } else if (nak ? f <= high : f < high) {
420 accepted = true;
421 base = nak ? f : f + 1;
422 increase(base - oldBase);
423 for (int i = oldBase; i < base; i++) {
424 if (acked[i] < 0) acked[i] = t;
425 }
426 if (nak) {
427 goBack();
428 } else if (next < base) {
429 next = base;
430 }
431 }
432 if (base > oldBase) {
433 length = s.timeout;
434 duplicates = 0;
435 }
436 if (accepted && !sr && !nak) timer = base == next ? -1 : t + length;
437 String label = nak ? 'NAK' : 'ACK';
438 emit(
439 'SENDER',
440 '${accepted ? 'GOT' : 'OLD'}$label a=$a #${e.number}${accepted ? ' base=$base' : ''}',
441 );
442 if (accepted && nak) decrease(false);
443 if (!accepted &&
444 !nak &&
445 !sr &&
446 s.dupack > 0 &&
447 base >= 1 &&
448 a == mod(base - 1)) {
449 duplicates++;
450 if (duplicates == s.dupack) {
451 fast++;
452 emit('SENDER', 'FASTRETX f=$base');
453 goBack();
454 decrease(false);
455 }
456 }
457 cwndLine(oldC, oldS);
458 }
459
460 void timeout(int f) {
461 int oldC = cwnd, oldS = threshold;
462 timeouts++;
463 emit('SENDER', 'TIMEOUT f=$f');
464 if (sr) {
465 timers[f] = -1;
466 queue.add(f);
467 } else {
468 goBack();
469 }
470 if (s.backoff) length = math.min(length * 2, 8 * s.timeout);
471 decrease(true);
472 cwndLine(oldC, oldS);
473 }
474
475 void send(int f) {
476 int id = ++s.data;
477 bool again = tx[f]++ > 0;
478 if (again) {
479 resent++;
480 } else {
481 first[f] = t;
482 }
483 emit('SENDER', '${again ? 'RESEND' : 'SEND'} f=$f seq=${mod(f)} #$id');
484 dataEvents.add(
485 Event(
486 t + s.delay + s.extra('DELAYDATA', id),
487 id,
488 f,
489 outcome('data', id),
490 'data',
491 ),
492 );
493 }
494
495 void arrivals(List<Event> events, bool data) {
496 List<Event> ready = events.where((e) => e.arrival == t).toList()
497 ..sort((a, b) => a.number.compareTo(b.number));
498 events.removeWhere((e) => e.arrival == t);
499 for (Event e in ready) {
500 if (e.outcome != 0) {
501 bool lost = e.outcome == 1;
502 if (data) {
503 if (lost) {
504 lostData++;
505 } else {
506 corruptData++;
507 }
508 } else {
509 if (lost) {
510 lostControl++;
511 } else {
512 corruptControl++;
513 }
514 }
515 emit(
516 data ? 'RECEIVER' : 'SENDER',
517 '${lost ? 'LOST' : 'CORRUPT'} ${e.kind} #${e.number}',
518 );
519 } else if (data) {
520 receive(e);
521 } else {
522 acknowledge(e);
523 }
524 }
525 }
526
527 void snapshot() {
528 StringBuffer x = StringBuffer(), y = StringBuffer();
529 for (int f = 0; f < s.n; f++) {
530 bool ack = sr ? marks[f] : f < base;
531 bool retransmit = sr ? queue.contains(f) : next <= f && f < high;
532 bool sent = sr ? f < high : f < next;
533 x.write(
534 ack
535 ? 'A'
536 : retransmit
537 ? 'R'
538 : sent
539 ? 'O'
540 : f < base + effective()
541 ? 'W'
542 : '.',
543 );
544 y.write(
545 f < r
546 ? 'D'
547 : sr && buffer.containsKey(f)
548 ? 'B'
549 : (sr ? f < r + s.w : f == r)
550 ? 'W'
551 : '.',
552 );
553 }
554 emit('ALWAYS', 'SNAPSHOT sender=$x receiver=$y');
555 }
556
557 void run() {
558 for (t = 0; t <= s.limit; t++) {
559 arrivals(dataEvents, true);
560 arrivals(controlEvents, false);
561 bool done = base == s.n;
562 if (!done) {
563 if (sr) {
564 for (int f = base; f < high; f++) {
565 if (timers[f] == t) timeout(f);
566 }
567 } else if (timer == t) {
568 timeout(base);
569 }
570 if (sr) {
571 int f = -1;
572 if (queue.isNotEmpty) {
573 f = queue.removeAt(0);
574 } else if (high < s.n &&
575 high < base + effective() &&
576 high * s.interval <= t) {
577 f = high++;
578 }
579 if (f >= 0) {
580 send(f);
581 timers[f] = t + length;
582 }
583 } else if (next < s.n &&
584 next < base + effective() &&
585 (next < high || next * s.interval <= t)) {
586 send(next);
587 high = math.max(high, ++next);
588 if (timer < 0) timer = t + length;
589 }
590 }
591 if (s.contains('SNAPSHOT', t)) snapshot();
592 if (done || t == s.limit) {
593 s.end = t;
594 if (done) s.completion = t;
595 emit('ALWAYS', done ? 'DONE' : 'ABORT');
596 break;
597 }
598 }
599 for (int f = 0; f < s.n; f++) {
600 output.add(
601 'frame $f: seq=${mod(f)} tx=${tx[f]} first=${tick(first[f])} delivered=${tick(delivery[f])} acked=${tick(acked[f])}',
602 );
603 }
604 output.add(
605 'totals: data=${s.data} resent=$resent control=$controlCount nak=$naks timeouts=$timeouts fastretx=$fast',
606 );
607 output.add(
608 'channel: lost_data=$lostData lost_ctrl=$lostControl corrupt_data=$corruptData corrupt_ctrl=$corruptControl',
609 );
610 output.add(
611 'receiver: delivered=${s.delivered} errors=$errors discarded=$discarded max_buffer=$maxBuffer',
612 );
613 String goodput = s.completion < 0
614 ? '-'
615 : fixed(s.delivered, s.completion, 2);
616 String latencyText = s.delivered == 0
617 ? '-'
618 : fixed(latency, s.delivered, 2);
619 output.add(
620 'result: completion=${tick(s.completion)} goodput=$goodput efficiency=${fixed(s.delivered * 100, s.data, 1)}% utilization=${fixed(s.data * 100, s.end + 1, 1)}% latency=$latencyText',
621 );
622 }
623}

625void main() {
626 int count = int.parse(stdin.readLineSync()!);
627 List<Scenario> scenarios = [];
628 for (int i = 0; i < count; i++) {
629 Scenario s = Scenario(stdin.readLineSync()!.split(' ')[1]);
630 while (true) {
631 String line = stdin.readLineSync()!;
632 if (line == 'END') break;
633 s.line(line.split(' '));
634 }
635 s.check();
636 scenarios.add(s);
637 output.add('== ${s.name} ==');
638 if (s.error != null) {
639 output.add('error: ${s.error}');
640 } else {
641 output.add(
642 'config: ${s.protocol} n=${s.n} w=${s.w} k=${s.k} T=${s.timeout} d=${s.delay}',
643 );
644 String random = s.random ? '${s.seed}/${s.loss}/${s.corruption}' : 'off';
645 output.add(
646 'options: interval=${s.interval} limit=${s.limit} dupack=${s.dupack == 0 ? 'off' : s.dupack} nak=${onOff(s.nak)} backoff=${onOff(s.backoff)} congestion=${onOff(s.congestion)} unsafe=${onOff(s.unsafe)} trace=${s.trace.toLowerCase()} random=$random',
647 );
648 Simulation(s).run();
649 }
650 }
651 output.add('== ranking ==');
652 List<Scenario> completed = scenarios.where((s) => s.completion >= 0).toList();
653 Map<Scenario, int> order = {
654 for (int i = 0; i < scenarios.length; i++) scenarios[i]: i,
655 };
656 completed.sort((a, b) {
657 int c = (b.delivered * a.completion).compareTo(a.delivered * b.completion);
658 if (c != 0) return c;
659 c = (b.delivered * a.data).compareTo(a.delivered * b.data);
660 return c != 0 ? c : order[a]!.compareTo(order[b]!);
661 });
662 for (int i = 0; i < completed.length; i++) {
663 Scenario s = completed[i];
664 output.add(
665 '${i + 1}. ${s.name} ${s.protocol} goodput=${fixed(s.delivered, s.completion, 2)} efficiency=${fixed(s.delivered * 100, s.data, 1)}%',
666 );
667 }
668 for (Scenario s in scenarios) {
669 if (s.completion < 0)
670 output.add('- ${s.name} ${s.error != null ? 'invalid' : 'aborted'}');
671 }
672 output.add('== protocols ==');
673 for (String p in ['SW', 'GBN', 'SR']) {
674 int runs = 0, d = 0, t = 0, x = 0;
675 for (Scenario s in completed) {
676 if (s.protocol == p) {
677 runs++;
678 d += s.delivered;
679 t += s.completion;
680 x += s.data;
681 }
682 }
683 if (runs > 0)
684 output.add(
685 '$p: runs=$runs goodput=${fixed(d, t, 2)} efficiency=${fixed(d * 100, x, 1)}%',
686 );
687 }
688 stdout.write('${output.join('\n')}\n');
689}
