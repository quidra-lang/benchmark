3const PROTOCOLS = ['SW', 'GBN', 'SR'];
4const KEYS = [
5 'PROTOCOL', 'FRAMES', 'WINDOW', 'SEQBITS', 'TIMEOUT', 'DELAY',
6 'INTERVAL', 'MAXTICKS', 'DUPACK', 'NAK', 'BACKOFF', 'CONGESTION',
7 'UNSAFE', 'TRACE', 'RANDOM', 'LOSEDATA', 'LOSECTRL', 'CORRUPTDATA',
8 'CORRUPTCTRL', 'DELAYDATA', 'DELAYCTRL', 'SNAPSHOT'
9];
10const REQUIRED = ['PROTOCOL', 'FRAMES', 'WINDOW', 'SEQBITS', 'TIMEOUT', 'DELAY'];
11const LIST_KEYS = new Set([
12 'LOSEDATA', 'LOSECTRL', 'CORRUPTDATA', 'CORRUPTCTRL',
13 'DELAYDATA', 'DELAYCTRL', 'SNAPSHOT'
14]);
15const DELAY_KEYS = new Set(['DELAYDATA', 'DELAYCTRL']);
16const CHOICES = {
17 PROTOCOL: PROTOCOLS, NAK: ['ON', 'OFF'], BACKOFF: ['ON', 'OFF'],
18 CONGESTION: ['ON', 'OFF'], UNSAFE: ['ON', 'OFF'],
19 TRACE: ['ALL', 'SENDER', 'RECEIVER', 'NONE']
20};
21const RANGES = {
22 FRAMES: [1, 60], WINDOW: [1, 32], SEQBITS: [1, 6],
23 TIMEOUT: [1, 50], DELAY: [1, 10], INTERVAL: [0, 10],
24 MAXTICKS: [1, 2000], DUPACK: [1, 10],
25 LOSEDATA: [1, 9999], LOSECTRL: [1, 9999],
26 CORRUPTDATA: [1, 9999], CORRUPTCTRL: [1, 9999], SNAPSHOT: [0, 2000]
27};
28const TRACE_SIDES = {
29 ALL: ['SENDER', 'RECEIVER'], SENDER: ['SENDER'], RECEIVER: ['RECEIVER'], NONE: []
30};

36function truncated(numerator, denominator, digits) {
37 const scale = 10 ** digits;
38 const value = Math.floor(numerator * scale / denominator);
39 return Math.floor(value / scale) + '.' + String(value % scale).padStart(digits, '0');
40}

42function tickText(tick) { return tick === null ? '-' : String(tick); }
43function mod(value, modulus) { return ((value % modulus) + modulus) % modulus; }
44function option(settings, key, fallback) { return (settings.get(key) ?? [fallback])[0]; }
45function numberSet(settings, key) { return new Set((settings.get(key) ?? []).map(Number)); }
46function delayMap(settings, key) {
47 return new Map((settings.get(key) ?? []).map(value => value.split(':').map(Number)));
48}
49function inRange(value, bounds) { return bounds[0] <= value && value <= bounds[1]; }
50function unique(values) { return new Set(values).size === values.length; }
51function validValue(key, values) {
52 if (Object.hasOwn(CHOICES, key)) return CHOICES[key].includes(values[0]);
53 if (!LIST_KEYS.has(key) && Object.hasOwn(RANGES, key)) {
54 return inRange(Number(values[0]), RANGES[key]);
55 }
56 if (key === 'RANDOM') {
57 const [seed, loss, corrupt] = values.map(Number);
58 return inRange(seed, [1, RANDOM_MODULUS - 1]) && loss >= 0 && corrupt >= 0 && loss + corrupt <= 100;
59 }
60 if (DELAY_KEYS.has(key)) {
61 const pairs = values.map(value => value.split(':').map(Number));
62 return pairs.every(([n, extra]) => inRange(n, [1, 9999]) && inRange(extra, [1, 50])) &&
63 unique(pairs.map(([n]) => n));
64 }
65 const numbers = values.map(Number);
66 return numbers.every(n => inRange(n, RANGES[key])) && unique(numbers);
67}

69class Config {
70 constructor(settings) {
71 this.protocol = option(settings, 'PROTOCOL', '');
72 this.frames = Number(option(settings, 'FRAMES', '0'));
73 this.window = Number(option(settings, 'WINDOW', '0'));
74 this.seqbits = Number(option(settings, 'SEQBITS', '0'));
75 this.timeout = Number(option(settings, 'TIMEOUT', '0'));
76 this.delay = Number(option(settings, 'DELAY', '0'));
77 this.interval = Number(option(settings, 'INTERVAL', '0'));
78 this.limit = Number(option(settings, 'MAXTICKS', '1000'));
79 this.dupack = settings.has('DUPACK') ? Number(settings.get('DUPACK')[0]) : null;
80 this.nak = option(settings, 'NAK', 'OFF') === 'ON';
81 this.backoff = option(settings, 'BACKOFF', 'OFF') === 'ON';
82 this.congestion = option(settings, 'CONGESTION', 'OFF') === 'ON';
83 this.unsafe = option(settings, 'UNSAFE', 'OFF') === 'ON';
84 this.trace = option(settings, 'TRACE', 'ALL');
85 this.random = settings.has('RANDOM') ? settings.get('RANDOM').map(Number) : null;
86 this.loseData = numberSet(settings, 'LOSEDATA');
87 this.loseCtrl = numberSet(settings, 'LOSECTRL');
88 this.corruptData = numberSet(settings, 'CORRUPTDATA');
89 this.corruptCtrl = numberSet(settings, 'CORRUPTCTRL');
90 this.delayData = delayMap(settings, 'DELAYDATA');
91 this.delayCtrl = delayMap(settings, 'DELAYCTRL');
92 this.snapshots = numberSet(settings, 'SNAPSHOT');
93 }

116function checkSettings(lines) {
117 const settings = new Map();
118 for (const [key, values] of lines) {
119 if (!KEYS.includes(key)) throw new SettingsError(`unknown key ${key}`);
120 if (settings.has(key)) throw new SettingsError(`duplicate key ${key}`);
121 const countMatches = key === 'RANDOM' ? values.length === 3 :
122 LIST_KEYS.has(key) ? values.length >= 1 : values.length === 1;
123 if (!countMatches) throw new SettingsError(`wrong number of values for ${key}`);
124 settings.set(key, values);
125 }
126 for (const key of REQUIRED) {
127 if (!settings.has(key)) throw new SettingsError(`missing key ${key}`);
128 }
129 for (const key of KEYS) {
130 if (settings.has(key) && !validValue(key, settings.get(key))) {
131 throw new SettingsError(`bad value for ${key}`);
132 }
133 }
134 const config = new Config(settings);
135 if (config.protocol === 'SR' && config.dupack !== null) {
136 throw new SettingsError('DUPACK not supported for SR');
137 }
138 const limit = config.windowLimit();
139 if ((config.protocol === 'SW' || !config.unsafe) && config.window > limit) {
140 throw new SettingsError(`window ${config.window} exceeds limit ${limit} for ${config.protocol}`);
141 }
142 const dataBoth = [...config.loseData].filter(n => config.corruptData.has(n));
143 if (dataBoth.length) throw new SettingsError(`data #${Math.min(...dataBoth)} both lost and corrupted`);
144 const ctrlBoth = [...config.loseCtrl].filter(n => config.corruptCtrl.has(n));
145 if (ctrlBoth.length) throw new SettingsError(`control #${Math.min(...ctrlBoth)} both lost and corrupted`);
146 return config;
147}

158class LossGenerator {
159 constructor(seed, loss, corruption) {
160 this.state = seed;
161 this.loss = loss;
162 this.corruption = corruption;
163 }
164
165 nextOutcome() {
166 this.state = RANDOM_MULTIPLIER * this.state % RANDOM_MODULUS;
167 const roll = this.state % 100;
168 if (roll < this.loss) return 'LOST';
169 if (roll < this.loss + this.corruption) return 'CORRUPT';
170 return 'NORMAL';
171 }

174class Channel {
175 constructor(delay, losses, corruptions, extraDelays) {
176 this.delay = delay;
177 this.losses = losses;
178 this.corruptions = corruptions;
179 this.extraDelays = extraDelays;
180 this.sent = 0;
181 this.lost = 0;
182 this.corrupted = 0;
183 this.inTransit = [];
184 }
185
186 send(kind, value, payload, tick, generator) {
187 const number = ++this.sent;
188 let outcome = generator === null ? 'NORMAL' : generator.nextOutcome();
189 if (this.losses.has(number)) outcome = 'LOST';
190 else if (this.corruptions.has(number)) outcome = 'CORRUPT';
191 const item = {
192 number, kind, value, payload, outcome,
193 arrival: tick + this.delay + (this.extraDelays.get(number) ?? 0)
194 };
195 this.inTransit.push(item);
196 return item;
197 }
198
199 arrivals(tick) {
200 const arrived = this.inTransit.filter(item => item.arrival === tick);
201 this.inTransit = this.inTransit.filter(item => item.arrival !== tick);
202 return arrived;
203 }
204
205 countDamage(item) {
206 if (item.outcome === 'LOST') ++this.lost;
207 else ++this.corrupted;
208 }
209}

211class Simulation {
212 constructor(config) {
213 this.config = config;
214 this.selective = config.protocol === 'SR';
215 this.modulus = 2 ** config.seqbits;
216 this.visible = new Set(TRACE_SIDES[config.trace]);
217 this.generator = config.random === null ? null : new LossGenerator(...config.random);
218 this.dataChannel = new Channel(config.delay, config.loseData, config.corruptData, config.delayData);
219 this.controlChannel = new Channel(config.delay, config.loseCtrl, config.corruptCtrl, config.delayCtrl);
220 this.records = Array.from({ length: config.frames }, () => new FrameRecord());
221 this.lines = [];
222 this.tick = 0;
223 this.base = 0;
224 this.framesSent = 0;
225 this.nextFrame = 0;
226 this.timerLength = config.timeout;
227 this.timer = null;
228 this.duplicates = 0;
229 this.acked = Array(config.frames).fill(false);
230 this.timers = Array(config.frames).fill(null);
231 this.queue = [];
232 this.cwnd = 1;
233 this.ssthresh = config.window;
234 this.roundAcks = 0;
235 this.expected = 0;
236 this.nakSent = false;
237 this.buffer = new Map();
238 this.resent = 0;
239 this.naks = 0;
240 this.timeouts = 0;
241 this.fastRetransmits = 0;
242 this.delivered = 0;
243 this.errors = 0;
244 this.discarded = 0;
245 this.maxBuffer = 0;
246 this.completion = null;
247 }
248
249 emit(...parts) { this.lines.push([this.tick, ...parts].join(' ')); }
250 trace(side, ...parts) { if (this.visible.has(side)) this.emit(...parts); }
251 frameOf(value) { return this.base + mod(value - this.base, this.modulus); }
252 effectiveWindow() { return this.config.congestion ? Math.min(this.config.window, this.cwnd) : this.config.window; }
253
254 run() {
255 for (let tick = 0; tick <= this.config.limit; ++tick) {
256 this.tick = tick;
257 for (const item of this.dataChannel.arrivals(tick)) {
258 if (item.outcome === 'NORMAL') this.receiveData(item);
259 else this.reportDamage(this.dataChannel, item, 'RECEIVER');
260 }
261 for (const item of this.controlChannel.arrivals(tick)) {
262 if (item.outcome === 'NORMAL') this.receiveControl(item);
263 else this.reportDamage(this.controlChannel, item, 'SENDER');
264 }
265 const completed = this.base === this.config.frames;
266 if (!completed) {
267 this.processTimeouts();
268 this.sendData();
269 }
270 if (this.config.snapshots.has(tick)) this.snapshot();
271 if (completed) {
272 this.completion = tick;
273 this.emit('DONE');
274 return;
275 }
276 }
277 this.emit('ABORT');
278 }
279
280 reportDamage(channel, item, side) {
281 channel.countDamage(item);
282 this.trace(side, item.outcome, item.kind, `#${item.number}`);
283 }
284
285 receiveData(item) {
286 this.trace('RECEIVER', 'RECV', `seq=${item.value}`, `#${item.number}`);
287 if (this.selective) this.selectiveReceive(item.value, item.payload);
288 else this.goBackNReceive(item.value, item.payload);
289 }
290
291 goBackNReceive(seq, payload) {
292 if (seq === this.expected % this.modulus) {
293 this.deliver(payload);
294 this.sendControl('ack', seq);
295 return;
296 }
297 this.discard(seq);
298 if (this.config.nak && !this.nakSent) {
299 this.sendControl('nak', this.expected % this.modulus);
300 this.nakSent = true;
301 } else if (this.expected > 0) {
302 this.sendControl('ack', (this.expected - 1) % this.modulus);
303 }
304 }
305
306 selectiveReceive(seq, payload) {
307 const offset = mod(seq - this.expected, this.modulus);
308 if (offset >= this.config.window) {
309 this.discard(seq);
310 if (mod(this.expected - seq, this.modulus) <= this.config.window) {
311 this.sendControl('ack', seq);
312 }
313 return;
314 }
315 const position = this.expected + offset;
316 if (this.buffer.has(position)) {
317 this.discard(seq);
318 this.sendControl('ack', seq);
319 } else if (offset === 0) {
320 this.deliver(payload);
321 while (this.buffer.has(this.expected)) {
322 const waiting = this.buffer.get(this.expected);
323 this.buffer.delete(this.expected);
324 this.deliver(waiting);
325 }
326 this.sendControl('ack', seq);
327 } else {
328 this.buffer.set(position, payload);
329 this.maxBuffer = Math.max(this.maxBuffer, this.buffer.size);
330 this.trace('RECEIVER', 'BUFFER', `f=${position}`);
331 this.sendControl('ack', seq);
332 if (this.config.nak && !this.nakSent) {
333 this.sendControl('nak', this.expected % this.modulus);
334 this.nakSent = true;
335 }
336 }
337 }
338
339 deliver(payload) {
340 const position = this.expected;
341 if (payload === position) {
342 ++this.delivered;
343 this.records[payload].delivered = this.tick;
344 this.trace('RECEIVER', 'DELIVER', `f=${position}`);
345 } else {
346 ++this.errors;
347 this.trace('RECEIVER', 'DELIVER', `f=${position}`, `wrong=${payload}`);
348 }
349 ++this.expected;
350 this.nakSent = false;
351 }
352
353 discard(seq) {
354 ++this.discarded;
355 this.trace('RECEIVER', 'DISCARD', `seq=${seq}`);
356 }
357
358 sendControl(kind, value) {
359 const item = this.controlChannel.send(kind, value, null, this.tick, this.generator);
360 if (kind === 'nak') ++this.naks;
361 this.trace('RECEIVER', kind.toUpperCase(), `a=${value}`, `#${item.number}`);
362 }
363
364 receiveControl(item) {
365 const before = [this.cwnd, this.ssthresh];
366 if (item.kind === 'ack') {
367 if (this.selective) this.selectiveAck(item);
368 else this.goBackNAck(item);
369 } else if (this.selective) this.selectiveNak(item);
370 else this.goBackNNak(item);
371 this.reportWindow(before);
372 }
373
374 controlLine(item, accepted) {
375 const parts = [`${accepted ? 'GOT' : 'OLD'}${item.kind.toUpperCase()}`,
376 `a=${item.value}`, `#${item.number}`];
377 if (accepted) parts.push(`base=${this.base}`);
378 this.trace('SENDER', ...parts);
379 }
380
381 goBackNAck(item) {
382 const frame = this.frameOf(item.value);
383 if (frame < this.framesSent) {
384 this.growWindow(frame - this.base + 1);
385 this.moveBase(frame + 1);
386 if (this.nextFrame < this.base) this.nextFrame = this.base;
387 this.timer = this.base === this.nextFrame ? null : this.tick + this.timerLength;
388 this.controlLine(item, true);
389 return;
390 }
391 this.controlLine(item, false);
392 if (this.config.dupack === null || this.base < 1) return;
393 if (item.value === (this.base - 1) % this.modulus) {
394 ++this.duplicates;
395 if (this.duplicates === this.config.dupack) {
396 ++this.fastRetransmits;
397 this.trace('SENDER', 'FASTRETX', `f=${this.base}`);
398 this.goBack();
399 this.halveWindow();
400 }
401 }
402 }
403
404 goBackNNak(item) {
405 const frame = this.frameOf(item.value);
406 if (frame > this.framesSent) {
407 this.controlLine(item, false);
408 return;
409 }
410 this.growWindow(frame - this.base);
411 this.moveBase(frame);
412 this.goBack();
413 this.halveWindow();
414 this.controlLine(item, true);
415 }
416
417 selectiveAck(item) {
418 const frame = this.frameOf(item.value);
419 if (frame >= this.framesSent || this.acked[frame]) {
420 this.controlLine(item, false);
421 return;
422 }
423 this.acked[frame] = true;
424 this.records[frame].acked = this.tick;
425 this.timers[frame] = null;
426 this.queue = this.queue.filter(waiting => waiting !== frame);
427 this.growWindow(1);
428 const previous = this.base;
429 while (this.base < this.framesSent && this.acked[this.base]) ++this.base;
430 if (this.base > previous) this.timerLength = this.config.timeout;
431 this.controlLine(item, true);
432 }
433
434 selectiveNak(item) {
435 const frame = this.frameOf(item.value);
436 if (frame >= this.framesSent || this.acked[frame]) {
437 this.controlLine(item, false);
438 return;
439 }
440 this.timers[frame] = null;
441 if (!this.queue.includes(frame)) this.queue.push(frame);
442 this.halveWindow();
443 this.controlLine(item, true);
444 }
445
446 moveBase(newBase) {
447 if (newBase <= this.base) return;
448 for (let frame = this.base; frame < newBase; ++frame) this.records[frame].acked = this.tick;
449 this.base = newBase;
450 this.timerLength = this.config.timeout;
451 this.duplicates = 0;
452 }
453
454 goBack() { this.nextFrame = this.base; this.timer = null; this.duplicates = 0; }
455
456 growWindow(count) {
457 if (!this.config.congestion) return;
458 for (let i = 0; i < count; ++i) {
459 if (this.cwnd >= this.config.window) continue;
460 if (this.cwnd < this.ssthresh) ++this.cwnd;
461 else {
462 ++this.roundAcks;
463 if (this.roundAcks === this.cwnd) {
464 ++this.cwnd;
465 this.roundAcks = 0;
466 }
467 }
468 }
469 }
470
471 halveWindow() {
472 if (!this.config.congestion) return;
473 this.ssthresh = Math.max(Math.floor(this.cwnd / 2), 1);
474 this.cwnd = this.ssthresh;
475 this.roundAcks = 0;
476 }
477
478 collapseWindow() {
479 if (!this.config.congestion) return;
480 this.ssthresh = Math.max(Math.floor(this.cwnd / 2), 1);
481 this.cwnd = 1;
482 this.roundAcks = 0;
483 }
484
485 reportWindow(before) {
486 if (this.cwnd !== before[0] || this.ssthresh !== before[1]) {
487 this.trace('SENDER', 'CWND', this.cwnd, `ssthresh=${this.ssthresh}`);
488 }
489 }
490
491 processTimeouts() {
492 if (this.selective) {
493 for (let frame = this.base; frame < this.framesSent; ++frame) {
494 if (this.timers[frame] === this.tick) {
495 this.timers[frame] = null;
496 this.queue.push(frame);
497 this.timeOut(frame);
498 }
499 }
500 } else if (this.timer === this.tick) {
501 this.timeOut(this.base);
502 this.goBack();
503 }
504 }
505
506 timeOut(frame) {
507 const before = [this.cwnd, this.ssthresh];
508 ++this.timeouts;
509 this.trace('SENDER', 'TIMEOUT', `f=${frame}`);
510 if (this.config.backoff) this.timerLength = Math.min(2 * this.timerLength, 8 * this.config.timeout);
511 this.collapseWindow();
512 this.reportWindow(before);
513 }
514
515 maySendNew(frame) {
516 return frame < this.config.frames && frame < this.base + this.effectiveWindow() &&
517 frame * this.config.interval <= this.tick;
518 }
519
520 sendData() { if (this.selective) this.selectiveSend(); else this.goBackNSend(); }
521
522 goBackNSend() {
523 const frame = this.nextFrame;
524 if (frame >= this.config.frames || frame >= this.base + this.effectiveWindow()) return;
525 if (frame >= this.framesSent && frame * this.config.interval > this.tick) return;
526 this.transmit(frame);
527 this.framesSent = Math.max(this.framesSent, frame + 1);
528 ++this.nextFrame;
529 if (this.timer === null) this.timer = this.tick + this.timerLength;
530 }
531
532 selectiveSend() {
533 let frame;
534 if (this.queue.length) frame = this.queue.shift();
535 else if (this.maySendNew(this.framesSent)) frame = this.framesSent++;
536 else return;
537 this.transmit(frame);
538 this.timers[frame] = this.tick + this.timerLength;
539 }

541 transmit(frame) {
542 const record = this.records[frame], seq = frame % this.modulus;
543 const item = this.dataChannel.send('data', seq, frame, this.tick, this.generator);
544 ++record.transmissions;
545 let label;
546 if (record.transmissions === 1) {
547 record.first = this.tick;
548 label = 'SEND';
549 } else {
550 ++this.resent;
551 label = 'RESEND';
552 }
553 this.trace('SENDER', label, `f=${frame}`, `seq=${seq}`, `#${item.number}`);
554 }
555
556 senderMark(frame) {
557 let acknowledged, waiting, sent;
558 if (this.selective) {
559 acknowledged = this.acked[frame];
560 waiting = this.queue.includes(frame);
561 sent = frame < this.framesSent;
562 } else {
563 acknowledged = frame < this.base;
564 waiting = this.nextFrame <= frame && frame < this.framesSent;
565 sent = frame < this.nextFrame;
566 }
567 if (acknowledged) return 'A';
568 if (waiting) return 'R';
569 if (sent) return 'O';
570 if (frame < this.base + this.effectiveWindow()) return 'W';
571 return '.';
572 }
573
574 receiverMark(position) {
575 if (position < this.expected) return 'D';
576 if (this.selective) {
577 if (this.buffer.has(position)) return 'B';
578 if (position < this.expected + this.config.window) return 'W';
579 } else if (position === this.expected) return 'W';
580 return '.';
581 }

583 snapshot() {
584 const positions = Array.from({length: this.config.frames}, (_, i) => i);
585 const sender = positions.map(frame => this.senderMark(frame)).join('');
586 const receiver = positions.map(position => this.receiverMark(position)).join('');
587 this.emit('SNAPSHOT', `sender=${sender}`, `receiver=${receiver}`);
588 }

590 dataCount() { return this.dataChannel.sent; }
591 goodputText() { return truncated(this.delivered, this.completion, 2); }
592 efficiencyText() { return truncated(100 * this.delivered, this.dataCount(), 1); }
593 latencyText() {
594 if (this.delivered === 0) return '-';
595 const total = this.records.reduce((sum, record) =>
596 sum + (record.delivered === null ? 0 : record.delivered - record.first), 0);
597 return truncated(total, this.delivered, 2);

599
600 report() {
601 const lines = this.records.map((record, frame) =>
602 `frame ${frame}: seq=${frame % this.modulus} tx=${record.transmissions} ` +
603 `first=${tickText(record.first)} delivered=${tickText(record.delivered)} ` +
604 `acked=${tickText(record.acked)}`);
605 const data = this.dataChannel, control = this.controlChannel;
606 lines.push(`totals: data=${data.sent} resent=${this.resent} control=${control.sent} ` +
607 `nak=${this.naks} timeouts=${this.timeouts} fastretx=${this.fastRetransmits}`);
608 lines.push(`channel: lost_data=${data.lost} lost_ctrl=${control.lost} ` +
609 `corrupt_data=${data.corrupted} corrupt_ctrl=${control.corrupted}`);
610 lines.push(`receiver: delivered=${this.delivered} errors=${this.errors} ` +
611 `discarded=${this.discarded} max_buffer=${this.maxBuffer}`);
612 const completion = this.completion === null ? '-' : String(this.completion);
613 const goodput = this.completion === null ? '-' : this.goodputText();
614 const utilization = truncated(100 * data.sent, this.tick + 1, 1);
615 lines.push(`result: completion=${completion} goodput=${goodput} ` +
616 `efficiency=${this.efficiencyText()}% utilization=${utilization}% latency=${this.latencyText()}`);
617 return lines;
618 }
619}
620
621class Scenario {
622 constructor(name, settings) {
623 this.name = name;
624 this.settings = settings;
625 this.config = null;
626 this.simulation = null;
627 }
628
629 completed() { return this.simulation !== null && this.simulation.completion !== null; }
630 status() { return this.simulation === null ? 'invalid' : 'aborted'; }
631
632 execute(output) {
633 output.push(`== ${this.name} ==`);
634 try { this.config = checkSettings(this.settings); }
635 catch (error) {
636 if (!(error instanceof SettingsError)) throw error;
637 output.push(`error: ${error.message}`);
638 return;
639 }
640 output.push(this.config.configLine(), this.config.optionsLine());
641 this.simulation = new Simulation(this.config);
642 this.simulation.run();
643 output.push(...this.simulation.lines, ...this.simulation.report());
644 }
645}
646
647function rankingLines(scenarios) {
648 const lines = ['== ranking =='];
649 const completed = scenarios.filter(scenario => scenario.completed());
650 completed.sort((a, b) => {
651 const x = a.simulation, y = b.simulation;
652 const first = y.delivered * x.completion - x.delivered * y.completion;
653 if (first !== 0) return first;
654 return y.delivered * x.dataCount() - x.delivered * y.dataCount();
655 });
656 completed.forEach((scenario, i) => {
657 const run = scenario.simulation;
658 lines.push(`${i + 1}. ${scenario.name} ${scenario.config.protocol} ` +
659 `goodput=${run.goodputText()} efficiency=${run.efficiencyText()}%`);
660 });
661 for (const scenario of scenarios) {
662 if (!scenario.completed()) lines.push(`- ${scenario.name} ${scenario.status()}`);
663 }
664 return lines;
665}
666
667function protocolLines(scenarios) {
668 const lines = ['== protocols =='];
669 for (const protocol of PROTOCOLS) {
670 const runs = scenarios.filter(s => s.completed() && s.config.protocol === protocol)
671 .map(s => s.simulation);
672 if (!runs.length) continue;
673 const delivered = runs.reduce((sum, r) => sum + r.delivered, 0);
674 const ticks = runs.reduce((sum, r) => sum + r.completion, 0);
675 const data = runs.reduce((sum, r) => sum + r.dataCount(), 0);
676 lines.push(`${protocol}: runs=${runs.length} goodput=${truncated(delivered, ticks, 2)} ` +
677 `efficiency=${truncated(100 * delivered, data, 1)}%`);
678 }
679 return lines;
680}

682function main() {
683 const lines = readFileSync(0, 'utf8').trimEnd().split(/\r?\n/);
684 const count = Number(lines[0]);
685 let index = 1;
686 const scenarios = [];
687 for (let i = 0; i < count; ++i) {
688 const name = lines[index++].split(' ')[1];
689 const settings = [];
690 while (lines[index] !== 'END') {
691 const [key, ...values] = lines[index++].split(' ');
692 settings.push([key, values]);
693 }
694 ++index;
695 scenarios.push(new Scenario(name, settings));
696 }
697 const output = [];
698 for (const scenario of scenarios) scenario.execute(output);
699 output.push(...rankingLines(scenarios), ...protocolLines(scenarios));
700 process.stdout.write(output.join('\n') + '\n');
701}
