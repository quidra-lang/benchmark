11 static String fixed(long numerator, long denominator, int digits) {
12 long scale = digits == 2 ? 100 : 10;
13 long v = denominator == 0 ? 0 : numerator * scale / denominator;
14 return v / scale + "." + String.format(Locale.ROOT, "%0" + digits + "d", v % scale);
15 }

22 static class Scenario {
23 String name, protocol = "", error, trace = "ALL";
24 Map<String, List<String>> values = new HashMap<>();
25 int n, w, k, timeout, delay, interval = 0, limit = 1000, dupack = 0;
26 boolean nak, backoff, congestion, unsafe, random;

32 Scenario(String name) {
33 this.name = name;
34 }
35 int number(String key) {
36 return Integer.parseInt(values.get(key).get(0));
37 }
38 boolean on(String key) {
39 return values.containsKey(key) && values.get(key).get(0).equals("ON");

41 void line(String[] words) {
42 if (error != null)
43 return;
44 String key = words[0];
45 int position = Arrays.asList(KEYS).indexOf(key);
46 if (position < 0) {
47 error = "unknown key " + key;
48 return;
49 }
50 if (values.containsKey(key)) {
51 error = "duplicate key " + key;
52 return;
53 }
54 int count = words.length - 1;
55 if (key.equals("RANDOM") ? count != 3 : position >= 15 ? count < 1 : count != 1) {
56 error = "wrong number of values for " + key;
57 return;
58 }
59 values.put(key, Arrays.asList(words).subList(1, words.length));
60 }
61 boolean valid(String key) {
62 List<String> v = values.get(key);
63 if (key.equals("PROTOCOL"))
64 return List.of("SW", "GBN", "SR").contains(v.get(0));
65 if (List.of("NAK", "BACKOFF", "CONGESTION", "UNSAFE").contains(key))
66 return List.of("ON", "OFF").contains(v.get(0));
67 if (key.equals("TRACE"))
68 return List.of("ALL", "SENDER", "RECEIVER", "NONE").contains(v.get(0));
69 if (key.equals("RANDOM")) {
70 long s = Long.parseLong(v.get(0));
71 long p = Long.parseLong(v.get(1)), c = Long.parseLong(v.get(2));
72 return s >= 1 && s <= 2147483646L && p >= 0 && c >= 0 && p + c <= 100;
73 }
74 int pos = Arrays.asList(KEYS).indexOf(key);
75 if (pos >= 15) {
76 Set<Integer> found = new HashSet<>();
77 Map<Integer, Integer> extra = new HashMap<>();
78 boolean delayed = key.startsWith("DELAY");
79 for (String item : v) {
80 String[] pair = item.split(":");
81 int a = Integer.parseInt(pair[0]);
82 if (a < (key.equals("SNAPSHOT") ? 0 : 1) ||
83 a > (key.equals("SNAPSHOT") ? 2000 : 9999) || !found.add(a))
84 return false;
85 if (delayed) {
86 int e = Integer.parseInt(pair[1]);
87 if (e < 1 || e > 50)
88 return false;
89 extra.put(a, e);
90 }
91 }
92 lists.put(key, found);
93 if (delayed)
94 delays.put(key, extra);
95 return true;
96 }
97 int a = number(key), lo = 1, hi = 0;
98 switch (key) {
99 case "FRAMES":
100 hi = 60;
101 break;
102 case "WINDOW":
103 hi = 32;
104 break;
105 case "SEQBITS":
106 hi = 6;
107 break;
108 case "TIMEOUT":
109 hi = 50;
110 break;
111 case "DELAY", "DUPACK":
112 hi = 10;
113 break;
114 case "INTERVAL":
115 lo = 0;
116 hi = 10;
117 break;
118 case "MAXTICKS":
119 hi = 2000;
120 break;
121 }
122 return a >= lo && a <= hi;
123 }
124 void check() {
125 if (error != null)
126 return;
127 for (int i = 0; i < 6; i++)
128 if (!values.containsKey(KEYS[i])) {
129 error = "missing key " + KEYS[i];
130 return;
131 }
132 for (String key : KEYS)
133 if (values.containsKey(key) && !valid(key)) {
134 error = "bad value for " + key;
135 return;
136 }
137 protocol = values.get("PROTOCOL").get(0);
138 n = number("FRAMES");
139 w = number("WINDOW");
140 k = number("SEQBITS");
141 timeout = number("TIMEOUT");
142 delay = number("DELAY");
143 if (values.containsKey("INTERVAL"))
144 interval = number("INTERVAL");
145 if (values.containsKey("MAXTICKS"))
146 limit = number("MAXTICKS");
147 if (values.containsKey("DUPACK"))
148 dupack = number("DUPACK");
149 nak = on("NAK");
150 backoff = on("BACKOFF");
151 congestion = on("CONGESTION");
152 unsafe = on("UNSAFE");
153 if (values.containsKey("TRACE"))
154 trace = values.get("TRACE").get(0);
155 random = values.containsKey("RANDOM");
156 if (random) {
157 List<String> v = values.get("RANDOM");
158 seed = Long.parseLong(v.get(0));
159 loss = Integer.parseInt(v.get(1));
160 corruption = Integer.parseInt(v.get(2));
161 }
162 if (protocol.equals("SR") && dupack != 0) {
163 error = "DUPACK not supported for SR";
164 return;
165 }
166 int upper = protocol.equals("SW") ? 1
167 : protocol.equals("GBN") ? (1 << k) - 1
168 : 1 << (k - 1);
169 if (w > upper && !(unsafe && !protocol.equals("SW"))) {
170 error = "window " + w + " exceeds limit " + upper + " for " + protocol;
171 return;
172 }
173 for (String kind : List.of("DATA", "CTRL")) {
174 TreeSet<Integer> both = new TreeSet<>(lists.getOrDefault("LOSE" + kind, Set.of()));
175 both.retainAll(lists.getOrDefault("CORRUPT" + kind, Set.of()));
176 if (!both.isEmpty()) {
177 error = (kind.equals("DATA") ? "data" : "control") + " #" + both.first() +
178 " both lost and corrupted";
179 return;
180 }
181 }
182 }
183 boolean contains(String key, int i) {
184 return lists.getOrDefault(key, Set.of()).contains(i);
185 }
186 int extra(String key, int i) {
187 return delays.getOrDefault(key, Map.of()).getOrDefault(i, 0);
188 }
189 }
190 record Event(int arrival, int number, int value, int outcome, String kind) {}
191 static class Simulation {
192 Scenario s;
193 int t, M, base, high, next, timer = -1, C, duplicates, r;
194 boolean sr, nakSent;
195 int cwnd = 1, threshold, growth;
196 int control, naks, resent, timeouts, fast, lostData, lostControl, corruptData,
197 corruptControl;
198 int errors, discarded, maxBuffer, latency;
199 long random;
200 int[] tx, first, delivery, acked, timers;
201 boolean[] marks;
202 List<Integer> queue = new ArrayList<>();
203 Map<Integer, Integer> buffer = new HashMap<>();
204 List<Event> dataEvents = new ArrayList<>(), controlEvents = new ArrayList<>();
205 Simulation(Scenario s) {
206 this.s = s;
207 M = 1 << s.k;
208 C = s.timeout;
209 threshold = s.w;
210 random = s.seed;
211 sr = s.protocol.equals("SR");
212 tx = new int[s.n];
213 first = new int[s.n];
214 delivery = new int[s.n];
215 acked = new int[s.n];
216 timers = new int[s.n];
217 marks = new boolean[s.n];
218 Arrays.fill(first, -1);
219 Arrays.fill(delivery, -1);
220 Arrays.fill(acked, -1);
221 Arrays.fill(timers, -1);
222 }
223 void emit(String side, String text) {
224 if (side.equals("ALWAYS") || s.trace.equals("ALL") || s.trace.equals(side))
225 output.add(t + " " + text);
226 }
227 int effective() {
228 return s.congestion ? Math.min(s.w, cwnd) : s.w;
229 }
230 int outcome(String kind, int number) {
231 int result = 0;
232 if (s.random) {
233 random = random * 48271L % 2147483647L;
234 int v = (int)(random % 100);
235 result = v < s.loss ? 1 : v < s.loss + s.corruption ? 2 : 0;
236 }
237 String suffix = kind.equals("data") ? "DATA" : "CTRL";
238 if (s.contains("LOSE" + suffix, number))
239 result = 1;
240 if (s.contains("CORRUPT" + suffix, number))
241 result = 2;
242 return result;
243 }
244 void control(boolean nak, int a) {
245 int id = ++control;
246 if (nak)
247 naks++;
248 String kind = nak ? "nak" : "ack";
249 emit("RECEIVER", kind.toUpperCase(Locale.ROOT) + " a=" + a + " #" + id);
250 controlEvents.add(
251 new Event(t + s.delay + s.extra("DELAYCTRL", id), id, a, outcome(kind, id), kind));
252 }
253 void deliver(int payload) {
254 if (payload == r) {
255 emit("RECEIVER", "DELIVER f=" + r);
256 if (delivery[payload] < 0)
257 delivery[payload] = t;
258 s.delivered++;
259 latency += t - first[payload];
260 } else {
261 emit("RECEIVER", "DELIVER f=" + r + " wrong=" + payload);
262 errors++;
263 }
264 r++;
265 nakSent = false;
266 }
267 void discard(int seq) {
268 discarded++;
269 emit("RECEIVER", "DISCARD seq=" + seq);
270 }
271 void receive(Event e) {
272 int seq = e.value() % M;
273 emit("RECEIVER", "RECV seq=" + seq + " #" + e.number());
274 if (!sr) {
275 if (seq == r % M) {
276 deliver(e.value());
277 control(false, seq);
278 } else {
279 discard(seq);
280 if (s.nak && !nakSent) {
281 control(true, r % M);
282 nakSent = true;
283 } else if (r > 0)
284 control(false, (r - 1) % M);
285 }
286 } else {
287 int offset = mod(seq - r, M);
288 if (offset < s.w) {
289 int position = r + offset;
290 if (buffer.containsKey(position)) {
291 discard(seq);
292 control(false, seq);
293 } else if (offset == 0) {
294 deliver(e.value());
295 while (buffer.containsKey(r))
296 deliver(buffer.remove(r));
297 control(false, seq);
298 } else {
299 buffer.put(position, e.value());
300 maxBuffer = Math.max(maxBuffer, buffer.size());
301 emit("RECEIVER", "BUFFER f=" + position);
302 control(false, seq);
303 if (s.nak && !nakSent) {
304 control(true, r % M);
305 nakSent = true;
306 }
307 }
308 } else {
309 discard(seq);
310 if (mod(r - seq, M) <= s.w)
311 control(false, seq);
312 }
313 }
314 }
315 void goBack() {
316 next = base;
317 timer = -1;
318 duplicates = 0;
319 }
320 void increase(int count) {
321 if (!s.congestion)
322 return;
323 for (int i = 0; i < count && cwnd < s.w; i++) {
324 if (cwnd < threshold)
325 cwnd++;
326 else if (++growth == cwnd) {
327 cwnd++;
328 growth = 0;
329 }
330 }
331 }
332 void decrease(boolean timeout) {
333 if (!s.congestion)
334 return;
335 threshold = Math.max(cwnd / 2, 1);
336 cwnd = timeout ? 1 : threshold;
337 growth = 0;
338 }
339 void cwndLine(int oldC, int oldS) {
340 if (s.congestion && (oldC != cwnd || oldS != threshold))
341 emit("SENDER", "CWND " + cwnd + " ssthresh=" + threshold);
342 }
343 void acknowledge(Event e) {
344 int oldC = cwnd, oldS = threshold, oldBase = base;
345 int a = e.value(), f = base + mod(a - base, M);
346 boolean nak = e.kind().equals("nak"), accepted = false;
347 if (sr) {
348 if (f < high && !marks[f]) {
349 accepted = true;
350 timers[f] = -1;
351 if (nak) {
352 if (!queue.contains(f))
353 queue.add(f);
354 } else {
355 marks[f] = true;
356 if (acked[f] < 0)
357 acked[f] = t;
358 queue.remove(Integer.valueOf(f));
359 while (base < high && marks[base])
360 base++;
361 increase(1);
362 }
363 }
364 } else if (nak ? f <= high : f < high) {
365 accepted = true;
366 base = nak ? f : f + 1;
367 increase(base - oldBase);
368 for (int i = oldBase; i < base; i++)
369 if (acked[i] < 0)
370 acked[i] = t;
371 if (nak)
372 goBack();
373 else if (next < base)
374 next = base;
375 }
376 if (base > oldBase) {
377 C = s.timeout;
378 duplicates = 0;
379 }
380 if (accepted && !sr && !nak)
381 timer = base == next ? -1 : t + C;
382 String label = nak ? "NAK" : "ACK";
383 emit("SENDER", (accepted ? "GOT" : "OLD") + label + " a=" + a + " #" + e.number() +
384 (accepted ? " base=" + base : ""));
385 if (accepted && nak)
386 decrease(false);
387 if (!accepted && !nak && !sr && s.dupack > 0 && base >= 1 && a == (base - 1) % M) {
388 duplicates++;
389 if (duplicates == s.dupack) {
390 fast++;
391 emit("SENDER", "FASTRETX f=" + base);
392 goBack();
393 decrease(false);
394 }
395 }
396 cwndLine(oldC, oldS);
397 }
398 void timeout(int f) {
399 int oldC = cwnd, oldS = threshold;
400 timeouts++;
401 emit("SENDER", "TIMEOUT f=" + f);
402 if (sr) {
403 timers[f] = -1;
404 queue.add(f);
405 } else
406 goBack();
407 if (s.backoff)
408 C = Math.min(C * 2, 8 * s.timeout);
409 decrease(true);
410 cwndLine(oldC, oldS);
411 }
412 void send(int f) {
413 int id = ++s.data;
414 boolean again = tx[f]++ > 0;
415 if (again)
416 resent++;
417 else
418 first[f] = t;
419 emit("SENDER", (again ? "RESEND" : "SEND") + " f=" + f + " seq=" + f % M + " #" + id);
420 dataEvents.add(new Event(t + s.delay + s.extra("DELAYDATA", id), id, f,
421 outcome("data", id), "data"));
422 }
423 void arrivals(List<Event> events, boolean data) {
424 List<Event> ready = events.stream()
425 .filter(e -> e.arrival() == t)
426 .sorted(Comparator.comparingInt(Event::number))
427 .toList();
428 events.removeAll(ready);
429 for (Event e : ready) {
430 if (e.outcome() != 0) {
431 boolean lost = e.outcome() == 1;
432 if (data) {
433 if (lost)
434 lostData++;
435 else
436 corruptData++;
437 } else {
438 if (lost)
439 lostControl++;
440 else
441 corruptControl++;
442 }
443 emit(data ? "RECEIVER" : "SENDER",
444 (lost ? "LOST" : "CORRUPT") + " " + e.kind() + " #" + e.number());
445 } else if (data)
446 receive(e);
447 else
448 acknowledge(e);
449 }
450 }
451 void snapshot() {
452 StringBuilder x = new StringBuilder(), y = new StringBuilder();
453 for (int f = 0; f < s.n; f++) {
454 boolean ack = sr ? marks[f] : f < base;
455 boolean retransmit = sr ? queue.contains(f) : next <= f && f < high;
456 boolean sent = sr ? f < high : f < next;
457 x.append(ack ? 'A'
458 : retransmit ? 'R'
459 : sent ? 'O'
460 : f < base + effective() ? 'W'
461 : '.');
462 y.append(f < r ? 'D'
463 : sr && buffer.containsKey(f) ? 'B'
464 : (sr ? f < r + s.w : f == r) ? 'W'
465 : '.');
466 }
467 emit("ALWAYS", "SNAPSHOT sender=" + x + " receiver=" + y);
468 }
469 void run() {
470 for (t = 0; t <= s.limit; t++) {
471 arrivals(dataEvents, true);
472 arrivals(controlEvents, false);
473 boolean done = base == s.n;
474 if (!done) {
475 if (sr) {
476 for (int f = base; f < high; f++)
477 if (timers[f] == t)
478 timeout(f);
479 } else if (timer == t)
480 timeout(base);
481 if (sr) {
482 int f = -1;
483 if (!queue.isEmpty())
484 f = queue.remove(0);
485 else if (high < s.n && high < base + effective() && high * s.interval <= t)
486 f = high++;
487 if (f >= 0) {
488 send(f);
489 timers[f] = t + C;
490 }
491 } else if (next < s.n && next < base + effective() &&
492 (next < high || next * s.interval <= t)) {
493 send(next);
494 high = Math.max(high, ++next);
495 if (timer < 0)
496 timer = t + C;
497 }
498 }
499 if (s.contains("SNAPSHOT", t))
500 snapshot();
501 if (done || t == s.limit) {
502 s.end = t;
503 if (done)
504 s.completion = t;
505 emit("ALWAYS", done ? "DONE" : "ABORT");
506 break;
507 }
508 }
509 for (int f = 0; f < s.n; f++)
510 output.add("frame " + f + ": seq=" + f % M + " tx=" + tx[f] +
511 " first=" + tick(first[f]) + " delivered=" + tick(delivery[f]) +
512 " acked=" + tick(acked[f]));
513 output.add("totals: data=" + s.data + " resent=" + resent + " control=" + control +
514 " nak=" + naks + " timeouts=" + timeouts + " fastretx=" + fast);
515 output.add("channel: lost_data=" + lostData + " lost_ctrl=" + lostControl +
516 " corrupt_data=" + corruptData + " corrupt_ctrl=" + corruptControl);
517 output.add("receiver: delivered=" + s.delivered + " errors=" + errors +
518 " discarded=" + discarded + " max_buffer=" + maxBuffer);
519 output.add("result: completion=" + tick(s.completion) + " goodput=" +
520 (s.completion < 0 ? "-" : fixed(s.delivered, s.completion, 2)) +
521 " efficiency=" + fixed(s.delivered * 100L, s.data, 1) +
522 "% utilization=" + fixed(s.data * 100L, s.end + 1, 1) +
523 "% latency=" + (s.delivered == 0 ? "-" : fixed(latency, s.delivered, 2)));
524 }
525 }
526 public static void main(String[] args) throws Exception {
527 BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
528 int count = Integer.parseInt(input.readLine());
529 List<Scenario> scenarios = new ArrayList<>();
530 for (int q = 0; q < count; q++) {
531 Scenario s = new Scenario(input.readLine().split(" ")[1]);
532 String line;
533 while (!(line = input.readLine()).equals("END"))
534 s.line(line.split(" "));
535 s.check();
536 scenarios.add(s);
537 output.add("== " + s.name + " ==");
538 if (s.error != null) {
539 output.add("error: " + s.error);
540 continue;
541 }
542 output.add("config: " + s.protocol + " n=" + s.n + " w=" + s.w + " k=" + s.k +
543 " T=" + s.timeout + " d=" + s.delay);
544 output.add("options: interval=" + s.interval + " limit=" + s.limit +
545 " dupack=" + (s.dupack == 0 ? "off" : s.dupack) +
546 " nak=" + (s.nak ? "on" : "off") + " backoff=" + (s.backoff ? "on" : "off") +
547 " congestion=" + (s.congestion ? "on" : "off") +
548 " unsafe=" + (s.unsafe ? "on" : "off") +
549 " trace=" + s.trace.toLowerCase(Locale.ROOT) + " random=" +
550 (s.random ? s.seed + "/" + s.loss + "/" + s.corruption : "off"));
551 new Simulation(s).run();
552 }
553 output.add("== ranking ==");
554 List<Scenario> completed =
555 new ArrayList<>(scenarios.stream().filter(s -> s.completion >= 0).toList());
556 completed.sort((a, b) -> {
557 int c =
558 Long.compare((long)b.delivered * a.completion, (long)a.delivered * b.completion);
559 return c != 0 ? c
560 : Long.compare((long)b.delivered * a.data, (long)a.delivered * b.data);
561 });
562 for (int i = 0; i < completed.size(); i++) {
563 Scenario s = completed.get(i);
564 output.add((i + 1) + ". " + s.name + " " + s.protocol +
565 " goodput=" + fixed(s.delivered, s.completion, 2) +
566 " efficiency=" + fixed(s.delivered * 100L, s.data, 1) + "%");
567 }
568 for (Scenario s : scenarios)
569 if (s.completion < 0)
570 output.add("- " + s.name + (s.error != null ? " invalid" : " aborted"));
571 output.add("== protocols ==");
572 for (String p : List.of("SW", "GBN", "SR")) {
573 int runs = 0, d = 0, t = 0, x = 0;
574 for (Scenario s : completed)
575 if (s.protocol.equals(p)) {
576 runs++;
577 d += s.delivered;
578 t += s.completion;
579 x += s.data;
580 }
581 if (runs > 0)
582 output.add(p + ": runs=" + runs + " goodput=" + fixed(d, t, 2) +
583 " efficiency=" + fixed(d * 100L, x, 1) + "%");
584 }
585 System.out.print(String.join("\n", output) + "\n");
586 }
