4const KEYS: [&str; 22] = ["PROTOCOL", "FRAMES", "WINDOW", "SEQBITS", "TIMEOUT", "DELAY", "INTERVAL", "MAXTICKS", "DUPACK", "NAK", "BACKOFF", "CONGESTION", "UNSAFE", "TRACE", "RANDOM", "LOSEDATA", "LOSECTRL", "CORRUPTDATA", "CORRUPTCTRL", "DELAYDATA", "DELAYCTRL", "SNAPSHOT"];
5fn fixed(numerator: i64, denominator: i64, digits: usize) -> String {
6 let scale = if digits == 2 { 100 } else { 10 };
7 let value = if denominator == 0 { 0 } else { numerator * scale / denominator };
8 format!("{}.{:0width$}", value / scale, value % scale, width = digits)
9}
10fn stamp(tick: i32) -> String { if tick < 0 { "-".into() } else { tick.to_string() } }

36fn validate(name: &str, lines: &[Vec<&str>]) -> Result<Config, String> {
37 let mut values = BTreeMap::<&str, Vec<&str>>::new();
38 for line in lines {
39 let key = line[0];
40 let Some(index) = KEYS.iter().position(|k| *k == key) else { return Err(format!("unknown key {key}")); };
41 if values.contains_key(key) { return Err(format!("duplicate key {key}")); }
42 let count = line.len() - 1;
43 if if key == "RANDOM" { count != 3 } else if index >= 15 { count == 0 } else { count != 1 } {
44 return Err(format!("wrong number of values for {key}"));
45 }
46 values.insert(key, line[1..].to_vec());
47 }
48 for key in &KEYS[..6] { if !values.contains_key(key) { return Err(format!("missing key {key}")); } }
49 for (index, key) in KEYS.iter().enumerate() {
50 let Some(v) = values.get(key) else { continue; };
51 let valid = match index {
52 0 => ["SW", "GBN", "SR"].contains(&v[0]),
53 9..=12 => ["ON", "OFF"].contains(&v[0]),
54 13 => ["ALL", "SENDER", "RECEIVER", "NONE"].contains(&v[0]),
55 14 => {
56 let seed: i64 = v[0].parse().unwrap();
57 let loss: i64 = v[1].parse().unwrap(); let corruption: i64 = v[2].parse().unwrap();
58 (1..=2147483646).contains(&seed) && loss >= 0 && corruption >= 0 && loss + corruption <= 100
59 }
60 15..=21 => {
61 let mut seen = BTreeSet::new();
62 let mut valid = true;
63 for value in v {
64 let parts: Vec<&str> = value.split(':').collect();
65 let number: i32 = parts[0].parse().unwrap();
66 let (low, high) = if index == 21 { (0, 2000) } else { (1, 9999) };
67 if !(low..=high).contains(&number) || !seen.insert(number) { valid = false; }
68 if index == 19 || index == 20 {
69 let extra: i32 = parts[1].parse().unwrap(); if !(1..=50).contains(&extra) { valid = false; }
70 }
71 }
72 valid
73 }
74 _ => {
75 let number: i32 = v[0].parse().unwrap();
76 let low = if index == 6 { 0 } else { 1 };
77 let high = match index { 1 => 60, 2 => 32, 3 => 6, 4 => 50, 7 => 2000, _ => 10 };
78 (low..=high).contains(&number)
79 }
80 };
81 if !valid { return Err(format!("bad value for {key}")); }
82 }
83 let get = |key: &str, fallback: &str| values.get(key).map_or(fallback, |v| v[0]).to_string();
84 let number = |key, fallback| get(key, fallback).parse::<i32>().unwrap();
85 let protocol = get("PROTOCOL", "").to_string();
86 let w = number("WINDOW", "0"); let k = number("SEQBITS", "0");
87 let unsafe_window = get("UNSAFE", "OFF") == "ON";
88 if protocol == "SR" && values.contains_key("DUPACK") { return Err("DUPACK not supported for SR".into()); }
89 let limit = if protocol == "SW" { 1 } else if protocol == "GBN" { (1 << k) - 1 } else { 1 << (k - 1) };
90 if w > limit && !(unsafe_window && protocol != "SW") { return Err(format!("window {w} exceeds limit {limit} for {protocol}")); }
91 let set = |key| values.get(key).map_or_else(BTreeSet::new, |v| v.iter().map(|s| s.parse().unwrap()).collect());
92 let delays = |key| values.get(key).map_or_else(BTreeMap::new, |v| v.iter().map(|s| {
93 let (a, b) = s.split_once(':').unwrap(); (a.parse().unwrap(), b.parse().unwrap())
94 }).collect());
95 let lost_data: BTreeSet<i32> = set("LOSEDATA"); let lost_control: BTreeSet<i32> = set("LOSECTRL");
96 let corrupt_data: BTreeSet<i32> = set("CORRUPTDATA"); let corrupt_control: BTreeSet<i32> = set("CORRUPTCTRL");
97 if let Some(i) = lost_data.intersection(&corrupt_data).next() { return Err(format!("data #{i} both lost and corrupted")); }
98 if let Some(i) = lost_control.intersection(&corrupt_control).next() { return Err(format!("control #{i} both lost and corrupted")); }
99 Ok(Config {
100 name: name.into(), protocol, trace: get("TRACE", "ALL").into(), n: number("FRAMES", "0") as usize, w, k,
101 timeout: number("TIMEOUT", "0"), delay: number("DELAY", "0"), interval: number("INTERVAL", "0"),
102 limit: number("MAXTICKS", "1000"), dupack: number("DUPACK", "0"), nak: get("NAK", "OFF") == "ON",
103 backoff: get("BACKOFF", "OFF") == "ON", congestion: get("CONGESTION", "OFF") == "ON", unsafe_window,
104 random: values.get("RANDOM").map(|v| (v[0].parse().unwrap(), v[1].parse().unwrap(), v[2].parse().unwrap())),
105 lost_data, lost_control, corrupt_data, corrupt_control, delay_data: delays("DELAYDATA"), delay_control: delays("DELAYCTRL"), snapshots: set("SNAPSHOT"),
106 })
107}

168 fn new(c: &'a Config) -> Self {
169 Self { c, output: Vec::new(), data_events: Vec::new(), control_events: Vec::new(), queue: VecDeque::new(), buffer: BTreeMap::new(),
170 acknowledged: vec![false; c.n], timers: vec![-1; c.n], tx: vec![0; c.n], first: vec![-1; c.n], delivered_at: vec![-1; c.n], acked_at: vec![-1; c.n],
171 tick: 0, base: 0, high: 0, next: 0, duration: c.timeout, timer: -1, duplicates: 0, receiver: 0, cwnd: 1, threshold: c.w, credit: 0,
172 random: c.random.map_or(0, |v| v.0), nak_sent: false, data: 0, control: 0, resent: 0, naks: 0, timeouts: 0, fast: 0,
173 lost_data: 0, lost_control: 0, corrupt_data: 0, corrupt_control: 0, delivered: 0, errors: 0, discarded: 0, max_buffer: 0, latency: 0 }
174 }
175 fn sr(&self) -> bool { self.c.protocol == "SR" }
176 fn modulus(&self) -> i32 { 1 << self.c.k }
177 fn effective(&self) -> usize { (if self.c.congestion { self.c.w.min(self.cwnd) } else { self.c.w }) as usize }
178 fn log(&mut self, side: &str, text: String) {
179 if self.c.trace == "ALL" || self.c.trace == side || side == "ALWAYS" { self.output.push(format!("{} {text}", self.tick)); }
180 }
181 fn outcome(&mut self, data: bool, number: i32) -> i32 {
182 let mut outcome = 0;
183 if let Some((_, loss, corruption)) = self.c.random {
184 self.random = self.random * 48271 % 2147483647;
185 let value = (self.random % 100) as i32;
186 outcome = if value < loss { 1 } else if value < loss + corruption { 2 } else { 0 };
187 }
188 if (if data { &self.c.lost_data } else { &self.c.lost_control }).contains(&number) { outcome = 1; }
189 if (if data { &self.c.corrupt_data } else { &self.c.corrupt_control }).contains(&number) { outcome = 2; }
190 outcome
191 }
192 fn send_control(&mut self, nak: bool, value: i32) {
193 self.control += 1; if nak { self.naks += 1; }
194 self.log("RECEIVER", format!("{} a={value} #{}", if nak { "NAK" } else { "ACK" }, self.control));
195 let outcome = self.outcome(false, self.control);
196 self.control_events.push(Event { arrival: self.tick + self.c.delay + self.c.delay_control.get(&self.control).copied().unwrap_or(0), number: self.control, value, outcome, nak });
197 }
198 fn deliver(&mut self, payload: i32) {
199 if payload == self.receiver {
200 self.log("RECEIVER", format!("DELIVER f={}", self.receiver)); self.delivered += 1;
201 let frame = payload as usize;
202 if self.delivered_at[frame] < 0 { self.delivered_at[frame] = self.tick; }
203 self.latency += self.tick - self.first[frame];
204 } else {
205 self.log("RECEIVER", format!("DELIVER f={} wrong={payload}", self.receiver)); self.errors += 1;
206 }
207 self.receiver += 1; self.nak_sent = false;
208 }
209 fn discard(&mut self, sequence: i32) { self.discarded += 1; self.log("RECEIVER", format!("DISCARD seq={sequence}")); }
210 fn receive(&mut self, e: Event) {
211 if e.outcome != 0 {
212 if e.outcome == 1 { self.lost_data += 1; } else { self.corrupt_data += 1; }
213 self.log("RECEIVER", format!("{} data #{}", if e.outcome == 1 { "LOST" } else { "CORRUPT" }, e.number)); return;
214 }
215 let sequence = e.value % self.modulus();
216 self.log("RECEIVER", format!("RECV seq={sequence} #{}", e.number));
217 if !self.sr() {
218 if sequence == self.receiver % self.modulus() { self.deliver(e.value); self.send_control(false, sequence); }
219 else {
220 self.discard(sequence);
221 if self.c.nak && !self.nak_sent { self.send_control(true, self.receiver % self.modulus()); self.nak_sent = true; }
222 else if self.receiver > 0 { self.send_control(false, (self.receiver - 1) % self.modulus()); }
223 }
224 return;
225 }
226 let offset = (sequence - self.receiver).rem_euclid(self.modulus());
227 let position = self.receiver + offset;
228 if offset < self.c.w {
229 if self.buffer.contains_key(&position) { self.discard(sequence); self.send_control(false, sequence); }
230 else if offset == 0 {
231 self.deliver(e.value);
232 while let Some(payload) = self.buffer.remove(&self.receiver) { self.deliver(payload); }
233 self.send_control(false, sequence);
234 } else {
235 self.buffer.insert(position, e.value); self.max_buffer = self.max_buffer.max(self.buffer.len());
236 self.log("RECEIVER", format!("BUFFER f={position}")); self.send_control(false, sequence);
237 if self.c.nak && !self.nak_sent { self.send_control(true, self.receiver % self.modulus()); self.nak_sent = true; }
238 }
239 } else {
240 self.discard(sequence);
241 if (self.receiver - sequence).rem_euclid(self.modulus()) <= self.c.w { self.send_control(false, sequence); }
242 }
243 }
244 fn go_back(&mut self) { self.next = self.base; self.timer = -1; self.duplicates = 0; }
245 fn increase(&mut self, count: usize) {
246 if !self.c.congestion { return; }
247 for _ in 0..count {
248 if self.cwnd >= self.c.w { break; }
249 if self.cwnd < self.threshold { self.cwnd += 1; }
250 else { self.credit += 1; if self.credit == self.cwnd { self.cwnd += 1; self.credit = 0; } }
251 }
252 }
253 fn decrease(&mut self, timeout: bool) {
254 if self.c.congestion { self.threshold = (self.cwnd / 2).max(1); self.cwnd = if timeout { 1 } else { self.threshold }; self.credit = 0; }
255 }
256 fn cwnd_line(&mut self, old: (i32, i32)) {
257 if self.c.congestion && old != (self.cwnd, self.threshold) { self.log("SENDER", format!("CWND {} ssthresh={}", self.cwnd, self.threshold)); }
258 }
259 fn control(&mut self, e: Event) {
260 let kind = if e.nak { "NAK" } else { "ACK" };
261 if e.outcome != 0 {
262 if e.outcome == 1 { self.lost_control += 1; } else { self.corrupt_control += 1; }
263 self.log("SENDER", format!("{} {} #{}", if e.outcome == 1 { "LOST" } else { "CORRUPT" }, kind.to_lowercase(), e.number)); return;
264 }
265 let old = (self.cwnd, self.threshold); let old_base = self.base;
266 let frame = self.base + (e.value - self.base as i32).rem_euclid(self.modulus()) as usize;
267 let accepted = if self.sr() { frame < self.high && !self.acknowledged[frame] } else if e.nak { frame <= self.high } else { frame < self.high };
268 if accepted {
269 if self.sr() {
270 self.timers[frame] = -1;
271 if e.nak { if !self.queue.contains(&frame) { self.queue.push_back(frame); } }
272 else {
273 self.acknowledged[frame] = true; self.queue.retain(|f| *f != frame);
274 if self.acked_at[frame] < 0 { self.acked_at[frame] = self.tick; }
275 while self.base < self.high && self.acknowledged[self.base] { self.base += 1; }
276 self.increase(1);
277 }
278 } else {
279 self.base = frame + usize::from(!e.nak); self.increase(self.base - old_base);
280 for f in old_base..self.base { if self.acked_at[f] < 0 { self.acked_at[f] = self.tick; } }
281 }
282 if self.base > old_base { self.duration = self.c.timeout; self.duplicates = 0; }
283 if !self.sr() {
284 if e.nak { self.go_back(); } else { self.next = self.next.max(self.base); self.timer = if self.base == self.next { -1 } else { self.tick + self.duration }; }
285 }
286 if e.nak { self.decrease(false); }
287 self.log("SENDER", format!("GOT{kind} a={} #{} base={}", e.value, e.number, self.base));
288 } else {
289 self.log("SENDER", format!("OLD{kind} a={} #{}", e.value, e.number));
290 if !self.sr() && !e.nak && self.c.dupack > 0 && self.base >= 1 && e.value == (self.base as i32 - 1) % self.modulus() {
291 self.duplicates += 1;
292 if self.duplicates == self.c.dupack { self.fast += 1; self.log("SENDER", format!("FASTRETX f={}", self.base)); self.go_back(); self.decrease(false); }
293 }
294 }
295 self.cwnd_line(old);
296 }
297 fn timeout(&mut self, frame: usize) {
298 let old = (self.cwnd, self.threshold);
299 self.timeouts += 1; self.log("SENDER", format!("TIMEOUT f={frame}"));
300 if self.sr() { self.timers[frame] = -1; if !self.queue.contains(&frame) { self.queue.push_back(frame); } } else { self.go_back(); }
301 if self.c.backoff { self.duration = (self.duration * 2).min(self.c.timeout * 8); }
302 self.decrease(true); self.cwnd_line(old);
303 }
304 fn send(&mut self, frame: usize) {
305 self.data += 1; let retry = self.tx[frame] > 0;
306 if retry { self.resent += 1; } else { self.first[frame] = self.tick; }
307 self.tx[frame] += 1;
308 self.log("SENDER", format!("{} f={frame} seq={} #{}", if retry { "RESEND" } else { "SEND" }, frame as i32 % self.modulus(), self.data));
309 let outcome = self.outcome(true, self.data);
310 self.data_events.push(Event { arrival: self.tick + self.c.delay + self.c.delay_data.get(&self.data).copied().unwrap_or(0), number: self.data, value: frame as i32, outcome, nak: false });
311 if self.sr() { self.timers[frame] = self.tick + self.duration; } else if self.timer < 0 { self.timer = self.tick + self.duration; }
312 }
313 fn snapshot(&mut self) {
314 let mut sender = String::new(); let mut receiver = String::new();
315 for f in 0..self.c.n {
316 sender.push(if if self.sr() { self.acknowledged[f] } else { f < self.base } { 'A' }
317 else if if self.sr() { self.queue.contains(&f) } else { self.next <= f && f < self.high } { 'R' }
318 else if if self.sr() { f < self.high } else { f < self.next } { 'O' }
319 else if f < self.base + self.effective() { 'W' } else { '.' });
320 receiver.push(if (f as i32) < self.receiver { 'D' } else if self.sr() && self.buffer.contains_key(&(f as i32)) { 'B' }
321 else if if self.sr() { (f as i32) < self.receiver + self.c.w } else { f as i32 == self.receiver } { 'W' } else { '.' });
322 }
323 self.log("ALWAYS", format!("SNAPSHOT sender={sender} receiver={receiver}"));
324 }
325 fn run(mut self, order: usize) -> (Summary, Vec<String>) {
326 let c = self.c;
327 self.output.push(format!("config: {} n={} w={} k={} T={} d={}", c.protocol, c.n, c.w, c.k, c.timeout, c.delay));
328 let on = |v| if v { "on" } else { "off" };
329 let random = c.random.map_or("off".into(), |(s, p, r)| format!("{s}/{p}/{r}"));
330 self.output.push(format!("options: interval={} limit={} dupack={} nak={} backoff={} congestion={} unsafe={} trace={} random={random}", c.interval, c.limit,
331 if c.dupack == 0 { "off".into() } else { c.dupack.to_string() }, on(c.nak), on(c.backoff), on(c.congestion), on(c.unsafe_window), c.trace.to_lowercase()));
332 let mut complete = false;
333 for tick in 0..=c.limit {
334 self.tick = tick;
335 let events: Vec<Event> = self.data_events.iter().filter(|e| e.arrival == tick).cloned().collect();
336 for e in events { self.receive(e); }
337 let events: Vec<Event> = self.control_events.iter().filter(|e| e.arrival == tick).cloned().collect();
338 for e in events { self.control(e); }
339 complete = self.base == c.n;
340 if !complete {
341 if self.sr() { for frame in self.base..self.high { if self.timers[frame] == tick { self.timeout(frame); } } }
342 else if self.timer == tick { self.timeout(self.base); }
343 if self.sr() {
344 if let Some(frame) = self.queue.pop_front() { self.send(frame); }
345 else if self.high < c.n && self.high < self.base + self.effective() && self.high as i32 * c.interval <= tick {
346 let frame = self.high; self.high += 1; self.send(frame);
347 }
348 } else if self.next < c.n && self.next < self.base + self.effective() && (self.next < self.high || self.next as i32 * c.interval <= tick) {
349 let frame = self.next; self.next += 1; self.high = self.high.max(self.next); self.send(frame);
350 }
351 }
352 if c.snapshots.contains(&tick) { self.snapshot(); }
353 if complete || tick == c.limit { self.log("ALWAYS", if complete { "DONE".into() } else { "ABORT".into() }); break; }
354 }
355 for f in 0..c.n { self.output.push(format!("frame {f}: seq={} tx={} first={} delivered={} acked={}", f as i32 % self.modulus(), self.tx[f], stamp(self.first[f]), stamp(self.delivered_at[f]), stamp(self.acked_at[f]))); }
356 self.output.push(format!("totals: data={} resent={} control={} nak={} timeouts={} fastretx={}", self.data, self.resent, self.control, self.naks, self.timeouts, self.fast));
357 self.output.push(format!("channel: lost_data={} lost_ctrl={} corrupt_data={} corrupt_ctrl={}", self.lost_data, self.lost_control, self.corrupt_data, self.corrupt_control));
358 self.output.push(format!("receiver: delivered={} errors={} discarded={} max_buffer={}", self.delivered, self.errors, self.discarded, self.max_buffer));
359 self.output.push(format!("result: completion={} goodput={} efficiency={}% utilization={}% latency={}",
360 if complete { self.tick.to_string() } else { "-".into() }, if complete { fixed(self.delivered as i64, self.tick as i64, 2) } else { "-".into() },
361 fixed(self.delivered as i64 * 100, self.data as i64, 1), fixed(self.data as i64 * 100, self.tick as i64 + 1, 1),
362 if self.delivered == 0 { "-".into() } else { fixed(self.latency as i64, self.delivered as i64, 2) }));
363 (Summary { name: c.name.clone(), protocol: c.protocol.clone(), status: if complete { "complete" } else { "aborted" }, delivered: self.delivered as i64, data: self.data as i64, completion: self.tick as i64, order }, self.output)
364 }
365}
366fn main() {
367 let mut input = String::new(); io::stdin().read_to_string(&mut input).unwrap();
368 let mut lines = input.lines(); let count: usize = lines.next().unwrap().parse().unwrap();
369 let mut output = Vec::new(); let mut results = Vec::new();
370 for order in 0..count {
371 let name = lines.next().unwrap().split_whitespace().nth(1).unwrap();
372 let mut settings = Vec::new();
373 for line in lines.by_ref() { if line == "END" { break; } settings.push(line.split_whitespace().collect()); }
374 output.push(format!("== {name} =="));
375 match validate(name, &settings) {
376 Ok(config) => { let (summary, timeline) = Simulation::new(&config).run(order); results.push(summary); output.extend(timeline); }
377 Err(message) => {
378 output.push(format!("error: {message}"));
379 results.push(Summary { name: name.into(), protocol: String::new(), status: "invalid", delivered: 0, data: 0, completion: 0, order });
380 }
381 }
382 }
383 output.push("== ranking ==".into());
384 let mut completed: Vec<&Summary> = results.iter().filter(|r| r.status == "complete").collect();
385 completed.sort_by(|a, b| (b.delivered * a.completion).cmp(&(a.delivered * b.completion))
386 .then_with(|| (b.delivered * a.data).cmp(&(a.delivered * b.data))).then(a.order.cmp(&b.order)));
387 for (rank, r) in completed.iter().enumerate() { output.push(format!("{}. {} {} goodput={} efficiency={}%", rank + 1, r.name, r.protocol, fixed(r.delivered, r.completion, 2), fixed(r.delivered * 100, r.data, 1))); }
388 for r in &results { if r.status != "complete" { output.push(format!("- {} {}", r.name, r.status)); } }
389 output.push("== protocols ==".into());
390 for protocol in ["SW", "GBN", "SR"] {
391 let runs: Vec<&&Summary> = completed.iter().filter(|r| r.protocol == protocol).collect();
392 if runs.is_empty() { continue; }
