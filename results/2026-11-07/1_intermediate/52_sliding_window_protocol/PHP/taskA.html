43function truncated(int $numerator, int $denominator, int $digits): string
44{
45 $scale = 10 ** $digits;
46 $value = intdiv($numerator * $scale, $denominator);
47 return intdiv($value, $scale) . '.' . str_pad((string) ($value % $scale), $digits, '0', STR_PAD_LEFT);
48}

87function validValue(string $key, array $values): bool
88{
89 if (isset(CHOICES[$key])) return in_array($values[0], CHOICES[$key], true);
90 if (!in_array($key, LIST_KEYS, true) && isset(RANGES[$key])) {
91 return inRange((int) $values[0], RANGES[$key]);
92 }
93 if ($key === 'RANDOM') {
94 [$seed, $loss, $corrupt] = array_map('intval', $values);
95 return inRange($seed, [1, RANDOM_MODULUS - 1]) &&
96 $loss >= 0 && $corrupt >= 0 && $loss + $corrupt <= 100;
97 }
98 if (in_array($key, DELAY_KEYS, true)) {
99 $numbers = [];
100 foreach ($values as $value) {
101 [$number, $extra] = array_map('intval', explode(':', $value));
102 if (!inRange($number, [1, 9999]) || !inRange($extra, [1, 50])) return false;
103 if (isset($numbers[$number])) return false;
104 $numbers[$number] = true;
105 }
106 return true;
107 }
108 $seen = [];
109 foreach ($values as $value) {
110 $number = (int) $value;
111 if (!inRange($number, RANGES[$key]) || isset($seen[$number])) return false;
112 $seen[$number] = true;
113 }
114 return true;
115}

196function checkSettings(array $lines): Config
197{
198 $settings = [];
199 foreach ($lines as [$key, $values]) {
200 if (!in_array($key, KEYS, true)) throw new SettingsError('unknown key ' . $key);
201 if (array_key_exists($key, $settings)) throw new SettingsError('duplicate key ' . $key);
202 $validCount = $key === 'RANDOM' ? count($values) === 3 :
203 (in_array($key, LIST_KEYS, true) ? count($values) >= 1 : count($values) === 1);
204 if (!$validCount) throw new SettingsError('wrong number of values for ' . $key);
205 $settings[$key] = $values;
206 }
207 foreach (REQUIRED as $key) {
208 if (!isset($settings[$key])) throw new SettingsError('missing key ' . $key);
209 }
210 foreach (KEYS as $key) {
211 if (isset($settings[$key]) && !validValue($key, $settings[$key])) {
212 throw new SettingsError('bad value for ' . $key);
213 }
214 }
215 $config = new Config($settings);
216 if ($config->protocol === 'SR' && $config->dupack !== null) {
217 throw new SettingsError('DUPACK not supported for SR');
218 }
219 $maxWindow = $config->windowLimit();
220 if (($config->protocol === 'SW' || !$config->unsafe) && $config->window > $maxWindow) {
221 throw new SettingsError('window ' . $config->window . ' exceeds limit ' .
222 $maxWindow . ' for ' . $config->protocol);
223 }
224 $both = array_intersect(array_keys($config->loseData), array_keys($config->corruptData));
225 if ($both) throw new SettingsError('data #' . min($both) . ' both lost and corrupted');
226 $both = array_intersect(array_keys($config->loseCtrl), array_keys($config->corruptCtrl));
227 if ($both) throw new SettingsError('control #' . min($both) . ' both lost and corrupted');
228 return $config;
229}

251final class LossGenerator
252{
253 public function __construct(
254 private int $state,
255 private int $loss,
256 private int $corruption
257 ) {}
258
259 public function nextOutcome(): string
260 {
261 $this->state = RANDOM_MULTIPLIER * $this->state % RANDOM_MODULUS;
262 $roll = $this->state % 100;
263 if ($roll < $this->loss) return 'LOST';
264 if ($roll < $this->loss + $this->corruption) return 'CORRUPT';
265 return 'NORMAL';
266 }

269final class Channel
270{
271 public int $sent = 0;
272 public int $lost = 0;
273 public int $corrupted = 0;
274 private array $inTransit = [];
275
276 public function __construct(
277 private int $delay,
278 private array $losses,
279 private array $corruptions,
280 private array $extraDelays
281 ) {}
282
283 public function send(string $kind, int $value, ?int $payload, int $tick,
284 ?LossGenerator $generator): Transmission
285 {
286 $number = ++$this->sent;
287 $outcome = $generator?->nextOutcome() ?? 'NORMAL';
288 if (isset($this->losses[$number])) $outcome = 'LOST';
289 elseif (isset($this->corruptions[$number])) $outcome = 'CORRUPT';
290 $item = new Transmission($number, $kind, $value, $payload, $outcome,
291 $tick + $this->delay + ($this->extraDelays[$number] ?? 0));
292 $this->inTransit[] = $item;
293 return $item;
294 }
295
296 public function arrivals(int $tick): array
297 {
298 $arrived = [];
299 $remaining = [];
300 foreach ($this->inTransit as $item) {
301 if ($item->arrival === $tick) $arrived[] = $item;
302 else $remaining[] = $item;
303 }
304 $this->inTransit = $remaining;
305 return $arrived;
306 }
307
308 public function countDamage(Transmission $item): void
309 {
310 if ($item->outcome === 'LOST') ++$this->lost;
311 else ++$this->corrupted;
312 }
313}

315final class Simulation
316{
317 public array $lines = [];
318 public ?int $completion = null;
319 public int $delivered = 0;
320 public Channel $dataChannel;
321 public Channel $controlChannel;
322 private Config $config;
323 private bool $selective;
324 private int $modulus;
325 private array $visible;
326 private ?LossGenerator $generator;
327 private array $records;
328 private int $tick = 0;
329 private int $base = 0;
330 private int $framesSent = 0;
331 private int $nextFrame = 0;
332 private int $timerLength;
333 private ?int $timer = null;
334 private int $duplicates = 0;
335 private array $acked;
336 private array $timers;
337 private array $queue = [];
338 private int $cwnd = 1;
339 private int $ssthresh;
340 private int $roundAcks = 0;
341 private int $expected = 0;
342 private bool $nakSent = false;
343 private array $buffer = [];
344 private int $resent = 0;
345 private int $naks = 0;
346 private int $timeouts = 0;
347 private int $fastRetransmits = 0;
348 private int $errors = 0;
349 private int $discarded = 0;
350 private int $maxBuffer = 0;
351
352 public function __construct(Config $config)
353 {
354 $this->config = $config;
355 $this->selective = $config->protocol === 'SR';
356 $this->modulus = 2 ** $config->seqbits;
357 $this->visible = TRACE_SIDES[$config->trace];
358 $this->generator = $config->random === null ? null : new LossGenerator(...$config->random);
359 $this->dataChannel = new Channel($config->delay, $config->loseData,
360 $config->corruptData, $config->delayData);
361 $this->controlChannel = new Channel($config->delay, $config->loseCtrl,
362 $config->corruptCtrl, $config->delayCtrl);
363 $this->records = array_fill(0, $config->frames, null);
364 for ($i = 0; $i < $config->frames; ++$i) $this->records[$i] = new FrameRecord();
365 $this->timerLength = $config->timeout;
366 $this->acked = array_fill(0, $config->frames, false);
367 $this->timers = array_fill(0, $config->frames, null);
368 $this->ssthresh = $config->window;
369 }
370
371 private function emit(mixed ...$parts): void
372 {
373 $this->lines[] = $this->tick . ' ' . implode(' ', $parts);
374 }
375
376 private function trace(string $side, mixed ...$parts): void
377 {
378 if (in_array($side, $this->visible, true)) $this->emit(...$parts);
379 }
380
381 private function frameOf(int $value): int
382 {
383 return $this->base + modulo($value - $this->base, $this->modulus);
384 }
385
386 private function effectiveWindow(): int
387 {
388 return $this->config->congestion ? min($this->config->window, $this->cwnd) : $this->config->window;
389 }
390
391 public function run(): void
392 {
393 for ($tick = 0; $tick <= $this->config->limit; ++$tick) {
394 $this->tick = $tick;
395 foreach ($this->dataChannel->arrivals($tick) as $item) {
396 if ($item->outcome === 'NORMAL') $this->receiveData($item);
397 else $this->reportDamage($this->dataChannel, $item, 'RECEIVER');
398 }
399 foreach ($this->controlChannel->arrivals($tick) as $item) {
400 if ($item->outcome === 'NORMAL') $this->receiveControl($item);
401 else $this->reportDamage($this->controlChannel, $item, 'SENDER');
402 }
403 $completed = $this->base === $this->config->frames;
404 if (!$completed) {
405 $this->processTimeouts();
406 $this->sendData();
407 }
408 if (isset($this->config->snapshots[$tick])) $this->snapshot();
409 if ($completed) {
410 $this->completion = $tick;
411 $this->emit('DONE');
412 return;
413 }
414 }
415 $this->emit('ABORT');
416 }
417
418 private function reportDamage(Channel $channel, Transmission $item, string $side): void
419 {
420 $channel->countDamage($item);
421 $this->trace($side, $item->outcome, $item->kind, '#' . $item->number);
422 }
423
424 private function receiveData(Transmission $item): void
425 {
426 $this->trace('RECEIVER', 'RECV', 'seq=' . $item->value, '#' . $item->number);
427 if ($this->selective) $this->selectiveReceive($item->value, $item->payload);
428 else $this->goBackNReceive($item->value, $item->payload);
429 }
430
431 private function goBackNReceive(int $seq, int $payload): void
432 {
433 if ($seq === $this->expected % $this->modulus) {
434 $this->deliver($payload);
435 $this->sendControl('ack', $seq);
436 return;
437 }
438 $this->discard($seq);
439 if ($this->config->nak && !$this->nakSent) {
440 $this->sendControl('nak', $this->expected % $this->modulus);
441 $this->nakSent = true;
442 } elseif ($this->expected > 0) {
443 $this->sendControl('ack', ($this->expected - 1) % $this->modulus);
444 }
445 }
446
447 private function selectiveReceive(int $seq, int $payload): void
448 {
449 $offset = modulo($seq - $this->expected, $this->modulus);
450 if ($offset >= $this->config->window) {
451 $this->discard($seq);
452 if (modulo($this->expected - $seq, $this->modulus) <= $this->config->window) {
453 $this->sendControl('ack', $seq);
454 }
455 return;
456 }
457 $position = $this->expected + $offset;
458 if (isset($this->buffer[$position])) {
459 $this->discard($seq);
460 $this->sendControl('ack', $seq);
461 } elseif ($offset === 0) {
462 $this->deliver($payload);
463 while (isset($this->buffer[$this->expected])) {
464 $waiting = $this->buffer[$this->expected];
465 unset($this->buffer[$this->expected]);
466 $this->deliver($waiting);
467 }
468 $this->sendControl('ack', $seq);
469 } else {
470 $this->buffer[$position] = $payload;
471 $this->maxBuffer = max($this->maxBuffer, count($this->buffer));
472 $this->trace('RECEIVER', 'BUFFER', 'f=' . $position);
473 $this->sendControl('ack', $seq);
474 if ($this->config->nak && !$this->nakSent) {
475 $this->sendControl('nak', $this->expected % $this->modulus);
476 $this->nakSent = true;
477 }
478 }
479 }
480
481 private function deliver(int $payload): void
482 {
483 $position = $this->expected;
484 if ($payload === $position) {
485 ++$this->delivered;
486 $this->records[$payload]->delivered = $this->tick;
487 $this->trace('RECEIVER', 'DELIVER', 'f=' . $position);
488 } else {
489 ++$this->errors;
490 $this->trace('RECEIVER', 'DELIVER', 'f=' . $position, 'wrong=' . $payload);
491 }
492 ++$this->expected;
493 $this->nakSent = false;
494 }
495
496 private function discard(int $seq): void
497 {
498 ++$this->discarded;
499 $this->trace('RECEIVER', 'DISCARD', 'seq=' . $seq);
500 }
501
502 private function sendControl(string $kind, int $value): void
503 {
504 $item = $this->controlChannel->send($kind, $value, null, $this->tick, $this->generator);
505 if ($kind === 'nak') ++$this->naks;
506 $this->trace('RECEIVER', strtoupper($kind), 'a=' . $value, '#' . $item->number);
507 }
508
509 private function receiveControl(Transmission $item): void
510 {
511 $before = [$this->cwnd, $this->ssthresh];
512 if ($item->kind === 'ack') {
513 if ($this->selective) $this->selectiveAck($item);
514 else $this->goBackNAck($item);
515 } elseif ($this->selective) {
516 $this->selectiveNak($item);
517 } else {
518 $this->goBackNNak($item);
519 }
520 $this->reportWindow($before);
521 }
522
523 private function controlLine(Transmission $item, bool $accepted): void
524 {
525 $parts = [($accepted ? 'GOT' : 'OLD') . strtoupper($item->kind),
526 'a=' . $item->value, '#' . $item->number];
527 if ($accepted) $parts[] = 'base=' . $this->base;
528 $this->trace('SENDER', ...$parts);
529 }
530
531 private function goBackNAck(Transmission $item): void
532 {
533 $frame = $this->frameOf($item->value);
534 if ($frame < $this->framesSent) {
535 $this->growWindow($frame - $this->base + 1);
536 $this->moveBase($frame + 1);
537 if ($this->nextFrame < $this->base) $this->nextFrame = $this->base;
538 $this->timer = $this->base === $this->nextFrame ? null : $this->tick + $this->timerLength;
539 $this->controlLine($item, true);
540 return;
541 }
542 $this->controlLine($item, false);
543 if ($this->config->dupack === null || $this->base < 1) return;
544 if ($item->value === ($this->base - 1) % $this->modulus) {
545 ++$this->duplicates;
546 if ($this->duplicates === $this->config->dupack) {
547 ++$this->fastRetransmits;
548 $this->trace('SENDER', 'FASTRETX', 'f=' . $this->base);
549 $this->goBack();
550 $this->halveWindow();
551 }
552 }
553 }
554
555 private function goBackNNak(Transmission $item): void
556 {
557 $frame = $this->frameOf($item->value);
558 if ($frame > $this->framesSent) {
559 $this->controlLine($item, false);
560 return;
561 }
562 $this->growWindow($frame - $this->base);
563 $this->moveBase($frame);
564 $this->goBack();
565 $this->halveWindow();
566 $this->controlLine($item, true);
567 }
568
569 private function selectiveAck(Transmission $item): void
570 {
571 $frame = $this->frameOf($item->value);
572 if ($frame >= $this->framesSent || $this->acked[$frame]) {
573 $this->controlLine($item, false);
574 return;
575 }
576 $this->acked[$frame] = true;
577 $this->records[$frame]->acked = $this->tick;
578 $this->timers[$frame] = null;
579 $this->queue = array_values(array_filter($this->queue,
580 static fn(int $waiting): bool => $waiting !== $frame));
581 $this->growWindow(1);
582 $previous = $this->base;
583 while ($this->base < $this->framesSent && $this->acked[$this->base]) ++$this->base;
584 if ($this->base > $previous) $this->timerLength = $this->config->timeout;
585 $this->controlLine($item, true);
586 }
587
588 private function selectiveNak(Transmission $item): void
589 {
590 $frame = $this->frameOf($item->value);
591 if ($frame >= $this->framesSent || $this->acked[$frame]) {
592 $this->controlLine($item, false);
593 return;
594 }
595 $this->timers[$frame] = null;
596 if (!in_array($frame, $this->queue, true)) $this->queue[] = $frame;
597 $this->halveWindow();
598 $this->controlLine($item, true);
599 }
600
601 private function moveBase(int $newBase): void
602 {
603 if ($newBase <= $this->base) return;
604 for ($f = $this->base; $f < $newBase; ++$f) $this->records[$f]->acked = $this->tick;
605 $this->base = $newBase;
606 $this->timerLength = $this->config->timeout;
607 $this->duplicates = 0;
608 }
609
610 private function goBack(): void
611 {
612 $this->nextFrame = $this->base;
613 $this->timer = null;
614 $this->duplicates = 0;
615 }
616
617 private function growWindow(int $count): void
618 {
619 if (!$this->config->congestion) return;
620 for ($i = 0; $i < $count; ++$i) {
621 if ($this->cwnd >= $this->config->window) continue;
622 if ($this->cwnd < $this->ssthresh) {
623 ++$this->cwnd;
624 } else {
625 ++$this->roundAcks;
626 if ($this->roundAcks === $this->cwnd) {
627 ++$this->cwnd;
628 $this->roundAcks = 0;
629 }
630 }
631 }
632 }
633
634 private function halveWindow(): void
635 {
636 if (!$this->config->congestion) return;
637 $this->ssthresh = max(intdiv($this->cwnd, 2), 1);
638 $this->cwnd = $this->ssthresh;
639 $this->roundAcks = 0;
640 }
641
642 private function collapseWindow(): void
643 {
644 if (!$this->config->congestion) return;
645 $this->ssthresh = max(intdiv($this->cwnd, 2), 1);
646 $this->cwnd = 1;
647 $this->roundAcks = 0;
648 }
649
650 private function reportWindow(array $before): void
651 {
652 if ([$this->cwnd, $this->ssthresh] !== $before) {
653 $this->trace('SENDER', 'CWND', $this->cwnd, 'ssthresh=' . $this->ssthresh);
654 }
655 }
656
657 private function processTimeouts(): void
658 {
659 if ($this->selective) {
660 for ($frame = $this->base; $frame < $this->framesSent; ++$frame) {
661 if ($this->timers[$frame] === $this->tick) {
662 $this->timers[$frame] = null;
663 $this->queue[] = $frame;
664 $this->timeOut($frame);
665 }
666 }
667 } elseif ($this->timer === $this->tick) {
668 $this->timeOut($this->base);
669 $this->goBack();
670 }
671 }
672
673 private function timeOut(int $frame): void
674 {
675 $before = [$this->cwnd, $this->ssthresh];
676 ++$this->timeouts;
677 $this->trace('SENDER', 'TIMEOUT', 'f=' . $frame);
678 if ($this->config->backoff) {
679 $this->timerLength = min(2 * $this->timerLength, 8 * $this->config->timeout);
680 }
681 $this->collapseWindow();
682 $this->reportWindow($before);
683 }
684
685 private function maySendNew(int $frame): bool
686 {
687 return $frame < $this->config->frames &&
688 $frame < $this->base + $this->effectiveWindow() &&
689 $frame * $this->config->interval <= $this->tick;
690 }
691
692 private function sendData(): void
693 {
694 if ($this->selective) $this->selectiveSend();
695 else $this->goBackNSend();
696 }
697
698 private function goBackNSend(): void
699 {
700 $frame = $this->nextFrame;
701 if ($frame >= $this->config->frames || $frame >= $this->base + $this->effectiveWindow()) return;
702 if ($frame >= $this->framesSent && $frame * $this->config->interval > $this->tick) return;
703 $this->transmit($frame);
704 $this->framesSent = max($this->framesSent, $frame + 1);
705 ++$this->nextFrame;
706 if ($this->timer === null) $this->timer = $this->tick + $this->timerLength;
707 }
708
709 private function selectiveSend(): void
710 {
711 if ($this->queue) {
712 $frame = array_shift($this->queue);
713 } elseif ($this->maySendNew($this->framesSent)) {
714 $frame = $this->framesSent++;
715 } else {
716 return;
717 }
718 $this->transmit($frame);
719 $this->timers[$frame] = $this->tick + $this->timerLength;
720 }
721
722 private function transmit(int $frame): void
723 {
724 $record = $this->records[$frame];
725 $seq = $frame % $this->modulus;
726 $item = $this->dataChannel->send('data', $seq, $frame, $this->tick, $this->generator);
727 ++$record->transmissions;
728 if ($record->transmissions === 1) {
729 $record->first = $this->tick;
730 $label = 'SEND';
731 } else {
732 ++$this->resent;
733 $label = 'RESEND';
734 }
735 $this->trace('SENDER', $label, 'f=' . $frame, 'seq=' . $seq, '#' . $item->number);
736 }
737
738 private function senderMark(int $frame): string
739 {
740 if ($this->selective) {
741 $acknowledged = $this->acked[$frame];
742 $waiting = in_array($frame, $this->queue, true);
743 $sent = $frame < $this->framesSent;
744 } else {
745 $acknowledged = $frame < $this->base;
746 $waiting = $this->nextFrame <= $frame && $frame < $this->framesSent;
747 $sent = $frame < $this->nextFrame;
748 }
749 if ($acknowledged) return 'A';
750 if ($waiting) return 'R';
751 if ($sent) return 'O';
752 if ($frame < $this->base + $this->effectiveWindow()) return 'W';
753 return '.';
754 }
755
756 private function receiverMark(int $position): string
757 {
758 if ($position < $this->expected) return 'D';
759 if ($this->selective) {
760 if (isset($this->buffer[$position])) return 'B';
761 if ($position < $this->expected + $this->config->window) return 'W';
762 } elseif ($position === $this->expected) {
763 return 'W';
764 }
765 return '.';
766 }
767
768 private function snapshot(): void
769 {
770 $sender = $receiver = '';
771 for ($frame = 0; $frame < $this->config->frames; ++$frame) {
772 $sender .= $this->senderMark($frame);
773 $receiver .= $this->receiverMark($frame);
774 }
775 $this->emit('SNAPSHOT', 'sender=' . $sender, 'receiver=' . $receiver);
776 }

778 public function dataCount(): int
779 {
780 return $this->dataChannel->sent;
781 }
782
783 public function goodputText(): string
784 {
785 return truncated($this->delivered, $this->completion, 2);
786 }
787
788 public function efficiencyText(): string
789 {
790 return truncated(100 * $this->delivered, $this->dataCount(), 1);

792
793 private function latencyText(): string
794 {
795 if ($this->delivered === 0) return '-';
796 $total = 0;
797 foreach ($this->records as $record) {
798 if ($record->delivered !== null) $total += $record->delivered - $record->first;
799 }
800 return truncated($total, $this->delivered, 2);
801 }

803 public function report(): array
804 {
805 $lines = [];
806 foreach ($this->records as $frame => $record) {
807 $lines[] = 'frame ' . $frame . ': seq=' . ($frame % $this->modulus) .
808 ' tx=' . $record->transmissions . ' first=' . tickText($record->first) .
809 ' delivered=' . tickText($record->delivered) . ' acked=' . tickText($record->acked);
810 }
811 $data = $this->dataChannel;
812 $control = $this->controlChannel;
813 $lines[] = 'totals: data=' . $data->sent . ' resent=' . $this->resent .
814 ' control=' . $control->sent . ' nak=' . $this->naks .
815 ' timeouts=' . $this->timeouts . ' fastretx=' . $this->fastRetransmits;
816 $lines[] = 'channel: lost_data=' . $data->lost . ' lost_ctrl=' . $control->lost .
817 ' corrupt_data=' . $data->corrupted . ' corrupt_ctrl=' . $control->corrupted;
818 $lines[] = 'receiver: delivered=' . $this->delivered . ' errors=' . $this->errors .
819 ' discarded=' . $this->discarded . ' max_buffer=' . $this->maxBuffer;
820 $completion = $this->completion === null ? '-' : (string) $this->completion;
821 $goodput = $this->completion === null ? '-' : $this->goodputText();
822 $utilization = truncated(100 * $data->sent, $this->tick + 1, 1);
823 $lines[] = 'result: completion=' . $completion . ' goodput=' . $goodput .
824 ' efficiency=' . $this->efficiencyText() . '% utilization=' . $utilization .
825 '% latency=' . $this->latencyText();
826 return $lines;
827 }

830final class Scenario
831{
832 public ?Config $config = null;
833 public ?Simulation $simulation = null;
834
835 public function __construct(public string $name, private array $settings) {}
836
837 public function completed(): bool
838 {
839 return $this->simulation !== null && $this->simulation->completion !== null;
840 }
841
842 public function status(): string
843 {
844 return $this->simulation === null ? 'invalid' : 'aborted';
845 }
846
847 public function execute(array &$output): void
848 {
849 $output[] = '== ' . $this->name . ' ==';
850 try {
851 $this->config = checkSettings($this->settings);
852 } catch (SettingsError $error) {
853 $output[] = 'error: ' . $error->getMessage();
854 return;
855 }
856 $output[] = $this->config->configLine();
857 $output[] = $this->config->optionsLine();
858 $this->simulation = new Simulation($this->config);
859 $this->simulation->run();
860 array_push($output, ...$this->simulation->lines, ...$this->simulation->report());
861 }
862}
863
864function rankingLines(array $scenarios): array
865{
866 $output = ['== ranking =='];
867 $completed = [];
868 foreach ($scenarios as $index => $scenario) {
869 if ($scenario->completed()) $completed[] = [$scenario, $index];
870 }
871 usort($completed, static function (array $a, array $b): int {
872 [$first, $ia] = $a;
873 [$second, $ib] = $b;
874 $left = $first->simulation;
875 $right = $second->simulation;
876 $goodput = $right->delivered * $left->completion - $left->delivered * $right->completion;
877 if ($goodput !== 0) return $goodput <=> 0;
878 $efficiency = $right->delivered * $left->dataCount() -
879 $left->delivered * $right->dataCount();
880 return $efficiency !== 0 ? $efficiency <=> 0 : $ia <=> $ib;
881 });
882 foreach ($completed as $rank => [$scenario]) {
883 $run = $scenario->simulation;
884 $output[] = ($rank + 1) . '. ' . $scenario->name . ' ' . $scenario->config->protocol .
885 ' goodput=' . $run->goodputText() . ' efficiency=' . $run->efficiencyText() . '%';
886 }
887 foreach ($scenarios as $scenario) {
888 if (!$scenario->completed()) $output[] = '- ' . $scenario->name . ' ' . $scenario->status();
889 }
890 return $output;
891}

893function protocolLines(array $scenarios): array
894{
895 $output = ['== protocols =='];
896 foreach (PROTOCOLS as $protocol) {
897 $runs = [];
898 foreach ($scenarios as $scenario) {
899 if ($scenario->completed() && $scenario->config->protocol === $protocol) {
900 $runs[] = $scenario->simulation;
901 }
902 }
903 if (!$runs) continue;
904 $delivered = $ticks = $data = 0;
905 foreach ($runs as $run) {
906 $delivered += $run->delivered;
907 $ticks += $run->completion;
908 $data += $run->dataCount();
909 }
910 $output[] = $protocol . ': runs=' . count($runs) .
911 ' goodput=' . truncated($delivered, $ticks, 2) .
912 ' efficiency=' . truncated(100 * $delivered, $data, 1) . '%';
913 }
914 return $output;
915}

917function readScenarios(array $lines): array
918{
919 $count = (int) $lines[0];
920 $index = 1;
921 $scenarios = [];
922 for ($number = 0; $number < $count; ++$number) {
