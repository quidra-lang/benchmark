5class Program {
6 static readonly string[] Keys = { "PROTOCOL", "FRAMES", "WINDOW", "SEQBITS",
7 "TIMEOUT", "DELAY", "INTERVAL", "MAXTICKS",
8 "DUPACK", "NAK", "BACKOFF", "CONGESTION",
9 "UNSAFE", "TRACE", "RANDOM", "LOSEDATA",
10 "LOSECTRL", "CORRUPTDATA", "CORRUPTCTRL", "DELAYDATA",
11 "DELAYCTRL", "SNAPSHOT" };
12 static readonly List<string> Output = new();
13 static int Mod(int a, int b) => (a % b + b) % b;
14 static string Fixed(long numerator, long denominator, int digits) {
15 long scale = digits == 2 ? 100 : 10;
16 long value = denominator == 0 ? 0 : numerator * scale / denominator;
17 return (value / scale) + "." + (value % scale).ToString(digits == 2 ? "D2" : "D1");
18 }
19 static string Tick(int tick) => tick < 0 ? "-" : tick.ToString();
20 class Config {
21 public string Name = "", Protocol = "", Trace = "ALL";
22 public int N, W, K, T, Delay, Interval, Limit = 1000, Dupack;
23 public bool Nak, Backoff, Congestion, Unsafe, Random;
24 public long Seed;
25 public int Loss, Corruption;
26 public HashSet<int> LostData = new(), LostControl = new(), CorruptData = new(),
27 CorruptControl = new(), Snapshots = new();
28 public Dictionary<int, int> DelayData = new(), DelayControl = new();
29 }
30 static string Validate(Config c, List<string[]> lines) {
31 Dictionary<string, string[]> values = new();
32 foreach (string[] line in lines) {
33 string key = line[0];
34 int index = Array.IndexOf(Keys, key), count = line.Length - 1;
35 if (index < 0)
36 return "unknown key " + key;
37 if (values.ContainsKey(key))
38 return "duplicate key " + key;
39 if (key == "RANDOM" ? count != 3 : index >= 15 ? count < 1 : count != 1)
40 return "wrong number of values for " + key;
41 values[key] = line.Skip(1).ToArray();
42 }
43 for (int i = 0; i < 6; ++i)
44 if (!values.ContainsKey(Keys[i]))
45 return "missing key " + Keys[i];
46 foreach (string key in Keys) {
47 if (!values.TryGetValue(key, out string[] v))
48 continue;
49 int index = Array.IndexOf(Keys, key);
50 bool valid = true;
51 if (key == "PROTOCOL")
52 valid = new[] { "SW", "GBN", "SR" }.Contains(v[0]);
53 else if (index >= 9 && index <= 12)
54 valid = v[0] == "ON" || v[0] == "OFF";
55 else if (key == "TRACE")
56 valid = new[] { "ALL", "SENDER", "RECEIVER", "NONE" }.Contains(v[0]);
57 else if (key == "RANDOM") {
58 long seed = long.Parse(v[0]);
59 int loss = int.Parse(v[1]), corruption = int.Parse(v[2]);
60 valid = seed >= 1 && seed <= 2147483646 && loss >= 0 && corruption >= 0 &&
61 (long)loss + corruption <= 100;
62 } else if (index >= 15) {
63 HashSet<int> seen = new();
64 foreach (string value in v) {
65 string[] parts = value.Split(':');
66 int number = int.Parse(parts[0]);
67 int lower = key == "SNAPSHOT" ? 0 : 1, upper = key == "SNAPSHOT" ? 2000 : 9999;
68 if (number < lower || number > upper || !seen.Add(number))
69 valid = false;
70 if (index == 19 || index == 20) {
71 int extra = int.Parse(parts[1]);
72 if (extra < 1 || extra > 50)
73 valid = false;
74 }
75 }
76 } else {
77 int number = int.Parse(v[0]);
78 int lower = key == "INTERVAL" ? 0 : 1;
79 int upper = key switch {
80 "FRAMES" => 60, "WINDOW" => 32, "SEQBITS" => 6, "TIMEOUT" => 50,
81 "DELAY" => 10, "INTERVAL" => 10, "MAXTICKS" => 2000,
82 _ => 10
83 };
84 valid = number >= lower && number <= upper;
85 }
86 if (!valid)
87 return "bad value for " + key;
88 }
89 string Get(string key, string fallback) =>
90 values.TryGetValue(key, out string[] v) ? v[0] : fallback;
91 c.Protocol = Get("PROTOCOL", "");
92 c.N = int.Parse(Get("FRAMES", "0"));
93 c.W = int.Parse(Get("WINDOW", "0"));
94 c.K = int.Parse(Get("SEQBITS", "0"));
95 c.T = int.Parse(Get("TIMEOUT", "0"));
96 c.Delay = int.Parse(Get("DELAY", "0"));
97 c.Interval = int.Parse(Get("INTERVAL", "0"));
98 c.Limit = int.Parse(Get("MAXTICKS", "1000"));
99 c.Dupack = int.Parse(Get("DUPACK", "0"));
100 c.Nak = Get("NAK", "OFF") == "ON";
101 c.Backoff = Get("BACKOFF", "OFF") == "ON";
102 c.Congestion = Get("CONGESTION", "OFF") == "ON";
103 c.Unsafe = Get("UNSAFE", "OFF") == "ON";
104 c.Trace = Get("TRACE", "ALL");
105 if (c.Protocol == "SR" && values.ContainsKey("DUPACK"))
106 return "DUPACK not supported for SR";
107 int limit = c.Protocol == "SW" ? 1 : c.Protocol == "GBN" ? (1 << c.K) - 1 : 1 << (c.K - 1);
108 if (c.W > limit && !(c.Unsafe && c.Protocol != "SW"))
109 return $"window {c.W} exceeds limit {limit} for {c.Protocol}";
110 if (values.TryGetValue("RANDOM", out string[] random)) {
111 c.Random = true;
112 c.Seed = long.Parse(random[0]);
113 c.Loss = int.Parse(random[1]);
114 c.Corruption = int.Parse(random[2]);
115 }
116 HashSet<int> Set(string key) =>
117 values.TryGetValue(key, out string[] v) ? v.Select(int.Parse).ToHashSet() : new();
118 Dictionary<int, int> Delays(string key) =>
119 values.TryGetValue(key, out string[] v)
120 ? v.Select(x => x.Split(':'))
121 .ToDictionary(x => int.Parse(x[0]), x => int.Parse(x[1]))
122 : new();
123 c.LostData = Set("LOSEDATA");
124 c.LostControl = Set("LOSECTRL");
125 c.CorruptData = Set("CORRUPTDATA");
126 c.CorruptControl = Set("CORRUPTCTRL");
127 c.Snapshots = Set("SNAPSHOT");
128 c.DelayData = Delays("DELAYDATA");
129 c.DelayControl = Delays("DELAYCTRL");
130 int conflict = c.LostData.Intersect(c.CorruptData).DefaultIfEmpty(-1).Min();
131 if (conflict >= 0)
132 return $"data #{conflict} both lost and corrupted";
133 conflict = c.LostControl.Intersect(c.CorruptControl).DefaultIfEmpty(-1).Min();
134 return conflict >= 0 ? $"control #{conflict} both lost and corrupted" : null;
135 }
136 record Transmission(int Arrival, int Number, int Value, int Outcome, bool Nak);
137 class Result {
138 public Config Config;
139 public string Status;
140 public int Delivered, Data, Completion;
141 public Result(Config c, string status) {
142 Config = c;
143 Status = status;
144 }
145 }
146 class Simulation {
147 readonly Config c;
148 readonly bool sr;
149 readonly int modulus;
150 readonly List<Transmission> dataEvents = new(), controlEvents = new();
151 readonly List<int> queue = new();
152 readonly Dictionary<int, int> buffer = new();
153 readonly bool[] acknowledged;
154 readonly int[] timers, tx, first, deliveredAt, ackedAt;
155 int tick, baseFrame, high, next, duration, timer = -1, duplicates, receiver, cwnd = 1,
156 threshold, credit;
157 int data, control, resent, naks, timeouts, fast, lostData, lostControl, corruptData,
158 corruptControl;
159 int delivered, errors, discarded, maxBuffer, latency;
160 bool nakSent;
161 long random;
162 public Simulation(Config config) {
163 c = config;
164 sr = c.Protocol == "SR";
165 modulus = 1 << c.K;
166 duration = c.T;
167 threshold = c.W;
168 random = c.Seed;
169 acknowledged = new bool[c.N];
170 timers = Enumerable.Repeat(-1, c.N).ToArray();
171 tx = new int[c.N];
172 first = Enumerable.Repeat(-1, c.N).ToArray();
173 deliveredAt = Enumerable.Repeat(-1, c.N).ToArray();
174 ackedAt = Enumerable.Repeat(-1, c.N).ToArray();
175 }
176 int Effective => c.Congestion ? Math.Min(c.W, cwnd) : c.W;
177 void Log(string side, string text) {
178 if (c.Trace == "ALL" || c.Trace == side || side == "ALWAYS")
179 Output.Add(tick + " " + text);
180 }
181 int Outcome(bool isData, int number) {
182 int outcome = 0;
183 if (c.Random) {
184 random = random * 48271 % 2147483647;
185 int value = (int)(random % 100);
186 outcome = value < c.Loss ? 1 : value < c.Loss + c.Corruption ? 2 : 0;
187 }
188 if ((isData ? c.LostData : c.LostControl).Contains(number))
189 outcome = 1;
190 if ((isData ? c.CorruptData : c.CorruptControl).Contains(number))
191 outcome = 2;
192 return outcome;
193 }
194 void SendControl(bool nak, int value) {
195 ++control;
196 if (nak)
197 ++naks;
198 Log("RECEIVER", $"{(nak ? "NAK" : "ACK")} a={value} #{control}");
199 controlEvents.Add(
200 new Transmission(tick + c.Delay + c.DelayControl.GetValueOrDefault(control),
201 control, value, Outcome(false, control), nak));
202 }
203 void Deliver(int payload) {
204 if (payload == receiver) {
205 Log("RECEIVER", $"DELIVER f={receiver}");
206 ++delivered;
207 if (deliveredAt[payload] < 0)
208 deliveredAt[payload] = tick;
209 latency += tick - first[payload];
210 } else {
211 Log("RECEIVER", $"DELIVER f={receiver} wrong={payload}");
212 ++errors;
213 }
214 ++receiver;
215 nakSent = false;
216 }
217 void Discard(int sequence) {
218 ++discarded;
219 Log("RECEIVER", $"DISCARD seq={sequence}");
220 }
221 void Receive(Transmission e) {
222 if (e.Outcome != 0) {
223 if (e.Outcome == 1)
224 ++lostData;
225 else
226 ++corruptData;
227 Log("RECEIVER", $"{(e.Outcome == 1 ? "LOST" : "CORRUPT")} data #{e.Number}");
228 return;
229 }
230 int sequence = e.Value % modulus;
231 Log("RECEIVER", $"RECV seq={sequence} #{e.Number}");
232 if (!sr) {
233 if (sequence == receiver % modulus) {
234 Deliver(e.Value);
235 SendControl(false, sequence);
236 } else {
237 Discard(sequence);
238 if (c.Nak && !nakSent) {
239 SendControl(true, receiver % modulus);
240 nakSent = true;
241 } else if (receiver > 0)
242 SendControl(false, (receiver - 1) % modulus);
243 }
244 return;
245 }
246 int offset = Mod(sequence - receiver, modulus), position = receiver + offset;
247 if (offset < c.W) {
248 if (buffer.ContainsKey(position)) {
249 Discard(sequence);
250 SendControl(false, sequence);
251 } else if (offset == 0) {
252 Deliver(e.Value);
253 while (buffer.Remove(receiver, out int payload))
254 Deliver(payload);
255 SendControl(false, sequence);
256 } else {
257 buffer[position] = e.Value;
258 maxBuffer = Math.Max(maxBuffer, buffer.Count);
259 Log("RECEIVER", $"BUFFER f={position}");
260 SendControl(false, sequence);
261 if (c.Nak && !nakSent) {
262 SendControl(true, receiver % modulus);
263 nakSent = true;
264 }
265 }
266 } else {
267 Discard(sequence);
268 if (Mod(receiver - sequence, modulus) <= c.W)
269 SendControl(false, sequence);
270 }
271 }
272 void GoBack() {
273 next = baseFrame;
274 timer = -1;
275 duplicates = 0;
276 }
277 void Increase(int count) {
278 if (!c.Congestion)
279 return;
280 for (int i = 0; i < count && cwnd < c.W; ++i) {
281 if (cwnd < threshold)
282 ++cwnd;
283 else if (++credit == cwnd) {
284 ++cwnd;
285 credit = 0;
286 }
287 }
288 }
289 void Decrease(bool timeout) {
290 if (!c.Congestion)
291 return;
292 threshold = Math.Max(cwnd / 2, 1);
293 cwnd = timeout ? 1 : threshold;
294 credit = 0;
295 }
296 void Cwnd(int oldWindow, int oldThreshold) {
297 if (c.Congestion && (oldWindow != cwnd || oldThreshold != threshold))
298 Log("SENDER", $"CWND {cwnd} ssthresh={threshold}");
299 }
300 void Control(Transmission e) {
301 string kind = e.Nak ? "NAK" : "ACK";
302 if (e.Outcome != 0) {
303 if (e.Outcome == 1)
304 ++lostControl;
305 else
306 ++corruptControl;
307 Log("SENDER",
308 $"{(e.Outcome == 1 ? "LOST" : "CORRUPT")} {kind.ToLowerInvariant()} #{e.Number}");
309 return;
310 }
311 int oldWindow = cwnd, oldThreshold = threshold, oldBase = baseFrame;
312 int frame = baseFrame + Mod(e.Value - baseFrame, modulus);
313 bool accepted = sr ? frame < high && !acknowledged[frame]
314 : e.Nak ? frame <= high
315 : frame < high;
316 if (accepted) {
317 if (sr) {
318 timers[frame] = -1;
319 if (e.Nak) {
320 if (!queue.Contains(frame))
321 queue.Add(frame);
322 } else {
323 acknowledged[frame] = true;
324 queue.Remove(frame);
325 if (ackedAt[frame] < 0)
326 ackedAt[frame] = tick;
327 while (baseFrame < high && acknowledged[baseFrame])
328 ++baseFrame;
329 Increase(1);
330 }
331 } else {
332 baseFrame = frame + (e.Nak ? 0 : 1);
333 Increase(baseFrame - oldBase);
334 for (int f = oldBase; f < baseFrame; ++f)
335 if (ackedAt[f] < 0)
336 ackedAt[f] = tick;
337 }
338 if (baseFrame > oldBase) {
339 duration = c.T;
340 duplicates = 0;
341 }
342 if (!sr) {
343 if (e.Nak)
344 GoBack();
345 else {
346 next = Math.Max(next, baseFrame);
347 timer = baseFrame == next ? -1 : tick + duration;
348 }
349 }
350 if (e.Nak)
351 Decrease(false);
352 Log("SENDER", $"GOT{kind} a={e.Value} #{e.Number} base={baseFrame}");
353 } else {
354 Log("SENDER", $"OLD{kind} a={e.Value} #{e.Number}");
355 if (!sr && !e.Nak && c.Dupack > 0 && baseFrame >= 1 &&
356 e.Value == (baseFrame - 1) % modulus) {
357 ++duplicates;
358 if (duplicates == c.Dupack) {
359 ++fast;
360 Log("SENDER", $"FASTRETX f={baseFrame}");
361 GoBack();
362 Decrease(false);
363 }
364 }
365 }
366 Cwnd(oldWindow, oldThreshold);
367 }
368 void Timeout(int frame) {
369 int oldWindow = cwnd, oldThreshold = threshold;
370 ++timeouts;
371 Log("SENDER", $"TIMEOUT f={frame}");
372 if (sr) {
373 timers[frame] = -1;
374 if (!queue.Contains(frame))
375 queue.Add(frame);
376 } else
377 GoBack();
378 if (c.Backoff)
379 duration = Math.Min(duration * 2, c.T * 8);
380 Decrease(true);
381 Cwnd(oldWindow, oldThreshold);
382 }
383 void Send(int frame) {
384 ++data;
385 bool retry = tx[frame] > 0;
386 if (retry)
387 ++resent;
388 else
389 first[frame] = tick;
390 ++tx[frame];
391 Log("SENDER", $"{(retry ? "RESEND" : "SEND")} f={frame} seq={frame % modulus} #{data}");
392 dataEvents.Add(new Transmission(tick + c.Delay + c.DelayData.GetValueOrDefault(data),
393 data, frame, Outcome(true, data), false));
394 if (sr)
395 timers[frame] = tick + duration;
396 else if (timer < 0)
397 timer = tick + duration;
398 }
399 void Snapshot() {
400 char[] sender = new char[c.N], receiving = new char[c.N];
401 for (int f = 0; f < c.N; ++f) {
402 sender[f] = (sr ? acknowledged[f] : f < baseFrame) ? 'A'
403 : (sr ? queue.Contains(f) : next <= f && f < high) ? 'R'
404 : (sr ? f < high : f < next) ? 'O'
405 : f < baseFrame + Effective ? 'W'
406 : '.';
407 receiving[f] = f < receiver ? 'D'
408 : sr && buffer.ContainsKey(f) ? 'B'
409 : (sr ? f < receiver + c.W : f == receiver) ? 'W'
410 : '.';
411 }
412 Log("ALWAYS", $"SNAPSHOT sender={new string(sender)} receiver={new string(receiving)}");
413 }
414 public Result Run() {
415 Output.Add($"config: {c.Protocol} n={c.N} w={c.W} k={c.K} T={c.T} d={c.Delay}");
416 string On(bool value) => value ? "on" : "off";
417 Output.Add(
418 $"options: interval={c.Interval} limit={c.Limit} dupack={(c.Dupack == 0 ? "off" : c.Dupack.ToString())} nak={On(c.Nak)} backoff={On(c.Backoff)} congestion={On(c.Congestion)} unsafe={On(c.Unsafe)} trace={c.Trace.ToLowerInvariant()} random={(c.Random ? $"{c.Seed}/{c.Loss}/{c.Corruption}" : "off")}");
419 bool complete = false;
420 for (tick = 0; tick <= c.Limit; ++tick) {
421 foreach (Transmission e in dataEvents.Where(e => e.Arrival == tick)
422 .OrderBy(e => e.Number)
423 .ToArray())
424 Receive(e);
425 foreach (Transmission e in controlEvents.Where(e => e.Arrival == tick)
426 .OrderBy(e => e.Number)
427 .ToArray())
428 Control(e);
429 complete = baseFrame == c.N;
430 if (!complete) {
431 if (sr) {
432 for (int f = baseFrame; f < high; ++f)
433 if (timers[f] == tick)
434 Timeout(f);
435 } else if (timer == tick)
436 Timeout(baseFrame);
437 if (sr) {
438 if (queue.Count > 0) {
439 int frame = queue[0];
440 queue.RemoveAt(0);
441 Send(frame);
442 } else if (high < c.N && high < baseFrame + Effective &&
443 high * c.Interval <= tick) {
444 int frame = high++;
445 Send(frame);
446 }
447 } else if (next < c.N && next < baseFrame + Effective &&
448 (next < high || next * c.Interval <= tick)) {
449 int frame = next++;
450 high = Math.Max(high, next);
451 Send(frame);
452 }
453 }
454 if (c.Snapshots.Contains(tick))
455 Snapshot();
456 if (complete || tick == c.Limit) {
457 Log("ALWAYS", complete ? "DONE" : "ABORT");
458 break;
459 }
460 }
461 for (int f = 0; f < c.N; ++f)
462 Output.Add(
463 $"frame {f}: seq={f % modulus} tx={tx[f]} first={Tick(first[f])} delivered={Tick(deliveredAt[f])} acked={Tick(ackedAt[f])}");
464 Output.Add(
465 $"totals: data={data} resent={resent} control={control} nak={naks} timeouts={timeouts} fastretx={fast}");
466 Output.Add(
467 $"channel: lost_data={lostData} lost_ctrl={lostControl} corrupt_data={corruptData} corrupt_ctrl={corruptControl}");
468 Output.Add(
469 $"receiver: delivered={delivered} errors={errors} discarded={discarded} max_buffer={maxBuffer}");
470 Output.Add(
471 $"result: completion={(complete ? tick.ToString() : "-")} goodput={(complete ? Fixed(delivered, tick, 2) : "-")} efficiency={Fixed(delivered * 100L, data, 1)}% utilization={Fixed(data * 100L, tick + 1, 1)}% latency={(delivered == 0 ? "-" : Fixed(latency, delivered, 2))}");
472 return new Result(c, complete ? "complete" : "aborted") { Delivered = delivered,
473 Data = data,
474 Completion = tick };
475 }
476 }
477 static void Main() {
478 int count = int.Parse(Console.ReadLine());
479 List<Result> results = new();
480 for (int i = 0; i < count; ++i) {
481 Config config = new() { Name = Console.ReadLine().Split(' ')[1] };
482 List<string[]> lines = new();
483 string line;
484 while ((line = Console.ReadLine()) != "END")
485 lines.Add(line.Split(' '));
486 Output.Add("== " + config.Name + " ==");
487 string error = Validate(config, lines);
488 if (error != null) {
489 Output.Add("error: " + error);
490 results.Add(new Result(config, "invalid"));
491 } else
492 results.Add(new Simulation(config).Run());
493 }
494 Output.Add("== ranking ==");
495 List<Result> completed = results.Where(r => r.Status == "complete").ToList();
496 completed.Sort((a, b) => {
497 int order =
498 ((long)b.Delivered * a.Completion).CompareTo((long)a.Delivered * b.Completion);
499 if (order == 0)
500 order = ((long)b.Delivered * a.Data).CompareTo((long)a.Delivered * b.Data);
501 return order != 0 ? order : results.IndexOf(a).CompareTo(results.IndexOf(b));
502 });
503 for (int i = 0; i < completed.Count; ++i) {
504 Result r = completed[i];
505 Output.Add(
506 $"{i + 1}. {r.Config.Name} {r.Config.Protocol} goodput={Fixed(r.Delivered, r.Completion, 2)} efficiency={Fixed(r.Delivered * 100L, r.Data, 1)}%");
507 }
508 foreach (Result r in results.Where(r => r.Status != "complete"))
509 Output.Add($"- {r.Config.Name} {r.Status}");
510 Output.Add("== protocols ==");
511 foreach (string protocol in new[] { "SW", "GBN", "SR" }) {
512 List<Result> runs = completed.Where(r => r.Config.Protocol == protocol).ToList();
513 if (runs.Count == 0)
514 continue;
515 long delivered = runs.Sum(r => (long)r.Delivered),
516 ticks = runs.Sum(r => (long)r.Completion), data = runs.Sum(r => (long)r.Data);
517 Output.Add(
518 $"{protocol}: runs={runs.Count} goodput={Fixed(delivered, ticks, 2)} efficiency={Fixed(delivered * 100, data, 1)}%");
519 }
520 Console.WriteLine(string.Join("\n", Output));
521 }
