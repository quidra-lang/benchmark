28func modulo(_ a: Int, _ b: Int) -> Int {
29 (a % b + b) % b
30}
31func fixed(_ numerator: Int, _ denominator: Int, _ digits: Int) -> String {
32 let scale = digits == 2 ? 100 : 10
33 let value = denominator == 0 ? 0 : numerator * scale / denominator
34 let fraction = String(value % scale)
35 return "\(value / scale)." + String(repeating: "0", count: digits - fraction.count) + fraction
36}
37func stamp(_ tick: Int) -> String {

66func validate(_ name: String, _ lines: [[String]]) throws -> Config {
67 var values: [String: [String]] = [:]
68 func fail(_ message: String) throws -> Never {
69 throw SettingsError(message: message)
70 }
71 for line in lines {
72 let key = line[0]
73 guard let index = keys.firstIndex(of: key) else {
74 try fail("unknown key \(key)")
75 }
76 if values[key] != nil {
77 try fail("duplicate key \(key)")
78 }
79 let count = line.count - 1
80 if key == "RANDOM" ? count != 3 : index >= 15 ? count < 1 : count != 1 {
81 try fail("wrong number of values for \(key)")
82 }
83 values[key] = Array(line.dropFirst())
84 }
85 for key in keys.prefix(6) {
86 if values[key] == nil {
87 try fail("missing key \(key)")
88 }
89 }
90 for (index, key) in keys.enumerated() {
91 guard let v = values[key] else {
92 continue
93 }
94 var valid = true
95 switch index {
96 case 0:
97 valid = ["SW", "GBN", "SR"].contains(v[0])
98 case 9...12:
99 valid = ["ON", "OFF"].contains(v[0])
100 case 13:
101 valid = ["ALL", "SENDER", "RECEIVER", "NONE"].contains(v[0])
102 case 14:
103 let numbers = v.map { Int($0)! }
104 valid = (1...2147483646)
105 .contains(numbers[0]) && numbers[1] >= 0 && numbers[2] >= 0 && numbers[1] + numbers[2] <= 100
106 case 15...21:
107 var seen: Set<Int> = []
108 for value in v {
109 let parts = value.split(separator: ":").map { Int($0)! }
110 let lower = index == 21 ? 0 : 1
111 let upper = index == 21 ? 2000 : 9999
112 if !(lower...upper).contains(parts[0]) || !seen.insert(parts[0]).inserted {
113 valid = false
114 }
115 if index == 19 || index == 20 {
116 if !(1...50).contains(parts[1]) {
117 valid = false
118 }
119 }
120 }
121 default:
122 let number = Int(v[0])!
123 let lower = index == 6 ? 0 : 1
124 let upper: Int
125 switch index {
126 case 1:
127 upper = 60
128 case 2:
129 upper = 32
130 case 3:
131 upper = 6
132 case 4:
133 upper = 50
134 case 7:
135 upper = 2000
136 default:
137 upper = 10
138 }
139 valid = (lower...upper).contains(number)
140 }
141 if !valid {
142 try fail("bad value for \(key)")
143 }
144 }
145 func get(_ key: String, _ fallback: String) -> String {
146 values[key]?.first ?? fallback
147 }
148 func number(_ key: String, _ fallback: Int) -> Int {
149 Int(get(key, String(fallback)))!
150 }
151 var c = Config(name: name)
152 c.protocolName = get("PROTOCOL", "")
153 c.n = number("FRAMES", 0)
154 c.w = number("WINDOW", 0)
155 c.k = number("SEQBITS", 0)
156 c.timeout = number("TIMEOUT", 0)
157 c.delay = number("DELAY", 0)
158 c.interval = number("INTERVAL", 0)
159 c.limit = number("MAXTICKS", 1000)
160 c.dupack = number("DUPACK", 0)
161 c.nak = get("NAK", "OFF") == "ON"
162 c.backoff = get("BACKOFF", "OFF") == "ON"
163 c.congestion = get("CONGESTION", "OFF") == "ON"
164 c.unsafeWindow = get("UNSAFE", "OFF") == "ON"
165 c.trace = get("TRACE", "ALL")
166 if c.protocolName == "SR" && values["DUPACK"] != nil {
167 try fail("DUPACK not supported for SR")
168 }
169 let upper = c.protocolName == "SW" ? 1 : c.protocolName == "GBN" ? (1 << c.k) - 1 : 1 << (c.k - 1)
170 if c.w > upper && !(c.unsafeWindow && c.protocolName != "SW") {
171 try fail("window \(c.w) exceeds limit \(upper) for \(c.protocolName)")
172 }
173 if let v = values["RANDOM"] {
174 c.random = (Int(v[0])!, Int(v[1])!, Int(v[2])!)
175 }
176 func set(_ key: String) -> Set<Int> {
177 Set((values[key] ?? []).map { Int($0)! })
178 }
179 func delays(_ key: String) -> [Int: Int] {
180 var result: [Int: Int] = [:]
181 for value in values[key] ?? [] {
182 let parts = value.split(separator: ":").map { Int($0)! }
183 result[parts[0]] = parts[1]
184 }
185 return result
186 }
187 c.lostData = set("LOSEDATA")
188 c.lostControl = set("LOSECTRL")
189 c.corruptData = set("CORRUPTDATA")
190 c.corruptControl = set("CORRUPTCTRL")
191 c.delayData = delays("DELAYDATA")
192 c.delayControl = delays("DELAYCTRL")
193 c.snapshots = set("SNAPSHOT")
194 if let conflict = c.lostData.intersection(c.corruptData)
195 .min()
196 {
197 try fail("data #\(conflict) both lost and corrupted")
198 }
199 if let conflict = c.lostControl.intersection(c.corruptControl)
200 .min()
201 {
202 try fail("control #\(conflict) both lost and corrupted")
203 }
204 return c
205}

222@MainActor
223final class Simulation {
224 let c: Config
225 let sr: Bool
226 let modulus: Int
227 var dataEvents: [Transmission] = []
228 var controlEvents: [Transmission] = []
229 var queue: [Int] = []
230 var buffer: [Int: Int] = [:]
231 var acknowledged: [Bool]
232 var timers: [Int]
233 var tx: [Int]
234 var first: [Int]
235 var deliveredAt: [Int]
236 var ackedAt: [Int]
237 var tick = 0
238 var base = 0
239 var high = 0
240 var next = 0
241 var duration: Int
242 var timer = -1
243 var duplicates = 0
244 var receiver = 0
245 var cwnd = 1
246 var threshold: Int
247 var credit = 0
248 var random: Int
249 var nakSent = false
250 var data = 0
251 var control = 0
252 var resent = 0
253 var naks = 0
254 var timeouts = 0
255 var fast = 0
256 var lostData = 0
257 var lostControl = 0
258 var corruptData = 0
259 var corruptControl = 0
260 var delivered = 0
261 var errors = 0
262 var discarded = 0
263 var maxBuffer = 0
264 var latency = 0
265 init(_ config: Config) {
266 c = config
267 sr = config.protocolName == "SR"
268 modulus = 1 << config.k
269 duration = config.timeout
270 threshold = config.w
271 random = config.random?.seed ?? 0
272 acknowledged = Array(repeating: false, count: config.n)
273 timers = Array(repeating: -1, count: config.n)
274 tx = Array(repeating: 0, count: config.n)
275 first = Array(repeating: -1, count: config.n)
276 deliveredAt = Array(repeating: -1, count: config.n)
277 ackedAt = Array(repeating: -1, count: config.n)
278 }
279 var effective: Int {
280 c.congestion ? min(c.w, cwnd) : c.w
281 }
282 func log(_ side: String, _ text: String) {
283 if c.trace == "ALL" || c.trace == side || side == "ALWAYS" {
284 output.append("\(tick) \(text)")
285 }
286 }
287 func outcome(_ isData: Bool, _ number: Int) -> Int {
288 var result = 0
289 if let option = c.random {
290 random = random * 48271 % 2147483647
291 let value = random % 100
292 result = value < option.loss ? 1 : value < option.loss + option.corruption ? 2 : 0
293 }
294 if (isData ? c.lostData : c.lostControl).contains(number) {
295 result = 1
296 }
297 if (isData ? c.corruptData : c.corruptControl).contains(number) {
298 result = 2
299 }
300 return result
301 }
302 func sendControl(_ nak: Bool, _ value: Int) {
303 control += 1
304 if nak {
305 naks += 1
306 }
307 log("RECEIVER", "\(nak ? "NAK" : "ACK") a=\(value) #\(control)")
308 let event = Transmission(
309 arrival: tick + c.delay + c.delayControl[control, default: 0],
310 number: control,
311 value: value,
312 outcome: outcome(false, control),
313 nak: nak
314 )
315 controlEvents.append(event)
316 }
317 func deliver(_ payload: Int) {
318 if payload == receiver {
319 log("RECEIVER", "DELIVER f=\(receiver)")
320 delivered += 1
321 if deliveredAt[payload] < 0 {
322 deliveredAt[payload] = tick
323 }
324 latency += tick - first[payload]
325 } else {
326 log("RECEIVER", "DELIVER f=\(receiver) wrong=\(payload)")
327 errors += 1
328 }
329 receiver += 1
330 nakSent = false
331 }
332 func discard(_ sequence: Int) {
333 discarded += 1
334 log("RECEIVER", "DISCARD seq=\(sequence)")
335 }
336 func receive(_ e: Transmission) {
337 if e.outcome != 0 {
338 if e.outcome == 1 {
339 lostData += 1
340 } else {
341 corruptData += 1
342 }
343 log("RECEIVER", "\(e.outcome == 1 ? "LOST" : "CORRUPT") data #\(e.number)")
344 return
345 }
346 let sequence = e.value % modulus
347 log("RECEIVER", "RECV seq=\(sequence) #\(e.number)")
348 if !sr {
349 if sequence == receiver % modulus {
350 deliver(e.value)
351 sendControl(false, sequence)
352 } else {
353 discard(sequence)
354 if c.nak && !nakSent {
355 sendControl(true, receiver % modulus)
356 nakSent = true
357 } else if receiver > 0 {
358 sendControl(false, (receiver - 1) % modulus)
359 }
360 }
361 return
362 }
363 let offset = modulo(sequence - receiver, modulus)
364 let position = receiver + offset
365 if offset < c.w {
366 if buffer[position] != nil {
367 discard(sequence)
368 sendControl(false, sequence)
369 } else if offset == 0 {
370 deliver(e.value)
371 while let payload = buffer.removeValue(forKey: receiver) {
372 deliver(payload)
373 }
374 sendControl(false, sequence)
375 } else {
376 buffer[position] = e.value
377 maxBuffer = max(maxBuffer, buffer.count)
378 log("RECEIVER", "BUFFER f=\(position)")
379 sendControl(false, sequence)
380 if c.nak && !nakSent {
381 sendControl(true, receiver % modulus)
382 nakSent = true
383 }
384 }
385 } else {
386 discard(sequence)
387 if modulo(receiver - sequence, modulus) <= c.w {
388 sendControl(false, sequence)
389 }
390 }
391 }
392 func goBack() {
393 next = base
394 timer = -1
395 duplicates = 0
396 }
397 func increase(_ count: Int) {
398 if !c.congestion {
399 return
400 }
401 for _ in 0..<count {
402 if cwnd >= c.w {
403 break
404 }
405 if cwnd < threshold {
406 cwnd += 1
407 } else {
408 credit += 1
409 if credit == cwnd {
410 cwnd += 1
411 credit = 0
412 }
413 }
414 }
415 }
416 func decrease(_ timeout: Bool) {
417 if !c.congestion {
418 return
419 }
420 threshold = max(cwnd / 2, 1)
421 cwnd = timeout ? 1 : threshold
422 credit = 0
423 }
424 func cwndLine(_ oldWindow: Int, _ oldThreshold: Int) {
425 if c.congestion && (oldWindow != cwnd || oldThreshold != threshold) {
426 log("SENDER", "CWND \(cwnd) ssthresh=\(threshold)")
427 }
428 }
429 func controlArrival(_ e: Transmission) {
430 let kind = e.nak ? "NAK" : "ACK"
431 if e.outcome != 0 {
432 if e.outcome == 1 {
433 lostControl += 1
434 } else {
435 corruptControl += 1
436 }
437 log("SENDER", "\(e.outcome == 1 ? "LOST" : "CORRUPT") \(kind.lowercased()) #\(e.number)")
438 return
439 }
440 let oldWindow = cwnd, oldThreshold = threshold, oldBase = base
441 let frame = base + modulo(e.value - base, modulus)
442 let accepted = sr ? frame < high && !acknowledged[frame] : e.nak ? frame <= high : frame < high
443 if accepted {
444 if sr {
445 timers[frame] = -1
446 if e.nak {
447 if !queue.contains(frame) {
448 queue.append(frame)
449 }
450 } else {
451 acknowledged[frame] = true
452 queue.removeAll { $0 == frame }
453 if ackedAt[frame] < 0 {
454 ackedAt[frame] = tick
455 }
456 while base < high && acknowledged[base] {
457 base += 1
458 }
459 increase(1)
460 }
461 } else {
462 base = frame + (e.nak ? 0 : 1)
463 increase(base - oldBase)
464 for f in oldBase..<base {
465 if ackedAt[f] < 0 {
466 ackedAt[f] = tick
467 }
468 }
469 }
470 if base > oldBase {
471 duration = c.timeout
472 duplicates = 0
473 }
474 if !sr {
475 if e.nak {
476 goBack()
477 } else {
478 next = max(next, base)
479 timer = base == next ? -1 : tick + duration
480 }
481 }
482 if e.nak {
483 decrease(false)
484 }
485 log("SENDER", "GOT\(kind) a=\(e.value) #\(e.number) base=\(base)")
486 } else {
487 log("SENDER", "OLD\(kind) a=\(e.value) #\(e.number)")
488 if !sr && !e.nak && c.dupack > 0 && base >= 1 && e.value == (base - 1) % modulus {
489 duplicates += 1
490 if duplicates == c.dupack {
491 fast += 1
492 log("SENDER", "FASTRETX f=\(base)")
493 goBack()
494 decrease(false)
495 }
496 }
497 }
498 cwndLine(oldWindow, oldThreshold)
499 }
500 func timeout(_ frame: Int) {
501 let oldWindow = cwnd, oldThreshold = threshold
502 timeouts += 1
503 log("SENDER", "TIMEOUT f=\(frame)")
504 if sr {
505 timers[frame] = -1
506 if !queue.contains(frame) {
507 queue.append(frame)
508 }
509 } else {
510 goBack()
511 }
512 if c.backoff {
513 duration = min(duration * 2, c.timeout * 8)
514 }
515 decrease(true)
516 cwndLine(oldWindow, oldThreshold)
517 }
518 func send(_ frame: Int) {
519 data += 1
520 let retry = tx[frame] > 0
521 if retry {
522 resent += 1
523 } else {
524 first[frame] = tick
525 }
526 tx[frame] += 1
527 log("SENDER", "\(retry ? "RESEND" : "SEND") f=\(frame) seq=\(frame % modulus) #\(data)")
528 let event = Transmission(
529 arrival: tick + c.delay + c.delayData[data, default: 0],
530 number: data,
531 value: frame,
532 outcome: outcome(true, data),
533 nak: false
534 )
535 dataEvents.append(event)
536 if sr {
537 timers[frame] = tick + duration
538 } else if timer < 0 {
539 timer = tick + duration
540 }
541 }
542 func snapshot() {
543 var sender = "", receiving = ""
544 for f in 0..<c.n {
545 if sr ? acknowledged[f] : f < base {
546 sender += "A"
547 } else if sr ? queue.contains(f) : next <= f && f < high {
548 sender += "R"
549 } else if sr ? f < high : f < next {
550 sender += "O"
551 } else {
552 sender += f < base + effective ? "W" : "."
553 }
554 if f < receiver {
555 receiving += "D"
556 } else if sr && buffer[f] != nil {
557 receiving += "B"
558 } else {
559 receiving += (sr ? f < receiver + c.w : f == receiver) ? "W" : "."
560 }
561 }
562 log("ALWAYS", "SNAPSHOT sender=\(sender) receiver=\(receiving)")
563 }
564 func run(_ order: Int) -> Result {
565 output.append("config: \(c.protocolName) n=\(c.n) w=\(c.w) k=\(c.k) T=\(c.timeout) d=\(c.delay)")
566 func on(_ value: Bool) -> String {
567 value ? "on" : "off"
568 }
569 let randomOption = c.random.map { "\($0.seed)/\($0.loss)/\($0.corruption)" } ?? "off"
570 let dupack = c.dupack == 0 ? "off" : String(c.dupack)
571 output
572 .append(
573 "options: interval=\(c.interval) limit=\(c.limit) dupack=\(dupack) nak=\(on(c.nak)) backoff=\(on(c.backoff)) congestion=\(on(c.congestion)) unsafe=\(on(c.unsafeWindow)) trace=\(c.trace.lowercased()) random=\(randomOption)"
574 )
575 var complete = false
576 for currentTick in 0...c.limit {
577 tick = currentTick
578 for event in dataEvents.filter({ $0.arrival == tick }) {
579 receive(event)
580 }
581 for event in controlEvents.filter({ $0.arrival == tick }) {
582 controlArrival(event)
583 }
584 complete = base == c.n
585 if !complete {
586 if sr {
587 for frame in base..<high {
588 if timers[frame] == tick {
589 timeout(frame)
590 }
591 }
592 } else if timer == tick {
593 timeout(base)
594 }
595 if sr {
596 if !queue.isEmpty {
597 send(queue.removeFirst())
598 } else if high < c.n && high < base + effective && high * c.interval <= tick {
599 let frame = high
600 high += 1
601 send(frame)
602 }
603 } else if next < c.n && next < base + effective && (next < high || next * c.interval <= tick) {
604 let frame = next
605 next += 1
606 high = max(high, next)
607 send(frame)
608 }
609 }
610 if c.snapshots.contains(tick) {
611 snapshot()
612 }
613 if complete || tick == c.limit {
614 log("ALWAYS", complete ? "DONE" : "ABORT")
615 break
616 }
617 }
618 for f in 0..<c.n {
619 output
620 .append(
621 "frame \(f): seq=\(f % modulus) tx=\(tx[f]) first=\(stamp(first[f])) delivered=\(stamp(deliveredAt[f])) acked=\(stamp(ackedAt[f]))"
622 )
623 }
624 output
625 .append(
626 "totals: data=\(data) resent=\(resent) control=\(control) nak=\(naks) timeouts=\(timeouts) fastretx=\(fast)"
627 )
628 output
629 .append(
630 "channel: lost_data=\(lostData) lost_ctrl=\(lostControl) corrupt_data=\(corruptData) corrupt_ctrl=\(corruptControl)"
631 )
632 output
633 .append("receiver: delivered=\(delivered) errors=\(errors) discarded=\(discarded) max_buffer=\(maxBuffer)")
634 let completion = complete ? String(tick) : "-"
635 let goodput = complete ? fixed(delivered, tick, 2) : "-"
636 let average = delivered == 0 ? "-" : fixed(latency, delivered, 2)
637 output
638 .append(
639 "result: completion=\(completion) goodput=\(goodput) efficiency=\(fixed(delivered * 100, data, 1))% utilization=\(fixed(data * 100, tick + 1, 1))% latency=\(average)"
640 )
641 return Result(
642 name: c.name,
643 protocolName: c.protocolName,
644 status: complete ? "complete" : "aborted",
645 delivered: delivered,
646 data: data,
647 completion: tick,
648 order: order
649 )
650 }
651}
652let count = Int(readLine()!)!
653var results: [Result] = []
654for order in 0..<count {
655 let name = String(readLine()!.split(separator: " ")[1])
656 var settings: [[String]] = []
657 while let line = readLine(), line != "END" {
658 settings.append(line.split(separator: " ").map(String.init))
659 }
660 output.append("== \(name) ==")
661 do {
662 let config = try validate(name, settings)
663 results.append(Simulation(config).run(order))
664 } catch let error as SettingsError {
665 output.append("error: \(error.message)")
666 results.append(Result(
667 name: name,
668 protocolName: "",
669 status: "invalid",
670 delivered: 0,
671 data: 0,
672 completion: 0,
673 order: order
674 ))
675 } catch { fatalError("Unexpected settings error") }
676}
677output.append("== ranking ==")
678let completed = results.filter { $0.status == "complete" }.sorted {
679 let left = $0.delivered * $1.completion, right = $1.delivered * $0.completion
680 if left != right {
681 return left > right
682 }
683 let leftEfficiency = $0.delivered * $1.data, rightEfficiency = $1.delivered * $0.data
684 if leftEfficiency != rightEfficiency {
685 return leftEfficiency > rightEfficiency
686 }
687 return $0.order < $1.order
688}
689for (index, result) in completed.enumerated() {
690 output
691 .append(
692 "\(index + 1). \(result.name) \(result.protocolName) goodput=\(fixed(result.delivered, result.completion, 2)) efficiency=\(fixed(result.delivered * 100, result.data, 1))%"
693 )
694}
695for result in results where result.status != "complete" {
696 output.append("- \(result.name) \(result.status)")
697}
698output.append("== protocols ==")
699for protocolName in ["SW", "GBN", "SR"] {
700 let runs = completed.filter { $0.protocolName == protocolName }
701 if runs.isEmpty {
702 continue
703 }
704 let delivered = runs.reduce(0) { $0 + $1.delivered }
705 let ticks = runs.reduce(0) { $0 + $1.completion }
706 let data = runs.reduce(0) { $0 + $1.data }
707 output
708 .append(
709 "\(protocolName): runs=\(runs.count) goodput=\(fixed(delivered, ticks, 2)) efficiency=\(fixed(delivered * 100, data, 1))%"
710 )
711}
712print(output.joined(separator: "\n"))
