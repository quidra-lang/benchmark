                    if k == "DELAYDATA" or k == "DELAYCTRL" then
                        local a, b = s:match("^([^:]+):([^:]+)$")
                        number, extra = tonumber(a), tonumber(b)
                        if not inrange(number, 1, 9999) or not inrange(extra, 1, 50) then
                            valid = false
                        end
                    if values[number] then
                        valid = false
                    end
                    values[number] = extra or true
                end
                c[k] = values
            end
            if not valid then
                return nil, "bad value for " .. k
    local c, err = validate(lines)
    local result
    if err then
        emit("error: " .. err)
        result = {invalid = true}
    else
        result = simulate(c)
