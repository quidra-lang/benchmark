12 'LOSEDATA', 'LOSECTRL', 'CORRUPTDATA', 'CORRUPTCTRL',
13 'DELAYDATA', 'DELAYCTRL', 'SNAPSHOT'
14]);
15const DELAY_KEYS = new Set(['DELAYDATA', 'DELAYCTRL']);
16const CHOICES: Record<string, string[]> = {
17 PROTOCOL: PROTOCOLS, NAK: ['ON', 'OFF'], BACKOFF: ['ON', 'OFF'],
18 CONGESTION: ['ON', 'OFF'], UNSAFE: ['ON', 'OFF'],
19 TRACE: ['ALL', 'SENDER', 'RECEIVER', 'NONE']

23 TIMEOUT: [1, 50], DELAY: [1, 10], INTERVAL: [0, 10],
24 MAXTICKS: [1, 2000], DUPACK: [1, 10],

26 CORRUPTDATA: [1, 9999], CORRUPTCTRL: [1, 9999], SNAPSHOT: [0, 2000]

49function truncated(numerator: number, denominator: number, digits: number): string {
50 const scale = 10 ** digits;
51 const value = Math.floor(numerator * scale / denominator);
52 return Math.floor(value / scale) + '.' + String(value % scale).padStart(digits, '0');
53}
54
55function tickText(tick: number | null): string { return tick === null ? '-' : String(tick); }
56function mod(value: number, modulus: number): number { return ((value % modulus) + modulus) % modulus; }
57function option(settings: Map<string, string[]>, key: string, fallback: string): string { return (settings.get(key) ?? [fallback])[0]; }
58function numberSet(settings: Map<string, string[]>, key: string): Set<number> { return new Set((settings.get(key) ?? []).map(Number)); }
59function delayMap(settings: Map<string, string[]>, key: string): Map<number, number> {
60 return new Map((settings.get(key) ?? []).map(value => value.split(':').map(Number) as [number, number]));
61}
62function inRange(value: number, bounds: [number, number]): boolean { return bounds[0] <= value && value <= bounds[1]; }
63function unique<T>(values: T[]): boolean { return new Set(values).size === values.length; }
64function validValue(key: string, values: string[]): boolean {
65 if (Object.hasOwn(CHOICES, key)) return CHOICES[key].includes(values[0]);
66 if (!LIST_KEYS.has(key) && Object.hasOwn(RANGES, key)) {
67 return inRange(Number(values[0]), RANGES[key]);
68 }
69 if (key === 'RANDOM') {
70 const [seed, loss, corrupt] = values.map(Number);
71 return inRange(seed, [1, RANDOM_MODULUS - 1]) && loss >= 0 && corrupt >= 0 && loss + corrupt <= 100;
72 }
73 if (DELAY_KEYS.has(key)) {
74 const pairs = values.map(value => value.split(':').map(Number));
75 return pairs.every(([n, extra]) => inRange(n, [1, 9999]) && inRange(extra, [1, 50])) &&
76 unique(pairs.map(([n]) => n));
77 }
78 const numbers = values.map(Number);
79 return numbers.every(n => inRange(n, RANGES[key])) && unique(numbers);
80}

106 constructor(settings: Map<string, string[]>) {
107 this.protocol = option(settings, 'PROTOCOL', '');
108 this.frames = Number(option(settings, 'FRAMES', '0'));
109 this.window = Number(option(settings, 'WINDOW', '0'));
110 this.seqbits = Number(option(settings, 'SEQBITS', '0'));
111 this.timeout = Number(option(settings, 'TIMEOUT', '0'));
112 this.delay = Number(option(settings, 'DELAY', '0'));
113 this.interval = Number(option(settings, 'INTERVAL', '0'));
114 this.limit = Number(option(settings, 'MAXTICKS', '1000'));
115 this.dupack = settings.has('DUPACK') ? Number(settings.get('DUPACK')![0]) : null;
116 this.nak = option(settings, 'NAK', 'OFF') === 'ON';
117 this.backoff = option(settings, 'BACKOFF', 'OFF') === 'ON';
118 this.congestion = option(settings, 'CONGESTION', 'OFF') === 'ON';
119 this.unsafe = option(settings, 'UNSAFE', 'OFF') === 'ON';
120 this.trace = option(settings, 'TRACE', 'ALL');
121 this.random = settings.has('RANDOM') ? settings.get('RANDOM')!.map(Number) as [number, number, number] : null;
122 this.loseData = numberSet(settings, 'LOSEDATA');
123 this.loseCtrl = numberSet(settings, 'LOSECTRL');
124 this.corruptData = numberSet(settings, 'CORRUPTDATA');
125 this.corruptCtrl = numberSet(settings, 'CORRUPTCTRL');
126 this.delayData = delayMap(settings, 'DELAYDATA');
127 this.delayCtrl = delayMap(settings, 'DELAYCTRL');
128 this.snapshots = numberSet(settings, 'SNAPSHOT');
129 }
130
131 windowLimit(): number {
132 if (this.protocol === 'SW') return 1;
133 if (this.protocol === 'GBN') return 2 ** this.seqbits - 1;
134 return 2 ** (this.seqbits - 1);
135 }
136
137 configLine(): string {
138 return `config: ${this.protocol} n=${this.frames} w=${this.window} k=${this.seqbits} T=${this.timeout} d=${this.delay}`;
139 }
140
141 optionsLine(): string {
142 const dupack = this.dupack === null ? 'off' : this.dupack;
143 const random = this.random === null ? 'off' : this.random.join('/');
144 const switchText = (flag: boolean): string => flag ? 'on' : 'off';
145 return `options: interval=${this.interval} limit=${this.limit} dupack=${dupack} ` +
146 `nak=${switchText(this.nak)} backoff=${switchText(this.backoff)} ` +
147 `congestion=${switchText(this.congestion)} unsafe=${switchText(this.unsafe)} ` +
148 `trace=${this.trace.toLowerCase()} random=${random}`;
149 }

152function checkSettings(lines: SettingLine[]): Config {
153 const settings = new Map<string, string[]>();
154 for (const [key, values] of lines) {
155 if (!KEYS.includes(key)) throw new SettingsError(`unknown key ${key}`);
156 if (settings.has(key)) throw new SettingsError(`duplicate key ${key}`);
157 const countMatches = key === 'RANDOM' ? values.length === 3 :
158 LIST_KEYS.has(key) ? values.length >= 1 : values.length === 1;
159 if (!countMatches) throw new SettingsError(`wrong number of values for ${key}`);
160 settings.set(key, values);
161 }
162 for (const key of REQUIRED) {
163 if (!settings.has(key)) throw new SettingsError(`missing key ${key}`);
164 }
165 for (const key of KEYS) {
166 if (settings.has(key) && !validValue(key, settings.get(key)!)) {
167 throw new SettingsError(`bad value for ${key}`);
168 }
169 }
170 const config = new Config(settings);
171 if (config.protocol === 'SR' && config.dupack !== null) {
172 throw new SettingsError('DUPACK not supported for SR');
173 }
174 const limit = config.windowLimit();
175 if ((config.protocol === 'SW' || !config.unsafe) && config.window > limit) {
176 throw new SettingsError(`window ${config.window} exceeds limit ${limit} for ${config.protocol}`);
177 }
178 const dataBoth = [...config.loseData].filter(n => config.corruptData.has(n));
179 if (dataBoth.length) throw new SettingsError(`data #${Math.min(...dataBoth)} both lost and corrupted`);
180 const ctrlBoth = [...config.loseCtrl].filter(n => config.corruptCtrl.has(n));
181 if (ctrlBoth.length) throw new SettingsError(`control #${Math.min(...ctrlBoth)} both lost and corrupted`);
182 return config;
183}

192class LossGenerator {
193 state: number;
194 readonly loss: number;
195 readonly corruption: number;
196
197 constructor(seed: number, loss: number, corruption: number) {
198 this.state = seed;
199 this.loss = loss;
200 this.corruption = corruption;
201 }
202
203 nextOutcome(): Outcome {
204 this.state = RANDOM_MULTIPLIER * this.state % RANDOM_MODULUS;
205 const roll = this.state % 100;
206 if (roll < this.loss) return 'LOST';
207 if (roll < this.loss + this.corruption) return 'CORRUPT';
208 return 'NORMAL';
209 }
210}

212class Channel {
213 readonly delay: number;
214 readonly losses: Set<number>;
215 readonly corruptions: Set<number>;
216 readonly extraDelays: Map<number, number>;
217 sent = 0;
218 lost = 0;
219 corrupted = 0;
220 inTransit: Transmission[] = [];
221
222 constructor(delay: number, losses: Set<number>, corruptions: Set<number>, extraDelays: Map<number, number>) {
223 this.delay = delay;
224 this.losses = losses;
225 this.corruptions = corruptions;
226 this.extraDelays = extraDelays;
227 }
228
229 send(kind: TransmissionKind, value: number, payload: number | null, tick: number, generator: LossGenerator | null): Transmission {
230 const number = ++this.sent;
231 let outcome = generator === null ? 'NORMAL' : generator.nextOutcome();
232 if (this.losses.has(number)) outcome = 'LOST';
233 else if (this.corruptions.has(number)) outcome = 'CORRUPT';
234 const item = {
235 number, kind, value, payload, outcome,
236 arrival: tick + this.delay + (this.extraDelays.get(number) ?? 0)
237 };
238 this.inTransit.push(item);
239 return item;
240 }
241
242 arrivals(tick: number): Transmission[] {
243 const arrived = this.inTransit.filter(item => item.arrival === tick);
244 this.inTransit = this.inTransit.filter(item => item.arrival !== tick);
245 return arrived;
246 }
247
248 countDamage(item: Transmission): void {
249 if (item.outcome === 'LOST') ++this.lost;
250 else ++this.corrupted;
251 }
252}

258 readonly visible: Set<string>;

261 readonly controlChannel: Channel;

263 readonly lines: string[];
264 tick: number;

311 this.ssthresh = config.window;

327 emit(...parts: (string | number)[]): void { this.lines.push([this.tick, ...parts].join(' ')); }
328 trace(side: string, ...parts: (string | number)[]): void { if (this.visible.has(side)) this.emit(...parts); }
329 frameOf(value: number): number { return this.base + mod(value - this.base, this.modulus); }
330 effectiveWindow(): number { return this.config.congestion ? Math.min(this.config.window, this.cwnd) : this.config.window; }
331
332 run(): void {
333 for (let tick = 0; tick <= this.config.limit; ++tick) {
334 this.tick = tick;
335 for (const item of this.dataChannel.arrivals(tick)) {
336 if (item.outcome === 'NORMAL') this.receiveData(item);
337 else this.reportDamage(this.dataChannel, item, 'RECEIVER');
338 }
339 for (const item of this.controlChannel.arrivals(tick)) {
340 if (item.outcome === 'NORMAL') this.receiveControl(item);
341 else this.reportDamage(this.controlChannel, item, 'SENDER');
342 }
343 const completed = this.base === this.config.frames;
344 if (!completed) {
345 this.processTimeouts();
346 this.sendData();
347 }
348 if (this.config.snapshots.has(tick)) this.snapshot();
349 if (completed) {
350 this.completion = tick;
351 this.emit('DONE');
352 return;
353 }
354 }
355 this.emit('ABORT');
356 }
357
358 reportDamage(channel: Channel, item: Transmission, side: string): void {
359 channel.countDamage(item);
360 this.trace(side, item.outcome, item.kind, `#${item.number}`);
361 }
362
363 receiveData(item: Transmission): void {
364 this.trace('RECEIVER', 'RECV', `seq=${item.value}`, `#${item.number}`);
365 if (this.selective) this.selectiveReceive(item.value, item.payload!);
366 else this.goBackNReceive(item.value, item.payload!);
367 }
368
369 goBackNReceive(seq: number, payload: number): void {
370 if (seq === this.expected % this.modulus) {
371 this.deliver(payload);
372 this.sendControl('ack', seq);
373 return;
374 }
375 this.discard(seq);
376 if (this.config.nak && !this.nakSent) {
377 this.sendControl('nak', this.expected % this.modulus);
378 this.nakSent = true;
379 } else if (this.expected > 0) {
380 this.sendControl('ack', (this.expected - 1) % this.modulus);
381 }
382 }
383
384 selectiveReceive(seq: number, payload: number): void {
385 const offset = mod(seq - this.expected, this.modulus);
386 if (offset >= this.config.window) {
387 this.discard(seq);
388 if (mod(this.expected - seq, this.modulus) <= this.config.window) {
389 this.sendControl('ack', seq);
390 }
391 return;
392 }
393 const position = this.expected + offset;
394 if (this.buffer.has(position)) {
395 this.discard(seq);
396 this.sendControl('ack', seq);
397 } else if (offset === 0) {
398 this.deliver(payload);
399 while (this.buffer.has(this.expected)) {
400 const waiting = this.buffer.get(this.expected);
401 this.buffer.delete(this.expected);
402 this.deliver(waiting!);
403 }
404 this.sendControl('ack', seq);
405 } else {
406 this.buffer.set(position, payload);
407 this.maxBuffer = Math.max(this.maxBuffer, this.buffer.size);
408 this.trace('RECEIVER', 'BUFFER', `f=${position}`);
409 this.sendControl('ack', seq);
410 if (this.config.nak && !this.nakSent) {
411 this.sendControl('nak', this.expected % this.modulus);
412 this.nakSent = true;
413 }
414 }
415 }
416
417 deliver(payload: number): void {
418 const position = this.expected;
419 if (payload === position) {
420 ++this.delivered;
421 this.records[payload].delivered = this.tick;
422 this.trace('RECEIVER', 'DELIVER', `f=${position}`);
423 } else {
424 ++this.errors;
425 this.trace('RECEIVER', 'DELIVER', `f=${position}`, `wrong=${payload}`);
426 }
427 ++this.expected;
428 this.nakSent = false;
429 }
430
431 discard(seq: number): void {
432 ++this.discarded;
433 this.trace('RECEIVER', 'DISCARD', `seq=${seq}`);
434 }
435
436 sendControl(kind: ControlKind, value: number): void {
437 const item = this.controlChannel.send(kind, value, null, this.tick, this.generator);
438 if (kind === 'nak') ++this.naks;
439 this.trace('RECEIVER', kind.toUpperCase(), `a=${value}`, `#${item.number}`);
440 }
441
442 receiveControl(item: Transmission): void {
443 const before: [number, number] = [this.cwnd, this.ssthresh];
444 if (item.kind === 'ack') {
445 if (this.selective) this.selectiveAck(item);
446 else this.goBackNAck(item);
447 } else if (this.selective) this.selectiveNak(item);
448 else this.goBackNNak(item);
449 this.reportWindow(before);
450 }
451
452 controlLine(item: Transmission, accepted: boolean): void {
453 const parts = [`${accepted ? 'GOT' : 'OLD'}${item.kind.toUpperCase()}`,
454 `a=${item.value}`, `#${item.number}`];
455 if (accepted) parts.push(`base=${this.base}`);
456 this.trace('SENDER', ...parts);
457 }
458
459 goBackNAck(item: Transmission): void {
460 const frame = this.frameOf(item.value);
461 if (frame < this.framesSent) {
462 this.growWindow(frame - this.base + 1);
463 this.moveBase(frame + 1);
464 if (this.nextFrame < this.base) this.nextFrame = this.base;
465 this.timer = this.base === this.nextFrame ? null : this.tick + this.timerLength;
466 this.controlLine(item, true);
467 return;
468 }
469 this.controlLine(item, false);
470 if (this.config.dupack === null || this.base < 1) return;
471 if (item.value === (this.base - 1) % this.modulus) {
472 ++this.duplicates;
473 if (this.duplicates === this.config.dupack) {
474 ++this.fastRetransmits;
475 this.trace('SENDER', 'FASTRETX', `f=${this.base}`);
476 this.goBack();
477 this.halveWindow();
478 }
479 }
480 }
481
482 goBackNNak(item: Transmission): void {
483 const frame = this.frameOf(item.value);
484 if (frame > this.framesSent) {
485 this.controlLine(item, false);
486 return;
487 }
488 this.growWindow(frame - this.base);
489 this.moveBase(frame);
490 this.goBack();
491 this.halveWindow();
492 this.controlLine(item, true);
493 }
494
495 selectiveAck(item: Transmission): void {
496 const frame = this.frameOf(item.value);
497 if (frame >= this.framesSent || this.acked[frame]) {
498 this.controlLine(item, false);
499 return;
500 }
501 this.acked[frame] = true;
502 this.records[frame].acked = this.tick;
503 this.timers[frame] = null;
504 this.queue = this.queue.filter(waiting => waiting !== frame);
505 this.growWindow(1);
506 const previous = this.base;
507 while (this.base < this.framesSent && this.acked[this.base]) ++this.base;
508 if (this.base > previous) this.timerLength = this.config.timeout;
509 this.controlLine(item, true);
510 }
511
512 selectiveNak(item: Transmission): void {
513 const frame = this.frameOf(item.value);
514 if (frame >= this.framesSent || this.acked[frame]) {
515 this.controlLine(item, false);
516 return;
517 }
518 this.timers[frame] = null;
519 if (!this.queue.includes(frame)) this.queue.push(frame);
520 this.halveWindow();
521 this.controlLine(item, true);
522 }
523
524 moveBase(newBase: number): void {
525 if (newBase <= this.base) return;
526 for (let frame = this.base; frame < newBase; ++frame) this.records[frame].acked = this.tick;
527 this.base = newBase;
528 this.timerLength = this.config.timeout;
529 this.duplicates = 0;
530 }
531
532 goBack(): void { this.nextFrame = this.base; this.timer = null; this.duplicates = 0; }
533
534 growWindow(count: number): void {
535 if (!this.config.congestion) return;
536 for (let i = 0; i < count; ++i) {
537 if (this.cwnd >= this.config.window) continue;
538 if (this.cwnd < this.ssthresh) ++this.cwnd;
539 else {
540 ++this.roundAcks;
541 if (this.roundAcks === this.cwnd) {
542 ++this.cwnd;
543 this.roundAcks = 0;
544 }
545 }
546 }
547 }
548
549 halveWindow(): void {
550 if (!this.config.congestion) return;
551 this.ssthresh = Math.max(Math.floor(this.cwnd / 2), 1);
552 this.cwnd = this.ssthresh;
553 this.roundAcks = 0;
554 }
555
556 collapseWindow(): void {
557 if (!this.config.congestion) return;
558 this.ssthresh = Math.max(Math.floor(this.cwnd / 2), 1);
559 this.cwnd = 1;
560 this.roundAcks = 0;
561 }
562
563 reportWindow(before: [number, number]): void {
564 if (this.cwnd !== before[0] || this.ssthresh !== before[1]) {
565 this.trace('SENDER', 'CWND', this.cwnd, `ssthresh=${this.ssthresh}`);
566 }
567 }
568
569 processTimeouts(): void {
570 if (this.selective) {
571 for (let frame = this.base; frame < this.framesSent; ++frame) {
572 if (this.timers[frame] === this.tick) {
573 this.timers[frame] = null;
574 this.queue.push(frame);
575 this.timeOut(frame);
576 }
577 }
578 } else if (this.timer === this.tick) {
579 this.timeOut(this.base);
580 this.goBack();
581 }
582 }
583
584 timeOut(frame: number): void {
585 const before: [number, number] = [this.cwnd, this.ssthresh];
586 ++this.timeouts;
587 this.trace('SENDER', 'TIMEOUT', `f=${frame}`);
588 if (this.config.backoff) this.timerLength = Math.min(2 * this.timerLength, 8 * this.config.timeout);
589 this.collapseWindow();
590 this.reportWindow(before);
591 }
592
593 maySendNew(frame: number): boolean {
594 return frame < this.config.frames && frame < this.base + this.effectiveWindow() &&
595 frame * this.config.interval <= this.tick;
596 }
597
598 sendData(): void { if (this.selective) this.selectiveSend(); else this.goBackNSend(); }
599
600 goBackNSend(): void {
601 const frame = this.nextFrame;
602 if (frame >= this.config.frames || frame >= this.base + this.effectiveWindow()) return;
603 if (frame >= this.framesSent && frame * this.config.interval > this.tick) return;
604 this.transmit(frame);
605 this.framesSent = Math.max(this.framesSent, frame + 1);
606 ++this.nextFrame;
607 if (this.timer === null) this.timer = this.tick + this.timerLength;
608 }
609
610 selectiveSend(): void {
611 let frame;
612 if (this.queue.length) frame = this.queue.shift();
613 else if (this.maySendNew(this.framesSent)) frame = this.framesSent++;
614 else return;
615 this.transmit(frame!);
616 this.timers[frame!] = this.tick + this.timerLength;
617 }
618
619 transmit(frame: number): void {
620 const record = this.records[frame], seq = frame % this.modulus;
621 const item = this.dataChannel.send('data', seq, frame, this.tick, this.generator);
622 ++record.transmissions;
623 let label;
624 if (record.transmissions === 1) {
625 record.first = this.tick;
626 label = 'SEND';
627 } else {
628 ++this.resent;
629 label = 'RESEND';
630 }
631 this.trace('SENDER', label, `f=${frame}`, `seq=${seq}`, `#${item.number}`);
632 }
633
634 senderMark(frame: number): string {
635 let acknowledged, waiting, sent;
636 if (this.selective) {
637 acknowledged = this.acked[frame];
638 waiting = this.queue.includes(frame);
639 sent = frame < this.framesSent;
640 } else {
641 acknowledged = frame < this.base;
642 waiting = this.nextFrame <= frame && frame < this.framesSent;
643 sent = frame < this.nextFrame;
644 }
645 if (acknowledged) return 'A';
646 if (waiting) return 'R';
647 if (sent) return 'O';
648 if (frame < this.base + this.effectiveWindow()) return 'W';
649 return '.';
650 }
651
652 receiverMark(position: number): string {
653 if (position < this.expected) return 'D';
654 if (this.selective) {
655 if (this.buffer.has(position)) return 'B';
656 if (position < this.expected + this.config.window) return 'W';
657 } else if (position === this.expected) return 'W';
658 return '.';
659 }
660
661 snapshot(): void {
662 const positions = Array.from({length: this.config.frames}, (_, i) => i);
663 const sender = positions.map(frame => this.senderMark(frame)).join('');
664 const receiver = positions.map(position => this.receiverMark(position)).join('');
665 this.emit('SNAPSHOT', `sender=${sender}`, `receiver=${receiver}`);
666 }
667
668 dataCount(): number { return this.dataChannel.sent; }
669 goodputText(): string { return truncated(this.delivered, this.completion!, 2); }
670 efficiencyText(): string { return truncated(100 * this.delivered, this.dataCount(), 1); }
671 latencyText(): string {
672 if (this.delivered === 0) return '-';
673 const total = this.records.reduce((sum, record) =>
674 sum + (record.delivered === null ? 0 : record.delivered - record.first!), 0);
675 return truncated(total, this.delivered, 2);
676 }
677
678 report(): string[] {
679 const lines = this.records.map((record, frame) =>
680 `frame ${frame}: seq=${frame % this.modulus} tx=${record.transmissions} ` +
681 `first=${tickText(record.first)} delivered=${tickText(record.delivered)} ` +
682 `acked=${tickText(record.acked)}`);
683 const data = this.dataChannel, control = this.controlChannel;
684 lines.push(`totals: data=${data.sent} resent=${this.resent} control=${control.sent} ` +
685 `nak=${this.naks} timeouts=${this.timeouts} fastretx=${this.fastRetransmits}`);
686 lines.push(`channel: lost_data=${data.lost} lost_ctrl=${control.lost} ` +
687 `corrupt_data=${data.corrupted} corrupt_ctrl=${control.corrupted}`);
688 lines.push(`receiver: delivered=${this.delivered} errors=${this.errors} ` +
689 `discarded=${this.discarded} max_buffer=${this.maxBuffer}`);
690 const completion = this.completion === null ? '-' : String(this.completion);
691 const goodput = this.completion === null ? '-' : this.goodputText();
692 const utilization = truncated(100 * data.sent, this.tick + 1, 1);
693 lines.push(`result: completion=${completion} goodput=${goodput} ` +
694 `efficiency=${this.efficiencyText()}% utilization=${utilization}% latency=${this.latencyText()}`);
695 return lines;
696 }
697}

699class Scenario {
700 readonly name: string;
701 readonly settings: SettingLine[];
702 config: Config | null = null;
703 simulation: Simulation | null = null;
704
705 constructor(name: string, settings: SettingLine[]) {
706 this.name = name;
707 this.settings = settings;
708
709 }
710
711 completed(): boolean { return this.simulation !== null && this.simulation.completion !== null; }
712 status(): string { return this.simulation === null ? 'invalid' : 'aborted'; }
713
714 execute(output: string[]): void {
715 output.push(`== ${this.name} ==`);
716 try { this.config = checkSettings(this.settings); }
717 catch (error) {
718 if (!(error instanceof SettingsError)) throw error;
719 output.push(`error: ${error.message}`);
720 return;
721 }
722 output.push(this.config.configLine(), this.config.optionsLine());
723 this.simulation = new Simulation(this.config);
724 this.simulation.run();
725 output.push(...this.simulation.lines, ...this.simulation.report());
726 }
727}
728
729function rankingLines(scenarios: Scenario[]): string[] {
730 const lines = ['== ranking =='];
731 const completed = scenarios.filter(scenario => scenario.completed());
732 completed.sort((a, b) => {
733 const x = a.simulation!, y = b.simulation!;
734 const first = y.delivered * x.completion! - x.delivered * y.completion!;
735 if (first !== 0) return first;
736 return y.delivered * x.dataCount() - x.delivered * y.dataCount();
737 });
738 completed.forEach((scenario, i) => {
739 const run = scenario.simulation!;
740 lines.push(`${i + 1}. ${scenario.name} ${scenario.config!.protocol} ` +
741 `goodput=${run.goodputText()} efficiency=${run.efficiencyText()}%`);
742 });
743 for (const scenario of scenarios) {
744 if (!scenario.completed()) lines.push(`- ${scenario.name} ${scenario.status()}`);
745 }
746 return lines;
747}
748
749function protocolLines(scenarios: Scenario[]): string[] {
750 const lines = ['== protocols =='];
751 for (const protocol of PROTOCOLS) {
752 const runs = scenarios.filter(s => s.completed() && s.config!.protocol === protocol)
753 .map(s => s.simulation!);
754 if (!runs.length) continue;
755 const delivered = runs.reduce((sum, r) => sum + r.delivered, 0);
756 const ticks = runs.reduce((sum, r) => sum + r.completion!, 0);
757 const data = runs.reduce((sum, r) => sum + r.dataCount(), 0);
758 lines.push(`${protocol}: runs=${runs.length} goodput=${truncated(delivered, ticks, 2)} ` +
759 `efficiency=${truncated(100 * delivered, data, 1)}%`);
760 }
761 return lines;
762}

764function main(): void {
765 const lines = readFileSync(0, 'utf8').trimEnd().split(/\r?\n/);
766 const count = Number(lines[0]);
767 let index = 1;
768 const scenarios: Scenario[] = [];
769 for (let i = 0; i < count; ++i) {
770 const name = lines[index++].split(' ')[1];
771 const settings: SettingLine[] = [];
772 while (lines[index] !== 'END') {
773 const [key, ...values] = lines[index++].split(' ');
774 settings.push([key, values]);
775 }
776 ++index;
777 scenarios.push(new Scenario(name, settings));
778 }
779 const output: string[] = [];
780 for (const scenario of scenarios) scenario.execute(output);
781 output.push(...rankingLines(scenarios), ...protocolLines(scenarios));
782 process.stdout.write(output.join('\n') + '\n');
783}
