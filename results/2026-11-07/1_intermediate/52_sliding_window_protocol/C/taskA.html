42static int mod(int a, int b) {
43 int r = a % b;
44 return r < 0 ? r + b : r;
45}
46static int minimum(int a, int b) {
47 return a < b ? a : b;

52static void fixed(char *out, int64_t numerator, int64_t denominator, int digits) {
53 int scale = digits == 2 ? 100 : 10;
54 int64_t value = denominator ? numerator * scale / denominator : 0;
55 sprintf(out, "%lld.%0*lld", (long long)(value / scale), digits, (long long)(value % scale));
56}

58 if (tick < 0)
59 return "-";

63static int validate(Config *c, char raw[22][201], int *present, char *error) {
64 for (int i = 0; i < 6; ++i)
65 if (!present[i]) {
66 sprintf(error, "missing key %s", keys[i]);
67 return 0;
68 }
69 for (int i = 0; i < 22; ++i) {
70 if (!present[i])
71 continue;
72 char copy[201];
73 strcpy(copy, raw[i]);
74 int valid = 1;
75 if (i == 0)
76 valid = !strcmp(copy, "SW") || !strcmp(copy, "GBN") || !strcmp(copy, "SR");
77 else if (i >= 9 && i <= 12)
78 valid = !strcmp(copy, "ON") || !strcmp(copy, "OFF");
79 else if (i == 13)
80 valid = !strcmp(copy, "ALL") || !strcmp(copy, "SENDER") || !strcmp(copy, "RECEIVER") ||
81 !strcmp(copy, "NONE");
82 else if (i == 14) {
83 long long seed, loss, corruption;
84 sscanf(copy, "%lld %lld %lld", &seed, &loss, &corruption);
85 valid = seed >= 1 && seed <= 2147483646LL && loss >= 0 && corruption >= 0 &&
86 loss + corruption <= 100;
87 } else if (i >= 15) {
88 unsigned char seen[10000] = {0};
89 for (char *p = strtok(copy, " "); p; p = strtok(NULL, " ")) {
90 int number = atoi(p), lower = i == 21 ? 0 : 1, upper = i == 21 ? 2000 : 9999;
91 if (number < lower || number > upper)
92 valid = 0;
93 else if (seen[number])
94 valid = 0;
95 else
96 seen[number] = 1;
97 if (i == 19 || i == 20) {
98 int extra = atoi(strchr(p, ':') + 1);
99 if (extra < 1 || extra > 50)
100 valid = 0;
101 }
102 }
103 } else {
104 int number = atoi(copy), lower = i == 6 ? 0 : 1;
105 int upper = i == 1 ? 60 : i == 2 ? 32 : i == 3 ? 6 : i == 4 ? 50 : i == 7 ? 2000 : 10;
106 valid = number >= lower && number <= upper;
107 }
108 if (!valid) {
109 sprintf(error, "bad value for %s", keys[i]);
110 return 0;
111 }
112 }
113 strcpy(c->protocol, raw[0]);
114 strcpy(c->trace, present[13] ? raw[13] : "ALL");
115 c->n = atoi(raw[1]);
116 c->w = atoi(raw[2]);
117 c->k = atoi(raw[3]);
118 c->timeout = atoi(raw[4]);
119 c->delay = atoi(raw[5]);
120 c->interval = present[6] ? atoi(raw[6]) : 0;
121 c->limit = present[7] ? atoi(raw[7]) : 1000;
122 c->dupack = present[8] ? atoi(raw[8]) : 0;
123 c->nak = present[9] && !strcmp(raw[9], "ON");
124 c->backoff = present[10] && !strcmp(raw[10], "ON");
125 c->congestion = present[11] && !strcmp(raw[11], "ON");
126 c->unsafe_window = present[12] && !strcmp(raw[12], "ON");
127 if (!strcmp(c->protocol, "SR") && present[8]) {
128 strcpy(error, "DUPACK not supported for SR");
129 return 0;
130 }
131 int limit = !strcmp(c->protocol, "SW") ? 1
132 : !strcmp(c->protocol, "GBN") ? (1 << c->k) - 1
133 : 1 << (c->k - 1);
134 if (c->w > limit && !(c->unsafe_window && strcmp(c->protocol, "SW"))) {
135 sprintf(error, "window %d exceeds limit %d for %s", c->w, limit, c->protocol);
136 return 0;
137 }
138 if (present[14]) {
139 long long seed;
140 sscanf(raw[14], "%lld %d %d", &seed, &c->loss, &c->corruption);
141 c->seed = seed;
142 c->random = 1;
143 }
144 for (int i = 15; i < 22; ++i)
145 if (present[i]) {
146 char copy[201];
147 strcpy(copy, raw[i]);
148 for (char *p = strtok(copy, " "); p; p = strtok(NULL, " ")) {
149 int number = atoi(p);
150 if (i == 15)
151 c->lost_data[number] = 1;
152 if (i == 16)
153 c->lost_control[number] = 1;
154 if (i == 17)
155 c->corrupt_data[number] = 1;
156 if (i == 18)
157 c->corrupt_control[number] = 1;
158 if (i == 19)
159 c->delay_data[number] = atoi(strchr(p, ':') + 1);
160 if (i == 20)
161 c->delay_control[number] = atoi(strchr(p, ':') + 1);
162 if (i == 21)
163 c->snapshots[number] = 1;
164 }
165 }
166 for (int i = 1; i <= 9999; ++i)
167 if (c->lost_data[i] && c->corrupt_data[i]) {
168 sprintf(error, "data #%d both lost and corrupted", i);
169 return 0;
170 }
171 for (int i = 1; i <= 9999; ++i)
172 if (c->lost_control[i] && c->corrupt_control[i]) {
173 sprintf(error, "control #%d both lost and corrupted", i);
174 return 0;
175 }
176 return 1;
177}
178static int visible(const Simulation *s, const char *side) {
179 return !strcmp(s->c->trace, "ALL") || !strcmp(s->c->trace, side) || !strcmp(side, "ALWAYS");
180}
181static int effective(const Simulation *s) {
182 return s->c->congestion ? minimum(s->c->w, s->cwnd) : s->c->w;
183}
184static int outcome(Simulation *s, int data, int number) {
185 int result = 0;
186 Config *c = s->c;
187 if (c->random) {
188 s->random = s->random * 48271 % 2147483647;
189 int value = (int)(s->random % 100);
190 result = value < c->loss ? 1 : value < c->loss + c->corruption ? 2 : 0;
191 }
192 if ((data ? c->lost_data : c->lost_control)[number])
193 result = 1;
194 if ((data ? c->corrupt_data : c->corrupt_control)[number])
195 result = 2;
196 return result;
197}
198static void send_control(Simulation *s, int nak, int value) {
199 ++s->control;
200 if (nak)
201 ++s->naks;
202 if (visible(s, "RECEIVER"))
203 printf("%d %s a=%d #%d\n", s->tick, nak ? "NAK" : "ACK", value, s->control);
204 Event e = {s->tick + s->c->delay + s->c->delay_control[s->control], s->control, value,
205 outcome(s, 0, s->control), nak};
206 s->control_events[s->control_count++] = e;
207}
208static void deliver(Simulation *s, int payload) {
209 if (payload == s->receiver) {
210 if (visible(s, "RECEIVER"))
211 printf("%d DELIVER f=%d\n", s->tick, s->receiver);
212 ++s->delivered;
213 if (s->delivered_at[payload] < 0)
214 s->delivered_at[payload] = s->tick;
215 s->latency += s->tick - s->first[payload];
216 } else {
217 if (visible(s, "RECEIVER"))
218 printf("%d DELIVER f=%d wrong=%d\n", s->tick, s->receiver, payload);
219 ++s->errors;
220 }
221 ++s->receiver;
222 s->nak_sent = 0;
223}
224static void discard(Simulation *s, int sequence) {
225 ++s->discarded;
226 if (visible(s, "RECEIVER"))
227 printf("%d DISCARD seq=%d\n", s->tick, sequence);
228}
229static int buffered(const Simulation *s, int position) {
230 for (int i = 0; i < s->buffered; ++i)
231 if (s->positions[i] == position)
232 return i;
233 return -1;
234}
235static void receive(Simulation *s, Event e) {
236 if (e.outcome) {
237 if (e.outcome == 1)
238 ++s->lost_data;
239 else
240 ++s->corrupt_data;
241 if (visible(s, "RECEIVER"))
242 printf("%d %s data #%d\n", s->tick, e.outcome == 1 ? "LOST" : "CORRUPT", e.number);
243 return;
244 }
245 int sequence = e.value % s->modulus;
246 if (visible(s, "RECEIVER"))
247 printf("%d RECV seq=%d #%d\n", s->tick, sequence, e.number);
248 if (!s->sr) {
249 if (sequence == s->receiver % s->modulus) {
250 deliver(s, e.value);
251 send_control(s, 0, sequence);
252 } else {
253 discard(s, sequence);
254 if (s->c->nak && !s->nak_sent) {
255 send_control(s, 1, s->receiver % s->modulus);
256 s->nak_sent = 1;
257 } else if (s->receiver > 0)
258 send_control(s, 0, (s->receiver - 1) % s->modulus);
259 }
260 return;
261 }
262 int offset = mod(sequence - s->receiver, s->modulus), position = s->receiver + offset;
263 if (offset < s->c->w) {
264 if (buffered(s, position) >= 0) {
265 discard(s, sequence);
266 send_control(s, 0, sequence);
267 } else if (offset == 0) {
268 deliver(s, e.value);
269 int i;
270 while ((i = buffered(s, s->receiver)) >= 0) {
271 int payload = s->payloads[i];
272 --s->buffered;
273 s->positions[i] = s->positions[s->buffered];
274 s->payloads[i] = s->payloads[s->buffered];
275 deliver(s, payload);
276 }
277 send_control(s, 0, sequence);
278 } else {
279 s->positions[s->buffered] = position;
280 s->payloads[s->buffered++] = e.value;
281 s->max_buffer = maximum(s->max_buffer, s->buffered);
282 if (visible(s, "RECEIVER"))
283 printf("%d BUFFER f=%d\n", s->tick, position);
284 send_control(s, 0, sequence);
285 if (s->c->nak && !s->nak_sent) {
286 send_control(s, 1, s->receiver % s->modulus);
287 s->nak_sent = 1;
288 }
289 }
290 } else {
291 discard(s, sequence);
292 if (mod(s->receiver - sequence, s->modulus) <= s->c->w)
293 send_control(s, 0, sequence);
294 }
295}
296static int queued(const Simulation *s, int frame) {
297 for (int i = 0; i < s->queued; ++i)
298 if (s->queue[i] == frame)
299 return i;
300 return -1;
301}
302static void dequeue(Simulation *s, int index) {
303 --s->queued;
304 memmove(s->queue + index, s->queue + index + 1, (size_t)(s->queued - index) * sizeof(int));
305}
306static void go_back(Simulation *s) {
307 s->next = s->base;
308 s->timer = -1;
309 s->duplicates = 0;
310}
311static void increase(Simulation *s, int count) {
312 if (!s->c->congestion)
313 return;
314 for (int i = 0; i < count && s->cwnd < s->c->w; ++i) {
315 if (s->cwnd < s->threshold)
316 ++s->cwnd;
317 else if (++s->credit == s->cwnd) {
318 ++s->cwnd;
319 s->credit = 0;
320 }
321 }
322}
323static void decrease(Simulation *s, int timeout) {
324 if (!s->c->congestion)
325 return;
326 s->threshold = maximum(s->cwnd / 2, 1);
327 s->cwnd = timeout ? 1 : s->threshold;
328 s->credit = 0;
329}
330static void cwnd_line(Simulation *s, int old_window, int old_threshold) {
331 if (s->c->congestion && (old_window != s->cwnd || old_threshold != s->threshold) &&
332 visible(s, "SENDER"))
333 printf("%d CWND %d ssthresh=%d\n", s->tick, s->cwnd, s->threshold);
334}
335static void control(Simulation *s, Event e) {
336 const char *kind = e.nak ? "NAK" : "ACK";
337 if (e.outcome) {
338 if (e.outcome == 1)
339 ++s->lost_control;
340 else
341 ++s->corrupt_control;
342 if (visible(s, "SENDER"))
343 printf("%d %s %s #%d\n", s->tick, e.outcome == 1 ? "LOST" : "CORRUPT",
344 e.nak ? "nak" : "ack", e.number);
345 return;
346 }
347 int old_window = s->cwnd, old_threshold = s->threshold, old_base = s->base;
348 int frame = s->base + mod(e.value - s->base, s->modulus);
349 int accepted = s->sr ? frame < s->high && !s->acknowledged[frame]
350 : e.nak ? frame <= s->high
351 : frame < s->high;
352 if (accepted) {
353 if (s->sr) {
354 s->timers[frame] = -1;
355 int at = queued(s, frame);
356 if (e.nak) {
357 if (at < 0)
358 s->queue[s->queued++] = frame;
359 } else {
360 s->acknowledged[frame] = 1;
361 if (at >= 0)
362 dequeue(s, at);
363 if (s->acked_at[frame] < 0)
364 s->acked_at[frame] = s->tick;
365 while (s->base < s->high && s->acknowledged[s->base])
366 ++s->base;
367 increase(s, 1);
368 }
369 } else {
370 s->base = frame + !e.nak;
371 increase(s, s->base - old_base);
372 for (int f = old_base; f < s->base; ++f)
373 if (s->acked_at[f] < 0)
374 s->acked_at[f] = s->tick;
375 }
376 if (s->base > old_base) {
377 s->duration = s->c->timeout;
378 s->duplicates = 0;
379 }
380 if (!s->sr) {
381 if (e.nak)
382 go_back(s);
383 else {
384 s->next = maximum(s->next, s->base);
385 s->timer = s->base == s->next ? -1 : s->tick + s->duration;
386 }
387 }
388 if (e.nak)
389 decrease(s, 0);
390 if (visible(s, "SENDER"))
391 printf("%d GOT%s a=%d #%d base=%d\n", s->tick, kind, e.value, e.number, s->base);
392 } else {
393 if (visible(s, "SENDER"))
394 printf("%d OLD%s a=%d #%d\n", s->tick, kind, e.value, e.number);
395 if (!s->sr && !e.nak && s->c->dupack && s->base >= 1 &&
396 e.value == (s->base - 1) % s->modulus) {
397 if (++s->duplicates == s->c->dupack) {
398 ++s->fast;
399 if (visible(s, "SENDER"))
400 printf("%d FASTRETX f=%d\n", s->tick, s->base);
401 go_back(s);
402 decrease(s, 0);
403 }
404 }
405 }
406 cwnd_line(s, old_window, old_threshold);
407}
408static void timeout(Simulation *s, int frame) {
409 int old_window = s->cwnd, old_threshold = s->threshold;
410 ++s->timeouts;
411 if (visible(s, "SENDER"))
412 printf("%d TIMEOUT f=%d\n", s->tick, frame);
413 if (s->sr) {
414 s->timers[frame] = -1;
415 if (queued(s, frame) < 0)
416 s->queue[s->queued++] = frame;
417 } else
418 go_back(s);
419 if (s->c->backoff)
420 s->duration = minimum(s->duration * 2, s->c->timeout * 8);
421 decrease(s, 1);
422 cwnd_line(s, old_window, old_threshold);
423}
424static void send_data(Simulation *s, int frame) {
425 ++s->data;
426 int retry = s->tx[frame] > 0;
427 if (retry)
428 ++s->resent;
429 else
430 s->first[frame] = s->tick;
431 ++s->tx[frame];
432 if (visible(s, "SENDER"))
433 printf("%d %s f=%d seq=%d #%d\n", s->tick, retry ? "RESEND" : "SEND", frame,
434 frame % s->modulus, s->data);
435 Event e = {s->tick + s->c->delay + s->c->delay_data[s->data], s->data, frame,
436 outcome(s, 1, s->data), 0};
437 s->data_events[s->data_count++] = e;
438 if (s->sr)
439 s->timers[frame] = s->tick + s->duration;
440 else if (s->timer < 0)
441 s->timer = s->tick + s->duration;
442}
443static void snapshot(Simulation *s) {
444 char sender[61], receiver[61];
445 for (int f = 0; f < s->c->n; ++f) {
446 sender[f] = (s->sr ? s->acknowledged[f] : f < s->base) ? 'A'
447 : (s->sr ? queued(s, f) >= 0 : s->next <= f && f < s->high) ? 'R'
448 : (s->sr ? f < s->high : f < s->next) ? 'O'
449 : f < s->base + effective(s) ? 'W'
450 : '.';
451 receiver[f] = f < s->receiver ? 'D'
452 : s->sr && buffered(s, f) >= 0 ? 'B'
453 : (s->sr ? f < s->receiver + s->c->w : f == s->receiver) ? 'W'
454 : '.';
455 }
456 sender[s->c->n] = receiver[s->c->n] = 0;
457 printf("%d SNAPSHOT sender=%s receiver=%s\n", s->tick, sender, receiver);
458}
459static Result simulate(Config *c, int order) {
460 Simulation *s = calloc(1, sizeof(Simulation));
461 s->c = c;
462 s->sr = !strcmp(c->protocol, "SR");
463 s->modulus = 1 << c->k;
464 s->duration = c->timeout;
465 s->timer = -1;
466 s->cwnd = 1;
467 s->threshold = c->w;
468 s->random = c->seed;
469 for (int f = 0; f < c->n; ++f)
470 s->timers[f] = s->first[f] = s->delivered_at[f] = s->acked_at[f] = -1;
471 printf("config: %s n=%d w=%d k=%d T=%d d=%d\n", c->protocol, c->n, c->w, c->k, c->timeout,
472 c->delay);
473 char dupack[20], random[64], trace[9];
474 if (c->dupack)
475 sprintf(dupack, "%d", c->dupack);
476 else
477 strcpy(dupack, "off");
478 if (c->random)
479 sprintf(random, "%lld/%d/%d", (long long)c->seed, c->loss, c->corruption);
480 else
481 strcpy(random, "off");
482 strcpy(trace, c->trace);
483 for (int i = 0; trace[i]; ++i)
484 if (trace[i] >= 'A' && trace[i] <= 'Z')
485 trace[i] += 'a' - 'A';
486 printf(
487 "options: interval=%d limit=%d dupack=%s nak=%s backoff=%s congestion=%s unsafe=%s trace=%s random=%s\n",
488 c->interval, c->limit, dupack, c->nak ? "on" : "off", c->backoff ? "on" : "off",
489 c->congestion ? "on" : "off", c->unsafe_window ? "on" : "off", trace, random);
490 int complete = 0;
491 for (s->tick = 0; s->tick <= c->limit; ++s->tick) {
492 for (int i = 0; i < s->data_count; ++i)
493 if (s->data_events[i].arrival == s->tick)
494 receive(s, s->data_events[i]);
495 for (int i = 0; i < s->control_count; ++i)
496 if (s->control_events[i].arrival == s->tick)
497 control(s, s->control_events[i]);
498 complete = s->base == c->n;
499 if (!complete) {
500 if (s->sr) {
501 for (int f = s->base; f < s->high; ++f)
502 if (s->timers[f] == s->tick)
503 timeout(s, f);
504 } else if (s->timer == s->tick)
505 timeout(s, s->base);
506 if (s->sr) {
507 if (s->queued) {
508 int frame = s->queue[0];
509 dequeue(s, 0);
510 send_data(s, frame);
511 } else if (s->high < c->n && s->high < s->base + effective(s) &&
512 s->high * c->interval <= s->tick) {
513 int frame = s->high++;
514 send_data(s, frame);
515 }
516 } else if (s->next < c->n && s->next < s->base + effective(s) &&
517 (s->next < s->high || s->next * c->interval <= s->tick)) {
518 int frame = s->next++;
519 s->high = maximum(s->high, s->next);
520 send_data(s, frame);
521 }
522 }
523 if (c->snapshots[s->tick])
524 snapshot(s);
525 if (complete || s->tick == c->limit) {
526 printf("%d %s\n", s->tick, complete ? "DONE" : "ABORT");
527 break;
528 }
529 }
530 for (int f = 0; f < c->n; ++f) {
531 char a[20], b[20], d[20];
532 printf("frame %d: seq=%d tx=%d first=%s delivered=%s acked=%s\n", f, f % s->modulus,
533 s->tx[f], stamp(s->first[f], a), stamp(s->delivered_at[f], b),
534 stamp(s->acked_at[f], d));
535 }
536 printf("totals: data=%d resent=%d control=%d nak=%d timeouts=%d fastretx=%d\n", s->data,
537 s->resent, s->control, s->naks, s->timeouts, s->fast);
538 printf("channel: lost_data=%d lost_ctrl=%d corrupt_data=%d corrupt_ctrl=%d\n", s->lost_data,
539 s->lost_control, s->corrupt_data, s->corrupt_control);
540 printf("receiver: delivered=%d errors=%d discarded=%d max_buffer=%d\n", s->delivered, s->errors,
541 s->discarded, s->max_buffer);
542 char completion[30], goodput[40], efficiency[40], utilization[40], latency[40];
543 if (complete) {
544 sprintf(completion, "%d", s->tick);
545 fixed(goodput, s->delivered, s->tick, 2);
546 } else {
547 strcpy(completion, "-");
548 strcpy(goodput, "-");
549 }
550 fixed(efficiency, s->delivered * 100LL, s->data, 1);
551 fixed(utilization, s->data * 100LL, s->tick + 1, 1);
552 if (s->delivered)
553 fixed(latency, s->latency, s->delivered, 2);
554 else
555 strcpy(latency, "-");
556 printf("result: completion=%s goodput=%s efficiency=%s%% utilization=%s%% latency=%s\n",
557 completion, goodput, efficiency, utilization, latency);
558 Result result = {0};
559 strcpy(result.name, c->name);
560 strcpy(result.protocol, c->protocol);
561 strcpy(result.status, complete ? "complete" : "aborted");
562 result.delivered = s->delivered;
563 result.data = s->data;
564 result.completion = s->tick;
565 result.order = order;
566 free(s);
567 return result;
568}
569static int compare(const Result *a, const Result *b) {
570 int64_t left = a->delivered * b->completion, right = b->delivered * a->completion;
571 if (left != right)
572 return left > right ? -1 : 1;
573 left = a->delivered * b->data;
574 right = b->delivered * a->data;
575 if (left != right)
576 return left > right ? -1 : 1;
577 return a->order - b->order;
578}
