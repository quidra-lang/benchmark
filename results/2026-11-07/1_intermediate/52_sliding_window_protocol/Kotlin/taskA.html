31fun fixed(n: Long, d: Long, digits: Int): String {
32 val scale = if (digits == 2) 100L else 10L
33 val v = if (d == 0L) 0L else n * scale / d
34 return "${v / scale}." + (v % scale).toString().padStart(digits, '0')
35}

41class Scenario(val name: String) {
42 var protocol = ""
43 var error: String? = null
44 var trace = "ALL"
45 val values = mutableMapOf<String, List<String>>()
46 val lists = mutableMapOf<String, Set<Int>>()
47 val delays = mutableMapOf<String, Map<Int, Int>>()
48 var n = 0
49 var w = 0
50 var k = 0
51 var timeout = 0
52 var delay = 0
53 var interval = 0
54 var limit = 1000
55 var dupack = 0
56 var nak = false
57 var backoff = false
58 var congestion = false
59 var unsafe = false
60 var random = false
61 var seed = 0L
62 var loss = 0
63 var corruption = 0
64 var completion = -1
65 var end = 0
66 var delivered = 0
67 var data = 0
68
69 fun number(key: String) = values.getValue(key)[0].toInt()
70
71 fun on(key: String) = values[key]?.get(0) == "ON"
72
73 fun line(words: List<String>) {
74 if (error != null) return
75 val key = words[0]
76 val pos = keys.indexOf(key)
77 if (pos < 0) {
78 error = "unknown key $key"
79 return
80 }
81 if (key in values) {
82 error = "duplicate key $key"
83 return
84 }
85 val count = words.size - 1
86 val bad = if (key == "RANDOM") count != 3 else if (pos >= 15) count < 1 else count != 1
87 if (bad) {
88 error = "wrong number of values for $key"
89 return
90 }
91 values[key] = words.drop(1)
92 }
93
94 fun valid(key: String): Boolean {
95 val v = values.getValue(key)
96 when (key) {
97 "PROTOCOL" -> return v[0] in listOf("SW", "GBN", "SR")
98 "NAK",
99 "BACKOFF",
100 "CONGESTION",
101 "UNSAFE" -> return v[0] in listOf("ON", "OFF")
102 "TRACE" -> return v[0] in listOf("ALL", "SENDER", "RECEIVER", "NONE")
103 "RANDOM" -> {
104 val s = v[0].toLong()
105 val p = v[1].toLong()
106 val c = v[2].toLong()
107 return s in 1..2147483646L && p >= 0 && c >= 0 && p + c <= 100
108 }
109 }
110 if (keys.indexOf(key) >= 15) {
111 val found = mutableSetOf<Int>()
112 val extra = mutableMapOf<Int, Int>()
113 val delayed = key.startsWith("DELAY")
114 for (item in v) {
115 val pair = item.split(':')
116 val a = pair[0].toInt()
117 val lo = if (key == "SNAPSHOT") 0 else 1
118 val hi = if (key == "SNAPSHOT") 2000 else 9999
119 if (a !in lo..hi || !found.add(a)) return false
120 if (delayed) {
121 val e = pair[1].toInt()
122 if (e !in 1..50) return false
123 extra[a] = e
124 }
125 }
126 lists[key] = found
127 if (delayed) delays[key] = extra
128 return true
129 }
130 val range =
131 when (key) {
132 "FRAMES" -> 1..60
133 "WINDOW" -> 1..32
134 "SEQBITS" -> 1..6
135 "TIMEOUT" -> 1..50
136 "DELAY",
137 "DUPACK" -> 1..10
138 "INTERVAL" -> 0..10
139 else -> 1..2000
140 }
141 return number(key) in range
142 }
143
144 fun check() {
145 if (error != null) return
146 for (key in keys.take(6)) if (key !in values) {
147 error = "missing key $key"
148 return
149 }
150 for (key in keys) if (key in values && !valid(key)) {
151 error = "bad value for $key"
152 return
153 }
154 protocol = values.getValue("PROTOCOL")[0]
155 n = number("FRAMES")
156 w = number("WINDOW")
157 k = number("SEQBITS")
158 timeout = number("TIMEOUT")
159 delay = number("DELAY")
160 if ("INTERVAL" in values) interval = number("INTERVAL")
161 if ("MAXTICKS" in values) limit = number("MAXTICKS")
162 if ("DUPACK" in values) dupack = number("DUPACK")
163 nak = on("NAK")
164 backoff = on("BACKOFF")
165 congestion = on("CONGESTION")
166 unsafe = on("UNSAFE")
167 if ("TRACE" in values) trace = values.getValue("TRACE")[0]
168 random = "RANDOM" in values
169 if (random) {
170 val v = values.getValue("RANDOM")
171 seed = v[0].toLong()
172 loss = v[1].toInt()
173 corruption = v[2].toInt()
174 }
175 if (protocol == "SR" && dupack > 0) {
176 error = "DUPACK not supported for SR"
177 return
178 }
179 val upper =
180 when (protocol) {
181 "SW" -> 1
182 "GBN" -> (1 shl k) - 1
183 else -> 1 shl (k - 1)
184 }
185 if (w > upper && !(unsafe && protocol != "SW")) {
186 error = "window $w exceeds limit $upper for $protocol"
187 return
188 }
189 for (kind in listOf("DATA", "CTRL")) {
190 val both = lists["LOSE$kind"].orEmpty().intersect(lists["CORRUPT$kind"].orEmpty())
191 if (both.isNotEmpty()) {
192 error =
193 (if (kind == "DATA") "data" else "control") +
194 " #${both.min()} both lost and corrupted"
195 return
196 }
197 }
198 }
199
200 fun contains(key: String, i: Int) = i in lists[key].orEmpty()
201
202 fun extra(key: String, i: Int) = delays[key]?.get(i) ?: 0
203}

213class Simulation(val s: Scenario) {
214 var t = 0
215 val modulus = 1 shl s.k
216 val sr = s.protocol == "SR"
217 var base = 0
218 var high = 0
219 var next = 0
220 var timer = -1
221 var length = s.timeout
222 var duplicates = 0
223 var r = 0
224 var nakSent = false
225 var cwnd = 1
226 var threshold = s.w
227 var growth = 0
228 var controlCount = 0
229 var naks = 0
230 var resent = 0
231 var timeouts = 0
232 var fast = 0
233 var lostData = 0
234 var lostControl = 0
235 var corruptData = 0
236 var corruptControl = 0
237 var errors = 0
238 var discarded = 0
239 var maxBuffer = 0
240 var latency = 0
241 var random = s.seed
242 val tx = IntArray(s.n)
243 val first = IntArray(s.n) { -1 }
244 val delivery = IntArray(s.n) { -1 }
245 val acked = IntArray(s.n) { -1 }
246 val timers = IntArray(s.n) { -1 }
247 val marks = BooleanArray(s.n)
248 val queue = mutableListOf<Int>()
249 val buffer = mutableMapOf<Int, Int>()
250 val dataEvents = mutableListOf<Event>()
251 val controlEvents = mutableListOf<Event>()
252
253 fun mod(a: Int) = Math.floorMod(a, modulus)
254
255 fun emit(side: String, text: String) {
256 if (side == "ALWAYS" || s.trace == "ALL" || s.trace == side) output.add("$t $text")
257 }
258
259 fun effective() = if (s.congestion) min(s.w, cwnd) else s.w
260
261 fun outcome(kind: String, number: Int): Int {
262 var result = 0
263 if (s.random) {
264 random = random * 48271L % 2147483647L
265 val v = (random % 100).toInt()
266 result = if (v < s.loss) 1 else if (v < s.loss + s.corruption) 2 else 0
267 }
268 val suffix = if (kind == "data") "DATA" else "CTRL"
269 if (s.contains("LOSE$suffix", number)) result = 1
270 if (s.contains("CORRUPT$suffix", number)) result = 2
271 return result
272 }
273
274 fun control(nak: Boolean, a: Int) {
275 val id = ++controlCount
276 if (nak) naks++
277 val kind = if (nak) "nak" else "ack"
278 emit("RECEIVER", "${kind.uppercase()} a=$a #$id")
279 controlEvents.add(
280 Event(t + s.delay + s.extra("DELAYCTRL", id), id, a, outcome(kind, id), kind),
281 )
282 }
283
284 fun deliver(payload: Int) {
285 if (payload == r) {
286 emit("RECEIVER", "DELIVER f=$r")
287 if (delivery[payload] < 0) delivery[payload] = t
288 s.delivered++
289 latency += t - first[payload]
290 } else {
291 emit("RECEIVER", "DELIVER f=$r wrong=$payload")
292 errors++
293 }
294 r++
295 nakSent = false
296 }
297
298 fun discard(seq: Int) {
299 discarded++
300 emit("RECEIVER", "DISCARD seq=$seq")
301 }
302
303 fun receive(e: Event) {
304 val seq = mod(e.value)
305 emit("RECEIVER", "RECV seq=$seq #${e.number}")
306 if (!sr) {
307 if (seq == mod(r)) {
308 deliver(e.value)
309 control(false, seq)
310 } else {
311 discard(seq)
312 if (s.nak && !nakSent) {
313 control(true, mod(r))
314 nakSent = true
315 } else if (r > 0) control(false, mod(r - 1))
316 }
317 } else {
318 val offset = mod(seq - r)
319 if (offset < s.w) {
320 val position = r + offset
321 if (position in buffer) {
322 discard(seq)
323 control(false, seq)
324 } else if (offset == 0) {
325 deliver(e.value)
326 while (r in buffer) deliver(buffer.remove(r)!!)
327 control(false, seq)
328 } else {
329 buffer[position] = e.value
330 maxBuffer = max(maxBuffer, buffer.size)
331 emit("RECEIVER", "BUFFER f=$position")
332 control(false, seq)
333 if (s.nak && !nakSent) {
334 control(true, mod(r))
335 nakSent = true
336 }
337 }
338 } else {
339 discard(seq)
340 if (mod(r - seq) <= s.w) control(false, seq)
341 }
342 }
343 }
344
345 fun goBack() {
346 next = base
347 timer = -1
348 duplicates = 0
349 }
350
351 fun increase(count: Int) {
352 if (!s.congestion) return
353 repeat(count) {
354 if (cwnd < s.w) {
355 if (cwnd < threshold) cwnd++
356 else {
357 growth++
358 if (growth == cwnd) {
359 cwnd++
360 growth = 0
361 }
362 }
363 }
364 }
365 }
366
367 fun decrease(timeout: Boolean) {
368 if (!s.congestion) return
369 threshold = max(cwnd / 2, 1)
370 cwnd = if (timeout) 1 else threshold
371 growth = 0
372 }
373
374 fun cwndLine(c: Int, h: Int) {
375 if (s.congestion && (c != cwnd || h != threshold))
376 emit("SENDER", "CWND $cwnd ssthresh=$threshold")
377 }
378
379 fun acknowledge(e: Event) {
380 val oldC = cwnd
381 val oldS = threshold
382 val oldBase = base
383 val a = e.value
384 val f = base + mod(a - base)
385 val nak = e.kind == "nak"
386 var accepted = false
387 if (sr) {
388 if (f < high && !marks[f]) {
389 accepted = true
390 timers[f] = -1
391 if (nak) {
392 if (f !in queue) queue.add(f)
393 } else {
394 marks[f] = true
395 if (acked[f] < 0) acked[f] = t
396 queue.remove(f)
397 while (base < high && marks[base]) base++
398 increase(1)
399 }
400 }
401 } else if (if (nak) f <= high else f < high) {
402 accepted = true
403 base = if (nak) f else f + 1
404 increase(base - oldBase)
405 for (i in oldBase until base) if (acked[i] < 0) acked[i] = t
406 if (nak) goBack() else if (next < base) next = base
407 }
408 if (base > oldBase) {
409 length = s.timeout
410 duplicates = 0
411 }
412 if (accepted && !sr && !nak) timer = if (base == next) -1 else t + length
413 val label = if (nak) "NAK" else "ACK"
414 emit(
415 "SENDER",
416 (if (accepted) "GOT" else "OLD") +
417 "$label a=$a #${e.number}" +
418 if (accepted) " base=$base" else "",
419 )
420 if (accepted && nak) decrease(false)
421 if (!accepted && !nak && !sr && s.dupack > 0 && base >= 1 && a == mod(base - 1)) {
422 duplicates++
423 if (duplicates == s.dupack) {
424 fast++
425 emit("SENDER", "FASTRETX f=$base")
426 goBack()
427 decrease(false)
428 }
429 }
430 cwndLine(oldC, oldS)
431 }
432
433 fun timeout(f: Int) {
434 val oldC = cwnd
435 val oldS = threshold
436 timeouts++
437 emit("SENDER", "TIMEOUT f=$f")
438 if (sr) {
439 timers[f] = -1
440 queue.add(f)
441 } else goBack()
442 if (s.backoff) length = min(length * 2, 8 * s.timeout)
443 decrease(true)
444 cwndLine(oldC, oldS)
445 }
446
447 fun send(f: Int) {
448 val id = ++s.data
449 val again = tx[f]++ > 0
450 if (again) resent++ else first[f] = t
451 emit("SENDER", (if (again) "RESEND" else "SEND") + " f=$f seq=${mod(f)} #$id")
452 dataEvents.add(
453 Event(t + s.delay + s.extra("DELAYDATA", id), id, f, outcome("data", id), "data"),
454 )
455 }
456
457 fun arrivals(events: MutableList<Event>, data: Boolean) {
458 val ready = events.filter { it.arrival == t }.sortedBy { it.number }
459 events.removeAll(ready.toSet())
460 for (e in ready) {
461 if (e.outcome != 0) {
462 val lost = e.outcome == 1
463 if (data) {
464 if (lost) lostData++ else corruptData++
465 } else {
466 if (lost) lostControl++ else corruptControl++
467 }
468 emit(
469 if (data) "RECEIVER" else "SENDER",
470 (if (lost) "LOST" else "CORRUPT") + " ${e.kind} #${e.number}",
471 )
472 } else if (data) receive(e) else acknowledge(e)
473 }
474 }

476 fun snapshot() {
477 val x = StringBuilder()
478 val y = StringBuilder()
479 for (f in 0 until s.n) {
480 val ack = if (sr) marks[f] else f < base
481 val retransmit = if (sr) f in queue else next <= f && f < high
482 val sent = if (sr) f < high else f < next
483 x.append(
484 if (ack) 'A'
485 else if (retransmit) 'R'
486 else if (sent) 'O' else if (f < base + effective()) 'W' else '.',
487 )
488 y.append(
489 if (f < r) 'D'
490 else if (sr && f in buffer) 'B'
491 else if (if (sr) f < r + s.w else f == r) 'W' else '.',
492 )
493 }
494 emit("ALWAYS", "SNAPSHOT sender=$x receiver=$y")
495 }
496
497 fun run() {
498 for (time in 0..s.limit) {
499 t = time
500 arrivals(dataEvents, true)
501 arrivals(controlEvents, false)
502 val done = base == s.n
503 if (!done) {
504 if (sr) {
505 for (f in base until high) if (timers[f] == t) timeout(f)
506 } else if (timer == t) timeout(base)
507 if (sr) {
508 var f = -1
509 if (queue.isNotEmpty()) f = queue.removeAt(0)
510 else if (high < s.n && high < base + effective() && high * s.interval <= t)
511 f = high++
512 if (f >= 0) {
513 send(f)
514 timers[f] = t + length
515 }
516 } else if (
517 next < s.n &&
518 next < base + effective() &&
519 (next < high || next * s.interval <= t)
520 ) {
521 send(next)
522 next++
523 high = max(high, next)
524 if (timer < 0) timer = t + length
525 }
526 }
527 if (s.contains("SNAPSHOT", t)) snapshot()
528 if (done || t == s.limit) {
529 s.end = t
530 if (done) s.completion = t
531 emit("ALWAYS", if (done) "DONE" else "ABORT")
532 break
533 }
534 }
535 for (f in 0 until s.n) output.add(
536 "frame $f: seq=${mod(f)} tx=${tx[f]} first=${tick(first[f])} delivered=${tick(delivery[f])} acked=${tick(acked[f])}",
537 )
538 output.add(
539 "totals: data=${s.data} resent=$resent control=$controlCount nak=$naks timeouts=$timeouts fastretx=$fast",
540 )
541 output.add(
542 "channel: lost_data=$lostData lost_ctrl=$lostControl corrupt_data=$corruptData corrupt_ctrl=$corruptControl",
543 )
544 output.add(
545 "receiver: delivered=${s.delivered} errors=$errors discarded=$discarded max_buffer=$maxBuffer",
546 )
547 val goodput =
548 if (s.completion < 0) "-" else fixed(s.delivered.toLong(), s.completion.toLong(), 2)
549 val latencyText =
550 if (s.delivered == 0) "-" else fixed(latency.toLong(), s.delivered.toLong(), 2)
551 output.add(
552 "result: completion=${tick(s.completion)} goodput=$goodput efficiency=${fixed(s.delivered * 100L, s.data.toLong(), 1)}% utilization=${fixed(s.data * 100L, (s.end + 1).toLong(), 1)}% latency=$latencyText",
553 )
554 }
555}

557fun main() {
558 val input = System.`in`.bufferedReader()
559 val scenarios = mutableListOf<Scenario>()
560 repeat(input.readLine().toInt()) {
561 val s = Scenario(input.readLine().split(' ')[1])
562 while (true) {
563 val line = input.readLine()
564 if (line == "END") break
565 s.line(line.split(' '))
566 }
567 s.check()
568 scenarios.add(s)
569 output.add("== ${s.name} ==")
570 if (s.error != null) output.add("error: ${s.error}")
571 else {
572 output.add(
573 "config: ${s.protocol} n=${s.n} w=${s.w} k=${s.k} T=${s.timeout} d=${s.delay}",
574 )
575 val random = if (s.random) "${s.seed}/${s.loss}/${s.corruption}" else "off"
576 output.add(
577 "options: interval=${s.interval} limit=${s.limit} dupack=${if (s.dupack == 0) "off" else s.dupack} nak=${onOff(s.nak)} backoff=${onOff(s.backoff)} congestion=${onOff(s.congestion)} unsafe=${onOff(s.unsafe)} trace=${s.trace.lowercase()} random=$random",
578 )
579 Simulation(s).run()
580 }
581 }
582 output.add("== ranking ==")
583 val completed =
584 scenarios
585 .filter { it.completion >= 0 }
586 .sortedWith { a, b ->
587 val c =
588 (b.delivered.toLong() * a.completion).compareTo(
589 a.delivered.toLong() * b.completion,
590 )
591 if (c != 0) c
592 else (b.delivered.toLong() * a.data).compareTo(a.delivered.toLong() * b.data)
593 }
594 for ((i, s) in completed.withIndex()) output.add(
595 "${i + 1}. ${s.name} ${s.protocol} goodput=${fixed(s.delivered.toLong(), s.completion.toLong(), 2)} efficiency=${fixed(s.delivered * 100L, s.data.toLong(), 1)}%",
596 )
597 for (s in scenarios) if (s.completion < 0)
598 output.add("- ${s.name} " + if (s.error != null) "invalid" else "aborted")
599 output.add("== protocols ==")
600 for (p in listOf("SW", "GBN", "SR")) {
601 val runs = completed.filter { it.protocol == p }
602 if (runs.isNotEmpty())
603 output.add(
604 "$p: runs=${runs.size} goodput=${fixed(runs.sumOf { it.delivered.toLong() }, runs.sumOf { it.completion.toLong() }, 2)} efficiency=${fixed(runs.sumOf { it.delivered * 100L }, runs.sumOf { it.data.toLong() }, 1)}%",
605 )
606 }
607 print(output.joinToString("\n", postfix = "\n"))
608}
