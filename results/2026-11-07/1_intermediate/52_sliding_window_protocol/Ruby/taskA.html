60def valid_value?(key, values)
61 return CHOICES[key].include?(values[0]) if CHOICES.key?(key)
62 return RANGES[key].cover?(values[0].to_i) if !LIST_KEYS.include?(key) && RANGES.key?(key)
63 if key == 'RANDOM'
64 seed, loss, corrupt = values.map(&:to_i)
65 return (1...RANDOM_MODULUS).cover?(seed) && loss >= 0 && corrupt >= 0 && loss + corrupt <= 100
66 end
67 if DELAY_KEYS.include?(key)
68 pairs = values.map { |value| value.split(':', 2).map(&:to_i) }
69 return pairs.all? { |n, extra| (1..9999).cover?(n) && (1..50).cover?(extra) } &&
70 pairs.map(&:first).uniq.length == pairs.length
71 end
72 values.all? { |value| RANGES.fetch(key).cover?(value.to_i) } &&
73 values.map(&:to_i).uniq.length == values.length
74end

129def check_settings(lines)
130 settings = {}
131 lines.each do |key, values|
132 raise SettingsError, "unknown key #{key}" unless KEYS.include?(key)
133 raise SettingsError, "duplicate key #{key}" if settings.key?(key)
134 count_ok = if key == 'RANDOM'
135 values.length == 3
136 elsif LIST_KEYS.include?(key)
137 !values.empty?
138 else
139 values.length == 1
140 end
141 raise SettingsError, "wrong number of values for #{key}" unless count_ok
142 settings[key] = values
143 end
144 REQUIRED.each do |key|
145 raise SettingsError, "missing key #{key}" unless settings.key?(key)
146 end
147 KEYS.each do |key|
148 if settings.key?(key) && !valid_value?(key, settings[key])
149 raise SettingsError, "bad value for #{key}"
150 end
151 end
152 config = Config.new(settings)
153 if config.protocol == 'SR' && !config.dupack.nil?
154 raise SettingsError, 'DUPACK not supported for SR'
155 end
156 if (config.protocol == 'SW' || !config.unsafe) && config.window > config.window_limit
157 raise SettingsError, "window #{config.window} exceeds limit #{config.window_limit} for #{config.protocol}"
158 end
159 both = config.lose_data.keys & config.corrupt_data.keys
160 raise SettingsError, "data ##{both.min} both lost and corrupted" unless both.empty?
161 both = config.lose_ctrl.keys & config.corrupt_ctrl.keys
162 raise SettingsError, "control ##{both.min} both lost and corrupted" unless both.empty?
163 config
164end

173class LossGenerator
174 def initialize(seed, loss, corruption)
175 @state, @loss, @corruption = seed, loss, corruption
176 end
177
178 def next_outcome
179 @state = RANDOM_MULTIPLIER * @state % RANDOM_MODULUS
180 roll = @state % 100
181 return 'LOST' if roll < @loss
182 return 'CORRUPT' if roll < @loss + @corruption
183 'NORMAL'
184 end
185end

187class Channel
188 attr_reader :sent, :lost, :corrupted
189
190 def initialize(delay, losses, corruptions, extra_delays)
191 @delay, @losses, @corruptions, @extra_delays = delay, losses, corruptions, extra_delays
192 @sent = @lost = @corrupted = 0
193 @in_transit = []
194 end
195
196 def send(kind, value, payload, tick, generator)
197 @sent += 1
198 outcome = generator.nil? ? 'NORMAL' : generator.next_outcome
199 outcome = if @losses.key?(@sent)
200 'LOST'
201 elsif @corruptions.key?(@sent)
202 'CORRUPT'
203 else
204 outcome
205 end
206 item = Transmission.new(@sent, kind, value, payload,
207 outcome, tick + @delay + @extra_delays.fetch(@sent, 0))
208 @in_transit << item
209 item
210 end
211
212 def arrivals(tick)
213 current, @in_transit = @in_transit.partition { |item| item.arrival == tick }
214 current
215 end
216
217 def count_damage(item)
218 item.outcome == 'LOST' ? @lost += 1 : @corrupted += 1
219 end
220end

222class Simulation
223 attr_reader :lines, :completion, :delivered, :data_channel, :config
224
225 def initialize(config)
226 @config = config
227 @selective = config.protocol == 'SR'
228 @modulus = 2**config.seqbits
229 @visible = TRACE_SIDES.fetch(config.trace)
230 @generator = config.random.nil? ? nil : LossGenerator.new(*config.random)
231 @data_channel = Channel.new(config.delay, config.lose_data, config.corrupt_data, config.delay_data)
232 @control_channel = Channel.new(config.delay, config.lose_ctrl, config.corrupt_ctrl, config.delay_ctrl)
233 @records = Array.new(config.frames) { FrameRecord.new }
234 @lines = []
235 @tick = @base = @frames_sent = @next_frame = 0
236 @timer_length = config.timeout
237 @timer = nil
238 @duplicates = 0
239 @acked = Array.new(config.frames, false)
240 @timers = Array.new(config.frames)
241 @queue = []
242 @cwnd = 1
243 @ssthresh = config.window
244 @round_acks = 0
245 @expected = 0
246 @nak_sent = false
247 @buffer = {}
248 @resent = @naks = @timeouts = @fast_retransmits = 0
249 @delivered = @errors = @discarded = @max_buffer = 0
250 @completion = nil
251 end
252
253 def emit(*parts)
254 @lines << ([@tick] + parts).join(' ')
255 end
256
257 def trace(side, *parts)
258 emit(*parts) if @visible.include?(side)
259 end
260
261 def frame_of(value)
262 @base + (value - @base) % @modulus
263 end
264
265 def effective_window
266 @config.congestion ? [@config.window, @cwnd].min : @config.window
267 end
268
269 def run
270 (0..@config.limit).each do |tick|
271 @tick = tick
272 @data_channel.arrivals(tick).each do |item|
273 item.outcome == 'NORMAL' ? receive_data(item) : report_damage(@data_channel, item, 'RECEIVER')
274 end
275 @control_channel.arrivals(tick).each do |item|
276 item.outcome == 'NORMAL' ? receive_control(item) : report_damage(@control_channel, item, 'SENDER')
277 end
278 completed = @base == @config.frames
279 unless completed
280 process_timeouts
281 send_data
282 end
283 snapshot if @config.snapshots.key?(tick)
284 if completed
285 @completion = tick
286 emit('DONE')
287 return
288 end
289 end
290 emit('ABORT')
291 end
292
293 def report_damage(channel, item, side)
294 channel.count_damage(item)
295 trace(side, item.outcome, item.kind, "##{item.number}")
296 end
297
298 def receive_data(item)
299 trace('RECEIVER', 'RECV', "seq=#{item.value}", "##{item.number}")
300 if @selective
301 selective_receive(item.value, item.payload)
302 else
303 go_back_n_receive(item.value, item.payload)
304 end
305 end
306
307 def go_back_n_receive(seq, payload)
308 if seq == @expected % @modulus
309 deliver(payload)
310 send_control('ack', seq)
311 return
312 end
313 discard(seq)
314 if @config.nak && !@nak_sent
315 send_control('nak', @expected % @modulus)
316 @nak_sent = true
317 elsif @expected > 0
318 send_control('ack', (@expected - 1) % @modulus)
319 end
320 end
321
322 def selective_receive(seq, payload)
323 offset = (seq - @expected) % @modulus
324 if offset >= @config.window
325 discard(seq)
326 send_control('ack', seq) if (@expected - seq) % @modulus <= @config.window
327 return
328 end
329 position = @expected + offset
330 if @buffer.key?(position)
331 discard(seq)
332 send_control('ack', seq)
333 elsif offset == 0
334 deliver(payload)
335 deliver(@buffer.delete(@expected)) while @buffer.key?(@expected)
336 send_control('ack', seq)
337 else
338 @buffer[position] = payload
339 @max_buffer = [@max_buffer, @buffer.length].max
340 trace('RECEIVER', 'BUFFER', "f=#{position}")
341 send_control('ack', seq)
342 if @config.nak && !@nak_sent
343 send_control('nak', @expected % @modulus)
344 @nak_sent = true
345 end
346 end
347 end
348
349 def deliver(payload)
350 position = @expected
351 if payload == position
352 @delivered += 1
353 @records[payload].delivered = @tick
354 trace('RECEIVER', 'DELIVER', "f=#{position}")
355 else
356 @errors += 1
357 trace('RECEIVER', 'DELIVER', "f=#{position}", "wrong=#{payload}")
358 end
359 @expected += 1
360 @nak_sent = false
361 end
362
363 def discard(seq)
364 @discarded += 1
365 trace('RECEIVER', 'DISCARD', "seq=#{seq}")
366 end
367
368 def send_control(kind, value)
369 item = @control_channel.send(kind, value, nil, @tick, @generator)
370 @naks += 1 if kind == 'nak'
371 trace('RECEIVER', kind.upcase, "a=#{value}", "##{item.number}")
372 end
373
374 def receive_control(item)
375 before = [@cwnd, @ssthresh]
376 if item.kind == 'ack'
377 @selective ? selective_ack(item) : go_back_n_ack(item)
378 else
379 @selective ? selective_nak(item) : go_back_n_nak(item)
380 end
381 report_window(before)
382 end
383
384 def control_line(item, accepted)
385 parts = ["#{accepted ? 'GOT' : 'OLD'}#{item.kind.upcase}", "a=#{item.value}", "##{item.number}"]
386 parts << "base=#{@base}" if accepted
387 trace('SENDER', *parts)
388 end
389
390 def go_back_n_ack(item)
391 frame = frame_of(item.value)
392 if frame < @frames_sent
393 grow_window(frame - @base + 1)
394 move_base(frame + 1)
395 @next_frame = @base if @next_frame < @base
396 @timer = @base == @next_frame ? nil : @tick + @timer_length
397 control_line(item, true)
398 return
399 end
400 control_line(item, false)
401 return if @config.dupack.nil? || @base < 1
402 if item.value == (@base - 1) % @modulus
403 @duplicates += 1
404 if @duplicates == @config.dupack
405 @fast_retransmits += 1
406 trace('SENDER', 'FASTRETX', "f=#{@base}")
407 go_back
408 halve_window
409 end
410 end
411 end
412
413 def go_back_n_nak(item)
414 frame = frame_of(item.value)
415 if frame > @frames_sent
416 control_line(item, false)
417 return
418 end
419 grow_window(frame - @base)
420 move_base(frame)
421 go_back
422 halve_window
423 control_line(item, true)
424 end
425
426 def selective_ack(item)
427 frame = frame_of(item.value)
428 if frame >= @frames_sent || @acked[frame]
429 control_line(item, false)
430 return
431 end
432 @acked[frame] = true
433 @records[frame].acked = @tick
434 @timers[frame] = nil
435 @queue.delete(frame)
436 grow_window(1)
437 previous = @base
438 @base += 1 while @base < @frames_sent && @acked[@base]
439 @timer_length = @config.timeout if @base > previous
440 control_line(item, true)
441 end
442
443 def selective_nak(item)
444 frame = frame_of(item.value)
445 if frame >= @frames_sent || @acked[frame]
446 control_line(item, false)
447 return
448 end
449 @timers[frame] = nil
450 @queue << frame unless @queue.include?(frame)
451 halve_window
452 control_line(item, true)
453 end
454
455 def move_base(new_base)
456 return if new_base <= @base
457 (@base...new_base).each { |frame| @records[frame].acked = @tick }
458 @base = new_base
459 @timer_length = @config.timeout
460 @duplicates = 0
461 end
462
463 def go_back
464 @next_frame = @base
465 @timer = nil
466 @duplicates = 0
467 end
468
469 def grow_window(count)
470 return unless @config.congestion
471 count.times do
472 next if @cwnd >= @config.window
473 if @cwnd < @ssthresh
474 @cwnd += 1
475 else
476 @round_acks += 1
477 if @round_acks == @cwnd
478 @cwnd += 1
479 @round_acks = 0
480 end
481 end
482 end
483 end
484
485 def halve_window
486 return unless @config.congestion
487 @ssthresh = [@cwnd / 2, 1].max
488 @cwnd = @ssthresh
489 @round_acks = 0
490 end
491
492 def collapse_window
493 return unless @config.congestion
494 @ssthresh = [@cwnd / 2, 1].max
495 @cwnd = 1
496 @round_acks = 0
497 end
498
499 def report_window(before)
500 trace('SENDER', 'CWND', @cwnd, "ssthresh=#{@ssthresh}") if [@cwnd, @ssthresh] != before
501 end
502
503 def process_timeouts
504 if @selective
505 (@base...@frames_sent).each do |frame|
506 next unless @timers[frame] == @tick
507 @timers[frame] = nil
508 @queue << frame
509 time_out(frame)
510 end
511 elsif @timer == @tick
512 time_out(@base)
513 go_back
514 end
515 end
516
517 def time_out(frame)
518 before = [@cwnd, @ssthresh]
519 @timeouts += 1
520 trace('SENDER', 'TIMEOUT', "f=#{frame}")
521 @timer_length = [2 * @timer_length, 8 * @config.timeout].min if @config.backoff
522 collapse_window
523 report_window(before)
524 end
525
526 def may_send_new?(frame)
527 frame < @config.frames && frame < @base + effective_window &&
528 frame * @config.interval <= @tick
529 end
530
531 def send_data
532 @selective ? selective_send : go_back_n_send
533 end
534
535 def go_back_n_send
536 frame = @next_frame
537 return if frame >= @config.frames || frame >= @base + effective_window
538 return if frame >= @frames_sent && frame * @config.interval > @tick
539 transmit(frame)
540 @frames_sent = [@frames_sent, frame + 1].max
541 @next_frame += 1
542 @timer = @tick + @timer_length if @timer.nil?
543 end
544
545 def selective_send
546 if !@queue.empty?
547 frame = @queue.shift
548 elsif may_send_new?(@frames_sent)
549 frame = @frames_sent
550 @frames_sent += 1
551 else
552 return
553 end
554 transmit(frame)
555 @timers[frame] = @tick + @timer_length
556 end
557
558 def transmit(frame)
559 record = @records[frame]
560 seq = frame % @modulus
561 item = @data_channel.send('data', seq, frame, @tick, @generator)
562 record.transmissions += 1
563 if record.transmissions == 1
564 record.first = @tick
565 label = 'SEND'
566 else
567 @resent += 1
568 label = 'RESEND'
569 end
570 trace('SENDER', label, "f=#{frame}", "seq=#{seq}", "##{item.number}")
571 end
572
573 def sender_mark(frame)
574 if @selective
575 acknowledged = @acked[frame]
576 waiting = @queue.include?(frame)
577 sent = frame < @frames_sent
578 else
579 acknowledged = frame < @base
580 waiting = @next_frame <= frame && frame < @frames_sent
581 sent = frame < @next_frame
582 end
583 return 'A' if acknowledged
584 return 'R' if waiting
585 return 'O' if sent
586 return 'W' if frame < @base + effective_window
587 '.'
588 end
589
590 def receiver_mark(position)
591 return 'D' if position < @expected
592 if @selective
593 return 'B' if @buffer.key?(position)
594 return 'W' if position < @expected + @config.window
595 elsif position == @expected
596 return 'W'
597 end
598 '.'
599 end
600
601 def snapshot
602 sender = (0...@config.frames).map { |f| sender_mark(f) }.join
603 receiver = (0...@config.frames).map { |f| receiver_mark(f) }.join
604 emit('SNAPSHOT', "sender=#{sender}", "receiver=#{receiver}")
605 end

608 @data_channel.sent
609 end
610
611 def goodput_text
612 truncated(@delivered, @completion, 2)
613 end
614

616 truncated(100 * @delivered, data_count, 1)
617 end
618
619 def latency_text
620 return '-' if @delivered == 0
621 total = @records.sum { |r| r.delivered.nil? ? 0 : r.delivered - r.first }
622 truncated(total, @delivered, 2)
623 end

625 def report
626 lines = @records.each_with_index.map do |record, frame|
627 "frame #{frame}: seq=#{frame % @modulus} tx=#{record.transmissions} " \
628 "first=#{tick_text(record.first)} delivered=#{tick_text(record.delivered)} " \
629 "acked=#{tick_text(record.acked)}"
630 end
631 data, control = @data_channel, @control_channel
632 lines << "totals: data=#{data.sent} resent=#{@resent} control=#{control.sent} " \
633 "nak=#{@naks} timeouts=#{@timeouts} fastretx=#{@fast_retransmits}"
634 lines << "channel: lost_data=#{data.lost} lost_ctrl=#{control.lost} " \
635 "corrupt_data=#{data.corrupted} corrupt_ctrl=#{control.corrupted}"
636 lines << "receiver: delivered=#{@delivered} errors=#{@errors} " \
637 "discarded=#{@discarded} max_buffer=#{@max_buffer}"
638 if @completion.nil?
639 completion = goodput = '-'
640 else
641 completion, goodput = @completion.to_s, goodput_text
642 end
643 utilization = truncated(100 * data.sent, @tick + 1, 1)
644 lines << "result: completion=#{completion} goodput=#{goodput} " \
645 "efficiency=#{efficiency_text}% utilization=#{utilization}% " \
646 "latency=#{latency_text}"
647 lines
648 end

686 completed = scenarios.each_with_index.select { |scenario, _| scenario.completed? }
687 completed.sort_by! do |scenario, index|

689 [-Rational(run.delivered, run.completion),

695 "goodput=#{run.goodput_text} efficiency=#{run.efficiency_text}%"
696 end
697 scenarios.each do |scenario|
698 lines << "- #{scenario.name} #{scenario.status}" unless scenario.completed?
699 end
700 lines
701end

704 lines = ['== protocols ==']
705 PROTOCOLS.each do |protocol|
