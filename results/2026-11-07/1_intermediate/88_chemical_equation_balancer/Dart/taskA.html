92 Composition sequence(){
93 var c=<String,int>{
94 };
95 var items=0;
96 while(true){
97 var ch=peek();
98 Composition item;
99 if(RegExp(r'^[A-Z]$').hasMatch(ch)){
100 var e=text[p++];
101 if(RegExp(r'^[a-z]$').hasMatch(peek()))e+=text[p++];
102 seen.add(e);
103 item={
104 e:1
105 };
106 }else if(ch=='('||ch=='['){
107 p++;
108 item=sequence();
109 if(peek()!=(ch=='('?')':']'))throw const FormatException();
110 p++;
111 }else break;
112 add(c,item,number());
113 items++;
114 }
115 if(items==0)throw const FormatException();
116 return c;
117 }
118 Formula parse(){
119 var c=sequence();
120 while(peek()=='.'){
121 p++;
122 var k=number();
123 add(c,sequence(),k);
124 }
125 var q=0;
126 if(peek()=='^'){
127 p++;
128 var k=number();
129 var ch=peek();
130 if(ch!='+'&&ch!='-')throw const FormatException();
131 p++;
132 q=ch=='+'?k:-k;
133 }
134 if(p!=text.length)throw const FormatException();
135 return Formula(c,q);
136 }

208(int,List<List<int>>) nullspace(List<List<int>> rows,int n){

210 var piv=<int>[];
211 var free=<int>[];
212 for(var j=0;j<n;j++){

217 free.add(j);
218 continue;
219 }
220 var row=a[r];
221 a[r]=a[i];
222 a[i]=row;
223 var t=a[r][j];

226 if(k!=r&&a[k][j].n!=0){
227 var f=a[k][j];
228 for(var z=0;z<n;z++)a[k][z]=a[k][z]-f*a[r][z];
229 }
230 }
231 piv.add(j);
232 }

234 for(var f in free){

242 for(var x in b)g=gcd(g,x);

520 return [...out,for(var i=0;i<list.length;i++)'${list[i].name} ${dec(qs[0]*v[i]*1000000~/d,6)} mol ${dec(qs[0]*v[i]*list[i].m~/d,3)} g${gas(list[i],qs[0]*v[i],d)}'];
521 }
522 if(cmd=='YIELD'){
523 var t=qs[0]*v[product]*list[product].m~/(v[target]*list[target].m);
524 return [...out,'theoretical ${dec(t,3)} g','actual ${dec(qs[1],3)} g',t==0?'yield undefined':'yield ${dec(qs[1]*10000~/t,2)}%'];
525 }
526 var l=0;
527 for(var i=1;i<r;i++){
528 if(qs[i]*v[l]*list[l].m<qs[l]*v[i]*list[i].m)l=i;
529 }
530 var d=v[l]*list[l].m;
531 out.addAll(['limiting ${list[l].name}','extent ${dec(qs[l]*1000000~/d,6)} mol']);
532 for(var i=r;i<list.length;i++)out.add('product ${list[i].name} ${dec(qs[l]*v[i]*list[i].m~/d,3)} g${gas(list[i],qs[l]*v[i],d)}');
533 for(var i=0;i<r;i++){
534 if(i!=l)out.add('remaining ${list[i].name} ${dec((qs[i]*d-qs[l]*v[i]*list[i].m)~/d,3)} g');
535 }
536 return out;
537 }
