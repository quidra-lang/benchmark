72class FormulaParser {
73 public int $p=0;
74 public array $seen=[];
75 function __construct(public string $text){
76 }
77 function peek(): string {
78 return $this->text[$this->p]??'';
79 }
80 function number(): int {
81 $start=$this->p;
82 while(preg_match('/^[0-9]$/D',$this->peek()))$this->p++;
83 if($start===$this->p)return 1;
84 if($this->text[$start]==='0')throw new InvalidArgumentException();
85 return (int)substr($this->text,$start,$this->p-$start);
86 }
87 function sequence(): array {
88 $c=[];
89 $items=0;
90 while(true){
91 $ch=$this->peek();
92 if(preg_match('/^[A-Z]$/D',$ch)){
93 $e=$ch;
94 $this->p++;
95 if(preg_match('/^[a-z]$/D',$this->peek())){
96 $e.=$this->peek();
97 $this->p++;
98 }
99 $this->seen[]=$e;
100 $item=[$e=>1];
101 }
102 elseif($ch==='('||$ch==='['){
103 $this->p++;
104 $item=$this->sequence();
105 if($this->peek()!==($ch==='('?')':']'))throw new InvalidArgumentException();
106 $this->p++;
107 }else break;
108 $k=$this->number();
109 addComposition($c,$item,$k);
110 $items++;
111 }
112 if(!$items)throw new InvalidArgumentException();
113 return $c;
114 }
115 function parse(): array {
116 $c=$this->sequence();
117 while($this->peek()==='.'){
118 $this->p++;
119 $k=$this->number();
120 addComposition($c,$this->sequence(),$k);
121 }
122 $q=0;
123 if($this->peek()==='^'){
124 $this->p++;
125 $k=$this->number();
126 $ch=$this->peek();
127 if($ch!=='+'&&$ch!=='-')throw new InvalidArgumentException();
128 $this->p++;
129 $q=$ch==='+'?$k:-$k;
130 }
131 if($this->p!==strlen($this->text))throw new InvalidArgumentException();
132 return ['c'=>$c,'q'=>$q];
133 }

163function equation(array $a,bool $check=false): array {
164 $valid=count($a)%2===1;
165 $arrow=-1;
166 foreach($a as $i=>$s){
167 $op=in_array($s,['+','->'],true);
168 if($i%2){
169 if(!$op)$valid=false;
170 if($s==='->'){
171 if($arrow>=0)$valid=false;
172 $arrow=$i;
173 }
174 }
175 elseif($op)$valid=false;
176 }
177 if(!$valid||$arrow<0)chemError('malformed equation');
178 $list=[];
179 for($i=0;$i<count($a);$i+=2)$list[]=species($a[$i],$check);

183 return ['list'=>$list,'r'=>$r,'sign'=>$sign];

208function nullspace(array $rows,int $n): array {
209 $a=array_map(fn($r)=>array_map(fn($x)=>new Fraction($x),$r),$rows);
210 $piv=[];
211 $free=[];
212 for($j=0;$j<$n;$j++){
213 $r=count($piv);
214 $i=$r;
215 while($i<count($a)&&!$a[$i][$j]->n)$i++;
216 if($i===count($a)){
217 $free[]=$j;
218 continue;
219 }
220 [$a[$r],$a[$i]]=[$a[$i],$a[$r]];
221 $t=$a[$r][$j];
222 $a[$r]=array_map(fn($x)=>$x->div($t),$a[$r]);
223 foreach($a as $k=>$row)if($k!==$r&&$row[$j]->n){
224 $t=$row[$j];
225 foreach($row as $z=>$x)$a[$k][$z]=$x->sub($t->mul($a[$r][$z]));
226 }
227 $piv[]=$j;
228 }
229 $basis=[];
230 foreach($free as $f){
231 $v=array_map(fn($_)=>new Fraction(),range(1,$n));

237 $g=array_reduce($b,fn($g,$x)=>gcd($g,$x),0);
238 $basis[]=array_map(fn($x)=>intdiv($x,$g),$b);
239 }
240 return [count($piv),$basis];
241}
242function coefficients(array $list,array $sign,?int $k=null): array {
243 [, $rows]=compositionMatrix($list,$sign);

247 if(!$d)throw new RuntimeException('IMPOSSIBLE');
248 $v=$basis[0];
249 if($v[0]<0)$v=array_map(fn($x)=>-$x,$v);
250 foreach(array_slice($v,0,$k??count($list)) as $x)if($x<=0)throw new RuntimeException('IMPOSSIBLE');
251 return $v;

507 }
508 if($cmd==='LIMITING'){
509 if(count($rest)!==$r)chemError("expected $r masses, got ".count($rest));
510 $qs=array_map('quantity',$rest);
511 }
512 if($cmd==='YIELD'){
513 foreach($list as $i=>$s)if($i<$r&&$s['name']===$rest[0]){
514 $target=$i;
515 break;
516 }
517 if($target<0)chemError($rest[0].' is not a reactant');
518 foreach($list as $i=>$s)if($i>=$r&&$s['name']===$rest[2]){
519 $product=$i;
520 break;
521 }
522 if($product<0)chemError($rest[2].' is not a product');
523 $qs=[quantity($rest[1]),quantity($rest[3])];
524 }
525 $v=coefficients($list,$sign);
526 $out=[notation($eq,$v)];
527 if($cmd==='BALANCE')return $out;
528 if($cmd==='DELTAH'){
529 $h=0;
530 foreach($list as $i=>$s)$h-=$sign[$i]*$v[$i]*$this->heats[$s['name']];
531 $out[]='dH = '.ent($h).' kJ ('.($h<0?'exothermic':($h>0?'endothermic':'thermoneutral')).')';
532 return $out;
533 }
534 if($cmd==='STOICH'){
535 $d=$v[$target]*$list[$target]['m'];
536 foreach($list as $i=>$s){
537 $x=$qs[0]*$v[$i];
538 $out[]=$s['name'].' '.dec(intdiv($x*1000000,$d),6).' mol '.dec(intdiv($x*$s['m'],$d),3).' g'.gas($s,$x,$d);
539 }
540 return $out;
541 }
542 if($cmd==='YIELD'){
543 $t=intdiv($qs[0]*$v[$product]*$list[$product]['m'],$v[$target]*$list[$target]['m']);
544 return array_merge($out,['theoretical '.dec($t,3).' g','actual '.dec($qs[1],3).' g',$t?'yield '.dec(intdiv($qs[1]*10000,$t),2).'%':'yield undefined']);
545 }
546 $l=0;
547 for($i=1;$i<$r;$i++)if($qs[$i]*$v[$l]*$list[$l]['m']<$qs[$l]*$v[$i]*$list[$i]['m'])$l=$i;
548 $d=$v[$l]*$list[$l]['m'];
549 $out[]='limiting '.$list[$l]['name'];
550 $out[]='extent '.dec(intdiv($qs[$l]*1000000,$d),6).' mol';
551 for($i=$r;$i<count($list);$i++)$out[]='product '.$list[$i]['name'].' '.dec(intdiv($qs[$l]*$v[$i]*$list[$i]['m'],$d),3).' g'.gas($list[$i],$qs[$l]*$v[$i],$d);
552 for($i=0;$i<$r;$i++)if($i!==$l)$out[]='remaining '.$list[$i]['name'].' '.dec(intdiv($qs[$i]*$d-$qs[$l]*$v[$i]*$list[$i]['m'],$d),3).' g';
553 return $out;
554 }
