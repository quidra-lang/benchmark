type Composition = BTreeMap<String, Int>;
fn order(c: &Composition) -> Vec<String> {
    let mut v: Vec<_> = c.keys().cloned().collect();
    if c.contains_key("C") {
        v.sort_by_key(|s| {
            if s == "C" {
                (0, s.clone())
            } else if s == "H" {
                (1, s.clone())
            } else {
                (2, s.clone())
            }
        })
    }
    v
}
fn hill(c: &Composition, q: Int) -> String {
    let mut s = order(c)
        .iter()
        .map(|e| {
            format!(
                "{}{}",
                e,
                if c[e] == 1 {
                    String::new()
                } else {
                    c[e].to_string()
                }
            )
        })
        .collect::<String>();
    if q != 0 {
        s.push_str(&format!(
            "^{}{}",
            if q.abs() == 1 {
                String::new()
            } else {
                q.abs().to_string()
            },
            if q > 0 { "+" } else { "-" }
        ))
    }
    s
}
