144 static class Parser {
145 final String text;
146 int p;
147 List<String> seen=new ArrayList<>();
148 Parser(String s){
149 text=s;
150 }
151 char peek(){
152 return p<text.length()?text.charAt(p):'\0';
153 }
154 long number(){
155 int start=p;
156 while(peek()>='0'&&peek()<='9')p++;
157 if(start==p)return 1;
158 if(text.charAt(start)=='0')throw new IllegalArgumentException();
159 return Long.parseLong(text.substring(start,p));
160 }
161 Map<String,Long> sequence(){
162 Map<String,Long> c=new HashMap<>();
163 int items=0;
164 while(true){
165 char ch=peek();
166 Map<String,Long> item;
167 if(ch>='A'&&ch<='Z'){
168 String e=""+text.charAt(p++);
169 if(peek()>='a'&&peek()<='z')e+=text.charAt(p++);
170 seen.add(e);
171 item=Map.of(e,1L);
172 }else if(ch=='('||ch=='['){
173 p++;
174 item=sequence();
175 if(peek()!=(ch=='('?')':']'))throw new IllegalArgumentException();
176 p++;
177 }else break;
178 c=add(c,item,number());
179 items++;
180 }
181 if(items==0)throw new IllegalArgumentException();
182 return c;
183 }
184 Formula parse(){
185 Map<String,Long> c=sequence();
186 while(peek()=='.'){
187 p++;
188 long k=number();
189 c=add(c,sequence(),k);
190 }
191 long q=0;
192 if(peek()=='^'){
193 p++;
194 long k=number();
195 char ch=peek();
196 if(ch!='+'&&ch!='-')throw new IllegalArgumentException();
197 p++;
198 q=ch=='+'?k:-k;
199 }
200 if(p!=text.length())throw new IllegalArgumentException();
201 return new Formula(c,q);
202 }
203 }

288 static Nullspace nullspace(long[][] rows,int n){
289 Fraction[][] a=new Fraction[rows.length][n];
290 for(int i=0;i<rows.length;i++)for(int j=0;j<n;j++)a[i][j]=new Fraction(rows[i][j]);
291 List<Integer> piv=new ArrayList<>(),free=new ArrayList<>();
292 for(int j=0;j<n;j++){
293 int r=piv.size(),i=r;
294 while(i<a.length&&a[i][j].n==0)i++;
295 if(i==a.length){
296 free.add(j);
297 continue;
298 }
299 Fraction[] row=a[r];
300 a[r]=a[i];
301 a[i]=row;
302 Fraction t=a[r][j];
303 for(int z=0;z<n;z++)a[r][z]=a[r][z].div(t);
304 for(int k=0;k<a.length;k++)if(k!=r&&a[k][j].n!=0){
305 Fraction f=a[k][j];
306 for(int z=0;z<n;z++)a[k][z]=a[k][z].sub(f.mul(a[r][z]));
307 }
308 piv.add(j);
309 }
310 List<long[]> basis=new ArrayList<>();
311 for(int f:free){
312 Fraction[] v=new Fraction[n];
313 Arrays.setAll(v,i->new Fraction(0));
314 v[f]=new Fraction(1);
315 for(int i=0;i<piv.size();i++)v[piv.get(i)]=new Fraction(-a[i][f].n,a[i][f].d);
316 long l=1;
317 for(Fraction x:v)l=l/gcd(l,x.d)*x.d;
318 long[] b=new long[n];
319 long g=0;
320 for(int i=0;i<n;i++){
321 b[i]=v[i].n*(l/v[i].d);
322 g=gcd(g,b[i]);
323 }
324 for(int i=0;i<n;i++)b[i]/=g;
325 basis.add(b);
326 }
327 return new Nullspace(piv.size(),basis);
328 }

533 for(int i=0;i<eq.list.size();i++)(i<eq.r?left:right).add(term(list.get(i).name(),v[i]));
534 for(int i=eq.list.size();i<list.size();i++)if(v[i]!=0)(v[i]>0?left:right).add(term(list.get(i).name(),Math.abs(v[i])));
535 List<String> out=new ArrayList<>(List.of(String.join(" + ",left)+" -> "+String.join(" + ",right)));
536 if(cmd.equals("HALF"))out.add(v[v.length-1]>0?"reduction":v[v.length-1]<0?"oxidation":"no electron transfer");
537 return out;
538 }
539 List<String> eqWords=a,rest=List.of();
540 if(List.of("STOICH","LIMITING","YIELD").contains(cmd)){
541 if(Collections.frequency(a,":")!=1)fail("invalid separator");
542 int i=a.indexOf(":");
543 eqWords=a.subList(0,i);
544 rest=a.subList(i+1,a.size());
545 if((cmd.equals("STOICH")&&rest.size()!=2)||(cmd.equals("YIELD")&&rest.size()!=4))fail("wrong argument count for "+cmd);
546 }
547 Equation eq=equation(eqWords,cmd.equals("CHECK"));
548 List<Species> list=eq.list;
549 int r=eq.r;
550 long[] signs=eq.signs();
551 if(cmd.equals("MATRIX")){
552 Matrix matrix=matrix(list,signs);
553 Nullspace ns=nullspace(matrix.rows,list.size());
554 List<String> out=new ArrayList<>();
555 out.add("columns: "+list.stream().map(Species::name).collect(Collectors.joining(" ")));
556 for(int i=0;i<matrix.rows.length;i++)out.add(matrix.labels.get(i)+": "+Arrays.stream(matrix.rows[i]).mapToObj(Long::toString).collect(Collectors.joining(" ")));
557 out.add("rank "+ns.rank+", nullity "+ns.basis.size());
558 for(long[] row:ns.basis)out.add("null: "+Arrays.stream(row).mapToObj(Long::toString).collect(Collectors.joining(" ")));
559 return out;
560 }
561 if(cmd.equals("CHECK")){
562 Matrix matrix=matrix(list,signs);
563 List<String> diff=new ArrayList<>();
564 for(int i=0;i<matrix.rows.length;i++){
565 long l=0,rr=0;
566 for(int j=0;j<list.size();j++)if(j<r)l+=matrix.rows[i][j]*list.get(j).coefficient;
567 else rr-=matrix.rows[i][j]*list.get(j).coefficient;
568 String label=matrix.labels.get(i);
569 if(l!=rr)diff.add(label+": "+(label.equals("charge")?signed(l):Long.toString(l))+" != "+(label.equals("charge")?signed(rr):Long.toString(rr)));
570 }
571 if(!diff.isEmpty()){
572 diff.addFirst("not balanced");
573 return diff;
574 }
575 long g=0;
576 for(Species s:list)g=gcd(g,s.coefficient);
577 return List.of(g==1?"balanced":"balanced, not in lowest terms (gcd "+g+")");
578 }
579 int target=-1,product=-1;
580 long[] qs={
581 };
582 if(cmd.equals("DELTAH"))for(Species s:list)if(!heats.containsKey(s.name()))fail("no enthalpy for "+s.name());
583 if(cmd.equals("STOICH")){
584 target=find(list,rest.get(0),0,list.size());
585 if(target<0)fail("species "+rest.get(0)+" not in equation");
586 qs=new long[]{
587 quantity(rest.get(1))
588 };
589 }
590 if(cmd.equals("LIMITING")){
591 if(rest.size()!=r)fail("expected "+r+" masses, got "+rest.size());
592 qs=new long[r];
593 for(int i=0;i<r;i++)qs[i]=quantity(rest.get(i));
594 }
595 if(cmd.equals("YIELD")){
596 target=find(list,rest.get(0),0,r);
597 if(target<0)fail(rest.get(0)+" is not a reactant");
598 product=find(list,rest.get(2),r,list.size());
599 if(product<0)fail(rest.get(2)+" is not a product");
600 qs=new long[]{
601 quantity(rest.get(1)),quantity(rest.get(3))
602 };
603 }
604 long[] v=coefficients(eq);
605 List<String> out=new ArrayList<>(List.of(notation(eq,v)));
606 if(cmd.equals("BALANCE"))return out;
607 if(cmd.equals("DELTAH")){
608 long h=0;
609 for(int i=0;i<list.size();i++)h-=signs[i]*v[i]*heats.get(list.get(i).name());
610 out.add("dH = "+ent(h)+" kJ ("+(h<0?"exothermic":h>0?"endothermic":"thermoneutral")+")");
611 return out;
612 }
613 if(cmd.equals("STOICH")){
614 long d=v[target]*list.get(target).mass();
615 for(int i=0;i<list.size();i++){
616 Species s=list.get(i);
617 long x=qs[0]*v[i];
618 out.add(s.name()+" "+dec(x*1000000/d,6)+" mol "+dec(x*s.mass()/d,3)+" g"+gas(s,x,d));
619 }
620 return out;
621 }
622 if(cmd.equals("YIELD")){
623 long t=qs[0]*v[product]*list.get(product).mass()/(v[target]*list.get(target).mass());
624 out.add("theoretical "+dec(t,3)+" g");
625 out.add("actual "+dec(qs[1],3)+" g");
626 out.add(t==0?"yield undefined":"yield "+dec(qs[1]*10000/t,2)+"%");
627 return out;
628 }
629 int l=0;
630 for(int i=1;i<r;i++)if(qs[i]*v[l]*list.get(l).mass()<qs[l]*v[i]*list.get(i).mass())l=i;
631 long d=v[l]*list.get(l).mass();
632 out.add("limiting "+list.get(l).name());
633 out.add("extent "+dec(qs[l]*1000000/d,6)+" mol");
634 for(int i=r;i<list.size();i++)out.add("product "+list.get(i).name()+" "+dec(qs[l]*v[i]*list.get(i).mass()/d,3)+" g"+gas(list.get(i),qs[l]*v[i],d));
635 for(int i=0;i<r;i++)if(i!=l)out.add("remaining "+list.get(i).name()+" "+dec((qs[i]*d-qs[l]*v[i]*list.get(i).mass())/d,3)+" g");
636 return out;
637 }
