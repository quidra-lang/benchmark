169 class Parser
170 {
171 readonly string text;
172 int p;
173 public readonly List<string> Seen=new();
174 public Parser(string s){
175 text=s;
176 }
177 char Peek()=>p<text.Length?text[p]:'\0';
178 long Number(){
179 int start=p;
180 while(Peek()>='0'&&Peek()<='9')p++;
181 if(start==p)return 1;
182 if(text[start]=='0')throw new FormatException();
183 return long.Parse(text[start..p]);
184 }
185 Comp Sequence(){
186 Comp c=new();
187 int items=0;
188 while(true){
189 char ch=Peek();
190 Comp item;
191 if(ch>='A'&&ch<='Z'){
192 string e=text[p++].ToString();
193 if(Peek()>='a'&&Peek()<='z')e+=text[p++];
194 Seen.Add(e);
195 item=new(){
196 {
197 e,1
198 }
199 };
200 }else if(ch=='('||ch=='['){
201 p++;
202 item=Sequence();
203 if(Peek()!=(ch=='('?')':']'))throw new FormatException();
204 p++;
205 }else break;
206 Add(c,item,Number());
207 items++;
208 }
209 if(items==0)throw new FormatException();
210 return c;
211 }
212 public Formula Parse(){
213 Comp c=Sequence();
214 while(Peek()=='.'){
215 p++;
216 long k=Number();
217 Add(c,Sequence(),k);
218 }
219 long q=0;
220 if(Peek()=='^'){
221 p++;
222 long k=Number();
223 char ch=Peek();
224 if(ch!='+'&&ch!='-')throw new FormatException();
225 p++;
226 q=ch=='+'?k:-k;
227 }
228 if(p!=text.Length)throw new FormatException();
229 return new(c,q);
230 }
231 }

297 static (int Rank,List<long[]> Basis) Nullspace(long[][] rows,int n){
298 Fraction[][] a=rows.Select(r=>r.Select(x=>new Fraction(x)).ToArray()).ToArray();
299 List<int> piv=new(),free=new();
300 for(int j=0;j<n;j++){
301 int r=piv.Count,i=r;
302 while(i<a.Length&&a[i][j].N==0)i++;
303 if(i==a.Length){
304 free.Add(j);
305 continue;
306 }
307 (a[r],a[i])=(a[i],a[r]);
308 Fraction t=a[r][j];
309 a[r]=a[r].Select(x=>x/t).ToArray();
310 for(int k=0;k<a.Length;k++)if(k!=r&&a[k][j].N!=0){
311 Fraction f=a[k][j];
312 for(int z=0;z<n;z++)a[k][z]=a[k][z]-f*a[r][z];
313 }
314 piv.Add(j);
315 }
316 List<long[]> basis=new();
317 foreach(int f in free){
318 Fraction[] v=Enumerable.Range(0,n).Select(_=>new Fraction(0)).ToArray();
319 v[f]=new(1);
320 for(int i=0;i<piv.Count;i++)v[piv[i]]=new(-a[i][f].N,a[i][f].D);
321 long l=1;
322 foreach(Fraction x in v)l=l/Gcd(l,x.D)*x.D;
323 long[] b=v.Select(x=>x.N*(l/x.D)).ToArray();
324 long g=b.Aggregate(0L,Gcd);
325 basis.Add(b.Select(x=>x/g).ToArray());
326 }
327 return(piv.Count,basis);
328 }

349 List<string> output=es.Select(e=>e+" "+Dec(1000*a[e]*W[f]/(W[e]*a[f]),3)).ToList();

353 foreach(string e in es){

579 }
580 if(diff.Count>0){
581 diff.Insert(0,"not balanced");
582 return diff;
583 }
584 long g=list.Select(s=>s.Coefficient).Aggregate(0L,Gcd);
585 return new(){
586 g==1?"balanced":$"balanced, not in lowest terms (gcd {g})"
587 };
588 }
589 int target=-1,product=-1;
590 long[] qs=Array.Empty<long>();
591 if(cmd=="DELTAH")foreach(Species s in list)if(!heats.ContainsKey(s.Name))Fail("no enthalpy for "+s.Name);
592 if(cmd=="STOICH"){
593 target=list.FindIndex(s=>s.Name==rest[0]);
594 if(target<0)Fail("species "+rest[0]+" not in equation");
595 qs=new[]{
596 Quantity(rest[1])
597 };
598 }
599 if(cmd=="LIMITING"){
600 if(rest.Length!=split)Fail($"expected {split} masses, got {rest.Length}");
601 qs=rest.Select(Quantity).ToArray();
602 }
603 if(cmd=="YIELD"){
604 target=list.FindIndex(0,split,s=>s.Name==rest[0]);
605 if(target<0)Fail(rest[0]+" is not a reactant");
606 product=list.FindIndex(split,list.Count-split,s=>s.Name==rest[2]);
607 if(product<0)Fail(rest[2]+" is not a product");
608 qs=new[]{
609 Quantity(rest[1]),Quantity(rest[3])
610 };
611 }
612 long[] v=Coefficients(list,signs);
613 List<string> output=new(){
614 Notation(eq,v)
615 };
616 if(cmd=="BALANCE")return output;
617 if(cmd=="DELTAH"){
618 long h=list.Select((s,i)=>-signs[i]*v[i]*heats[s.Name]).Sum();
619 output.Add($"dH = {Ent(h)} kJ ({(h<0?"exothermic":h>0?"endothermic":"thermoneutral")})");
620 return output;
621 }
622 if(cmd=="STOICH"){
623 long d=v[target]*list[target].Mass;
624 output.AddRange(list.Select((s,i)=>$"{s.Name} {Dec(qs[0]*v[i]*1000000/d,6)} mol {Dec(qs[0]*v[i]*s.Mass/d,3)} g"+Gas(s,qs[0]*v[i],d)));
625 return output;
626 }
627 if(cmd=="YIELD"){
628 long t=qs[0]*v[product]*list[product].Mass/(v[target]*list[target].Mass);
629 output.AddRange(new[]{
630 $"theoretical {Dec(t,3)} g",$"actual {Dec(qs[1],3)} g",t==0?"yield undefined":$"yield {Dec(qs[1]*10000/t,2)}%"
631 });
632 return output;
633 }
634 int limiting=0;
635 for(int i=1;i<split;i++)if(qs[i]*v[limiting]*list[limiting].Mass<qs[limiting]*v[i]*list[i].Mass)limiting=i;
636 long divisor=v[limiting]*list[limiting].Mass;
637 output.Add("limiting "+list[limiting].Name);
638 output.Add("extent "+Dec(qs[limiting]*1000000/divisor,6)+" mol");
639 for(int i=split;i<list.Count;i++)output.Add($"product {list[i].Name} {Dec(qs[limiting]*v[i]*list[i].Mass/divisor,3)} g"+Gas(list[i],qs[limiting]*v[i],divisor));
640 for(int i=0;i<split;i++)if(i!=limiting)output.Add($"remaining {list[i].Name} {Dec((qs[i]*divisor-qs[limiting]*v[i]*list[i].Mass)/divisor,3)} g");
641 return output;
642 }
