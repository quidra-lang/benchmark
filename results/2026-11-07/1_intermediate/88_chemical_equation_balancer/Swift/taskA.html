86func gcd(_ a: Int, _ b: Int) -> Int {
87 var x = abs(a), y = abs(b)

93func dec(_ n: Int, _ d: Int) -> String {
94 var p = 1
95 for _ in 0..<d {
96 p *= 10
97 }
98 let f = String(n % p)
99 return "\(n / p)." + String(repeating: "0", count: d - f.count) + f
100}

110struct Fraction: CustomStringConvertible {
111 let n: Int, d: Int
112 init(_ a: Int, _ b: Int = 1) {
113 let g = gcd(a, b)
114 n = a / g * (b < 0 ? -1 : 1)
115 d = abs(b) / g
116 }
117 static func - (a: Fraction, b: Fraction) -> Fraction {
118 Fraction(a.n * b.d - b.n * a.d, a.d * b.d)
119 }
120 static func * (a: Fraction, b: Fraction) -> Fraction {
121 Fraction(a.n * b.n, a.d * b.d)
122 }
123 static func / (a: Fraction, b: Fraction) -> Fraction {
124 Fraction(a.n * b.d, a.d * b.n)
125 }
126 var description: String {
127 signed(n) + (d == 1 ? "" : "/\(d)")
128 }
129}

185struct Parser {
186 let chars: [Character]
187 var p = 0
188 var seen: [String] = []
189 init(_ text: String) {
190 chars = Array(text)
191 }
192 func peek() -> Character {
193 p < chars.count ? chars[p] : "\0"
194 }
195 mutating func number() throws -> Int {
196 let start = p
197 while peek() >= "0" && peek() <= "9" {
198 p += 1
199 }
200 if start == p {
201 return 1
202 }
203 if chars[start] == "0" {
204 throw ChemError.message("syntax")
205 }
206 return Int(String(chars[start..<p]))!
207 }
208 mutating func sequence() throws -> Composition {
209 var c: Composition = [:]
210 var items = 0
211 while true {
212 let ch = peek()
213 let item: Composition
214 if ch >= "A" && ch <= "Z" {
215 var e = String(chars[p])
216 p += 1
217 if peek() >= "a" && peek() <= "z" {
218 e += String(chars[p])
219 p += 1
220 }
221 seen.append(e)
222 item = [e: 1]
223 } else if ch == "(" || ch == "[" {
224 p += 1
225 item = try sequence()
226 if peek() != (ch == "(" ? ")" : "]") {
227 throw ChemError.message("syntax")
228 }
229 p += 1
230 } else {
231 break
232 }
233 let k = try number()
234 c = add(c, item, k)
235 items += 1
236 }
237 if items == 0 {
238 throw ChemError.message("syntax")
239 }
240 return c
241 }
242 mutating func parse() throws -> Formula {
243 var c = try sequence()
244 while peek() == "." {
245 p += 1
246 let k = try number()
247 c = add(c, try sequence(), k)
248 }
249 var q = 0
250 if peek() == "^" {
251 p += 1
252 let k = try number()
253 let ch = peek()
254 if ch != "+" && ch != "-" {
255 throw ChemError.message("syntax")
256 }
257 p += 1
258 q = ch == "+" ? k :- k
259 }
260 if p != chars.count {
261 throw ChemError.message("syntax")
262 }
263 return Formula(c: c, q: q)
264 }
265}
266func formula(_ text: String, _ word: String? = nil) throws -> Formula {
267 var p = Parser(text)
268 let f: Formula
269 do {
270 f = try p.parse()
271 } catch {
272 return try fail("invalid formula " + (word ?? text))
273 }
274 for e in p.seen {
275 if W[e] == nil {
276 return try fail("unknown element " + e)
277 }
278 }
279 return f
280}
281func species(_ word: String, _ check: Bool = false) throws -> Species {
282 let state = ["(s)", "(l)", "(g)", "(aq)"].first {
283 word.hasSuffix($0)
284 }
285 ?? ""
286 var f = String(word.dropLast(state.count))
287 var k = 1
288 if check {
289 let chars = Array(f)
290 let n = chars.prefix {
291 $0 >= "0" && $0 <= "9"
292 }.count
293 if n > 0 {
294 if chars[0] == "0" {
295 return try fail("invalid formula " + word)
296 }
297 k = Int(String(chars.prefix(n)))!
298 f = String(chars.dropFirst(n))
299 }
300 }
301 return Species(f: f, state: state, coefficient: k, formula: try formula(f, word))
302}
303func equation(_ a: [String], _ check: Bool = false) throws -> Equation {
304 var valid = a.count % 2 == 1
305 var arrow = -1
306 for i in a.indices {
307 let op = a[i] == "+" || a[i] == "->"
308 if i % 2 == 1 {
309 if !op {
310 valid = false
311 }
312 if a[i] == "->" {
313 if arrow >= 0 {
314 valid = false
315 }
316 arrow = i
317 }
318 } else if op {
319 valid = false
320 }
321 }
322 if !valid || arrow < 0 {
323 return try fail("malformed equation")
324 }
325 var list: [Species] = []
326 for i in stride(from: 0, to: a.count, by: 2) {
327 list.append(try species(a[i], check))
328 }
329 return Equation(list: list, r: (arrow + 1) / 2)
330}
331func quantity(_ s: String) throws -> Int {
332 if !matches(s, "\\A(0|[1-9][0-9]{0,3})(\\.[0-9]{1,3})?\\z") {
333 return try fail("invalid quantity " + s)
334 }
335 let p = s.components(separatedBy: ".")
336 let q = Int(p[0])! * 1000 + (p.count == 1 ? 0 : Int(p[1] + String(repeating: "0", count: 3 - p[1].count))!)
337 if q == 0 {
338 return try fail("invalid quantity " + s)
339 }
340 return q
341}
342func enthalpy(_ s: String) throws -> Int {
343 if !matches(s, "\\A-?(0|[1-9][0-9]{0,4})(\\.[0-9]{1,2})?\\z") {
344 return try fail("invalid enthalpy " + s)
345 }
346 let p = (s.hasPrefix("-") ? String(s.dropFirst()) : s).components(separatedBy: ".")
347 return (s.hasPrefix("-") ? -1 : 1) * (Int(p[0])! * 100 + (p.count == 1 ? 0 : Int(p[1] + String(
348 repeating: "0",
349 count: 2 - p[1].count
350 ))!))
351}
352func matrix(_ list: [Species], _ signs: [Int]) -> ([String], [[Int]]) {
353 var all: Composition = [:]
354 for s in list {
355 all = add(all, s.formula.c)
356 }
357 let labels = order(all) + ["charge"]
358 return (labels, labels.map {
359 e in list.enumerated().map {
360 i, s in signs[i] * (e == "charge" ? s.formula.q : s.formula.c[e, default: 0])
361 }
362 })
363}
364func nullspace(_ rows: [[Int]], _ n: Int) -> (Int, [[Int]]) {
365 var a = rows.map {
366 $0.map {
367 Fraction($0)
368 }
369 }
370 var piv: [Int] = [], free: [Int] = []
371 for j in 0..<n {
372 let r = piv.count
373 guard let i = (r..<a.count).first(where: {
374 a[$0][j].n != 0
375 }) else {
376 free.append(j)
377 continue
378 }
379 a.swapAt(r, i)
380 let t = a[r][j]
381 a[r] = a[r].map {
382 $0 / t
383 }
384 for k in a.indices where k != r && a[k][j].n != 0 {
385 let f = a[k][j]
386 for z in 0..<n {
387 a[k][z] = a[k][z] - f * a[r][z]
388 }
389 }
390 piv.append(j)
391 }
392 var basis: [[Int]] = []
393 for f in free {
394 var v = Array(repeating: Fraction(0), count: n)
395 v[f] = Fraction(1)
396 for (i, j) in piv.enumerated() {
397 v[j] = Fraction(-a[i][f].n, a[i][f].d)
398 }
399 var l = 1
400 for x in v {
401 l = l / gcd(l, x.d) * x.d
402 }
403 let b = v.map {
404 $0.n * (l / $0.d)
405 }
406 let g = b.reduce(0, gcd)
407 basis.append(b.map {
408 $0 / g
409 })
410 }
411 return (piv.count, basis)
412}
413func coefficients(_ list: [Species], _ signs: [Int], _ required: Int? = nil) throws -> [Int] {
414 let (_, rows) = matrix(list, signs)
415 let (_, basis) = nullspace(rows, list.count)
416 if basis.count >= 2 {
417 throw ChemError.message("AMBIGUOUS (nullity \(basis.count))")
418 }
419 if basis.isEmpty {
420 throw ChemError.message("IMPOSSIBLE")
421 }
422 var v = basis[0]
423 if v[0] < 0 {
424 v = v.map { -$0
425 }
426 }
427 if v.prefix(required ?? list.count).contains(where: {
428 $0 <= 0
429 }) {
430 throw ChemError.message("IMPOSSIBLE")
431 }
432 return v
433}

446func ratios(_ a: Composition) -> [String] {
447 let es = order(a)
448 var f = es[0]
449 for e in es where a[e]! * W[f]! < a[f]! * W[e]! {
450 f = e
451 }
452 let out = es.map {
453 "\($0) \(dec(1000 * a[$0]! * W[f]! / (W[$0]! * a[f]!), 3))"
454 }
455 for k in 1...12 {
456 var c: Composition = [:]
457 var ok = true
458 for e in es {
459 let n = k * a[e]! * W[f]!, d = W[e]! * a[f]!, i = (2 * n + d) / (2 * d)
460 c[e] = i
461 if 20 * abs(n - i * d) > d {
462 ok = false
463 }
464 }
465 if ok {
466 return out + ["\(hill(c)) (k = \(k))"]
467 }
468 }
469 return out + ["no simple ratio"]
470}
471func group(_ f: String, _ k: Int) -> String {
472 let c = String(f.split(separator: "^")[0])
473 if k == 1 {
474 return c
475 }
476 let chars = Array(c)
477 var depth = 0, last = -1
478 if chars[0] == "[" {
479 for i in chars.indices {
480 if chars[i] == "[" {
481 depth += 1
482 }
483 if chars[i] == "]" {
484 depth -= 1
485 if depth == 0 {
486 last = i
487 break
488 }
489 }
490 }
491 }
492 return (matches(c, "\\A[A-Z][a-z]?\\z") || last == chars.count - 1 ? c : "(\(c))") + String(k)
493}

520class Chemistry {
521 var heats: Composition = [:]
522 func execute(_ line: String) throws -> [String] {
523 let words = line.components(separatedBy: " "), cmd = words[0], a = Array(words.dropFirst()), n = a.count
524 guard let e = counts[cmd] else {
525 return try fail("unknown command " + cmd)
526 }
527 if (e >= 0 && n != e) || (e == -1 && n < 1) || (e == -2 && (n < 2 || n % 2 != 0)) || (e == -3 && n < 2) {
528 return try fail("wrong argument count for " + cmd)
529 }
530 if cmd == "ELEMENT" {
531 guard let w = W[a[0]] else {
532 return try fail("unknown element " + a[0])
533 }
534 return ["\(a[0]) (Z = \(symbols.firstIndex(of: a[0])! + 1)) \(dec(w, 3)) u"]
535 }
536 if cmd == "HEATS" {
537 return heats.isEmpty ? ["no enthalpies"] : heats.keys.sorted().map {
538 "\($0) \(ent(heats[$0]!)) kJ/mol"
539 }
540 }
541 if cmd == "HEAT" {
542 _ = try species(a[0])
543 let h = try enthalpy(a[1])
544 let text = "dHf(\(a[0])) = \(ent(h)) kJ/mol" + (heats[a[0]] == nil ? "" : " (was \(ent(heats[a[0]]!)))")
545 heats[a[0]] = h
546 return [text]
547 }
548 if cmd == "DERIVE" {
549 var b: Composition = [:]
550 for i in stride(from: 0, to: n, by: 2) {
551 let s = a[i]
552 if W[s] == nil {
553 return try fail("unknown element " + s)
554 }
555 if b[s] != nil {
556 return try fail("duplicate element " + s)
557 }
558 let q = try quantity(a[i + 1])
559 if q > 100000 {
560 return try fail("invalid percentage " + a[i + 1])
561 }
562 b[s] = q
563 }
564 let t = b.values.reduce(0,+)
565 if abs(t - 100000) > 1000 {
566 return try fail("percentages sum to " + dec(t, 3))
567 }
568 return ratios(b)
569 }
570 if cmd == "COMBUST" {
571 let qs = try a.map {
572 try quantity($0)
573 }
574 let sample = qs[0], c = qs[1] * 12011 / 44009, h = qs[2] * 2016 / 18015, o = sample - c - h
575 if c == 0 || h == 0 || 100 * o < -sample {
576 return try fail("inconsistent masses")
577 }
578 var b = ["C": c, "H": h]
579 if 100 * o > sample {
580 b["O"] = o
581 }
582 return order(b).map {
583 "mass \($0) \(dec(b[$0]!, 3)) g"
584 } + ratios(b)
585 }
586 if ["PARSE", "MASS", "PERCENT", "EMPIRICAL", "DBE", "OXSTATE", "MOLES", "GRAMS", "MOLECULAR", "SALT"]
587 .contains(cmd)
588 {
589 let f = try formula(a[0])
590 let c = f.c, q = f.q, m = f.molarMass, es = order(c)
591 if cmd == "PARSE" {
592 return ["\(hill(c, q)) (\(es.map { "\($0) \(c[$0]!)" }.joined(separator: ", ")); charge \(signed(q)))"]
593 }
594 if cmd == "MASS" {
595 return ["M(\(a[0])) = \(dec(m, 3)) g/mol"]
596 }
597 if cmd == "PERCENT" {
598 return es.map {
599 "\($0) \(dec(10000 * c[$0]! * W[$0]! / m, 2))%"
600 }
601 }
602 if cmd == "EMPIRICAL" {
603 let g = c.values.reduce(0, gcd)
604 return ["\(hill(c.mapValues { $0 / g })) (n = \(g))"]
605 }
606 if cmd == "DBE" {
607 if q != 0 {
608 return try fail("charged formula " + a[0])
609 }
610 var t = 2
611 for s in es {
612 guard let v = valence[s] else {
613 return try fail("no valence for " + s)
614 }
615 t += c[s]! * (v - 2)
616 }
617 return ["DBE = \(t < 0 ? "-" : "")\(abs(t) / 2).\(t % 2 != 0 ? 5 : 0)"]
618 }
619 if cmd == "OXSTATE" {
620 var st: [String: Fraction] = [:]
621 var last = ""
622 func assign(_ e: String, _ n: Int) {
623 st[e] = Fraction(n)
624 last = e
625 }
626 if es.count == 1 {
627 st[es[0]] = Fraction(q, c[es[0]]!)
628 } else {
629 if c["F"] != nil {
630 assign("F", -1)
631 }
632 for s in es {
633 if let x = metals[s] {
634 assign(s, x)
635 }
636 }
637 if c["H"] != nil {

639 $0 == "H" || metals[$0] != nil

642 if c["O"] != nil {
643 assign("O", -2)
644 } else if c["F"] == nil {
645 for s in ["Cl", "Br", "I"] where c[s] != nil {
646 assign(s, -1)
647 break
648 }
649 }
650 let u = es.filter {
651 st[$0] == nil
652 }
653 if u.count > 1 {
654 return ["indeterminate: " + u.joined(separator: " ")]
655 }
656 let total = st.reduce(0) {
657 $0 + c[$1.key]! * $1.value.n
658 }
659 if !u.isEmpty {
660 st[u[0]] = Fraction(q - total, c[u[0]]!)
661 } else if total != q {
662 st[last] = Fraction(q - total + c[last]! * st[last]!.n, c[last]!)
663 }
664 }
665 return [es.map {
666 "\($0) \(st[$0]!)"
667 }.joined(separator: ", ")]
668 }
669 if cmd == "SALT" {
670 let b = try formula(a[1])
671 if q <= 0 {
672 return try fail("not a cation " + a[0])

674 if b.q >= 0 {
675 return try fail("not an anion " + a[1])
676 }
677 for s in a where s.contains(".") {
678 return try fail("hydrate ion " + s)
679 }

681 return ["\(group(a[0], x))\(group(a[1], y)) (\(hill(z)), \(dec(mass(z), 3)) g/mol)"]
682 }
683 if cmd == "MOLECULAR" && q != 0 {
684 return try fail("charged formula " + a[0])
685 }
686 let x = try quantity(a[1])
687 if cmd == "MOLES" {
688 return ["\(dec(x, 3)) g \(a[0]) = \(dec(x * 1000000 / m, 6)) mol"]
689 }
690 if cmd == "GRAMS" {
691 return ["\(dec(x, 3)) mol \(a[0]) = \(dec(x * m / 1000, 3)) g"]
692 }
693 let k = (2 * x + m) / (2 * m)
694 return k == 0 || 10 * abs(x - k * m) > m ? ["no integer multiple (ratio \(dec(1000 * x / m, 3)))"] :
695 ["\(hill(scale(c, k))) (n = \(k), \(dec(k * m, 3)) g/mol)"]
696 }
697 if cmd == "REDOX" || cmd == "HALF" {

699 return try fail("invalid medium " + a[0])

702 for s in eq.list where ["H^+", "OH^-", "H2O"].contains(s.f) {
703 return try fail("reserved species " + s.name)
704 }
705 var aux = [try species(a[0] == "acidic" ? "H^+" : "OH^-"), try species("H2O")]
706 if cmd == "HALF" {
707 aux.append(Species(f: "e^-", state: "", coefficient: 1, formula: Formula(c: [:], q: -1)))
708 }
709 let list = eq.list + aux, v = try coefficients(
710 list,
711 eq.signs + Array(repeating: 1, count: aux.count),
712 eq.list.count
713 )
714 var left: [String] = [], right: [String] = []
715 for i in eq.list.indices {
716 if i < eq.r {
717 left.append(term(list[i].name, v[i]))
718 } else {
719 right.append(term(list[i].name, v[i]))
720 }
721 }
722 for i in eq.list.count..<list.count {
723 if v[i] > 0 {
724 left.append(term(list[i].name, v[i]))
725 } else if v[i] < 0 {
726 right.append(term(list[i].name, -v[i]))
727 }
728 }
729 var out = [left.joined(separator: " + ") + " -> " + right.joined(separator: " + ")]
730 if cmd == "HALF" {
731 out.append(v.last! > 0 ? "reduction" : v.last! < 0 ? "oxidation" : "no electron transfer")
732 }
733 return out
734 }
735 var eqWords = a, rest: [String] = []
736 if ["STOICH", "LIMITING", "YIELD"].contains(cmd) {
737 if a.filter({
738 $0 == ":"
739 }).count != 1 {
740 return try fail("invalid separator")
741 }
742 let i = a.firstIndex(of: ":")!
743 eqWords = Array(a.prefix(i))
744 rest = Array(a.dropFirst(i + 1))
745 if (cmd == "STOICH" && rest.count != 2) || (cmd == "YIELD" && rest.count != 4) {
746 return try fail("wrong argument count for " + cmd)
747 }
748 }
749 let eq = try equation(eqWords, cmd == "CHECK")
750 let list = eq.list, r = eq.r
751 if cmd == "MATRIX" {
752 let (labels, rows) = matrix(list, eq.signs)
753 let (rank, basis) = nullspace(rows, list.count)
754 return ["columns: " + list.map {
755 $0.name
756 }.joined(separator: " ")] + rows.enumerated().map {
757 "\(labels[$0.offset]): " + $0.element.map {
758 String($0)
759 }.joined(separator: " ")
760 } + ["rank \(rank), nullity \(basis.count)"] + basis.map {
761 "null: " + $0.map {
762 String($0)
763 }.joined(separator: " ")
764 }
765 }
766 if cmd == "CHECK" {
767 let (labels, rows) = matrix(list, eq.signs)
768 var diff: [String] = []
769 for (i, row) in rows.enumerated() {
770 let l = (0..<r).reduce(0) {
771 $0 + row[$1] * list[$1].coefficient
772 }, rr = -(r..<list.count).reduce(0) {
773 $0 + row[$1] * list[$1].coefficient
774 }
775 if l != rr {
776 diff
777 .append(
778 "\(labels[i]): \(labels[i] == "charge" ? signed(l) : String(l)) != \(labels[i] == "charge" ? signed(rr) : String(rr))"
779 )
780 }
781 }
782 if !diff.isEmpty {
783 return ["not balanced"] + diff
784 }
785 let g = list.map {
786 $0.coefficient
787 }.reduce(0, gcd)
788 return [g == 1 ? "balanced" : "balanced, not in lowest terms (gcd \(g))"]
789 }
790 var target = -1, product = -1
791 var qs: [Int] = []
792 if cmd == "DELTAH" {
793 for s in list where heats[s.name] == nil {
794 return try fail("no enthalpy for " + s.name)
795 }
796 }
797 if cmd == "STOICH" {
798 target = list.firstIndex {
799 $0.name == rest[0]
800 }
801 ?? -1
802 if target < 0 {
803 return try fail("species \(rest[0]) not in equation")
804 }
805 qs = [try quantity(rest[1])]
806 }
807 if cmd == "LIMITING" {
808 if rest.count != r {
809 return try fail("expected \(r) masses, got \(rest.count)")
810 }
811 qs = try rest.map {
812 try quantity($0)
813 }
814 }
815 if cmd == "YIELD" {
816 target = (0..<r).first {
817 list[$0].name == rest[0]
818 }
819 ?? -1
820 if target < 0 {
821 return try fail(rest[0] + " is not a reactant")
822 }
823 product = (r..<list.count).first {
824 list[$0].name == rest[2]
825 }
826 ?? -1
827 if product < 0 {
828 return try fail(rest[2] + " is not a product")
829 }
830 qs = [try quantity(rest[1]), try quantity(rest[3])]
831 }
832 let v = try coefficients(list, eq.signs)
833 var out = [notation(eq, v)]
834 if cmd == "BALANCE" {
835 return out
836 }
837 if cmd == "DELTAH" {
838 let h = list.indices.reduce(0) {
839 $0 - eq.signs[$1] * v[$1] * heats[list[$1].name]!
840 }
841 return out + ["dH = \(ent(h)) kJ (\(h < 0 ? "exothermic" : h > 0 ? "endothermic" : "thermoneutral"))"]
842 }
843 if cmd == "STOICH" {
844 let d = v[target] * list[target].molarMass
845 return out + list.enumerated().map {
846 i, s in
847 "\(s.name) \(dec(qs[0] * v[i] * 1000000 / d, 6)) mol \(dec(qs[0] * v[i] * s.molarMass / d, 3)) g" +
848 gas(
849 s,
850 qs[0] * v[i],
851 d
852 )
853 }
854 }
855 if cmd == "YIELD" {
856 let t = qs[0] * v[product] * list[product].molarMass / (v[target] * list[target].molarMass)
857 return out + [
858 "theoretical \(dec(t, 3)) g",
859 "actual \(dec(qs[1], 3)) g",
860 t == 0 ? "yield undefined" : "yield \(dec(qs[1] * 10000 / t, 2))%"
861 ]
862 }
863 var l = 0
864 for i in 1..<r where qs[i] * v[l] * list[l].molarMass < qs[l] * v[i] * list[i].molarMass {
865 l = i
866 }
867 let d = v[l] * list[l].molarMass
868 out += ["limiting " + list[l].name, "extent \(dec(qs[l] * 1000000 / d, 6)) mol"]
869 for i in r..<list.count {
870 out.append("product \(list[i].name) \(dec(qs[l] * v[i] * list[i].molarMass / d, 3)) g" + gas(
871 list[i],
872 qs[l] * v[i],
873 d
874 ))
875 }
876 for i in 0..<r where i != l {
877 out.append("remaining \(list[i].name) \(dec((qs[i] * d - qs[l] * v[i] * list[i].molarMass) / d, 3)) g")
878 }
879 return out
880 }
881}
