67static Int gcd_int(Int a, Int b) {
68 if (a < 0)
69 a = -a;
70 if (b < 0)
71 b = -b;
72 while (b) {
73 Int r = a % b;
74 a = b;
75 b = r;
76 }
77 return a;
78}

80static Rational rational(Int n, Int d) {
81 if (d < 0) {
82 n = -n;
83 d = -d;
84 }
85 Int g = gcd_int(n, d);
86 return (Rational){n / g, d / g};
87}

89static Rational subtract(Rational a, Rational b) {
90 return rational(a.n * b.d - b.n * a.d, a.d * b.d);
91}
92
93static Rational multiply(Rational a, Rational b) {
94 Int g = gcd_int(a.n, b.d), h = gcd_int(b.n, a.d);
95 return rational((a.n / g) * (b.n / h), (a.d / h) * (b.d / g));
96}
97
98static Rational divide(Rational a, Rational b) {
99 return multiply(a, rational(b.d, b.n));
100}

113static int hill_order(const Int *atoms, int *order) {
114 int n = 0;
115 for (int i = 0; i < 54; ++i)
116 if (atoms[i])
117 order[n++] = i;
118 qsort(order, (size_t)n, sizeof *order, name_compare);
119 if (atoms[5]) {
120 int sorted[54], k = 0;
121 sorted[k++] = 5;
122 if (atoms[0])
123 sorted[k++] = 0;
124 for (int i = 0; i < n; ++i)
125 if (order[i] != 5 && order[i] != 0)
126 sorted[k++] = order[i];
127 memcpy(order, sorted, (size_t)n * sizeof *order);
128 }
129 return n;
130}

175static Int number(Parser *p) {
176 if (!isdigit((unsigned char)p->text[p->pos]))
177 return 1;
178 if (p->text[p->pos] == '0')
179 p->valid = 0;
180 Int n = 0;
181 while (isdigit((unsigned char)p->text[p->pos]))
182 n = n * 10 + p->text[p->pos++] - '0';
183 return n;
184}

186static void add_atoms(Formula *a, Formula b, Int factor) {
187 for (int i = 0; i < 54; ++i) {
188 Int value = b.atoms[i] * factor + a->atoms[i];
189 a->atoms[i] = value > 1000000000 ? 1000000000 : value;
190 }
191}

193static Formula sequence(Parser *p) {
194 Formula result = {0};
195 int count = 0;
196 for (;;) {
197 char c = p->text[p->pos];
198 Formula item = {0};
199 if (c >= 'A' && c <= 'Z') {
200 char sym[3] = {c, 0, 0};
201 ++p->pos;
202 c = p->text[p->pos];
203 if (c >= 'a' && c <= 'z')
204 sym[1] = p->text[p->pos++];
205 int e = element(sym);
206 if (e >= 0)
207 item.atoms[e] = 1;
208 else if (!p->unknown[0])
209 strcpy(p->unknown, sym);
210 } else if (c == '(' || c == '[') {
211 ++p->pos;
212 item = sequence(p);
213 if (p->text[p->pos] != (c == '(' ? ')' : ']')) {
214 p->valid = 0;
215 break;
216 }
217 ++p->pos;
218 } else
219 break;
220 ++count;
221 Int k = number(p);
222 add_atoms(&result, item, k);
223 }
224 if (!count)
225 p->valid = 0;
226 return result;
227}

229static Formula read_formula(const char *text, const char *word) {
230 Parser p = {text, 0, 1, {0}};
231 Formula result = sequence(&p);
232 while (text[p.pos] == '.') {
233 ++p.pos;
234 Int k = number(&p);
235 Formula part = sequence(&p);
236 add_atoms(&result, part, k);
237 }
238 if (text[p.pos] == '^') {
239 ++p.pos;
240 Int k = number(&p);
241 if (text[p.pos] == '+' || text[p.pos] == '-') {
242 result.charge = text[p.pos++] == '+' ? k : -k;
243 } else
244 p.valid = 0;
245 }
246 if (!p.valid || text[p.pos])
247 fail("invalid formula %s", word);
248 if (p.unknown[0])
249 fail("unknown element %s", p.unknown);
250 return result;
251}

260static Species read_species(const char *word, int coefficients) {
261 Species s = {0};
262 char text[128];
263 strcpy(text, word);
264 const char *states[] = {"(s)", "(l)", "(g)", "(aq)"};
265 size_t n = strlen(text);
266 for (int i = 0; i < 4; ++i) {
267 size_t k = strlen(states[i]);
268 if (n >= k && !strcmp(text + n - k, states[i])) {
269 strcpy(s.state, states[i]);
270 text[n - k] = 0;
271 break;
272 }
273 }
274 char *core = text;
275 s.coefficient = 1;
276 if (coefficients && isdigit((unsigned char)*core)) {
277 if (*core == '0')
278 fail("invalid formula %s", word);
279 s.coefficient = strtoll(core, &core, 10);
280 }
281 s.formula = read_formula(core, word);
282 strcpy(s.core, core);
283 snprintf(s.name, sizeof s.name, "%s%s", core, s.state);
284 return s;
285}

287static Equation read_equation(char **args, int n, int coefficients) {
288 Equation e = {0};
289 int arrow = -1;
290 if (n < 3 || !(n % 2))
291 fail("malformed equation");
292 for (int i = 0; i < n; ++i) {
293 if (i % 2) {
294 if (!strcmp(args[i], "->")) {
295 if (arrow >= 0)
296 fail("malformed equation");
297 arrow = i;
298 } else if (strcmp(args[i], "+"))
299 fail("malformed equation");
300 } else if (!strcmp(args[i], "+") || !strcmp(args[i], "->"))
301 fail("malformed equation");
302 }
303 if (arrow < 0)
304 fail("malformed equation");
305 e.reactants = (arrow + 1) / 2;
306 for (int i = 0; i < n; i += 2) {
307 e.species[e.count] = read_species(args[i], coefficients);
308 e.signs[e.count] = i < arrow ? 1 : -1;
309 ++e.count;
310 }
311 return e;
312}

314static Int scaled_value(const char *word, int whole_digits, int fraction_digits, int signed_value,
315 int positive) {
316 const char *p = word;
317 int negative = signed_value && *p == '-';
318 if (negative)
319 ++p;
320 const char *start = p;
321 Int value = 0;
322 while (isdigit((unsigned char)*p)) {
323 if (p - start < whole_digits)
324 value = value * 10 + *p - '0';
325 ++p;
326 }
327 int n = (int)(p - start), valid = n >= 1 && n <= whole_digits && (n == 1 || *start != '0');
328 int f = 0;
329 if (*p == '.') {
330 ++p;
331 start = p;
332 while (isdigit((unsigned char)*p)) {
333 if (f < fraction_digits)
334 value = value * 10 + *p - '0';
335 ++p;
336 ++f;
337 }
338 if (p == start || f > fraction_digits)
339 valid = 0;
340 }
341 if (*p)
342 valid = 0;
343 while (f++ < fraction_digits)
344 value *= 10;
345 if (!valid || (positive && !value))
346 fail("invalid %s %s", signed_value ? "enthalpy" : "quantity", word);
347 return negative ? -value : value;
348}

354static int composition_matrix(Equation e, Int matrix[55][13], int *labels) {
355 Formula union_formula = {0};
356 for (int j = 0; j < e.count; ++j)
357 for (int i = 0; i < 54; ++i)
358 if (e.species[j].formula.atoms[i])
359 union_formula.atoms[i] = 1;
360 int m = hill_order(union_formula.atoms, labels);
361 labels[m] = -1;
362 for (int i = 0; i <= m; ++i)
363 for (int j = 0; j < e.count; ++j)
364 matrix[i][j] = e.signs[j] * (i == m ? e.species[j].formula.charge
365 : e.species[j].formula.atoms[labels[i]]);
366 return m + 1;
367}

369static int null_space(Int matrix[55][13], int m, int n, Int basis[13][13], int *rank) {
370 Rational a[55][13];
371 int pivots[13], free_columns[13], free_count = 0, r = 0;
372 for (int i = 0; i < m; ++i)
373 for (int j = 0; j < n; ++j)
374 a[i][j] = rational(matrix[i][j], 1);
375 for (int j = 0; j < n; ++j) {
376 int p = r;
377 while (p < m && !a[p][j].n)
378 ++p;
379 if (p == m) {
380 free_columns[free_count++] = j;
381 continue;
382 }
383 for (int k = 0; k < n; ++k) {
384 Rational t = a[r][k];
385 a[r][k] = a[p][k];
386 a[p][k] = t;
387 }
388 Rational divisor = a[r][j];
389 for (int k = 0; k < n; ++k)
390 a[r][k] = divide(a[r][k], divisor);
391 for (int i = 0; i < m; ++i)
392 if (i != r) {
393 Rational factor = a[i][j];
394 for (int k = 0; k < n; ++k)
395 a[i][k] = subtract(a[i][k], multiply(factor, a[r][k]));
396 }
397 pivots[r++] = j;
398 }
399 for (int b = 0; b < free_count; ++b) {
400 Rational values[13];
401 for (int j = 0; j < n; ++j)
402 values[j] = rational(j == free_columns[b], 1);
403 for (int i = 0; i < r; ++i)
404 values[pivots[i]] = rational(-a[i][free_columns[b]].n, a[i][free_columns[b]].d);
405 Int multiple = 1, divisor = 0;
406 for (int j = 0; j < n; ++j)
407 multiple = multiple / gcd_int(multiple, values[j].d) * values[j].d;
408 for (int j = 0; j < n; ++j) {
409 basis[b][j] = values[j].n * (multiple / values[j].d);
410 divisor = gcd_int(divisor, basis[b][j]);
411 }
412 for (int j = 0; j < n; ++j)
413 basis[b][j] /= divisor;
414 }
415 *rank = r;
416 return free_count;
417}

419static void coefficients(Equation e, int required, Int *values) {
420 Int matrix[55][13], basis[13][13];
421 int labels[55], rank;
422 int m = composition_matrix(e, matrix, labels);
423 int d = null_space(matrix, m, e.count, basis, &rank);
424 if (d != 1) {
425 if (d)
426 append("AMBIGUOUS (nullity %d)\n", d);
427 else
428 append("IMPOSSIBLE\n");
429 longjmp(command_end, 1);
430 }
431 Int sign = basis[0][0] < 0 ? -1 : 1;
432 for (int i = 0; i < e.count; ++i)
433 values[i] = basis[0][i] * sign;
434 for (int i = 0; i < required; ++i)
435 if (values[i] <= 0) {
436 append("IMPOSSIBLE\n");
437 longjmp(command_end, 1);
438 }
439}

465static void ratio_report(Int *amounts) {
466 int order[54], n = hill_order(amounts, order), f = order[0];
467 for (int i = 1; i < n; ++i) {
468 int e = order[i];
469 if (amounts[e] * weights[f] < amounts[f] * weights[e])
470 f = e;
471 }
472 for (int i = 0; i < n; ++i) {
473 int e = order[i];
474 append("%s ", symbols[e]);
475 decimal(1000 * amounts[e] * weights[f] / (weights[e] * amounts[f]), 3);
476 append("\n");
477 }
478 for (int k = 1; k <= 12; ++k) {
479 Formula counts = {0};
480 int valid = 1;
481 for (int i = 0; i < n; ++i) {
482 int e = order[i];
483 Int a = k * amounts[e] * weights[f], b = weights[e] * amounts[f];
484 counts.atoms[e] = (2 * a + b) / (2 * b);
485 if (20 * llabs(a - counts.atoms[e] * b) > b)
486 valid = 0;
487 }
488 if (valid) {
489 hill(counts, 0);
490 append(" (k = %d)\n", k);
491 return;
492 }
493 }
494 append("no simple ratio\n");
495}

505static void oxidation(Formula f) {
506 int order[54], n = hill_order(f.atoms, order), assigned[54] = {0}, last = -1;
507 Rational values[54];
508 for (int i = 0; i < 54; ++i)
509 values[i] = rational(0, 1);
510 if (n == 1) {
511 assigned[order[0]] = 1;
512 values[order[0]] = rational(f.charge, f.atoms[order[0]]);
513 } else {
514 int e = 8;
515 if (f.atoms[e]) {
516 assigned[e] = 1;
517 values[e] = rational(-1, 1);
518 last = e;
519 }
520 for (int i = 0; i < n; ++i) {
521 e = order[i];
522 int v = fixed_metal(e);
523 if (v) {
524 assigned[e] = 1;
525 values[e] = rational(v, 1);
526 last = e;
527 }
528 }
529 if (f.atoms[0]) {
530 int only_metals = 1;
531 for (int i = 0; i < n; ++i)
532 if (order[i] && !fixed_metal(order[i]))
533 only_metals = 0;
534 assigned[0] = 1;
535 values[0] = rational(only_metals ? -1 : 1, 1);
536 last = 0;
537 }
538 if (f.atoms[7]) {
539 assigned[7] = 1;
540 values[7] = rational(-2, 1);
541 last = 7;
542 }
543 if (!f.atoms[7] && !f.atoms[8]) {
544 int halogens[] = {16, 34, 52};
545 for (int i = 0; i < 3; ++i)
546 if (f.atoms[halogens[i]]) {
547 e = halogens[i];
548 assigned[e] = 1;
549 values[e] = rational(-1, 1);
550 last = e;
551 break;
552 }
553 }
554 int unknown = 0, missing = -1;
555 Int sum = 0;
556 for (int i = 0; i < n; ++i) {
557 e = order[i];
558 if (!assigned[e]) {
559 ++unknown;
560 missing = e;
561 } else
562 sum += f.atoms[e] * values[e].n;
563 }
564 if (unknown >= 2) {
565 append("indeterminate:");
566 for (int i = 0; i < n; ++i)
567 if (!assigned[order[i]])
568 append(" %s", symbols[order[i]]);
569 append("\n");
570 return;
571 }
572 if (unknown == 1)
573 values[missing] = rational(f.charge - sum, f.atoms[missing]);
574 else if (sum != f.charge)
575 values[last] = rational(f.charge - sum + f.atoms[last] * values[last].n, f.atoms[last]);
576 }
577 for (int i = 0; i < n; ++i) {
578 if (i)
579 append(", ");
580 append("%s ", symbols[order[i]]);
581 signed_rational(values[order[i]]);
582 }
583 append("\n");
584}

586static void ion_part(const char *formula, Int k) {
587 char core[128];
588 strcpy(core, formula);
589 char *charge = strchr(core, '^');
590 if (charge)
591 *charge = 0;
592 int single = core[0] >= 'A' && core[0] <= 'Z' &&
593 (core[1] == 0 || (core[1] >= 'a' && core[1] <= 'z' && core[2] == 0));
594 int whole_bracket = 0;
595 if (core[0] == '[') {
596 int depth = 0;
597 for (int i = 0; core[i]; ++i) {
598 if (core[i] == '[')
599 ++depth;
600 if (core[i] == ']' && !--depth) {
601 whole_bracket = core[i + 1] == 0;
602 break;
603 }
604 }
605 }
606 if (k == 1)
607 append("%s", core);
608 else if (single || whole_bracket)
609 append("%s%lld", core, (long long)k);
610 else
611 append("(%s)%lld", core, (long long)k);
612}

640static void execute(char **words, int count) {
641 const char *cmd = words[0];
642 char **a = words + 1;
643 int n = count - 1;
644 const char *commands[] = {"ELEMENT", "PARSE", "MASS", "PERCENT", "EMPIRICAL", "DBE",
645 "OXSTATE", "MOLES", "GRAMS", "MOLECULAR", "SALT", "DERIVE",
646 "COMBUST", "BALANCE", "MATRIX", "CHECK", "REDOX", "HALF",
647 "STOICH", "LIMITING", "YIELD", "HEAT", "DELTAH", "HEATS"};
648 int id = -1;
649 for (int i = 0; i < 24; ++i)
650 if (!strcmp(cmd, commands[i]))
651 id = i;
652 if (id < 0)
653 fail("unknown command %s", cmd);
654 int valid = 1;
655 if (id <= 6)
656 valid = n == 1;
657 else if (id <= 10 || id == 21)
658 valid = n == 2;
659 else if (id == 11)
660 valid = n >= 2 && n % 2 == 0;
661 else if (id == 12)
662 valid = n == 3;
663 else if ((id >= 13 && id <= 15) || id == 22)
664 valid = n >= 1;
665 else if (id == 16 || id == 17)
666 valid = n >= 2;
667 else if (id == 23)
668 valid = n == 0;
669 if (!valid)
670 fail("wrong argument count for %s", cmd);
671 if (id == 0) {
672 int e = element(a[0]);
673 if (e < 0)
674 fail("unknown element %s", a[0]);
675 append("%s (Z = %d) ", a[0], e + 1);
676 decimal(weights[e], 3);
677 append(" u\n");
678 } else if (id >= 1 && id <= 10) {
679 Formula f = read_formula(a[0], a[0]);
680 Int mass = molar_mass(f);
681 int order[54], m = hill_order(f.atoms, order);
682 if (id == 1) {
