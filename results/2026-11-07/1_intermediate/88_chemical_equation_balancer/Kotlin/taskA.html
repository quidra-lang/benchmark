91 fun sequence():Composition {
92 var c:Composition=emptyMap();
93 var items=0;
94 while(true){
95 val ch=peek();
96 val item:Composition;
97 if(ch in 'A'..'Z'){
98 var e=text[p++].toString();
99 if(peek() in 'a'..'z')e+=text[p++];
100 seen+=e;
101 item=mapOf(e to 1L)
102 }else if(ch=='('||ch=='['){
103 p++;
104 item=sequence();
105 if(peek()!=(if(ch=='(')')'else ']'))throw IllegalArgumentException();
106 p++
107 }else break;
108 c=add(c,item,number());
109 items++
110 };
111 if(items==0)throw IllegalArgumentException();
112 return c
113 }
114 fun parse():Formula {
115 var c=sequence();
116 while(peek()=='.'){
117 p++;
118 val k=number();
119 c=add(c,sequence(),k)
120 };
121 var q=0L;
122 if(peek()=='^'){
123 p++;
124 val k=number();
125 val ch=peek();
126 if(ch!='+'&&ch!='-')throw IllegalArgumentException();
127 p++;
128 q=if(ch=='+')k else -k
129 };
130 if(p!=text.length)throw IllegalArgumentException();
131 return Formula(c,q)
132 }

535 return out+listOf("theoretical ${dec(t,3)} g","actual ${dec(qs[1],3)} g",if(t==0L)"yield undefined"else "yield ${dec(qs[1]*10000/t,2)}%")
536 }
537 var l=0;
538 for(i in 1 until r)if(qs[i]*v[l]*list[l].mass<qs[l]*v[i]*list[i].mass)l=i;
539 val d=v[l]*list[l].mass;
540 out+="limiting ${list[l].name}";
541 out+="extent ${dec(qs[l]*1000000/d,6)} mol";
542 for(i in r until list.size)out+="product ${list[i].name} ${dec(qs[l]*v[i]*list[i].mass/d,3)} g"+gas(list[i],qs[l]*v[i],d);
543 for(i in 0 until r)if(i!=l)out+="remaining ${list[i].name} ${dec((qs[i]*d-qs[l]*v[i]*list[i].mass)/d,3)} g";
544 return out
545 }
