176 ordered = sorted(elements)

209class FormulaParser:
210 def __init__(self, text):
211 self.text = text
212 self.pos = 0
213 self.symbols = []
214
215 def peek(self):
216 return self.text[self.pos] if self.pos < len(self.text) else ""
217
218 def parse(self):
219 composition = self.sequence()
220 while self.peek() == ".":
221 self.pos += 1
222 multiplier = self.number()
223 add_scaled(composition, self.sequence(), multiplier)
224 charge = 0
225 if self.peek() == "^":
226 self.pos += 1
227 magnitude = self.number()
228 sign = self.peek()
229 if sign not in ("+", "-"):
230 raise ValueError
231 self.pos += 1
232 charge = magnitude if sign == "+" else -magnitude
233 if self.pos != len(self.text):
234 raise ValueError
235 return Formula(composition, charge)
236
237 def sequence(self):
238 composition = Counter()
239 while True:
240 char = self.peek()
241 if char.isupper():
242 item = Counter({self.symbol(): 1})
243 elif char == "(":
244 item = self.group(")")
245 elif char == "[":
246 item = self.group("]")
247 else:
248 break
249 add_scaled(composition, item, self.number())
250 if not composition:
251 raise ValueError
252 return composition
253
254 def symbol(self):
255 start = self.pos
256 self.pos += 1
257 if self.peek().islower():
258 self.pos += 1
259 symbol = self.text[start : self.pos]
260 self.symbols.append(symbol)
261 return symbol
262
263 def group(self, closing):
264 self.pos += 1
265 composition = self.sequence()
266 if self.peek() != closing:
267 raise ValueError
268 self.pos += 1
269 return composition
270
271 def number(self):
272 start = self.pos
273 while self.peek().isdigit():
274 self.pos += 1
275 digits = self.text[start : self.pos]
276 if not digits:
277 return 1
278 if digits.startswith("0"):
279 raise ValueError
280 return int(digits)

283def read_formula(text, word=None):
284 parser = FormulaParser(text)
285 try:
286 formula = parser.parse()
287 except ValueError:
288 raise ChemError(f"invalid formula {word or text}") from None
289 for symbol in parser.symbols:
290 if symbol not in WEIGHTS:
291 raise ChemError(f"unknown element {symbol}")
292 return formula

295def read_species(word, allow_coefficient=False):
296 state = next((suffix for suffix in STATES if word.endswith(suffix)), "")
297 text = word.removesuffix(state)
298 coefficient = 1
299 if allow_coefficient:
300 match = LEADING_DIGITS.match(text)
301 if match:
302 if text.startswith("0"):
303 raise ChemError(f"invalid formula {word}")
304 coefficient = int(match.group())
305 text = text[match.end() :]
306 return Species(text, read_formula(text, word), state, coefficient)

309def read_equation(words, allow_coefficients=False):
310 operators = words[1::2]
311 if (
312 len(words) % 2 == 0
313 or any(word not in ("+", "->") for word in operators)
314 or operators.count("->") != 1
315 or any(word in ("+", "->") for word in words[::2])
316 ):
317 raise ChemError("malformed equation")
318 species = [read_species(word, allow_coefficients) for word in words[::2]]
319 split = operators.index("->") + 1
320 return Equation(species[:split], species[split:])

342def composition_matrix(species, signs):
343 formulas = [item.formula for item in species]
344 elements = hill_order(
345 {element for formula in formulas for element in formula.composition}
346 )
347 rows = [
348 [sign * formula.composition[element] for formula, sign in zip(formulas, signs)]
349 for element in elements
350 ]
351 rows.append([sign * formula.charge for formula, sign in zip(formulas, signs)])
352 return elements + ["charge"], rows

355def null_space(rows, column_count):
356 matrix = [[Fraction(entry) for entry in row] for row in rows]
357 pivot_columns = []
358 free_columns = []
359 for column in range(column_count):
360 rank = len(pivot_columns)
361 pivot_row = next(
362 (i for i in range(rank, len(matrix)) if matrix[i][column] != 0), None
363 )
364 if pivot_row is None:
365 free_columns.append(column)
366 continue
367 matrix[rank], matrix[pivot_row] = matrix[pivot_row], matrix[rank]
368 pivot = matrix[rank][column]
369 matrix[rank] = [entry / pivot for entry in matrix[rank]]
370 for i, row in enumerate(matrix):
371 factor = row[column]
372 if i != rank and factor != 0:
373 matrix[i] = [
374 entry - factor * base for entry, base in zip(row, matrix[rank])
375 ]
376 pivot_columns.append(column)
377 basis = []
378 for free in free_columns:
379 values = [Fraction(0)] * column_count
380 values[free] = Fraction(1)
381 for row, pivot_column in enumerate(pivot_columns):
382 values[pivot_column] = -matrix[row][free]
383 multiple = lcm(*(value.denominator for value in values))
384 integers = [int(value * multiple) for value in values]
385 divisor = gcd(*integers)
386 basis.append([integer // divisor for integer in integers])
387 return len(pivot_columns), basis

390def determine_coefficients(species, signs, required):
391 _, rows = composition_matrix(species, signs)
392 _, basis = null_space(rows, len(species))
393 if len(basis) >= 2:
394 raise NoCoefficients(f"AMBIGUOUS (nullity {len(basis)})")
395 if not basis:
396 raise NoCoefficients("IMPOSSIBLE")
397 coefficients = basis[0]
398 if coefficients[0] < 0:
399 coefficients = [-value for value in coefficients]
400 if any(value <= 0 for value in coefficients[:required]):
401 raise NoCoefficients("IMPOSSIBLE")
402 return coefficients

445def ratio_report(amounts):
446 elements = hill_order(amounts)
447 moles = {
448 element: Fraction(amounts[element], WEIGHTS[element]) for element in elements
449 }
450 smallest = min(moles.values())
451 ratios = {element: moles[element] / smallest for element in elements}
452 lines = [
453 f"{element} {decimal(floor(1000 * ratios[element]), 3)}" for element in elements
454 ]
455 for multiplier in range(1, 13):
456 counts = {element: round(multiplier * ratios[element]) for element in elements}
457 if all(
458 abs(multiplier * ratios[element] - counts[element]) <= Fraction(1, 20)
459 for element in elements
460 ):
461 lines.append(f"{hill_notation(counts)} (k = {multiplier})")
462 return lines
463 lines.append("no simple ratio")
464 return lines

520def assign_oxidation_states(elements):
521 assigned = {}
522 if "F" in elements:
523 assigned["F"] = -1
524 for element in elements:
525 if element in FIXED_METALS:
526 assigned[element] = FIXED_METALS[element]
527 if "H" in elements:
528 only_metals = all(
529 element in FIXED_METALS for element in elements if element != "H"
530 )
531 assigned["H"] = -1 if only_metals else 1
532 if "O" in elements:
533 assigned["O"] = -2
534 elif "F" not in elements:
535 for halogen in ("Cl", "Br", "I"):
536 if halogen in elements:
537 assigned[halogen] = -1
538 break
539 return assigned

542def command_oxstate(args):
543 formula = read_formula(args[0])
544 composition = formula.composition
545 charge = formula.charge
546 elements = hill_order(composition)
547 if len(elements) == 1:
548 states = {elements[0]: Fraction(charge, composition[elements[0]])}
549 else:
550 assigned = assign_oxidation_states(elements)
551 undetermined = [element for element in elements if element not in assigned]
552 if len(undetermined) >= 2:
553 return ["indeterminate: " + " ".join(undetermined)]
554 total = sum(composition[element] * value for element, value in assigned.items())
555 states = {element: Fraction(value) for element, value in assigned.items()}
556 if undetermined:
557 unknown = undetermined[0]
558 states[unknown] = Fraction(charge - total, composition[unknown])
559 elif total != charge:
560 last = list(assigned)[-1]
561 others = total - composition[last] * assigned[last]
562 states[last] = Fraction(charge - others, composition[last])
563 return [", ".join(f"{element} {signed(states[element])}" for element in elements)]

595def closing_bracket(text):
596 depth = 0
597 for index, char in enumerate(text):
598 if char == "[":
599 depth += 1
600 elif char == "]":
601 depth -= 1
602 if depth == 0:
603 return index

618def command_salt(args):
619 cation_word, anion_word = args
620 cation = read_formula(cation_word)
621 anion = read_formula(anion_word)
622 if cation.charge <= 0:
623 raise ChemError(f"not a cation {cation_word}")
624 if anion.charge >= 0:
625 raise ChemError(f"not an anion {anion_word}")
626 for word in args:
627 if "." in word:
628 raise ChemError(f"hydrate ion {word}")
629 divisor = gcd(cation.charge, anion.charge)
630 cation_count = -anion.charge // divisor
631 anion_count = cation.charge // divisor
632 salt = Formula(scaled(cation.composition, cation_count), 0)
633 add_scaled(salt.composition, anion.composition, anion_count)
634 name = group_notation(cation_word, cation_count) + group_notation(
635 anion_word, anion_count
636 )
637 notation = hill_notation(salt.composition)
638 return [f"{name} ({notation}, {decimal(salt.molar_mass, 3)} g/mol)"]

693def command_check(args):
694 equation = read_equation(args, allow_coefficients=True)
695 species = equation.species
696 labels, rows = composition_matrix(species, equation.signs)
697 coefficients = [item.coefficient for item in species]
698 split = len(equation.reactants)
699 differences = []
700 for label, row in zip(labels, rows):
701 amounts = [value * entry for value, entry in zip(coefficients, row)]
702 left = sum(amounts[:split])
703 right = -sum(amounts[split:])
704 if left != right:
705 show = signed if label == "charge" else str
706 differences.append(f"{label}: {show(left)} != {show(right)}")
707 if differences:
708 return ["not balanced", *differences]

710 if divisor == 1:
711 return ["balanced"]
712 return [f"balanced, not in lowest terms (gcd {divisor})"]

715def balance_in_solution(args, with_electrons):
716 medium = args[0]
717 if medium not in SOLUTION_IONS:
718 raise ChemError(f"invalid medium {medium}")
719 equation = read_equation(args[1:])
720 species = equation.species
721 for item in species:
722 if item.formula_text in RESERVED_SPECIES:
723 raise ChemError(f"reserved species {item.name}")
724 auxiliary = [SOLUTION_IONS[medium], WATER]
725 if with_electrons:
726 auxiliary.append(ELECTRON)
727 coefficients = determine_coefficients(
728 species + auxiliary, equation.signs + [1] * len(auxiliary), len(species)
729 )
730 terms = [term(item.name, value) for item, value in zip(species, coefficients)]
731 split = len(equation.reactants)
732 left = terms[:split]
733 right = terms[split:]
734 for item, value in zip(auxiliary, coefficients[len(species) :]):
735 if value > 0:
736 left.append(term(item.name, value))
737 elif value < 0:
738 right.append(term(item.name, -value))
739 lines = [join_sides(left, right)]
740 if with_electrons:
741 electrons = coefficients[-1]
742 if electrons > 0:
743 lines.append("reduction")
744 elif electrons < 0:
745 lines.append("oxidation")
746 else:
747 lines.append("no electron transfer")
748 return lines

769 quantity = read_quantity(quantity_word)
770 coefficients = balance(equation)
771 divisor = coefficients[index] * species[index].formula.molar_mass
772 lines = [equation_notation(equation, coefficients)]
773 for item, value in zip(species, coefficients):
774 amount = quantity * value
775 moles = decimal(amount * 1000000 // divisor, 6)
776 grams = decimal(amount * item.formula.molar_mass // divisor, 3)
777 lines.append(
778 f"{item.name} {moles} mol {grams} g" + gas_volume(item, amount, divisor)
779 )
780 return lines

783def command_limiting(args):
784 equation_words, mass_words = split_at_separator(args)
785 equation = read_equation(equation_words)
786 reactants = equation.reactants
787 if len(mass_words) != len(reactants):
788 raise ChemError(f"expected {len(reactants)} masses, got {len(mass_words)}")
789 masses = [read_quantity(word) for word in mass_words]
790 coefficients = balance(equation)
791 reaction_masses = [
792 value * reactant.formula.molar_mass
793 for value, reactant in zip(coefficients, reactants)
794 ]
795 extents = [Fraction(mass, weight) for mass, weight in zip(masses, reaction_masses)]
796 limiting = extents.index(min(extents))
797 limiting_mass = masses[limiting]
798 divisor = reaction_masses[limiting]
799 lines = [
800 equation_notation(equation, coefficients),
801 f"limiting {reactants[limiting].name}",
802 f"extent {decimal(limiting_mass * 1000000 // divisor, 6)} mol",
803 ]
804 for product, value in zip(equation.products, coefficients[len(reactants) :]):
805 amount = limiting_mass * value
806 grams = decimal(amount * product.formula.molar_mass // divisor, 3)
807 volume = gas_volume(product, amount, divisor)
808 lines.append(f"product {product.name} {grams} g{volume}")
809 for index, reactant in enumerate(reactants):
810 if index != limiting:
811 consumed = limiting_mass * reaction_masses[index]
812 remaining = (masses[index] * divisor - consumed) // divisor
813 lines.append(f"remaining {reactant.name} {decimal(remaining, 3)} g")
814 return lines

817def command_yield(args):
818 equation_words, rest = split_at_separator(args)
819 if len(rest) != 4:
820 raise ChemError("wrong argument count for YIELD")
821 equation = read_equation(equation_words)
822 reactant_name, reactant_word, product_name, product_word = rest
823 reactant_index = find_species(equation.reactants, reactant_name)
824 if reactant_index is None:
825 raise ChemError(f"{reactant_name} is not a reactant")
826 product_index = find_species(equation.products, product_name)
827 if product_index is None:
828 raise ChemError(f"{product_name} is not a product")
829 reactant_mass = read_quantity(reactant_word)
830 actual = read_quantity(product_word)
831 coefficients = balance(equation)
832 reactant_coefficient = coefficients[reactant_index]
833 product_coefficient = coefficients[len(equation.reactants) + product_index]
834 reactant_molar_mass = equation.reactants[reactant_index].formula.molar_mass
835 product_molar_mass = equation.products[product_index].formula.molar_mass
836 produced = reactant_mass * product_coefficient * product_molar_mass
837 theoretical = produced // (reactant_coefficient * reactant_molar_mass)
838 lines = [
839 equation_notation(equation, coefficients),
840 f"theoretical {decimal(theoretical, 3)} g",
841 f"actual {decimal(actual, 3)} g",
842 ]
843 if theoretical == 0:
844 lines.append("yield undefined")
845 else:
846 lines.append(f"yield {decimal(actual * 10000 // theoretical, 2)}%")
847 return lines

861def command_deltah(args):
862 equation = read_equation(args)
863 for item in equation.species:
864 if item.name not in formation_enthalpies:
865 raise ChemError(f"no enthalpy for {item.name}")
866 coefficients = balance(equation)
867 contributions = [
868 value * formation_enthalpies[item.name]
869 for item, value in zip(equation.species, coefficients)
870 ]
871 split = len(equation.reactants)
872 change = sum(contributions[split:]) - sum(contributions[:split])
873 if change < 0:
874 kind = "exothermic"
875 elif change > 0:
876 kind = "endothermic"
877 else:
878 kind = "thermoneutral"
879 return [
880 equation_notation(equation, coefficients),
881 f"dH = {enthalpy_notation(change)} kJ ({kind})",
882 ]

934def execute(line):
935 name, *args = line.split(" ")
936 try:
937 if name not in COMMANDS:
938 raise ChemError(f"unknown command {name}")
939 handler, accepts = COMMANDS[name]
940 if not accepts(len(args)):
941 raise ChemError(f"wrong argument count for {name}")
942 return handler(args)
943 except ChemError as error:
944 return [f"error: {error}"]
945 except NoCoefficients as result:
946 return [str(result)]
