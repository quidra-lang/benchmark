hill_order <- function(c) {
  es <- sort(names(c), method = 'radix')
  if ('C' %in% es) {
    c('C', if ('H' %in% es) 'H', setdiff(es, c('C', 'H')))
  } else {
    es
  }
hill <- function(c, q = 0) {
  es <- hill_order(c)
  s <- paste0(es, ifelse(c[es] == 1, '', num(c[es])), collapse = '')
  if (q != 0) {
    s <- paste0(
      s,
      '^',
      if (abs(q) == 1) '' else num(abs(q)),
      if (q > 0) '+' else '-'
    )
  }
  s
