182function nullspace(rows, n) {

561 for (let i = 1; i < r; i++) if (qs[i] * v[l] * list[l].m < qs[l] * v[i] * list[i].m) l = i;
562 const d = v[l] * list[l].m;
563 out.push(
564 "limiting " + list[l].name,
565 "extent " + dec(Math.floor((qs[l] * 1000000) / d), 6) + " mol",
566 );
567 for (let i = r; i < list.length; i++)
568 out.push(
569 `product ${list[i].name} ${dec(Math.floor((qs[l] * v[i] * list[i].m) / d), 3)} g` +
570 gas(list[i], qs[l] * v[i], d),
571 );
572 for (let i = 0; i < r; i++)
573 if (i !== l)
574 out.push(
575 `remaining ${list[i].name} ${dec(Math.floor((qs[i] * d - qs[l] * v[i] * list[i].m) / d), 3)} g`,
576 );
577 return out;
