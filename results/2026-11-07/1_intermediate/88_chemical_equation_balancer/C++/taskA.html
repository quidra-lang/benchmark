176struct Parser{
177 string text;
178 size_t p=0;
179 vector<string> seen;
180 char peek()const{
181 return p<text.size()?text[p]:'\0';
182 }
183 Integer number(){
184 size_t start=p;
185 while(isdigit(peek()))++p;
186 if(start==p)return 1;
187 if(text[start]=='0')throw invalid_argument("");
188 return stoll(text.substr(start,p-start));
189 }
190 Composition sequence(){
191 Composition c;
192 int items=0;
193 while(true){
194 char ch=peek();
195 Composition item;
196 if(isupper(ch)){
197 string e(1,text[p++]);
198 if(islower(peek()))e+=text[p++];
199 seen.push_back(e);
200 item[e]=1;
201 }else if(ch=='('||ch=='['){
202 ++p;
203 item=sequence();
204 if(peek()!=(ch=='('?')':']'))throw invalid_argument("");
205 ++p;
206 }else break;
207 Integer k=number();
208 add(c,item,k);
209 ++items;
210 }
211 if(!items)throw invalid_argument("");
212 return c;
213 }
214 Formula parse(){
215 Formula f{
216 sequence(),0
217 };
218 while(peek()=='.'){
219 ++p;
220 Integer k=number();
221 Composition c=sequence();
222 add(f.c,c,k);
223 }
224 if(peek()=='^'){
225 ++p;
226 Integer k=number();
227 char ch=peek();
228 if(ch!='+'&&ch!='-')throw invalid_argument("");
229 ++p;
230 f.q=ch=='+'?k:-k;
231 }
232 if(p!=text.size())throw invalid_argument("");
233 return f;
234 }

351Nullspace nullspace(const vector<vector<Integer>>& rows,size_t n){
352 vector<vector<Fraction>> a;
353 for(const auto& row:rows){
354 vector<Fraction> r;
355 for(Integer x:row)r.emplace_back(x);
356 a.push_back(r);
357 }
358 vector<size_t> piv,free;
359 for(size_t j=0;j<n;++j){
360 size_t r=piv.size(),i=r;
361 while(i<a.size()&&!a[i][j].n)++i;
362 if(i==a.size()){
363 free.push_back(j);
364 continue;
365 }
366 swap(a[r],a[i]);
367 Fraction t=a[r][j];
368 for(Fraction& x:a[r])x=x/t;
369 for(size_t k=0;k<a.size();++k)if(k!=r&&a[k][j].n){
370 Fraction f=a[k][j];
371 for(size_t z=0;z<n;++z)a[k][z]=a[k][z]-f*a[r][z];
372 }
373 piv.push_back(j);
374 }
375 Nullspace result{
376 int(piv.size()),{
377 }
378 };
379 for(size_t f:free){
380 vector<Fraction> v(n);
381 v[f]=Fraction(1);
382 for(size_t i=0;i<piv.size();++i)v[piv[i]]=Fraction(-a[i][f].n,a[i][f].d);
383 Integer l=1;
384 for(Fraction x:v)l=lcm(l,x.d);
385 vector<Integer> b;
386 Integer g=0;
387 for(Fraction x:v){
388 Integer z=x.n*(l/x.d);
389 b.push_back(z);
390 g=gcd(g,z);
391 }
392 for(Integer& x:b)x/=g;
393 result.basis.push_back(b);
394 }
395 return result;
396}

673 }
674 if(cmd=="REDOX"||cmd=="HALF"){
675 string medium=a[0];
676 if(medium!="acidic"&&medium!="basic")fail("invalid medium "+medium);
677 Equation eq=equation(vector<string>(a.begin()+1,a.end()));
678 for(const Species& s:eq.list)if(s.f=="H^+"||s.f=="OH^-"||s.f=="H2O")fail("reserved species "+s.name());
679 vector<Species> aux={
680 species(medium=="acidic"?"H^+":"OH^-"),species("H2O")
681 };
682 if(cmd=="HALF")aux.push_back({
683 "e^-","",1,{
684 {
685 },-1
686 }
687 });
688 vector<Species> list=eq.list;
689 list.insert(list.end(),aux.begin(),aux.end());
690 vector<Integer> signs=eq.signs();
691 signs.insert(signs.end(),aux.size(),1);
692 vector<Integer> v=coefficients(list,signs,eq.list.size());
693 vector<string> left,right;
694 for(size_t i=0;i<eq.list.size();++i)(i<eq.r?left:right).push_back(term(list[i].name(),v[i]));
695 for(size_t i=eq.list.size();i<list.size();++i)if(v[i])(v[i]>0?left:right).push_back(term(list[i].name(),abs(v[i])));
696 vector<string> out={
697 join(left," + ")+" -> "+join(right," + ")
698 };
699 if(cmd=="HALF")out.push_back(v.back()>0?"reduction":v.back()<0?"oxidation":"no electron transfer");
700 return out;
701 }
702 vector<string> words=a,rest;
703 if(cmd=="STOICH"||cmd=="LIMITING"||cmd=="YIELD"){
704 if(count(a.begin(),a.end(),":")!=1)fail("invalid separator");
705 auto p=find(a.begin(),a.end(),":");
706 words.assign(a.begin(),p);
707 rest.assign(p+1,a.end());
708 if((cmd=="STOICH"&&rest.size()!=2)||(cmd=="YIELD"&&rest.size()!=4))fail("wrong argument count for "+cmd);
709 }
710 Equation eq=equation(words,cmd=="CHECK");
711 const vector<Species>& list=eq.list;
712 size_t r=eq.r;
713 vector<Integer> signs=eq.signs();
714 if(cmd=="MATRIX"){
715 Matrix m=matrix(list,signs);
716 Nullspace ns=nullspace(m.rows,list.size());
717 vector<string> names,out;
718 for(const Species& s:list)names.push_back(s.name());
719 out.push_back("columns: "+join(names));
720 for(size_t i=0;i<m.rows.size();++i){
721 vector<string> row;
722 for(Integer x:m.rows[i])row.push_back(to_string(x));
723 out.push_back(m.labels[i]+": "+join(row));
724 }
725 out.push_back("rank "+to_string(ns.rank)+", nullity "+to_string(ns.basis.size()));
726 for(const auto& v:ns.basis){
727 vector<string> row;
728 for(Integer x:v)row.push_back(to_string(x));
729 out.push_back("null: "+join(row));
730 }
731 return out;
732 }
733 if(cmd=="CHECK"){
734 Matrix m=matrix(list,signs);
735 vector<string> diff;
736 for(size_t i=0;i<m.rows.size();++i){
737 Integer l=0,rr=0;
738 for(size_t j=0;j<list.size();++j)(j<r?l:rr)+=m.rows[i][j]*list[j].coefficient;
739 rr=-rr;
740 if(l!=rr)diff.push_back(m.labels[i]+": "+(m.labels[i]=="charge"?signedInt(l):to_string(l))+" != "+(m.labels[i]=="charge"?signedInt(rr):to_string(rr)));
741 }
742 if(!diff.empty()){
743 diff.insert(diff.begin(),"not balanced");
744 return diff;
745 }
746 Integer g=0;
747 for(const Species& s:list)g=gcd(g,s.coefficient);
748 return {
749 g==1?"balanced":"balanced, not in lowest terms (gcd "+to_string(g)+")"
750 };
751 }
752 int target=-1,product=-1;
753 vector<Integer> qs;
754 if(cmd=="DELTAH")for(const Species& s:list)if(!heats.count(s.name()))fail("no enthalpy for "+s.name());
755 if(cmd=="STOICH"){
756 for(size_t i=0;i<list.size();++i)if(list[i].name()==rest[0]){
757 target=i;
758 break;
759 }
760 if(target<0)fail("species "+rest[0]+" not in equation");
761 qs.push_back(quantity(rest[1]));
762 }
763 if(cmd=="LIMITING"){
764 if(rest.size()!=r)fail("expected "+to_string(r)+" masses, got "+to_string(rest.size()));
765 for(const string& x:rest)qs.push_back(quantity(x));
766 }
767 if(cmd=="YIELD"){
768 for(size_t i=0;i<r;++i)if(list[i].name()==rest[0]){
769 target=i;
770 break;
771 }
772 if(target<0)fail(rest[0]+" is not a reactant");
773 for(size_t i=r;i<list.size();++i)if(list[i].name()==rest[2]){
774 product=i;
775 break;
776 }
777 if(product<0)fail(rest[2]+" is not a product");
778 qs={
779 quantity(rest[1]),quantity(rest[3])
780 };
781 }
782 vector<Integer> v=coefficients(list,signs,list.size());
783 vector<string> out={
784 notation(eq,v)
785 };
786 if(cmd=="BALANCE")return out;
787 if(cmd=="DELTAH"){
788 Integer h=0;
789 for(size_t i=0;i<list.size();++i)h-=signs[i]*v[i]*heats.at(list[i].name());
790 out.push_back("dH = "+ent(h)+" kJ ("+(h<0?"exothermic":h>0?"endothermic":"thermoneutral")+")");
791 return out;
792 }
793 if(cmd=="STOICH"){
794 Integer d=v[target]*list[target].mass();
795 for(size_t i=0;i<list.size();++i){
796 Integer x=qs[0]*v[i];
797 out.push_back(list[i].name()+" "+dec(x*1000000/d,6)+" mol "+dec(x*list[i].mass()/d,3)+" g"+gas(list[i],x,d));
798 }
799 return out;
800 }
801 if(cmd=="YIELD"){
802 Integer t=qs[0]*v[product]*list[product].mass()/(v[target]*list[target].mass());
803 out.push_back("theoretical "+dec(t,3)+" g");
804 out.push_back("actual "+dec(qs[1],3)+" g");
805 out.push_back(t?"yield "+dec(qs[1]*10000/t,2)+"%":"yield undefined");
806 return out;
807 }
808 size_t l=0;
809 for(size_t i=1;i<r;++i)if(qs[i]*v[l]*list[l].mass()<qs[l]*v[i]*list[i].mass())l=i;
810 Integer d=v[l]*list[l].mass();
811 out.push_back("limiting "+list[l].name());
812 out.push_back("extent "+dec(qs[l]*1000000/d,6)+" mol");
813 for(size_t i=r;i<list.size();++i)out.push_back("product "+list[i].name()+" "+dec(qs[l]*v[i]*list[i].mass()/d,3)+" g"+gas(list[i],qs[l]*v[i],d));
814 for(size_t i=0;i<r;++i)if(i!=l)out.push_back("remaining "+list[i].name()+" "+dec((qs[i]*d-qs[l]*v[i]*list[i].mass())/d,3)+" g");
815 return out;
816 }
817};
