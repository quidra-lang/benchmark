122 if (/[a-z]/.test(text[p] || "!")) e += text[p++];

203function nullspace(rows: number[][], n: number) {
204 const a = rows.map((r) => r.map((x) => new Fraction(x))),
205 piv: number[] = [],
206 free: number[] = [];
207 for (let j = 0; j < n; j++) {
208 const r = piv.length,
209 i = a.findIndex((row, k) => k >= r && row[j].n !== 0);
210 if (i < 0) {
211 free.push(j);
212 continue;
213 }
214 [a[r], a[i]] = [a[i], a[r]];
215 const t = a[r][j];
216 a[r] = a[r].map((x) => x.div(t));
217 for (let k = 0; k < a.length; k++)
218 if (k !== r && a[k][j].n) {
219 const f = a[k][j];
220 a[k] = a[k].map((x, z) => x.sub(f.mul(a[r][z])));
221 }
222 piv.push(j);
223 }
224 const basis = free.map((f) => {
225 const v = Array.from({ length: n }, () => new Fraction(0));
226 v[f] = new Fraction(1);
227 piv.forEach((j, i) => (v[j] = a[i][f].neg()));
228 let l = 1;
229 for (const x of v) l = (l / gcd(l, x.d)) * x.d;
230 const b = v.map((x) => x.n * (l / x.d));
231 const g = b.reduce(gcd, 0);
232 return b.map((x) => x / g);
233 });
234 return { rank: piv.length, basis };

562 `dH = ${ent(h)} kJ (${h < 0 ? "exothermic" : h > 0 ? "endothermic" : "thermoneutral"})`,
563 ];
564 }
565 if (cmd === "STOICH") {
566 const d = v[target] * list[target].m;
567 return [
568 ...out,
569 ...list.map(
570 (s, i) =>
571 `${s.name} ${dec(Math.floor((qs[0] * v[i] * 1000000) / d), 6)} mol ${dec(Math.floor((qs[0] * v[i] * s.m) / d), 3)} g` +
572 gas(s, qs[0] * v[i], d),
573 ),
574 ];
575 }
576 if (cmd === "YIELD") {
577 const t = Math.floor((qs[0] * v[product] * list[product].m) / (v[target] * list[target].m));
578 return [
579 ...out,
580 `theoretical ${dec(t, 3)} g`,
581 `actual ${dec(qs[1], 3)} g`,
582 t ? "yield " + dec(Math.floor((qs[1] * 10000) / t), 2) + "%" : "yield undefined",
583 ];
584 }
585 let l = 0;
586 for (let i = 1; i < r; i++) if (qs[i] * v[l] * list[l].m < qs[l] * v[i] * list[i].m) l = i;
587 const d = v[l] * list[l].m;
588 out.push(
589 "limiting " + list[l].name,
590 "extent " + dec(Math.floor((qs[l] * 1000000) / d), 6) + " mol",
591 );
592 for (let i = r; i < list.length; i++)
593 out.push(
594 `product ${list[i].name} ${dec(Math.floor((qs[l] * v[i] * list[i].m) / d), 3)} g` +
595 gas(list[i], qs[l] * v[i], d),
596 );
597 for (let i = 0; i < r; i++)
598 if (i !== l)
599 out.push(
600 `remaining ${list[i].name} ${dec(Math.floor((qs[i] * d - qs[l] * v[i] * list[i].m) / d), 3)} g`,
601 );
602 return out;
603}
