func order(c Composition) []string {
	es := []string{}
	for e := range c {
		es = append(es, e)
	}
	sort.Strings(es)
	if _, ok := c["C"]; !ok {
		return es
	}
	out := []string{"C"}
	if _, ok := c["H"]; ok {
		out = append(out, "H")
	}
	for _, e := range es {
		if e != "C" && e != "H" {
			out = append(out, e)
		}
	}
	return out
func hill(c Composition, q int64) string {
	var s strings.Builder
	for _, e := range order(c) {
		s.WriteString(e)
		if c[e] != 1 {
			s.WriteString(strconv.FormatInt(c[e], 10))
		}
	}
	if q != 0 {
		s.WriteByte('^')
		if abs(q) != 1 {
			s.WriteString(strconv.FormatInt(abs(q), 10))
		}
		if q > 0 {
			s.WriteByte('+')
		} else {
			s.WriteByte('-')
		}
	}
	return s.String()
