145def fail_chem(text) = raise ChemError, text
146def dec(n, d) = "#{n / 10 ** d}.#{(n % 10 ** d).to_s.rjust(d, "0")}"
147def ent(n) = (n < 0 ? "-" : "") + dec(n.abs, 2)
148def signed(n)
149 return "0" if n == 0

152end

154def order(c)
155 es = c.keys.sort
156 es.include?("C") ? ["C", *(es.include?("H") ? ["H"] : []), *(es - %w[C H])] : es

160 b.each { |e, n| a[e] = a.fetch(e, 0) + n * k }

164def scale(c, k) = c.transform_values { |n| n * k }

166 order(c).map { |e| e + (c[e] == 1 ? "" : c[e].to_s) }.join +

168end

170class FormulaParser
171 attr_reader :symbols
172 def initialize(text)
173 @text = text
174 @p = 0
175 @symbols = []
176 end
177
178 def peek = @text[@p] || ""
179 def number
180 start = @p
181 @p += 1 while peek.match?(/\A[0-9]\z/)
182 return 1 if start == @p
183 raise ArgumentError if @text[start] == "0"
184 @text[start...@p].to_i
185 end
186
187 def sequence
188 c = {}
189 items = 0
190 loop do
191 ch = peek
192 if ch.match?(/\A[A-Z]\z/)
193 e = ch
194 @p += 1
195 if peek.match?(/\A[a-z]\z/)
196 e += peek
197 @p += 1
198 end
199
200 @symbols << e
201 item = {e => 1}
202 elsif ["(", "["].include?(ch)
203 @p += 1
204 item = sequence
205 raise ArgumentError unless peek == (ch == "(" ? ")" : "]")
206 @p += 1
207 else
208 break
209 end
210
211 add(c, item, number)
212 items += 1
213 end
214
215 raise ArgumentError if items == 0
216 c
217 end
218
219 def parse
220 c = sequence
221 while peek == "."
222 @p += 1
223 k = number
224 add(c, sequence, k)
225 end
226
227 q = 0
228 if peek == "^"
229 @p += 1
230 k = number
231 ch = peek
232 raise ArgumentError unless ["+", "-"].include?(ch)
233 @p += 1
234 q = ch == "+" ? k : -k
235 end
236
237 raise ArgumentError unless @p == @text.length
238 Formula.new(c, q)
239 end
240end

242def formula(text, word = text)
243 p = FormulaParser.new(text)
244 begin
245 f = p.parse
246 rescue ArgumentError
247 fail_chem("invalid formula #{word}")
248 end
249
250 p.symbols.each { |e| fail_chem("unknown element #{e}") unless WEIGHTS.key?(e) }
251 f
252end

254def species(word, check = false)
255 state = %w[(s) (l) (g) (aq)].find { |s| word.end_with?(s) } || ""
256 f = state.empty? ? word : word[0...-state.length]
257 k = 1
258 if check && (m = /\A[0-9]+/.match(f))
259 fail_chem("invalid formula #{word}") if f.start_with?("0")
260 k = m[0].to_i
261 f = f[m[0].length..]
262 end
263
264 Species.new(f, state, k, formula(f, word))
265end

267def equation(a, check = false)
268 ops = a.each_index.select(&:odd?).map { |i| a[i] }
269 if a.length.even? ||
270 ops.count("->") != 1 ||
271 ops.any? { |s| !%w[+ ->].include?(s) } ||
272 a.each_index.any? { |i| i.even? && %w[+ ->].include?(a[i]) }
273 fail_chem("malformed equation")
274 end
275
276 Equation.new(a.each_index.select(&:even?).map { |i| species(a[i], check) }, (a.index("->") + 1) / 2)
277end

279def quantity(s)
280 fail_chem("invalid quantity #{s}") unless /\A(0|[1-9][0-9]{0,3})(?:\.([0-9]{1,3}))?\z/.match?(s)
281 a, b = s.split(".")
282 q = a.to_i * 1000 + (b || "").ljust(3, "0").to_i
283 fail_chem("invalid quantity #{s}") if q == 0
284 q
285end

287def enthalpy(s)
288 fail_chem("invalid enthalpy #{s}") unless /\A-?(0|[1-9][0-9]{0,4})(?:\.[0-9]{1,2})?\z/.match?(s)
289 a, b = s.delete_prefix("-").split(".")
290 (s.start_with?("-") ? -1 : 1) * (a.to_i * 100 + (b || "").ljust(2, "0").to_i)
291end

293def matrix(list, signs)
294 all = {}
295 list.each { |s| add(all, s.c) }
296 labels = order(all) + ["charge"]
297 [labels, labels.map { |e| list.each_with_index.map { |s, i| signs[i] * (e == "charge" ? s.q : s.c.fetch(e, 0)) } }]
298end

300def nullspace(rows, n)
301 a = rows.map { |r| r.map(&:to_r) }
302 piv = []
303 free = []
304 n.times do |j|
305 r = piv.length
306 i = (r...a.length).find { |k| a[k][j] != 0 }
307 if i.nil?
308 free << j
309 next
310 end
311
312 a[r], a[i] = a[i], a[r]
313 t = a[r][j]
314 a[r] = a[r].map { |x| x / t }
315 a.each_index do |k|
316 next if k == r || a[k][j] == 0
317 t = a[k][j]
318 a[k] = a[k].each_with_index.map { |x, z| x - t * a[r][z] }
319 end
320
321 piv << j
322 end
323
324 basis = free.map do |f|
325 v = Array.new(n, 0.to_r)
326 v[f] = 1.to_r
327 piv.each_with_index { |j, i| v[j] = -a[i][f] }
328 l = v.map(&:denominator).reduce(1, :lcm)
329 b = v.map { |x| (x * l).to_i }
330 g = b.reduce(0) { |t, x| t.gcd(x) }
331 b.map { |x| x / g }
332 end
333
334 [piv.length, basis]
335end

337def coefficients(list, signs, k = list.length)
338 _, basis = nullspace(matrix(list, signs)[1], list.length)
339 raise NoCoefficients, "AMBIGUOUS (nullity #{basis.length})" if basis.length >= 2
340 raise NoCoefficients, "IMPOSSIBLE" if basis.empty?
341 v = basis[0]
342 v = v.map { |x| -x } if v[0] < 0
343 raise NoCoefficients, "IMPOSSIBLE" if v.take(k).any? { |x| x <= 0 }
344 v
345end

347def term(s, k) = k == 1 ? s : k.to_s + s
348def notation(eq, v)
349 t = eq.list.each_with_index.map { |s, i| term(s.name, v[i]) }
350 t.take(eq.r).join(" + ") + " -> " + t.drop(eq.r).join(" + ")
351end

353def gas(s, n, d) = s.state == "(g)" ? " #{dec(n * 22414 / d, 3)} L" : ""
354def ratios(a)
355 es = order(a)
356 f = es.min_by { |e| Rational(a[e], WEIGHTS[e]) }
357 out = es.map { |e| "#{e} #{dec(1000 * a[e] * WEIGHTS[f] / (WEIGHTS[e] * a[f]), 3)}" }
358 (1..12).each do |k|
359 c = {}
360 ok = true
361 es.each do |e|
362 n = k * a[e] * WEIGHTS[f]
363 d = WEIGHTS[e] * a[f]
364 i = (2 * n + d) / (2 * d)
365 c[e] = i
366 ok = false if 20 * (n - i * d).abs > d
367 end
368
369 return out + ["#{hill(c)} (k = #{k})"] if ok
370 end
371
372 out + ["no simple ratio"]
373end

375def group(f, k)
376 c = f.split("^")[0]
377 return c if k == 1
378 depth = 0
379 last = nil
380 if c.start_with?("[")
381 c.chars.each_with_index { |ch, i|
382 depth += 1 if ch == "["
383 if ch == "]"
384 depth -= 1
385 if depth == 0
386 last = i
387 break
388 end
389 end
390 }
391 end
392
393 (/\A[A-Z][a-z]?\z/.match?(c) || last == c.length - 1 ? c : "(#{c})") + k.to_s
394end

398 "PARSE" => 1,
399 "MASS" => 1,
400 "PERCENT" => 1,
401 "EMPIRICAL" => 1,
402 "DBE" => 1,
403 "OXSTATE" => 1,
404 "MOLES" => 2,
405 "GRAMS" => 2,
406 "MOLECULAR" => 2,
407 "SALT" => 2,
408 "HEAT" => 2,
409 "COMBUST" => 3,
410 "HEATS" => 0,
411 "DERIVE" => -2,
412 "BALANCE" => -1,
413 "MATRIX" => -1,
414 "CHECK" => -1,
415 "DELTAH" => -1,
416 "REDOX" => -3,
417 "HALF" => -3,
418 "STOICH" => -4,
419 "LIMITING" => -4,
420 "YIELD" => -4

423class Chemistry
424 def initialize = @heats = {}
425 def execute(line)
426 cmd, *a = line.split(" ")
427 n = a.length
428 fail_chem("unknown command #{cmd}") unless COUNTS.key?(cmd)
429 e = COUNTS[cmd]
430 if (e >= 0 && n != e) || (e == -1 && n < 1) || (e == -2 && (n < 2 || n.odd?)) || (e == -3 && n < 2)
431 fail_chem("wrong argument count for #{cmd}")
432 end
433
434 case cmd
435 when "ELEMENT"
436 s = a[0]
437 fail_chem("unknown element #{s}") unless WEIGHTS.key?(s)
438 return ["#{s} (Z = #{SYMBOLS.index(s) + 1}) #{dec(WEIGHTS[s], 3)} u"]
439 when "HEATS"
440 return @heats.empty? ? ["no enthalpies"] : @heats.keys.sort.map { |s| "#{s} #{ent(@heats[s])} kJ/mol" }
441 when "HEAT"
442 species(a[0])
443 h = enthalpy(a[1])
444 s = "dHf(#{a[0]}) = #{ent(h)} kJ/mol"
445 s += " (was #{ent(@heats[a[0]])})" if @heats.key?(a[0])
446 @heats[a[0]] = h
447 return [s]
448 when "DERIVE"
449 b = {}
450 a.each_slice(2) do |s, w|
451 fail_chem("unknown element #{s}") unless WEIGHTS.key?(s)
452 fail_chem("duplicate element #{s}") if b.key?(s)
453 q = quantity(w)
454 fail_chem("invalid percentage #{w}") if q > 100000
455 b[s] = q
456 end
457
458 t = b.values.sum
459 fail_chem("percentages sum to #{dec(t, 3)}") if (t - 100000).abs > 1000
460 return ratios(b)
461 when "COMBUST"
462 s, p, r = a.map { |x| quantity(x) }
463 c = p * 12011 / 44009
464 h = r * 2016 / 18015
465 o = s - c - h
466 fail_chem("inconsistent masses") if c == 0 || h == 0 || 100 * o < -s
467 b = {"C" => c, "H" => h}
468 b["O"] = o if 100 * o > s
469 return b.map { |el, m| "mass #{el} #{dec(m, 3)} g" } + ratios(b)
470 end
471
472 if %w[PARSE MASS PERCENT EMPIRICAL DBE OXSTATE MOLES GRAMS MOLECULAR SALT].include?(cmd)
473 f = formula(a[0])
474 c = f.c
475 q = f.q
476 m = f.mass
477 es = order(c)
478 case cmd
479 when "PARSE"
480 return ["#{hill(c, q)} (#{es.map { |el| "#{el} #{c[el]}" }.join(", ")}; charge #{signed(q)})"]
481 when "MASS"
482 return ["M(#{a[0]}) = #{dec(m, 3)} g/mol"]
483 when "PERCENT"
484 return es.map { |el| "#{el} #{dec(10000 * c[el] * WEIGHTS[el] / m, 2)}%" }
485 when "EMPIRICAL"
486 g = c.values.reduce(:gcd)
487 return ["#{hill(c.transform_values { |x| x / g })} (n = #{g})"]
488 when "DBE"
489 fail_chem("charged formula #{a[0]}") unless q == 0
490 t = 2
491 es.each { |el|
492 fail_chem("no valence for #{el}") unless VALENCE.key?(el)
493 t += c[el] * (VALENCE[el] - 2)
494 }
495 return ["DBE = #{t < 0 ? "-" : ""}#{t.abs / 2}.#{t.odd? ? 5 : 0}"]
496 when "OXSTATE"
497 st = {}
498 if es.length == 1
499 st[es[0]] = Rational(q, c[es[0]])
500 else
501 st["F"] = -1 if c.key?("F")
502 es.each { |el| st[el] = METALS[el] if METALS.key?(el) }
503 st["H"] = es.all? { |el| el == "H" || METALS.key?(el) } ? -1 : 1 if c.key?("H")
504 if c.key?("O")
505 st["O"] = -2
506 elsif !c.key?("F")
507 el = %w[Cl Br I].find { |x| c.key?(x) }
508 st[el] = -1 if el
509 end
510
511 u = es - st.keys
512 return ["indeterminate: " + u.join(" ")] if u.length >= 2
513 total = st.sum { |el, x| c[el] * x }
514 if u.length == 1
515 st[u[0]] = Rational(q - total, c[u[0]])
516 elsif total != q
517 el = st.keys.last
518 st[el] = Rational(q - total + c[el] * st[el], c[el])
519 end
520 end
521
522 return [es.map { |el| "#{el} #{signed(st[el])}" }.join(", ")]
523 when "SALT"
524 b = formula(a[1])
525 fail_chem("not a cation #{a[0]}") if q <= 0
526 fail_chem("not an anion #{a[1]}") if b.q >= 0
527 a.each { |w| fail_chem("hydrate ion #{w}") if w.include?(".") }
528 g = q.gcd(b.q)
529 x = -b.q / g
530 y = q / g
531 z = add(scale(c, x), b.c, y)
532 return ["#{group(a[0], x)}#{group(a[1], y)} (#{hill(z)}, #{dec(Formula.new(z, 0).mass, 3)} g/mol)"]
533 end
534
535 fail_chem("charged formula #{a[0]}") if cmd == "MOLECULAR" && q != 0
536 v = quantity(a[1])
537 return ["#{dec(v, 3)} g #{a[0]} = #{dec(v * 1000000 / m, 6)} mol"] if cmd == "MOLES"
538 return ["#{dec(v, 3)} mol #{a[0]} = #{dec(v * m / 1000, 3)} g"] if cmd == "GRAMS"
539 k = (2 * v + m) / (2 * m)
540 return ["no integer multiple (ratio #{dec(1000 * v / m, 3)})"] if k == 0 || 10 * (v - k * m).abs > m
541 return ["#{hill(scale(c, k))} (n = #{k}, #{dec(k * m, 3)} g/mol)"]
542 end
543
544 if %w[REDOX HALF].include?(cmd)
545 fail_chem("invalid medium #{a[0]}") unless %w[acidic basic].include?(a[0])
546 eq = equation(a.drop(1))
547 eq.list.each { |s| fail_chem("reserved species #{s.name}") if %w[H^+ OH^- H2O].include?(s.f) }
548 aux = [species(a[0] == "acidic" ? "H^+" : "OH^-"), species("H2O")]
549 aux << Species.new("e^-", "", 1, Formula.new({}, -1)) if cmd == "HALF"
550 v = coefficients(eq.list + aux, eq.signs + Array.new(aux.length, 1), eq.list.length)
551 left = []
552 right = []
553 eq.list.each_with_index { |s, i| (i < eq.r ? left : right) << term(s.name, v[i]) }
554 aux.each_with_index { |s, i|
555 k = v[eq.list.length + i]
556 (k > 0 ? left : right) << term(s.name, k.abs) unless k == 0
557 }
558 out = [left.join(" + ") + " -> " + right.join(" + ")]
559 out << (v[-1] > 0 ? "reduction" : v[-1] < 0 ? "oxidation" : "no electron transfer") if cmd == "HALF"
560 return out
561 end
562
563 words = a
564 rest = []
565 if %w[STOICH LIMITING YIELD].include?(cmd)
566 fail_chem("invalid separator") unless a.count(":") == 1
567 i = a.index(":")
568 words = a.take(i)
569 rest = a.drop(i + 1)
570 if (cmd == "STOICH" && rest.length != 2) || (cmd == "YIELD" && rest.length != 4)
571 fail_chem("wrong argument count for #{cmd}")
572 end
573 end
574
575 eq = equation(words, cmd == "CHECK")
576 list = eq.list
577 r = eq.r
578 if cmd == "MATRIX"
579 labels, rows = matrix(list, eq.signs)
580 rank, basis = nullspace(rows, list.length)
581 return ["columns: " + list.map(&:name).join(" ")] +
582 rows.each_with_index.map { |row, i| labels[i] + ": " + row.join(" ") } +
583 ["rank #{rank}, nullity #{basis.length}"] +
584 basis.map { |v| "null: " + v.join(" ") }
585 end
586
587 if cmd == "CHECK"
588 labels, rows = matrix(list, eq.signs)
589 diff = []
590 rows.each_with_index do |row, i|
591 l = (0...r).sum { |j| row[j] * list[j].k }
592 rr = -(r...list.length).sum { |j| row[j] * list[j].k }
593 unless l == rr
594 diff << "#{labels[i]}: #{labels[i] == "charge" ? signed(l) : l} != #{labels[i] == "charge" ? signed(rr) : rr}"
595 end
596 end
597
598 return ["not balanced"] + diff unless diff.empty?
599 g = list.map(&:k).reduce(:gcd)
600 return [g == 1 ? "balanced" : "balanced, not in lowest terms (gcd #{g})"]
601 end
602
603 qs = []
604 target = product = nil
605 case cmd
606 when "DELTAH"
607 list.each { |s| fail_chem("no enthalpy for #{s.name}") unless @heats.key?(s.name) }
608 when "STOICH"
609 target = list.index { |s| s.name == rest[0] }
610 fail_chem("species #{rest[0]} not in equation") if target.nil?
611 qs = [quantity(rest[1])]
612 when "LIMITING"
613 fail_chem("expected #{r} masses, got #{rest.length}") unless rest.length == r
614 qs = rest.map { |s| quantity(s) }
615 when "YIELD"
616 target = (0...r).find { |i| list[i].name == rest[0] }
617 fail_chem("#{rest[0]} is not a reactant") if target.nil?
618 product = (r...list.length).find { |i| list[i].name == rest[2] }
619 fail_chem("#{rest[2]} is not a product") if product.nil?
620 qs = [quantity(rest[1]), quantity(rest[3])]
621 end
622
623 v = coefficients(list, eq.signs)
624 out = [notation(eq, v)]
625 case cmd
626 when "BALANCE"
627 out
628 when "DELTAH"
629 h = list.each_index.sum { |i| -eq.signs[i] * v[i] * @heats[list[i].name] }
630 out + ["dH = #{ent(h)} kJ (#{h < 0 ? "exothermic" : h > 0 ? "endothermic" : "thermoneutral"})"]
631 when "STOICH"
632 d = v[target] * list[target].mass
633 out +
634 list.each_with_index.map { |s, i|
635 "#{s.name} #{dec(qs[0] * v[i] * 1000000 / d, 6)} mol #{dec(qs[0] * v[i] * s.mass / d, 3)} g" +
636 gas(s, qs[0] * v[i], d)
637 }
638 when "YIELD"
639 t = qs[0] * v[product] * list[product].mass / (v[target] * list[target].mass)
640 out +
641 [
642 "theoretical #{dec(t, 3)} g",
643 "actual #{dec(qs[1], 3)} g",
644 t == 0 ? "yield undefined" : "yield #{dec(qs[1] * 10000 / t, 2)}%"
645 ]
646 when "LIMITING"
647 l = (0...r).min_by { |i| Rational(qs[i], v[i] * list[i].mass) }
648 d = v[l] * list[l].mass
649 out += ["limiting #{list[l].name}", "extent #{dec(qs[l] * 1000000 / d, 6)} mol"]
650 (r...list.length).each { |i|
651 out << "product #{list[i].name} #{dec(qs[l] * v[i] * list[i].mass / d, 3)} g" + gas(list[i], qs[l] * v[i], d)
652 }
653 (0...r).each { |i|
654 out << "remaining #{list[i].name} #{dec((qs[i] * d - qs[l] * v[i] * list[i].mass) / d, 3)} g" unless i == l
655 }
656 out
657 end
658
659 rescue ChemError => e
660 ["error: " + e.message]
661 rescue NoCoefficients => e
662 [e.message]
663 end
664end
