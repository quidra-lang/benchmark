def order(c)
  es = c.keys.sort
  es.include?("C") ? ["C", *(es.include?("H") ? ["H"] : []), *(es - %w[C H])] : es
def hill(c, q = 0)

    (q == 0 ? "" : "^" + (q.abs == 1 ? "" : q.abs.to_s) + (q > 0 ? "+" : "-"))
