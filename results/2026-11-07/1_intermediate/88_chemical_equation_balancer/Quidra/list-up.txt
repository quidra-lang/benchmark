string[] symbols()
    return "H He Li Be B C N O F Ne Na Mg Al Si P S Cl Ar K Ca Sc Ti V Cr Mn Fe Co Ni Cu Zn Ga Ge As Se Br Kr Rb Sr Y Zr Nb Mo Tc Ru Rh Pd Ag Cd In Sn Sb Te I Xe".split(" ")
int[] hill_order(int[] counts)
    names := symbols()
    present := [for i := (nat(0):<#names)[] if counts[i] > 0 then names[i]]
    ordered := present.sorted()
    if counts[5] > 0
        ordered = ["C"].concat(if counts[0] > 0 then ["H"] else []).concat([for name := ordered[] if name != "C" and name != "H" then name])
    return [for name := ordered[] then locate(names, name)]

string hill(int[] counts, charge := 0)
    result := ""
    names := symbols()
    for element := hill_order(counts)[]
        result += names[element] + (if counts[element] == 1 then "" else string(counts[element]))
    if charge != 0
        result += "^" + (if absolute(charge) == 1 then "" else string(absolute(charge))) + (if charge > 0 then "+" else "-")
    return result
