121function number!(p)
122 start=p.p
123 while isdigit(peek(p))
124 p.p+=1
125 end
126 start==p.p && return 1
127 p.text[start]=='0' && throw(ArgumentError(""))
128 parse(Int, p.text[start:(p.p-1)])
129end
130function sequence!(p)
131 c=Dict{String,Int}()
132 items=0
133 while true
134 ch=peek(p)
135 if 'A'<=ch<='Z'
136 e=string(ch)
137 p.p+=1
138 if 'a'<=peek(p)<='z'
139 e*=string(peek(p))
140 p.p+=1
141 end
142 push!(p.seen, e)
143 item=Dict(e=>1)
144 elseif ch=='('||ch=='['
145 p.p+=1
146 item=sequence!(p)
147 peek(p)!=(ch=='(' ? ')' : ']') && throw(ArgumentError(""))
148 p.p+=1
149 else
150 break
151 end
152 k=number!(p)
153 add!(c, item, k)
154 items+=1
155 end
156 items==0 && throw(ArgumentError(""))
157 c
158end
159function formula(text, word = text)
160 p=Parser(text, 1, String[])
161 c=Dict{String,Int}()
162 q=0
163 try
164 c=sequence!(p)
165 while peek(p)=='.'
166 p.p+=1
167 k=number!(p)
168 add!(c, sequence!(p), k)
169 end
170 if peek(p)=='^'
171 p.p+=1
172 k=number!(p)
173 ch=peek(p)
174 !(ch in ['+', '-']) && throw(ArgumentError(""))
175 p.p+=1
176 q=ch=='+' ? k : -k
177 end
178 p.p!=length(text)+1 && throw(ArgumentError(""))
179 catch e
180 e isa ArgumentError || rethrow()
181 fail("invalid formula "*word)
182 end
183 for e in p.seen
184 haskey(W, e)||fail("unknown element "*e)
185 end
186 Formula(c, q)
187end
188function species(word, check = false)
189 state=""

191 k=1
192 for s in ["(s)", "(l)", "(g)", "(aq)"]
193 if endswith(word, s)
194 state=s
195 f=word[1:(end-length(s))]

198 end

200 startswith(f, "0")&&fail("invalid formula "*word)
201 k=parse(Int, m.match)
202 f=f[(length(m.match)+1):end]
203 end

205end
206function equation(a, check = false)
207 ops=a[2:2:end]
208 (
209 iseven(length(a))||count(==("->"), ops)!=1||any(x->!(x in ["+", "->"]), ops)||any(
210 x->x in ["+", "->"],
211 a[1:2:end],
212 )
213 )&&fail("malformed equation")
214 list=[species(x, check) for x in a[1:2:end]]
215 r=div(findfirst(==("->"), a), 2)
216 (list = list, r = r, sign = [i<=r ? 1 : -1 for i in eachindex(list)])
217end
218function quantity(s)
219 occursin(r"^(0|[1-9][0-9]{0,3})(\.[0-9]{1,3})?$", s)||fail("invalid quantity "*s)
220 parts=split(s, '.')
221 q=parse(Int, parts[1])*1000+(length(parts)==1 ? 0 : parse(Int, rpad(parts[2], 3, '0')))
222 q==0&&fail("invalid quantity "*s)
223 q
224end
225function enthalpy(s)
226 occursin(r"^-?(0|[1-9][0-9]{0,4})(\.[0-9]{1,2})?$", s)||fail("invalid enthalpy "*s)
227 parts=split(startswith(s, "-") ? s[2:end] : s, '.')
228 (startswith(s, "-") ? -1 : 1)*(
229 parse(Int, parts[1])*100+(length(parts)==1 ? 0 : parse(Int, rpad(parts[2], 2, '0')))
230 )
231end

238 rows=[

245end
246function nullspace(rows, n)
247 a=[Rational{Int}.(r) for r in rows]
248 piv=Int[]
249 free=Int[]
250 for j in 1:n
251 r=length(piv)+1
252 i=findfirst(k->a[k][j]!=0, r:length(a))
253 if i===nothing
254 push!(free, j)
255 continue
256 end
257 i+=r-1
258 a[r], a[i]=a[i], a[r]
259 t=a[r][j]
260 a[r]=a[r] ./ t
261 for k in eachindex(a)
262 if k!=r&&a[k][j]!=0
263 t=a[k][j]
264 a[k]=a[k] .- t .* a[r]
265 end
266 end
267 push!(piv, j)
268 end
269 basis=Vector{Int}[]
270 for f in free
271 v=fill(0//1, n)
272 v[f]=1//1
273 for (i, j) in enumerate(piv)
274 v[j]=-a[i][f]
275 end
276 l=foldl(lcm, denominator.(v); init = 1)
277 b=[numerator(x)*div(l, denominator(x)) for x in v]
278 g=foldl(gcd, b; init = 0)
279 push!(basis, div.(b, g))
280 end
281 length(piv), basis
282end
283function coefficients(list, sign, k = length(list))
284 _, rows=matrix(list, sign)
285 _, basis=nullspace(rows, length(list))
286 d=length(basis)
287 d>=2&&throw(NoCoefficients("AMBIGUOUS (nullity $d)"))
288 d==0&&throw(NoCoefficients("IMPOSSIBLE"))
289 v=basis[1]
290 v[1]<0&&(v=-v)
291 any(<=(0), v[1:k])&&throw(NoCoefficients("IMPOSSIBLE"))
292 v
293end

300function ratios(a)
301 es=hillorder(a)
302 f=es[argmin([a[e]//W[e] for e in es])]
303 out=[e*" "*dec(div(1000*a[e]*W[f], W[e]*a[f]), 3) for e in es]
304 for k in 1:12
305 c=Dict{String,Int}()
306 ok=true
307 for e in es
308 n=k*a[e]*W[f]
309 d=W[e]*a[f]
310 i=div(2*n+d, 2*d)
311 c[e]=i
312 20*abs(n-i*d)>d&&(ok=false)
313 end
314 ok&&return vcat(out, [hill(c)*" (k = $k)"])
315 end
316 vcat(out, ["no simple ratio"])
317end

326 if ch==']'
327 depth-=1
328 if depth==0
329 last=i

332 end
333 end
334 end
335 (occursin(r"^[A-Z][a-z]?$", c)||last==length(c) ? c : "("*c*")")*string(k)
336end

363function execute(line, heats)
364 words=String.(split(line))
365 cmd=popfirst!(words)
366 a=words
367 n=length(a)
368 haskey(COUNTS, cmd)||fail("unknown command "*cmd)
369 e=COUNTS[cmd]
370 ((e>=0&&n!=e)||(e==-1&&n<1)||(e==-2&&(n<2||isodd(n)))||(e==-3&&n<2))&&fail(
371 "wrong argument count for "*cmd,
372 )
373 if cmd=="ELEMENT"
374 haskey(W, a[1])||fail("unknown element "*a[1])
375 return ["$(a[1]) (Z = $(findfirst(==(a[1]),SYMBOLS))) $(dec(W[a[1]],3)) u"]
376 elseif cmd=="HEATS"
377 return isempty(heats) ? ["no enthalpies"] :
378 [s*" "*ent(heats[s])*" kJ/mol" for s in sort(collect(keys(heats)))]
379 elseif cmd=="HEAT"
380 species(a[1])
381 h=enthalpy(a[2])
382 s="dHf($(a[1])) = $(ent(h)) kJ/mol"*(
383 haskey(heats, a[1]) ? " (was $(ent(heats[a[1]])))" : ""
384 )
385 heats[a[1]]=h
386 return [s]
387 elseif cmd=="DERIVE"
388 b=Dict{String,Int}()
389 for i in 1:2:n
390 s=a[i]
391 haskey(W, s)||fail("unknown element "*s)
392 haskey(b, s)&&fail("duplicate element "*s)
393 q=quantity(a[i+1])
394 q>100000&&fail("invalid percentage "*a[i+1])
395 b[s]=q
396 end
397 t=sum(values(b))
398 abs(t-100000)>1000&&fail("percentages sum to "*dec(t, 3))
399 return ratios(b)
400 elseif cmd=="COMBUST"
401 s, p, r=quantity.(a)
402 c=div(p*12011, 44009)
403 h=div(r*2016, 18015)
404 o=s-c-h
405 (c==0||h==0||100*o < -s)&&fail("inconsistent masses")
406 b=Dict("C"=>c, "H"=>h)
407 100*o>s&&(b["O"]=o)
408 return vcat(["mass $e $(dec(b[e],3)) g" for e in hillorder(b)], ratios(b))
409 end
410 if cmd in [
411 "PARSE",
412 "MASS",
413 "PERCENT",
414 "EMPIRICAL",
415 "DBE",
416 "OXSTATE",
417 "MOLES",
418 "GRAMS",
419 "MOLECULAR",
420 "SALT",
421 ]
422 f=formula(a[1])
423 c=f.c
424 q=f.q
425 m=mass(c)
426 es=hillorder(c)
427 cmd=="PARSE"&&return [
428 hill(c, q)*" ("*join(["$e $(c[e])" for e in es], ", ")*"; charge "*signed(q)*")",
429 ]
430 cmd=="MASS"&&return ["M($(a[1])) = $(dec(m,3)) g/mol"]
431 cmd=="PERCENT"&&return ["$e $(dec(div(10000*c[e]*W[e],m),2))%" for e in es]
432 if cmd=="EMPIRICAL"
433 g=foldl(gcd, values(c))
434 return [hill(Dict(e=>div(x, g) for (e, x) in c))*" (n = $g)"]
435 end
436 if cmd=="DBE"
437 q!=0&&fail("charged formula "*a[1])
438 t=2
439 for e in es
440 haskey(VALENCE, e)||fail("no valence for "*e)
441 t+=c[e]*(VALENCE[e]-2)
442 end
443 return ["DBE = "*(t<0 ? "-" : "")*string(div(abs(t), 2))*"."*(isodd(t) ? "5" : "0")]
444 elseif cmd=="OXSTATE"
445 st=Dict{String,Rational{Int}}()
446 last=""
447 if length(es)==1
448 st[es[1]]=q//c[es[1]]
449 else
450 if haskey(c, "F")
451 st["F"]=-1
452 last="F"
453 end
454 for e in es
455 if haskey(METALS, e)
456 st[e]=METALS[e]
457 last=e
458 end
459 end
460 if haskey(c, "H")
461 st["H"]=all(e->e=="H"||haskey(METALS, e), es) ? -1 : 1
462 last="H"
463 end
464 if haskey(c, "O")
465 st["O"]=-2
466 last="O"
467 elseif !haskey(c, "F")
468 for e in ["Cl", "Br", "I"]
469 if haskey(c, e)
470 st[e]=-1
471 last=e
472 break
473 end
474 end
475 end
476 u=filter(e->!haskey(st, e), es)
477 length(u)>1&&return ["indeterminate: "*join(u, " ")]
478 total=sum(c[e]*x for (e, x) in st; init = 0//1)
479 if !isempty(u)
480 st[u[1]]=(q-total)/c[u[1]]
481 elseif total!=q
482 st[last]=(q-total+c[last]*st[last])/c[last]
483 end
484 end
485 return [join([e*" "*signed(st[e]) for e in es], ", ")]
486 elseif cmd=="SALT"
487 b=formula(a[2])
488 q<=0&&fail("not a cation "*a[1])
489 b.q>=0&&fail("not an anion "*a[2])
490 for s in a
491 occursin('.', s)&&fail("hydrate ion "*s)
492 end
493 g=gcd(q, b.q)
494 x=div(-b.q, g)
495 y=div(q, g)
496 z=add!(scaled(c, x), b.c, y)
497 return [group(a[1], x)*group(a[2], y)*" ("*hill(z)*", "*dec(mass(z), 3)*" g/mol)"]
498 end
499 cmd=="MOLECULAR"&&q!=0&&fail("charged formula "*a[1])
500 x=quantity(a[2])
501 cmd=="MOLES"&&return ["$(dec(x,3)) g $(a[1]) = $(dec(div(x*1000000,m),6)) mol"]
502 cmd=="GRAMS"&&return ["$(dec(x,3)) mol $(a[1]) = $(dec(div(x*m,1000),3)) g"]
503 k=div(2*x+m, 2*m)
504 (k==0||10*abs(x-k*m)>m)&&return ["no integer multiple (ratio $(dec(div(1000*x,m),3)))"]
505 return [hill(scaled(c, k))*" (n = $k, $(dec(k*m,3)) g/mol)"]
506 end
507 if cmd in ["REDOX", "HALF"]
508 a[1] in ["acidic", "basic"]||fail("invalid medium "*a[1])
509 eq=equation(a[2:end])
510 for s in eq.list
511 s.f in ["H^+", "OH^-", "H2O"]&&fail("reserved species "*name(s))
512 end
513 aux=[species(a[1]=="acidic" ? "H^+" : "OH^-"), species("H2O")]
514 cmd=="HALF"&&push!(aux, Species("e^-", "", 1, Formula(Dict{String,Int}(), -1)))
515 list=vcat(eq.list, aux)
516 v=coefficients(list, vcat(eq.sign, ones(Int, length(aux))), length(eq.list))
517 left=String[]
518 right=String[]
519 for i in eachindex(eq.list)
520 push!(i<=eq.r ? left : right, term(name(list[i]), v[i]))
521 end
522 for i in (length(eq.list)+1):length(list)
523 v[i]!=0&&push!(v[i]>0 ? left : right, term(name(list[i]), abs(v[i])))
524 end
525 out=[join(left, " + ")*" -> "*join(right, " + ")]
526 cmd=="HALF"&&push!(
527 out,
528 v[end]>0 ? "reduction" : v[end]<0 ? "oxidation" : "no electron transfer",
529 )
530 return out
531 end
532 words=a
533 rest=String[]
534 if cmd in ["STOICH", "LIMITING", "YIELD"]
535 count(==(":"), a)==1||fail("invalid separator")
536 i=findfirst(==(":"), a)
537 words=a[1:(i-1)]
538 rest=a[(i+1):end]
539 ((cmd=="STOICH"&&length(rest)!=2)||(cmd=="YIELD"&&length(rest)!=4))&&fail(
540 "wrong argument count for "*cmd,
541 )
542 end
543 eq=equation(words, cmd=="CHECK")
544 list=eq.list
545 r=eq.r
546 if cmd=="MATRIX"
547 labels, rows=matrix(list, eq.sign)
548 rank, basis=nullspace(rows, length(list))
549 return vcat(
550 ["columns: "*join(name.(list), " ")],
551 [labels[i]*": "*join(row, " ") for (i, row) in enumerate(rows)],
552 ["rank $rank, nullity $(length(basis))"],
553 ["null: "*join(v, " ") for v in basis],
554 )
555 elseif cmd=="CHECK"
556 labels, rows=matrix(list, eq.sign)
557 diff=String[]
558 for (i, row) in enumerate(rows)
559 l=sum(row[j]*list[j].coefficient for j in 1:r)
560 rr=-sum(row[j]*list[j].coefficient for j in (r+1):length(list))
561 l!=rr&&push!(
562 diff,
563 labels[i]*": "*(labels[i]=="charge" ? signed(l) : string(l))*" != "*(
564 labels[i]=="charge" ? signed(rr) : string(rr)
565 ),
566 )
567 end
568 !isempty(diff)&&return vcat(["not balanced"], diff)
569 g=foldl(gcd, [s.coefficient for s in list])
570 return [g==1 ? "balanced" : "balanced, not in lowest terms (gcd $g)"]
571 end
572 qs=Int[]
573 target=nothing
574 product=nothing
575 if cmd=="DELTAH"
576 for s in list
577 haskey(heats, name(s))||fail("no enthalpy for "*name(s))
578 end
579 elseif cmd=="STOICH"
580 target=findfirst(s->name(s)==rest[1], list)
581 target===nothing&&fail("species $(rest[1]) not in equation")
582 qs=[quantity(rest[2])]
583 elseif cmd=="LIMITING"
584 length(rest)==r||fail("expected $r masses, got $(length(rest))")
585 qs=quantity.(rest)
586 elseif cmd=="YIELD"
587 target=findfirst(i->name(list[i])==rest[1], 1:r)
588 target===nothing&&fail(rest[1]*" is not a reactant")
589 product=findfirst(i->name(list[i])==rest[3], (r+1):length(list))
590 product===nothing&&fail(rest[3]*" is not a product")
591 product+=r
592 qs=[quantity(rest[2]), quantity(rest[4])]
593 end
594 v=coefficients(list, eq.sign)
595 out=[notation(eq, v)]
596 cmd=="BALANCE"&&return out
597 if cmd=="DELTAH"
598 h=sum(-eq.sign[i]*v[i]*heats[name(list[i])] for i in eachindex(list))
599 push!(
600 out,
601 "dH = $(ent(h)) kJ ("*(h<0 ? "exothermic" : h>0 ? "endothermic" : "thermoneutral")*")",
602 )
603 elseif cmd=="STOICH"
604 d=v[target]*mass(list[target])
605 append!(
606 out,
607 [
608 "$(name(s)) $(dec(div(qs[1]*v[i]*1000000,d),6)) mol $(dec(div(qs[1]*v[i]*mass(s),d),3)) g"*gas(
609 s,
610 qs[1]*v[i],
611 d,
612 ) for (i, s) in enumerate(list)
613 ],
614 )
615 elseif cmd=="YIELD"
616 t=div(qs[1]*v[product]*mass(list[product]), v[target]*mass(list[target]))
617 append!(
618 out,
619 [
620 "theoretical $(dec(t,3)) g",
621 "actual $(dec(qs[2],3)) g",
622 t==0 ? "yield undefined" : "yield $(dec(div(qs[2]*10000,t),2))%",
623 ],
624 )
625 else
626 l=argmin([qs[i]//(v[i]*mass(list[i])) for i in 1:r])
627 d=v[l]*mass(list[l])
628 append!(out, ["limiting "*name(list[l]), "extent $(dec(div(qs[l]*1000000,d),6)) mol"])
629 for i in (r+1):length(list)
630 push!(
631 out,
632 "product $(name(list[i])) $(dec(div(qs[l]*v[i]*mass(list[i]),d),3)) g"*gas(
633 list[i],
634 qs[l]*v[i],
635 d,
636 ),
637 )
638 end
639 for i in 1:r
640 i!=l&&push!(
641 out,
642 "remaining $(name(list[i])) $(dec(div(qs[i]*d-qs[l]*v[i]*mass(list[i]),d),3)) g",
643 )
644 end
645 end
646 out
647end
648function main()
649 heats=Dict{String,Int}()
650 for _ in 1:parse(Int, readline())
651 line=readline()
652 try
653 foreach(println, execute(line, heats))
654 catch e
655 if e isa ChemError
656 println("error: ", e.message)
657 elseif e isa NoCoefficients
658 println(e.message)
