1const METHODS = Set(["GET", "HEAD", "POST", "PUT", "DELETE", "PATCH", "OPTIONS"])
2const REASONS = [
3 "fields", "date", "time", "ip", "method", "path", "status", "bytes", "latency"
4]
5const CLASSES = ["1xx", "2xx", "3xx", "4xx", "5xx"]
6const FILTER_KEYS = ["METHOD", "STATUS", "IP", "PATH"]
7const MONTH_DAYS = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]
8const FORMS = Dict(
9 "SUMMARY" => String[], "MALFORMED" => String[],
10 "STATUS" => String[], "METHODS" => String[],
11 "TOP" => ["kind", "limit"], "HOURLY" => String[],
12 "TIMELINE" => ["minutes"], "BANDWIDTH" => String[],
13 "VISITORS" => String[], "LATENCY" => ["path"],
14 "HISTOGRAM" => ["width"], "APDEX" => ["threshold"],
15 "ERRORS" => String[], "SLOWEST" => ["limit"],
16 "SESSIONS" => ["ip", "minutes"], "BURST" => ["length", "times"],
17 "PEAK" => ["length"], "GAPS" => ["length"],
18 "IP" => ["ip"], "RANGE" => ["date", "time", "date", "time"],
19 "FILTER" => ["key", "value"], "UNFILTER" => ["subject"],
20 "RESET" => String[], "STATE" => String[]
21)
22const REQUIRE_RECORDS = Set([
23 "STATUS", "METHODS", "TOP", "HOURLY", "TIMELINE", "BANDWIDTH",
24 "VISITORS", "HISTOGRAM", "APDEX", "ERRORS", "SLOWEST", "BURST", "PEAK", "GAPS"
25])
26const RANGES = Dict(
27 "limit" => (1, 100), "minutes" => (1, 1440), "width" => (1, 100000),
28 "threshold" => (1, 100000), "length" => (1, 86400), "times" => (1, 1000)
29)
30const ERRORS = Dict(
31 "limit" => "invalid number", "minutes" => "invalid number",
32 "width" => "invalid number", "threshold" => "invalid number",
33 "length" => "invalid number", "times" => "invalid number",
34 "kind" => "invalid kind", "path" => "invalid path", "ip" => "invalid ip",
35 "date" => "invalid timestamp", "time" => "invalid timestamp",
36 "key" => "invalid key", "subject" => "invalid key", "value" => "invalid value"
37)

47function tokens(line)
48 value = strip(line)
49 isempty(value) && return String[]
50 return String.(split(value, r"\s+"))
51end
52
53digits(s) = occursin(r"^[0-9]+$", s)
54canonical(s, maxlen) = !isempty(s) && length(s) <= maxlen &&
55 digits(s) && (length(s) == 1 || s[1] != '0')
56
57leap(year) = year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
58year_days(year) = leap(year) ? 366 : 365
59month_days(year, month) = month == 2 && leap(year) ? 29 : MONTH_DAYS[month]
60
61function valid_date(s)
62 occursin(r"^[0-9]{4}-[0-9]{2}-[0-9]{2}$", s) || return false
63 year = parse(Int, s[1:4])
64 month = parse(Int, s[6:7])
65 day = parse(Int, s[9:10])
66 return 2000 <= year <= 2059 && 1 <= month <= 12 &&
67 1 <= day <= month_days(year, month)
68end
69
70function valid_time(s)
71 occursin(r"^[0-9]{2}:[0-9]{2}:[0-9]{2}$", s) || return false
72 return parse(Int, s[1:2]) < 24 && parse(Int, s[4:5]) < 60 &&
73 parse(Int, s[7:8]) < 60
74end
75
76function valid_ip(s)
77 parts = split(s, '.'; keepempty=true)
78 length(parts) == 4 || return false
79 return all(p -> canonical(p, 3) && parse(Int, p) <= 255, parts)
80end
81
82valid_status(s) = occursin(r"^[0-9]{3}$", s) &&
83 100 <= parse(Int, s) <= 599
84
85function valid_filter_value(key, value)
86 if key == "METHOD"
87 return value in METHODS
88 elseif key == "STATUS"
89 return value in CLASSES || valid_status(value)
90 elseif key == "IP"
91 return valid_ip(value)
92 end
93 return startswith(value, "/")
94end
95
96function valid_argument(kind, value, args)
97 if haskey(RANGES, kind)
98 low, high = RANGES[kind]
99 return canonical(value, 6) && low <= parse(Int, value) <= high
100 end
101 kind == "kind" && return value in ["PATHS", "IPS"]
102 kind == "path" && return startswith(value, "/")
103 kind == "ip" && return valid_ip(value)
104 kind == "date" && return valid_date(value)
105 kind == "time" && return valid_time(value)
106 kind == "key" && return value in FILTER_KEYS
107 if kind == "subject"
108 return value == "RANGE" || value in FILTER_KEYS
109 end
110 return valid_filter_value(args[1], value)
111end
112
113function malformed_reason(fields)
114 length(fields) == 8 || return "fields"
115 valid_date(fields[1]) || return "date"
116 valid_time(fields[2]) || return "time"
117 valid_ip(fields[3]) || return "ip"
118 fields[4] in METHODS || return "method"
119 startswith(fields[5], "/") || return "path"
120 valid_status(fields[6]) || return "status"
121 canonical(fields[7], 8) || return "bytes"
122 canonical(fields[8], 6) || return "latency"
123 return nothing
124end

126function seconds_of(date, clock)
127 year = parse(Int, date[1:4])
128 month = parse(Int, date[6:7])
129 days = parse(Int, date[9:10]) - 1
130 for y in 2000:year-1
131 days += year_days(y)
132 end
133 for m in 1:month-1
134 days += month_days(year, m)
135 end
136 return days * 86400 + parse(Int, clock[1:2]) * 3600 +
137 parse(Int, clock[4:5]) * 60 + parse(Int, clock[7:8])
138end
139
140padded(n, width=2) = lpad(string(n), width, '0')
141
142function interval_label(seconds)
143 days = seconds ÷ 86400
144 minutes = seconds % 86400 ÷ 60
145 year = 2000
146 while days >= year_days(year)
147 days -= year_days(year)
148 year += 1
149 end
150 month = 1
151 while days >= month_days(year, month)
152 days -= month_days(year, month)
153 month += 1
154 end
155 return string(padded(year, 4), "-", padded(month), "-", padded(days + 1),
156 " ", padded(minutes ÷ 60), ":", padded(minutes % 60))
157end

159function normalize_path(raw)
160 path = split(raw, '?'; limit=2)[1]
161 parts = String[]
162 for part in split(path, '/'; keepempty=true)
163 if part == ".."
164 isempty(parts) || pop!(parts)
165 elseif part != "" && part != "."
166 push!(parts, String(part))
167 end
168 end
169 for i in eachindex(parts)
170 s = parts[i]
171 if digits(s)
172 parts[i] = "{id}"
173 elseif length(s) >= 8 && occursin(r"^[0-9a-f]+$", s) &&
174 occursin(r"[0-9]", s)
175 parts[i] = "{hex}"
176 end
177 end
178 return "/" * join(parts, "/")
179end
180
181function compare_ip(a, b)
182 left = parse.(Int, split(a, '.'))
183 right = parse.(Int, split(b, '.'))
184 for i in 1:4
185 left[i] != right[i] && return left[i] < right[i]
186 end
187 return false
188end
189
190function status_matches(filter, status)
191 if filter in CLASSES
192 return status ÷ 100 == parse(Int, filter[1:1])
193 end
194 return status == parse(Int, filter)
195end
196
197function decimal(numerator, denominator, places)
198 scale = 10 ^ places
199 result = (numerator * scale) ÷ denominator
200 return string(result ÷ scale, ".", padded(result % scale, places))
201end
202
203percent(a, b) = decimal(a * 100, b, 2) * "%"
204bar(count, maximum) = repeat("#", (count * 30 + maximum - 1) ÷ maximum)
205
206function bytes_text(size)
207 size >= 1073741824 && return decimal(size, 1073741824, 1) * "GB"
208 size >= 1048576 && return decimal(size, 1048576, 1) * "MB"
209 size >= 1024 && return decimal(size, 1024, 1) * "KB"
210 return string(size, "B")
211end

251function ranked(counts; tie=(a,b)->a<b)
252 return sort!(collect(keys(counts));

256function class_counts(records)
257 counts = zeros(Int, 5)
258 for r in records
259 counts[r.status ÷ 100] += 1

264function bucket_counts(values)
265 first, last = minimum(values), maximum(values)
266 ensure(last - first + 1 <= 60, "too many buckets")
267 counts = zeros(Int, last - first + 1)
268 for value in values
269 counts[value - first + 1] += 1
270 end
271 return first, counts
272end

274percentile(values, rank) = values[(rank * length(values) + 99) ÷ 100]

276function peak(records, width)
277 first = 1
278 best = 0
279 start = 1
280 last = 1
281 for i in eachindex(records)
282 while records[i].seconds - records[first].seconds >= width
283 first += 1
284 end
285 if i - first + 1 > best
286 best = i - first + 1
287 start = first
288 last = i
289 end
290 end
291 return (size=best, start=records[start], last=records[last])
292end
293
294function apdex(records, threshold)
295 sat = count(r -> r.status < 500 && r.latency <= threshold, records)
296 tol = count(r -> r.status < 500 &&
297 threshold < r.latency <= threshold * 4, records)
298 n = length(records)
299 points, total = 2 * sat + tol, 2 * n
300 rating = if points * 100 >= 94 * total
301 "excellent"
302 elseif points * 100 >= 85 * total
303 "good"
304 elseif points * 100 >= 70 * total
305 "fair"
306 elseif points * 100 >= 50 * total
307 "poor"
308 else
309 "unacceptable"
310 end
311 return (count=n, satisfied=sat, tolerating=tol, frustrated=n-sat-tol,
312 points=points, total=total, text=decimal(points, total, 3), rating=rating)
313end

315mutable struct Analyzer
316 records::Vector{Record}
317 malformed::Vector{Tuple{Int,String}}
318 linecount::Int
319 filters::Dict{String,Union{Nothing,String}}
320 timerange::Union{Nothing,Tuple{Int,Int,String,String}}
321 output::Vector{String}
322end
323
324function Analyzer(lines::Vector{String})
325 records = Record[]
326 malformed = Tuple{Int,String}[]
327 for (index, line) in enumerate(lines)
328 fields = tokens(line)
329 reason = malformed_reason(fields)
330 if reason === nothing
331 push!(records, record_of(index, fields))
332 else
333 push!(malformed, (index, reason))
334 end
335 end
336 sort!(records; by=r -> (r.seconds, r.line))
337 filters = Dict{String,Union{Nothing,String}}(key => nothing for key in FILTER_KEYS)
338 return Analyzer(records, malformed, length(lines), filters, nothing, String[])
339end
340
341function emit!(a::Analyzer, args...)
342 push!(a.output, join(string.(args), " "))
343end
344
345function selected(a::Analyzer)
346 result = Record[]
347 for r in a.records
348 window = a.timerange
349 window !== nothing && (r.seconds < window[1] || r.seconds > window[2]) && continue
350 a.filters["METHOD"] !== nothing && r.method != a.filters["METHOD"] && continue
351 a.filters["STATUS"] !== nothing &&
352 !status_matches(a.filters["STATUS"], r.status) && continue
353 a.filters["IP"] !== nothing && r.ip != a.filters["IP"] && continue
354 a.filters["PATH"] !== nothing && r.path != a.filters["PATH"] && continue
355 push!(result, r)
356 end
357 return result
358end

360function bar_line!(a, label, number, maximum)
361 if number == 0
362 emit!(a, label, number)
363 else
364 emit!(a, label, number, bar(number, maximum))
365 end
366end

368function run!(a::Analyzer, name, args, rows)
369 if name == "SUMMARY"
370 emit!(a, "SUMMARY", "lines", a.linecount, "valid", length(a.records),
371 "malformed", length(a.malformed))
372 emit!(a, "requests", length(rows), "bytes", sum((r.bytes for r in rows); init=0),
373 "ips", length(Set(r.ip for r in rows)),
374 "paths", length(Set(r.path for r in rows)))
375 if isempty(rows)
376 emit!(a, "span", "none")
377 else
378 emit!(a, "span", first(rows).timestamp, last(rows).timestamp,
379 last(rows).seconds - first(rows).seconds)
380 end
381
382 elseif name == "MALFORMED"
383 emit!(a, "MALFORMED", length(a.malformed))
384 for reason in REASONS
385 indices = [index for (index, why) in a.malformed if why == reason]
386 isempty(indices) || emit!(a, reason, length(indices), indices...)
387 end
388
389 elseif name == "STATUS"
390 emit!(a, "STATUS", length(rows))
391 grouped = class_counts(rows)
392 for i in eachindex(CLASSES)
393 emit!(a, CLASSES[i], grouped[i], percent(grouped[i], length(rows)))
394 end
395 status = Dict{Int,Int}()
396 for row in rows
397 status[row.status] = get(status, row.status, 0) + 1
398 end
399 for code in sort!(collect(keys(status)))
400 emit!(a, code, status[code], percent(status[code], length(rows)))
401 end
402
403 elseif name == "METHODS"
404 values = count_by(rows, r -> r.method)
405 emit!(a, "METHODS", length(rows))
406 for method in ranked(values)
407 emit!(a, method, values[method], percent(values[method], length(rows)))
408 end
409
410 elseif name == "TOP"
411 kind = args[1]
412 values = count_by(rows, r -> kind == "PATHS" ? r.path : r.ip)
413 names = ranked(values; tie=kind == "IPS" ? compare_ip : ((a,b)->a<b))
414 shown = min(parse(Int, args[2]), length(names))
415 emit!(a, "TOP", kind, shown, "of", length(names))
416 rank = 0
417 for i in 1:shown
418 (i == 1 || values[names[i]] != values[names[i-1]]) && (rank = i)
419 emit!(a, rank, names[i], values[names[i]], percent(values[names[i]], length(rows)))
420 end
421
422 elseif name == "HOURLY"
423 values = zeros(Int, 24)
424 for row in rows
425 values[row.hour+1] += 1
426 end
427 highest = maximum(values)
428 firsthour = findfirst(==(highest), values)-1
429 emit!(a, "HOURLY", length(rows), "peak", padded(firsthour), highest)
430 for i in 1:24
431 bar_line!(a, padded(i-1), values[i], highest)
432 end
433
434 elseif name == "TIMELINE"
435 minutes = parse(Int, args[1])
436 width = minutes * 60
437 firstbucket, counts = bucket_counts([r.seconds ÷ width for r in rows])
438 emit!(a, "TIMELINE", minutes, length(counts))
439 largest = maximum(counts)
440 for i in eachindex(counts)
441 bar_line!(a, interval_label((firstbucket + i - 1) * width), counts[i], largest)
442 end
443
444 elseif name == "BANDWIDTH"
445 days = group_by(rows, r -> r.date)
446 volume = sum(r.bytes for r in rows)
447 emit!(a, "BANDWIDTH", length(days), volume, bytes_text(volume))
448 for date in sort!(collect(keys(days)))
449 daybytes = sum(r.bytes for r in days[date])
450 emit!(a, date, length(days[date]), daybytes, bytes_text(daybytes))
451 end
452
453 elseif name == "VISITORS"
454 firstvisit = Dict{String,String}()
455 for row in rows
456 if !haskey(firstvisit, row.ip) || row.date < firstvisit[row.ip]
457 firstvisit[row.ip] = row.date
458 end
459 end
460 days = group_by(rows, r -> r.date)
461 emit!(a, "VISITORS", length(days), length(firstvisit))
462 for date in sort!(collect(keys(days)))
463 ips = Set(row.ip for row in days[date])
464 fresh = count(ip -> firstvisit[ip] == date, ips)
465 emit!(a, date, "unique", length(ips), "new", fresh, "returning", length(ips) - fresh)
466 end
467
468 elseif name == "LATENCY"
469 label = isempty(args) ? "all" : normalize_path(args[1])
470 chosen = isempty(args) ? rows : filter(r -> r.path == label, rows)
471 if isempty(chosen)
472 emit!(a, "LATENCY", "no records")
473 else
474 values = sort!([r.latency for r in chosen])
475 n, trim = length(values), length(values) ÷ 10
476 trimmed = values[trim+1:n-trim]
477 emit!(a, "LATENCY", label, "count", n, "min", first(values),
478 "p50", percentile(values, 50), "p90", percentile(values, 90),
479 "p99", percentile(values, 99), "max", last(values),
480 "mean", decimal(sum(values), n, 2),
481 "trimmed", decimal(sum(trimmed), length(trimmed), 2))
482 end
483
484 elseif name == "HISTOGRAM"
485 width = parse(Int, args[1])
486 firstbucket, counts = bucket_counts([r.latency ÷ width for r in rows])
487 emit!(a, "HISTOGRAM", width, length(counts))
488 largest = maximum(counts)
489 for i in eachindex(counts)
490 low = (firstbucket + i - 1) * width
491 bar_line!(a, string(low, "-", low + width - 1), counts[i], largest)
492 end
493
494 elseif name == "APDEX"
495 threshold = parse(Int, args[1])
496 overall = apdex(rows, threshold)
497 emit!(a, "APDEX", threshold, overall.text, overall.rating,
498 "satisfied", overall.satisfied, "tolerating", overall.tolerating,
499 "frustrated", overall.frustrated)
500 paths = [(path=path, score=apdex(group, threshold))
501 for (path, group) in group_by(rows, r -> r.path)]
502 sort!(paths; lt=(x,y)->begin
503 a_score, b_score = x.score, y.score
504 left = a_score.points * b_score.total
505 right = b_score.points * a_score.total
506 left != right && return left < right
507 a_score.count != b_score.count && return a_score.count > b_score.count
508 return x.path < y.path
509 end)
510 for item in paths
511 emit!(a, item.path, item.score.text, item.score.rating, item.score.count)
512 end
513
514 elseif name == "ERRORS"
515 failed = count(r -> r.status >= 400, rows)
516 emit!(a, "ERRORS", failed, "of", length(rows), percent(failed, length(rows)))
517 paths = NamedTuple[]
518 for (path, group) in group_by(rows, r -> r.path)
519 client = count(r -> r.status ÷ 100 == 4, group)
520 server = count(r -> r.status ÷ 100 == 5, group)
521 client + server > 0 && push!(paths,
522 (path=path, count=length(group), client=client, server=server))
523 end
524 sort!(paths; lt=(x,y)->begin
525 xf, yf = x.client+x.server, y.client+y.server
526 yf * x.count != xf * y.count && return yf * x.count < xf * y.count
527 yf != xf && return yf > xf
528 return x.path < y.path
529 end)
530 for row in paths
531 emit!(a, row.path, row.count, row.client, row.server,
532 percent(row.client + row.server, row.count))
533 end
534
535 elseif name == "SLOWEST"
536 endpoints = NamedTuple[]
537 for (key, group) in group_by(rows, r -> r.method * "\u0001" * r.path)
538 pair = split(key, '\u0001'; limit=2)
539 push!(endpoints, (method=pair[1], path=pair[2], count=length(group),
540 total=sum(r.latency for r in group),
541 maximum=maximum(r.latency for r in group)))
542 end
543 sort!(endpoints; lt=(x,y)->begin
544 left, right = y.total * x.count, x.total * y.count
545 left != right && return left < right
546 x.maximum != y.maximum && return x.maximum > y.maximum
547 x.path != y.path && return x.path < y.path
548 return x.method < y.method
549 end)
550 shown = min(parse(Int, args[1]), length(endpoints))
551 emit!(a, "SLOWEST", shown, "of", length(endpoints))
552 for i in 1:shown
553 row = endpoints[i]
554 emit!(a, i, row.method, row.path, "count", row.count,
555 "avg", decimal(row.total, row.count, 2), "max", row.maximum)
556 end
557
558 elseif name == "SESSIONS"
559 ip = args[1]
560 minutes = length(args) == 2 ? parse(Int, args[2]) : 30
561 found = filter(r -> r.ip == ip, rows)
562 if isempty(found)
563 emit!(a, "SESSIONS", "no records")
564 else
565 sessions = [Record[found[1]]]
566 for i in 2:length(found)
567 if found[i].seconds - found[i-1].seconds <= minutes * 60
568 push!(sessions[end], found[i])
569 else
570 push!(sessions, Record[found[i]])
571 end
572 end
573 duration = sum(last(group).seconds - first(group).seconds for group in sessions)
574 emit!(a, "SESSIONS", ip, "timeout", minutes, "count", length(sessions),
575 "requests", length(found),
576 "avg_duration", decimal(duration, length(sessions), 1),
577 "avg_requests", decimal(length(found), length(sessions), 1))
578 for (i, session) in enumerate(sessions)
579 emit!(a, i, first(session).timestamp, last(session).timestamp,
580 last(session).seconds-first(session).seconds, length(session),
581 length(Set(r.path for r in session)))
582 end
583 end
584
585 elseif name == "BURST"
586 width, threshold = parse(Int, args[1]), parse(Int, args[2])
587 failed = filter(r -> r.status >= 400, rows)
588 peaks = [(ip=ip, window=peak(group, width))
589 for (ip, group) in group_by(failed, r -> r.ip)]
590 filter!(x -> x.window.size >= threshold, peaks)
591 sort!(peaks; lt=(x,y)->x.window.size != y.window.size ?
592 x.window.size > y.window.size : compare_ip(x.ip, y.ip))
593 emit!(a, "BURST", width, threshold, length(peaks))
594 for item in peaks
595 emit!(a, item.ip, item.window.size,
596 item.window.start.timestamp, item.window.last.timestamp)
597 end
598
599 elseif name == "PEAK"
600 width = parse(Int, args[1])
601 value = peak(rows, width)
602 emit!(a, "PEAK", width, value.size,
603 value.start.timestamp, value.last.timestamp)
604
605 elseif name == "GAPS"
606 threshold = parse(Int, args[1])
607 gaps = Tuple{Record,Record}[]
608 for i in 2:length(rows)
609 if rows[i].seconds - rows[i-1].seconds >= threshold
610 push!(gaps, (rows[i-1], rows[i]))
611 end
612 end
613 emit!(a, "GAPS", threshold, length(gaps))
614 for (before, after) in gaps
615 emit!(a, before.timestamp, after.timestamp, after.seconds-before.seconds)
616 end
617
618 elseif name == "IP"
619 ip = args[1]
620 found = filter(r -> r.ip == ip, rows)
621 if isempty(found)
622 emit!(a, "IP", "no records")
623 else
624 paths = count_by(found, r -> r.path)
625 emit!(a, "IP", ip, "requests", length(found),
626 "bytes", sum(r.bytes for r in found),
627 "errors", count(r -> r.status >= 400, found),
628 "paths", length(paths))
629 emit!(a, "first", first(found).timestamp, "last", last(found).timestamp)
630 methods = count_by(found, r -> r.method)
631 parts = Any["methods"]
632 for method in ranked(methods)
633 append!(parts, (method, methods[method]))
634 end
635 emit!(a, parts...)
636 classes = class_counts(found)
637 parts = Any["classes"]
638 for i in eachindex(CLASSES)
639 append!(parts, (CLASSES[i], classes[i]))
640 end
641 emit!(a, parts...)
642 for path in Iterators.take(ranked(paths), 3)
643 emit!(a, "path", path, paths[path])
644 end
645 end
646
647 elseif name == "RANGE"
648 firsttime = seconds_of(args[1], args[2])
649 lasttime = seconds_of(args[3], args[4])
650 ensure(firsttime <= lasttime, "invalid range")
651 fromlabel, tolabel = args[1] * " " * args[2], args[3] * " " * args[4]
652 a.timerange = (firsttime, lasttime, fromlabel, tolabel)
653 emit!(a, "RANGE", fromlabel, tolabel, "active", length(selected(a)))
654
655 elseif name == "FILTER"
656 key, value = args
657 key == "PATH" && (value = normalize_path(value))
658 a.filters[key] = value
659 emit!(a, "FILTER", key, value, "active", length(selected(a)))
660
661 elseif name == "UNFILTER"
662 key = args[1]
663 if key == "RANGE"
664 ensure(a.timerange !== nothing, "not filtered")
665 a.timerange = nothing
666 else
667 ensure(a.filters[key] !== nothing, "not filtered")
668 a.filters[key] = nothing
669 end
670 emit!(a, "UNFILTER", key, "active", length(selected(a)))
671
672 elseif name == "RESET"
673 a.timerange = nothing
674 for key in FILTER_KEYS
675 a.filters[key] = nothing
676 end
677 emit!(a, "RESET", "active", length(selected(a)))
678
679 elseif name == "STATE"
680 value = a.timerange === nothing ? "-" :
681 a.timerange[3] * " " * a.timerange[4]
682 parts = Any["STATE", "active", length(rows), "range", value]
683 for key in FILTER_KEYS
684 append!(parts, (lowercase(key),
685 a.filters[key] === nothing ? "-" : a.filters[key]))
686 end
687 emit!(a, parts...)
688 end
689end
690
691function execute!(a::Analyzer, line)
692 fields = tokens(line)
693 name = isempty(fields) ? "" : fields[1]
694 args = fields[2:end]
695 try
696 ensure(haskey(FORMS, name), "unknown query")
697 form = FORMS[name]
698 optional = name in ["LATENCY", "SESSIONS"] && length(args) == length(form)-1
699 ensure(length(args) == length(form) || optional, "wrong arguments")
700 for i in eachindex(args)
701 ensure(valid_argument(form[i], args[i], args), ERRORS[form[i]])
702 end
703 rows = selected(a)
704 if name in REQUIRE_RECORDS && isempty(rows)
705 emit!(a, name, "no records")
706 else
707 run!(a, name, args, rows)
708 end
709 catch error
710 error isa QueryError || rethrow()
711 emit!(a, "ERROR", error.message)
712 end
713end

715function main()
716 lines = readlines(stdin)
717 separator = findfirst(==("---"), lines)
718 ending = findnext(==("END"), lines, separator + 1)
719 analyzer = Analyzer(lines[1:separator-1])
720 for i in separator+1:ending-1
721 execute!(analyzer, lines[i])
722 end
723 isempty(analyzer.output) || println(join(analyzer.output, "\n"))
724end
