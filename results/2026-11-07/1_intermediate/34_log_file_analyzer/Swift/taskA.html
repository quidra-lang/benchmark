8 "SUMMARY": [], "MALFORMED": [], "STATUS": [], "METHODS": [],
9 "TOP": ["kind", "limit"], "HOURLY": [], "TIMELINE": ["minutes"],

43func require(_ condition: Bool, _ message: String) throws {
44 if !condition { throw QueryError(message: message) }
45}

47func words(_ line: String) -> [String] {
48 line.split(separator: " ").map(String.init)
49}
50
51func matches(_ text: String, _ pattern: String) -> Bool {
52 text.range(of: pattern, options: .regularExpression) != nil
53}
54
55func canonical(_ text: String, _ maximum: Int) -> Bool {
56 text.utf8.count <= maximum && matches(text, "^(0|[1-9][0-9]*)$")
57}
58
59func parseDate(_ text: String) -> Date? {
60 guard matches(text, "^[0-9]{4}-[0-9]{2}-[0-9]{2}$") else { return nil }
61 let parts = text.split(separator: "-").map { Int($0)! }
62 guard (2000...2059).contains(parts[0]), (1...12).contains(parts[1]),
63 (1...31).contains(parts[2]),
64 let date = calendar.date(from: DateComponents(year: parts[0], month: parts[1], day: parts[2]))
65 else { return nil }
66 let actual = calendar.dateComponents([.year, .month, .day], from: date)
67 return actual.year == parts[0] && actual.month == parts[1] && actual.day == parts[2] ? date : nil
68}
69
70func validTime(_ text: String) -> Bool {
71 guard matches(text, "^[0-9]{2}:[0-9]{2}:[0-9]{2}$") else { return false }
72 let parts = text.split(separator: ":").map { Int($0)! }
73 return parts[0] < 24 && parts[1] < 60 && parts[2] < 60
74}
75
76func validIP(_ text: String) -> Bool {
77 let parts = text.split(separator: ".", omittingEmptySubsequences: false).map(String.init)
78 return parts.count == 4 && parts.allSatisfy { canonical($0, 3) && Int($0)! <= 255 }
79}
80
81func validStatus(_ text: String) -> Bool {
82 matches(text, "^[0-9]{3}$") && (100...599).contains(Int(text)!)
83}
84
85func validFilter(_ key: String, _ value: String) -> Bool {
86 switch key {
87 case "METHOD": return methods.contains(value)
88 case "STATUS": return statusClasses.contains(value) || validStatus(value)
89 case "IP": return validIP(value)
90 default: return value.hasPrefix("/")
91 }
92}
93
94func validArgument(_ kind: String, _ value: String, _ arguments: [String]) -> Bool {
95 if let range = ranges[kind] { return canonical(value, 6) && range.contains(Int(value)!) }
96 switch kind {
97 case "kind": return value == "PATHS" || value == "IPS"
98 case "path": return value.hasPrefix("/")
99 case "ip": return validIP(value)
100 case "date": return parseDate(value) != nil
101 case "time": return validTime(value)
102 case "key": return filterKeys.contains(value)
103 case "subject": return value == "RANGE" || filterKeys.contains(value)
104 default: return validFilter(arguments[0], value)
105 }
106}
107
108func malformedReason(_ fields: [String]) -> String? {
109 if fields.count != 8 { return "fields" }
110 if parseDate(fields[0]) == nil { return "date" }
111 if !validTime(fields[1]) { return "time" }
112 if !validIP(fields[2]) { return "ip" }
113 if !methods.contains(fields[3]) { return "method" }
114 if !fields[4].hasPrefix("/") { return "path" }
115 if !validStatus(fields[5]) { return "status" }
116 if !canonical(fields[6], 8) { return "bytes" }
117 if !canonical(fields[7], 6) { return "latency" }
118 return nil
119}

121func secondsOf(_ date: String, _ time: String) -> Int {
122 let days = calendar.dateComponents([.day], from: epoch, to: parseDate(date)!).day!
123 let parts = time.split(separator: ":").map { Int($0)! }
124 return days * 86400 + parts[0] * 3600 + parts[1] * 60 + parts[2]
125}
126
127func intervalLabel(_ seconds: Int) -> String {
128 let date = calendar.date(byAdding: .day, value: seconds / 86400, to: epoch)!
129 let parts = calendar.dateComponents([.year, .month, .day], from: date)
130 return String(format: "%04d-%02d-%02d %02d:%02d", parts.year!, parts.month!, parts.day!,
131 seconds % 86400 / 3600, seconds % 3600 / 60)
132}

134func normalizePath(_ text: String) -> String {
135 let raw = String(text.prefix { $0 != "?" })
136 var parts: [String] = []
137 for part in raw.split(separator: "/") {
138 if part == ".." {
139 if !parts.isEmpty { parts.removeLast() }
140 } else if part != "." {
141 parts.append(String(part))
142 }
143 }
144 return "/" + parts.map { part in
145 if matches(part, "^[0-9]+$") { return "{id}" }
146 if part.count >= 8 && matches(part, "^[0-9a-f]+$") && matches(part, "[0-9]") { return "{hex}" }
147 return part
148 }.joined(separator: "/")
149}
150
151func ipNumber(_ text: String) -> Int {
152 text.split(separator: ".").reduce(0) { $0 * 256 + Int($1)! }
153}
154
155func statusMatches(_ value: String, _ status: Int) -> Bool {
156 statusClasses.contains(value) ? status / 100 == Int(value.prefix(1))! : status == Int(value)!
157}
158
159func decimal(_ numerator: Int, _ denominator: Int, _ places: Int) -> String {
160 let scale = [10, 100, 1000][places - 1]
161 let value = numerator * scale / denominator
162 let digits = String(value % scale)
163 return "\(value / scale)." + String(repeating: "0", count: places - digits.count) + digits
164}
165
166func percent(_ part: Int, _ total: Int) -> String {
167 decimal(part * 100, total, 2) + "%"
168}
169
170func bytesText(_ count: Int) -> String {
171 for (unit, suffix) in [(1073741824, "GB"), (1048576, "MB"), (1024, "KB")] {
172 if count >= unit { return decimal(count, unit, 1) + suffix }
173 }
174 return "\(count)B"
175}
176
177func frequencies<T: Hashable>(_ values: [T]) -> [T: Int] {
178 Dictionary(values.map { ($0, 1) }, uniquingKeysWith: +)
179}
180
181func ranked(_ counts: [String: Int], ip: Bool = false) -> [String] {
182 counts.keys.sorted { left, right in
183 if counts[left]! != counts[right]! { return counts[left]! > counts[right]! }
184 return ip ? ipNumber(left) < ipNumber(right) : left < right
185 }
186}

188struct LogRecord {
189 let line: Int
190 let date: String
191 let timestamp: String
192 let seconds: Int
193 let hour: Int
194 let ip: String
195 let method: String
196 let path: String
197 let status: Int
198 let bytes: Int
199 let latency: Int
200
201 init(line: Int, fields: [String]) {
202 self.line = line
203 date = fields[0]
204 timestamp = fields[0] + " " + fields[1]
205 seconds = secondsOf(fields[0], fields[1])
206 hour = Int(fields[1].prefix(2))!
207 ip = fields[2]
208 method = fields[3]
209 path = normalizePath(fields[4])
210 status = Int(fields[5])!
211 bytes = Int(fields[6])!
212 latency = Int(fields[7])!
213 }
214}

216func sum(_ rows: [LogRecord], _ key: KeyPath<LogRecord, Int>) -> Int {
217 rows.reduce(0) { $0 + $1[keyPath: key] }
218}
219
220func classCounts(_ rows: [LogRecord]) -> [Int] {
221 var counts = Array(repeating: 0, count: 5)
222 for row in rows { counts[row.status / 100 - 1] += 1 }
223 return counts
224}
225
226func buckets(_ values: [Int]) throws -> (first: Int, counts: [Int]) {
227 let first = values.min()!, width = values.max()! - first + 1
228 try require(width <= 60, "too many buckets")
229 var counts = Array(repeating: 0, count: width)
230 for value in values { counts[value - first] += 1 }
231 return (first, counts)
232}
233
234func percentile(_ values: [Int], _ rank: Int) -> Int {
235 values[(rank * values.count + 99) / 100 - 1]
236}
237
238func peak(_ rows: [LogRecord], _ width: Int) -> (count: Int, first: LogRecord, last: LogRecord) {
239 var first = 0, bestFirst = 0, bestLast = 0, count = 0
240 for last in rows.indices {
241 while rows[last].seconds - rows[first].seconds >= width { first += 1 }
242 if last - first + 1 > count {
243 count = last - first + 1
244 bestFirst = first
245 bestLast = last
246 }
247 }
248 return (count, rows[bestFirst], rows[bestLast])
249}

251struct Apdex {
252 let count: Int
253 let satisfied: Int
254 let tolerating: Int
255 var frustrated: Int { count - satisfied - tolerating }
256 var points: Int { satisfied * 2 + tolerating }
257 var total: Int { count * 2 }
258 var text: String { decimal(points, total, 3) }
259 var rating: String {
260 for (limit, name) in [(94, "excellent"), (85, "good"), (70, "fair"), (50, "poor")] {
261 if points * 100 >= total * limit { return name }
262 }
263 return "unacceptable"
264 }
265 init(_ rows: [LogRecord], _ threshold: Int) {
266 count = rows.count
267 satisfied = rows.filter { $0.status < 500 && $0.latency <= threshold }.count
268 tolerating = rows.filter { $0.status < 500 && $0.latency > threshold && $0.latency <= threshold * 4 }.count
269 }
270}

272final class Analyzer {
273 private var records: [LogRecord] = []
274 private var malformed: [(line: Int, reason: String)] = []
275 private let lineCount: Int
276 private var conditions: [String: String] = [:]
277 private var range: (start: Int, end: Int, startText: String, endText: String)?
278 private var output: [String] = []
279
280 init(_ lines: [String]) {
281 lineCount = lines.count
282 for (i, line) in lines.enumerated() {
283 let fields = words(line)
284 if let reason = malformedReason(fields) { malformed.append((i + 1, reason)) }
285 else { records.append(LogRecord(line: i + 1, fields: fields)) }
286 }
287 records.sort { ($0.seconds, $0.line) < ($1.seconds, $1.line) }
288 }
289
290 private func emit(_ parts: Any...) {
291 output.append(parts.map { String(describing: $0) }.joined(separator: " "))
292 }
293
294 private func selected() -> [LogRecord] {
295 records.filter { row in
296 if let range, row.seconds < range.start || row.seconds > range.end { return false }
297 if let value = conditions["METHOD"], row.method != value { return false }
298 if let value = conditions["STATUS"], !statusMatches(value, row.status) { return false }
299 if let value = conditions["IP"], row.ip != value { return false }
300 return conditions["PATH"] == nil || row.path == conditions["PATH"]!
301 }
302 }
303
304 private func barLine(_ label: String, _ count: Int, _ maximum: Int) {
305 if count == 0 { emit(label, count) }
306 else { emit(label, count, String(repeating: "#", count: (count * 30 + maximum - 1) / maximum)) }
307 }
308
309 func execute(_ line: String) -> [String] {
310 output = []
311 let tokens = words(line), name = words(line).first ?? ""
312 let args = Array(tokens.dropFirst())
313 do {
314 try require(forms[name] != nil, "unknown query")
315 let form = forms[name]!
316 let optional = (name == "LATENCY" || name == "SESSIONS") && args.count == form.count - 1
317 try require(args.count == form.count || optional, "wrong arguments")
318 for i in args.indices {
319 try require(validArgument(form[i], args[i], args), argumentErrors[form[i]]!)
320 }
321 let rows = selected()
322 if requireRecords.contains(name) && rows.isEmpty { emit(name, "no records") }
323 else { try run(name, args, rows) }
324 } catch let error as QueryError {
325 output = ["ERROR " + error.message]
326 } catch {
327 fatalError("Unexpected failure: \(error)")
328 }
329 return output
330 }
331
332 private func run(_ name: String, _ a: [String], _ rows: [LogRecord]) throws {
333 switch name {
334 case "SUMMARY":
335 emit("SUMMARY", "lines", lineCount, "valid", records.count, "malformed", malformed.count)
336 emit("requests", rows.count, "bytes", sum(rows, \.bytes),
337 "ips", Set(rows.map(\.ip)).count, "paths", Set(rows.map(\.path)).count)
338 if let first = rows.first, let last = rows.last {
339 emit("span", first.timestamp, last.timestamp, last.seconds - first.seconds)
340 } else { emit("span", "none") }
341 case "MALFORMED":
342 emit("MALFORMED", malformed.count)
343 for reason in reasons {
344 let numbers = malformed.filter { $0.reason == reason }.map { String($0.line) }
345 if !numbers.isEmpty { emit(reason, numbers.count, numbers.joined(separator: " ")) }
346 }
347 case "STATUS":
348 emit("STATUS", rows.count)
349 let classes = classCounts(rows), counts = frequencies(rows.map(\.status))
350 for i in classes.indices { emit(statusClasses[i], classes[i], percent(classes[i], rows.count)) }
351 for status in counts.keys.sorted() { emit(status, counts[status]!, percent(counts[status]!, rows.count)) }
352 case "METHODS":
353 emit("METHODS", rows.count)
354 let counts = frequencies(rows.map(\.method))
355 for method in ranked(counts) { emit(method, counts[method]!, percent(counts[method]!, rows.count)) }
356 case "TOP":
357 let kind = a[0]
358 let counts = frequencies(rows.map { kind == "PATHS" ? $0.path : $0.ip })
359 let keys = ranked(counts, ip: kind == "IPS"), shown = min(Int(a[1])!, counts.count)
360 emit("TOP", kind, shown, "of", keys.count)
361 var rank = 0
362 for i in 0..<shown {
363 if i == 0 || counts[keys[i]] != counts[keys[i - 1]] { rank = i + 1 }
364 emit(rank, keys[i], counts[keys[i]]!, percent(counts[keys[i]]!, rows.count))
365 }
366 case "HOURLY":
367 var counts = Array(repeating: 0, count: 24)
368 for row in rows { counts[row.hour] += 1 }
369 let maximum = counts.max()!
370 emit("HOURLY", rows.count, "peak", String(format: "%02d", counts.firstIndex(of: maximum)!), maximum)
371 for i in counts.indices { barLine(String(format: "%02d", i), counts[i], maximum) }
372 case "TIMELINE":
373 let minutes = Int(a[0])!, width = Int(a[0])! * 60
374 let b = try buckets(rows.map { $0.seconds / width })
375 emit("TIMELINE", minutes, b.counts.count)
376 for i in b.counts.indices { barLine(intervalLabel((b.first + i) * width), b.counts[i], b.counts.max()!) }
377 case "BANDWIDTH":
378 let days = Dictionary(grouping: rows, by: \.date), total = sum(rows, \.bytes)
379 emit("BANDWIDTH", days.count, total, bytesText(total))
380 for date in days.keys.sorted() {
381 let group = days[date]!, total = sum(days[date]!, \.bytes)
382 emit(date, group.count, total, bytesText(total))
383 }
384 case "VISITORS":
385 var firstDates: [String: String] = [:]
386 for row in rows where firstDates[row.ip] == nil { firstDates[row.ip] = row.date }
387 let days = Dictionary(grouping: rows, by: \.date)
388 emit("VISITORS", days.count, firstDates.count)
389 for date in days.keys.sorted() {
390 let ips = Set(days[date]!.map(\.ip)), fresh = Set(days[date]!.map(\.ip)).filter { firstDates[$0] == date }.count
391 emit(date, "unique", ips.count, "new", fresh, "returning", ips.count - fresh)
392 }
393 case "LATENCY":
394 let label = a.isEmpty ? "all" : normalizePath(a[0])
395 let chosen = a.isEmpty ? rows : rows.filter { $0.path == label }
396 guard !chosen.isEmpty else { emit("LATENCY", "no records"); return }
397 let values = chosen.map(\.latency).sorted(), count = chosen.count, trim = chosen.count / 10
398 let trimmed = values[trim..<(count - trim)]
399 emit("LATENCY", label, "count", count, "min", values.first!, "p50", percentile(values, 50),
400 "p90", percentile(values, 90), "p99", percentile(values, 99), "max", values.last!,
401 "mean", decimal(values.reduce(0, +), count, 2),
402 "trimmed", decimal(trimmed.reduce(0, +), trimmed.count, 2))
403 case "HISTOGRAM":
404 let width = Int(a[0])!, b = try buckets(rows.map { $0.latency / Int(a[0])! })
405 emit("HISTOGRAM", width, b.counts.count)
406 for i in b.counts.indices {
407 let lower = (b.first + i) * width
408 barLine("\(lower)-\(lower + width - 1)", b.counts[i], b.counts.max()!)
409 }
410 case "APDEX":
411 let threshold = Int(a[0])!, overall = Apdex(rows, Int(a[0])!)
412 emit("APDEX", threshold, overall.text, overall.rating, "satisfied", overall.satisfied,
413 "tolerating", overall.tolerating, "frustrated", overall.frustrated)
414 let paths = Dictionary(grouping: rows, by: \.path).map { (path: $0.key, score: Apdex($0.value, threshold)) }
415 .sorted { x, y in
416 let left = x.score.points * y.score.total, right = y.score.points * x.score.total
417 if left != right { return left < right }
418 if x.score.count != y.score.count { return x.score.count > y.score.count }
419 return x.path < y.path
420 }
421 for row in paths { emit(row.path, row.score.text, row.score.rating, row.score.count) }
422 case "ERRORS":
423 let failed = rows.filter { $0.status >= 400 }.count
424 emit("ERRORS", failed, "of", rows.count, percent(failed, rows.count))
425 let paths = Dictionary(grouping: rows, by: \.path).map { entry in
426 (path: entry.key, count: entry.value.count,
427 client: entry.value.filter { $0.status / 100 == 4 }.count,
428 server: entry.value.filter { $0.status / 100 == 5 }.count)
429 }.filter { $0.client + $0.server > 0 }.sorted { x, y in
430 let xf = x.client + x.server, yf = y.client + y.server
431 if xf * y.count != yf * x.count { return xf * y.count > yf * x.count }
432 if xf != yf { return xf > yf }
433 return x.path < y.path
434 }
435 for row in paths { emit(row.path, row.count, row.client, row.server, percent(row.client + row.server, row.count)) }
436 case "SLOWEST":
437 let paths = Dictionary(grouping: rows, by: { $0.method + "\u{1}" + $0.path }).values.map { group in
438 (method: group[0].method, path: group[0].path, count: group.count,
439 total: sum(group, \.latency), maximum: group.map(\.latency).max()!)
440 }.sorted { x, y in
441 if x.total * y.count != y.total * x.count { return x.total * y.count > y.total * x.count }
442 if x.maximum != y.maximum { return x.maximum > y.maximum }
443 return (x.path, x.method) < (y.path, y.method)
444 }
445 let shown = min(Int(a[0])!, paths.count)
446 emit("SLOWEST", shown, "of", paths.count)
447 for (i, row) in paths.prefix(shown).enumerated() {
448 emit(i + 1, row.method, row.path, "count", row.count, "avg", decimal(row.total, row.count, 2), "max", row.maximum)
449 }
450 case "SESSIONS":
451 let ip = a[0], minutes = a.count == 2 ? Int(a[1])! : 30
452 let found = rows.filter { $0.ip == ip }
453 guard !found.isEmpty else { emit("SESSIONS", "no records"); return }
454 var sessions = [[found[0]]]
455 for i in 1..<found.count {
456 if found[i].seconds - found[i - 1].seconds <= minutes * 60 { sessions[sessions.count - 1].append(found[i]) }
457 else { sessions.append([found[i]]) }
458 }
459 let duration = sessions.reduce(0) { $0 + $1.last!.seconds - $1.first!.seconds }
460 emit("SESSIONS", ip, "timeout", minutes, "count", sessions.count, "requests", found.count,
461 "avg_duration", decimal(duration, sessions.count, 1), "avg_requests", decimal(found.count, sessions.count, 1))
462 for (i, session) in sessions.enumerated() {
463 emit(i + 1, session.first!.timestamp, session.last!.timestamp,
464 session.last!.seconds - session.first!.seconds, session.count, Set(session.map(\.path)).count)
465 }
466 case "BURST":
467 let width = Int(a[0])!, threshold = Int(a[1])!
468 let groups = Dictionary(grouping: rows.filter { $0.status >= 400 }, by: \.ip)
469 let peaks = groups.map { (ip: $0.key, window: peak($0.value, width)) }.filter { $0.window.count >= threshold }
470 .sorted { x, y in
471 if x.window.count != y.window.count { return x.window.count > y.window.count }
472 return ipNumber(x.ip) < ipNumber(y.ip)
473 }
474 emit("BURST", width, threshold, peaks.count)
475 for row in peaks { emit(row.ip, row.window.count, row.window.first.timestamp, row.window.last.timestamp) }
476 case "PEAK":
477 let width = Int(a[0])!, window = peak(rows, Int(a[0])!)
478 emit("PEAK", width, window.count, window.first.timestamp, window.last.timestamp)
479 case "GAPS":
480 let width = Int(a[0])!
481 let gaps = rows.indices.dropFirst().filter { rows[$0].seconds - rows[$0 - 1].seconds >= width }
482 emit("GAPS", width, gaps.count)
483 for i in gaps { emit(rows[i - 1].timestamp, rows[i].timestamp, rows[i].seconds - rows[i - 1].seconds) }
484 case "IP":
485 let found = rows.filter { $0.ip == a[0] }
486 guard !found.isEmpty else { emit("IP", "no records"); return }
487 let paths = frequencies(found.map(\.path)), methods = frequencies(found.map(\.method))
488 emit("IP", a[0], "requests", found.count, "bytes", sum(found, \.bytes),
489 "errors", found.filter { $0.status >= 400 }.count, "paths", paths.count)
490 emit("first", found.first!.timestamp, "last", found.last!.timestamp)
491 emit("methods", ranked(methods).map { "\($0) \(methods[$0]!)" }.joined(separator: " "))
492 let counts = classCounts(found)
493 emit("classes", counts.indices.map { "\(statusClasses[$0]) \(counts[$0])" }.joined(separator: " "))
494 for path in ranked(paths).prefix(3) { emit("path", path, paths[path]!) }
495 case "RANGE":
496 let start = secondsOf(a[0], a[1]), end = secondsOf(a[2], a[3])
497 try require(start <= end, "invalid range")
498 range = (start, end, a[0] + " " + a[1], a[2] + " " + a[3])
499 emit("RANGE", range!.startText, range!.endText, "active", selected().count)
500 case "FILTER":
501 let key = a[0], value = a[0] == "PATH" ? normalizePath(a[1]) : a[1]
502 conditions[key] = value
503 emit("FILTER", key, value, "active", selected().count)
504 case "UNFILTER":
505 let key = a[0]
506 if key == "RANGE" {
507 try require(range != nil, "not filtered")
508 range = nil
509 } else {
510 try require(conditions[key] != nil, "not filtered")
511 conditions.removeValue(forKey: key)
512 }
513 emit("UNFILTER", key, "active", selected().count)
514 case "RESET":
515 range = nil
516 conditions.removeAll()
