static const char *decimal(int64_t numerator, int64_t denominator, int places, char *buffer) {

    int64_t value = numerator * scale / denominator;
    sprintf(buffer, "%" PRId64 ".%0*" PRId64, value / scale, places, value % scale);
    return buffer;
