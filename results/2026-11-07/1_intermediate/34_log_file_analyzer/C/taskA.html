114static bool digits(const char *text) {
115 if (!*text) return false;
116 for (; *text; ++text)
117 if (*text < '0' || *text > '9') return false;
118 return true;

121static bool canonical(const char *text, size_t limit) {
122 size_t n = strlen(text);
123 return n > 0 && n <= limit && digits(text) && (n == 1 || text[0] != '0');

126static bool leap(int year) {
127 return year % 4 == 0 && (year % 100 != 0 || year % 400 == 0);

130static int month_days(int year, int month) {
131 static const int days[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
132 return month == 2 && leap(year) ? 29 : days[month - 1];

135static bool valid_date(const char *text) {
136 if (strlen(text) != 10 || text[4] != '-' || text[7] != '-') return false;
137 for (int i = 0; i < 10; ++i)
138 if (i != 4 && i != 7 && (text[i] < '0' || text[i] > '9')) return false;
139 int year = atoi(text), month = atoi(text + 5), day = atoi(text + 8);
140 return year >= 2000 && year <= 2059 && month >= 1 && month <= 12 &&
141 day >= 1 && day <= month_days(year, month);
142}

144static bool valid_time(const char *text) {
145 if (strlen(text) != 8 || text[2] != ':' || text[5] != ':') return false;
146 for (int i = 0; i < 8; ++i)
147 if (i != 2 && i != 5 && (text[i] < '0' || text[i] > '9')) return false;
148 return atoi(text) < 24 && atoi(text + 3) < 60 && atoi(text + 6) < 60;
149}

151static bool parse_ip(const char *text, uint32_t *number) {
152 uint32_t result = 0;
153 for (int i = 0; i < 4; ++i) {
154 const char *start = text;
155 int value = 0;
156 while (*text >= '0' && *text <= '9') {
157 if (text - start >= 3) return false;
158 value = value * 10 + *text++ - '0';
159 }
160 if (text == start || (text - start > 1 && *start == '0') || value > 255)
161 return false;
162 result = result * 256 + (uint32_t)value;
163 if (i < 3) {
164 if (*text != '.') return false;
165 ++text;
166 } else if (*text) return false;
167 }
168 *number = result;
169 return true;
170}

185static bool filter_value(const char *key, const char *value) {
186 uint32_t ip;
187 switch (name_index(key, filter_keys, 4)) {
188 case 0: return name_index(value, methods, 7) >= 0;
189 case 1: return status_class(value) || valid_status(value);
190 case 2: return parse_ip(value, &ip);
191 case 3: return value[0] == '/';
192 default: return false;
193 }

196static bool valid_argument(Argument kind, const char *value, char **args) {
197 static const int upper[] = {100, 1440, 100000, 100000, 86400, 1000};
198 if (kind <= TIMES_ARG)
199 return canonical(value, 6) && atoi(value) >= 1 && atoi(value) <= upper[kind];
200 uint32_t ip;
201 switch (kind) {
202 case KIND_ARG: return strcmp(value, "PATHS") == 0 || strcmp(value, "IPS") == 0;
203 case PATH_ARG: return value[0] == '/';
204 case IP_ARG: return parse_ip(value, &ip);
205 case DATE_ARG: return valid_date(value);
206 case TIME_ARG: return valid_time(value);
207 case KEY_ARG: return name_index(value, filter_keys, 4) >= 0;
208 case SUBJECT_ARG: return strcmp(value, "RANGE") == 0 || name_index(value, filter_keys, 4) >= 0;
209 case VALUE_ARG: return filter_value(args[0], value);
210 default: return false;
211 }

228static int64_t seconds_of(const char *date, const char *clock) {
229 int year = atoi(date), month = atoi(date + 5);
230 int64_t days = atoi(date + 8) - 1;
231 for (int y = 2000; y < year; ++y) days += leap(y) ? 366 : 365;
232 for (int m = 1; m < month; ++m) days += month_days(year, m);
233 return days * 86400 + atoi(clock) * 3600 + atoi(clock + 3) * 60 + atoi(clock + 6);
234}

236static void interval_label(int64_t seconds, char *text) {
237 int days = (int)(seconds / 86400), year = 2000, month = 1;
238 int minutes = (int)(seconds % 86400 / 60);
239 while (days >= (leap(year) ? 366 : 365)) {
240 days -= leap(year) ? 366 : 365;
241 ++year;
242 }
243 while (days >= month_days(year, month)) days -= month_days(year, month++);
244 sprintf(text, "%04d-%02d-%02d %02d:%02d", year, month, days + 1, minutes / 60, minutes % 60);
245}

247static void normalize_path(const char *raw, char *result) {
248 char copy[MAX_LINE + 1], *parts[MAX_LINE], *p;
249 strcpy(copy, raw);
250 p = strchr(copy, '?');
251 if (p) *p = '\0';
252 int n = 0;
253 for (p = strtok(copy, "/"); p; p = strtok(NULL, "/")) {
254 if (strcmp(p, "..") == 0) {
255 if (n) --n;
256 } else if (strcmp(p, ".") != 0) parts[n++] = p;
257 }
258 strcpy(result, "/");
259 for (int i = 0; i < n; ++i) {
260 p = parts[i];
261 bool hex = strlen(p) >= 8, has_digit = false;
262 for (const char *c = p; *c; ++c) {
263 bool digit = *c >= '0' && *c <= '9';
264 has_digit |= digit;
265 hex &= digit || (*c >= 'a' && *c <= 'f');
266 }
267 if (i) strcat(result, "/");
268 strcat(result, digits(p) ? "{id}" : hex && has_digit ? "{hex}" : p);
269 }
270}

272static const char *decimal(int64_t numerator, int64_t denominator, int places, char *buffer) {
273 int64_t scale = places == 1 ? 10 : places == 2 ? 100 : 1000;
274 int64_t value = numerator * scale / denominator;
275 sprintf(buffer, "%" PRId64 ".%0*" PRId64, value / scale, places, value % scale);
276 return buffer;
277}

285static const char *bytes_text(int64_t bytes, char *buffer) {
286 int64_t unit = bytes >= 1073741824 ? 1073741824 : bytes >= 1048576 ? 1048576 : 1024;
287 if (bytes < 1024) sprintf(buffer, "%" PRId64 "B", bytes);
288 else {
289 decimal(bytes, unit, 1, buffer);
290 strcat(buffer, unit == 1073741824 ? "GB" : unit == 1048576 ? "MB" : "KB");
291 }
292 return buffer;
293}

313static int key_order(const Record *a, const Record *b, GroupKey key) {
314 if (key == BY_IP) return compare_integer(a->ip_number, b->ip_number);
315 if (key == BY_METHOD) return strcmp(a->method, b->method);
316 if (key == BY_DATE) return strcmp(a->date, b->date);
317 int order = strcmp(a->path, b->path);
318 return order || key != BY_ENDPOINT ? order : strcmp(a->method, b->method);
319}

321static int grouped_record_order(const void *left, const void *right) {
322 const Record *a = *(Record *const *)left, *b = *(Record *const *)right;
323 int order = key_order(a, b, group_key);
324 return order ? order : record_order(a, b);
325}

327static void accumulate(Group *group, Record *row, int threshold) {
328 ++group->count;
329 group->bytes += row->bytes;
330 group->latency += row->latency;
331 if (row->latency > group->maximum) group->maximum = row->latency;
332 group->client += row->status / 100 == 4;
333 group->server += row->status / 100 == 5;
334 if (row->status < 500) {
335 group->satisfied += row->latency <= threshold;
336 group->tolerating += row->latency > threshold && row->latency <= 4 * threshold;
337 }
338}

340static int group_rows(Record **rows, int n, GroupKey key, int threshold,
341 Group *groups, Record **ordered) {
342 memcpy(ordered, rows, (size_t)n * sizeof(*rows));
343 group_key = key;
344 qsort(ordered, n, sizeof(*ordered), grouped_record_order);
345 int count = 0;
346 for (int i = 0; i < n; ++i) {
347 if (i == 0 || key_order(ordered[i - 1], ordered[i], key))
348 groups[count++] = (Group){.sample = ordered[i], .begin = i};
349 accumulate(&groups[count - 1], ordered[i], threshold);
350 }
351 return count;
352}

354static int unique_count(Record **rows, int n, GroupKey key) {
355 Record *ordered[MAX_RECORDS];
356 memcpy(ordered, rows, (size_t)n * sizeof(*rows));
357 group_key = key;
358 qsort(ordered, n, sizeof(*ordered), grouped_record_order);
359 int count = 0;
360 for (int i = 0; i < n; ++i)
361 count += i == 0 || key_order(ordered[i - 1], ordered[i], key) != 0;
362 return count;
363}

368 return order ? order : key_order(a->sample, b->sample, group_key);

371static int error_order(const void *left, const void *right) {
372 const Group *a = left, *b = right;
373 int64_t x = a->client + a->server, y = b->client + b->server;
374 int order = compare_integer(y * a->count, x * b->count);
375 if (!order) order = compare_integer(y, x);
376 return order ? order : strcmp(a->sample->path, b->sample->path);
377}

379static int slow_order(const void *left, const void *right) {
380 const Group *a = left, *b = right;
381 int order = compare_integer(b->latency * a->count, a->latency * b->count);
382 if (!order) order = compare_integer(b->maximum, a->maximum);
383 return order ? order : key_order(a->sample, b->sample, BY_ENDPOINT);
384}

386static int score_order(const void *left, const void *right) {
387 const Group *a = left, *b = right;
388 int64_t x = 2 * a->satisfied + a->tolerating, y = 2 * b->satisfied + b->tolerating;
389 int order = compare_integer(x * b->count, y * a->count);
390 if (!order) order = compare_integer(b->count, a->count);
391 return order ? order : strcmp(a->sample->path, b->sample->path);
392}

394static const char *rating(const Group *group) {
395 int64_t points = 100 * (2 * group->satisfied + group->tolerating);
396 int64_t total = 2 * group->count;
397 if (points >= 94 * total) return "excellent";
398 if (points >= 85 * total) return "good";
399 if (points >= 70 * total) return "fair";
400 if (points >= 50 * total) return "poor";
401 return "unacceptable";
402}

404static Window peak(Record **rows, int n, int length) {
405 Window best = {0};
406 int first = 0;
407 for (int last = 0; last < n; ++last) {
408 while (rows[last]->seconds - rows[first]->seconds >= length) ++first;
409 if (last - first + 1 > best.count)

411 }

413}

421static int selected(Analyzer *analyzer, Record **rows) {
422 int count = 0;
423 for (int i = 0; i < analyzer->record_count; ++i) {
424 Record *r = &analyzer->records[i];
425 if (analyzer->has_range && (r->seconds < analyzer->first_second || r->seconds > analyzer->last_second)) continue;
426 if (*analyzer->filters[0] && strcmp(r->method, analyzer->filters[0])) continue;
427 if (*analyzer->filters[1] && !status_matches(analyzer->filters[1], r->status)) continue;
428 if (*analyzer->filters[2] && strcmp(r->ip, analyzer->filters[2])) continue;
429 if (*analyzer->filters[3] && strcmp(r->path, analyzer->filters[3])) continue;
430 rows[count++] = r;
431 }
432 return count;
433}

439static void run_query(Analyzer *analyzer, Query query, int argc, char **args, Record **rows, int n) {
440 Group groups[MAX_RECORDS];
441 Record *ordered[MAX_RECORDS];
442 char first[64], second[64], label[PATH_SIZE];
443 switch (query) {
444 case SUMMARY: {
445 int64_t bytes = 0;
446 for (int i = 0; i < n; ++i) bytes += rows[i]->bytes;
447 printf("SUMMARY lines %d valid %d malformed %d\n", analyzer->line_count, analyzer->record_count, analyzer->malformed_count);
448 printf("requests %d bytes %" PRId64 " ips %d paths %d\n", n, bytes,
449 unique_count(rows, n, BY_IP), unique_count(rows, n, BY_PATH));
450 if (!n) puts("span none");
451 else printf("span %s %s %" PRId64 "\n", rows[0]->timestamp, rows[n - 1]->timestamp, rows[n - 1]->seconds - rows[0]->seconds);
452 break;
453 }
454 case MALFORMED:
455 printf("MALFORMED %d\n", analyzer->malformed_count);
456 for (int reason = 0; reason < 9; ++reason) {
457 int count = 0;
458 for (int i = 0; i < analyzer->malformed_count; ++i) count += analyzer->malformed[i].reason == reason;
459 if (!count) continue;
460 printf("%s %d", reasons[reason], count);
461 for (int i = 0; i < analyzer->malformed_count; ++i)
462 if (analyzer->malformed[i].reason == reason) printf(" %d", analyzer->malformed[i].line);
463 putchar('\n');
464 }
465 break;
466 case STATUS: {
467 int classes[5] = {0}, statuses[600] = {0};
468 for (int i = 0; i < n; ++i) { ++classes[rows[i]->status / 100 - 1]; ++statuses[rows[i]->status]; }
469 printf("STATUS %d\n", n);
470 for (int i = 0; i < 5; ++i) printf("%dxx %d %s\n", i + 1, classes[i], percentage(classes[i], n, first));
471 for (int i = 100; i < 600; ++i)
472 if (statuses[i]) printf("%d %d %s\n", i, statuses[i], percentage(statuses[i], n, first));
473 break;
474 }
475 case METHODS:
476 case TOP: {
477 GroupKey key = query == METHODS ? BY_METHOD : strcmp(args[0], "IPS") == 0 ? BY_IP : BY_PATH;
478 int count = group_rows(rows, n, key, 0, groups, ordered);
479 qsort(groups, count, sizeof(*groups), count_order);
480 int limit = query == METHODS ? count : atoi(args[1]);
481 if (limit > count) limit = count;
482 if (query == METHODS) printf("METHODS %d\n", n);
483 else printf("TOP %s %d of %d\n", args[0], limit, count);
484 int rank = 0;
485 for (int i = 0; i < limit; ++i) {
486 Group *g = &groups[i];
487 const char *text = key == BY_IP ? g->sample->ip : key == BY_METHOD ? g->sample->method : g->sample->path;
488 if (i == 0 || g->count != groups[i - 1].count) rank = i + 1;
489 if (query == TOP) printf("%d ", rank);
490 printf("%s %d %s\n", text, g->count, percentage(g->count, n, first));
491 }
492 break;
493 }
494 case HOURLY: {
495 int counts[24] = {0}, maximum = 0, hour = 0;
496 for (int i = 0; i < n; ++i) ++counts[rows[i]->hour];
497 for (int i = 0; i < 24; ++i)
498 if (counts[i] > maximum) { maximum = counts[i]; hour = i; }
499 printf("HOURLY %d peak %02d %d\n", n, hour, maximum);
500 for (int i = 0; i < 24; ++i) { sprintf(label, "%02d", i); bar_line(label, counts[i], maximum); }
501 break;
502 }
503 case TIMELINE:
504 case HISTOGRAM: {
505 int parameter = atoi(args[0]), width = query == TIMELINE ? parameter * 60 : parameter;
506 int64_t low = INT64_MAX, high = 0, values[MAX_RECORDS];
507 for (int i = 0; i < n; ++i) {
508 values[i] = (query == TIMELINE ? rows[i]->seconds : rows[i]->latency) / width;
509 if (values[i] < low) low = values[i];
510 if (values[i] > high) high = values[i];
511 }
512 if (high - low + 1 > 60) { puts("ERROR too many buckets"); break; }
513 int counts[60] = {0}, maximum = 0, count = (int)(high - low + 1);
514 for (int i = 0; i < n; ++i) ++counts[values[i] - low];
515 for (int i = 0; i < count; ++i) if (counts[i] > maximum) maximum = counts[i];
516 printf("%s %d %d\n", forms[query].name, parameter, count);
517 for (int i = 0; i < count; ++i) {
518 int64_t start = (low + i) * width;
519 if (query == TIMELINE) interval_label(start, label);
520 else sprintf(label, "%" PRId64 "-%" PRId64, start, start + width - 1);
521 bar_line(label, counts[i], maximum);
522 }
523 break;
524 }
525 case BANDWIDTH: {
526 int count = group_rows(rows, n, BY_DATE, 0, groups, ordered);
527 int64_t total = 0;
528 for (int i = 0; i < count; ++i) total += groups[i].bytes;
529 printf("BANDWIDTH %d %" PRId64 " %s\n", count, total, bytes_text(total, first));
530 for (int i = 0; i < count; ++i)
531 printf("%s %d %" PRId64 " %s\n", groups[i].sample->date, groups[i].count, groups[i].bytes, bytes_text(groups[i].bytes, first));
532 break;
533 }
534 case VISITORS: {
535 Group ips[MAX_RECORDS];
536 Record *ip_rows[MAX_RECORDS];
537 int ip_count = group_rows(rows, n, BY_IP, 0, ips, ip_rows);
538 int count = group_rows(rows, n, BY_DATE, 0, groups, ordered);
539 printf("VISITORS %d %d\n", count, ip_count);
540 for (int i = 0; i < count; ++i) {
541 Group *g = &groups[i];
542 int fresh = 0;
543 for (int j = 0; j < ip_count; ++j) fresh += strcmp(ips[j].sample->date, g->sample->date) == 0;
544 int unique = unique_count(ordered + g->begin, g->count, BY_IP);
545 printf("%s unique %d new %d returning %d\n", g->sample->date, unique, fresh, unique - fresh);
546 }
547 break;
548 }
549 case LATENCY: {
550 if (argc) normalize_path(args[0], label); else strcpy(label, "all");
551 int values[MAX_RECORDS], count = 0;
552 for (int i = 0; i < n; ++i)
553 if (!argc || strcmp(rows[i]->path, label) == 0) values[count++] = rows[i]->latency;
554 if (!count) { puts("LATENCY no records"); break; }
555 qsort(values, count, sizeof(*values), latency_order);
556 int trim = count / 10;
557 int64_t sum = 0, trimmed = 0;
558 for (int i = 0; i < count; ++i) {
559 sum += values[i];
560 if (i >= trim && i < count - trim) trimmed += values[i];
561 }
562 printf("LATENCY %s count %d min %d p50 %d p90 %d p99 %d max %d mean %s trimmed %s\n",
563 label, count, values[0], values[(50 * count + 99) / 100 - 1],
564 values[(90 * count + 99) / 100 - 1], values[(99 * count + 99) / 100 - 1], values[count - 1],
565 decimal(sum, count, 2, first), decimal(trimmed, count - 2 * trim, 2, second));
566 break;
567 }
568 case APDEX: {
569 int threshold = atoi(args[0]);
570 Group overall = {0};
571 for (int i = 0; i < n; ++i) accumulate(&overall, rows[i], threshold);
572 printf("APDEX %d %s %s satisfied %d tolerating %d frustrated %d\n", threshold,
573 decimal(2 * overall.satisfied + overall.tolerating, 2 * n, 3, first), rating(&overall),
574 overall.satisfied, overall.tolerating, n - overall.satisfied - overall.tolerating);
575 int count = group_rows(rows, n, BY_PATH, threshold, groups, ordered);
576 qsort(groups, count, sizeof(*groups), score_order);
577 for (int i = 0; i < count; ++i) {
578 Group *g = &groups[i];
579 printf("%s %s %s %d\n", g->sample->path,
580 decimal(2 * g->satisfied + g->tolerating, 2 * g->count, 3, first), rating(g), g->count);
581 }
582 break;
583 }
584 case ERRORS: {
585 int count = group_rows(rows, n, BY_PATH, 0, groups, ordered), failed = 0;
586 for (int i = 0; i < count; ++i) failed += groups[i].client + groups[i].server;
587 printf("ERRORS %d of %d %s\n", failed, n, percentage(failed, n, first));
588 qsort(groups, count, sizeof(*groups), error_order);
589 for (int i = 0; i < count; ++i) {
590 Group *g = &groups[i];
591 if (g->client + g->server)
592 printf("%s %d %d %d %s\n", g->sample->path, g->count, g->client, g->server, percentage(g->client + g->server, g->count, first));
593 }
594 break;
595 }
596 case SLOWEST: {
597 int count = group_rows(rows, n, BY_ENDPOINT, 0, groups, ordered);
598 qsort(groups, count, sizeof(*groups), slow_order);
599 int limit = atoi(args[0]);
600 if (limit > count) limit = count;
601 printf("SLOWEST %d of %d\n", limit, count);
602 for (int i = 0; i < limit; ++i) {
603 Group *g = &groups[i];
604 printf("%d %s %s count %d avg %s max %d\n", i + 1, g->sample->method, g->sample->path,
605 g->count, decimal(g->latency, g->count, 2, first), g->maximum);
606 }
607 break;
608 }
609 case SESSIONS: {
610 int count = 0, timeout = argc == 2 ? atoi(args[1]) : 30;
611 for (int i = 0; i < n; ++i) if (strcmp(rows[i]->ip, args[0]) == 0) ordered[count++] = rows[i];
612 if (!count) { puts("SESSIONS no records"); break; }
613 int starts[MAX_RECORDS + 1], sessions = 1;
614 starts[0] = 0;
615 for (int i = 1; i < count; ++i)
616 if (ordered[i]->seconds - ordered[i - 1]->seconds > timeout * 60) starts[sessions++] = i;
617 starts[sessions] = count;
618 int64_t duration = 0;
619 for (int i = 0; i < sessions; ++i) duration += ordered[starts[i + 1] - 1]->seconds - ordered[starts[i]]->seconds;
620 printf("SESSIONS %s timeout %d count %d requests %d avg_duration %s avg_requests %s\n",
621 args[0], timeout, sessions, count, decimal(duration, sessions, 1, first), decimal(count, sessions, 1, second));
622 for (int i = 0; i < sessions; ++i) {
623 Record *begin = ordered[starts[i]], *end = ordered[starts[i + 1] - 1];
624 int length = starts[i + 1] - starts[i];
625 printf("%d %s %s %" PRId64 " %d %d\n", i + 1, begin->timestamp, end->timestamp,
626 end->seconds - begin->seconds, length, unique_count(ordered + starts[i], length, BY_PATH));
627 }
628 break;
629 }
630 case BURST: {
631 Record *failed[MAX_RECORDS];
632 Burst bursts[MAX_RECORDS];
633 int length = atoi(args[0]), minimum = atoi(args[1]), failure_count = 0, count = 0;
634 for (int i = 0; i < n; ++i) if (rows[i]->status >= 400) failed[failure_count++] = rows[i];
635 int group_count = group_rows(failed, failure_count, BY_IP, 0, groups, ordered);
636 for (int i = 0; i < group_count; ++i) {
637 Group *g = &groups[i];
638 Window window = peak(ordered + g->begin, g->count, length);
639 if (window.count >= minimum)
640 bursts[count++] = (Burst){g->sample->ip, g->sample->ip_number, window,
641 ordered[g->begin + window.first], ordered[g->begin + window.last]};
642 }
643 qsort(bursts, count, sizeof(*bursts), burst_order);
644 printf("BURST %d %d %d\n", length, minimum, count);
645 for (int i = 0; i < count; ++i)
646 printf("%s %d %s %s\n", bursts[i].ip, bursts[i].window.count, bursts[i].first->timestamp, bursts[i].last->timestamp);
647 break;
648 }
649 case PEAK: {
650 int length = atoi(args[0]);
651 Window window = peak(rows, n, length);
652 printf("PEAK %d %d %s %s\n", length, window.count, rows[window.first]->timestamp, rows[window.last]->timestamp);
653 break;
654 }
655 case GAPS: {
656 int length = atoi(args[0]), count = 0;
657 for (int i = 1; i < n; ++i) count += rows[i]->seconds - rows[i - 1]->seconds >= length;
658 printf("GAPS %d %d\n", length, count);
659 for (int i = 1; i < n; ++i)
660 if (rows[i]->seconds - rows[i - 1]->seconds >= length)
661 printf("%s %s %" PRId64 "\n", rows[i - 1]->timestamp, rows[i]->timestamp, rows[i]->seconds - rows[i - 1]->seconds);
662 break;
663 }
664 case IP: {
665 Record *found[MAX_RECORDS];
666 int count = 0, failures = 0, classes[5] = {0};
667 int64_t bytes = 0;
668 for (int i = 0; i < n; ++i) if (strcmp(rows[i]->ip, args[0]) == 0) {
669 found[count++] = rows[i];
670 bytes += rows[i]->bytes;
671 failures += rows[i]->status >= 400;
672 ++classes[rows[i]->status / 100 - 1];
673 }
674 if (!count) { puts("IP no records"); break; }
675 printf("IP %s requests %d bytes %" PRId64 " errors %d paths %d\n", args[0], count, bytes, failures, unique_count(found, count, BY_PATH));
676 printf("first %s last %s\n", found[0]->timestamp, found[count - 1]->timestamp);
677 int size = group_rows(found, count, BY_METHOD, 0, groups, ordered);
678 qsort(groups, size, sizeof(*groups), count_order);
679 printf("methods");
680 for (int i = 0; i < size; ++i) printf(" %s %d", groups[i].sample->method, groups[i].count);
681 printf("\nclasses");
682 for (int i = 0; i < 5; ++i) printf(" %dxx %d", i + 1, classes[i]);
683 putchar('\n');
684 size = group_rows(found, count, BY_PATH, 0, groups, ordered);
685 qsort(groups, size, sizeof(*groups), count_order);
686 for (int i = 0; i < size && i < 3; ++i) printf("path %s %d\n", groups[i].sample->path, groups[i].count);
687 break;
688 }
689 case RANGE: {
690 int64_t start = seconds_of(args[0], args[1]), end = seconds_of(args[2], args[3]);
691 if (start > end) { puts("ERROR invalid range"); break; }
692 analyzer->has_range = true;
693 analyzer->first_second = start;
694 analyzer->last_second = end;
695 sprintf(analyzer->first_timestamp, "%s %s", args[0], args[1]);
696 sprintf(analyzer->last_timestamp, "%s %s", args[2], args[3]);
697 printf("RANGE %s %s active %d\n", analyzer->first_timestamp, analyzer->last_timestamp, selected(analyzer, ordered));
698 break;
699 }
700 case FILTER: {
701 int key = name_index(args[0], filter_keys, 4);
702 if (key == 3) normalize_path(args[1], analyzer->filters[key]);
703 else strcpy(analyzer->filters[key], args[1]);
704 printf("FILTER %s %s active %d\n", args[0], analyzer->filters[key], selected(analyzer, ordered));
705 break;
706 }
707 case UNFILTER: {
708 int key = name_index(args[0], filter_keys, 4);
709 if (key < 0) {
710 if (!analyzer->has_range) { puts("ERROR not filtered"); break; }
711 analyzer->has_range = false;
712 } else {
713 if (!*analyzer->filters[key]) { puts("ERROR not filtered"); break; }
714 analyzer->filters[key][0] = '\0';
715 }
716 printf("UNFILTER %s active %d\n", args[0], selected(analyzer, ordered));
717 break;
718 }
719 case RESET:
720 analyzer->has_range = false;
721 for (int i = 0; i < 4; ++i) analyzer->filters[i][0] = '\0';
722 printf("RESET active %d\n", selected(analyzer, ordered));
723 break;
724 case STATE:
725 printf("STATE active %d range ", n);
726 if (analyzer->has_range) printf("%s %s", analyzer->first_timestamp, analyzer->last_timestamp);
727 else putchar('-');
728 for (int i = 0; i < 4; ++i) printf(" %s %s", lower_keys[i], *analyzer->filters[i] ? analyzer->filters[i] : "-");
729 putchar('\n');
730 break;
731 }
732}

734static void execute(Analyzer *analyzer, char *line) {
735 char *parts[MAX_LINE];
736 int count = words(line, parts), query = -1;
737 if (count) for (int i = 0; i <= STATE; ++i)
738 if (strcmp(parts[0], forms[i].name) == 0) { query = i; break; }
739 if (query < 0) { puts("ERROR unknown query"); return; }
740 const QueryForm *form = &forms[query];
741 int argc = count - 1;
742 char **args = parts + 1;
743 if (argc < form->minimum || argc > form->maximum) { puts("ERROR wrong arguments"); return; }
744 for (int i = 0; i < argc; ++i) {
745 Argument kind = form->arguments[i];
746 if (!valid_argument(kind, args[i], args)) { printf("ERROR %s\n", argument_errors[kind]); return; }
747 }
748 Record *rows[MAX_RECORDS];
749 int n = selected(analyzer, rows);
750 if (!n && form->needs_records) { printf("%s no records\n", form->name); return; }
751 run_query(analyzer, (Query)query, argc, args, rows, n);
752}
753
754int main(void) {
755 static Analyzer analyzer;
756 char line[MAX_LINE + 3];
757 bool queries = false;
758 while (fgets(line, sizeof(line), stdin)) {
759 line[strcspn(line, "\r\n")] = '\0';
760 if (queries) {
761 if (strcmp(line, "END") == 0) break;
762 execute(&analyzer, line);
763 } else if (strcmp(line, "---") == 0) {
764 queries = true;
765 qsort(analyzer.records, analyzer.record_count, sizeof(Record), record_order);
766 } else {
767 ++analyzer.line_count;
768 char *parts[MAX_LINE];
769 int count = words(line, parts), reason = malformed_reason(parts, count);
770 if (reason >= 0) {
771 analyzer.malformed[analyzer.malformed_count++] = (Malformed){analyzer.line_count, reason};
772 continue;
773 }
774 Record *r = &analyzer.records[analyzer.record_count++];
775 r->line = analyzer.line_count;
776 strcpy(r->date, parts[0]);
777 sprintf(r->timestamp, "%s %s", parts[0], parts[1]);
778 r->hour = atoi(parts[1]);
779 r->seconds = seconds_of(parts[0], parts[1]);
780 strcpy(r->ip, parts[2]);
781 parse_ip(parts[2], &r->ip_number);
782 strcpy(r->method, parts[3]);
783 normalize_path(parts[4], r->path);
784 r->status = atoi(parts[5]);
785 r->bytes = strtoll(parts[6], NULL, 10);
786 r->latency = atoi(parts[7]);
787 }
788 }
789 return ferror(stdin) ? EXIT_FAILURE : EXIT_SUCCESS;
