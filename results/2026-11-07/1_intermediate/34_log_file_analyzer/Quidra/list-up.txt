string padded(int value, int width)
    text := string(value)
    return "0" * nat(math.max(0, width - int(#text))) + text
string decimal(int numerator, int denominator, int places)
    scale := 10 ^ places
    value := numerator * scale / denominator
    return "{value / scale}.{padded(value % scale, places)}"

string percentage(int amount, int total)
    return decimal(amount * 100, total, 2) + "%"
void run_query(LogData &data, string command, const string[] args, const int[] ids)
    n := int(#ids)
    elif command == "STATUS"
        emit("STATUS {n}")
        int[] codes := data.statuses[ids]
        for: 1 <= category <= 5
            subset := codes[codes >= category * 100]
            count := int(#subset[subset < (category + 1) * 100])
            emit("{category}xx {count} {percentage(count, n)}")
        previous := -1
        for code := codes.sorted()[]
            if code == previous
                continue
            previous = code
            subset := codes[codes >= code]
            count := int(#subset[subset <= code])
            emit("{code} {count} {percentage(count, n)}")
    members := data.selected()
    if #members == 0 and one_of(command, "STATUS METHODS TOP HOURLY TIMELINE BANDWIDTH VISITORS HISTOGRAM APDEX ERRORS SLOWEST BURST PEAK GAPS")
        emit("{command} no records")
        return
    run_query(&data, command, args, members)
