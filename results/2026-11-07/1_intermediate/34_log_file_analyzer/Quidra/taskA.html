172int seconds_of(string date, string clock)
173 year := int(date[0:<4])
174 month := int(date[5:<7])
175 int days := int(date[8:<10]) - 1
176 for: 2000 <= prior < year
177 days += if leap(prior) then 366 else 365
178 for: 1 <= prior < month
179 days += month_days(year, prior)
180 return days * 86400 + int(clock[0:<2]) * 3600 + int(clock[3:<5]) * 60 + int(clock[6:<8])

186string interval_label(int seconds)
187 days := seconds / 86400
188 minutes := seconds % 86400 / 60
189 year := 2000
190 while days >= (if leap(year) then 366 else 365)
191 days -= if leap(year) then 366 else 365
192 year += 1
193 month := 1
194 while days >= month_days(year, month)
195 days -= month_days(year, month)
196 month += 1
197 return "{year}-{padded(month, 2)}-{padded(days + 1, 2)} {padded(minutes / 60, 2)}:{padded(minutes % 60, 2)}"

199string normalize_path(string raw)
200 string[] parts := []
201 path := raw.split("?")[0]
202 for part := path.split("/")[]
203 if part == ".."
204 if #parts > 0
205 parts = parts[0:<int(#parts) - 1]
206 elif part != "" and part != "."
207 parts = parts.append(part)
208 result := "/"
209 for: position < #parts
210 part := parts[position]
211 if digits(part)
212 part = "{{id}}"
213 elif #part >= 8
214 hexadecimal := true
215 has_digit := false
216 for: offset < #part
217 hexadecimal = hexadecimal and "0123456789abcdef".contains(part[offset])
218 has_digit = has_digit or "0123456789".contains(part[offset])
219 if hexadecimal and has_digit
220 part = "{{hex}}"
221 if position > 0
222 result += "/"
223 result += part
224 return result

401bool group_before(const Group left, const Group right, string mode)
402 comparison := right.count - left.count
403 if mode == "APDEX"
404 comparison = left.points() * right.count - right.points() * left.count
405 if comparison == 0
406 comparison = right.count - left.count
407 elif mode == "ERRORS"
408 left_errors := left.client + left.server
409 right_errors := right.client + right.server
410 comparison = right_errors * left.count - left_errors * right.count
411 if comparison == 0
412 comparison = right_errors - left_errors
413 elif mode == "SLOWEST"
414 comparison = right.total_latency * left.count - left.total_latency * right.count
415 if comparison == 0
416 comparison = right.maximum - left.maximum
417 elif mode == "BURST"
418 comparison = right.peak_count - left.peak_count
419 return if comparison != 0 then comparison < 0 else left.tie < right.tie

421Group[] ranked(const Group[] values, string mode)
422 length := int(#values)
423 if length < 2
424 return values
425 middle := length / 2
426 left := ranked(values[0:<middle], mode)
427 right := ranked(values[middle:], mode)
428 Group[] result := []
429 i := 0
430 j := 0
431 while i < int(#left) and j < int(#right)
432 if group_before(right[j], left[i], mode)
433 result = result.append(right[j])
434 j += 1
435 else
436 result = result.append(left[i])
437 i += 1
438 if i < int(#left)
439 result = result.concat(left[i:])
440 if j < int(#right)
441 result = result.concat(right[j:])
442 return result

445 begin := 0
446 best := 0

677 elif command == "RANGE"

692 emit("ERROR not filtered")

697 string &destination

700 elif key == "STATUS"

704 else
705 &destination = &data.path_filter

708 emit("ERROR not filtered")
709 return
710 destination = ""
711 else
712 destination = if key == "PATH" then normalize_path(args[1]) else args[1]
713 emit("FILTER {key} {destination} active {#data.selected()}")

715 emit("UNFILTER {key} active {#data.selected()}")
716 elif command == "RESET"
717 data.method_filter = ""
718 data.status_filter = ""
719 data.ip_filter = ""
720 data.path_filter = ""
721 data.first_second = -1
722 data.last_second = -1
723 emit("RESET active {#data.selected()}")
724 elif command == "STATE"
725 period := if data.first_second < 0 then "-" else data.first_label + " " + data.last_label
726 method := if data.method_filter == "" then "-" else data.method_filter
727 status := if data.status_filter == "" then "-" else data.status_filter
728 ip := if data.ip_filter == "" then "-" else data.ip_filter
729 path := if data.path_filter == "" then "-" else data.path_filter

736 emit("ERROR unknown query")
