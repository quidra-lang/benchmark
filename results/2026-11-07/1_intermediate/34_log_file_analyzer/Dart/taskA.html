45List<String> tokens(String text) {
46 final trimmed = text.trim();
47 return trimmed.isEmpty ? <String>[] : trimmed.split(RegExp(r'\s+'));
48}
49
50bool digits(String s) => RegExp(r'^[0-9]+$').hasMatch(s);
51
52bool canonical(String s, int maxLength) =>
53 s.isNotEmpty && s.length <= maxLength && digits(s) &&
54 (s.length == 1 || s[0] != '0');
55
56bool leap(int year) =>
57 year % 4 == 0 && (year % 100 != 0 || year % 400 == 0);
58
59int yearDays(int year) => leap(year) ? 366 : 365;
60
61int monthDays(int year, int month) =>
62 month == 2 && leap(year) ? 29 : daysByMonth[month - 1];
63
64bool validDate(String s) {
65 if (!RegExp(r'^[0-9]{4}-[0-9]{2}-[0-9]{2}$').hasMatch(s)) return false;
66 final y = int.parse(s.substring(0, 4));
67 final m = int.parse(s.substring(5, 7));
68 final d = int.parse(s.substring(8, 10));
69 return y >= 2000 && y <= 2059 && m >= 1 && m <= 12 &&
70 d >= 1 && d <= monthDays(y, m);
71}
72
73bool validTime(String s) {
74 if (!RegExp(r'^[0-9]{2}:[0-9]{2}:[0-9]{2}$').hasMatch(s)) return false;
75 return int.parse(s.substring(0, 2)) <= 23 &&
76 int.parse(s.substring(3, 5)) <= 59 &&
77 int.parse(s.substring(6, 8)) <= 59;
78}
79
80bool validIp(String s) {
81 final parts = s.split('.');
82 return parts.length == 4 &&
83 parts.every((p) => canonical(p, 3) && int.parse(p) <= 255);
84}
85
86bool validStatus(String s) =>
87 RegExp(r'^[0-9]{3}$').hasMatch(s) &&
88 int.parse(s) >= 100 && int.parse(s) <= 599;
89
90bool validFilterValue(String key, String s) {
91 if (key == 'METHOD') return methods.contains(s);
92 if (key == 'STATUS') return classes.contains(s) || validStatus(s);
93 if (key == 'IP') return validIp(s);
94 return s.startsWith('/');
95}
96
97bool validArgument(String kind, String value, List<String> args) {
98 final range = ranges[kind];
99 if (range != null) {
100 return canonical(value, 6) &&
101 int.parse(value) >= range[0] && int.parse(value) <= range[1];
102 }
103 switch (kind) {
104 case 'kind': return value == 'PATHS' || value == 'IPS';
105 case 'path': return value.startsWith('/');
106 case 'ip': return validIp(value);
107 case 'date': return validDate(value);
108 case 'time': return validTime(value);
109 case 'key': return keys.contains(value);
110 case 'subject': return value == 'RANGE' || keys.contains(value);
111 default: return validFilterValue(args[0], value);
112 }
113}
114
115String? malformedReason(List<String> fields) {
116 if (fields.length != 8) return 'fields';
117 if (!validDate(fields[0])) return 'date';
118 if (!validTime(fields[1])) return 'time';
119 if (!validIp(fields[2])) return 'ip';
120 if (!methods.contains(fields[3])) return 'method';
121 if (!fields[4].startsWith('/')) return 'path';
122 if (!validStatus(fields[5])) return 'status';
123 if (!canonical(fields[6], 8)) return 'bytes';
124 if (!canonical(fields[7], 6)) return 'latency';
125 return null;
126}

128int secondsOf(String date, String time) {
129 final year = int.parse(date.substring(0, 4));
130 final month = int.parse(date.substring(5, 7));
131 final day = int.parse(date.substring(8, 10));
132 var days = day - 1;
133 for (var y = 2000; y < year; y++) days += yearDays(y);
134 for (var m = 1; m < month; m++) days += monthDays(year, m);
135 return days * 86400 + int.parse(time.substring(0, 2)) * 3600 +
136 int.parse(time.substring(3, 5)) * 60 +
137 int.parse(time.substring(6, 8));
138}
139
140String pad2(int n) => n.toString().padLeft(2, '0');
141
142String intervalLabel(int seconds) {
143 var days = seconds ~/ 86400;
144 final minutes = (seconds % 86400) ~/ 60;
145 var year = 2000, month = 1;
146 while (days >= yearDays(year)) {
147 days -= yearDays(year);
148 year++;
149 }
150 while (days >= monthDays(year, month)) {
151 days -= monthDays(year, month);
152 month++;
153 }
154 return '$year-' + pad2(month) + '-' + pad2(days + 1) +
155 ' ' + pad2(minutes ~/ 60) + ':' + pad2(minutes % 60);
156}

158String normalizePath(String raw) {
159 final parts = <String>[];
160 for (final part in raw.split('?').first.split('/')) {
161 if (part == '..') {
162 if (parts.isNotEmpty) parts.removeLast();
163 } else if (part.isNotEmpty && part != '.') {
164 parts.add(part);
165 }
166 }
167 final normalized = parts.map((s) {
168 if (digits(s)) return '{id}';
169 if (s.length >= 8 && RegExp(r'^[0-9a-f]+$').hasMatch(s) &&
170 RegExp(r'[0-9]').hasMatch(s)) return '{hex}';
171 return s;
172 });
173 return '/' + normalized.join('/');
174}

176int compareIp(String a, String b) {
177 final x = a.split('.').map(int.parse).toList();
178 final y = b.split('.').map(int.parse).toList();
179 for (var i = 0; i < 4; i++) {
180 if (x[i] != y[i]) return x[i].compareTo(y[i]);
181 }
182 return 0;
183}
184
185bool statusMatches(String match, int status) =>
186 classes.contains(match) ? status ~/ 100 == int.parse(match[0]) :
187 status == int.parse(match);
188
189String decimal(int numerator, int denominator, int places) {
190 final scale = places == 1 ? 10 : places == 2 ? 100 : 1000;
191 final n = numerator * scale ~/ denominator;
192 return (n ~/ scale).toString() + '.' +
193 (n % scale).toString().padLeft(places, '0');
194}
195
196String percent(int count, int total) =>
197 decimal(count * 100, total, 2) + '%';
198
199String bar(int count, int maximum) =>
200 '#' * ((count * 30 + maximum - 1) ~/ maximum);
201
202String bytesText(int count) {
203 if (count >= 1073741824) return decimal(count, 1073741824, 1) + 'GB';
204 if (count >= 1048576) return decimal(count, 1048576, 1) + 'MB';
205 if (count >= 1024) return decimal(count, 1024, 1) + 'KB';
206 return count.toString() + 'B';
207}

209int total<T>(Iterable<T> items, int Function(T) value) =>
210 items.fold<int>(0, (sum, item) => sum + value(item));
211
212Map<String, int> counts(Iterable<String> values) {
213 final result = <String, int>{};
214 for (final s in values) result[s] = (result[s] ?? 0) + 1;
215 return result;
216}
217
218List<String> byCount(Map<String, int> numbers, [int Function(String, String)? compare]) {
219 compare ??= (a, b) => a.compareTo(b);
220 final sorted = numbers.keys.toList();
221 sorted.sort((a, b) {
222 final n = numbers[b]!.compareTo(numbers[a]!);
223 return n != 0 ? n : compare!(a, b);
224 });
225 return sorted;
226}

228class LogRecord {
229 final int line, hour, seconds, status, bytes, latency;
230 final String date, timestamp, ip, method, path;
231 LogRecord(this.line, List<String> fields)
232 : date = fields[0],
233 timestamp = fields[0] + ' ' + fields[1],
234 hour = int.parse(fields[1].substring(0, 2)),
235 seconds = secondsOf(fields[0], fields[1]),
236 ip = fields[2],
237 method = fields[3],
238 path = normalizePath(fields[4]),
239 status = int.parse(fields[5]),
240 bytes = int.parse(fields[6]),
241 latency = int.parse(fields[7]);
242}

244Map<String, List<LogRecord>> groups(
245 Iterable<LogRecord> records, String Function(LogRecord) key) {
246 final result = <String, List<LogRecord>>{};
247 for (final row in records) {
248 result.putIfAbsent(key(row), () => <LogRecord>[]).add(row);
249 }
250 return result;
251}

253List<int> classCounts(Iterable<LogRecord> records) {
254 final numbers = List<int>.filled(5, 0);
255 for (final row in records) numbers[row.status ~/ 100 - 1]++;
256 return numbers;

259class Buckets {
260 final int first;
261 final List<int> counts;
262
263 Buckets._(this.first, this.counts);
264
265 factory Buckets(List<int> values) {
266 final first = values.reduce(math.min);
267 final width = values.reduce(math.max) - first + 1;
268 ensure(width <= 60, 'too many buckets');
269 final counts = List<int>.filled(width, 0);
270 for (final n in values) counts[n - first]++;
271 return Buckets._(first, counts);
272 }
273}

278class WindowPeak {
279 final int count;
280 final LogRecord start, end;
281 WindowPeak(this.count, this.start, this.end);
282}

284WindowPeak peak(List<LogRecord> records, int length) {
285 var first = 0, count = 0, begin = 0, end = 0;
286 for (var i = 0; i < records.length; i++) {
287 while (records[i].seconds - records[first].seconds >= length) first++;
288 if (i - first + 1 > count) {
289 count = i - first + 1;
290 begin = first;
291 end = i;
292 }
293 }
294 return WindowPeak(count, records[begin], records[end]);
295}

297class Apdex {
298 final int count, satisfied, tolerating, frustrated, points, all;
299 final String notation, rating;
300 Apdex(List<LogRecord> records, int threshold)
301 : count = records.length,
302 satisfied = records.where((r) =>
303 r.status < 500 && r.latency <= threshold).length,
304 tolerating = records.where((r) =>
305 r.status < 500 && r.latency > threshold &&
306 r.latency <= 4 * threshold).length,
307 frustrated = records.where((r) =>
308 r.status >= 500 || r.latency > 4 * threshold).length,
309 points = _points(records, threshold),
310 all = 2 * records.length,
311 notation = decimal(_points(records, threshold), 2 * records.length, 3),
312 rating = _rating(records, threshold);
313
314 static int _points(List<LogRecord> records, int threshold) =>
315 2 * records.where((r) =>
316 r.status < 500 && r.latency <= threshold).length +
317 records.where((r) => r.status < 500 &&
318 r.latency > threshold && r.latency <= 4 * threshold).length;
319
320 static String _rating(List<LogRecord> records, int threshold) {
321 final numerator = 100 * _points(records, threshold);
322 final denominator = 2 * records.length;
323 if (numerator >= 94 * denominator) return 'excellent';
324 if (numerator >= 85 * denominator) return 'good';
325 if (numerator >= 70 * denominator) return 'fair';
326 if (numerator >= 50 * denominator) return 'poor';
327 return 'unacceptable';
328 }
329}

331class Analyzer {
332 final records = <LogRecord>[];
333 final malformed = <(int, String)>[];
334 final output = <String>[];
335 final conditions = <String, String?>{
336 'METHOD': null, 'STATUS': null, 'IP': null, 'PATH': null
337 };
338 final int lines;
339 int? rangeStart, rangeEnd;
340 String? fromLabel, toLabel;
341
342 Analyzer(List<String> logLines) : lines = logLines.length {
343 for (var i = 0; i < logLines.length; i++) {
344 final fields = tokens(logLines[i]);
345 final reason = malformedReason(fields);
346 if (reason == null) {
347 records.add(LogRecord(i + 1, fields));
348 } else {
349 malformed.add((i + 1, reason));
350 }
351 }
352 records.sort((a, b) {
353 final byTime = a.seconds.compareTo(b.seconds);
354 return byTime == 0 ? a.line.compareTo(b.line) : byTime;
355 });
356 }
357
358 void emit(List<Object> values) {
359 output.add(values.join(' '));
360 }
361
362 List<LogRecord> selected() => records.where((r) {
363 if (rangeStart != null &&
364 (r.seconds < rangeStart! || r.seconds > rangeEnd!)) return false;
365 if (conditions['METHOD'] != null && conditions['METHOD'] != r.method) return false;
366 if (conditions['STATUS'] != null &&
367 !statusMatches(conditions['STATUS']!, r.status)) return false;
368 if (conditions['IP'] != null && conditions['IP'] != r.ip) return false;
369 return conditions['PATH'] == null || conditions['PATH'] == r.path;
370 }).toList();
371
372 void barLine(String label, int count, int maximum) {
373 emit([label, count, if (count > 0) bar(count, maximum)]);
374 }
375
376 void execute(String line) {
377 final parts = tokens(line);
378 final name = parts.isEmpty ? '' : parts.first;
379 final args = parts.skip(1).toList();
380 try {
381 ensure(forms.containsKey(name), 'unknown query');
382 final form = forms[name]!;
383 final optional = (name == 'LATENCY' || name == 'SESSIONS') &&
384 args.length == form.length - 1;
385 ensure(args.length == form.length || optional, 'wrong arguments');
386 for (var i = 0; i < args.length; i++) {
387 ensure(validArgument(form[i], args[i], args), errors[form[i]]!);
388 }
389 final rows = selected();
390 if (needRecords.contains(name) && rows.isEmpty) {
391 emit([name, 'no records']);
392 } else {
393 run(name, args, rows);
394 }
395 } on QueryError catch (error) {
396 emit(['ERROR', error.message]);
397 }
398 }
399
400 void run(String name, List<String> a, List<LogRecord> rows) {
401 switch (name) {
402 case 'SUMMARY': {
403 emit(['SUMMARY', 'lines', lines, 'valid', records.length,
404 'malformed', malformed.length]);
405 emit(['requests', rows.length, 'bytes', total(rows, (r) => r.bytes),
406 'ips', rows.map((r) => r.ip).toSet().length,
407 'paths', rows.map((r) => r.path).toSet().length]);
408 if (rows.isEmpty) {
409 emit(['span', 'none']);
410 } else {
411 emit(['span', rows.first.timestamp, rows.last.timestamp,
412 rows.last.seconds - rows.first.seconds]);
413 }
414 break;
415
416 }
417 case 'MALFORMED': {
418 emit(['MALFORMED', malformed.length]);
419 for (final reason in reasons) {
420 final found = malformed.where((m) => m.$2 == reason)
421 .map((m) => m.$1).toList();
422 if (found.isNotEmpty) emit([reason, found.length, ...found]);
423 }
424 break;
425
426 }
427 case 'STATUS': {
428 emit(['STATUS', rows.length]);
429 final cs = classCounts(rows);
430 for (var i = 0; i < 5; i++) emit([classes[i], cs[i], percent(cs[i], rows.length)]);
431 final statuses = <int, int>{};
432 for (final row in rows) {
433 statuses[row.status] = (statuses[row.status] ?? 0) + 1;
434 }
435 final sorted = statuses.keys.toList()..sort();
436 for (final status in sorted) {
437 emit([status, statuses[status]!, percent(statuses[status]!, rows.length)]);
438 }
439 break;
440
441 }
442 case 'METHODS': {
443 final values = counts(rows.map((r) => r.method));
444 emit(['METHODS', rows.length]);
445 for (final method in byCount(values)) {
446 emit([method, values[method]!, percent(values[method]!, rows.length)]);
447 }
448 break;
449
450 }
451 case 'TOP': {
452 final kind = a[0];
453 final values = counts(rows.map((r) => kind == 'PATHS' ? r.path : r.ip));
454 final sorted = byCount(values, kind == 'IPS' ? compareIp : null);
455 final limit = math.min(int.parse(a[1]), sorted.length);
456 emit(['TOP', kind, limit, 'of', sorted.length]);
457 var rank = 0;
458 for (var i = 0; i < limit; i++) {
459 if (i == 0 || values[sorted[i]] != values[sorted[i - 1]]) rank = i + 1;
460 emit([rank, sorted[i], values[sorted[i]]!,
461 percent(values[sorted[i]]!, rows.length)]);
462 }
463 break;
464
465 }
466 case 'HOURLY': {
467 final values = List<int>.filled(24, 0);
468 for (final row in rows) values[row.hour]++;
469 final highest = values.reduce(math.max);
470 emit(['HOURLY', rows.length, 'peak', pad2(values.indexOf(highest)), highest]);
471 for (var i = 0; i < 24; i++) barLine(pad2(i), values[i], highest);
472 break;
473
474 }
475 case 'TIMELINE': {
476 final minutes = int.parse(a[0]), step = int.parse(a[0]) * 60;
477 final buckets = Buckets(rows.map((r) => r.seconds ~/ step).toList());
478 emit(['TIMELINE', minutes, buckets.counts.length]);
479 final highest = buckets.counts.reduce(math.max);
480 for (var i = 0; i < buckets.counts.length; i++) {
481 barLine(intervalLabel((buckets.first + i) * step), buckets.counts[i], highest);
482 }
483 break;
484
485 }
486 case 'BANDWIDTH': {
487 final byDay = groups(rows, (r) => r.date);
488 final size = total(rows, (r) => r.bytes);
489 emit(['BANDWIDTH', byDay.length, size, bytesText(size)]);
490 final dates = byDay.keys.toList()..sort();
491 for (final date in dates) {
492 final items = byDay[date]!;
493 final bytes = total(items, (r) => r.bytes);
494 emit([date, items.length, bytes, bytesText(bytes)]);
495 }
496 break;
497
498 }
499 case 'VISITORS': {
500 final firstDate = <String, String>{};
501 for (final row in rows) {
502 if (!firstDate.containsKey(row.ip) ||
503 row.date.compareTo(firstDate[row.ip]!) < 0) firstDate[row.ip] = row.date;
504 }
505 final byDay = groups(rows, (r) => r.date);
506 emit(['VISITORS', byDay.length, firstDate.length]);
507 final dates = byDay.keys.toList()..sort();
508 for (final date in dates) {
509 final ips = byDay[date]!.map((r) => r.ip).toSet();
510 final fresh = ips.where((ip) => firstDate[ip] == date).length;
511 emit([date, 'unique', ips.length, 'new', fresh, 'returning', ips.length - fresh]);
512 }
513 break;
514
515 }
516 case 'LATENCY': {
517 final label = a.isEmpty ? 'all' : normalizePath(a[0]);
518 final selectedRows = a.isEmpty ? rows :
519 rows.where((r) => r.path == label).toList();
520 if (selectedRows.isEmpty) {
521 emit(['LATENCY', 'no records']);
522 break;
523 }
524 final values = selectedRows.map((r) => r.latency).toList()..sort();
525 final n = values.length, trim = values.length ~/ 10;
526 final trimmed = values.sublist(trim, n - trim);
527 emit(['LATENCY', label, 'count', n, 'min', values.first,
528 'p50', percentile(values, 50), 'p90', percentile(values, 90),
529 'p99', percentile(values, 99), 'max', values.last,
530 'mean', decimal(total(values, (x) => x), n, 2),
531 'trimmed', decimal(total(trimmed, (x) => x), trimmed.length, 2)]);
532 break;
533
534 }
535 case 'HISTOGRAM': {
536 final width = int.parse(a[0]);
537 final buckets = Buckets(rows.map((r) => r.latency ~/ width).toList());
538 emit(['HISTOGRAM', width, buckets.counts.length]);
539 final highest = buckets.counts.reduce(math.max);
540 for (var i = 0; i < buckets.counts.length; i++) {
541 final lo = (buckets.first + i) * width;
542 barLine(lo.toString() + '-' + (lo + width - 1).toString(),
543 buckets.counts[i], highest);
544 }
545 break;
546
547 }
548 case 'APDEX': {
549 final threshold = int.parse(a[0]);
550 final all = Apdex(rows, threshold);
551 emit(['APDEX', threshold, all.notation, all.rating, 'satisfied',
552 all.satisfied, 'tolerating', all.tolerating, 'frustrated', all.frustrated]);
553 final byPath = groups(rows, (r) => r.path).entries.map((e) =>
554 (path: e.key, score: Apdex(e.value, threshold))).toList();
555 byPath.sort((x, y) {
556 final n = (x.score.points * y.score.all)
557 .compareTo(y.score.points * x.score.all);
558 if (n != 0) return n;
559 final size = y.score.count.compareTo(x.score.count);
560 return size != 0 ? size : x.path.compareTo(y.path);
561 });
562 for (final item in byPath) {
563 emit([item.path, item.score.notation, item.score.rating, item.score.count]);
564 }
565 break;
566
567 }
568 case 'ERRORS': {
569 final failed = rows.where((r) => r.status >= 400).length;
570 emit(['ERRORS', failed, 'of', rows.length, percent(failed, rows.length)]);
571 final byPath = groups(rows, (r) => r.path).entries.map((e) =>
572 (path: e.key, count: e.value.length,
573 client: e.value.where((r) => r.status ~/ 100 == 4).length,
574 server: e.value.where((r) => r.status ~/ 100 == 5).length))
575 .where((e) => e.client + e.server > 0).toList();
576 byPath.sort((x, y) {
577 final xf = x.client + x.server, yf = y.client + y.server;
578 final n = yf * x.count - xf * y.count;
579 if (n != 0) return n;
580 final ties = yf.compareTo(xf);
581 return ties != 0 ? ties : x.path.compareTo(y.path);
582 });
583 for (final p in byPath) {
584 emit([p.path, p.count, p.client, p.server,
585 percent(p.client + p.server, p.count)]);
586 }
587 break;
588
589 }
590 case 'SLOWEST': {
591 final entries = groups(rows, (r) => r.method + '\u0001' + r.path);
592 final endpoints = entries.entries.map((e) {
593 final pair = e.key.split('\u0001');
594 return (method: pair[0], path: pair[1], count: e.value.length,
595 sum: total(e.value, (r) => r.latency),
596 maxLatency: e.value.map((r) => r.latency).reduce(math.max));
597 }).toList();
598 endpoints.sort((x, y) {
599 final n = y.sum * x.count - x.sum * y.count;
600 if (n != 0) return n;
601 final m = y.maxLatency.compareTo(x.maxLatency);
602 if (m != 0) return m;
603 final path = x.path.compareTo(y.path);
604 return path != 0 ? path : x.method.compareTo(y.method);
605 });
606 final limit = math.min(int.parse(a[0]), endpoints.length);
607 emit(['SLOWEST', limit, 'of', endpoints.length]);
608 for (var i = 0; i < limit; i++) {
609 final p = endpoints[i];
610 emit([i + 1, p.method, p.path, 'count', p.count,
611 'avg', decimal(p.sum, p.count, 2), 'max', p.maxLatency]);
612 }
613 break;
614
615 }
616 case 'SESSIONS': {
617 final ip = a[0], timeout = a.length == 1 ? 30 : int.parse(a[1]);
618 final found = rows.where((r) => r.ip == ip).toList();
619 if (found.isEmpty) {
620 emit(['SESSIONS', 'no records']);
621 break;
622 }
623 final sessions = <List<LogRecord>>[[found.first]];
624 for (var i = 1; i < found.length; i++) {
625 if (found[i].seconds - found[i - 1].seconds <= timeout * 60) {
626 sessions.last.add(found[i]);
627 } else {
628 sessions.add([found[i]]);
629 }
630 }
631 final duration = total(sessions, (s) => s.last.seconds - s.first.seconds);
632 emit(['SESSIONS', ip, 'timeout', timeout, 'count', sessions.length,
633 'requests', found.length, 'avg_duration', decimal(duration, sessions.length, 1),
634 'avg_requests', decimal(found.length, sessions.length, 1)]);
635 for (var i = 0; i < sessions.length; i++) {
636 final s = sessions[i];
637 emit([i + 1, s.first.timestamp, s.last.timestamp,
638 s.last.seconds - s.first.seconds, s.length,
639 s.map((r) => r.path).toSet().length]);
640 }
641 break;
642
643 }
644 case 'BURST': {
645 final length = int.parse(a[0]), count = int.parse(a[1]);
646 final ips = groups(rows.where((r) => r.status >= 400), (r) => r.ip);
647 final bursts = ips.entries.map((e) =>
648 (ip: e.key, window: peak(e.value, length)))
649 .where((e) => e.window.count >= count).toList();
650 bursts.sort((x, y) {
651 final n = y.window.count.compareTo(x.window.count);
652 return n != 0 ? n : compareIp(x.ip, y.ip);
653 });
654 emit(['BURST', length, count, bursts.length]);
655 for (final entry in bursts) {
656 emit([entry.ip, entry.window.count,
657 entry.window.start.timestamp, entry.window.end.timestamp]);
658 }
659 break;
660
661 }
662 case 'PEAK': {
663 final length = int.parse(a[0]), result = peak(rows, int.parse(a[0]));
664 emit(['PEAK', length, result.count, result.start.timestamp, result.end.timestamp]);
665 break;
666
667 }
668 case 'GAPS': {
669 final length = int.parse(a[0]);
670 final gaps = <(LogRecord, LogRecord)>[];
671 for (var i = 1; i < rows.length; i++) {
672 if (rows[i].seconds - rows[i - 1].seconds >= length) {
673 gaps.add((rows[i - 1], rows[i]));
674 }
675 }
676 emit(['GAPS', length, gaps.length]);
677 for (final g in gaps) {
678 emit([g.$1.timestamp, g.$2.timestamp,
679 g.$2.seconds - g.$1.seconds]);
680 }
681 break;
682
683 }
684 case 'IP': {
685 final ip = a[0], found = rows.where((r) => r.ip == a[0]).toList();
686 if (found.isEmpty) {
687 emit(['IP', 'no records']);
688 break;
689 }
690 final byPath = counts(found.map((r) => r.path));
691 emit(['IP', ip, 'requests', found.length,
692 'bytes', total(found, (r) => r.bytes),
693 'errors', found.where((r) => r.status >= 400).length,
694 'paths', byPath.length]);
695 emit(['first', found.first.timestamp, 'last', found.last.timestamp]);
696 final byMethod = counts(found.map((r) => r.method));
697 emit(['methods', for (final m in byCount(byMethod)) ...[m, byMethod[m]!] ]);
698 final cs = classCounts(found);
699 emit(['classes', for (var i = 0; i < 5; i++) ...[classes[i], cs[i]]]);
700 for (final p in byCount(byPath).take(3)) {
701 emit(['path', p, byPath[p]!]);
702 }
703 break;
704
705 }
706 case 'RANGE': {
707 final start = secondsOf(a[0], a[1]), end = secondsOf(a[2], a[3]);
708 ensure(start <= end, 'invalid range');
709 rangeStart = start;
710 rangeEnd = end;
711 fromLabel = a[0] + ' ' + a[1];
712 toLabel = a[2] + ' ' + a[3];
713 emit(['RANGE', fromLabel!, toLabel!, 'active', selected().length]);
714 break;
715
716 }
717 case 'FILTER': {
718 final key = a[0], raw = a[1];
719 conditions[key] = key == 'PATH' ? normalizePath(raw) : raw;
720 emit(['FILTER', key, conditions[key]!, 'active', selected().length]);
721 break;
722
723 }
724 case 'UNFILTER': {
725 final key = a[0];
726 if (key == 'RANGE') {
727 ensure(rangeStart != null, 'not filtered');
728 rangeStart = rangeEnd = null;
729 fromLabel = toLabel = null;
730 } else {
731 ensure(conditions[key] != null, 'not filtered');
732 conditions[key] = null;
733 }
734 emit(['UNFILTER', key, 'active', selected().length]);
735 break;
736
737 }
738 case 'RESET': {
739 rangeStart = rangeEnd = null;
740 fromLabel = toLabel = null;
741 for (final key in keys) conditions[key] = null;
742 emit(['RESET', 'active', selected().length]);
743 break;
744
745 }
746 case 'STATE': {
747 emit(['STATE', 'active', rows.length, 'range',
748 rangeStart == null ? '-' : fromLabel! + ' ' + toLabel!,
749 for (final key in keys) ...[key.toLowerCase(), conditions[key] ?? '-']]);
750 break;
751 }
752 }
753 }
754}

756void main() {
757 final input = stdin.readAsLinesSync();
758 final separator = input.indexOf('---');
759 final end = input.indexOf('END', separator + 1);
760 final analyzer = Analyzer(input.sublist(0, separator));
761 for (var i = separator + 1; i < end; i++) analyzer.execute(input[i]);
762 if (analyzer.output.isNotEmpty) {
763 stdout.writeln(analyzer.output.join('\n'));
764 }
765}
