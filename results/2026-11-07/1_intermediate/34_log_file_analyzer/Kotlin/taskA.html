23 "STATUS", "METHODS", "TOP", "HOURLY", "TIMELINE", "BANDWIDTH",

39private val integerPattern = Regex("[0-9]+")

52private fun tokens(line: String): List<String> =
53 line.trim().takeIf { it.isNotEmpty() }?.split(Regex("\\s+")) ?: emptyList()
54
55private fun canonical(text: String, max: Int): Boolean =
56 text.isNotEmpty() && text.length <= max && integerPattern.matches(text) &&
57 (text.length == 1 || text[0] != '0')
58
59private fun leap(year: Int): Boolean =
60 year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
61
62private fun daysInYear(year: Int): Int = if (leap(year)) 366 else 365
63private fun daysInMonth(year: Int, month: Int): Int =
64 if (month == 2 && leap(year)) 29 else monthLengths[month - 1]
65
66private fun validDate(s: String): Boolean {
67 if (!datePattern.matches(s)) return false
68 val year = s.substring(0, 4).toInt()
69 val month = s.substring(5, 7).toInt()
70 val day = s.substring(8, 10).toInt()
71 return year in 2000..2059 && month in 1..12 &&
72 day in 1..daysInMonth(year, month)
73}
74
75private fun validTime(s: String): Boolean =
76 timePattern.matches(s) && s.substring(0, 2).toInt() < 24 &&
77 s.substring(3, 5).toInt() < 60 && s.substring(6, 8).toInt() < 60
78
79private fun validIp(s: String): Boolean {
80 val parts = s.split('.')
81 return parts.size == 4 && parts.all { canonical(it, 3) && it.toInt() <= 255 }
82}
83
84private fun validStatus(s: String): Boolean =
85 statusPattern.matches(s) && s.toInt() in 100..599
86
87private fun validFilterValue(key: String, value: String): Boolean = when (key) {
88 "METHOD" -> value in methods
89 "STATUS" -> value in statusClasses || validStatus(value)
90 "IP" -> validIp(value)
91 else -> value.startsWith("/")
92}
93
94private fun validArgument(kind: String, text: String, args: List<String>): Boolean {
95 ranges[kind]?.let { return canonical(text, 6) && text.toInt() in it }
96 return when (kind) {
97 "kind" -> text == "PATHS" || text == "IPS"
98 "path" -> text.startsWith("/")
99 "ip" -> validIp(text)
100 "date" -> validDate(text)
101 "time" -> validTime(text)
102 "key" -> text in filterKeys
103 "subject" -> text == "RANGE" || text in filterKeys
104 else -> validFilterValue(args[0], text)
105 }
106}

108private fun malformedReason(fields: List<String>): String? {
109 if (fields.size != 8) return "fields"
110 if (!validDate(fields[0])) return "date"
111 if (!validTime(fields[1])) return "time"
112 if (!validIp(fields[2])) return "ip"
113 if (fields[3] !in methods) return "method"
114 if (!fields[4].startsWith("/")) return "path"
115 if (!validStatus(fields[5])) return "status"
116 if (!canonical(fields[6], 8)) return "bytes"
117 if (!canonical(fields[7], 6)) return "latency"
118 return null
119}

121private fun secondsOf(date: String, clock: String): Long {
122 val year = date.substring(0, 4).toInt()
123 val month = date.substring(5, 7).toInt()
124 var day = date.substring(8, 10).toLong() - 1
125 for (y in 2000 until year) day += daysInYear(y)
126 for (m in 1 until month) day += daysInMonth(year, m)
127 return day * 86400 + clock.substring(0, 2).toLong() * 3600 +
128 clock.substring(3, 5).toLong() * 60 + clock.substring(6, 8).toLong()
129}
130
131private fun pad(n: Long) = n.toString().padStart(2, '0')
132
133private fun intervalLabel(seconds: Long): String {
134 var days = seconds / 86400
135 val minutes = seconds % 86400 / 60
136 var year = 2000
137 while (days >= daysInYear(year)) days -= daysInYear(year++)
138 var month = 1
139 while (days >= daysInMonth(year, month)) days -= daysInMonth(year, month++)
140 return year.toString() + "-" + pad(month.toLong()) + "-" + pad(days + 1) +
141 " " + pad(minutes / 60) + ":" + pad(minutes % 60)
142}

144private fun normalizePath(value: String): String {
145 val parts = mutableListOf<String>()
146 for (part in value.substringBefore('?').split('/')) {
147 when (part) {
148 ".." -> if (parts.isNotEmpty()) parts.removeAt(parts.lastIndex)
149 "", "." -> {}
150 else -> parts.add(part)
151 }
152 }
153 return "/" + parts.joinToString("/") {
154 when {
155 integerPattern.matches(it) -> "{id}"
156 it.length >= 8 && hexPattern.matches(it) &&
157 digitPattern.containsMatchIn(it) -> "{hex}"
158 else -> it
159 }
160 }
161}
162
163private fun compareIp(a: String, b: String): Int {
164 val left = a.split('.').map { it.toInt() }
165 val right = b.split('.').map { it.toInt() }
166 for (i in 0..3) if (left[i] != right[i]) return left[i].compareTo(right[i])
167 return 0
168}
169
170private fun statusMatches(filter: String, value: Int): Boolean =
171 if (filter in statusClasses) value / 100 == filter[0].digitToInt()
172 else value == filter.toInt()
173
174private fun decimal(numerator: Long, denominator: Long, places: Int): String {
175 val factor = when (places) { 1 -> 10L; 2 -> 100L; else -> 1000L }
176 val scaled = numerator * factor / denominator
177 return (scaled / factor).toString() + "." +
178 (scaled % factor).toString().padStart(places, '0')
179}
180
181private fun percent(part: Int, whole: Int): String =
182 decimal(part.toLong() * 100, whole.toLong(), 2) + "%"
183
184private fun bar(count: Int, maximum: Int): String =
185 "#".repeat((count * 30 + maximum - 1) / maximum)
186
187private fun bytesText(n: Long): String = when {
188 n >= 1073741824 -> decimal(n, 1073741824, 1) + "GB"
189 n >= 1048576 -> decimal(n, 1048576, 1) + "MB"
190 n >= 1024 -> decimal(n, 1024, 1) + "KB"
191 else -> n.toString() + "B"
192}

194private fun <T> frequency(values: Iterable<T>): Map<T, Int> {
195 val result = mutableMapOf<T, Int>()
196 for (v in values) result[v] = result.getOrDefault(v, 0) + 1
197 return result
198}

200private fun ranked(
201 counts: Map<String, Int>,
202 compare: Comparator<String> = Comparator.naturalOrder()
203): List<String> = counts.keys.sortedWith { a, b ->
204 val n = counts.getValue(b).compareTo(counts.getValue(a))
205 if (n != 0) n else compare.compare(a, b)
206}

220private fun classCounts(records: List<Record>): IntArray {
221 val result = IntArray(5)
222 for (r in records) result[r.status / 100 - 1]++
223 return result

226private data class Buckets(val first: Long, val counts: IntArray)
227
228private fun buckets(values: List<Long>): Buckets {
229 val first = values.min()
230 val span = values.max() - first + 1
231 requireValue(span <= 60, "too many buckets")
232 val counts = IntArray(span.toInt())
233 for (v in values) counts[(v - first).toInt()]++
234 return Buckets(first, counts)
235}

237private fun percentile(values: List<Int>, rank: Int): Int =
238 values[(rank * values.size + 99) / 100 - 1]
239
240private data class Window(val count: Int, val first: Record, val last: Record)
241
242private fun windowPeak(records: List<Record>, length: Int): Window {
243 var begin = 0
244 var count = 0
245 var first = 0
246 var last = 0
247 for (i in records.indices) {
248 while (records[i].seconds - records[begin].seconds >= length) begin++
249 if (i - begin + 1 > count) {
250 count = i - begin + 1
251 first = begin
252 last = i
253 }
254 }
255 return Window(count, records[first], records[last])
256}

258private data class Apdex(
259 val count: Int, val satisfied: Int, val tolerating: Int, val frustrated: Int,
260 val points: Long, val total: Long, val text: String, val rating: String
261)
262
263private fun apdex(records: List<Record>, threshold: Int): Apdex {
264 val satisfied = records.count { it.status < 500 && it.latency <= threshold }
265 val tolerating = records.count {
266 it.status < 500 && it.latency > threshold && it.latency <= threshold * 4
267 }
268 val n = records.size
269 val points = satisfied * 2L + tolerating
270 val total = n * 2L
271 val rating = when {
272 100 * points >= 94 * total -> "excellent"
273 100 * points >= 85 * total -> "good"
274 100 * points >= 70 * total -> "fair"
275 100 * points >= 50 * total -> "poor"
276 else -> "unacceptable"
277 }
278 return Apdex(n, satisfied, tolerating, n - satisfied - tolerating,
279 points, total, decimal(points, total, 3), rating)
280}

282private class LogAnalyzer(lines: List<String>) {
283 private val records = mutableListOf<Record>()
284 private val malformed = mutableListOf<Pair<Int, String>>()
285 private val lineCount = lines.size
286 private val filters = mutableMapOf<String, String?>(
287 "METHOD" to null, "STATUS" to null, "IP" to null, "PATH" to null
288 )
289 private var from: Long? = null
290 private var until: Long? = null
291 private var fromText: String? = null
292 private var untilText: String? = null
293 val output = mutableListOf<String>()
294
295 init {
296 lines.forEachIndexed { index, line ->
297 val values = tokens(line)
298 val reason = malformedReason(values)
299 if (reason != null) malformed.add(index + 1 to reason)
300 else records.add(Record(index + 1, values))
301 }
302 records.sortWith(compareBy<Record> { it.seconds }.thenBy { it.line })
303 }
304
305 private fun emit(vararg parts: Any) {
306 output.add(parts.joinToString(" "))
307 }
308
309 private fun selected(): List<Record> = records.filter { r ->
310 (from == null || r.seconds >= from!! && r.seconds <= until!!) &&
311 (filters["METHOD"] == null || r.method == filters["METHOD"]) &&
312 (filters["STATUS"] == null || statusMatches(filters["STATUS"]!!, r.status)) &&
313 (filters["IP"] == null || r.ip == filters["IP"]) &&
314 (filters["PATH"] == null || r.path == filters["PATH"])
315 }
316
317 private fun barLine(label: String, count: Int, maximum: Int) {
318 if (count > 0) emit(label, count, bar(count, maximum))
319 else emit(label, count)
320 }
321
322 fun execute(line: String) {
323 val fields = tokens(line)
324 val name = fields.firstOrNull() ?: ""
325 val args = fields.drop(1)
326 try {
327 requireValue(name in forms, "unknown query")
328 val expected = forms.getValue(name)
329 val optional = (name == "LATENCY" || name == "SESSIONS") &&
330 args.size == expected.size - 1
331 requireValue(args.size == expected.size || optional, "wrong arguments")
332 args.forEachIndexed { i, value ->
333 requireValue(validArgument(expected[i], value, args),
334 errors.getValue(expected[i]))
335 }
336 val rows = selected()
337 if (name in requireRecords && rows.isEmpty()) {
338 emit(name, "no records")
339 return
340 }
341 run(name, args, rows)
342 } catch (error: QueryError) {
343 emit("ERROR", error.message ?: "")
344 }
345 }
346
347 private fun run(name: String, args: List<String>, rows: List<Record>) {
348 when (name) {
349 "SUMMARY" -> {
350 emit("SUMMARY", "lines", lineCount, "valid", records.size,
351 "malformed", malformed.size)
352 emit("requests", rows.size, "bytes", rows.sumOf { it.size },
353 "ips", rows.map { it.ip }.toSet().size,
354 "paths", rows.map { it.path }.toSet().size)
355 if (rows.isEmpty()) emit("span", "none")
356 else emit("span", rows.first().timestamp, rows.last().timestamp,
357 rows.last().seconds - rows.first().seconds)
358 }
359 "MALFORMED" -> {
360 emit("MALFORMED", malformed.size)
361 for (reason in reasons) {
362 val matching = malformed.filter { it.second == reason }.map { it.first }
363 if (matching.isNotEmpty()) emit(reason, matching.size, *matching.toTypedArray())
364 }
365 }
366 "STATUS" -> {
367 emit("STATUS", rows.size)
368 val category = classCounts(rows)
369 for (i in 0..4) emit(statusClasses[i], category[i], percent(category[i], rows.size))
370 val statuses = frequency(rows.map { it.status })
371 for (s in statuses.keys.sorted())
372 emit(s, statuses.getValue(s), percent(statuses.getValue(s), rows.size))
373 }
374 "METHODS" -> {
375 emit("METHODS", rows.size)
376 val counts = frequency(rows.map { it.method })
377 for (method in ranked(counts))
378 emit(method, counts.getValue(method), percent(counts.getValue(method), rows.size))
379 }
380 "TOP" -> {
381 val kind = args[0]
382 val counts = frequency(rows.map { if (kind == "PATHS") it.path else it.ip })
383 val keys = ranked(counts, if (kind == "IPS") Comparator { a, b ->
384 compareIp(a, b)
385 } else Comparator.naturalOrder())
386 val shown = minOf(args[1].toInt(), keys.size)
387 emit("TOP", kind, shown, "of", keys.size)
388 var rank = 0
389 for (i in 0 until shown) {
390 if (i == 0 || counts[keys[i]] != counts[keys[i - 1]]) rank = i + 1
391 emit(rank, keys[i], counts.getValue(keys[i]),
392 percent(counts.getValue(keys[i]), rows.size))
393 }
394 }
395 "HOURLY" -> {
396 val counts = IntArray(24)
397 for (r in rows) counts[r.hour]++
398 val maximum = counts.max()
399 emit("HOURLY", rows.size, "peak", pad(counts.indexOf(maximum).toLong()), maximum)
400 for (i in counts.indices) barLine(pad(i.toLong()), counts[i], maximum)
401 }
402 "TIMELINE" -> {
403 val minutes = args[0].toInt()
404 val width = minutes * 60L
405 val buckets = buckets(rows.map { it.seconds / width })
406 emit("TIMELINE", minutes, buckets.counts.size)
407 val maximum = buckets.counts.max()
408 buckets.counts.forEachIndexed { i, n ->
409 barLine(intervalLabel((buckets.first + i) * width), n, maximum)
410 }
411 }
412 "BANDWIDTH" -> {
413 val byDate = rows.groupBy { it.date }
414 val size = rows.sumOf { it.size }
415 emit("BANDWIDTH", byDate.size, size, bytesText(size))
416 for (date in byDate.keys.sorted()) {
417 val records = byDate.getValue(date)
418 val total = records.sumOf { it.size }
419 emit(date, records.size, total, bytesText(total))
420 }
421 }
422 "VISITORS" -> {
423 val firstDates = mutableMapOf<String, String>()
424 for (r in rows) {
425 if (firstDates[r.ip] == null || r.date < firstDates.getValue(r.ip))
426 firstDates[r.ip] = r.date
427 }
428 val byDate = rows.groupBy { it.date }
429 emit("VISITORS", byDate.size, firstDates.size)
430 for (date in byDate.keys.sorted()) {
431 val ips = byDate.getValue(date).map { it.ip }.toSet()
432 val fresh = ips.count { firstDates[it] == date }
433 emit(date, "unique", ips.size, "new", fresh,
434 "returning", ips.size - fresh)
435 }
436 }
437 "LATENCY" -> {
438 val label = if (args.isEmpty()) "all" else normalizePath(args[0])
439 val chosen = if (args.isEmpty()) rows else rows.filter { it.path == label }
440 if (chosen.isEmpty()) {
441 emit("LATENCY", "no records")
442 } else {
443 val values = chosen.map { it.latency }.sorted()
444 val count = values.size
445 val trim = count / 10
446 val trimmed = values.subList(trim, count - trim)
447 emit("LATENCY", label, "count", count, "min", values.first(),
448 "p50", percentile(values, 50), "p90", percentile(values, 90),
449 "p99", percentile(values, 99), "max", values.last(),
450 "mean", decimal(values.sumOf { it.toLong() }, count.toLong(), 2),
451 "trimmed", decimal(trimmed.sumOf { it.toLong() }, trimmed.size.toLong(), 2))
452 }
453 }
454 "HISTOGRAM" -> {
455 val width = args[0].toInt()
456 val buckets = buckets(rows.map { (it.latency / width).toLong() })
457 emit("HISTOGRAM", width, buckets.counts.size)
458 val maximum = buckets.counts.max()
459 buckets.counts.forEachIndexed { i, n ->
460 val start = (buckets.first + i) * width
461 barLine(start.toString() + "-" + (start + width - 1), n, maximum)
462 }
463 }
464 "APDEX" -> {
465 val threshold = args[0].toInt()
466 val overall = apdex(rows, threshold)
467 emit("APDEX", threshold, overall.text, overall.rating, "satisfied",
468 overall.satisfied, "tolerating", overall.tolerating,
469 "frustrated", overall.frustrated)
470 val paths = rows.groupBy { it.path }.map { (path, group) ->
471 path to apdex(group, threshold)
472 }.sortedWith { a, b ->
473 val ratio = (a.second.points * b.second.total)
474 .compareTo(b.second.points * a.second.total)
475 if (ratio != 0) ratio
476 else {
477 val count = b.second.count.compareTo(a.second.count)
478 if (count != 0) count else a.first.compareTo(b.first)
479 }
480 }
481 for ((path, score) in paths) emit(path, score.text, score.rating, score.count)
482 }
483 "ERRORS" -> {
484 val failed = rows.count { it.status >= 400 }
485 emit("ERRORS", failed, "of", rows.size, percent(failed, rows.size))
486 data class Info(val path: String, val count: Int, val client: Int, val server: Int)
487 val result = rows.groupBy { it.path }.map { (path, group) ->
488 Info(path, group.size, group.count { it.status / 100 == 4 },
489 group.count { it.status / 100 == 5 })
490 }.filter { it.client + it.server > 0 }.sortedWith { a, b ->
491 val ax = a.client + a.server
492 val bx = b.client + b.server
493 val ratio = (bx.toLong() * a.count).compareTo(ax.toLong() * b.count)
494 if (ratio != 0) ratio
495 else {
496 val count = bx.compareTo(ax)
497 if (count != 0) count else a.path.compareTo(b.path)
498 }
499 }
500 for (info in result) {
501 emit(info.path, info.count, info.client, info.server,
502 percent(info.client + info.server, info.count))
503 }
504 }
505 "SLOWEST" -> {
506 data class Endpoint(
507 val method: String, val path: String, val count: Int,
508 val sum: Long, val maximum: Int
509 )
510 val result = rows.groupBy { it.method to it.path }.map { (key, group) ->
511 Endpoint(key.first, key.second, group.size,
512 group.sumOf { it.latency.toLong() }, group.maxOf { it.latency })
513 }.sortedWith { a, b ->
514 val avg = (b.sum * a.count).compareTo(a.sum * b.count)
515 if (avg != 0) avg
516 else {
517 val max = b.maximum.compareTo(a.maximum)
518 if (max != 0) max
519 else {
520 val path = a.path.compareTo(b.path)
521 if (path != 0) path else a.method.compareTo(b.method)
522 }
523 }
524 }
525 val limit = minOf(args[0].toInt(), result.size)
526 emit("SLOWEST", limit, "of", result.size)
527 for ((i, item) in result.take(limit).withIndex()) {
528 emit(i + 1, item.method, item.path, "count", item.count,
529 "avg", decimal(item.sum, item.count.toLong(), 2), "max", item.maximum)
530 }
531 }
532 "SESSIONS" -> {
533 val ip = args[0]
534 val minutes = if (args.size == 2) args[1].toInt() else 30
535 val chosen = rows.filter { it.ip == ip }
536 if (chosen.isEmpty()) {
537 emit("SESSIONS", "no records")
538 } else {
539 val sessions = mutableListOf(mutableListOf(chosen.first()))
540 for (i in 1 until chosen.size) {
541 if (chosen[i].seconds - chosen[i - 1].seconds <= minutes * 60L)
542 sessions.last().add(chosen[i])
543 else sessions.add(mutableListOf(chosen[i]))
544 }
545 val duration = sessions.sumOf { it.last().seconds - it.first().seconds }
546 emit("SESSIONS", ip, "timeout", minutes, "count", sessions.size,
547 "requests", chosen.size, "avg_duration",
548 decimal(duration, sessions.size.toLong(), 1),
549 "avg_requests", decimal(chosen.size.toLong(), sessions.size.toLong(), 1))
550 for ((i, session) in sessions.withIndex()) {
551 emit(i + 1, session.first().timestamp, session.last().timestamp,
552 session.last().seconds - session.first().seconds,
553 session.size, session.map { it.path }.toSet().size)
554 }
555 }
556 }
557 "BURST" -> {
558 val length = args[0].toInt()
559 val threshold = args[1].toInt()
560 val bursts = rows.filter { it.status >= 400 }.groupBy { it.ip }
561 .map { (ip, group) -> ip to windowPeak(group, length) }
562 .filter { it.second.count >= threshold }
563 .sortedWith { a, b ->
564 val count = b.second.count.compareTo(a.second.count)
565 if (count != 0) count else compareIp(a.first, b.first)
566 }
567 emit("BURST", length, threshold, bursts.size)
568 for ((ip, window) in bursts) {
569 emit(ip, window.count, window.first.timestamp, window.last.timestamp)
570 }
571 }
572 "PEAK" -> {
573 val length = args[0].toInt()
574 val peak = windowPeak(rows, length)
575 emit("PEAK", length, peak.count, peak.first.timestamp, peak.last.timestamp)
576 }
577 "GAPS" -> {
578 val length = args[0].toLong()
579 val gaps = mutableListOf<Pair<Record, Record>>()
580 for (i in 1 until rows.size) {
581 if (rows[i].seconds - rows[i - 1].seconds >= length)
582 gaps.add(rows[i - 1] to rows[i])
583 }
584 emit("GAPS", length, gaps.size)
585 for ((start, end) in gaps)
586 emit(start.timestamp, end.timestamp, end.seconds - start.seconds)
587 }
588 "IP" -> {
589 val ip = args[0]
590 val found = rows.filter { it.ip == ip }
591 if (found.isEmpty()) {
592 emit("IP", "no records")
593 } else {
594 val counts = frequency(found.map { it.path })
595 emit("IP", ip, "requests", found.size, "bytes",
596 found.sumOf { it.size }, "errors", found.count { it.status >= 400 },
597 "paths", counts.size)
598 emit("first", found.first().timestamp, "last", found.last().timestamp)
599 val methods = frequency(found.map { it.method })
600 val methodParts = mutableListOf<Any>("methods")
601 for (method in ranked(methods)) {
602 methodParts.add(method)
603 methodParts.add(methods.getValue(method))
604 }
605 emit(*methodParts.toTypedArray())
606 val categories = classCounts(found)
607 val parts = mutableListOf<Any>("classes")
608 for (i in categories.indices) {
609 parts.add(statusClasses[i])
610 parts.add(categories[i])
611 }
612 emit(*parts.toTypedArray())
613 for (path in ranked(counts).take(3))
614 emit("path", path, counts.getValue(path))
615 }
616 }
617 "RANGE" -> {
618 val start = secondsOf(args[0], args[1])
619 val end = secondsOf(args[2], args[3])
620 requireValue(start <= end, "invalid range")
621 from = start
622 until = end
623 fromText = args[0] + " " + args[1]
624 untilText = args[2] + " " + args[3]
625 emit("RANGE", fromText!!, untilText!!, "active", selected().size)
626 }
627 "FILTER" -> {
628 val key = args[0]
629 val value = if (key == "PATH") normalizePath(args[1]) else args[1]
630 filters[key] = value
631 emit("FILTER", key, value, "active", selected().size)
632 }
633 "UNFILTER" -> {
634 val key = args[0]
635 if (key == "RANGE") {
636 requireValue(from != null, "not filtered")
637 from = null
638 until = null
639 fromText = null
640 untilText = null
641 } else {
642 requireValue(filters[key] != null, "not filtered")
643 filters[key] = null
644 }
645 emit("UNFILTER", key, "active", selected().size)
646 }
647 "RESET" -> {
648 from = null
649 until = null
650 fromText = null
651 untilText = null
652 for (key in filterKeys) filters[key] = null
653 emit("RESET", "active", selected().size)
654 }
655 "STATE" -> {
656 val parts = mutableListOf<Any>("STATE", "active", rows.size,
657 "range", if (from == null) "-" else fromText!! + " " + untilText!!)
658 for (key in filterKeys) {
659 parts.add(key.lowercase(Locale.ROOT))
660 parts.add(filters[key] ?: "-")
661 }
662 emit(*parts.toTypedArray())
663 }
664 }
665 }
666}
