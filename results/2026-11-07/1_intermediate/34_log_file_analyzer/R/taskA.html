32words <- function(line) strsplit(trimws(line), " +")[[1L]]
33canonical <- function(s, width) {
34 nchar(s) <= width && grepl("^(0|[1-9][0-9]*)$", s)
35}
36valid_date <- function(s) {
37 if (!grepl("^[0-9]{4}-[0-9]{2}-[0-9]{2}$", s)) return(FALSE)
38 year <- as.integer(substr(s, 1, 4))
39 year >= 2000 && year <= 2059 && !is.na(as.Date(s, format = "%Y-%m-%d"))
40}
41valid_time <- function(s) {
42 grepl("^[0-9]{2}:[0-9]{2}:[0-9]{2}$", s) &&
43 as.integer(substr(s, 1, 2)) < 24 &&
44 as.integer(substr(s, 4, 5)) < 60 && as.integer(substr(s, 7, 8)) < 60
45}
46valid_ip <- function(s) {
47 if (!grepl("^[0-9]+\\.[0-9]+\\.[0-9]+\\.[0-9]+$", s)) return(FALSE)
48 parts <- strsplit(s, ".", fixed = TRUE)[[1L]]
49 all(vapply(parts, function(p) canonical(p, 3) && as.numeric(p) <= 255, logical(1)))
50}
51valid_status <- function(s) {
52 grepl("^[0-9]{3}$", s) && as.integer(s) >= 100 && as.integer(s) <= 599
53}
54valid_filter <- function(key, value) {
55 switch(key,
56 METHOD = value %in% methods,
57 STATUS = value %in% classes || valid_status(value),
58 IP = valid_ip(value),
59 PATH = startsWith(value, "/")
60 )
61}
62valid_argument <- function(kind, value, args) {
63 if (kind %in% names(ranges)) {
64 bounds <- ranges[[kind]]
65 return(canonical(value, 6) && as.numeric(value) >= bounds[1] &&
66 as.numeric(value) <= bounds[2])
67 }
68 switch(kind,
69 kind = value %in% c("PATHS", "IPS"), path = startsWith(value, "/"),
70 ip = valid_ip(value), date = valid_date(value), time = valid_time(value),
71 key = value %in% filter_keys, subject = value %in% c(filter_keys, "RANGE"),
72 value = valid_filter(args[1], value)
73 )
74}
75malformed_reason <- function(f) {
76 if (length(f) != 8) return("fields")
77 if (!valid_date(f[1])) return("date")
78 if (!valid_time(f[2])) return("time")
79 if (!valid_ip(f[3])) return("ip")
80 if (!f[4] %in% methods) return("method")
81 if (!startsWith(f[5], "/")) return("path")
82 if (!valid_status(f[6])) return("status")
83 if (!canonical(f[7], 8)) return("bytes")
84 if (!canonical(f[8], 6)) return("latency")
85 NA_character_
86}
87seconds_of <- function(date, clock) {
88 days <- as.numeric(as.Date(date, format = "%Y-%m-%d") - as.Date("2000-01-01"))
89 hms <- as.numeric(strsplit(clock, ":", fixed = TRUE)[[1L]])
90 days * 86400 + sum(hms * c(3600, 60, 1))
91}
92interval_label <- function(seconds) {
93 date <- as.Date(seconds %/% 86400, origin = "2000-01-01")
94 minutes <- seconds %% 86400 %/% 60
95 sprintf("%s %02d:%02d", format(date, "%Y-%m-%d"),
96 as.integer(minutes %/% 60), as.integer(minutes %% 60))
97}
98normalize_path <- function(raw) {
99 parts <- character()
100 for (part in strsplit(sub("\\?.*$", "", raw), "/", fixed = TRUE)[[1L]]) {
101 if (part == "..") {
102 if (length(parts)) parts <- head(parts, -1L)
103 } else if (part != "" && part != ".") {
104 parts <- c(parts, part)
105 }
106 }
107 numeric_part <- grepl("^[0-9]+$", parts)
108 hex_part <- nchar(parts) >= 8 & grepl("^[0-9a-f]+$", parts) & grepl("[0-9]", parts)
109 parts[hex_part] <- "{hex}"
110 parts[numeric_part] <- "{id}"
111 paste0("/", paste(parts, collapse = "/"))
112}
113ip_number <- function(ips) {
114 vapply(strsplit(ips, ".", fixed = TRUE),
115 function(p) sum(as.numeric(p) * 256^(3:0)), numeric(1))
116}
117status_matches <- function(value, status) {
118 if (value %in% classes) status %/% 100 == as.numeric(substr(value, 1, 1))
119 else status == as.numeric(value)
120}
121fixed <- function(numerator, denominator, places) {
122 scale <- 10^places
123 value <- (numerator * scale) %/% denominator
124 paste0(sprintf("%.0f", value %/% scale), ".",
125 sprintf(paste0("%0", places, "d"), as.integer(value %% scale)))
126}
127percent <- function(part, total) paste0(fixed(part * 100, total, 2), "%")
128bytes_text <- function(value) {
129 if (value >= 1073741824) return(paste0(fixed(value, 1073741824, 1), "GB"))
130 if (value >= 1048576) return(paste0(fixed(value, 1048576, 1), "MB"))
131 if (value >= 1024) return(paste0(fixed(value, 1024, 1), "KB"))
132 paste0(sprintf("%.0f", value), "B")
133}
134key_counts <- function(values, ip = FALSE) {
135 keys <- unique(values)
136 counts <- tabulate(match(values, keys), nbins = length(keys))
137 index <- order(-counts, if (ip) ip_number(keys) else keys, method = "radix")
138 data.frame(key = keys[index], count = counts[index], stringsAsFactors = FALSE)
139}
140class_counts <- function(rows) tabulate(rows$status %/% 100, nbins = 5)
141buckets <- function(values) {
142 first <- min(values)
143 width <- max(values) - first + 1
144 ensure(width <= 60, "too many buckets")
145 list(first = first, counts = tabulate(as.integer(values - first + 1), nbins = width))
146}
147percentile <- function(values, rank) values[(rank * length(values) + 99) %/% 100]
148window_peak <- function(rows, width) {
149 first <- 1L
150 best <- c(count = 0, first = 1, last = 1)
151 for (last in seq_len(nrow(rows))) {
152 while (rows$seconds[last] - rows$seconds[first] >= width) first <- first + 1L
153 if (last - first + 1 > best["count"]) {
154 best <- c(count = last - first + 1, first = first, last = last)
155 }
156 }
157 best
158}
159apdex <- function(rows, threshold) {
160 satisfied <- sum(rows$status < 500 & rows$latency <= threshold)
161 tolerating <- sum(rows$status < 500 & rows$latency > threshold & rows$latency <= 4 * threshold)
162 n <- nrow(rows)
163 c(count = n, satisfied = satisfied, tolerating = tolerating,
164 frustrated = n - satisfied - tolerating, points = 2 * satisfied + tolerating, total = 2 * n)
165}
166apdex_rating <- function(score) {
167 points <- score["points"] * 100
168 total <- score["total"]
169 if (points >= 94 * total) "excellent"
170 else if (points >= 85 * total) "good"
171 else if (points >= 70 * total) "fair"
172 else if (points >= 50 * total) "poor"
173 else "unacceptable"
174}
175fraction_rank <- function(numerator, denominator) {
176 vapply(seq_along(numerator), function(i) {
177 sum(numerator * denominator[i] < numerator[i] * denominator)
178 }, numeric(1))
179}

181make_analyzer <- function(log_lines) {
182 parsed <- lapply(log_lines, words)
183 bad <- vapply(parsed, malformed_reason, character(1))
184 records <- data.frame(line = integer(), date = character(), timestamp = character(),
185 hour = integer(), seconds = numeric(), ip = character(), method = character(),
186 path = character(), status = integer(), bytes = numeric(), latency = numeric())
187 valid <- which(is.na(bad))
188 if (length(valid)) {
189 records <- do.call(rbind, lapply(valid, function(i) {
190 f <- parsed[[i]]
191 data.frame(line = i, date = f[1], timestamp = paste(f[1], f[2]),
192 hour = as.integer(substr(f[2], 1, 2)), seconds = seconds_of(f[1], f[2]),
193 ip = f[3], method = f[4], path = normalize_path(f[5]),
194 status = as.integer(f[6]), bytes = as.numeric(f[7]), latency = as.numeric(f[8]),
195 stringsAsFactors = FALSE)
196 }))
197 records <- records[order(records$seconds, records$line), , drop = FALSE]
198 }
199 conditions <- setNames(rep(NA_character_, length(filter_keys)), filter_keys)
200 time_range <- NULL
201
202 selected <- function() {
203 keep <- rep(TRUE, nrow(records))
204 if (!is.null(time_range)) {
205 keep <- keep & records$seconds >= time_range$start & records$seconds <= time_range$end
206 }
207 if (!is.na(conditions["METHOD"])) keep <- keep & records$method == conditions["METHOD"]
208 if (!is.na(conditions["STATUS"])) keep <- keep & status_matches(conditions["STATUS"], records$status)
209 if (!is.na(conditions["IP"])) keep <- keep & records$ip == conditions["IP"]
210 if (!is.na(conditions["PATH"])) keep <- keep & records$path == conditions["PATH"]
211 records[keep, , drop = FALSE]
212 }

214 run <- function(name, args) {
215 output <- character()
216 emit <- function(...) {
217 parts <- lapply(list(...), function(x) if (is.numeric(x)) sprintf("%.0f", x) else as.character(x))
218 output <<- c(output, paste(unlist(parts, use.names = FALSE), collapse = " "))
219 }
220 bar_line <- function(label, count, maximum) {
221 if (count == 0) emit(label, count)
222 else emit(label, count, strrep("#", (count * 30 + maximum - 1) %/% maximum))
223 }
224 rows <- selected()
225 n <- nrow(rows)
226 if (name %in% require_records && n == 0) return(paste(name, "no records"))
227 switch(name,
228 SUMMARY = {
229 emit("SUMMARY", "lines", length(log_lines), "valid", nrow(records), "malformed", sum(!is.na(bad)))
230 emit("requests", n, "bytes", sum(rows$bytes), "ips", length(unique(rows$ip)), "paths", length(unique(rows$path)))
231 if (n == 0) emit("span", "none")
232 else emit("span", rows$timestamp[1], rows$timestamp[n], rows$seconds[n] - rows$seconds[1])
233 },
234 MALFORMED = {
235 emit("MALFORMED", sum(!is.na(bad)))
236 for (reason in reasons) {
237 indices <- which(bad == reason)
238 if (length(indices)) emit(reason, length(indices), indices)
239 }
240 },
241 STATUS = {
242 emit("STATUS", n)
243 counts <- class_counts(rows)
244 for (i in seq_along(classes)) emit(classes[i], counts[i], percent(counts[i], n))
245 codes <- sort(unique(rows$status))
246 for (code in codes) {
247 count <- sum(rows$status == code)
248 emit(code, count, percent(count, n))
249 }
250 },
251 METHODS = {
252 emit("METHODS", n)
253 counts <- key_counts(rows$method)
254 for (i in seq_len(nrow(counts))) emit(counts$key[i], counts$count[i], percent(counts$count[i], n))
255 },
256 TOP = {
257 counts <- key_counts(if (args[1] == "PATHS") rows$path else rows$ip, ip = args[1] == "IPS")
258 shown <- min(as.numeric(args[2]), nrow(counts))
259 emit("TOP", args[1], shown, "of", nrow(counts))
260 ranks <- match(counts$count, counts$count)
261 for (i in seq_len(shown)) emit(ranks[i], counts$key[i], counts$count[i], percent(counts$count[i], n))
262 },
263 HOURLY = {
264 counts <- tabulate(rows$hour + 1L, nbins = 24)
265 emit("HOURLY", n, "peak", sprintf("%02d", which.max(counts) - 1L), max(counts))
266 for (i in seq_along(counts)) bar_line(sprintf("%02d", i - 1L), counts[i], max(counts))
267 },
268 TIMELINE = {
269 minutes <- as.numeric(args[1])
270 width <- minutes * 60
271 b <- buckets(rows$seconds %/% width)
272 emit("TIMELINE", minutes, length(b$counts))
273 for (i in seq_along(b$counts)) bar_line(interval_label((b$first + i - 1) * width), b$counts[i], max(b$counts))
274 },
275 BANDWIDTH = {
276 dates <- sort(unique(rows$date), method = "radix")
277 total <- sum(rows$bytes)
278 emit("BANDWIDTH", length(dates), total, bytes_text(total))
279 for (date in dates) {
280 group <- rows[rows$date == date, , drop = FALSE]
281 size <- sum(group$bytes)
282 emit(date, nrow(group), size, bytes_text(size))
283 }
284 },
285 VISITORS = {
286 dates <- sort(unique(rows$date), method = "radix")
287 first <- rows[!duplicated(rows$ip), c("ip", "date"), drop = FALSE]
288 emit("VISITORS", length(dates), nrow(first))
289 for (date in dates) {
290 ips <- unique(rows$ip[rows$date == date])
291 fresh <- sum(first$date[match(ips, first$ip)] == date)
292 emit(date, "unique", length(ips), "new", fresh, "returning", length(ips) - fresh)
293 }
294 },
295 LATENCY = {
296 label <- if (length(args)) normalize_path(args[1]) else "all"
297 values <- if (length(args)) rows$latency[rows$path == label] else rows$latency
298 if (!length(values)) emit("LATENCY", "no records")
299 else {
300 values <- sort(values)
301 count <- length(values)
302 trim <- count %/% 10
303 trimmed <- values[seq.int(trim + 1, count - trim)]
304 emit("LATENCY", label, "count", count, "min", values[1],
305 "p50", percentile(values, 50), "p90", percentile(values, 90),
306 "p99", percentile(values, 99), "max", values[count],
307 "mean", fixed(sum(values), count, 2), "trimmed", fixed(sum(trimmed), length(trimmed), 2))
308 }
309 },
310 HISTOGRAM = {
311 width <- as.numeric(args[1])
312 b <- buckets(rows$latency %/% width)
313 emit("HISTOGRAM", width, length(b$counts))
314 for (i in seq_along(b$counts)) {
315 lower <- (b$first + i - 1) * width
316 bar_line(sprintf("%.0f-%.0f", lower, lower + width - 1), b$counts[i], max(b$counts))
317 }
318 },
319 APDEX = {
320 threshold <- as.numeric(args[1])
321 s <- apdex(rows, threshold)
322 emit("APDEX", threshold, fixed(s["points"], s["total"], 3), apdex_rating(s),
323 "satisfied", s["satisfied"], "tolerating", s["tolerating"], "frustrated", s["frustrated"])
324 groups <- split(seq_len(n), rows$path)
325 scores <- t(vapply(groups, function(i) apdex(rows[i, , drop = FALSE], threshold), numeric(6)))
326 ranking <- order(fraction_rank(scores[, "points"], scores[, "total"]),
327 -scores[, "count"], names(groups), method = "radix")
328 for (i in ranking) {
329 s <- scores[i, ]
330 emit(names(groups)[i], fixed(s["points"], s["total"], 3), apdex_rating(s), s["count"])
331 }
332 },
333 ERRORS = {
334 failed <- sum(rows$status >= 400)
335 emit("ERRORS", failed, "of", n, percent(failed, n))
336 groups <- split(seq_len(n), rows$path)
337 stats <- t(vapply(groups, function(i) {
338 c(count = length(i), client = sum(rows$status[i] %/% 100 == 4), server = sum(rows$status[i] %/% 100 == 5))
339 }, numeric(3)))
340 failures <- stats[, "client"] + stats[, "server"]
341 ranking <- order(-fraction_rank(failures, stats[, "count"]), -failures, names(groups), method = "radix")
342 for (i in ranking[failures[ranking] > 0]) {
343 emit(names(groups)[i], stats[i, "count"], stats[i, "client"], stats[i, "server"], percent(failures[i], stats[i, "count"]))
344 }
345 },
346 SLOWEST = {
347 groups <- split(seq_len(n), paste(rows$method, rows$path, sep = "\001"))
348 stats <- do.call(rbind, lapply(groups, function(i) {
349 data.frame(method = rows$method[i[1]], path = rows$path[i[1]], count = length(i),
350 total = sum(rows$latency[i]), maximum = max(rows$latency[i]), stringsAsFactors = FALSE)
351 }))
352 ranking <- order(-fraction_rank(stats$total, stats$count), -stats$maximum, stats$path, stats$method, method = "radix")
353 shown <- min(as.numeric(args[1]), nrow(stats))
354 emit("SLOWEST", shown, "of", nrow(stats))
355 for (rank in seq_len(shown)) {
356 s <- stats[ranking[rank], ]
357 emit(rank, s$method, s$path, "count", s$count, "avg", fixed(s$total, s$count, 2), "max", s$maximum)
358 }
359 },
360 SESSIONS = {
361 ip <- args[1]
362 minutes <- if (length(args) == 2) as.numeric(args[2]) else 30
363 found <- rows[rows$ip == ip, , drop = FALSE]
364 if (!nrow(found)) emit("SESSIONS", "no records")
365 else {
366 starts <- which(c(TRUE, diff(found$seconds) > minutes * 60))
367 ends <- c(starts[-1] - 1L, nrow(found))
368 durations <- found$seconds[ends] - found$seconds[starts]
369 emit("SESSIONS", ip, "timeout", minutes, "count", length(starts), "requests", nrow(found),
370 "avg_duration", fixed(sum(durations), length(starts), 1),
371 "avg_requests", fixed(nrow(found), length(starts), 1))
372 for (i in seq_along(starts)) {
373 emit(i, found$timestamp[starts[i]], found$timestamp[ends[i]], durations[i],
374 ends[i] - starts[i] + 1, length(unique(found$path[seq.int(starts[i], ends[i])])))
375 }
376 }
377 },
378 BURST = {
379 width <- as.numeric(args[1])
380 threshold <- as.numeric(args[2])
381 failed <- rows[rows$status >= 400, , drop = FALSE]
382 groups <- split(seq_len(nrow(failed)), failed$ip)
383 peaks <- lapply(groups, function(i) window_peak(failed[i, , drop = FALSE], width))
384 sizes <- vapply(peaks, function(p) unname(p["count"]), numeric(1))
385 ranking <- order(-sizes, ip_number(names(groups)), method = "radix")
386 ranking <- ranking[sizes[ranking] >= threshold]
387 emit("BURST", width, threshold, length(ranking))
388 for (i in ranking) {
389 p <- peaks[[i]]
390 group <- failed[groups[[i]], , drop = FALSE]
391 emit(names(groups)[i], p["count"], group$timestamp[p["first"]], group$timestamp[p["last"]])
392 }
393 },
394 PEAK = {
395 width <- as.numeric(args[1])
396 p <- window_peak(rows, width)
397 emit("PEAK", width, p["count"], rows$timestamp[p["first"]], rows$timestamp[p["last"]])
398 },
399 GAPS = {
400 width <- as.numeric(args[1])
401 gaps <- which(diff(rows$seconds) >= width)
402 emit("GAPS", width, length(gaps))
403 for (i in gaps) emit(rows$timestamp[i], rows$timestamp[i + 1L], rows$seconds[i + 1L] - rows$seconds[i])
404 },
405 IP = {
406 ip <- args[1]
407 found <- rows[rows$ip == ip, , drop = FALSE]
408 if (!nrow(found)) emit("IP", "no records")
409 else {
410 paths <- key_counts(found$path)
411 emit("IP", ip, "requests", nrow(found), "bytes", sum(found$bytes),
412 "errors", sum(found$status >= 400), "paths", nrow(paths))
413 emit("first", found$timestamp[1], "last", tail(found$timestamp, 1))
414 counts <- key_counts(found$method)
415 emit("methods", as.vector(t(cbind(counts$key, sprintf("%d", counts$count)))))
416 emit("classes", as.vector(t(cbind(classes, sprintf("%d", class_counts(found))))))
417 for (i in seq_len(min(3, nrow(paths)))) emit("path", paths$key[i], paths$count[i])
418 }
419 },
420 RANGE = {
421 start <- seconds_of(args[1], args[2])
422 end <- seconds_of(args[3], args[4])
423 ensure(start <= end, "invalid range")
424 time_range <<- list(start = start, end = end, from = paste(args[1], args[2]), to = paste(args[3], args[4]))
425 emit("RANGE", time_range$from, time_range$to, "active", nrow(selected()))
426 },
427 FILTER = {
428 key <- args[1]
429 value <- if (key == "PATH") normalize_path(args[2]) else args[2]
430 conditions[key] <<- value
431 emit("FILTER", key, value, "active", nrow(selected()))
432 },
433 UNFILTER = {
434 key <- args[1]
435 if (key == "RANGE") {
436 ensure(!is.null(time_range), "not filtered")
437 time_range <<- NULL
438 } else {
439 ensure(!is.na(conditions[key]), "not filtered")
440 conditions[key] <<- NA_character_
441 }
442 emit("UNFILTER", key, "active", nrow(selected()))
443 },
444 RESET = {
445 time_range <<- NULL
446 conditions[] <<- NA_character_
447 emit("RESET", "active", nrow(selected()))
448 },
449 STATE = {
450 window <- if (is.null(time_range)) "-" else paste(time_range$from, time_range$to)
451 values <- ifelse(is.na(conditions), "-", conditions)
452 emit("STATE", "active", n, "range", window,
453 as.vector(t(cbind(tolower(filter_keys), values))))
454 }
455 )
456 output
457 }

459 function(line) {
460 tryCatch({
461 fields <- words(line)
462 name <- fields[1]
463 args <- fields[-1]
464 ensure(name %in% names(forms), "unknown query")
465 form <- forms[[name]]
466 optional <- name %in% c("LATENCY", "SESSIONS") && length(args) == length(form) - 1
467 ensure(length(args) == length(form) || optional, "wrong arguments")
468 for (i in seq_along(args)) ensure(valid_argument(form[i], args[i], args), argument_errors[[form[i]]])
469 run(name, args)
470 }, query_error = function(error) paste("ERROR", error$message))
471 }
472}

474main <- function() {
475 input <- readLines(stdin(), warn = FALSE)
476 separator <- match("---", input)
477 queries <- input[seq.int(separator + 1L, length(input))]
478 end <- match("END", queries)

480 for (line in queries[seq_len(end - 1L)]) {
481 cat(paste(execute(line), collapse = "\n"), "\n", sep = "")
482 }
