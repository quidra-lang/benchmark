11 'SUMMARY' => [], 'MALFORMED' => [], 'STATUS' => [], 'METHODS' => [],

42function check(bool $condition, string $reason): void
43{
44 if (!$condition) throw new CommandError($reason);
45}

47function validDigits(string $text): bool
48{
49 return preg_match('/\A[0-9]+\z/', $text) === 1;
50}
51
52function canonicalInteger(string $text, int $maxLength): bool
53{
54 return validDigits($text) && strlen($text) <= $maxLength &&
55 (strlen($text) === 1 || $text[0] !== '0');
56}
57
58function leapYear(int $year): bool
59{
60 return $year % 4 === 0 && ($year % 100 !== 0 || $year % 400 === 0);
61}
62
63function daysInYear(int $year): int
64{
65 return leapYear($year) ? 366 : 365;
66}
67
68function daysInMonth(int $year, int $month): int
69{
70 return $month === 2 && leapYear($year) ? 29 : MONTH_DAYS[$month - 1];
71}
72
73function validDate(string $text): bool
74{
75 if (preg_match('/\A[0-9]{4}-[0-9]{2}-[0-9]{2}\z/', $text) !== 1) return false;
76 [$year, $month, $day] = array_map('intval', explode('-', $text));
77 return $year >= 2000 && $year <= 2059 && $month >= 1 && $month <= 12 &&
78 $day >= 1 && $day <= daysInMonth($year, $month);
79}
80
81function validTime(string $text): bool
82{
83 if (preg_match('/\A[0-9]{2}:[0-9]{2}:[0-9]{2}\z/', $text) !== 1) return false;
84 [$hour, $minute, $second] = array_map('intval', explode(':', $text));
85 return $hour <= 23 && $minute <= 59 && $second <= 59;
86}
87
88function validIP(string $text): bool
89{
90 $parts = explode('.', $text);
91 if (count($parts) !== 4) return false;
92 foreach ($parts as $part) {
93 if (!canonicalInteger($part, 3) || (int)$part > 255) return false;
94 }
95 return true;
96}
97
98function validStatus(string $text): bool
99{
100 return preg_match('/\A[0-9]{3}\z/', $text) === 1 &&
101 (int)$text >= 100 && (int)$text <= 599;
102}
103
104function validFilter(string $key, string $value): bool
105{
106 return match ($key) {
107 'METHOD' => in_array($value, METHODS, true),
108 'STATUS' => in_array($value, STATUS_CLASSES, true) || validStatus($value),
109 'IP' => validIP($value),
110 default => str_starts_with($value, '/'),
111 };
112}
113
114function validArgument(string $type, string $text, array $arguments): bool
115{
116 if (isset(ARGUMENT_RANGES[$type])) {
117 [$low, $high] = ARGUMENT_RANGES[$type];
118 return canonicalInteger($text, 6) && (int)$text >= $low && (int)$text <= $high;
119 }
120 return match ($type) {
121 'kind' => in_array($text, ['PATHS', 'IPS'], true),
122 'path' => str_starts_with($text, '/'),
123 'ip' => validIP($text),
124 'date' => validDate($text),
125 'time' => validTime($text),
126 'key' => in_array($text, FILTER_KEYS, true),
127 'subject' => $text === 'RANGE' || in_array($text, FILTER_KEYS, true),
128 default => validFilter($arguments[0], $text),
129 };
130}
131
132function malformedReason(array $fields): ?string
133{
134 if (count($fields) !== 8) return 'fields';
135 [$date, $clock, $ip, $method, $path, $status, $bytes, $latency] = $fields;
136 if (!validDate($date)) return 'date';
137 if (!validTime($clock)) return 'time';
138 if (!validIP($ip)) return 'ip';
139 if (!in_array($method, METHODS, true)) return 'method';
140 if (!str_starts_with($path, '/')) return 'path';
141 if (!validStatus($status)) return 'status';
142 if (!canonicalInteger($bytes, 8)) return 'bytes';
143 if (!canonicalInteger($latency, 6)) return 'latency';
144 return null;
145}
146
147function secondsOf(string $date, string $clock): int
148{
149 [$year, $month, $day] = array_map('intval', explode('-', $date));
150 [$hour, $minute, $second] = array_map('intval', explode(':', $clock));
151 $days = $day - 1;
152 for ($y = 2000; $y < $year; ++$y) $days += daysInYear($y);
153 for ($m = 1; $m < $month; ++$m) $days += daysInMonth($year, $m);
154 return $days * 86400 + $hour * 3600 + $minute * 60 + $second;
155}
156
157function intervalLabel(int $seconds): string
158{
159 $days = intdiv($seconds, 86400);
160 $rest = $seconds % 86400;
161 $year = 2000;
162 while ($days >= daysInYear($year)) $days -= daysInYear($year++);
163 $month = 1;
164 while ($days >= daysInMonth($year, $month)) $days -= daysInMonth($year, $month++);
165 return sprintf('%d-%02d-%02d %02d:%02d', $year, $month, $days + 1,
166 intdiv($rest, 3600), intdiv($rest % 3600, 60));
167}

169function normalizePath(string $raw): string
170{
171 $parts = [];
172 foreach (explode('/', explode('?', $raw, 2)[0]) as $part) {
173 if ($part === '..') array_pop($parts);
174 elseif ($part !== '' && $part !== '.') $parts[] = $part;
175 }
176 return '/' . implode('/', array_map(static function (string $part): string {
177 if (validDigits($part)) return '{id}';
178 if (strlen($part) >= 8 && preg_match('/\A[0-9a-f]+\z/', $part) === 1 &&
179 preg_match('/[0-9]/', $part) === 1) return '{hex}';
180 return $part;
181 }, $parts));
182}

184function compareIP(string $a, string $b): int
185{
186 $left = array_map('intval', explode('.', $a));
187 $right = array_map('intval', explode('.', $b));
188 for ($i = 0; $i < 4; ++$i) {
189 if ($left[$i] !== $right[$i]) return $left[$i] <=> $right[$i];
190 }
191 return 0;
192}

194function decimal(int $numerator, int $denominator, int $places): string
195{
196 $scale = 10 ** $places;
197 $value = intdiv($numerator * $scale, $denominator);
198 return intdiv($value, $scale) . '.' . str_pad((string)($value % $scale), $places, '0', STR_PAD_LEFT);
199}
200
201function percentage(int $numerator, int $denominator): string
202{
203 return decimal($numerator * 100, $denominator, 2) . '%';
204}
205
206function bar(int $count, int $maximum): string
207{
208 return str_repeat('#', intdiv($count * 30 + $maximum - 1, $maximum));
209}
210
211function sizeNotation(int $bytes): string
212{
213 foreach ([[1073741824, 'GB'], [1048576, 'MB'], [1024, 'KB']] as [$base, $unit]) {
214 if ($bytes >= $base) return decimal($bytes, $base, 1) . $unit;
215 }
216 return $bytes . 'B';
217}

219function statusCounts(array $records): array
220{
221 $counts = array_fill(0, 5, 0);
222 foreach ($records as $r) ++$counts[intdiv($r->status, 100) - 1];
223 return $counts;
224}

226function tally(array $records, callable $key): array
227{
228 $counts = [];
229 foreach ($records as $r) {
230 $value = $key($r);
231 $counts[$value] = ($counts[$value] ?? 0) + 1;
232 }
233 return $counts;
234}
235
236function grouped(array $records, callable $key): array
237{
238 $groups = [];
239 foreach ($records as $r) $groups[$key($r)][] = $r;
240 return $groups;
241}
242
243function orderedByCount(array $counts, ?callable $tie = null): array
244{
245 $keys = array_keys($counts);
246 usort($keys, static function ($a, $b) use ($counts, $tie): int {
247 $cmp = $counts[$b] <=> $counts[$a];
248 if ($cmp !== 0) return $cmp;
249 return $tie ? $tie((string)$a, (string)$b) : strcmp((string)$a, (string)$b);
250 });
251 return $keys;
252}

254function buckets(array $numbers): array
255{
256 $first = min($numbers);
257 $last = max($numbers);
258 check($last - $first + 1 <= 60, 'too many buckets');
259 $counts = array_fill(0, $last - $first + 1, 0);
260 foreach ($numbers as $n) ++$counts[$n - $first];
261 return [$first, $counts];
262}
263
264function percentile(array $sorted, int $rank): int
265{
266 return $sorted[intdiv($rank * count($sorted) + 99, 100) - 1];
267}
268
269function bestWindow(array $records, int $length): array
270{
271 $size = 0;
272 $start = $end = null;
273 $first = 0;
274 foreach ($records as $last => $r) {
275 while ($r->seconds - $records[$first]->seconds >= $length) ++$first;
276 if ($last - $first + 1 > $size) {
277 $size = $last - $first + 1;
278 $start = $records[$first];
279 $end = $r;
280 }
281 }
282 return [$size, $start, $end];
283}

285function apdex(array $records, int $threshold): array
286{
287 $sat = $tol = 0;
288 foreach ($records as $r) {
289 if ($r->status >= 500) continue;
290 if ($r->latency <= $threshold) ++$sat;
291 elseif ($r->latency <= 4 * $threshold) ++$tol;
292 }
293 $count = count($records);
294 $frustrated = $count - $sat - $tol;
295 $points = 2 * $sat + $tol;
296 $total = 2 * $count;
297 $rating = 'unacceptable';
298 foreach ([[94, 'excellent'], [85, 'good'], [70, 'fair'], [50, 'poor']] as [$minimum, $name]) {
299 if ($points * 100 >= $minimum * $total) { $rating = $name; break; }
300 }
301 return [decimal($points, $total, 3), $rating, $sat, $tol,
302 $frustrated, $count, $points, $total];
303}

332final class LogAnalyzer
333{
334 private array $records = [];
335 private array $malformed = [];
336 private array $conditions = [];
337 private ?array $range = null;
338 private array $output = [];
339 private int $lineCount;
340
341 public function __construct(array $lines)
342 {
343 $this->lineCount = count($lines);
344 foreach ($lines as $index => $line) {
345 $fields = preg_split('/\s+/', trim($line));
346 $reason = malformedReason($fields);
347 if ($reason === null) $this->records[] = new LogRecord($index + 1, $fields);
348 else $this->malformed[] = [$index + 1, $reason];
349 }
350 usort($this->records, static fn(LogRecord $a, LogRecord $b): int =>
351 ($a->seconds <=> $b->seconds) ?: ($a->line <=> $b->line));
352 foreach (FILTER_KEYS as $key) $this->conditions[$key] = null;
353 }
354
355 private function emit(...$parts): void
356 {
357 $this->output[] = implode(' ', array_map('strval', $parts));
358 }
359
360 private function barLine(string $label, int $count, int $maximum): void
361 {
362 if ($count) $this->emit($label, $count, bar($count, $maximum));
363 else $this->emit($label, 0);
364 }
365
366 private function selected(): array
367 {
368 return array_values(array_filter($this->records, function (LogRecord $r): bool {
369 if ($this->range !== null &&
370 ($r->seconds < $this->range[2] || $r->seconds > $this->range[3])) return false;
371 if ($this->conditions['METHOD'] !== null && $r->method !== $this->conditions['METHOD']) return false;
372 if ($this->conditions['STATUS'] !== null) {
373 $status = $this->conditions['STATUS'];
374 if (in_array($status, STATUS_CLASSES, true)) {
375 if (intdiv($r->status, 100) !== (int)$status[0]) return false;
376 } elseif ($r->status !== (int)$status) return false;
377 }
378 if ($this->conditions['IP'] !== null && $r->ip !== $this->conditions['IP']) return false;
379 return $this->conditions['PATH'] === null || $r->path === $this->conditions['PATH'];
380 }));
381 }
382
383 public function run(string $line): void
384 {
385 $tokens = preg_split('/\s+/', trim($line));
386 $name = array_shift($tokens);
387 try {
388 check(array_key_exists($name, QUERY_FORMS), 'unknown query');
389 $form = QUERY_FORMS[$name];
390 $size = count($form);
391 $optional = in_array($name, OPTIONAL_LAST, true) && count($tokens) === $size - 1;
392 check(count($tokens) === $size || $optional, 'wrong arguments');
393 foreach ($tokens as $i => $argument) {
394 $kind = $form[$i];
395 check(validArgument($kind, $argument, $tokens), ARGUMENT_ERRORS[$kind]);
396 }
397 $records = $this->selected();
398 if (in_array($name, NEEDS_TARGETS, true) && !$records) {
399 $this->emit($name, 'no records');
400 return;
401 }
402 $this->command($name, $tokens, $records);
403 } catch (CommandError $error) {
404 $this->emit('ERROR', $error->getMessage());
405 }
406 }
407
408 private function command(string $name, array $a, array $records): void
409 {
410 $count = count($records);
411 switch ($name) {
412 case 'SUMMARY':
413 $this->emit('SUMMARY', 'lines', $this->lineCount, 'valid', count($this->records),
414 'malformed', count($this->malformed));
415 $this->emit('requests', $count, 'bytes', array_sum(array_map(static fn($r) => $r->size, $records)),
416 'ips', count(array_unique(array_map(static fn($r) => $r->ip, $records))),
417 'paths', count(array_unique(array_map(static fn($r) => $r->path, $records))));
418 if (!$records) $this->emit('span', 'none');
419 else {
420 $last = $records[$count - 1];
421 $this->emit('span', $records[0]->timestamp, $last->timestamp, $last->seconds - $records[0]->seconds);
422 }
423 break;
424 case 'MALFORMED':
425 $this->emit('MALFORMED', count($this->malformed));
426 foreach (REASONS as $reason) {
427 $numbers = [];
428 foreach ($this->malformed as [$index, $cause]) {
429 if ($cause === $reason) $numbers[] = $index;
430 }
431 if ($numbers) $this->emit($reason, count($numbers), ...$numbers);
432 }
433 break;
434 case 'STATUS':
435 $this->emit('STATUS', $count);
436 foreach (statusCounts($records) as $i => $total) {
437 $this->emit(STATUS_CLASSES[$i], $total, percentage($total, $count));
438 }
439 $statuses = tally($records, static fn($r) => $r->status);
440 ksort($statuses, SORT_NUMERIC);
441 foreach ($statuses as $status => $total) $this->emit($status, $total, percentage($total, $count));
442 break;
443 case 'METHODS':
444 $totals = tally($records, static fn($r) => $r->method);
445 $this->emit('METHODS', $count);
446 foreach (orderedByCount($totals) as $method) {
447 $this->emit($method, $totals[$method], percentage($totals[$method], $count));
448 }
449 break;
450 case 'TOP':
451 [$kind, $limit] = $a;
452 $totals = tally($records, static fn($r) => $kind === 'PATHS' ? $r->path : $r->ip);
453 $keys = orderedByCount($totals, $kind === 'IPS' ? 'compareIP' : null);
454 $shown = min((int)$limit, count($keys));
455 $this->emit('TOP', $kind, $shown, 'of', count($keys));
456 $rank = 0;
457 for ($i = 0; $i < $shown; ++$i) {
458 $key = $keys[$i];
459 if ($i === 0 || $totals[$key] !== $totals[$keys[$i - 1]]) $rank = $i + 1;
460 $this->emit($rank, $key, $totals[$key], percentage($totals[$key], $count));
461 }
462 break;
463 case 'HOURLY':
464 $totals = array_fill(0, 24, 0);
465 foreach ($records as $r) ++$totals[$r->hour];
466 $largest = max($totals);
467 $this->emit('HOURLY', $count, 'peak', sprintf('%02d', array_search($largest, $totals, true)), $largest);
468 foreach ($totals as $hour => $total) $this->barLine(sprintf('%02d', $hour), $total, $largest);
469 break;
470 case 'TIMELINE':
471 $minutes = (int)$a[0];
472 $width = $minutes * 60;
473 [$first, $totals] = buckets(array_map(static fn($r) => intdiv($r->seconds, $width), $records));
474 $this->emit('TIMELINE', $minutes, count($totals));
475 $largest = max($totals);
476 foreach ($totals as $i => $total) $this->barLine(intervalLabel(($first + $i) * $width), $total, $largest);
477 break;
478 case 'BANDWIDTH':
479 $days = grouped($records, static fn($r) => $r->date);
480 $bytes = array_sum(array_map(static fn($r) => $r->size, $records));
481 $this->emit('BANDWIDTH', count($days), $bytes, sizeNotation($bytes));
482 ksort($days, SORT_STRING);
483 foreach ($days as $date => $items) {
484 $amount = array_sum(array_map(static fn($r) => $r->size, $items));
485 $this->emit($date, count($items), $amount, sizeNotation($amount));
486 }
487 break;
488 case 'VISITORS':
489 $first = [];
490 foreach ($records as $r) {
491 if (!isset($first[$r->ip]) || strcmp($r->date, $first[$r->ip]) < 0) $first[$r->ip] = $r->date;
492 }
493 $days = grouped($records, static fn($r) => $r->date);
494 $this->emit('VISITORS', count($days), count($first));
495 ksort($days, SORT_STRING);
496 foreach ($days as $date => $items) {
497 $ips = array_unique(array_map(static fn($r) => $r->ip, $items));
498 $new = count(array_filter($ips, static fn($ip) => $first[$ip] === $date));
499 $this->emit($date, 'unique', count($ips), 'new', $new, 'returning', count($ips) - $new);
500 }
501 break;
502 case 'LATENCY':
503 $label = $a ? normalizePath($a[0]) : 'all';
504 $items = $a ? array_values(array_filter($records, static fn($r) => $r->path === $label)) : $records;
505 if (!$items) { $this->emit('LATENCY', 'no records'); break; }
506 $times = array_map(static fn($r) => $r->latency, $items);
507 sort($times, SORT_NUMERIC);
508 $n = count($times);
509 $trim = intdiv($n, 10);
510 $trimmed = array_slice($times, $trim, $n - 2 * $trim);
511 $this->emit('LATENCY', $label, 'count', $n, 'min', $times[0],
512 'p50', percentile($times, 50), 'p90', percentile($times, 90),
513 'p99', percentile($times, 99), 'max', $times[$n - 1],
514 'mean', decimal(array_sum($times), $n, 2),
515 'trimmed', decimal(array_sum($trimmed), count($trimmed), 2));
516 break;
517 case 'HISTOGRAM':
518 $width = (int)$a[0];
519 [$first, $totals] = buckets(array_map(static fn($r) => intdiv($r->latency, $width), $records));
520 $this->emit('HISTOGRAM', $width, count($totals));
521 $largest = max($totals);
522 foreach ($totals as $i => $total) {
523 $low = ($first + $i) * $width;
524 $this->barLine($low . '-' . ($low + $width - 1), $total, $largest);
525 }
526 break;
527 case 'APDEX':
528 $threshold = (int)$a[0];
529 [$notation, $rating, $sat, $tol, $frustrated] = apdex($records, $threshold);
530 $this->emit('APDEX', $threshold, $notation, $rating, 'satisfied', $sat,
531 'tolerating', $tol, 'frustrated', $frustrated);
532 $groups = grouped($records, static fn($r) => $r->path);
533 $scores = [];
534 foreach ($groups as $path => $items) $scores[] = [$path, apdex($items, $threshold)];
535 usort($scores, static function (array $a, array $b): int {
536 $x = $a[1]; $y = $b[1];
537 return ($x[6] * $y[7] <=> $y[6] * $x[7]) ?: ($y[5] <=> $x[5]) ?: strcmp($a[0], $b[0]);
538 });
539 foreach ($scores as [$path, $score]) $this->emit($path, $score[0], $score[1], $score[5]);
540 break;
541 case 'ERRORS':
542 $failed = count(array_filter($records, static fn($r) => $r->status >= 400));
543 $this->emit('ERRORS', $failed, 'of', $count, percentage($failed, $count));
544 $groups = grouped($records, static fn($r) => $r->path);
545 $rows = [];
546 foreach ($groups as $path => $items) {
547 $client = count(array_filter($items, static fn($r) => intdiv($r->status, 100) === 4));
548 $server = count(array_filter($items, static fn($r) => intdiv($r->status, 100) === 5));
549 if ($client + $server > 0) $rows[] = [$path, count($items), $client, $server];
550 }
551 usort($rows, static function (array $a, array $b): int {
552 $f1 = $a[2] + $a[3];
553 $f2 = $b[2] + $b[3];
554 return ($f2 * $a[1] <=> $f1 * $b[1]) ?: ($f2 <=> $f1) ?: strcmp($a[0], $b[0]);
555 });
556 foreach ($rows as [$path, $n, $client, $server]) {
557 $this->emit($path, $n, $client, $server, percentage($client + $server, $n));
558 }
559 break;
560 case 'SLOWEST':
561 $groups = grouped($records, static fn($r) => $r->method . "\x1F" . $r->path);
562 $rows = [];
563 foreach ($groups as $key => $items) {
564 [$method, $path] = explode("\x1F", $key, 2);
565 $rows[] = [$method, $path, count($items),
566 array_sum(array_map(static fn($r) => $r->latency, $items)),
567 max(array_map(static fn($r) => $r->latency, $items))];
568 }
569 usort($rows, static fn(array $a, array $b): int =>
570 ($b[3] * $a[2] <=> $a[3] * $b[2]) ?: ($b[4] <=> $a[4]) ?:
571 strcmp($a[1], $b[1]) ?: strcmp($a[0], $b[0]));
572 $shown = min((int)$a[0], count($rows));
573 $this->emit('SLOWEST', $shown, 'of', count($rows));
574 foreach (array_slice($rows, 0, $shown) as $i => [$method, $path, $n, $sum, $max]) {
575 $this->emit($i + 1, $method, $path, 'count', $n,
576 'avg', decimal($sum, $n, 2), 'max', $max);
577 }
578 break;
579 case 'SESSIONS':
580 $ip = $a[0];
581 $minutes = isset($a[1]) ? (int)$a[1] : 30;
582 $items = array_values(array_filter($records, static fn($r) => $r->ip === $ip));
583 if (!$items) { $this->emit('SESSIONS', 'no records'); break; }
584 $sessions = [[$items[0]]];
585 for ($i = 1, $n = count($items); $i < $n; ++$i) {
586 $last = count($sessions) - 1;
587 if ($items[$i]->seconds - $items[$i - 1]->seconds <= $minutes * 60)
588 $sessions[$last][] = $items[$i];
589 else $sessions[] = [$items[$i]];
590 }
591 $durations = 0;
592 foreach ($sessions as $session) $durations += end($session)->seconds - $session[0]->seconds;
593 $this->emit('SESSIONS', $ip, 'timeout', $minutes, 'count', count($sessions),
594 'requests', count($items), 'avg_duration', decimal($durations, count($sessions), 1),
595 'avg_requests', decimal(count($items), count($sessions), 1));
596 foreach ($sessions as $i => $session) {
597 $first = $session[0]; $last = end($session);
598 $paths = array_unique(array_map(static fn($r) => $r->path, $session));
599 $this->emit($i + 1, $first->timestamp, $last->timestamp,
600 $last->seconds - $first->seconds, count($session), count($paths));
601 }
602 break;
603 case 'BURST':
604 [$length, $times] = array_map('intval', $a);
605 $groups = grouped(array_values(array_filter($records, static fn($r) => $r->status >= 400)),
606 static fn($r) => $r->ip);
607 $rows = [];
608 foreach ($groups as $ip => $items) {
609 [$size, $first, $last] = bestWindow($items, $length);
610 if ($size >= $times) $rows[] = [$ip, $size, $first, $last];
611 }
612 usort($rows, static fn(array $a, array $b): int =>
613 ($b[1] <=> $a[1]) ?: compareIP($a[0], $b[0]));
614 $this->emit('BURST', $length, $times, count($rows));
615 foreach ($rows as [$ip, $size, $first, $last]) {
616 $this->emit($ip, $size, $first->timestamp, $last->timestamp);
617 }
618 break;
619 case 'PEAK':
620 $length = (int)$a[0];
621 [$size, $first, $last] = bestWindow($records, $length);
622 $this->emit('PEAK', $length, $size, $first->timestamp, $last->timestamp);
623 break;
624 case 'GAPS':
625 $length = (int)$a[0];
626 $gaps = [];
627 for ($i = 1; $i < $count; ++$i) {
628 if ($records[$i]->seconds - $records[$i - 1]->seconds >= $length)
629 $gaps[] = [$records[$i - 1], $records[$i]];
630 }
631 $this->emit('GAPS', $length, count($gaps));
632 foreach ($gaps as [$before, $after]) {
633 $this->emit($before->timestamp, $after->timestamp, $after->seconds - $before->seconds);
634 }
635 break;
636 case 'IP':
637 $ip = $a[0];
638 $items = array_values(array_filter($records, static fn($r) => $r->ip === $ip));
639 if (!$items) { $this->emit('IP', 'no records'); break; }
640 $paths = tally($items, static fn($r) => $r->path);
641 $this->emit('IP', $ip, 'requests', count($items),
642 'bytes', array_sum(array_map(static fn($r) => $r->size, $items)),
643 'errors', count(array_filter($items, static fn($r) => $r->status >= 400)),
644 'paths', count($paths));
645 $this->emit('first', $items[0]->timestamp, 'last', end($items)->timestamp);
646 $methods = tally($items, static fn($r) => $r->method);
647 $parts = ['methods'];
648 foreach (orderedByCount($methods) as $method) array_push($parts, $method, $methods[$method]);
649 $this->emit(...$parts);
650 $parts = ['classes'];
651 foreach (statusCounts($items) as $i => $total) array_push($parts, STATUS_CLASSES[$i], $total);
652 $this->emit(...$parts);
653 foreach (array_slice(orderedByCount($paths), 0, 3) as $path) {
654 $this->emit('path', $path, $paths[$path]);
655 }
656 break;
657 case 'RANGE':
658 $start = $a[0] . ' ' . $a[1];
659 $end = $a[2] . ' ' . $a[3];
660 $first = secondsOf($a[0], $a[1]);
661 $last = secondsOf($a[2], $a[3]);
662 check($first <= $last, 'invalid range');
663 $this->range = [$start, $end, $first, $last];
664 $this->emit('RANGE', $start, $end, 'active', count($this->selected()));
665 break;
666 case 'FILTER':
667 [$key, $value] = $a;
668 if ($key === 'PATH') $value = normalizePath($value);
669 $this->conditions[$key] = $value;
670 $this->emit('FILTER', $key, $value, 'active', count($this->selected()));
671 break;
672 case 'UNFILTER':
673 $key = $a[0];
674 if ($key === 'RANGE') {
675 check($this->range !== null, 'not filtered');
676 $this->range = null;
677 } else {
678 check($this->conditions[$key] !== null, 'not filtered');
679 $this->conditions[$key] = null;
680 }
681 $this->emit('UNFILTER', $key, 'active', count($this->selected()));
682 break;
683 case 'RESET':
684 $this->range = null;
685 foreach (FILTER_KEYS as $key) $this->conditions[$key] = null;
686 $this->emit('RESET', 'active', count($this->selected()));
687 break;
688 case 'STATE':
689 $parts = ['STATE', 'active', $count, 'range'];
690 if ($this->range === null) $parts[] = '-';
691 else array_push($parts, $this->range[0], $this->range[1]);
692 foreach (FILTER_KEYS as $key) {
693 array_push($parts, strtolower($key), $this->conditions[$key] ?? '-');
694 }
695 $this->emit(...$parts);
696 break;
697 }
698 }
