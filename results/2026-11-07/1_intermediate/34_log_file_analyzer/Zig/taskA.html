73fn words(text: []const u8, storage: *[101][]const u8) []const []const u8 {
74 var iter = std.mem.tokenizeScalar(u8, text, ' ');
75 var count: usize = 0;
76 while (iter.next()) |word| {
77 storage[count] = word;
78 count += 1;
79 }
80 return storage[0..count];

83fn digits(text: []const u8) bool {
84 if (text.len == 0) return false;
85 for (text) |c| if (c < '0' or c > '9') return false;
86 return true;
87}

89fn canonical(text: []const u8, maximum: usize) bool {
90 return text.len > 0 and text.len <= maximum and digits(text) and
91 (text.len == 1 or text[0] != '0');
92}

94fn leap(year: i64) bool {
95 return @mod(year, 4) == 0 and (@mod(year, 100) != 0 or @mod(year, 400) == 0);
96}

98fn monthDays(year: i64, month: i64) i64 {
99 const days = [_]i64{ 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31 };
100 return if (month == 2 and leap(year)) 29 else days[@intCast(month - 1)];
101}

103fn validDate(text: []const u8) bool {
104 if (text.len != 10 or text[4] != '-' or text[7] != '-') return false;
105 if (!digits(text[0..4]) or !digits(text[5..7]) or !digits(text[8..10])) return false;
106 const year = number(text[0..4]);
107 const month = number(text[5..7]);
108 const day = number(text[8..10]);
109 return year >= 2000 and year <= 2059 and month >= 1 and month <= 12 and
110 day >= 1 and day <= monthDays(year, month);
111}
112
113fn validClock(text: []const u8) bool {
114 if (text.len != 8 or text[2] != ':' or text[5] != ':') return false;
115 return digits(text[0..2]) and digits(text[3..5]) and digits(text[6..8]) and
116 number(text[0..2]) < 24 and number(text[3..5]) < 60 and number(text[6..8]) < 60;
117}

119fn parseIp(text: []const u8) ?u32 {
120 var iter = std.mem.splitScalar(u8, text, '.');
121 var count: usize = 0;
122 var result: u32 = 0;
123 while (iter.next()) |part| {
124 if (count == 4 or !canonical(part, 3) or number(part) > 255) return null;
125 result = (result << 8) | @as(u32, @intCast(number(part)));
126 count += 1;
127 }
128 return if (count == 4) result else null;
129}

131fn validStatus(text: []const u8) bool {
132 return text.len == 3 and digits(text) and number(text) >= 100 and number(text) <= 599;
133}
134
135fn statusClass(text: []const u8) bool {
136 return text.len == 3 and text[0] >= '1' and text[0] <= '5' and same(text[1..], "xx");
137}
138
139fn statusMatches(pattern: []const u8, status: i64) bool {
140 return if (statusClass(pattern)) @divTrunc(status, 100) == pattern[0] - '0' else status == number(pattern);
141}

143fn filterValue(key: []const u8, value: []const u8) bool {
144 const index = indexOf(key, &filter_keys) orelse return false;
145 return switch (index) {
146 0 => indexOf(value, &methods) != null,
147 1 => validStatus(value) or statusClass(value),
148 2 => parseIp(value) != null,
149 3 => std.mem.startsWith(u8, value, "/"),
150 else => unreachable,
151 };
152}

154fn argumentError(kind: Argument, value: []const u8, args: []const []const u8) ?[]const u8 {
155 switch (kind) {
156 .limit, .minutes, .width, .threshold, .length, .times => {
157 const maximum: i64 = switch (kind) {
158 .limit => 100,
159 .minutes => 1440,
160 .width, .threshold => 100000,
161 .length => 86400,
162 .times => 1000,
163 else => unreachable,
164 };
165 if (!canonical(value, 6) or number(value) < 1 or number(value) > maximum)
166 return "invalid number";
167 },
168 .kind => if (!same(value, "PATHS") and !same(value, "IPS")) return "invalid kind",
169 .path => if (!std.mem.startsWith(u8, value, "/")) return "invalid path",
170 .ip => if (parseIp(value) == null) return "invalid ip",
171 .date => if (!validDate(value)) return "invalid timestamp",
172 .clock => if (!validClock(value)) return "invalid timestamp",
173 .key => if (indexOf(value, &filter_keys) == null) return "invalid key",
174 .subject => if (indexOf(value, &filter_keys) == null and !same(value, "RANGE")) return "invalid key",
175 .value => if (!filterValue(args[0], value)) return "invalid value",
176 }
177 return null;
178}

206fn malformedReason(fields: []const []const u8) ?usize {
207 if (fields.len != 8) return 0;
208 if (!validDate(fields[0])) return 1;
209 if (!validClock(fields[1])) return 2;
210 if (parseIp(fields[2]) == null) return 3;
211 if (indexOf(fields[3], &methods) == null) return 4;
212 if (!std.mem.startsWith(u8, fields[4], "/")) return 5;
213 if (!validStatus(fields[5])) return 6;
214 if (!canonical(fields[6], 8)) return 7;
215 if (!canonical(fields[7], 6)) return 8;
216 return null;
217}

219fn secondsOf(date: []const u8, clock: []const u8) i64 {
220 const year = number(date[0..4]);
221 const month = number(date[5..7]);
222 var days = number(date[8..10]) - 1;
223 var y: i64 = 2000;
224 while (y < year) : (y += 1) days += if (leap(y)) @as(i64, 366) else 365;
225 var m: i64 = 1;
226 while (m < month) : (m += 1) days += monthDays(year, m);
227 return days * 86400 + number(clock[0..2]) * 3600 + number(clock[3..5]) * 60 + number(clock[6..8]);
228}

230fn intervalLabel(allocator: Allocator, seconds: i64) ![]const u8 {
231 var days = @divTrunc(seconds, 86400);
232 const minutes = @divTrunc(@mod(seconds, 86400), 60);
233 var year: i64 = 2000;
234 while (days >= (if (leap(year)) @as(i64, 366) else 365)) : (year += 1)
235 days -= if (leap(year)) @as(i64, 366) else 365;
236 var month: i64 = 1;
237 while (days >= monthDays(year, month)) : (month += 1) days -= monthDays(year, month);
238 return std.fmt.allocPrint(allocator, "{d:0>4}-{d:0>2}-{d:0>2} {d:0>2}:{d:0>2}", .{ year, month, days + 1, @divTrunc(minutes, 60), @mod(minutes, 60) });
239}

241fn normalizePath(allocator: Allocator, raw: []const u8) ![]const u8 {
242 const end = std.mem.indexOfScalar(u8, raw, '?') orelse raw.len;
243 var parts: [101][]const u8 = undefined;
244 var count: usize = 0;
245 var iter = std.mem.splitScalar(u8, raw[0..end], '/');
246 while (iter.next()) |part| {
247 if (same(part, "..")) {
248 if (count > 0) count -= 1;
249 } else if (part.len > 0 and !same(part, ".")) {
250 parts[count] = part;
251 count += 1;
252 }
253 }
254 for (parts[0..count]) |*part| {
255 if (digits(part.*)) {
256 part.* = "{id}";
257 } else if (part.len >= 8) {
258 var hex = true;
259 var has_digit = false;
260 for (part.*) |c| {
261 const digit = c >= '0' and c <= '9';
262 hex = hex and (digit or (c >= 'a' and c <= 'f'));
263 has_digit = has_digit or digit;
264 }
265 if (hex and has_digit) part.* = "{hex}";
266 }
267 }
268 const joined = try std.mem.join(allocator, "/", parts[0..count]);
269 return std.fmt.allocPrint(allocator, "/{s}", .{joined});
270}

272fn decimal(allocator: Allocator, numerator: i64, denominator: i64, comptime places: usize) ![]const u8 {
273 const scale: i64 = switch (places) {
274 1 => 10,
275 2 => 100,
276 3 => 1000,
277 else => @compileError("unsupported decimal width"),
278 };
279 const value = @divTrunc(numerator * scale, denominator);
280 const format = "{d}.{d:0>" ++ std.fmt.comptimePrint("{d}", .{places}) ++ "}";
281 return std.fmt.allocPrint(allocator, format, .{ @divTrunc(value, scale), @mod(value, scale) });
282}

284fn percentage(allocator: Allocator, part: i64, total: i64) ![]const u8 {
285 return std.fmt.allocPrint(allocator, "{s}%", .{try decimal(allocator, part * 100, total, 2)});
286}

288fn bytesText(allocator: Allocator, bytes: i64) ![]const u8 {
289 if (bytes < 1024) return std.fmt.allocPrint(allocator, "{d}B", .{bytes});
290 const unit: i64 = if (bytes >= 1073741824) 1073741824 else if (bytes >= 1048576) 1048576 else 1024;
291 const label: []const u8 = if (unit == 1073741824) "GB" else if (unit == 1048576) "MB" else "KB";
292 return std.fmt.allocPrint(allocator, "{s}{s}", .{ try decimal(allocator, bytes, unit, 1), label });
293}

335 if (kind == .ip) return std.math.order(a.ip_number, b.ip_number);
336 const order = std.mem.order(u8, keyText(a, kind), keyText(b, kind));

346const Stats = struct {
347 count: i64 = 0,
348 bytes: i64 = 0,
349 latency: i64 = 0,
350 maximum: i64 = 0,
351 client: i64 = 0,
352 server: i64 = 0,
353 satisfied: i64 = 0,
354 tolerating: i64 = 0,
355
356 fn of(rows: Rows, threshold: i64) Stats {
357 var result = Stats{ .count = integer(rows.len) };
358 for (rows) |row| {
359 result.bytes += row.bytes;
360 result.latency += row.latency;
361 result.maximum = @max(result.maximum, row.latency);
362 result.client += @intFromBool(@divTrunc(row.status, 100) == 4);
363 result.server += @intFromBool(@divTrunc(row.status, 100) == 5);
364 if (row.status < 500) {
365 result.satisfied += @intFromBool(row.latency <= threshold);
366 result.tolerating += @intFromBool(row.latency > threshold and row.latency <= threshold * 4);
367 }
368 }
369 return result;
370 }
371
372 fn points(self: Stats) i64 {
373 return 2 * self.satisfied + self.tolerating;
374 }
375
376 fn rating(self: Stats) []const u8 {
377 const value = 100 * self.points();
378 const total = 2 * self.count;
379 return if (value >= 94 * total) "excellent" else if (value >= 85 * total) "good" else if (value >= 70 * total) "fair" else if (value >= 50 * total) "poor" else "unacceptable";
380 }
381};

383const Group = struct { rows: Rows, stats: Stats };

385fn grouped(allocator: Allocator, rows: Rows, kind: GroupKey, threshold: i64) ![]Group {
386 const ordered = try allocator.dupe(*const Record, rows);
387 std.mem.sort(*const Record, ordered, kind, rowLess);
388 const groups = try allocator.alloc(Group, rows.len);
389 var first: usize = 0;
390 var count: usize = 0;
391 while (first < ordered.len) {
392 var end = first + 1;
393 while (end < ordered.len and keyOrder(ordered[first], ordered[end], kind) == .eq) : (end += 1) {}
394 const members = ordered[first..end];
395 groups[count] = .{ .rows = members, .stats = Stats.of(members, threshold) };
396 count += 1;
397 first = end;
398 }
399 return groups[0..count];
400}

402fn distinctCount(allocator: Allocator, rows: Rows, kind: GroupKey) !usize {
403 const ordered = try allocator.dupe(*const Record, rows);
404 std.mem.sort(*const Record, ordered, kind, rowLess);
405 var result: usize = 0;
406 for (ordered, 0..) |row, i| {
407 if (i == 0 or keyOrder(ordered[i - 1], row, kind) != .eq) result += 1;
408 }
409 return result;
410}

412const Rank = struct { mode: RankMode, key: GroupKey };
413
414fn groupLess(rank: Rank, a: Group, b: Group) bool {
415 const x = a.stats;
416 const y = b.stats;
417 var difference: i64 = 0;
418 switch (rank.mode) {
419 .count => difference = y.count - x.count,
420 .apdex => {
421 difference = x.points() * y.count - y.points() * x.count;
422 if (difference == 0) difference = y.count - x.count;
423 },
424 .errors => {
425 difference = (y.client + y.server) * x.count - (x.client + x.server) * y.count;
426 if (difference == 0) difference = y.client + y.server - x.client - x.server;
427 },
428 .slowest => {
429 difference = y.latency * x.count - x.latency * y.count;
430 if (difference == 0) difference = y.maximum - x.maximum;
431 },
432 }
433 return if (difference != 0) difference < 0 else keyOrder(a.rows[0], b.rows[0], rank.key) == .lt;
434}

438fn peak(rows: Rows, width: i64) Window {
439 var result = Window{ .count = 0, .first = rows[0], .last = rows[0] };
440 var first: usize = 0;
441 for (rows, 0..) |row, last| {
442 while (row.seconds - rows[first].seconds >= width) first += 1;
443 if (last - first + 1 > result.count)
444 result = .{ .count = last - first + 1, .first = rows[first], .last = row };
445 }
446 return result;
447}

470const Analyzer = struct {
471 records: [max_records]Record = undefined,
472 record_count: usize = 0,
473 malformed: [max_records]BadLine = undefined,
474 malformed_count: usize = 0,
475 line_count: usize = 0,
476 filters: [4][max_path]u8 = undefined,
477 filter_lengths: [4]usize = @splat(0),
478 range: ?DateRange = null,

480 fn ingest(self: *Analyzer, allocator: Allocator, input: []const u8) !void {
481 self.line_count += 1;
482 const line = try allocator.dupe(u8, input);
483 var storage: [101][]const u8 = undefined;
484 const fields = words(line, &storage);
485 if (malformedReason(fields)) |reason| {
486 self.malformed[self.malformed_count] = .{ .number = self.line_count, .reason = reason };
487 self.malformed_count += 1;
488 return;
489 }
490 self.records[self.record_count] = .{
491 .line = self.line_count,
492 .date = fields[0],
493 .stamp = try std.fmt.allocPrint(allocator, "{s} {s}", .{ fields[0], fields[1] }),
494 .ip = fields[2],
495 .method = fields[3],
496 .path = try normalizePath(allocator, fields[4]),
497 .ip_number = parseIp(fields[2]).?,
498 .hour = @intCast(number(fields[1][0..2])),
499 .seconds = secondsOf(fields[0], fields[1]),
500 .status = number(fields[5]),
501 .bytes = number(fields[6]),
502 .latency = number(fields[7]),
503 };
504 self.record_count += 1;
505 }

507 fn filter(self: *const Analyzer, index: usize) []const u8 {
508 return self.filters[index][0..self.filter_lengths[index]];
509 }

511 fn selected(self: *const Analyzer, allocator: Allocator) !Rows {
512 const result = try allocator.alloc(*const Record, self.record_count);
513 var count: usize = 0;
514 for (self.records[0..self.record_count]) |*row| {
515 if (self.range) |range| {
516 if (row.seconds < range.first or row.seconds > range.last) continue;
517 }
518 if (self.filter_lengths[0] > 0 and !same(row.method, self.filter(0))) continue;
519 if (self.filter_lengths[1] > 0 and !statusMatches(self.filter(1), row.status)) continue;
520 if (self.filter_lengths[2] > 0 and !same(row.ip, self.filter(2))) continue;
521 if (self.filter_lengths[3] > 0 and !same(row.path, self.filter(3))) continue;
522 result[count] = row;
523 count += 1;
524 }
525 return result[0..count];
526 }

528 fn execute(self: *Analyzer, allocator: Allocator, line: []const u8, out: *Writer) !void {
529 var storage: [101][]const u8 = undefined;
530 const fields = words(line, &storage);
531 const name = if (fields.len == 0) "" else fields[0];
532 const query = std.meta.stringToEnum(Query, name) orelse {
533 try out.writeAll("ERROR unknown query\n");
534 return;
535 };
536 const args = fields[1..];
537 const form = queryForm(query);
538 const minimum = form.len - @as(usize, @intFromBool(query == .LATENCY or query == .SESSIONS));
539 if (args.len < minimum or args.len > form.len) {
540 try out.writeAll("ERROR wrong arguments\n");
541 return;
542 }
543 for (args, 0..) |arg, i| if (argumentError(form[i], arg, args)) |reason| {
544 try out.print("ERROR {s}\n", .{reason});
545 return;
546 };
547 const rows = try self.selected(allocator);
548 if (rows.len == 0 and requiresRecords(query)) {
549 try out.print("{s} no records\n", .{name});
550 return;
551 }
552 try self.run(allocator, query, args, rows, out);
553 }
554
555 fn run(self: *Analyzer, allocator: Allocator, query: Query, args: []const []const u8, rows: Rows, out: *Writer) !void {
556 const n = integer(rows.len);
557 switch (query) {
558 .SUMMARY => {
559 const total = Stats.of(rows, 0);
560 try out.print("SUMMARY lines {d} valid {d} malformed {d}\n", .{ self.line_count, self.record_count, self.malformed_count });
561 try out.print("requests {d} bytes {d} ips {d} paths {d}\n", .{ n, total.bytes, try distinctCount(allocator, rows, .ip), try distinctCount(allocator, rows, .path) });
562 if (rows.len == 0) try out.writeAll("span none\n") else try out.print("span {s} {s} {d}\n", .{ rows[0].stamp, rows[rows.len - 1].stamp, rows[rows.len - 1].seconds - rows[0].seconds });
563 },
564 .MALFORMED => {
565 try out.print("MALFORMED {d}\n", .{self.malformed_count});
566 for (reasons, 0..) |reason, index| {
567 var count: usize = 0;
568 for (self.malformed[0..self.malformed_count]) |bad| if (bad.reason == index) {
569 count += 1;
570 };
571 if (count == 0) continue;
572 try out.print("{s} {d}", .{ reason, count });
573 for (self.malformed[0..self.malformed_count]) |bad| if (bad.reason == index)
574 try out.print(" {d}", .{bad.number});
575 try out.writeByte('\n');
576 }
577 },
578 .STATUS => {
579 var classes: [5]i64 = @splat(0);
580 var statuses: [600]i64 = @splat(0);
581 for (rows) |row| {
582 classes[@intCast(@divTrunc(row.status, 100) - 1)] += 1;
583 statuses[@intCast(row.status)] += 1;
584 }
585 try out.print("STATUS {d}\n", .{n});
586 for (classes, 1..) |count, category|
587 try out.print("{d}xx {d} {s}\n", .{ category, count, try percentage(allocator, count, n) });
588 for (statuses, 0..) |count, status| if (count > 0)
589 try out.print("{d} {d} {s}\n", .{ status, count, try percentage(allocator, count, n) });
590 },
591 .METHODS, .TOP => {
592 const kind: GroupKey = if (query == .METHODS) .method else if (same(args[0], "IPS")) .ip else .path;
593 const groups = try grouped(allocator, rows, kind, 0);
594 std.mem.sort(Group, groups, Rank{ .mode = .count, .key = kind }, groupLess);
595 const limit = if (query == .METHODS) groups.len else @min(@as(usize, @intCast(number(args[1]))), groups.len);
596 if (query == .METHODS) try out.print("METHODS {d}\n", .{n}) else try out.print("TOP {s} {d} of {d}\n", .{ args[0], limit, groups.len });
597 var rank: usize = 0;
598 for (groups[0..limit], 0..) |group, i| {
599 if (i == 0 or group.stats.count != groups[i - 1].stats.count) rank = i + 1;
600 if (query == .TOP) try out.print("{d} ", .{rank});
601 try out.print("{s} {d} {s}\n", .{ keyText(group.rows[0], kind), group.stats.count, try percentage(allocator, group.stats.count, n) });
602 }
603 },
604 .HOURLY => {
605 var counts: [24]usize = @splat(0);
606 for (rows) |row| counts[row.hour] += 1;
607 const maximum = std.mem.max(usize, &counts);
608 const first = std.mem.indexOfScalar(usize, &counts, maximum).?;
609 try out.print("HOURLY {d} peak {d:0>2} {d}\n", .{ n, first, maximum });
610 for (counts, 0..) |count, hour|
611 try barLine(out, try std.fmt.allocPrint(allocator, "{d:0>2}", .{hour}), count, maximum);
612 },
613 .TIMELINE, .HISTOGRAM => {
614 const parameter = number(args[0]);
615 const width = if (query == .TIMELINE) parameter * 60 else parameter;
616 const values = try allocator.alloc(i64, rows.len);
617 for (rows, 0..) |row, i| values[i] = @divTrunc(if (query == .TIMELINE) row.seconds else row.latency, width);
618 const low = std.mem.min(i64, values);
619 const high = std.mem.max(i64, values);
620 if (high - low + 1 > 60) {
621 try out.writeAll("ERROR too many buckets\n");
622 return;
623 }
624 var buffer: [60]usize = @splat(0);
625 const counts = buffer[0..@as(usize, @intCast(high - low + 1))];
626 for (values) |value| counts[@intCast(value - low)] += 1;
627 const maximum = std.mem.max(usize, counts);
628 try out.print("{s} {d} {d}\n", .{ @tagName(query), parameter, counts.len });
629 for (counts, 0..) |count, i| {
630 const start = (low + integer(i)) * width;
631 const label = if (query == .TIMELINE) try intervalLabel(allocator, start) else try std.fmt.allocPrint(allocator, "{d}-{d}", .{ start, start + width - 1 });
632 try barLine(out, label, count, maximum);
633 }
634 },
635 .BANDWIDTH => {
636 const groups = try grouped(allocator, rows, .date, 0);
637 const total = Stats.of(rows, 0);
638 try out.print("BANDWIDTH {d} {d} {s}\n", .{ groups.len, total.bytes, try bytesText(allocator, total.bytes) });
639 for (groups) |group| try out.print("{s} {d} {d} {s}\n", .{
640 group.rows[0].date, group.stats.count, group.stats.bytes, try bytesText(allocator, group.stats.bytes),
641 });
642 },
643 .VISITORS => {
644 const days = try grouped(allocator, rows, .date, 0);
645 const ips = try grouped(allocator, rows, .ip, 0);
646 try out.print("VISITORS {d} {d}\n", .{ days.len, ips.len });
647 for (days) |day| {
648 const unique = try distinctCount(allocator, day.rows, .ip);
649 var fresh: usize = 0;
650 for (ips) |ip| if (same(ip.rows[0].date, day.rows[0].date)) {
651 fresh += 1;
652 };
653 try out.print("{s} unique {d} new {d} returning {d}\n", .{ day.rows[0].date, unique, fresh, unique - fresh });
654 }
655 },
656 .LATENCY => {
657 const label = if (args.len == 0) "all" else try normalizePath(allocator, args[0]);
658 const buffer = try allocator.alloc(i64, rows.len);
659 var count: usize = 0;
660 for (rows) |row| if (args.len == 0 or same(row.path, label)) {
661 buffer[count] = row.latency;
662 count += 1;
663 };
664 if (count == 0) {
665 try out.writeAll("LATENCY no records\n");
666 return;
667 }
668 const values = buffer[0..count];
669 std.mem.sort(i64, values, {}, latencyLess);
670 const trim = count / 10;
671 var total: i64 = 0;
672 var trimmed: i64 = 0;
673 for (values, 0..) |value, i| {
674 total += value;
675 if (i >= trim and i < count - trim) trimmed += value;
676 }
677 try out.print("LATENCY {s} count {d} min {d} p50 {d} p90 {d} p99 {d} max {d} mean {s} trimmed {s}\n", .{
678 label, count, values[0], values[(50 * count + 99) / 100 - 1], values[(90 * count + 99) / 100 - 1],
679 values[(99 * count + 99) / 100 - 1], values[count - 1], try decimal(allocator, total, integer(count), 2), try decimal(allocator, trimmed, integer(count - 2 * trim), 2),
680 });
681 },
682 .APDEX => {
683 const threshold = number(args[0]);
684 const total = Stats.of(rows, threshold);
685 try out.print("APDEX {d} {s} {s} satisfied {d} tolerating {d} frustrated {d}\n", .{
686 threshold, try decimal(allocator, total.points(), 2 * n, 3), total.rating(),
687 total.satisfied, total.tolerating, n - total.satisfied - total.tolerating,
688 });
689 const groups = try grouped(allocator, rows, .path, threshold);
690 std.mem.sort(Group, groups, Rank{ .mode = .apdex, .key = .path }, groupLess);
691 for (groups) |group| try out.print("{s} {s} {s} {d}\n", .{
692 group.rows[0].path, try decimal(allocator, group.stats.points(), 2 * group.stats.count, 3), group.stats.rating(), group.stats.count,
693 });
694 },
695 .ERRORS => {
696 const total = Stats.of(rows, 0);
697 const failed = total.client + total.server;
698 try out.print("ERRORS {d} of {d} {s}\n", .{ failed, n, try percentage(allocator, failed, n) });
699 const groups = try grouped(allocator, rows, .path, 0);
700 std.mem.sort(Group, groups, Rank{ .mode = .errors, .key = .path }, groupLess);
701 for (groups) |group| {
702 const stats = group.stats;
703 if (stats.client + stats.server == 0) continue;
704 try out.print("{s} {d} {d} {d} {s}\n", .{ group.rows[0].path, stats.count, stats.client, stats.server, try percentage(allocator, stats.client + stats.server, stats.count) });
705 }
706 },
707 .SLOWEST => {
708 const groups = try grouped(allocator, rows, .endpoint, 0);
709 std.mem.sort(Group, groups, Rank{ .mode = .slowest, .key = .endpoint }, groupLess);
710 const limit = @min(@as(usize, @intCast(number(args[0]))), groups.len);
711 try out.print("SLOWEST {d} of {d}\n", .{ limit, groups.len });
712 for (groups[0..limit], 1..) |group, rank|
713 try out.print("{d} {s} {s} count {d} avg {s} max {d}\n", .{ rank, group.rows[0].method, group.rows[0].path, group.stats.count, try decimal(allocator, group.stats.latency, group.stats.count, 2), group.stats.maximum });
714 },
715 .SESSIONS => {
716 const found = try ipRows(allocator, rows, args[0]);
717 if (found.len == 0) {
718 try out.writeAll("SESSIONS no records\n");
719 return;
720 }
721 const minutes = if (args.len == 2) number(args[1]) else 30;
722 const starts = try allocator.alloc(usize, found.len + 1);
723 starts[0] = 0;
724 var count: usize = 1;
725 for (1..found.len) |i| if (found[i].seconds - found[i - 1].seconds > minutes * 60) {
726 starts[count] = i;
727 count += 1;
728 };
729 starts[count] = found.len;
730 var duration: i64 = 0;
731 for (0..count) |i| duration += found[starts[i + 1] - 1].seconds - found[starts[i]].seconds;
732 try out.print("SESSIONS {s} timeout {d} count {d} requests {d} avg_duration {s} avg_requests {s}\n", .{
733 args[0], minutes, count, found.len, try decimal(allocator, duration, integer(count), 1),
734 try decimal(allocator, integer(found.len), integer(count), 1),
735 });
736 for (0..count) |i| {
737 const session = found[starts[i]..starts[i + 1]];
738 const first = session[0];
739 const last = session[session.len - 1];
740 try out.print("{d} {s} {s} {d} {d} {d}\n", .{ i + 1, first.stamp, last.stamp, last.seconds - first.seconds, session.len, try distinctCount(allocator, session, .path) });
741 }
742 },
743 .BURST => {
744 const width = number(args[0]);
745 const threshold = number(args[1]);
746 const failed = try allocator.alloc(*const Record, rows.len);
747 var failure_count: usize = 0;
748 for (rows) |row| if (row.status >= 400) {
749 failed[failure_count] = row;
750 failure_count += 1;
751 };
752 const groups = try grouped(allocator, failed[0..failure_count], .ip, 0);
753 const buffer = try allocator.alloc(Window, groups.len);
754 var count: usize = 0;
755 for (groups) |group| {
756 const window = peak(group.rows, width);
757 if (integer(window.count) >= threshold) {
758 buffer[count] = window;
759 count += 1;
760 }
761 }
762 const matches = buffer[0..count];
763 std.mem.sort(Window, matches, {}, windowLess);
764 try out.print("BURST {d} {d} {d}\n", .{ width, threshold, count });
765 for (matches) |window| try out.print("{s} {d} {s} {s}\n", .{ window.first.ip, window.count, window.first.stamp, window.last.stamp });
766 },
767 .PEAK => {
768 const width = number(args[0]);
769 const window = peak(rows, width);
770 try out.print("PEAK {d} {d} {s} {s}\n", .{ width, window.count, window.first.stamp, window.last.stamp });
771 },
772 .GAPS => {
773 const width = number(args[0]);
774 var count: usize = 0;
775 for (1..rows.len) |i| if (rows[i].seconds - rows[i - 1].seconds >= width) {
776 count += 1;
777 };
778 try out.print("GAPS {d} {d}\n", .{ width, count });
779 for (1..rows.len) |i| if (rows[i].seconds - rows[i - 1].seconds >= width)
780 try out.print("{s} {s} {d}\n", .{ rows[i - 1].stamp, rows[i].stamp, rows[i].seconds - rows[i - 1].seconds });
781 },
782 .IP => {
783 const found = try ipRows(allocator, rows, args[0]);
784 if (found.len == 0) {
785 try out.writeAll("IP no records\n");
786 return;
787 }
788 const total = Stats.of(found, 0);
789 try out.print("IP {s} requests {d} bytes {d} errors {d} paths {d}\n", .{
790 args[0], found.len, total.bytes, total.client + total.server, try distinctCount(allocator, found, .path),
791 });
792 try out.print("first {s} last {s}\nmethods", .{ found[0].stamp, found[found.len - 1].stamp });
793 const method_groups = try grouped(allocator, found, .method, 0);
794 std.mem.sort(Group, method_groups, Rank{ .mode = .count, .key = .method }, groupLess);
795 for (method_groups) |group| try out.print(" {s} {d}", .{ group.rows[0].method, group.stats.count });
796 try out.writeAll("\nclasses");
797 var classes: [5]usize = @splat(0);
798 for (found) |row| classes[@intCast(@divTrunc(row.status, 100) - 1)] += 1;
799 for (classes, 1..) |count, category| try out.print(" {d}xx {d}", .{ category, count });
800 try out.writeByte('\n');
801 const path_groups = try grouped(allocator, found, .path, 0);
802 std.mem.sort(Group, path_groups, Rank{ .mode = .count, .key = .path }, groupLess);
803 for (path_groups[0..@min(3, path_groups.len)]) |group|
804 try out.print("path {s} {d}\n", .{ group.rows[0].path, group.stats.count });
805 },
806 .RANGE => {
807 const first = secondsOf(args[0], args[1]);
808 const last = secondsOf(args[2], args[3]);
809 if (first > last) {
810 try out.writeAll("ERROR invalid range\n");
811 return;
812 }
813 var range = DateRange{ .first = first, .last = last, .first_label = undefined, .last_label = undefined };
814 _ = try std.fmt.bufPrint(&range.first_label, "{s} {s}", .{ args[0], args[1] });
815 _ = try std.fmt.bufPrint(&range.last_label, "{s} {s}", .{ args[2], args[3] });
816 self.range = range;
817 try out.print("RANGE {s} {s} active {d}\n", .{ range.first_label, range.last_label, (try self.selected(allocator)).len });
818 },
819 .FILTER => {
820 const key = indexOf(args[0], &filter_keys).?;
821 const value = if (key == 3) try normalizePath(allocator, args[1]) else args[1];
822 std.mem.copyForwards(u8, self.filters[key][0..value.len], value);
823 self.filter_lengths[key] = value.len;
824 try out.print("FILTER {s} {s} active {d}\n", .{ args[0], value, (try self.selected(allocator)).len });
825 },
826 .UNFILTER => {
827 if (same(args[0], "RANGE")) {
828 if (self.range == null) {
829 try out.writeAll("ERROR not filtered\n");
830 return;
831 }
832 self.range = null;
833 } else {
834 const key = indexOf(args[0], &filter_keys).?;
835 if (self.filter_lengths[key] == 0) {
836 try out.writeAll("ERROR not filtered\n");
837 return;
838 }
839 self.filter_lengths[key] = 0;
840 }
841 try out.print("UNFILTER {s} active {d}\n", .{ args[0], (try self.selected(allocator)).len });
842 },
843 .RESET => {
844 self.range = null;
845 self.filter_lengths = @splat(0);
846 try out.print("RESET active {d}\n", .{(try self.selected(allocator)).len});
847 },
848 .STATE => {
849 try out.print("STATE active {d} range ", .{n});
850 if (self.range) |range| try out.print("{s} {s}", .{ range.first_label, range.last_label }) else try out.writeByte('-');
851 for (lower_keys, 0..) |label, key|
852 try out.print(" {s} {s}", .{ label, if (self.filter_lengths[key] == 0) "-" else self.filter(key) });
853 try out.writeByte('\n');
854 },
855 }
856 }

859pub fn main(init: std.process.Init) !void {
860 var persistent = std.heap.ArenaAllocator.init(std.heap.page_allocator);
861 defer persistent.deinit();
862 var input_buffer: [4096]u8 = undefined;
863 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
864 var output_buffer: [8192]u8 = undefined;
865 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
866 const out = &output.interface;
867 var analyzer = Analyzer{};
868 var queries = false;
869 while (try input.interface.takeDelimiter('\n')) |line| {
870 if (!queries) {
871 if (same(line, "---")) {
872 queries = true;
873 std.mem.sort(Record, analyzer.records[0..analyzer.record_count], {}, chronological);
874 } else try analyzer.ingest(persistent.allocator(), line);
875 } else {
876 if (same(line, "END")) break;
877 var scratch = std.heap.ArenaAllocator.init(std.heap.page_allocator);
878 defer scratch.deinit();
879 try analyzer.execute(scratch.allocator(), line, out);
880 }
881 }
882 try out.flush();
883}
