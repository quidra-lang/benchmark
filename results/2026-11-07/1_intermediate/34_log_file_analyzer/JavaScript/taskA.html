3const METHODS = ['GET', 'HEAD', 'POST', 'PUT', 'DELETE', 'PATCH', 'OPTIONS'];
4const REASONS = ['fields', 'date', 'time', 'ip', 'method', 'path', 'status', 'bytes', 'latency'];
5const STATUS_CLASSES = ['1xx', '2xx', '3xx', '4xx', '5xx'];
6const FILTER_KEYS = ['METHOD', 'STATUS', 'IP', 'PATH'];
7const MONTH_DAYS = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
8const FORMS = {
9 SUMMARY: [], MALFORMED: [], STATUS: [], METHODS: [], TOP: ['kind', 'limit'],
10 HOURLY: [], TIMELINE: ['minutes'], BANDWIDTH: [], VISITORS: [],
11 LATENCY: ['path'], HISTOGRAM: ['width'], APDEX: ['threshold'], ERRORS: [],
12 SLOWEST: ['limit'], SESSIONS: ['ip', 'minutes'], BURST: ['length', 'times'],
13 PEAK: ['length'], GAPS: ['length'], IP: ['ip'],
14 RANGE: ['date', 'time', 'date', 'time'], FILTER: ['key', 'value'],
15 UNFILTER: ['subject'], RESET: [], STATE: []
16};
17const OPTIONAL = new Set(['LATENCY', 'SESSIONS']);
18const REQUIRE_RECORDS = new Set([
19 'STATUS', 'METHODS', 'TOP', 'HOURLY', 'TIMELINE', 'BANDWIDTH',
20 'VISITORS', 'HISTOGRAM', 'APDEX', 'ERRORS', 'SLOWEST', 'BURST', 'PEAK', 'GAPS'
21]);
22const RANGES = {
23 limit: [1, 100], minutes: [1, 1440], width: [1, 100000],
24 threshold: [1, 100000], length: [1, 86400], times: [1, 1000]
25};
26const ERRORS = {
27 limit: 'invalid number', minutes: 'invalid number', width: 'invalid number',
28 threshold: 'invalid number', length: 'invalid number', times: 'invalid number',
29 kind: 'invalid kind', path: 'invalid path', ip: 'invalid ip',
30 date: 'invalid timestamp', time: 'invalid timestamp',
31 key: 'invalid key', subject: 'invalid key', value: 'invalid value'
32};
33const DAY_SECONDS = 86400;
34const MAX_BUCKETS = 60;

36class QueryError extends Error {}

38function check(condition, reason) {
39 if (!condition) throw new QueryError(reason);
40}
41function digits(text) {
42 return /^[0-9]+$/.test(text);
43}
44function canonical(text, maxDigits) {
45 return typeof text === 'string' && digits(text) && text.length <= maxDigits &&
46 (text.length === 1 || text[0] !== '0');
47}
48function leapYear(year) {
49 return year % 4 === 0 && (year % 100 !== 0 || year % 400 === 0);
50}
51function daysInYear(year) {
52 return leapYear(year) ? 366 : 365;
53}
54function daysInMonth(year, month) {
55 return month === 2 && leapYear(year) ? 29 : MONTH_DAYS[month - 1];
56}
57function validDate(text) {
58 if (!/^\d{4}-\d{2}-\d{2}$/.test(text)) return false;
59 const [year, month, day] = text.split('-').map(Number);
60 return year >= 2000 && year <= 2059 && month >= 1 && month <= 12 &&
61 day >= 1 && day <= daysInMonth(year, month);
62}
63function validTime(text) {
64 if (!/^\d{2}:\d{2}:\d{2}$/.test(text)) return false;
65 const [hour, minute, second] = text.split(':').map(Number);
66 return hour < 24 && minute < 60 && second < 60;
67}
68function validIp(text) {
69 const parts = text.split('.');
70 return parts.length === 4 && parts.every(part => canonical(part, 3) && Number(part) <= 255);
71}
72function validStatus(text) {
73 return /^\d{3}$/.test(text) && Number(text) >= 100 && Number(text) <= 599;
74}
75function validFilterValue(kind, value) {
76 if (kind === 'METHOD') return METHODS.includes(value);
77 if (kind === 'STATUS') return STATUS_CLASSES.includes(value) || validStatus(value);
78 if (kind === 'IP') return validIp(value);
79 return value.startsWith('/');
80}
81function validArgument(type, text, args) {
82 if (Object.hasOwn(RANGES, type)) {
83 const [low, high] = RANGES[type];
84 return canonical(text, 6) && Number(text) >= low && Number(text) <= high;
85 }
86 switch (type) {
87 case 'kind': return ['PATHS', 'IPS'].includes(text);
88 case 'path': return text.startsWith('/');
89 case 'ip': return validIp(text);
90 case 'date': return validDate(text);
91 case 'time': return validTime(text);
92 case 'key': return FILTER_KEYS.includes(text);
93 case 'subject': return text === 'RANGE' || FILTER_KEYS.includes(text);
94 default: return validFilterValue(args[0], text);
95 }
96}
97function malformedReason(fields) {
98 if (fields.length !== 8) return 'fields';
99 const [date, clock, ip, method, path, status, bytes, latency] = fields;
100 if (!validDate(date)) return 'date';
101 if (!validTime(clock)) return 'time';
102 if (!validIp(ip)) return 'ip';
103 if (!METHODS.includes(method)) return 'method';
104 if (!path.startsWith('/')) return 'path';
105 if (!validStatus(status)) return 'status';
106 if (!canonical(bytes, 8)) return 'bytes';
107 if (!canonical(latency, 6)) return 'latency';
108 return null;
109}
110function dayNumber(year, month, day) {
111 let days = day - 1;
112 for (let y = 2000; y < year; ++y) days += daysInYear(y);
113 for (let m = 1; m < month; ++m) days += daysInMonth(year, m);
114 return days;
115}
116function secondsOf(date, clock) {
117 const [year, month, day] = date.split('-').map(Number);
118 const [hour, minute, second] = clock.split(':').map(Number);
119 return dayNumber(year, month, day) * DAY_SECONDS + hour * 3600 + minute * 60 + second;
120}
121function intervalLabel(seconds) {
122 let days = Math.floor(seconds / DAY_SECONDS);
123 let rest = seconds % DAY_SECONDS;
124 let year = 2000;
125 while (days >= daysInYear(year)) days -= daysInYear(year++);
126 let month = 1;
127 while (days >= daysInMonth(year, month)) days -= daysInMonth(year, month++);
128 return year + '-' + String(month).padStart(2, '0') + '-' + String(days + 1).padStart(2, '0') +
129 ' ' + String(Math.floor(rest / 3600)).padStart(2, '0') + ':' +
130 String(Math.floor(rest % 3600 / 60)).padStart(2, '0');
131}
132function normalizePath(raw) {
133 const parts = [];
134 for (const item of raw.split('?')[0].split('/')) {
135 if (item === '..') parts.pop();
136 else if (item !== '' && item !== '.') parts.push(item);
137 }
138 return '/' + parts.map(part => {
139 if (digits(part)) return '{id}';
140 if (part.length >= 8 && /^[0-9a-f]+$/.test(part) && /[0-9]/.test(part)) return '{hex}';
141 return part;
142 }).join('/');
143}
144function compareIp(a, b) {
145 const left = a.split('.').map(Number), right = b.split('.').map(Number);
146 for (let i = 0; i < 4; ++i) if (left[i] !== right[i]) return left[i] - right[i];
147 return 0;
148}
149function statusMatches(filter, status) {
150 return STATUS_CLASSES.includes(filter) ? Math.floor(status / 100) === Number(filter[0]) : status === Number(filter);
151}
152function decimal(numerator, denominator, places) {
153 const scale = 10 ** places;
154 const value = Math.floor(numerator * scale / denominator);
155 return Math.floor(value / scale) + '.' + String(value % scale).padStart(places, '0');
156}
157function percent(numerator, denominator) {
158 return decimal(numerator * 100, denominator, 2) + '%';
159}
160function bar(count, maximum) {
161 return '#'.repeat(Math.floor((count * 30 + maximum - 1) / maximum));
162}
163function bytesText(size) {
164 for (const [unit, suffix] of [[1073741824, 'GB'], [1048576, 'MB'], [1024, 'KB']]) {
165 if (size >= unit) return decimal(size, unit, 1) + suffix;
166 }
167 return size + 'B';
168}
169function sum(records, key) {
170 return records.reduce((total, record) => total + record[key], 0);
171}
172function grouped(records, key) {
173 const groups = new Map();
174 for (const row of records) {
175 const k = typeof key === 'function' ? key(row) : row[key];
176 if (!groups.has(k)) groups.set(k, []);
177 groups.get(k).push(row);
178 }
179 return groups;
180}
181function tallied(values) {
182 const counts = new Map();
183 for (const value of values) counts.set(value, (counts.get(value) ?? 0) + 1);
184 return counts;
185}
186function sortedByCount(counts, comparator = (a, b) => a < b ? -1 : (a > b ? 1 : 0)) {
187 return [...counts.keys()].sort((a, b) => counts.get(b) - counts.get(a) || comparator(a, b));
188}
189function statusClassCounts(records) {
190 const counts = [0, 0, 0, 0, 0];
191 for (const record of records) ++counts[Math.floor(record.status / 100) - 1];
192 return counts;
193}
194function bucketCounts(numbers) {
195 const first = Math.min(...numbers), last = Math.max(...numbers);
196 check(last - first + 1 <= MAX_BUCKETS, 'too many buckets');
197 const counts = Array(last - first + 1).fill(0);
198 for (const n of numbers) ++counts[n - first];
199 return [first, counts];
200}
201function percentile(sorted, rank) {
202 return sorted[Math.floor((rank * sorted.length + 99) / 100) - 1];
203}
204function windowPeak(records, length) {
205 let size = 0, start = null, end = null;
206 let first = 0;
207 for (let last = 0; last < records.length; ++last) {
208 while (records[last].seconds - records[first].seconds >= length) ++first;
209 if (last - first + 1 > size) {
210 size = last - first + 1;
211 start = records[first];
212 end = records[last];
213 }
214 }
215 return {size, start, end};
216}
217function apdexScore(records, threshold) {
218 const satisfied = records.filter(r => r.status < 500 && r.latency <= threshold).length;
219 const tolerating = records.filter(r => r.status < 500 && r.latency > threshold && r.latency <= 4 * threshold).length;
220 const frustrated = records.length - satisfied - tolerating;
221 const points = 2 * satisfied + tolerating, total = 2 * records.length;
222 const rating = points * 100 >= 94 * total ? 'excellent' :
223 points * 100 >= 85 * total ? 'good' :
224 points * 100 >= 70 * total ? 'fair' :
225 points * 100 >= 50 * total ? 'poor' : 'unacceptable';
226 return {count: records.length, satisfied, tolerating, frustrated, points, total,
227 notation: decimal(points, total, 3), rating};
228}

230class LogAnalyzer {
231 constructor(lines) {
232 this.records = [];
233 this.malformed = [];
234 this.lineCount = lines.length;
235 for (let i = 0; i < lines.length; ++i) {
236 const fields = lines[i].trim().split(/\s+/);
237 const reason = malformedReason(fields);
238 if (reason !== null) {
239 this.malformed.push({line: i + 1, reason});
240 continue;
241 }
242 const [date, clock, ip, method, rawPath, status, bytes, latency] = fields;
243 this.records.push({line: i + 1, date, hour: Number(clock.slice(0, 2)),
244 timestamp: date + ' ' + clock, seconds: secondsOf(date, clock), ip,
245 method, path: normalizePath(rawPath), status: Number(status),
246 size: Number(bytes), latency: Number(latency)});
247 }
248 this.records.sort((a, b) => a.seconds - b.seconds || a.line - b.line);
249 this.conditions = {METHOD: null, STATUS: null, IP: null, PATH: null};
250 this.range = null;
251 this.output = [];
252 }
253 emit(...parts) {
254 this.output.push(parts.join(' '));
255 }
256 selected() {
257 return this.records.filter(r => {
258 if (this.range && (r.seconds < this.range.startSeconds || r.seconds > this.range.endSeconds)) return false;
259 if (this.conditions.METHOD !== null && r.method !== this.conditions.METHOD) return false;
260 if (this.conditions.STATUS !== null && !statusMatches(this.conditions.STATUS, r.status)) return false;
261 if (this.conditions.IP !== null && r.ip !== this.conditions.IP) return false;
262 return this.conditions.PATH === null || r.path === this.conditions.PATH;
263 });
264 }
265 barLine(label, count, maximum) {
266 this.emit(label, count, ...(count ? [bar(count, maximum)] : []));
267 }
268 execute(line) {
269 const [name, ...args] = line.trim().split(/\s+/);
270 try {
271 check(Object.hasOwn(FORMS, name), 'unknown query');
272 const form = FORMS[name];
273 const optional = OPTIONAL.has(name) && args.length === form.length - 1;
274 check(args.length === form.length || optional, 'wrong arguments');
275 for (let i = 0; i < args.length; ++i) {
276 check(validArgument(form[i], args[i], args), ERRORS[form[i]]);
277 }
278 const records = this.selected();
279 if (REQUIRE_RECORDS.has(name) && records.length === 0) {
280 this.emit(name, 'no records');
281 return;
282 }
283 this.run(name, args, records);
284 } catch (error) {
285 if (!(error instanceof QueryError)) throw error;
286 this.emit('ERROR', error.message);
287 }
288 }
289 run(name, a, records) {
290 switch (name) {
291 case 'SUMMARY': {
292 this.emit('SUMMARY', 'lines', this.lineCount, 'valid', this.records.length,
293 'malformed', this.malformed.length);
294 this.emit('requests', records.length, 'bytes', sum(records, 'size'),
295 'ips', new Set(records.map(r => r.ip)).size,
296 'paths', new Set(records.map(r => r.path)).size);
297 if (!records.length) this.emit('span', 'none');
298 else this.emit('span', records[0].timestamp, records.at(-1).timestamp,
299 records.at(-1).seconds - records[0].seconds);
300 break;
301 }
302 case 'MALFORMED': {
303 this.emit('MALFORMED', this.malformed.length);
304 for (const reason of REASONS) {
305 const lines = this.malformed.filter(x => x.reason === reason).map(x => x.line);
306 if (lines.length) this.emit(reason, lines.length, ...lines);
307 }
308 break;
309 }
310 case 'STATUS': {
311 this.emit('STATUS', records.length);
312 const classes = statusClassCounts(records);
313 STATUS_CLASSES.forEach((c, i) => this.emit(c, classes[i], percent(classes[i], records.length)));
314 const counts = tallied(records.map(r => r.status));
315 for (const s of [...counts.keys()].sort((x, y) => x - y))
316 this.emit(s, counts.get(s), percent(counts.get(s), records.length));
317 break;
318 }
319 case 'METHODS': {
320 const counts = tallied(records.map(r => r.method));
321 this.emit('METHODS', records.length);
322 for (const method of sortedByCount(counts))
323 this.emit(method, counts.get(method), percent(counts.get(method), records.length));
324 break;
325 }
326 case 'TOP': {
327 const [kind, token] = a;
328 const counts = tallied(records.map(r => kind === 'PATHS' ? r.path : r.ip));
329 const keys = sortedByCount(counts, kind === 'IPS' ? compareIp : undefined);
330 const shown = Math.min(Number(token), keys.length);
331 this.emit('TOP', kind, shown, 'of', keys.length);
332 let rank = 0;
333 keys.slice(0, shown).forEach((key, i) => {
334 if (i === 0 || counts.get(key) !== counts.get(keys[i - 1])) rank = i + 1;
335 this.emit(rank, key, counts.get(key), percent(counts.get(key), records.length));
336 });
337 break;
338 }
339 case 'HOURLY': {
340 const counts = Array(24).fill(0);
341 for (const r of records) ++counts[r.hour];
342 const maximum = Math.max(...counts);
343 this.emit('HOURLY', records.length, 'peak', String(counts.indexOf(maximum)).padStart(2, '0'), maximum);
344 counts.forEach((count, h) => this.barLine(String(h).padStart(2, '0'), count, maximum));
345 break;
346 }
347 case 'TIMELINE': {
348 const minutes = Number(a[0]), width = minutes * 60;
349 const [first, counts] = bucketCounts(records.map(r => Math.floor(r.seconds / width)));
350 this.emit('TIMELINE', minutes, counts.length);
351 const maximum = Math.max(...counts);
352 counts.forEach((count, i) => this.barLine(intervalLabel((first + i) * width), count, maximum));
353 break;
354 }
355 case 'BANDWIDTH': {
356 const days = grouped(records, 'date'), bytes = sum(records, 'size');
357 this.emit('BANDWIDTH', days.size, bytes, bytesText(bytes));
358 for (const date of [...days.keys()].sort()) {
359 const volume = sum(days.get(date), 'size');
360 this.emit(date, days.get(date).length, volume, bytesText(volume));
361 }
362 break;
363 }
364 case 'VISITORS': {
365 const first = new Map();
366 for (const r of records) if (!first.has(r.ip) || r.date < first.get(r.ip)) first.set(r.ip, r.date);
367 const days = grouped(records, 'date');
368 this.emit('VISITORS', days.size, first.size);
369 for (const date of [...days.keys()].sort()) {
370 const ips = new Set(days.get(date).map(r => r.ip));
371 let newcomers = 0;
372 for (const ip of ips) if (first.get(ip) === date) ++newcomers;
373 this.emit(date, 'unique', ips.size, 'new', newcomers, 'returning', ips.size - newcomers);
374 }
375 break;
376 }
377 case 'LATENCY': {
378 const label = a.length ? normalizePath(a[0]) : 'all';
379 const rows = a.length ? records.filter(r => r.path === label) : records;
380 if (!rows.length) { this.emit('LATENCY', 'no records'); break; }
381 const values = rows.map(r => r.latency).sort((x, y) => x - y);
382 const n = values.length, trim = Math.floor(n / 10);
383 const trimmed = values.slice(trim, n - trim);
384 this.emit('LATENCY', label, 'count', n, 'min', values[0],
385 'p50', percentile(values, 50), 'p90', percentile(values, 90),
386 'p99', percentile(values, 99), 'max', values.at(-1),
387 'mean', decimal(values.reduce((v, x) => v + x, 0), n, 2),
388 'trimmed', decimal(trimmed.reduce((v, x) => v + x, 0), trimmed.length, 2));
389 break;
390 }
391 case 'HISTOGRAM': {
392 const width = Number(a[0]);
393 const [first, counts] = bucketCounts(records.map(r => Math.floor(r.latency / width)));
394 this.emit('HISTOGRAM', width, counts.length);
395 const maximum = Math.max(...counts);
396 counts.forEach((count, i) => {
397 const lo = (first + i) * width;
398 this.barLine(lo + '-' + (lo + width - 1), count, maximum);
399 });
400 break;
401 }
402 case 'APDEX': {
403 const threshold = Number(a[0]), overall = apdexScore(records, threshold);
404 this.emit('APDEX', threshold, overall.notation, overall.rating,
405 'satisfied', overall.satisfied, 'tolerating', overall.tolerating,
406 'frustrated', overall.frustrated);
407 const paths = [...grouped(records, 'path')].map(([path, rows]) => ({path, ...apdexScore(rows, threshold)}));
408 paths.sort((x, y) => x.points * y.total - y.points * x.total || y.count - x.count ||
409 (x.path < y.path ? -1 : x.path > y.path ? 1 : 0));
410 for (const row of paths) this.emit(row.path, row.notation, row.rating, row.count);
411 break;
412 }
413 case 'ERRORS': {
414 const failed = records.filter(r => r.status >= 400);
415 this.emit('ERRORS', failed.length, 'of', records.length, percent(failed.length, records.length));
416 const paths = [...grouped(records, 'path')].map(([path, rows]) => ({
417 path, count: rows.length,
418 client: rows.filter(r => Math.floor(r.status / 100) === 4).length,
419 server: rows.filter(r => Math.floor(r.status / 100) === 5).length
420 })).filter(row => row.client + row.server > 0);
421 paths.sort((x, y) => {
422 const xf = x.client + x.server, yf = y.client + y.server;
423 return yf * x.count - xf * y.count || yf - xf || (x.path < y.path ? -1 : 1);
424 });
425 for (const row of paths) this.emit(row.path, row.count, row.client, row.server,
426 percent(row.client + row.server, row.count));
427 break;
428 }
429 case 'SLOWEST': {
430 const groups = grouped(records, r => r.method + '\u0001' + r.path);
431 const rows = [...groups].map(([key, items]) => {
432 const [method, path] = key.split('\u0001');
433 return {method, path, count: items.length,
434 total: sum(items, 'latency'), maximum: Math.max(...items.map(r => r.latency))};
435 });
436 rows.sort((x, y) => y.total * x.count - x.total * y.count || y.maximum - x.maximum ||
437 (x.path < y.path ? -1 : x.path > y.path ? 1 : (x.method < y.method ? -1 : 1)));
438 const shown = Math.min(Number(a[0]), rows.length);
439 this.emit('SLOWEST', shown, 'of', rows.length);
440 rows.slice(0, shown).forEach((row, i) =>
441 this.emit(i + 1, row.method, row.path, 'count', row.count,
442 'avg', decimal(row.total, row.count, 2), 'max', row.maximum));
443 break;
444 }
445 case 'SESSIONS': {
446 const ip = a[0], minutes = a.length > 1 ? Number(a[1]) : 30;
447 const found = records.filter(r => r.ip === ip);
448 if (!found.length) { this.emit('SESSIONS', 'no records'); break; }
449 const sessions = [[found[0]]];
450 for (let i = 1; i < found.length; ++i) {
451 const current = found[i], prev = found[i - 1];
452 if (current.seconds - prev.seconds <= minutes * 60) sessions.at(-1).push(current);
453 else sessions.push([current]);
454 }
455 const duration = sessions.reduce((v, session) => v + session.at(-1).seconds - session[0].seconds, 0);
456 this.emit('SESSIONS', ip, 'timeout', minutes, 'count', sessions.length,
457 'requests', found.length, 'avg_duration', decimal(duration, sessions.length, 1),
458 'avg_requests', decimal(found.length, sessions.length, 1));
459 sessions.forEach((session, i) => this.emit(i + 1, session[0].timestamp,
460 session.at(-1).timestamp, session.at(-1).seconds - session[0].seconds,
461 session.length, new Set(session.map(r => r.path)).size));
462 break;
463 }
464 case 'BURST': {
465 const [length, times] = a.map(Number);
466 const ips = grouped(records.filter(r => r.status >= 400), 'ip');
467 const rows = [...ips].map(([ip, items]) => ({ip, ...windowPeak(items, length)}))
468 .filter(x => x.size >= times);
469 rows.sort((x, y) => y.size - x.size || compareIp(x.ip, y.ip));
470 this.emit('BURST', length, times, rows.length);
471 for (const row of rows) this.emit(row.ip, row.size, row.start.timestamp, row.end.timestamp);
472 break;
473 }
474 case 'PEAK': {
475 const window = windowPeak(records, Number(a[0]));
476 this.emit('PEAK', Number(a[0]), window.size, window.start.timestamp, window.end.timestamp);
477 break;
478 }
479 case 'GAPS': {
480 const length = Number(a[0]);
481 const gaps = [];
482 for (let i = 1; i < records.length; ++i) {
483 if (records[i].seconds - records[i - 1].seconds >= length)
484 gaps.push([records[i - 1], records[i]]);
485 }
486 this.emit('GAPS', length, gaps.length);
487 for (const [prev, curr] of gaps)
488 this.emit(prev.timestamp, curr.timestamp, curr.seconds - prev.seconds);
489 break;
490 }
491 case 'IP': {
492 const ip = a[0], found = records.filter(r => r.ip === ip);
493 if (!found.length) { this.emit('IP', 'no records'); break; }
494 const paths = tallied(found.map(r => r.path));
495 this.emit('IP', ip, 'requests', found.length, 'bytes', sum(found, 'size'),
496 'errors', found.filter(r => r.status >= 400).length, 'paths', paths.size);
497 this.emit('first', found[0].timestamp, 'last', found.at(-1).timestamp);
498 const methods = tallied(found.map(r => r.method));
499 this.emit('methods', ...sortedByCount(methods).flatMap(method => [method, methods.get(method)]));
500 this.emit('classes', ...STATUS_CLASSES.flatMap((label, i) => [label, statusClassCounts(found)[i]]));
501 for (const path of sortedByCount(paths).slice(0, 3)) this.emit('path', path, paths.get(path));
502 break;
503 }
504 case 'RANGE': {
505 const start = a[0] + ' ' + a[1], end = a[2] + ' ' + a[3];
506 const startSeconds = secondsOf(a[0], a[1]), endSeconds = secondsOf(a[2], a[3]);
507 check(startSeconds <= endSeconds, 'invalid range');
508 this.range = {start, end, startSeconds, endSeconds};
509 this.emit('RANGE', start, end, 'active', this.selected().length);
510 break;
511 }
512 case 'FILTER': {
513 const [key, raw] = a;
514 const value = key === 'PATH' ? normalizePath(raw) : raw;
515 this.conditions[key] = value;
516 this.emit('FILTER', key, value, 'active', this.selected().length);
517 break;
518 }
519 case 'UNFILTER': {
520 const key = a[0];
521 if (key === 'RANGE') {
522 check(this.range !== null, 'not filtered');
523 this.range = null;
524 } else {
525 check(this.conditions[key] !== null, 'not filtered');
526 this.conditions[key] = null;
527 }
528 this.emit('UNFILTER', key, 'active', this.selected().length);
529 break;
530 }
531 case 'RESET': {
532 this.range = null;
533 for (const key of FILTER_KEYS) this.conditions[key] = null;
534 this.emit('RESET', 'active', this.selected().length);
535 break;
536 }
537 case 'STATE': {
538 const range = this.range === null ? '-' : this.range.start + ' ' + this.range.end;
539 this.emit('STATE', 'active', records.length, 'range', range,
540 ...FILTER_KEYS.flatMap(key => [key.toLowerCase(), this.conditions[key] ?? '-']));
541 break;
542 }
543 }
544 }
545}

548 const lines = readFileSync(0, 'utf8').split(/\r?\n/);
549 const separator = lines.indexOf('---');
550 const end = lines.indexOf('END', separator + 1);
551 const analyzer = new LogAnalyzer(lines.slice(0, separator));
552 for (let i = separator + 1; i < end; ++i) analyzer.execute(lines[i]);
553 if (analyzer.output.length) process.stdout.write(analyzer.output.join('\n') + '\n');
554}
