53func includes(a []string, s string) bool {
54 for _, x := range a {
55 if x == s {
56 return true
57 }
58 }
59 return false
60}
61func parse(s string) int { v, _ := strconv.Atoi(s); return v }
62func canonical(s string, maxDigits int) bool {
63 return len(s) > 0 && len(s) <= maxDigits && digitsPattern.MatchString(s) &&
64 (len(s) == 1 || s[0] != '0')
65}
66func leapYear(y int) bool { return y%4 == 0 && (y%100 != 0 || y%400 == 0) }
67func daysInYear(y int) int {
68 if leapYear(y) {
69 return 366
70 }
71 return 365
72}
73func daysInMonth(y, m int) int {
74 if m == 2 && leapYear(y) {
75 return 29
76 }
77 return monthDays[m-1]
78}
79func validDate(s string) bool {
80 if !datePattern.MatchString(s) {
81 return false
82 }
83 y, m, d := parse(s[:4]), parse(s[5:7]), parse(s[8:10])
84 return y >= 2000 && y <= 2059 && m >= 1 && m <= 12 &&
85 d >= 1 && d <= daysInMonth(y, m)
86}
87func validTime(s string) bool {
88 if !timePattern.MatchString(s) {
89 return false
90 }
91 return parse(s[:2]) < 24 && parse(s[3:5]) < 60 && parse(s[6:8]) < 60
92}
93func validIP(s string) bool {
94 parts := strings.Split(s, ".")
95 if len(parts) != 4 {
96 return false
97 }
98 for _, p := range parts {
99 if !canonical(p, 3) || parse(p) > 255 {
100 return false
101 }
102 }
103 return true
104}
105func validStatus(s string) bool {
106 return statusPattern.MatchString(s) && parse(s) >= 100 && parse(s) <= 599
107}
108func validFilterValue(kind, value string) bool {
109 switch kind {
110 case "METHOD":
111 return includes(methods, value)
112 case "STATUS":
113 return includes(statusClasses, value) || validStatus(value)
114 case "IP":
115 return validIP(value)
116 default:
117 return strings.HasPrefix(value, "/")
118 }
119}
120func validArgument(kind, value string, args []string) bool {
121 if pair, ok := argumentRanges[kind]; ok {
122 return canonical(value, 6) && parse(value) >= pair[0] && parse(value) <= pair[1]
123 }
124 switch kind {
125 case "kind":
126 return value == "PATHS" || value == "IPS"
127 case "path":
128 return strings.HasPrefix(value, "/")
129 case "ip":
130 return validIP(value)
131 case "date":
132 return validDate(value)
133 case "time":
134 return validTime(value)
135 case "key":
136 return includes(filterKeys, value)
137 case "subject":
138 return value == "RANGE" || includes(filterKeys, value)
139 default:
140 return validFilterValue(args[0], value)
141 }
142}
143func malformedReason(fields []string) string {
144 if len(fields) != 8 {
145 return "fields"
146 }
147 if !validDate(fields[0]) {
148 return "date"
149 }
150 if !validTime(fields[1]) {
151 return "time"
152 }
153 if !validIP(fields[2]) {
154 return "ip"
155 }
156 if !includes(methods, fields[3]) {
157 return "method"
158 }
159 if !strings.HasPrefix(fields[4], "/") {
160 return "path"
161 }
162 if !validStatus(fields[5]) {
163 return "status"
164 }
165 if !canonical(fields[6], 8) {
166 return "bytes"
167 }
168 if !canonical(fields[7], 6) {
169 return "latency"
170 }
171 return ""
172}
173func dayNumber(y, m, d int) int {
174 days := d - 1
175 for yy := 2000; yy < y; yy++ {
176 days += daysInYear(yy)
177 }
178 for mm := 1; mm < m; mm++ {
179 days += daysInMonth(y, mm)
180 }
181 return days
182}
183func secondsOf(date, clock string) int {
184 y, m, d := parse(date[:4]), parse(date[5:7]), parse(date[8:10])
185 return dayNumber(y, m, d)*86400 + parse(clock[:2])*3600 +
186 parse(clock[3:5])*60 + parse(clock[6:8])
187}
188func intervalLabel(seconds int) string {
189 days, rest, year := seconds/86400, seconds%86400, 2000
190 for days >= daysInYear(year) {
191 days -= daysInYear(year)
192 year++
193 }
194 month := 1
195 for days >= daysInMonth(year, month) {
196 days -= daysInMonth(year, month)
197 month++
198 }
199 return fmt.Sprintf("%d-%02d-%02d %02d:%02d", year, month, days+1, rest/3600, rest%3600/60)
200}
201func normalizePath(raw string) string {
202 raw = strings.SplitN(raw, "?", 2)[0]
203 parts := make([]string, 0)
204 for _, item := range strings.Split(raw, "/") {
205 if item == ".." {
206 if len(parts) > 0 {
207 parts = parts[:len(parts)-1]
208 }
209 continue
210 }
211 if item == "" || item == "." {
212 continue
213 }
214 if digitsPattern.MatchString(item) {
215 item = "{id}"
216 } else if len(item) >= 8 && hexPattern.MatchString(item) && hexDigitPattern.MatchString(item) {
217 item = "{hex}"
218 }
219 parts = append(parts, item)
220 }
221 return "/" + strings.Join(parts, "/")
222}
223func compareIP(a, b string) int {
224 left, right := strings.Split(a, "."), strings.Split(b, ".")
225 for i := 0; i < 4; i++ {
226 diff := parse(left[i]) - parse(right[i])
227 if diff != 0 {
228 return diff
229 }
230 }
231 return 0
232}
233func statusMatches(filter string, status int) bool {
234 if includes(statusClasses, filter) {
235 return status/100 == int(filter[0]-'0')
236 }
237 return status == parse(filter)
238}
239func decimal(numerator, denominator, places int) string {
240 scale := 1
241 for range places {
242 scale *= 10
243 }
244 value := numerator * scale / denominator
245 return fmt.Sprintf("%d.%0*d", value/scale, places, value%scale)
246}
247func percent(numerator, denominator int) string { return decimal(numerator*100, denominator, 2) + "%" }
248func bar(count, maximum int) string { return strings.Repeat("#", (count*30+maximum-1)/maximum) }
249func bytesText(size int) string {
250 for _, pair := range []struct {
251 unit int
252 suffix string
253 }{
254 {1073741824, "GB"}, {1048576, "MB"}, {1024, "KB"},
255 } {
256 if size >= pair.unit {
257 return decimal(size, pair.unit, 1) + pair.suffix
258 }
259 }
260 return strconv.Itoa(size) + "B"
261}
262func sortedCounts(counts map[string]int, ip bool) []string {
263 keys := make([]string, 0, len(counts))
264 for k := range counts {
265 keys = append(keys, k)
266 }
267 sort.Slice(keys, func(i, j int) bool {
268 if counts[keys[i]] != counts[keys[j]] {
269 return counts[keys[i]] > counts[keys[j]]
270 }
271 if ip {
272 return compareIP(keys[i], keys[j]) < 0
273 }
274 return keys[i] < keys[j]
275 })
276 return keys
277}
278func bucketCounts(numbers []int) (int, []int, bool) {
279 first, last := numbers[0], numbers[0]
280 for _, n := range numbers {
281 if n < first {
282 first = n
283 }
284 if n > last {
285 last = n
286 }
287 }
288 if last-first+1 > 60 {
289 return 0, nil, false
290 }
291 counts := make([]int, last-first+1)
292 for _, n := range numbers {
293 counts[n-first]++
294 }
295 return first, counts, true
296}
297func percentile(sorted []int, rank int) int { return sorted[(rank*len(sorted)+99)/100-1] }

308}

331func uniqueCount(rows []Record, key func(Record) string) int { return len(grouped(rows, key)) }
332func latestPeak(rows []Record, length int) Window {
333 result := Window{}
334 first := 0
335 for last := range rows {
336 for rows[last].Seconds-rows[first].Seconds >= length {
337 first++
338 }
339 if last-first+1 > result.Size {
340 result = Window{Size: last - first + 1, Start: rows[first], End: rows[last]}
341 }
342 }
343 return result
344}
345func apdexScore(rows []Record, threshold int) Apdex {
346 satisfied, tolerating := 0, 0
347 for _, r := range rows {
348 if r.Status < 500 && r.Latency <= threshold {
349 satisfied++
350 } else if r.Status < 500 && r.Latency > threshold && r.Latency <= 4*threshold {
351 tolerating++
352 }
353 }
354 points, total := 2*satisfied+tolerating, 2*len(rows)
355 rating := "unacceptable"
356 switch {
357 case points*100 >= 94*total:
358 rating = "excellent"
359 case points*100 >= 85*total:
360 rating = "good"
361 case points*100 >= 70*total:
362 rating = "fair"
363 case points*100 >= 50*total:
364 rating = "poor"
365 }
366 return Apdex{Count: len(rows), Satisfied: satisfied, Tolerating: tolerating,
367 Frustrated: len(rows) - satisfied - tolerating, Points: points, Total: total,
368 Notation: decimal(points, total, 3), Rating: rating}
369}

400func newAnalyzer(lines []string) *Analyzer {
401 a := &Analyzer{LineCount: len(lines), Conditions: make(map[string]string)}
402 for i, line := range lines {
403 fields := strings.Fields(line)
404 reason := malformedReason(fields)
405 if reason != "" {
406 a.Malformed = append(a.Malformed, Malformed{Line: i + 1, Reason: reason})
407 continue
408 }
409 date, clock := fields[0], fields[1]
410 a.Records = append(a.Records, Record{
411 Line: i + 1, Hour: parse(clock[:2]), Seconds: secondsOf(date, clock),
412 Status: parse(fields[5]), Size: parse(fields[6]), Latency: parse(fields[7]),
413 Date: date, Timestamp: date + " " + clock, IP: fields[2], Method: fields[3],
414 Path: normalizePath(fields[4]),
415 })
416 }
417 sort.Slice(a.Records, func(i, j int) bool {
418 x, y := a.Records[i], a.Records[j]
419 if x.Seconds != y.Seconds {
420 return x.Seconds < y.Seconds
421 }
422 return x.Line < y.Line
423 })
424 return a
425}
426func (a *Analyzer) emit(parts ...any) {
427 words := make([]string, len(parts))
428 for i, v := range parts {
429 words[i] = fmt.Sprint(v)
430 }
431 a.Output = append(a.Output, strings.Join(words, " "))
432}
433func (a *Analyzer) selected() []Record {
434 out := make([]Record, 0, len(a.Records))
435 for _, r := range a.Records {
436 if a.Range != nil && (r.Seconds < a.Range.StartSeconds || r.Seconds > a.Range.EndSeconds) {
437 continue
438 }
439 if f := a.Conditions["METHOD"]; f != "" && f != r.Method {
440 continue
441 }
442 if f := a.Conditions["STATUS"]; f != "" && !statusMatches(f, r.Status) {
443 continue
444 }
445 if f := a.Conditions["IP"]; f != "" && f != r.IP {
446 continue
447 }
448 if f := a.Conditions["PATH"]; f != "" && f != r.Path {
449 continue
450 }
451 out = append(out, r)
452 }
453 return out
454}
455func (a *Analyzer) barLine(label string, count, maximum int) {
456 if count > 0 {
457 a.emit(label, count, bar(count, maximum))
458 } else {
459 a.emit(label, count)
460 }
461}
462func (a *Analyzer) execute(line string) {
463 tokens := strings.Fields(line)
464 name := tokens[0]
465 args := tokens[1:]
466 form, ok := forms[name]
467 if !ok {
468 a.emit("ERROR", "unknown query")
469 return
470 }
471 optional := (name == "LATENCY" || name == "SESSIONS") && len(args) == len(form)-1
472 if len(args) != len(form) && !optional {
473 a.emit("ERROR", "wrong arguments")
474 return
475 }
476 for i, v := range args {
477 if !validArgument(form[i], v, args) {
478 a.emit("ERROR", argumentErrors[form[i]])
479 return
480 }
481 }
482 rows := a.selected()
483 if requireRecords[name] && len(rows) == 0 {
484 a.emit(name, "no records")
485 return
486 }
487 if msg := a.run(name, args, rows); msg != "" {
488 a.emit("ERROR", msg)
489 }
490}
491func stringValues(rows []Record, f func(Record) string) []string {
492 values := make([]string, len(rows))
493 for i, r := range rows {
494 values[i] = f(r)
495 }
496 return values
497}
498func sortedMapKeys[T any](group map[string]T) []string {
499 keys := make([]string, 0, len(group))
500 for k := range group {
501 keys = append(keys, k)
502 }
503 sort.Strings(keys)
504 return keys
505}

507func (a *Analyzer) run(name string, args []string, rows []Record) string {
508 switch name {
509 case "SUMMARY":
510 a.emit("SUMMARY", "lines", a.LineCount, "valid", len(a.Records), "malformed", len(a.Malformed))
511 a.emit("requests", len(rows), "bytes", sumRecords(rows, "size"),
512 "ips", uniqueCount(rows, func(r Record) string { return r.IP }),
513 "paths", uniqueCount(rows, func(r Record) string { return r.Path }))
514 if len(rows) == 0 {
515 a.emit("span", "none")
516 } else {
517 first, last := rows[0], rows[len(rows)-1]
518 a.emit("span", first.Timestamp, last.Timestamp, last.Seconds-first.Seconds)
519 }
520 case "MALFORMED":
521 a.emit("MALFORMED", len(a.Malformed))
522 for _, reason := range reasons {
523 lines := make([]any, 0)
524 for _, row := range a.Malformed {
525 if row.Reason == reason {
526 lines = append(lines, row.Line)
527 }
528 }
529 if len(lines) > 0 {
530 a.emit(append([]any{reason, len(lines)}, lines...)...)
531 }
532 }
533 case "STATUS":
534 a.emit("STATUS", len(rows))
535 classes := countStatusClasses(rows)
536 for i, kind := range statusClasses {
537 a.emit(kind, classes[i], percent(classes[i], len(rows)))
538 }
539 statuses := make(map[int]int)
540 for _, r := range rows {
541 statuses[r.Status]++
542 }
543 codes := make([]int, 0, len(statuses))
544 for code := range statuses {
545 codes = append(codes, code)
546 }
547 sort.Ints(codes)
548 for _, code := range codes {
549 a.emit(code, statuses[code], percent(statuses[code], len(rows)))
550 }
551 case "METHODS":
552 counts := tallied(stringValues(rows, func(r Record) string { return r.Method }))
553 a.emit("METHODS", len(rows))
554 for _, method := range sortedCounts(counts, false) {
555 a.emit(method, counts[method], percent(counts[method], len(rows)))
556 }
557 case "TOP":
558 kind, limit := args[0], parse(args[1])
559 ip := kind == "IPS"
560 counts := tallied(stringValues(rows, func(r Record) string {
561 if ip {
562 return r.IP
563 }
564 return r.Path
565 }))
566 keys := sortedCounts(counts, ip)
567 shown := min(limit, len(keys))
568 a.emit("TOP", kind, shown, "of", len(keys))
569 rank := 0
570 for i, k := range keys[:shown] {
571 if i == 0 || counts[k] != counts[keys[i-1]] {
572 rank = i + 1
573 }
574 a.emit(rank, k, counts[k], percent(counts[k], len(rows)))
575 }
576 case "HOURLY":
577 var counts [24]int
578 for _, r := range rows {
579 counts[r.Hour]++
580 }
581 top, peak := 0, 0
582 for i, n := range counts {
583 if n > top {
584 top = n
585 peak = i
586 }
587 }
588 a.emit("HOURLY", len(rows), "peak", fmt.Sprintf("%02d", peak), top)
589 for i, n := range counts {
590 a.barLine(fmt.Sprintf("%02d", i), n, top)
591 }
592 case "TIMELINE":
593 minutes := parse(args[0])
594 width := minutes * 60
595 values := make([]int, len(rows))
596 for i, r := range rows {
597 values[i] = r.Seconds / width
598 }
599 first, counts, ok := bucketCounts(values)
600 if !ok {
601 return "too many buckets"
602 }
603 top := 0
604 for _, n := range counts {
605 top = max(top, n)
606 }
607 a.emit("TIMELINE", minutes, len(counts))
608 for i, n := range counts {
609 a.barLine(intervalLabel((first+i)*width), n, top)
610 }
611 case "BANDWIDTH":
612 days := grouped(rows, func(r Record) string { return r.Date })
613 volume := sumRecords(rows, "size")
614 a.emit("BANDWIDTH", len(days), volume, bytesText(volume))
615 for _, date := range sortedMapKeys(days) {
616 data := days[date]
617 count := sumRecords(data, "size")
618 a.emit(date, len(data), count, bytesText(count))
619 }
620 case "VISITORS":
621 first := make(map[string]string)
622 for _, r := range rows {
623 if date, ok := first[r.IP]; !ok || r.Date < date {
624 first[r.IP] = r.Date
625 }
626 }
627 days := grouped(rows, func(r Record) string { return r.Date })
628 a.emit("VISITORS", len(days), len(first))
629 for _, date := range sortedMapKeys(days) {
630 ips := make(map[string]bool)
631 for _, r := range days[date] {
632 ips[r.IP] = true
633 }
634 newcomers := 0
635 for ip := range ips {
636 if first[ip] == date {
637 newcomers++
638 }
639 }
640 a.emit(date, "unique", len(ips), "new", newcomers, "returning", len(ips)-newcomers)
641 }
642 case "LATENCY":
643 label := "all"
644 filtered := rows
645 if len(args) > 0 {
646 label = normalizePath(args[0])
647 filtered = make([]Record, 0)
648 for _, r := range rows {
649 if r.Path == label {
650 filtered = append(filtered, r)
651 }
652 }
653 }
654 if len(filtered) == 0 {
655 a.emit("LATENCY", "no records")
656 break
657 }
658 values := make([]int, len(filtered))
659 for i, r := range filtered {
660 values[i] = r.Latency
661 }
662 sort.Ints(values)
663 n := len(values)
664 trim := n / 10
665 trimmed := values[trim : n-trim]
666 total, trimmedTotal := 0, 0
667 for _, v := range values {
668 total += v
669 }
670 for _, v := range trimmed {
671 trimmedTotal += v
672 }
673 a.emit("LATENCY", label, "count", n, "min", values[0],
674 "p50", percentile(values, 50), "p90", percentile(values, 90),
675 "p99", percentile(values, 99), "max", values[n-1],
676 "mean", decimal(total, n, 2), "trimmed", decimal(trimmedTotal, len(trimmed), 2))
677 case "HISTOGRAM":
678 width := parse(args[0])
679 values := make([]int, len(rows))
680 for i, r := range rows {
681 values[i] = r.Latency / width
682 }
683 first, counts, ok := bucketCounts(values)
684 if !ok {
685 return "too many buckets"
686 }
687 top := 0
688 for _, n := range counts {
689 top = max(top, n)
690 }
691 a.emit("HISTOGRAM", width, len(counts))
692 for i, n := range counts {
693 lo := (first + i) * width
694 a.barLine(fmt.Sprintf("%d-%d", lo, lo+width-1), n, top)
695 }
696 case "APDEX":
697 threshold := parse(args[0])
698 overall := apdexScore(rows, threshold)
699 a.emit("APDEX", threshold, overall.Notation, overall.Rating,
700 "satisfied", overall.Satisfied, "tolerating", overall.Tolerating,
701 "frustrated", overall.Frustrated)
702 groupedPaths := grouped(rows, func(r Record) string { return r.Path })
703 type scoreRow struct {
704 Path string
705 Score Apdex
706 }
707 scores := make([]scoreRow, 0, len(groupedPaths))
708 for path, items := range groupedPaths {
709 scores = append(scores, scoreRow{path, apdexScore(items, threshold)})
710 }
711 sort.Slice(scores, func(i, j int) bool {
712 x, y := scores[i], scores[j]
713 left, right := x.Score.Points*y.Score.Total, y.Score.Points*x.Score.Total
714 if left != right {
715 return left < right
716 }
717 if x.Score.Count != y.Score.Count {
718 return x.Score.Count > y.Score.Count
719 }
720 return x.Path < y.Path
721 })
722 for _, row := range scores {
723 a.emit(row.Path, row.Score.Notation, row.Score.Rating, row.Score.Count)
724 }
725 case "ERRORS":
726 failed := 0
727 for _, r := range rows {
728 if r.Status >= 400 {
729 failed++
730 }
731 }
732 a.emit("ERRORS", failed, "of", len(rows), percent(failed, len(rows)))
733 groups := grouped(rows, func(r Record) string { return r.Path })
734 type errorRow struct {
735 Path string
736 Count, Client, Server int
737 }
738 errors := make([]errorRow, 0)
739 for path, items := range groups {
740 item := errorRow{Path: path, Count: len(items)}
741 for _, r := range items {
742 if r.Status/100 == 4 {
743 item.Client++
744 }
745 if r.Status/100 == 5 {
746 item.Server++
747 }
748 }
749 if item.Client+item.Server > 0 {
750 errors = append(errors, item)
751 }
752 }
753 sort.Slice(errors, func(i, j int) bool {
754 x, y := errors[i], errors[j]
755 xf, yf := x.Client+x.Server, y.Client+y.Server
756 if xf*y.Count != yf*x.Count {
757 return xf*y.Count > yf*x.Count
758 }
759 if xf != yf {
760 return xf > yf
761 }
762 return x.Path < y.Path
763 })
764 for _, row := range errors {
765 a.emit(row.Path, row.Count, row.Client, row.Server,
766 percent(row.Client+row.Server, row.Count))
767 }
768 case "SLOWEST":
769 groups := grouped(rows, func(r Record) string { return r.Method + "\x01" + r.Path })
770 type slowRow struct {
771 Method, Path string
772 Count, Total, Maximum int
773 }
774 slow := make([]slowRow, 0, len(groups))
775 for key, items := range groups {
776 pair := strings.Split(key, "\x01")
777 current := slowRow{Method: pair[0], Path: pair[1], Count: len(items),
778 Total: sumRecords(items, "latency")}
779 for _, r := range items {
780 current.Maximum = max(current.Maximum, r.Latency)
781 }
782 slow = append(slow, current)
783 }
784 sort.Slice(slow, func(i, j int) bool {
785 x, y := slow[i], slow[j]
786 if x.Total*y.Count != y.Total*x.Count {
787 return x.Total*y.Count > y.Total*x.Count
788 }
789 if x.Maximum != y.Maximum {
790 return x.Maximum > y.Maximum
791 }
792 if x.Path != y.Path {
793 return x.Path < y.Path
794 }
795 return x.Method < y.Method
796 })
797 shown := min(parse(args[0]), len(slow))
798 a.emit("SLOWEST", shown, "of", len(slow))
799 for i, r := range slow[:shown] {
800 a.emit(i+1, r.Method, r.Path, "count", r.Count,
801 "avg", decimal(r.Total, r.Count, 2), "max", r.Maximum)
802 }
803 case "SESSIONS":
804 ip := args[0]
805 minutes := 30
806 if len(args) > 1 {
807 minutes = parse(args[1])
808 }
809 found := make([]Record, 0)
810 for _, r := range rows {
811 if r.IP == ip {
812 found = append(found, r)
813 }
814 }
815 if len(found) == 0 {
816 a.emit("SESSIONS", "no records")
817 break
818 }
819 sessions := [][]Record{{found[0]}}
820 for i := 1; i < len(found); i++ {
821 prev := found[i-1]
822 current := found[i]
823 if current.Seconds-prev.Seconds <= minutes*60 {
824 last := len(sessions) - 1
825 sessions[last] = append(sessions[last], current)
826 } else {
827 sessions = append(sessions, []Record{current})
828 }
829 }
830 duration := 0
831 for _, session := range sessions {
832 duration += session[len(session)-1].Seconds - session[0].Seconds
833 }
834 a.emit("SESSIONS", ip, "timeout", minutes, "count", len(sessions),
835 "requests", len(found), "avg_duration", decimal(duration, len(sessions), 1),
836 "avg_requests", decimal(len(found), len(sessions), 1))
837 for i, session := range sessions {
838 start, end := session[0], session[len(session)-1]
839 a.emit(i+1, start.Timestamp, end.Timestamp, end.Seconds-start.Seconds,
840 len(session), uniqueCount(session, func(r Record) string { return r.Path }))
841 }
842 case "BURST":
843 length, times := parse(args[0]), parse(args[1])
844 failures := make([]Record, 0)
845 for _, r := range rows {
846 if r.Status >= 400 {
847 failures = append(failures, r)
848 }
849 }
850 ips := grouped(failures, func(r Record) string { return r.IP })
851 type burstRow struct {
852 IP string
853 Window Window
854 }
855 bursts := make([]burstRow, 0, len(ips))
856 for ip, items := range ips {
857 window := latestPeak(items, length)
858 if window.Size >= times {
859 bursts = append(bursts, burstRow{ip, window})
860 }
861 }
862 sort.Slice(bursts, func(i, j int) bool {
863 x, y := bursts[i], bursts[j]
864 if x.Window.Size != y.Window.Size {
865 return x.Window.Size > y.Window.Size
866 }
867 return compareIP(x.IP, y.IP) < 0
868 })
869 a.emit("BURST", length, times, len(bursts))
870 for _, row := range bursts {
871 a.emit(row.IP, row.Window.Size, row.Window.Start.Timestamp, row.Window.End.Timestamp)
872 }
873 case "PEAK":
874 length := parse(args[0])
875 window := latestPeak(rows, length)
876 a.emit("PEAK", length, window.Size, window.Start.Timestamp, window.End.Timestamp)
877 case "GAPS":
878 length := parse(args[0])
879 count := 0
880 for i := 1; i < len(rows); i++ {
881 if rows[i].Seconds-rows[i-1].Seconds >= length {
882 count++
883 }
884 }
885 a.emit("GAPS", length, count)
886 for i := 1; i < len(rows); i++ {
887 if rows[i].Seconds-rows[i-1].Seconds >= length {
888 a.emit(rows[i-1].Timestamp, rows[i].Timestamp, rows[i].Seconds-rows[i-1].Seconds)
889 }
890 }
891 case "IP":
892 ip := args[0]
893 found := make([]Record, 0)
894 for _, r := range rows {
895 if r.IP == ip {
896 found = append(found, r)
897 }
898 }
899 if len(found) == 0 {
900 a.emit("IP", "no records")
901 break
902 }
903 paths := tallied(stringValues(found, func(r Record) string { return r.Path }))
904 errors := 0
905 for _, r := range found {
906 if r.Status >= 400 {
907 errors++
908 }
909 }
910 a.emit("IP", ip, "requests", len(found), "bytes", sumRecords(found, "size"),
911 "errors", errors, "paths", len(paths))
912 a.emit("first", found[0].Timestamp, "last", found[len(found)-1].Timestamp)
913 methods := tallied(stringValues(found, func(r Record) string { return r.Method }))
914 info := []any{"methods"}
915 for _, method := range sortedCounts(methods, false) {
916 info = append(info, method, methods[method])
917 }
918 a.emit(info...)
919 classes := countStatusClasses(found)
920 info = []any{"classes"}
921 for i, c := range statusClasses {
922 info = append(info, c, classes[i])
923 }
924 a.emit(info...)
925 keys := sortedCounts(paths, false)
926 for _, path := range keys[:min(3, len(keys))] {
927 a.emit("path", path, paths[path])
928 }
929 case "RANGE":
930 start, end := args[0]+" "+args[1], args[2]+" "+args[3]
931 startSeconds, endSeconds := secondsOf(args[0], args[1]), secondsOf(args[2], args[3])
932 if startSeconds > endSeconds {
933 return "invalid range"
934 }
935 a.Range = &TimeRange{Start: start, End: end, StartSeconds: startSeconds, EndSeconds: endSeconds}
936 a.emit("RANGE", start, end, "active", len(a.selected()))
937 case "FILTER":
938 key, value := args[0], args[1]
939 if key == "PATH" {
940 value = normalizePath(value)
941 }
942 a.Conditions[key] = value
943 a.emit("FILTER", key, value, "active", len(a.selected()))
944 case "UNFILTER":
945 key := args[0]
946 if key == "RANGE" {
947 if a.Range == nil {
948 return "not filtered"
949 }
950 a.Range = nil
951 } else {
952 if a.Conditions[key] == "" {
953 return "not filtered"
954 }
955 delete(a.Conditions, key)
956 }
957 a.emit("UNFILTER", key, "active", len(a.selected()))
958 case "RESET":
959 a.Range = nil
960 a.Conditions = make(map[string]string)
961 a.emit("RESET", "active", len(a.selected()))
962 case "STATE":
963 timeRange := "-"
964 if a.Range != nil {
965 timeRange = a.Range.Start + " " + a.Range.End
966 }
967 info := []any{"STATE", "active", len(rows), "range", timeRange}
968 for _, key := range filterKeys {
969 value := a.Conditions[key]
970 if value == "" {
971 value = "-"
972 }
973 info = append(info, strings.ToLower(key), value)
974 }
975 a.emit(info...)
976 }
977 return ""
978}

980func main() {
981 scanner := bufio.NewScanner(os.Stdin)
982 scanner.Buffer(make([]byte, 4096), 1024*1024)
983 lines := make([]string, 0)
984 for scanner.Scan() {
985 lines = append(lines, scanner.Text())
986 }
987 separator := -1
988 end := -1
989 for i, line := range lines {
990 if separator < 0 && line == "---" {
991 separator = i
992 continue
993 }
994 if separator >= 0 && line == "END" {
995 end = i
996 break
997 }
998 }
999 analyzer := newAnalyzer(lines[:separator])
1000 for _, line := range lines[separator+1 : end] {
1001 analyzer.execute(line)
1002 }
1003 if len(analyzer.Output) > 0 {
1004 fmt.Println(strings.Join(analyzer.Output, "\n"))
1005 }
1006}
