47def digits?(text)
48 /\A[0-9]+\z/.match?(text)
49end
50
51def canonical?(text, maximum)
52 digits?(text) && text.length <= maximum && (text.length == 1 || text[0] != '0')
53end
54
55def leap_year?(year)
56 (year % 4).zero? && (!(year % 100).zero? || (year % 400).zero?)
57end
58
59def year_days(year)
60 leap_year?(year) ? 366 : 365
61end
62
63def month_days(year, month)
64 month == 2 && leap_year?(year) ? 29 : MONTH_DAYS[month - 1]
65end
66
67def valid_date?(text)
68 return false unless /\A[0-9]{4}-[0-9]{2}-[0-9]{2}\z/.match?(text)
69 year, month, day = text.split('-').map(&:to_i)
70 year.between?(2000, 2059) && month.between?(1, 12) &&
71 day.between?(1, month_days(year, month))
72end
73
74def valid_time?(text)
75 return false unless /\A[0-9]{2}:[0-9]{2}:[0-9]{2}\z/.match?(text)
76 hour, minute, second = text.split(':').map(&:to_i)
77 hour <= 23 && minute <= 59 && second <= 59
78end
79
80def valid_ip?(text)
81 parts = text.split('.', -1)
82 parts.length == 4 && parts.all? { |part| canonical?(part, 3) && part.to_i <= 255 }
83end
84
85def valid_status?(text)
86 text.length == 3 && digits?(text) && text.to_i.between?(100, 599)
87end
88
89def valid_filter_value?(key, value)
90 case key
91 when 'METHOD' then METHODS.include?(value)
92 when 'STATUS' then STATUS_CLASSES.include?(value) || valid_status?(value)
93 when 'IP' then valid_ip?(value)
94 else value.start_with?('/')
95 end
96end
97
98def valid_argument?(kind, value, args)
99 if NUMBER_RANGES.key?(kind)
100 low, high = NUMBER_RANGES.fetch(kind)
101 return canonical?(value, 6) && value.to_i.between?(low, high)
102 end
103 case kind
104 when 'kind' then TOP_KINDS.include?(value)
105 when 'path' then value.start_with?('/')
106 when 'ip' then valid_ip?(value)
107 when 'date' then valid_date?(value)
108 when 'time' then valid_time?(value)
109 when 'key' then FILTER_KEYS.include?(value)
110 when 'subject' then FILTER_KEYS.include?(value) || value == 'RANGE'
111 else valid_filter_value?(args[0], value)
112 end
113end

115def malformed_reason(fields)
116 return 'fields' unless fields.length == 8
117 date, clock, ip, method, path, status, bytes, latency = fields
118 return 'date' unless valid_date?(date)
119 return 'time' unless valid_time?(clock)
120 return 'ip' unless valid_ip?(ip)
121 return 'method' unless METHODS.include?(method)
122 return 'path' unless path.start_with?('/')
123 return 'status' unless valid_status?(status)
124 return 'bytes' unless canonical?(bytes, 8)
125 return 'latency' unless canonical?(latency, 6)
126 nil
127end

129def day_number(year, month, day)
130 days = day - 1
131 (2000...year).each { |y| days += year_days(y) }
132 (1...month).each { |m| days += month_days(year, m) }
133 days
134end
135
136def seconds_of(date, clock)
137 year, month, day = date.split('-').map(&:to_i)
138 hour, minute, second = clock.split(':').map(&:to_i)
139 day_number(year, month, day) * 86_400 + hour * 3600 + minute * 60 + second
140end
141
142def interval_label(seconds)
143 days, rest = seconds.divmod(86_400)
144 year = 2000
145 while days >= year_days(year)
146 days -= year_days(year)
147 year += 1
148 end
149 month = 1
150 while days >= month_days(year, month)
151 days -= month_days(year, month)
152 month += 1
153 end
154 format('%d-%02d-%02d %02d:%02d', year, month, days + 1, rest / 3600, rest % 3600 / 60)
155end

157def normalize_path(text)
158 parts = []
159 text.split('?', 2)[0].split('/').each do |part|
160 case part
161 when '..' then parts.pop
162 when '', '.' then next
163 else parts << part
164 end
165 end
166 '/' + parts.map do |part|
167 if digits?(part)
168 '{id}'
169 elsif part.length >= 8 && /\A[0-9a-f]+\z/.match?(part) && /[0-9]/.match?(part)
170 '{hex}'
171 else
172 part
173 end
174 end.join('/')
175end

177def ip_order(ip)
178 ip.split('.').map(&:to_i)

181def status_matches?(value, status)
182 STATUS_CLASSES.include?(value) ? status / 100 == value[0].to_i : status == value.to_i
183end

185def decimal(numerator, denominator, places)
186 scale = 10**places
187 whole, fraction = ((numerator * scale) / denominator).divmod(scale)
188 "#{whole}.#{fraction.to_s.rjust(places, '0')}"
189end
190
191def percentage(part, whole)
192 decimal(part * 100, whole, 2) + '%'
193end
194
195def bar(count, maximum)
196 '#' * ((count * BAR_WIDTH + maximum - 1) / maximum)
197end

199def byte_notation(size)
200 BYTE_UNITS.each do |unit, suffix|
201 return decimal(size, unit, 1) + suffix if size >= unit
202 end
203 "#{size}B"
204end

210def by_count(counts)
211 counts.keys.sort_by { |key| [-counts[key], block_given? ? yield(key) : key] }
212end

214def class_counts(records)
215 counts = Array.new(5, 0)
216 records.each { |record| counts[record.status / 100 - 1] += 1 }
217 counts
218end
219
220def bucket_counts(numbers)
221 first, last = numbers.minmax
222 raise QueryError, 'too many buckets' if last - first + 1 > MAX_BUCKETS
223 counts = Array.new(last - first + 1, 0)
224 numbers.each { |number| counts[number - first] += 1 }
225 [first, counts]
226end

228def percentile(values, rank)
229 values[(rank * values.length + 99) / 100 - 1]
230end

232class Record
233 attr_reader :line_number, :date, :hour, :timestamp, :seconds, :ip, :method, :path,
234 :status, :size, :latency
235
236 def initialize(line_number, fields)
237 date, clock, @ip, @method, path, status, size, latency = fields
238 @line_number = line_number
239 @date = date
240 @hour = clock[0, 2].to_i
241 @timestamp = "#{date} #{clock}"
242 @seconds = seconds_of(date, clock)
243 @path = normalize_path(path)
244 @status, @size, @latency = status.to_i, size.to_i, latency.to_i
245 end
246
247 def order
248 [@seconds, @line_number]
249 end
250
251 def error?
252 @status >= 400
253 end
254end

256class TimeRange
257 attr_reader :start, :end, :start_seconds, :end_seconds
258
259 def initialize(start_date, start_time, end_date, end_time)
260 @start = "#{start_date} #{start_time}"
261 @end = "#{end_date} #{end_time}"
262 @start_seconds = seconds_of(start_date, start_time)

271class Filters
272 attr_accessor :range
273 attr_reader :conditions
274
275 def initialize
276 @range = nil
277 @conditions = FILTER_KEYS.to_h { |key| [key, nil] }
278 end
279
280 def accepts?(record)
281 return false if @range && !@range.contains?(record.seconds)
282 return false if @conditions['METHOD'] && record.method != @conditions['METHOD']
283 return false if @conditions['STATUS'] && !status_matches?(@conditions['STATUS'], record.status)
284 return false if @conditions['IP'] && record.ip != @conditions['IP']
285 @conditions['PATH'].nil? || record.path == @conditions['PATH']
286 end
287end

289class Window
290 attr_reader :size, :start, :end
291
292 def initialize(records, length)
293 @size, @start, @end = 0, nil, nil
294 first = 0
295 records.each_with_index do |record, last|
296 first += 1 while record.seconds - records[first].seconds >= length
297 next unless last - first + 1 > @size
298 @size = last - first + 1
299 @start, @end = records[first], record
300 end
301 end
302end

304class ApdexScore
305 attr_reader :count, :satisfied, :tolerating, :frustrated, :score
306
307 def initialize(records, threshold)
308 @count = records.length
309 @satisfied = records.count { |r| r.status < 500 && r.latency <= threshold }
310 @tolerating = records.count { |r| r.status < 500 && r.latency > threshold && r.latency <= 4 * threshold }
311 @frustrated = @count - @satisfied - @tolerating
312 @points = 2 * @satisfied + @tolerating
313 @score = Rational(@points, 2 * @count)
314 end
315
316 def notation
317 decimal(@points, 2 * @count, 3)
318 end
319
320 def rating
321 RATINGS.each { |minimum, name| return name if @score >= minimum }
322 'unacceptable'
323 end
324end

357class LogAnalyzer
358 attr_reader :output
359
360 def initialize(log_lines)
361 @line_count = log_lines.length
362 @records = []
363 @malformed = []
364 log_lines.each_with_index do |line, index|
365 fields = line.split
366 reason = malformed_reason(fields)
367 if reason.nil?
368 @records << Record.new(index + 1, fields)
369 else
370 @malformed << [index + 1, reason]
371 end
372 end
373 @records.sort_by!(&:order)
374 @filters = Filters.new
375 @output = []
376 @handlers = {
377 'SUMMARY' => :summary, 'MALFORMED' => :malformed_lines, 'STATUS' => :status_breakdown,
378 'METHODS' => :method_breakdown, 'TOP' => :top, 'HOURLY' => :hourly,
379 'TIMELINE' => :timeline, 'BANDWIDTH' => :bandwidth, 'VISITORS' => :visitors,
380 'LATENCY' => :latency, 'HISTOGRAM' => :histogram, 'APDEX' => :apdex,
381 'ERRORS' => :errors, 'SLOWEST' => :slowest, 'SESSIONS' => :sessions,
382 'BURST' => :burst, 'PEAK' => :peak, 'GAPS' => :gaps, 'IP' => :ip_details,
383 'RANGE' => :set_range, 'FILTER' => :add_filter, 'UNFILTER' => :remove_filter,
384 'RESET' => :reset, 'STATE' => :state
385 }
386 end
387
388 def emit(*parts)
389 @output << parts.join(' ')
390 end

392 def bar_line(label, count, maximum)
393 if count.zero?
394 emit(label, count)
395 else
396 emit(label, count, bar(count, maximum))
397 end

404 def summary(_arguments, records)
405 emit('SUMMARY', 'lines', @line_count, 'valid', @records.length,
406 'malformed', @malformed.length)
407 emit('requests', records.length, 'bytes', records.sum(&:size),
408 'ips', records.map(&:ip).uniq.length, 'paths', records.map(&:path).uniq.length)
409 if records.empty?
410 emit('span', 'none')
411 else
412 first, last = records.first, records.last
413 emit('span', first.timestamp, last.timestamp, last.seconds - first.seconds)
414 end

419 REASONS.each do |reason|

421 emit(reason, numbers.length, *numbers) unless numbers.empty?
422 end

445 def top(arguments, records)
446 kind, limit = arguments[0], arguments[1].to_i
447 counts = records.map { |r| kind == 'PATHS' ? r.path : r.ip }.tally
448 keys = kind == 'PATHS' ? by_count(counts) : by_count(counts) { |ip| ip_order(ip) }
449 shown = [limit, keys.length].min
450 emit('TOP', kind, shown, 'of', keys.length)
451 rank = 0
452 keys.take(shown).each_with_index do |key, index|
453 rank = index + 1 if index.zero? || counts[key] != counts[keys[index - 1]]
454 emit(rank, key, counts[key], percentage(counts[key], records.length))
455 end
456 end

458 def hourly(_arguments, records)
459 counts = Array.new(24, 0)
460 records.each { |r| counts[r.hour] += 1 }
461 maximum = counts.max
462 peak_hour = counts.index(maximum)
463 emit('HOURLY', records.length, 'peak', format('%02d', peak_hour), maximum)
464 counts.each_with_index do |amount, hour|
465 bar_line(format('%02d', hour), amount, maximum)

469 def timeline(arguments, records)
470 minutes = arguments[0].to_i
471 width = minutes * 60
472 first, counts = bucket_counts(records.map { |r| r.seconds / width })
473 emit('TIMELINE', minutes, counts.length)
474 maximum = counts.max
475 counts.each_with_index do |amount, offset|

490 def visitors(_arguments, records)
491 first_days = {}
492 records.each do |r|
493 first_days[r.ip] = r.date if !first_days.key?(r.ip) || r.date < first_days[r.ip]
494 end
495 days = records.group_by(&:date)
496 emit('VISITORS', days.length, first_days.length)
497 days.keys.sort.each do |date|
498 ips = days[date].map(&:ip).uniq
499 first = ips.count { |ip| first_days[ip] == date }
500 emit(date, 'unique', ips.length, 'new', first, 'returning', ips.length - first)
501 end

504 def latency(arguments, records)
505 if arguments.empty?
506 name = 'all'
507 selected = records
508 else
509 name = normalize_path(arguments[0])
510 selected = records.select { |r| r.path == name }
511 end
512 if selected.empty?
513 emit('LATENCY', 'no records')
514 return
515 end
516 values = selected.map(&:latency).sort
517 count = values.length
518 trim = count / 10
519 trimmed = values[trim, count - 2 * trim]
520 emit('LATENCY', name, 'count', count, 'min', values.first,
521 'p50', percentile(values, 50), 'p90', percentile(values, 90),
522 'p99', percentile(values, 99), 'max', values.last,
523 'mean', decimal(values.sum, count, 2),
524 'trimmed', decimal(trimmed.sum, trimmed.length, 2))
525 end

528 width = arguments[0].to_i
529 first, counts = bucket_counts(records.map { |r| r.latency / width })
530 emit('HISTOGRAM', width, counts.length)
531 maximum = counts.max

538 def apdex(arguments, records)
539 threshold = arguments[0].to_i
540 overall = ApdexScore.new(records, threshold)
541 emit('APDEX', threshold, overall.notation, overall.rating,
542 'satisfied', overall.satisfied, 'tolerating', overall.tolerating,

574 def sessions(arguments, records)
575 ip = arguments[0]
576 minutes = arguments.length == 2 ? arguments[1].to_i : DEFAULT_TIMEOUT
577 selected = records.select { |r| r.ip == ip }
578 if selected.empty?
579 emit('SESSIONS', 'no records')
580 return
581 end
582 sessions = [[selected[0]]]
583 selected.each_cons(2) do |previous, current|
584 if current.seconds - previous.seconds <= minutes * 60
585 sessions[-1] << current
586 else
587 sessions << [current]
588 end
589 end
590 durations = sessions.sum { |s| s.last.seconds - s.first.seconds }
591 emit('SESSIONS', ip, 'timeout', minutes, 'count', sessions.length,
592 'requests', selected.length, 'avg_duration', decimal(durations, sessions.length, 1),
593 'avg_requests', decimal(selected.length, sessions.length, 1))
594 sessions.each_with_index do |session, index|
595 first, last = session.first, session.last
596 emit(index + 1, first.timestamp, last.timestamp, last.seconds - first.seconds,
597 session.length, session.map(&:path).uniq.length)
598 end
599 end

601 def burst(arguments, records)
602 length, times = arguments.map(&:to_i)
603 failures = records.select(&:error?)
604 matches = failures.group_by(&:ip).filter_map do |ip, group|
605 window = Window.new(group, length)
606 [ip, window] if window.size >= times
607 end
608 matches.sort_by! { |ip, window| [-window.size, ip_order(ip)] }
609 emit('BURST', length, times, matches.length)
610 matches.each do |ip, window|
611 emit(ip, window.size, window.start.timestamp, window.end.timestamp)
612 end
613 end

630 selected = records.select { |r| r.ip == ip }
631 if selected.empty?
632 emit('IP', 'no records')
633 return
634 end
635 paths = selected.map(&:path).tally
636 emit('IP', ip, 'requests', selected.length, 'bytes', selected.sum(&:size),

656 def add_filter(arguments, _records)
657 key, value = arguments
658 value = normalize_path(value) if key == 'PATH'
659 @filters.conditions[key] = value
660 emit('FILTER', key, value, 'active', targets.length)
661 end
662
663 def remove_filter(arguments, _records)
664 key = arguments[0]
665 if key == 'RANGE'
666 raise QueryError, 'not filtered' if @filters.range.nil?
667 @filters.range = nil
668 else
669 raise QueryError, 'not filtered' if @filters.conditions[key].nil?
670 @filters.conditions[key] = nil
671 end
672 emit('UNFILTER', key, 'active', targets.length)
673 end
674
675 def reset(_arguments, _records)
676 @filters = Filters.new
677 emit('RESET', 'active', targets.length)
678 end
679
680 def state(_arguments, records)
681 range = @filters.range
682 range_text = range.nil? ? '-' : "#{range.start} #{range.end}"
683 parts = ['STATE', 'active', records.length, 'range', range_text]
684 FILTER_KEYS.each do |key|
685 value = @filters.conditions[key]
686 parts.concat([key.downcase, value.nil? ? '-' : value])
687 end
688 emit(*parts)
689 end
690
691 def execute(line)
692 name, *args = line.split
693 begin
694 form = QUERY_FORMS[name]
695 raise QueryError, 'unknown query' if form.nil?
696 optional = OPTIONAL_LAST.include?(name) && args.length == form.length - 1
697 raise QueryError, 'wrong arguments' if args.length != form.length && !optional
698 form.zip(args).each do |kind, value|
699 next if value.nil?
700 raise QueryError, ARGUMENT_ERRORS[kind] unless valid_argument?(kind, value, args)
701 end
702 records = targets
703 if NEEDS_TARGETS.include?(name) && records.empty?
704 emit(name, 'no records')
705 else
706 send(@handlers.fetch(name), args, records)
707 end
708 rescue QueryError => e
709 emit('ERROR', e.message)
710 end
711 end
712end
