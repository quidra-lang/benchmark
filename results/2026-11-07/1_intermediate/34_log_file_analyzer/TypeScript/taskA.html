3const METHODS = ['GET', 'HEAD', 'POST', 'PUT', 'DELETE', 'PATCH', 'OPTIONS'];
4const REASONS = ['fields', 'date', 'time', 'ip', 'method', 'path', 'status', 'bytes', 'latency'];
5const STATUS_CLASSES = ['1xx', '2xx', '3xx', '4xx', '5xx'];
6const FILTER_KEYS = ['METHOD', 'STATUS', 'IP', 'PATH'];
7const MONTH_DAYS = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
8const FORMS: Record<string, string[]> = {
9 SUMMARY: [], MALFORMED: [], STATUS: [], METHODS: [], TOP: ['kind', 'limit'],
10 HOURLY: [], TIMELINE: ['minutes'], BANDWIDTH: [], VISITORS: [],
11 LATENCY: ['path'], HISTOGRAM: ['width'], APDEX: ['threshold'], ERRORS: [],
12 SLOWEST: ['limit'], SESSIONS: ['ip', 'minutes'], BURST: ['length', 'times'],
13 PEAK: ['length'], GAPS: ['length'], IP: ['ip'],
14 RANGE: ['date', 'time', 'date', 'time'], FILTER: ['key', 'value'],
15 UNFILTER: ['subject'], RESET: [], STATE: []
16};
17const OPTIONAL = new Set(['LATENCY', 'SESSIONS']);
18const REQUIRE_RECORDS = new Set([
19 'STATUS', 'METHODS', 'TOP', 'HOURLY', 'TIMELINE', 'BANDWIDTH',
20 'VISITORS', 'HISTOGRAM', 'APDEX', 'ERRORS', 'SLOWEST', 'BURST', 'PEAK', 'GAPS'
21]);
22const RANGES: Record<string, [number, number]> = {
23 limit: [1, 100], minutes: [1, 1440], width: [1, 100000],
24 threshold: [1, 100000], length: [1, 86400], times: [1, 1000]
25};
26const ERRORS: Record<string, string> = {
27 limit: 'invalid number', minutes: 'invalid number', width: 'invalid number',
28 threshold: 'invalid number', length: 'invalid number', times: 'invalid number',
29 kind: 'invalid kind', path: 'invalid path', ip: 'invalid ip',
30 date: 'invalid timestamp', time: 'invalid timestamp',
31 key: 'invalid key', subject: 'invalid key', value: 'invalid value'
32};
33const DAY_SECONDS = 86400;
34const MAX_BUCKETS = 60;

36type LogRecord = {line: number; date: string; hour: number; timestamp: string; seconds: number; ip: string; method: string; path: string; status: number; size: number; latency: number};
37type MalformedLine = {line: number; reason: string};

44function check(condition: boolean, reason: string): asserts condition {
45 if (!condition) throw new QueryError(reason);
46}
47function digits(text: string): boolean {
48 return /^[0-9]+$/.test(text);
49}
50function canonical(text: string, maxDigits: number): boolean {
51 return typeof text === 'string' && digits(text) && text.length <= maxDigits &&
52 (text.length === 1 || text[0] !== '0');
53}
54function leapYear(year: number): boolean {
55 return year % 4 === 0 && (year % 100 !== 0 || year % 400 === 0);
56}
57function daysInYear(year: number): number {
58 return leapYear(year) ? 366 : 365;
59}
60function daysInMonth(year: number, month: number): number {
61 return month === 2 && leapYear(year) ? 29 : MONTH_DAYS[month - 1];
62}
63function validDate(text: string): boolean {
64 if (!/^\d{4}-\d{2}-\d{2}$/.test(text)) return false;
65 const [year, month, day] = text.split('-').map(Number);
66 return year >= 2000 && year <= 2059 && month >= 1 && month <= 12 &&
67 day >= 1 && day <= daysInMonth(year, month);
68}
69function validTime(text: string): boolean {
70 if (!/^\d{2}:\d{2}:\d{2}$/.test(text)) return false;
71 const [hour, minute, second] = text.split(':').map(Number);
72 return hour < 24 && minute < 60 && second < 60;
73}
74function validIp(text: string): boolean {
75 const parts = text.split('.');
76 return parts.length === 4 && parts.every(part => canonical(part, 3) && Number(part) <= 255);
77}
78function validStatus(text: string): boolean {
79 return /^\d{3}$/.test(text) && Number(text) >= 100 && Number(text) <= 599;
80}
81function validFilterValue(kind: string, value: string): boolean {
82 if (kind === 'METHOD') return METHODS.includes(value);
83 if (kind === 'STATUS') return STATUS_CLASSES.includes(value) || validStatus(value);
84 if (kind === 'IP') return validIp(value);
85 return value.startsWith('/');
86}
87function validArgument(type: string, text: string, args: string[]): boolean {
88 if (Object.hasOwn(RANGES, type)) {
89 const [low, high] = RANGES[type];
90 return canonical(text, 6) && Number(text) >= low && Number(text) <= high;
91 }
92 switch (type) {
93 case 'kind': return ['PATHS', 'IPS'].includes(text);
94 case 'path': return text.startsWith('/');
95 case 'ip': return validIp(text);
96 case 'date': return validDate(text);
97 case 'time': return validTime(text);
98 case 'key': return FILTER_KEYS.includes(text);
99 case 'subject': return text === 'RANGE' || FILTER_KEYS.includes(text);
100 default: return validFilterValue(args[0], text);
101 }
102}
103function malformedReason(fields: string[]): string | null {
104 if (fields.length !== 8) return 'fields';
105 const [date, clock, ip, method, path, status, bytes, latency] = fields;
106 if (!validDate(date)) return 'date';
107 if (!validTime(clock)) return 'time';
108 if (!validIp(ip)) return 'ip';
109 if (!METHODS.includes(method)) return 'method';
110 if (!path.startsWith('/')) return 'path';
111 if (!validStatus(status)) return 'status';
112 if (!canonical(bytes, 8)) return 'bytes';
113 if (!canonical(latency, 6)) return 'latency';
114 return null;
115}
116function dayNumber(year: number, month: number, day: number): number {
117 let days = day - 1;
118 for (let y = 2000; y < year; ++y) days += daysInYear(y);
119 for (let m = 1; m < month; ++m) days += daysInMonth(year, m);
120 return days;
121}
122function secondsOf(date: string, clock: string): number {
123 const [year, month, day] = date.split('-').map(Number);
124 const [hour, minute, second] = clock.split(':').map(Number);
125 return dayNumber(year, month, day) * DAY_SECONDS + hour * 3600 + minute * 60 + second;
126}
127function intervalLabel(seconds: number): string {
128 let days = Math.floor(seconds / DAY_SECONDS);
129 let rest = seconds % DAY_SECONDS;
130 let year = 2000;
131 while (days >= daysInYear(year)) days -= daysInYear(year++);
132 let month = 1;
133 while (days >= daysInMonth(year, month)) days -= daysInMonth(year, month++);
134 return year + '-' + String(month).padStart(2, '0') + '-' + String(days + 1).padStart(2, '0') +
135 ' ' + String(Math.floor(rest / 3600)).padStart(2, '0') + ':' +
136 String(Math.floor(rest % 3600 / 60)).padStart(2, '0');
137}
138function normalizePath(raw: string): string {
139 const parts = [];
140 for (const item of raw.split('?')[0].split('/')) {
141 if (item === '..') parts.pop();
142 else if (item !== '' && item !== '.') parts.push(item);
143 }
144 return '/' + parts.map(part => {
145 if (digits(part)) return '{id}';
146 if (part.length >= 8 && /^[0-9a-f]+$/.test(part) && /[0-9]/.test(part)) return '{hex}';
147 return part;
148 }).join('/');
149}
150function compareIp(a: string, b: string): number {
151 const left = a.split('.').map(Number), right = b.split('.').map(Number);
152 for (let i = 0; i < 4; ++i) if (left[i] !== right[i]) return left[i] - right[i];
153 return 0;
154}
155function statusMatches(filter: string, status: number): boolean {
156 return STATUS_CLASSES.includes(filter) ? Math.floor(status / 100) === Number(filter[0]) : status === Number(filter);
157}
158function decimal(numerator: number, denominator: number, places: number): string {
159 const scale = 10 ** places;
160 const value = Math.floor(numerator * scale / denominator);
161 return Math.floor(value / scale) + '.' + String(value % scale).padStart(places, '0');
162}
163function percent(numerator: number, denominator: number): string {
164 return decimal(numerator * 100, denominator, 2) + '%';
165}
166function bar(count: number, maximum: number): string {
167 return '#'.repeat(Math.floor((count * 30 + maximum - 1) / maximum));
168}
169function bytesText(size: number): string {
170 for (const [unit, suffix] of ([[1073741824, 'GB'], [1048576, 'MB'], [1024, 'KB']] as [number, string][])) {
171 if (size >= unit) return decimal(size, unit, 1) + suffix;
172 }
173 return size + 'B';
174}
175function sum(records: LogRecord[], key: "size" | "latency"): number {
176 return records.reduce((total, record) => total + record[key], 0);
177}
178function grouped(records: LogRecord[], key: keyof LogRecord | ((r: LogRecord) => string)): Map<string, LogRecord[]> {
179 const groups = new Map<string, LogRecord[]>();
180 for (const row of records) {
181 const k = typeof key === 'function' ? key(row) : String(row[key]);
182 if (!groups.has(k)) groups.set(k, []);
183 groups.get(k)!.push(row);
184 }
185 return groups;
186}
187function tallied<T extends string | number>(values: T[]): Map<T, number> {
188 const counts = new Map<T, number>();
189 for (const value of values) counts.set(value, (counts.get(value) ?? 0) + 1);
190 return counts;
191}
192function sortedByCount<T extends string | number>(counts: Map<T, number>, comparator: (a: T, b: T) => number = (a, b) => a < b ? -1 : (a > b ? 1 : 0)): T[] {
193 return [...counts.keys()].sort((a, b) => (counts.get(b)! - counts.get(a)!) || comparator(a, b));
194}
195function statusClassCounts(records: LogRecord[]): number[] {
196 const counts = [0, 0, 0, 0, 0];
197 for (const record of records) ++counts[Math.floor(record.status / 100) - 1];
198 return counts;
199}
200function bucketCounts(numbers: number[]): [number, number[]] {
201 const first = Math.min(...numbers), last = Math.max(...numbers);
202 check(last - first + 1 <= MAX_BUCKETS, 'too many buckets');
203 const counts = Array(last - first + 1).fill(0);
204 for (const n of numbers) ++counts[n - first];
205 return [first, counts];
206}
207function percentile(sorted: number[], rank: number): number {
208 return sorted[Math.floor((rank * sorted.length + 99) / 100) - 1];
209}
210function windowPeak(records: LogRecord[], length: number): PeakWindow {
211 let size = 0, start: LogRecord | null = null, end: LogRecord | null = null;
212 let first = 0;
213 for (let last = 0; last < records.length; ++last) {
214 while (records[last].seconds - records[first].seconds >= length) ++first;
215 if (last - first + 1 > size) {
216 size = last - first + 1;
217 start = records[first];
218 end = records[last];
219 }
220 }
221 return {size, start, end};
222}
223function apdexScore(records: LogRecord[], threshold: number): Apdex {
224 const satisfied = records.filter(r => r.status < 500 && r.latency <= threshold).length;
225 const tolerating = records.filter(r => r.status < 500 && r.latency > threshold && r.latency <= 4 * threshold).length;
226 const frustrated = records.length - satisfied - tolerating;
227 const points = 2 * satisfied + tolerating, total = 2 * records.length;
228 const rating = points * 100 >= 94 * total ? 'excellent' :
229 points * 100 >= 85 * total ? 'good' :
230 points * 100 >= 70 * total ? 'fair' :
231 points * 100 >= 50 * total ? 'poor' : 'unacceptable';
232 return {count: records.length, satisfied, tolerating, frustrated, points, total,
233 notation: decimal(points, total, 3), rating};
234}

236class LogAnalyzer {
237 records: LogRecord[] = [];
238 malformed: MalformedLine[] = [];
239 lineCount: number;
240 conditions: Record<string, string | null>;
241 range: FilterRange | null = null;
242 output: string[] = [];
243
244 constructor(lines: string[]) {
245 this.lineCount = lines.length;
246 for (let i = 0; i < lines.length; ++i) {
247 const fields = lines[i].trim().split(/\s+/);
248 const reason = malformedReason(fields);
249 if (reason !== null) {
250 this.malformed.push({line: i + 1, reason});
251 continue;
252 }
253 const [date, clock, ip, method, rawPath, status, bytes, latency] = fields;
254 this.records.push({line: i + 1, date, hour: Number(clock.slice(0, 2)),
255 timestamp: date + ' ' + clock, seconds: secondsOf(date, clock), ip,
256 method, path: normalizePath(rawPath), status: Number(status),
257 size: Number(bytes), latency: Number(latency)});
258 }
259 this.records.sort((a, b) => a.seconds - b.seconds || a.line - b.line);
260 this.conditions = {METHOD: null, STATUS: null, IP: null, PATH: null};
261
262 }
263 emit(...parts: (number | string)[]): void {
264 this.output.push(parts.join(' '));
265 }
266 selected(): LogRecord[] {
267 return this.records.filter(r => {
268 if (this.range && (r.seconds < this.range.startSeconds || r.seconds > this.range.endSeconds)) return false;
269 if (this.conditions.METHOD !== null && r.method !== this.conditions.METHOD) return false;
270 if (this.conditions.STATUS !== null && !statusMatches(this.conditions.STATUS, r.status)) return false;
271 if (this.conditions.IP !== null && r.ip !== this.conditions.IP) return false;
272 return this.conditions.PATH === null || r.path === this.conditions.PATH;
273 });
274 }
275 barLine(label: string, count: number, maximum: number): void {
276 this.emit(label, count, ...(count ? [bar(count, maximum)] : []));
277 }
278 execute(line: string): void {
279 const [name, ...args] = line.trim().split(/\s+/);
280 try {
281 check(Object.hasOwn(FORMS, name), 'unknown query');
282 const form = FORMS[name];
283 const optional = OPTIONAL.has(name) && args.length === form.length - 1;
284 check(args.length === form.length || optional, 'wrong arguments');
285 for (let i = 0; i < args.length; ++i) {
286 check(validArgument(form[i], args[i], args), ERRORS[form[i]]);
287 }
288 const records = this.selected();
289 if (REQUIRE_RECORDS.has(name) && records.length === 0) {
290 this.emit(name, 'no records');
291 return;
292 }
293 this.run(name, args, records);
294 } catch (error) {
295 if (!(error instanceof QueryError)) throw error;
296 this.emit('ERROR', error.message);
297 }
298 }
299 run(name: string, a: string[], records: LogRecord[]): void {
300 switch (name) {
301 case 'SUMMARY': {
302 this.emit('SUMMARY', 'lines', this.lineCount, 'valid', this.records.length,
303 'malformed', this.malformed.length);
304 this.emit('requests', records.length, 'bytes', sum(records, 'size'),
305 'ips', new Set(records.map(r => r.ip)).size,
306 'paths', new Set(records.map(r => r.path)).size);
307 if (!records.length) this.emit('span', 'none');
308 else this.emit('span', records[0].timestamp, records.at(-1)!.timestamp,
309 records.at(-1)!.seconds - records[0].seconds);
310 break;
311 }
312 case 'MALFORMED': {
313 this.emit('MALFORMED', this.malformed.length);
314 for (const reason of REASONS) {
315 const lines = this.malformed.filter(x => x.reason === reason).map(x => x.line);
316 if (lines.length) this.emit(reason, lines.length, ...lines);
317 }
318 break;
319 }
320 case 'STATUS': {
321 this.emit('STATUS', records.length);
322 const classes = statusClassCounts(records);
323 STATUS_CLASSES.forEach((c, i) => this.emit(c, classes[i], percent(classes[i], records.length)));
324 const counts = tallied(records.map(r => r.status));
325 for (const s of [...counts.keys()].sort((x, y) => x - y))
326 this.emit(s, counts.get(s)!, percent(counts.get(s)!, records.length));
327 break;
328 }
329 case 'METHODS': {
330 const counts = tallied(records.map(r => r.method));
331 this.emit('METHODS', records.length);
332 for (const method of sortedByCount(counts))
333 this.emit(method, counts.get(method)!, percent(counts.get(method)!, records.length));
334 break;
335 }
336 case 'TOP': {
337 const [kind, token] = a;
338 const counts = tallied(records.map(r => kind === 'PATHS' ? r.path : r.ip));
339 const keys = sortedByCount(counts, kind === 'IPS' ? compareIp : undefined);
340 const shown = Math.min(Number(token), keys.length);
341 this.emit('TOP', kind, shown, 'of', keys.length);
342 let rank = 0;
343 keys.slice(0, shown).forEach((key, i) => {
344 if (i === 0 || counts.get(key)! !== counts.get(keys[i - 1])) rank = i + 1;
345 this.emit(rank, key, counts.get(key)!, percent(counts.get(key)!, records.length));
346 });
347 break;
348 }
349 case 'HOURLY': {
350 const counts = Array(24).fill(0);
351 for (const r of records) ++counts[r.hour];
352 const maximum = Math.max(...counts);
353 this.emit('HOURLY', records.length, 'peak', String(counts.indexOf(maximum)).padStart(2, '0'), maximum);
354 counts.forEach((count, h) => this.barLine(String(h).padStart(2, '0'), count, maximum));
355 break;
356 }
357 case 'TIMELINE': {
358 const minutes = Number(a[0]), width = minutes * 60;
359 const [first, counts] = bucketCounts(records.map(r => Math.floor(r.seconds / width)));
360 this.emit('TIMELINE', minutes, counts.length);
361 const maximum = Math.max(...counts);
362 counts.forEach((count, i) => this.barLine(intervalLabel((first + i) * width), count, maximum));
363 break;
364 }
365 case 'BANDWIDTH': {
366 const days = grouped(records, 'date'), bytes = sum(records, 'size');
367 this.emit('BANDWIDTH', days.size, bytes, bytesText(bytes));
368 for (const date of [...days.keys()].sort()) {
369 const volume = sum(days.get(date)!, 'size');
370 this.emit(date, days.get(date)!.length, volume, bytesText(volume));
371 }
372 break;
373 }
374 case 'VISITORS': {
375 const first = new Map();
376 for (const r of records) if (!first.has(r.ip) || r.date < first.get(r.ip)) first.set(r.ip, r.date);
377 const days = grouped(records, 'date');
378 this.emit('VISITORS', days.size, first.size);
379 for (const date of [...days.keys()].sort()) {
380 const ips = new Set(days.get(date)!.map(r => r.ip));
381 let newcomers = 0;
382 for (const ip of ips) if (first.get(ip) === date) ++newcomers;
383 this.emit(date, 'unique', ips.size, 'new', newcomers, 'returning', ips.size - newcomers);
384 }
385 break;
386 }
387 case 'LATENCY': {
388 const label = a.length ? normalizePath(a[0]) : 'all';
389 const rows = a.length ? records.filter(r => r.path === label) : records;
390 if (!rows.length) { this.emit('LATENCY', 'no records'); break; }
391 const values = rows.map(r => r.latency).sort((x, y) => x - y);
392 const n = values.length, trim = Math.floor(n / 10);
393 const trimmed = values.slice(trim, n - trim);
394 this.emit('LATENCY', label, 'count', n, 'min', values[0],
395 'p50', percentile(values, 50), 'p90', percentile(values, 90),
396 'p99', percentile(values, 99), 'max', values.at(-1)!,
397 'mean', decimal(values.reduce((v, x) => v + x, 0), n, 2),
398 'trimmed', decimal(trimmed.reduce((v, x) => v + x, 0), trimmed.length, 2));
399 break;
400 }
401 case 'HISTOGRAM': {
402 const width = Number(a[0]);
403 const [first, counts] = bucketCounts(records.map(r => Math.floor(r.latency / width)));
404 this.emit('HISTOGRAM', width, counts.length);
405 const maximum = Math.max(...counts);
406 counts.forEach((count, i) => {
407 const lo = (first + i) * width;
408 this.barLine(lo + '-' + (lo + width - 1), count, maximum);
409 });
410 break;
411 }
412 case 'APDEX': {
413 const threshold = Number(a[0]), overall = apdexScore(records, threshold);
414 this.emit('APDEX', threshold, overall.notation, overall.rating,
415 'satisfied', overall.satisfied, 'tolerating', overall.tolerating,
416 'frustrated', overall.frustrated);
417 const paths = [...grouped(records, 'path')].map(([path, rows]) => ({path, ...apdexScore(rows, threshold)}));
418 paths.sort((x, y) => x.points * y.total - y.points * x.total || y.count - x.count ||
419 (x.path < y.path ? -1 : x.path > y.path ? 1 : 0));
420 for (const row of paths) this.emit(row.path, row.notation, row.rating, row.count);
421 break;
422 }
423 case 'ERRORS': {
424 const failed = records.filter(r => r.status >= 400);
425 this.emit('ERRORS', failed.length, 'of', records.length, percent(failed.length, records.length));
426 const paths = [...grouped(records, 'path')].map(([path, rows]) => ({
427 path, count: rows.length,
428 client: rows.filter(r => Math.floor(r.status / 100) === 4).length,
429 server: rows.filter(r => Math.floor(r.status / 100) === 5).length
430 })).filter(row => row.client + row.server > 0);
431 paths.sort((x, y) => {
432 const xf = x.client + x.server, yf = y.client + y.server;
433 return yf * x.count - xf * y.count || yf - xf || (x.path < y.path ? -1 : 1);
434 });
435 for (const row of paths) this.emit(row.path, row.count, row.client, row.server,
436 percent(row.client + row.server, row.count));
437 break;
438 }
439 case 'SLOWEST': {
440 const groups = grouped(records, r => r.method + '\u0001' + r.path);
441 const rows = [...groups].map(([key, items]) => {
442 const [method, path] = key.split('\u0001');
443 return {method, path, count: items.length,
444 total: sum(items, 'latency'), maximum: Math.max(...items.map(r => r.latency))};
445 });
446 rows.sort((x, y) => y.total * x.count - x.total * y.count || y.maximum - x.maximum ||
447 (x.path < y.path ? -1 : x.path > y.path ? 1 : (x.method < y.method ? -1 : 1)));
448 const shown = Math.min(Number(a[0]), rows.length);
449 this.emit('SLOWEST', shown, 'of', rows.length);
450 rows.slice(0, shown).forEach((row, i) =>
451 this.emit(i + 1, row.method, row.path, 'count', row.count,
452 'avg', decimal(row.total, row.count, 2), 'max', row.maximum));
453 break;
454 }
455 case 'SESSIONS': {
456 const ip = a[0], minutes = a.length > 1 ? Number(a[1]) : 30;
457 const found = records.filter(r => r.ip === ip);
458 if (!found.length) { this.emit('SESSIONS', 'no records'); break; }
459 const sessions = [[found[0]]];
460 for (let i = 1; i < found.length; ++i) {
461 const current = found[i], prev = found[i - 1];
462 if (current.seconds - prev.seconds <= minutes * 60) sessions.at(-1)!.push(current);
463 else sessions.push([current]);
464 }
465 const duration = sessions.reduce((v, session) => v + session.at(-1)!.seconds - session[0].seconds, 0);
466 this.emit('SESSIONS', ip, 'timeout', minutes, 'count', sessions.length,
467 'requests', found.length, 'avg_duration', decimal(duration, sessions.length, 1),
468 'avg_requests', decimal(found.length, sessions.length, 1));
469 sessions.forEach((session, i) => this.emit(i + 1, session[0].timestamp,
470 session.at(-1)!.timestamp, session.at(-1)!.seconds - session[0].seconds,
471 session.length, new Set(session.map(r => r.path)).size));
472 break;
473 }
474 case 'BURST': {
475 const [length, times] = a.map(Number);
476 const ips = grouped(records.filter(r => r.status >= 400), 'ip');
477 const rows = [...ips].map(([ip, items]) => ({ip, ...windowPeak(items, length)}))
478 .filter(x => x.size >= times);
479 rows.sort((x, y) => y.size - x.size || compareIp(x.ip, y.ip));
480 this.emit('BURST', length, times, rows.length);
481 for (const row of rows) this.emit(row.ip, row.size, row.start!.timestamp, row.end!.timestamp);
482 break;
483 }
484 case 'PEAK': {
485 const window = windowPeak(records, Number(a[0]));
486 this.emit('PEAK', Number(a[0]), window.size, window.start!.timestamp, window.end!.timestamp);
487 break;
488 }
489 case 'GAPS': {
490 const length = Number(a[0]);
491 const gaps = [];
492 for (let i = 1; i < records.length; ++i) {
493 if (records[i].seconds - records[i - 1].seconds >= length)
494 gaps.push([records[i - 1], records[i]]);
495 }
496 this.emit('GAPS', length, gaps.length);
497 for (const [prev, curr] of gaps)
498 this.emit(prev.timestamp, curr.timestamp, curr.seconds - prev.seconds);
499 break;
500 }
501 case 'IP': {
502 const ip = a[0], found = records.filter(r => r.ip === ip);
503 if (!found.length) { this.emit('IP', 'no records'); break; }
504 const paths = tallied(found.map(r => r.path));
505 this.emit('IP', ip, 'requests', found.length, 'bytes', sum(found, 'size'),
506 'errors', found.filter(r => r.status >= 400).length, 'paths', paths.size);
507 this.emit('first', found[0].timestamp, 'last', found.at(-1)!.timestamp);
508 const methods = tallied(found.map(r => r.method));
509 this.emit('methods', ...sortedByCount(methods).flatMap(method => [method, methods.get(method)!]));
510 this.emit('classes', ...STATUS_CLASSES.flatMap((label, i) => [label, statusClassCounts(found)[i]]));
511 for (const path of sortedByCount(paths).slice(0, 3)) this.emit('path', path, paths.get(path)!);
512 break;
513 }
514 case 'RANGE': {
515 const start = a[0] + ' ' + a[1], end = a[2] + ' ' + a[3];
516 const startSeconds = secondsOf(a[0], a[1]), endSeconds = secondsOf(a[2], a[3]);
517 check(startSeconds <= endSeconds, 'invalid range');
518 this.range = {start, end, startSeconds, endSeconds};
519 this.emit('RANGE', start, end, 'active', this.selected().length);
520 break;
521 }
522 case 'FILTER': {
523 const [key, raw] = a;
524 const value = key === 'PATH' ? normalizePath(raw) : raw;
525 this.conditions[key] = value;
526 this.emit('FILTER', key, value, 'active', this.selected().length);
527 break;
528 }
529 case 'UNFILTER': {
530 const key = a[0];
531 if (key === 'RANGE') {
532 check(this.range !== null, 'not filtered');
533 this.range = null;
534 } else {
535 check(this.conditions[key] !== null, 'not filtered');
536 this.conditions[key] = null;
537 }
538 this.emit('UNFILTER', key, 'active', this.selected().length);
539 break;
540 }
541 case 'RESET': {
542 this.range = null;
543 for (const key of FILTER_KEYS) this.conditions[key] = null;
544 this.emit('RESET', 'active', this.selected().length);
545 break;
546 }
547 case 'STATE': {
548 const range = this.range === null ? '-' : this.range.start + ' ' + this.range.end;
549 this.emit('STATE', 'active', records.length, 'range', range,
550 ...FILTER_KEYS.flatMap(key => [key.toLowerCase(), this.conditions[key] ?? '-']));
551 break;
552 }
553 }
554 }
555}

564}
