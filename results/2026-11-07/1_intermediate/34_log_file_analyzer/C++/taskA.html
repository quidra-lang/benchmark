73vector<string> split(const string &s, char delimiter) {
74 vector<string> parts;
75 size_t begin = 0;
76 for (size_t i = 0; i <= s.size(); ++i) {
77 if (i == s.size() || s[i] == delimiter) {
78 parts.push_back(s.substr(begin, i - begin));
79 begin = i + 1;
80 }
81 }
82 return parts;
83}

86 return !s.empty() && all_of(s.begin(), s.end(), [](char c) {
87 return c >= '0' && c <= '9';

92 return s.size() <= limit && digits(s) &&
93 (s.size() == 1 || s.front() != '0');

96bool leap_year(int year) {
97 return year % 4 == 0 && (year % 100 != 0 || year % 400 == 0);
98}

100int year_days(int year) {
101 return leap_year(year) ? 366 : 365;

104int month_days(int year, int month) {
105 return month == 2 && leap_year(year) ? 29 : MONTH_DAYS[month - 1];

108bool valid_date(const string &s) {
109 if (s.size() != 10 || s[4] != '-' || s[7] != '-') return false;
110 string value = s.substr(0, 4) + s.substr(5, 2) + s.substr(8, 2);
111 if (!digits(value)) return false;
112 int year = stoi(s.substr(0, 4));
113 int month = stoi(s.substr(5, 2));
114 int day = stoi(s.substr(8, 2));
115 return year >= 2000 && year <= 2059 && month >= 1 && month <= 12 &&
116 day >= 1 && day <= month_days(year, month);
117}
118
119bool valid_time(const string &s) {
120 if (s.size() != 8 || s[2] != ':' || s[5] != ':') return false;
121 if (!digits(s.substr(0, 2) + s.substr(3, 2) + s.substr(6, 2))) return false;
122 return stoi(s.substr(0, 2)) < 24 && stoi(s.substr(3, 2)) < 60 &&
123 stoi(s.substr(6, 2)) < 60;
124}
125
126bool valid_ip(const string &ip) {
127 vector<string> parts = split(ip, '.');
128 if (parts.size() != 4) return false;
129 for (const string &part : parts)
130 if (!canonical(part, 3) || stoi(part) > 255) return false;
131 return true;
132}

134bool valid_status(const string &s) {
135 return s.size() == 3 && digits(s) && stoi(s) >= 100 && stoi(s) <= 599;
136}

138bool contains(const vector<string> &values, const string &s) {
139 return find(values.begin(), values.end(), s) != values.end();

142bool filter_value(const string &key, const string &value) {
143 if (key == "METHOD") return contains(METHODS, value);
144 if (key == "STATUS") return contains(CLASSES, value) || valid_status(value);
145 if (key == "IP") return valid_ip(value);
146 return !value.empty() && value[0] == '/';
147}
148
149bool valid_argument(const string &kind, const string &value,
150 const vector<string> &args) {
151 auto range = RANGES.find(kind);
152 if (range != RANGES.end()) {
153 return canonical(value, 6) && stoi(value) >= range->second.first &&
154 stoi(value) <= range->second.second;
155 }
156 if (kind == "kind") return value == "PATHS" || value == "IPS";
157 if (kind == "path") return !value.empty() && value[0] == '/';
158 if (kind == "ip") return valid_ip(value);
159 if (kind == "date") return valid_date(value);
160 if (kind == "time") return valid_time(value);
161 if (kind == "key") return contains(KEYS, value);
162 if (kind == "subject") return value == "RANGE" || contains(KEYS, value);
163 return filter_value(args[0], value);
164}

166string malformed_reason(const vector<string> &fields) {
167 if (fields.size() != 8) return "fields";
168 if (!valid_date(fields[0])) return "date";
169 if (!valid_time(fields[1])) return "time";
170 if (!valid_ip(fields[2])) return "ip";
171 if (!contains(METHODS, fields[3])) return "method";
172 if (fields[4].empty() || fields[4][0] != '/') return "path";
173 if (!valid_status(fields[5])) return "status";
174 if (!canonical(fields[6], 8)) return "bytes";
175 if (!canonical(fields[7], 6)) return "latency";
176 return "";
177}

179int64_t seconds_of(const string &date, const string &clock) {
180 int year = stoi(date.substr(0, 4));
181 int month = stoi(date.substr(5, 2));
182 int64_t days = stoi(date.substr(8, 2)) - 1;
183 for (int y = 2000; y < year; ++y) days += year_days(y);
184 for (int m = 1; m < month; ++m) days += month_days(year, m);
185 return days * 86400 + stoi(clock.substr(0, 2)) * 3600 +
186 stoi(clock.substr(3, 2)) * 60 + stoi(clock.substr(6, 2));
187}
188
189string interval_label(int64_t seconds) {
190 int64_t days = seconds / 86400;
191 int64_t minutes = seconds % 86400 / 60;
192 int year = 2000;
193 while (days >= year_days(year)) days -= year_days(year++);
194 int month = 1;
195 while (days >= month_days(year, month)) days -= month_days(year, month++);
196 ostringstream out;
197 out << setfill('0') << setw(4) << year << "-" << setw(2) << month
198 << "-" << setw(2) << days + 1 << " "
199 << setw(2) << minutes / 60 << ":" << setw(2) << minutes % 60;
200 return out.str();
201}

203string normalize_path(const string &raw) {
204 string path = raw.substr(0, raw.find('?'));
205 vector<string> parts;
206 for (const string &part : split(path, '/')) {
207 if (part == "..") {
208 if (!parts.empty()) parts.pop_back();
209 } else if (!part.empty() && part != ".") {
210 parts.push_back(part);
211 }
212 }
213 for (string &part : parts) {
214 if (digits(part)) {
215 part = "{id}";
216 } else if (part.size() >= 8 &&
217 all_of(part.begin(), part.end(), [](char c) {
218 return (c >= '0' && c <= '9') || (c >= 'a' && c <= 'f');
219 }) &&
220 any_of(part.begin(), part.end(), [](char c) {
221 return c >= '0' && c <= '9';
222 })) {
223 part = "{hex}";
224 }
225 }
226 string result = "/";
227 for (size_t i = 0; i < parts.size(); ++i) {
228 if (i > 0) result += "/";
229 result += parts[i];
230 }
231 return result;
232}

234bool ip_less(const string &a, const string &b) {
235 vector<string> x = split(a, '.');
236 vector<string> y = split(b, '.');
237 for (size_t i = 0; i < 4; ++i) {
238 if (stoi(x[i]) != stoi(y[i])) return stoi(x[i]) < stoi(y[i]);
239 }
240 return false;
241}

243bool status_matches(const string &filter, int status) {
244 return contains(CLASSES, filter) ? status / 100 == filter[0] - '0'
245 : status == stoi(filter);
246}

248string decimal(int64_t numerator, int64_t denominator, int places) {
249 int64_t scale = places == 1 ? 10 : places == 2 ? 100 : 1000;
250 int64_t value = numerator * scale / denominator;
251 ostringstream out;
252 out << value / scale << "." << setfill('0') << setw(places) << value % scale;
253 return out.str();
254}

258}

264string bytes_text(int64_t size) {
265 if (size >= 1073741824) return decimal(size, 1073741824, 1) + "GB";
266 if (size >= 1048576) return decimal(size, 1048576, 1) + "MB";
267 if (size >= 1024) return decimal(size, 1024, 1) + "KB";
268 return to_string(size) + "B";

301vector<string> by_count(const map<string, int> &counts, bool ip_order = false) {
302 vector<string> names;
303 for (const auto &entry : counts) names.push_back(entry.first);
304 sort(names.begin(), names.end(), [&](const string &a, const string &b) {
305 if (counts.at(a) != counts.at(b)) return counts.at(a) > counts.at(b);
306 return ip_order ? ip_less(a, b) : a < b;
307 });
308 return names;
309}

329struct Buckets {
330 int64_t first;
331 vector<int> counts;
332};
333
334Buckets bucket_counts(const vector<int64_t> &values) {
335 auto low_high = minmax_element(values.begin(), values.end());
336 int64_t first = *low_high.first;
337 int64_t last = *low_high.second;
338 require_value(last - first + 1 <= 60, "too many buckets");
339 Buckets result{first, vector<int>(last - first + 1, 0)};
340 for (int64_t value : values) result.counts[value - first]++;
341 return result;
342}

348struct Window {
349 int count = 0;
350 Record first{}, last{};
351};
352
353Window window_peak(const vector<Record> &records, int width) {
354 Window result;
355 size_t first = 0;
356 for (size_t last = 0; last < records.size(); ++last) {
357 while (records[last].seconds - records[first].seconds >= width) ++first;
358 if (static_cast<int>(last - first + 1) > result.count) {
359 result.count = last - first + 1;
360 result.first = records[first];
361 result.last = records[last];
362 }
363 }
364 return result;
365}

367struct Apdex {
368 int count = 0, satisfied = 0, tolerating = 0, frustrated = 0;
369 int64_t points = 0, total = 0;
370 string notation, rating;
371};
372
373Apdex apdex_score(const vector<Record> &records, int threshold) {
374 Apdex result;
375 result.count = records.size();
376 for (const Record &r : records) {
377 if (r.status >= 500) continue;
378 if (r.latency <= threshold) ++result.satisfied;
379 else if (r.latency <= 4 * threshold) ++result.tolerating;
380 }
381 result.frustrated = result.count - result.satisfied - result.tolerating;
382 result.points = result.satisfied * 2LL + result.tolerating;
383 result.total = result.count * 2LL;
384 result.rating = result.points * 100 >= result.total * 94 ? "excellent" :
385 result.points * 100 >= result.total * 85 ? "good" :
386 result.points * 100 >= result.total * 70 ? "fair" :
387 result.points * 100 >= result.total * 50 ? "poor" : "unacceptable";
388 result.notation = decimal(result.points, result.total, 3);
389 return result;
390}

411 ostringstream out;
412 for (size_t i = 0; i < parts.size(); ++i) {
413 if (i > 0) out << " ";
414 out << parts[i];
415 }
416 output.push_back(out.str());

419 vector<Record> selected() const {
420 vector<Record> result;
421 for (const Record &r : records) {
422 if (has_range && (r.seconds < from || r.seconds > through)) continue;
423 if (!filters.at("METHOD").empty() && filters.at("METHOD") != r.method) continue;
424 if (!filters.at("STATUS").empty() &&
425 !status_matches(filters.at("STATUS"), r.status)) continue;
426 if (!filters.at("IP").empty() && filters.at("IP") != r.ip) continue;
427 if (!filters.at("PATH").empty() && filters.at("PATH") != r.path) continue;
428 result.push_back(r);
429 }
430 return result;
431 }

434 if (count == 0) emit(label, count);
435 else emit(label, count, bar(count, maximum));

437
438 void run(const string &name, const vector<string> &a, const vector<Record> &rows) {
439 if (name == "SUMMARY") {
440 set<string> ips, paths;
441 for (const Record &r : rows) {
442 ips.insert(r.ip);
443 paths.insert(r.path);
444 }
445 emit("SUMMARY", "lines", line_count, "valid", records.size(),
446 "malformed", malformed.size());
447 emit("requests", rows.size(), "bytes", sum_bytes(rows),
448 "ips", ips.size(), "paths", paths.size());
449 if (rows.empty()) emit("span", "none");
450 else emit("span", rows.front().timestamp, rows.back().timestamp,
451 rows.back().seconds - rows.front().seconds);
452
453 } else if (name == "MALFORMED") {
454 emit("MALFORMED", malformed.size());
455 for (const string &reason : REASONS) {
456 vector<string> parts = {reason};
457 int n = 0;
458 for (const auto &row : malformed) {
459 if (row.second == reason) {
460 parts.push_back(to_string(row.first));
461 ++n;
462 }
463 }
464 if (n > 0) {
465 parts.insert(parts.begin() + 1, to_string(n));
466 emit_vector(parts);
467 }
468 }
469
470 } else if (name == "STATUS") {
471 emit("STATUS", rows.size());
472 auto class_count = class_counts(rows);
473 for (int i = 0; i < 5; ++i)
474 emit(CLASSES[i], class_count[i], percent(class_count[i], rows.size()));
475 map<int, int> status;
476 for (const Record &r : rows) status[r.status]++;
477 for (const auto &entry : status)
478 emit(entry.first, entry.second, percent(entry.second, rows.size()));
479
480 } else if (name == "METHODS") {
481 emit("METHODS", rows.size());
482 auto counts = count_by(rows, [](const Record &r) { return r.method; });
483 for (const string &method : by_count(counts))
484 emit(method, counts[method], percent(counts[method], rows.size()));
485
486 } else if (name == "TOP") {
487 const string &kind = a[0];
488 auto counts = count_by(rows, [&](const Record &r) {
489 return kind == "PATHS" ? r.path : r.ip;
490 });
491 vector<string> names = by_count(counts, kind == "IPS");
492 int shown = min(stoi(a[1]), static_cast<int>(names.size()));
493 emit("TOP", kind, shown, "of", names.size());
494 int rank = 0;
495 for (int i = 0; i < shown; ++i) {
496 if (i == 0 || counts[names[i]] != counts[names[i-1]]) rank = i+1;
497 emit(rank, names[i], counts[names[i]], percent(counts[names[i]], rows.size()));
498 }
499
500 } else if (name == "HOURLY") {
501 array<int, 24> counts{};
502 for (const Record &r : rows) counts[r.hour]++;
503 int maximum = *max_element(counts.begin(), counts.end());
504 int first = find(counts.begin(), counts.end(), maximum) - counts.begin();
505 ostringstream hour;
506 hour << setfill('0') << setw(2) << first;
507 emit("HOURLY", rows.size(), "peak", hour.str(), maximum);
508 for (int i = 0; i < 24; ++i) {
509 ostringstream label;
510 label << setfill('0') << setw(2) << i;
511 bar_line(label.str(), counts[i], maximum);
512 }
513
514 } else if (name == "TIMELINE") {
515 int minutes = stoi(a[0]);
516 int64_t width = minutes * 60LL;
517 vector<int64_t> bucket_numbers;
518 for (const Record &r : rows) bucket_numbers.push_back(r.seconds / width);
519 Buckets buckets = bucket_counts(bucket_numbers);
520 emit("TIMELINE", minutes, buckets.counts.size());
521 int maximum = *max_element(buckets.counts.begin(), buckets.counts.end());
522 for (size_t i = 0; i < buckets.counts.size(); ++i)
523 bar_line(interval_label((buckets.first + i) * width),
524 buckets.counts[i], maximum);
525
526 } else if (name == "BANDWIDTH") {
527 auto days = group_by(rows, [](const Record &r) { return r.date; });
528 int64_t total = sum_bytes(rows);
529 emit("BANDWIDTH", days.size(), total, bytes_text(total));
530 for (const auto &entry : days) {
531 int64_t bytes = sum_bytes(entry.second);
532 emit(entry.first, entry.second.size(), bytes, bytes_text(bytes));
533 }
534
535 } else if (name == "VISITORS") {
536 map<string, string> first_date;
537 for (const Record &r : rows) {
538 if (first_date.find(r.ip) == first_date.end() || r.date < first_date[r.ip])
539 first_date[r.ip] = r.date;
540 }
541 auto days = group_by(rows, [](const Record &r) { return r.date; });
542 emit("VISITORS", days.size(), first_date.size());
543 for (const auto &entry : days) {
544 set<string> ips;
545 for (const Record &r : entry.second) ips.insert(r.ip);
546 int fresh = 0;
547 for (const string &ip : ips)
548 if (first_date[ip] == entry.first) ++fresh;
549 emit(entry.first, "unique", ips.size(), "new", fresh,
550 "returning", ips.size() - fresh);
551 }
552
553 } else if (name == "LATENCY") {
554 string label = a.empty() ? "all" : normalize_path(a[0]);
555 vector<int> values;
556 for (const Record &r : rows)
557 if (a.empty() || r.path == label) values.push_back(r.latency);
558 if (values.empty()) {
559 emit("LATENCY", "no records");
560 } else {
561 sort(values.begin(), values.end());
562 int n = values.size(), trim = n / 10;
563 int64_t total = accumulate(values.begin(), values.end(), int64_t{0});
564 int64_t trimmed_total = accumulate(
565 values.begin() + trim, values.end() - trim, int64_t{0});
566 emit("LATENCY", label, "count", n, "min", values.front(),
567 "p50", percentile(values, 50), "p90", percentile(values, 90),
568 "p99", percentile(values, 99), "max", values.back(),
569 "mean", decimal(total, n, 2),
570 "trimmed", decimal(trimmed_total, n - trim * 2, 2));
571 }
572
573 } else if (name == "HISTOGRAM") {
574 int width = stoi(a[0]);
575 vector<int64_t> bucket_numbers;
576 for (const Record &r : rows) bucket_numbers.push_back(r.latency / width);
577 Buckets buckets = bucket_counts(bucket_numbers);
578 emit("HISTOGRAM", width, buckets.counts.size());
579 int maximum = *max_element(buckets.counts.begin(), buckets.counts.end());
580 for (size_t i = 0; i < buckets.counts.size(); ++i) {
581 int64_t low = (buckets.first + i) * width;
582 bar_line(to_string(low) + "-" + to_string(low + width - 1),
583 buckets.counts[i], maximum);
584 }
585
586 } else if (name == "APDEX") {
587 int threshold = stoi(a[0]);
588 Apdex overall = apdex_score(rows, threshold);
589 emit("APDEX", threshold, overall.notation, overall.rating,
590 "satisfied", overall.satisfied, "tolerating", overall.tolerating,
591 "frustrated", overall.frustrated);
592 struct Item { string path; Apdex score; };
593 vector<Item> paths;
594 auto groups = group_by(rows, [](const Record &r) { return r.path; });
595 for (const auto &entry : groups)
596 paths.push_back({entry.first, apdex_score(entry.second, threshold)});
597 sort(paths.begin(), paths.end(), [](const Item &a, const Item &b) {
598 int64_t left = a.score.points * b.score.total;
599 int64_t right = b.score.points * a.score.total;
600 if (left != right) return left < right;
601 if (a.score.count != b.score.count)
602 return a.score.count > b.score.count;
603 return a.path < b.path;
604 });
605 for (const Item &item : paths)
606 emit(item.path, item.score.notation, item.score.rating, item.score.count);
607
608 } else if (name == "ERRORS") {
609 int failed = 0;
610 for (const Record &r : rows) if (r.status >= 400) ++failed;
611 emit("ERRORS", failed, "of", rows.size(), percent(failed, rows.size()));
612 struct Item { string path; int count, client, server; };
613 vector<Item> paths;
614 auto groups = group_by(rows, [](const Record &r) { return r.path; });
615 for (const auto &entry : groups) {
616 int client = 0, server = 0;
617 for (const Record &r : entry.second) {
618 if (r.status / 100 == 4) ++client;
619 if (r.status / 100 == 5) ++server;
620 }
621 if (client + server > 0)
622 paths.push_back({entry.first, static_cast<int>(entry.second.size()),
623 client, server});
624 }
625 sort(paths.begin(), paths.end(), [](const Item &a, const Item &b) {
626 int64_t af = a.client + a.server, bf = b.client + b.server;
627 if (bf * a.count != af * b.count)
628 return bf * a.count < af * b.count;
629 if (af != bf) return bf > af;
630 return a.path < b.path;
631 });
632 for (const Item &row : paths)
633 emit(row.path, row.count, row.client, row.server,
634 percent(row.client + row.server, row.count));
635
636 } else if (name == "SLOWEST") {
637 struct Item {
638 string method, path;
639 int count, maximum;
640 int64_t total;
641 };
642 vector<Item> endpoints;
643 auto groups = group_by(rows, [](const Record &r) {
644 return r.method + "\1" + r.path;
645 });
646 for (const auto &entry : groups) {
647 size_t point = entry.first.find('\1');
648 int maximum = 0;
649 for (const Record &r : entry.second)
650 maximum = max(maximum, r.latency);
651 endpoints.push_back({
652 entry.first.substr(0, point), entry.first.substr(point + 1),
653 static_cast<int>(entry.second.size()), maximum, sum_latency(entry.second)
654 });
655 }
656 sort(endpoints.begin(), endpoints.end(), [](const Item &a, const Item &b) {
657 if (b.total * a.count != a.total * b.count)
658 return b.total * a.count < a.total * b.count;
659 if (a.maximum != b.maximum) return a.maximum > b.maximum;
660 if (a.path != b.path) return a.path < b.path;
661 return a.method < b.method;
662 });
663 int shown = min(stoi(a[0]), static_cast<int>(endpoints.size()));
664 emit("SLOWEST", shown, "of", endpoints.size());
665 for (int i = 0; i < shown; ++i) {
666 const Item &e = endpoints[i];
667 emit(i+1, e.method, e.path, "count", e.count,
668 "avg", decimal(e.total, e.count, 2), "max", e.maximum);
669 }
670
671 } else if (name == "SESSIONS") {
672 string ip = a[0];
673 int minutes = a.size() == 2 ? stoi(a[1]) : 30;
674 vector<Record> found;
675 for (const Record &r : rows) if (r.ip == ip) found.push_back(r);
676 if (found.empty()) {
677 emit("SESSIONS", "no records");
678 } else {
679 vector<vector<Record>> sessions{{found.front()}};
680 for (size_t i = 1; i < found.size(); ++i) {
681 if (found[i].seconds - found[i-1].seconds <= minutes * 60)
682 sessions.back().push_back(found[i]);
683 else sessions.push_back({found[i]});
684 }
685 int64_t duration = 0;
686 for (const auto &session : sessions)
687 duration += session.back().seconds - session.front().seconds;
688 emit("SESSIONS", ip, "timeout", minutes, "count", sessions.size(),
689 "requests", found.size(),
690 "avg_duration", decimal(duration, sessions.size(), 1),
691 "avg_requests", decimal(found.size(), sessions.size(), 1));
692 for (size_t i = 0; i < sessions.size(); ++i) {
693 const auto &session = sessions[i];
694 set<string> paths;
695 for (const Record &r : session) paths.insert(r.path);
696 emit(i+1, session.front().timestamp, session.back().timestamp,
697 session.back().seconds - session.front().seconds,
698 session.size(), paths.size());
699 }
700 }
701
702 } else if (name == "BURST") {
703 int width = stoi(a[0]), threshold = stoi(a[1]);
704 vector<Record> failed;
705 for (const Record &r : rows)
706 if (r.status >= 400) failed.push_back(r);
707 auto groups = group_by(failed, [](const Record &r) { return r.ip; });
708 struct Item { string ip; Window window; };
709 vector<Item> bursts;
710 for (const auto &entry : groups) {
711 Window peak = window_peak(entry.second, width);
712 if (peak.count >= threshold) bursts.push_back({entry.first, peak});
713 }
714 sort(bursts.begin(), bursts.end(), [](const Item &a, const Item &b) {
715 if (a.window.count != b.window.count)
716 return a.window.count > b.window.count;
717 return ip_less(a.ip, b.ip);
718 });
719 emit("BURST", width, threshold, bursts.size());
720 for (const Item &item : bursts)
721 emit(item.ip, item.window.count,
722 item.window.first.timestamp, item.window.last.timestamp);
723
724 } else if (name == "PEAK") {
725 int width = stoi(a[0]);
726 Window peak = window_peak(rows, width);
727 emit("PEAK", width, peak.count,
728 peak.first.timestamp, peak.last.timestamp);
729
730 } else if (name == "GAPS") {
731 int width = stoi(a[0]);
732 vector<pair<Record, Record>> gaps;
733 for (size_t i = 1; i < rows.size(); ++i) {
734 if (rows[i].seconds - rows[i-1].seconds >= width)
735 gaps.push_back({rows[i-1], rows[i]});
736 }
737 emit("GAPS", width, gaps.size());
738 for (const auto &gap : gaps)
739 emit(gap.first.timestamp, gap.second.timestamp,
740 gap.second.seconds - gap.first.seconds);
741
742 } else if (name == "IP") {
743 string ip = a[0];
744 vector<Record> found;
745 for (const Record &r : rows) if (r.ip == ip) found.push_back(r);
746 if (found.empty()) {
747 emit("IP", "no records");
748 } else {
749 auto paths = count_by(found, [](const Record &r) { return r.path; });
750 int failures = 0;
751 for (const Record &r : found) if (r.status >= 400) ++failures;
752 emit("IP", ip, "requests", found.size(), "bytes", sum_bytes(found),
753 "errors", failures, "paths", paths.size());
754 emit("first", found.front().timestamp, "last", found.back().timestamp);
755 auto methods = count_by(found, [](const Record &r) { return r.method; });
756 vector<string> method_parts{"methods"};
757 for (const string &method : by_count(methods)) {
758 method_parts.push_back(method);
759 method_parts.push_back(to_string(methods[method]));
760 }
761 emit_vector(method_parts);
762 auto grouped_classes = class_counts(found);
763 vector<string> class_parts{"classes"};
764 for (size_t i = 0; i < CLASSES.size(); ++i) {
765 class_parts.push_back(CLASSES[i]);
766 class_parts.push_back(to_string(grouped_classes[i]));
767 }
768 emit_vector(class_parts);
769 auto ranking = by_count(paths);
770 for (size_t i = 0; i < ranking.size() && i < 3; ++i)
771 emit("path", ranking[i], paths[ranking[i]]);
772 }
773
774 } else if (name == "RANGE") {
775 int64_t first = seconds_of(a[0], a[1]);
776 int64_t last = seconds_of(a[2], a[3]);
777 require_value(first <= last, "invalid range");
778 from = first;
779 through = last;
780 from_label = a[0] + " " + a[1];
781 through_label = a[2] + " " + a[3];
782 has_range = true;
783 emit("RANGE", from_label, through_label, "active", selected().size());
784
785 } else if (name == "FILTER") {
786 const string &key = a[0];
787 string value = key == "PATH" ? normalize_path(a[1]) : a[1];
788 filters[key] = value;
789 emit("FILTER", key, value, "active", selected().size());
790
791 } else if (name == "UNFILTER") {
792 const string &key = a[0];
793 if (key == "RANGE") {
794 require_value(has_range, "not filtered");
795 has_range = false;
796 } else {
797 require_value(!filters[key].empty(), "not filtered");
798 filters[key].clear();
799 }
800 emit("UNFILTER", key, "active", selected().size());
801
802 } else if (name == "RESET") {
803 has_range = false;
804 for (const string &key : KEYS) filters[key].clear();
805 emit("RESET", "active", selected().size());
806
807 } else if (name == "STATE") {
808 vector<string> parts{
809 "STATE", "active", to_string(rows.size()), "range",
810 has_range ? from_label + " " + through_label : "-"
811 };
812 for (const string &key : KEYS) {
813 string lowered = key;
814 transform(lowered.begin(), lowered.end(), lowered.begin(),
815 [](unsigned char c) { return tolower(c); });
816 parts.push_back(lowered);
817 parts.push_back(filters[key].empty() ? "-" : filters[key]);
818 }
819 emit_vector(parts);
820 }
821 }
822
823public:
824 explicit Analyzer(const vector<string> &lines) : line_count(lines.size()) {
825 for (size_t i = 0; i < lines.size(); ++i) {
826 vector<string> fields = words(lines[i]);
827 string reason = malformed_reason(fields);
828 if (!reason.empty()) malformed.push_back({static_cast<int>(i+1), reason});
829 else records.push_back(make_record(i+1, fields));
830 }
831 sort(records.begin(), records.end(), [](const Record &a, const Record &b) {
832 return a.seconds < b.seconds ||
833 (a.seconds == b.seconds && a.line < b.line);
834 });
835 for (const string &key : KEYS) filters[key] = "";
836 }
837
838 void execute(const string &line) {
839 vector<string> parts = words(line);
840 string name = parts.empty() ? "" : parts.front();
841 vector<string> args;
842 if (!parts.empty()) args.assign(parts.begin()+1, parts.end());
843 try {
844 auto form = FORMS.find(name);
845 require_value(form != FORMS.end(), "unknown query");
846 const auto &expected = form->second;
847 bool optional = (name == "LATENCY" || name == "SESSIONS") &&
848 args.size() + 1 == expected.size();
849 require_value(args.size() == expected.size() || optional, "wrong arguments");
850 for (size_t i = 0; i < args.size(); ++i)
851 require_value(valid_argument(expected[i], args[i], args),
852 ERRORS.at(expected[i]));
853 vector<Record> rows = selected();
854 if (REQUIRED.count(name) && rows.empty()) emit(name, "no records");
855 else run(name, args, rows);
856 } catch (const QueryError &error) {
857 emit("ERROR", error.what());
858 }
859 }
860
861 void finish() const {
862 for (const string &line : output) cout << line << '\n';
863 }
864};

866int main() {
867 ios::sync_with_stdio(false);
868 cin.tie(nullptr);
869
870 vector<string> logs, queries;
871 bool query_part = false;
872 string line;
873 while (getline(cin, line)) {
874 if (!query_part) {
875 if (line == "---") query_part = true;
876 else logs.push_back(line);
877 } else if (line == "END") {
878 break;
879 } else {
880 queries.push_back(line);
881 }
882 }
883
884 Analyzer analyzer(logs);
885 for (const string &query : queries) analyzer.execute(query);
886 analyzer.finish();
887}
