string decimal(int64_t numerator, int64_t denominator, int places) {
    int64_t scale = places == 1 ? 10 : places == 2 ? 100 : 1000;
    int64_t value = numerator * scale / denominator;
    ostringstream out;
    out << value / scale << "." << setfill('0') << setw(places) << value % scale;
    return out.str();
