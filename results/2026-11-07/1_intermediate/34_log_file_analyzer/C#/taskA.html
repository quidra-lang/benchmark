95 static string[] Tokens(string line)
96 {
97 string s = line.Trim();
98 return s.Length == 0 ? Array.Empty<string>() : Regex.Split(s, @"\s+");
99 }
100
101 static bool Canonical(string s, int maximum)
102 {
103 return s.Length >= 1 && s.Length <= maximum &&
104 s.All(c => c >= '0' && c <= '9') && (s.Length == 1 || s[0] != '0');
105 }
106
107 static bool Leap(int year)
108 {
109 return year % 4 == 0 && (year % 100 != 0 || year % 400 == 0);
110 }
111
112 static int DaysInYear(int year) { return Leap(year) ? 366 : 365; }
113
114 static int DaysInMonth(int year, int month)
115 {
116 return month == 2 && Leap(year) ? 29 : MonthDays[month - 1];
117 }
118
119 static bool ValidDate(string s)
120 {
121 if (!Regex.IsMatch(s, @"^[0-9]{4}-[0-9]{2}-[0-9]{2}$")) return false;
122 int year = int.Parse(s.Substring(0, 4), CultureInfo.InvariantCulture);
123 int month = int.Parse(s.Substring(5, 2), CultureInfo.InvariantCulture);
124 int day = int.Parse(s.Substring(8, 2), CultureInfo.InvariantCulture);
125 return year >= 2000 && year <= 2059 && month >= 1 && month <= 12 &&
126 day >= 1 && day <= DaysInMonth(year, month);
127 }
128
129 static bool ValidTime(string s)
130 {
131 if (!Regex.IsMatch(s, @"^[0-9]{2}:[0-9]{2}:[0-9]{2}$")) return false;
132 return int.Parse(s.Substring(0, 2), CultureInfo.InvariantCulture) < 24 &&
133 int.Parse(s.Substring(3, 2), CultureInfo.InvariantCulture) < 60 &&
134 int.Parse(s.Substring(6, 2), CultureInfo.InvariantCulture) < 60;
135 }
136
137 static bool ValidIp(string s)
138 {
139 string[] parts = s.Split('.');
140 return parts.Length == 4 && parts.All(p =>
141 Canonical(p, 3) && int.Parse(p, CultureInfo.InvariantCulture) <= 255);
142 }
143
144 static bool ValidStatus(string s)
145 {
146 return Regex.IsMatch(s, @"^[0-9]{3}$") &&
147 int.Parse(s, CultureInfo.InvariantCulture) >= 100 &&
148 int.Parse(s, CultureInfo.InvariantCulture) <= 599;
149 }
150
151 static bool FilterValue(string kind, string value)
152 {
153 if (kind == "METHOD") return Methods.Contains(value);
154 if (kind == "STATUS") return Classes.Contains(value) || ValidStatus(value);
155 if (kind == "IP") return ValidIp(value);
156 return value.StartsWith("/", StringComparison.Ordinal);
157 }
158
159 static bool ValidArgument(string kind, string s, string[] args)
160 {
161 if (Ranges.TryGetValue(kind, out var range))
162 {
163 if (!Canonical(s, 6)) return false;
164 int n = int.Parse(s, CultureInfo.InvariantCulture);
165 return n >= range.Lower && n <= range.Upper;
166 }
167 switch (kind)
168 {
169 case "kind": return s == "PATHS" || s == "IPS";
170 case "path": return s.StartsWith("/", StringComparison.Ordinal);
171 case "ip": return ValidIp(s);
172 case "date": return ValidDate(s);
173 case "time": return ValidTime(s);
174 case "key": return Keys.Contains(s);
175 case "subject": return s == "RANGE" || Keys.Contains(s);
176 default: return FilterValue(args[0], s);
177 }
178 }
179
180 static string? Malformed(string[] fields)
181 {
182 if (fields.Length != 8) return "fields";
183 if (!ValidDate(fields[0])) return "date";
184 if (!ValidTime(fields[1])) return "time";
185 if (!ValidIp(fields[2])) return "ip";
186 if (!Methods.Contains(fields[3])) return "method";
187 if (!fields[4].StartsWith("/", StringComparison.Ordinal)) return "path";
188 if (!ValidStatus(fields[5])) return "status";
189 if (!Canonical(fields[6], 8)) return "bytes";
190 if (!Canonical(fields[7], 6)) return "latency";
191 return null;
192 }

194 static long SecondsOf(string date, string clock)
195 {
196 int year = int.Parse(date.Substring(0, 4), CultureInfo.InvariantCulture);
197 int month = int.Parse(date.Substring(5, 2), CultureInfo.InvariantCulture);
198 int day = int.Parse(date.Substring(8, 2), CultureInfo.InvariantCulture);
199 long days = day - 1;
200 for (int y = 2000; y < year; y++) days += DaysInYear(y);
201 for (int m = 1; m < month; m++) days += DaysInMonth(year, m);
202 return days * 86400 + int.Parse(clock.Substring(0, 2), CultureInfo.InvariantCulture) * 3600 +
203 int.Parse(clock.Substring(3, 2), CultureInfo.InvariantCulture) * 60 +
204 int.Parse(clock.Substring(6, 2), CultureInfo.InvariantCulture);
205 }
206
207 static string IntervalLabel(long seconds)
208 {
209 long days = seconds / 86400, rest = seconds % 86400;
210 int year = 2000, month = 1;
211 while (days >= DaysInYear(year)) days -= DaysInYear(year++);
212 while (days >= DaysInMonth(year, month)) days -= DaysInMonth(year, month++);
213 return $"{year:D4}-{month:D2}-{days + 1:D2} {rest / 3600:D2}:{rest % 3600 / 60:D2}";
214 }
215
216 static string NormalizePath(string raw)
217 {
218 List<string> parts = new List<string>();
219 foreach (string part in raw.Split('?')[0].Split('/'))
220 {
221 if (part == "..")
222 {
223 if (parts.Count > 0) parts.RemoveAt(parts.Count - 1);
224 }
225 else if (part != "" && part != ".") parts.Add(part);
226 }
227 for (int i = 0; i < parts.Count; i++)
228 {
229 if (parts[i].All(c => c >= '0' && c <= '9')) parts[i] = "{id}";
230 else if (parts[i].Length >= 8 &&
231 parts[i].All(c => c >= '0' && c <= '9' || c >= 'a' && c <= 'f') &&
232 parts[i].Any(c => c >= '0' && c <= '9')) parts[i] = "{hex}";
233 }
234 return "/" + string.Join("/", parts);
235 }

237 static int CompareIp(string a, string b)
238 {
239 int[] x = a.Split('.').Select(int.Parse).ToArray();
240 int[] y = b.Split('.').Select(int.Parse).ToArray();
241 for (int i = 0; i < 4; i++) if (x[i] != y[i]) return x[i].CompareTo(y[i]);
242 return 0;
243 }

245 static bool StatusMatches(string filter, int status)
246 {
247 return Classes.Contains(filter) ? status / 100 == filter[0] - '0' :
248 status == int.Parse(filter, CultureInfo.InvariantCulture);
249 }
250
251 static string Decimal(long numerator, long denominator, int places)
252 {
253 long scale = places == 1 ? 10 : places == 2 ? 100 : 1000;
254 long n = numerator * scale / denominator;
255 return (n / scale).ToString(CultureInfo.InvariantCulture) + "." +
256 (n % scale).ToString(CultureInfo.InvariantCulture).PadLeft(places, '0');
257 }
258
259 static string Percent(long numerator, long denominator)
260 {
261 return Decimal(numerator * 100, denominator, 2) + "%";
262 }
263
264 static string Bar(int count, int maximum)
265 {
266 return new string('#', (count * 30 + maximum - 1) / maximum);
267 }
268
269 static string BytesText(long bytes)
270 {
271 if (bytes >= 1073741824) return Decimal(bytes, 1073741824, 1) + "GB";
272 if (bytes >= 1048576) return Decimal(bytes, 1048576, 1) + "MB";
273 if (bytes >= 1024) return Decimal(bytes, 1024, 1) + "KB";
274 return bytes.ToString(CultureInfo.InvariantCulture) + "B";
275 }

277 static List<string> ByCount(Dictionary<string, int> counts, Comparison<string>? comparison = null)
278 {
279 var result = counts.Keys.ToList();
280 comparison ??= (a, b) => StringComparer.Ordinal.Compare(a, b);
281 result.Sort((a, b) =>
282 {
283 int n = counts[b].CompareTo(counts[a]);
284 return n != 0 ? n : comparison(a, b);
285 });
286 return result;
287 }

289 static Dictionary<string, int> CountStrings(IEnumerable<string> values)
290 {
291 Dictionary<string, int> counts = new Dictionary<string, int>(StringComparer.Ordinal);
292 foreach (string value in values)
293 {
294 counts.TryGetValue(value, out int count);
295 counts[value] = count + 1;
296 }
297 return counts;

300 static Dictionary<string, List<LogRecord>> Groups(IEnumerable<LogRecord> rows,
301 Func<LogRecord, string> key)
302 {
303 var result = new Dictionary<string, List<LogRecord>>(StringComparer.Ordinal);
304 foreach (LogRecord row in rows)
305 {
306 string name = key(row);
307 if (!result.TryGetValue(name, out var list)) result[name] = list = new List<LogRecord>();
308 list.Add(row);
309 }
310 return result;
311 }

313 static int[] ClassCounts(IEnumerable<LogRecord> rows)
314 {
315 int[] counts = new int[5];
316 foreach (LogRecord row in rows) counts[row.Status / 100 - 1]++;
317 return counts;

320 static (long First, int[] Counts) Buckets(IEnumerable<long> values)
321 {
322 long[] numbers = values.ToArray();
323 long first = numbers.Min(), last = numbers.Max();
324 Check(last - first + 1 <= 60, "too many buckets");
325 int[] counts = new int[(int)(last - first + 1)];
326 foreach (long n in numbers) counts[(int)(n - first)]++;
327 return (first, counts);
328 }

330 static int Percentile(int[] values, int rank)
331 {
332 return values[(rank * values.Length + 99) / 100 - 1];
333 }

335 static WindowResult Window(List<LogRecord> records, int length)
336 {
337 WindowResult result = new WindowResult();
338 int first = 0;
339 for (int last = 0; last < records.Count; last++)
340 {
341 while (records[last].Seconds - records[first].Seconds >= length) first++;
342 if (last - first + 1 > result.Count)
343 {
344 result.Count = last - first + 1;
345 result.Start = records[first];
346 result.End = records[last];
347 }
348 }
349 return result;
350 }

352 static ApdexResult Apdex(List<LogRecord> records, int threshold)
353 {
354 var r = new ApdexResult();
355 r.Count = records.Count;
356 r.Satisfied = records.Count(x => x.Status < 500 && x.Latency <= threshold);
357 r.Tolerating = records.Count(x => x.Status < 500 &&
358 x.Latency > threshold && x.Latency <= 4 * threshold);
359 r.Frustrated = r.Count - r.Satisfied - r.Tolerating;
360 r.Points = 2L * r.Satisfied + r.Tolerating;
361 r.Total = 2L * r.Count;
362 r.Rating = r.Points * 100 >= 94 * r.Total ? "excellent" :
363 r.Points * 100 >= 85 * r.Total ? "good" :
364 r.Points * 100 >= 70 * r.Total ? "fair" :
365 r.Points * 100 >= 50 * r.Total ? "poor" : "unacceptable";
366 r.Notation = Decimal(r.Points, r.Total, 3);
367 return r;
368 }

370 sealed class LogAnalyzer
371 {
372 readonly List<LogRecord> records = new List<LogRecord>();
373 readonly List<(int Line, string Reason)> malformed = new List<(int, string)>();
374 readonly int lineCount;
375 readonly Dictionary<string, string?> filters = new Dictionary<string, string?>(StringComparer.Ordinal);
376 long? from, through;
377 string? fromLabel, throughLabel;
378 public readonly List<string> Output = new List<string>();
379
380 public LogAnalyzer(List<string> lines)
381 {
382 lineCount = lines.Count;
383 for (int i = 0; i < lines.Count; i++)
384 {
385 string[] fields = Tokens(lines[i]);
386 string? reason = Malformed(fields);
387 if (reason != null)
388 {
389 malformed.Add((i + 1, reason));
390 continue;
391 }
392 records.Add(new LogRecord
393 {
394 Line = i + 1, Date = fields[0],
395 Hour = int.Parse(fields[1].Substring(0, 2), CultureInfo.InvariantCulture),
396 Timestamp = fields[0] + " " + fields[1],
397 Seconds = SecondsOf(fields[0], fields[1]), Ip = fields[2],
398 Method = fields[3], Path = NormalizePath(fields[4]),
399 Status = int.Parse(fields[5], CultureInfo.InvariantCulture),
400 Bytes = long.Parse(fields[6], CultureInfo.InvariantCulture),
401 Latency = int.Parse(fields[7], CultureInfo.InvariantCulture)
402 });
403 }
404 records.Sort((a, b) =>
405 {
406 int n = a.Seconds.CompareTo(b.Seconds);
407 return n != 0 ? n : a.Line.CompareTo(b.Line);
408 });
409 foreach (string key in Keys) filters[key] = null;
410 }
411
412 void Emit(params object[] parts)
413 {
414 Output.Add(string.Join(" ", parts.Select(p =>
415 Convert.ToString(p, CultureInfo.InvariantCulture) ?? "")));
416 }
417
418 List<LogRecord> Selected()
419 {
420 return records.Where(r =>
421 (!from.HasValue || r.Seconds >= from.Value && r.Seconds <= through!.Value) &&
422 (filters["METHOD"] == null || r.Method == filters["METHOD"]) &&
423 (filters["STATUS"] == null || StatusMatches(filters["STATUS"]!, r.Status)) &&
424 (filters["IP"] == null || r.Ip == filters["IP"]) &&
425 (filters["PATH"] == null || r.Path == filters["PATH"])).ToList();
426 }
427
428 void BarLine(string label, int count, int maximum)
429 {
430 if (count == 0) Emit(label, count);
431 else Emit(label, count, Bar(count, maximum));
432 }
433
434 public void Execute(string line)
435 {
436 string[] tokens = Tokens(line);
437 string name = tokens.Length == 0 ? "" : tokens[0];
438 string[] args = tokens.Skip(1).ToArray();
439 try
440 {
441 Check(Forms.TryGetValue(name, out var form), "unknown query");
442 form = Forms[name];
443 bool optional = (name == "LATENCY" || name == "SESSIONS") &&
444 args.Length == form.Length - 1;
445 Check(args.Length == form.Length || optional, "wrong arguments");
446 for (int i = 0; i < args.Length; i++)
447 Check(ValidArgument(form[i], args[i], args), Errors[form[i]]);
448 List<LogRecord> selected = Selected();
449 if (Required.Contains(name) && selected.Count == 0)
450 {
451 Emit(name, "no records");
452 return;
453 }
454 Run(name, args, selected);
455 }
456 catch (QueryError error)
457 {
458 Emit("ERROR", error.Message);
459 }
460 }
461
462 void Run(string name, string[] args, List<LogRecord> rows)
463 {
464 switch (name)
465 {
466 case "SUMMARY":
467 Emit("SUMMARY", "lines", lineCount, "valid", records.Count,
468 "malformed", malformed.Count);
469 Emit("requests", rows.Count, "bytes", rows.Sum(r => r.Bytes),
470 "ips", rows.Select(r => r.Ip).Distinct().Count(),
471 "paths", rows.Select(r => r.Path).Distinct().Count());
472 if (rows.Count == 0) Emit("span", "none");
473 else Emit("span", rows[0].Timestamp, rows[^1].Timestamp,
474 rows[^1].Seconds - rows[0].Seconds);
475 break;
476 case "MALFORMED":
477 Emit("MALFORMED", malformed.Count);
478 foreach (string reason in Reasons)
479 {
480 var matches = malformed.Where(m => m.Reason == reason)
481 .Select(m => m.Line).ToArray();
482 if (matches.Length != 0)
483 Emit(new object[] {reason, matches.Length}
484 .Concat(matches.Cast<object>()).ToArray());
485 }
486 break;
487 case "STATUS":
488 {
489 Emit("STATUS", rows.Count);
490 int[] classes = ClassCounts(rows);
491 for (int i = 0; i < 5; i++)
492 Emit(Classes[i], classes[i], Percent(classes[i], rows.Count));
493 foreach (var group in rows.GroupBy(r => r.Status).OrderBy(g => g.Key))
494 Emit(group.Key, group.Count(), Percent(group.Count(), rows.Count));
495 break;
496 }
497 case "METHODS":
498 {
499 Emit("METHODS", rows.Count);
500 var counts = CountStrings(rows.Select(r => r.Method));
501 foreach (string method in ByCount(counts))
502 Emit(method, counts[method], Percent(counts[method], rows.Count));
503 break;
504 }
505 case "TOP":
506 {
507 string kind = args[0];
508 var counts = CountStrings(rows.Select(r => kind == "PATHS" ? r.Path : r.Ip));
509 var keys = ByCount(counts, kind == "IPS" ? CompareIp : null);
510 int shown = Math.Min(int.Parse(args[1], CultureInfo.InvariantCulture), keys.Count);
511 Emit("TOP", kind, shown, "of", keys.Count);
512 int rank = 0;
513 for (int i = 0; i < shown; i++)
514 {
515 if (i == 0 || counts[keys[i]] != counts[keys[i - 1]]) rank = i + 1;
516 Emit(rank, keys[i], counts[keys[i]], Percent(counts[keys[i]], rows.Count));
517 }
518 break;
519 }
520 case "HOURLY":
521 {
522 int[] counts = new int[24];
523 foreach (var r in rows) counts[r.Hour]++;
524 int maximum = counts.Max();
525 Emit("HOURLY", rows.Count, "peak", Array.IndexOf(counts, maximum).ToString("D2"),
526 maximum);
527 for (int i = 0; i < 24; i++) BarLine(i.ToString("D2"), counts[i], maximum);
528 break;
529 }
530 case "TIMELINE":
531 {
532 int minutes = int.Parse(args[0], CultureInfo.InvariantCulture);
533 long width = minutes * 60L;
534 var buckets = Buckets(rows.Select(r => r.Seconds / width));
535 Emit("TIMELINE", minutes, buckets.Counts.Length);
536 int maximum = buckets.Counts.Max();
537 for (int i = 0; i < buckets.Counts.Length; i++)
538 BarLine(IntervalLabel((buckets.First + i) * width), buckets.Counts[i], maximum);
539 break;
540 }
541 case "BANDWIDTH":
542 {
543 var groups = Groups(rows, r => r.Date);
544 long total = rows.Sum(r => r.Bytes);
545 Emit("BANDWIDTH", groups.Count, total, BytesText(total));
546 foreach (string date in groups.Keys.OrderBy(s => s, StringComparer.Ordinal))
547 {
548 var entries = groups[date];
549 long bytes = entries.Sum(r => r.Bytes);
550 Emit(date, entries.Count, bytes, BytesText(bytes));
551 }
552 break;
553 }
554 case "VISITORS":
555 {
556 var first = new Dictionary<string, string>(StringComparer.Ordinal);
557 foreach (var r in rows)
558 {
559 if (!first.ContainsKey(r.Ip) ||
560 StringComparer.Ordinal.Compare(r.Date, first[r.Ip]) < 0)
561 first[r.Ip] = r.Date;
562 }
563 var groups = Groups(rows, r => r.Date);
564 Emit("VISITORS", groups.Count, first.Count);
565 foreach (string date in groups.Keys.OrderBy(s => s, StringComparer.Ordinal))
566 {
567 var ips = groups[date].Select(r => r.Ip).Distinct().ToList();
568 int newcomers = ips.Count(ip => first[ip] == date);
569 Emit(date, "unique", ips.Count, "new", newcomers,
570 "returning", ips.Count - newcomers);
571 }
572 break;
573 }
574 case "LATENCY":
575 {
576 string label = args.Length > 0 ? NormalizePath(args[0]) : "all";
577 var selected = args.Length > 0 ? rows.Where(r => r.Path == label).ToList() : rows;
578 if (selected.Count == 0) { Emit("LATENCY", "no records"); break; }
579 int[] values = selected.Select(r => r.Latency).OrderBy(x => x).ToArray();
580 int n = values.Length, trim = n / 10;
581 int[] trimmed = values.Skip(trim).Take(n - 2 * trim).ToArray();
582 Emit("LATENCY", label, "count", n, "min", values[0],
583 "p50", Percentile(values, 50), "p90", Percentile(values, 90),
584 "p99", Percentile(values, 99), "max", values[^1],
585 "mean", Decimal(values.Sum(v => (long)v), n, 2),
586 "trimmed", Decimal(trimmed.Sum(v => (long)v), trimmed.Length, 2));
587 break;
588 }
589 case "HISTOGRAM":
590 {
591 int width = int.Parse(args[0], CultureInfo.InvariantCulture);
592 var buckets = Buckets(rows.Select(r => (long)r.Latency / width));
593 Emit("HISTOGRAM", width, buckets.Counts.Length);
594 int maximum = buckets.Counts.Max();
595 for (int i = 0; i < buckets.Counts.Length; i++)
596 {
597 long lo = (buckets.First + i) * width;
598 BarLine($"{lo}-{lo + width - 1}", buckets.Counts[i], maximum);
599 }
600 break;
601 }
602 case "APDEX":
603 {
604 int threshold = int.Parse(args[0], CultureInfo.InvariantCulture);
605 var overall = Apdex(rows, threshold);
606 Emit("APDEX", threshold, overall.Notation, overall.Rating, "satisfied",
607 overall.Satisfied, "tolerating", overall.Tolerating, "frustrated", overall.Frustrated);
608 var paths = Groups(rows, r => r.Path).Select(group =>
609 (Path: group.Key, Score: Apdex(group.Value, threshold))).ToList();
610 paths.Sort((a, b) =>
611 {
612 long x = a.Score.Points * b.Score.Total;
613 long y = b.Score.Points * a.Score.Total;
614 int c = x.CompareTo(y);
615 if (c != 0) return c;
616 c = b.Score.Count.CompareTo(a.Score.Count);
617 return c != 0 ? c : StringComparer.Ordinal.Compare(a.Path, b.Path);
618 });
619 foreach (var p in paths) Emit(p.Path, p.Score.Notation, p.Score.Rating, p.Score.Count);
620 break;
621 }
622 case "ERRORS":
623 {
624 int errors = rows.Count(r => r.Status >= 400);
625 Emit("ERRORS", errors, "of", rows.Count, Percent(errors, rows.Count));
626 var data = Groups(rows, r => r.Path).Select(group =>
627 (Path: group.Key, Count: group.Value.Count,
628 Client: group.Value.Count(r => r.Status / 100 == 4),
629 Server: group.Value.Count(r => r.Status / 100 == 5)))
630 .Where(p => p.Client + p.Server > 0).ToList();
631 data.Sort((a, b) =>
632 {
633 long af = a.Client + a.Server, bf = b.Client + b.Server;
634 long x = bf * a.Count - af * b.Count;
635 if (x != 0) return x < 0 ? -1 : 1;
636 int n = (int)(bf - af);
637 return n != 0 ? n : StringComparer.Ordinal.Compare(a.Path, b.Path);
638 });
639 foreach (var p in data)
640 Emit(p.Path, p.Count, p.Client, p.Server,
641 Percent(p.Client + p.Server, p.Count));
642 break;
643 }
644 case "SLOWEST":
645 {
646 var groups = Groups(rows, r => r.Method + "\u0001" + r.Path);
647 var endpoints = groups.Select(group =>
648 {
649 string[] key = group.Key.Split('\u0001');
650 return (Method: key[0], Path: key[1], Count: group.Value.Count,
651 Total: group.Value.Sum(r => (long)r.Latency),
652 Maximum: group.Value.Max(r => r.Latency));
653 }).ToList();
654 endpoints.Sort((a, b) =>
655 {
656 long x = b.Total * a.Count - a.Total * b.Count;
657 if (x != 0) return x < 0 ? -1 : 1;
658 int n = b.Maximum.CompareTo(a.Maximum);
659 if (n != 0) return n;
660 n = StringComparer.Ordinal.Compare(a.Path, b.Path);
661 return n != 0 ? n : StringComparer.Ordinal.Compare(a.Method, b.Method);
662 });
663 int shown = Math.Min(int.Parse(args[0], CultureInfo.InvariantCulture), endpoints.Count);
664 Emit("SLOWEST", shown, "of", endpoints.Count);
665 for (int i = 0; i < shown; i++)
666 {
667 var row = endpoints[i];
668 Emit(i + 1, row.Method, row.Path, "count", row.Count,
669 "avg", Decimal(row.Total, row.Count, 2), "max", row.Maximum);
670 }
671 break;
672 }
673 case "SESSIONS":
674 {
675 string ip = args[0];
676 int minutes = args.Length > 1 ? int.Parse(args[1], CultureInfo.InvariantCulture) : 30;
677 var found = rows.Where(r => r.Ip == ip).ToList();
678 if (found.Count == 0) { Emit("SESSIONS", "no records"); break; }
679 var sessions = new List<List<LogRecord>> {new List<LogRecord> {found[0]}};
680 for (int i = 1; i < found.Count; i++)
681 {
682 if (found[i].Seconds - found[i - 1].Seconds <= minutes * 60L)
683 sessions[^1].Add(found[i]);
684 else sessions.Add(new List<LogRecord> {found[i]});
685 }
686 long duration = sessions.Sum(s => s[^1].Seconds - s[0].Seconds);
687 Emit("SESSIONS", ip, "timeout", minutes, "count", sessions.Count,
688 "requests", found.Count, "avg_duration", Decimal(duration, sessions.Count, 1),
689 "avg_requests", Decimal(found.Count, sessions.Count, 1));
690 for (int i = 0; i < sessions.Count; i++)
691 {
692 var s = sessions[i];
693 Emit(i + 1, s[0].Timestamp, s[^1].Timestamp,
694 s[^1].Seconds - s[0].Seconds, s.Count,
695 s.Select(r => r.Path).Distinct().Count());
696 }
697 break;
698 }
699 case "BURST":
700 {
701 int length = int.Parse(args[0], CultureInfo.InvariantCulture);
702 int threshold = int.Parse(args[1], CultureInfo.InvariantCulture);
703 var groups = Groups(rows.Where(r => r.Status >= 400), r => r.Ip);
704 var bursts = groups.Select(group => (Ip: group.Key, Window: Window(group.Value, length)))
705 .Where(x => x.Window.Count >= threshold).ToList();
706 bursts.Sort((a, b) =>
707 {
708 int n = b.Window.Count.CompareTo(a.Window.Count);
709 return n != 0 ? n : CompareIp(a.Ip, b.Ip);
710 });
711 Emit("BURST", length, threshold, bursts.Count);
712 foreach (var b in bursts)
713 Emit(b.Ip, b.Window.Count, b.Window.Start.Timestamp, b.Window.End.Timestamp);
714 break;
715 }
716 case "PEAK":
717 {
718 int length = int.Parse(args[0], CultureInfo.InvariantCulture);
719 var peak = Window(rows, length);
720 Emit("PEAK", length, peak.Count, peak.Start.Timestamp, peak.End.Timestamp);
721 break;
722 }
723 case "GAPS":
724 {
725 int length = int.Parse(args[0], CultureInfo.InvariantCulture);
726 var gaps = new List<(LogRecord Before, LogRecord After)>();
727 for (int i = 1; i < rows.Count; i++)
728 if (rows[i].Seconds - rows[i - 1].Seconds >= length)
729 gaps.Add((rows[i - 1], rows[i]));
730 Emit("GAPS", length, gaps.Count);
731 foreach (var g in gaps)
732 Emit(g.Before.Timestamp, g.After.Timestamp,
733 g.After.Seconds - g.Before.Seconds);
734 break;
735 }
736 case "IP":
737 {
738 string ip = args[0];
739 var found = rows.Where(r => r.Ip == ip).ToList();
740 if (found.Count == 0) { Emit("IP", "no records"); break; }
741 var paths = CountStrings(found.Select(r => r.Path));
742 Emit("IP", ip, "requests", found.Count, "bytes", found.Sum(r => r.Bytes),
743 "errors", found.Count(r => r.Status >= 400), "paths", paths.Count);
744 Emit("first", found[0].Timestamp, "last", found[^1].Timestamp);
745 var methods = CountStrings(found.Select(r => r.Method));
746 var parts = new List<object> {"methods"};
747 foreach (string method in ByCount(methods))
748 {
749 parts.Add(method);
750 parts.Add(methods[method]);
751 }
752 Emit(parts.ToArray());
753 var classes = ClassCounts(found);
754 parts = new List<object> {"classes"};
755 for (int i = 0; i < 5; i++)
756 {
757 parts.Add(Classes[i]);
758 parts.Add(classes[i]);
759 }
760 Emit(parts.ToArray());
761 foreach (string path in ByCount(paths).Take(3))
762 Emit("path", path, paths[path]);
763 break;
764 }
765 case "RANGE":
766 {
767 long start = SecondsOf(args[0], args[1]);
768 long end = SecondsOf(args[2], args[3]);
769 Check(start <= end, "invalid range");
770 from = start;
771 through = end;
772 fromLabel = args[0] + " " + args[1];
773 throughLabel = args[2] + " " + args[3];
774 Emit("RANGE", fromLabel, throughLabel, "active", Selected().Count);
775 break;
776 }
777 case "FILTER":
778 {
779 string key = args[0];
780 string value = key == "PATH" ? NormalizePath(args[1]) : args[1];
781 filters[key] = value;
782 Emit("FILTER", key, value, "active", Selected().Count);
783 break;
784 }
785 case "UNFILTER":
786 {
787 string key = args[0];
788 if (key == "RANGE")
789 {
790 Check(from.HasValue, "not filtered");
791 from = null;
792 through = null;
793 fromLabel = null;
794 throughLabel = null;
795 }
796 else
797 {
798 Check(filters[key] != null, "not filtered");
799 filters[key] = null;
800 }
801 Emit("UNFILTER", key, "active", Selected().Count);
802 break;
803 }
804 case "RESET":
805 from = null;
806 through = null;
807 fromLabel = null;
808 throughLabel = null;
809 foreach (string key in Keys) filters[key] = null;
810 Emit("RESET", "active", Selected().Count);
811 break;
812 case "STATE":
813 {
814 var parts = new List<object>
815 {"STATE", "active", rows.Count, "range",
816 from.HasValue ? fromLabel + " " + throughLabel : "-"};
817 foreach (string key in Keys)
818 {
819 parts.Add(key.ToLowerInvariant());
820 parts.Add(filters[key] ?? "-");
821 }
822 Emit(parts.ToArray());
823 break;
824 }
825 }
826 }
827 }

829 static void Main()
830 {
831 List<string> logLines = new List<string>();
832 bool inQueries = false;
833 var queries = new List<string>();
834 string? line;
835 while ((line = Console.ReadLine()) != null)
836 {
837 if (!inQueries)
838 {
839 if (line == "---") inQueries = true;
840 else logLines.Add(line);
841 }
842 else if (line == "END") break;
843 else queries.Add(line);
844 }
845
846 LogAnalyzer analyzer = new LogAnalyzer(logLines);
847 foreach (string query in queries) analyzer.Execute(query);
848 foreach (string output in analyzer.Output) Console.WriteLine(output);
849 }
