4const METHODS: [&str; 7] = ["GET", "HEAD", "POST", "PUT", "DELETE", "PATCH", "OPTIONS"];
5const REASONS: [&str; 9] = ["fields", "date", "time", "ip", "method", "path", "status", "bytes", "latency"];
6const CLASSES: [&str; 5] = ["1xx", "2xx", "3xx", "4xx", "5xx"];
7const KEYS: [&str; 4] = ["METHOD", "STATUS", "IP", "PATH"];
8const REQUIRE_RECORDS: [&str; 14] = [
9 "STATUS", "METHODS", "TOP", "HOURLY", "TIMELINE", "BANDWIDTH", "VISITORS",
10 "HISTOGRAM", "APDEX", "ERRORS", "SLOWEST", "BURST", "PEAK", "GAPS",
11];
12type QueryResult = Result<Vec<String>, &'static str>;

14fn ensure(condition: bool, message: &'static str) -> Result<(), &'static str> {
15 if condition { Ok(()) } else { Err(message) }
16}

18fn argument_kinds(name: &str) -> Option<&'static [&'static str]> {
19 match name {
20 "SUMMARY" | "MALFORMED" | "STATUS" | "METHODS" | "HOURLY" | "BANDWIDTH"
21 | "VISITORS" | "ERRORS" | "RESET" | "STATE" => Some(&[]),
22 "TOP" => Some(&["kind", "limit"]),
23 "TIMELINE" => Some(&["minutes"]),
24 "LATENCY" => Some(&["path"]),
25 "HISTOGRAM" => Some(&["width"]),
26 "APDEX" => Some(&["threshold"]),
27 "SLOWEST" => Some(&["limit"]),
28 "SESSIONS" => Some(&["ip", "minutes"]),
29 "BURST" => Some(&["length", "times"]),
30 "PEAK" | "GAPS" => Some(&["length"]),
31 "IP" => Some(&["ip"]),
32 "RANGE" => Some(&["date", "time", "date", "time"]),
33 "FILTER" => Some(&["key", "value"]),
34 "UNFILTER" => Some(&["subject"]),
35 _ => None,
36 }
37}

39fn digits(value: &str) -> bool {
40 !value.is_empty() && value.bytes().all(|b| b.is_ascii_digit())
41}
42
43fn canonical(value: &str, limit: usize) -> bool {
44 value.len() <= limit && digits(value) && (value.len() == 1 || !value.starts_with('0'))
45}
46
47fn leap(year: i32) -> bool {
48 year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
49}
50
51fn year_days(year: i32) -> i64 {
52 if leap(year) { 366 } else { 365 }
53}
54
55fn month_days(year: i32, month: usize) -> i64 {
56 if month == 2 && leap(year) { 29 }
57 else { [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31][month - 1] }
58}

60fn date_parts(value: &str) -> Option<(i32, usize, i64)> {
61 let fields: Vec<&str> = value.split('-').collect();
62 if fields.len() != 3 || fields[0].len() != 4 || fields[1].len() != 2
63 || fields[2].len() != 2 || !fields.iter().all(|s| digits(s)) {
64 return None;
65 }
66 let year = fields[0].parse().ok()?;
67 let month = fields[1].parse().ok()?;
68 let day = fields[2].parse().ok()?;
69 if !(2000..=2059).contains(&year) || !(1..=12).contains(&month)
70 || day < 1 || day > month_days(year, month) {
71 None
72 } else {
73 Some((year, month, day))
74 }
75}
76
77fn clock_parts(value: &str) -> Option<[i64; 3]> {
78 let fields: Vec<&str> = value.split(':').collect();
79 if fields.len() != 3 || !fields.iter().all(|s| s.len() == 2 && digits(s)) {
80 return None;
81 }
82 let parts = [fields[0].parse().ok()?, fields[1].parse().ok()?, fields[2].parse().ok()?];
83 if parts[0] < 24 && parts[1] < 60 && parts[2] < 60 { Some(parts) } else { None }
84}
85
86fn valid_ip(value: &str) -> bool {
87 let parts: Vec<&str> = value.split('.').collect();
88 parts.len() == 4 && parts.iter().all(|s| canonical(s, 3) && s.parse::<u16>().unwrap() <= 255)
89}
90
91fn valid_status(value: &str) -> bool {
92 value.len() == 3 && digits(value) && (100..=599).contains(&value.parse::<u16>().unwrap())
93}
94
95fn valid_filter(key: &str, value: &str) -> bool {
96 match key {
97 "METHOD" => METHODS.contains(&value),
98 "STATUS" => CLASSES.contains(&value) || valid_status(value),
99 "IP" => valid_ip(value),
100 _ => value.starts_with('/'),
101 }
102}
103
104fn validate_argument(kind: &str, value: &str, args: &[&str]) -> Result<(), &'static str> {
105 let bound = match kind {
106 "limit" => Some(100), "minutes" => Some(1440),
107 "width" | "threshold" => Some(100000), "length" => Some(86400),
108 "times" => Some(1000), _ => None,
109 };
110 if let Some(maximum) = bound {
111 return ensure(canonical(value, 6) && (1..=maximum).contains(&value.parse::<u32>().unwrap()), "invalid number");
112 }
113 let (valid, reason) = match kind {
114 "kind" => (value == "PATHS" || value == "IPS", "invalid kind"),
115 "path" => (value.starts_with('/'), "invalid path"),
116 "ip" => (valid_ip(value), "invalid ip"),
117 "date" => (date_parts(value).is_some(), "invalid timestamp"),
118 "time" => (clock_parts(value).is_some(), "invalid timestamp"),
119 "key" => (KEYS.contains(&value), "invalid key"),
120 "subject" => (value == "RANGE" || KEYS.contains(&value), "invalid key"),
121 _ => (valid_filter(args[0], value), "invalid value"),
122 };
123 ensure(valid, reason)
124}

126fn malformed_reason(fields: &[&str]) -> Option<&'static str> {
127 if fields.len() != 8 { return Some("fields"); }
128 if date_parts(fields[0]).is_none() { return Some("date"); }
129 if clock_parts(fields[1]).is_none() { return Some("time"); }
130 if !valid_ip(fields[2]) { return Some("ip"); }
131 if !METHODS.contains(&fields[3]) { return Some("method"); }
132 if !fields[4].starts_with('/') { return Some("path"); }
133 if !valid_status(fields[5]) { return Some("status"); }
134 if !canonical(fields[6], 8) { return Some("bytes"); }
135 if !canonical(fields[7], 6) { return Some("latency"); }
136 None
137}

139fn seconds_of(date: &str, clock: &str) -> i64 {
140 let (year, month, day) = date_parts(date).unwrap();
141 let [hour, minute, second] = clock_parts(clock).unwrap();
142 let days = day - 1 + (2000..year).map(year_days).sum::<i64>()
143 + (1..month).map(|m| month_days(year, m)).sum::<i64>();
144 days * 86400 + hour * 3600 + minute * 60 + second
145}
146
147fn interval_label(seconds: i64) -> String {
148 let mut days = seconds / 86400;
149 let mut year = 2000;
150 while days >= year_days(year) { days -= year_days(year); year += 1; }
151 let mut month = 1;
152 while days >= month_days(year, month) { days -= month_days(year, month); month += 1; }
153 format!("{year:04}-{month:02}-{:02} {:02}:{:02}", days + 1, seconds % 86400 / 3600, seconds % 3600 / 60)
154}

156fn normalize_path(value: &str) -> String {
157 let mut parts = Vec::new();
158 for part in value.split('?').next().unwrap().split('/') {
159 match part {
160 ".." => { parts.pop(); }
161 "" | "." => {}
162 _ => parts.push(part),
163 }
164 }
165 let parts: Vec<&str> = parts.into_iter().map(|part| {
166 if digits(part) { "{id}" }
167 else if part.len() >= 8 && part.bytes().all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b))
168 && part.bytes().any(|b| b.is_ascii_digit()) { "{hex}" }
169 else { part }
170 }).collect();
171 format!("/{}", parts.join("/"))
172}
173
174fn ip_number(value: &str) -> u64 {
175 value.split('.').fold(0, |n, part| n * 256 + part.parse::<u64>().unwrap())
176}
177
178fn status_matches(value: &str, status: u16) -> bool {
179 if CLASSES.contains(&value) { status / 100 == (value.as_bytes()[0] - b'0') as u16 }
180 else { status == value.parse::<u16>().unwrap() }
181}
182
183fn decimal(numerator: u64, denominator: u64, places: usize) -> String {
184 let scale = 10_u64.pow(places as u32);
185 let value = numerator * scale / denominator;
186 format!("{}.{:0width$}", value / scale, value % scale, width = places)
187}
188
189fn percent(part: usize, whole: usize) -> String {
190 format!("{}%", decimal(part as u64 * 100, whole as u64, 2))
191}
192
193fn bytes_text(value: u64) -> String {
194 for (unit, suffix) in [(1073741824, "GB"), (1048576, "MB"), (1024, "KB")] {
195 if value >= unit { return format!("{}{suffix}", decimal(value, unit, 1)); }
196 }
197 format!("{value}B")
198}

225fn groups<'a, K: Ord>(rows: &[&'a Record], key: impl Fn(&'a Record) -> K) -> BTreeMap<K, Vec<&'a Record>> {
226 let mut result: BTreeMap<K, Vec<&Record>> = BTreeMap::new();
227 for &row in rows { result.entry(key(row)).or_default().push(row); }
228 result
229}
230
231fn counts<'a>(values: impl Iterator<Item = &'a str>) -> BTreeMap<&'a str, usize> {
232 let mut result = BTreeMap::new();
233 for value in values { *result.entry(value).or_insert(0) += 1; }
234 result
235}
236
237fn ranked<'a>(counts: &BTreeMap<&'a str, usize>, ip: bool) -> Vec<&'a str> {
238 let mut names: Vec<&str> = counts.keys().copied().collect();
239 names.sort_by(|a, b| {
240 counts.get(*b).unwrap().cmp(counts.get(*a).unwrap()).then_with(|| {
241 if ip { ip_number(a).cmp(&ip_number(b)) } else { a.cmp(b) }
242 })
243 });
244 names
245}
246
247fn class_counts(rows: &[&Record]) -> [usize; 5] {
248 let mut result = [0; 5];
249 for row in rows { result[(row.status / 100 - 1) as usize] += 1; }
250 result
251}
252
253fn buckets(values: impl Iterator<Item = i64>) -> Result<(i64, Vec<usize>), &'static str> {
254 let values: Vec<i64> = values.collect();
255 let first = *values.iter().min().unwrap();
256 let width = values.iter().max().unwrap() - first + 1;
257 ensure(width <= 60, "too many buckets")?;
258 let mut counts = vec![0; width as usize];
259 for value in values { counts[(value - first) as usize] += 1; }
260 Ok((first, counts))
261}
262
263fn percentile(values: &[u64], rank: usize) -> u64 {
264 values[(rank * values.len() + 99) / 100 - 1]
265}

267struct Peak<'a> {
268 count: usize,
269 first: &'a Record,
270 last: &'a Record,
271}
272
273fn peak<'a>(rows: &[&'a Record], width: i64) -> Peak<'a> {
274 let mut first = 0;
275 let mut result = Peak { count: 0, first: rows[0], last: rows[0] };
276 for last in 0..rows.len() {
277 while rows[last].seconds - rows[first].seconds >= width { first += 1; }
278 if last - first + 1 > result.count {
279 result = Peak { count: last - first + 1, first: rows[first], last: rows[last] };
280 }
281 }
282 result
283}

285struct Apdex {
286 count: usize,
287 satisfied: usize,
288 tolerating: usize,
289}
290
291impl Apdex {
292 fn new(rows: &[&Record], threshold: u64) -> Self {
293 Self {
294 count: rows.len(),
295 satisfied: rows.iter().filter(|r| r.status < 500 && r.latency <= threshold).count(),
296 tolerating: rows.iter().filter(|r| r.status < 500 && r.latency > threshold && r.latency <= 4 * threshold).count(),
297 }
298 }
299 fn points(&self) -> u64 { (2 * self.satisfied + self.tolerating) as u64 }
300 fn total(&self) -> u64 { (2 * self.count) as u64 }
301 fn text(&self) -> String { decimal(self.points(), self.total(), 3) }
302 fn rating(&self) -> &'static str {
303 for (limit, rating) in [(94, "excellent"), (85, "good"), (70, "fair"), (50, "poor")] {
304 if self.points() * 100 >= self.total() * limit { return rating; }
305 }
306 "unacceptable"
307 }
308}

317struct Analyzer {
318 line_count: usize,
319 records: Vec<Record>,
320 malformed: Vec<(usize, &'static str)>,
321 conditions: BTreeMap<String, String>,
322 range: Option<TimeRange>,
323}
324
325impl Analyzer {
326 fn new(lines: &[String]) -> Self {
327 let mut analyzer = Self {
328 line_count: lines.len(), records: Vec::new(), malformed: Vec::new(),
329 conditions: BTreeMap::new(), range: None,
330 };
331 for (i, line) in lines.iter().enumerate() {
332 let fields: Vec<&str> = line.split_ascii_whitespace().collect();
333 if let Some(reason) = malformed_reason(&fields) { analyzer.malformed.push((i + 1, reason)); }
334 else { analyzer.records.push(Record::new(i + 1, &fields)); }
335 }
336 analyzer.records.sort_by_key(|r| (r.seconds, r.line));
337 analyzer
338 }
339
340 fn selected(&self) -> Vec<&Record> {
341 self.records.iter().filter(|row| {
342 if let Some(range) = &self.range {
343 if row.seconds < range.start || row.seconds > range.end { return false; }
344 }
345 for (key, value) in &self.conditions {
346 let matches = match key.as_str() {
347 "METHOD" => row.method == *value,
348 "STATUS" => status_matches(value, row.status),
349 "IP" => row.ip == *value,
350 _ => row.path == *value,
351 };
352 if !matches { return false; }
353 }
354 true
355 }).collect()
356 }

358 fn execute(&mut self, line: &str) -> Vec<String> {
359 self.command(line).unwrap_or_else(|error| vec![format!("ERROR {error}")])
360 }
361
362 fn command(&mut self, line: &str) -> QueryResult {
363 let fields: Vec<&str> = line.split_ascii_whitespace().collect();
364 let name = fields.first().copied().unwrap_or("");
365 let args = fields.get(1..).unwrap_or(&[]);
366 let form = argument_kinds(name).ok_or("unknown query")?;
367 let optional = matches!(name, "LATENCY" | "SESSIONS") && args.len() + 1 == form.len();
368 ensure(args.len() == form.len() || optional, "wrong arguments")?;
369 for (i, argument) in args.iter().enumerate() { validate_argument(form[i], argument, args)?; }
370 match name {
371 "RANGE" => {
372 let start = seconds_of(args[0], args[1]);
373 let end = seconds_of(args[2], args[3]);
374 ensure(start <= end, "invalid range")?;
375 let start_text = format!("{} {}", args[0], args[1]);
376 let end_text = format!("{} {}", args[2], args[3]);
377 self.range = Some(TimeRange { start, end, start_text: start_text.clone(), end_text: end_text.clone() });
378 Ok(vec![format!("RANGE {start_text} {end_text} active {}", self.selected().len())])
379 }
380 "FILTER" => {
381 let value = if args[0] == "PATH" { normalize_path(args[1]) } else { args[1].to_string() };
382 self.conditions.insert(args[0].to_string(), value.clone());
383 Ok(vec![format!("FILTER {} {value} active {}", args[0], self.selected().len())])
384 }
385 "UNFILTER" => {
386 if args[0] == "RANGE" {
387 ensure(self.range.is_some(), "not filtered")?;
388 self.range = None;
389 } else {
390 ensure(self.conditions.contains_key(args[0]), "not filtered")?;
391 self.conditions.remove(args[0]);
392 }
393 Ok(vec![format!("UNFILTER {} active {}", args[0], self.selected().len())])
394 }
395 "RESET" => {
396 self.range = None;
397 self.conditions.clear();
398 Ok(vec![format!("RESET active {}", self.selected().len())])
399 }
400 _ => self.query(name, args),
401 }
402 }
403
404 fn query(&self, name: &str, args: &[&str]) -> QueryResult {
405 let rows = self.selected();
406 let n = rows.len();
407 if REQUIRE_RECORDS.contains(&name) && n == 0 { return Ok(vec![format!("{name} no records")]); }
408 let mut output = Vec::new();
409 match name {
410 "SUMMARY" => {
411 output.push(format!("SUMMARY lines {} valid {} malformed {}", self.line_count, self.records.len(), self.malformed.len()));
412 output.push(format!("requests {n} bytes {} ips {} paths {}",
413 rows.iter().map(|r| r.bytes).sum::<u64>(), counts(rows.iter().map(|r| r.ip.as_str())).len(),
414 counts(rows.iter().map(|r| r.path.as_str())).len()));
415 if n == 0 { output.push("span none".into()); }
416 else { output.push(format!("span {} {} {}", rows[0].timestamp, rows[n - 1].timestamp, rows[n - 1].seconds - rows[0].seconds)); }
417 }
418 "MALFORMED" => {
419 output.push(format!("MALFORMED {}", self.malformed.len()));
420 for reason in REASONS {
421 let numbers: Vec<String> = self.malformed.iter().filter(|(_, why)| *why == reason).map(|(line, _)| line.to_string()).collect();
422 if !numbers.is_empty() { output.push(format!("{reason} {} {}", numbers.len(), numbers.join(" "))); }
423 }
424 }
425 "STATUS" => {
426 output.push(format!("STATUS {n}"));
427 for (i, count) in class_counts(&rows).iter().enumerate() {
428 output.push(format!("{} {count} {}", CLASSES[i], percent(*count, n)));
429 }
430 let mut statuses = BTreeMap::new();
431 for row in &rows { *statuses.entry(row.status).or_insert(0) += 1; }
432 for (status, count) in statuses { output.push(format!("{status} {count} {}", percent(count, n))); }
433 }
434 "METHODS" => {
435 output.push(format!("METHODS {n}"));
436 let counts = counts(rows.iter().map(|r| r.method.as_str()));
437 for key in ranked(&counts, false) { output.push(format!("{key} {} {}", counts[key], percent(counts[key], n))); }
438 }
439 "TOP" => {
440 let kind = args[0];
441 let counts = counts(rows.iter().map(|r| if kind == "PATHS" { r.path.as_str() } else { r.ip.as_str() }));
442 let keys = ranked(&counts, kind == "IPS");
443 let shown = args[1].parse::<usize>().unwrap().min(keys.len());
444 output.push(format!("TOP {kind} {shown} of {}", keys.len()));
445 let mut rank = 0;
446 for i in 0..shown {
447 if i == 0 || counts[keys[i]] != counts[keys[i - 1]] { rank = i + 1; }
448 output.push(format!("{rank} {} {} {}", keys[i], counts[keys[i]], percent(counts[keys[i]], n)));
449 }
450 }
451 "HOURLY" => {
452 let mut counts = [0; 24];
453 for row in &rows { counts[row.hour] += 1; }
454 let maximum = *counts.iter().max().unwrap();
455 let first = counts.iter().position(|&c| c == maximum).unwrap();
456 output.push(format!("HOURLY {n} peak {first:02} {maximum}"));
457 for (hour, &count) in counts.iter().enumerate() { output.push(bar_line(&format!("{hour:02}"), count, maximum)); }
458 }
459 "TIMELINE" => {
460 let minutes = args[0].parse::<i64>().unwrap();
461 let width = minutes * 60;
462 let (first, counts) = buckets(rows.iter().map(|r| r.seconds / width))?;
463 output.push(format!("TIMELINE {minutes} {}", counts.len()));
464 let maximum = *counts.iter().max().unwrap();
465 for (i, &count) in counts.iter().enumerate() {
466 output.push(bar_line(&interval_label((first + i as i64) * width), count, maximum));
467 }
468 }
469 "BANDWIDTH" => {
470 let days = groups(&rows, |r| r.date.as_str());
471 let total: u64 = rows.iter().map(|r| r.bytes).sum();
472 output.push(format!("BANDWIDTH {} {total} {}", days.len(), bytes_text(total)));
473 for (date, group) in days {
474 let bytes: u64 = group.iter().map(|r| r.bytes).sum();
475 output.push(format!("{date} {} {bytes} {}", group.len(), bytes_text(bytes)));
476 }
477 }
478 "VISITORS" => {
479 let mut first = BTreeMap::new();
480 for row in &rows { first.entry(row.ip.as_str()).or_insert(row.date.as_str()); }
481 let days = groups(&rows, |r| r.date.as_str());
482 output.push(format!("VISITORS {} {}", days.len(), first.len()));
483 for (date, group) in days {
484 let ips = counts(group.iter().map(|r| r.ip.as_str()));
485 let fresh = ips.keys().filter(|&&ip| first[ip] == date).count();
486 output.push(format!("{date} unique {} new {fresh} returning {}", ips.len(), ips.len() - fresh));
487 }
488 }
489 "LATENCY" => {
490 let label = args.first().map(|p| normalize_path(p)).unwrap_or_else(|| "all".into());
491 let mut values: Vec<u64> = rows.iter().filter(|r| args.is_empty() || r.path == label).map(|r| r.latency).collect();
492 if values.is_empty() { output.push("LATENCY no records".into()); }
493 else {
494 values.sort_unstable();
495 let count = values.len();
496 let trim = count / 10;
497 let trimmed = &values[trim..count - trim];
498 output.push(format!("LATENCY {label} count {count} min {} p50 {} p90 {} p99 {} max {} mean {} trimmed {}",
499 values[0], percentile(&values, 50), percentile(&values, 90), percentile(&values, 99), values[count - 1],
500 decimal(values.iter().sum(), count as u64, 2), decimal(trimmed.iter().sum(), trimmed.len() as u64, 2)));
501 }
502 }
503 "HISTOGRAM" => {
504 let width = args[0].parse::<i64>().unwrap();
505 let (first, counts) = buckets(rows.iter().map(|r| r.latency as i64 / width))?;
506 output.push(format!("HISTOGRAM {width} {}", counts.len()));
507 let maximum = *counts.iter().max().unwrap();
508 for (i, &count) in counts.iter().enumerate() {
509 let low = (first + i as i64) * width;
510 output.push(bar_line(&format!("{low}-{}", low + width - 1), count, maximum));
511 }
512 }
513 "APDEX" => {
514 let threshold = args[0].parse().unwrap();
515 let total = Apdex::new(&rows, threshold);
516 output.push(format!("APDEX {threshold} {} {} satisfied {} tolerating {} frustrated {}",
517 total.text(), total.rating(), total.satisfied, total.tolerating, total.count - total.satisfied - total.tolerating));
518 let mut paths: Vec<(&str, Apdex)> = groups(&rows, |r| r.path.as_str()).into_iter()
519 .map(|(path, group)| (path, Apdex::new(&group, threshold))).collect();
520 paths.sort_by(|(xp, x), (yp, y)| (x.points() * y.total()).cmp(&(y.points() * x.total()))
521 .then_with(|| y.count.cmp(&x.count)).then_with(|| xp.cmp(yp)));
522 for (path, score) in paths { output.push(format!("{path} {} {} {}", score.text(), score.rating(), score.count)); }
523 }
524 "ERRORS" => {
525 let failed = rows.iter().filter(|r| r.status >= 400).count();
526 output.push(format!("ERRORS {failed} of {n} {}", percent(failed, n)));
527 struct Path<'a> { name: &'a str, count: usize, client: usize, server: usize }
528 let mut paths: Vec<Path> = groups(&rows, |r| r.path.as_str()).into_iter().map(|(name, group)| Path {
529 name, count: group.len(), client: group.iter().filter(|r| r.status / 100 == 4).count(),
530 server: group.iter().filter(|r| r.status / 100 == 5).count(),
531 }).filter(|p| p.client + p.server > 0).collect();
532 paths.sort_by(|x, y| {
533 let xf = x.client + x.server;
534 let yf = y.client + y.server;
535 (yf as u64 * x.count as u64).cmp(&(xf as u64 * y.count as u64))
536 .then_with(|| yf.cmp(&xf)).then_with(|| x.name.cmp(y.name))
537 });
538 for p in paths { output.push(format!("{} {} {} {} {}", p.name, p.count, p.client, p.server, percent(p.client + p.server, p.count))); }
539 }
540 "SLOWEST" => {
541 struct Endpoint<'a> { method: &'a str, path: &'a str, count: usize, total: u64, maximum: u64 }
542 let mut endpoints: Vec<Endpoint> = groups(&rows, |r| (r.method.as_str(), r.path.as_str())).into_iter()
543 .map(|((method, path), group)| Endpoint {
544 method, path, count: group.len(), total: group.iter().map(|r| r.latency).sum(),
545 maximum: group.iter().map(|r| r.latency).max().unwrap(),
546 }).collect();
547 endpoints.sort_by(|x, y| (y.total * x.count as u64).cmp(&(x.total * y.count as u64))
548 .then_with(|| y.maximum.cmp(&x.maximum)).then_with(|| x.path.cmp(y.path)).then_with(|| x.method.cmp(y.method)));
549 let shown = args[0].parse::<usize>().unwrap().min(endpoints.len());
550 output.push(format!("SLOWEST {shown} of {}", endpoints.len()));
551 for (i, e) in endpoints.iter().take(shown).enumerate() {
552 output.push(format!("{} {} {} count {} avg {} max {}", i + 1, e.method, e.path, e.count, decimal(e.total, e.count as u64, 2), e.maximum));
553 }
554 }
555 "SESSIONS" => {
556 let ip = args[0];
557 let minutes = args.get(1).map(|s| s.parse::<i64>().unwrap()).unwrap_or(30);
558 let found: Vec<&Record> = rows.iter().copied().filter(|r| r.ip == ip).collect();
559 if found.is_empty() { output.push("SESSIONS no records".into()); }
560 else {
561 let mut sessions = vec![vec![found[0]]];
562 for i in 1..found.len() {
563 if found[i].seconds - found[i - 1].seconds <= minutes * 60 { sessions.last_mut().unwrap().push(found[i]); }
564 else { sessions.push(vec![found[i]]); }
565 }
566 let duration: u64 = sessions.iter().map(|s| (s.last().unwrap().seconds - s[0].seconds) as u64).sum();
567 output.push(format!("SESSIONS {ip} timeout {minutes} count {} requests {} avg_duration {} avg_requests {}",
568 sessions.len(), found.len(), decimal(duration, sessions.len() as u64, 1), decimal(found.len() as u64, sessions.len() as u64, 1)));
569 for (i, session) in sessions.iter().enumerate() {
570 let last = session.last().unwrap();
571 output.push(format!("{} {} {} {} {} {}", i + 1, session[0].timestamp, last.timestamp, last.seconds - session[0].seconds,
572 session.len(), counts(session.iter().map(|r| r.path.as_str())).len()));
573 }
574 }
575 }
576 "BURST" => {
577 let width = args[0].parse().unwrap();
578 let threshold = args[1].parse::<usize>().unwrap();
579 let failed: Vec<&Record> = rows.iter().copied().filter(|r| r.status >= 400).collect();
580 let mut peaks: Vec<(&str, Peak)> = groups(&failed, |r| r.ip.as_str()).into_iter().map(|(ip, group)| (ip, peak(&group, width)))
581 .filter(|(_, p)| p.count >= threshold).collect();
582 peaks.sort_by(|(xi, x), (yi, y)| y.count.cmp(&x.count).then_with(|| ip_number(xi).cmp(&ip_number(yi))));
583 output.push(format!("BURST {width} {threshold} {}", peaks.len()));
584 for (ip, p) in peaks { output.push(format!("{ip} {} {} {}", p.count, p.first.timestamp, p.last.timestamp)); }
585 }
586 "PEAK" => {
587 let width = args[0].parse().unwrap();
588 let p = peak(&rows, width);
589 output.push(format!("PEAK {width} {} {} {}", p.count, p.first.timestamp, p.last.timestamp));
590 }
591 "GAPS" => {
592 let width = args[0].parse::<i64>().unwrap();
593 let gaps: Vec<_> = rows.windows(2).filter(|p| p[1].seconds - p[0].seconds >= width).collect();
594 output.push(format!("GAPS {width} {}", gaps.len()));
595 for gap in gaps { output.push(format!("{} {} {}", gap[0].timestamp, gap[1].timestamp, gap[1].seconds - gap[0].seconds)); }
596 }
597 "IP" => {
598 let ip = args[0];
599 let found: Vec<&Record> = rows.iter().copied().filter(|r| r.ip == ip).collect();
600 if found.is_empty() { output.push("IP no records".into()); }
601 else {
602 let paths = counts(found.iter().map(|r| r.path.as_str()));
603 output.push(format!("IP {ip} requests {} bytes {} errors {} paths {}", found.len(), found.iter().map(|r| r.bytes).sum::<u64>(),
604 found.iter().filter(|r| r.status >= 400).count(), paths.len()));
605 output.push(format!("first {} last {}", found[0].timestamp, found.last().unwrap().timestamp));
606 let methods = counts(found.iter().map(|r| r.method.as_str()));
607 let method_parts: Vec<String> = ranked(&methods, false).iter().map(|m| format!("{m} {}", methods[*m])).collect();
608 output.push(format!("methods {}", method_parts.join(" ")));
609 let category_parts: Vec<String> = class_counts(&found).iter().enumerate().map(|(i, count)| format!("{} {count}", CLASSES[i])).collect();
610 output.push(format!("classes {}", category_parts.join(" ")));
611 for path in ranked(&paths, false).into_iter().take(3) { output.push(format!("path {path} {}", paths[path])); }
612 }
613 }
614 "STATE" => {
615 let range = self.range.as_ref().map(|r| format!("{} {}", r.start_text, r.end_text)).unwrap_or_else(|| "-".into());
616 let conditions: Vec<String> = KEYS.iter().map(|key| format!("{} {}", key.to_ascii_lowercase(),
617 self.conditions.get(*key).map(String::as_str).unwrap_or("-"))).collect();
618 output.push(format!("STATE active {n} range {range} {}", conditions.join(" ")));
619 }
620 _ => unreachable!("Only validated query names are dispatched"),
621 }
622 Ok(output)
623 }
624}

626fn bar_line(label: &str, count: usize, maximum: usize) -> String {
627 if count == 0 { format!("{label} 0") }
628 else { format!("{label} {count} {}", "#".repeat((count * 30 + maximum - 1) / maximum)) }
629}

631fn main() -> io::Result<()> {
632 let stdin = io::stdin();
633 let mut input = stdin.lock().lines();
634 let mut logs = Vec::new();
635 for line in input.by_ref() {
636 let line = line?;
637 if line == "---" { break; }
638 logs.push(line);
639 }
640 let mut analyzer = Analyzer::new(&logs);
641 let stdout = io::stdout();
642 let mut output = io::BufWriter::new(stdout.lock());
643 for line in input {
644 let line = line?;
645 if line == "END" { break; }
646 for result in analyzer.execute(&line) { writeln!(output, "{result}")?; }
647 }
648 output.flush()
649}
