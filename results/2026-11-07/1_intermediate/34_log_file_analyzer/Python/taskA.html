85def is_digits(text):

89def is_canonical(text, max_length):
90 if not is_digits(text) or len(text) > max_length:
91 return False
92 return len(text) == 1 or text[0] != "0"

95def is_leap(year):
96 return year % 4 == 0 and year % 100 != 0 or year % 400 == 0

99def year_days(year):
100 return 366 if is_leap(year) else 365

103def month_days(year, month):
104 if month == 2 and is_leap(year):
105 return 29
106 return MONTH_DAYS[month - 1]

109def is_valid_date(text):
110 if len(text) != 10 or text[4] != "-" or text[7] != "-":
111 return False
112 parts = (text[:4], text[5:7], text[8:])
113 if not all(is_digits(part) for part in parts):
114 return False
115 year, month, day = (int(part) for part in parts)
116 if not 2000 <= year <= 2059 or not 1 <= month <= 12:
117 return False
118 return 1 <= day <= month_days(year, month)
119
120
121def is_valid_time(text):
122 if len(text) != 8 or text[2] != ":" or text[5] != ":":
123 return False
124 parts = (text[:2], text[3:5], text[6:])
125 if not all(is_digits(part) for part in parts):
126 return False
127 hour, minute, second = (int(part) for part in parts)
128 return hour <= 23 and minute <= 59 and second <= 59

131def is_valid_ip(text):
132 parts = text.split(".")
133 if len(parts) != 4:
134 return False
135 return all(is_canonical(part, 3) and int(part) <= 255 for part in parts)

138def is_valid_status(text):
139 return len(text) == 3 and is_digits(text) and 100 <= int(text) <= 599

142def is_valid_value(key, text):
143 if key == "METHOD":
144 return text in METHODS
145 if key == "STATUS":
146 return text in STATUS_CLASSES or is_valid_status(text)
147 if key == "IP":
148 return is_valid_ip(text)
149 return text.startswith("/")

152def is_valid_argument(kind, text, arguments):
153 if kind in NUMBER_RANGES:
154 low, high = NUMBER_RANGES[kind]
155 return is_canonical(text, 6) and low <= int(text) <= high
156 if kind == "kind":
157 return text in TOP_KINDS
158 if kind == "path":
159 return text.startswith("/")
160 if kind == "ip":
161 return is_valid_ip(text)
162 if kind == "date":
163 return is_valid_date(text)
164 if kind == "time":
165 return is_valid_time(text)
166 if kind == "key":
167 return text in FILTER_KEYS
168 if kind == "subject":
169 return text in FILTER_KEYS or text == "RANGE"
170 return is_valid_value(arguments[0], text)

173def malformed_reason(fields):
174 if len(fields) != 8:
175 return "fields"
176 date, clock, ip, method, path, status, size, latency = fields
177 if not is_valid_date(date):
178 return "date"
179 if not is_valid_time(clock):
180 return "time"
181 if not is_valid_ip(ip):
182 return "ip"
183 if method not in METHODS:
184 return "method"
185 if not path.startswith("/"):
186 return "path"
187 if not is_valid_status(status):
188 return "status"
189 if not is_canonical(size, 8):
190 return "bytes"
191 if not is_canonical(latency, 6):
192 return "latency"
193 return None

196def day_number(year, month, day):
197 days = day - 1
198 for earlier_year in range(2000, year):
199 days += year_days(earlier_year)
200 for earlier_month in range(1, month):
201 days += month_days(year, earlier_month)
202 return days

205def seconds_of(date, clock):
206 year, month, day = int(date[:4]), int(date[5:7]), int(date[8:])
207 hour, minute, second = int(clock[:2]), int(clock[3:5]), int(clock[6:])
208 return day_number(year, month, day) * 86400 + hour * 3600 + minute * 60 + second

211def interval_label(seconds):
212 days, rest = divmod(seconds, 86400)
213 year = 2000
214 while days >= year_days(year):
215 days -= year_days(year)
216 year += 1
217 month = 1
218 while days >= month_days(year, month):
219 days -= month_days(year, month)
220 month += 1
221 return f"{year}-{month:02d}-{days + 1:02d} {rest // 3600:02d}:{rest % 3600 // 60:02d}"

224def path_part(part):
225 if is_digits(part):
226 return "{id}"
227 is_hex = all(character in HEX_DIGITS for character in part)
228 has_digit = any("0" <= character <= "9" for character in part)
229 if len(part) >= 8 and is_hex and has_digit:
230 return "{hex}"
231 return part

234def normalize_path(text):
235 kept = []
236 for part in text.split("?", 1)[0].split("/"):
237 if part == "..":
238 if kept:
239 kept.pop()
240 elif part not in ("", "."):
241 kept.append(part)
242 return "/" + "/".join(path_part(part) for part in kept)

255def decimal(numerator, denominator, places):
256 scale = 10 ** places
257 whole, fraction = divmod(numerator * scale // denominator, scale)
258 return f"{whole}." + str(fraction).zfill(places)

294def bucket_counts(numbers):
295 first, last = min(numbers), max(numbers)
296 if last - first + 1 > MAX_BUCKETS:
297 raise QueryError("too many buckets")
298 counts = [0] * (last - first + 1)
299 for number in numbers:
300 counts[number - first] += 1
301 return first, counts

341class Filters:
342 def __init__(self):
343 self.range = None
344 self.conditions = dict.fromkeys(FILTER_KEYS)
345
346 def accepts(self, record):
347 if self.range is not None and not self.range.contains(record.seconds):
348 return False
349 method = self.conditions["METHOD"]
350 if method is not None and record.method != method:
351 return False
352 status = self.conditions["STATUS"]
353 if status is not None and not status_matches(status, record.status):
354 return False
355 ip = self.conditions["IP"]
356 if ip is not None and record.ip != ip:
357 return False
358 path = self.conditions["PATH"]
359 return path is None or record.path == path

362class Window:
363 def __init__(self, records, length):
364 self.size = 0
365 self.start = None
366 self.end = None
367 first = 0
368 for last, record in enumerate(records):
369 while record.seconds - records[first].seconds >= length:
370 first += 1
371 if last - first + 1 > self.size:
372 self.size = last - first + 1
373 self.start = records[first]
374 self.end = record

377class ApdexScore:
378 def __init__(self, records, threshold):
379 self.count = len(records)
380 self.satisfied = 0
381 self.tolerating = 0
382 for record in records:
383 if record.status >= 500:
384 continue
385 if record.latency <= threshold:
386 self.satisfied += 1
387 elif record.latency <= 4 * threshold:
388 self.tolerating += 1
389 self.frustrated = self.count - self.satisfied - self.tolerating
390 self.points = 2 * self.satisfied + self.tolerating
391 self.score = Fraction(self.points, 2 * self.count)
392
393 def notation(self):
394 return decimal(self.points, 2 * self.count, 3)
395
396 def rating(self):
397 for minimum, name in RATINGS:
398 if self.score >= minimum:
399 return name
400 return "unacceptable"

519 def top(self, arguments, records):
520 kind, limit = arguments[0], int(arguments[1])
521 if kind == "PATHS":
522 counts = Counter(record.path for record in records)
523 keys = by_count(counts)
524 else:
525 counts = Counter(record.ip for record in records)
526 keys = by_count(counts, ip_order)
527 shown = min(limit, len(keys))
528 self.emit("TOP", kind, shown, "of", len(keys))
529 rank = 0
530 for index, key in enumerate(keys[:shown]):
531 if index == 0 or counts[key] != counts[keys[index - 1]]:
532 rank = index + 1
533 self.emit(rank, key, counts[key], percentage(counts[key], len(records)))

562 def visitors(self, arguments, records):
563 first_days = {}

566 first_days[record.ip] = record.date

574 def latency(self, arguments, records):

576 target = normalize_path(arguments[0])
577 selected = [record for record in records if record.path == target]

579 target = "all"

645 def sessions(self, arguments, records):
646 ip = arguments[0]
647 minutes = int(arguments[1]) if len(arguments) == 2 else DEFAULT_TIMEOUT
648 selected = [record for record in records if record.ip == ip]
649 if not selected:
650 self.emit("SESSIONS", "no records")
651 return
652 sessions = [[selected[0]]]
653 for previous, record in zip(selected, selected[1:]):
654 if record.seconds - previous.seconds <= minutes * 60:
655 sessions[-1].append(record)
656 else:
657 sessions.append([record])
658 durations = sum(session[-1].seconds - session[0].seconds for session in sessions)
659 self.emit(
660 "SESSIONS", ip, "timeout", minutes, "count", len(sessions),
661 "requests", len(selected),
662 "avg_duration", decimal(durations, len(sessions), 1),
663 "avg_requests", decimal(len(selected), len(sessions), 1),
664 )
665 for number, session in enumerate(sessions, 1):
666 first, last = session[0], session[-1]
667 paths = {record.path for record in session}
668 self.emit(
669 number, first.timestamp, last.timestamp,
670 last.seconds - first.seconds, len(session), len(paths),
671 )

673 def burst(self, arguments, records):
674 length, times = int(arguments[0]), int(arguments[1])
675 failures = [record for record in records if record.is_error()]
676 matches = []
677 for ip, group in group_by(failures, lambda record: record.ip).items():
678 window = Window(group, length)
679 if window.size >= times:
680 matches.append((ip, window))
681 matches.sort(key=lambda item: (-item[1].size, ip_order(item[0])))
682 self.emit("BURST", length, times, len(matches))
683 for ip, window in matches:
684 self.emit(ip, window.size, window.start.timestamp, window.end.timestamp)

701 def ip_details(self, arguments, records):
702 ip = arguments[0]
703 selected = [record for record in records if record.ip == ip]
704 if not selected:
705 self.emit("IP", "no records")
706 return
707 total = sum(record.size for record in selected)
708 failed = sum(record.is_error() for record in selected)
709 paths = Counter(record.path for record in selected)
710 self.emit(
711 "IP", ip, "requests", len(selected), "bytes", total,
712 "errors", failed, "paths", len(paths),
713 )
714 self.emit("first", selected[0].timestamp, "last", selected[-1].timestamp)
715 methods = Counter(record.method for record in selected)
716 parts = ["methods"]
717 for method in by_count(methods):
718 parts.extend((method, methods[method]))
719 self.emit(*parts)
720 parts = ["classes"]
721 for label, count in zip(STATUS_CLASSES, class_counts(selected)):
722 parts.extend((label, count))

724 for path in by_count(paths)[:PATH_LINES]:

757 def state(self, arguments, records):
758 time_range = self.filters.range
759 range_text = "-" if time_range is None else time_range.start + " " + time_range.end
760 parts = ["STATE", "active", len(records), "range", range_text]
761 for key in FILTER_KEYS:
762 value = self.filters.conditions[key]
763 parts.extend((key.lower(), "-" if value is None else value))
764 self.emit(*parts)

766 def execute(self, line):
767 name, *arguments = line.split()
768 try:
769 if name not in QUERY_FORMS:
770 raise QueryError("unknown query")
771 form = QUERY_FORMS[name]
772 omitted = name in OPTIONAL_LAST and len(arguments) == len(form) - 1
773 if len(arguments) != len(form) and not omitted:
774 raise QueryError("wrong arguments")
775 for kind, text in zip(form, arguments):
776 if not is_valid_argument(kind, text, arguments):
777 raise QueryError(ARGUMENT_ERRORS[kind])
778 records = self.targets()
779 if name in NEEDS_TARGETS and not records:
780 self.emit(name, "no records")
781 else:
782 self.handlers[name](arguments, records)
783 except QueryError as error:
784 self.emit("ERROR", error)
