def decimal(numerator, denominator, places):
    scale = 10 ** places
    whole, fraction = divmod(numerator * scale // denominator, scale)
    return f"{whole}." + str(fraction).zfill(places)
def percentage(part, whole):
    return decimal(part * 100, whole, 2) + "%"
