62 private static List<String> tokens(String line) {
63 String text = line.trim();
64 return text.isEmpty() ? List.of() : Arrays.asList(text.split("\\s+"));
65 }

67 private static boolean digits(String text) {
68 return text.matches("[0-9]+");
69 }
70
71 private static boolean canonical(String text, int maximum) {
72 return text.length() > 0 && text.length() <= maximum && digits(text) &&
73 (text.length() == 1 || text.charAt(0) != '0');
74 }
75
76 private static boolean validDate(String text) {
77 if (!text.matches("[0-9]{4}-[0-9]{2}-[0-9]{2}")) return false;
78 int year = Integer.parseInt(text.substring(0, 4));
79 if (year < 2000 || year > 2059) return false;
80 try {
81 LocalDate.parse(text);
82 return true;
83 } catch (DateTimeParseException error) {
84 return false;
85 }
86 }
87
88 private static boolean validTime(String text) {
89 return text.matches("[0-9]{2}:[0-9]{2}:[0-9]{2}") &&
90 Integer.parseInt(text.substring(0, 2)) < 24 &&
91 Integer.parseInt(text.substring(3, 5)) < 60 &&
92 Integer.parseInt(text.substring(6, 8)) < 60;
93 }
94
95 private static boolean validIp(String ip) {
96 String[] parts = ip.split("\\.", -1);
97 if (parts.length != 4) return false;
98 for (String part : parts) {
99 if (!canonical(part, 3) || Integer.parseInt(part) > 255) return false;
100 }
101 return true;
102 }
103
104 private static boolean validStatus(String text) {
105 return text.matches("[0-9]{3}") &&
106 Integer.parseInt(text) >= 100 && Integer.parseInt(text) <= 599;
107 }
108
109 private static boolean filterValue(String key, String value) {
110 return switch (key) {
111 case "METHOD" -> METHODS.contains(value);
112 case "STATUS" -> CLASSES.contains(value) || validStatus(value);
113 case "IP" -> validIp(value);
114 default -> value.startsWith("/");
115 };
116 }
117
118 private static boolean validArgument(String kind, String value, List<String> args) {
119 int[] range = RANGES.get(kind);
120 if (range != null) {
121 if (!canonical(value, 6)) return false;
122 int number = Integer.parseInt(value);
123 return number >= range[0] && number <= range[1];
124 }
125 return switch (kind) {
126 case "kind" -> value.equals("PATHS") || value.equals("IPS");
127 case "path" -> value.startsWith("/");
128 case "ip" -> validIp(value);
129 case "date" -> validDate(value);
130 case "time" -> validTime(value);
131 case "key" -> KEYS.contains(value);
132 case "subject" -> value.equals("RANGE") || KEYS.contains(value);
133 default -> filterValue(args.get(0), value);
134 };
135 }

137 private static String malformedReason(List<String> fields) {
138 if (fields.size() != 8) return "fields";
139 if (!validDate(fields.get(0))) return "date";
140 if (!validTime(fields.get(1))) return "time";
141 if (!validIp(fields.get(2))) return "ip";
142 if (!METHODS.contains(fields.get(3))) return "method";
143 if (!fields.get(4).startsWith("/")) return "path";
144 if (!validStatus(fields.get(5))) return "status";
145 if (!canonical(fields.get(6), 8)) return "bytes";
146 if (!canonical(fields.get(7), 6)) return "latency";
147 return null;
148 }

150 private static long secondsOf(String date, String time) {
151 long days = ChronoUnit.DAYS.between(EPOCH, LocalDate.parse(date));
152 return days * 86400 +
153 Integer.parseInt(time.substring(0, 2)) * 3600 +
154 Integer.parseInt(time.substring(3, 5)) * 60 +
155 Integer.parseInt(time.substring(6, 8));
156 }
157
158 private static String intervalLabel(long seconds) {
159 LocalDate date = EPOCH.plusDays(seconds / 86400);
160 long minutes = seconds % 86400 / 60;
161 return String.format(Locale.ROOT, "%s %02d:%02d",
162 date, minutes / 60, minutes % 60);
163 }

165 private static String normalizePath(String raw) {
166 String path = raw.split("\\?", 2)[0];
167 List<String> parts = new ArrayList<>();
168 for (String item : path.split("/", -1)) {
169 if (item.equals("..")) {
170 if (!parts.isEmpty()) parts.removeLast();
171 } else if (!item.isEmpty() && !item.equals(".")) {
172 parts.add(item);
173 }
174 }
175 for (int i = 0; i < parts.size(); i++) {
176 String item = parts.get(i);
177 if (digits(item)) parts.set(i, "{id}");
178 else if (item.length() >= 8 && item.matches("[0-9a-f]+") &&
179 item.matches(".*[0-9].*")) parts.set(i, "{hex}");
180 }
181 return "/" + String.join("/", parts);
182 }

184 private static int compareIp(String a, String b) {
185 String[] left = a.split("\\."), right = b.split("\\.");
186 for (int i = 0; i < 4; i++) {
187 int n = Integer.compare(Integer.parseInt(left[i]), Integer.parseInt(right[i]));
188 if (n != 0) return n;
189 }
190 return 0;
191 }

193 private static boolean statusMatches(String filter, int status) {
194 return CLASSES.contains(filter) ?
195 status / 100 == filter.charAt(0) - '0' : status == Integer.parseInt(filter);
196 }

198 private static String decimal(long numerator, long denominator, int places) {
199 long scale = switch (places) { case 1 -> 10; case 2 -> 100; default -> 1000; };
200 long number = numerator * scale / denominator;
201 return number / scale + "." +
202 String.format(Locale.ROOT, "%0" + places + "d", number % scale);
203 }
204
205 private static String percent(int part, int whole) {
206 return decimal(part * 100L, whole, 2) + "%";
207 }

209 private static String bar(int count, int maximum) {
210 return "#".repeat((count * 30 + maximum - 1) / maximum);
211 }

213 private static String bytesText(long bytes) {
214 if (bytes >= 1073741824) return decimal(bytes, 1073741824, 1) + "GB";
215 if (bytes >= 1048576) return decimal(bytes, 1048576, 1) + "MB";
216 if (bytes >= 1024) return decimal(bytes, 1024, 1) + "KB";
217 return bytes + "B";
218 }

220 private static Map<String, Integer> counts(Collection<String> values) {
221 Map<String, Integer> result = new HashMap<>();
222 for (String item : values) result.merge(item, 1, Integer::sum);
223 return result;

226 private static List<String> byCount(Map<String, Integer> counts, Comparator<String> tie) {
227 List<String> result = new ArrayList<>(counts.keySet());
228 result.sort((a, b) -> {
229 int n = Integer.compare(counts.get(b), counts.get(a));
230 return n != 0 ? n : tie.compare(a, b);
231 });
232 return result;
233 }

235 private static List<String> byCount(Map<String, Integer> counts) {
236 return byCount(counts, String::compareTo);
237 }

244 Row(int line, List<String> parts) {

261 for (Row row : rows) {
262 result.computeIfAbsent(key.apply(row), k -> new ArrayList<>()).add(row);
263 }

275 private static Buckets buckets(List<Long> values) {
276 long first = Collections.min(values);
277 long last = Collections.max(values);
278 check(last - first + 1 <= 60, "too many buckets");
279 int[] counts = new int[(int)(last - first + 1)];
280 for (long value : values) counts[(int)(value - first)]++;
281 return new Buckets(first, counts);
282 }

288 private record Peak(int size, Row start, Row end) { }
289
290 private static Peak peak(List<Row> rows, int width) {
291 int first = 0, size = 0, start = 0, end = 0;
292 for (int last = 0; last < rows.size(); last++) {
293 while (rows.get(last).seconds - rows.get(first).seconds >= width) first++;
294 if (last - first + 1 > size) {
295 size = last - first + 1;
296 start = first;
297 end = last;
298 }
299 }
300 return new Peak(size, rows.get(start), rows.get(end));
301 }

303 private record Apdex(int count, int satisfied, int tolerating, int frustrated,
304 long points, long total, String notation, String rating) { }
305
306 private static Apdex apdex(List<Row> rows, int threshold) {
307 int satisfied = 0, tolerating = 0;
308 for (Row row : rows) {
309 if (row.status >= 500) continue;
310 if (row.latency <= threshold) satisfied++;
311 else if (row.latency <= 4 * threshold) tolerating++;
312 }
313 int n = rows.size();
314 long points = satisfied * 2L + tolerating, total = n * 2L;
315 String rating = points * 100 >= total * 94 ? "excellent" :
316 points * 100 >= total * 85 ? "good" :
317 points * 100 >= total * 70 ? "fair" :
318 points * 100 >= total * 50 ? "poor" : "unacceptable";
319 return new Apdex(n, satisfied, tolerating, n - satisfied - tolerating,
320 points, total, decimal(points, total, 3), rating);
321 }

323 private static class Analyzer {
324 private final List<Row> records = new ArrayList<>();
325 private final List<Map.Entry<Integer, String>> malformed = new ArrayList<>();
326 private final int lineCount;
327 private final Map<String, String> filters = new HashMap<>();
328 private Long from, to;
329 private String fromLabel, toLabel;
330 final List<String> output = new ArrayList<>();
331
332 Analyzer(List<String> lines) {
333 lineCount = lines.size();
334 for (int i = 0; i < lines.size(); i++) {
335 List<String> fields = tokens(lines.get(i));
336 String reason = malformedReason(fields);
337 if (reason != null) malformed.add(Map.entry(i + 1, reason));
338 else records.add(new Row(i + 1, fields));
339 }
340 records.sort(Comparator.comparingLong((Row r) -> r.seconds)
341 .thenComparingInt(r -> r.line));
342 for (String key : KEYS) filters.put(key, null);
343 }
344
345 private void emit(Object... values) {
346 output.add(Arrays.stream(values).map(String::valueOf)
347 .collect(Collectors.joining(" ")));
348 }
349
350 private List<Row> selected() {
351 return records.stream().filter(r ->
352 (from == null || r.seconds >= from && r.seconds <= to) &&
353 (filters.get("METHOD") == null || r.method.equals(filters.get("METHOD"))) &&
354 (filters.get("STATUS") == null ||
355 statusMatches(filters.get("STATUS"), r.status)) &&
356 (filters.get("IP") == null || r.ip.equals(filters.get("IP"))) &&
357 (filters.get("PATH") == null || r.path.equals(filters.get("PATH"))))
358 .collect(Collectors.toList());
359 }
360
361 private void barLine(String label, int count, int maximum) {
362 if (count == 0) emit(label, count);
363 else emit(label, count, bar(count, maximum));
364 }
365
366 void execute(String line) {
367 List<String> parts = tokens(line);
368 String name = parts.isEmpty() ? "" : parts.get(0);
369 List<String> args = parts.isEmpty() ? List.of() : parts.subList(1, parts.size());
370 try {
371 check(FORMS.containsKey(name), "unknown query");
372 List<String> form = FORMS.get(name);
373 boolean optional = (name.equals("LATENCY") || name.equals("SESSIONS")) &&
374 args.size() == form.size() - 1;
375 check(args.size() == form.size() || optional, "wrong arguments");
376 for (int i = 0; i < args.size(); i++) {
377 check(validArgument(form.get(i), args.get(i), args), ERRORS.get(form.get(i)));
378 }
379 List<Row> rows = selected();
380 if (REQUIRED.contains(name) && rows.isEmpty()) {
381 emit(name, "no records");
382 return;
383 }
384 run(name, args, rows);
385 } catch (QueryError error) {
386 emit("ERROR", error.getMessage());
387 }
388 }

390 private void run(String name, List<String> a, List<Row> rows) {
391 switch (name) {
392 case "SUMMARY" -> {
393 emit("SUMMARY", "lines", lineCount, "valid", records.size(),
394 "malformed", malformed.size());
395 emit("requests", rows.size(), "bytes",
396 rows.stream().mapToLong(r -> r.bytes).sum(),
397 "ips", rows.stream().map(r -> r.ip).distinct().count(),
398 "paths", rows.stream().map(r -> r.path).distinct().count());
399 if (rows.isEmpty()) emit("span", "none");
400 else emit("span", rows.getFirst().timestamp, rows.getLast().timestamp,
401 rows.getLast().seconds - rows.getFirst().seconds);
402 }
403 case "MALFORMED" -> {
404 emit("MALFORMED", malformed.size());
405 for (String reason : REASONS) {
406 List<Integer> indices = malformed.stream()
407 .filter(x -> x.getValue().equals(reason))
408 .map(Map.Entry::getKey).toList();
409 if (!indices.isEmpty()) {
410 List<Object> parts = new ArrayList<>();
411 parts.add(reason);
412 parts.add(indices.size());
413 parts.addAll(indices);
414 emit(parts.toArray());
415 }
416 }
417 }
418 case "STATUS" -> {
419 emit("STATUS", rows.size());
420 int[] classes = classCounts(rows);
421 for (int i = 0; i < 5; i++)
422 emit(CLASSES.get(i), classes[i], percent(classes[i], rows.size()));
423 Map<Integer, Long> status = rows.stream().collect(
424 Collectors.groupingBy(r -> r.status, Collectors.counting()));
425 for (int code : new TreeSet<>(status.keySet())) {
426 int count = status.get(code).intValue();
427 emit(code, count, percent(count, rows.size()));
428 }
429 }
430 case "METHODS" -> {
431 Map<String, Integer> counts = counts(
432 rows.stream().map(r -> r.method).toList());
433 emit("METHODS", rows.size());
434 for (String method : byCount(counts))
435 emit(method, counts.get(method), percent(counts.get(method), rows.size()));
436 }
437 case "TOP" -> {
438 String kind = a.get(0);
439 Map<String, Integer> counts = counts(rows.stream()
440 .map(r -> kind.equals("PATHS") ? r.path : r.ip).toList());
441 List<String> ranked = byCount(counts,
442 kind.equals("IPS") ? Main::compareIp : String::compareTo);
443 int shown = Math.min(Integer.parseInt(a.get(1)), ranked.size());
444 emit("TOP", kind, shown, "of", ranked.size());
445 int rank = 0;
446 for (int i = 0; i < shown; i++) {
447 if (i == 0 || !counts.get(ranked.get(i)).equals(counts.get(ranked.get(i-1))))
448 rank = i + 1;
449 String value = ranked.get(i);
450 emit(rank, value, counts.get(value), percent(counts.get(value), rows.size()));
451 }
452 }
453 case "HOURLY" -> {
454 int[] counts = new int[24];
455 for (Row r : rows) counts[r.hour]++;
456 int maximum = Arrays.stream(counts).max().orElseThrow();
457 int first = 0;
458 while (counts[first] != maximum) first++;
459 emit("HOURLY", rows.size(), "peak",
460 String.format(Locale.ROOT, "%02d", first), maximum);
461 for (int h = 0; h < 24; h++)
462 barLine(String.format(Locale.ROOT, "%02d", h), counts[h], maximum);
463 }
464 case "TIMELINE" -> {
465 int minutes = Integer.parseInt(a.get(0));
466 long width = minutes * 60L;
467 Buckets buckets = buckets(rows.stream().map(r -> r.seconds / width).toList());
468 emit("TIMELINE", minutes, buckets.counts.length);
469 int maximum = Arrays.stream(buckets.counts).max().orElseThrow();
470 for (int i = 0; i < buckets.counts.length; i++)
471 barLine(intervalLabel((buckets.first + i) * width), buckets.counts[i], maximum);
472 }
473 case "BANDWIDTH" -> {
474 Map<String, List<Row>> days = groups(rows, r -> r.date);
475 long volume = rows.stream().mapToLong(r -> r.bytes).sum();
476 emit("BANDWIDTH", days.size(), volume, bytesText(volume));
477 for (String date : new TreeSet<>(days.keySet())) {
478 List<Row> part = days.get(date);
479 long bytes = part.stream().mapToLong(r -> r.bytes).sum();
480 emit(date, part.size(), bytes, bytesText(bytes));
481 }
482 }
483 case "VISITORS" -> {
484 Map<String, String> first = new HashMap<>();
485 for (Row r : rows) {
486 if (!first.containsKey(r.ip) || r.date.compareTo(first.get(r.ip)) < 0)
487 first.put(r.ip, r.date);
488 }
489 Map<String, List<Row>> days = groups(rows, r -> r.date);
490 emit("VISITORS", days.size(), first.size());
491 for (String date : new TreeSet<>(days.keySet())) {
492 Set<String> ips = days.get(date).stream()
493 .map(r -> r.ip).collect(Collectors.toSet());
494 long fresh = ips.stream().filter(ip -> first.get(ip).equals(date)).count();
495 emit(date, "unique", ips.size(), "new", fresh,
496 "returning", ips.size() - fresh);
497 }
498 }
499 case "LATENCY" -> {
500 String label = a.isEmpty() ? "all" : normalizePath(a.get(0));
501 List<Row> chosen = a.isEmpty() ? rows : rows.stream()
502 .filter(r -> r.path.equals(label)).toList();
503 if (chosen.isEmpty()) {
504 emit("LATENCY", "no records");
505 break;
506 }
507 List<Integer> numbers = chosen.stream().map(r -> r.latency)
508 .sorted().toList();
509 int n = numbers.size(), trim = n / 10;
510 List<Integer> trimmed = numbers.subList(trim, n-trim);
511 long sum = numbers.stream().mapToLong(i -> i).sum();
512 long trimmedSum = trimmed.stream().mapToLong(i -> i).sum();
513 emit("LATENCY", label, "count", n, "min", numbers.getFirst(),
514 "p50", percentile(numbers, 50), "p90", percentile(numbers, 90),
515 "p99", percentile(numbers, 99), "max", numbers.getLast(),
516 "mean", decimal(sum, n, 2),
517 "trimmed", decimal(trimmedSum, trimmed.size(), 2));
518 }
519 case "HISTOGRAM" -> {
520 int width = Integer.parseInt(a.get(0));
521 Buckets buckets = buckets(rows.stream()
522 .map(r -> (long)(r.latency / width)).toList());
523 emit("HISTOGRAM", width, buckets.counts.length);
524 int maximum = Arrays.stream(buckets.counts).max().orElseThrow();
525 for (int i = 0; i < buckets.counts.length; i++) {
526 long lower = (buckets.first + i) * width;
527 barLine(lower + "-" + (lower + width - 1), buckets.counts[i], maximum);
528 }
529 }
530 case "APDEX" -> {
531 int threshold = Integer.parseInt(a.get(0));
532 Apdex overall = apdex(rows, threshold);
533 emit("APDEX", threshold, overall.notation, overall.rating,
534 "satisfied", overall.satisfied, "tolerating", overall.tolerating,
535 "frustrated", overall.frustrated);
536 record PathScore(String path, Apdex score) { }
537 List<PathScore> paths = groups(rows, r -> r.path).entrySet().stream()
538 .map(e -> new PathScore(e.getKey(), apdex(e.getValue(), threshold)))
539 .collect(Collectors.toCollection(ArrayList::new));
540 paths.sort((x, y) -> {
541 int n = Long.compare(x.score.points * y.score.total,
542 y.score.points * x.score.total);
543 if (n != 0) return n;
544 n = Integer.compare(y.score.count, x.score.count);
545 return n != 0 ? n : x.path.compareTo(y.path);
546 });
547 for (PathScore p : paths)
548 emit(p.path, p.score.notation, p.score.rating, p.score.count);
549 }
550 case "ERRORS" -> {
551 long failed = rows.stream().filter(r -> r.status >= 400).count();
552 emit("ERRORS", failed, "of", rows.size(), percent((int)failed, rows.size()));
553 record ErrorPath(String path, int count, int client, int server) { }
554 List<ErrorPath> paths = new ArrayList<>();
555 for (var entry : groups(rows, r -> r.path).entrySet()) {
556 List<Row> group = entry.getValue();
557 int client = (int)group.stream().filter(r -> r.status / 100 == 4).count();
558 int server = (int)group.stream().filter(r -> r.status / 100 == 5).count();
559 if (client + server > 0)
560 paths.add(new ErrorPath(entry.getKey(), group.size(), client, server));
561 }
562 paths.sort((x, y) -> {
563 long af = x.client + x.server, bf = y.client + y.server;
564 int c = Long.compare(bf * x.count, af * y.count);
565 if (c != 0) return c;
566 c = Long.compare(bf, af);
567 return c != 0 ? c : x.path.compareTo(y.path);
568 });
569 for (ErrorPath p : paths)
570 emit(p.path, p.count, p.client, p.server,
571 percent(p.client + p.server, p.count));
572 }
573 case "SLOWEST" -> {
574 record Endpoint(String method, String path, int count, long sum, int max) { }
575 List<Endpoint> endpoints = new ArrayList<>();
576 for (var entry : groups(rows, r -> r.method + "\u0001" + r.path).entrySet()) {
577 String[] pair = entry.getKey().split("\u0001", 2);
578 List<Row> group = entry.getValue();
579 endpoints.add(new Endpoint(pair[0], pair[1], group.size(),
580 group.stream().mapToLong(r -> r.latency).sum(),
581 group.stream().mapToInt(r -> r.latency).max().orElseThrow()));
582 }
583 endpoints.sort((x, y) -> {
584 int n = Long.compare(y.sum * x.count, x.sum * y.count);
585 if (n != 0) return n;
586 n = Integer.compare(y.max, x.max);
587 if (n != 0) return n;
588 n = x.path.compareTo(y.path);
589 return n != 0 ? n : x.method.compareTo(y.method);
590 });
591 int limit = Math.min(Integer.parseInt(a.get(0)), endpoints.size());
592 emit("SLOWEST", limit, "of", endpoints.size());
593 for (int i = 0; i < limit; i++) {
594 Endpoint e = endpoints.get(i);
595 emit(i+1, e.method, e.path, "count", e.count,
596 "avg", decimal(e.sum, e.count, 2), "max", e.max);
597 }
598 }
599 case "SESSIONS" -> {
600 String ip = a.get(0);
601 int minutes = a.size() == 2 ? Integer.parseInt(a.get(1)) : 30;
602 List<Row> chosen = rows.stream().filter(r -> r.ip.equals(ip)).toList();
603 if (chosen.isEmpty()) {
604 emit("SESSIONS", "no records");
605 break;
606 }
607 List<List<Row>> sessions = new ArrayList<>();
608 sessions.add(new ArrayList<>(List.of(chosen.getFirst())));
609 for (int i = 1; i < chosen.size(); i++) {
610 Row next = chosen.get(i);
611 if (next.seconds - chosen.get(i-1).seconds <= minutes * 60L)
612 sessions.getLast().add(next);
613 else sessions.add(new ArrayList<>(List.of(next)));
614 }
615 long duration = 0;
616 for (List<Row> session : sessions)
617 duration += session.getLast().seconds - session.getFirst().seconds;
618 emit("SESSIONS", ip, "timeout", minutes, "count", sessions.size(),
619 "requests", chosen.size(),
620 "avg_duration", decimal(duration, sessions.size(), 1),
621 "avg_requests", decimal(chosen.size(), sessions.size(), 1));
622 for (int i = 0; i < sessions.size(); i++) {
623 List<Row> session = sessions.get(i);
624 emit(i + 1, session.getFirst().timestamp, session.getLast().timestamp,
625 session.getLast().seconds - session.getFirst().seconds,
626 session.size(), session.stream().map(r -> r.path).distinct().count());
627 }
628 }
629 case "BURST" -> {
630 int length = Integer.parseInt(a.get(0));
631 int threshold = Integer.parseInt(a.get(1));
632 record Burst(String ip, Peak window) { }
633 List<Burst> bursts = new ArrayList<>();
634 Map<String, List<Row>> failed = groups(
635 rows.stream().filter(r -> r.status >= 400).toList(), r -> r.ip);
636 for (var group : failed.entrySet()) {
637 Peak window = peak(group.getValue(), length);
638 if (window.size >= threshold) bursts.add(new Burst(group.getKey(), window));
639 }
640 bursts.sort((x, y) -> {
641 int n = Integer.compare(y.window.size, x.window.size);
642 return n != 0 ? n : compareIp(x.ip, y.ip);
643 });
644 emit("BURST", length, threshold, bursts.size());
645 for (Burst b : bursts)
646 emit(b.ip, b.window.size, b.window.start.timestamp, b.window.end.timestamp);
647 }
648 case "PEAK" -> {
649 int length = Integer.parseInt(a.get(0));
650 Peak window = peak(rows, length);
651 emit("PEAK", length, window.size, window.start.timestamp, window.end.timestamp);
652 }
653 case "GAPS" -> {
654 int length = Integer.parseInt(a.get(0));
655 List<Map.Entry<Row, Row>> gaps = new ArrayList<>();
656 for (int i = 1; i < rows.size(); i++) {
657 if (rows.get(i).seconds - rows.get(i-1).seconds >= length)
658 gaps.add(Map.entry(rows.get(i-1), rows.get(i)));
659 }
660 emit("GAPS", length, gaps.size());
661 for (var gap : gaps)
662 emit(gap.getKey().timestamp, gap.getValue().timestamp,
663 gap.getValue().seconds - gap.getKey().seconds);
664 }
665 case "IP" -> {
666 String ip = a.get(0);
667 List<Row> found = rows.stream().filter(r -> r.ip.equals(ip)).toList();
668 if (found.isEmpty()) {
669 emit("IP", "no records");
670 break;
671 }
672 Map<String, Integer> paths = counts(
673 found.stream().map(r -> r.path).toList());
674 emit("IP", ip, "requests", found.size(), "bytes",
675 found.stream().mapToLong(r -> r.bytes).sum(),
676 "errors", found.stream().filter(r -> r.status >= 400).count(),
677 "paths", paths.size());
678 emit("first", found.getFirst().timestamp, "last", found.getLast().timestamp);
679 Map<String, Integer> methods = counts(
680 found.stream().map(r -> r.method).toList());
681 List<Object> methodLine = new ArrayList<>(List.of("methods"));
682 for (String method : byCount(methods)) {
683 methodLine.add(method);
684 methodLine.add(methods.get(method));
685 }
686 emit(methodLine.toArray());
687 List<Object> classLine = new ArrayList<>(List.of("classes"));
688 int[] classes = classCounts(found);
689 for (int i = 0; i < 5; i++) {
690 classLine.add(CLASSES.get(i));
691 classLine.add(classes[i]);
692 }
693 emit(classLine.toArray());
694 for (String path : byCount(paths).stream().limit(3).toList())
695 emit("path", path, paths.get(path));
696 }
697 case "RANGE" -> {
698 long start = secondsOf(a.get(0), a.get(1));
699 long end = secondsOf(a.get(2), a.get(3));
700 check(start <= end, "invalid range");
701 from = start;
702 to = end;
703 fromLabel = a.get(0) + " " + a.get(1);
704 toLabel = a.get(2) + " " + a.get(3);
705 emit("RANGE", fromLabel, toLabel, "active", selected().size());
706 }
707 case "FILTER" -> {
708 String key = a.get(0);
709 String value = key.equals("PATH") ? normalizePath(a.get(1)) : a.get(1);
710 filters.put(key, value);
711 emit("FILTER", key, value, "active", selected().size());
712 }
713 case "UNFILTER" -> {
714 String key = a.get(0);
715 if (key.equals("RANGE")) {
716 check(from != null, "not filtered");
717 from = null;
718 to = null;
719 fromLabel = null;
720 toLabel = null;
721 } else {
722 check(filters.get(key) != null, "not filtered");
723 filters.put(key, null);
724 }
725 emit("UNFILTER", key, "active", selected().size());
726 }
727 case "RESET" -> {
728 from = null;
729 to = null;
730 fromLabel = null;
731 toLabel = null;
732 for (String key : KEYS) filters.put(key, null);
733 emit("RESET", "active", selected().size());
734 }
735 case "STATE" -> {
736 List<Object> parts = new ArrayList<>();
737 parts.addAll(List.of("STATE", "active", rows.size(), "range"));
738 parts.add(from == null ? "-" : fromLabel + " " + toLabel);
739 for (String key : KEYS) {
740 parts.add(key.toLowerCase(Locale.ROOT));
741 parts.add(filters.get(key) == null ? "-" : filters.get(key));
742 }
743 emit(parts.toArray());
744 }
745 }
746 }
747 }

749 public static void main(String[] args) throws Exception {
750 BufferedReader input = new BufferedReader(
751 new InputStreamReader(System.in, StandardCharsets.UTF_8));
752 List<String> logs = new ArrayList<>();
753 List<String> queries = new ArrayList<>();
754 String line;
755 boolean readingQueries = false;
756 while ((line = input.readLine()) != null) {
757 if (!readingQueries) {
758 if (line.equals("---")) readingQueries = true;
759 else logs.add(line);
760 } else if (line.equals("END")) break;
761 else queries.add(line);
762 }
763 Analyzer analyzer = new Analyzer(logs);
764 for (String query : queries) analyzer.execute(query);
765 if (!analyzer.output.isEmpty()) System.out.println(String.join("\n", analyzer.output));
766 }
