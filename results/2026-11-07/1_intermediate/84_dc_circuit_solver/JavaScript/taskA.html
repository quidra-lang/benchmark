7function gcd(a, b) {
8 while (b !== 0n) [a, b] = [b, a % b];
9 return a;

11class Rational {
12 constructor(n, d = 1n) {
13 if (d < 0n) {
14 n = -n;
15 d = -d;
16 }
17 const g = gcd(n < 0n ? -n : n, d);
18 this.n = n / g;
19 this.d = d / g;
20 }
21 add(b) {
22 return new Rational(this.n * b.d + b.n * this.d, this.d * b.d);
23 }
24 sub(b) {
25 return new Rational(this.n * b.d - b.n * this.d, this.d * b.d);
26 }
27 mul(b) {
28 return new Rational(this.n * b.n, this.d * b.d);
29 }
30 div(b) {
31 return new Rational(this.n * b.d, this.d * b.n);
32 }
33 neg() {
34 return new Rational(-this.n, this.d);
35 }
36 compare(b) {
37 const difference = this.n * b.d - b.n * this.d;
38 return difference < 0n ? -1 : difference > 0n ? 1 : 0;
39 }
40 get zero() {
41 return this.n === 0n;
42 }
43 get positive() {
44 return this.n > 0n;
45 }
46 toString() {
47 return this.d === 1n ? String(this.n) : `${this.n}/${this.d}`;
48 }
49 value() {
50 const scaled = ((this.n < 0n ? -this.n : this.n) * 1000000n) / this.d;
51 return `${this} (${this.n < 0n ? "-" : ""}${scaled / 1000000n}.${String(scaled % 1000000n).padStart(6, "0")})`;
52 }
53}
54const ZERO = new Rational(0n),
55 ONE = new Rational(1n);
56function readNumber(word) {
57 let text = word,
58 multiplier = ONE;
59 for (const [suffix, value] of [
60 ["meg", new Rational(1000000n)],
61 ["k", new Rational(1000n)],
62 ["m", new Rational(1n, 1000n)],
63 ["u", new Rational(1n, 1000000n)],
64 ]) {
65 if (text.endsWith(suffix)) {
66 text = text.slice(0, -suffix.length);
67 multiplier = value;
68 break;
69 }
70 }
71 const negative = text.startsWith("-");
72 if (negative) text = text.slice(1);
73 if (!/^[0-9]+(?:\.[0-9]+|\/[0-9]+)?$/.test(text)) return null;
74 let numerator, denominator;
75 if (text.includes("/")) {
76 const parts = text.split("/");
77 numerator = BigInt(parts[0]);
78 denominator = BigInt(parts[1]);
79 if (denominator === 0n) return null;
80 } else if (text.includes(".")) {
81 const parts = text.split(".");
82 numerator = BigInt(parts.join(""));
83 denominator = 10n ** BigInt(parts[1].length);
84 } else {
85 numerator = BigInt(text);
86 denominator = 1n;
87 }
88 return new Rational(negative ? -numerator : numerator, denominator).mul(multiplier);
89}

110function solve(a, b) {
111 const n = a.length,
112 rows = a.map((row, i) => [...row, b[i]]);
113 for (let k = 0; k < n; k++) {
114 let r = k;
115 while (r < n && rows[r][k].zero) r++;
116 if (r === n) return null;
117 [rows[k], rows[r]] = [rows[r], rows[k]];
118 const pivot = rows[k][k];
119 for (let j = 0; j <= n; j++) rows[k][j] = rows[k][j].div(pivot);
120 for (let i = 0; i < n; i++)
121 if (i !== k && !rows[i][k].zero) {
122 const factor = rows[i][k];
123 for (let j = 0; j <= n; j++) rows[i][j] = rows[i][j].sub(factor.mul(rows[k][j]));
124 }
125 }
126 return rows.map((row) => row[n]);
127}
128function reachable(start, edges) {
129 const seen = new Set([start]),
130 pending = [start];
131 while (pending.length > 0) {
132 const node = pending.pop();
133 for (const [a, b] of edges) {
134 const next = a === node ? b : b === node ? a : null;
135 if (next !== null && !seen.has(next)) {
136 seen.add(next);
137 pending.push(next);
138 }
139 }
140 }
141 return seen;

143class Groups {
144 constructor(nodes) {
145 this.parent = new Map(nodes.map((node) => [node, node]));
146 }
147 find(node) {
148 const parent = this.parent.get(node);
149 if (parent === node) return node;
150 const root = this.find(parent);
151 this.parent.set(node, root);
152 return root;
153 }
154 join(a, b) {
155 a = this.find(a);
156 b = this.find(b);
157 if (a !== b) this.parent.set(a < b ? b : a, a < b ? a : b);
158 }

162 constructor() {
163 this.elements = [];

165 nodes(includeGround = false) {
166 const names = new Set(this.elements.flatMap((element) => element.terminals));
167 if (includeGround) names.add("0");

170 circuitNodes() {
171 return this.nodes().filter((node) => node !== "0");
172 }
173 elementCheck(name) {
174 const element = this.elements.find((element) => element.name === name);
175 if (element === undefined) fail("unknown element " + name);
176 return element;
177 }
178 nodeCheck(name) {
179 if (name !== "0" && !this.circuitNodes().includes(name)) fail("unknown node " + name);
180 }
181 add(words) {
182 const name = words[0],
183 type = name[0];
184 if (!"RVIEGFH".includes(type)) fail("unknown element type " + name);
185 const expected = "EG".includes(type) ? 6 : "FH".includes(type) ? 5 : 4;
186 if (words.length !== expected) fail("wrong field count for " + name);
187 if (this.elements.some((element) => element.name === name))
188 fail("duplicate element " + name);
189 const terminals = words.slice(1, "EG".includes(type) ? 5 : 3);
190 for (const node of terminals) if (!validNode(node)) fail("invalid node " + node);
191 const reference = "FH".includes(type) ? words[3] : null;
192 if (
193 reference !== null &&
194 !this.elements.some((element) => element.name === reference && element.name[0] === "V")
195 )
196 fail("unknown voltage source " + reference);
197 const value = numberCheck(words[words.length - 1]);
198 if (type === "R" && !value.positive) fail("nonpositive resistance " + name);
199 this.elements.push({ name, terminals, reference, value });
200 }
201 floating() {
202 const edges = this.elements
203 .filter((element) => "RVEH".includes(element.name[0]))
204 .map((element) => [element.terminals[0], element.terminals[1]]);
205 const seen = reachable("0", edges);
206 return this.circuitNodes().filter((node) => !seen.has(node));
207 }
208 loops() {
209 const groups = new Groups(this.nodes(true)),
210 result = [];
211 for (const element of this.elements.filter(voltageType)) {
212 const [a, b] = element.terminals;
213 if (groups.find(a) === groups.find(b)) result.push(element.name);
214 groups.join(a, b);
215 }
216 return result;
217 }
218 structure() {
219 if (this.circuitNodes().length === 0) fail("no nodes");
220 const floating = this.floating();
221 if (floating.length > 0) fail("floating node " + floating[0]);
222 const loops = this.loops();
223 if (loops.length > 0) fail("voltage source loop at " + loops[0]);
224 }
225 equations(overrides = new Map()) {
226 const nodes = this.circuitNodes(),
227 sources = this.elements.filter(voltageType),
228 count = nodes.length + sources.length;
229 const nodeIndex = new Map(nodes.map((node, i) => [node, i]));
230 const sourceIndex = new Map(sources.map((source, i) => [source.name, nodes.length + i]));
231 const a = matrix(count),
232 b = zeros(count);
233 const node = (name) => (name === "0" ? -1 : nodeIndex.get(name));
234 const source = (name) => sourceIndex.get(name);
235 const stamp = (row, column, value) => {
236 if (row >= 0 && column >= 0) a[row][column] = a[row][column].add(value);
237 };
238 const rhs = (row, value) => {
239 if (row >= 0) b[row] = b[row].add(value);
240 };
241 for (const element of this.elements) {
242 const [n1, n2, c1, c2] = element.terminals,
243 type = element.name[0];
244 const x = node(n1),
245 y = node(n2),
246 w = overrides.get(element.name) ?? element.value;
247 if (type === "R") {
248 const g = ONE.div(w);
249 stamp(x, x, g);
250 stamp(y, y, g);
251 stamp(x, y, g.neg());
252 stamp(y, x, g.neg());
253 } else if (type === "I") {
254 rhs(x, w.neg());
255 rhs(y, w);
256 } else if (type === "G") {
257 stamp(x, node(c1), w);
258 stamp(y, node(c2), w);
259 stamp(x, node(c2), w.neg());
260 stamp(y, node(c1), w.neg());
261 } else if (type === "F") {
262 stamp(x, source(element.reference), w);
263 stamp(y, source(element.reference), w.neg());
264 } else {
265 const s = source(element.name);
266 stamp(x, s, ONE);
267 stamp(s, x, ONE);
268 stamp(y, s, ONE.neg());
269 stamp(s, y, ONE.neg());
270 if (type === "V") rhs(s, w);
271 else if (type === "E") {
272 stamp(s, node(c1), w.neg());
273 stamp(s, node(c2), w);
274 } else stamp(s, source(element.reference), w.neg());
275 }
276 }
277 return { nodes, sources, count, nodeIndex, sourceIndex, a, b };
278 }
279 potential(equations, solution, name) {
280 return name === "0" ? ZERO : solution[equations.nodeIndex.get(name)];
281 }
282 sourceCurrent(equations, solution, name) {
283 return solution[equations.sourceIndex.get(name)];
284 }
285 current(equations, solution, element) {
286 const [n1, n2, c1, c2] = element.terminals,
287 type = element.name[0];
288 if (type === "R")
289 return this.potential(equations, solution, n1)
290 .sub(this.potential(equations, solution, n2))
291 .div(element.value);
292 if (type === "I") return element.value;
293 if (type === "G")
294 return element.value.mul(
295 this.potential(equations, solution, c1).sub(
296 this.potential(equations, solution, c2),
297 ),
298 );
299 if (type === "F")
300 return element.value.mul(this.sourceCurrent(equations, solution, element.reference));
301 return this.sourceCurrent(equations, solution, element.name);
302 }
303 injection(equations, a, b) {
304 const column = zeros(equations.count);
305 if (a !== "0") {
306 const i = equations.nodeIndex.get(a);
307 column[i] = column[i].add(ONE);
308 }
309 if (b !== "0") {
310 const i = equations.nodeIndex.get(b);
311 column[i] = column[i].sub(ONE);
312 }
313 return solve(equations.a, column);
314 }
315 alone(equations, element, unit) {
316 const column = zeros(equations.count),
317 value = unit ? ONE : element.value;
318 if (element.name[0] === "V") column[equations.sourceIndex.get(element.name)] = value;
319 else {
320 const [a, b] = element.terminals;
321 if (a !== "0") {
322 const i = equations.nodeIndex.get(a);
323 column[i] = column[i].sub(value);
324 }
325 if (b !== "0") {
326 const i = equations.nodeIndex.get(b);
327 column[i] = column[i].add(value);
328 }
329 }
330 return solve(equations.a, column);
331 }
332 resistance(a, b) {
333 const groups = new Groups(this.nodes(true));
334 for (const element of this.elements)
335 if (element.name[0] === "V") groups.join(element.terminals[0], element.terminals[1]);
336 a = groups.find(a);
337 b = groups.find(b);
338 if (a === b) return ["Req = " + ZERO.value()];
339 const edges = [];
340 for (const element of this.elements)
341 if (element.name[0] === "R") {
342 const x = groups.find(element.terminals[0]),
343 y = groups.find(element.terminals[1]);
344 if (x !== y) edges.push({ x, y, value: ONE.div(element.value) });
345 }
346 const seen = reachable(
347 a,
348 edges.map((edge) => [edge.x, edge.y]),
349 );
350 if (!seen.has(b)) return ["Req = INFINITE"];
351 const nodes = [...seen].filter((node) => node !== b).sort(),
352 indices = new Map(nodes.map((node, i) => [node, i]));
353 const g = matrix(nodes.length),
354 column = zeros(nodes.length);
355 const stamp = (row, col, value) => {
356 if (row !== b && col !== b) {
357 const i = indices.get(row),
358 j = indices.get(col);
359 g[i][j] = g[i][j].add(value);
360 }
361 };
362 for (const edge of edges)
363 if (seen.has(edge.x)) {
364 stamp(edge.x, edge.x, edge.value);
365 stamp(edge.y, edge.y, edge.value);
366 stamp(edge.x, edge.y, edge.value.neg());
367 stamp(edge.y, edge.x, edge.value.neg());
368 }
369 column[indices.get(a)] = ONE;
370 return ["Req = " + solve(g, column)[indices.get(a)].value()];
371 }
372 reduce(a, b) {
373 if (a === b) return ["Req = " + ZERO.value()];
374 let list = this.elements
375 .filter((element) => element.name[0] === "R")
376 .map((element) => ({
377 name: element.name,
378 a: element.terminals[0],
379 b: element.terminals[1],
380 value: element.value,
381 }));
382 const connected = reachable(
383 a,
384 list.map((element) => [element.a, element.b]),
385 );
386 if (!connected.has(b)) return ["Req = INFINITE"];
387 list = list.filter((element) => connected.has(element.a));
388 const output = [];
389 let serial = 0;
390 while (true) {
391 let remove = list.findIndex((element) => element.a === element.b);
392 if (remove >= 0) {
393 output.push("remove " + list[remove].name);
394 list.splice(remove, 1);
395 continue;
396 }
397 const degree = new Map();
398 for (const element of list)
399 for (const node of [element.a, element.b])
400 degree.set(node, (degree.get(node) ?? 0) + 1);
401 const internal = [...degree.keys()].filter((node) => node !== a && node !== b).sort();
402 const dangling = internal.find((node) => degree.get(node) === 1);
403 if (dangling !== undefined) {
404 remove = list.findIndex(
405 (element) => element.a === dangling || element.b === dangling,
406 );
407 output.push("remove " + list[remove].name);
408 list.splice(remove, 1);
409 continue;
410 }
411 let first = -1,
412 second = -1,
413 parallel = false;
414 search: for (let i = 0; i < list.length; i++)
415 for (let j = i + 1; j < list.length; j++) {
416 const x = list[i],
417 y = list[j];
418 if ((x.a === y.a && x.b === y.b) || (x.a === y.b && x.b === y.a)) {
419 first = i;
420 second = j;
421 parallel = true;
422 break search;
423 }
424 }
425 let center = null;
426 if (!parallel) {
427 center = internal.find((node) => degree.get(node) === 2) ?? null;
428 if (center === null) break;
429 const indices = list
430 .map((element, i) => (element.a === center || element.b === center ? i : -1))
431 .filter((i) => i >= 0);
432 [first, second] = indices;
433 }
434 const x = list[first],
435 y = list[second],
436 name = "#" + ++serial;
437 const value = parallel
438 ? x.value.mul(y.value).div(x.value.add(y.value))
439 : x.value.add(y.value);
440 const left = parallel ? x.a : x.a === center ? x.b : x.a;
441 const right = parallel ? x.b : y.a === center ? y.b : y.a;
442 output.push(
443 `${name} = ${x.name} ${parallel ? "||" : "+"} ${y.name} = ${value.value()}`,
444 );
445 list.splice(second, 1);
446 list.splice(first, 1);
447 list.push({ name, a: left, b: right, value });
448 }
449 output.push(
450 list.length === 1
451 ? "Req = " + list[0].value.value()
452 : `not series-parallel: ${list.length} resistors remain`,
453 );
454 return output;
455 }
456 check() {
457 const nodes = this.circuitNodes();
458 if (nodes.length === 0) return ["no nodes"];
459 const output = [];
460 for (const node of nodes) {
461 let degree = 0;
462 for (const element of this.elements) {
463 if (element.terminals[0] === node) degree++;
464 if (element.terminals[1] === node) degree++;
465 }
466 if (degree < 2) output.push("dangling node " + node);
467 }
468 for (const node of this.floating()) output.push("floating node " + node);
469 for (const name of this.loops()) output.push("voltage source loop at " + name);
470 if (output.length === 0) {
471 const eq = this.equations();
472 if (solve(eq.a, eq.b) === null) output.push("singular matrix");
473 }
474 return output.length === 0 ? ["ok"] : output;
475 }

478 args = words.slice(1);

552 if (command === ".SWEEP") {

555 to = numberCheck(args[2]),
556 step = numberCheck(args[3]);
557 this.nodeCheck(args[4]);

586 else if (command === ".TF") {

592 ".VOLTAGE",
593 ".KCL",
594 ".THEVENIN",
595 ".NORTON",

597 ".SENS",

609 "unknowns: " +
610 [
611 ...eq.nodes.map((node) => `V(${node})`),
612 ...eq.sources.map((element) => `I(${element.name})`),
613 ].join(" "),
614 ...labels.map((label, i) => `${label}: ${eq.a[i].join(" ")} | ${eq.b[i]}`),
615 ];
616 }
617 const solution = solve(eq.a, eq.b);
618 if (solution === null) fail("singular matrix");
619 const potential = (name, values = solution) => this.potential(eq, values, name);
620 const output = [];
621 if (command === ".OP") {
622 for (const node of eq.nodes) output.push(`V(${node}) = ${potential(node).value()}`);
623 for (const element of eq.sources)
624 output.push(
625 `I(${element.name}) = ${this.sourceCurrent(eq, solution, element.name).value()}`,
626 );

628 output.push(`I(${selected.name}) = ${this.current(eq, solution, selected).value()}`);

635 delivered = ZERO,
636 total = ZERO;

640 .mul(this.current(eq, solution, element));
641 output.push(`P(${element.name}) = ${power.value()}`);
642 if (power.positive) absorbed = absorbed.add(power);
643 else if (!power.zero) delivered = delivered.sub(power);

645 }
646 output.push(
647 "absorbed = " + absorbed.value(),
648 "delivered = " + delivered.value(),
649 "total = " + total.value(),
650 );
651 } else if (command === ".KCL") {
652 let sum = ZERO;
653 for (const element of this.elements)
654 if (element.terminals[0] === args[0] || element.terminals[1] === args[0]) {
655 const current =
656 element.terminals[0] === element.terminals[1]
657 ? ZERO
658 : element.terminals[0] === args[0]
659 ? this.current(eq, solution, element)
660 : this.current(eq, solution, element).neg();
661 output.push(`I(${element.name}) leaving ${args[0]} = ${current.value()}`);
662 sum = sum.add(current);
663 }
664 output.push("sum = " + sum.value());
665 } else if ([".THEVENIN", ".NORTON", ".MAXPOWER"].includes(command)) {
666 const voltage = potential(args[0]).sub(potential(args[1])),
667 z = this.injection(eq, args[0], args[1]);
668 const resistance = potential(args[0], z).sub(potential(args[1], z));
669 if (command === ".THEVENIN")
670 output.push("Vth = " + voltage.value(), "Rth = " + resistance.value());
671 else if (command === ".NORTON") {
672 if (resistance.zero) fail("zero thevenin resistance");
673 output.push(
674 "In = " + voltage.div(resistance).value(),
675 "Rn = " + resistance.value(),
676 );
677 } else {
678 if (!resistance.positive) fail("nonpositive thevenin resistance");
679 output.push(
680 "RL = " + resistance.value(),
681 "Pmax = " + voltage.mul(voltage).div(new Rational(4n).mul(resistance)).value(),
682 );
683 }
684 } else if (command === ".TF") {
685 const z = this.alone(eq, selected, true);
686 output.push("gain = " + potential(args[0], z).value());
687 if (selected.name[0] === "V") {
688 const current = this.sourceCurrent(eq, z, selected.name);
689 output.push(
690 "input resistance = " +
691 (current.zero ? "INFINITE" : ONE.neg().div(current).value()),
692 );
693 } else
694 output.push(
695 "input resistance = " +
696 potential(selected.terminals[1], z)
697 .sub(potential(selected.terminals[0], z))
698 .value(),
699 );
700 output.push(
701 "output resistance = " +
702 potential(args[0], this.injection(eq, args[0], "0")).value(),
703 );
704 } else if (command === ".SENS") {
705 for (const element of this.elements) {
706 const column = zeros(eq.count),
707 [a, b, c, d] = element.terminals;
708 const add = (node, value) => {
709 if (node !== "0") {
710 const i = eq.nodeIndex.get(node);
711 column[i] = column[i].add(value);
712 }
713 };
714 const type = element.name[0];
715 if (type === "R") {
716 const t = this.current(eq, solution, element).div(element.value);
717 add(a, t);
718 add(b, t.neg());
719 } else if (type === "I") {
720 add(a, ONE.neg());
721 add(b, ONE);
722 } else if (type === "G" || type === "F") {
723 const t =
724 type === "G"
725 ? potential(c).sub(potential(d))
726 : this.sourceCurrent(eq, solution, element.reference);
727 add(a, t.neg());
728 add(b, t);
729 } else
730 column[eq.sourceIndex.get(element.name)] =
731 type === "V"
732 ? ONE
733 : type === "E"
734 ? potential(c).sub(potential(d))
735 : this.sourceCurrent(eq, solution, element.reference);
736 output.push(
737 `dV(${args[0]})/d${element.name} = ${potential(args[0], solve(eq.a, column)).value()}`,
738 );
739 }
740 } else if (command === ".ZPARAM") {
741 const first = this.injection(eq, args[0], args[1]),
742 second = this.injection(eq, args[2], args[3]);
743 output.push(
744 "z11 = " + potential(args[0], first).sub(potential(args[1], first)).value(),
745 "z12 = " + potential(args[0], second).sub(potential(args[1], second)).value(),
746 "z21 = " + potential(args[2], first).sub(potential(args[3], first)).value(),
747 "z22 = " + potential(args[2], second).sub(potential(args[3], second)).value(),
748 );
749 } else if (command === ".SUPERPOSITION") {
750 let total = ZERO;
751 for (const element of this.elements.filter(independent)) {
752 const value = potential(args[0], this.alone(eq, element, false));
753 output.push(`${element.name}: V(${args[0]}) = ${value.value()}`);
754 total = total.add(value);
755 }
756 output.push(`total: V(${args[0]}) = ${total.value()}`);
757 }
758 return output;
759 }
760}
