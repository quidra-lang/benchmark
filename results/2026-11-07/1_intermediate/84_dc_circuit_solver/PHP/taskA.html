4class Fraction
5{
6 public int $p;
7 public int $q;
8
9 public function __construct(int $p = 0, int $q = 1)
10 {
11 if ($q < 0) {
12 $p = -$p;
13 $q = -$q;
14 }
15 $a = abs($p);
16 $b = $q;
17 while ($b !== 0) {
18 [$a, $b] = [$b, $a % $b];
19 }
20 $this->p = intdiv($p, $a);
21 $this->q = intdiv($q, $a);
22 }
23
24 public function add(Fraction $other): Fraction
25 {
26 return new Fraction($this->p * $other->q + $other->p * $this->q, $this->q * $other->q);
27 }
28
29 public function sub(Fraction $other): Fraction
30 {
31 return new Fraction($this->p * $other->q - $other->p * $this->q, $this->q * $other->q);
32 }
33
34 public function mul(Fraction $other): Fraction
35 {
36 return new Fraction($this->p * $other->p, $this->q * $other->q);
37 }
38
39 public function div(Fraction $other): Fraction
40 {
41 return new Fraction($this->p * $other->q, $this->q * $other->p);
42 }
43
44 public function neg(): Fraction
45 {
46 return new Fraction(-$this->p, $this->q);
47 }
48
49 public function compare(Fraction $other): int
50 {
51 return ($this->p * $other->q) <=> ($other->p * $this->q);
52 }
53
54 public function fraction(): string
55 {
56 return $this->q === 1 ? (string)$this->p : $this->p . '/' . $this->q;
57 }
58
59 public function formatted(): string
60 {
61 $scaled = intdiv(abs($this->p) * 1000000, $this->q);
62 $decimal = ($this->p < 0 ? '-' : '') . intdiv($scaled, 1000000) . '.' . str_pad((string)($scaled % 1000000), 6, '0', STR_PAD_LEFT);
63 return $this->fraction() . ' (' . $decimal . ')';
64 }
65}

78function rationalNumber(string $text): ?Fraction
79{
80 $multiplier = new Fraction(1);
81 foreach (['meg' => new Fraction(1000000), 'k' => new Fraction(1000), 'm' => new Fraction(1, 1000), 'u' => new Fraction(1, 1000000)] as $suffix => $value) {
82 if (str_ends_with($text, $suffix)) {
83 $text = substr($text, 0, -strlen($suffix));
84 $multiplier = $value;
85 break;
86 }
87 }
88 if (!preg_match('/\A-?[0-9]+(?:\.[0-9]+|\/[0-9]+)?\z/', $text)) {
89 return null;
90 }
91 if (str_contains($text, '/')) {
92 [$p, $q] = explode('/', $text);
93 if ((int)$q === 0) {
94 return null;
95 }
96 $number = new Fraction((int)$p, (int)$q);
97 } elseif (str_contains($text, '.')) {
98 [$whole, $digits] = explode('.', $text);
99 $q = 10 ** strlen($digits);
100 $negative = str_starts_with($text, '-');
101 $p = abs((int)$whole) * $q + (int)$digits;
102 $number = new Fraction($negative ? -$p : $p, $q);
103 } else {
104 $number = new Fraction((int)$text);
105 }
106 return $number->mul($multiplier);
107}

109function solve(array $matrix, array $rhs): ?array
110{
111 $n = count($rhs);
112 $rows = [];
113 foreach ($matrix as $i => $row) {
114 $rows[] = array_merge($row, [$rhs[$i]]);
115 }
116 for ($k = 0; $k < $n; $k++) {
117 $pivot = null;
118 for ($i = $k; $i < $n; $i++) {
119 if ($rows[$i][$k]->p !== 0) {
120 $pivot = $i;
121 break;
122 }
123 }
124 if ($pivot === null) {
125 return null;
126 }
127 if ($pivot !== $k) {
128 [$rows[$k], $rows[$pivot]] = [$rows[$pivot], $rows[$k]];
129 }
130 $divisor = $rows[$k][$k];
131 for ($j = 0; $j <= $n; $j++) {
132 $rows[$k][$j] = $rows[$k][$j]->div($divisor);
133 }
134 for ($i = 0; $i < $n; $i++) {
135 if ($i === $k || $rows[$i][$k]->p === 0) {
136 continue;
137 }
138 $factor = $rows[$i][$k];
139 for ($j = 0; $j <= $n; $j++) {
140 $rows[$i][$j] = $rows[$i][$j]->sub($factor->mul($rows[$k][$j]));
141 }
142 }
143 }
144 return array_map(fn(array $row) => $row[$n], $rows);
145}

147function reachable(string $start, array $edges): array
148{
149 $seen = [$start => true];
150 $queue = [$start];
151 while ($queue !== []) {
152 $node = array_shift($queue);
153 foreach ($edges as [$a, $b]) {
154 if ($a !== $node && $b !== $node) {
155 continue;
156 }
157 $other = $a === $node ? $b : $a;
158 if (!isset($seen[$other])) {
159 $seen[$other] = true;
160 $queue[] = $other;
161 }
162 }
163 }
164 return $seen;
165}

167class Groups
168{
169 private array $parent = [];
170
171 public function __construct(array $nodes)
172 {
173 foreach ($nodes as $node) {
174 $this->parent[$node] = $node;
175 }
176 }
177
178 public function find(string $node): string
179 {
180 if ($this->parent[$node] !== $node) {
181 $this->parent[$node] = $this->find($this->parent[$node]);
182 }
183 return $this->parent[$node];
184 }
185
186 public function join(string $a, string $b): void
187 {
188 $a = $this->find($a);
189 $b = $this->find($b);
190 if (strcmp($a, $b) < 0) {
191 $this->parent[$b] = $a;
192 } else {
193 $this->parent[$a] = $b;
194 }
195 }
196}

249 private function add(array $words): array
250 {
251 $name = $words[0];
252 $type = $name[0];
253 if (!in_array($type, ['R', 'V', 'I', 'E', 'G', 'F', 'H'], true)) {
254 throw new CircuitError('unknown element type ' . $name);
255 }
256 $count = in_array($type, ['E', 'G'], true) ? 6 : (in_array($type, ['F', 'H'], true) ? 5 : 4);
257 if (count($words) !== $count) {
258 throw new CircuitError('wrong field count for ' . $name);
259 }
260 foreach ($this->elements as $item) {
261 if ($item->name === $name) {
262 throw new CircuitError('duplicate element ' . $name);
263 }
264 }
265 $terminals = array_slice($words, 1, in_array($type, ['E', 'G'], true) ? 4 : 2);
266 foreach ($terminals as $terminal) {
267 if (!preg_match('/\A(?:0|[a-z][a-z0-9]{0,7})\z/', $terminal)) {
268 throw new CircuitError('invalid node ' . $terminal);
269 }
270 }
271 $reference = in_array($type, ['F', 'H'], true) ? $words[3] : null;
272 if ($reference !== null) {
273 $found = false;
274 foreach ($this->elements as $item) {
275 if ($item->name === $reference && $item->type === 'V') {
276 $found = true;
277 break;
278 }
279 }
280 if (!$found) {
281 throw new CircuitError('unknown voltage source ' . $reference);
282 }
283 }
284 $value = $this->number($words[array_key_last($words)]);
285 if ($type === 'R' && $value->p <= 0) {
286 throw new CircuitError('nonpositive resistance ' . $name);
287 }
288 $this->elements[] = new Element($name, $type, $terminals, $value, $reference);
289 return [];
290 }

292 private function grounding(): array
293 {
294 $edges = [];
295 foreach ($this->elements as $item) {
296 if (in_array($item->type, ['R', 'V', 'E', 'H'], true)) {
297 $edges[] = array_slice($item->nodes, 0, 2);

299 }
300 return reachable('0', $edges);

303 private function voltageLoops(): array
304 {
305 $groups = new Groups(array_merge(['0'], $this->nodes()));
306 $result = [];
307 foreach ($this->sources() as $item) {
308 [$a, $b] = $item->nodes;
309 if ($groups->find($a) === $groups->find($b)) {
310 $result[] = $item->name;
311 }
312 $groups->join($a, $b);
313 }
314 return $result;

317 private function structureCheck(): void
318 {
319 $nodes = $this->nodes();
320 if ($nodes === []) {
321 throw new CircuitError('no nodes');
322 }
323 $grounded = $this->grounding();
324 foreach ($nodes as $name) {
325 if (!isset($grounded[$name])) {
326 throw new CircuitError('floating node ' . $name);

331 throw new CircuitError('voltage source loop at ' . $loops[0]);
332 }

335 private function equations(array $overrides = []): array
336 {
337 $labels = array_merge($this->nodes(), array_map(fn(Element $item) => $item->name, $this->sources()));
338 $index = array_flip($labels);
339 $n = count($labels);
340 $matrix = array_fill(0, $n, array_fill(0, $n, new Fraction()));
341 $rhs = array_fill(0, $n, new Fraction());
342 $put = function (string $row, string $column, Fraction $value) use (&$matrix, $index): void {
343 if ($row !== '0' && $column !== '0') {
344 $i = $index[$row];
345 $j = $index[$column];
346 $matrix[$i][$j] = $matrix[$i][$j]->add($value);
347 }
348 };
349 $putRhs = function (string $row, Fraction $value) use (&$rhs, $index): void {
350 if ($row !== '0') {
351 $i = $index[$row];
352 $rhs[$i] = $rhs[$i]->add($value);
353 }
354 };
355 foreach ($this->elements as $item) {
356 [$a, $b] = $item->nodes;
357 $c = $item->nodes[2] ?? null;
358 $d = $item->nodes[3] ?? null;
359 $value = $overrides[$item->name] ?? $item->value;
360 $s = $item->name;
361 if ($item->type === 'R') {
362 $g = (new Fraction(1))->div($value);
363 $put($a, $a, $g);
364 $put($b, $b, $g);
365 $put($a, $b, $g->neg());
366 $put($b, $a, $g->neg());
367 } elseif ($item->type === 'I') {
368 $putRhs($a, $value->neg());
369 $putRhs($b, $value);
370 } elseif ($item->type === 'G') {
371 $put($a, $c, $value);
372 $put($b, $d, $value);
373 $put($a, $d, $value->neg());
374 $put($b, $c, $value->neg());
375 } elseif ($item->type === 'F') {
376 $put($a, $item->reference, $value);
377 $put($b, $item->reference, $value->neg());
378 } else {
379 $put($a, $s, new Fraction(1));
380 $put($s, $a, new Fraction(1));
381 $put($b, $s, new Fraction(-1));
382 $put($s, $b, new Fraction(-1));
383 if ($item->type === 'V') {
384 $putRhs($s, $value);
385 } elseif ($item->type === 'E') {
386 $put($s, $c, $value->neg());
387 $put($s, $d, $value);
388 } else {
389 $put($s, $item->reference, $value->neg());
390 }
391 }
392 }
393 return [$matrix, $rhs, $labels, $index];
394 }

396 private function potential(array $solution, array $index, string $name): Fraction

401 private function current(Element $item, array $solution, array $index): Fraction
402 {
403 [$a, $b] = $item->nodes;
404 return match ($item->type) {
405 'R' => $this->potential($solution, $index, $a)->sub($this->potential($solution, $index, $b))->div($item->value),
406 'V', 'E', 'H' => $solution[$index[$item->name]],
407 'I' => $item->value,
408 'G' => $item->value->mul($this->potential($solution, $index, $item->nodes[2])->sub($this->potential($solution, $index, $item->nodes[3]))),
409 'F' => $item->value->mul($solution[$index[$item->reference]])
410 };
411 }

413 private function injection(array $matrix, array $index, string $a, string $b): array
414 {
415 $rhs = array_fill(0, count($matrix), new Fraction());
416 if ($a !== '0') {
417 $rhs[$index[$a]] = $rhs[$index[$a]]->add(new Fraction(1));
418 }
419 if ($b !== '0') {
420 $rhs[$index[$b]] = $rhs[$index[$b]]->sub(new Fraction(1));

425 private function sourceSolution(array $matrix, array $index, Element $item, bool $unit = false): array
426 {
427 $rhs = array_fill(0, count($matrix), new Fraction());
428 $value = $unit ? new Fraction(1) : $item->value;
429 if ($item->type === 'V') {
430 $rhs[$index[$item->name]] = $value;
431 } else {
432 [$a, $b] = $item->nodes;
433 if ($a !== '0') {
434 $rhs[$index[$a]] = $rhs[$index[$a]]->sub($value);
435 }
436 if ($b !== '0') {
437 $rhs[$index[$b]] = $rhs[$index[$b]]->add($value);
438 }
439 }
440 return solve($matrix, $rhs);

443 private function analysis(string $command, array $args): array
444 {
445 if ($command === '.CURRENT') {
446 $this->element($args[0]);
447 } elseif (in_array($command, ['.VOLTAGE', '.THEVENIN', '.NORTON', '.MAXPOWER', '.ZPARAM'], true)) {
448 foreach ($args as $name) {
449 $this->node($name);
450 }
451 } elseif (in_array($command, ['.KCL', '.SENS', '.SUPERPOSITION'], true)) {
452 $this->node($args[0]);
453 } elseif ($command === '.TF') {
454 $this->node($args[0]);
455 $source = $this->element($args[1]);
456 if (!in_array($source->type, ['V', 'I'], true)) {
457 throw new CircuitError('not an independent source ' . $source->name);
458 }
459 }
460 $this->structureCheck();
461 [$matrix, $rhs, , $index] = $this->equations();
462 $solution = solve($matrix, $rhs) ?? throw new CircuitError('singular matrix');
463 $voltage = fn(string $name, ?array $values = null) => $this->potential($values ?? $solution, $index, $name);
464 $lines = [];
465 switch ($command) {
466 case '.OP':
467 foreach ($this->nodes() as $name) {
468 $lines[] = 'V(' . $name . ') = ' . $voltage($name)->formatted();
469 }
470 foreach ($this->sources() as $item) {
471 $lines[] = 'I(' . $item->name . ') = ' . $solution[$index[$item->name]]->formatted();
472 }
473 break;
474 case '.CURRENT':
475 $lines[] = 'I(' . $args[0] . ') = ' . $this->current($this->element($args[0]), $solution, $index)->formatted();
476 break;
477 case '.VOLTAGE':
478 $lines[] = 'V(' . $args[0] . ',' . $args[1] . ') = ' . $voltage($args[0])->sub($voltage($args[1]))->formatted();
479 break;
480 case '.POWER':
481 $absorbed = new Fraction();
482 $delivered = new Fraction();
483 $total = new Fraction();
484 foreach ($this->elements as $item) {
485 $power = $voltage($item->nodes[0])->sub($voltage($item->nodes[1]))->mul($this->current($item, $solution, $index));
486 if ($power->p > 0) {
487 $absorbed = $absorbed->add($power);
488 }
489 if ($power->p < 0) {
490 $delivered = $delivered->sub($power);
491 }
492 $total = $total->add($power);
493 $lines[] = 'P(' . $item->name . ') = ' . $power->formatted();
494 }
495 $lines[] = 'absorbed = ' . $absorbed->formatted();
496 $lines[] = 'delivered = ' . $delivered->formatted();
497 $lines[] = 'total = ' . $total->formatted();
498 break;
499 case '.KCL':
500 $a = $args[0];
501 $total = new Fraction();
502 foreach ($this->elements as $item) {
503 [$x, $y] = $item->nodes;
504 if ($x !== $a && $y !== $a) {
505 continue;
506 }
507 $value = $x === $y ? new Fraction() : $this->current($item, $solution, $index)->mul(new Fraction($x === $a ? 1 : -1));
508 $total = $total->add($value);
509 $lines[] = 'I(' . $item->name . ') leaving ' . $a . ' = ' . $value->formatted();
510 }
511 $lines[] = 'sum = ' . $total->formatted();
512 break;
513 case '.THEVENIN':
514 case '.NORTON':
515 case '.MAXPOWER':
516 [$a, $b] = $args;
517 $vth = $voltage($a)->sub($voltage($b));
518 $z = $this->injection($matrix, $index, $a, $b);
519 $rth = $voltage($a, $z)->sub($voltage($b, $z));
520 if ($command === '.NORTON') {
521 if ($rth->p === 0) {
522 throw new CircuitError('zero thevenin resistance');
523 }
524 $lines = ['In = ' . $vth->div($rth)->formatted(), 'Rn = ' . $rth->formatted()];
525 } elseif ($command === '.MAXPOWER') {
526 if ($rth->p <= 0) {
527 throw new CircuitError('nonpositive thevenin resistance');
528 }
529 $lines = ['RL = ' . $rth->formatted(), 'Pmax = ' . $vth->mul($vth)->div((new Fraction(4))->mul($rth))->formatted()];
530 } else {
531 $lines = ['Vth = ' . $vth->formatted(), 'Rth = ' . $rth->formatted()];
532 }
533 break;
534 case '.TF':
535 [$a, $name] = $args;
536 $item = $this->element($name);
537 $z = $this->sourceSolution($matrix, $index, $item, true);
538 $lines[] = 'gain = ' . $voltage($a, $z)->formatted();
539 if ($item->type === 'V') {
540 $j = $z[$index[$name]];
541 $lines[] = $j->p === 0 ? 'input resistance = INFINITE' : 'input resistance = ' . (new Fraction(-1))->div($j)->formatted();
542 } else {
543 $lines[] = 'input resistance = ' . $voltage($item->nodes[1], $z)->sub($voltage($item->nodes[0], $z))->formatted();
544 }
545 $lines[] = 'output resistance = ' . $voltage($a, $this->injection($matrix, $index, $a, '0'))->formatted();
546 break;
547 case '.SENS':
548 $a = $args[0];
549 foreach ($this->elements as $item) {
550 $d = array_fill(0, count($matrix), new Fraction());
551 $add = function (string $row, Fraction $value) use (&$d, $index): void {
552 if ($row !== '0') {
553 $d[$index[$row]] = $d[$index[$row]]->add($value);
554 }
555 };
556 [$x, $y] = $item->nodes;
557 if ($item->type === 'R') {
558 $t = $this->current($item, $solution, $index)->div($item->value);
559 $add($x, $t);
560 $add($y, $t->neg());
561 } elseif ($item->type === 'I') {
562 $add($x, new Fraction(-1));
563 $add($y, new Fraction(1));
564 } elseif ($item->type === 'G' || $item->type === 'F') {
565 $t = $item->type === 'G' ? $voltage($item->nodes[2])->sub($voltage($item->nodes[3])) : $solution[$index[$item->reference]];
566 $add($x, $t->neg());
567 $add($y, $t);
568 } elseif ($item->type === 'V') {
569 $add($item->name, new Fraction(1));
570 } elseif ($item->type === 'E') {
571 $add($item->name, $voltage($item->nodes[2])->sub($voltage($item->nodes[3])));
572 } else {
573 $add($item->name, $solution[$index[$item->reference]]);
574 }
575 $lines[] = 'dV(' . $a . ')/d' . $item->name . ' = ' . $voltage($a, solve($matrix, $d))->formatted();
576 }
577 break;
578 case '.ZPARAM':
579 [$a, $b, $c, $d] = $args;
580 $z1 = $this->injection($matrix, $index, $a, $b);
581 $z2 = $this->injection($matrix, $index, $c, $d);
582 $lines = ['z11 = ' . $voltage($a, $z1)->sub($voltage($b, $z1))->formatted(),
583 'z12 = ' . $voltage($a, $z2)->sub($voltage($b, $z2))->formatted(),
584 'z21 = ' . $voltage($c, $z1)->sub($voltage($d, $z1))->formatted(),
585 'z22 = ' . $voltage($c, $z2)->sub($voltage($d, $z2))->formatted()];
586 break;
587 case '.SUPERPOSITION':
588 $a = $args[0];
589 $total = new Fraction();
590 foreach ($this->elements as $item) {
591 if (!in_array($item->type, ['V', 'I'], true)) {
592 continue;
593 }
594 $value = $voltage($a, $this->sourceSolution($matrix, $index, $item));
595 $total = $total->add($value);
596 $lines[] = $item->name . ': V(' . $a . ') = ' . $value->formatted();
597 }
598 $lines[] = 'total: V(' . $a . ') = ' . $total->formatted();
599 break;
600 }
601 return $lines;
602 }

604 private function equivalent(string $a, string $b): array
605 {
606 $groups = new Groups(array_merge(['0'], $this->nodes()));
607 foreach ($this->elements as $item) {
608 if ($item->type === 'V') {
609 $groups->join($item->nodes[0], $item->nodes[1]);
610 }
611 }
612 $a = $groups->find($a);
613 $b = $groups->find($b);
614 if ($a === $b) {
615 return ['Req = 0 (0.000000)'];
616 }
617 $edges = [];
618 foreach ($this->elements as $item) {
619 if ($item->type !== 'R') {
620 continue;
621 }
622 $x = $groups->find($item->nodes[0]);
623 $y = $groups->find($item->nodes[1]);
624 if ($x !== $y) {
625 $edges[] = [$x, $y, (new Fraction(1))->div($item->value)];
626 }
627 }
628 $connected = reachable($a, $edges);
629 if (!isset($connected[$b])) {
630 return ['Req = INFINITE'];
631 }
632 $labels = array_values(array_filter(array_keys($connected), fn($name) => (string)$name !== $b));
633 sort($labels, SORT_STRING);
634 $index = array_flip($labels);
635 $matrix = array_fill(0, count($labels), array_fill(0, count($labels), new Fraction()));
636 $rhs = array_fill(0, count($labels), new Fraction());
637 $rhs[$index[$a]] = new Fraction(1);
638 foreach ($edges as [$x, $y, $g]) {
639 if (!isset($connected[$x])) {
640 continue;
641 }
642 if ($x !== $b) {
643 $matrix[$index[$x]][$index[$x]] = $matrix[$index[$x]][$index[$x]]->add($g);
644 }
645 if ($y !== $b) {
646 $matrix[$index[$y]][$index[$y]] = $matrix[$index[$y]][$index[$y]]->add($g);
647 }
648 if ($x !== $b && $y !== $b) {
649 $matrix[$index[$x]][$index[$y]] = $matrix[$index[$x]][$index[$y]]->sub($g);
650 $matrix[$index[$y]][$index[$x]] = $matrix[$index[$y]][$index[$x]]->sub($g);
651 }
652 }
653 return ['Req = ' . solve($matrix, $rhs)[$index[$a]]->formatted()];
654 }

656 private function reduceResistors(string $a, string $b): array
657 {
658 if ($a === $b) {
659 return ['Req = 0 (0.000000)'];
660 }
661 $list = array_values(array_filter($this->elements, fn(Element $item) => $item->type === 'R'));
662 $connected = reachable($a, array_map(fn(Element $item) => $item->nodes, $list));
663 if (!isset($connected[$b])) {
664 return ['Req = INFINITE'];
665 }
666 $list = array_values(array_filter($list, fn(Element $item) => isset($connected[$item->nodes[0]])));
667 $lines = [];
668 $serial = 0;
669 while (true) {
670 $self = null;
671 foreach ($list as $i => $item) {
672 if ($item->nodes[0] === $item->nodes[1]) {
673 $self = $i;
674 break;
675 }
676 }
677 if ($self !== null) {
678 $lines[] = 'remove ' . $list[$self]->name;
679 array_splice($list, $self, 1);
680 continue;
681 }
682 $degree = [];
683 foreach ($list as $item) {
684 foreach ($item->nodes as $name) {
685 $degree[$name] = ($degree[$name] ?? 0) + 1;
686 }
687 }
688 $internal = array_values(array_filter(array_keys($degree), fn($name) => (string)$name !== $a && (string)$name !== $b));
689 sort($internal, SORT_STRING);
690 $leaf = null;
691 foreach ($internal as $name) {
692 if ($degree[$name] === 1) {
693 $leaf = (string)$name;
694 break;
695 }
696 }
697 if ($leaf !== null) {
698 foreach ($list as $i => $item) {
699 if (in_array($leaf, $item->nodes, true)) {
700 $lines[] = 'remove ' . $item->name;
701 array_splice($list, $i, 1);
702 break;
703 }
704 }
705 continue;
706 }
707 $pair = null;
708 for ($i = 0; $i < count($list); $i++) {
709 for ($j = $i + 1; $j < count($list); $j++) {
710 $x = $list[$i]->nodes;
711 $y = $list[$j]->nodes;
712 if (($x[0] === $y[0] && $x[1] === $y[1]) || ($x[0] === $y[1] && $x[1] === $y[0])) {
713 $pair = [$i, $j];
714 break 2;
715 }
716 }
717 }
718 if ($pair !== null) {
719 $x = $list[$pair[0]];
720 $y = $list[$pair[1]];
721 $value = $x->value->mul($y->value)->div($x->value->add($y->value));
722 $terminals = $x->nodes;
723 $operator = '||';
724 } else {
725 $middle = null;
726 foreach ($internal as $name) {
727 if ($degree[$name] === 2) {
728 $middle = (string)$name;
729 break;
730 }
731 }
732 if ($middle === null) {
733 break;
734 }
735 $pair = [];
736 foreach ($list as $i => $item) {
737 if (in_array($middle, $item->nodes, true)) {
738 $pair[] = $i;
739 }
740 }
741 $x = $list[$pair[0]];
742 $y = $list[$pair[1]];
743 $terminals = [$x->nodes[0] === $middle ? $x->nodes[1] : $x->nodes[0], $y->nodes[0] === $middle ? $y->nodes[1] : $y->nodes[0]];
744 $value = $x->value->add($y->value);
745 $operator = '+';
746 }
747 $name = '#' . ++$serial;
748 $lines[] = $name . ' = ' . $x->name . ' ' . $operator . ' ' . $y->name . ' = ' . $value->formatted();
749 $list = array_values(array_filter($list, fn($i) => !in_array($i, $pair, true), ARRAY_FILTER_USE_KEY));
750 $list[] = new Element($name, 'R', $terminals, $value);
751 }
752 $lines[] = count($list) === 1 ? 'Req = ' . $list[0]->value->formatted() : 'not series-parallel: ' . count($list) . ' resistors remain';
753 return $lines;
754 }

756 private function command(array $words): array
757 {
758 $cmd = array_shift($words);
759 $args = $words;
760 if (!array_key_exists($cmd, self::COUNTS)) {
761 throw new CircuitError('unknown command ' . $cmd);
762 }
763 if (count($args) !== self::COUNTS[$cmd]) {
764 throw new CircuitError('wrong argument count for ' . $cmd);
765 }
766 if (in_array($cmd, self::ANALYSES, true)) {
767 return $this->analysis($cmd, $args);
768 }
769 switch ($cmd) {
770 case '.REQ':
771 case '.REDUCE':
772 foreach ($args as $name) {
773 $this->node($name);
774 }
775 return $cmd === '.REQ' ? $this->equivalent($args[0], $args[1]) : $this->reduceResistors($args[0], $args[1]);
776 case '.SWEEP':
777 $item = $this->element($args[0]);
778 $from = $this->number($args[1]);
779 $to = $this->number($args[2]);
780 $step = $this->number($args[3]);
781 $this->node($args[4]);
782 if ($step->p <= 0) {
783 throw new CircuitError('invalid sweep step');
784 }
785 if ($from->compare($to) > 0) {
786 throw new CircuitError('invalid sweep range');
787 }
788 $range = $to->sub($from)->div($step);
789 $count = intdiv($range->p, $range->q) + 1;
790 if ($count > 20) {
791 throw new CircuitError('too many sweep points');
792 }
793 if ($item->type === 'R' && $from->p <= 0) {
794 throw new CircuitError('nonpositive resistance ' . $item->name);
795 }
796 $this->structureCheck();
797 $lines = [];
798 for ($k = 0; $k < $count; $k++) {
799 $x = $from->add((new Fraction($k))->mul($step));
800 [$matrix, $rhs, , $index] = $this->equations([$item->name => $x]);
801 $solution = solve($matrix, $rhs);
802 $lines[] = $item->name . ' = ' . $x->fraction() . ': ' . ($solution === null ? 'singular matrix' : 'V(' . $args[4] . ') = ' . $this->potential($solution, $index, $args[4])->formatted());
803 }
804 return $lines;
805 case '.MATRIX':
806 $this->structureCheck();
807 [$matrix, $rhs, $labels] = $this->equations();
808 $unknowns = array_merge(array_map(fn($name) => 'V(' . $name . ')', $this->nodes()), array_map(fn(Element $item) => 'I(' . $item->name . ')', $this->sources()));
809 $lines = ['unknowns: ' . implode(' ', $unknowns)];
810 foreach ($matrix as $i => $row) {
811 $lines[] = $labels[$i] . ': ' . implode(' ', array_map(fn(Fraction $value) => $value->fraction(), $row)) . ' | ' . $rhs[$i]->fraction();
812 }
813 return $lines;
814 case '.CHECK':
815 if ($this->nodes() === []) {
816 return ['no nodes'];
817 }
818 $lines = [];
819 foreach ($this->nodes() as $name) {
820 $degree = 0;
821 foreach ($this->elements as $item) {
822 $degree += (int)($item->nodes[0] === $name) + (int)($item->nodes[1] === $name);
823 }
824 if ($degree < 2) {
825 $lines[] = 'dangling node ' . $name;
826 }
827 }
828 $grounded = $this->grounding();
829 foreach ($this->nodes() as $name) {
830 if (!isset($grounded[$name])) {
831 $lines[] = 'floating node ' . $name;
832 }
833 }
834 foreach ($this->voltageLoops() as $name) {
835 $lines[] = 'voltage source loop at ' . $name;
836 }
837 if ($lines === []) {
838 [$matrix, $rhs] = $this->equations();
839 if (solve($matrix, $rhs) === null) {
840 $lines[] = 'singular matrix';
841 }
842 }
843 return $lines === [] ? ['ok'] : $lines;
844 case '.ALTER':
845 $item = $this->element($args[0]);
846 $value = $this->number($args[1]);
847 if ($item->type === 'R' && $value->p <= 0) {
848 throw new CircuitError('nonpositive resistance ' . $item->name);
849 }
850 $line = 'altered ' . $item->name . ': ' . $item->value->fraction() . ' -> ' . $value->fraction();
851 $item->value = $value;
852 return [$line];
853 case '.DELETE':
854 $item = $this->element($args[0]);
855 foreach ($this->elements as $other) {
856 if (in_array($other->type, ['F', 'H'], true) && $other->reference === $item->name) {
857 throw new CircuitError($item->name . ' is referenced by ' . $other->name);
858 }
859 }
860 $this->elements = array_values(array_filter($this->elements, fn(Element $other) => $other !== $item));
861 return ['deleted ' . $item->name];
862 case '.CLEAR':
863 $line = 'cleared ' . count($this->elements) . ' elements';
864 $this->elements = [];
865 return [$line];
866 case '.NODES':
867 if ($this->elements === []) {
868 return ['no nodes'];
869 }
870 $all = [];
871 foreach ($this->elements as $item) {
872 foreach ($item->nodes as $name) {
873 $all[$name] = true;
874 }
875 }
876 $names = array_keys($all);
877 sort($names, SORT_STRING);
878 $lines = [];
879 foreach ($names as $name) {
880 $name = (string)$name;
881 $users = array_filter($this->elements, fn(Element $item) => in_array($name, $item->nodes, true));
882 $lines[] = $name . ': ' . implode(' ', array_map(fn(Element $item) => $item->name, $users));
883 }
884 return $lines;
885 case '.PRINT':
886 if ($this->elements === []) {
887 return ['empty netlist'];
888 }
889 return array_map(function (Element $item): string {
890 $words = array_merge([$item->name], $item->nodes);
891 if ($item->reference !== null) {
892 $words[] = $item->reference;
893 }
894 $words[] = $item->value->fraction();
895 return implode(' ', $words);
896 }, $this->elements);
897 }
898 return [];
899 }

901 public function run(): void
902 {
903 $number = 0;
904 while (($input = fgets(STDIN)) !== false) {
905 $number++;
906 $line = str_ends_with($input, "\n") ? substr($input, 0, -1) : $input;
907 if ($line === '.END') {
908 break;
909 }
910 if (str_starts_with($line, '*')) {
911 continue;
912 }
913 try {
914 $words = explode(' ', $line);
915 $output = str_starts_with($line, '.') ? $this->command($words) : $this->add($words);
916 foreach ($output as $text) {
917 echo $text, "\n";
918 }
919 } catch (CircuitError $error) {
920 echo 'error: line ', $number, ': ', $error->getMessage(), "\n";
921 }
922 }
923 }
924}
925
926(new Circuit())->run();
