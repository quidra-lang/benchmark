1struct CircuitError <: Exception
2 message::String
3end
4circuit_error(message) = throw(CircuitError(message))
5
6function fraction_text(value)
7 p, q = numerator(value), denominator(value)
8 q == 1 ? string(p) : string(p, "/", q)
9end
10function value_text(value)
11 p, q = numerator(value), denominator(value)
12 scaled = div(abs(p) * 1000000, q)
13 decimal = (p < 0 ? "-" : "") * string(div(scaled, 1000000)) * "." *
14 lpad(string(rem(scaled, 1000000)), 6, '0')
15 fraction_text(value) * " (" * decimal * ")"
16end
17function parse_value(text)
18 multiplier = 1 // 1
19 for (suffix, scale) in (("meg", 1000000 // 1), ("k", 1000 // 1),
20 ("m", 1 // 1000), ("u", 1 // 1000000))
21 if endswith(text, suffix)
22 text = text[1:end - length(suffix)]
23 multiplier = scale
24 break
25 end
26 end
27 occursin(r"^-?[0-9]+(?:\.[0-9]+|/[0-9]+)?$", text) || return nothing
28 if occursin('/', text)
29 parts = split(text, '/')
30 p, q = parse(Int, parts[1]), parse(Int, parts[2])
31 q == 0 && return nothing
32 value = p // q
33 elseif occursin('.', text)
34 parts = split(text, '.')
35 q = 10 ^ length(parts[2])
36 p = abs(parse(Int, parts[1])) * q + parse(Int, parts[2])
37 value = (startswith(text, "-") ? -p : p) // q
38 else
39 value = parse(Int, text) // 1
40 end
41 value * multiplier
42end
43function required_value(text)
44 value = parse_value(text)
45 value === nothing && circuit_error("invalid value $text")
46 value
47end
48function solve_system(matrix, rhs)
49 n = length(rhs)
50 rows = hcat(matrix, rhs)
51 for k in 1:n
52 pivot = 0
53 for i in k:n
54 if rows[i, k] != 0
55 pivot = i
56 break
57 end
58 end
59 pivot == 0 && return nothing
60 if pivot != k
61 saved = rows[k, :]
62 rows[k, :] = rows[pivot, :]
63 rows[pivot, :] = saved
64 end
65 divisor = rows[k, k]
66 for j in 1:n + 1
67 rows[k, j] /= divisor
68 end
69 for i in 1:n
70 (i == k || rows[i, k] == 0) && continue
71 factor = rows[i, k]
72 for j in 1:n + 1
73 rows[i, j] -= factor * rows[k, j]
74 end
75 end
76 end
77 rows[:, n + 1]
78end
79function reached_nodes(start, edges)
80 reached = Set([String(start)])
81 pending = [String(start)]
82 head = 1
83 while head <= length(pending)
84 node = pending[head]
85 head += 1
86 for edge in edges
87 a, b = edge[1], edge[2]
88 (a == node || b == node) || continue
89 other = a == node ? b : a
90 if !(other in reached)
91 push!(reached, other)
92 push!(pending, other)
93 end
94 end
95 end
96 reached
97end
98mutable struct CircuitGroups
99 parent::Dict{String,String}
100end
101CircuitGroups(nodes) = CircuitGroups(Dict(node => node for node in nodes))
102function group_of(groups, node)
103 parent = groups.parent[node]
104 parent == node && return parent
105 root = group_of(groups, parent)
106 groups.parent[node] = root
107 root
108end
109function join_groups!(groups, a, b)
110 a, b = group_of(groups, a), group_of(groups, b)
111 if isless(a, b)
112 groups.parent[b] = a
113 else
114 groups.parent[a] = b
115 end
116end

128Circuit() = Circuit(CircuitElement[])
129const VOLTAGE_KINDS = ('V', 'E', 'H')
130const COMMAND_COUNTS = Dict(".OP" => 0, ".CURRENT" => 1, ".VOLTAGE" => 2, ".POWER" => 0,
131 ".KCL" => 1, ".THEVENIN" => 2, ".NORTON" => 2, ".MAXPOWER" => 2, ".TF" => 2,
132 ".SENS" => 1, ".ZPARAM" => 4, ".SUPERPOSITION" => 1, ".SWEEP" => 5, ".MATRIX" => 0,
133 ".REQ" => 2, ".REDUCE" => 2, ".CHECK" => 0, ".ALTER" => 2, ".DELETE" => 1,
134 ".CLEAR" => 0, ".NODES" => 0, ".PRINT" => 0)
135const ANALYSES = Set([".OP", ".CURRENT", ".VOLTAGE", ".POWER", ".KCL", ".THEVENIN",
136 ".NORTON", ".MAXPOWER", ".TF", ".SENS", ".ZPARAM", ".SUPERPOSITION"])

138function circuit_nodes(c)
139 sort(unique([node for item in c.elements for node in item.nodes if node != "0"]))
140end
141voltage_sources(c) = [item for item in c.elements if item.kind in VOLTAGE_KINDS]
142function required_element(c, name)
143 for item in c.elements
144 item.name == name && return item
145 end
146 circuit_error("unknown element $name")
147end
148function require_node(c, name)
149 (name == "0" || name in circuit_nodes(c)) || circuit_error("unknown node $name")
150end
151function add_element!(c, fields)
152 name = fields[1]
153 kind = name[1]
154 kind in ('R', 'V', 'I', 'E', 'G', 'F', 'H') || circuit_error("unknown element type $name")
155 count = kind in ('E', 'G') ? 6 : kind in ('F', 'H') ? 5 : 4
156 length(fields) == count || circuit_error("wrong field count for $name")
157 any(item -> item.name == name, c.elements) && circuit_error("duplicate element $name")
158 terminals = String.(fields[2:(kind in ('E', 'G') ? 5 : 3)])
159 for node in terminals
160 occursin(r"^(?:0|[a-z][a-z0-9]{0,7})$", node) || circuit_error("invalid node $node")
161 end
162 reference = kind in ('F', 'H') ? String(fields[4]) : ""
163 if !isempty(reference) && !any(item -> item.name == reference && item.kind == 'V', c.elements)
164 circuit_error("unknown voltage source $reference")
165 end
166 value = required_value(fields[end])
167 kind == 'R' && value <= 0 && circuit_error("nonpositive resistance $name")
168 push!(c.elements, CircuitElement(name, kind, terminals, value, reference))
169 String[]
170end
171grounded_nodes(c) = reached_nodes("0", [item.nodes[1:2] for item in c.elements if item.kind in ('R', 'V', 'E', 'H')])
172function voltage_loops(c)
173 groups = CircuitGroups(["0"; circuit_nodes(c)])
174 result = String[]
175 for item in voltage_sources(c)
176 a, b = item.nodes[1:2]
177 group_of(groups, a) == group_of(groups, b) && push!(result, item.name)
178 join_groups!(groups, a, b)
179 end
180 result
181end
182function check_structure(c)
183 nodes = circuit_nodes(c)
184 isempty(nodes) && circuit_error("no nodes")
185 grounded = grounded_nodes(c)
186 for node in nodes
187 node in grounded || circuit_error("floating node $node")
188 end
189 loops = voltage_loops(c)
190 isempty(loops) || circuit_error("voltage source loop at $(loops[1])")
191end
192function add_rhs!(rhs, index, row, value)
193 row != "0" && (rhs[index[row]] += value)
194end
195function circuit_equations(c, overrides=Dict{String,Rational{Int}}())
196 labels = [circuit_nodes(c); [item.name for item in voltage_sources(c)]]
197 index = Dict(name => i for (i, name) in enumerate(labels))
198 n = length(labels)
199 matrix = fill(0 // 1, n, n)
200 rhs = fill(0 // 1, n)
201 function put(row, column, value)
202 if row != "0" && column != "0"
203 matrix[index[row], index[column]] += value
204 end
205 end
206 for item in c.elements
207 a, b = item.nodes[1:2]
208 value = get(overrides, item.name, item.value)
209 name = item.name
210 if item.kind == 'R'
211 g = inv(value)
212 put(a, a, g)
213 put(b, b, g)
214 put(a, b, -g)
215 put(b, a, -g)
216 elseif item.kind == 'I'
217 add_rhs!(rhs, index, a, -value)
218 add_rhs!(rhs, index, b, value)
219 elseif item.kind == 'G'
220 x, y = item.nodes[3:4]
221 put(a, x, value)
222 put(b, y, value)
223 put(a, y, -value)
224 put(b, x, -value)
225 elseif item.kind == 'F'
226 put(a, item.reference, value)
227 put(b, item.reference, -value)
228 else
229 put(a, name, 1)
230 put(name, a, 1)
231 put(b, name, -1)
232 put(name, b, -1)
233 if item.kind == 'V'
234 add_rhs!(rhs, index, name, value)
235 elseif item.kind == 'E'
236 x, y = item.nodes[3:4]
237 put(name, x, -value)
238 put(name, y, value)
239 else
240 put(name, item.reference, -value)
241 end
242 end
243 end
244 matrix, rhs, labels, index
245end
246potential(solution, index, node) = node == "0" ? 0 // 1 : solution[index[node]]
247function element_current(item, solution, index)
248 a, b = item.nodes[1:2]
249 kind = item.kind
250 kind == 'R' && return (potential(solution, index, a) - potential(solution, index, b)) / item.value
251 kind in VOLTAGE_KINDS && return solution[index[item.name]]
252 kind == 'I' && return item.value
253 kind == 'G' && return item.value *
254 (potential(solution, index, item.nodes[3]) - potential(solution, index, item.nodes[4]))
255 item.value * solution[index[item.reference]]
256end
257function injection_solution(matrix, index, a, b)
258 rhs = fill(0 // 1, size(matrix, 1))
259 add_rhs!(rhs, index, a, 1)
260 add_rhs!(rhs, index, b, -1)
261 solve_system(matrix, rhs)
262end
263function source_solution(matrix, index, item, unit=false)
264 rhs = fill(0 // 1, size(matrix, 1))
265 value = unit ? 1 // 1 : item.value
266 if item.kind == 'V'
267 add_rhs!(rhs, index, item.name, value)
268 else
269 a, b = item.nodes[1:2]
270 add_rhs!(rhs, index, a, -value)
271 add_rhs!(rhs, index, b, value)
272 end
273 solve_system(matrix, rhs)
274end
275function analyze_circuit(c, command, args)
276 if command == ".CURRENT"
277 required_element(c, args[1])
278 elseif command in (".VOLTAGE", ".THEVENIN", ".NORTON", ".MAXPOWER", ".ZPARAM")
279 for node in args
280 require_node(c, node)
281 end
282 elseif command in (".KCL", ".SENS", ".SUPERPOSITION")
283 require_node(c, args[1])
284 elseif command == ".TF"
285 require_node(c, args[1])
286 item = required_element(c, args[2])
287 item.kind in ('V', 'I') || circuit_error("not an independent source $(item.name)")
288 end
289 check_structure(c)
290 matrix, rhs, labels, index = circuit_equations(c)
291 solution = solve_system(matrix, rhs)
292 solution === nothing && circuit_error("singular matrix")
293 voltage(node, values=solution) = potential(values, index, node)
294 lines = String[]
295 if command == ".OP"
296 for node in circuit_nodes(c)
297 push!(lines, "V($node) = " * value_text(voltage(node)))
298 end
299 for item in voltage_sources(c)
300 push!(lines, "I($(item.name)) = " * value_text(solution[index[item.name]]))
301 end
302 elseif command == ".CURRENT"
303 push!(lines, "I($(args[1])) = " * value_text(element_current(required_element(c, args[1]), solution, index)))
304 elseif command == ".VOLTAGE"
305 push!(lines, "V($(args[1]),$(args[2])) = " * value_text(voltage(args[1]) - voltage(args[2])))
306 elseif command == ".POWER"
307 absorbed = delivered = total = 0 // 1
308 for item in c.elements
309 power = (voltage(item.nodes[1]) - voltage(item.nodes[2])) * element_current(item, solution, index)
310 power > 0 && (absorbed += power)
311 power < 0 && (delivered -= power)
312 total += power
313 push!(lines, "P($(item.name)) = " * value_text(power))
314 end
315 append!(lines, ["absorbed = " * value_text(absorbed),
316 "delivered = " * value_text(delivered), "total = " * value_text(total)])
317 elseif command == ".KCL"
318 a = args[1]
319 total = 0 // 1
320 for item in c.elements
321 x, y = item.nodes[1:2]
322 (x == a || y == a) || continue
323 value = x == y ? 0 // 1 : (x == a ? 1 : -1) * element_current(item, solution, index)
324 total += value
325 push!(lines, "I($(item.name)) leaving $a = " * value_text(value))
326 end
327 push!(lines, "sum = " * value_text(total))
328 elseif command in (".THEVENIN", ".NORTON", ".MAXPOWER")
329 a, b = args
330 vth = voltage(a) - voltage(b)
331 z = injection_solution(matrix, index, a, b)
332 rth = voltage(a, z) - voltage(b, z)
333 if command == ".NORTON"
334 rth == 0 && circuit_error("zero thevenin resistance")
335 append!(lines, ["In = " * value_text(vth / rth), "Rn = " * value_text(rth)])
336 elseif command == ".MAXPOWER"
337 rth <= 0 && circuit_error("nonpositive thevenin resistance")
338 append!(lines, ["RL = " * value_text(rth), "Pmax = " * value_text(vth * vth / (4 * rth))])
339 else
340 append!(lines, ["Vth = " * value_text(vth), "Rth = " * value_text(rth)])
341 end
342 elseif command == ".TF"
343 a, name = args
344 item = required_element(c, name)
345 z = source_solution(matrix, index, item, true)
346 push!(lines, "gain = " * value_text(voltage(a, z)))
347 if item.kind == 'V'
348 j = z[index[name]]
349 push!(lines, j == 0 ? "input resistance = INFINITE" : "input resistance = " * value_text(-inv(j)))
350 else
351 push!(lines, "input resistance = " * value_text(voltage(item.nodes[2], z) - voltage(item.nodes[1], z)))
352 end
353 push!(lines, "output resistance = " * value_text(voltage(a, injection_solution(matrix, index, a, "0"))))
354 elseif command == ".SENS"
355 a = args[1]
356 for item in c.elements
357 d = fill(0 // 1, length(rhs))
358 x, y = item.nodes[1:2]
359 if item.kind == 'R'
360 t = element_current(item, solution, index) / item.value
361 add_rhs!(d, index, x, t)
362 add_rhs!(d, index, y, -t)
363 elseif item.kind == 'I'
364 add_rhs!(d, index, x, -1)
365 add_rhs!(d, index, y, 1)
366 elseif item.kind in ('G', 'F')
367 t = item.kind == 'G' ? voltage(item.nodes[3]) - voltage(item.nodes[4]) : solution[index[item.reference]]
368 add_rhs!(d, index, x, -t)
369 add_rhs!(d, index, y, t)
370 elseif item.kind == 'V'
371 add_rhs!(d, index, item.name, 1)
372 elseif item.kind == 'E'
373 add_rhs!(d, index, item.name, voltage(item.nodes[3]) - voltage(item.nodes[4]))
374 else
375 add_rhs!(d, index, item.name, solution[index[item.reference]])
376 end
377 push!(lines, "dV($a)/d$(item.name) = " * value_text(voltage(a, solve_system(matrix, d))))
378 end
379 elseif command == ".ZPARAM"
380 a, b, x, y = args
381 z1 = injection_solution(matrix, index, a, b)
382 z2 = injection_solution(matrix, index, x, y)
383 append!(lines, ["z11 = " * value_text(voltage(a, z1) - voltage(b, z1)),
384 "z12 = " * value_text(voltage(a, z2) - voltage(b, z2)),
385 "z21 = " * value_text(voltage(x, z1) - voltage(y, z1)),
386 "z22 = " * value_text(voltage(x, z2) - voltage(y, z2))])
387 elseif command == ".SUPERPOSITION"
388 a = args[1]
389 total = 0 // 1
390 for item in c.elements
391 item.kind in ('V', 'I') || continue
392 value = voltage(a, source_solution(matrix, index, item))
393 total += value
394 push!(lines, "$(item.name): V($a) = " * value_text(value))
395 end
396 push!(lines, "total: V($a) = " * value_text(total))
397 end
398 lines
399end

401function equivalent_resistance(c, a, b)
402 groups = CircuitGroups(["0"; circuit_nodes(c)])
403 for item in c.elements
404 item.kind == 'V' && join_groups!(groups, item.nodes[1], item.nodes[2])
405 end
406 a, b = group_of(groups, a), group_of(groups, b)
407 a == b && return ["Req = 0 (0.000000)"]
408 edges = Tuple{String,String,Rational{Int}}[]
409 for item in c.elements
410 item.kind == 'R' || continue
411 x, y = group_of(groups, item.nodes[1]), group_of(groups, item.nodes[2])
412 x != y && push!(edges, (x, y, inv(item.value)))
413 end
414 connected = reached_nodes(a, edges)
415 b in connected || return ["Req = INFINITE"]
416 labels = sort([node for node in connected if node != b])
417 index = Dict(node => i for (i, node) in enumerate(labels))
418 matrix = fill(0 // 1, length(labels), length(labels))
419 rhs = fill(0 // 1, length(labels))
420 rhs[index[a]] = 1 // 1
421 for (x, y, g) in edges
422 x in connected || continue
423 x != b && (matrix[index[x], index[x]] += g)
424 y != b && (matrix[index[y], index[y]] += g)
425 if x != b && y != b
426 matrix[index[x], index[y]] -= g
427 matrix[index[y], index[x]] -= g
428 end
429 end
430 ["Req = " * value_text(solve_system(matrix, rhs)[index[a]])]
431end
432function reduce_resistors(c, a, b)
433 a == b && return ["Req = 0 (0.000000)"]
434 resistors = [item for item in c.elements if item.kind == 'R']
435 connected = reached_nodes(a, [item.nodes for item in resistors])
436 b in connected || return ["Req = INFINITE"]
437 filter!(item -> item.nodes[1] in connected, resistors)
438 lines = String[]
439 serial = 0
440 while true
441 self = findfirst(item -> item.nodes[1] == item.nodes[2], resistors)
442 if self !== nothing
443 push!(lines, "remove $(resistors[self].name)")
444 deleteat!(resistors, self)
445 continue
446 end
447 degree = Dict{String,Int}()
448 for item in resistors, node in item.nodes
449 degree[node] = get(degree, node, 0) + 1
450 end
451 internal = sort([node for node in keys(degree) if node != a && node != b])
452 leaf = findfirst(node -> degree[node] == 1, internal)
453 if leaf !== nothing
454 i = findfirst(item -> internal[leaf] in item.nodes, resistors)
455 push!(lines, "remove $(resistors[i].name)")
456 deleteat!(resistors, i)
457 continue
458 end
459 pair = nothing
460 for i in eachindex(resistors)
461 for j in i + 1:length(resistors)
462 if sort(resistors[i].nodes) == sort(resistors[j].nodes)
463 pair = (i, j)
464 break
465 end
466 end
467 pair !== nothing && break
468 end
469 if pair !== nothing
470 x, y = resistors[pair[1]], resistors[pair[2]]
471 value = x.value * y.value / (x.value + y.value)
472 terminals = x.nodes
473 operator = "||"
474 else
475 middle = findfirst(node -> degree[node] == 2, internal)
476 middle === nothing && break
477 node = internal[middle]
478 pair = findall(item -> node in item.nodes, resistors)
479 x, y = resistors[pair[1]], resistors[pair[2]]
480 terminals = [x.nodes[1] == node ? x.nodes[2] : x.nodes[1],
481 y.nodes[1] == node ? y.nodes[2] : y.nodes[1]]
482 value = x.value + y.value
483 operator = "+"
484 end
485 serial += 1
486 name = "#$serial"
487 push!(lines, "$name = $(x.name) $operator $(y.name) = " * value_text(value))
488 deleteat!(resistors, [pair[1], pair[2]])
489 push!(resistors, CircuitElement(name, 'R', terminals, value, ""))
490 end
491 push!(lines, length(resistors) == 1 ? "Req = " * value_text(resistors[1].value) :
492 "not series-parallel: $(length(resistors)) resistors remain")
493 lines
494end
495function circuit_command!(c, fields)
496 command = fields[1]
497 args = fields[2:end]
498 haskey(COMMAND_COUNTS, command) || circuit_error("unknown command $command")
499 length(args) == COMMAND_COUNTS[command] || circuit_error("wrong argument count for $command")
500 command in ANALYSES && return analyze_circuit(c, command, args)
501 if command in (".REQ", ".REDUCE")
502 for node in args
503 require_node(c, node)
504 end
505 return command == ".REQ" ? equivalent_resistance(c, args[1], args[2]) : reduce_resistors(c, args[1], args[2])
506 elseif command == ".SWEEP"
507 item = required_element(c, args[1])
508 from = required_value(args[2])
509 to = required_value(args[3])
510 step = required_value(args[4])
511 require_node(c, args[5])
512 step <= 0 && circuit_error("invalid sweep step")
513 from > to && circuit_error("invalid sweep range")
514 range = (to - from) / step
515 point_count = fld(numerator(range), denominator(range)) + 1
516 point_count > 20 && circuit_error("too many sweep points")
517 item.kind == 'R' && from <= 0 && circuit_error("nonpositive resistance $(item.name)")
518 check_structure(c)
519 lines = String[]
520 for k in 0:point_count - 1
521 x = from + k * step
522 matrix, rhs, labels, index = circuit_equations(c, Dict(item.name => x))
523 solution = solve_system(matrix, rhs)
524 result = solution === nothing ? "singular matrix" : "V($(args[5])) = " * value_text(potential(solution, index, args[5]))
525 push!(lines, "$(item.name) = $(fraction_text(x)): $result")
526 end
527 return lines
528 elseif command == ".MATRIX"
529 check_structure(c)
530 matrix, rhs, labels, index = circuit_equations(c)
531 unknowns = [["V($node)" for node in circuit_nodes(c)]; ["I($(item.name))" for item in voltage_sources(c)]]
532 lines = ["unknowns: " * join(unknowns, " ")]
533 for i in eachindex(labels)
534 push!(lines, labels[i] * ": " * join(fraction_text.(matrix[i, :]), " ") * " | " * fraction_text(rhs[i]))
535 end
536 return lines
537 elseif command == ".CHECK"
538 nodes = circuit_nodes(c)
539 isempty(nodes) && return ["no nodes"]
540 lines = String[]
541 for node in nodes
542 degree = sum((count(==(node), item.nodes[1:2]) for item in c.elements); init=0)
543 degree < 2 && push!(lines, "dangling node $node")
544 end
545 grounded = grounded_nodes(c)
546 for node in nodes
547 node in grounded || push!(lines, "floating node $node")
548 end
549 append!(lines, ["voltage source loop at $name" for name in voltage_loops(c)])
550 if isempty(lines)
551 matrix, rhs, labels, index = circuit_equations(c)
552 solve_system(matrix, rhs) === nothing && push!(lines, "singular matrix")
553 end
554 return isempty(lines) ? ["ok"] : lines
555 elseif command == ".ALTER"
556 item = required_element(c, args[1])
557 value = required_value(args[2])
558 item.kind == 'R' && value <= 0 && circuit_error("nonpositive resistance $(item.name)")
559 line = "altered $(item.name): $(fraction_text(item.value)) -> $(fraction_text(value))"
560 item.value = value
561 return [line]
562 elseif command == ".DELETE"
563 item = required_element(c, args[1])
564 for other in c.elements
565 if other.kind in ('F', 'H') && other.reference == item.name
566 circuit_error("$(item.name) is referenced by $(other.name)")
567 end
568 end
569 filter!(other -> other !== item, c.elements)
570 return ["deleted $(item.name)"]
571 elseif command == ".CLEAR"
572 line = "cleared $(length(c.elements)) elements"
573 empty!(c.elements)
574 return [line]
575 elseif command == ".NODES"
576 isempty(c.elements) && return ["no nodes"]
577 nodes = sort(unique([node for item in c.elements for node in item.nodes]))
578 return [node * ": " * join((item.name for item in c.elements if node in item.nodes), " ") for node in nodes]
579 end
580 isempty(c.elements) && return ["empty netlist"]
581 lines = String[]
582 for item in c.elements
583 words = [item.name; item.nodes]
584 !isempty(item.reference) && push!(words, item.reference)
585 push!(words, fraction_text(item.value))
586 push!(lines, join(words, " "))
587 end
588 lines
589end
590function main()
591 circuit = Circuit()
592 for (number, line) in enumerate(eachline())
593 line == ".END" && break
594 startswith(line, "*") && continue
