10 static void Fail(string reason) => throw new Problem(reason);
11 readonly record struct Rational : IComparable<Rational>
12 {
13 public readonly long Numerator, Denominator;
14 public static readonly Rational Zero = new(0), One = new(1);
15 public Rational(long numerator, long denominator = 1)
16 {
17 if (denominator < 0)
18 {
19 numerator = -numerator;
20 denominator = -denominator;
21 }
22 long a = Math.Abs(numerator), b = denominator;
23 while (b != 0)
24 {
25 long next = a % b;
26 a = b;
27 b = next;
28 }
29 Numerator = numerator / a;
30 Denominator = denominator / a;
31 }
32 public static Rational operator +(Rational a, Rational b) => new(a.Numerator * b.Denominator + b.Numerator * a.Denominator, a.Denominator * b.Denominator);
33 public static Rational operator -(Rational a) => new(-a.Numerator, a.Denominator);
34 public static Rational operator -(Rational a, Rational b) => a + -b;
35 public static Rational operator *(Rational a, Rational b) => new(a.Numerator * b.Numerator, a.Denominator * b.Denominator);
36 public static Rational operator /(Rational a, Rational b) => new(a.Numerator * b.Denominator, a.Denominator * b.Numerator);
37 public int CompareTo(Rational b) => (Numerator * b.Denominator).CompareTo(b.Numerator * Denominator);
38 public override string ToString() => Denominator == 1 ? Numerator.ToString(CultureInfo.InvariantCulture) : $"{Numerator}/{Denominator}";
39 public string Value()
40 {
41 long scaled = Math.Abs(Numerator) * 1000000 / Denominator;
42 return $"{this} ({(Numerator < 0 ? "-" : "")}{scaled / 1000000}.{scaled % 1000000:000000})";
43 }
44 }

46 static bool Digits(string text) => text.Length > 0 && text.All(c => c is >= '0' and <= '9');
47
48 static Rational Number(string original)
49 {
50 string text = original;
51 Rational multiplier = Rational.One;
52 if (text.EndsWith("meg", StringComparison.Ordinal))
53 {
54 text = text[..^3];
55 multiplier = new(1000000);
56 }
57 else if (text.EndsWith('k'))
58 {
59 text = text[..^1];
60 multiplier = new(1000);
61 }
62 else if (text.EndsWith('m'))
63 {
64 text = text[..^1];
65 multiplier = new(1, 1000);
66 }
67 else if (text.EndsWith('u'))
68 {
69 text = text[..^1];
70 multiplier = new(1, 1000000);
71 }
72 bool negative = text.StartsWith('-');
73 if (negative) text = text[1..];
74 Rational? value = null;
75 if (Digits(text)) value = new Rational(long.Parse(text));
76 else
77 {
78 int dot = text.IndexOf('.'), slash = text.IndexOf('/');
79 if (dot >= 0 && Digits(text[..dot]) && Digits(text[(dot + 1)..]))
80 {
81 long denominator = 1;
82 for (int i = dot + 1; i < text.Length; i++) denominator *= 10;
83 value = new Rational(long.Parse(text[..dot] + text[(dot + 1)..]), denominator);
84 }
85 else if (slash >= 0 && Digits(text[..slash]) && Digits(text[(slash + 1)..]))
86 {
87 long denominator = long.Parse(text[(slash + 1)..]);
88 if (denominator != 0) value = new Rational(long.Parse(text[..slash]), denominator);
89 }
90 }
91 if (!value.HasValue) Fail("invalid value " + original);
92 Rational result = value!.Value * multiplier;
93 return negative ? -result : result;
94 }

96 static bool ValidNode(string name) => name == "0" || name.Length is >= 1 and <= 8 && name[0] is >= 'a' and <= 'z' && name.All(c => c is >= 'a' and <= 'z' or >= '0' and <= '9');

110 class Groups
111 {
112 readonly Dictionary<string, string> parents = new(StringComparer.Ordinal);
113 public Groups(IEnumerable<string> nodes)
114 {
115 foreach (string node in nodes) parents[node] = node;
116 }
117 public string Find(string node)
118 {
119 string parent = parents[node];
120 if (parent != node) parents[node] = parent = Find(parent);
121 return parent;
122 }
123 public bool Join(string a, string b)
124 {
125 a = Find(a);
126 b = Find(b);
127 if (a == b) return false;
128 if (StringComparer.Ordinal.Compare(a, b) < 0) parents[b] = a;
129 else parents[a] = b;
130 return true;
131 }

134 static Rational[] Zero(int count) => Enumerable.Repeat(Rational.Zero, count).ToArray();
135
136 static Rational[][] ZeroMatrix(int count) => Enumerable.Range(0, count).Select(_ => Zero(count)).ToArray();
137
138 static Rational[]? Solve(Rational[][] matrix, Rational[] column)
139 {
140 int count = column.Length;
141 Rational[][] rows = Enumerable.Range(0, count).Select(i => matrix[i].Append(column[i]).ToArray()).ToArray();
142 for (int k = 0; k < count; k++)
143 {
144 int pivot = k;
145 while (pivot < count && rows[pivot][k].Numerator == 0) pivot++;
146 if (pivot == count) return null;
147 (rows[k], rows[pivot]) = (rows[pivot], rows[k]);
148 Rational value = rows[k][k];
149 for (int j = 0; j <= count; j++) rows[k][j] /= value;
150 for (int i = 0; i < count; i++) if (i != k && rows[i][k].Numerator != 0)
151 {
152 Rational factor = rows[i][k];
153 for (int j = 0; j <= count; j++) rows[i][j] -= factor * rows[k][j];
154 }
155 }
156 return Enumerable.Range(0, count).Select(i => rows[i][count]).ToArray();
157 }

159 class Model
160 {
161 public List<string> Nodes;
162 public List<Element> Sources;
163 public Dictionary<string, int> Indices = new(StringComparer.Ordinal);
164 public Rational[][] A;
165 public Rational[] B;
166 public Model(List<Element> elements, List<string> nodes, Element? changed = null, Rational replacement = default)
167 {
168 Nodes = nodes;
169 Sources = elements.Where(e => e.Voltage).ToList();
170 int count = nodes.Count + Sources.Count;
171 for (int i = 0; i < nodes.Count; i++) Indices[nodes[i]] = i;
172 for (int i = 0; i < Sources.Count; i++) Indices[Sources[i].Name] = nodes.Count + i;
173 A = ZeroMatrix(count);
174 B = Zero(count);
175 foreach (Element e in elements)
176 {
177 Rational w = e == changed ? replacement : e.Value;
178 switch (e.Kind)
179 {
180 case 'R':
181 Rational g = Rational.One / w;
182 Add(e.N1, e.N1, g);
183 Add(e.N2, e.N2, g);
184 Add(e.N1, e.N2, -g);
185 Add(e.N2, e.N1, -g);
186 break;
187 case 'I':
188 Add(B, e.N1, -w);
189 Add(B, e.N2, w);
190 break;
191 case 'G':
192 Add(e.N1, e.C1!, w);
193 Add(e.N2, e.C2!, w);
194 Add(e.N1, e.C2!, -w);
195 Add(e.N2, e.C1!, -w);
196 break;
197 case 'F':
198 Add(e.N1, e.Reference!, w);
199 Add(e.N2, e.Reference!, -w);
200 break;
201 default:
202 Add(e.N1, e.Name, Rational.One);
203 Add(e.Name, e.N1, Rational.One);
204 Add(e.N2, e.Name, -Rational.One);
205 Add(e.Name, e.N2, -Rational.One);
206 if (e.Kind == 'V') Add(B, e.Name, w);
207 else if (e.Kind == 'E')
208 {
209 Add(e.Name, e.C1!, -w);
210 Add(e.Name, e.C2!, w);
211 }
212 else Add(e.Name, e.Reference!, -w);
213 break;
214 }
215 }
216 }
217 public void Add(string row, string col, Rational value)
218 {
219 if (Indices.TryGetValue(row, out int i) && Indices.TryGetValue(col, out int j)) A[i][j] += value;
220 }
221 public void Add(Rational[] column, string row, Rational value)
222 {
223 if (Indices.TryGetValue(row, out int i)) column[i] += value;
224 }
225 public Rational Voltage(Rational[] solution, string node) => node == "0" ? Rational.Zero : solution[Indices[node]];
226 public Rational Current(Rational[] solution, Element e) => e.Kind switch
227 {
228 'R' => (Voltage(solution, e.N1) - Voltage(solution, e.N2)) / e.Value,
229 'I' => e.Value,
230 'G' => e.Value * (Voltage(solution, e.C1!) - Voltage(solution, e.C2!)),
231 'F' => e.Value * solution[Indices[e.Reference!]],
232 _ => solution[Indices[e.Name]]
233 };
234 public Rational[] Injection(string from, string to)
235 {
236 Rational[] rhs = Zero(B.Length);
237 Add(rhs, from, Rational.One);
238 Add(rhs, to, -Rational.One);
239 return Solve(A, rhs)!;
240 }
241 public Rational[] Alone(Element source, bool unit)
242 {
243 Rational[] rhs = Zero(B.Length);
244 Rational value = unit ? Rational.One : source.Value;
245 if (source.Kind == 'V') Add(rhs, source.Name, value);
246 else
247 {
248 Add(rhs, source.N1, -value);
249 Add(rhs, source.N2, value);
250 }
251 return Solve(A, rhs)!;
252 }

267 static HashSet<string> Reach(string start, List<Element> edges)
268 {
269 HashSet<string> reached = new(StringComparer.Ordinal) { start };
270 bool changed;
271 do
272 {
273 int before = reached.Count;
274 foreach (Element e in edges)
275 {
276 if (reached.Contains(e.N1)) reached.Add(e.N2);
277 if (reached.Contains(e.N2)) reached.Add(e.N1);
278 }
279 changed = before != reached.Count;
280 }
281 while (changed);
282 return reached;
283 }

308 void AddElement(string[] w)
309 {
310 string name = w[0];
311 char kind = name[0];
312 if (!"RVIEGFH".Contains(kind)) Fail("unknown element type " + name);
313 int count = kind is 'E' or 'G' ? 6 : kind is 'F' or 'H' ? 5 : 4;
314 if (w.Length != count) Fail("wrong field count for " + name);
315 if (Find(name) != null) Fail("duplicate element " + name);
316 for (int i = 1; i <= (count == 6 ? 4 : 2); i++) if (!ValidNode(w[i])) Fail("invalid node " + w[i]);
317 if (kind is 'F' or 'H')
318 {
319 Element? source = Find(w[3]);
320 if (source == null || source.Kind != 'V') Fail("unknown voltage source " + w[3]);
321 }
322 Rational value = Number(w[count - 1]);
323 if (kind == 'R' && value.Numerator <= 0) Fail("nonpositive resistance " + name);
324 Element e = new(name, w[1], w[2], value);
325 if (count == 6)
326 {
327 e.C1 = w[3];
328 e.C2 = w[4];
329 }
330 else if (count == 5) e.Reference = w[3];
331 elements.Add(e);
332 }

359 string Resistance(string from, string to)
360 {
361 Groups groups = new(Nodes(true).Append("0"));
362 foreach (Element e in elements) if (e.Kind == 'V') groups.Join(e.N1, e.N2);
363 string a = groups.Find(from), b = groups.Find(to);
364 if (a == b) return "Req = " + Rational.Zero.Value();
365 List<Element> edges = new();
366 foreach (Element e in elements.Where(e => e.Kind == 'R'))
367 {
368 string p = groups.Find(e.N1), q = groups.Find(e.N2);
369 if (p != q) edges.Add(new(e.Name, p, q, e.Value));
370 }
371 var reached = Reach(a, edges);
372 if (!reached.Contains(b)) return "Req = INFINITE";
373 List<string> unknowns = reached.Where(n => n != b).Order(StringComparer.Ordinal).ToList();
374 Dictionary<string, int> index = new(StringComparer.Ordinal);
375 for (int i = 0; i < unknowns.Count; i++) index[unknowns[i]] = i;
376 Rational[][] g = ZeroMatrix(unknowns.Count);
377 Rational[] rhs = Zero(unknowns.Count);
378 rhs[index[a]] = Rational.One;
379 foreach (Element e in edges) if (reached.Contains(e.N1))
380 {
381 Rational value = Rational.One / e.Value;
382 bool hasP = index.TryGetValue(e.N1, out int p), hasQ = index.TryGetValue(e.N2, out int q);
383 if (hasP) g[p][p] += value;
384 if (hasQ) g[q][q] += value;
385 if (hasP && hasQ)
386 {
387 g[p][q] -= value;
388 g[q][p] -= value;
389 }
390 }
391 return "Req = " + Solve(g, rhs)![index[a]].Value();
392 }

394 List<string> Reduce(string a, string b)
395 {
396 if (a == b) return new() { "Req = " + Rational.Zero.Value() };
397 List<Element> list = elements.Where(e => e.Kind == 'R').ToList();
398 var reached = Reach(a, list);
399 if (!reached.Contains(b)) return new() { "Req = INFINITE" };
400 list.RemoveAll(e => !reached.Contains(e.N1));
401 List<string> output = new();
402 int sequence = 0;
403 while (true)
404 {
405 int remove = list.FindIndex(e => e.N1 == e.N2);
406 if (remove >= 0)
407 {
408 output.Add("remove " + list[remove].Name);
409 list.RemoveAt(remove);
410 continue;
411 }
412 SortedDictionary<string, int> degrees = new(StringComparer.Ordinal);
413 foreach (Element e in list)
414 {
415 degrees[e.N1] = degrees.GetValueOrDefault(e.N1) + 1;
416 degrees[e.N2] = degrees.GetValueOrDefault(e.N2) + 1;
417 }
418 degrees.Remove(a);
419 degrees.Remove(b);
420 string? leaf = degrees.FirstOrDefault(entry => entry.Value == 1).Key;
421 if (leaf != null)
422 {
423 remove = list.FindIndex(e => e.N1 == leaf || e.N2 == leaf);
424 output.Add("remove " + list[remove].Name);
425 list.RemoveAt(remove);
426 continue;
427 }
428 int x = -1, y = -1;
429 bool parallel = false;
430 for (int i = 0; i < list.Count && x < 0; i++) for (int j = i + 1; j < list.Count; j++)
431 {
432 Element p = list[i], q = list[j];
433 if (p.N1 == q.N1 && p.N2 == q.N2 || p.N1 == q.N2 && p.N2 == q.N1)
434 {
435 x = i;
436 y = j;
437 parallel = true;
438 break;
439 }
440 }
441 string? middle = null;
442 if (x < 0)
443 {
444 middle = degrees.FirstOrDefault(entry => entry.Value == 2).Key;
445 if (middle == null) break;
446 for (int i = 0; i < list.Count; i++) if (list[i].N1 == middle || list[i].N2 == middle)
447 {
448 if (x < 0) x = i;
449 else y = i;
450 }
451 }
452 Element first = list[x], second = list[y];
453 Rational value = parallel ? first.Value * second.Value / (first.Value + second.Value) : first.Value + second.Value;
454 string left = parallel ? first.N1 : first.N1 == middle ? first.N2 : first.N1, right = parallel ? first.N2 : second.N1 == middle ? second.N2 : second.N1;
455 Element next = new("#" + (++sequence), left, right, value);
456 list.RemoveAt(y);
457 list.RemoveAt(x);
458 list.Add(next);
459 output.Add(next.Name + " = " + first.Name + (parallel ? " || " : " + ") + second.Name + " = " + value.Value());
460 }
461 output.Add(list.Count == 1 ? "Req = " + list[0].Value.Value() : $"not series-parallel: {list.Count} resistors remain");
462 return output;
463 }

465 List<string> Command(string[] w)
466 {
467 string c = w[0];
468 if (!Arity.TryGetValue(c, out int arity)) Fail("unknown command " + c);
469 if (w.Length - 1 != arity) Fail("wrong argument count for " + c);
470 List<string> output = new();
471 switch (c)
472 {
473 case ".ALTER":
474 {
475 Element e = Get(w[1]);
476 Rational value = Number(w[2]);
477 if (e.Kind == 'R' && value.Numerator <= 0) Fail("nonpositive resistance " + e.Name);
478 output.Add($"altered {e.Name}: {e.Value} -> {value}");
479 e.Value = value;
480 return output;
481 }
482 case ".DELETE":
483 {
484 Element e = Get(w[1]);
485 foreach (Element other in elements) if (other.Reference == w[1]) Fail(w[1] + " is referenced by " + other.Name);
486 elements.Remove(e);
487 return new() { "deleted " + e.Name };
488 }
489 case ".CLEAR":
490 {
491 int count = elements.Count;
492 elements.Clear();
493 return new() { $"cleared {count} elements" };
494 }
495 case ".PRINT":
496 return elements.Count == 0 ? new() { "empty netlist" } : elements.Select(e => e.ToString()).ToList();
497 case ".NODES":
498 foreach (string name in Nodes(true)) output.Add(name + ": " + string.Join(" ", elements.Where(e => e.Terminals.Contains(name)).Select(e => e.Name)));
499 return output.Count == 0 ? new() { "no nodes" } : output;
500 case ".REQ":
501 case ".REDUCE":
502 Node(w[1]);
503 Node(w[2]);
504 return c == ".REQ" ? new() { Resistance(w[1], w[2]) } : Reduce(w[1], w[2]);
505 case ".CHECK":
506 {
507 if (Nodes().Count == 0) return new() { "no nodes" };
508 foreach (string node in Nodes())
509 {
510 int degree = elements.Sum(e => (e.N1 == node ? 1 : 0) + (e.N2 == node ? 1 : 0));
511 if (degree < 2) output.Add("dangling node " + node);
512 }
513 output.AddRange(Floating().Select(n => "floating node " + n));
514 output.AddRange(Loops().Select(n => "voltage source loop at " + n));
515 if (output.Count == 0)
516 {
517 Model model = new(elements, Nodes());
518 if (Solve(model.A, model.B) == null) output.Add("singular matrix");
519 }
520 return output.Count == 0 ? new() { "ok" } : output;
521 }
522 case ".SWEEP":
523 {
524 Element e = Get(w[1]);
525 Rational from = Number(w[2]), to = Number(w[3]), step = Number(w[4]);
526 Node(w[5]);
527 if (step.Numerator <= 0) Fail("invalid sweep step");
528 if (from.CompareTo(to) > 0) Fail("invalid sweep range");
529 Rational span = (to - from) / step;
530 long count = span.Numerator / span.Denominator + 1;
531 if (count > 20) Fail("too many sweep points");
532 if (e.Kind == 'R' && from.Numerator <= 0) Fail("nonpositive resistance " + e.Name);
533 Structure();
534 for (int k = 0; k < count; k++)
535 {
536 Rational value = from + step * new Rational(k);
537 Model model = new(elements, Nodes(), e, value);
538 Rational[]? solution = Solve(model.A, model.B);
539 output.Add(e.Name + " = " + value + ": " + (solution == null ? "singular matrix" : $"V({w[5]}) = {model.Voltage(solution, w[5]).Value()}"));
540 }
541 return output;
542 }
543 }
544 Element? selected = null;
545 switch (c)
546 {
547 case ".CURRENT":
548 selected = Get(w[1]);
549 break;
550 case ".VOLTAGE":
551 case ".THEVENIN":
552 case ".NORTON":
553 case ".MAXPOWER":
554 Node(w[1]);
555 Node(w[2]);
556 break;
557 case ".KCL":
558 case ".SENS":
559 case ".SUPERPOSITION":
560 Node(w[1]);
561 break;
562 case ".ZPARAM":
563 for (int i = 1; i <= 4; i++) Node(w[i]);
564 break;
565 case ".TF":
566 Node(w[1]);
567 selected = Get(w[2]);
568 if (!selected.Independent) Fail("not an independent source " + selected.Name);
569 break;
570 }
571 Structure();
572 Model circuit = new(elements, Nodes());
573 if (c == ".MATRIX")
574 {
575 output.Add("unknowns: " + string.Join(" ", circuit.Nodes.Select(n => "V(" + n + ")").Concat(circuit.Sources.Select(e => "I(" + e.Name + ")"))));
576 for (int i = 0; i < circuit.B.Length; i++)
577 {
578 string label = i < circuit.Nodes.Count ? circuit.Nodes[i] : circuit.Sources[i - circuit.Nodes.Count].Name;
579 output.Add(label + ": " + string.Join(" ", circuit.A[i].Select(q => q.ToString())) + " | " + circuit.B[i]);
580 }
581 return output;
582 }
583 Rational[] point = Solve(circuit.A, circuit.B) ?? throw new Problem("singular matrix");
584 switch (c)
585 {
586 case ".OP":
587 foreach (string node in circuit.Nodes) output.Add($"V({node}) = {circuit.Voltage(point, node).Value()}");
588 foreach (Element e in circuit.Sources) output.Add($"I({e.Name}) = {circuit.Current(point, e).Value()}");
589 break;
590 case ".CURRENT":
591 output.Add($"I({selected!.Name}) = {circuit.Current(point, selected).Value()}");
592 break;
593 case ".VOLTAGE":
594 output.Add($"V({w[1]},{w[2]}) = {(circuit.Voltage(point, w[1]) - circuit.Voltage(point, w[2])).Value()}");
595 break;
596 case ".POWER":
597 {
598 Rational absorbed = Rational.Zero, delivered = Rational.Zero, total = Rational.Zero;
599 foreach (Element e in elements)
600 {
601 Rational power = (circuit.Voltage(point, e.N1) - circuit.Voltage(point, e.N2)) * circuit.Current(point, e);
602 output.Add($"P({e.Name}) = {power.Value()}");
603 if (power.Numerator > 0) absorbed += power;
604 if (power.Numerator < 0) delivered -= power;
605 total += power;
606 }
607 output.Add("absorbed = " + absorbed.Value());
608 output.Add("delivered = " + delivered.Value());
609 output.Add("total = " + total.Value());
610 break;
611 }
612 case ".KCL":
613 {
614 Rational total = Rational.Zero;
615 foreach (Element e in elements) if (e.N1 == w[1] || e.N2 == w[1])
616 {
617 Rational value = e.N1 == e.N2 ? Rational.Zero : e.N1 == w[1] ? circuit.Current(point, e) : -circuit.Current(point, e);
618 output.Add($"I({e.Name}) leaving {w[1]} = {value.Value()}");
619 total += value;
620 }
621 output.Add("sum = " + total.Value());
622 break;
623 }
624 case ".THEVENIN":
625 case ".NORTON":
626 case ".MAXPOWER":
627 {
628 Rational voltage = circuit.Voltage(point, w[1]) - circuit.Voltage(point, w[2]);
629 Rational[] injection = circuit.Injection(w[1], w[2]);
630 Rational resistance = circuit.Voltage(injection, w[1]) - circuit.Voltage(injection, w[2]);
631 if (c == ".NORTON" && resistance.Numerator == 0) Fail("zero thevenin resistance");
632 if (c == ".MAXPOWER" && resistance.Numerator <= 0) Fail("nonpositive thevenin resistance");
633 if (c == ".THEVENIN")
634 {
635 output.Add("Vth = " + voltage.Value());
636 output.Add("Rth = " + resistance.Value());
637 }
638 else if (c == ".NORTON")
639 {
640 output.Add("In = " + (voltage / resistance).Value());
641 output.Add("Rn = " + resistance.Value());
642 }
643 else
644 {
645 output.Add("RL = " + resistance.Value());
646 output.Add("Pmax = " + (voltage * voltage / (new Rational(4) * resistance)).Value());
647 }
648 break;
649 }
650 case ".TF":
651 {
652 Rational[] unit = circuit.Alone(selected!, true);
653 output.Add("gain = " + circuit.Voltage(unit, w[1]).Value());
654 if (selected!.Kind == 'V')
655 {
656 Rational current = unit[circuit.Indices[selected.Name]];
657 output.Add("input resistance = " + (current.Numerator == 0 ? "INFINITE" : (-Rational.One / current).Value()));
658 }
659 else output.Add("input resistance = " + (circuit.Voltage(unit, selected.N2) - circuit.Voltage(unit, selected.N1)).Value());
660 output.Add("output resistance = " + circuit.Voltage(circuit.Injection(w[1], "0"), w[1]).Value());
661 break;
662 }
663 case ".SENS":
664 foreach (Element e in elements)
665 {
666 Rational[] rhs = Zero(circuit.B.Length);
667 switch (e.Kind)
668 {
669 case 'R':
670 {
671 Rational value = circuit.Current(point, e) / e.Value;
672 circuit.Add(rhs, e.N1, value);
673 circuit.Add(rhs, e.N2, -value);
674 break;
675 }
676 case 'I':
677 circuit.Add(rhs, e.N1, -Rational.One);
678 circuit.Add(rhs, e.N2, Rational.One);
679 break;
680 case 'G':
681 case 'F':
682 {
683 Rational value = e.Kind == 'G' ? circuit.Voltage(point, e.C1!) - circuit.Voltage(point, e.C2!) : point[circuit.Indices[e.Reference!]];
684 circuit.Add(rhs, e.N1, -value);
685 circuit.Add(rhs, e.N2, value);
686 break;
687 }
688 case 'V':
689 circuit.Add(rhs, e.Name, Rational.One);
690 break;
691 case 'E':
692 circuit.Add(rhs, e.Name, circuit.Voltage(point, e.C1!) - circuit.Voltage(point, e.C2!));
693 break;
694 case 'H':
695 circuit.Add(rhs, e.Name, point[circuit.Indices[e.Reference!]]);
696 break;
697 }
698 output.Add($"dV({w[1]})/d{e.Name} = {circuit.Voltage(Solve(circuit.A, rhs)!, w[1]).Value()}");
699 }
700 break;
701 case ".ZPARAM":
702 {
703 Rational[] first = circuit.Injection(w[1], w[2]), second = circuit.Injection(w[3], w[4]);
704 output.Add("z11 = " + (circuit.Voltage(first, w[1]) - circuit.Voltage(first, w[2])).Value());
705 output.Add("z12 = " + (circuit.Voltage(second, w[1]) - circuit.Voltage(second, w[2])).Value());
706 output.Add("z21 = " + (circuit.Voltage(first, w[3]) - circuit.Voltage(first, w[4])).Value());
707 output.Add("z22 = " + (circuit.Voltage(second, w[3]) - circuit.Voltage(second, w[4])).Value());
708 break;
709 }
710 case ".SUPERPOSITION":
711 {
712 Rational total = Rational.Zero;
713 foreach (Element e in elements) if (e.Independent)
714 {
715 Rational value = circuit.Voltage(circuit.Alone(e, false), w[1]);
716 output.Add($"{e.Name}: V({w[1]}) = {value.Value()}");
717 total += value;
718 }
719 output.Add($"total: V({w[1]}) = {total.Value()}");
720 break;
721 }
722 }
723 return output;
724 }

730 string? line;

739 {

742 }

746 }
747 }
