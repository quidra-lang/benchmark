25 record Rational(long numerator, long denominator) implements Comparable<Rational> {
26 static final Rational ZERO = new Rational(0, 1), ONE = new Rational(1, 1);
27 Rational {
28 if (denominator < 0) {
29 numerator = -numerator;
30 denominator = -denominator;
31 }
32 long g = gcd(Math.abs(numerator), denominator);
33 numerator /= g;
34 denominator /= g;
35 }
36 static long gcd(long a, long b) {
37 while (b != 0) {
38 long next = a % b;
39 a = b;
40 b = next;
41 }
42 return a;
43 }
44 Rational add(Rational b) {
45 return new Rational(numerator * b.denominator + b.numerator * denominator, denominator * b.denominator);
46 }
47 Rational negate() {
48 return new Rational(-numerator, denominator);
49 }
50 Rational subtract(Rational b) {
51 return add(b.negate());
52 }
53 Rational multiply(Rational b) {
54 return new Rational(numerator * b.numerator, denominator * b.denominator);
55 }
56 Rational divide(Rational b) {
57 return new Rational(numerator * b.denominator, denominator * b.numerator);
58 }
59 public int compareTo(Rational b) {
60 return Long.compare(numerator * b.denominator, b.numerator * denominator);
61 }
62 public String toString() {
63 return denominator == 1 ? Long.toString(numerator) : numerator + "/" + denominator;
64 }
65 String value() {
66 long scaled = Math.abs(numerator) * 1000000 / denominator;
67 String tail = Long.toString(scaled % 1000000);
68 return this + " (" + (numerator < 0 ? "-" : "") + scaled / 1000000 + "." + "0".repeat(6 - tail.length()) + tail + ")";
69 }
70 }

72 static boolean digits(String text) {
73 if (text.isEmpty()) return false;
74 for (char c : text.toCharArray()) if (c < '0' || c > '9') return false;
75 return true;
76 }

78 static Rational number(String original) {
79 String text = original;
80 Rational multiplier = Rational.ONE;
81 if (text.endsWith("meg")) {
82 text = text.substring(0, text.length() - 3);
83 multiplier = new Rational(1000000, 1);
84 } else if (text.endsWith("k")) {
85 text = text.substring(0, text.length() - 1);
86 multiplier = new Rational(1000, 1);
87 } else if (text.endsWith("m")) {
88 text = text.substring(0, text.length() - 1);
89 multiplier = new Rational(1, 1000);
90 } else if (text.endsWith("u")) {
91 text = text.substring(0, text.length() - 1);
92 multiplier = new Rational(1, 1000000);
93 }
94 boolean negative = text.startsWith("-");
95 if (negative) text = text.substring(1);
96 Rational value = null;
97 if (digits(text)) value = new Rational(Long.parseLong(text), 1);
98 else {
99 int dot = text.indexOf('.'), slash = text.indexOf('/');
100 if (dot >= 0 && digits(text.substring(0, dot)) && digits(text.substring(dot + 1))) {
101 long denominator = 1;
102 for (int i = dot + 1; i < text.length(); i++) denominator *= 10;
103 value = new Rational(Long.parseLong(text.substring(0, dot) + text.substring(dot + 1)), denominator);
104 } else if (slash >= 0 && digits(text.substring(0, slash)) && digits(text.substring(slash + 1))) {
105 long denominator = Long.parseLong(text.substring(slash + 1));
106 if (denominator != 0) value = new Rational(Long.parseLong(text.substring(0, slash)), denominator);
107 }
108 }
109 if (value == null) fail("invalid value " + original);
110 value = value.multiply(multiplier);
111 return negative ? value.negate() : value;
112 }

115 if (name.equals("0")) return true;
116 if (name.isEmpty() || name.length() > 8 || name.charAt(0) < 'a' || name.charAt(0) > 'z') return false;
117 for (char c : name.toCharArray()) if (!(c >= 'a' && c <= 'z' || c >= '0' && c <= '9')) return false;
118 return true;

146 static class Groups {
147 Map<String, String> parents = new TreeMap<>();
148 Groups(Iterable<String> nodes) {
149 for (String node : nodes) parents.put(node, node);
150 }
151 String find(String node) {
152 String parent = parents.get(node);
153 if (!parent.equals(node)) parents.put(node, parent = find(parent));
154 return parent;
155 }
156 boolean join(String a, String b) {
157 a = find(a);
158 b = find(b);
159 if (a.equals(b)) return false;
160 if (a.compareTo(b) < 0) parents.put(b, a);
161 else parents.put(a, b);
162 return true;
163 }

172 static Rational[][] zeroMatrix(int count) {
173 Rational[][] matrix = new Rational[count][];
174 for (int i = 0; i < count; i++) matrix[i] = zero(count);
175 return matrix;

178 static Rational[] solve(Rational[][] matrix, Rational[] column) {
179 int count = column.length;
180 Rational[][] rows = new Rational[count][count + 1];
181 for (int i = 0; i < count; i++) {
182 System.arraycopy(matrix[i], 0, rows[i], 0, count);
183 rows[i][count] = column[i];
184 }
185 for (int k = 0; k < count; k++) {
186 int pivot = k;
187 while (pivot < count && rows[pivot][k].numerator == 0) pivot++;
188 if (pivot == count) return null;
189 Rational[] swap = rows[k];
190 rows[k] = rows[pivot];
191 rows[pivot] = swap;
192 Rational value = rows[k][k];
193 for (int j = 0; j <= count; j++) rows[k][j] = rows[k][j].divide(value);
194 for (int i = 0; i < count; i++) if (i != k && rows[i][k].numerator != 0) {
195 Rational factor = rows[i][k];
196 for (int j = 0; j <= count; j++) rows[i][j] = rows[i][j].subtract(factor.multiply(rows[k][j]));
197 }
198 }
199 Rational[] solution = new Rational[count];
200 for (int i = 0; i < count; i++) solution[i] = rows[i][count];
201 return solution;
202 }

204 static class Model {
205 List<String> nodes;
206 List<Element> sources;
207 Map<String, Integer> indices = new HashMap<>();
208 Rational[][] a;
209 Rational[] b;
210 Model(List<Element> elements, List<String> nodes, Element changed, Rational replacement) {
211 this.nodes = nodes;
212 sources = elements.stream().filter(Element::voltage).toList();
213 int n = nodes.size() + sources.size();
214 for (int i = 0; i < nodes.size(); i++) indices.put(nodes.get(i), i);
215 for (int i = 0; i < sources.size(); i++) indices.put(sources.get(i).name, nodes.size() + i);
216 a = zeroMatrix(n);
217 b = zero(n);
218 for (Element e : elements) {
219 Rational w = e == changed ? replacement : e.value, negative = w.negate();
220 switch (e.kind) {
221 case 'R' -> {
222 Rational g = Rational.ONE.divide(w);
223 add(e.n1, e.n1, g);
224 add(e.n2, e.n2, g);
225 add(e.n1, e.n2, g.negate());
226 add(e.n2, e.n1, g.negate());
227 }
228 case 'I' -> {
229 add(b, e.n1, negative);
230 add(b, e.n2, w);
231 }
232 case 'G' -> {
233 add(e.n1, e.c1, w);
234 add(e.n2, e.c2, w);
235 add(e.n1, e.c2, negative);
236 add(e.n2, e.c1, negative);
237 }
238 case 'F' -> {
239 add(e.n1, e.reference, w);
240 add(e.n2, e.reference, negative);
241 }
242 default -> {
243 add(e.n1, e.name, Rational.ONE);
244 add(e.name, e.n1, Rational.ONE);
245 add(e.n2, e.name, Rational.ONE.negate());
246 add(e.name, e.n2, Rational.ONE.negate());
247 if (e.kind == 'V') add(b, e.name, w);
248 else if (e.kind == 'E') {
249 add(e.name, e.c1, negative);
250 add(e.name, e.c2, w);
251 } else add(e.name, e.reference, negative);
252 }
253 }
254 }
255 }

258 if (i != null && j != null) a[i][j] = a[i][j].add(value);

260 void add(Rational[] column, String row, Rational value) {
261 Integer i = indices.get(row);
262 if (i != null) column[i] = column[i].add(value);

264 Rational voltage(Rational[] solution, String node) {
265 return node.equals("0") ? Rational.ZERO : solution[indices.get(node)];
266 }
267 Rational current(Rational[] solution, Element e) {
268 return switch (e.kind) {
269 case 'R' -> voltage(solution, e.n1).subtract(voltage(solution, e.n2)).divide(e.value);
270 case 'I' -> e.value;
271 case 'G' -> e.value.multiply(voltage(solution, e.c1).subtract(voltage(solution, e.c2)));
272 case 'F' -> e.value.multiply(solution[indices.get(e.reference)]);
273 default -> solution[indices.get(e.name)];
274 };
275 }
276 Rational[] injection(String from, String to) {
277 Rational[] rhs = zero(b.length);
278 add(rhs, from, Rational.ONE);
279 add(rhs, to, Rational.ONE.negate());
280 return solve(a, rhs);
281 }
282 Rational[] alone(Element source, boolean unit) {
283 Rational[] rhs = zero(b.length);
284 Rational value = unit ? Rational.ONE : source.value;
285 if (source.kind == 'V') add(rhs, source.name, value);
286 else {
287 add(rhs, source.n1, value.negate());
288 add(rhs, source.n2, value);
289 }
290 return solve(a, rhs);
291 }

317 Set<String> reach(String start, List<Element> edges) {
318 Set<String> reached = new HashSet<>();
319 reached.add(start);
320 boolean changed;
321 do {
322 int before = reached.size();
323 for (Element e : edges) {
324 if (reached.contains(e.n1)) reached.add(e.n2);
325 if (reached.contains(e.n2)) reached.add(e.n1);
326 }
327 changed = reached.size() != before;
328 }
329 while (changed);
330 return reached;
331 }

355 void element(String[] words) {
356 String name = words[0];
357 char kind = name.charAt(0);
358 if ("RVIEGFH".indexOf(kind) < 0) fail("unknown element type " + name);
359 int count = kind == 'E' || kind == 'G' ? 6 : kind == 'F' || kind == 'H' ? 5 : 4;
360 if (words.length != count) fail("wrong field count for " + name);
361 if (find(name) != null) fail("duplicate element " + name);
362 for (int i = 1; i <= (count == 6 ? 4 : 2); i++) if (!validNode(words[i])) fail("invalid node " + words[i]);
363 if (kind == 'F' || kind == 'H') {
364 Element source = find(words[3]);
365 if (source == null || source.kind != 'V') fail("unknown voltage source " + words[3]);
366 }
367 Rational value = number(words[count - 1]);
368 if (kind == 'R' && value.numerator <= 0) fail("nonpositive resistance " + name);
369 Element e = new Element(name, words[1], words[2], value);
370 if (count == 6) {
371 e.c1 = words[3];
372 e.c2 = words[4];
373 } else if (count == 5) e.reference = words[3];
374 elements.add(e);
375 }

401 String resistance(String from, String to) {
402 TreeSet<String> names = new TreeSet<>(nodes(true));
403 names.add("0");
404 Groups groups = new Groups(names);
405 for (Element e : elements) if (e.kind == 'V') groups.join(e.n1, e.n2);
406 String a = groups.find(from), b = groups.find(to);
407 if (a.equals(b)) return "Req = " + Rational.ZERO.value();
408 List<Element> edges = new ArrayList<>();
409 for (Element e : elements) if (e.kind == 'R') {
410 String p = groups.find(e.n1), q = groups.find(e.n2);
411 if (!p.equals(q)) edges.add(new Element(e.name, p, q, e.value));
412 }
413 Set<String> reached = reach(a, edges);
414 if (!reached.contains(b)) return "Req = INFINITE";
415 List<String> unknowns = reached.stream().filter(n -> !n.equals(b)).sorted().toList();
416 Map<String, Integer> index = new HashMap<>();
417 for (int i = 0; i < unknowns.size(); i++) index.put(unknowns.get(i), i);
418 Rational[][] g = zeroMatrix(unknowns.size());
419 Rational[] rhs = zero(unknowns.size());
420 rhs[index.get(a)] = Rational.ONE;
421 for (Element e : edges) if (reached.contains(e.n1)) {
422 Rational value = Rational.ONE.divide(e.value);
423 Integer p = index.get(e.n1), q = index.get(e.n2);
424 if (p != null) g[p][p] = g[p][p].add(value);
425 if (q != null) g[q][q] = g[q][q].add(value);
426 if (p != null && q != null) {
427 g[p][q] = g[p][q].subtract(value);
428 g[q][p] = g[q][p].subtract(value);
429 }
430 }
431 return "Req = " + solve(g, rhs)[index.get(a)].value();
432 }

434 List<String> reduce(String a, String b) {
435 if (a.equals(b)) return List.of("Req = " + Rational.ZERO.value());
436 List<Element> list = new ArrayList<>(elements.stream().filter(e -> e.kind == 'R').toList());
437 Set<String> reached = reach(a, list);
438 if (!reached.contains(b)) return List.of("Req = INFINITE");
439 list.removeIf(e -> !reached.contains(e.n1));
440 List<String> out = new ArrayList<>();
441 int sequence = 0;
442 while (true) {
443 int remove = -1;
444 for (int i = 0; i < list.size(); i++) if (list.get(i).n1.equals(list.get(i).n2)) {
445 remove = i;
446 break;
447 }
448 if (remove >= 0) {
449 out.add("remove " + list.remove(remove).name);
450 continue;
451 }
452 TreeMap<String, Integer> degrees = new TreeMap<>();
453 for (Element e : list) {
454 degrees.merge(e.n1, 1, Integer::sum);
455 degrees.merge(e.n2, 1, Integer::sum);
456 }
457 degrees.remove(a);
458 degrees.remove(b);
459 String leaf = null;
460 for (Map.Entry<String, Integer> entry : degrees.entrySet()) if (entry.getValue() == 1) {
461 leaf = entry.getKey();
462 break;
463 }
464 if (leaf != null) {
465 for (int i = 0; i < list.size(); i++) if (list.get(i).n1.equals(leaf) || list.get(i).n2.equals(leaf)) {
466 out.add("remove " + list.remove(i).name);
467 break;
468 }
469 continue;
470 }
471 int x = -1, y = -1;
472 boolean parallel = false;
473 search: for (int i = 0; i < list.size(); i++) for (int j = i + 1; j < list.size(); j++) {
474 Element p = list.get(i), q = list.get(j);
475 if (p.n1.equals(q.n1) && p.n2.equals(q.n2) || p.n1.equals(q.n2) && p.n2.equals(q.n1)) {
476 x = i;
477 y = j;
478 parallel = true;
479 break search;
480 }
481 }
482 String middle = null;
483 if (x < 0) {
484 for (Map.Entry<String, Integer> entry : degrees.entrySet()) if (entry.getValue() == 2) {
485 middle = entry.getKey();
486 break;
487 }
488 if (middle == null) break;
489 for (int i = 0; i < list.size(); i++) if (list.get(i).n1.equals(middle) || list.get(i).n2.equals(middle)) {
490 if (x < 0) x = i;
491 else y = i;
492 }
493 }
494 Element p = list.get(x), q = list.get(y);
495 Rational value = parallel ? p.value.multiply(q.value).divide(p.value.add(q.value)) : p.value.add(q.value);
496 String left = parallel ? p.n1 : p.n1.equals(middle) ? p.n2 : p.n1, right = parallel ? p.n2 : q.n1.equals(middle) ? q.n2 : q.n1;
497 Element next = new Element("#" + (++sequence), left, right, value);
498 list.remove(y);
499 list.remove(x);
500 list.add(next);
501 out.add(next.name + " = " + p.name + (parallel ? " || " : " + ") + q.name + " = " + value.value());
502 }
503 out.add(list.size() == 1 ? "Req = " + list.getFirst().value.value() : "not series-parallel: " + list.size() + " resistors remain");
504 return out;
505 }

507 List<String> command(String[] w) {
508 String c = w[0];
509 if (!ARITY.containsKey(c)) fail("unknown command " + c);
510 if (w.length - 1 != ARITY.get(c)) fail("wrong argument count for " + c);
511 List<String> out = new ArrayList<>();
512 switch (c) {
513 case ".ALTER" -> {
514 Element e = get(w[1]);
515 Rational value = number(w[2]);
516 if (e.kind == 'R' && value.numerator <= 0) fail("nonpositive resistance " + e.name);
517 out.add("altered " + e.name + ": " + e.value + " -> " + value);
518 e.value = value;
519 return out;
520 }
521 case ".DELETE" -> {
522 Element e = get(w[1]);
523 for (Element other : elements) if (w[1].equals(other.reference)) fail(w[1] + " is referenced by " + other.name);
524 elements.remove(e);
525 return List.of("deleted " + e.name);
526 }
527 case ".CLEAR" -> {
528 int count = elements.size();
529 elements.clear();
530 return List.of("cleared " + count + " elements");
531 }
532 case ".PRINT" -> {
533 return elements.isEmpty() ? List.of("empty netlist") : elements.stream().map(Element::toString).toList();
534 }
535 case ".NODES" -> {
536 for (String name : nodes(true)) out.add(name + ": " + String.join(" ", elements.stream().filter(e -> e.terminals().contains(name)).map(e -> e.name).toList()));
537 return out.isEmpty() ? List.of("no nodes") : out;
538 }
539 case ".REQ", ".REDUCE" -> {
540 node(w[1]);
541 node(w[2]);
542 return c.equals(".REQ") ? List.of(resistance(w[1], w[2])) : reduce(w[1], w[2]);
543 }
544 case ".CHECK" -> {
545 if (nodes(false).isEmpty()) return List.of("no nodes");
546 for (String node : nodes(false)) {
547 int degree = 0;
548 for (Element e : elements) {
549 if (e.n1.equals(node)) degree++;
550 if (e.n2.equals(node)) degree++;
551 }
552 if (degree < 2) out.add("dangling node " + node);
553 }
554 for (String node : floating()) out.add("floating node " + node);
555 for (String name : loops()) out.add("voltage source loop at " + name);
556 if (out.isEmpty()) {
557 Model model = new Model(elements, nodes(false), null, null);
558 if (solve(model.a, model.b) == null) out.add("singular matrix");
559 }
560 return out.isEmpty() ? List.of("ok") : out;
561 }
562 case ".SWEEP" -> {
563 Element e = get(w[1]);
564 Rational from = number(w[2]), to = number(w[3]), step = number(w[4]);
565 node(w[5]);
566 if (step.numerator <= 0) fail("invalid sweep step");
567 if (from.compareTo(to) > 0) fail("invalid sweep range");
568 Rational span = to.subtract(from).divide(step);
569 long count = span.numerator / span.denominator + 1;
570 if (count > 20) fail("too many sweep points");
571 if (e.kind == 'R' && from.numerator <= 0) fail("nonpositive resistance " + e.name);
572 structure();
573 for (int k = 0; k < count; k++) {
574 Rational value = from.add(step.multiply(new Rational(k, 1)));
575 Model model = new Model(elements, nodes(false), e, value);
576 Rational[] solution = solve(model.a, model.b);
577 out.add(e.name + " = " + value + ": " + (solution == null ? "singular matrix" : "V(" + w[5] + ") = " + model.voltage(solution, w[5]).value()));
578 }
579 return out;
580 }
581 }
582 Element selected = null;
583 switch (c) {
584 case ".CURRENT" -> selected = get(w[1]);
585 case ".VOLTAGE", ".THEVENIN", ".NORTON", ".MAXPOWER" -> {
586 node(w[1]);
587 node(w[2]);
588 }
589 case ".KCL", ".SENS", ".SUPERPOSITION" -> node(w[1]);
590 case ".ZPARAM" -> {
591 for (int i = 1; i <= 4; i++) node(w[i]);
592 }
593 case ".TF" -> {
594 node(w[1]);
595 selected = get(w[2]);
596 if (!selected.independent()) fail("not an independent source " + selected.name);
597 }
598 }
599 structure();
600 Model model = new Model(elements, nodes(false), null, null);
601 if (c.equals(".MATRIX")) {
602 List<String> names = new ArrayList<>();
603 names.addAll(model.nodes.stream().map(n -> "V(" + n + ")").toList());
604 names.addAll(model.sources.stream().map(e -> "I(" + e.name + ")").toList());
605 out.add("unknowns: " + String.join(" ", names));
606 for (int i = 0; i < model.b.length; i++) {
607 String label = i < model.nodes.size() ? model.nodes.get(i) : model.sources.get(i - model.nodes.size()).name;
608 out.add(label + ": " + String.join(" ", Arrays.stream(model.a[i]).map(Rational::toString).toList()) + " | " + model.b[i]);
609 }
610 return out;
611 }
612 Rational[] solution = solve(model.a, model.b);
613 if (solution == null) fail("singular matrix");
614 switch (c) {
615 case ".OP" -> {
616 for (String node : model.nodes) out.add("V(" + node + ") = " + model.voltage(solution, node).value());
617 for (Element e : model.sources) out.add("I(" + e.name + ") = " + model.current(solution, e).value());
618 }
619 case ".CURRENT" -> out.add("I(" + selected.name + ") = " + model.current(solution, selected).value());
620 case ".VOLTAGE" -> out.add("V(" + w[1] + "," + w[2] + ") = " + model.voltage(solution, w[1]).subtract(model.voltage(solution, w[2])).value());
621 case ".POWER" -> {
622 Rational absorbed = Rational.ZERO, delivered = Rational.ZERO, total = Rational.ZERO;
623 for (Element e : elements) {
624 Rational power = model.voltage(solution, e.n1).subtract(model.voltage(solution, e.n2)).multiply(model.current(solution, e));
625 out.add("P(" + e.name + ") = " + power.value());
626 if (power.numerator > 0) absorbed = absorbed.add(power);
627 if (power.numerator < 0) delivered = delivered.subtract(power);
628 total = total.add(power);
629 }
630 out.add("absorbed = " + absorbed.value());
631 out.add("delivered = " + delivered.value());
632 out.add("total = " + total.value());
633 }
634 case ".KCL" -> {
635 Rational total = Rational.ZERO;
636 for (Element e : elements) if (e.n1.equals(w[1]) || e.n2.equals(w[1])) {
637 Rational value = e.n1.equals(e.n2) ? Rational.ZERO : e.n1.equals(w[1]) ? model.current(solution, e) : model.current(solution, e).negate();
638 out.add("I(" + e.name + ") leaving " + w[1] + " = " + value.value());
639 total = total.add(value);
640 }
641 out.add("sum = " + total.value());
642 }
643 case ".THEVENIN", ".NORTON", ".MAXPOWER" -> {
644 Rational voltage = model.voltage(solution, w[1]).subtract(model.voltage(solution, w[2]));
645 Rational[] injection = model.injection(w[1], w[2]);
646 Rational resistance = model.voltage(injection, w[1]).subtract(model.voltage(injection, w[2]));
647 if (c.equals(".NORTON") && resistance.numerator == 0) fail("zero thevenin resistance");
648 if (c.equals(".MAXPOWER") && resistance.numerator <= 0) fail("nonpositive thevenin resistance");
649 if (c.equals(".THEVENIN")) {
650 out.add("Vth = " + voltage.value());
651 out.add("Rth = " + resistance.value());
652 } else if (c.equals(".NORTON")) {
653 out.add("In = " + voltage.divide(resistance).value());
654 out.add("Rn = " + resistance.value());
655 } else {
656 out.add("RL = " + resistance.value());
657 out.add("Pmax = " + voltage.multiply(voltage).divide(new Rational(4, 1).multiply(resistance)).value());
658 }
659 }
660 case ".TF" -> {
661 Rational[] unit = model.alone(selected, true);
662 out.add("gain = " + model.voltage(unit, w[1]).value());
663 if (selected.kind == 'V') {
664 Rational current = unit[model.indices.get(selected.name)];
665 out.add("input resistance = " + (current.numerator == 0 ? "INFINITE" : Rational.ONE.negate().divide(current).value()));
666 } else out.add("input resistance = " + model.voltage(unit, selected.n2).subtract(model.voltage(unit, selected.n1)).value());
667 out.add("output resistance = " + model.voltage(model.injection(w[1], "0"), w[1]).value());
668 }
669 case ".SENS" -> {
670 for (Element e : elements) {
671 Rational[] rhs = zero(model.b.length);
672 switch (e.kind) {
673 case 'R' -> {
674 Rational value = model.current(solution, e).divide(e.value);
675 model.add(rhs, e.n1, value);
676 model.add(rhs, e.n2, value.negate());
677 }
678 case 'I' -> {
679 model.add(rhs, e.n1, Rational.ONE.negate());
680 model.add(rhs, e.n2, Rational.ONE);
681 }
682 case 'G', 'F' -> {
683 Rational value = e.kind == 'G' ? model.voltage(solution, e.c1).subtract(model.voltage(solution, e.c2)) : solution[model.indices.get(e.reference)];
684 model.add(rhs, e.n1, value.negate());
685 model.add(rhs, e.n2, value);
686 }
687 case 'V' -> model.add(rhs, e.name, Rational.ONE);
688 case 'E' -> model.add(rhs, e.name, model.voltage(solution, e.c1).subtract(model.voltage(solution, e.c2)));
689 case 'H' -> model.add(rhs, e.name, solution[model.indices.get(e.reference)]);
690 }
691 out.add("dV(" + w[1] + ")/d" + e.name + " = " + model.voltage(solve(model.a, rhs), w[1]).value());
692 }
693 }
694 case ".ZPARAM" -> {
695 Rational[] first = model.injection(w[1], w[2]), second = model.injection(w[3], w[4]);
696 out.add("z11 = " + model.voltage(first, w[1]).subtract(model.voltage(first, w[2])).value());
697 out.add("z12 = " + model.voltage(second, w[1]).subtract(model.voltage(second, w[2])).value());
698 out.add("z21 = " + model.voltage(first, w[3]).subtract(model.voltage(first, w[4])).value());
699 out.add("z22 = " + model.voltage(second, w[3]).subtract(model.voltage(second, w[4])).value());
700 }
701 case ".SUPERPOSITION" -> {
702 Rational total = Rational.ZERO;
703 for (Element e : elements) if (e.independent()) {
704 Rational value = model.voltage(model.alone(e, false), w[1]);
705 out.add(e.name + ": V(" + w[1] + ") = " + value.value());
706 total = total.add(value);
707 }
708 out.add("total: V(" + w[1] + ") = " + total.value());
709 }
710 default -> throw new IllegalStateException();
711 }
712 return out;
713 }
714 public static void main(String[] args) throws Exception {
715 BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
716 Main circuit = new Main();
717 String line;
718 int number = 0;
719 while ((line = reader.readLine()) != null) {
720 number++;
721 if (line.equals(".END")) break;
722 if (line.startsWith("*")) continue;
723 String[] words = line.split(" ");
724 try {
725 if (line.startsWith(".")) for (String output : circuit.command(words)) System.out.println(output);
726 else circuit.element(words);
727 } catch (Problem error) {
728 System.out.println("error: line " + number + ": " + error.getMessage());
729 }
730 }
731 }
