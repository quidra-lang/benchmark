void emit(string text)
    println(text)

bool includes_text(string[] items, string value)
    for item := items[]
        if item == value
            return true
    return false

int index_of(string[] items, string value)
    for: i < #items
        if items[i] == value
            return int(i)
    return -1

bool digits(string text)
    if text == ""
        return false
    for: i < #text
        if not "0123456789".contains(text[i])
            return false
    return true

int greatest_divisor(int a, int b)
    while b != 0
        a, b = b, a % b
    return a

class Fraction(numerator := 0, denominator := 1)
    Fraction normalized()
        p := numerator
        q := denominator
        if q < 0
            p = -p
            q = -q
        divisor := greatest_divisor(if p < 0 then -p else p, q)
        return Fraction(p / divisor, q / divisor)

    Fraction plus(Fraction other)
        return Fraction(numerator * other.denominator + other.numerator * denominator, denominator * other.denominator).normalized()

    Fraction minus(Fraction other)
        return Fraction(numerator * other.denominator - other.numerator * denominator, denominator * other.denominator).normalized()

    Fraction times(Fraction other)
        return Fraction(numerator * other.numerator, denominator * other.denominator).normalized()

    Fraction over(Fraction other)
        return Fraction(numerator * other.denominator, denominator * other.numerator).normalized()

    Fraction negative()
        return Fraction(-numerator, denominator)

    bool is_zero()
        return numerator == 0

    bool less_than(Fraction other)
        return numerator * other.denominator < other.numerator * denominator

    string fraction_text()
        return if denominator == 1 then string(numerator) else "{numerator}/{denominator}"

    string value_text()
        magnitude := if numerator < 0 then -numerator else numerator
        scaled := magnitude * 1000000 / denominator
        tail := string(scaled % 1000000)
        string decimal := (if numerator < 0 then "-" else "") + "{scaled / 1000000}." + "0" * (6 - #tail) + tail
        return fraction_text() + " (" + decimal + ")"

Fraction | error parse_value(string original)
    text := original
    multiplier := Fraction(1)
    if text.ends_with("meg")
        multiplier = Fraction(1000000)
        text = text[:<#text - 3]
    elif text.ends_with("k")
        multiplier = Fraction(1000)
        text = text[:<#text - 1]
    elif text.ends_with("m")
        multiplier = Fraction(1, 1000)
        text = text[:<#text - 1]
    elif text.ends_with("u")
        multiplier = Fraction(1, 1000000)
        text = text[:<#text - 1]
    negative := text.starts_with("-")
    if negative
        text = text[1:]
    value := Fraction()
    valid := false
    if digits(text)
        value = Fraction(int(text))
        valid = true
    else
        parts := text.split(".")
        if #parts == 2 and digits(parts[0]) and digits(parts[1])
            scale := 1
            for: i < #parts[1]
                scale *= 10
            value = Fraction(int(parts[0]) * scale + int(parts[1]), scale).normalized()
            valid = true
        else
            parts = text.split("/")
            if #parts == 2 and digits(parts[0]) and digits(parts[1])
                bottom := int(parts[1])
                if bottom != 0
                    value = Fraction(int(parts[0]), bottom).normalized()
                    valid = true
    if not valid
        return error("invalid value " + original)
    value = value.times(multiplier)
    return if negative then value.negative() else value

bool valid_node(string text)
    if text == "0"
        return true
    if #text == 0 or #text > 8 or not "abcdefghijklmnopqrstuvwxyz".contains(text[0])
        return false
    for: i < #text
        if not "abcdefghijklmnopqrstuvwxyz0123456789".contains(text[i])
            return false
    return true

class Element(string name, string n1, string n2, Fraction value, c1 := "", c2 := "", vname := "")
class Link(string a, string b)
class ColumnEntry(string key, Fraction value)
string[] terminals(Element item)
    return if item.c1 == "" then [item.n1, item.n2] else [item.n1, item.n2, item.c1, item.c2]

string[] all_nodes(Element[] items, ground := false)
    string[] result := []
    for item := items[]
        for node := terminals(item)[]
            if (ground or node != "0") and not includes_text(result, node)
                result = result.append(node)
    return result.sorted()

string[] reachable(string start, Link[] links)
    result := [start]
    nat cursor := 0
    while cursor < #result
        current := result[cursor]
        cursor += 1
        for item := links[]
            if item.a == current and not includes_text(result, item.b)
                result = result.append(item.b)
            if item.b == current and not includes_text(result, item.a)
                result = result.append(item.a)
    return result

string[] floating_nodes(Element[] items)
    Link[] links := []
    for item := items[]
        if includes_text(["R", "V", "E", "H"], item.name[0])
            links = links.append(Link(item.n1, item.n2))
    grounded := reachable("0", links)
    return [for node := all_nodes(items)[] if not includes_text(grounded, node) then node]

string[] voltage_loops(Element[] items)
    string[] result := []
    Link[] links := []
    for item := items[]
        if includes_text(["V", "E", "H"], item.name[0])
            if includes_text(reachable(item.n1, links), item.n2)
                result = result.append(item.name)
            links = links.append(Link(item.n1, item.n2))
    return result

void | error check_structure(Element[] items)
    if #all_nodes(items) == 0
        return error("no nodes")
    floating := floating_nodes(items)
    if #floating > 0
        return error("floating node " + floating[0])
    loops := voltage_loops(items)
    if #loops > 0
        return error("voltage source loop at " + loops[0])
    return void

Fraction[][] zero_matrix(nat size)
    return [for i := (0:<nat(size))[] then [for j := (0:<nat(size))[] then Fraction()]]

void add_matrix(Fraction[][] &rows, string[] keys, string row, string column, Fraction value)
    r := index_of(keys, row)
    c := index_of(keys, column)
    if r >= 0 and c >= 0
        rows[r][c] = rows[r][c].plus(value)

void add_column(Fraction[] &values, string[] keys, string key, Fraction value)
    index := index_of(keys, key)
    if index >= 0
        values[index] = values[index].plus(value)

Fraction[] | none gauss_jordan(Fraction[][] matrix, Fraction[] column)
    size := #matrix
    rows := [for i := (0:<nat(size))[] then matrix[i].append(column[i])]
    for: k < size
        pivot := -1
        for: k <= r < size
            if not rows[r][k].is_zero()
                pivot = int(r)
                break
        if pivot < 0
            return none
        previous := rows[k]
        rows[k] = rows[pivot]
        rows[pivot] = previous
        divisor := rows[k][k]
        for: j <= size
            rows[k][j] = rows[k][j].over(divisor)
        for: i < size
            factor := rows[i][k]
            if i != k and not factor.is_zero()
                for: j <= size
                    rows[i][j] = rows[i][j].minus(factor.times(rows[k][j]))
    return [for i := (0:<nat(size))[] then rows[i][size]]

ColumnEntry[] source_entries(Element item, Fraction value)
    return if item.name[0] == "I" then [ColumnEntry(item.n1, value.negative()), ColumnEntry(item.n2, value)] else [ColumnEntry(item.name, value)]

class Solution(string[] keys, Fraction[] values)
    Fraction voltage(string node)
        return if node == "0" then Fraction() else values[index_of(keys, node)]
class Equations(Element[] netlist)
    nodes := all_nodes(netlist)
    Element[] voltage_sources := [for item := netlist[] if includes_text(["V", "E", "H"], item.name[0]) then item]
    string[] unknowns := nodes.concat([for item := voltage_sources[] then item.name])
    matrix := zero_matrix(#unknowns)
    Fraction[] rhs := [for i := (nat(0):<#unknowns)[] then Fraction()]

    void .build()
        one := Fraction(1)
        for item := netlist[]
            kind := item.name[0]
            if kind == "R"
                g := one.over(item.value)
                add_matrix(&matrix, unknowns, item.n1, item.n1, g)
                add_matrix(&matrix, unknowns, item.n2, item.n2, g)
                add_matrix(&matrix, unknowns, item.n1, item.n2, g.negative())
                add_matrix(&matrix, unknowns, item.n2, item.n1, g.negative())
            elif kind == "G"
                add_matrix(&matrix, unknowns, item.n1, item.c1, item.value)
                add_matrix(&matrix, unknowns, item.n2, item.c2, item.value)
                add_matrix(&matrix, unknowns, item.n1, item.c2, item.value.negative())
                add_matrix(&matrix, unknowns, item.n2, item.c1, item.value.negative())
            if kind == "V" or kind == "I"
                for entry := source_entries(item, item.value)[]
                    add_column(&rhs, unknowns, entry.key, entry.value)
    Solution | error solve(Fraction[] values)
        match gauss_jordan(matrix, values)
            Fraction[] result
                return Solution(unknowns, result)
            none
                return error("singular matrix")
class Circuit()
    Element[] elements := []

    int find_element(string name)
        for: i < #elements
            if elements[i].name == name
                return int(i)
        return -1

    nat | error check_element(string name)
        int index := find_element(name)
        if index < 0
            return error("unknown element " + name)
        return nat(index)

    void | error check_node(string node)
        if node != "0" and not includes_text(all_nodes(elements), node)
            return error("unknown node " + node)
        return void
    void | error .add_element(string[] words)
        name := words[0]
        kind := name[0]
        if not includes_text(["R", "V", "I", "E", "G", "F", "H"], kind)
            return error("unknown element type " + name)
        nat count := if kind == "E" or kind == "G" then 6 else (if kind == "F" or kind == "H" then 5 else 4)
        if #words != count
            return error("wrong field count for " + name)
        if find_element(name) >= 0
            return error("duplicate element " + name)
        nat node_count := if kind == "E" or kind == "G" then 4 else 2
        for: 1 <= i <= node_count
            if not valid_node(words[i])
                return error("invalid node " + words[i])
        vname := ""
        value := try parse_value(words[-1])
        if kind == "R" and value.numerator <= 0
            return error("nonpositive resistance " + name)
        c1 := if node_count == 4 then words[3] else ""
        c2 := if node_count == 4 then words[4] else ""
        elements = elements.append(Element(name, words[1], words[2], value, c1, c2, vname))
        return void
    string[] | error .run_command(string[] words)
        commands := [".OP", ".CURRENT", ".VOLTAGE", ".POWER", ".KCL", ".THEVENIN", ".NORTON", ".MAXPOWER", ".TF", ".SENS", ".ZPARAM", ".SUPERPOSITION", ".SWEEP", ".MATRIX", ".REQ", ".REDUCE", ".CHECK", ".ALTER", ".DELETE", ".CLEAR", ".NODES", ".PRINT"]
        nat[] counts := [0, 1, 2, 0, 1, 2, 2, 2, 2, 1, 4, 1, 5, 0, 2, 2, 0, 2, 1, 0, 0, 0]
        command := words[0]
        command_index := index_of(commands, command)
        if command_index < 0
            return error("unknown command " + command)
        if #words != counts[command_index] + 1
            return error("wrong argument count for " + command)
        args := words[1:]
        if command == ".SWEEP"
            nat index := try check_element(args[0])
            first := try parse_value(args[1])
            last := try parse_value(args[2])
            step := try parse_value(args[3])
            try check_node(args[4])
            if step.numerator <= 0
                return error("invalid sweep step")
            if last.less_than(first)
                return error("invalid sweep range")
            span := last.minus(first).over(step)
            count := span.numerator / span.denominator + 1
            if count > 20
                return error("too many sweep points")
            if elements[index].name[0] == "R" and first.numerator <= 0
                return error("nonpositive resistance " + args[0])
            try check_structure(elements)
            string[] output := []
            for: k < count
                value := first.plus(step.times(Fraction(k)))
                trial := elements
                trial[index].value = value
                equations := Equations(trial)
                .equations.build()
                prefix := args[0] + " = " + value.fraction_text() + ": "
                match equations.solve(equations.rhs)
                    Solution solution
                        output = output.append(prefix + "V(" + args[4] + ") = " + solution.voltage(args[4]).value_text())
                    error failure
                        output = output.append(prefix + "singular matrix")
            return output
void read_circuit()
    circuit := Circuit()
    line := ""
    nat number := 0
    while true
        scan(&line)
        number += 1
        if line == ".END"
            break
        if line.starts_with("*")
            continue
        words := line.split(" ")
        if line.starts_with(".")
            match .circuit.run_command(words)
                string[] output
                    for text := output[]
                        emit(text)
                error failure
                    emit("error: line {number}: " + failure.message)
        else
            match .circuit.add_element(words)
                void
                    void
                error failure
                    emit("error: line {number}: " + failure.message)

read_circuit()
