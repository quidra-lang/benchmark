72def parse_number(word):
73 multiplier = 1
74 for suffix, factor in SUFFIXES:
75 if word.endswith(suffix):
76 word = word.removesuffix(suffix)
77 multiplier = factor
78 break
79 sign = 1
80 if word.startswith("-"):
81 word = word[1:]
82 sign = -1
83 if not MANTISSA_PATTERN.fullmatch(word):
84 return None
85 try:
86 mantissa = Fraction(word)
87 except ZeroDivisionError:
88 return None
89 return sign * multiplier * mantissa

113def reachable(start, links):
114 neighbors = defaultdict(list)
115 for u, v in links:
116 neighbors[u].append(v)
117 neighbors[v].append(u)
118 seen = {start}
119 pending = [start]
120 while pending:
121 node = pending.pop()
122 for neighbor in neighbors[node]:
123 if neighbor not in seen:
124 seen.add(neighbor)
125 pending.append(neighbor)
126 return seen

157def gauss_jordan(matrix, column):
158 size = len(matrix)
159 rows = [row + [value] for row, value in zip(matrix, column)]
160 for k in range(size):
161 pivot = next((r for r in range(k, size) if rows[r][k] != 0), None)
162 if pivot is None:
163 return None
164 rows[k], rows[pivot] = rows[pivot], rows[k]
165 divisor = rows[k][k]
166 rows[k] = [value / divisor for value in rows[k]]
167 for i in range(size):
168 factor = rows[i][k]
169 if i != k and factor != 0:
170 rows[i] = [
171 value - factor * pivot_value
172 for value, pivot_value in zip(rows[i], rows[k])
173 ]
174 return [row[size] for row in rows]

183def sensitivity_entries(element, operating):
184 n1, n2 = element.n1, element.n2
185 match element.kind:
186 case "R":
187 t = operating.element_current(element) / element.value
188 return [(n1, t), (n2, -t)]
189 case "I":
190 return [(n1, -1), (n2, 1)]
191 case "G":
192 t = operating.difference(element.c1, element.c2)
193 return [(n1, -t), (n2, t)]
194 case "F":
195 t = operating.current(element.vname)
196 return [(n1, -t), (n2, t)]
197 case "V":
198 return [(element.name, 1)]
199 case "E":
200 return [(element.name, operating.difference(element.c1, element.c2))]
201 case "H":
202 return [(element.name, operating.current(element.vname))]

210 def voltage(self, node):
211 if node == GROUND:
212 return Fraction(0)
213 return self.values[self.index[node]]
214
215 def difference(self, node1, node2):
216 return self.voltage(node1) - self.voltage(node2)

218 def current(self, name):
219 return self.values[self.index[name]]

221 def element_current(self, element):
222 match element.kind:
223 case "R":
224 return self.difference(element.n1, element.n2) / element.value
225 case "I":
226 return element.value
227 case "G":
228 return element.value * self.difference(element.c1, element.c2)
229 case "F":
230 return element.value * self.current(element.vname)
231 case _:
232 return self.current(element.name)

235class Equations:
236 def __init__(self, elements):
237 self.nodes = circuit_nodes(elements)
238 self.voltage_sources = [e for e in elements if e.kind in VOLTAGE_SOURCES]
239 self.unknowns = self.nodes + [e.name for e in self.voltage_sources]
240 self.index = {key: i for i, key in enumerate(self.unknowns)}
241 size = len(self.unknowns)
242 self.matrix = [[Fraction(0)] * size for _ in range(size)]
243 for element in elements:
244 self.stamp(element)
245 self.rhs = self.column(
246 entry
247 for e in elements
248 if e.kind in INDEPENDENT_SOURCES
249 for entry in source_entries(e, e.value)
250 )
251
252 def add(self, row, column, value):
253 if row != GROUND and column != GROUND:
254 self.matrix[self.index[row]][self.index[column]] += value
255
256 def stamp(self, element):
257 n1, n2, w = element.n1, element.n2, element.value
258 match element.kind:
259 case "R":
260 g = 1 / w
261 self.add(n1, n1, g)
262 self.add(n2, n2, g)
263 self.add(n1, n2, -g)
264 self.add(n2, n1, -g)
265 case "G":
266 self.add(n1, element.c1, w)
267 self.add(n2, element.c2, w)
268 self.add(n1, element.c2, -w)
269 self.add(n2, element.c1, -w)
270 case "F":
271 self.add(n1, element.vname, w)
272 self.add(n2, element.vname, -w)
273 case "V" | "E" | "H":
274 s = element.name
275 self.add(n1, s, 1)
276 self.add(s, n1, 1)
277 self.add(n2, s, -1)
278 self.add(s, n2, -1)
279 if element.kind == "E":
280 self.add(s, element.c1, -w)
281 self.add(s, element.c2, w)
282 elif element.kind == "H":
283 self.add(s, element.vname, -w)

285 def column(self, entries):
286 column = [Fraction(0)] * len(self.unknowns)
287 for key, value in entries:
288 if key != GROUND:
289 column[self.index[key]] += value
290 return column

293 values = gauss_jordan(self.matrix, column)

305def reduction_step(resistors, a, b, names):
306 for resistor in resistors:
307 if resistor.end1 == resistor.end2:
308 resistors.remove(resistor)
309 return f"remove {resistor.name}"
310 degree = Counter(end for r in resistors for end in r.ends)
311 internal = sorted(node for node in degree if node not in (a, b))
312 for node in internal:
313 if degree[node] == 1:
314 resistor = next(r for r in resistors if node in r.ends)
315 resistors.remove(resistor)
316 return f"remove {resistor.name}"
317 for i, x in enumerate(resistors):
318 partners = [r for r in resistors[i + 1 :] if set(r.ends) == set(x.ends)]
319 if partners:
320 y = partners[0]
321 resistance = x.resistance * y.resistance / (x.resistance + y.resistance)
322 merged = Resistor(next(names), x.end1, x.end2, resistance)
323 resistors.remove(x)
324 resistors.remove(y)
325 resistors.append(merged)
326 return f"{merged.name} = {x.name} || {y.name} = {format_value(resistance)}"
327 for node in internal:
328 if degree[node] == 2:
329 x, y = [r for r in resistors if node in r.ends]
330 resistance = x.resistance + y.resistance
331 u, v = x.other_end(node), y.other_end(node)
332 merged = Resistor(next(names), u, v, resistance)
333 resistors.remove(x)
334 resistors.remove(y)
335 resistors.append(merged)
336 return f"{merged.name} = {x.name} + {y.name} = {format_value(resistance)}"
337 return None

365 def add_element(self, words):
366 name = words[0]
367 kind = name[0]
368 if kind not in FIELD_COUNTS:
369 raise CircuitError(f"unknown element type {name}")
370 if len(words) != FIELD_COUNTS[kind]:
371 raise CircuitError(f"wrong field count for {name}")
372 if self.find(name) is not None:
373 raise CircuitError(f"duplicate element {name}")

375 for node in nodes:

383 raise CircuitError(f"unknown voltage source {vname}")

386 raise CircuitError(f"nonpositive resistance {name}")
387 n1, n2, *controls = nodes
388 self.elements.append(Element(name, n1, n2, value, *controls, vname=vname))

540 def sweep(self, name, first_word, last_word, step_word, a):
541 element = self.check_element(name)
542 first = check_number(first_word)
543 last = check_number(last_word)
544 step = check_number(step_word)
545 self.check_node(a)
546 if step <= 0:
547 raise CircuitError("invalid sweep step")
548 if first > last:
549 raise CircuitError("invalid sweep range")
550 point_count = (last - first) // step + 1
551 if point_count > MAX_SWEEP_POINTS:
552 raise CircuitError("too many sweep points")
553 if element.kind == "R" and first <= 0:
554 raise CircuitError(f"nonpositive resistance {name}")
555 check_structure(self.elements)
556 lines = []
557 for k in range(point_count):
558 x = first + k * step
559 trial = [replace(e, value=x) if e is element else e for e in self.elements]
560 equations = Equations(trial)
561 solution = equations.solve(equations.rhs)
562 if solution is None:
563 lines.append(f"{name} = {x}: singular matrix")
564 else:
565 voltage = format_value(solution.voltage(a))
566 lines.append(f"{name} = {x}: V({a}) = {voltage}")
567 return lines

581 def req(self, a, b):
582 self.check_node(a)
583 self.check_node(b)
584 wires = [(e.n1, e.n2) for e in self.elements if e.kind == "V"]
585 group = {}
586 for node in [GROUND, *circuit_nodes(self.elements)]:
587 if node not in group:
588 for member in reachable(node, wires):
589 group[member] = node
590 if group[a] == group[b]:
591 return ["Req = 0 (0.000000)"]
592 edges = [
593 (group[e.n1], group[e.n2], 1 / e.value)
594 for e in self.elements
595 if e.kind == "R" and group[e.n1] != group[e.n2]
596 ]
597 reached = reachable(group[a], [(p, q) for p, q, _ in edges])
598 if group[b] not in reached:
599 return ["Req = INFINITE"]
600 index = {g: i for i, g in enumerate(sorted(reached - {group[b]}))}
601 size = len(index)
602 matrix = [[Fraction(0)] * size for _ in range(size)]
603 for p, q, conductance in edges:
604 if p not in reached:
605 continue
606 stamps = (
607 (p, p, conductance),
608 (q, q, conductance),
609 (p, q, -conductance),
610 (q, p, -conductance),
611 )
612 for row, column, value in stamps:
613 if row in index and column in index:
614 matrix[index[row]][index[column]] += value
615 injected = [Fraction(0)] * size
616 injected[index[group[a]]] = Fraction(1)
617 solution = gauss_jordan(matrix, injected)
618 return [f"Req = {format_value(solution[index[group[a]]])}"]

620 def reduce(self, a, b):
621 self.check_node(a)
622 self.check_node(b)
623 if a == b:
624 return ["Req = 0 (0.000000)"]
625 resistors = [
626 Resistor(e.name, e.n1, e.n2, e.value)
627 for e in self.elements
628 if e.kind == "R"
629 ]
630 connected = reachable(a, [r.ends for r in resistors])
631 if b not in connected:
632 return ["Req = INFINITE"]
633 resistors = [r for r in resistors if r.end1 in connected]
634 names = (f"#{k}" for k in count(1))
635 lines = []
636 while (line := reduction_step(resistors, a, b, names)) is not None:
637 lines.append(line)
638 if len(resistors) == 1:
639 lines.append(f"Req = {format_value(resistors[0].resistance)}")
640 else:
641 lines.append(f"not series-parallel: {len(resistors)} resistors remain")
642 return lines

644 def check(self):
645 nodes = circuit_nodes(self.elements)
646 if not nodes:
647 return ["no nodes"]
648 degree = Counter(end for e in self.elements for end in (e.n1, e.n2))
649 lines = [f"dangling node {node}" for node in nodes if degree[node] < 2]
650 lines += [f"floating node {node}" for node in floating_nodes(self.elements)]
651 lines += [
652 f"voltage source loop at {name}" for name in voltage_loops(self.elements)
653 ]
654 if not lines:
655 equations = Equations(self.elements)
656 if equations.solve(equations.rhs) is None:
657 lines.append("singular matrix")
658 return lines or ["ok"]

682 def list_nodes(self):
683 if not self.elements:
684 return ["no nodes"]
