25function gcd(a: bigint, b: bigint): bigint {
26 while (b !== 0n) [a, b] = [b, a % b];
27 return a;
28}
29class Rational {
30 readonly n: bigint;
31 readonly d: bigint;
32 constructor(n: bigint, d = 1n) {
33 if (d < 0n) {
34 n = -n;
35 d = -d;
36 }
37 const g = gcd(n < 0n ? -n : n, d);
38 this.n = n / g;
39 this.d = d / g;
40 }
41 add(b: Rational): Rational {
42 return new Rational(this.n * b.d + b.n * this.d, this.d * b.d);
43 }
44 sub(b: Rational): Rational {
45 return new Rational(this.n * b.d - b.n * this.d, this.d * b.d);
46 }
47 mul(b: Rational): Rational {
48 return new Rational(this.n * b.n, this.d * b.d);
49 }
50 div(b: Rational): Rational {
51 return new Rational(this.n * b.d, this.d * b.n);
52 }
53 neg(): Rational {
54 return new Rational(-this.n, this.d);
55 }
56 compare(b: Rational): number {
57 const difference = this.n * b.d - b.n * this.d;
58 return difference < 0n ? -1 : difference > 0n ? 1 : 0;
59 }
60 get zero(): boolean {
61 return this.n === 0n;
62 }
63 get positive(): boolean {
64 return this.n > 0n;
65 }
66 toString(): string {
67 return this.d === 1n ? String(this.n) : `${this.n}/${this.d}`;
68 }
69 value(): string {
70 const scaled = ((this.n < 0n ? -this.n : this.n) * 1000000n) / this.d;
71 return `${this} (${this.n < 0n ? "-" : ""}${scaled / 1000000n}.${String(scaled % 1000000n).padStart(6, "0")})`;
72 }
73}
74const ZERO = new Rational(0n),
75 ONE = new Rational(1n);
76function readNumber(word: string): Rational | null {
77 let text = word,
78 multiplier = ONE;
79 const multipliers: Array<[string, Rational]> = [
80 ["meg", new Rational(1000000n)],
81 ["k", new Rational(1000n)],
82 ["m", new Rational(1n, 1000n)],
83 ["u", new Rational(1n, 1000000n)],
84 ];
85 for (const [suffix, value] of multipliers) {
86 if (text.endsWith(suffix)) {
87 text = text.slice(0, -suffix.length);
88 multiplier = value;
89 break;
90 }
91 }
92 const negative = text.startsWith("-");
93 if (negative) text = text.slice(1);
94 if (!/^[0-9]+(?:\.[0-9]+|\/[0-9]+)?$/.test(text)) return null;
95 let numerator: bigint, denominator: bigint;
96 if (text.includes("/")) {
97 const parts = text.split("/");
98 numerator = BigInt(parts[0]);
99 denominator = BigInt(parts[1]);
100 if (denominator === 0n) return null;
101 } else if (text.includes(".")) {
102 const parts = text.split(".");
103 numerator = BigInt(parts.join(""));
104 denominator = 10n ** BigInt(parts[1].length);
105 } else {
106 numerator = BigInt(text);
107 denominator = 1n;
108 }
109 return new Rational(negative ? -numerator : numerator, denominator).mul(multiplier);
110}

131function solve(a: Rational[][], b: Rational[]): Rational[] | null {
132 const n = a.length,
133 rows = a.map((row, i) => [...row, b[i]]);
134 for (let k = 0; k < n; k++) {
135 let r = k;
136 while (r < n && rows[r][k].zero) r++;
137 if (r === n) return null;
138 [rows[k], rows[r]] = [rows[r], rows[k]];
139 const pivot = rows[k][k];
140 for (let j = 0; j <= n; j++) rows[k][j] = rows[k][j].div(pivot);
141 for (let i = 0; i < n; i++)
142 if (i !== k && !rows[i][k].zero) {
143 const factor = rows[i][k];
144 for (let j = 0; j <= n; j++) rows[i][j] = rows[i][j].sub(factor.mul(rows[k][j]));
145 }
146 }
147 return rows.map((row) => row[n]);
148}
149function reachable(start: string, edges: string[][]): Set<string> {
150 const seen = new Set([start]),
151 pending = [start];
152 while (pending.length > 0) {
153 const node = pending.pop()!;
154 for (const [a, b] of edges) {
155 const next = a === node ? b : b === node ? a : null;
156 if (next !== null && !seen.has(next)) {
157 seen.add(next);
158 pending.push(next);
159 }
160 }
161 }
162 return seen;
163}
164class Groups {
165 readonly parent: Map<string, string>;
166 constructor(nodes: string[]) {
167 this.parent = new Map<string, string>(nodes.map((node) => [node, node]));
168 }
169 find(node: string): string {
170 const parent = this.parent.get(node)!;
171 if (parent === node) return node;
172 const root = this.find(parent);
173 this.parent.set(node, root);
174 return root;
175 }
176 join(a: string, b: string): void {
177 a = this.find(a);
178 b = this.find(b);
179 if (a !== b) this.parent.set(a < b ? b : a, a < b ? a : b);
180 }
181}

190 circuitNodes(): string[] {
191 return this.nodes().filter((node) => node !== "0");
192 }
193 elementCheck(name: string): CircuitElement {
194 const element = this.elements.find((element) => element.name === name);
195 if (element === undefined) fail("unknown element " + name);
196 return element;
197 }
198 nodeCheck(name: string): void {
199 if (name !== "0" && !this.circuitNodes().includes(name)) fail("unknown node " + name);
200 }
201 add(words: string[]): void {
202 const name = words[0],
203 type = name[0];
204 if (!"RVIEGFH".includes(type)) fail("unknown element type " + name);
205 const expected = "EG".includes(type) ? 6 : "FH".includes(type) ? 5 : 4;
206 if (words.length !== expected) fail("wrong field count for " + name);
207 if (this.elements.some((element) => element.name === name))
208 fail("duplicate element " + name);
209 const terminals = words.slice(1, "EG".includes(type) ? 5 : 3);
210 for (const node of terminals) if (!validNode(node)) fail("invalid node " + node);
211 const reference = "FH".includes(type) ? words[3] : null;
212 if (
213 reference !== null &&
214 !this.elements.some((element) => element.name === reference && element.name[0] === "V")
215 )
216 fail("unknown voltage source " + reference);
217 const value = numberCheck(words[words.length - 1]);
218 if (type === "R" && !value.positive) fail("nonpositive resistance " + name);
219 this.elements.push({ name, terminals, reference, value });
220 }
221 floating(): string[] {
222 const edges = this.elements
223 .filter((element) => "RVEH".includes(element.name[0]))
224 .map((element) => [element.terminals[0], element.terminals[1]]);
225 const seen = reachable("0", edges);
226 return this.circuitNodes().filter((node) => !seen.has(node));
227 }
228 loops(): string[] {
229 const groups = new Groups(this.nodes(true)),
230 result: string[] = [];
231 for (const element of this.elements.filter(voltageType)) {
232 const [a, b] = element.terminals;
233 if (groups.find(a) === groups.find(b)) result.push(element.name);
234 groups.join(a, b);
235 }
236 return result;
237 }
238 structure(): void {
239 if (this.circuitNodes().length === 0) fail("no nodes");
240 const floating = this.floating();
241 if (floating.length > 0) fail("floating node " + floating[0]);
242 const loops = this.loops();
243 if (loops.length > 0) fail("voltage source loop at " + loops[0]);
244 }
245 equations(overrides = new Map<string, Rational>()): Equations {
246 const nodes = this.circuitNodes(),
247 sources = this.elements.filter(voltageType),
248 count = nodes.length + sources.length;
249 const nodeIndex = new Map<string, number>(nodes.map((node, i) => [node, i]));
250 const sourceIndex = new Map<string, number>(
251 sources.map((source, i) => [source.name, nodes.length + i]),
252 );
253 const a = matrix(count),
254 b = zeros(count);
255 const node = (name: string): number => (name === "0" ? -1 : nodeIndex.get(name)!);
256 const source = (name: string | null): number => sourceIndex.get(name!)!;
257 const stamp = (row: number, column: number, value: Rational): void => {
258 if (row >= 0 && column >= 0) a[row][column] = a[row][column].add(value);
259 };
260 const rhs = (row: number, value: Rational): void => {
261 if (row >= 0) b[row] = b[row].add(value);
262 };
263 for (const element of this.elements) {
264 const [n1, n2, c1, c2] = element.terminals,
265 type = element.name[0];
266 const x = node(n1),
267 y = node(n2),
268 w = overrides.get(element.name) ?? element.value;
269 if (type === "R") {
270 const g = ONE.div(w);
271 stamp(x, x, g);
272 stamp(y, y, g);
273 stamp(x, y, g.neg());
274 stamp(y, x, g.neg());
275 } else if (type === "I") {
276 rhs(x, w.neg());
277 rhs(y, w);
278 } else if (type === "G") {
279 stamp(x, node(c1), w);
280 stamp(y, node(c2), w);
281 stamp(x, node(c2), w.neg());
282 stamp(y, node(c1), w.neg());
283 } else if (type === "F") {
284 stamp(x, source(element.reference), w);
285 stamp(y, source(element.reference), w.neg());
286 } else {
287 const s = source(element.name);
288 stamp(x, s, ONE);
289 stamp(s, x, ONE);
290 stamp(y, s, ONE.neg());
291 stamp(s, y, ONE.neg());
292 if (type === "V") rhs(s, w);
293 else if (type === "E") {
294 stamp(s, node(c1), w.neg());
295 stamp(s, node(c2), w);
296 } else stamp(s, source(element.reference), w.neg());
297 }
298 }
299 return { nodes, sources, count, nodeIndex, sourceIndex, a, b };
300 }
301 potential(equations: Equations, solution: Rational[], name: string): Rational {
302 return name === "0" ? ZERO : solution[equations.nodeIndex.get(name)!];
303 }
304 sourceCurrent(equations: Equations, solution: Rational[], name: string | null): Rational {
305 return solution[equations.sourceIndex.get(name!)!];
306 }
307 current(equations: Equations, solution: Rational[], element: CircuitElement): Rational {
308 const [n1, n2, c1, c2] = element.terminals,
309 type = element.name[0];
310 if (type === "R")
311 return this.potential(equations, solution, n1)
312 .sub(this.potential(equations, solution, n2))
313 .div(element.value);
314 if (type === "I") return element.value;
315 if (type === "G")
316 return element.value.mul(
317 this.potential(equations, solution, c1).sub(
318 this.potential(equations, solution, c2),
319 ),
320 );
321 if (type === "F")
322 return element.value.mul(this.sourceCurrent(equations, solution, element.reference));
323 return this.sourceCurrent(equations, solution, element.name);
324 }
325 injection(equations: Equations, a: string, b: string): Rational[] {
326 const column = zeros(equations.count);
327 if (a !== "0") {
328 const i = equations.nodeIndex.get(a)!;
329 column[i] = column[i].add(ONE);
330 }
331 if (b !== "0") {
332 const i = equations.nodeIndex.get(b)!;
333 column[i] = column[i].sub(ONE);
334 }
335 return solve(equations.a, column)!;
336 }
337 alone(equations: Equations, element: CircuitElement, unit: boolean): Rational[] {
338 const column = zeros(equations.count),
339 value = unit ? ONE : element.value;
340 if (element.name[0] === "V") column[equations.sourceIndex.get(element.name)!] = value;
341 else {
342 const [a, b] = element.terminals;
343 if (a !== "0") {
344 const i = equations.nodeIndex.get(a)!;
345 column[i] = column[i].sub(value);
346 }
347 if (b !== "0") {
348 const i = equations.nodeIndex.get(b)!;
349 column[i] = column[i].add(value);
350 }
351 }
352 return solve(equations.a, column)!;
353 }
354 resistance(a: string, b: string): string[] {
355 const groups = new Groups(this.nodes(true));
356 for (const element of this.elements)
357 if (element.name[0] === "V") groups.join(element.terminals[0], element.terminals[1]);
358 a = groups.find(a);
359 b = groups.find(b);
360 if (a === b) return ["Req = " + ZERO.value()];
361 const edges: ResistanceEdge[] = [];
362 for (const element of this.elements)
363 if (element.name[0] === "R") {
364 const x = groups.find(element.terminals[0]),
365 y = groups.find(element.terminals[1]);
366 if (x !== y) edges.push({ x, y, value: ONE.div(element.value) });
367 }
368 const seen = reachable(
369 a,
370 edges.map((edge) => [edge.x, edge.y]),
371 );
372 if (!seen.has(b)) return ["Req = INFINITE"];
373 const nodes = [...seen].filter((node) => node !== b).sort(),
374 indices = new Map(nodes.map((node, i) => [node, i]));
375 const g = matrix(nodes.length),
376 column = zeros(nodes.length);
377 const stamp = (row: string, col: string, value: Rational): void => {
378 if (row !== b && col !== b) {
379 const i = indices.get(row)!,
380 j = indices.get(col)!;
381 g[i][j] = g[i][j].add(value);
382 }
383 };
384 for (const edge of edges)
385 if (seen.has(edge.x)) {
386 stamp(edge.x, edge.x, edge.value);
387 stamp(edge.y, edge.y, edge.value);
388 stamp(edge.x, edge.y, edge.value.neg());
389 stamp(edge.y, edge.x, edge.value.neg());
390 }
391 column[indices.get(a)!] = ONE;
392 return ["Req = " + solve(g, column)![indices.get(a)!].value()];
393 }
394 reduce(a: string, b: string): string[] {
395 if (a === b) return ["Req = " + ZERO.value()];
396 let list: Resistor[] = this.elements
397 .filter((element) => element.name[0] === "R")
398 .map((element) => ({
399 name: element.name,
400 a: element.terminals[0],
401 b: element.terminals[1],
402 value: element.value,
403 }));
404 const connected = reachable(
405 a,
406 list.map((element) => [element.a, element.b]),
407 );
408 if (!connected.has(b)) return ["Req = INFINITE"];
409 list = list.filter((element) => connected.has(element.a));
410 const output: string[] = [];
411 let serial = 0;
412 while (true) {
413 let remove = list.findIndex((element) => element.a === element.b);
414 if (remove >= 0) {
415 output.push("remove " + list[remove].name);
416 list.splice(remove, 1);
417 continue;
418 }
419 const degree = new Map<string, number>();
420 for (const element of list)
421 for (const node of [element.a, element.b])
422 degree.set(node, (degree.get(node) ?? 0) + 1);
423 const internal = [...degree.keys()].filter((node) => node !== a && node !== b).sort();
424 const dangling = internal.find((node) => degree.get(node) === 1);
425 if (dangling !== undefined) {
426 remove = list.findIndex(
427 (element) => element.a === dangling || element.b === dangling,
428 );
429 output.push("remove " + list[remove].name);
430 list.splice(remove, 1);
431 continue;
432 }
433 let first = -1,
434 second = -1,
435 parallel = false;
436 search: for (let i = 0; i < list.length; i++)
437 for (let j = i + 1; j < list.length; j++) {
438 const x = list[i],
439 y = list[j];
440 if ((x.a === y.a && x.b === y.b) || (x.a === y.b && x.b === y.a)) {
441 first = i;
442 second = j;
443 parallel = true;
444 break search;
445 }
446 }
447 let center: string | null = null;
448 if (!parallel) {
449 center = internal.find((node) => degree.get(node) === 2) ?? null;
450 if (center === null) break;
451 const indices = list
452 .map((element, i) => (element.a === center || element.b === center ? i : -1))
453 .filter((i) => i >= 0);
454 [first, second] = indices;
455 }
456 const x = list[first],
457 y = list[second],
458 name = "#" + ++serial;
459 const value = parallel
460 ? x.value.mul(y.value).div(x.value.add(y.value))
461 : x.value.add(y.value);
462 const left = parallel ? x.a : x.a === center ? x.b : x.a;
463 const right = parallel ? x.b : y.a === center ? y.b : y.a;
464 output.push(
465 `${name} = ${x.name} ${parallel ? "||" : "+"} ${y.name} = ${value.value()}`,
466 );
467 list.splice(second, 1);
468 list.splice(first, 1);
469 list.push({ name, a: left, b: right, value });
470 }
471 output.push(
472 list.length === 1
473 ? "Req = " + list[0].value.value()
474 : `not series-parallel: ${list.length} resistors remain`,
475 );
476 return output;
477 }
478 check(): string[] {
479 const nodes = this.circuitNodes();
480 if (nodes.length === 0) return ["no nodes"];
481 const output: string[] = [];
482 for (const node of nodes) {
483 let degree = 0;
484 for (const element of this.elements) {
485 if (element.terminals[0] === node) degree++;
486 if (element.terminals[1] === node) degree++;
487 }
488 if (degree < 2) output.push("dangling node " + node);
489 }
490 for (const node of this.floating()) output.push("floating node " + node);
491 for (const name of this.loops()) output.push("voltage source loop at " + name);
492 if (output.length === 0) {
493 const eq = this.equations();
494 if (solve(eq.a, eq.b) === null) output.push("singular matrix");
495 }
496 return output.length === 0 ? ["ok"] : output;
497 }
498 command(words: string[]): string[] {
499 const command = words[0],
500 args = words.slice(1);
501 const counts = new Map<string, number>([
502 [".OP", 0],
503 [".CURRENT", 1],
504 [".VOLTAGE", 2],
505 [".POWER", 0],
506 [".KCL", 1],
507 [".THEVENIN", 2],
508 [".NORTON", 2],
509 [".MAXPOWER", 2],
510 [".TF", 2],
511 [".SENS", 1],
512 [".ZPARAM", 4],
513 [".SUPERPOSITION", 1],
514 [".SWEEP", 5],
515 [".MATRIX", 0],
516 [".REQ", 2],
517 [".REDUCE", 2],
518 [".CHECK", 0],
519 [".ALTER", 2],
520 [".DELETE", 1],
521 [".CLEAR", 0],
522 [".NODES", 0],
523 [".PRINT", 0],
524 ]);

531 }

534 user = this.elements.find((other) => other.reference === element.name);
535 if (user !== undefined) fail(element.name + " is referenced by " + user.name);
536 this.elements.splice(this.elements.indexOf(element), 1);

538 }
539 if (command === ".ALTER") {
540 const element = this.elementCheck(args[0]),
541 value = numberCheck(args[1]);
542 if (element.name[0] === "R" && !value.positive)
543 fail("nonpositive resistance " + element.name);
544 const output = `altered ${element.name}: ${element.value} -> ${value}`;
545 element.value = value;
546 return [output];
547 }
548 if (command === ".NODES") {
549 const nodes = this.nodes();
550 return nodes.length === 0
551 ? ["no nodes"]
552 : nodes.map(
553 (node) =>
554 node +
555 ": " +
556 this.elements
557 .filter((element) => element.terminals.includes(node))
558 .map((element) => element.name)

560 );
561 }
562 if (command === ".PRINT")

568 ...element.terminals,
569 ...(element.reference === null ? [] : [element.reference]),

571 ].join(" "),
572 );
573 if (command === ".CHECK") return this.check();
574 if (command === ".SWEEP") {
575 const element = this.elementCheck(args[0]);
576 const from = numberCheck(args[1]),
577 to = numberCheck(args[2]),
578 step = numberCheck(args[3]);
579 this.nodeCheck(args[4]);
580 if (!step.positive) fail("invalid sweep step");
581 if (from.compare(to) > 0) fail("invalid sweep range");
582 const span = to.sub(from).div(step),
583 count = span.n / span.d + 1n;
584 if (count > 20n) fail("too many sweep points");
585 if (element.name[0] === "R" && !from.positive)
586 fail("nonpositive resistance " + element.name);

588 const output: string[] = [];
589 for (let k = 0n; k < count; k++) {
590 const value = from.add(step.mul(new Rational(k))),
591 eq = this.equations(new Map([[element.name, value]])),
592 solution = solve(eq.a, eq.b);
593 output.push(
594 `${element.name} = ${value}: ${solution === null ? "singular matrix" : `V(${args[4]}) = ${this.potential(eq, solution, args[4]).value()}`}`,
595 );
596 }
597 return output;
598 }
599 if (command === ".REQ" || command === ".REDUCE") {
600 this.nodeCheck(args[0]);
601 this.nodeCheck(args[1]);
602 return command === ".REQ"
603 ? this.resistance(args[0], args[1])
604 : this.reduce(args[0], args[1]);
605 }
606 let selected: CircuitElement | null = null;
607 if (command === ".CURRENT") selected = this.elementCheck(args[0]);
608 else if (command === ".TF") {
609 this.nodeCheck(args[0]);
610 selected = this.elementCheck(args[1]);
611 if (!independent(selected)) fail("not an independent source " + selected!.name);
612 } else if (
613 [
614 ".VOLTAGE",
615 ".KCL",
616 ".THEVENIN",
617 ".NORTON",
618 ".MAXPOWER",
619 ".SENS",
620 ".ZPARAM",
621 ".SUPERPOSITION",
622 ].includes(command)
623 ) {
624 for (const node of args) this.nodeCheck(node);
625 }
626 this.structure();
627 const eq = this.equations();
628 if (command === ".MATRIX") {
629 const labels = [...eq.nodes, ...eq.sources.map((element) => element.name)];
630 return [
631 "unknowns: " +
632 [
633 ...eq.nodes.map((node) => `V(${node})`),
634 ...eq.sources.map((element) => `I(${element.name})`),
635 ].join(" "),
636 ...labels.map((label, i) => `${label}: ${eq.a[i].join(" ")} | ${eq.b[i]}`),
637 ];
638 }
639 const solution = solve(eq.a, eq.b);
640 if (solution === null) fail("singular matrix");
641 const potential = (name: string, values: Rational[] = solution): Rational =>
642 this.potential(eq, values, name);
643 const output: string[] = [];
644 if (command === ".OP") {
645 for (const node of eq.nodes) output.push(`V(${node}) = ${potential(node).value()}`);
646 for (const element of eq.sources)
647 output.push(
648 `I(${element.name}) = ${this.sourceCurrent(eq, solution, element.name).value()}`,
649 );
650 } else if (command === ".CURRENT")
651 output.push(`I(${selected!.name}) = ${this.current(eq, solution, selected!).value()}`);
652 else if (command === ".VOLTAGE")
653 output.push(
654 `V(${args[0]},${args[1]}) = ${potential(args[0]).sub(potential(args[1])).value()}`,
655 );
656 else if (command === ".POWER") {
657 let absorbed = ZERO,
658 delivered = ZERO,
659 total = ZERO;
660 for (const element of this.elements) {
661 const power = potential(element.terminals[0])
662 .sub(potential(element.terminals[1]))
663 .mul(this.current(eq, solution, element));
664 output.push(`P(${element.name}) = ${power.value()}`);
665 if (power.positive) absorbed = absorbed.add(power);
666 else if (!power.zero) delivered = delivered.sub(power);
667 total = total.add(power);
668 }
669 output.push(
670 "absorbed = " + absorbed.value(),
671 "delivered = " + delivered.value(),
672 "total = " + total.value(),
673 );
674 } else if (command === ".KCL") {
675 let sum = ZERO;
676 for (const element of this.elements)
677 if (element.terminals[0] === args[0] || element.terminals[1] === args[0]) {
678 const current =
679 element.terminals[0] === element.terminals[1]
680 ? ZERO
681 : element.terminals[0] === args[0]
682 ? this.current(eq, solution, element)
683 : this.current(eq, solution, element).neg();
684 output.push(`I(${element.name}) leaving ${args[0]} = ${current.value()}`);
685 sum = sum.add(current);
686 }
687 output.push("sum = " + sum.value());
688 } else if ([".THEVENIN", ".NORTON", ".MAXPOWER"].includes(command)) {
689 const voltage = potential(args[0]).sub(potential(args[1])),
690 z = this.injection(eq, args[0], args[1]);
691 const resistance = potential(args[0], z).sub(potential(args[1], z));
692 if (command === ".THEVENIN")
693 output.push("Vth = " + voltage.value(), "Rth = " + resistance.value());
694 else if (command === ".NORTON") {
695 if (resistance.zero) fail("zero thevenin resistance");
696 output.push(
697 "In = " + voltage.div(resistance).value(),
698 "Rn = " + resistance.value(),
699 );
700 } else {
701 if (!resistance.positive) fail("nonpositive thevenin resistance");
702 output.push(
703 "RL = " + resistance.value(),
704 "Pmax = " + voltage.mul(voltage).div(new Rational(4n).mul(resistance)).value(),
705 );
706 }
707 } else if (command === ".TF") {
708 const z = this.alone(eq, selected!, true);
709 output.push("gain = " + potential(args[0], z).value());
710 if (selected!.name[0] === "V") {
711 const current = this.sourceCurrent(eq, z, selected!.name);
712 output.push(
713 "input resistance = " +
714 (current.zero ? "INFINITE" : ONE.neg().div(current).value()),
715 );
716 } else
717 output.push(
718 "input resistance = " +
719 potential(selected!.terminals[1], z)
720 .sub(potential(selected!.terminals[0], z))
721 .value(),
722 );
723 output.push(
724 "output resistance = " +
725 potential(args[0], this.injection(eq, args[0], "0")).value(),
726 );
727 } else if (command === ".SENS") {
728 for (const element of this.elements) {
729 const column = zeros(eq.count),
730 [a, b, c, d] = element.terminals;
731 const add = (node: string, value: Rational): void => {
732 if (node !== "0") {
733 const i = eq.nodeIndex.get(node)!;
734 column[i] = column[i].add(value);
735 }
736 };
737 const type = element.name[0];
738 if (type === "R") {
739 const t = this.current(eq, solution, element).div(element.value);
740 add(a, t);
741 add(b, t.neg());
742 } else if (type === "I") {
743 add(a, ONE.neg());
744 add(b, ONE);
745 } else if (type === "G" || type === "F") {
746 const t =
747 type === "G"
748 ? potential(c).sub(potential(d))
749 : this.sourceCurrent(eq, solution, element.reference);
750 add(a, t.neg());
751 add(b, t);
752 } else
753 column[eq.sourceIndex.get(element.name)!] =
754 type === "V"
755 ? ONE
756 : type === "E"
757 ? potential(c).sub(potential(d))
758 : this.sourceCurrent(eq, solution, element.reference);
759 output.push(
760 `dV(${args[0]})/d${element.name} = ${potential(args[0], solve(eq.a, column)!).value()}`,
761 );
762 }
763 } else if (command === ".ZPARAM") {
764 const first = this.injection(eq, args[0], args[1]),
765 second = this.injection(eq, args[2], args[3]);
766 output.push(
767 "z11 = " + potential(args[0], first).sub(potential(args[1], first)).value(),
768 "z12 = " + potential(args[0], second).sub(potential(args[1], second)).value(),
769 "z21 = " + potential(args[2], first).sub(potential(args[3], first)).value(),
770 "z22 = " + potential(args[2], second).sub(potential(args[3], second)).value(),
771 );
772 } else if (command === ".SUPERPOSITION") {
773 let total = ZERO;
774 for (const element of this.elements.filter(independent)) {
775 const value = potential(args[0], this.alone(eq, element, false));
776 output.push(`${element.name}: V(${args[0]}) = ${value.value()}`);
777 total = total.add(value);
778 }
779 output.push(`total: V(${args[0]}) = ${total.value()}`);
780 }
781 return output;
782 }
783}

786 output: string[] = [];
787let lineNumber = 0;
788for (const line of fs.readFileSync(0, "utf8").split(/\r?\n/)) {
789 lineNumber++;
790 if (line === ".END") break;
791 if (line.startsWith("*")) continue;
792 const words = line.split(" ");
793 try {
794 if (line.startsWith(".")) output.push(...circuit.command(words));
795 else circuit.add(words);
796 } catch (error) {
797 if (!(error instanceof Problem)) throw error;
798 output.push(`error: line ${lineNumber}: ${error.message}`);
799 }
800}
801if (output.length > 0) process.stdout.write(output.join("\n") + "\n");
