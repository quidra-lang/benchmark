10class Rational implements Comparable<Rational> {
11 final BigInt n, d;
12 Rational._(this.n, this.d);
13 factory Rational(BigInt n, [BigInt? denominator]) {
14 var d = denominator ?? BigInt.one;
15 if (d.isNegative) {
16 n = -n;
17 d = -d;
18 }
19 final divisor = n.abs().gcd(d);
20 return Rational._(n ~/ divisor, d ~/ divisor);
21 }
22 Rational operator +(Rational other) =>
23 Rational(n * other.d + other.n * d, d * other.d);
24 Rational operator -(Rational other) =>
25 Rational(n * other.d - other.n * d, d * other.d);
26 Rational operator *(Rational other) => Rational(n * other.n, d * other.d);
27 Rational operator /(Rational other) => Rational(n * other.d, d * other.n);
28 Rational operator -() => Rational(-n, d);
29 @override
30 int compareTo(Rational other) => (n * other.d).compareTo(other.n * d);
31 bool get zero => n == BigInt.zero;
32 bool get positive => n > BigInt.zero;
33 @override
34 String toString() => d == BigInt.one ? '$n' : '$n/$d';
35 String value() {
36 final million = BigInt.from(1000000), scaled = n.abs() * million ~/ d;
37 return '$this (${n.isNegative ? '-' : ''}${scaled ~/ million}.${(scaled % million).toString().padLeft(6, '0')})';
38 }
39}

44Rational? readNumber(String word) {
45 var text = word, multiplier = one;
46 final multipliers = {
47 'meg': rational(1000000),
48 'k': rational(1000),
49 'm': Rational(BigInt.one, BigInt.from(1000)),
50 'u': Rational(BigInt.one, BigInt.from(1000000)),
51 };
52 for (final entry in multipliers.entries) {
53 if (text.endsWith(entry.key)) {
54 text = text.substring(0, text.length - entry.key.length);
55 multiplier = entry.value;
56 break;
57 }
58 }
59 final negative = text.startsWith('-');
60 if (negative) text = text.substring(1);
61 if (!RegExp(r'^[0-9]+(?:\.[0-9]+|/[0-9]+)?$').hasMatch(text)) return null;
62 BigInt numerator, denominator;
63 if (text.contains('/')) {
64 final parts = text.split('/');
65 numerator = BigInt.parse(parts[0]);
66 denominator = BigInt.parse(parts[1]);
67 if (denominator == BigInt.zero) return null;
68 } else if (text.contains('.')) {
69 final parts = text.split('.');
70 numerator = BigInt.parse(parts.join(''));
71 denominator = BigInt.from(10).pow(parts[1].length);
72 } else {
73 numerator = BigInt.parse(text);
74 denominator = BigInt.one;
75 }
76 return Rational(negative ? -numerator : numerator, denominator) * multiplier;
77}

82 return value;

102List<Rational>? solve(List<List<Rational>> a, List<Rational> b) {
103 final n = a.length;
104 final rows = List.generate(n, (i) => [...a[i], b[i]]);
105 for (var k = 0; k < n; k++) {
106 var r = k;
107 while (r < n && rows[r][k].zero) {
108 r++;
109 }
110 if (r == n) return null;
111 final old = rows[k];
112 rows[k] = rows[r];
113 rows[r] = old;
114 final pivot = rows[k][k];
115 for (var j = 0; j <= n; j++) {
116 rows[k][j] = rows[k][j] / pivot;
117 }
118 for (var i = 0; i < n; i++) {
119 if (i != k && !rows[i][k].zero) {
120 final factor = rows[i][k];
121 for (var j = 0; j <= n; j++) {
122 rows[i][j] = rows[i][j] - factor * rows[k][j];
123 }
124 }
125 }
126 }
127 return rows.map((row) => row[n]).toList();
128}

130Set<String> reachable(String start, List<(String, String)> edges) {
131 final seen = <String>{start}, pending = <String>[start];
132 while (pending.isNotEmpty) {
133 final node = pending.removeLast();
134 for (final (a, b) in edges) {
135 final next = a == node
136 ? b
137 : b == node
138 ? a
139 : null;
140 if (next != null && seen.add(next)) pending.add(next);
141 }
142 }
143 return seen;
144}

146class Groups {
147 final Map<String, String> parent;
148 Groups(List<String> nodes) : parent = {for (final node in nodes) node: node};
149 String find(String node) {
150 final previous = parent[node]!;
151 if (previous == node) return node;
152 final root = find(previous);
153 parent[node] = root;
154 return root;
155 }
156
157 void join(String a, String b) {
158 a = find(a);
159 b = find(b);
160 if (a == b) return;
161 if (a.compareTo(b) < 0) {
162 parent[b] = a;
163 } else {
164 parent[a] = b;
165 }
166 }

272 Equations equations([Map<String, Rational> overrides = const {}]) {
273 final names = circuitNodes(),
274 sources = elements.where(voltageType).toList();
275 final count = names.length + sources.length;
276 final nodeIndex = {for (var i = 0; i < names.length; i++) names[i]: i};
277 final sourceIndex = {
278 for (var i = 0; i < sources.length; i++)
279 sources[i].name: names.length + i,
280 };
281 final a = matrix(count), b = zeros(count);
282 int node(String name) => name == '0' ? -1 : nodeIndex[name]!;
283 int source(String? name) => sourceIndex[name]!;
284 void stamp(int row, int column, Rational value) {
285 if (row >= 0 && column >= 0) a[row][column] = a[row][column] + value;
286 }
287
288 void rhs(int row, Rational value) {
289 if (row >= 0) b[row] = b[row] + value;
290 }
291
292 for (final element in elements) {
293 final terminals = element.terminals, type = element.name[0];
294 final x = node(terminals[0]),
295 y = node(terminals[1]),
296 w = overrides[element.name] ?? element.value;
297 if (type == 'R') {
298 final g = one / w;
299 stamp(x, x, g);
300 stamp(y, y, g);
301 stamp(x, y, -g);
302 stamp(y, x, -g);
303 } else if (type == 'I') {
304 rhs(x, -w);
305 rhs(y, w);
306 } else if (type == 'G') {
307 stamp(x, node(terminals[2]), w);
308 stamp(y, node(terminals[3]), w);
309 stamp(x, node(terminals[3]), -w);
310 stamp(y, node(terminals[2]), -w);
311 } else if (type == 'F') {
312 stamp(x, source(element.reference), w);
313 stamp(y, source(element.reference), -w);
314 } else {
315 final s = source(element.name);
316 stamp(x, s, one);
317 stamp(s, x, one);
318 stamp(y, s, -one);
319 stamp(s, y, -one);
320 if (type == 'V') {
321 rhs(s, w);
322 } else if (type == 'E') {
323 stamp(s, node(terminals[2]), -w);
324 stamp(s, node(terminals[3]), w);
325 } else {
326 stamp(s, source(element.reference), -w);
327 }
328 }
329 }
330 return Equations(names, sources, count, nodeIndex, sourceIndex, a, b);
331 }

333 Rational potential(
334 Equations equations,
335 List<Rational> solution,
336 String name,
337 ) => name == '0' ? zero : solution[equations.nodeIndex[name]!];
338 Rational sourceCurrent(
339 Equations equations,
340 List<Rational> solution,
341 String? name,
342 ) => solution[equations.sourceIndex[name]!];
343 Rational current(
344 Equations equations,
345 List<Rational> solution,
346 Element element,
347 ) {
348 final t = element.terminals, type = element.name[0];
349 if (type == 'R')
350 return (potential(equations, solution, t[0]) -
351 potential(equations, solution, t[1])) /
352 element.value;
353 if (type == 'I') return element.value;
354 if (type == 'G')
355 return element.value *
356 (potential(equations, solution, t[2]) -
357 potential(equations, solution, t[3]));
358 if (type == 'F')
359 return element.value *
360 sourceCurrent(equations, solution, element.reference);
361 return sourceCurrent(equations, solution, element.name);
362 }

364 List<Rational> injection(Equations equations, String a, String b) {

367 final i = equations.nodeIndex[a]!;
368 column[i] = column[i] + one;
369 }
370 if (b != '0') {
371 final i = equations.nodeIndex[b]!;
372 column[i] = column[i] - one;
373 }

377 List<Rational> alone(Equations equations, Element element, bool unit) {
378 final column = zeros(equations.count), value = unit ? one : element.value;
379 if (element.name[0] == 'V') {
380 column[equations.sourceIndex[element.name]!] = value;
381 } else {
382 final a = element.terminals[0], b = element.terminals[1];
383 if (a != '0') {
384 final i = equations.nodeIndex[a]!;
385 column[i] = column[i] - value;
386 }
387 if (b != '0') {
388 final i = equations.nodeIndex[b]!;
389 column[i] = column[i] + value;
390 }
391 }
392 return solve(equations.a, column)!;

395 List<String> resistance(String a, String b) {
396 final groups = Groups(nodes(includeGround: true));
397 for (final element in elements) {
398 if (element.name[0] == 'V')
399 groups.join(element.terminals[0], element.terminals[1]);
400 }
401 a = groups.find(a);
402 b = groups.find(b);
403 if (a == b) return ['Req = ${zero.value()}'];
404 final edges = <Edge>[];
405 for (final element in elements) {
406 if (element.name[0] == 'R') {
407 final x = groups.find(element.terminals[0]),
408 y = groups.find(element.terminals[1]);
409 if (x != y) edges.add(Edge(x, y, one / element.value));
410 }
411 }
412 final seen = reachable(a, edges.map((edge) => (edge.a, edge.b)).toList());
413 if (!seen.contains(b)) return ['Req = INFINITE'];
414 final names = seen.where((name) => name != b).toList()..sort();
415 final indices = {for (var i = 0; i < names.length; i++) names[i]: i};
416 final g = matrix(names.length), column = zeros(names.length);
417 void stamp(String row, String col, Rational value) {
418 if (row != b && col != b) {
419 final i = indices[row]!, j = indices[col]!;
420 g[i][j] = g[i][j] + value;
421 }
422 }
423
424 for (final edge in edges) {
425 if (seen.contains(edge.a)) {
426 stamp(edge.a, edge.a, edge.value);
427 stamp(edge.b, edge.b, edge.value);
428 stamp(edge.a, edge.b, -edge.value);
429 stamp(edge.b, edge.a, -edge.value);
430 }
431 }
432 column[indices[a]!] = one;
433 return ['Req = ${solve(g, column)![indices[a]!].value()}'];
434 }

436 List<String> reduce(String a, String b) {
437 if (a == b) return ['Req = ${zero.value()}'];
438 var list = elements
439 .where((element) => element.name[0] == 'R')
440 .map(
441 (element) => Resistor(
442 element.name,
443 element.terminals[0],
444 element.terminals[1],
445 element.value,
446 ),
447 )
448 .toList();
449 final connected = reachable(
450 a,
451 list.map((element) => (element.a, element.b)).toList(),
452 );
453 if (!connected.contains(b)) return ['Req = INFINITE'];
454 list = list.where((element) => connected.contains(element.a)).toList();
455 final output = <String>[];
456 var serial = 0;
457 while (true) {
458 var remove = list.indexWhere((element) => element.a == element.b);
459 if (remove >= 0) {
460 output.add('remove ${list.removeAt(remove).name}');
461 continue;
462 }
463 final degree = <String, int>{};
464 for (final element in list) {
465 for (final node in [element.a, element.b]) {
466 degree[node] = (degree[node] ?? 0) + 1;
467 }
468 }
469 final internal =
470 degree.keys.where((node) => node != a && node != b).toList()..sort();
471 String? dangling;
472 for (final node in internal) {
473 if (degree[node] == 1) {
474 dangling = node;
475 break;
476 }
477 }
478 if (dangling != null) {
479 remove = list.indexWhere(
480 (element) => element.a == dangling || element.b == dangling,
481 );
482 output.add('remove ${list.removeAt(remove).name}');
483 continue;
484 }
485 var first = -1, second = -1, parallel = false;
486 search:
487 for (var i = 0; i < list.length; i++) {
488 for (var j = i + 1; j < list.length; j++) {
489 final x = list[i], y = list[j];
490 if (x.a == y.a && x.b == y.b || x.a == y.b && x.b == y.a) {
491 first = i;
492 second = j;
493 parallel = true;
494 break search;
495 }
496 }
497 }
498 String? center;
499 if (!parallel) {
500 for (final node in internal) {
501 if (degree[node] == 2) {
502 center = node;
503 break;
504 }
505 }
506 if (center == null) break;
507 final indices = <int>[];
508 for (var i = 0; i < list.length; i++) {
509 if (list[i].a == center || list[i].b == center) indices.add(i);
510 }
511 first = indices[0];
512 second = indices[1];
513 }
514 final x = list[first], y = list[second], name = '#${++serial}';
515 final value = parallel
516 ? x.value * y.value / (x.value + y.value)
517 : x.value + y.value;
518 final left = parallel
519 ? x.a
520 : x.a == center
521 ? x.b
522 : x.a;
523 final right = parallel
524 ? x.b
525 : y.a == center
526 ? y.b
527 : y.a;
528 output.add(
529 '$name = ${x.name} ${parallel ? '||' : '+'} ${y.name} = ${value.value()}',
530 );
531 list.removeAt(second);
532 list.removeAt(first);
533 list.add(Resistor(name, left, right, value));
534 }
535 output.add(
536 list.length == 1
537 ? 'Req = ${list[0].value.value()}'
538 : 'not series-parallel: ${list.length} resistors remain',
539 );
540 return output;
541 }

543 List<String> check() {
544 final names = circuitNodes();
545 if (names.isEmpty) return ['no nodes'];
546 final output = <String>[];
547 for (final node in names) {
548 var degree = 0;
549 for (final element in elements) {
550 if (element.terminals[0] == node) degree++;
551 if (element.terminals[1] == node) degree++;
552 }
553 if (degree < 2) output.add('dangling node $node');
554 }
555 for (final node in floating()) {
556 output.add('floating node $node');
557 }
558 for (final name in loops()) {
559 output.add('voltage source loop at $name');
560 }
561 if (output.isEmpty) {
562 final eq = equations();
563 if (solve(eq.a, eq.b) == null) output.add('singular matrix');
564 }
565 return output.isEmpty ? ['ok'] : output;
566 }

568 List<String> command(List<String> words) {
569 final command = words[0], args = words.sublist(1);
570 const counts = {
571 '.OP': 0,
572 '.CURRENT': 1,
573 '.VOLTAGE': 2,
574 '.POWER': 0,
575 '.KCL': 1,
576 '.THEVENIN': 2,
577 '.NORTON': 2,
578 '.MAXPOWER': 2,
579 '.TF': 2,
580 '.SENS': 1,
581 '.ZPARAM': 4,
582 '.SUPERPOSITION': 1,
583 '.SWEEP': 5,
584 '.MATRIX': 0,
585 '.REQ': 2,
586 '.REDUCE': 2,
587 '.CHECK': 0,
588 '.ALTER': 2,
589 '.DELETE': 1,
590 '.CLEAR': 0,
591 '.NODES': 0,
592 '.PRINT': 0,
593 };
594 if (!counts.containsKey(command)) fail('unknown command $command');
595 if (args.length != counts[command])
596 fail('wrong argument count for $command');
597 if (command == '.CLEAR') {
598 final count = elements.length;
599 elements = [];
600 return ['cleared $count elements'];
601 }
602 if (command == '.DELETE') {
603 final element = elementCheck(args[0]);
604 for (final user in elements) {
605 if (user.reference == element.name)
606 fail('${element.name} is referenced by ${user.name}');
607 }
608 elements.remove(element);
609 return ['deleted ${element.name}'];
610 }
611 if (command == '.ALTER') {
612 final element = elementCheck(args[0]), value = numberCheck(args[1]);
613 if (element.name[0] == 'R' && !value.positive)
614 fail('nonpositive resistance ${element.name}');
615 final output = 'altered ${element.name}: ${element.value} -> $value';
616 element.value = value;
617 return [output];
618 }
619 if (command == '.NODES') {
620 final names = nodes();
621 return names.isEmpty
622 ? ['no nodes']
623 : names
624 .map(
625 (node) =>
626 '$node: ${elements.where((element) => element.terminals.contains(node)).map((element) => element.name).join(' ')}',
627 )
628 .toList();
629 }
630 if (command == '.PRINT') {
631 return elements.isEmpty
632 ? ['empty netlist']
633 : elements
634 .map(
635 (element) => [
636 element.name,
637 ...element.terminals,
638 if (element.reference != null) element.reference!,
639 '${element.value}',
640 ].join(' '),
641 )
642 .toList();
643 }
644 if (command == '.CHECK') return check();
645 if (command == '.SWEEP') {
646 final element = elementCheck(args[0]);
647 final from = numberCheck(args[1]),
648 to = numberCheck(args[2]),
649 step = numberCheck(args[3]);
650 nodeCheck(args[4]);
651 if (!step.positive) fail('invalid sweep step');
652 if (from.compareTo(to) > 0) fail('invalid sweep range');
653 final span = (to - from) / step, count = span.n ~/ span.d + BigInt.one;
654 if (count > BigInt.from(20)) fail('too many sweep points');
655 if (element.name[0] == 'R' && !from.positive)
656 fail('nonpositive resistance ${element.name}');
657 structure();
658 final output = <String>[];
659 for (var k = BigInt.zero; k < count; k += BigInt.one) {
660 final value = from + step * Rational(k),
661 eq = equations({element.name: value});
662 final solution = solve(eq.a, eq.b);
663 output.add(
664 '${element.name} = $value: ${solution == null ? 'singular matrix' : 'V(${args[4]}) = ${potential(eq, solution, args[4]).value()}'}',
665 );
666 }
667 return output;
668 }
669 if (command == '.REQ' || command == '.REDUCE') {
670 nodeCheck(args[0]);
671 nodeCheck(args[1]);
672 return command == '.REQ'
673 ? resistance(args[0], args[1])
674 : reduce(args[0], args[1]);
675 }
676 Element? selected;
677 if (command == '.CURRENT') {
678 selected = elementCheck(args[0]);
679 } else if (command == '.TF') {
680 nodeCheck(args[0]);
681 selected = elementCheck(args[1]);
682 if (!independent(selected))
683 fail('not an independent source ${selected.name}');
684 } else if ([
685 '.VOLTAGE',
686 '.KCL',
687 '.THEVENIN',
688 '.NORTON',
689 '.MAXPOWER',
690 '.SENS',
691 '.ZPARAM',
692 '.SUPERPOSITION',
693 ].contains(command)) {
694 for (final node in args) {
695 nodeCheck(node);
696 }
697 }
698 structure();
699 final eq = equations();
700 if (command == '.MATRIX') {
701 final labels = [
702 ...eq.nodes,
703 ...eq.sources.map((element) => element.name),
704 ];
705 return [
706 'unknowns: ${[...eq.nodes.map((node) => 'V($node)'), ...eq.sources.map((element) => 'I(${element.name})')].join(' ')}',
707 for (var i = 0; i < labels.length; i++)
708 '${labels[i]}: ${eq.a[i].join(' ')} | ${eq.b[i]}',
709 ];
710 }
711 final solution = solve(eq.a, eq.b);
712 if (solution == null) fail('singular matrix');
713 Rational voltage(String name, [List<Rational>? values]) =>
714 potential(eq, values ?? solution, name);
715 final output = <String>[];
716 if (command == '.OP') {
717 for (final node in eq.nodes) {
718 output.add('V($node) = ${voltage(node).value()}');
719 }
720 for (final element in eq.sources) {
721 output.add(
722 'I(${element.name}) = ${sourceCurrent(eq, solution, element.name).value()}',
723 );
724 }
725 } else if (command == '.CURRENT') {
726 final element = selected!;
727 output.add(
728 'I(${element.name}) = ${current(eq, solution, element).value()}',
729 );
730 } else if (command == '.VOLTAGE') {
731 output.add(
732 'V(${args[0]},${args[1]}) = ${(voltage(args[0]) - voltage(args[1])).value()}',
733 );
734 } else if (command == '.POWER') {
735 var absorbed = zero, delivered = zero, total = zero;
736 for (final element in elements) {
737 final power =
738 (voltage(element.terminals[0]) - voltage(element.terminals[1])) *
739 current(eq, solution, element);
740 output.add('P(${element.name}) = ${power.value()}');
741 if (power.positive) {
742 absorbed = absorbed + power;
743 } else if (!power.zero) {
744 delivered = delivered - power;
745 }
746 total = total + power;
747 }
748 output.addAll([
749 'absorbed = ${absorbed.value()}',
750 'delivered = ${delivered.value()}',
751 'total = ${total.value()}',
752 ]);
753 } else if (command == '.KCL') {
754 var sum = zero;
755 for (final element in elements) {
756 if (element.terminals[0] == args[0] ||
757 element.terminals[1] == args[0]) {
758 final value = element.terminals[0] == element.terminals[1]
759 ? zero
760 : element.terminals[0] == args[0]
761 ? current(eq, solution, element)
762 : -current(eq, solution, element);
763 output.add(
764 'I(${element.name}) leaving ${args[0]} = ${value.value()}',
765 );
766 sum = sum + value;
767 }
768 }
769 output.add('sum = ${sum.value()}');
770 } else if (['.THEVENIN', '.NORTON', '.MAXPOWER'].contains(command)) {
771 final value = voltage(args[0]) - voltage(args[1]),
772 z = injection(eq, args[0], args[1]);
773 final resistance = voltage(args[0], z) - voltage(args[1], z);
774 if (command == '.THEVENIN') {
775 output.addAll([
776 'Vth = ${value.value()}',
777 'Rth = ${resistance.value()}',
778 ]);
779 } else if (command == '.NORTON') {
780 if (resistance.zero) fail('zero thevenin resistance');
781 output.addAll([
782 'In = ${(value / resistance).value()}',
783 'Rn = ${resistance.value()}',
784 ]);
785 } else {
786 if (!resistance.positive) fail('nonpositive thevenin resistance');
787 output.addAll([
788 'RL = ${resistance.value()}',
789 'Pmax = ${(value * value / (rational(4) * resistance)).value()}',
790 ]);
791 }
792 } else if (command == '.TF') {
793 final source = selected!;
794 final z = alone(eq, source, true);
795 output.add('gain = ${voltage(args[0], z).value()}');
796 if (source.name[0] == 'V') {
797 final value = sourceCurrent(eq, z, source.name);
798 output.add(
799 'input resistance = ${value.zero ? 'INFINITE' : (-one / value).value()}',
800 );
801 } else {
802 output.add(
803 'input resistance = ${(voltage(source.terminals[1], z) - voltage(source.terminals[0], z)).value()}',
804 );
805 }
806 output.add(
807 'output resistance = ${voltage(args[0], injection(eq, args[0], '0')).value()}',
808 );
809 } else if (command == '.SENS') {
810 for (final element in elements) {
811 final column = zeros(eq.count),
812 t = element.terminals,
813 type = element.name[0];
814 void addNode(String node, Rational value) {
815 if (node != '0') {
816 final i = eq.nodeIndex[node]!;
817 column[i] = column[i] + value;
818 }
819 }
820
821 if (type == 'R') {
822 final value = current(eq, solution, element) / element.value;
823 addNode(t[0], value);
824 addNode(t[1], -value);
825 } else if (type == 'I') {
826 addNode(t[0], -one);
827 addNode(t[1], one);
828 } else if (type == 'G' || type == 'F') {
829 final value = type == 'G'
830 ? voltage(t[2]) - voltage(t[3])
831 : sourceCurrent(eq, solution, element.reference);
832 addNode(t[0], -value);
833 addNode(t[1], value);
834 } else {
835 column[eq.sourceIndex[element.name]!] = type == 'V'
836 ? one
837 : type == 'E'
838 ? voltage(t[2]) - voltage(t[3])
839 : sourceCurrent(eq, solution, element.reference);
840 }
841 output.add(
842 'dV(${args[0]})/d${element.name} = ${voltage(args[0], solve(eq.a, column)!).value()}',
843 );
844 }
845 } else if (command == '.ZPARAM') {
846 final first = injection(eq, args[0], args[1]),
847 second = injection(eq, args[2], args[3]);
848 output.addAll([
849 'z11 = ${(voltage(args[0], first) - voltage(args[1], first)).value()}',
850 'z12 = ${(voltage(args[0], second) - voltage(args[1], second)).value()}',
851 'z21 = ${(voltage(args[2], first) - voltage(args[3], first)).value()}',
852 'z22 = ${(voltage(args[2], second) - voltage(args[3], second)).value()}',
853 ]);
854 } else if (command == '.SUPERPOSITION') {
855 var total = zero;
856 for (final element in elements.where(independent)) {
857 final value = voltage(args[0], alone(eq, element, false));
858 output.add('${element.name}: V(${args[0]}) = ${value.value()}');
859 total = total + value;
860 }
861 output.add('total: V(${args[0]}) = ${total.value()}');
862 }
863 return output;
864 }
865}

867void main() {
868 final circuit = Circuit(), output = <String>[];
869 var lineNumber = 0;
870 String? line;
871 while ((line = stdin.readLineSync()) != null) {
872 final input = line!;
873 lineNumber++;
874 if (input == '.END') break;
875 if (input.startsWith('*')) continue;
876 final words = input.split(' ');
877 try {
878 if (input.startsWith('.')) {
879 output.addAll(circuit.command(words));
880 } else {
881 circuit.add(words);
882 }
883 } on Problem catch (error) {
884 output.add('error: line $lineNumber: ${error.message}');
885 }
886 }
887 if (output.isNotEmpty) stdout.write('${output.join('\n')}\n');
888}
