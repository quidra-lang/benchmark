package main
import (
	"bufio"
	"fmt"
	"os"
	"sort"
	"strconv"
	"strings"
)
type rational struct{ numerator, denominator int64 }
func gcd(a, b int64) int64 {
	if a < 0 {
		a = -a
	}
	for b != 0 {
		a, b = b, a%b
	}
	return a
}
func fraction(n, d int64) rational {
	if d < 0 {
		n, d = -n, -d
	}
	common := gcd(n, d)
	return rational{n / common, d / common}
}
var zero = rational{0, 1}
var one = rational{1, 1}
func (a rational) add(b rational) rational {
	return fraction(a.numerator*b.denominator+b.numerator*a.denominator, a.denominator*b.denominator)
}
func (a rational) subtract(b rational) rational {
	return fraction(a.numerator*b.denominator-b.numerator*a.denominator, a.denominator*b.denominator)
}
func (a rational) multiply(b rational) rational {
	return fraction(a.numerator*b.numerator, a.denominator*b.denominator)
}
func (a rational) divide(b rational) rational {
	return fraction(a.numerator*b.denominator, a.denominator*b.numerator)
}
func (a rational) negate() rational { return rational{-a.numerator, a.denominator} }
func (a rational) compare(b rational) int {
	x, y := a.numerator*b.denominator, b.numerator*a.denominator
	if x < y {
		return -1
	}
	if x > y {
		return 1
	}
	return 0
}
func (a rational) String() string {
	if a.denominator == 1 {
		return strconv.FormatInt(a.numerator, 10)
	}
	return fmt.Sprintf("%d/%d", a.numerator, a.denominator)
}
func (a rational) value() string {
	n, sign := a.numerator, ""
	if n < 0 {
		n, sign = -n, "-"
	}
	scaled := n * 1000000 / a.denominator
	return fmt.Sprintf("%s (%s%d.%06d)", a.String(), sign, scaled/1000000, scaled%1000000)
}
func digits(text string) bool {
	if text == "" {
		return false
	}
	for i := range text {
		if text[i] < '0' || text[i] > '9' {
			return false
		}
	}
	return true
}
func integer(text string) int64 { value, _ := strconv.ParseInt(text, 10, 64); return value }
func parseNumber(text string) (rational, bool) {
	multiplier := one
	if strings.HasSuffix(text, "meg") {
		multiplier, text = rational{1000000, 1}, strings.TrimSuffix(text, "meg")
	} else if strings.HasSuffix(text, "k") {
		multiplier, text = rational{1000, 1}, strings.TrimSuffix(text, "k")
	} else if strings.HasSuffix(text, "m") {
		multiplier, text = rational{1, 1000}, strings.TrimSuffix(text, "m")
	} else if strings.HasSuffix(text, "u") {
		multiplier, text = rational{1, 1000000}, strings.TrimSuffix(text, "u")
	}
	negative := strings.HasPrefix(text, "-")
	if negative {
		text = text[1:]
	}
	result := zero
	if digits(text) {
		result = fraction(integer(text), 1)
	} else if parts := strings.Split(text, "."); len(parts) == 2 && digits(parts[0]) && digits(parts[1]) {
		denominator := int64(1)
		for range parts[1] {
			denominator *= 10
		}
		result = fraction(integer(parts[0])*denominator+integer(parts[1]), denominator)
	} else if parts := strings.Split(text, "/"); len(parts) == 2 && digits(parts[0]) && digits(parts[1]) && integer(parts[1]) != 0 {
		result = fraction(integer(parts[0]), integer(parts[1]))
	} else {
		return zero, false
	}
	result = result.multiply(multiplier)
	if negative {
		result = result.negate()
	}
	return result, true
}
func validNode(name string) bool {
	if name == "0" {
		return true
	}
	if len(name) < 1 || len(name) > 8 || name[0] < 'a' || name[0] > 'z' {
		return false
	}
	for i := range name {
		if !(name[i] >= 'a' && name[i] <= 'z' || name[i] >= '0' && name[i] <= '9') {
			return false
		}
	}
	return true
}
type element struct {
	name, n1, n2, c1, c2, reference string
	kind                            byte
	value                           rational
}
func (e element) terminals() []string {
	result := []string{e.n1, e.n2}
	if e.kind == 'E' || e.kind == 'G' {
		result = append(result, e.c1, e.c2)
	}
	return result
}
func voltageSource(e element) bool { return e.kind == 'V' || e.kind == 'E' || e.kind == 'H' }
type circuit struct{ elements []element }
type commandError string
func fail(format string, args ...any) { panic(commandError(fmt.Sprintf(format, args...))) }
func sorted(set map[string]bool) []string {
	result := []string{}
	for name := range set {
		result = append(result, name)
	}
	sort.Strings(result)
	return result
}
func (c *circuit) nodes(includeGround bool) []string {
	set := map[string]bool{}
	for _, e := range c.elements {
		for _, node := range e.terminals() {
			if includeGround || node != "0" {
				set[node] = true
			}
		}
	}
	return sorted(set)
}
func (c *circuit) node(name string) {
	if name == "0" {
		return
	}
	for _, node := range c.nodes(false) {
		if name == node {
			return
		}
	}
	fail("unknown node %s", name)
}
func (c *circuit) index(name string) int {
	for i, e := range c.elements {
		if e.name == name {
			return i
		}
	}
	fail("unknown element %s", name)
	return -1
}
func number(text string) rational {
	value, ok := parseNumber(text)
	if !ok {
		fail("invalid value %s", text)
	}
	return value
}
func (c *circuit) add(fields []string) {
	name, kind := fields[0], fields[0][0]
	expected := 4
	if kind == 'E' || kind == 'G' {
		expected = 6
	}
	if kind == 'F' || kind == 'H' {
		expected = 5
	}
	if !strings.ContainsRune("RVIEGFH", rune(kind)) {
		fail("unknown element type %s", name)
	}
	if len(fields) != expected {
		fail("wrong field count for %s", name)
	}
	for _, e := range c.elements {
		if e.name == name {
			fail("duplicate element %s", name)
		}
	}
	nodes := fields[1:3]
	if kind == 'E' || kind == 'G' {
		nodes = fields[1:5]
	}
	for _, node := range nodes {
		if !validNode(node) {
			fail("invalid node %s", node)
		}
	}
	e := element{name: name, kind: kind, n1: fields[1], n2: fields[2]}
	if kind == 'E' || kind == 'G' {
		e.c1, e.c2 = fields[3], fields[4]
	}
	e.value = number(fields[len(fields)-1])
	if kind == 'R' && e.value.numerator <= 0 {
		fail("nonpositive resistance %s", name)
	}
	c.elements = append(c.elements, e)
}
type unionFind struct{ parent map[string]string }
func newUnion(nodes []string) *unionFind {
	u := &unionFind{map[string]string{"0": "0"}}
	for _, name := range nodes {
		u.parent[name] = name
	}
	return u
}
func (u *unionFind) root(name string) string {
	if u.parent[name] != name {
		u.parent[name] = u.root(u.parent[name])
	}
	return u.parent[name]
}
func (u *unionFind) join(a, b string) {
	a, b = u.root(a), u.root(b)
	if a > b {
		a, b = b, a
	}
	u.parent[b] = a
}
func (c *circuit) floating() []string {
	u := newUnion(c.nodes(false))
	for _, e := range c.elements {
		if e.kind == 'R' || voltageSource(e) {
			u.join(e.n1, e.n2)
		}
	}
	result := []string{}
	for _, name := range c.nodes(false) {
		if u.root(name) != u.root("0") {
			result = append(result, name)
		}
	}
	return result
}
func (c *circuit) sourceLoops() []string {
	u := newUnion(c.nodes(false))
	result := []string{}
	for _, e := range c.elements {
		if voltageSource(e) {
			if u.root(e.n1) == u.root(e.n2) {
				result = append(result, e.name)
			}
			u.join(e.n1, e.n2)
		}
	}
	return result
}
func (c *circuit) structure() {
	if len(c.nodes(false)) == 0 {
		fail("no nodes")
	}
	if floating := c.floating(); len(floating) > 0 {
		fail("floating node %s", floating[0])
	}
	if loops := c.sourceLoops(); len(loops) > 0 {
		fail("voltage source loop at %s", loops[0])
	}
}
type equations struct {
	a                             [][]rational
	b                             []rational
	names, labels, nodes, sources []string
	indices                       map[string]int
}
func zeroVector(n int) []rational {
	result := make([]rational, n)
	for i := range result {
		result[i] = zero
	}
	return result
}
func zeroMatrix(n int) [][]rational {
	result := make([][]rational, n)
	for i := range result {
		result[i] = zeroVector(n)
	}
	return result
}
func (m *equations) add(row, column string, value rational) {
	if row != "0" && column != "0" {
		i, j := m.indices[row], m.indices[column]
		m.a[i][j] = m.a[i][j].add(value)
	}
}
func (m *equations) rhs(vector []rational, row string, value rational) {
	if row != "0" {
		i := m.indices[row]
		vector[i] = vector[i].add(value)
	}
}
func (c *circuit) build() *equations {
	m := &equations{nodes: c.nodes(false), indices: map[string]int{}}
	for _, node := range m.nodes {
		m.indices[node] = len(m.names)
		m.names = append(m.names, "V("+node+")")
		m.labels = append(m.labels, node)
	}
	for _, e := range c.elements {
		if voltageSource(e) {
			m.indices[e.name] = len(m.names)
			m.sources = append(m.sources, e.name)
			m.names = append(m.names, "I("+e.name+")")
			m.labels = append(m.labels, e.name)
		}
	}
	m.a, m.b = zeroMatrix(len(m.names)), zeroVector(len(m.names))
	for _, e := range c.elements {
		w := e.value
		switch e.kind {
		case 'R':
			g := one.divide(w)
			m.add(e.n1, e.n1, g)
			m.add(e.n2, e.n2, g)
			m.add(e.n1, e.n2, g.negate())
			m.add(e.n2, e.n1, g.negate())
		case 'I':
			m.rhs(m.b, e.n1, w.negate())
			m.rhs(m.b, e.n2, w)
		case 'G':
			m.add(e.n1, e.c1, w)
			m.add(e.n2, e.c2, w)
			m.add(e.n1, e.c2, w.negate())
			m.add(e.n2, e.c1, w.negate())
		}
	}
	return m
}
func solve(a [][]rational, b []rational) []rational {
	n := len(b)
	rows := make([][]rational, n)
	for i := range rows {
		rows[i] = append(append([]rational(nil), a[i]...), b[i])
	}
	for k := 0; k < n; k++ {
		pivot := k
		for pivot < n && rows[pivot][k].numerator == 0 {
			pivot++
		}
		if pivot == n {
			return nil
		}
		rows[k], rows[pivot] = rows[pivot], rows[k]
		divisor := rows[k][k]
		for j := 0; j <= n; j++ {
			rows[k][j] = rows[k][j].divide(divisor)
		}
		for i := 0; i < n; i++ {
			if i != k {
				factor := rows[i][k]
				if factor.numerator != 0 {
					for j := 0; j <= n; j++ {
						rows[i][j] = rows[i][j].subtract(factor.multiply(rows[k][j]))
					}
				}
			}
		}
	}
	result := zeroVector(n)
	for i := range result {
		result[i] = rows[i][n]
	}
	return result
}
func (m *equations) potential(solution []rational, node string) rational {
	if node == "0" {
		return zero
	}
	return solution[m.indices[node]]
}
var argumentCounts = map[string]int{
	".ZPARAM": 4, ".SUPERPOSITION": 1, ".SWEEP": 5, ".MATRIX": 0,
}
func (c *circuit) command(fields []string) []string {
	name, args := fields[0], fields[1:]
	count, known := argumentCounts[name]
	if !known {
		fail("unknown command %s", name)
	}
	if len(args) != count {
		fail("wrong argument count for %s", name)
	}
	lines := []string{}
	emit := func(format string, values ...any) {
		lines = append(lines, fmt.Sprintf(format, values...))
	}
	switch name {
	case ".SWEEP":
		index := c.index(args[0])
		from := number(args[1])
		to := number(args[2])
		step := number(args[3])
		c.node(args[4])
		if step.numerator <= 0 {
			fail("invalid sweep step")
		}
		if from.compare(to) > 0 {
			fail("invalid sweep range")
		}
		ratio := to.subtract(from).divide(step)
		points := ratio.numerator/ratio.denominator + 1
		if points > 20 {
			fail("too many sweep points")
		}
		if c.elements[index].kind == 'R' && from.numerator <= 0 {
			fail("nonpositive resistance %s", args[0])
		}
		c.structure()
		temporary := circuit{elements: append([]element(nil), c.elements...)}
		for k := int64(0); k < points; k++ {
			value := from.add(step.multiply(rational{k, 1}))
			temporary.elements[index].value = value
			m := temporary.build()
			solution := solve(m.a, m.b)
			if solution == nil {
				emit("%s = %s: singular matrix", args[0], value.String())
			} else {
				emit("%s = %s: V(%s) = %s", args[0], value.String(), args[4], m.potential(solution, args[4]).value())
			}
		}
		return lines
	}
}
func (c *circuit) process(line string, number int, out *bufio.Writer) {
	defer func() {
		if caught := recover(); caught != nil {
			if reason, ok := caught.(commandError); ok {
				fmt.Fprintf(out, "error: line %d: %s\n", number, reason)
			} else {
				panic(caught)
			}
		}
	}()
	if line[0] == '*' {
		return
	}
	fields := strings.Split(line, " ")
	if line[0] != '.' {
		c.add(fields)
		return
	}
	for _, text := range c.command(fields) {
		fmt.Fprintln(out, text)
	}
}
func main() {
	scanner := bufio.NewScanner(os.Stdin)
	out := bufio.NewWriter(os.Stdout)
	defer out.Flush()
	c := circuit{}
	lineNumber := 0
	for scanner.Scan() {
		lineNumber++
		line := scanner.Text()
		if line == ".END" {
			break
		}
		c.process(line, lineNumber, out)
	}
}
