5private class Q(numerator: Long, denominator: Long = 1) : Comparable<Q> {
6 val p: Long
7 val q: Long
8 init {
9 var a = kotlin.math.abs(numerator)
10 var b = kotlin.math.abs(denominator)
11 while (b != 0L) {
12 val r = a % b
13 a = b
14 b = r
15 }
16 val sign = if (denominator < 0) -1 else 1
17 p = sign * numerator / a
18 q = kotlin.math.abs(denominator) / a
19 }
20 operator fun plus(b: Q) = Q(p * b.q + b.p * q, q * b.q)
21 operator fun minus(b: Q) = Q(p * b.q - b.p * q, q * b.q)
22 operator fun times(b: Q) = Q(p * b.p, q * b.q)
23 operator fun div(b: Q) = Q(p * b.q, q * b.p)
24 operator fun unaryMinus() = Q(-p, q)
25 override fun compareTo(other: Q) = (p * other.q).compareTo(other.p * q)
26 override fun toString() = if (q == 1L) "$p" else "$p/$q"
27
28 fun value(): String {
29 val h = kotlin.math.abs(p) * 1000000 / q
30 return "$this (${if (p < 0) "-" else ""}${h / 1000000}.${(h % 1000000).toString().padStart(6, '0')})"
31 }
32}
33private val zero = Q(0)
34private val one = Q(1)

36private fun number(raw: String): Q {
37 var s = raw
38 var multiplier = one
39 for ((suffix, factor) in listOf("meg" to Q(1000000), "k" to Q(1000), "m" to Q(1, 1000), "u" to Q(1, 1000000))) {
40 if (s.endsWith(suffix)) {
41 s = s.dropLast(suffix.length)
42 multiplier = factor
43 break
44 }
45 }
46 var sign = 1L
47 if (s.startsWith('-')) {
48 sign = -1
49 s = s.drop(1)
50 }
51 val value = when {
52 s.matches(Regex("[0-9]+")) -> Q(s.toLong())
53 s.matches(Regex("[0-9]+\\.[0-9]+")) -> {
54 val parts = s.split('.')
55 Q((parts[0] + parts[1]).toLong(), parts[1].fold(1L) {
56 n, _ -> n * 10
57 })
58 }
59 s.matches(Regex("[0-9]+/[0-9]+")) -> {
60 val parts = s.split('/')
61 if (parts[1].toLong() == 0L) fail("invalid value $raw")
62 Q(parts[0].toLong(), parts[1].toLong())
63 } else -> fail("invalid value $raw")
64 }
65 return Q(sign) * value * multiplier
66}

78private class Union(nodes: Collection<String>) {
79 private val parent = nodes.associateWith {
80 it
81 }.toMutableMap()
82
83 fun root(x: String): String {
84 val p = parent.getValue(x)
85 if (p == x) return x
86 val r = root(p)
87 parent[x] = r
88 return r
89 }
90
91 fun join(a: String, b: String) {
92 val x = root(a)
93 val y = root(b)
94 if (x < y) parent[y] = x else parent[x] = y
95 }
96}

98private fun reach(start: String, edges: List<Pair<String, String>>): Set<String> {
99 val found = mutableSetOf(start)
100 val queue = ArrayDeque<String>()
101 queue.add(start)
102 while (queue.isNotEmpty()) {
103 val a = queue.removeFirst()
104 for ((u, v) in edges) {
105 val b = if (a == u) v else if (a == v) u else continue
106 if (found.add(b)) queue.add(b)
107 }
108 }
109 return found
110}

112private fun solve(a: Array<Array<Q>>, b: Array<Q>): Array<Q>? {
113 val n = b.size
114 val m = Array(n) {
115 i -> Array(n + 1) {
116 j -> if (j == n) b[i] else a[i][j]
117 }
118 }
119 for (k in 0 until n) {
120 val r = (k until n).firstOrNull {
121 m[it][k].p != 0L
122 } ?: return null
123 val temp = m[k]
124 m[k] = m[r]
125 m[r] = temp
126 val pivot = m[k][k]
127 for (j in 0..n) m[k][j] = m[k][j] / pivot
128 for (i in 0 until n) if (i != k && m[i][k].p != 0L) {
129 val f = m[i][k]
130 for (j in 0..n) m[i][j] = m[i][j] - f * m[k][j]
131 }
132 }
133 return Array(n) {
134 m[it][n]
135 }
136}

138private class Equations(val elements: List<Element>) {
139 val nodes = elements.flatMap {
140 it.nodes
141 }.filter {
142 it != "0"
143 }.distinct().sorted()
144 val sources = elements.filter {
145 it.voltage
146 }
147 val labels = nodes + sources.map {
148 it.name
149 }
150 val index = labels.withIndex().associate {
151 it.value to it.index
152 }
153 val a = Array(labels.size) {
154 Array(labels.size) {
155 zero
156 }
157 }
158 val b = Array(labels.size) {
159 zero
160 }
161 init {
162 for (e in elements) {
163 val w = e.value
164 when (e.kind) {
165 'R' -> {
166 val g = one / w
167 add(e.a, e.a, g)
168 add(e.b, e.b, g)
169 add(e.a, e.b, -g)
170 add(e.b, e.a, -g)
171 }
172 'I' -> {
173 add(b, e.a, -w)
174 add(b, e.b, w)
175 }
176 'G' -> {
177 add(e.a, e.nodes[2], w)
178 add(e.b, e.nodes[3], w)
179 add(e.a, e.nodes[3], -w)
180 add(e.b, e.nodes[2], -w)
181 }
182 'F' -> {
183 add(e.a, e.reference, w)
184 add(e.b, e.reference, -w)
185 } else -> {
186 add(e.a, e.name, one)
187 add(e.name, e.a, one)
188 add(e.b, e.name, -one)
189 add(e.name, e.b, -one)
190 when (e.kind) {
191 'V' -> add(b, e.name, w)
192 'E' -> {
193 add(e.name, e.nodes[2], -w)
194 add(e.name, e.nodes[3], w)
195 }
196 'H' -> add(e.name, e.reference, -w)
197 }
198 }
199 }
200 }
201 }

203 fun add(row: String, column: String, value: Q) {
204 if (row != "0" && column != "0") {
205 val i = index.getValue(row)
206 val j = index.getValue(column)
207 a[i][j] = a[i][j] + value
208 }
209 }

211 fun add(rhs: Array<Q>, row: String, value: Q) {
212 if (row != "0") {
213 val i = index.getValue(row)
214 rhs[i] = rhs[i] + value
215 }
216 }
217
218 fun v(y: Array<Q>, node: String) = if (node == "0") zero else y[index.getValue(node)]
219
220 fun injection(x: String, y: String): Array<Q> {
221 val rhs = Array(labels.size) {
222 zero
223 }
224 add(rhs, x, one)
225 add(rhs, y, -one)
226 return solve(a, rhs)!!
227 }
228
229 fun alone(e: Element, unit: Boolean = false): Array<Q> {
230 val rhs = Array(labels.size) {
231 zero
232 }
233 val w = if (unit) one else e.value
234 if (e.kind == 'V') add(rhs, e.name, w) else {
235 add(rhs, e.a, -w)
236 add(rhs, e.b, w)
237 }
238 return solve(a, rhs)!!
239 }
240
241 fun current(e: Element, y: Array<Q>): Q = when (e.kind) {
242 'R' -> (v(y, e.a) - v(y, e.b)) / e.value
243 'I' -> e.value
244 'G' -> e.value * (v(y, e.nodes[2]) - v(y, e.nodes[3]))
245 'F' -> e.value * y[index.getValue(e.reference)]
246 else -> y[index.getValue(e.name)]
247 }
248}

299 fun add(w: List<String>) {
300 val name = w[0]
301 val kind = name[0]
302 if (kind !in "RVIEGFH") fail("unknown element type $name")
303 val count = if (kind in "RVI") 4 else if (kind in "EG") 6 else 5
304 if (w.size != count) fail("wrong field count for $name")
305 if (elements.any {
306 it.name == name
307 }) fail("duplicate element $name")
308 val ns = if (kind in "EG") w.subList(1, 5) else w.subList(1, 3)
309 for (n in ns) if (!validNode(n)) fail("invalid node $n")
310 val reference = if (kind in "FH") w[3] else ""
311 if (kind in "FH" && elements.none {
312 it.name == reference && it.kind == 'V'
313 }) fail("unknown voltage source $reference")
314 val value = number(w.last())
315 if (kind == 'R' && value <= zero) fail("nonpositive resistance $name")
316 elements.add(Element(name, ns.toList(), reference, value))
317 }

319 fun resistance(a: String, b: String): List<String> {
320 val u = Union(nodes() + "0")
321 for (e in elements) if (e.kind == 'V') u.join(e.a, e.b)
322 val x = u.root(a)
323 val y = u.root(b)
324 if (x == y) return listOf("Req = ${zero.value()}")
325 val edges = elements.filter {
326 it.kind == 'R' && u.root(it.a) != u.root(it.b)
327 }
328 val connected = reach(x, edges.map {
329 u.root(it.a) to u.root(it.b)
330 })
331 if (y !in connected) return listOf("Req = INFINITE")
332 val groups = connected.filter {
333 it != y
334 }.sorted()
335 val index = groups.withIndex().associate {
336 it.value to it.index
337 }
338 val matrix = Array(groups.size) {
339 Array(groups.size) {
340 zero
341 }
342 }
343 val rhs = Array(groups.size) {
344 zero
345 }
346
347 fun stamp(p: String, q: String, v: Q) {
348 if (p != y && q != y) {
349 val i = index.getValue(p)
350 val j = index.getValue(q)
351 matrix[i][j] = matrix[i][j] + v
352 }
353 }
354 for (e in edges) if (u.root(e.a) in connected) {
355 val p = u.root(e.a)
356 val q = u.root(e.b)
357 val g = one / e.value
358 stamp(p, p, g)
359 stamp(q, q, g)
360 stamp(p, q, -g)
361 stamp(q, p, -g)
362 }
363 rhs[index.getValue(x)] = one
364 return listOf("Req = ${solve(matrix, rhs)!![index.getValue(x)].value()}")
365 }

367 fun reduce(a: String, b: String): List<String> {
368 if (a == b) return listOf("Req = ${zero.value()}")
369 var list = elements.filter {
370 it.kind == 'R'
371 }.map {
372 Resistor(it.name, it.a, it.b, it.value)
373 }.toMutableList()
374 val connected = reach(a, list.map {
375 it.a to it.b
376 })
377 if (b !in connected) return listOf("Req = INFINITE")
378 list = list.filter {
379 it.a in connected
380 }.toMutableList()
381 val out = mutableListOf<String>()
382 var created = 0
383 while (true) {
384 val self = list.indexOfFirst {
385 it.a == it.b
386 }
387 if (self >= 0) {
388 out.add("remove ${list.removeAt(self).name}")
389 continue
390 }
391 val degree = sortedMapOf<String, Int>()
392 for (r in list) {
393 degree[r.a] = (degree[r.a] ?: 0) + 1
394 degree[r.b] = (degree[r.b] ?: 0) + 1
395 }
396 val dangling = degree.keys.firstOrNull {
397 it != a && it != b && degree[it] == 1
398 }
399 if (dangling != null) {
400 val i = list.indexOfFirst {
401 it.a == dangling || it.b == dangling
402 }
403 out.add("remove ${list.removeAt(i).name}")
404 continue
405 }
406 var pair: Pair<Int, Int>? = null
407 outer@ for (i in list.indices) for (j in i + 1 until list.size) {
408 val x = list[i]
409 val y = list[j]
410 if (x.a == y.a && x.b == y.b || x.a == y.b && x.b == y.a) {
411 pair = i to j
412 break@outer
413 }
414 }
415 if (pair != null) {
416 val (i, j) = pair
417 val x = list[i]
418 val y = list[j]
419 val value = x.value * y.value / (x.value + y.value)
420 val r = Resistor("#${++created}", x.a, x.b, value)
421 list.removeAt(j)
422 list.removeAt(i)
423 list.add(r)
424 out.add("${r.name} = ${x.name} || ${y.name} = ${value.value()}")
425 continue
426 }
427 val p = degree.keys.firstOrNull {
428 it != a && it != b && degree[it] == 2
429 } ?: break
430 val indices = list.indices.filter {
431 list[it].a == p || list[it].b == p
432 }
433 val i = indices[0]
434 val j = indices[1]
435 val x = list[i]
436 val y = list[j]
437 val value = x.value + y.value
438 val r = Resistor("#${++created}", if (x.a == p) x.b else x.a, if (y.a == p) y.b else y.a, value)
439 list.removeAt(j)
440 list.removeAt(i)
441 list.add(r)
442 out.add("${r.name} = ${x.name} + ${y.name} = ${value.value()}")
443 }
444 out.add(if (list.size == 1) "Req = ${list[0].value.value()}" else "not series-parallel: ${list.size} resistors remain")
445 return out
446 }

448 fun command(w: List<String>): List<String> {
449 val c = w[0]
450 val counts = mapOf(".OP" to 0, ".CURRENT" to 1, ".VOLTAGE" to 2, ".POWER" to 0, ".KCL" to 1, ".THEVENIN" to 2, ".NORTON" to 2, ".MAXPOWER" to 2, ".TF" to 2, ".SENS" to 1, ".ZPARAM" to 4, ".SUPERPOSITION" to 1, ".SWEEP" to 5, ".MATRIX" to 0, ".REQ" to 2, ".REDUCE" to 2, ".CHECK" to 0, ".ALTER" to 2, ".DELETE" to 1, ".CLEAR" to 0, ".NODES" to 0, ".PRINT" to 0)
451 val count = counts[c] ?: fail("unknown command $c")
452 if (w.size != count + 1) fail("wrong argument count for $c")
453 when (c) {
454 ".CLEAR" -> {
455 val n = elements.size
456 elements.clear()
457 return listOf("cleared $n elements")
458 }
459 ".PRINT" -> return if (elements.isEmpty()) listOf("empty netlist") else elements.map {
460 e -> (listOf(e.name) + e.nodes + (if (e.reference.isEmpty()) emptyList() else listOf(e.reference)) + e.value.toString()).joinToString(" ")
461 }
462 ".NODES" -> {
463 val ns = elements.flatMap {
464 it.nodes
465 }.distinct().sorted()
466 return if (ns.isEmpty()) listOf("no nodes") else ns.map {
467 n -> "$n: " + elements.filter {
468 n in it.nodes
469 }.joinToString(" ") {
470 it.name
471 }
472 }
473 }
474 ".DELETE" -> {
475 val e = get(w[1])
476 elements.firstOrNull {
477 it.kind in "FH" && it.reference == e.name
478 }?.let {
479 fail("${e.name} is referenced by ${it.name}")
480 }
481 elements.remove(e)
482 return listOf("deleted ${e.name}")
483 }
484 ".ALTER" -> {
485 val e = get(w[1])
486 val v = number(w[2])
487 if (e.kind == 'R' && v <= zero) fail("nonpositive resistance ${e.name}")
488 val old = e.value
489 e.value = v
490 return listOf("altered ${e.name}: $old -> $v")
491 }
492 ".CHECK" -> {
493 val ns = nodes()
494 if (ns.isEmpty()) return listOf("no nodes")
495 val out = mutableListOf<String>()
496 for (n in ns) if (elements.sumOf {
497 (if (it.a == n) 1 else 0) + (if (it.b == n) 1 else 0)
498 } < 2) out.add("dangling node $n")
499 out.addAll(floating().map {
500 "floating node $it"
501 })
502 out.addAll(loops().map {
503 "voltage source loop at $it"
504 })
505 if (out.isEmpty()) {
506 val e = Equations(elements)
507 if (solve(e.a, e.b) == null) out.add("singular matrix")
508 }
509 return out.ifEmpty {
510 listOf("ok")
511 }
512 }
513 ".REQ", ".REDUCE" -> {
514 node(w[1])
515 node(w[2])
516 return if (c == ".REQ") resistance(w[1], w[2]) else reduce(w[1], w[2])
517 }
518 ".SWEEP" -> {
519 val e = get(w[1])
520 val from = number(w[2])
521 val to = number(w[3])
522 val step = number(w[4])
523 node(w[5])
524 if (step <= zero) fail("invalid sweep step")
525 if (from > to) fail("invalid sweep range")
526 val ratio = (to - from) / step
527 val k = ratio.p / ratio.q + 1
528 if (k > 20) fail("too many sweep points")
529 if (e.kind == 'R' && from <= zero) fail("nonpositive resistance ${e.name}")
530 structure()
531 return (0 until k.toInt()).map {
532 i ->
533 val x = from + Q(i.toLong()) * step
534 val eq = Equations(elements.map {
535 if (it === e) it.copy(value = x) else it
536 })
537 val y = solve(eq.a, eq.b)
538 "${e.name} = $x: " + if (y == null) "singular matrix" else "V(${w[5]}) = ${eq.v(y, w[5]).value()}"
539 }
540 }
541 }
542 when (c) {
543 ".CURRENT" -> get(w[1])
544 ".VOLTAGE", ".THEVENIN", ".NORTON", ".MAXPOWER" -> {
545 node(w[1])
546 node(w[2])
547 }
548 ".KCL", ".SENS", ".SUPERPOSITION" -> node(w[1])
549 ".ZPARAM" -> w.drop(1).forEach(::node)
550 ".TF" -> {
551 node(w[1])
552 val e = get(w[2])
553 if (!e.independent) fail("not an independent source ${e.name}")
554 }
555 }
556 structure()
557 val eq = Equations(elements)
558 if (c == ".MATRIX") return listOf("unknowns: " + (eq.nodes.map {
559 "V($it)"
560 } + eq.sources.map {
561 "I(${it.name})"
562 }).joinToString(" ")) + eq.labels.indices.map {
563 i -> "${eq.labels[i]}: " + eq.a[i].joinToString(" ") + " | ${eq.b[i]}"
564 }
565 val y = solve(eq.a, eq.b) ?: fail("singular matrix")
566
567 fun v(n: String) = eq.v(y, n)
568 val out = mutableListOf<String>()
569 when (c) {
570 ".OP" -> {
571 for (n in eq.nodes) out.add("V($n) = ${v(n).value()}")
572 for (e in eq.sources) out.add("I(${e.name}) = ${eq.current(e, y).value()}")
573 }
574 ".CURRENT" -> out.add("I(${w[1]}) = ${eq.current(get(w[1]), y).value()}")
575 ".VOLTAGE" -> out.add("V(${w[1]},${w[2]}) = ${(v(w[1]) - v(w[2])).value()}")
576 ".POWER" -> {
577 var absorbed = zero
578 var delivered = zero
579 var total = zero
580 for (e in elements) {
581 val p = (v(e.a) - v(e.b)) * eq.current(e, y)
582 out.add("P(${e.name}) = ${p.value()}")
583 if (p > zero) absorbed += p
584 if (p < zero) delivered -= p
585 total += p
586 }
587 out.add("absorbed = ${absorbed.value()}")
588 out.add("delivered = ${delivered.value()}")
589 out.add("total = ${total.value()}")
590 }
591 ".KCL" -> {
592 var sum = zero
593 val n = w[1]
594 for (e in elements) if (e.a == n || e.b == n) {
595 val i = if (e.a == n && e.b == n) zero else if (e.a == n) eq.current(e, y) else -eq.current(e, y)
596 sum += i
597 out.add("I(${e.name}) leaving $n = ${i.value()}")
598 }
599 out.add("sum = ${sum.value()}")
600 }
601 ".THEVENIN", ".NORTON", ".MAXPOWER" -> {
602 val z = eq.injection(w[1], w[2])
603 val voltage = v(w[1]) - v(w[2])
604 val resistance = eq.v(z, w[1]) - eq.v(z, w[2])
605 if (c == ".NORTON" && resistance.p == 0L) fail("zero thevenin resistance")
606 if (c == ".MAXPOWER" && resistance <= zero) fail("nonpositive thevenin resistance")
607 when (c) {
608 ".THEVENIN" -> {
609 out.add("Vth = ${voltage.value()}")
610 out.add("Rth = ${resistance.value()}")
611 }
612 ".NORTON" -> {
613 out.add("In = ${(voltage / resistance).value()}")
614 out.add("Rn = ${resistance.value()}")
615 } else -> {
616 out.add("RL = ${resistance.value()}")
617 out.add("Pmax = ${(voltage * voltage / (Q(4) * resistance)).value()}")
618 }
619 }
620 }
621 ".TF" -> {
622 val e = get(w[2])
623 val z = eq.alone(e, true)
624 out.add("gain = ${eq.v(z, w[1]).value()}")
625 if (e.kind == 'V') {
626 val i = z[eq.index.getValue(e.name)]
627 out.add("input resistance = " + if (i.p == 0L) "INFINITE" else (-one / i).value())
628 } else out.add("input resistance = ${(eq.v(z, e.b) - eq.v(z, e.a)).value()}")
629 out.add("output resistance = ${eq.v(eq.injection(w[1], "0"), w[1]).value()}")
630 }
631 ".SENS" -> for (e in elements) {
632 val d = Array(eq.labels.size) {
633 zero
634 }
635 when (e.kind) {
636 'R' -> {
637 val t = eq.current(e, y) / e.value
638 eq.add(d, e.a, t)
639 eq.add(d, e.b, -t)
640 }
641 'I' -> {
642 eq.add(d, e.a, -one)
643 eq.add(d, e.b, one)
644 }
645 'G', 'F' -> {
646 val t = if (e.kind == 'G') v(e.nodes[2]) - v(e.nodes[3]) else y[eq.index.getValue(e.reference)]
647 eq.add(d, e.a, -t)
648 eq.add(d, e.b, t)
649 }
650 'V' -> eq.add(d, e.name, one)
651 'E' -> eq.add(d, e.name, v(e.nodes[2]) - v(e.nodes[3]))
652 'H' -> eq.add(d, e.name, y[eq.index.getValue(e.reference)])
653 }
654 out.add("dV(${w[1]})/d${e.name} = ${eq.v(solve(eq.a, d)!!, w[1]).value()}")
655 }
656 ".ZPARAM" -> {
657 val z1 = eq.injection(w[1], w[2])
658 val z2 = eq.injection(w[3], w[4])
659 out.add("z11 = ${(eq.v(z1, w[1]) - eq.v(z1, w[2])).value()}")
660 out.add("z12 = ${(eq.v(z2, w[1]) - eq.v(z2, w[2])).value()}")
661 out.add("z21 = ${(eq.v(z1, w[3]) - eq.v(z1, w[4])).value()}")
662 out.add("z22 = ${(eq.v(z2, w[3]) - eq.v(z2, w[4])).value()}")
663 }
664 ".SUPERPOSITION" -> {
665 var total = zero
666 for (e in elements.filter {
667 it.independent
668 }) {
669 val value = eq.v(eq.alone(e), w[1])
670 total += value
671 out.add("${e.name}: V(${w[1]}) = ${value.value()}")
672 }
673 out.add("total: V(${w[1]}) = ${total.value()}")
674 }
675 }
676 return out
677 }
678}
