1class CircuitError < StandardError
2end

4Element = Struct.new(:name, :type, :nodes, :value, :reference)

6def fraction(value)
7 value.denominator == 1 ? value.numerator.to_s : "#{value.numerator}/#{value.denominator}"
8end
9
10def formatted(value)
11 scaled = value.numerator.abs * 1_000_000 / value.denominator
12 decimal = (value.negative? ? '-' : '') + (scaled / 1_000_000).to_s + '.' + (scaled % 1_000_000).to_s.rjust(6, '0')
13 fraction(value) + ' (' + decimal + ')'
14end
15
16def rational_number(text)
17 multiplier = Rational(1)
18 { 'meg' => Rational(1_000_000), 'k' => Rational(1000), 'm' => Rational(1, 1000), 'u' => Rational(1, 1_000_000) }.each do |suffix, value|
19 next unless text.end_with?(suffix)
20 text = text.delete_suffix(suffix)
21 multiplier = value
22 break
23 end
24 return nil unless text.match?(/\A-?[0-9]+(?:\.[0-9]+|\/[0-9]+)?\z/)
25 return nil if text.include?('/') && text.split('/')[1].to_i == 0
26 Rational(text) * multiplier
27end

29def solve(matrix, rhs)
30 n = rhs.length
31 rows = matrix.each_with_index.map { |row, i| row + [rhs[i]] }
32 n.times do |k|
33 pivot = (k...n).find { |i| rows[i][k] != 0 }
34 return nil unless pivot
35 rows[k], rows[pivot] = rows[pivot], rows[k] if pivot != k
36 divisor = rows[k][k]
37 rows[k] = rows[k].map { |entry| entry / divisor }
38 n.times do |i|
39 next if i == k || rows[i][k] == 0
40 factor = rows[i][k]
41 rows[i] = rows[i].each_with_index.map { |entry, j| entry - factor * rows[k][j] }
42 end
43 end
44 rows.map(&:last)
45end

47def reachable(start, edges)
48 seen = Set[start]
49 queue = [start]
50 until queue.empty?
51 node = queue.shift
52 edges.each do |a, b|
53 next unless a == node || b == node
54 other = a == node ? b : a
55 next if seen.include?(other)
56 seen << other
57 queue << other
58 end
59 end
60 seen
61end

63class Groups
64 def initialize(nodes)
65 @parent = nodes.to_h { |node| [node, node] }
66 end
67
68 def find(node)
69 @parent[node] = find(@parent[node]) unless @parent[node] == node
70 @parent[node]
71 end
72
73 def join(a, b)
74 a, b = find(a), find(b)
75 @parent[[a, b].max] = [a, b].min
76 end
77end

79class Circuit
80 COUNTS = { '.OP' => 0, '.CURRENT' => 1, '.VOLTAGE' => 2, '.POWER' => 0, '.KCL' => 1,
81 '.THEVENIN' => 2, '.NORTON' => 2, '.MAXPOWER' => 2, '.TF' => 2, '.SENS' => 1,
82 '.ZPARAM' => 4, '.SUPERPOSITION' => 1, '.SWEEP' => 5, '.MATRIX' => 0, '.REQ' => 2,
83 '.REDUCE' => 2, '.CHECK' => 0, '.ALTER' => 2, '.DELETE' => 1, '.CLEAR' => 0,
84 '.NODES' => 0, '.PRINT' => 0 }.freeze
85 ANALYSES = %w[.OP .CURRENT .VOLTAGE .POWER .KCL .THEVENIN .NORTON .MAXPOWER .TF .SENS .ZPARAM .SUPERPOSITION].freeze

87 def initialize
88 @elements = []
89 end

91 def nodes
92 @elements.flat_map(&:nodes).uniq.reject { |node| node == '0' }.sort
93 end
94
95 def sources
96 @elements.select { |element| %w[V E H].include?(element.type) }
97 end
98
99 def element(name)
100 @elements.find { |item| item.name == name } || raise(CircuitError, "unknown element #{name}")
101 end
102
103 def node(name)
104 raise CircuitError, "unknown node #{name}" unless name == '0' || nodes.include?(name)
105 end
106
107 def number(text)
108 rational_number(text) || raise(CircuitError, "invalid value #{text}")
109 end
110
111 def add(words)
112 name = words[0]
113 type = name[0]
114 raise CircuitError, "unknown element type #{name}" unless %w[R V I E G F H].include?(type)
115 count = %w[E G].include?(type) ? 6 : %w[F H].include?(type) ? 5 : 4
116 raise CircuitError, "wrong field count for #{name}" unless words.length == count
117 raise CircuitError, "duplicate element #{name}" if @elements.any? { |item| item.name == name }
118 terminals = words[1, %w[E G].include?(type) ? 4 : 2]
119 terminals.each do |terminal|
120 raise CircuitError, "invalid node #{terminal}" unless terminal.match?(/\A(?:0|[a-z][a-z0-9]{0,7})\z/)
121 end
122 reference = %w[F H].include?(type) ? words[3] : nil
123 if reference && !@elements.any? { |item| item.name == reference && item.type == 'V' }
124 raise CircuitError, "unknown voltage source #{reference}"
125 end
126 value = number(words[-1])
127 raise CircuitError, "nonpositive resistance #{name}" if type == 'R' && value <= 0
128 @elements << Element.new(name, type, terminals, value, reference)
129 []
130 end

132 def grounding
133 reachable('0', @elements.select { |item| %w[R V E H].include?(item.type) }.map { |item| item.nodes[0, 2] })
134 end
135
136 def voltage_loops
137 groups = Groups.new(['0'] + nodes)
138 result = []
139 sources.each do |item|
140 a, b = item.nodes
141 result << item.name if groups.find(a) == groups.find(b)
142 groups.join(a, b)
143 end
144 result
145 end

147 def structure_check
148 names = nodes
149 raise CircuitError, 'no nodes' if names.empty?
150 grounded = grounding
151 floating = names.find { |name| !grounded.include?(name) }
152 raise CircuitError, "floating node #{floating}" if floating
153 loop_name = voltage_loops.first
154 raise CircuitError, "voltage source loop at #{loop_name}" if loop_name
155 end
156
157 def equations(overrides = {})
158 labels = nodes + sources.map(&:name)
159 index = labels.each_with_index.to_h
160 n = labels.length
161 matrix = Array.new(n) { Array.new(n, Rational(0)) }
162 rhs = Array.new(n, Rational(0))
163 put = lambda do |row, column, value|
164 matrix[index[row]][index[column]] += value if row != '0' && column != '0'
165 end
166 put_rhs = lambda do |row, value|
167 rhs[index[row]] += value unless row == '0'
168 end
169 @elements.each do |item|
170 a, b, c, d = item.nodes
171 value = overrides.fetch(item.name, item.value)
172 s = item.name
173 case item.type
174 when 'R'
175 g = 1 / value
176 put.call(a, a, g)
177 put.call(b, b, g)
178 put.call(a, b, -g)
179 put.call(b, a, -g)
180 when 'I'
181 put_rhs.call(a, -value)
182 put_rhs.call(b, value)
183 when 'G'
184 put.call(a, c, value)
185 put.call(b, d, value)
186 put.call(a, d, -value)
187 put.call(b, c, -value)
188 when 'F'
189 put.call(a, item.reference, value)
190 put.call(b, item.reference, -value)
191 else
192 put.call(a, s, 1)
193 put.call(s, a, 1)
194 put.call(b, s, -1)
195 put.call(s, b, -1)
196 case item.type
197 when 'V'
198 put_rhs.call(s, value)
199 when 'E'
200 put.call(s, c, -value)
201 put.call(s, d, value)
202 when 'H'
203 put.call(s, item.reference, -value)
204 end
205 end
206 end
207 [matrix, rhs, labels, index]
208 end
209
210 def potential(solution, index, name)
211 name == '0' ? Rational(0) : solution[index[name]]
212 end
213
214 def current(item, solution, index)
215 a, b, c, d = item.nodes
216 case item.type
217 when 'R' then (potential(solution, index, a) - potential(solution, index, b)) / item.value
218 when 'V', 'E', 'H' then solution[index[item.name]]
219 when 'I' then item.value
220 when 'G' then item.value * (potential(solution, index, c) - potential(solution, index, d))
221 when 'F' then item.value * solution[index[item.reference]]
222 end
223 end
224
225 def injection(matrix, index, a, b)
226 rhs = Array.new(matrix.length, Rational(0))
227 rhs[index[a]] += 1 unless a == '0'
228 rhs[index[b]] -= 1 unless b == '0'
229 solve(matrix, rhs)
230 end
231
232 def source_solution(matrix, index, item, unit = false)
233 rhs = Array.new(matrix.length, Rational(0))
234 value = unit ? Rational(1) : item.value
235 if item.type == 'V'
236 rhs[index[item.name]] += value
237 else
238 a, b = item.nodes
239 rhs[index[a]] -= value unless a == '0'
240 rhs[index[b]] += value unless b == '0'
241 end
242 solve(matrix, rhs)
243 end

245 def analysis(command, args)
246 case command
247 when '.CURRENT'
248 element(args[0])
249 when '.VOLTAGE', '.THEVENIN', '.NORTON', '.MAXPOWER', '.ZPARAM'
250 args.each { |name| node(name) }
251 when '.KCL', '.SENS', '.SUPERPOSITION'
252 node(args[0])
253 when '.TF'
254 node(args[0])
255 source = element(args[1])
256 raise CircuitError, "not an independent source #{source.name}" unless %w[V I].include?(source.type)
257 end
258 structure_check
259 matrix, rhs, _, index = equations
260 solution = solve(matrix, rhs) || raise(CircuitError, 'singular matrix')
261 voltage = ->(name, values = solution) { potential(values, index, name) }
262 lines = []
263 case command
264 when '.OP'
265 nodes.each { |name| lines << "V(#{name}) = #{formatted(voltage.call(name))}" }
266 sources.each { |item| lines << "I(#{item.name}) = #{formatted(solution[index[item.name]])}" }
267 when '.CURRENT'
268 lines << "I(#{args[0]}) = #{formatted(current(element(args[0]), solution, index))}"
269 when '.VOLTAGE'
270 lines << "V(#{args[0]},#{args[1]}) = #{formatted(voltage.call(args[0]) - voltage.call(args[1]))}"
271 when '.POWER'
272 absorbed = delivered = total = Rational(0)
273 @elements.each do |item|
274 power = (voltage.call(item.nodes[0]) - voltage.call(item.nodes[1])) * current(item, solution, index)
275 absorbed += power if power > 0
276 delivered -= power if power < 0
277 total += power
278 lines << "P(#{item.name}) = #{formatted(power)}"
279 end
280 lines += ["absorbed = #{formatted(absorbed)}", "delivered = #{formatted(delivered)}", "total = #{formatted(total)}"]
281 when '.KCL'
282 a = args[0]
283 total = Rational(0)
284 @elements.each do |item|
285 x, y = item.nodes
286 next unless x == a || y == a
287 value = x == y ? Rational(0) : (x == a ? 1 : -1) * current(item, solution, index)
288 total += value
289 lines << "I(#{item.name}) leaving #{a} = #{formatted(value)}"
290 end
291 lines << "sum = #{formatted(total)}"
292 when '.THEVENIN', '.NORTON', '.MAXPOWER'
293 a, b = args
294 vth = voltage.call(a) - voltage.call(b)
295 z = injection(matrix, index, a, b)
296 rth = voltage.call(a, z) - voltage.call(b, z)
297 if command == '.NORTON'
298 raise CircuitError, 'zero thevenin resistance' if rth == 0
299 lines = ["In = #{formatted(vth / rth)}", "Rn = #{formatted(rth)}"]
300 elsif command == '.MAXPOWER'
301 raise CircuitError, 'nonpositive thevenin resistance' if rth <= 0
302 lines = ["RL = #{formatted(rth)}", "Pmax = #{formatted(vth * vth / (4 * rth))}"]
303 else
304 lines = ["Vth = #{formatted(vth)}", "Rth = #{formatted(rth)}"]
305 end
306 when '.TF'
307 a, name = args
308 item = element(name)
309 z = source_solution(matrix, index, item, true)
310 lines << "gain = #{formatted(voltage.call(a, z))}"
311 if item.type == 'V'
312 j = z[index[name]]
313 lines << (j == 0 ? 'input resistance = INFINITE' : "input resistance = #{formatted(-1 / j)}")
314 else
315 lines << "input resistance = #{formatted(voltage.call(item.nodes[1], z) - voltage.call(item.nodes[0], z))}"
316 end
317 lines << "output resistance = #{formatted(voltage.call(a, injection(matrix, index, a, '0')))}"
318 when '.SENS'
319 a = args[0]
320 @elements.each do |item|
321 d = Array.new(matrix.length, Rational(0))
322 add = ->(row, value) { d[index[row]] += value unless row == '0' }
323 x, y, c, e = item.nodes
324 case item.type
325 when 'R'
326 t = current(item, solution, index) / item.value
327 add.call(x, t)
328 add.call(y, -t)
329 when 'I'
330 add.call(x, -1)
331 add.call(y, 1)
332 when 'G', 'F'
333 t = item.type == 'G' ? voltage.call(c) - voltage.call(e) : solution[index[item.reference]]
334 add.call(x, -t)
335 add.call(y, t)
336 when 'V'
337 add.call(item.name, 1)
338 when 'E'
339 add.call(item.name, voltage.call(c) - voltage.call(e))
340 when 'H'
341 add.call(item.name, solution[index[item.reference]])
342 end
343 lines << "dV(#{a})/d#{item.name} = #{formatted(voltage.call(a, solve(matrix, d)))}"
344 end
345 when '.ZPARAM'
346 a, b, c, d = args
347 z1 = injection(matrix, index, a, b)
348 z2 = injection(matrix, index, c, d)
349 lines = ["z11 = #{formatted(voltage.call(a, z1) - voltage.call(b, z1))}",
350 "z12 = #{formatted(voltage.call(a, z2) - voltage.call(b, z2))}",
351 "z21 = #{formatted(voltage.call(c, z1) - voltage.call(d, z1))}",
352 "z22 = #{formatted(voltage.call(c, z2) - voltage.call(d, z2))}"]
353 when '.SUPERPOSITION'
354 a = args[0]
355 total = Rational(0)
356 @elements.select { |item| %w[V I].include?(item.type) }.each do |item|
357 value = voltage.call(a, source_solution(matrix, index, item))
358 total += value
359 lines << "#{item.name}: V(#{a}) = #{formatted(value)}"
360 end
361 lines << "total: V(#{a}) = #{formatted(total)}"
362 end
363 lines
364 end

366 def equivalent(a, b)
367 groups = Groups.new(['0'] + nodes)
368 @elements.select { |item| item.type == 'V' }.each { |item| groups.join(*item.nodes) }
369 a, b = groups.find(a), groups.find(b)
370 return ['Req = 0 (0.000000)'] if a == b
371 edges = @elements.select { |item| item.type == 'R' }.filter_map do |item|
372 x, y = item.nodes.map { |name| groups.find(name) }
373 [x, y, 1 / item.value] unless x == y
374 end
375 connected = reachable(a, edges.map { |x, y, _| [x, y] })
376 return ['Req = INFINITE'] unless connected.include?(b)
377 labels = (connected.to_a - [b]).sort
378 index = labels.each_with_index.to_h
379 matrix = Array.new(labels.length) { Array.new(labels.length, Rational(0)) }
380 rhs = Array.new(labels.length, Rational(0))
381 rhs[index[a]] = Rational(1)
382 edges.each do |x, y, g|
383 next unless connected.include?(x)
384 matrix[index[x]][index[x]] += g unless x == b
385 matrix[index[y]][index[y]] += g unless y == b
386 if x != b && y != b
387 matrix[index[x]][index[y]] -= g
388 matrix[index[y]][index[x]] -= g
389 end
390 end
391 ["Req = #{formatted(solve(matrix, rhs)[index[a]])}"]
392 end

394 def reduce_resistors(a, b)
395 return ['Req = 0 (0.000000)'] if a == b
396 list = @elements.select { |item| item.type == 'R' }.map(&:dup)
397 connected = reachable(a, list.map { |item| item.nodes })
398 return ['Req = INFINITE'] unless connected.include?(b)
399 list.select! { |item| connected.include?(item.nodes[0]) }
400 lines = []
401 serial = 0
402 loop do
403 self_loop = list.index { |item| item.nodes[0] == item.nodes[1] }
404 if self_loop
405 lines << "remove #{list.delete_at(self_loop).name}"
406 next
407 end
408 degree = Hash.new(0)
409 list.each { |item| item.nodes.each { |name| degree[name] += 1 } }
410 internal = (degree.keys - [a, b]).sort
411 leaf = internal.find { |name| degree[name] == 1 }
412 if leaf
413 i = list.index { |item| item.nodes.include?(leaf) }
414 lines << "remove #{list.delete_at(i).name}"
415 next
416 end
417 pair = nil
418 list.each_index do |i|
419 j = ((i + 1)...list.length).find { |k| list[i].nodes.sort == list[k].nodes.sort }
420 if j
421 pair = [i, j]
422 break
423 end
424 end
425 if pair
426 x, y = pair.map { |i| list[i] }
427 value = x.value * y.value / (x.value + y.value)
428 terminals = x.nodes
429 operator = '||'
430 else
431 middle = internal.find { |name| degree[name] == 2 }
432 break unless middle
433 pair = list.each_index.select { |i| list[i].nodes.include?(middle) }
434 x, y = pair.map { |i| list[i] }
435 terminals = [x.nodes.find { |name| name != middle }, y.nodes.find { |name| name != middle }]
436 value = x.value + y.value
437 operator = '+'
438 end
439 serial += 1
440 name = "##{serial}"
441 lines << "#{name} = #{x.name} #{operator} #{y.name} = #{formatted(value)}"
442 list = list.each_with_index.reject { |_, i| pair.include?(i) }.map(&:first)
443 list << Element.new(name, 'R', terminals, value)
444 end
445 lines << (list.length == 1 ? "Req = #{formatted(list[0].value)}" : "not series-parallel: #{list.length} resistors remain")
446 lines
447 end

449 def command(words)
450 cmd, *args = words
451 raise CircuitError, "unknown command #{cmd}" unless COUNTS.key?(cmd)
452 raise CircuitError, "wrong argument count for #{cmd}" unless args.length == COUNTS[cmd]
453 return analysis(cmd, args) if ANALYSES.include?(cmd)
454 case cmd
455 when '.REQ', '.REDUCE'
456 args.each { |name| node(name) }
457 cmd == '.REQ' ? equivalent(*args) : reduce_resistors(*args)
458 when '.SWEEP'
459 item = element(args[0])
460 from, to, step = args[1, 3].map { |value| number(value) }
461 node(args[4])
462 raise CircuitError, 'invalid sweep step' if step <= 0
463 raise CircuitError, 'invalid sweep range' if from > to
464 count = ((to - from) / step).floor + 1
465 raise CircuitError, 'too many sweep points' if count > 20
466 raise CircuitError, "nonpositive resistance #{item.name}" if item.type == 'R' && from <= 0
467 structure_check
468 Array.new(count) do |k|
469 x = from + k * step
470 matrix, rhs, _, index = equations(item.name => x)
471 solution = solve(matrix, rhs)
472 prefix = "#{item.name} = #{fraction(x)}: "
473 prefix + (solution ? "V(#{args[4]}) = #{formatted(potential(solution, index, args[4]))}" : 'singular matrix')
474 end
475 when '.MATRIX'
476 structure_check
477 matrix, rhs, labels, = equations
478 unknowns = nodes.map { |name| "V(#{name})" } + sources.map { |item| "I(#{item.name})" }
479 ["unknowns: #{unknowns.join(' ')}"] + matrix.each_with_index.map { |row, i| "#{labels[i]}: #{row.map { |value| fraction(value) }.join(' ')} | #{fraction(rhs[i])}" }
480 when '.CHECK'
481 return ['no nodes'] if nodes.empty?
482 lines = []
483 nodes.each do |name|
484 degree = @elements.sum { |item| item.nodes[0, 2].count(name) }
485 lines << "dangling node #{name}" if degree < 2
486 end
487 grounded = grounding
488 nodes.each { |name| lines << "floating node #{name}" unless grounded.include?(name) }
489 voltage_loops.each { |name| lines << "voltage source loop at #{name}" }
490 if lines.empty?
491 matrix, rhs, = equations
492 lines << 'singular matrix' unless solve(matrix, rhs)
493 end
494 lines.empty? ? ['ok'] : lines
495 when '.ALTER'
496 item = element(args[0])
497 value = number(args[1])
498 raise CircuitError, "nonpositive resistance #{item.name}" if item.type == 'R' && value <= 0
499 line = "altered #{item.name}: #{fraction(item.value)} -> #{fraction(value)}"
500 item.value = value
501 [line]
502 when '.DELETE'
503 item = element(args[0])
504 user = @elements.find { |other| %w[F H].include?(other.type) && other.reference == item.name }
505 raise CircuitError, "#{item.name} is referenced by #{user.name}" if user
506 @elements.delete(item)
507 ["deleted #{item.name}"]
508 when '.CLEAR'
509 line = "cleared #{@elements.length} elements"
510 @elements.clear
511 [line]
512 when '.NODES'
513 return ['no nodes'] if @elements.empty?
514 @elements.flat_map(&:nodes).uniq.sort.map do |name|
515 "#{name}: #{@elements.select { |item| item.nodes.include?(name) }.map(&:name).join(' ')}"
516 end
517 when '.PRINT'
518 return ['empty netlist'] if @elements.empty?
519 @elements.map { |item| ([item.name] + item.nodes + [item.reference, fraction(item.value)].compact).join(' ') }
520 end
521 end
522
523 def run
524 STDIN.each_line.with_index(1) do |input, number|
525 line = input.delete_suffix("\n")
526 break if line == '.END'
527 next if line.start_with?('*')
528 begin
529 output = line.start_with?('.') ? command(line.split(' ')) : add(line.split(' '))
530 output.each { |text| puts text }
531 rescue CircuitError => error
532 puts "error: line #{number}: #{error.message}"
533 end
534 end
535 end
536end
