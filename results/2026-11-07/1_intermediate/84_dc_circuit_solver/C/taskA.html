18static long long absolute(long long x) {
19 return x < 0 ? -x : x;
20}
21
22static Rational rational(long long numerator, long long denominator) {
23 if (denominator < 0) {
24 numerator = -numerator;
25 denominator = -denominator;
26 }
27 long long a = absolute(numerator), b = denominator;
28 while (b) {
29 long long next = a % b;
30 a = b;
31 b = next;
32 }
33 return (Rational){numerator / a, denominator / a};
34}

36static Rational add(Rational a, Rational b) {
37 return rational(a.numerator * b.denominator + b.numerator * a.denominator, a.denominator * b.denominator);
38}

40static Rational negate(Rational a) {
41 return (Rational){-a.numerator, a.denominator};
42}
43
44static Rational subtract(Rational a, Rational b) {
45 return add(a, negate(b));
46}

48static Rational multiply(Rational a, Rational b) {
49 return rational(a.numerator * b.numerator, a.denominator * b.denominator);
50}
51
52static Rational divide(Rational a, Rational b) {
53 return rational(a.numerator * b.denominator, a.denominator * b.numerator);
54}

56static int compare(Rational a, Rational b) {
57 long long x = a.numerator * b.denominator, y = b.numerator * a.denominator;
58 return (x > y) - (x < y);
59}

70}

105static Rational number(const char *original) {
106 char text[101];
107 strcpy(text, original);
108 Rational multiplier = ONE;
109 size_t length = strlen(text);
110 if (ends(text, "meg")) {
111 text[length - 3] = '\0';
112 multiplier = rational(1000000, 1);
113 } else if (ends(text, "k")) {
114 text[length - 1] = '\0';
115 multiplier = rational(1000, 1);
116 } else if (ends(text, "m")) {
117 text[length - 1] = '\0';
118 multiplier = rational(1, 1000);
119 } else if (ends(text, "u")) {
120 text[length - 1] = '\0';
121 multiplier = rational(1, 1000000);
122 }
123 char *mantissa = text;
124 bool negative = *mantissa == '-';
125 if (negative) mantissa++;
126 Rational result = ZERO;
127 bool valid = false;
128 if (digits(mantissa)) {
129 result = rational(strtoll(mantissa, NULL, 10), 1);
130 valid = true;
131 } else {
132 char *separator = strpbrk(mantissa, "./");
133 if (separator) {
134 char kind = *separator;
135 *separator = '\0';
136 const char *tail = separator + 1;
137 if (digits(mantissa) && digits(tail)) {
138 long long left = strtoll(mantissa, NULL, 10), right = strtoll(tail, NULL, 10);
139 if (kind == '/') {
140 if (right) {
141 result = rational(left, right);
142 valid = true;
143 }
144 } else {
145 long long denominator = 1;
146 for (const char *p = tail; *p; p++) denominator *= 10;
147 result = rational(left * denominator + right, denominator);
148 valid = true;
149 }
150 }
151 }
152 }
153 if (!valid) fail("invalid value %s", original);
154 result = multiply(result, multiplier);
155 return negative ? negate(result) : result;
156}

204static Nodes nodes(bool include_ground, bool force_ground) {

207 for (int i = 0; i < element_count; i++) {

210 add_node(&result, e->n2);
211 if (*e->c1) {

213 add_node(&result, e->c2);

215 }
216 if (!include_ground) for (int i = 0; i < result.count; i++) if (equal(result.names[i], "0")) {
217 memmove(result.names + i, result.names + i + 1, (size_t)(result.count - i - 1) * sizeof result.names[0]);
218 result.count--;
219 break;
220 }
221 qsort(result.names, (size_t)result.count, sizeof result.names[0], node_compare);

236static int root(int *parents, int i) {
237 while (parents[i] != i) {
238 parents[i] = parents[parents[i]];
239 i = parents[i];
240 }
241 return i;

244static bool join(int *parents, int a, int b) {
245 a = root(parents, a);
246 b = root(parents, b);
247 if (a == b) return false;
248 if (a < b) parents[b] = a;
249 else parents[a] = b;
250 return true;
251}

254 result[start] = true;
255 bool changed;
256 do {
257 changed = false;
258 for (int i = 0; i < count; i++) if (result[i]) for (int j = 0; j < count; j++) if (graph[i][j] && !result[j]) {
259 result[j] = true;
260 changed = true;
261 }
262 }
263 while (changed);
264}

303static bool solve(const Matrix *matrix, const Rational *column, int count, Rational *solution) {
304 Rational rows[MAX_UNKNOWNS][MAX_UNKNOWNS + 1];
305 for (int i = 0; i < count; i++) {
306 for (int j = 0; j < count; j++) rows[i][j] = matrix->cells[i][j];
307 rows[i][count] = column[i];
308 }
309 for (int k = 0; k < count; k++) {
310 int pivot = k;
311 while (pivot < count && !rows[pivot][k].numerator) pivot++;
312 if (pivot == count) return false;
313 if (pivot != k) for (int j = 0; j <= count; j++) {
314 Rational swap = rows[k][j];
315 rows[k][j] = rows[pivot][j];
316 rows[pivot][j] = swap;
317 }
318 Rational value = rows[k][k];
319 for (int j = 0; j <= count; j++) rows[k][j] = divide(rows[k][j], value);
320 for (int i = 0; i < count; i++) if (i != k && rows[i][k].numerator) {
321 Rational factor = rows[i][k];
322 for (int j = 0; j <= count; j++) rows[i][j] = subtract(rows[i][j], multiply(factor, rows[k][j]));
323 }
324 }
325 for (int i = 0; i < count; i++) solution[i] = rows[i][count];
326 return true;
327}

352static void build(Model *m, int changed, Rational replacement) {
353 memset(m, 0, sizeof *m);
354 m->nodes = nodes(false, false);
355 for (int i = 0; i < element_count; i++) if (voltage_type(&elements[i])) m->sources[m->source_count++] = i;
356 m->count = m->nodes.count + m->source_count;
357 zero_matrix(&m->a, m->count);
358 zero_column(m->b, m->count);
359 for (int i = 0; i < element_count; i++) {
360 Element *e = &elements[i];
361 Rational w = i == changed ? replacement : e->value;
362 switch (e->kind) {
363 case 'R':
364 {
365 Rational g = divide(ONE, w);
366 matrix_add(m, e->n1, e->n1, g);
367 matrix_add(m, e->n2, e->n2, g);
368 matrix_add(m, e->n1, e->n2, negate(g));
369 matrix_add(m, e->n2, e->n1, negate(g));
370 break;
371 }
372 case 'I':
373 column_add(m, m->b, e->n1, negate(w));
374 column_add(m, m->b, e->n2, w);
375 break;
376 case 'G':
377 matrix_add(m, e->n1, e->c1, w);
378 matrix_add(m, e->n2, e->c2, w);
379 matrix_add(m, e->n1, e->c2, negate(w));
380 matrix_add(m, e->n2, e->c1, negate(w));
381 break;
382 case 'F':
383 matrix_add(m, e->n1, e->reference, w);
384 matrix_add(m, e->n2, e->reference, negate(w));
385 break;
386 default:
387 matrix_add(m, e->n1, e->name, ONE);
388 matrix_add(m, e->name, e->n1, ONE);
389 matrix_add(m, e->n2, e->name, negate(ONE));
390 matrix_add(m, e->name, e->n2, negate(ONE));
391 if (e->kind == 'V') column_add(m, m->b, e->name, w);
392 else if (e->kind == 'E') {
393 matrix_add(m, e->name, e->c1, negate(w));
394 matrix_add(m, e->name, e->c2, w);
395 } else matrix_add(m, e->name, e->reference, negate(w));
396 break;
397 }
398 }
399}

409static Rational current(const Model *m, const Rational *solution, const Element *e) {
410 switch (e->kind) {
411 case 'R':
412 return divide(difference(m, solution, e->n1, e->n2), e->value);
413 case 'I':
414 return e->value;
415 case 'G':
416 return multiply(e->value, difference(m, solution, e->c1, e->c2));
417 case 'F':
418 return multiply(e->value, solution[model_index(m, e->reference)]);
419 default:
420 return solution[model_index(m, e->name)];
421 }
422}

425 Rational rhs[MAX_UNKNOWNS];
426 zero_column(rhs, m->count);
427 column_add(m, rhs, a, ONE);

432static void alone(const Model *m, const Element *source, bool unit, Rational *solution) {
433 Rational rhs[MAX_UNKNOWNS];
434 zero_column(rhs, m->count);
435 Rational q = unit ? ONE : source->value;
436 if (source->kind == 'V') column_add(m, rhs, source->name, q);
437 else {
438 column_add(m, rhs, source->n1, negate(q));
439 column_add(m, rhs, source->n2, q);
440 }
441 solve(&m->a, rhs, m->count, solution);
442}

444static void add_element(char **w, int argc) {
445 const char *name = w[0];
446 char kind = *name;
447 if (!strchr("RVIEGFH", kind)) fail("unknown element type %s", name);
448 int count = kind == 'E' || kind == 'G' ? 6 : kind == 'F' || kind == 'H' ? 5 : 4;
449 if (argc != count) fail("wrong field count for %s", name);
450 if (find_element(name) >= 0) fail("duplicate element %s", name);
451 for (int i = 1; i <= (count == 6 ? 4 : 2); i++) if (!valid_node(w[i])) fail("invalid node %s", w[i]);
452 if (kind == 'F' || kind == 'H') {
453 int source = find_element(w[3]);
454 if (source < 0 || elements[source].kind != 'V') fail("unknown voltage source %s", w[3]);
455 }
456 Rational q = number(w[count - 1]);
457 if (kind == 'R' && q.numerator <= 0) fail("nonpositive resistance %s", name);
458 Element e = {0};
459 strcpy(e.name, name);
460 e.kind = kind;
461 strcpy(e.n1, w[1]);
462 strcpy(e.n2, w[2]);
463 if (count == 6) {
464 strcpy(e.c1, w[3]);
465 strcpy(e.c2, w[4]);
466 } else if (count == 5) strcpy(e.reference, w[3]);
467 e.value = q;
468 elements[element_count++] = e;
469}

499static void resistance(const char *from, const char *to) {
500 Nodes all = nodes(true, true);
501 int parents[MAX_NODES];
502 for (int i = 0; i < all.count; i++) parents[i] = i;
503 for (int i = 0; i < element_count; i++) if (elements[i].kind == 'V') join(parents, node_index(&all, elements[i].n1), node_index(&all, elements[i].n2));
504 int a = root(parents, node_index(&all, from)), b = root(parents, node_index(&all, to));
505 if (a == b) {
506 output_value("Req", ZERO);
507 return;
508 }
509 bool graph[MAX_NODES][MAX_NODES] = {
510 {false}
511 }, connected[MAX_NODES] = {false};
512 for (int i = 0; i < element_count; i++) if (elements[i].kind == 'R') {
513 int p = root(parents, node_index(&all, elements[i].n1)), q = root(parents, node_index(&all, elements[i].n2));
514 if (p != q) graph[p][q] = graph[q][p] = true;
515 }
516 reached(graph, all.count, a, connected);
517 if (!connected[b]) {
518 puts("Req = INFINITE");
519 return;
520 }
521 int index[MAX_NODES], count = 0;
522 for (int i = 0; i < all.count; i++) index[i] = connected[i] && i != b ? count++ : -1;
523 Matrix g;
524 Rational rhs[MAX_UNKNOWNS], solution[MAX_UNKNOWNS];
525 zero_matrix(&g, count);
526 zero_column(rhs, count);
527 rhs[index[a]] = ONE;
528 for (int i = 0; i < element_count; i++) if (elements[i].kind == 'R') {
529 int p = root(parents, node_index(&all, elements[i].n1)), q = root(parents, node_index(&all, elements[i].n2));
530 if (p == q || !connected[p]) continue;
531 Rational value = divide(ONE, elements[i].value);
532 p = index[p];
533 q = index[q];
534 if (p >= 0) g.cells[p][p] = add(g.cells[p][p], value);
535 if (q >= 0) g.cells[q][q] = add(g.cells[q][q], value);
536 if (p >= 0 && q >= 0) {
537 g.cells[p][q] = subtract(g.cells[p][q], value);
538 g.cells[q][p] = subtract(g.cells[q][p], value);
539 }
540 }
541 solve(&g, rhs, count, solution);
542 output_value("Req", solution[index[a]]);
543}

550static void reduce(const char *from, const char *to) {
551 if (equal(from, to)) {
552 output_value("Req", ZERO);
553 return;
554 }
555 Nodes all = nodes(true, true);
556 bool graph[MAX_NODES][MAX_NODES] = {
557 {false}
558 }, connected[MAX_NODES] = {false};
559 for (int i = 0; i < element_count; i++) if (elements[i].kind == 'R') {
560 int a = node_index(&all, elements[i].n1), b = node_index(&all, elements[i].n2);
561 graph[a][b] = graph[b][a] = true;
562 }
563 reached(graph, all.count, node_index(&all, from), connected);
564 if (!connected[node_index(&all, to)]) {
565 puts("Req = INFINITE");
566 return;
567 }
568 Element list[MAX_ELEMENTS];
569 int count = 0, sequence = 0;
570 for (int i = 0; i < element_count; i++) if (elements[i].kind == 'R' && connected[node_index(&all, elements[i].n1)]) list[count++] = elements[i];
571 while (true) {
572 int removed = -1;
573 for (int i = 0; i < count; i++) if (equal(list[i].n1, list[i].n2)) {
574 removed = i;
575 break;
576 }
577 if (removed >= 0) {
578 printf("remove %s\n", list[removed].name);
579 remove_resistor(list, &count, removed);
580 continue;
581 }
582 int degrees[MAX_NODES] = {0};
583 for (int i = 0; i < count; i++) {
584 degrees[node_index(&all, list[i].n1)]++;
585 degrees[node_index(&all, list[i].n2)]++;
586 }
587 int leaf = -1;
588 for (int i = 0; i < all.count; i++) if (!equal(all.names[i], from) && !equal(all.names[i], to) && degrees[i] == 1) {
589 leaf = i;
590 break;
591 }
592 if (leaf >= 0) {
593 for (int i = 0; i < count; i++) if (equal(list[i].n1, all.names[leaf]) || equal(list[i].n2, all.names[leaf])) {
594 printf("remove %s\n", list[i].name);
595 remove_resistor(list, &count, i);
596 break;
597 }
598 continue;
599 }
600 int x = -1, y = -1;
601 bool parallel = false;
602 for (int i = 0; i < count && x < 0; i++) for (int j = i + 1; j < count; j++) if ((equal(list[i].n1, list[j].n1) && equal(list[i].n2, list[j].n2)) || (equal(list[i].n1, list[j].n2) && equal(list[i].n2, list[j].n1))) {
603 x = i;
604 y = j;
605 parallel = true;
606 break;
607 }
608 int middle = -1;
609 if (x < 0) {
610 for (int i = 0; i < all.count; i++) if (!equal(all.names[i], from) && !equal(all.names[i], to) && degrees[i] == 2) {
611 middle = i;
612 break;
613 }
614 if (middle < 0) break;
615 for (int i = 0; i < count; i++) if (equal(list[i].n1, all.names[middle]) || equal(list[i].n2, all.names[middle])) {
616 if (x < 0) x = i;
617 else y = i;
618 }
619 }
620 Element a = list[x], b = list[y], next = {0};
621 snprintf(next.name, sizeof next.name, "#%d", ++sequence);
622 next.kind = 'R';
623 next.value = parallel ? divide(multiply(a.value, b.value), add(a.value, b.value)) : add(a.value, b.value);
624 strcpy(next.n1, parallel ? a.n1 : equal(a.n1, all.names[middle]) ? a.n2 : a.n1);
625 strcpy(next.n2, parallel ? a.n2 : equal(b.n1, all.names[middle]) ? b.n2 : b.n1);
626 remove_resistor(list, &count, y);
627 remove_resistor(list, &count, x);
628 list[count++] = next;
629 printf("%s = %s %s %s = ", next.name, a.name, parallel ? "||" : "+", b.name);
630 value(next.value);
631 putchar('\n');
632 }
633 if (count == 1) output_value("Req", list[0].value);
634 else printf("not series-parallel: %d resistors remain\n", count);
635}

637static void command(char **w, int argc) {
638 const char *c = w[0];
639 int arity = -1;
640 for (size_t i = 0; i < sizeof commands / sizeof commands[0]; i++) if (equal(c, commands[i].name)) arity = commands[i].arity;
641 if (arity < 0) fail("unknown command %s", c);
642 if (argc - 1 != arity) fail("wrong argument count for %s", c);
643 if (equal(c, ".ALTER")) {
644 int index = get_element(w[1]);
645 Rational next = number(w[2]);
646 Element *e = &elements[index];
647 if (e->kind == 'R' && next.numerator <= 0) fail("nonpositive resistance %s", e->name);
648 printf("altered %s: ", e->name);
649 fraction(e->value);
650 fputs(" -> ", stdout);
651 fraction(next);
652 putchar('\n');
653 e->value = next;
654 return;
655 }
656 if (equal(c, ".DELETE")) {
657 int index = get_element(w[1]);
658 for (int i = 0; i < element_count; i++) if (equal(elements[i].reference, w[1])) fail("%s is referenced by %s", w[1], elements[i].name);
659 printf("deleted %s\n", elements[index].name);
660 memmove(elements + index, elements + index + 1, (size_t)(element_count - index - 1) * sizeof(Element));
661 element_count--;
662 return;
663 }
664 if (equal(c, ".CLEAR")) {
665 printf("cleared %d elements\n", element_count);
666 element_count = 0;
667 return;
668 }
669 if (equal(c, ".PRINT")) {
670 if (!element_count) puts("empty netlist");
671 for (int i = 0; i < element_count; i++) {
672 Element *e = &elements[i];
673 printf("%s %s %s", e->name, e->n1, e->n2);
674 if (*e->c1) printf(" %s %s", e->c1, e->c2);
675 else if (*e->reference) printf(" %s", e->reference);
676 putchar(' ');
677 fraction(e->value);
678 putchar('\n');
679 }
680 return;
681 }
682 if (equal(c, ".NODES")) {
683 Nodes all = nodes(true, false);
684 if (!all.count) puts("no nodes");
685 for (int n = 0; n < all.count; n++) {
686 printf("%s:", all.names[n]);
687 for (int i = 0; i < element_count; i++) {
688 Element *e = &elements[i];
689 if (equal(e->n1, all.names[n]) || equal(e->n2, all.names[n]) || equal(e->c1, all.names[n]) || equal(e->c2, all.names[n])) printf(" %s", e->name);
690 }
691 putchar('\n');
692 }
693 return;
694 }
695 if (equal(c, ".REQ") || equal(c, ".REDUCE")) {
696 check_node(w[1]);
697 check_node(w[2]);
698 if (equal(c, ".REQ")) resistance(w[1], w[2]);
699 else reduce(w[1], w[2]);
700 return;
701 }
702 Model model;
703 if (equal(c, ".CHECK")) {
704 Nodes actual = nodes(false, false);
705 if (!actual.count) {
706 puts("no nodes");
707 return;
708 }
709 bool any = false;
710 for (int n = 0; n < actual.count; n++) {
711 int degree = 0;
712 for (int i = 0; i < element_count; i++) {
713 degree += equal(elements[i].n1, actual.names[n]);
714 degree += equal(elements[i].n2, actual.names[n]);
715 }
716 if (degree < 2) {
717 printf("dangling node %s\n", actual.names[n]);
718 any = true;
719 }
720 }
721 Nodes all = nodes(true, true);
722 bool connected[MAX_NODES] = {false};
723 floating(&all, connected);
724 for (int i = 0; i < all.count; i++) if (!connected[i] && !equal(all.names[i], "0")) {
725 printf("floating node %s\n", all.names[i]);
726 any = true;
727 }
728 int loops[MAX_ELEMENTS], count = source_loops(&all, loops);
729 for (int i = 0; i < count; i++) {
730 printf("voltage source loop at %s\n", elements[loops[i]].name);
731 any = true;
732 }
733 if (!any) {
734 Rational solution[MAX_UNKNOWNS];
735 build(&model, -1, ZERO);
736 if (!solve(&model.a, model.b, model.count, solution)) puts("singular matrix");
737 else puts("ok");
738 }
739 return;
740 }
741 if (equal(c, ".SWEEP")) {
742 int index = get_element(w[1]);
743 Rational from = number(w[2]);
744 Rational to = number(w[3]);
745 Rational step = number(w[4]);
746 check_node(w[5]);
747 if (step.numerator <= 0) fail("invalid sweep step");
748 if (compare(from, to) > 0) fail("invalid sweep range");
749 Rational span = divide(subtract(to, from), step);
750 long long count = span.numerator / span.denominator + 1;
751 if (count > 20) fail("too many sweep points");
752 if (elements[index].kind == 'R' && from.numerator <= 0) fail("nonpositive resistance %s", elements[index].name);
753 structure();
754 for (int k = 0; k < count; k++) {
755 Rational q = add(from, multiply(step, rational(k, 1))), solution[MAX_UNKNOWNS];
756 build(&model, index, q);
757 bool valid = solve(&model.a, model.b, model.count, solution);
758 printf("%s = ", elements[index].name);
759 fraction(q);
760 if (!valid) puts(": singular matrix");
761 else {
762 printf(": V(%s) = ", w[5]);
763 value(potential(&model, solution, w[5]));
764 putchar('\n');
765 }
766 }
767 return;
768 }
769 int selected = -1;
770 if (equal(c, ".CURRENT")) selected = get_element(w[1]);
771 else if (equal(c, ".VOLTAGE") || equal(c, ".THEVENIN") || equal(c, ".NORTON") || equal(c, ".MAXPOWER")) {
772 check_node(w[1]);
773 check_node(w[2]);
774 } else if (equal(c, ".KCL") || equal(c, ".SENS") || equal(c, ".SUPERPOSITION")) check_node(w[1]);
775 else if (equal(c, ".ZPARAM")) for (int i = 1; i <= 4; i++) check_node(w[i]);
776 else if (equal(c, ".TF")) {
777 check_node(w[1]);
778 selected = get_element(w[2]);
779 if (!independent(&elements[selected])) fail("not an independent source %s", elements[selected].name);
780 }
781 structure();
782 build(&model, -1, ZERO);
783 if (equal(c, ".MATRIX")) {
784 fputs("unknowns:", stdout);
785 for (int i = 0; i < model.nodes.count; i++) printf(" V(%s)", model.nodes.names[i]);
786 for (int i = 0; i < model.source_count; i++) printf(" I(%s)", elements[model.sources[i]].name);
787 putchar('\n');
788 for (int i = 0; i < model.count; i++) {
789 printf("%s:", i < model.nodes.count ? model.nodes.names[i] : elements[model.sources[i - model.nodes.count]].name);
790 for (int j = 0; j < model.count; j++) {
791 putchar(' ');
792 fraction(model.a.cells[i][j]);
793 }
794 fputs(" | ", stdout);
795 fraction(model.b[i]);
796 putchar('\n');
797 }
798 return;
799 }
800 Rational point[MAX_UNKNOWNS];
801 if (!solve(&model.a, model.b, model.count, point)) fail("singular matrix");
802 if (equal(c, ".OP")) {
803 for (int i = 0; i < model.nodes.count; i++) {
804 printf("V(%s) = ", model.nodes.names[i]);
805 value(point[i]);
806 putchar('\n');
807 }
808 for (int i = 0; i < model.source_count; i++) {
809 printf("I(%s) = ", elements[model.sources[i]].name);
810 value(point[model.nodes.count + i]);
811 putchar('\n');
812 }
813 } else if (equal(c, ".CURRENT")) {
814 printf("I(%s) = ", elements[selected].name);
815 value(current(&model, point, &elements[selected]));
816 putchar('\n');
817 } else if (equal(c, ".VOLTAGE")) {
818 printf("V(%s,%s) = ", w[1], w[2]);
819 value(difference(&model, point, w[1], w[2]));
820 putchar('\n');
821 } else if (equal(c, ".POWER")) {
822 Rational absorbed = ZERO, delivered = ZERO, total = ZERO;
823 for (int i = 0; i < element_count; i++) {
824 Element *e = &elements[i];
825 Rational power = multiply(difference(&model, point, e->n1, e->n2), current(&model, point, e));
826 printf("P(%s) = ", e->name);
827 value(power);
828 putchar('\n');
829 if (power.numerator > 0) absorbed = add(absorbed, power);
830 if (power.numerator < 0) delivered = subtract(delivered, power);
831 total = add(total, power);
832 }
833 output_value("absorbed", absorbed);
834 output_value("delivered", delivered);
835 output_value("total", total);
836 } else if (equal(c, ".KCL")) {
837 Rational sum = ZERO;
838 for (int i = 0; i < element_count; i++) {
839 Element *e = &elements[i];
840 if (equal(e->n1, w[1]) || equal(e->n2, w[1])) {
841 Rational q = equal(e->n1, e->n2) ? ZERO : equal(e->n1, w[1]) ? current(&model, point, e) : negate(current(&model, point, e));
842 printf("I(%s) leaving %s = ", e->name, w[1]);
843 value(q);
844 putchar('\n');
845 sum = add(sum, q);
846 }
847 }
848 output_value("sum", sum);
849 } else if (equal(c, ".THEVENIN") || equal(c, ".NORTON") || equal(c, ".MAXPOWER")) {
850 Rational voltage = difference(&model, point, w[1], w[2]), auxiliary[MAX_UNKNOWNS];
851 injection(&model, w[1], w[2], auxiliary);
852 Rational resistance = difference(&model, auxiliary, w[1], w[2]);
853 if (equal(c, ".NORTON") && !resistance.numerator) fail("zero thevenin resistance");
854 if (equal(c, ".MAXPOWER") && resistance.numerator <= 0) fail("nonpositive thevenin resistance");
855 if (equal(c, ".THEVENIN")) {
856 output_value("Vth", voltage);
857 output_value("Rth", resistance);
858 } else if (equal(c, ".NORTON")) {
859 output_value("In", divide(voltage, resistance));
860 output_value("Rn", resistance);
861 } else {
862 output_value("RL", resistance);
863 output_value("Pmax", divide(multiply(voltage, voltage), multiply(rational(4, 1), resistance)));
864 }
865 } else if (equal(c, ".TF")) {
866 Element *e = &elements[selected];
867 Rational unit[MAX_UNKNOWNS], auxiliary[MAX_UNKNOWNS];
868 alone(&model, e, true, unit);
869 output_value("gain", potential(&model, unit, w[1]));
870 if (e->kind == 'V') {
871 Rational q = unit[model_index(&model, e->name)];
872 if (!q.numerator) puts("input resistance = INFINITE");
873 else output_value("input resistance", divide(negate(ONE), q));
874 } else output_value("input resistance", difference(&model, unit, e->n2, e->n1));
875 injection(&model, w[1], "0", auxiliary);
876 output_value("output resistance", potential(&model, auxiliary, w[1]));
877 } else if (equal(c, ".SENS")) {
878 for (int i = 0; i < element_count; i++) {
879 Element *e = &elements[i];
880 Rational rhs[MAX_UNKNOWNS], auxiliary[MAX_UNKNOWNS];
881 zero_column(rhs, model.count);
882 switch (e->kind) {
883 case 'R':
884 {
885 Rational q = divide(current(&model, point, e), e->value);
886 column_add(&model, rhs, e->n1, q);
887 column_add(&model, rhs, e->n2, negate(q));
888 break;
889 }
890 case 'I':
891 column_add(&model, rhs, e->n1, negate(ONE));
892 column_add(&model, rhs, e->n2, ONE);
893 break;
894 case 'G':
895 case 'F':
896 {
897 Rational q = e->kind == 'G' ? difference(&model, point, e->c1, e->c2) : point[model_index(&model, e->reference)];
898 column_add(&model, rhs, e->n1, negate(q));
899 column_add(&model, rhs, e->n2, q);
900 break;
901 }
902 case 'V':
903 column_add(&model, rhs, e->name, ONE);
904 break;
905 case 'E':
906 column_add(&model, rhs, e->name, difference(&model, point, e->c1, e->c2));
907 break;
908 case 'H':
909 column_add(&model, rhs, e->name, point[model_index(&model, e->reference)]);
910 break;
911 }
912 solve(&model.a, rhs, model.count, auxiliary);
913 printf("dV(%s)/d%s = ", w[1], e->name);
914 value(potential(&model, auxiliary, w[1]));
915 putchar('\n');
916 }
917 } else if (equal(c, ".ZPARAM")) {
918 Rational first[MAX_UNKNOWNS], second[MAX_UNKNOWNS];
919 injection(&model, w[1], w[2], first);
920 injection(&model, w[3], w[4], second);
921 output_value("z11", difference(&model, first, w[1], w[2]));
922 output_value("z12", difference(&model, second, w[1], w[2]));
923 output_value("z21", difference(&model, first, w[3], w[4]));
924 output_value("z22", difference(&model, second, w[3], w[4]));
925 } else if (equal(c, ".SUPERPOSITION")) {
926 Rational sum = ZERO;
927 for (int i = 0; i < element_count; i++) if (independent(&elements[i])) {
928 Rational auxiliary[MAX_UNKNOWNS];
929 alone(&model, &elements[i], false, auxiliary);
930 Rational q = potential(&model, auxiliary, w[1]);
931 printf("%s: V(%s) = ", elements[i].name, w[1]);
932 value(q);
933 putchar('\n');
934 sum = add(sum, q);
935 }
936 printf("total: V(%s) = ", w[1]);
937 value(sum);
938 putchar('\n');
939 }
