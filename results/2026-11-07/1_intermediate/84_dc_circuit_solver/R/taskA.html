6positive_divmod <- function(a, b) {
7 q <- floor(a / b)
8 r <- a - q * b
9 if (r < 0) {
10 q <- q - 1
11 r <- r + b
12 } else if (r >= b) {
13 q <- q + 1
14 r <- r - b
15 }
16 c(q, r)
17}
18rational <- function(p = 0, q = 1) {
19 if (p == 0) return(c(0, 1))
20 if (q < 0) {
21 p <- -p
22 q <- -q
23 }
24 a <- abs(p)
25 b <- q
26 while (b != 0) {
27 remainder <- positive_divmod(a, b)[[2]]
28 a <- b
29 b <- remainder
30 }
31 c(p / a, q / a)
32}

49rational_number <- function(text) {
50 multiplier <- rational(1)
51 suffixes <- list(meg = rational(1000000), k = rational(1000), m = rational(1, 1000), u = rational(1, 1000000))
52 for (suffix in names(suffixes)) {
53 if (endsWith(text, suffix)) {
54 text <- substr(text, 1L, nchar(text) - nchar(suffix))
55 multiplier <- suffixes[[suffix]]
56 break
57 }
58 }
59 if (!grepl("^-?[0-9]+([.][0-9]+|/[0-9]+)?$", text)) return(NULL)
60 if (grepl("/", text, fixed = TRUE)) {
61 parts <- strsplit(text, "/", fixed = TRUE)[[1]]
62 q <- as.numeric(parts[[2]])
63 if (q == 0) return(NULL)
64 value <- rational(as.numeric(parts[[1]]), q)
65 } else if (grepl(".", text, fixed = TRUE)) {
66 parts <- strsplit(text, ".", fixed = TRUE)[[1]]
67 q <- 10 ^ nchar(parts[[2]])
68 p <- abs(as.numeric(parts[[1]])) * q + as.numeric(parts[[2]])
69 value <- rational(if (startsWith(text, "-")) -p else p, q)
70 } else {
71 value <- rational(as.numeric(text))
72 }
73 multiply(value, multiplier)
74}

76solve_system <- function(matrix, rhs) {
77 n <- length(rhs)
78 rows <- lapply(seq_len(n), function(i) c(matrix[[i]], list(rhs[[i]])))
79 for (k in seq_len(n)) {
80 pivot <- NULL
81 for (i in seq.int(k, n)) {
82 if (rows[[i]][[k]][[1]] != 0) {
83 pivot <- i
84 break
85 }
86 }
87 if (is.null(pivot)) return(NULL)
88 if (pivot != k) {
89 row <- rows[[k]]
90 rows[[k]] <- rows[[pivot]]
91 rows[[pivot]] <- row
92 }
93 divisor <- rows[[k]][[k]]
94 rows[[k]] <- lapply(rows[[k]], function(entry) divide(entry, divisor))
95 for (i in seq_len(n)) {
96 if (i == k || rows[[i]][[k]][[1]] == 0) next
97 factor <- rows[[i]][[k]]
98 for (j in seq_len(n + 1L)) rows[[i]][[j]] <- subtract(rows[[i]][[j]], multiply(factor, rows[[k]][[j]]))
99 }
100 }
101 lapply(rows, function(row) row[[n + 1L]])
102}
103reachable <- function(start, edges) {
104 seen <- start
105 queue <- start
106 while (length(queue) != 0L) {
107 node <- queue[[1]]
108 queue <- queue[-1L]
109 for (edge in edges) {
110 a <- edge[[1]]
111 b <- edge[[2]]
112 if (a != node && b != node) next
113 other <- if (a == node) b else a
114 if (!(other %in% seen)) {
115 seen <- c(seen, other)
116 queue <- c(queue, other)
117 }
118 }
119 }
120 seen
121}

256 a <- item$nodes[[1]]

261 I = item$value,

416 equivalent <- function(a, b) {
417 groups <- new_groups(c("0", nodes()))
418 for (item in state$elements) if (item$type == "V") groups$join(item$nodes[[1]], item$nodes[[2]])
419 a <- groups$find(a)
420 b <- groups$find(b)
421 if (a == b) return("Req = 0 (0.000000)")
422 edges <- list()
423 for (item in state$elements) {
424 if (item$type != "R") next
425 x <- groups$find(item$nodes[[1]])
426 y <- groups$find(item$nodes[[2]])
427 if (x != y) edges[[length(edges) + 1L]] <- list(x, y, divide(rational(1), item$value))
428 }
429 connected <- reachable(a, edges)
430 if (!(b %in% connected)) return("Req = INFINITE")
431 labels <- sort(setdiff(connected, b), method = "radix")
432 index <- setNames(as.list(seq_along(labels)), labels)
433 matrix <- replicate(length(labels), zeros(length(labels)), simplify = FALSE)
434 rhs <- zeros(length(labels))
435 rhs[[index[[a]]]] <- rational(1)
436 for (edge in edges) {
437 x <- edge[[1]]
438 y <- edge[[2]]
439 g <- edge[[3]]
440 if (!(x %in% connected)) next
441 if (x != b) matrix[[index[[x]]]][[index[[x]]]] <- add(matrix[[index[[x]]]][[index[[x]]]], g)
442 if (y != b) matrix[[index[[y]]]][[index[[y]]]] <- add(matrix[[index[[y]]]][[index[[y]]]], g)
443 if (x != b && y != b) {
444 matrix[[index[[x]]]][[index[[y]]]] <- subtract(matrix[[index[[x]]]][[index[[y]]]], g)
445 matrix[[index[[y]]]][[index[[x]]]] <- subtract(matrix[[index[[y]]]][[index[[x]]]], g)
446 }
447 }
448 paste0("Req = ", formatted(solve_system(matrix, rhs)[[index[[a]]]]))
449 }
450 reduce_resistors <- function(a, b) {
451 if (a == b) return("Req = 0 (0.000000)")
452 list <- Filter(function(item) item$type == "R", state$elements)
453 connected <- reachable(a, lapply(list, function(item) item$nodes))
454 if (!(b %in% connected)) return("Req = INFINITE")
455 list <- Filter(function(item) item$nodes[[1]] %in% connected, list)
456 lines <- character()
457 serial <- 0L
458 repeat {
459 self <- which(vapply(list, function(item) item$nodes[[1]] == item$nodes[[2]], logical(1)))
460 if (length(self) != 0L) {
461 lines <- c(lines, paste0("remove ", list[[self[[1]]]]$name))
462 list <- list[-self[[1]]]
463 next
464 }
465 ends <- unlist(lapply(list, function(item) item$nodes), use.names = FALSE)
466 names <- sort(unique(ends), method = "radix")
467 degrees <- setNames(vapply(names, function(name) sum(ends == name), integer(1)), names)
468 internal <- setdiff(names, c(a, b))
469 leaves <- internal[degrees[internal] == 1L]
470 if (length(leaves) != 0L) {
471 i <- which(vapply(list, function(item) leaves[[1]] %in% item$nodes, logical(1)))[[1]]
472 lines <- c(lines, paste0("remove ", list[[i]]$name))
473 list <- list[-i]
474 next
475 }
476 pair <- NULL
477 for (i in seq_along(list)) {
478 if (i == length(list)) next
479 for (j in seq.int(i + 1L, length(list))) {
480 if (identical(sort(list[[i]]$nodes, method = "radix"), sort(list[[j]]$nodes, method = "radix"))) {
481 pair <- c(i, j)
482 break
483 }
484 }
485 if (!is.null(pair)) break
486 }
487 if (!is.null(pair)) {
488 x <- list[[pair[[1]]]]
489 y <- list[[pair[[2]]]]
490 value <- divide(multiply(x$value, y$value), add(x$value, y$value))
491 terminals <- x$nodes
492 operator <- "||"
493 } else {
494 middle <- internal[degrees[internal] == 2L]
495 if (length(middle) == 0L) break
496 p <- middle[[1]]
497 pair <- which(vapply(list, function(item) p %in% item$nodes, logical(1)))
498 x <- list[[pair[[1]]]]
499 y <- list[[pair[[2]]]]
500 terminals <- c(x$nodes[x$nodes != p], y$nodes[y$nodes != p])
501 value <- add(x$value, y$value)
502 operator <- "+"
503 }
504 serial <- serial + 1L
505 name <- paste0("#", serial)
506 lines <- c(lines, paste0(name, " = ", x$name, " ", operator, " ", y$name, " = ", formatted(value)))
507 list <- list[-pair]
508 list[[length(list) + 1L]] <- list(name = name, type = "R", nodes = terminals, value = value)
509 }
510 c(lines, if (length(list) == 1L) paste0("Req = ", formatted(list[[1]]$value)) else paste0("not series-parallel: ", length(list), " resistors remain"))
511 }

516 if (length(args) != counts[[cmd]]) circuit_error(paste0("wrong argument count for ", cmd))
517 if (cmd %in% analyses) return(analysis(cmd, args))
518 if (cmd %in% c(".REQ", ".REDUCE")) {
519 for (name in args) node(name)
520 return(if (cmd == ".REQ") equivalent(args[[1]], args[[2]]) else reduce_resistors(args[[1]], args[[2]]))
521 }
522 if (cmd == ".SWEEP") {
523 item <- element(args[[1]])
524 from <- number(args[[2]])
525 to <- number(args[[3]])
526 step <- number(args[[4]])
527 node(args[[5]])
528 if (step[[1]] <= 0) circuit_error("invalid sweep step")
529 if (compare(from, to) > 0) circuit_error("invalid sweep range")
530 range <- divide(subtract(to, from), step)
531 count <- positive_divmod(range[[1]], range[[2]])[[1]] + 1L
532 if (count > 20L) circuit_error("too many sweep points")
533 if (item$type == "R" && from[[1]] <= 0) circuit_error(paste0("nonpositive resistance ", item$name))
534 structure_check()
535 lines <- character()
536 for (k in seq_len(count) - 1L) {
537 x <- add(from, multiply(rational(k), step))
538 system <- equations(setNames(list(x), item$name))
539 solution <- solve_system(system$matrix, system$rhs)
540 text <- if (is.null(solution)) "singular matrix" else paste0("V(", args[[5]], ") = ", formatted(potential(solution, system$index, args[[5]])))
541 lines <- c(lines, paste0(item$name, " = ", fraction(x), ": ", text))
542 }
543 return(lines)
544 }
545 if (cmd == ".MATRIX") {
546 structure_check()
547 system <- equations()
548 unknowns <- c(paste0("V(", nodes(), ")"), vapply(sources(), function(item) paste0("I(", item$name, ")"), character(1)))
549 lines <- paste0("unknowns: ", paste(unknowns, collapse = " "))
550 for (i in seq_along(system$matrix)) lines <- c(lines, paste0(system$labels[[i]], ": ", paste(vapply(system$matrix[[i]], fraction, character(1)), collapse = " "), " | ", fraction(system$rhs[[i]])))
551 return(lines)
552 }
553 if (cmd == ".CHECK") {
554 if (length(nodes()) == 0L) return("no nodes")
555 lines <- character()
556 for (name in nodes()) {
557 degree <- sum(vapply(state$elements, function(item) sum(item$nodes[1:2] == name), integer(1)))
558 if (degree < 2L) lines <- c(lines, paste0("dangling node ", name))
559 }
560 grounded <- grounding()
561 for (name in nodes()) if (!(name %in% grounded)) lines <- c(lines, paste0("floating node ", name))
562 for (name in voltage_loops()) lines <- c(lines, paste0("voltage source loop at ", name))
563 if (length(lines) == 0L) {
564 system <- equations()
565 if (is.null(solve_system(system$matrix, system$rhs))) lines <- "singular matrix"
566 }
567 return(if (length(lines) == 0L) "ok" else lines)
568 }
569 if (cmd == ".ALTER") {
570 i <- element_index(args[[1]])
571 item <- state$elements[[i]]
572 value <- number(args[[2]])
573 if (item$type == "R" && value[[1]] <= 0) circuit_error(paste0("nonpositive resistance ", item$name))
574 line <- paste0("altered ", item$name, ": ", fraction(item$value), " -> ", fraction(value))
575 state$elements[[i]]$value <- value
576 return(line)
577 }
578 if (cmd == ".DELETE") {
579 i <- element_index(args[[1]])
580 item <- state$elements[[i]]
581 for (other in state$elements) if (other$type %in% c("F", "H") && other$reference == item$name) circuit_error(paste0(item$name, " is referenced by ", other$name))
582 state$elements <- state$elements[-i]
583 return(paste0("deleted ", item$name))
584 }
585 if (cmd == ".CLEAR") {
586 line <- paste0("cleared ", length(state$elements), " elements")
587 state$elements <- list()
588 return(line)
589 }
590 if (cmd == ".NODES") {
591 if (length(state$elements) == 0L) return("no nodes")
592 names <- sort(unique(unlist(lapply(state$elements, function(item) item$nodes), use.names = FALSE)), method = "radix")
593 return(vapply(names, function(name) {
594 users <- vapply(Filter(function(item) name %in% item$nodes, state$elements), function(item) item$name, character(1))
595 paste0(name, ": ", paste(users, collapse = " "))
596 }, character(1)))
597 }
598 if (length(state$elements) == 0L) return("empty netlist")
599 vapply(state$elements, function(item) paste(c(item$name, item$nodes, item$reference, fraction(item$value)), collapse = " "), character(1))
600 }
601 input <- file("stdin", open = "r")
602 on.exit(close(input))
603 number_line <- 0L
604 repeat {
605 lines <- readLines(input, n = 1L, warn = FALSE)
606 if (length(lines) == 0L || lines[[1]] == ".END") break
607 number_line <- number_line + 1L
608 line <- lines[[1]]
609 if (startsWith(line, "*")) next
610 output <- tryCatch({
611 words <- strsplit(line, " ", fixed = TRUE)[[1]]
612 if (startsWith(line, ".")) command(words) else add_element(words)
613 }, circuit_error = function(error) paste0("error: line ", number_line, ": ", error$message))
614 for (text in output) cat(text, "\n", sep = "")
615 }
616}
