98function digits(value: bigint, radices: number[]): number[] {
99 const result = new Array<number>(radices.length);
100 for (let i = radices.length - 1; i >= 0; i--) {
101 result[i] = Number(value % BigInt(radices[i]));
102 value /= BigInt(radices[i]);
103 }
104 return result;

115function encode(cell: Cell): string | null {
116 let { x, y } = cell;
117 const p = cell.p;
118 if (cell.system === "GEOHASH") {
119 const a = Math.floor((5 * p + 1) / 2);
120 const b = Math.floor(5 * p / 2);
121 const xs = x.toString(2).padStart(a, "0");
122 const ys = y.toString(2).padStart(b, "0");
123 let bits = "";
124 for (let i = 0; i < 5 * p; i++) {
125 bits += i % 2 === 0 ? xs[Math.floor(i / 2)] : ys[Math.floor(i / 2)];
126 }
127 let code = "";
128 for (let i = 0; i < bits.length; i += 5) {
129 code += geohash[parseInt(bits.slice(i, i + 5), 2)];
130 }
131 return code;
132 }
133 if (cell.system === "MAIDENHEAD") {
134 const count = p / 2;
135 const radices = [18, 10, 24, 10].slice(0, count);
136 const xs = digits(x, radices);
137 const ys = digits(y, radices);
138 return xs.map((value, i) => maiden[i][value] + maiden[i][ys[i]]).join("");
139 }
140 if (cell.system === "PLUSCODE") {
141 let tail = "";
142 if (p === 11) {
143 tail = plus[Number(4n * (y % 5n) + x % 4n)];
144 x /= 4n;
145 y /= 5n;
146 }
147 const count = Math.min(p, 10) / 2;
148 const xs = digits(x, [18, ...new Array(count - 1).fill(20)]);
149 const ys = digits(y, [9, ...new Array(count - 1).fill(20)]);
150 const code = xs.map((value, i) => plus[ys[i]] + plus[value]).join("") + tail;
151 return p >= 8 ? code.slice(0, 8) + "+" + code.slice(8) : code + "0".repeat(8 - p) + "+";
152 }
153 if (cell.system === "GARS") {
154 let suffix = "";
155 if (p === 7) {
156 suffix = String(1n + x % 3n + 3n * (2n - y % 3n));
157 x /= 3n;
158 y /= 3n;
159 }
160 if (p >= 6) {
161 suffix = String(1n + x % 2n + 2n * (1n - y % 2n)) + suffix;
162 x /= 2n;
163 y /= 2n;
164 }
165 return String(x + 1n).padStart(3, "0") + l24[Number(y / 24n)] + l24[Number(y % 24n)] + suffix;
166 }
167 if (cell.system === "GEOREF") {
168 let tail = "";
169 if (p >= 8) {
170 const m = (p - 4) / 2;
171 const radix = BigInt(6 * 10 ** (m - 1));
172 tail = String(x % radix).padStart(m, "0") + String(y % radix).padStart(m, "0");
173 x /= radix;
174 y /= radix;
175 }
176 if (p === 2) {
177 return l24[Number(x)] + l24[Number(y)];
178 }
179 return l24[Number(x / 15n)] + l24[Number(y / 15n)] + l24[Number(x % 15n)] + l24[Number(y % 15n)] + tail;
180 }
181 const index = sequences.MESH.indexOf(p);
182 const scale = BigInt([1, 8, 80, 160, 320, 640][index]);
183 const u = x - 280n * scale;
184 const v = y - 135n * scale;
185 if (u < 0n || u >= 80n * scale || v < 0n || v >= 100n * scale) {
186 return null;
187 }
188 const xs = digits(u, [80, 8, 10, 2, 2, 2].slice(0, index + 1));
189 const ys = digits(v, [100, 8, 10, 2, 2, 2].slice(0, index + 1));
190 let code = String(ys[0]).padStart(2, "0") + String(xs[0]).padStart(2, "0");
191 for (let i = 1; i < xs.length; i++) {
192 code += i <= 2 ? String(ys[i]) + xs[i] : String(1 + xs[i] + 2 * ys[i]);
193 }
194 return code;
195}

197function decode(system: string, code: string): Cell {
198 let p = code.length;
199 let x = 0n;
200 let y = 0n;
201 if (system === "GEOHASH") {
202 if (p < 1 || p > 9) {
203 invalid();
204 }
205 const bits = code.split("").map(c => character(geohash, c).toString(2).padStart(5, "0")).join("");
206 for (let i = 0; i < bits.length; i++) {
207 if (i % 2 === 0) {
208 x = 2n * x + BigInt(bits[i]);
209 } else {
210 y = 2n * y + BigInt(bits[i]);
211 }
212 }
213 } else if (system === "MAIDENHEAD") {
214 if (!sequences[system].includes(p)) {
215 invalid();
216 }
217 for (let i = 0; i < p / 2; i++) {
218 const radix = BigInt(maiden[i].length);
219 x = x * radix + BigInt(character(maiden[i], code[2 * i]));
220 y = y * radix + BigInt(character(maiden[i], code[2 * i + 1]));
221 }
222 } else if (system === "PLUSCODE") {
223 const parts = code.split("+");
224 if (parts.length !== 2 || parts[0].length !== 8) {
225 invalid();
226 }
227 const zero = parts[0].indexOf("0");
228 let raw = "";
229 if (zero >= 0) {
230 if (![2, 4, 6].includes(zero) || parts[0].slice(zero) !== "0".repeat(8 - zero) || parts[1] !== "") {
231 invalid();
232 }
233 raw = parts[0].slice(0, zero);
234 } else {
235 if (![0, 2, 3].includes(parts[1].length)) {
236 invalid();
237 }
238 raw = parts[0] + parts[1];
239 }
240 const values = raw.split("").map(c => character(plus, c));
241 if (values[0] > 8 || values[1] > 17) {
242 invalid();
243 }
244 p = values.length;
245 for (let i = 0; i < Math.min(p, 10); i += 2) {
246 y = y * BigInt(i === 0 ? 9 : 20) + BigInt(values[i]);
247 x = x * BigInt(i === 0 ? 18 : 20) + BigInt(values[i + 1]);
248 }
249 if (p === 11) {
250 x = 4n * x + BigInt(values[10] % 4);
251 y = 5n * y + BigInt(Math.floor(values[10] / 4));
252 }
253 } else if (system === "GARS") {
254 if (!/^[0-9]{3}[A-Z]{2}[1-4]?[1-9]?$/.test(code) || p < 5 || p > 7) {
255 invalid();
256 }
257 const band = Number(code.slice(0, 3));
258 if (band < 1 || band > 720) {
259 invalid();
260 }
261 x = BigInt(band - 1);
262 y = BigInt(24 * character(l24.slice(0, 15), code[3]) + character(l24, code[4]));
263 if (p >= 6) {
264 const q = Number(code[5]) - 1;
265 if (q < 0 || q > 3) {
266 invalid();
267 }
268 x = 2n * x + BigInt(q % 2);
269 y = 2n * y + BigInt(1 - Math.floor(q / 2));
270 }
271 if (p === 7) {
272 const k = Number(code[6]) - 1;
273 x = 3n * x + BigInt(k % 3);
274 y = 3n * y + BigInt(2 - Math.floor(k / 3));
275 }
276 } else if (system === "GEOREF") {
277 if (!sequences[system].includes(p)) {
278 invalid();
279 }
280 x = BigInt(character(l24, code[0]));
281 y = BigInt(character(l24.slice(0, 12), code[1]));
282 if (p >= 4) {
283 x = 15n * x + BigInt(character(l24.slice(0, 15), code[2]));
284 y = 15n * y + BigInt(character(l24.slice(0, 15), code[3]));
285 }
286 if (p >= 8) {
287 const m = (p - 4) / 2;
288 const tail = code.slice(4);
289 if (!/^[0-9]+$/.test(tail)) {
290 invalid();
291 }
292 const radix = 6 * 10 ** (m - 1);
293 const e = Number(tail.slice(0, m));
294 const n = Number(tail.slice(m));
295 if (e >= radix || n >= radix) {
296 invalid();
297 }
298 x = x * BigInt(radix) + BigInt(e);
299 y = y * BigInt(radix) + BigInt(n);
300 }
301 } else {
302 if (!sequences.MESH.includes(p) || !/^[0-9]+$/.test(code) || Number(code.slice(2, 4)) > 79) {
303 invalid();
304 }
305 let u = BigInt(code.slice(2, 4));
306 let v = BigInt(code.slice(0, 2));
307 if (p >= 6) {
308 if (Number(code[4]) > 7 || Number(code[5]) > 7) {
309 invalid();
310 }
311 v = 8n * v + BigInt(code[4]);
312 u = 8n * u + BigInt(code[5]);
313 }
314 if (p >= 8) {
315 v = 10n * v + BigInt(code[6]);
316 u = 10n * u + BigInt(code[7]);
317 }
318 for (let i = 8; i < p; i++) {
319 const q = Number(code[i]) - 1;
320 if (q < 0 || q > 3) {
321 invalid();
322 }
323 u = 2n * u + BigInt(q % 2);
324 v = 2n * v + BigInt(Math.floor(q / 2));
325 }
326 const scale = BigInt([1, 8, 80, 160, 320, 640][sequences.MESH.indexOf(p)]);
327 x = 280n * scale + u;
328 y = 135n * scale + v;
329 }
330 return grid(system, p, x, y);
331}

339 }

361 const index = latitude ? cell.y : cell.x;

371function relate(a: Cell, b: Cell): string {
372 const aw = { n: a.x, d: a.w };
373 const ae = { n: a.x + 1n, d: a.w };
374 const as = { n: a.y, d: a.h };
375 const an = { n: a.y + 1n, d: a.h };
376 const bw = { n: b.x, d: b.w };
377 const be = { n: b.x + 1n, d: b.w };
378 const bs = { n: b.y, d: b.h };
379 const bn = { n: b.y + 1n, d: b.h };
380 const inside = comparison(aw, bw) >= 0 && comparison(ae, be) <= 0 && comparison(as, bs) >= 0 && comparison(an, bn) <= 0;
381 const contains = comparison(bw, aw) >= 0 && comparison(be, ae) <= 0 && comparison(bs, as) >= 0 && comparison(bn, an) <= 0;
382 if (inside && contains) {
383 return "EQUAL";
384 }
385 if (inside) {
386 return "INSIDE";
387 }
388 if (contains) {
389 return "CONTAINS";
390 }
391 const x = comparison(aw, be) < 0 && comparison(bw, ae) < 0;
392 const y = comparison(as, bn) < 0 && comparison(bs, an) < 0;
393 if (x && y) {
394 return "OVERLAP";
395 }
396 const yTouch = comparison(as, bn) <= 0 && comparison(bs, an) <= 0;
397 const xTouch = comparison(aw, be) <= 0 && comparison(bw, ae) <= 0;
398 const wrapped = a.x + 1n === a.w && b.x === 0n || b.x + 1n === b.w && a.x === 0n;
399 return yTouch && (xTouch || wrapped) ? "TOUCH" : "DISJOINT";
400}

402function compact(cells: Cell[]): string {
403 const system = cells[0].system;
404 const sequence = sequences[system];
405 const selected = new Map<string, Cell>();
406 cells.sort((a, b) => sequence.indexOf(a.p) - sequence.indexOf(b.p));
407 for (const cell of cells) {
408 const index = sequence.indexOf(cell.p);
409 if (!sequence.slice(0, index).some(p => selected.has(key(ancestor(cell, p))))) {
410 selected.set(key(cell), cell);
411 }
412 }
413 while (true) {
414 let merged = false;
415 for (const cell of selected.values()) {
416 const index = sequence.indexOf(cell.p);
417 if (!index) {
418 continue;
419 }
420 const parent = ancestor(cell, sequence[index - 1]);
421 const rx = cell.w / parent.w;
422 const ry = cell.h / parent.h;
423 if (rx * ry > BigInt(selected.size)) {
424 continue;
425 }
426 const children: Cell[] = [];
427 for (let j = 0n; j < ry; j++) {
428 for (let i = 0n; i < rx; i++) {
429 children.push(grid(system, cell.p, parent.x * rx + i, parent.y * ry + j));
430 }
431 }
432 if (children.every(child => selected.has(key(child)))) {
433 for (const child of children) {
434 selected.delete(key(child));
435 }
436 selected.set(key(parent), parent);
437 merged = true;
438 break;
439 }
440 }
441 if (!merged) {
442 break;
443 }
444 }
445 const sorted = [...selected.values()].sort((a, b) =>
446 sequence.indexOf(a.p) - sequence.indexOf(b.p) || (a.y === b.y ? 0 : a.y > b.y ? -1 : 1) ||
447 (a.x === b.x ? 0 : a.x < b.x ? -1 : 1));
448 return sorted.map(cell => encode(cell)!).join(" ");
449}

479function command(line: string): string {
480 const [name, ...args] = line.split(" ");
481 if (name === "COMPACT") {
482 if (args.length < 2 || args.length > 101) {
483 invalid();
484 }
485 const system = systemName(args[0]);
486 return compact(args.slice(1).map(code => decode(system, code)));
487 }
488 if (!Object.hasOwn(arities, name) || !arities[name].includes(args.length)) {
489 invalid();
490 }
491 if (name === "ENCODE" || name === "SIZE" || name === "COVER") {
492 const system = systemName(args[0]);
493 if (name === "ENCODE") {
494 const lat = coordinate(args[1], true);
495 const lon = coordinate(args[2], false);
496 return encode(point(system, precision(system, args[3]), lat, lon)) ?? "OUT OF AREA";
497 }
498 const p = precision(system, args[1]);
499 const cell = grid(system, p);
500 if (name === "SIZE") {
501 const scale = BigInt([1, 8, 80, 160, 320, 640][sequences.MESH.indexOf(p)] ?? 0);
502 const count = system === "MESH" ? 8000n * scale * scale : cell.w * cell.h;
503 return "w=" + coordinateText(360000000n / cell.w) + " h=" + coordinateText(180000000n / cell.h) + " cells=" + count;
504 }
505 const lat1 = coordinate(args[2], true);
506 const lon1 = coordinate(args[3], false);
507 const lat2 = coordinate(args[4], true);
508 const lon2 = coordinate(args[5], false);
509 if (lat1 > lat2) {
510 invalid();
511 }
512 const a = point(system, p, lat1, lon1);
513 const b = point(system, p, lat2, lon2);
514 let columns = b.x - a.x + 1n;
515 if (lon1 > lon2) {
516 columns += cell.w;
517 if (columns > cell.w) {
518 columns = cell.w;
519 }
520 }
521 if (columns * (b.y - a.y + 1n) > 100n) {
522 return "TOO MANY";
523 }
524 const codes: string[] = [];
525 for (let y = b.y; y >= a.y; y--) {
526 for (let i = 0n; i < columns; i++) {
527 codes.push(encode(grid(system, p, (a.x + i) % cell.w, y)) ?? "-");
528 }
529 }
530 return codes.join(" ");
531 }
532 if (name === "SHORTEN" || name === "RECOVER") {
533 const lat = coordinate(args[1], true);
534 const lon = coordinate(args[2], false);
535 if (name === "SHORTEN") {
536 const cell = decode("PLUSCODE", args[0]);
537 if (cell.p < 8) {
538 invalid();
539 }
540 const cy = center(cell, true);
541 const cx = center(cell, false);
542 const dy = cy.n - lat * cy.d;
543 const dx = cx.n - lon * cx.d;
544 const close = (distance: bigint): boolean =>
545 (dy < 0n ? -dy : dy) < distance * cy.d && (dx < 0n ? -dx : dx) < distance * cx.d;
546 return close(15000n) ? args[0].slice(6) : close(300000n) ? args[0].slice(4) : args[0];
547 }
548 const parts = args[0].split("+");
549 if (parts.length !== 2 || ![2, 4].includes(parts[0].length) || ![0, 2, 3].includes(parts[1].length)) {
550 invalid();
551 }
552 for (const c of parts.join("")) {
553 character(plus, c);
554 }
555 const removed = 8 - parts[0].length;
556 const prefix = encode(point("PLUSCODE", 10, lat, lon))!.slice(0, removed);
557 const cell = decode("PLUSCODE", prefix + args[0]);
558 const cy = center(cell, true);
559 const cx = center(cell, false);
560 const distance = removed === 4 ? 1000000n : 50000n;
561 const my = distance * cell.h / 180000000n;
562 const mx = distance * cell.w / 360000000n;
563 if ((lat + distance / 2n) * cy.d < cy.n && cy.n - distance * cy.d >= -90000000n * cy.d) {
564 cell.y -= my;
565 } else if ((lat - distance / 2n) * cy.d > cy.n && cy.n + distance * cy.d <= 90000000n * cy.d) {
566 cell.y += my;
567 }
568 if ((lon + distance / 2n) * cx.d < cx.n) {
569 cell.x = mod(cell.x - mx, cell.w);
570 } else if ((lon - distance / 2n) * cx.d > cx.n) {
571 cell.x = mod(cell.x + mx, cell.w);
572 }
573 return encode(cell)!;
574 }
575 if (name === "RELATE") {
576 return relate(decode(systemName(args[0]), args[1]), decode(systemName(args[2]), args[3]));
577 }
578 if (["DECODE", "NEIGHBORS", "PARENT", "CHILDREN", "COMMON", "CONTAINS", "CONVERT", "STEPS", "PATH"].includes(name)) {
579 const system = systemName(args[0]);
580 const cell = decode(system, args[1]);
581 const sequence = sequences[system];
582 const index = sequence.indexOf(cell.p);

604 return "NONE";
605 }

608 const ry = child.h / cell.h;

611 }

614 for (let i = 0n; i < rx; i++) {
615 codes.push(encode(grid(system, child.p, rx * cell.x + i, ry * cell.y + j)) ?? "-");
616 }

619 }
620 if (name === "CONTAINS") {
621 const lat = coordinate(args[2], true);
622 const lon = coordinate(args[3], false);

624 }

626 if (args[2] !== "TO") {
627 invalid();
628 }
629 const target = systemName(args[3]);
630 const converted = grid(target, precision(target, args[4]));
631 converted.x = (2n * cell.x + 1n) * converted.w / (2n * cell.w);
632 converted.y = (2n * cell.y + 1n) * converted.h / (2n * cell.h);
633 return encode(converted) ?? "OUT OF AREA";
634 }
635 const other = decode(system, args[2]);
636 if (name === "COMMON") {
637 const last = Math.min(index, sequence.indexOf(other.p));
638 for (let i = last; i >= 0; i--) {
639 const a = ancestor(cell, sequence[i]);
640 const b = ancestor(other, sequence[i]);

642 return encode(a)!;
643 }

645 return "NONE";
646 }
647 if (cell.p !== other.p) {
648 return "MISMATCH";

650 const [dx, dy] = offsets(cell, other);

652 return "dx=" + dx + " dy=" + dy;

655 const ay = dy < 0n ? -dy : dy;
656 const n = ax > ay ? ax : ay;

658 return "TOO MANY";
659 }
660 if (!n) {
661 return encode(cell)!;
662 }
663 const codes: string[] = [];
664 for (let i = 0n; i <= n; i++) {
665 const x = mod(cell.x + floor(2n * i * dx + n, 2n * n), cell.w);
666 const y = cell.y + floor(2n * i * dy + n, 2n * n);
667 codes.push(encode(grid(system, cell.p, x, y)) ?? "-");
668 }
669 return codes.join(" ");
670 }
671 return otherCommand(name, args);
672}

674function otherCommand(name: string, args: string[]): string {
675 if (name === "UTMZONE") {
676 const lat = coordinate(args[0], true);
677 const lon = coordinate(args[1], false);
678 if (lat < -80000000n || lat >= 84000000n) {
679 return "NONE";
680 }

682 const band = bands[Math.min(index, 19)];
683 let zone = (lon + 180000000n) / 6000000n + 1n;
684 if (band === "V" && lon >= 3000000n && lon < 12000000n) {
685 zone = 32n;
686 }
687 if (band === "X" && lon >= 0n && lon < 42000000n) {
688 zone = lon < 9000000n ? 31n : lon < 21000000n ? 33n : lon < 33000000n ? 35n : 37n;
689 }

693 const z = integer(args[2], 1, 23);

698 code += String((x >> BigInt(i) & 1n) + 2n * (y >> BigInt(i) & 1n));

711 }

713 }
714 if (name === "UNOSGB" || name === "UNMGRS") {

718 if (name === "UNOSGB") {

720 invalid();
721 }

725 n = BigInt(500000 * (3 - Math.floor(first / 5)) + 100000 * (4 - Math.floor(second / 5)));
726 if (e < 0n || e >= 700000n || n < 0n || n >= 1300000n) {

772}
