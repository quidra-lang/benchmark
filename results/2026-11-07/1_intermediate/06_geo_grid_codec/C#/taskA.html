36 static long coordinate(string s, bool latitude) {
37 require(matches(s,"-?[0-9]{1,3}(\\.[0-9]{1,6})?"));
38 bool negative = s.StartsWith("-");
39 string[] pieces = (negative ? s.Substring(1) : s).Split('.');
40 long value = long.Parse(pieces[0])*1000000;
41 if (pieces.Length == 2) value += long.Parse(pieces[1].PadRight(6,'0'));
42 if (negative) value = -value;
43 require(latitude ? value >= -90000000 && value <= 90000000 : value >= -180000000 && value < 180000000);
44 return value;
45 }

56 static Cell grid(string system, int level, long x, long y) {
57 int p = precisions(system)[level];
58 long w, h;
59 switch (system) {
60 case "GEOHASH":
61 w=1L<<((5*p+1)/2);
62 h=1L<<(5*p/2);
63 break;
64 case "MAIDENHEAD":
65 w=new long[]{18,180,4320,43200}[level];
66 h=w;
67 break;
68 case "PLUSCODE":
69 w=new long[]{18,360,7200,144000,2880000,11520000}[level];
70 h=new long[]{9,180,3600,72000,1440000,7200000}[level];
71 break;
72 case "GARS":
73 w=new long[]{720,1440,4320}[level];
74 h=new long[]{360,720,2160}[level];
75 break;
76 case "GEOREF":
77 w=new long[]{24,360,21600,216000,2160000}[level];
78 h=new long[]{12,180,10800,108000,1080000}[level];
79 break;
80 default:
81 w=360*MESH_SCALE[level];
82 h=270*MESH_SCALE[level];
83 break;
84 }
85 return new Cell(system,level,x,y,w,h);

107 static long[] radix(long value, long[] bases, int count) {
108 long[] result = new long[count];
109 for (int i = count-1; i >= 0; i--) {
110 result[i] = value%bases[i];
111 value /= bases[i];
112 }
113 return result;

115 static string code(Cell c) {
116 string system = c.system;
117 int p = precisions(system)[c.level];
118 long x = c.x, y = c.y;
119 StringBuilder result = new StringBuilder();
120 switch (system) {
121 case "GEOHASH": {
122 int ax=(5*p+1)/2, ay=5*p/2, group=0;
123 for (int bit = 0; bit < 5*p; bit++) {
124 int b = bit%2 == 0 ? (int)(x>>(--ax)&1) : (int)(y>>(--ay)&1);
125 group = group*2+b;
126 if (bit%5 == 4) {
127 result.Append(GH[group]);
128 group=0;
129 }
130 }
131 return result.ToString();
132 }
133 case "MAIDENHEAD": {
134 int count=p/2;
135 long[] a=radix(x,new long[]{18,10,24,10},count), b=radix(y,new long[]{18,10,24,10},count);
136 for (int i=0;i<count;i++) result.Append(MAIDEN[i][(int)a[i]]).Append(MAIDEN[i][(int)b[i]]);
137 return result.ToString();
138 }
139 case "PLUSCODE": {
140 int extra = p == 11 ? (int)(4*(y%5)+x%4) : -1;
141 if (p == 11) {
142 x/=4;
143 y/=5;
144 }
145 int count=Math.Min(p,10)/2;
146 long[] a=radix(x,new long[]{18,20,20,20,20},count), b=radix(y,new long[]{9,20,20,20,20},count);
147 for (int i=0;i<count;i++) result.Append(PLUS[(int)b[i]]).Append(PLUS[(int)a[i]]);
148 if (extra >= 0) result.Append(PLUS[extra]);
149 string text=result.ToString();
150 return p < 8 ? text+new string('0',8-p)+"+" : text.Substring(0,8)+"+"+text.Substring(8);
151 }
152 case "GARS": {
153 long scale=c.level==0 ? 1 : c.level==1 ? 2 : 6;
154 long bx=x/scale, by=y/scale;
155 result.Append(pad(bx+1,3)).Append(L24[(int)(by/24)]).Append(L24[(int)(by%24)]);
156 if (p>=6) {
157 long divisor=p==7 ? 3 : 1;
158 result.Append(1+x/divisor%2+2*(1-y/divisor%2));
159 }
160 if (p==7) result.Append(1+x%3+3*(2-y%3));
161 return result.ToString();
162 }
163 case "GEOREF": {
164 if (p==2) return ""+L24[(int)x]+L24[(int)y];
165 int m=(p-4)/2;
166 long scale=p==4 ? 1 : 6*power(m-1), bx=x/scale, by=y/scale;
167 result.Append(L24[(int)(bx/15)]).Append(L24[(int)(by/15)])
168 .Append(L24[(int)(bx%15)]).Append(L24[(int)(by%15)]);
169 if (p>=8) result.Append(pad(x%scale,m)).Append(pad(y%scale,m));
170 return result.ToString();
171 }
172 default: {
173 long scale=MESH_SCALE[c.level], u=x-280*scale, v=y-135*scale;
174 if (u<0 || u>=80*scale || v<0 || v>=100*scale) return null;
175 int count=c.level+1;
176 long[] a=radix(u,new long[]{80,8,10,2,2,2},count), b=radix(v,new long[]{100,8,10,2,2,2},count);
177 result.Append(pad(b[0],2)).Append(pad(a[0],2));
178 for (int i=1;i<count;i++) {
179 if (i<=2) result.Append(b[i]).Append(a[i]);
180 else result.Append(1+a[i]+2*b[i]);
181 }
182 return result.ToString();
183 }
184 }

186 static Cell decode(string system, string text) {
187 precisions(system);
188 int p=text.Length;
189 long x=0,y=0;
190 switch (system) {
191 case "GEOHASH": {
192 require(p>=1 && p<=9);
193 int bit=0;
194 foreach (char ch in text) {
195 int d=digit(ch,GH);
196 for (int i=4;i>=0;i--) {
197 if (bit%2==0) x=x*2+(d>>i&1);
198 else y=y*2+(d>>i&1);
199 bit++;
200 }
201 }
202 break;
203 }
204 case "MAIDENHEAD": {
205 level(system,p);
206 for (int i=0;i<p/2;i++) {
207 x=x*MAIDEN[i].Length+digit(text[2*i],MAIDEN[i]);
208 y=y*MAIDEN[i].Length+digit(text[2*i+1],MAIDEN[i]);
209 }
210 break;
211 }
212 case "PLUSCODE": {
213 require(p>=9 && text.IndexOf('+')==8 && text.LastIndexOf('+')==8);
214 int firstZero=text.Substring(0,8).IndexOf('0');
215 int before=firstZero<0 ? 8 : firstZero, after=p-9;
216 require(before==2 || before==4 || before==6 || before==8);
217 require(before==8 ? after==0 || after==2 || after==3 : after==0);
218 for (int i=before;i<8;i++) require(text[i]=='0');
219 string sequence=text.Substring(0,before)+text.Substring(9);
220 p=sequence.Length;
221 require(digit(sequence[0],PLUS)<=8 && digit(sequence[1],PLUS)<=17);
222 for (int i=0;i<Math.Min(p,10);i+=2) {
223 y=y*20+digit(sequence[i],PLUS);
224 x=x*20+digit(sequence[i+1],PLUS);
225 }
226 if (p==11) {
227 int d=digit(sequence[10],PLUS);
228 x=x*4+d%4;
229 y=y*5+d/4;
230 }
231 break;
232 }
233 case "GARS": {
234 require(p>=5 && p<=7);
235 long e=digits(text.Substring(0,3));
236 require(e>=1 && e<=720);
237 x=e-1;
238 y=digit(text[3],L24.Substring(0,15))*24+digit(text[4],L24);
239 if (p>=6) {
240 int q=digit(text[5],"1234");
241 x=x*2+q%2;
242 y=y*2+1-q/2;
243 }
244 if (p==7) {
245 int q=digit(text[6],"123456789");
246 x=x*3+q%3;
247 y=y*3+2-q/3;
248 }
249 break;
250 }
251 case "GEOREF": {
252 level(system,p);
253 x=digit(text[0],L24);
254 y=digit(text[1],L24.Substring(0,12));
255 if (p>=4) {
256 x=x*15+digit(text[2],L24.Substring(0,15));
257 y=y*15+digit(text[3],L24.Substring(0,15));
258 }
259 if (p>=8) {
260 int m=(p-4)/2;
261 long scale=6*power(m-1), a=digits(text.Substring(4,m)), b=digits(text.Substring(4+m));
262 require(a<scale && b<scale);
263 x=x*scale+a;
264 y=y*scale+b;
265 }
266 break;
267 }
268 default: {
269 int l=level(system,p);
270 digits(text);
271 long v=digits(text.Substring(0,2)), u=digits(text.Substring(2,2));
272 require(u<80);
273 if (p>=6) {
274 v=v*8+digit(text[4],"01234567");
275 u=u*8+digit(text[5],"01234567");
276 }
277 if (p>=8) {
278 v=v*10+digit(text[6],"0123456789");
279 u=u*10+digit(text[7],"0123456789");
280 }
281 for (int i=8;i<p;i++) {
282 int q=digit(text[i],"1234");
283 v=v*2+q/2;
284 u=u*2+q%2;
285 }
286 x=u+280*MESH_SCALE[l];
287 y=v+135*MESH_SCALE[l];
288 break;
289 }
290 }
291 Cell result=grid(system,level(system,p),x,y);
292 require(text.Equals(code(result)));
293 return result;
294 }

298 }

309 }

316 require(matches(a,"[0-9]{1,5}") && matches(b,"[0-9]{1,5}") && a.Length==b.Length);
317 long size=power(5-a.Length);
318 return new long[]{digits(a)*size,digits(b)*size,size};

320 static string columns(long zone) { return zone%3==1 ? "ABCDEFGH" : zone%3==2 ? "JKLMNPQR" : "STUVWXYZ"; }
321 static string auxiliary(string[] t) {
322 string command=t[0];
323 int n=t.Length-1;
324 switch (command) {
325 case "UTMZONE": {
326 require(n==2);
327 long lat=coordinate(t[1],true), lon=coordinate(t[2],false);
328 if (lat< -80000000 || lat>=84000000) return "NONE";
329 char band=BANDS[(int)Math.Min((lat+80000000)/8000000,19)];
330 long zone=(lon+180000000)/6000000+1;
331 if (band=='V' && lon>=3000000 && lon<12000000) zone=32;
332 if (band=='X') {
333 if (lon>=0 && lon<9000000) zone=31;
334 else if (lon>=9000000 && lon<21000000) zone=33;
335 else if (lon>=21000000 && lon<33000000) zone=35;
336 else if (lon>=33000000 && lon<42000000) zone=37;
337 }
338 return zone+band.ToString();
339 }
340 case "OSGB": {
341 require(n==3);
342 long e=range(t[1],0,699999), north=range(t[2],0,1299999);
343 int d=(int)range(t[3],0,5);
344 string letters=""+G25[(int)((3-north/500000)*5+2+e/500000)]+G25[(int)((4-north/100000%5)*5+e/100000%5)];
345 return letters+(d==0 ? "" : " "+pad(e%100000,5).Substring(0,d)+" "+pad(north%100000,5).Substring(0,d));
346 }
347 case "UNOSGB": {
348 require(n==1 || n==3);
349 require(t[1].Length==2);
350 int a=digit(t[1][0],G25), b=digit(t[1][1],G25);
351 long e=500000*(a%5-2)+100000*(b%5), north=500000*(3-a/5)+100000*(4-b/5), size=100000;
352 require(e>=0 && e<700000 && north>=0 && north<1300000);
353 if (n==3) {
354 long[] values=detail(t[2],t[3]);
355 e+=values[0];
356 north+=values[1];
357 size=values[2];
358 }
359 return "sw="+e+","+north+" size="+size;
360 }
361 case "MGRS": {
362 require(n==5);
363 long zone=range(t[1],1,60), e=range(t[3],100000,899999), north=range(t[4],0,9999999);
364 require(t[2].Length==1);
365 digit(t[2][0],BANDS);
366 int d=(int)range(t[5],0,5), offset=zone%2==0 ? 5 : 0;
367 string square=""+columns(zone)[(int)(e/100000-1)]+MROWS[(int)((north/100000+offset)%20)];
368 return zone+t[2]+" "+square+(d==0 ? "" : " "+pad(e%100000,5).Substring(0,d)+" "+pad(north%100000,5).Substring(0,d));
369 }
370 case "UNMGRS": {
371 require(n==2 || n==4);
372 require(matches(t[1],"[1-9][0-9]?[CDEFGHJKLMNPQRSTUVWX]"));
373 long zone=long.Parse(t[1].Substring(0,t[1].Length-1));
374 require(zone<=60 && t[2].Length==2);
375 int a=digit(t[2][0],columns(zone)), b=digit(t[2][1],MROWS), offset=zone%2==0 ? 5 : 0;
376 long e=100000*(a+1), north=100000*mod(b-offset,20), size=100000;
377 if (n==4) {
378 long[] values=detail(t[3],t[4]);
379 e+=values[0];
380 north+=values[1];
381 size=values[2];
382 }
383 return t[1]+" sw="+e+","+north+" size="+size;
384 }
385 case "QUADKEY": {
386 require(n==3);
387 int z=(int)range(t[3],1,23);
388 long x=range(t[1],0,(1L<<z)-1), y=range(t[2],0,(1L<<z)-1);
389 StringBuilder result=new StringBuilder();
390 for (int i=z-1;i>=0;i--) result.Append((x>>i&1)+2*(y>>i&1));
391 return result.ToString();
392 }
393 case "UNQUADKEY": {
394 require(n==1 && matches(t[1],"[0-3]{1,23}"));
395 long x=0,y=0;
396 foreach (char c in t[1]) {
397 int d=c-'0';
398 x=x*2+d%2;
399 y=y*2+d/2;
400 }
401 return "x="+x+" y="+y+" z="+t[1].Length;
402 }
403 default: throw new Invalid();
404 }
405 }
406 static string execute(string line) {
407 string[] t=line.Split(' ');
408 string command=t[0];
409 if (!counts.ContainsKey(command) && !command.Equals("COMPACT")) return auxiliary(t);
410 if (command.Equals("COMPACT")) require(t.Length>=3 && t.Length<=102);
411 else require(t.Length==counts[command]+1);
412 if (command.Equals("SHORTEN") || command.Equals("RECOVER")) {
413 long lat=coordinate(t[2],true), lon=coordinate(t[3],false);
414 if (command.Equals("SHORTEN")) {
415 Cell cShort=decode("PLUSCODE",t[1]);
416 require(precisions(cShort.system)[cShort.level]>=8);
417 long ynShort= -90000000*2*cShort.h+180000000*(2*cShort.y+1), xnShort= -180000000*2*cShort.w+360000000*(2*cShort.x+1);
418 if (Math.Abs(ynShort-lat*2*cShort.h)<15000*2*cShort.h && Math.Abs(xnShort-lon*2*cShort.w)<15000*2*cShort.w) return t[1].Substring(6);
419 if (Math.Abs(ynShort-lat*2*cShort.h)<300000*2*cShort.h && Math.Abs(xnShort-lon*2*cShort.w)<300000*2*cShort.w) return t[1].Substring(4);
420 return t[1];
421 }
422 int plus=t[1].IndexOf('+');
423 require(plus==2 || plus==4);
424 require(t[1].LastIndexOf('+')==plus);
425 int tail=t[1].Length-plus-1;
426 require(tail==0 || tail==2 || tail==3);
427 foreach (char ch in t[1].Replace("+","")) digit(ch,PLUS);
428 int missing=8-plus;
429 string prefix=code(point("PLUSCODE",4,lat,lon)).Substring(0,missing);
430 Cell c=decode("PLUSCODE",prefix+t[1]);
431 long d=missing==4 ? 1000000 : 50000, x=c.x, y=c.y;
432 long yn= -90000000*2*c.h+180000000*(2*y+1), xn= -180000000*2*c.w+360000000*(2*x+1);
433 long my=d*c.h/180000000, mx=d*c.w/360000000;
434 if ((lat+d/2)*2*c.h<yn && yn-d*2*c.h>= -90000000*2*c.h) y-=my;
435 else if ((lat-d/2)*2*c.h>yn && yn+d*2*c.h<=90000000*2*c.h) y+=my;
436 if ((lon+d/2)*2*c.w<xn) x=mod(x-mx,c.w);
437 else if ((lon-d/2)*2*c.w>xn) x=mod(x+mx,c.w);
438 return code(grid("PLUSCODE",c.level,x,y));
439 }
440 string system=t[1];
441 precisions(system);
442 if (command.Equals("ENCODE")) {
443 Cell c=point(system,level(system,integer(t[4])),coordinate(t[2],true),coordinate(t[3],false));
444 string encoded=code(c);
445 return encoded==null ? "OUT OF AREA" : encoded;
446 }
447 if (command.Equals("SIZE")) {
448 int l=level(system,integer(t[2]));
449 Cell c=grid(system,l,0,0);
450 long count=system.Equals("MESH") ? 8000*MESH_SCALE[l]*MESH_SCALE[l] : c.w*c.h;
451 return "w="+coordinateFormat(360000000/c.w)+" h="+coordinateFormat(180000000/c.h)+" cells="+count;
452 }
453 if (command.Equals("COVER")) {
454 int l=level(system,integer(t[2]));
455 long lat1=coordinate(t[3],true), lon1=coordinate(t[4],false), lat2=coordinate(t[5],true), lon2=coordinate(t[6],false);
456 require(lat1<=lat2);
457 Cell a=point(system,l,lat1,lon1), b=point(system,l,lat2,lon2);
458 long columns=lon1<=lon2 ? b.x-a.x+1 : Math.Min(a.w,b.x-a.x+1+a.w);
459 if (columns*(b.y-a.y+1)>100) return "TOO MANY";
460 List<string> result=new();
461 for (long y=b.y;y>=a.y;y--) {
462 for (long i=0;i<columns;i++) result.Add(shown(grid(system,l,(a.x+i)%a.w,y)));
463 }
464 return string.Join(" ",result);
465 }
466 if (command.Equals("COMPACT")) {
467 HashSet<Cell> cells=new();
468 for (int i=2;i<t.Length;i++) cells.Add(decode(system,t[i]));
469 var covered = cells.Where(c => cells.Any(a => a.level<c.level && ancestor(c,a.level) == a)).ToList();
470 cells.ExceptWith(covered);
471 bool changed=true;
472 while (changed) {
473 changed=false;
474 foreach (Cell child in cells.ToList()) {
475 if (child.level==0) continue;
476 Cell parent=ancestor(child,child.level-1), g=grid(system,child.level,0,0);
477 long rx=g.w/parent.w, ry=g.h/parent.h;
478 if (rx*ry>cells.Count) continue;
479 HashSet<Cell> siblings=new();
480 for (long j=0;j<ry;j++) {
481 for (long i=0;i<rx;i++) siblings.Add(grid(system,child.level,parent.x*rx+i,parent.y*ry+j));
482 }
483 if (cells.IsSupersetOf(siblings)) {
484 cells.ExceptWith(siblings);
485 cells.Add(parent);
486 changed=true;
487 break;
488 }
489 }
490 }
491 List<Cell> result=new(cells);
492 result = result.OrderBy(c => c.level).ThenByDescending(c => c.y).ThenBy(c => c.x).ToList();
493 return string.Join(" ",result.Select(code));
494 }
495 Cell primary=decode(system,t[2]);
496 switch (command) {
497 case "DECODE": return "sw="+coordinateFormat(boundary(primary,true,0,false))+","+coordinateFormat(boundary(primary,false,0,false))+
498 " ne="+coordinateFormat(boundary(primary,true,1,false))+","+coordinateFormat(boundary(primary,false,1,false))+
499 " c="+coordinateFormat(boundary(primary,true,0,true))+","+coordinateFormat(boundary(primary,false,0,true));
500 case "NEIGHBORS": {
501 int[] dx={0,1,1,1,0,-1,-1,-1}, dy={1,1,0,-1,-1,-1,0,1};
502 List<string> result=new();
503 for (int i=0;i<8;i++) result.Add(primary.y+dy[i]<0 || primary.y+dy[i]>=primary.h ? "-" : shown(grid(system,primary.level,mod(primary.x+dx[i],primary.w),primary.y+dy[i])));
504 return string.Join(" ",result);
505 }
506 case "PARENT": return primary.level==0 ? "NONE" : code(ancestor(primary,primary.level-1));
507 case "CHILDREN": {
508 if (primary.level+1==precisions(system).Length) return "NONE";
509 Cell g=grid(system,primary.level+1,0,0);
510 long rx=g.w/primary.w, ry=g.h/primary.h;
511 if (rx*ry>100) return "TOO MANY";
512 List<string> result=new();
513 for (long j=ry-1;j>=0;j--) {
514 for (long i=0;i<rx;i++) result.Add(shown(grid(system,g.level,primary.x*rx+i,primary.y*ry+j)));
515 }
516 return string.Join(" ",result);
517 }
518 case "COMMON": {
519 Cell b=decode(system,t[3]);
520 for (int l=Math.Min(primary.level,b.level);l>=0;l--) {
521 Cell parent=ancestor(primary,l);
522 if (parent.Equals(ancestor(b,l))) return code(parent);
523 }
524 return "NONE";
525 }
526 case "CONTAINS": return primary.Equals(point(system,primary.level,coordinate(t[3],true),coordinate(t[4],false))) ? "YES" : "NO";
527 case "CONVERT": {
528 require(t[3].Equals("TO"));
529 Cell g=grid(t[4],level(t[4],integer(t[5])),0,0);
530 string result=code(grid(g.system,g.level,(2*primary.x+1)*g.w/(2*primary.w),(2*primary.y+1)*g.h/(2*primary.h)));
531 return result==null ? "OUT OF AREA" : result;
532 }
533 case "STEPS":
534 case "PATH": {
535 Cell b=decode(system,t[3]);
536 if (primary.level!=b.level) return "MISMATCH";
537 long[] delta=steps(primary,b);
538 if (command.Equals("STEPS")) return "dx="+delta[0]+" dy="+delta[1];
539 long n=Math.Max(Math.Abs(delta[0]),Math.Abs(delta[1]));
540 if (n+1>100) return "TOO MANY";
541 List<string> result=new();
542 for (long i=0;i<=n;i++) {
543 long dx=n==0 ? 0 : floor(2*i*delta[0]+n,2*n), dy=n==0 ? 0 : floor(2*i*delta[1]+n,2*n);
544 result.Add(shown(grid(system,primary.level,mod(primary.x+dx,primary.w),primary.y+dy)));
545 }
546 return string.Join(" ",result);
547 }
548 case "RELATE": {
549 Cell b=decode(t[3],t[4]);
550 bool ab=inside(primary,b), ba=inside(b,primary);
551 if (ab && ba) return "EQUAL";
552 if (ab) return "INSIDE";
553 if (ba) return "CONTAINS";
554 int ew=compare(primary.x+1,primary.w,b.x,b.w), we=compare(b.x+1,b.w,primary.x,primary.w);
555 int ns=compare(primary.y+1,primary.h,b.y,b.h), sn=compare(b.y+1,b.h,primary.y,primary.h);
556 if (ew>0 && we>0 && ns>0 && sn>0) return "OVERLAP";
557 bool dateline=primary.x+1==primary.w && b.x==0 || b.x+1==b.w && primary.x==0;
558 return ns>=0 && sn>=0 && (ew>=0 && we>=0 || dateline) ? "TOUCH" : "DISJOINT";
559 }
560 default: throw new Invalid();
561 }
