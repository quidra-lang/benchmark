75Integer integer(const string &s, Integer minimum, Integer maximum) {
76 bool negative = !s.empty() && s[0] == '-';
77 string body = negative ? s.substr(1) : s;
78 require(digits(body, 1, 9));
79 Integer value = std::stoll(s);
80 require(value >= minimum && value <= maximum);
81 return value;

83Integer location(const string &s, bool latitude) {
84 bool negative = !s.empty() && s[0] == '-';
85 string body = negative ? s.substr(1) : s;
86 vector<string> parts = split(body, '.');
87 require(parts.size() == 1 || parts.size() == 2);
88 require(digits(parts[0], 1, 3));
89 string fraction = parts.size() == 2 ? parts[1] : "";
90 if (parts.size() == 2) require(digits(fraction, 1, 6));
91 fraction.append(6 - fraction.size(), '0');
92 Integer value = std::stoll(parts[0]) * 1000000 + std::stoll(fraction);
93 if (negative) value = -value;
94 require(latitude ? value >= -90000000 && value <= 90000000 : value >= -180000000 && value < 180000000);
95 return value;

128string encode(int system, Cell c) {
129 const Grid &g = grids[system][c.precision];
130 int p = g.precision;
131 Integer x = c.x, y = c.y;
132 string result;
133 if (system == 0) {
134 int a = (5 * p + 1) / 2, b = 5 * p / 2, value = 0;
135 for (int i = 0; i < 5 * p; ++i) {
136 int bit = i % 2 == 0 ? int(x >> (a - 1 - i / 2) & 1) : int(y >> (b - 1 - i / 2) & 1);
137 value = value * 2 + bit;
138 if (i % 5 == 4) {
139 result += hashAlphabet[value];
140 value = 0;
141 }
142 }
143 } else if (system == 1 || system == 2) {
144 int count = system == 1 ? p / 2 : std::min(p, 10) / 2;
145 int tail = -1;
146 if (system == 2 && p == 11) {
147 tail = int(4 * (y % 5) + x % 4);
148 x /= 4;
149 y /= 5;
150 }
151 vector<int> xd(count), yd(count);
152 for (int i = count - 1; i >= 0; --i) {
153 int rx = system == 1 ? int(maidenAlphabets[i].size()) : i == 0 ? 18 : 20;
154 int ry = system == 1 ? rx : i == 0 ? 9 : 20;
155 xd[i] = int(x % rx);
156 yd[i] = int(y % ry);
157 x /= rx;
158 y /= ry;
159 }
160 for (int i = 0; i < count; ++i) {
161 if (system == 1) {
162 result += maidenAlphabets[i][xd[i]];
163 result += maidenAlphabets[i][yd[i]];
164 } else {
165 result += plusAlphabet[yd[i]];
166 result += plusAlphabet[xd[i]];
167 }
168 }
169 if (tail >= 0) result += plusAlphabet[tail];
170 if (system == 2) {
171 if (p < 8) result += string(8 - p, '0');
172 result.insert(8, "+");
173 }
174 } else if (system == 3) {
175 Integer scale = p == 5 ? 1 : p == 6 ? 2 : 6;
176 Integer baseY = y / scale;
177 result = padded(x / scale + 1, 3) + letters24[baseY / 24] + letters24[baseY % 24];
178 if (p >= 6) {
179 Integer xx = p == 7 ? x / 3 : x, yy = p == 7 ? y / 3 : y;
180 result += char('1' + xx % 2 + 2 * (1 - yy % 2));
181 }
182 if (p == 7) result += char('1' + x % 3 + 3 * (2 - y % 3));
183 } else if (system == 4) {
184 int m = (p - 4) / 2;
185 Integer scale = p < 8 ? 1 : m == 2 ? 60 : m == 3 ? 600 : 6000;
186 Integer xx = x / scale, yy = y / scale;
187 if (p == 2) return string(1, letters24[x]) + letters24[y];
188 result = string(1, letters24[xx / 15]) + letters24[yy / 15] + letters24[xx % 15] + letters24[yy % 15];
189 if (p >= 8) result += padded(x % scale, m) + padded(y % scale, m);
190 } else {
191 Integer scale = meshScale[c.precision];
192 x -= 280 * scale;
193 y -= 135 * scale;
194 if (x < 0 || x >= 80 * scale || y < 0 || y >= 100 * scale) return "";
195 int count = c.precision + 1;
196 const int radices[] = {0, 8, 10, 2, 2, 2};
197 vector<int> xd(count), yd(count);
198 for (int i = count - 1; i >= 1; --i) {
199 xd[i] = int(x % radices[i]);
200 yd[i] = int(y % radices[i]);
201 x /= radices[i];
202 y /= radices[i];
203 }
204 result = padded(y, 2) + padded(x, 2);
205 for (int i = 1; i < count; ++i) {
206 if (i <= 2) {
207 result += char('0' + yd[i]);
208 result += char('0' + xd[i]);
209 } else result += char('1' + xd[i] + 2 * yd[i]);
210 }
211 }
212 return result;
213}
214Cell decode(int system, const string &code) {
215 int p = int(code.size());
216 Integer x = 0, y = 0;
217 if (system == 0) {
218 require(p >= 1 && p <= 9);
219 int position = 0;
220 for (char c : code) {
221 int value = digit(hashAlphabet, c);
222 for (int bit = 4; bit >= 0; --bit) {
223 if (position++ % 2 == 0) x = x * 2 + (value >> bit & 1);
224 else y = y * 2 + (value >> bit & 1);
225 }
226 }
227 } else if (system == 1) {
228 precisionId(system, p);
229 for (int i = 0; i < p / 2; ++i) {
230 x = x * Integer(maidenAlphabets[i].size()) + digit(maidenAlphabets[i], code[2 * i]);
231 y = y * Integer(maidenAlphabets[i].size()) + digit(maidenAlphabets[i], code[2 * i + 1]);
232 }
233 } else if (system == 2) {
234 require(p >= 9 && code[8] == '+' && std::count(code.begin(), code.end(), '+') == 1);
235 int prefix = 0;
236 while (prefix < 8 && code[prefix] != '0') {
237 digit(plusAlphabet, code[prefix]);
238 ++prefix;
239 }
240 require(prefix == 2 || prefix == 4 || prefix == 6 || prefix == 8);
241 for (int i = prefix; i < 8; ++i) require(code[i] == '0');
242 int suffix = p - 9;
243 require(prefix == 8 ? suffix == 0 || suffix == 2 || suffix == 3 : suffix == 0);
244 require(digit(plusAlphabet, code[0]) <= 8 && digit(plusAlphabet, code[1]) <= 17);
245 string sequence = code.substr(0, prefix) + code.substr(9);
246 p = int(sequence.size());
247 for (int i = 0; i < std::min(p, 10) / 2; ++i) {
248 y = y * (i == 0 ? 9 : 20) + digit(plusAlphabet, sequence[2 * i]);
249 x = x * (i == 0 ? 18 : 20) + digit(plusAlphabet, sequence[2 * i + 1]);
250 }
251 if (p == 11) {
252 int tail = digit(plusAlphabet, sequence[10]);
253 x = x * 4 + tail % 4;
254 y = y * 5 + tail / 4;
255 }
256 } else if (system == 3) {
257 require(p >= 5 && p <= 7 && digits(code.substr(0, 3), 3, 3));
258 x = integer(code.substr(0, 3), 1, 720) - 1;
259 int a = digit(letters24, code[3]), b = digit(letters24, code[4]);
260 require(a < 15);
261 y = 24 * a + b;
262 if (p >= 6) {
263 require(code[5] >= '1' && code[5] <= '4');
264 int q = code[5] - '1';
265 x = 2 * x + q % 2;
266 y = 2 * y + 1 - q / 2;
267 }
268 if (p == 7) {
269 require(code[6] >= '1' && code[6] <= '9');
270 int q = code[6] - '1';
271 x = 3 * x + q % 3;
272 y = 3 * y + 2 - q / 3;
273 }
274 } else if (system == 4) {
275 precisionId(system, p);
276 x = digit(letters24, code[0]);
277 y = digit(letters24, code[1]);
278 require(y < 12);
279 if (p >= 4) {
280 int a = digit(letters24, code[2]), b = digit(letters24, code[3]);
281 require(a < 15 && b < 15);
282 x = 15 * x + a;
283 y = 15 * y + b;
284 }
285 if (p >= 8) {
286 int m = (p - 4) / 2;
287 Integer scale = m == 2 ? 60 : m == 3 ? 600 : 6000;
288 require(digits(code.substr(4, m), m, m) && digits(code.substr(4 + m), m, m));
289 x = x * scale + integer(code.substr(4, m), 0, scale - 1);
290 y = y * scale + integer(code.substr(4 + m), 0, scale - 1);
291 }
292 } else {
293 int precision = precisionId(system, p);
294 require(digits(code, p, p));
295 y = integer(code.substr(0, 2), 0, 99);
296 x = integer(code.substr(2, 2), 0, 79);
297 if (p >= 6) {
298 require(code[4] <= '7' && code[5] <= '7');
299 y = y * 8 + code[4] - '0';
300 x = x * 8 + code[5] - '0';
301 }
302 if (p >= 8) {
303 y = y * 10 + code[6] - '0';
304 x = x * 10 + code[7] - '0';
305 }
306 for (int i = 8; i < p; ++i) {
307 require(code[i] >= '1' && code[i] <= '4');
308 int q = code[i] - '1';
309 x = x * 2 + q % 2;
310 y = y * 2 + q / 2;
311 }
312 x += 280 * meshScale[precision];
313 y += 135 * meshScale[precision];
314 }
315 return {precisionId(system, p), x, y};
316}

326string relation(int firstSystem, Cell a, int secondSystem, Cell b) {
327 const Grid &ga = grids[firstSystem][a.precision], &gb = grids[secondSystem][b.precision];
328 int west = compare(a.x, ga.width, b.x, gb.width), east = compare(a.x + 1, ga.width, b.x + 1, gb.width);
329 int south = compare(a.y, ga.height, b.y, gb.height), north = compare(a.y + 1, ga.height, b.y + 1, gb.height);
330 if (!west && !east && !south && !north) return "EQUAL";
331 if (west >= 0 && east <= 0 && south >= 0 && north <= 0) return "INSIDE";
332 if (west <= 0 && east >= 0 && south <= 0 && north >= 0) return "CONTAINS";
333 int xy1 = compare(a.x, ga.width, b.x + 1, gb.width), xy2 = compare(b.x, gb.width, a.x + 1, ga.width);
334 int yy1 = compare(a.y, ga.height, b.y + 1, gb.height), yy2 = compare(b.y, gb.height, a.y + 1, ga.height);
335 if (xy1 < 0 && xy2 < 0 && yy1 < 0 && yy2 < 0) return "OVERLAP";
336 bool wrap = a.x == 0 && b.x + 1 == gb.width || b.x == 0 && a.x + 1 == ga.width;
337 if (yy1 <= 0 && yy2 <= 0 && (xy1 <= 0 && xy2 <= 0 || wrap)) return "TOUCH";
338 return "DISJOINT";
339}
340void prune(int system, std::set<Cell> &cells) {
341 vector<Cell> removed;
342 for (Cell c : cells) {
343 for (Cell other : cells) {
344 if (other.precision < c.precision && ancestor(system, c, other.precision) == other) {
345 removed.push_back(c);
346 break;
347 }
348 }
349 }
350 for (Cell c : removed) cells.erase(c);
351}
352string compact(int system, const vector<string> &codes) {
353 std::set<Cell> cells;
354 for (const string &code : codes) cells.insert(decode(system, code));
355 prune(system, cells);
356 for (;;) {
357 bool changed = false;
358 for (Cell c : cells) {
359 if (!c.precision) continue;
360 Cell parent = ancestor(system, c, c.precision - 1);
361 const Grid &g = grids[system][c.precision], &pg = grids[system][parent.precision];
362 Integer needed = (g.width / pg.width) * (g.height / pg.height);
363 if (needed > Integer(cells.size())) continue;
364 vector<Cell> children;
365 for (Cell other : cells) if (other.precision == c.precision && ancestor(system, other, parent.precision) == parent) children.push_back(other);
366 if (Integer(children.size()) != needed) continue;
367 for (Cell child : children) cells.erase(child);
368 cells.insert(parent);
369 prune(system, cells);
370 changed = true;
371 break;
372 }
373 if (!changed) break;
374 }
375 vector<Cell> ordered(cells.begin(), cells.end());
376 std::sort(ordered.begin(), ordered.end(), [](Cell a, Cell b) {
377 if (a.precision != b.precision) return a.precision < b.precision;
378 if (a.y != b.y) return a.y > b.y;
379 return a.x < b.x;
380 });
381 vector<string> result;
382 for (Cell c : ordered) result.push_back(encode(system, c));
383 return join(result);
384}

386Integer squareDigits(const vector<string> &a, int start, Integer &e, Integer &n) {
387 if (int(a.size()) == start) return 100000;
388 require(digits(a[start], 1, 5) && digits(a[start + 1], 1, 5) && a[start].size() == a[start + 1].size());
389 Integer scale = 1;
390 for (int i = int(a[start].size()); i < 5; ++i) scale *= 10;
391 e += integer(a[start], 0, 99999) * scale;
392 n += integer(a[start + 1], 0, 99999) * scale;
393 return scale;
394}
395string execute(const vector<string> &a) {
396 static const std::array<string, 23> names = {
397 "ENCODE", "DECODE", "NEIGHBORS", "PARENT", "CHILDREN", "COMMON",
398 "COMPACT", "CONTAINS", "COVER", "CONVERT", "STEPS", "PATH", "RELATE",
399 "SIZE", "SHORTEN", "RECOVER", "UTMZONE", "OSGB", "UNOSGB", "MGRS",
400 "UNMGRS", "QUADKEY", "UNQUADKEY"
401 };
402 const auto found = std::find(names.begin(), names.end(), a[0]);
403 require(found != names.end());
404 int command = int(found - names.begin());
405 const int lengths[] = {5,3,3,3,3,4,0,5,7,6,4,4,5,3,4,4,3,4,0,6,0,4,2};
406 if (command == 6) require(a.size() >= 3 && a.size() <= 102);
407 else if (command == 18) require(a.size() == 2 || a.size() == 4);
408 else if (command == 20) require(a.size() == 3 || a.size() == 5);
409 else require(int(a.size()) == lengths[command]);
410 if (command <= 13) {
411 int system = systemId(a[1]);
412 if (command == 0) {
413 Integer lat = location(a[2], true), lon = location(a[3], false);
414 int precision = precisionArgument(system, a[4]);
415 string code = encode(system, point(system, precision, lat, lon));
416 return code.empty() ? "OUT OF AREA" : code;
417 }
418 if (command == 8) {
419 int precision = precisionArgument(system, a[2]);
420 Integer lat1 = location(a[3], true), lon1 = location(a[4], false), lat2 = location(a[5], true), lon2 = location(a[6], false);
421 require(lat1 <= lat2);
422 Cell first = point(system, precision, lat1, lon1), last = point(system, precision, lat2, lon2);
423 const Grid &g = grids[system][precision];
424 Integer columns = lon1 <= lon2 ? last.x - first.x + 1 : std::min(g.width, last.x - first.x + 1 + g.width);
425 if ((last.y - first.y + 1) * columns > 100) return "TOO MANY";
426 vector<string> result;
427 for (Integer y = last.y; y >= first.y; --y) {
428 for (Integer i = 0; i < columns; ++i) result.push_back(display(system, {precision, (first.x + i) % g.width, y}));
429 }
430 return join(result);
431 }
432 if (command == 13) {
433 int precision = precisionArgument(system, a[2]);
434 const Grid &g = grids[system][precision];
435 Integer count = system == 5 ? 8000 * meshScale[precision] * meshScale[precision] : g.width * g.height;
436 return "w=" + coordinate(360000000 / g.width) + " h=" + coordinate(180000000 / g.height) + " cells=" + std::to_string(count);
437 }
438 if (command == 6) return compact(system, vector<string>(a.begin() + 2, a.end()));
439 Cell c = decode(system, a[2]);
440 const Grid &g = grids[system][c.precision];
441 if (command == 1) {
442 Integer south = -90000000 + 180000000 * c.y / g.height, west = -180000000 + 360000000 * c.x / g.width;
443 Integer north = -90000000 + 180000000 * (c.y + 1) / g.height, east = -180000000 + 360000000 * (c.x + 1) / g.width;
444 Integer cy = floorRatio(centerNumerator(c.y, g.height, 180000000), 2 * g.height), cx = floorRatio(centerNumerator(c.x, g.width, 360000000), 2 * g.width);
445 return "sw=" + coordinate(south) + "," + coordinate(west) + " ne=" + coordinate(north) + "," + coordinate(east) + " c=" + coordinate(cy) + "," + coordinate(cx);
446 }
447 if (command == 2) {
448 const int dx[] = {0,1,1,1,0,-1,-1,-1}, dy[] = {1,1,0,-1,-1,-1,0,1};
449 vector<string> result;
450 for (int i = 0; i < 8; ++i) result.push_back(c.y + dy[i] < 0 || c.y + dy[i] >= g.height ? "-" : display(system, {c.precision, modulo(c.x + dx[i], g.width), c.y + dy[i]}));
451 return join(result);
452 }
453 if (command == 3) return c.precision == 0 ? "NONE" : encode(system, ancestor(system, c, c.precision - 1));
454 if (command == 4) {
455 if (c.precision + 1 == int(grids[system].size())) return "NONE";
456 const Grid &next = grids[system][c.precision + 1];
457 Integer rx = next.width / g.width, ry = next.height / g.height;
458 if (rx * ry > 100) return "TOO MANY";
459 vector<string> result;
460 for (Integer j = ry - 1; j >= 0; --j) {
461 for (Integer i = 0; i < rx; ++i) result.push_back(display(system, {c.precision + 1, rx * c.x + i, ry * c.y + j}));
462 }
463 return join(result);
464 }
465 if (command == 7) {
466 Integer lat = location(a[3], true), lon = location(a[4], false);
467 return point(system, c.precision, lat, lon) == c ? "YES" : "NO";
468 }
469 if (command == 9) {
470 require(a[3] == "TO");
471 int destination = systemId(a[4]), precision = precisionArgument(destination, a[5]);
472 const Grid &next = grids[destination][precision];
473 string code = encode(destination, {precision, (2 * c.x + 1) * next.width / (2 * g.width), (2 * c.y + 1) * next.height / (2 * g.height)});
474 return code.empty() ? "OUT OF AREA" : code;
475 }
476 if (command == 12) return relation(system, c, systemId(a[3]), decode(systemId(a[3]), a[4]));
477 Cell other = decode(system, a[3]);
478 if (command == 5) {
479 for (int precision = std::min(c.precision, other.precision); precision >= 0; --precision) {
480 Cell parent = ancestor(system, c, precision);
481 if (parent == ancestor(system, other, precision)) return encode(system, parent);
482 }
483 return "NONE";
484 }
485 if (c.precision != other.precision) return "MISMATCH";
