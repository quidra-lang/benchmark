25const SYSTEMS: [&str; 6] = ["GEOHASH", "MAIDENHEAD", "PLUSCODE", "GARS", "GEOREF", "MESH"];
26const HASH: &[u8] = b"0123456789bcdefghjkmnpqrstuvwxyz";

37fn floor_ratio(a: Integer, b: Integer) -> Integer { a.div_euclid(b) }
38fn modulo(a: Integer, b: Integer) -> Integer { a.rem_euclid(b) }
39fn padded(value: Integer, width: usize) -> String { format!("{value:0width$}") }
40fn coordinate(value: Integer) -> String {
41 let magnitude = value.abs();
42 format!("{}{}.{:06}", if value < 0 { "-" } else { "" }, magnitude / 1_000_000, magnitude % 1_000_000)
43}
44fn digits(s: &str, low: usize, high: usize) -> bool {
45 s.len() >= low && s.len() <= high && s.bytes().all(|c| c.is_ascii_digit())
46}
47fn integer(s: &str, low: Integer, high: Integer) -> Answer<Integer> {
48 let negative = s.starts_with('-');
49 let body = if negative { &s[1..] } else { s };
50 require!(digits(body, 1, 9));
51 let value: Integer = s.parse().unwrap();
52 require!(value >= low && value <= high);
53 Ok(value)
54}
55fn location(s: &str, latitude: bool) -> Answer<Integer> {
56 let negative = s.starts_with('-');
57 let body = if negative { &s[1..] } else { s };
58 let parts: Vec<&str> = body.split('.').collect();
59 require!(parts.len() == 1 || parts.len() == 2);
60 require!(digits(parts[0], 1, 3));
61 let mut fraction = if parts.len() == 2 { parts[1].to_owned() } else { String::new() };
62 if parts.len() == 2 { require!(digits(&fraction, 1, 6)); }
63 let mut value = integer(parts[0], 0, 999)? * 1_000_000;
64 fraction.push_str(&"0".repeat(6 - fraction.len()));
65 value += integer(&fraction, 0, 999999)?;
66 if negative { value = -value; }
67 require!(if latitude { (-90000000..=90000000).contains(&value) } else { (-180000000..180000000).contains(&value) });
68 Ok(value)
69}
70fn system_id(s: &str) -> Answer<usize> { SYSTEMS.iter().position(|name| *name == s).ok_or(()) }
71fn digit(alphabet: &[u8], c: u8) -> Answer<Integer> {
72 alphabet.iter().position(|v| *v == c).map(|i| i as Integer).ok_or(())
73}
74fn center_numerator(position: Integer, divisions: Integer, span: Integer) -> Integer {
75 span * (2 * position + 1 - divisions)
76}
77fn compare(a: Integer, b: Integer, c: Integer, d: Integer) -> std::cmp::Ordering {
78 (a * d).cmp(&(c * b))
79}
80fn mgrs_columns(zone: Integer) -> &'static [u8] {
81 match zone % 3 {
82 1 => b"ABCDEFGH",
83 2 => b"JKLMNPQR",
84 _ => b"STUVWXYZ",
85 }
86}
87fn square_digits(a: &[&str], start: usize, e: &mut Integer, n: &mut Integer) -> Answer<Integer> {
88 if a.len() == start { return Ok(100000); }
89 require!(digits(a[start], 1, 5) && digits(a[start + 1], 1, 5) && a[start].len() == a[start + 1].len());
90 let scale = 10_i64.pow((5 - a[start].len()) as u32);
91 *e += integer(a[start], 0, 99999)? * scale;
92 *n += integer(a[start + 1], 0, 99999)? * scale;
93 Ok(scale)
94}
95impl Codec {
96 fn new() -> Self {
97 let mut grids: [Vec<Grid>; 6] = Default::default();
98 for p in 1..=9 {
99 grids[0].push(Grid { precision: p, width: 1 << ((5 * p + 1) / 2), height: 1 << (5 * p / 2) });
100 }
101 let definitions: [&[(Integer, Integer, Integer)]; 5] = [
102 &[(2,18,18), (4,180,180), (6,4320,4320), (8,43200,43200)],
103 &[(2,18,9), (4,360,180), (6,7200,3600), (8,144000,72000), (10,2880000,1440000), (11,11520000,7200000)],
104 &[(5,720,360), (6,1440,720), (7,4320,2160)],
105 &[(2,24,12), (4,360,180), (8,21600,10800), (10,216000,108000), (12,2160000,1080000)],
106 &[(4,360,270), (6,2880,2160), (8,28800,21600), (9,57600,43200), (10,115200,86400), (11,230400,172800)],
107 ];
108 for (i, entries) in definitions.iter().enumerate() {
109 for &(precision, width, height) in entries.iter() { grids[i + 1].push(Grid { precision, width, height }); }
110 }
111 Self { grids }
112 }
113 fn precision_id(&self, system: usize, p: Integer) -> Answer<usize> {
114 self.grids[system].iter().position(|g| g.precision == p).ok_or(())
115 }
116 fn precision_argument(&self, system: usize, s: &str) -> Answer<usize> {
117 self.precision_id(system, integer(s, 0, 999999999)?)
118 }
119 fn point(&self, system: usize, precision: usize, latitude: Integer, longitude: Integer) -> Cell {
120 let g = self.grids[system][precision];
121 Cell { precision, x: (longitude + 180000000) * g.width / 360000000, y: ((latitude + 90000000) * g.height / 180000000).min(g.height - 1) }
122 }
123 fn ancestor(&self, system: usize, c: Cell, precision: usize) -> Cell {
124 let g = self.grids[system][c.precision];
125 let parent = self.grids[system][precision];
126 Cell { precision, x: c.x * parent.width / g.width, y: c.y * parent.height / g.height }
127 }
128 fn encode(&self, system: usize, c: Cell) -> String {
129 let p = self.grids[system][c.precision].precision as usize;
130 let (mut x, mut y) = (c.x, c.y);
131 let mut result = String::new();
132 match system {
133 0 => {
134 let (a, b, mut value) = ((5 * p + 1) / 2, 5 * p / 2, 0);
135 for i in 0..5 * p {
136 let bit = if i % 2 == 0 { x >> (a - 1 - i / 2) & 1 } else { y >> (b - 1 - i / 2) & 1 };
137 value = value * 2 + bit;
138 if i % 5 == 4 {
139 result.push(HASH[value as usize] as char);
140 value = 0;
141 }
142 }
143 }
144 1 | 2 => {
145 let count = if system == 1 { p / 2 } else { p.min(10) / 2 };
146 let tail = if system == 2 && p == 11 {
147 let value = 4 * (y % 5) + x % 4;
148 x /= 4;
149 y /= 5;
150 Some(value)
151 } else { None };
152 let (mut xd, mut yd) = (vec![0; count], vec![0; count]);
153 for i in (0..count).rev() {
154 let rx = if system == 1 { MAIDEN_ALPHABETS[i].len() as Integer } else if i == 0 { 18 } else { 20 };
155 let ry = if system == 1 { rx } else if i == 0 { 9 } else { 20 };
156 xd[i] = (x % rx) as usize;
157 yd[i] = (y % ry) as usize;
158 x /= rx;
159 y /= ry;
160 }
161 for i in 0..count {
162 if system == 1 {
163 result.push(MAIDEN_ALPHABETS[i][xd[i]] as char);
164 result.push(MAIDEN_ALPHABETS[i][yd[i]] as char);
165 } else {
166 result.push(PLUS[yd[i]] as char);
167 result.push(PLUS[xd[i]] as char);
168 }
169 }
170 if let Some(value) = tail { result.push(PLUS[value as usize] as char); }
171 if system == 2 {
172 if p < 8 { result.push_str(&"0".repeat(8 - p)); }
173 result.insert(8, '+');
174 }
175 }
176 3 => {
177 let scale = if p == 5 { 1 } else if p == 6 { 2 } else { 6 };
178 let base_y = y / scale;
179 result = padded(x / scale + 1, 3);
180 result.push(LETTERS[(base_y / 24) as usize] as char);
181 result.push(LETTERS[(base_y % 24) as usize] as char);
182 if p >= 6 {
183 let (xx, yy) = if p == 7 { (x / 3, y / 3) } else { (x, y) };
184 result.push((b'1' + (xx % 2 + 2 * (1 - yy % 2)) as u8) as char);
185 }
186 if p == 7 { result.push((b'1' + (x % 3 + 3 * (2 - y % 3)) as u8) as char); }
187 }
188 4 => {
189 if p == 2 { return format!("{}{}", LETTERS[x as usize] as char, LETTERS[y as usize] as char); }
190 let m = (p - 4) / 2;
191 let scale = if p < 8 { 1 } else if m == 2 { 60 } else if m == 3 { 600 } else { 6000 };
192 let (xx, yy) = (x / scale, y / scale);
193 for value in [xx / 15, yy / 15, xx % 15, yy % 15] { result.push(LETTERS[value as usize] as char); }
194 if p >= 8 { result.push_str(&(padded(x % scale, m) + &padded(y % scale, m))); }
195 }
196 _ => {
197 let scale = MESH_SCALE[c.precision];
198 x -= 280 * scale;
199 y -= 135 * scale;
200 if x < 0 || x >= 80 * scale || y < 0 || y >= 100 * scale { return result; }
201 let count = c.precision + 1;
202 let radices = [0, 8, 10, 2, 2, 2];
203 let (mut xd, mut yd) = (vec![0; count], vec![0; count]);
204 for i in (1..count).rev() {
205 xd[i] = x % radices[i];
206 yd[i] = y % radices[i];
207 x /= radices[i];
208 y /= radices[i];
209 }
210 result = padded(y, 2) + &padded(x, 2);
211 for i in 1..count {
212 if i <= 2 {
213 result.push((b'0' + yd[i] as u8) as char);
214 result.push((b'0' + xd[i] as u8) as char);
215 } else { result.push((b'1' + (xd[i] + 2 * yd[i]) as u8) as char); }
216 }
217 }
218 }
219 result
220 }
221 fn decode(&self, system: usize, code: &str) -> Answer<Cell> {
222 let bytes = code.as_bytes();
223 let mut p = bytes.len();
224 let (mut x, mut y) = (0, 0);
225 match system {
226 0 => {
227 require!((1..=9).contains(&p));
228 let mut position = 0;
229 for &c in bytes {
230 let value = digit(HASH, c)?;
231 for bit in (0..=4).rev() {
232 if position % 2 == 0 {
233 x = 2 * x + (value >> bit & 1);
234 } else {
235 y = 2 * y + (value >> bit & 1);
236 }
237 position += 1;
238 }
239 }
240 }
241 1 => {
242 self.precision_id(system, p as Integer)?;
243 for i in 0..p / 2 {
244 x = x * MAIDEN_ALPHABETS[i].len() as Integer + digit(MAIDEN_ALPHABETS[i], bytes[2 * i])?;
245 y = y * MAIDEN_ALPHABETS[i].len() as Integer + digit(MAIDEN_ALPHABETS[i], bytes[2 * i + 1])?;
246 }
247 }
248 2 => {
249 require!(p >= 9 && bytes[8] == b'+' && bytes.iter().filter(|&&c| c == b'+').count() == 1);
250 let mut prefix = 0;
251 while prefix < 8 && bytes[prefix] != b'0' {
252 digit(PLUS, bytes[prefix])?;
253 prefix += 1;
254 }
255 require!([2, 4, 6, 8].contains(&prefix));
256 require!(bytes[prefix..8].iter().all(|&c| c == b'0'));
257 let suffix = p - 9;
258 require!(if prefix == 8 { [0, 2, 3].contains(&suffix) } else { suffix == 0 });
259 require!(digit(PLUS, bytes[0])? <= 8 && digit(PLUS, bytes[1])? <= 17);
260 let sequence: Vec<u8> = bytes[..prefix].iter().chain(bytes[9..].iter()).copied().collect();
261 p = sequence.len();
262 for i in 0..p.min(10) / 2 {
263 y = y * (if i == 0 { 9 } else { 20 }) + digit(PLUS, sequence[2 * i])?;
264 x = x * (if i == 0 { 18 } else { 20 }) + digit(PLUS, sequence[2 * i + 1])?;
265 }
266 if p == 11 {
267 let tail = digit(PLUS, sequence[10])?;
268 x = x * 4 + tail % 4;
269 y = y * 5 + tail / 4;
270 }
271 }
272 3 => {
273 require!((5..=7).contains(&p) && digits(&code[..3], 3, 3));
274 x = integer(&code[..3], 1, 720)? - 1;
275 let (a, b) = (digit(LETTERS, bytes[3])?, digit(LETTERS, bytes[4])?);
276 require!(a < 15);
277 y = 24 * a + b;
278 if p >= 6 {
279 require!((b'1'..=b'4').contains(&bytes[5]));
280 let q = Integer::from(bytes[5] - b'1');
281 x = 2 * x + q % 2;
282 y = 2 * y + 1 - q / 2;
283 }
284 if p == 7 {
285 require!((b'1'..=b'9').contains(&bytes[6]));
286 let q = Integer::from(bytes[6] - b'1');
287 x = 3 * x + q % 3;
288 y = 3 * y + 2 - q / 3;
289 }
290 }
291 4 => {
292 self.precision_id(system, p as Integer)?;
293 x = digit(LETTERS, bytes[0])?;
294 y = digit(LETTERS, bytes[1])?;
295 require!(y < 12);
296 if p >= 4 {
297 let (a, b) = (digit(LETTERS, bytes[2])?, digit(LETTERS, bytes[3])?);
298 require!(a < 15 && b < 15);
299 x = 15 * x + a;
300 y = 15 * y + b;
301 }
302 if p >= 8 {
303 let m = (p - 4) / 2;
304 let scale = if m == 2 { 60 } else if m == 3 { 600 } else { 6000 };
305 require!(digits(&code[4..4 + m], m, m) && digits(&code[4 + m..], m, m));
306 x = x * scale + integer(&code[4..4 + m], 0, scale - 1)?;
307 y = y * scale + integer(&code[4 + m..], 0, scale - 1)?;
308 }
309 }
310 _ => {
311 let precision = self.precision_id(system, p as Integer)?;
312 require!(digits(code, p, p));
313 y = integer(&code[..2], 0, 99)?;
314 x = integer(&code[2..4], 0, 79)?;
315 if p >= 6 {
316 require!(bytes[4] <= b'7' && bytes[5] <= b'7');
317 y = y * 8 + Integer::from(bytes[4] - b'0');
318 x = x * 8 + Integer::from(bytes[5] - b'0');
319 }
320 if p >= 8 {
321 y = y * 10 + Integer::from(bytes[6] - b'0');
322 x = x * 10 + Integer::from(bytes[7] - b'0');
323 }
324 for &c in &bytes[8.min(p)..] {
325 require!((b'1'..=b'4').contains(&c));
326 let q = Integer::from(c - b'1');
327 x = x * 2 + q % 2;
328 y = y * 2 + q / 2;
329 }
330 x += 280 * MESH_SCALE[precision];
331 y += 135 * MESH_SCALE[precision];
332 }
333 }
334 Ok(Cell { precision: self.precision_id(system, p as Integer)?, x, y })
335 }
336 fn display(&self, system: usize, c: Cell) -> String {
337 let code = self.encode(system, c);
338 if code.is_empty() { "-".to_owned() } else { code }
339 }
340 fn relation(&self, sa: usize, a: Cell, sb: usize, b: Cell) -> &'static str {
341 use std::cmp::Ordering::{Equal, Greater, Less};
342 let (ga, gb) = (self.grids[sa][a.precision], self.grids[sb][b.precision]);
343 let west = compare(a.x, ga.width, b.x, gb.width);
344 let east = compare(a.x + 1, ga.width, b.x + 1, gb.width);
345 let south = compare(a.y, ga.height, b.y, gb.height);
346 let north = compare(a.y + 1, ga.height, b.y + 1, gb.height);
347 if [west, east, south, north].iter().all(|v| *v == Equal) { return "EQUAL"; }
348 if west != Less && east != Greater && south != Less && north != Greater { return "INSIDE"; }
349 if west != Greater && east != Less && south != Greater && north != Less { return "CONTAINS"; }
350 let x1 = compare(a.x, ga.width, b.x + 1, gb.width);
351 let x2 = compare(b.x, gb.width, a.x + 1, ga.width);
352 let y1 = compare(a.y, ga.height, b.y + 1, gb.height);
353 let y2 = compare(b.y, gb.height, a.y + 1, ga.height);
354 if [x1, x2, y1, y2].iter().all(|v| *v == Less) { return "OVERLAP"; }
355 let wrap = a.x == 0 && b.x + 1 == gb.width || b.x == 0 && a.x + 1 == ga.width;
356 if y1 != Greater && y2 != Greater && (x1 != Greater && x2 != Greater || wrap) { return "TOUCH"; }
357 "DISJOINT"
358 }
359 fn prune(&self, system: usize, cells: &mut BTreeSet<Cell>) {
360 let removed: Vec<Cell> = cells.iter().copied().filter(|&c| cells.iter().any(|&other| other.precision < c.precision && self.ancestor(system, c, other.precision) == other)).collect();
361 for c in removed { cells.remove(&c); }
362 }
363 fn compact(&self, system: usize, codes: &[&str]) -> Answer<String> {
364 let mut cells = BTreeSet::new();
365 for code in codes { cells.insert(self.decode(system, code)?); }
366 self.prune(system, &mut cells);
367 loop {
368 let mut merge = None;
369 for &c in &cells {
370 if c.precision == 0 { continue; }
371 let parent = self.ancestor(system, c, c.precision - 1);
372 let (g, pg) = (self.grids[system][c.precision], self.grids[system][parent.precision]);
373 let needed = g.width / pg.width * (g.height / pg.height);
374 if needed > cells.len() as Integer { continue; }
375 let children: Vec<Cell> = cells.iter().copied().filter(|&other| other.precision == c.precision && self.ancestor(system, other, parent.precision) == parent).collect();
376 if children.len() as Integer == needed {
377 merge = Some((parent, children));
378 break;
379 }
380 }
381 if let Some((parent, children)) = merge {
382 for child in children { cells.remove(&child); }
383 cells.insert(parent);
384 self.prune(system, &mut cells);
385 } else { break; }
386 }
387 let mut ordered: Vec<Cell> = cells.into_iter().collect();
388 ordered.sort_by_key(|c| (c.precision, -c.y, c.x));
389 Ok(ordered.iter().map(|&c| self.encode(system, c)).collect::<Vec<_>>().join(" "))
390 }
391 fn execute(&self, a: &[&str]) -> Answer<String> {
392 const NAMES: [&str; 23] = [
393 "ENCODE", "DECODE", "NEIGHBORS", "PARENT", "CHILDREN", "COMMON",
394 "COMPACT", "CONTAINS", "COVER", "CONVERT", "STEPS", "PATH", "RELATE",
395 "SIZE", "SHORTEN", "RECOVER", "UTMZONE", "OSGB", "UNOSGB", "MGRS",
396 "UNMGRS", "QUADKEY", "UNQUADKEY",
397 ];
398 const LENGTHS: [usize; 23] = [5,3,3,3,3,4,0,5,7,6,4,4,5,3,4,4,3,4,0,6,0,4,2];
399 let command = NAMES.iter().position(|name| *name == a[0]).ok_or(())?;
400 require!(match command {
401 6 => (3..=102).contains(&a.len()),
402 18 => a.len() == 2 || a.len() == 4,
403 20 => a.len() == 3 || a.len() == 5,
404 _ => a.len() == LENGTHS[command],
405 });
406 if command <= 13 {
407 let system = system_id(a[1])?;
408 if command == 0 {
409 let (lat, lon) = (location(a[2], true)?, location(a[3], false)?);
410 let precision = self.precision_argument(system, a[4])?;
411 let code = self.encode(system, self.point(system, precision, lat, lon));
412 return Ok(if code.is_empty() { "OUT OF AREA".to_owned() } else { code });
413 }
414 if command == 8 {
415 let precision = self.precision_argument(system, a[2])?;
416 let (lat1, lon1, lat2, lon2) = (location(a[3], true)?, location(a[4], false)?, location(a[5], true)?, location(a[6], false)?);
417 require!(lat1 <= lat2);
418 let (first, last) = (self.point(system, precision, lat1, lon1), self.point(system, precision, lat2, lon2));
419 let g = self.grids[system][precision];
420 let columns = if lon1 <= lon2 { last.x - first.x + 1 } else { g.width.min(last.x - first.x + 1 + g.width) };
421 if (last.y - first.y + 1) * columns > 100 { return Ok("TOO MANY".to_owned()); }
422 let mut result = Vec::new();
423 for y in (first.y..=last.y).rev() {
424 for i in 0..columns { result.push(self.display(system, Cell { precision, x: (first.x + i) % g.width, y })); }
425 }
426 return Ok(result.join(" "));
427 }
428 if command == 13 {
429 let precision = self.precision_argument(system, a[2])?;
430 let g = self.grids[system][precision];
431 let count = if system == 5 { 8000 * MESH_SCALE[precision] * MESH_SCALE[precision] } else { g.width * g.height };
432 return Ok(format!("w={} h={} cells={}", coordinate(360000000 / g.width), coordinate(180000000 / g.height), count));
433 }
434 if command == 6 { return self.compact(system, &a[2..]); }
435 let c = self.decode(system, a[2])?;
436 let g = self.grids[system][c.precision];
437 match command {
438 1 => {
439 let (south, west) = (-90000000 + 180000000 * c.y / g.height, -180000000 + 360000000 * c.x / g.width);
440 let (north, east) = (-90000000 + 180000000 * (c.y + 1) / g.height, -180000000 + 360000000 * (c.x + 1) / g.width);
441 let (cy, cx) = (floor_ratio(center_numerator(c.y, g.height, 180000000), 2 * g.height), floor_ratio(center_numerator(c.x, g.width, 360000000), 2 * g.width));
442 return Ok(format!("sw={},{} ne={},{} c={},{}", coordinate(south), coordinate(west), coordinate(north), coordinate(east), coordinate(cy), coordinate(cx)));
443 }
444 2 => {
445 let dx = [0,1,1,1,0,-1,-1,-1];
446 let dy = [1,1,0,-1,-1,-1,0,1];
447 let result: Vec<String> = (0..8).map(|i| {
448 if c.y + dy[i] < 0 || c.y + dy[i] >= g.height { "-".to_owned() }
449 else { self.display(system, Cell { precision: c.precision, x: modulo(c.x + dx[i], g.width), y: c.y + dy[i] }) }
450 }).collect();
451 return Ok(result.join(" "));
452 }
453 3 => return Ok(if c.precision == 0 { "NONE".to_owned() } else { self.encode(system, self.ancestor(system, c, c.precision - 1)) }),
454 4 => {
455 if c.precision + 1 == self.grids[system].len() { return Ok("NONE".to_owned()); }
456 let next = self.grids[system][c.precision + 1];
457 let (rx, ry) = (next.width / g.width, next.height / g.height);
458 if rx * ry > 100 { return Ok("TOO MANY".to_owned()); }
459 let mut result = Vec::new();
460 for j in (0..ry).rev() {
461 for i in 0..rx { result.push(self.display(system, Cell { precision: c.precision + 1, x: rx * c.x + i, y: ry * c.y + j })); }
462 }
463 return Ok(result.join(" "));
464 }
465 7 => {
466 let (lat, lon) = (location(a[3], true)?, location(a[4], false)?);
467 return Ok((if self.point(system, c.precision, lat, lon) == c { "YES" } else { "NO" }).to_owned());
468 }
469 9 => {
470 require!(a[3] == "TO");
471 let destination = system_id(a[4])?;
472 let precision = self.precision_argument(destination, a[5])?;
473 let next = self.grids[destination][precision];
474 let code = self.encode(destination, Cell { precision, x: (2 * c.x + 1) * next.width / (2 * g.width), y: (2 * c.y + 1) * next.height / (2 * g.height) });
475 return Ok(if code.is_empty() { "OUT OF AREA".to_owned() } else { code });
476 }
477 12 => {
478 let other_system = system_id(a[3])?;
479 return Ok(self.relation(system, c, other_system, self.decode(other_system, a[4])?).to_owned());
480 }
481 _ => {}
482 }
483 let other = self.decode(system, a[3])?;
484 if command == 5 {
485 for precision in (0..=c.precision.min(other.precision)).rev() {
486 let parent = self.ancestor(system, c, precision);
487 if parent == self.ancestor(system, other, precision) { return Ok(self.encode(system, parent)); }
488 }
489 return Ok("NONE".to_owned());
490 }
491 if c.precision != other.precision { return Ok("MISMATCH".to_owned()); }
492 let (mut dx, dy) = (modulo(other.x - c.x, g.width), other.y - c.y);
493 if dx > g.width / 2 { dx -= g.width; }
494 if command == 10 { return Ok(format!("dx={} dy={}", dx, dy)); }
495 let n = dx.abs().max(dy.abs());
496 if n + 1 > 100 { return Ok("TOO MANY".to_owned()); }
497 let mut result = Vec::new();
498 for i in 0..=n {
499 let x = c.x + if n == 0 { 0 } else { floor_ratio(2 * i * dx + n, 2 * n) };
500 let y = c.y + if n == 0 { 0 } else { floor_ratio(2 * i * dy + n, 2 * n) };
501 result.push(self.display(system, Cell { precision: c.precision, x: modulo(x, g.width), y }));
502 }
503 return Ok(result.join(" "));
504 }
505 if command == 14 || command == 15 {
506 let (lat, lon) = (location(a[2], true)?, location(a[3], false)?);
507 let mut full = a[1].to_owned();
508 let mut removed = 0;
509 if command == 15 {
510 let plus = full.find('+').ok_or(())?;
511 require!((plus == 2 || plus == 4) && full.bytes().filter(|&c| c == b'+').count() == 1);
512 let suffix = full.len() - plus - 1;
513 require!([0, 2, 3].contains(&suffix));
514 for c in full.bytes().filter(|&c| c != b'+') { digit(PLUS, c)?; }
515 removed = 8 - plus;
516 let prefix = self.encode(2, self.point(2, self.precision_id(2, 10)?, lat, lon));
517 full = prefix[..removed].to_owned() + &full;
518 }
519 let mut c = self.decode(2, &full)?;
520 let g = self.grids[2][c.precision];
521 require!(g.precision >= 8);
522 let (cy, cx) = (center_numerator(c.y, g.height, 180000000), center_numerator(c.x, g.width, 360000000));
523 let (yd, xd) = (2 * g.height, 2 * g.width);
524 if command == 14 {
525 if (cy - lat * yd).abs() < 15000 * yd && (cx - lon * xd).abs() < 15000 * xd { return Ok(full[6..].to_owned()); }
526 if (cy - lat * yd).abs() < 300000 * yd && (cx - lon * xd).abs() < 300000 * xd { return Ok(full[4..].to_owned()); }
527 return Ok(full);
528 }
529 let distance = if removed == 4 { 1000000 } else { 50000 };
530 let (my, mx) = (distance * g.height / 180000000, distance * g.width / 360000000);
531 if (lat + distance / 2) * yd < cy && cy - distance * yd >= -90000000 * yd { c.y -= my; }
532 else if (lat - distance / 2) * yd > cy && cy + distance * yd <= 90000000 * yd { c.y += my; }
533 if (lon + distance / 2) * xd < cx { c.x = modulo(c.x - mx, g.width); }
534 else if (lon - distance / 2) * xd > cx { c.x = modulo(c.x + mx, g.width); }
535 return Ok(self.encode(2, c));
536 }
537 if command == 16 {
538 let (lat, lon) = (location(a[1], true)?, location(a[2], false)?);
539 if !(-80000000..84000000).contains(&lat) { return Ok("NONE".to_owned()); }
540 let band = BANDS[((lat + 80000000) / 8000000).min(19) as usize];
541 let mut zone = (lon + 180000000) / 6000000 + 1;
542 if band == b'V' && (3000000..12000000).contains(&lon) { zone = 32; }
543 if band == b'X' && (0..42000000).contains(&lon) {
544 zone = if lon < 9000000 { 31 } else if lon < 21000000 { 33 } else if lon < 33000000 { 35 } else { 37 };
545 }
546 return Ok(format!("{}{}", zone, band as char));
547 }
548 if command == 17 || command == 19 {
549 let (e, n, d);
550 let mut prefix;
551 if command == 17 {
552 e = integer(a[1], 0, 699999)?;
553 n = integer(a[2], 0, 1299999)?;
554 d = integer(a[3], 0, 5)? as usize;
555 prefix = format!("{}{}", OS_LETTERS[(5 * (3 - n / 500000) + 2 + e / 500000) as usize] as char, OS_LETTERS[(5 * (4 - n / 100000 % 5) + e / 100000 % 5) as usize] as char);
556 } else {
557 let zone = integer(a[1], 1, 60)?;
558 require!(a[2].len() == 1);
559 digit(BANDS, a[2].as_bytes()[0])?;
560 e = integer(a[3], 100000, 899999)?;
561 n = integer(a[4], 0, 9999999)?;
562 d = integer(a[5], 0, 5)? as usize;
563 prefix = format!("{}{} {}{}", zone, a[2], mgrs_columns(zone)[(e / 100000 - 1) as usize] as char, MGRS_ROWS[((n / 100000 + if zone % 2 == 0 { 5 } else { 0 }) % 20) as usize] as char);
564 }
565 if d > 0 { prefix.push_str(&format!(" {} {}", &padded(e % 100000, 5)[..d], &padded(n % 100000, 5)[..d])); }
566 return Ok(prefix);
567 }
568 if command == 18 || command == 20 {
569 let (mut e, mut n, scale);
570 let mut prefix = String::new();
571 if command == 18 {
572 require!(a[1].len() == 2);
573 let (first, second) = (digit(OS_LETTERS, a[1].as_bytes()[0])?, digit(OS_LETTERS, a[1].as_bytes()[1])?);
574 e = 500000 * (first % 5 - 2) + 100000 * (second % 5);
575 n = 500000 * (3 - first / 5) + 100000 * (4 - second / 5);
576 require!((0..700000).contains(&e) && (0..1300000).contains(&n));
577 scale = square_digits(a, 2, &mut e, &mut n)?;
578 } else {
579 require!(a[1].len() == 2 || a[1].len() == 3);
580 let zone_text = &a[1][..a[1].len() - 1];
581 require!(digits(zone_text, 1, 2) && zone_text.as_bytes()[0] != b'0');
582 let zone = integer(zone_text, 1, 60)?;
583 digit(BANDS, a[1].as_bytes()[a[1].len() - 1])?;
584 require!(a[2].len() == 2);
585 e = 100000 * (digit(mgrs_columns(zone), a[2].as_bytes()[0])? + 1);
586 n = 100000 * modulo(digit(MGRS_ROWS, a[2].as_bytes()[1])? - (if zone % 2 == 0 { 5 } else { 0 }), 20);
587 scale = square_digits(a, 3, &mut e, &mut n)?;
588 prefix = a[1].to_owned() + " ";
589 }
590 return Ok(format!("{}sw={},{} size={}", prefix, e, n, scale));
591 }
592 if command == 21 {
593 let z = integer(a[3], 1, 23)?;
594 let (x, y) = (integer(a[1], 0, (1 << z) - 1)?, integer(a[2], 0, (1 << z) - 1)?);
595 let key: String = (0..z).rev().map(|i| (b'0' + ((x >> i & 1) + 2 * (y >> i & 1)) as u8) as char).collect();
596 return Ok(key);
597 }
598 require!((1..=23).contains(&a[1].len()));
599 let (mut x, mut y) = (0, 0);
600 for c in a[1].bytes() {
601 require!((b'0'..=b'3').contains(&c));
602 let q = Integer::from(c - b'0');
603 x = 2 * x + q % 2;
604 y = 2 * y + q / 2;
605 }
606 Ok(format!("x={} y={} z={}", x, y, a[1].len()))
607 }
608}
609fn main() {
610 let mut input = String::new();
611 io::stdin().read_to_string(&mut input).unwrap();
612 let mut lines = input.lines();
613 let count: usize = lines.next().unwrap().parse().unwrap();
614 let codec = Codec::new();
615 let mut output = String::new();
616 for _ in 0..count {
617 let arguments: Vec<&str> = lines.next().unwrap().split(' ').collect();
618 let answer = codec.execute(&arguments).unwrap_or_else(|_| "INVALID".to_owned());
619 output.push_str(&answer);
620 output.push('\n');
621 }
622 print!("{}", output);
623}
