22int coordinate(String text, bool latitude) {

25 var n = int.parse(parts[0]) * 1000000 + (parts.length == 1 ? 0 : int.parse(parts[1].padRight(6, '0')));
26 if (negative) n = -n;
27 if (latitude ? n < -90000000 || n > 90000000 : n < -180000000 || n >= 180000000) invalid();

51(int, int) grid(int s, int p) {
52 return switch (s) {
53 0 => (1 << ((5 * p + 1) ~/ 2), 1 << (5 * p ~/ 2)),
54 1 => {2: (18, 18), 4: (180, 180), 6: (4320, 4320), 8: (43200, 43200)}[p]!,
55 2 => {2: (18, 9), 4: (360, 180), 6: (7200, 3600), 8: (144000, 72000), 10: (2880000, 1440000), 11: (11520000, 7200000)}[p]!,
56 3 => {5: (720, 360), 6: (1440, 720), 7: (4320, 2160)}[p]!,
57 4 => {2: (24, 12), 4: (360, 180), 8: (21600, 10800), 10: (216000, 108000), 12: (2160000, 1080000)}[p]!,
58 _ => (360 * meshScale(p), 270 * meshScale(p)),

61int meshScale(int p) => {4: 1, 6: 8, 8: 80, 9: 160, 10: 320, 11: 640}[p]!;
62List<int> radixDigits(int value, List<int> radices) {
63 final result = List.filled(radices.length, 0);
64 for (var i = radices.length - 1; i >= 0; i--) {
65 result[i] = value % radices[i];
66 value ~/= radices[i];
67 }
68 return result;

79Cell containing(int s, int p, int lat, int lon) {

83String? encode(Cell c) {
84 var x = c.x, y = c.y;
85 final p = c.p, result = StringBuffer();
86 switch (c.s) {
87 case 0:
88 var xb = (5 * p + 1) ~/ 2, yb = 5 * p ~/ 2, value = 0;
89 for (var i = 0; i < 5 * p; i++) {
90 final bit = i % 2 == 0 ? (x >> --xb) & 1 : (y >> --yb) & 1;
91 value = 2 * value + bit;
92 if (i % 5 == 4) {
93 result.write(geohash[value]);
94 value = 0;
95 }
96 }
97 case 1:
98 final radices = [18, 10, 24, 10].take(p ~/ 2).toList();
99 final xs = radixDigits(x, radices), ys = radixDigits(y, radices);
100 for (var i = 0; i < xs.length; i++) result.write('${maidenSets[i][xs[i]]}${maidenSets[i][ys[i]]}');
101 case 2:
102 final last = p == 11 ? 4 * (y % 5) + x % 4 : -1;
103 if (p == 11) {
104 x ~/= 4;
105 y ~/= 5;
106 }
107 final k = (p == 11 ? 10 : p) ~/ 2;
108 final xs = radixDigits(x, [18, ...List.filled(k - 1, 20)]), ys = radixDigits(y, [9, ...List.filled(k - 1, 20)]);
109 for (var i = 0; i < k; i++) result.write('${plusDigits[ys[i]]}${plusDigits[xs[i]]}');
110 if (last >= 0) result.write(plusDigits[last]);
111 final text = result.toString();
112 return p < 8 ? '${text.padRight(8, '0')}+' : '${text.substring(0, 8)}+${text.substring(8)}';
113 case 3:
114 final area = p == 7 ? 1 + x % 3 + 3 * (2 - y % 3) : 0;
115 if (p == 7) {
116 x ~/= 3;
117 y ~/= 3;
118 }
119 final quadrant = p >= 6 ? 1 + x % 2 + 2 * (1 - y % 2) : 0;
120 if (p >= 6) {
121 x ~/= 2;
122 y ~/= 2;
123 }
124 result.write('${decimal(x + 1, 3)}${l24[y ~/ 24]}${l24[y % 24]}');
125 if (p >= 6) result.write(quadrant);
126 if (p == 7) result.write(area);
127 case 4:
128 final m = (p - 4) ~/ 2, scale = p >= 8 ? 6 * pow(10, m - 1).toInt() : 1;
129 final east = x % scale, north = y % scale;
130 if (p >= 8) {
131 x ~/= scale;
132 y ~/= scale;
133 }
134 final a = p >= 4 ? x ~/ 15 : x, b = p >= 4 ? y ~/ 15 : y;
135 result.write('${l24[a]}${l24[b]}');
136 if (p >= 4) result.write('${l24[x % 15]}${l24[y % 15]}');
137 if (p >= 8) result.write('${decimal(east, m)}${decimal(north, m)}');
138 case 5:
139 final scale = meshScale(p), u = x - 280 * scale, v = y - 135 * scale;
140 if (u < 0 || u >= 80 * scale || v < 0 || v >= 100 * scale) return null;
141 final count = precisions[5].indexOf(p) + 1;
142 final us = radixDigits(u, [80, 8, 10, 2, 2, 2].take(count).toList());
143 final vs = radixDigits(v, [100, 8, 10, 2, 2, 2].take(count).toList());
144 result.write('${decimal(vs[0], 2)}${decimal(us[0], 2)}');
145 for (var i = 1; i < count; i++) result.write(i <= 2 ? '${vs[i]}${us[i]}' : '${1 + us[i] + 2 * vs[i]}');
146 }
147 return result.toString();
148}
149Cell decode(int s, String code) {
150 final p = s == 2 ? 0 : code.length;
151 if (s != 2 && !precisions[s].contains(p)) invalid();
152 var x = 0, y = 0, actual = p;
153 switch (s) {
154 case 0:
155 var bitIndex = 0;
156 for (final ch in code.split('')) {
157 final value = index(geohash, ch);
158 for (var b = 4; b >= 0; b--) {
159 if (bitIndex++ % 2 == 0)
160 x = 2 * x + ((value >> b) & 1);
161 else
162 y = 2 * y + ((value >> b) & 1);
163 }
164 }
165 case 1:
166 const radices = [18, 10, 24, 10];
167 for (var i = 0; i < p ~/ 2; i++) {
168 x = x * radices[i] + index(maidenSets[i], code[2 * i]);
169 y = y * radices[i] + index(maidenSets[i], code[2 * i + 1]);
170 }
171 case 2:
172 if (code.length < 9 || code[8] != '+' || '+'.allMatches(code).length != 1) invalid();
173 final zero = code.substring(0, 8).indexOf('0'), before = zero < 0 ? 8 : zero;
174 if (![2, 4, 6, 8].contains(before)) invalid();
175 if (zero >= 0 && (code.substring(before, 8) != '0' * (8 - before) || code.length != 9)) invalid();
176 if (zero < 0 && ![0, 2, 3].contains(code.length - 9)) invalid();
177 final text = code.substring(0, before) + code.substring(9);
178 actual = text.length;
179 final values = text.split('').map((c) => index(plusDigits, c)).toList();
180 if (values[0] > 8 || values[1] > 17) invalid();
181 final pairs = (actual == 11 ? 10 : actual) ~/ 2;
182 for (var i = 0; i < pairs; i++) {
183 y = y * (i == 0 ? 9 : 20) + values[2 * i];
184 x = x * (i == 0 ? 18 : 20) + values[2 * i + 1];
185 }
186 if (actual == 11) {
187 x = 4 * x + values.last % 4;
188 y = 5 * y + values.last ~/ 4;
189 }
190 case 3:
191 final band = digits(code.substring(0, 3));
192 if (band < 1 || band > 720) invalid();
193 final a = index(l24, code[3]), b = index(l24, code[4]);
194 if (a >= 15) invalid();
195 x = band - 1;
196 y = a * 24 + b;
197 if (p >= 6) {
198 final q = index('1234', code[5]);
199 x = 2 * x + q % 2;
200 y = 2 * y + 1 - q ~/ 2;
201 }
202 if (p == 7) {
203 final q = index('123456789', code[6]);
204 x = 3 * x + q % 3;
205 y = 3 * y + 2 - q ~/ 3;
206 }
207 case 4:
208 x = index(l24, code[0]);
209 y = index(l24, code[1]);
210 if (y >= 12) invalid();
211 if (p >= 4) {
212 final a = index(l24, code[2]), b = index(l24, code[3]);
213 if (a >= 15 || b >= 15) invalid();
214 x = 15 * x + a;
215 y = 15 * y + b;
216 }
217 if (p >= 8) {
218 final m = (p - 4) ~/ 2, scale = 6 * pow(10, (p - 4) ~/ 2 - 1).toInt();
219 final a = digits(code.substring(4, 4 + m)), b = digits(code.substring(4 + m));
220 if (a >= scale || b >= scale) invalid();
221 x = x * scale + a;
222 y = y * scale + b;
223 }
224 case 5:
225 digits(code);
226 y = int.parse(code.substring(0, 2));
227 x = int.parse(code.substring(2, 4));
228 if (x > 79) invalid();
229 if (p >= 6) {
230 final a = digits(code[4]), b = digits(code[5]);
231 if (a > 7 || b > 7) invalid();
232 y = 8 * y + a;
233 x = 8 * x + b;
234 }
235 if (p >= 8) {
236 y = 10 * y + digits(code[6]);
237 x = 10 * x + digits(code[7]);
238 }
239 for (var i = 8; i < p; i++) {
240 final q = index('1234', code[i]);
241 x = 2 * x + q % 2;
242 y = 2 * y + q ~/ 2;
243 }
244 x += 280 * meshScale(p);
245 y += 135 * meshScale(p);
246 }
247 return Cell(s, actual, x, y);
248}

297 final dy = d.y - c.y;
298 if (op == 'STEPS') return 'dx=$dx dy=$dy';
299 final n = max(dx.abs(), dy.abs());
300 if (n + 1 > 100) return 'TOO MANY';
301 return [for (var i = 0; i <= n; i++) encode(Cell(c.s, c.p, (c.x + (n == 0 ? 0 : rounded(i * dx, n))) % c.w, c.y + (n == 0 ? 0 : rounded(i * dy, n)))) ?? '-'].join(' ');

303 if (op == 'COMPACT') {

306 for (final code in a.skip(2)) {
307 final c = decode(s, code);

313 cells.remove(c.key);

315 }
316 }

320 changed = false;

329 cells[parent.key] = parent;

336 return ordered.map((c) => encode(c)!).join(' ');

339 arity(4);
