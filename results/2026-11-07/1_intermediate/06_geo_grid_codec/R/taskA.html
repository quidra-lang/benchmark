75encode <- function(sys, p, x, y) {
76 if (sys == "GEOHASH") {
77 a <- fd(5*p+1, 2)
78 b <- fd(5*p, 2)
79 value <- 0
80 out <- ""
81 for (i in seq_len(5*p) - 1) {
82 if (i %% 2 == 0) {
83 a <- a - 1
84 bit <- md(fd(x, 2^a), 2)
85 } else {
86 b <- b - 1
87 bit <- md(fd(y, 2^b), 2)
88 }
89 value <- 2*value + bit
90 if (i %% 5 == 4) {
91 out <- paste0(out, char_at(gh, value))
92 value <- 0
93 }
94 }
95 return(out)
96 }
97 if (sys == "MAIDENHEAD") {
98 k <- p/2
99 radices <- c(18,10,24,10)[seq_len(k)]
100 xs <- radix_digits(x, radices)
101 ys <- radix_digits(y, radices)
102 return(paste0(char_at(mh[seq_len(k)], xs), char_at(mh[seq_len(k)], ys), collapse = ""))
103 }
104 if (sys == "PLUSCODE") {
105 tail <- ""
106 if (p == 11) {
107 tail <- char_at(pc, 4*md(y,5) + md(x,4))
108 x <- fd(x,4)
109 y <- fd(y,5)
110 }
111 k <- fd(min(p,10),2)
112 radices <- rep(20, k)
113 radices[1L] <- 18
114 xs <- radix_digits(x, radices)
115 radices[1L] <- 9
116 ys <- radix_digits(y, radices)
117 out <- paste0(paste0(char_at(pc, ys), char_at(pc, xs), collapse = ""), tail)
118 if (p < 8) return(paste0(out, strrep("0", 8-p), "+"))
119 return(paste0(substr(out,1,8), "+", substring(out,9)))
120 }
121 if (sys == "GARS") {
122 tail <- ""
123 if (p == 7) {
124 tail <- num(1 + md(x,3) + 3*(2-md(y,3)))
125 x <- fd(x,3)
126 y <- fd(y,3)
127 }
128 if (p >= 6) {
129 tail <- paste0(num(1 + md(x,2) + 2*(1-md(y,2))), tail)
130 x <- fd(x,2)
131 y <- fd(y,2)
132 }
133 return(paste0(sprintf("%03d", as.integer(x+1)), char_at(l24,fd(y,24)), char_at(l24,md(y,24)), tail))
134 }
135 if (sys == "GEOREF") {
136 tail <- ""
137 if (p >= 8) {
138 m <- (p-4)/2
139 scale <- 6*10^(m-1)
140 tail <- sprintf("%0*d%0*d", as.integer(m), as.integer(md(x,scale)), as.integer(m), as.integer(md(y,scale)))
141 x <- fd(x,scale)
142 y <- fd(y,scale)
143 }
144 if (p >= 4) {
145 tail <- paste0(char_at(l24,md(x,15)), char_at(l24,md(y,15)), tail)
146 x <- fd(x,15)
147 y <- fd(y,15)
148 }
149 return(paste0(char_at(l24,x), char_at(l24,y), tail))
150 }
151 level <- match(p, precs$MESH)
152 scale <- c(1,8,80,160,320,640)[level]
153 u <- x - 280*scale
154 v <- y - 135*scale
155 if (u < 0 || u >= 80*scale || v < 0 || v >= 100*scale) return("-")
156 radices <- c(80,8,10,2,2,2)[seq_len(level)]
157 us <- radix_digits(u, radices)
158 radices[1L] <- 100
159 vs <- radix_digits(v, radices)
160 out <- sprintf("%02d%02d", as.integer(vs[1L]), as.integer(us[1L]))
161 if (level >= 2L) {
162 for (i in 2:level) {
163 if (i <= 3L) out <- paste0(out, num(vs[i]), num(us[i])) else out <- paste0(out, num(1+us[i]+2*vs[i]))
164 }
165 }
166 out

168decode <- function(sys, code) {
169 p <- nchar(code, type = "bytes")
170 x <- y <- 0
171 if (sys != "PLUSCODE" && !(p %in% precs[[sys]])) invalid()
172 if (sys == "GEOHASH") {
173 bit_index <- 0
174 for (char in chars(code)) {
175 value <- index_char(gh, char)
176 for (i in 4:0) {
177 bit <- md(fd(value,2^i),2)
178 if (bit_index %% 2 == 0) x <- 2*x+bit else y <- 2*y+bit
179 bit_index <- bit_index + 1
180 }
181 }
182 } else if (sys == "MAIDENHEAD") {
183 for (i in seq_len(p/2)) {
184 radix <- nchar(mh[i])
185 x <- x*radix + index_char(mh[i], substr(code,2*i-1,2*i-1))
186 y <- y*radix + index_char(mh[i], substr(code,2*i,2*i))
187 }
188 } else if (sys == "PLUSCODE") {
189 plus <- gregexpr("+", code, fixed = TRUE)[[1L]]
190 if (length(plus) != 1L || plus[1L] != 9L) invalid()
191 before <- substr(code,1,8)
192 after <- substring(code,10)
193 zero <- regexpr("0", before, fixed = TRUE)[1L]
194 prefix <- if (zero < 0L) 8L else zero-1L
195 if (!(prefix %in% c(2,4,6,8))) invalid()
196 if (prefix < 8L && (substring(before,prefix+1L) != strrep("0",8L-prefix) || after != "")) invalid()
197 if (prefix == 8L && !(nchar(after) %in% c(0,2,3))) invalid()
198 sequence <- paste0(substr(before,1,prefix),after)
199 p <- nchar(sequence)
200 values <- vapply(chars(sequence), function(char) index_char(pc,char), numeric(1L))
201 if (values[1L] > 8 || values[2L] > 17) invalid()
202 for (i in seq_len(fd(min(p,10),2))) {
203 y <- y*(if (i == 1L) 9 else 20) + values[2*i-1]
204 x <- x*(if (i == 1L) 18 else 20) + values[2*i]
205 }
206 if (p == 11L) {
207 x <- 4*x + md(values[11L],4)
208 y <- 5*y + fd(values[11L],4)
209 }
210 } else if (sys == "GARS") {
211 if (!grepl("^[0-9]{3}", code, perl = TRUE)) invalid()
212 x <- as.numeric(substr(code,1,3)) - 1
213 y <- 24*index_char(substr(l24,1,15),substr(code,4,4)) + index_char(l24,substr(code,5,5))
214 if (p >= 6L) {
215 q <- index_char("1234",substr(code,6,6))
216 x <- 2*x + md(q,2)
217 y <- 2*y + 1-fd(q,2)
218 }
219 if (p == 7L) {
220 q <- index_char("123456789",substr(code,7,7))
221 x <- 3*x + md(q,3)
222 y <- 3*y + 2-fd(q,3)
223 }
224 } else if (sys == "GEOREF") {
225 x <- index_char(l24,substr(code,1,1))
226 y <- index_char(substr(l24,1,12),substr(code,2,2))
227 if (p >= 4L) {
228 x <- 15*x + index_char(substr(l24,1,15),substr(code,3,3))
229 y <- 15*y + index_char(substr(l24,1,15),substr(code,4,4))
230 }
231 if (p >= 8L) {
232 m <- (p-4)/2
233 tail <- substring(code,5)
234 if (!grepl("^[0-9]+$",tail,perl=TRUE)) invalid()
235 scale <- 6*10^(m-1)
236 e <- as.numeric(substr(tail,1,m))
237 n <- as.numeric(substring(tail,m+1))
238 if (e >= scale || n >= scale) invalid()
239 x <- x*scale + e
240 y <- y*scale + n
241 }
242 } else {
243 if (!grepl("^[0-9]+$",code,perl=TRUE)) invalid()
244 level <- match(p,precs$MESH)
245 u <- as.numeric(substr(code,3,4))
246 v <- as.numeric(substr(code,1,2))
247 if (u > 79) invalid()
248 if (level >= 2L) {
249 for (i in 2:level) {
250 if (i <= 3L) {
251 base <- if (i == 2L) 8 else 10
252 a <- as.numeric(substr(code,2*i+1,2*i+1))
253 b <- as.numeric(substr(code,2*i+2,2*i+2))
254 if (a >= base || b >= base) invalid()
255 v <- v*base + a
256 u <- u*base + b
257 } else {
258 q <- index_char("1234",substr(code,i+5,i+5))
259 u <- 2*u + md(q,2)
260 v <- 2*v + fd(q,2)
261 }
262 }
263 }
264 scale <- c(1,8,80,160,320,640)[level]
265 x <- u + 280*scale
266 y <- v + 135*scale
267 }
268 size <- grid(sys,p)
269 if (x < 0 || x >= size[1L] || y < 0 || y >= size[2L] || encode(sys,p,x,y) != code) invalid()
270 c(p,x,y)
271}

296compact_cells <- function(sys,cells) {
297 set <- new.env(hash=TRUE,parent=emptyenv())
298 for (cell in cells) set[[cell_key(cell)]] <- cell
299 repeat {
300 changed <- FALSE
301 for (key in ls(set,all.names=TRUE)) {
302 cell <- set[[key]]
303 index <- match(cell[1L],precs[[sys]])
304 for (i in seq_len(index-1L)) {
305 if (!is.null(set[[cell_key(ancestor(sys,cell,precs[[sys]][i]))]])) {
306 rm(list=key,envir=set)
307 changed <- TRUE
308 break
309 }
310 }
311 }
312 for (index in rev(seq_len(length(precs[[sys]])-1L)+1L)) {
313 p <- precs[[sys]][index]
314 q <- precs[[sys]][index-1L]
315 size <- grid(sys,p)
316 target <- grid(sys,q)
317 needed <- fd(size[1L],target[1L])*fd(size[2L],target[2L])
318 groups <- list()
319 parents <- list()
320 for (key in ls(set,all.names=TRUE)) {
321 cell <- set[[key]]
322 if (cell[1L] == p) {
323 parent <- ancestor(sys,cell,q)
324 pk <- cell_key(parent)
325 groups[[pk]] <- c(groups[[pk]],key)
326 parents[[pk]] <- parent
327 }
328 }
329 for (key in names(groups)) {
330 if (length(groups[[key]]) == needed) {
331 rm(list=groups[[key]],envir=set)
332 set[[key]] <- parents[[key]]
333 changed <- TRUE
334 }
335 }
336 }
337 if (!changed) break
338 }
339 cells <- lapply(ls(set,all.names=TRUE),function(key) set[[key]])
340 p <- vapply(cells,function(cell) cell[1L],numeric(1L))
341 x <- vapply(cells,function(cell) cell[2L],numeric(1L))
342 y <- vapply(cells,function(cell) cell[3L],numeric(1L))
343 cells[order(p,-y,x,method="radix")]

345execute <- function(line) {
346 all <- strsplit(line," ",fixed=TRUE)[[1L]]
347 command <- all[1L]
348 t <- all[-1L]

352 if (command == "COMPACT") {

354 } else if (is.null(counts[[command]]) || !(length(t) %in% counts[[command]])) invalid()

365 } else if (lon < 21000000) {
366 zone <- 33
367 } else if (lon < 33000000) {

370 zone <- 37

416 y <- integer_arg(t[2L],0,2^z-1)

438 y <- cell[3L]

440 size <- grid("PLUSCODE",p)

442 h <- size[2L]
443 dy <- abs((-90000000-lat)*2*h+180000000*(2*y+1))
444 dx <- abs((-180000000-lon)*2*w+360000000*(2*x+1))

446 if (dy < 300000*2*h && dx < 300000*2*w) return(substring(t[1L],5))

449 if (!grepl("^([23456789CFGHJMPQRVWX]{2}|[23456789CFGHJMPQRVWX]{4})\\+([23456789CFGHJMPQRVWX]{2,3})?$",t[1L],perl=TRUE)) invalid()
450 r <- 9L-regexpr("+",t[1L],fixed=TRUE)[1L]

455 x <- cell[2L]
456 y <- cell[3L]

458 w <- size[1L]
459 h <- size[2L]
460 d <- if (r == 4L) 1000000 else 50000
461 my <- fd(d*h,180000000)
462 mx <- fd(d*w,360000000)
463 cy <- 180000000*(2*y+1)
464 cx <- 360000000*(2*x+1)
465 if ((lat+d/2+90000000)*2*h < cy && cy >= d*2*h) {
466 y <- y-my
467 } else if ((lat-d/2+90000000)*2*h > cy && cy <= (180000000-d)*2*h) {
468 y <- y+my
469 }
470 if ((lon+d/2+180000000)*2*w < cx) {
471 x <- md(x-mx,w)
472 } else if ((lon-d/2+180000000)*2*w > cx) {
473 x <- md(x+mx,w)
474 }
475 return(encode("PLUSCODE",p,x,y))
476 }
477 sys <- system_name(t[1L])
478 if (command %in% c("SIZE","ENCODE","COVER")) {
479 p <- precision(sys,t[if (command == "ENCODE") 4L else 2L])
480 size <- grid(sys,p)
481 w <- size[1L]
482 h <- size[2L]
483 if (command == "SIZE") {
484 cells <- w*h
485 if (sys == "MESH") {
486 scale <- c(1,8,80,160,320,640)[match(p,precs[[sys]])]
487 cells <- 8000*scale*scale
488 }
489 return(paste0("w=",coord_text(fd(360000000,w))," h=",coord_text(fd(180000000,h))," cells=",num(cells)))
490 }
491 if (command == "ENCODE") {
492 point <- point_cell(sys,p,coordinate(t[2L],TRUE),coordinate(t[3L],FALSE))
493 code <- encode(sys,p,point[1L],point[2L])
