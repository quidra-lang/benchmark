5private const val GH = "0123456789bcdefghjkmnpqrstuvwxyz"
6private const val L24 = "ABCDEFGHJKLMNPQRSTUVWXYZ"
7private const val PLUS = "23456789CFGHJMPQRVWX"
8private const val G25 = "ABCDEFGHJKLMNOPQRSTUVWXYZ"
9private const val BANDS = "CDEFGHJKLMNPQRSTUVWX"
10private const val MROWS = "ABCDEFGHJKLMNPQRSTUV"
11private val precisionTable = mapOf(
12 "GEOHASH" to intArrayOf(1,2,3,4,5,6,7,8,9), "MAIDENHEAD" to intArrayOf(2,4,6,8),
13 "PLUSCODE" to intArrayOf(2,4,6,8,10,11), "GARS" to intArrayOf(5,6,7),
14 "GEOREF" to intArrayOf(2,4,8,10,12), "MESH" to intArrayOf(4,6,8,9,10,11)

16private val counts = mapOf("ENCODE" to 4,"DECODE" to 2,"NEIGHBORS" to 2,"PARENT" to 2,"CHILDREN" to 2,
17 "COMMON" to 3,"CONTAINS" to 4,"COVER" to 6,"CONVERT" to 5,"STEPS" to 3,"PATH" to 3,
18 "RELATE" to 4,"SIZE" to 2,"SHORTEN" to 3,"RECOVER" to 3)
19private val meshScale = longArrayOf(1,8,80,160,320,640)
20private val maiden = arrayOf("ABCDEFGHIJKLMNOPQR","0123456789","abcdefghijklmnopqrstuvwx","0123456789")
21private data class Cell(val system: String, val level: Int, val x: Long, val y: Long, val w: Long, val h: Long)
22private class Invalid : RuntimeException()
23private fun need(condition: Boolean) { if (!condition) throw Invalid() }
24private fun integer(s: String): Long {
25 need(s.matches(Regex("-?[0-9]{1,9}")))
26 return s.toLong()
27}
28private fun range(s: String, low: Long, high: Long): Long = integer(s).also { need(it in low..high) }
29private fun coordinate(s: String, latitude: Boolean): Long {
30 need(s.matches(Regex("-?[0-9]{1,3}(\\.[0-9]{1,6})?")))
31 val negative = s.startsWith('-')
32 val pieces = s.removePrefix("-").split('.')
33 var value = pieces[0].toLong()*1000000
34 if (pieces.size == 2) value += pieces[1].padEnd(6,'0').toLong()
35 if (negative) value = -value
36 need(if (latitude) value in -90000000L..90000000L else value >= -180000000 && value < 180000000)
37 return value
38}
39private fun precisions(system: String): IntArray = precisionTable[system] ?: throw Invalid()
40private fun level(system: String, precision: Long): Int {
41 val result = precisions(system).indexOf(precision.toInt())
42 need(result >= 0)
43 return result
44}
45private fun grid(system: String, level: Int, x: Long = 0, y: Long = 0): Cell {
46 val p = precisions(system)[level]
47 val widths = when (system) {
48 "GEOHASH" -> longArrayOf(1L shl ((5*p+1)/2),1L shl (5*p/2))
49 "MAIDENHEAD" -> longArrayOf(longArrayOf(18,180,4320,43200)[level],longArrayOf(18,180,4320,43200)[level])
50 "PLUSCODE" -> longArrayOf(longArrayOf(18,360,7200,144000,2880000,11520000)[level],longArrayOf(9,180,3600,72000,1440000,7200000)[level])
51 "GARS" -> longArrayOf(longArrayOf(720,1440,4320)[level],longArrayOf(360,720,2160)[level])
52 "GEOREF" -> longArrayOf(longArrayOf(24,360,21600,216000,2160000)[level],longArrayOf(12,180,10800,108000,1080000)[level])
53 else -> longArrayOf(360*meshScale[level],270*meshScale[level])
54 }
55 return Cell(system,level,x,y,widths[0],widths[1])
56}
57private fun point(system: String, level: Int, lat: Long, lon: Long): Cell {
58 val g = grid(system,level)
59 return grid(system,level,(lon+180000000)*g.w/360000000,min((lat+90000000)*g.h/180000000,g.h-1))
60}
61private fun digit(c: Char, alphabet: String): Int = alphabet.indexOf(c).also { need(it >= 0) }
62private fun digits(s: String): Long {
63 need(s.isNotEmpty() && s.all { it in '0'..'9' })
64 return s.toLong()
65}
66private fun pad(n: Long, width: Int) = n.toString().padStart(width,'0')
67private fun power(n: Int): Long {
68 var value = 1L
69 repeat(n) { value *= 10 }
70 return value
71}
72private fun radix(value: Long, bases: LongArray, count: Int): LongArray {
73 var rest = value
74 val result = LongArray(count)
75 for (i in count-1 downTo 0) {
76 result[i] = rest%bases[i]
77 rest /= bases[i]
78 }
79 return result
80}
81private fun code(c: Cell): String? {
82 val p = precisions(c.system)[c.level]
83 var x = c.x
84 var y = c.y
85 val result = StringBuilder()
86 when (c.system) {
87 "GEOHASH" -> {
88 var ax = (5*p+1)/2
89 var ay = 5*p/2
90 var group = 0
91 for (bit in 0 until 5*p) {
92 val b = if (bit%2 == 0) (x shr --ax) and 1 else (y shr --ay) and 1
93 group = group*2+b.toInt()
94 if (bit%5 == 4) {
95 result.append(GH[group])
96 group = 0
97 }
98 }
99 }
100 "MAIDENHEAD" -> {
101 val a = radix(x,longArrayOf(18,10,24,10),p/2)
102 val b = radix(y,longArrayOf(18,10,24,10),p/2)
103 for (i in a.indices) result.append(maiden[i][a[i].toInt()]).append(maiden[i][b[i].toInt()])
104 }
105 "PLUSCODE" -> {
106 val extra = if (p == 11) (4*(y%5)+x%4).toInt() else -1
107 if (p == 11) {
108 x /= 4
109 y /= 5
110 }
111 val count = min(p,10)/2
112 val a = radix(x,longArrayOf(18,20,20,20,20),count)
113 val b = radix(y,longArrayOf(9,20,20,20,20),count)
114 for (i in 0 until count) result.append(PLUS[b[i].toInt()]).append(PLUS[a[i].toInt()])
115 if (extra >= 0) result.append(PLUS[extra])
116 val text = result.toString()
117 return if (p < 8) text+"0".repeat(8-p)+"+" else text.take(8)+"+"+text.drop(8)
118 }
119 "GARS" -> {
120 val scale = if (c.level == 0) 1 else if (c.level == 1) 2 else 6
121 val bx = x/scale
122 val by = y/scale
123 result.append(pad(bx+1,3)).append(L24[(by/24).toInt()]).append(L24[(by%24).toInt()])
124 if (p >= 6) {
125 val divisor = if (p == 7) 3 else 1
126 result.append(1+x/divisor%2+2*(1-y/divisor%2))
127 }
128 if (p == 7) result.append(1+x%3+3*(2-y%3))
129 }
130 "GEOREF" -> {
131 if (p == 2) return "${L24[x.toInt()]}${L24[y.toInt()]}"
132 val m = (p-4)/2
133 val scale = if (p == 4) 1L else 6*power(m-1)
134 val bx = x/scale
135 val by = y/scale
136 result.append(L24[(bx/15).toInt()]).append(L24[(by/15).toInt()])
137 .append(L24[(bx%15).toInt()]).append(L24[(by%15).toInt()])
138 if (p >= 8) result.append(pad(x%scale,m)).append(pad(y%scale,m))
139 }
140 else -> {
141 val scale = meshScale[c.level]
142 val u = x-280*scale
143 val v = y-135*scale
144 if (u < 0 || u >= 80*scale || v < 0 || v >= 100*scale) return null
145 val count = c.level+1
146 val a = radix(u,longArrayOf(80,8,10,2,2,2),count)
147 val b = radix(v,longArrayOf(100,8,10,2,2,2),count)
148 result.append(pad(b[0],2)).append(pad(a[0],2))
149 for (i in 1 until count) {
150 if (i <= 2) result.append(b[i]).append(a[i])
151 else result.append(1+a[i]+2*b[i])
152 }
153 }
154 }
155 return result.toString()
156}
157private fun decode(system: String, text: String): Cell {
158 precisions(system)
159 var p = text.length
160 var x = 0L
161 var y = 0L
162 when (system) {
163 "GEOHASH" -> {
164 need(p in 1..9)
165 var bit = 0
166 for (ch in text) {
167 val d = digit(ch,GH)
168 for (i in 4 downTo 0) {
169 if (bit%2 == 0) x = x*2+((d shr i) and 1)
170 else y = y*2+((d shr i) and 1)
171 bit++
172 }
173 }
174 }
175 "MAIDENHEAD" -> {
176 level(system,p.toLong())
177 for (i in 0 until p/2) {
178 x = x*maiden[i].length+digit(text[2*i],maiden[i])
179 y = y*maiden[i].length+digit(text[2*i+1],maiden[i])
180 }
181 }
182 "PLUSCODE" -> {
183 need(p >= 9 && text.indexOf('+') == 8 && text.lastIndexOf('+') == 8)
184 val firstZero = text.take(8).indexOf('0')
185 val before = if (firstZero < 0) 8 else firstZero
186 val after = p-9
187 need(before in listOf(2,4,6,8))
188 need(if (before == 8) after in listOf(0,2,3) else after == 0)
189 for (i in before until 8) need(text[i] == '0')
190 val sequence = text.take(before)+text.drop(9)
191 p = sequence.length
192 need(digit(sequence[0],PLUS) <= 8 && digit(sequence[1],PLUS) <= 17)
193 for (i in 0 until min(p,10) step 2) {
194 y = y*20+digit(sequence[i],PLUS)
195 x = x*20+digit(sequence[i+1],PLUS)
196 }
197 if (p == 11) {
198 val d = digit(sequence[10],PLUS)
199 x = x*4+d%4
200 y = y*5+d/4
201 }
202 }
203 "GARS" -> {
204 need(p in 5..7)
205 val e = digits(text.take(3))
206 need(e in 1L..720L)
207 x = e-1
208 y = (digit(text[3],L24.take(15))*24+digit(text[4],L24)).toLong()
209 if (p >= 6) {
210 val q = digit(text[5],"1234")
211 x = x*2+q%2
212 y = y*2+1-q/2
213 }
214 if (p == 7) {
215 val q = digit(text[6],"123456789")
216 x = x*3+q%3
217 y = y*3+2-q/3
218 }
219 }
220 "GEOREF" -> {
221 level(system,p.toLong())
222 x = digit(text[0],L24).toLong()
223 y = digit(text[1],L24.take(12)).toLong()
224 if (p >= 4) {
225 x = x*15+digit(text[2],L24.take(15))
226 y = y*15+digit(text[3],L24.take(15))
227 }
228 if (p >= 8) {
229 val m = (p-4)/2
230 val scale = 6*power(m-1)
231 val a = digits(text.substring(4,4+m))
232 val b = digits(text.drop(4+m))
233 need(a < scale && b < scale)
234 x = x*scale+a
235 y = y*scale+b
236 }
237 }
238 else -> {
239 val l = level(system,p.toLong())
240 digits(text)
241 var v = digits(text.take(2))
242 var u = digits(text.substring(2,4))
243 need(u < 80)
244 if (p >= 6) {
245 v = v*8+digit(text[4],"01234567")
246 u = u*8+digit(text[5],"01234567")
247 }
248 if (p >= 8) {
249 v = v*10+digit(text[6],"0123456789")
250 u = u*10+digit(text[7],"0123456789")
251 }
252 for (i in 8 until p) {
253 val q = digit(text[i],"1234")
254 v = v*2+q/2
255 u = u*2+q%2
256 }
257 x = u+280*meshScale[l]
258 y = v+135*meshScale[l]
259 }
260 }
261 val result = grid(system,level(system,p.toLong()),x,y)
262 need(text == code(result))
263 return result
264}
265private fun ancestor(c: Cell, level: Int): Cell {
266 val g = grid(c.system,level)
267 return grid(c.system,level,c.x*g.w/c.w,c.y*g.h/c.h)
268}
269private fun shown(c: Cell) = code(c) ?: "-"
270private fun coordinateFormat(n: Long) = (if (n < 0) "-" else "")+"${abs(n)/1000000}.${pad(abs(n)%1000000,6)}"
271private fun boundary(c: Cell, latitude: Boolean, offset: Int, center: Boolean): Long {
272 val extent = if (latitude) 180000000L else 360000000L
273 val total = if (latitude) c.h else c.w
274 val value = if (latitude) c.y else c.x
275 return -extent/2+if (center) extent*(2*value+1)/(2*total) else extent*(value+offset)/total
276}
277private fun compare(a: Long, ad: Long, b: Long, bd: Long) = (a*bd).compareTo(b*ad)
278private fun inside(a: Cell, b: Cell) = compare(a.x,a.w,b.x,b.w) >= 0 && compare(a.x+1,a.w,b.x+1,b.w) <= 0 &&
279 compare(a.y,a.h,b.y,b.h) >= 0 && compare(a.y+1,a.h,b.y+1,b.h) <= 0
280private fun steps(a: Cell, b: Cell): LongArray {
281 var dx = Math.floorMod(b.x-a.x,a.w)
282 if (dx > a.w/2) dx -= a.w
283 return longArrayOf(dx,b.y-a.y)
284}
285private fun detail(a: String, b: String): LongArray {
286 need(a.matches(Regex("[0-9]{1,5}")) && b.matches(Regex("[0-9]{1,5}")) && a.length == b.length)
287 val size = power(5-a.length)
288 return longArrayOf(digits(a)*size,digits(b)*size,size)
289}
290private fun columns(zone: Long) = if (zone%3 == 1L) "ABCDEFGH" else if (zone%3 == 2L) "JKLMNPQR" else "STUVWXYZ"
291private fun auxiliary(t: List<String>): String {
292 val n = t.size-1
293 when (t[0]) {
294 "UTMZONE" -> {
295 need(n == 2)
296 val lat = coordinate(t[1],true)
297 val lon = coordinate(t[2],false)
298 if (lat < -80000000 || lat >= 84000000) return "NONE"
299 val band = BANDS[min((lat+80000000)/8000000,19).toInt()]
300 var zone = (lon+180000000)/6000000+1
301 if (band == 'V' && lon >= 3000000 && lon < 12000000) zone = 32
302 if (band == 'X') {
303 if (lon >= 0 && lon < 9000000) zone = 31
304 else if (lon >= 9000000 && lon < 21000000) zone = 33
305 else if (lon >= 21000000 && lon < 33000000) zone = 35
306 else if (lon >= 33000000 && lon < 42000000) zone = 37
307 }
308 return "$zone$band"
309 }
310 "OSGB" -> {
311 need(n == 3)
312 val e = range(t[1],0,699999)
313 val north = range(t[2],0,1299999)
314 val d = range(t[3],0,5).toInt()
315 val letters = "${G25[((3-north/500000)*5+2+e/500000).toInt()]}${G25[((4-north/100000%5)*5+e/100000%5).toInt()]}"
316 return letters+if (d == 0) "" else " ${pad(e%100000,5).take(d)} ${pad(north%100000,5).take(d)}"
317 }
318 "UNOSGB" -> {
319 need(n == 1 || n == 3)
320 need(t[1].length == 2)
321 val a = digit(t[1][0],G25)
322 val b = digit(t[1][1],G25)
323 var e = (500000*(a%5-2)+100000*(b%5)).toLong()
324 var north = (500000*(3-a/5)+100000*(4-b/5)).toLong()
325 var size = 100000L
326 need(e >= 0 && e < 700000 && north >= 0 && north < 1300000)
327 if (n == 3) {
328 val values = detail(t[2],t[3])
329 e += values[0]
330 north += values[1]
331 size = values[2]
332 }
333 return "sw=$e,$north size=$size"
334 }
335 "MGRS" -> {
336 need(n == 5)
337 val zone = range(t[1],1,60)
338 val e = range(t[3],100000,899999)
339 val north = range(t[4],0,9999999)
340 need(t[2].length == 1)
341 digit(t[2][0],BANDS)
342 val d = range(t[5],0,5).toInt()
343 val offset = if (zone%2 == 0L) 5 else 0
344 val square = "${columns(zone)[(e/100000-1).toInt()]}${MROWS[((north/100000+offset)%20).toInt()]}"
345 return "$zone${t[2]} $square"+if (d == 0) "" else " ${pad(e%100000,5).take(d)} ${pad(north%100000,5).take(d)}"
346 }
347 "UNMGRS" -> {
348 need(n == 2 || n == 4)
349 need(t[1].matches(Regex("[1-9][0-9]?[CDEFGHJKLMNPQRSTUVWX]")))
350 val zone = t[1].dropLast(1).toLong()
351 need(zone <= 60 && t[2].length == 2)
352 val a = digit(t[2][0],columns(zone))
353 val b = digit(t[2][1],MROWS)
354 val offset = if (zone%2 == 0L) 5 else 0
355 var e = 100000L*(a+1)
356 var north = 100000*Math.floorMod((b-offset).toLong(),20)
357 var size = 100000L
358 if (n == 4) {
359 val values = detail(t[3],t[4])
360 e += values[0]
361 north += values[1]
362 size = values[2]
363 }
364 return "${t[1]} sw=$e,$north size=$size"
365 }
366 "QUADKEY" -> {
367 need(n == 3)
368 val z = range(t[3],1,23).toInt()
369 val x = range(t[1],0,(1L shl z)-1)
370 val y = range(t[2],0,(1L shl z)-1)
371 return (z-1 downTo 0).joinToString("") { (((x shr it) and 1)+2*((y shr it) and 1)).toString() }
372 }
373 "UNQUADKEY" -> {
374 need(n == 1 && t[1].matches(Regex("[0-3]{1,23}")))
375 var x = 0L
376 var y = 0L
377 for (ch in t[1]) {
378 val d = ch-'0'
379 x = x*2+d%2
380 y = y*2+d/2
381 }
382 return "x=$x y=$y z=${t[1].length}"
383 }
384 else -> throw Invalid()
385 }
386}
387private fun execute(line: String): String {
388 val t = line.split(' ')
389 val command = t[0]
390 if (command !in counts && command != "COMPACT") return auxiliary(t)
391 need(if (command == "COMPACT") t.size in 3..102 else t.size == counts.getValue(command)+1)
392 if (command == "SHORTEN" || command == "RECOVER") {
393 val lat = coordinate(t[2],true)
394 val lon = coordinate(t[3],false)
395 if (command == "SHORTEN") {
396 val c = decode("PLUSCODE",t[1])
397 need(precisions(c.system)[c.level] >= 8)
398 val yn = -90000000*2*c.h+180000000*(2*c.y+1)
399 val xn = -180000000*2*c.w+360000000*(2*c.x+1)
400 if (abs(yn-lat*2*c.h) < 15000*2*c.h && abs(xn-lon*2*c.w) < 15000*2*c.w) return t[1].drop(6)
401 if (abs(yn-lat*2*c.h) < 300000*2*c.h && abs(xn-lon*2*c.w) < 300000*2*c.w) return t[1].drop(4)
402 return t[1]
403 }
404 val plus = t[1].indexOf('+')
405 need(plus == 2 || plus == 4)
406 need(t[1].lastIndexOf('+') == plus)
407 need(t[1].length-plus-1 in listOf(0,2,3))
408 for (ch in t[1].replace("+","")) digit(ch,PLUS)
409 val missing = 8-plus
410 val prefix = code(point("PLUSCODE",4,lat,lon))!!.take(missing)
411 val c = decode("PLUSCODE",prefix+t[1])
412 val d = if (missing == 4) 1000000L else 50000L
413 var x = c.x
414 var y = c.y
415 val yn = -90000000*2*c.h+180000000*(2*y+1)
416 val xn = -180000000*2*c.w+360000000*(2*x+1)
417 val my = d*c.h/180000000
418 val mx = d*c.w/360000000
419 if ((lat+d/2)*2*c.h < yn && yn-d*2*c.h >= -90000000*2*c.h) y -= my
420 else if ((lat-d/2)*2*c.h > yn && yn+d*2*c.h <= 90000000*2*c.h) y += my
421 if ((lon+d/2)*2*c.w < xn) x = Math.floorMod(x-mx,c.w)
422 else if ((lon-d/2)*2*c.w > xn) x = Math.floorMod(x+mx,c.w)
423 return code(grid("PLUSCODE",c.level,x,y))!!
424 }
425 val system = t[1]
426 precisions(system)
427 when (command) {
428 "ENCODE" -> return code(point(system,level(system,integer(t[4])),coordinate(t[2],true),coordinate(t[3],false))) ?: "OUT OF AREA"
429 "SIZE" -> {
430 val l = level(system,integer(t[2]))
431 val c = grid(system,l)
432 val count = if (system == "MESH") 8000*meshScale[l]*meshScale[l] else c.w*c.h
433 return "w=${coordinateFormat(360000000/c.w)} h=${coordinateFormat(180000000/c.h)} cells=$count"
434 }
435 "COVER" -> {
436 val l = level(system,integer(t[2]))
437 val lat1 = coordinate(t[3],true)
438 val lon1 = coordinate(t[4],false)
439 val lat2 = coordinate(t[5],true)
440 val lon2 = coordinate(t[6],false)
441 need(lat1 <= lat2)
442 val a = point(system,l,lat1,lon1)
443 val b = point(system,l,lat2,lon2)
444 val columns = if (lon1 <= lon2) b.x-a.x+1 else min(a.w,b.x-a.x+1+a.w)
445 if (columns*(b.y-a.y+1) > 100) return "TOO MANY"
446 val result = mutableListOf<String>()
447 for (y in b.y downTo a.y) {
448 for (i in 0 until columns) result.add(shown(grid(system,l,(a.x+i)%a.w,y)))
449 }
450 return result.joinToString(" ")
451 }
452 "COMPACT" -> {
453 val cells = t.drop(2).map { decode(system,it) }.toMutableSet()
454 val covered = cells.filter { c -> cells.any { a -> a.level < c.level && ancestor(c,a.level) == a } }
455 cells.removeAll(covered.toSet())
456 var changed = true
457 while (changed) {
458 changed = false
459 for (child in cells.toList()) {
460 if (child.level == 0) continue
461 val parent = ancestor(child,child.level-1)
462 val g = grid(system,child.level)
463 val rx = g.w/parent.w
464 val ry = g.h/parent.h
465 if (rx*ry > cells.size) continue
466 val siblings = mutableSetOf<Cell>()
467 for (j in 0 until ry) {
468 for (i in 0 until rx) siblings.add(grid(system,child.level,parent.x*rx+i,parent.y*ry+j))
469 }
470 if (cells.containsAll(siblings)) {
471 cells.removeAll(siblings)
472 cells.add(parent)
473 changed = true
474 break
475 }
476 }
477 }
478 return cells.sortedWith(compareBy<Cell> { it.level }.thenByDescending { it.y }.thenBy { it.x }).joinToString(" ") { code(it)!! }
479 }
480 }
481 val a = decode(system,t[2])
482 when (command) {
483 "DECODE" -> return "sw=${coordinateFormat(boundary(a,true,0,false))},${coordinateFormat(boundary(a,false,0,false))}"+
484 " ne=${coordinateFormat(boundary(a,true,1,false))},${coordinateFormat(boundary(a,false,1,false))}"+
485 " c=${coordinateFormat(boundary(a,true,0,true))},${coordinateFormat(boundary(a,false,0,true))}"
486 "NEIGHBORS" -> {
487 val dx = intArrayOf(0,1,1,1,0,-1,-1,-1)
488 val dy = intArrayOf(1,1,0,-1,-1,-1,0,1)
489 return (0..7).joinToString(" ") { i -> if (a.y+dy[i] < 0 || a.y+dy[i] >= a.h) "-" else shown(grid(system,a.level,Math.floorMod(a.x+dx[i],a.w),a.y+dy[i])) }
490 }
491 "PARENT" -> return if (a.level == 0) "NONE" else code(ancestor(a,a.level-1))!!
492 "CHILDREN" -> {
493 if (a.level+1 == precisions(system).size) return "NONE"
494 val g = grid(system,a.level+1)
495 val rx = g.w/a.w
496 val ry = g.h/a.h
497 if (rx*ry > 100) return "TOO MANY"
498 val result = mutableListOf<String>()
499 for (j in ry-1 downTo 0) {
500 for (i in 0 until rx) result.add(shown(grid(system,g.level,a.x*rx+i,a.y*ry+j)))
501 }
502 return result.joinToString(" ")
503 }
504 "COMMON" -> {
505 val b = decode(system,t[3])
506 for (l in min(a.level,b.level) downTo 0) {
507 val parent = ancestor(a,l)
508 if (parent == ancestor(b,l)) return code(parent)!!
509 }
510 return "NONE"
511 }
512 "CONTAINS" -> return if (a == point(system,a.level,coordinate(t[3],true),coordinate(t[4],false))) "YES" else "NO"
513 "CONVERT" -> {
514 need(t[3] == "TO")
515 val g = grid(t[4],level(t[4],integer(t[5])))
516 return code(grid(g.system,g.level,(2*a.x+1)*g.w/(2*a.w),(2*a.y+1)*g.h/(2*a.h))) ?: "OUT OF AREA"
517 }
518 "STEPS", "PATH" -> {
519 val b = decode(system,t[3])
520 if (a.level != b.level) return "MISMATCH"
521 val delta = steps(a,b)
522 if (command == "STEPS") return "dx=${delta[0]} dy=${delta[1]}"
523 val n = max(abs(delta[0]),abs(delta[1]))
524 if (n+1 > 100) return "TOO MANY"
525 return (0L..n).joinToString(" ") { i ->
526 val dx = if (n == 0L) 0 else Math.floorDiv(2*i*delta[0]+n,2*n)
527 val dy = if (n == 0L) 0 else Math.floorDiv(2*i*delta[1]+n,2*n)
528 shown(grid(system,a.level,Math.floorMod(a.x+dx,a.w),a.y+dy))
529 }
530 }
531 "RELATE" -> {
532 val b = decode(t[3],t[4])
533 val ab = inside(a,b)
534 val ba = inside(b,a)
535 if (ab && ba) return "EQUAL"
536 if (ab) return "INSIDE"
537 if (ba) return "CONTAINS"
538 val ew = compare(a.x+1,a.w,b.x,b.w)
539 val we = compare(b.x+1,b.w,a.x,a.w)
540 val ns = compare(a.y+1,a.h,b.y,b.h)
541 val sn = compare(b.y+1,b.h,a.y,a.h)
542 if (ew > 0 && we > 0 && ns > 0 && sn > 0) return "OVERLAP"
543 val dateline = a.x+1 == a.w && b.x == 0L || b.x+1 == b.w && a.x == 0L
544 return if (ns >= 0 && sn >= 0 && (ew >= 0 && we >= 0 || dateline)) "TOUCH" else "DISJOINT"
545 }
546 else -> throw Invalid()
547 }
548}
549fun main() {
550 val input = System.`in`.bufferedReader()
551 val output = System.out.bufferedWriter()
552 repeat(input.readLine().toInt()) {
553 val result = try { execute(input.readLine()) } catch (_: Invalid) { "INVALID" }
554 output.write(result)
555 output.newLine()
556 }
557 output.flush()
558}
