58def coordinate(text, latitude):
59 require(COORDINATE_PATTERN.fullmatch(text) is not None)
60 negative = text.startswith("-")
61 unsigned = text[1:] if negative else text
62 whole, dot, fraction = unsigned.partition(".")
63 value = int(whole) * 1000000 + int(fraction.ljust(6, "0") if dot else "0")
64 if negative:
65 value = -value
66 if latitude:
67 require(-90000000 <= value <= 90000000)
68 else:
69 require(-180000000 <= value < 180000000)
70 return value

99def radix_digits(value, radices):
100 digits = []
101 for radix in reversed(radices):
102 value, digit = divmod(value, radix)
103 digits.append(digit)
104 return digits[::-1]

107def radix_value(digits, radices):
108 value = 0
109 for digit, radix in zip(digits, radices):
110 value = value * radix + digit
111 return value

114def encode_geohash(level, x, y):
115 longitude_bits = (5 * level + 1) // 2
116 latitude_bits = 5 * level // 2
117 characters = []
118 group = 0
119 for bit in range(5 * level):
120 if bit % 2 == 0:
121 longitude_bits -= 1
122 value = (x >> longitude_bits) & 1
123 else:
124 latitude_bits -= 1
125 value = (y >> latitude_bits) & 1
126 group = 2 * group + value
127 if bit % 5 == 4:
128 characters.append(GEOHASH_DIGITS[group])
129 group = 0
130 return "".join(characters)

133def encode_maidenhead(level, x, y):
134 sets = MAIDENHEAD_SETS[:level // 2]
135 radices = [len(alphabet) for alphabet in sets]
136 xs = radix_digits(x, radices)
137 ys = radix_digits(y, radices)
138 return "".join(alphabet[a] + alphabet[b] for alphabet, a, b in zip(sets, xs, ys))

141def encode_pluscode(level, x, y):
142 suffix = ""
143 if level == 11:
144 x, column = divmod(x, 4)
145 y, row = divmod(y, 5)
146 suffix = PLUS_DIGITS[4 * row + column]
147 pairs = min(level, 10) // 2
148 xs = radix_digits(x, [18] + [20] * (pairs - 1))
149 ys = radix_digits(y, [9] + [20] * (pairs - 1))
150 digits = "".join(PLUS_DIGITS[b] + PLUS_DIGITS[a] for a, b in zip(xs, ys)) + suffix
151 if level < 8:
152 return digits + "0" * (8 - level) + "+"
153 return digits[:8] + "+" + digits[8:]

156def encode_gars(level, x, y):
157 area = ""
158 quadrant = ""
159 if level == 7:
160 x, column = divmod(x, 3)
161 y, row = divmod(y, 3)
162 area = str(1 + column + 3 * (2 - row))
163 if level >= 6:
164 x, column = divmod(x, 2)
165 y, row = divmod(y, 2)
166 quadrant = str(1 + column + 2 * (1 - row))
167 return f"{x + 1:03d}" + L24[y // 24] + L24[y % 24] + quadrant + area

170def encode_georef(level, x, y):
171 minutes = ""
172 if level >= 8:
173 places = (level - 4) // 2
174 radix = 6 * 10 ** (places - 1)
175 x, easting = divmod(x, radix)
176 y, northing = divmod(y, radix)
177 minutes = f"{easting:0{places}d}{northing:0{places}d}"
178 degrees = ""
179 if level >= 4:
180 x, east_degree = divmod(x, 15)
181 y, north_degree = divmod(y, 15)
182 degrees = L24[east_degree] + L24[north_degree]
183 return L24[x] + L24[y] + degrees + minutes

186def encode_mesh(level, x, y):
187 scale = MESH_SCALES[level]
188 u = x - 280 * scale
189 v = y - 135 * scale
190 if not (0 <= u < 80 * scale and 0 <= v < 100 * scale):
191 return None
192 count = PRECISIONS["MESH"].index(level) + 1
193 us = radix_digits(u, (80, 8, 10, 2, 2, 2)[:count])
194 vs = radix_digits(v, (100, 8, 10, 2, 2, 2)[:count])
195 code = f"{vs[0]:02d}{us[0]:02d}"
196 if count >= 2:
197 code += str(vs[1]) + str(us[1])
198 if count >= 3:
199 code += str(vs[2]) + str(us[2])
200 for index in range(3, count):
201 code += str(1 + us[index] + 2 * vs[index])
202 return code

215class Cell:
216 def __init__(self, system, level, x, y):
217 self.system = system
218 self.level = level
219 self.x = x
220 self.y = y
221 self.width, self.height = dimensions(system, level)

223 def code(self, missing="-"):
224 result = ENCODERS[self.system](self.level, self.x, self.y)
225 return missing if result is None else result

227 def ancestor(self, level):
228 width, height = dimensions(self.system, level)
229 return Cell(self.system, level, self.x * width // self.width, self.y * height // self.height)

232 return self.level, self.x, self.y

234 def center(self):
235 latitude = Fraction((2 * self.y + 1) * 180000000, 2 * self.height) - 90000000
236 longitude = Fraction((2 * self.x + 1) * 360000000, 2 * self.width) - 180000000
237 return latitude, longitude

239 def normalized_bounds(self):
240 return (

246def point_cell(system, level, latitude, longitude):
247 width, height = dimensions(system, level)
248 x = (longitude + 180000000) * width // 360000000
249 y = min((latitude + 90000000) * height // 180000000, height - 1)
250 return Cell(system, level, x, y)

253def decode_geohash(code):
254 require(1 <= len(code) <= 9 and all(c in GEOHASH_DIGITS for c in code))
255 x = 0
256 y = 0
257 index = 0
258 for character in code:
259 value = GEOHASH_DIGITS.index(character)
260 for bit in range(4, -1, -1):
261 digit = (value >> bit) & 1
262 if index % 2 == 0:
263 x = 2 * x + digit
264 else:
265 y = 2 * y + digit
266 index += 1
267 return len(code), x, y

270def decode_maidenhead(code):
271 require(len(code) in PRECISIONS["MAIDENHEAD"])
272 x = 0
273 y = 0
274 for index in range(len(code) // 2):
275 alphabet = MAIDENHEAD_SETS[index]
276 east, north = code[2 * index:2 * index + 2]
277 require(east in alphabet and north in alphabet)
278 x = x * len(alphabet) + alphabet.index(east)
279 y = y * len(alphabet) + alphabet.index(north)
280 return len(code), x, y

283def decode_pluscode(code):
284 require(code.count("+") == 1 and code.find("+") == 8)
285 before, after = code.split("+")
286 if "0" in before:
287 count = before.index("0")
288 require(count in (2, 4, 6) and before[count:] == "0" * (8 - count))
289 require(after == "")
290 digits = before[:count]
291 else:
292 require(len(after) in (0, 2, 3))
293 digits = before + after
294 require(all(character in PLUS_DIGITS for character in digits))
295 require(PLUS_DIGITS.index(digits[0]) <= 8 and PLUS_DIGITS.index(digits[1]) <= 17)
296 level = len(digits)
297 x = 0
298 y = 0
299 for index in range(min(level, 10) // 2):
300 y = 20 * y + PLUS_DIGITS.index(digits[2 * index])
301 x = 20 * x + PLUS_DIGITS.index(digits[2 * index + 1])
302 if level == 11:
303 refinement = PLUS_DIGITS.index(digits[-1])
304 x = 4 * x + refinement % 4
305 y = 5 * y + refinement // 4
306 return level, x, y

309def decode_gars(code):
310 require(len(code) in PRECISIONS["GARS"])
311 require(re.fullmatch(r"[0-9]{3}", code[:3]) is not None)
312 band = int(code[:3])
313 require(1 <= band <= 720 and code[3] in L24[:15] and code[4] in L24)
314 x = band - 1
315 y = 24 * L24.index(code[3]) + L24.index(code[4])
316 if len(code) >= 6:
317 require(code[5] in "1234")
318 quadrant = int(code[5]) - 1
319 x = 2 * x + quadrant % 2
320 y = 2 * y + 1 - quadrant // 2
321 if len(code) == 7:
322 require(code[6] in "123456789")
323 area = int(code[6]) - 1
324 x = 3 * x + area % 3
325 y = 3 * y + 2 - area // 3
326 return len(code), x, y

329def decode_georef(code):
330 require(len(code) in PRECISIONS["GEOREF"])
331 require(code[0] in L24 and code[1] in L24[:12])
332 x = L24.index(code[0])
333 y = L24.index(code[1])
334 if len(code) >= 4:
335 require(code[2] in L24[:15] and code[3] in L24[:15])
336 x = 15 * x + L24.index(code[2])
337 y = 15 * y + L24.index(code[3])
338 if len(code) >= 8:
339 places = (len(code) - 4) // 2
340 require(re.fullmatch(r"[0-9]+", code[4:]) is not None)
341 east = int(code[4:4 + places])
342 north = int(code[4 + places:])
343 radix = 6 * 10 ** (places - 1)
344 require(east < radix and north < radix)
345 x = radix * x + east
346 y = radix * y + north
347 return len(code), x, y

350def decode_mesh(code):
351 require(len(code) in PRECISIONS["MESH"])
352 require(re.fullmatch(r"[0-9]+", code) is not None)
353 us = [int(code[2:4])]
354 vs = [int(code[:2])]
355 require(us[0] <= 79)
356 if len(code) >= 6:
357 require(code[4] in "01234567" and code[5] in "01234567")
358 vs.append(int(code[4]))
359 us.append(int(code[5]))
360 if len(code) >= 8:
361 vs.append(int(code[6]))
362 us.append(int(code[7]))
363 for character in code[8:]:
364 require(character in "1234")
365 value = int(character) - 1
366 us.append(value % 2)
367 vs.append(value // 2)
368 count = len(us)
369 scale = MESH_SCALES[len(code)]
370 x = 280 * scale + radix_value(us, (80, 8, 10, 2, 2, 2)[:count])
371 y = 135 * scale + radix_value(vs, (100, 8, 10, 2, 2, 2)[:count])
372 return len(code), x, y

471def compact(arguments):
472 system = system_name(arguments[0])
473 cells = {}
474 levels = PRECISIONS[system]
475 for code in arguments[1:]:
476 cell = decode_cell(system, code)
477 cells[cell.identity()] = cell
478 for key, cell in list(cells.items()):
479 index = levels.index(cell.level)
480 if any(cell.ancestor(level).identity() in cells for level in levels[:index]):
481 del cells[key]
482 while True:
483 merged = False
484 for index in range(len(levels) - 1, 0, -1):
485 level, previous = levels[index], levels[index - 1]
486 width, height = dimensions(system, level)
487 old_width, old_height = dimensions(system, previous)
488 required = (width // old_width) * (height // old_height)
489 families = {}
490 for cell in cells.values():
491 if cell.level == level:
492 ancestor = cell.ancestor(previous)
493 families.setdefault(ancestor.identity(), []).append(cell)
494 for parent_key, siblings in families.items():
495 if len(siblings) == required:
496 for sibling in siblings:
497 del cells[sibling.identity()]
498 ancestor = siblings[0].ancestor(previous)
499 cells[parent_key] = ancestor
500 merged = True
501 if merged:
502 break
503 if not merged:
504 break
505 ordered = sorted(cells.values(), key=lambda cell: (cell.level, -cell.y, cell.x))
506 return " ".join(cell.code() for cell in ordered)

579def relate(arguments):
580 first = decode_cell(system_name(arguments[0]), arguments[1])
581 second = decode_cell(system_name(arguments[2]), arguments[3])
582 left1, right1, bottom1, top1 = first.normalized_bounds()
583 left2, right2, bottom2, top2 = second.normalized_bounds()
584 if (left1, right1, bottom1, top1) == (left2, right2, bottom2, top2):
585 return "EQUAL"
586 if left2 <= left1 and right1 <= right2 and bottom2 <= bottom1 and top1 <= top2:
587 return "INSIDE"
588 if left1 <= left2 and right2 <= right1 and bottom1 <= bottom2 and top2 <= top1:
589 return "CONTAINS"
590 shared_left, shared_right = max(left1, left2), min(right1, right2)
591 shared_bottom, shared_top = max(bottom1, bottom2), min(top1, top2)
592 if shared_left < shared_right and shared_bottom < shared_top:
593 return "OVERLAP"
594 if shared_left <= shared_right and shared_bottom <= shared_top:
595 return "TOUCH"
596 dateline = (right1 == 1 and left2 == 0) or (right2 == 1 and left1 == 0)
597 if dateline and shared_bottom <= shared_top:
598 return "TOUCH"
599 return "DISJOINT"

630def recover(arguments):
631 short = arguments[0]
632 require(short.count("+") == 1)
633 before, after = short.split("+")
634 require(len(before) in (2, 4) and len(after) in (0, 2, 3))
635 require(all(c in PLUS_DIGITS for c in before + after))
636 latitude = coordinate(arguments[1], True)
637 longitude = coordinate(arguments[2], False)
638 removed = 8 - len(before)
639 prefix = point_cell("PLUSCODE", 10, latitude, longitude).code()[:removed]
640 cell = decode_cell("PLUSCODE", prefix + short)
641 cy, cx = cell.center()
642 distance = 1000000 if removed == 4 else 50000
643 rows = distance * cell.height // 180000000
644 columns = distance * cell.width // 360000000
645 if latitude + distance // 2 < cy and cy - distance >= -90000000:
646 cell.y -= rows
647 elif latitude - distance // 2 > cy and cy + distance <= 90000000:
648 cell.y += rows
649 if longitude + distance // 2 < cx:
650 cell.x = (cell.x - columns) % cell.width
651 elif longitude - distance // 2 > cx:
652 cell.x = (cell.x + columns) % cell.width
653 return cell.code()

656def utmzone(arguments):
657 latitude = coordinate(arguments[0], True)
658 longitude = coordinate(arguments[1], False)
659 if latitude < -80000000 or latitude >= 84000000:
660 return "NONE"
661 band = BANDS[min((latitude + 80000000) // 8000000, 19)]
662 zone = (longitude + 180000000) // 6000000 + 1
663 if band == "V" and 3000000 <= longitude < 12000000:
664 zone = 32
665 elif band == "X":
666 if 0 <= longitude < 9000000:
667 zone = 31
668 elif 9000000 <= longitude < 21000000:
669 zone = 33
670 elif 21000000 <= longitude < 33000000:
671 zone = 35
672 elif 33000000 <= longitude < 42000000:
673 zone = 37
674 return str(zone) + band

677def metre_digits(easting, northing, places):
678 return f"{easting % 100000:05d}"[:places], f"{northing % 100000:05d}"[:places]

681def square_offsets(arguments):
682 if not arguments:
683 return 0, 0, 100000
684 east, north = arguments
685 require(re.fullmatch(r"[0-9]{1,5}", east) is not None)
686 require(re.fullmatch(r"[0-9]{1,5}", north) is not None)
687 require(len(east) == len(north))
688 square_size = 10 ** (5 - len(east))
689 return int(east) * square_size, int(north) * square_size, square_size

692def osgb(arguments):
693 easting = integer(arguments[0], 0, 699999)
694 northing = integer(arguments[1], 0, 1299999)
695 places = integer(arguments[2], 0, 5)
696 first = 5 * (3 - northing // 500000) + 2 + easting // 500000
697 second = 5 * (4 - northing // 100000 % 5) + easting // 100000 % 5
698 letters = OSGB_LETTERS[first] + OSGB_LETTERS[second]
699 if places == 0:
700 return letters
701 east, north = metre_digits(easting, northing, places)
702 return f"{letters} {east} {north}"

705def unosgb(arguments):
706 letters = arguments[0]
707 require(len(letters) == 2 and all(c in OSGB_LETTERS for c in letters))
708 row1, column1 = divmod(OSGB_LETTERS.index(letters[0]), 5)
709 row2, column2 = divmod(OSGB_LETTERS.index(letters[1]), 5)
710 easting = 500000 * (column1 - 2) + 100000 * column2
711 northing = 500000 * (3 - row1) + 100000 * (4 - row2)
712 require(0 <= easting < 700000 and 0 <= northing < 1300000)
713 east_offset, north_offset, square_size = square_offsets(arguments[1:])
714 return f"sw={easting + east_offset},{northing + north_offset} size={square_size}"

717def mgrs(arguments):
718 zone = integer(arguments[0], 1, 60)
719 band = arguments[1]
720 require(len(band) == 1 and band in BANDS)
721 easting = integer(arguments[2], 100000, 899999)
722 northing = integer(arguments[3], 0, 9999999)
723 places = integer(arguments[4], 0, 5)
724 column = MGRS_COLUMNS[zone % 3][easting // 100000 - 1]
725 shift = 5 if zone % 2 == 0 else 0
726 row = MGRS_ROWS[(northing // 100000 + shift) % 20]
727 prefix = f"{zone}{band} {column}{row}"
728 if places == 0:
729 return prefix
730 east, north = metre_digits(easting, northing, places)
731 return f"{prefix} {east} {north}"

734def unmgrs(arguments):
735 designator, square = arguments[:2]
736 require(re.fullmatch(r"[1-9][0-9]?[CDEFGHJKLMNPQRSTUVWX]", designator) is not None)
737 zone = int(designator[:-1])
738 require(zone <= 60)
739 require(len(square) == 2)
740 columns = MGRS_COLUMNS[zone % 3]
741 require(square[0] in columns and square[1] in MGRS_ROWS)
742 easting = 100000 * (columns.index(square[0]) + 1)
743 shift = 5 if zone % 2 == 0 else 0
744 northing = 100000 * ((MGRS_ROWS.index(square[1]) - shift) % 20)
745 east_offset, north_offset, square_size = square_offsets(arguments[2:])
746 return f"{designator} sw={easting + east_offset},{northing + north_offset} size={square_size}"

749def quadkey(arguments):
750 zoom = integer(arguments[2], 1, 23)
751 x = integer(arguments[0], 0, (1 << zoom) - 1)
752 y = integer(arguments[1], 0, (1 << zoom) - 1)
753 return "".join(str(((x >> bit) & 1) + 2 * ((y >> bit) & 1)) for bit in range(zoom - 1, -1, -1))

756def unquadkey(arguments):
757 key = arguments[0]
758 require(1 <= len(key) <= 23 and all(character in "0123" for character in key))
759 x = 0
760 y = 0
761 for character in key:
762 digit = int(character)
763 x = 2 * x + digit % 2
764 y = 2 * y + digit // 2
765 return f"x={x} y={y} z={len(key)}"
