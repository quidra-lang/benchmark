70function encode(string $sys, int $p, int $x, int $y): string
71{
72 if ($sys === 'GEOHASH') {
73 $a = intdiv(5 * $p + 1, 2);
74 $b = intdiv(5 * $p, 2);
75 $value = 0;
76 $out = '';
77 for ($i = 0; $i < 5 * $p; ++$i) {
78 if ($i % 2 === 0) {
79 --$a;
80 $bit = intdiv($x, 2 ** $a) % 2;
81 } else {
82 --$b;
83 $bit = intdiv($y, 2 ** $b) % 2;
84 }
85 $value = 2 * $value + $bit;
86 if ($i % 5 === 4) {
87 $out .= GH[$value];
88 $value = 0;
89 }
90 }
91 return $out;
92 }
93 if ($sys === 'MAIDENHEAD') {
94 $k = intdiv($p, 2);
95 $radices = array_slice([18,10,24,10], 0, $k);
96 $xs = digits($x, $radices);
97 $ys = digits($y, $radices);
98 $out = '';
99 for ($i = 0; $i < $k; ++$i) $out .= MH[$i][$xs[$i]] . MH[$i][$ys[$i]];
100 return $out;
101 }
102 if ($sys === 'PLUSCODE') {
103 $tail = '';
104 if ($p === 11) {
105 $tail = PC[4 * ($y % 5) + $x % 4];
106 $x = intdiv($x, 4);
107 $y = intdiv($y, 5);
108 }
109 $k = intdiv(min($p, 10), 2);
110 $radices = array_fill(0, $k, 20);
111 $radices[0] = 18;
112 $xs = digits($x, $radices);
113 $radices[0] = 9;
114 $ys = digits($y, $radices);
115 $out = '';
116 for ($i = 0; $i < $k; ++$i) $out .= PC[$ys[$i]] . PC[$xs[$i]];
117 $out .= $tail;
118 if ($p < 8) return $out . str_repeat('0', 8 - $p) . '+';
119 return substr($out, 0, 8) . '+' . substr($out, 8);
120 }
121 if ($sys === 'GARS') {
122 $tail = '';
123 if ($p === 7) {
124 $tail = (string) (1 + $x % 3 + 3 * (2 - $y % 3));
125 $x = intdiv($x, 3);
126 $y = intdiv($y, 3);
127 }
128 if ($p >= 6) {
129 $tail = (1 + $x % 2 + 2 * (1 - $y % 2)) . $tail;
130 $x = intdiv($x, 2);
131 $y = intdiv($y, 2);
132 }
133 return sprintf('%03d', $x + 1) . L24[intdiv($y, 24)] . L24[$y % 24] . $tail;
134 }
135 if ($sys === 'GEOREF') {
136 $tail = '';
137 if ($p >= 8) {
138 $m = intdiv($p - 4, 2);
139 $scale = 6 * 10 ** ($m - 1);
140 $tail = sprintf('%0*d%0*d', $m, $x % $scale, $m, $y % $scale);
141 $x = intdiv($x, $scale);
142 $y = intdiv($y, $scale);
143 }
144 if ($p >= 4) {
145 $tail = L24[$x % 15] . L24[$y % 15] . $tail;
146 $x = intdiv($x, 15);
147 $y = intdiv($y, 15);
148 }
149 return L24[$x] . L24[$y] . $tail;
150 }
151 $level = array_search($p, PRECS['MESH'], true);
152 $scale = [1,8,80,160,320,640][$level];
153 $u = $x - 280 * $scale;
154 $v = $y - 135 * $scale;
155 if ($u < 0 || $u >= 80 * $scale || $v < 0 || $v >= 100 * $scale) return '-';
156 $radices = array_slice([80,8,10,2,2,2], 0, $level + 1);
157 $us = digits($u, $radices);
158 $radices[0] = 100;
159 $vs = digits($v, $radices);
160 $out = sprintf('%02d%02d', $vs[0], $us[0]);
161 for ($i = 1; $i <= $level; ++$i) {
162 if ($i <= 2) $out .= $vs[$i] . $us[$i];
163 else $out .= 1 + $us[$i] + 2 * $vs[$i];
164 }
165 return $out;
166}
167function decode(string $sys, string $code): array
168{
169 $p = strlen($code);
170 $x = 0;
171 $y = 0;
172 if ($sys !== 'PLUSCODE' && !in_array($p, PRECS[$sys], true)) invalid();
173 if ($sys === 'GEOHASH') {
174 $bitIndex = 0;
175 foreach (str_split($code) as $char) {
176 $value = index_char(GH, $char);
177 for ($i = 4; $i >= 0; --$i) {
178 $bit = intdiv($value, 2 ** $i) % 2;
179 if ($bitIndex % 2 === 0) $x = 2 * $x + $bit;
180 else $y = 2 * $y + $bit;
181 ++$bitIndex;
182 }
183 }
184 } elseif ($sys === 'MAIDENHEAD') {
185 for ($i = 0; $i < intdiv($p, 2); ++$i) {
186 $radix = strlen(MH[$i]);
187 $x = $x * $radix + index_char(MH[$i], $code[2 * $i]);
188 $y = $y * $radix + index_char(MH[$i], $code[2 * $i + 1]);
189 }
190 } elseif ($sys === 'PLUSCODE') {
191 if (substr_count($code, '+') !== 1 || strpos($code, '+') !== 8) invalid();
192 $before = substr($code, 0, 8);
193 $after = substr($code, 9);
194 $prefix = strcspn($before, '0');
195 if (!in_array($prefix, [2,4,6,8], true)) invalid();
196 if ($prefix < 8 && (substr($before, $prefix) !== str_repeat('0', 8 - $prefix) || $after !== '')) invalid();
197 if ($prefix === 8 && !in_array(strlen($after), [0,2,3], true)) invalid();
198 $sequence = substr($before, 0, $prefix) . $after;
199 $p = strlen($sequence);
200 $values = array_map(fn($char) => index_char(PC, $char), str_split($sequence));
201 if ($values[0] > 8 || $values[1] > 17) invalid();
202 for ($i = 0; $i < intdiv(min($p, 10), 2); ++$i) {
203 $y = $y * ($i === 0 ? 9 : 20) + $values[2 * $i];
204 $x = $x * ($i === 0 ? 18 : 20) + $values[2 * $i + 1];
205 }
206 if ($p === 11) {
207 $x = 4 * $x + $values[10] % 4;
208 $y = 5 * $y + intdiv($values[10], 4);
209 }
210 } elseif ($sys === 'GARS') {
211 if (!preg_match('/^[0-9]{3}/D', $code)) invalid();
212 $x = (int) substr($code, 0, 3) - 1;
213 $y = 24 * index_char(substr(L24, 0, 15), $code[3]) + index_char(L24, $code[4]);
214 if ($p >= 6) {
215 $q = index_char('1234', $code[5]);
216 $x = 2 * $x + $q % 2;
217 $y = 2 * $y + 1 - intdiv($q, 2);
218 }
219 if ($p === 7) {
220 $q = index_char('123456789', $code[6]);
221 $x = 3 * $x + $q % 3;
222 $y = 3 * $y + 2 - intdiv($q, 3);
223 }
224 } elseif ($sys === 'GEOREF') {
225 $x = index_char(L24, $code[0]);
226 $y = index_char(substr(L24, 0, 12), $code[1]);
227 if ($p >= 4) {
228 $x = 15 * $x + index_char(substr(L24, 0, 15), $code[2]);
229 $y = 15 * $y + index_char(substr(L24, 0, 15), $code[3]);
230 }
231 if ($p >= 8) {
232 $m = intdiv($p - 4, 2);
233 $tail = substr($code, 4);
234 if (!preg_match('/^[0-9]+$/D', $tail)) invalid();
235 $scale = 6 * 10 ** ($m - 1);
236 $e = (int) substr($tail, 0, $m);
237 $n = (int) substr($tail, $m);
238 if ($e >= $scale || $n >= $scale) invalid();
239 $x = $x * $scale + $e;
240 $y = $y * $scale + $n;
241 }
242 } else {
243 if (!preg_match('/^[0-9]+$/D', $code)) invalid();
244 $level = array_search($p, PRECS['MESH'], true);
245 $u = (int) substr($code, 2, 2);
246 $v = (int) substr($code, 0, 2);
247 if ($u > 79) invalid();
248 for ($i = 1; $i <= $level; ++$i) {
249 if ($i <= 2) {
250 $base = $i === 1 ? 8 : 10;
251 $a = (int) $code[2 + 2 * $i];
252 $b = (int) $code[3 + 2 * $i];
253 if ($a >= $base || $b >= $base) invalid();
254 $v = $v * $base + $a;
255 $u = $u * $base + $b;
256 } else {
257 $q = index_char('1234', $code[$i + 5]);
258 $u = 2 * $u + $q % 2;
259 $v = 2 * $v + intdiv($q, 2);
260 }
261 }
262 $scale = [1,8,80,160,320,640][$level];
263 $x = $u + 280 * $scale;
264 $y = $v + 135 * $scale;
265 }
266 [$w, $h] = grid($sys, $p);
267 if ($x < 0 || $x >= $w || $y < 0 || $y >= $h || encode($sys, $p, $x, $y) !== $code) invalid();
268 return [$p, $x, $y];
269}

301function compact_cells(string $sys, array $cells): array
302{
303 $set = [];
304 foreach ($cells as $cell) $set[implode(':', $cell)] = $cell;
305 do {
306 $changed = false;
307 foreach ($set as $key => $cell) {
308 $index = array_search($cell[0], PRECS[$sys], true);
309 for ($i = 0; $i < $index; ++$i) {
310 if (isset($set[implode(':', ancestor($sys, $cell, PRECS[$sys][$i]))])) {
311 unset($set[$key]);
312 $changed = true;
313 break;
314 }
315 }
316 }
317 for ($index = count(PRECS[$sys]) - 1; $index >= 1; --$index) {
318 [$p, $q] = [PRECS[$sys][$index], PRECS[$sys][$index - 1]];
319 [$w, $h] = grid($sys, $p);
320 [$v, $u] = grid($sys, $q);
321 $needed = intdiv($w, $v) * intdiv($h, $u);
322 $groups = [];
323 foreach ($set as $key => $cell) {
324 if ($cell[0] === $p) {
325 $parent = ancestor($sys, $cell, $q);
326 $pk = implode(':', $parent);
327 $groups[$pk]['parent'] = $parent;
328 $groups[$pk]['keys'][] = $key;
329 }
330 }
331 foreach ($groups as $key => $group) {
332 if (count($group['keys']) === $needed) {
333 foreach ($group['keys'] as $child) unset($set[$child]);
334 $set[$key] = $group['parent'];
335 $changed = true;
336 }
337 }
338 }
339 } while ($changed);
340 $cells = array_values($set);
341 usort($cells, fn($a, $b) => [$a[0], -$a[2], $a[1]] <=> [$b[0], -$b[2], $b[1]]);
342 return $cells;
343}
344function execute(string $line): string

358 if ($band === 'V' && $lon >= 3000000 && $lon < 12000000) $zone = 32;

399 $r = index_char(ROWS, $t[1][1]);

401 $n = 100000 * md($r - ($zone % 2 === 0 ? 5 : 0), 20);
402 [$de, $dn, $size] = square_digits($t, 2);

404 }
405 if ($command === 'QUADKEY') {
406 $z = integer($t[2], 1, 23);
407 $x = integer($t[0], 0, 2 ** $z - 1);
408 $y = integer($t[1], 0, 2 ** $z - 1);
409 $out = '';
410 for ($i = $z - 1; $i >= 0; --$i) $out .= intdiv($x, 2 ** $i) % 2 + 2 * (intdiv($y, 2 ** $i) % 2);

414 if (!preg_match('/^[0-3]{1,23}$/D', $t[0])) invalid();
415 $x = 0;
416 $y = 0;
417 foreach (str_split($t[0]) as $char) {
418 $v = (int) $char;
419 $x = 2 * $x + $v % 2;
420 $y = 2 * $y + intdiv($v, 2);
421 }

425 $lat = coordinate($t[1], true);
426 $lon = coordinate($t[2], false);
427 if ($command === 'SHORTEN') {
428 [$p, $x, $y] = decode('PLUSCODE', $t[0]);

439 [$px, $py] = point_cell('PLUSCODE', 10, $lat, $lon);
440 $full = substr(encode('PLUSCODE', 10, $px, $py), 0, $r) . $t[0];
441 [$p, $x, $y] = decode('PLUSCODE', $full);
442 [$w, $h] = grid('PLUSCODE', $p);
443 $d = $r === 4 ? 1000000 : 50000;
444 $my = intdiv($d * $h, 180000000);
445 $mx = intdiv($d * $w, 360000000);
446 $cy = 180000000 * (2 * $y + 1);
447 $cx = 360000000 * (2 * $x + 1);
448 if (($lat + intdiv($d, 2) + 90000000) * 2 * $h < $cy && $cy >= $d * 2 * $h) $y -= $my;
449 elseif (($lat - intdiv($d, 2) + 90000000) * 2 * $h > $cy && $cy <= (180000000 - $d) * 2 * $h) $y += $my;

451 elseif (($lon - intdiv($d, 2) + 180000000) * 2 * $w > $cx) $x = md($x + $mx, $w);
452 return encode('PLUSCODE', $p, $x, $y);
453 }
454 $sys = system_name($t[0]);
455 if ($command === 'SIZE' || $command === 'ENCODE' || $command === 'COVER') {
456 $p = precision($sys, $t[$command === 'ENCODE' ? 3 : 1]);
457 [$w, $h] = grid($sys, $p);
458 if ($command === 'SIZE') {
459 $cells = $w * $h;
460 if ($sys === 'MESH') {
461 $scale = [1,8,80,160,320,640][array_search($p, PRECS[$sys], true)];
462 $cells = 8000 * $scale * $scale;
463 }
464 return 'w=' . coord_text(intdiv(360000000, $w)) . ' h=' . coord_text(intdiv(180000000, $h)) . " cells=$cells";
465 }
466 if ($command === 'ENCODE') {
467 [$x, $y] = point_cell($sys, $p, coordinate($t[1], true), coordinate($t[2], false));
468 $code = encode($sys, $p, $x, $y);
469 return $code === '-' ? 'OUT OF AREA' : $code;
470 }
471 $lat1 = coordinate($t[2], true);
472 $lon1 = coordinate($t[3], false);
473 $lat2 = coordinate($t[4], true);
