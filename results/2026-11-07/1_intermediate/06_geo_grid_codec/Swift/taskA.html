46func coordinate(_ s: String, _ latitude: Bool) throws -> Int {
47 try need(matches(s,"-?[0-9]{1,3}(\\.[0-9]{1,6})?"))
48 let negative = s.hasPrefix("-")
49 let pieces = (negative ? s.drop(1) : s).split(separator:".").map(String.init)
50 var value = Int(pieces[0])!*1000000
51 if pieces.count == 2 { value += Int(pieces[1]+String(repeating:"0",count:6-pieces[1].count))! }
52 if negative { value = -value }
53 try need(latitude ? value >= -90000000 && value <= 90000000 : value >= -180000000 && value < 180000000)
54 return value
55}

60func level(_ system: String, _ precision: Int) throws -> Int {

64func grid(_ system: String, _ level: Int, _ x: Int = 0, _ y: Int = 0) throws -> Cell {
65 let p = try precisions(system)[level]
66 let w: Int
67 let h: Int
68 switch system {
69 case "GEOHASH":
70 w = 1 << ((5*p+1)/2)
71 h = 1 << (5*p/2)
72 case "MAIDENHEAD":
73 w = [18,180,4320,43200][level]
74 h = w
75 case "PLUSCODE":
76 w = [18,360,7200,144000,2880000,11520000][level]
77 h = [9,180,3600,72000,1440000,7200000][level]
78 case "GARS":
79 w = [720,1440,4320][level]
80 h = [360,720,2160][level]
81 case "GEOREF":
82 w = [24,360,21600,216000,2160000][level]
83 h = [12,180,10800,108000,1080000][level]
84 default:
85 w = 360*meshScale[level]
86 h = 270*meshScale[level]
87 }
88 return Cell(system:system,level:level,x:x,y:y,w:w,h:h)
89}

110func radix(_ value: Int, _ bases: [Int], _ count: Int) -> [Int] {
111 var rest = value
112 var result = [Int](repeating:0,count:count)
113 for i in stride(from:count-1,through:0,by:-1) {
114 result[i] = rest%bases[i]
115 rest /= bases[i]
116 }
117 return result
118}
119func code(_ c: Cell) -> String? {
120 let p = precisionTable[c.system]![c.level]
121 var x = c.x
122 var y = c.y
123 var result = ""
124 switch c.system {
125 case "GEOHASH":
126 var ax = (5*p+1)/2
127 var ay = 5*p/2
128 var group = 0
129 for bit in 0..<5*p {
130 let b: Int
131 if bit%2 == 0 {
132 ax -= 1
133 b = (x >> ax)&1
134 } else {
135 ay -= 1
136 b = (y >> ay)&1
137 }
138 group = group*2+b
139 if bit%5 == 4 {
140 result.append(gh[group])
141 group = 0
142 }
143 }
144 case "MAIDENHEAD":
145 let a = radix(x,[18,10,24,10],p/2)
146 let b = radix(y,[18,10,24,10],p/2)
147 for i in a.indices {
148 result.append(maiden[i][a[i]])
149 result.append(maiden[i][b[i]])
150 }
151 case "PLUSCODE":
152 let extra = p == 11 ? 4*(y%5)+x%4 : -1
153 if p == 11 {
154 x /= 4
155 y /= 5
156 }
157 let count = min(p,10)/2
158 let a = radix(x,[18,20,20,20,20],count)
159 let b = radix(y,[9,20,20,20,20],count)
160 for i in 0..<count {
161 result.append(plusAlphabet[b[i]])
162 result.append(plusAlphabet[a[i]])
163 }
164 if extra >= 0 { result.append(plusAlphabet[extra]) }
165 return p < 8 ? result+String(repeating:"0",count:8-p)+"+" : result.take(8)+"+"+result.drop(8)
166 case "GARS":
167 let scale = c.level == 0 ? 1 : c.level == 1 ? 2 : 6
168 let bx = x/scale
169 let by = y/scale
170 result = pad(bx+1,3)+String(l24[by/24])+String(l24[by%24])
171 if p >= 6 {
172 let divisor = p == 7 ? 3 : 1
173 result += String(1+x/divisor%2+2*(1-y/divisor%2))
174 }
175 if p == 7 { result += String(1+x%3+3*(2-y%3)) }
176 case "GEOREF":
177 if p == 2 { return String(l24[x])+String(l24[y]) }
178 let m = (p-4)/2
179 let scale = p == 4 ? 1 : 6*power(m-1)
180 let bx = x/scale
181 let by = y/scale
182 result = String(l24[bx/15])+String(l24[by/15])+String(l24[bx%15])+String(l24[by%15])
183 if p >= 8 { result += pad(x%scale,m)+pad(y%scale,m) }
184 default:
185 let scale = meshScale[c.level]
186 let u = x-280*scale
187 let v = y-135*scale
188 if u < 0 || u >= 80*scale || v < 0 || v >= 100*scale { return nil }
189 let count = c.level+1
190 let a = radix(u,[80,8,10,2,2,2],count)
191 let b = radix(v,[100,8,10,2,2,2],count)
192 result = pad(b[0],2)+pad(a[0],2)
193 if count > 1 {
194 for i in 1..<count {
195 if i <= 2 { result += String(b[i])+String(a[i]) }
196 else { result += String(1+a[i]+2*b[i]) }
197 }
198 }
199 }
200 return result
201}
202func decode(_ system: String, _ text: String) throws -> Cell {
203 _ = try precisions(system)
204 var p = text.count
205 var x = 0
206 var y = 0
207 switch system {
208 case "GEOHASH":
209 try need(p >= 1 && p <= 9)
210 var bit = 0
211 for ch in text {
212 let d = try digit(ch,gh)
213 for i in stride(from:4,through:0,by:-1) {
214 if bit%2 == 0 { x = x*2+((d >> i)&1) }
215 else { y = y*2+((d >> i)&1) }
216 bit += 1
217 }
218 }
219 case "MAIDENHEAD":
220 _ = try level(system,p)
221 for i in 0..<p/2 {
222 x = try x*maiden[i].count+digit(text[2*i],maiden[i])
223 y = try y*maiden[i].count+digit(text[2*i+1],maiden[i])
224 }
225 case "PLUSCODE":
226 try need(p >= 9 && text.firstIndex(of:"+") == text.index(text.startIndex,offsetBy:8) && text.filter { $0 == "+" }.count == 1)
227 let firstZero = Array(text.take(8)).firstIndex(of:"0")
228 let before = firstZero ?? 8
229 let after = p-9
230 try need([2,4,6,8].contains(before))
231 try need(before == 8 ? [0,2,3].contains(after) : after == 0)
232 if before < 8 { for i in before..<8 { try need(text[i] == "0") } }
233 let sequence = text.take(before)+text.drop(9)
234 p = sequence.count
235 try need(digit(sequence[0],plusAlphabet) <= 8 && digit(sequence[1],plusAlphabet) <= 17)
236 for i in stride(from:0,to:min(p,10),by:2) {
237 y = try y*20+digit(sequence[i],plusAlphabet)
238 x = try x*20+digit(sequence[i+1],plusAlphabet)
239 }
240 if p == 11 {
241 let d = try digit(sequence[10],plusAlphabet)
242 x = x*4+d%4
243 y = y*5+d/4
244 }
245 case "GARS":
246 try need(p >= 5 && p <= 7)
247 let e = try digits(text.take(3))
248 try need(e >= 1 && e <= 720)
249 x = e-1
250 y = try digit(text[3],l24.take(15))*24+digit(text[4],l24)
251 if p >= 6 {
252 let q = try digit(text[5],"1234")
253 x = x*2+q%2
254 y = y*2+1-q/2
255 }
256 if p == 7 {
257 let q = try digit(text[6],"123456789")
258 x = x*3+q%3
259 y = y*3+2-q/3
260 }
261 case "GEOREF":
262 _ = try level(system,p)
263 x = try digit(text[0],l24)
264 y = try digit(text[1],l24.take(12))
265 if p >= 4 {
266 x = try x*15+digit(text[2],l24.take(15))
267 y = try y*15+digit(text[3],l24.take(15))
268 }
269 if p >= 8 {
270 let m = (p-4)/2
271 let scale = 6*power(m-1)
272 let a = try digits(text.slice(4,4+m))
273 let b = try digits(text.drop(4+m))
274 try need(a < scale && b < scale)
275 x = x*scale+a
276 y = y*scale+b
277 }
278 default:
279 let l = try level(system,p)
280 _ = try digits(text)
281 var v = try digits(text.take(2))
282 var u = try digits(text.slice(2,4))
283 try need(u < 80)
284 if p >= 6 {
285 v = try v*8+digit(text[4],"01234567")
286 u = try u*8+digit(text[5],"01234567")
287 }
288 if p >= 8 {
289 v = try v*10+digit(text[6],"0123456789")
290 u = try u*10+digit(text[7],"0123456789")
291 }
292 if p > 8 {
293 for i in 8..<p {
294 let q = try digit(text[i],"1234")
295 v = v*2+q/2
296 u = u*2+q%2
297 }
298 }
299 x = u+280*meshScale[l]
300 y = v+135*meshScale[l]
301 }
302 let result = try grid(system,level(system,p),x,y)
303 try need(text == code(result))
304 return result
305}

312func boundary(_ c: Cell, _ latitude: Bool, _ offset: Int, _ center: Bool) -> Int {
313 let extent = latitude ? 180000000 : 360000000
314 let total = latitude ? c.h : c.w
315 let value = latitude ? c.y : c.x

318func compare(_ a: Int, _ ad: Int, _ b: Int, _ bd: Int) -> Int {
319 let left = a*bd
320 let right = b*ad
321 return left < right ? -1 : left > right ? 1 : 0
322}
323func inside(_ a: Cell, _ b: Cell) -> Bool {
324 compare(a.x,a.w,b.x,b.w) >= 0 && compare(a.x+1,a.w,b.x+1,b.w) <= 0 &&
325 compare(a.y,a.h,b.y,b.h) >= 0 && compare(a.y+1,a.h,b.y+1,b.h) <= 0
326}
327func mod(_ a: Int, _ n: Int) -> Int { (a%n+n)%n }
328func floorDiv(_ a: Int, _ n: Int) -> Int { a/n-(a%n < 0 ? 1 : 0) }
329func steps(_ a: Cell, _ b: Cell) -> (Int,Int) {
330 var dx = mod(b.x-a.x,a.w)
331 if dx > a.w/2 { dx -= a.w }
332 return (dx,b.y-a.y)
333}
334func detail(_ a: String, _ b: String) throws -> (Int,Int,Int) {
335 try need(matches(a,"[0-9]{1,5}") && matches(b,"[0-9]{1,5}") && a.count == b.count)
336 let size = power(5-a.count)
337 return try (digits(a)*size,digits(b)*size,size)
338}
339func columns(_ zone: Int) -> String { zone%3 == 1 ? "ABCDEFGH" : zone%3 == 2 ? "JKLMNPQR" : "STUVWXYZ" }
340func auxiliary(_ t: [String]) throws -> String {
341 let n = t.count-1
342 switch t[0] {
343 case "UTMZONE":
344 try need(n == 2)
345 let lat = try coordinate(t[1],true)
346 let lon = try coordinate(t[2],false)
347 if lat < -80000000 || lat >= 84000000 { return "NONE" }
348 let band = bands[min((lat+80000000)/8000000,19)]
349 var zone = (lon+180000000)/6000000+1
350 if band == "V" && lon >= 3000000 && lon < 12000000 { zone = 32 }
351 if band == "X" {
352 if lon >= 0 && lon < 9000000 { zone = 31 }
353 else if lon >= 9000000 && lon < 21000000 { zone = 33 }
354 else if lon >= 21000000 && lon < 33000000 { zone = 35 }
355 else if lon >= 33000000 && lon < 42000000 { zone = 37 }
356 }
357 return String(zone)+String(band)
358 case "OSGB":
359 try need(n == 3)
360 let e = try range(t[1],0,699999)
361 let north = try range(t[2],0,1299999)
362 let d = try range(t[3],0,5)
363 let letters = String(g25[(3-north/500000)*5+2+e/500000])+String(g25[(4-north/100000%5)*5+e/100000%5])
364 return letters+(d == 0 ? "" : " "+pad(e%100000,5).take(d)+" "+pad(north%100000,5).take(d))
365 case "UNOSGB":
366 try need(n == 1 || n == 3)
367 try need(t[1].count == 2)
368 let a = try digit(t[1][0],g25)
369 let b = try digit(t[1][1],g25)
370 var e = 500000*(a%5-2)+100000*(b%5)
371 var north = 500000*(3-a/5)+100000*(4-b/5)
372 var size = 100000
373 try need(e >= 0 && e < 700000 && north >= 0 && north < 1300000)
374 if n == 3 {
375 let values = try detail(t[2],t[3])
376 e += values.0
377 north += values.1
378 size = values.2
379 }
380 return "sw=\(e),\(north) size=\(size)"
381 case "MGRS":
382 try need(n == 5)
383 let zone = try range(t[1],1,60)
384 let e = try range(t[3],100000,899999)
385 let north = try range(t[4],0,9999999)
386 try need(t[2].count == 1)
387 _ = try digit(t[2][0],bands)
388 let d = try range(t[5],0,5)
389 let offset = zone%2 == 0 ? 5 : 0
390 let square = String(columns(zone)[e/100000-1])+String(mrows[(north/100000+offset)%20])
391 return String(zone)+t[2]+" "+square+(d == 0 ? "" : " "+pad(e%100000,5).take(d)+" "+pad(north%100000,5).take(d))
392 case "UNMGRS":
393 try need(n == 2 || n == 4)
394 try need(matches(t[1],"[1-9][0-9]?[CDEFGHJKLMNPQRSTUVWX]"))
395 let zone = Int(t[1].dropLast())!
396 try need(zone <= 60 && t[2].count == 2)
397 let a = try digit(t[2][0],columns(zone))
398 let b = try digit(t[2][1],mrows)
399 let offset = zone%2 == 0 ? 5 : 0
400 var e = 100000*(a+1)
401 var north = 100000*mod(b-offset,20)
402 var size = 100000
403 if n == 4 {
404 let values = try detail(t[3],t[4])
405 e += values.0
406 north += values.1
407 size = values.2
408 }
409 return t[1]+" sw=\(e),\(north) size=\(size)"
410 case "QUADKEY":
411 try need(n == 3)
412 let z = try range(t[3],1,23)
413 let x = try range(t[1],0,(1 << z)-1)
414 let y = try range(t[2],0,(1 << z)-1)
415 return stride(from:z-1,through:0,by:-1).map { String(((x >> $0)&1)+2*((y >> $0)&1)) }.joined()
416 case "UNQUADKEY":
417 try need(n == 1 && matches(t[1],"[0-3]{1,23}"))
418 var x = 0
419 var y = 0
420 for ch in t[1].utf8 {
421 let d = Int(ch)-48
422 x = x*2+d%2
423 y = y*2+d/2
424 }
425 return "x=\(x) y=\(y) z=\(t[1].count)"
426 default: throw Invalid.argument
427 }

429func execute(_ line: String) throws -> String {
430 let t = line.split(separator:" ").map(String.init)
431 let command = t[0]
432 if counts[command] == nil && command != "COMPACT" { return try auxiliary(t) }
433 try need(command == "COMPACT" ? t.count >= 3 && t.count <= 102 : t.count == counts[command]!+1)
434 if command == "SHORTEN" || command == "RECOVER" {
435 let lat = try coordinate(t[2],true)
436 let lon = try coordinate(t[3],false)
437 if command == "SHORTEN" {
438 let c = try decode("PLUSCODE",t[1])
439 try need(precisions(c.system)[c.level] >= 8)
440 let yn = -90000000*2*c.h+180000000*(2*c.y+1)
441 let xn = -180000000*2*c.w+360000000*(2*c.x+1)
442 if abs(yn-lat*2*c.h) < 15000*2*c.h && abs(xn-lon*2*c.w) < 15000*2*c.w { return t[1].drop(6) }
443 if abs(yn-lat*2*c.h) < 300000*2*c.h && abs(xn-lon*2*c.w) < 300000*2*c.w { return t[1].drop(4) }
444 return t[1]
445 }
446 let plus = Array(t[1]).firstIndex(of:"+") ?? -1
447 try need(plus == 2 || plus == 4)
448 try need(t[1].filter { $0 == "+" }.count == 1 && [0,2,3].contains(t[1].count-plus-1))
449 for ch in t[1] where ch != "+" { _ = try digit(ch,plusAlphabet) }
450 let missing = 8-plus
451 let prefix = try code(point("PLUSCODE",4,lat,lon))!.take(missing)
452 let c = try decode("PLUSCODE",prefix+t[1])
453 let d = missing == 4 ? 1000000 : 50000
454 var x = c.x
455 var y = c.y
456 let yn = -90000000*2*c.h+180000000*(2*y+1)
457 let xn = -180000000*2*c.w+360000000*(2*x+1)
458 let my = d*c.h/180000000
459 let mx = d*c.w/360000000
460 if (lat+d/2)*2*c.h < yn && yn-d*2*c.h >= -90000000*2*c.h { y -= my }
461 else if (lat-d/2)*2*c.h > yn && yn+d*2*c.h <= 90000000*2*c.h { y += my }
462 if (lon+d/2)*2*c.w < xn { x = mod(x-mx,c.w) }
463 else if (lon-d/2)*2*c.w > xn { x = mod(x+mx,c.w) }
464 return try code(grid("PLUSCODE",c.level,x,y))!
465 }
466 let system = t[1]
467 _ = try precisions(system)
468 switch command {
469 case "ENCODE": return try code(point(system,level(system,integer(t[4])),coordinate(t[2],true),coordinate(t[3],false))) ?? "OUT OF AREA"
470 case "SIZE":
471 let l = try level(system,integer(t[2]))
472 let c = try grid(system,l)
473 let count = system == "MESH" ? 8000*meshScale[l]*meshScale[l] : c.w*c.h
474 return "w=\(coordinateFormat(360000000/c.w)) h=\(coordinateFormat(180000000/c.h)) cells=\(count)"
475 case "COVER":
476 let l = try level(system,integer(t[2]))
477 let lat1 = try coordinate(t[3],true)
478 let lon1 = try coordinate(t[4],false)
479 let lat2 = try coordinate(t[5],true)
480 let lon2 = try coordinate(t[6],false)
481 try need(lat1 <= lat2)
482 let a = try point(system,l,lat1,lon1)
483 let b = try point(system,l,lat2,lon2)
484 let columns = lon1 <= lon2 ? b.x-a.x+1 : min(a.w,b.x-a.x+1+a.w)
485 if columns*(b.y-a.y+1) > 100 { return "TOO MANY" }
486 var result: [String] = []
487 for y in stride(from:b.y,through:a.y,by:-1) {
488 for i in 0..<columns { result.append(try shown(grid(system,l,(a.x+i)%a.w,y))) }
489 }
490 return result.joined(separator:" ")
491 case "COMPACT":
492 var cells = Set(try t.dropFirst(2).map { try decode(system,$0) })
493 let covered = try cells.filter { c in
494 try cells.contains { a in try a.level < c.level && ancestor(c,a.level) == a }
495 }
496 cells.subtract(covered)
497 var changed = true
498 while changed {
499 changed = false
500 for child in Array(cells) {
501 if child.level == 0 { continue }
502 let parent = try ancestor(child,child.level-1)
503 let g = try grid(system,child.level)
504 let rx = g.w/parent.w
505 let ry = g.h/parent.h
506 if rx*ry > cells.count { continue }
507 var siblings = Set<Cell>()
508 for j in 0..<ry {
509 for i in 0..<rx { siblings.insert(try grid(system,child.level,parent.x*rx+i,parent.y*ry+j)) }
510 }
511 if cells.isSuperset(of:siblings) {
512 cells.subtract(siblings)
513 cells.insert(parent)
514 changed = true
515 break
516 }
517 }
518 }
519 return cells.sorted {
520 if $0.level != $1.level { return $0.level < $1.level }
521 if $0.y != $1.y { return $0.y > $1.y }
522 return $0.x < $1.x
523 }.map { code($0)! }.joined(separator:" ")
524 default: break
525 }

536 }.joined(separator:" ")
537 case "PARENT": return try a.level == 0 ? "NONE" : code(ancestor(a,a.level-1))!
538 case "CHILDREN":
539 if a.level+1 == precisionTable[system]!.count { return "NONE" }
540 let g = try grid(system,a.level+1)
541 let rx = g.w/a.w
542 let ry = g.h/a.h
543 if rx*ry > 100 { return "TOO MANY" }
544 var result: [String] = []
545 for j in stride(from:ry-1,through:0,by:-1) {
546 for i in 0..<rx { result.append(try shown(grid(system,g.level,a.x*rx+i,a.y*ry+j))) }
547 }
548 return result.joined(separator:" ")
549 case "COMMON":
550 let b = try decode(system,t[3])
551 for l in stride(from:min(a.level,b.level),through:0,by:-1) {
552 let parent = try ancestor(a,l)
553 if try parent == ancestor(b,l) { return code(parent)! }
554 }
555 return "NONE"
556 case "CONTAINS": return try a == point(system,a.level,coordinate(t[3],true),coordinate(t[4],false)) ? "YES" : "NO"
557 case "CONVERT":
558 try need(t[3] == "TO")

560 return try code(grid(g.system,g.level,(2*a.x+1)*g.w/(2*a.w),(2*a.y+1)*g.h/(2*a.h))) ?? "OUT OF AREA"
561 case "STEPS", "PATH":
562 let b = try decode(system,t[3])
563 if a.level != b.level { return "MISMATCH" }
564 let (dx,dy) = steps(a,b)
565 if command == "STEPS" { return "dx=\(dx) dy=\(dy)" }
566 let n = max(abs(dx),abs(dy))
567 if n+1 > 100 { return "TOO MANY" }
568 return try (0...n).map { i in
569 let ox = n == 0 ? 0 : floorDiv(2*i*dx+n,2*n)
570 let oy = n == 0 ? 0 : floorDiv(2*i*dy+n,2*n)
571 return try shown(grid(system,a.level,mod(a.x+ox,a.w),a.y+oy))
572 }.joined(separator:" ")
573 case "RELATE":
574 let b = try decode(t[3],t[4])
575 let ab = inside(a,b)
576 let ba = inside(b,a)
577 if ab && ba { return "EQUAL" }
578 if ab { return "INSIDE" }
579 if ba { return "CONTAINS" }
580 let ew = compare(a.x+1,a.w,b.x,b.w)
581 let we = compare(b.x+1,b.w,a.x,a.w)
582 let ns = compare(a.y+1,a.h,b.y,b.h)
583 let sn = compare(b.y+1,b.h,a.y,a.h)
584 if ew > 0 && we > 0 && ns > 0 && sn > 0 { return "OVERLAP" }
585 let dateline = a.x+1 == a.w && b.x == 0 || b.x+1 == b.w && a.x == 0
586 return ns >= 0 && sn >= 0 && (ew >= 0 && we >= 0 || dateline) ? "TOUCH" : "DISJOINT"
587 default: throw Invalid.argument
588 }
