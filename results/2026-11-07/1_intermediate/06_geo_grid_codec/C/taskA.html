107 require(value >= low && value <= high);

110static Integer location(const char *s, bool latitude) {
111 bool negative = s[0] == '-';
112 const char *body = s + (negative ? 1 : 0);
113 const char *dot = strchr(body, '.');
114 size_t head_length = dot ? (size_t)(dot - body) : strlen(body);
115 const char *head = slice(body, 0, head_length);
116 require(digits(head,1,3));
117 Integer value = integer(head,0,999) * 1000000;
118 if (dot) {
119 const char *fraction = dot + 1;
120 require(digits(fraction,1,6));
121 Integer tail = integer(fraction,0,999999);
122 for (size_t i = strlen(fraction); i < 6; ++i) tail *= 10;
123 value += tail;
124 }
125 if (negative) value = -value;
126 require(latitude ? value >= -90000000 && value <= 90000000 : value >= -180000000 && value < 180000000);
127 return value;
128}

165static char *encode(int system, Cell c) {
166 int p = grids[system][c.precision].precision;
167 Integer x = c.x, y = c.y;
168 Text result = {0};
169 if (system == 0) {
170 int a = (5*p+1)/2, b = 5*p/2, value = 0;
171 for (int i = 0; i < 5*p; ++i) {
172 int bit = i % 2 == 0 ? (int)(x >> (a-1-i/2) & 1) : (int)(y >> (b-1-i/2) & 1);
173 value = value * 2 + bit;
174 if (i % 5 == 4) {
175 character(&result,hash_alphabet[value]);
176 value = 0;
177 }
178 }
179 } else if (system == 1 || system == 2) {
180 int count = system == 1 ? p/2 : (int)minimum(p,10)/2;
181 int tail = -1;
182 if (system == 2 && p == 11) {
183 tail = (int)(4*(y%5)+x%4);
184 x /= 4;
185 y /= 5;
186 }
187 int xd[5], yd[5];
188 for (int i = count - 1; i >= 0; --i) {
189 int rx = system == 1 ? (int)strlen(maiden_alphabets[i]) : i == 0 ? 18 : 20;
190 int ry = system == 1 ? rx : i == 0 ? 9 : 20;
191 xd[i] = (int)(x % rx);
192 yd[i] = (int)(y % ry);
193 x /= rx;
194 y /= ry;
195 }
196 for (int i = 0; i < count; ++i) {
197 if (system == 1) {
198 character(&result,maiden_alphabets[i][xd[i]]);
199 character(&result,maiden_alphabets[i][yd[i]]);
200 } else {
201 character(&result,plus_alphabet[yd[i]]);
202 character(&result,plus_alphabet[xd[i]]);
203 }
204 }
205 if (tail >= 0) character(&result,plus_alphabet[tail]);
206 if (system == 2) {
207 while (result.length < 8) character(&result,'0');
208 memmove(result.data+9,result.data+8,result.length-8+1);
209 result.data[8] = '+';
210 ++result.length;
211 }
212 } else if (system == 3) {
213 Integer scale = p == 5 ? 1 : p == 6 ? 2 : 6;
214 Integer base_y = y/scale;
215 append(&result,padded(x/scale+1,3));
216 character(&result,letters24[base_y/24]);
217 character(&result,letters24[base_y%24]);
218 if (p >= 6) {
219 Integer xx = p == 7 ? x/3 : x, yy = p == 7 ? y/3 : y;
220 character(&result,(char)('1'+xx%2+2*(1-yy%2)));
221 }
222 if (p == 7) character(&result,(char)('1'+x%3+3*(2-y%3)));
223 } else if (system == 4) {
224 if (p == 2) return format("%c%c",letters24[x],letters24[y]);
225 int m = (p-4)/2;
226 Integer scale = p < 8 ? 1 : m == 2 ? 60 : m == 3 ? 600 : 6000;
227 Integer xx = x/scale, yy = y/scale;
228 append(&result,format("%c%c%c%c",letters24[xx/15],letters24[yy/15],letters24[xx%15],letters24[yy%15]));
229 if (p >= 8) {
230 append(&result,padded(x%scale,m));
231 append(&result,padded(y%scale,m));
232 }
233 } else {
234 Integer scale = mesh_scale[c.precision];
235 x -= 280*scale;
236 y -= 135*scale;
237 if (x < 0 || x >= 80*scale || y < 0 || y >= 100*scale) return copy_text(&result);
238 int count = c.precision+1;
239 const int radices[] = {0,8,10,2,2,2};
240 int xd[6], yd[6];
241 for (int i = count-1; i >= 1; --i) {
242 xd[i] = (int)(x%radices[i]);
243 yd[i] = (int)(y%radices[i]);
244 x /= radices[i];
245 y /= radices[i];
246 }
247 append(&result,padded(y,2));
248 append(&result,padded(x,2));
249 for (int i = 1; i < count; ++i) {
250 if (i <= 2) {
251 character(&result,(char)('0'+yd[i]));
252 character(&result,(char)('0'+xd[i]));
253 } else character(&result,(char)('1'+xd[i]+2*yd[i]));
254 }
255 }
256 return copy_text(&result);
257}
258static Cell decode(int system, const char *code) {
259 int p = (int)strlen(code);
260 Integer x = 0, y = 0;
261 if (system == 0) {
262 require(p >= 1 && p <= 9);
263 int position = 0;
264 for (int i = 0; i < p; ++i) {
265 int value = digit(hash_alphabet,code[i]);
266 for (int bit = 4; bit >= 0; --bit) {
267 if (position++ % 2 == 0) x = 2*x+(value>>bit&1);
268 else y = 2*y+(value>>bit&1);
269 }
270 }
271 } else if (system == 1) {
272 precision_id(system,p);
273 for (int i = 0; i < p/2; ++i) {
274 x = x*(Integer)strlen(maiden_alphabets[i])+digit(maiden_alphabets[i],code[2*i]);
275 y = y*(Integer)strlen(maiden_alphabets[i])+digit(maiden_alphabets[i],code[2*i+1]);
276 }
277 } else if (system == 2) {
278 int plus_count = 0;
279 for (int i = 0; i < p; ++i) plus_count += code[i] == '+';
280 require(p >= 9 && code[8] == '+' && plus_count == 1);
281 int prefix = 0;
282 while (prefix < 8 && code[prefix] != '0') {
283 digit(plus_alphabet,code[prefix]);
284 ++prefix;
285 }
286 require(prefix == 2 || prefix == 4 || prefix == 6 || prefix == 8);
287 for (int i = prefix; i < 8; ++i) require(code[i] == '0');
288 int suffix = p-9;
289 require(prefix == 8 ? suffix == 0 || suffix == 2 || suffix == 3 : suffix == 0);
290 require(digit(plus_alphabet,code[0]) <= 8 && digit(plus_alphabet,code[1]) <= 17);
291 char *sequence = format("%s%s",slice(code,0,(size_t)prefix),code+9);
292 p = (int)strlen(sequence);
293 for (int i = 0; i < (int)minimum(p,10)/2; ++i) {
294 y = y*(i == 0 ? 9 : 20)+digit(plus_alphabet,sequence[2*i]);
295 x = x*(i == 0 ? 18 : 20)+digit(plus_alphabet,sequence[2*i+1]);
296 }
297 if (p == 11) {
298 int tail = digit(plus_alphabet,sequence[10]);
299 x = x*4+tail%4;
300 y = y*5+tail/4;
301 }
302 } else if (system == 3) {
303 require(p >= 5 && p <= 7);
304 require(digits(slice(code,0,3),3,3));
305 x = integer(slice(code,0,3),1,720)-1;
306 int a = digit(letters24,code[3]), b = digit(letters24,code[4]);
307 require(a < 15);
308 y = 24*a+b;
309 if (p >= 6) {
310 require(code[5] >= '1' && code[5] <= '4');
311 int q = code[5]-'1';
312 x = 2*x+q%2;
313 y = 2*y+1-q/2;
314 }
315 if (p == 7) {
316 require(code[6] >= '1' && code[6] <= '9');
317 int q = code[6]-'1';
318 x = 3*x+q%3;
319 y = 3*y+2-q/3;
320 }
321 } else if (system == 4) {
322 precision_id(system,p);
323 x = digit(letters24,code[0]);
324 y = digit(letters24,code[1]);
325 require(y < 12);
326 if (p >= 4) {
327 int a = digit(letters24,code[2]), b = digit(letters24,code[3]);
328 require(a < 15 && b < 15);
329 x = 15*x+a;
330 y = 15*y+b;
331 }
332 if (p >= 8) {
333 int m = (p-4)/2;
334 Integer scale = m == 2 ? 60 : m == 3 ? 600 : 6000;
335 const char *e = slice(code,4,(size_t)m), *n = slice(code,4+(size_t)m,(size_t)m);
336 require(digits(e,m,m) && digits(n,m,m));
337 x = x*scale+integer(e,0,scale-1);
338 y = y*scale+integer(n,0,scale-1);
339 }
340 } else {
341 int precision = precision_id(system,p);
342 require(digits(code,p,p));
343 y = integer(slice(code,0,2),0,99);
344 x = integer(slice(code,2,2),0,79);
345 if (p >= 6) {
346 require(code[4] <= '7' && code[5] <= '7');
347 y = y*8+code[4]-'0';
348 x = x*8+code[5]-'0';
349 }
350 if (p >= 8) {
351 y = y*10+code[6]-'0';
352 x = x*10+code[7]-'0';
353 }
354 for (int i = 8; i < p; ++i) {
355 require(code[i] >= '1' && code[i] <= '4');
356 int q = code[i]-'1';
357 x = x*2+q%2;
358 y = y*2+q/2;
359 }
360 x += 280*mesh_scale[precision];
361 y += 135*mesh_scale[precision];
362 }
363 return (Cell){precision_id(system,p),x,y};
364}

374static const char *relation(int sa, Cell a, int sb, Cell b) {
375 Grid ga = grids[sa][a.precision], gb = grids[sb][b.precision];
376 int west = compare(a.x,ga.width,b.x,gb.width), east = compare(a.x+1,ga.width,b.x+1,gb.width);
377 int south = compare(a.y,ga.height,b.y,gb.height), north = compare(a.y+1,ga.height,b.y+1,gb.height);
378 if (!west && !east && !south && !north) return "EQUAL";
379 if (west >= 0 && east <= 0 && south >= 0 && north <= 0) return "INSIDE";
380 if (west <= 0 && east >= 0 && south <= 0 && north >= 0) return "CONTAINS";
381 int x1 = compare(a.x,ga.width,b.x+1,gb.width), x2 = compare(b.x,gb.width,a.x+1,ga.width);
382 int y1 = compare(a.y,ga.height,b.y+1,gb.height), y2 = compare(b.y,gb.height,a.y+1,ga.height);
383 if (x1 < 0 && x2 < 0 && y1 < 0 && y2 < 0) return "OVERLAP";
384 bool wrap = (a.x == 0 && b.x+1 == gb.width) || (b.x == 0 && a.x+1 == ga.width);
385 if (y1 <= 0 && y2 <= 0 && ((x1 <= 0 && x2 <= 0) || wrap)) return "TOUCH";
386 return "DISJOINT";
387}
388static int prune(int system, Cell *cells, int count) {
389 bool removed[100] = {false};
390 for (int i = 0; i < count; ++i) {
391 for (int j = 0; j < count; ++j) {
392 if (cells[j].precision < cells[i].precision && equal(ancestor(system,cells[i],cells[j].precision),cells[j])) {
393 removed[i] = true;
394 break;
395 }
396 }
397 }
398 int length = 0;
399 for (int i = 0; i < count; ++i) if (!removed[i]) cells[length++] = cells[i];
400 return length;
401}
402static int cell_order(const void *left, const void *right) {
403 const Cell *a = left, *b = right;
404 if (a->precision != b->precision) return a->precision < b->precision ? -1 : 1;
405 if (a->y != b->y) return a->y > b->y ? -1 : 1;
406 return a->x < b->x ? -1 : a->x > b->x ? 1 : 0;
407}
408static char *compact(int system, char **codes, int code_count) {
409 Cell cells[100];
410 int count = 0;
411 for (int i = 0; i < code_count; ++i) {
412 Cell c = decode(system,codes[i]);
413 bool duplicate = false;
414 for (int j = 0; j < count; ++j) duplicate |= equal(c,cells[j]);
415 if (!duplicate) cells[count++] = c;
416 }
417 count = prune(system,cells,count);
418 for (;;) {
419 bool changed = false;
420 for (int i = 0; i < count; ++i) {
421 Cell c = cells[i];
422 if (!c.precision) continue;
423 Cell parent = ancestor(system,c,c.precision-1);
424 Grid g = grids[system][c.precision], pg = grids[system][parent.precision];
425 Integer needed = (g.width/pg.width)*(g.height/pg.height);
426 if (needed > count) continue;
427 bool children[100];
428 int matches = 0;
429 for (int j = 0; j < count; ++j) {
430 children[j] = cells[j].precision == c.precision && equal(ancestor(system,cells[j],parent.precision),parent);
431 matches += children[j];
432 }
433 if (matches != needed) continue;
434 int length = 0;
435 for (int j = 0; j < count; ++j) if (!children[j]) cells[length++] = cells[j];
436 cells[length++] = parent;
437 count = prune(system,cells,length);
438 changed = true;
439 break;
440 }
441 if (!changed) break;
442 }
443 qsort(cells,(size_t)count,sizeof(*cells),cell_order);
444 Text result = {0};
445 for (int i = 0; i < count; ++i) item(&result,encode(system,cells[i]));
446 return copy_text(&result);
447}

458static const char *execute(char **a, int count) {
459 const char *names[] = {
460 "ENCODE", "DECODE", "NEIGHBORS", "PARENT", "CHILDREN", "COMMON",
461 "COMPACT", "CONTAINS", "COVER", "CONVERT", "STEPS", "PATH", "RELATE",
462 "SIZE", "SHORTEN", "RECOVER", "UTMZONE", "OSGB", "UNOSGB", "MGRS",
463 "UNMGRS", "QUADKEY", "UNQUADKEY"
464 };
465 const int lengths[] = {5,3,3,3,3,4,0,5,7,6,4,4,5,3,4,4,3,4,0,6,0,4,2};
466 int command = -1;
467 for (int i = 0; i < 23; ++i) if (!strcmp(a[0],names[i])) command = i;
468 require(command >= 0);
469 if (command == 6) require(count >= 3 && count <= 102);
470 else if (command == 18) require(count == 2 || count == 4);
471 else if (command == 20) require(count == 3 || count == 5);
472 else require(count == lengths[command]);
473 if (command <= 13) {
474 int system = system_id(a[1]);
475 if (command == 0) {
476 Integer lat = location(a[2],true), lon = location(a[3],false);
477 int precision = precision_argument(system,a[4]);
478 const char *code = encode(system,point(system,precision,lat,lon));
479 return *code ? code : "OUT OF AREA";
480 }
481 if (command == 8) {
482 int precision = precision_argument(system,a[2]);
483 Integer lat1 = location(a[3],true), lon1 = location(a[4],false), lat2 = location(a[5],true), lon2 = location(a[6],false);
484 require(lat1 <= lat2);
485 Cell first = point(system,precision,lat1,lon1), last = point(system,precision,lat2,lon2);
486 Grid g = grids[system][precision];
487 Integer columns = lon1 <= lon2 ? last.x-first.x+1 : minimum(g.width,last.x-first.x+1+g.width);
488 if ((last.y-first.y+1)*columns > 100) return "TOO MANY";
489 Text result = {0};
490 for (Integer y = last.y; y >= first.y; --y) {
491 for (Integer i = 0; i < columns; ++i) item(&result,display(system,(Cell){precision,(first.x+i)%g.width,y}));
492 }
493 return copy_text(&result);
494 }
495 if (command == 13) {
496 int precision = precision_argument(system,a[2]);
497 Grid g = grids[system][precision];
498 Integer total = system == 5 ? 8000*mesh_scale[precision]*mesh_scale[precision] : g.width*g.height;
499 return format("w=%s h=%s cells=%lld",coordinate(360000000/g.width),coordinate(180000000/g.height),total);
500 }
501 if (command == 6) return compact(system,a+2,count-2);
502 Cell c = decode(system,a[2]);
503 Grid g = grids[system][c.precision];
504 if (command == 1) {
