51fn compareUnsigned(a: u64, b: u64) Relation {
52 return if (a < b) .LT else if (a > b) .GT else .EQ;
53}
54fn compareSigned(a: i64, b: i64) Relation {
55 return if (a < b) .LT else if (a > b) .GT else .EQ;
56}
57fn bitLength(value: u64) i32 {
58 var n = value;
59 var length: i32 = 0;
60 while (n != 0) : (n >>= 1) length += 1;
61 return length;
62}
63fn gcd(a: u64, b: u64) u64 {
64 var x = a;
65 var y = b;
66 while (y != 0) {
67 const remainder = x % y;
68 x = y;
69 y = remainder;
70 }
71 return x;
72}
73const Scaled = struct { n: u64, d: u64 };
74fn scaled(n: u64, d: u64, exponent: i32) Scaled {
75 return if (exponent >= 0) .{ .n = n << @intCast(exponent), .d = d } else .{ .n = n, .d = d << @intCast(-exponent) };
76}
77fn integerSqrt(value: u64) u64 {
78 if (value == 0) return 0;
79 var root = @as(u64, 1) << @intCast(@divTrunc(bitLength(value) + 1, 2));
80 while (true) {
81 const next = (root + value / root) / 2;
82 if (next >= root) return root;
83 root = next;
84 }
85}
86fn increment(mode: Mode, negative: bool, whole: u64, inexact: bool, halfway: Relation) bool {
87 if (!inexact) return false;
88 return switch (mode) {
89 .RNE => halfway == .GT or halfway == .EQ and whole % 2 != 0,
90 .RNA => halfway != .LT,
91 .RTZ => false,
92 .RUP => !negative,
93 .RDN => negative,
94 };
95}
96const Units = struct { whole: u64, inexact: bool };
97fn roundUnits(exact: Exact, exponent: i32, mode: Mode, square_root: bool) Units {
98 const ratio = scaled(exact.n, exact.d, exact.exponent - (if (square_root) 2 * exponent else exponent));
99 var whole = if (square_root) integerSqrt(ratio.n / ratio.d) else ratio.n / ratio.d;
100 const inexact = if (square_root) whole * whole * ratio.d != ratio.n else ratio.n % ratio.d != 0;
101 const midpoint = 2 * whole + 1;
102 const halfway = if (square_root) compareUnsigned(4 * ratio.n, ratio.d * midpoint * midpoint) else compareUnsigned(2 * (ratio.n % ratio.d), ratio.d);
103 if (increment(mode, exact.negative, whole, inexact, halfway)) whole += 1;
104 return .{ .whole = whole, .inexact = inexact };
105}

107const Flags = struct {
108 bits: u8 = 0,
109 const invalid: u8 = 1;
110 const divbyzero: u8 = 2;
111 const overflow: u8 = 4;
112 const underflow: u8 = 8;
113 const inexact: u8 = 16;
114 pub fn format(value: Flags, out: *Writer) Writer.Error!void {
115 try out.writeAll("flags=");
116 if (value.bits == 0) return out.writeAll("none");
117 const names = [_][]const u8{ "invalid", "divbyzero", "overflow", "underflow", "inexact" };
118 var first = true;
119 for (names, 0..) |name, i| if (value.bits & (@as(u8, 1) << @intCast(i)) != 0) {
120 if (!first) try out.writeByte(',');
121 try out.writeAll(name);
122 first = false;
123 };
124 }

166const Exact = struct {
167 n: u64 = 0,
168 exponent: i32 = 0,
169 d: u64 = 1,
170 negative: bool = false,
171
172 fn negated(exact: Exact) Exact {
173 var result = exact;
174 result.negative = !result.negative;
175 return result;
176 }
177 fn log2Floor(exact: Exact) i32 {
178 const t = bitLength(exact.n) - bitLength(exact.d) + exact.exponent;
179 const ratio = scaled(exact.n, exact.d, exact.exponent - t);
180 return if (ratio.n < ratio.d) t - 1 else t;
181 }
182 fn product(a: Exact, b: Exact) Exact {
183 return .{ .n = a.n * b.n, .exponent = a.exponent + b.exponent, .d = a.d * b.d, .negative = a.negative != b.negative };
184 }
185 fn sum(terms: []const Exact, mode: Mode) Exact {
186 var exponent: ?i32 = null;
187 for (terms) |term| if (term.n != 0 and (exponent == null or term.exponent < exponent.?)) {
188 exponent = term.exponent;
189 };
190 if (exponent == null) {
191 var same = true;
192 for (terms) |term| same = same and term.negative == terms[0].negative;
193 return .{ .negative = if (same) terms[0].negative else mode == .RDN };
194 }
195 var numerator: i64 = 0;
196 for (terms) |term| {
197 if (term.n == 0) continue;
198 const aligned: i64 = @intCast(term.n << @intCast(term.exponent - exponent.?));
199 numerator += if (term.negative) -aligned else aligned;
200 }
201 return .{ .n = @intCast(if (numerator < 0) -numerator else numerator), .exponent = exponent.?, .negative = numerator < 0 or numerator == 0 and mode == .RDN };
202 }
203 fn decimal(exact: Exact, keep_zero_sign: bool) Decimal {
204 return .{ .exact = exact, .keep_zero_sign = keep_zero_sign };
205 }
206 pub fn format(exact: Exact, out: *Writer) Writer.Error!void {
207 if (exact.n == 0) return out.writeByte('0');
208 const ratio = scaled(exact.n, exact.d, exact.exponent);
209 const factor = gcd(ratio.n, ratio.d);
210 if (exact.negative) try out.writeByte('-');
211 try out.print("{d}", .{ratio.n / factor});
212 if (ratio.d != factor) try out.print("/{d}", .{ratio.d / factor});
213 }
214};
215const Decimal = struct {
216 exact: Exact,
217 keep_zero_sign: bool = false,
218 pub fn format(value: Decimal, out: *Writer) Writer.Error!void {
219 const exact = value.exact;
220 const ratio = scaled(exact.n, exact.d, exact.exponent);
221 if (exact.negative and (exact.n != 0 or value.keep_zero_sign)) try out.writeByte('-');
222 try out.print("{d}", .{ratio.n / ratio.d});
223 var remainder = ratio.n % ratio.d;
224 if (remainder == 0) return;
225 try out.writeByte('.');
226 while (remainder != 0) {
227 remainder *= 10;
228 try out.writeByte(@intCast('0' + remainder / ratio.d));
229 remainder %= ratio.d;
230 }
231 }

234const Fields = struct { negative: bool, exponent: u32, fraction: u32, q: i32 };
235const MiniFormat = struct {
236 name: Name,
237 e: u6,
238 m: u6,
239 width: u6,
240 bias: i32,
241 emin: i32,
242 emax: i32,
243 hidden: u32,
244 exponent_mask: u32,
245 sign_bit: u32,
246 infinity: u32,
247
248 fn init(name: []const u8, e: u6, m: u6) MiniFormat {
249 const bias: i32 = @intCast((@as(u32, 1) << @intCast(e - 1)) - 1);
250 const hidden = @as(u32, 1) << @intCast(m);
251 const mask = (@as(u32, 1) << @intCast(e)) - 1;
252 return .{ .name = .init(name), .e = e, .m = m, .width = 1 + e + m, .bias = bias, .emin = 1 - bias, .emax = bias, .hidden = hidden, .exponent_mask = mask, .sign_bit = @as(u32, 1) << @intCast(e + m), .infinity = mask * hidden };
253 }
254 fn fields(f: MiniFormat, pattern: u32) Fields {
255 const exponent = (pattern >> @intCast(f.m)) & f.exponent_mask;
256 return .{ .negative = pattern & f.sign_bit != 0, .exponent = exponent, .fraction = pattern & (f.hidden - 1), .q = (if (exponent == 0) f.emin else @as(i32, @intCast(exponent)) - f.bias) - f.m };
257 }
258 fn isNaN(f: MiniFormat, pattern: u32) bool {
259 return f.fields(pattern).exponent == f.exponent_mask and f.fields(pattern).fraction != 0;
260 }
261 fn isSignaling(f: MiniFormat, pattern: u32) bool {
262 return f.isNaN(pattern) and f.fields(pattern).fraction < f.hidden / 2;
263 }
264 fn isInfinity(f: MiniFormat, pattern: u32) bool {
265 return pattern & (f.sign_bit - 1) == f.infinity;
266 }
267 fn isZero(f: MiniFormat, pattern: u32) bool {
268 return pattern & (f.sign_bit - 1) == 0;
269 }
270 fn isFinite(f: MiniFormat, pattern: u32) bool {
271 return f.fields(pattern).exponent != f.exponent_mask;
272 }
273 fn nan(f: MiniFormat) u32 {
274 return f.infinity + f.hidden / 2;
275 }
276 fn zero(f: MiniFormat, negative: bool) u32 {
277 return if (negative) f.sign_bit else 0;
278 }
279 fn inf(f: MiniFormat, negative: bool) u32 {
280 return f.zero(negative) + f.infinity;
281 }
282 fn flip(f: MiniFormat, pattern: u32) u32 {
283 return pattern ^ f.sign_bit;
284 }
285 fn exact(f: MiniFormat, pattern: u32) Exact {
286 const parts = f.fields(pattern);
287 return .{ .n = if (parts.exponent == 0) parts.fraction else f.hidden + parts.fraction, .exponent = parts.q, .negative = parts.negative };
288 }
289 fn bits(f: MiniFormat, pattern: u32) BitText {
290 return .{ .f = f, .pattern = pattern };
291 }
292 fn hex(f: MiniFormat, pattern: u32) Hex {
293 return .{ .value = pattern, .width = (f.width + 3) / 4 };
294 }
295 fn valueText(f: MiniFormat, pattern: u32) ValueText {
296 return .{ .f = f, .pattern = pattern };
297 }
298 fn patternText(f: MiniFormat, pattern: u32) PatternText {
299 return .{ .f = f, .pattern = pattern };
300 }
301 fn resultText(f: MiniFormat, result: Result) ResultText {
302 return .{ .f = f, .result = result };
303 }
304 fn overflow(f: MiniFormat, negative: bool, mode: Mode) Result {
305 const infinity = mode == .RNE or mode == .RNA or mode == .RUP and !negative or mode == .RDN and negative;
306 return .{ .pattern = f.zero(negative) + (if (infinity) f.infinity else f.infinity - 1), .flags = Flags.overflow | Flags.inexact };
307 }
308 fn roundNonzero(f: MiniFormat, exact_value: Exact, mode: Mode, square_root: bool) Result {
309 const logarithm = exact_value.log2Floor();
310 const t = if (square_root) @divFloor(logarithm, 2) else logarithm;
311 const negative = if (square_root) false else exact_value.negative;
312 if (t > f.emax) return f.overflow(negative, mode);
313 var exponent = @max(t, f.emin) - f.m;
314 const units = roundUnits(exact_value, exponent, mode, square_root);
315 var n = units.whole;
316 if (n == 2 * f.hidden) {
317 n = f.hidden;
318 exponent += 1;
319 }
320 if (exponent > f.emax - f.m) return f.overflow(negative, mode);
321 const magnitude = if (n >= f.hidden) @as(u32, @intCast(exponent + f.m + f.bias)) * f.hidden + @as(u32, @intCast(n)) - f.hidden else @as(u32, @intCast(n));
322 var flags: u8 = if (units.inexact) Flags.inexact else 0;
323 if (units.inexact and t < f.emin and (t < f.emin - 1 or roundUnits(exact_value, t - f.m, mode, square_root).whole < 2 * f.hidden)) flags |= Flags.underflow;
324 return .{ .pattern = f.zero(negative) + magnitude, .flags = flags };
325 }
326 fn round(f: MiniFormat, exact_value: Exact, mode: Mode) Result {
327 return if (exact_value.n == 0) .{ .pattern = f.zero(exact_value.negative) } else f.roundNonzero(exact_value, mode, false);
328 }

372const RealInput = union(enum) {

377 return switch (input) {

381 };

384 switch (input) {
385 .finite => |number| try out.print("{f}", .{number}),
386 .infinity => |negative| try out.writeAll(if (negative) "-inf" else "inf"),

392fn calculate(f: MiniFormat, operation: Operation, mode: Mode, inputs: []const u32) Result {
393 var any_nan = false;
394 var signaling = false;
395 for (inputs) |pattern| {
396 any_nan = any_nan or f.isNaN(pattern);
397 signaling = signaling or f.isSignaling(pattern);
398 }
399 if (any_nan) return .{ .pattern = f.nan(), .flags = if (signaling) Flags.invalid else 0 };
400 const a = inputs[0];
401 const negative = f.fields(a).negative;
402 const invalid: Result = .{ .pattern = f.nan(), .flags = Flags.invalid };
403 if (operation == .SQRT) {
404 if (f.isZero(a)) return .{ .pattern = a };
405 if (negative) return invalid;
406 if (f.isInfinity(a)) return .{ .pattern = a };
407 return f.roundNonzero(f.exact(a), mode, true);
408 }
409 const b = if (operation == .SUB) f.flip(inputs[1]) else inputs[1];
410 const sigma = negative != f.fields(b).negative;
411 if (operation == .ADD or operation == .SUB) {
412 if (f.isInfinity(a) and f.isInfinity(b) and sigma) return invalid;
413 if (f.isInfinity(a)) return .{ .pattern = a };
414 if (f.isInfinity(b)) return .{ .pattern = b };
415 return f.round(Exact.sum(&.{ f.exact(a), f.exact(b) }, mode), mode);
416 }
417 if (operation == .MUL) {
418 if (f.isInfinity(a) or f.isInfinity(b)) return if (f.isZero(a) or f.isZero(b)) invalid else .{ .pattern = f.inf(sigma) };
419 if (f.isZero(a) or f.isZero(b)) return .{ .pattern = f.zero(sigma) };
420 return f.round(Exact.product(f.exact(a), f.exact(b)), mode);
421 }
422 if (operation == .DIV) {
423 if (f.isInfinity(a)) return if (f.isInfinity(b)) invalid else .{ .pattern = f.inf(sigma) };
424 if (f.isInfinity(b)) return .{ .pattern = f.zero(sigma) };
425 if (f.isZero(b)) return if (f.isZero(a)) invalid else .{ .pattern = f.inf(sigma), .flags = Flags.divbyzero };
426 if (f.isZero(a)) return .{ .pattern = f.zero(sigma) };
427 const x = f.exact(a);
428 const y = f.exact(b);
429 return f.round(.{ .n = x.n * y.d, .exponent = x.exponent - y.exponent, .d = x.d * y.n, .negative = sigma }, mode);
430 }
431 const c = inputs[2];
432 if (f.isInfinity(a) or f.isInfinity(b)) {
433 if (f.isZero(a) or f.isZero(b) or f.isInfinity(c) and f.fields(c).negative != sigma) return invalid;
434 return .{ .pattern = f.inf(sigma) };
435 }
436 if (f.isInfinity(c)) return .{ .pattern = c };
437 return f.round(Exact.sum(&.{ Exact.product(f.exact(a), f.exact(b)), f.exact(c) }, mode), mode);
438}

440fn nextUp(f: MiniFormat, pattern: u32) u32 {
441 return if (pattern == f.infinity) pattern else if (f.isZero(pattern)) 1 else if (f.fields(pattern).negative) pattern - 1 else pattern + 1;

447const Program = struct {
448 formats: [13]MiniFormat = undefined,
449 format_count: usize = 3,
450 scratch: Allocator,
451 out: *Writer,
452 error_message: []const u8 = "",
453
454 fn init(scratch: Allocator, out: *Writer) Program {
455 var program: Program = .{ .scratch = scratch, .out = out };
456 program.formats[0] = .init("HALF", 5, 10);
457 program.formats[1] = .init("E5M2", 5, 2);
458 program.formats[2] = .init("E4M3", 4, 3);
459 return program;
460 }
461 fn fail(program: *Program, comptime fmt: []const u8, args: anytype) Error {
462 program.error_message = std.fmt.allocPrint(program.scratch, fmt, args) catch return error.OutOfMemory;
463 return error.BadCommand;
464 }
465 fn findFormat(program: *const Program, name: []const u8) ?MiniFormat {
466 for (program.formats[0..program.format_count]) |f| if (f.name.matches(name)) return f;
467 return null;
468 }
469 fn getFormat(program: *Program, name: []const u8) Error!MiniFormat {
470 return program.findFormat(name) orelse return program.fail("UNKNOWN FORMAT {s}", .{name});
471 }
472 fn parseMode(program: *Program, words: []const []const u8, index: usize) Error!Mode {
473 if (index == words.len) return .RNE;
474 return std.meta.stringToEnum(Mode, words[index]) orelse return program.fail("BAD MODE {s}", .{words[index]});
475 }
476 fn parseReal(program: *Program, word: []const u8) Error!RealInput {
477 if (equal(word, "nan")) return .nan;
478 if (equal(word, "inf") or equal(word, "-inf")) return .{ .infinity = word[0] == '-' };
479 if (std.mem.startsWith(u8, word, "0b") or std.mem.startsWith(u8, word, "0x")) return program.fail("BAD OPERAND {s}", .{word});
480 const negative = word[0] == '-';
481 const body = if (negative) word[1..] else word;
482 if (std.mem.indexOfScalar(u8, body, '/')) |slash| {
483 const n = parseNumber(body[0..slash]);
484 const d = parseNumber(body[slash + 1 ..]);
485 if (d == 0) return program.fail("BAD OPERAND {s}", .{word});
486 return .{ .finite = .{ .n = n, .d = d, .negative = negative } };
487 }
488 var n: u64 = 0;
489 var d: u64 = 1;
490 var fractional = false;
491 for (body) |ch| {
492 if (ch == '.') {
493 fractional = true;
494 continue;
495 }
496 n = n * 10 + ch - '0';
497 if (fractional) d *= 10;
498 }
499 return .{ .finite = .{ .n = n, .d = d, .negative = negative } };
500 }
501 fn operand(program: *Program, f: MiniFormat, word: []const u8) Error!u32 {
502 const bits = std.mem.startsWith(u8, word, "0b");
503 const hex = std.mem.startsWith(u8, word, "0x");
504 if (bits or hex) {
505 const width: usize = if (bits) f.width else (f.width + 3) / 4;
506 if (word.len - 2 != width) return program.fail("BAD OPERAND {s}", .{word});
507 const pattern = std.fmt.parseInt(u32, word[2..], if (bits) 2 else 16) catch unreachable;
508 if (pattern >= @as(u32, 1) << @intCast(f.width)) return program.fail("BAD OPERAND {s}", .{word});
509 return pattern;
510 }
511 return (try program.parseReal(word)).encoded(f, .RNE).pattern;
512 }
513 fn operands(program: *Program, f: MiniFormat, words: []const []const u8, destination: []u32) Error!void {
514 for (words, destination) |word, *pattern| pattern.* = try program.operand(f, word);
515 }
516 fn summary(program: *Program, f: MiniFormat, mode: Mode, patterns: []const u32, result: Result) Writer.Error!void {
517 var terms: [20]Exact = undefined;
518 for (patterns, 0..) |pattern, i| terms[i] = f.exact(pattern);
519 const exact = Exact.sum(terms[0..patterns.len], mode);
520 try line(program.out, "RESULT {f}", .{f.resultText(result)});
521 try line(program.out, "EXACT {f}", .{exact.decimal(false)});
522 try line(program.out, "ROUNDED {f}", .{f.resultText(f.round(exact, mode))});
523 try line(program.out, "DIFF {f}", .{Difference{ .f = f, .pattern = result.pattern, .exact = exact }});
524 }
525 fn summation(program: *Program, command: Command, f: MiniFormat, mode: Mode, patterns: []const u32) Error!void {
526 for (patterns) |pattern| if (!f.isFinite(pattern)) return program.fail("NOT FINITE", .{});
527 try line(program.out, "{s} {f} {s} count={d}", .{ @tagName(command), f.name, @tagName(mode), patterns.len });
528 var sum = patterns[0];
529 var compensation: u32 = 0;
530 var flags: u8 = 0;
531 for (patterns[1..], 1..) |pattern, i| {
532 if (command == .SUM) {
533 const added = calculate(f, .ADD, mode, &.{ sum, pattern });
534 sum = added.pattern;
535 flags |= added.flags;
536 try line(program.out, "PARTIAL {d} {f}", .{ i + 1, f.resultText(added) });
537 } else {
538 const y = calculate(f, .SUB, mode, &.{ pattern, compensation });
539 const t = calculate(f, .ADD, mode, &.{ sum, y.pattern });
540 const d = calculate(f, .SUB, mode, &.{ t.pattern, sum });
541 const c = calculate(f, .SUB, mode, &.{ d.pattern, y.pattern });
542 sum = t.pattern;
543 compensation = c.pattern;
544 const step = y.flags | t.flags | d.flags | c.flags;
545 flags |= step;
546 try line(program.out, "KSTEP {d} sum={f} comp={f} {f}", .{ i + 1, f.valueText(sum), f.valueText(compensation), Flags{ .bits = step } });
547 }
548 }
549 try program.summary(f, mode, patterns, .{ .pattern = sum, .flags = flags });
550 }
551 fn dot(program: *Program, f: MiniFormat, mode: Mode, a: []const u32, b: []const u32) Error!void {
552 if (a.len != b.len) return program.fail("LENGTH MISMATCH", .{});
553 for (a) |pattern| if (!f.isFinite(pattern)) return program.fail("NOT FINITE", .{});
554 for (b) |pattern| if (!f.isFinite(pattern)) return program.fail("NOT FINITE", .{});
555 var naive = calculate(f, .MUL, mode, &.{ a[0], b[0] });
556 var chained = naive;
557 for (1..a.len) |i| {
558 const product = calculate(f, .MUL, mode, &.{ a[i], b[i] });
559 const added = calculate(f, .ADD, mode, &.{ naive.pattern, product.pattern });
560 naive = .{ .pattern = added.pattern, .flags = naive.flags | product.flags | added.flags };
561 const fused = calculate(f, .FMA, mode, &.{ a[i], b[i], chained.pattern });
562 chained = .{ .pattern = fused.pattern, .flags = chained.flags | fused.flags };
563 }
564 var terms: [10]Exact = undefined;
565 for (a, b, 0..) |x, y, i| terms[i] = Exact.product(f.exact(x), f.exact(y));
566 const exact = Exact.sum(terms[0..a.len], mode);
567 try line(program.out, "DOT {f} {s} n={d}", .{ f.name, @tagName(mode), a.len });
568 try line(program.out, "NAIVE {f}", .{f.resultText(naive)});
569 try line(program.out, "CHAINED {f}", .{f.resultText(chained)});
570 try line(program.out, "EXACT {f}", .{exact.decimal(false)});
571 try line(program.out, "FUSED {f}", .{f.resultText(f.round(exact, mode))});
572 try line(program.out, "DIFF naive={f} chained={f}", .{ Difference{ .f = f, .pattern = naive.pattern, .exact = exact }, Difference{ .f = f, .pattern = chained.pattern, .exact = exact } });
573 }
574 fn table(program: *Program, f: MiniFormat) Error!void {
575 if (f.width > 8) return program.fail("TABLE TOO LARGE", .{});
576 const count = f.infinity;
577 try line(program.out, "TABLE {f} count={d}", .{ f.name, count });
578 var pattern: u32 = 0;
579 while (pattern < count) : (pattern += 1) {
580 try program.out.print("ENTRY {d} {f} gap=", .{ pattern, f.patternText(pattern) });
581 if (pattern == count - 1) try line(program.out, "-", .{}) else {
582 const gap = Exact.sum(&.{ f.exact(pattern + 1), f.exact(pattern).negated() }, .RNE);
583 try line(program.out, "{f}", .{gap.decimal(false)});
584 }
585 }
586 try line(program.out, "MAX {f}", .{f.patternText(f.infinity - 1)});
587 try line(program.out, "MINNORMAL {f}", .{f.patternText(f.hidden)});
588 try line(program.out, "MINSUBNORMAL {f}", .{f.patternText(1)});
589 const epsilon: Exact = .{ .n = 1, .exponent = -@as(i32, f.m) };
590 try line(program.out, "EPSILON {f}", .{epsilon.decimal(false)});
591 try line(program.out, "PATTERNS finite={d} infinite=2 nan={d} total={d}", .{ 2 * count, 2 * (f.hidden - 1), @as(u32, 1) << @intCast(f.width) });
592 }

594 fn execute(program: *Program, words: []const []const u8) Error!void {
595 const command = std.meta.stringToEnum(Command, words[0]).?;
596 const out = program.out;
597 if (command == .FORMAT) {
598 const name = words[1];
599 const e = parseNumber(words[2]);
600 const m = parseNumber(words[3]);
601 if (program.findFormat(name) != null) return program.fail("FORMAT EXISTS {s}", .{name});
602 if (e < 2 or e > 5 or m < 1 or m > 10) return program.fail("BAD FORMAT", .{});
603 const f: MiniFormat = .init(name, @intCast(e), @intCast(m));
604 program.formats[program.format_count] = f;
605 program.format_count += 1;
606 try line(out, "FORMAT {f} e={d} m={d} bits={d} bias={d} emin={d} emax={d}", .{ f.name, e, m, f.width, f.bias, f.emin, f.emax });
607 return;
608 }
609 const f = try program.getFormat(words[1]);
610 switch (command) {
611 .ENCODE => {
612 const input = try program.parseReal(words[2]);
613 const mode = try program.parseMode(words, 3);
614 try line(out, "ENCODE {f} {s} {f} {f}", .{ f.name, @tagName(mode), input, f.resultText(input.encoded(f, mode)) });
615 },
616 .CONVERT => {
617 const g = try program.getFormat(words[2]);
618 const a = try program.operand(f, words[3]);
619 const mode = try program.parseMode(words, 4);
620 const result: Result = if (f.isNaN(a)) .{ .pattern = g.nan(), .flags = if (f.isSignaling(a)) Flags.invalid else 0 } else if (f.isInfinity(a)) .{ .pattern = g.inf(f.fields(a).negative) } else g.round(f.exact(a), mode);
621 try line(out, "CONVERT {f} {f} {s} {f}", .{ f.name, g.name, @tagName(mode), g.resultText(result) });
622 },
623 .ADD, .SUB, .MUL, .DIV, .SQRT, .FMA => {
624 const operation = command.operation().?;
625 const count: usize = if (operation == .SQRT) 1 else if (operation == .FMA) 3 else 2;
626 var patterns: [3]u32 = undefined;
627 try program.operands(f, words[2 .. 2 + count], patterns[0..count]);
628 const mode = try program.parseMode(words, 2 + count);
629 try line(out, "{s} {f} {s} {f}", .{ @tagName(command), f.name, @tagName(mode), f.resultText(calculate(f, operation, mode, patterns[0..count])) });
630 },
631 .SUM, .KAHAN => {
632 const mode = try program.parseMode(words, 2);
633 var patterns: [20]u32 = undefined;
634 const count = words.len - 3;
635 try program.operands(f, words[3..], patterns[0..count]);
636 try program.summation(command, f, mode, patterns[0..count]);
637 },
638 .DOT => {
639 const mode = try program.parseMode(words, 2);
640 var separator: usize = 3;
641 while (!equal(words[separator], "|")) separator += 1;
642 var a: [10]u32 = undefined;
643 var b: [10]u32 = undefined;
644 const a_count = separator - 3;
645 const b_count = words.len - separator - 1;
646 try program.operands(f, words[3..separator], a[0..a_count]);
647 try program.operands(f, words[separator + 1 ..], b[0..b_count]);
648 try program.dot(f, mode, a[0..a_count], b[0..b_count]);
649 },
650 .TABLE => try program.table(f),
651 .DECODE => {
652 const a = try program.operand(f, words[2]);
653 const parts = f.fields(a);
654 const classification: []const u8 = if (f.isNaN(a)) (if (f.isSignaling(a)) "snan" else "qnan") else if (f.isInfinity(a)) "infinity" else if (f.isZero(a)) "zero" else if (parts.exponent == 0) "subnormal" else "normal";
655 try line(out, "DECODE {f} {f} {f} {s} sign={d}", .{ f.name, f.bits(a), f.hex(a), classification, @intFromBool(parts.negative) });
656 try out.print("FIELDS exponent={d} fraction={d}", .{ parts.exponent, parts.fraction });
657 if (f.isFinite(a)) {
658 try line(out, " unbiased={d} significand={d}.{f}", .{ parts.q + f.m, @intFromBool(parts.exponent != 0), Binary{ .value = parts.fraction, .width = f.m } });
659 try line(out, "VALUE {f} {f}", .{ f.exact(a), f.valueText(a) });
660 } else if (f.isNaN(a)) try line(out, " payload={d}", .{parts.fraction % (f.hidden / 2)}) else try out.writeByte('\n');
661 },
662 .COMPARE => {
663 const a = try program.operand(f, words[2]);
664 const b = try program.operand(f, words[3]);
665 const total_order = compareSigned(orderKey(f, a), orderKey(f, b));
666 const numeric = if (f.isNaN(a) or f.isNaN(b)) "UN" else if (f.isZero(a) and f.isZero(b)) "EQ" else @tagName(total_order);
667 const flags: u8 = if (f.isSignaling(a) or f.isSignaling(b)) Flags.invalid else 0;
668 try line(out, "COMPARE {f} {f} {f} {s} total={s} {f}", .{ f.name, f.hex(a), f.hex(b), numeric, @tagName(total_order), Flags{ .bits = flags } });
669 },
670 .TOINT => {
671 const a = try program.operand(f, words[2]);
672 const kind = std.meta.stringToEnum(IntegerType, words[3]) orelse return program.fail("BAD TYPE {s}", .{words[3]});
673 const mode = try program.parseMode(words, 4);
674 const range: struct { minimum: i64, maximum: i64 } = switch (kind) {
675 .I8 => .{ .minimum = -128, .maximum = 127 },
676 .U8 => .{ .minimum = 0, .maximum = 255 },
677 .I16 => .{ .minimum = -32768, .maximum = 32767 },
678 .U16 => .{ .minimum = 0, .maximum = 65535 },
679 .I32 => .{ .minimum = -2147483648, .maximum = 2147483647 },
680 .U32 => .{ .minimum = 0, .maximum = 4294967295 },
681 };
682 try out.print("TOINT {f} {s} {s} ", .{ f.name, @tagName(kind), @tagName(mode) });
683 if (!f.isFinite(a)) try line(out, "invalid flags=invalid", .{}) else {
684 const units = roundUnits(f.exact(a), 0, mode, false);
685 const whole: i64 = @intCast(units.whole);
686 const integer = if (f.fields(a).negative) -whole else whole;
687 if (integer < range.minimum or integer > range.maximum) try line(out, "invalid flags=invalid", .{}) else try line(out, "{d} {f}", .{ integer, Flags{ .bits = if (units.inexact) Flags.inexact else 0 } });
688 }
689 },
690 .NEXTUP, .NEXTDOWN, .ULP => {
691 const a = try program.operand(f, words[2]);
692 const result: Result = if (f.isNaN(a)) .{ .pattern = f.nan(), .flags = if (f.isSignaling(a)) Flags.invalid else 0 } else .{ .pattern = switch (command) {
693 .NEXTUP => nextUp(f, a),
694 .NEXTDOWN => f.flip(nextUp(f, f.flip(a))),
695 .ULP => if (f.isInfinity(a)) f.infinity else f.round(.{ .n = 1, .exponent = f.fields(a).q }, .RNE).pattern,
696 else => unreachable,
697 } };
698 try line(out, "{s} {f} {f}", .{ @tagName(command), f.name, f.resultText(result) });
699 },
700 .FORMAT => unreachable,
701 }
702 }
703};

705fn readWords(gpa: Allocator, input: *std.Io.Reader) ![]const []const u8 {
706 var buffer: Writer.Allocating = .init(gpa);
707 defer buffer.deinit();
708 _ = try input.streamDelimiterEnding(&buffer.writer, '\n');
709 if (input.seek < input.end) input.toss(1) else if (buffer.written().len == 0) return error.EndOfStream;
710 const text = try buffer.toOwnedSlice();
711 var words: std.ArrayList([]const u8) = .empty;
712 var parts = std.mem.tokenizeAny(u8, text, " \r");
713 while (parts.next()) |word| try words.append(gpa, word);
714 return try words.toOwnedSlice(gpa);
715}

717pub fn main(init: std.process.Init) !void {
718 var scratch = std.heap.ArenaAllocator.init(std.heap.page_allocator);
719 defer scratch.deinit();
720 var input_buffer: [4096]u8 = undefined;
721 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
722 var output_buffer: [4096]u8 = undefined;
723 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
724 var program: Program = .init(scratch.allocator(), &output.interface);
725 const count = try std.fmt.parseInt(usize, (try readWords(scratch.allocator(), &input.interface))[0], 10);
726 for (0..count) |_| {
727 _ = scratch.reset(.retain_capacity);
728 const words = try readWords(scratch.allocator(), &input.interface);
729 var response: Writer.Allocating = .init(scratch.allocator());
730 defer response.deinit();
731 program.out = &response.writer;
732 program.execute(words) catch |err| {
733 if (err != error.BadCommand) return err;
734 try output.interface.print("ERROR {s}\n", .{program.error_message});
735 continue;
736 };
737 try output.interface.writeAll(response.written());
738 }
739 try output.interface.flush();
740}
