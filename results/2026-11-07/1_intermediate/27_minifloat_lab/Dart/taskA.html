17Never fail(String message) => throw CommandError(message);
18String modeText(Mode mode) => mode.name.toUpperCase();
19String binary(int value, int width) => value.toRadixString(2).padLeft(width, '0');
20String relation(int a, int b) => a < b ? 'LT' : a > b ? 'GT' : 'EQ';
21String flagsText(int flags) {
22 final names = [for (final flag in Flag.values) if (flags & flag.bit != 0) flag.name];
23 return 'flags=${names.isEmpty ? 'none' : names.join(',')}';
24}

26class Result {

28 const Result(this.pattern, [this.flags = 0]);

31typedef Units = ({int integer, bool inexact});
32typedef Fields = ({bool negative, int exponent, int fraction, int q});
33
34(int, int) scaled(int numerator, int denominator, int exponent) =>
35 exponent >= 0 ? (numerator << exponent, denominator) : (numerator, denominator << -exponent);
36
37int integerSqrt(int value) {
38 if (value == 0) return 0;
39 var root = 1 << ((value.bitLength + 1) ~/ 2);
40 while (true) {
41 final next = (root + value ~/ root) ~/ 2;
42 if (next >= root) return root;
43 root = next;
44 }
45}
46
47bool increment(Mode mode, bool negative, int integer, bool inexact, int halfway) {
48 if (!inexact) return false;
49 return switch (mode) {
50 Mode.rne => halfway > 0 || halfway == 0 && integer.isOdd,
51 Mode.rna => halfway >= 0,
52 Mode.rtz => false,
53 Mode.rup => !negative,
54 Mode.rdn => negative,
55 };
56}
57
58Units roundUnits(Exact value, int exponent, Mode mode) {
59 final (n, d) = scaled(value.numerator, value.denominator, value.exponent - exponent);
60 var integer = n ~/ d;
61 final remainder = n % d;
62 if (increment(mode, value.negative, integer, remainder != 0, 2 * remainder - d)) integer++;
63 return (integer: integer, inexact: remainder != 0);
64}
65
66Units roundSqrtUnits(Exact value, int exponent, Mode mode) {
67 final (n, d) = scaled(value.numerator, value.denominator, value.exponent - 2 * exponent);
68 var integer = integerSqrt(n ~/ d);
69 final inexact = integer * integer * d != n;
70 final midpoint = 2 * integer + 1;
71 if (increment(mode, false, integer, inexact, 4 * n - d * midpoint * midpoint)) integer++;
72 return (integer: integer, inexact: inexact);
73}

75class Exact {
76 final int numerator, exponent, denominator;
77 final bool negative;
78 const Exact(this.numerator, {this.exponent = 0, this.denominator = 1, this.negative = false});
79
80 Exact negated() => Exact(numerator, exponent: exponent, denominator: denominator, negative: !negative);
81
82 int log2Floor() {
83 final t = numerator.bitLength - denominator.bitLength + exponent;
84 final (n, d) = scaled(numerator, denominator, exponent - t);
85 return n < d ? t - 1 : t;
86 }
87
88 String fraction() {
89 if (numerator == 0) return '0';
90 var (n, d) = scaled(numerator, denominator, exponent);
91 final divisor = n.gcd(d);
92 n ~/= divisor;
93 d ~/= divisor;
94 return '${negative ? '-' : ''}$n${d == 1 ? '' : '/$d'}';
95 }
96
97 String decimal({bool keepZeroSign = false}) {
98 final (n, d) = scaled(numerator, denominator, exponent);
99 final text = StringBuffer();
100 if (negative && (numerator != 0 || keepZeroSign)) text.write('-');
101 text.write(n ~/ d);
102 var remainder = n % d;
103 if (remainder != 0) {
104 text.write('.');
105 while (remainder != 0) {
106 remainder *= 10;
107 text.write(String.fromCharCode(48 + remainder ~/ d));
108 remainder %= d;
109 }
110 }
111 return text.toString();
112 }
113
114 static Exact product(Exact a, Exact b) => Exact(
115 a.numerator * b.numerator,
116 exponent: a.exponent + b.exponent,
117 denominator: a.denominator * b.denominator,
118 negative: a.negative != b.negative,
119 );
120
121 static Exact sum(List<Exact> terms, Mode mode) {
122 int? exponent;
123 for (final term in terms) {
124 if (term.numerator != 0 && (exponent == null || term.exponent < exponent)) exponent = term.exponent;
125 }
126 if (exponent == null) {
127 final same = terms.every((term) => term.negative == terms.first.negative);
128 return Exact(0, negative: same ? terms.first.negative : mode == Mode.rdn);
129 }
130 var numerator = 0;
131 for (final term in terms) {
132 if (term.numerator == 0) continue;
133 final aligned = term.numerator << (term.exponent - exponent);
134 numerator += term.negative ? -aligned : aligned;
135 }
136 return Exact(numerator.abs(), exponent: exponent,
137 negative: numerator < 0 || numerator == 0 && mode == Mode.rdn);
138 }
139}

141class MiniFormat {
142 final String name;
143 final int e, m;
144 const MiniFormat(this.name, this.e, this.m);
145 int get width => 1 + e + m;
146 int get bias => (1 << (e - 1)) - 1;
147 int get minExponent => 1 - bias;
148 int get maxExponent => bias;
149 int get hidden => 1 << m;
150 int get exponentMask => (1 << e) - 1;
151 int get signBit => 1 << (e + m);
152 int get infinity => exponentMask * hidden;
153 int get nan => infinity + hidden ~/ 2;
154 int get maximum => infinity - 1;
155
156 Fields fields(int pattern) {
157 final exponent = (pattern >> m) & exponentMask;
158 return (negative: pattern & signBit != 0, exponent: exponent, fraction: pattern & (hidden - 1),
159 q: (exponent == 0 ? minExponent : exponent - bias) - m);
160 }
161 bool isNaN(int pattern) => fields(pattern).exponent == exponentMask && fields(pattern).fraction != 0;
162 bool isSignaling(int pattern) => isNaN(pattern) && fields(pattern).fraction < hidden ~/ 2;
163 bool isInfinity(int pattern) => pattern & (signBit - 1) == infinity;
164 bool isZero(int pattern) => pattern & (signBit - 1) == 0;
165 bool isFinite(int pattern) => fields(pattern).exponent != exponentMask;
166 int zero(bool negative) => negative ? signBit : 0;
167 int inf(bool negative) => zero(negative) + infinity;
168 int flip(int pattern) => pattern ^ signBit;
169
170 Exact exact(int pattern) {
171 final f = fields(pattern);
172 return Exact(f.exponent == 0 ? f.fraction : hidden + f.fraction, exponent: f.q, negative: f.negative);
173 }
174 String bits(int pattern) {
175 final f = fields(pattern);
176 return '${f.negative ? 1 : 0}|${binary(f.exponent, e)}|${binary(f.fraction, m)}';
177 }
178 String hex(int pattern) => '0x${pattern.toRadixString(16).toUpperCase().padLeft((width + 3) ~/ 4, '0')}';
179 String value(int pattern) => isNaN(pattern) ? 'nan' : isInfinity(pattern)
180 ? (fields(pattern).negative ? '-inf' : 'inf') : exact(pattern).decimal(keepZeroSign: true);
181 String patternText(int pattern) => '${bits(pattern)} ${hex(pattern)} ${value(pattern)}';
182 String resultText(Result result) => '${patternText(result.pattern)} ${flagsText(result.flags)}';
183
184 Result overflow(bool negative, Mode mode) {
185 final toInfinity = mode == Mode.rne || mode == Mode.rna || mode == Mode.rup && !negative || mode == Mode.rdn && negative;
186 return Result(zero(negative) + (toInfinity ? infinity : maximum), Flag.overflow.bit | Flag.inexact.bit);
187 }
188
189 Result roundNonzero(bool negative, int t, Mode mode, Units Function(int) unitsAt) {
190 if (t > maxExponent) return overflow(negative, mode);
191 var exponent = (t > minExponent ? t : minExponent) - m;
192 final units = unitsAt(exponent);
193 var n = units.integer;
194 if (n == 2 * hidden) {
195 n = hidden;
196 exponent++;
197 }
198 if (exponent > maxExponent - m) return overflow(negative, mode);
199 final magnitude = n >= hidden ? (exponent + m + bias) * hidden + n - hidden : n;
200 var flags = units.inexact ? Flag.inexact.bit : 0;
201 if (units.inexact && t < minExponent &&
202 (t < minExponent - 1 || unitsAt(t - m).integer < 2 * hidden)) flags |= Flag.underflow.bit;
203 return Result(zero(negative) + magnitude, flags);
204 }
205
206 Result round(Exact value, Mode mode) => value.numerator == 0 ? Result(zero(value.negative))
207 : roundNonzero(value.negative, value.log2Floor(), mode, (q) => roundUnits(value, q, mode));
208
209 Result sqrt(Exact value, Mode mode) {
210 final log = value.log2Floor();
211 final t = log >= 0 ? log ~/ 2 : -((-log + 1) ~/ 2);
212 return roundNonzero(false, t, mode, (q) => roundSqrtUnits(value, q, mode));
213 }
214}

216class RealInput {
217 final RealKind kind;
218 final Exact number;
219 const RealInput(this.kind, this.number);
220 String get text => kind == RealKind.finite ? number.fraction() : kind == RealKind.nan
221 ? 'nan' : number.negative ? '-inf' : 'inf';
222}
223
224Mode modeAt(List<String> words, int index) => index == words.length ? Mode.rne : switch (words[index]) {
225 'RNE' => Mode.rne, 'RNA' => Mode.rna, 'RTZ' => Mode.rtz, 'RUP' => Mode.rup, 'RDN' => Mode.rdn,
226 _ => fail('BAD MODE ${words[index]}'),
227};
228
229RealInput realInput(String word) {
230 if (word == 'nan') return const RealInput(RealKind.nan, Exact(0));
231 if (word == 'inf' || word == '-inf') return RealInput(RealKind.infinity, Exact(0, negative: word == '-inf'));
232 if (word.startsWith('0b') || word.startsWith('0x')) fail('BAD OPERAND $word');
233 final negative = word.startsWith('-');
234 var body = negative ? word.substring(1) : word;
235 final slash = body.indexOf('/');
236 if (slash >= 0) {
237 final n = int.parse(body.substring(0, slash));
238 final d = int.parse(body.substring(slash + 1));
239 if (d == 0) fail('BAD OPERAND $word');
240 return RealInput(RealKind.finite, Exact(n, denominator: d, negative: negative));
241 }
242 final dot = body.indexOf('.');
243 var denominator = 1;
244 if (dot >= 0) {
245 for (var i = dot + 1; i < body.length; i++) denominator *= 10;
246 body = body.replaceRange(dot, dot + 1, '');
247 }
248 return RealInput(RealKind.finite, Exact(int.parse(body), denominator: denominator, negative: negative));
249}

251Result encoded(MiniFormat f, RealInput input, Mode mode) => switch (input.kind) {
252 RealKind.nan => Result(f.nan),
253 RealKind.infinity => Result(f.inf(input.number.negative)),
254 RealKind.finite => f.round(input.number, mode),
255};
256
257int operand(MiniFormat f, String word) {
258 final bits = word.startsWith('0b'), hex = word.startsWith('0x');
259 if (bits || hex) {
260 if (word.length - 2 != (bits ? f.width : (f.width + 3) ~/ 4)) fail('BAD OPERAND $word');
261 final pattern = int.parse(word.substring(2), radix: bits ? 2 : 16);
262 if (pattern >= 1 << f.width) fail('BAD OPERAND $word');
263 return pattern;
264 }
265 return encoded(f, realInput(word), Mode.rne).pattern;
266}

268Result calculate(MiniFormat f, Operation operation, Mode mode, List<int> inputs) {
269 if (inputs.any(f.isNaN)) return Result(f.nan, inputs.any(f.isSignaling) ? Flag.invalid.bit : 0);
270 final a = inputs[0], negative = f.fields(inputs[0]).negative;
271 final invalid = Result(f.nan, Flag.invalid.bit);
272 if (operation == Operation.sqrt) {
273 if (f.isZero(a)) return Result(a);
274 if (negative) return invalid;
275 if (f.isInfinity(a)) return Result(a);
276 return f.sqrt(f.exact(a), mode);
277 }
278 final b = operation == Operation.sub ? f.flip(inputs[1]) : inputs[1];
279 final sigma = negative != f.fields(b).negative;
280 if (operation == Operation.add || operation == Operation.sub) {
281 if (f.isInfinity(a) && f.isInfinity(b) && sigma) return invalid;
282 if (f.isInfinity(a)) return Result(a);
283 if (f.isInfinity(b)) return Result(b);
284 return f.round(Exact.sum([f.exact(a), f.exact(b)], mode), mode);
285 }
286 if (operation == Operation.mul) {
287 if (f.isInfinity(a) || f.isInfinity(b)) return f.isZero(a) || f.isZero(b) ? invalid : Result(f.inf(sigma));
288 if (f.isZero(a) || f.isZero(b)) return Result(f.zero(sigma));
289 return f.round(Exact.product(f.exact(a), f.exact(b)), mode);
290 }
291 if (operation == Operation.div) {
292 if (f.isInfinity(a)) return f.isInfinity(b) ? invalid : Result(f.inf(sigma));
293 if (f.isInfinity(b)) return Result(f.zero(sigma));
294 if (f.isZero(b)) return f.isZero(a) ? invalid : Result(f.inf(sigma), Flag.divbyzero.bit);
295 if (f.isZero(a)) return Result(f.zero(sigma));
296 final x = f.exact(a), y = f.exact(b);
297 return f.round(Exact(x.numerator * y.denominator, exponent: x.exponent - y.exponent,
298 denominator: x.denominator * y.numerator, negative: sigma), mode);
299 }
300 final c = inputs[2];
301 if (f.isInfinity(a) || f.isInfinity(b)) {
302 if (f.isZero(a) || f.isZero(b) || f.isInfinity(c) && f.fields(c).negative != sigma) return invalid;
303 return Result(f.inf(sigma));
304 }
305 if (f.isInfinity(c)) return Result(c);
306 return f.round(Exact.sum([Exact.product(f.exact(a), f.exact(b)), f.exact(c)], mode), mode);
307}

309int nextUp(MiniFormat f, int pattern) => pattern == f.infinity ? pattern
310 : f.isZero(pattern) ? 1 : pattern + (f.fields(pattern).negative ? -1 : 1);
311String difference(MiniFormat f, int pattern, Exact exact) => !f.isFinite(pattern) ? f.value(pattern)
312 : Exact.sum([f.exact(pattern), exact.negated()], Mode.rne).fraction();

314class Laboratory {
315 final formats = <String, MiniFormat>{
316 'HALF': const MiniFormat('HALF', 5, 10), 'E5M2': const MiniFormat('E5M2', 5, 2), 'E4M3': const MiniFormat('E4M3', 4, 3),
317 };
318 static const integerTypes = <String, (int, int)>{
319 'I8': (-128, 127), 'U8': (0, 255), 'I16': (-32768, 32767), 'U16': (0, 65535),
320 'I32': (-2147483648, 2147483647), 'U32': (0, 4294967295),
321 };
322 static const operations = {
323 'ADD': Operation.add, 'SUB': Operation.sub, 'MUL': Operation.mul,
324 'DIV': Operation.div, 'SQRT': Operation.sqrt, 'FMA': Operation.fma,
325 };
326 final List<String> output = [];
327 void emit(String line) => output.add(line);
328 MiniFormat format(String name) => formats[name] ?? fail('UNKNOWN FORMAT $name');
329
330 List<int> operands(MiniFormat f, List<String> words, int begin, int end) =>
331 [for (var i = begin; i < end; i++) operand(f, words[i])];
332
333 void summary(MiniFormat f, Mode mode, List<int> patterns, Result result) {
334 final exact = Exact.sum(patterns.map(f.exact).toList(), mode);
335 emit('RESULT ${f.resultText(result)}');
336 emit('EXACT ${exact.decimal()}');
337 emit('ROUNDED ${f.resultText(f.round(exact, mode))}');
338 emit('DIFF ${difference(f, result.pattern, exact)}');
339 }
340
341 void summation(String command, MiniFormat f, Mode mode, List<int> patterns) {
342 if (!patterns.every(f.isFinite)) fail('NOT FINITE');
343 emit('$command ${f.name} ${modeText(mode)} count=${patterns.length}');
344 var sum = patterns.first, compensation = 0, flags = 0;
345 for (var i = 1; i < patterns.length; i++) {
346 if (command == 'SUM') {
347 final added = calculate(f, Operation.add, mode, [sum, patterns[i]]);
348 sum = added.pattern;
349 flags |= added.flags;
350 emit('PARTIAL ${i + 1} ${f.resultText(added)}');
351 } else {
352 final y = calculate(f, Operation.sub, mode, [patterns[i], compensation]);
353 final t = calculate(f, Operation.add, mode, [sum, y.pattern]);
354 final d = calculate(f, Operation.sub, mode, [t.pattern, sum]);
355 final c = calculate(f, Operation.sub, mode, [d.pattern, y.pattern]);
356 sum = t.pattern;
357 compensation = c.pattern;
358 final step = y.flags | t.flags | d.flags | c.flags;
359 flags |= step;
360 emit('KSTEP ${i + 1} sum=${f.value(sum)} comp=${f.value(compensation)} ${flagsText(step)}');
361 }
362 }
363 summary(f, mode, patterns, Result(sum, flags));
364 }

366 void dot(MiniFormat f, Mode mode, List<int> a, List<int> b) {
367 if (a.length != b.length) fail('LENGTH MISMATCH');
368 if (!a.every(f.isFinite) || !b.every(f.isFinite)) fail('NOT FINITE');
369 var naive = calculate(f, Operation.mul, mode, [a[0], b[0]]);
370 var chained = naive;
371 for (var i = 1; i < a.length; i++) {
372 final product = calculate(f, Operation.mul, mode, [a[i], b[i]]);
373 final added = calculate(f, Operation.add, mode, [naive.pattern, product.pattern]);
374 naive = Result(added.pattern, naive.flags | product.flags | added.flags);
375 final fused = calculate(f, Operation.fma, mode, [a[i], b[i], chained.pattern]);
376 chained = Result(fused.pattern, chained.flags | fused.flags);
377 }
378 final exact = Exact.sum([for (var i = 0; i < a.length; i++) Exact.product(f.exact(a[i]), f.exact(b[i]))], mode);
379 emit('DOT ${f.name} ${modeText(mode)} n=${a.length}');
380 emit('NAIVE ${f.resultText(naive)}');
381 emit('CHAINED ${f.resultText(chained)}');
382 emit('EXACT ${exact.decimal()}');
383 emit('FUSED ${f.resultText(f.round(exact, mode))}');
384 emit('DIFF naive=${difference(f, naive.pattern, exact)} chained=${difference(f, chained.pattern, exact)}');
385 }

387 void table(MiniFormat f) {
388 if (f.width > 8) fail('TABLE TOO LARGE');
389 final count = f.infinity;
390 emit('TABLE ${f.name} count=$count');
391 for (var pattern = 0; pattern < count; pattern++) {
392 final gap = pattern == count - 1 ? '-'
393 : Exact.sum([f.exact(pattern + 1), f.exact(pattern).negated()], Mode.rne).decimal();
394 emit('ENTRY $pattern ${f.patternText(pattern)} gap=$gap');
395 }
396 emit('MAX ${f.patternText(f.maximum)}');
397 emit('MINNORMAL ${f.patternText(f.hidden)}');
398 emit('MINSUBNORMAL ${f.patternText(1)}');
399 emit('EPSILON ${Exact(1, exponent: -f.m).decimal()}');
400 emit('PATTERNS finite=${2 * count} infinite=2 nan=${2 * (f.hidden - 1)} total=${1 << f.width}');
401 }

403 List<String> execute(String line) {
404 output.clear();
405 final words = line.split(' '), command = words[0];
406 if (command == 'FORMAT') {
407 final name = words[1], e = int.parse(words[2]), m = int.parse(words[3]);
408 if (formats.containsKey(name)) fail('FORMAT EXISTS $name');
409 if (e < 2 || e > 5 || m < 1 || m > 10) fail('BAD FORMAT');
410 final f = MiniFormat(name, e, m);
411 formats[name] = f;
412 emit('FORMAT $name e=$e m=$m bits=${f.width} bias=${f.bias} emin=${f.minExponent} emax=${f.maxExponent}');
413 return output;
414 }
415 final f = format(words[1]);
416 final operation = operations[command];
417 if (command == 'ENCODE') {
418 final input = realInput(words[2]);
419 final mode = modeAt(words, 3);
420 emit('ENCODE ${f.name} ${modeText(mode)} ${input.text} ${f.resultText(encoded(f, input, mode))}');
421 } else if (command == 'CONVERT') {
422 final g = format(words[2]);
423 final a = operand(f, words[3]);
424 final mode = modeAt(words, 4);
425 final result = f.isNaN(a) ? Result(g.nan, f.isSignaling(a) ? Flag.invalid.bit : 0)
426 : f.isInfinity(a) ? Result(g.inf(f.fields(a).negative)) : g.round(f.exact(a), mode);
427 emit('CONVERT ${f.name} ${g.name} ${modeText(mode)} ${g.resultText(result)}');
428 } else if (command == 'SUM' || command == 'KAHAN' || command == 'DOT') {
429 final mode = modeAt(words, 2);
430 if (command == 'DOT') {
431 final separator = words.indexOf('|');
432 final a = operands(f, words, 3, separator);
433 final b = operands(f, words, separator + 1, words.length);
434 dot(f, mode, a, b);
435 } else {
436 summation(command, f, mode, operands(f, words, 3, words.length));
437 }
438 } else if (command == 'TABLE') {
439 table(f);
440 } else if (operation != null) {
441 final count = operation == Operation.sqrt ? 1 : operation == Operation.fma ? 3 : 2;
442 final patterns = operands(f, words, 2, 2 + count);
443 final mode = modeAt(words, 2 + count);
444 emit('$command ${f.name} ${modeText(mode)} ${f.resultText(calculate(f, operation, mode, patterns))}');
445 } else {
446 final a = operand(f, words[2]);
447 final fields = f.fields(a);
448 if (command == 'DECODE') {
449 final classification = f.isNaN(a) ? (f.isSignaling(a) ? 'snan' : 'qnan') : f.isInfinity(a) ? 'infinity'
450 : f.isZero(a) ? 'zero' : fields.exponent == 0 ? 'subnormal' : 'normal';
451 emit('DECODE ${f.name} ${f.bits(a)} ${f.hex(a)} $classification sign=${fields.negative ? 1 : 0}');
452 final heading = 'FIELDS exponent=${fields.exponent} fraction=${fields.fraction}';
453 if (f.isFinite(a)) {
454 emit('$heading unbiased=${fields.q + f.m} significand=${fields.exponent == 0 ? 0 : 1}.${binary(fields.fraction, f.m)}');
455 emit('VALUE ${f.exact(a).fraction()} ${f.value(a)}');
456 } else {
457 emit(f.isNaN(a) ? '$heading payload=${fields.fraction % (f.hidden ~/ 2)}' : heading);
458 }
459 } else if (command == 'COMPARE') {
460 final b = operand(f, words[3]);
461 int key(int pattern) => f.fields(pattern).negative ? -(pattern - f.signBit) - 1 : pattern;
462 final order = relation(key(a), key(b));
463 final numeric = f.isNaN(a) || f.isNaN(b) ? 'UN' : f.isZero(a) && f.isZero(b) ? 'EQ' : order;
464 final flags = f.isSignaling(a) || f.isSignaling(b) ? Flag.invalid.bit : 0;
465 emit('COMPARE ${f.name} ${f.hex(a)} ${f.hex(b)} $numeric total=$order ${flagsText(flags)}');
466 } else if (command == 'TOINT') {
467 final range = integerTypes[words[3]] ?? fail('BAD TYPE ${words[3]}');
468 final mode = modeAt(words, 4);
469 final prefix = 'TOINT ${f.name} ${words[3]} ${modeText(mode)}';
470 if (!f.isFinite(a)) {
471 emit('$prefix invalid flags=invalid');
472 } else {
473 final units = roundUnits(f.exact(a), 0, mode);
474 final integer = fields.negative ? -units.integer : units.integer;
475 emit(integer < range.$1 || integer > range.$2 ? '$prefix invalid flags=invalid'
476 : '$prefix $integer ${flagsText(units.inexact ? Flag.inexact.bit : 0)}');
477 }
478 } else {
479 final Result result;
480 if (f.isNaN(a)) {
481 result = Result(f.nan, f.isSignaling(a) ? Flag.invalid.bit : 0);
482 } else if (command == 'NEXTUP') {
483 result = Result(nextUp(f, a));
484 } else if (command == 'NEXTDOWN') {
485 result = Result(f.flip(nextUp(f, f.flip(a))));
486 } else {
487 result = Result(f.isInfinity(a) ? f.infinity : f.round(Exact(1, exponent: fields.q), Mode.rne).pattern);
488 }
489 emit('$command ${f.name} ${f.resultText(result)}');
490 }
491 }
492 return output;
493 }
494}
