37def decimal_text(numerator, exponent):
38 sign = "-" if numerator < 0 else ""
39 numerator = abs(numerator)
40 if exponent >= 0:
41 return sign + str(numerator << exponent)
42 denominator = 1 << -exponent
43 whole, remainder = divmod(numerator, denominator)
44 digits = []
45 while remainder != 0:
46 digit, remainder = divmod(remainder * 10, denominator)
47 digits.append(str(digit))
48 text = sign + str(whole)
49 if digits:
50 text += "." + "".join(digits)
51 return text

62def remainder_class(remainder, divisor):
63 if remainder == 0:
64 return NO_REMAINDER
65 if 2 * remainder < divisor:
66 return BELOW_HALF
67 if 2 * remainder == divisor:
68 return AT_HALF
69 return ABOVE_HALF

72def rounds_up(mode, sign, value, remainder):
73 if mode == "RNE":
74 return remainder == ABOVE_HALF or (remainder == AT_HALF and value % 2 == 1)
75 if mode == "RNA":
76 return remainder >= AT_HALF
77 if mode == "RUP":
78 return remainder != NO_REMAINDER and sign == 0
79 if mode == "RDN":
80 return remainder != NO_REMAINDER and sign == 1
81 return False

84class Ratio:
85 def __init__(self, numerator, exponent, denominator):
86 common = gcd(numerator, denominator)
87 numerator //= common
88 denominator //= common
89 while numerator != 0 and numerator % 2 == 0:
90 numerator //= 2
91 exponent += 1
92 while denominator % 2 == 0:
93 denominator //= 2
94 exponent -= 1
95 self.numerator = numerator
96 self.exponent = exponent
97 self.denominator = denominator
98
99 def floor_log2(self):
100 shift = self.numerator.bit_length() - self.denominator.bit_length()
101 if shift >= 0:
102 below = self.numerator < self.denominator << shift
103 else:
104 below = self.numerator << -shift < self.denominator
105 return self.exponent + shift - (1 if below else 0)
106
107 def split(self, exponent):
108 numerator = self.numerator
109 denominator = self.denominator
110 shift = self.exponent - exponent
111 if shift >= 0:
112 numerator <<= shift
113 else:
114 denominator <<= -shift
115 whole, remainder = divmod(numerator, denominator)
116 return whole, remainder_class(remainder, denominator)
117
118 def fraction_text(self, negative):
119 if self.numerator == 0:
120 return "0"
121 numerator = self.numerator
122 denominator = self.denominator
123 if self.exponent >= 0:
124 numerator <<= self.exponent
125 else:
126 denominator <<= -self.exponent
127 text = ("-" if negative else "") + str(numerator)
128 if denominator != 1:
129 text += "/" + str(denominator)
130 return text

133class Root:
134 def __init__(self, significand, exponent):
135 if exponent % 2 != 0:
136 significand *= 2
137 exponent -= 1
138 self.radicand = significand
139 self.half_exponent = exponent // 2
140
141 def floor_log2(self):
142 return (self.radicand.bit_length() - 1) // 2 + self.half_exponent
143
144 def split(self, exponent):
145 radicand = self.radicand << (2 * (self.half_exponent - exponent))
146 whole = isqrt(radicand)
147 if whole * whole == radicand:
148 return whole, NO_REMAINDER
149 if 4 * radicand < (2 * whole + 1) ** 2:
150 return whole, BELOW_HALF
151 return whole, ABOVE_HALF

154def parse_number(text):
155 negative = text.startswith("-")
156 body = text[1:] if negative else text
157 if "/" in body:
158 numerator_text, denominator_text = body.split("/")
159 denominator = int(denominator_text)
160 if denominator == 0:
161 raise CommandError(f"BAD OPERAND {text}")
162 return negative, Ratio(int(numerator_text), 0, denominator)
163 whole, _, fraction = body.partition(".")
164 return negative, Ratio(int(whole + fraction), 0, 10 ** len(fraction))

167class Format:
168 def __init__(self, name, exponent_bits, fraction_bits):
169 self.name = name
170 self.exponent_bits = exponent_bits
171 self.fraction_bits = fraction_bits
172 self.width = 1 + exponent_bits + fraction_bits
173 self.bias = 2 ** (exponent_bits - 1) - 1
174 self.emin = 1 - self.bias
175 self.emax = self.bias
176 self.top_field = 2 ** exponent_bits - 1
177 self.hidden_bit = 1 << fraction_bits
178 self.sign_bit = 1 << (exponent_bits + fraction_bits)
179 self.hex_digits = (self.width + 3) // 4
180 self.nan = self.compose(0, self.top_field, self.hidden_bit // 2)
181 self.largest = self.compose(0, self.top_field - 1, self.hidden_bit - 1)
182
183 def compose(self, sign, field, fraction):
184 return sign * self.sign_bit + (field << self.fraction_bits) + fraction
185
186 def sign(self, pattern):
187 return pattern // self.sign_bit
188
189 def field(self, pattern):
190 return (pattern >> self.fraction_bits) & self.top_field
191
192 def fraction(self, pattern):
193 return pattern & (self.hidden_bit - 1)
194
195 def is_nan(self, pattern):
196 return self.field(pattern) == self.top_field and self.fraction(pattern) != 0
197
198 def is_signaling(self, pattern):
199 return self.is_nan(pattern) and self.fraction(pattern) < self.hidden_bit // 2
200
201 def is_infinity(self, pattern):
202 return self.field(pattern) == self.top_field and self.fraction(pattern) == 0
203
204 def is_finite(self, pattern):
205 return self.field(pattern) != self.top_field
206
207 def is_zero(self, pattern):
208 return pattern % self.sign_bit == 0
209
210 def flip(self, pattern):
211 return pattern ^ self.sign_bit
212
213 def zero(self, sign):
214 return self.compose(sign, 0, 0)
215
216 def infinity(self, sign):
217 return self.compose(sign, self.top_field, 0)
218
219 def special(self, text):
220 if text == "nan":
221 return self.nan
222 return self.infinity(1 if text == "-inf" else 0)
223
224 def term(self, pattern):
225 field = self.field(pattern)
226 fraction = self.fraction(pattern)
227 if field == 0:
228 return self.sign(pattern), fraction, self.emin - self.fraction_bits
229 return self.sign(pattern), self.hidden_bit + fraction, field - self.bias - self.fraction_bits
230
231 def power_of_two(self, exponent):
232 if exponent >= self.emin:
233 return self.compose(0, exponent + self.bias, 0)
234 return 1 << (exponent - self.emin + self.fraction_bits)
235
236 def order_key(self, pattern):
237 if self.sign(pattern) == 0:
238 return pattern
239 return -(pattern - self.sign_bit) - 1
240
241 def classify(self, pattern):
242 field = self.field(pattern)
243 fraction = self.fraction(pattern)
244 if field == self.top_field:
245 if fraction == 0:
246 return "infinity"
247 return "qnan" if fraction >= self.hidden_bit // 2 else "snan"
248 if field == 0:
249 return "zero" if fraction == 0 else "subnormal"
250 return "normal"

252 def bit_text(self, pattern):
253 field = f"{self.field(pattern):0{self.exponent_bits}b}"
254 fraction = f"{self.fraction(pattern):0{self.fraction_bits}b}"
255 return f"{self.sign(pattern)}|{field}|{fraction}"
256
257 def hex_text(self, pattern):
258 return f"0x{pattern:0{self.hex_digits}X}"
259
260 def value_text(self, pattern):
261 if self.is_nan(pattern):
262 return "nan"
263 if self.is_infinity(pattern):
264 return "-inf" if self.sign(pattern) == 1 else "inf"
265 sign, significand, exponent = self.term(pattern)
266 return ("-" if sign == 1 else "") + decimal_text(significand, exponent)
267
268 def pattern_text(self, pattern):
269 return f"{self.bit_text(pattern)} {self.hex_text(pattern)} {self.value_text(pattern)}"
270
271 def result_text(self, pattern, flags):
272 return f"{self.pattern_text(pattern)} {flag_text(flags)}"

274 def overflow(self, sign, mode):
275 if mode in ("RNE", "RNA") or (mode == "RUP" and sign == 0) or (mode == "RDN" and sign == 1):
276 return self.infinity(sign), OVERFLOW | INEXACT
277 return self.largest + sign * self.sign_bit, OVERFLOW | INEXACT

279 def tiny_after_rounding(self, sign, magnitude, mode, top):
280 if top < self.emin - 1:
281 return True
282 value, remainder = magnitude.split(top - self.fraction_bits)
283 if rounds_up(mode, sign, value, remainder):
284 value += 1
285 return value < 2 * self.hidden_bit
286
287 def round(self, sign, magnitude, mode):
288 top = magnitude.floor_log2()
289 if top > self.emax:
290 return self.overflow(sign, mode)
291 exponent = max(top, self.emin) - self.fraction_bits
292 value, remainder = magnitude.split(exponent)
293 if rounds_up(mode, sign, value, remainder):
294 value += 1
295 if value == 2 * self.hidden_bit:
296 value = self.hidden_bit
297 exponent += 1
298 if exponent > self.emax - self.fraction_bits:
299 return self.overflow(sign, mode)
300 flags = 0
301 if remainder != NO_REMAINDER:
302 flags |= INEXACT
303 if top < self.emin and self.tiny_after_rounding(sign, magnitude, mode, top):
304 flags |= UNDERFLOW
305 if value >= self.hidden_bit:
306 field = exponent + self.fraction_bits + self.bias
307 return self.compose(sign, field, value - self.hidden_bit), flags
308 return self.compose(sign, 0, value), flags
309
310 def encode(self, negative, ratio, mode):
311 sign = 1 if negative else 0
312 if ratio.numerator == 0:
313 return self.zero(sign), 0
314 return self.round(sign, ratio, mode)

317def exact_sum(terms):
318 exponents = [exponent for _, significand, exponent in terms if significand != 0]
319 lowest = min(exponents) if exponents else 0
320 total = 0
321 for sign, significand, exponent in terms:
322 if significand != 0:
323 part = significand << (exponent - lowest)
324 total += -part if sign == 1 else part
325 return total, lowest

328def round_sum(fmt, terms, mode):
329 total, exponent = exact_sum(terms)
330 if total == 0:
331 first_sign, _, _ = terms[0]
332 if all(significand == 0 and sign == first_sign for sign, significand, _ in terms):
333 return fmt.zero(first_sign), 0
334 return fmt.zero(1 if mode == "RDN" else 0), 0
335 return fmt.round(1 if total < 0 else 0, Ratio(abs(total), exponent, 1), mode)

338def aligned_difference(first, first_exponent, second, second_exponent):
339 lowest = min(first_exponent, second_exponent)
340 difference = (first << (first_exponent - lowest)) - (second << (second_exponent - lowest))
341 return difference, lowest

344def error_text(fmt, pattern, exact):
345 if not fmt.is_finite(pattern):
346 return fmt.value_text(pattern)
347 sign, significand, exponent = fmt.term(pattern)
348 value = -significand if sign == 1 else significand
349 difference, lowest = aligned_difference(value, exponent, *exact)

353def product_term(fmt, a, b):
354 a_sign, a_significand, a_exponent = fmt.term(a)
355 b_sign, b_significand, b_exponent = fmt.term(b)
356 return a_sign ^ b_sign, a_significand * b_significand, a_exponent + b_exponent

359def nan_rule(fmt, operands):
360 if not any(fmt.is_nan(pattern) for pattern in operands):
361 return None
362 signaling = any(fmt.is_signaling(pattern) for pattern in operands)
363 return fmt.nan, INVALID if signaling else 0

366def add(fmt, a, b, mode):
367 result = nan_rule(fmt, (a, b))
368 if result is not None:
369 return result
370 if fmt.is_infinity(a) and fmt.is_infinity(b) and fmt.sign(a) != fmt.sign(b):
371 return fmt.nan, INVALID
372 if fmt.is_infinity(a):
373 return a, 0
374 if fmt.is_infinity(b):
375 return b, 0
376 return round_sum(fmt, [fmt.term(a), fmt.term(b)], mode)

379def subtract(fmt, a, b, mode):

383def multiply(fmt, a, b, mode):
384 result = nan_rule(fmt, (a, b))
385 if result is not None:
386 return result
387 sigma = fmt.sign(a) ^ fmt.sign(b)
388 if fmt.is_infinity(a) or fmt.is_infinity(b):
389 if fmt.is_zero(a) or fmt.is_zero(b):
390 return fmt.nan, INVALID
391 return fmt.infinity(sigma), 0
392 if fmt.is_zero(a) or fmt.is_zero(b):
393 return fmt.zero(sigma), 0
394 sign, significand, exponent = product_term(fmt, a, b)
395 return fmt.round(sign, Ratio(significand, exponent, 1), mode)

398def divide(fmt, a, b, mode):
399 result = nan_rule(fmt, (a, b))
400 if result is not None:
401 return result
402 sigma = fmt.sign(a) ^ fmt.sign(b)
403 if fmt.is_infinity(a):
404 if fmt.is_infinity(b):
405 return fmt.nan, INVALID
406 return fmt.infinity(sigma), 0
407 if fmt.is_infinity(b):
408 return fmt.zero(sigma), 0
409 if fmt.is_zero(b):
410 if fmt.is_zero(a):
411 return fmt.nan, INVALID
412 return fmt.infinity(sigma), DIVBYZERO
413 if fmt.is_zero(a):
414 return fmt.zero(sigma), 0
415 _, dividend, dividend_exponent = fmt.term(a)
416 _, divisor, divisor_exponent = fmt.term(b)
417 quotient = Ratio(dividend, dividend_exponent - divisor_exponent, divisor)
418 return fmt.round(sigma, quotient, mode)

421def square_root(fmt, a, mode):
422 result = nan_rule(fmt, (a,))
423 if result is not None:
424 return result
425 if fmt.is_zero(a):
426 return a, 0
427 if fmt.sign(a) == 1:
428 return fmt.nan, INVALID

435def fused_multiply_add(fmt, a, b, c, mode):
436 result = nan_rule(fmt, (a, b, c))
437 if result is not None:
438 return result
439 sigma = fmt.sign(a) ^ fmt.sign(b)
440 if fmt.is_infinity(a) or fmt.is_infinity(b):
441 if fmt.is_zero(a) or fmt.is_zero(b):
442 return fmt.nan, INVALID
443 if fmt.is_infinity(c) and fmt.sign(c) != sigma:
444 return fmt.nan, INVALID
445 return fmt.infinity(sigma), 0
446 if fmt.is_infinity(c):
447 return c, 0
448 return round_sum(fmt, [product_term(fmt, a, b), fmt.term(c)], mode)

509def operand(fmt, text):
510 if text.startswith("0b"):
511 digits = text[2:]
512 if len(digits) != fmt.width:
513 raise CommandError(f"BAD OPERAND {text}")
514 return int(digits, 2)
515 if text.startswith("0x"):
516 digits = text[2:]
517 if len(digits) != fmt.hex_digits or int(digits, 16) >= 1 << fmt.width:
518 raise CommandError(f"BAD OPERAND {text}")
519 return int(digits, 16)
520 if text in SPECIAL_VALUES:
521 return fmt.special(text)
522 negative, ratio = parse_number(text)
523 pattern, _ = fmt.encode(negative, ratio, "RNE")
524 return pattern

544class MinifloatLab:
545 def __init__(self):
546 self.formats = {
547 name: Format(name, exponent_bits, fraction_bits)
548 for name, exponent_bits, fraction_bits in BUILTIN_FORMATS

550 self.output = []
551 self.handlers = {
552 "FORMAT": self.define_format,
553 "ENCODE": self.encode,

556 "COMPARE": self.compare,
557 "TOINT": self.to_integer,

564 def emit(self, *parts):
565 self.output.append(" ".join(str(part) for part in parts))

567 def lookup(self, name):
568 if name not in self.formats:
569 raise CommandError(f"UNKNOWN FORMAT {name}")
570 return self.formats[name]

572 def define_format(self, arguments):
573 name, exponent_text, fraction_text = arguments
574 if name in self.formats:
575 raise CommandError(f"FORMAT EXISTS {name}")
576 exponent_bits, fraction_bits = int(exponent_text), int(fraction_text)
577 if not (2 <= exponent_bits <= 5 and 1 <= fraction_bits <= 10):
578 raise CommandError("BAD FORMAT")
579 fmt = Format(name, exponent_bits, fraction_bits)
580 self.formats[name] = fmt
581 self.emit(
582 "FORMAT", name, f"e={exponent_bits}", f"m={fraction_bits}", f"bits={fmt.width}",
583 f"bias={fmt.bias}", f"emin={fmt.emin}", f"emax={fmt.emax}",
584 )

586 def encode(self, arguments):
587 fmt = self.lookup(arguments[0])
588 text = arguments[1]
589 if text.startswith("0b") or text.startswith("0x"):
590 raise CommandError(f"BAD OPERAND {text}")
591 if text in SPECIAL_VALUES:
592 mode = optional_mode(arguments, 2)
593 value, pattern, flags = text, fmt.special(text), 0
594 else:
595 negative, ratio = parse_number(text)
596 mode = optional_mode(arguments, 2)
597 pattern, flags = fmt.encode(negative, ratio, mode)
598 value = ratio.fraction_text(negative)
599 self.emit("ENCODE", fmt.name, mode, value, fmt.result_text(pattern, flags))

601 def decode(self, arguments):
602 fmt = self.lookup(arguments[0])
603 pattern = operand(fmt, arguments[1])
604 self.emit(
605 "DECODE", fmt.name, fmt.bit_text(pattern), fmt.hex_text(pattern),
606 fmt.classify(pattern), f"sign={fmt.sign(pattern)}",
607 )
608 field = fmt.field(pattern)
609 fraction = fmt.fraction(pattern)
610 fields = [f"exponent={field}", f"fraction={fraction}"]
611 if fmt.is_finite(pattern):
612 sign, significand, exponent = fmt.term(pattern)
613 leading = "0" if field == 0 else "1"
614 fields.append(f"unbiased={exponent + fmt.fraction_bits}")
615 fields.append(f"significand={leading}.{fraction:0{fmt.fraction_bits}b}")
616 self.emit("FIELDS", *fields)
617 number = Ratio(significand, exponent, 1).fraction_text(sign == 1)
618 self.emit("VALUE", number, fmt.value_text(pattern))
619 elif fmt.is_infinity(pattern):
620 self.emit("FIELDS", *fields)
621 else:
622 self.emit("FIELDS", *fields, f"payload={fraction % (fmt.hidden_bit // 2)}")

624 def arithmetic(self, command, arguments):
625 count, operation = ARITHMETIC[command]
626 fmt = self.lookup(arguments[0])
627 operands = [operand(fmt, text) for text in arguments[1:count + 1]]
628 mode = optional_mode(arguments, count + 1)
629 pattern, flags = operation(fmt, *operands, mode)
630 self.emit(command, fmt.name, mode, fmt.result_text(pattern, flags))

632 def convert(self, arguments):

636 mode = optional_mode(arguments, 3)

640 def compare(self, arguments):
641 fmt = self.lookup(arguments[0])
642 a = operand(fmt, arguments[1])
643 b = operand(fmt, arguments[2])
644 flags = INVALID if fmt.is_signaling(a) or fmt.is_signaling(b) else 0
645 total = ordering(fmt.order_key(a), fmt.order_key(b))
646 if fmt.is_nan(a) or fmt.is_nan(b):
647 relation = "UN"
648 elif fmt.is_zero(a) and fmt.is_zero(b):
649 relation = "EQ"
650 else:
651 relation = total
652 self.emit(
653 "COMPARE", fmt.name, fmt.hex_text(a), fmt.hex_text(b),
654 relation, f"total={total}", flag_text(flags),
655 )

657 def neighbor(self, command, arguments):
658 fmt = self.lookup(arguments[0])
659 value = operand(fmt, arguments[1])
660 pattern, flags = NEIGHBORS[command](fmt, value)

663 def to_integer(self, arguments):
664 fmt = self.lookup(arguments[0])
665 value = operand(fmt, arguments[1])
666 kind = arguments[2]
667 if kind not in INTEGER_TYPES:
668 raise CommandError(f"BAD TYPE {kind}")
669 mode = optional_mode(arguments, 3)
670 prefix = ("TOINT", fmt.name, kind, mode)
671 if not fmt.is_finite(value):
672 self.emit(*prefix, "invalid", flag_text(INVALID))
673 return
674 sign, significand, exponent = fmt.term(value)
675 whole, remainder = Ratio(significand, exponent, 1).split(0)
676 if rounds_up(mode, sign, whole, remainder):
677 whole += 1
678 result = -whole if sign == 1 else whole
679 lowest, highest = INTEGER_TYPES[kind]
680 if not lowest <= result <= highest:
681 self.emit(*prefix, "invalid", flag_text(INVALID))
682 return
683 flags = INEXACT if remainder != NO_REMAINDER else 0
684 self.emit(*prefix, result, flag_text(flags))

686 def table(self, arguments):
687 fmt = self.lookup(arguments[0])
688 if fmt.width > 8:
689 raise CommandError("TABLE TOO LARGE")
690 count = fmt.top_field * fmt.hidden_bit
691 self.emit("TABLE", fmt.name, f"count={count}")
692 for index in range(count):
693 if index == count - 1:
694 gap = "-"
695 else:
696 _, upper, upper_exponent = fmt.term(index + 1)
697 _, lower, lower_exponent = fmt.term(index)
698 gap = decimal_text(*aligned_difference(upper, upper_exponent, lower, lower_exponent))
699 self.emit("ENTRY", index, fmt.pattern_text(index), f"gap={gap}")
700 self.emit("MAX", fmt.pattern_text(fmt.largest))
701 self.emit("MINNORMAL", fmt.pattern_text(fmt.hidden_bit))
702 self.emit("MINSUBNORMAL", fmt.pattern_text(1))
703 self.emit("EPSILON", decimal_text(1, -fmt.fraction_bits))
704 self.emit(
705 "PATTERNS", f"finite={2 * count}", "infinite=2",
706 f"nan={2 * (fmt.hidden_bit - 1)}", f"total={2 ** fmt.width}",
707 )

709 def summation_arguments(self, arguments):
710 fmt = self.lookup(arguments[0])
711 mode = checked_mode(arguments[1])
712 values = [operand(fmt, text) for text in arguments[2:]]
713 require_finite(fmt, values)
714 return fmt, mode, values

716 def summary(self, fmt, mode, values, result, flags):
717 terms = [fmt.term(value) for value in values]
718 exact = exact_sum(terms)
719 rounded, rounded_flags = round_sum(fmt, terms, mode)
720 self.emit("RESULT", fmt.pattern_text(result), flag_text(flags))
721 self.emit("EXACT", decimal_text(*exact))
722 self.emit("ROUNDED", fmt.result_text(rounded, rounded_flags))
723 self.emit("DIFF", error_text(fmt, result, exact))

725 def sequential_sum(self, arguments):
726 fmt, mode, values = self.summation_arguments(arguments)
727 self.emit("SUM", fmt.name, mode, f"count={len(values)}")
728 total = values[0]
729 flags = 0
730 for index in range(1, len(values)):
731 total, step_flags = add(fmt, total, values[index], mode)
732 flags |= step_flags
733 self.emit("PARTIAL", index + 1, fmt.result_text(total, step_flags))
734 self.summary(fmt, mode, values, total, flags)

736 def kahan_sum(self, arguments):
737 fmt, mode, values = self.summation_arguments(arguments)
738 self.emit("KAHAN", fmt.name, mode, f"count={len(values)}")
739 total = values[0]
740 compensation = fmt.zero(0)
741 flags = 0
742 for index in range(1, len(values)):
743 corrected, corrected_flags = subtract(fmt, values[index], compensation, mode)
744 updated, updated_flags = add(fmt, total, corrected, mode)
745 change, change_flags = subtract(fmt, updated, total, mode)
746 compensation, compensation_flags = subtract(fmt, change, corrected, mode)
747 step_flags = corrected_flags | updated_flags | change_flags | compensation_flags
748 total = updated
749 flags |= step_flags
750 self.emit(
751 "KSTEP", index + 1, f"sum={fmt.value_text(total)}",
752 f"comp={fmt.value_text(compensation)}", flag_text(step_flags),
753 )
754 self.summary(fmt, mode, values, total, flags)

756 def dot_product(self, arguments):
757 fmt = self.lookup(arguments[0])
758 mode = checked_mode(arguments[1])
759 separator = arguments.index("|")
760 left = [operand(fmt, text) for text in arguments[2:separator]]
761 right = [operand(fmt, text) for text in arguments[separator + 1:]]
762 if len(left) != len(right):
763 raise CommandError("LENGTH MISMATCH")
764 require_finite(fmt, left + right)
765 naive, naive_flags = multiply(fmt, left[0], right[0], mode)
766 for index in range(1, len(left)):
767 product, product_flags = multiply(fmt, left[index], right[index], mode)
768 naive, sum_flags = add(fmt, naive, product, mode)
769 naive_flags |= product_flags | sum_flags
770 chained, chained_flags = multiply(fmt, left[0], right[0], mode)
771 for index in range(1, len(left)):
772 chained, step_flags = fused_multiply_add(fmt, left[index], right[index], chained, mode)
773 chained_flags |= step_flags
774 terms = [product_term(fmt, a, b) for a, b in zip(left, right)]
775 exact = exact_sum(terms)
776 fused, fused_flags = round_sum(fmt, terms, mode)
777 self.emit("DOT", fmt.name, mode, f"n={len(left)}")
778 self.emit("NAIVE", fmt.result_text(naive, naive_flags))
779 self.emit("CHAINED", fmt.result_text(chained, chained_flags))
780 self.emit("EXACT", decimal_text(*exact))
781 self.emit("FUSED", fmt.result_text(fused, fused_flags))
782 self.emit(
783 "DIFF", f"naive={error_text(fmt, naive, exact)}",
784 f"chained={error_text(fmt, chained, exact)}",
785 )

787 def execute(self, line):
788 command, *arguments = line.split(" ")
789 try:
790 if command in ARITHMETIC:
791 self.arithmetic(command, arguments)
792 elif command in NEIGHBORS:
793 self.neighbor(command, arguments)
794 else:
795 self.handlers[command](arguments)
796 except CommandError as error:
797 self.emit("ERROR", error)
