36Mode parse_mode(const std::vector<std::string>& words, std::size_t index) {
37 if (index == words.size()) return Mode::RNE;
38 const auto found = std::find(mode_names.begin(), mode_names.end(), words[index]);

45std::string flags_text(unsigned flags) {
46 static const std::array<std::string, 5> names{"invalid", "divbyzero", "overflow", "underflow", "inexact"};
47 std::string result;
48 for (std::size_t i = 0; i < names.size(); ++i) {
49 if (!(flags & (1u << i))) continue;
50 if (!result.empty()) result += ',';
51 result += names[i];
52 }
53 return "flags=" + (result.empty() ? "none" : result);
54}

56std::string binary(int value, int width) {
57 std::string text(width, '0');
58 for (int i = 0; i < width; ++i) if (value & (1 << i)) text[width - 1 - i] = '1';
59 return text;

62int bit_length(Integer n) {
63 int length = 0;
64 for (; n; n >>= 1) ++length;
65 return length;
66}

68std::pair<Integer, Integer> scaled(Integer n, Integer d, int exponent) {
69 if (exponent >= 0) return {n << exponent, d};
70 return {n, d << -exponent};
71}
72
73Integer integer_sqrt(Integer n) {
74 if (n == 0) return 0;
75 Integer root = Integer{1} << ((bit_length(n) + 1) / 2);
76 for (;;) {
77 const Integer next = (root + n / root) / 2;
78 if (next >= root) return root;
79 root = next;
80 }
81}

83struct Exact {
84 Integer n;
85 int exponent;
86 Integer denominator;
87 bool negative;
88
89 Exact(Integer n = 0, int exponent = 0, Integer denominator = 1, bool negative = false)
90 : n(n), exponent(exponent), denominator(denominator), negative(negative) {}
91
92 Exact negated() const { return {n, exponent, denominator, !negative}; }
93
94 int log2_floor() const {
95 int t = bit_length(n) - bit_length(denominator) + exponent;
96 const auto [numerator, divisor] = scaled(n, denominator, exponent - t);
97 if (numerator < divisor) --t;
98 return t;
99 }
100
101 std::string fraction() const {
102 if (n == 0) return "0";
103 auto [numerator, divisor] = scaled(n, denominator, exponent);
104 const Integer factor = std::gcd(numerator, divisor);
105 numerator /= factor;
106 divisor /= factor;
107 return (negative ? "-" : "") + std::to_string(numerator)
108 + (divisor == 1 ? "" : "/" + std::to_string(divisor));
109 }
110
111 std::string decimal(bool keep_zero_sign = false) const {
112 const auto [numerator, divisor] = scaled(n, denominator, exponent);
113 std::string text = std::to_string(numerator / divisor);
114 Integer remainder = numerator % divisor;
115 if (remainder) {
116 text += '.';
117 while (remainder) {
118 remainder *= 10;
119 text += static_cast<char>('0' + remainder / divisor);
120 remainder %= divisor;
121 }
122 }
123 if (negative && (n != 0 || keep_zero_sign)) text.insert(text.begin(), '-');
124 return text;
125 }
126};

128Exact exact_product(const Exact& a, const Exact& b) {
129 return {a.n * b.n, a.exponent + b.exponent, a.denominator * b.denominator, a.negative != b.negative};
130}
131
132Exact exact_sum(const std::vector<Exact>& terms, Mode mode) {
133 std::optional<int> exponent;
134 for (const Exact& term : terms) {
135 if (term.n != 0 && (!exponent || term.exponent < *exponent)) exponent = term.exponent;
136 }
137 if (!exponent) {
138 const bool same_sign = std::all_of(terms.begin(), terms.end(), [&](const Exact& term) {
139 return term.negative == terms.front().negative;
140 });
141 return {0, 0, 1, same_sign ? terms.front().negative : mode == Mode::RDN};
142 }
143 Integer numerator = 0;
144 for (const Exact& term : terms) {
145 if (!term.n) continue;
146 const Integer aligned = term.n << (term.exponent - *exponent);
147 numerator += term.negative ? -aligned : aligned;
148 }
149 const bool negative = numerator < 0 || (numerator == 0 && mode == Mode::RDN);
150 return {numerator < 0 ? -numerator : numerator, *exponent, 1, negative};
151}
152
153bool increment(Mode mode, bool negative, Integer integer, bool inexact, Integer halfway) {
154 if (!inexact) return false;
155 switch (mode) {
156 case Mode::RNE: return halfway > 0 || (halfway == 0 && integer % 2 != 0);
157 case Mode::RNA: return halfway >= 0;
158 case Mode::RTZ: return false;
159 case Mode::RUP: return !negative;
160 case Mode::RDN: return negative;
161 }
162 return false;
163}
164
165Units rounded_units(const Exact& value, int exponent, Mode mode) {
166 const auto [n, d] = scaled(value.n, value.denominator, value.exponent - exponent);
167 Integer integer = n / d;
168 const Integer remainder = n % d;
169 if (increment(mode, value.negative, integer, remainder != 0, 2 * remainder - d)) ++integer;
170 return {integer, remainder != 0};
171}
172
173Units rounded_sqrt_units(const Exact& value, int exponent, Mode mode) {
174 const auto [n, d] = scaled(value.n, value.denominator, value.exponent - 2 * exponent);
175 Integer integer = integer_sqrt(n / d);
176 const bool inexact = integer * integer * d != n;
177 const Integer midpoint = 2 * integer + 1;
178 if (increment(mode, false, integer, inexact, 4 * n - d * midpoint * midpoint)) ++integer;
179 return {integer, inexact};
180}

182struct Fields { bool negative; int exponent, fraction, q; };

184class Format {
185 template<class UnitsAt>
186 Result round_nonzero(bool negative, int t, Mode mode, UnitsAt units_at) const {
187 if (t > emax) return overflow(negative, mode);
188 int exponent = std::max(t, emin) - m;
189 const Units units = units_at(exponent);
190 Integer n = units.integer;
191 if (n == 2 * hidden) {
192 n = hidden;
193 ++exponent;
194 }
195 if (exponent > emax - m) return overflow(negative, mode);
196 const int magnitude = n >= hidden
197 ? (exponent + m + bias) * hidden + static_cast<int>(n - hidden) : static_cast<int>(n);
198 unsigned flags = units.inexact ? Inexact : 0;
199 if (units.inexact && t < emin && (t < emin - 1 || units_at(t - m).integer < 2 * hidden)) flags |= Underflow;
200 return {zero(negative) + magnitude, flags};
201 }
202
203public:
204 std::string name;
205 int e, m, w, bias, emin, emax, hidden, exponent_mask, sign_bit, infinity, nan, maximum;
206
207 Format(std::string name, int e, int m)
208 : name(std::move(name)), e(e), m(m), w(1 + e + m), bias((1 << (e - 1)) - 1),
209 emin(1 - bias), emax(bias), hidden(1 << m), exponent_mask((1 << e) - 1),
210 sign_bit(1 << (e + m)), infinity(exponent_mask * hidden), nan(infinity + hidden / 2), maximum(infinity - 1) {}
211
212 Fields fields(int pattern) const {
213 const bool negative = (pattern & sign_bit) != 0;
214 const int exponent = (pattern >> m) & exponent_mask;
215 const int fraction = pattern & (hidden - 1);
216 return {negative, exponent, fraction, (exponent == 0 ? emin : exponent - bias) - m};
217 }
218
219 bool is_nan(int pattern) const {
220 const Fields f = fields(pattern);
221 return f.exponent == exponent_mask && f.fraction != 0;
222 }
223 bool is_signaling(int pattern) const { return is_nan(pattern) && fields(pattern).fraction < hidden / 2; }
224 bool is_infinity(int pattern) const { return (pattern & (sign_bit - 1)) == infinity; }
225 bool is_zero(int pattern) const { return (pattern & (sign_bit - 1)) == 0; }
226 bool is_finite(int pattern) const { return fields(pattern).exponent != exponent_mask; }
227 int zero(bool negative) const { return negative ? sign_bit : 0; }
228 int inf(bool negative) const { return zero(negative) + infinity; }
229 int flip(int pattern) const { return pattern ^ sign_bit; }
230
231 Exact exact(int pattern) const {
232 const Fields f = fields(pattern);
233 return {f.exponent == 0 ? f.fraction : hidden + f.fraction, f.q, 1, f.negative};
234 }
235
236 std::string bits(int pattern) const {
237 const Fields f = fields(pattern);
238 return std::string(f.negative ? "1|" : "0|") + binary(f.exponent, e) + '|' + binary(f.fraction, m);
239 }
240
241 std::string hex(int pattern) const {
242 std::ostringstream out;
243 out << "0x" << std::uppercase << std::hex << std::setw((w + 3) / 4) << std::setfill('0') << pattern;
244 return out.str();
245 }
246
247 std::string value(int pattern) const {
248 if (is_nan(pattern)) return "nan";
249 if (is_infinity(pattern)) return fields(pattern).negative ? "-inf" : "inf";
250 return exact(pattern).decimal(true);
251 }
252
253 std::string pattern_text(int pattern) const { return bits(pattern) + ' ' + hex(pattern) + ' ' + value(pattern); }
254 std::string result_text(Result result) const { return pattern_text(result.pattern) + ' ' + flags_text(result.flags); }
255
256 Result overflow(bool negative, Mode mode) const {
257 const bool to_infinity = mode == Mode::RNE || mode == Mode::RNA
258 || (mode == Mode::RUP && !negative) || (mode == Mode::RDN && negative);
259 return {zero(negative) + (to_infinity ? infinity : maximum), Overflow | Inexact};
260 }
261
262 Result round(const Exact& value, Mode mode) const {
263 if (value.n == 0) return {zero(value.negative)};
264 return round_nonzero(value.negative, value.log2_floor(), mode,
265 [&](int exponent) { return rounded_units(value, exponent, mode); });
266 }
267
268 Result sqrt(const Exact& value, Mode mode) const {
269 const int log = value.log2_floor();
270 const int t = log >= 0 ? log / 2 : -((-log + 1) / 2);
271 return round_nonzero(false, t, mode, [&](int exponent) { return rounded_sqrt_units(value, exponent, mode); });
272 }

275enum class RealKind { Finite, Infinity, NaN };

278Integer decimal_integer(const std::string& text) {
279 Integer value = 0;
280 for (char digit : text) value = value * 10 + digit - '0';
281 return value;
282}
283
284RealInput parse_real(const std::string& word) {
285 if (word == "nan") return {RealKind::NaN, {}};
286 if (word == "inf" || word == "-inf") return {RealKind::Infinity, {0, 0, 1, word == "-inf"}};
287 if (word.compare(0, 2, "0b") == 0 || word.compare(0, 2, "0x") == 0) fail("BAD OPERAND " + word);
288 const bool negative = word.front() == '-';
289 std::string body = negative ? word.substr(1) : word;
290 const auto slash = body.find('/');
291 if (slash != std::string::npos) {
292 const Integer n = decimal_integer(body.substr(0, slash));
293 const Integer d = decimal_integer(body.substr(slash + 1));
294 if (d == 0) fail("BAD OPERAND " + word);
295 return {RealKind::Finite, {n, 0, d, negative}};
296 }
297 Integer denominator = 1;
298 const auto dot = body.find('.');
299 if (dot != std::string::npos) {
300 for (std::size_t i = dot + 1; i < body.size(); ++i) denominator *= 10;
301 body.erase(dot, 1);
302 }
303 return {RealKind::Finite, {decimal_integer(body), 0, denominator, negative}};
304}

306Result encoded(const Format& format, const RealInput& input, Mode mode) {
307 if (input.kind == RealKind::NaN) return {format.nan};
308 if (input.kind == RealKind::Infinity) return {format.inf(input.exact.negative)};
309 return format.round(input.exact, mode);
310}

312int operand(const Format& format, const std::string& word) {
313 const bool bits = word.compare(0, 2, "0b") == 0;
314 const bool hex = word.compare(0, 2, "0x") == 0;
315 if (bits || hex) {
316 const auto width = static_cast<std::size_t>(bits ? format.w : (format.w + 3) / 4);
317 if (word.size() - 2 != width) fail("BAD OPERAND " + word);
318 int pattern = 0;
319 for (std::size_t i = 2; i < word.size(); ++i)
320 pattern = pattern * (bits ? 2 : 16) + (word[i] <= '9' ? word[i] - '0' : word[i] - 'A' + 10);
321 if (pattern >= (1 << format.w)) fail("BAD OPERAND " + word);
322 return pattern;
323 }
324 return encoded(format, parse_real(word), Mode::RNE).pattern;
325}

327std::optional<Operation> parse_operation(const std::string& command) {
328 if (command == "ADD") return Operation::Add;
329 if (command == "SUB") return Operation::Sub;
330 if (command == "MUL") return Operation::Mul;
331 if (command == "DIV") return Operation::Div;
332 if (command == "SQRT") return Operation::Sqrt;
333 if (command == "FMA") return Operation::Fma;
334 return std::nullopt;
335}
336
337Result operation(const Format& format, Operation op, std::vector<int> inputs, Mode mode) {
338 if (op == Operation::Sub) {
339 inputs[1] = format.flip(inputs[1]);
340 op = Operation::Add;
341 }
342 const bool nan = std::any_of(inputs.begin(), inputs.end(), [&](int p) { return format.is_nan(p); });
343 if (nan) {
344 const bool signaling = std::any_of(inputs.begin(), inputs.end(), [&](int p) { return format.is_signaling(p); });
345 return {format.nan, signaling ? Invalid : 0u};
346 }
347 const int a = inputs[0];
348 const bool negative = format.fields(a).negative;
349 const Result invalid{format.nan, Invalid};
350 if (op == Operation::Sqrt) {
351 if (format.is_zero(a)) return {a};
352 if (negative) return invalid;
353 if (format.is_infinity(a)) return {a};
354 return format.sqrt(format.exact(a), mode);
355 }
356 const int b = inputs[1];
357 const bool sigma = negative != format.fields(b).negative;
358 if (op == Operation::Add) {
359 if (format.is_infinity(a) && format.is_infinity(b) && sigma) return invalid;
360 if (format.is_infinity(a)) return {a};
361 if (format.is_infinity(b)) return {b};
362 return format.round(exact_sum({format.exact(a), format.exact(b)}, mode), mode);
363 }
364 if (op == Operation::Mul) {
365 if (format.is_infinity(a) || format.is_infinity(b))
366 return format.is_zero(a) || format.is_zero(b) ? invalid : Result{format.inf(sigma)};
367 if (format.is_zero(a) || format.is_zero(b)) return {format.zero(sigma)};
368 return format.round(exact_product(format.exact(a), format.exact(b)), mode);
369 }
370 if (op == Operation::Div) {
371 if (format.is_infinity(a)) return format.is_infinity(b) ? invalid : Result{format.inf(sigma)};
372 if (format.is_infinity(b)) return {format.zero(sigma)};
373 if (format.is_zero(b)) return format.is_zero(a) ? invalid : Result{format.inf(sigma), DivByZero};
374 if (format.is_zero(a)) return {format.zero(sigma)};
375 const Exact x = format.exact(a), y = format.exact(b);
376 return format.round({x.n * y.denominator, x.exponent - y.exponent, x.denominator * y.n, sigma}, mode);
377 }
378 const int c = inputs[2];
379 if (format.is_infinity(a) || format.is_infinity(b)) {
380 if (format.is_zero(a) || format.is_zero(b) || (format.is_infinity(c) && format.fields(c).negative != sigma)) return invalid;
381 return {format.inf(sigma)};
382 }
383 if (format.is_infinity(c)) return {c};
384 return format.round(exact_sum({exact_product(format.exact(a), format.exact(b)), format.exact(c)}, mode), mode);
385}

387int next_up(const Format& format, int pattern) {
388 if (pattern == format.infinity) return pattern;
389 if (format.is_zero(pattern)) return 1;
390 return pattern + (format.fields(pattern).negative ? -1 : 1);
391}

430 static void summation(const std::string& command, const Format& format, Mode mode, const std::vector<int>& patterns) {
431 if (!all_finite(format, patterns)) fail("NOT FINITE");
432 std::cout << command << ' ' << format.name << ' ' << mode_text(mode) << " count=" << patterns.size() << '\n';
433 int sum = patterns[0], compensation = 0;
434 unsigned flags = 0;
435 for (std::size_t i = 1; i < patterns.size(); ++i) {
436 if (command == "SUM") {
437 const Result added = operation(format, Operation::Add, {sum, patterns[i]}, mode);
438 sum = added.pattern;
439 flags |= added.flags;
440 std::cout << "PARTIAL " << i + 1 << ' ' << format.result_text(added) << '\n';
441 } else {
442 const Result y = operation(format, Operation::Sub, {patterns[i], compensation}, mode);
443 const Result t = operation(format, Operation::Add, {sum, y.pattern}, mode);
444 const Result d = operation(format, Operation::Sub, {t.pattern, sum}, mode);
445 const Result c = operation(format, Operation::Sub, {d.pattern, y.pattern}, mode);
446 sum = t.pattern;
447 compensation = c.pattern;
448 const unsigned step_flags = y.flags | t.flags | d.flags | c.flags;
449 flags |= step_flags;
450 std::cout << "KSTEP " << i + 1 << " sum=" << format.value(sum) << " comp=" << format.value(compensation)
451 << ' ' << flags_text(step_flags) << '\n';
452 }
453 }
454 summary(format, mode, patterns, {sum, flags});

457 static void dot(const Format& format, Mode mode, const std::vector<int>& a, const std::vector<int>& b) {
458 if (a.size() != b.size()) fail("LENGTH MISMATCH");
459 if (!all_finite(format, a) || !all_finite(format, b)) fail("NOT FINITE");
460 Result naive = operation(format, Operation::Mul, {a[0], b[0]}, mode);
461 Result chained = naive;
462 for (std::size_t i = 1; i < a.size(); ++i) {
463 const Result product = operation(format, Operation::Mul, {a[i], b[i]}, mode);
464 const Result added = operation(format, Operation::Add, {naive.pattern, product.pattern}, mode);
465 naive = {added.pattern, naive.flags | product.flags | added.flags};
466 const Result fused = operation(format, Operation::Fma, {a[i], b[i], chained.pattern}, mode);
467 chained = {fused.pattern, chained.flags | fused.flags};
468 }
469 std::vector<Exact> terms;
470 for (std::size_t i = 0; i < a.size(); ++i) terms.push_back(exact_product(format.exact(a[i]), format.exact(b[i])));
471 const Exact exact = exact_sum(terms, mode);
472 std::cout << "DOT " << format.name << ' ' << mode_text(mode) << " n=" << a.size() << '\n'
473 << "NAIVE " << format.result_text(naive) << '\n'
474 << "CHAINED " << format.result_text(chained) << '\n'
475 << "EXACT " << exact.decimal() << '\n'
476 << "FUSED " << format.result_text(format.round(exact, mode)) << '\n'
477 << "DIFF naive=" << difference(format, naive.pattern, exact)
478 << " chained=" << difference(format, chained.pattern, exact) << '\n';
479 }

505 void command(const std::vector<std::string>& words) {
506 const std::string& command = words[0];
507 if (command == "FORMAT") {
508 const std::string& name = words[1];
509 const int e = static_cast<int>(decimal_integer(words[2])), m = static_cast<int>(decimal_integer(words[3]));
510 if (formats.find(name) != formats.end()) fail("FORMAT EXISTS " + name);
511 if (e < 2 || e > 5 || m < 1 || m > 10) fail("BAD FORMAT");
512 const auto entry = formats.emplace(name, Format(name, e, m));
513 const Format& f = entry.first->second;
514 std::cout << "FORMAT " << name << " e=" << e << " m=" << m << " bits=" << f.w
515 << " bias=" << f.bias << " emin=" << f.emin << " emax=" << f.emax << '\n';
516 return;
517 }
518 const Format& f = format(words[1]);
519 if (command == "ENCODE") {
520 const RealInput input = parse_real(words[2]);
521 const Mode mode = parse_mode(words, 3);
522 const std::string value = input.kind == RealKind::Finite ? input.exact.fraction() : words[2];
523 std::cout << "ENCODE " << f.name << ' ' << mode_text(mode) << ' ' << value << ' '
524 << f.result_text(encoded(f, input, mode)) << '\n';
525 } else if (command == "CONVERT") {
526 const Format& g = format(words[2]);
527 const int a = operand(f, words[3]);
528 const Mode mode = parse_mode(words, 4);
529 Result result;
530 if (f.is_nan(a)) result = {g.nan, f.is_signaling(a) ? Invalid : 0u};
531 else if (f.is_infinity(a)) result = {g.inf(f.fields(a).negative)};
532 else result = g.round(f.exact(a), mode);
533 std::cout << "CONVERT " << f.name << ' ' << g.name << ' ' << mode_text(mode) << ' ' << g.result_text(result) << '\n';
534 } else if (command == "SUM" || command == "KAHAN" || command == "DOT") {
535 const Mode mode = parse_mode(words, 2);
536 if (command == "DOT") {
537 const auto separator = static_cast<std::size_t>(std::find(words.begin(), words.end(), "|") - words.begin());
538 const auto a = operands(f, words, 3, separator);
539 const auto b = operands(f, words, separator + 1, words.size());
540 dot(f, mode, a, b);
541 } else {
542 const auto patterns = operands(f, words, 3, words.size());
543 summation(command, f, mode, patterns);
544 }
545 } else if (command == "TABLE") table(f);
546 else if (const auto op = parse_operation(command)) {
547 const std::size_t count = *op == Operation::Sqrt ? 1 : *op == Operation::Fma ? 3 : 2;
548 const auto patterns = operands(f, words, 2, 2 + count);
549 const Mode mode = parse_mode(words, 2 + count);
550 std::cout << command << ' ' << f.name << ' ' << mode_text(mode) << ' '
551 << f.result_text(operation(f, *op, patterns, mode)) << '\n';
552 } else {
553 const int a = operand(f, words[2]);
554 const Fields fields = f.fields(a);
555 if (command == "DECODE") {
556 const std::string classification = f.is_nan(a) ? (f.is_signaling(a) ? "snan" : "qnan")
557 : f.is_infinity(a) ? "infinity" : f.is_zero(a) ? "zero" : fields.exponent == 0 ? "subnormal" : "normal";
558 std::cout << "DECODE " << f.name << ' ' << f.bits(a) << ' ' << f.hex(a) << ' ' << classification
559 << " sign=" << (fields.negative ? 1 : 0) << '\n';
560 std::cout << "FIELDS exponent=" << fields.exponent << " fraction=" << fields.fraction;
561 if (f.is_finite(a)) {
562 std::cout << " unbiased=" << fields.q + f.m << " significand=" << (fields.exponent == 0 ? '0' : '1')
563 << '.' << binary(fields.fraction, f.m) << '\n'
564 << "VALUE " << f.exact(a).fraction() << ' ' << f.value(a) << '\n';
565 } else {
566 if (f.is_nan(a)) std::cout << " payload=" << fields.fraction % (f.hidden / 2);
567 std::cout << '\n';
568 }
569 } else if (command == "COMPARE") {
570 const int b = operand(f, words[3]);
571 auto key = [&](int p) { return f.fields(p).negative ? -(p - f.sign_bit) - 1 : p; };
572 const std::string order = relation(key(a), key(b));
573 const std::string numeric = f.is_nan(a) || f.is_nan(b) ? "UN"
574 : f.is_zero(a) && f.is_zero(b) ? "EQ" : order;
575 const unsigned flags = f.is_signaling(a) || f.is_signaling(b) ? Invalid : 0;
576 std::cout << "COMPARE " << f.name << ' ' << f.hex(a) << ' ' << f.hex(b) << ' ' << numeric
577 << " total=" << order << ' ' << flags_text(flags) << '\n';
578 } else if (command == "TOINT") {
579 const auto type = integer_types.find(words[3]);
580 if (type == integer_types.end()) fail("BAD TYPE " + words[3]);
581 const Mode mode = parse_mode(words, 4);
582 std::cout << "TOINT " << f.name << ' ' << words[3] << ' ' << mode_text(mode) << ' ';
583 if (!f.is_finite(a)) std::cout << "invalid flags=invalid\n";
584 else {
585 const Units units = rounded_units(f.exact(a), 0, mode);
586 const Integer integer = fields.negative ? -units.integer : units.integer;
587 const auto [minimum, maximum] = type->second;
588 if (integer < minimum || integer > maximum) std::cout << "invalid flags=invalid\n";
589 else std::cout << integer << ' ' << flags_text(units.inexact ? Inexact : 0) << '\n';
590 }
591 } else {
592 Result result;
593 if (f.is_nan(a)) result = {f.nan, f.is_signaling(a) ? Invalid : 0u};
594 else if (command == "NEXTUP") result = {next_up(f, a)};
595 else if (command == "NEXTDOWN") result = {f.flip(next_up(f, f.flip(a)))};
596 else result = {f.is_infinity(a) ? f.infinity : f.round({1, fields.q}, Mode::RNE).pattern};
597 std::cout << command << ' ' << f.name << ' ' << f.result_text(result) << '\n';
598 }
599 }
600 }
601};
