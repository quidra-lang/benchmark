19static class IntegerMath
20{
21 public static int BitLength(long value)
22 {
23 int length = 0;
24 for (; value != 0; value >>= 1) length++;
25 return length;
26 }
27
28 public static (long Numerator, long Denominator) Scaled(long numerator, long denominator, int exponent) =>
29 exponent >= 0 ? (numerator << exponent, denominator) : (numerator, denominator << -exponent);
30
31 public static long Gcd(long a, long b)
32 {
33 while (b != 0) (a, b) = (b, a % b);
34 return a;
35 }
36
37 public static long IntegerSqrt(long value)
38 {
39 if (value == 0) return 0;
40 long root = 1L << ((BitLength(value) + 1) / 2);
41 while (true)
42 {
43 long next = (root + value / root) / 2;
44 if (next >= root) return root;
45 root = next;
46 }
47 }
48
49 public static bool Increment(Mode mode, bool negative, long integer, bool inexact, long halfway)
50 {
51 if (!inexact) return false;
52 return mode switch
53 {
54 Mode.Rne => halfway > 0 || halfway == 0 && integer % 2 != 0,
55 Mode.Rna => halfway >= 0,
56 Mode.Rtz => false,
57 Mode.Rup => !negative,
58 Mode.Rdn => negative,
59 _ => throw new ArgumentOutOfRangeException(nameof(mode))
60 };
61 }
62
63 public static Units RoundUnits(Exact value, int exponent, Mode mode)
64 {
65 var (n, d) = Scaled(value.Numerator, value.Denominator, value.Exponent - exponent);
66 long integer = n / d, remainder = n % d;
67 if (Increment(mode, value.Negative, integer, remainder != 0, 2 * remainder - d)) integer++;
68 return new(integer, remainder != 0);
69 }
70
71 public static Units RoundSqrtUnits(Exact value, int exponent, Mode mode)
72 {
73 var (n, d) = Scaled(value.Numerator, value.Denominator, value.Exponent - 2 * exponent);
74 long integer = IntegerSqrt(n / d);
75 bool inexact = integer * integer * d != n;
76 long midpoint = 2 * integer + 1;
77 if (Increment(mode, false, integer, inexact, 4 * n - d * midpoint * midpoint)) integer++;
78 return new(integer, inexact);
79 }
80}

82readonly record struct Exact(long Numerator, int Exponent = 0, long Denominator = 1, bool Negative = false)
83{
84 public Exact Negated() => this with { Negative = !Negative };
85
86 public int Log2Floor()
87 {
88 int exponent = BitLength(Numerator) - BitLength(Denominator) + Exponent;
89 var (n, d) = Scaled(Numerator, Denominator, Exponent - exponent);
90 return n < d ? exponent - 1 : exponent;
91 }
92
93 public string Fraction()
94 {
95 if (Numerator == 0) return "0";
96 var (n, d) = Scaled(Numerator, Denominator, Exponent);
97 long factor = Gcd(n, d);
98 n /= factor;
99 d /= factor;
100 return (Negative ? "-" : "") + n + (d == 1 ? "" : "/" + d);
101 }
102
103 public string Decimal(bool keepZeroSign = false)
104 {
105 var (n, d) = Scaled(Numerator, Denominator, Exponent);
106 var text = new StringBuilder();
107 if (Negative && (Numerator != 0 || keepZeroSign)) text.Append('-');
108 text.Append(n / d);
109 long remainder = n % d;
110 if (remainder != 0)
111 {
112 text.Append('.');
113 while (remainder != 0)
114 {
115 remainder *= 10;
116 text.Append((char)('0' + remainder / d));
117 remainder %= d;
118 }
119 }
120 return text.ToString();
121 }
122
123 public static Exact Product(Exact a, Exact b) =>
124 new(a.Numerator * b.Numerator, a.Exponent + b.Exponent, a.Denominator * b.Denominator, a.Negative != b.Negative);
125
126 public static Exact Sum(IReadOnlyList<Exact> terms, Mode mode)
127 {
128 int? exponent = null;
129 foreach (var term in terms)
130 if (term.Numerator != 0 && (exponent is null || term.Exponent < exponent.Value)) exponent = term.Exponent;
131 if (exponent is null)
132 {
133 bool same = terms.All(term => term.Negative == terms[0].Negative);
134 return new(0, Negative: same ? terms[0].Negative : mode == Mode.Rdn);
135 }
136 long numerator = 0;
137 foreach (var term in terms)
138 {
139 if (term.Numerator == 0) continue;
140 long aligned = term.Numerator << (term.Exponent - exponent.Value);
141 numerator += term.Negative ? -aligned : aligned;
142 }
143 return new(Math.Abs(numerator), exponent.Value, Negative: numerator < 0 || numerator == 0 && mode == Mode.Rdn);
144 }
145}

147static class Notation
148{
149 public static string ModeText(Mode mode) => mode.ToString().ToUpperInvariant();
150 public static string Binary(int value, int width) => Convert.ToString(value, 2).PadLeft(width, '0');
151 public static string Relation(int a, int b) => a < b ? "LT" : a > b ? "GT" : "EQ";
152
153 public static string FlagsText(FpFlags flags)
154 {
155 (FpFlags Flag, string Name)[] names =
156 [
157 (FpFlags.Invalid, "invalid"), (FpFlags.DivByZero, "divbyzero"), (FpFlags.Overflow, "overflow"),
158 (FpFlags.Underflow, "underflow"), (FpFlags.Inexact, "inexact")
159 ];
160 var raised = new List<string>();
161 foreach (var (flag, name) in names) if ((flags & flag) != 0) raised.Add(name);
162 return "flags=" + (raised.Count == 0 ? "none" : string.Join(',', raised));
163 }
164}

166sealed record MiniFormat(string Name, int E, int M)
167{
168 public int Width => 1 + E + M;
169 public int Bias => (1 << (E - 1)) - 1;
170 public int MinExponent => 1 - Bias;
171 public int MaxExponent => Bias;
172 public int Hidden => 1 << M;
173 public int ExponentMask => (1 << E) - 1;
174 public int SignBit => 1 << (E + M);
175 public int Infinity => ExponentMask * Hidden;
176 public int NaN => Infinity + Hidden / 2;
177 public int Maximum => Infinity - 1;
178
179 public Fields Fields(int pattern)
180 {
181 int exponent = (pattern >> M) & ExponentMask;
182 return new((pattern & SignBit) != 0, exponent, pattern & (Hidden - 1), (exponent == 0 ? MinExponent : exponent - Bias) - M);
183 }
184 public bool IsNaN(int pattern) => Fields(pattern).Exponent == ExponentMask && Fields(pattern).Fraction != 0;
185 public bool IsSignaling(int pattern) => IsNaN(pattern) && Fields(pattern).Fraction < Hidden / 2;
186 public bool IsInfinity(int pattern) => (pattern & (SignBit - 1)) == Infinity;
187 public bool IsZero(int pattern) => (pattern & (SignBit - 1)) == 0;
188 public bool IsFinite(int pattern) => Fields(pattern).Exponent != ExponentMask;
189 public int Zero(bool negative) => negative ? SignBit : 0;
190 public int Inf(bool negative) => Zero(negative) + Infinity;
191 public int Flip(int pattern) => pattern ^ SignBit;
192
193 public Exact Exact(int pattern)
194 {
195 var fields = Fields(pattern);
196 return new(fields.Exponent == 0 ? fields.Fraction : Hidden + fields.Fraction, fields.Q, Negative: fields.Negative);
197 }
198 public string Bits(int pattern)
199 {
200 var fields = Fields(pattern);
201 return $"{(fields.Negative ? 1 : 0)}|{Notation.Binary(fields.Exponent, E)}|{Notation.Binary(fields.Fraction, M)}";
202 }
203 public string Hex(int pattern) => "0x" + pattern.ToString($"X{(Width + 3) / 4}", CultureInfo.InvariantCulture);
204 public string Value(int pattern) => IsNaN(pattern) ? "nan" : IsInfinity(pattern) ? (Fields(pattern).Negative ? "-inf" : "inf") : Exact(pattern).Decimal(true);
205 public string PatternText(int pattern) => $"{Bits(pattern)} {Hex(pattern)} {Value(pattern)}";
206 public string ResultText(Result result) => $"{PatternText(result.Pattern)} {Notation.FlagsText(result.Flags)}";
207
208 Result Overflow(bool negative, Mode mode)
209 {
210 bool infinity = mode is Mode.Rne or Mode.Rna || mode == Mode.Rup && !negative || mode == Mode.Rdn && negative;
211 return new(Zero(negative) + (infinity ? Infinity : Maximum), FpFlags.Overflow | FpFlags.Inexact);
212 }
213
214 Result RoundNonzero(bool negative, int t, Mode mode, Func<int, Units> unitsAt)
215 {
216 if (t > MaxExponent) return Overflow(negative, mode);
217 int exponent = Math.Max(t, MinExponent) - M;
218 var units = unitsAt(exponent);
219 long n = units.Integer;
220 if (n == 2 * Hidden) { n = Hidden; exponent++; }
221 if (exponent > MaxExponent - M) return Overflow(negative, mode);
222 int magnitude = n >= Hidden ? (exponent + M + Bias) * Hidden + (int)n - Hidden : (int)n;
223 var flags = units.Inexact ? FpFlags.Inexact : FpFlags.None;
224 if (units.Inexact && t < MinExponent && (t < MinExponent - 1 || unitsAt(t - M).Integer < 2 * Hidden)) flags |= FpFlags.Underflow;
225 return new(Zero(negative) + magnitude, flags);
226 }
227
228 public Result Round(Exact value, Mode mode) => value.Numerator == 0 ? new(Zero(value.Negative))
229 : RoundNonzero(value.Negative, value.Log2Floor(), mode, exponent => RoundUnits(value, exponent, mode));
230
231 public Result Sqrt(Exact value, Mode mode)
232 {
233 int log = value.Log2Floor();
234 int exponent = log >= 0 ? log / 2 : -((-log + 1) / 2);
235 return RoundNonzero(false, exponent, mode, q => RoundSqrtUnits(value, q, mode));
236 }
237}

244static class Parsing
245{
246 public static Mode ModeAt(string[] words, int index) => index == words.Length ? Mode.Rne : words[index] switch
247 {
248 "RNE" => Mode.Rne, "RNA" => Mode.Rna, "RTZ" => Mode.Rtz, "RUP" => Mode.Rup, "RDN" => Mode.Rdn,
249 _ => throw new CommandError($"BAD MODE {words[index]}")
250 };
251
252 public static RealInput Real(string word)
253 {
254 if (word == "nan") return new(RealKind.NaN, new(0));
255 if (word is "inf" or "-inf") return new(RealKind.Infinity, new(0, Negative: word == "-inf"));
256 if (word.StartsWith("0b", StringComparison.Ordinal) || word.StartsWith("0x", StringComparison.Ordinal)) throw new CommandError($"BAD OPERAND {word}");
257 bool negative = word[0] == '-';
258 string body = negative ? word[1..] : word;
259 int slash = body.IndexOf('/');
260 if (slash >= 0)
261 {
262 long n = long.Parse(body[..slash]), d = long.Parse(body[(slash + 1)..]);
263 if (d == 0) throw new CommandError($"BAD OPERAND {word}");
264 return new(RealKind.Finite, new(n, Denominator: d, Negative: negative));
265 }
266 int dot = body.IndexOf('.');
267 long denominator = 1;
268 if (dot >= 0)
269 {
270 for (int i = dot + 1; i < body.Length; i++) denominator *= 10;
271 body = body.Remove(dot, 1);
272 }
273 return new(RealKind.Finite, new(long.Parse(body), Denominator: denominator, Negative: negative));
274 }
275
276 public static Result Encoded(MiniFormat format, RealInput input, Mode mode) => input.Kind switch
277 {
278 RealKind.NaN => new(format.NaN),
279 RealKind.Infinity => new(format.Inf(input.Number.Negative)),
280 _ => format.Round(input.Number, mode)
281 };
282
283 public static int Operand(MiniFormat format, string word)
284 {
285 bool bits = word.StartsWith("0b", StringComparison.Ordinal), hex = word.StartsWith("0x", StringComparison.Ordinal);
286 if (bits || hex)
287 {
288 if (word.Length - 2 != (bits ? format.Width : (format.Width + 3) / 4)) throw new CommandError($"BAD OPERAND {word}");
289 int pattern = Convert.ToInt32(word[2..], bits ? 2 : 16);
290 if (pattern >= 1 << format.Width) throw new CommandError($"BAD OPERAND {word}");
291 return pattern;
292 }
293 return Encoded(format, Real(word), Mode.Rne).Pattern;
294 }
295}

297static class Arithmetic
298{
299 public static Result Apply(MiniFormat format, Operation operation, Mode mode, params int[] inputs)
300 {
301 if (inputs.Any(format.IsNaN)) return new(format.NaN, inputs.Any(format.IsSignaling) ? FpFlags.Invalid : FpFlags.None);
302 int a = inputs[0];
303 bool negative = format.Fields(a).Negative;
304 var invalid = new Result(format.NaN, FpFlags.Invalid);
305 if (operation == Operation.Sqrt)
306 {
307 if (format.IsZero(a)) return new(a);
308 if (negative) return invalid;
309 if (format.IsInfinity(a)) return new(a);
310 return format.Sqrt(format.Exact(a), mode);
311 }
312 int b = operation == Operation.Sub ? format.Flip(inputs[1]) : inputs[1];
313 bool sigma = negative != format.Fields(b).Negative;
314 if (operation is Operation.Add or Operation.Sub)
315 {
316 if (format.IsInfinity(a) && format.IsInfinity(b) && sigma) return invalid;
317 if (format.IsInfinity(a)) return new(a);
318 if (format.IsInfinity(b)) return new(b);
319 return format.Round(Exact.Sum([format.Exact(a), format.Exact(b)], mode), mode);
320 }
321 if (operation == Operation.Mul)
322 {
323 if (format.IsInfinity(a) || format.IsInfinity(b)) return format.IsZero(a) || format.IsZero(b) ? invalid : new(format.Inf(sigma));
324 if (format.IsZero(a) || format.IsZero(b)) return new(format.Zero(sigma));
325 return format.Round(Exact.Product(format.Exact(a), format.Exact(b)), mode);
326 }
327 if (operation == Operation.Div)
328 {
329 if (format.IsInfinity(a)) return format.IsInfinity(b) ? invalid : new(format.Inf(sigma));
330 if (format.IsInfinity(b)) return new(format.Zero(sigma));
331 if (format.IsZero(b)) return format.IsZero(a) ? invalid : new(format.Inf(sigma), FpFlags.DivByZero);
332 if (format.IsZero(a)) return new(format.Zero(sigma));
333 var x = format.Exact(a);
334 var y = format.Exact(b);
335 return format.Round(new(x.Numerator * y.Denominator, x.Exponent - y.Exponent, x.Denominator * y.Numerator, sigma), mode);
336 }
337 int c = inputs[2];
338 if (format.IsInfinity(a) || format.IsInfinity(b))
339 {
340 if (format.IsZero(a) || format.IsZero(b) || format.IsInfinity(c) && format.Fields(c).Negative != sigma) return invalid;
341 return new(format.Inf(sigma));
342 }
343 if (format.IsInfinity(c)) return new(c);
344 return format.Round(Exact.Sum([Exact.Product(format.Exact(a), format.Exact(b)), format.Exact(c)], mode), mode);
345 }
346
347 public static int NextUp(MiniFormat format, int pattern) => pattern == format.Infinity ? pattern
348 : format.IsZero(pattern) ? 1 : pattern + (format.Fields(pattern).Negative ? -1 : 1);
349
350 public static string Difference(MiniFormat format, int pattern, Exact exact) => !format.IsFinite(pattern) ? format.Value(pattern)
351 : Exact.Sum([format.Exact(pattern), exact.Negated()], Mode.Rne).Fraction();
352}

354sealed class Laboratory
355{
356 readonly Dictionary<string, MiniFormat> formats = new(StringComparer.Ordinal)
357 {
358 ["HALF"] = new("HALF", 5, 10), ["E5M2"] = new("E5M2", 5, 2), ["E4M3"] = new("E4M3", 4, 3)
359 };
360 static readonly Dictionary<string, (long Minimum, long Maximum)> integerTypes = new(StringComparer.Ordinal)
361 {
362 ["I8"] = (-128, 127), ["U8"] = (0, 255), ["I16"] = (-32768, 32767), ["U16"] = (0, 65535),
363 ["I32"] = (-2147483648L, 2147483647L), ["U32"] = (0, 4294967295L)
364 };
365 static readonly Dictionary<string, Operation> operations = new(StringComparer.Ordinal)
366 {
367 ["ADD"] = Operation.Add, ["SUB"] = Operation.Sub, ["MUL"] = Operation.Mul,
368 ["DIV"] = Operation.Div, ["SQRT"] = Operation.Sqrt, ["FMA"] = Operation.Fma
369 };
370 readonly List<string> output = [];
371 void Emit(string line) => output.Add(line);
372 MiniFormat Format(string name) => formats.TryGetValue(name, out var format) ? format : throw new CommandError($"UNKNOWN FORMAT {name}");
373
374 static int[] Operands(MiniFormat format, string[] words, int begin, int end)
375 {
376 int[] patterns = new int[end - begin];
377 for (int i = begin; i < end; i++) patterns[i - begin] = Parsing.Operand(format, words[i]);
378 return patterns;
379 }
380
381 void Summary(MiniFormat format, Mode mode, int[] patterns, Result result)
382 {
383 var exact = Exact.Sum(patterns.Select(format.Exact).ToArray(), mode);
384 Emit($"RESULT {format.ResultText(result)}");
385 Emit($"EXACT {exact.Decimal()}");
386 Emit($"ROUNDED {format.ResultText(format.Round(exact, mode))}");
387 Emit($"DIFF {Arithmetic.Difference(format, result.Pattern, exact)}");
388 }
389
390 void Summation(string command, MiniFormat format, Mode mode, int[] patterns)
391 {
392 if (!patterns.All(format.IsFinite)) throw new CommandError("NOT FINITE");
393 Emit($"{command} {format.Name} {Notation.ModeText(mode)} count={patterns.Length}");
394 int sum = patterns[0], compensation = 0;
395 var flags = FpFlags.None;
396 for (int i = 1; i < patterns.Length; i++)
397 {
398 if (command == "SUM")
399 {
400 var added = Arithmetic.Apply(format, Operation.Add, mode, sum, patterns[i]);
401 sum = added.Pattern;
402 flags |= added.Flags;
403 Emit($"PARTIAL {i + 1} {format.ResultText(added)}");
404 }
405 else
406 {
407 var y = Arithmetic.Apply(format, Operation.Sub, mode, patterns[i], compensation);
408 var t = Arithmetic.Apply(format, Operation.Add, mode, sum, y.Pattern);
409 var d = Arithmetic.Apply(format, Operation.Sub, mode, t.Pattern, sum);
410 var c = Arithmetic.Apply(format, Operation.Sub, mode, d.Pattern, y.Pattern);
411 sum = t.Pattern;
412 compensation = c.Pattern;
413 var step = y.Flags | t.Flags | d.Flags | c.Flags;
414 flags |= step;
415 Emit($"KSTEP {i + 1} sum={format.Value(sum)} comp={format.Value(compensation)} {Notation.FlagsText(step)}");
416 }
417 }
418 Summary(format, mode, patterns, new(sum, flags));
419 }

421 void Dot(MiniFormat format, Mode mode, int[] a, int[] b)
422 {
423 if (a.Length != b.Length) throw new CommandError("LENGTH MISMATCH");
424 if (!a.All(format.IsFinite) || !b.All(format.IsFinite)) throw new CommandError("NOT FINITE");
425 var naive = Arithmetic.Apply(format, Operation.Mul, mode, a[0], b[0]);
426 var chained = naive;
427 for (int i = 1; i < a.Length; i++)
428 {
429 var product = Arithmetic.Apply(format, Operation.Mul, mode, a[i], b[i]);
430 var added = Arithmetic.Apply(format, Operation.Add, mode, naive.Pattern, product.Pattern);
431 naive = new(added.Pattern, naive.Flags | product.Flags | added.Flags);
432 var fused = Arithmetic.Apply(format, Operation.Fma, mode, a[i], b[i], chained.Pattern);
433 chained = new(fused.Pattern, chained.Flags | fused.Flags);
434 }
435 var terms = new Exact[a.Length];
436 for (int i = 0; i < a.Length; i++) terms[i] = Exact.Product(format.Exact(a[i]), format.Exact(b[i]));
437 var exact = Exact.Sum(terms, mode);
438 Emit($"DOT {format.Name} {Notation.ModeText(mode)} n={a.Length}");
439 Emit($"NAIVE {format.ResultText(naive)}");
440 Emit($"CHAINED {format.ResultText(chained)}");
441 Emit($"EXACT {exact.Decimal()}");
442 Emit($"FUSED {format.ResultText(format.Round(exact, mode))}");
443 Emit($"DIFF naive={Arithmetic.Difference(format, naive.Pattern, exact)} chained={Arithmetic.Difference(format, chained.Pattern, exact)}");
444 }
445
446 void Table(MiniFormat format)
447 {
448 if (format.Width > 8) throw new CommandError("TABLE TOO LARGE");
449 int count = format.Infinity;
450 Emit($"TABLE {format.Name} count={count}");
451 for (int pattern = 0; pattern < count; pattern++)
452 {
453 string gap = pattern == count - 1 ? "-" : Exact.Sum([format.Exact(pattern + 1), format.Exact(pattern).Negated()], Mode.Rne).Decimal();
454 Emit($"ENTRY {pattern} {format.PatternText(pattern)} gap={gap}");
455 }
456 Emit($"MAX {format.PatternText(format.Maximum)}");
457 Emit($"MINNORMAL {format.PatternText(format.Hidden)}");
458 Emit($"MINSUBNORMAL {format.PatternText(1)}");
459 Emit($"EPSILON {new Exact(1, -format.M).Decimal()}");
460 Emit($"PATTERNS finite={2 * count} infinite=2 nan={2 * (format.Hidden - 1)} total={1 << format.Width}");
461 }

463 public IReadOnlyList<string> Execute(string line)
464 {
465 output.Clear();
466 string[] words = line.Split(' ');
467 string command = words[0];
468 if (command == "FORMAT")
469 {
470 string name = words[1];
471 int e = int.Parse(words[2]), m = int.Parse(words[3]);
472 if (formats.ContainsKey(name)) throw new CommandError($"FORMAT EXISTS {name}");
473 if (e < 2 || e > 5 || m < 1 || m > 10) throw new CommandError("BAD FORMAT");
474 var format = new MiniFormat(name, e, m);
475 formats.Add(name, format);
476 Emit($"FORMAT {name} e={e} m={m} bits={format.Width} bias={format.Bias} emin={format.MinExponent} emax={format.MaxExponent}");
477 return output;
478 }
479 var f = Format(words[1]);
480 if (command == "ENCODE")
481 {
482 var input = Parsing.Real(words[2]);
483 var mode = Parsing.ModeAt(words, 3);
484 Emit($"ENCODE {f.Name} {Notation.ModeText(mode)} {input.Text} {f.ResultText(Parsing.Encoded(f, input, mode))}");
485 }
486 else if (command == "CONVERT")
487 {
488 var g = Format(words[2]);
489 int a = Parsing.Operand(f, words[3]);
490 var mode = Parsing.ModeAt(words, 4);
491 Result result = f.IsNaN(a) ? new(g.NaN, f.IsSignaling(a) ? FpFlags.Invalid : FpFlags.None)
492 : f.IsInfinity(a) ? new(g.Inf(f.Fields(a).Negative)) : g.Round(f.Exact(a), mode);
493 Emit($"CONVERT {f.Name} {g.Name} {Notation.ModeText(mode)} {g.ResultText(result)}");
494 }
495 else if (command is "SUM" or "KAHAN" or "DOT")
496 {
497 var mode = Parsing.ModeAt(words, 2);
498 if (command == "DOT")
499 {
500 int separator = Array.IndexOf(words, "|");
501 var a = Operands(f, words, 3, separator);
502 var b = Operands(f, words, separator + 1, words.Length);
503 Dot(f, mode, a, b);
504 }
505 else Summation(command, f, mode, Operands(f, words, 3, words.Length));
506 }
507 else if (command == "TABLE") Table(f);
508 else if (operations.TryGetValue(command, out var op))
509 {
510 int count = op == Operation.Sqrt ? 1 : op == Operation.Fma ? 3 : 2;
511 var patterns = Operands(f, words, 2, 2 + count);
512 var mode = Parsing.ModeAt(words, 2 + count);
513 Emit($"{command} {f.Name} {Notation.ModeText(mode)} {f.ResultText(Arithmetic.Apply(f, op, mode, patterns))}");
514 }
515 else
516 {
517 int a = Parsing.Operand(f, words[2]);
518 var fields = f.Fields(a);
519 if (command == "DECODE")
520 {
521 string classification = f.IsNaN(a) ? (f.IsSignaling(a) ? "snan" : "qnan") : f.IsInfinity(a) ? "infinity"
522 : f.IsZero(a) ? "zero" : fields.Exponent == 0 ? "subnormal" : "normal";
523 Emit($"DECODE {f.Name} {f.Bits(a)} {f.Hex(a)} {classification} sign={(fields.Negative ? 1 : 0)}");
524 string heading = $"FIELDS exponent={fields.Exponent} fraction={fields.Fraction}";
525 if (f.IsFinite(a))
526 {
527 Emit($"{heading} unbiased={fields.Q + f.M} significand={(fields.Exponent == 0 ? 0 : 1)}.{Notation.Binary(fields.Fraction, f.M)}");
528 Emit($"VALUE {f.Exact(a).Fraction()} {f.Value(a)}");
529 }
530 else Emit(f.IsNaN(a) ? $"{heading} payload={fields.Fraction % (f.Hidden / 2)}" : heading);
531 }
532 else if (command == "COMPARE")
533 {
534 int b = Parsing.Operand(f, words[3]);
535 int Key(int pattern) => f.Fields(pattern).Negative ? -(pattern - f.SignBit) - 1 : pattern;
536 string order = Notation.Relation(Key(a), Key(b));
537 string numeric = f.IsNaN(a) || f.IsNaN(b) ? "UN" : f.IsZero(a) && f.IsZero(b) ? "EQ" : order;
538 var flags = f.IsSignaling(a) || f.IsSignaling(b) ? FpFlags.Invalid : FpFlags.None;
539 Emit($"COMPARE {f.Name} {f.Hex(a)} {f.Hex(b)} {numeric} total={order} {Notation.FlagsText(flags)}");
540 }
541 else if (command == "TOINT")
542 {
543 if (!integerTypes.TryGetValue(words[3], out var range)) throw new CommandError($"BAD TYPE {words[3]}");
544 var mode = Parsing.ModeAt(words, 4);
545 string prefix = $"TOINT {f.Name} {words[3]} {Notation.ModeText(mode)}";
546 if (!f.IsFinite(a)) Emit($"{prefix} invalid flags=invalid");
547 else
548 {
549 var units = RoundUnits(f.Exact(a), 0, mode);
550 long integer = fields.Negative ? -units.Integer : units.Integer;
551 Emit(integer < range.Minimum || integer > range.Maximum ? $"{prefix} invalid flags=invalid"
552 : $"{prefix} {integer} {Notation.FlagsText(units.Inexact ? FpFlags.Inexact : FpFlags.None)}");
553 }
554 }
555 else
556 {
557 Result result;
558 if (f.IsNaN(a)) result = new(f.NaN, f.IsSignaling(a) ? FpFlags.Invalid : FpFlags.None);
559 else if (command == "NEXTUP") result = new(Arithmetic.NextUp(f, a));
560 else if (command == "NEXTDOWN") result = new(f.Flip(Arithmetic.NextUp(f, f.Flip(a))));
561 else result = new(f.IsInfinity(a) ? f.Infinity : f.Round(new(1, fields.Q), Mode.Rne).Pattern);
562 Emit($"{command} {f.Name} {f.ResultText(result)}");
563 }
564 }
565 return output;
566 }
567}
