24function gcd(a, b) {
25 while (b !== 0n) [a, b] = [b, a % b];
26 return a;

29function scaled(numerator, denominator, exponent) {
30 return exponent >= 0
31 ? [numerator << BigInt(exponent), denominator]
32 : [numerator, denominator << BigInt(-exponent)];
33}

35function integerSqrt(value) {
36 if (value === 0n) return 0n;
37 let root = 1n << BigInt((value.toString(2).length + 1) >> 1);
38 while (true) {
39 const next = (root + value / root) >> 1n;
40 if (next >= root) return root;
41 root = next;
42 }
43}

45class Exact {
46 constructor(numerator, exponent = 0, denominator = 1n, sign = 0) {
47 this.n = numerator;
48 this.e = exponent;
49 this.d = denominator;
50 this.sign = sign;
51 }
52
53 negated() { return new Exact(this.n, this.e, this.d, this.sign ^ 1); }
54
55 log2Floor() {
56 let result = this.n.toString(2).length - this.d.toString(2).length + this.e;
57 const [n, d] = scaled(this.n, this.d, this.e - result);
58 if (n < d) result--;
59 return result;
60 }
61
62 fraction() {
63 if (this.n === 0n) return '0';
64 let [n, d] = scaled(this.n, this.d, this.e);
65 const factor = gcd(n, d);
66 n /= factor;
67 d /= factor;
68 return (this.sign ? '-' : '') + n + (d === 1n ? '' : `/${d}`);
69 }
70
71 decimal(keepZeroSign = false) {
72 const [n, d] = scaled(this.n, this.d, this.e);
73 let text = String(n / d);
74 let remainder = n % d;
75 if (remainder !== 0n) {
76 text += '.';
77 while (remainder !== 0n) {
78 remainder *= 10n;
79 text += remainder / d;
80 remainder %= d;
81 }
82 }
83 return (this.sign && (this.n !== 0n || keepZeroSign) ? '-' : '') + text;
84 }
85}

87function exactProduct(a, b) {
88 return new Exact(a.n * b.n, a.e + b.e, a.d * b.d, a.sign ^ b.sign);
89}

91function exactSum(terms, mode) {
92 const nonzero = terms.filter(term => term.n !== 0n);
93 if (nonzero.length === 0) {
94 const sameSign = terms.every(term => term.sign === terms[0].sign);
95 return new Exact(0n, 0, 1n, sameSign ? terms[0].sign : Number(mode === 'RDN'));
96 }
97 const exponent = Math.min(...nonzero.map(term => term.e));
98 let numerator = 0n;
99 for (const term of nonzero) {
100 const aligned = term.n << BigInt(term.e - exponent);
101 numerator += term.sign ? -aligned : aligned;
102 }
103 const sign = numerator < 0n ? 1 : numerator === 0n ? Number(mode === 'RDN') : 0;
104 return new Exact(numerator < 0n ? -numerator : numerator, exponent, 1n, sign);
105}

107function increment(mode, sign, integer, inexact, halfway) {
108 if (!inexact) return false;
109 if (mode === 'RNE') return halfway > 0n || halfway === 0n && integer % 2n === 1n;
110 if (mode === 'RNA') return halfway >= 0n;
111 if (mode === 'RUP') return sign === 0;
112 if (mode === 'RDN') return sign === 1;
113 return false;
114}

116function roundedUnits(value, exponent, mode) {
117 const [n, d] = scaled(value.n, value.d, value.e - exponent);
118 let integer = n / d;
119 const remainder = n % d;
120 const inexact = remainder !== 0n;
121 if (increment(mode, value.sign, integer, inexact, 2n * remainder - d)) integer++;
122 return { integer, inexact };
123}
124
125function roundedSqrtUnits(value, exponent, mode) {
126 const [n, d] = scaled(value.n, value.d, value.e - 2 * exponent);
127 let integer = integerSqrt(n / d);
128 const inexact = integer * integer * d !== n;
129 const twiceMidpoint = 2n * integer + 1n;
130 const halfway = 4n * n - d * twiceMidpoint * twiceMidpoint;
131 if (increment(mode, 0, integer, inexact, halfway)) integer++;
132 return { integer, inexact };
133}

135class Format {
136 constructor(name, e, m) {
137 this.name = name;
138 this.e = e;
139 this.m = m;
140 this.w = 1 + e + m;
141 this.bias = (1 << (e - 1)) - 1;
142 this.emin = 1 - this.bias;
143 this.emax = this.bias;
144 this.hidden = 1 << m;
145 this.exponentMask = (1 << e) - 1;
146 this.signBit = 1 << (e + m);
147 this.infinity = this.exponentMask * this.hidden;
148 this.nan = this.infinity + (this.hidden >> 1);
149 this.maximum = this.infinity - 1;
150 }
151
152 fields(pattern) {
153 const sign = pattern & this.signBit ? 1 : 0;
154 const exponent = (pattern >> this.m) & this.exponentMask;
155 const fraction = pattern & (this.hidden - 1);
156 const q = (exponent === 0 ? this.emin : exponent - this.bias) - this.m;
157 return { sign, exponent, fraction, q };
158 }
159
160 isNaN(pattern) { const f = this.fields(pattern); return f.exponent === this.exponentMask && f.fraction !== 0; }
161 isSignaling(pattern) { return this.isNaN(pattern) && this.fields(pattern).fraction < (this.hidden >> 1); }
162 isInfinity(pattern) { return (pattern & (this.signBit - 1)) === this.infinity; }
163 isZero(pattern) { return (pattern & (this.signBit - 1)) === 0; }
164 isFinite(pattern) { return this.fields(pattern).exponent !== this.exponentMask; }
165 zero(sign) { return sign * this.signBit; }
166 inf(sign) { return this.zero(sign) + this.infinity; }
167 flip(pattern) { return pattern ^ this.signBit; }
168
169 exact(pattern) {
170 const f = this.fields(pattern);
171 return new Exact(BigInt(f.exponent === 0 ? f.fraction : this.hidden + f.fraction), f.q, 1n, f.sign);
172 }
173
174 bits(pattern) {
175 const f = this.fields(pattern);
176 return `${f.sign}|${f.exponent.toString(2).padStart(this.e, '0')}|${f.fraction.toString(2).padStart(this.m, '0')}`;
177 }
178
179 hex(pattern) { return '0x' + pattern.toString(16).toUpperCase().padStart((this.w + 3) >> 2, '0'); }
180
181 value(pattern) {
182 if (this.isNaN(pattern)) return 'nan';
183 if (this.isInfinity(pattern)) return this.fields(pattern).sign ? '-inf' : 'inf';
184 return this.exact(pattern).decimal(true);
185 }
186
187 patternText(pattern) { return `${this.bits(pattern)} ${this.hex(pattern)} ${this.value(pattern)}`; }
188 resultText(result) { return `${this.patternText(result.pattern)} ${flagsText(result.flags)}`; }
189
190 overflow(sign, mode) {
191 const infinity = mode === 'RNE' || mode === 'RNA' || mode === 'RUP' && sign === 0 || mode === 'RDN' && sign === 1;
192 return { pattern: this.zero(sign) + (infinity ? this.infinity : this.maximum), flags: OVERFLOW | INEXACT };
193 }
194
195 finishRound(sign, t, exponent, units, unbounded, mode) {
196 const hidden = BigInt(this.hidden);
197 let n = units.integer;
198 if (n === 2n * hidden) {
199 n = hidden;
200 exponent++;
201 }
202 if (exponent > this.emax - this.m) return this.overflow(sign, mode);
203 const magnitude = n >= hidden
204 ? (exponent + this.m + this.bias) * this.hidden + Number(n - hidden)
205 : Number(n);
206 let flags = units.inexact ? INEXACT : 0;
207 if (units.inexact && t < this.emin && (t < this.emin - 1 || unbounded().integer < 2n * hidden)) {
208 flags |= UNDERFLOW;
209 }
210 return { pattern: this.zero(sign) + magnitude, flags };
211 }
212
213 round(value, mode) {
214 if (value.n === 0n) return { pattern: this.zero(value.sign), flags: 0 };
215 const t = value.log2Floor();
216 if (t > this.emax) return this.overflow(value.sign, mode);
217 const exponent = Math.max(t, this.emin) - this.m;
218 return this.finishRound(value.sign, t, exponent, roundedUnits(value, exponent, mode),
219 () => roundedUnits(value, t - this.m, mode), mode);
220 }
221
222 sqrt(value, mode) {
223 const log = value.log2Floor();
224 const t = log >= 0 ? log >> 1 : -((-log + 1) >> 1);
225 const exponent = Math.max(t, this.emin) - this.m;
226 return this.finishRound(0, t, exponent, roundedSqrtUnits(value, exponent, mode),
227 () => roundedSqrtUnits(value, t - this.m, mode), mode);
228 }
229}

231function parseReal(word) {
232 if (word === 'nan' || word === 'inf' || word === '-inf') return { special: word };
233 if (word.startsWith('0b') || word.startsWith('0x')) fail(`BAD OPERAND ${word}`);
234 const sign = word.startsWith('-') ? 1 : 0;
235 const body = sign ? word.slice(1) : word;
236 if (body.includes('/')) {
237 const [p, q] = body.split('/').map(BigInt);
238 if (q === 0n) fail(`BAD OPERAND ${word}`);
239 return { exact: new Exact(p, 0, q, sign) };
240 }
241 const [whole, fraction = ''] = body.split('.');
242 return { exact: new Exact(BigInt(whole + fraction), 0, 10n ** BigInt(fraction.length), sign) };
243}

245function encoded(format, real, mode) {
246 if (real.special === 'nan') return { pattern: format.nan, flags: 0 };
247 if (real.special) return { pattern: format.inf(Number(real.special === '-inf')), flags: 0 };
248 return format.round(real.exact, mode);
249}

251function operand(format, word) {
252 if (word.startsWith('0b')) {
253 if (word.length - 2 !== format.w) fail(`BAD OPERAND ${word}`);
254 return Number(BigInt(word));
255 }
256 if (word.startsWith('0x')) {
257 const pattern = Number(BigInt(word));
258 if (word.length - 2 !== ((format.w + 3) >> 2) || pattern >= (1 << format.w)) fail(`BAD OPERAND ${word}`);
259 return pattern;
260 }
261 return encoded(format, parseReal(word), 'RNE').pattern;
262}

264function operation(format, command, inputs, mode) {
265 if (command === 'SUB') return operation(format, 'ADD', [inputs[0], format.flip(inputs[1])], mode);
266 if (inputs.some(pattern => format.isNaN(pattern))) {
267 return { pattern: format.nan, flags: inputs.some(pattern => format.isSignaling(pattern)) ? INVALID : 0 };
268 }
269 const [a, b, c] = inputs;
270 const result = (pattern, flags = 0) => ({ pattern, flags });
271 const invalid = () => result(format.nan, INVALID);
272 const aSign = format.fields(a).sign;
273 if (command === 'SQRT') {
274 if (format.isZero(a)) return result(a);
275 if (aSign) return invalid();
276 if (format.isInfinity(a)) return result(a);
277 return format.sqrt(format.exact(a), mode);
278 }
279 const sigma = aSign ^ format.fields(b).sign;
280 if (command === 'ADD') {
281 if (format.isInfinity(a) && format.isInfinity(b) && sigma) return invalid();
282 if (format.isInfinity(a)) return result(a);
283 if (format.isInfinity(b)) return result(b);
284 return format.round(exactSum([format.exact(a), format.exact(b)], mode), mode);
285 }
286 if (command === 'MUL') {
287 if (format.isInfinity(a) || format.isInfinity(b)) {
288 return format.isZero(a) || format.isZero(b) ? invalid() : result(format.inf(sigma));
289 }
290 if (format.isZero(a) || format.isZero(b)) return result(format.zero(sigma));
291 return format.round(exactProduct(format.exact(a), format.exact(b)), mode);
292 }
293 if (command === 'DIV') {
294 if (format.isInfinity(a)) return format.isInfinity(b) ? invalid() : result(format.inf(sigma));
295 if (format.isInfinity(b)) return result(format.zero(sigma));
296 if (format.isZero(b)) return format.isZero(a) ? invalid() : result(format.inf(sigma), DIVBYZERO);
297 if (format.isZero(a)) return result(format.zero(sigma));
298 const x = format.exact(a), y = format.exact(b);
299 return format.round(new Exact(x.n * y.d, x.e - y.e, x.d * y.n, sigma), mode);
300 }
301 if (format.isInfinity(a) || format.isInfinity(b)) {
302 if (format.isZero(a) || format.isZero(b) || format.isInfinity(c) && format.fields(c).sign !== sigma) return invalid();
303 return result(format.inf(sigma));
304 }
305 if (format.isInfinity(c)) return result(c);
306 return format.round(exactSum([exactProduct(format.exact(a), format.exact(b)), format.exact(c)], mode), mode);
307}

309function nextUp(format, pattern) {
310 if (pattern === format.infinity) return pattern;
311 if (format.isZero(pattern)) return 1;
312 return pattern + (format.fields(pattern).sign ? -1 : 1);
313}

315function difference(format, pattern, exact) {
316 if (!format.isFinite(pattern)) return format.value(pattern);
317 return exactSum([format.exact(pattern), exact.negated()], 'RNE').fraction();
318}

322class Laboratory {
323 constructor() {
324 this.formats = new Map([['HALF', new Format('HALF', 5, 10)], ['E5M2', new Format('E5M2', 5, 2)], ['E4M3', new Format('E4M3', 4, 3)]]);
325 this.output = [];
326 }
327
328 emit(line) { this.output.push(line); }
329
330 format(name) {
331 if (!this.formats.has(name)) fail(`UNKNOWN FORMAT ${name}`);
332 return this.formats.get(name);
333 }
334
335 summary(format, mode, patterns, pattern, flags) {
336 const exact = exactSum(patterns.map(pattern => format.exact(pattern)), mode);
337 this.emit(`RESULT ${format.resultText({ pattern, flags })}`);
338 this.emit(`EXACT ${exact.decimal()}`);
339 this.emit(`ROUNDED ${format.resultText(format.round(exact, mode))}`);
340 this.emit(`DIFF ${difference(format, pattern, exact)}`);
341 }
342
343 summation(command, format, mode, patterns) {
344 if (patterns.some(pattern => !format.isFinite(pattern))) fail('NOT FINITE');
345 this.emit(`${command} ${format.name} ${mode} count=${patterns.length}`);
346 let sum = patterns[0], compensation = 0, flags = 0;
347 for (let i = 1; i < patterns.length; i++) {
348 if (command === 'SUM') {
349 const added = operation(format, 'ADD', [sum, patterns[i]], mode);
350 sum = added.pattern;
351 flags |= added.flags;
352 this.emit(`PARTIAL ${i + 1} ${format.resultText(added)}`);
353 } else {
354 const y = operation(format, 'SUB', [patterns[i], compensation], mode);
355 const t = operation(format, 'ADD', [sum, y.pattern], mode);
356 const d = operation(format, 'SUB', [t.pattern, sum], mode);
357 const c = operation(format, 'SUB', [d.pattern, y.pattern], mode);
358 sum = t.pattern;
359 compensation = c.pattern;
360 const stepFlags = y.flags | t.flags | d.flags | c.flags;
361 flags |= stepFlags;
362 this.emit(`KSTEP ${i + 1} sum=${format.value(sum)} comp=${format.value(compensation)} ${flagsText(stepFlags)}`);
363 }
364 }
365 this.summary(format, mode, patterns, sum, flags);
366 }

368 dot(format, mode, a, b) {
369 if (a.length !== b.length) fail('LENGTH MISMATCH');
370 if ([...a, ...b].some(pattern => !format.isFinite(pattern))) fail('NOT FINITE');
371 let naive = operation(format, 'MUL', [a[0], b[0]], mode);
372 let chained = { ...naive };
373 for (let i = 1; i < a.length; i++) {
374 const product = operation(format, 'MUL', [a[i], b[i]], mode);
375 const added = operation(format, 'ADD', [naive.pattern, product.pattern], mode);
376 naive = { pattern: added.pattern, flags: naive.flags | product.flags | added.flags };
377 const fused = operation(format, 'FMA', [a[i], b[i], chained.pattern], mode);
378 chained = { pattern: fused.pattern, flags: chained.flags | fused.flags };
379 }
380 const exact = exactSum(a.map((pattern, i) => exactProduct(format.exact(pattern), format.exact(b[i]))), mode);
381 this.emit(`DOT ${format.name} ${mode} n=${a.length}`);
382 this.emit(`NAIVE ${format.resultText(naive)}`);
383 this.emit(`CHAINED ${format.resultText(chained)}`);
384 this.emit(`EXACT ${exact.decimal()}`);
385 this.emit(`FUSED ${format.resultText(format.round(exact, mode))}`);
386 this.emit(`DIFF naive=${difference(format, naive.pattern, exact)} chained=${difference(format, chained.pattern, exact)}`);
387 }
388
389 table(format) {
390 if (format.w > 8) fail('TABLE TOO LARGE');
391 const count = format.infinity;
392 this.emit(`TABLE ${format.name} count=${count}`);
393 for (let pattern = 0; pattern < count; pattern++) {
394 const gap = pattern === count - 1 ? '-'
395 : exactSum([format.exact(pattern + 1), format.exact(pattern).negated()], 'RNE').decimal();
396 this.emit(`ENTRY ${pattern} ${format.patternText(pattern)} gap=${gap}`);
397 }
398 this.emit(`MAX ${format.patternText(format.maximum)}`);
399 this.emit(`MINNORMAL ${format.patternText(format.hidden)}`);
400 this.emit(`MINSUBNORMAL ${format.patternText(1)}`);
401 this.emit(`EPSILON ${new Exact(1n, -format.m).decimal()}`);
402 this.emit(`PATTERNS finite=${2 * count} infinite=2 nan=${2 * (format.hidden - 1)} total=${1 << format.w}`);
403 }

405 command(line) {
406 const words = line.split(' ');
407 const command = words[0];
408 if (command === 'FORMAT') {
409 const name = words[1], e = Number(words[2]), m = Number(words[3]);
410 if (this.formats.has(name)) fail(`FORMAT EXISTS ${name}`);
411 if (e < 2 || e > 5 || m < 1 || m > 10) fail('BAD FORMAT');
412 const format = new Format(name, e, m);
413 this.formats.set(name, format);
414 this.emit(`FORMAT ${name} e=${e} m=${m} bits=${format.w} bias=${format.bias} emin=${format.emin} emax=${format.emax}`);
415 return;
416 }
417
418 const format = this.format(words[1]);
419 const name = format.name;
420 if (command === 'ENCODE') {
421 const real = parseReal(words[2]);
422 const mode = modeOf(words[3]);
423 const value = real.special ?? real.exact.fraction();
424 this.emit(`ENCODE ${name} ${mode} ${value} ${format.resultText(encoded(format, real, mode))}`);
425 } else if (command === 'CONVERT') {
426 const target = this.format(words[2]);
427 const a = operand(format, words[3]);
428 const mode = modeOf(words[4]);
429 let result;
430 if (format.isNaN(a)) result = { pattern: target.nan, flags: format.isSignaling(a) ? INVALID : 0 };
431 else if (format.isInfinity(a)) result = { pattern: target.inf(format.fields(a).sign), flags: 0 };
432 else result = target.round(format.exact(a), mode);
433 this.emit(`CONVERT ${name} ${target.name} ${mode} ${target.resultText(result)}`);
434 } else if (command === 'SUM' || command === 'KAHAN' || command === 'DOT') {
435 const mode = modeOf(words[2]);
436 if (command === 'DOT') {
437 const separator = words.indexOf('|');
438 const a = words.slice(3, separator).map(word => operand(format, word));
439 const b = words.slice(separator + 1).map(word => operand(format, word));
440 this.dot(format, mode, a, b);
441 } else {
442 const patterns = words.slice(3).map(word => operand(format, word));
443 this.summation(command, format, mode, patterns);
444 }
445 } else if (command === 'TABLE') this.table(format);
446 else if (['ADD', 'SUB', 'MUL', 'DIV', 'SQRT', 'FMA'].includes(command)) {
447 const count = command === 'SQRT' ? 1 : command === 'FMA' ? 3 : 2;
448 const patterns = words.slice(2, 2 + count).map(word => operand(format, word));
449 const mode = modeOf(words[2 + count]);
450 this.emit(`${command} ${name} ${mode} ${format.resultText(operation(format, command, patterns, mode))}`);
451 } else {
452 const a = operand(format, words[2]);
453 const fields = format.fields(a);
454 if (command === 'DECODE') {
455 const classification = format.isNaN(a) ? format.isSignaling(a) ? 'snan' : 'qnan'
456 : format.isInfinity(a) ? 'infinity' : format.isZero(a) ? 'zero' : fields.exponent === 0 ? 'subnormal' : 'normal';
457 this.emit(`DECODE ${name} ${format.bits(a)} ${format.hex(a)} ${classification} sign=${fields.sign}`);
458 let line = `FIELDS exponent=${fields.exponent} fraction=${fields.fraction}`;
459 if (format.isFinite(a)) {
460 line += ` unbiased=${fields.q + format.m} significand=${fields.exponent === 0 ? 0 : 1}.${fields.fraction.toString(2).padStart(format.m, '0')}`;
461 this.emit(line);
462 this.emit(`VALUE ${format.exact(a).fraction()} ${format.value(a)}`);
463 } else {
464 if (format.isNaN(a)) line += ` payload=${fields.fraction % (format.hidden >> 1)}`;
465 this.emit(line);
466 }
467 } else if (command === 'COMPARE') {
468 const b = operand(format, words[3]);
469 const key = pattern => format.fields(pattern).sign ? -(pattern - format.signBit) - 1 : pattern;
470 const order = relation(key(a), key(b));
471 const numeric = format.isNaN(a) || format.isNaN(b) ? 'UN'
472 : format.isZero(a) && format.isZero(b) ? 'EQ' : order;
473 const flags = format.isSignaling(a) || format.isSignaling(b) ? INVALID : 0;
474 this.emit(`COMPARE ${name} ${format.hex(a)} ${format.hex(b)} ${numeric} total=${order} ${flagsText(flags)}`);
475 } else if (command === 'TOINT') {
476 const type = words[3];
477 if (!TYPES.has(type)) fail(`BAD TYPE ${type}`);
478 const mode = modeOf(words[4]);
479 const prefix = `TOINT ${name} ${type} ${mode}`;
480 if (!format.isFinite(a)) this.emit(`${prefix} invalid flags=invalid`);
481 else {
482 const units = roundedUnits(format.exact(a), 0, mode);
483 const integer = fields.sign ? -units.integer : units.integer;
484 const [minimum, maximum] = TYPES.get(type);
485 if (integer < minimum || integer > maximum) this.emit(`${prefix} invalid flags=invalid`);
486 else this.emit(`${prefix} ${integer} ${flagsText(units.inexact ? INEXACT : 0)}`);
487 }
488 } else {
489 let result;
490 if (format.isNaN(a)) result = { pattern: format.nan, flags: format.isSignaling(a) ? INVALID : 0 };
491 else if (command === 'NEXTUP') result = { pattern: nextUp(format, a), flags: 0 };
492 else if (command === 'NEXTDOWN') result = { pattern: format.flip(nextUp(format, format.flip(a))), flags: 0 };
493 else {
494 const pattern = format.isInfinity(a) ? format.infinity : format.round(new Exact(1n, fields.q), 'RNE').pattern;
495 result = { pattern, flags: 0 };
496 }
497 this.emit(`${command} ${name} ${format.resultText(result)}`);
498 }
499 }
500 }
