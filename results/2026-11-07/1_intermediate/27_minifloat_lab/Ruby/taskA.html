15Term = Data.define(:negative, :magnitude, :exponent) do
16 def product(other)
17 Term.new(negative != other.negative, magnitude * other.magnitude, exponent + other.exponent)
18 end
19
20 def negated = with(negative: !negative)
21 def exact = Exact.make(negative, magnitude, exponent)
22
23 def decimal
24 return '0' if magnitude.zero?
25
26 numerator, denominator = exact.scaled(0)
27 whole, remainder = numerator.divmod(denominator)
28 text = "#{negative ? '-' : ''}#{whole}"
29 text << '.' unless remainder.zero?
30 until remainder.zero?
31 digit, remainder = (remainder * 10).divmod(denominator)
32 text << (48 + digit).chr
33 end
34 text
35 end
36end

38def sum_terms(terms, mode)
39 exponent = terms.reject { |term| term.magnitude.zero? }.map(&:exponent).min || 0
40 sum = terms.sum do |term|
41 next 0 if term.magnitude.zero?
42
43 aligned = term.magnitude << (term.exponent - exponent)
44 term.negative ? -aligned : aligned
45 end
46 if sum.zero?
47 same_zero = terms.all? { |term| term.magnitude.zero? && term.negative == terms.first.negative }
48 return Term.new(same_zero ? terms.first.negative : mode == :RDN, 0, 0)
49 end
50 Term.new(sum.negative?, sum.abs, exponent)
51end

53Exact = Data.define(:negative, :numerator, :exponent, :denominator) do
54 def self.make(negative, numerator, exponent = 0, denominator = 1)
55 return new(negative, 0, 0, 1) if numerator.zero?
56
57 common = numerator.gcd(denominator)
58 numerator /= common
59 denominator /= common
60 while numerator.even?
61 numerator >>= 1
62 exponent += 1
63 end
64 while denominator.even?
65 denominator >>= 1
66 exponent -= 1
67 end
68 new(negative, numerator, exponent, denominator)
69 end
70
71 def scaled(quantum)
72 shift = exponent - quantum
73 shift >= 0 ? [numerator << shift, denominator] : [numerator, denominator << -shift]
74 end
75
76 def floor_log
77 power = numerator.bit_length - denominator.bit_length
78 below = power >= 0 ? numerator < (denominator << power) : (numerator << -power) < denominator
79 exponent + power - (below ? 1 : 0)
80 end
81
82 def fraction
83 return '0' if numerator.zero?
84
85 n, d = scaled(0)
86 "#{negative ? '-' : ''}#{n}#{d == 1 ? '' : "/#{d}"}"
87 end
88end

90Quantized = Data.define(:whole, :inexact, :half) do
91 def self.from(value, quantum, square_root)
92 numerator, denominator = value.scaled(square_root ? 2 * quantum : quantum)
93 unless square_root
94 whole, remainder = numerator.divmod(denominator)
95 return new(whole, !remainder.zero?, remainder <=> denominator - remainder)
96 end
97 whole = Integer.sqrt(numerator / denominator)
98 midpoint = 2 * whole + 1
99 new(whole, whole * whole * denominator != numerator, 4 * numerator <=> midpoint * midpoint * denominator)
100 end
101
102 def rounded(mode, negative)
103 increment = case mode
104 when :RNE then half.positive? || half.zero? && whole.odd?
105 when :RNA then half >= 0
106 when :RTZ then false
107 when :RUP then inexact && !negative
108 when :RDN then inexact && negative
109 end
110 whole + (increment ? 1 : 0)
111 end
112end

117class MiniFormat
118 attr_reader :name, :e, :m, :w, :bias, :emin, :emax, :sign, :base, :infinity, :nan
119
120 def initialize(name, e, m)
121 @name, @e, @m = name.freeze, e, m
122 @w = 1 + e + m
123 @bias = (1 << (e - 1)) - 1
124 @emin = 1 - bias
125 @emax = bias
126 @sign = 1 << (e + m)
127 @base = 1 << m
128 @infinity = ((1 << e) - 1) << m
129 @nan = infinity | (base / 2)
130 freeze
131 end
132
133 def negative?(pattern) = (pattern & sign) != 0
134 def zero?(pattern) = (pattern & (sign - 1)).zero?
135 def signed(pattern, negative) = pattern | (negative ? sign : 0)
136
137 def kind(pattern)
138 magnitude = pattern & (sign - 1)
139 fraction = pattern & (base - 1)
140 if magnitude >= infinity
141 return :infinity if fraction.zero?
142
143 return fraction >= base / 2 ? :qnan : :snan
144 end
145 return :zero if magnitude.zero?
146
147 magnitude < base ? :subnormal : :normal
148 end
149
150 def finite?(pattern) = %i[zero subnormal normal].include?(kind(pattern))
151 def nan?(pattern) = %i[qnan snan].include?(kind(pattern))
152
153 def term(pattern)
154 field = (pattern & (sign - 1)) >> m
155 fraction = pattern & (base - 1)
156 Term.new(negative?(pattern), field.zero? ? fraction : base + fraction,
157 field.zero? ? emin - m : field - bias - m)
158 end
159
160 def overflow(mode, negative)
161 to_infinity = %i[RNE RNA].include?(mode) || mode == :RUP && !negative || mode == :RDN && negative
162 Rounded.new(signed(to_infinity ? infinity : infinity - 1, negative), OVERFLOW | INEXACT)
163 end
164
165 def round(value, mode, square_root = false)
166 return Rounded.new(signed(0, value.negative), 0) if value.numerator.zero?
167
168 t = value.floor_log
169 t = t.div(2) if square_root
170 return overflow(mode, value.negative) if t > emax
171
172 quantum = [t, emin].max - m
173 divided = Quantized.from(value, quantum, square_root)
174 significand = divided.rounded(mode, value.negative)
175 if significand == 2 * base
176 significand = base
177 quantum += 1
178 end
179 return overflow(mode, value.negative) if quantum > emax - m
180
181 flags = divided.inexact ? INEXACT : 0
182 if divided.inexact && t < emin
183 tiny = t < emin - 1 || Quantized.from(value, t - m, square_root).rounded(mode, value.negative) < 2 * base
184 flags |= UNDERFLOW if tiny
185 end
186 magnitude = if significand < base
187 significand
188 else
189 ((quantum + m + bias) << m) | (significand - base)
190 end
191 Rounded.new(signed(magnitude, value.negative), flags)
192 end
193
194 def round_term(term, mode) = round(term.exact, mode)
195 def round_sum(terms, mode) = round_term(sum_terms(terms, mode), mode)
196
197 def arithmetic(operation, a, b, c, mode)
198 if operation == :SUB
199 b ^= sign
200 operation = :ADD
201 end
202 inputs = operation == :SQRT ? [a] : operation == :FMA ? [a, b, c] : [a, b]
203 if inputs.any? { |p| nan?(p) }
204 return Rounded.new(nan, inputs.any? { |p| kind(p) == :snan } ? INVALID : 0)
205 end
206 ai, bi = kind(a) == :infinity, kind(b) == :infinity
207 sigma = negative?(a) != negative?(b)
208 invalid = Rounded.new(nan, INVALID)
209 case operation
210 when :ADD
211 return invalid if ai && bi && sigma
212 return Rounded.new(ai ? a : b, 0) if ai || bi
213
214 round_sum([term(a), term(b)], mode)
215 when :MUL
216 return zero?(a) || zero?(b) ? invalid : Rounded.new(signed(infinity, sigma), 0) if ai || bi
217
218 round_term(term(a).product(term(b)), mode)
219 when :DIV
220 return bi ? invalid : Rounded.new(signed(infinity, sigma), 0) if ai
221 return Rounded.new(signed(0, sigma), 0) if bi
222 return zero?(a) ? invalid : Rounded.new(signed(infinity, sigma), DIVBYZERO) if zero?(b)
223 return Rounded.new(signed(0, sigma), 0) if zero?(a)
224
225 x, y = term(a), term(b)
226 round(Exact.make(sigma, x.magnitude, x.exponent - y.exponent, y.magnitude), mode)
227 when :SQRT
228 return Rounded.new(a, 0) if zero?(a)
229 return invalid if negative?(a)
230 return Rounded.new(a, 0) if ai
231
232 round(term(a).exact, mode, true)
233 when :FMA
234 if ai || bi
235 return invalid if zero?(a) || zero?(b) || kind(c) == :infinity && negative?(c) != sigma
236
237 return Rounded.new(signed(infinity, sigma), 0)
238 end
239 return Rounded.new(c, 0) if kind(c) == :infinity
240
241 round_sum([term(a).product(term(b)), term(c)], mode)
242 end
243 end

245 def raw_operand(word, encoding)
246 if word.start_with?('0b', '0x')
247 binary = word.start_with?('0b')
248 width = binary ? w : (w + 3) / 4
249 raise CommandError, "ERROR BAD OPERAND #{word}" if encoding || word.length - 2 != width
250
251 pattern = word[2..].to_i(binary ? 2 : 16)
252 raise CommandError, "ERROR BAD OPERAND #{word}" if pattern >= 2 * sign
253
254 return Operand.new(pattern, nil)
255 end
256 case word
257 when 'nan' then return Operand.new(nan, nil)
258 when 'inf' then return Operand.new(infinity, nil)
259 when '-inf' then return Operand.new(infinity | sign, nil)
260 end
261 negative = word.start_with?('-')
262 digits = word.delete_prefix('-')
263 if digits.include?('/')
264 numerator, denominator = digits.split('/').map { |part| part.to_i(10) }
265 raise CommandError, "ERROR BAD OPERAND #{word}" if denominator.zero?
266 elsif digits.include?('.')
267 integer, fraction = digits.split('.')
268 numerator = (integer + fraction).to_i(10)
269 denominator = 10**fraction.length
270 else
271 numerator, denominator = digits.to_i(10), 1
272 end
273 Operand.new(0, Exact.make(negative, numerator, 0, denominator))
274 end

276 def operand(word)
277 value = raw_operand(word, false)
278 value.number ? round(value.number, :RNE).pattern : value.pattern
279 end

281 def bits(pattern)
282 "#{negative?(pattern) ? 1 : 0}|#{binary((pattern & (sign - 1)) >> m, e)}|#{binary(pattern & (base - 1), m)}"
283 end
284 def hex(pattern) = format('0x%0*X', (w + 3) / 4, pattern)
285
286 def value(pattern)
287 case kind(pattern)
288 when :qnan, :snan then 'nan'
289 when :infinity then negative?(pattern) ? '-inf' : 'inf'
290 when :zero then negative?(pattern) ? '-0' : '0'
291 else term(pattern).decimal
292 end
293 end
294
295 def pattern_text(pattern) = "#{bits(pattern)} #{hex(pattern)} #{value(pattern)}"
296 def result_text(result) = "#{pattern_text(result.pattern)} #{flag_text(result.flags)}"
297 def difference(pattern, exact_sum)
298 finite?(pattern) ? sum_terms([term(pattern), exact_sum.negated], :RNE).exact.fraction : value(pattern)
299 end
300 def order_key(pattern) = negative?(pattern) ? -(pattern - sign) - 1 : pattern
301
302 def next_up(pattern)
303 return pattern if pattern == infinity
304 return 1 if zero?(pattern)
305
306 negative?(pattern) ? pattern - 1 : pattern + 1
307 end
308end
309
310def binary(value, width) = value.to_s(2).rjust(width, '0')
311
312def flag_text(flags)
313 names = FLAG_NAMES.each_with_index.filter_map { |name, i| name unless (flags & (1 << i)).zero? }
314 "flags=#{names.empty? ? 'none' : names.join(',')}"
315end

317def parse_mode(word)
318 mode = word.to_sym
319 raise CommandError, "ERROR BAD MODE #{word}" unless MODES.include?(mode)
320
321 mode
322end
323
324def mode_at(words, index) = words[index] ? parse_mode(words[index]) : :RNE
325def integer_range(word) = INTEGER_RANGES.fetch(word) { raise CommandError, "ERROR BAD TYPE #{word}" }
326def relation(a, b) = a < b ? 'LT' : a == b ? 'EQ' : 'GT'

328def summation(f, words)
329 command = words[0]
330 mode = parse_mode(words[2])
331 values = []
332 left_count = 0
333 words.drop(3).each do |word|
334 if word == '|'
335 left_count = values.length
336 else
337 values << f.operand(word)
338 end
339 end
340 if command == 'DOT' && left_count != values.length - left_count
341 raise CommandError, 'ERROR LENGTH MISMATCH'
342 end
343 raise CommandError, 'ERROR NOT FINITE' unless values.all? { |p| f.finite?(p) }
344
345 if command == 'DOT'
346 terms = Array.new(left_count) { |i| f.term(values[i]).product(f.term(values[left_count + i])) }
347 exact_sum = sum_terms(terms, mode)
348 first = f.arithmetic(:MUL, values.first, values[left_count], 0, mode)
349 naive = chained = first
350 (1...left_count).each do |i|
351 multiplied = f.arithmetic(:MUL, values[i], values[left_count + i], 0, mode)
352 added = f.arithmetic(:ADD, naive.pattern, multiplied.pattern, 0, mode)
353 naive = Rounded.new(added.pattern, naive.flags | multiplied.flags | added.flags)
354 end
355 (1...left_count).each do |i|
356 fused = f.arithmetic(:FMA, values[i], values[left_count + i], chained.pattern, mode)
357 chained = Rounded.new(fused.pattern, chained.flags | fused.flags)
358 end
359 return [
360 "DOT #{f.name} #{mode} n=#{left_count}",
361 "NAIVE #{f.result_text(naive)}",
362 "CHAINED #{f.result_text(chained)}",
363 "EXACT #{exact_sum.decimal}",
364 "FUSED #{f.result_text(f.round_term(exact_sum, mode))}",
365 "DIFF naive=#{f.difference(naive.pattern, exact_sum)} chained=#{f.difference(chained.pattern, exact_sum)}"
366 ]
367 end
368 exact_sum = sum_terms(values.map { |p| f.term(p) }, mode)
369 result = Rounded.new(values.first, 0)
370 compensation = 0
371 out = ["#{command} #{f.name} #{mode} count=#{values.length}"]
372 values.each_with_index do |value, i|
373 next if i.zero?
374
375 if command == 'SUM'
376 added = f.arithmetic(:ADD, result.pattern, value, 0, mode)
377 result = Rounded.new(added.pattern, result.flags | added.flags)
378 out << "PARTIAL #{i + 1} #{f.result_text(added)}"
379 else
380 y = f.arithmetic(:SUB, value, compensation, 0, mode)
381 t = f.arithmetic(:ADD, result.pattern, y.pattern, 0, mode)
382 d = f.arithmetic(:SUB, t.pattern, result.pattern, 0, mode)
383 c = f.arithmetic(:SUB, d.pattern, y.pattern, 0, mode)
384 flags = y.flags | t.flags | d.flags | c.flags
385 result = Rounded.new(t.pattern, result.flags | flags)
386 compensation = c.pattern
387 out << "KSTEP #{i + 1} sum=#{f.value(result.pattern)} comp=#{f.value(compensation)} #{flag_text(flags)}"
388 end
389 end
390 out << "RESULT #{f.result_text(result)}"
391 out << "EXACT #{exact_sum.decimal}"
392 out << "ROUNDED #{f.result_text(f.round_term(exact_sum, mode))}"
393 out << "DIFF #{f.difference(result.pattern, exact_sum)}"
394 out
395end

397def table(f)
398 raise CommandError, 'ERROR TABLE TOO LARGE' if f.w > 8
399
400 out = ["TABLE #{f.name} count=#{f.infinity}"]
401 f.infinity.times do |p|
402 gap = p + 1 == f.infinity ? '-' : sum_terms([f.term(p + 1), f.term(p).negated], :RNE).decimal
403 out << "ENTRY #{p} #{f.pattern_text(p)} gap=#{gap}"
404 end
405 out << "MAX #{f.pattern_text(f.infinity - 1)}"
406 out << "MINNORMAL #{f.pattern_text(f.base)}"
407 out << "MINSUBNORMAL #{f.pattern_text(1)}"
408 out << "EPSILON #{Term.new(false, 1, -f.m).decimal}"
409 out << "PATTERNS finite=#{2 * f.infinity} infinite=2 nan=#{2 * (f.base - 1)} total=#{2 * f.sign}"
410 out
411end

413class Laboratory
414 def initialize
415 @formats = {
416 'HALF' => MiniFormat.new('HALF', 5, 10), 'E5M2' => MiniFormat.new('E5M2', 5, 2),
417 'E4M3' => MiniFormat.new('E4M3', 4, 3)
418 }
419 end
420
421 def find_format(name)
422 @formats.fetch(name) { raise CommandError, "ERROR UNKNOWN FORMAT #{name}" }
423 end
424
425 def execute(words)
426 command = words[0]
427 if command == 'FORMAT'
428 name = words[1]
429 raise CommandError, "ERROR FORMAT EXISTS #{name}" if @formats.key?(name)
430
431 e, m = words[2].to_i(10), words[3].to_i(10)
432 raise CommandError, 'ERROR BAD FORMAT' unless (2..5).cover?(e) && (1..10).cover?(m)
433
434 f = MiniFormat.new(name, e, m)
435 @formats[name] = f
436 return ["FORMAT #{name} e=#{e} m=#{m} bits=#{f.w} bias=#{f.bias} emin=#{f.emin} emax=#{f.emax}"]
437 end
438 f = find_format(words[1])
439 return table(f) if command == 'TABLE'
440 return summation(f, words) if %w[SUM KAHAN DOT].include?(command)
441
442 if command == 'CONVERT'
443 g = find_format(words[2])
444 a = f.operand(words[3])
445 mode = mode_at(words, 4)
446 result = case f.kind(a)
447 when :qnan then Rounded.new(g.nan, 0)
448 when :snan then Rounded.new(g.nan, INVALID)
449 when :infinity then Rounded.new(g.signed(g.infinity, f.negative?(a)), 0)
450 when :zero then Rounded.new(g.signed(0, f.negative?(a)), 0)
451 else g.round_term(f.term(a), mode)
452 end
453 return ["CONVERT #{f.name} #{g.name} #{mode} #{g.result_text(result)}"]
454 end
455 if command == 'ENCODE'
456 value = f.raw_operand(words[2], true)
457 mode = mode_at(words, 3)
458 result = value.number ? f.round(value.number, mode) : Rounded.new(value.pattern, 0)
459 source = value.number ? value.number.fraction : f.value(value.pattern)
460 return ["ENCODE #{f.name} #{mode} #{source} #{f.result_text(result)}"]
461 end
462 a = f.operand(words[2])
463 case command
464 when 'DECODE'
465 kind = f.kind(a)
466 exponent = (a & (f.sign - 1)) >> f.m
467 fraction = a & (f.base - 1)
468 out = ["DECODE #{f.name} #{f.bits(a)} #{f.hex(a)} #{kind} sign=#{f.negative?(a) ? 1 : 0}"]
469 if f.finite?(a)
470 term = f.term(a)
471 out << "FIELDS exponent=#{exponent} fraction=#{fraction} unbiased=#{term.exponent + f.m} significand=#{kind == :normal ? 1 : 0}.#{binary(fraction, f.m)}"
472 out << "VALUE #{term.exact.fraction} #{f.value(a)}"
473 elsif kind == :infinity
474 out << "FIELDS exponent=#{exponent} fraction=#{fraction}"
475 else
476 out << "FIELDS exponent=#{exponent} fraction=#{fraction} payload=#{fraction % (f.base / 2)}"
477 end
478 out
479 when 'COMPARE'
480 b = f.operand(words[3])
481 ak, bk = f.order_key(a), f.order_key(b)
482 numeric = if f.nan?(a) || f.nan?(b)
483 'UN'
484 elsif f.zero?(a) && f.zero?(b)
485 'EQ'
486 else
487 relation(ak, bk)
488 end
489 flags = f.kind(a) == :snan || f.kind(b) == :snan ? INVALID : 0
490 ["COMPARE #{f.name} #{f.hex(a)} #{f.hex(b)} #{numeric} total=#{relation(ak, bk)} #{flag_text(flags)}"]
491 when 'TOINT'
492 range = integer_range(words[3])
493 mode = mode_at(words, 4)
494 prefix = "TOINT #{f.name} #{words[3]} #{mode}"
495 return ["#{prefix} invalid flags=invalid"] unless f.finite?(a)
496
497 value = f.term(a).exact
498 divided = Quantized.from(value, 0, false)
499 magnitude = divided.rounded(mode, value.negative)
500 result = value.negative ? -magnitude : magnitude
501 return ["#{prefix} invalid flags=invalid"] unless range.cover?(result)
502
503 ["#{prefix} #{result} #{flag_text(divided.inexact ? INEXACT : 0)}"]
504 when 'NEXTUP', 'NEXTDOWN', 'ULP'
505 if f.nan?(a)
506 result = Rounded.new(f.nan, f.kind(a) == :snan ? INVALID : 0)
507 else
508 pattern = case command
509 when 'NEXTUP' then f.next_up(a)
510 when 'NEXTDOWN' then f.next_up(a ^ f.sign) ^ f.sign
511 else
512 f.kind(a) == :infinity ? f.infinity : f.round_term(Term.new(false, 1, f.term(a).exponent), :RNE).pattern
513 end
514 result = Rounded.new(pattern, 0)
515 end
516 ["#{command} #{f.name} #{f.result_text(result)}"]
517 else
518 operation = command.to_sym
519 arity = operation == :SQRT ? 1 : operation == :FMA ? 3 : 2
520 b = arity >= 2 ? f.operand(words[3]) : 0
521 c = arity == 3 ? f.operand(words[4]) : 0
522 mode = mode_at(words, arity + 2)
523 ["#{command} #{f.name} #{mode} #{f.result_text(f.arithmetic(operation, a, b, c, mode))}"]
524 end
525 end
526end
