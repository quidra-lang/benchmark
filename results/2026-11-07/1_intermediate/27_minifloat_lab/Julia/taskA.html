21struct Exact
22 n::Int64
23 e::Int
24 d::Int64
25 negative::Bool
26 function Exact(n::Integer, e::Int = 0, d::Integer = 1, negative::Bool = false)
27 numerator, denominator = Int64(n), Int64(d)
28 if numerator == 0
29 return new(0, 0, 1, negative)
30 end
31 factor = gcd(numerator, denominator)
32 numerator ÷= factor
33 denominator ÷= factor
34 shift = trailing_zeros(numerator)
35 numerator >>= shift
36 e += shift
37 shift = trailing_zeros(denominator)
38 denominator >>= shift
39 e -= shift
40 new(numerator, e, denominator, negative)
41 end
42end
43negated(value::Exact) = Exact(value.n, value.e, value.d, !value.negative)
44product(a::Exact, b::Exact) = Exact(a.n * b.n, a.e + b.e, a.d * b.d, xor(a.negative, b.negative))

46function scaled(value::Exact, quantum::Int)
47 shift = value.e - quantum
48 shift >= 0 ? (value.n << shift, value.d) : (value.n, value.d << -shift)
49end
50function log_two_floor(value::Exact)
51 power = leading_zeros(value.d) - leading_zeros(value.n)
52 below = power >= 0 ? value.n < (value.d << power) : (value.n << -power) < value.d
53 value.e + power - Int(below)
54end
55function fraction_text(value::Exact)
56 value.n == 0 && return "0"
57 numerator, denominator = scaled(value, 0)
58 string(value.negative ? "-" : "", numerator, denominator == 1 ? "" : "/$denominator")
59end
60function decimal_text(value::Exact; keep_zero_sign::Bool = false)
61 numerator, denominator = scaled(value, 0)
62 output = IOBuffer()
63 value.negative && (value.n != 0 || keep_zero_sign) && print(output, '-')
64 print(output, numerator ÷ denominator)
65 remainder = numerator % denominator
66 remainder != 0 && print(output, '.')
67 while remainder != 0
68 remainder *= 10
69 print(output, remainder ÷ denominator)
70 remainder %= denominator
71 end
72 String(take!(output))
73end
74function exact_sum(terms::Vector{Exact}, mode::Mode)
75 nonzero = filter(term -> term.n != 0, terms)
76 if isempty(nonzero)
77 same = all(term -> term.negative == terms[1].negative, terms)
78 return Exact(0, 0, 1, same ? terms[1].negative : mode == RDN)
79 end
80 exponent = minimum(term.e for term in nonzero)
81 total = sum((term.negative ? -term.n : term.n) << (term.e - exponent) for term in nonzero)
82 Exact(abs(total), exponent, 1, total < 0 || (total == 0 && mode == RDN))
83end

90function integer_sqrt(value::Int64)
91 value == 0 && return Int64(0)
92 root, next = value, (value + 1) ÷ 2
93 while next < root
94 root = next
95 next = (root + value ÷ root) ÷ 2
96 end
97 root
98end
99function quantize(value::Exact, exponent::Int, square_root::Bool)
100 numerator, denominator = scaled(value, square_root ? 2 * exponent : exponent)
101 if !square_root
102 remainder = numerator % denominator
103 return Units(numerator ÷ denominator, remainder != 0, cmp(remainder, denominator - remainder))
104 end
105 integer = integer_sqrt(numerator ÷ denominator)
106 midpoint = 2 * integer + 1
107 Units(integer, integer * integer * denominator != numerator, cmp(4 * numerator, midpoint * midpoint * denominator))
108end
109function rounded_units(units::Units, mode::Mode, negative::Bool)
110 increment = if mode == RNE
111 units.halfway > 0 || (units.halfway == 0 && isodd(units.value))
112 elseif mode == RNA
113 units.halfway >= 0
114 elseif mode == RUP
115 units.inexact && !negative
116 elseif mode == RDN
117 units.inexact && negative
118 else
119 false
120 end
121 units.value + Int(increment)
122end

133struct MiniFormat
134 name::String

146function MiniFormat(name::AbstractString, e::Int, m::Int)
147 bias = (1 << (e - 1)) - 1
148 hidden = 1 << m
149 infinity = ((1 << e) - 1) * hidden
150 MiniFormat(String(name), e, m, 1 + e + m, bias, 1 - bias, bias, hidden, 1 << (e + m), infinity, infinity + hidden ÷ 2)
151end
152function fields(format::MiniFormat, pattern::Int)
153 exponent = (pattern & (format.sign_bit - 1)) >> format.m
154 (negative = (pattern & format.sign_bit) != 0, exponent = exponent,
155 fraction = pattern & (format.hidden - 1), q = (exponent == 0 ? format.emin : exponent - format.bias) - format.m)
156end
157function classification(format::MiniFormat, pattern::Int)
158 magnitude = pattern & (format.sign_bit - 1)
159 fraction = pattern & (format.hidden - 1)
160 if magnitude >= format.infinity
161 fraction == 0 && return INFINITY
162 return fraction >= format.hidden ÷ 2 ? QNAN : SNAN
163 end
164 magnitude == 0 ? ZERO : magnitude < format.hidden ? SUBNORMAL : NORMAL
165end
166is_nan(format::MiniFormat, pattern::Int) = classification(format, pattern) in (QNAN, SNAN)
167is_signaling(format::MiniFormat, pattern::Int) = classification(format, pattern) == SNAN
168is_finite(format::MiniFormat, pattern::Int) = classification(format, pattern) in (ZERO, SUBNORMAL, NORMAL)
169is_infinite(format::MiniFormat, pattern::Int) = classification(format, pattern) == INFINITY
170is_zero(format::MiniFormat, pattern::Int) = (pattern & (format.sign_bit - 1)) == 0
171zero_pattern(format::MiniFormat, negative::Bool) = negative ? format.sign_bit : 0
172infinity_pattern(format::MiniFormat, negative::Bool) = zero_pattern(format, negative) + format.infinity
173flip(format::MiniFormat, pattern::Int) = xor(pattern, format.sign_bit)
174function pattern_exact(format::MiniFormat, pattern::Int)
175 field = fields(format, pattern)
176 Exact(field.fraction + (field.exponent == 0 ? 0 : format.hidden), field.q, 1, field.negative)
177end
178function bits_text(format::MiniFormat, pattern::Int)
179 field = fields(format, pattern)
180 string(Int(field.negative), '|', string(field.exponent; base = 2, pad = format.e), '|', string(field.fraction; base = 2, pad = format.m))
181end
182hex_text(format::MiniFormat, pattern::Int) = "0x" * uppercase(string(pattern; base = 16, pad = cld(format.w, 4)))
183function value_text(format::MiniFormat, pattern::Int)
184 is_nan(format, pattern) && return "nan"
185 is_infinite(format, pattern) && return (fields(format, pattern).negative ? "-inf" : "inf")
186 decimal_text(pattern_exact(format, pattern); keep_zero_sign = true)
187end
188function flags_text(flags::UInt8)
189 names = ("invalid", "divbyzero", "overflow", "underflow", "inexact")
190 raised = [name for (index, name) in enumerate(names) if (flags & (UInt8(1) << (index - 1))) != 0]
191 "flags=" * (isempty(raised) ? "none" : join(raised, ','))
192end
193pattern_text(format::MiniFormat, pattern::Int) = "$(bits_text(format, pattern)) $(hex_text(format, pattern)) $(value_text(format, pattern))"
194result_text(format::MiniFormat, result::Result) = "$(pattern_text(format, result.pattern)) $(flags_text(result.flags))"

196function overflow_result(format::MiniFormat, mode::Mode, negative::Bool)
197 infinite = mode in (RNE, RNA) || (mode == RUP && !negative) || (mode == RDN && negative)
198 Result(zero_pattern(format, negative) + (infinite ? format.infinity : format.infinity - 1), OVERFLOW | INEXACT)
199end
200function round_exact(format::MiniFormat, value::Exact, mode::Mode; square_root::Bool = false)
201 value.n == 0 && return Result(zero_pattern(format, value.negative))
202 order = log_two_floor(value)
203 square_root && (order = fld(order, 2))
204 order > format.emax && return overflow_result(format, mode, value.negative)
205 exponent = max(order, format.emin) - format.m
206 units = quantize(value, exponent, square_root)
207 n = rounded_units(units, mode, value.negative)
208 if n == 2 * format.hidden
209 n = Int64(format.hidden)
210 exponent += 1
211 end
212 exponent > format.emax - format.m && return overflow_result(format, mode, value.negative)
213 flags = units.inexact ? INEXACT : UInt8(0)
214 if units.inexact && order < format.emin
215 tiny = order < format.emin - 1 || rounded_units(quantize(value, order - format.m, square_root), mode, value.negative) < 2 * format.hidden
216 tiny && (flags |= UNDERFLOW)
217 end
218 magnitude = n < format.hidden ? Int(n) : (exponent + format.m + format.bias) * format.hidden + Int(n - format.hidden)
219 Result(zero_pattern(format, value.negative) + magnitude, flags)
220end

222function raw_operand(format::MiniFormat, word::AbstractString; encoding::Bool = false)
223 if startswith(word, "0b") || startswith(word, "0x")
224 binary = startswith(word, "0b")
225 width = binary ? format.w : cld(format.w, 4)
226 (encoding || length(word) - 2 != width) && fail("BAD OPERAND $word")
227 pattern = parse(Int, word[3:end]; base = binary ? 2 : 16)
228 pattern >= 2 * format.sign_bit && fail("BAD OPERAND $word")
229 return InputValue(pattern, nothing)
230 end
231 word == "nan" && return InputValue(format.nan, nothing)
232 word == "inf" && return InputValue(format.infinity, nothing)
233 word == "-inf" && return InputValue(infinity_pattern(format, true), nothing)
234 negative = startswith(word, "-")
235 body = negative ? word[2:end] : word
236 denominator = Int64(1)
237 if occursin('/', body)
238 parts = split(body, '/')
239 numerator = parse(Int64, parts[1]; base = 10)
240 denominator = parse(Int64, parts[2]; base = 10)
241 denominator == 0 && fail("BAD OPERAND $word")
242 elseif occursin('.', body)
243 parts = split(body, '.')
244 numerator = parse(Int64, parts[1] * parts[2]; base = 10)
245 denominator = Int64(10)^length(parts[2])
246 else
247 numerator = parse(Int64, body; base = 10)
248 end
249 InputValue(0, Exact(numerator, 0, denominator, negative))
250end
251function operand(format::MiniFormat, word::AbstractString)
252 value = raw_operand(format, word)
253 isnothing(value.number) ? value.pattern : round_exact(format, value.number, RNE).pattern
254end

256function arithmetic(format::MiniFormat, operation::Operation, inputs::Vector{Int}, mode::Mode)
257 if operation == SUB
258 return arithmetic(format, ADD, [inputs[1], flip(format, inputs[2])], mode)
259 end
260 if any(pattern -> is_nan(format, pattern), inputs)
261 return Result(format.nan, any(pattern -> is_signaling(format, pattern), inputs) ? INVALID : UInt8(0))
262 end
263 invalid = Result(format.nan, INVALID)
264 a = inputs[1]
265 negative = fields(format, a).negative
266 if operation == SQRT
267 is_zero(format, a) && return Result(a)
268 negative && return invalid
269 is_infinite(format, a) && return Result(a)
270 return round_exact(format, pattern_exact(format, a), mode; square_root = true)
271 end
272 b = inputs[2]
273 sigma = xor(negative, fields(format, b).negative)
274 if operation == ADD
275 is_infinite(format, a) && is_infinite(format, b) && sigma && return invalid
276 is_infinite(format, a) && return Result(a)
277 is_infinite(format, b) && return Result(b)
278 return round_exact(format, exact_sum([pattern_exact(format, a), pattern_exact(format, b)], mode), mode)
279 elseif operation == MUL
280 if is_infinite(format, a) || is_infinite(format, b)
281 return is_zero(format, a) || is_zero(format, b) ? invalid : Result(infinity_pattern(format, sigma))
282 end
283 return round_exact(format, product(pattern_exact(format, a), pattern_exact(format, b)), mode)
284 elseif operation == DIV
285 is_infinite(format, a) && return (is_infinite(format, b) ? invalid : Result(infinity_pattern(format, sigma)))
286 is_infinite(format, b) && return Result(zero_pattern(format, sigma))
287 is_zero(format, b) && return (is_zero(format, a) ? invalid : Result(infinity_pattern(format, sigma), DIVBYZERO))
288 is_zero(format, a) && return Result(zero_pattern(format, sigma))
289 x, y = pattern_exact(format, a), pattern_exact(format, b)
290 return round_exact(format, Exact(x.n * y.d, x.e - y.e, x.d * y.n, sigma), mode)
291 end
292 c = inputs[3]
293 if is_infinite(format, a) || is_infinite(format, b)
294 (is_zero(format, a) || is_zero(format, b) || (is_infinite(format, c) && fields(format, c).negative != sigma)) && return invalid
295 return Result(infinity_pattern(format, sigma))
296 end
297 is_infinite(format, c) && return Result(c)
298 terms = [product(pattern_exact(format, a), pattern_exact(format, b)), pattern_exact(format, c)]
299 round_exact(format, exact_sum(terms, mode), mode)
300end
301function next_up(format::MiniFormat, pattern::Int)
302 pattern == format.infinity && return pattern
303 is_zero(format, pattern) && return 1
304 pattern + (fields(format, pattern).negative ? -1 : 1)
305end
306function difference_text(format::MiniFormat, pattern::Int, value::Exact)
307 is_finite(format, pattern) || return value_text(format, pattern)
308 fraction_text(exact_sum([pattern_exact(format, pattern), negated(value)], RNE))
309end
310relation(a::Integer, b::Integer) = a < b ? "LT" : a > b ? "GT" : "EQ"

312function summation(output::IO, command::AbstractString, format::MiniFormat, mode::Mode, patterns::Vector{Int})
313 all(pattern -> is_finite(format, pattern), patterns) || fail("NOT FINITE")
314 line(output, command, format.name, mode, "count=$(length(patterns))")
315 current, compensation, flags = patterns[1], 0, UInt8(0)
316 for index in 2:length(patterns)
317 if command == "SUM"
318 added = arithmetic(format, ADD, [current, patterns[index]], mode)
319 current = added.pattern
320 flags |= added.flags
321 line(output, "PARTIAL", index, result_text(format, added))
322 else
323 y = arithmetic(format, SUB, [patterns[index], compensation], mode)
324 t = arithmetic(format, ADD, [current, y.pattern], mode)
325 d = arithmetic(format, SUB, [t.pattern, current], mode)
326 correction = arithmetic(format, SUB, [d.pattern, y.pattern], mode)
327 current, compensation = t.pattern, correction.pattern
328 step_flags = y.flags | t.flags | d.flags | correction.flags
329 flags |= step_flags
330 line(output, "KSTEP", index, "sum=$(value_text(format, current))", "comp=$(value_text(format, compensation))", flags_text(step_flags))
331 end
332 end
333 value = exact_sum([pattern_exact(format, pattern) for pattern in patterns], mode)
334 line(output, "RESULT", result_text(format, Result(current, flags)))
335 line(output, "EXACT", decimal_text(value))
336 line(output, "ROUNDED", result_text(format, round_exact(format, value, mode)))
337 line(output, "DIFF", difference_text(format, current, value))
338end
339function dot_product(output::IO, format::MiniFormat, mode::Mode, a::Vector{Int}, b::Vector{Int})
340 length(a) == length(b) || fail("LENGTH MISMATCH")
341 all(pattern -> is_finite(format, pattern), [a; b]) || fail("NOT FINITE")
342 naive = arithmetic(format, MUL, [a[1], b[1]], mode)
343 chained = naive
344 for index in 2:length(a)
345 multiplied = arithmetic(format, MUL, [a[index], b[index]], mode)
346 added = arithmetic(format, ADD, [naive.pattern, multiplied.pattern], mode)
347 naive = Result(added.pattern, naive.flags | multiplied.flags | added.flags)
348 fused = arithmetic(format, FMA, [a[index], b[index], chained.pattern], mode)
349 chained = Result(fused.pattern, chained.flags | fused.flags)
350 end
351 value = exact_sum([product(pattern_exact(format, a[index]), pattern_exact(format, b[index])) for index in eachindex(a)], mode)
352 line(output, "DOT", format.name, mode, "n=$(length(a))")
353 line(output, "NAIVE", result_text(format, naive))
354 line(output, "CHAINED", result_text(format, chained))
355 line(output, "EXACT", decimal_text(value))
356 line(output, "FUSED", result_text(format, round_exact(format, value, mode)))
357 line(output, "DIFF", "naive=$(difference_text(format, naive.pattern, value))", "chained=$(difference_text(format, chained.pattern, value))")
358end
359function table_lines(output::IO, format::MiniFormat)
360 format.w <= 8 || fail("TABLE TOO LARGE")
361 count = format.infinity
362 line(output, "TABLE", format.name, "count=$count")
363 for pattern in 0:(count - 1)
364 gap = pattern == count - 1 ? "-" : decimal_text(exact_sum([pattern_exact(format, pattern + 1), negated(pattern_exact(format, pattern))], RNE))
365 line(output, "ENTRY", pattern, pattern_text(format, pattern), "gap=$gap")
366 end
367 line(output, "MAX", pattern_text(format, format.infinity - 1))
368 line(output, "MINNORMAL", pattern_text(format, format.hidden))
369 line(output, "MINSUBNORMAL", pattern_text(format, 1))
370 line(output, "EPSILON", decimal_text(Exact(1, -format.m)))
371 line(output, "PATTERNS", "finite=$(2 * count)", "infinite=2", "nan=$(2 * (format.hidden - 1))", "total=$(2 * format.sign_bit)")
372end

374function format_named(formats::Dict{String,MiniFormat}, name::AbstractString)
375 haskey(formats, name) || fail("UNKNOWN FORMAT $name")
376 formats[name]

378function mode_word(words::Vector{<:AbstractString}, index::Int)
379 index > length(words) && return RNE
380 mode = get(MODES, words[index], nothing)
381 isnothing(mode) && fail("BAD MODE $(words[index])")
382 mode
383end
384function execute(formats::Dict{String,MiniFormat}, words::Vector{<:AbstractString})
385 command = words[1]
386 output = IOBuffer()
387 if command == "FORMAT"
388 name = words[2]
389 e, m = parse(Int, words[3]), parse(Int, words[4])
390 haskey(formats, name) && fail("FORMAT EXISTS $name")
391 (2 <= e <= 5 && 1 <= m <= 10) || fail("BAD FORMAT")
392 format = MiniFormat(name, e, m)
393 formats[format.name] = format
394 line(output, "FORMAT", name, "e=$e", "m=$m", "bits=$(format.w)", "bias=$(format.bias)", "emin=$(format.emin)", "emax=$(format.emax)")
395 return String(take!(output))
396 end
397 format = format_named(formats, words[2])
398 name = format.name
399 if command == "ENCODE"
400 input = raw_operand(format, words[3]; encoding = true)
401 mode = mode_word(words, 4)
402 value = isnothing(input.number) ? value_text(format, input.pattern) : fraction_text(input.number)
403 encoded = isnothing(input.number) ? Result(input.pattern) : round_exact(format, input.number, mode)
404 line(output, "ENCODE", name, mode, value, result_text(format, encoded))
405 elseif command == "CONVERT"
406 target = format_named(formats, words[3])
407 a = operand(format, words[4])
408 mode = mode_word(words, 5)
409 converted = if is_nan(format, a)
410 Result(target.nan, is_signaling(format, a) ? INVALID : UInt8(0))
411 elseif is_infinite(format, a)
412 Result(infinity_pattern(target, fields(format, a).negative))
413 else
414 round_exact(target, pattern_exact(format, a), mode)
415 end
416 line(output, "CONVERT", name, target.name, mode, result_text(target, converted))
417 elseif command in ("SUM", "KAHAN", "DOT")
418 mode = mode_word(words, 3)
419 if command == "DOT"
420 separator = findfirst(==("|"), words)::Int
421 a = [operand(format, word) for word in words[4:(separator - 1)]]
422 b = [operand(format, word) for word in words[(separator + 1):end]]
423 dot_product(output, format, mode, a, b)
424 else
425 patterns = [operand(format, word) for word in words[4:end]]
426 summation(output, command, format, mode, patterns)
427 end
428 elseif command == "TABLE"
429 table_lines(output, format)
430 elseif haskey(OPERATIONS, command)
431 operation = OPERATIONS[command]
432 count = operation == SQRT ? 1 : operation == FMA ? 3 : 2
433 patterns = [operand(format, word) for word in words[3:(2 + count)]]
434 mode = mode_word(words, 3 + count)
435 line(output, command, name, mode, result_text(format, arithmetic(format, operation, patterns, mode)))
436 else
437 a = operand(format, words[3])
438 field = fields(format, a)
439 if command == "DECODE"
440 kind = classification(format, a)
441 line(output, "DECODE", name, bits_text(format, a), hex_text(format, a), lowercase(string(kind)), "sign=$(Int(field.negative))")
442 description = "FIELDS exponent=$(field.exponent) fraction=$(field.fraction)"
443 if is_finite(format, a)
444 description *= " unbiased=$(field.q + format.m) significand=$(field.exponent == 0 ? 0 : 1).$(string(field.fraction; base = 2, pad = format.m))"
445 println(output, description)
446 line(output, "VALUE", fraction_text(pattern_exact(format, a)), value_text(format, a))
447 else
448 is_nan(format, a) && (description *= " payload=$(field.fraction % (format.hidden ÷ 2))")
449 println(output, description)
450 end
451 elseif command == "COMPARE"
452 b = operand(format, words[4])
453 key(pattern) = fields(format, pattern).negative ? -(pattern - format.sign_bit) - 1 : pattern
454 total = relation(key(a), key(b))
455 numeric = is_nan(format, a) || is_nan(format, b) ? "UN" : is_zero(format, a) && is_zero(format, b) ? "EQ" : total
456 flags = is_signaling(format, a) || is_signaling(format, b) ? INVALID : UInt8(0)
457 line(output, "COMPARE", name, hex_text(format, a), hex_text(format, b), numeric, "total=$total", flags_text(flags))
458 elseif command == "TOINT"
459 type = words[4]
460 haskey(TYPES, type) || fail("BAD TYPE $type")
461 mode = mode_word(words, 5)
462 if !is_finite(format, a)
463 line(output, "TOINT", name, type, mode, "invalid flags=invalid")
464 else
465 units = quantize(pattern_exact(format, a), 0, false)
466 magnitude = rounded_units(units, mode, field.negative)
467 integer = field.negative ? -magnitude : magnitude
468 minimum, maximum = TYPES[type]
469 if integer < minimum || integer > maximum
470 line(output, "TOINT", name, type, mode, "invalid flags=invalid")
471 else
472 line(output, "TOINT", name, type, mode, integer, flags_text(units.inexact ? INEXACT : UInt8(0)))
473 end
474 end
475 else
476 value = if is_nan(format, a)
477 Result(format.nan, is_signaling(format, a) ? INVALID : UInt8(0))
478 elseif command == "NEXTUP"
479 Result(next_up(format, a))
480 elseif command == "NEXTDOWN"
481 Result(flip(format, next_up(format, flip(format, a))))
482 else
483 Result(is_infinite(format, a) ? format.infinity : round_exact(format, Exact(1, field.q), RNE).pattern)
484 end
485 line(output, command, name, result_text(format, value))
486 end
487 end
488 String(take!(output))
489end

491function main()
492 count = parse(Int, readline())
493 formats = Dict("HALF" => MiniFormat("HALF", 5, 10), "E5M2" => MiniFormat("E5M2", 5, 2), "E4M3" => MiniFormat("E4M3", 4, 3))
494 for _ in 1:count
495 words = split(readline())
496 try
497 print(execute(formats, words))
498 catch error
499 error isa CommandError || rethrow()
500 println("ERROR ", error.message)
501 end
502 end
