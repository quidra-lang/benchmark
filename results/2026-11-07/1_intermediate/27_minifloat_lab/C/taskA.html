55static Format *add_format(const char *name, int e, int m) {
56 Format *f = &formats[format_count++];
57 strcpy(f->name, name);
58 f->e = e;
59 f->m = m;
60 f->w = 1 + e + m;
61 f->bias = (1 << (e - 1)) - 1;
62 f->emin = 1 - f->bias;
63 f->emax = f->bias;
64 f->sign = UINT32_C(1) << (e + m);
65 f->base = UINT32_C(1) << m;
66 f->infinity = ((UINT32_C(1) << e) - 1) << m;
67 f->nan = f->infinity | (f->base >> 1);
68 return f;
69}

71static Class classify(const Format *f, Pattern p) {
72 Pattern magnitude = p & (f->sign - 1), fraction = p & (f->base - 1);
73 if (magnitude >= f->infinity) {
74 if (!fraction) return INFINITY_VALUE;
75 return fraction >= f->base / 2 ? QNAN : SNAN;
76 }
77 if (!magnitude) return ZERO;
78 return magnitude < f->base ? SUBNORMAL : NORMAL;
79}

81static bool is_nan(const Format *f, Pattern p) { return classify(f, p) >= QNAN; }
82static bool is_finite(const Format *f, Pattern p) { return classify(f, p) <= NORMAL; }
83static bool is_zero(const Format *f, Pattern p) { return !(p & (f->sign - 1)); }
84static bool negative(const Format *f, Pattern p) { return (p & f->sign) != 0; }
85static Pattern with_sign(const Format *f, Pattern p, bool sign) { return p | (sign ? f->sign : 0); }

87static Term term(const Format *f, Pattern p) {
88 Pattern field = (p & (f->sign - 1)) >> f->m, fraction = p & (f->base - 1);
89 return (Term){ negative(f, p), field ? f->base + fraction : fraction,
90 field ? (int)field - f->bias - f->m : f->emin - f->m };
91}

93static uint64_t gcd(uint64_t a, uint64_t b) {
94 while (b) { uint64_t remainder = a % b; a = b; b = remainder; }
95 return a;
96}

98static Exact exact(bool sign, uint64_t numerator, int exponent, uint64_t denominator) {
99 if (!numerator) return (Exact){ sign, 0, 1, 0 };
100 uint64_t common = gcd(numerator, denominator);
101 numerator /= common;
102 denominator /= common;
103 while (!(numerator & 1)) { numerator >>= 1; exponent++; }
104 while (!(denominator & 1)) { denominator >>= 1; exponent--; }
105 return (Exact){ sign, numerator, denominator, exponent };
106}

108static Exact exact_term(Term value) { return exact(value.negative, value.magnitude, value.exponent, 1); }
109static Term product(Term a, Term b) {
110 return (Term){ a.negative != b.negative, a.magnitude * b.magnitude, a.exponent + b.exponent };
111}

113static Term sum_terms(const Term *terms, size_t count, Mode mode) {
114 bool any = false, all_same_zero = true;
115 int exponent = 0;
116 for (size_t i = 0; i < count; i++) {
117 if (terms[i].magnitude) {
118 if (!any || terms[i].exponent < exponent) exponent = terms[i].exponent;
119 any = true;
120 all_same_zero = false;
121 } else if (terms[i].negative != terms[0].negative) all_same_zero = false;
122 }
123 int64_t sum = 0;
124 for (size_t i = 0; i < count; i++) {
125 if (!terms[i].magnitude) continue;
126 uint64_t aligned = terms[i].magnitude << (terms[i].exponent - exponent);
127 sum += terms[i].negative ? -(int64_t)aligned : (int64_t)aligned;
128 }
129 if (!sum) return (Term){ all_same_zero ? terms[0].negative : mode == RDN, 0, 0 };
130 return (Term){ sum < 0, (uint64_t)(sum < 0 ? -sum : sum), exponent };
131}

133static int bit_length(uint64_t value) {
134 int length = 0;
135 while (value) { length++; value >>= 1; }
136 return length;
137}

139static int floor_log(Exact value) {
140 int power = bit_length(value.numerator) - bit_length(value.denominator);
141 bool below = power >= 0 ? value.numerator < (value.denominator << power)
142 : (value.numerator << -power) < value.denominator;
143 return value.exponent + power - below;
144}

146static uint64_t integer_sqrt(uint64_t value) {
147 if (!value) return 0;
148 uint64_t low = 0, high = 1;
149 while (high <= value / high) high <<= 1;
150 while (low + 1 < high) {
151 uint64_t middle = (low + high) / 2;
152 if (middle <= value / middle) low = middle; else high = middle;
153 }
154 return low;
155}

157static int compare_unsigned(uint64_t a, uint64_t b) { return (a > b) - (a < b); }

159static Quantized quantize(Exact value, int quantum, bool square_root) {
160 uint64_t numerator = value.numerator, denominator = value.denominator;
161 int shift = value.exponent - (square_root ? 2 * quantum : quantum);
162 if (shift >= 0) numerator <<= shift; else denominator <<= -shift;
163 if (!square_root) {
164 uint64_t remainder = numerator % denominator;
165 return (Quantized){ numerator / denominator, remainder != 0, compare_unsigned(remainder, denominator - remainder) };
166 }
167 uint64_t whole = integer_sqrt(numerator / denominator);
168 uint64_t midpoint = 2 * whole + 1;
169 return (Quantized){ whole, whole * whole * denominator != numerator,
170 compare_unsigned(4 * numerator, midpoint * midpoint * denominator) };
171}

173static bool increment(Quantized value, Mode mode, bool sign) {
174 switch (mode) {
175 case RNE: return value.half > 0 || (value.half == 0 && (value.whole & 1));
176 case RNA: return value.half >= 0;
177 case RTZ: return false;
178 case RUP: return value.inexact && !sign;
179 case RDN: return value.inexact && sign;
180 }
181 return false;
182}

184static Result overflow_result(const Format *f, Mode mode, bool sign) {
185 bool infinity = mode == RNE || mode == RNA || (mode == RUP && !sign) || (mode == RDN && sign);
186 return (Result){ with_sign(f, infinity ? f->infinity : f->infinity - 1, sign), OVERFLOW | INEXACT };
187}

189static Result round_exact(const Format *f, Exact value, Mode mode, bool square_root) {
190 if (!value.numerator) return (Result){ with_sign(f, 0, value.negative), 0 };
191 int t = floor_log(value);
192 if (square_root) t = t < 0 ? -((-t + 1) / 2) : t / 2;
193 if (t > f->emax) return overflow_result(f, mode, value.negative);
194 int quantum = (t > f->emin ? t : f->emin) - f->m;
195 Quantized divided = quantize(value, quantum, square_root);
196 uint64_t significand = divided.whole + increment(divided, mode, value.negative);
197 if (significand == 2 * f->base) { significand = f->base; quantum++; }
198 if (quantum > f->emax - f->m) return overflow_result(f, mode, value.negative);
199 Flags flags = divided.inexact ? INEXACT : 0;
200 if (divided.inexact && t < f->emin) {
201 bool tiny = t < f->emin - 1;
202 if (!tiny) {
203 Quantized unbounded = quantize(value, t - f->m, square_root);
204 tiny = unbounded.whole + increment(unbounded, mode, value.negative) < 2 * f->base;
205 }
206 if (tiny) flags |= UNDERFLOW;
207 }
208 Pattern magnitude = significand < f->base ? (Pattern)significand
209 : ((Pattern)(quantum + f->m + f->bias) << f->m) | (Pattern)(significand - f->base);
210 return (Result){ with_sign(f, magnitude, value.negative), flags };
211}

213static Result rounded_term(const Format *f, Term value, Mode mode) { return round_exact(f, exact_term(value), mode, false); }
214static Result rounded_sum(const Format *f, const Term *terms, size_t count, Mode mode) {
215 return rounded_term(f, sum_terms(terms, count, mode), mode);

218static Text fraction_text(Exact value) {
219 if (!value.numerator) return text("0");
220 uint64_t numerator = value.numerator, denominator = value.denominator;
221 if (value.exponent >= 0) numerator <<= value.exponent; else denominator <<= -value.exponent;
222 if (denominator == 1) return text("%s%" PRIu64, value.negative ? "-" : "", numerator);
223 return text("%s%" PRIu64 "/%" PRIu64, value.negative ? "-" : "", numerator, denominator);
224}

226static Text decimal_text(Term term_value) {
227 Exact value = exact_term(term_value);
228 if (!value.numerator) return text("0");
229 uint64_t numerator = value.numerator, denominator = value.denominator;
230 if (value.exponent >= 0) numerator <<= value.exponent; else denominator <<= -value.exponent;
231 Text result = text("%s%" PRIu64, value.negative ? "-" : "", numerator / denominator);
232 uint64_t remainder = numerator % denominator;
233 size_t length = strlen(result.data);
234 if (remainder) result.data[length++] = '.';
235 while (remainder) {
236 remainder *= 10;
237 result.data[length++] = (char)('0' + remainder / denominator);
238 remainder %= denominator;
239 }
240 result.data[length] = '\0';
241 return result;
242}

244static Text binary_text(Pattern value, int width) {
245 Text result;
246 for (int i = 0; i < width; i++) result.data[i] = (char)('0' + ((value >> (width - i - 1)) & 1));
247 result.data[width] = '\0';
248 return result;

257static Text value_text(const Format *f, Pattern p) {
258 if (is_nan(f, p)) return text("nan");

284static Result arithmetic(const Format *f, Operation operation, Pattern a, Pattern b, Pattern c, Mode mode) {
285 if (operation == SUB) { b ^= f->sign; operation = ADD; }
286 Pattern inputs[] = { a, b, c };
287 size_t count = operation == SQRT ? 1 : operation == FMA ? 3 : 2;
288 bool nan = false;
289 Flags nan_flags = 0;
290 for (size_t i = 0; i < count; i++) {
291 nan |= is_nan(f, inputs[i]);
292 if (classify(f, inputs[i]) == SNAN) nan_flags |= INVALID;
293 }
294 if (nan) return (Result){ f->nan, nan_flags };
295 bool ai = classify(f, a) == INFINITY_VALUE, bi = classify(f, b) == INFINITY_VALUE;
296 bool sigma = negative(f, a) != negative(f, b);
297 switch (operation) {
298 case ADD:
299 if (ai && bi && sigma) return (Result){ f->nan, INVALID };
300 if (ai || bi) return (Result){ ai ? a : b, 0 };
301 return rounded_sum(f, (Term[]){ term(f, a), term(f, b) }, 2, mode);
302 case MUL:
303 if (ai || bi) return is_zero(f, a) || is_zero(f, b)
304 ? (Result){ f->nan, INVALID } : (Result){ with_sign(f, f->infinity, sigma), 0 };
305 return rounded_term(f, product(term(f, a), term(f, b)), mode);
306 case DIV:
307 if (ai) return bi ? (Result){ f->nan, INVALID } : (Result){ with_sign(f, f->infinity, sigma), 0 };
308 if (bi) return (Result){ with_sign(f, 0, sigma), 0 };
309 if (is_zero(f, b)) return is_zero(f, a) ? (Result){ f->nan, INVALID }
310 : (Result){ with_sign(f, f->infinity, sigma), DIVBYZERO };
311 if (is_zero(f, a)) return (Result){ with_sign(f, 0, sigma), 0 };
312 {
313 Term x = term(f, a), y = term(f, b);
314 return round_exact(f, exact(sigma, x.magnitude, x.exponent - y.exponent, y.magnitude), mode, false);
315 }
316 case SQRT:
317 if (is_zero(f, a)) return (Result){ a, 0 };
318 if (negative(f, a)) return (Result){ f->nan, INVALID };
319 if (ai) return (Result){ a, 0 };
320 return round_exact(f, exact_term(term(f, a)), mode, true);
321 case FMA:
322 if (ai || bi) {
323 if (is_zero(f, a) || is_zero(f, b) || (classify(f, c) == INFINITY_VALUE && negative(f, c) != sigma)) {
324 return (Result){ f->nan, INVALID };
325 }
326 return (Result){ with_sign(f, f->infinity, sigma), 0 };
327 }
328 if (classify(f, c) == INFINITY_VALUE) return (Result){ c, 0 };
329 return rounded_sum(f, (Term[]){ product(term(f, a), term(f, b)), term(f, c) }, 2, mode);
330 case SUB: break;
331 }
332 return (Result){ 0, 0 };
333}

335static bool read_operand(const Format *f, const char *word, bool encoding, Operand *operand) {
336 *operand = (Operand){0};
337 if (!strncmp(word, "0b", 2) || !strncmp(word, "0x", 2)) {
338 bool bits = word[1] == 'b';
339 if (encoding || strlen(word + 2) != (size_t)(bits ? f->w : (f->w + 3) / 4)) return false;
340 unsigned long value = strtoul(word + 2, NULL, bits ? 2 : 16);
341 if (value >= 2 * f->sign) return false;
342 operand->pattern = (Pattern)value;
343 return true;
344 }
345 if (!strcmp(word, "nan") || !strcmp(word, "inf") || !strcmp(word, "-inf")) {
346 operand->special = true;
347 operand->pattern = !strcmp(word, "nan") ? f->nan : with_sign(f, f->infinity, word[0] == '-');
348 return true;
349 }
350 bool sign = word[0] == '-';
351 const char *digits = word + sign, *slash = strchr(digits, '/');
352 uint64_t numerator = 0, denominator = 1;
353 if (slash) {
354 numerator = strtoull(digits, NULL, 10);
355 denominator = strtoull(slash + 1, NULL, 10);
356 if (!denominator) return false;
357 } else {
358 bool fractional = false;
359 for (const char *p = digits; *p; p++) {
360 if (*p == '.') { fractional = true; continue; }
361 numerator = numerator * 10 + (unsigned)(*p - '0');
362 if (fractional) denominator *= 10;
363 }
364 }
365 operand->exact = exact(sign, numerator, 0, denominator);
366 if (!encoding) operand->pattern = round_exact(f, operand->exact, RNE, false).pattern;
367 return true;
368}

370static bool checked_operand(const Format *f, const char *word, bool encoding, Operand *operand) {
371 if (read_operand(f, word, encoding, operand)) return true;
372 printf("ERROR BAD OPERAND %s\n", word);
373 return false;
374}

395static void summation(const Format *f, const char *command, char **words, int word_count) {
396 Mode mode;
397 if (!read_mode(words[2], &mode)) return;
398 Pattern values[20];
399 size_t count = 0, left_count = 0;
400 bool dot = !strcmp(command, "DOT");
401 for (int i = 3; i < word_count; i++) {
402 if (!strcmp(words[i], "|")) { left_count = count; continue; }
403 Operand operand;
404 if (!checked_operand(f, words[i], false, &operand)) return;
405 values[count++] = operand.pattern;
406 }
407 if (dot && left_count != count - left_count) { puts("ERROR LENGTH MISMATCH"); return; }
408 for (size_t i = 0; i < count; i++) if (!is_finite(f, values[i])) { puts("ERROR NOT FINITE"); return; }
409 Term terms[20];
410 if (dot) {
411 for (size_t i = 0; i < left_count; i++) terms[i] = product(term(f, values[i]), term(f, values[left_count + i]));
412 Term exact_sum = sum_terms(terms, left_count, mode);
413 Result first = arithmetic(f, MUL, values[0], values[left_count], 0, mode);
414 Result naive = first, chained = first;
415 for (size_t i = 1; i < left_count; i++) {
416 Result multiplied = arithmetic(f, MUL, values[i], values[left_count + i], 0, mode);
417 Result added = arithmetic(f, ADD, naive.pattern, multiplied.pattern, 0, mode);
418 naive = (Result){ added.pattern, naive.flags | multiplied.flags | added.flags };
419 }
420 for (size_t i = 1; i < left_count; i++) {
421 Result fused = arithmetic(f, FMA, values[i], values[left_count + i], chained.pattern, mode);
422 chained = (Result){ fused.pattern, chained.flags | fused.flags };
423 }
424 Result fused = rounded_term(f, exact_sum, mode);
425 printf("DOT %s %s n=%zu\n", f->name, mode_names[mode], left_count);
426 printf("NAIVE %s\n", result_text(f, naive).data);
427 printf("CHAINED %s\n", result_text(f, chained).data);
428 printf("EXACT %s\n", decimal_text(exact_sum).data);
429 printf("FUSED %s\n", result_text(f, fused).data);
430 printf("DIFF naive=%s chained=%s\n", difference_text(f, naive.pattern, exact_sum).data, difference_text(f, chained.pattern, exact_sum).data);
431 return;
432 }
433 for (size_t i = 0; i < count; i++) terms[i] = term(f, values[i]);
434 Term exact_sum = sum_terms(terms, count, mode);
435 Result result = { values[0], 0 };
436 Pattern compensation = 0;
437 printf("%s %s %s count=%zu\n", command, f->name, mode_names[mode], count);
438 for (size_t i = 1; i < count; i++) {
439 if (!strcmp(command, "SUM")) {
440 Result next = arithmetic(f, ADD, result.pattern, values[i], 0, mode);
441 result = (Result){ next.pattern, result.flags | next.flags };
442 printf("PARTIAL %zu %s\n", i + 1, result_text(f, next).data);
443 } else {
444 Result y = arithmetic(f, SUB, values[i], compensation, 0, mode);
445 Result t = arithmetic(f, ADD, result.pattern, y.pattern, 0, mode);
446 Result d = arithmetic(f, SUB, t.pattern, result.pattern, 0, mode);
447 Result c = arithmetic(f, SUB, d.pattern, y.pattern, 0, mode);
448 Flags step_flags = y.flags | t.flags | d.flags | c.flags;
449 result = (Result){ t.pattern, result.flags | step_flags };
450 compensation = c.pattern;
451 printf("KSTEP %zu sum=%s comp=%s %s\n", i + 1, value_text(f, result.pattern).data,
452 value_text(f, compensation).data, flags_text(step_flags).data);
453 }
454 }
455 printf("RESULT %s\n", result_text(f, result).data);
456 printf("EXACT %s\n", decimal_text(exact_sum).data);
457 printf("ROUNDED %s\n", result_text(f, rounded_term(f, exact_sum, mode)).data);
458 printf("DIFF %s\n", difference_text(f, result.pattern, exact_sum).data);
459}

461static void table_output(const Format *f) {
462 if (f->w > 8) { puts("ERROR TABLE TOO LARGE"); return; }
463 Pattern count = f->infinity;
464 printf("TABLE %s count=%" PRIu32 "\n", f->name, count);
465 for (Pattern p = 0; p < count; p++) {
466 Text gap = text("-");
467 if (p + 1 < count) {
468 Term previous = term(f, p);
469 previous.negative = true;
470 gap = decimal_text(sum_terms((Term[]){ term(f, p + 1), previous }, 2, RNE));
471 }
472 printf("ENTRY %" PRIu32 " %s gap=%s\n", p, pattern_text(f, p).data, gap.data);
473 }
474 printf("MAX %s\n", pattern_text(f, count - 1).data);
475 printf("MINNORMAL %s\n", pattern_text(f, f->base).data);
476 printf("MINSUBNORMAL %s\n", pattern_text(f, 1).data);
477 printf("EPSILON %s\n", decimal_text((Term){ false, 1, -f->m }).data);
478 printf("PATTERNS finite=%" PRIu32 " infinite=2 nan=%" PRIu32 " total=%" PRIu32 "\n", 2 * count, 2 * (f->base - 1), 2 * f->sign);
479}

493static void execute(char **words, int count) {
494 const char *command = words[0];
495 if (!strcmp(command, "FORMAT")) {
496 if (find_format(words[1])) { printf("ERROR FORMAT EXISTS %s\n", words[1]); return; }
497 int e = atoi(words[2]), m = atoi(words[3]);
498 if (e < 2 || e > 5 || m < 1 || m > 10) { puts("ERROR BAD FORMAT"); return; }
499 Format *f = add_format(words[1], e, m);
500 printf("FORMAT %s e=%d m=%d bits=%d bias=%d emin=%d emax=%d\n", f->name, f->e, f->m, f->w, f->bias, f->emin, f->emax);
501 return;
502 }
503 Format *f = checked_format(words[1]);
504 if (!f) return;
505 if (!strcmp(command, "TABLE")) { table_output(f); return; }
506 if (!strcmp(command, "SUM") || !strcmp(command, "KAHAN") || !strcmp(command, "DOT")) {
507 summation(f, command, words, count);
508 return;
509 }
510 Mode mode = RNE;
511 if (!strcmp(command, "CONVERT")) {
512 Format *g = checked_format(words[2]);
513 if (!g) return;
514 Operand a;
515 if (!checked_operand(f, words[3], false, &a)) return;
516 if (count == 5 && !read_mode(words[4], &mode)) return;
517 Result result;
518 Class kind = classify(f, a.pattern);
519 if (kind >= QNAN) result = (Result){ g->nan, kind == SNAN ? INVALID : 0 };
520 else if (kind == INFINITY_VALUE || kind == ZERO) {
521 result = (Result){ with_sign(g, kind == ZERO ? 0 : g->infinity, negative(f, a.pattern)), 0 };
522 } else result = rounded_term(g, term(f, a.pattern), mode);
523 printf("CONVERT %s %s %s %s\n", f->name, g->name, mode_names[mode], result_text(g, result).data);
524 return;
525 }
526 Operand a;
527 if (!checked_operand(f, words[2], !strcmp(command, "ENCODE"), &a)) return;
528 if (!strcmp(command, "ENCODE")) {
529 if (count == 4 && !read_mode(words[3], &mode)) return;
530 Result result = a.special ? (Result){ a.pattern, 0 } : round_exact(f, a.exact, mode, false);
531 Text source = a.special ? value_text(f, a.pattern) : fraction_text(a.exact);
532 printf("ENCODE %s %s %s %s\n", f->name, mode_names[mode], source.data, result_text(f, result).data);
533 return;
534 }
535 if (!strcmp(command, "DECODE")) {
536 Class kind = classify(f, a.pattern);
537 Pattern exponent = (a.pattern & (f->sign - 1)) >> f->m, fraction = a.pattern & (f->base - 1);
538 printf("DECODE %s %s %s %s sign=%d\n", f->name, bits_text(f, a.pattern).data,
539 hex_text(f, a.pattern).data, class_names[kind], negative(f, a.pattern));
540 if (kind <= NORMAL) {
541 Term value = term(f, a.pattern);
542 printf("FIELDS exponent=%" PRIu32 " fraction=%" PRIu32 " unbiased=%d significand=%d.%s\n",
543 exponent, fraction, value.exponent + f->m, kind == NORMAL, binary_text(fraction, f->m).data);
544 printf("VALUE %s %s\n", fraction_text(exact_term(value)).data, value_text(f, a.pattern).data);
545 } else if (kind == INFINITY_VALUE) {
546 printf("FIELDS exponent=%" PRIu32 " fraction=%" PRIu32 "\n", exponent, fraction);
547 } else {
548 printf("FIELDS exponent=%" PRIu32 " fraction=%" PRIu32 " payload=%" PRIu32 "\n", exponent, fraction, fraction % (f->base / 2));
549 }
550 return;
551 }
552 if (!strcmp(command, "COMPARE")) {
553 Operand b;
554 if (!checked_operand(f, words[3], false, &b)) return;
555 int64_t ak = order_key(f, a.pattern), bk = order_key(f, b.pattern);
556 const char *numeric = is_nan(f, a.pattern) || is_nan(f, b.pattern) ? "UN"
557 : is_zero(f, a.pattern) && is_zero(f, b.pattern) ? "EQ" : relation(ak, bk);
558 Flags flags = classify(f, a.pattern) == SNAN || classify(f, b.pattern) == SNAN ? INVALID : 0;
559 printf("COMPARE %s %s %s %s total=%s %s\n", f->name, hex_text(f, a.pattern).data,
560 hex_text(f, b.pattern).data, numeric, relation(ak, bk), flags_text(flags).data);
561 return;
562 }
563 if (!strcmp(command, "TOINT")) {
564 const IntegerType *type = NULL;
565 for (size_t i = 0; i < sizeof integer_types / sizeof *integer_types; i++) {
566 if (!strcmp(words[3], integer_types[i].name)) type = &integer_types[i];
567 }
568 if (!type) { printf("ERROR BAD TYPE %s\n", words[3]); return; }
569 if (count == 5 && !read_mode(words[4], &mode)) return;
570 bool invalid = !is_finite(f, a.pattern);
571 int64_t result = 0;
572 Flags flags = 0;
573 if (!invalid) {
574 Exact value = exact_term(term(f, a.pattern));
575 Quantized divided = quantize(value, 0, false);
576 uint64_t magnitude = divided.whole + increment(divided, mode, value.negative);
577 result = value.negative ? -(int64_t)magnitude : (int64_t)magnitude;
578 invalid = result < type->minimum || result > type->maximum;
579 flags = divided.inexact ? INEXACT : 0;
580 }
581 if (invalid) printf("TOINT %s %s %s invalid flags=invalid\n", f->name, type->name, mode_names[mode]);
582 else printf("TOINT %s %s %s %" PRId64 " %s\n", f->name, type->name, mode_names[mode], result, flags_text(flags).data);
583 return;
584 }
585 if (!strcmp(command, "NEXTUP") || !strcmp(command, "NEXTDOWN") || !strcmp(command, "ULP")) {
586 Result result = { 0, 0 };
587 Class kind = classify(f, a.pattern);
588 if (kind >= QNAN) result = (Result){ f->nan, kind == SNAN ? INVALID : 0 };
589 else if (!strcmp(command, "ULP")) {
590 result.pattern = kind == INFINITY_VALUE ? f->infinity
591 : rounded_term(f, (Term){ false, 1, term(f, a.pattern).exponent }, RNE).pattern;
592 } else if (!strcmp(command, "NEXTUP")) result.pattern = next_up(f, a.pattern);
593 else result.pattern = next_up(f, a.pattern ^ f->sign) ^ f->sign;
594 printf("%s %s %s\n", command, f->name, result_text(f, result).data);
595 return;
596 }
597 for (int i = 0; i < 6; i++) if (!strcmp(command, operation_names[i])) {
598 Operation operation = (Operation)i;
599 int arity = operation == SQRT ? 1 : operation == FMA ? 3 : 2;
600 Operand b = {0}, c = {0};
601 if (arity >= 2 && !checked_operand(f, words[3], false, &b)) return;
602 if (arity == 3 && !checked_operand(f, words[4], false, &c)) return;
603 if (count == arity + 3 && !read_mode(words[arity + 2], &mode)) return;
604 Result result = arithmetic(f, operation, a.pattern, b.pattern, c.pattern, mode);
605 printf("%s %s %s %s\n", command, f->name, mode_names[mode], result_text(f, result).data);
606 return;
607 }
608}

610int main(void) {
611 add_format("HALF", 5, 10);
612 add_format("E5M2", 5, 2);
613 add_format("E4M3", 4, 3);
614 char line[1024];
615 if (!fgets(line, sizeof line, stdin)) return 0;
616 int commands = atoi(line);
617 for (int i = 0; i < commands && fgets(line, sizeof line, stdin); i++) {
618 char *words[32];
619 int count = 0;
620 for (char *word = strtok(line, " \r\n"); word; word = strtok(NULL, " \r\n")) words[count++] = word;
621 execute(words, count);
622 }
623 return 0;
