38function bitLength(int $value): int
39{
40 $length = 0;
41 for (; $value !== 0; $value >>= 1) $length++;
42 return $length;
43}

45function scaled(int $n, int $d, int $exponent): array
46{
47 return $exponent >= 0 ? [$n << $exponent, $d] : [$n, $d << -$exponent];
48}

50function gcd(int $a, int $b): int
51{
52 while ($b !== 0) [$a, $b] = [$b, $a % $b];
53 return $a;
54}

56function integerSqrt(int $value): int
57{
58 if ($value === 0) return 0;
59 $root = 1 << intdiv(bitLength($value) + 1, 2);
60 while (true) {
61 $next = intdiv($root + intdiv($value, $root), 2);
62 if ($next >= $root) return $root;
63 $root = $next;
64 }
65}

67function increment(Mode $mode, bool $negative, int $whole, bool $inexact, int $halfway): bool
68{
69 if (!$inexact) return false;
70 return match ($mode) {
71 Mode::RNE => $halfway > 0 || $halfway === 0 && $whole % 2 !== 0,
72 Mode::RNA => $halfway >= 0,
73 Mode::RTZ => false,
74 Mode::RUP => !$negative,
75 Mode::RDN => $negative,
76 };
77}

79function roundUnits(Exact $value, int $exponent, Mode $mode): Units
80{
81 [$n, $d] = scaled($value->numerator, $value->denominator, $value->exponent - $exponent);
82 $whole = intdiv($n, $d);
83 $remainder = $n % $d;
84 if (increment($mode, $value->negative, $whole, $remainder !== 0, 2 * $remainder - $d)) $whole++;
85 return new Units($whole, $remainder !== 0);
86}
87
88function roundSqrtUnits(Exact $value, int $exponent, Mode $mode): Units
89{
90 [$n, $d] = scaled($value->numerator, $value->denominator, $value->exponent - 2 * $exponent);
91 $whole = integerSqrt(intdiv($n, $d));
92 $inexact = $whole * $whole * $d !== $n;
93 $midpoint = 2 * $whole + 1;
94 if (increment($mode, false, $whole, $inexact, 4 * $n - $d * $midpoint * $midpoint)) $whole++;
95 return new Units($whole, $inexact);
96}

98final readonly class Exact
99{
100 public function __construct(
101 public int $numerator,
102 public int $exponent = 0,
103 public int $denominator = 1,
104 public bool $negative = false,
105 ) {}
106
107 public function negated(): self
108 {
109 return new self($this->numerator, $this->exponent, $this->denominator, !$this->negative);
110 }
111
112 public function log2Floor(): int
113 {
114 $t = bitLength($this->numerator) - bitLength($this->denominator) + $this->exponent;
115 [$n, $d] = scaled($this->numerator, $this->denominator, $this->exponent - $t);
116 return $n < $d ? $t - 1 : $t;
117 }
118
119 public function fraction(): string
120 {
121 if ($this->numerator === 0) return '0';
122 [$n, $d] = scaled($this->numerator, $this->denominator, $this->exponent);
123 $factor = gcd($n, $d);
124 $n = intdiv($n, $factor);
125 $d = intdiv($d, $factor);
126 return ($this->negative ? '-' : '') . $n . ($d === 1 ? '' : '/' . $d);
127 }
128
129 public function decimal(bool $keepZeroSign = false): string
130 {
131 [$n, $d] = scaled($this->numerator, $this->denominator, $this->exponent);
132 $text = ($this->negative && ($this->numerator !== 0 || $keepZeroSign) ? '-' : '') . intdiv($n, $d);
133 $remainder = $n % $d;
134 if ($remainder !== 0) {
135 $text .= '.';
136 while ($remainder !== 0) {
137 $remainder *= 10;
138 $text .= chr(48 + intdiv($remainder, $d));
139 $remainder %= $d;
140 }
141 }
142 return $text;
143 }
144
145 public static function product(self $a, self $b): self
146 {
147 return new self($a->numerator * $b->numerator, $a->exponent + $b->exponent,
148 $a->denominator * $b->denominator, $a->negative !== $b->negative);
149 }
150
151 public static function sum(array $terms, Mode $mode): self
152 {
153 $exponent = null;
154 foreach ($terms as $term) {
155 if ($term->numerator !== 0 && ($exponent === null || $term->exponent < $exponent)) $exponent = $term->exponent;
156 }
157 if ($exponent === null) {
158 $same = true;
159 foreach ($terms as $term) $same = $same && $term->negative === $terms[0]->negative;
160 return new self(0, negative: $same ? $terms[0]->negative : $mode === Mode::RDN);
161 }
162 $numerator = 0;
163 foreach ($terms as $term) {
164 if ($term->numerator === 0) continue;
165 $aligned = $term->numerator << ($term->exponent - $exponent);
166 $numerator += $term->negative ? -$aligned : $aligned;
167 }
168 return new self(abs($numerator), $exponent, negative: $numerator < 0 || $numerator === 0 && $mode === Mode::RDN);
169 }
170}

175final readonly class MiniFormat
176{
177 public int $width, $bias, $minExponent, $maxExponent, $hidden, $exponentMask, $signBit, $infinity;
178
179 public function __construct(public string $name, public int $e, public int $m)
180 {
181 $this->width = 1 + $e + $m;
182 $this->bias = (1 << ($e - 1)) - 1;
183 $this->minExponent = 1 - $this->bias;
184 $this->maxExponent = $this->bias;
185 $this->hidden = 1 << $m;
186 $this->exponentMask = (1 << $e) - 1;
187 $this->signBit = 1 << ($e + $m);
188 $this->infinity = $this->exponentMask * $this->hidden;
189 }
190
191 public function fields(int $pattern): Fields
192 {
193 $exponent = ($pattern >> $this->m) & $this->exponentMask;
194 return new Fields(($pattern & $this->signBit) !== 0, $exponent, $pattern & ($this->hidden - 1),
195 ($exponent === 0 ? $this->minExponent : $exponent - $this->bias) - $this->m);
196 }
197 public function isNaN(int $pattern): bool { return $this->fields($pattern)->exponent === $this->exponentMask && $this->fields($pattern)->fraction !== 0; }
198 public function isSignaling(int $pattern): bool { return $this->isNaN($pattern) && $this->fields($pattern)->fraction < intdiv($this->hidden, 2); }
199 public function isInfinity(int $pattern): bool { return ($pattern & ($this->signBit - 1)) === $this->infinity; }
200 public function isZero(int $pattern): bool { return ($pattern & ($this->signBit - 1)) === 0; }
201 public function isFinite(int $pattern): bool { return $this->fields($pattern)->exponent !== $this->exponentMask; }
202 public function nan(): int { return $this->infinity + intdiv($this->hidden, 2); }
203 public function zero(bool $negative): int { return $negative ? $this->signBit : 0; }
204 public function inf(bool $negative): int { return $this->zero($negative) + $this->infinity; }
205 public function flip(int $pattern): int { return $pattern ^ $this->signBit; }
206
207 public function exact(int $pattern): Exact
208 {
209 $f = $this->fields($pattern);
210 return new Exact($f->exponent === 0 ? $f->fraction : $this->hidden + $f->fraction, $f->q, negative: $f->negative);
211 }
212 public function bits(int $pattern): string
213 {
214 $f = $this->fields($pattern);
215 return ($f->negative ? '1' : '0') . '|' . binary($f->exponent, $this->e) . '|' . binary($f->fraction, $this->m);
216 }
217 public function hex(int $pattern): string
218 {
219 return '0x' . str_pad(strtoupper(dechex($pattern)), intdiv($this->width + 3, 4), '0', STR_PAD_LEFT);
220 }
221 public function value(int $pattern): string
222 {
223 if ($this->isNaN($pattern)) return 'nan';
224 if ($this->isInfinity($pattern)) return $this->fields($pattern)->negative ? '-inf' : 'inf';
225 return $this->exact($pattern)->decimal(true);
226 }
227 public function patternText(int $pattern): string { return $this->bits($pattern) . ' ' . $this->hex($pattern) . ' ' . $this->value($pattern); }
228 public function resultText(Result $result): string { return $this->patternText($result->pattern) . ' ' . Flags::text($result->flags); }
229
230 private function overflow(bool $negative, Mode $mode): Result
231 {
232 $infinity = $mode === Mode::RNE || $mode === Mode::RNA || $mode === Mode::RUP && !$negative || $mode === Mode::RDN && $negative;
233 return new Result($this->zero($negative) + ($infinity ? $this->infinity : $this->infinity - 1), Flags::OVERFLOW | Flags::INEXACT);
234 }
235
236 private function roundNonzero(bool $negative, int $t, Mode $mode, Closure $unitsAt): Result
237 {
238 if ($t > $this->maxExponent) return $this->overflow($negative, $mode);
239 $exponent = max($t, $this->minExponent) - $this->m;
240 $units = $unitsAt($exponent);
241 $n = $units->whole;
242 if ($n === 2 * $this->hidden) { $n = $this->hidden; $exponent++; }
243 if ($exponent > $this->maxExponent - $this->m) return $this->overflow($negative, $mode);
244 $magnitude = $n >= $this->hidden ? ($exponent + $this->m + $this->bias) * $this->hidden + $n - $this->hidden : $n;
245 $flags = $units->inexact ? Flags::INEXACT : 0;
246 if ($units->inexact && $t < $this->minExponent &&
247 ($t < $this->minExponent - 1 || $unitsAt($t - $this->m)->whole < 2 * $this->hidden)) $flags |= Flags::UNDERFLOW;
248 return new Result($this->zero($negative) + $magnitude, $flags);
249 }
250
251 public function round(Exact $value, Mode $mode): Result
252 {
253 if ($value->numerator === 0) return new Result($this->zero($value->negative));
254 return $this->roundNonzero($value->negative, $value->log2Floor(), $mode,
255 static fn(int $q): Units => roundUnits($value, $q, $mode));
256 }
257 public function sqrt(Exact $value, Mode $mode): Result
258 {
259 $log = $value->log2Floor();
260 $t = $log >= 0 ? intdiv($log, 2) : -intdiv(-$log + 1, 2);
261 return $this->roundNonzero(false, $t, $mode, static fn(int $q): Units => roundSqrtUnits($value, $q, $mode));
262 }
263}

289function realInput(string $word): RealInput
290{
291 if ($word === 'nan') return new RealInput(RealKind::NaN, new Exact(0));
292 if ($word === 'inf' || $word === '-inf') return new RealInput(RealKind::Infinity, new Exact(0, negative: $word === '-inf'));
293 if (str_starts_with($word, '0b') || str_starts_with($word, '0x')) fail('BAD OPERAND ' . $word);
294 $negative = $word[0] === '-';
295 $body = $negative ? substr($word, 1) : $word;
296 $slash = strpos($body, '/');
297 if ($slash !== false) {
298 $n = (int)substr($body, 0, $slash);
299 $d = (int)substr($body, $slash + 1);
300 if ($d === 0) fail('BAD OPERAND ' . $word);
301 return new RealInput(RealKind::Finite, new Exact($n, denominator: $d, negative: $negative));
302 }
303 $dot = strpos($body, '.');
304 $denominator = 1;
305 if ($dot !== false) {
306 for ($i = $dot + 1; $i < strlen($body); $i++) $denominator *= 10;
307 $body = str_replace('.', '', $body);
308 }
309 return new RealInput(RealKind::Finite, new Exact((int)$body, denominator: $denominator, negative: $negative));
310}

312function operand(MiniFormat $f, string $word): int
313{
314 $bits = str_starts_with($word, '0b');
315 $hex = str_starts_with($word, '0x');
316 if ($bits || $hex) {
317 if (strlen($word) - 2 !== ($bits ? $f->width : intdiv($f->width + 3, 4))) fail('BAD OPERAND ' . $word);
318 $pattern = intval(substr($word, 2), $bits ? 2 : 16);
319 if ($pattern >= (1 << $f->width)) fail('BAD OPERAND ' . $word);
320 return $pattern;
321 }
322 return realInput($word)->encoded($f, Mode::RNE)->pattern;
323}

325function calculate(MiniFormat $f, Operation $operation, Mode $mode, array $inputs): Result
326{
327 $anyNaN = false;
328 $signaling = false;
329 foreach ($inputs as $pattern) {
330 $anyNaN = $anyNaN || $f->isNaN($pattern);
331 $signaling = $signaling || $f->isSignaling($pattern);
332 }
333 if ($anyNaN) return new Result($f->nan(), $signaling ? Flags::INVALID : 0);
334 $a = $inputs[0];
335 $negative = $f->fields($a)->negative;
336 $invalid = new Result($f->nan(), Flags::INVALID);
337 if ($operation === Operation::SQRT) {
338 if ($f->isZero($a)) return new Result($a);
339 if ($negative) return $invalid;
340 if ($f->isInfinity($a)) return new Result($a);
341 return $f->sqrt($f->exact($a), $mode);
342 }
343 $b = $operation === Operation::SUB ? $f->flip($inputs[1]) : $inputs[1];
344 $sigma = $negative !== $f->fields($b)->negative;
345 if ($operation === Operation::ADD || $operation === Operation::SUB) {
346 if ($f->isInfinity($a) && $f->isInfinity($b) && $sigma) return $invalid;
347 if ($f->isInfinity($a)) return new Result($a);
348 if ($f->isInfinity($b)) return new Result($b);
349 return $f->round(Exact::sum([$f->exact($a), $f->exact($b)], $mode), $mode);
350 }
351 if ($operation === Operation::MUL) {
352 if ($f->isInfinity($a) || $f->isInfinity($b)) return $f->isZero($a) || $f->isZero($b) ? $invalid : new Result($f->inf($sigma));
353 if ($f->isZero($a) || $f->isZero($b)) return new Result($f->zero($sigma));
354 return $f->round(Exact::product($f->exact($a), $f->exact($b)), $mode);
355 }
356 if ($operation === Operation::DIV) {
357 if ($f->isInfinity($a)) return $f->isInfinity($b) ? $invalid : new Result($f->inf($sigma));
358 if ($f->isInfinity($b)) return new Result($f->zero($sigma));
359 if ($f->isZero($b)) return $f->isZero($a) ? $invalid : new Result($f->inf($sigma), Flags::DIVBYZERO);
360 if ($f->isZero($a)) return new Result($f->zero($sigma));
361 $x = $f->exact($a);
362 $y = $f->exact($b);
363 return $f->round(new Exact($x->numerator * $y->denominator, $x->exponent - $y->exponent,
364 $x->denominator * $y->numerator, $sigma), $mode);
365 }
366 $c = $inputs[2];
367 if ($f->isInfinity($a) || $f->isInfinity($b)) {
368 if ($f->isZero($a) || $f->isZero($b) || $f->isInfinity($c) && $f->fields($c)->negative !== $sigma) return $invalid;
369 return new Result($f->inf($sigma));
370 }
371 if ($f->isInfinity($c)) return new Result($c);
372 return $f->round(Exact::sum([Exact::product($f->exact($a), $f->exact($b)), $f->exact($c)], $mode), $mode);
373}

423 private function summation(string $command, MiniFormat $f, Mode $mode, array $patterns): void
424 {
425 foreach ($patterns as $pattern) if (!$f->isFinite($pattern)) fail('NOT FINITE');
426 $count = count($patterns);
427 $this->emit("$command {$f->name} {$mode->value} count=$count");
428 $sum = $patterns[0];
429 $compensation = 0;
430 $flags = 0;
431 for ($i = 1; $i < $count; $i++) {
432 $stepNumber = $i + 1;
433 if ($command === 'SUM') {
434 $added = calculate($f, Operation::ADD, $mode, [$sum, $patterns[$i]]);
435 $sum = $added->pattern;
436 $flags |= $added->flags;
437 $this->emit("PARTIAL $stepNumber " . $f->resultText($added));
438 } else {
439 $y = calculate($f, Operation::SUB, $mode, [$patterns[$i], $compensation]);
440 $t = calculate($f, Operation::ADD, $mode, [$sum, $y->pattern]);
441 $d = calculate($f, Operation::SUB, $mode, [$t->pattern, $sum]);
442 $c = calculate($f, Operation::SUB, $mode, [$d->pattern, $y->pattern]);
443 $sum = $t->pattern;
444 $compensation = $c->pattern;
445 $step = $y->flags | $t->flags | $d->flags | $c->flags;
446 $flags |= $step;
447 $this->emit("KSTEP $stepNumber sum=" . $f->value($sum) . ' comp=' . $f->value($compensation) . ' ' . Flags::text($step));
448 }
449 }
450 $this->summary($f, $mode, $patterns, new Result($sum, $flags));
451 }

453 private function dot(MiniFormat $f, Mode $mode, array $a, array $b): void
454 {
455 if (count($a) !== count($b)) fail('LENGTH MISMATCH');
456 foreach ($a as $pattern) if (!$f->isFinite($pattern)) fail('NOT FINITE');
457 foreach ($b as $pattern) if (!$f->isFinite($pattern)) fail('NOT FINITE');
458 $count = count($a);
459 $naive = calculate($f, Operation::MUL, $mode, [$a[0], $b[0]]);
460 $chained = $naive;
461 for ($i = 1; $i < $count; $i++) {
462 $product = calculate($f, Operation::MUL, $mode, [$a[$i], $b[$i]]);
463 $added = calculate($f, Operation::ADD, $mode, [$naive->pattern, $product->pattern]);
464 $naive = new Result($added->pattern, $naive->flags | $product->flags | $added->flags);
465 $fused = calculate($f, Operation::FMA, $mode, [$a[$i], $b[$i], $chained->pattern]);
466 $chained = new Result($fused->pattern, $chained->flags | $fused->flags);
467 }
468 $terms = [];
469 for ($i = 0; $i < $count; $i++) $terms[] = Exact::product($f->exact($a[$i]), $f->exact($b[$i]));
470 $exact = Exact::sum($terms, $mode);
471 $this->emit("DOT {$f->name} {$mode->value} n=$count");
472 $this->emit('NAIVE ' . $f->resultText($naive));
473 $this->emit('CHAINED ' . $f->resultText($chained));
474 $this->emit('EXACT ' . $exact->decimal());
475 $this->emit('FUSED ' . $f->resultText($f->round($exact, $mode)));
476 $this->emit('DIFF naive=' . difference($f, $naive->pattern, $exact) . ' chained=' . difference($f, $chained->pattern, $exact));
477 }

479 private function table(MiniFormat $f): void
480 {
481 if ($f->width > 8) fail('TABLE TOO LARGE');
482 $count = $f->infinity;
483 $this->emit("TABLE {$f->name} count=$count");
484 for ($pattern = 0; $pattern < $count; $pattern++) {
485 $gap = $pattern === $count - 1 ? '-'
486 : Exact::sum([$f->exact($pattern + 1), $f->exact($pattern)->negated()], Mode::RNE)->decimal();
487 $this->emit("ENTRY $pattern " . $f->patternText($pattern) . " gap=$gap");
488 }
489 $this->emit('MAX ' . $f->patternText($f->infinity - 1));
490 $this->emit('MINNORMAL ' . $f->patternText($f->hidden));
491 $this->emit('MINSUBNORMAL ' . $f->patternText(1));
492 $this->emit('EPSILON ' . (new Exact(1, -$f->m))->decimal());
493 $this->emit('PATTERNS finite=' . (2 * $count) . ' infinite=2 nan=' . (2 * ($f->hidden - 1)) . ' total=' . (1 << $f->width));
494 }

496 public function execute(string $line): array
497 {
498 $this->output = [];
499 $words = explode(' ', $line);
500 $command = $words[0];
501 if ($command === 'FORMAT') {
502 $name = $words[1];
503 $e = (int)$words[2];
504 $m = (int)$words[3];
505 if (isset($this->formats[$name])) fail('FORMAT EXISTS ' . $name);
506 if ($e < 2 || $e > 5 || $m < 1 || $m > 10) fail('BAD FORMAT');
507 $f = new MiniFormat($name, $e, $m);
508 $this->formats[$name] = $f;
509 $this->emit("FORMAT $name e=$e m=$m bits={$f->width} bias={$f->bias} emin={$f->minExponent} emax={$f->maxExponent}");
510 return $this->output;
511 }
512 $f = $this->format($words[1]);
513 $operation = Operation::tryFrom($command);
514 if ($command === 'ENCODE') {
515 $input = realInput($words[2]);
516 $mode = modeAt($words, 3);
517 $this->emit("ENCODE {$f->name} {$mode->value} " . $input->text() . ' ' . $f->resultText($input->encoded($f, $mode)));
518 } elseif ($command === 'CONVERT') {
519 $g = $this->format($words[2]);
520 $a = operand($f, $words[3]);
521 $mode = modeAt($words, 4);
522 if ($f->isNaN($a)) $result = new Result($g->nan(), $f->isSignaling($a) ? Flags::INVALID : 0);
523 elseif ($f->isInfinity($a)) $result = new Result($g->inf($f->fields($a)->negative));
524 else $result = $g->round($f->exact($a), $mode);
525 $this->emit("CONVERT {$f->name} {$g->name} {$mode->value} " . $g->resultText($result));
526 } elseif ($command === 'SUM' || $command === 'KAHAN' || $command === 'DOT') {
527 $mode = modeAt($words, 2);
528 if ($command === 'DOT') {
529 $separator = array_search('|', $words, true);
530 $a = $this->operands($f, $words, 3, $separator);
531 $b = $this->operands($f, $words, $separator + 1, count($words));
532 $this->dot($f, $mode, $a, $b);
533 } else $this->summation($command, $f, $mode, $this->operands($f, $words, 3, count($words)));
534 } elseif ($command === 'TABLE') {
535 $this->table($f);
536 } elseif ($operation !== null) {
537 $count = $operation === Operation::SQRT ? 1 : ($operation === Operation::FMA ? 3 : 2);
538 $patterns = $this->operands($f, $words, 2, 2 + $count);
539 $mode = modeAt($words, 2 + $count);
540 $this->emit("$command {$f->name} {$mode->value} " . $f->resultText(calculate($f, $operation, $mode, $patterns)));
541 } else {
542 $a = operand($f, $words[2]);
543 $fields = $f->fields($a);
544 if ($command === 'DECODE') {
545 if ($f->isNaN($a)) $classification = $f->isSignaling($a) ? 'snan' : 'qnan';
546 elseif ($f->isInfinity($a)) $classification = 'infinity';
547 elseif ($f->isZero($a)) $classification = 'zero';
548 else $classification = $fields->exponent === 0 ? 'subnormal' : 'normal';
549 $this->emit("DECODE {$f->name} " . $f->bits($a) . ' ' . $f->hex($a) . " $classification sign=" . ($fields->negative ? 1 : 0));
550 $heading = "FIELDS exponent={$fields->exponent} fraction={$fields->fraction}";
551 if ($f->isFinite($a)) {
552 $this->emit($heading . ' unbiased=' . ($fields->q + $f->m) . ' significand=' . ($fields->exponent === 0 ? '0.' : '1.') . binary($fields->fraction, $f->m));
553 $this->emit('VALUE ' . $f->exact($a)->fraction() . ' ' . $f->value($a));
554 } else $this->emit($f->isNaN($a) ? $heading . ' payload=' . ($fields->fraction % intdiv($f->hidden, 2)) : $heading);
555 } elseif ($command === 'COMPARE') {
556 $b = operand($f, $words[3]);
557 $key = static fn(int $p): int => $f->fields($p)->negative ? -($p - $f->signBit) - 1 : $p;
558 $order = relation($key($a), $key($b));
559 $numeric = $f->isNaN($a) || $f->isNaN($b) ? 'UN' : ($f->isZero($a) && $f->isZero($b) ? 'EQ' : $order);
560 $flags = $f->isSignaling($a) || $f->isSignaling($b) ? Flags::INVALID : 0;
561 $this->emit("COMPARE {$f->name} " . $f->hex($a) . ' ' . $f->hex($b) . " $numeric total=$order " . Flags::text($flags));
562 } elseif ($command === 'TOINT') {
563 [$minimum, $maximum] = self::INTEGER_TYPES[$words[3]] ?? fail('BAD TYPE ' . $words[3]);
564 $mode = modeAt($words, 4);
565 $prefix = "TOINT {$f->name} {$words[3]} {$mode->value}";
566 if (!$f->isFinite($a)) $this->emit($prefix . ' invalid flags=invalid');
567 else {
568 $units = roundUnits($f->exact($a), 0, $mode);
569 $integer = $fields->negative ? -$units->whole : $units->whole;
570 $this->emit($integer < $minimum || $integer > $maximum ? $prefix . ' invalid flags=invalid'
571 : $prefix . " $integer " . Flags::text($units->inexact ? Flags::INEXACT : 0));
572 }
573 } else {
574 if ($f->isNaN($a)) $result = new Result($f->nan(), $f->isSignaling($a) ? Flags::INVALID : 0);
575 elseif ($command === 'NEXTUP') $result = new Result(nextUp($f, $a));
576 elseif ($command === 'NEXTDOWN') $result = new Result($f->flip(nextUp($f, $f->flip($a))));
577 else $result = new Result($f->isInfinity($a) ? $f->infinity : $f->round(new Exact(1, $fields->q), Mode::RNE)->pattern);
578 $this->emit("$command {$f->name} " . $f->resultText($result));
579 }
580 }
581 return $this->output;
582 }
583}
