1local flags = { invalid = 1, divbyzero = 2, overflow = 4, underflow = 8, inexact = 16 }
2local modes = { RNE = true, RNA = true, RTZ = true, RUP = true, RDN = true }
3local operations = { ADD = 2, SUB = 2, MUL = 2, DIV = 2, SQRT = 1, FMA = 3 }
4local integer_types = {
5 I8 = {-128, 127}, U8 = {0, 255}, I16 = {-32768, 32767},
6 U16 = {0, 65535}, I32 = {-2147483648, 2147483647}, U32 = {0, 4294967295},
7}

9local function fail(format, ...)
10 error({command_error = true, message = string.format(format, ...)}, 0)
11end

13local function flags_text(value)
14 local names = {}
15 for i, name in ipairs({"invalid", "divbyzero", "overflow", "underflow", "inexact"}) do
16 if (value & (1 << (i - 1))) ~= 0 then names[#names + 1] = name end
17 end
18 return "flags=" .. (#names == 0 and "none" or table.concat(names, ","))
19end

21local function result(pattern, raised)
22 return {pattern = pattern, flags = raised or 0}
23end

25local function scaled(n, d, exponent)
26 if exponent >= 0 then return n << exponent, d end
27 return n, d << -exponent
28end
29
30local function bit_length(value)
31 local length = 0
32 while value ~= 0 do value, length = value >> 1, length + 1 end
33 return length
34end
35
36local function gcd(a, b)
37 while b ~= 0 do a, b = b, a % b end
38 return a
39end

41local function integer_sqrt(value)
42 if value == 0 then return 0 end
43 local root = 1 << ((bit_length(value) + 1) // 2)
44 while true do
45 local next_root = (root + value // root) // 2
46 if next_root >= root then return root end
47 root = next_root
48 end
49end

51local function increment(mode, negative, whole, inexact, halfway)
52 if not inexact then return false end
53 if mode == "RNE" then return halfway > 0 or halfway == 0 and whole % 2 ~= 0 end
54 if mode == "RNA" then return halfway >= 0 end
55 if mode == "RUP" then return not negative end
56 if mode == "RDN" then return negative end
57 return false
58end
59
60local function round_units(value, exponent, mode)
61 local n, d = scaled(value.n, value.d, value.exponent - exponent)
62 local whole, remainder = n // d, n % d
63 if increment(mode, value.negative, whole, remainder ~= 0, 2 * remainder - d) then whole = whole + 1 end
64 return whole, remainder ~= 0
65end
66
67local function round_sqrt_units(value, exponent, mode)
68 local n, d = scaled(value.n, value.d, value.exponent - 2 * exponent)
69 local whole = integer_sqrt(n // d)
70 local inexact = whole * whole * d ~= n
71 local midpoint = 2 * whole + 1
72 if increment(mode, false, whole, inexact, 4 * n - d * midpoint * midpoint) then whole = whole + 1 end
73 return whole, inexact
74end

76local Exact = {}
77Exact.__index = Exact
78
79function Exact.new(n, exponent, d, negative)
80 return setmetatable({n = n, exponent = exponent or 0, d = d or 1, negative = negative or false}, Exact)
81end
82
83function Exact:negated()
84 return Exact.new(self.n, self.exponent, self.d, not self.negative)
85end
86
87function Exact:log2_floor()
88 local t = bit_length(self.n) - bit_length(self.d) + self.exponent
89 local n, d = scaled(self.n, self.d, self.exponent - t)
90 return n < d and t - 1 or t
91end
92
93function Exact:fraction()
94 if self.n == 0 then return "0" end
95 local n, d = scaled(self.n, self.d, self.exponent)
96 local factor = gcd(n, d)
97 n, d = n // factor, d // factor
98 return (self.negative and "-" or "") .. n .. (d == 1 and "" or "/" .. d)
99end
100
101function Exact:decimal(keep_zero_sign)
102 local n, d = scaled(self.n, self.d, self.exponent)
103 local text = {(self.negative and (self.n ~= 0 or keep_zero_sign) and "-" or "") .. (n // d)}
104 local remainder = n % d
105 if remainder ~= 0 then
106 text[#text + 1] = "."
107 while remainder ~= 0 do
108 remainder = remainder * 10
109 text[#text + 1] = string.char(48 + remainder // d)
110 remainder = remainder % d
111 end
112 end
113 return table.concat(text)
114end
115
116function Exact.product(a, b)
117 return Exact.new(a.n * b.n, a.exponent + b.exponent, a.d * b.d, a.negative ~= b.negative)
118end
119
120function Exact.sum(terms, mode)
121 local exponent
122 for _, term in ipairs(terms) do
123 if term.n ~= 0 and (exponent == nil or term.exponent < exponent) then exponent = term.exponent end
124 end
125 if exponent == nil then
126 local negative, same = terms[1].negative, true
127 for _, term in ipairs(terms) do same = same and term.negative == terms[1].negative end
128 if not same then negative = mode == "RDN" end
129 return Exact.new(0, 0, 1, negative)
130 end
131 local numerator = 0
132 for _, term in ipairs(terms) do
133 if term.n ~= 0 then
134 local aligned = term.n << (term.exponent - exponent)
135 numerator = numerator + (term.negative and -aligned or aligned)
136 end
137 end
138 return Exact.new(math.abs(numerator), exponent, 1, numerator < 0 or numerator == 0 and mode == "RDN")
139end

141local function binary(value, width)
142 local digits = {}
143 for i = width - 1, 0, -1 do digits[#digits + 1] = ((value >> i) & 1) == 0 and "0" or "1" end
144 return table.concat(digits)
145end

147local function relation(a, b)
148 return a < b and "LT" or a > b and "GT" or "EQ"
149end

152MiniFormat.__index = MiniFormat

154function MiniFormat.new(name, e, m)
155 local bias, hidden, mask = (1 << (e - 1)) - 1, 1 << m, (1 << e) - 1
156 return setmetatable({name = name, e = e, m = m, width = 1 + e + m, bias = bias,
157 emin = 1 - bias, emax = bias, hidden = hidden, exponent_mask = mask,
158 sign_bit = 1 << (e + m), infinity = mask * hidden}, MiniFormat)
159end
160
161function MiniFormat:fields(pattern)
162 local exponent = (pattern >> self.m) & self.exponent_mask
163 return {negative = (pattern & self.sign_bit) ~= 0, exponent = exponent, fraction = pattern & (self.hidden - 1),
164 q = (exponent == 0 and self.emin or exponent - self.bias) - self.m}
165end
166function MiniFormat:is_nan(pattern)
167 local f = self:fields(pattern)
168 return f.exponent == self.exponent_mask and f.fraction ~= 0
169end
170function MiniFormat:is_signaling(pattern) return self:is_nan(pattern) and self:fields(pattern).fraction < self.hidden // 2 end
171function MiniFormat:is_infinity(pattern) return (pattern & (self.sign_bit - 1)) == self.infinity end
172function MiniFormat:is_zero(pattern) return (pattern & (self.sign_bit - 1)) == 0 end
173function MiniFormat:is_finite(pattern) return self:fields(pattern).exponent ~= self.exponent_mask end
174function MiniFormat:nan() return self.infinity + self.hidden // 2 end
175function MiniFormat:zero(negative) return negative and self.sign_bit or 0 end
176function MiniFormat:inf(negative) return self:zero(negative) + self.infinity end
177function MiniFormat:flip(pattern) return pattern ~ self.sign_bit end
178
179function MiniFormat:exact(pattern)
180 local f = self:fields(pattern)
181 return Exact.new(f.exponent == 0 and f.fraction or self.hidden + f.fraction, f.q, 1, f.negative)
182end
183function MiniFormat:bits(pattern)
184 local f = self:fields(pattern)
185 return (f.negative and "1" or "0") .. "|" .. binary(f.exponent, self.e) .. "|" .. binary(f.fraction, self.m)
186end
187function MiniFormat:hex(pattern) return string.format("0x%0" .. ((self.width + 3) // 4) .. "X", pattern) end
188function MiniFormat:value(pattern)
189 if self:is_nan(pattern) then return "nan" end
190 if self:is_infinity(pattern) then return self:fields(pattern).negative and "-inf" or "inf" end
191 return self:exact(pattern):decimal(true)
192end
193function MiniFormat:pattern_text(pattern) return self:bits(pattern) .. " " .. self:hex(pattern) .. " " .. self:value(pattern) end
194function MiniFormat:result_text(value) return self:pattern_text(value.pattern) .. " " .. flags_text(value.flags) end
195
196function MiniFormat:overflow(negative, mode)
197 local infinity = mode == "RNE" or mode == "RNA" or mode == "RUP" and not negative or mode == "RDN" and negative
198 return result(self:zero(negative) + (infinity and self.infinity or self.infinity - 1), flags.overflow | flags.inexact)
199end
200
201function MiniFormat:round_nonzero(negative, t, mode, units_at)
202 if t > self.emax then return self:overflow(negative, mode) end
203 local exponent = math.max(t, self.emin) - self.m
204 local n, inexact = units_at(exponent)
205 if n == 2 * self.hidden then n, exponent = self.hidden, exponent + 1 end
206 if exponent > self.emax - self.m then return self:overflow(negative, mode) end
207 local magnitude = n >= self.hidden and (exponent + self.m + self.bias) * self.hidden + n - self.hidden or n
208 local raised = inexact and flags.inexact or 0
209 if inexact and t < self.emin then
210 local tiny = t < self.emin - 1
211 if not tiny then
212 local unbounded = units_at(t - self.m)
213 tiny = unbounded < 2 * self.hidden
214 end
215 if tiny then raised = raised | flags.underflow end
216 end
217 return result(self:zero(negative) + magnitude, raised)
218end
219
220function MiniFormat:round(value, mode)
221 if value.n == 0 then return result(self:zero(value.negative)) end
222 return self:round_nonzero(value.negative, value:log2_floor(), mode, function(q) return round_units(value, q, mode) end)
223end
224function MiniFormat:sqrt(value, mode)
225 return self:round_nonzero(false, value:log2_floor() // 2, mode, function(q) return round_sqrt_units(value, q, mode) end)
226end

228local function mode_at(words, index)
229 if words[index] == nil then return "RNE" end
230 if not modes[words[index]] then fail("BAD MODE %s", words[index]) end
231 return words[index]
232end

234local function real_input(word)
235 if word == "nan" then return {kind = "nan", number = Exact.new(0)} end
236 if word == "inf" or word == "-inf" then return {kind = "infinity", number = Exact.new(0, 0, 1, word == "-inf")} end
237 local prefix = word:sub(1, 2)
238 if prefix == "0b" or prefix == "0x" then fail("BAD OPERAND %s", word) end
239 local negative = word:sub(1, 1) == "-"
240 local body = negative and word:sub(2) or word
241 local slash = body:find("/", 1, true)
242 if slash then
243 local n, d = tonumber(body:sub(1, slash - 1), 10), tonumber(body:sub(slash + 1), 10)
244 if d == 0 then fail("BAD OPERAND %s", word) end
245 return {kind = "finite", number = Exact.new(n, 0, d, negative)}
246 end
247 local dot = body:find(".", 1, true)
248 local denominator = 1
249 if dot then
250 for _ = dot + 1, #body do denominator = denominator * 10 end
251 body = body:gsub("%.", "")
252 end
253 return {kind = "finite", number = Exact.new(tonumber(body, 10), 0, denominator, negative)}
254end

256local function input_text(input)
257 if input.kind == "nan" then return "nan" end
258 if input.kind == "infinity" then return input.number.negative and "-inf" or "inf" end
259 return input.number:fraction()
260end

262local function encoded(f, input, mode)
263 if input.kind == "nan" then return result(f:nan()) end
264 if input.kind == "infinity" then return result(f:inf(input.number.negative)) end
265 return f:round(input.number, mode)
266end

268local function operand(f, word)
269 local prefix = word:sub(1, 2)
270 if prefix == "0b" or prefix == "0x" then
271 local bits = prefix == "0b"
272 if #word - 2 ~= (bits and f.width or (f.width + 3) // 4) then fail("BAD OPERAND %s", word) end
273 local pattern = tonumber(word:sub(3), bits and 2 or 16)
274 if pattern >= (1 << f.width) then fail("BAD OPERAND %s", word) end
275 return pattern
276 end
277 return encoded(f, real_input(word), "RNE").pattern
278end

280local function calculate(f, operation, mode, inputs)
281 local any_nan, signaling = false, false
282 for _, pattern in ipairs(inputs) do
283 any_nan = any_nan or f:is_nan(pattern)
284 signaling = signaling or f:is_signaling(pattern)
285 end
286 if any_nan then return result(f:nan(), signaling and flags.invalid or 0) end
287 local a = inputs[1]
288 local negative = f:fields(a).negative
289 local invalid = result(f:nan(), flags.invalid)
290 if operation == "SQRT" then
291 if f:is_zero(a) then return result(a) end
292 if negative then return invalid end
293 if f:is_infinity(a) then return result(a) end
294 return f:sqrt(f:exact(a), mode)
295 end
296 local b = operation == "SUB" and f:flip(inputs[2]) or inputs[2]
297 local sigma = negative ~= f:fields(b).negative
298 if operation == "ADD" or operation == "SUB" then
299 if f:is_infinity(a) and f:is_infinity(b) and sigma then return invalid end
300 if f:is_infinity(a) then return result(a) end
301 if f:is_infinity(b) then return result(b) end
302 return f:round(Exact.sum({f:exact(a), f:exact(b)}, mode), mode)
303 end
304 if operation == "MUL" then
305 if f:is_infinity(a) or f:is_infinity(b) then
306 return (f:is_zero(a) or f:is_zero(b)) and invalid or result(f:inf(sigma))
307 end
308 if f:is_zero(a) or f:is_zero(b) then return result(f:zero(sigma)) end
309 return f:round(Exact.product(f:exact(a), f:exact(b)), mode)
310 end
311 if operation == "DIV" then
312 if f:is_infinity(a) then return f:is_infinity(b) and invalid or result(f:inf(sigma)) end
313 if f:is_infinity(b) then return result(f:zero(sigma)) end
314 if f:is_zero(b) then return f:is_zero(a) and invalid or result(f:inf(sigma), flags.divbyzero) end
315 if f:is_zero(a) then return result(f:zero(sigma)) end
316 local x, y = f:exact(a), f:exact(b)
317 return f:round(Exact.new(x.n * y.d, x.exponent - y.exponent, x.d * y.n, sigma), mode)
318 end
319 local c = inputs[3]
320 if f:is_infinity(a) or f:is_infinity(b) then
321 if f:is_zero(a) or f:is_zero(b) or f:is_infinity(c) and f:fields(c).negative ~= sigma then return invalid end
322 return result(f:inf(sigma))
323 end
324 if f:is_infinity(c) then return result(c) end
325 return f:round(Exact.sum({Exact.product(f:exact(a), f:exact(b)), f:exact(c)}, mode), mode)
326end

328local function next_up(f, pattern)
329 if pattern == f.infinity then return pattern end
330 if f:is_zero(pattern) then return 1 end
331 return pattern + (f:fields(pattern).negative and -1 or 1)
332end

334local function difference(f, pattern, exact)
335 if not f:is_finite(pattern) then return f:value(pattern) end
336 return Exact.sum({f:exact(pattern), exact:negated()}, "RNE"):fraction()
337end

358 local patterns = {}

363function Laboratory:summary(f, mode, patterns, value)
364 local terms = {}
365 for _, pattern in ipairs(patterns) do terms[#terms + 1] = f:exact(pattern) end
366 local exact = Exact.sum(terms, mode)
367 self:emit("RESULT %s", f:result_text(value))
368 self:emit("EXACT %s", exact:decimal())
369 self:emit("ROUNDED %s", f:result_text(f:round(exact, mode)))
370 self:emit("DIFF %s", difference(f, value.pattern, exact))
371end

373function Laboratory:summation(command, f, mode, patterns)
374 for _, pattern in ipairs(patterns) do if not f:is_finite(pattern) then fail("NOT FINITE") end end
375 self:emit("%s %s %s count=%d", command, f.name, mode, #patterns)
376 local sum, compensation, raised = patterns[1], 0, 0
377 for i = 2, #patterns do
378 if command == "SUM" then
379 local added = calculate(f, "ADD", mode, {sum, patterns[i]})
380 sum, raised = added.pattern, raised | added.flags
381 self:emit("PARTIAL %d %s", i, f:result_text(added))
382 else
383 local y = calculate(f, "SUB", mode, {patterns[i], compensation})
384 local t = calculate(f, "ADD", mode, {sum, y.pattern})
385 local d = calculate(f, "SUB", mode, {t.pattern, sum})
386 local c = calculate(f, "SUB", mode, {d.pattern, y.pattern})
387 sum, compensation = t.pattern, c.pattern
388 local step = y.flags | t.flags | d.flags | c.flags
389 raised = raised | step
390 self:emit("KSTEP %d sum=%s comp=%s %s", i, f:value(sum), f:value(compensation), flags_text(step))
391 end
392 end
393 self:summary(f, mode, patterns, result(sum, raised))
394end

396function Laboratory:dot(f, mode, a, b)
397 if #a ~= #b then fail("LENGTH MISMATCH") end
398 for _, pattern in ipairs(a) do if not f:is_finite(pattern) then fail("NOT FINITE") end end
399 for _, pattern in ipairs(b) do if not f:is_finite(pattern) then fail("NOT FINITE") end end
400 local naive = calculate(f, "MUL", mode, {a[1], b[1]})
401 local chained = naive
402 for i = 2, #a do
403 local product = calculate(f, "MUL", mode, {a[i], b[i]})
404 local added = calculate(f, "ADD", mode, {naive.pattern, product.pattern})
405 naive = result(added.pattern, naive.flags | product.flags | added.flags)
406 local fused = calculate(f, "FMA", mode, {a[i], b[i], chained.pattern})
407 chained = result(fused.pattern, chained.flags | fused.flags)
408 end
409 local terms = {}
410 for i = 1, #a do terms[i] = Exact.product(f:exact(a[i]), f:exact(b[i])) end
411 local exact = Exact.sum(terms, mode)
412 self:emit("DOT %s %s n=%d", f.name, mode, #a)
413 self:emit("NAIVE %s", f:result_text(naive))
414 self:emit("CHAINED %s", f:result_text(chained))
415 self:emit("EXACT %s", exact:decimal())
416 self:emit("FUSED %s", f:result_text(f:round(exact, mode)))
417 self:emit("DIFF naive=%s chained=%s", difference(f, naive.pattern, exact), difference(f, chained.pattern, exact))
418end

420function Laboratory:table(f)
421 if f.width > 8 then fail("TABLE TOO LARGE") end
422 local count = f.infinity
423 self:emit("TABLE %s count=%d", f.name, count)
424 for pattern = 0, count - 1 do
425 local gap = "-"
426 if pattern < count - 1 then gap = Exact.sum({f:exact(pattern + 1), f:exact(pattern):negated()}, "RNE"):decimal() end
427 self:emit("ENTRY %d %s gap=%s", pattern, f:pattern_text(pattern), gap)
428 end
429 self:emit("MAX %s", f:pattern_text(f.infinity - 1))
430 self:emit("MINNORMAL %s", f:pattern_text(f.hidden))
431 self:emit("MINSUBNORMAL %s", f:pattern_text(1))
432 self:emit("EPSILON %s", Exact.new(1, -f.m):decimal())
433 self:emit("PATTERNS finite=%d infinite=2 nan=%d total=%d", 2 * count, 2 * (f.hidden - 1), 1 << f.width)
434end

436function Laboratory:execute(line)
437 self.output = {}
438 local words = {}
439 for word in line:gmatch("%S+") do words[#words + 1] = word end
440 local command = words[1]
441 if command == "FORMAT" then
442 local name, e, m = words[2], tonumber(words[3], 10), tonumber(words[4], 10)
443 if self.formats[name] then fail("FORMAT EXISTS %s", name) end
444 if e < 2 or e > 5 or m < 1 or m > 10 then fail("BAD FORMAT") end
445 local f = MiniFormat.new(name, e, m)
446 self.formats[name] = f
447 self:emit("FORMAT %s e=%d m=%d bits=%d bias=%d emin=%d emax=%d", name, e, m, f.width, f.bias, f.emin, f.emax)
448 return self.output
449 end
450 local f = self:format(words[2])
451 if command == "ENCODE" then
452 local input = real_input(words[3])
453 local mode = mode_at(words, 4)
454 self:emit("ENCODE %s %s %s %s", f.name, mode, input_text(input), f:result_text(encoded(f, input, mode)))
455 elseif command == "CONVERT" then
456 local g = self:format(words[3])
457 local a = operand(f, words[4])
458 local mode = mode_at(words, 5)
459 local value
460 if f:is_nan(a) then value = result(g:nan(), f:is_signaling(a) and flags.invalid or 0)
461 elseif f:is_infinity(a) then value = result(g:inf(f:fields(a).negative))
462 else value = g:round(f:exact(a), mode) end
463 self:emit("CONVERT %s %s %s %s", f.name, g.name, mode, g:result_text(value))
464 elseif command == "SUM" or command == "KAHAN" or command == "DOT" then
465 local mode = mode_at(words, 3)
466 if command == "DOT" then
467 local separator = 4
468 while words[separator] ~= "|" do separator = separator + 1 end
469 local a = operands(f, words, 4, separator - 1)
470 local b = operands(f, words, separator + 1, #words)
471 self:dot(f, mode, a, b)
472 else self:summation(command, f, mode, operands(f, words, 4, #words)) end
473 elseif command == "TABLE" then
474 self:table(f)
475 elseif operations[command] then
476 local count = operations[command]
477 local patterns = operands(f, words, 3, 2 + count)
478 local mode = mode_at(words, 3 + count)
479 self:emit("%s %s %s %s", command, f.name, mode, f:result_text(calculate(f, command, mode, patterns)))
480 else
481 local a = operand(f, words[3])
482 local fields = f:fields(a)
483 if command == "DECODE" then
484 local classification
485 if f:is_nan(a) then classification = f:is_signaling(a) and "snan" or "qnan"
486 elseif f:is_infinity(a) then classification = "infinity"
487 elseif f:is_zero(a) then classification = "zero"
488 else classification = fields.exponent == 0 and "subnormal" or "normal" end
489 self:emit("DECODE %s %s %s %s sign=%d", f.name, f:bits(a), f:hex(a), classification, fields.negative and 1 or 0)
490 local heading = string.format("FIELDS exponent=%d fraction=%d", fields.exponent, fields.fraction)
491 if f:is_finite(a) then
492 self:emit("%s unbiased=%d significand=%d.%s", heading, fields.q + f.m, fields.exponent == 0 and 0 or 1, binary(fields.fraction, f.m))
493 self:emit("VALUE %s %s", f:exact(a):fraction(), f:value(a))
494 elseif f:is_nan(a) then self:emit("%s payload=%d", heading, fields.fraction % (f.hidden // 2))
495 else self:emit("%s", heading) end
496 elseif command == "COMPARE" then
497 local b = operand(f, words[4])
498 local function key(p) return f:fields(p).negative and -(p - f.sign_bit) - 1 or p end
499 local order = relation(key(a), key(b))
500 local numeric = order
501 if f:is_nan(a) or f:is_nan(b) then numeric = "UN"
502 elseif f:is_zero(a) and f:is_zero(b) then numeric = "EQ" end
503 local raised = (f:is_signaling(a) or f:is_signaling(b)) and flags.invalid or 0
504 self:emit("COMPARE %s %s %s %s total=%s %s", f.name, f:hex(a), f:hex(b), numeric, order, flags_text(raised))
505 elseif command == "TOINT" then
506 local range = integer_types[words[4]]
507 if not range then fail("BAD TYPE %s", words[4]) end
508 local mode = mode_at(words, 5)
509 local prefix = string.format("TOINT %s %s %s", f.name, words[4], mode)
510 if not f:is_finite(a) then self:emit("%s invalid flags=invalid", prefix)
511 else
512 local whole, inexact = round_units(f:exact(a), 0, mode)
513 local integer = fields.negative and -whole or whole
514 if integer < range[1] or integer > range[2] then self:emit("%s invalid flags=invalid", prefix)
515 else self:emit("%s %d %s", prefix, integer, flags_text(inexact and flags.inexact or 0)) end
516 end
517 else
518 local value
519 if f:is_nan(a) then value = result(f:nan(), f:is_signaling(a) and flags.invalid or 0)
520 elseif command == "NEXTUP" then value = result(next_up(f, a))
521 elseif command == "NEXTDOWN" then value = result(f:flip(next_up(f, f:flip(a))))
522 else value = result(f:is_infinity(a) and f.infinity or f:round(Exact.new(1, fields.q), "RNE").pattern) end
523 self:emit("%s %s %s", command, f.name, f:result_text(value))
524 end
525 end
526 return self.output
527end
