18private data class Term(val negative: Boolean, val magnitude: Long, val exponent: Int) {
19 fun product(other: Term) = Term(negative != other.negative, magnitude * other.magnitude, exponent + other.exponent)
20 fun negated() = copy(negative = !negative)
21 fun exact() = Exact.of(negative, magnitude, exponent, 1)
22 fun decimal(): String {
23 if (magnitude == 0L) return "0"
24 val (numerator, denominator) = exact().scaled(0)
25 val out = StringBuilder(if (negative) "-" else "")
26 out.append(numerator / denominator)
27 var remainder = numerator % denominator
28 if (remainder != 0L) out.append('.')
29 while (remainder != 0L) {
30 remainder *= 10
31 out.append(('0'.code + remainder / denominator).toInt().toChar())
32 remainder %= denominator
33 }
34 return out.toString()
35 }
36}

38private fun sumTerms(terms: List<Term>, mode: Mode): Term {
39 val exponent = terms.filter { it.magnitude != 0L }.minOfOrNull { it.exponent } ?: 0
40 var sum = 0L
41 for (term in terms) {
42 if (term.magnitude == 0L) continue
43 val magnitude = term.magnitude shl (term.exponent - exponent)
44 sum += if (term.negative) -magnitude else magnitude
45 }
46 if (sum == 0L) {
47 val sameZero = terms.all { it.magnitude == 0L && it.negative == terms.first().negative }
48 return Term(if (sameZero) terms.first().negative else mode == Mode.RDN, 0, 0)
49 }
50 return Term(sum < 0, if (sum < 0) -sum else sum, exponent)
51}

53private class Exact private constructor(val negative: Boolean, val numerator: Long, val exponent: Int, val denominator: Long) {
54 companion object {
55 fun of(negative: Boolean, numerator: Long, exponent: Int, denominator: Long): Exact {
56 if (numerator == 0L) return Exact(negative, 0, 0, 1)
57 var a = numerator
58 var b = denominator
59 while (b != 0L) {
60 val remainder = a % b
61 a = b
62 b = remainder
63 }
64 var n = numerator / a
65 var d = denominator / a
66 var q = exponent
67 var shift = n.countTrailingZeroBits()
68 n = n shr shift
69 q += shift
70 shift = d.countTrailingZeroBits()
71 d = d shr shift
72 q -= shift
73 return Exact(negative, n, q, d)
74 }
75 }
76
77 fun scaled(quantum: Int): Pair<Long, Long> {
78 val shift = exponent - quantum
79 return if (shift >= 0) (numerator shl shift) to denominator else numerator to (denominator shl -shift)
80 }
81 fun floorLog(): Int {
82 val power = denominator.countLeadingZeroBits() - numerator.countLeadingZeroBits()
83 val below = if (power >= 0) numerator < (denominator shl power) else (numerator shl -power) < denominator
84 return exponent + power - if (below) 1 else 0
85 }
86 fun fraction(): String {
87 if (numerator == 0L) return "0"
88 val (n, d) = scaled(0)
89 return (if (negative) "-" else "") + n + if (d == 1L) "" else "/$d"
90 }
91}

93private fun integerSqrt(value: Long): Long {
94 if (value == 0L) return 0
95 var root = value
96 var next = (value + 1) / 2
97 while (next < root) {
98 root = next
99 next = (root + value / root) / 2
100 }
101 return root
102}

104private data class Quantized(val whole: Long, val inexact: Boolean, val half: Int) {
105 companion object {
106 fun of(value: Exact, quantum: Int, squareRoot: Boolean): Quantized {
107 val (numerator, denominator) = value.scaled(if (squareRoot) 2 * quantum else quantum)
108 if (!squareRoot) {
109 val remainder = numerator % denominator
110 return Quantized(numerator / denominator, remainder != 0L, remainder.compareTo(denominator - remainder))
111 }
112 val whole = integerSqrt(numerator / denominator)
113 val midpoint = 2 * whole + 1
114 return Quantized(whole, whole * whole * denominator != numerator,
115 (4 * numerator).compareTo(midpoint * midpoint * denominator))
116 }
117 }
118 fun rounded(mode: Mode, negative: Boolean): Long {
119 val increment = when (mode) {
120 Mode.RNE -> half > 0 || half == 0 && (whole and 1L) != 0L
121 Mode.RNA -> half >= 0
122 Mode.RTZ -> false
123 Mode.RUP -> inexact && !negative
124 Mode.RDN -> inexact && negative
125 }
126 return whole + if (increment) 1 else 0
127 }
128}

134private class Format(val name: String, val e: Int, val m: Int) {
135 val w = 1 + e + m
136 val bias = (1 shl (e - 1)) - 1
137 val emin = 1 - bias
138 val emax = bias
139 val sign = 1 shl (e + m)
140 val base = 1 shl m
141 val infinity = ((1 shl e) - 1) shl m
142 val nan = infinity or (base / 2)
143
144 fun negative(p: Int) = (p and sign) != 0
145 fun zero(p: Int) = (p and (sign - 1)) == 0
146 fun signed(p: Int, negative: Boolean) = p or if (negative) sign else 0
147 fun kind(p: Int): Kind {
148 val magnitude = p and (sign - 1)
149 val fraction = p and (base - 1)
150 return when {
151 magnitude >= infinity -> when {
152 fraction == 0 -> Kind.INFINITY
153 fraction >= base / 2 -> Kind.QNAN
154 else -> Kind.SNAN
155 }
156 magnitude == 0 -> Kind.ZERO
157 magnitude < base -> Kind.SUBNORMAL
158 else -> Kind.NORMAL
159 }
160 }
161 fun finite(p: Int) = kind(p) in setOf(Kind.ZERO, Kind.SUBNORMAL, Kind.NORMAL)
162 fun isNaN(p: Int) = kind(p) == Kind.QNAN || kind(p) == Kind.SNAN
163 fun term(p: Int): Term {
164 val field = (p and (sign - 1)) shr m
165 val fraction = p and (base - 1)
166 return Term(negative(p), (if (field == 0) fraction else base + fraction).toLong(),
167 if (field == 0) emin - m else field - bias - m)
168 }
169 fun overflow(mode: Mode, negative: Boolean): Rounded {
170 val toInfinity = mode == Mode.RNE || mode == Mode.RNA || mode == Mode.RUP && !negative || mode == Mode.RDN && negative
171 return Rounded(signed(if (toInfinity) infinity else infinity - 1, negative), OVERFLOW or INEXACT)
172 }
173 fun round(value: Exact, mode: Mode, squareRoot: Boolean = false): Rounded {
174 if (value.numerator == 0L) return Rounded(signed(0, value.negative))
175 var t = value.floorLog()
176 if (squareRoot) t = t.floorDiv(2)
177 if (t > emax) return overflow(mode, value.negative)
178 var quantum = maxOf(t, emin) - m
179 val divided = Quantized.of(value, quantum, squareRoot)
180 var significand = divided.rounded(mode, value.negative)
181 if (significand == 2L * base) {
182 significand = base.toLong()
183 quantum++
184 }
185 if (quantum > emax - m) return overflow(mode, value.negative)
186 var flags = if (divided.inexact) INEXACT else 0
187 if (divided.inexact && t < emin) {
188 val tiny = t < emin - 1 || Quantized.of(value, t - m, squareRoot).rounded(mode, value.negative) < 2L * base
189 if (tiny) flags = flags or UNDERFLOW
190 }
191 val magnitude = if (significand < base) significand.toInt()
192 else ((quantum + m + bias) shl m) or (significand - base).toInt()
193 return Rounded(signed(magnitude, value.negative), flags)
194 }
195 fun roundTerm(term: Term, mode: Mode) = round(term.exact(), mode)
196 fun roundSum(terms: List<Term>, mode: Mode) = roundTerm(sumTerms(terms, mode), mode)

198 fun arithmetic(operation: Operation, a: Int, bInput: Int, c: Int, mode: Mode): Rounded {
199 val b = if (operation == Operation.SUB) bInput xor sign else bInput
200 val op = if (operation == Operation.SUB) Operation.ADD else operation
201 val inputs = when (op) {
202 Operation.SQRT -> listOf(a)
203 Operation.FMA -> listOf(a, b, c)
204 else -> listOf(a, b)
205 }
206 if (inputs.any(::isNaN)) return Rounded(nan, if (inputs.any { kind(it) == Kind.SNAN }) INVALID else 0)
207 val ai = kind(a) == Kind.INFINITY
208 val bi = kind(b) == Kind.INFINITY
209 val sigma = negative(a) != negative(b)
210 val invalid = Rounded(nan, INVALID)
211 return when (op) {
212 Operation.ADD -> when {
213 ai && bi && sigma -> invalid
214 ai -> Rounded(a)
215 bi -> Rounded(b)
216 else -> roundSum(listOf(term(a), term(b)), mode)
217 }
218 Operation.MUL -> if (ai || bi) {
219 if (zero(a) || zero(b)) invalid else Rounded(signed(infinity, sigma))
220 } else roundTerm(term(a).product(term(b)), mode)
221 Operation.DIV -> when {
222 ai -> if (bi) invalid else Rounded(signed(infinity, sigma))
223 bi -> Rounded(signed(0, sigma))
224 zero(b) -> if (zero(a)) invalid else Rounded(signed(infinity, sigma), DIVBYZERO)
225 zero(a) -> Rounded(signed(0, sigma))
226 else -> {
227 val x = term(a)
228 val y = term(b)
229 round(Exact.of(sigma, x.magnitude, x.exponent - y.exponent, y.magnitude), mode)
230 }
231 }
232 Operation.SQRT -> when {
233 zero(a) -> Rounded(a)
234 negative(a) -> invalid
235 ai -> Rounded(a)
236 else -> round(term(a).exact(), mode, true)
237 }
238 Operation.FMA -> when {
239 ai || bi -> if (zero(a) || zero(b) || kind(c) == Kind.INFINITY && negative(c) != sigma) invalid
240 else Rounded(signed(infinity, sigma))
241 kind(c) == Kind.INFINITY -> Rounded(c)
242 else -> roundSum(listOf(term(a).product(term(b)), term(c)), mode)
243 }
244 Operation.SUB -> error("unreachable operation")
245 }
246 }

248 fun rawOperand(word: String, encoding: Boolean): Operand {
249 if (word.startsWith("0b") || word.startsWith("0x")) {
250 val binary = word.startsWith("0b")
251 val width = if (binary) w else (w + 3) / 4
252 if (encoding || word.length - 2 != width) fail("ERROR BAD OPERAND $word")
253 val p = word.substring(2).toInt(if (binary) 2 else 16)
254 if (p >= 2 * sign) fail("ERROR BAD OPERAND $word")
255 return Operand(pattern = p)
256 }
257 when (word) {
258 "nan" -> return Operand(pattern = nan)
259 "inf" -> return Operand(pattern = infinity)
260 "-inf" -> return Operand(pattern = infinity or sign)
261 }
262 val negative = word.startsWith('-')
263 val digits = word.removePrefix("-")
264 val slash = digits.indexOf('/')
265 val dot = digits.indexOf('.')
266 var denominator = 1L
267 val numerator = when {
268 slash >= 0 -> {
269 denominator = digits.substring(slash + 1).toLong()
270 if (denominator == 0L) fail("ERROR BAD OPERAND $word")
271 digits.substring(0, slash).toLong()
272 }
273 dot >= 0 -> {
274 repeat(digits.length - dot - 1) { denominator *= 10 }
275 (digits.substring(0, dot) + digits.substring(dot + 1)).toLong()
276 }
277 else -> digits.toLong()
278 }
279 return Operand(number = Exact.of(negative, numerator, 0, denominator))
280 }
281 fun operand(word: String): Int {
282 val value = rawOperand(word, false)
283 return value.number?.let { round(it, Mode.RNE).pattern } ?: value.pattern
284 }
285 fun bits(p: Int) = "${if (negative(p)) 1 else 0}|${binary((p and (sign - 1)) shr m, e)}|${binary(p and (base - 1), m)}"
286 fun hex(p: Int) = "0x" + p.toString(16).uppercase(Locale.ROOT).padStart((w + 3) / 4, '0')
287 fun value(p: Int): String = when (kind(p)) {
288 Kind.QNAN, Kind.SNAN -> "nan"
289 Kind.INFINITY -> if (negative(p)) "-inf" else "inf"
290 Kind.ZERO -> if (negative(p)) "-0" else "0"
291 else -> term(p).decimal()
292 }
293 fun pattern(p: Int) = "${bits(p)} ${hex(p)} ${value(p)}"
294 fun result(result: Rounded) = "${pattern(result.pattern)} ${flagText(result.flags)}"
295 fun difference(p: Int, exactSum: Term) = if (finite(p)) sumTerms(listOf(term(p), exactSum.negated()), Mode.RNE).exact().fraction() else value(p)
296 fun orderKey(p: Int) = if (negative(p)) -(p - sign) - 1 else p
297 fun nextUp(p: Int) = when {
298 p == infinity -> p
299 zero(p) -> 1
300 negative(p) -> p - 1
301 else -> p + 1
302 }
303}

305private fun binary(value: Int, width: Int) = value.toString(2).padStart(width, '0')
306private fun flagText(flags: Int): String {
307 val names = listOf("invalid", "divbyzero", "overflow", "underflow", "inexact")
308 val raised = names.filterIndexed { i, _ -> (flags and (1 shl i)) != 0 }
309 return "flags=" + if (raised.isEmpty()) "none" else raised.joinToString(",")
310}
311private fun parseMode(word: String) = Mode.entries.find { it.name == word } ?: fail("ERROR BAD MODE $word")
312private fun modeAt(words: List<String>, index: Int) = words.getOrNull(index)?.let(::parseMode) ?: Mode.RNE
313private fun integerRange(word: String) = when (word) {
314 "I8" -> Bounds(-128, 127)
315 "U8" -> Bounds(0, 255)
316 "I16" -> Bounds(-32768, 32767)
317 "U16" -> Bounds(0, 65535)
318 "I32" -> Bounds(-2147483648, 2147483647)
319 "U32" -> Bounds(0, 4294967295)
320 else -> fail("ERROR BAD TYPE $word")
321}
322private fun relation(a: Int, b: Int) = if (a < b) "LT" else if (a == b) "EQ" else "GT"

324private fun summation(f: Format, words: List<String>): String {
325 val command = words[0]
326 val mode = parseMode(words[2])
327 val values = mutableListOf<Int>()
328 var leftCount = 0
329 for (word in words.drop(3)) {
330 if (word == "|") leftCount = values.size else values.add(f.operand(word))
331 }
332 if (command == "DOT" && leftCount != values.size - leftCount) fail("ERROR LENGTH MISMATCH")
333 if (values.any { !f.finite(it) }) fail("ERROR NOT FINITE")
334 val out = StringBuilder()
335 if (command == "DOT") {
336 val terms = List(leftCount) { i -> f.term(values[i]).product(f.term(values[leftCount + i])) }
337 val exactSum = sumTerms(terms, mode)
338 val first = f.arithmetic(Operation.MUL, values.first(), values[leftCount], 0, mode)
339 var naive = first
340 var chained = first
341 for (i in 1 until leftCount) {
342 val multiplied = f.arithmetic(Operation.MUL, values[i], values[leftCount + i], 0, mode)
343 val added = f.arithmetic(Operation.ADD, naive.pattern, multiplied.pattern, 0, mode)
344 naive = Rounded(added.pattern, naive.flags or multiplied.flags or added.flags)
345 }
346 for (i in 1 until leftCount) {
347 val fused = f.arithmetic(Operation.FMA, values[i], values[leftCount + i], chained.pattern, mode)
348 chained = Rounded(fused.pattern, chained.flags or fused.flags)
349 }
350 out.appendLine("DOT ${f.name} $mode n=$leftCount")
351 out.appendLine("NAIVE ${f.result(naive)}")
352 out.appendLine("CHAINED ${f.result(chained)}")
353 out.appendLine("EXACT ${exactSum.decimal()}")
354 out.appendLine("FUSED ${f.result(f.roundTerm(exactSum, mode))}")
355 out.appendLine("DIFF naive=${f.difference(naive.pattern, exactSum)} chained=${f.difference(chained.pattern, exactSum)}")
356 return out.toString()
357 }
358 val exactSum = sumTerms(values.map(f::term), mode)
359 var result = Rounded(values.first())
360 var compensation = 0
361 out.appendLine("$command ${f.name} $mode count=${values.size}")
362 for (i in 1 until values.size) {
363 if (command == "SUM") {
364 val next = f.arithmetic(Operation.ADD, result.pattern, values[i], 0, mode)
365 result = Rounded(next.pattern, result.flags or next.flags)
366 out.appendLine("PARTIAL ${i + 1} ${f.result(next)}")
367 } else {
368 val y = f.arithmetic(Operation.SUB, values[i], compensation, 0, mode)
369 val t = f.arithmetic(Operation.ADD, result.pattern, y.pattern, 0, mode)
370 val d = f.arithmetic(Operation.SUB, t.pattern, result.pattern, 0, mode)
371 val c = f.arithmetic(Operation.SUB, d.pattern, y.pattern, 0, mode)
372 val flags = y.flags or t.flags or d.flags or c.flags
373 result = Rounded(t.pattern, result.flags or flags)
374 compensation = c.pattern
375 out.appendLine("KSTEP ${i + 1} sum=${f.value(result.pattern)} comp=${f.value(compensation)} ${flagText(flags)}")
376 }
377 }
378 out.appendLine("RESULT ${f.result(result)}")
379 out.appendLine("EXACT ${exactSum.decimal()}")
380 out.appendLine("ROUNDED ${f.result(f.roundTerm(exactSum, mode))}")
381 out.appendLine("DIFF ${f.difference(result.pattern, exactSum)}")
382 return out.toString()
383}

385private fun table(f: Format): String {
386 if (f.w > 8) fail("ERROR TABLE TOO LARGE")
387 return buildString {
388 appendLine("TABLE ${f.name} count=${f.infinity}")
389 for (p in 0 until f.infinity) {
390 val gap = if (p + 1 == f.infinity) "-" else sumTerms(listOf(f.term(p + 1), f.term(p).negated()), Mode.RNE).decimal()
391 appendLine("ENTRY $p ${f.pattern(p)} gap=$gap")
392 }
393 appendLine("MAX ${f.pattern(f.infinity - 1)}")
394 appendLine("MINNORMAL ${f.pattern(f.base)}")
395 appendLine("MINSUBNORMAL ${f.pattern(1)}")
396 appendLine("EPSILON ${Term(false, 1, -f.m).decimal()}")
397 appendLine("PATTERNS finite=${2 * f.infinity} infinite=2 nan=${2 * (f.base - 1)} total=${2 * f.sign}")
398 }
399}

401private class Laboratory {
402 private val formats = mutableMapOf(
403 "HALF" to Format("HALF", 5, 10), "E5M2" to Format("E5M2", 5, 2), "E4M3" to Format("E4M3", 4, 3)
404 )
405 private fun format(name: String) = formats[name] ?: fail("ERROR UNKNOWN FORMAT $name")
406
407 fun execute(words: List<String>): String {
408 val command = words[0]
409 if (command == "FORMAT") {
410 if (words[1] in formats) fail("ERROR FORMAT EXISTS ${words[1]}")
411 val e = words[2].toInt()
412 val m = words[3].toInt()
413 if (e !in 2..5 || m !in 1..10) fail("ERROR BAD FORMAT")
414 val f = Format(words[1], e, m)
415 formats[f.name] = f
416 return "FORMAT ${f.name} e=$e m=$m bits=${f.w} bias=${f.bias} emin=${f.emin} emax=${f.emax}\n"
417 }
418 val f = format(words[1])
419 if (command == "TABLE") return table(f)
420 if (command in listOf("SUM", "KAHAN", "DOT")) return summation(f, words)
421 if (command == "CONVERT") {
422 val g = format(words[2])
423 val a = f.operand(words[3])
424 val mode = modeAt(words, 4)
425 val result = when (f.kind(a)) {
426 Kind.QNAN -> Rounded(g.nan)
427 Kind.SNAN -> Rounded(g.nan, INVALID)
428 Kind.INFINITY -> Rounded(g.signed(g.infinity, f.negative(a)))
429 Kind.ZERO -> Rounded(g.signed(0, f.negative(a)))
430 else -> g.roundTerm(f.term(a), mode)
431 }
432 return "CONVERT ${f.name} ${g.name} $mode ${g.result(result)}\n"
433 }
434 if (command == "ENCODE") {
435 val value = f.rawOperand(words[2], true)
436 val mode = modeAt(words, 3)
437 val result = value.number?.let { f.round(it, mode) } ?: Rounded(value.pattern)
438 val source = value.number?.fraction() ?: f.value(value.pattern)
439 return "ENCODE ${f.name} $mode $source ${f.result(result)}\n"
440 }
441 val a = f.operand(words[2])
442 val out = StringBuilder()
443 when (command) {
444 "DECODE" -> {
445 val kind = f.kind(a)
446 val exponent = (a and (f.sign - 1)) shr f.m
447 val fraction = a and (f.base - 1)
448 out.appendLine("DECODE ${f.name} ${f.bits(a)} ${f.hex(a)} $kind sign=${if (f.negative(a)) 1 else 0}")
449 if (f.finite(a)) {
450 val term = f.term(a)
451 out.appendLine("FIELDS exponent=$exponent fraction=$fraction unbiased=${term.exponent + f.m} significand=${if (kind == Kind.NORMAL) 1 else 0}.${binary(fraction, f.m)}")
452 out.appendLine("VALUE ${term.exact().fraction()} ${f.value(a)}")
453 } else if (kind == Kind.INFINITY) out.appendLine("FIELDS exponent=$exponent fraction=$fraction")
454 else out.appendLine("FIELDS exponent=$exponent fraction=$fraction payload=${fraction % (f.base / 2)}")
455 }
456 "COMPARE" -> {
457 val b = f.operand(words[3])
458 val ak = f.orderKey(a)
459 val bk = f.orderKey(b)
460 val numeric = when {
461 f.isNaN(a) || f.isNaN(b) -> "UN"
462 f.zero(a) && f.zero(b) -> "EQ"
463 else -> relation(ak, bk)
464 }
465 val flags = if (f.kind(a) == Kind.SNAN || f.kind(b) == Kind.SNAN) INVALID else 0
466 out.appendLine("COMPARE ${f.name} ${f.hex(a)} ${f.hex(b)} $numeric total=${relation(ak, bk)} ${flagText(flags)}")
467 }
468 "TOINT" -> {
469 val bounds = integerRange(words[3])
470 val mode = modeAt(words, 4)
471 val prefix = "TOINT ${f.name} ${words[3]} $mode"
472 if (!f.finite(a)) out.appendLine("$prefix invalid flags=invalid")
473 else {
474 val value = f.term(a).exact()
475 val divided = Quantized.of(value, 0, false)
476 val magnitude = divided.rounded(mode, value.negative)
477 val result = if (value.negative) -magnitude else magnitude
478 if (result < bounds.minimum || result > bounds.maximum) out.appendLine("$prefix invalid flags=invalid")
479 else out.appendLine("$prefix $result ${flagText(if (divided.inexact) INEXACT else 0)}")
480 }
481 }
482 "NEXTUP", "NEXTDOWN", "ULP" -> {
483 val result = if (f.isNaN(a)) Rounded(f.nan, if (f.kind(a) == Kind.SNAN) INVALID else 0)
484 else Rounded(when (command) {
485 "NEXTUP" -> f.nextUp(a)
486 "NEXTDOWN" -> f.nextUp(a xor f.sign) xor f.sign
487 else -> if (f.kind(a) == Kind.INFINITY) f.infinity
488 else f.roundTerm(Term(false, 1, f.term(a).exponent), Mode.RNE).pattern
489 })
490 out.appendLine("$command ${f.name} ${f.result(result)}")
491 }
492 else -> {
493 val operation = Operation.valueOf(command)
494 val arity = when (operation) { Operation.SQRT -> 1; Operation.FMA -> 3; else -> 2 }
495 val b = if (arity >= 2) f.operand(words[3]) else 0
496 val c = if (arity == 3) f.operand(words[4]) else 0
497 val mode = modeAt(words, arity + 2)
498 out.appendLine("$command ${f.name} $mode ${f.result(f.arithmetic(operation, a, b, c, mode))}")
499 }
500 }
501 return out.toString()
502 }
503}

505fun main() {
506 val input = System.`in`.bufferedReader()
507 val output = System.out.bufferedWriter()
508 val count = input.readLine().toInt()
509 val laboratory = Laboratory()
510 repeat(count) {
511 val response = try { laboratory.execute(input.readLine().split(' ')) }
512 catch (error: CommandError) { "${error.message}\n" }
513 output.write(response)
514 }
515 output.flush()
516}
