27 record Term(boolean negative, long magnitude, int exponent) {
28 Term product(Term other) {
29 return new Term(negative != other.negative, magnitude * other.magnitude, exponent + other.exponent);
30 }
31 Term negated() { return new Term(!negative, magnitude, exponent); }
32 Exact exact() { return new Exact(negative, magnitude, exponent, 1); }
33 String decimal() {
34 if (magnitude == 0) return "0";
35 Scaled value = exact().scaled(0);
36 StringBuilder out = new StringBuilder(negative ? "-" : "");
37 out.append(value.numerator / value.denominator);
38 long remainder = value.numerator % value.denominator;
39 if (remainder != 0) out.append('.');
40 while (remainder != 0) {
41 remainder *= 10;
42 out.append((char) ('0' + remainder / value.denominator));
43 remainder %= value.denominator;
44 }
45 return out.toString();
46 }
47 }

49 static Term sumTerms(Mode mode, Term... terms) {
50 int exponent = 0;
51 boolean any = false, sameZero = true;
52 for (Term term : terms) {
53 if (term.magnitude != 0) {
54 if (!any || term.exponent < exponent) exponent = term.exponent;
55 any = true;
56 sameZero = false;
57 } else if (term.negative != terms[0].negative) sameZero = false;
58 }
59 long sum = 0;
60 for (Term term : terms) {
61 if (term.magnitude == 0) continue;
62 long aligned = term.magnitude << (term.exponent - exponent);
63 sum += term.negative ? -aligned : aligned;
64 }
65 if (sum == 0) return new Term(sameZero ? terms[0].negative : mode == Mode.RDN, 0, 0);
66 return new Term(sum < 0, Math.abs(sum), exponent);
67 }

69 record Scaled(long numerator, long denominator) {}
70 record Exact(boolean negative, long numerator, int exponent, long denominator) {
71 Exact {
72 if (numerator == 0) {
73 exponent = 0;
74 denominator = 1;
75 } else {
76 long a = numerator, b = denominator;
77 while (b != 0) { long remainder = a % b; a = b; b = remainder; }
78 numerator /= a;
79 denominator /= a;
80 int shift = Long.numberOfTrailingZeros(numerator);
81 numerator >>= shift;
82 exponent += shift;
83 shift = Long.numberOfTrailingZeros(denominator);
84 denominator >>= shift;
85 exponent -= shift;
86 }
87 }
88 Scaled scaled(int quantum) {
89 int shift = exponent - quantum;
90 return shift >= 0 ? new Scaled(numerator << shift, denominator) : new Scaled(numerator, denominator << -shift);
91 }
92 int floorLog() {
93 int power = Long.numberOfLeadingZeros(denominator) - Long.numberOfLeadingZeros(numerator);
94 boolean below = power >= 0 ? numerator < (denominator << power) : (numerator << -power) < denominator;
95 return exponent + power - (below ? 1 : 0);
96 }
97 String fraction() {
98 if (numerator == 0) return "0";
99 Scaled value = scaled(0);
100 String sign = negative ? "-" : "";
101 return sign + value.numerator + (value.denominator == 1 ? "" : "/" + value.denominator);
102 }
103 }

105 static long integerSqrt(long value) {
106 if (value == 0) return 0;
107 long root = value, next = (value + 1) / 2;
108 while (next < root) {
109 root = next;
110 next = (root + value / root) / 2;
111 }
112 return root;
113 }

115 record Quantized(long whole, boolean inexact, int half) {
116 static Quantized of(Exact value, int quantum, boolean squareRoot) {
117 Scaled scaled = value.scaled(squareRoot ? 2 * quantum : quantum);
118 long numerator = scaled.numerator, denominator = scaled.denominator;
119 if (!squareRoot) {
120 long remainder = numerator % denominator;
121 return new Quantized(numerator / denominator, remainder != 0, Long.compare(remainder, denominator - remainder));
122 }
123 long whole = integerSqrt(numerator / denominator), midpoint = 2 * whole + 1;
124 return new Quantized(whole, whole * whole * denominator != numerator,
125 Long.compare(4 * numerator, midpoint * midpoint * denominator));
126 }
127 long rounded(Mode mode, boolean negative) {
128 boolean increment = switch (mode) {
129 case RNE -> half > 0 || half == 0 && (whole & 1) != 0;
130 case RNA -> half >= 0;
131 case RTZ -> false;
132 case RUP -> inexact && !negative;
133 case RDN -> inexact && negative;
134 };
135 return whole + (increment ? 1 : 0);
136 }
137 }

145 static final class Format {
146 final String name;
147 final int e, m, w, bias, emin, emax, sign, base, infinity, nan;
148 Format(String name, int e, int m) {
149 this.name = name;
150 this.e = e;
151 this.m = m;
152 w = 1 + e + m;
153 bias = (1 << (e - 1)) - 1;
154 emin = 1 - bias;
155 emax = bias;
156 sign = 1 << (e + m);
157 base = 1 << m;
158 infinity = ((1 << e) - 1) << m;
159 nan = infinity | base / 2;
160 }
161 boolean negative(int p) { return (p & sign) != 0; }
162 boolean zero(int p) { return (p & (sign - 1)) == 0; }
163 int signed(int p, boolean negative) { return p | (negative ? sign : 0); }
164 Kind kind(int p) {
165 int magnitude = p & (sign - 1), fraction = p & (base - 1);
166 if (magnitude >= infinity) {
167 if (fraction == 0) return Kind.INFINITY;
168 return fraction >= base / 2 ? Kind.QNAN : Kind.SNAN;
169 }
170 if (magnitude == 0) return Kind.ZERO;
171 return magnitude < base ? Kind.SUBNORMAL : Kind.NORMAL;
172 }
173 boolean finite(int p) { return kind(p).ordinal() <= Kind.NORMAL.ordinal(); }
174 boolean isNaN(int p) { return kind(p) == Kind.QNAN || kind(p) == Kind.SNAN; }
175 Term term(int p) {
176 int field = (p & (sign - 1)) >> m, fraction = p & (base - 1);
177 return new Term(negative(p), field == 0 ? fraction : base + fraction,
178 field == 0 ? emin - m : field - bias - m);
179 }
180 Rounded overflow(Mode mode, boolean negative) {
181 boolean toInfinity = mode == Mode.RNE || mode == Mode.RNA || mode == Mode.RUP && !negative || mode == Mode.RDN && negative;
182 return new Rounded(signed(toInfinity ? infinity : infinity - 1, negative), OVERFLOW | INEXACT);
183 }
184 Rounded round(Exact value, Mode mode, boolean squareRoot) {
185 if (value.numerator == 0) return new Rounded(signed(0, value.negative));
186 int t = value.floorLog();
187 if (squareRoot) t = Math.floorDiv(t, 2);
188 if (t > emax) return overflow(mode, value.negative);
189 int quantum = Math.max(t, emin) - m;
190 Quantized divided = Quantized.of(value, quantum, squareRoot);
191 long significand = divided.rounded(mode, value.negative);
192 if (significand == 2L * base) { significand = base; quantum++; }
193 if (quantum > emax - m) return overflow(mode, value.negative);
194 int flags = divided.inexact ? INEXACT : 0;
195 if (divided.inexact && t < emin) {
196 boolean tiny = t < emin - 1 || Quantized.of(value, t - m, squareRoot).rounded(mode, value.negative) < 2L * base;
197 if (tiny) flags |= UNDERFLOW;
198 }
199 int magnitude = significand < base ? (int) significand
200 : (quantum + m + bias) << m | (int) (significand - base);
201 return new Rounded(signed(magnitude, value.negative), flags);
202 }
203 Rounded roundTerm(Term term, Mode mode) { return round(term.exact(), mode, false); }
204 Rounded roundSum(Mode mode, Term... terms) { return roundTerm(sumTerms(mode, terms), mode); }

206 Rounded arithmetic(Operation operation, int a, int b, int c, Mode mode) {
207 if (operation == Operation.SUB) { b ^= sign; operation = Operation.ADD; }
208 int[] inputs = {a, b, c};
209 int count = operation == Operation.SQRT ? 1 : operation == Operation.FMA ? 3 : 2;
210 boolean hasNaN = false;
211 int nanFlags = 0;
212 for (int i = 0; i < count; i++) {
213 hasNaN |= isNaN(inputs[i]);
214 if (kind(inputs[i]) == Kind.SNAN) nanFlags |= INVALID;
215 }
216 if (hasNaN) return new Rounded(nan, nanFlags);
217 boolean ai = kind(a) == Kind.INFINITY, bi = kind(b) == Kind.INFINITY;
218 boolean negative = negative(a) != negative(b);
219 Rounded invalid = new Rounded(nan, INVALID);
220 return switch (operation) {
221 case ADD -> {
222 if (ai && bi && negative) yield invalid;
223 if (ai || bi) yield new Rounded(ai ? a : b);
224 yield roundSum(mode, term(a), term(b));
225 }
226 case MUL -> {
227 if (ai || bi) yield zero(a) || zero(b) ? invalid : new Rounded(signed(infinity, negative));
228 yield roundTerm(term(a).product(term(b)), mode);
229 }
230 case DIV -> {
231 if (ai) yield bi ? invalid : new Rounded(signed(infinity, negative));
232 if (bi) yield new Rounded(signed(0, negative));
233 if (zero(b)) yield zero(a) ? invalid : new Rounded(signed(infinity, negative), DIVBYZERO);
234 if (zero(a)) yield new Rounded(signed(0, negative));
235 Term x = term(a), y = term(b);
236 yield round(new Exact(negative, x.magnitude, x.exponent - y.exponent, y.magnitude), mode, false);
237 }
238 case SQRT -> {
239 if (zero(a)) yield new Rounded(a);
240 if (negative(a)) yield invalid;
241 if (ai) yield new Rounded(a);
242 yield round(term(a).exact(), mode, true);
243 }
244 case FMA -> {
245 if (ai || bi) {
246 if (zero(a) || zero(b) || kind(c) == Kind.INFINITY && negative(c) != negative) yield invalid;
247 yield new Rounded(signed(infinity, negative));
248 }
249 if (kind(c) == Kind.INFINITY) yield new Rounded(c);
250 yield roundSum(mode, term(a).product(term(b)), term(c));
251 }
252 case SUB -> throw new AssertionError();
253 };
254 }

256 Input rawOperand(String word, boolean encoding) {
257 if (word.startsWith("0b") || word.startsWith("0x")) {
258 boolean binary = word.startsWith("0b");
259 int width = binary ? w : (w + 3) / 4;
260 if (encoding || word.length() - 2 != width) throw new CommandError("ERROR BAD OPERAND " + word);
261 int p = Integer.parseInt(word.substring(2), binary ? 2 : 16);
262 if (p >= 2 * sign) throw new CommandError("ERROR BAD OPERAND " + word);
263 return new Input(p, null);
264 }
265 switch (word) {
266 case "nan": return new Input(nan, null);
267 case "inf": return new Input(infinity, null);
268 case "-inf": return new Input(infinity | sign, null);
269 }
270 boolean negative = word.startsWith("-");
271 String digits = negative ? word.substring(1) : word;
272 int slash = digits.indexOf('/'), dot = digits.indexOf('.');
273 long numerator, denominator = 1;
274 if (slash >= 0) {
275 numerator = Long.parseLong(digits.substring(0, slash));
276 denominator = Long.parseLong(digits.substring(slash + 1));
277 if (denominator == 0) throw new CommandError("ERROR BAD OPERAND " + word);
278 } else if (dot >= 0) {
279 numerator = Long.parseLong(digits.substring(0, dot) + digits.substring(dot + 1));
280 for (int i = dot + 1; i < digits.length(); i++) denominator *= 10;
281 } else numerator = Long.parseLong(digits);
282 return new Input(0, new Exact(negative, numerator, 0, denominator));
283 }
284 int operand(String word) {
285 Input value = rawOperand(word, false);
286 return value.number == null ? value.pattern : round(value.number, Mode.RNE, false).pattern;
287 }
288 String bits(int p) { return (negative(p) ? 1 : 0) + "|" + binary((p & (sign - 1)) >> m, e) + "|" + binary(p & (base - 1), m); }
289 String hex(int p) { return String.format(Locale.ROOT, "0x%0" + (w + 3) / 4 + "X", p); }
290 String value(int p) {
291 return switch (kind(p)) {
292 case QNAN, SNAN -> "nan";
293 case INFINITY -> negative(p) ? "-inf" : "inf";
294 case ZERO -> negative(p) ? "-0" : "0";
295 default -> term(p).decimal();
296 };
297 }
298 String pattern(int p) { return bits(p) + " " + hex(p) + " " + value(p); }
299 String result(Rounded result) { return pattern(result.pattern) + " " + flagText(result.flags); }
300 String difference(int p, Term exactSum) {
301 return finite(p) ? sumTerms(Mode.RNE, term(p), exactSum.negated()).exact().fraction() : value(p);
302 }
303 long orderKey(int p) { return negative(p) ? -(long) (p - sign) - 1 : p; }
304 int nextUp(int p) {
305 if (p == infinity) return p;
306 if (zero(p)) return 1;
307 return negative(p) ? p - 1 : p + 1;
308 }
309 }

311 static String binary(int value, int width) {
312 String digits = Integer.toBinaryString(value);
313 return "0".repeat(width - digits.length()) + digits;

315 static String flagText(int flags) {
316 String[] names = {"invalid", "divbyzero", "overflow", "underflow", "inexact"};
317 StringJoiner out = new StringJoiner(",");
318 for (int i = 0; i < names.length; i++) if ((flags & (1 << i)) != 0) out.add(names[i]);
319 return "flags=" + (out.length() == 0 ? "none" : out.toString());
320 }
321 static Mode parseMode(String word) {
322 try { return Mode.valueOf(word); }
323 catch (IllegalArgumentException error) { throw new CommandError("ERROR BAD MODE " + word); }
324 }
325 static Mode optionalMode(String[] words, int index) { return index == words.length ? Mode.RNE : parseMode(words[index]); }
326 static Bounds integerRange(String word) {
327 return switch (word) {
328 case "I8" -> new Bounds(-128, 127);
329 case "U8" -> new Bounds(0, 255);
330 case "I16" -> new Bounds(-32768, 32767);
331 case "U16" -> new Bounds(0, 65535);
332 case "I32" -> new Bounds(-2147483648L, 2147483647L);
333 case "U32" -> new Bounds(0, 4294967295L);
334 default -> throw new CommandError("ERROR BAD TYPE " + word);
335 };

340 static String summation(Format f, String[] words) {
341 String command = words[0];
342 Mode mode = parseMode(words[2]);
343 List<Integer> values = new ArrayList<>();
344 int leftCount = 0;
345 for (int i = 3; i < words.length; i++) {
346 if (words[i].equals("|")) leftCount = values.size();
347 else values.add(f.operand(words[i]));
348 }
349 if (command.equals("DOT") && leftCount != values.size() - leftCount) throw new CommandError("ERROR LENGTH MISMATCH");
350 for (int p : values) if (!f.finite(p)) throw new CommandError("ERROR NOT FINITE");
351 StringBuilder out = new StringBuilder();
352 if (command.equals("DOT")) {
353 Term[] terms = new Term[leftCount];
354 for (int i = 0; i < leftCount; i++) terms[i] = f.term(values.get(i)).product(f.term(values.get(leftCount + i)));
355 Term exactSum = sumTerms(mode, terms);
356 Rounded first = f.arithmetic(Operation.MUL, values.getFirst(), values.get(leftCount), 0, mode);
357 Rounded naive = first, chained = first;
358 for (int i = 1; i < leftCount; i++) {
359 Rounded multiplied = f.arithmetic(Operation.MUL, values.get(i), values.get(leftCount + i), 0, mode);
360 Rounded added = f.arithmetic(Operation.ADD, naive.pattern, multiplied.pattern, 0, mode);
361 naive = new Rounded(added.pattern, naive.flags | multiplied.flags | added.flags);
362 }
363 for (int i = 1; i < leftCount; i++) {
364 Rounded fused = f.arithmetic(Operation.FMA, values.get(i), values.get(leftCount + i), chained.pattern, mode);
365 chained = new Rounded(fused.pattern, chained.flags | fused.flags);
366 }
367 line(out, "DOT %s %s n=%d", f.name, mode, leftCount);
368 line(out, "NAIVE %s", f.result(naive));
369 line(out, "CHAINED %s", f.result(chained));
370 line(out, "EXACT %s", exactSum.decimal());
371 line(out, "FUSED %s", f.result(f.roundTerm(exactSum, mode)));
372 line(out, "DIFF naive=%s chained=%s", f.difference(naive.pattern, exactSum), f.difference(chained.pattern, exactSum));
373 return out.toString();
374 }
375 Term[] terms = values.stream().map(f::term).toArray(Term[]::new);
376 Term exactSum = sumTerms(mode, terms);
377 Rounded result = new Rounded(values.getFirst());
378 int compensation = 0;
379 line(out, "%s %s %s count=%d", command, f.name, mode, values.size());
380 for (int i = 1; i < values.size(); i++) {
381 if (command.equals("SUM")) {
382 Rounded next = f.arithmetic(Operation.ADD, result.pattern, values.get(i), 0, mode);
383 result = new Rounded(next.pattern, result.flags | next.flags);
384 line(out, "PARTIAL %d %s", i + 1, f.result(next));
385 } else {
386 Rounded y = f.arithmetic(Operation.SUB, values.get(i), compensation, 0, mode);
387 Rounded t = f.arithmetic(Operation.ADD, result.pattern, y.pattern, 0, mode);
388 Rounded d = f.arithmetic(Operation.SUB, t.pattern, result.pattern, 0, mode);
389 Rounded c = f.arithmetic(Operation.SUB, d.pattern, y.pattern, 0, mode);
390 int flags = y.flags | t.flags | d.flags | c.flags;
391 result = new Rounded(t.pattern, result.flags | flags);
392 compensation = c.pattern;
393 line(out, "KSTEP %d sum=%s comp=%s %s", i + 1, f.value(result.pattern), f.value(compensation), flagText(flags));
394 }
395 }
396 line(out, "RESULT %s", f.result(result));
397 line(out, "EXACT %s", exactSum.decimal());
398 line(out, "ROUNDED %s", f.result(f.roundTerm(exactSum, mode)));
399 line(out, "DIFF %s", f.difference(result.pattern, exactSum));
400 return out.toString();
401 }

403 static String table(Format f) {
404 if (f.w > 8) throw new CommandError("ERROR TABLE TOO LARGE");
405 StringBuilder out = new StringBuilder();
406 line(out, "TABLE %s count=%d", f.name, f.infinity);
407 for (int p = 0; p < f.infinity; p++) {
408 String gap = p + 1 == f.infinity ? "-" : sumTerms(Mode.RNE, f.term(p + 1), f.term(p).negated()).decimal();
409 line(out, "ENTRY %d %s gap=%s", p, f.pattern(p), gap);
410 }
411 line(out, "MAX %s", f.pattern(f.infinity - 1));
412 line(out, "MINNORMAL %s", f.pattern(f.base));
413 line(out, "MINSUBNORMAL %s", f.pattern(1));
414 line(out, "EPSILON %s", new Term(false, 1, -f.m).decimal());
415 line(out, "PATTERNS finite=%d infinite=2 nan=%d total=%d", 2 * f.infinity, 2 * (f.base - 1), 2 * f.sign);
416 return out.toString();
417 }

425 static String execute(Map<String, Format> formats, String[] words) {
426 String command = words[0];
427 StringBuilder out = new StringBuilder();
428 if (command.equals("FORMAT")) {
429 if (formats.containsKey(words[1])) throw new CommandError("ERROR FORMAT EXISTS " + words[1]);
430 int e = Integer.parseInt(words[2]), m = Integer.parseInt(words[3]);
431 if (e < 2 || e > 5 || m < 1 || m > 10) throw new CommandError("ERROR BAD FORMAT");
432 Format f = new Format(words[1], e, m);
433 formats.put(f.name, f);
434 line(out, "FORMAT %s e=%d m=%d bits=%d bias=%d emin=%d emax=%d", f.name, e, m, f.w, f.bias, f.emin, f.emax);
435 return out.toString();
436 }
437 Format f = findFormat(formats, words[1]);
438 if (command.equals("TABLE")) return table(f);
439 if (command.equals("SUM") || command.equals("KAHAN") || command.equals("DOT")) return summation(f, words);
440 if (command.equals("CONVERT")) {
441 Format g = findFormat(formats, words[2]);
442 int a = f.operand(words[3]);
443 Mode mode = optionalMode(words, 4);
444 Rounded result = switch (f.kind(a)) {
445 case QNAN -> new Rounded(g.nan);
446 case SNAN -> new Rounded(g.nan, INVALID);
447 case INFINITY -> new Rounded(g.signed(g.infinity, f.negative(a)));
448 case ZERO -> new Rounded(g.signed(0, f.negative(a)));
449 default -> g.roundTerm(f.term(a), mode);
450 };
451 line(out, "CONVERT %s %s %s %s", f.name, g.name, mode, g.result(result));
452 return out.toString();
453 }
454 if (command.equals("ENCODE")) {
455 Input value = f.rawOperand(words[2], true);
456 Mode mode = optionalMode(words, 3);
457 Rounded result = value.number == null ? new Rounded(value.pattern) : f.round(value.number, mode, false);
458 String source = value.number == null ? f.value(value.pattern) : value.number.fraction();
459 line(out, "ENCODE %s %s %s %s", f.name, mode, source, f.result(result));
460 return out.toString();
461 }
462 int a = f.operand(words[2]);
463 switch (command) {
464 case "DECODE" -> {
465 Kind kind = f.kind(a);
466 int exponent = (a & (f.sign - 1)) >> f.m, fraction = a & (f.base - 1);
467 line(out, "DECODE %s %s %s %s sign=%d", f.name, f.bits(a), f.hex(a), kind, f.negative(a) ? 1 : 0);
468 if (f.finite(a)) {
469 Term term = f.term(a);
470 line(out, "FIELDS exponent=%d fraction=%d unbiased=%d significand=%d.%s", exponent, fraction,
471 term.exponent + f.m, kind == Kind.NORMAL ? 1 : 0, binary(fraction, f.m));
472 line(out, "VALUE %s %s", term.exact().fraction(), f.value(a));
473 } else if (kind == Kind.INFINITY) line(out, "FIELDS exponent=%d fraction=%d", exponent, fraction);
474 else line(out, "FIELDS exponent=%d fraction=%d payload=%d", exponent, fraction, fraction % (f.base / 2));
475 }
476 case "COMPARE" -> {
477 int b = f.operand(words[3]);
478 long ak = f.orderKey(a), bk = f.orderKey(b);
479 String numeric = f.isNaN(a) || f.isNaN(b) ? "UN" : f.zero(a) && f.zero(b) ? "EQ" : relation(ak, bk);
480 int flags = f.kind(a) == Kind.SNAN || f.kind(b) == Kind.SNAN ? INVALID : 0;
481 line(out, "COMPARE %s %s %s %s total=%s %s", f.name, f.hex(a), f.hex(b), numeric, relation(ak, bk), flagText(flags));
482 }
483 case "TOINT" -> {
484 Bounds bounds = integerRange(words[3]);
485 Mode mode = optionalMode(words, 4);
486 boolean invalid = !f.finite(a);
487 long result = 0;
488 int flags = 0;
489 if (!invalid) {
490 Exact value = f.term(a).exact();
491 Quantized divided = Quantized.of(value, 0, false);
492 long magnitude = divided.rounded(mode, value.negative);
493 result = value.negative ? -magnitude : magnitude;
494 invalid = result < bounds.minimum || result > bounds.maximum;
495 flags = divided.inexact ? INEXACT : 0;
496 }
497 if (invalid) line(out, "TOINT %s %s %s invalid flags=invalid", f.name, words[3], mode);
498 else line(out, "TOINT %s %s %s %d %s", f.name, words[3], mode, result, flagText(flags));
499 }
500 case "NEXTUP", "NEXTDOWN", "ULP" -> {
501 Rounded result;
502 if (f.isNaN(a)) result = new Rounded(f.nan, f.kind(a) == Kind.SNAN ? INVALID : 0);
503 else {
504 int pattern = switch (command) {
505 case "NEXTUP" -> f.nextUp(a);
506 case "NEXTDOWN" -> f.nextUp(a ^ f.sign) ^ f.sign;
507 default -> f.kind(a) == Kind.INFINITY ? f.infinity
508 : f.roundTerm(new Term(false, 1, f.term(a).exponent), Mode.RNE).pattern;
509 };
510 result = new Rounded(pattern);
511 }
512 line(out, "%s %s %s", command, f.name, f.result(result));
513 }
514 default -> {
515 Operation operation = Operation.valueOf(command);
516 int arity = operation == Operation.SQRT ? 1 : operation == Operation.FMA ? 3 : 2;
517 int b = arity >= 2 ? f.operand(words[3]) : 0;
518 int c = arity == 3 ? f.operand(words[4]) : 0;
519 Mode mode = optionalMode(words, arity + 2);
520 line(out, "%s %s %s %s", command, f.name, mode, f.result(f.arithmetic(operation, a, b, c, mode)));
521 }
522 }
523 return out.toString();
524 }

537 }
