46function gcd(a: bigint, b: bigint): bigint {
47 while (b !== 0n) [a, b] = [b, a % b];
48 return a;
49}
50function scaled(numerator: bigint, denominator: bigint, exponent: number): readonly [bigint, bigint] {
51 return exponent >= 0
52 ? [numerator << BigInt(exponent), denominator]
53 : [numerator, denominator << BigInt(-exponent)];
54}
55function integerSqrt(value: bigint): bigint {
56 if (value === 0n) return 0n;
57 let root = 1n << BigInt((value.toString(2).length + 1) >> 1);
58 while (true) {
59 const next = (root + value / root) >> 1n;
60 if (next >= root) return root;
61 root = next;
62 }
63}

65class Exact {
66 constructor(
67 readonly n: bigint,
68 readonly e = 0,
69 readonly d = 1n,
70 readonly sign: Sign = 0,
71 ) {}
72
73 negated(): Exact { return new Exact(this.n, this.e, this.d, oppositeSign(this.sign)); }
74 log2Floor(): number {
75 let exponent = this.n.toString(2).length - this.d.toString(2).length + this.e;
76 const [n, d] = scaled(this.n, this.d, this.e - exponent);
77 if (n < d) exponent--;
78 return exponent;
79 }
80 fraction(): string {
81 if (this.n === 0n) return '0';
82 let [n, d] = scaled(this.n, this.d, this.e);
83 const divisor = gcd(n, d);
84 n /= divisor;
85 d /= divisor;
86 return (this.sign ? '-' : '') + n + (d === 1n ? '' : `/${d}`);
87 }
88 decimal(keepZeroSign = false): string {
89 const [n, d] = scaled(this.n, this.d, this.e);
90 let text = String(n / d);
91 let remainder = n % d;
92 if (remainder !== 0n) {
93 text += '.';
94 while (remainder !== 0n) {
95 remainder *= 10n;
96 text += remainder / d;
97 remainder %= d;
98 }
99 }
100 return (this.sign && (this.n !== 0n || keepZeroSign) ? '-' : '') + text;
101 }
102}
103function exactProduct(a: Exact, b: Exact): Exact {
104 return new Exact(a.n * b.n, a.e + b.e, a.d * b.d, xorSign(a.sign, b.sign));
105}
106function exactSum(terms: readonly Exact[], mode: Mode): Exact {
107 const nonzero = terms.filter(term => term.n !== 0n);
108 if (nonzero.length === 0) {
109 const sameSign = terms.every(term => term.sign === terms[0].sign);
110 return new Exact(0n, 0, 1n, sameSign ? terms[0].sign : mode === 'RDN' ? 1 : 0);
111 }
112 const exponent = Math.min(...nonzero.map(term => term.e));
113 let numerator = 0n;
114 for (const term of nonzero) {
115 const aligned = term.n << BigInt(term.e - exponent);
116 numerator += term.sign ? -aligned : aligned;
117 }
118 const sign: Sign = numerator < 0n || numerator === 0n && mode === 'RDN' ? 1 : 0;
119 return new Exact(numerator < 0n ? -numerator : numerator, exponent, 1n, sign);
120}
121function increment(mode: Mode, sign: Sign, integer: bigint, inexact: boolean, halfway: bigint): boolean {
122 if (!inexact) return false;
123 switch (mode) {
124 case 'RNE': return halfway > 0n || halfway === 0n && integer % 2n === 1n;
125 case 'RNA': return halfway >= 0n;
126 case 'RTZ': return false;
127 case 'RUP': return sign === 0;
128 case 'RDN': return sign === 1;
129 }
130}
131function roundedUnits(value: Exact, exponent: number, mode: Mode): Units {
132 const [n, d] = scaled(value.n, value.d, value.e - exponent);
133 let integer = n / d;
134 const remainder = n % d;
135 const inexact = remainder !== 0n;
136 if (increment(mode, value.sign, integer, inexact, 2n * remainder - d)) integer++;
137 return { integer, inexact };
138}
139function roundedSqrtUnits(value: Exact, exponent: number, mode: Mode): Units {
140 const [n, d] = scaled(value.n, value.d, value.e - 2 * exponent);
141 let integer = integerSqrt(n / d);
142 const inexact = integer * integer * d !== n;
143 const midpoint = 2n * integer + 1n;
144 const halfway = 4n * n - d * midpoint * midpoint;
145 if (increment(mode, 0, integer, inexact, halfway)) integer++;
146 return { integer, inexact };
147}

149class MiniFormat {
150 readonly w: number;
151 readonly bias: number;
152 readonly emin: number;
153 readonly emax: number;
154 readonly hidden: number;
155 readonly exponentMask: number;
156 readonly signBit: number;
157 readonly infinity: number;
158 readonly nan: number;
159 readonly maximum: number;
160
161 constructor(readonly name: string, readonly e: number, readonly m: number) {
162 this.w = 1 + e + m;
163 this.bias = (1 << (e - 1)) - 1;
164 this.emin = 1 - this.bias;
165 this.emax = this.bias;
166 this.hidden = 1 << m;
167 this.exponentMask = (1 << e) - 1;
168 this.signBit = 1 << (e + m);
169 this.infinity = this.exponentMask * this.hidden;
170 this.nan = this.infinity + (this.hidden >> 1);
171 this.maximum = this.infinity - 1;
172 }
173 fields(pattern: number): Fields {
174 const sign: Sign = pattern & this.signBit ? 1 : 0;
175 const exponent = (pattern >> this.m) & this.exponentMask;
176 const fraction = pattern & (this.hidden - 1);
177 const q = (exponent === 0 ? this.emin : exponent - this.bias) - this.m;
178 return { sign, exponent, fraction, q };
179 }
180 isNaN(pattern: number): boolean {
181 const field = this.fields(pattern);
182 return field.exponent === this.exponentMask && field.fraction !== 0;
183 }
184 isSignaling(pattern: number): boolean { return this.isNaN(pattern) && this.fields(pattern).fraction < (this.hidden >> 1); }
185 isInfinity(pattern: number): boolean { return (pattern & (this.signBit - 1)) === this.infinity; }
186 isZero(pattern: number): boolean { return (pattern & (this.signBit - 1)) === 0; }
187 isFinite(pattern: number): boolean { return this.fields(pattern).exponent !== this.exponentMask; }
188 classification(pattern: number): PatternClass {
189 if (this.isNaN(pattern)) return this.isSignaling(pattern) ? 'snan' : 'qnan';
190 if (this.isInfinity(pattern)) return 'infinity';
191 if (this.isZero(pattern)) return 'zero';
192 return this.fields(pattern).exponent === 0 ? 'subnormal' : 'normal';
193 }
194 zero(sign: Sign): number { return sign * this.signBit; }
195 inf(sign: Sign): number { return this.zero(sign) + this.infinity; }
196 flip(pattern: number): number { return pattern ^ this.signBit; }
197 exact(pattern: number): Exact {
198 const field = this.fields(pattern);
199 return new Exact(BigInt(field.exponent === 0 ? field.fraction : this.hidden + field.fraction), field.q, 1n, field.sign);
200 }
201 bits(pattern: number): string {
202 const field = this.fields(pattern);
203 return `${field.sign}|${field.exponent.toString(2).padStart(this.e, '0')}|${field.fraction.toString(2).padStart(this.m, '0')}`;
204 }
205 hex(pattern: number): string { return '0x' + pattern.toString(16).toUpperCase().padStart((this.w + 3) >> 2, '0'); }
206 value(pattern: number): string {
207 if (this.isNaN(pattern)) return 'nan';
208 if (this.isInfinity(pattern)) return this.fields(pattern).sign ? '-inf' : 'inf';
209 return this.exact(pattern).decimal(true);
210 }
211 patternText(pattern: number): string { return `${this.bits(pattern)} ${this.hex(pattern)} ${this.value(pattern)}`; }
212 resultText(value: Result): string { return `${this.patternText(value.pattern)} ${flagsText(value.flags)}`; }
213 overflow(sign: Sign, mode: Mode): Result {
214 const infinite = mode === 'RNE' || mode === 'RNA' || mode === 'RUP' && sign === 0 || mode === 'RDN' && sign === 1;
215 return result(this.zero(sign) + (infinite ? this.infinity : this.maximum), OVERFLOW | INEXACT);
216 }
217 private finishRound(sign: Sign, order: number, exponent: number, units: Units, unbounded: () => Units, mode: Mode): Result {
218 const hidden = BigInt(this.hidden);
219 let n = units.integer;
220 if (n === 2n * hidden) {
221 n = hidden;
222 exponent++;
223 }
224 if (exponent > this.emax - this.m) return this.overflow(sign, mode);
225 const magnitude = n >= hidden ? (exponent + this.m + this.bias) * this.hidden + Number(n - hidden) : Number(n);
226 let flags = units.inexact ? INEXACT : 0;
227 if (units.inexact && order < this.emin && (order < this.emin - 1 || unbounded().integer < 2n * hidden)) flags |= UNDERFLOW;
228 return result(this.zero(sign) + magnitude, flags);
229 }
230 round(value: Exact, mode: Mode): Result {
231 if (value.n === 0n) return result(this.zero(value.sign));
232 const order = value.log2Floor();
233 if (order > this.emax) return this.overflow(value.sign, mode);
234 const exponent = Math.max(order, this.emin) - this.m;
235 return this.finishRound(value.sign, order, exponent, roundedUnits(value, exponent, mode),
236 () => roundedUnits(value, order - this.m, mode), mode);
237 }
238 sqrt(value: Exact, mode: Mode): Result {
239 const log = value.log2Floor();
240 const order = log >= 0 ? log >> 1 : -((-log + 1) >> 1);
241 const exponent = Math.max(order, this.emin) - this.m;
242 return this.finishRound(0, order, exponent, roundedSqrtUnits(value, exponent, mode),
243 () => roundedSqrtUnits(value, order - this.m, mode), mode);
244 }
245}

247function parseReal(word: string): Real {
248 if (word === 'nan' || word === 'inf' || word === '-inf') return { kind: 'special', value: word };
249 if (word.startsWith('0b') || word.startsWith('0x')) fail(`BAD OPERAND ${word}`);
250 const sign: Sign = word.startsWith('-') ? 1 : 0;
251 const body = sign ? word.slice(1) : word;
252 if (body.includes('/')) {
253 const [p, q] = body.split('/').map(BigInt);
254 if (q === 0n) fail(`BAD OPERAND ${word}`);
255 return { kind: 'exact', value: new Exact(p, 0, q, sign) };
256 }
257 const [whole, fraction = ''] = body.split('.');
258 return { kind: 'exact', value: new Exact(BigInt(whole + fraction), 0, 10n ** BigInt(fraction.length), sign) };
259}
260function encoded(format: MiniFormat, real: Real, mode: Mode): Result {
261 if (real.kind === 'exact') return format.round(real.value, mode);
262 if (real.value === 'nan') return result(format.nan);
263 return result(format.inf(real.value === '-inf' ? 1 : 0));
264}
265function operand(format: MiniFormat, word: string): number {
266 if (word.startsWith('0b')) {
267 if (word.length - 2 !== format.w) fail(`BAD OPERAND ${word}`);
268 return Number(BigInt(word));
269 }
270 if (word.startsWith('0x')) {
271 const pattern = Number(BigInt(word));
272 if (word.length - 2 !== ((format.w + 3) >> 2) || pattern >= (1 << format.w)) fail(`BAD OPERAND ${word}`);
273 return pattern;
274 }
275 return encoded(format, parseReal(word), 'RNE').pattern;
276}
277function operation(format: MiniFormat, command: ArithmeticCommand, inputs: readonly number[], mode: Mode): Result {
278 if (command === 'SUB') return operation(format, 'ADD', [inputs[0], format.flip(inputs[1])], mode);
279 if (inputs.some(pattern => format.isNaN(pattern))) {
280 return result(format.nan, inputs.some(pattern => format.isSignaling(pattern)) ? INVALID : 0);
281 }
282 const [a, b, c] = inputs;
283 const invalid = (): Result => result(format.nan, INVALID);
284 const sign = format.fields(a).sign;
285 if (command === 'SQRT') {
286 if (format.isZero(a)) return result(a);
287 if (sign) return invalid();
288 if (format.isInfinity(a)) return result(a);
289 return format.sqrt(format.exact(a), mode);
290 }
291 const sigma = xorSign(sign, format.fields(b).sign);
292 switch (command) {
293 case 'ADD':
294 if (format.isInfinity(a) && format.isInfinity(b) && sigma) return invalid();
295 if (format.isInfinity(a)) return result(a);
296 if (format.isInfinity(b)) return result(b);
297 return format.round(exactSum([format.exact(a), format.exact(b)], mode), mode);
298 case 'MUL':
299 if (format.isInfinity(a) || format.isInfinity(b)) return format.isZero(a) || format.isZero(b) ? invalid() : result(format.inf(sigma));
300 if (format.isZero(a) || format.isZero(b)) return result(format.zero(sigma));
301 return format.round(exactProduct(format.exact(a), format.exact(b)), mode);
302 case 'DIV': {
303 if (format.isInfinity(a)) return format.isInfinity(b) ? invalid() : result(format.inf(sigma));
304 if (format.isInfinity(b)) return result(format.zero(sigma));
305 if (format.isZero(b)) return format.isZero(a) ? invalid() : result(format.inf(sigma), DIVBYZERO);
306 if (format.isZero(a)) return result(format.zero(sigma));
307 const x = format.exact(a), y = format.exact(b);
308 return format.round(new Exact(x.n * y.d, x.e - y.e, x.d * y.n, sigma), mode);
309 }
310 case 'FMA':
311 if (format.isInfinity(a) || format.isInfinity(b)) {
312 if (format.isZero(a) || format.isZero(b) || format.isInfinity(c) && format.fields(c).sign !== sigma) return invalid();
313 return result(format.inf(sigma));
314 }
315 if (format.isInfinity(c)) return result(c);
316 return format.round(exactSum([exactProduct(format.exact(a), format.exact(b)), format.exact(c)], mode), mode);
317 }
318}
319function nextUp(format: MiniFormat, pattern: number): number {
320 if (pattern === format.infinity) return pattern;
321 if (format.isZero(pattern)) return 1;
322 return pattern + (format.fields(pattern).sign ? -1 : 1);
323}
324function difference(format: MiniFormat, pattern: number, exact: Exact): string {
325 if (!format.isFinite(pattern)) return format.value(pattern);
326 return exactSum([format.exact(pattern), exact.negated()], 'RNE').fraction();
327}
328const relation = (a: number, b: number): Comparison => a < b ? 'LT' : a > b ? 'GT' : 'EQ';

330class Laboratory {
331 private readonly formats = new Map<string, MiniFormat>([
332 ['HALF', new MiniFormat('HALF', 5, 10)], ['E5M2', new MiniFormat('E5M2', 5, 2)], ['E4M3', new MiniFormat('E4M3', 4, 3)],
333 ]);
334 readonly output: string[] = [];
335
336 emit(line: string): void { this.output.push(line); }
337 private format(name: string): MiniFormat {
338 const format = this.formats.get(name);
339 if (!format) fail(`UNKNOWN FORMAT ${name}`);
340 return format;
341 }
342 private summary(format: MiniFormat, mode: Mode, patterns: readonly number[], pattern: number, flags: number): void {
343 const exact = exactSum(patterns.map(pattern => format.exact(pattern)), mode);
344 this.emit(`RESULT ${format.resultText(result(pattern, flags))}`);
345 this.emit(`EXACT ${exact.decimal()}`);
346 this.emit(`ROUNDED ${format.resultText(format.round(exact, mode))}`);
347 this.emit(`DIFF ${difference(format, pattern, exact)}`);
348 }
349 private summation(command: SummationCommand, format: MiniFormat, mode: Mode, patterns: readonly number[]): void {
350 if (patterns.some(pattern => !format.isFinite(pattern))) fail('NOT FINITE');
351 this.emit(`${command} ${format.name} ${mode} count=${patterns.length}`);
352 let sum = patterns[0], compensation = 0, flags = 0;
353 for (let index = 1; index < patterns.length; index++) {
354 if (command === 'SUM') {
355 const added = operation(format, 'ADD', [sum, patterns[index]], mode);
356 sum = added.pattern;
357 flags |= added.flags;
358 this.emit(`PARTIAL ${index + 1} ${format.resultText(added)}`);
359 } else {
360 const y = operation(format, 'SUB', [patterns[index], compensation], mode);
361 const t = operation(format, 'ADD', [sum, y.pattern], mode);
362 const d = operation(format, 'SUB', [t.pattern, sum], mode);
363 const correction = operation(format, 'SUB', [d.pattern, y.pattern], mode);
364 sum = t.pattern;
365 compensation = correction.pattern;
366 const stepFlags = y.flags | t.flags | d.flags | correction.flags;
367 flags |= stepFlags;
368 this.emit(`KSTEP ${index + 1} sum=${format.value(sum)} comp=${format.value(compensation)} ${flagsText(stepFlags)}`);
369 }
370 }
371 this.summary(format, mode, patterns, sum, flags);
372 }
373 private dot(format: MiniFormat, mode: Mode, a: readonly number[], b: readonly number[]): void {
374 if (a.length !== b.length) fail('LENGTH MISMATCH');
375 if ([...a, ...b].some(pattern => !format.isFinite(pattern))) fail('NOT FINITE');
376 let naive = operation(format, 'MUL', [a[0], b[0]], mode);
377 let chained = naive;
378 for (let index = 1; index < a.length; index++) {
379 const product = operation(format, 'MUL', [a[index], b[index]], mode);
380 const added = operation(format, 'ADD', [naive.pattern, product.pattern], mode);
381 naive = result(added.pattern, naive.flags | product.flags | added.flags);
382 const fused = operation(format, 'FMA', [a[index], b[index], chained.pattern], mode);
383 chained = result(fused.pattern, chained.flags | fused.flags);
384 }
385 const exact = exactSum(a.map((pattern, index) => exactProduct(format.exact(pattern), format.exact(b[index]))), mode);
386 this.emit(`DOT ${format.name} ${mode} n=${a.length}`);
387 this.emit(`NAIVE ${format.resultText(naive)}`);
388 this.emit(`CHAINED ${format.resultText(chained)}`);
389 this.emit(`EXACT ${exact.decimal()}`);
390 this.emit(`FUSED ${format.resultText(format.round(exact, mode))}`);
391 this.emit(`DIFF naive=${difference(format, naive.pattern, exact)} chained=${difference(format, chained.pattern, exact)}`);
392 }
393 private table(format: MiniFormat): void {
394 if (format.w > 8) fail('TABLE TOO LARGE');
395 const count = format.infinity;
396 this.emit(`TABLE ${format.name} count=${count}`);
397 for (let pattern = 0; pattern < count; pattern++) {
398 const gap = pattern === count - 1 ? '-'
399 : exactSum([format.exact(pattern + 1), format.exact(pattern).negated()], 'RNE').decimal();
400 this.emit(`ENTRY ${pattern} ${format.patternText(pattern)} gap=${gap}`);
401 }
402 this.emit(`MAX ${format.patternText(format.maximum)}`);
403 this.emit(`MINNORMAL ${format.patternText(format.hidden)}`);
404 this.emit(`MINSUBNORMAL ${format.patternText(1)}`);
405 this.emit(`EPSILON ${new Exact(1n, -format.m).decimal()}`);
406 this.emit(`PATTERNS finite=${2 * count} infinite=2 nan=${2 * (format.hidden - 1)} total=${1 << format.w}`);
407 }
408 execute(words: readonly string[]): void {
409 const command = words[0] as Command;
410 if (command === 'FORMAT') {
411 const name = words[1], e = Number(words[2]), m = Number(words[3]);
412 if (this.formats.has(name)) fail(`FORMAT EXISTS ${name}`);
413 if (e < 2 || e > 5 || m < 1 || m > 10) fail('BAD FORMAT');
414 const format = new MiniFormat(name, e, m);
415 this.formats.set(name, format);
416 this.emit(`FORMAT ${name} e=${e} m=${m} bits=${format.w} bias=${format.bias} emin=${format.emin} emax=${format.emax}`);
417 return;
418 }
419 const format = this.format(words[1]);
420 const name = format.name;
421 if (command === 'ENCODE') {
422 const real = parseReal(words[2]);
423 const mode = modeOf(words[3]);
424 const value = real.kind === 'special' ? real.value : real.value.fraction();
425 this.emit(`ENCODE ${name} ${mode} ${value} ${format.resultText(encoded(format, real, mode))}`);
426 } else if (command === 'CONVERT') {
427 const target = this.format(words[2]);
428 const a = operand(format, words[3]);
429 const mode = modeOf(words[4]);
430 const converted = format.isNaN(a) ? result(target.nan, format.isSignaling(a) ? INVALID : 0)
431 : format.isInfinity(a) ? result(target.inf(format.fields(a).sign)) : target.round(format.exact(a), mode);
432 this.emit(`CONVERT ${name} ${target.name} ${mode} ${target.resultText(converted)}`);
433 } else if (command === 'SUM' || command === 'KAHAN' || command === 'DOT') {
434 const mode = modeOf(words[2]);
435 if (command === 'DOT') {
436 const separator = words.indexOf('|');
437 const a = words.slice(3, separator).map(word => operand(format, word));
438 const b = words.slice(separator + 1).map(word => operand(format, word));
439 this.dot(format, mode, a, b);
440 } else this.summation(command, format, mode, words.slice(3).map(word => operand(format, word)));
441 } else if (command === 'TABLE') this.table(format);
442 else if (isArithmetic(command)) {
443 const count = command === 'SQRT' ? 1 : command === 'FMA' ? 3 : 2;
444 const patterns = words.slice(2, 2 + count).map(word => operand(format, word));
445 const mode = modeOf(words[2 + count]);
446 this.emit(`${command} ${name} ${mode} ${format.resultText(operation(format, command, patterns, mode))}`);
447 } else {
448 const a = operand(format, words[2]);
449 const field = format.fields(a);
450 if (command === 'DECODE') {
451 this.emit(`DECODE ${name} ${format.bits(a)} ${format.hex(a)} ${format.classification(a)} sign=${field.sign}`);
452 let line = `FIELDS exponent=${field.exponent} fraction=${field.fraction}`;
453 if (format.isFinite(a)) {
454 line += ` unbiased=${field.q + format.m} significand=${field.exponent === 0 ? 0 : 1}.${field.fraction.toString(2).padStart(format.m, '0')}`;
455 this.emit(line);
456 this.emit(`VALUE ${format.exact(a).fraction()} ${format.value(a)}`);
457 } else {
458 if (format.isNaN(a)) line += ` payload=${field.fraction % (format.hidden >> 1)}`;
459 this.emit(line);
460 }
461 } else if (command === 'COMPARE') {
462 const b = operand(format, words[3]);
463 const key = (pattern: number): number => format.fields(pattern).sign ? -(pattern - format.signBit) - 1 : pattern;
464 const total = relation(key(a), key(b));
465 const numeric: Comparison | 'UN' = format.isNaN(a) || format.isNaN(b) ? 'UN'
466 : format.isZero(a) && format.isZero(b) ? 'EQ' : total;
467 const flags = format.isSignaling(a) || format.isSignaling(b) ? INVALID : 0;
468 this.emit(`COMPARE ${name} ${format.hex(a)} ${format.hex(b)} ${numeric} total=${total} ${flagsText(flags)}`);
469 } else if (command === 'TOINT') {
470 const type = typeOf(words[3]);
471 const mode = modeOf(words[4]);
472 const prefix = `TOINT ${name} ${type} ${mode}`;
473 if (!format.isFinite(a)) this.emit(`${prefix} invalid flags=invalid`);
474 else {
475 const units = roundedUnits(format.exact(a), 0, mode);
476 const integer = field.sign ? -units.integer : units.integer;
477 const [minimum, maximum] = TYPES[type];
478 if (integer < minimum || integer > maximum) this.emit(`${prefix} invalid flags=invalid`);
479 else this.emit(`${prefix} ${integer} ${flagsText(units.inexact ? INEXACT : 0)}`);
480 }
481 } else {
482 let value: Result;
483 if (format.isNaN(a)) value = result(format.nan, format.isSignaling(a) ? INVALID : 0);
484 else if (command === 'NEXTUP') value = result(nextUp(format, a));
485 else if (command === 'NEXTDOWN') value = result(format.flip(nextUp(format, format.flip(a))));
486 else value = result(format.isInfinity(a) ? format.infinity : format.round(new Exact(1n, field.q), 'RNE').pattern);
487 this.emit(`${command} ${name} ${format.resultText(value)}`);
488 }
489 }
490 }
491}

493const input = readFileSync(0, 'utf8').split(/\r?\n/);
494const laboratory = new Laboratory();
495for (const line of input.slice(1, Number(input[0]) + 1)) {
496 try { laboratory.execute(line.split(' ')); }
497 catch (problem) {
498 if (!(problem instanceof CommandError)) throw problem;
499 laboratory.emit(`ERROR ${problem.message}`);
500 }
501}
