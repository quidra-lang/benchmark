14struct Flags: OptionSet, Sendable {

53func padded(_ value: Int, radix: Int, width: Int) -> String {
54 let text = String(value, radix: radix, uppercase: true)
55 return String(repeating: "0", count: max(0, width - text.count)) + text

58func bitLength(_ value: Int64) -> Int {
59 var n = value, length = 0
60 while n != 0 {
61 n >>= 1
62 length += 1
63 }
64 return length
65}

67func scaled(_ numerator: Int64, _ denominator: Int64, exponent: Int) -> (Int64, Int64) {
68 exponent >= 0 ? (numerator << exponent, denominator) : (numerator, denominator << (-exponent))
69}
70
71func gcd(_ first: Int64, _ second: Int64) -> Int64 {
72 var a = first, b = second
73 while b != 0 { (a, b) = (b, a % b) }
74 return a
75}

77func integerSqrt(_ value: Int64) -> Int64 {
78 if value == 0 { return 0 }
79 var root = Int64(1) << ((bitLength(value) + 1) / 2)
80 while true {
81 let next = (root + value / root) / 2
82 if next >= root { return root }
83 root = next
84 }
85}

87struct Exact {
88 let n: Int64
89 let exponent: Int
90 let denominator: Int64
91 let negative: Bool
92
93 init(_ n: Int64 = 0, exponent: Int = 0, denominator: Int64 = 1, negative: Bool = false) {
94 self.n = n
95 self.exponent = exponent
96 self.denominator = denominator
97 self.negative = negative
98 }
99
100 func negated() -> Exact { Exact(n, exponent: exponent, denominator: denominator, negative: !negative) }
101
102 func log2Floor() -> Int {
103 var t = bitLength(n) - bitLength(denominator) + exponent
104 let (numerator, divisor) = scaled(n, denominator, exponent: exponent - t)
105 if numerator < divisor { t -= 1 }
106 return t
107 }
108
109 func fraction() -> String {
110 if n == 0 { return "0" }
111 var (numerator, divisor) = scaled(n, denominator, exponent: exponent)
112 let factor = gcd(numerator, divisor)
113 numerator /= factor
114 divisor /= factor
115 return (negative ? "-" : "") + String(numerator) + (divisor == 1 ? "" : "/\(divisor)")
116 }
117
118 func decimal(keepZeroSign: Bool = false) -> String {
119 let (numerator, divisor) = scaled(n, denominator, exponent: exponent)
120 var text = String(numerator / divisor)
121 var remainder = numerator % divisor
122 if remainder != 0 {
123 text += "."
124 while remainder != 0 {
125 remainder *= 10
126 text += String(remainder / divisor)
127 remainder %= divisor
128 }
129 }
130 return (negative && (n != 0 || keepZeroSign) ? "-" : "") + text
131 }
132}
133
134func exactProduct(_ a: Exact, _ b: Exact) -> Exact {
135 Exact(a.n * b.n, exponent: a.exponent + b.exponent,
136 denominator: a.denominator * b.denominator, negative: a.negative != b.negative)
137}
138
139func exactSum(_ terms: [Exact], mode: Mode) -> Exact {
140 let nonzero = terms.filter { $0.n != 0 }
141 if nonzero.isEmpty {
142 let sign = terms.allSatisfy { $0.negative == terms[0].negative } ? terms[0].negative : mode == .rdn
143 return Exact(negative: sign)
144 }
145 let exponent = nonzero.map(\.exponent).min()!
146 var numerator: Int64 = 0
147 for term in nonzero {
148 let aligned = term.n << (term.exponent - exponent)
149 numerator += term.negative ? -aligned : aligned
150 }
151 let negative = numerator < 0 || numerator == 0 && mode == .rdn
152 return Exact(abs(numerator), exponent: exponent, negative: negative)
153}
154
155func shouldIncrement(_ mode: Mode, negative: Bool, integer: Int64, inexact: Bool, halfway: Int64) -> Bool {
156 if !inexact { return false }
157 switch mode {
158 case .rne: return halfway > 0 || halfway == 0 && integer % 2 != 0
159 case .rna: return halfway >= 0
160 case .rtz: return false
161 case .rup: return !negative
162 case .rdn: return negative
163 }
164}
165
166func roundedUnits(_ value: Exact, exponent: Int, mode: Mode) -> Units {
167 let (n, d) = scaled(value.n, value.denominator, exponent: value.exponent - exponent)
168 var integer = n / d
169 let remainder = n % d
170 if shouldIncrement(mode, negative: value.negative, integer: integer, inexact: remainder != 0, halfway: 2 * remainder - d) {
171 integer += 1
172 }
173 return Units(integer: integer, inexact: remainder != 0)
174}
175
176func roundedSqrtUnits(_ value: Exact, exponent: Int, mode: Mode) -> Units {
177 let (n, d) = scaled(value.n, value.denominator, exponent: value.exponent - 2 * exponent)
178 var integer = integerSqrt(n / d)
179 let inexact = integer * integer * d != n
180 let midpoint = 2 * integer + 1
181 if shouldIncrement(mode, negative: false, integer: integer, inexact: inexact, halfway: 4 * n - d * midpoint * midpoint) {
182 integer += 1
183 }
184 return Units(integer: integer, inexact: inexact)
185}

187struct Fields {
188 let negative: Bool
189 let exponent: Int
190 let fraction: Int
191 let q: Int
192}

194struct Format {
195 let name: String
196 let e: Int
197 let m: Int
198 var w: Int { 1 + e + m }
199 var bias: Int { (1 << (e - 1)) - 1 }
200 var emin: Int { 1 - bias }
201 var emax: Int { bias }
202 var hidden: Int { 1 << m }
203 var exponentMask: Int { (1 << e) - 1 }
204 var signBit: Int { 1 << (e + m) }
205 var infinity: Int { exponentMask * hidden }
206 var nan: Int { infinity + hidden / 2 }
207 var maximum: Int { infinity - 1 }
208
209 func fields(_ pattern: Int) -> Fields {
210 let exponent = (pattern >> m) & exponentMask
211 return Fields(negative: pattern & signBit != 0, exponent: exponent,
212 fraction: pattern & (hidden - 1), q: (exponent == 0 ? emin : exponent - bias) - m)
213 }
214
215 func isNaN(_ pattern: Int) -> Bool {
216 let f = fields(pattern)
217 return f.exponent == exponentMask && f.fraction != 0
218 }
219 func isSignaling(_ pattern: Int) -> Bool { isNaN(pattern) && fields(pattern).fraction < hidden / 2 }
220 func isInfinity(_ pattern: Int) -> Bool { pattern & (signBit - 1) == infinity }
221 func isZero(_ pattern: Int) -> Bool { pattern & (signBit - 1) == 0 }
222 func isFinite(_ pattern: Int) -> Bool { fields(pattern).exponent != exponentMask }
223 func zero(_ negative: Bool) -> Int { negative ? signBit : 0 }
224 func inf(_ negative: Bool) -> Int { zero(negative) + infinity }
225 func flip(_ pattern: Int) -> Int { pattern ^ signBit }
226
227 func exact(_ pattern: Int) -> Exact {
228 let f = fields(pattern)
229 return Exact(Int64(f.exponent == 0 ? f.fraction : hidden + f.fraction), exponent: f.q, negative: f.negative)
230 }
231
232 func bits(_ pattern: Int) -> String {
233 let f = fields(pattern)
234 return "\(f.negative ? 1 : 0)|\(padded(f.exponent, radix: 2, width: e))|\(padded(f.fraction, radix: 2, width: m))"
235 }
236
237 func hex(_ pattern: Int) -> String { "0x" + padded(pattern, radix: 16, width: (w + 3) / 4) }
238
239 func value(_ pattern: Int) -> String {
240 if isNaN(pattern) { return "nan" }
241 if isInfinity(pattern) { return fields(pattern).negative ? "-inf" : "inf" }
242 return exact(pattern).decimal(keepZeroSign: true)
243 }
244
245 func patternText(_ pattern: Int) -> String { "\(bits(pattern)) \(hex(pattern)) \(value(pattern))" }
246 func resultText(_ result: Result) -> String { "\(patternText(result.pattern)) \(result.flags.text)" }
247
248 func overflow(negative: Bool, mode: Mode) -> Result {
249 let toInfinity = mode == .rne || mode == .rna || mode == .rup && !negative || mode == .rdn && negative
250 return Result(pattern: zero(negative) + (toInfinity ? infinity : maximum), flags: [.overflow, .inexact])
251 }
252
253 private func roundNonzero(negative: Bool, t: Int, mode: Mode, unitsAt: (Int) -> Units) -> Result {
254 if t > emax { return overflow(negative: negative, mode: mode) }
255 var exponent = max(t, emin) - m
256 let units = unitsAt(exponent)
257 var n = units.integer
258 if n == 2 * Int64(hidden) {
259 n = Int64(hidden)
260 exponent += 1
261 }
262 if exponent > emax - m { return overflow(negative: negative, mode: mode) }
263 let magnitude = n >= Int64(hidden) ? (exponent + m + bias) * hidden + Int(n) - hidden : Int(n)
264 var flags: Flags = units.inexact ? .inexact : []
265 if units.inexact && t < emin && (t < emin - 1 || unitsAt(t - m).integer < 2 * Int64(hidden)) {
266 flags.formUnion(.underflow)
267 }
268 return Result(pattern: zero(negative) + magnitude, flags: flags)
269 }
270
271 func round(_ value: Exact, mode: Mode) -> Result {
272 if value.n == 0 { return Result(pattern: zero(value.negative)) }
273 return roundNonzero(negative: value.negative, t: value.log2Floor(), mode: mode) {
274 roundedUnits(value, exponent: $0, mode: mode)
275 }
276 }
277
278 func sqrt(_ value: Exact, mode: Mode) -> Result {
279 let log = value.log2Floor()
280 let t = log >= 0 ? log / 2 : -((-log + 1) / 2)
281 return roundNonzero(negative: false, t: t, mode: mode) {
282 roundedSqrtUnits(value, exponent: $0, mode: mode)
283 }
284 }

299func parseReal(_ word: String) throws -> RealInput {
300 if word == "nan" { return .nan }
301 if word == "inf" || word == "-inf" { return .infinity(word == "-inf") }
302 if word.hasPrefix("0b") || word.hasPrefix("0x") { throw CommandError("BAD OPERAND \(word)") }
303 let negative = word.hasPrefix("-")
304 let body = negative ? String(word.dropFirst()) : word
305 if body.contains("/") {
306 let parts = body.split(separator: "/")
307 let n = Int64(parts[0])!, d = Int64(parts[1])!
308 if d == 0 { throw CommandError("BAD OPERAND \(word)") }
309 return .finite(Exact(n, denominator: d, negative: negative))
310 }
311 let parts = body.split(separator: ".")
312 let fraction = parts.count == 2 ? String(parts[1]) : ""
313 var denominator: Int64 = 1
314 for _ in fraction { denominator *= 10 }
315 let n = Int64(String(parts[0]) + fraction)!
316 return .finite(Exact(n, denominator: denominator, negative: negative))
317}

319func encoded(_ format: Format, _ input: RealInput, mode: Mode) -> Result {
320 switch input {
321 case .nan: return Result(pattern: format.nan)
322 case let .infinity(negative): return Result(pattern: format.inf(negative))
323 case let .finite(value): return format.round(value, mode: mode)
324 }
325}

327func operand(_ format: Format, _ word: String) throws -> Int {
328 if word.hasPrefix("0b") || word.hasPrefix("0x") {
329 let bits = word.hasPrefix("0b")
330 let width = bits ? format.w : (format.w + 3) / 4
331 if word.utf8.count - 2 != width { throw CommandError("BAD OPERAND \(word)") }
332 let pattern = Int(word.dropFirst(2), radix: bits ? 2 : 16)!
333 if pattern >= (1 << format.w) { throw CommandError("BAD OPERAND \(word)") }
334 return pattern
335 }
336 return encoded(format, try parseReal(word), mode: .rne).pattern
337}

339func operation(_ format: Format, _ requested: Arithmetic, _ arguments: [Int], mode: Mode) -> Result {
340 var op = requested, inputs = arguments
341 if op == .sub {
342 inputs[1] = format.flip(inputs[1])
343 op = .add
344 }
345 if inputs.contains(where: { format.isNaN($0) }) {
346 return Result(pattern: format.nan, flags: inputs.contains(where: { format.isSignaling($0) }) ? .invalid : [])
347 }
348 let a = inputs[0]
349 let negative = format.fields(a).negative
350 let invalid = Result(pattern: format.nan, flags: .invalid)
351 if op == .sqrt {
352 if format.isZero(a) { return Result(pattern: a) }
353 if negative { return invalid }
354 if format.isInfinity(a) { return Result(pattern: a) }
355 return format.sqrt(format.exact(a), mode: mode)
356 }
357 let b = inputs[1]
358 let sigma = negative != format.fields(b).negative
359 if op == .add {
360 if format.isInfinity(a) && format.isInfinity(b) && sigma { return invalid }
361 if format.isInfinity(a) { return Result(pattern: a) }
362 if format.isInfinity(b) { return Result(pattern: b) }
363 return format.round(exactSum([format.exact(a), format.exact(b)], mode: mode), mode: mode)
364 }
365 if op == .mul {
366 if format.isInfinity(a) || format.isInfinity(b) {
367 return format.isZero(a) || format.isZero(b) ? invalid : Result(pattern: format.inf(sigma))
368 }
369 if format.isZero(a) || format.isZero(b) { return Result(pattern: format.zero(sigma)) }
370 return format.round(exactProduct(format.exact(a), format.exact(b)), mode: mode)
371 }
372 if op == .div {
373 if format.isInfinity(a) { return format.isInfinity(b) ? invalid : Result(pattern: format.inf(sigma)) }
374 if format.isInfinity(b) { return Result(pattern: format.zero(sigma)) }
375 if format.isZero(b) { return format.isZero(a) ? invalid : Result(pattern: format.inf(sigma), flags: .divByZero) }
376 if format.isZero(a) { return Result(pattern: format.zero(sigma)) }
377 let x = format.exact(a), y = format.exact(b)
378 let quotient = Exact(x.n * y.denominator, exponent: x.exponent - y.exponent,
379 denominator: x.denominator * y.n, negative: sigma)
380 return format.round(quotient, mode: mode)
381 }
382 let c = inputs[2]
383 if format.isInfinity(a) || format.isInfinity(b) {
384 if format.isZero(a) || format.isZero(b) || format.isInfinity(c) && format.fields(c).negative != sigma { return invalid }
385 return Result(pattern: format.inf(sigma))
386 }
387 if format.isInfinity(c) { return Result(pattern: c) }
388 return format.round(exactSum([exactProduct(format.exact(a), format.exact(b)), format.exact(c)], mode: mode), mode: mode)
389}

391func nextUp(_ format: Format, _ pattern: Int) -> Int {
392 if pattern == format.infinity { return pattern }
393 if format.isZero(pattern) { return 1 }
394 return pattern + (format.fields(pattern).negative ? -1 : 1)

426 func summation(_ command: String, format: Format, mode: Mode, patterns: [Int]) throws {
427 guard patterns.allSatisfy({ format.isFinite($0) }) else { throw CommandError("NOT FINITE") }
428 emit("\(command) \(format.name) \(mode.rawValue) count=\(patterns.count)")
429 var sum = patterns[0], compensation = 0
430 var flags: Flags = []
431 for i in 1..<patterns.count {
432 if command == "SUM" {
433 let added = operation(format, .add, [sum, patterns[i]], mode: mode)
434 sum = added.pattern
435 flags.formUnion(added.flags)
436 emit("PARTIAL \(i + 1) \(format.resultText(added))")
437 } else {
438 let y = operation(format, .sub, [patterns[i], compensation], mode: mode)
439 let t = operation(format, .add, [sum, y.pattern], mode: mode)
440 let d = operation(format, .sub, [t.pattern, sum], mode: mode)
441 let c = operation(format, .sub, [d.pattern, y.pattern], mode: mode)
442 sum = t.pattern
443 compensation = c.pattern
444 let stepFlags = y.flags.union(t.flags).union(d.flags).union(c.flags)
445 flags.formUnion(stepFlags)
446 emit("KSTEP \(i + 1) sum=\(format.value(sum)) comp=\(format.value(compensation)) \(stepFlags.text)")
447 }
448 }
449 summary(format, mode: mode, patterns: patterns, result: Result(pattern: sum, flags: flags))
450 }

452 func dot(_ format: Format, mode: Mode, a: [Int], b: [Int]) throws {
453 guard a.count == b.count else { throw CommandError("LENGTH MISMATCH") }
454 guard (a + b).allSatisfy({ format.isFinite($0) }) else { throw CommandError("NOT FINITE") }
455 var naive = operation(format, .mul, [a[0], b[0]], mode: mode)
456 var chained = naive
457 for i in 1..<a.count {
458 let product = operation(format, .mul, [a[i], b[i]], mode: mode)
459 let added = operation(format, .add, [naive.pattern, product.pattern], mode: mode)
460 naive = Result(pattern: added.pattern, flags: naive.flags.union(product.flags).union(added.flags))
461 let fused = operation(format, .fma, [a[i], b[i], chained.pattern], mode: mode)
462 chained = Result(pattern: fused.pattern, flags: chained.flags.union(fused.flags))
463 }
464 let terms = a.indices.map { exactProduct(format.exact(a[$0]), format.exact(b[$0])) }
465 let exact = exactSum(terms, mode: mode)
466 emit("DOT \(format.name) \(mode.rawValue) n=\(a.count)")
467 emit("NAIVE \(format.resultText(naive))")
468 emit("CHAINED \(format.resultText(chained))")
469 emit("EXACT \(exact.decimal())")
470 emit("FUSED \(format.resultText(format.round(exact, mode: mode)))")
471 emit("DIFF naive=\(difference(format, naive.pattern, exact)) chained=\(difference(format, chained.pattern, exact))")
472 }

474 func table(_ format: Format) throws {
475 if format.w > 8 { throw CommandError("TABLE TOO LARGE") }
476 let count = format.infinity
477 emit("TABLE \(format.name) count=\(count)")
478 for pattern in 0..<count {
479 let gap = pattern == count - 1 ? "-"
480 : exactSum([format.exact(pattern + 1), format.exact(pattern).negated()], mode: .rne).decimal()
481 emit("ENTRY \(pattern) \(format.patternText(pattern)) gap=\(gap)")
482 }
483 emit("MAX \(format.patternText(format.maximum))")
484 emit("MINNORMAL \(format.patternText(format.hidden))")
485 emit("MINSUBNORMAL \(format.patternText(1))")
486 emit("EPSILON \(Exact(1, exponent: -format.m).decimal())")
487 emit("PATTERNS finite=\(2 * count) infinite=2 nan=\(2 * (format.hidden - 1)) total=\(1 << format.w)")
488 }

490 func command(_ line: String) throws {
491 let words = line.split(separator: " ").map(String.init)
492 let command = words[0]
493 if command == "FORMAT" {
494 let name = words[1], e = Int(words[2])!, m = Int(words[3])!
495 if formats[name] != nil { throw CommandError("FORMAT EXISTS \(name)") }
496 guard (2...5).contains(e), (1...10).contains(m) else { throw CommandError("BAD FORMAT") }
497 let format = Format(name: name, e: e, m: m)
498 formats[name] = format
499 emit("FORMAT \(name) e=\(e) m=\(m) bits=\(format.w) bias=\(format.bias) emin=\(format.emin) emax=\(format.emax)")
500 return
501 }
502 let f = try format(words[1])
503 if command == "ENCODE" {
504 let value = try parseReal(words[2])
505 let mode = try parseMode(words, at: 3)
506 emit("ENCODE \(f.name) \(mode.rawValue) \(value.text) \(f.resultText(encoded(f, value, mode: mode)))")
507 } else if command == "CONVERT" {
508 let g = try format(words[2])
509 let a = try operand(f, words[3])
510 let mode = try parseMode(words, at: 4)
511 let result: Result
512 if f.isNaN(a) { result = Result(pattern: g.nan, flags: f.isSignaling(a) ? .invalid : []) }
513 else if f.isInfinity(a) { result = Result(pattern: g.inf(f.fields(a).negative)) }
514 else { result = g.round(f.exact(a), mode: mode) }
515 emit("CONVERT \(f.name) \(g.name) \(mode.rawValue) \(g.resultText(result))")
516 } else if command == "SUM" || command == "KAHAN" || command == "DOT" {
517 let mode = try parseMode(words, at: 2)
518 if command == "DOT" {
519 let separator = words.firstIndex(of: "|")!
520 let a = try words[3..<separator].map { try operand(f, $0) }
521 let b = try words[(separator + 1)...].map { try operand(f, $0) }
522 try dot(f, mode: mode, a: a, b: b)
523 } else {
524 let patterns = try words[3...].map { try operand(f, $0) }
525 try summation(command, format: f, mode: mode, patterns: patterns)
526 }
527 } else if command == "TABLE" { try table(f) }
528 else if let op = Arithmetic(rawValue: command) {
529 let patterns = try words[2..<(2 + op.arity)].map { try operand(f, $0) }
530 let mode = try parseMode(words, at: 2 + op.arity)
531 emit("\(command) \(f.name) \(mode.rawValue) \(f.resultText(operation(f, op, patterns, mode: mode)))")
532 } else {
533 let a = try operand(f, words[2])
534 let fields = f.fields(a)
535 if command == "DECODE" {
536 let classification = f.isNaN(a) ? (f.isSignaling(a) ? "snan" : "qnan")
537 : f.isInfinity(a) ? "infinity" : f.isZero(a) ? "zero" : fields.exponent == 0 ? "subnormal" : "normal"
538 emit("DECODE \(f.name) \(f.bits(a)) \(f.hex(a)) \(classification) sign=\(fields.negative ? 1 : 0)")
539 var line = "FIELDS exponent=\(fields.exponent) fraction=\(fields.fraction)"
540 if f.isFinite(a) {
541 line += " unbiased=\(fields.q + f.m) significand=\(fields.exponent == 0 ? 0 : 1).\(padded(fields.fraction, radix: 2, width: f.m))"
542 emit(line)
543 emit("VALUE \(f.exact(a).fraction()) \(f.value(a))")
544 } else {
545 if f.isNaN(a) { line += " payload=\(fields.fraction % (f.hidden / 2))" }
546 emit(line)
547 }
548 } else if command == "COMPARE" {
549 let b = try operand(f, words[3])
550 func key(_ pattern: Int) -> Int { f.fields(pattern).negative ? -(pattern - f.signBit) - 1 : pattern }
551 let order = relation(key(a), key(b))
552 let numeric = f.isNaN(a) || f.isNaN(b) ? "UN" : f.isZero(a) && f.isZero(b) ? "EQ" : order
553 let flags: Flags = f.isSignaling(a) || f.isSignaling(b) ? .invalid : []
554 emit("COMPARE \(f.name) \(f.hex(a)) \(f.hex(b)) \(numeric) total=\(order) \(flags.text)")
555 } else if command == "TOINT" {
556 guard let range = integerTypes[words[3]] else { throw CommandError("BAD TYPE \(words[3])") }
557 let mode = try parseMode(words, at: 4)
558 let prefix = "TOINT \(f.name) \(words[3]) \(mode.rawValue)"
559 if !f.isFinite(a) { emit("\(prefix) invalid flags=invalid") }
560 else {
561 let units = roundedUnits(f.exact(a), exponent: 0, mode: mode)
562 let integer = fields.negative ? -units.integer : units.integer
563 if !range.contains(integer) { emit("\(prefix) invalid flags=invalid") }
564 else {
565 let flags: Flags = units.inexact ? .inexact : []
566 emit("\(prefix) \(integer) \(flags.text)")
567 }
568 }
569 } else {
570 let result: Result
571 if f.isNaN(a) { result = Result(pattern: f.nan, flags: f.isSignaling(a) ? .invalid : []) }
572 else if command == "NEXTUP" { result = Result(pattern: nextUp(f, a)) }
573 else if command == "NEXTDOWN" { result = Result(pattern: f.flip(nextUp(f, f.flip(a)))) }
574 else {
575 let pattern = f.isInfinity(a) ? f.infinity : f.round(Exact(1, exponent: fields.q), mode: .rne).pattern
576 result = Result(pattern: pattern)
577 }
578 emit("\(command) \(f.name) \(f.resultText(result))")
579 }
580 }
581 }
582}
