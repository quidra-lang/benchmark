6type Pattern = u32;
7type Flags = u8;
8const INVALID: Flags = 1;
9const DIVBYZERO: Flags = 2;
10const OVERFLOW: Flags = 4;
11const UNDERFLOW: Flags = 8;
12const INEXACT: Flags = 16;

14#[derive(Clone, Copy, PartialEq)]
15enum Mode { Rne, Rna, Rtz, Rup, Rdn }
16
17impl Mode {
18 fn parse(word: &str) -> Result<Self, String> {
19 match word {
20 "RNE" => Ok(Self::Rne), "RNA" => Ok(Self::Rna), "RTZ" => Ok(Self::Rtz),
21 "RUP" => Ok(Self::Rup), "RDN" => Ok(Self::Rdn),
22 _ => Err(format!("ERROR BAD MODE {word}")),
23 }
24 }
25
26 fn optional(word: Option<&&str>) -> Result<Self, String> {
27 word.map_or(Ok(Self::Rne), |word| Self::parse(word))
28 }
29}
30
31impl fmt::Display for Mode {
32 fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
33 f.write_str(match self { Self::Rne => "RNE", Self::Rna => "RNA", Self::Rtz => "RTZ", Self::Rup => "RUP", Self::Rdn => "RDN" })
34 }
35}

37#[derive(Clone, Copy, PartialEq)]
38enum Class { Zero, Subnormal, Normal, Infinity, QuietNan, SignalingNan }

40impl fmt::Display for Class {

42 f.write_str(match self {
43 Self::Zero => "zero", Self::Subnormal => "subnormal", Self::Normal => "normal",
44 Self::Infinity => "infinity", Self::QuietNan => "qnan", Self::SignalingNan => "snan",
45 })
46 }

49#[derive(Clone, Copy, PartialEq)]
50enum Operation { Add, Sub, Mul, Div, Sqrt, Fma }

52#[derive(Clone, Copy)]
53struct Term { negative: bool, magnitude: u64, exponent: i32 }
54
55impl Term {
56 fn product(self, other: Self) -> Self {
57 Self { negative: self.negative != other.negative, magnitude: self.magnitude * other.magnitude, exponent: self.exponent + other.exponent }
58 }
59
60 fn exact(self) -> Exact { Exact::new(self.negative, self.magnitude, self.exponent, 1) }
61
62 fn decimal(self) -> String {
63 if self.magnitude == 0 { return "0".into(); }
64 let value = self.exact();
65 let (numerator, denominator) = value.scaled(0);
66 let sign = if value.negative { "-" } else { "" };
67 let mut output = format!("{sign}{}", numerator / denominator);
68 let mut remainder = numerator % denominator;
69 if remainder != 0 { output.push('.'); }
70 while remainder != 0 {
71 remainder *= 10;
72 output.push(char::from(b'0' + (remainder / denominator) as u8));
73 remainder %= denominator;
74 }
75 output
76 }
77}
78
79fn sum_terms(terms: &[Term], mode: Mode) -> Term {
80 let exponent = terms.iter().filter(|term| term.magnitude != 0).map(|term| term.exponent).min().unwrap_or(0);
81 let mut sum = 0i64;
82 for term in terms.iter().filter(|term| term.magnitude != 0) {
83 let magnitude = (term.magnitude << (term.exponent - exponent)) as i64;
84 sum += if term.negative { -magnitude } else { magnitude };
85 }
86 let negative = if sum == 0 {
87 if terms.iter().all(|term| term.magnitude == 0 && term.negative == terms[0].negative) { terms[0].negative }
88 else { mode == Mode::Rdn }
89 } else { sum < 0 };
90 Term { negative, magnitude: sum.unsigned_abs(), exponent }
91}

93#[derive(Clone, Copy)]
94struct Exact { negative: bool, numerator: u64, denominator: u64, exponent: i32 }
95
96impl Exact {
97 fn new(negative: bool, mut numerator: u64, mut exponent: i32, mut denominator: u64) -> Self {
98 if numerator == 0 { return Self { negative, numerator: 0, denominator: 1, exponent: 0 }; }
99 let (mut a, mut b) = (numerator, denominator);
100 while b != 0 { (a, b) = (b, a % b); }
101 numerator /= a;
102 denominator /= a;
103 let shift = numerator.trailing_zeros();
104 numerator >>= shift;
105 exponent += shift as i32;
106 let shift = denominator.trailing_zeros();
107 denominator >>= shift;
108 exponent -= shift as i32;
109 Self { negative, numerator, denominator, exponent }
110 }
111
112 fn scaled(self, quantum: i32) -> (u64, u64) {
113 let shift = self.exponent - quantum;
114 if shift >= 0 { (self.numerator << shift, self.denominator) }
115 else { (self.numerator, self.denominator << -shift) }
116 }
117
118 fn floor_log(self) -> i32 {
119 let power = self.denominator.leading_zeros() as i32 - self.numerator.leading_zeros() as i32;
120 let below = if power >= 0 { self.numerator < self.denominator << power }
121 else { self.numerator << -power < self.denominator };
122 self.exponent + power - i32::from(below)
123 }
124
125 fn fraction(self) -> String {
126 if self.numerator == 0 { return "0".into(); }
127 let (numerator, denominator) = self.scaled(0);
128 let sign = if self.negative { "-" } else { "" };
129 if denominator == 1 { format!("{sign}{numerator}") }
130 else { format!("{sign}{numerator}/{denominator}") }
131 }
132}

134#[derive(Clone, Copy)]
135struct Quantized { whole: u64, inexact: bool, half: Ordering }
136
137impl Quantized {
138 fn new(value: Exact, quantum: i32, square_root: bool) -> Self {
139 let (numerator, denominator) = value.scaled(if square_root { 2 * quantum } else { quantum });
140 if !square_root {
141 let remainder = numerator % denominator;
142 return Self { whole: numerator / denominator, inexact: remainder != 0, half: remainder.cmp(&(denominator - remainder)) };
143 }
144 let whole = (numerator / denominator).isqrt();
145 let midpoint = 2 * whole + 1;
146 Self { whole, inexact: whole * whole * denominator != numerator, half: (4 * numerator).cmp(&(midpoint * midpoint * denominator)) }
147 }
148
149 fn increment(self, mode: Mode, negative: bool) -> u64 {
150 u64::from(match mode {
151 Mode::Rne => self.half == Ordering::Greater || (self.half == Ordering::Equal && self.whole & 1 != 0),
152 Mode::Rna => self.half != Ordering::Less,
153 Mode::Rtz => false,
154 Mode::Rup => self.inexact && !negative,
155 Mode::Rdn => self.inexact && negative,
156 })
157 }
158}

162
163impl Rounded {
164 fn plain(pattern: Pattern) -> Self { Self { pattern, flags: 0 } }

167enum Input { Number(Exact), Pattern(Pattern) }

169struct Format {
170 name: String,
171 e: i32, m: i32, w: i32, bias: i32, emin: i32, emax: i32,
172 sign: Pattern, base: Pattern, infinity: Pattern, nan: Pattern,
173}
174
175impl Format {
176 fn new(name: &str, e: i32, m: i32) -> Self {
177 let bias = (1 << (e - 1)) - 1;
178 let base = 1 << m;
179 let infinity = ((1 << e) - 1) << m;
180 Self { name: name.into(), e, m, w: 1 + e + m, bias, emin: 1 - bias, emax: bias,
181 sign: 1 << (e + m), base, infinity, nan: infinity | (base / 2) }
182 }
183
184 fn negative(&self, p: Pattern) -> bool { p & self.sign != 0 }
185 fn signed(&self, p: Pattern, negative: bool) -> Pattern { p | if negative { self.sign } else { 0 } }
186 fn zero(&self, p: Pattern) -> bool { p & (self.sign - 1) == 0 }
187 fn class(&self, p: Pattern) -> Class {
188 let magnitude = p & (self.sign - 1);
189 let fraction = p & (self.base - 1);
190 if magnitude >= self.infinity {
191 if fraction == 0 { Class::Infinity }
192 else if fraction >= self.base / 2 { Class::QuietNan } else { Class::SignalingNan }
193 } else if magnitude == 0 { Class::Zero }
194 else if magnitude < self.base { Class::Subnormal } else { Class::Normal }
195 }
196
197 fn finite(&self, p: Pattern) -> bool { matches!(self.class(p), Class::Zero | Class::Subnormal | Class::Normal) }
198 fn is_nan(&self, p: Pattern) -> bool { matches!(self.class(p), Class::QuietNan | Class::SignalingNan) }
199 fn term(&self, p: Pattern) -> Term {
200 let field = (p & (self.sign - 1)) >> self.m;
201 let fraction = p & (self.base - 1);
202 Term { negative: self.negative(p), magnitude: u64::from(if field == 0 { fraction } else { self.base + fraction }),
203 exponent: if field == 0 { self.emin - self.m } else { field as i32 - self.bias - self.m } }
204 }
205
206 fn overflow(&self, mode: Mode, negative: bool) -> Rounded {
207 let infinity = matches!(mode, Mode::Rne | Mode::Rna) || (mode == Mode::Rup && !negative) || (mode == Mode::Rdn && negative);
208 Rounded { pattern: self.signed(if infinity { self.infinity } else { self.infinity - 1 }, negative), flags: OVERFLOW | INEXACT }
209 }
210
211 fn round(&self, value: Exact, mode: Mode, square_root: bool) -> Rounded {
212 if value.numerator == 0 { return Rounded::plain(self.signed(0, value.negative)); }
213 let mut t = value.floor_log();
214 if square_root { t = t.div_euclid(2); }
215 if t > self.emax { return self.overflow(mode, value.negative); }
216 let mut quantum = t.max(self.emin) - self.m;
217 let divided = Quantized::new(value, quantum, square_root);
218 let mut significand = divided.whole + divided.increment(mode, value.negative);
219 let base = u64::from(self.base);
220 if significand == 2 * base { significand = base; quantum += 1; }
221 if quantum > self.emax - self.m { return self.overflow(mode, value.negative); }
222 let mut flags = if divided.inexact { INEXACT } else { 0 };
223 if divided.inexact && t < self.emin {
224 let tiny = t < self.emin - 1 || {
225 let unbounded = Quantized::new(value, t - self.m, square_root);
226 unbounded.whole + unbounded.increment(mode, value.negative) < 2 * base
227 };
228 if tiny { flags |= UNDERFLOW; }
229 }
230 let magnitude = if significand < base { significand as Pattern }
231 else { ((quantum + self.m + self.bias) as Pattern) << self.m | (significand - base) as Pattern };
232 Rounded { pattern: self.signed(magnitude, value.negative), flags }
233 }
234
235 fn round_term(&self, term: Term, mode: Mode) -> Rounded { self.round(term.exact(), mode, false) }
236 fn round_sum(&self, terms: &[Term], mode: Mode) -> Rounded { self.round_term(sum_terms(terms, mode), mode) }
237
238 fn arithmetic(&self, mut operation: Operation, a: Pattern, mut b: Pattern, c: Pattern, mode: Mode) -> Rounded {
239 if operation == Operation::Sub { b ^= self.sign; operation = Operation::Add; }
240 let inputs = [a, b, c];
241 let count = match operation { Operation::Sqrt => 1, Operation::Fma => 3, _ => 2 };
242 if inputs[..count].iter().any(|&p| self.is_nan(p)) {
243 let flags = if inputs[..count].iter().any(|&p| self.class(p) == Class::SignalingNan) { INVALID } else { 0 };
244 return Rounded { pattern: self.nan, flags };
245 }
246 let ai = self.class(a) == Class::Infinity;
247 let bi = self.class(b) == Class::Infinity;
248 let negative = self.negative(a) != self.negative(b);
249 let invalid = Rounded { pattern: self.nan, flags: INVALID };
250 match operation {
251 Operation::Add => {
252 if ai && bi && negative { invalid }
253 else if ai || bi { Rounded::plain(if ai { a } else { b }) }
254 else { self.round_sum(&[self.term(a), self.term(b)], mode) }
255 }
256 Operation::Mul => {
257 if ai || bi {
258 if self.zero(a) || self.zero(b) { invalid } else { Rounded::plain(self.signed(self.infinity, negative)) }
259 } else { self.round_term(self.term(a).product(self.term(b)), mode) }
260 }
261 Operation::Div => {
262 if ai { if bi { invalid } else { Rounded::plain(self.signed(self.infinity, negative)) } }
263 else if bi { Rounded::plain(self.signed(0, negative)) }
264 else if self.zero(b) {
265 if self.zero(a) { invalid } else { Rounded { pattern: self.signed(self.infinity, negative), flags: DIVBYZERO } }
266 } else if self.zero(a) { Rounded::plain(self.signed(0, negative)) }
267 else {
268 let (x, y) = (self.term(a), self.term(b));
269 self.round(Exact::new(negative, x.magnitude, x.exponent - y.exponent, y.magnitude), mode, false)
270 }
271 }
272 Operation::Sqrt => {
273 if self.zero(a) { Rounded::plain(a) }
274 else if self.negative(a) { invalid }
275 else if ai { Rounded::plain(a) }
276 else { self.round(self.term(a).exact(), mode, true) }
277 }
278 Operation::Fma => {
279 if ai || bi {
280 if self.zero(a) || self.zero(b) || (self.class(c) == Class::Infinity && self.negative(c) != negative) { invalid }
281 else { Rounded::plain(self.signed(self.infinity, negative)) }
282 } else if self.class(c) == Class::Infinity { Rounded::plain(c) }
283 else { self.round_sum(&[self.term(a).product(self.term(b)), self.term(c)], mode) }
284 }
285 Operation::Sub => unreachable!(),
286 }
287 }

289 fn raw_operand(&self, word: &str, encoding: bool) -> Result<Input, String> {
290 let invalid = || format!("ERROR BAD OPERAND {word}");
291 if word.starts_with("0b") || word.starts_with("0x") {
292 let binary = word.starts_with("0b");
293 let width = if binary { self.w } else { (self.w + 3) / 4 } as usize;
294 if encoding || word.len() - 2 != width { return Err(invalid()); }
295 let pattern = Pattern::from_str_radix(&word[2..], if binary { 2 } else { 16 }).unwrap();
296 return if pattern < 2 * self.sign { Ok(Input::Pattern(pattern)) } else { Err(invalid()) };
297 }
298 match word {
299 "nan" => return Ok(Input::Pattern(self.nan)),
300 "inf" => return Ok(Input::Pattern(self.infinity)),
301 "-inf" => return Ok(Input::Pattern(self.infinity | self.sign)),
302 _ => {}
303 }
304 let negative = word.starts_with('-');
305 let digits = word.strip_prefix('-').unwrap_or(word);
306 let (numerator, denominator) = if let Some((p, q)) = digits.split_once('/') {
307 (p.parse().unwrap(), q.parse().unwrap())
308 } else if let Some((integer, fraction)) = digits.split_once('.') {
309 (format!("{integer}{fraction}").parse().unwrap(), 10u64.pow(fraction.len() as u32))
310 } else { (digits.parse().unwrap(), 1) };
311 if denominator == 0 { return Err(invalid()); }
312 Ok(Input::Number(Exact::new(negative, numerator, 0, denominator)))
313 }
314
315 fn operand(&self, word: &str) -> Result<Pattern, String> {
316 Ok(match self.raw_operand(word, false)? {
317 Input::Pattern(pattern) => pattern,
318 Input::Number(value) => self.round(value, Mode::Rne, false).pattern,
319 })
320 }

322 fn bits(&self, p: Pattern) -> String {
323 format!("{}|{:0ew$b}|{:0mw$b}", u8::from(self.negative(p)), (p & (self.sign - 1)) >> self.m,
324 p & (self.base - 1), ew = self.e as usize, mw = self.m as usize)
325 }
326
327 fn hex(&self, p: Pattern) -> String { format!("0x{p:0width$X}", width = ((self.w + 3) / 4) as usize) }
328 fn value(&self, p: Pattern) -> String {
329 match self.class(p) {
330 Class::QuietNan | Class::SignalingNan => "nan".into(),
331 Class::Infinity => if self.negative(p) { "-inf" } else { "inf" }.into(),
332 Class::Zero => if self.negative(p) { "-0" } else { "0" }.into(),
333 _ => self.term(p).decimal(),
334 }
335 }
336
337 fn pattern(&self, p: Pattern) -> String { format!("{} {} {}", self.bits(p), self.hex(p), self.value(p)) }
338 fn result(&self, result: Rounded) -> String { format!("{} {}", self.pattern(result.pattern), flag_text(result.flags)) }
339 fn difference(&self, p: Pattern, mut exact_sum: Term) -> String {
340 if !self.finite(p) { return self.value(p); }
341 exact_sum.negative = !exact_sum.negative;
342 sum_terms(&[self.term(p), exact_sum], Mode::Rne).exact().fraction()
343 }
344
345 fn order_key(&self, p: Pattern) -> i64 { if self.negative(p) { -i64::from(p - self.sign) - 1 } else { i64::from(p) } }
346 fn next_up(&self, p: Pattern) -> Pattern {
347 if p == self.infinity { p } else if self.zero(p) { 1 }
348 else if self.negative(p) { p - 1 } else { p + 1 }
349 }
350}

352fn flag_text(flags: Flags) -> String {
353 let names = ["invalid", "divbyzero", "overflow", "underflow", "inexact"];
354 let set: Vec<_> = names.into_iter().enumerate().filter_map(|(i, name)| (flags & (1 << i) != 0).then_some(name)).collect();
355 format!("flags={}", if set.is_empty() { "none".into() } else { set.join(",") })
356}

358fn relation(a: i64, b: i64) -> &'static str { match a.cmp(&b) { Ordering::Less => "LT", Ordering::Equal => "EQ", Ordering::Greater => "GT" } }

360fn integer_range(word: &str) -> Result<(i64, i64), String> {
361 match word {
362 "I8" => Ok((-128, 127)), "U8" => Ok((0, 255)), "I16" => Ok((-32768, 32767)),

365 }

368fn summation(f: &Format, words: &[&str]) -> Result<String, String> {
369 let command = words[0];
370 let mode = Mode::parse(words[2])?;
371 let mut values = Vec::new();
372 let mut left_count = 0;
373 for &word in &words[3..] {
374 if word == "|" { left_count = values.len(); } else { values.push(f.operand(word)?); }
375 }
376 if command == "DOT" && left_count != values.len() - left_count { return Err("ERROR LENGTH MISMATCH".into()); }
377 if values.iter().any(|&p| !f.finite(p)) { return Err("ERROR NOT FINITE".into()); }
378 let mut output = String::new();
379 if command == "DOT" {
380 let terms: Vec<_> = (0..left_count).map(|i| f.term(values[i]).product(f.term(values[left_count + i]))).collect();
381 let exact_sum = sum_terms(&terms, mode);
382 let first = f.arithmetic(Operation::Mul, values[0], values[left_count], 0, mode);
383 let (mut naive, mut chained) = (first, first);
384 for i in 1..left_count {
385 let multiplied = f.arithmetic(Operation::Mul, values[i], values[left_count + i], 0, mode);
386 let added = f.arithmetic(Operation::Add, naive.pattern, multiplied.pattern, 0, mode);
387 naive = Rounded { pattern: added.pattern, flags: naive.flags | multiplied.flags | added.flags };
388 }
389 for i in 1..left_count {
390 let fused = f.arithmetic(Operation::Fma, values[i], values[left_count + i], chained.pattern, mode);
391 chained = Rounded { pattern: fused.pattern, flags: chained.flags | fused.flags };
392 }
393 writeln!(output, "DOT {} {mode} n={left_count}", f.name).unwrap();
394 writeln!(output, "NAIVE {}", f.result(naive)).unwrap();
395 writeln!(output, "CHAINED {}", f.result(chained)).unwrap();
396 writeln!(output, "EXACT {}", exact_sum.decimal()).unwrap();
397 writeln!(output, "FUSED {}", f.result(f.round_term(exact_sum, mode))).unwrap();
398 writeln!(output, "DIFF naive={} chained={}", f.difference(naive.pattern, exact_sum), f.difference(chained.pattern, exact_sum)).unwrap();
399 return Ok(output);
400 }
401 let terms: Vec<_> = values.iter().map(|&p| f.term(p)).collect();
402 let exact_sum = sum_terms(&terms, mode);
403 let mut result = Rounded::plain(values[0]);
404 let mut compensation = 0;
405 writeln!(output, "{command} {} {mode} count={}", f.name, values.len()).unwrap();
406 for (i, &value) in values.iter().enumerate().skip(1) {
407 if command == "SUM" {
408 let next = f.arithmetic(Operation::Add, result.pattern, value, 0, mode);
409 result = Rounded { pattern: next.pattern, flags: result.flags | next.flags };
410 writeln!(output, "PARTIAL {} {}", i + 1, f.result(next)).unwrap();
411 } else {
412 let y = f.arithmetic(Operation::Sub, value, compensation, 0, mode);
413 let t = f.arithmetic(Operation::Add, result.pattern, y.pattern, 0, mode);
414 let d = f.arithmetic(Operation::Sub, t.pattern, result.pattern, 0, mode);
415 let c = f.arithmetic(Operation::Sub, d.pattern, y.pattern, 0, mode);
416 let flags = y.flags | t.flags | d.flags | c.flags;
417 result = Rounded { pattern: t.pattern, flags: result.flags | flags };
418 compensation = c.pattern;
419 writeln!(output, "KSTEP {} sum={} comp={} {}", i + 1, f.value(result.pattern), f.value(compensation), flag_text(flags)).unwrap();
420 }
421 }
422 writeln!(output, "RESULT {}", f.result(result)).unwrap();
423 writeln!(output, "EXACT {}", exact_sum.decimal()).unwrap();
424 writeln!(output, "ROUNDED {}", f.result(f.round_term(exact_sum, mode))).unwrap();
425 writeln!(output, "DIFF {}", f.difference(result.pattern, exact_sum)).unwrap();
426 Ok(output)
427}

429fn table(f: &Format) -> Result<String, String> {
430 if f.w > 8 { return Err("ERROR TABLE TOO LARGE".into()); }
431 let mut output = format!("TABLE {} count={}\n", f.name, f.infinity);
432 for p in 0..f.infinity {
433 let gap = if p + 1 == f.infinity { "-".into() } else {
434 let mut previous = f.term(p);
435 previous.negative = true;
436 sum_terms(&[f.term(p + 1), previous], Mode::Rne).decimal()
437 };
438 writeln!(output, "ENTRY {p} {} gap={gap}", f.pattern(p)).unwrap();
439 }
440 writeln!(output, "MAX {}", f.pattern(f.infinity - 1)).unwrap();
441 writeln!(output, "MINNORMAL {}", f.pattern(f.base)).unwrap();
442 writeln!(output, "MINSUBNORMAL {}", f.pattern(1)).unwrap();
443 writeln!(output, "EPSILON {}", Term { negative: false, magnitude: 1, exponent: -f.m }.decimal()).unwrap();
444 writeln!(output, "PATTERNS finite={} infinite=2 nan={} total={}", 2 * f.infinity, 2 * (f.base - 1), 2 * f.sign).unwrap();
445 Ok(output)
446}

448fn find_format<'a>(formats: &'a BTreeMap<String, Format>, name: &str) -> Result<&'a Format, String> {

452fn execute(formats: &mut BTreeMap<String, Format>, words: &[&str]) -> Result<String, String> {
453 let command = words[0];
454 if command == "FORMAT" {
455 if formats.contains_key(words[1]) { return Err(format!("ERROR FORMAT EXISTS {}", words[1])); }
456 let e = words[2].parse().unwrap();
457 let m = words[3].parse().unwrap();
458 if !(2..=5).contains(&e) || !(1..=10).contains(&m) { return Err("ERROR BAD FORMAT".into()); }
459 let f = Format::new(words[1], e, m);
460 let output = format!("FORMAT {} e={e} m={m} bits={} bias={} emin={} emax={}\n", f.name, f.w, f.bias, f.emin, f.emax);
461 formats.insert(f.name.clone(), f);
462 return Ok(output);
463 }
464 let f = find_format(formats, words[1])?;
465 if command == "TABLE" { return table(f); }
466 if matches!(command, "SUM" | "KAHAN" | "DOT") { return summation(f, words); }
467 if command == "CONVERT" {
468 let g = find_format(formats, words[2])?;
469 let a = f.operand(words[3])?;
470 let mode = Mode::optional(words.get(4))?;
471 let result = match f.class(a) {
472 Class::QuietNan => Rounded::plain(g.nan),
473 Class::SignalingNan => Rounded { pattern: g.nan, flags: INVALID },
474 Class::Infinity => Rounded::plain(g.signed(g.infinity, f.negative(a))),
475 Class::Zero => Rounded::plain(g.signed(0, f.negative(a))),
476 _ => g.round_term(f.term(a), mode),
477 };
478 return Ok(format!("CONVERT {} {} {mode} {}\n", f.name, g.name, g.result(result)));
479 }
480 if command == "ENCODE" {
481 let value = f.raw_operand(words[2], true)?;
482 let mode = Mode::optional(words.get(3))?;
483 let (source, result) = match value {
484 Input::Number(value) => (value.fraction(), f.round(value, mode, false)),
485 Input::Pattern(p) => (f.value(p), Rounded::plain(p)),
486 };
487 return Ok(format!("ENCODE {} {mode} {source} {}\n", f.name, f.result(result)));
488 }
489 let a = f.operand(words[2])?;
490 match command {
491 "DECODE" => {
492 let kind = f.class(a);
493 let exponent = (a & (f.sign - 1)) >> f.m;
494 let fraction = a & (f.base - 1);
495 let mut output = format!("DECODE {} {} {} {kind} sign={}\n", f.name, f.bits(a), f.hex(a), u8::from(f.negative(a)));
496 if f.finite(a) {
497 let term = f.term(a);
498 writeln!(output, "FIELDS exponent={exponent} fraction={fraction} unbiased={} significand={}.{:0width$b}",
499 term.exponent + f.m, u8::from(kind == Class::Normal), fraction, width = f.m as usize).unwrap();
500 writeln!(output, "VALUE {} {}", term.exact().fraction(), f.value(a)).unwrap();
501 } else if kind == Class::Infinity {
502 writeln!(output, "FIELDS exponent={exponent} fraction={fraction}").unwrap();
503 } else {
504 writeln!(output, "FIELDS exponent={exponent} fraction={fraction} payload={}", fraction % (f.base / 2)).unwrap();
505 }
506 Ok(output)
507 }
508 "COMPARE" => {
509 let b = f.operand(words[3])?;
510 let (ak, bk) = (f.order_key(a), f.order_key(b));
511 let numeric = if f.is_nan(a) || f.is_nan(b) { "UN" }
512 else if f.zero(a) && f.zero(b) { "EQ" } else { relation(ak, bk) };
513 let flags = if f.class(a) == Class::SignalingNan || f.class(b) == Class::SignalingNan { INVALID } else { 0 };
514 Ok(format!("COMPARE {} {} {} {numeric} total={} {}\n", f.name, f.hex(a), f.hex(b), relation(ak, bk), flag_text(flags)))
515 }
516 "TOINT" => {
517 let (minimum, maximum) = integer_range(words[3])?;
518 let mode = Mode::optional(words.get(4))?;
519 let prefix = format!("TOINT {} {} {mode}", f.name, words[3]);
520 if !f.finite(a) { return Ok(format!("{prefix} invalid flags=invalid\n")); }
521 let value = f.term(a).exact();
522 let divided = Quantized::new(value, 0, false);
523 let magnitude = (divided.whole + divided.increment(mode, value.negative)) as i64;
524 let result = if value.negative { -magnitude } else { magnitude };
525 if result < minimum || result > maximum { Ok(format!("{prefix} invalid flags=invalid\n")) }
526 else { Ok(format!("{prefix} {result} {}\n", flag_text(if divided.inexact { INEXACT } else { 0 }))) }
527 }
528 "NEXTUP" | "NEXTDOWN" | "ULP" => {
529 let result = if f.is_nan(a) {
530 Rounded { pattern: f.nan, flags: if f.class(a) == Class::SignalingNan { INVALID } else { 0 } }
531 } else {
532 let pattern = match command {
533 "NEXTUP" => f.next_up(a),
534 "NEXTDOWN" => f.next_up(a ^ f.sign) ^ f.sign,
535 _ => if f.class(a) == Class::Infinity { f.infinity }
536 else { f.round_term(Term { negative: false, magnitude: 1, exponent: f.term(a).exponent }, Mode::Rne).pattern },
537 };
538 Rounded::plain(pattern)
539 };
540 Ok(format!("{command} {} {}\n", f.name, f.result(result)))
541 }
542 _ => {
543 let operation = match command {
544 "ADD" => Operation::Add, "SUB" => Operation::Sub, "MUL" => Operation::Mul,
545 "DIV" => Operation::Div, "SQRT" => Operation::Sqrt, "FMA" => Operation::Fma,
546 _ => unreachable!(),
547 };
548 let arity = match operation { Operation::Sqrt => 1, Operation::Fma => 3, _ => 2 };
549 let b = if arity >= 2 { f.operand(words[3])? } else { 0 };
550 let c = if arity == 3 { f.operand(words[4])? } else { 0 };
551 let mode = Mode::optional(words.get(arity + 2))?;
552 Ok(format!("{command} {} {mode} {}\n", f.name, f.result(f.arithmetic(operation, a, b, c, mode))))
553 }
554 }
555}
556
557fn main() -> io::Result<()> {
558 let mut input = String::new();
559 io::stdin().read_to_string(&mut input)?;
560 let mut lines = input.lines();
561 let count: usize = lines.next().unwrap().parse().unwrap();
562 let mut formats = BTreeMap::new();
563 for (name, e, m) in [("HALF", 5, 10), ("E5M2", 5, 2), ("E4M3", 4, 3)] {
564 formats.insert(name.into(), Format::new(name, e, m));
565 }
566 let mut output = io::BufWriter::new(io::stdout().lock());
567 for line in lines.take(count) {
568 let words: Vec<_> = line.split_whitespace().collect();
569 match execute(&mut formats, &words) {
570 Ok(response) => output.write_all(response.as_bytes())?,
571 Err(error) => writeln!(output, "{error}")?,
572 }
573 }
574 output.flush()
575}
