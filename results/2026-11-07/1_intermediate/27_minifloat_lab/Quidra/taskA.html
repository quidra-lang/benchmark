29class Format(string name, int e, int m)
30 width := 1 + e + m
31 bias := 2 ^ (e - 1) - 1
32 emin := 1 - bias
33 emax := bias
34 hidden := 2 ^ m
35 exponent_mask := 2 ^ e - 1
36 sign_bit := 2 ^ (e + m)
37 infinity := exponent_mask * hidden
38 nan_pattern := infinity + hidden / 2
39 maximum := infinity - 1

103Exact exact_sum(const Exact[] terms, Mode mode)
104 nonzero := [for term := terms[] if term.n != 0 then term]
105 if #nonzero == 0
106 same := true
107 for term := terms[]
108 if term.negative != terms[0].negative
109 same = false
110 return Exact(0, negative = if same then terms[0].negative else mode == Mode.RDN)
111 exponent := [for term := nonzero[] then term.exponent].min()
112 aligned := [for term := nonzero[] then (if term.negative then -1 else 1) * term.n * 2 ^ (term.exponent - exponent)]
113 total := aligned.sum()
114 return Exact(math.abs(total), exponent, negative = if total == 0 then mode == Mode.RDN else total < 0)

152int integer_sqrt(int value)
153 if value == 0
154 return 0
155 root := 1
156 while root * root <= value
157 root *= 2
158 while true
159 next := (root + value / root) / 2
160 if next >= root

164bool increment(Mode mode, bool negative, int integer, bool inexact, int midpoint)
165 if not inexact
166 return false
167 if mode == Mode.RNE
168 return midpoint > 0 or midpoint == 0 and integer % 2 == 1
169 if mode == Mode.RNA
170 return midpoint >= 0
171 if mode == Mode.RUP
172 return not negative
173 return mode == Mode.RDN and negative

175Units round_units(const Exact exact, int unit_exponent, Mode mode, square_root := false)
176 ratio := scaled(exact, if square_root then 2 * unit_exponent else unit_exponent)
177 int integer
178 bool inexact
179 int midpoint
180 if square_root
181 integer = integer_sqrt(ratio.n / ratio.d)
182 inexact = integer * integer * ratio.d != ratio.n
183 midpoint = 4 * ratio.n - ratio.d * (2 * integer + 1) ^ 2
184 else
185 integer = ratio.n / ratio.d
186 remainder := ratio.n % ratio.d
187 inexact = remainder != 0
188 midpoint = 2 * remainder - ratio.d
189 if increment(mode, exact.negative, integer, inexact, midpoint)
190 integer += 1
191 return Units(integer, inexact)

194 exponent := pattern / fmt.hidden % (fmt.exponent_mask + 1)
195 fraction := pattern % fmt.hidden
196 q := (if exponent == 0 then fmt.emin else exponent - fmt.bias) - fmt.m

229Result overflow_result(const Format fmt, Mode mode, bool negative)
230 infinite := mode == Mode.RNE or mode == Mode.RNA or mode == Mode.RUP and not negative or mode == Mode.RDN and negative
231 pattern := zero_pattern(fmt, negative) + (if infinite then fmt.infinity else fmt.maximum)

234Result rounded(const Format fmt, const Exact exact, Mode mode, square_root := false)
235 if exact.n == 0
236 return Result(zero_pattern(fmt, exact.negative))
237 order := log_floor(exact)
238 if square_root
239 order = if order >= 0 then order / 2 else -((-order + 1) / 2)
240 if order > fmt.emax
241 return overflow_result(fmt, mode, exact.negative)
242 exponent := math.max(order, fmt.emin) - fmt.m
243 units := round_units(exact, exponent, mode, square_root)
244 significand := units.value
245 if significand == 2 * fmt.hidden
246 significand = fmt.hidden
247 exponent += 1
248 if exponent > fmt.emax - fmt.m
249 return overflow_result(fmt, mode, exact.negative)
250 magnitude := if significand >= fmt.hidden then (exponent + fmt.m + fmt.bias) * fmt.hidden + significand - fmt.hidden else significand
251 underflow := false
252 if units.inexact and order < fmt.emin
253 underflow = true
254 if order == fmt.emin - 1
255 unbounded := round_units(exact, order - fmt.m, mode, square_root)
256 underflow = unbounded.value < 2 * fmt.hidden
257 return Result(zero_pattern(fmt, exact.negative) + magnitude, Flags(underflow = underflow, inexact = units.inexact))

295Exact | Special | error parse_real(string word)
296 if word == "nan"
297 return Special.NaN
298 if word == "inf"
299 return Special.PositiveInfinity
300 if word == "-inf"
301 return Special.NegativeInfinity
302 if word.starts_with("0b") or word.starts_with("0x")
303 return error("BAD OPERAND {word}")
304 negative := word.starts_with("-")
305 body := if negative then word[1:] else word
306 parts := body.split("/")
307 if #parts == 2
308 numerator := int(parts[0])
309 denominator := int(parts[1])
310 if denominator == 0
311 return error("BAD OPERAND {word}")
312 return Exact(numerator, d = denominator, negative = negative)
313 parts = body.split(".")
314 numerator := int(parts.join(""))
315 denominator := if #parts == 2 then 10 ^ int(#parts[1]) else 1
316 return Exact(numerator, d = denominator, negative = negative)

348Result operation(const Format fmt, string op, const int[] patterns, Mode mode)
349 a := patterns[0]
350 if op == "SUB"
351 return operation(fmt, "ADD", [a, flip(fmt, patterns[1])], mode)
352 has_nan := false
353 signaling := false
354 for pattern := patterns[]
355 has_nan = has_nan or is_nan(fmt, pattern)
356 signaling = signaling or is_signaling(fmt, pattern)
357 if has_nan
358 return Result(fmt.nan_pattern, Flags(invalid = signaling))
359 if op == "SQRT"
360 if is_zero(fmt, a)
361 return Result(a)
362 if a >= fmt.sign_bit
363 return invalid_result(fmt)
364 if is_infinity(fmt, a)
365 return Result(a)
366 return rounded(fmt, from_pattern(fmt, a), mode, true)
367 b := patterns[1]
368 negative := (a >= fmt.sign_bit) != (b >= fmt.sign_bit)
369 if op == "ADD"
370 if is_infinity(fmt, a) and is_infinity(fmt, b) and negative
371 return invalid_result(fmt)
372 if is_infinity(fmt, a)
373 return Result(a)
374 if is_infinity(fmt, b)
375 return Result(b)
376 return rounded(fmt, exact_sum([from_pattern(fmt, a), from_pattern(fmt, b)], mode), mode)
377 if op == "MUL"
378 if is_infinity(fmt, a) or is_infinity(fmt, b)
379 return if is_zero(fmt, a) or is_zero(fmt, b) then invalid_result(fmt) else Result(infinity_pattern(fmt, negative))
380 if is_zero(fmt, a) or is_zero(fmt, b)
381 return Result(zero_pattern(fmt, negative))
382 return rounded(fmt, product(from_pattern(fmt, a), from_pattern(fmt, b)), mode)
383 if op == "DIV"
384 if is_infinity(fmt, a)
385 return if is_infinity(fmt, b) then invalid_result(fmt) else Result(infinity_pattern(fmt, negative))
386 if is_infinity(fmt, b)
387 return Result(zero_pattern(fmt, negative))
388 if is_zero(fmt, b)
389 return if is_zero(fmt, a) then invalid_result(fmt) else Result(infinity_pattern(fmt, negative), Flags(divbyzero = true))
390 if is_zero(fmt, a)
391 return Result(zero_pattern(fmt, negative))
392 left := from_pattern(fmt, a)
393 right := from_pattern(fmt, b)
394 return rounded(fmt, Exact(left.n, left.exponent - right.exponent, right.n, negative), mode)
395 c := patterns[2]
396 if is_infinity(fmt, a) or is_infinity(fmt, b)
397 if is_zero(fmt, a) or is_zero(fmt, b) or is_infinity(fmt, c) and (c >= fmt.sign_bit) != negative
398 return invalid_result(fmt)
399 return Result(infinity_pattern(fmt, negative))
400 if is_infinity(fmt, c)
401 return Result(c)
402 multiplied := product(from_pattern(fmt, a), from_pattern(fmt, b))
403 return rounded(fmt, exact_sum([multiplied, from_pattern(fmt, c)], mode), mode)

430void | error summation(string op, const Format fmt, Mode mode, const int[] patterns)
431 for pattern := patterns[]
432 if not is_finite(fmt, pattern)
433 return error("NOT FINITE")
434 emit("{op} {fmt.name} {mode_text(mode)} count={#patterns}")
435 total := patterns[0]
436 compensation := 0
437 flags := Flags()
438 for: 1 <= index < int(#patterns)
439 if op == "SUM"
440 added := operation(fmt, "ADD", [total, patterns[index]], mode)
441 total = added.pattern
442 flags = combined(flags, added.flags)
443 emit("PARTIAL {index + 1} " + result_text(fmt, added))
444 else
445 y_value := operation(fmt, "SUB", [patterns[index], compensation], mode)
446 t_value := operation(fmt, "ADD", [total, y_value.pattern], mode)
447 d_value := operation(fmt, "SUB", [t_value.pattern, total], mode)
448 c_value := operation(fmt, "SUB", [d_value.pattern, y_value.pattern], mode)
449 total = t_value.pattern
450 compensation = c_value.pattern
451 step_flags := combined(combined(y_value.flags, t_value.flags), combined(d_value.flags, c_value.flags))
452 flags = combined(flags, step_flags)
453 emit("KSTEP {index + 1} sum={value_text(fmt, total)} comp={value_text(fmt, compensation)} {flag_text(step_flags)}")
454 summary(fmt, mode, patterns, Result(total, flags))
455 return void

457void | error dot_product(const Format fmt, Mode mode, const int[] left, const int[] right)
458 if #left != #right
459 return error("LENGTH MISMATCH")
460 for pattern := left.concat(right)[]
461 if not is_finite(fmt, pattern)
462 return error("NOT FINITE")
463 naive := operation(fmt, "MUL", [left[0], right[0]], mode)
464 chained := naive
465 for: 1 <= index < int(#left)
466 multiplied := operation(fmt, "MUL", [left[index], right[index]], mode)
467 added := operation(fmt, "ADD", [naive.pattern, multiplied.pattern], mode)
468 naive = Result(added.pattern, combined(naive.flags, combined(multiplied.flags, added.flags)))
469 fused := operation(fmt, "FMA", [left[index], right[index], chained.pattern], mode)
470 chained = Result(fused.pattern, combined(chained.flags, fused.flags))
471 terms := [for index := left.indices()[] then product(from_pattern(fmt, left[index]), from_pattern(fmt, right[index]))]
472 exact := exact_sum(terms, mode)
473 emit("DOT {fmt.name} {mode_text(mode)} n={#left}")
474 emit("NAIVE " + result_text(fmt, naive))
475 emit("CHAINED " + result_text(fmt, chained))
476 emit("EXACT " + decimal_text(exact))
477 emit("FUSED " + result_text(fmt, rounded(fmt, exact, mode)))
478 emit("DIFF naive={difference(fmt, naive.pattern, exact)} chained={difference(fmt, chained.pattern, exact)}")
479 return void

481void | error table_output(const Format fmt)
482 if fmt.width > 8
483 return error("TABLE TOO LARGE")
484 emit("TABLE {fmt.name} count={fmt.infinity}")
485 for: 0 <= pattern < fmt.infinity
486 gap := "-"

493 emit("EPSILON " + decimal_text(Exact(1, -fmt.m)))
494 emit("PATTERNS finite={2 * fmt.infinity} infinite=2 nan={2 * (fmt.hidden - 1)} total={2 * fmt.sign_bit}")
495 return void

519void | error command(map<string, Format> &formats, const string[] words)
520 op := words[0]
521 name := words[1]
522 if op == "FORMAT"
523 if formats.has(name)
524 return error("FORMAT EXISTS {name}")
525 e := int(words[2])
526 m := int(words[3])
527 if e < 2 or e > 5 or m < 1 or m > 10
528 return error("BAD FORMAT")
529 added := Format(name, e, m)
530 .formats.set(name, added)
531 emit("FORMAT {name} e={e} m={m} bits={added.width} bias={added.bias} emin={added.emin} emax={added.emax}")
532 return void
533 fmt := try lookup_format(formats, name)
534 if op == "ENCODE"
535 Exact | Special value := try parse_real(words[2])
536 mode := try optional_mode(words, 3)
537 string text
538 match value
539 Exact exact
540 text = fraction_text(exact)
541 Special special
542 text = special_text(special)
543 emit("ENCODE {name} {mode_text(mode)} {text} " + result_text(fmt, encoded(fmt, value, mode)))
544 elif op == "CONVERT"
545 destination := try lookup_format(formats, words[2])
546 pattern := try operand(fmt, words[3])
547 mode := try optional_mode(words, 4)
548 Result converted
549 if is_nan(fmt, pattern)
550 converted = Result(destination.nan_pattern, Flags(invalid = is_signaling(fmt, pattern)))
551 elif is_infinity(fmt, pattern)
552 converted = Result(infinity_pattern(destination, pattern >= fmt.sign_bit))
553 else
554 converted = rounded(destination, from_pattern(fmt, pattern), mode)
555 emit("CONVERT {name} {destination.name} {mode_text(mode)} " + result_text(destination, converted))
556 elif op == "SUM" or op == "KAHAN" or op == "DOT"
557 mode := try parse_mode(words[2])
558 if op == "DOT"
559 separator := 3
560 while words[separator] != "|"
561 separator += 1
562 left := try operands(fmt, words[3:<separator])
563 right := try operands(fmt, words[separator + 1:])
564 try dot_product(fmt, mode, left, right)
565 else
566 patterns := try operands(fmt, words[3:])
567 try summation(op, fmt, mode, patterns)
568 elif op == "TABLE"
569 try table_output(fmt)
570 elif op == "ADD" or op == "SUB" or op == "MUL" or op == "DIV" or op == "SQRT" or op == "FMA"
571 count := if op == "SQRT" then 1 elif op == "FMA" then 3 else 2
572 patterns := try operands(fmt, words[2:<2 + count])
573 mode := try optional_mode(words, 2 + count)
574 emit("{op} {name} {mode_text(mode)} " + result_text(fmt, operation(fmt, op, patterns, mode)))
575 else
576 pattern := try operand(fmt, words[2])
577 parts := decode_fields(fmt, pattern)
578 if op == "DECODE"
579 classification := if is_nan(fmt, pattern) then (if is_signaling(fmt, pattern) then "snan" else "qnan") elif is_infinity(fmt, pattern) then "infinity" elif is_zero(fmt, pattern) then "zero" elif parts.exponent == 0 then "subnormal" else "normal"
580 sign := if parts.negative then 1 else 0
581 emit("DECODE {name} {bit_text(fmt, pattern)} {hex_text(fmt, pattern)} {classification} sign={sign}")
582 text := "FIELDS exponent={parts.exponent} fraction={parts.fraction}"
583 if is_finite(fmt, pattern)
584 leading := if parts.exponent == 0 then 0 else 1
585 emit(text + " unbiased={parts.q + fmt.m} significand={leading}." + radix_text(parts.fraction, 2, fmt.m))
586 emit("VALUE {fraction_text(from_pattern(fmt, pattern))} {value_text(fmt, pattern)}")
587 else
588 if is_nan(fmt, pattern)
589 text += " payload={parts.fraction % (fmt.hidden / 2)}"
590 emit(text)
591 elif op == "COMPARE"
592 other := try operand(fmt, words[3])
593 total_order := relation(order_key(fmt, pattern), order_key(fmt, other))
594 numeric := if is_nan(fmt, pattern) or is_nan(fmt, other) then "UN" elif is_zero(fmt, pattern) and is_zero(fmt, other) then "EQ" else total_order
595 flags := Flags(invalid = is_signaling(fmt, pattern) or is_signaling(fmt, other))
596 emit("COMPARE {name} {hex_text(fmt, pattern)} {hex_text(fmt, other)} {numeric} total={total_order} {flag_text(flags)}")
597 elif op == "TOINT"
