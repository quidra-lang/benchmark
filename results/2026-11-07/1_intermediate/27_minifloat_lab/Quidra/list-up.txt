class Fields(bool negative, int exponent, int fraction, int q)
class Flags(invalid := false, divbyzero := false, overflow := false, underflow := false, inexact := false)
class Result(int pattern, flags := Flags())
    hidden := 2 ^ m
    exponent_mask := 2 ^ e - 1
    sign_bit := 2 ^ (e + m)
    infinity := exponent_mask * hidden
    nan_pattern := infinity + hidden / 2
Fields decode_fields(const Format fmt, int pattern)
    exponent := pattern / fmt.hidden % (fmt.exponent_mask + 1)
    fraction := pattern % fmt.hidden
    q := (if exponent == 0 then fmt.emin else exponent - fmt.bias) - fmt.m
    return Fields(pattern >= fmt.sign_bit, exponent, fraction, q)

bool is_nan(const Format fmt, int pattern)
    parts := decode_fields(fmt, pattern)
    return parts.exponent == fmt.exponent_mask and parts.fraction != 0
bool is_signaling(const Format fmt, int pattern)
    return is_nan(fmt, pattern) and pattern % fmt.hidden < fmt.hidden / 2
bool is_zero(const Format fmt, int pattern)
    return pattern % fmt.sign_bit == 0
int flip(const Format fmt, int pattern)
    return if pattern >= fmt.sign_bit then pattern - fmt.sign_bit else pattern + fmt.sign_bit
int next_up(const Format fmt, int pattern)
    if pattern == fmt.infinity
        return pattern
    if is_zero(fmt, pattern)
        return 1
    return pattern + (if pattern >= fmt.sign_bit then -1 else 1)

int order_key(const Format fmt, int pattern)
    return if pattern >= fmt.sign_bit then fmt.sign_bit - pattern - 1 else pattern

string relation(int left, int right)
