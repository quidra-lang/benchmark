42scaled <- function(n, d, exponent) {

46log_two_floor <- function(value) {
47 exponent <- sizeinbase(value$n, 2L) - sizeinbase(value$d, 2L) + value$e
48 ratio <- scaled(value$n, value$d, value$e - exponent)
49 exponent - if (ratio$n < ratio$d) 1L else 0L
50}
51integer_sqrt <- function(value) {
52 if (value == 0L) return(ZERO)
53 root <- power_two((sizeinbase(value, 2L) + 1L) %/% 2L)
54 repeat {
55 next_root <- (root + value %/% root) %/% 2L
56 if (next_root >= root) return(root)
57 root <- next_root
58 }

60fraction_text <- function(value) {
61 if (value$n == 0L) return("0")
62 ratio <- scaled(value$n, value$d, value$e)
63 divisor <- gcd.bigz(ratio$n, ratio$d)
64 n <- ratio$n %/% divisor
65 d <- ratio$d %/% divisor
66 paste0(if (value$negative) "-" else "", as.character(n), if (d == 1L) "" else paste0("/", as.character(d)))
67}
68decimal_text <- function(value, keep_zero_sign = FALSE) {
69 ratio <- scaled(value$n, value$d, value$e)
70 text <- as.character(ratio$n %/% ratio$d)
71 remainder <- ratio$n %% ratio$d
72 if (remainder != 0L) {
73 text <- paste0(text, ".")
74 while (remainder != 0L) {
75 remainder <- remainder * 10L
76 text <- paste0(text, as.character(remainder %/% ratio$d))
77 remainder <- remainder %% ratio$d
78 }
79 }
80 paste0(if (value$negative && (value$n != 0L || keep_zero_sign)) "-" else "", text)
81}
82exact_product <- function(a, b) exact(a$n * b$n, a$e + b$e, a$d * b$d, xor(a$negative, b$negative))
83exact_sum <- function(terms, mode) {
84 nonzero <- terms[vapply(terms, function(term) term$n != 0L, logical(1L))]
85 if (!length(nonzero)) {
86 signs <- vapply(terms, function(term) term$negative, logical(1L))
87 return(exact(negative = if (all(signs == signs[1L])) signs[1L] else mode == "RDN"))
88 }
89 exponent <- min(vapply(nonzero, function(term) term$e, integer(1L)))
90 aligned <- c_bigz(lapply(nonzero, function(term) {
91 value <- term$n * power_two(term$e - exponent)
92 if (term$negative) -value else value
93 }))
94 total <- sum(aligned)
95 exact(abs(total), exponent, negative = total < 0L || (total == 0L && mode == "RDN"))
96}

98increment <- function(mode, negative, integer, inexact, halfway) {
99 if (!inexact) return(FALSE)
100 switch(mode,
101 RNE = halfway > 0L || (halfway == 0L && integer %% 2L == 1L),
102 RNA = halfway >= 0L, RTZ = FALSE, RUP = !negative, RDN = negative)
103}
104rounded_units <- function(value, exponent, mode) {
105 ratio <- scaled(value$n, value$d, value$e - exponent)
106 integer <- ratio$n %/% ratio$d
107 remainder <- ratio$n %% ratio$d
108 inexact <- remainder != 0L
109 if (increment(mode, value$negative, integer, inexact, 2L * remainder - ratio$d)) integer <- integer + 1L
110 list(value = integer, inexact = inexact)
111}
112rounded_sqrt_units <- function(value, exponent, mode) {
113 ratio <- scaled(value$n, value$d, value$e - 2L * exponent)
114 integer <- integer_sqrt(ratio$n %/% ratio$d)
115 inexact <- integer * integer * ratio$d != ratio$n
116 midpoint <- 2L * integer + 1L
117 halfway <- 4L * ratio$n - ratio$d * midpoint * midpoint
118 if (increment(mode, FALSE, integer, inexact, halfway)) integer <- integer + 1L
119 list(value = integer, inexact = inexact)
120}

122new_format <- function(name, e, m) {
123 bias <- bitwShiftL(1L, e - 1L) - 1L
124 hidden <- bitwShiftL(1L, m)
125 exponent_mask <- bitwShiftL(1L, e) - 1L
126 infinity <- exponent_mask * hidden
127 list(name = name, e = e, m = m, w = 1L + e + m, bias = bias, emin = 1L - bias,
128 emax = bias, hidden = hidden, exponent_mask = exponent_mask,
129 sign_bit = bitwShiftL(1L, e + m), infinity = infinity,
130 nan = infinity + hidden %/% 2L, maximum = infinity - 1L)
131}
132fields <- function(format, pattern) {
133 exponent <- pattern %/% format$hidden %% (format$exponent_mask + 1L)
134 list(negative = bitwAnd(pattern, format$sign_bit) != 0L,
135 exponent = exponent, fraction = pattern %% format$hidden,
136 q = (if (exponent == 0L) format$emin else exponent - format$bias) - format$m)
137}
138is_nan <- function(format, pattern) {
139 field <- fields(format, pattern)
140 field$exponent == format$exponent_mask && field$fraction != 0L
141}
142is_signaling <- function(format, pattern) is_nan(format, pattern) && fields(format, pattern)$fraction < format$hidden %/% 2L
143is_infinite <- function(format, pattern) bitwAnd(pattern, format$sign_bit - 1L) == format$infinity
144is_zero <- function(format, pattern) bitwAnd(pattern, format$sign_bit - 1L) == 0L
145is_finite <- function(format, pattern) fields(format, pattern)$exponent != format$exponent_mask
146zero_pattern <- function(format, negative) if (negative) format$sign_bit else 0L
147infinity_pattern <- function(format, negative) zero_pattern(format, negative) + format$infinity
148flip <- function(format, pattern) bitwXor(pattern, format$sign_bit)
149pattern_exact <- function(format, pattern) {

153}

171overflow_result <- function(format, negative, mode) {
172 infinite <- mode %in% c("RNE", "RNA") || (mode == "RUP" && !negative) || (mode == "RDN" && negative)
173 result(zero_pattern(format, negative) + if (infinite) format$infinity else format$maximum, bitwOr(OVERFLOW, INEXACT))
174}
175finish_round <- function(format, negative, order, exponent, units, unbounded, mode) {
176 n <- units$value
177 if (n == 2L * format$hidden) {
178 n <- as.bigz(format$hidden)
179 exponent <- exponent + 1L
180 }
181 if (exponent > format$emax - format$m) return(overflow_result(format, negative, mode))
182 magnitude <- if (n >= format$hidden) (exponent + format$m + format$bias) * format$hidden + small_integer(n - format$hidden) else small_integer(n)
183 flags <- if (units$inexact) INEXACT else 0L
184 if (units$inexact && order < format$emin && (order < format$emin - 1L || unbounded()$value < 2L * format$hidden)) {
185 flags <- bitwOr(flags, UNDERFLOW)
186 }
187 result(zero_pattern(format, negative) + magnitude, flags)
188}
189round_exact <- function(format, value, mode) {
190 if (value$n == 0L) return(result(zero_pattern(format, value$negative)))
191 order <- log_two_floor(value)
192 if (order > format$emax) return(overflow_result(format, value$negative, mode))
193 exponent <- max(order, format$emin) - format$m
194 finish_round(format, value$negative, order, exponent, rounded_units(value, exponent, mode),
195 function() rounded_units(value, order - format$m, mode), mode)
196}
197round_sqrt <- function(format, value, mode) {
198 order <- log_two_floor(value) %/% 2L
199 exponent <- max(order, format$emin) - format$m
200 finish_round(format, FALSE, order, exponent, rounded_sqrt_units(value, exponent, mode),
201 function() rounded_sqrt_units(value, order - format$m, mode), mode)
202}

204parse_real <- function(word) {
205 if (word %in% c("inf", "-inf", "nan")) return(list(special = word))
206 if (startsWith(word, "0b") || startsWith(word, "0x")) fail(paste("BAD OPERAND", word))
207 negative <- startsWith(word, "-")
208 body <- if (negative) substring(word, 2L) else word
209 if (grepl("/", body, fixed = TRUE)) {
210 parts <- strsplit(body, "/", fixed = TRUE)[[1L]]
211 numerator <- decimal_integer(parts[1L])
212 denominator <- decimal_integer(parts[2L])
213 if (denominator == 0L) fail(paste("BAD OPERAND", word))
214 return(list(exact = exact(numerator, d = denominator, negative = negative)))
215 }
216 parts <- strsplit(body, ".", fixed = TRUE)[[1L]]
217 fraction <- if (length(parts) == 2L) parts[2L] else ""
218 list(exact = exact(decimal_integer(paste0(parts[1L], fraction)), d = as.bigz(10L)^nchar(fraction), negative = negative))
219}
220encoded <- function(format, real, mode) {
221 if (!is.null(real$special)) return(result(switch(real$special, nan = format$nan, inf = format$infinity, "-inf" = infinity_pattern(format, TRUE))))
222 round_exact(format, real$exact, mode)
223}
224operand <- function(format, word) {
225 if (startsWith(word, "0b")) {
226 if (nchar(word) - 2L != format$w) fail(paste("BAD OPERAND", word))
227 return(small_integer(as.bigz(word)))
228 }
229 if (startsWith(word, "0x")) {
230 pattern <- small_integer(as.bigz(word))
231 if (nchar(word) - 2L != (format$w + 3L) %/% 4L || pattern >= bitwShiftL(1L, format$w)) fail(paste("BAD OPERAND", word))
232 return(pattern)
233 }
234 encoded(format, parse_real(word), "RNE")$pattern
235}

237operation <- function(format, command, inputs, mode) {
238 if (command == "SUB") return(operation(format, "ADD", c(inputs[1L], flip(format, inputs[2L])), mode))
239 if (any(vapply(inputs, function(pattern) is_nan(format, pattern), logical(1L)))) {
240 return(result(format$nan, if (any(vapply(inputs, function(pattern) is_signaling(format, pattern), logical(1L)))) INVALID else 0L))
241 }
242 invalid <- function() result(format$nan, INVALID)
243 a <- inputs[1L]
244 negative <- fields(format, a)$negative
245 if (command == "SQRT") {
246 if (is_zero(format, a)) return(result(a))
247 if (negative) return(invalid())
248 if (is_infinite(format, a)) return(result(a))
249 return(round_sqrt(format, pattern_exact(format, a), mode))
250 }
251 b <- inputs[2L]
252 sigma <- xor(negative, fields(format, b)$negative)
253 if (command == "ADD") {
254 if (is_infinite(format, a) && is_infinite(format, b) && sigma) return(invalid())
255 if (is_infinite(format, a)) return(result(a))
256 if (is_infinite(format, b)) return(result(b))
257 return(round_exact(format, exact_sum(list(pattern_exact(format, a), pattern_exact(format, b)), mode), mode))
258 }
259 if (command == "MUL") {
260 if (is_infinite(format, a) || is_infinite(format, b)) {
261 return(if (is_zero(format, a) || is_zero(format, b)) invalid() else result(infinity_pattern(format, sigma)))
262 }
263 if (is_zero(format, a) || is_zero(format, b)) return(result(zero_pattern(format, sigma)))
264 return(round_exact(format, exact_product(pattern_exact(format, a), pattern_exact(format, b)), mode))
265 }
266 if (command == "DIV") {
267 if (is_infinite(format, a)) return(if (is_infinite(format, b)) invalid() else result(infinity_pattern(format, sigma)))
268 if (is_infinite(format, b)) return(result(zero_pattern(format, sigma)))
269 if (is_zero(format, b)) return(if (is_zero(format, a)) invalid() else result(infinity_pattern(format, sigma), DIVBYZERO))
270 if (is_zero(format, a)) return(result(zero_pattern(format, sigma)))
271 x <- pattern_exact(format, a)
272 y <- pattern_exact(format, b)
273 return(round_exact(format, exact(x$n * y$d, x$e - y$e, x$d * y$n, sigma), mode))
274 }
275 c <- inputs[3L]
276 if (is_infinite(format, a) || is_infinite(format, b)) {
277 if (is_zero(format, a) || is_zero(format, b) || (is_infinite(format, c) && fields(format, c)$negative != sigma)) return(invalid())
278 return(result(infinity_pattern(format, sigma)))
279 }
280 if (is_infinite(format, c)) return(result(c))
281 terms <- list(exact_product(pattern_exact(format, a), pattern_exact(format, b)), pattern_exact(format, c))
282 round_exact(format, exact_sum(terms, mode), mode)
283}

295summation <- function(command, format, mode, patterns) {
296 if (any(!vapply(patterns, function(pattern) is_finite(format, pattern), logical(1L)))) fail("NOT FINITE")
297 output <- paste(command, format$name, mode, paste0("count=", length(patterns)))
298 current <- patterns[1L]
299 compensation <- 0L
300 flags <- 0L
301 for (index in seq_along(patterns)[-1L]) {
302 if (command == "SUM") {
303 added <- operation(format, "ADD", c(current, patterns[index]), mode)
304 current <- added$pattern
305 flags <- bitwOr(flags, added$flags)
306 output <- c(output, paste("PARTIAL", index, result_text(format, added)))
307 } else {
308 y <- operation(format, "SUB", c(patterns[index], compensation), mode)
309 t <- operation(format, "ADD", c(current, y$pattern), mode)
310 d <- operation(format, "SUB", c(t$pattern, current), mode)
311 correction <- operation(format, "SUB", c(d$pattern, y$pattern), mode)
312 current <- t$pattern
313 compensation <- correction$pattern
314 step_flags <- combine_flags(y$flags, t$flags, d$flags, correction$flags)
315 flags <- bitwOr(flags, step_flags)
316 output <- c(output, paste("KSTEP", index, paste0("sum=", value_text(format, current)),
317 paste0("comp=", value_text(format, compensation)), flags_text(step_flags)))
318 }
319 }
320 value <- exact_sum(lapply(patterns, function(pattern) pattern_exact(format, pattern)), mode)
321 c(output, paste("RESULT", result_text(format, result(current, flags))),
322 paste("EXACT", decimal_text(value)), paste("ROUNDED", result_text(format, round_exact(format, value, mode))),
323 paste("DIFF", difference_text(format, current, value)))
324}
325dot_product <- function(format, mode, a, b) {
326 if (length(a) != length(b)) fail("LENGTH MISMATCH")
327 if (any(!vapply(c(a, b), function(pattern) is_finite(format, pattern), logical(1L)))) fail("NOT FINITE")
328 naive <- operation(format, "MUL", c(a[1L], b[1L]), mode)
329 chained <- naive
330 for (index in seq_along(a)[-1L]) {
331 product <- operation(format, "MUL", c(a[index], b[index]), mode)
332 added <- operation(format, "ADD", c(naive$pattern, product$pattern), mode)
333 naive <- result(added$pattern, combine_flags(naive$flags, product$flags, added$flags))
334 fused <- operation(format, "FMA", c(a[index], b[index], chained$pattern), mode)
335 chained <- result(fused$pattern, bitwOr(chained$flags, fused$flags))
336 }
337 terms <- lapply(seq_along(a), function(index) exact_product(pattern_exact(format, a[index]), pattern_exact(format, b[index])))
338 value <- exact_sum(terms, mode)
339 c(paste("DOT", format$name, mode, paste0("n=", length(a))),
340 paste("NAIVE", result_text(format, naive)), paste("CHAINED", result_text(format, chained)),
341 paste("EXACT", decimal_text(value)), paste("FUSED", result_text(format, round_exact(format, value, mode))),
342 paste("DIFF", paste0("naive=", difference_text(format, naive$pattern, value)),
343 paste0("chained=", difference_text(format, chained$pattern, value))))
344}

350 paste("ENTRY", pattern, pattern_text(format, pattern), paste0("gap=", gap))

356 paste0("total=", bitwShiftL(1L, format$w))))

366run_command <- function(line) {
367 words <- strsplit(line, " ", fixed = TRUE)[[1L]]
368 command <- words[1L]
369 if (command == "FORMAT") {
370 name <- words[2L]
371 e <- as.integer(words[3L])
372 m <- as.integer(words[4L])
373 if (exists(name, envir = formats, inherits = FALSE)) fail(paste("FORMAT EXISTS", name))
374 if (e < 2L || e > 5L || m < 1L || m > 10L) fail("BAD FORMAT")
375 format <- new_format(name, e, m)
376 formats[[name]] <- format
377 return(paste("FORMAT", name, paste0("e=", e), paste0("m=", m), paste0("bits=", format$w),
378 paste0("bias=", format$bias), paste0("emin=", format$emin), paste0("emax=", format$emax)))
379 }
380 format <- format_named(words[2L])
381 name <- format$name
382 if (command == "ENCODE") {
383 real <- parse_real(words[3L])
384 mode <- mode_word(words, 4L)
385 value <- if (is.null(real$special)) fraction_text(real$exact) else real$special
386 return(paste("ENCODE", name, mode, value, result_text(format, encoded(format, real, mode))))
387 }
388 if (command == "CONVERT") {
389 target <- format_named(words[3L])
390 a <- operand(format, words[4L])
391 mode <- mode_word(words, 5L)
392 converted <- if (is_nan(format, a)) result(target$nan, if (is_signaling(format, a)) INVALID else 0L)
393 else if (is_infinite(format, a)) result(infinity_pattern(target, fields(format, a)$negative))
394 else round_exact(target, pattern_exact(format, a), mode)
395 return(paste("CONVERT", name, target$name, mode, result_text(target, converted)))
396 }
397 if (command %in% c("SUM", "KAHAN", "DOT")) {
398 mode <- mode_word(words, 3L)
399 if (command == "DOT") {
400 separator <- match("|", words)
401 a <- vapply(words[seq.int(4L, separator - 1L)], function(word) operand(format, word), integer(1L))
402 b <- vapply(words[seq.int(separator + 1L, length(words))], function(word) operand(format, word), integer(1L))
403 return(dot_product(format, mode, a, b))
404 }
405 patterns <- vapply(words[-seq_len(3L)], function(word) operand(format, word), integer(1L))
406 return(summation(command, format, mode, patterns))
407 }
408 if (command == "TABLE") return(table_lines(format))
409 if (command %in% c("ADD", "SUB", "MUL", "DIV", "SQRT", "FMA")) {
410 count <- if (command == "SQRT") 1L else if (command == "FMA") 3L else 2L
411 patterns <- vapply(words[seq.int(3L, 2L + count)], function(word) operand(format, word), integer(1L))
412 mode <- mode_word(words, 3L + count)
413 return(paste(command, name, mode, result_text(format, operation(format, command, patterns, mode))))
414 }
415 a <- operand(format, words[3L])
416 field <- fields(format, a)
417 if (command == "DECODE") {
418 classification <- if (is_nan(format, a)) if (is_signaling(format, a)) "snan" else "qnan"
419 else if (is_infinite(format, a)) "infinity" else if (is_zero(format, a)) "zero"
420 else if (field$exponent == 0L) "subnormal" else "normal"
421 header <- paste("DECODE", name, bits_text(format, a), hex_text(format, a), classification, paste0("sign=", as.integer(field$negative)))
422 description <- paste("FIELDS", paste0("exponent=", field$exponent), paste0("fraction=", field$fraction))
423 if (is_finite(format, a)) {
424 description <- paste(description, paste0("unbiased=", field$q + format$m),
425 paste0("significand=", if (field$exponent == 0L) "0" else "1", ".", binary_text(field$fraction, format$m)))
426 return(c(header, description, paste("VALUE", fraction_text(pattern_exact(format, a)), value_text(format, a))))
427 }
428 if (is_nan(format, a)) description <- paste(description, paste0("payload=", field$fraction %% (format$hidden %/% 2L)))
429 return(c(header, description))
430 }
431 if (command == "COMPARE") {
432 b <- operand(format, words[4L])
433 key <- function(pattern) if (fields(format, pattern)$negative) -(pattern - format$sign_bit) - 1L else pattern
434 total <- relation(key(a), key(b))
435 numeric <- if (is_nan(format, a) || is_nan(format, b)) "UN" else if (is_zero(format, a) && is_zero(format, b)) "EQ" else total
436 flags <- if (is_signaling(format, a) || is_signaling(format, b)) INVALID else 0L
437 return(paste("COMPARE", name, hex_text(format, a), hex_text(format, b), numeric, paste0("total=", total), flags_text(flags)))
438 }
439 if (command == "TOINT") {
440 type <- words[4L]
441 if (!type %in% names(TYPES)) fail(paste("BAD TYPE", type))
442 mode <- mode_word(words, 5L)
443 prefix <- paste("TOINT", name, type, mode)
444 if (!is_finite(format, a)) return(paste(prefix, "invalid flags=invalid"))
445 units <- rounded_units(pattern_exact(format, a), 0L, mode)
446 integer <- if (field$negative) -units$value else units$value
447 bounds <- TYPES[[type]]
448 if (integer < bounds[1L] || integer > bounds[2L]) return(paste(prefix, "invalid flags=invalid"))
449 return(paste(prefix, as.character(integer), flags_text(if (units$inexact) INEXACT else 0L)))
450 }
451 value <- if (is_nan(format, a)) result(format$nan, if (is_signaling(format, a)) INVALID else 0L)
452 else if (command == "NEXTUP") result(next_up(format, a))
453 else if (command == "NEXTDOWN") result(flip(format, next_up(format, flip(format, a))))
454 else result(if (is_infinite(format, a)) format$infinity else round_exact(format, exact(ONE, field$q), "RNE")$pattern)
455 paste(command, name, result_text(format, value))
456}
