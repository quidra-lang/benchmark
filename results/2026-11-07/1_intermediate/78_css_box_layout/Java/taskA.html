105 static boolean integer(String s) {
106 if (s.isEmpty() || s.length() > 3) return false;
107 for (int i = 0; i < s.length(); i++) if (s.charAt(i) < '0' || s.charAt(i) > '9') return false;
108 return true;
109 }

112 return integer(s) && Integer.parseInt(s) > 0;

131 static Map<String, String> declaration(String property, String value) {
132 String[] tokens = value.split(" ", -1);
133 Map<String, String> result = new LinkedHashMap<>();
134 if (oneOf(property, "margin padding border gap")) {
135 boolean gap = property.equals("gap");
136 if (tokens.length < 1 || tokens.length > (gap ? 2 : 4)) return null;
137 for (String token : tokens) if (!integer(token) || property.equals("border") && !oneOf(token, "0 1")) return null;
138 if (gap) {
139 result.put("row-gap", tokens[0]);
140 result.put("column-gap", tokens[tokens.length - 1]);
141 } else {
142 String[] sides = {"top", "right", "bottom", "left"};
143 String[] expanded = {tokens[0], tokens.length > 1 ? tokens[1] : tokens[0], tokens.length > 2 ? tokens[2] : tokens[0], tokens.length > 3 ? tokens[3] : tokens.length > 1 ? tokens[1] : tokens[0]};
144 for (int i = 0; i < 4; i++) result.put(property + "-" + sides[i], expanded[i]);
145 }
146 return result;
147 }
148 if (!INITIAL.containsKey(property)) return null;
149 boolean valid;
150 if (property.startsWith("grid-template-")) {
151 valid = value.equals("none");
152 if (!valid) {
153 valid = true;
154 for (String token : tokens) if (!integer(token) && !token.equals("auto") && !(token.endsWith("fr") && positive(token.substring(0, token.length() - 2)))) valid = false;
155 }
156 } else if (property.equals("grid-column") || property.equals("grid-row")) {
157 valid = tokens.length == 1 && (value.equals("auto") || positive(value))
158 || tokens.length == 2 && tokens[0].equals("span") && positive(tokens[1])
159 || tokens.length == 3 && positive(tokens[0]) && tokens[1].equals("/") && positive(tokens[2]) && Integer.parseInt(tokens[0]) < Integer.parseInt(tokens[2])
160 || tokens.length == 4 && positive(tokens[0]) && tokens[1].equals("/") && tokens[2].equals("span") && positive(tokens[3]);
161 } else {
162 if (tokens.length != 1) return null;
163 valid = switch (property) {
164 case "display" -> oneOf(value, "block flex grid none");
165 case "position" -> oneOf(value, "static relative absolute");
166 case "box-sizing" -> oneOf(value, "content-box border-box");
167 case "width", "height", "flex-basis" -> value.equals("auto") || integer(value) || percentage(value);
168 case "min-width", "min-height" -> integer(value) || percentage(value);
169 case "max-width", "max-height" -> value.equals("none") || integer(value) || percentage(value);
170 case "top", "right", "bottom", "left" -> value.equals("auto") || signed(value);
171 case "order" -> signed(value);
172 case "flex-direction" -> oneOf(value, "row row-reverse column column-reverse");
173 case "flex-wrap" -> oneOf(value, "nowrap wrap");
174 case "justify-content" -> oneOf(value, "start end center space-between space-around space-evenly");
175 case "align-items" -> oneOf(value, "stretch start end center");
176 case "align-self" -> oneOf(value, "auto stretch start end center");
177 default -> property.startsWith("border-") ? oneOf(value, "0 1") : integer(value);
178 };
179 }
180 if (!valid) return null;
181 result.put(property, value);
182 return result;
183 }

192 static int resolve(Element e, String property, int reference, boolean horizontal) {
193 String value = e.get(property);
194 if (value.equals("auto") || value.equals("none")) return -1;
195 int n;
196 if (value.endsWith("%")) {
197 if (reference < 0) return -1;
198 n = (int)((long)reference * Integer.parseInt(value.substring(0, value.length() - 1)) / 100);
199 } else n = Integer.parseInt(value);
200 return e.get("box-sizing").equals("border-box") ? Math.max(0, n - (horizontal ? e.hf() : e.vf())) : n;
201 }

203 static int clamp(Element e, int value, int reference, boolean horizontal) {
204 String axis = horizontal ? "width" : "height";
205 int low = resolve(e, "min-" + axis, reference, horizontal), high = resolve(e, "max-" + axis, reference, horizontal);
206 if (high >= 0) value = Math.min(value, high);
207 return Math.max(value, Math.max(0, low));

210 static int[] distribute(int total, long[] weights) {
211 int[] result = new int[weights.length];
212 long sum = 0;
213 for (long w : weights) sum += w;
214 if (sum == 0) return result;
215 int remaining = total;
216 for (int i = 0; i < weights.length; i++) {
217 result[i] = (int)((long)total * weights[i] / sum);
218 remaining -= result[i];
219 }
220 for (int i = 0; i < weights.length && remaining > 0; i++) if (weights[i] > 0) {
221 result[i]++;
222 remaining--;
223 }
224 return result;
225 }

250 static Grid grid(Element e) {
251 List<GridItem> items = new ArrayList<>();
252 int nc = tracks(e.get("grid-template-columns"), true).size(), nr = tracks(e.get("grid-template-rows"), false).size();
253 for (Element child : flow(e, true)) {
254 GridItem item = new GridItem(child);
255 items.add(item);
256 nc = Math.max(nc, item.column == 0 ? item.columns : item.column + item.columns - 1);
257 }
258 for (GridItem item : items) if (item.row > 0 && item.column > 0) item.placed = true;
259 int cr = 1, cc = 1;
260 for (GridItem item : items) {
261 if (item.placed) continue;
262 if (item.row > 0) {
263 int c = 1;
264 while (c <= nc - item.columns + 1 && overlaps(items, item, item.row, c)) c++;
265 item.column = c <= nc - item.columns + 1 ? c : 1;
266 } else if (item.column > 0) {
267 if (item.column < cc) cr++;
268 int r = cr;
269 while (overlaps(items, item, r, item.column)) r++;
270 item.row = r;
271 cr = r;
272 cc = item.column + item.columns;
273 } else {
274 int r = cr, c = cc;
275 while (true) {
276 if (c + item.columns - 1 > nc) {
277 r++;
278 c = 1;
279 } else if (overlaps(items, item, r, c)) c++;
280 else break;
281 }
282 item.row = r;
283 item.column = c;
284 cr = r;
285 cc = c + item.columns;
286 }
287 item.placed = true;
288 }
289 for (GridItem item : items) nr = Math.max(nr, item.row + item.rows - 1);
290 return new Grid(items, nc, nr);
291 }

293 static int[] trackSizes(Element e, Grid grid, boolean columns, int available, boolean intrinsic) {
294 int count = columns ? grid.columns : grid.rows;
295 List<Track> specs = tracks(e.get(columns ? "grid-template-columns" : "grid-template-rows"), columns);
296 while (specs.size() < count) specs.add(new Track('a', 0));
297 int[] sizes = new int[count];
298 for (int i = 0; i < count; i++) {
299 Track track = specs.get(i);
300 if (track.kind == 'n') sizes[i] = track.value;
301 else if (track.kind == 'a' || available < 0) {
302 for (GridItem item : grid.items) if ((columns ? item.column : item.row) == i + 1 && (columns ? item.columns : item.rows) == 1) sizes[i] = Math.max(sizes[i], intrinsic ? contribution(item.e) : item.contribution);
303 }
304 }
305 int gap = e.number(columns ? "column-gap" : "row-gap");
306 int leftover = available - Arrays.stream(sizes).sum() - gap * Math.max(0, count - 1);
307 if (available >= 0 && leftover > 0) {
308 boolean fr = specs.stream().anyMatch(t -> t.kind == 'f');
309 long[] weights = new long[count];
310 for (int i = 0; i < count; i++) if (fr ? specs.get(i).kind == 'f' : specs.get(i).kind == 'a') weights[i] = fr ? specs.get(i).value : 1;
311 int[] extra = distribute(leftover, weights);
312 for (int i = 0; i < count; i++) sizes[i] += extra[i];
313 }
314 return sizes;
315 }

317 static int maxContent(Element e) {
318 if (e.intrinsic >= 0) return e.intrinsic;
319 int result = 0;
320 if (e.text != null) result = e.text.length();
321 else if (e.get("display").equals("flex") && e.get("flex-direction").startsWith("row")) {
322 List<Element> children = flow(e, false);
323 for (Element child : children) result += contribution(child);
324 result += e.number("column-gap") * Math.max(0, children.size() - 1);
325 } else if (e.get("display").equals("grid")) {
326 Grid g = grid(e);
327 result = Arrays.stream(trackSizes(e, g, true, -1, true)).sum() + e.number("column-gap") * (g.columns - 1);
328 } else for (Element child : flow(e, false)) result = Math.max(result, contribution(child));
329 e.intrinsic = result;
330 return result;
331 }

351 static int layout(Element e, int widthReference, int heightReference, int forcedWidth, int forcedHeight) {
352 int w = resolve(e, "width", widthReference, true);
353 e.cw = forcedWidth >= 0 ? forcedWidth : clamp(e, w < 0 ? Math.max(0, widthReference - e.hm() - e.hf()) : w, widthReference, true);
354 if (forcedHeight < 0) {
355 int h = resolve(e, "height", heightReference, false);
356 if (h >= 0) forcedHeight = clamp(e, h, heightReference, false);
357 }
358 int a;
359 if (e.text != null) {
360 e.lines = new ArrayList<>();
361 String current = "";
362 for (String word : e.text.split(" ")) {
363 if (current.isEmpty()) current = word;
364 else if (current.length() + 1 + word.length() <= e.cw) current += " " + word;
365 else {
366 e.lines.add(current);
367 current = word;
368 }
369 }
370 e.lines.add(current);
371 a = e.lines.size();
372 } else if (e.get("display").equals("flex")) a = flex(e, forcedHeight, heightReference);
373 else if (e.get("display").equals("grid")) a = gridLayout(e, forcedHeight);
374 else {
375 a = 0;
376 int previousBottom = 0;
377 boolean first = true;
378 for (Element child : flow(e, false)) {
379 layout(child, e.cw, forcedHeight, -1, -1);
380 int y = a + (first ? child.number("margin-top") : Math.max(previousBottom, child.number("margin-top")));
381 position(e, child, child.number("margin-left"), y);
382 a = y + child.bh();
383 previousBottom = child.number("margin-bottom");
384 first = false;
385 }
386 if (!first) a += previousBottom;
387 }
388 e.ch = forcedHeight >= 0 ? forcedHeight : clamp(e, a, heightReference, false);
389 return a;
390 }

392 static int flex(Element e, int k, int heightReference) {
393 boolean horizontal = e.get("flex-direction").startsWith("row"), reverse = e.get("flex-direction").endsWith("reverse");
394 int reference = horizontal ? e.cw : k, gap = e.number(horizontal ? "column-gap" : "row-gap");
395 List<FlexItem> items = new ArrayList<>();
396 for (Element child : flow(e, true)) {
397 FlexItem item = new FlexItem(child);
398 items.add(item);
399 item.outer = horizontal ? child.hm() + child.hf() : child.vm() + child.vf();
400 if (!horizontal) {
401 int r = Math.max(0, e.cw - child.hm() - child.hf()), width = resolve(child, "width", e.cw, true);
402 item.cross = clamp(child, width >= 0 ? width : align(e, child).equals("stretch") ? r : Math.min(maxContent(child), r), e.cw, true);
403 }
404 item.base = resolve(child, "flex-basis", reference, horizontal);
405 if (item.base < 0) item.base = resolve(child, horizontal ? "width" : "height", reference, horizontal);
406 if (item.base < 0) item.base = horizontal ? maxContent(child) : layout(child, e.cw, k, item.cross, -1);
407 item.hypothetical = clamp(child, item.base, reference, horizontal);
408 }
409 int length = reference;
410 if (length < 0) {
411 length = gap * Math.max(0, items.size() - 1);
412 for (FlexItem item : items) length += item.hypothetical + item.outer;
413 length = clamp(e, length, heightReference, false);
414 }
415 List<List<FlexItem>> lines = new ArrayList<>();
416 int occupied = 0;
417 for (FlexItem item : items) {
418 if (lines.isEmpty() || horizontal && e.get("flex-wrap").equals("wrap") && occupied + gap + item.hypothetical + item.outer > length) {
419 lines.add(new ArrayList<>());
420 occupied = 0;
421 }
422 List<FlexItem> line = lines.get(lines.size() - 1);
423 if (!line.isEmpty()) occupied += gap;
424 occupied += item.hypothetical + item.outer;
425 line.add(item);
426 }
427 int crossStart = 0;
428 for (List<FlexItem> line : lines) {
429 int total = gap * (line.size() - 1);
430 for (FlexItem item : line) total += item.hypothetical + item.outer;
431 boolean grow = total < length;
432 for (FlexItem item : line) {
433 item.weight = grow ? item.e.number("flex-grow") : (long)item.e.number("flex-shrink") * item.base;
434 item.size = item.hypothetical;
435 item.frozen = item.weight == 0 || (grow ? item.base > item.hypothetical : item.base < item.hypothetical);
436 }
437 while (true) {
438 int free = length - gap * (line.size() - 1), remaining = 0;
439 long[] weights = new long[line.size()];
440 for (int i = 0; i < line.size(); i++) {
441 FlexItem item = line.get(i);
442 free -= (item.frozen ? item.size : item.base) + item.outer;
443 if (!item.frozen) {
444 weights[i] = item.weight;
445 remaining++;
446 }
447 }
448 if (remaining == 0) break;
449 int[] shares = distribute(Math.max(0, grow ? free : -free), weights), violations = new int[line.size()];
450 int violation = 0;
451 for (int i = 0; i < line.size(); i++) {
452 FlexItem item = line.get(i);
453 if (!item.frozen) {
454 int tentative = item.base + (grow ? shares[i] : -shares[i]);
455 item.size = clamp(item.e, tentative, reference, horizontal);
456 violations[i] = item.size - tentative;
457 violation += violations[i];
458 }
459 }
460 for (int i = 0; i < line.size(); i++) if (violation == 0 || violation > 0 && violations[i] > 0 || violation < 0 && violations[i] < 0) line.get(i).frozen = true;
461 }
462 int lc = horizontal ? 0 : e.cw;
463 if (horizontal) {
464 for (FlexItem item : line) {
465 layout(item.e, e.cw, k, item.size, -1);
466 item.cross = item.e.ch;
467 lc = Math.max(lc, item.cross + item.e.vf() + item.e.vm());
468 }
469 if (lines.size() == 1) lc = k >= 0 ? k : clamp(e, lc, heightReference, false);
470 }
471 int used = gap * (line.size() - 1);
472 for (FlexItem item : line) used += item.size + item.outer;
473 int space = length - used;
474 int[] before = new int[line.size() + 1];
475 if (space > 0) {
476 String justify = e.get("justify-content");
477 if (justify.equals("end")) before[0] = space;
478 else if (justify.equals("center")) before[0] = space / 2;
479 else if (justify.startsWith("space-")) {
480 long[] weights = new long[before.length];
481 Arrays.fill(weights, 1);
482 if (justify.equals("space-between")) {
483 weights[0] = 0;
484 weights[weights.length - 1] = 0;
485 } else if (justify.equals("space-around")) for (int i = 1; i < weights.length - 1; i++) weights[i] = 2;
486 before = distribute(space, weights);
487 }
488 }
489 int cursor = 0;
490 for (int i = 0; i < line.size(); i++) {
491 FlexItem item = line.get(i);
492 Element child = item.e;
493 String alignment = align(e, child);
494 if (horizontal) {
495 if (alignment.equals("stretch") && child.get("height").equals("auto")) {
496 item.cross = clamp(child, Math.max(0, lc - child.vm() - child.vf()), k, false);
497 layout(child, e.cw, k, item.size, item.cross);
498 } else layout(child, e.cw, k, item.size, -1);
499 } else layout(child, e.cw, k, item.cross, item.size);
500 int cross = crossStart + offset(alignment, lc - item.cross - (horizontal ? child.vf() + child.vm() : child.hf() + child.hm())) + child.number(horizontal ? "margin-top" : "margin-left");
501 cursor += before[i];
502 int main = reverse ? length - cursor - child.number(horizontal ? "margin-right" : "margin-bottom") - item.size - (horizontal ? child.hf() : child.vf()) : cursor + child.number(horizontal ? "margin-left" : "margin-top");
503 position(e, child, horizontal ? main : cross, horizontal ? cross : main);
504 cursor += item.size + item.outer + gap;
505 }
506 crossStart += lc + e.number("row-gap");
507 }
508 return horizontal ? lines.isEmpty() ? 0 : crossStart - e.number("row-gap") : length;
509 }

523 static int gridLayout(Element e, int k) {
524 Grid g = grid(e);
525 int cg = e.number("column-gap"), rg = e.number("row-gap");
526 int[] columns = trackSizes(e, g, true, e.cw, true);
527 for (GridItem item : g.items) {
528 item.aw = area(columns, item.column, item.columns, cg);
529 item.ax = prefix(columns, item.column - 1, cg);
530 int width = resolve(item.e, "width", item.aw, true);
531 item.width = clamp(item.e, width < 0 ? Math.max(0, item.aw - item.e.hm() - item.e.hf()) : width, item.aw, true);
532 layout(item.e, item.aw, -1, item.width, -1);
533 item.contribution = item.e.bh() + item.e.vm();
534 }
535 int[] rows = trackSizes(e, g, false, k, false);
536 for (GridItem item : g.items) {
537 int ah = area(rows, item.row, item.rows, rg), ay = prefix(rows, item.row - 1, rg);
538 String alignment = align(e, item.e);
539 int fh = alignment.equals("stretch") && item.e.get("height").equals("auto") ? clamp(item.e, Math.max(0, ah - item.e.vm() - item.e.vf()), ah, false) : -1;
540 layout(item.e, item.aw, ah, item.width, fh);
541 position(e, item.e, item.ax + item.e.number("margin-left"), ay + offset(alignment, ah - item.e.bh() - item.e.vm()) + item.e.number("margin-top"));
542 }
543 return Arrays.stream(rows).sum() + rg * Math.max(0, g.rows - 1);
544 }

546 static void relayout() {
547 for (Element e : elements) {
548 e.displayed = !e.get("display").equals("none") && (e.parent == null || e.parent.displayed);
549 e.intrinsic = -1;
550 e.containing = null;
551 }
552 Element root = elements.get(0);
553 if (root.displayed && !root.get("position").equals("absolute")) {
554 layout(root, viewportWidth, viewportHeight, -1, -1);
555 root.rx = root.number("margin-left");
556 root.ry = root.number("margin-top");
557 }
558 for (Element e : elements) if (e.displayed && e.get("position").equals("absolute")) {
559 Element cb = e.parent;
560 while (cb != null && cb.get("position").equals("static")) cb = cb.parent;
561 e.containing = cb;
562 int width = cb == null ? viewportWidth : cb.cw + cb.number("padding-left") + cb.number("padding-right"), height = cb == null ? viewportHeight : cb.ch + cb.number("padding-top") + cb.number("padding-bottom");
563 int left = e.get("left").equals("auto") ? 0 : e.number("left"), right = e.get("right").equals("auto") ? 0 : e.number("right"), top = e.get("top").equals("auto") ? 0 : e.number("top"), bottom = e.get("bottom").equals("auto") ? 0 : e.number("bottom");
564 int w = resolve(e, "width", width, true);
565 if (e.get("width").equals("auto")) {
566 int r = Math.max(0, width - left - right - e.hm() - e.hf());
567 w = !e.get("left").equals("auto") && !e.get("right").equals("auto") ? r : Math.min(maxContent(e), r);
568 }
569 int h = e.get("height").equals("auto") && !e.get("top").equals("auto") && !e.get("bottom").equals("auto") ? clamp(e, Math.max(0, height - top - bottom - e.vm() - e.vf()), height, false) : -1;
570 layout(e, width, height, clamp(e, w, width, true), h);
571 e.rx = !e.get("left").equals("auto") ? left + e.number("margin-left") : !e.get("right").equals("auto") ? width - right - e.number("margin-right") - e.bw() : e.number("margin-left");
572 e.ry = !e.get("top").equals("auto") ? top + e.number("margin-top") : !e.get("bottom").equals("auto") ? height - bottom - e.number("margin-bottom") - e.bh() : e.number("margin-top");
573 }
574 for (Element e : elements) if (e.displayed) {
575 Element ref = e.get("position").equals("absolute") ? e.containing : e.parent;
576 e.x = e.rx + (ref == null ? 0 : ref.x + (e.get("position").equals("absolute") ? ref.number("border-left") : 0));
577 e.y = e.ry + (ref == null ? 0 : ref.y + (e.get("position").equals("absolute") ? ref.number("border-top") : 0));
578 if (e.get("position").equals("relative")) {
579 e.x += !e.get("left").equals("auto") ? e.number("left") : !e.get("right").equals("auto") ? -e.number("right") : 0;
580 e.y += !e.get("top").equals("auto") ? e.number("top") : !e.get("bottom").equals("auto") ? -e.number("bottom") : 0;
581 }
582 }
583 }

585 static int[][] snapshot() {
586 int[][] result = new int[elements.size()][5];
587 for (Element e : elements) result[e.index] = new int[]{e.displayed ? 1 : 0, e.x, e.y, e.bw(), e.bh()};
588 return result;

591 static String changes(int[][] old) {
592 List<String> changed = new ArrayList<>();
593 int[][] now = snapshot();
594 for (Element e : elements) if (old[e.index][0] != now[e.index][0] || e.displayed && !Arrays.equals(old[e.index], now[e.index])) changed.add(e.id);
595 return changed.isEmpty() ? "no change" : String.join(" ", changed);
596 }

602 static void render(StringBuilder out) {
603 char[][] screen = new char[viewportHeight][viewportWidth];
604 for (char[] row : screen) Arrays.fill(row, '.');
605 for (Element e : elements) if (e.displayed) {
606 for (int y = Math.max(0, e.y); y < Math.min(viewportHeight, e.y + e.bh()); y++) for (int x = Math.max(0, e.x); x < Math.min(viewportWidth, e.x + e.bw()); x++) {
607 boolean horizontal = e.number("border-top") == 1 && y == e.y || e.number("border-bottom") == 1 && y == e.y + e.bh() - 1, vertical = e.number("border-left") == 1 && x == e.x || e.number("border-right") == 1 && x == e.x + e.bw() - 1;
608 screen[y][x] = horizontal ? vertical ? '+' : '-' : vertical ? '|' : ' ';
609 }
610 if (e.text != null) for (int i = 0; i < e.lines.size(); i++) {
611 int y = e.cy() + i;
612 if (y < 0 || y >= viewportHeight) continue;
613 String line = e.lines.get(i);
614 for (int j = 0; j < line.length(); j++) {
615 int x = e.cx() + j;
616 if (x >= 0 && x < viewportWidth) screen[y][x] = line.charAt(j);
617 }
618 }
619 }
620 String edge = "#".repeat(viewportWidth + 2);
621 out.append("render ").append(viewportWidth).append('x').append(viewportHeight).append('\n').append(edge).append('\n');
622 for (char[] row : screen) out.append('#').append(row).append("#\n");
623 out.append(edge).append('\n');
624 }

626 static void command(String line, StringBuilder out) {
627 String[] t = line.split(" ");
628 Element e = t.length > 1 ? elements.stream().filter(a -> a.id.equals(t[1])).findFirst().orElse(null) : null;
629 switch (t[0]) {
630 case "LAYOUT" -> {
631 out.append("layout ").append(viewportWidth).append('x').append(viewportHeight).append('\n');
632 for (Element a : elements) out.append(" ".repeat(a.depth)).append(a.displayed ? rectangle(a.id, a.x, a.y, a.bw(), a.bh()) : a.id + " none\n");
633 }
634 case "BOX" -> {
635 if (e == null) out.append("error: unknown element ").append(t[1]).append('\n');
636 else if (!e.displayed) out.append("box ").append(e.id).append(": none\n");
637 else {
638 out.append("box ").append(e.id).append('\n');
639 out.append(rectangle("margin", e.x - e.number("margin-left"), e.y - e.number("margin-top"), e.bw() + e.hm(), e.bh() + e.vm()));
640 out.append(rectangle("border", e.x, e.y, e.bw(), e.bh()));
641 out.append(rectangle("padding", e.x + e.number("border-left"), e.y + e.number("border-top"), e.cw + e.number("padding-left") + e.number("padding-right"), e.ch + e.number("padding-top") + e.number("padding-bottom")));
642 out.append(rectangle("content", e.cx(), e.cy(), e.cw, e.ch));
643 }
644 }
645 case "RENDER" -> render(out);
646 case "HIT" -> {
647 int x = Integer.parseInt(t[1]), y = Integer.parseInt(t[2]);
648 if (x >= viewportWidth || y >= viewportHeight) out.append("error: point ").append(x).append(' ').append(y).append(" outside viewport\n");
649 else {
650 Element hit = null;
651 for (Element a : elements) if (a.displayed && x >= a.x && x < a.x + a.bw() && y >= a.y && y < a.y + a.bh()) hit = a;
652 List<String> path = new ArrayList<>();
653 for (; hit != null; hit = hit.parent) path.add(0, hit.id);
654 out.append("hit ").append(x).append(' ').append(y).append(": ").append(path.isEmpty() ? "none" : String.join(" > ", path)).append('\n');
655 }
656 }
657 case "SET" -> {
658 String value = line.substring(t[0].length() + t[1].length() + t[2].length() + 3);
659 if (e == null) out.append("error: unknown element ").append(t[1]).append('\n');
660 else if (!known(t[2])) out.append("error: unknown property ").append(t[2]).append('\n');
661 else {
662 Map<String, String> values = declaration(t[2], value);
663 if (values == null) out.append("error: invalid value for ").append(t[2]).append(": ").append(value).append('\n');
664 else {
665 int[][] old = snapshot();
666 e.style.putAll(values);
667 relayout();
668 out.append("set ").append(e.id).append(' ').append(t[2]).append(' ').append(value).append(": ").append(changes(old)).append('\n');
669 }
670 }
671 }
672 case "VIEWPORT" -> {
673 int w = Integer.parseInt(t[1]), h = Integer.parseInt(t[2]);
674 if (w < 1 || w > 80 || h < 1 || h > 40) out.append("error: invalid viewport ").append(w).append(' ').append(h).append('\n');
675 else {
676 int[][] old = snapshot();
677 viewportWidth = w;
678 viewportHeight = h;
679 relayout();
680 out.append("viewport ").append(w).append(' ').append(h).append(": ").append(changes(old)).append('\n');
681 }
682 }
683 case "OVERFLOW" -> {
684 List<String> list = new ArrayList<>();
685 for (Element a : elements) if (a.displayed) {
686 boolean overflow = a.text != null && (a.lines.size() > a.ch || a.lines.stream().anyMatch(s -> s.length() > a.cw));
687 if (a.text == null) for (Element child : flow(a, false)) if (child.x - child.number("margin-left") < a.cx() || child.y - child.number("margin-top") < a.cy() || child.x + child.bw() + child.number("margin-right") > a.cx() + a.cw || child.y + child.bh() + child.number("margin-bottom") > a.cy() + a.ch) overflow = true;
688 if (overflow) list.add(a.id);
689 }
690 out.append("overflow: ").append(list.isEmpty() ? "none" : String.join(" ", list)).append('\n');
691 }
692 default -> out.append("error: unknown command ").append(t[0]).append('\n');
693 }
694 }

696 public static void main(String[] args) throws Exception {
697 BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
698 String[] dimensions = reader.readLine().split(" ");
699 viewportWidth = Integer.parseInt(dimensions[0]);
700 viewportHeight = Integer.parseInt(dimensions[1]);
701 List<Rule> rules = new ArrayList<>();
702 String line;
703 while (!(line = reader.readLine()).equals("---")) {
704 int brace = line.indexOf(" { ");
705 List<Declaration> declarations = new ArrayList<>();
706 for (String raw : line.substring(brace + 3, line.length() - 2).split("; ")) {
707 int colon = raw.indexOf(": ");
708 Map<String, String> values = declaration(raw.substring(0, colon), raw.substring(colon + 2));
709 if (values != null) for (Map.Entry<String, String> entry : values.entrySet()) declarations.add(new Declaration(entry.getKey(), entry.getValue()));
710 }
711 rules.add(new Rule(line.substring(0, brace).split(", "), declarations));
712 }
713 Element[] ancestors = new Element[11];
714 while (!(line = reader.readLine()).equals("===")) {
715 int spaces = 0;
716 while (line.charAt(spaces) == ' ') spaces++;
717 Element e = new Element();
718 e.depth = spaces / 2;
719 e.index = elements.size();
720 String body = line.substring(spaces);
721 int end = body.indexOf(' ');
722 String head = end < 0 ? body : body.substring(0, end);
723 if (end >= 0) e.text = body.substring(end + 2, body.length() - 1);
724 String[] parts = head.split("[#.]");
725 e.type = parts[0];
726 e.id = parts[1];
727 for (int i = 2; i < parts.length; i++) e.classes.add(parts[i]);
728 if (e.depth > 0) {
729 e.parent = ancestors[e.depth - 1];
730 e.parent.children.add(e);
731 }
732 ancestors[e.depth] = e;
733 elements.add(e);
734 Map<String, Integer> scores = new HashMap<>();
735 for (Rule rule : rules) {
736 int score = -1;
737 for (String selector : rule.selectors) {
738 if (selector.startsWith("#") && selector.substring(1).equals(e.id)) score = Math.max(score, 2);
739 else if (selector.startsWith(".") && e.classes.contains(selector.substring(1))) score = Math.max(score, 1);
740 else if (selector.equals(e.type)) score = Math.max(score, 0);
741 }
742 if (score >= 0) for (Declaration d : rule.declarations) if (score >= scores.getOrDefault(d.property, -1)) {
743 e.style.put(d.property, d.value);
744 scores.put(d.property, score);
745 }
746 }
747 }
748 relayout();
749 while (!(line = reader.readLine()).equals("END")) {
750 StringBuilder out = new StringBuilder();
751 command(line, out);
752 System.out.print(out);
753 }
754 }
