70func contains(value string, choices ...string) bool {
71 for _, choice := range choices {
72 if value == choice {
73 return true
74 }
75 }
76 return false

78func unsigned(s string) bool {
79 if len(s) < 1 || len(s) > 3 {
80 return false
81 }
82 for _, c := range s {
83 if c < '0' || c > '9' {
84 return false
85 }
86 }
87 return true
88}
89func positive(s string) bool { return unsigned(s) && number(s) > 0 }
90func signed(s string) bool { return unsigned(s) || strings.HasPrefix(s, "-") && unsigned(s[1:]) }
91func percentage(s string) bool { return strings.HasSuffix(s, "%") && unsigned(s[:len(s)-1]) }
92func gridValue(value string) bool {
93 parts := strings.Split(value, " ")
94 if len(parts) == 1 {
95 return parts[0] == "auto" || positive(parts[0])
96 }
97 if len(parts) == 2 {
98 return parts[0] == "span" && positive(parts[1])
99 }
100 if len(parts) == 3 {
101 return positive(parts[0]) && parts[1] == "/" && positive(parts[2]) && number(parts[0]) < number(parts[2])
102 }
103 return len(parts) == 4 && positive(parts[0]) && parts[1] == "/" && parts[2] == "span" && positive(parts[3])
104}
105func valid(property, value string) ([]declaration, bool) {
106 parts := strings.Split(value, " ")
107 if contains(property, "margin", "padding", "border", "gap") {
108 limit := 4
109 if property == "gap" {
110 limit = 2
111 }
112 if len(parts) < 1 || len(parts) > limit {
113 return nil, false
114 }
115 for _, part := range parts {
116 if !unsigned(part) || property == "border" && !contains(part, "0", "1") {
117 return nil, false
118 }
119 }
120 if property == "gap" {
121 second := parts[0]
122 if len(parts) == 2 {
123 second = parts[1]
124 }
125 return []declaration{{"row-gap", parts[0]}, {"column-gap", second}}, true
126 }
127 values := [4]string{parts[0], parts[0], parts[0], parts[0]}
128 if len(parts) > 1 {
129 values[1], values[3] = parts[1], parts[1]
130 }
131 if len(parts) > 2 {
132 values[2] = parts[2]
133 }
134 if len(parts) > 3 {
135 values[3] = parts[3]
136 }
137 result := []declaration{}
138 for i, side := range []string{"top", "right", "bottom", "left"} {
139 result = append(result, declaration{property + "-" + side, values[i]})
140 }
141 return result, true
142 }
143 if _, known := initial[property]; !known {
144 return nil, false
145 }
146 good := false
147 if strings.HasPrefix(property, "grid-template-") {
148 good = value == "none"
149 if !good {
150 good = true
151 for _, part := range parts {
152 if !unsigned(part) && part != "auto" && !(strings.HasSuffix(part, "fr") && positive(part[:len(part)-2])) {
153 good = false
154 break
155 }
156 }
157 }
158 } else if property == "grid-column" || property == "grid-row" {
159 good = gridValue(value)
160 } else if len(parts) == 1 {
161 switch property {
162 case "display":
163 good = contains(value, "block", "flex", "grid", "none")
164 case "position":
165 good = contains(value, "static", "relative", "absolute")
166 case "box-sizing":
167 good = contains(value, "content-box", "border-box")
168 case "width", "height", "flex-basis":
169 good = value == "auto" || unsigned(value) || percentage(value)
170 case "min-width", "min-height":
171 good = unsigned(value) || percentage(value)
172 case "max-width", "max-height":
173 good = value == "none" || unsigned(value) || percentage(value)
174 case "top", "right", "bottom", "left":
175 good = value == "auto" || signed(value)
176 case "flex-direction":
177 good = contains(value, "row", "row-reverse", "column", "column-reverse")
178 case "flex-wrap":
179 good = contains(value, "nowrap", "wrap")
180 case "justify-content":
181 good = contains(value, "start", "end", "center", "space-between", "space-around", "space-evenly")
182 case "align-items":
183 good = contains(value, "stretch", "start", "end", "center")
184 case "align-self":
185 good = contains(value, "auto", "stretch", "start", "end", "center")
186 case "order":
187 good = signed(value)
188 default:
189 if strings.HasPrefix(property, "border-") {
190 good = contains(value, "0", "1")
191 } else {
192 good = unsigned(value)
193 }
194 }
195 }
196 if !good {
197 return nil, false
198 }
199 return []declaration{{property, value}}, true
200}
201func (e *element) n(property string) int { return number(e.style[property]) }
202func (e *element) frame(horizontal bool) int {
203 if horizontal {
204 return e.n("padding-left") + e.n("padding-right") + e.n("border-left") + e.n("border-right")
205 }
206 return e.n("padding-top") + e.n("padding-bottom") + e.n("border-top") + e.n("border-bottom")
207}
208func (e *element) margin(horizontal bool) int {
209 if horizontal {
210 return e.n("margin-left") + e.n("margin-right")
211 }
212 return e.n("margin-top") + e.n("margin-bottom")
213}
214func (e *element) resolve(property string, reference *int, horizontal bool) *int {
215 value := e.style[property]
216 if value == "auto" || value == "none" {
217 return nil
218 }
219 n := 0
220 if strings.HasSuffix(value, "%") {
221 if reference == nil {
222 return nil
223 }
224 n = *reference * number(value[:len(value)-1]) / 100
225 } else {
226 n = number(value)
227 }
228 if e.style["box-sizing"] == "border-box" {
229 n = max(0, n-e.frame(horizontal))
230 }
231 return pointer(n)
232}
233func (e *element) clamp(value int, reference *int, horizontal bool) int {
234 axis := "height"
235 if horizontal {
236 axis = "width"
237 }
238 lower, upper := e.resolve("min-"+axis, reference, horizontal), e.resolve("max-"+axis, reference, horizontal)
239 if upper != nil {
240 value = min(value, *upper)
241 }
242 if lower != nil {
243 value = max(value, *lower)
244 }
245 return value
246}

281func distribute(total int, weights []int) []int {
282 result := make([]int, len(weights))
283 sum := 0
284 for _, w := range weights {
285 sum += w
286 }
287 if sum == 0 {
288 return result
289 }
290 used := 0
291 for i, w := range weights {
292 result[i] = total * w / sum
293 used += result[i]
294 }
295 for i, w := range weights {
296 if used == total {
297 break
298 }
299 if w > 0 {
300 result[i]++
301 used++
302 }
303 }
304 return result
305}
306func gridSpec(value string) (int, int) {
307 p := strings.Split(value, " ")
308 if value == "auto" {
309 return 0, 1
310 }
311 if p[0] == "span" {
312 return 0, number(p[1])
313 }
314 start := number(p[0])
315 if len(p) == 1 {
316 return start, 1
317 }
318 if len(p) == 3 {
319 return start, number(p[2]) - start
320 }
321 return start, number(p[3])
322}
323func tracks(value string, columns bool) []track {
324 if value == "none" {
325 if columns {
326 return []track{{kind: "auto"}}
327 }
328 return nil
329 }
330 result := []track{}
331 for _, word := range strings.Split(value, " ") {
332 t := track{kind: "fixed", value: number(word)}
333 if word == "auto" {
334 t.kind = "auto"
335 } else if strings.HasSuffix(word, "fr") {
336 t.kind = "fr"
337 t.value = number(word[:len(word)-2])
338 }
339 result = append(result, t)
340 }
341 return result
342}
343func (d *document) placement(e *element) ([]placement, int, int) {
344 columns := len(tracks(e.style["grid-template-columns"], true))
345 rows := len(tracks(e.style["grid-template-rows"], false))
346 items := []placement{}
347 for _, child := range e.items() {
348 c, cs := gridSpec(child.style["grid-column"])
349 r, rs := gridSpec(child.style["grid-row"])
350 items = append(items, placement{child, r, c, rs, cs})
351 if c > 0 {
352 columns = max(columns, c+cs-1)
353 } else {
354 columns = max(columns, cs)
355 }
356 }
357 occupied := map[[2]int]bool{}
358 overlaps := func(p placement, r, c int) bool {
359 for y := r; y < r+p.rows; y++ {
360 for x := c; x < c+p.cols; x++ {
361 if occupied[[2]int{y, x}] {
362 return true
363 }
364 }
365 }
366 return false
367 }
368 mark := func(p placement) {
369 for y := p.row; y < p.row+p.rows; y++ {
370 for x := p.col; x < p.col+p.cols; x++ {
371 occupied[[2]int{y, x}] = true
372 }
373 }
374 }
375 for _, p := range items {
376 if p.row > 0 && p.col > 0 {
377 mark(p)
378 }
379 }
380 cr, cc := 1, 1
381 for i := range items {
382 p := &items[i]
383 if p.row > 0 && p.col > 0 {
384 continue
385 }
386 if p.row > 0 {
387 p.col = 1
388 for c := 1; c <= columns-p.cols+1; c++ {
389 if !overlaps(*p, p.row, c) {
390 p.col = c
391 break
392 }
393 }
394 } else if p.col > 0 {
395 if p.col < cc {
396 cr++
397 }
398 r := cr
399 for overlaps(*p, r, p.col) {
400 r++
401 }
402 p.row, cr, cc = r, r, p.col+p.cols
403 } else {
404 r, c := cr, cc
405 for {
406 if c+p.cols-1 > columns {
407 r++
408 c = 1
409 } else if overlaps(*p, r, c) {
410 c++
411 } else {
412 break
413 }
414 }
415 p.row, p.col, cr, cc = r, c, r, c+p.cols
416 }
417 mark(*p)
418 }
419 for _, p := range items {
420 rows = max(rows, p.row+p.rows-1)
421 }
422 return items, columns, rows
423}
424func trackSizes(spec []track, count int, available *int, gap int, items []placement, contributions []int, horizontal bool) []int {
425 sizes, weights := make([]int, count), make([]int, count)
426 kinds := make([]string, count)
427 hasFR := false
428 for i := 0; i < count; i++ {
429 t := track{kind: "auto"}
430 if i < len(spec) {
431 t = spec[i]
432 }
433 kinds[i] = t.kind
434 if t.kind == "fixed" {
435 sizes[i] = t.value
436 } else if t.kind == "auto" || available == nil {
437 for j, p := range items {
438 start, span := p.col, p.cols
439 if !horizontal {
440 start, span = p.row, p.rows
441 }
442 if start == i+1 && span == 1 {
443 sizes[i] = max(sizes[i], contributions[j])
444 }
445 }
446 }
447 if t.kind == "fr" {
448 weights[i], hasFR = t.value, true
449 }
450 }
451 if available != nil && count > 0 {
452 left := *available - sum(sizes) - gap*(count-1)
453 if left > 0 {
454 if !hasFR {
455 for i, kind := range kinds {
456 if kind == "auto" {
457 weights[i] = 1
458 }
459 }
460 }
461 extra := distribute(left, weights)
462 for i := range sizes {
463 sizes[i] += extra[i]
464 }
465 }
466 }
467 return sizes
468}

476func (d *document) contribution(e *element) int {
477 value := e.resolve("width", nil, true)
478 width := 0
479 if value == nil {
480 width = d.maxContent(e)
481 } else {
482 width = *value
483 }
484 return e.clamp(width, nil, true) + e.frame(true) + e.margin(true)
485}
486func (d *document) maxContent(e *element) int {
487 if e.text != "" {
488 return len(e.text)
489 }
490 children := e.flow()
491 if e.style["display"] == "flex" && strings.HasPrefix(e.style["flex-direction"], "row") {
492 value := 0
493 for _, child := range children {
494 value += d.contribution(child)
495 }
496 if len(children) > 0 {
497 value += e.n("column-gap") * (len(children) - 1)
498 }
499 return value
500 }
501 if e.style["display"] == "grid" {
502 items, nc, _ := d.placement(e)
503 values := make([]int, len(items))
504 for i, p := range items {
505 values[i] = d.contribution(p.e)
506 }
507 sizes := trackSizes(tracks(e.style["grid-template-columns"], true), nc, nil, e.n("column-gap"), items, values, true)
508 return sum(sizes) + e.n("column-gap")*(nc-1)
509 }
510 value := 0
511 for _, child := range children {
512 value = max(value, d.contribution(child))
513 }
514 return value
515}
516func wrap(text string, width int) []string {
517 lines := []string{}
518 for _, word := range strings.Split(text, " ") {
519 if len(lines) == 0 || len(lines[len(lines)-1])+1+len(word) > width {
520 lines = append(lines, word)
521 } else {
522 lines[len(lines)-1] += " " + word
523 }
524 }
525 return lines
526}
527func (d *document) layoutElement(e *element, cw int, ch, fw, fh *int) int {
528 if fw != nil {
529 e.cw = *fw
530 } else {
531 value := e.resolve("width", pointer(cw), true)
532 width := max(0, cw-e.margin(true)-e.frame(true))
533 if value != nil {
534 width = *value
535 }
536 e.cw = e.clamp(width, pointer(cw), true)
537 }
538 if fh == nil {
539 if value := e.resolve("height", ch, false); value != nil {
540 fh = pointer(e.clamp(*value, ch, false))
541 }
542 }
543 a := 0
544 e.lines = nil
545 if e.text != "" {
546 e.lines = wrap(e.text, e.cw)
547 a = len(e.lines)
548 } else {
549 switch e.style["display"] {
550 case "flex":
551 a = d.flex(e, fh, ch)
552 case "grid":
553 a = d.grid(e, fh)
554 default:
555 a = d.block(e, fh)
556 }
557 }
558 if fh != nil {
559 e.ch = *fh
560 } else {
561 e.ch = e.clamp(a, ch, false)
562 }
563 return a
564}
565func position(e, child *element, x, y int) {
566 child.rx = x + e.n("border-left") + e.n("padding-left")
567 child.ry = y + e.n("border-top") + e.n("padding-top")
568}
569func (d *document) block(e *element, k *int) int {
570 bottom, previous, count := 0, 0, 0
571 for _, child := range e.flow() {
572 d.layoutElement(child, e.cw, k, nil, nil)
573 y := child.n("margin-top")
574 if count > 0 {
575 y = bottom + max(previous, y)
576 }
577 position(e, child, child.n("margin-left"), y)
578 bottom, previous = y+child.ch+child.frame(false), child.n("margin-bottom")
579 count++
580 }
581 if count == 0 {
582 return 0
583 }
584 return bottom + previous
585}

593func (d *document) flex(e *element, k, ch *int) int {
594 horizontal := strings.HasPrefix(e.style["flex-direction"], "row")
595 reversed := strings.HasSuffix(e.style["flex-direction"], "reverse")
596 mainRef, gap := k, e.n("row-gap")
597 if horizontal {
598 mainRef, gap = pointer(e.cw), e.n("column-gap")
599 }
600 items := []*flexItem{}
601 for _, child := range e.items() {
602 item := &flexItem{e: child, outer: child.frame(horizontal) + child.margin(horizontal)}
603 if !horizontal {
604 r := max(0, e.cw-child.frame(true)-child.margin(true))
605 c := r
606 if child.style["width"] != "auto" {
607 c = *child.resolve("width", pointer(e.cw), true)
608 } else if child.align(e) != "stretch" {
609 c = min(d.maxContent(child), r)
610 }
611 item.cross = child.clamp(c, pointer(e.cw), true)
612 }
613 base := child.resolve("flex-basis", mainRef, horizontal)
614 if base == nil {
615 property := "height"
616 if horizontal {
617 property = "width"
618 }
619 base = child.resolve(property, mainRef, horizontal)
620 }
621 if base != nil {
622 item.base = *base
623 } else if horizontal {
624 item.base = d.maxContent(child)
625 } else {
626 item.base = d.layoutElement(child, e.cw, k, pointer(item.cross), nil)
627 }
628 item.hypothetical = child.clamp(item.base, mainRef, horizontal)
629 items = append(items, item)
630 }
631 length := 0
632 if horizontal {
633 length = e.cw
634 } else if k != nil {
635 length = *k
636 } else {
637 for _, item := range items {
638 length += item.hypothetical + item.outer
639 }
640 if len(items) > 0 {
641 length += gap * (len(items) - 1)
642 }
643 length = e.clamp(length, ch, false)
644 }
645 lines := [][]*flexItem{}
646 used := 0
647 for _, item := range items {
648 if len(lines) == 0 {
649 lines = append(lines, []*flexItem{})
650 used = 0
651 }
652 last := len(lines) - 1
653 if horizontal && e.style["flex-wrap"] == "wrap" && len(lines[last]) > 0 && used+gap+item.hypothetical+item.outer > length {
654 lines = append(lines, []*flexItem{})
655 last++
656 used = 0
657 }
658 if len(lines[last]) > 0 {
659 used += gap
660 }
661 lines[last] = append(lines[last], item)
662 used += item.hypothetical + item.outer
663 }
664 crossStart, totalCross := 0, 0
665 for _, line := range lines {
666 gaps := gap * (len(line) - 1)
667 hypothetical := gaps
668 for _, item := range line {
669 hypothetical += item.hypothetical + item.outer
670 }
671 growing := hypothetical < length
672 for _, item := range line {
673 if growing {
674 item.weight = item.e.n("flex-grow")
675 } else {
676 item.weight = item.e.n("flex-shrink") * item.base
677 }
678 item.frozen = item.weight == 0 || growing && item.base > item.hypothetical || !growing && item.base < item.hypothetical
679 item.size = item.hypothetical
680 }
681 for {
682 free, unfrozen := length-gaps, 0
683 weights := make([]int, len(line))
684 for i, item := range line {
685 if item.frozen {
686 free -= item.size + item.outer
687 } else {
688 free -= item.base + item.outer
689 weights[i] = item.weight
690 unfrozen++
691 }
692 }
693 if unfrozen == 0 {
694 break
695 }
696 amount := max(0, free)
697 if !growing {
698 amount = max(0, -free)
699 }
700 assignments := distribute(amount, weights)
701 violations := make([]int, len(line))
702 totalViolation := 0
703 for i, item := range line {
704 if item.frozen {
705 continue
706 }
707 tentative := item.base + assignments[i]
708 if !growing {
709 tentative = item.base - assignments[i]
710 }
711 item.size = item.e.clamp(tentative, mainRef, horizontal)
712 violations[i] = item.size - tentative
713 totalViolation += violations[i]
714 }
715 for i, item := range line {
716 if !item.frozen && (totalViolation == 0 || totalViolation > 0 && violations[i] > 0 || totalViolation < 0 && violations[i] < 0) {
717 item.frozen = true
718 }
719 }
720 }
721 lc := e.cw
722 if horizontal {
723 lc = 0
724 for _, item := range line {
725 d.layoutElement(item.e, e.cw, k, pointer(item.size), nil)
726 item.cross = item.e.ch
727 lc = max(lc, item.cross+item.e.frame(false)+item.e.margin(false))
728 }
729 if len(lines) == 1 {
730 if k != nil {
731 lc = *k
732 } else {
733 lc = e.clamp(lc, ch, false)
734 }
735 }
736 }
737 remaining := length - gaps
738 for _, item := range line {
739 remaining -= item.size + item.outer
740 }
741 spaces := make([]int, len(line))
742 if remaining > 0 {
743 switch e.style["justify-content"] {
744 case "end":
745 spaces[0] = remaining
746 case "center":
747 spaces[0] = remaining / 2
748 case "space-between", "space-around", "space-evenly":
749 weights := make([]int, len(line)+1)
750 for i := range weights {
751 switch e.style["justify-content"] {
752 case "space-between":
753 if i > 0 && i < len(line) {
754 weights[i] = 1
755 }
756 case "space-around":
757 weights[i] = 2
758 if i == 0 || i == len(line) {
759 weights[i] = 1
760 }
761 default:
762 weights[i] = 1
763 }
764 }
765 copy(spaces, distribute(remaining, weights))
766 }
767 }
768 cursor := 0
769 for i, item := range line {
770 child, c := item.e, item.cross
771 align := child.align(e)
772 if horizontal {
773 if align == "stretch" && child.style["height"] == "auto" {
774 c = child.clamp(max(0, lc-child.frame(false)-child.margin(false)), k, false)
775 d.layoutElement(child, e.cw, k, pointer(item.size), pointer(c))
776 } else {
777 d.layoutElement(child, e.cw, k, pointer(item.size), nil)
778 c = child.ch
779 }
780 } else {
781 d.layoutElement(child, e.cw, k, pointer(c), pointer(item.size))
782 }
783 q := lc - c - child.frame(!horizontal) - child.margin(!horizontal)
784 offset := 0
785 if q > 0 {
786 if align == "end" {
787 offset = q
788 } else if align == "center" {
789 offset = q / 2
790 }
791 }
792 cursor += spaces[i]
793 side := "margin-top"
794 if horizontal {
795 side = "margin-left"
796 }
797 if reversed {
798 if horizontal {
799 side = "margin-right"
800 } else {
801 side = "margin-bottom"
802 }
803 }
804 main := cursor + child.n(side)
805 if reversed {
806 main = length - cursor - child.n(side) - item.size - child.frame(horizontal)
807 }
808 cross := crossStart + offset + child.n("margin-top")
809 if !horizontal {
810 cross = offset + child.n("margin-left")
811 }
812 if horizontal {
813 position(e, child, main, cross)
814 } else {
815 position(e, child, cross, main)
816 }
817 cursor += item.size + item.outer + gap
818 }
819 crossStart += lc + e.n("row-gap")
820 totalCross += lc
821 }
822 if !horizontal {
823 return length
824 }
825 if len(lines) > 0 {
826 totalCross += e.n("row-gap") * (len(lines) - 1)
827 }
828 return totalCross
829}
830func area(sizes []int, start, span, gap int) (int, int) {
831 return sum(sizes[start-1:start-1+span]) + gap*(span-1), sum(sizes[:start-1]) + gap*(start-1)
832}
833func (d *document) grid(e *element, k *int) int {
834 items, nc, nr := d.placement(e)
835 contributions := make([]int, len(items))
836 for i, p := range items {
837 contributions[i] = d.contribution(p.e)
838 }
839 columns := trackSizes(tracks(e.style["grid-template-columns"], true), nc, pointer(e.cw), e.n("column-gap"), items, contributions, true)
840 widths, positions, actual := make([]int, len(items)), make([]int, len(items)), make([]int, len(items))
841 for i, p := range items {
842 aw, ax := area(columns, p.col, p.cols, e.n("column-gap"))
843 value := p.e.resolve("width", pointer(aw), true)
844 gw := max(0, aw-p.e.frame(true)-p.e.margin(true))
845 if value != nil {
846 gw = *value
847 }
848 gw = p.e.clamp(gw, pointer(aw), true)
849 widths[i], positions[i], actual[i] = aw, ax, gw
850 d.layoutElement(p.e, aw, nil, pointer(gw), nil)
851 contributions[i] = p.e.ch + p.e.frame(false) + p.e.margin(false)
852 }
853 rows := trackSizes(tracks(e.style["grid-template-rows"], false), nr, k, e.n("row-gap"), items, contributions, false)
854 for i, p := range items {
855 ah, ay := area(rows, p.row, p.rows, e.n("row-gap"))
856 child := p.e
857 var fh *int
858 if child.align(e) == "stretch" && child.style["height"] == "auto" {
859 fh = pointer(child.clamp(max(0, ah-child.frame(false)-child.margin(false)), pointer(ah), false))
860 }
861 d.layoutElement(child, widths[i], pointer(ah), pointer(actual[i]), fh)
862 q := ah - child.ch - child.frame(false) - child.margin(false)
863 offset := 0
864 if q > 0 {
865 if child.align(e) == "end" {
866 offset = q
867 } else if child.align(e) == "center" {
868 offset = q / 2
869 }
870 }
871 position(e, child, positions[i]+child.n("margin-left"), ay+offset+child.n("margin-top"))
872 }
873 if nr == 0 {
874 return 0
875 }
876 return sum(rows) + e.n("row-gap")*(nr-1)
877}
878func (d *document) cascade() {
879 for _, e := range d.elements {
880 e.style = map[string]string{}
881 for property, value := range initial {
882 e.style[property] = value
883 }
884 specificities := map[string]int{}
885 for _, r := range d.rules {
886 specificity := -1
887 for _, selector := range r.selectors {
888 if strings.HasPrefix(selector, "#") {
889 if e.id == selector[1:] {
890 specificity = max(specificity, 2)
891 }
892 } else if strings.HasPrefix(selector, ".") {
893 if e.classes[selector[1:]] {
894 specificity = max(specificity, 1)
895 }
896 } else if e.kind == selector {
897 specificity = max(specificity, 0)
898 }
899 }
900 if specificity < 0 {
901 continue
902 }
903 for _, decl := range r.declarations {
904 old, exists := specificities[decl.property]
905 if !exists || specificity >= old {
906 e.style[decl.property] = decl.value
907 specificities[decl.property] = specificity
908 }
909 }
910 }
911 for property, value := range e.inline {
912 e.style[property] = value
913 }
914 e.displayed = e.style["display"] != "none" && (e.parent == nil || e.parent.displayed)
915 e.containing = nil
916 }
917}
918func (d *document) layout() {
919 d.cascade()
920 root := d.elements[0]
921 if root.displayed && root.style["position"] != "absolute" {
922 d.layoutElement(root, d.width, pointer(d.height), nil, nil)
923 root.rx, root.ry = root.n("margin-left"), root.n("margin-top")
924 }
925 for _, e := range d.elements {
926 if !e.displayed || e.style["position"] != "absolute" {
927 continue
928 }
929 ancestor := e.parent
930 for ancestor != nil && ancestor.style["position"] == "static" {
931 ancestor = ancestor.parent
932 }
933 e.containing = ancestor
934 cbw, cbh := d.width, d.height
935 if ancestor != nil {
936 cbw = ancestor.cw + ancestor.n("padding-left") + ancestor.n("padding-right")
937 cbh = ancestor.ch + ancestor.n("padding-top") + ancestor.n("padding-bottom")
938 }
939 value := e.resolve("width", pointer(cbw), true)
940 w := 0
941 if value != nil {
942 w = *value
943 } else {
944 r := cbw - e.margin(true) - e.frame(true)
945 if e.style["left"] != "auto" {
946 r -= e.n("left")
947 }
948 if e.style["right"] != "auto" {
949 r -= e.n("right")
950 }
951 r = max(0, r)
952 if e.style["left"] != "auto" && e.style["right"] != "auto" {
953 w = r
954 } else {
955 w = min(d.maxContent(e), r)
956 }
957 }
958 w = e.clamp(w, pointer(cbw), true)
959 var fh *int
960 if e.style["height"] == "auto" && e.style["top"] != "auto" && e.style["bottom"] != "auto" {
961 fh = pointer(e.clamp(max(0, cbh-e.n("top")-e.n("bottom")-e.margin(false)-e.frame(false)), pointer(cbh), false))
962 }
963 d.layoutElement(e, cbw, pointer(cbh), pointer(w), fh)
964 e.rx = e.n("margin-left")
965 if e.style["left"] != "auto" {
966 e.rx += e.n("left")
967 } else if e.style["right"] != "auto" {
968 e.rx = cbw - e.n("right") - e.n("margin-right") - e.cw - e.frame(true)
969 }
970 e.ry = e.n("margin-top")
971 if e.style["top"] != "auto" {
972 e.ry += e.n("top")
973 } else if e.style["bottom"] != "auto" {
974 e.ry = cbh - e.n("bottom") - e.n("margin-bottom") - e.ch - e.frame(false)
975 }
976 }
977 for _, e := range d.elements {
978 if !e.displayed {
979 continue
980 }
981 x, y := 0, 0
982 if e.style["position"] == "absolute" {
983 if e.containing != nil {
984 x, y = e.containing.x+e.containing.n("border-left"), e.containing.y+e.containing.n("border-top")
985 }
986 } else if e.parent != nil {
987 x, y = e.parent.x, e.parent.y
988 }
989 e.x, e.y = x+e.rx, y+e.ry
990 if e.style["position"] == "relative" {
991 if e.style["left"] != "auto" {
992 e.x += e.n("left")
993 } else if e.style["right"] != "auto" {
994 e.x -= e.n("right")
995 }
996 if e.style["top"] != "auto" {
997 e.y += e.n("top")
998 } else if e.style["bottom"] != "auto" {
999 e.y -= e.n("bottom")
1000 }
1001 }
1002 }
1003}

1007func (d *document) snapshot() [][5]int {
1008 result := make([][5]int, len(d.elements))
1009 for i, e := range d.elements {
1010 if e.displayed {
1011 r := e.border()
1012 result[i] = [5]int{1, r.x, r.y, r.w, r.h}
1013 }
1014 }
1015 return result

1017func (d *document) changed(before [][5]int) string {
1018 after := d.snapshot()
1019 names := []string{}
1020 for i, e := range d.elements {
1021 if before[i] != after[i] {
1022 names = append(names, e.id)
1023 }
1024 }
1025 if len(names) == 0 {

1027 }

1030func (d *document) render() {
1031 screen := make([][]byte, d.height)
1032 for y := range screen {
1033 screen[y] = []byte(strings.Repeat(".", d.width))
1034 }
1035 draw := func(x, y int, c byte) {
1036 if x >= 0 && x < d.width && y >= 0 && y < d.height {
1037 screen[y][x] = c
1038 }
1039 }
1040 for _, e := range d.elements {
1041 if !e.displayed {
1042 continue
1043 }
1044 r := e.border()
1045 for y := max(0, r.y); y < min(d.height, r.y+r.h); y++ {
1046 for x := max(0, r.x); x < min(d.width, r.x+r.w); x++ {
1047 horizontal := e.n("border-top") == 1 && y == r.y || e.n("border-bottom") == 1 && y == r.y+r.h-1
1048 vertical := e.n("border-left") == 1 && x == r.x || e.n("border-right") == 1 && x == r.x+r.w-1
1049 c := byte(' ')
1050 if horizontal && vertical {
1051 c = '+'
1052 } else if horizontal {
1053 c = '-'
1054 } else if vertical {
1055 c = '|'
1056 }
1057 draw(x, y, c)
1058 }
1059 }
1060 content := e.content()
1061 for i, line := range e.lines {
1062 for j := 0; j < len(line); j++ {
1063 draw(content.x+j, content.y+i, line[j])
1064 }
1065 }
1066 }
1067 d.emit("render %dx%d", d.width, d.height)
1068 d.emit("%s", strings.Repeat("#", d.width+2))
1069 for _, row := range screen {
1070 d.emit("#%s#", row)
1071 }
1072 d.emit("%s", strings.Repeat("#", d.width+2))
1073}
1074func (d *document) command(line string) {
1075 fields := strings.Split(line, " ")
1076 command, args := fields[0], fields[1:]
1077 switch command {
1078 case "LAYOUT":
1079 d.emit("layout %dx%d", d.width, d.height)
1080 for _, e := range d.elements {
1081 indent := strings.Repeat(" ", e.depth*2)
1082 if e.displayed {
1083 r := e.border()
1084 d.emit("%s%s %d %d %d %d", indent, e.id, r.x, r.y, r.w, r.h)
1085 } else {
1086 d.emit("%s%s none", indent, e.id)
1087 }
1088 }
1089 case "BOX":
1090 e := d.byID[args[0]]
1091 if e == nil {
1092 d.emit("error: unknown element %s", args[0])
1093 return
1094 }
1095 if !e.displayed {
1096 d.emit("box %s: none", e.id)
1097 return
1098 }
1099 d.emit("box %s", e.id)
1100 r := e.border()
1101 padding := rectangle{
1102 r.x + e.n("border-left"), r.y + e.n("border-top"),
1103 e.cw + e.n("padding-left") + e.n("padding-right"),
1104 e.ch + e.n("padding-top") + e.n("padding-bottom"),
1105 }
1106 names := []string{"margin", "border", "padding", "content"}
1107 for i, box := range []rectangle{e.marginBox(), r, padding, e.content()} {
1108 d.emit("%s %d %d %d %d", names[i], box.x, box.y, box.w, box.h)
1109 }
1110 case "RENDER":
1111 d.render()
1112 case "HIT":
1113 x, y := number(args[0]), number(args[1])
1114 if x >= d.width || y >= d.height {
1115 d.emit("error: point %d %d outside viewport", x, y)
1116 return
1117 }
1118 var found *element
1119 for _, e := range d.elements {
1120 r := e.border()
1121 if e.displayed && x >= r.x && x < r.x+r.w && y >= r.y && y < r.y+r.h {
1122 found = e
1123 }
1124 }
1125 path := []string{}
1126 for found != nil {
1127 path = append(path, found.id)
1128 found = found.parent
1129 }
1130 for i, j := 0, len(path)-1; i < j; i, j = i+1, j-1 {
1131 path[i], path[j] = path[j], path[i]
1132 }
1133 value := "none"
1134 if len(path) > 0 {
1135 value = strings.Join(path, " > ")
1136 }
1137 d.emit("hit %d %d: %s", x, y, value)
1138 case "SET":
1139 e := d.byID[args[0]]
1140 property, value := args[1], strings.Join(args[2:], " ")
1141 if e == nil {
1142 d.emit("error: unknown element %s", args[0])
1143 return
1144 }
1145 if _, exists := initial[property]; !exists && !contains(property, "margin", "padding", "border", "gap") {
1146 d.emit("error: unknown property %s", property)
1147 return
1148 }
1149 declarations, ok := valid(property, value)
1150 if !ok {
1151 d.emit("error: invalid value for %s: %s", property, value)
1152 return
1153 }
1154 before := d.snapshot()
1155 for _, decl := range declarations {
1156 e.inline[decl.property] = decl.value
1157 }
1158 d.layout()
1159 d.emit("set %s %s %s: %s", e.id, property, value, d.changed(before))
1160 case "VIEWPORT":
1161 w, h := number(args[0]), number(args[1])
1162 if w < 1 || w > 80 || h < 1 || h > 40 {
1163 d.emit("error: invalid viewport %d %d", w, h)
1164 return
1165 }
1166 before := d.snapshot()
1167 d.width, d.height = w, h
1168 d.layout()
1169 d.emit("viewport %d %d: %s", w, h, d.changed(before))
1170 case "OVERFLOW":
1171 names := []string{}
1172 for _, e := range d.elements {
1173 if !e.displayed {
1174 continue
1175 }
1176 overflow := false
1177 if e.text != "" {
1178 overflow = len(e.lines) > e.ch
1179 for _, line := range e.lines {
1180 overflow = overflow || len(line) > e.cw
1181 }
1182 } else {
1183 content := e.content()
1184 for _, child := range e.flow() {
1185 r := child.marginBox()
1186 if r.x < content.x || r.y < content.y || r.x+r.w > content.x+content.w || r.y+r.h > content.y+content.h {
1187 overflow = true
1188 }
1189 }
1190 }
1191 if overflow {
1192 names = append(names, e.id)
1193 }
1194 }
1195 value := "none"
1196 if len(names) > 0 {
1197 value = strings.Join(names, " ")
1198 }
1199 d.emit("overflow: %s", value)
1200 default:
1201 d.emit("error: unknown command %s", command)
1202 }
1203}
1204func main() {
1205 for _, prefix := range []string{"margin", "padding", "border"} {
1206 for _, side := range []string{"top", "right", "bottom", "left"} {
1207 initial[prefix+"-"+side] = "0"
1208 }
1209 }
1210 input, _ := io.ReadAll(os.Stdin)
1211 lines := strings.Split(strings.TrimSuffix(string(input), "\n"), "\n")
1212 index := 0
1213 next := func() string { line := strings.TrimSuffix(lines[index], "\r"); index++; return line }
1214 dimensions := strings.Split(next(), " ")
1215 d := document{width: number(dimensions[0]), height: number(dimensions[1]), byID: map[string]*element{}}
1216 for index < len(lines) {
1217 line := next()
1218 if line == "---" {
1219 break
1220 }
1221 left, right, _ := strings.Cut(line, " { ")
1222 r := rule{selectors: strings.Split(left, ", ")}
1223 for _, part := range strings.Split(strings.TrimSuffix(right, " }"), "; ") {
1224 property, value, _ := strings.Cut(part, ": ")
1225 if declarations, ok := valid(property, value); ok {
1226 r.declarations = append(r.declarations, declarations...)
1227 }
1228 }
1229 d.rules = append(d.rules, r)
1230 }
1231 stack := []*element{}
1232 for index < len(lines) {
1233 line := next()
1234 if line == "===" {
1235 break
1236 }
1237 spaces := len(line) - len(strings.TrimLeft(line, " "))
1238 depth := spaces / 2
1239 head, text, _ := strings.Cut(line[spaces:], " ")
1240 if text != "" {
1241 text = text[1 : len(text)-1]
1242 }
1243 kind, rest, _ := strings.Cut(head, "#")
1244 parts := strings.Split(rest, ".")
1245 e := &element{kind: kind, id: parts[0], text: text, classes: map[string]bool{}, inline: map[string]string{}, depth: depth, index: len(d.elements)}
1246 for _, class := range parts[1:] {
1247 e.classes[class] = true
1248 }
1249 if depth > 0 {
1250 e.parent = stack[depth-1]
1251 e.parent.children = append(e.parent.children, e)
1252 }
1253 stack = append(stack[:depth], e)
1254 d.elements = append(d.elements, e)
1255 d.byID[e.id] = e
1256 }
1257 d.layout()
1258 for index < len(lines) {
1259 line := next()
1260 if line == "END" {
1261 break
1262 }
1263 d.command(line)
1264 }
1265 out := bufio.NewWriter(os.Stdout)
1266 defer out.Flush()
1267 for _, line := range d.output {
1268 fmt.Fprintln(out, line)
1269 }
1270}
