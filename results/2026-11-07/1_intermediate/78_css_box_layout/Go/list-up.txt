func (e *element) frame(horizontal bool) int {
	if horizontal {
		return e.n("padding-left") + e.n("padding-right") + e.n("border-left") + e.n("border-right")
	}
	return e.n("padding-top") + e.n("padding-bottom") + e.n("border-top") + e.n("border-bottom")
func (e *element) border() rectangle {
	return rectangle{e.x, e.y, e.cw + e.frame(true), e.ch + e.frame(false)}
}
func (e *element) content() rectangle {
	return rectangle{e.x + e.n("border-left") + e.n("padding-left"), e.y + e.n("border-top") + e.n("padding-top"), e.cw, e.ch}
}

	}
	if fh == nil {
