122 public static function declaration(string $property, string $value): ?array
123 {
124 if (!self::known($property)) {
125 return null;
126 }
127 $tokens = explode(' ', $value);
128 if (in_array($property, ['margin', 'padding', 'border'], true)) {
129 if (count($tokens) < 1 || count($tokens) > 4) {
130 return null;
131 }
132 foreach ($tokens as $token) {
133 if ($property === 'border' ? !in_array($token, ['0', '1'], true) : !self::integer($token)) {
134 return null;
135 }
136 }
137 $expanded = [$tokens[0], $tokens[1] ?? $tokens[0], $tokens[2] ?? $tokens[0], $tokens[3] ?? $tokens[1] ?? $tokens[0]];
138 $result = [];
139 foreach (self::SIDES as $i => $side) {
140 $result[] = ["$property-$side", $expanded[$i]];
141 }
142 return $result;
143 }
144 if ($property === 'gap') {
145 if (count($tokens) > 2) {
146 return null;
147 }
148 foreach ($tokens as $token) {
149 if (!self::integer($token)) {
150 return null;
151 }
152 }
153 return [['row-gap', $tokens[0]], ['column-gap', $tokens[count($tokens) - 1]]];
154 }
155 $valid = false;
156 if ($property === 'grid-template-columns' || $property === 'grid-template-rows') {
157 $valid = $value === 'none';
158 if (!$valid) {
159 $valid = true;
160 foreach ($tokens as $token) {
161 if ($token !== 'auto' && !self::integer($token) && !self::fraction($token)) {
162 $valid = false;
163 }
164 }
165 }
166 } elseif ($property === 'grid-column' || $property === 'grid-row') {
167 $valid = $value === 'auto'
168 || (count($tokens) === 1 && self::positive($tokens[0]))
169 || (count($tokens) === 2 && $tokens[0] === 'span' && self::positive($tokens[1]))
170 || (count($tokens) === 3 && self::positive($tokens[0]) && $tokens[1] === '/' && self::positive($tokens[2]) && (int)$tokens[0] < (int)$tokens[2])
171 || (count($tokens) === 4 && self::positive($tokens[0]) && $tokens[1] === '/' && $tokens[2] === 'span' && self::positive($tokens[3]));
172 } elseif (count($tokens) === 1) {
173 if (isset(self::CHOICES[$property])) {
174 $valid = in_array($value, self::CHOICES[$property], true);
175 } elseif (in_array($property, ['width', 'height', 'flex-basis'], true)) {
176 $valid = $value === 'auto' || self::integer($value) || self::percentage($value);
177 } elseif (in_array($property, ['min-width', 'min-height'], true)) {
178 $valid = self::integer($value) || self::percentage($value);
179 } elseif (in_array($property, ['max-width', 'max-height'], true)) {
180 $valid = $value === 'none' || self::integer($value) || self::percentage($value);
181 } elseif (in_array($property, ['top', 'right', 'bottom', 'left'], true)) {
182 $valid = $value === 'auto' || self::signed($value);
183 } elseif ($property === 'order') {
184 $valid = self::signed($value);
185 } elseif (str_starts_with($property, 'border-')) {
186 $valid = in_array($value, ['0', '1'], true);
187 } else {
188 $valid = self::integer($value);
189 }
190 }
191 return $valid ? [[$property, $value]] : null;
192 }

208 private function resolve(Element $element, string $property, ?int $reference, bool $horizontal): ?int
209 {
210 $value = $element->style[$property];
211 if ($value === 'auto' || $value === 'none' || (str_ends_with($value, '%') && $reference === null)) {
212 return null;
213 }
214 $result = str_ends_with($value, '%') ? intdiv($reference * (int)$value, 100) : (int)$value;
215 return $element->style['box-sizing'] === 'border-box' ? max(0, $result - $element->frame($horizontal)) : $result;

218 private function clamp(Element $element, int $value, ?int $reference, bool $horizontal): int
219 {
220 $axis = $horizontal ? 'width' : 'height';
221 $lower = $this->resolve($element, 'min-' . $axis, $reference, $horizontal) ?? 0;
222 $upper = $this->resolve($element, 'max-' . $axis, $reference, $horizontal);
223 if ($upper !== null) {
224 $value = min($value, $upper);
225 }
226 return max($value, $lower);

229 private function distribute(int $total, array $weights): array
230 {
231 $sum = array_sum($weights);
232 $result = array_fill(0, count($weights), 0);
233 if ($sum === 0) {
234 return $result;
235 }
236 foreach ($weights as $i => $weight) {
237 $result[$i] = intdiv($total * $weight, $sum);
238 }
239 $remainder = $total - array_sum($result);
240 foreach ($weights as $i => $weight) {
241 if ($remainder === 0) {
242 break;
243 }
244 if ($weight > 0) {
245 $result[$i]++;
246 $remainder--;
247 }
248 }
249 return $result;
250 }

265 private function placement(Element $element): array
266 {
267 $columns = $element->style['grid-template-columns'] === 'none' ? ['auto'] : explode(' ', $element->style['grid-template-columns']);
268 $rows = $element->style['grid-template-rows'] === 'none' ? [] : explode(' ', $element->style['grid-template-rows']);
269 $items = [];
270 $nc = count($columns);
271 foreach ($this->children($element, true) as $child) {
272 [$column, $cs] = $this->gridPosition($child->style['grid-column']);
273 [$row, $rs] = $this->gridPosition($child->style['grid-row']);
274 $items[] = ['element' => $child, 'column' => $column, 'row' => $row, 'cs' => $cs, 'rs' => $rs];
275 $nc = max($nc, $column === 0 ? $cs : $column + $cs - 1);
276 }
277 $occupied = [];
278 $overlaps = function (array $item, int $row, int $column) use (&$occupied): bool {
279 for ($r = $row; $r < $row + $item['rs']; $r++) {
280 for ($c = $column; $c < $column + $item['cs']; $c++) {
281 if (isset($occupied["$r,$c"])) {
282 return true;
283 }
284 }
285 }
286 return false;
287 };
288 $occupy = function (array $item) use (&$occupied): void {
289 for ($r = $item['row']; $r < $item['row'] + $item['rs']; $r++) {
290 for ($c = $item['column']; $c < $item['column'] + $item['cs']; $c++) {
291 $occupied["$r,$c"] = true;
292 }
293 }
294 };
295 foreach ($items as $item) {
296 if ($item['row'] > 0 && $item['column'] > 0) {
297 $occupy($item);
298 }
299 }
300 $cr = $cc = 1;
301 foreach ($items as &$item) {
302 if ($item['row'] > 0 && $item['column'] > 0) {
303 continue;
304 }
305 if ($item['row'] > 0) {
306 $column = 1;
307 while ($column + $item['cs'] - 1 <= $nc && $overlaps($item, $item['row'], $column)) {
308 $column++;
309 }
310 $item['column'] = $column + $item['cs'] - 1 <= $nc ? $column : 1;
311 } elseif ($item['column'] > 0) {
312 if ($item['column'] < $cc) {
313 $cr++;
314 }
315 $row = $cr;
316 while ($overlaps($item, $row, $item['column'])) {
317 $row++;
318 }
319 $item['row'] = $cr = $row;
320 $cc = $item['column'] + $item['cs'];
321 } else {
322 $row = $cr;
323 $column = $cc;
324 while (true) {
325 if ($column + $item['cs'] - 1 > $nc) {
326 $row++;
327 $column = 1;
328 } elseif ($overlaps($item, $row, $column)) {
329 $column++;
330 } else {
331 break;
332 }
333 }
334 $item['row'] = $cr = $row;
335 $item['column'] = $column;
336 $cc = $column + $item['cs'];
337 }
338 $occupy($item);
339 }
340 unset($item);
341 $nr = count($rows);
342 foreach ($items as $item) {
343 $nr = max($nr, $item['row'] + $item['rs'] - 1);
344 }
345 return ['items' => $items, 'columns' => array_pad($columns, $nc, 'auto'), 'rows' => array_pad($rows, $nr, 'auto')];
346 }

348 private function trackSizes(array $grid, bool $horizontal, ?int $available, int $gap, array $contributions): array
349 {
350 $specifications = $grid[$horizontal ? 'columns' : 'rows'];
351 $fractional = false;
352 foreach ($specifications as $specification) {
353 $fractional = $fractional || self::fraction($specification);
354 }
355 $weights = $sizes = [];
356 foreach ($specifications as $i => $specification) {
357 $weights[] = $fractional ? (self::fraction($specification) ? (int)$specification : 0) : ($specification === 'auto' ? 1 : 0);
358 $size = 0;
359 if (self::integer($specification)) {
360 $size = (int)$specification;
361 } elseif ($specification === 'auto' || $available === null) {
362 foreach ($grid['items'] as $j => $item) {
363 if ($item[$horizontal ? 'column' : 'row'] === $i + 1 && $item[$horizontal ? 'cs' : 'rs'] === 1) {
364 $size = max($size, $contributions[$j]);
365 }
366 }
367 }
368 $sizes[] = $size;
369 }
370 if ($available !== null) {
371 $leftover = $available - array_sum($sizes) - $gap * (count($sizes) - 1);
372 $additions = $this->distribute(max($leftover, 0), $weights);
373 foreach ($additions as $i => $addition) {
374 $sizes[$i] += $addition;
375 }
376 }
377 return $sizes;
378 }

380 private function maxContent(Element $element): int
381 {
382 if ($element->maximum !== null) {
383 return $element->maximum;
384 }
385 $items = $this->children($element);
386 if ($element->text !== null) {
387 $value = strlen($element->text);
388 } elseif ($element->style['display'] === 'flex' && str_starts_with($element->style['flex-direction'], 'row')) {
389 $value = array_sum(array_map(fn($child) => $this->contribution($child), $items)) + (int)$element->style['column-gap'] * max(0, count($items) - 1);
390 } elseif ($element->style['display'] === 'grid') {
391 $grid = $this->placement($element);
392 $values = array_map(fn($item) => $this->contribution($item['element']), $grid['items']);
393 $value = array_sum($this->trackSizes($grid, true, null, (int)$element->style['column-gap'], $values)) + (int)$element->style['column-gap'] * (count($grid['columns']) - 1);
394 } else {
395 $value = $items === [] ? 0 : max(array_map(fn($child) => $this->contribution($child), $items));
396 }
397 return $element->maximum = $value;
398 }

411 private function layout(Element $element, int $cwReference, ?int $chReference, ?int $fw = null, ?int $fh = null): int
412 {
413 $element->cw = $fw ?? $this->clamp($element, $this->resolve($element, 'width', $cwReference, true) ?? max(0, $cwReference - $element->frame(true) - $element->margins(true)), $cwReference, true);
414 if ($fh === null) {
415 $height = $this->resolve($element, 'height', $chReference, false);
416 if ($height !== null) {
417 $fh = $this->clamp($element, $height, $chReference, false);
418 }
419 }
420 if ($element->text !== null) {
421 $element->lines = [];
422 foreach (explode(' ', $element->text) as $word) {
423 $last = count($element->lines) - 1;
424 if ($last >= 0 && strlen($element->lines[$last]) + 1 + strlen($word) <= $element->cw) {
425 $element->lines[$last] .= ' ' . $word;
426 } else {
427 $element->lines[] = $word;
428 }
429 }
430 $natural = count($element->lines);
431 } elseif ($element->style['display'] === 'flex') {
432 $natural = $this->flex($element, $fh, $chReference);
433 } elseif ($element->style['display'] === 'grid') {
434 $natural = $this->gridLayout($element, $fh);
435 } else {
436 $natural = $this->block($element, $fh);
437 }
438 $element->natural = $natural;
439 $element->ch = $fh ?? $this->clamp($element, $natural, $chReference, false);

443 private function block(Element $element, ?int $reference): int
444 {
445 $bottom = 0;
446 $previous = null;
447 foreach ($this->children($element) as $child) {
448 $this->layout($child, $element->cw, $reference);
449 $y = $bottom + ($previous === null ? $child->margin[0] : max($previous->margin[2], $child->margin[0]));
450 $this->positionChild($element, $child, $child->margin[3], $y);
451 $bottom = $y + $child->height();
452 $previous = $child;
453 }
454 return $previous === null ? 0 : $bottom + $previous->margin[2];

457 private function flex(Element $element, ?int $reference, ?int $heightReference): int
458 {
459 $horizontal = str_starts_with($element->style['flex-direction'], 'row');
460 $reverse = str_ends_with($element->style['flex-direction'], 'reverse');
461 $mainReference = $horizontal ? $element->cw : $reference;
462 $gap = (int)$element->style[$horizontal ? 'column-gap' : 'row-gap'];
463 $lineGap = (int)$element->style['row-gap'];
464 $items = [];
465 foreach ($this->children($element, true) as $child) {
466 $alignment = $this->align($element, $child);
467 $cross = null;
468 if (!$horizontal) {
469 $room = max(0, $element->cw - $child->frame(true) - $child->margins(true));
470 $cross = $this->resolve($child, 'width', $element->cw, true) ?? ($alignment === 'stretch' ? $room : min($this->maxContent($child), $room));
471 $cross = $this->clamp($child, $cross, $element->cw, true);
472 }
473 $base = $this->resolve($child, 'flex-basis', $mainReference, $horizontal) ?? $this->resolve($child, $horizontal ? 'width' : 'height', $mainReference, $horizontal);
474 $base ??= $horizontal ? $this->maxContent($child) : $this->layout($child, $element->cw, $reference, $cross);
475 $hypothetical = $this->clamp($child, $base, $mainReference, $horizontal);
476 $items[] = ['element' => $child, 'align' => $alignment, 'cross' => $cross, 'base' => $base, 'hypothetical' => $hypothetical, 'outer' => $child->frame($horizontal) + $child->margins($horizontal)];
477 }
478 $mainLength = $horizontal ? $element->cw : $reference;
479 if ($mainLength === null) {
480 $natural = array_sum(array_map(fn($item) => $item['hypothetical'] + $item['outer'], $items)) + $gap * max(0, count($items) - 1);
481 $mainLength = $this->clamp($element, $natural, $heightReference, false);
482 }
483 $lines = [];
484 foreach ($items as $item) {
485 $last = count($lines) - 1;
486 $currentLength = $last < 0 ? 0 : array_sum(array_map(fn($i) => $i['hypothetical'] + $i['outer'], $lines[$last])) + $gap * count($lines[$last]);
487 if ($last < 0 || ($horizontal && $element->style['flex-wrap'] === 'wrap' && $currentLength + $item['hypothetical'] + $item['outer'] > $mainLength)) {
488 $lines[] = [];
489 $last++;
490 }
491 $lines[$last][] = $item;
492 }
493 $crossStart = 0;
494 foreach ($lines as $line) {
495 $totalGap = $gap * (count($line) - 1);
496 $growing = array_sum(array_map(fn($i) => $i['hypothetical'] + $i['outer'], $line)) + $totalGap < $mainLength;
497 foreach ($line as &$item) {
498 $child = $item['element'];
499 $item['weight'] = $growing ? (int)$child->style['flex-grow'] : (int)$child->style['flex-shrink'] * $item['base'];
500 $item['size'] = $item['hypothetical'];
501 $item['frozen'] = $item['weight'] === 0 || ($growing ? $item['base'] > $item['hypothetical'] : $item['base'] < $item['hypothetical']);
502 }
503 unset($item);
504 while (true) {
505 $active = [];
506 $free = $mainLength - $totalGap;
507 foreach ($line as $i => $item) {
508 $free -= ($item['frozen'] ? $item['size'] : $item['base']) + $item['outer'];
509 if (!$item['frozen']) {
510 $active[] = $i;
511 }
512 }
513 if ($active === []) {
514 break;
515 }
516 $assigned = $this->distribute(max(0, $growing ? $free : -$free), array_map(fn($i) => $line[$i]['weight'], $active));
517 $violation = 0;
518 foreach ($active as $j => $i) {
519 $tentative = $line[$i]['base'] + ($growing ? $assigned[$j] : -$assigned[$j]);
520 $line[$i]['size'] = $this->clamp($line[$i]['element'], $tentative, $mainReference, $horizontal);
521 $line[$i]['violation'] = $line[$i]['size'] - $tentative;
522 $violation += $line[$i]['violation'];
523 }
524 foreach ($active as $i) {
525 if ($violation === 0 || ($violation > 0 ? $line[$i]['violation'] > 0 : $line[$i]['violation'] < 0)) {
526 $line[$i]['frozen'] = true;
527 }
528 }
529 }
530 if ($horizontal) {
531 foreach ($line as &$item) {
532 $this->layout($item['element'], $element->cw, $reference, $item['size']);
533 $item['cross'] = $item['element']->ch;
534 }
535 unset($item);
536 }
537 $outerCross = max(array_map(fn($i) => $i['cross'] + $i['element']->frame(!$horizontal) + $i['element']->margins(!$horizontal), $line));
538 $lineCross = !$horizontal ? $element->cw : (count($lines) === 1 ? ($reference ?? $this->clamp($element, $outerCross, $heightReference, false)) : $outerCross);
539 $remaining = $mainLength - $totalGap - array_sum(array_map(fn($i) => $i['size'] + $i['outer'], $line));
540 $spaces = array_fill(0, count($line), 0);
541 if ($remaining > 0) {
542 $kind = $element->style['justify-content'];
543 if ($kind === 'end') {
544 $spaces[0] = $remaining;
545 } elseif ($kind === 'center') {
546 $spaces[0] = intdiv($remaining, 2);
547 } elseif (in_array($kind, ['space-between', 'space-around', 'space-evenly'], true)) {
548 $weights = [];
549 for ($i = 0; $i <= count($line); $i++) {
550 $weights[] = $kind === 'space-evenly' ? 1 : (($i === 0 || $i === count($line)) ? ($kind === 'space-between' ? 0 : 1) : ($kind === 'space-between' ? 1 : 2));
551 }
552 $spaces = array_slice($this->distribute($remaining, $weights), 0, count($line));
553 }
554 }
555 $cursor = 0;
556 foreach ($line as $i => $item) {
557 $child = $item['element'];
558 $cross = $item['cross'];
559 if ($horizontal) {
560 if ($item['align'] === 'stretch' && $child->style['height'] === 'auto') {
561 $cross = $this->clamp($child, max(0, $lineCross - $child->frame(false) - $child->margins(false)), $reference, false);
562 $this->layout($child, $element->cw, $reference, $item['size'], $cross);
563 } else {
564 $this->layout($child, $element->cw, $reference, $item['size']);
565 }
566 } else {
567 $this->layout($child, $element->cw, $reference, $cross, $item['size']);
568 }
569 $room = $lineCross - $cross - $child->frame(!$horizontal) - $child->margins(!$horizontal);
570 $offset = $room > 0 ? ($item['align'] === 'end' ? $room : ($item['align'] === 'center' ? intdiv($room, 2) : 0)) : 0;
571 $cursor += $spaces[$i];
572 $size = $item['size'] + $item['outer'];
573 $main = ($reverse ? $mainLength - $cursor - $size : $cursor) + $child->margin[$horizontal ? 3 : 0];
574 $crossPosition = $crossStart + $offset + $child->margin[$horizontal ? 0 : 3];
575 $this->positionChild($element, $child, $horizontal ? $main : $crossPosition, $horizontal ? $crossPosition : $main);
576 $cursor += $size + $gap;
577 }
578 $crossStart += $lineCross + $lineGap;
579 }
580 return $horizontal ? ($lines === [] ? 0 : $crossStart - $lineGap) : $mainLength;
581 }

583 private function gridLayout(Element $element, ?int $reference): int
584 {
585 $grid = $this->placement($element);
586 $columnGap = (int)$element->style['column-gap'];
587 $rowGap = (int)$element->style['row-gap'];
588 $columns = $this->trackSizes($grid, true, $element->cw, $columnGap, array_map(fn($i) => $this->contribution($i['element']), $grid['items']));
589 foreach ($grid['items'] as &$item) {
590 $child = $item['element'];
591 $start = $item['column'] - 1;
592 $item['aw'] = array_sum(array_slice($columns, $start, $item['cs'])) + $columnGap * ($item['cs'] - 1);
593 $item['ax'] = array_sum(array_slice($columns, 0, $start)) + $columnGap * $start;
594 $item['gw'] = $this->clamp($child, $this->resolve($child, 'width', $item['aw'], true) ?? max(0, $item['aw'] - $child->frame(true) - $child->margins(true)), $item['aw'], true);
595 $this->layout($child, $item['aw'], null, $item['gw']);
596 $item['rowContribution'] = $child->height() + $child->margins(false);
597 }
598 unset($item);
599 $rows = $this->trackSizes($grid, false, $reference, $rowGap, array_column($grid['items'], 'rowContribution'));
600 foreach ($grid['items'] as $item) {
601 $child = $item['element'];
602 $start = $item['row'] - 1;
603 $areaHeight = array_sum(array_slice($rows, $start, $item['rs'])) + $rowGap * ($item['rs'] - 1);
604 $areaY = array_sum(array_slice($rows, 0, $start)) + $rowGap * $start;
605 $alignment = $this->align($element, $child);
606 $height = $alignment === 'stretch' && $child->style['height'] === 'auto' ? $this->clamp($child, max(0, $areaHeight - $child->frame(false) - $child->margins(false)), $areaHeight, false) : null;
607 $this->layout($child, $item['aw'], $areaHeight, $item['gw'], $height);
608 $room = $areaHeight - $child->height() - $child->margins(false);
609 $offset = $room > 0 ? ($alignment === 'end' ? $room : ($alignment === 'center' ? intdiv($room, 2) : 0)) : 0;
610 $this->positionChild($element, $child, $item['ax'] + $child->margin[3], $areaY + $offset + $child->margin[0]);
611 }
612 return $rows === [] ? 0 : array_sum($rows) + $rowGap * (count($rows) - 1);
613 }

615 private function relayout(): void
616 {
617 foreach ($this->elements as $element) {
618 $element->style = self::defaults();
619 $priorities = [];
620 foreach ($this->rules as [$selectors, $declarations]) {
621 $specificity = -1;
622 foreach ($selectors as $selector) {
623 if ($selector[0] === '#') {
624 $match = substr($selector, 1) === $element->id ? 2 : -1;
625 } elseif ($selector[0] === '.') {
626 $match = in_array(substr($selector, 1), $element->classes, true) ? 1 : -1;
627 } else {
628 $match = $selector === $element->type ? 0 : -1;
629 }
630 $specificity = max($specificity, $match);
631 }
632 if ($specificity < 0) {
633 continue;
634 }
635 foreach ($declarations as [$property, $value]) {
636 if (!isset($priorities[$property]) || $specificity >= $priorities[$property]) {
637 $element->style[$property] = $value;
638 $priorities[$property] = $specificity;
639 }
640 }
641 }
642 $element->style = array_replace($element->style, $element->overrides);
643 $element->margin = $element->padding = $element->border = [];
644 foreach (self::SIDES as $side) {
645 $element->margin[] = (int)$element->style["margin-$side"];
646 $element->padding[] = (int)$element->style["padding-$side"];
647 $element->border[] = (int)$element->style["border-$side"];
648 }
649 $element->displayed = $element->style['display'] !== 'none' && ($element->parent === null || $element->parent->displayed);
650 $element->maximum = $element->contribution = null;
651 $element->lines = [];
652 $element->containing = null;
653 }
654 $root = $this->elements[0];
655 if ($root->displayed && $root->style['position'] !== 'absolute') {
656 $this->layout($root, $this->width, $this->height);
657 $root->rx = $root->margin[3];
658 $root->ry = $root->margin[0];
659 }
660 foreach ($this->elements as $element) {
661 if (!$element->displayed || $element->style['position'] !== 'absolute') {
662 continue;
663 }
664 $ancestor = $element->parent;
665 while ($ancestor !== null && $ancestor->style['position'] === 'static') {
666 $ancestor = $ancestor->parent;
667 }
668 $element->containing = $ancestor;
669 $cbw = $ancestor === null ? $this->width : $ancestor->cw + $ancestor->padding[1] + $ancestor->padding[3];
670 $cbh = $ancestor === null ? $this->height : $ancestor->ch + $ancestor->padding[0] + $ancestor->padding[2];
671 $offsets = array_map(fn($p) => $element->style[$p] === 'auto' ? null : (int)$element->style[$p], ['left', 'right', 'top', 'bottom']);
672 [$left, $right, $top, $bottom] = $offsets;
673 $width = $this->resolve($element, 'width', $cbw, true);
674 if ($width === null) {
675 $room = max(0, $cbw - ($left ?? 0) - ($right ?? 0) - $element->margins(true) - $element->frame(true));
676 $width = $left !== null && $right !== null ? $room : min($this->maxContent($element), $room);
677 }
678 $fw = $this->clamp($element, $width, $cbw, true);
679 $fh = $element->style['height'] === 'auto' && $top !== null && $bottom !== null ? $this->clamp($element, max(0, $cbh - $top - $bottom - $element->margins(false) - $element->frame(false)), $cbh, false) : null;
680 $this->layout($element, $cbw, $cbh, $fw, $fh);
681 $element->rx = $left !== null ? $left + $element->margin[3] : ($right !== null ? $cbw - $right - $element->margin[1] - $element->width() : $element->margin[3]);
682 $element->ry = $top !== null ? $top + $element->margin[0] : ($bottom !== null ? $cbh - $bottom - $element->margin[2] - $element->height() : $element->margin[0]);
683 }
684 foreach ($this->elements as $element) {
685 if (!$element->displayed) {
686 continue;
687 }
688 $reference = $element->style['position'] === 'absolute' ? $element->containing : $element->parent;
689 $x = $reference?->x ?? 0;
690 $y = $reference?->y ?? 0;
691 if ($element->style['position'] === 'absolute' && $reference !== null) {
692 $x += $reference->border[3];
693 $y += $reference->border[0];
694 }
695 $element->x = $x + $element->rx;
696 $element->y = $y + $element->ry;
697 if ($element->style['position'] === 'relative') {
698 [$left, $right, $top, $bottom] = array_map(fn($p) => $element->style[$p] === 'auto' ? null : (int)$element->style[$p], ['left', 'right', 'top', 'bottom']);
699 $element->x += $left ?? ($right === null ? 0 : -$right);
700 $element->y += $top ?? ($bottom === null ? 0 : -$bottom);
701 }
702 }
703 }

705 private function boxes(Element $element): array
706 {
707 [$x, $y, $w, $h] = [$element->x, $element->y, $element->width(), $element->height()];
708 [$m, $b, $p] = [$element->margin, $element->border, $element->padding];
709 return [
710 [$x - $m[3], $y - $m[0], $w + $m[1] + $m[3], $h + $m[0] + $m[2]],
711 [$x, $y, $w, $h],
712 [$x + $b[3], $y + $b[0], $w - $b[1] - $b[3], $h - $b[0] - $b[2]],
713 [$x + $b[3] + $p[3], $y + $b[0] + $p[0], $element->cw, $element->ch]
714 ];
715 }

734 private function render(): void
735 {
736 $screen = array_fill(0, $this->height, str_repeat('.', $this->width));
737 foreach ($this->elements as $element) {
738 if (!$element->displayed) {
739 continue;
740 }
741 [$x, $y, $w, $h] = $this->boxes($element)[1];
742 for ($row = max($y, 0); $row < min($y + $h, $this->height); $row++) {
743 for ($column = max($x, 0); $column < min($x + $w, $this->width); $column++) {
744 $horizontal = ($element->border[0] === 1 && $row === $y) || ($element->border[2] === 1 && $row === $y + $h - 1);
745 $vertical = ($element->border[3] === 1 && $column === $x) || ($element->border[1] === 1 && $column === $x + $w - 1);
746 $screen[$row][$column] = $horizontal && $vertical ? '+' : ($horizontal ? '-' : ($vertical ? '|' : ' '));
747 }
748 }
749 if ($element->text !== null) {
750 [$cx, $cy] = $this->boxes($element)[3];
751 foreach ($element->lines as $i => $line) {
752 $row = $cy + $i;
753 if ($row < 0 || $row >= $this->height) {
754 continue;
755 }
756 for ($j = 0; $j < strlen($line); $j++) {
757 if ($cx + $j >= 0 && $cx + $j < $this->width) {
758 $screen[$row][$cx + $j] = $line[$j];
759 }
760 }
761 }
762 }
763 }
764 $edge = str_repeat('#', $this->width + 2);
765 echo "render {$this->width}x{$this->height}\n$edge\n";
766 foreach ($screen as $row) {
767 echo '#', $row, "#\n";
768 }
769 echo $edge, "\n";
770 }

772 public function execute(string $line): void
773 {
774 $args = explode(' ', $line);
775 $command = array_shift($args);
776 switch ($command) {
777 case 'LAYOUT':
778 echo "layout {$this->width}x{$this->height}\n";
779 foreach ($this->elements as $e) {
780 echo str_repeat(' ', $e->depth), $e->id, $e->displayed ? " {$e->x} {$e->y} " . $e->width() . ' ' . $e->height() : ' none', "\n";
781 }
782 break;
783 case 'BOX':
784 $element = $this->ids[$args[0]] ?? null;
785 if ($element === null) {
786 echo "error: unknown element {$args[0]}\n";
787 } elseif (!$element->displayed) {
788 echo "box {$element->id}: none\n";
789 } else {
790 echo "box {$element->id}\n";
791 $boxes = $this->boxes($element);
792 foreach (['margin', 'border', 'padding', 'content'] as $i => $name) {
793 echo $name, ' ', implode(' ', $boxes[$i]), "\n";
794 }
795 }
796 break;
797 case 'RENDER':
798 $this->render();
799 break;
800 case 'HIT':
801 [$x, $y] = array_map('intval', $args);
802 if ($x >= $this->width || $y >= $this->height) {
803 echo "error: point $x $y outside viewport\n";
804 } else {
805 $found = null;
806 foreach ($this->elements as $e) {
807 if ($e->displayed && $e->x <= $x && $x < $e->x + $e->width() && $e->y <= $y && $y < $e->y + $e->height()) {
808 $found = $e;
809 }
810 }
811 $path = [];
812 while ($found !== null) {
813 array_unshift($path, $found->id);
814 $found = $found->parent;
815 }
816 echo "hit $x $y: ", $path === [] ? 'none' : implode(' > ', $path), "\n";
817 }
818 break;
819 case 'SET':
820 [$id, $property] = array_slice($args, 0, 2);
821 $value = implode(' ', array_slice($args, 2));
822 if (!isset($this->ids[$id])) {
823 echo "error: unknown element $id\n";
824 } elseif (!self::known($property)) {
825 echo "error: unknown property $property\n";
826 } elseif (($declarations = self::declaration($property, $value)) === null) {
827 echo "error: invalid value for $property: $value\n";
828 } else {
829 $before = $this->snapshot();
830 foreach ($declarations as [$p, $v]) {
831 $this->ids[$id]->overrides[$p] = $v;
832 }
833 $this->relayout();
834 echo "set $id $property $value: ", $this->changes($before), "\n";
835 }
836 break;
837 case 'VIEWPORT':
838 [$width, $height] = array_map('intval', $args);
839 if ($width < 1 || $width > 80 || $height < 1 || $height > 40) {
840 echo "error: invalid viewport $width $height\n";
841 } else {
842 $before = $this->snapshot();
843 $this->width = $width;
844 $this->height = $height;
845 $this->relayout();
846 echo "viewport $width $height: ", $this->changes($before), "\n";
847 }
848 break;
849 case 'OVERFLOW':
850 $overflow = [];
851 foreach ($this->elements as $e) {
852 if (!$e->displayed) {
853 continue;
854 }
855 if ($e->text !== null) {
856 $over = max(array_map('strlen', $e->lines)) > $e->cw || count($e->lines) > $e->ch;
857 } else {
858 [$cx, $cy, $cw, $ch] = $this->boxes($e)[3];
859 $over = false;
860 foreach ($this->children($e) as $child) {
861 [$x, $y, $w, $h] = $this->boxes($child)[0];
862 if ($x < $cx || $y < $cy || $x + $w > $cx + $cw || $y + $h > $cy + $ch) {
863 $over = true;
864 }
865 }
866 }
867 if ($over) {
868 $overflow[] = $e->id;
869 }
870 }
871 echo 'overflow: ', $overflow === [] ? 'none' : implode(' ', $overflow), "\n";
872 break;
873 default:
874 echo "error: unknown command $command\n";
875 }
876 }
877}

879$input = array_map(fn($line) => rtrim($line, "\r\n"), file('php://stdin'));
880[$width, $height] = array_map('intval', explode(' ', $input[0]));
881$i = 1;
882$rules = [];
883while ($input[$i] !== '---') {
884 [$selectors, $body] = explode(' { ', $input[$i++], 2);
885 $declarations = [];
886 foreach (explode('; ', substr($body, 0, -2)) as $text) {
887 [$property, $value] = explode(': ', $text, 2);
888 $declarations = array_merge($declarations, BoxLayout::declaration($property, $value) ?? []);
889 }
890 $rules[] = [explode(', ', $selectors), $declarations];
891}
892$i++;
893$elements = $stack = [];
894while ($input[$i] !== '===') {
895 $line = $input[$i++];
896 $depth = intdiv(strspn($line, ' '), 2);
897 $parts = explode(' ', ltrim($line, ' '), 2);
898 [$type, $rest] = explode('#', $parts[0], 2);
899 $classes = explode('.', $rest);
900 $id = array_shift($classes);
901 $parent = $depth === 0 ? null : $stack[$depth - 1];
902 $text = isset($parts[1]) ? substr($parts[1], 1, -1) : null;
903 $element = new Element($type, $id, $classes, $text, $parent, $depth, count($elements));
904 if ($parent !== null) {
905 $parent->children[] = $element;
906 }
907 $elements[] = $element;
908 $stack[$depth] = $element;
909}
910$i++;
911$layout = new BoxLayout($width, $height, $rules, $elements);
912while ($input[$i] !== 'END') {
913 $layout->execute($input[$i++]);
914}
