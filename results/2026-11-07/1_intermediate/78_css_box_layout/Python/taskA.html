49def is_grid_line(tokens):
50 match tokens:
51 case ['auto']:
52 return True
53 case ['span', count]:
54 return is_positive(count)
55 case [start]:
56 return is_positive(start)
57 case [start, '/', 'span', count]:

105def expand_declaration(prop, value):
106 tokens = value.split(' ')
107 if prop in LONGHANDS:
108 return [(prop, value)] if LONGHANDS[prop][1](tokens) else None
109 if prop == 'gap' and len(tokens) <= 2 and all(map(is_non_negative, tokens)):
110 return [('row-gap', tokens[0]), ('column-gap', tokens[-1])]
111 if prop in BOX_SHORTHANDS and len(tokens) <= 4:
112 is_valid = LONGHANDS[f'{prop}-top'][1]
113 if all(is_valid([token]) for token in tokens):
114 picks = SHORTHAND_PICKS[len(tokens)]
115 return [(f'{prop}-{side}', tokens[pick]) for side, pick in zip(SIDES, picks)]
116 return None

255def cascaded_style(element, rules):
256 winners = {}
257 for selectors, declarations in rules:
258 specificity = max(selector_specificity(selector, element) for selector in selectors)
259 if specificity < 0:
260 continue
261 for prop, value in declarations:
262 if prop not in winners or specificity >= winners[prop][0]:
263 winners[prop] = (specificity, value)
264 return [(prop, value) for prop, (_, value) in winners.items()]

306def distribute(total, weights):
307 weight_sum = sum(weights)
308 shares = [total * weight // weight_sum for weight in weights]
309 remainder = total - sum(shares)
310 for index, weight in enumerate(weights):
311 if remainder > 0 and weight > 0:
312 shares[index] += 1
313 remainder -= 1
314 return shares

321def layout(element, width_ref, height_ref, fixed_width=None, fixed_height=None):
322 if fixed_width is None:
323 width = resolve(element, 'width', width_ref, True)
324 if width is None:
325 width = max(0, width_ref - element.outer(True))
326 fixed_width = clamp(element, width, True, width_ref)
327 if fixed_height is None:
328 height = resolve(element, 'height', height_ref, False)
329 if height is not None:
330 fixed_height = clamp(element, height, False, height_ref)

332 display = element.style['display']
333 if element.text is not None:
334 element.lines = wrap(element.text, fixed_width)
335 content_height = len(element.lines)

338 elif display == 'grid':

340 else:
341 content_height = block_layout(element, fixed_width, fixed_height)
342 if fixed_height is None:
343 fixed_height = clamp(element, content_height, False, height_ref)
344 element.ch = fixed_height
345 return content_height

394def resolve_main_sizes(line, main_length, gap, horizontal, main_ref):
395 gaps = gap_total(gap, len(line))
396 growing = sum(item.hypothetical + item.outer for item in line) + gaps < main_length
397 weights = []
398 unfrozen = []
399 for index, item in enumerate(line):
400 if growing:
401 weight = item.element.number('flex-grow')
402 else:
403 weight = item.element.number('flex-shrink') * item.base
404 weights.append(weight)
405 item.size = item.hypothetical
406 flexible = item.base <= item.hypothetical if growing else item.base >= item.hypothetical
407 if weight > 0 and flexible:
408 unfrozen.append(index)
409 while unfrozen:
410 free = main_length - gaps - sum(
411 (item.base if index in unfrozen else item.size) + item.outer
412 for index, item in enumerate(line)
413 )
414 amount = max(0, free) if growing else max(0, -free)
415 shares = distribute(amount, [weights[index] for index in unfrozen])
416 violations = {}
417 for index, share in zip(unfrozen, shares):
418 item = line[index]
419 tentative = item.base + share if growing else item.base - share
420 item.size = clamp(item.element, tentative, horizontal, main_ref)
421 violations[index] = item.size - tentative
422 total = sum(violations.values())
423 if total > 0:
424 unfrozen = [index for index in unfrozen if violations[index] <= 0]
425 elif total < 0:
426 unfrozen = [index for index in unfrozen if violations[index] >= 0]
427 else:
428 unfrozen = []

431def justify_spaces(justify, remaining, count):
432 spaces = [0] * count
433 if remaining <= 0 or justify == 'start':
434 return spaces
435 if justify == 'end':
436 spaces[0] = remaining
437 elif justify == 'center':
438 spaces[0] = remaining // 2
439 else:
440 edge, inner = JUSTIFY_WEIGHTS[justify]
441 weights = [edge] + [inner] * (count - 1) + [edge]
442 if sum(weights) > 0:
443 spaces = distribute(remaining, weights)[:count]
444 return spaces

466def flex_layout(container, content_width, content_height, height_ref):
467 direction = container.style['flex-direction']
468 horizontal = direction in ('row', 'row-reverse')
469 row_gap = container.number('row-gap')
470 gap = container.number('column-gap') if horizontal else row_gap
471 main_ref = content_width if horizontal else content_height
472 items = [
473 flex_item(container, element, horizontal, content_width, content_height)
474 for element in container.items()
475 ]
476
477 if horizontal:
478 main_length = content_width
479 elif content_height is not None:
480 main_length = content_height
481 else:
482 total = sum(item.hypothetical + item.outer for item in items) + gap_total(gap, len(items))
483 main_length = clamp(container, total, False, height_ref)
484
485 lines = []
486 if horizontal and container.style['flex-wrap'] == 'wrap':
487 for item in items:
488 if lines:
489 used = sum(other.hypothetical + other.outer for other in lines[-1])
490 if used + gap * len(lines[-1]) + item.hypothetical + item.outer <= main_length:
491 lines[-1].append(item)
492 continue
493 lines.append([item])
494 elif items:
495 lines.append(items)
496
497 reverse = direction.endswith('-reverse')
498 justify = container.style['justify-content']
499 cross_start = 0
500 line_crosses = []
501 for line in lines:
502 resolve_main_sizes(line, main_length, gap, horizontal, main_ref)
503 if horizontal:
504 for item in line:
505 layout(item.element, content_width, content_height, item.size)
506 item.cross = item.element.ch

508 if len(lines) > 1:
509 line_cross = largest
510 elif content_height is not None:
511 line_cross = content_height
512 else:
513 line_cross = clamp(container, largest, False, height_ref)
514 else:
515 line_cross = content_width
516
517 positions = main_positions(line, main_length, gap, justify, reverse, horizontal)
518 for item, main_position in zip(line, positions):
519 element = item.element
520 if not horizontal:
521 layout(element, content_width, content_height, item.cross, item.size)
522 elif item.align == 'stretch' and element.style['height'] == 'auto':
523 stretched = max(0, line_cross - element.outer(False))
524 item.cross = clamp(element, stretched, False, content_height)
525 layout(element, content_width, content_height, item.size, item.cross)
526 else:
527 layout(element, content_width, content_height, item.size)
528 free = line_cross - item.cross - element.outer(not horizontal)
529 offset = align_offset(item.align, free)
530 if horizontal:
531 container.place_child(
532 element, main_position, cross_start + offset + element.margin.top
533 )
534 else:
535 container.place_child(
536 element, cross_start + offset + element.margin.left, main_position
537 )
538 line_crosses.append(line_cross)
539 cross_start += line_cross + row_gap
540
541 if horizontal:
542 return sum(line_crosses) + gap_total(row_gap, len(line_crosses))
543 return main_length

577def place_grid_items(container, items):
578 requests = [
579 (*parse_grid_line(item.style['grid-row']), *parse_grid_line(item.style['grid-column']))
580 for item in items
581 ]
582 column_count = max(
583 [len(column_specs(container))]
584 + [span if start is None else start + span - 1 for _, _, start, span in requests]
585 )
586 occupied = set()
587 areas = [None] * len(items)
588
589 def fits(row, column, row_span, column_span):
590 return occupied.isdisjoint(area_cells(row, column, row_span, column_span))
591
592 def occupy(index, row, column):
593 _, row_span, _, column_span = requests[index]
594 areas[index] = Area(row, column, row_span, column_span)
595 occupied.update(area_cells(row, column, row_span, column_span))
596
597 for index, (row, _, column, _) in enumerate(requests):
598 if row is not None and column is not None:
599 occupy(index, row, column)
600
601 cursor_row, cursor_column = 1, 1
602 for index, (row, row_span, column, column_span) in enumerate(requests):
603 if row is not None and column is not None:
604 continue
605 if row is not None:
606 candidates = range(1, column_count - column_span + 2)
607 column = next(
608 (c for c in candidates if fits(row, c, row_span, column_span)),
609 1,
610 )
611 elif column is not None:
612 if column < cursor_column:
613 cursor_row += 1
614 row = cursor_row
615 while not fits(row, column, row_span, column_span):
616 row += 1
617 cursor_row, cursor_column = row, column + column_span
618 else:
619 row, column = cursor_row, cursor_column
620 while column + column_span - 1 > column_count or not fits(
621 row, column, row_span, column_span
622 ):
623 if column + column_span - 1 > column_count:
624 row += 1
625 column = 1
626 else:
627 column += 1
628 cursor_row, cursor_column = row, column + column_span
629 occupy(index, row, column)
630 return areas, column_count

633def track_sizes(specs, count, available, gap, contributions):
634 tracks = specs + ['auto'] * (count - len(specs))
635 sizes = []
636 for index, spec in enumerate(tracks, 1):
637 if spec == 'auto' or (spec.endswith('fr') and available is None):
638 sizes.append(
639 max(
640 (size for start, span, size in contributions if start == index and span == 1),
641 default=0,
642 )
643 )
644 elif spec.endswith('fr'):
645 sizes.append(0)
646 else:
647 sizes.append(int(spec))
648 if available is not None:
649 leftover = available - sum(sizes) - gap * (len(tracks) - 1)
650 weights = [int(spec[:-2]) if spec.endswith('fr') else 0 for spec in tracks]
651 if not any(weights):
652 weights = [1 if spec == 'auto' else 0 for spec in tracks]
653 if leftover > 0 and any(weights):
654 for index, share in enumerate(distribute(leftover, weights)):
655 sizes[index] += share
656 return sizes

691
692 row_specs = track_specs(container, 'grid-template-rows')
693 row_count = max([len(row_specs)] + [area.row + area.row_span - 1 for area in areas])
694 rows = track_sizes(row_specs, row_count, content_height, row_gap, row_contributions)
695 for item, area, (area_width, width) in zip(items, areas, widths):
696 area_height = span_size(rows, area.row, area.row_span, row_gap)
697 align = item_align(container, item)
698 if align == 'stretch' and item.style['height'] == 'auto':
699 stretched = max(0, area_height - item.outer(False))
700 height = clamp(item, stretched, False, area_height)
701 layout(item, area_width, area_height, width, height)
702 else:
703 layout(item, area_width, area_height, width)
704 offset = align_offset(align, area_height - item.ch - item.outer(False))
705 container.place_child(
706 item,
707 span_start(columns, area.column, column_gap) + item.margin.left,
708 span_start(rows, area.row, row_gap) + offset + item.margin.top,
709 )
710 return sum(rows) + gap_total(row_gap, row_count)

713def max_content_width(element):
714 if element.text is not None:
715 return len(element.text)
716 children = element.in_flow_children()
717 display = element.style['display']
718 gap = element.number('column-gap')
719 if display == 'flex' and element.style['flex-direction'] in ('row', 'row-reverse'):
720 return sum(map(width_contribution, children)) + gap_total(gap, len(children))
721 if display == 'grid':
722 items = element.items()
723 areas, column_count = place_grid_items(element, items)
724 contributions = [
725 (area.column, area.column_span, width_contribution(item))
726 for item, area in zip(items, areas)
727 ]
728 sizes = track_sizes(column_specs(element), column_count, None, gap, contributions)
729 return sum(sizes) + gap_total(gap, column_count)
730 return max(map(width_contribution, children), default=0)

733def width_contribution(element):
734 width = resolve(element, 'width', None, True)
735 if width is None:
736 width = max_content_width(element)
737 return clamp(element, width, True, None) + element.outer(True)

740def positioned_ancestor(element):
741 ancestor = element.parent
742 while ancestor is not None and ancestor.style['position'] == 'static':
743 ancestor = ancestor.parent

747def absolute_offset(start, end, margin_start, margin_end, free):
748 if start is not None:
749 return start + margin_start
750 if end is not None:
751 return free - end - margin_end
752 return margin_start

755def relative_shift(start, end):
756 if start is not None:
757 return start
758 if end is not None:
759 return -end
760 return 0

763def layout_absolute(element, viewport_width, viewport_height):
764 block = positioned_ancestor(element)
765 if block is None:
766 block_width, block_height = viewport_width, viewport_height
767 else:
768 _, _, block_width, block_height = block.padding_box()
769 top, right, bottom, left = (inset(element, side) for side in SIDES)
770
771 width = resolve(element, 'width', block_width, True)
772 if width is None:
773 room = max(0, block_width - (left or 0) - (right or 0) - element.outer(True))
774 if left is not None and right is not None:
775 width = room
776 else:
777 width = min(max_content_width(element), room)
778 width = clamp(element, width, True, block_width)
779 height = None
780 if element.style['height'] == 'auto' and top is not None and bottom is not None:
781 room = max(0, block_height - top - bottom - element.outer(False))
782 height = clamp(element, room, False, block_height)
783 layout(element, block_width, block_height, width, height)
784
785 margin = element.margin
786 element.rel_x = absolute_offset(
787 left, right, margin.left, margin.right, block_width - element.border_width
788 )
789 element.rel_y = absolute_offset(
790 top, bottom, margin.top, margin.bottom, block_height - element.border_height
791 )

794def finalize_position(element):
795 position = element.style['position']
796 if position == 'absolute':
797 block = positioned_ancestor(element)
798 origin_x, origin_y = (0, 0) if block is None else block.padding_box()[:2]
799 elif element.parent is None:
800 origin_x, origin_y = 0, 0
801 else:
802 origin_x, origin_y = element.parent.x, element.parent.y
803 element.x = origin_x + element.rel_x
804 element.y = origin_y + element.rel_y
805 if position == 'relative':
806 top, right, bottom, left = (inset(element, side) for side in SIDES)
807 element.x += relative_shift(left, right)
808 element.y += relative_shift(top, bottom)

811def contains(box, x, y):
812 left, top, width, height = box
813 return left <= x < left + width and top <= y < top + height

816def overflows(element):
817 if element.text is not None:
818 longest = max(map(len, element.lines))
819 return longest > element.cw or len(element.lines) > element.ch
820 left, top, width, height = element.content_box()
821 for child in element.in_flow_children():
822 x, y, w, h = child.margin_box()
823 if x < left or y < top or x + w > left + width or y + h > top + height:
824 return True
825 return False

828def format_box(box):
829 return ' '.join(map(str, box))

832class Document:
833 def __init__(self, width, height, elements):
834 self.width = width
835 self.height = height
836 self.elements = elements
837 self.by_id = {element.id: element for element in elements}
838 self.perform_layout()
839
840 def perform_layout(self):
841 for element in self.elements:
842 parent_displayed = element.parent is None or element.parent.displayed
843 element.displayed = parent_displayed and element.style['display'] != 'none'
844 root = self.elements[0]
845 if root.displayed and root.style['position'] != 'absolute':
846 layout(root, self.width, self.height)
847 root.rel_x, root.rel_y = root.margin.left, root.margin.top
848 for element in self.elements:
849 if element.displayed and element.style['position'] == 'absolute':
850 layout_absolute(element, self.width, self.height)
851 for element in self.elements:
852 if element.displayed:
853 finalize_position(element)
854
855 def snapshot(self):
856 return [element.border_box() if element.displayed else None for element in self.elements]
857
858 def changes(self, before):
859 after = self.snapshot()
860 changed = [
861 element.id for element, old, new in zip(self.elements, before, after) if old != new
862 ]
863 return ' '.join(changed) or 'no change'
864
865 def execute(self, line):
866 name, *args = line.split(' ', 3)
867 match name:
868 case 'LAYOUT':
869 return self.report_layout()
870 case 'BOX':
871 return self.report_box(args[0])
872 case 'RENDER':
873 return self.render()
874 case 'HIT':
875 return self.hit(int(args[0]), int(args[1]))
876 case 'SET':
877 return self.set_property(*args)
878 case 'VIEWPORT':
879 return self.resize(int(args[0]), int(args[1]))
880 case 'OVERFLOW':
881 return self.report_overflow()
882 return [f'error: unknown command {name}']
883
884 def report_layout(self):
885 lines = [f'layout {self.width}x{self.height}']
886 for element in self.elements:
887 indent = ' ' * element.depth
888 box = format_box(element.border_box()) if element.displayed else 'none'
889 lines.append(f'{indent}{element.id} {box}')
890 return lines

892 def report_box(self, element_id):
893 element = self.by_id.get(element_id)
894 if element is None:
895 return [f'error: unknown element {element_id}']
896 if not element.displayed:
897 return [f'box {element_id}: none']
898 return [
899 f'box {element_id}',
900 f'margin {format_box(element.margin_box())}',
901 f'border {format_box(element.border_box())}',
902 f'padding {format_box(element.padding_box())}',
903 f'content {format_box(element.content_box())}',
904 ]

906 def render(self):
907 screen = [['.'] * self.width for _ in range(self.height)]
908 for element in self.elements:
909 if element.displayed:
910 self.draw(screen, element)
911 edge = '#' * (self.width + 2)
912 rows = ['#' + ''.join(row) + '#' for row in screen]
913 return [f'render {self.width}x{self.height}', edge, *rows, edge]
914
915 def draw(self, screen, element):
916 left, top, width, height = element.border_box()
917 right, bottom = left + width - 1, top + height - 1
918 border = element.border
919 for y in range(max(top, 0), min(bottom + 1, self.height)):
920 for x in range(max(left, 0), min(right + 1, self.width)):
921 on_row = (border.top and y == top) or (border.bottom and y == bottom)
922 on_column = (border.left and x == left) or (border.right and x == right)
923 if on_row and on_column:
924 screen[y][x] = '+'
925 elif on_row:
926 screen[y][x] = '-'
927 elif on_column:
928 screen[y][x] = '|'
929 else:
930 screen[y][x] = ' '
931 if element.text is not None:
932 text_left, text_top, _, _ = element.content_box()
933 for row, text_line in enumerate(element.lines):
934 for column, char in enumerate(text_line):
935 x, y = text_left + column, text_top + row
936 if 0 <= x < self.width and 0 <= y < self.height:
937 screen[y][x] = char

939 def hit(self, x, y):
940 if x >= self.width or y >= self.height:
941 return [f'error: point {x} {y} outside viewport']
942 target = None
943 for element in self.elements:
944 if element.displayed and contains(element.border_box(), x, y):
945 target = element
946 if target is None:
947 return [f'hit {x} {y}: none']
948 path = []
949 while target is not None:
950 path.append(target.id)
951 target = target.parent
952 return [f'hit {x} {y}: ' + ' > '.join(reversed(path))]
953
954 def set_property(self, element_id, prop, value):
955 element = self.by_id.get(element_id)
956 if element is None:
957 return [f'error: unknown element {element_id}']
958 if prop not in KNOWN_PROPERTIES:
959 return [f'error: unknown property {prop}']
960 declarations = expand_declaration(prop, value)
961 if declarations is None:
962 return [f'error: invalid value for {prop}: {value}']
963 before = self.snapshot()
964 element.update_style(declarations)
965 self.perform_layout()
966 return [f'set {element_id} {prop} {value}: {self.changes(before)}']
967
968 def resize(self, width, height):
969 if not (1 <= width <= 80 and 1 <= height <= 40):
970 return [f'error: invalid viewport {width} {height}']
971 before = self.snapshot()
972 self.width = width
973 self.height = height
974 self.perform_layout()
975 return [f'viewport {width} {height}: {self.changes(before)}']
976
977 def report_overflow(self):
978 overflowing = [
979 element.id for element in self.elements if element.displayed and overflows(element)
980 ]
981 return ['overflow: ' + (' '.join(overflowing) or 'none')]
