31class BoxLayout
32 SIDES = %w[top right bottom left].freeze
33 DEFAULTS = {
34 'display' => 'block', 'position' => 'static', 'box-sizing' => 'content-box',
35 'width' => 'auto', 'height' => 'auto', 'min-width' => '0', 'min-height' => '0',
36 'max-width' => 'none', 'max-height' => 'none', 'top' => 'auto', 'right' => 'auto',
37 'bottom' => 'auto', 'left' => 'auto', 'flex-direction' => 'row', 'flex-wrap' => 'nowrap',
38 'justify-content' => 'start', 'align-items' => 'stretch', 'align-self' => 'auto',
39 'row-gap' => '0', 'column-gap' => '0', 'order' => '0', 'flex-grow' => '0',
40 'flex-shrink' => '1', 'flex-basis' => 'auto', 'grid-template-columns' => 'none',
41 'grid-template-rows' => 'none', 'grid-column' => 'auto', 'grid-row' => 'auto'
42 }.merge(%w[margin padding border].product(SIDES).to_h { |prefix, side| ["#{prefix}-#{side}", '0'] }).freeze
43 CHOICES = {
44 'display' => %w[block flex grid none], 'position' => %w[static relative absolute],
45 'box-sizing' => %w[content-box border-box], 'flex-direction' => %w[row row-reverse column column-reverse],
46 'flex-wrap' => %w[nowrap wrap], 'justify-content' => %w[start end center space-between space-around space-evenly],
47 'align-items' => %w[stretch start end center], 'align-self' => %w[auto stretch start end center]
48 }.freeze

56 def self.integer?(text)
57 /\A[0-9]{1,3}\z/.match?(text)
58 end

60 def self.positive?(text)
61 integer?(text) && text.to_i.positive?

64 def self.signed?(text)
65 integer?(text) || (text.start_with?('-') && integer?(text[1..]))
66 end

68 def self.percentage?(text)
69 text.end_with?('%') && integer?(text[0...-1])
70 end

72 def self.fraction?(text)
73 text.end_with?('fr') && positive?(text[0...-2])

76 def self.known?(property)
77 DEFAULTS.key?(property) || %w[margin padding border gap].include?(property)
78 end

80 def self.declaration(property, value)
81 return nil unless known?(property)
82 tokens = value.split(' ')
83 if %w[margin padding border].include?(property)
84 return nil unless (1..4).cover?(tokens.length) && tokens.all? { |t| property == 'border' ? %w[0 1].include?(t) : integer?(t) }
85 a, b, c, d = tokens
86 return SIDES.zip([a, b || a, c || a, d || b || a]).map { |side, v| ["#{property}-#{side}", v] }
87 end
88 if property == 'gap'
89 return nil unless (1..2).cover?(tokens.length) && tokens.all? { |t| integer?(t) }
90 return [['row-gap', tokens[0]], ['column-gap', tokens[-1]]]
91 end
92 if %w[grid-template-columns grid-template-rows].include?(property)
93 valid = value == 'none' || tokens.all? { |t| t == 'auto' || integer?(t) || fraction?(t) }
94 elsif %w[grid-column grid-row].include?(property)
95 valid = value == 'auto' || (tokens.length == 1 && positive?(tokens[0])) ||
96 (tokens.length == 2 && tokens[0] == 'span' && positive?(tokens[1])) ||
97 (tokens.length == 3 && positive?(tokens[0]) && tokens[1] == '/' && positive?(tokens[2]) && tokens[0].to_i < tokens[2].to_i) ||
98 (tokens.length == 4 && positive?(tokens[0]) && tokens[1] == '/' && tokens[2] == 'span' && positive?(tokens[3]))
99 elsif tokens.length == 1
100 valid = if CHOICES.key?(property)
101 CHOICES[property].include?(value)
102 elsif %w[width height flex-basis].include?(property)
103 value == 'auto' || integer?(value) || percentage?(value)
104 elsif %w[min-width min-height].include?(property)
105 integer?(value) || percentage?(value)
106 elsif %w[max-width max-height].include?(property)
107 value == 'none' || integer?(value) || percentage?(value)
108 elsif %w[top right bottom left].include?(property)
109 value == 'auto' || signed?(value)
110 elsif property == 'order'
111 signed?(value)
112 elsif property.start_with?('border-')
113 %w[0 1].include?(value)
114 else
115 integer?(value)
116 end
117 end
118 valid ? [[property, value]] : nil
119 end

130 def resolve(element, property, reference, horizontal)
131 value = element.style[property]
132 return nil if %w[auto none].include?(value) || (value.end_with?('%') && reference.nil?)
133 result = value.end_with?('%') ? reference * value.to_i / 100 : value.to_i
134 element.style['box-sizing'] == 'border-box' ? [0, result - element.frame(horizontal)].max : result
135 end

137 def clamp(element, value, reference, horizontal)
138 axis = horizontal ? 'width' : 'height'
139 lower = resolve(element, 'min-' + axis, reference, horizontal) || 0
140 upper = resolve(element, 'max-' + axis, reference, horizontal)
141 value = [value, upper].min if upper
142 [value, lower].max
143 end

145 def distribute(total, weights)
146 sum = weights.sum
147 return Array.new(weights.length, 0) if sum.zero?
148 result = weights.map { |weight| total * weight / sum }
149 remainder = total - result.sum
150 weights.each_index do |i|
151 break if remainder.zero?
152 if weights[i].positive?
153 result[i] += 1
154 remainder -= 1
155 end
156 end
157 result
158 end

160 def grid_position(value)
161 tokens = value.split(' ')
162 return [0, 1] if value == 'auto'
163 return [0, tokens[1].to_i] if tokens[0] == 'span'
164 start = tokens[0].to_i
165 [start, tokens.length == 1 ? 1 : tokens[2] == 'span' ? tokens[3].to_i : tokens[2].to_i - start]
166 end

168 def placement(element)
169 columns = element.style['grid-template-columns'] == 'none' ? ['auto'] : element.style['grid-template-columns'].split(' ')
170 rows = element.style['grid-template-rows'] == 'none' ? [] : element.style['grid-template-rows'].split(' ')
171 items = children(element, true).map do |child|
172 column, cs = grid_position(child.style['grid-column'])
173 row, rs = grid_position(child.style['grid-row'])
174 { element: child, column: column, row: row, cs: cs, rs: rs }
175 end
176 nc = ([columns.length] + items.map { |i| i[:column].zero? ? i[:cs] : i[:column] + i[:cs] - 1 }).max
177 occupied = {}
178 overlaps = lambda do |item, row, column|
179 (row...(row + item[:rs])).any? { |r| (column...(column + item[:cs])).any? { |c| occupied[[r, c]] } }
180 end
181 occupy = lambda do |item|
182 (item[:row]...(item[:row] + item[:rs])).each { |r| (item[:column]...(item[:column] + item[:cs])).each { |c| occupied[[r, c]] = true } }
183 end
184 items.each { |item| occupy.call(item) if item[:row].positive? && item[:column].positive? }
185 cr = cc = 1
186 items.each do |item|
187 next if item[:row].positive? && item[:column].positive?
188 if item[:row].positive?
189 column = (1..(nc - item[:cs] + 1)).find { |c| !overlaps.call(item, item[:row], c) }
190 item[:column] = column || 1
191 elsif item[:column].positive?
192 cr += 1 if item[:column] < cc
193 row = cr
194 row += 1 while overlaps.call(item, row, item[:column])
195 item[:row] = cr = row
196 cc = item[:column] + item[:cs]
197 else
198 row, column = cr, cc
199 loop do
200 if column + item[:cs] - 1 > nc
201 row += 1
202 column = 1
203 elsif overlaps.call(item, row, column)
204 column += 1
205 else
206 break
207 end
208 end
209 item[:row], item[:column] = row, column
210 cr, cc = row, column + item[:cs]
211 end
212 occupy.call(item)
213 end
214 nr = ([rows.length] + items.map { |i| i[:row] + i[:rs] - 1 }).max
215 { items: items, columns: columns + Array.new(nc - columns.length, 'auto'), rows: rows + Array.new(nr - rows.length, 'auto') }
216 end

218 def track_sizes(grid, horizontal, available, gap, contributions)
219 specifications = grid[horizontal ? :columns : :rows]
220 fractional = specifications.any? { |t| self.class.fraction?(t) }
221 weights = []
222 sizes = specifications.each_with_index.map do |specification, i|
223 weights << (fractional ? (self.class.fraction?(specification) ? specification.to_i : 0) : (specification == 'auto' ? 1 : 0))
224 if self.class.integer?(specification)
225 specification.to_i
226 elsif specification == 'auto' || available.nil?
227 grid[:items].each_index.select { |j| grid[:items][j][horizontal ? :column : :row] == i + 1 && grid[:items][j][horizontal ? :cs : :rs] == 1 }.map { |j| contributions[j] }.max || 0
228 else
229 0
230 end
231 end
232 if available
233 leftover = available - sizes.sum - gap * (sizes.length - 1)
234 additions = distribute([leftover, 0].max, weights)
235 sizes = sizes.zip(additions).map(&:sum)
236 end
237 sizes
238 end

240 def max_content(element)
241 return element.maximum unless element.maximum.nil?
242 items = children(element)
243 value = if element.text
244 element.text.length
245 elsif element.style['display'] == 'flex' && element.style['flex-direction'].start_with?('row')
246 items.empty? ? 0 : items.sum { |e| contribution(e) } + element.style['column-gap'].to_i * (items.length - 1)
247 elsif element.style['display'] == 'grid'
248 grid = placement(element)
249 track_sizes(grid, true, nil, element.style['column-gap'].to_i, grid[:items].map { |i| contribution(i[:element]) }).sum + element.style['column-gap'].to_i * (grid[:columns].length - 1)
250 else
251 items.map { |e| contribution(e) }.max || 0
252 end
253 element.maximum = value
254 end

265 def layout(element, cw_reference, ch_reference, fw = nil, fh = nil)
266 element.cw = fw || clamp(element, resolve(element, 'width', cw_reference, true) || [0, cw_reference - element.frame(true) - element.margins(true)].max, cw_reference, true)
267 unless fh
268 height = resolve(element, 'height', ch_reference, false)
269 fh = clamp(element, height, ch_reference, false) if height
270 end
271 if element.text
272 element.lines = []
273 element.text.split(' ').each do |word|
274 if !element.lines.empty? && element.lines.last.length + 1 + word.length <= element.cw
275 element.lines[-1] += ' ' + word
276 else
277 element.lines << word
278 end
279 end
280 natural = element.lines.length
281 else
282 natural = case element.style['display']
283 when 'flex' then flex(element, fh, ch_reference)
284 when 'grid' then grid_layout(element, fh)
285 else block(element, fh)
286 end
287 end
288 element.natural = natural
289 element.ch = fh || clamp(element, natural, ch_reference, false)
290 natural
291 end

293 def block(element, reference)
294 bottom = 0
295 previous = nil
296 children(element).each do |child|
297 layout(child, element.cw, reference)
298 y = bottom + (previous ? [previous.margin[2], child.margin[0]].max : child.margin[0])
299 position_child(element, child, child.margin[3], y)
300 bottom = y + child.height
301 previous = child
302 end
303 previous ? bottom + previous.margin[2] : 0
304 end

306 def flex(element, reference, height_reference)
307 horizontal = element.style['flex-direction'].start_with?('row')
308 reverse = element.style['flex-direction'].end_with?('reverse')
309 main_reference = horizontal ? element.cw : reference
310 gap = element.style[horizontal ? 'column-gap' : 'row-gap'].to_i
311 line_gap = element.style['row-gap'].to_i
312 items = children(element, true).map do |child|
313 alignment = align(element, child)
314 cross = nil
315 unless horizontal
316 room = [0, element.cw - child.frame(true) - child.margins(true)].max
317 cross = resolve(child, 'width', element.cw, true)
318 cross ||= alignment == 'stretch' ? room : [max_content(child), room].min
319 cross = clamp(child, cross, element.cw, true)
320 end
321 base = resolve(child, 'flex-basis', main_reference, horizontal) || resolve(child, horizontal ? 'width' : 'height', main_reference, horizontal)
322 base ||= horizontal ? max_content(child) : layout(child, element.cw, reference, cross)
323 hypothetical = clamp(child, base, main_reference, horizontal)
324 { element: child, align: alignment, cross: cross, base: base, hypothetical: hypothetical, outer: child.frame(horizontal) + child.margins(horizontal) }
325 end
326 main_length = if horizontal
327 element.cw
328 elsif reference
329 reference
330 else
331 clamp(element, items.sum { |i| i[:hypothetical] + i[:outer] } + gap * [items.length - 1, 0].max, height_reference, false)
332 end
333 lines = []
334 items.each do |item|
335 current = lines.last
336 if current.nil? || (horizontal && element.style['flex-wrap'] == 'wrap' && current.sum { |i| i[:hypothetical] + i[:outer] } + gap * current.length + item[:hypothetical] + item[:outer] > main_length)
337 lines << []
338 end
339 lines.last << item
340 end
341 cross_start = 0
342 lines.each do |line|
343 total_gap = gap * (line.length - 1)
344 growing = line.sum { |i| i[:hypothetical] + i[:outer] } + total_gap < main_length
345 line.each do |item|
346 child = item[:element]
347 item[:weight] = growing ? child.style['flex-grow'].to_i : child.style['flex-shrink'].to_i * item[:base]
348 item[:size] = item[:hypothetical]
349 item[:frozen] = item[:weight].zero? || (growing ? item[:base] > item[:hypothetical] : item[:base] < item[:hypothetical])
350 end
351 until line.all? { |i| i[:frozen] }
352 free = main_length - total_gap - line.sum { |i| (i[:frozen] ? i[:size] : i[:base]) + i[:outer] }
353 active = line.reject { |i| i[:frozen] }
354 assigned = distribute([growing ? free : -free, 0].max, active.map { |i| i[:weight] })
355 active.each_with_index do |item, i|
356 tentative = item[:base] + (growing ? assigned[i] : -assigned[i])
357 item[:size] = clamp(item[:element], tentative, main_reference, horizontal)
358 item[:violation] = item[:size] - tentative
359 end
360 violation = active.sum { |i| i[:violation] }
361 active.each { |i| i[:frozen] = true if violation.zero? || (violation.positive? ? i[:violation].positive? : i[:violation].negative?) }
362 end
363 if horizontal
364 line.each do |item|
365 layout(item[:element], element.cw, reference, item[:size])
366 item[:cross] = item[:element].ch
367 end
368 end
369 outer_cross = line.map { |i| i[:cross] + i[:element].frame(!horizontal) + i[:element].margins(!horizontal) }.max || 0
370 line_cross = if !horizontal
371 element.cw
372 elsif lines.length == 1
373 reference || clamp(element, outer_cross, height_reference, false)
374 else
375 outer_cross
376 end
377 remaining = main_length - total_gap - line.sum { |i| i[:size] + i[:outer] }
378 spaces = Array.new(line.length, 0)
379 if remaining.positive?
380 case element.style['justify-content']
381 when 'end' then spaces[0] = remaining
382 when 'center' then spaces[0] = remaining / 2
383 when 'space-between', 'space-around', 'space-evenly'
384 kind = element.style['justify-content']
385 weights = Array.new(line.length + 1) { |i| kind == 'space-evenly' ? 1 : i.zero? || i == line.length ? (kind == 'space-between' ? 0 : 1) : (kind == 'space-between' ? 1 : 2) }
386 spaces = distribute(remaining, weights).take(line.length)
387 end
388 end
389 cursor = 0
390 line.each_with_index do |item, i|
391 child = item[:element]
392 cross = item[:cross]
393 if horizontal
394 if item[:align] == 'stretch' && child.style['height'] == 'auto'
395 cross = clamp(child, [0, line_cross - child.frame(false) - child.margins(false)].max, reference, false)
396 layout(child, element.cw, reference, item[:size], cross)
397 else
398 layout(child, element.cw, reference, item[:size])
399 end
400 else
401 layout(child, element.cw, reference, cross, item[:size])
402 end
403 room = line_cross - cross - child.frame(!horizontal) - child.margins(!horizontal)
404 offset = room.positive? ? (item[:align] == 'end' ? room : item[:align] == 'center' ? room / 2 : 0) : 0
405 cursor += spaces[i]
406 size = item[:size] + item[:outer]
407 main = reverse ? main_length - cursor - size + child.margin[horizontal ? 3 : 0] : cursor + child.margin[horizontal ? 3 : 0]
408 cross_position = cross_start + offset + child.margin[horizontal ? 0 : 3]
409 position_child(element, child, horizontal ? main : cross_position, horizontal ? cross_position : main)
410 cursor += size + gap
411 end
412 cross_start += line_cross + line_gap
413 end
414 horizontal ? (lines.empty? ? 0 : cross_start - line_gap) : main_length
415 end

417 def grid_layout(element, reference)
418 grid = placement(element)
419 column_gap = element.style['column-gap'].to_i
420 row_gap = element.style['row-gap'].to_i
421 columns = track_sizes(grid, true, element.cw, column_gap, grid[:items].map { |i| contribution(i[:element]) })
422 grid[:items].each do |item|
423 child = item[:element]
424 start = item[:column] - 1
425 item[:aw] = columns[start, item[:cs]].sum + column_gap * (item[:cs] - 1)
426 item[:ax] = columns.take(start).sum + column_gap * start
427 item[:gw] = clamp(child, resolve(child, 'width', item[:aw], true) || [0, item[:aw] - child.frame(true) - child.margins(true)].max, item[:aw], true)
428 layout(child, item[:aw], nil, item[:gw])
429 item[:row_contribution] = child.height + child.margins(false)
430 end
431 rows = track_sizes(grid, false, reference, row_gap, grid[:items].map { |i| i[:row_contribution] })
432 grid[:items].each do |item|
433 child = item[:element]
434 start = item[:row] - 1
435 area_height = rows[start, item[:rs]].sum + row_gap * (item[:rs] - 1)
436 area_y = rows.take(start).sum + row_gap * start
437 alignment = align(element, child)
438 height = alignment == 'stretch' && child.style['height'] == 'auto' ? clamp(child, [0, area_height - child.frame(false) - child.margins(false)].max, area_height, false) : nil
439 layout(child, item[:aw], area_height, item[:gw], height)
440 room = area_height - child.height - child.margins(false)
441 offset = room.positive? ? (alignment == 'end' ? room : alignment == 'center' ? room / 2 : 0) : 0
442 position_child(element, child, item[:ax] + child.margin[3], area_y + offset + child.margin[0])
443 end
444 rows.empty? ? 0 : rows.sum + row_gap * (rows.length - 1)
445 end

447 def relayout
448 @elements.each do |element|
449 element.style = DEFAULTS.dup
450 priorities = {}
451 @rules.each do |selectors, declarations|
452 matches = selectors.filter_map do |selector|
453 selector.start_with?('#') ? (selector[1..] == element.id ? 2 : nil) : selector.start_with?('.') ? (element.classes.include?(selector[1..]) ? 1 : nil) : (selector == element.type ? 0 : nil)
454 end
455 next if matches.empty?
456 specificity = matches.max
457 declarations.each do |property, value|
458 if !priorities.key?(property) || specificity >= priorities[property]
459 element.style[property] = value
460 priorities[property] = specificity
461 end
462 end
463 end
464 element.style.merge!(element.overrides)
465 element.margin, element.padding, element.border = %w[margin padding border].map { |prefix| SIDES.map { |side| element.style["#{prefix}-#{side}"].to_i } }
466 element.displayed = element.style['display'] != 'none' && (!element.parent || element.parent.displayed)
467 element.maximum = element.contribution = nil
468 element.lines = []
469 element.containing = nil
470 end
471 root = @elements.first
472 if root.displayed && root.style['position'] != 'absolute'
473 layout(root, @width, @height)
474 root.rx, root.ry = root.margin[3], root.margin[0]
475 end
476 @elements.each do |element|
477 next unless element.displayed && element.style['position'] == 'absolute'
478 ancestor = element.parent
479 ancestor = ancestor.parent while ancestor && ancestor.style['position'] == 'static'
480 element.containing = ancestor
481 cbw = ancestor ? ancestor.cw + ancestor.padding[1] + ancestor.padding[3] : @width
482 cbh = ancestor ? ancestor.ch + ancestor.padding[0] + ancestor.padding[2] : @height
483 offsets = %w[left right top bottom].map { |p| element.style[p] == 'auto' ? nil : element.style[p].to_i }
484 left, right, top, bottom = offsets
485 width = resolve(element, 'width', cbw, true)
486 unless width
487 room = [0, cbw - (left || 0) - (right || 0) - element.margins(true) - element.frame(true)].max
488 width = left && right ? room : [max_content(element), room].min
489 end
490 fw = clamp(element, width, cbw, true)
491 fh = element.style['height'] == 'auto' && top && bottom ? clamp(element, [0, cbh - top - bottom - element.margins(false) - element.frame(false)].max, cbh, false) : nil
492 layout(element, cbw, cbh, fw, fh)
493 element.rx = left ? left + element.margin[3] : right ? cbw - right - element.margin[1] - element.width : element.margin[3]
494 element.ry = top ? top + element.margin[0] : bottom ? cbh - bottom - element.margin[2] - element.height : element.margin[0]
495 end
496 @elements.each do |element|
497 next unless element.displayed
498 reference = element.style['position'] == 'absolute' ? element.containing : element.parent
499 x = reference ? reference.x : 0
500 y = reference ? reference.y : 0
501 if element.style['position'] == 'absolute' && reference
502 x += reference.border[3]
503 y += reference.border[0]
504 end
505 element.x, element.y = x + element.rx, y + element.ry
506 if element.style['position'] == 'relative'
507 left, right, top, bottom = %w[left right top bottom].map { |p| element.style[p] == 'auto' ? nil : element.style[p].to_i }
508 element.x += left || (right ? -right : 0)
509 element.y += top || (bottom ? -bottom : 0)
510 end
511 end
512 end

514 def boxes(element)
515 x, y, w, h = element.x, element.y, element.width, element.height
516 m, b, p = element.margin, element.border, element.padding
517 [[x - m[3], y - m[0], w + m[1] + m[3], h + m[0] + m[2]], [x, y, w, h],
518 [x + b[3], y + b[0], w - b[1] - b[3], h - b[0] - b[2]],

522 def snapshot
523 @elements.map { |e| e.displayed ? [e.x, e.y, e.width, e.height] : nil }
524 end

526 def changes(before)
527 changed = @elements.each_index.select { |i| before[i] != snapshot[i] }.map { |i| @elements[i].id }
528 changed.empty? ? 'no change' : changed.join(' ')
529 end

531 def render
532 screen = Array.new(@height) { '.' * @width }
533 @elements.each do |element|
534 next unless element.displayed
535 x, y, w, h = boxes(element)[1]
536 ([y, 0].max...[y + h, @height].min).each do |row|
537 ([x, 0].max...[x + w, @width].min).each do |column|
538 horizontal = (element.border[0] == 1 && row == y) || (element.border[2] == 1 && row == y + h - 1)
539 vertical = (element.border[3] == 1 && column == x) || (element.border[1] == 1 && column == x + w - 1)
540 screen[row][column] = horizontal && vertical ? '+' : horizontal ? '-' : vertical ? '|' : ' '
541 end
542 end
543 next unless element.text
544 cx, cy = boxes(element)[3].take(2)
545 element.lines.each_with_index do |line, i|
546 row = cy + i
547 next unless (0...@height).cover?(row)
548 line.each_char.with_index { |char, j| screen[row][cx + j] = char if (0...@width).cover?(cx + j) }
549 end
550 end
551 puts "render #{@width}x#{@height}", '#' * (@width + 2)
552 screen.each { |row| puts '#' + row + '#' }
553 puts '#' * (@width + 2)
554 end

556 def execute(line)
557 command, *args = line.split(' ')
558 case command
559 when 'LAYOUT'
560 puts "layout #{@width}x#{@height}"
561 @elements.each { |e| puts ' ' * e.depth + e.id + (e.displayed ? " #{e.x} #{e.y} #{e.width} #{e.height}" : ' none') }
562 when 'BOX'
563 element = @ids[args[0]]
564 if !element
565 puts "error: unknown element #{args[0]}"
566 elsif !element.displayed
567 puts "box #{element.id}: none"
568 else
569 puts "box #{element.id}"
570 %w[margin border padding content].zip(boxes(element)).each { |name, rect| puts name + ' ' + rect.join(' ') }
571 end
572 when 'RENDER'
573 render
574 when 'HIT'
575 x, y = args.map(&:to_i)
576 if x >= @width || y >= @height
577 puts "error: point #{x} #{y} outside viewport"
578 else
579 element = @elements.reverse.find { |e| e.displayed && e.x <= x && x < e.x + e.width && e.y <= y && y < e.y + e.height }
580 path = []
581 while element
582 path.unshift(element.id)
583 element = element.parent
584 end
585 puts "hit #{x} #{y}: #{path.empty? ? 'none' : path.join(' > ')}"
586 end
587 when 'SET'
588 id, property = args.take(2)
589 value = args.drop(2).join(' ')
590 if !@ids.key?(id)
591 puts "error: unknown element #{id}"
592 elsif !self.class.known?(property)
593 puts "error: unknown property #{property}"
594 elsif !(declarations = self.class.declaration(property, value))
595 puts "error: invalid value for #{property}: #{value}"
596 else
597 before = snapshot
598 declarations.each { |p, v| @ids[id].overrides[p] = v }
599 relayout
600 puts "set #{id} #{property} #{value}: #{changes(before)}"
601 end
602 when 'VIEWPORT'
603 width, height = args.map(&:to_i)
604 if !(1..80).cover?(width) || !(1..40).cover?(height)
605 puts "error: invalid viewport #{width} #{height}"
606 else
607 before = snapshot
608 @width, @height = width, height
609 relayout
610 puts "viewport #{width} #{height}: #{changes(before)}"
611 end
612 when 'OVERFLOW'
613 overflow = @elements.select do |e|
614 next false unless e.displayed
615 if e.text
616 e.lines.map(&:length).max > e.cw || e.lines.length > e.ch
617 else
618 cx, cy, cw, ch = boxes(e)[3]
619 children(e).any? do |child|
620 x, y, w, h = boxes(child)[0]
621 x < cx || y < cy || x + w > cx + cw || y + h > cy + ch
622 end
623 end
624 end.map(&:id)
625 puts 'overflow: ' + (overflow.empty? ? 'none' : overflow.join(' '))
626 else
627 puts "error: unknown command #{command}"
628 end
629 end
630end

632input = STDIN.each_line.map(&:chomp)
633width, height = input.shift.split(' ').map(&:to_i)
634rules = []
635until input.first == '---'
636 selectors, body = input.shift.split(' { ', 2)
637 declarations = body.delete_suffix(' }').split('; ').flat_map do |text|
638 property, value = text.split(': ', 2)
639 BoxLayout.declaration(property, value) || []
640 end
641 rules << [selectors.split(', '), declarations]
642end
643input.shift
644elements = []
645stack = []
646until input.first == '==='
647 line = input.shift
648 depth = line[/\A */].length / 2
649 head, text = line.lstrip.split(' ', 2)
650 type, rest = head.split('#', 2)
651 id, *classes = rest.split('.')
652 parent = depth.zero? ? nil : stack[depth - 1]
653 element = Element.new(type, id, classes, text ? text[1...-1] : nil, parent, depth)
654 parent.children << element if parent
655 elements << element
656 stack[depth] = element
657end
658input.shift
659layout = BoxLayout.new(width, height, rules, elements)
660input.each do |line|
661 break if line == 'END'
662 layout.execute(line)
663end
