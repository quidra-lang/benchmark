44natural_token(s) = occursin(r"\A[0-9]{1,3}\z", s)
45positive_token(s) = natural_token(s) && parse(Int, s)>0
46signed_token(s) = natural_token(startswith(s, "-") ? s[2:end] : s)
47percent_token(s) = endswith(s, "%") && natural_token(s[1:(end-1)])
48fr_token(s) = endswith(s, "fr") && positive_token(s[1:(end-2)])
49function gridline_valid(t)
50 t==["auto"] && return true
51 length(t)==1 && return positive_token(t[1])
52 length(t)==2 && return t[1]=="span" && positive_token(t[2])
53 length(t)==3 && return positive_token(t[1]) &&
54 t[2]=="/" &&
55 positive_token(t[3]) &&
56 parse(Int, t[1])<parse(Int, t[3])
57 length(t)==4 && return positive_token(t[1]) && t[2]=="/" && t[3]=="span" && positive_token(t[4])
58 false
59end
60known_property(p) = haskey(INITIAL, p) || p in ("margin", "padding", "border", "gap")
61function valid_longhand(p, t)
62 if p in ("grid-template-columns", "grid-template-rows")
63 return t==["none"] || all(s->s=="auto" || natural_token(s) || fr_token(s), t)
64 end
65 p in ("grid-column", "grid-row") && return gridline_valid(t)
66 length(t)==1 || return false
67 s=t[1]
68 haskey(CHOICES, p) && return s in CHOICES[p]
69 p in ("width", "height", "flex-basis") &&
70 return s=="auto" || natural_token(s) || percent_token(s)
71 p in ("min-width", "min-height") && return natural_token(s) || percent_token(s)
72 p in ("max-width", "max-height") && return s=="none" || natural_token(s) || percent_token(s)
73 startswith(p, "border-") && return s in ("0", "1")
74 p in SIDES && return s=="auto" || signed_token(s)
75 p=="order" && return signed_token(s)
76 natural_token(s)
77end
78function expanddecl(p, value)
79 t=String.(split(value, ' '))
80 if haskey(INITIAL, p)
81 return valid_longhand(p, t) ? [p=>value] : nothing
82 end
83 if p=="gap"
84 1<=length(t)<=2 && all(natural_token, t) || return nothing
85 return ["row-gap"=>t[1], "column-gap"=>t[end]]
86 end

88 1<=length(t)<=4 && all(s->valid_longhand("$p-top", [s]), t) || return nothing
89 picks=((1, 1, 1, 1), (1, 2, 1, 2), (1, 2, 3, 2), (1, 2, 3, 4))[length(t)]

91 end

163function distribute(total, weights)
164 denominator=sum(weights)
165 shares=[div(total*w, denominator) for w in weights]
166 remainder=total-sum(shares)
167 for i in eachindex(weights)
168 if remainder>0 && weights[i]>0
169 shares[i]+=1
170 remainder-=1
171 end
172 end
173 shares
174end

193function gridplacement(d, e, ordered)
194 requests=[
195 (
196 gridrequest(d.elements[i].style["grid-row"])...,
197 gridrequest(d.elements[i].style["grid-column"])...,
198 ) for i in ordered
199 ]
200 nc=maximum([length(columns(e)); [c==0 ? cs : c+cs-1 for (_, _, c, cs) in requests]])
201 areas=fill((0, 0, 0, 0), length(ordered))
202 occupied=Set{Tuple{Int,Int}}()
203 fits(r, c, rs, cs) = all(!((y, x) in occupied) for y in r:(r+rs-1) for x in c:(c+cs-1))
204 function occupy!(i, r, c)
205 _, rs, _, cs=requests[i]
206 areas[i]=(r, c, rs, cs)
207 for y in r:(r+rs-1), x in c:(c+cs-1)
208 push!(occupied, (y, x))
209 end
210 end
211 for (i, (r, _, c, _)) in enumerate(requests)
212 r!=0 && c!=0 && occupy!(i, r, c)
213 end
214 cr=1
215 cc=1
216 for (i, (r, rs, c, cs)) in enumerate(requests)
217 r!=0 && c!=0 && continue
218 if r!=0
219 c=1
220 for candidate in 1:(nc-cs+1)
221 if fits(r, candidate, rs, cs)
222 c=candidate
223 break
224 end
225 end
226 elseif c!=0
227 c<cc && (cr+=1)
228 r=cr
229 while !fits(r, c, rs, cs)
230 r+=1
231 end
232 cr=r
233 cc=c+cs
234 else
235 r=cr
236 c=cc
237 while true
238 if c+cs-1>nc
239 r+=1
240 c=1
241 elseif !fits(r, c, rs, cs)
242 c+=1
243 else
244 break
245 end
246 end
247 cr=r
248 cc=c+cs
249 end
250 occupy!(i, r, c)
251 end
252 areas, nc
253end
254function tracksizes(specs, n, available, gap, contributions)
255 specifications=[specs; fill("auto", n-length(specs))]
256 sizes=zeros(Int, n)
257 for (i, s) in enumerate(specifications)
258 if s=="auto" || (endswith(s, "fr") && available===nothing)
259 sizes[i]=maximum(
260 (v for (start, span, v) in contributions if start==i && span==1);
261 init = 0,
262 )
263 elseif !endswith(s, "fr")
264 sizes[i]=parse(Int, s)
265 end
266 end
267 if available!==nothing
268 leftover=available-sum(sizes)-gap*(n-1)
269 weights=[endswith(s, "fr") ? parse(Int, s[1:(end-2)]) : 0 for s in specifications]
270 all(==(0), weights) && (weights=Int[s=="auto" for s in specifications])
271 leftover>0 && any(>(0), weights) && (sizes.+=distribute(leftover, weights))
272 end
273 sizes
274end

349function main_sizes!(d, line, lengthmain, gap, horizontal, reference)
350 gaps=gaptotal(gap, length(line))
351 growing=sum(it.hypothetical+it.outside for it in line)+gaps<lengthmain
352 weights=[
353 parse(Int, d.elements[it.index].style[growing ? "flex-grow" : "flex-shrink"])*(
354 growing ? 1 : it.base
355 ) for it in line
356 ]
357 active=Int[]
358 for (j, it) in enumerate(line)
359 it.size=it.hypothetical
360 weights[j]>0 &&
361 (growing ? it.base<=it.hypothetical : it.base>=it.hypothetical) &&
362 push!(active, j)
363 end
364 while !isempty(active)
365 free=lengthmain-gaps-sum(
366 (j in active ? it.base : it.size)+it.outside for (j, it) in enumerate(line)
367 )
368 shares=distribute(max(0, growing ? free : -free), weights[active])
369 violations=zeros(Int, length(line))
370 for (j, share) in zip(active, shares)
371 it=line[j]
372 tentative=it.base+(growing ? share : -share)
373 it.size=constrain(d.elements[it.index], tentative, horizontal, reference)
374 violations[j]=it.size-tentative
375 end
376 total=sum(violations)
377 active=total>0 ? [j for j in active if violations[j]<=0] :
378 total<0 ? [j for j in active if violations[j]>=0] : Int[]
379 end
380end
381function justifyspaces(justify, remaining, n)
382 spaces=zeros(Int, n)
383 remaining<=0 && return spaces
384 if justify=="end"
385 spaces[1]=remaining
386 elseif justify=="center"
387 spaces[1]=div(remaining, 2)
388 elseif justify!="start"
389 ends, middle=justify=="space-between" ? (0, 1) : justify=="space-around" ? (1, 2) : (1, 1)
390 weights=[ends; fill(middle, n-1); ends]
391 sum(weights)>0 && (spaces=distribute(remaining, weights)[1:n])
392 end
393 spaces
394end

592 fh=nothing
593 if e.style["height"]=="auto" && top!==nothing && bottom!==nothing
594 fh=constrain(e, max(0, bh-top-bottom-outer(e, false)), false, bh)
595 end
596 layout!(d, i, bw, bh, fw, fh)
597 e.rx=absoffset(
598 left,
599 right,
600 edge(e, "margin", "left"),
601 edge(e, "margin", "right"),
602 bw-borderwidth(e),
603 )
604 e.ry=absoffset(
605 top,
606 bottom,
607 edge(e, "margin", "top"),
608 edge(e, "margin", "bottom"),
609 bh-borderheight(e),
610 )
611 end
612 for (i, e) in enumerate(d.elements)
613 e.displayed || continue
614 p=e.style["position"]=="absolute" ? ancestor(d, i) : e.parent
615 ox, oy=p==0 ? (0, 0) :
616 e.style["position"]=="absolute" ? paddingbox(d.elements[p])[1:2] :
617 (d.elements[p].x, d.elements[p].y)
618 e.x=ox+e.rx
619 e.y=oy+e.ry
620 if e.style["position"]=="relative"
621 e.x+=relshift(inset(e, "left"), inset(e, "right"))
622 e.y+=relshift(inset(e, "top"), inset(e, "bottom"))
623 end
624 end
625end
626snapshot(d) = [e.displayed ? borderbox(e) : nothing for e in d.elements]
627function changes(d, before)
628 ids=[e.id for (e, old, new) in zip(d.elements, before, snapshot(d)) if old!=new]
629 isempty(ids) ? "no change" : join(ids, ' ')
630end
631boxtext(box) = join(box, ' ')
632function overflows(d, e)
633 e.text!==nothing && return maximum(length, e.lines)>e.cw || length(e.lines)>e.ch
634 x, y, w, h=contentbox(e)
635 for j in flow(d, e)
636 a, b, c, f=marginbox(d.elements[j])
637 (a<x || b<y || a+c>x+w || b+f>y+h) && return true
638 end
639 false
640end
641function render(d)
642 screen=fill('.', d.height, d.width)
643 for e in d.elements
644 e.displayed || continue
645 x, y, w, h=borderbox(e)
646 for yy in max(0, y):(min(d.height, y+h)-1), xx in max(0, x):(min(d.width, x+w)-1)
647 onrow=(edge(e, "border", "top")==1 && yy==y) ||
648 (edge(e, "border", "bottom")==1 && yy==y+h-1)
649 oncol=(edge(e, "border", "left")==1 && xx==x) ||
650 (edge(e, "border", "right")==1 && xx==x+w-1)
651 screen[yy+1, xx+1]=onrow ? (oncol ? '+' : '-') : (oncol ? '|' : ' ')
652 end
653 if e.text!==nothing
654 tx, ty, _, _=contentbox(e)
655 for (row, text) in enumerate(e.lines), (col, c) in enumerate(text)
656 xx=tx+col-1
657 yy=ty+row-1
658 0<=xx<d.width && 0<=yy<d.height && (screen[yy+1, xx+1]=c)
659 end
660 end
661 end
662 println("render $(d.width)x$(d.height)")
663 boundary=repeat("#", d.width+2)
664 println(boundary)
665 for row in eachrow(screen)
666 println("#", join(row), "#")
667 end
668 println(boundary)
669end
670function execute!(d, line)
671 p=String.(split(line, ' '; limit = 4))
672 command=p[1]
673 if command=="LAYOUT"
674 println("layout $(d.width)x$(d.height)")
675 for e in d.elements
676 println(repeat(" ", e.depth), e.id, " ", e.displayed ? boxtext(borderbox(e)) : "none")
677 end
678 elseif command=="BOX"
679 i=findfirst(e->e.id==p[2], d.elements)
680 if i===nothing
681 println("error: unknown element $(p[2])")
682 elseif !d.elements[i].displayed
683 println("box $(p[2]): none")
684 else
685 e=d.elements[i]
686 println("box $(e.id)")
687 for (kind, box) in (
688 ("margin", marginbox(e)),
689 ("border", borderbox(e)),
690 ("padding", paddingbox(e)),
691 ("content", contentbox(e)),
692 )
693 println(kind, " ", boxtext(box))
694 end
695 end
696 elseif command=="RENDER"
697 render(d)
698 elseif command=="HIT"
699 x=parse(Int, p[2])
700 y=parse(Int, p[3])
701 if x>=d.width || y>=d.height
702 println("error: point $x $y outside viewport")
703 return
704 end
705 target=0
706 for (i, e) in enumerate(d.elements)
707 a, b, w, h=borderbox(e)
708 e.displayed && a<=x<a+w && b<=y<b+h && (target=i)
709 end
710 path=String[]
711 while target!=0
712 pushfirst!(path, d.elements[target].id)
713 target=d.elements[target].parent
714 end
715 println("hit $x $y: ", isempty(path) ? "none" : join(path, " > "))
716 elseif command=="SET"
717 id, property, value=p[2:4]
718 i=findfirst(e->e.id==id, d.elements)
719 if i===nothing
720 println("error: unknown element $id")
721 return
722 end
723 if !known_property(property)
724 println("error: unknown property $property")
725 return
726 end
727 declarations=expanddecl(property, value)
728 if declarations===nothing
729 println("error: invalid value for $property: $value")
730 return
731 end
732 before=snapshot(d)
733 merge!(d.elements[i].style, Dict(declarations))
734 performlayout!(d)
735 println("set $id $property $value: $(changes(d,before))")
736 elseif command=="VIEWPORT"
737 w=parse(Int, p[2])
738 h=parse(Int, p[3])
739 if !(1<=w<=80 && 1<=h<=40)
740 println("error: invalid viewport $w $h")
741 return
742 end
743 before=snapshot(d)
744 d.width=w
745 d.height=h
746 performlayout!(d)
747 println("viewport $w $h: $(changes(d,before))")
748 elseif command=="OVERFLOW"
749 ids=[e.id for e in d.elements if e.displayed && overflows(d, e)]
750 println("overflow: ", isempty(ids) ? "none" : join(ids, ' '))
751 else
752 println("error: unknown command $command")
753 end
754end
755function main()
756 w, h=parse.(Int, split(readline()))
757 rules=Tuple{Vector{String},Vector{Pair{String,String}}}[]
758 while true
759 line=readline()
760 line=="---" && break
761 selectors, body=split(line, " { "; limit = 2)
762 declarations=Pair{String,String}[]
763 for declaration in split(body[1:(end-2)], "; ")
764 property, value=String.(split(declaration, ": "; limit = 2))
765 expanded=expanddecl(property, value)
766 expanded!==nothing && append!(declarations, expanded)
767 end
768 push!(rules, (String.(split(selectors, ", ")), declarations))
769 end
770 elements=CssElement[]
771 stack=Int[]
772 while true
773 line=readline()
774 line=="===" && break
775 head=lstrip(line, ' ')
776 depth=div(length(line)-length(head), 2)
777 pieces=split(head, ' '; limit = 2)
778 tag, names=split(pieces[1], '#')
779 names=String.(split(names, '.'))
780 text=length(pieces)==2 ? String(pieces[2][2:(end-1)]) : nothing
781 resize!(stack, depth)
782 parent=isempty(stack) ? 0 : stack[end]
783 e=CssElement(
784 String(tag),
785 names[1],
786 names[2:end],
787 text,
788 parent,
789 depth,
790 Int[],
791 copy(INITIAL),
792 false,
793 0,
794 0,
795 String[],
796 0,
797 0,
798 0,
799 0,
800 )
801 winners=Dict{String,Int}()
802 for (selectors, declarations) in rules
803 specificity=-1
804 for selector in selectors
805 score=startswith(selector, "#") ? (selector[2:end]==e.id ? 2 : -1) :
806 startswith(selector, ".") ? (selector[2:end] in e.classes ? 1 : -1) :
807 (selector==e.tag ? 0 : -1)
808 specificity=max(specificity, score)
809 end
810 specificity<0 && continue
811 for (property, value) in declarations
812 if specificity>=get(winners, property, -1)
813 winners[property]=specificity
814 e.style[property]=value
815 end
816 end
817 end
818 push!(elements, e)
819 parent!=0 && push!(elements[parent].children, length(elements))
820 push!(stack, length(elements))
821 end
822 document=Document(w, h, elements, fill(-1, length(elements)))
823 performlayout!(document)
824 while true
825 line=readline()
826 line=="END" && break
827 execute!(document, line)
828 end
829end
830main()
