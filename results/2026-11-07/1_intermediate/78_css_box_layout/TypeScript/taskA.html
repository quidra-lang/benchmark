133function declarations(property: string, value: string): Declaration[] | null {

176 else if (sides.includes(property)) valid = value === "auto" || signed(value);

181function distribute(total: number, weights: number[]): number[] {
182 const sum = weights.reduce((a, b) => a + b, 0);
183 if (sum === 0) return weights.map(() => 0);
184 const result = weights.map((weight) => Math.floor((total * weight) / sum));
185 let remainder = total - result.reduce((a, b) => a + b, 0);
186 for (let i = 0; i < weights.length && remainder > 0; i++)
187 if (weights[i] > 0) {
188 result[i]++;
189 remainder--;
190 }
191 return result;

331 function overlaps(item: GridItem, r: number, c: number): boolean {

337 item.row = r;
338 item.column = c;
339 for (let y = r; y < r + item.rs; y++)
340 for (let x = c; x < c + item.cs; x++) occupied.add(y + ":" + x);

351 put(item, item.row, column <= nc - item.cs + 1 ? column : 1);
352 } else if (item.column !== null) {

358 cc = item.column + item.cs;
359 } else {
360 let row = cr,
361 column = cc;
362 while (true) {
363 if (column + item.cs - 1 > nc) {
364 row++;

366 } else if (overlaps(item, row, column)) column++;
367 else break;
368 }
369 put(item, row, column);
