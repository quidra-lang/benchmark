107static void *allocate(Arena *arena, size_t size) {
108 void *pointer = calloc(1, size ? size : 1);
109 Allocation *allocation = malloc(sizeof *allocation);
110 if (!pointer || !allocation) exit(1);
111 allocation->pointer = pointer;
112 allocation->next = arena->head;
113 arena->head = allocation;
114 return pointer;
115}

117static void clear_arena(Arena *arena) {
118 while (arena->head) {
119 Allocation *next = arena->head->next;
120 free(arena->head->pointer);
121 free(arena->head);
122 arena->head = next;
123 }
124}

136static void add_string(Arena *arena, Strings *strings, char *value) {
137 if (strings->count == strings->capacity) {
138 int capacity = strings->capacity ? strings->capacity * 2 : 8;
139 char **items = allocate(arena, (size_t)capacity * sizeof *items);
140 if (strings->count) memcpy(items, strings->items, (size_t)strings->count * sizeof *items);
141 strings->items = items;
142 strings->capacity = capacity;
143 }
144 strings->items[strings->count++] = value;
145}

147static Strings split(Arena *arena, const char *text, const char *delimiter) {
148 Strings result = {0};
149 size_t length = strlen(delimiter);
150 const char *start = text, *next;
151 while ((next = strstr(start, delimiter))) {
152 add_string(arena, &result, part(arena, start, (size_t)(next - start)));
153 start = next + length;
154 }
155 add_string(arena, &result, copy(arena, start));
156 return result;
157}

159static char *read_line(void) {
160 size_t length = 0, capacity = 128;
161 char *line = malloc(capacity);
162 int ch;
163 if (!line) exit(1);
164 while ((ch = getchar()) != EOF && ch != '\n') {
165 if (length + 1 >= capacity) {
166 capacity *= 2;
167 char *grown = realloc(line, capacity);
168 if (!grown) exit(1);
169 line = grown;
170 }
171 line[length++] = (char)ch;
172 }
173 if (ch == EOF && length == 0) {
174 free(line);
175 return NULL;
176 }
177 line[length] = '\0';
178 return line;
179}

185static bool one_of(const char *value, const char *choices) {
186 size_t length = strlen(value);
187 const char *start = choices;
188 while (*start) {
189 const char *end = strchr(start, ' ');
190 size_t n = end ? (size_t)(end - start) : strlen(start);
191 if (n == length && memcmp(start, value, n) == 0) return true;
192 if (!end) break;
193 start = end + 1;
194 }
195 return false;
196}

231static int parse_declaration(int property, const char *value, Declaration result[4]) {
232 if (property < 0) return 0;
233 Strings tokens = split(&work, value, " ");
234 if (property >= MARGIN) {
235 bool gap = property == GAP;
236 if (tokens.count < 1 || tokens.count > (gap ? 2 : 4)) return 0;
237 for (int i = 0; i < tokens.count; i++) if (!integer(tokens.items[i]) || (property == BORDER && !one_of(tokens.items[i], "0 1"))) return 0;
238 if (gap) {
239 result[0] = (Declaration){ROW_GAP, tokens.items[0]};
240 result[1] = (Declaration){COLUMN_GAP, tokens.items[tokens.count - 1]};
241 return 2;
242 }
243 int first = property == MARGIN ? MT : property == PADDING ? PT : BT;
244 int indices[] = {0, tokens.count > 1 ? 1 : 0, tokens.count > 2 ? 2 : 0, tokens.count > 3 ? 3 : tokens.count > 1 ? 1 : 0};
245 for (int i = 0; i < 4; i++) result[i] = (Declaration){first + i, tokens.items[indices[i]]};
246 return 4;
247 }
248 bool valid = false;
249 if (property == COLUMNS || property == ROWS) {
250 valid = true;
251 if (!equal(value, "none")) for (int i = 0; i < tokens.count; i++) {
252 char *token = tokens.items[i];
253 size_t n = strlen(token);
254 if (!integer(token) && !equal(token, "auto") && !(n > 2 && equal(token + n - 2, "fr") && positive(part(&work, token, n - 2)))) valid = false;
255 }
256 } else if (property == GRID_COLUMN || property == GRID_ROW) {
257 char **t = tokens.items;
258 valid = (tokens.count == 1 && (equal(value, "auto") || positive(value)))
259 || (tokens.count == 2 && equal(t[0], "span") && positive(t[1]))
260 || (tokens.count == 3 && positive(t[0]) && equal(t[1], "/") && positive(t[2]) && atoi(t[0]) < atoi(t[2]))
261 || (tokens.count == 4 && positive(t[0]) && equal(t[1], "/") && equal(t[2], "span") && positive(t[3]));
262 } else {
263 if (tokens.count != 1) return 0;
264 switch (property) {
265 case DISPLAY:
266 valid = one_of(value, "block flex grid none");
267 break;
268 case POSITION:
269 valid = one_of(value, "static relative absolute");
270 break;
271 case BOX_SIZING:
272 valid = one_of(value, "content-box border-box");
273 break;
274 case WIDTH:
275 case HEIGHT:
276 case BASIS:
277 valid = equal(value, "auto") || integer(value) || percentage(value);
278 break;
279 case MIN_WIDTH:
280 case MIN_HEIGHT:
281 valid = integer(value) || percentage(value);
282 break;
283 case MAX_WIDTH:
284 case MAX_HEIGHT:
285 valid = equal(value, "none") || integer(value) || percentage(value);
286 break;
287 case TOP:
288 case RIGHT:
289 case BOTTOM:
290 case LEFT:
291 valid = equal(value, "auto") || signed_integer(value);
292 break;
293 case ORDER:
294 valid = signed_integer(value);
295 break;
296 case DIRECTION:
297 valid = one_of(value, "row row-reverse column column-reverse");
298 break;
299 case WRAP:
300 valid = one_of(value, "nowrap wrap");
301 break;
302 case JUSTIFY:
303 valid = one_of(value, "start end center space-between space-around space-evenly");
304 break;
305 case ALIGN_ITEMS:
306 valid = one_of(value, "stretch start end center");
307 break;
308 case ALIGN_SELF:
309 valid = one_of(value, "auto stretch start end center");
310 break;
311 default:
312 valid = property >= BT && property <= BL ? one_of(value, "0 1") : integer(value);
313 break;
314 }
315 }
316 if (!valid) return 0;
317 result[0] = (Declaration){property, (char *)value};
318 return 1;
319}

358 ElementList result = {0};

360 Element *child = e->children[i];
361 if (child->displayed && !equal(child->style[POSITION], "absolute")) result.items[result.count++] = child;
362 }
363 if (sorted) for (int i = 1; i < result.count; i++) {
364 Element *item = result.items[i];
365 int j = i;
366 while (j > 0 && number(result.items[j - 1], ORDER) > number(item, ORDER)) {
367 result.items[j] = result.items[j - 1];
368 j--;
369 }
370 result.items[j] = item;
371 }
372 return result;

393static int *distribute(int total, const long long *weights, int count) {
394 int *result = allocate(&layout_memory, (size_t)count * sizeof *result);
395 long long sum = 0;
396 for (int i = 0; i < count; i++) sum += weights[i];
397 if (!sum) return result;
398 int remaining = total;
399 for (int i = 0; i < count; i++) {
400 result[i] = (int)((long long)total * weights[i] / sum);
401 remaining -= result[i];
402 }
403 for (int i = 0; i < count && remaining > 0; i++) if (weights[i] > 0) {
404 result[i]++;
405 remaining--;
406 }
407 return result;
408}

451static Grid place_grid(Element *e) {
452 Grid result = {0};
453 ElementList children = flow(e, true);
454 tracks(e->style[COLUMNS], true, &result.columns);
455 tracks(e->style[ROWS], false, &result.rows);
456 for (int i = 0; i < children.count; i++) {
457 GridItem *item = &result.items[result.count++];
458 item->element = children.items[i];
459 placement(item->element->style[GRID_COLUMN], &item->column, &item->columns);
460 placement(item->element->style[GRID_ROW], &item->row, &item->rows);
461 result.columns = maximum(result.columns, item->column ? item->column + item->columns - 1 : item->columns);
462 }
463 for (int i = 0; i < result.count; i++) {
464 GridItem *item = &result.items[i];
465 if (item->row && item->column) item->placed = true;
466 }
467 int cr = 1, cc = 1;
468 for (int i = 0; i < result.count; i++) {
469 GridItem *item = &result.items[i];
470 if (item->placed) continue;
471 if (item->row) {
472 int c = 1;
473 while (c <= result.columns - item->columns + 1 && overlaps(&result, item, item->row, c)) c++;
474 item->column = c <= result.columns - item->columns + 1 ? c : 1;
475 } else if (item->column) {
476 if (item->column < cc) cr++;
477 int r = cr;
478 while (overlaps(&result, item, r, item->column)) r++;
479 item->row = r;
480 cr = r;
481 cc = item->column + item->columns;
482 } else {
483 int r = cr, c = cc;
484 while (true) {
485 if (c + item->columns - 1 > result.columns) {
486 r++;
487 c = 1;
488 } else if (overlaps(&result, item, r, c)) c++;
489 else break;
490 }
491 item->row = r;
492 item->column = c;
493 cr = r;
494 cc = c + item->columns;
495 }
496 item->placed = true;
497 }
498 for (int i = 0; i < result.count; i++) result.rows = maximum(result.rows, result.items[i].row + result.items[i].rows - 1);
499 return result;
500}

503static int *track_sizes(Element *e, Grid *grid, bool columns, int available, bool intrinsic) {
504 int count = columns ? grid->columns : grid->rows, explicit_count;
505 Track *explicit_tracks = tracks(e->style[columns ? COLUMNS : ROWS], columns, &explicit_count), *specs = allocate(&layout_memory, (size_t)count * sizeof *specs);
506 for (int i = 0; i < count; i++) specs[i] = i < explicit_count ? explicit_tracks[i] : (Track){'a', 0};
507 int *sizes = allocate(&layout_memory, (size_t)count * sizeof *sizes);
508 for (int i = 0; i < count; i++) {
509 if (specs[i].kind == 'n') sizes[i] = specs[i].value;
510 else if (specs[i].kind == 'a' || available < 0) for (int j = 0; j < grid->count; j++) {
511 GridItem *item = &grid->items[j];
512 if ((columns ? item->column : item->row) == i + 1 && (columns ? item->columns : item->rows) == 1) sizes[i] = maximum(sizes[i], intrinsic ? contribution(item->element) : item->contribution);
513 }
514 }
515 int gap = number(e, columns ? COLUMN_GAP : ROW_GAP), leftover = available - gap * maximum(0, count - 1);
516 for (int i = 0; i < count; i++) leftover -= sizes[i];
517 if (available >= 0 && leftover > 0) {
518 bool fr = false;
519 for (int i = 0; i < count; i++) if (specs[i].kind == 'f') fr = true;
520 long long *weights = allocate(&layout_memory, (size_t)count * sizeof *weights);
521 for (int i = 0; i < count; i++) if (fr ? specs[i].kind == 'f' : specs[i].kind == 'a') weights[i] = fr ? specs[i].value : 1;
522 int *extra = distribute(leftover, weights, count);
523 for (int i = 0; i < count; i++) sizes[i] += extra[i];
524 }
525 return sizes;
526}

528static int max_content(Element *e) {

530 int result = 0;
531 if (e->text) result = (int)strlen(e->text);
532 else if (equal(e->style[DISPLAY], "flex") && strncmp(e->style[DIRECTION], "row", 3) == 0) {
533 ElementList children = flow(e, false);

543 for (int i = 0; i < children.count; i++) result = maximum(result, contribution(children.items[i]));
544 }
545 e->intrinsic = result;
546 return result;

568static int flex(Element *e, int k, int height_reference) {
569 bool horizontal = strncmp(e->style[DIRECTION], "row", 3) == 0, reverse = strstr(e->style[DIRECTION], "reverse") != NULL;
570 int reference = horizontal ? e->cw : k, gap = number(e, horizontal ? COLUMN_GAP : ROW_GAP);
571 ElementList children = flow(e, true);
572 FlexItem items[60] = {0};
573 for (int i = 0; i < children.count; i++) {
574 FlexItem *item = &items[i];
575 Element *child = children.items[i];
576 item->element = child;
577 item->outer = horizontal ? hm(child) + hf(child) : vm(child) + vf(child);
578 if (!horizontal) {
579 int r = maximum(0, e->cw - hm(child) - hf(child)), width = resolve(child, WIDTH, e->cw, true);
580 item->cross = clamp(child, width >= 0 ? width : equal(align(e, child), "stretch") ? r : minimum(max_content(child), r), e->cw, true);
581 }
582 item->base = resolve(child, BASIS, reference, horizontal);
583 if (item->base < 0) item->base = resolve(child, horizontal ? WIDTH : HEIGHT, reference, horizontal);
584 if (item->base < 0) item->base = horizontal ? max_content(child) : layout(child, e->cw, k, item->cross, -1);
585 item->hypothetical = clamp(child, item->base, reference, horizontal);
586 }
587 int length = reference;
588 if (length < 0) {
589 length = gap * maximum(0, children.count - 1);
590 for (int i = 0; i < children.count; i++) length += items[i].hypothetical + items[i].outer;
591 length = clamp(e, length, height_reference, false);
592 }
593 int starts[60], ends[60], line_count = 0, occupied = 0;
594 for (int i = 0; i < children.count; i++) {
595 FlexItem *item = &items[i];
596 if (line_count == 0 || (horizontal && equal(e->style[WRAP], "wrap") && occupied + gap + item->hypothetical + item->outer > length)) {
597 starts[line_count] = i;
598 ends[line_count] = i;
599 line_count++;
600 occupied = 0;
601 }
602 if (ends[line_count - 1] > starts[line_count - 1]) occupied += gap;
603 occupied += item->hypothetical + item->outer;
604 ends[line_count - 1] = i + 1;
605 }
606 int cross_start = 0;
607 for (int line = 0; line < line_count; line++) {
608 int start = starts[line], end = ends[line], count = end - start, total = gap * (count - 1);
609 for (int i = start; i < end; i++) total += items[i].hypothetical + items[i].outer;
610 bool grow = total < length;
611 for (int i = start; i < end; i++) {
612 FlexItem *item = &items[i];
613 item->weight = grow ? number(item->element, GROW) : (long long)number(item->element, SHRINK) * item->base;
614 item->size = item->hypothetical;
615 item->frozen = item->weight == 0 || (grow ? item->base > item->hypothetical : item->base < item->hypothetical);
616 }
617 while (true) {
618 int free_space = length - gap * (count - 1), remaining = 0;
619 long long weights[60] = {0};
620 for (int i = start; i < end; i++) {
621 FlexItem *item = &items[i];
622 free_space -= (item->frozen ? item->size : item->base) + item->outer;
623 if (!item->frozen) {
624 weights[i - start] = item->weight;
625 remaining++;
626 }
627 }
628 if (!remaining) break;
629 int *shares = distribute(maximum(0, grow ? free_space : -free_space), weights, count), violations[60] = {0}, violation = 0;
630 for (int i = start; i < end; i++) {
631 FlexItem *item = &items[i];
632 if (!item->frozen) {
633 int tentative = item->base + (grow ? shares[i - start] : -shares[i - start]);
634 item->size = clamp(item->element, tentative, reference, horizontal);
635 violations[i - start] = item->size - tentative;
636 violation += violations[i - start];
637 }
638 }
639 for (int i = start; i < end; i++) if (violation == 0 || (violation > 0 && violations[i - start] > 0) || (violation < 0 && violations[i - start] < 0)) items[i].frozen = true;
640 }
641 int lc = horizontal ? 0 : e->cw;
642 if (horizontal) {
643 for (int i = start; i < end; i++) {
644 FlexItem *item = &items[i];
645 layout(item->element, e->cw, k, item->size, -1);
646 item->cross = item->element->ch;
647 lc = maximum(lc, item->cross + vf(item->element) + vm(item->element));
648 }
649 if (line_count == 1) lc = k >= 0 ? k : clamp(e, lc, height_reference, false);
650 }
651 int used = gap * (count - 1);
652 for (int i = start; i < end; i++) used += items[i].size + items[i].outer;
653 int space = length - used, *before = allocate(&layout_memory, (size_t)(count + 1) * sizeof *before);
654 if (space > 0) {
655 const char *justify = e->style[JUSTIFY];
656 if (equal(justify, "end")) before[0] = space;
657 else if (equal(justify, "center")) before[0] = space / 2;
658 else if (strncmp(justify, "space-", 6) == 0) {
659 long long weights[61];
660 for (int i = 0; i <= count; i++) weights[i] = 1;
661 if (equal(justify, "space-between")) weights[0] = weights[count] = 0;
662 else if (equal(justify, "space-around")) for (int i = 1; i < count; i++) weights[i] = 2;
663 before = distribute(space, weights, count + 1);
664 }
665 }
666 int cursor = 0;
667 for (int i = start; i < end; i++) {
668 FlexItem *item = &items[i];
669 Element *child = item->element;
670 const char *alignment = align(e, child);
671 if (horizontal) {
672 if (equal(alignment, "stretch") && equal(child->style[HEIGHT], "auto")) {
673 item->cross = clamp(child, maximum(0, lc - vm(child) - vf(child)), k, false);
674 layout(child, e->cw, k, item->size, item->cross);
675 } else layout(child, e->cw, k, item->size, -1);
676 } else layout(child, e->cw, k, item->cross, item->size);
677 int cross = cross_start + offset(alignment, lc - item->cross - (horizontal ? vf(child) + vm(child) : hf(child) + hm(child))) + number(child, horizontal ? MT : ML);
678 cursor += before[i - start];
679 int main = reverse ? length - cursor - number(child, horizontal ? MR : MB) - item->size - (horizontal ? hf(child) : vf(child)) : cursor + number(child, horizontal ? ML : MT);
680 position(e, child, horizontal ? main : cross, horizontal ? cross : main);
681 cursor += item->size + item->outer + gap;
682 }
683 cross_start += lc + number(e, ROW_GAP);
684 }
685 return horizontal ? line_count ? cross_start - number(e, ROW_GAP) : 0 : length;
686}

700static int grid_layout(Element *e, int k) {
701 Grid grid = place_grid(e);
702 int cg = number(e, COLUMN_GAP), rg = number(e, ROW_GAP), *columns = track_sizes(e, &grid, true, e->cw, true);
703 for (int i = 0; i < grid.count; i++) {
704 GridItem *item = &grid.items[i];
705 Element *child = item->element;
706 item->aw = area(columns, item->column, item->columns, cg);
707 item->ax = prefix(columns, item->column - 1, cg);
708 int width = resolve(child, WIDTH, item->aw, true);
709 item->width = clamp(child, width < 0 ? maximum(0, item->aw - hm(child) - hf(child)) : width, item->aw, true);
710 layout(child, item->aw, -1, item->width, -1);
711 item->contribution = bh(child) + vm(child);
712 }
713 int *rows = track_sizes(e, &grid, false, k, false);
714 for (int i = 0; i < grid.count; i++) {
715 GridItem *item = &grid.items[i];
716 Element *child = item->element;
717 int ah = area(rows, item->row, item->rows, rg), ay = prefix(rows, item->row - 1, rg);
718 const char *alignment = align(e, child);
719 int forced_height = equal(alignment, "stretch") && equal(child->style[HEIGHT], "auto") ? clamp(child, maximum(0, ah - vm(child) - vf(child)), ah, false) : -1;
720 layout(child, item->aw, ah, item->width, forced_height);
721 position(e, child, item->ax + number(child, ML), ay + offset(alignment, ah - bh(child) - vm(child)) + number(child, MT));
722 }
723 int result = rg * maximum(0, grid.rows - 1);
724 for (int i = 0; i < grid.rows; i++) result += rows[i];
725 return result;
726}

728static int layout(Element *e, int width_reference, int height_reference, int forced_width, int forced_height) {
729 int width = resolve(e, WIDTH, width_reference, true);
730 e->cw = forced_width >= 0 ? forced_width : clamp(e, width < 0 ? maximum(0, width_reference - hm(e) - hf(e)) : width, width_reference, true);
731 if (forced_height < 0) {
732 int height = resolve(e, HEIGHT, height_reference, false);
733 if (height >= 0) forced_height = clamp(e, height, height_reference, false);
734 }
735 int a;
736 if (e->text) {
737 e->lines = (Strings){0};
738 Strings words = split(&layout_memory, e->text, " ");
739 char *current = words.items[0];
740 for (int i = 1; i < words.count; i++) {
741 char *word = words.items[i];
742 size_t length = strlen(current), word_length = strlen(word);
743 if (length + 1 + word_length <= (size_t)e->cw) {
744 char *joined = allocate(&layout_memory, length + word_length + 2);
745 memcpy(joined, current, length);
746 joined[length] = ' ';
747 memcpy(joined + length + 1, word, word_length + 1);
748 current = joined;
749 } else {
750 add_string(&layout_memory, &e->lines, current);
751 current = word;
752 }
753 }
754 add_string(&layout_memory, &e->lines, current);
755 a = e->lines.count;
756 } else if (equal(e->style[DISPLAY], "flex")) a = flex(e, forced_height, height_reference);
757 else if (equal(e->style[DISPLAY], "grid")) a = grid_layout(e, forced_height);
758 else {
759 a = 0;
760 int previous_bottom = 0;
761 ElementList children = flow(e, false);
762 for (int i = 0; i < children.count; i++) {
763 Element *child = children.items[i];
764 layout(child, e->cw, forced_height, -1, -1);
765 int y = a + (i == 0 ? number(child, MT) : maximum(previous_bottom, number(child, MT)));
766 position(e, child, number(child, ML), y);
767 a = y + bh(child);
768 previous_bottom = number(child, MB);
769 }
770 if (children.count) a += previous_bottom;
771 }
772 e->ch = forced_height >= 0 ? forced_height : clamp(e, a, height_reference, false);
773 return a;
774}

776static void relayout(void) {
777 clear_arena(&layout_memory);
778 for (int i = 0; i < element_count; i++) {
779 Element *e = &elements[i];
780 e->displayed = !equal(e->style[DISPLAY], "none") && (!e->parent || e->parent->displayed);
781 e->intrinsic = -1;
782 e->containing = NULL;
783 e->lines = (Strings){0};
784 }
785 Element *root = &elements[0];
786 if (root->displayed && !equal(root->style[POSITION], "absolute")) {
787 layout(root, viewport_width, viewport_height, -1, -1);
788 root->rx = number(root, ML);
789 root->ry = number(root, MT);
790 }
791 for (int i = 0; i < element_count; i++) {
792 Element *e = &elements[i];
793 if (!e->displayed || !equal(e->style[POSITION], "absolute")) continue;
794 Element *cb = e->parent;
795 while (cb && equal(cb->style[POSITION], "static")) cb = cb->parent;
796 e->containing = cb;
797 int width = cb ? cb->cw + number(cb, PL) + number(cb, PR) : viewport_width, height = cb ? cb->ch + number(cb, PT) + number(cb, PB) : viewport_height;
798 int left = equal(e->style[LEFT], "auto") ? 0 : number(e, LEFT), right = equal(e->style[RIGHT], "auto") ? 0 : number(e, RIGHT), top = equal(e->style[TOP], "auto") ? 0 : number(e, TOP), bottom = equal(e->style[BOTTOM], "auto") ? 0 : number(e, BOTTOM);
799 int w = resolve(e, WIDTH, width, true);
800 if (equal(e->style[WIDTH], "auto")) {
801 int r = maximum(0, width - left - right - hm(e) - hf(e));
802 w = !equal(e->style[LEFT], "auto") && !equal(e->style[RIGHT], "auto") ? r : minimum(max_content(e), r);
803 }
804 int h = equal(e->style[HEIGHT], "auto") && !equal(e->style[TOP], "auto") && !equal(e->style[BOTTOM], "auto") ? clamp(e, maximum(0, height - top - bottom - vm(e) - vf(e)), height, false) : -1;
805 layout(e, width, height, clamp(e, w, width, true), h);
806 e->rx = !equal(e->style[LEFT], "auto") ? left + number(e, ML) : !equal(e->style[RIGHT], "auto") ? width - right - number(e, MR) - bw(e) : number(e, ML);
807 e->ry = !equal(e->style[TOP], "auto") ? top + number(e, MT) : !equal(e->style[BOTTOM], "auto") ? height - bottom - number(e, MB) - bh(e) : number(e, MT);
808 }
809 for (int i = 0; i < element_count; i++) {
810 Element *e = &elements[i];
811 if (!e->displayed) continue;
812 bool absolute = equal(e->style[POSITION], "absolute");
813 Element *reference = absolute ? e->containing : e->parent;
814 e->x = e->rx + (reference ? reference->x + (absolute ? number(reference, BL) : 0) : 0);
815 e->y = e->ry + (reference ? reference->y + (absolute ? number(reference, BT) : 0) : 0);
816 if (equal(e->style[POSITION], "relative")) {
817 e->x += !equal(e->style[LEFT], "auto") ? number(e, LEFT) : !equal(e->style[RIGHT], "auto") ? -number(e, RIGHT) : 0;
818 e->y += !equal(e->style[TOP], "auto") ? number(e, TOP) : !equal(e->style[BOTTOM], "auto") ? -number(e, BOTTOM) : 0;
819 }
820 }
821}

884static void command(const char *line) {
885 Strings tokens = split(&work, line, " ");
886 char **t = tokens.items;
887 Element *e = tokens.count > 1 ? find_element(t[1]) : NULL;
888 if (equal(t[0], "LAYOUT")) {
889 printf("layout %dx%d\n", viewport_width, viewport_height);
890 for (int i = 0; i < element_count; i++) {
891 Element *item = &elements[i];
892 for (int n = 0; n < item->depth * 2; n++) putchar(' ');
893 if (item->displayed) rectangle(item->id, item->x, item->y, bw(item), bh(item));
894 else printf("%s none\n", item->id);
895 }
896 } else if (equal(t[0], "BOX")) {
897 if (!e) printf("error: unknown element %s\n", t[1]);
898 else if (!e->displayed) printf("box %s: none\n", e->id);
899 else {
900 printf("box %s\n", e->id);
901 rectangle("margin", e->x - number(e, ML), e->y - number(e, MT), bw(e) + hm(e), bh(e) + vm(e));
902 rectangle("border", e->x, e->y, bw(e), bh(e));
903 rectangle("padding", e->x + number(e, BL), e->y + number(e, BT), e->cw + number(e, PL) + number(e, PR), e->ch + number(e, PT) + number(e, PB));
904 rectangle("content", cx(e), cy(e), e->cw, e->ch);
905 }
906 } else if (equal(t[0], "RENDER")) render();
907 else if (equal(t[0], "HIT")) {
908 int x = atoi(t[1]), y = atoi(t[2]);
909 if (x >= viewport_width || y >= viewport_height) printf("error: point %d %d outside viewport\n", x, y);
910 else {
911 Element *hit = NULL, *path[11];
912 int count = 0;
913 for (int i = 0; i < element_count; i++) {
914 Element *item = &elements[i];
915 if (item->displayed && x >= item->x && x < item->x + bw(item) && y >= item->y && y < item->y + bh(item)) hit = item;
916 }
917 printf("hit %d %d: ", x, y);
918 for (; hit; hit = hit->parent) path[count++] = hit;
919 if (!count) printf("none");
920 for (int i = count - 1; i >= 0; i--) printf("%s%s", i == count - 1 ? "" : " > ", path[i]->id);
921 putchar('\n');
922 }
923 } else if (equal(t[0], "SET")) {
924 const char *value = line + strlen(t[0]) + strlen(t[1]) + strlen(t[2]) + 3;
925 int property = property_index(t[2]);
926 Declaration values[4];
927 if (!e) printf("error: unknown element %s\n", t[1]);
928 else if (property < 0) printf("error: unknown property %s\n", t[2]);
929 else {
930 int count = parse_declaration(property, value, values);
931 if (!count) printf("error: invalid value for %s: %s\n", t[2], value);
932 else {
933 Box old[60];
934 snapshot(old);
935 for (int i = 0; i < count; i++) e->style[values[i].property] = copy(&permanent, values[i].value);
936 relayout();
937 printf("set %s %s %s: ", e->id, t[2], value);
938 print_changes(old);
939 }
940 }
941 } else if (equal(t[0], "VIEWPORT")) {
942 int width = atoi(t[1]), height = atoi(t[2]);
943 if (width < 1 || width > 80 || height < 1 || height > 40) printf("error: invalid viewport %d %d\n", width, height);
944 else {
945 Box old[60];
946 snapshot(old);
947 viewport_width = width;
948 viewport_height = height;
949 relayout();
950 printf("viewport %d %d: ", width, height);
951 print_changes(old);
952 }
953 } else if (equal(t[0], "OVERFLOW")) {
954 bool any = false;
955 printf("overflow: ");
956 for (int i = 0; i < element_count; i++) {
957 Element *item = &elements[i];
958 if (!item->displayed) continue;
959 bool overflow = false;
960 if (item->text) {
961 overflow = item->lines.count > item->ch;
962 for (int j = 0; j < item->lines.count; j++) if ((int)strlen(item->lines.items[j]) > item->cw) overflow = true;
963 } else {
964 ElementList children = flow(item, false);
965 for (int j = 0; j < children.count; j++) {
966 Element *child = children.items[j];
967 if (child->x - number(child, ML) < cx(item) || child->y - number(child, MT) < cy(item) || child->x + bw(child) + number(child, MR) > cx(item) + item->cw || child->y + bh(child) + number(child, MB) > cy(item) + item->ch) overflow = true;
968 }
969 }
970 if (overflow) {
971 printf("%s%s", any ? " " : "", item->id);
972 any = true;
973 }
974 }
975 if (!any) printf("none");
976 putchar('\n');
977 } else printf("error: unknown command %s\n", t[0]);
978}

980static void read_rule(const char *line) {
981 Rule *rule = &rules[rule_count++];
982 const char *brace = strstr(line, " { ");
983 char *selector_text = part(&permanent, line, (size_t)(brace - line));
984 rule->selectors = split(&permanent, selector_text, ", ");
985 Strings declarations = split(&work, part(&work, brace + 3, strlen(brace + 3) - 2), "; ");
986 for (int i = 0; i < declarations.count; i++) {
987 const char *raw = declarations.items[i], *colon = strstr(raw, ": ");
988 int property = property_index(part(&work, raw, (size_t)(colon - raw)));
989 Declaration values[4];
990 int count = parse_declaration(property, colon + 2, values);
991 for (int j = 0; j < count; j++) rule->declarations[rule->count++] = (Declaration){values[j].property, copy(&permanent, values[j].value)};
992 }
993}

995static void read_element(const char *line, Element *ancestors[11]) {
996 int spaces = 0;
997 while (line[spaces] == ' ') spaces++;
998 Element *e = &elements[element_count];
999 e->index = element_count++;
1000 e->depth = spaces / 2;
1001 const char *body = line + spaces, *space = strchr(body, ' ');
1002 char *head = space ? part(&work, body, (size_t)(space - body)) : copy(&work, body);
1003 if (space) e->text = part(&permanent, space + 2, strlen(space + 2) - 1);
1004 char *hash = strchr(head, '#');
1005 e->type = part(&permanent, head, (size_t)(hash - head));
1006 Strings parts = split(&permanent, hash + 1, ".");
1007 e->id = parts.items[0];
1008 for (int i = 1; i < parts.count; i++) add_string(&permanent, &e->classes, parts.items[i]);
1009 if (e->depth > 0) {
1010 e->parent = ancestors[e->depth - 1];
1011 e->parent->children[e->parent->child_count++] = e;
1012 }
1013 ancestors[e->depth] = e;
1014 int scores[PROPERTY_COUNT];
1015 for (int i = 0; i < PROPERTY_COUNT; i++) {
1016 scores[i] = -1;
1017 e->style[i] = initial_values[i];
1018 }
1019 for (int i = 0; i < rule_count; i++) {
1020 Rule *rule = &rules[i];
1021 int score = -1;
1022 for (int j = 0; j < rule->selectors.count; j++) {
1023 const char *selector = rule->selectors.items[j];
1024 if (selector[0] == '#' && equal(selector + 1, e->id)) score = maximum(score, 2);
1025 else if (selector[0] == '.') {
1026 for (int k = 0; k < e->classes.count; k++) if (equal(selector + 1, e->classes.items[k])) score = maximum(score, 1);
1027 } else if (equal(selector, e->type)) score = maximum(score, 0);
1028 }
1029 if (score >= 0) for (int j = 0; j < rule->count; j++) {
1030 Declaration d = rule->declarations[j];
1031 if (score >= scores[d.property]) {
1032 e->style[d.property] = d.value;
1033 scores[d.property] = score;
1034 }
1035 }
1036 }
1037}

1042 sscanf(line, "%d %d", &viewport_width, &viewport_height);

1045 read_rule(line);

1057 relayout();
1058 while ((line = read_line()) && !equal(line, "END")) {

1062 }
1063 free(line);
1064 clear_arena(&layout_memory);
1065 clear_arena(&work);
1066 clear_arena(&permanent);
