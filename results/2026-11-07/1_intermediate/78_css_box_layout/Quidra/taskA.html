603 int value := resolve_length(item, "width", aw, true)
604 int gw := constrain(item, if value < 0 then maximum(aw - outer(item, true), 0) else value, true, aw)
605 .layout_node(item, aw, -1, gw)
606 area_widths = area_widths.append(aw)
607 widths = widths.append(gw)
608 row_contributions = row_contributions.append(Contribution(area.row, area.row_span, nodes[item].ch + outer(item, false)))

610 nr := int(#specs)
611 for area := placement.areas[]
612 nr = maximum(nr, area.row + area.row_span - 1)
613 rows := track_sizes(specs, nr, k, rg, row_contributions)
614 for: i < #ordered
615 item := ordered[i]
616 area := placement.areas[i]
617 ah := span_size(rows, area.row, area.row_span, rg)
618 string align := item_align(index, item)
619 if align == "stretch" and property(item, "height") == "auto"
620 int fh := constrain(item, maximum(ah - outer(item, false), 0), false, ah)
621 .layout_node(item, area_widths[i], ah, widths[i], fh)
622 else
623 .layout_node(item, area_widths[i], ah, widths[i])
624 offset := align_offset(align, ah - nodes[item].ch - outer(item, false))
625 .place_child(index, item, span_start(columns, area.column, cg) + edge(item, "margin", "left"), span_start(rows, area.row, rg) + offset + edge(item, "margin", "top"))
626 return total_ints(rows) + gap_total(rg, nat(nr))

647 void .perform_layout()
648 intrinsic = [for i := (nat(0):<#nodes)[] then -1]
649 for: i < #nodes
650 nodes[i].displayed = property(i, "display") != "none" and (nodes[i].parent < 0 or nodes[nodes[i].parent].displayed)
651 if nodes[0].displayed and property(0, "position") != "absolute"
652 .layout_node(0, viewport_width, viewport_height)
653 nodes[0].rx = edge(0, "margin", "left")
654 nodes[0].ry = edge(0, "margin", "top")
655 for: i < #nodes
656 if not nodes[i].displayed or property(i, "position") != "absolute"
657 continue
658 int ancestor_id := positioned_ancestor(i)
659 int bw := if ancestor_id < 0 then viewport_width else padding_box(nat(ancestor_id)).w
660 int bh := if ancestor_id < 0 then viewport_height else padding_box(nat(ancestor_id)).h
661 has_left := property(i, "left") != "auto"
662 has_right := property(i, "right") != "auto"
663 has_top := property(i, "top") != "auto"
664 has_bottom := property(i, "bottom") != "auto"
665 int left := inset(i, "left")
666 int right := inset(i, "right")
667 int top := inset(i, "top")
668 int bottom := inset(i, "bottom")
669 int w := resolve_length(i, "width", bw, true)
670 if w < 0
671 room := maximum(bw - left - right - outer(i, true), 0)
672 w = if has_left and has_right then room else minimum(.max_content(i), room)
673 int fw := constrain(i, w, true, bw)
674 fh := -1
675 if property(i, "height") == "auto" and has_top and has_bottom
676 fh = constrain(i, maximum(bh - top - bottom - outer(i, false), 0), false, bh)
677 .layout_node(i, bw, bh, fw, fh)
678 nodes[i].rx = if has_left then left + edge(i, "margin", "left") else (if has_right then bw - right - edge(i, "margin", "right") - border_width(i) else edge(i, "margin", "left"))
679 nodes[i].ry = if has_top then top + edge(i, "margin", "top") else (if has_bottom then bh - bottom - edge(i, "margin", "bottom") - border_height(i) else edge(i, "margin", "top"))
680 for: i < #nodes
681 if not nodes[i].displayed
682 continue
683 absolute_position := property(i, "position") == "absolute"
684 int ancestor_id := if absolute_position then positioned_ancestor(i) else nodes[i].parent
685 ox := 0
686 oy := 0
687 if ancestor_id >= 0
688 CssBox origin := if absolute_position then padding_box(nat(ancestor_id)) else border_box(nat(ancestor_id))
689 ox = origin.x
690 oy = origin.y
691 nodes[i].x = ox + nodes[i].rx
692 nodes[i].y = oy + nodes[i].ry
693 if property(i, "position") == "relative"
694 nodes[i].x += if property(i, "left") != "auto" then inset(i, "left") else -inset(i, "right")
695 nodes[i].y += if property(i, "top") != "auto" then inset(i, "top") else -inset(i, "bottom")

709 if nodes[index].text != ""
710 longest := 0
711 for line := nodes[index].lines[]
712 longest = maximum(longest, int(#line))

721 void render_screen()
722 screen := [for row := (0:<int(viewport_height))[] then [for column := (0:<int(viewport_width))[] then "."]]
723 for: i < #nodes
724 if not nodes[i].displayed
725 continue
726 CssBox box := border_box(i)
727 for: maximum(box.y, 0) <= row < minimum(box.y + box.h, viewport_height)
728 for: maximum(box.x, 0) <= column < minimum(box.x + box.w, viewport_width)
729 on_row := (edge(i, "border", "top") == 1 and row == box.y) or (edge(i, "border", "bottom") == 1 and row == box.y + box.h - 1)
730 on_column := (edge(i, "border", "left") == 1 and column == box.x) or (edge(i, "border", "right") == 1 and column == box.x + box.w - 1)
731 screen[row][column] = if on_row then (if on_column then "+" else "-") else (if on_column then "|" else " ")
732 if nodes[i].text != ""
733 CssBox content := content_box(i)
734 for: row < #nodes[i].lines
735 line := nodes[i].lines[row]
736 for: column < #line
737 x := content.x + int(column)
738 y := content.y + int(row)
739 if x >= 0 and x < viewport_width and y >= 0 and y < viewport_height
740 screen[y][x] = line[column]
741 emit("render {viewport_width}x{viewport_height}")
742 string boundary := "#" * nat(viewport_width + 2)
743 emit(boundary)
744 for row := screen[]
745 emit("#" + row.join("") + "#")
746 emit(boundary)

748 void .execute(string line)
749 parts := line.split(" ")
750 command := parts[0]
751 if command == "LAYOUT"
752 emit("layout {viewport_width}x{viewport_height}")
753 for: i < #nodes
754 emit(" " * (2 * nodes[i].depth) + nodes[i].id + " " + (if nodes[i].displayed then box_text(border_box(i)) else "none"))
755 elif command == "BOX" or command == "SET"
756 id := parts[1]
757 found := -1
758 for: i < #nodes
759 if nodes[i].id == id
760 found = int(i)
761 break
762 if found < 0
763 emit("error: unknown element " + id)
764 return
765 index := nat(found)
766 if command == "BOX"
767 if not nodes[index].displayed
768 emit("box " + id + ": none")
769 else
770 emit("box " + id)
771 emit("margin " + box_text(margin_box(index)))
772 emit("border " + box_text(border_box(index)))
773 emit("padding " + box_text(padding_box(index)))
774 emit("content " + box_text(content_box(index)))
775 else
776 prop := parts[2]
777 value := parts[3:].join(" ")
778 if not known_property(prop)
779 emit("error: unknown property " + prop)
780 return
781 declarations := expand_declaration(prop, value)
782 if #declarations == 0
783 emit("error: invalid value for " + prop + ": " + value)
784 return
785 string[] before := snapshot()
786 for declaration := declarations[]
787 .nodes[index].style.set(declaration.prop, declaration.value)
788 .perform_layout()
789 emit("set " + id + " " + prop + " " + value + ": " + changes(before))
790 elif command == "RENDER"
791 render_screen()
792 elif command == "HIT"
793 x := int(parts[1])
794 y := int(parts[2])
795 if x >= viewport_width or y >= viewport_height
796 emit("error: point {x} {y} outside viewport")
797 return
798 target := -1
799 for: i < #nodes
800 CssBox box := border_box(i)
801 if nodes[i].displayed and x >= box.x and x < box.x + box.w and y >= box.y and y < box.y + box.h
802 target = int(i)
803 string[] path := []
804 while target >= 0
805 path = [nodes[target].id].concat(path)
806 target = nodes[target].parent
807 emit("hit {x} {y}: " + (if #path == 0 then "none" else path.join(" > ")))
808 elif command == "VIEWPORT"
809 w := int(parts[1])
810 h := int(parts[2])
811 if w < 1 or w > 80 or h < 1 or h > 40
812 emit("error: invalid viewport {w} {h}")
813 return
814 string[] before := snapshot()
815 viewport_width = w
816 viewport_height = h
817 .perform_layout()
818 emit("viewport {w} {h}: " + changes(before))
819 elif command == "OVERFLOW"
820 string[] ids := []
821 for: i < #nodes
822 if nodes[i].displayed and overflows(i)
823 ids = ids.append(nodes[i].id)
824 emit("overflow: " + (if #ids == 0 then "none" else ids.join(" ")))
825 else
826 emit("error: unknown command " + command)

828void run_program()
829 line := ""
830 scan(&line)
831 dimensions := line.split(" ")
832 Rule[] rules := []
833 while true
834 scan(&line)
835 if line == "---"
836 break
837 boundary := find_text(line, " {{ ")
838 selectors := line[:<boundary].split(", ")
839 body := line[boundary + 3:<#line - 2]
840 Declaration[] declarations := []
841 for declaration := body.split("; ")[]
842 colon := find_text(declaration, ": ")
843 declarations = declarations.concat(expand_declaration(declaration[:<colon], declaration[colon + 2:]))
844 rules = rules.append(Rule(selectors, declarations))
845 CssNode[] elements := []
846 nat[] stack := []
847 while true
848 scan(&line)
849 if line == "==="
850 break
851 nat spaces := 0
852 while spaces < #line and line[spaces] == " "
853 spaces += 1
854 depth := spaces / 2
855 rest := line[spaces:]
856 nat head_end := 0
857 while head_end < #rest and rest[head_end] != " "
858 head_end += 1
859 head := rest[:<head_end]
860 text := if head_end == #rest then "" else rest[head_end + 2:<#rest - 1]
861 hash_at := find_text(head, "#")
862 names := head[hash_at + 1:].split(".")
863 stack = stack[:<depth]
864 parent := if #stack == 0 then -1 else int(stack[-1])
865 node := CssNode(head[:<hash_at], names[0], names[1:], text, parent, depth)
866 map<string, int> winners
867 for rule := rules[]
868 specificity := -1
869 for selector := rule.selectors[]
870 score := -1
871 if selector.starts_with("#")
872 if selector[1:] == node.id
873 score = 2
874 elif selector.starts_with(".")
875 if has_text(node.classes, selector[1:])
876 score = 1
877 elif selector == node.tag
878 score = 0
879 specificity = maximum(specificity, score)
880 if specificity < 0
881 continue
882 for declaration := rule.declarations[]
883 previous := -1
884 match winners.get(declaration.prop)
885 int entry
886 previous = entry
887 none
888 void
889 if specificity >= previous
890 .winners.set(declaration.prop, specificity)
891 .node.style.set(declaration.prop, declaration.value)
892 index := #elements
893 elements = elements.append(node)
894 if parent >= 0
895 elements[parent].children = elements[parent].children.append(index)
896 stack = stack.append(index)
897 document := Document(int(dimensions[0]), int(dimensions[1]), elements)
898 .document.perform_layout()
899 while true
900 scan(&line)
901 if line == "END"
902 break
903 .document.execute(line)
