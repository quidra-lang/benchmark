92 static Dictionary<string, string> ParseDeclaration(string property, string value)
93 {
94 string[] tokens = value.Split(' ');
95 Dictionary<string, string> result = new();
96 if (OneOf(property, "margin padding border gap"))
97 {
98 bool gap = property == "gap";
99 if (tokens.Length < 1 || tokens.Length > (gap ? 2 : 4)) return null;
100 foreach (string token in tokens) if (!Integer(token) || property == "border" && !OneOf(token, "0 1")) return null;
101 if (gap)
102 {
103 result["row-gap"] = tokens[0];
104 result["column-gap"] = tokens[^1];
105 }
106 else
107 {
108 string[] sides = { "top", "right", "bottom", "left" };
109 string[] expanded = { tokens[0], tokens.Length > 1 ? tokens[1] : tokens[0], tokens.Length > 2 ? tokens[2] : tokens[0], tokens.Length > 3 ? tokens[3] : tokens.Length > 1 ? tokens[1] : tokens[0] };
110 for (int i = 0; i < 4; i++) result[property + "-" + sides[i]] = expanded[i];
111 }
112 return result;
113 }
114 if (!Initial.ContainsKey(property)) return null;
115 bool valid;
116 if (property.StartsWith("grid-template-", StringComparison.Ordinal)) valid = value == "none" || tokens.All(t => Integer(t) || t == "auto" || t.EndsWith("fr", StringComparison.Ordinal) && Positive(t[..^2]));
117 else if (property is "grid-column" or "grid-row")
118 valid = tokens.Length == 1 && (value == "auto" || Positive(value))
119 || tokens.Length == 2 && tokens[0] == "span" && Positive(tokens[1])
120 || tokens.Length == 3 && Positive(tokens[0]) && tokens[1] == "/" && Positive(tokens[2]) && int.Parse(tokens[0], CultureInfo.InvariantCulture) < int.Parse(tokens[2], CultureInfo.InvariantCulture)
121 || tokens.Length == 4 && Positive(tokens[0]) && tokens[1] == "/" && tokens[2] == "span" && Positive(tokens[3]);
122 else
123 {
124 if (tokens.Length != 1) return null;
125 valid = property switch
126 {
127 "display" => OneOf(value, "block flex grid none"),
128 "position" => OneOf(value, "static relative absolute"),
129 "box-sizing" => OneOf(value, "content-box border-box"),
130 "width" or "height" or "flex-basis" => value == "auto" || Integer(value) || Percentage(value),
131 "min-width" or "min-height" => Integer(value) || Percentage(value),
132 "max-width" or "max-height" => value == "none" || Integer(value) || Percentage(value),
133 "top" or "right" or "bottom" or "left" => value == "auto" || Signed(value),
134 "order" => Signed(value),
135 "flex-direction" => OneOf(value, "row row-reverse column column-reverse"),
136 "flex-wrap" => OneOf(value, "nowrap wrap"),
137 "justify-content" => OneOf(value, "start end center space-between space-around space-evenly"),
138 "align-items" => OneOf(value, "stretch start end center"),
139 "align-self" => OneOf(value, "auto stretch start end center"),
140 _ => property.StartsWith("border-", StringComparison.Ordinal) ? OneOf(value, "0 1") : Integer(value)
141 };
142 }
143 if (!valid) return null;
144 result[property] = value;
145 return result;
146 }

154 static int Resolve(Element element, string property, int reference, bool horizontal)
155 {
156 string value = element.Get(property);
157 if (value is "auto" or "none") return -1;
158 int n;
159 if (value.EndsWith('%'))
160 {
161 if (reference < 0) return -1;
162 n = (int)((long)reference * int.Parse(value[..^1], CultureInfo.InvariantCulture) / 100);
163 }
164 else n = int.Parse(value, CultureInfo.InvariantCulture);
165 return element.Get("box-sizing") == "border-box" ? Math.Max(0, n - (horizontal ? element.Hf : element.Vf)) : n;
166 }

168 static int Clamp(Element element, int value, int reference, bool horizontal)
169 {
170 string axis = horizontal ? "width" : "height";
171 int low = Resolve(element, "min-" + axis, reference, horizontal), high = Resolve(element, "max-" + axis, reference, horizontal);
172 if (high >= 0) value = Math.Min(value, high);
173 return Math.Max(value, Math.Max(0, low));
174 }

176 static int[] Distribute(int total, long[] weights)
177 {
178 int[] result = new int[weights.Length];
179 long sum = weights.Sum();
180 if (sum == 0) return result;
181 int remaining = total;
182 for (int i = 0; i < weights.Length; i++)
183 {
184 result[i] = (int)((long)total * weights[i] / sum);
185 remaining -= result[i];
186 }
187 for (int i = 0; i < weights.Length && remaining > 0; i++) if (weights[i] > 0)
188 {
189 result[i]++;
190 remaining--;
191 }
192 return result;
193 }

196 {
197 string[] t = value.Split(' ');
198 if (value == "auto") return new[] { 0, 1 };

201 return new[] { start, t.Length == 1 ? 1 : t.Length == 3 ? int.Parse(t[2], CultureInfo.InvariantCulture) - start : int.Parse(t[3], CultureInfo.InvariantCulture) };

216 static bool Overlaps(List<GridItem> items, GridItem item, int row, int column) => items.Any(other => other.Placed && row < other.Row + other.Rows && other.Row < row + item.Rows && column < other.Column + other.Columns && other.Column < column + item.Columns);

218 static Grid PlaceGrid(Element element)
219 {
220 List<GridItem> items = new();
221 int nc = Tracks(element.Get("grid-template-columns"), true).Count, nr = Tracks(element.Get("grid-template-rows"), false).Count;
222 foreach (Element child in Flow(element, true))
223 {
224 GridItem item = new(child);
225 items.Add(item);
226 nc = Math.Max(nc, item.Column == 0 ? item.Columns : item.Column + item.Columns - 1);
227 }
228 foreach (GridItem item in items) if (item.Row > 0 && item.Column > 0) item.Placed = true;
229 int cr = 1, cc = 1;
230 foreach (GridItem item in items)
231 {
232 if (item.Placed) continue;
233 if (item.Row > 0)
234 {
235 int c = 1;
236 while (c <= nc - item.Columns + 1 && Overlaps(items, item, item.Row, c)) c++;
237 item.Column = c <= nc - item.Columns + 1 ? c : 1;
238 }
239 else if (item.Column > 0)
240 {
241 if (item.Column < cc) cr++;
242 int r = cr;
243 while (Overlaps(items, item, r, item.Column)) r++;
244 item.Row = r;
245 cr = r;
246 cc = item.Column + item.Columns;
247 }
248 else
249 {
250 int r = cr, c = cc;
251 while (true)
252 {
253 if (c + item.Columns - 1 > nc)
254 {
255 r++;
256 c = 1;
257 }
258 else if (Overlaps(items, item, r, c)) c++;
259 else break;
260 }
261 item.Row = r;
262 item.Column = c;
263 cr = r;
264 cc = c + item.Columns;
265 }
266 item.Placed = true;
267 }
268 foreach (GridItem item in items) nr = Math.Max(nr, item.Row + item.Rows - 1);
269 return new Grid(items, nc, nr);
270 }

272 static int[] TrackSizes(Element element, Grid grid, bool columns, int available, bool intrinsic)
273 {
274 int count = columns ? grid.Columns : grid.Rows;
275 List<Track> specs = Tracks(element.Get(columns ? "grid-template-columns" : "grid-template-rows"), columns);
276 while (specs.Count < count) specs.Add(new Track('a', 0));
277 int[] sizes = new int[count];
278 for (int i = 0; i < count; i++)
279 {
280 Track track = specs[i];
281 if (track.Kind == 'n') sizes[i] = track.Value;
282 else if (track.Kind == 'a' || available < 0) foreach (GridItem item in grid.Items) if ((columns ? item.Column : item.Row) == i + 1 && (columns ? item.Columns : item.Rows) == 1) sizes[i] = Math.Max(sizes[i], intrinsic ? Contribution(item.Element) : item.Contribution);
283 }
284 int gap = element.Number(columns ? "column-gap" : "row-gap");
285 int leftover = available - sizes.Sum() - gap * Math.Max(0, count - 1);
286 if (available >= 0 && leftover > 0)
287 {
288 bool fr = specs.Any(t => t.Kind == 'f');
289 long[] weights = new long[count];
290 for (int i = 0; i < count; i++) if (fr ? specs[i].Kind == 'f' : specs[i].Kind == 'a') weights[i] = fr ? specs[i].Value : 1;
291 int[] extra = Distribute(leftover, weights);
292 for (int i = 0; i < count; i++) sizes[i] += extra[i];
293 }
294 return sizes;
295 }

297 static int MaxContent(Element element)
298 {
299 if (element.Intrinsic >= 0) return element.Intrinsic;
300 int result = 0;
301 if (element.Text != null) result = element.Text.Length;
302 else if (element.Get("display") == "flex" && element.Get("flex-direction").StartsWith("row", StringComparison.Ordinal))
303 {
304 List<Element> children = Flow(element, false);
305 result = children.Sum(Contribution) + element.Number("column-gap") * Math.Max(0, children.Count - 1);
306 }
307 else if (element.Get("display") == "grid")
308 {
309 Grid grid = PlaceGrid(element);
310 result = TrackSizes(element, grid, true, -1, true).Sum() + element.Number("column-gap") * (grid.Columns - 1);
311 }
312 else foreach (Element child in Flow(element, false)) result = Math.Max(result, Contribution(child));
313 element.Intrinsic = result;
314 return result;
315 }

333 static int Layout(Element element, int widthReference, int heightReference, int forcedWidth, int forcedHeight)
334 {
335 int w = Resolve(element, "width", widthReference, true);
336 element.Cw = forcedWidth >= 0 ? forcedWidth : Clamp(element, w < 0 ? Math.Max(0, widthReference - element.Hm - element.Hf) : w, widthReference, true);
337 if (forcedHeight < 0)
338 {
339 int h = Resolve(element, "height", heightReference, false);
340 if (h >= 0) forcedHeight = Clamp(element, h, heightReference, false);
341 }
342 int a;
343 if (element.Text != null)
344 {
345 element.Lines = new();
346 string current = "";
347 foreach (string word in element.Text.Split(' '))
348 {
349 if (current.Length == 0) current = word;
350 else if (current.Length + 1 + word.Length <= element.Cw) current += " " + word;
351 else
352 {
353 element.Lines.Add(current);
354 current = word;
355 }
356 }
357 element.Lines.Add(current);
358 a = element.Lines.Count;
359 }
360 else if (element.Get("display") == "flex") a = Flex(element, forcedHeight, heightReference);
361 else if (element.Get("display") == "grid") a = GridLayout(element, forcedHeight);
362 else
363 {
364 a = 0;
365 int previousBottom = 0;
366 bool first = true;
367 foreach (Element child in Flow(element, false))
368 {
369 Layout(child, element.Cw, forcedHeight, -1, -1);
370 int y = a + (first ? child.Number("margin-top") : Math.Max(previousBottom, child.Number("margin-top")));
371 Position(element, child, child.Number("margin-left"), y);
372 a = y + child.Bh;
373 previousBottom = child.Number("margin-bottom");
374 first = false;
375 }
376 if (!first) a += previousBottom;
377 }
378 element.Ch = forcedHeight >= 0 ? forcedHeight : Clamp(element, a, heightReference, false);
379 return a;
380 }

382 static int Flex(Element element, int k, int heightReference)
383 {
384 bool horizontal = element.Get("flex-direction").StartsWith("row", StringComparison.Ordinal), reverse = element.Get("flex-direction").EndsWith("reverse", StringComparison.Ordinal);
385 int reference = horizontal ? element.Cw : k, gap = element.Number(horizontal ? "column-gap" : "row-gap");
386 List<FlexItem> items = new();
387 foreach (Element child in Flow(element, true))
388 {
389 FlexItem item = new(child);
390 items.Add(item);
391 item.Outer = horizontal ? child.Hm + child.Hf : child.Vm + child.Vf;
392 if (!horizontal)
393 {
394 int r = Math.Max(0, element.Cw - child.Hm - child.Hf), width = Resolve(child, "width", element.Cw, true);
395 item.Cross = Clamp(child, width >= 0 ? width : Align(element, child) == "stretch" ? r : Math.Min(MaxContent(child), r), element.Cw, true);
396 }
397 item.Base = Resolve(child, "flex-basis", reference, horizontal);
398 if (item.Base < 0) item.Base = Resolve(child, horizontal ? "width" : "height", reference, horizontal);
399 if (item.Base < 0) item.Base = horizontal ? MaxContent(child) : Layout(child, element.Cw, k, item.Cross, -1);
400 item.Hypothetical = Clamp(child, item.Base, reference, horizontal);
401 }
402 int length = reference;
403 if (length < 0)
404 {
405 length = gap * Math.Max(0, items.Count - 1);
406 foreach (FlexItem item in items) length += item.Hypothetical + item.Outer;
407 length = Clamp(element, length, heightReference, false);
408 }
409 List<List<FlexItem>> lines = new();
410 int occupied = 0;
411 foreach (FlexItem item in items)
412 {
413 if (lines.Count == 0 || horizontal && element.Get("flex-wrap") == "wrap" && occupied + gap + item.Hypothetical + item.Outer > length)
414 {
415 lines.Add(new());
416 occupied = 0;
417 }
418 List<FlexItem> line = lines[^1];
419 if (line.Count > 0) occupied += gap;
420 occupied += item.Hypothetical + item.Outer;
421 line.Add(item);
422 }
423 int crossStart = 0;
424 foreach (List<FlexItem> line in lines)
425 {
426 int total = gap * (line.Count - 1);
427 foreach (FlexItem item in line) total += item.Hypothetical + item.Outer;
428 bool grow = total < length;
429 foreach (FlexItem item in line)
430 {
431 item.Weight = grow ? item.Element.Number("flex-grow") : (long)item.Element.Number("flex-shrink") * item.Base;
432 item.Size = item.Hypothetical;
433 item.Frozen = item.Weight == 0 || (grow ? item.Base > item.Hypothetical : item.Base < item.Hypothetical);
434 }
435 while (true)
436 {
437 int free = length - gap * (line.Count - 1), remaining = 0;
438 long[] weights = new long[line.Count];
439 for (int i = 0; i < line.Count; i++)
440 {
441 FlexItem item = line[i];
442 free -= (item.Frozen ? item.Size : item.Base) + item.Outer;
443 if (!item.Frozen)
444 {
445 weights[i] = item.Weight;
446 remaining++;
447 }
448 }
449 if (remaining == 0) break;
450 int[] shares = Distribute(Math.Max(0, grow ? free : -free), weights), violations = new int[line.Count];
451 int violation = 0;
452 for (int i = 0; i < line.Count; i++)
453 {
454 FlexItem item = line[i];
455 if (!item.Frozen)
456 {
457 int tentative = item.Base + (grow ? shares[i] : -shares[i]);
458 item.Size = Clamp(item.Element, tentative, reference, horizontal);
459 violations[i] = item.Size - tentative;
460 violation += violations[i];
461 }
462 }
463 for (int i = 0; i < line.Count; i++) if (violation == 0 || violation > 0 && violations[i] > 0 || violation < 0 && violations[i] < 0) line[i].Frozen = true;
464 }
465 int lc = horizontal ? 0 : element.Cw;
466 if (horizontal)
467 {
468 foreach (FlexItem item in line)
469 {
470 Layout(item.Element, element.Cw, k, item.Size, -1);
471 item.Cross = item.Element.Ch;
472 lc = Math.Max(lc, item.Cross + item.Element.Vf + item.Element.Vm);
473 }
474 if (lines.Count == 1) lc = k >= 0 ? k : Clamp(element, lc, heightReference, false);
475 }
476 int used = gap * (line.Count - 1);
477 foreach (FlexItem item in line) used += item.Size + item.Outer;
478 int space = length - used;
479 int[] before = new int[line.Count + 1];
480 if (space > 0)
481 {
482 string justify = element.Get("justify-content");
483 if (justify == "end") before[0] = space;
484 else if (justify == "center") before[0] = space / 2;
485 else if (justify.StartsWith("space-", StringComparison.Ordinal))
486 {
487 long[] weights = Enumerable.Repeat(1L, before.Length).ToArray();
488 if (justify == "space-between")
489 {
490 weights[0] = 0;
491 weights[^1] = 0;
492 }
493 else if (justify == "space-around") for (int i = 1; i < weights.Length - 1; i++) weights[i] = 2;
494 before = Distribute(space, weights);
495 }
496 }
497 int cursor = 0;
498 for (int i = 0; i < line.Count; i++)
499 {
500 FlexItem item = line[i];
501 Element child = item.Element;
502 string alignment = Align(element, child);
503 if (horizontal)
504 {
505 if (alignment == "stretch" && child.Get("height") == "auto")
506 {
507 item.Cross = Clamp(child, Math.Max(0, lc - child.Vm - child.Vf), k, false);
508 Layout(child, element.Cw, k, item.Size, item.Cross);
509 }
510 else Layout(child, element.Cw, k, item.Size, -1);
511 }
512 else Layout(child, element.Cw, k, item.Cross, item.Size);
513 int cross = crossStart + Offset(alignment, lc - item.Cross - (horizontal ? child.Vf + child.Vm : child.Hf + child.Hm)) + child.Number(horizontal ? "margin-top" : "margin-left");
514 cursor += before[i];
515 int main = reverse ? length - cursor - child.Number(horizontal ? "margin-right" : "margin-bottom") - item.Size - (horizontal ? child.Hf : child.Vf) : cursor + child.Number(horizontal ? "margin-left" : "margin-top");
516 Position(element, child, horizontal ? main : cross, horizontal ? cross : main);
517 cursor += item.Size + item.Outer + gap;
518 }
519 crossStart += lc + element.Number("row-gap");
520 }
521 return horizontal ? lines.Count == 0 ? 0 : crossStart - element.Number("row-gap") : length;
522 }

538 static int GridLayout(Element element, int k)
539 {
540 Grid grid = PlaceGrid(element);
541 int cg = element.Number("column-gap"), rg = element.Number("row-gap");
542 int[] columns = TrackSizes(element, grid, true, element.Cw, true);
543 foreach (GridItem item in grid.Items)
544 {
545 item.Aw = Area(columns, item.Column, item.Columns, cg);
546 item.Ax = Prefix(columns, item.Column - 1, cg);
547 int width = Resolve(item.Element, "width", item.Aw, true);
548 item.Width = Clamp(item.Element, width < 0 ? Math.Max(0, item.Aw - item.Element.Hm - item.Element.Hf) : width, item.Aw, true);
549 Layout(item.Element, item.Aw, -1, item.Width, -1);
550 item.Contribution = item.Element.Bh + item.Element.Vm;
551 }
552 int[] rows = TrackSizes(element, grid, false, k, false);
553 foreach (GridItem item in grid.Items)
554 {
555 int ah = Area(rows, item.Row, item.Rows, rg), ay = Prefix(rows, item.Row - 1, rg);
556 string alignment = Align(element, item.Element);
557 int fh = alignment == "stretch" && item.Element.Get("height") == "auto" ? Clamp(item.Element, Math.Max(0, ah - item.Element.Vm - item.Element.Vf), ah, false) : -1;
558 Layout(item.Element, item.Aw, ah, item.Width, fh);
559 Position(element, item.Element, item.Ax + item.Element.Number("margin-left"), ay + Offset(alignment, ah - item.Element.Bh - item.Element.Vm) + item.Element.Number("margin-top"));
560 }
561 return rows.Sum() + rg * Math.Max(0, grid.Rows - 1);
562 }

564 static void Relayout()
565 {
566 foreach (Element element in Elements)
567 {
568 element.Displayed = element.Get("display") != "none" && (element.Parent == null || element.Parent.Displayed);
569 element.Intrinsic = -1;
570 element.Containing = null;
571 }
572 Element root = Elements[0];
573 if (root.Displayed && root.Get("position") != "absolute")
574 {
575 Layout(root, viewportWidth, viewportHeight, -1, -1);
576 root.Rx = root.Number("margin-left");
577 root.Ry = root.Number("margin-top");
578 }
579 foreach (Element element in Elements) if (element.Displayed && element.Get("position") == "absolute")
580 {
581 Element cb = element.Parent;
582 while (cb != null && cb.Get("position") == "static") cb = cb.Parent;
583 element.Containing = cb;
584 int width = cb == null ? viewportWidth : cb.Cw + cb.Number("padding-left") + cb.Number("padding-right"), height = cb == null ? viewportHeight : cb.Ch + cb.Number("padding-top") + cb.Number("padding-bottom");
585 int left = element.Get("left") == "auto" ? 0 : element.Number("left"), right = element.Get("right") == "auto" ? 0 : element.Number("right"), top = element.Get("top") == "auto" ? 0 : element.Number("top"), bottom = element.Get("bottom") == "auto" ? 0 : element.Number("bottom");
586 int w = Resolve(element, "width", width, true);
587 if (element.Get("width") == "auto")
588 {
589 int r = Math.Max(0, width - left - right - element.Hm - element.Hf);
590 w = element.Get("left") != "auto" && element.Get("right") != "auto" ? r : Math.Min(MaxContent(element), r);
591 }
592 int h = element.Get("height") == "auto" && element.Get("top") != "auto" && element.Get("bottom") != "auto" ? Clamp(element, Math.Max(0, height - top - bottom - element.Vm - element.Vf), height, false) : -1;
593 Layout(element, width, height, Clamp(element, w, width, true), h);
594 element.Rx = element.Get("left") != "auto" ? left + element.Number("margin-left") : element.Get("right") != "auto" ? width - right - element.Number("margin-right") - element.Bw : element.Number("margin-left");
595 element.Ry = element.Get("top") != "auto" ? top + element.Number("margin-top") : element.Get("bottom") != "auto" ? height - bottom - element.Number("margin-bottom") - element.Bh : element.Number("margin-top");
596 }
597 foreach (Element element in Elements) if (element.Displayed)
598 {
599 Element reference = element.Get("position") == "absolute" ? element.Containing : element.Parent;
600 element.X = element.Rx + (reference == null ? 0 : reference.X + (element.Get("position") == "absolute" ? reference.Number("border-left") : 0));
601 element.Y = element.Ry + (reference == null ? 0 : reference.Y + (element.Get("position") == "absolute" ? reference.Number("border-top") : 0));
602 if (element.Get("position") == "relative")
603 {
604 element.X += element.Get("left") != "auto" ? element.Number("left") : element.Get("right") != "auto" ? -element.Number("right") : 0;
605 element.Y += element.Get("top") != "auto" ? element.Number("top") : element.Get("bottom") != "auto" ? -element.Number("bottom") : 0;
606 }
607 }
608 }

610 static int[][] Snapshot() => Elements.Select(e => new[] { e.Displayed ? 1 : 0, e.X, e.Y, e.Bw, e.Bh }).ToArray();

612 static string Changes(int[][] old)
613 {
614 int[][] now = Snapshot();
615 List<string> changed = Elements.Where(e => old[e.Index][0] != now[e.Index][0] || e.Displayed && !old[e.Index].SequenceEqual(now[e.Index])).Select(e => e.Id).ToList();
616 return changed.Count == 0 ? "no change" : string.Join(" ", changed);
617 }

619 static string Rectangle(string name, int x, int y, int width, int height) => $"{name} {x} {y} {width} {height}\n";

621 static void Render(StringBuilder output)
622 {
623 char[][] screen = Enumerable.Range(0, viewportHeight).Select(_ => Enumerable.Repeat('.', viewportWidth).ToArray()).ToArray();
624 foreach (Element element in Elements) if (element.Displayed)
625 {
626 for (int y = Math.Max(0, element.Y); y < Math.Min(viewportHeight, element.Y + element.Bh); y++) for (int x = Math.Max(0, element.X); x < Math.Min(viewportWidth, element.X + element.Bw); x++)
627 {
628 bool horizontal = element.Number("border-top") == 1 && y == element.Y || element.Number("border-bottom") == 1 && y == element.Y + element.Bh - 1, vertical = element.Number("border-left") == 1 && x == element.X || element.Number("border-right") == 1 && x == element.X + element.Bw - 1;
629 screen[y][x] = horizontal ? vertical ? '+' : '-' : vertical ? '|' : ' ';
630 }
631 if (element.Text != null) for (int i = 0; i < element.Lines.Count; i++)
632 {
633 int y = element.Cy + i;
634 if (y < 0 || y >= viewportHeight) continue;
635 string line = element.Lines[i];
636 for (int j = 0; j < line.Length; j++)
637 {
638 int x = element.Cx + j;
639 if (x >= 0 && x < viewportWidth) screen[y][x] = line[j];
640 }
641 }
642 }
643 string edge = new('#', viewportWidth + 2);
644 output.Append($"render {viewportWidth}x{viewportHeight}\n{edge}\n");
645 foreach (char[] row in screen) output.Append('#').Append(row).Append("#\n");
646 output.Append(edge).Append('\n');
647 }

649 static void Command(string line, StringBuilder output)
650 {
651 string[] t = line.Split(' ');
652 Element element = t.Length > 1 ? Elements.FirstOrDefault(e => e.Id == t[1]) : null;
653 switch (t[0])
654 {
655 case "LAYOUT":
656 output.Append($"layout {viewportWidth}x{viewportHeight}\n");
657 foreach (Element e in Elements) output.Append(new string(' ', 2 * e.Depth)).Append(e.Displayed ? Rectangle(e.Id, e.X, e.Y, e.Bw, e.Bh) : $"{e.Id} none\n");
658 break;
659 case "BOX":
660 if (element == null) output.Append($"error: unknown element {t[1]}\n");
661 else if (!element.Displayed) output.Append($"box {element.Id}: none\n");
662 else
663 {
664 output.Append($"box {element.Id}\n");
665 output.Append(Rectangle("margin", element.X - element.Number("margin-left"), element.Y - element.Number("margin-top"), element.Bw + element.Hm, element.Bh + element.Vm));
666 output.Append(Rectangle("border", element.X, element.Y, element.Bw, element.Bh));
667 output.Append(Rectangle("padding", element.X + element.Number("border-left"), element.Y + element.Number("border-top"), element.Cw + element.Number("padding-left") + element.Number("padding-right"), element.Ch + element.Number("padding-top") + element.Number("padding-bottom")));
668 output.Append(Rectangle("content", element.Cx, element.Cy, element.Cw, element.Ch));
669 }
670 break;
671 case "RENDER":
672 Render(output);
673 break;
674 case "HIT":
675 {
676 int x = int.Parse(t[1], CultureInfo.InvariantCulture), y = int.Parse(t[2], CultureInfo.InvariantCulture);
677 if (x >= viewportWidth || y >= viewportHeight) output.Append($"error: point {x} {y} outside viewport\n");
678 else
679 {
680 Element hit = Elements.LastOrDefault(e => e.Displayed && x >= e.X && x < e.X + e.Bw && y >= e.Y && y < e.Y + e.Bh);
681 List<string> path = new();
682 for (; hit != null; hit = hit.Parent) path.Insert(0, hit.Id);
683 output.Append($"hit {x} {y}: {(path.Count == 0 ? "none" : string.Join(" > ", path))}\n");
684 }
685 break;
686 }
687 case "SET":
688 {
689 string value = line[(t[0].Length + t[1].Length + t[2].Length + 3)..];
690 if (element == null) output.Append($"error: unknown element {t[1]}\n");
691 else if (!Known(t[2])) output.Append($"error: unknown property {t[2]}\n");
692 else
693 {
694 Dictionary<string, string> values = ParseDeclaration(t[2], value);
695 if (values == null) output.Append($"error: invalid value for {t[2]}: {value}\n");
696 else
697 {
698 int[][] old = Snapshot();
699 foreach (var pair in values) element.Style[pair.Key] = pair.Value;
700 Relayout();
701 output.Append($"set {element.Id} {t[2]} {value}: {Changes(old)}\n");
702 }
703 }
704 break;
705 }
706 case "VIEWPORT":
707 {
708 int w = int.Parse(t[1], CultureInfo.InvariantCulture), h = int.Parse(t[2], CultureInfo.InvariantCulture);
709 if (w < 1 || w > 80 || h < 1 || h > 40) output.Append($"error: invalid viewport {w} {h}\n");
710 else
711 {
712 int[][] old = Snapshot();
713 viewportWidth = w;
714 viewportHeight = h;
715 Relayout();
716 output.Append($"viewport {w} {h}: {Changes(old)}\n");
717 }
718 break;
719 }
720 case "OVERFLOW":
721 {
722 List<string> list = new();
723 foreach (Element e in Elements) if (e.Displayed)
724 {
725 bool overflow = e.Text != null && (e.Lines.Count > e.Ch || e.Lines.Any(s => s.Length > e.Cw));
726 if (e.Text == null) foreach (Element child in Flow(e, false)) if (child.X - child.Number("margin-left") < e.Cx || child.Y - child.Number("margin-top") < e.Cy || child.X + child.Bw + child.Number("margin-right") > e.Cx + e.Cw || child.Y + child.Bh + child.Number("margin-bottom") > e.Cy + e.Ch) overflow = true;
727 if (overflow) list.Add(e.Id);
728 }
729 output.Append($"overflow: {(list.Count == 0 ? "none" : string.Join(" ", list))}\n");
730 break;
731 }
732 default:
733 output.Append($"error: unknown command {t[0]}\n");
734 break;
735 }
736 }

738 static void Main()
739 {
740 CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
741 string[] dimensions = Console.ReadLine().Split(' ');
742 viewportWidth = int.Parse(dimensions[0], CultureInfo.InvariantCulture);
743 viewportHeight = int.Parse(dimensions[1], CultureInfo.InvariantCulture);
744 List<Rule> rules = new();
745 string line;
746 while ((line = Console.ReadLine()) != "---")
747 {
748 int brace = line.IndexOf(" { ", StringComparison.Ordinal);
749 List<Declaration> declarations = new();
750 foreach (string raw in line[(brace + 3)..^2].Split("; ", StringSplitOptions.None))
751 {
752 int colon = raw.IndexOf(": ", StringComparison.Ordinal);
753 Dictionary<string, string> values = ParseDeclaration(raw[..colon], raw[(colon + 2)..]);
754 if (values != null) foreach (var pair in values) declarations.Add(new Declaration(pair.Key, pair.Value));
755 }
756 rules.Add(new Rule(line[..brace].Split(", ", StringSplitOptions.None), declarations));
757 }
758 Element[] ancestors = new Element[11];
759 while ((line = Console.ReadLine()) != "===")
760 {
761 int spaces = 0;
762 while (line[spaces] == ' ') spaces++;
763 Element element = new() { Depth = spaces / 2, Index = Elements.Count };
764 string body = line[spaces..];
765 int end = body.IndexOf(' ');
766 string head = end < 0 ? body : body[..end];
767 if (end >= 0) element.Text = body[(end + 2)..^1];
768 string[] parts = head.Split(new[] { '#', '.' });
769 element.Type = parts[0];
770 element.Id = parts[1];
771 element.Classes.AddRange(parts.Skip(2));
772 if (element.Depth > 0)
773 {
774 element.Parent = ancestors[element.Depth - 1];
775 element.Parent.Children.Add(element);
776 }
777 ancestors[element.Depth] = element;
778 Elements.Add(element);
779 Dictionary<string, int> scores = new();
780 foreach (Rule rule in rules)
781 {
782 int score = -1;
783 foreach (string selector in rule.Selectors)
784 {
785 if (selector.StartsWith('#') && selector[1..] == element.Id) score = Math.Max(score, 2);
786 else if (selector.StartsWith('.') && element.Classes.Contains(selector[1..])) score = Math.Max(score, 1);
787 else if (selector == element.Type) score = Math.Max(score, 0);
788 }
789 if (score >= 0) foreach (Declaration declaration in rule.Declarations) if (score >= scores.GetValueOrDefault(declaration.Property, -1))
790 {
791 element.Style[declaration.Property] = declaration.Value;
792 scores[declaration.Property] = score;
793 }
794 }
795 }
796 Relayout();
797 while ((line = Console.ReadLine()) != "END")
798 {
799 StringBuilder output = new();
800 Command(line, output);
801 Console.Write(output);
802 }
803 }
