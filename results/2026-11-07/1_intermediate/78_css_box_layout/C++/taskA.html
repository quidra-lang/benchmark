17static std::vector<std::string> split(const std::string& text, const std::string& separator) {
18 std::vector<std::string> result;
19 std::size_t start = 0;
20 while (true) {
21 std::size_t end = text.find(separator, start);
22 result.push_back(text.substr(start, end == std::string::npos ? end : end - start));
23 if (end == std::string::npos) return result;
24 start = end + separator.size();
25 }
26}

28static std::string trim(const std::string& text) {
29 std::size_t first = text.find_first_not_of(' ');
30 return first == std::string::npos ? "" : text.substr(first, text.find_last_not_of(' ') - first + 1);
31}

33static std::string join(const std::vector<std::string>& names, const std::string& separator = " ") {
34 std::string result;
35 for (const std::string& name : names) {
36 if (!result.empty()) result += separator;
37 result += name;
38 }
39 return result;
40}

62static bool integer(const std::string& text) {
63 return !text.empty() && text.size() <= 3 && std::all_of(text.begin(), text.end(), [](char c) { return c >= '0' && c <= '9'; });
64}

66static bool positive(const std::string& text) { return integer(text) && std::stoi(text) > 0; }
67static bool signedInteger(const std::string& text) { return integer(text) || (!text.empty() && text[0] == '-' && integer(text.substr(1))); }
68static bool percentage(const std::string& text) { return !text.empty() && text.back() == '%' && integer(text.substr(0, text.size() - 1)); }
69static bool fraction(const std::string& text) { return text.size() > 2 && text.substr(text.size() - 2) == "fr" && positive(text.substr(0, text.size() - 2)); }

71static bool known(const std::string& property) {
72 return initialValues().count(property) || property == "margin" || property == "padding" || property == "border" || property == "gap";
73}

75static std::optional<Declarations> declaration(const std::string& property, const std::string& value) {
76 if (!known(property)) return std::nullopt;
77 std::vector<std::string> tokens = split(value, " ");
78 if (property == "margin" || property == "padding" || property == "border") {
79 if (tokens.size() > 4) return std::nullopt;
80 for (const std::string& token : tokens) {
81 if (property == "border" ? token != "0" && token != "1" : !integer(token)) return std::nullopt;
82 }
83 std::array<std::string, 4> expanded = {tokens[0], tokens.size() >= 2 ? tokens[1] : tokens[0],
84 tokens.size() >= 3 ? tokens[2] : tokens[0], tokens.size() == 4 ? tokens[3] : (tokens.size() >= 2 ? tokens[1] : tokens[0])};
85 Declarations result;
86 const char* sides[] = {"top", "right", "bottom", "left"};
87 for (int i = 0; i < 4; ++i) result.emplace_back(property + '-' + sides[i], expanded[i]);
88 return result;
89 }
90 if (property == "gap") {
91 if (tokens.size() > 2 || !std::all_of(tokens.begin(), tokens.end(), integer)) return std::nullopt;
92 return Declarations{{"row-gap", tokens[0]}, {"column-gap", tokens.back()}};
93 }
94 bool valid = false;
95 if (property == "grid-template-columns" || property == "grid-template-rows") {
96 valid = value == "none" || std::all_of(tokens.begin(), tokens.end(), [](const std::string& token) { return token == "auto" || integer(token) || fraction(token); });
97 } else if (property == "grid-column" || property == "grid-row") {
98 valid = value == "auto" || (tokens.size() == 1 && positive(tokens[0]))
99 || (tokens.size() == 2 && tokens[0] == "span" && positive(tokens[1]))
100 || (tokens.size() == 3 && positive(tokens[0]) && tokens[1] == "/" && positive(tokens[2]) && std::stoi(tokens[0]) < std::stoi(tokens[2]))
101 || (tokens.size() == 4 && positive(tokens[0]) && tokens[1] == "/" && tokens[2] == "span" && positive(tokens[3]));
102 } else if (tokens.size() == 1) {
103 static const std::map<std::string, std::set<std::string>> choices = {
104 {"display", {"block", "flex", "grid", "none"}}, {"position", {"static", "relative", "absolute"}},
105 {"box-sizing", {"content-box", "border-box"}}, {"flex-direction", {"row", "row-reverse", "column", "column-reverse"}},
106 {"flex-wrap", {"nowrap", "wrap"}}, {"justify-content", {"start", "end", "center", "space-between", "space-around", "space-evenly"}},
107 {"align-items", {"stretch", "start", "end", "center"}}, {"align-self", {"auto", "stretch", "start", "end", "center"}}
108 };
109 auto choice = choices.find(property);
110 if (choice != choices.end()) valid = choice->second.count(value) != 0;
111 else if (property == "width" || property == "height" || property == "flex-basis") valid = value == "auto" || integer(value) || percentage(value);
112 else if (property == "min-width" || property == "min-height") valid = integer(value) || percentage(value);
113 else if (property == "max-width" || property == "max-height") valid = value == "none" || integer(value) || percentage(value);
114 else if (property == "top" || property == "right" || property == "bottom" || property == "left") valid = value == "auto" || signedInteger(value);
115 else if (property == "order") valid = signedInteger(value);
116 else if (property.compare(0, 7, "border-") == 0) valid = value == "0" || value == "1";
117 else valid = integer(value);
118 }
119 return valid ? std::optional<Declarations>(Declarations{{property, value}}) : std::nullopt;
120}

187 std::optional<int> resolve(int index, const std::string& property, std::optional<int> reference, bool horizontal) const {
188 const Element& element = elements[index];
189 const std::string& value = element.get(property);
190 if (value == "auto" || value == "none" || (value.back() == '%' && !reference)) return std::nullopt;
191 int result = value.back() == '%' ? static_cast<int>(static_cast<long long>(*reference) * std::stoi(value) / 100) : std::stoi(value);
192 if (element.get("box-sizing") == "border-box") result = std::max(0, result - element.frame(horizontal));
193 return result;
194 }

196 int clamp(int index, int value, std::optional<int> reference, bool horizontal) const {
197 std::string axis = horizontal ? "width" : "height";
198 int lower = resolve(index, "min-" + axis, reference, horizontal).value_or(0);
199 std::optional<int> upper = resolve(index, "max-" + axis, reference, horizontal);
200 if (upper) value = std::min(value, *upper);
201 return std::max(value, lower);
202 }

204 static std::vector<int> distribute(int total, const std::vector<int>& weights) {
205 int sum = std::accumulate(weights.begin(), weights.end(), 0), assigned = 0;
206 std::vector<int> result(weights.size(), 0);
207 if (sum == 0) return result;
208 for (std::size_t i = 0; i < weights.size(); ++i) {
209 result[i] = static_cast<int>(static_cast<long long>(total) * weights[i] / sum);
210 assigned += result[i];
211 }
212 for (std::size_t i = 0; i < weights.size() && assigned < total; ++i) {
213 if (weights[i] > 0) {
214 ++result[i];
215 ++assigned;
216 }
217 }
218 return result;
219 }

221 static std::pair<int, int> gridPosition(const std::string& value) {
222 std::vector<std::string> parts = split(value, " ");
223 if (value == "auto") return {0, 1};
224 if (parts[0] == "span") return {0, std::stoi(parts[1])};
225 int start = std::stoi(parts[0]);
226 if (parts.size() == 1) return {start, 1};
227 return {start, parts[2] == "span" ? std::stoi(parts[3]) : std::stoi(parts[2]) - start};
228 }

230 Grid placement(int index) const {
231 const Element& element = elements[index];
232 Grid grid;
233 grid.columnTracks = element.get("grid-template-columns") == "none" ? std::vector<std::string>{"auto"} : split(element.get("grid-template-columns"), " ");
234 grid.rowTracks = element.get("grid-template-rows") == "none" ? std::vector<std::string>{} : split(element.get("grid-template-rows"), " ");
235 grid.columns = static_cast<int>(grid.columnTracks.size());
236 grid.rows = static_cast<int>(grid.rowTracks.size());
237 for (int child : children(index, true)) {
238 auto [column, columnSpan] = gridPosition(elements[child].get("grid-column"));
239 auto [row, rowSpan] = gridPosition(elements[child].get("grid-row"));
240 grid.items.push_back({child, column, row, columnSpan, rowSpan});
241 grid.columns = std::max(grid.columns, column == 0 ? columnSpan : column + columnSpan - 1);
242 }
243 std::set<std::pair<int, int>> occupied;
244 auto overlaps = [&](const Placement& item, int row, int column) {
245 for (int r = row; r < row + item.rowSpan; ++r) {
246 for (int c = column; c < column + item.columnSpan; ++c) {
247 if (occupied.count({r, c})) return true;
248 }
249 }
250 return false;
251 };
252 auto occupy = [&](const Placement& item) {
253 for (int r = item.row; r < item.row + item.rowSpan; ++r) {
254 for (int c = item.column; c < item.column + item.columnSpan; ++c) occupied.insert({r, c});
255 }
256 };
257 for (const Placement& item : grid.items) {
258 if (item.row && item.column) occupy(item);
259 }
260 int cursorRow = 1, cursorColumn = 1;
261 for (Placement& item : grid.items) {
262 if (item.row && item.column) continue;
263 if (item.row) {
264 int column = 1;
265 while (column + item.columnSpan - 1 <= grid.columns && overlaps(item, item.row, column)) ++column;
266 item.column = column + item.columnSpan - 1 <= grid.columns ? column : 1;
267 } else if (item.column) {
268 if (item.column < cursorColumn) ++cursorRow;
269 item.row = cursorRow;
270 while (overlaps(item, item.row, item.column)) ++item.row;
271 cursorRow = item.row;
272 cursorColumn = item.column + item.columnSpan;
273 } else {
274 item.row = cursorRow;
275 item.column = cursorColumn;
276 while (true) {
277 if (item.column + item.columnSpan - 1 > grid.columns) {
278 ++item.row;
279 item.column = 1;
280 }
281 else if (overlaps(item, item.row, item.column)) ++item.column;
282 else break;
283 }
284 cursorRow = item.row;
285 cursorColumn = item.column + item.columnSpan;
286 }
287 occupy(item);
288 }
289 for (const Placement& item : grid.items) grid.rows = std::max(grid.rows, item.row + item.rowSpan - 1);
290 return grid;
291 }

293 std::vector<int> trackSizes(const Grid& grid, bool horizontal, std::optional<int> available, int gap, const std::vector<int>& contributions) const {
294 int count = horizontal ? grid.columns : grid.rows;
295 std::vector<std::string> specifications = horizontal ? grid.columnTracks : grid.rowTracks;
296 specifications.resize(count, "auto");
297 std::vector<int> sizes(count, 0), weights(count, 0);
298 bool hasFractions = std::any_of(specifications.begin(), specifications.end(), fraction);
299 for (int i = 0; i < count; ++i) {
300 const std::string& specification = specifications[i];
301 if (integer(specification)) sizes[i] = std::stoi(specification);
302 else {
303 if (specification == "auto" || !available) {
304 for (std::size_t j = 0; j < grid.items.size(); ++j) {
305 const Placement& item = grid.items[j];
306 if ((horizontal ? item.column : item.row) == i + 1 && (horizontal ? item.columnSpan : item.rowSpan) == 1) sizes[i] = std::max(sizes[i], contributions[j]);
307 }
308 }
309 weights[i] = hasFractions ? (fraction(specification) ? std::stoi(specification) : 0) : (specification == "auto" ? 1 : 0);
310 }
311 }
312 if (available && count > 0) {
313 int leftover = *available - std::accumulate(sizes.begin(), sizes.end(), 0) - gap * (count - 1);
314 if (leftover > 0) {
315 std::vector<int> additions = distribute(leftover, weights);
316 for (int i = 0; i < count; ++i) sizes[i] += additions[i];
317 }
318 }
319 return sizes;
320 }

322 int contribution(int index) {
323 Element& element = elements[index];
324 if (!element.contribution) {
325 int content = resolve(index, "width", std::nullopt, true).value_or(maxContent(index));
326 element.contribution = clamp(index, content, std::nullopt, true) + element.frame(true) + element.margins(true);
327 }
328 return *element.contribution;
329 }
330
331 int maxContent(int index) {
332 Element& element = elements[index];
333 if (element.maxContent) return *element.maxContent;
334 int result = 0;
335 if (element.text) result = static_cast<int>(element.text->size());
336 else if (element.get("display") == "grid") {
337 Grid grid = placement(index);
338 std::vector<int> contributions;
339 for (const Placement& item : grid.items) contributions.push_back(contribution(item.element));
340 std::vector<int> sizes = trackSizes(grid, true, std::nullopt, element.number("column-gap"), contributions);
341 result = std::accumulate(sizes.begin(), sizes.end(), 0) + element.number("column-gap") * (grid.columns - 1);
342 } else {
343 std::vector<int> items = children(index);
344 bool row = element.get("display") == "flex" && (element.get("flex-direction") == "row" || element.get("flex-direction") == "row-reverse");
345 for (int child : items) result = row ? result + contribution(child) : std::max(result, contribution(child));
346 if (row && !items.empty()) result += element.number("column-gap") * (static_cast<int>(items.size()) - 1);
347 }
348 element.maxContent = result;
349 return result;
350 }

358 int bottom = 0, previousMargin = 0;
359 bool first = true;

377 int flex(int index, std::optional<int> reference, std::optional<int> heightReference) {
378 const Element& container = elements[index];
379 std::string direction = container.get("flex-direction");
380 bool horizontal = direction == "row" || direction == "row-reverse";
381 bool reverse = direction == "row-reverse" || direction == "column-reverse";
382 int gap = container.number(horizontal ? "column-gap" : "row-gap"), crossGap = container.number("row-gap");
383 std::optional<int> mainReference = horizontal ? std::optional<int>(container.cw) : reference;
384 struct Item {
385 int element, base, hypothetical, outer, cross = 0, size = 0;
386 std::string alignment;
387 };
388 std::vector<Item> items;
389 for (int child : children(index, true)) {
390 const Element& item = elements[child];
391 std::string alignment = align(index, child);
392 int cross = 0;
393 if (!horizontal) {
394 int room = std::max(0, container.cw - item.margins(true) - item.frame(true));
395 cross = item.get("width") != "auto" ? *resolve(child, "width", container.cw, true)
396 : (alignment == "stretch" ? room : std::min(maxContent(child), room));
397 cross = clamp(child, cross, container.cw, true);
398 }
399 std::optional<int> base = resolve(child, "flex-basis", mainReference, horizontal);
400 if (!base) base = resolve(child, horizontal ? "width" : "height", mainReference, horizontal);
401 if (!base) base = horizontal ? maxContent(child) : layout(child, container.cw, reference, cross);
402 items.push_back({child, *base, clamp(child, *base, mainReference, horizontal), item.frame(horizontal) + item.margins(horizontal), cross, 0, alignment});
403 }
404 int mainLength = horizontal ? container.cw : reference.value_or(0);
405 if (!horizontal && !reference) {
406 for (const Item& item : items) mainLength += item.hypothetical + item.outer;
407 if (!items.empty()) mainLength += gap * (static_cast<int>(items.size()) - 1);
408 mainLength = clamp(index, mainLength, heightReference, false);
409 }
410 std::vector<std::vector<std::size_t>> lines;
411 int used = 0;
412 for (std::size_t i = 0; i < items.size(); ++i) {
413 int outer = items[i].hypothetical + items[i].outer;
414 if (lines.empty() || (horizontal && container.get("flex-wrap") == "wrap" && used + gap + outer > mainLength)) {
415 lines.push_back({});
416 used = 0;
417 }
418 if (!lines.back().empty()) used += gap;
419 lines.back().push_back(i);
420 used += outer;
421 }
422 int crossStart = 0;
423 for (const auto& line : lines) {
424 int gaps = gap * (static_cast<int>(line.size()) - 1), hypothetical = gaps;
425 for (std::size_t i : line) hypothetical += items[i].hypothetical + items[i].outer;
426 bool growing = hypothetical < mainLength;
427 std::vector<bool> frozen(line.size(), false);
428 std::vector<int> weights(line.size(), 0), violations(line.size(), 0);
429 for (std::size_t j = 0; j < line.size(); ++j) {
430 Item& item = items[line[j]];
431 weights[j] = elements[item.element].number(growing ? "flex-grow" : "flex-shrink") * (growing ? 1 : item.base);
432 frozen[j] = weights[j] == 0 || (growing ? item.base > item.hypothetical : item.base < item.hypothetical);
433 item.size = item.hypothetical;
434 }
435 while (std::find(frozen.begin(), frozen.end(), false) != frozen.end()) {
436 int free = mainLength - gaps;
437 std::vector<int> activeWeights = weights;
438 for (std::size_t j = 0; j < line.size(); ++j) {
439 const Item& item = items[line[j]];
440 free -= (frozen[j] ? item.size : item.base) + item.outer;
441 if (frozen[j]) activeWeights[j] = 0;
442 }
443 std::vector<int> shares = distribute(std::max(0, growing ? free : -free), activeWeights);
444 int violation = 0;
445 for (std::size_t j = 0; j < line.size(); ++j) {
446 if (frozen[j]) continue;
447 Item& item = items[line[j]];
448 int tentative = item.base + (growing ? shares[j] : -shares[j]);
449 item.size = clamp(item.element, tentative, mainReference, horizontal);
450 violations[j] = item.size - tentative;
451 violation += violations[j];
452 }
453 for (std::size_t j = 0; j < line.size(); ++j) {
454 if (violation == 0 || (violation > 0 && violations[j] > 0) || (violation < 0 && violations[j] < 0)) frozen[j] = true;
455 }
456 }
457 int lineCross = horizontal ? 0 : container.cw;
458 for (std::size_t i : line) {
459 Item& item = items[i];
460 if (horizontal) {
461 layout(item.element, container.cw, reference, item.size);
462 item.cross = elements[item.element].ch;
463 lineCross = std::max(lineCross, item.cross + elements[item.element].frame(false) + elements[item.element].margins(false));
464 }
465 }
466 if (horizontal && lines.size() == 1) lineCross = reference ? *reference : clamp(index, lineCross, heightReference, false);
467 int remainder = mainLength - gaps;
468 for (std::size_t i : line) remainder -= items[i].size + items[i].outer;
469 std::vector<int> spaces(line.size() + 1, 0);
470 const std::string& justify = container.get("justify-content");
471 if (remainder > 0) {
472 if (justify == "end") spaces[0] = remainder;
473 else if (justify == "center") spaces[0] = remainder / 2;
474 else if (justify != "start") {
475 std::vector<int> weightsOfSpaces(spaces.size(), justify == "space-around" ? 2 : 1);
476 if (justify == "space-between") weightsOfSpaces.front() = weightsOfSpaces.back() = 0;
477 else if (justify == "space-around") weightsOfSpaces.front() = weightsOfSpaces.back() = 1;
478 spaces = distribute(remainder, weightsOfSpaces);
479 }
480 }
481 int cursor = 0;
482 for (std::size_t j = 0; j < line.size(); ++j) {
483 Item& item = items[line[j]];
484 Element& child = elements[item.element];
485 if (horizontal) {
486 std::optional<int> forcedHeight;
487 if (item.alignment == "stretch" && child.get("height") == "auto") forcedHeight = clamp(item.element, std::max(0, lineCross - child.margins(false) - child.frame(false)), reference, false);
488 layout(item.element, container.cw, reference, item.size, forcedHeight);
489 item.cross = child.ch;
490 } else layout(item.element, container.cw, reference, item.cross, item.size);
491 int cross = crossStart + alignmentOffset(lineCross - item.cross - child.frame(!horizontal) - child.margins(!horizontal), item.alignment)
492 + child.margin[horizontal ? 0 : 3];
493 cursor += spaces[j];
494 int startMargin = child.margin[horizontal ? (reverse ? 1 : 3) : (reverse ? 2 : 0)];
495 int main = reverse ? mainLength - cursor - startMargin - item.size - child.frame(horizontal) : cursor + startMargin;
496 positionChild(index, item.element, horizontal ? main : cross, horizontal ? cross : main);
497 cursor += item.size + item.outer + gap;
498 }
499 crossStart += lineCross + crossGap;
500 }
501 return horizontal ? (lines.empty() ? 0 : crossStart - crossGap) : mainLength;
502 }
503
504 int gridLayout(int index, std::optional<int> reference) {
505 Grid grid = placement(index);
506 const Element& container = elements[index];
507 int columnGap = container.number("column-gap"), rowGap = container.number("row-gap");
508 std::vector<int> contributions;
509 for (const Placement& item : grid.items) contributions.push_back(contribution(item.element));
510 std::vector<int> columns = trackSizes(grid, true, container.cw, columnGap, contributions), rowContributions;
511 for (Placement& item : grid.items) {
512 const Element& child = elements[item.element];
513 item.areaX = std::accumulate(columns.begin(), columns.begin() + item.column - 1, 0) + columnGap * (item.column - 1);
514 item.areaWidth = std::accumulate(columns.begin() + item.column - 1, columns.begin() + item.column + item.columnSpan - 1, 0) + columnGap * (item.columnSpan - 1);
515 int content = resolve(item.element, "width", item.areaWidth, true).value_or(std::max(0, item.areaWidth - child.margins(true) - child.frame(true)));
516 item.width = clamp(item.element, content, item.areaWidth, true);
517 layout(item.element, item.areaWidth, std::nullopt, item.width);
518 item.rowContribution = child.height() + child.margins(false);
519 rowContributions.push_back(item.rowContribution);
520 }
521 std::vector<int> rows = trackSizes(grid, false, reference, rowGap, rowContributions);
522 for (const Placement& item : grid.items) {
523 const Element& child = elements[item.element];
524 int areaY = std::accumulate(rows.begin(), rows.begin() + item.row - 1, 0) + rowGap * (item.row - 1);
525 int areaHeight = std::accumulate(rows.begin() + item.row - 1, rows.begin() + item.row + item.rowSpan - 1, 0) + rowGap * (item.rowSpan - 1);
526 std::string alignment = align(index, item.element);
527 std::optional<int> forcedHeight;
528 if (alignment == "stretch" && child.get("height") == "auto") forcedHeight = clamp(item.element, std::max(0, areaHeight - child.margins(false) - child.frame(false)), areaHeight, false);
529 layout(item.element, item.areaWidth, areaHeight, item.width, forcedHeight);
530 int offset = alignmentOffset(areaHeight - child.height() - child.margins(false), alignment);
531 positionChild(index, item.element, item.areaX + child.margin[3], areaY + offset + child.margin[0]);
532 }
533 return grid.rows == 0 ? 0 : std::accumulate(rows.begin(), rows.end(), 0) + rowGap * (grid.rows - 1);
534 }
535
536 int layout(int index, int widthReference, std::optional<int> heightReference, std::optional<int> forcedWidth = std::nullopt, std::optional<int> forcedHeight = std::nullopt) {
537 Element& element = elements[index];
538 element.cw = forcedWidth ? *forcedWidth : clamp(index, resolve(index, "width", widthReference, true).value_or(std::max(0, widthReference - element.margins(true) - element.frame(true))), widthReference, true);
539 if (!forcedHeight) {
540 std::optional<int> resolved = resolve(index, "height", heightReference, false);
541 if (resolved) forcedHeight = clamp(index, *resolved, heightReference, false);
542 }
543 int natural = 0;
544 element.lines.clear();
545 if (element.text) {
546 for (const std::string& word : split(*element.text, " ")) {
547 if (element.lines.empty() || static_cast<int>(element.lines.back().size() + 1 + word.size()) > element.cw) element.lines.push_back(word);
548 else element.lines.back() += ' ' + word;
549 }
550 natural = static_cast<int>(element.lines.size());
551 } else if (element.get("display") == "flex") natural = flex(index, forcedHeight, heightReference);
552 else if (element.get("display") == "grid") natural = gridLayout(index, forcedHeight);
553 else natural = block(index, forcedHeight);
554 element.naturalHeight = natural;
555 element.ch = forcedHeight ? *forcedHeight : clamp(index, natural, heightReference, false);
556 return natural;
557 }

559 int containingBlock(int index) const {
560 for (int parent = elements[index].parent; parent >= 0; parent = elements[parent].parent) {
561 if (elements[parent].get("position") != "static") return parent;
562 }
563 return -1;
564 }

566 void recompute() {
567 const char* sides[] = {"top", "right", "bottom", "left"};
568 for (std::size_t i = 0; i < elements.size(); ++i) {
569 Element& element = elements[i];
570 element.style = initialValues();
571 std::map<std::string, int> specificities;
572 for (const StyleRule& rule : rules) {
573 int specificity = -1;
574 for (const std::string& selector : rule.selectors) {
575 if (selector[0] == '#' && selector.substr(1) == element.id) specificity = 2;
576 else if (selector[0] == '.' && element.classes.count(selector.substr(1))) specificity = std::max(specificity, 1);
577 else if (selector == element.type) specificity = std::max(specificity, 0);
578 }
579 if (specificity < 0) continue;
580 for (const auto& [property, value] : rule.declarations) {
581 auto previous = specificities.find(property);
582 if (previous == specificities.end() || specificity >= previous->second) {
583 element.style[property] = value;
584 specificities[property] = specificity;
585 }
586 }
587 }
588 for (const auto& [property, value] : element.overrides) element.style[property] = value;
589 for (int side = 0; side < 4; ++side) {
590 element.margin[side] = element.number(std::string("margin-") + sides[side]);
591 element.padding[side] = element.number(std::string("padding-") + sides[side]);
592 element.border[side] = element.number(std::string("border-") + sides[side]);
593 }
594 element.displayed = element.get("display") != "none" && (element.parent < 0 || elements[element.parent].displayed);
595 element.maxContent.reset();
596 element.contribution.reset();
597 }
598 if (elements[0].displayed && elements[0].get("position") != "absolute") {
599 layout(0, width, height);
600 elements[0].relativeX = elements[0].margin[3];
601 elements[0].relativeY = elements[0].margin[0];
602 }
603 for (std::size_t i = 0; i < elements.size(); ++i) {
604 Element& element = elements[i];
605 if (!element.displayed || element.get("position") != "absolute") continue;
606 int block = containingBlock(static_cast<int>(i));
607 int blockWidth = block < 0 ? width : elements[block].cw + elements[block].padding[1] + elements[block].padding[3];
608 int blockHeight = block < 0 ? height : elements[block].ch + elements[block].padding[0] + elements[block].padding[2];
609 auto inset = [&](const std::string& property) -> std::optional<int> { return element.get(property) == "auto" ? std::nullopt : std::optional<int>(element.number(property)); };
610 std::optional<int> left = inset("left"), right = inset("right"), top = inset("top"), bottom = inset("bottom");
611 int contentWidth;
612 if (element.get("width") != "auto") contentWidth = *resolve(static_cast<int>(i), "width", blockWidth, true);
613 else {
614 int room = std::max(0, blockWidth - left.value_or(0) - right.value_or(0) - element.margins(true) - element.frame(true));
615 contentWidth = left && right ? room : std::min(maxContent(static_cast<int>(i)), room);
616 }
617 contentWidth = clamp(static_cast<int>(i), contentWidth, blockWidth, true);
618 std::optional<int> forcedHeight;
619 if (element.get("height") == "auto" && top && bottom) forcedHeight = clamp(static_cast<int>(i), std::max(0, blockHeight - *top - *bottom - element.margins(false) - element.frame(false)), blockHeight, false);
620 layout(static_cast<int>(i), blockWidth, blockHeight, contentWidth, forcedHeight);
621 element.relativeX = left ? *left + element.margin[3] : (right ? blockWidth - *right - element.margin[1] - element.width() : element.margin[3]);
622 element.relativeY = top ? *top + element.margin[0] : (bottom ? blockHeight - *bottom - element.margin[2] - element.height() : element.margin[0]);
623 }
624 for (std::size_t i = 0; i < elements.size(); ++i) {
625 Element& element = elements[i];
626 if (!element.displayed) continue;
627 int parent = element.get("position") == "absolute" ? containingBlock(static_cast<int>(i)) : element.parent;
628 element.x = parent < 0 ? 0 : elements[parent].x;
629 element.y = parent < 0 ? 0 : elements[parent].y;
630 if (element.get("position") == "absolute" && parent >= 0) {
631 element.x += elements[parent].border[3];
632 element.y += elements[parent].border[0];
633 }
634 element.x += element.relativeX;
635 element.y += element.relativeY;
636 if (element.get("position") == "relative") {
637 element.x += element.get("left") != "auto" ? element.number("left") : (element.get("right") != "auto" ? -element.number("right") : 0);
638 element.y += element.get("top") != "auto" ? element.number("top") : (element.get("bottom") != "auto" ? -element.number("bottom") : 0);
639 }
640 }
641 }
642
643 Rect rectangle(int index, const std::string& box) const {
644 const Element& element = elements[index];
645 Rect rect{element.x, element.y, element.width(), element.height()};
646 if (box == "margin") {
647 rect.x -= element.margin[3];
648 rect.y -= element.margin[0];
649 rect.width += element.margins(true);
650 rect.height += element.margins(false);
651 } else if (box == "padding" || box == "content") {
652 rect.x += element.border[3];
653 rect.y += element.border[0];
654 rect.width -= element.border[1] + element.border[3];
655 rect.height -= element.border[0] + element.border[2];
656 if (box == "content") {
657 rect.x += element.padding[3];
658 rect.y += element.padding[0];
659 rect.width = element.cw;
660 rect.height = element.ch;
661 }
662 }
663 return rect;
664 }

666 using Snapshot = std::vector<std::optional<std::array<int, 4>>>;
667 Snapshot snapshot() const {
668 Snapshot result;
669 for (const Element& element : elements) {
670 if (element.displayed) result.push_back(std::array<int, 4>{element.x, element.y, element.width(), element.height()});
671 else result.push_back(std::nullopt);
672 }
673 return result;
674 }
675
676 std::string changed(const Snapshot& before) const {
677 Snapshot after = snapshot();
678 std::vector<std::string> names;
679 for (std::size_t i = 0; i < before.size(); ++i) {
680 if (before[i] != after[i]) names.push_back(elements[i].id);
681 }
682 return names.empty() ? "no change" : join(names);
683 }
684
685 void render() const {
686 std::vector<std::string> screen(height, std::string(width, '.'));
687 for (std::size_t i = 0; i < elements.size(); ++i) {
688 const Element& element = elements[i];
689 if (!element.displayed) continue;
690 Rect border = rectangle(static_cast<int>(i), "border"), content = rectangle(static_cast<int>(i), "content");
691 for (int y = std::max(0, border.y); y < std::min(height, border.y + border.height); ++y) {
692 for (int x = std::max(0, border.x); x < std::min(width, border.x + border.width); ++x) {
693 bool horizontal = (element.border[0] && y == border.y) || (element.border[2] && y == border.y + border.height - 1);
694 bool vertical = (element.border[3] && x == border.x) || (element.border[1] && x == border.x + border.width - 1);
695 screen[y][x] = horizontal ? (vertical ? '+' : '-') : (vertical ? '|' : ' ');
696 }
697 }
698 for (std::size_t row = 0; row < element.lines.size(); ++row) {
699 int y = content.y + static_cast<int>(row);
700 if (y < 0 || y >= height) continue;
701 for (std::size_t column = 0; column < element.lines[row].size(); ++column) {
702 int x = content.x + static_cast<int>(column);
703 if (x >= 0 && x < width) screen[y][x] = element.lines[row][column];
704 }
705 }
706 }
707 std::string edge(width + 2, '#');
708 std::cout << "render " << width << 'x' << height << '\n' << edge << '\n';
709 for (const std::string& row : screen) std::cout << '#' << row << "#\n";
710 std::cout << edge << '\n';
711 }
712
713 void execute(const std::string& line) {
714 std::vector<std::string> parts = split(line, " ");
715 const std::string& command = parts[0];
716 if (command == "LAYOUT") {
717 std::cout << "layout " << width << 'x' << height << '\n';
718 for (const Element& element : elements) {
719 std::cout << std::string(2 * element.depth, ' ') << element.id;
720 if (!element.displayed) std::cout << " none\n";
721 else std::cout << ' ' << element.x << ' ' << element.y << ' ' << element.width() << ' ' << element.height() << '\n';
722 }
723 } else if (command == "BOX") {
724 auto found = ids.find(parts[1]);
725 if (found == ids.end()) {
726 std::cout << "error: unknown element " << parts[1] << '\n';
727 return;
728 }
729 const Element& element = elements[found->second];
730 if (!element.displayed) std::cout << "box " << parts[1] << ": none\n";
731 else {
732 std::cout << "box " << parts[1] << '\n';
733 for (const std::string& box : {"margin", "border", "padding", "content"}) {
734 Rect rect = rectangle(found->second, box);
735 std::cout << box << ' ' << rect.x << ' ' << rect.y << ' ' << rect.width << ' ' << rect.height << '\n';
736 }
737 }
738 } else if (command == "RENDER") render();
739 else if (command == "HIT") {
740 int x = std::stoi(parts[1]), y = std::stoi(parts[2]);
741 if (x >= width || y >= height) {
742 std::cout << "error: point " << x << ' ' << y << " outside viewport\n";
743 return;
744 }
745 int found = -1;
746 for (std::size_t i = 0; i < elements.size(); ++i) {
747 if (elements[i].displayed && rectangle(static_cast<int>(i), "border").contains(x, y)) found = static_cast<int>(i);
748 }
749 std::vector<std::string> path;
750 for (int i = found; i >= 0; i = elements[i].parent) path.push_back(elements[i].id);
751 std::reverse(path.begin(), path.end());
752 std::cout << "hit " << x << ' ' << y << ": " << (path.empty() ? "none" : join(path, " > ")) << '\n';
753 } else if (command == "SET") {
754 auto found = ids.find(parts[1]);
755 if (found == ids.end()) {
756 std::cout << "error: unknown element " << parts[1] << '\n';
757 return;
758 }
759 if (!known(parts[2])) {
760 std::cout << "error: unknown property " << parts[2] << '\n';
761 return;
762 }
763 std::string value = join(std::vector<std::string>(parts.begin() + 3, parts.end()));
764 std::optional<Declarations> declarations = declaration(parts[2], value);
765 if (!declarations) {
766 std::cout << "error: invalid value for " << parts[2] << ": " << value << '\n';
767 return;
768 }
769 Snapshot before = snapshot();
770 for (const auto& [property, text] : *declarations) elements[found->second].overrides[property] = text;
771 recompute();
772 std::cout << "set " << parts[1] << ' ' << parts[2] << ' ' << value << ": " << changed(before) << '\n';
773 } else if (command == "VIEWPORT") {
774 int w = std::stoi(parts[1]), h = std::stoi(parts[2]);
775 if (w < 1 || w > 80 || h < 1 || h > 40) {
776 std::cout << "error: invalid viewport " << w << ' ' << h << '\n';
777 return;
778 }
779 Snapshot before = snapshot();
780 width = w;
781 height = h;
782 recompute();
783 std::cout << "viewport " << w << ' ' << h << ": " << changed(before) << '\n';
784 } else if (command == "OVERFLOW") {
785 std::vector<std::string> names;
786 for (std::size_t i = 0; i < elements.size(); ++i) {
787 const Element& element = elements[i];
788 if (!element.displayed) continue;
789 bool overflowing = false;
790 if (element.text) {
791 overflowing = static_cast<int>(element.lines.size()) > element.ch;
792 for (const std::string& text : element.lines) overflowing = overflowing || static_cast<int>(text.size()) > element.cw;
793 } else {
794 Rect content = rectangle(static_cast<int>(i), "content");
795 for (int child : children(static_cast<int>(i))) {
796 Rect margin = rectangle(child, "margin");
797 if (margin.x < content.x || margin.y < content.y || margin.x + margin.width > content.x + content.width || margin.y + margin.height > content.y + content.height) overflowing = true;
798 }
799 }
800 if (overflowing) names.push_back(element.id);
801 }
802 std::cout << "overflow: " << (names.empty() ? "none" : join(names)) << '\n';
803 } else std::cout << "error: unknown command " << command << '\n';
804 }

806public:
807 void run() {
808 std::cin >> width >> height;
809 std::string line;
810 std::getline(std::cin, line);
811 while (std::getline(std::cin, line) && line != "---") {
812 std::size_t brace = line.find('{');
813 StyleRule rule{split(trim(line.substr(0, brace)), ", "), {}};
814 for (const std::string& text : split(trim(line.substr(brace + 1, line.size() - brace - 2)), "; ")) {
815 std::size_t colon = text.find(": ");
816 std::optional<Declarations> declarations = declaration(text.substr(0, colon), text.substr(colon + 2));
817 if (declarations) rule.declarations.insert(rule.declarations.end(), declarations->begin(), declarations->end());
818 }
819 rules.push_back(std::move(rule));
820 }
821 std::vector<int> ancestors;
822 while (std::getline(std::cin, line) && line != "===") {
823 std::size_t first = line.find_first_not_of(' '), space = line.find(' ', first);
824 std::string head = line.substr(first, space == std::string::npos ? space : space - first);
825 std::size_t hash = head.find('#');
826 std::vector<std::string> names = split(head.substr(hash + 1), ".");
827 Element element;
828 element.type = head.substr(0, hash);
829 element.id = names[0];
830 element.depth = static_cast<int>(first / 2);
831 element.classes.insert(names.begin() + 1, names.end());
832 if (space != std::string::npos) element.text = line.substr(space + 2, line.size() - space - 3);
833 if (element.depth > 0) element.parent = ancestors[element.depth - 1];
834 int index = static_cast<int>(elements.size());
835 if (element.parent >= 0) elements[element.parent].children.push_back(index);
836 ids[element.id] = index;
837 elements.push_back(std::move(element));
838 ancestors.resize(elements.back().depth + 1);
839 ancestors.back() = index;
840 }
841 recompute();
842 while (std::getline(std::cin, line) && line != "END") execute(line);
843 }
