139function distribute(total, weights) {
140 const sum = weights.reduce((a, b) => a + b, 0);
141 if (sum === 0) return weights.map(() => 0);
142 const result = weights.map((weight) => Math.floor((total * weight) / sum));
143 let remainder = total - result.reduce((a, b) => a + b, 0);
144 for (let i = 0; i < weights.length && remainder > 0; i++)
145 if (weights[i] > 0) {
146 result[i]++;
147 remainder--;
148 }
149 return result;
