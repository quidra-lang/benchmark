23bool uint(String text) => text.isNotEmpty && text.length <= 3 && !RegExp(r'[^0-9]').hasMatch(text);
24bool positive(String text) => uint(text) && int.parse(text) > 0;
25bool signed(String text) => uint(text.startsWith('-') ? text.substring(1) : text);
26bool percentage(String text) => text.endsWith('%') && uint(text.substring(0, text.length - 1));
27bool fraction(String text) => text.endsWith('fr') && positive(text.substring(0, text.length - 2));
28bool gridValue(String value) {
29 final words = value.split(' ');
30 return value == 'auto' || words.length == 1 && positive(words[0]) ||
31 words.length == 2 && words[0] == 'span' && positive(words[1]) ||
32 words.length == 3 && positive(words[0]) && words[1] == '/' && positive(words[2]) && int.parse(words[0]) < int.parse(words[2]) ||
33 words.length == 4 && positive(words[0]) && words[1] == '/' && words[2] == 'span' && positive(words[3]);
34}
35List<Declaration>? declarations(String property, String value) {
36 if (!known.contains(property)) return null;
37 final words = value.split(' ');
38 if (['margin', 'padding', 'border'].contains(property)) {
39 if (words.isEmpty || words.length > 4 || !words.every((word) => property == 'border' ? word == '0' || word == '1' : uint(word))) return null;
40 final values = words.length == 1 ? [words[0], words[0], words[0], words[0]]
41 : words.length == 2 ? [words[0], words[1], words[0], words[1]]
42 : words.length == 3 ? [words[0], words[1], words[2], words[1]] : words;
43 return [for (var i = 0; i < 4; i++) ('$property-${sides[i]}', values[i])];
44 }
45 if (property == 'gap') return words.length <= 2 && words.every(uint)
46 ? [('row-gap', words[0]), ('column-gap', words.length == 1 ? words[0] : words[1])] : null;
47 bool valid;
48 if (property.startsWith('grid-template-')) { valid = value == 'none' || words.every((word) => uint(word) || fraction(word) || word == 'auto'); }
49 else if (property == 'grid-column' || property == 'grid-row') { valid = gridValue(value); }
50 else if (words.length != 1) { valid = false; }
51 else if (enumerations.containsKey(property)) { valid = enumerations[property]!.contains(value); }
52 else if (property == 'width' || property == 'height' || property == 'flex-basis') { valid = value == 'auto' || uint(value) || percentage(value); }
53 else if (property.startsWith('min-')) { valid = uint(value) || percentage(value); }
54 else if (property.startsWith('max-')) { valid = value == 'none' || uint(value) || percentage(value); }
55 else if (property.startsWith('border-')) { valid = value == '0' || value == '1'; }
56 else if (sides.contains(property)) { valid = value == 'auto' || signed(value); }
57 else if (property == 'order') { valid = signed(value); }
58 else { valid = uint(value); }
59 return valid ? [(property, value)] : null;
60}
61int sum(Iterable<int> values) => values.fold<int>(0, (a, b) => a + b);
62int maximumValue(Iterable<int> values) => values.fold<int>(0, math.max);
63List<int> distribute(int total, List<int> weights) {
64 final denominator = sum(weights);
65 if (denominator == 0) return List<int>.filled(weights.length, 0);
66 final result = weights.map((weight) => total * weight ~/ denominator).toList();
67 var remainder = total - sum(result);
68 for (var i = 0; i < weights.length && remainder > 0; i++) {
69 if (weights[i] > 0) { result[i]++; remainder--; }
70 }
71 return result;
72}

158Grid placement(Element element) {
159 final columns = element.value('grid-template-columns') == 'none' ? ['auto'] : element.value('grid-template-columns').split(' ');
160 final rows = element.value('grid-template-rows') == 'none' ? <String>[] : element.value('grid-template-rows').split(' ');
161 (int?, int) axis(String value) {
162 final words = value.split(' ');
163 if (value == 'auto') return (null, 1);
164 if (words[0] == 'span') return (null, int.parse(words[1]));
165 return (int.parse(words[0]), words.length == 1 ? 1 : words.length == 3 ? int.parse(words[2]) - int.parse(words[0]) : int.parse(words[3]));
166 }
167 final items = element.items().map((node) {
168 final (column, cs) = axis(node.value('grid-column'));
169 final (row, rs) = axis(node.value('grid-row'));
170 return GridItem(node, column, row, cs, rs);
171 }).toList();
172 final nc = math.max(columns.length, maximumValue(items.map((item) => item.column == null ? item.cs : item.column! + item.cs - 1)));
173 final occupied = <(int, int)>{};
174 bool overlaps(GridItem item, int r, int c) {
175 for (var y = r; y < r + item.rs; y++) {
176 for (var x = c; x < c + item.cs; x++) { if (occupied.contains((y, x))) return true; }
177 }
178 return false;
179 }
180 void put(GridItem item, int r, int c) {
181 item.row = r;
182 item.column = c;
183 for (var y = r; y < r + item.rs; y++) {
184 for (var x = c; x < c + item.cs; x++) { occupied.add((y, x)); }
185 }
186 }
187 final pending = items.where((item) => item.row == null || item.column == null).toList();
188 for (final item in items) { if (item.row != null && item.column != null) put(item, item.row!, item.column!); }
189 var cr = 1, cc = 1;
190 for (final item in pending) {
191 if (item.row != null) {
192 var column = 1;
193 while (column <= nc - item.cs + 1 && overlaps(item, item.row!, column)) { column++; }
194 put(item, item.row!, column <= nc - item.cs + 1 ? column : 1);
195 } else if (item.column != null) {
196 if (item.column! < cc) cr++;
197 var row = cr;
198 while (overlaps(item, row, item.column!)) { row++; }
199 put(item, row, item.column!);
200 cr = row;
201 cc = item.column! + item.cs;
202 } else {
203 var row = cr, column = cc;
204 while (true) {
205 if (column + item.cs - 1 > nc) { row++; column = 1; }
206 else if (overlaps(item, row, column)) { column++; }
207 else { break; }
208 }
209 put(item, row, column);
210 cr = row;
211 cc = column + item.cs;
212 }
213 }
214 return Grid(columns, rows, items, nc, math.max(rows.length, maximumValue(items.map((item) => item.row! + item.rs - 1))));

216List<int> trackSizes(List<String> explicit, int count, int? available, int gap, List<GridItem> items, bool horizontal, List<int> contributions) {
217 final tracks = [...explicit, ...List<String>.filled(math.max(0, count - explicit.length), 'auto')];
218 final sizes = <int>[];
219 for (var i = 0; i < tracks.length; i++) {
220 final track = tracks[i];
221 if (uint(track)) { sizes.add(int.parse(track)); }
222 else if (fraction(track) && available != null) { sizes.add(0); }
223 else {
224 var size = 0;
225 for (var j = 0; j < items.length; j++) {
226 final item = items[j];
227 if ((horizontal ? item.column : item.row) == i + 1 && (horizontal ? item.cs : item.rs) == 1) size = math.max(size, contributions[j]);
228 }
229 sizes.add(size);
230 }
231 }
232 if (available != null) {
233 final remaining = available - sum(sizes) - gap * math.max(0, count - 1), hasFr = tracks.any(fraction);
234 if (remaining > 0) {
235 final weights = tracks.map((track) => hasFr ? fraction(track) ? int.parse(track.substring(0, track.length - 2)) : 0 : track == 'auto' ? 1 : 0).toList();
236 final shares = distribute(remaining, weights);
237 for (var i = 0; i < sizes.length; i++) { sizes[i] += shares[i]; }
238 }
239 }
240 return sizes;
241}
242int maximum(Element element) {
243 final cached = maximumCache[element];
244 if (cached != null) return cached;
245 int result;
246 if (element.text != null) { result = element.text!.length; }
247 else if (element.value('display') == 'grid') {
248 final grid = placement(element), gap = element.number('column-gap');
249 result = sum(trackSizes(grid.columns, grid.nc, null, gap, grid.items, true, grid.items.map((item) => contribution(item.node)).toList())) + gap * (grid.nc - 1);
250 } else {
251 final children = element.flow();
252 result = element.value('display') == 'flex' && element.value('flex-direction').startsWith('row')
253 ? sum(children.map(contribution)) + element.number('column-gap') * math.max(0, children.length - 1)
254 : maximumValue(children.map(contribution));
255 }
256 maximumCache[element] = result;
257 return result;
258}
259int layout(Element element, int widthReference, int? heightReference, [int? forcedWidth, int? forcedHeight]) {
260 element.cw = forcedWidth ?? element.clamp(element.resolve('width', widthReference, true) ?? math.max(0, widthReference - element.margin(true) - element.frame(true)), true, widthReference);
261 var fixed = forcedHeight;
262 if (fixed == null) {
263 final resolved = element.resolve('height', heightReference, false);
264 if (resolved != null) fixed = element.clamp(resolved, false, heightReference);
265 }
266 var content = 0;
267 element.lines = [];
268 if (element.text != null) {
269 for (final word in element.text!.split(' ')) {
270 if (element.lines.isEmpty || element.lines.last.length + 1 + word.length > element.cw) { element.lines.add(word); }
271 else { element.lines[element.lines.length - 1] += ' $word'; }
272 }
273 content = element.lines.length;
274 } else if (element.value('display') == 'flex') { content = flex(element, fixed, heightReference); }
275 else if (element.value('display') == 'grid') { content = gridLayout(element, fixed); }
276 else {
277 var bottom = 0, previousMargin = 0;
278 for (final child in element.flow()) {
279 layout(child, element.cw, fixed);
280 final top = bottom + math.max(previousMargin, child.side('margin', 'top'));
281 element.place(child, child.side('margin', 'left'), top);
282 bottom = top + child.borderHeight();
283 previousMargin = child.side('margin', 'bottom');
284 }
285 content = bottom + previousMargin;
286 }
287 element.ch = fixed ?? element.clamp(content, false, heightReference);
288 return content;

290int flex(Element element, int? k, int? heightReference) {
291 final horizontal = element.value('flex-direction').startsWith('row'), reverse = element.value('flex-direction').endsWith('reverse');
292 final reference = horizontal ? element.cw : k, gap = element.number(horizontal ? 'column-gap' : 'row-gap'), lineGap = element.number('row-gap');
293 final items = <FlexItem>[];
294 for (final node in element.items()) {
295 final alignment = align(element, node);
296 var cross = 0;
297 if (!horizontal) {
298 final available = math.max(0, element.cw - node.margin(true) - node.frame(true));
299 cross = node.clamp(node.value('width') != 'auto' ? node.resolve('width', element.cw, true)!
300 : alignment == 'stretch' ? available : math.min(maximum(node), available), true, element.cw);
301 }
302 final base = node.resolve('flex-basis', reference, horizontal) ?? node.resolve(horizontal ? 'width' : 'height', reference, horizontal)
303 ?? (horizontal ? maximum(node) : layout(node, element.cw, k, cross));
304 final hypothetical = node.clamp(base, horizontal, reference), outer = node.frame(horizontal) + node.margin(horizontal);
305 items.add(FlexItem(node, alignment, cross, base, hypothetical, outer));
306 }
307 final main = horizontal ? element.cw : k ?? element.clamp(sum(items.map((item) => item.hypothetical + item.outer)) + gap * math.max(0, items.length - 1), false, heightReference);
308 final lines = <List<FlexItem>>[];
309 for (final item in items) {
310 if (lines.isEmpty || horizontal && element.value('flex-wrap') == 'wrap' &&
311 sum(lines.last.map((other) => other.hypothetical + other.outer)) + gap * lines.last.length + item.hypothetical + item.outer > main) {
312 lines.add([item]);
313 } else { lines.last.add(item); }
314 }
315 var crossStart = 0;
316 for (final line in lines) {
317 final growing = sum(line.map((item) => item.hypothetical + item.outer)) + gap * (line.length - 1) < main;
318 for (final item in line) {
319 item.weight = growing ? item.node.number('flex-grow') : item.node.number('flex-shrink') * item.base;
320 item.frozen = item.weight == 0 || (growing ? item.base > item.hypothetical : item.base < item.hypothetical);
321 }
322 while (line.any((item) => !item.frozen)) {
323 final free = main - gap * (line.length - 1) - sum(line.map((item) => (item.frozen ? item.target : item.base) + item.outer));
324 final shares = distribute(math.max(0, growing ? free : -free), line.map((item) => item.frozen ? 0 : item.weight).toList());
325 final violations = List<int>.filled(line.length, 0);
326 for (var i = 0; i < line.length; i++) {
327 final item = line[i];
328 if (item.frozen) continue;
329 final tentative = item.base + (growing ? shares[i] : -shares[i]);
330 item.target = item.node.clamp(tentative, horizontal, reference);
331 violations[i] = item.target - tentative;
332 }
333 final total = sum(violations);
334 for (var i = 0; i < line.length; i++) {
335 if (total == 0 || (total > 0 ? violations[i] > 0 : violations[i] < 0)) line[i].frozen = true;
336 }
337 }
338 if (horizontal) {
339 for (final item in line) { layout(item.node, element.cw, k, item.target); item.cross = item.node.ch; }
340 }
341 final outerCross = maximumValue(line.map((item) => item.cross + item.node.frame(!horizontal) + item.node.margin(!horizontal)));
342 final lineCross = horizontal ? lines.length == 1 ? k ?? element.clamp(outerCross, false, heightReference) : outerCross : element.cw;
343 final remaining = main - sum(line.map((item) => item.target + item.outer)) - gap * (line.length - 1);
344 final spaces = List<int>.filled(line.length, 0), justification = element.value('justify-content');
345 if (remaining > 0) {
346 if (justification == 'end') { spaces[0] = remaining; }
347 else if (justification == 'center') { spaces[0] = remaining ~/ 2; }
348 else if (justification.startsWith('space-')) {
349 final weights = List<int>.generate(line.length + 1, (i) => justification == 'space-evenly' ? 1
350 : i == 0 || i == line.length ? justification == 'space-between' ? 0 : 1 : justification == 'space-between' ? 1 : 2);
351 final shares = distribute(remaining, weights);
352 for (var i = 0; i < spaces.length; i++) { spaces[i] = shares[i]; }
353 }
354 }
355 var cursor = 0;
356 for (var i = 0; i < line.length; i++) {
357 final item = line[i], node = line[i].node;
358 if (horizontal && item.alignment == 'stretch' && node.value('height') == 'auto') {
359 item.cross = node.clamp(math.max(0, lineCross - node.margin(false) - node.frame(false)), false, k);
360 layout(node, element.cw, k, item.target, item.cross);
361 } else if (horizontal) { layout(node, element.cw, k, item.target); item.cross = node.ch; }
362 else { layout(node, element.cw, k, item.cross, item.target); }
363 final spare = lineCross - item.cross - node.frame(!horizontal) - node.margin(!horizontal);
364 final offset = spare > 0 ? item.alignment == 'end' ? spare : item.alignment == 'center' ? spare ~/ 2 : 0 : 0;
365 cursor += spaces[i];
366 final startMargin = node.side('margin', horizontal ? reverse ? 'right' : 'left' : reverse ? 'bottom' : 'top');
367 final mainPosition = reverse ? main - cursor - startMargin - (horizontal ? node.borderWidth() : node.borderHeight()) : cursor + startMargin;
368 final crossPosition = crossStart + offset + node.side('margin', horizontal ? 'top' : 'left');
369 element.place(node, horizontal ? mainPosition : crossPosition, horizontal ? crossPosition : mainPosition);
370 cursor += item.target + item.outer + gap;
371 }
372 crossStart += lineCross + lineGap;
373 }
374 return horizontal ? lines.isEmpty ? 0 : crossStart - lineGap : main;
375}
376(int, int) area(List<int> sizes, int start, int span, int gap) =>
377 (sum(sizes.skip(start - 1).take(span)) + gap * (span - 1), sum(sizes.take(start - 1)) + gap * (start - 1));
378int gridLayout(Element element, int? k) {
379 final grid = placement(element), columnGap = element.number('column-gap'), rowGap = element.number('row-gap');
380 final columns = trackSizes(grid.columns, grid.nc, element.cw, columnGap, grid.items, true, grid.items.map((item) => contribution(item.node)).toList());
381 for (final item in grid.items) {
382 final node = item.node;
383 final (aw, ax) = area(columns, item.column!, item.cs, columnGap);
384 item.aw = aw;
385 item.ax = ax;
386 item.gw = node.clamp(node.resolve('width', aw, true) ?? math.max(0, aw - node.margin(true) - node.frame(true)), true, aw);
387 layout(node, aw, null, item.gw);
388 item.contribution = node.borderHeight() + node.margin(false);
389 }
390 final rows = trackSizes(grid.rows, grid.nr, k, rowGap, grid.items, false, grid.items.map((item) => item.contribution).toList());
391 for (final item in grid.items) {
392 final node = item.node, alignment = align(element, item.node);
393 final (ah, ay) = area(rows, item.row!, item.rs, rowGap);
394 if (alignment == 'stretch' && node.value('height') == 'auto') {
395 layout(node, item.aw, ah, item.gw, node.clamp(math.max(0, ah - node.margin(false) - node.frame(false)), false, ah));
396 } else { layout(node, item.aw, ah, item.gw); }
397 final spare = ah - node.borderHeight() - node.margin(false);
398 final offset = spare > 0 ? alignment == 'end' ? spare : alignment == 'center' ? spare ~/ 2 : 0 : 0;
399 element.place(node, item.ax + node.side('margin', 'left'), ay + offset + node.side('margin', 'top'));
400 }
401 return sum(rows) + rowGap * math.max(0, grid.nr - 1);
402}
403void relayout() {
404 maximumCache.clear();
405 for (final node in elements) {
406 node.style = Map.of(initial);
407 final ranks = <String, int>{};
408 for (final rule in rules) {
409 final matching = rule.selectors.where((selector) => selector[0] == '#' ? selector.substring(1) == node.id
410 : selector[0] == '.' ? node.classes.contains(selector.substring(1)) : selector == node.type).toList();
411 if (matching.isEmpty) continue;
412 final rank = maximumValue(matching.map((selector) => selector[0] == '#' ? 2 : selector[0] == '.' ? 1 : 0));
413 for (final (property, value) in rule.declarations) {
414 if (!ranks.containsKey(property) || rank >= ranks[property]!) { node.style[property] = value; ranks[property] = rank; }
415 }
416 }
417 node.style.addAll(node.overrides);
418 node.shown = node.value('display') != 'none' && (node.parent?.shown ?? true);
419 node.containing = null;
420 }
421 final root = elements.first;
422 if (root.shown && root.value('position') != 'absolute') {
423 layout(root, width, height);
424 root.rx = root.side('margin', 'left');
425 root.ry = root.side('margin', 'top');
426 }
427 for (final node in elements) {
428 if (!node.shown || node.value('position') != 'absolute') continue;
429 var containing = node.parent;
430 while (containing != null && containing.value('position') == 'static') { containing = containing.parent; }
431 node.containing = containing;
432 final cbw = containing == null ? width : containing.cw + containing.side('padding', 'left') + containing.side('padding', 'right');
433 final cbh = containing == null ? height : containing.ch + containing.side('padding', 'top') + containing.side('padding', 'bottom');
434 final left = node.value('left') == 'auto' ? null : node.number('left'), right = node.value('right') == 'auto' ? null : node.number('right');
435 final top = node.value('top') == 'auto' ? null : node.number('top'), bottom = node.value('bottom') == 'auto' ? null : node.number('bottom');
436 final available = math.max(0, cbw - (left ?? 0) - (right ?? 0) - node.margin(true) - node.frame(true));
437 final fw = node.clamp(node.value('width') != 'auto' ? node.resolve('width', cbw, true)!
438 : left != null && right != null ? available : math.min(maximum(node), available), true, cbw);
439 final fh = node.value('height') == 'auto' && top != null && bottom != null
440 ? node.clamp(math.max(0, cbh - top - bottom - node.margin(false) - node.frame(false)), false, cbh) : null;
441 layout(node, cbw, cbh, fw, fh);
442 node.rx = left != null ? left + node.side('margin', 'left') : right != null ? cbw - right - node.side('margin', 'right') - node.borderWidth() : node.side('margin', 'left');
443 node.ry = top != null ? top + node.side('margin', 'top') : bottom != null ? cbh - bottom - node.side('margin', 'bottom') - node.borderHeight() : node.side('margin', 'top');
444 }
445 for (final node in elements) {
446 if (!node.shown) continue;
447 final absolute = node.value('position') == 'absolute', reference = node.value('position') == 'absolute' ? node.containing : node.parent;
448 node.x = (reference == null ? 0 : reference.x + (absolute ? reference.side('border', 'left') : 0)) + node.rx;
449 node.y = (reference == null ? 0 : reference.y + (absolute ? reference.side('border', 'top') : 0)) + node.ry;
450 if (node.value('position') == 'relative') {
451 node.x += node.value('left') != 'auto' ? node.number('left') : node.value('right') != 'auto' ? -node.number('right') : 0;
452 node.y += node.value('top') != 'auto' ? node.number('top') : node.value('bottom') != 'auto' ? -node.number('bottom') : 0;
453 }
454 }
455}
456Boxes boxes(Element node) {
457 final border = Rect(node.x, node.y, node.borderWidth(), node.borderHeight());
458 final margin = Rect(node.x - node.side('margin', 'left'), node.y - node.side('margin', 'top'), border.w + node.margin(true), border.h + node.margin(false));
459 final padding = Rect(node.x + node.side('border', 'left'), node.y + node.side('border', 'top'),
460 border.w - node.side('border', 'left') - node.side('border', 'right'), border.h - node.side('border', 'top') - node.side('border', 'bottom'));
461 return Boxes(margin, border, padding, Rect(padding.x + node.side('padding', 'left'), padding.y + node.side('padding', 'top'), node.cw, node.ch));
462}
463List<String> snapshot() => elements.map((node) => node.shown ? boxes(node).border.text : 'none').toList();
464String changed(List<String> before) {
465 final names = <String>[];
466 for (var i = 0; i < elements.length; i++) {
467 final node = elements[i];
468 if (before[i] != (node.shown ? boxes(node).border.text : 'none')) names.add(node.id);
469 }
470 return names.isEmpty ? 'no change' : names.join(' ');
471}
472void render() {
473 final screen = List.generate(height, (_) => List<String>.filled(width, '.'));
474 void paint(int x, int y, String c) { if (x >= 0 && x < width && y >= 0 && y < height) screen[y][x] = c; }
475 for (final node in elements) {
476 if (!node.shown) continue;
477 final bw = node.borderWidth(), bh = node.borderHeight();
478 for (var y = math.max(0, node.y); y < math.min(height, node.y + bh); y++) {
479 for (var x = math.max(0, node.x); x < math.min(width, node.x + bw); x++) {
480 final horizontal = node.side('border', 'top') == 1 && y == node.y || node.side('border', 'bottom') == 1 && y == node.y + bh - 1;
481 final vertical = node.side('border', 'left') == 1 && x == node.x || node.side('border', 'right') == 1 && x == node.x + bw - 1;
482 screen[y][x] = horizontal ? vertical ? '+' : '-' : vertical ? '|' : ' ';
483 }
484 }
485 if (node.text != null) {
486 final content = boxes(node).content;
487 for (var i = 0; i < node.lines.length; i++) {
488 final line = node.lines[i];
489 for (var j = 0; j < line.length; j++) { paint(content.x + j, content.y + i, line[j]); }
490 }
491 }
492 }
493 final edge = ''.padRight(width + 2, '#');
494 output.addAll(['render ${width}x$height', edge, ...screen.map((row) => '#${row.join()}#'), edge]);
495}
496void execute(String line) {
497 final words = line.split(' ');
498 final command = words.first, args = words.skip(1).toList();
499 final id = args.isEmpty ? '' : args[0];
500 if (command == 'LAYOUT') {
501 output.add('layout ${width}x$height');
502 for (final node in elements) { output.add('${''.padRight(node.depth * 2)}${node.id} ${node.shown ? boxes(node).border.text : 'none'}'); }
503 } else if (command == 'BOX') {
504 final node = byId[id];
505 if (node == null) { output.add('error: unknown element $id'); return; }
506 if (!node.shown) { output.add('box $id: none'); }
507 else {
508 output.add('box $id');
509 for (final entry in boxes(node).entries.entries) { output.add('${entry.key} ${entry.value.text}'); }
510 }
511 } else if (command == 'RENDER') { render(); }
512 else if (command == 'HIT') {
513 final x = int.parse(args[0]), y = int.parse(args[1]);
514 if (x >= width || y >= height) { output.add('error: point $x $y outside viewport'); return; }
515 Element? selected;
516 for (final node in elements) {
517 if (node.shown && x >= node.x && y >= node.y && x < node.x + node.borderWidth() && y < node.y + node.borderHeight()) selected = node;
518 }
519 final path = <String>[];
520 for (var current = selected; current != null; current = current.parent) { path.add(current.id); }
521 output.add('hit $x $y: ${path.isEmpty ? 'none' : path.reversed.join(' > ')}');
522 } else if (command == 'SET') {
523 final node = byId[id], property = args[1], value = args.skip(2).join(' ');
524 if (node == null) { output.add('error: unknown element $id'); return; }
525 if (!known.contains(property)) { output.add('error: unknown property $property'); return; }
526 final expanded = declarations(property, value);
527 if (expanded == null) { output.add('error: invalid value for $property: $value'); return; }
528 final before = snapshot();
529 for (final (key, item) in expanded) { node.overrides[key] = item; }
530 relayout();
531 output.add('set $id $property $value: ${changed(before)}');
532 } else if (command == 'VIEWPORT') {
533 final w = int.parse(args[0]), h = int.parse(args[1]);
534 if (w < 1 || w > 80 || h < 1 || h > 40) { output.add('error: invalid viewport $w $h'); return; }
535 final before = snapshot();
536 width = w;
537 height = h;
538 relayout();
539 output.add('viewport $w $h: ${changed(before)}');
540 } else if (command == 'OVERFLOW') {
541 final overflowing = <String>[];
542 for (final node in elements) {
543 if (!node.shown) continue;
544 bool overflow;
545 if (node.text != null) { overflow = maximumValue(node.lines.map((line) => line.length)) > node.cw || node.lines.length > node.ch; }
546 else {
547 final content = boxes(node).content;
548 overflow = node.flow().any((child) {
549 final margin = boxes(child).margin;
550 return margin.x < content.x || margin.y < content.y || margin.x + margin.w > content.x + content.w || margin.y + margin.h > content.y + content.h;
551 });
552 }
553 if (overflow) overflowing.add(node.id);
554 }
555 output.add('overflow: ${overflowing.isEmpty ? 'none' : overflowing.join(' ')}');
556 } else { output.add('error: unknown command $command'); }
557}
558void main() {
559 final dimensions = stdin.readLineSync()!.split(' ').map(int.parse).toList();
560 width = dimensions[0];
561 height = dimensions[1];
562 String line;
563 while ((line = stdin.readLineSync()!) != '---') {
564 final split = line.indexOf(' { '), expanded = <Declaration>[];
565 for (final item in line.substring(split + 3, line.length - 2).split('; ')) {
566 final colon = item.indexOf(': ');
567 expanded.addAll(declarations(item.substring(0, colon), item.substring(colon + 2)) ?? []);
568 }
569 rules.add(StyleRule(line.substring(0, split).split(', '), expanded));
570 }
571 final stack = <Element>[];
572 while ((line = stdin.readLineSync()!) != '===') {
573 var spaces = 0;
574 while (spaces < line.length && line[spaces] == ' ') { spaces++; }
575 final depth = spaces ~/ 2, text = line.substring(spaces), space = line.substring(spaces).indexOf(' ');
576 final head = space < 0 ? text : text.substring(0, space), hash = (space < 0 ? text : text.substring(0, space)).indexOf('#');
577 final parts = head.substring(hash + 1).split('.'), parent = depth == 0 ? null : stack[depth - 1];
578 final node = Element(head.substring(0, hash), parts[0], parts.skip(1).toList(), space < 0 ? null : text.substring(space + 2, text.length - 1), depth, parent, elements.length);
579 parent?.children.add(node);
580 elements.add(node);
581 byId[node.id] = node;
582 stack.length = depth;
583 stack.add(node);
584 }
585 relayout();
586 while ((line = stdin.readLineSync()!) != 'END') { execute(line); }
587 if (output.isNotEmpty) stdout.write('${output.join('\n')}\n');
588}
