27func natural(_ s: String) -> Bool {
28 (1...3).contains(s.utf8.count) && s.utf8.allSatisfy {
29 $0 >= 48 && $0 <= 57
30 }
31}

33func positive(_ s: String) -> Bool {

37func signed(_ s: String) -> Bool {

41func percent(_ s: String) -> Bool {

49func gridValue(_ t: [String]) -> Bool {
50 if t == ["auto"] {
51 return true
52 }
53 if t.count == 1 {
54 return positive(t[0])
55 }
56 if t.count == 2 {
57 return t[0] == "span" && positive(t[1])
58 }
59 if t.count == 3 {
60 return positive(t[0]) && t[1] == "/" && positive(t[2]) && Int(t[0])! < Int(t[2])!
61 }
62 if t.count == 4 {
63 return positive(t[0]) && t[1] == "/" && t[2] == "span" && positive(t[3])
64 }
65 return false
66}

68func valid(_ prop: String, _ t: [String]) -> Bool {
69 if prop.hasPrefix("grid-template-") {
70 return t == ["none"] || t.allSatisfy {
71 $0 == "auto" || natural($0) || fraction($0)
72 }
73 }
74 if prop == "grid-column" || prop == "grid-row" {
75 return gridValue(t)
76 }
77 if t.count != 1 {
78 return false
79 }
80 let v = t[0]
81 switch prop {
82 case "display":
83 return ["block", "flex", "grid", "none"].contains(v)
84 case "position":
85 return ["static", "relative", "absolute"].contains(v)
86 case "box-sizing":
87 return ["content-box", "border-box"].contains(v)
88 case "width", "height", "flex-basis":
89 return v == "auto" || natural(v) || percent(v)
90 case "min-width", "min-height":
91 return natural(v) || percent(v)
92 case "max-width", "max-height":
93 return v == "none" || natural(v) || percent(v)
94 case "top", "right", "bottom", "left":
95 return v == "auto" || signed(v)
96 case "order":
97 return signed(v)
98 case "flex-direction":
99 return ["row", "row-reverse", "column", "column-reverse"].contains(v)
100 case "flex-wrap":
101 return ["nowrap", "wrap"].contains(v)
102 case "justify-content":
103 return ["start", "end", "center", "space-between", "space-around", "space-evenly"].contains(v)
104 case "align-items":
105 return ["stretch", "start", "end", "center"].contains(v)
106 case "align-self":
107 return ["auto", "stretch", "start", "end", "center"].contains(v)
108 default:
109 return prop.hasPrefix("border-") ? v == "0" || v == "1" : natural(v)
110 }
111}

113func declaration(_ prop: String, _ value: String) -> [(String, String)]? {
114 let t = value.components(separatedBy: " ")
115 if initial[prop] != nil {
116 return valid(prop, t) ? [(prop, value)] : nil
117 }
118 if prop == "gap" {
119 if !(1...2).contains(t.count) || !t.allSatisfy(natural) {
120 return nil
121 }
122 return [("row-gap", t[0]), ("column-gap", t.last!)]
123 }
124 if shorthands.contains(prop) && (1...4).contains(t.count) && t.allSatisfy({
125 valid("\(prop)-top", [$0])
126 }) {
127 let pick: [Int]
128 switch t.count {
129 case 1:
130 pick = [0, 0, 0, 0]
131 case 2:
132 pick = [0, 1, 0, 1]
133 case 3:
134 pick = [0, 1, 2, 1]
135 default:
136 pick = [0, 1, 2, 3]
137 }
138 return sides.indices.map {
139 ("\(prop)-\(sides[$0])", t[pick[$0]])
140 }
141 }
142 return nil
143}

296func distribution(_ total: Int, _ weights: [Int]) -> [Int] {
297 let sum = weights.reduce(0, +)
298 if sum == 0 {
299 return Array(repeating: 0, count: weights.count)
300 }
301 var out = weights.map {
302 total * $0 / sum
303 }
304 var rest = total - out.reduce(0, +)
305 for i in weights.indices where rest > 0 && weights[i] > 0 {
306 out[i] += 1
307 rest -= 1
308 }
309 return out
310}

329func grid(_ e: Element) -> Grid {
330 let items = e.items
331 let columns = e.s("grid-template-columns") == "none" ? ["auto"] : e.s("grid-template-columns").components(separatedBy: " ")
332 let rows = e.s("grid-template-rows") == "none" ? [] : e.s("grid-template-rows").components(separatedBy: " ")
333 var areas = items.map {
334 child -> Area in let (c, cs) = placement(child.s("grid-column"))
335 let (r, rs) = placement(child.s("grid-row"))
336 return Area(row: r, col: c, rows: rs, cols: cs)
337 }
338 let nc = max(columns.count, areas.map {
339 $0.col == 0 ? $0.cols : $0.col + $0.cols - 1
340 }.max() ?? 0)
341 var occupied = Set<Cell>()
342 func mark(_ a: Area) {
343 for r in a.row..<(a.row + a.rows) {
344 for c in a.col..<(a.col + a.cols) {
345 occupied.insert(Cell(row: r, col: c))
346 }
347 }
348 }
349 func overlaps(_ a: Area, _ row: Int, _ col: Int) -> Bool {
350 (row..<(row + a.rows)).contains {
351 r in (col..<(col + a.cols)).contains {
352 c in occupied.contains(Cell(row: r, col: c))
353 }
354 }
355 }
356 for a in areas where a.row != 0 && a.col != 0 {
357 mark(a)
358 }
359 var cr = 1
360 var cc = 1
361 for i in areas.indices {
362 var a = areas[i]
363 if a.row != 0 && a.col != 0 {
364 continue
365 }
366 if a.row != 0 {
367 a.col = (1...(nc - a.cols + 1)).first {
368 !overlaps(a, a.row, $0)
369 } ?? 1
370 } else if a.col != 0 {
371 if a.col < cc {
372 cr += 1
373 }
374 var r = cr
375 while overlaps(a, r, a.col) {
376 r += 1
377 }
378 a.row = r
379 cr = r
380 cc = a.col + a.cols
381 } else {
382 var r = cr
383 var c = cc
384 while true {
385 if c + a.cols - 1 > nc {
386 r += 1
387 c = 1
388 } else if overlaps(a, r, c) {
389 c += 1
390 } else {
391 break
392 }
393 }
394 a.row = r
395 a.col = c
396 cr = r
397 cc = c + a.cols
398 }
399 areas[i] = a
400 mark(a)
401 }
402 let nr = max(rows.count, areas.map {
403 $0.row + $0.rows - 1
404 }.max() ?? 0)
405 return Grid(items: items, areas: areas, columns: columns, rows: rows, nc: nc, nr: nr)
406}

408func tracks(_ g: Grid, _ horizontal: Bool, _ available: Int?, _ gap: Int, _ contributions: [Int]) -> [Int] {
409 let count = horizontal ? g.nc : g.nr
410 let explicit = horizontal ? g.columns : g.rows
411 let specs = (0..<count).map {
412 $0 < explicit.count ? explicit[$0] : "auto"
413 }
414 var sizes = specs.enumerated().map {
415 i, spec -> Int in
416 if natural(spec) {
417 return Int(spec)!
418 }
419 if fraction(spec) && available != nil {
420 return 0
421 }
422 return g.areas.indices.filter {
423 j in let a = g.areas[j]
424 return (horizontal ? a.col : a.row) == i + 1 && (horizontal ? a.cols : a.rows) == 1
425 }.map {
426 contributions[$0]
427 }.max() ?? 0
428 }
429 if let u = available {
430 let left = u - sizes.reduce(0, +) - gap * max(0, count - 1)
431 if left > 0 {
432 let hasFr = specs.contains(where: fraction)
433 let weights = specs.map {
434 v -> Int in if hasFr {
435 return fraction(v) ? Int(v.dropLast(2))! : 0
436 }
437 return v == "auto" ? 1 : 0
438 }
439 let extra = distribution(left, weights)
440 for i in sizes.indices {
441 sizes[i] += extra[i]
442 }
443 }
444 }
445 return sizes
446}

478@discardableResult
479func layout(_ e: Element, _ cwRef: Int, _ chRef: Int?, _ fw: Int? = nil, _ givenHeight: Int? = nil) -> Int {
480 e.cw = fw ?? e.clamp(e.resolve("width", cwRef, true) ?? max(0, cwRef - e.margin(true) - e.frame(true)), cwRef, true)
481 let fh = givenHeight ?? e.resolve("height", chRef, false).map {
482 e.clamp($0, chRef, false)
483 }
484 let a: Int
485 if let text = e.text {
486 e.lines = wrap(text, e.cw)
487 a = e.lines.count
488 } else if e.s("display") == "flex" {
489 a = flex(e, fh, chRef)
490 } else if e.s("display") == "grid" {
491 a = gridLayout(e, fh)
492 } else {
493 var bottom = 0
494 var previous = 0
495 var first = true
496 for child in e.flow {
497 layout(child, e.cw, fh)
498 let y = bottom + (first ? child.edge("margin", 0) : max(previous, child.edge("margin", 0)))
499 e.place(child, child.edge("margin", 3), y)
500 bottom = y + child.borderHeight
501 previous = child.edge("margin", 2)
502 first = false
503 }
504 a = bottom + previous
505 }
506 e.ch = fh ?? e.clamp(a, chRef, false)
507 return a
508}

514func flex(_ e: Element, _ k: Int?, _ heightReference: Int?) -> Int {
515 let items = e.items
516 let horizontal = e.s("flex-direction").hasPrefix("row")
517 let reverse = e.s("flex-direction").hasSuffix("reverse")
518 let gap = e.n(horizontal ? "column-gap" : "row-gap")
519 let lineGap = e.n("row-gap")
520 let reference: Int? = horizontal ? e.cw : k
521 var cross = Array(repeating: 0, count: items.count)
522 var base = cross
523 var hypothetical = cross
524 let outer = items.map {
525 $0.frame(horizontal) + $0.margin(horizontal)
526 }
527 for (i, child) in items.enumerated() {
528 if !horizontal {
529 let r = max(0, e.cw - child.margin(true) - child.frame(true))
530 cross[i] = child.clamp(child.resolve("width", e.cw, true) ?? (e.align(child) == "stretch" ? r : min(maxContent(child), r)), e.cw, true)
531 }
532 base[i] = child.resolve("flex-basis", reference, horizontal) ?? child.resolve(horizontal ? "width" : "height", reference, horizontal) ?? (horizontal ? maxContent(child) : layout(child, e.cw, k, cross[i]))
533 hypothetical[i] = child.clamp(base[i], reference, horizontal)
534 }
535 let intrinsic = items.indices.map {
536 hypothetical[$0] + outer[$0]
537 }.reduce(0, +) + gap * max(0, items.count - 1)
538 let length = horizontal ? e.cw : k ?? e.clamp(intrinsic, heightReference, false)
539 var lines = [[Int]]()
540 var used = 0
541 for i in items.indices {
542 let size = hypothetical[i] + outer[i]
543 if lines.isEmpty || (horizontal && e.s("flex-wrap") == "wrap" && used + gap + size > length) {
544 lines.append([i])
545 used = size
546 } else {
547 lines[lines.count - 1].append(i)
548 used += gap + size
549 }
550 }
551 var lineStart = 0
552 for line in lines {
553 let totalGap = gap * (line.count - 1)
554 let grow = line.map {
555 hypothetical[$0] + outer[$0]
556 }.reduce(0, +) + totalGap < length
557 let weights = line.map {
558 grow ? items[$0].n("flex-grow") : items[$0].n("flex-shrink") * base[$0]
559 }
560 var target = line.map {
561 hypothetical[$0]
562 }
563 var frozen = line.enumerated().map {
564 j, i in weights[j] == 0 || (grow ? base[i] > hypothetical[i] : base[i] < hypothetical[i])
565 }
566 while frozen.contains(false) {
567 let free = length - totalGap - line.indices.map {
568 (frozen[$0] ? target[$0] : base[line[$0]]) + outer[line[$0]]
569 }.reduce(0, +)
570 let assignments = distribution(max(0, grow ? free : -free), line.indices.map {
571 frozen[$0] ? 0 : weights[$0]
572 })
573 var violations = Array(repeating: 0, count: line.count)
574 for j in line.indices where !frozen[j] {
575 let tentative = base[line[j]] + (grow ? assignments[j] : -assignments[j])
576 target[j] = items[line[j]].clamp(tentative, reference, horizontal)
577 violations[j] = target[j] - tentative
578 }
579 let violation = violations.reduce(0, +)
580 for j in line.indices where violation == 0 || violation > 0 && violations[j] > 0 || violation < 0 && violations[j] < 0 {
581 frozen[j] = true
582 }
583 }
584 if horizontal {
585 for j in line.indices {
586 let i = line[j]
587 layout(items[i], e.cw, k, target[j])
588 cross[i] = items[i].ch
589 }
590 }
591 var lineCross = horizontal ? line.map {
592 cross[$0] + items[$0].frame(false) + items[$0].margin(false)
593 }.max()! : e.cw
594 if horizontal && lines.count == 1 {
595 lineCross = k ?? e.clamp(lineCross, heightReference, false)
596 }
597 let remaining = length - line.indices.map {
598 target[$0] + outer[line[$0]]
599 }.reduce(0, +) - totalGap
600 var spaces = Array(repeating: 0, count: line.count)
601 if remaining > 0 {
602 let mode = e.s("justify-content")
603 if mode == "end" {
604 spaces[0] = remaining
605 } else if mode == "center" {
606 spaces[0] = remaining / 2
607 } else if mode.hasPrefix("space-") {
608 let w = (0...line.count).map {
609 j -> Int in
610 if mode == "space-evenly" {
611 return 1
612 }
613 if j == 0 || j == line.count {
614 return mode == "space-between" ? 0 : 1
615 }
616 return mode == "space-around" ? 2 : 1
617 }
618 spaces = Array(distribution(remaining, w).prefix(line.count))
619 }
620 }
621 var cursor = 0
622 for j in line.indices {
623 let i = line[j]
624 let child = items[i]
625 let align = e.align(child)
626 if horizontal {
627 if align == "stretch" && child.s("height") == "auto" {
628 cross[i] = child.clamp(max(0, lineCross - child.margin(false) - child.frame(false)), k, false)
629 layout(child, e.cw, k, target[j], cross[i])
630 } else {
631 layout(child, e.cw, k, target[j])
632 cross[i] = child.ch
633 }
634 } else {
635 layout(child, e.cw, k, cross[i], target[j])
636 }
637 cursor += spaces[j]
638 let main = reverse ? length - cursor - target[j] - outer[i] : cursor
639 let crossPosition = lineStart + offset(lineCross - cross[i] - child.frame(!horizontal) - child.margin(!horizontal), align)
640 if horizontal {
641 e.place(child, main + child.edge("margin", 3), crossPosition + child.edge("margin", 0))
642 } else {
643 e.place(child, crossPosition + child.edge("margin", 3), main + child.edge("margin", 0))
644 }
645 cursor += target[j] + outer[i] + gap
646 }
647 lineStart += lineCross + lineGap
648 }
649 return horizontal ? (lines.isEmpty ? 0 : lineStart - lineGap) : length
650}

652func gridLayout(_ e: Element, _ k: Int?) -> Int {
653 let g = grid(e)
654 let colGap = e.n("column-gap")
655 let rowGap = e.n("row-gap")
656 let columns = tracks(g, true, e.cw, colGap, g.items.map(contribution))
657 var widths = Array(repeating: 0, count: g.items.count)
658 var areaWidths = widths
659 let contributions = g.items.indices.map {
660 i -> Int in
661 let a = g.areas[i]
662 let child = g.items[i]
663 let aw = columns[(a.col - 1)..<(a.col - 1 + a.cols)].reduce(0, +) + colGap * (a.cols - 1)
664 areaWidths[i] = aw
665 widths[i] = child.clamp(child.resolve("width", aw, true) ?? max(0, aw - child.margin(true) - child.frame(true)), aw, true)
666 layout(child, aw, nil, widths[i])
667 return child.borderHeight + child.margin(false)
668 }
669 let rows = tracks(g, false, k, rowGap, contributions)
670 for i in g.items.indices {
671 let a = g.areas[i]
672 let child = g.items[i]
673 let align = e.align(child)
674 let ah = rows[(a.row - 1)..<(a.row - 1 + a.rows)].reduce(0, +) + rowGap * (a.rows - 1)
675 let ax = columns.prefix(a.col - 1).reduce(0, +) + colGap * (a.col - 1)
676 let ay = rows.prefix(a.row - 1).reduce(0, +) + rowGap * (a.row - 1)
677 let fh: Int? = align == "stretch" && child.s("height") == "auto" ? child.clamp(max(0, ah - child.margin(false) - child.frame(false)), ah, false) : nil
678 layout(child, areaWidths[i], ah, widths[i], fh)
679 let dy = offset(ah - child.borderHeight - child.margin(false), align)
680 e.place(child, ax + child.edge("margin", 3), ay + dy + child.edge("margin", 0))
681 }
682 return rows.reduce(0, +) + rowGap * max(0, g.nr - 1)
683}

685final class Document {
686 var width: Int
687 var height: Int
688 let rules: [Rule]
689 let elements: [Element]
690 let byId: [String: Element]
691 init(_ width: Int, _ height: Int, _ rules: [Rule], _ elements: [Element]) {
692 self.width = width
693 self.height = height
694 self.rules = rules
695 self.elements = elements
696 byId = Dictionary(uniqueKeysWithValues: elements.map {
697 ($0.id, $0)
698 })
699 }
700 func containing(_ e: Element) -> Element? {
701 var p = e.parent
702 while let node = p {
703 if node.s("position") != "static" {
704 return node
705 }
706 p = node.parent
707 }
708 return nil
709 }
710 func reflow() {
711 for e in elements {
712 e.style = initial
713 var specificity = [String: Int]()
714 for rule in rules {
715 let best = rule.selectors.compactMap {
716 s -> Int? in
717 if s.hasPrefix("#") {
718 return String(s.dropFirst()) == e.id ? 2 : nil
719 }
720 if s.hasPrefix(".") {
721 return e.classes.contains(String(s.dropFirst())) ? 1 : nil
722 }
723 return s == e.kind ? 0 : nil
724 }.max()
725 if let best = best {
726 for (p, v) in rule.values where best >= (specificity[p] ?? -1) {
727 e.style[p] = v
728 specificity[p] = best
729 }
730 }
731 }
732 for (p, v) in e.overrides {
733 e.style[p] = v
734 }
735 e.displayed = e.s("display") != "none" && (e.parent?.displayed ?? true)
736 }
737 let root = elements[0]
738 if root.displayed && root.s("position") != "absolute" {
739 layout(root, width, height)
740 root.rx = root.edge("margin", 3)
741 root.ry = root.edge("margin", 0)
742 }
743 for e in elements where e.displayed && e.s("position") == "absolute" {
744 let block = containing(e)
745 let cbw = block.map {
746 $0.cw + $0.edge("padding", 1) + $0.edge("padding", 3)
747 } ?? width
748 let cbh = block.map {
749 $0.ch + $0.edge("padding", 0) + $0.edge("padding", 2)
750 } ?? height
751 func side(_ s: String) -> Int? {
752 e.s(s) == "auto" ? nil : e.n(s)
753 }
754 let left = side("left")
755 let right = side("right")
756 let top = side("top")
757 let bottom = side("bottom")
758 let r = max(0, cbw - (left ?? 0) - (right ?? 0) - e.margin(true) - e.frame(true))
759 let w = e.resolve("width", cbw, true) ?? (left != nil && right != nil ? r : min(maxContent(e), r))
760 let fw = e.clamp(w, cbw, true)
761 let fh: Int? = e.s("height") == "auto" && top != nil && bottom != nil ? e.clamp(max(0, cbh - top! - bottom! - e.margin(false) - e.frame(false)), cbh, false) : nil
762 layout(e, cbw, cbh, fw, fh)
763 e.rx = left.map {
764 $0 + e.edge("margin", 3)
765 } ?? right.map {
766 cbw - $0 - e.edge("margin", 1) - e.borderWidth
767 } ?? e.edge("margin", 3)
768 e.ry = top.map {
769 $0 + e.edge("margin", 0)
770 } ?? bottom.map {
771 cbh - $0 - e.edge("margin", 2) - e.borderHeight
772 } ?? e.edge("margin", 0)
773 }
774 for e in elements where e.displayed {
775 let absolute = e.s("position") == "absolute"
776 let ref = absolute ? containing(e) : e.parent
777 let bx = ref.map {
778 $0.x + (absolute ? $0.edge("border", 3) : 0)
779 } ?? 0
780 let by = ref.map {
781 $0.y + (absolute ? $0.edge("border", 0) : 0)
782 } ?? 0
783 var dx = 0
784 var dy = 0
785 if e.s("position") == "relative" {
786 dx = e.s("left") != "auto" ? e.n("left") : e.s("right") != "auto" ? -e.n("right") : 0
787 dy = e.s("top") != "auto" ? e.n("top") : e.s("bottom") != "auto" ? -e.n("bottom") : 0
788 }
789 e.x = bx + e.rx + dx
790 e.y = by + e.ry + dy
791 }
792 }
793 func snapshot() -> [Rect?] {
794 elements.map {
795 $0.displayed ? $0.border : nil
796 }
797 }
798 func changed(_ old: [Rect?]) -> String {
799 let now = snapshot()
800 let ids = elements.indices.filter {
801 old[$0] != now[$0]
802 }.map {
803 elements[$0].id
804 }
805 return ids.isEmpty ? "no change" : ids.joined(separator: " ")
806 }
807 func render() -> [String] {
808 var screen = Array(repeating: Array(repeating: Character("."), count: width), count: height)
809 for e in elements where e.displayed {
810 let b = e.border
811 let ymin = max(0, b.y)
812 let ymax = min(height, b.y + b.h)
813 let xmin = max(0, b.x)
814 let xmax = min(width, b.x + b.w)
815 if ymin < ymax && xmin < xmax {
816 for y in ymin..<ymax {
817 for x in xmin..<xmax {
818 let horizontal = e.edge("border", 0) == 1 && y == b.y || e.edge("border", 2) == 1 && y == b.y + b.h - 1
819 let vertical = e.edge("border", 3) == 1 && x == b.x || e.edge("border", 1) == 1 && x == b.x + b.w - 1
820 screen[y][x] = horizontal && vertical ? "+" : horizontal ? "-" : vertical ? "|" : " "
821 }
822 }
823 }
824 if e.text != nil {
825 let c = e.content
826 for (i, line) in e.lines.enumerated() {
827 for (j, ch) in line.enumerated() {
828 let x = c.x + j
829 let y = c.y + i
830 if x >= 0 && x < width && y >= 0 && y < height {
831 screen[y][x] = ch
832 }
833 }
834 }
835 }
836 }
837 let frame = String(repeating: "#", count: width + 2)
838 return ["render \(width)x\(height)", frame] + screen.map {
839 "#" + String($0) + "#"
840 } + [frame]
841 }
842 func command(_ line: String) -> [String] {
843 let a = line.components(separatedBy: " ")
844 switch a[0] {
845 case "LAYOUT":
846 return ["layout \(width)x\(height)"] + elements.map {
847 String(repeating: " ", count: $0.depth) + $0.id + ($0.displayed ? " \($0.border)" : " none")
848 }
849 case "BOX":
850 guard let e = byId[a[1]] else {
851 return ["error: unknown element \(a[1])"]
852 }
853 if !e.displayed {
854 return ["box \(e.id): none"]
855 }
856 return ["box \(e.id)", "margin \(e.marginBox)", "border \(e.border)", "padding \(e.padding)", "content \(e.content)"]
857 case "RENDER":
858 return render()
859 case "HIT":
860 let x = Int(a[1])!
861 let y = Int(a[2])!
862 if x >= width || y >= height {
863 return ["error: point \(x) \(y) outside viewport"]
864 }
865 var e = elements.last {
866 $0.displayed && $0.border.contains(x, y)
867 }
868 var path = [String]()
869 while let node = e {
870 path.append(node.id)
871 e = node.parent
872 }
873 return ["hit \(x) \(y): " + (path.isEmpty ? "none" : path.reversed().joined(separator: " > "))]
874 case "SET":
875 let id = a[1]
876 let p = a[2]
877 let value = a.dropFirst(3).joined(separator: " ")
878 guard let e = byId[id] else {
879 return ["error: unknown element \(id)"]
880 }
881 if initial[p] == nil && !shorthands.contains(p) {
882 return ["error: unknown property \(p)"]
883 }
884 guard let values = declaration(p, value) else {
885 return ["error: invalid value for \(p): \(value)"]
886 }
887 let old = snapshot()
888 for (p, v) in values {
889 e.overrides[p] = v
890 }
891 reflow()
892 return ["set \(id) \(p) \(value): \(changed(old))"]
893 case "VIEWPORT":
894 let w = Int(a[1])!
895 let h = Int(a[2])!
896 if !(1...80).contains(w) || !(1...40).contains(h) {
897 return ["error: invalid viewport \(w) \(h)"]
898 }
899 let old = snapshot()
900 width = w
901 height = h
902 reflow()
903 return ["viewport \(w) \(h): \(changed(old))"]
904 case "OVERFLOW":
905 let ids = elements.filter {
906 e in e.displayed && (e.text != nil ? e.lines.contains {
907 $0.count > e.cw
908 } || e.lines.count > e.ch : e.flow.contains {
909 !e.content.contains($0.marginBox)
910 })
911 }.map {
912 $0.id
913 }
914 return ["overflow: " + (ids.isEmpty ? "none" : ids.joined(separator: " "))]
915 default:
916 return ["error: unknown command \(a[0])"]
917 }
918 }
919}
920
921let size = readLine()!.split(separator: " ").map {
922 Int($0)!
923}
924
925var rules = [Rule]()
926while let line = readLine(), line != "---" {
927 let range = line.range(of: " { ")!
928 let selectors = String(line[..<range.lowerBound]).components(separatedBy: ", ")
929 let body = String(line[range.upperBound...].dropLast(2))
930 let values = body.components(separatedBy: "; ").flatMap {
931 d -> [(String, String)] in let colon = d.range(of: ": ")!
932 return declaration(String(d[..<colon.lowerBound]), String(d[colon.upperBound...])) ?? []
933 }
934 rules.append(Rule(selectors: selectors, values: values))
935}
936
937var elements = [Element]()
938
939var ancestors = [Element]()
940while let line = readLine(), line != "===" {
941 let spaces = line.prefix {
942 $0 == " "
943 }.count
944 let depth = spaces / 2
945 let body = String(line.dropFirst(spaces))
946 let head: String
947 let text: String?
948 if let at = body.firstIndex(of: " ") {
949 head = String(body[..<at])
950 text = String(body[body.index(at, offsetBy: 2)...].dropLast())
951 } else {
952 head = body
953 text = nil
954 }
955 let hash = head.firstIndex(of: "#")!
956 let parts = String(head[head.index(after: hash)...]).components(separatedBy: ".")
957 let e = Element(String(head[..<hash]), parts[0], Set(parts.dropFirst()), text, depth == 0 ? nil : ancestors[depth - 1], elements.count)
958 e.parent?.children.append(e)
959 elements.append(e)
960 ancestors = Array(ancestors.prefix(depth))
961 ancestors.append(e)
962}
963
964let document = Document(size[0], size[1], rules, elements)
965document.reflow()
966
967var output = ""
968while let line = readLine(), line != "END" {
969 for result in document.command(line) {
970 output += result + "\n"
971 }
972}
973print(output, terminator: "")
