33private fun gridValue(t: List<String>): Boolean = when {
34 t == listOf("auto") -> true
35 t.size == 1 -> positive(t[0])
36 t.size == 2 -> t[0] == "span" && positive(t[1])
37 t.size == 3 -> positive(t[0]) && t[1] == "/" && positive(t[2]) && t[0].toInt() < t[2].toInt()
38 t.size == 4 -> positive(t[0]) && t[1] == "/" && t[2] == "span" && positive(t[3])
39 else -> false
40}

42private fun valid(prop: String, t: List<String>): Boolean {
43 if (prop.startsWith("grid-template-")) return t == listOf("none") || t.all {
44 it == "auto" || natural(it) || fraction(it)
45 }
46 if (prop == "grid-column" || prop == "grid-row") return gridValue(t)
47 if (t.size != 1) return false
48 val v = t[0]
49 return when (prop) {
50 "display" -> v in setOf("block", "flex", "grid", "none")
51 "position" -> v in setOf("static", "relative", "absolute")
52 "box-sizing" -> v in setOf("content-box", "border-box")
53 "width", "height", "flex-basis" -> v == "auto" || natural(v) || percent(v)
54 "min-width", "min-height" -> natural(v) || percent(v)
55 "max-width", "max-height" -> v == "none" || natural(v) || percent(v)
56 "top", "right", "bottom", "left" -> v == "auto" || signed(v)
57 "order" -> signed(v)
58 "flex-direction" -> v in setOf("row", "row-reverse", "column", "column-reverse")
59 "flex-wrap" -> v == "nowrap" || v == "wrap"
60 "justify-content" -> v in setOf("start", "end", "center", "space-between", "space-around", "space-evenly")
61 "align-items" -> v in setOf("stretch", "start", "end", "center")
62 "align-self" -> v in setOf("auto", "stretch", "start", "end", "center")
63 else -> if (prop.startsWith("border-")) v == "0" || v == "1" else natural(v)
64 }
65}

67private fun declaration(prop: String, value: String): List<Pair<String, String>>? {
68 val t = value.split(' ')
69 if (prop in initial) return if (valid(prop, t)) listOf(prop to value) else null
70 if (prop == "gap") {
71 if (t.size !in 1..2 || !t.all(::natural)) return null
72 return listOf("row-gap" to t[0], "column-gap" to t.last())
73 }
74 if (prop in shorthands && t.size in 1..4 && t.all {
75 valid("$prop-top", listOf(it))
76 }) {
77 val pick = when (t.size) {
78 1 -> listOf(0, 0, 0, 0)
79 2 -> listOf(0, 1, 0, 1)
80 3 -> listOf(0, 1, 2, 1)
81 else -> listOf(0, 1, 2, 3)
82 }
83 return sides.indices.map {
84 "$prop-${sides[it]}" to t[pick[it]]
85 }
86 }
87 return null
88}

159private fun distribution(total: Int, weights: List<Int>): List<Int> {
160 val sum = weights.sum()
161 if (sum == 0) return List(weights.size) {
162 0
163 }
164 val out = weights.map {
165 (total.toLong() * it / sum).toInt()
166 }.toMutableList()
167 var rest = total - out.sum()
168 for (i in weights.indices) if (rest > 0 && weights[i] > 0) {
169 out[i]++
170 rest--
171 }
172 return out
173}

186private fun grid(e: Element): Grid {
187 val items = e.items()
188 val columns = if (e.s("grid-template-columns") == "none") listOf("auto") else e.s("grid-template-columns").split(' ')
189 val rows = if (e.s("grid-template-rows") == "none") emptyList() else e.s("grid-template-rows").split(' ')
190 val areas = items.map {
191 val (c, cs) = placement(it.s("grid-column"))
192 val (r, rs) = placement(it.s("grid-row"))
193 Area(r, c, rs, cs)
194 }
195 val nc = maxOf(columns.size, areas.maxOfOrNull {
196 if (it.col == 0) it.cols else it.col + it.cols - 1
197 }?: 0)
198 val occupied = mutableSetOf<Pair<Int, Int>>()
199 fun mark(a: Area) {
200 for (r in a.row until a.row + a.rows) for (c in a.col until a.col + a.cols) occupied.add(r to c)
201 }
202 fun overlaps(a: Area, row: Int, col: Int): Boolean = (row until row + a.rows).any {
203 r -> (col until col + a.cols).any {
204 c -> (r to c) in occupied
205 }
206 }
207 for (a in areas) if (a.row != 0 && a.col != 0) mark(a)
208 var cr = 1
209 var cc = 1
210 for (a in areas) {
211 if (a.row != 0 && a.col != 0) continue
212 if (a.row != 0) {
213 a.col = (1..(nc - a.cols + 1)).firstOrNull {
214 !overlaps(a, a.row, it)
215 }?: 1
216 } else if (a.col != 0) {
217 if (a.col < cc) cr++
218 var r = cr
219 while (overlaps(a, r, a.col)) r++
220 a.row = r
221 cr = r
222 cc = a.col + a.cols
223 } else {
224 var r = cr
225 var c = cc
226 while (true) {
227 if (c + a.cols - 1 > nc) {
228 r++
229 c = 1
230 } else if (overlaps(a, r, c)) c++
231 else break
232 }
233 a.row = r
234 a.col = c
235 cr = r
236 cc = c + a.cols
237 }
238 mark(a)
239 }
240 val nr = maxOf(rows.size, areas.maxOfOrNull {
241 it.row + it.rows - 1
242 }?: 0)
243 return Grid(items, areas, columns, rows, nc, nr)
244}

246private fun tracks(g: Grid, horizontal: Boolean, available: Int?, gap: Int, contributions: List<Int>): List<Int> {
247 val count = if (horizontal) g.nc else g.nr
248 val explicit = if (horizontal) g.columns else g.rows
249 val specs = List(count) {
250 explicit.getOrElse(it) {
251 "auto"
252 }
253 }
254 val sizes = specs.mapIndexed {
255 i, spec ->
256 when {
257 natural(spec) -> spec.toInt()
258 fraction(spec) && available != null -> 0
259 else -> g.areas.indices.filter {
260 val a = g.areas[it]
261 (if (horizontal) a.col else a.row) == i + 1 && (if (horizontal) a.cols else a.rows) == 1
262 }.maxOfOrNull {
263 contributions[it]
264 }?: 0
265 }
266 }.toMutableList()
267 if (available != null) {
268 val left = available - sizes.sum() - gap * maxOf(0, count - 1)
269 if (left > 0) {
270 val hasFr = specs.any(::fraction)
271 val weights = specs.map {
272 if (hasFr) {
273 if (fraction(it)) it.dropLast(2).toInt() else 0
274 } else if (it == "auto") 1 else 0
275 }
276 val extra = distribution(left, weights)
277 for (i in sizes.indices) sizes[i] += extra[i]
278 }
279 }
280 return sizes
281}

308private fun layout(e: Element, cwRef: Int, chRef: Int?, fw: Int? = null, givenHeight: Int? = null): Int {
309 e.cw = fw ?: e.clamp(e.resolve("width", cwRef, true) ?: maxOf(0, cwRef - e.margin(true) - e.frame(true)), cwRef, true)
310 val fh = givenHeight ?: e.resolve("height", chRef, false)?.let {
311 e.clamp(it, chRef, false)
312 }
313 val a = when {
314 e.text != null -> {
315 e.lines = wrap(e.text, e.cw)
316 e.lines.size
317 }
318 e.s("display") == "flex" -> flex(e, fh, chRef)
319 e.s("display") == "grid" -> gridLayout(e, fh)
320 else -> {
321 var bottom = 0
322 var previous = 0
323 var first = true
324 for (child in e.flow()) {
325 layout(child, e.cw, fh)
326 val y = bottom + if (first) child.edge("margin", 0) else maxOf(previous, child.edge("margin", 0))
327 e.place(child, child.edge("margin", 3), y)
328 bottom = y + child.borderHeight()
329 previous = child.edge("margin", 2)
330 first = false
331 }
332 bottom + previous
333 }
334 }
335 e.ch = fh ?: e.clamp(a, chRef, false)
336 return a
337}

345private fun flex(e: Element, k: Int?, heightReference: Int?): Int {
346 val items = e.items()
347 val horizontal = e.s("flex-direction").startsWith("row")
348 val reverse = e.s("flex-direction").endsWith("reverse")
349 val gap = e.n(if (horizontal) "column-gap" else "row-gap")
350 val lineGap = e.n("row-gap")
351 val reference = if (horizontal) e.cw else k
352 val cross = IntArray(items.size)
353 val base = IntArray(items.size)
354 val hypothetical = IntArray(items.size)
355 val outer = items.map {
356 it.frame(horizontal) + it.margin(horizontal)
357 }
358 for ((i, child) in items.withIndex()) {
359 if (!horizontal) {
360 val r = maxOf(0, e.cw - child.margin(true) - child.frame(true))
361 cross[i] = child.clamp(child.resolve("width", e.cw, true) ?: if (e.align(child) == "stretch") r else minOf(maxContent(child), r), e.cw, true)
362 }
363 base[i] = child.resolve("flex-basis", reference, horizontal) ?: child.resolve(if (horizontal) "width" else "height", reference, horizontal) ?: if (horizontal) maxContent(child) else layout(child, e.cw, k, cross[i])
364 hypothetical[i] = child.clamp(base[i], reference, horizontal)
365 }
366 val length = if (horizontal) e.cw else k ?: e.clamp(items.indices.sumOf {
367 hypothetical[it] + outer[it]
368 }+ gap * maxOf(0, items.size - 1), heightReference, false)
369 val lines = mutableListOf<MutableList<Int>>()
370 var used = 0
371 for (i in items.indices) {
372 val size = hypothetical[i] + outer[i]
373 if (lines.isEmpty() || (horizontal && e.s("flex-wrap") == "wrap" && used + gap + size > length)) {
374 lines.add(mutableListOf(i))
375 used = size
376 } else {
377 if (lines.last().isNotEmpty()) used += gap
378 lines.last().add(i)
379 used += size
380 }
381 }
382 var lineStart = 0
383 for (line in lines) {
384 val totalGap = gap * (line.size - 1)
385 val grow = line.sumOf {
386 hypothetical[it] + outer[it]
387 }+ totalGap < length
388 val weights = line.map {
389 if (grow) items[it].n("flex-grow") else items[it].n("flex-shrink") * base[it]
390 }
391 val target = line.map {
392 hypothetical[it]
393 }.toMutableList()
394 val frozen = line.mapIndexed {
395 j, i -> weights[j] == 0 || if (grow) base[i] > hypothetical[i] else base[i] < hypothetical[i]
396 }.toMutableList()
397 while (frozen.any {
398 !it
399 }) {
400 val free = length - totalGap - line.indices.sumOf {
401 j -> (if (frozen[j]) target[j] else base[line[j]]) + outer[line[j]]
402 }
403 val assignments = distribution(maxOf(0, if (grow) free else -free), line.indices.map {
404 if (frozen[it]) 0 else weights[it]
405 })
406 val violations = IntArray(line.size)
407 for (j in line.indices) if (!frozen[j]) {
408 val tentative = base[line[j]] + if (grow) assignments[j] else -assignments[j]
409 target[j] = items[line[j]].clamp(tentative, reference, horizontal)
410 violations[j] = target[j] - tentative
411 }
412 val violation = violations.sum()
413 for (j in line.indices) if (violation == 0 || violation > 0 && violations[j] > 0 || violation < 0 && violations[j] < 0) frozen[j] = true
414 }
415 if (horizontal) for (j in line.indices) {
416 val i = line[j]
417 layout(items[i], e.cw, k, target[j])
418 cross[i] = items[i].ch
419 }
420 var lineCross = if (!horizontal) e.cw else line.maxOf {
421 cross[it] + items[it].frame(false) + items[it].margin(false)
422 }
423 if (horizontal && lines.size == 1) lineCross = k ?: e.clamp(lineCross, heightReference, false)
424 val remaining = length - line.indices.sumOf {
425 target[it] + outer[line[it]]
426 } - totalGap
427 val spaces = MutableList(line.size) {
428 0
429 }
430 if (remaining > 0) when (e.s("justify-content")) {
431 "end" -> spaces[0] = remaining
432 "center" -> spaces[0] = remaining / 2
433 "space-between", "space-around", "space-evenly" -> {
434 val mode = e.s("justify-content")
435 val w = List(line.size + 1) {
436 j -> if (mode == "space-evenly") 1 else if (j == 0 || j == line.size) {
437 if (mode == "space-between") 0 else 1
438 } else if (mode == "space-around") 2 else 1
439 }
440 val assigned = distribution(remaining, w)
441 for (j in spaces.indices) spaces[j] = assigned[j]
442 }
443 }
444 var cursor = 0
445 for (j in line.indices) {
446 val i = line[j]
447 val child = items[i]
448 val align = e.align(child)
449 if (horizontal) {
450 if (align == "stretch" && child.s("height") == "auto") {
451 cross[i] = child.clamp(maxOf(0, lineCross - child.margin(false) - child.frame(false)), k, false)
452 layout(child, e.cw, k, target[j], cross[i])
453 } else {
454 layout(child, e.cw, k, target[j])
455 cross[i] = child.ch
456 }
457 } else layout(child, e.cw, k, cross[i], target[j])
458 cursor += spaces[j]
459 val main = if (reverse) length - cursor - target[j] - outer[i] else cursor
460 val crossPosition = lineStart + offset(lineCross - cross[i] - child.frame(!horizontal) - child.margin(!horizontal), align)
461 if (horizontal) e.place(child, main + child.edge("margin", 3), crossPosition + child.edge("margin", 0))
462 else e.place(child, crossPosition + child.edge("margin", 3), main + child.edge("margin", 0))
463 cursor += target[j] + outer[i] + gap
464 }
465 lineStart += lineCross + lineGap
466 }
467 return if (horizontal) {
468 if (lines.isEmpty()) 0 else lineStart - lineGap
469 } else length
470}

472private fun gridLayout(e: Element, k: Int?): Int {
473 val g = grid(e)
474 val colGap = e.n("column-gap")
475 val rowGap = e.n("row-gap")
476 val columns = tracks(g, true, e.cw, colGap, g.items.map(::contribution))
477 val widths = IntArray(g.items.size)
478 val areas = IntArray(g.items.size)
479 val contributions = g.items.indices.map {
480 i ->
481 val a = g.areas[i]
482 val child = g.items[i]
483 val aw = columns.subList(a.col - 1, a.col - 1 + a.cols).sum() + colGap * (a.cols - 1)
484 areas[i] = aw
485 widths[i] = child.clamp(child.resolve("width", aw, true) ?: maxOf(0, aw - child.margin(true) - child.frame(true)), aw, true)
486 layout(child, aw, null, widths[i])
487 child.borderHeight() + child.margin(false)
488 }
489 val rows = tracks(g, false, k, rowGap, contributions)
490 for (i in g.items.indices) {
491 val a = g.areas[i]
492 val child = g.items[i]
493 val align = e.align(child)
494 val ah = rows.subList(a.row - 1, a.row - 1 + a.rows).sum() + rowGap * (a.rows - 1)
495 val ax = columns.take(a.col - 1).sum() + colGap * (a.col - 1)
496 val ay = rows.take(a.row - 1).sum() + rowGap * (a.row - 1)
497 val fh = if (align == "stretch" && child.s("height") == "auto") child.clamp(maxOf(0, ah - child.margin(false) - child.frame(false)), ah, false) else null
498 layout(child, areas[i], ah, widths[i], fh)
499 val dy = offset(ah - child.borderHeight() - child.margin(false), align)
500 e.place(child, ax + child.edge("margin", 3), ay + dy + child.edge("margin", 0))
501 }
502 return rows.sum() + rowGap * maxOf(0, g.nr - 1)
503}

505private class Document(var width: Int, var height: Int, val rules: List<Rule>, val elements: List<Element>) {
506 val byId = elements.associateBy {
507 it.id
508 }
509 fun containing(e: Element): Element? {
510 var p = e.parent
511 while (p != null) {
512 if (p.s("position") != "static") return p
513 p = p.parent
514 }
515 return null
516 }
517 fun reflow() {
518 for (e in elements) {
519 e.style = initial.toMutableMap()
520 val specificity = mutableMapOf<String, Int>()
521 for (rule in rules) {
522 val best = rule.selectors.mapNotNull {
523 s -> when {
524 s.startsWith('#') -> if (s.drop(1) == e.id) 2 else null
525 s.startsWith('.') -> if (s.drop(1) in e.classes) 1 else null
526 else -> if (s == e.type) 0 else null
527 }
528 }.maxOrNull() ?: continue
529 for ((p, v) in rule.values) if (best >= (specificity[p] ?: -1)) {
530 e.style[p] = v
531 specificity[p] = best
532 }
533 }
534 e.style.putAll(e.overrides)
535 e.displayed = e.s("display") != "none" && (e.parent?.displayed ?: true)
536 }
537 val root = elements[0]
538 if (root.displayed && root.s("position") != "absolute") {
539 layout(root, width, height)
540 root.rx = root.edge("margin", 3)
541 root.ry = root.edge("margin", 0)
542 }
543 for (e in elements) if (e.displayed && e.s("position") == "absolute") {
544 val block = containing(e)
545 val cbw = block?.let {
546 it.cw + it.edge("padding", 1) + it.edge("padding", 3)
547 }?: width
548 val cbh = block?.let {
549 it.ch + it.edge("padding", 0) + it.edge("padding", 2)
550 }?: height
551 fun side(s: String) = e.s(s).let {
552 if (it == "auto") null else it.toInt()
553 }
554 val left = side("left")
555 val right = side("right")
556 val top = side("top")
557 val bottom = side("bottom")
558 val r = maxOf(0, cbw - (left ?: 0) - (right ?: 0) - e.margin(true) - e.frame(true))
559 val w = e.resolve("width", cbw, true) ?: if (left != null && right != null) r else minOf(maxContent(e), r)
560 val fw = e.clamp(w, cbw, true)
561 val fh = if (e.s("height") == "auto" && top != null && bottom != null) e.clamp(maxOf(0, cbh - top - bottom - e.margin(false) - e.frame(false)), cbh, false) else null
562 layout(e, cbw, cbh, fw, fh)
563 e.rx = if (left != null) left + e.edge("margin", 3) else if (right != null) cbw - right - e.edge("margin", 1) - e.borderWidth() else e.edge("margin", 3)
564 e.ry = if (top != null) top + e.edge("margin", 0) else if (bottom != null) cbh - bottom - e.edge("margin", 2) - e.borderHeight() else e.edge("margin", 0)
565 }
566 for (e in elements) if (e.displayed) {
567 val ref = if (e.s("position") == "absolute") containing(e) else e.parent
568 val bx = if (e.s("position") == "absolute") ref?.let {
569 it.x + it.edge("border", 3)
570 }?: 0 else ref?.x ?: 0
571 val by = if (e.s("position") == "absolute") ref?.let {
572 it.y + it.edge("border", 0)
573 }?: 0 else ref?.y ?: 0
574 var dx = 0
575 var dy = 0
576 if (e.s("position") == "relative") {
577 dx = if (e.s("left") != "auto") e.n("left") else if (e.s("right") != "auto") -e.n("right") else 0
578 dy = if (e.s("top") != "auto") e.n("top") else if (e.s("bottom") != "auto") -e.n("bottom") else 0
579 }
580 e.x = bx + e.rx + dx
581 e.y = by + e.ry + dy
582 }
583 }
584 fun snapshot() = elements.map {
585 if (it.displayed) it.border() else null
586 }
587 fun changed(old: List<Rect?>): String = elements.indices.filter {
588 old[it] != if (elements[it].displayed) elements[it].border() else null
589 }.joinToString(" ") {
590 elements[it].id
591 }.ifEmpty {
592 "no change"
593 }
594 fun render(): List<String> {
595 val screen = Array(height) {
596 CharArray(width) {
597 '.'
598 }
599 }
600 for (e in elements) if (e.displayed) {
601 val b = e.border()
602 for (y in maxOf(0, b.y) until minOf(height, b.y + b.h)) for (x in maxOf(0, b.x) until minOf(width, b.x + b.w)) {
603 val horizontal = e.edge("border", 0) == 1 && y == b.y || e.edge("border", 2) == 1 && y == b.y + b.h - 1
604 val vertical = e.edge("border", 3) == 1 && x == b.x || e.edge("border", 1) == 1 && x == b.x + b.w - 1
605 screen[y][x] = if (horizontal && vertical) '+' else if (horizontal) '-' else if (vertical) '|' else ' '
606 }
607 if (e.text != null) {
608 val c = e.content()
609 for ((i, line) in e.lines.withIndex()) for ((j, ch) in line.withIndex()) {
610 val x = c.x + j
611 val y = c.y + i
612 if (x in 0 until width && y in 0 until height) screen[y][x] = ch
613 }
614 }
615 }
616 val frame = "#".repeat(width + 2)
617 return listOf("render ${width}x$height", frame) + screen.map {
618 "#${it.concatToString()}#"
619 }+ frame
620 }
621 fun command(line: String): List<String> {
622 val a = line.split(' ')
623 return when (a[0]) {
624 "LAYOUT" -> listOf("layout ${width}x$height") + elements.map {
625 " ".repeat(it.depth) + it.id + if (it.displayed) " ${it.border()}" else " none"
626 }
627 "BOX" -> {
628 val e = byId[a[1]] ?: return listOf("error: unknown element ${a[1]}")
629 if (!e.displayed) listOf("box ${e.id}: none") else listOf("box ${e.id}", "margin ${e.marginBox()}", "border ${e.border()}", "padding ${e.padding()}", "content ${e.content()}")
630 }
631 "RENDER" -> render()
632 "HIT" -> {
633 val x = a[1].toInt()
634 val y = a[2].toInt()
635 if (x >= width || y >= height) listOf("error: point $x $y outside viewport")
636 else {
637 var e = elements.lastOrNull {
638 it.displayed && it.border().contains(x, y)
639 }
640 val path = mutableListOf<String>()
641 while (e != null) {
642 path.add(e.id)
643 e = e.parent
644 }
645 listOf("hit $x $y: " + if (path.isEmpty()) "none" else path.reversed().joinToString(" > "))
646 }
647 }
648 "SET" -> {
649 val id = a[1]
650 val p = a[2]
651 val value = a.drop(3).joinToString(" ")
652 val e = byId[id] ?: return listOf("error: unknown element $id")
653 if (p !in initial && p !in shorthands) return listOf("error: unknown property $p")
654 val values = declaration(p, value) ?: return listOf("error: invalid value for $p: $value")
655 val old = snapshot()
656 e.overrides.putAll(values)
657 reflow()
658 listOf("set $id $p $value: ${changed(old)}")
659 }
660 "VIEWPORT" -> {
661 val w = a[1].toInt()
662 val h = a[2].toInt()
663 if (w !in 1..80 || h !in 1..40) listOf("error: invalid viewport $w $h")
664 else {
665 val old = snapshot()
666 width = w
667 height = h
668 reflow()
669 listOf("viewport $w $h: ${changed(old)}")
670 }
671 }
672 "OVERFLOW" -> {
673 val ids = elements.filter {
674 e -> e.displayed && if (e.text != null) e.lines.any {
675 it.length > e.cw
676 } || e.lines.size > e.ch else e.flow().any {
677 !e.content().contains(it.marginBox())
678 }
679 }.joinToString(" ") {
680 it.id
681 }
682 listOf("overflow: " + ids.ifEmpty {
683 "none"
684 })
685 } else -> listOf("error: unknown command ${a[0]}")
686 }
687 }
688}

690fun main() {
691 val input = System.`in`.bufferedReader()
692 val size = input.readLine().split(' ').map(String::toInt)
693 val rules = mutableListOf<Rule>()
694 while (true) {
695 val line = input.readLine()
696 if (line == "---") break
697 val at = line.indexOf(" { ")
698 val selectors = line.take(at).split(", ")
699 val values = line.substring(at + 3, line.length - 2).split("; ").flatMap {
700 d -> val colon = d.indexOf(": ")
701 declaration(d.take(colon), d.substring(colon + 2)) ?: emptyList()
702 }
703 rules.add(Rule(selectors, values))
704 }
705 val elements = mutableListOf<Element>()
706 val ancestors = mutableListOf<Element>()
707 while (true) {
708 val line = input.readLine()
709 if (line == "===") break
710 val spaces = line.takeWhile {
711 it == ' '
712 }.length
713 val depth = spaces / 2
714 val body = line.drop(spaces)
715 val at = body.indexOf(' ')
716 val head = if (at < 0) body else body.take(at)
717 val text = if (at < 0) null else body.substring(at + 2, body.length - 1)
718 val hash = head.indexOf('#')
719 val parts = head.substring(hash + 1).split('.')
720 val e = Element(head.take(hash), parts[0], parts.drop(1).toSet(), text, if (depth == 0) null else ancestors[depth - 1], elements.size)
721 e.parent?.children?.add(e)
722 elements.add(e)
723 while (ancestors.size > depth) ancestors.removeAt(ancestors.lastIndex)
724 ancestors.add(e)
725 }
726 val document = Document(size[0], size[1], rules, elements)
727 document.reflow()
728 val output = StringBuilder()
729 while (true) {
730 val line = input.readLine() ?: break
731 if (line == "END") break
732 for (result in document.command(line)) output.append(result).append('\n')
733 }
734 print(output)
735}
