1def num(n) = [:n, n]
2def txt(s) = [:s, s]
3def err(s) = [:e, s]
4def ref(c, r, f = 0) = [:r, c, r, f]
5def idx(r) = r[2] * 26 + r[1]
6def label(i) = (65 + i % 26).chr + (i / 26 + 1).to_s
7def inside(r) = (0...26).cover?(r[1]) && (0...99).cover?(r[2])
8def area_pair(a, b)
9 c, r, f = a[1..]
10 d, t, g = b[1..]
11 ac, ar, bc, br = f & 1, f & 2, g & 1, g & 2
12 if c > d
13 c, d, ac, bc = d, c, bc, ac
14 end
15
16 if r > t
17 r, t, ar, br = t, r, br, ar
18 end
19
20 [:g, ref(c, r, ac | ar), ref(d, t, bc | br)]
21end

23def address(s)
24 raise "bad address" unless /\A[A-Z][1-9][0-9]?\z/.match?(s)
25 ref(s.getbyte(0) - 65, s[1..].to_i - 1)
26end

28def area(s)
29 parts = s.split(":", -1)
30 raise "bad address" if parts.size > 2
31 a = address(parts[0])
32 b = parts.size == 1 ? a : address(parts[1])
33 area_pair(a, b)
34end

36def cells(g)
37 (g[1][2]..g[2][2]).flat_map { |r| (g[1][1]..g[2][1]).map { |c| r * 26 + c } }
38end

40def contains(g, r) = (g[1][1]..g[2][1]).cover?(r[1]) && (g[1][2]..g[2][2]).cover?(r[2])
41LEVELS = [%w[= <> < <= > >=], ["&"], %w[+ -], %w[* /]].freeze
42def prec(op) = LEVELS.index { |xs| xs.include?(op) }

44class Parser
45 def syntax(p) = raise "syntax error at #{p}"
46 def initialize(s)
47 @tokens = []
48 i = 1
49 while i < s.size
50 p = i + 1
51 ch = s[i]
52 if ch == " "
53 i += 1
54 next
55 end
56
57 if /[0-9]/.match?(ch)
58 word = s[i..][/\A[0-9]+/]
59 syntax(p) if word.size >= 10
60 @tokens << [num(word.to_i), p]
61 i += word.size
62 elsif ch == "\""
63 j = s.index("\"", i + 1)
64 syntax(p) unless j
65 @tokens << [txt(s[i + 1...j]), p]
66 i = j + 1
67 elsif /[A-Za-z$]/.match?(ch)
68 word = s[i..][/\A[A-Za-z0-9$]+/].upcase
69 i += word.size
70 if (m = /\A(\$?)([A-Z])(\$?)([1-9][0-9]?)\z/.match(word))
71 x = ref(m[2].ord - 65, m[4].to_i - 1, (m[1].empty? ? 0 : 1) | (m[3].empty? ? 0 : 2))
72 elsif /\A[A-Z][A-Z0-9]*\z/.match?(word)
73 x = [s[i..].lstrip.start_with?("(") ? :c : :w, word]
74 else
75 syntax(p)
76 end
77
78 @tokens << [x, p]
79 else
80 op = s[i, 2]
81 op = ch unless %w[<> <= >=].include?(op)
82 syntax(p) unless "+-*/&(),:=<>".include?(ch)
83 @tokens << [[:t, op], p]
84 i += op.size
85 end
86 end
87
88 @tokens << [[:end], s.size + 1]
89 @at = 0
90 end

92 def take(op)
93 if @tokens[@at][0] == [:t, op]
94 @at += 1
95 true
96 else
97 false
98 end
99 end
100
101 def expect(op)
102 syntax(@tokens[@at][1]) unless take(op)
103 end
104
105 def expression(level = 0)
106 return primary if level == 4
107 a = expression(level + 1)
108 while @tokens[@at][0][0] == :t && LEVELS[level].include?(@tokens[@at][0][1])
109 op = @tokens[@at][0][1]
110 @at += 1
111 a = [:b, op, a, expression(level + 1)]
112 end
113
114 a
115 end
116
117 def primary
118 return [:u, primary] if take("-")
119 x, p = @tokens[@at]
120 @at += 1
121 case x[0]
122 when :n, :s, :w
123 x
124 when :r
125 return x unless take(":")
126 syntax(@tokens[@at][1]) unless @tokens[@at][0][0] == :r
127 y = @tokens[@at][0]
128 @at += 1
129 area_pair(x, y)
130 when :c
131 expect("(")
132 args = []
133 unless take(")")
134 args << expression
135 args << expression while take(",")
136 expect(")")
137 end
138
139 [:f, x[1], args]
140 when :t
141 syntax(p) unless x[1] == "("
142 a = expression
143 expect(")")
144 a
145 else
146 syntax(p)
147 end
148 end
149
150 def parse
151 tree = expression
152 syntax(@tokens[@at][1]) unless @tokens[@at][0][0] == :end
153 tree
154 end
155end

157def canonical(x)
158 case x[0]
159 when :n
160 x[1].to_s
161 when :s
162 "\"" + x[1] + "\""
163 when :w
164 x[1]
165 when :x
166 "#REF!"
167 when :r
168 (x[3] & 1 == 0 ? "" : "$") + (65 + x[1]).chr + (x[3] & 2 == 0 ? "" : "$") + (x[2] + 1).to_s
169 when :g
170 canonical(x[1]) + ":" + canonical(x[2])
171 when :f
172 x[1] + "(" + x[2].map { |a| canonical(a) }.join(",") + ")"
173 when :u
174 "-" + (x[1][0] == :b ? "(" + canonical(x[1]) + ")" : canonical(x[1]))
175 when :b
176 a, b = canonical(x[2]), canonical(x[3])
177 a = "(" + a + ")" if x[2][0] == :b && prec(x[2][1]) < prec(x[1])
178 b = "(" + b + ")" if x[3][0] == :b && prec(x[3][1]) <= prec(x[1])
179 a + x[1] + b
180 end
181end

183def display(x) = x.nil? ? "" : x[1].to_s
184def notation(x) = x.nil? ? "(empty)" : x[0] == :"=" ? "=" + canonical(x[1]) : x[0] == :s ? "\"" + x[1] + "\"" : display(
185 x
186)
187def to_number(x) = x.nil? ? num(0) : %i[s g].include?(x[0]) ? err("#VALUE!") : x
188def to_string(x) = x.nil? ? txt("") : x[0] == :n ? txt(x[1].to_s) : x[0] == :g ? err("#VALUE!") : x
189def quotient(a, b) = (a < 0) == (b < 0) ? a.abs / b.abs : -(a.abs / b.abs)
190def rewrite(x, &f)
191 return nil unless x
192 case x[0]
193 when :r, :g
194 f.call(x)
195 when :f
196 [:f, x[1], x[2].map { |a| rewrite(a, &f) }]
197 when :b
198 [:b, x[1], rewrite(x[2], &f), rewrite(x[3], &f)]
199 when :u, :"="
200 [x[0], rewrite(x[1], &f)]
201 else
202 x
203 end
204end
205
206def dependencies(x, out = {})
207 return out unless x
208 case x[0]
209 when :r
210 out[idx(x)] = true
211 when :g
212 cells(x).each { |i| out[i] = true }
213 when :f
214 x[2].each { |a| dependencies(a, out) }
215 when :b
216 dependencies(x[2], out)
217 dependencies(x[3], out)
218 when :u, :"="
219 dependencies(x[1], out)
220 end
221
222 out
223end

225class Sheet
226 def initialize
227 @contents = Array.new(2574)
228 @values = @contents.dup
229 @deps = Array.new(2574) { {} }
230 @history = []
231 end

233 def evaluate(x)
234 case x[0]
235 when :n, :s, :g
236 x
237 when :r
238 @values[idx(x)]
239 when :w
240 err("#NAME?")
241 when :x
242 err("#REF!")
243 when :u
244 a = to_number(evaluate(x[1]))
245 a[0] == :e ? a : num(-a[1])
246 when :b
247 a, b = evaluate(x[2]), evaluate(x[3])
248 return a if a&.first == :e
249 return b if b&.first == :e
250 op = x[1]
251 if prec(op) == 0
252 return err("#VALUE!") if a&.first == :g || b&.first == :g
253 blank = a&.first == :s || b&.first == :s ? txt("") : num(0)
254 a ||= blank
255 b ||= blank
256 c = a[0] != b[0] ? (a[0] == :n ? -1 : 1) : a[1] <=> b[1]
257 return num(
258 (
259 case op
260 when "="
261 c == 0
262 when "<>"
263 c != 0
264 when "<"
265 c < 0
266 when "<="
267 c <= 0
268 when ">"
269 c > 0
270 else
271 c >= 0
272 end
273 ) ? 1 : 0
274 )
275 end
276
277 a, b = op == "&" ? [to_string(a), to_string(b)] : [to_number(a), to_number(b)]
278 return a if a[0] == :e
279 return b if b[0] == :e
280 case op
281 when "&"
282 txt(a[1] + b[1])
283 when "+"
284 num(a[1] + b[1])
285 when "-"
286 num(a[1] - b[1])
287 when "*"
288 num(a[1] * b[1])
289 else
290 b[1] == 0 ? err("#DIV/0!") : num(quotient(a[1], b[1]))
291 end
292
293 when :f
294 call(x[1], x[2])
295 end
296 end

298 def call(name, args)
299 if %w[SUM MIN MAX COUNT AVERAGE AND OR CONCAT].include?(name)
300 return err("#VALUE!") if args.empty?
301 items = args.flat_map { |a|
302 v = evaluate(a)
303 v&.first == :g ? cells(v).map { |i| @values[i] } : [v]
304 }
305 ns = items.select { |a| a&.first == :n }.map { |a| a[1] }
306 return num(ns.size) if name == "COUNT"
307 error = items.find { |a| a&.first == :e }
308 return error if error
309 case name
310 when "SUM"
311 num(ns.sum)
312 when "MIN"
313 num(ns.min || 0)
314 when "MAX"
315 num(ns.max || 0)
316 when "AVERAGE"
317 ns.empty? ? err("#DIV/0!") : num(quotient(ns.sum, ns.size))
318 when "AND"
319 ns.empty? ? err("#VALUE!") : num(ns.all? { |n| n != 0 } ? 1 : 0)
320 when "OR"
321 ns.empty? ? err("#VALUE!") : num(ns.any? { |n| n != 0 } ? 1 : 0)
322 else
323 txt(items.map { |a| to_string(a)[1] }.join)
324 end
325 elsif name == "IF"
326 return err("#VALUE!") unless [2, 3].include?(args.size)
327 c = to_number(evaluate(args[0]))
328 return c if c[0] == :e
329 c[1] != 0 ? evaluate(args[1]) : args.size == 3 ? evaluate(args[2]) : num(0)
330 elsif %w[ABS NOT LEN].include?(name)
331 return err("#VALUE!") unless args.size == 1
332 v = evaluate(args[0])
333 v = name == "LEN" ? to_string(v) : to_number(v)
334 return v if v[0] == :e
335 num(name == "LEN" ? v[1].size : name == "ABS" ? v[1].abs : v[1] == 0 ? 1 : 0)
336 else
337 err("#NAME?")
338 end
339 end

341 def recalculate
342 @deps = @contents.map { |x| x&.first == :"=" ? dependencies(x[1]) : {} }
343 @values = @contents.map { |x| x&.first == :"=" ? nil : x }
344 indices = {}
345 low = {}
346 stack = []
347 on = {}
348 cyclic = {}
349 visit = lambda do |i|
350 indices[i] = low[i] = indices.size
351 stack << i
352 on[i] = true
353 @deps[i].each_key do |j|
354 next unless @contents[j]&.first == :"="
355 if !indices.key?(j)
356 visit.call(j)
357 low[i] = [low[i], low[j]].min
358 elsif on[j]
359 low[i] = [low[i], indices[j]].min
360 end
361 end
362
363 if low[i] == indices[i]
364 part = []
365 loop {
366 j = stack.pop
367 on.delete(j)
368 part << j
369 break if j == i
370 }
371 part.each { |j| cyclic[j] = true } if part.size > 1 || @deps[i].key?(i)
372 end
373 end
374
375 @contents.each_index { |i| visit.call(i) if @contents[i]&.first == :"=" && !indices.key?(i) }
376 done = {}
377 calculate = lambda do |i|
378 return unless @contents[i]&.first == :"=" && !done[i]
379 done[i] = true
380 if cyclic[i]
381 @values[i] = err("#CYCLE!")
382 return
383 end
384
385 @deps[i].each_key { |j| calculate.call(j) }
386 v = evaluate(@contents[i][1])
387 @values[i] = v.nil? ? num(0) : v[0] == :g ? err("#VALUE!") : v
388 end
389
390 @contents.each_index { |i| calculate.call(i) }
391 end

393 def replace(next_contents)
394 old = @values
395 @contents = next_contents
396 recalculate
397 old.each_index.count { |i| old[i] != @values[i] }
398 end

400 def describe(i) = "#{label(i)}: #{notation(@contents[i])} -> #{notation(@values[i])}"
401 def execute(line)
402 name, rest = line.split(" ", 2)
403 case name
404 when "SHOW"
405 return [describe(idx(address(rest)))]
406 when "LIST"
407 out = @contents.each_index.filter_map { |i| describe(i) if @contents[i] }
408 return out.empty? ? ["(no cells)"] : out
409 when "DEPS"
410 i = idx(address(rest))
411 format = -> (xs) { xs.empty? ? "none" : xs.sort.map { |j| label(j) }.join(",") }
412 return [
413 "#{rest}: precedents=#{format.call(@deps[i].keys)} dependents=#{format.call(@deps.each_index.select { |j| @deps[j].key?(i) })}"
414 ]
415 when "PRINT"
416 g = area(rest)
417 rs = g[1][2]..g[2][2]
418 cs = g[1][1]..g[2][1]
419 rw = (g[2][2] + 1).to_s.size
420 ws = cs.map { |c| [1, *rs.map { |r| display(@values[r * 26 + c]).size }].max }
421 sep = "+" + "-" * rw + "+" + ws.map { |w| "-" * w + "+" }.join
422 out = [sep, "|" + " " * rw + "|" + cs.each_with_index.map { |c, j| (65 + c).chr.ljust(ws[j]) + "|" }.join, sep]
423 rs.each { |r|
424 out <<
425 "|" +
426 (r + 1).to_s.rjust(rw) +
427 "|" +
428 cs
429 .each_with_index
430 .map { |c, j|
431 v = @values[r * 26 + c]
432 (v&.first == :n ? display(v).rjust(ws[j]) : display(v).ljust(ws[j])) + "|"
433 }
434 .join
435 }
436 return out << sep
437 when "UNDO"
438 raise "nothing to undo" if @history.empty?
439 command, old = @history.pop
440 return ["UNDO #{command} changed=#{replace(old)}"]
441 end
442
443 next_contents = @contents.dup
444 target = nil
445 case name
446 when "SET"
447 a, s = rest.split(" ", 2)
448 target = idx(address(a))
449 next_contents[target] = if s.start_with?("=")
450 [:"=", Parser.new(s).parse]
451 elsif /\A"[^"]*"\z/.match?(s)
452 txt(s[1...-1])
453 elsif /\A(?:0|-?[1-9][0-9]{0,8})\z/.match?(s)
454 num(s.to_i)
455 else
456 raise "bad value"
457 end
458
459 when "CLEAR"
460 cells(area(rest)).each { |i| next_contents[i] = nil }
461 when "COPY", "MOVE"
462 a, b = rest.split(" ")
463 source = area(a)
464 dest = address(b)
465 dc = dest[1] - source[1][1]
466 dr = dest[2] - source[1][2]
467 shift = -> (r, rel) { ref(r[1] + (rel && r[3] & 1 != 0 ? 0 : dc), r[2] + (rel && r[3] & 2 != 0 ? 0 : dr), r[3]) }
468 destination = area_pair(shift.call(source[1], false), shift.call(source[2], false))
469 raise "out of grid" unless inside(destination[1]) && inside(destination[2])
470 transform = lambda do |x|
471 if x[0] == :r
472 if name == "COPY"
473 r = shift.call(x, true)
474 inside(r) ? r : [:x]
475 else
476 contains(source, x) ? shift.call(x, false) : contains(destination, x) ? [:x] : x
477 end
478 elsif name == "MOVE" && !(contains(source, x[1]) && contains(source, x[2]))
479 x
480 else
481 a = shift.call(x[1], name == "COPY")
482 b = shift.call(x[2], name == "COPY")
483 inside(a) && inside(b) ? area_pair(a, b) : [:x]
484 end
485 end
486
487 next_contents.map! { |x| rewrite(x, &transform) } if name == "MOVE"
488 moving = cells(source).map { |i| [i, name == "COPY" ? rewrite(@contents[i], &transform) : next_contents[i]] }
489 moving.each { |i, _| next_contents[i] = nil } if name == "MOVE"
490 moving.each { |i, x| next_contents[i + dr * 26 + dc] = x }
491 when "INSERTROW", "DELETEROW", "INSERTCOL", "DELETECOL"
492 row = name.end_with?("ROW")
493 insert = name.start_with?("INSERT")
494 raise "bad address" unless (row ? /\A[1-9][0-9]?\z/ : /\A[A-Z]\z/).match?(rest)
495 at = row ? rest.to_i - 1 : rest.ord - 65
496 limit = row ? 99 : 26
497 axis = -> (x) { x[row ? 2 : 1] }
498 change = -> (x, v) { row ? ref(x[1], v, x[3]) : ref(v, x[2], x[3]) }
499 if insert && @contents.each_index.any? { |i| @contents[i] && (row ? i / 26 : i % 26) == limit - 1 }
500 raise "last #{row ? "row" : "column"} not empty"
501 end
502
503 transform = lambda do |x|
504 if x[0] == :r
505 v = axis.call(x)
506 if !insert && v == at
507 [:x]
508 else
509 n = v + (insert ? (v >= at ? 1 : 0) : (v > at ? -1 : 0))
510 n >= limit ? [:x] : change.call(x, n)
511 end
512 else
513 a = axis.call(x[1])
514 b = axis.call(x[2])
515 u = a + (insert ? (a >= at ? 1 : 0) : (a > at ? -1 : 0))
516 v = b + (insert ? (b >= at ? 1 : 0) : (b >= at ? -1 : 0))
517 (!insert && a == at && b == at) || v >= limit ? [:x] : area_pair(change.call(x[1], u), change.call(x[2], v))
518 end
519 end
520
521 next_contents = Array.new(2574)
522 @contents.each_index do |i|
523 next unless @contents[i]
524 r = ref(i % 26, i / 26)
525 v = axis.call(r)
526 next if !insert && v == at
527 t = change.call(r, v + (insert ? (v >= at ? 1 : 0) : (v > at ? -1 : 0)))
528 next_contents[idx(t)] = rewrite(@contents[i], &transform)
529 end
530 else
531 raise "unknown command"
532 end
533
534 @history << [name, @contents]
535 changed = replace(next_contents)
536 [name + (name == "SET" ? " #{label(target)}=#{notation(@values[target])}" : "") + " changed=#{changed}"]
537 end
538end
