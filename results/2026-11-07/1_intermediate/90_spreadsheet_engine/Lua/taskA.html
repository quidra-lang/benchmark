14local function ref(c, r, f)
15 return {"R", c, r, f or 0}

17local function id(r)
18 return r[3] * 26 + r[2] + 1
19end
20local function label(i)
21 i = i - 1
22 return string.char(65 + i % 26) .. (i // 26 + 1)
23end
24local function inside(r)
25 return r[2] >= 0 and r[2] < 26 and r[3] >= 0 and r[3] < 99
26end
27local function range(a, b)
28 local c, r, f, d, t, g = a[2], a[3], a[4], b[2], b[3], b[4]
29 local ac, ar, bc, br = f & 1, f & 2, g & 1, g & 2
30 if c > d then
31 c, d, ac, bc = d, c, bc, ac
32 end
33 if r > t then
34 r, t, ar, br = t, r, br, ar
35 end
36 return {"G", ref(c, r, ac | ar), ref(d, t, bc | br)}
37end
38local function address(s)
39 local c, r = s:match("^([A-Z])([1-9]%d?)$")
40 if not c then
41 fail("bad address")
42 end
43 return ref(c:byte() - 65, tonumber(r) - 1)
44end
45local function area(s)
46 local p = s:find(":", 1, true)
47 if not p then
48 local a = address(s)
49 return range(a, a)
50 end
51 local a = address(s:sub(1, p - 1))
52 local b = address(s:sub(p + 1))
53 return range(a, b)
54end
55local function cells(g)
56 local out = {}
57 for r = g[2][3], g[3][3] do
58 for c = g[2][2], g[3][2] do
59 out[#out + 1] = r * 26 + c + 1
60 end
61 end
62 return out

64local function contains(g, r)
65 return r[2] >= g[2][2] and r[2] <= g[3][2] and r[3] >= g[2][3] and r[3] <= g[3][3]
66end
67local levels = {
68 {["="] = true, ["<>"] = true, ["<"] = true, ["<="] = true, [">"] = true, [">="] = true},
69 {["&"] = true},
70 {["+"] = true, ["-"] = true},
71 {["*"] = true, ["/"] = true}
72}
73local function prec(op)
74 for i, x in ipairs(levels) do
75 if x[op] then
76 return i
77 end
78 end
79 return 0

81local function parse(s)
82 local ts, i = {}, 2
83 local function syntax(p)
84 fail("syntax error at " .. p)
85 end
86 while i <= #s do
87 local p, ch = i, s:sub(i, i)
88 if ch == " " then
89 i = i + 1
90 elseif ch:match("%d") then
91 local w = s:sub(i):match("^%d+")
92 if #w >= 10 then
93 syntax(p)
94 end
95 ts[#ts + 1] = {num(tonumber(w)), p}
96 i = i + #w
97 elseif ch == '"' then
98 local j = s:find('"', i + 1, true)
99 if not j then
100 syntax(p)
101 end
102 ts[#ts + 1] = {text(s:sub(i + 1, j - 1)), p}
103 i = j + 1
104 elseif ch:match("[A-Za-z$]") then
105 local w = s:sub(i):match("^[A-Za-z0-9$]+"):upper()
106 i = i + #w
107 local ac, c, ar, r = w:match("^(%$?)([A-Z])(%$?)([1-9]%d?)$")
108 local x
109 if c then
110 x =
111 ref(
112 c:byte() - 65,
113 tonumber(r) - 1,
114 (ac ~= "" and 1 or 0) | (ar ~= "" and 2 or 0)
115 )
116 elseif w:match("^[A-Z][A-Z0-9]*$") then
117 x = {s:sub(i):match("^ *%(") and "C" or "W", w}
118 else
119 syntax(p)
120 end
121 ts[#ts + 1] = {x, p}
122 else
123 local op = s:sub(i, i + 1)
124 if op ~= "<>" and op ~= "<=" and op ~= ">=" then
125 op = ch
126 end
127 if not ("+-*/&(),:=<>"):find(ch, 1, true) then
128 syntax(p)
129 end
130 ts[#ts + 1] = {{"T", op}, p}
131 i = i + #op
132 end
133 end
134 ts[#ts + 1] = {{"END"}, #s + 1}
135 i = 1
136 local function take(op)
137 if ts[i][1][1] == "T" and ts[i][1][2] == op then
138 i = i + 1
139 return true
140 end
141 return false
142 end
143 local function expect(op)
144 if not take(op) then
145 syntax(ts[i][2])
146 end
147 end
148 local expression, primary
149 expression = function(level)
150 level = level or 1
151 if level == 5 then
152 return primary()
153 end
154 local a = expression(level + 1)
155 while ts[i][1][1] == "T" and levels[level][ts[i][1][2]] do
156 local op = ts[i][1][2]
157 i = i + 1
158 a = {"B", op, a, expression(level + 1)}
159 end
160 return a
161 end
162 primary = function()
163 if take("-") then
164 return {"U", primary()}
165 end
166 local x, p = ts[i][1], ts[i][2]
167 i = i + 1
168 if x[1] == "N" or x[1] == "S" or x[1] == "W" then
169 return x
170 end
171 if x[1] == "R" then
172 if not take(":") then
173 return x
174 end
175 if ts[i][1][1] ~= "R" then
176 syntax(ts[i][2])
177 end
178 local b = ts[i][1]
179 i = i + 1
180 return range(x, b)
181 end
182 if x[1] == "C" then
183 expect("(")
184 local args = {}
185 if not take(")") then
186 args[#args + 1] = expression()
187 while take(",") do
188 args[#args + 1] = expression()
189 end
190 expect(")")
191 end
192 return {"F", x[2], args}
193 end
194 if x[1] == "T" and x[2] == "(" then
195 local a = expression()
196 expect(")")
197 return a
198 end
199 syntax(p)
200 end
201 local tree = expression()
202 if ts[i][1][1] ~= "END" then
203 syntax(ts[i][2])
204 end
205 return tree
206end
207local canonical
208canonical = function(x)
209 local k = x[1]
210 if k == "N" then
211 return tostring(x[2])
212 elseif k == "S" then
213 return '"' .. x[2] .. '"'
214 elseif k == "W" then
215 return x[2]
216 elseif k == "X" then
217 return "#REF!"
218 elseif k == "R" then
219 return (x[4] & 1 ~= 0 and "$" or "") ..
220 string.char(65 + x[2]) .. (x[4] & 2 ~= 0 and "$" or "") .. (x[3] + 1)
221 elseif k == "G" then
222 return canonical(x[2]) .. ":" .. canonical(x[3])
223 elseif k == "F" then
224 local a = {}
225 for i, v in ipairs(x[3]) do
226 a[i] = canonical(v)
227 end
228 return x[2] .. "(" .. table.concat(a, ",") .. ")"
229 elseif k == "U" then
230 local a = canonical(x[2])
231 return "-" .. (x[2][1] == "B" and "(" .. a .. ")" or a)
232 else
233 local a, b = canonical(x[3]), canonical(x[4])
234 if x[3][1] == "B" and prec(x[3][2]) < prec(x[2]) then
235 a = "(" .. a .. ")"
236 end
237 if x[4][1] == "B" and prec(x[4][2]) <= prec(x[2]) then
238 b = "(" .. b .. ")"
239 end
240 return a .. x[2] .. b
241 end
242end
243local function display(x)
244 return x == EMPTY and "" or tostring(x[2])
245end
246local function notation(x)
247 if x == EMPTY then
248 return "(empty)"
249 elseif x[1] == "=" then
250 return "=" .. canonical(x[2])
251 elseif x[1] == "S" then
252 return '"' .. x[2] .. '"'
253 else
254 return display(x)
255 end
256end
257local function number(x)
258 if x == EMPTY then
259 return num(0)
260 elseif x[1] == "S" or x[1] == "G" then
261 return err("#VALUE!")
262 else
263 return x
264 end
265end
266local function string_value(x)
267 if x == EMPTY then
268 return text("")
269 elseif x[1] == "N" then
270 return text(tostring(x[2]))
271 elseif x[1] == "G" then
272 return err("#VALUE!")
273 else
274 return x
275 end
276end
277local function quotient(a, b)
278 local q = math.abs(a) // math.abs(b)
279 return (a < 0) == (b < 0) and q or -q
280end
281local function rewrite(x, f)
282 local k = x[1]
283 if k == "R" or k == "G" then
284 return f(x)
285 elseif k == "F" then
286 local a = {}
287 for i, v in ipairs(x[3]) do
288 a[i] = rewrite(v, f)
289 end
290 return {"F", x[2], a}
291 elseif k == "B" then
292 return {"B", x[2], rewrite(x[3], f), rewrite(x[4], f)}
293 elseif k == "U" or k == "=" then
294 return {k, rewrite(x[2], f)}
295 else
296 return x
297 end
298end
299local function dependencies(x, out)
300 out = out or {}
301 local k = x[1]
302 if k == "R" then
303 out[id(x)] = true
304 elseif k == "G" then
305 for _, i in ipairs(cells(x)) do
306 out[i] = true
307 end
308 elseif k == "F" then
309 for _, a in ipairs(x[3]) do
310 dependencies(a, out)
311 end
312 elseif k == "B" then
313 dependencies(x[3], out)
314 dependencies(x[4], out)
315 elseif k == "U" or k == "=" then
316 dependencies(x[2], out)
317 end
318 return out
319end

331local evaluate, call
332local aggregate = {
333 SUM = true,
334 MIN = true,
335 MAX = true,
336 COUNT = true,
337 AVERAGE = true,
338 AND = true,
339 OR = true,
340 CONCAT = true
341}
342evaluate = function(x)
343 local k = x[1]
344 if k == "N" or k == "S" or k == "G" then
345 return x
346 elseif k == "R" then
347 return values[id(x)]
348 elseif k == "W" then
349 return err("#NAME?")
350 elseif k == "X" then
351 return err("#REF!")
352 elseif k == "U" then
353 local a = number(evaluate(x[2]))
354 return a[1] == "E" and a or num(-a[2])
355 elseif k == "F" then
356 return call(x[2], x[3])
357 end
358 local a, b = evaluate(x[3]), evaluate(x[4])
359 local op = x[2]
360 if a[1] == "E" then
361 return a
362 elseif b[1] == "E" then
363 return b
364 end
365 if prec(op) == 1 then
366 if a[1] == "G" or b[1] == "G" then
367 return err("#VALUE!")
368 end
369 local blank = (a[1] == "S" or b[1] == "S") and text("") or num(0)
370 if a == EMPTY then
371 a = blank
372 end
373 if b == EMPTY then
374 b = blank
375 end
376 local c
377 if a[1] ~= b[1] then
378 c = a[1] == "N" and -1 or 1
379 elseif a[2] == b[2] then
380 c = 0
381 elseif a[2] < b[2] then
382 c = -1
383 else
384 c = 1
385 end
386 local yes =
387 (op == "=" and c == 0) or (op == "<>" and c ~= 0) or (op == "<" and c < 0) or
388 (op == "<=" and c <= 0) or
389 (op == ">" and c > 0) or
390 (op == ">=" and c >= 0)
391 return num(yes and 1 or 0)
392 end
393 a = op == "&" and string_value(a) or number(a)
394 b = op == "&" and string_value(b) or number(b)
395 if a[1] == "E" then
396 return a
397 elseif b[1] == "E" then
398 return b
399 end
400 if op == "&" then
401 return text(a[2] .. b[2])
402 elseif op == "+" then
403 return num(a[2] + b[2])
404 elseif op == "-" then
405 return num(a[2] - b[2])
406 elseif op == "*" then
407 return num(a[2] * b[2])
408 end
409 return b[2] == 0 and err("#DIV/0!") or num(quotient(a[2], b[2]))
410end
411call = function(name, args)
412 if aggregate[name] then
413 if #args == 0 then
414 return err("#VALUE!")
415 end
416 local items, ns = {}, {}
417 for _, a in ipairs(args) do
418 local v = evaluate(a)
419 if v[1] == "G" then
420 for _, i in ipairs(cells(v)) do
421 items[#items + 1] = values[i]
422 end
423 else
424 items[#items + 1] = v
425 end
426 end
427 for _, v in ipairs(items) do
428 if v[1] == "N" then
429 ns[#ns + 1] = v[2]
430 end
431 end
432 if name == "COUNT" then
433 return num(#ns)
434 end
435 for _, v in ipairs(items) do
436 if v[1] == "E" then
437 return v
438 end
439 end
440 if name == "CONCAT" then
441 local a = {}
442 for i, v in ipairs(items) do
443 a[i] = string_value(v)[2]
444 end
445 return text(table.concat(a))
446 end
447 local sum, lo, hi, all, any = 0, ns[1] or 0, ns[1] or 0, true, false
448 for _, n in ipairs(ns) do
449 sum = sum + n
450 lo = math.min(lo, n)
451 hi = math.max(hi, n)
452 all = all and n ~= 0
453 any = any or n ~= 0
454 end
455 if name == "SUM" then
456 return num(sum)
457 elseif name == "MIN" then
458 return num(lo)
459 elseif name == "MAX" then
460 return num(hi)
461 end
462 if #ns == 0 then
463 return err(name == "AVERAGE" and "#DIV/0!" or "#VALUE!")
464 end
465 if name == "AVERAGE" then
466 return num(quotient(sum, #ns))
467 end
468 local yes
469 if name == "AND" then
470 yes = all
471 else
472 yes = any
473 end
474 return num(yes and 1 or 0)
475 end
476 if name == "IF" then
477 if #args ~= 2 and #args ~= 3 then
478 return err("#VALUE!")
479 end
480 local c = number(evaluate(args[1]))
481 if c[1] == "E" then
482 return c
483 end
484 if c[2] ~= 0 then
485 return evaluate(args[2])
486 elseif #args == 3 then
487 return evaluate(args[3])
488 else
489 return num(0)
490 end
491 end
492 if name ~= "ABS" and name ~= "NOT" and name ~= "LEN" then
493 return err("#NAME?")
494 end
495 if #args ~= 1 then
496 return err("#VALUE!")
497 end
498 local v = evaluate(args[1])
499 v = name == "LEN" and string_value(v) or number(v)
500 if v[1] == "E" then
501 return v
502 end
503 return num(
504 name == "LEN" and #v[2] or name == "ABS" and math.abs(v[2]) or (v[2] == 0 and 1 or 0)
505 )
506end
507local function recalculate()
508 local indices, low, stack, on, cyclic, serial = {}, {}, {}, {}, {}, 0
509 for i = 1, 2574 do
510 deps[i] = contents[i][1] == "=" and dependencies(contents[i][2]) or {}
511 values[i] = contents[i][1] == "=" and EMPTY or contents[i]
512 end
513 local visit
514 visit = function(i)
515 indices[i] = serial
516 low[i] = serial
517 serial = serial + 1
518 stack[#stack + 1] = i
519 on[i] = true
520 for j in pairs(deps[i]) do
521 if contents[j][1] == "=" then
522 if not indices[j] then
523 visit(j)
524 low[i] = math.min(low[i], low[j])
525 elseif on[j] then
526 low[i] = math.min(low[i], indices[j])
527 end
528 end
529 end
530 if low[i] == indices[i] then
531 local part, j = {}, nil
532 repeat
533 j = table.remove(stack)
534 on[j] = nil
535 part[#part + 1] = j
536 until j == i
537 if #part > 1 or deps[i][i] then
538 for _, v in ipairs(part) do
539 cyclic[v] = true
540 end
541 end
542 end
543 end
544 for i = 1, 2574 do
545 if contents[i][1] == "=" and not indices[i] then
546 visit(i)
547 end
548 end
549 local done, calculate = {}
550 calculate = function(i)
551 if contents[i][1] ~= "=" or done[i] then
552 return
553 end
554 done[i] = true
555 if cyclic[i] then
556 values[i] = err("#CYCLE!")
557 return
558 end
559 for j in pairs(deps[i]) do
560 calculate(j)
561 end
562 local v = evaluate(contents[i][2])
563 values[i] = v == EMPTY and num(0) or v[1] == "G" and err("#VALUE!") or v
564 end
565 for i = 1, 2574 do
566 calculate(i)
567 end
568end
569local function replace(next_grid)
570 local old = values
571 values = grid()
572 contents = next_grid
573 recalculate()
574 local count = 0
575 for i = 1, 2574 do
576 if notation(old[i]) ~= notation(values[i]) then
577 count = count + 1
578 end
579 end
580 return count
581end

583 return label(i) .. ": " .. notation(contents[i]) .. " -> " .. notation(values[i])
584end
585local function format(xs)
586 local a = {}
587 for i in pairs(xs) do
588 a[#a + 1] = i
589 end
590 table.sort(a)
591 for j, i in ipairs(a) do
592 a[j] = label(i)

595end
596local function execute(line)
597 local name, rest = line:match("^(%S+) ?(.*)$")
598 if name == "SHOW" then
599 return {describe(id(address(rest)))}
600 end
601 if name == "LIST" then
602 local out = {}
603 for i = 1, 2574 do
604 if contents[i] ~= EMPTY then
605 out[#out + 1] = describe(i)
606 end
607 end
608 return #out == 0 and {"(no cells)"} or out
609 end
610 if name == "DEPS" then
611 local i = id(address(rest))
612 local ds = {}
613 for j = 1, 2574 do
614 if deps[j][i] then
615 ds[j] = true
616 end
617 end
618 return {rest .. ": precedents=" .. format(deps[i]) .. " dependents=" .. format(ds)}
619 end
620 if name == "PRINT" then
621 local g = area(rest)
622 local rw = #tostring(g[3][3] + 1)
623 local ws = {}
624 local sep = "+" .. string.rep("-", rw) .. "+"
625 local head = "|" .. string.rep(" ", rw) .. "|"
626 for c = g[2][2], g[3][2] do
627 local w = 1
628 for r = g[2][3], g[3][3] do
629 w = math.max(w, #display(values[r * 26 + c + 1]))
630 end
631 ws[#ws + 1] = w
632 sep = sep .. string.rep("-", w) .. "+"
633 head = head .. string.char(65 + c) .. string.rep(" ", w - 1) .. "|"
634 end
635 local out = {sep, head, sep}
636 for r = g[2][3], g[3][3] do
637 local s = "|" .. string.format("%" .. rw .. "d", r + 1) .. "|"
638 for j, w in ipairs(ws) do
639 local v = values[r * 26 + g[2][2] + j]
640 local t = display(v)
641 local pad = string.rep(" ", w - #t)
642 s = s .. (v[1] == "N" and pad .. t or t .. pad) .. "|"
643 end
644 out[#out + 1] = s
645 end
646 out[#out + 1] = sep
647 return out
648 end
649 if name == "UNDO" then
650 if #history == 0 then
651 fail("nothing to undo")
652 end
653 local e = table.remove(history)
654 return {"UNDO " .. e[1] .. " changed=" .. replace(e[2])}
655 end
656 local next_grid, target = {}, nil
657 for i = 1, 2574 do
658 next_grid[i] = contents[i]
659 end
660 if name == "SET" then
661 local a, s = rest:match("^(%S+) (.*)$")
662 target = id(address(a))
663 if s:sub(1, 1) == "=" then
664 next_grid[target] = {"=", parse(s)}
665 elseif s:match('^"[^"]*"$') then
666 next_grid[target] = text(s:sub(2, -2))
667 elseif s == "0" or (s:match("^%-?[1-9]%d*$") and #(s:gsub("^-", "")) <= 9) then
668 next_grid[target] = num(tonumber(s))
669 else
670 fail("bad value")
671 end
672 elseif name == "CLEAR" then
673 for _, i in ipairs(cells(area(rest))) do
674 next_grid[i] = EMPTY
675 end
676 elseif name == "COPY" or name == "MOVE" then
677 local a, b = rest:match("^(%S+) (%S+)$")
678 local source = area(a)
679 local dest = address(b)
680 local dc, dr = dest[2] - source[2][2], dest[3] - source[2][3]
681 local function shift(r, rel)
682 return ref(
683 r[2] + (rel and r[4] & 1 ~= 0 and 0 or dc),
684 r[3] + (rel and r[4] & 2 ~= 0 and 0 or dr),
685 r[4]
686 )
687 end
688 local destination = range(shift(source[2], false), shift(source[3], false))
689 if not inside(destination[2]) or not inside(destination[3]) then
690 fail("out of grid")
691 end
692 local function transform(x)
693 if x[1] == "R" then
694 if name == "COPY" then
695 local r = shift(x, true)
696 return inside(r) and r or INVALID
697 end
698 return contains(source, x) and shift(x, false) or
699 contains(destination, x) and INVALID or
700 x
701 end
702 if name == "MOVE" and not (contains(source, x[2]) and contains(source, x[3])) then
703 return x
704 end
705 local u, v = shift(x[2], name == "COPY"), shift(x[3], name == "COPY")
706 return inside(u) and inside(v) and range(u, v) or INVALID
707 end
708 if name == "MOVE" then
709 for i = 1, 2574 do
710 next_grid[i] = rewrite(next_grid[i], transform)
711 end
712 end
713 local moving = {}
714 for _, i in ipairs(cells(source)) do
715 moving[#moving + 1] = {
716 i,
717 name == "COPY" and rewrite(contents[i], transform) or next_grid[i]
718 }
719 end
720 if name == "MOVE" then
721 for _, e in ipairs(moving) do
722 next_grid[e[1]] = EMPTY
723 end
724 end
725 for _, e in ipairs(moving) do
726 next_grid[e[1] + dr * 26 + dc] = e[2]
727 end
728 elseif name == "INSERTROW" or name == "DELETEROW" or name == "INSERTCOL" or name == "DELETECOL" then
729 local row, insert = name:sub(-3) == "ROW", name:sub(1, 6) == "INSERT"
730 if not (row and rest:match("^[1-9]%d?$") or not row and rest:match("^[A-Z]$")) then
731 fail("bad address")
732 end
733 local at = row and tonumber(rest) - 1 or rest:byte() - 65
734 local limit = row and 99 or 26
735 local function axis(x)
736 return x[row and 3 or 2]
737 end
738 local function change(x, v)
739 return row and ref(x[2], v, x[4]) or ref(v, x[3], x[4])
740 end
741 if insert then
742 for i = 1, 2574 do
743 if contents[i] ~= EMPTY and (row and (i - 1) // 26 or (i - 1) % 26) == limit - 1 then
744 fail("last " .. (row and "row" or "column") .. " not empty")
745 end
