117static Node *range(Node *a, Node *b) {
118 int c = a->n, d = b->n, r = a->row, t = b->row, ac = a->flags & 1, bc = b->flags & 1,
119 ar = a->flags & 2, br = b->flags & 2, z;
120 if (c > d) {
121 z = c;
122 c = d;
123 d = z;
124 z = ac;
125 ac = bc;
126 bc = z;
127 }
128 if (r > t) {
129 z = r;
130 r = t;
131 t = z;
132 z = ar;
133 ar = br;
134 br = z;
135 }
136 return node('G', 0, "", reference(c, r, ac | ar), reference(d, t, bc | br));
137}
138static bool row_number(const char *s) {
139 size_t n = strlen(s);
140 return (n == 1 || n == 2) && s[0] >= '1' && s[0] <= '9' &&
141 (n == 1 || (s[1] >= '0' && s[1] <= '9'));
142}

162static int precedence(const char *s) {
163 if (!strcmp(s, "&"))
164 return 1;
165 if (!strcmp(s, "+") || !strcmp(s, "-"))
166 return 2;
167 if (!strcmp(s, "*") || !strcmp(s, "/"))
168 return 3;
169 if (!strcmp(s, "=") || !strcmp(s, "<>") || !strcmp(s, "<") || !strcmp(s, "<=") ||
170 !strcmp(s, ">") || !strcmp(s, ">="))
171 return 0;
172 return -1;
173}
174static Node *word_reference(const char *w) {
175 int i = 0, f = 0;
176 if (w[i] == '$') {
177 f |= 1;
178 i++;
179 }
180 if (w[i] < 'A' || w[i] > 'Z')
181 return NULL;
182 int c = w[i++] - 'A';
183 if (w[i] == '$') {
184 f |= 2;
185 i++;
186 }
187 if (!row_number(w + i))
188 return NULL;
189 return reference(c, atoi(w + i) - 1, f);
190}
191static bool letter(char c) {
192 return (c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z');
193}
194static void tokenize(Parser *p, const char *s) {
195 size_t n = strlen(s);
196 for (size_t i = 1; i < n;) {
197 int pos = (int)i + 1;
198 char ch = s[i];
199 if (ch == ' ') {
200 i++;
201 continue;
202 }
203 Node *x;
204 if (isdigit((unsigned char)ch)) {
205 size_t j = i;
206 while (j < n && isdigit((unsigned char)s[j]))
207 j++;
208 if (j - i >= 10)
209 syntax(pos);
210 char *w = copy_text(s + i, j - i);
211 x = num(atoi(w));
212 free(w);
213 i = j;
214 } else if (ch == '"') {
215 const char *end = strchr(s + i + 1, '"');
216 if (!end)
217 syntax(pos);
218 char *w = copy_text(s + i + 1, (size_t)(end - s - i - 1));
219 x = text(w);
220 free(w);
221 i = (size_t)(end - s) + 1;
222 } else if (letter(ch) || ch == '$') {
223 size_t j = i;
224 while (j < n && (letter(s[j]) || isdigit((unsigned char)s[j]) || s[j] == '$'))
225 j++;
226 char *w = copy_text(s + i, j - i);
227 for (char *q = w; *q; q++)
228 *q = (char)toupper((unsigned char)*q);
229 i = j;
230 x = word_reference(w);
231 if (!x) {
232 bool valid = w[0] >= 'A' && w[0] <= 'Z';
233 for (char *q = w; *q; q++)
234 if (!(*q >= 'A' && *q <= 'Z') && !isdigit((unsigned char)*q))
235 valid = false;
236 if (!valid)
237 syntax(pos);
238 while (j < n && s[j] == ' ')
239 j++;
240 x = node(j < n && s[j] == '(' ? 'C' : 'W', 0, w, NULL, NULL);
241 }
242 free(w);
243 } else {
244 char op[3] = {ch, 0, 0};
245 if ((ch == '<' && (s[i + 1] == '>' || s[i + 1] == '=')) ||
246 (ch == '>' && s[i + 1] == '='))
247 op[1] = s[i + 1];
248 if (!strchr("+-*/&(),:=<>", ch))
249 syntax(pos);
250 x = node('T', 0, op, NULL, NULL);
251 i += strlen(op);
252 }
253 p->tokens[p->size++] = (Token){x, pos};
254 }
255 p->tokens[p->size++] = (Token){node('Z', 0, "", NULL, NULL), (int)n + 1};
256}
257static bool take(Parser *p, const char *op) {
258 Node *x = p->tokens[p->at].value;
259 if (x->kind == 'T' && !strcmp(x->text, op)) {
260 p->at++;
261 return true;
262 }
263 return false;
264}
265static void expect(Parser *p, const char *op) {
266 if (!take(p, op))
267 syntax(p->tokens[p->at].position);
268}
269static Node *expression(Parser *p, int level);
270static Node *primary(Parser *p) {
271 if (take(p, "-"))
272 return node('U', 0, "", primary(p), NULL);
273 Token t = p->tokens[p->at++];
274 Node *x = t.value;
275 if (x->kind == 'N' || x->kind == 'S' || x->kind == 'W')
276 return x;
277 if (x->kind == 'R') {
278 if (!take(p, ":"))
279 return x;
280 if (p->tokens[p->at].value->kind != 'R')
281 syntax(p->tokens[p->at].position);
282 return range(x, p->tokens[p->at++].value);
283 }
284 if (x->kind == 'C') {
285 expect(p, "(");
286 Node *call = node('F', 0, x->text, NULL, NULL);
287 Node *args[101];
288 int count = 0;
289 if (!take(p, ")")) {
290 args[count++] = expression(p, 0);
291 while (take(p, ","))
292 args[count++] = expression(p, 0);
293 expect(p, ")");
294 }
295 call->argc = count;
296 call->args = allocate((size_t)count * sizeof *call->args);
297 memcpy(call->args, args, (size_t)count * sizeof *args);
298 return call;
299 }
300 if (x->kind == 'T' && !strcmp(x->text, "(")) {
301 Node *a = expression(p, 0);
302 expect(p, ")");
303 return a;
304 }
305 syntax(t.position);
306 return NULL;
307}
308static Node *expression(Parser *p, int level) {
309 if (level == 4)
310 return primary(p);
311 Node *a = expression(p, level + 1);
312 while (p->tokens[p->at].value->kind == 'T' &&
313 precedence(p->tokens[p->at].value->text) == level) {
314 char *op = p->tokens[p->at++].value->text;
315 Node *b = expression(p, level + 1);
316 a = node('B', 0, op, a, b);
317 }
318 return a;
319}
320static Node *parse_formula(const char *s) {
321 Parser p = {0};
322 tokenize(&p, s);
323 Node *x = expression(&p, 0);
324 if (p.tokens[p.at].value->kind != 'Z')
325 syntax(p.tokens[p.at].position);
326 return x;
327}
328static void canonical(Node *x, char *out) {
329 char s[40];
330 switch (x->kind) {
331 case 'N':
332 sprintf(s, "%d", x->n);
333 strcat(out, s);
334 break;
335 case 'S':
336 strcat(out, "\"");
337 strcat(out, x->text);
338 strcat(out, "\"");
339 break;
340 case 'W':
341 strcat(out, x->text);
342 break;
343 case 'X':
344 strcat(out, "#REF!");
345 break;
346 case 'R':
347 sprintf(s, "%s%c%s%d", x->flags & 1 ? "$" : "", 'A' + x->n, x->flags & 2 ? "$" : "",
348 x->row + 1);
349 strcat(out, s);
350 break;
351 case 'G':
352 canonical(x->a, out);
353 strcat(out, ":");
354 canonical(x->b, out);
355 break;
356 case 'F':
357 strcat(out, x->text);
358 strcat(out, "(");
359 for (int i = 0; i < x->argc; i++) {
360 if (i)
361 strcat(out, ",");
362 canonical(x->args[i], out);
363 }
364 strcat(out, ")");
365 break;
366 case 'U':
367 strcat(out, "-");
368 if (x->a->kind == 'B')
369 strcat(out, "(");
370 canonical(x->a, out);
371 if (x->a->kind == 'B')
372 strcat(out, ")");
373 break;
374 case 'B': {
375 bool a = x->a->kind == 'B' && precedence(x->a->text) < precedence(x->text),
376 b = x->b->kind == 'B' && precedence(x->b->text) <= precedence(x->text);
377 if (a)
378 strcat(out, "(");
379 canonical(x->a, out);
380 if (a)
381 strcat(out, ")");
382 strcat(out, x->text);
383 if (b)
384 strcat(out, "(");
385 canonical(x->b, out);
386 if (b)
387 strcat(out, ")");
388 break;
389 }
390 }
391}

425typedef Node *(*Transform)(Node *, void *);
426static Node *rewrite(Node *x, Transform f, void *context) {
427 if (!x)
428 return NULL;
429 if (x->kind == 'R' || x->kind == 'G')
430 return f(x, context);
431 Node *a = rewrite(x->a, f, context), *b = rewrite(x->b, f, context);
432 bool changed = a != x->a || b != x->b;
433 Node **args = allocate((size_t)x->argc * sizeof *args);
434 for (int i = 0; i < x->argc; i++) {
435 args[i] = rewrite(x->args[i], f, context);
436 changed |= args[i] != x->args[i];
437 }
438 if (!changed) {
439 free(args);
440 return x;
441 }
442 Node *v = node(x->kind, x->n, x->text, a, b);
443 v->args = args;
444 v->argc = x->argc;
445 return v;
446}
447static void dependencies(Node *x, bool *out) {
448 if (!x)
449 return;
450 if (x->kind == 'R')
451 out[id(x)] = true;
452 else if (x->kind == 'G') {
453 for (int r = x->a->row; r <= x->b->row; r++)
454 for (int c = x->a->n; c <= x->b->n; c++)
455 out[r * 26 + c] = true;
456 } else {
457 dependencies(x->a, out);
458 dependencies(x->b, out);
459 for (int i = 0; i < x->argc; i++)
460 dependencies(x->args[i], out);
461 }
462}
463static Node *evaluate(Sheet *s, Node *x);
464static Node *aggregate(Sheet *s, Node *x) {
465 if (!x->argc)
466 return error_value("#VALUE!");
467 int count = 0, lo = 0, hi = 0;
468 long long sum = 0;
469 bool all = true, any = false;
470 char joined[101] = "";
471 for (int a = 0; a < x->argc; a++) {
472 Node *v = evaluate(s, x->args[a]);
473 int r1 = 0, r2 = 0, c1 = 0, c2 = 0;
474 if (v && v->kind == 'G') {
475 r1 = v->a->row;
476 r2 = v->b->row;
477 c1 = v->a->n;
478 c2 = v->b->n;
479 }
480 for (int r = r1; r <= r2; r++)
481 for (int c = c1; c <= c2; c++) {
482 Node *item = v && v->kind == 'G' ? s->values[r * 26 + c] : v;
483 if (item && item->kind == 'E') {
484 if (strcmp(x->text, "COUNT"))
485 return item;
486 continue;
487 }
488 if (item && item->kind == 'N') {
489 int n = item->n;
490 if (!count || n < lo)
491 lo = n;
492 if (!count || n > hi)
493 hi = n;
494 count++;
495 sum += n;
496 all = all && n != 0;
497 any = any || n != 0;
498 }
499 if (!strcmp(x->text, "CONCAT")) {
500 Node *t = to_string(item);
501 strcat(joined, t->text);
502 }
503 }
504 }
505 const char *name = x->text;
506 if (!strcmp(name, "COUNT"))
507 return num(count);
508 if (!strcmp(name, "CONCAT"))
509 return text(joined);
510 if (!strcmp(name, "SUM"))
511 return num((int)sum);
512 if (!strcmp(name, "MIN"))
513 return num(lo);
514 if (!strcmp(name, "MAX"))
515 return num(hi);
516 if (!count)
517 return error_value(!strcmp(name, "AVERAGE") ? "#DIV/0!" : "#VALUE!");
518 if (!strcmp(name, "AVERAGE"))
519 return num((int)(sum / count));
520 return num(!strcmp(name, "AND") ? all : any);
521}
522static Node *evaluate(Sheet *s, Node *x) {
523 if (x->kind == 'N' || x->kind == 'S' || x->kind == 'G')
524 return x;
525 if (x->kind == 'R')
526 return s->values[id(x)];
527 if (x->kind == 'W')
528 return error_value("#NAME?");
529 if (x->kind == 'X')
530 return error_value("#REF!");
531 if (x->kind == 'U') {
532 Node *a = to_number(evaluate(s, x->a));
533 return a->kind == 'E' ? a : num(-a->n);
534 }
535 if (x->kind == 'B') {
536 Node *a = evaluate(s, x->a);
537 Node *b = evaluate(s, x->b);
538 const char *op = x->text;
539 if (a && a->kind == 'E')
540 return a;
541 if (b && b->kind == 'E')
542 return b;
543 if (precedence(op) == 0) {
544 if ((a && a->kind == 'G') || (b && b->kind == 'G'))
545 return error_value("#VALUE!");
546 bool strings = (a && a->kind == 'S') || (b && b->kind == 'S');
547 if (!a)
548 a = strings ? text("") : num(0);
549 if (!b)
550 b = strings ? text("") : num(0);
551 int c = a->kind != b->kind ? (a->kind == 'N' ? -1 : 1)
552 : a->kind == 'N' ? (a->n > b->n) - (a->n < b->n)
553 : strcmp(a->text, b->text);
554 return num(!strcmp(op, "=") ? c == 0
555 : !strcmp(op, "<>") ? c != 0
556 : !strcmp(op, "<") ? c < 0
557 : !strcmp(op, "<=") ? c <= 0
558 : !strcmp(op, ">") ? c > 0
559 : c >= 0);
560 }
561 a = !strcmp(op, "&") ? to_string(a) : to_number(a);
562 b = !strcmp(op, "&") ? to_string(b) : to_number(b);
563 if (a->kind == 'E')
564 return a;
565 if (b->kind == 'E')
566 return b;
567 if (!strcmp(op, "&")) {
568 char joined[101];
569 strcpy(joined, a->text);
570 strcat(joined, b->text);
571 return text(joined);
572 }
573 if (!strcmp(op, "+"))
574 return num(a->n + b->n);
575 if (!strcmp(op, "-"))
576 return num(a->n - b->n);
577 if (!strcmp(op, "*"))
578 return num(a->n * b->n);
579 return b->n ? num(a->n / b->n) : error_value("#DIV/0!");
580 }
581 const char *name = x->text;
582 const char *names[] = {"SUM", "MIN", "MAX", "COUNT", "AVERAGE", "AND", "OR", "CONCAT"};
583 for (int i = 0; i < 8; i++)
584 if (!strcmp(name, names[i]))
585 return aggregate(s, x);
586 if (!strcmp(name, "IF")) {
587 if (x->argc != 2 && x->argc != 3)
588 return error_value("#VALUE!");
589 Node *c = to_number(evaluate(s, x->args[0]));
590 if (c->kind == 'E')
591 return c;
592 return c->n ? evaluate(s, x->args[1]) : x->argc == 3 ? evaluate(s, x->args[2]) : num(0);
593 }
594 if (strcmp(name, "ABS") && strcmp(name, "NOT") && strcmp(name, "LEN"))
595 return error_value("#NAME?");
596 if (x->argc != 1)
597 return error_value("#VALUE!");
598 Node *v = evaluate(s, x->args[0]);
599 v = !strcmp(name, "LEN") ? to_string(v) : to_number(v);
600 return v->kind == 'E' ? v
601 : num(!strcmp(name, "LEN") ? (int)strlen(v->text)
602 : !strcmp(name, "ABS") ? abs(v->n)
603 : v->n == 0);
604}
605static void visit(Sheet *s, int i) {
606 s->indices[i] = s->low[i] = s->serial++;
607 s->stack[s->stack_size++] = i;
608 s->on[i] = true;
609 for (int k = 0; k < s->counts[i]; k++) {
610 int j = s->deps[i][k];
611 if (s->contents[j] && s->contents[j]->kind == '=') {
612 if (s->indices[j] < 0) {
613 visit(s, j);
614 if (s->low[j] < s->low[i])
615 s->low[i] = s->low[j];
616 } else if (s->on[j] && s->indices[j] < s->low[i])
617 s->low[i] = s->indices[j];
618 }
619 }
620 if (s->low[i] == s->indices[i]) {
621 int part[200], count = 0, j;
622 do {
623 j = s->stack[--s->stack_size];
624 s->on[j] = false;
625 part[count++] = j;
626 } while (j != i);
627 bool self = false;
628 for (int k = 0; k < s->counts[i]; k++)
629 self |= s->deps[i][k] == i;
630 if (count > 1 || self)
631 for (int k = 0; k < count; k++)
632 s->cyclic[part[k]] = true;
633 }
634}
635static void calculate(Sheet *s, int i) {
636 if (!s->contents[i] || s->contents[i]->kind != '=' || s->done[i])
637 return;
638 s->done[i] = true;
639 if (s->cyclic[i]) {
640 s->values[i] = error_value("#CYCLE!");
641 return;
642 }
643 for (int k = 0; k < s->counts[i]; k++)
644 calculate(s, s->deps[i][k]);
645 Node *v = evaluate(s, s->contents[i]->a);
646 s->values[i] = !v ? num(0) : v->kind == 'G' ? error_value("#VALUE!") : v;
647}
648static int replace(Sheet *s, Node **next) {
649 Allocation *old_allocations = value_allocations;
650 value_allocations = NULL;
651 Node *old[2574];
652 memcpy(old, s->values, sizeof old);
653 memcpy(s->contents, next, sizeof s->contents);
654 s->serial = s->stack_size = 0;
655 memset(s->on, 0, sizeof s->on);
656 memset(s->cyclic, 0, sizeof s->cyclic);
657 memset(s->done, 0, sizeof s->done);
658 for (int i = 0; i < 2574; i++) {
659 free(s->deps[i]);
660 s->deps[i] = NULL;
661 s->counts[i] = 0;
662 s->indices[i] = -1;
663 Node *x = s->contents[i];
664 if (x && x->kind == '=') {
665 bool found[2574] = {0};
666 dependencies(x->a, found);
667 for (int j = 0; j < 2574; j++)
668 if (found[j])
669 s->counts[i]++;
670 s->deps[i] = allocate((size_t)s->counts[i] * sizeof(int));
671 int k = 0;
672 for (int j = 0; j < 2574; j++)
673 if (found[j])
674 s->deps[i][k++] = j;
675 s->values[i] = NULL;
676 } else
677 s->values[i] = x;
678 }
679 for (int i = 0; i < 2574; i++)
680 if (s->contents[i] && s->contents[i]->kind == '=' && s->indices[i] < 0)
681 visit(s, i);
682 evaluating = true;
683 for (int i = 0; i < 2574; i++)
684 calculate(s, i);
685 evaluating = false;
686 int changed = 0;
687 for (int i = 0; i < 2574; i++) {
688 Node *a = old[i], *b = s->values[i];
689 if (!a || !b) {
690 changed += a != b;
691 continue;
692 }
693 changed +=
694 a->kind != b->kind || (a->kind == 'N' ? a->n != b->n : strcmp(a->text, b->text) != 0);
695 }
696 release_values(old_allocations);
697 return changed;
698}
699static void describe(Sheet *s, int i) {
700 char a[8], c[4096], v[104];
701 label(i, a);
702 notation(s->contents[i], c);
703 notation(s->values[i], v);
704 printf("%s: %s -> %s\n", a, c, v);
705}
706static void print_ids(bool *found) {
707 bool any = false;
708 for (int i = 0; i < 2574; i++)
709 if (found[i]) {
710 char a[8];
711 label(i, a);
712 printf("%s%s", any ? "," : "", a);
713 any = true;
714 }
715 if (!any)
716 printf("none");
717}
718typedef struct {
719 bool copy;
720 Node *source, *destination;
721 int dc, dr;
722} Transfer;
723static Node *shift(Node *r, Transfer *t, bool relative) {
724 return reference(r->n + (relative && (r->flags & 1) ? 0 : t->dc),
725 r->row + (relative && (r->flags & 2) ? 0 : t->dr), r->flags);
726}
727static Node *transfer_ref(Node *x, void *context) {
728 Transfer *t = context;
729 if (x->kind == 'R') {
730 if (t->copy) {
731 Node *r = shift(x, t, true);
732 return inside(r) ? r : invalid;
733 }
734 return contains(t->source, x) ? shift(x, t, false)
735 : contains(t->destination, x) ? invalid
736 : x;
737 }
738 if (!t->copy && !(contains(t->source, x->a) && contains(t->source, x->b)))
739 return x;
740 Node *a = shift(x->a, t, t->copy), *b = shift(x->b, t, t->copy);
741 return inside(a) && inside(b) ? range(a, b) : invalid;
742}
743typedef struct {
744 bool row, insert;
745 int at, limit;
746} Lines;
747static int axis(Node *x, Lines *t) {
748 return t->row ? x->row : x->n;
749}
750static Node *change(Node *x, int v, Lines *t) {
751 return reference(t->row ? x->n : v, t->row ? v : x->row, x->flags);
752}
753static Node *line_ref(Node *x, void *context) {
754 Lines *t = context;
755 if (x->kind == 'R') {
756 int v = axis(x, t);
757 if (!t->insert && v == t->at)
758 return invalid;
759 int n = v + (t->insert ? (v >= t->at) : -(v > t->at));
760 return n >= t->limit ? invalid : change(x, n, t);
761 }
762 int a = axis(x->a, t), b = axis(x->b, t);
763 if (!t->insert && a == t->at && b == t->at)
764 return invalid;
765 int u = a + (t->insert ? (a >= t->at) : -(a > t->at)),
766 v = b + (t->insert ? (b >= t->at) : -(b >= t->at));
767 return v >= t->limit ? invalid : range(change(x->a, u, t), change(x->b, v, t));
768}
769static bool valid_integer(const char *s) {
770 if (!strcmp(s, "0"))
771 return true;
772 size_t i = s[0] == '-' ? 1 : 0, n = strlen(s);
773 if (n <= i || n - i > 9 || s[i] < '1' || s[i] > '9')
774 return false;
775 for (i++; i < n; i++)
776 if (!isdigit((unsigned char)s[i]))
777 return false;
778 return true;
779}
780static void execute(Sheet *s, char *line) {
781 char *rest = strchr(line, ' ');
782 if (rest)
783 *rest++ = 0;
784 else
785 rest = line + strlen(line);
786 char *name = line;
787 if (!strcmp(name, "SHOW")) {
788 describe(s, id(address(rest)));
789 return;
790 }
791 if (!strcmp(name, "LIST")) {
792 bool any = false;
793 for (int i = 0; i < 2574; i++)
794 if (s->contents[i]) {
795 describe(s, i);
796 any = true;
797 }
798 if (!any)
799 puts("(no cells)");
800 return;
801 }
802 if (!strcmp(name, "DEPS")) {
803 int i = id(address(rest));
804 bool ps[2574] = {0}, ds[2574] = {0};
805 for (int k = 0; k < s->counts[i]; k++)
806 ps[s->deps[i][k]] = true;
807 for (int j = 0; j < 2574; j++)
808 for (int k = 0; k < s->counts[j]; k++)
809 if (s->deps[j][k] == i)
810 ds[j] = true;
811 printf("%s: precedents=", rest);
812 print_ids(ps);
813 printf(" dependents=");
814 print_ids(ds);
815 putchar('\n');
816 return;
817 }
818 if (!strcmp(name, "PRINT")) {
819 Node *g = area(rest);
820 int rw = g->b->row + 1 >= 10 ? 2 : 1, widths[26], cols = g->b->n - g->a->n + 1;
821 char sep[4096] = "+", head[4096] = "|", value[104];
822 for (int j = 0; j < rw; j++) {
823 strcat(sep, "-");
824 strcat(head, " ");
825 }
826 strcat(sep, "+");
827 strcat(head, "|");
828 for (int j = 0; j < cols; j++) {
829 int c = g->a->n + j, w = 1;
830 for (int r = g->a->row; r <= g->b->row; r++) {
831 display(s->values[r * 26 + c], value);
832 if ((int)strlen(value) > w)
833 w = (int)strlen(value);
834 }
835 widths[j] = w;
836 for (int k = 0; k < w; k++)
837 strcat(sep, "-");
838 strcat(sep, "+");
839 size_t len = strlen(head);
840 head[len] = (char)('A' + c);
841 head[len + 1] = 0;
842 for (int k = 1; k < w; k++)
843 strcat(head, " ");
844 strcat(head, "|");
845 }
846 puts(sep);
847 puts(head);
848 puts(sep);
849 for (int r = g->a->row; r <= g->b->row; r++) {
850 printf("|%*d|", rw, r + 1);
851 for (int j = 0; j < cols; j++) {
852 Node *v = s->values[r * 26 + g->a->n + j];
853 display(v, value);
854 if (v && v->kind == 'N')
855 printf("%*s|", widths[j], value);
856 else
857 printf("%-*s|", widths[j], value);
858 }
859 putchar('\n');
860 }
861 puts(sep);
862 return;
863 }
864 if (!strcmp(name, "UNDO")) {
865 if (!s->history_size)
866 fail("nothing to undo");
867 History h = s->history[--s->history_size];
868 int count = replace(s, h.contents);
869 free(h.contents);
870 printf("UNDO %s changed=%d\n", h.name, count);
871 return;
872 }
873 Node *next[2574];
874 memcpy(next, s->contents, sizeof next);
875 int target = -1;
876 if (!strcmp(name, "SET")) {
877 char *v = strchr(rest, ' ');
878 *v++ = 0;
879 target = id(address(rest));
880 size_t n = strlen(v);
881 if (v[0] == '=')
882 next[target] = node('=', 0, "", parse_formula(v), NULL);
883 else if (n >= 2 && v[0] == '"' && v[n - 1] == '"' && !memchr(v + 1, '"', n - 2)) {
884 char *w = copy_text(v + 1, n - 2);
885 next[target] = text(w);
886 free(w);
887 } else if (valid_integer(v))
888 next[target] = num(atoi(v));
889 else
890 fail("bad value");
891 } else if (!strcmp(name, "CLEAR")) {
892 Node *g = area(rest);
893 for (int r = g->a->row; r <= g->b->row; r++)
894 for (int c = g->a->n; c <= g->b->n; c++)
895 next[r * 26 + c] = NULL;
896 } else if (!strcmp(name, "COPY") || !strcmp(name, "MOVE")) {
897 char *b = strchr(rest, ' ');
898 *b++ = 0;
899 Node *source = area(rest), *dest = address(b);
900 Transfer t = {!strcmp(name, "COPY"), source, NULL, dest->n - source->a->n,
901 dest->row - source->a->row};
902 t.destination = range(shift(source->a, &t, false), shift(source->b, &t, false));
903 if (!inside(t.destination->a) || !inside(t.destination->b))
904 fail("out of grid");
905 if (!t.copy)
906 for (int i = 0; i < 2574; i++)
907 next[i] = rewrite(next[i], transfer_ref, &t);
908 Node *moving[2574];
909 memcpy(moving, next, sizeof moving);
910 if (!t.copy)
911 for (int r = source->a->row; r <= source->b->row; r++)
912 for (int c = source->a->n; c <= source->b->n; c++)
913 next[r * 26 + c] = NULL;
914 for (int r = source->a->row; r <= source->b->row; r++)
915 for (int c = source->a->n; c <= source->b->n; c++) {
916 int i = r * 26 + c;
917 next[i + t.dr * 26 + t.dc] =
918 t.copy ? rewrite(s->contents[i], transfer_ref, &t) : moving[i];
919 }
920 } else if (!strcmp(name, "INSERTROW") || !strcmp(name, "DELETEROW") ||
921 !strcmp(name, "INSERTCOL") || !strcmp(name, "DELETECOL")) {
922 bool row = strstr(name, "ROW") != NULL, insert = !strncmp(name, "INSERT", 6);
923 if (row ? !row_number(rest) : !(strlen(rest) == 1 && rest[0] >= 'A' && rest[0] <= 'Z'))
924 fail("bad address");
925 Lines t = {row, insert, row ? atoi(rest) - 1 : rest[0] - 'A', row ? 99 : 26};
926 if (insert)
927 for (int i = 0; i < 2574; i++)
928 if (s->contents[i] && (row ? i / 26 : i % 26) == t.limit - 1)
929 fail(row ? "last row not empty" : "last column not empty");
930 memset(next, 0, sizeof next);
931 for (int i = 0; i < 2574; i++)
932 if (s->contents[i]) {
933 Node *r = reference(i % 26, i / 26, 0);
934 int v = axis(r, &t);
935 if (!insert && v == t.at)
936 continue;
937 Node *u = change(r, v + (insert ? (v >= t.at) : -(v > t.at)), &t);
938 next[id(u)] = rewrite(s->contents[i], line_ref, &t);
939 }
940 } else
941 fail("unknown command");
942 History *h = &s->history[s->history_size++];
943 strcpy(h->name, name);
944 h->contents = allocate(sizeof s->contents);
945 memcpy(h->contents, s->contents, sizeof s->contents);
946 int count = replace(s, next);
947 printf("%s", name);
948 if (target >= 0) {
949 char a[8], v[104];
950 label(target, a);
951 notation(s->values[target], v);
952 printf(" %s=%s", a, v);
953 }
954 printf(" changed=%d\n", count);
955}
