21function make_range(a, b)
22 c, d, r, t=a.n, b.n, a.row, b.row
23 ac, bc, ar, br=a.flags&1, b.flags&1, a.flags&2, b.flags&2
24 if c>d
25 c, d, ac, bc=d, c, bc, ac
26 end
27 if r>t
28 r, t, ar, br=t, r, br, ar
29 end
30 node('G', 0, "", reference(c, r, ac|ar), reference(d, t, bc|br))
31end
32function address(s)
33 occursin(r"\A[A-Z][1-9][0-9]?\z", s) || throw(ArgumentError("bad address"))
34 reference(Int(s[1])-65, parse(Int, s[2:end])-1)
35end
36function area(s)
37 p=split(s, ':'; keepempty = true)
38 length(p)<=2 || throw(ArgumentError("bad address"))
39 a=address(p[1])
40 b=length(p)==1 ? a : address(p[2])
41 make_range(a, b)
42end
43cells(g) = [r*26+c+1 for r=g.a.row:g.b.row for c=g.a.n:g.b.n]
44contains_cell(g, r) = g.a.n<=r.n<=g.b.n && g.a.row<=r.row<=g.b.row
45const LEVELS =
46 [Set(["=", "<>", "<", "<=", ">", ">="]), Set(["&"]), Set(["+", "-"]), Set(["*", "/"])]
47precedence(op) = findfirst(xs->op in xs, LEVELS)
48function parse_formula(s)
49 tokens=Tuple{Node,Int}[]
50 i=2
51 syntax(p) = throw(ArgumentError("syntax error at $p"))
52 while i<=length(s)
53 p=i
54 ch=s[i]
55 if ch==' '
56 i+=1
57 continue
58 end
59 if isdigit(ch)
60 j=i
61 while j<=length(s)&&isdigit(s[j])
62 j+=1
63 end
64 j-i<10 || syntax(p)
65 push!(tokens, (num(parse(Int, s[i:(j-1)])), p))
66 i=j
67 elseif ch=='"'
68 j=findnext('"', s, i+1)
69 j===nothing && syntax(p)
70 push!(tokens, (text(s[(i+1):(j-1)]), p))
71 i=j+1
72 elseif isletter(ch)||ch=='\$'
73 j=i
74 while j<=length(s)&&(isletter(s[j])||isdigit(s[j])||s[j]=='\$')
75 j+=1
76 end
77 w=uppercase(s[i:(j-1)])
78 i=j
79 m=match(r"\A(\$?)([A-Z])(\$?)([1-9][0-9]?)\z", w)
80 if m!==nothing
81 x=reference(
82 Int(m[2][1])-65,
83 parse(Int, m[4])-1,
84 (isempty(m[1]) ? 0 : 1)|(isempty(m[3]) ? 0 : 2),
85 )
86 elseif occursin(r"\A[A-Z][A-Z0-9]*\z", w)
87 x=node(startswith(lstrip(s[j:end]), "(") ? 'C' : 'W', 0, w)
88 else
89 syntax(p)
90 end
91 push!(tokens, (x, p))
92 else
93 op=s[i:min(i+1, length(s))]
94 op in ("<>", "<=", ">=") || (op=string(ch))
95 ch in "+-*/&(),:=<>" || syntax(p)
96 push!(tokens, (node('T', 0, op), p))
97 i+=length(op)
98 end
99 end
100 push!(tokens, (node('Z'), length(s)+1))
101 i=1
102 function take(op)
103 if tokens[i][1].kind=='T'&&tokens[i][1].text==op
104 i+=1
105 return true
106 end
107 false
108 end
109 expect(op) = take(op) || syntax(tokens[i][2])
110 function expression(level = 1)
111 level==5 && return primary()
112 a=expression(level+1)
113 while tokens[i][1].kind=='T'&&tokens[i][1].text in LEVELS[level]
114 op=tokens[i][1].text
115 i+=1
116 a=node('B', 0, op, a, expression(level+1))
117 end
118 a
119 end
120 function primary()
121 take("-") && return node('U', 0, "", primary())
122 x, p=tokens[i]
123 i+=1
124 x.kind in ('N', 'S', 'W') && return x
125 if x.kind=='R'
126 take(":") || return x
127 tokens[i][1].kind=='R' || syntax(tokens[i][2])
128 b=tokens[i][1]
129 i+=1
130 return make_range(x, b)
131 elseif x.kind=='C'
132 expect("(")
133 args=Node[]
134 if !take(")")
135 push!(args, expression())
136 while take(",")
137 push!(args, expression())
138 end
139 expect(")")
140 end
141 return node('F', 0, x.text, nothing, nothing, args)
142 elseif x.kind=='T'&&x.text=="("
143 a=expression()
144 expect(")")
145 return a
146 end
147 syntax(p)
148 end
149 result=expression()
150 tokens[i][1].kind=='Z' || syntax(tokens[i][2])
151 result
152end
153function canonical(x)
154 k=x.kind
155 k=='N' && return string(x.n)
156 k=='S' && return "\""*x.text*"\""
157 k=='W' && return x.text
158 k=='X' && return "#REF!"
159 k=='R' &&
160 return (x.flags&1!=0 ? "\$" : "")*string(Char(65+x.n))*(x.flags&2!=0 ? "\$" : "")*string(
161 x.row+1,
162 )
163 k=='G' && return canonical(x.a)*":"*canonical(x.b)
164 k=='F' && return x.text*"("*join(canonical.(x.args), ",")*")"
165 k=='U' && return "-"*(x.a.kind=='B' ? "("*canonical(x.a)*")" : canonical(x.a))
166 a=canonical(x.a)
167 b=canonical(x.b)
168 x.a.kind=='B'&&precedence(x.a.text)<precedence(x.text) && (a="("*a*")")
169 x.b.kind=='B'&&precedence(x.b.text)<=precedence(x.text) && (b="("*b*")")
170 a*x.text*b
171end
172display_value(x) = x===nothing ? "" : x.kind=='N' ? string(x.n) : x.text
173notation(x) =
174 x===nothing ? "(empty)" :
175 x.kind=='=' ? "="*canonical(x.a) : x.kind=='S' ? "\""*x.text*"\"" : display_value(x)
176to_number(x) = x===nothing ? num(0) : x.kind in ('S', 'G') ? err("#VALUE!") : x
177to_text(x) =
178 x===nothing ? text("") : x.kind=='N' ? text(string(x.n)) : x.kind=='G' ? err("#VALUE!") : x
179function rewrite(x, f)
180 x===nothing && return nothing
181 x.kind in ('R', 'G') && return f(x)
182 a=rewrite(x.a, f)
183 b=rewrite(x.b, f)
184 args=Node[rewrite(v, f) for v in x.args]
185 a===x.a&&b===x.b&&all(args[j]===x.args[j] for j in eachindex(args)) ? x :
186 node(x.kind, x.n, x.text, a, b, args)
187end
188function dependencies(x, out = Set{Int}())
189 x===nothing && return out
190 if x.kind=='R'
191 push!(out, id(x))
192 elseif x.kind=='G'
193 union!(out, cells(x))
194 else
195 dependencies(x.a, out)
196 dependencies(x.b, out)
197 foreach(a->dependencies(a, out), x.args)
198 end
199 out
200end

210function evaluate(sheet, x)
211 k=x.kind
212 k in ('N', 'S', 'G') && return x
213 k=='R' && return sheet.values[id(x)]
214 k=='W' && return err("#NAME?")
215 k=='X' && return err("#REF!")
216 if k=='U'
217 a=to_number(evaluate(sheet, x.a))
218 return a.kind=='E' ? a : num(-a.n)
219 end
220 if k=='B'
221 a=evaluate(sheet, x.a)
222 b=evaluate(sheet, x.b)
223 op=x.text
224 a!==nothing&&a.kind=='E' && return a
225 b!==nothing&&b.kind=='E' && return b
226 if precedence(op)==1
227 ((a!==nothing&&a.kind=='G')||(b!==nothing&&b.kind=='G')) && return err("#VALUE!")
228 strings=(a!==nothing&&a.kind=='S')||(b!==nothing&&b.kind=='S')
229 blank=strings ? text("") : num(0)
230 a===nothing && (a=blank)
231 b===nothing && (b=blank)
232 c=a.kind!=b.kind ? (a.kind=='N' ? -1 : 1) :
233 a.kind=='N' ? cmp(a.n, b.n) : cmp(a.text, b.text)
234 yes=op=="=" ? c==0 :
235 op=="<>" ? c!=0 : op=="<" ? c<0 : op=="<=" ? c<=0 : op==">" ? c>0 : c>=0
236 return num(Int(yes))
237 end
238 convert=op=="&" ? to_text : to_number
239 a=convert(a)
240 b=convert(b)
241 a.kind=='E' && return a
242 b.kind=='E' && return b
243 return op=="&" ? text(a.text*b.text) :
244 op=="+" ? num(a.n+b.n) :
245 op=="-" ? num(a.n-b.n) :
246 op=="*" ? num(a.n*b.n) : b.n==0 ? err("#DIV/0!") : num(div(a.n, b.n))
247 end
248 name=x.text
249 args=x.args
250 if name in ("SUM", "MIN", "MAX", "COUNT", "AVERAGE", "AND", "OR", "CONCAT")
251 isempty(args) && return err("#VALUE!")
252 items=Union{Nothing,Node}[]
253 for a in args
254 v=evaluate(sheet, a)
255 if v!==nothing&&v.kind=='G'
256 append!(items, sheet.values[cells(v)])
257 else
258 push!(items, v)
259 end
260 end
261 ns=[v.n for v in items if v!==nothing&&v.kind=='N']
262 name=="COUNT" && return num(length(ns))
263 for v in items
264 v!==nothing&&v.kind=='E' && return v
265 end
266 name=="CONCAT" && return text(join(to_text(v).text for v in items))
267 name=="SUM" && return num(sum(ns))
268 name=="MIN" && return num(isempty(ns) ? 0 : minimum(ns))
269 name=="MAX" && return num(isempty(ns) ? 0 : maximum(ns))
270 isempty(ns) && return err(name=="AVERAGE" ? "#DIV/0!" : "#VALUE!")
271 name=="AVERAGE" && return num(div(sum(ns), length(ns)))
272 return num(Int(name=="AND" ? all(!=(0), ns) : any(!=(0), ns)))
273 elseif name=="IF"
274 length(args) in (2, 3) || return err("#VALUE!")
275 c=to_number(evaluate(sheet, args[1]))
276 c.kind=='E' && return c
277 return c.n!=0 ? evaluate(sheet, args[2]) :
278 length(args)==3 ? evaluate(sheet, args[3]) : num(0)
279 end
280 name in ("ABS", "NOT", "LEN") || return err("#NAME?")
281 length(args)==1 || return err("#VALUE!")
282 v=evaluate(sheet, args[1])
283 v=name=="LEN" ? to_text(v) : to_number(v)
284 v.kind=='E' ? v : num(name=="LEN" ? length(v.text) : name=="ABS" ? abs(v.n) : Int(v.n==0))
285end
286function recalculate!(sheet)
287 sheet.deps=[x!==nothing&&x.kind=='=' ? dependencies(x.a) : Set{Int}() for x in sheet.contents]
288 sheet.values=Union{Nothing,Node}[x!==nothing&&x.kind=='=' ? nothing : x for x in sheet.contents]
289 indices=fill(-1, 2574)
290 low=zeros(Int, 2574)
291 on=falses(2574)
292 cyclic=falses(2574)
293 done=falses(2574)
294 stack=Int[]
295 serial=0
296 function visit(i)
297 indices[i]=low[i]=serial
298 serial+=1
299 push!(stack, i)
300 on[i]=true
301 for j in sheet.deps[i]
302 if sheet.contents[j]!==nothing&&sheet.contents[j].kind=='='
303 if indices[j]<0
304 visit(j)
305 low[i]=min(low[i], low[j])
306 elseif on[j]
307 low[i]=min(low[i], indices[j])
308 end
309 end
310 end
311 if low[i]==indices[i]
312 part=Int[]
313 while true
314 j=pop!(stack)
315 on[j]=false
316 push!(part, j)
317 j==i && break
318 end
319 if length(part)>1||i in sheet.deps[i]
320 cyclic[part].=true
321 end
322 end
323 end
324 for i=1:2574
325 sheet.contents[i]!==nothing&&sheet.contents[i].kind=='='&&indices[i]<0 && visit(i)
326 end
327 function calculate(i)
328 (sheet.contents[i]===nothing||sheet.contents[i].kind!='='||done[i]) && return
329 done[i]=true
330 if cyclic[i]
331 sheet.values[i]=err("#CYCLE!")
332 return
333 end
334 foreach(calculate, sheet.deps[i])
335 v=evaluate(sheet, sheet.contents[i].a)
336 sheet.values[i]=v===nothing ? num(0) : v.kind=='G' ? err("#VALUE!") : v
337 end
338 foreach(calculate, 1:2574)
339end
340function replace_contents!(sheet, next)
341 old=sheet.values
342 sheet.contents=next
343 recalculate!(sheet)
344 count(i->notation(old[i])!=notation(sheet.values[i]), 1:2574)
345end
346describe(sheet, i) = label(i)*": "*notation(sheet.contents[i])*" -> "*notation(sheet.values[i])
347format_cells(xs) = isempty(xs) ? "none" : join(label.(sort!(collect(xs))), ",")
348function execute!(sheet, line)
349 parts=split(line, ' '; limit = 2)
350 name=parts[1]
351 rest=length(parts)==2 ? parts[2] : ""
352 name=="SHOW" && return [describe(sheet, id(address(rest)))]
353 if name=="LIST"
354 out=[describe(sheet, i) for i=1:2574 if sheet.contents[i]!==nothing]
355 return isempty(out) ? ["(no cells)"] : out
356 end
357 if name=="DEPS"
358 i=id(address(rest))
359 ds=[j for j=1:2574 if i in sheet.deps[j]]
360 return [rest*": precedents="*format_cells(sheet.deps[i])*" dependents="*format_cells(ds)]
361 end
362 if name=="PRINT"
363 g=area(rest)
364 rw=length(string(g.b.row+1))
365 cs=g.a.n:g.b.n
366 rs=g.a.row:g.b.row
367 ws=[max(1, maximum(length(display_value(sheet.values[r*26+c+1])) for r in rs)) for c in cs]
368 sep="+"*"-"^rw*"+"*join("-"^w*"+" for w in ws)
369 head="|"*" "^rw*"|"*join(rpad(string(Char(65+c)), w)*"|" for (c, w) in zip(cs, ws))
370 out=[sep, head, sep]
371 for r in rs
372 row="|"*lpad(string(r+1), rw)*"|"
373 for (c, w) in zip(cs, ws)
374 v=sheet.values[r*26+c+1]
375 row*=(
376 v!==nothing&&v.kind=='N' ? lpad(display_value(v), w) :
377 rpad(display_value(v), w)
378 )*"|"
379 end
380 push!(out, row)
381 end
382 push!(out, sep)
383 return out
384 end
385 if name=="UNDO"
386 isempty(sheet.history) && throw(ArgumentError("nothing to undo"))
387 command, old=pop!(sheet.history)
388 return ["UNDO "*command*" changed="*string(replace_contents!(sheet, old))]
389 end
390 next=copy(sheet.contents)
391 target=0
392 if name=="SET"
393 p=split(rest, ' '; limit = 2)
394 target=id(address(p[1]))
395 s=p[2]
396 if startswith(s, "=")
397 next[target]=node('=', 0, "", parse_formula(s))
398 elseif occursin(r"\A\"[^\"]*\"\z", s)
399 next[target]=text(s[2:(end-1)])
400 elseif occursin(r"\A(0|-?[1-9][0-9]{0,8})\z", s)
401 next[target]=num(parse(Int, s))
402 else
403 throw(ArgumentError("bad value"))
404 end
405 elseif name=="CLEAR"
406 next[cells(area(rest))].=nothing
407 elseif name in ("COPY", "MOVE")
408 p=split(rest, ' '; limit = 2)
409 source=area(p[1])
410 dest=address(p[2])
411 dc=dest.n-source.a.n
412 dr=dest.row-source.a.row
413 shift(r, relative) = reference(
414 r.n+(relative&&r.flags&1!=0 ? 0 : dc),
415 r.row+(relative&&r.flags&2!=0 ? 0 : dr),
416 r.flags,
417 )
418 destination=make_range(shift(source.a, false), shift(source.b, false))
419 inside(destination.a)&&inside(destination.b) || throw(ArgumentError("out of grid"))
420 function transform(x)
421 if x.kind=='R'
422 if name=="COPY"
423 r=shift(x, true)
424 return inside(r) ? r : INVALID
425 end
426 return contains_cell(source, x) ? shift(x, false) :
427 contains_cell(destination, x) ? INVALID : x
428 end
429 name=="MOVE"&&!(contains_cell(source, x.a)&&contains_cell(source, x.b)) && return x
430 a=shift(x.a, name=="COPY")
431 b=shift(x.b, name=="COPY")
432 inside(a)&&inside(b) ? make_range(a, b) : INVALID
433 end
434 name=="MOVE" && (next=Union{Nothing,Node}[rewrite(x, transform) for x in next])
435 ids=cells(source)
436 moving=[name=="COPY" ? rewrite(sheet.contents[i], transform) : next[i] for i in ids]
437 name=="MOVE" && (next[ids].=nothing)
438 for (i, x) in zip(ids, moving)
439 next[i+dr*26+dc]=x
440 end
441 elseif name in ("INSERTROW", "DELETEROW", "INSERTCOL", "DELETECOL")
442 row=endswith(name, "ROW")
443 insert=startswith(name, "INSERT")
444 occursin(row ? r"\A[1-9][0-9]?\z" : r"\A[A-Z]\z", rest) ||
445 throw(ArgumentError("bad address"))
446 at=row ? parse(Int, rest)-1 : Int(rest[1])-65
447 limit=row ? 99 : 26
448 axis(x) = row ? x.row : x.n
449 change(x, v) = reference(row ? x.n : v, row ? v : x.row, x.flags)
450 if insert&&any(
451 i->sheet.contents[i]!==nothing&&(row ? div(i-1, 26) : (i-1)%26)==limit-1,
452 1:2574,
453 )
454 throw(ArgumentError("last "*(row ? "row" : "column")*" not empty"))
455 end
456 function transform_axis(x)
457 if x.kind=='R'
458 v=axis(x)
459 !insert&&v==at && return INVALID
460 n=v+(insert ? Int(v>=at) : -Int(v>at))
461 return n>=limit ? INVALID : change(x, n)
462 end
463 a=axis(x.a)
464 b=axis(x.b)
465 !insert&&a==at&&b==at && return INVALID
466 u=a+(insert ? Int(a>=at) : -Int(a>at))
467 v=b+(insert ? Int(b>=at) : -Int(b>=at))
468 v>=limit ? INVALID : make_range(change(x.a, u), change(x.b, v))
469 end
470 next=grid()
471 for i=1:2574
472 sheet.contents[i]===nothing && continue
473 r=reference((i-1)%26, div(i-1, 26))
474 v=axis(r)
475 !insert&&v==at && continue
476 t=change(r, v+(insert ? Int(v>=at) : -Int(v>at)))
477 next[id(t)]=rewrite(sheet.contents[i], transform_axis)
478 end
479 else
480 throw(ArgumentError("unknown command"))
481 end
482 push!(sheet.history, (String(name), sheet.contents))
483 changed=replace_contents!(sheet, next)
484 [
485 name*(name=="SET" ? " "*label(target)*"="*notation(sheet.values[target]) : "")*" changed="*string(
486 changed,
487 ),
488 ]
489end
