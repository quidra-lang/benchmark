78parse_formula <- function(s) {
79 tokens <- list()
80 i <- 2L
81 n <- nchar(s)
82 syntax <- function(p) fail(paste("syntax error at", p))
83 add <- function(x, p) {
84 tokens[[length(tokens) + 1L]] <<- list(x, p)
85 }
86 while (i <= n) {
87 p <- i
88 ch <- substr(s, i, i)
89 if (ch == " ") {
90 i <- i + 1L
91 next
92 }
93 if (grepl("[0-9]", ch)) {
94 w <- regmatches(substring(s, i), regexpr("^[0-9]+", substring(s, i)))
95 if (nchar(w) >= 10) {
96 syntax(p)
97 }
98 add(num(as.numeric(w)), p)
99 i <- i + nchar(w)
100 } else if (ch == '"') {
101 j <- i + 1L
102 while (j <= n && substr(s, j, j) != '"') {
103 j <- j + 1L
104 }
105 if (j > n) {
106 syntax(p)
107 }
108 add(text(substr(s, i + 1L, j - 1L)), p)
109 i <- j + 1L
110 } else if (grepl("[A-Za-z$]", ch)) {
111 w <- toupper(regmatches(
112 substring(s, i),
113 regexpr("^[A-Za-z0-9$]+", substring(s, i))
114 ))
115 i <- i + nchar(w)
116 m <- regmatches(w, regexec("^(\\$?)([A-Z])(\\$?)([1-9][0-9]?)$", w))[[1]]
117 if (length(m)) {
118 x <- ref(
119 utf8ToInt(m[3]) - 65L,
120 as.integer(m[5]) - 1L,
121 as.integer(m[2] != "") + 2L * as.integer(m[4] != "")
122 )
123 } else if (grepl("^[A-Z][A-Z0-9]*$", w)) {
124 x <- list(if (grepl("^ *\\(", substring(s, i))) "C" else "W", w)
125 } else {
126 syntax(p)
127 }
128 add(x, p)
129 } else {
130 op <- substr(s, i, i + 1L)
131 if (!op %in% c("<>", "<=", ">=")) {
132 op <- ch
133 }
134 if (!ch %in% strsplit("+-*/&(),:=<>", "", fixed = TRUE)[[1]]) {
135 syntax(p)
136 }
137 add(list("T", op), p)
138 i <- i + nchar(op)
139 }
140 }
141 add(list("END"), n + 1L)
142 i <- 1L
143 take <- function(op) {
144 if (identical(tokens[[i]][[1]], list("T", op))) {
145 i <<- i + 1L
146 TRUE
147 } else {
148 FALSE
149 }
150 }
151 expect <- function(op) {
152 if (!take(op)) syntax(tokens[[i]][[2]])
153 }
154 expression <- function(level = 1L) {
155 if (level == 5L) {
156 return(primary())
157 }
158 a <- expression(level + 1L)
159 while (
160 tokens[[i]][[1]][[1]] == "T" && tokens[[i]][[1]][[2]] %in% levels[[level]]
161 ) {
162 op <- tokens[[i]][[1]][[2]]
163 i <<- i + 1L
164 a <- list("B", op, a, expression(level + 1L))
165 }
166 a
167 }
168 primary <- function() {
169 if (take("-")) {
170 return(list("U", primary()))
171 }
172 x <- tokens[[i]][[1]]
173 p <- tokens[[i]][[2]]
174 i <<- i + 1L
175 if (x[[1]] %in% c("N", "S", "W")) {
176 return(x)
177 }
178 if (x[[1]] == "R") {
179 if (!take(":")) {
180 return(x)
181 }
182 if (tokens[[i]][[1]][[1]] != "R") {
183 syntax(tokens[[i]][[2]])
184 }
185 b <- tokens[[i]][[1]]
186 i <<- i + 1L
187 return(make_range(x, b))
188 }
189 if (x[[1]] == "C") {
190 expect("(")
191 args <- list()
192 if (!take(")")) {
193 args[[1]] <- expression()
194 while (take(",")) {
195 args[[length(args) + 1L]] <- expression()
196 }
197 expect(")")
198 }
199 return(list("F", x[[2]], args))
200 }
201 if (identical(x, list("T", "("))) {
202 a <- expression()
203 expect(")")
204 return(a)
205 }
206 syntax(p)
207 }
208 tree <- expression()
209 if (tokens[[i]][[1]][[1]] != "END") {
210 syntax(tokens[[i]][[2]])
211 }
212 tree
213}

379 }
380 }
381 }
382 return(num(as.integer(switch(
383 op,
384 "=" = c == 0,
385 "<>" = c != 0,
386 "<" = c < 0,

410call_function <- function(name, args) {
411 if (
412 name %in% c("SUM", "MIN", "MAX", "COUNT", "AVERAGE", "AND", "OR", "CONCAT")
413 ) {
414 if (!length(args)) {
415 return(err("#VALUE!"))
416 }
417 items <- list()
418 for (a in args) {
419 v <- evaluate(a)
420 items <- c(items, if (v[[1]] == "G") sheet$values[cells(v)] else list(v))
421 }
422 ns <- vapply(
423 Filter(function(v) v[[1]] == "N", items),
424 function(v) v[[2]],
425 numeric(1)
426 )
427 if (name == "COUNT") {
428 return(num(length(ns)))
429 }
430 for (v in items) {
431 if (v[[1]] == "E") return(v)
432 }
433 return(switch(
434 name,
435 SUM = num(sum(ns)),
436 MIN = num(if (length(ns)) min(ns) else 0),
437 MAX = num(if (length(ns)) max(ns) else 0),
438 AVERAGE = if (length(ns)) {
439 num(trunc(sum(ns) / length(ns)))
440 } else {
441 err("#DIV/0!")
442 },
443 AND = if (length(ns)) num(as.integer(all(ns != 0))) else err("#VALUE!"),
444 OR = if (length(ns)) num(as.integer(any(ns != 0))) else err("#VALUE!"),
445 CONCAT = text(paste(
446 vapply(items, function(v) to_string(v)[[2]], ""),
447 collapse = ""
448 ))
449 ))
450 }
451 if (name == "IF") {
452 if (!length(args) %in% c(2L, 3L)) {
453 return(err("#VALUE!"))
454 }
455 c <- to_number(evaluate(args[[1]]))
456 if (c[[1]] == "E") {
457 return(c)
458 }
459 return(
460 if (c[[2]] != 0) {
461 evaluate(args[[2]])
462 } else if (length(args) == 3L) {
463 evaluate(args[[3]])
464 } else {
465 num(0)
466 }
467 )
468 }
469 if (!name %in% c("ABS", "NOT", "LEN")) {
470 return(err("#NAME?"))
471 }
472 if (length(args) != 1L) {
473 return(err("#VALUE!"))
474 }
475 v <- evaluate(args[[1]])
476 v <- if (name == "LEN") to_string(v) else to_number(v)
477 if (v[[1]] == "E") {
478 v
479 } else {
480 num(
481 if (name == "LEN") {
482 nchar(v[[2]])
483 } else if (name == "ABS") {

486 as.integer(v[[2]] == 0)

488 )

503 visit <- function(i) {
