11 static bool Matches(string value, string pattern) => Regex.IsMatch(value,

13 static Expr Node(char k, int n, string s, Expr a, Expr b, List<Expr> args) {

16 static Expr Leaf(char k, int n, string s) {
17 return Node(k, n, s, null, null, []);

20 return Leaf('N', n, "");

23 return Leaf('S', 0, s);

26 return Leaf('E', 0, s);

28 static Expr Ref(int c, int r, int f) {
29 return new Expr('R', c, r, f, "", null, null, []);

31 static readonly Expr INVALID = Leaf('X', 0, "");
32 static int Id(Expr r) {
33 return r.Row * 26 + r.N;
34 }
35 static string Label(int i) {
36 return "" + (char)('A' + i % 26) + (i / 26 + 1);
37 }
38 static bool Inside(Expr r) {
39 return r.N >= 0 && r.N < 26 && r.Row >= 0 && r.Row < 99;
40 }
41 static Expr Range(Expr a, Expr b) {
42 int c = a.N, d = b.N, r = a.Row, t = b.Row, ac = a.Flags & 1, bc = b.Flags & 1,
43 ar = a.Flags & 2, br = b.Flags & 2;
44 if (c > d) {
45 int z = c;
46 c = d;
47 d = z;
48 z = ac;
49 ac = bc;
50 bc = z;
51 }
52 if (r > t) {
53 int z = r;
54 r = t;
55 t = z;
56 z = ar;
57 ar = br;
58 br = z;
59 }
60 return Node('G', 0, "", Ref(c, r, ac | ar), Ref(d, t, bc | br), []);
61 }
62 static Expr Address(string s) {
63 if (!Matches(s, "[A-Z][1-9][0-9]?"))
64 throw new ArgumentException("bad address");
65 return Ref(s[0] - 'A', int.Parse(s.Substring(1)) - 1, 0);
66 }
67 static Expr Area(string s) {
68 int p = s.IndexOf(':');
69 if (p < 0) {
70 Expr a = Address(s);
71 return Range(a, a);
72 }
73 Expr a = Address(s.Substring(0, p)), b = Address(s.Substring(p + 1));
74 return Range(a, b);
75 }
76 static List<int> Cells(Expr g) {
77 List<int> resultRows = new();
78 for (int r = g.A.Row; r <= g.B.Row; r++)
79 for (int c = g.A.N; c <= g.B.N; c++)
80 resultRows.Add(r * 26 + c);
81 return resultRows;
82 }
83 static bool Contains(Expr g, Expr r) {
84 return r.N >= g.A.N && r.N <= g.B.N && r.Row >= g.A.Row && r.Row <= g.B.Row;
85 }
86 static readonly List<HashSet<string>> LEVELS = [
87 new HashSet<string> { "=", "<>", "<", "<=", ">", ">=" }, new HashSet<string> { "&" },
88 new HashSet<string> { "+", "-" }, new HashSet<string> { "*", "/" }
89 ];
90 static int Precedence(string op) {
91 for (int i = 0; i < 4; i++)
92 if (LEVELS[i].Contains(op))
93 return i;
94 return -1;

96 record Token(Expr Value, int Position);
97 sealed class Parser {
98 readonly List<Token> tokens = new();
99 int at;
100 void Syntax(int p) {
101 throw new ArgumentException("syntax error at " + p);
102 }
103 public Parser(string s) {
104 for (int i = 1; i < s.Length;) {
105 int p = i + 1;
106 char ch = s[i];
107 if (ch == ' ') {
108 i++;
109 continue;
110 }
111 Expr x;
112 if (ch >= '0' && ch <= '9') {
113 int j = i;
114 while (j < s.Length && char.IsDigit(s[j]))
115 j++;
116 if (j - i >= 10)
117 Syntax(p);
118 x = Num(int.Parse(s.Substring(i, j - i)));
119 i = j;
120 } else if (ch == '"') {
121 int j = s.IndexOf('"', i + 1);
122 if (j < 0)
123 Syntax(p);
124 x = Text(s.Substring(i + 1, j - i - 1));
125 i = j + 1;
126 } else if (char.IsLetter(ch) || ch == '$') {
127 int j = i;
128 while (j < s.Length && (char.IsLetterOrDigit(s[j]) || s[j] == '$'))
129 j++;
130 string w = s.Substring(i, j - i).ToUpperInvariant();
131 i = j;
132 Match m = Regex.Match(w, "\\A(\\$?)([A-Z])(\\$?)([1-9][0-9]?)\\z");
133 if (m.Success)
134 x = Ref(m.Groups[2].Value[0] - 'A', int.Parse(m.Groups[4].Value) - 1,
135 (m.Groups[1].Value.Length == 0 ? 0 : 1) |
136 (m.Groups[3].Value.Length == 0 ? 0 : 2));
137 else if (Matches(w, "[A-Z][A-Z0-9]*"))
138 x = Leaf(s.Substring(j).TrimStart().StartsWith("(") ? 'C' : 'W', 0, w);
139 else {
140 Syntax(p);
141 return;
142 }
143 } else {
144 string op = s.Substring(i, Math.Min(2, s.Length - i));
145 if (!new HashSet<string> { "<>", "<=", ">=" }.Contains(op))
146 op = "" + ch;
147 if ("+-*/&(),:=<>".IndexOf(ch) < 0)
148 Syntax(p);
149 x = Leaf('T', 0, op);
150 i += op.Length;
151 }
152 tokens.Add(new Token(x, p));
153 }
154 tokens.Add(new Token(Leaf('Z', 0, ""), s.Length + 1));
155 }
156 bool Take(string op) {
157 Expr x = tokens[at].Value;
158 if (x.Kind == 'T' && x.Text.Equals(op)) {
159 at++;
160 return true;
161 }
162 return false;
163 }
164 void Expect(string op) {
165 if (!Take(op))
166 Syntax(tokens[at].Position);
167 }
168 Expr Expression(int level) {
169 if (level == 4)
170 return Primary();
171 Expr a = Expression(level + 1);
172 while (tokens[at].Value.Kind == 'T' && LEVELS[level].Contains(tokens[at].Value.Text)) {
173 string op = tokens[at++].Value.Text;
174 a = Node('B', 0, op, a, Expression(level + 1), []);
175 }
176 return a;
177 }
178 Expr Primary() {
179 if (Take("-"))
180 return Node('U', 0, "", Primary(), null, []);
181 Token t = tokens[at++];
182 Expr x = t.Value;
183 if (x.Kind == 'N' || x.Kind == 'S' || x.Kind == 'W')
184 return x;
185 if (x.Kind == 'R') {
186 if (!Take(":"))
187 return x;
188 if (tokens[at].Value.Kind != 'R')
189 Syntax(tokens[at].Position);
190 return Range(x, tokens[at++].Value);
191 }
192 if (x.Kind == 'C') {
193 Expect("(");
194 List<Expr> args = new();
195 if (!Take(")")) {
196 args.Add(Expression(0));
197 while (Take(","))
198 args.Add(Expression(0));
199 Expect(")");
200 }
201 return Node('F', 0, x.Text, null, null, args);
202 }
203 if (x.Kind == 'T' && x.Text.Equals("(")) {
204 Expr a = Expression(0);
205 Expect(")");
206 return a;
207 }
208 Syntax(t.Position);
209 return null;
210 }
211 public Expr Parse() {
212 Expr x = Expression(0);
213 if (tokens[at].Value.Kind != 'Z')
214 Syntax(tokens[at].Position);
215 return x;
216 }
217 }
218 static string Canonical(Expr x) {
219 switch (x.Kind) {
220 case 'N':
221 return "" + x.N;
222 case 'S':
223 return "\"" + x.Text + "\"";
224 case 'W':
225 return x.Text;
226 case 'X':
227 return "#REF!";
228 case 'R':
229 return (x.Flags % 2 != 0 ? "$" : "") + (char)('A' + x.N) +
230 ((x.Flags & 2) != 0 ? "$" : "") + (x.Row + 1);
231 case 'G':
232 return Canonical(x.A) + ":" + Canonical(x.B);
233 case 'F': {
234 return x.Text + "(" + string.Join(",", x.Args.Select(Canonical)) + ")";
235 }
236 case 'U':
237 return "-" + (x.A.Kind == 'B' ? "(" + Canonical(x.A) + ")" : Canonical(x.A));
238 case 'B': {
239 string a = Canonical(x.A), b = Canonical(x.B);
240 if (x.A.Kind == 'B' && Precedence(x.A.Text) < Precedence(x.Text))
241 a = "(" + a + ")";
242 if (x.B.Kind == 'B' && Precedence(x.B.Text) <= Precedence(x.Text))
243 b = "(" + b + ")";
244 return a + x.Text + b;
245 }
246 default:
247 throw new InvalidOperationException();
248 }
249 }
250 static string Display(Expr x) {
251 return x == null ? "" : x.Kind == 'N' ? "" + x.N : x.Text;
252 }
253 static string Notation(Expr x) {
254 return x == null ? "(empty)"
255 : x.Kind == '=' ? "=" + Canonical(x.A)
256 : x.Kind == 'S' ? "\"" + x.Text + "\""
257 : Display(x);
258 }
259 static Expr ToNumber(Expr x) {
260 return x == null ? Num(0) : x.Kind == 'S' || x.Kind == 'G' ? Error("#VALUE!") : x;
261 }
262 static Expr ToText(Expr x) {
263 return x == null ? Text("")
264 : x.Kind == 'N' ? Text("" + x.N)
265 : x.Kind == 'G' ? Error("#VALUE!")
266 : x;
267 }
268 static Expr Rewrite(Expr x, Func<Expr, Expr> f) {
269 if (x == null)
270 return null;
271 if (x.Kind == 'R' || x.Kind == 'G')
272 return f(x);
273 Expr a = Rewrite(x.A, f), b = Rewrite(x.B, f);
274 List<Expr> args = new();
275 bool changed = a != x.A || b != x.B;
276 foreach (Expr old in x.Args) {
277 Expr v = Rewrite(old, f);
278 changed |= v != old;
279 args.Add(v);
280 }
281 return changed ? Node(x.Kind, x.N, x.Text, a, b, args) : x;
282 }
283 static void Dependencies(Expr x, SortedSet<int> resultRows) {
284 if (x == null)
285 return;
286 if (x.Kind == 'R')
287 resultRows.Add(Id(x));
288 else if (x.Kind == 'G')
289 resultRows.UnionWith(Cells(x));
290 else {
291 Dependencies(x.A, resultRows);
292 Dependencies(x.B, resultRows);
293 foreach (Expr a in x.Args)
294 Dependencies(a, resultRows);
295 }

306 Expr Evaluate(Expr x) {
307 if (x.Kind == 'N' || x.Kind == 'S' || x.Kind == 'G')
308 return x;
309 if (x.Kind == 'R')
310 return values[Id(x)];
311 if (x.Kind == 'W')
312 return Error("#NAME?");
313 if (x.Kind == 'X')
314 return Error("#REF!");
315 if (x.Kind == 'U') {
316 Expr a = ToNumber(Evaluate(x.A));
317 return a.Kind == 'E' ? a : Num(-a.N);
318 }
319 if (x.Kind == 'B') {
320 Expr a = Evaluate(x.A), b = Evaluate(x.B);
321 if (a != null && a.Kind == 'E')
322 return a;
323 if (b != null && b.Kind == 'E')
324 return b;
325 string op = x.Text;
326 if (Precedence(op) == 0) {
327 if ((a != null && a.Kind == 'G') || (b != null && b.Kind == 'G'))
328 return Error("#VALUE!");
329 bool strings = (a != null && a.Kind == 'S') || (b != null && b.Kind == 'S');
330 if (a == null)
331 a = strings ? Text("") : Num(0);
332 if (b == null)
333 b = strings ? Text("") : Num(0);
334 int c = a.Kind != b.Kind ? (a.Kind == 'N' ? -1 : 1)
335 : a.Kind == 'N' ? a.N.CompareTo(b.N)
336 : string.CompareOrdinal(a.Text, b.Text);
337 bool yes = op switch { "=" => c == 0, "<>" => c != 0, "<" => c < 0,
338 "<=" => c <= 0, ">" => c > 0,
339 _ => c >= 0 };
340 return Num(yes ? 1 : 0);
341 }
342 a = op.Equals("&") ? ToText(a) : ToNumber(a);
343 b = op.Equals("&") ? ToText(b) : ToNumber(b);
344 if (a.Kind == 'E')
345 return a;
346 if (b.Kind == 'E')
347 return b;
348 return op switch { "&" => Text(a.Text + b.Text), "+" => Num(a.N + b.N),
349 "-" => Num(a.N - b.N), "*" => Num(a.N * b.N),
350 _ => b.N == 0 ? Error("#DIV/0!") : Num(a.N / b.N) };
351 }
352 string name = x.Text;
353 List<Expr> args = x.Args;
354 if (new HashSet<string> { "SUM", "MIN", "MAX", "COUNT", "AVERAGE", "AND", "OR",
355 "CONCAT" }
356 .Contains(name)) {
357 if (args.Count == 0)
358 return Error("#VALUE!");
359 List<Expr> items = new();
360 List<int> ns = new();
361 foreach (Expr arg in args) {
362 Expr v = Evaluate(arg);
363 if (v != null && v.Kind == 'G') {
364 foreach (int i in Cells(v))
365 items.Add(values[i]);
366 } else
367 items.Add(v);
368 }
369 foreach (Expr v in items)
370 if (v != null && v.Kind == 'N')
371 ns.Add(v.N);
372 if (name.Equals("COUNT"))
373 return Num(ns.Count);
374 foreach (Expr v in items)
375 if (v != null && v.Kind == 'E')
376 return v;
377 if (name.Equals("CONCAT")) {
378 StringBuilder s = new StringBuilder();
379 foreach (Expr v in items)
380 s.Append(ToText(v).Text);
381 return Text(s.ToString());
382 }
383 int sum = 0;
384 foreach (int n in ns)
385 sum += n;
386 if (name.Equals("SUM"))
387 return Num(sum);
388 if (name.Equals("MIN"))
389 return Num(ns.Count == 0 ? 0 : ns.Min());
390 if (name.Equals("MAX"))
391 return Num(ns.Count == 0 ? 0 : ns.Max());
392 if (ns.Count == 0)
393 return Error(name.Equals("AVERAGE") ? "#DIV/0!" : "#VALUE!");
394 if (name.Equals("AVERAGE"))
395 return Num(sum / ns.Count);
396 bool result = name.Equals("AND");
397 foreach (int n in ns)
398 result = name.Equals("AND") ? result && n != 0 : result || n != 0;
399 return Num(result ? 1 : 0);
400 }
401 if (name.Equals("IF")) {
402 if (args.Count != 2 && args.Count != 3)
403 return Error("#VALUE!");
404 Expr c = ToNumber(Evaluate(args[0]));
405 if (c.Kind == 'E')
406 return c;
407 return c.N != 0 ? Evaluate(args[1]) : args.Count == 3 ? Evaluate(args[2]) : Num(0);
408 }
409 if (!new HashSet<string> { "ABS", "NOT", "LEN" }.Contains(name))
410 return Error("#NAME?");
411 if (args.Count != 1)
412 return Error("#VALUE!");
413 Expr scalar = Evaluate(args[0]);
414 scalar = name.Equals("LEN") ? ToText(scalar) : ToNumber(scalar);
415 return scalar.Kind == 'E' ? scalar
416 : Num(name.Equals("LEN") ? scalar.Text.Length
417 : name.Equals("ABS") ? Math.Abs(scalar.N)
418 : scalar.N == 0 ? 1
419 : 0);

421 void Recalculate() {
422 for (int i = 0; i < 2574; i++) {
423 deps[i].Clear();
424 if (contents[i] != null && contents[i].Kind == '=') {
425 Dependencies(contents[i].A, deps[i]);
426 values[i] = null;
427 } else
428 values[i] = contents[i];
429 }
430 int serial = 0;
431 int[] indices = new int[2574], low = new int[2574];
432 Array.Fill(indices, -1);
433 bool[] on = new bool[2574], cyclic = new bool[2574], done = new bool[2574];
434 Stack<int> stack = new();
435 void Visit(int i) {
436 indices[i] = low[i] = serial++;
437 stack.Push(i);
438 on[i] = true;
439 foreach (int j in deps[i])
440 if (contents[j] != null && contents[j].Kind == '=') {
441 if (indices[j] < 0) {
442 Visit(j);
443 low[i] = Math.Min(low[i], low[j]);
444 } else if (on[j])
445 low[i] = Math.Min(low[i], indices[j]);
446 }
447 if (low[i] == indices[i]) {
448 List<int> part = new();
449 int j;
450 do {
451 j = stack.Pop();
452 on[j] = false;
453 part.Add(j);
454 } while (j != i);
455 if (part.Count > 1 || deps[i].Contains(i))
456 foreach (int k in part)
457 cyclic[k] = true;
458 }
459 }
460 void Calculate(int i) {
461 if (contents[i] == null || contents[i].Kind != '=' || done[i])
462 return;
463 done[i] = true;
464 if (cyclic[i]) {
465 values[i] = Error("#CYCLE!");
466 return;
467 }
468 foreach (int j in deps[i])
469 Calculate(j);
470 Expr v = Evaluate(contents[i].A);
471 values[i] = v == null ? Num(0) : v.Kind == 'G' ? Error("#VALUE!") : v;
472 }
473 for (int i = 0; i < 2574; i++)
474 if (contents[i] != null && contents[i].Kind == '=' && indices[i] < 0)
475 Visit(i);
476 for (int i = 0; i < 2574; i++)
477 Calculate(i);
478 }

612 throw new ArgumentException("bad address");

619 }
620 if (insert)
621 for (int i = 0; i < 2574; i++)
622 if (contents[i] != null && (row ? i / 26 : i % 26) == limit - 1)
623 throw new ArgumentException("last " + (row ? "row" : "column") +
624 " not empty");
625 Func<Expr, Expr> transform = x => {
626 if (x.Kind == 'R') {
627 int coordinate = Axis(x);
628 if (!insert && coordinate == at)
629 return INVALID;
630 int n = coordinate +
631 (insert ? (coordinate >= at ? 1 : 0) : (coordinate > at ? -1 : 0));
632 return n >= limit ? INVALID : Change(x, n);
633 }
634 int a = Axis(x.A), b = Axis(x.B);
635 if (!insert && a == at && b == at)
636 return INVALID;
637 int u = a + (insert ? (a >= at ? 1 : 0) : (a > at ? -1 : 0)),
638 v = b + (insert ? (b >= at ? 1 : 0) : (b >= at ? -1 : 0));
639 return v >= limit ? INVALID : Range(Change(x.A, u), Change(x.B, v));
640 };
641 next = new Expr[2574];
642 for (int i = 0; i < 2574; i++)
643 if (contents[i] != null) {
644 Expr r = Ref(i % 26, i / 26, 0);
645 int v = Axis(r);
646 if (!insert && v == at)
647 continue;
648 Expr t = Change(r, v + (insert ? (v >= at ? 1 : 0) : (v > at ? -1 : 0)));
649 next[Id(t)] = Rewrite(contents[i], transform);
650 }
651 } else
652 throw new ArgumentException("unknown command");
653 history.Add(new History(name, contents));
654 int changed = Replace(next);
655 return
656 [name +
657 (name.Equals("SET") ? " " + Label(target) + "=" + Notation(values[target]) : "") +
658 " changed=" + changed];
659 }
660 }
