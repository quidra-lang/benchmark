func num(n int) *node           { return &node{kind: 'N', n: n} }
func text(s string) *node       { return &node{kind: 'S', text: s} }
func errorValue(s string) *node { return &node{kind: 'E', text: s} }
func number(x *node) *node {
	if x == nil {
		return num(0)
	}
	if x.kind == 'S' || x.kind == 'G' {
		return errorValue("#VALUE!")
	}
	return x
func (s *sheet) evaluate(x *node) *node {
	switch x.kind {
	case 'N', 'S', 'G':
		return x
	case 'R':
		return s.values[id(x)]
	case 'U':
		a := number(s.evaluate(x.a))
		if a.kind == 'E' {
			return a
		}
		return num(-a.n)
