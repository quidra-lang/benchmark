181def make_range(first, last):
182 if first.col > last.col:
183 first, last = (
184 replace(first, col=last.col, absolute_col=last.absolute_col),

201def syntax_error(position):
202 return CommandError(f"syntax error at {position}")

205def tokenize(formula):
206 tokens = []
207 index = 1
208 while index < len(formula):
209 char = formula[index]
210 position = index + 1
211 if char == " ":
212 index += 1
213 elif char.isdigit():
214 digits = DIGITS_RE.match(formula, index)[0]
215 if len(digits) >= 10:
216 raise syntax_error(position)
217 tokens.append(Token("number", int(digits), position))
218 index += len(digits)
219 elif char == '"':
220 end = formula.find('"', index + 1)
221 if end < 0:
222 raise syntax_error(position)
223 tokens.append(Token("string", formula[index + 1 : end], position))
224 index = end + 1
225 elif char.isalpha() or char == "$":
226 word = WORD_RE.match(formula, index)[0]
227 index += len(word)
228 tokens.append(classify_word(word.upper(), formula[index:], position))
229 elif formula[index : index + 2] in TWO_CHAR_SYMBOLS:
230 tokens.append(Token("symbol", formula[index : index + 2], position))
231 index += 2
232 elif char in SYMBOLS:
233 tokens.append(Token("symbol", char, position))
234 index += 1
235 else:
236 raise syntax_error(position)
237 tokens.append(Token("end", None, len(formula) + 1))
238 return tokens

241def classify_word(word, rest, position):
242 match = REFERENCE_RE.fullmatch(word)
243 if match:
244 absolute_col, letter, absolute_row, row = match.groups()
245 reference = Reference(
246 COLUMNS.index(letter) + 1, int(row), absolute_col == "$", absolute_row == "$"
247 )
248 return Token("reference", reference, position)
249 if NAME_RE.fullmatch(word):
250 kind = "function" if rest.lstrip(" ").startswith("(") else "name"
251 return Token(kind, word, position)
252 raise syntax_error(position)

255class Parser:
256 def __init__(self, tokens):
257 self.tokens = tokens
258 self.index = 0
259
260 def accept(self, symbols):
261 token = self.tokens[self.index]
262 if token.kind == "symbol" and token.value in symbols:
263 self.index += 1
264 return token.value
265 return None
266
267 def expect(self, symbol):
268 if not self.accept({symbol}):
269 raise syntax_error(self.tokens[self.index].position)
270
271 def formula(self):
272 tree = self.expression()
273 token = self.tokens[self.index]
274 if token.kind != "end":
275 raise syntax_error(token.position)
276 return tree
277
278 def expression(self, level=0):
279 if level == len(OPERATOR_LEVELS):
280 return self.unary()
281 left = self.expression(level + 1)
282 while op := self.accept(OPERATOR_LEVELS[level]):
283 left = BinaryOp(op, left, self.expression(level + 1))
284 return left
285
286 def unary(self):
287 if self.accept({"-"}):
288 return Negate(self.unary())
289 return self.primary()
290
291 def primary(self):
292 token = self.tokens[self.index]
293 self.index += 1
294 match token.kind:
295 case "number":
296 return Number(token.value)
297 case "string":
298 return Text(token.value)
299 case "name":
300 return Name(token.value)
301 case "reference":
302 if not self.accept({":"}):
303 return token.value
304 last = self.tokens[self.index]
305 if last.kind != "reference":
306 raise syntax_error(last.position)
307 self.index += 1
308 return make_range(token.value, last.value)
309 case "function":
310 self.expect("(")
311 args = []
312 if not self.accept({")"}):
313 args.append(self.expression())
314 while self.accept({","}):
315 args.append(self.expression())
316 self.expect(")")
317 return Call(token.value, tuple(args))
318 case "symbol" if token.value == "(":
319 inner = self.expression()
320 self.expect(")")
321 return inner
322 raise syntax_error(token.position)

325def parse_content(text):
326 if text.startswith("="):
327 return Parser(tokenize(text)).formula()
328 if len(text) >= 2 and text[0] == text[-1] == '"' and '"' not in text[1:-1]:
329 return text[1:-1]
330 if INTEGER_RE.fullmatch(text):
331 return int(text)
332 raise CommandError("bad value")

335def canonical(node):
336 match node:
337 case Number(value):
338 return str(value)
339 case Text(value):
340 return f'"{value}"'
341 case Name(word):
342 return word
343 case Reference() | RangeRef():
344 return str(node)
345 case InvalidRef():
346 return "#REF!"
347 case Call(name, args):
348 return f"{name}({','.join(map(canonical, args))})"
349 case Negate(operand):
350 text = canonical(operand)
351 return f"-({text})" if isinstance(operand, BinaryOp) else f"-{text}"
352 case BinaryOp(op, left, right):
353 left_text = canonical(left)
354 right_text = canonical(right)
355 if isinstance(left, BinaryOp) and PRECEDENCE[left.op] < PRECEDENCE[op]:
356 left_text = f"({left_text})"
357 if isinstance(right, BinaryOp) and PRECEDENCE[right.op] <= PRECEDENCE[op]:
358 right_text = f"({right_text})"
359 return left_text + op + right_text

386def collect_precedents(node):
387 match node:
388 case Reference():
389 return {node.cell}
390 case RangeRef():
391 return set(node.area.cells())
392 case Call(args=args):
393 return set().union(*map(collect_precedents, args))
394 case Negate(operand):
395 return collect_precedents(operand)
396 case BinaryOp(left=left, right=right):
397 return collect_precedents(left) | collect_precedents(right)
398 return set()

401def rewrite(node, on_reference, on_range):
402 match node:
403 case Reference():
404 return on_reference(node)
405 case RangeRef():
406 return on_range(node)
407 case Call(name, args):
408 return Call(name, tuple(rewrite(arg, on_reference, on_range) for arg in args))
409 case Negate(operand):
410 return Negate(rewrite(operand, on_reference, on_range))
411 case BinaryOp(op, left, right):
412 return BinaryOp(
413 op, rewrite(left, on_reference, on_range), rewrite(right, on_reference, on_range)

447def compare(x, y):
448 if isinstance(x, str) and isinstance(y, int):
449 return 1
450 if isinstance(x, int) and isinstance(y, str):
451 return -1
452 blank = "" if isinstance(x, str) or isinstance(y, str) else 0
453 x = blank if x is None else x
454 y = blank if y is None else y
455 return (x > y) - (x < y)

459 result = abs(a) // abs(b)

463def strongly_connected_components(graph):
464 index = {}
465 lowlink = {}
466 stack = []
467 on_stack = set()
468 components = []
469
470 def visit(node):
471 index[node] = lowlink[node] = len(index)
472 stack.append(node)
473 on_stack.add(node)
474 for successor in graph[node]:
475 if successor not in index:
476 visit(successor)
477 lowlink[node] = min(lowlink[node], lowlink[successor])
478 elif successor in on_stack:
479 lowlink[node] = min(lowlink[node], index[successor])
480 if lowlink[node] == index[node]:
481 component = []
482 while True:
483 member = stack.pop()
484 on_stack.remove(member)
485 component.append(member)
486 if member == node:
487 break
488 components.append(component)
489
490 for node in graph:
491 if node not in index:
492 visit(node)
493 return components

496class Sheet:
497 def __init__(self):
498 self.contents = {}
499 self.values = {}
500 self.precedents = {}
501 self.history = []

503 def execute(self, line):
504 name, _, argument = line.partition(" ")
505 match name:
506 case "SET":
507 self.set_cell(argument)
508 case "CLEAR":
509 self.modify(name, self.cleared(argument))
510 case "COPY":
511 self.modify(name, self.copied(*argument.split(" ")))
512 case "MOVE":
513 self.modify(name, self.moved(*argument.split(" ")))
514 case "INSERTROW" | "DELETEROW" | "INSERTCOL" | "DELETECOL":
515 self.modify(name, self.lines_changed(name, argument))
516 case "PRINT":
517 self.print_range(argument)
518 case "SHOW":
519 cell = parse_address(argument)
520 print(self.describe(argument, cell))
521 case "DEPS":
522 self.show_dependencies(argument)
523 case "LIST":
524 self.list_cells()
525 case "UNDO":
526 self.undo()
527 case _:
528 raise CommandError("unknown command")

530 def commit(self, name, contents):
531 self.history.append((name, self.contents))
532 return self.replace_contents(contents)

534 def modify(self, name, contents):
535 print(f"{name} changed={self.commit(name, contents)}")

537 def replace_contents(self, contents):
538 old_values = self.values
539 self.contents = contents
540 self.recalculate()
541 cells = old_values.keys() | self.values.keys()
542 return sum(1 for cell in cells if old_values.get(cell) != self.values.get(cell))

544 def recalculate(self):
545 formulas = {
546 cell: content for cell, content in self.contents.items() if isinstance(content, Node)
547 }
548 self.precedents = {cell: collect_precedents(tree) for cell, tree in formulas.items()}
549 graph = {
550 cell: [other for other in cells if other in formulas]
551 for cell, cells in self.precedents.items()
552 }
553 self.values = {
554 cell: content for cell, content in self.contents.items() if cell not in formulas
555 }
556 for component in strongly_connected_components(graph):
557 in_cycle = len(component) > 1 or component[0] in graph[component[0]]
558 for cell in component:
559 if in_cycle:
560 self.values[cell] = ErrorValue.CYCLE
561 else:
562 self.values[cell] = self.formula_value(formulas[cell])

564 def formula_value(self, tree):
565 value = self.evaluate(tree)
566 if value is None:
567 return 0
568 if isinstance(value, CellRange):
569 return ErrorValue.VALUE
570 return value
571
572 def evaluate(self, node):
573 match node:
574 case Number(value) | Text(value):
575 return value
576 case Reference():
577 return self.values.get(node.cell)
578 case RangeRef():
579 return node.area
580 case Name():
581 return ErrorValue.NAME
582 case InvalidRef():
583 return ErrorValue.REF
584 case Call(name, args):
585 return self.call(name, args)
586 case Negate(operand):
587 number = to_number(self.evaluate(operand))
588 return number if isinstance(number, ErrorValue) else -number
589 case BinaryOp(op, left, right):
590 return self.apply_operator(op, self.evaluate(left), self.evaluate(right))
591
592 def apply_operator(self, op, x, y):
593 if isinstance(x, ErrorValue):
594 return x
595 if isinstance(y, ErrorValue):
596 return y
597 if op in COMPARISONS:
598 if isinstance(x, CellRange) or isinstance(y, CellRange):
599 return ErrorValue.VALUE
600 return 1 if COMPARISONS[op](compare(x, y), 0) else 0
601 convert = to_string if op == "&" else to_number
602 a = convert(x)
603 if isinstance(a, ErrorValue):
604 return a
605 b = convert(y)
606 if isinstance(b, ErrorValue):
607 return b
608 match op:
609 case "&" | "+":
610 return a + b
611 case "-":
612 return a - b
613 case "*":
614 return a * b
615 return ErrorValue.DIV0 if b == 0 else quotient(a, b)
616
617 def call(self, name, args):
618 if name in AGGREGATES:
619 return self.aggregate(name, args) if args else ErrorValue.VALUE
620 if name == "IF":
621 if len(args) not in (2, 3):
622 return ErrorValue.VALUE
623 condition = to_number(self.evaluate(args[0]))
624 if isinstance(condition, ErrorValue):
625 return condition
626 if condition != 0:
627 return self.evaluate(args[1])
628 return self.evaluate(args[2]) if len(args) == 3 else 0
629 if name not in ("ABS", "NOT", "LEN"):
630 return ErrorValue.NAME
631 if len(args) != 1:
632 return ErrorValue.VALUE
633 value = self.evaluate(args[0])
634 if name == "LEN":
635 text = to_string(value)
636 return text if isinstance(text, ErrorValue) else len(text)
637 number = to_number(value)
638 if isinstance(number, ErrorValue):
639 return number
640 if name == "ABS":
641 return abs(number)
642 return 1 if number == 0 else 0

644 def aggregate(self, name, args):
645 items = []
646 for arg in args:
647 value = self.evaluate(arg)
648 if isinstance(value, CellRange):
649 items.extend(self.values.get(cell) for cell in value.cells())
650 else:
651 items.append(value)
652 numbers = [item for item in items if isinstance(item, int)]
653 if name == "COUNT":
654 return len(numbers)
655 error = next((item for item in items if isinstance(item, ErrorValue)), None)
656 if error is not None:
657 return error
658 match name:
659 case "SUM":
660 return sum(numbers)
661 case "MIN":
662 return min(numbers, default=0)
663 case "MAX":
664 return max(numbers, default=0)
665 case "AVERAGE":
666 return quotient(sum(numbers), len(numbers)) if numbers else ErrorValue.DIV0
667 case "AND" | "OR":
668 if not numbers:
669 return ErrorValue.VALUE
670 test = all if name == "AND" else any
671 return 1 if test(numbers) else 0
672 return "".join(map(to_string, items))

705 def shift(ref):
706 return replace(

708 row=ref.row if ref.absolute_row else ref.row + rows,

712 def on_reference(ref):

716 def on_range(area):
717 first = shift(area.first)
718 last = shift(area.last)
719 if first.cell.inside_sheet() and last.cell.inside_sheet():
720 return make_range(first, last)
721 return INVALID

723 contents = dict(self.contents)
724 for cell in source.cells():
725 content = self.contents.get(cell)
726 if isinstance(content, Node):
727 content = rewrite(content, on_reference, on_range)

735 if source.contains(ref.cell):
736 return replace(ref, row=ref.row + rows, col=ref.col + cols)

741 return RangeRef(on_reference(area.first), on_reference(area.last))

744 contents = self.rewritten(on_reference, on_range)

747 place(contents, cell.shifted(rows, cols), content)

750 def lines_changed(self, name, argument):
751 if name.endswith("ROW"):
752 if not ROW_RE.fullmatch(argument):
753 raise CommandError("bad address")
754 axis, index, limit, label = "row", int(argument), ROWS, "row"
755 else:
756 if not COLUMN_RE.fullmatch(argument):
757 raise CommandError("bad address")
758 axis, index, limit, label = "col", COLUMNS.index(argument) + 1, len(COLUMNS), "column"
759 if name.startswith("DELETE"):
760 return self.line_deleted(axis, index)
761 if any(getattr(cell, axis) == limit for cell in self.contents):
762 raise CommandError(f"last {label} not empty")
763 return self.line_inserted(axis, index, limit)

765 def line_inserted(self, axis, index, limit):
766 def bump(item):
767 value = getattr(item, axis)
768 return with_axis(item, axis, value + 1) if value >= index else item
769
770 def on_reference(ref):
771 ref = bump(ref)
772 return INVALID if getattr(ref, axis) > limit else ref
773
774 def on_range(area):
775 last = bump(area.last)
776 if getattr(last, axis) > limit:
777 return INVALID
778 return RangeRef(bump(area.first), last)
779
780 contents = self.rewritten(on_reference, on_range)
781 return {bump(cell): content for cell, content in contents.items()}

783 def line_deleted(self, axis, index):
784 def pull_back(item, threshold):
785 value = getattr(item, axis)
786 return with_axis(item, axis, value - 1) if value >= threshold else item
787
788 def on_reference(ref):
789 return INVALID if getattr(ref, axis) == index else pull_back(ref, index + 1)

791 def on_range(area):
792 if getattr(area.first, axis) == index == getattr(area.last, axis):
793 return INVALID
794 return RangeRef(pull_back(area.first, index + 1), pull_back(area.last, index))
795
796 contents = self.rewritten(on_reference, on_range)
797 return {
798 pull_back(cell, index + 1): content
799 for cell, content in contents.items()
800 if getattr(cell, axis) != index
801 }

803 def print_range(self, argument):
804 area = parse_range(argument)
805 rows = range(area.top, area.bottom + 1)
806 cols = range(area.left, area.right + 1)
807 texts = {cell: display(self.values.get(cell)) for cell in area.cells()}
808 label_width = len(str(area.bottom))
809 widths = [max(1, max(len(texts[Cell(row, col)]) for row in rows)) for col in cols]
810 separator = "+" + "-" * label_width + "+" + "".join("-" * width + "+" for width in widths)
811 header = "".join(COLUMNS[col - 1].ljust(width) + "|" for col, width in zip(cols, widths))
812 print(separator)
813 print("|" + " " * label_width + "|" + header)
814 print(separator)
815 for row in rows:
816 line = "|" + str(row).rjust(label_width) + "|"
817 for col, width in zip(cols, widths):
818 cell = Cell(row, col)
819 if isinstance(self.values.get(cell), int):
820 line += texts[cell].rjust(width) + "|"
821 else:
822 line += texts[cell].ljust(width) + "|"
823 print(line)
824 print(separator)
