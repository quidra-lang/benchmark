37 static Expr range(Expr a, Expr b) {
38 int c = a.n, d = b.n, r = a.row, t = b.row, ac = a.flags & 1, bc = b.flags & 1,
39 ar = a.flags & 2, br = b.flags & 2;
40 if (c > d) {
41 int z = c;
42 c = d;
43 d = z;
44 z = ac;
45 ac = bc;
46 bc = z;
47 }
48 if (r > t) {
49 int z = r;
50 r = t;
51 t = z;
52 z = ar;
53 ar = br;
54 br = z;
55 }
56 return node('G', 0, "", ref(c, r, ac | ar), ref(d, t, bc | br), List.of());

97 Parser(String s) {
98 for (int i = 1; i < s.length();) {
99 int p = i + 1;
100 char ch = s.charAt(i);
101 if (ch == ' ') {
102 i++;
103 continue;
104 }
105 Expr x;
106 if (ch >= '0' && ch <= '9') {
107 int j = i;
108 while (j < s.length() && Character.isDigit(s.charAt(j)))
109 j++;
110 if (j - i >= 10)
111 syntax(p);
112 x = num(Integer.parseInt(s.substring(i, j)));
113 i = j;
114 } else if (ch == '"') {
115 int j = s.indexOf('"', i + 1);
116 if (j < 0)
117 syntax(p);
118 x = text(s.substring(i + 1, j));
119 i = j + 1;
120 } else if (Character.isLetter(ch) || ch == '$') {
121 int j = i;
122 while (j < s.length() &&
123 (Character.isLetterOrDigit(s.charAt(j)) || s.charAt(j) == '$'))
124 j++;
125 String w = s.substring(i, j).toUpperCase(Locale.ROOT);
126 i = j;
127 Matcher m = Pattern.compile("(\\$?)([A-Z])(\\$?)([1-9][0-9]?)").matcher(w);
128 if (m.matches())
129 x = ref(m.group(2).charAt(0) - 'A', Integer.parseInt(m.group(4)) - 1,
130 (m.group(1).isEmpty() ? 0 : 1) | (m.group(3).isEmpty() ? 0 : 2));
131 else if (w.matches("[A-Z][A-Z0-9]*"))
132 x = leaf(s.substring(j).stripLeading().startsWith("(") ? 'C' : 'W', 0, w);
133 else {
134 syntax(p);
135 return;
136 }
137 } else {
138 String op = s.substring(i, Math.min(i + 2, s.length()));
139 if (!Set.of("<>", "<=", ">=").contains(op))
140 op = "" + ch;
141 if ("+-*/&(),:=<>".indexOf(ch) < 0)
142 syntax(p);
143 x = leaf('T', 0, op);
144 i += op.length();
145 }
146 tokens.add(new Token(x, p));
147 }
148 tokens.add(new Token(leaf('Z', 0, ""), s.length() + 1));
149 }
150 boolean take(String op) {
151 Expr x = tokens.get(at).value;
152 if (x.kind == 'T' && x.text.equals(op)) {
153 at++;
154 return true;
155 }
156 return false;
157 }
158 void expect(String op) {
159 if (!take(op))
160 syntax(tokens.get(at).position);
161 }
162 Expr expression(int level) {
163 if (level == 4)
164 return primary();
165 Expr a = expression(level + 1);
166 while (tokens.get(at).value.kind == 'T' &&
167 LEVELS.get(level).contains(tokens.get(at).value.text)) {
168 String op = tokens.get(at++).value.text;
169 a = node('B', 0, op, a, expression(level + 1), List.of());
170 }
171 return a;
172 }
173 Expr primary() {
174 if (take("-"))
175 return node('U', 0, "", primary(), null, List.of());
176 Token t = tokens.get(at++);
177 Expr x = t.value;
178 if (x.kind == 'N' || x.kind == 'S' || x.kind == 'W')
179 return x;
180 if (x.kind == 'R') {
181 if (!take(":"))
182 return x;
183 if (tokens.get(at).value.kind != 'R')
184 syntax(tokens.get(at).position);
185 return range(x, tokens.get(at++).value);
186 }
187 if (x.kind == 'C') {
188 expect("(");
189 List<Expr> args = new ArrayList<>();
190 if (!take(")")) {
191 args.add(expression(0));
192 while (take(","))
193 args.add(expression(0));
194 expect(")");
195 }
196 return node('F', 0, x.text, null, null, List.copyOf(args));
197 }
198 if (x.kind == 'T' && x.text.equals("(")) {
199 Expr a = expression(0);
200 expect(")");
201 return a;
202 }
203 syntax(t.position);
204 return null;
205 }
206 Expr parse() {
207 Expr x = expression(0);
208 if (tokens.get(at).value.kind != 'Z')
209 syntax(tokens.get(at).position);
210 return x;
211 }

213 static String canonical(Expr x) {
214 switch (x.kind) {
215 case 'N':
216 return "" + x.n;
217 case 'S':
218 return "\"" + x.text + "\"";
219 case 'W':
220 return x.text;
221 case 'X':
222 return "#REF!";
223 case 'R':
224 return (x.flags % 2 != 0 ? "$" : "") + (char)('A' + x.n) +
225 ((x.flags & 2) != 0 ? "$" : "") + (x.row + 1);
226 case 'G':
227 return canonical(x.a) + ":" + canonical(x.b);
228 case 'F': {
229 StringJoiner j = new StringJoiner(",", x.text + "(", ")");
230 for (Expr a : x.args)
231 j.add(canonical(a));
232 return j.toString();
233 }
234 case 'U':
235 return "-" + (x.a.kind == 'B' ? "(" + canonical(x.a) + ")" : canonical(x.a));
236 case 'B': {
237 String a = canonical(x.a), b = canonical(x.b);
238 if (x.a.kind == 'B' && precedence(x.a.text) < precedence(x.text))
239 a = "(" + a + ")";
240 if (x.b.kind == 'B' && precedence(x.b.text) <= precedence(x.text))
241 b = "(" + b + ")";
242 return a + x.text + b;
243 }
244 default:
245 throw new IllegalStateException();
246 }
247 }

266 static Expr rewrite(Expr x, UnaryOperator<Expr> f) {
267 if (x == null)
268 return null;
269 if (x.kind == 'R' || x.kind == 'G')
270 return f.apply(x);
271 Expr a = rewrite(x.a, f), b = rewrite(x.b, f);
272 List<Expr> args = new ArrayList<>();
273 boolean changed = a != x.a || b != x.b;
274 for (Expr old : x.args) {
275 Expr v = rewrite(old, f);
276 changed |= v != old;
277 args.add(v);
278 }
279 return changed ? node(x.kind, x.n, x.text, a, b, List.copyOf(args)) : x;
280 }
281 static void dependencies(Expr x, Set<Integer> out) {
282 if (x == null)
283 return;
284 if (x.kind == 'R')
285 out.add(id(x));
286 else if (x.kind == 'G')
287 out.addAll(cells(x));
288 else {
289 dependencies(x.a, out);
290 dependencies(x.b, out);
291 for (Expr a : x.args)
292 dependencies(a, out);
293 }

438 class Graph {
439 int serial;
440 int[] index = new int[2574], low = new int[2574];
441 boolean[] on = new boolean[2574], cyclic = new boolean[2574],
442 done = new boolean[2574];
443 Deque<Integer> stack = new ArrayDeque<>();
444 Graph() {
445 Arrays.fill(index, -1);
446 }
447 void visit(int i) {
448 index[i] = low[i] = serial++;
449 stack.push(i);
450 on[i] = true;
451 for (int j : deps.get(i))
452 if (contents[j] != null && contents[j].kind == '=') {
453 if (index[j] < 0) {
454 visit(j);
455 low[i] = Math.min(low[i], low[j]);
456 } else if (on[j])
457 low[i] = Math.min(low[i], index[j]);
458 }
459 if (low[i] == index[i]) {
460 List<Integer> part = new ArrayList<>();
461 int j;
462 do {
463 j = stack.pop();
464 on[j] = false;
465 part.add(j);
466 } while (j != i);
467 if (part.size() > 1 || deps.get(i).contains(i))
468 for (int k : part)
469 cyclic[k] = true;
470 }
471 }
472 void calculate(int i) {
473 if (contents[i] == null || contents[i].kind != '=' || done[i])
474 return;
475 done[i] = true;
476 if (cyclic[i]) {
477 values[i] = error("#CYCLE!");
478 return;
479 }
480 for (int j : deps.get(i))
481 calculate(j);
482 Expr v = evaluate(contents[i].a);
483 values[i] = v == null ? num(0) : v.kind == 'G' ? error("#VALUE!") : v;
484 }
485 }

638 class Axis {
639 int get(Expr x) {
640 return row ? x.row : x.n;
641 }
642 Expr set(Expr x, int v) {
643 return ref(row ? x.n : v, row ? v : x.row, x.flags);
644 }
645 }
646 Axis axis = new Axis();
647 if (insert)
648 for (int i = 0; i < 2574; i++)
649 if (contents[i] != null && (row ? i / 26 : i % 26) == limit - 1)
650 throw new IllegalArgumentException("last " + (row ? "row" : "column") +
651 " not empty");
652 UnaryOperator<Expr> transform = x -> {
653 if (x.kind == 'R') {
654 int v = axis.get(x);
655 if (!insert && v == at)
656 return INVALID;
657 int n = v + (insert ? (v >= at ? 1 : 0) : (v > at ? -1 : 0));
658 return n >= limit ? INVALID : axis.set(x, n);
659 }
660 int a = axis.get(x.a), b = axis.get(x.b);
661 if (!insert && a == at && b == at)
662 return INVALID;
663 int u = a + (insert ? (a >= at ? 1 : 0) : (a > at ? -1 : 0)),
664 v = b + (insert ? (b >= at ? 1 : 0) : (b >= at ? -1 : 0));
665 return v >= limit ? INVALID : range(axis.set(x.a, u), axis.set(x.b, v));
666 };
667 next = new Expr[2574];
668 for (int i = 0; i < 2574; i++)
669 if (contents[i] != null) {
670 Expr r = ref(i % 26, i / 26, 0);
671 int v = axis.get(r);
672 if (!insert && v == at)
673 continue;
674 Expr t = axis.set(r, v + (insert ? (v >= at ? 1 : 0) : (v > at ? -1 : 0)));
675 next[id(t)] = rewrite(contents[i], transform);
676 }
677 } else
678 throw new IllegalArgumentException("unknown command");
679 history.add(new History(name, contents));
680 int changed = replace(next);
681 return List.of(
682 name +
683 (name.equals("SET") ? " " + label(target) + "=" + notation(values[target]) : "") +
684 " changed=" + changed);
685 }
