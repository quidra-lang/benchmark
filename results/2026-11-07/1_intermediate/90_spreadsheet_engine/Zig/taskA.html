87fn addr(s: S) ?Ref {
88 if (s.len < 2 or s.len > 3 or s[0] < 'A' or s[0] > 'Z' or s[1] == '0') return null;
89 for (s[1..]) |c| if (!std.ascii.isDigit(c)) return null;
90 const r = std.fmt.parseInt(i32, s[1..], 10) catch return null;
91 if (r < 1 or r > 99) return null;
92 return .{ .c = s[0] - 'A' + 1, .r = r };
93}

95fn normalize(a0: Ref, b0: Ref) Rect {
96 var a = a0;
97 var b = b0;
98 if (a.c > b.c) {
99 std.mem.swap(i32, &a.c, &b.c);
100 std.mem.swap(bool, &a.ac, &b.ac);
101 }
102 if (a.r > b.r) {
103 std.mem.swap(i32, &a.r, &b.r);
104 std.mem.swap(bool, &a.ar, &b.ar);
105 }
106 return .{ .a = a, .b = b };
107}

144fn canonical(n: *Node) anyerror!S {
145 return switch (n.k) {
146 .num => fmt("{d}", .{n.n}),
147 .str => fmt("\"{s}\"", .{n.s}),
148 .name => n.s,
149 .ref => refText(n.a),
150 .range => fmt("{s}:{s}", .{ try refText(n.a), try refText(n.b) }),
151 .invalid => "#REF!",
152 .neg => if (n.args[0].k == .binary) fmt("-({s})", .{try canonical(n.args[0])}) else fmt("-{s}", .{try canonical(n.args[0])}),
153 .binary => blk: {
154 const x = n.args[0];
155 const y = n.args[1];
156 var l = try canonical(x);
157 var r = try canonical(y);
158 if (x.k == .binary and precedence(x.s) < precedence(n.s)) l = try fmt("({s})", .{l});
159 if (y.k == .binary and precedence(y.s) <= precedence(n.s)) r = try fmt("({s})", .{r});
160 break :blk fmt("{s}{s}{s}", .{ l, n.s, r });
161 },
162 .call => blk: {
163 var s = try fmt("{s}(", .{n.s});
164 for (n.args, 0..) |a, i| s = try fmt("{s}{s}{s}", .{ s, if (i == 0) @as(S, "") else ",", try canonical(a) });
165 break :blk fmt("{s})", .{s});
166 },
167 };

170const Token = struct { k: enum { num, str, name, ref, fun, sym, end }, s: S, p: usize, a: Ref = .{ .c = 0, .r = 0 } };

172fn reference(s: S) ?Ref {
173 var i: usize = 0;
174 var r: Ref = .{ .c = 0, .r = 0 };
175 if (i < s.len and s[i] == '$') {
176 r.ac = true;
177 i += 1;
178 }
179 if (i == s.len or s[i] < 'A' or s[i] > 'Z') return null;
180 r.c = s[i] - 'A' + 1;
181 i += 1;
182 if (i < s.len and s[i] == '$') {
183 r.ar = true;
184 i += 1;
185 }
186 const start = i;
187 if (start == s.len or s[start] == '0') return null;
188 while (i < s.len and std.ascii.isDigit(s[i])) i += 1;
189 if (i != s.len) return null;
190 r.r = std.fmt.parseInt(i32, s[start..], 10) catch return null;
191 return if (valid(r)) r else null;
192}

194fn lex(s: S) ![]Token {
195 var ts: std.ArrayList(Token) = .empty;
196 var i: usize = 1;
197 while (i < s.len) {
198 if (s[i] == ' ') {
199 i += 1;
200 continue;
201 }
202 const start = i;
203 const c = s[i];
204 var t: Token = .{ .k = .sym, .s = "", .p = i + 1 };
205 if (std.ascii.isDigit(c)) {
206 while (i < s.len and std.ascii.isDigit(s[i])) i += 1;
207 if (i - start > 9) return fail("syntax error at {d}", .{start + 1});
208 t.k = .num;
209 } else if (c == '"') {
210 i += 1;
211 const begin = i;
212 while (i < s.len and s[i] != '"') i += 1;
213 if (i == s.len) return fail("syntax error at {d}", .{start + 1});
214 t.k = .str;
215 t.s = s[begin..i];
216 i += 1;
217 try ts.append(A, t);
218 continue;
219 } else if (std.ascii.isAlphabetic(c) or c == '$') {
220 while (i < s.len and (std.ascii.isAlphanumeric(s[i]) or s[i] == '$')) i += 1;
221 const word = try A.dupe(u8, s[start..i]);
222 for (word) |*ch| ch.* = std.ascii.toUpper(ch.*);
223 t.s = word;
224 if (reference(word)) |r| {
225 t.k = .ref;
226 t.a = r;
227 } else {
228 if (!std.ascii.isUpper(word[0])) return fail("syntax error at {d}", .{start + 1});
229 for (word) |ch| if (!std.ascii.isUpper(ch) and !std.ascii.isDigit(ch)) return fail("syntax error at {d}", .{start + 1});
230 var j = i;
231 while (j < s.len and s[j] == ' ') j += 1;
232 t.k = if (j < s.len and s[j] == '(') .fun else .name;
233 }
234 try ts.append(A, t);
235 continue;
236 } else {
237 if (std.mem.indexOfScalar(u8, "+-*/&(),:=<>", c) == null) return fail("syntax error at {d}", .{i + 1});
238 i += 1;
239 if (i < s.len and ((c == '<' and (s[i] == '>' or s[i] == '=')) or (c == '>' and s[i] == '='))) i += 1;
240 }
241 t.s = s[start..i];
242 try ts.append(A, t);
243 }
244 try ts.append(A, .{ .k = .end, .s = "", .p = s.len + 1 });
245 return ts.toOwnedSlice(A);
246}

248const Parser = struct {
249 ts: []Token,
250 i: usize = 0,
251 fn accept(p: *Parser, s: S) bool {
252 if (!eq(p.ts[p.i].s, s)) return false;
253 p.i += 1;
254 return true;
255 }
256 fn bad(p: *Parser) anyerror {
257 return fail("syntax error at {d}", .{p.ts[p.i].p});
258 }
259 fn primary(p: *Parser) anyerror!*Node {
260 const t = p.ts[p.i];
261 if (p.accept("-")) return node(.{ .k = .neg, .args = try A.dupe(*Node, &.{try p.primary()}) });
262 if (p.accept("(")) {
263 const a = try p.expression(1);
264 if (!p.accept(")")) return p.bad();
265 return a;
266 }
267 p.i += 1;
268 switch (t.k) {
269 .num => return node(.{ .k = .num, .n = try std.fmt.parseInt(i64, t.s, 10) }),
270 .str => return node(.{ .k = .str, .s = t.s }),
271 .name => return node(.{ .k = .name, .s = t.s }),
272 .ref => {
273 if (p.accept(":")) {
274 const b = p.ts[p.i];
275 if (b.k != .ref) return p.bad();
276 p.i += 1;
277 const r = normalize(t.a, b.a);
278 return node(.{ .k = .range, .a = r.a, .b = r.b });
279 }
280 return node(.{ .k = .ref, .a = t.a });
281 },
282 .fun => {
283 if (!p.accept("(")) return p.bad();
284 var args: std.ArrayList(*Node) = .empty;
285 if (!p.accept(")")) {
286 while (true) {
287 try args.append(A, try p.expression(1));
288 if (p.accept(")")) break;
289 if (!p.accept(",")) return p.bad();
290 }
291 }
292 return node(.{ .k = .call, .s = t.s, .args = try args.toOwnedSlice(A) });
293 },
294 else => {
295 p.i -= 1;
296 return p.bad();
297 },
298 }
299 }
300 fn expression(p: *Parser, min: u8) anyerror!*Node {
301 var n = try p.primary();
302 while (p.ts[p.i].k == .sym) {
303 const op = p.ts[p.i].s;
304 if (eq(op, ")") or eq(op, ",") or eq(op, ":") or eq(op, "(")) break;
305 const pr = precedence(op);
306 if (pr < min) break;
307 p.i += 1;
308 const r = try p.expression(pr + 1);
309 n = try node(.{ .k = .binary, .s = op, .args = try A.dupe(*Node, &.{ n, r }) });
310 }
311 return n;
312 }

338 switch (n.k) {

340 .range => {

349fn visit(i: usize) anyerror!void {
350 visited[i] = 1;
351 discovery[i] = clock;
352 lowlink[i] = clock;
353 clock += 1;
354 try stack.append(A, i);
355 for (0..size) |j| if (deps[i][j] and cells[j].formula) {
356 if (visited[j] == 0) {
357 try visit(j);
358 lowlink[i] = @min(lowlink[i], lowlink[j]);
359 } else if (visited[j] == 1) lowlink[i] = @min(lowlink[i], discovery[j]);
360 };
361 if (lowlink[i] == discovery[i]) {
362 var component: std.ArrayList(usize) = .empty;
363 while (true) {
364 const j = stack.pop().?;
365 visited[j] = 2;
366 try component.append(A, j);
367 if (j == i) break;
368 }
369 if (component.items.len > 1 or deps[i][i]) for (component.items) |j| {
370 cycles[j] = true;
371 };
372 }
373}

375fn cellValue(i: usize) anyerror!Value {
376 if (done[i]) return values[i];
377 done[i] = true;
378 if (cycles[i]) {
379 values[i] = err("#CYCLE!");
380 return values[i];
381 }
382 if (cells[i].n) |n| {
383 var v = try eval(n);
384 if (cells[i].formula) {
385 if (v.k == .empty) v = num(0);
386 if (v.k == .range) v = err("#VALUE!");
387 }
388 values[i] = v;
389 } else values[i] = .{};
390 return values[i];
391}

393fn eval(n: *Node) anyerror!Value {
394 switch (n.k) {
395 .num => return num(n.n),
396 .str => return .{ .k = .str, .s = n.s },
397 .name => return err("#NAME?"),
398 .invalid => return err("#REF!"),
399 .ref => return cellValue(idx(n.a)),
400 .range => return .{ .k = .range, .a = n.a, .b = n.b },
401 .neg => {
402 const v = numeric(try eval(n.args[0]));
403 return if (v.k == .err) v else num(-v.n);
404 },
405 .binary => {
406 var x = try eval(n.args[0]);
407 var y = try eval(n.args[1]);
408 if (x.k == .err) return x;
409 if (y.k == .err) return y;
410 if (eq(n.s, "&")) {
411 x = try string(x);
412 y = try string(y);
413 if (x.k == .err) return x;
414 if (y.k == .err) return y;
415 return .{ .k = .str, .s = try fmt("{s}{s}", .{ x.s, y.s }) };
416 }
417 if (precedence(n.s) >= 3) {
418 x = numeric(x);
419 y = numeric(y);
420 if (x.k == .err) return x;
421 if (y.k == .err) return y;
422 if (eq(n.s, "+")) return num(x.n + y.n);
423 if (eq(n.s, "-")) return num(x.n - y.n);
424 if (eq(n.s, "*")) return num(x.n * y.n);
425 return if (y.n == 0) err("#DIV/0!") else num(@divTrunc(x.n, y.n));
426 }
427 if (x.k == .range or y.k == .range) return err("#VALUE!");
428 var cmp: i32 = 0;
429 if (x.k == .str and y.k == .num) cmp = 1 else if (x.k == .num and y.k == .str) cmp = -1 else if (x.k == .str or y.k == .str) {
430 const ord = std.mem.order(u8, x.s, y.s);
431 cmp = if (ord == .lt) -1 else if (ord == .gt) 1 else 0;
432 } else cmp = if (x.n < y.n) -1 else if (x.n > y.n) 1 else 0;
433 return num(@intFromBool(if (eq(n.s, "=")) cmp == 0 else if (eq(n.s, "<>")) cmp != 0 else if (eq(n.s, "<")) cmp < 0 else if (eq(n.s, "<=")) cmp <= 0 else if (eq(n.s, ">")) cmp > 0 else cmp >= 0));
434 },
435 .call => return function(n),
436 }
437}

439fn function(n: *Node) anyerror!Value {
440 const s = n.s;
441 const aggregate = eq(s, "SUM") or eq(s, "MIN") or eq(s, "MAX") or eq(s, "COUNT") or eq(s, "AVERAGE") or eq(s, "AND") or eq(s, "OR") or eq(s, "CONCAT");
442 if (!aggregate and !eq(s, "IF") and !eq(s, "ABS") and !eq(s, "NOT") and !eq(s, "LEN")) return err("#NAME?");
443 if ((aggregate and n.args.len == 0) or (eq(s, "IF") and n.args.len != 2 and n.args.len != 3) or (!aggregate and !eq(s, "IF") and n.args.len != 1)) return err("#VALUE!");
444 if (eq(s, "IF")) {
445 const c = numeric(try eval(n.args[0]));
446 if (c.k == .err) return c;
447 return if (c.n != 0) eval(n.args[1]) else if (n.args.len == 3) eval(n.args[2]) else num(0);
448 }
449 if (!aggregate) {
450 var x = try eval(n.args[0]);
451 if (eq(s, "LEN")) {
452 x = try string(x);
453 return if (x.k == .err) x else num(@intCast(x.s.len));
454 }
455 x = numeric(x);
456 if (x.k == .err) return x;
457 return num(if (eq(s, "NOT")) @intFromBool(x.n == 0) else @intCast(@abs(x.n)));
458 }
459 var items: std.ArrayList(Value) = .empty;
460 for (n.args) |a| {
461 const v = try eval(a);
462 if (v.k == .range) {
463 for (0..size) |i| if (inside(at(i), .{ .a = v.a, .b = v.b })) {
464 try items.append(A, try cellValue(i));
465 };
466 } else try items.append(A, v);
467 }
468 var count: i64 = 0;
469 var sum: i64 = 0;
470 var min: i64 = 0;
471 var max: i64 = 0;
472 var all = true;
473 var any = false;
474 var text: S = "";
475 for (items.items) |v| {
476 if (v.k == .err and !eq(s, "COUNT")) return v;
477 if (v.k == .num) {
478 if (count == 0) {
479 min = v.n;
480 max = v.n;
481 }
482 count += 1;
483 sum += v.n;
484 min = @min(min, v.n);
485 max = @max(max, v.n);
486 all = all and v.n != 0;
487 any = any or v.n != 0;
488 }
489 if (eq(s, "CONCAT")) {
490 const x = try string(v);
491 text = try fmt("{s}{s}", .{ text, x.s });
492 }
493 }
494 if (eq(s, "COUNT")) return num(count);
495 if (eq(s, "CONCAT")) return .{ .k = .str, .s = text };
496 if (eq(s, "MIN")) return num(min);
497 if (eq(s, "MAX")) return num(max);
498 if (eq(s, "AVERAGE")) return if (count == 0) err("#DIV/0!") else num(@divTrunc(sum, count));
499 if (eq(s, "AND") or eq(s, "OR")) return if (count == 0) err("#VALUE!") else num(@intFromBool(if (eq(s, "AND")) all else any));
500 return num(sum);
501}

503fn recalc() !void {
504 for (0..size) |i| {
505 @memset(&deps[i], false);
506 if (cells[i].formula) dependencies(cells[i].n.?, &deps[i]);

521const Rewrite = struct { mode: enum { copy, move, insert, delete }, source: Rect = .{ .a = .{ .c = 0, .r = 0 }, .b = .{ .c = 0, .r = 0 } }, dest: Rect = .{ .a = .{ .c = 0, .r = 0 }, .b = .{ .c = 0, .r = 0 } }, dc: i32 = 0, dr: i32 = 0, col: bool = false, pos: i32 = 0 };
522
523fn rewrite(n: *Node, rule: Rewrite) anyerror!*Node {
524 var x = n.*;
525 if (x.k == .ref or x.k == .range) {
526 var rs = [_]Ref{ x.a, x.b };
527 var invalid = false;
528 const range = x.k == .range;
529 if (rule.mode == .move and range and !(inside(rs[0], rule.source) and inside(rs[1], rule.source))) return n;
530 for (rs[0..if (range) @as(usize, 2) else 1], 0..) |*r, i| {
531 switch (rule.mode) {
532 .copy => {
533 if (!r.ac) r.c += rule.dc;
534 if (!r.ar) r.r += rule.dr;
535 },
536 .move => {
537 if (inside(r.*, rule.source)) {
538 r.c += rule.dc;
539 r.r += rule.dr;
540 } else if (!range and inside(r.*, rule.dest)) invalid = true;
541 },
542 .insert => {
543 const v = if (rule.col) &r.c else &r.r;
544 if (v.* >= rule.pos) v.* += 1;
545 },
546 .delete => {
547 const v = if (rule.col) &r.c else &r.r;
548 if (!range) {
549 if (v.* == rule.pos) invalid = true;
550 if (v.* > rule.pos) v.* -= 1;
551 } else {
552 const v0 = if (rule.col) x.a.c else x.a.r;
553 const v1 = if (rule.col) x.b.c else x.b.r;
554 if (v0 == rule.pos and v1 == rule.pos) invalid = true;
555 if (v.* > rule.pos or (i == 1 and v.* == rule.pos)) v.* -= 1;
556 }
557 },
558 }
559 if (!valid(r.*)) invalid = true;
560 }
561 if (invalid) return node(.{ .k = .invalid });
562 if (range) {
563 const r = normalize(rs[0], rs[1]);
564 x.a = r.a;
565 x.b = r.b;
566 } else x.a = rs[0];
567 } else if (x.args.len > 0) {
568 const args = try A.alloc(*Node, x.args.len);
569 for (x.args, 0..) |v, i| args[i] = try rewrite(v, rule);
570 x.args = args;
571 }
572 return node(x);
573}

592fn table(r: Rect) !void {
593 var widths: [26]usize = @as([26]usize, @splat(1));
594 for (0..size) |i| if (inside(at(i), r)) {
595 widths[i % 26] = @max(widths[i % 26], (try display(values[i], false)).len);
596 };
597 const rw: usize = if (r.b.r >= 10) 2 else 1;
598 var sep: std.ArrayList(u8) = .empty;
599 try sep.append(A, '+');
600 try sep.appendNTimes(A, '-', rw);
601 try sep.append(A, '+');
602 var c = r.a.c;
603 while (c <= r.b.c) : (c += 1) {
604 try sep.appendNTimes(A, '-', widths[@intCast(c - 1)]);
605 try sep.append(A, '+');
606 }
607 try line(sep.items);
608 try put("|");
609 try pad(rw, ' ');
610 try put("|");
611 c = r.a.c;
612 while (c <= r.b.c) : (c += 1) {
613 try put(try fmt("{c}", .{@as(u8, @intCast(c - 1)) + 'A'}));
614 try pad(widths[@intCast(c - 1)] - 1, ' ');
615 try put("|");
616 }
617 try put("\n");
618 try line(sep.items);
619 var row = r.a.r;
620 while (row <= r.b.r) : (row += 1) {
621 const rs = try fmt("{d}", .{row});
622 try put("|");
623 try pad(rw - rs.len, ' ');
624 try put(rs);
625 try put("|");
626 c = r.a.c;
627 while (c <= r.b.c) : (c += 1) {
628 const v = values[idx(.{ .r = row, .c = c })];
629 const s = try display(v, false);
630 const w = widths[@intCast(c - 1)];
631 if (v.k == .num) try pad(w - s.len, ' ');
632 try put(s);
633 if (v.k != .num) try pad(w - s.len, ' ');
634 try put("|");
635 }
636 try put("\n");
637 }
638 try line(sep.items);
639}

641fn command(text: S) anyerror!void {
642 const split = std.mem.indexOfScalar(u8, text, ' ');
643 const c = if (split) |i| text[0..i] else text;
644 const rest = if (split) |i| text[i + 1 ..] else "";
645 const space = std.mem.indexOfScalar(u8, rest, ' ');
646 const a = if (space) |i| rest[0..i] else rest;
647 const b = if (space) |i| rest[i + 1 ..] else "";
648 if (eq(c, "LIST")) {
649 var any = false;
650 for (cells, 0..) |v, i| if (v.n != null) {
651 any = true;
652 try show(i);
653 };
654 if (!any) try line("(no cells)");
655 return;
656 }
657 if (eq(c, "SHOW") or eq(c, "DEPS")) {
658 const p = addr(a) orelse return fail("bad address", .{});
659 const i = idx(p);
660 if (eq(c, "SHOW")) {
661 try show(i);
662 return;
663 }
664 var pre: S = "";
665 var dep: S = "";
666 for (0..size) |j| {
667 if (deps[i][j]) pre = try fmt("{s}{s}{s}", .{ pre, if (pre.len == 0) @as(S, "") else ",", try refText(at(j)) });
668 if (deps[j][i]) dep = try fmt("{s}{s}{s}", .{ dep, if (dep.len == 0) @as(S, "") else ",", try refText(at(j)) });
669 }
670 try line(try fmt("{s}: precedents={s} dependents={s}", .{ a, if (pre.len == 0) @as(S, "none") else pre, if (dep.len == 0) @as(S, "none") else dep }));
671 return;
672 }
673 if (eq(c, "PRINT")) {
674 try table(rectangle(a) orelse return fail("bad address", .{}));
675 return;
676 }
677 const old = values;
678 var target: ?Ref = null;
679 var newcell: Cell = .{};
680 var rect: Rect = undefined;
681 var dest: Rect = undefined;
682 var rule: Rewrite = undefined;
683 var axis: i32 = 0;
684 const row = eq(c, "INSERTROW") or eq(c, "DELETEROW");
685 const col = eq(c, "INSERTCOL") or eq(c, "DELETECOL");
686 const insert = eq(c, "INSERTROW") or eq(c, "INSERTCOL");
687 if (eq(c, "SET")) {
688 target = addr(a) orelse return fail("bad address", .{});
689 if (b.len > 0 and b[0] == '=') {
690 var p = Parser{ .ts = try lex(b) };
691 const n = try p.expression(1);
692 if (p.ts[p.i].k != .end) return p.bad();
693 newcell = .{ .n = n, .formula = true };
694 } else if (b.len >= 2 and b[0] == '"' and b[b.len - 1] == '"' and std.mem.indexOfScalar(u8, b[1 .. b.len - 1], '"') == null) newcell.n = try node(.{ .k = .str, .s = b[1 .. b.len - 1] }) else {
695 const off: usize = if (b.len > 0 and b[0] == '-') 1 else 0;
696 if (!eq(b, "0")) {
697 if (b.len <= off or b.len - off > 9 or b[off] < '1' or b[off] > '9') return fail("bad value", .{});
698 for (b[off..]) |ch| if (!std.ascii.isDigit(ch)) return fail("bad value", .{});
699 }
700 newcell.n = try node(.{ .k = .num, .n = std.fmt.parseInt(i64, b, 10) catch return fail("bad value", .{}) });
701 }
702 } else if (eq(c, "CLEAR")) rect = rectangle(a) orelse return fail("bad address", .{}) else if (eq(c, "COPY") or eq(c, "MOVE")) {
703 rect = rectangle(a) orelse return fail("bad address", .{});
704 const p = addr(b) orelse return fail("bad address", .{});
705 const dc = p.c - rect.a.c;
706 const dr = p.r - rect.a.r;
707 dest = .{ .a = p, .b = .{ .c = rect.b.c + dc, .r = rect.b.r + dr } };
708 if (!valid(dest.b)) return fail("out of grid", .{});
709 rule = .{ .mode = if (eq(c, "COPY")) .copy else .move, .source = rect, .dest = dest, .dc = dc, .dr = dr };
710 } else if (row or col) {
711 if (row) {
712 const p = addr(try fmt("A{s}", .{a})) orelse return fail("bad address", .{});
713 axis = p.r;
714 } else {
715 if (a.len != 1 or a[0] < 'A' or a[0] > 'Z') return fail("bad address", .{});
716 axis = a[0] - 'A' + 1;
717 }
718 if (insert) {
719 for (cells, 0..) |v, i| if (v.n != null and (if (col) at(i).c == 26 else at(i).r == 99)) return fail("last {s} not empty", .{if (col) @as(S, "column") else "row"});
720 }
721 rule = .{ .mode = if (insert) .insert else .delete, .col = col, .pos = axis };
722 } else if (eq(c, "UNDO")) {
723 if (history.items.len == 0) return fail("nothing to undo", .{});
724 const h = history.pop().?;
725 @memcpy(&cells, h.cells);
726 try recalc();
727 var changed: usize = 0;
728 for (values, old) |v, x| if (!equal(v, x)) {
729 changed += 1;
730 };
731 try line(try fmt("UNDO {s} changed={d}", .{ h.name, changed }));
732 return;
733 } else return fail("unknown command", .{});
734 try history.append(A, .{ .cells = try A.dupe(Cell, &cells), .name = c });
735 if (eq(c, "SET")) cells[idx(target.?)] = newcell else if (eq(c, "CLEAR")) {
736 for (0..size) |i| if (inside(at(i), rect)) {
737 cells[i] = .{};
738 };
739 } else if (eq(c, "COPY") or eq(c, "MOVE")) {
740 if (eq(c, "MOVE")) {
741 for (&cells) |*v| if (v.formula) {
742 v.n = try rewrite(v.n.?, rule);
743 };
744 }
745 const original = cells;
746 if (eq(c, "MOVE")) for (0..size) |i| {
747 if (inside(at(i), rect)) cells[i] = .{};
748 };
749 for (0..size) |i| if (inside(at(i), rect)) {
750 const p = at(i);
751 const j = idx(.{ .c = p.c + rule.dc, .r = p.r + rule.dr });
752 cells[j] = original[i];
753 if (eq(c, "COPY") and cells[j].formula) cells[j].n = try rewrite(cells[j].n.?, rule);
754 };
755 } else {
756 const original = cells;
757 cells = @as([size]Cell, @splat(.{}));
758 for (0..size) |i| {
759 var p = at(i);
760 const v = if (col) &p.c else &p.r;
761 if (!insert and v.* == axis) continue;
762 if (v.* >= axis) v.* += if (insert) @as(i32, 1) else -1;
763 if (valid(p)) cells[idx(p)] = original[i];
764 }
765 for (&cells) |*v| if (v.formula) {
766 v.n = try rewrite(v.n.?, rule);
767 };
768 }
769 try recalc();
770 var changed: usize = 0;
771 for (values, old) |v, x| if (!equal(v, x)) {
772 changed += 1;
773 };
774 if (eq(c, "SET")) try line(try fmt("SET {s}={s} changed={d}", .{ a, try display(values[idx(target.?)], true), changed })) else try line(try fmt("{s} changed={d}", .{ c, changed }));
775}

777pub fn main(init: std.process.Init) !void {
778 var ib: [8192]u8 = undefined;
779 var input = std.Io.File.stdin().readerStreaming(init.io, &ib);
780 const text = try input.interface.allocRemaining(A, .unlimited);
781 var lines = std.mem.splitScalar(u8, text, '\n');
782 const n = try std.fmt.parseInt(usize, lines.next() orelse "0", 10);
783 try recalc();
784 for (0..n) |_| {
785 command(std.mem.trimEnd(u8, lines.next() orelse "", "\r")) catch |e| {
786 if (e != error.Invalid) return e;
787 try line(try fmt("ERROR {s}", .{why}));
788 };
789 }
790 var ob: [8192]u8 = undefined;
791 var sink = std.Io.File.stdout().writerStreaming(init.io, &ob);
792 try sink.interface.writeAll(out.items);
793 try sink.interface.flush();
794}
