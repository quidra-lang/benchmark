7struct Reference {
8 col: i32,
9 row: i32,

14 Empty,

27fn num(n: i64) -> Tree {
28 Rc::new(Node::Int(n))
29}
30fn text(s: String) -> Tree {
31 Rc::new(Node::Text(s))
32}
33fn error(s: &'static str) -> Tree {
34 Rc::new(Node::Error(s))
35}

37 Rc::new(Node::Invalid)
38}

40 Rc::new(Node::Empty)
41}
42fn id(r: &Reference) -> usize {
43 (r.row * 26 + r.col) as usize
44}
45fn label(i: usize) -> String {
46 format!("{}{}", (b'A' + (i % 26) as u8) as char, i / 26 + 1)
47}
48fn inside(r: &Reference) -> bool {
49 (0..26).contains(&r.col) && (0..99).contains(&r.row)
50}
51fn normalized(a: &Reference, b: &Reference) -> (Reference, Reference) {
52 let (mut a, mut b) = (a.clone(), b.clone());
53 if a.col > b.col {
54 std::mem::swap(&mut a.col, &mut b.col);
55 let ac = a.flags & 1;
56 let bc = b.flags & 1;
57 a.flags = (a.flags & 2) | bc;
58 b.flags = (b.flags & 2) | ac;
59 }
60 if a.row > b.row {
61 std::mem::swap(&mut a.row, &mut b.row);
62 let ar = a.flags & 2;
63 let br = b.flags & 2;
64 a.flags = (a.flags & 1) | br;
65 b.flags = (b.flags & 1) | ar;
66 }
67 (a, b)
68}
69#[derive(Clone)]
70struct Area {
71 a: Reference,
72 b: Reference,
73}
74impl Area {
75 fn cells(&self) -> Vec<usize> {
76 (self.a.row..=self.b.row)
77 .flat_map(|r| (self.a.col..=self.b.col).map(move |c| (r * 26 + c) as usize))
78 .collect()
79 }
80 fn contains(&self, r: &Reference) -> bool {
81 (self.a.col..=self.b.col).contains(&r.col) && (self.a.row..=self.b.row).contains(&r.row)
82 }
83}
84fn valid_row(s: &str) -> bool {
85 let b = s.as_bytes();
86 (b.len() == 1 || b.len() == 2)
87 && b[0] >= b'1'
88 && b[0] <= b'9'
89 && (b.len() == 1 || b[1].is_ascii_digit())
90}
91fn address(s: &str) -> Result<Reference, String> {
92 let b = s.as_bytes();
93 if b.len() < 2 || !b[0].is_ascii_uppercase() || !valid_row(&s[1..]) {
94 return Err("bad address".into());
95 }
96 Ok(Reference {
97 col: (b[0] - b'A') as i32,
98 row: s[1..].parse::<i32>().unwrap() - 1,
99 flags: 0,
100 })
101}
102fn area(s: &str) -> Result<Area, String> {
103 let p: Vec<_> = s.split(':').collect();
104 if p.len() > 2 {
105 return Err("bad address".into());
106 }
107 let a = address(p[0])?;
108 let b = if p.len() == 1 {
109 a.clone()
110 } else {
111 address(p[1])?
112 };
113 let (a, b) = normalized(&a, &b);
114 Ok(Area { a, b })
115}
116fn reference(s: &str) -> Option<Reference> {
117 let b = s.as_bytes();
118 let (mut i, mut f) = (0, 0);
119 if b.get(i) == Some(&b'$') {
120 f |= 1;
121 i += 1
122 }
123 let c = *b.get(i)?;
124 if !c.is_ascii_uppercase() {
125 return None;
126 }
127 i += 1;
128 if b.get(i) == Some(&b'$') {
129 f |= 2;
130 i += 1
131 }
132 if !valid_row(&s[i..]) {
133 return None;
134 }
135 Some(Reference {
136 col: (c - b'A') as i32,
137 row: s[i..].parse::<i32>().ok()? - 1,
138 flags: f,
139 })
140}
141fn precedence(op: &str) -> i32 {
142 match op {
143 "=" | "<>" | "<" | "<=" | ">" | ">=" => 0,
144 "&" => 1,
145 "+" | "-" => 2,
146 "*" | "/" => 3,
147 _ => -1,
148 }
149}
150#[derive(Clone)]
151enum Token {
152 Value(Tree),
153 Call(String),
154 Symbol(String),
155 End,
156}
157struct Parser {
158 tokens: Vec<(Token, usize)>,
159 at: usize,
160}
161fn syntax(p: usize) -> String {
162 format!("syntax error at {p}")
163}
164impl Parser {
165 fn new(s: &str) -> Result<Self, String> {
166 let bytes = s.as_bytes();
167 let mut i = 1;
168 let mut tokens = Vec::new();
169 while i < bytes.len() {
170 let (p, ch) = (i + 1, bytes[i]);
171 if ch == b' ' {
172 i += 1;
173 continue;
174 }
175 let token;
176 if ch.is_ascii_digit() {
177 let mut j = i;
178 while j < bytes.len() && bytes[j].is_ascii_digit() {
179 j += 1
180 }
181 if j - i >= 10 {
182 return Err(syntax(p));
183 }
184 token = Token::Value(num(s[i..j].parse().unwrap()));
185 i = j;
186 } else if ch == b'"' {
187 let j = s[i + 1..]
188 .find('"')
189 .map(|v| i + 1 + v)
190 .ok_or_else(|| syntax(p))?;
191 token = Token::Value(text(s[i + 1..j].into()));
192 i = j + 1;
193 } else if ch.is_ascii_alphabetic() || ch == b'$' {
194 let mut j = i;
195 while j < bytes.len() && (bytes[j].is_ascii_alphanumeric() || bytes[j] == b'$') {
196 j += 1
197 }
198 let w = s[i..j].to_ascii_uppercase();
199 i = j;
200 if let Some(r) = reference(&w) {
201 token = Token::Value(Rc::new(Node::Ref(r)))
202 } else if w.as_bytes()[0].is_ascii_uppercase()
203 && w.bytes().all(|c| c.is_ascii_alphanumeric())
204 {
205 token = if s[j..].trim_start_matches(' ').starts_with('(') {
206 Token::Call(w)
207 } else {
208 Token::Value(Rc::new(Node::Name(w)))
209 };
210 } else {
211 return Err(syntax(p));
212 }
213 } else {
214 let pair = &s[i..(i + 2).min(s.len())];
215 let op = if ["<>", "<=", ">="].contains(&pair) {
216 pair
217 } else {
218 &s[i..i + 1]
219 };
220 if !b"+-*/&(),:=<>".contains(&ch) {
221 return Err(syntax(p));
222 }
223 token = Token::Symbol(op.into());
224 i += op.len();
225 }
226 tokens.push((token, p));
227 }
228 tokens.push((Token::End, s.len() + 1));
229 Ok(Self { tokens, at: 0 })
230 }
231 fn take(&mut self, op: &str) -> bool {
232 if matches!(&self.tokens[self.at].0,Token::Symbol(s) if s==op) {
233 self.at += 1;
234 true
235 } else {
236 false
237 }
238 }
239 fn expect(&mut self, op: &str) -> Result<(), String> {
240 if self.take(op) {
241 Ok(())
242 } else {
243 Err(syntax(self.tokens[self.at].1))
244 }
245 }
246 fn expression(&mut self, level: i32) -> Result<Tree, String> {
247 if level == 4 {
248 return self.primary();
249 }
250 let mut a = self.expression(level + 1)?;
251 while let Token::Symbol(op) = self.tokens[self.at].0.clone() {
252 if precedence(&op) != level {
253 break;
254 }
255 self.at += 1;
256 let b = self.expression(level + 1)?;
257 a = Rc::new(Node::Binary(op, a, b));
258 }
259 Ok(a)
260 }
261 fn primary(&mut self) -> Result<Tree, String> {
262 if self.take("-") {
263 return Ok(Rc::new(Node::Neg(self.primary()?)));
264 }
265 let (t, p) = self.tokens[self.at].clone();
266 self.at += 1;
267 match t {
268 Token::Value(x) => {
269 if let Node::Ref(a) = x.as_ref() {
270 if self.take(":") {
271 let Token::Value(y) = self.tokens[self.at].0.clone() else {
272 return Err(syntax(self.tokens[self.at].1));
273 };
274 let Node::Ref(b) = y.as_ref() else {
275 return Err(syntax(self.tokens[self.at].1));
276 };
277 self.at += 1;
278 let (a, b) = normalized(a, b);
279 return Ok(Rc::new(Node::Range(a, b)));
280 }
281 }
282 Ok(x)
283 }
284 Token::Call(name) => {
285 self.expect("(")?;
286 let mut args = Vec::new();
287 if !self.take(")") {
288 args.push(self.expression(0)?);
289 while self.take(",") {
290 args.push(self.expression(0)?)
291 }
292 self.expect(")")?
293 }
294 Ok(Rc::new(Node::Call(name, args)))
295 }
296 Token::Symbol(op) if op == "(" => {
297 let a = self.expression(0)?;
298 self.expect(")")?;
299 Ok(a)
300 }
301 _ => Err(syntax(p)),
302 }
303 }
304 fn parse(mut self) -> Result<Tree, String> {
305 let a = self.expression(0)?;
306 if !matches!(self.tokens[self.at].0, Token::End) {
307 return Err(syntax(self.tokens[self.at].1));
308 }
309 Ok(a)
310 }
311}
312fn reference_text(r: &Reference) -> String {
313 format!(
314 "{}{}{}{}",
315 if r.flags & 1 != 0 { "$" } else { "" },
316 (b'A' + r.col as u8) as char,
317 if r.flags & 2 != 0 { "$" } else { "" },
318 r.row + 1
319 )
320}
321fn canonical(x: &Tree) -> String {
322 match x.as_ref() {
323 Node::Int(n) => n.to_string(),
324 Node::Text(s) => format!("\"{s}\""),
325 Node::Name(s) => s.clone(),
326 Node::Invalid => "#REF!".into(),
327 Node::Ref(r) => reference_text(r),
328 Node::Range(a, b) => format!("{}:{}", reference_text(a), reference_text(b)),
329 Node::Call(name, args) => format!(
330 "{}({})",
331 name,
332 args.iter().map(canonical).collect::<Vec<_>>().join(",")
333 ),
334 Node::Neg(a) => {
335 if matches!(a.as_ref(), Node::Binary(..)) {
336 format!("-({})", canonical(a))
337 } else {
338 format!("-{}", canonical(a))
339 }
340 }
341 Node::Binary(op, a, b) => {
342 let (mut left, mut right) = (canonical(a), canonical(b));
343 if let Node::Binary(p, ..) = a.as_ref() {
344 if precedence(p) < precedence(op) {
345 left = format!("({left})")
346 }
347 }
348 if let Node::Binary(p, ..) = b.as_ref() {
349 if precedence(p) <= precedence(op) {
350 right = format!("({right})")
351 }
352 }
353 format!("{left}{op}{right}")
354 }
355 _ => unreachable!(),
356 }
357}
358fn display(x: &Tree) -> String {
359 match x.as_ref() {
360 Node::Empty => String::new(),
361 Node::Int(n) => n.to_string(),
362 Node::Text(s) => s.clone(),
363 Node::Error(s) => (*s).into(),
364 _ => unreachable!(),
365 }
366}
367fn notation(x: &Tree) -> String {
368 match x.as_ref() {
369 Node::Empty => "(empty)".into(),
370 Node::Formula(a) => format!("={}", canonical(a)),
371 Node::Text(s) => format!("\"{s}\""),
372 _ => display(x),
373 }
374}
375fn number(x: &Tree) -> Tree {
376 match x.as_ref() {
377 Node::Empty => num(0),
378 Node::Text(_) | Node::Range(..) => error("#VALUE!"),
379 _ => x.clone(),
380 }
381}
382fn string_value(x: &Tree) -> Tree {
383 match x.as_ref() {
384 Node::Empty => text(String::new()),
385 Node::Int(n) => text(n.to_string()),
386 Node::Range(..) => error("#VALUE!"),
387 _ => x.clone(),
388 }
389}
390fn rewrite(x: &Tree, f: &impl Fn(&Node) -> Tree) -> Tree {
391 match x.as_ref() {
392 Node::Ref(_) | Node::Range(..) => f(x.as_ref()),
393 Node::Formula(a) => Rc::new(Node::Formula(rewrite(a, f))),
394 Node::Call(name, args) => Rc::new(Node::Call(
395 name.clone(),
396 args.iter().map(|a| rewrite(a, f)).collect(),
397 )),
398 Node::Neg(a) => Rc::new(Node::Neg(rewrite(a, f))),
399 Node::Binary(op, a, b) => Rc::new(Node::Binary(op.clone(), rewrite(a, f), rewrite(b, f))),
400 _ => x.clone(),
401 }
402}
403fn dependencies(x: &Tree, out: &mut BTreeSet<usize>) {
404 match x.as_ref() {
405 Node::Ref(r) => {
406 out.insert(id(r));
407 }
408 Node::Range(a, b) => out.extend(
409 Area {
410 a: a.clone(),
411 b: b.clone(),
412 }
413 .cells(),
414 ),
415 Node::Call(_, args) => {
416 for a in args {
417 dependencies(a, out)
418 }
419 }
420 Node::Neg(a) | Node::Formula(a) => dependencies(a, out),
421 Node::Binary(_, a, b) => {
422 dependencies(a, out);
423 dependencies(b, out)
424 }
425 _ => (),
426 }
427}
428fn evaluate(x: &Tree, values: &[Tree]) -> Tree {
429 match x.as_ref() {
430 Node::Int(_) | Node::Text(_) | Node::Range(..) => x.clone(),
431 Node::Ref(r) => values[id(r)].clone(),
432 Node::Name(_) => error("#NAME?"),
433 Node::Invalid => error("#REF!"),
434 Node::Neg(a) => {
435 let a = number(&evaluate(a, values));
436 match a.as_ref() {
437 Node::Error(_) => a.clone(),
438 Node::Int(n) => num(-n),
439 _ => unreachable!(),
440 }
441 }
442 Node::Binary(op, a, b) => {
443 let (mut a, mut b) = (evaluate(a, values), evaluate(b, values));
444 if matches!(a.as_ref(), Node::Error(_)) {
445 return a;
446 }
447 if matches!(b.as_ref(), Node::Error(_)) {
448 return b;
449 }
450 if precedence(op) == 0 {
451 if matches!(a.as_ref(), Node::Range(..)) || matches!(b.as_ref(), Node::Range(..)) {
452 return error("#VALUE!");
453 }
454 let blank =
455 if matches!(a.as_ref(), Node::Text(_)) || matches!(b.as_ref(), Node::Text(_)) {
456 text(String::new())
457 } else {
458 num(0)
459 };
460 if matches!(a.as_ref(), Node::Empty) {
461 a = blank.clone()
462 }
463 if matches!(b.as_ref(), Node::Empty) {
464 b = blank
465 }
466 let c = match (a.as_ref(), b.as_ref()) {
467 (Node::Int(a), Node::Int(b)) => a.cmp(b),
468 (Node::Text(a), Node::Text(b)) => a.cmp(b),
469 (Node::Int(_), Node::Text(_)) => std::cmp::Ordering::Less,
470 _ => std::cmp::Ordering::Greater,
471 };
472 let yes = match op.as_str() {
473 "=" => c.is_eq(),
474 "<>" => !c.is_eq(),
475 "<" => c.is_lt(),
476 "<=" => !c.is_gt(),
477 ">" => c.is_gt(),
478 _ => !c.is_lt(),
479 };
480 return num(i64::from(yes));
481 }
482 if op == "&" {
483 a = string_value(&a);
484 b = string_value(&b)
485 } else {
486 a = number(&a);
487 b = number(&b)
488 }
489 if matches!(a.as_ref(), Node::Error(_)) {
490 return a;
491 }
492 if matches!(b.as_ref(), Node::Error(_)) {
493 return b;
494 }
495 if let (Node::Text(a), Node::Text(b)) = (a.as_ref(), b.as_ref()) {
496 return text(format!("{a}{b}"));
497 }
498 let (Node::Int(a), Node::Int(b)) = (a.as_ref(), b.as_ref()) else {
499 unreachable!()
500 };
501 match op.as_str() {
502 "+" => num(a + b),
503 "-" => num(a - b),
504 "*" => num(a * b),
505 _ => {
506 if *b == 0 {
507 error("#DIV/0!")
508 } else {
509 num(a / b)
510 }
511 }
512 }
513 }
514 Node::Call(name, args) => {
515 if [
516 "SUM", "MIN", "MAX", "COUNT", "AVERAGE", "AND", "OR", "CONCAT",
517 ]
518 .contains(&name.as_str())
519 {
520 if args.is_empty() {
521 return error("#VALUE!");
522 }
523 let mut items = Vec::new();
524 for a in args {
525 let v = evaluate(a, values);
526 if let Node::Range(a, b) = v.as_ref() {
527 items.extend(
528 Area {
529 a: a.clone(),
530 b: b.clone(),
531 }
532 .cells()
533 .iter()
534 .map(|i| values[*i].clone()),
535 )
536 } else {
537 items.push(v)
538 }
539 }
540 let ns: Vec<i64> = items
541 .iter()
542 .filter_map(|v| {
543 if let Node::Int(n) = v.as_ref() {
544 Some(*n)
545 } else {
546 None
547 }
548 })
549 .collect();
550 if name == "COUNT" {
551 return num(ns.len() as i64);
552 }
553 for v in &items {
554 if matches!(v.as_ref(), Node::Error(_)) {
555 return v.clone();
556 }
557 }
558 if name == "CONCAT" {
559 return text(
560 items
561 .iter()
562 .map(|v| display(&string_value(v)))
563 .collect::<String>(),
564 );
565 }
566 let sum: i64 = ns.iter().sum();
567 return match name.as_str() {
568 "SUM" => num(sum),
569 "MIN" => num(*ns.iter().min().unwrap_or(&0)),
570 "MAX" => num(*ns.iter().max().unwrap_or(&0)),
571 "AVERAGE" => {
572 if ns.is_empty() {
573 error("#DIV/0!")
574 } else {
575 num(sum / ns.len() as i64)
576 }
577 }
578 "AND" => {
579 if ns.is_empty() {
580 error("#VALUE!")
581 } else {
582 num(i64::from(ns.iter().all(|n| *n != 0)))
583 }
584 }
585 _ => {
586 if ns.is_empty() {
587 error("#VALUE!")
588 } else {
589 num(i64::from(ns.iter().any(|n| *n != 0)))
590 }
591 }
592 };
593 }
594 if name == "IF" {
595 if args.len() != 2 && args.len() != 3 {
596 return error("#VALUE!");
597 }
598 let c = number(&evaluate(&args[0], values));
599 return match c.as_ref() {
600 Node::Error(_) => c.clone(),
601 Node::Int(n) => {
602 if *n != 0 {
603 evaluate(&args[1], values)
604 } else if args.len() == 3 {
605 evaluate(&args[2], values)
606 } else {
607 num(0)
608 }
609 }
610 _ => unreachable!(),
611 };
612 }
613 if !["ABS", "NOT", "LEN"].contains(&name.as_str()) {
614 return error("#NAME?");
615 }
616 if args.len() != 1 {
617 return error("#VALUE!");
618 }
619 let v = evaluate(&args[0], values);
620 let v = if name == "LEN" {
621 string_value(&v)
622 } else {
623 number(&v)
624 };
625 match v.as_ref() {
626 Node::Error(_) => v.clone(),
627 Node::Text(s) => num(s.len() as i64),
628 Node::Int(n) => num(if name == "ABS" {
629 n.abs()
630 } else {
631 i64::from(*n == 0)
632 }),
633 _ => unreachable!(),
634 }
635 }
636 _ => unreachable!(),
637 }
638}
639struct Graph<'a> {
640 deps: &'a [Vec<usize>],
641 formulas: &'a [bool],
642 index: Vec<usize>,
643 low: Vec<usize>,
644 on: Vec<bool>,
645 cyclic: Vec<bool>,
646 stack: Vec<usize>,
647 serial: usize,
648}
649impl Graph<'_> {
650 fn visit(&mut self, i: usize) {
651 self.index[i] = self.serial;
652 self.low[i] = self.serial;
653 self.serial += 1;
654 self.stack.push(i);
655 self.on[i] = true;
656 for j in self.deps[i].clone() {
657 if self.formulas[j] {
658 if self.index[j] == usize::MAX {
659 self.visit(j);
660 self.low[i] = self.low[i].min(self.low[j])
661 } else if self.on[j] {
662 self.low[i] = self.low[i].min(self.index[j])
663 }
664 }
665 }
666 if self.low[i] == self.index[i] {
667 let mut part = Vec::new();
668 loop {
669 let j = self.stack.pop().unwrap();
670 self.on[j] = false;
671 part.push(j);
672 if j == i {
673 break;
674 }
675 }
676 if part.len() > 1 || self.deps[i].contains(&i) {
677 for j in part {
678 self.cyclic[j] = true
679 }
680 }
681 }
682 }
683}
684fn calculate(
685 i: usize,
686 contents: &[Tree],
687 deps: &[Vec<usize>],
688 cyclic: &[bool],
689 done: &mut [bool],
690 values: &mut [Tree],
691) {
692 let Node::Formula(tree) = contents[i].as_ref() else {
693 return;
694 };
695 if done[i] {
696 return;
697 }
698 done[i] = true;
699 if cyclic[i] {
700 values[i] = error("#CYCLE!");
701 return;
702 }
703 for j in &deps[i] {
704 calculate(*j, contents, deps, cyclic, done, values)
705 }
706 let v = evaluate(tree, values);
707 values[i] = match v.as_ref() {
708 Node::Empty => num(0),
709 Node::Range(..) => error("#VALUE!"),
710 _ => v.clone(),
711 };
712}
713struct Sheet {
714 contents: Vec<Tree>,
715 values: Vec<Tree>,
716 deps: Vec<Vec<usize>>,
717 history: Vec<(String, Vec<Tree>)>,
718}
719impl Sheet {
720 fn new() -> Self {
721 Self {
722 contents: vec![empty(); 2574],
723 values: vec![empty(); 2574],
724 deps: vec![Vec::new(); 2574],
725 history: Vec::new(),
726 }
727 }
728 fn replace(&mut self, next: Vec<Tree>) -> usize {
729 let old = self.values.clone();
730 self.contents = next;
731 let formulas: Vec<_> = self
732 .contents
733 .iter()
734 .map(|x| matches!(x.as_ref(), Node::Formula(_)))
735 .collect();
736 self.deps = self
737 .contents
738 .iter()
739 .map(|x| {
740 let mut out = BTreeSet::new();
741 if matches!(x.as_ref(), Node::Formula(_)) {
742 dependencies(x, &mut out)
743 }
744 out.into_iter().collect()
745 })
746 .collect();
747 self.values = self
748 .contents
749 .iter()
750 .map(|x| {
751 if matches!(x.as_ref(), Node::Formula(_)) {
752 empty()
753 } else {
754 x.clone()
755 }
756 })
757 .collect();
758 let mut graph = Graph {
759 deps: &self.deps,
760 formulas: &formulas,
761 index: vec![usize::MAX; 2574],
762 low: vec![0; 2574],
763 on: vec![false; 2574],
764 cyclic: vec![false; 2574],
765 stack: Vec::new(),
766 serial: 0,
767 };
768 for i in 0..2574 {
769 if formulas[i] && graph.index[i] == usize::MAX {
770 graph.visit(i)
771 }
772 }
773 let mut done = vec![false; 2574];
774 for i in 0..2574 {
775 calculate(
776 i,
777 &self.contents,
778 &self.deps,
779 &graph.cyclic,
780 &mut done,
781 &mut self.values,
782 )
783 }
784 old.iter().zip(&self.values).filter(|(a, b)| a != b).count()
785 }
786 fn describe(&self, i: usize) -> String {
787 format!(
788 "{}: {} -> {}",
789 label(i),
790 notation(&self.contents[i]),
791 notation(&self.values[i])
792 )
793 }
794 fn execute(&mut self, line: &str) -> Result<Vec<String>, String> {
795 let (name, rest) = line.split_once(' ').unwrap_or((line, ""));
796 match name {
797 "SHOW" => return Ok(vec![self.describe(id(&address(rest)?))]),
798 "LIST" => {
799 let out: Vec<_> = (0..2574)
800 .filter(|i| !matches!(self.contents[*i].as_ref(), Node::Empty))
801 .map(|i| self.describe(i))
802 .collect();
803 return Ok(if out.is_empty() {
804 vec!["(no cells)".into()]
805 } else {
806 out
807 });
808 }
809 "DEPS" => {
810 let i = id(&address(rest)?);
811 let ds: Vec<_> = (0..2574).filter(|j| self.deps[*j].contains(&i)).collect();
812 let fmt = |xs: &[usize]| {
813 if xs.is_empty() {
814 "none".into()
815 } else {
816 xs.iter().map(|i| label(*i)).collect::<Vec<_>>().join(",")
817 }
818 };
819 return Ok(vec![format!(
820 "{rest}: precedents={} dependents={}",
821 fmt(&self.deps[i]),
822 fmt(&ds)
823 )]);
824 }
825 "PRINT" => {
826 let g = area(rest)?;
827 let rw = (g.b.row + 1).to_string().len();
828 let widths: Vec<_> = (g.a.col..=g.b.col)
829 .map(|c| {
830 (g.a.row..=g.b.row)
831 .map(|r| display(&self.values[(r * 26 + c) as usize]).len())
832 .max()
833 .unwrap()
834 .max(1)
835 })
836 .collect();
837 let sep = format!(
838 "+{}+{}",
839 "-".repeat(rw),
840 widths
841 .iter()
842 .map(|w| "-".repeat(*w) + "+")
843 .collect::<String>()
844 );
845 let header = format!(
846 "|{}|{}",
847 " ".repeat(rw),
848 widths
849 .iter()
850 .enumerate()
851 .map(|(j, w)| format!(
852 "{:<width$}|",
853 (b'A' + g.a.col as u8 + j as u8) as char,
854 width = *w
855 ))
856 .collect::<String>()
857 );
858 let mut out = vec![sep.clone(), header, sep.clone()];
859 for r in g.a.row..=g.b.row {
860 let mut row = format!("|{:>width$}|", r + 1, width = rw);
861 for (j, w) in widths.iter().enumerate() {
862 let v = &self.values[(r * 26 + g.a.col) as usize + j];
863 let s = display(v);
864 row += &if matches!(v.as_ref(), Node::Int(_)) {
865 format!("{s:>width$}|", width = *w)
866 } else {
867 format!("{s:<width$}|", width = *w)
868 }
869 }
870 out.push(row)
871 }
872 out.push(sep);
873 return Ok(out);
874 }
875 "UNDO" => {
876 let (command, old) = self.history.pop().ok_or("nothing to undo")?;
877 return Ok(vec![format!(
878 "UNDO {command} changed={}",
879 self.replace(old)
880 )]);
881 }
882 _ => (),
883 }
884 let mut next = self.contents.clone();
885 let mut target = 0;
886 match name {
887 "SET" => {
888 let (a, s) = rest.split_once(' ').unwrap();
889 target = id(&address(a)?);
890 next[target] = if s.starts_with('=') {
891 Rc::new(Node::Formula(Parser::new(s)?.parse()?))
892 } else if s.len() >= 2
893 && s.starts_with('"')
894 && s.ends_with('"')
895 && !s[1..s.len() - 1].contains('"')
896 {
897 text(s[1..s.len() - 1].into())
898 } else if s == "0" || {
899 let d = s.strip_prefix('-').unwrap_or(s);
900 d.len() <= 9
901 && !d.is_empty()
902 && d.as_bytes()[0] >= b'1'
903 && d.as_bytes()[0] <= b'9'
904 && d.bytes().all(|c| c.is_ascii_digit())
905 } {
906 num(s.parse().unwrap())
907 } else {
908 return Err("bad value".into());
909 };
910 }
911 "CLEAR" => {
912 for i in area(rest)?.cells() {
913 next[i] = empty()
914 }
915 }
916 "COPY" | "MOVE" => {
917 let (a, b) = rest.split_once(' ').unwrap();
918 let source = area(a)?;
919 let dest = address(b)?;
920 let (dc, dr) = (dest.col - source.a.col, dest.row - source.a.row);
921 let shift = |r: &Reference, relative: bool| Reference {
922 col: r.col + if relative && r.flags & 1 != 0 { 0 } else { dc },
923 row: r.row + if relative && r.flags & 2 != 0 { 0 } else { dr },
924 flags: r.flags,
925 };
926 let destination = Area {
927 a: shift(&source.a, false),
928 b: shift(&source.b, false),
929 };
930 if !inside(&destination.a) || !inside(&destination.b) {
931 return Err("out of grid".into());
932 }
933 let transform = |x: &Node| -> Tree {
934 match x {
935 Node::Ref(r) => {
936 if name == "COPY" {
937 let r = shift(r, true);
938 if inside(&r) {
939 Rc::new(Node::Ref(r))
940 } else {
941 invalid()
942 }
943 } else if source.contains(r) {
944 Rc::new(Node::Ref(shift(r, false)))
945 } else if destination.contains(r) {
946 invalid()
947 } else {
948 Rc::new(x.clone())
949 }
950 }
951 Node::Range(a, b) => {
952 if name == "MOVE" && !(source.contains(a) && source.contains(b)) {
953 return Rc::new(x.clone());
954 }
955 let (a, b) = (shift(a, name == "COPY"), shift(b, name == "COPY"));
956 if inside(&a) && inside(&b) {
957 let (a, b) = normalized(&a, &b);
958 Rc::new(Node::Range(a, b))
959 } else {
960 invalid()
961 }
962 }
963 _ => unreachable!(),
964 }
965 };
966 if name == "MOVE" {
967 next = next.iter().map(|x| rewrite(x, &transform)).collect()
968 }
969 let ids = source.cells();
970 let moving: Vec<_> = ids
971 .iter()
972 .map(|i| {
973 if name == "COPY" {
974 rewrite(&self.contents[*i], &transform)
975 } else {
976 next[*i].clone()
977 }
978 })
979 .collect();
980 if name == "MOVE" {
981 for i in &ids {
982 next[*i] = empty()
983 }
984 }
985 for (i, x) in ids.into_iter().zip(moving) {
986 next[(i as i32 + dr * 26 + dc) as usize] = x
987 }
988 }
989 "INSERTROW" | "DELETEROW" | "INSERTCOL" | "DELETECOL" => {
990 let row = name.ends_with("ROW");
991 let insert = name.starts_with("INSERT");
992 let limit = if row { 99 } else { 26 };
993 let at = if row {
994 if !valid_row(rest) {
995 return Err("bad address".into());
996 }
997 rest.parse::<i32>().unwrap() - 1
998 } else {
999 if rest.len() != 1 || !rest.as_bytes()[0].is_ascii_uppercase() {
1000 return Err("bad address".into());
1001 }
1002 (rest.as_bytes()[0] - b'A') as i32
1003 };
1004 let axis = |r: &Reference| if row { r.row } else { r.col };
1005 let change = |r: &Reference, v: i32| Reference {
1006 col: if row { r.col } else { v },
1007 row: if row { v } else { r.row },
1008 flags: r.flags,
1009 };
1010 if insert
1011 && (0..2574).any(|i| {
1012 !matches!(self.contents[i].as_ref(), Node::Empty)
1013 && (if row { i as i32 / 26 } else { i as i32 % 26 }) == limit - 1
1014 })
1015 {
1016 return Err(format!(
1017 "last {} not empty",
1018 if row { "row" } else { "column" }
1019 ));
1020 }
1021 let transform = |x: &Node| -> Tree {
1022 match x {
1023 Node::Ref(r) => {
1024 let v = axis(r);
1025 if !insert && v == at {
1026 return invalid();
1027 }
1028 let n = v + if insert {
1029 i32::from(v >= at)
1030 } else {
1031 -i32::from(v > at)
1032 };
1033 if n >= limit {
1034 invalid()
1035 } else {
1036 Rc::new(Node::Ref(change(r, n)))
1037 }
1038 }
1039 Node::Range(a, b) => {
1040 let (u, v) = (axis(a), axis(b));
1041 if !insert && u == at && v == at {
1042 return invalid();
1043 }
1044 let c = u + if insert {
1045 i32::from(u >= at)
1046 } else {
1047 -i32::from(u > at)
1048 };
1049 let d = v + if insert {
1050 i32::from(v >= at)
1051 } else {
1052 -i32::from(v >= at)
1053 };
1054 if d >= limit {
1055 invalid()
1056 } else {
1057 Rc::new(Node::Range(change(a, c), change(b, d)))
1058 }
1059 }
1060 _ => unreachable!(),
1061 }
1062 };
1063 next = vec![empty(); 2574];
1064 for i in 0..2574 {
1065 if matches!(self.contents[i].as_ref(), Node::Empty) {
1066 continue;
1067 }
1068 let r = Reference {
1069 col: (i % 26) as i32,
1070 row: (i / 26) as i32,
1071 flags: 0,
1072 };
1073 let v = axis(&r);
1074 if !insert && v == at {
1075 continue;
1076 }
1077 let t = change(
1078 &r,
1079 v + if insert {
1080 i32::from(v >= at)
1081 } else {
1082 -i32::from(v > at)
1083 },
1084 );
1085 next[id(&t)] = rewrite(&self.contents[i], &transform)
1086 }
1087 }
1088 _ => return Err("unknown command".into()),
1089 }
1090 self.history.push((name.into(), self.contents.clone()));
1091 let changed = self.replace(next);
1092 Ok(vec![format!(
1093 "{}{} changed={changed}",
1094 name,
1095 if name == "SET" {
1096 format!(" {}={}", label(target), notation(&self.values[target]))
1097 } else {
1098 String::new()
1099 }
1100 )])
1101 }
1102}
