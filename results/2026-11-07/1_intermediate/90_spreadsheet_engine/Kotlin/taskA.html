30fun range(a: Node, b: Node): Node {
31 var c = a.n
32 var d = b.n
33 var r = a.row
34 var t = b.row
35 var ac = a.flags and 1
36 var bc = b.flags and 1
37 var ar = a.flags and 2
38 var br = b.flags and 2
39 if (c > d) {
40 val z = c
41 c = d
42 d = z
43 val f = ac
44 ac = bc
45 bc = f
46 }
47 if (r > t) {
48 val z = r
49 r = t
50 t = z
51 val f = ar
52 ar = br
53 br = f
54 }
55 return Node('G', a = ref(c, r, ac or ar), b = ref(d, t, bc or br))
56}

58fun address(s: String): Node {
59 if (!Regex("[A-Z][1-9][0-9]?").matches(s)) throw Problem("bad address")
60 return ref(s[0] - 'A', s.substring(1).toInt() - 1)

64 val p = s.indexOf(':')
65 if (p < 0) {

69 val a = address(s.substring(0, p))

85class Parser(s: String) {
86 private val tokens = mutableListOf<Token>()
87 private var at = 0
88
89 private fun syntax(p: Int): Nothing = throw Problem("syntax error at $p")
90
91 init {
92 var i = 1
93 while (i < s.length) {
94 val p = i + 1
95 val ch = s[i]
96 if (ch == ' ') {
97 i++
98 continue
99 }
100 val x: Node
101 if (ch in '0'..'9') {
102 var j = i
103 while (j < s.length && s[j] in '0'..'9') j++
104 if (j - i >= 10) syntax(p)
105 x = num(s.substring(i, j).toInt())
106 i = j
107 } else if (ch == '"') {
108 val j = s.indexOf('"', i + 1)
109 if (j < 0) syntax(p)
110 x = text(s.substring(i + 1, j))
111 i = j + 1
112 } else if (ch.isLetter() || ch == '$') {
113 var j = i
114 while (j < s.length && (s[j].isLetterOrDigit() || s[j] == '$')) j++
115 val w = s.substring(i, j).uppercase()
116 i = j
117 val m = Regex("(\\$?)([A-Z])(\\$?)([1-9][0-9]?)").matchEntire(w)
118 x =
119 if (m != null)
120 ref(
121 m.groupValues[2][0] - 'A',
122 m.groupValues[4].toInt() - 1,
123 (if (m.groupValues[1].isEmpty()) 0 else 1) or
124 (if (m.groupValues[3].isEmpty()) 0 else 2),
125 )
126 else if (Regex("[A-Z][A-Z0-9]*").matches(w))
127 Node(if (s.substring(j).trimStart().startsWith("(")) 'C' else 'W', text = w)
128 else syntax(p)
129 } else {
130 var op = s.substring(i, minOf(i + 2, s.length))
131 if (op !in setOf("<>", "<=", ">=")) op = ch.toString()
132 if (ch !in "+-*/&(),:=<>") syntax(p)
133 x = Node('T', text = op)
134 i += op.length
135 }
136 tokens += Token(x, p)
137 }
138 tokens += Token(Node('Z'), s.length + 1)
139 }
140
141 private fun take(op: String): Boolean {
142 val x = tokens[at].value
143 if (x.kind == 'T' && x.text == op) {
144 at++
145 return true
146 }
147 return false
148 }
149
150 private fun expect(op: String) {
151 if (!take(op)) syntax(tokens[at].position)
152 }
153
154 private fun expression(level: Int = 0): Node {
155 if (level == 4) return primary()
156 var a = expression(level + 1)
157 while (tokens[at].value.kind == 'T' && tokens[at].value.text in levels[level]) {
158 val op = tokens[at++].value.text
159 a = Node('B', text = op, a = a, b = expression(level + 1))
160 }
161 return a
162 }
163
164 private fun primary(): Node {
165 if (take("-")) return Node('U', a = primary())
166 val t = tokens[at++]
167 val x = t.value
168 if (x.kind in "NSW") return x
169 if (x.kind == 'R') {
170 if (!take(":")) return x
171 if (tokens[at].value.kind != 'R') syntax(tokens[at].position)
172 return range(x, tokens[at++].value)
173 }
174 if (x.kind == 'C') {
175 expect("(")
176 val args = mutableListOf<Node>()
177 if (!take(")")) {
178 args += expression()
179 while (take(",")) args += expression()
180 expect(")")
181 }
182 return Node('F', text = x.text, args = args.toList())
183 }
184 if (x.kind == 'T' && x.text == "(") {
185 val a = expression()
186 expect(")")
187 return a
188 }
189 syntax(t.position)
190 }
191
192 fun parse(): Node {
193 val x = expression()
194 if (tokens[at].value.kind != 'Z') syntax(tokens[at].position)
195 return x
196 }
197}

199fun canonical(x: Node): String =
200 when (x.kind) {
201 'N' -> x.n.toString()
202 'S' -> "\"${x.text}\""
203 'W' -> x.text
204 'X' -> "#REF!"
205 'R' ->
206 (if (x.flags and 1 != 0) "$" else "") +
207 ('A'.code + x.n).toChar() +
208 (if (x.flags and 2 != 0) "$" else "") +
209 (x.row + 1)
210 'G' -> canonical(x.a!!) + ":" + canonical(x.b!!)
211 'F' -> x.text + "(" + x.args.joinToString(",", transform = ::canonical) + ")"
212 'U' -> "-" + (if (x.a!!.kind == 'B') "(" + canonical(x.a) + ")" else canonical(x.a))
213 else -> {
214 var a = canonical(x.a!!)
215 var b = canonical(x.b!!)
216 if (x.a.kind == 'B' && precedence(x.a.text) < precedence(x.text)) a = "($a)"
217 if (x.b.kind == 'B' && precedence(x.b.text) <= precedence(x.text)) b = "($b)"
218 a + x.text + b
219 }
220 }

222fun display(x: Node?) = if (x == null) "" else if (x.kind == 'N') x.n.toString() else x.text
223
224fun notation(x: Node?) =
225 if (x == null) "(empty)"
226 else if (x.kind == '=') "=" + canonical(x.a!!)
227 else if (x.kind == 'S') "\"${x.text}\"" else display(x)

229fun number(x: Node?) = if (x == null) num(0) else if (x.kind in "SG") errorValue("#VALUE!") else x
230
231fun stringValue(x: Node?) =
232 if (x == null) text("")
233 else if (x.kind == 'N') text(x.n.toString())
234 else if (x.kind == 'G') errorValue("#VALUE!") else x

236fun rewrite(x: Node?, f: (Node) -> Node): Node? {
237 if (x == null) return null
238 if (x.kind in "RG") return f(x)
239 return x.copy(a = rewrite(x.a, f), b = rewrite(x.b, f), args = x.args.map { rewrite(it, f)!! })
240}

242fun dependencies(x: Node?, out: MutableSet<Int> = mutableSetOf()): Set<Int> {
243 if (x == null) return out
244 when (x.kind) {
245 'R' -> out.add(id(x))
246 'G' -> out.addAll(cells(x))
247 else -> {
248 dependencies(x.a, out)
249 dependencies(x.b, out)
250 x.args.forEach { dependencies(it, out) }
251 }
252 }
253 return out
254}

256class Sheet {
257 private var contents = arrayOfNulls<Node>(2574)
258 private var values = arrayOfNulls<Node>(2574)
259 private var deps = List<Set<Int>>(2574) { emptySet() }
260 private val history = mutableListOf<Pair<String, Array<Node?>>>()

262 private fun evaluate(x: Node): Node? {
263 if (x.kind in "NSG") return x
264 if (x.kind == 'R') return values[id(x)]
265 if (x.kind == 'W') return errorValue("#NAME?")
266 if (x.kind == 'X') return errorValue("#REF!")
267 if (x.kind == 'U') {
268 val a = number(evaluate(x.a!!))
269 return if (a.kind == 'E') a else num(-a.n)
270 }
271 if (x.kind == 'B') {
272 var a = evaluate(x.a!!)
273 var b = evaluate(x.b!!)
274 val op = x.text
275 if (a?.kind == 'E') return a
276 if (b?.kind == 'E') return b
277 if (precedence(op) == 0) {
278 if (a?.kind == 'G' || b?.kind == 'G') return errorValue("#VALUE!")
279 val blank = if (a?.kind == 'S' || b?.kind == 'S') text("") else num(0)
280 a = a ?: blank
281 b = b ?: blank
282 val c =
283 if (a.kind != b.kind) {
284 if (a.kind == 'N') -1 else 1
285 } else if (a.kind == 'N') a.n.compareTo(b.n) else a.text.compareTo(b.text)
286 val yes =
287 when (op) {
288 "=" -> c == 0
289 "<>" -> c != 0
290 "<" -> c < 0
291 "<=" -> c <= 0
292 ">" -> c > 0
293 else -> c >= 0
294 }
295 return num(if (yes) 1 else 0)
296 }
297 val u = if (op == "&") stringValue(a) else number(a)
298 val v = if (op == "&") stringValue(b) else number(b)
299 if (u.kind == 'E') return u
300 if (v.kind == 'E') return v
301 return when (op) {
302 "&" -> text(u.text + v.text)
303 "+" -> num(u.n + v.n)
304 "-" -> num(u.n - v.n)
305 "*" -> num(u.n * v.n)
306 else -> if (v.n == 0) errorValue("#DIV/0!") else num(u.n / v.n)
307 }
308 }
309 val name = x.text
310 val args = x.args
311 if (name in setOf("SUM", "MIN", "MAX", "COUNT", "AVERAGE", "AND", "OR", "CONCAT")) {
312 if (args.isEmpty()) return errorValue("#VALUE!")
313 val items = args.flatMap {
314 val v = evaluate(it)
315 if (v?.kind == 'G') cells(v).map { i -> values[i] } else listOf(v)
316 }
317 val ns = items.filterNotNull().filter { it.kind == 'N' }.map { it.n }
318 if (name == "COUNT") return num(ns.size)
319 items
320 .firstOrNull { it?.kind == 'E' }
321 ?.let {
322 return it
323 }
324 return when (name) {
325 "SUM" -> num(ns.sum())
326 "MIN" -> num(ns.minOrNull() ?: 0)
327 "MAX" -> num(ns.maxOrNull() ?: 0)
328 "AVERAGE" -> if (ns.isEmpty()) errorValue("#DIV/0!") else num(ns.sum() / ns.size)
329 "AND" ->
330 if (ns.isEmpty()) errorValue("#VALUE!")
331 else num(if (ns.all { it != 0 }) 1 else 0)
332 "OR" ->
333 if (ns.isEmpty()) errorValue("#VALUE!")
334 else num(if (ns.any { it != 0 }) 1 else 0)
335 else -> text(items.joinToString("") { stringValue(it).text })
336 }
337 }
338 if (name == "IF") {
339 if (args.size !in 2..3) return errorValue("#VALUE!")
340 val c = number(evaluate(args[0]))
341 if (c.kind == 'E') return c
342 return if (c.n != 0) evaluate(args[1])
343 else if (args.size == 3) evaluate(args[2]) else num(0)
344 }
345 if (name !in setOf("ABS", "NOT", "LEN")) return errorValue("#NAME?")
346 if (args.size != 1) return errorValue("#VALUE!")
347 val v = if (name == "LEN") stringValue(evaluate(args[0])) else number(evaluate(args[0]))
348 return if (v.kind == 'E') v
349 else
350 num(
351 if (name == "LEN") v.text.length
352 else if (name == "ABS") kotlin.math.abs(v.n) else if (v.n == 0) 1 else 0,
353 )
354 }

356 private fun recalculate() {
357 deps = contents.map { if (it?.kind == '=') dependencies(it.a) else emptySet() }
358 values = contents.map { if (it?.kind == '=') null else it }.toTypedArray()
359 val indices = IntArray(2574) { -1 }
360 val low = IntArray(2574)
361 val on = BooleanArray(2574)
362 val cyclic = BooleanArray(2574)
363 val done = BooleanArray(2574)
364 val stack = mutableListOf<Int>()
365 var serial = 0
366 fun visit(i: Int) {
367 indices[i] = serial
368 low[i] = serial++
369 stack.add(i)
370 on[i] = true
371 for (j in deps[i]) if (contents[j]?.kind == '=') {
372 if (indices[j] < 0) {
373 visit(j)
374 low[i] = minOf(low[i], low[j])
375 } else if (on[j]) low[i] = minOf(low[i], indices[j])
376 }
377 if (low[i] == indices[i]) {
378 val part = mutableListOf<Int>()
379 var j: Int
380 do {
381 j = stack.removeAt(stack.lastIndex)
382 on[j] = false
383 part.add(j)
384 } while (j != i)
385 if (part.size > 1 || i in deps[i]) for (k in part) cyclic[k] = true
386 }
387 }
388 for (i in 0 until 2574) if (contents[i]?.kind == '=' && indices[i] < 0) visit(i)
389 fun calculate(i: Int) {
390 if (contents[i]?.kind != '=' || done[i]) return
391 done[i] = true
392 if (cyclic[i]) {
393 values[i] = errorValue("#CYCLE!")
394 return
395 }
396 deps[i].forEach(::calculate)
397 val v = evaluate(contents[i]!!.a!!)
398 values[i] = if (v == null) num(0) else if (v.kind == 'G') errorValue("#VALUE!") else v
399 }
400 for (i in 0 until 2574) calculate(i)
401 }

403 private fun replace(next: Array<Node?>): Int {
404 val old = values
405 contents = next
406 recalculate()
407 return old.indices.count { old[it] != values[it] }
408 }

415 fun execute(line: String): List<String> {
416 val p = line.indexOf(' ')
417 val name = if (p < 0) line else line.substring(0, p)
418 val rest = if (p < 0) "" else line.substring(p + 1)
419 when (name) {
420 "SHOW" -> return listOf(describe(id(address(rest))))
421 "LIST" -> {
422 val out = contents.indices.filter { contents[it] != null }.map(::describe)
423 return if (out.isEmpty()) listOf("(no cells)") else out
424 }
425 "DEPS" -> {
426 val i = id(address(rest))
427 return listOf(
428 "$rest: precedents=${formatted(deps[i])} dependents=${formatted(deps.indices.filter { i in deps[it] })}",
429 )
430 }
431 "PRINT" -> {
432 val g = area(rest)
433 val rw = (g.b!!.row + 1).toString().length
434 val cs = g.a!!.n..g.b.n
435 val rs = g.a.row..g.b.row
436 val ws = cs.map { c ->
437 maxOf(1, rs.maxOf { r -> display(values[r * 26 + c]).length })
438 }
439 val sep = "+" + "-".repeat(rw) + "+" + ws.joinToString("") { "-".repeat(it) + "+" }
440 val head =
441 "|" +
442 " ".repeat(rw) +
443 "|" +
444 cs.zip(ws).joinToString("") { (c, w) ->
445 ('A'.code + c).toChar().toString().padEnd(w) + "|"
446 }
447 val out = mutableListOf(sep, head, sep)
448 for (r in rs) {
449 var row = "|" + (r + 1).toString().padStart(rw) + "|"
450 for ((c, w) in cs.zip(ws)) {
451 val v = values[r * 26 + c]
452 row +=
453 (if (v?.kind == 'N') display(v).padStart(w) else display(v).padEnd(w)) +
454 "|"
455 }
456 out += row
457 }
458 out += sep
459 return out
460 }
461 "UNDO" -> {
462 if (history.isEmpty()) throw Problem("nothing to undo")
463 val (command, old) = history.removeAt(history.lastIndex)
464 return listOf("UNDO $command changed=${replace(old)}")
465 }
466 }
467 var next = contents.copyOf()
468 var target = -1
469 when (name) {
470 "SET" -> {
471 val p = rest.indexOf(' ')
472 target = id(address(rest.substring(0, p)))
473 val s = rest.substring(p + 1)
474 next[target] =
475 if (s.startsWith("=")) Node('=', a = Parser(s).parse())
476 else if (Regex("\"[^\"]*\"").matches(s)) text(s.substring(1, s.length - 1))
477 else if (Regex("0|-?[1-9][0-9]{0,8}").matches(s)) num(s.toInt())
478 else throw Problem("bad value")
479 }
480 "CLEAR" -> for (i in cells(area(rest))) next[i] = null
481 "COPY",
482 "MOVE" -> {
483 val p = rest.indexOf(' ')
484 val source = area(rest.substring(0, p))
485 val dest = address(rest.substring(p + 1))
486 val dc = dest.n - source.a!!.n
487 val dr = dest.row - source.a.row
488 fun shift(r: Node, relative: Boolean) =
489 ref(
490 r.n + (if (relative && r.flags and 1 != 0) 0 else dc),
491 r.row + (if (relative && r.flags and 2 != 0) 0 else dr),
492 r.flags,
493 )
494 val destination = range(shift(source.a, false), shift(source.b!!, false))
495 if (!inside(destination.a!!) || !inside(destination.b!!))
496 throw Problem("out of grid")
497 val transform: (Node) -> Node = { x ->
498 if (x.kind == 'R') {
499 if (name == "COPY") {
500 val a = shift(x, true)
501 if (inside(a)) a else invalid
502 } else if (contains(source, x)) shift(x, false)
503 else if (contains(destination, x)) invalid else x
504 } else if (
505 name == "MOVE" && !(contains(source, x.a!!) && contains(source, x.b!!))
506 )
507 x
508 else {
509 val a = shift(x.a!!, name == "COPY")
510 val b = shift(x.b!!, name == "COPY")
511 if (inside(a) && inside(b)) range(a, b) else invalid
512 }
513 }
514 if (name == "MOVE") next = next.map { rewrite(it, transform) }.toTypedArray()
515 val ids = cells(source)
516 val moving = ids.map {
517 if (name == "COPY") rewrite(contents[it], transform) else next[it]
518 }
519 if (name == "MOVE") for (i in ids) next[i] = null
520 for (j in ids.indices) next[ids[j] + dr * 26 + dc] = moving[j]
521 }
522 "INSERTROW",
523 "DELETEROW",
524 "INSERTCOL",
525 "DELETECOL" -> {
526 val row = name.endsWith("ROW")
527 val insert = name.startsWith("INSERT")
528 if (!Regex(if (row) "[1-9][0-9]?" else "[A-Z]").matches(rest))
529 throw Problem("bad address")
530 val at = if (row) rest.toInt() - 1 else rest[0] - 'A'
531 val limit = if (row) 99 else 26
532 fun axis(x: Node) = if (row) x.row else x.n
533 fun change(x: Node, v: Int) =
534 ref(if (row) x.n else v, if (row) v else x.row, x.flags)
535 if (
536 insert &&
537 contents.indices.any {
538 contents[it] != null && (if (row) it / 26 else it % 26) == limit - 1
539 }
540 )
541 throw Problem("last ${if(row)"row" else "column"} not empty")
542 val transform: (Node) -> Node = { x ->
543 if (x.kind == 'R') {
544 val v = axis(x)
545 val n =
546 v +
547 (if (insert) {
548 if (v >= at) 1 else 0
549 } else if (v > at) -1 else 0)
550 if (!insert && v == at || n >= limit) invalid else change(x, n)
551 } else {
552 val a = axis(x.a!!)
553 val b = axis(x.b!!)
554 val u =
555 a +
556 (if (insert) {
557 if (a >= at) 1 else 0
558 } else if (a > at) -1 else 0)
559 val v =
560 b +
561 (if (insert) {
562 if (b >= at) 1 else 0
563 } else if (b >= at) -1 else 0)
564 if (!insert && a == at && b == at || v >= limit) invalid
565 else range(change(x.a, u), change(x.b, v))
566 }
567 }
568 next = arrayOfNulls(2574)
569 for (i in contents.indices) if (contents[i] != null) {
570 val r = ref(i % 26, i / 26)
571 val v = axis(r)
572 if (!insert && v == at) continue
573 val t =
574 change(
575 r,
576 v +
577 (if (insert) {
578 if (v >= at) 1 else 0
579 } else if (v > at) -1 else 0),
580 )
581 next[id(t)] = rewrite(contents[i], transform)
582 }
583 }
584 else -> throw Problem("unknown command")
585 }
586 history += name to contents
587 val changed = replace(next)
588 return listOf(
589 name +
590 (if (name == "SET") " ${label(target)}=${notation(values[target])}" else "") +
591 " changed=$changed",
592 )
593 }
594}

596fun main() {
597 val input = System.`in`.bufferedReader()
598 val n = input.readLine().toInt()
599 val sheet = Sheet()
600 val out = StringBuilder()
601 repeat(n) {
602 val line = input.readLine()
603 try {
604 for (s in sheet.execute(line)) out.append(s).append('\n')
605 } catch (e: Problem) {
606 out.append("ERROR ").append(e.message).append('\n')
607 }
608 }
609 print(out)
