19decimal <- function(numerator, denominator, places) {
20 scale <- 10^places
21 whole <- floor(numerator / denominator)
22 remainder <- numerator - whole * denominator
23 fraction <- floor(remainder * scale / denominator)
24 paste0(integer(whole), ".", sprintf(paste0("%0", places, "d"), as.integer(fraction)))
25}

46parse_policy <- function(tokens) {
47 name <- tokens[1]
48 if (!(name %in% names(policy_parameters))) fail(paste("unknown policy", name))
49 count <- policy_parameters[[name]]
50 if (length(tokens) - 1 != count)
51 fail(paste(name, "needs", count, "parameters"))
52 values <- numeric(count)
53 for (i in seq_len(count)) {
54 token <- tokens[i + 1]
55 values[i] <- as.numeric(if (name == "STERMAN" && i <= 3)
56 gsub(".", "", token, fixed = TRUE) else token)
57 }
58 valid <- switch(
59 name,
60 FIXED = 0 <= values[1] && values[1] <= 1000,
61 PASS = TRUE,
62 BASESTOCK = 0 <= values[1] && values[1] <= 10000,
63 MINMAX = 0 <= values[1] && values[1] < values[2] && values[2] <= 10000,
64 MOVAVG = 1 <= values[1] && values[1] <= 52 &&
65 0 <= values[2] && values[2] <= 10000,
66 SMOOTH = 1 <= values[1] && values[1] <= 10 &&
67 0 <= values[2] && values[2] <= 10000,
68 STERMAN = all(values[1:3] >= 0 & values[1:3] <= 100) &&
69 0 <= values[4] && values[4] <= 10000
70 )
71 if (!valid) fail(paste("invalid parameter for", name))
72 list(name = name, args = values, text = paste(tokens, collapse = " "))
73}

126positions <- function(simulation) {
127 supply <- numeric(4)
128 position <- numeric(4)
129 for (index in 1:4) {
130 stage <- simulation$stages[[index]]
131 upstream <- if (index < 4)
132 simulation$stages[[index + 1]]$backlog else simulation$plant_backlog
133 line <- sum(stage$orders) + upstream + sum(stage$shipping)
134 supply[index] <- line
135 position[index] <- stage$inventory + line - stage$backlog
136 }
137 list(supply = supply, position = position)
138}

140policy_order <- function(simulation, index, incoming, line, position) {
141 stage <- simulation$stages[[index]]
142 name <- stage$policy$name
143 args <- stage$policy$args
144 week <- simulation$week + 1
145 demand <- signal(simulation, index, week, incoming)
146 if (name == "FIXED") return(args[1])
147 if (name == "PASS") return(incoming[index])
148 if (name == "BASESTOCK") return(max(0, args[1] - position))
149 if (name == "MINMAX")
150 return(if (position <= args[1]) args[2] - position else 0)
151 if (name == "MOVAVG") {
152 count <- min(args[1], week)
153 observations <- vapply(seq.int(week - count + 1, week),
154 function(u) signal(simulation, index, u, incoming), numeric(1))
155 forecast <- floor(sum(observations) / count)
156 delay <- simulation$initials[[index]]$shipping +
157 simulation$initials[[index]]$orders
158 return(max(0, args[2] + forecast * delay - position))
159 }
160 if (name == "SMOOTH") {
161 return(max(0, demand + floor((args[2] - position) / args[1])))
162 }
163 theta <- args[1]
164 alpha <- args[2]
165 beta <- args[3]
166 target <- args[4]
167 stage$expected <- floor((100 * theta * demand +
168 (100 - theta) * stage$expected) / 100)
169 net <- stage$inventory - stage$backlog
170 max(0, floor((100 * stage$expected +
171 alpha * (100 * (target - net) - beta * line)) / 10000))
172}

174simulation_step <- function(simulation) {
175 received <- numeric(4)
176 for (index in 1:4) {
177 stage <- simulation$stages[[index]]
178 received[index] <- stage$shipping[1]
179 stage$shipping <- stage$shipping[-1]
180 stage$inventory <- stage$inventory + received[index]
181 }
182 incoming <- c(simulation$demands[simulation$week + 1],
183 vapply(simulation$stages[1:3], function(stage)
184 stage$orders[1], numeric(1)))
185 plant_incoming <- simulation$stages[[4]]$orders[1]
186 for (stage in simulation$stages) stage$orders <- stage$orders[-1]
187
188 records <- list()
189 for (index in 1:4) {
190 stage <- simulation$stages[[index]]
191 before <- stage$backlog
192 requested <- before + incoming[index]
193 shipped <- min(stage$inventory, requested)
194 if (!is.null(stage$capacity)) shipped <- min(shipped, stage$capacity)
195 stage$inventory <- stage$inventory - shipped
196 stage$backlog <- requested - shipped
197 if (index > 1) {
198 downstream <- simulation$stages[[index - 1]]
199 downstream$shipping <- c(downstream$shipping, shipped)
200 }
201 records[[index]] <- list(
202 received = received[index], incoming = incoming[index], shipped = shipped,
203 before = before,
204 ontime = min(max(0, shipped - before), incoming[index]),
205 inventory = stage$inventory, backlog = stage$backlog
206 )
207 }
208 requested <- simulation$plant_backlog + plant_incoming
209 produced <- if (is.null(simulation$plant_capacity))
210 requested else min(requested, simulation$plant_capacity)
211 simulation$plant_backlog <- requested - produced
212 factory <- simulation$stages[[4]]
213 factory$shipping <- c(factory$shipping, produced)
214
215 pipeline <- positions(simulation)
216 orders <- vapply(1:4, function(i)
217 policy_order(simulation, i, incoming, pipeline$supply[i],
218 pipeline$position[i]), numeric(1))
219 for (index in 1:4) {
220 stage <- simulation$stages[[index]]
221 quantity <- floor((orders[index] + stage$lot - 1) / stage$lot) * stage$lot
222 stage$orders <- c(stage$orders, quantity)
223 record <- records[[index]]
224 record$order <- quantity
225 record$holding <- stage$holding * stage$inventory
226 record$backlog_cost <- stage$backlog_rate * stage$backlog
227 record$cost <- record$holding + record$backlog_cost
228 stage$records[[length(stage$records) + 1]] <- record
229 records[[index]] <- record
230 }
231 simulation$plant_records[[length(simulation$plant_records) + 1]] <-
232 list(incoming = plant_incoming, produced = produced,
233 backlog = simulation$plant_backlog)
234 simulation$week <- simulation$week + 1
235 records
236}

284 current <- game$simulation
285 trial_state <- new_simulation(game$initials, current$demands)
286 trial_state$visibility <- current$visibility
287 trial_state$plant_capacity <- current$plant_capacity

305generate <- function(arguments) {
306 weeks <- as.numeric(arguments[1])
307 base <- as.numeric(arguments[2])
308 spread <- as.numeric(arguments[3])
309 seed <- as.numeric(arguments[4])
310 changes <- list()
311 positions <- if (length(arguments) > 4)
312 seq.int(5, length(arguments), by = 3) else integer()
313 for (position in positions) {
314 change <- list(kind = arguments[position],
315 week = as.numeric(arguments[position + 1]),
316 amount = as.numeric(arguments[position + 2]))
317 if (!(1 <= change$week && change$week <= weeks))
318 fail(paste("week", change$week, "outside horizon"))
319 changes[[length(changes) + 1]] <- change
320 }
321 levels <- numeric(weeks)
322 for (week in seq_len(weeks)) {
323 level <- base
324 for (change in changes) {
325 if (change$kind == "STEP" && week >= change$week) {
326 level <- level + change$amount
327 } else if (change$kind == "PULSE" && week == change$week) {
328 level <- level + change$amount
329 } else if (change$kind == "RAMP" && week >= change$week) {
330 level <- level + change$amount * (week - change$week + 1)
331 }
332 }
333 if (level < 0) fail(paste("negative level at week", week))
334 levels[week] <- level
335 }
336 values <- numeric(weeks)
337 for (week in seq_len(weeks)) {
338 seed <- (48271 * seed) %% 2147483647
339 values[week] <- levels[week] + seed %% spread
340 }
341 set_demand(values)
342}

477show_leadtime <- function(index) {
478 simulation <- game$simulation
479 stage <- simulation$stages[[index]]
480 initial <- game$initials[[index]]
481 nominal <- initial$shipping + initial$orders
482 cumulative <- cumsum(record_values(stage$records, "received")) -
483 nominal * initial$quantity
484 ordered <- 0
485 count <- 0
486 leads <- numeric()
487 for (t in seq_along(stage$records)) {
488 record <- stage$records[[t]]
489 ordered <- ordered + record$order
490 if (record$order == 0) next
491 count <- count + 1
492 if (t < length(stage$records)) {
493 for (u in seq.int(t + 1, length(stage$records))) {
494 if (cumulative[u] >= ordered) {
495 leads <- c(leads, u - t)
496 break
497 }
498 }
499 }
500 }
501 emit("LEADTIME", stage_names[index], "nominal", nominal,
502 "orders", count, "completed", length(leads),
503 "open", count - length(leads),
504 "min", if (length(leads)) min(leads) else "-",
505 "avg", if (length(leads)) decimal(sum(leads), length(leads), 2) else "-",
506 "max", if (length(leads)) max(leads) else "-")

551run_command <- function(command, a) {
552 simulation <- game$simulation
553 if (command == "DEMAND") {
554 set_demand(as.numeric(a))
555 } else if (command == "GENERATE") {
556 generate(a)
557 } else if (command == "RUN") {
558 if (simulation$week == length(simulation$demands))
559 fail(paste("horizon reached at week", length(simulation$demands)))
560 first <- simulation$week + 1
561 count <- min(as.numeric(a[1]), length(simulation$demands) - simulation$week)
562 total_cost <- 0
563 for (i in seq_len(count)) {
564 records <- simulation_step(simulation)
565 for (index in 1:4) {
566 record <- records[[index]]
567 total_cost <- total_cost + record$cost
568 if (record$before == 0 && record$backlog > 0) {
569 emit("STOCKOUT", stage_names[index], "week", simulation$week,
570 "backlog", record$backlog)
571 } else if (record$before > 0 && record$backlog == 0) {
572 emit("RECOVER", stage_names[index], "week", simulation$week)
573 }
574 }
575 }
576 emit("RUN", paste0(first, "-", simulation$week), "demand",
577 sum(simulation$demands[seq.int(first, simulation$week)]),
578 "cost", money(total_cost))
579 if (simulation$week == length(simulation$demands))
580 emit("HORIZON", simulation$week)
581 } else if (command == "RESET") {
582 initialize()
583 emit("RESET", "week", 0)
584 } else if (command == "SAVE") {
585 assign(a[1], clone_state(simulation), envir = game$snapshots)
586 emit("SAVE", a[1], "week", simulation$week)
587 } else if (command == "LOAD") {
588 if (!exists(a[1], envir = game$snapshots, inherits = FALSE))
589 fail(paste("no snapshot", a[1]))
590 game$simulation <- clone_state(get(a[1], envir = game$snapshots,
591 inherits = FALSE))
592 emit("LOAD", a[1], "week", game$simulation$week)
593 } else if (command == "SET") {
594 indices <- target(a[1])
595 policy <- parse_policy(a[-1])
596 for (index in indices) set_policy(simulation, index, policy)
597 emit("SET", a[1], policy$text)
598 } else if (command == "BATCH") {
599 index <- stage_index(a[1])
600 lot <- as.numeric(a[2])
601 if (lot < 1 || lot > 100) fail(paste("invalid batch size", a[2]))
602 simulation$stages[[index]]$lot <- lot
603 emit("BATCH", a[1], lot)
604 } else if (command == "CAPACITY") {
605 index <- if (a[1] == "PLANT") NA_integer_ else stage_index(a[1])
606 capacity <- if (a[2] == "OFF") NULL else as.numeric(a[2])
607 if (!is.null(capacity) && capacity > 10000)
608 fail(paste("invalid capacity", a[2]))
609 if (is.na(index)) {
610 simulation$plant_capacity <- capacity
611 } else {
612 simulation$stages[[index]]$capacity <- capacity
613 }
614 emit("CAPACITY", a[1], a[2])
615 } else if (command == "COST") {
616 index <- stage_index(a[1])
617 holding <- as.numeric(a[2])
618 backlog <- as.numeric(a[3])
619 if (holding > 1000 || backlog > 1000) fail("invalid cost")
620 stage <- simulation$stages[[index]]
621 stage$holding <- holding
622 stage$backlog_rate <- backlog
623 emit("COST", a[1], "holding", holding, "backlog", backlog)
624 } else if (command == "VISIBILITY") {
625 if (!(a[1] %in% c("ON", "OFF"))) fail(paste("unknown visibility", a[1]))
626 simulation$visibility <- a[1]
627 emit("VISIBILITY", a[1])
628 } else if (command == "SHOCK") {
629 week <- as.numeric(a[1])
630 value <- as.numeric(a[2])
631 if (week < 1 || week > length(simulation$demands))
632 fail(paste("week", a[1], "outside horizon"))
633 if (week <= simulation$week) fail(paste("week", a[1], "already simulated"))
634 old <- simulation$demands[week]
635 simulation$demands[week] <- value
636 emit("SHOCK", "week", week, old, "->", value)
637 } else if (command == "STATUS") {
638 show_status()
639 } else if (command == "SETTINGS") {
640 emit("SETTINGS")
641 for (index in 1:4) {
642 stage <- simulation$stages[[index]]
643 initial <- game$initials[[index]]
644 emit(stage_names[index], "holding", stage$holding,
645 "backlog", stage$backlog_rate, "batch", stage$lot,
646 "capacity", if (is.null(stage$capacity)) "OFF" else stage$capacity,
647 "delays", initial$shipping, initial$orders)
648 }
649 emit("PLANT", "capacity",
650 if (is.null(simulation$plant_capacity)) "OFF" else simulation$plant_capacity)
651 emit("VISIBILITY", simulation$visibility)
652 } else if (command == "WEEK") {
653 require_weeks()
654 week <- as.numeric(a[1])
655 if (week < 1 || week > simulation$week) fail(paste("bad week", a[1]))
656 show_week(week)
657 } else if (command == "TABLE") {
658 index <- stage_index(a[1])
659 require_weeks()
660 bounds <- range_check(a[-1], simulation$week)
661 show_table(index, bounds[1], bounds[2])
662 } else if (command == "CHAIN") {
663 require_weeks()
664 bounds <- range_check(a, simulation$week)
665 first <- bounds[1]
666 last <- bounds[2]
667 emit("CHAIN", paste0(first, "-", last))
668 header <- c(right("week", 4), right("demand", 6),
669 vapply(stage_names, right, character(1), width = 11))
670 emit(paste(header, collapse = " "))
671 for (week in seq.int(first, last)) {
672 orders <- vapply(simulation$stages, function(stage)
673 stage$records[[week]]$order, numeric(1))
674 entries <- c(right(week, 4), right(simulation$demands[week], 6),
675 vapply(orders, right, character(1), width = 11))
676 emit(paste(entries, collapse = " "))
677 }
678 } else if (command == "DEMANDS") {
679 bounds <- range_check(a, length(simulation$demands))
680 values <- simulation$demands[seq.int(bounds[1], bounds[2])]
681 emit("DEMANDS", paste0(bounds[1], "-", bounds[2]),
682 "total", sum(values), "min", min(values), "max", max(values))
683 emit(paste(values, collapse = " "))
684 } else if (command == "COSTS") {
685 show_costs()
686 } else if (command == "SUMMARY") {
687 require_weeks()
688 emit("SUMMARY", "weeks", paste0("1-", simulation$week))
689 for (index in 1:4) {
690 records <- simulation$stages[[index]]$records
691 orders <- record_values(records, "order")
692 emit(stage_names[index],
693 "avg_inv", decimal(record_sum(records, "inventory"), simulation$week, 2),
694 "avg_back", decimal(record_sum(records, "backlog"), simulation$week, 2),
695 "orders", sum(orders), "min", min(orders), "max", max(orders),
696 "zero_weeks", sum(orders == 0))
697 }
698 } else if (command == "BALANCE") {
699 emit("BALANCE", "week", simulation$week)
700 for (index in 1:4) {
701 stage <- simulation$stages[[index]]
702 records <- stage$records
703 emit(stage_names[index], "start", game$initials[[index]]$inventory,
704 "received", record_sum(records, "received"),
705 "shipped", record_sum(records, "shipped"),
706 "end", stage$inventory, "owed", stage$backlog)

708 initial_goods <- sum(vapply(game$initials, function(initial)

710 emit("SYSTEM", "initial", initial_goods,
711 "produced", record_sum(simulation$plant_records, "produced"),
712 "delivered", record_sum(simulation$stages[[1]]$records, "shipped"),
713 "stock", sum(vapply(simulation$stages, function(stage) stage$inventory, numeric(1))),
714 "transit", sum(vapply(simulation$stages, function(stage) sum(stage$shipping), numeric(1))))
715 } else if (command == "FILLRATE") {
716 require_weeks()
717 emit("FILLRATE", "weeks", paste0("1-", simulation$week))
718 for (index in 1:4) {
719 records <- simulation$stages[[index]]$records
720 result <- fill_text(records)
721 stockouts <- sum(record_values(records, "backlog") > 0)
722 emit(stage_names[index], "fill", result$fill, "ontime", result$ontime,
723 "ordered", result$ordered, "stockout_weeks", stockouts,
724 "service", paste0(decimal((simulation$week - stockouts) * 100,
725 simulation$week, 2), "%"))
726 }
727 } else if (command == "BULLWHIP") {
728 require_weeks()
729 show_bullwhip()
730 } else if (command == "PEAKS") {
731 require_weeks()
732 show_peaks()
733 } else if (command == "LEADTIME") {
734 index <- stage_index(a[1])
735 require_weeks()
736 show_leadtime(index)
737 } else if (command %in% c("PLOT", "HISTOGRAM")) {
738 index <- stage_index(a[1])
739 mode_check(a[2])
740 if (command == "HISTOGRAM") {
741 width <- as.numeric(a[3])
742 if (width < 1 || width > 100) fail(paste("invalid width", a[3]))
743 }
744 require_weeks()
745 if (command == "PLOT") {
746 bounds <- range_check(a[-(1:2)], simulation$week)
747 show_plot(index, a[2], bounds[1], bounds[2])
748 } else {
749 show_histogram(index, a[2], width)

751 } else if (command == "COMPARE") {
752 policies <- list()
753 start <- 1
754 for (end in seq_len(length(a) + 1)) {
755 if (end == length(a) + 1 || a[end] == "/") {
756 policies[[length(policies) + 1]] <- parse_policy(a[seq.int(start, end - 1)])
757 start <- end + 1
758 }
759 }
760 rows <- list()
761 for (i in seq_along(policies)) {
762 policy <- policies[[i]]
763 simulation_trial <- trial(rep(list(policy), 4))
764 cost <- sum(sum_costs(simulation_trial))
765 fill <- fill_text(simulation_trial$stages[[1]]$records)$fill
766 demand_var <- variation(simulation_trial$demands)
767 factory_var <- variation(record_values(simulation_trial$stages[[4]]$records, "order"))
768 ratio <- if (demand_var == 0) "n/a" else decimal(factory_var, demand_var, 4)
769 rows[[i]] <- list(cost = cost, index = i, policy = policy$text,
770 fill = fill, ratio = ratio)
771 }
772 emit("COMPARE", length(policies), "weeks", length(simulation$demands))
773 ordering <- order(vapply(rows, function(row) row$cost, numeric(1)),
774 vapply(rows, function(row) row$index, numeric(1)))
775 for (rank in seq_along(ordering)) {
776 row <- rows[[ordering[rank]]]
777 emit(rank, row$policy, "total", money(row$cost),
778 "fill", row$fill, "bullwhip", row$ratio)
779 }
780 } else if (command == "OPTIMIZE") {
781 indices <- target(a[1])
782 low <- as.numeric(a[2])
783 high <- as.numeric(a[3])
784 if (low > high) fail(paste("bad range", a[2], a[3]))
785 if (high - low > 200) fail("range too large")
786 best <- NULL
787 for (value in seq.int(low, high)) {
788 policies <- lapply(simulation$stages, function(stage) stage$policy)
789 policy <- list(name = "BASESTOCK", args = value,
790 text = paste("BASESTOCK", value))
791 for (index in indices) policies[[index]] <- policy
792 simulation_trial <- trial(policies)
793 costs <- sum_costs(simulation_trial)
794 objective <- sum(costs[indices])
795 row <- c(objective, value, sum(costs))
796 if (is.null(best) || row[1] < best[1] ||
797 (row[1] == best[1] && row[2] < best[2])) best <- row
798 }
799 emit("OPTIMIZE", a[1], "BASESTOCK", best[2],
800 "cost", money(best[1]), "total", money(best[3]))
801 } else {
802 fail(paste("unknown command", command))
803 }
