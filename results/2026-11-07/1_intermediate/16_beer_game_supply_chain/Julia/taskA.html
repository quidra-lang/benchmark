3const NAMES = ["RETAILER", "WHOLESALER", "DISTRIBUTOR", "FACTORY"]
4const PARAMETERS = Dict("FIXED" => 1, "PASS" => 0, "BASESTOCK" => 1,
5 "MINMAX" => 2, "MOVAVG" => 2, "SMOOTH" => 2, "STERMAN" => 4)
6const MODES = Dict("ORDERS" => "order", "INV" => "inventory", "BACK" => "backlog")

14money(value::Integer) = string(div(value, 100), ".", lpad(string(mod(value, 100)), 2, '0'))

16function decimal(numerator::Integer, denominator::Integer, places::Int)
17 scale = 10^places
18 value = fld(numerator * scale, denominator)
19 string(fld(value, scale), ".", lpad(string(mod(value, scale)), places, '0'))
20end

22variation(values) = length(values) * sum(v^2 for v in values) - sum(values)^2

30function parse_policy(tokens::Vector{String})
31 name = tokens[1]
32 haskey(PARAMETERS, name) || fail("unknown policy " * name)
33 count = PARAMETERS[name]
34 length(tokens) - 1 == count || fail("$name needs $count parameters")
35 args = [parse(Int, name == "STERMAN" && i <= 3 ?
36 replace(text, "." => "") : text)
37 for (i, text) in enumerate(tokens[2:end])]
38 valid = if name == "FIXED"
39 0 <= args[1] <= 1000
40 elseif name == "PASS"
41 true
42 elseif name == "BASESTOCK"
43 0 <= args[1] <= 10000
44 elseif name == "MINMAX"
45 0 <= args[1] < args[2] <= 10000
46 elseif name == "MOVAVG"
47 1 <= args[1] <= 52 && 0 <= args[2] <= 10000
48 elseif name == "SMOOTH"
49 1 <= args[1] <= 10 && 0 <= args[2] <= 10000
50 else
51 all(0 <= x <= 100 for x in args[1:3]) && 0 <= args[4] <= 10000
52 end
53 valid || fail("invalid parameter for " * name)
54 Policy(name, args, join(tokens, " "))
55end

106function signal(s::Simulation, index::Int, week::Int, current=nothing)
107 if s.visibility == "ON"
108 return s.demands[week]
109 end
110 if current !== nothing && week == s.week + 1
111 return current[index]
112 end
113 s.stages[index].records[week]["incoming"]
114end

116function set_policy!(s::Simulation, index::Int, policy::Policy)
117 stage = s.stages[index]
118 stage.policy = policy
119 if policy.name == "STERMAN"
120 stage.expected = 100 * (s.week == 0 ?
121 s.initials[index].quantity : signal(s, index, s.week))
122 end
123end

125function positions(s::Simulation)
126 supply, position = Int[], Int[]
127 for (index, stage) in enumerate(s.stages)
128 upstream = index < 4 ? s.stages[index+1].backlog : s.plant_backlog
129 line = sum(stage.orders) + upstream + sum(stage.shipping)
130 push!(supply, line)
131 push!(position, stage.inventory + line - stage.backlog)
132 end
133 return supply, position
134end
135
136function policy_order!(s::Simulation, index::Int, incoming, line::Int, position::Int)
137 stage = s.stages[index]
138 name, args = stage.policy.name, stage.policy.args
139 week = s.week + 1
140 observed = signal(s, index, week, incoming)
141 if name == "FIXED"
142 return args[1]
143 elseif name == "PASS"
144 return incoming[index]
145 elseif name == "BASESTOCK"
146 return max(0, args[1] - position)
147 elseif name == "MINMAX"
148 return position <= args[1] ? args[2] - position : 0
149 elseif name == "MOVAVG"
150 k, target = args
151 count = min(k, week)
152 forecast = fld(sum(signal(s, index, u, incoming)
153 for u in week-count+1:week), count)
154 delay = s.initials[index].shipping + s.initials[index].orders
155 return max(0, target + forecast * delay - position)
156 elseif name == "SMOOTH"
157 k, target = args
158 return max(0, observed + fld(target - position, k))
159 else
160 theta, alpha, beta, target = args
161 stage.expected = fld(100 * theta * observed +
162 (100 - theta) * stage.expected, 100)
163 net = stage.inventory - stage.backlog
164 return max(0, fld(100 * stage.expected +
165 alpha * (100 * (target - net) - beta * line), 10000))
166 end
167end

169function step!(s::Simulation)
170 received = Int[]
171 for stage in s.stages
172 quantity = popfirst!(stage.shipping)
173 stage.inventory += quantity
174 push!(received, quantity)
175 end
176 incoming = [s.demands[s.week + 1]]
177 append!(incoming, (popfirst!(s.stages[index].orders) for index in 1:3))
178 plant_incoming = popfirst!(s.stages[4].orders)
179 records = Dict{String,Int}[]
180 for (index, stage) in enumerate(s.stages)
181 before = stage.backlog
182 requested = before + incoming[index]
183 shipped = min(stage.inventory, requested)
184 if stage.capacity !== nothing
185 shipped = min(shipped, stage.capacity)
186 end
187 stage.inventory -= shipped
188 stage.backlog = requested - shipped
189 if index > 1
190 push!(s.stages[index-1].shipping, shipped)
191 end
192 push!(records, Dict(
193 "received" => received[index], "incoming" => incoming[index],
194 "shipped" => shipped, "before" => before,
195 "ontime" => min(max(0, shipped - before), incoming[index]),
196 "inventory" => stage.inventory, "backlog" => stage.backlog,
197 ))
198 end
199 requested = s.plant_backlog + plant_incoming
200 produced = s.plant_capacity === nothing ?
201 requested : min(requested, s.plant_capacity)
202 s.plant_backlog = requested - produced
203 push!(s.stages[4].shipping, produced)
204 supply, position = positions(s)
205 for (index, stage) in enumerate(s.stages)
206 order = policy_order!(s, index, incoming, supply[index], position[index])
207 quantity = cld(order, stage.lot) * stage.lot
208 push!(stage.orders, quantity)
209 record = records[index]
210 record["order"] = quantity
211 record["holding"] = stage.holding * stage.inventory
212 record["backlog_cost"] = stage.backlog_rate * stage.backlog
213 record["cost"] = record["holding"] + record["backlog_cost"]
214 push!(stage.records, record)
215 end
216 push!(s.plant_records, Dict(
217 "incoming" => plant_incoming, "produced" => produced,
218 "backlog" => s.plant_backlog,
219 ))
220 s.week += 1
221 records
222end

224costs(s::Simulation) = [sum(r["cost"] for r in stage.records) for stage in s.stages]

254function range_check(args::Vector{String}, limit::Int)
255 isempty(args) && return (1, limit)
256 first, last = parse.(Int, args)
257 1 <= first <= last <= limit || fail("bad range $(args[1]) $(args[2])")
258 return (first, last)
259end

281function trial(g::Game, policies::Vector{Policy})
282 current = g.simulation
283 alternative = Simulation(g.initials, current.demands)
284 alternative.visibility = current.visibility
285 alternative.plant_capacity = current.plant_capacity
286 for index in 1:4
287 stage, previous = alternative.stages[index], current.stages[index]
288 stage.lot = previous.lot
289 stage.capacity = previous.capacity
290 stage.holding = previous.holding
291 stage.backlog_rate = previous.backlog_rate
292 set_policy!(alternative, index, policies[index])
293 end
294 while alternative.week < length(alternative.demands)
295 step!(alternative)
296 end
297 alternative
298end

307function generate!(g::Game, args::Vector{String})
308 weeks, base, spread, seed = parse.(Int, args[1:4])
309 changes = [(args[i], parse(Int, args[i+1]), parse(Int, args[i+2]))
310 for i in 5:3:length(args)]
311 for (_, week, _) in changes
312 1 <= week <= weeks || fail("week $week outside horizon")
313 end
314 levels = Int[]
315 for week in 1:weeks
316 level = base
317 for (kind, start, amount) in changes
318 if kind == "STEP" && week >= start
319 level += amount
320 elseif kind == "PULSE" && week == start
321 level += amount
322 elseif kind == "RAMP" && week >= start
323 level += amount * (week - start + 1)
324 end
325 end
326 level >= 0 || fail("negative level at week $week")
327 push!(levels, level)
328 end
329 amounts = Int[]
330 for level in levels
331 seed = mod(48271 * seed, 2147483647)
332 push!(amounts, level + mod(seed, spread))
333 end
334 demand!(g, amounts)
335end

405function bullwhip!(g::Game)
406 s = g.simulation
407 n = s.week
408 demands = s.demands[1:n]
409 demand_var = variation(demands)
410 emit!(g, "BULLWHIP weeks 1-$n demand mean",
411 decimal(sum(demands), n, 4), "var",
412 decimal(demand_var, n*n, 4))
413 previous = demand_var
414 for (index, stage) in enumerate(s.stages)
415 orders = [r["order"] for r in stage.records]
416 v = variation(orders)
417 emit!(g, NAMES[index], "mean", decimal(sum(orders), n, 4),
418 "var", decimal(v, n*n, 4),
419 "ratio", demand_var == 0 ? "n/a" : decimal(v, demand_var, 4),
420 "step", previous == 0 ? "n/a" : decimal(v, previous, 4))
421 previous = v
422 end
423end

425function peaks!(g::Game)
426 s = g.simulation
427 emit!(g, "PEAKS weeks 1-$(s.week)")
428 for (index, stage) in enumerate(s.stages)
429 inventories = [r["inventory"] for r in stage.records]
430 backlogs = [r["backlog"] for r in stage.records]
431 max_inventory, max_backlog = maximum(inventories), maximum(backlogs)
432 longest, run, start, best_start, best_end = 0, 0, 0, 0, 0
433 for (week, backlog) in enumerate(backlogs)
434 if backlog > 0
435 if run == 0
436 start = week
437 end
438 run += 1
439 if run > longest
440 longest, best_start, best_end = run, start, week
441 end
442 else
443 run = 0
444 end
445 end
446 inventory_week = max_inventory == 0 ? "-" : string(findfirst(==(max_inventory), inventories))
447 backlog_week = max_backlog == 0 ? "-" : string(findfirst(==(max_backlog), backlogs))
448 stockout = longest > 0 ? "$longest weeks $best_start-$best_end" : "0"
449 emit!(g, NAMES[index], "max_inv", max_inventory, "week", inventory_week,
450 "max_back", max_backlog, "week", backlog_week,
451 "longest_stockout", stockout)
452 end
453end

455function leadtime!(g::Game, index::Int)
456 s = g.simulation
457 stage, initial = s.stages[index], g.initials[index]
458 nominal = initial.shipping + initial.orders
459 cumulative = cumsum([r["received"] for r in stage.records]) .-
460 nominal * initial.quantity
461 ordered, count = 0, 0
462 leads = Int[]
463 for (t, record) in enumerate(stage.records)
464 ordered += record["order"]
465 if record["order"] == 0
466 continue
467 end
468 count += 1
469 for u in t+1:length(stage.records)
470 if cumulative[u] >= ordered
471 push!(leads, u - t)
472 break
473 end
474 end
475 end
476 emit!(g, "LEADTIME", NAMES[index], "nominal", nominal, "orders", count,
477 "completed", length(leads), "open", count - length(leads),
478 "min", isempty(leads) ? "-" : string(minimum(leads)),
479 "avg", isempty(leads) ? "-" : decimal(sum(leads), length(leads), 2),
480 "max", isempty(leads) ? "-" : string(maximum(leads)))
481end

483function plot!(g::Game, index::Int, mode::String, first::Int, last::Int)
484 displayed = values(g, index, mode)[first:last]
485 maximum_value = maximum(displayed)
486 if mode == "NET"
487 minimum_value = minimum(displayed)
488 emit!(g, "PLOT", NAMES[index], mode, "$first-$last",
489 "min", minimum_value, "max", maximum_value)
490 scale = maximum(abs.(displayed))
491 for (week, value) in enumerate(displayed)
492 length_ = scale == 0 ? 0 : fld(abs(value) * 25, scale)
493 left = value < 0 ?
494 repeat(" ", 25 - length_) * repeat("-", length_) : repeat(" ", 25)
495 right = value > 0 ? repeat("+", length_) : ""
496 emit!(g, "$(lpad(string(first + week - 1), 3)) $left|$right $value")
497 end
498 else
499 emit!(g, "PLOT", NAMES[index], mode, "$first-$last", "max", maximum_value)
500 for (week, value) in enumerate(displayed)
501 length_ = maximum_value == 0 ? 0 : fld(value * 50, maximum_value)
502 emit!(g, "$(lpad(string(first + week - 1), 3)) |$(repeat("#", length_)) $value")
503 end
504 end

507function command!(g::Game, command::String, args::Vector{String})
508 s = g.simulation
509 if command == "DEMAND"
510 demand!(g, parse.(Int, args))
511 elseif command == "GENERATE"
512 generate!(g, args)
513 elseif command == "RUN"
514 s.week < length(s.demands) ||
515 fail("horizon reached at week $(length(s.demands))")
516 first = s.week + 1
517 count = min(parse(Int, args[1]), length(s.demands) - s.week)
518 cost = 0
519 for _ in 1:count
520 records = step!(s)
521 for (index, r) in enumerate(records)
522 cost += r["cost"]
523 if r["before"] == 0 && r["backlog"] > 0
524 emit!(g, "STOCKOUT", NAMES[index], "week", s.week,
525 "backlog", r["backlog"])
526 elseif r["before"] > 0 && r["backlog"] == 0
527 emit!(g, "RECOVER", NAMES[index], "week", s.week)
528 end
529 end
530 end
531 emit!(g, "RUN $first-$(s.week) demand",
532 sum(s.demands[first:s.week]), "cost", money(cost))
533 if s.week == length(s.demands)
534 emit!(g, "HORIZON", s.week)
535 end
536 elseif command == "RESET"
537 initialize!(g)
538 emit!(g, "RESET week 0")
539 elseif command == "SAVE"
540 g.snapshots[args[1]] = deepcopy(s)
541 emit!(g, "SAVE", args[1], "week", s.week)
542 elseif command == "LOAD"
543 haskey(g.snapshots, args[1]) || fail("no snapshot " * args[1])
544 g.simulation = deepcopy(g.snapshots[args[1]])
545 emit!(g, "LOAD", args[1], "week", g.simulation.week)
546 elseif command == "SET"
547 indices = target(args[1])
548 policy = parse_policy(args[2:end])
549 for index in indices
550 set_policy!(s, index, policy)
551 end
552 emit!(g, "SET", args[1], policy.display)
553 elseif command == "BATCH"
554 index = stage_index(args[1])
555 lot = parse(Int, args[2])
556 1 <= lot <= 100 || fail("invalid batch size " * args[2])
557 s.stages[index].lot = lot
558 emit!(g, "BATCH", args[1], lot)
559 elseif command == "CAPACITY"
560 index = args[1] == "PLANT" ? nothing : stage_index(args[1])
561 capacity = args[2] == "OFF" ? nothing : parse(Int, args[2])
562 if capacity !== nothing && capacity > 10000
563 fail("invalid capacity " * args[2])
564 end
565 if index === nothing
566 s.plant_capacity = capacity
567 else
568 s.stages[index].capacity = capacity
569 end
570 emit!(g, "CAPACITY", args...)
571 elseif command == "COST"
572 index = stage_index(args[1])
573 holding, backlog = parse.(Int, args[2:3])
574 holding <= 1000 && backlog <= 1000 || fail("invalid cost")
575 s.stages[index].holding = holding
576 s.stages[index].backlog_rate = backlog
577 emit!(g, "COST", args[1], "holding", holding, "backlog", backlog)
578 elseif command == "VISIBILITY"
579 args[1] in ("ON", "OFF") || fail("unknown visibility " * args[1])
580 s.visibility = args[1]
581 emit!(g, "VISIBILITY", args[1])
582 elseif command == "SHOCK"
583 week, amount = parse.(Int, args)
584 1 <= week <= length(s.demands) ||
585 fail("week $(args[1]) outside horizon")
586 week > s.week || fail("week $(args[1]) already simulated")
587 before = s.demands[week]
588 s.demands[week] = amount
589 emit!(g, "SHOCK week", week, before, "->", amount)
590 elseif command == "STATUS"
591 show_status!(g)
592 elseif command == "SETTINGS"
593 emit!(g, "SETTINGS")
594 for (index, stage) in enumerate(s.stages)
595 initial = g.initials[index]
596 emit!(g, NAMES[index], "holding", stage.holding,
597 "backlog", stage.backlog_rate, "batch", stage.lot,
598 "capacity", stage.capacity === nothing ? "OFF" : stage.capacity,
599 "delays", initial.shipping, initial.orders)
600 end
601 emit!(g, "PLANT capacity", s.plant_capacity === nothing ? "OFF" : s.plant_capacity)
602 emit!(g, "VISIBILITY", s.visibility)
603 elseif command == "WEEK"
604 require_weeks(g)
605 week = parse(Int, args[1])
606 1 <= week <= s.week || fail("bad week " * args[1])
607 show_week!(g, week)
608 elseif command == "TABLE"
609 index = stage_index(args[1])
610 require_weeks(g)
611 first, last = range_check(args[2:end], s.week)
612 table_display!(g, index, first, last)
613 elseif command == "CHAIN"
614 require_weeks(g)
615 first, last = range_check(args, s.week)
616 emit!(g, "CHAIN $first-$last")
617 emit!(g, "$(lpad("week", 4)) $(lpad("demand", 6)) " *
618 join(lpad.(NAMES, 11), " "))
619 for week in first:last
620 orders = [r.records[week]["order"] for r in s.stages]
621 emit!(g, "$(lpad(string(week), 4)) $(lpad(string(s.demands[week]), 6)) " *
622 join((lpad(string(v), 11) for v in orders), " "))
623 end
624 elseif command == "DEMANDS"
625 first, last = range_check(args, length(s.demands))
626 amounts = s.demands[first:last]
627 emit!(g, "DEMANDS $first-$last total", sum(amounts),
628 "min", minimum(amounts), "max", maximum(amounts))
629 emit!(g, join(amounts, " "))
630 elseif command == "COSTS"
631 show_costs!(g)
632 elseif command == "SUMMARY"
633 require_weeks(g)
634 emit!(g, "SUMMARY weeks 1-$(s.week)")
635 for (index, stage) in enumerate(s.stages)
636 r = stage.records
637 orders = [v["order"] for v in r]
638 emit!(g, NAMES[index],
639 "avg_inv", decimal(sum(v["inventory"] for v in r), s.week, 2),
640 "avg_back", decimal(sum(v["backlog"] for v in r), s.week, 2),
641 "orders", sum(orders), "min", minimum(orders),
642 "max", maximum(orders), "zero_weeks", count(==(0), orders))
643 end
644 elseif command == "BALANCE"
645 emit!(g, "BALANCE week", s.week)
646 for (index, stage) in enumerate(s.stages)
647 emit!(g, NAMES[index], "start", g.initials[index].inventory,
648 "received", sum(r["received"] for r in stage.records),
649 "shipped", sum(r["shipped"] for r in stage.records),
650 "end", stage.inventory, "owed", stage.backlog)
651 end
652 initial_goods = sum(i.inventory + i.quantity * i.shipping for i in g.initials)
653 emit!(g, "SYSTEM initial", initial_goods,
654 "produced", sum(r["produced"] for r in s.plant_records),
655 "delivered", sum(r["shipped"] for r in s.stages[1].records),
656 "stock", sum(stage.inventory for stage in s.stages),
657 "transit", sum(sum(stage.shipping) for stage in s.stages))
658 elseif command == "FILLRATE"
659 require_weeks(g)
660 emit!(g, "FILLRATE weeks 1-$(s.week)")
661 for (index, stage) in enumerate(s.stages)
662 ordered, ontime, fill = fill_text(stage.records)
663 stockouts = count(r -> r["backlog"] > 0, stage.records)
664 emit!(g, NAMES[index], "fill", fill, "ontime", ontime,
665 "ordered", ordered, "stockout_weeks", stockouts,
666 "service", decimal((s.week - stockouts) * 100, s.week, 2) * "%")
667 end
668 elseif command == "BULLWHIP"
669 require_weeks(g)
670 bullwhip!(g)
671 elseif command == "PEAKS"
672 require_weeks(g)
673 peaks!(g)
674 elseif command == "LEADTIME"
675 index = stage_index(args[1])
676 require_weeks(g)
677 leadtime!(g, index)
678 elseif command in ("PLOT", "HISTOGRAM")
679 index = stage_index(args[1])
680 mode_check(args[2])
681 if command == "HISTOGRAM"
682 width = parse(Int, args[3])
683 1 <= width <= 100 || fail("invalid width " * args[3])
684 end
685 require_weeks(g)
686 if command == "PLOT"
687 first, last = range_check(args[3:end], s.week)
688 plot!(g, index, args[2], first, last)
689 else
690 counts = Dict{Int,Int}()
691 for value in values(g, index, args[2])
692 bucket = fld(value, width)
693 counts[bucket] = get(counts, bucket, 0) + 1
694 end
695 emit!(g, "HISTOGRAM", args[1], args[2], "width", width,
696 "weeks 1-$(s.week)")
697 for bucket in minimum(keys(counts)):maximum(keys(counts))
698 n = get(counts, bucket, 0)
699 emit!(g, "$(bucket*width)..$(bucket*width+width-1) $n" *
700 (n == 0 ? "" : " " * repeat("*", n)))
701 end
702 end
703 elseif command == "COMPARE"
704 policies = Policy[]
705 first = 1
706 for last in 1:(length(args)+1)
707 if last > length(args) || args[last] == "/"
708 push!(policies, parse_policy(args[first:last-1]))
709 first = last + 1
710 end
711 end
712 rows = Tuple{Int,Int,String,String,String}[]
713 for (index, policy) in enumerate(policies)
714 simulated = trial(g, fill(policy, 4))
715 objective = sum(costs(simulated))
716 _, _, fill = fill_text(simulated.stages[1].records)
717 demand_var = variation(simulated.demands)
718 factory_var = variation([r["order"] for r in simulated.stages[4].records])
719 ratio = demand_var == 0 ? "n/a" : decimal(factory_var, demand_var, 4)
720 push!(rows, (objective, index, policy.display, fill, ratio))
721 end
722 emit!(g, "COMPARE", length(policies), "weeks", length(s.demands))
723 for (rank, row) in enumerate(sort(rows))
724 cost, _, policy, fill, ratio = row
725 emit!(g, rank, policy, "total", money(cost), "fill", fill,
726 "bullwhip", ratio)
727 end
728 elseif command == "OPTIMIZE"
729 indices = target(args[1])
730 low, high = parse.(Int, args[2:3])
731 low <= high || fail("bad range $(args[2]) $(args[3])")
732 high - low <= 200 || fail("range too large")
733 best = nothing
734 for level in low:high
735 policies = [stage.policy for stage in s.stages]
736 for index in indices
737 policies[index] = Policy("BASESTOCK", [level], "BASESTOCK $level")
738 end
739 simulated = trial(g, policies)
740 costs_ = costs(simulated)
741 candidate = (sum(costs_[indices]), level, sum(costs_))
742 if best === nothing || candidate[1] < best[1] ||
743 (candidate[1] == best[1] && candidate[2] < best[2])
744 best = candidate
745 end
746 end
747 emit!(g, "OPTIMIZE", args[1], "BASESTOCK", best[2],
748 "cost", money(best[1]), "total", money(best[3]))
749 else
750 fail("unknown command " * command)
751 end
752end
753
754function execute!(g::Game, line::String)
755 fields = String.(split(line))
756 command = fields[1]
757 try
758 command!(g, command, fields[2:end])
759 catch error
760 if error isa CommandError
761 emit!(g, "ERROR", error.message)
762 else
763 rethrow()
764 end
765 end
766end

768function main()
769 settings = [String.(split(readline(stdin))) for _ in 1:4]
770 game = Game(settings)
771 for line in eachline(stdin)
772 line == "END" && break
773 execute!(game, line)
774 end
775 if !isempty(game.output)
776 println(join(game.output, "\n"))
777 end
778end
