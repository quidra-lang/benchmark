20def decimal(numerator, denominator, places):
21 scale = 10 ** places
22 value = numerator * scale // denominator

26def variation(values):
27 return len(values) * sum(value * value for value in values) - sum(values) ** 2

30def parse_policy(tokens):
31 name = tokens[0]
32 if name not in PARAMETERS:
33 raise CommandError("unknown policy " + name)
34 count = PARAMETERS[name]
35 if len(tokens) - 1 != count:
36 raise CommandError(f"{name} needs {count} parameters")
37 values = [int(token.replace(".", "")) if name == "STERMAN" and index < 3 else int(token)
38 for index, token in enumerate(tokens[1:])]
39 if name == "FIXED":
40 valid = 0 <= values[0] <= 1000
41 elif name == "PASS":
42 valid = True
43 elif name == "BASESTOCK":
44 valid = 0 <= values[0] <= 10000
45 elif name == "MINMAX":
46 valid = 0 <= values[0] < values[1] <= 10000
47 elif name == "MOVAVG":
48 valid = 1 <= values[0] <= 52 and 0 <= values[1] <= 10000
49 elif name == "SMOOTH":
50 valid = 1 <= values[0] <= 10 and 0 <= values[1] <= 10000
51 else:
52 valid = all(0 <= value <= 100 for value in values[:3]) and 0 <= values[3] <= 10000
53 if not valid:
54 raise CommandError("invalid parameter for " + name)
55 return name, values, " ".join(tokens)

58class Stage:
59 def __init__(self, initial):
60 self.inventory = initial["inventory"]
61 self.backlog = 0
62 self.shipping = [initial["quantity"]] * initial["shipping"]
63 self.orders = [initial["quantity"]] * initial["orders"]
64 self.policy = initial["policy"]
65 self.expected = 100 * initial["quantity"]
66 self.holding = initial["holding"]
67 self.backlog_rate = initial["backlog"]
68 self.lot = 1
69 self.capacity = None
70 self.records = []

73class Simulation:
74 def __init__(self, initials, demands):
75 self.initials = initials
76 self.week = 0
77 self.demands = list(demands)
78 self.stages = [Stage(initial) for initial in initials]
79 self.plant_backlog = 0
80 self.plant_capacity = None
81 self.plant_records = []
82 self.visibility = "OFF"
83
84 def signal(self, index, week, current=None):
85 if self.visibility == "ON":
86 return self.demands[week - 1]
87 if current is not None and week == self.week + 1:
88 return current[index]
89 return self.stages[index].records[week - 1]["incoming"]
90
91 def set_policy(self, index, policy):
92 stage = self.stages[index]
93 stage.policy = policy
94 if policy[0] == "STERMAN":
95 stage.expected = 100 * (self.signal(index, self.week) if self.week else self.initials[index]["quantity"])
96
97 def positions(self):
98 supply = []
99 position = []
100 for index, stage in enumerate(self.stages):
101 upstream_backlog = self.stages[index + 1].backlog if index < 3 else self.plant_backlog
102 line = sum(stage.orders) + upstream_backlog + sum(stage.shipping)
103 supply.append(line)
104 position.append(stage.inventory + line - stage.backlog)
105 return supply, position
106
107 def policy_order(self, index, incoming, line, position):
108 stage = self.stages[index]
109 name, args, _ = stage.policy
110 week = self.week + 1
111 signal = self.signal(index, week, incoming)
112 if name == "FIXED":
113 return args[0]
114 if name == "PASS":
115 return incoming[index]
116 if name == "BASESTOCK":
117 return max(0, args[0] - position)
118 if name == "MINMAX":
119 return args[1] - position if position <= args[0] else 0
120 if name == "MOVAVG":
121 k, target = args
122 count = min(k, week)
123 forecast = sum(self.signal(index, u, incoming) for u in range(week - count + 1, week + 1)) // count
124 delay = self.initials[index]["shipping"] + self.initials[index]["orders"]
125 return max(0, target + forecast * delay - position)
126 if name == "SMOOTH":
127 k, target = args
128 return max(0, signal + (target - position) // k)
129 theta, alpha, beta, target = args
130 stage.expected = (100 * theta * signal + (100 - theta) * stage.expected) // 100
131 net = stage.inventory - stage.backlog
132 return max(0, (100 * stage.expected + alpha * (100 * (target - net) - beta * line)) // 10000)

134 def step(self):
135 received = []
136 for stage in self.stages:
137 quantity = stage.shipping.pop(0)
138 stage.inventory += quantity
139 received.append(quantity)
140 incoming = [self.demands[self.week]]
141 incoming.extend(self.stages[index].orders.pop(0) for index in range(3))
142 plant_incoming = self.stages[3].orders.pop(0)
143 records = []
144 for index, stage in enumerate(self.stages):
145 before = stage.backlog
146 requested = before + incoming[index]
147 shipped = min(stage.inventory, requested)
148 if stage.capacity is not None:
149 shipped = min(shipped, stage.capacity)
150 stage.inventory -= shipped
151 stage.backlog = requested - shipped
152 if index:
153 self.stages[index - 1].shipping.append(shipped)
154 records.append({
155 "received": received[index], "incoming": incoming[index], "shipped": shipped,
156 "before": before, "ontime": min(max(0, shipped - before), incoming[index]),
157 "inventory": stage.inventory, "backlog": stage.backlog,
158 })
159 requested = self.plant_backlog + plant_incoming
160 produced = requested if self.plant_capacity is None else min(requested, self.plant_capacity)
161 self.plant_backlog = requested - produced
162 self.stages[3].shipping.append(produced)
163 supply, positions = self.positions()
164 orders = [self.policy_order(index, incoming, supply[index], positions[index]) for index in range(4)]
165 for index, stage in enumerate(self.stages):
166 quantity = (orders[index] + stage.lot - 1) // stage.lot * stage.lot
167 stage.orders.append(quantity)
168 record = records[index]
169 record["order"] = quantity
170 record["holding"] = stage.holding * stage.inventory
171 record["backlog_cost"] = stage.backlog_rate * stage.backlog
172 record["cost"] = record["holding"] + record["backlog_cost"]
173 stage.records.append(record)
174 self.plant_records.append({"incoming": plant_incoming, "produced": produced, "backlog": self.plant_backlog})
175 self.week += 1
176 return records
177
178 def costs(self):
179 return [sum(record["cost"] for record in stage.records) for stage in self.stages]

182class Game:
183 def __init__(self, settings):
184 self.initials = []
185 for tokens in settings:
186 self.initials.append({
187 "inventory": int(tokens[1]), "quantity": int(tokens[2]),
188 "shipping": int(tokens[3]), "orders": int(tokens[4]),
189 "holding": int(tokens[5]), "backlog": int(tokens[6]),
190 "policy": parse_policy(tokens[7:]),
191 })
192 self.base = []
193 self.simulation = Simulation(self.initials, [])
194 self.snapshots = {}
195 self.output = []

198 self.output.append(" ".join(str(part) for part in parts))

212 def range_check(self, arguments, limit):
213 if not arguments:
214 return 1, limit
215 first, last = map(int, arguments)

230 def initialize(self, demands=None):
231 if demands is not None:
232 self.base = list(demands)

235 def trial(self, policies):
236 current = self.simulation
237 trial = Simulation(self.initials, current.demands)
238 trial.visibility = current.visibility
239 trial.plant_capacity = current.plant_capacity
240 for index, stage in enumerate(trial.stages):
241 previous = current.stages[index]
242 stage.lot = previous.lot
243 stage.capacity = previous.capacity
244 stage.holding = previous.holding
245 stage.backlog_rate = previous.backlog_rate
246 trial.set_policy(index, policies[index])
247 while trial.week < len(trial.demands):
248 trial.step()
249 return trial

251 def demand(self, values):

255 def generate(self, arguments):
256 weeks, base, spread, seed = map(int, arguments[:4])
257 changes = [(arguments[i], int(arguments[i + 1]), int(arguments[i + 2])) for i in range(4, len(arguments), 3)]
258 for _, week, _ in changes:
259 if not 1 <= week <= weeks:
260 raise CommandError(f"week {week} outside horizon")
261 levels = []
262 for week in range(1, weeks + 1):
263 level = base
264 for kind, start, amount in changes:
265 if kind == "STEP" and week >= start:
266 level += amount
267 elif kind == "PULSE" and week == start:
268 level += amount
269 elif kind == "RAMP" and week >= start:
270 level += amount * (week - start + 1)
271 if level < 0:
272 raise CommandError(f"negative level at week {week}")
273 levels.append(level)
274 values = []
275 for level in levels:
276 seed = 48271 * seed % 2147483647
277 values.append(level + seed % spread)
278 self.demand(values)

331 def bullwhip(self):
332 s = self.simulation
333 n = s.week
334 demands = s.demands[:n]
335 demand_var = variation(demands)
336 self.emit("BULLWHIP", "weeks", f"1-{n}", "demand", "mean", decimal(sum(demands), n, 4), "var", decimal(demand_var, n * n, 4))
337 previous = demand_var
338 for index, stage in enumerate(s.stages):
339 values = [r["order"] for r in stage.records]
340 v = variation(values)
341 self.emit(NAMES[index], "mean", decimal(sum(values), n, 4), "var", decimal(v, n * n, 4),
342 "ratio", "n/a" if demand_var == 0 else decimal(v, demand_var, 4),
343 "step", "n/a" if previous == 0 else decimal(v, previous, 4))
344 previous = v

346 def peaks(self):
347 s = self.simulation
348 self.emit("PEAKS", "weeks", f"1-{s.week}")
349 for index, stage in enumerate(s.stages):
350 inventories = [r["inventory"] for r in stage.records]
351 backlogs = [r["backlog"] for r in stage.records]
352 max_inventory = max(inventories)
353 max_backlog = max(backlogs)
354 longest = run = 0
355 start = best_start = best_end = 0
356 for week, backlog in enumerate(backlogs, 1):
357 if backlog > 0:
358 if run == 0:
359 start = week
360 run += 1
361 if run > longest:
362 longest, best_start, best_end = run, start, week
363 else:
364 run = 0
365 self.emit(NAMES[index], "max_inv", max_inventory, "week", inventories.index(max_inventory) + 1 if max_inventory else "-",
366 "max_back", max_backlog, "week", backlogs.index(max_backlog) + 1 if max_backlog else "-",
367 "longest_stockout", f"{longest} weeks {best_start}-{best_end}" if longest else 0)

369 def leadtime(self, index):
370 stage = self.simulation.stages[index]
371 initial = self.initials[index]
372 nominal = initial["shipping"] + initial["orders"]
373 cumulative = [received - nominal * initial["quantity"]
374 for received in accumulate(r["received"] for r in stage.records)]
375 ordered = count = 0
376 leads = []
377 for t, r in enumerate(stage.records):
378 ordered += r["order"]
379 if r["order"] == 0:
380 continue
381 count += 1
382 for u in range(t + 1, len(stage.records)):
383 if cumulative[u] >= ordered:
384 leads.append(u - t)
385 break
386 self.emit("LEADTIME", NAMES[index], "nominal", nominal, "orders", count, "completed", len(leads),
387 "open", count - len(leads), "min", min(leads) if leads else "-",
388 "avg", decimal(sum(leads), len(leads), 2) if leads else "-", "max", max(leads) if leads else "-")

404 length = value * 50 // maximum if maximum else 0
405 self.emit(f"{week:3d} |{'#' * length} {value}")

407 def command(self, command, a):
408 s = self.simulation
409 if command == "DEMAND":
410 self.demand(list(map(int, a)))
411 elif command == "GENERATE":
412 self.generate(a)
413 elif command == "RUN":
414 if s.week == len(s.demands):
415 raise CommandError(f"horizon reached at week {len(s.demands)}")
416 first = s.week + 1
417 count = min(int(a[0]), len(s.demands) - s.week)
418 cost = 0
419 for _ in range(count):
420 records = s.step()
421 for index, r in enumerate(records):
422 cost += r["cost"]
423 if r["before"] == 0 and r["backlog"] > 0:
424 self.emit("STOCKOUT", NAMES[index], "week", s.week, "backlog", r["backlog"])
425 elif r["before"] > 0 and r["backlog"] == 0:
426 self.emit("RECOVER", NAMES[index], "week", s.week)
427 self.emit("RUN", f"{first}-{s.week}", "demand", sum(s.demands[first - 1:s.week]), "cost", money(cost))
428 if s.week == len(s.demands):
429 self.emit("HORIZON", s.week)
430 elif command == "RESET":
431 self.initialize()
432 self.emit("RESET", "week", 0)
433 elif command == "SAVE":
434 self.snapshots[a[0]] = copy.deepcopy(s)
435 self.emit("SAVE", a[0], "week", s.week)
436 elif command == "LOAD":
437 if a[0] not in self.snapshots:
438 raise CommandError("no snapshot " + a[0])
439 self.simulation = copy.deepcopy(self.snapshots[a[0]])
440 self.emit("LOAD", a[0], "week", self.simulation.week)
441 elif command == "SET":
442 indices = self.target(a[0])
443 policy = parse_policy(a[1:])
444 for index in indices:
445 s.set_policy(index, policy)
446 self.emit("SET", a[0], policy[2])
447 elif command == "BATCH":
448 index = self.stage_index(a[0])
449 lot = int(a[1])
450 if not 1 <= lot <= 100:
451 raise CommandError("invalid batch size " + a[1])
452 s.stages[index].lot = lot
453 self.emit("BATCH", a[0], lot)
454 elif command == "CAPACITY":
455 index = None if a[0] == "PLANT" else self.stage_index(a[0])
456 capacity = None if a[1] == "OFF" else int(a[1])
457 if capacity is not None and capacity > 10000:
458 raise CommandError("invalid capacity " + a[1])
459 if index is None:
460 s.plant_capacity = capacity
461 else:
462 s.stages[index].capacity = capacity
463 self.emit("CAPACITY", a[0], a[1])
464 elif command == "COST":
465 index = self.stage_index(a[0])
466 holding, backlog = map(int, a[1:])
467 if holding > 1000 or backlog > 1000:
468 raise CommandError("invalid cost")
469 s.stages[index].holding = holding
470 s.stages[index].backlog_rate = backlog
471 self.emit("COST", a[0], "holding", holding, "backlog", backlog)
472 elif command == "VISIBILITY":
473 if a[0] not in ("ON", "OFF"):
474 raise CommandError("unknown visibility " + a[0])
475 s.visibility = a[0]
476 self.emit("VISIBILITY", a[0])
477 elif command == "SHOCK":
478 week, demand = map(int, a)
479 if not 1 <= week <= len(s.demands):
480 raise CommandError("week " + a[0] + " outside horizon")
481 if week <= s.week:
482 raise CommandError("week " + a[0] + " already simulated")
483 old = s.demands[week - 1]
484 s.demands[week - 1] = demand
485 self.emit("SHOCK", "week", week, old, "->", demand)
486 elif command == "STATUS":
487 self.show_status()
488 elif command == "SETTINGS":
489 self.emit("SETTINGS")
490 for index, stage in enumerate(s.stages):
491 initial = self.initials[index]
492 self.emit(NAMES[index], "holding", stage.holding, "backlog", stage.backlog_rate,
493 "batch", stage.lot, "capacity", "OFF" if stage.capacity is None else stage.capacity,
494 "delays", initial["shipping"], initial["orders"])
495 self.emit("PLANT", "capacity", "OFF" if s.plant_capacity is None else s.plant_capacity)
496 self.emit("VISIBILITY", s.visibility)
497 elif command == "WEEK":
498 self.require_weeks()
499 week = int(a[0])
500 if not 1 <= week <= s.week:
501 raise CommandError("bad week " + a[0])
502 self.show_week(week)
503 elif command == "TABLE":
504 index = self.stage_index(a[0])
505 self.require_weeks()
506 self.table(index, *self.range_check(a[1:], s.week))
507 elif command == "CHAIN":
508 self.require_weeks()
509 first, last = self.range_check(a, s.week)
510 self.emit("CHAIN", f"{first}-{last}")
511 self.emit(f"{'week':>4} {'demand':>6} " + " ".join(name.rjust(11) for name in NAMES))
512 for week in range(first, last + 1):
513 orders = [stage.records[week - 1]["order"] for stage in s.stages]
514 self.emit(f"{week:4d} {s.demands[week - 1]:6d} " + " ".join(str(value).rjust(11) for value in orders))
515 elif command == "DEMANDS":
516 first, last = self.range_check(a, len(s.demands))
517 values = s.demands[first - 1:last]
518 self.emit("DEMANDS", f"{first}-{last}", "total", sum(values), "min", min(values), "max", max(values))
519 self.emit(*values)
520 elif command == "COSTS":
521 self.costs()
522 elif command == "SUMMARY":
523 self.require_weeks()
524 self.emit("SUMMARY", "weeks", f"1-{s.week}")
525 for index, stage in enumerate(s.stages):
526 orders = [r["order"] for r in stage.records]
527 self.emit(NAMES[index], "avg_inv", decimal(sum(r["inventory"] for r in stage.records), s.week, 2),
528 "avg_back", decimal(sum(r["backlog"] for r in stage.records), s.week, 2),
529 "orders", sum(orders), "min", min(orders), "max", max(orders), "zero_weeks", orders.count(0))
530 elif command == "BALANCE":
531 self.emit("BALANCE", "week", s.week)
532 for index, stage in enumerate(s.stages):
533 self.emit(NAMES[index], "start", self.initials[index]["inventory"],
534 "received", sum(r["received"] for r in stage.records), "shipped", sum(r["shipped"] for r in stage.records),
535 "end", stage.inventory, "owed", stage.backlog)
536 initial_goods = sum(i["inventory"] + i["quantity"] * i["shipping"] for i in self.initials)
537 self.emit("SYSTEM", "initial", initial_goods, "produced", sum(r["produced"] for r in s.plant_records),
538 "delivered", sum(r["shipped"] for r in s.stages[0].records), "stock", sum(stage.inventory for stage in s.stages),
539 "transit", sum(sum(stage.shipping) for stage in s.stages))
540 elif command == "FILLRATE":
541 self.require_weeks()
542 self.emit("FILLRATE", "weeks", f"1-{s.week}")
543 for index, stage in enumerate(s.stages):
544 ordered, ontime, fill = self.fill_text(stage.records)
545 stockouts = sum(r["backlog"] > 0 for r in stage.records)
546 self.emit(NAMES[index], "fill", fill, "ontime", ontime, "ordered", ordered, "stockout_weeks", stockouts,
547 "service", decimal((s.week - stockouts) * 100, s.week, 2) + "%")
548 elif command == "BULLWHIP":
549 self.require_weeks()
550 self.bullwhip()
551 elif command == "PEAKS":
552 self.require_weeks()
553 self.peaks()
554 elif command == "LEADTIME":
555 index = self.stage_index(a[0])
556 self.require_weeks()
557 self.leadtime(index)
558 elif command in ("PLOT", "HISTOGRAM"):
559 index = self.stage_index(a[0])
560 self.mode_check(a[1])
561 if command == "HISTOGRAM":
562 width = int(a[2])
563 if not 1 <= width <= 100:
564 raise CommandError("invalid width " + a[2])
565 self.require_weeks()
566 if command == "PLOT":
567 self.plot(index, a[1], *self.range_check(a[2:], s.week))
568 else:
569 bins = Counter(value // width for value in self.values(index, a[1]))
570 self.emit("HISTOGRAM", a[0], a[1], "width", width, "weeks", f"1-{s.week}")
571 for bucket in range(min(bins), max(bins) + 1):
572 count = bins.get(bucket, 0)
573 self.emit(f"{bucket * width}..{bucket * width + width - 1}", count, *(["*" * count] if count else []))
574 elif command == "COMPARE":
575 policies = []
576 start = 0
577 for end in range(len(a) + 1):
578 if end == len(a) or a[end] == "/":
579 policies.append(parse_policy(a[start:end]))
580 start = end + 1
581 rows = []
582 for index, policy in enumerate(policies):
583 trial = self.trial([policy] * 4)
584 cost = sum(trial.costs())
585 _, _, fill = self.fill_text(trial.stages[0].records)
586 demand_variation = variation(trial.demands)
587 factory_variation = variation([r["order"] for r in trial.stages[3].records])
588 ratio = "n/a" if demand_variation == 0 else decimal(factory_variation, demand_variation, 4)
589 rows.append((cost, index, policy[2], fill, ratio))
590 self.emit("COMPARE", len(policies), "weeks", len(s.demands))
591 for rank, (cost, _, policy, fill, ratio) in enumerate(sorted(rows), 1):
592 self.emit(rank, policy, "total", money(cost), "fill", fill, "bullwhip", ratio)
593 elif command == "OPTIMIZE":
594 indices = self.target(a[0])
595 low, high = map(int, a[1:])
596 if low > high:
597 raise CommandError(f"bad range {a[1]} {a[2]}")
598 if high - low > 200:
599 raise CommandError("range too large")
600 best = None
601 for target in range(low, high + 1):
602 policies = [stage.policy for stage in s.stages]
603 for index in indices:
604 policies[index] = ("BASESTOCK", [target], f"BASESTOCK {target}")
605 trial = self.trial(policies)
606 costs = trial.costs()
607 objective = sum(costs[index] for index in indices)
608 row = objective, target, sum(costs)
609 if best is None or row[:2] < best[:2]:
610 best = row
611 self.emit("OPTIMIZE", a[0], "BASESTOCK", best[1], "cost", money(best[0]), "total", money(best[2]))
612 else:
613 raise CommandError("unknown command " + command)

616 command, *arguments = line.split()
617 try:
618 self.command(command, arguments)
619 except CommandError as error:
620 self.emit("ERROR", error)
