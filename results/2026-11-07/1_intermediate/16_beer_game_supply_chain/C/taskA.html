73static char *read_line(void) {
74 size_t length = 0, capacity = 128;
75 char *line = resize(NULL, capacity);
76 int ch;
77 while ((ch = getchar()) != EOF && ch != '\n') {
78 if (length + 1 == capacity) line = resize(line, capacity *= 2);
79 line[length++] = (char)ch;
80 }
81 if (!length && ch == EOF) { free(line); return NULL; }
82 line[length] = '\0';
83 return line;
84}

86static char **words(char *line, int *count) {
87 int capacity = 16;
88 char **tokens = resize(NULL, (size_t)capacity * sizeof *tokens);
89 *count = 0;
90 for (char *token = strtok(line, " "); token; token = strtok(NULL, " ")) {
91 if (*count == capacity) tokens = resize(tokens, (size_t)(capacity *= 2) * sizeof *tokens);
92 tokens[(*count)++] = token;
93 }
94 return tokens;
95}

113static Text decimal(Number numerator, Number denominator, int places) {
114 Text result;
115 if (!denominator) { strcpy(result.text, "n/a"); return result; }
116 Number scale = places == 2 ? 100 : 10000;
117 Number value = numerator * scale / denominator;
118 snprintf(result.text, sizeof result.text, "%" PRId64 ".%0*" PRId64,
119 value / scale, places, value % scale);
120 return result;
121}

149static Statistics statistics(const Number *values, int count) {
150 Statistics result = { 0, INT64_MAX, INT64_MIN };
151 for (int i = 0; i < count; ++i) {
152 result.total += values[i];
153 result.minimum = minimum(result.minimum, values[i]);
154 result.maximum = maximum(result.maximum, values[i]);
155 }
156 return result;

159static Number variation(const Number *values, int count) {
160 Number squares = 0, total = sum(values, count);
161 for (int i = 0; i < count; ++i) squares += values[i] * values[i];
162 return count * squares - total * total;
163}

195static bool parse_policy(char **args, int count, Policy *result) {
196 int kind = -1;
197 for (int i = 0; i < 7; ++i) if (equal(args[0], policy_names[i])) kind = i;
198 if (kind < 0) return fail("unknown policy %s", args[0]);
199 const int parameter_counts[] = { 1, 0, 1, 2, 2, 2, 4 };
200 int parameters = parameter_counts[kind];
201 if (count != parameters + 1) return fail("%s needs %d parameters", args[0], parameters);
202 Policy p = { .kind = (PolicyKind)kind };
203 for (int i = 0; i < parameters; ++i) {
204 p.values[i] = number(args[i + 1]);
205 if (kind == STERMAN && i < 3)
206 p.values[i] = p.values[i] * 100 + number(strchr(args[i + 1], '.') + 1);
207 }
208 Number *v = p.values;
209 bool valid = false;
210 switch (p.kind) {
211 case FIXED: valid = v[0] <= 1000; break;
212 case PASS: valid = true; break;
213 case BASESTOCK: valid = v[0] <= 10000; break;
214 case MINMAX: valid = v[0] < v[1] && v[1] <= 10000; break;
215 case MOVAVG: valid = v[0] >= 1 && v[0] <= 52 && v[1] <= 10000; break;
216 case SMOOTH: valid = v[0] >= 1 && v[0] <= 10 && v[1] <= 10000; break;
217 case STERMAN: valid = v[0] <= 100 && v[1] <= 100 && v[2] <= 100 && v[3] <= 10000; break;
218 }
219 if (!valid) return fail("invalid parameter for %s", args[0]);
220 for (int i = 0; i < count; ++i) {
221 if (i) strcat(p.notation, " ");
222 strcat(p.notation, args[i]);
223 }
224 *result = p;
225 return true;
226}

228static void initialize(Simulation *sim, const Number *demand, int horizon) {
229 memset(sim, 0, sizeof *sim);
230 sim->horizon = horizon;
231 memcpy(sim->demand, demand, (size_t)horizon * sizeof *demand);
232 sim->plant_capacity = -1;
233 for (int i = 0; i < 4; ++i) {
234 Stage *s = &sim->stages[i];
235 Initial *cfg = &initial[i];
236 s->inventory = cfg->inventory;
237 s->shipping_delay = cfg->shipping_delay;
238 s->order_delay = cfg->order_delay;
239 for (int j = 0; j < 4; ++j) s->shipping[j] = s->orders[j] = cfg->quantity;
240 s->policy = cfg->policy;
241 s->expected = 100 * cfg->quantity;
242 s->holding_rate = cfg->holding;
243 s->backlog_rate = cfg->backlog;
244 s->lot = 1;
245 s->capacity = -1;
246 }
247}

260 Number value = pipeline[0];

262 pipeline[length - 1] = 0;

266static void step(Simulation *sim) {
267 Record rows[4] = { 0 };
268 PlantRecord plant = { 0 };
269 for (int i = 0; i < 4; ++i) {
270 Stage *s = &sim->stages[i];
271 rows[i].received = pop(s->shipping, s->shipping_delay);
272 s->inventory += rows[i].received;
273 }
274 rows[0].incoming = sim->demand[sim->week];
275 for (int i = 0; i < 4; ++i) {
276 Stage *s = &sim->stages[i];
277 Number incoming = pop(s->orders, s->order_delay);
278 if (i == 3) plant.incoming = incoming;
279 else rows[i + 1].incoming = incoming;
280 }
281 for (int i = 0; i < 4; ++i) {
282 Stage *s = &sim->stages[i];
283 Record *r = &rows[i];
284 r->before = s->backlog;
285 Number requested = s->backlog + r->incoming;
286 r->shipped = minimum(s->inventory, requested);
287 if (s->capacity >= 0) r->shipped = minimum(r->shipped, s->capacity);
288 s->inventory -= r->shipped;
289 s->backlog = requested - r->shipped;
290 r->ontime = minimum(maximum(0, r->shipped - r->before), r->incoming);
291 r->inventory = s->inventory;
292 r->backlog = s->backlog;
293 if (i) {
294 Stage *downstream = &sim->stages[i - 1];
295 downstream->shipping[downstream->shipping_delay - 1] = r->shipped;
296 }
297 }
298 plant.produced = sim->plant_backlog + plant.incoming;
299 if (sim->plant_capacity >= 0) plant.produced = minimum(plant.produced, sim->plant_capacity);
300 sim->plant_backlog += plant.incoming - plant.produced;
301 plant.backlog = sim->plant_backlog;
302 Stage *factory = &sim->stages[3];
303 factory->shipping[factory->shipping_delay - 1] = plant.produced;
304 Number supplies[4];
305 for (int i = 0; i < 4; ++i) supplies[i] = supply(sim, i);
306 for (int i = 0; i < 4; ++i) {
307 Stage *s = &sim->stages[i];
308 Record *r = &rows[i];
309 Number position = s->inventory + supplies[i] - s->backlog;
310 Number demand = sim->visible ? sim->demand[sim->week] : r->incoming;
311 Number *p = s->policy.values, quantity = 0;
312 switch (s->policy.kind) {
313 case FIXED: quantity = p[0]; break;
314 case PASS: quantity = r->incoming; break;
315 case BASESTOCK: quantity = maximum(0, p[0] - position); break;
316 case MINMAX: quantity = position <= p[0] ? p[1] - position : 0; break;
317 case MOVAVG: {
318 int count = (int)minimum(p[0], sim->week + 1);
319 Number total = demand;
320 for (int t = sim->week + 1 - count; t < sim->week; ++t) total += signal(sim, i, t);
321 Number forecast = total / count;
322 quantity = maximum(0, p[1] + forecast * (s->shipping_delay + s->order_delay) - position);
323 break;
324 }
325 case SMOOTH: quantity = maximum(0, demand + floor_div(p[1] - position, p[0])); break;
326 case STERMAN: {
327 s->expected = floor_div(100 * p[0] * demand + (100 - p[0]) * s->expected, 100);
328 Number net = s->inventory - s->backlog;
329 quantity = maximum(0, floor_div(100 * s->expected + p[1] * (100 * (p[3] - net) - p[2] * supplies[i]), 10000));
330 break;
331 }
332 }
333 r->order = (quantity + s->lot - 1) / s->lot * s->lot;
334 s->orders[s->order_delay - 1] = r->order;
335 r->holding_cost = s->holding_rate * s->inventory;
336 r->backlog_cost = s->backlog_rate * s->backlog;
337 s->records[sim->week] = *r;
338 }
339 sim->plant_records[sim->week++] = plant;
340}

342static void trial(Simulation *sim, const Policy *policies) {
343 initialize(sim, current.demand, current.horizon);
344 sim->visible = current.visible;
345 sim->plant_capacity = current.plant_capacity;
346 for (int i = 0; i < 4; ++i) {
347 Stage *s = &sim->stages[i], *old = &current.stages[i];
348 s->policy = policies[i];
349 s->lot = old->lot;
350 s->capacity = old->capacity;
351 s->holding_rate = old->holding_rate;
352 s->backlog_rate = old->backlog_rate;
353 }
354 while (sim->week < sim->horizon) step(sim);
355}

409static void execute(char **a, int count) {
410 const char *op = a[0], *id = count > 1 ? a[1] : "";
411 Simulation *sim = &current;
412 int n = sim->week;
413 if (equal(op, "DEMAND")) {
414 Number demand[104];
415 for (int i = 1; i < count; ++i) demand[i - 1] = number(a[i]);
416 set_demand(demand, count - 1);
417 } else if (equal(op, "GENERATE")) {
418 int horizon = (int)number(id);
419 for (int i = 5; i < count; i += 3) {
420 Number week = number(a[i + 1]);
421 if (week < 1 || week > horizon) { fail("week %s outside horizon", a[i + 1]); return; }
422 }
423 Number demand[104], random = number(a[4]);
424 for (int t = 1; t <= horizon; ++t) {
425 Number level = number(a[2]);
426 for (int i = 5; i < count; i += 3) {
427 Number elapsed = t - number(a[i + 1]), amount = number(a[i + 2]);
428 if (elapsed < 0) continue;
429 if (equal(a[i], "STEP")) level += amount;
430 else if (equal(a[i], "RAMP")) level += amount * (elapsed + 1);
431 else if (!elapsed) level += amount;
432 }
433 if (level < 0) { fail("negative level at week %d", t); return; }
434 random = 48271 * random % 2147483647;
435 demand[t - 1] = level + random % number(a[3]);
436 }
437 set_demand(demand, horizon);
438 } else if (equal(op, "RUN")) {
439 if (n == sim->horizon) { fail("horizon reached at week %d", n); return; }
440 int end = (int)minimum(sim->horizon, n + number(id));
441 Number total = 0;
442 while (sim->week < end) {
443 step(sim);
444 for (int i = 0; i < 4; ++i) {
445 Record r = sim->stages[i].records[sim->week - 1];
446 total += cost(r);
447 if (!r.before && r.backlog > 0) printf("STOCKOUT %s week %d backlog %" PRId64 "\n", stage_names[i], sim->week, r.backlog);
448 if (r.before > 0 && !r.backlog) printf("RECOVER %s week %d\n", stage_names[i], sim->week);
449 }
450 }
451 printf("RUN %d-%d demand %" PRId64 " cost %s\n", n + 1, end, sum(sim->demand + n, end - n), money(total).text);
452 if (end == sim->horizon) printf("HORIZON %d\n", end);
453 } else if (equal(op, "RESET")) {
454 initialize(sim, base_demand, base_horizon);
455 puts("RESET week 0");
456 } else if (equal(op, "SAVE") || equal(op, "LOAD")) {
457 int i = 0;
458 while (i < snapshot_count && !equal(snapshots[i].name, id)) ++i;
459 if (equal(op, "LOAD")) {
460 if (i == snapshot_count) { fail("no snapshot %s", id); return; }
461 *sim = *snapshots[i].state;
462 } else {
463 if (i == snapshot_count) {
464 strcpy(snapshots[i].name, id);
465 snapshots[i].state = resize(NULL, sizeof *sim);
466 ++snapshot_count;
467 }
468 *snapshots[i].state = *sim;
469 }
470 printf("%s %s week %d\n", op, id, sim->week);
471 } else if (equal(op, "SET")) {
472 int selected;
473 Policy policy;
474 if (!target(id, &selected) || !parse_policy(a + 2, count - 2, &policy)) return;
475 for (int i = 0; i < 4; ++i) {
476 if (selected >= 0 && selected != i) continue;
477 sim->stages[i].policy = policy;
478 if (policy.kind == STERMAN) sim->stages[i].expected = 100 * (n ? signal(sim, i, n - 1) : initial[i].quantity);
479 }
480 printf("SET %s %s\n", id, policy.notation);
481 } else if (equal(op, "BATCH") || equal(op, "CAPACITY") || equal(op, "COST")) {
482 int i = equal(op, "CAPACITY") && equal(id, "PLANT") ? 4 : stage_index(id);
483 if (i < 0) return;
484 if (equal(op, "BATCH")) {
485 Number lot = number(a[2]);
486 if (lot < 1 || lot > 100) { fail("invalid batch size %s", a[2]); return; }
487 sim->stages[i].lot = lot;
488 printf("BATCH %s %" PRId64 "\n", id, lot);
489 } else if (equal(op, "CAPACITY")) {
490 Number cap = equal(a[2], "OFF") ? -1 : number(a[2]);
491 if (cap > 10000) { fail("invalid capacity %s", a[2]); return; }
492 if (i == 4) sim->plant_capacity = cap;
493 else sim->stages[i].capacity = cap;
494 printf("CAPACITY %s %s\n", id, a[2]);
495 } else {
496 Number holding = number(a[2]), backlog = number(a[3]);
497 if (holding > 1000 || backlog > 1000) { fail("invalid cost"); return; }
498 sim->stages[i].holding_rate = holding;
499 sim->stages[i].backlog_rate = backlog;
500 printf("COST %s holding %" PRId64 " backlog %" PRId64 "\n", id, holding, backlog);
501 }
502 } else if (equal(op, "VISIBILITY")) {
503 if (!equal(id, "ON") && !equal(id, "OFF")) { fail("unknown visibility %s", id); return; }
504 sim->visible = equal(id, "ON");
505 printf("VISIBILITY %s\n", id);
506 } else if (equal(op, "SHOCK")) {
507 int week = (int)number(id);
508 if (week < 1 || week > sim->horizon) { fail("week %s outside horizon", id); return; }
509 if (week <= n) { fail("week %s already simulated", id); return; }
510 Number old = sim->demand[week - 1];
511 sim->demand[week - 1] = number(a[2]);
512 printf("SHOCK week %d %" PRId64 " -> %" PRId64 "\n", week, old, sim->demand[week - 1]);
513 } else if (equal(op, "STATUS")) {
514 printf("STATUS week %d of %d\n", n, sim->horizon);
515 for (int i = 0; i < 4; ++i) {
516 Stage *s = &sim->stages[i];
517 printf("%s inv %" PRId64 " back %" PRId64 " transit ", stage_names[i], s->inventory, s->backlog);
518 pipeline(s->shipping, s->shipping_delay);
519 fputs(" pending ", stdout);
520 pipeline(s->orders, s->order_delay);
521 Number line = supply(sim, i);
522 printf(" supply %" PRId64 " position %" PRId64 " policy %s", line, s->inventory + line - s->backlog, s->policy.notation);
523 if (s->policy.kind == STERMAN) printf(" expect %s", money(s->expected).text);
524 putchar('\n');
525 }
526 printf("PLANT back %" PRId64 "\n", sim->plant_backlog);
527 } else if (equal(op, "SETTINGS")) {
528 puts("SETTINGS");
529 for (int i = 0; i < 4; ++i) {
530 Stage *s = &sim->stages[i];
531 printf("%s holding %" PRId64 " backlog %" PRId64 " batch %" PRId64 " capacity %s delays %d %d\n",
532 stage_names[i], s->holding_rate, s->backlog_rate, s->lot, capacity_text(s->capacity).text, s->shipping_delay, s->order_delay);
533 }
534 printf("PLANT capacity %s\nVISIBILITY %s\n", capacity_text(sim->plant_capacity).text, sim->visible ? "ON" : "OFF");
535 } else if (equal(op, "WEEK")) {
536 if (!require_weeks()) return;
537 int week = (int)number(id);
538 if (week < 1 || week > n) { fail("bad week %s", id); return; }
539 printf("WEEK %d demand %" PRId64 "\n", week, sim->demand[week - 1]);
540 for (int i = 0; i < 4; ++i) {
541 Record r = sim->stages[i].records[week - 1];
542 printf("%s recv %" PRId64 " in %" PRId64 " ship %" PRId64 " inv %" PRId64 " back %" PRId64 " order %" PRId64 " cost %s\n",
543 stage_names[i], r.received, r.incoming, r.shipped, r.inventory, r.backlog, r.order, money(cost(r)).text);
544 }
545 PlantRecord p = sim->plant_records[week - 1];
546 printf("PLANT in %" PRId64 " made %" PRId64 " back %" PRId64 "\n", p.incoming, p.produced, p.backlog);
547 } else if (equal(op, "TABLE") || equal(op, "CHAIN") || equal(op, "DEMANDS")) {
548 int i = -1, skip = equal(op, "TABLE") ? 2 : 1;
549 if (skip == 2 && (i = stage_index(id)) < 0) return;
550 if (!equal(op, "DEMANDS") && !require_weeks()) return;
551 Range r;
552 if (!range(a + skip, count - skip, equal(op, "DEMANDS") ? sim->horizon : n, &r)) return;
553 if (equal(op, "TABLE")) {
554 printf("TABLE %s %d-%d\nweek in ship recv inv back order cost\n", id, r.first + 1, r.end);
555 for (int t = r.first; t < r.end; ++t) {
556 Record v = sim->stages[i].records[t];
557 printf("%4d %6" PRId64 " %6" PRId64 " %6" PRId64 " %6" PRId64 " %6" PRId64 " %6" PRId64 " %9s\n",
558 t + 1, v.incoming, v.shipped, v.received, v.inventory, v.backlog, v.order, money(cost(v)).text);
559 }
560 } else if (equal(op, "CHAIN")) {
561 printf("CHAIN %d-%d\n%4s %6s", r.first + 1, r.end, "week", "demand");
562 for (int j = 0; j < 4; ++j) printf(" %11s", stage_names[j]);
563 putchar('\n');
564 for (int t = r.first; t < r.end; ++t) {
565 printf("%4d %6" PRId64, t + 1, sim->demand[t]);
566 for (int j = 0; j < 4; ++j) printf(" %11" PRId64, sim->stages[j].records[t].order);
567 putchar('\n');
568 }
569 } else {
570 Statistics s = statistics(sim->demand + r.first, r.end - r.first);
571 printf("DEMANDS %d-%d total %" PRId64 " min %" PRId64 " max %" PRId64 "\n", r.first + 1, r.end, s.total, s.minimum, s.maximum);
572 for (int t = r.first; t < r.end; ++t) printf("%s%" PRId64, t == r.first ? "" : " ", sim->demand[t]);
573 putchar('\n');
574 }
575 } else if (equal(op, "COSTS")) {
576 printf("COSTS week %d\n", n);
577 Costs total = { 0 };
578 for (int i = 0; i < 4; ++i) {
579 Costs c = stage_costs(sim, i);
580 printf("%s holding %s backlog %s total %s\n", stage_names[i], money(c.holding).text, money(c.backlog).text, money(c.holding + c.backlog).text);
581 total.holding += c.holding;
582 total.backlog += c.backlog;
583 }
584 printf("TOTAL holding %s backlog %s total %s\n", money(total.holding).text, money(total.backlog).text, money(total.holding + total.backlog).text);
585 } else if (equal(op, "SUMMARY")) {
586 if (!require_weeks()) return;
587 printf("SUMMARY weeks 1-%d\n", n);
588 for (int i = 0; i < 4; ++i) {
589 Number inventory = 0, backlog = 0, orders[104];
590 int zero = 0;
591 values(sim, i, ORDERS, orders);
592 Statistics s = statistics(orders, n);
593 for (int t = 0; t < n; ++t) {
594 Record r = sim->stages[i].records[t];
595 inventory += r.inventory;
596 backlog += r.backlog;
597 zero += r.order == 0;
598 }
599 printf("%s avg_inv %s avg_back %s orders %" PRId64 " min %" PRId64 " max %" PRId64 " zero_weeks %d\n",
600 stage_names[i], decimal(inventory, n, 2).text, decimal(backlog, n, 2).text, s.total, s.minimum, s.maximum, zero);
601 }
602 } else if (equal(op, "BALANCE")) {
603 printf("BALANCE week %d\n", n);
604 Number goods = 0, stock = 0, transit = 0, delivered = 0, produced = 0;
605 for (int i = 0; i < 4; ++i) {
606 Stage *s = &sim->stages[i];
607 Number received = 0, shipped = 0;
608 for (int t = 0; t < n; ++t) { received += s->records[t].received; shipped += s->records[t].shipped; }
609 printf("%s start %" PRId64 " received %" PRId64 " shipped %" PRId64 " end %" PRId64 " owed %" PRId64 "\n",
610 stage_names[i], initial[i].inventory, received, shipped, s->inventory, s->backlog);
611 goods += initial[i].inventory + initial[i].quantity * s->shipping_delay;
612 stock += s->inventory;
613 transit += sum(s->shipping, s->shipping_delay);
614 if (!i) delivered = shipped;
615 }
616 for (int t = 0; t < n; ++t) produced += sim->plant_records[t].produced;
617 printf("SYSTEM initial %" PRId64 " produced %" PRId64 " delivered %" PRId64 " stock %" PRId64 " transit %" PRId64 "\n", goods, produced, delivered, stock, transit);
618 } else if (equal(op, "FILLRATE")) {
619 if (!require_weeks()) return;
620 printf("FILLRATE weeks 1-%d\n", n);
621 for (int i = 0; i < 4; ++i) {
622 Number ontime = 0, incoming = 0;
623 int stockouts = 0;
624 for (int t = 0; t < n; ++t) {
625 Record r = sim->stages[i].records[t];
626 ontime += r.ontime; incoming += r.incoming; stockouts += r.backlog > 0;
627 }
628 printf("%s fill %s ontime %" PRId64 " ordered %" PRId64 " stockout_weeks %d service %s\n",
629 stage_names[i], percentage(ontime, incoming).text, ontime, incoming, stockouts, percentage(n - stockouts, n).text);
630 }
631 } else if (equal(op, "BULLWHIP")) {
632 if (!require_weeks()) return;
633 Number demand_var = variation(sim->demand, n), previous = demand_var;
634 printf("BULLWHIP weeks 1-%d demand mean %s var %s\n", n, decimal(sum(sim->demand, n), n, 4).text, decimal(demand_var, n * n, 4).text);
635 for (int i = 0; i < 4; ++i) {
636 Number orders[104];
637 values(sim, i, ORDERS, orders);
638 Number v = variation(orders, n);
639 printf("%s mean %s var %s ratio %s step %s\n", stage_names[i], decimal(sum(orders, n), n, 4).text,
640 decimal(v, n * n, 4).text, decimal(v, demand_var, 4).text, decimal(v, previous, 4).text);
641 previous = v;
642 }
643 } else if (equal(op, "PEAKS")) {
644 if (!require_weeks()) return;
645 printf("PEAKS weeks 1-%d\n", n);
646 for (int i = 0; i < 4; ++i) {
647 Number inventory = 0, backlog = 0;
648 int inv_week = 0, back_week = 0, streak = 0, longest = 0, last = 0;
649 for (int t = 0; t < n; ++t) {
650 Record r = sim->stages[i].records[t];
651 if (r.inventory > inventory) { inventory = r.inventory; inv_week = t + 1; }
652 if (r.backlog > backlog) { backlog = r.backlog; back_week = t + 1; }
653 streak = r.backlog > 0 ? streak + 1 : 0;
654 if (streak > longest) { longest = streak; last = t + 1; }
655 }
656 printf("%s max_inv %" PRId64 " week %s max_back %" PRId64 " week %s longest_stockout %d",
657 stage_names[i], inventory, week_text(inv_week).text, backlog, week_text(back_week).text, longest);
658 if (longest) printf(" weeks %d-%d", last - longest + 1, last);
659 putchar('\n');
660 }
661 } else if (equal(op, "LEADTIME")) {
662 int i = stage_index(id);
663 if (i < 0 || !require_weeks()) return;
664 Stage *s = &sim->stages[i];
665 int nominal = s->shipping_delay + s->order_delay;
666 Number arrivals[104], received = -nominal * initial[i].quantity, ordered = 0, total = 0;
667 for (int t = 0; t < n; ++t) { received += s->records[t].received; arrivals[t] = received; }
668 int orders = 0, completed = 0, low = INT_MAX, high = 0;
669 for (int t = 0; t < n; ++t) {
670 ordered += s->records[t].order;
671 if (!s->records[t].order) continue;
672 ++orders;
673 for (int u = t + 1; u < n; ++u) {
674 if (arrivals[u] < ordered) continue;
675 int lead = u - t;
676 if (lead < low) low = lead;
677 if (lead > high) high = lead;
678 total += lead;
679 ++completed;
680 break;
681 }
682 }
683 printf("LEADTIME %s nominal %d orders %d completed %d open %d", id, nominal, orders, completed, orders - completed);
684 if (!completed) puts(" min - avg - max -");
685 else printf(" min %d avg %s max %d\n", low, decimal(total, completed, 2).text, high);
686 } else if (equal(op, "PLOT") || equal(op, "HISTOGRAM")) {
687 int i = stage_index(id);
688 if (i < 0) return;
689 int metric = metric_index(a[2]);
690 if (metric < 0) return;
691 bool histogram = equal(op, "HISTOGRAM");
692 Number width = histogram ? number(a[3]) : 0;
693 if (histogram && (width < 1 || width > 100)) { fail("invalid width %s", a[3]); return; }
694 if (!require_weeks()) return;
695 Range r = { 0, n };
696 if (!histogram && !range(a + 3, count - 3, n, &r)) return;
697 Number all[104];
698 values(sim, i, (Metric)metric, all);
699 Statistics stats = statistics(all + r.first, r.end - r.first);
700 if (histogram) {
701 printf("HISTOGRAM %s %s width %" PRId64 " weeks 1-%d\n", id, a[2], width, n);
702 Number last = floor_div(stats.maximum, width);
703 for (Number bin = floor_div(stats.minimum, width); bin <= last; ++bin) {
704 int frequency = 0;
705 for (int t = 0; t < n; ++t) frequency += floor_div(all[t], width) == bin;
706 printf("%" PRId64 "..%" PRId64 " %d", bin * width, bin * width + width - 1, frequency);
707 if (frequency) { putchar(' '); repeat('*', frequency); }
708 putchar('\n');
709 }
710 } else if (metric == NET) {
711 printf("PLOT %s NET %d-%d min %" PRId64 " max %" PRId64 "\n", id, r.first + 1, r.end, stats.minimum, stats.maximum);
712 Number magnitude = maximum(absolute(stats.minimum), absolute(stats.maximum));
713 for (int t = r.first; t < r.end; ++t) {
714 Number bars = magnitude ? absolute(all[t]) * 25 / magnitude : 0;
715 printf("%3d ", t + 1);
716 repeat(' ', all[t] < 0 ? 25 - bars : 25);
717 if (all[t] < 0) repeat('-', bars);
718 putchar('|');
719 if (all[t] > 0) repeat('+', bars);
720 printf(" %" PRId64 "\n", all[t]);
721 }
722 } else {
723 printf("PLOT %s %s %d-%d max %" PRId64 "\n", id, a[2], r.first + 1, r.end, stats.maximum);
724 for (int t = r.first; t < r.end; ++t) {
725 printf("%3d |", t + 1);
726 repeat('#', stats.maximum ? all[t] * 50 / stats.maximum : 0);
727 printf(" %" PRId64 "\n", all[t]);
728 }
729 }
730 } else if (equal(op, "COMPARE")) {
731 Policy policies[8];
732 Comparison results[8];
733 int start = 1, length = 0;
734 for (int i = 1; i <= count; ++i) {
735 if (i < count && !equal(a[i], "/")) continue;
736 if (!parse_policy(a + start, i - start, &policies[length++])) return;
737 start = i + 1;
738 }
739 for (int i = 0; i < length; ++i) {
740 Policy all[4];
741 for (int j = 0; j < 4; ++j) all[j] = policies[i];
742 Simulation candidate;
743 trial(&candidate, all);
744 Comparison r = { .policy = policies[i], .index = i, .cost = total_cost(&candidate) };
745 for (int t = 0; t < candidate.week; ++t) {
746 r.ontime += candidate.stages[0].records[t].ontime;
747 r.incoming += candidate.stages[0].records[t].incoming;
748 }
749 Number orders[104];
750 values(&candidate, 3, ORDERS, orders);
751 r.variation = variation(orders, candidate.week);
752 results[i] = r;
753 }
754 qsort(results, (size_t)length, sizeof *results, compare_results);
755 Number demand_variation = variation(sim->demand, sim->horizon);
756 printf("COMPARE %d weeks %d\n", length, sim->horizon);
757 for (int i = 0; i < length; ++i) {
758 Comparison r = results[i];
759 printf("%d %s total %s fill %s bullwhip %s\n", i + 1, r.policy.notation, money(r.cost).text,
760 percentage(r.ontime, r.incoming).text, decimal(r.variation, demand_variation, 4).text);
761 }
762 } else if (equal(op, "OPTIMIZE")) {
763 int selected;
764 if (!target(id, &selected)) return;
765 Number low = number(a[2]), high = number(a[3]);
766 if (low > high) { fail("bad range %s %s", a[2], a[3]); return; }
767 if (high - low > 200) { fail("range too large"); return; }
768 Number best = low, best_cost = INT64_MAX, best_total = 0;
769 for (Number stock = low; stock <= high; ++stock) {
770 Policy policies[4];
771 for (int i = 0; i < 4; ++i)
772 policies[i] = selected < 0 || selected == i ? (Policy){ .kind = BASESTOCK, .values = { stock } } : sim->stages[i].policy;
773 Simulation candidate;
774 trial(&candidate, policies);
775 Number total = total_cost(&candidate), objective = total;
776 if (selected >= 0) {
777 Costs c = stage_costs(&candidate, selected);
778 objective = c.holding + c.backlog;
779 }
780 if (objective < best_cost) { best = stock; best_cost = objective; best_total = total; }
781 }
782 printf("OPTIMIZE %s BASESTOCK %" PRId64 " cost %s total %s\n", id, best, money(best_cost).text, money(best_total).text);
783 } else fail("unknown command %s", op);
784}
