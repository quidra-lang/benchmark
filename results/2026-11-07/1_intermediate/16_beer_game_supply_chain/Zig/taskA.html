14fn sum(values: []const i64) i64 {
15 var total: i64 = 0;
16 for (values) |value| total += value;
17 return total;
18}
19fn variation(values: []const i64) i64 {
20 var squares: i64 = 0;
21 for (values) |v| squares += v * v;
22 const total = sum(values);
23 return @as(i64, @intCast(values.len)) * squares - total * total;
24}

75 fn init(initial: [4]Initial, demand: [104]i64, horizon: usize) Simulation {
76 var result = Simulation{ .horizon = horizon, .demand = demand, .stages = undefined };
77 for (initial, 0..) |cfg, i| result.stages[i] = .{
78 .inventory = cfg.inventory,
79 .shipping = @splat(cfg.quantity),
80 .orders = @splat(cfg.quantity),
81 .shipping_delay = cfg.shipping_delay,
82 .order_delay = cfg.order_delay,
83 .policy = cfg.policy,
84 .expected = cfg.quantity * 100,
85 .holding_rate = cfg.holding_rate,
86 .backlog_rate = cfg.backlog_rate,

88 return result;

90 fn supply(self: *const Simulation, i: usize) i64 {
91 const s = &self.stages[i];
92 const upstream = if (i == 3) self.plant_backlog else self.stages[i + 1].backlog;
93 return sum(s.orders[0..s.order_delay]) + upstream + sum(s.shipping[0..s.shipping_delay]);
94 }
95 fn signal(self: *const Simulation, stage: usize, week: usize) i64 {
96 return if (self.visible) self.demand[week] else self.stages[stage].records[week].incoming;
97 }
98 fn pop(pipeline: []i64) i64 {
99 const first = pipeline[0];
100 for (0..pipeline.len - 1) |i| pipeline[i] = pipeline[i + 1];
101 pipeline[pipeline.len - 1] = 0;
102 return first;
103 }
104 fn step(self: *Simulation) void {
105 var rows: [4]Record = @splat(.{});
106 for (&self.stages, 0..) |*s, i| {
107 rows[i].received = pop(s.shipping[0..s.shipping_delay]);
108 s.inventory += rows[i].received;
109 }
110 rows[0].incoming = self.demand[self.week];
111 var plant = PlantRecord{};
112 for (&self.stages, 0..) |*s, i| {
113 const incoming = pop(s.orders[0..s.order_delay]);
114 if (i == 3) plant.incoming = incoming else rows[i + 1].incoming = incoming;
115 }
116 for (&self.stages, 0..) |*s, i| {
117 const r = &rows[i];
118 r.before = s.backlog;
119 const requested = s.backlog + r.incoming;
120 r.shipped = @min(s.inventory, requested, s.capacity orelse std.math.maxInt(i64));
121 s.inventory -= r.shipped;
122 s.backlog = requested - r.shipped;
123 r.ontime = @min(@max(0, r.shipped - r.before), r.incoming);
124 r.inventory = s.inventory;
125 r.backlog = s.backlog;
126 if (i > 0) {
127 const downstream = &self.stages[i - 1];
128 downstream.shipping[downstream.shipping_delay - 1] = r.shipped;
129 }
130 }
131 plant.produced = @min(self.plant_backlog + plant.incoming, self.plant_capacity orelse std.math.maxInt(i64));
132 self.plant_backlog += plant.incoming - plant.produced;
133 plant.backlog = self.plant_backlog;
134 self.stages[3].shipping[self.stages[3].shipping_delay - 1] = plant.produced;
135 var supplies: [4]i64 = undefined;
136 for (0..4) |i| supplies[i] = self.supply(i);
137 for (&self.stages, 0..) |*s, i| {
138 const r = &rows[i];
139 const position = s.inventory + supplies[i] - s.backlog;
140 const demand = if (self.visible) self.demand[self.week] else r.incoming;
141 const p = s.policy.values;
142 const quantity = switch (s.policy.kind) {
143 .FIXED => p[0],
144 .PASS => r.incoming,
145 .BASESTOCK => @max(0, p[0] - position),
146 .MINMAX => if (position <= p[0]) p[1] - position else 0,
147 .MOVAVG => blk: {
148 const m = @min(@as(usize, @intCast(p[0])), self.week + 1);
149 var signals = demand;
150 for (self.week + 1 - m..self.week) |u| signals += self.signal(i, u);
151 const forecast = @divFloor(signals, @as(i64, @intCast(m)));
152 break :blk @max(0, p[1] + forecast * @as(i64, @intCast(s.shipping_delay + s.order_delay)) - position);
153 },
154 .SMOOTH => @max(0, demand + @divFloor(p[1] - position, p[0])),
155 .STERMAN => blk: {
156 s.expected = @divFloor(100 * p[0] * demand + (100 - p[0]) * s.expected, 100);
157 const net = s.inventory - s.backlog;
158 break :blk @max(0, @divFloor(100 * s.expected + p[1] * (100 * (p[3] - net) - p[2] * supplies[i]), 10000));
159 },
160 };
161 r.order = @divTrunc(quantity + s.lot - 1, s.lot) * s.lot;
162 s.orders[s.order_delay - 1] = r.order;
163 r.holding_cost = s.holding_rate * s.inventory;
164 r.backlog_cost = s.backlog_rate * s.backlog;
165 s.records[self.week] = r.*;
166 }
167 self.plant_records[self.week] = plant;
168 self.week += 1;
169 }
170 fn costs(self: *const Simulation) [4]i64 {
171 var result: [4]i64 = @splat(0);
172 for (self.stages, 0..) |s, i| for (s.records[0..self.week]) |r| {
173 result[i] += r.cost();
174 };
175 return result;
176 }
177 fn values(self: *const Simulation, stage: usize, metric: Metric) [104]i64 {
178 var result: [104]i64 = @splat(0);
179 for (self.stages[stage].records[0..self.week], 0..) |r, i| result[i] = r.value(metric);
180 return result;
181 }

183const Statistics = struct {
184 total: i64 = 0,
185 minimum: i64 = std.math.maxInt(i64),
186 maximum: i64 = std.math.minInt(i64),
187 fn of(values: []const i64) Statistics {
188 var s: Statistics = .{};
189 for (values) |v| {

211 fn decimal(self: *Game, comptime places: usize, numerator: i64, denominator: i64) Allocator.Error![]const u8 {
212 if (denominator == 0) return "n/a";
213 const scale: i64 = if (places == 2) 100 else 10000;
214 const scaled: u64 = @intCast(@divTrunc(numerator * scale, denominator));
215 const unit: u64 = @intCast(scale);
216 return if (places == 2) std.fmt.allocPrint(self.gpa, "{d}.{d:0>2}", .{ scaled / unit, scaled % unit }) else std.fmt.allocPrint(self.gpa, "{d}.{d:0>4}", .{ scaled / unit, scaled % unit });
217 }
218 fn money(self: *Game, cents: i64) Allocator.Error![]const u8 {
219 return self.decimal(2, cents, 100);
220 }
221 fn percentage(self: *Game, numerator: i64, denominator: i64) Allocator.Error![]const u8 {
222 if (denominator == 0) return "n/a";
223 return std.fmt.allocPrint(self.gpa, "{s}%", .{try self.decimal(2, numerator * 100, denominator)});
224 }

244 fn range(self: *Game, args: []const []const u8, limit: usize) Error!Range {
245 if (args.len == 0) return .{ .first = 0, .end = limit };
246 const first: usize = @intCast(integer(args[0]));
247 const last: usize = @intCast(integer(args[1]));
248 if (first == 0 or first > last or last > limit) return self.fail("bad range {s} {s}", .{ args[0], args[1] });
249 return .{ .first = first - 1, .end = last };
250 }
251 fn policy(self: *Game, args: []const []const u8) Error!Policy {
252 const kind = std.meta.stringToEnum(PolicyKind, args[0]) orelse return self.fail("unknown policy {s}", .{args[0]});
253 const count: usize = switch (kind) {
254 .PASS => 0,
255 .FIXED, .BASESTOCK => 1,
256 .MINMAX, .MOVAVG, .SMOOTH => 2,
257 .STERMAN => 4,
258 };
259 if (args.len != count + 1) return self.fail("{s} needs {d} parameters", .{ args[0], count });
260 var result = Policy{ .kind = kind, .text = try std.mem.join(self.gpa, " ", args) };
261 for (args[1..], 0..) |arg, i| result.values[i] = if (kind == .STERMAN and i < 3) integer(arg[0..1]) * 100 + integer(arg[2..]) else integer(arg);
262 const p = result.values;
263 const valid = switch (kind) {
264 .PASS => true,
265 .FIXED => p[0] <= 1000,
266 .BASESTOCK => p[0] <= 10000,
267 .MINMAX => p[0] < p[1] and p[1] <= 10000,
268 .MOVAVG => p[0] >= 1 and p[0] <= 52 and p[1] <= 10000,
269 .SMOOTH => p[0] >= 1 and p[0] <= 10 and p[1] <= 10000,
270 .STERMAN => p[0] <= 100 and p[1] <= 100 and p[2] <= 100 and p[3] <= 10000,
271 };
272 if (!valid) return self.fail("invalid parameter for {s}", .{args[0]});
273 return result;
274 }
275 fn reset(self: *Game) void {

277 }
278 fn setDemand(self: *Game, demand: [104]i64, horizon: usize) Error!void {
279 self.base = demand;
280 self.base_horizon = horizon;
281 self.reset();
282 const stats = Statistics.of(demand[0..horizon]);
283 try self.out.print("DEMAND weeks {d} total {d} min {d} max {d}\n", .{ horizon, stats.total, stats.minimum, stats.maximum });
284 }
285 fn trial(self: *Game, policies: [4]Policy) Simulation {
286 var result = Simulation.init(self.initial, self.sim.demand, self.sim.horizon);
287 result.visible = self.sim.visible;
288 result.plant_capacity = self.sim.plant_capacity;
289 for (&result.stages, 0..) |*s, i| {
290 const current = &self.sim.stages[i];
291 s.policy = policies[i];
292 s.lot = current.lot;
293 s.capacity = current.capacity;
294 s.holding_rate = current.holding_rate;
295 s.backlog_rate = current.backlog_rate;
296 }
297 while (result.week < result.horizon) result.step();
298 return result;
299 }
300 fn pipeline(self: *Game, values: []const i64) Error!void {
301 for (values, 0..) |value, i| try self.out.print("{s}{d}", .{ if (i == 0) "" else ",", value });
302 }
303 fn execute(self: *Game, a: []const []const u8) Error!void {
304 const op = std.meta.stringToEnum(Command, a[0]) orelse return self.fail("unknown command {s}", .{a[0]});
305 const id: []const u8 = if (a.len > 1) a[1] else "";
306 const sim = &self.sim;
307 const n: i64 = @intCast(sim.week);
308 switch (op) {
309 .DEMAND => {
310 var demand: [104]i64 = @splat(0);
311 for (a[1..], 0..) |value, i| demand[i] = integer(value);
312 try self.setDemand(demand, a.len - 1);
313 },
314 .GENERATE => {
315 const horizon: usize = @intCast(integer(id));
316 var i: usize = 5;
317 while (i < a.len) : (i += 3) {
318 const week = integer(a[i + 1]);
319 if (week < 1 or week > @as(i64, @intCast(horizon))) return self.fail("week {s} outside horizon", .{a[i + 1]});
320 }
321 var demand: [104]i64 = @splat(0);
322 var random = integer(a[4]);
323 for (0..horizon) |t| {
324 var level = integer(a[2]);
325 i = 5;
326 while (i < a.len) : (i += 3) {
327 const elapsed = @as(i64, @intCast(t + 1)) - integer(a[i + 1]);
328 if (elapsed < 0) continue;
329 const value = integer(a[i + 2]);
330 if (equal(a[i], "STEP")) level += value else if (equal(a[i], "RAMP")) level += value * (elapsed + 1) else if (elapsed == 0) {
331 level += value;
332 }
333 }
334 if (level < 0) return self.fail("negative level at week {d}", .{t + 1});
335 random = @mod(48271 * random, 2147483647);
336 demand[t] = level + @mod(random, integer(a[3]));
337 }
338 try self.setDemand(demand, horizon);
339 },
340 .RUN => {
341 if (sim.week == sim.horizon) return self.fail("horizon reached at week {d}", .{sim.horizon});
342 const first = sim.week;
343 const end = @min(sim.horizon, sim.week + @as(usize, @intCast(integer(id))));
344 var cost: i64 = 0;
345 while (sim.week < end) {
346 sim.step();
347 for (&sim.stages, 0..) |*s, i| {
348 const r = s.records[sim.week - 1];
349 cost += r.cost();
350 if (r.before == 0 and r.backlog > 0) try self.out.print("STOCKOUT {s} week {d} backlog {d}\n", .{ names[i], sim.week, r.backlog });
351 if (r.before > 0 and r.backlog == 0) try self.out.print("RECOVER {s} week {d}\n", .{ names[i], sim.week });
352 }
353 }
354 try self.out.print("RUN {d}-{d} demand {d} cost {s}\n", .{ first + 1, sim.week, sum(sim.demand[first..sim.week]), try self.money(cost) });
355 if (sim.week == sim.horizon) try self.out.print("HORIZON {d}\n", .{sim.horizon});
356 },
357 .RESET => {
358 self.reset();
359 try self.out.writeAll("RESET week 0\n");
360 },
361 .SAVE => {
362 const snapshot = try self.gpa.create(Simulation);
363 snapshot.* = sim.*;
364 try self.snapshots.put(self.gpa, id, snapshot);
365 try self.out.print("SAVE {s} week {d}\n", .{ id, sim.week });
366 },
367 .LOAD => {
368 const snapshot = self.snapshots.get(id) orelse return self.fail("no snapshot {s}", .{id});
369 sim.* = snapshot.*;
370 try self.out.print("LOAD {s} week {d}\n", .{ id, sim.week });
371 },
372 .SET => {
373 const selected = try self.target(id);
374 const p = try self.policy(a[2..]);
375 for (&sim.stages, 0..) |*s, i| {
376 if (selected != null and selected.? != i) continue;
377 s.policy = p;
378 if (p.kind == .STERMAN) s.expected = 100 * (if (sim.week == 0) self.initial[i].quantity else sim.signal(i, sim.week - 1));
379 }
380 try self.out.print("SET {s} {s}\n", .{ id, p.text });
381 },
382 .BATCH, .CAPACITY, .COST => {
383 const index = if (op == .CAPACITY and equal(id, "PLANT")) 4 else try self.stage(id);
384 if (op == .BATCH) {
385 const lot = integer(a[2]);
386 if (lot < 1 or lot > 100) return self.fail("invalid batch size {s}", .{a[2]});
387 sim.stages[index].lot = lot;
388 try self.out.print("BATCH {s} {d}\n", .{ id, lot });
389 } else if (op == .CAPACITY) {
390 const capacity: ?i64 = if (equal(a[2], "OFF")) null else integer(a[2]);
391 if (capacity != null and capacity.? > 10000) return self.fail("invalid capacity {s}", .{a[2]});
392 if (index == 4) sim.plant_capacity = capacity else sim.stages[index].capacity = capacity;
393 try self.out.print("CAPACITY {s} {s}\n", .{ id, a[2] });
394 } else {
395 const holding = integer(a[2]);
396 const backlog = integer(a[3]);
397 if (holding > 1000 or backlog > 1000) return self.fail("invalid cost", .{});
398 sim.stages[index].holding_rate = holding;
399 sim.stages[index].backlog_rate = backlog;
400 try self.out.print("COST {s} holding {d} backlog {d}\n", .{ id, holding, backlog });
401 }
402 },
403 .VISIBILITY => {
404 if (!equal(id, "ON") and !equal(id, "OFF")) return self.fail("unknown visibility {s}", .{id});
405 sim.visible = equal(id, "ON");
406 try self.out.print("VISIBILITY {s}\n", .{id});
407 },
408 .SHOCK => {
409 const week: usize = @intCast(integer(id));
410 if (week == 0 or week > sim.horizon) return self.fail("week {s} outside horizon", .{id});
411 if (week <= sim.week) return self.fail("week {s} already simulated", .{id});
412 const old = sim.demand[week - 1];
413 sim.demand[week - 1] = integer(a[2]);
414 try self.out.print("SHOCK week {d} {d} -> {d}\n", .{ week, old, sim.demand[week - 1] });
415 },
416 .STATUS => {
417 try self.out.print("STATUS week {d} of {d}\n", .{ sim.week, sim.horizon });
418 for (&sim.stages, 0..) |*s, i| {
419 try self.out.print("{s} inv {d} back {d} transit ", .{ names[i], s.inventory, s.backlog });
420 try self.pipeline(s.shipping[0..s.shipping_delay]);
421 try self.out.writeAll(" pending ");
422 try self.pipeline(s.orders[0..s.order_delay]);
423 const supply = sim.supply(i);
424 try self.out.print(" supply {d} position {d} policy {s}", .{ supply, s.inventory + supply - s.backlog, s.policy.text });
425 if (s.policy.kind == .STERMAN) try self.out.print(" expect {s}", .{try self.money(s.expected)});
426 try self.out.writeAll("\n");
427 }
428 try self.out.print("PLANT back {d}\n", .{sim.plant_backlog});
429 },
430 .SETTINGS => {
431 try self.out.writeAll("SETTINGS\n");
432 for (&sim.stages, 0..) |*s, i| try self.out.print("{s} holding {d} backlog {d} batch {d} capacity {s} delays {d} {d}\n", .{ names[i], s.holding_rate, s.backlog_rate, s.lot, try self.capacityText(s.capacity), s.shipping_delay, s.order_delay });
433 try self.out.print("PLANT capacity {s}\nVISIBILITY {s}\n", .{ try self.capacityText(sim.plant_capacity), if (sim.visible) "ON" else "OFF" });
434 },
435 .WEEK => {
436 try self.requireWeeks();
437 const week: usize = @intCast(integer(id));
438 if (week == 0 or week > sim.week) return self.fail("bad week {s}", .{id});
439 try self.out.print("WEEK {d} demand {d}\n", .{ week, sim.demand[week - 1] });
440 for (&sim.stages, 0..) |*s, i| {
441 const r = s.records[week - 1];
442 try self.out.print("{s} recv {d} in {d} ship {d} inv {d} back {d} order {d} cost {s}\n", .{ names[i], r.received, r.incoming, r.shipped, r.inventory, r.backlog, r.order, try self.money(r.cost()) });
443 }
444 const p = sim.plant_records[week - 1];
445 try self.out.print("PLANT in {d} made {d} back {d}\n", .{ p.incoming, p.produced, p.backlog });
446 },
447 .TABLE => {
448 const i = try self.stage(id);
449 try self.requireWeeks();
450 const r = try self.range(a[2..], sim.week);
451 try self.out.print("TABLE {s} {d}-{d}\nweek in ship recv inv back order cost\n", .{ id, r.first + 1, r.end });
452 for (r.first..r.end) |t| {
453 const row = sim.stages[i].records[t];
454 try self.out.print("{d: >4} {d: >6} {d: >6} {d: >6} {d: >6} {d: >6} {d: >6} {s: >9}\n", .{ t + 1, row.incoming, row.shipped, row.received, row.inventory, row.backlog, row.order, try self.money(row.cost()) });
455 }
456 },
457 .CHAIN => {
458 try self.requireWeeks();
459 const r = try self.range(a[1..], sim.week);
460 try self.out.print("CHAIN {d}-{d}\n{s: >4} {s: >6}", .{ r.first + 1, r.end, "week", "demand" });
461 for (names) |name| try self.out.print(" {s: >11}", .{name});
462 try self.out.writeAll("\n");
463 for (r.first..r.end) |t| {

740fn readLine(gpa: Allocator, input: *std.Io.Reader) !?[]const u8 {
741 var line: std.Io.Writer.Allocating = .init(gpa);
742 defer line.deinit();
743 _ = try input.streamDelimiterEnding(&line.writer, '\n');
744 if (input.seek < input.end) input.toss(1) else if (line.written().len == 0) return null;
745 return try line.toOwnedSlice();
746}
747fn tokenize(gpa: Allocator, line: []const u8) Allocator.Error![]const []const u8 {
748 var tokens: std.ArrayList([]const u8) = .empty;
749 var it = std.mem.splitScalar(u8, line, ' ');
750 while (it.next()) |token| try tokens.append(gpa, token);
751 return tokens.items;
752}
753pub fn main(init: std.process.Init) !void {
754 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
755 defer arena.deinit();
756 const gpa = arena.allocator();
757 var input_buffer: [4096]u8 = undefined;
758 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
759 var output_buffer: [8192]u8 = undefined;
760 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
761 var game = Game{ .gpa = gpa, .out = &output.interface };
762 for (0..4) |i| {
763 const a = try tokenize(gpa, (try readLine(gpa, &input.interface)).?);
764 game.initial[i] = .{ .inventory = integer(a[1]), .quantity = integer(a[2]), .shipping_delay = @intCast(integer(a[3])), .order_delay = @intCast(integer(a[4])), .holding_rate = integer(a[5]), .backlog_rate = integer(a[6]), .policy = try game.policy(a[7..]) };
765 }
766 game.reset();
767 while (try readLine(gpa, &input.interface)) |line| {
768 const a = try tokenize(gpa, line);
769 if (equal(a[0], "END")) break;
770 game.execute(a) catch |err| {
771 if (err != error.BadCommand) return err;
772 try output.interface.print("ERROR {s}\n", .{game.error_message});
773 };
774 }
775 try output.interface.flush();
776}
