3type Policy = {name: string; args: bigint[]; text: string};
4type Initial = {inventory: bigint; quantity: bigint; shipping: number; orders: number; holding: bigint; backlog: bigint; policy: Policy};
5type RecordRow = Record<string, bigint>;
6type Stage = {inventory: bigint; backlog: bigint; shipping: bigint[]; orders: bigint[]; policy: Policy; expected: bigint; holding: bigint; backlogRate: bigint; lot: bigint; capacity: bigint | null; records: RecordRow[]};
7type State = {week: number; demands: bigint[]; stages: Stage[]; plantBacklog: bigint; plantCapacity: bigint | null; plantRecords: RecordRow[]; visibility: string};
8const NAMES = ["RETAILER", "WHOLESALER", "DISTRIBUTOR", "FACTORY"];
9const PARAMETERS: Record<string, number> = {FIXED: 1, PASS: 0, BASESTOCK: 1, MINMAX: 2, MOVAVG: 2, SMOOTH: 2, STERMAN: 4};
10const MODES: Record<string, string> = {ORDERS: "order", INV: "inventory", BACK: "backlog"};
11const sum = (v: bigint[]): bigint => v.reduce((a, b) => a + b, 0n);
12const min = (...v: bigint[]): bigint => v.reduce((a, b) => a < b ? a : b);
13const max = (...v: bigint[]): bigint => v.reduce((a, b) => a > b ? a : b);
14const abs = (v: bigint): bigint => v < 0n ? -v : v;
15const compare = (a: bigint, b: bigint): number => a < b ? -1 : a > b ? 1 : 0;
16const floorDiv = (n: bigint, d: bigint): bigint => n < 0n ? -((-n + d - 1n) / d) : n / d;
17const money = (v: bigint): string => v / 100n + "." + String(v % 100n).padStart(2, "0");
18const variation = (v: bigint[]): bigint => BigInt(v.length) * sum(v.map(x => x * x)) - sum(v) ** 2n;
19function decimal(p: bigint, q: number | bigint, places: number): string {
20 const scale = 10n ** BigInt(places);
21 const value = p * scale / BigInt(q);
22 return value / scale + "." + String(value % scale).padStart(places, "0");
23}

26function parsePolicy(tokens: string[]): Policy {
27 const name = tokens[0];
28 if (!Object.hasOwn(PARAMETERS, name)) throw new CommandError("unknown policy " + name);
29 const count = PARAMETERS[name];
30 if (tokens.length - 1 !== count) throw new CommandError(name + " needs " + count + " parameters");
31 const a = tokens.slice(1).map((t, i) => BigInt(name === "STERMAN" && i < 3 ? t.replace(".", "") : t));
32 let valid = true;
33 if (name === "FIXED") valid = a[0] >= 0n && a[0] <= 1000n;
34 else if (name === "BASESTOCK") valid = a[0] >= 0n && a[0] <= 10000n;
35 else if (name === "MINMAX") valid = a[0] >= 0n && a[0] < a[1] && a[1] <= 10000n;
36 else if (name === "MOVAVG") valid = a[0] >= 1n && a[0] <= 52n && a[1] >= 0n && a[1] <= 10000n;
37 else if (name === "SMOOTH") valid = a[0] >= 1n && a[0] <= 10n && a[1] >= 0n && a[1] <= 10000n;
38 else if (name === "STERMAN") valid = a.slice(0, 3).every(v => v >= 0n && v <= 100n) && a[3] >= 0n && a[3] <= 10000n;
39 if (!valid) throw new CommandError("invalid parameter for " + name);
40 return {name, args: a, text: tokens.join(" ")};
41}

43class Simulation {
44 initials: Initial[];
45 week = 0;
46 demands: bigint[];
47 stages: Stage[];
48 plantBacklog = 0n;
49 plantCapacity: bigint | null = null;
50 plantRecords: RecordRow[] = [];
51 visibility = "OFF";
52
53 constructor(initials: Initial[], demands: bigint[]) {
54 this.initials = initials;
55 this.demands = [...demands];
56 this.stages = initials.map(i => ({inventory: i.inventory, backlog: 0n,
57 shipping: Array(i.shipping).fill(i.quantity), orders: Array(i.orders).fill(i.quantity),
58 policy: i.policy, expected: 100n * i.quantity, holding: i.holding, backlogRate: i.backlog,
59 lot: 1n, capacity: null, records: []}));
60 }
61
62 snapshot(): State {
63 return structuredClone({week: this.week, demands: this.demands, stages: this.stages,
64 plantBacklog: this.plantBacklog, plantCapacity: this.plantCapacity, plantRecords: this.plantRecords, visibility: this.visibility});
65 }
66
67 static restore(initials: Initial[], state: State): Simulation {
68 return Object.assign(new Simulation(initials, []), structuredClone(state));
69 }
70
71 signal(index: number, week: number, current: bigint[] | null = null): bigint {
72 if (this.visibility === "ON") return this.demands[week - 1];
73 if (current !== null && week === this.week + 1) return current[index];
74 return this.stages[index].records[week - 1].incoming;
75 }
76
77 setPolicy(index: number, policy: Policy): void {
78 const s = this.stages[index];
79 s.policy = policy;
80 if (policy.name === "STERMAN") s.expected = 100n * (this.week ? this.signal(index, this.week) : this.initials[index].quantity);
81 }
82
83 positions(): {supply: bigint[]; positions: bigint[]} {
84 const supply = this.stages.map((s, i) => sum(s.orders) + (i < 3 ? this.stages[i + 1].backlog : this.plantBacklog) + sum(s.shipping));
85 return {supply, positions: this.stages.map((s, i) => s.inventory + supply[i] - s.backlog)};
86 }
87
88 policyOrder(index: number, incoming: bigint[], line: bigint, position: bigint): bigint {
89 const s = this.stages[index];
90 const {name, args: a} = s.policy;
91 const week = this.week + 1;
92 const signal = this.signal(index, week, incoming);
93 if (name === "FIXED") return a[0];
94 if (name === "PASS") return incoming[index];
95 if (name === "BASESTOCK") return max(0n, a[0] - position);
96 if (name === "MINMAX") return position <= a[0] ? a[1] - position : 0n;
97 if (name === "MOVAVG") {
98 const count = Math.min(Number(a[0]), week);
99 const signals = Array.from({length: count}, (_, j) => this.signal(index, week - count + j + 1, incoming));
100 const forecast = sum(signals) / BigInt(count);
101 const delay = this.initials[index].shipping + this.initials[index].orders;
102 return max(0n, a[1] + forecast * BigInt(delay) - position);
103 }
104 if (name === "SMOOTH") return max(0n, signal + floorDiv(a[1] - position, a[0]));
105 const [theta, alpha, beta, target] = a;
106 s.expected = (100n * theta * signal + (100n - theta) * s.expected) / 100n;
107 const net = s.inventory - s.backlog;
108 return max(0n, floorDiv(100n * s.expected + alpha * (100n * (target - net) - beta * line), 10000n));
109 }
110
111 step(): RecordRow[] {
112 const received = this.stages.map(s => {
113 const q = s.shipping.shift()!;
114 s.inventory += q;
115 return q;
116 });
117 const incoming = [this.demands[this.week], ...this.stages.slice(0, 3).map(s => s.orders.shift()!)];
118 const plantIncoming = this.stages[3].orders.shift()!;
119 const records: RecordRow[] = [];
120 this.stages.forEach((s, i) => {
121 const before = s.backlog;
122 const requested = before + incoming[i];
123 const shipped = min(s.inventory, requested, s.capacity ?? requested);
124 s.inventory -= shipped;
125 s.backlog = requested - shipped;
126 if (i > 0) this.stages[i - 1].shipping.push(shipped);
127 records.push({received: received[i], incoming: incoming[i], shipped, before,
128 ontime: min(max(0n, shipped - before), incoming[i]), inventory: s.inventory, backlog: s.backlog});
129 });
130 const requested = this.plantBacklog + plantIncoming;
131 const produced = min(requested, this.plantCapacity ?? requested);
132 this.plantBacklog = requested - produced;
133 this.stages[3].shipping.push(produced);
134 const {supply, positions} = this.positions();
135 const orders = this.stages.map((_, i) => this.policyOrder(i, incoming, supply[i], positions[i]));
136 this.stages.forEach((s, i) => {
137 const quantity = (orders[i] + s.lot - 1n) / s.lot * s.lot;
138 s.orders.push(quantity);
139 const r = records[i];
140 r.order = quantity;
141 r.holding = s.holding * s.inventory;
142 r.backlog_cost = s.backlogRate * s.backlog;
143 r.cost = r.holding + r.backlog_cost;
144 s.records.push(r);
145 });
146 this.plantRecords.push({incoming: plantIncoming, produced, backlog: this.plantBacklog});
147 this.week++;
148 return records;
149 }
150
151 costs(): bigint[] {
152 return this.stages.map(s => sum(s.records.map(r => r.cost)));
153 }
154}

188 if (!a.length) return [1, limit];

207 trial(policies: Policy[]): Simulation {
208 const current = this.sim;
209 const trial = new Simulation(this.initials, current.demands);
210 trial.visibility = current.visibility;
211 trial.plantCapacity = current.plantCapacity;
212 trial.stages.forEach((s, i) => {
213 const old = current.stages[i];
214 s.lot = old.lot;
215 s.capacity = old.capacity;
216 s.holding = old.holding;
217 s.backlogRate = old.backlogRate;
218 trial.setPolicy(i, policies[i]);
219 });
220 while (trial.week < trial.demands.length) trial.step();
221 return trial;
222 }

224 demand(values: bigint[]): void {
225 this.initialize(values);
226 this.emit("DEMAND", "weeks", values.length, "total", sum(values), "min", min(...values), "max", max(...values));
227 }
228
229 generate(a: string[]): void {
230 const weeks = Number(a[0]);
231 const base = BigInt(a[1]);
232 const spread = BigInt(a[2]);
233 let seed = BigInt(a[3]);
234 const changes: {kind: string; week: number; amount: bigint}[] = [];
235 for (let i = 4; i < a.length; i += 3) {
236 const week = Number(a[i + 1]);
237 if (week < 1 || week > weeks) throw new CommandError("week " + a[i + 1] + " outside horizon");
238 changes.push({kind: a[i], week, amount: BigInt(a[i + 2])});
239 }
240 const levels: bigint[] = [];
241 for (let week = 1; week <= weeks; week++) {
242 let level = base;
243 for (const c of changes) {
244 if ((c.kind === "STEP" && week >= c.week) || (c.kind === "PULSE" && week === c.week)) level += c.amount;
245 else if (c.kind === "RAMP" && week >= c.week) level += c.amount * BigInt(week - c.week + 1);
246 }
247 if (level < 0n) throw new CommandError("negative level at week " + week);
248 levels.push(level);
249 }
250 const values = levels.map(level => {
251 seed = 48271n * seed % 2147483647n;
252 return level + seed % spread;
253 });
254 this.demand(values);
255 }
256
257 status(): void {
258 const s = this.sim;
259 this.emit("STATUS", "week", s.week, "of", s.demands.length);
260 const {supply, positions} = s.positions();
261 s.stages.forEach((stage, i) => {
262 this.emit(NAMES[i], "inv", stage.inventory, "back", stage.backlog,
263 "transit", stage.shipping.join(","), "pending", stage.orders.join(","), "supply", supply[i],
264 "position", positions[i], "policy", stage.policy.text, ...(stage.policy.name === "STERMAN" ? ["expect", money(stage.expected)] : []));
265 });
266 this.emit("PLANT", "back", s.plantBacklog);
267 }

269 fill(records: RecordRow[]): {ordered: bigint; ontime: bigint; text: string} {
270 const ordered = sum(records.map(r => r.incoming));
271 const ontime = sum(records.map(r => r.ontime));
272 return {ordered, ontime, text: ordered === 0n ? "n/a" : decimal(ontime * 100n, ordered, 2) + "%"};
273 }
274
275 peaks(): void {
276 const s = this.sim;
277 this.emit("PEAKS", "weeks", "1-" + s.week);
278 s.stages.forEach((stage, i) => {
279 const inventories = stage.records.map(r => r.inventory);
280 const backlogs = stage.records.map(r => r.backlog);
281 const mi = max(...inventories);
282 const mb = max(...backlogs);
283 let run = 0;
284 let longest = 0;
285 let start = 0;
286 let bestStart = 0;
287 let bestEnd = 0;
288 backlogs.forEach((backlog, j) => {
289 if (backlog > 0n) {
290 if (run === 0) start = j + 1;
291 run++;
292 if (run > longest) {
293 longest = run;
294 bestStart = start;
295 bestEnd = j + 1;
296 }
297 } else run = 0;
298 });
299 this.emit(NAMES[i], "max_inv", mi, "week", mi ? inventories.indexOf(mi) + 1 : "-",
300 "max_back", mb, "week", mb ? backlogs.indexOf(mb) + 1 : "-",
301 "longest_stockout", longest ? longest + " weeks " + bestStart + "-" + bestEnd : 0);
302 });
303 }

305 leadtime(index: number): void {
306 const records = this.sim.stages[index].records;
307 const initial = this.initials[index];
308 const nominal = initial.shipping + initial.orders;
309 let received = 0n;
310 const cumulative = records.map(r => {
311 received += r.received;
312 return received - BigInt(nominal) * initial.quantity;
313 });
314 let ordered = 0n;
315 let count = 0;
316 const leads: number[] = [];
317 records.forEach((r, t) => {
318 ordered += r.order;
319 if (r.order === 0n) return;
320 count++;
321 for (let u = t + 1; u < records.length; u++) {
322 if (cumulative[u] >= ordered) {
323 leads.push(u - t);
324 break;
325 }
326 }
327 });
328 this.emit("LEADTIME", NAMES[index], "nominal", nominal, "orders", count, "completed", leads.length,
329 "open", count - leads.length, "min", leads.length ? Math.min(...leads) : "-",
330 "avg", leads.length ? decimal(BigInt(leads.reduce((a, b) => a + b, 0)), leads.length, 2) : "-", "max", leads.length ? Math.max(...leads) : "-");
331 }

333 plot(index: number, mode: string, first: number, last: number): void {
334 const values = this.values(index, mode).slice(first - 1, last);
335 const maximum = max(...values);
336 if (mode === "NET") {
337 this.emit("PLOT", NAMES[index], mode, first + "-" + last, "min", min(...values), "max", maximum);
338 const scale = max(...values.map(abs));
339 values.forEach((value, i) => {
340 const length = scale === 0n ? 0 : Number(abs(value) * 25n / scale);
341 const left = value < 0n ? " ".repeat(25 - length) + "-".repeat(length) : " ".repeat(25);
342 const right = value > 0n ? "+".repeat(length) : "";
343 this.emit(String(first + i).padStart(3, " ") + " " + left + "|" + right + " " + value);
344 });
345 } else {
346 this.emit("PLOT", NAMES[index], mode, first + "-" + last, "max", maximum);
347 values.forEach((value, i) => {
348 const length = maximum === 0n ? 0 : Number(value * 50n / maximum);
349 this.emit(String(first + i).padStart(3, " ") + " |" + "#".repeat(length) + " " + value);
350 });
351 }

354 command(cmd: string, a: string[]): void {
355 const s = this.sim;
356 switch (cmd) {
357 case "DEMAND":
358 this.demand(a.map(BigInt));
359 break;
360 case "GENERATE":
361 this.generate(a);
362 break;
363 case "RUN": {
364 if (s.week === s.demands.length) throw new CommandError("horizon reached at week " + s.demands.length);
365 const first = s.week + 1;
366 const count = Math.min(Number(a[0]), s.demands.length - s.week);
367 let cost = 0n;
368 for (let j = 0; j < count; j++) {
369 s.step().forEach((r, i) => {
370 cost += r.cost;
371 if (r.before === 0n && r.backlog > 0n) this.emit("STOCKOUT", NAMES[i], "week", s.week, "backlog", r.backlog);
372 else if (r.before > 0n && r.backlog === 0n) this.emit("RECOVER", NAMES[i], "week", s.week);
373 });
374 }
375 this.emit("RUN", first + "-" + s.week, "demand", sum(s.demands.slice(first - 1, s.week)), "cost", money(cost));
376 if (s.week === s.demands.length) this.emit("HORIZON", s.week);
377 break;
378 }
379 case "RESET":
380 this.initialize();
381 this.emit("RESET", "week", 0);
382 break;
383 case "SAVE":
384 this.snapshots.set(a[0], s.snapshot());
385 this.emit("SAVE", a[0], "week", s.week);
386 break;
387 case "LOAD": {
388 const state = this.snapshots.get(a[0]);
389 if (!state) throw new CommandError("no snapshot " + a[0]);
390 this.sim = Simulation.restore(this.initials, state);
391 this.emit("LOAD", a[0], "week", this.sim.week);
392 break;
393 }
394 case "SET": {
395 const indices = this.target(a[0]);
396 const policy = parsePolicy(a.slice(1));
397 for (const i of indices) s.setPolicy(i, policy);
398 this.emit("SET", a[0], policy.text);
399 break;
400 }
401 case "BATCH": {
402 const i = this.stageIndex(a[0]);
403 const lot = BigInt(a[1]);
404 if (lot < 1n || lot > 100n) throw new CommandError("invalid batch size " + a[1]);
405 s.stages[i].lot = lot;
406 this.emit("BATCH", a[0], lot);
407 break;
408 }
409 case "CAPACITY": {
410 const i = a[0] === "PLANT" ? null : this.stageIndex(a[0]);
411 const capacity = a[1] === "OFF" ? null : BigInt(a[1]);
412 if (capacity !== null && capacity > 10000n) throw new CommandError("invalid capacity " + a[1]);
413 if (i === null) s.plantCapacity = capacity;
414 else s.stages[i].capacity = capacity;
415 this.emit("CAPACITY", a[0], a[1]);
416 break;
417 }
418 case "COST": {
419 const i = this.stageIndex(a[0]);
420 const holding = BigInt(a[1]);
421 const backlog = BigInt(a[2]);
422 if (holding > 1000n || backlog > 1000n) throw new CommandError("invalid cost");
423 s.stages[i].holding = holding;
424 s.stages[i].backlogRate = backlog;
425 this.emit("COST", a[0], "holding", holding, "backlog", backlog);
426 break;
427 }
428 case "VISIBILITY":
429 if (!["ON", "OFF"].includes(a[0])) throw new CommandError("unknown visibility " + a[0]);
430 s.visibility = a[0];
431 this.emit("VISIBILITY", a[0]);
432 break;
433 case "SHOCK": {
434 const week = Number(a[0]);
435 if (week < 1 || week > s.demands.length) throw new CommandError("week " + a[0] + " outside horizon");
436 if (week <= s.week) throw new CommandError("week " + a[0] + " already simulated");
437 const old = s.demands[week - 1];
438 s.demands[week - 1] = BigInt(a[1]);
439 this.emit("SHOCK", "week", week, old, "->", BigInt(a[1]));
440 break;
441 }
442 case "STATUS":
443 this.status();
444 break;
445 case "SETTINGS":
446 this.emit("SETTINGS");
447 s.stages.forEach((stage, i) => this.emit(NAMES[i], "holding", stage.holding, "backlog", stage.backlogRate,
448 "batch", stage.lot, "capacity", stage.capacity === null ? "OFF" : stage.capacity, "delays", this.initials[i].shipping, this.initials[i].orders));
449 this.emit("PLANT", "capacity", s.plantCapacity === null ? "OFF" : s.plantCapacity);
450 this.emit("VISIBILITY", s.visibility);
451 break;
452 case "WEEK": {
453 this.requireWeeks();
454 const week = Number(a[0]);
455 if (week < 1 || week > s.week) throw new CommandError("bad week " + a[0]);
456 this.emit("WEEK", week, "demand", s.demands[week - 1]);
457 s.stages.forEach((stage, i) => {
458 const r = stage.records[week - 1];
459 this.emit(NAMES[i], "recv", r.received, "in", r.incoming, "ship", r.shipped, "inv", r.inventory, "back", r.backlog, "order", r.order, "cost", money(r.cost));
460 });
461 const p = s.plantRecords[week - 1];
462 this.emit("PLANT", "in", p.incoming, "made", p.produced, "back", p.backlog);
463 break;
464 }
465 case "TABLE": {
466 const i = this.stageIndex(a[0]);
467 this.requireWeeks();
468 const [first, last] = this.range(a.slice(1), s.week);
469 this.emit("TABLE", a[0], first + "-" + last);
470 const widths = [4, 6, 6, 6, 6, 6, 6, 9];
471 this.emit(["week", "in", "ship", "recv", "inv", "back", "order", "cost"].map((v, j) => v.padStart(widths[j], " ")).join(" "));
472 for (let week = first; week <= last; week++) {
473 const r = s.stages[i].records[week - 1];
474 const values = [week, r.incoming, r.shipped, r.received, r.inventory, r.backlog, r.order, money(r.cost)];
475 this.emit(values.map((v, j) => String(v).padStart(widths[j], " ")).join(" "));
476 }
477 break;
478 }
479 case "CHAIN": {
480 this.requireWeeks();
481 const [first, last] = this.range(a, s.week);
482 this.emit("CHAIN", first + "-" + last);
483 this.emit("week".padStart(4, " ") + " " + "demand".padStart(6, " ") + " " + NAMES.map(n => n.padStart(11, " ")).join(" "));
484 for (let week = first; week <= last; week++) this.emit(String(week).padStart(4, " ") + " " +
485 String(s.demands[week - 1]).padStart(6, " ") + " " + s.stages.map(stage => String(stage.records[week - 1].order).padStart(11, " ")).join(" "));
486 break;
487 }
488 case "DEMANDS": {
489 const [first, last] = this.range(a, s.demands.length);
490 const values = s.demands.slice(first - 1, last);
491 this.emit("DEMANDS", first + "-" + last, "total", sum(values), "min", min(...values), "max", max(...values));
492 this.emit(...values);
493 break;
494 }
495 case "COSTS": {
496 this.emit("COSTS", "week", s.week);
497 let totalHolding = 0n;
498 let totalBacklog = 0n;
499 s.stages.forEach((stage, i) => {
500 const holding = sum(stage.records.map(r => r.holding));
501 const backlog = sum(stage.records.map(r => r.backlog_cost));
502 totalHolding += holding;
503 totalBacklog += backlog;
504 this.emit(NAMES[i], "holding", money(holding), "backlog", money(backlog), "total", money(holding + backlog));
505 });
506 this.emit("TOTAL", "holding", money(totalHolding), "backlog", money(totalBacklog), "total", money(totalHolding + totalBacklog));
507 break;
508 }
509 case "SUMMARY":
510 this.requireWeeks();
511 this.emit("SUMMARY", "weeks", "1-" + s.week);
512 s.stages.forEach((stage, i) => {
513 const orders = stage.records.map(r => r.order);
514 this.emit(NAMES[i], "avg_inv", decimal(sum(stage.records.map(r => r.inventory)), s.week, 2),
515 "avg_back", decimal(sum(stage.records.map(r => r.backlog)), s.week, 2), "orders", sum(orders),
516 "min", min(...orders), "max", max(...orders), "zero_weeks", orders.filter(v => v === 0n).length);
517 });
518 break;
519 case "BALANCE":
520 this.emit("BALANCE", "week", s.week);
521 s.stages.forEach((stage, i) => this.emit(NAMES[i], "start", this.initials[i].inventory,
522 "received", sum(stage.records.map(r => r.received)), "shipped", sum(stage.records.map(r => r.shipped)), "end", stage.inventory, "owed", stage.backlog));
523 this.emit("SYSTEM", "initial", sum(this.initials.map(i => i.inventory + i.quantity * BigInt(i.shipping))),
524 "produced", sum(s.plantRecords.map(r => r.produced)), "delivered", sum(s.stages[0].records.map(r => r.shipped)),
525 "stock", sum(s.stages.map(stage => stage.inventory)), "transit", sum(s.stages.map(stage => sum(stage.shipping))));
526 break;
527 case "FILLRATE":
528 this.requireWeeks();
529 this.emit("FILLRATE", "weeks", "1-" + s.week);
530 s.stages.forEach((stage, i) => {
531 const f = this.fill(stage.records);
532 const stockouts = stage.records.filter(r => r.backlog > 0n).length;
533 this.emit(NAMES[i], "fill", f.text, "ontime", f.ontime, "ordered", f.ordered, "stockout_weeks", stockouts,
534 "service", decimal(BigInt((s.week - stockouts) * 100), s.week, 2) + "%");
535 });
536 break;
537 case "BULLWHIP": {
538 this.requireWeeks();
539 const n = s.week;
540 const demand = s.demands.slice(0, n);
541 const dv = variation(demand);
542 this.emit("BULLWHIP", "weeks", "1-" + n, "demand", "mean", decimal(sum(demand), n, 4), "var", decimal(dv, n * n, 4));
543 let previous = dv;
544 s.stages.forEach((stage, i) => {
545 const values = stage.records.map(r => r.order);
546 const v = variation(values);
547 this.emit(NAMES[i], "mean", decimal(sum(values), n, 4), "var", decimal(v, n * n, 4),
548 "ratio", dv === 0n ? "n/a" : decimal(v, dv, 4), "step", previous === 0n ? "n/a" : decimal(v, previous, 4));
549 previous = v;
550 });
551 break;
552 }
553 case "PEAKS":
554 this.requireWeeks();
555 this.peaks();
556 break;
557 case "LEADTIME": {
558 const i = this.stageIndex(a[0]);
559 this.requireWeeks();
560 this.leadtime(i);
561 break;
562 }
563 case "PLOT":
564 case "HISTOGRAM": {
565 const i = this.stageIndex(a[0]);
566 this.mode(a[1]);
567 const width = cmd === "HISTOGRAM" ? BigInt(a[2]) : 1n;
568 if (cmd === "HISTOGRAM" && (width < 1n || width > 100n)) throw new CommandError("invalid width " + a[2]);
569 this.requireWeeks();
570 if (cmd === "PLOT") {
571 const [first, last] = this.range(a.slice(2), s.week);
572 this.plot(i, a[1], first, last);
573 } else {
574 const bins = new Map<bigint, number>();
575 for (const value of this.values(i, a[1])) {
576 const bucket = floorDiv(value, width);
577 bins.set(bucket, (bins.get(bucket) ?? 0) + 1);
578 }
579 this.emit("HISTOGRAM", a[0], a[1], "width", width, "weeks", "1-" + s.week);
580 for (let bucket = min(...bins.keys()); bucket <= max(...bins.keys()); bucket++) {
581 const count = bins.get(bucket) ?? 0;
582 this.emit(bucket * width + ".." + (bucket * width + width - 1n), count, ...(count ? ["*".repeat(count)] : []));
583 }
584 }
585 break;
586 }
587 case "COMPARE": {
588 const policies: Policy[] = [];
589 let start = 0;
590 for (let end = 0; end <= a.length; end++) {
591 if (end === a.length || a[end] === "/") {
592 policies.push(parsePolicy(a.slice(start, end)));
593 start = end + 1;
594 }
595 }
596 const rows = policies.map((policy, index) => {
597 const trial = this.trial([policy, policy, policy, policy]);
598 const cost = sum(trial.costs());
599 const fill = this.fill(trial.stages[0].records).text;
600 const dv = variation(trial.demands);
601 const fv = variation(trial.stages[3].records.map(r => r.order));
602 return {policy, index, cost, fill, ratio: dv === 0n ? "n/a" : decimal(fv, dv, 4)};
603 });
604 rows.sort((a, b) => compare(a.cost, b.cost) || a.index - b.index);
605 this.emit("COMPARE", policies.length, "weeks", s.demands.length);
606 rows.forEach((r, i) => this.emit(i + 1, r.policy.text, "total", money(r.cost), "fill", r.fill, "bullwhip", r.ratio));
607 break;
608 }
609 case "OPTIMIZE": {
610 const indices = this.target(a[0]);
611 const low = Number(a[1]);
612 const high = Number(a[2]);
613 if (low > high) throw new CommandError("bad range " + a[1] + " " + a[2]);
614 if (high - low > 200) throw new CommandError("range too large");
615 let best: {objective: bigint; target: number; total: bigint} | null = null;
616 for (let target = low; target <= high; target++) {
617 const policies = s.stages.map(stage => stage.policy);
618 for (const i of indices) policies[i] = {name: "BASESTOCK", args: [BigInt(target)], text: "BASESTOCK " + target};
619 const costs = this.trial(policies).costs();
620 const objective = sum(indices.map(i => costs[i]));
621 if (best === null || objective < best.objective) best = {objective, target, total: sum(costs)};
622 }
623 this.emit("OPTIMIZE", a[0], "BASESTOCK", best!.target, "cost", money(best!.objective), "total", money(best!.total));
624 break;
625 }
626 default:
627 throw new CommandError("unknown command " + cmd);
628 }
629 }

634 this.command(cmd, args);

636 if (!(error instanceof CommandError)) throw error;
637 this.emit("ERROR", error.message);
638 }
639 }
