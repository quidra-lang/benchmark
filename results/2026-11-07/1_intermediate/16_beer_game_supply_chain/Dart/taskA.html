17int total(Iterable<int> values) => values.fold(0, (a, b) => a + b);
18int floorDiv(int a, int b) => a < 0 ? -((-a + b - 1) ~/ b) : a ~/ b;
19
20String money(int value) =>
21 '${value ~/ 100}.${(value % 100).toString().padLeft(2, '0')}';
22
23String decimal(int numerator, int denominator, int places) {
24 final scale = [1, 10, 100, 1000, 10000][places];
25 final fixed = floorDiv(numerator * scale, denominator);
26 return '${fixed ~/ scale}.${(fixed % scale).toString().padLeft(places, '0')}';
27}
28
29int variation(List<int> values) {
30 final n = values.length;
31 return n * total(values.map((v) => v * v)) -
32 total(values) * total(values);
33}

42Policy parsePolicy(List<String> tokens) {
43 final name = tokens.first;
44 final count = parameterCounts[name];
45 if (count == null) throw CommandError('unknown policy $name');
46 if (tokens.length - 1 != count) {
47 throw CommandError('$name needs $count parameters');
48 }
49 final args = <int>[];
50 for (var i = 1; i < tokens.length; ++i) {
51 args.add(int.parse(name == 'STERMAN' && i <= 3
52 ? tokens[i].replaceAll('.', '') : tokens[i]));
53 }
54 final valid = switch (name) {
55 'FIXED' => args[0] >= 0 && args[0] <= 1000,
56 'PASS' => true,
57 'BASESTOCK' => args[0] >= 0 && args[0] <= 10000,
58 'MINMAX' => args[0] >= 0 && args[0] < args[1] && args[1] <= 10000,
59 'MOVAVG' => args[0] >= 1 && args[0] <= 52 &&
60 args[1] >= 0 && args[1] <= 10000,
61 'SMOOTH' => args[0] >= 1 && args[0] <= 10 &&
62 args[1] >= 0 && args[1] <= 10000,
63 _ => args.take(3).every((v) => v >= 0 && v <= 100) &&
64 args[3] >= 0 && args[3] <= 10000,
65 };
66 if (!valid) throw CommandError('invalid parameter for $name');
67 return Policy(name, args, tokens.join(' '));
68}

147 int signal(int index, int targetWeek, [List<int>? current]) {
148 if (visibility == 'ON') return demands[targetWeek - 1];
149 if (current != null && targetWeek == week + 1) return current[index];
150 return stages[index].records[targetWeek - 1]['incoming']!;
151 }
152
153 void setPolicy(int index, Policy policy) {
154 final stage = stages[index];
155 stage.policy = policy;
156 if (policy.name == 'STERMAN') {
157 stage.expected = 100 * (week == 0
158 ? initials[index].quantity : signal(index, week));
159 }
160 }
161
162 (List<int>, List<int>) positions() {
163 final supply = <int>[], positions = <int>[];
164 for (var i = 0; i < stages.length; ++i) {
165 final stage = stages[i];
166 final upstream = i < 3 ? stages[i + 1].backlog : plantBacklog;
167 final line = total(stage.orders) + upstream + total(stage.shipping);
168 supply.add(line);
169 positions.add(stage.inventory + line - stage.backlog);
170 }
171 return (supply, positions);
172 }
173
174 int policyOrder(int index, List<int> incoming, int line, int position) {
175 final stage = stages[index];
176 final policy = stage.policy;
177 final args = policy.args;
178 final nextWeek = week + 1;
179 final demandSignal = signal(index, nextWeek, incoming);
180
181 if (policy.name == 'FIXED') return args[0];
182 if (policy.name == 'PASS') return incoming[index];
183 if (policy.name == 'BASESTOCK') {
184 return args[0] > position ? args[0] - position : 0;
185 }
186 if (policy.name == 'MINMAX') {
187 return position <= args[0] ? args[1] - position : 0;
188 }
189 if (policy.name == 'MOVAVG') {
190 final count = args[0] < nextWeek ? args[0] : nextWeek;
191 var sum = 0;
192 for (var w = nextWeek - count + 1; w <= nextWeek; ++w) {
193 sum += signal(index, w, incoming);
194 }
195 final forecast = sum ~/ count;
196 final initial = initials[index];
197 final order = args[1] +
198 forecast * (initial.shippingDelay + initial.orderDelay) - position;
199 return order > 0 ? order : 0;
200 }
201 if (policy.name == 'SMOOTH') {
202 final order = demandSignal + floorDiv(args[1] - position, args[0]);
203 return order > 0 ? order : 0;
204 }
205 final theta = args[0], alpha = args[1], beta = args[2], target = args[3];
206 stage.expected = floorDiv(
207 100 * theta * demandSignal + (100 - theta) * stage.expected, 100);
208 final net = stage.inventory - stage.backlog;
209 final order = floorDiv(
210 100 * stage.expected +
211 alpha * (100 * (target - net) - beta * line), 10000);
212 return order > 0 ? order : 0;
213 }
214
215 List<Map<String, int>> step() {
216 final received = <int>[];
217 for (final stage in stages) {
218 final quantity = stage.shipping.removeAt(0);
219 stage.inventory += quantity;
220 received.add(quantity);
221 }
222 final incoming = <int>[demands[week]];
223 for (var i = 0; i < 3; i++) {
224 incoming.add(stages[i].orders.removeAt(0));
225 }
226 final plantIncoming = stages[3].orders.removeAt(0);
227 final records = <Map<String, int>>[];
228 for (var i = 0; i < stages.length; i++) {
229 final stage = stages[i];
230 final before = stage.backlog;
231 final requested = before + incoming[i];
232 var shipped = stage.inventory < requested ? stage.inventory : requested;
233 if (stage.capacity != null && shipped > stage.capacity!) {
234 shipped = stage.capacity!;
235 }
236 stage.inventory -= shipped;
237 stage.backlog = requested - shipped;
238 if (i > 0) stages[i - 1].shipping.add(shipped);
239 final ontime = (shipped - before) > 0 ? shipped - before : 0;
240 records.add({
241 'received': received[i], 'incoming': incoming[i],
242 'shipped': shipped, 'before': before,
243 'ontime': ontime < incoming[i] ? ontime : incoming[i],
244 'inventory': stage.inventory, 'backlog': stage.backlog,
245 });
246 }
247 final requested = plantBacklog + plantIncoming;
248 final produced = plantCapacity == null || requested < plantCapacity!
249 ? requested : plantCapacity!;
250 plantBacklog = requested - produced;
251 stages[3].shipping.add(produced);
252 final (supply, positions) = this.positions();
253 for (var i = 0; i < stages.length; ++i) {
254 final stage = stages[i];
255 final order = policyOrder(i, incoming, supply[i], positions[i]);
256 final quantity = ((order + stage.lot - 1) ~/ stage.lot) * stage.lot;
257 stage.orders.add(quantity);
258 final record = records[i];
259 record['order'] = quantity;
260 record['holding'] = stage.holding * stage.inventory;
261 record['backlog_cost'] = stage.backlogRate * stage.backlog;
262 record['cost'] = record['holding']! + record['backlog_cost']!;
263 stage.records.add(record);
264 }
265 plantRecords.add({
266 'incoming': plantIncoming, 'produced': produced, 'backlog': plantBacklog,
267 });
268 week++;
269 return records;
270 }
271
272 List<int> costs() => stages.map((s) =>
273 total(s.records.map((r) => r['cost']!))).toList();
274}

354 void generate(List<String> args) {
355 final weeks = int.parse(args[0]), base = int.parse(args[1]),
356 spread = int.parse(args[2]);
357 var seed = int.parse(args[3]);
358 final changes = <(String, int, int)>[];
359 for (var i = 4; i < args.length; i += 3) {
360 changes.add((args[i], int.parse(args[i+1]), int.parse(args[i+2])));
361 }
362 for (final (kind, week, amount) in changes) {
363 if (week < 1 || week > weeks) {
364 throw CommandError('week $week outside horizon');
365 }
366 }
367 final levels = <int>[];
368 for (var week = 1; week <= weeks; ++week) {
369 var level = base;
370 for (final (kind, start, amount) in changes) {
371 if (kind == 'STEP' && week >= start) {
372 level += amount;
373 } else if (kind == 'PULSE' && week == start) {
374 level += amount;
375 } else if (kind == 'RAMP' && week >= start) {
376 level += amount * (week - start + 1);
377 }
378 }
379 if (level < 0) throw CommandError('negative level at week $week');
380 levels.add(level);
381 }
382 final result = <int>[];
383 for (final level in levels) {
384 seed = 48271 * seed % 2147483647;
385 result.add(level + seed % spread);
386 }
387 demand(result);
388 }

459 final s = simulation, n = simulation.week;

468 final variance = variation(orders);

471 'ratio', demandVariation == 0 ? 'n/a' : decimal(variance, demandVariation, 4),
472 'step', previous == 0 ? 'n/a' : decimal(variance, previous, 4)]);
473 previous = variance;
474 }

477 void peaks() {
478 final s = simulation;
479 emit(['PEAKS', 'weeks', '1-${s.week}']);
480 for (var i = 0; i < stages.length; ++i) {
481 final records = s.stages[i].records;
482 final inventories = records.map((r) => r['inventory']!).toList();
483 final backlogs = records.map((r) => r['backlog']!).toList();
484 final maxInv = inventories.reduce((a, b) => a > b ? a : b);
485 final maxBack = backlogs.reduce((a, b) => a > b ? a : b);
486 var longest = 0, run = 0, start = 0, bestStart = 0, bestEnd = 0;
487 for (var week = 1; week <= backlogs.length; week++) {
488 if (backlogs[week - 1] > 0) {
489 if (run == 0) start = week;
490 run++;
491 if (run > longest) {
492 longest = run;
493 bestStart = start;
494 bestEnd = week;
495 }
496 } else {
497 run = 0;
498 }
499 }
500 emit([stages[i], 'max_inv', maxInv, 'week',
501 maxInv != 0 ? inventories.indexOf(maxInv) + 1 : '-',
502 'max_back', maxBack, 'week',
503 maxBack != 0 ? backlogs.indexOf(maxBack) + 1 : '-',
504 'longest_stockout', longest == 0 ? 0 : '$longest weeks $bestStart-$bestEnd']);
505 }

508 void leadtime(int index) {
509 final s = simulation, stage = s.stages[index], initial = initials[index];
510 final nominal = initial.shippingDelay + initial.orderDelay;
511 final cumulative = <int>[];
512 var received = 0;
513 for (final record in stage.records) {
514 received += record['received']!;
515 cumulative.add(received - nominal * initial.quantity);
516 }
517 var ordered = 0, count = 0;
518 final leads = <int>[];
519 for (var t = 0; t < stage.records.length; t++) {
520 final record = stage.records[t];
521 ordered += record['order']!;
522 if (record['order'] == 0) continue;
523 count++;
524 for (var u = t + 1; u < stage.records.length; ++u) {
525 if (cumulative[u] >= ordered) {
526 leads.add(u - t);
527 break;
528 }
529 }
530 }
531 emit(['LEADTIME', stages[index], 'nominal', nominal,
532 'orders', count, 'completed', leads.length, 'open', count - leads.length,
533 'min', leads.isEmpty ? '-' : leads.reduce((a, b) => a < b ? a : b),
534 'avg', leads.isEmpty ? '-' : decimal(total(leads), leads.length, 2),
535 'max', leads.isEmpty ? '-' : leads.reduce((a, b) => a > b ? a : b)]);
536 }

561 }

565 void command(String cmd, List<String> a) {
566 final s = simulation;
567 switch (cmd) {
568 case 'DEMAND':
569 demand(a.map(int.parse).toList());
570 break;
571 case 'GENERATE':
572 generate(a);
573 break;
574 case 'RUN':
575 if (s.week == s.demands.length) {
576 throw CommandError('horizon reached at week ${s.demands.length}');
577 }
578 final first = s.week + 1;
579 final count = int.parse(a[0]) < s.demands.length - s.week
580 ? int.parse(a[0]) : s.demands.length - s.week;
581 var cost = 0;
582 for (var i = 0; i < count; ++i) {
583 final records = s.step();
584 for (var stage = 0; stage < stages.length; stage++) {
585 final record = records[stage];
586 cost += record['cost']!;
587 if (record['before'] == 0 && record['backlog']! > 0) {
588 emit(['STOCKOUT', stages[stage], 'week', s.week,
589 'backlog', record['backlog']!]);
590 } else if (record['before']! > 0 && record['backlog'] == 0) {
591 emit(['RECOVER', stages[stage], 'week', s.week]);
592 }
593 }
594 }
595 emit(['RUN', '$first-${s.week}', 'demand',
596 total(s.demands.sublist(first-1, s.week)), 'cost', money(cost)]);
597 if (s.week == s.demands.length) emit(['HORIZON', s.week]);
598 break;
599 case 'RESET':
600 initialize();
601 emit(['RESET', 'week', 0]);
602 break;
603 case 'SAVE':
604 snapshots[a[0]] = s.clone();
605 emit(['SAVE', a[0], 'week', s.week]);
606 break;
607 case 'LOAD':
608 if (!snapshots.containsKey(a[0])) throw CommandError('no snapshot ${a[0]}');
609 simulation = snapshots[a[0]]!.clone();
610 emit(['LOAD', a[0], 'week', simulation.week]);
611 break;
612 case 'SET':
613 final indices = target(a[0]);
614 final policy = parsePolicy(a.sublist(1));
615 for (final index in indices) {
616 s.setPolicy(index, policy);
617 }
618 emit(['SET', a[0], policy.display]);
619 break;
620 case 'BATCH':
621 final index = stageIndex(a[0]);
622 final lot = int.parse(a[1]);
623 if (lot < 1 || lot > 100) throw CommandError('invalid batch size ${a[1]}');
624 s.stages[index].lot = lot;
625 emit(['BATCH', a[0], lot]);
626 break;
627 case 'CAPACITY':
628 final index = a[0] == 'PLANT' ? null : stageIndex(a[0]);
629 final capacity = a[1] == 'OFF' ? null : int.parse(a[1]);
630 if (capacity != null && capacity > 10000) {
631 throw CommandError('invalid capacity ${a[1]}');
632 }
633 if (index == null) {
634 s.plantCapacity = capacity;
635 } else {
636 s.stages[index].capacity = capacity;
637 }
638 emit(['CAPACITY', a[0], a[1]]);
639 break;
640 case 'COST':
641 final index = stageIndex(a[0]);
642 final holding = int.parse(a[1]), backlog = int.parse(a[2]);
643 if (holding > 1000 || backlog > 1000) throw CommandError('invalid cost');
644 s.stages[index].holding = holding;
645 s.stages[index].backlogRate = backlog;
646 emit(['COST', a[0], 'holding', holding, 'backlog', backlog]);
647 break;
648 case 'VISIBILITY':
649 if (a[0] != 'ON' && a[0] != 'OFF') {
650 throw CommandError('unknown visibility ${a[0]}');
651 }
652 s.visibility = a[0];
653 emit(['VISIBILITY', a[0]]);
654 break;
655 case 'SHOCK':
656 final week = int.parse(a[0]), value = int.parse(a[1]);
657 if (week < 1 || week > s.demands.length) {
658 throw CommandError('week ${a[0]} outside horizon');
659 }
660 if (week <= s.week) throw CommandError('week ${a[0]} already simulated');
661 final old = s.demands[week - 1];
662 s.demands[week - 1] = value;
663 emit(['SHOCK', 'week', week, old, '->', value]);
664 break;
665 case 'STATUS':
666 status();
667 break;
668 case 'SETTINGS':
669 emit(['SETTINGS']);
670 for (var i = 0; i < stages.length; ++i) {
671 final stage = s.stages[i], initial = initials[i];
672 emit([stages[i], 'holding', stage.holding, 'backlog', stage.backlogRate,
673 'batch', stage.lot, 'capacity', stage.capacity ?? 'OFF',
674 'delays', initial.shippingDelay, initial.orderDelay]);
675 }
676 emit(['PLANT', 'capacity', s.plantCapacity ?? 'OFF']);
677 emit(['VISIBILITY', s.visibility]);
678 break;
679 case 'WEEK':
680 requireWeeks();
681 final week = int.parse(a[0]);
682 if (week < 1 || week > s.week) throw CommandError('bad week ${a[0]}');
683 showWeek(week);
684 break;
685 case 'TABLE':
686 final index = stageIndex(a[0]);
687 requireWeeks();
688 final (first, last) = range(a.sublist(1), s.week);
689 table(index, first, last);
690 break;
691 case 'CHAIN':
692 requireWeeks();
693 final (first, last) = range(a, s.week);
694 emit(['CHAIN', '$first-$last']);
695 emit(['${'week'.padLeft(4)} ${'demand'.padLeft(6)} '
696 '${stages.map((s) => s.padLeft(11)).join(' ')}']);
697 for (var week = first; week <= last; week++) {
698 final orders = s.stages.map((stage) =>
699 stage.records[week - 1]['order']!.toString().padLeft(11)).join(' ');
700 emit(['${week.toString().padLeft(4)} '
701 '${s.demands[week-1].toString().padLeft(6)} $orders']);
702 }
703 break;
704 case 'DEMANDS':
705 final (first, last) = range(a, s.demands.length);
706 final values = s.demands.sublist(first-1, last);
707 emit(['DEMANDS', '$first-$last', 'total', total(values),
708 'min', values.reduce((a,b) => a < b ? a : b),
709 'max', values.reduce((a,b) => a > b ? a : b)]);
710 emit(values.cast<Object>());
711 break;
712 case 'COSTS':
713 costs();
714 break;
715 case 'SUMMARY':
716 requireWeeks();
717 emit(['SUMMARY', 'weeks', '1-${s.week}']);
718 for (var i = 0; i < stages.length; ++i) {
719 final records = s.stages[i].records;
720 final orders = records.map((r) => r['order']!).toList();
721 emit([stages[i], 'avg_inv',
722 decimal(total(records.map((r) => r['inventory']!)), s.week, 2),
723 'avg_back', decimal(total(records.map((r) => r['backlog']!)), s.week, 2),
724 'orders', total(orders),
725 'min', orders.reduce((a,b) => a < b ? a : b),
726 'max', orders.reduce((a,b) => a > b ? a : b),
727 'zero_weeks', orders.where((o) => o == 0).length]);
728 }
729 break;
730 case 'BALANCE':
731 emit(['BALANCE', 'week', s.week]);
732 for (var i = 0; i < stages.length; ++i) {
733 final record = s.stages[i].records;
734 emit([stages[i], 'start', initials[i].inventory,
735 'received', total(record.map((r) => r['received']!)),
736 'shipped', total(record.map((r) => r['shipped']!)),
737 'end', s.stages[i].inventory, 'owed', s.stages[i].backlog]);
738 }
739 final initialGoods = total(initials.map((i) =>
740 i.inventory + i.quantity * i.shippingDelay));
741 emit(['SYSTEM', 'initial', initialGoods,
742 'produced', total(s.plantRecords.map((r) => r['produced']!)),
743 'delivered', total(s.stages[0].records.map((r) => r['shipped']!)),
744 'stock', total(s.stages.map((stage) => stage.inventory)),
745 'transit', total(s.stages.map((stage) => total(stage.shipping)))]);
746 break;
747 case 'FILLRATE':
748 requireWeeks();
749 emit(['FILLRATE', 'weeks', '1-${s.week}']);
750 for (var i = 0; i < stages.length; ++i) {
751 final stage = s.stages[i];
752 final (ordered, ontime, fill) = fillText(stage.records);
753 final stockouts = stage.records.where((r) => r['backlog']! > 0).length;
754 emit([stages[i], 'fill', fill, 'ontime', ontime, 'ordered', ordered,
755 'stockout_weeks', stockouts,
756 'service', '${decimal((s.week-stockouts)*100, s.week, 2)}%']);
757 }
758 break;
759 case 'BULLWHIP':
760 requireWeeks();
761 bullwhip();
762 break;
763 case 'PEAKS':
764 requireWeeks();
765 peaks();
766 break;
767 case 'LEADTIME':
768 final index = stageIndex(a[0]);
769 requireWeeks();
770 leadtime(index);
771 break;
772 case 'PLOT':
773 case 'HISTOGRAM':
774 final index = stageIndex(a[0]);
775 checkMode(a[1]);
776 int? width;
777 if (cmd == 'HISTOGRAM') {
778 width = int.parse(a[2]);
779 if (width < 1 || width > 100) {
780 throw CommandError('invalid width ${a[2]}');
781 }
782 }
783 requireWeeks();
784 if (cmd == 'PLOT') {
785 final (first, last) = range(a.sublist(2), s.week);
786 plot(index, a[1], first, last);
787 } else {
788 final values = this.values(index, a[1]);
789 final bins = <int, int>{};
790 for (final value in values) {
791 final bucket = floorDiv(value, width!);
792 bins[bucket] = (bins[bucket] ?? 0) + 1;
793 }
794 emit(['HISTOGRAM', a[0], a[1], 'width', width!,
795 'weeks', '1-${s.week}']);
796 final keys = bins.keys.toList()..sort();
797 for (var bucket = keys.first; bucket <= keys.last; bucket++) {
798 final count = bins[bucket] ?? 0;
799 final parts = <Object>['${bucket * width}..${bucket * width + width - 1}', count];
800 if (count != 0) parts.add(List.filled(count, '*').join());
801 emit(parts);
802 }
803 }
804 break;
805 case 'COMPARE':
806 final policies = <Policy>[];
807 var start = 0;
808 for (var end = 0; end <= a.length; end++) {
809 if (end == a.length || a[end] == '/') {
810 policies.add(parsePolicy(a.sublist(start, end)));
811 start = end + 1;
812 }
813 }
814 final rows = <(int, int, String, String, String)>[];
815 for (var index = 0; index < policies.length; index++) {
816 final trial = this.trial(List.filled(4, policies[index]));
817 final cost = total(trial.costs());
818 final (_, __, fill) = fillText(trial.stages[0].records);
819 final demandVariance = variation(trial.demands);
820 final factoryVariance = variation(trial.stages[3].records
821 .map((r) => r['order']!).toList());
822 final ratio = demandVariance == 0 ? 'n/a' :
823 decimal(factoryVariance, demandVariance, 4);
824 rows.add((cost, index, policies[index].display, fill, ratio));
825 }
826 rows.sort((a,b) => a.$1 != b.$1 ? a.$1.compareTo(b.$1) : a.$2.compareTo(b.$2));
827 emit(['COMPARE', policies.length, 'weeks', s.demands.length]);
828 for (var i = 0; i < rows.length; i++) {
829 final item = rows[i];
830 emit([i+1, item.$3, 'total', money(item.$1),
831 'fill', item.$4, 'bullwhip', item.$5]);
832 }
833 break;
834 case 'OPTIMIZE':
835 final indices = target(a[0]);
836 final low = int.parse(a[1]), high = int.parse(a[2]);
837 if (low > high) throw CommandError('bad range ${a[1]} ${a[2]}');
838 if (high - low > 200) throw CommandError('range too large');
839 (int, int, int)? best;
840 for (var target = low; target <= high; ++target) {
841 final policies = s.stages.map((stage) => stage.policy).toList();
842 for (final index in indices) {
843 policies[index] = Policy('BASESTOCK', [target], 'BASESTOCK $target');
844 }
845 final trial = this.trial(policies);
846 final costs = trial.costs();
847 final objective = total(indices.map((i) => costs[i]));
848 final candidate = (objective, target, total(costs));
849 if (best == null || candidate.$1 < best.$1 ||
850 (candidate.$1 == best.$1 && candidate.$2 < best.$2)) {
851 best = candidate;
852 }
853 }
854 emit(['OPTIMIZE', a[0], 'BASESTOCK', best!.$2,
855 'cost', money(best.$1), 'total', money(best.$3)]);
856 break;
857 default:
858 throw CommandError('unknown command $cmd');
859 }
860 }
861
862 void execute(String line) {
863 final tokens = line.trim().split(' ');
864 final command = tokens.first;
865 try {
866 this.command(command, tokens.sublist(1));
867 } on CommandError catch (error) {
868 emit(['ERROR', error.message]);
869 }
870 }
871}

873void main() {
874 final initial = <List<String>>[];
875 for (var i = 0; i < 4; ++i) {
876 initial.add(stdin.readLineSync()!.split(' '));
877 }
878 final game = Game(initial);
879 while (true) {
880 final line = stdin.readLineSync();
881 if (line == null || line == 'END') break;
882 game.execute(line);
883 }
884 if (game.output.isNotEmpty) stdout.write('${game.output.join('\n')}\n');
885}
