43Policy | error parse_policy(const string[] words)
44 name := words[0]
45 int count
46 if name == "PASS"
47 count = 0
48 elif name == "FIXED" or name == "BASESTOCK"
49 count = 1
50 elif name == "MINMAX" or name == "MOVAVG" or name == "SMOOTH"
51 count = 2
52 elif name == "STERMAN"
53 count = 4
54 else
55 return error("unknown policy " + name)
56 if int(#words) != count + 1
57 return error("{name} needs {count} parameters")
58 int[] parameters := []
59 for: 0 <= index < count
60 text := words[index + 1]
61 if name == "STERMAN" and index < 3
62 parts := text.split(".")
63 parameters = parameters.append(100 * number(parts[0]) + number(parts[1]))
64 else
65 parameters = parameters.append(number(text))
66 valid := true
67 if name == "FIXED"
68 valid = parameters[0] <= 1000
69 elif name == "BASESTOCK"
70 valid = parameters[0] <= 10000
71 elif name == "MINMAX"
72 valid = parameters[0] < parameters[1] and parameters[1] <= 10000
73 elif name == "MOVAVG" or name == "SMOOTH"
74 valid = 1 <= parameters[0] <= (if name == "MOVAVG" then 52 else 10) and parameters[1] <= 10000
75 elif name == "STERMAN"
76 valid = parameters[0] <= 100 and parameters[1] <= 100 and parameters[2] <= 100 and parameters[3] <= 10000
77 if not valid
78 return error("invalid parameter for " + name)
79 return Policy(name, parameters, words.join(" "))

119void assign_policy(Simulation &simulation, int index, const Policy policy)

123int policy_order(Simulation &simulation, int index, int supply, int position)
124 &stage := &simulation.stages[index]
125 policy := stage.policy
126 parameters := policy.parameters
127 week := simulation.week + 1
128 incoming := stage.records[week - 1].incoming
129 demand_signal := signal(simulation, index, week)
130 if policy.name == "FIXED"
131 return parameters[0]
132 if policy.name == "PASS"
133 return incoming
134 if policy.name == "BASESTOCK"
135 return math.max(0, parameters[0] - position)
136 if policy.name == "MINMAX"
137 return if position <= parameters[0] then parameters[1] - position else 0
138 if policy.name == "MOVAVG"
139 count := math.min(parameters[0], week)
140 int[] signals := []
141 for: week - count + 1 <= int previous <= week
142 signals = signals.append(signal(simulation, index, previous))
143 forecast := signals.sum() / count
144 return math.max(0, parameters[1] + forecast * (stage.initial.shipping_delay + stage.initial.order_delay) - position)
145 if policy.name == "SMOOTH"
146 return math.max(0, demand_signal + floor_div(parameters[1] - position, parameters[0]))
147 stage.expected = floor_div(100 * parameters[0] * demand_signal + (100 - parameters[0]) * stage.expected, 100)
148 net := stage.inventory - stage.backlog
149 return math.max(0, floor_div(100 * stage.expected + parameters[1] * (100 * (parameters[3] - net) - parameters[2] * supply), 10000))

151void advance_week(Simulation &simulation)
152 int[] received := []
153 for &stage := &simulation.stages[]
154 received = received.append(stage.shipping[0])
155 stage.inventory += stage.shipping[0]
156 stage.shipping = stage.shipping[1:]
157 incoming := [simulation.demand[simulation.week]]
158 for &stage := &simulation.stages[]
159 incoming = incoming.append(stage.pending[0])
160 stage.pending = stage.pending[1:]
161 for: 0 <= index < 4
162 &stage := &simulation.stages[index]
163 before := stage.backlog
164 requested := before + incoming[index]
165 shipped := math.min(stage.inventory, requested)
166 if stage.capacity >= 0
167 shipped = math.min(shipped, stage.capacity)
168 stage.inventory -= shipped
169 stage.backlog = requested - shipped
170 ontime := math.min(incoming[index], math.max(0, shipped - before))
171 stage.records = stage.records.append(Record(received[index], incoming[index], shipped, before, ontime, stage.inventory, stage.backlog, 0, 0, 0, 0))
172 if index > 0
173 simulation.stages[index - 1].shipping = simulation.stages[index - 1].shipping.append(shipped)
174 requested := simulation.plant_backlog + incoming[4]
175 produced := if simulation.plant_capacity < 0 then requested else math.min(requested, simulation.plant_capacity)
176 simulation.plant_backlog = requested - produced
177 simulation.stages[3].shipping = simulation.stages[3].shipping.append(produced)
178 simulation.plant_records = simulation.plant_records.append(PlantRecord(incoming[4], produced, simulation.plant_backlog))
179 int[] supplies := []
180 int[] positions := []
181 for: 0 <= index < 4
182 supply := supply_line(simulation, index)
183 supplies = supplies.append(supply)
184 positions = positions.append(simulation.stages[index].inventory + supply - simulation.stages[index].backlog)
185 for: 0 <= index < 4
186 quantity := policy_order(&simulation, index, supplies[index], positions[index])
187 &stage := &simulation.stages[index]
188 quantity = (quantity + stage.lot - 1) / stage.lot * stage.lot
189 stage.pending = stage.pending.append(quantity)
190 &record := &stage.records[simulation.week]
191 record.order = quantity
192 record.holding = stage.inventory * stage.holding_rate
193 record.owing = stage.backlog * stage.backlog_rate
194 record.cost = record.holding + record.owing
195 simulation.week += 1

216Simulation trial(const Game game, const Policy[] policies)
217 result := fresh(game.initial, game.current.demand)
218 result.visible = game.current.visible
219 result.plant_capacity = game.current.plant_capacity
220 for: 0 <= index < 4
221 &stage := &result.stages[index]
222 current := game.current.stages[index]
223 stage.lot = current.lot
224 stage.capacity = current.capacity
225 stage.holding_rate = current.holding_rate
226 stage.backlog_rate = current.backlog_rate
227 assign_policy(&result, index, policies[index])
228 while result.week < int(#result.demand)
229 advance_week(&result)
230 return result
