12sealed class Policy
13{
14 static readonly Dictionary<string, int> ParameterCounts = new()
15 {
16 ["FIXED"] = 1, ["PASS"] = 0, ["BASESTOCK"] = 1,
17 ["MINMAX"] = 2, ["MOVAVG"] = 2, ["SMOOTH"] = 2, ["STERMAN"] = 4
18 };
19
20 public string Name { get; }
21 public long[] Values { get; }
22 public string Notation { get; }
23
24 public Policy(string name, long[] values, string notation)
25 {
26 Name = name;
27 Values = values;
28 Notation = notation;
29 }
30
31 public static Policy Parse(IReadOnlyList<string> tokens)
32 {
33 string name = tokens.Count > 0 ? tokens[0] : "";
34 if (!ParameterCounts.TryGetValue(name, out int expected))
35 throw new CommandError("unknown policy " + name);
36 if (tokens.Count != expected + 1)
37 throw new CommandError($"{name} needs {expected} parameters");
38
39 var values = new long[expected];
40 for (int i = 0; i < expected; ++i)
41 {
42 string text = tokens[i + 1];
43 values[i] = long.Parse(name == "STERMAN" && i < 3
44 ? text.Replace(".", "") : text, CultureInfo.InvariantCulture);
45 }
46
47 bool valid = name switch
48 {
49 "FIXED" => values[0] >= 0 && values[0] <= 1000,
50 "PASS" => true,
51 "BASESTOCK" => values[0] >= 0 && values[0] <= 10000,
52 "MINMAX" => values[0] >= 0 && values[0] < values[1] && values[1] <= 10000,
53 "MOVAVG" => values[0] >= 1 && values[0] <= 52 &&
54 values[1] >= 0 && values[1] <= 10000,
55 "SMOOTH" => values[0] >= 1 && values[0] <= 10 &&
56 values[1] >= 0 && values[1] <= 10000,
57 _ => values.Take(3).All(value => value >= 0 && value <= 100) &&
58 values[3] >= 0 && values[3] <= 10000
59 };
60 if (!valid) throw new CommandError("invalid parameter for " + name);
61 return new Policy(name, values, string.Join(" ", tokens));
62 }
63
64 public static Policy BaseStock(int target) =>
65 new("BASESTOCK", new long[] { target }, $"BASESTOCK {target}");
66}

180 public long Signal(int index, int week, long[]? current = null)
181 {
182 if (Visibility == "ON") return Demands[week - 1];
183 if (current != null && week == Week + 1) return current[index];
184 return Stages[index].Records[week - 1].Incoming;

187 public void SetPolicy(int index, Policy policy)
188 {
189 Stage stage = Stages[index];
190 stage.Policy = policy;
191 if (policy.Name == "STERMAN")
192 {
193 long signal = Week > 0
194 ? Signal(index, Week)
195 : Initials[index].Quantity;
196 stage.Expected = 100 * signal;
197 }
198 }
199
200 public (long[] Supply, long[] Position) Positions()
201 {
202 var supply = new long[4];
203 var position = new long[4];
204 for (int i = 0; i < 4; ++i)
205 {
206 Stage stage = Stages[i];
207 long upstreamBacklog = i < 3 ? Stages[i + 1].Backlog : PlantBacklog;
208 long line = stage.Orders.Sum() + upstreamBacklog + stage.Shipping.Sum();
209 supply[i] = line;
210 position[i] = stage.Inventory + line - stage.Backlog;
211 }
212 return (supply, position);
213 }

215 static long FloorDiv(long numerator, long denominator)
216 {
217 long quotient = numerator / denominator;
218 return numerator % denominator < 0 ? quotient - 1 : quotient;
219 }
220
221 long PolicyOrder(int index, long[] incoming, long line, long position)
222 {
223 Stage stage = Stages[index];
224 string name = stage.Policy.Name;
225 long[] args = stage.Policy.Values;
226 int week = Week + 1;
227 long signal = Signal(index, week, incoming);
228
229 switch (name)
230 {
231 case "FIXED":
232 return args[0];
233 case "PASS":
234 return incoming[index];
235 case "BASESTOCK":
236 return Math.Max(0, args[0] - position);
237 case "MINMAX":
238 return position <= args[0] ? args[1] - position : 0;
239 case "MOVAVG":
240 {
241 int count = (int)Math.Min(args[0], week);
242 long signals = 0;
243 for (int u = week - count + 1; u <= week; ++u)
244 signals += Signal(index, u, incoming);
245 long forecast = signals / count;
246 Initial initial = Initials[index];
247 int delay = initial.ShippingDelay + initial.OrderDelay;
248 return Math.Max(0, args[1] + forecast * delay - position);
249 }
250 case "SMOOTH":
251 return Math.Max(0, signal + FloorDiv(args[1] - position, args[0]));
252 default:
253 {
254 long theta = args[0];
255 long alpha = args[1];
256 long beta = args[2];
257 long target = args[3];
258 stage.Expected = FloorDiv(
259 100 * theta * signal + (100 - theta) * stage.Expected, 100);
260 long net = stage.Inventory - stage.Backlog;
261 return Math.Max(0, FloorDiv(
262 100 * stage.Expected + alpha * (100 * (target - net) - beta * line),
263 10000));
264 }
265 }
266 }
267
268 public WeekRecord[] Step()
269 {
270 var received = new long[4];
271 var incoming = new long[4];
272 for (int i = 0; i < 4; ++i)
273 {
274 Stage stage = Stages[i];
275 received[i] = stage.Shipping[0];
276 stage.Shipping.RemoveAt(0);
277 stage.Inventory += received[i];
278 }
279
280 incoming[0] = Demands[Week];
281 for (int i = 1; i < 4; ++i)
282 {
283 incoming[i] = Stages[i - 1].Orders[0];
284 Stages[i - 1].Orders.RemoveAt(0);
285 }
286 long plantIncoming = Stages[3].Orders[0];
287 Stages[3].Orders.RemoveAt(0);
288
289 var records = new WeekRecord[4];
290 for (int i = 0; i < 4; ++i)
291 {
292 Stage stage = Stages[i];
293 long before = stage.Backlog;
294 long requested = before + incoming[i];
295 long shipped = Math.Min(stage.Inventory, requested);
296 if (stage.Capacity.HasValue)
297 shipped = Math.Min(shipped, stage.Capacity.Value);
298 stage.Inventory -= shipped;
299 stage.Backlog = requested - shipped;
300 if (i > 0) Stages[i - 1].Shipping.Add(shipped);
301 records[i] = new WeekRecord
302 {
303 Received = received[i],
304 Incoming = incoming[i],
305 Shipped = shipped,
306 Before = before,
307 Ontime = Math.Min(Math.Max(0, shipped - before), incoming[i]),
308 Inventory = stage.Inventory,
309 Backlog = stage.Backlog
310 };
311 }
312
313 long requestedPlant = PlantBacklog + plantIncoming;
314 long produced = PlantCapacity.HasValue
315 ? Math.Min(requestedPlant, PlantCapacity.Value)
316 : requestedPlant;
317 PlantBacklog = requestedPlant - produced;
318 Stages[3].Shipping.Add(produced);
319
320 var (supply, positions) = Positions();
321 var orders = new long[4];
322 for (int i = 0; i < 4; ++i)
323 orders[i] = PolicyOrder(i, incoming, supply[i], positions[i]);
324
325 for (int i = 0; i < 4; ++i)
326 {
327 Stage stage = Stages[i];
328 long lot = stage.Lot;
329 WeekRecord record = records[i];
330 record.Order = ((orders[i] + lot - 1) / lot) * lot;
331 stage.Orders.Add(record.Order);
332 record.Holding = stage.HoldingRate * stage.Inventory;
333 record.BacklogCost = stage.BacklogRate * stage.Backlog;
334 record.Cost = record.Holding + record.BacklogCost;
335 stage.Records.Add(record);
336 }
337 PlantRecords.Add(new PlantRecord(plantIncoming, produced, PlantBacklog));
338 ++Week;
339 return records;
340 }
341
342 public long[] TotalCosts()
343 {
344 var result = new long[4];
345 for (int i = 0; i < 4; ++i)
346 result[i] = Stages[i].Records.Sum(record => record.Cost);
347 return result;
348 }
349}

373 static string Decimal(long numerator, long denominator, int places)
374 {
375 long scale = 1;
376 for (int i = 0; i < places; ++i) scale *= 10;
377 long scaled = numerator * scale / denominator;
378 return (scaled / scale) + "." +
379 (scaled % scale).ToString("D" + places, CultureInfo.InvariantCulture);

382 static long Variation(IEnumerable<long> source)
383 {
384 long[] values = source.ToArray();
385 long total = values.Sum();
386 long squares = values.Sum(value => value * value);
387 return values.Length * squares - total * total;
388 }

469 void Generate(IReadOnlyList<string> args)
470 {
471 int weeks = Whole(args[0]);
472 long baseLevel = Num(args[1]), spread = Num(args[2]);
473 long seed = Num(args[3]);
474 for (int i = 4; i < args.Count; i += 3)
475 {
476 int week = Whole(args[i + 1]);
477 if (week < 1 || week > weeks)
478 throw new CommandError($"week {args[i + 1]} outside horizon");
479 }
480
481 var levels = new List<long>();
482 for (int week = 1; week <= weeks; ++week)
483 {
484 long level = baseLevel;
485 for (int i = 4; i < args.Count; i += 3)
486 {
487 string kind = args[i];
488 int first = Whole(args[i + 1]);
489 long amount = Num(args[i + 2]);
490 if (kind == "STEP" && week >= first) level += amount;
491 else if (kind == "PULSE" && week == first) level += amount;
492 else if (kind == "RAMP" && week >= first)
493 level += amount * (week - first + 1);
494 }
495 if (level < 0)
496 throw new CommandError($"negative level at week {week}");
497 levels.Add(level);
498 }
499
500 var demands = new List<long>();
501 foreach (long level in levels)
502 {
503 seed = 48271 * seed % 2147483647L;
504 demands.Add(level + seed % spread);
505 }
506 Demand(demands);
507 }

589 void Bullwhip()
590 {
591 Simulation s = simulation;
592 int n = s.Week;
593 var demands = s.Demands.Take(n).ToList();
594 long demandVariation = Variation(demands);
595 Emit("BULLWHIP", "weeks", $"1-{n}", "demand", "mean",
596 Decimal(demands.Sum(), n, 4), "var",
597 Decimal(demandVariation, (long)n * n, 4));
598 long previous = demandVariation;
599 for (int i = 0; i < 4; ++i)
600 {
601 var orders = s.Stages[i].Records.Select(record => record.Order).ToList();
602 long variation = Variation(orders);
603 Emit(Names[i], "mean", Decimal(orders.Sum(), n, 4),
604 "var", Decimal(variation, (long)n * n, 4),
605 "ratio", demandVariation == 0 ? "n/a" : Decimal(variation, demandVariation, 4),
606 "step", previous == 0 ? "n/a" : Decimal(variation, previous, 4));
607 previous = variation;
608 }
609 }

611 void Peaks()
612 {
613 Simulation s = simulation;
614 Emit("PEAKS", "weeks", $"1-{s.Week}");
615 for (int index = 0; index < 4; ++index)
616 {
617 var records = s.Stages[index].Records;
618 long maxInv = 0, maxBack = 0;
619 int invWeek = 0, backWeek = 0;
620 int longest = 0, run = 0, start = 0, bestStart = 0, bestEnd = 0;
621 for (int i = 0; i < records.Count; ++i)
622 {
623 WeekRecord r = records[i];
624 int week = i + 1;
625 if (r.Inventory > maxInv) { maxInv = r.Inventory; invWeek = week; }
626 if (r.Backlog > maxBack) { maxBack = r.Backlog; backWeek = week; }
627 if (r.Backlog > 0)
628 {
629 if (run == 0) start = week;
630 ++run;
631 if (run > longest)
632 {
633 longest = run;
634 bestStart = start;
635 bestEnd = week;
636 }
637 }
638 else run = 0;
639 }
640 string stockout = longest == 0 ? "0" :
641 $"{longest} weeks {bestStart}-{bestEnd}";
642 Emit(Names[index], "max_inv", maxInv, "week", invWeek == 0 ? "-" : invWeek,
643 "max_back", maxBack, "week", backWeek == 0 ? "-" : backWeek,
644 "longest_stockout", stockout);
645 }
646 }

648 void Leadtime(int index)
649 {
650 Stage stage = simulation.Stages[index];
651 Initial initial = initials[index];
652 int nominal = initial.ShippingDelay + initial.OrderDelay;
653 var arrivals = new long[stage.Records.Count];
654 long received = -nominal * initial.Quantity;
655 for (int i = 0; i < arrivals.Length; ++i)
656 {
657 received += stage.Records[i].Received;
658 arrivals[i] = received;
659 }
660 long ordered = 0;
661 int count = 0;
662 var leads = new List<int>();
663 for (int t = 0; t < stage.Records.Count; ++t)
664 {
665 long amount = stage.Records[t].Order;
666 ordered += amount;
667 if (amount == 0) continue;
668 ++count;
669 for (int u = t + 1; u < stage.Records.Count; ++u)
670 {
671 if (arrivals[u] >= ordered)
672 {
673 leads.Add(u - t);
674 break;
675 }
676 }
677 }
678 Emit("LEADTIME", Names[index], "nominal", nominal, "orders", count,
679 "completed", leads.Count, "open", count - leads.Count,
680 "min", leads.Count > 0 ? leads.Min() : "-",
681 "avg", leads.Count > 0 ? Decimal(leads.Sum(), leads.Count, 2) : "-",
682 "max", leads.Count > 0 ? leads.Max() : "-");
683 }

685 void Plot(int index, string kind, int first, int last)
686 {
687 List<long> values = Values(index, kind).Skip(first - 1).Take(last - first + 1).ToList();
688 long maximum = values.Max();
689 if (kind == "NET")
690 {
691 long minimum = values.Min();
692 long scale = values.Max(value => Math.Abs(value));
693 Emit("PLOT", Names[index], "NET", $"{first}-{last}",
694 "min", minimum, "max", maximum);
695 for (int i = 0; i < values.Count; ++i)
696 {
697 long value = values[i];
698 int length = scale == 0 ? 0 : (int)(Math.Abs(value) * 25 / scale);
699 string left = value < 0
700 ? new string(' ', 25 - length) + new string('-', length)
701 : new string(' ', 25);
702 string right = value > 0 ? new string('+', length) : "";
703 Emit($"{(i + first).ToString().PadLeft(3)} {left}|{right} {value}");
704 }
705 }
706 else
707 {
708 Emit("PLOT", Names[index], kind, $"{first}-{last}", "max", maximum);
709 for (int i = 0; i < values.Count; ++i)
710 {
711 long value = values[i];
712 int length = maximum == 0 ? 0 : (int)(value * 50 / maximum);
713 Emit($"{(i + first).ToString().PadLeft(3)} |{new string('#', length)} {value}");
714 }
715 }

718 void Histogram(int index, string kind, int width)
719 {
720 var counts = new SortedDictionary<long, int>();
721 foreach (long value in Values(index, kind))
722 {
723 long bucket = Math.DivRem(value, width, out long remainder);
724 if (remainder < 0) --bucket;
725 counts[bucket] = counts.GetValueOrDefault(bucket, 0) + 1;
726 }
727 Emit("HISTOGRAM", Names[index], kind, "width", width,
728 "weeks", $"1-{simulation.Week}");
729 for (long bucket = counts.Keys.First(); bucket <= counts.Keys.Last(); ++bucket)
730 {
731 int count = counts.GetValueOrDefault(bucket, 0);
732 string line = $"{bucket * width}..{bucket * width + width - 1} {count}";
733 if (count > 0) line += " " + new string('*', count);
734 Emit(line);
735 }
736 }

738 void Compare(IReadOnlyList<string> tokens)
739 {
740 var policies = new List<Policy>();
741 int start = 0;
742 for (int end = 0; end <= tokens.Count; ++end)
743 {
744 if (end == tokens.Count || tokens[end] == "/")
745 {
746 policies.Add(Policy.Parse(tokens.Skip(start).Take(end - start).ToArray()));
747 start = end + 1;
748 }
749 }
750
751 var results = new List<(long Cost, int Original, string Label, string Fill, string Ratio)>();
752 for (int i = 0; i < policies.Count; ++i)
753 {
754 Policy policy = policies[i];
755 Simulation trial = Trial(new[] { policy, policy, policy, policy });
756 long cost = trial.TotalCosts().Sum();
757 string fill = FillText(trial.Stages[0].Records).Fill;
758 long demandVariation = Variation(trial.Demands);
759 long factoryVariation = Variation(trial.Stages[3].Records.Select(r => r.Order));
760 string ratio = demandVariation == 0
761 ? "n/a" : Decimal(factoryVariation, demandVariation, 4);
762 results.Add((cost, i, policy.Notation, fill, ratio));
763 }
764
765 Emit("COMPARE", policies.Count, "weeks", simulation.Demands.Count);
766 int rank = 0;
767 foreach (var row in results.OrderBy(r => r.Cost).ThenBy(r => r.Original))
768 Emit(++rank, row.Label, "total", Money(row.Cost),
769 "fill", row.Fill, "bullwhip", row.Ratio);
770 }

772 void Optimize(IReadOnlyList<string> args)
773 {
774 List<int> indices = Target(args[0]);
775 int low = Whole(args[1]), high = Whole(args[2]);
776 if (low > high) throw new CommandError($"bad range {args[1]} {args[2]}");
777 if (high - low > 200) throw new CommandError("range too large");
778
779 long bestCost = long.MaxValue, bestTotal = 0;
780 int bestStock = low;
781 for (int stock = low; stock <= high; ++stock)
782 {
783 Policy[] policies = simulation.Stages.Select(stage => stage.Policy).ToArray();
784 foreach (int index in indices) policies[index] = Policy.BaseStock(stock);
785 Simulation trial = Trial(policies);
786 long[] costs = trial.TotalCosts();
787 long objective = indices.Sum(index => costs[index]);
788 if (objective < bestCost)
789 {
790 bestCost = objective;
791 bestTotal = costs.Sum();
792 bestStock = stock;
793 }
794 }
795 Emit("OPTIMIZE", args[0], "BASESTOCK", bestStock,
796 "cost", Money(bestCost), "total", Money(bestTotal));
797 }

799 void Command(string command, IReadOnlyList<string> a)
800 {
801 Simulation s = simulation;
802 switch (command)
803 {
804 case "DEMAND":
805 Demand(a.Select(Num).ToList());
806 break;
807 case "GENERATE":
808 Generate(a);
809 break;
810 case "RUN":
811 {
812 if (s.Week == s.Demands.Count)
813 throw new CommandError($"horizon reached at week {s.Demands.Count}");
814 int first = s.Week + 1;
815 int count = Math.Min(Whole(a[0]), s.Demands.Count - s.Week);
816 long cost = 0;
817 for (int step = 0; step < count; ++step)
818 {
819 WeekRecord[] records = s.Step();
820 for (int i = 0; i < 4; ++i)
821 {
822 WeekRecord record = records[i];
823 cost += record.Cost;
824 if (record.Before == 0 && record.Backlog > 0)
825 Emit("STOCKOUT", Names[i], "week", s.Week,
826 "backlog", record.Backlog);
827 else if (record.Before > 0 && record.Backlog == 0)
828 Emit("RECOVER", Names[i], "week", s.Week);
829 }
830 }
831 Emit("RUN", $"{first}-{s.Week}", "demand",
832 s.Demands.Skip(first - 1).Take(s.Week - first + 1).Sum(),
833 "cost", Money(cost));
834 if (s.Week == s.Demands.Count) Emit("HORIZON", s.Week);
835 break;
836 }
837 case "RESET":
838 Initialize();
839 Emit("RESET", "week", 0);
840 break;
841 case "SAVE":
842 snapshots[a[0]] = s.Copy();
843 Emit("SAVE", a[0], "week", s.Week);
844 break;
845 case "LOAD":
846 {
847 if (!snapshots.TryGetValue(a[0], out Simulation? saved))
848 throw new CommandError("no snapshot " + a[0]);
849 simulation = saved.Copy();
850 Emit("LOAD", a[0], "week", simulation.Week);
851 break;
852 }
853 case "SET":
854 {
855 List<int> targets = Target(a[0]);
856 Policy policy = Policy.Parse(a.Skip(1).ToArray());
857 foreach (int index in targets) s.SetPolicy(index, policy);
858 Emit("SET", a[0], policy.Notation);
859 break;
860 }
861 case "BATCH":
862 {
863 int index = StageIndex(a[0]);
864 int lot = Whole(a[1]);
865 if (lot < 1 || lot > 100)
866 throw new CommandError("invalid batch size " + a[1]);
867 s.Stages[index].Lot = lot;
868 Emit("BATCH", a[0], lot);
869 break;
870 }
871 case "CAPACITY":
872 {
873 int index = a[0] == "PLANT" ? -1 : StageIndex(a[0]);
874 long? capacity = a[1] == "OFF" ? null : Num(a[1]);
875 if (capacity > 10000)
876 throw new CommandError("invalid capacity " + a[1]);
877 if (index < 0) s.PlantCapacity = capacity;
878 else s.Stages[index].Capacity = capacity;
879 Emit("CAPACITY", a[0], a[1]);
880 break;
881 }
882 case "COST":
883 {
884 int index = StageIndex(a[0]);
885 long holding = Num(a[1]), backlog = Num(a[2]);
886 if (holding > 1000 || backlog > 1000)
887 throw new CommandError("invalid cost");
888 s.Stages[index].HoldingRate = holding;
889 s.Stages[index].BacklogRate = backlog;
890 Emit("COST", a[0], "holding", holding, "backlog", backlog);
891 break;
892 }
893 case "VISIBILITY":
894 if (a[0] is not ("ON" or "OFF"))
895 throw new CommandError("unknown visibility " + a[0]);
896 s.Visibility = a[0];
897 Emit("VISIBILITY", a[0]);
898 break;
899 case "SHOCK":
900 {
901 int week = Whole(a[0]);
902 long demand = Num(a[1]);
903 if (week < 1 || week > s.Demands.Count)
904 throw new CommandError($"week {a[0]} outside horizon");
905 if (week <= s.Week)
906 throw new CommandError($"week {a[0]} already simulated");
907 long previous = s.Demands[week - 1];
908 s.Demands[week - 1] = demand;
909 Emit("SHOCK", "week", week, previous, "->", demand);
910 break;
911 }
912 case "STATUS":
913 Status();
914 break;
915 case "SETTINGS":
916 Emit("SETTINGS");
917 for (int i = 0; i < 4; ++i)
918 {
919 Stage stage = s.Stages[i];
920 Initial initial = initials[i];
921 Emit(Names[i], "holding", stage.HoldingRate, "backlog", stage.BacklogRate,
922 "batch", stage.Lot, "capacity", stage.Capacity?.ToString() ?? "OFF",
923 "delays", initial.ShippingDelay, initial.OrderDelay);
924 }
925 Emit("PLANT", "capacity", s.PlantCapacity?.ToString() ?? "OFF");
926 Emit("VISIBILITY", s.Visibility);
927 break;
928 case "WEEK":
929 {
930 RequireWeeks();
931 int week = Whole(a[0]);
932 if (week < 1 || week > s.Week)
933 throw new CommandError("bad week " + a[0]);
934 ShowWeek(week);
935 break;
936 }
937 case "TABLE":
938 {
939 int index = StageIndex(a[0]);
940 RequireWeeks();
941 var (first, last) = Range(a.Skip(1).ToArray(), s.Week);
942 Table(index, first, last);
943 break;
944 }
945 case "CHAIN":
946 {
947 RequireWeeks();
948 var (first, last) = Range(a, s.Week);
949 Emit("CHAIN", $"{first}-{last}");
950 Emit("week".PadLeft(4) + " " + "demand".PadLeft(6) + " " +
951 string.Join(" ", Names.Select(name => name.PadLeft(11))));
952 for (int week = first; week <= last; ++week)
953 {
954 string row = week.ToString().PadLeft(4) + " " +
955 s.Demands[week - 1].ToString().PadLeft(6) + " " +
956 string.Join(" ", s.Stages.Select(stage =>
957 stage.Records[week - 1].Order.ToString().PadLeft(11)));
958 Emit(row);
959 }
960 break;
961 }
962 case "DEMANDS":
963 {
964 var (first, last) = Range(a, s.Demands.Count);
965 List<long> values = s.Demands.Skip(first - 1).Take(last - first + 1).ToList();
966 Emit("DEMANDS", $"{first}-{last}", "total", values.Sum(),
967 "min", values.Min(), "max", values.Max());
968 Emit(string.Join(" ", values));
969 break;
970 }
971 case "COSTS":
972 Costs();
973 break;
974 case "SUMMARY":
975 RequireWeeks();
976 Emit("SUMMARY", "weeks", $"1-{s.Week}");
977 for (int i = 0; i < 4; ++i)
978 {
979 List<WeekRecord> records = s.Stages[i].Records;
980 long[] orders = records.Select(r => r.Order).ToArray();
981 Emit(Names[i], "avg_inv", Decimal(records.Sum(r => r.Inventory), s.Week, 2),
982 "avg_back", Decimal(records.Sum(r => r.Backlog), s.Week, 2),
983 "orders", orders.Sum(), "min", orders.Min(), "max", orders.Max(),
984 "zero_weeks", orders.Count(value => value == 0));
985 }
986 break;
987 case "BALANCE":
988 {
989 Emit("BALANCE", "week", s.Week);
990 for (int i = 0; i < 4; ++i)
991 {
992 Stage stage = s.Stages[i];
993 Emit(Names[i], "start", initials[i].Inventory,
994 "received", stage.Records.Sum(r => r.Received),
995 "shipped", stage.Records.Sum(r => r.Shipped),
996 "end", stage.Inventory, "owed", stage.Backlog);
997 }
998 long initialGoods = initials.Sum(i => i.Inventory + i.Quantity * i.ShippingDelay);
999 Emit("SYSTEM", "initial", initialGoods,
1000 "produced", s.PlantRecords.Sum(r => r.Produced),
1001 "delivered", s.Stages[0].Records.Sum(r => r.Shipped),
1002 "stock", s.Stages.Sum(stage => stage.Inventory),
1003 "transit", s.Stages.Sum(stage => stage.Shipping.Sum()));
1004 break;
1005 }
1006 case "FILLRATE":
1007 RequireWeeks();
1008 Emit("FILLRATE", "weeks", $"1-{s.Week}");
1009 for (int i = 0; i < 4; ++i)
1010 {
1011 List<WeekRecord> records = s.Stages[i].Records;
1012 var (ordered, ontime, fill) = FillText(records);
1013 int stockouts = records.Count(record => record.Backlog > 0);
1014 Emit(Names[i], "fill", fill, "ontime", ontime, "ordered", ordered,
1015 "stockout_weeks", stockouts, "service",
1016 Decimal((s.Week - stockouts) * 100, s.Week, 2) + "%");
1017 }
1018 break;
1019 case "BULLWHIP":
1020 RequireWeeks();
1021 Bullwhip();
1022 break;
1023 case "PEAKS":
1024 RequireWeeks();
1025 Peaks();
1026 break;
1027 case "LEADTIME":
1028 {
1029 int index = StageIndex(a[0]);
1030 RequireWeeks();
1031 Leadtime(index);
1032 break;
1033 }
1034 case "PLOT":
1035 case "HISTOGRAM":
1036 {
1037 int index = StageIndex(a[0]);
1038 CheckMode(a[1]);
1039 if (command == "HISTOGRAM")
1040 {
1041 int width = Whole(a[2]);
1042 if (width < 1 || width > 100)
1043 throw new CommandError("invalid width " + a[2]);
1044 }
1045 RequireWeeks();
1046 if (command == "PLOT")
1047 {
1048 var (first, last) = Range(a.Skip(2).ToArray(), s.Week);
1049 Plot(index, a[1], first, last);
1050 }
1051 else Histogram(index, a[1], Whole(a[2]));
1052 break;
1053 }
1054 case "COMPARE":
1055 Compare(a);
1056 break;
1057 case "OPTIMIZE":
1058 Optimize(a);
1059 break;
1060 default:
1061 throw new CommandError("unknown command " + command);
1062 }
1063 }

1065 public void Execute(string line)
1066 {
1067 string[] fields = line.Split(' ');
1068 string command = fields[0];
1069 try
1070 {
1071 Command(command, fields.Skip(1).ToArray());
1072 }
1073 catch (CommandError error)
1074 {
1075 Emit("ERROR", error.Message);
1076 }
1077 }
