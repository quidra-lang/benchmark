1NAMES = %w[RETAILER WHOLESALER DISTRIBUTOR FACTORY].freeze
2PARAMETERS = {'FIXED' => 1, 'PASS' => 0, 'BASESTOCK' => 1, 'MINMAX' => 2,
3 'MOVAVG' => 2, 'SMOOTH' => 2, 'STERMAN' => 4}.freeze
4MODES = {'ORDERS' => :order, 'INV' => :inventory, 'BACK' => :backlog}.freeze
5class CommandError < StandardError; end

7def money(value)
8 format('%d.%02d', value / 100, value % 100)
9end
10
11def decimal(numerator, denominator, places)
12 scale = 10**places
13 value = numerator * scale / denominator
14 format("%d.%0#{places}d", value / scale, value % scale)
15end
16
17def variation(values)
18 values.length * values.sum { |v| v * v } - values.sum**2
19end
20
21def parse_policy(tokens)
22 name = tokens[0]
23 raise CommandError, "unknown policy #{name}" unless PARAMETERS.key?(name)
24 count = PARAMETERS[name]
25 raise CommandError, "#{name} needs #{count} parameters" if tokens.length - 1 != count
26 values = tokens.drop(1).each_with_index.map { |t, i| name == 'STERMAN' && i < 3 ? t.delete('.').to_i : t.to_i }
27 valid = case name
28 when 'FIXED' then (0..1000).cover?(values[0])
29 when 'PASS' then true
30 when 'BASESTOCK' then (0..10000).cover?(values[0])
31 when 'MINMAX' then 0 <= values[0] && values[0] < values[1] && values[1] <= 10000
32 when 'MOVAVG' then (1..52).cover?(values[0]) && (0..10000).cover?(values[1])
33 when 'SMOOTH' then (1..10).cover?(values[0]) && (0..10000).cover?(values[1])
34 else values.first(3).all? { |v| (0..100).cover?(v) } && (0..10000).cover?(values[3])
35 end
36 raise CommandError, "invalid parameter for #{name}" unless valid
37 {name: name, args: values, text: tokens.join(' ')}
38end

40class Simulation
41 attr_accessor :week, :demands, :stages, :plant_backlog, :plant_capacity, :plant_records, :visibility
42
43 def initialize(initials, demands)
44 @initials = initials
45 @week = 0
46 @demands = demands.dup
47 @stages = initials.map do |i|
48 {
49 inventory: i[:inventory], backlog: 0,
50 shipping: Array.new(i[:shipping], i[:quantity]), orders: Array.new(i[:orders], i[:quantity]),
51 policy: i[:policy], expected: 100 * i[:quantity], holding: i[:holding],
52 backlog_rate: i[:backlog], lot: 1, capacity: nil, records: []
53 }
54 end
55 @plant_backlog = 0
56 @plant_capacity = nil
57 @plant_records = []
58 @visibility = 'OFF'
59 end
60
61 def signal(index, week, current = nil)
62 return @demands[week - 1] if @visibility == 'ON'
63 return current[index] if current && week == @week + 1
64 @stages[index][:records][week - 1][:incoming]
65 end
66
67 def set_policy(index, policy)
68 stage = @stages[index]
69 stage[:policy] = policy
70 if policy[:name] == 'STERMAN'
71 stage[:expected] = 100 * (@week.positive? ? signal(index, @week) : @initials[index][:quantity])
72 end
73 end
74
75 def positions
76 supply = @stages.each_with_index.map do |s, i|
77 s[:orders].sum + (i < 3 ? @stages[i + 1][:backlog] : @plant_backlog) + s[:shipping].sum
78 end
79 [supply, @stages.each_with_index.map { |s, i| s[:inventory] + supply[i] - s[:backlog] }]
80 end
81
82 def policy_order(index, incoming, line, position)
83 s = @stages[index]
84 name = s[:policy][:name]
85 a = s[:policy][:args]
86 week = @week + 1
87 signal = signal(index, week, incoming)
88 case name
89 when 'FIXED' then a[0]
90 when 'PASS' then incoming[index]
91 when 'BASESTOCK' then [0, a[0] - position].max
92 when 'MINMAX' then position <= a[0] ? a[1] - position : 0
93 when 'MOVAVG'
94 count = [a[0], week].min
95 forecast = ((week - count + 1)..week).sum { |u| signal(index, u, incoming) } / count
96 delay = @initials[index][:shipping] + @initials[index][:orders]
97 [0, a[1] + forecast * delay - position].max
98 when 'SMOOTH'
99 [0, signal + (a[1] - position).div(a[0])].max
100 else
101 theta, alpha, beta, target = a
102 s[:expected] = (100 * theta * signal + (100 - theta) * s[:expected]) / 100
103 net = s[:inventory] - s[:backlog]
104 [0, (100 * s[:expected] + alpha * (100 * (target - net) - beta * line)).div(10000)].max
105 end
106 end
107
108 def step
109 received = @stages.map do |s|
110 quantity = s[:shipping].shift
111 s[:inventory] += quantity
112 quantity
113 end
114 incoming = [@demands[@week], *@stages.first(3).map { |s| s[:orders].shift }]
115 plant_incoming = @stages[3][:orders].shift
116 records = []
117 @stages.each_with_index do |s, i|
118 before = s[:backlog]
119 requested = before + incoming[i]
120 shipped = [s[:inventory], requested, s[:capacity] || requested].min
121 s[:inventory] -= shipped
122 s[:backlog] = requested - shipped
123 @stages[i - 1][:shipping] << shipped if i.positive?
124 records << {
125 received: received[i], incoming: incoming[i], shipped: shipped, before: before,
126 ontime: [[0, shipped - before].max, incoming[i]].min,
127 inventory: s[:inventory], backlog: s[:backlog]
128 }
129 end
130 requested = @plant_backlog + plant_incoming
131 produced = [requested, @plant_capacity || requested].min
132 @plant_backlog = requested - produced
133 @stages[3][:shipping] << produced
134 supply, positions = positions()
135 orders = @stages.each_index.map { |i| policy_order(i, incoming, supply[i], positions[i]) }
136 @stages.each_with_index do |s, i|
137 quantity = (orders[i] + s[:lot] - 1) / s[:lot] * s[:lot]
138 s[:orders] << quantity
139 r = records[i]
140 r[:order] = quantity
141 r[:holding] = s[:holding] * s[:inventory]
142 r[:backlog_cost] = s[:backlog_rate] * s[:backlog]
143 r[:cost] = r[:holding] + r[:backlog_cost]
144 s[:records] << r
145 end
146 @plant_records << {incoming: plant_incoming, produced: produced, backlog: @plant_backlog}
147 @week += 1
148 records
149 end
150
151 def costs
152 @stages.map { |s| s[:records].sum { |r| r[:cost] } }
153 end
154end

159 def initialize(settings)
160 @initials = settings.map do |a|
161 {
162 inventory: a[1].to_i, quantity: a[2].to_i, shipping: a[3].to_i, orders: a[4].to_i,
163 holding: a[5].to_i, backlog: a[6].to_i, policy: parse_policy(a.drop(7))

173 @output << parts.join(' ')

179 index

183 text == 'ALL' ? [0, 1, 2, 3] : [stage_index(text)]

191 return [1, limit] if args.empty?
192 first, last = args.map(&:to_i)

199 end

202 @sim.stages[index][:records].map { |r| mode == 'NET' ? r[:inventory] - r[:backlog] : r[MODES[mode]] }

206 @base = demands.dup unless demands.nil?

210 def trial(policies)
211 current = @sim
212 trial = Simulation.new(@initials, current.demands)
213 trial.visibility = current.visibility
214 trial.plant_capacity = current.plant_capacity
215 trial.stages.each_with_index do |s, i|
216 old = current.stages[i]
217 s[:lot] = old[:lot]
218 s[:capacity] = old[:capacity]
219 s[:holding] = old[:holding]
220 s[:backlog_rate] = old[:backlog_rate]
221 trial.set_policy(i, policies[i])
222 end
223 trial.step while trial.week < trial.demands.length
224 trial
225 end

227 def demand(values)
228 initialize_sim(values)
229 emit('DEMAND', 'weeks', values.length, 'total', values.sum, 'min', values.min, 'max', values.max)
230 end

232 def generate(a)
233 weeks, base, spread, seed = a.first(4).map(&:to_i)
234 changes = a.drop(4).each_slice(3).map { |kind, week, amount| [kind, week.to_i, amount.to_i] }
235 changes.each do |_, week, _|
236 raise CommandError, "week #{week} outside horizon" unless (1..weeks).cover?(week)
237 end
238 levels = (1..weeks).map do |week|
239 level = base
240 changes.each do |kind, start, amount|
241 if (kind == 'STEP' && week >= start) || (kind == 'PULSE' && week == start)
242 level += amount
243 elsif kind == 'RAMP' && week >= start
244 level += amount * (week - start + 1)
245 end
246 end
247 raise CommandError, "negative level at week #{week}" if level.negative?
248 level
249 end
250 demands = levels.map do |level|
251 seed = 48271 * seed % 2147483647
252 level + seed % spread
253 end
254 demand(demands)
255 end

257 def status
258 s = @sim
259 emit('STATUS', 'week', s.week, 'of', s.demands.length)
260 supply, positions = s.positions
261 s.stages.each_with_index do |stage, i|
262 parts = [NAMES[i], 'inv', stage[:inventory], 'back', stage[:backlog],
263 'transit', stage[:shipping].join(','), 'pending', stage[:orders].join(','),
264 'supply', supply[i], 'position', positions[i], 'policy', stage[:policy][:text]]
265 parts.push('expect', money(stage[:expected])) if stage[:policy][:name] == 'STERMAN'
266 emit(*parts)
267 end
268 emit('PLANT', 'back', s.plant_backlog)
269 end

271 def fill(records)
272 ordered = records.sum { |r| r[:incoming] }
273 ontime = records.sum { |r| r[:ontime] }
274 [ordered, ontime, ordered.zero? ? 'n/a' : decimal(ontime * 100, ordered, 2) + '%']
275 end

277 def peaks
278 s = @sim
279 emit('PEAKS', 'weeks', "1-#{s.week}")
280 s.stages.each_with_index do |stage, i|
281 inventories = stage[:records].map { |r| r[:inventory] }
282 backlogs = stage[:records].map { |r| r[:backlog] }
283 mi = inventories.max
284 mb = backlogs.max
285 run = 0
286 longest = 0
287 start = 0
288 best_start = 0
289 best_end = 0
290 backlogs.each_with_index do |backlog, j|
291 if backlog.positive?
292 start = j + 1 if run.zero?
293 run += 1
294 if run > longest
295 longest = run
296 best_start = start
297 best_end = j + 1
298 end
299 else
300 run = 0
301 end
302 end
303 emit(NAMES[i], 'max_inv', mi, 'week', mi.positive? ? inventories.index(mi) + 1 : '-',
304 'max_back', mb, 'week', mb.positive? ? backlogs.index(mb) + 1 : '-',
305 'longest_stockout', longest.positive? ? "#{longest} weeks #{best_start}-#{best_end}" : 0)
306 end
307 end

309 def leadtime(index)
310 records = @sim.stages[index][:records]
311 initial = @initials[index]
312 nominal = initial[:shipping] + initial[:orders]
313 received = 0
314 cumulative = records.map do |r|
315 received += r[:received]
316 received - nominal * initial[:quantity]
317 end
318 ordered = 0
319 count = 0
320 leads = []
321 records.each_with_index do |r, t|
322 ordered += r[:order]
323 next if r[:order].zero?
324 count += 1
325 ((t + 1)...records.length).each do |u|
326 if cumulative[u] >= ordered
327 leads << u - t
328 break
329 end
330 end
331 end
332 emit('LEADTIME', NAMES[index], 'nominal', nominal, 'orders', count, 'completed', leads.length,
333 'open', count - leads.length, 'min', leads.empty? ? '-' : leads.min,
334 'avg', leads.empty? ? '-' : decimal(leads.sum, leads.length, 2), 'max', leads.empty? ? '-' : leads.max)
335 end

337 def plot(index, mode, first, last)
338 values = values(index, mode)[(first - 1)...last]
339 maximum = values.max
340 if mode == 'NET'
341 emit('PLOT', NAMES[index], mode, "#{first}-#{last}", 'min', values.min, 'max', maximum)
342 scale = values.map(&:abs).max
343 values.each_with_index do |value, i|
344 length = scale.zero? ? 0 : value.abs * 25 / scale
345 left = value.negative? ? ' ' * (25 - length) + '-' * length : ' ' * 25
346 right = value.positive? ? '+' * length : ''
347 emit(format('%3d', first + i) + " #{left}|#{right} #{value}")
348 end
349 else
350 emit('PLOT', NAMES[index], mode, "#{first}-#{last}", 'max', maximum)
351 values.each_with_index do |value, i|
352 length = maximum.zero? ? 0 : value * 50 / maximum
353 emit(format('%3d', first + i) + ' |' + '#' * length + " #{value}")
354 end
355 end
356 end

358 def command(cmd, a)
359 s = @sim
360 case cmd
361 when 'DEMAND'
362 demand(a.map(&:to_i))
363 when 'GENERATE'
364 generate(a)
365 when 'RUN'
366 raise CommandError, "horizon reached at week #{s.demands.length}" if s.week == s.demands.length
367 first = s.week + 1
368 count = [a[0].to_i, s.demands.length - s.week].min
369 cost = 0
370 count.times do
371 s.step.each_with_index do |r, i|
372 cost += r[:cost]
373 if r[:before].zero? && r[:backlog].positive?
374 emit('STOCKOUT', NAMES[i], 'week', s.week, 'backlog', r[:backlog])
375 elsif r[:before].positive? && r[:backlog].zero?
376 emit('RECOVER', NAMES[i], 'week', s.week)
377 end
378 end
379 end
380 emit('RUN', "#{first}-#{s.week}", 'demand', s.demands[(first - 1)...s.week].sum, 'cost', money(cost))
381 emit('HORIZON', s.week) if s.week == s.demands.length
382 when 'RESET'
383 initialize_sim
384 emit('RESET', 'week', 0)
385 when 'SAVE'
386 @snapshots[a[0]] = Marshal.dump(s)
387 emit('SAVE', a[0], 'week', s.week)
388 when 'LOAD'
389 raise CommandError, "no snapshot #{a[0]}" unless @snapshots.key?(a[0])
390 @sim = Marshal.load(@snapshots[a[0]])
391 emit('LOAD', a[0], 'week', @sim.week)
392 when 'SET'
393 indices = target(a[0])
394 policy = parse_policy(a.drop(1))
395 indices.each { |i| s.set_policy(i, policy) }
396 emit('SET', a[0], policy[:text])
397 when 'BATCH'
398 i = stage_index(a[0])
399 lot = a[1].to_i
400 raise CommandError, "invalid batch size #{a[1]}" unless (1..100).cover?(lot)
401 s.stages[i][:lot] = lot
402 emit('BATCH', a[0], lot)
403 when 'CAPACITY'
404 i = a[0] == 'PLANT' ? nil : stage_index(a[0])
405 capacity = a[1] == 'OFF' ? nil : a[1].to_i
406 raise CommandError, "invalid capacity #{a[1]}" if capacity && capacity > 10000
407 if i.nil?
408 s.plant_capacity = capacity
409 else
410 s.stages[i][:capacity] = capacity
411 end
412 emit('CAPACITY', a[0], a[1])
413 when 'COST'
414 i = stage_index(a[0])
415 holding, backlog = a.drop(1).map(&:to_i)
416 raise CommandError, 'invalid cost' if holding > 1000 || backlog > 1000
417 s.stages[i][:holding] = holding
418 s.stages[i][:backlog_rate] = backlog
419 emit('COST', a[0], 'holding', holding, 'backlog', backlog)
420 when 'VISIBILITY'
421 raise CommandError, "unknown visibility #{a[0]}" unless %w[ON OFF].include?(a[0])
422 s.visibility = a[0]
423 emit('VISIBILITY', a[0])
424 when 'SHOCK'
425 week, demand = a.map(&:to_i)
426 raise CommandError, "week #{a[0]} outside horizon" unless (1..s.demands.length).cover?(week)
427 raise CommandError, "week #{a[0]} already simulated" if week <= s.week
428 old = s.demands[week - 1]
429 s.demands[week - 1] = demand
430 emit('SHOCK', 'week', week, old, '->', demand)
431 when 'STATUS'
432 status
433 when 'SETTINGS'
434 emit('SETTINGS')
435 s.stages.each_with_index do |stage, i|
436 emit(NAMES[i], 'holding', stage[:holding], 'backlog', stage[:backlog_rate], 'batch', stage[:lot],
437 'capacity', stage[:capacity].nil? ? 'OFF' : stage[:capacity],
438 'delays', @initials[i][:shipping], @initials[i][:orders])
439 end
440 emit('PLANT', 'capacity', s.plant_capacity.nil? ? 'OFF' : s.plant_capacity)
441 emit('VISIBILITY', s.visibility)
442 when 'WEEK'
443 require_weeks
444 week = a[0].to_i
445 raise CommandError, "bad week #{a[0]}" unless (1..s.week).cover?(week)
446 emit('WEEK', week, 'demand', s.demands[week - 1])
447 s.stages.each_with_index do |stage, i|
448 r = stage[:records][week - 1]
449 emit(NAMES[i], 'recv', r[:received], 'in', r[:incoming], 'ship', r[:shipped],
450 'inv', r[:inventory], 'back', r[:backlog], 'order', r[:order], 'cost', money(r[:cost]))
451 end
452 p = s.plant_records[week - 1]
453 emit('PLANT', 'in', p[:incoming], 'made', p[:produced], 'back', p[:backlog])
454 when 'TABLE'
455 i = stage_index(a[0])
456 require_weeks
457 first, last = range_check(a.drop(1), s.week)
458 emit('TABLE', a[0], "#{first}-#{last}")
459 widths = [4, 6, 6, 6, 6, 6, 6, 9]
460 emit(%w[week in ship recv inv back order cost].zip(widths).map { |v, width| v.rjust(width) }.join(' '))
461 (first..last).each do |week|
462 r = s.stages[i][:records][week - 1]
463 values = [week, r[:incoming], r[:shipped], r[:received], r[:inventory], r[:backlog], r[:order], money(r[:cost])]
464 emit(values.zip(widths).map { |v, width| v.to_s.rjust(width) }.join(' '))
465 end
466 when 'CHAIN'
467 require_weeks
468 first, last = range_check(a, s.week)
469 emit('CHAIN', "#{first}-#{last}")
470 emit('week'.rjust(4) + ' ' + 'demand'.rjust(6) + ' ' + NAMES.map { |n| n.rjust(11) }.join(' '))
471 (first..last).each do |week|
472 orders = s.stages.map { |stage| stage[:records][week - 1][:order].to_s.rjust(11) }.join(' ')
473 emit(week.to_s.rjust(4) + ' ' + s.demands[week - 1].to_s.rjust(6) + ' ' + orders)
474 end
475 when 'DEMANDS'
476 first, last = range_check(a, s.demands.length)
477 values = s.demands[(first - 1)...last]
478 emit('DEMANDS', "#{first}-#{last}", 'total', values.sum, 'min', values.min, 'max', values.max)
479 emit(*values)
480 when 'COSTS'
481 emit('COSTS', 'week', s.week)
482 total_holding = 0
483 total_backlog = 0
484 s.stages.each_with_index do |stage, i|
485 holding = stage[:records].sum { |r| r[:holding] }
486 backlog = stage[:records].sum { |r| r[:backlog_cost] }
487 total_holding += holding
488 total_backlog += backlog
489 emit(NAMES[i], 'holding', money(holding), 'backlog', money(backlog), 'total', money(holding + backlog))
490 end
491 emit('TOTAL', 'holding', money(total_holding), 'backlog', money(total_backlog), 'total', money(total_holding + total_backlog))
492 when 'SUMMARY'
493 require_weeks
494 emit('SUMMARY', 'weeks', "1-#{s.week}")
495 s.stages.each_with_index do |stage, i|
496 orders = stage[:records].map { |r| r[:order] }
497 emit(NAMES[i], 'avg_inv', decimal(stage[:records].sum { |r| r[:inventory] }, s.week, 2),
498 'avg_back', decimal(stage[:records].sum { |r| r[:backlog] }, s.week, 2),
499 'orders', orders.sum, 'min', orders.min, 'max', orders.max, 'zero_weeks', orders.count(0))
500 end
501 when 'BALANCE'
502 emit('BALANCE', 'week', s.week)
503 s.stages.each_with_index do |stage, i|
504 emit(NAMES[i], 'start', @initials[i][:inventory], 'received', stage[:records].sum { |r| r[:received] },
505 'shipped', stage[:records].sum { |r| r[:shipped] }, 'end', stage[:inventory], 'owed', stage[:backlog])
506 end
507 initial = @initials.sum { |i| i[:inventory] + i[:quantity] * i[:shipping] }
508 emit('SYSTEM', 'initial', initial, 'produced', s.plant_records.sum { |r| r[:produced] },
509 'delivered', s.stages[0][:records].sum { |r| r[:shipped] }, 'stock', s.stages.sum { |stage| stage[:inventory] },
510 'transit', s.stages.sum { |stage| stage[:shipping].sum })
511 when 'FILLRATE'
512 require_weeks
513 emit('FILLRATE', 'weeks', "1-#{s.week}")
514 s.stages.each_with_index do |stage, i|
515 ordered, ontime, text = fill(stage[:records])
516 stockouts = stage[:records].count { |r| r[:backlog].positive? }
517 emit(NAMES[i], 'fill', text, 'ontime', ontime, 'ordered', ordered, 'stockout_weeks', stockouts,
518 'service', decimal((s.week - stockouts) * 100, s.week, 2) + '%')
519 end
520 when 'BULLWHIP'
521 require_weeks
522 n = s.week
523 demands = s.demands.first(n)
524 dv = variation(demands)
525 emit('BULLWHIP', 'weeks', "1-#{n}", 'demand', 'mean', decimal(demands.sum, n, 4), 'var', decimal(dv, n * n, 4))
526 previous = dv
527 s.stages.each_with_index do |stage, i|
528 values = stage[:records].map { |r| r[:order] }
529 v = variation(values)
530 emit(NAMES[i], 'mean', decimal(values.sum, n, 4), 'var', decimal(v, n * n, 4),
531 'ratio', dv.zero? ? 'n/a' : decimal(v, dv, 4), 'step', previous.zero? ? 'n/a' : decimal(v, previous, 4))
532 previous = v
533 end
534 when 'PEAKS'
535 require_weeks
536 peaks
537 when 'LEADTIME'
538 i = stage_index(a[0])
539 require_weeks
540 leadtime(i)
541 when 'PLOT', 'HISTOGRAM'
542 i = stage_index(a[0])
543 mode_check(a[1])
544 width = cmd == 'HISTOGRAM' ? a[2].to_i : 1
545 raise CommandError, "invalid width #{a[2]}" if cmd == 'HISTOGRAM' && !(1..100).cover?(width)
546 require_weeks
547 if cmd == 'PLOT'
548 plot(i, a[1], *range_check(a.drop(2), s.week))
549 else
550 bins = values(i, a[1]).map { |value| value.div(width) }.tally
551 emit('HISTOGRAM', a[0], a[1], 'width', width, 'weeks', "1-#{s.week}")
552 (bins.keys.min..bins.keys.max).each do |bucket|
553 count = bins.fetch(bucket, 0)
554 emit("#{bucket * width}..#{bucket * width + width - 1}", count, *(count.positive? ? ['*' * count] : []))
555 end
556 end
557 when 'COMPARE'
558 policies = []
559 start = 0
560 (0..a.length).each do |finish|
561 next unless finish == a.length || a[finish] == '/'
562 policies << parse_policy(a[start...finish])
563 start = finish + 1
564 end
565 rows = policies.each_with_index.map do |policy, index|
566 trial = trial([policy] * 4)
567 cost = trial.costs.sum
568 text = fill(trial.stages[0][:records])[2]
569 dv = variation(trial.demands)
