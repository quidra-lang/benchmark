50 long total = 0;

52 for (long value : values) {

54 squares += value * value;

56 return values.size() * squares - total * total;

83 static Policy read(List<String> tokens) {
84 String name = tokens.isEmpty() ? "" : tokens.get(0);
85 Integer expected = PARAMETER_COUNTS.get(name);
86 if (expected == null) fail("unknown policy " + name);
87 if (tokens.size() - 1 != expected)
88 fail(name + " needs " + expected + " parameters");
89 long[] values = new long[expected];
90 for (int i = 0; i < expected; i++) {
91 String text = tokens.get(i + 1);
92 values[i] = num(name.equals("STERMAN") && i < 3
93 ? text.replace(".", "") : text);
94 }
95 boolean valid;
96 switch (name) {
97 case "FIXED":
98 valid = values[0] >= 0 && values[0] <= 1000;
99 break;
100 case "PASS":
101 valid = true;
102 break;
103 case "BASESTOCK":
104 valid = values[0] >= 0 && values[0] <= 10000;
105 break;
106 case "MINMAX":
107 valid = values[0] >= 0 && values[0] < values[1] &&
108 values[1] <= 10000;
109 break;
110 case "MOVAVG":
111 valid = values[0] >= 1 && values[0] <= 52 &&
112 values[1] >= 0 && values[1] <= 10000;
113 break;
114 case "SMOOTH":
115 valid = values[0] >= 1 && values[0] <= 10 &&
116 values[1] >= 0 && values[1] <= 10000;
117 break;
118 default:
119 valid = values[3] >= 0 && values[3] <= 10000;
120 for (int i = 0; i < 3; i++)
121 valid &= values[i] >= 0 && values[i] <= 100;
122 }
123 if (!valid) fail("invalid parameter for " + name);
124 return new Policy(name, values, String.join(" ", tokens));
125 }

272 long[][] positions() {
273 long[][] result = new long[2][4];
274 for (int i = 0; i < 4; i++) {
275 Stage stage = stages[i];
276 long upstreamBacklog = i < 3 ? stages[i + 1].backlog : plantBacklog;
277 long line = sum(stage.orders) + upstreamBacklog + sum(stage.shipping);

284 long policyOrder(int index, long[] incoming, long line, long position) {
285 Stage stage = stages[index];
286 Policy policy = stage.policy;
287 long[] a = policy.values;
288 int t = week + 1;
289 long demand = signal(index, t, incoming);
290 switch (policy.name) {
291 case "FIXED":
292 return a[0];
293 case "PASS":
294 return incoming[index];
295 case "BASESTOCK":
296 return Math.max(0L, a[0] - position);
297 case "MINMAX":
298 return position <= a[0] ? a[1] - position : 0;
299 case "MOVAVG": {
300 int count = (int) Math.min(a[0], t);
301 long forecastSum = 0;
302 for (int u = t - count + 1; u <= t; u++)
303 forecastSum += signal(index, u, incoming);
304 long forecast = forecastSum / count;
305 Initial initial = initials.get(index);
306 int delay = initial.shipping + initial.orders;
307 return Math.max(0L, a[1] + forecast * delay - position);
308 }
309 case "SMOOTH":
310 return Math.max(0L, demand + Math.floorDiv(a[1] - position, a[0]));
311 default: {
312 long theta = a[0], alpha = a[1], beta = a[2], target = a[3];
313 stage.expected = Math.floorDiv(
314 100 * theta * demand + (100 - theta) * stage.expected, 100);
315 long net = stage.inventory - stage.backlog;
316 return Math.max(0L, Math.floorDiv(
317 100 * stage.expected + alpha * (100 * (target - net) - beta * line),
318 10000));
319 }
320 }
321 }

323 Record[] step() {
324 long[] received = new long[4];
325 long[] incoming = new long[4];
326 for (int i = 0; i < 4; i++) {
327 Stage stage = stages[i];
328 received[i] = stage.shipping.remove(0);
329 stage.inventory += received[i];
330 }
331
332 incoming[0] = demands.get(week);
333 for (int i = 1; i < 4; i++)
334 incoming[i] = stages[i - 1].orders.remove(0);
335 long plantIncoming = stages[3].orders.remove(0);
336
337 Record[] newRecords = new Record[4];
338 for (int i = 0; i < 4; i++) {
339 Stage stage = stages[i];
340 Record record = new Record();
341 record.received = received[i];
342 record.incoming = incoming[i];
343 record.before = stage.backlog;
344 long requested = stage.backlog + incoming[i];
345 long shipped = Math.min(stage.inventory, requested);
346 if (stage.capacity != null)
347 shipped = Math.min(shipped, stage.capacity);
348 stage.inventory -= shipped;
349 stage.backlog = requested - shipped;
350 if (i > 0) stages[i - 1].shipping.add(shipped);
351 record.shipped = shipped;
352 record.ontime = Math.min(Math.max(0L, shipped - record.before), incoming[i]);
353 record.inventory = stage.inventory;
354 record.backlog = stage.backlog;
355 newRecords[i] = record;
356 }
357
358 long request = plantBacklog + plantIncoming;
359 long produced = plantCapacity == null ? request : Math.min(request, plantCapacity);
360 plantBacklog = request - produced;
361 stages[3].shipping.add(produced);
362
363 long[][] positions = positions();
364 long[] policyOrders = new long[4];
365 for (int i = 0; i < 4; i++)
366 policyOrders[i] = policyOrder(i, incoming, positions[0][i], positions[1][i]);
367
368 for (int i = 0; i < 4; i++) {
369 Stage stage = stages[i];
370 Record record = newRecords[i];
371 long lot = stage.lot;
372 record.order = ((policyOrders[i] + lot - 1) / lot) * lot;
373 stage.orders.add(record.order);
374 record.holding = stage.holding * stage.inventory;
375 record.backlogCost = stage.backlogRate * stage.backlog;
376 record.cost = record.holding + record.backlogCost;
377 stage.records.add(record);
378 }
379 plantRecords.add(new PlantRecord(plantIncoming, produced, plantBacklog));
380 week++;
381 return newRecords;
382 }
383
384 long[] totalCosts() {
385 long[] totals = new long[4];
386 for (int i = 0; i < 4; i++)
387 for (Record r : stages[i].records)
388 totals[i] += r.cost;
389 return totals;
390 }
391 }

486 void generate(List<String> arguments) {
487 int weeks = whole(arguments.get(0));
488 long level = num(arguments.get(1)), spread = num(arguments.get(2));
489 long seed = num(arguments.get(3));
490 List<String> changes = arguments.subList(4, arguments.size());
491 for (int i = 0; i < changes.size(); i += 3) {
492 int first = whole(changes.get(i + 1));
493 if (first < 1 || first > weeks)
494 fail("week " + changes.get(i + 1) + " outside horizon");
495 }
496 List<Long> levels = new ArrayList<>();
497 for (int t = 1; t <= weeks; t++) {
498 long current = level;
499 for (int i = 0; i < changes.size(); i += 3) {
500 String kind = changes.get(i);
501 int first = whole(changes.get(i + 1));
502 long amount = num(changes.get(i + 2));
503 if (kind.equals("STEP") && t >= first) current += amount;
504 else if (kind.equals("PULSE") && t == first) current += amount;
505 else if (kind.equals("RAMP") && t >= first)
506 current += amount * (t - first + 1);
507 }
508 if (current < 0) fail("negative level at week " + t);
509 levels.add(current);
510 }
511 List<Long> series = new ArrayList<>();
512 for (long current : levels) {
513 seed = (48271L * seed) % 2147483647L;
514 series.add(current + seed % spread);
515 }
516 demand(series);
517 }

603 emit("BULLWHIP", "weeks", "1-" + n, "demand", "mean",

614 "step", previous == 0 ? "n/a" : decimal(v, previous, 4));
615 previous = v;

643 bestEnd = week;

650 longest + " weeks " + bestStart + "-" + bestEnd;
651 emit(NAMES[i], "max_inv", maxInv, "week", invWeek == 0 ? "-" : invWeek,
652 "max_back", maxBack, "week", backWeek == 0 ? "-" : backWeek,

657 void leadtime(int index) {
658 Stage stage = sim.stages[index];
659 Initial initial = initials.get(index);
660 int nominal = initial.shipping + initial.orders;
661 long[] arrivals = new long[stage.records.size()];
662 long received = -nominal * initial.quantity;
663 for (int i = 0; i < arrivals.length; i++) {
664 received += stage.records.get(i).received;
665 arrivals[i] = received;
666 }
667 long ordered = 0;
668 int count = 0;
669 List<Integer> leads = new ArrayList<>();
670 for (int t = 0; t < stage.records.size(); t++) {
671 long amount = stage.records.get(t).order;
672 ordered += amount;
673 if (amount == 0) continue;
674 count++;
675 for (int u = t + 1; u < stage.records.size(); u++) {
676 if (arrivals[u] >= ordered) {
677 leads.add(u - t);
678 break;
679 }
680 }
681 }
682 String min = "-", avg = "-", max = "-";
683 if (!leads.isEmpty()) {
684 min = String.valueOf(Collections.min(leads));
685 max = String.valueOf(Collections.max(leads));
686 long total = 0;
687 for (int lead : leads) total += lead;
688 avg = decimal(total, leads.size(), 2);
689 }
690 emit("LEADTIME", NAMES[index], "nominal", nominal, "orders", count,
691 "completed", leads.size(), "open", count - leads.size(),
692 "min", min, "avg", avg, "max", max);
693 }

695 void plot(int index, String kind, int first, int last) {
696 List<Long> values = values(index, kind).subList(first - 1, last);
697 long high = Collections.max(values);
698 if (kind.equals("NET")) {
699 long low = Collections.min(values), scale = 0;
700 for (long v : values) scale = Math.max(scale, Math.abs(v));
701 emit("PLOT", NAMES[index], kind, first + "-" + last,
702 "min", low, "max", high);
703 for (int j = 0; j < values.size(); j++) {
704 long v = values.get(j);
705 int length = scale == 0 ? 0 : (int) (Math.abs(v) * 25 / scale);
706 String left = v < 0
707 ? " ".repeat(25 - length) + "-".repeat(length) : " ".repeat(25);
708 String right = v > 0 ? "+".repeat(length) : "";
709 emit(pad(first + j, 3) + " " + left + "|" + right + " " + v);
710 }
711 } else {
712 emit("PLOT", NAMES[index], kind, first + "-" + last, "max", high);
713 for (int j = 0; j < values.size(); j++) {
714 long v = values.get(j);
715 int length = high == 0 ? 0 : (int) (v * 50 / high);
716 emit(pad(first + j, 3) + " |" + "#".repeat(length) + " " + v);
717 }
718 }
719 }

721 void histogram(int index, String kind, int width) {
722 List<Long> values = values(index, kind);
723 TreeMap<Long, Integer> bins = new TreeMap<>();
724 for (long v : values) {
725 long bucket = Math.floorDiv(v, width);
726 bins.merge(bucket, 1, Integer::sum);
727 }
728 emit("HISTOGRAM", NAMES[index], kind, "width", width,
729 "weeks", "1-" + sim.week);
730 for (long b = bins.firstKey(); b <= bins.lastKey(); b++) {
731 int count = bins.getOrDefault(b, 0);
732 String line = (b * width) + ".." + (b * width + width - 1) + " " + count;
733 if (count != 0) line += " " + "*".repeat(count);
734 emit(line);
735 }

738 void command(String command, List<String> a) {
739 Simulation s = sim;
740 switch (command) {
741 case "DEMAND": {
742 List<Long> demands = new ArrayList<>();
743 for (String v : a) demands.add(num(v));
744 demand(demands);
745 break;
746 }
747 case "GENERATE":
748 generate(a);
749 break;
750 case "RUN": {
751 if (s.week == s.demands.size())
752 fail("horizon reached at week " + s.demands.size());
753 int first = s.week + 1;
754 int steps = Math.min(whole(a.get(0)), s.demands.size() - s.week);
755 long cost = 0;
756 for (int k = 0; k < steps; k++) {
757 Record[] records = s.step();
758 for (int i = 0; i < 4; i++) {
759 Record r = records[i];
760 cost += r.cost;
761 if (r.before == 0 && r.backlog > 0)
762 emit("STOCKOUT", NAMES[i], "week", s.week,
763 "backlog", r.backlog);
764 else if (r.before > 0 && r.backlog == 0)
765 emit("RECOVER", NAMES[i], "week", s.week);
766 }
767 }
768 long totalDemand = sum(s.demands.subList(first - 1, s.week));
769 emit("RUN", first + "-" + s.week, "demand", totalDemand, "cost", money(cost));
770 if (s.week == s.demands.size()) emit("HORIZON", s.week);
771 break;
772 }
773 case "RESET":
774 initialize(null);
775 emit("RESET", "week", 0);
776 break;
777 case "SAVE":
778 snapshots.put(a.get(0), new Simulation(s));
779 emit("SAVE", a.get(0), "week", s.week);
780 break;
781 case "LOAD": {
782 Simulation snapshot = snapshots.get(a.get(0));
783 if (snapshot == null) fail("no snapshot " + a.get(0));
784 sim = new Simulation(snapshot);
785 emit("LOAD", a.get(0), "week", sim.week);
786 break;
787 }
788 case "SET": {
789 List<Integer> indices = target(a.get(0));
790 Policy policy = Policy.read(a.subList(1, a.size()));
791 for (int index : indices) s.setPolicy(index, policy);
792 emit("SET", a.get(0), policy.notation);
793 break;
794 }
795 case "BATCH": {
796 int index = stageIndex(a.get(0));
797 int lot = whole(a.get(1));
798 if (lot < 1 || lot > 100)
799 fail("invalid batch size " + a.get(1));
800 s.stages[index].lot = lot;
801 emit("BATCH", a.get(0), lot);
802 break;
803 }
804 case "CAPACITY": {
805 int index = a.get(0).equals("PLANT") ? -1 : stageIndex(a.get(0));
806 Long capacity = a.get(1).equals("OFF") ? null : num(a.get(1));
807 if (capacity != null && capacity > 10000)
808 fail("invalid capacity " + a.get(1));
809 if (index < 0) s.plantCapacity = capacity;
810 else s.stages[index].capacity = capacity;
811 emit("CAPACITY", a.get(0), a.get(1));
812 break;
813 }
814 case "COST": {
815 int index = stageIndex(a.get(0));
816 long holding = num(a.get(1)), backlog = num(a.get(2));
817 if (holding > 1000 || backlog > 1000) fail("invalid cost");
818 s.stages[index].holding = holding;
819 s.stages[index].backlogRate = backlog;
820 emit("COST", a.get(0), "holding", holding, "backlog", backlog);
821 break;
822 }
823 case "VISIBILITY":
824 if (!a.get(0).equals("ON") && !a.get(0).equals("OFF"))
825 fail("unknown visibility " + a.get(0));
826 s.visibility = a.get(0);
827 emit("VISIBILITY", a.get(0));
828 break;
829 case "SHOCK": {
830 int week = whole(a.get(0));
831 long demand = num(a.get(1));
832 if (week < 1 || week > s.demands.size())
833 fail("week " + a.get(0) + " outside horizon");
834 if (week <= s.week)
835 fail("week " + a.get(0) + " already simulated");
836 long old = s.demands.set(week - 1, demand);
837 emit("SHOCK", "week", week, old, "->", demand);
838 break;
839 }
840 case "STATUS":
841 status();
842 break;
843 case "SETTINGS": {
844 emit("SETTINGS");
845 for (int i = 0; i < 4; i++) {
846 Stage stage = s.stages[i];
847 Initial initial = initials.get(i);
848 emit(NAMES[i], "holding", stage.holding, "backlog", stage.backlogRate,
849 "batch", stage.lot, "capacity",
850 stage.capacity == null ? "OFF" : stage.capacity,
851 "delays", initial.shipping, initial.orders);
852 }
853 emit("PLANT", "capacity",
854 s.plantCapacity == null ? "OFF" : s.plantCapacity);
855 emit("VISIBILITY", s.visibility);
856 break;
857 }
858 case "WEEK": {
859 requireWeeks();
860 int week = whole(a.get(0));
861 if (week < 1 || week > s.week)
862 fail("bad week " + a.get(0));
863 showWeek(week);
864 break;
865 }
866 case "TABLE": {
867 int index = stageIndex(a.get(0));
868 requireWeeks();
869 int[] range = range(a.subList(1, a.size()), s.week);
870 table(index, range[0], range[1]);
871 break;
872 }
873 case "CHAIN": {
874 requireWeeks();
875 int[] range = range(a, s.week);
876 emit("CHAIN", range[0] + "-" + range[1]);
877 String header = pad("week", 4) + " " + pad("demand", 6);
878 for (String name : NAMES) header += " " + pad(name, 11);
879 emit(header);
880 for (int week = range[0]; week <= range[1]; week++) {
881 String row = pad(week, 4) + " " + pad(s.demands.get(week - 1), 6);
882 for (Stage stage : s.stages)
883 row += " " + pad(stage.records.get(week - 1).order, 11);
884 emit(row);
885 }
886 break;
887 }
888 case "DEMANDS": {
889 int[] range = range(a, s.demands.size());
890 List<Long> values = s.demands.subList(range[0] - 1, range[1]);
891 emit("DEMANDS", range[0] + "-" + range[1], "total", sum(values),
892 "min", Collections.min(values), "max", Collections.max(values));
893 StringBuilder line = new StringBuilder();
894 for (long v : values) {
895 if (line.length() > 0) line.append(' ');
896 line.append(v);
897 }
898 emit(line.toString());
899 break;
900 }
901 case "COSTS":
902 costs();
903 break;
904 case "SUMMARY": {
905 requireWeeks();
906 emit("SUMMARY", "weeks", "1-" + s.week);
907 for (int i = 0; i < 4; i++) {
908 long inv = 0, back = 0, orders = 0;
909 long least = Long.MAX_VALUE, most = 0;
910 int zeros = 0;
911 for (Record r : s.stages[i].records) {
912 inv += r.inventory;
913 back += r.backlog;
914 orders += r.order;
915 least = Math.min(least, r.order);
916 most = Math.max(most, r.order);
917 if (r.order == 0) zeros++;
918 }
919 emit(NAMES[i], "avg_inv", decimal(inv, s.week, 2),
920 "avg_back", decimal(back, s.week, 2),
921 "orders", orders, "min", least, "max", most,
922 "zero_weeks", zeros);
923 }
924 break;
925 }
926 case "BALANCE": {
927 emit("BALANCE", "week", s.week);
928 for (int i = 0; i < 4; i++) {
929 long received = 0, shipped = 0;
930 for (Record r : s.stages[i].records) {
931 received += r.received;
932 shipped += r.shipped;
933 }
934 emit(NAMES[i], "start", initials.get(i).inventory,
935 "received", received, "shipped", shipped,
936 "end", s.stages[i].inventory, "owed", s.stages[i].backlog);
937 }
938 long initialGoods = 0, produced = 0, delivered = 0, stock = 0, transit = 0;
939 for (Initial init : initials)
940 initialGoods += init.inventory + init.quantity * init.shipping;
941 for (PlantRecord r : s.plantRecords) produced += r.produced;
942 for (Record r : s.stages[0].records) delivered += r.shipped;
943 for (Stage stage : s.stages) {
944 stock += stage.inventory;
945 transit += sum(stage.shipping);
946 }
947 emit("SYSTEM", "initial", initialGoods, "produced", produced,
948 "delivered", delivered, "stock", stock, "transit", transit);
949 break;
950 }
951 case "FILLRATE": {
952 requireWeeks();
953 emit("FILLRATE", "weeks", "1-" + s.week);
954 for (int i = 0; i < 4; i++) {
955 List<Record> records = s.stages[i].records;
956 long[] counts = fillCounts(records);
957 int stockouts = 0;
958 for (Record r : records)
959 if (r.backlog > 0) stockouts++;
960 emit(NAMES[i], "fill", fillText(records),
961 "ontime", counts[1], "ordered", counts[0],
962 "stockout_weeks", stockouts,
963 "service", decimal((long) (s.week - stockouts) * 100, s.week, 2) + "%");
964 }
965 break;
966 }
967 case "BULLWHIP":
968 requireWeeks();
969 bullwhip();
970 break;
971 case "PEAKS":
972 requireWeeks();
973 peaks();
974 break;
975 case "LEADTIME": {
976 int index = stageIndex(a.get(0));
977 requireWeeks();
978 leadtime(index);
979 break;
980 }
981 case "PLOT":
982 case "HISTOGRAM": {
983 int index = stageIndex(a.get(0));
984 mode(a.get(1));
985 if (command.equals("HISTOGRAM")) {
986 int width = whole(a.get(2));
987 if (width < 1 || width > 100)
988 fail("invalid width " + a.get(2));
989 }
990 requireWeeks();
991 if (command.equals("PLOT")) {
992 int[] interval = range(a.subList(2, a.size()), s.week);
993 plot(index, a.get(1), interval[0], interval[1]);
994 } else {
995 histogram(index, a.get(1), whole(a.get(2)));
996 }
997 break;
998 }
999 case "COMPARE": {
1000 List<Policy> policies = new ArrayList<>();
1001 int start = 0;
1002 for (int end = 0; end <= a.size(); end++) {
1003 if (end == a.size() || a.get(end).equals("/")) {
1004 policies.add(Policy.read(a.subList(start, end)));
1005 start = end + 1;
1006 }

1008 class Row {
1009 long cost;
1010 int original;
1011 String label, fill, ratio;
1012 Row(long cost, int original, String label, String fill, String ratio) {
1013 this.cost = cost;
1014 this.original = original;
1015 this.label = label;
1016 this.fill = fill;
1017 this.ratio = ratio;
1018 }
1019 }
1020 List<Row> rows = new ArrayList<>();
1021 for (int index = 0; index < policies.size(); index++) {
1022 Policy policy = policies.get(index);
1023 Policy[] all = {policy, policy, policy, policy};
1024 Simulation trial = trial(all);
1025 long cost = 0;
1026 for (long value : trial.totalCosts()) cost += value;
1027 long dv = variation(trial.demands);
1028 List<Long> factory = new ArrayList<>();
1029 for (Record r : trial.stages[3].records) factory.add(r.order);
1030 long fv = variation(factory);
1031 String ratio = dv == 0 ? "n/a" : decimal(fv, dv, 4);
1032 String fill = fillText(trial.stages[0].records);
1033 rows.add(new Row(cost, index, policy.notation, fill, ratio));
1034 }
1035 rows.sort(Comparator.comparingLong((Row r) -> r.cost)
1036 .thenComparingInt(r -> r.original));
1037 emit("COMPARE", policies.size(), "weeks", s.demands.size());
1038 int rank = 0;
1039 for (Row row : rows)
1040 emit(++rank, row.label, "total", money(row.cost),
1041 "fill", row.fill, "bullwhip", row.ratio);
1042 break;
1043 }
1044 case "OPTIMIZE": {
1045 List<Integer> indices = target(a.get(0));
1046 int low = whole(a.get(1)), high = whole(a.get(2));
1047 if (low > high) fail("bad range " + a.get(1) + " " + a.get(2));
1048 if (high - low > 200) fail("range too large");
1049 long bestCost = Long.MAX_VALUE, bestTotal = 0;
1050 int bestValue = low;
1051 for (int candidate = low; candidate <= high; candidate++) {
1052 Policy[] policies = new Policy[4];
1053 for (int i = 0; i < 4; i++) policies[i] = s.stages[i].policy;
1054 for (int i : indices) policies[i] = Policy.baseStock(candidate);
1055 Simulation trial = trial(policies);
1056 long[] costs = trial.totalCosts();
1057 long objective = 0, total = 0;
1058 for (long cost : costs) total += cost;
1059 for (int i : indices) objective += costs[i];
1060 if (objective < bestCost) {
1061 bestCost = objective;
1062 bestTotal = total;
1063 bestValue = candidate;
1064 }
1065 }
1066 emit("OPTIMIZE", a.get(0), "BASESTOCK", bestValue,
1067 "cost", money(bestCost), "total", money(bestTotal));
1068 break;
1069 }
1070 default:
1071 fail("unknown command " + command);
1072 }

1076 String[] fields = line.split(" ");
1077 String command = fields[0];
1078 List<String> args = Arrays.asList(fields).subList(1, fields.length);
1079 try {
1080 command(command, args);
1081 } catch (CommandError error) {
1082 emit("ERROR", error.getMessage());
1083 }
1084 }
