24function decimal(int $numerator, int $denominator, int $places): string
25{
26 $whole = intdiv($numerator, $denominator);
27 $remainder = $numerator % $denominator;
28 $fraction = '';
29 for ($i = 0; $i < $places; ++$i) {
30 $remainder *= 10;
31 $fraction .= (string) intdiv($remainder, $denominator);
32 $remainder %= $denominator;
33 }
34 return $whole . '.' . $fraction;
35}

37function floorDiv(int $numerator, int $denominator): int
38{
39 $quotient = intdiv($numerator, $denominator);
40 return $numerator % $denominator < 0 ? $quotient - 1 : $quotient;
41}

43function variation(array $values): int
44{
45 $sum = array_sum($values);
46 $squares = 0;
47 foreach ($values as $value) {
48 $squares += $value * $value;
49 }
50 return count($values) * $squares - $sum * $sum;
51}

53function parsePolicy(array $tokens): array
54{
55 $name = $tokens[0];
56 if (!array_key_exists($name, POLICY_PARAMETERS)) {
57 fail('unknown policy ' . $name);
58 }
59 $count = POLICY_PARAMETERS[$name];
60 if (count($tokens) - 1 !== $count) {
61 fail($name . ' needs ' . $count . ' parameters');
62 }
63 $values = [];
64 foreach (array_slice($tokens, 1) as $i => $token) {
65 $values[] = (int) ($name === 'STERMAN' && $i < 3
66 ? str_replace('.', '', $token) : $token);
67 }
68 $valid = match ($name) {
69 'FIXED' => 0 <= $values[0] && $values[0] <= 1000,
70 'PASS' => true,
71 'BASESTOCK' => 0 <= $values[0] && $values[0] <= 10000,
72 'MINMAX' => 0 <= $values[0] && $values[0] < $values[1] && $values[1] <= 10000,
73 'MOVAVG' => 1 <= $values[0] && $values[0] <= 52 &&
74 0 <= $values[1] && $values[1] <= 10000,
75 'SMOOTH' => 1 <= $values[0] && $values[0] <= 10 &&
76 0 <= $values[1] && $values[1] <= 10000,
77 'STERMAN' => max(array_slice($values, 0, 3)) <= 100 &&
78 min(array_slice($values, 0, 3)) >= 0 &&
79 0 <= $values[3] && $values[3] <= 10000,
80 };
81 if (!$valid) {
82 fail('invalid parameter for ' . $name);
83 }
84 return ['name' => $name, 'args' => $values, 'text' => implode(' ', $tokens)];
85}

118final class Simulation
119{
120 public int $week = 0;
121 public array $demands;
122 public array $stages;
123 public int $plantBacklog = 0;
124 public ?int $plantCapacity = null;
125 public array $plantRecords = [];
126 public string $visibility = 'OFF';
127
128 public function __construct(public array $initials, array $demands)
129 {
130 $this->demands = $demands;
131 $this->stages = array_map(static fn(array $initial): Stage => new Stage($initial), $initials);
132 }
133
134 public function signal(int $index, int $week, ?array $current = null): int
135 {
136 if ($this->visibility === 'ON') {
137 return $this->demands[$week - 1];
138 }
139 if ($current !== null && $week === $this->week + 1) {
140 return $current[$index];
141 }
142 return $this->stages[$index]->records[$week - 1]['incoming'];
143 }
144
145 public function setPolicy(int $index, array $policy): void
146 {
147 $stage = $this->stages[$index];
148 $stage->policy = $policy;
149 if ($policy['name'] === 'STERMAN') {
150 $signal = $this->week
151 ? $this->signal($index, $this->week)
152 : $this->initials[$index]['quantity'];
153 $stage->expected = 100 * $signal;
154 }
155 }
156
157 public function positions(): array
158 {
159 $supply = [];
160 $position = [];
161 foreach ($this->stages as $index => $stage) {
162 $upstream = $index < 3
163 ? $this->stages[$index + 1]->backlog : $this->plantBacklog;
164 $line = array_sum($stage->orders) + $upstream + array_sum($stage->shipping);
165 $supply[] = $line;
166 $position[] = $stage->inventory + $line - $stage->backlog;
167 }
168 return [$supply, $position];
169 }
170
171 private function policyOrder(int $index, array $incoming, int $line, int $position): int
172 {
173 $stage = $this->stages[$index];
174 $name = $stage->policy['name'];
175 $a = $stage->policy['args'];
176 $week = $this->week + 1;
177 $demand = $this->signal($index, $week, $incoming);
178
179 if ($name === 'FIXED') {
180 return $a[0];
181 }
182 if ($name === 'PASS') {
183 return $incoming[$index];
184 }
185 if ($name === 'BASESTOCK') {
186 return max(0, $a[0] - $position);
187 }
188 if ($name === 'MINMAX') {
189 return $position <= $a[0] ? $a[1] - $position : 0;
190 }
191 if ($name === 'MOVAVG') {
192 $count = min($a[0], $week);
193 $total = 0;
194 for ($u = $week - $count + 1; $u <= $week; ++$u) {
195 $total += $this->signal($index, $u, $incoming);
196 }
197 $forecast = intdiv($total, $count);
198 $delay = $this->initials[$index]['shipping'] +
199 $this->initials[$index]['orders'];
200 return max(0, $a[1] + $forecast * $delay - $position);
201 }
202 if ($name === 'SMOOTH') {
203 return max(0, $demand + floorDiv($a[1] - $position, $a[0]));
204 }
205
206 [$theta, $alpha, $beta, $target] = $a;
207 $stage->expected = floorDiv(
208 100 * $theta * $demand + (100 - $theta) * $stage->expected, 100
209 );
210 $net = $stage->inventory - $stage->backlog;
211 return max(0, floorDiv(
212 100 * $stage->expected + $alpha * (100 * ($target - $net) - $beta * $line),
213 10000
214 ));
215 }
216
217 public function step(): array
218 {
219 $received = [];
220 foreach ($this->stages as $stage) {
221 $quantity = array_shift($stage->shipping);
222 $stage->inventory += $quantity;
223 $received[] = $quantity;
224 }
225
226 $incoming = [
227 $this->demands[$this->week],
228 $this->stages[0]->orders[0],
229 $this->stages[1]->orders[0],
230 $this->stages[2]->orders[0],
231 ];
232 $plantIncoming = $this->stages[3]->orders[0];
233 foreach ($this->stages as $stage) {
234 array_shift($stage->orders);
235 }
236
237 $records = [];
238 foreach ($this->stages as $index => $stage) {
239 $before = $stage->backlog;
240 $requested = $before + $incoming[$index];
241 $shipped = min($stage->inventory, $requested);
242 if ($stage->capacity !== null) {
243 $shipped = min($shipped, $stage->capacity);
244 }
245 $stage->inventory -= $shipped;
246 $stage->backlog = $requested - $shipped;
247 if ($index > 0) {
248 $this->stages[$index - 1]->shipping[] = $shipped;
249 }
250 $records[] = [
251 'received' => $received[$index], 'incoming' => $incoming[$index],
252 'shipped' => $shipped, 'before' => $before,
253 'ontime' => min(max(0, $shipped - $before), $incoming[$index]),
254 'inventory' => $stage->inventory, 'backlog' => $stage->backlog,
255 ];
256 }
257
258 $requested = $this->plantBacklog + $plantIncoming;
259 $produced = $this->plantCapacity === null
260 ? $requested : min($requested, $this->plantCapacity);
261 $this->plantBacklog = $requested - $produced;
262 $this->stages[3]->shipping[] = $produced;
263
264 [$supply, $positions] = $this->positions();
265 $orders = [];
266 foreach ($this->stages as $index => $stage) {
267 $orders[] = $this->policyOrder(
268 $index, $incoming, $supply[$index], $positions[$index]
269 );
270 }
271 foreach ($this->stages as $index => $stage) {
272 $quantity = intdiv($orders[$index] + $stage->lot - 1, $stage->lot) * $stage->lot;
273 $stage->orders[] = $quantity;
274 $record = &$records[$index];
275 $record['order'] = $quantity;
276 $record['holding'] = $stage->holding * $stage->inventory;
277 $record['backlog_cost'] = $stage->backlogRate * $stage->backlog;
278 $record['cost'] = $record['holding'] + $record['backlog_cost'];
279 $stage->records[] = $record;
280 unset($record);
281 }
282 $this->plantRecords[] = [
283 'incoming' => $plantIncoming,
284 'produced' => $produced, 'backlog' => $this->plantBacklog,
285 ];
286 ++$this->week;
287 return $records;
288 }
289
290 public function costs(): array
291 {
292 return array_map(static fn(Stage $stage): int => $stage->totalCost(), $this->stages);
293 }
294}

415 private function generate(array $arguments): void
416 {
417 [$weeks, $base, $spread, $seed] = array_map('intval', array_slice($arguments, 0, 4));
418 $changes = [];
419 for ($i = 4; $i < count($arguments); $i += 3) {
420 $change = [
421 'kind' => $arguments[$i],
422 'week' => (int) $arguments[$i + 1],
423 'amount' => (int) $arguments[$i + 2],
424 ];
425 if ($change['week'] < 1 || $change['week'] > $weeks) {
426 fail('week ' . $change['week'] . ' outside horizon');
427 }
428 $changes[] = $change;
429 }
430 $levels = [];
431 for ($week = 1; $week <= $weeks; ++$week) {
432 $level = $base;
433 foreach ($changes as $change) {
434 switch ($change['kind']) {
435 case 'STEP':
436 if ($week >= $change['week']) {
437 $level += $change['amount'];
438 }
439 break;
440 case 'PULSE':
441 if ($week === $change['week']) {
442 $level += $change['amount'];
443 }
444 break;
445 case 'RAMP':
446 if ($week >= $change['week']) {
447 $level += $change['amount'] * ($week - $change['week'] + 1);
448 }
449 break;
450 }
451 }
452 if ($level < 0) {
453 fail('negative level at week ' . $week);
454 }
455 $levels[] = $level;
456 }
457 $values = [];
458 foreach ($levels as $level) {
459 $seed = 48271 * $seed % 2147483647;
460 $values[] = $level + $seed % $spread;
461 }
462 $this->demand($values);
463 }

623 private function showLeadtime(int $index): void
624 {
625 $stage = $this->simulation->stages[$index];
626 $initial = $this->initials[$index];
627 $nominal = $initial['shipping'] + $initial['orders'];
628 $received = 0;
629 $cumulative = [];
630 foreach ($stage->records as $record) {
631 $received += $record['received'];
632 $cumulative[] = $received - $nominal * $initial['quantity'];
633 }
634 $ordered = 0;
635 $count = 0;
636 $leads = [];
637 foreach ($stage->records as $t => $record) {
638 $ordered += $record['order'];
639 if ($record['order'] === 0) {
640 continue;
641 }
642 ++$count;
643 for ($u = $t + 1; $u < count($stage->records); ++$u) {
644 if ($cumulative[$u] >= $ordered) {
645 $leads[] = $u - $t;
646 break;
647 }
648 }
649 }
650 $this->emit(
651 'LEADTIME', STAGE_NAMES[$index], 'nominal', $nominal,
652 'orders', $count, 'completed', count($leads),
653 'open', $count - count($leads),
654 'min', $leads ? min($leads) : '-',
655 'avg', $leads ? decimal(array_sum($leads), count($leads), 2) : '-',
656 'max', $leads ? max($leads) : '-'
657 );
658 }

680 $right = $value > 0 ? str_repeat('+', $width) : '';
681 $this->emit(sprintf('%3d %s|%s %d', $first + $i, $left, $right, $value));
682 }

684 $this->emit(
685 'PLOT', STAGE_NAMES[$index], $mode, $first . '-' . $last, 'max', $maximum
686 );
687 foreach ($values as $i => $value) {
688 $width = $maximum ? intdiv($value * 50, $maximum) : 0;
689 $this->emit(sprintf('%3d |%s %d', $first + $i, str_repeat('#', $width), $value));
690 }
691 }

694 private function showHistogram(int $index, string $mode, int $width): void
695 {
696 $counts = [];
697 foreach ($this->values($index, $mode) as $value) {
698 $bin = floorDiv($value, $width);
699 $counts[$bin] = ($counts[$bin] ?? 0) + 1;
700 }
701 $this->emit(
702 'HISTOGRAM', STAGE_NAMES[$index], $mode,
703 'width', $width, 'weeks', '1-' . $this->simulation->week
704 );
705 for ($bin = min(array_keys($counts)); $bin <= max(array_keys($counts)); ++$bin) {
706 $count = $counts[$bin] ?? 0;
707 $text = ($bin * $width) . '..' . ($bin * $width + $width - 1) . ' ' . $count;
708 if ($count) {
709 $text .= ' ' . str_repeat('*', $count);
710 }
711 $this->emit($text);
712 }

715 private function command(string $command, array $a): void
716 {
717 $s = $this->simulation;
718 switch ($command) {
719 case 'DEMAND':
720 $this->demand(array_map('intval', $a));
721 break;
722 case 'GENERATE':
723 $this->generate($a);
724 break;
725 case 'RUN':
726 if ($s->week === count($s->demands)) {
727 fail('horizon reached at week ' . count($s->demands));
728 }
729 $first = $s->week + 1;
730 $count = min((int) $a[0], count($s->demands) - $s->week);
731 $cost = 0;
732 for ($t = 0; $t < $count; ++$t) {
733 $records = $s->step();
734 foreach ($records as $index => $record) {
735 $cost += $record['cost'];
736 if ($record['before'] === 0 && $record['backlog'] > 0) {
737 $this->emit(
738 'STOCKOUT', STAGE_NAMES[$index], 'week', $s->week,
739 'backlog', $record['backlog']
740 );
741 } elseif ($record['before'] > 0 && $record['backlog'] === 0) {
742 $this->emit('RECOVER', STAGE_NAMES[$index], 'week', $s->week);
743 }
744 }
745 }
746 $demand = array_sum(array_slice(
747 $s->demands, $first - 1, $s->week - $first + 1
748 ));
749 $this->emit('RUN', $first . '-' . $s->week, 'demand', $demand, 'cost', money($cost));
750 if ($s->week === count($s->demands)) {
751 $this->emit('HORIZON', $s->week);
752 }
753 break;
754 case 'RESET':
755 $this->initialize();
756 $this->emit('RESET', 'week', 0);
757 break;
758 case 'SAVE':
759 $this->snapshots[$a[0]] = unserialize(serialize($s));
760 $this->emit('SAVE', $a[0], 'week', $s->week);
761 break;
762 case 'LOAD':
763 if (!array_key_exists($a[0], $this->snapshots)) {
764 fail('no snapshot ' . $a[0]);
765 }
766 $this->simulation = unserialize(serialize($this->snapshots[$a[0]]));
767 $this->emit('LOAD', $a[0], 'week', $this->simulation->week);
768 break;
769 case 'SET':
770 $indices = $this->target($a[0]);
771 $policy = parsePolicy(array_slice($a, 1));
772 foreach ($indices as $index) {
773 $s->setPolicy($index, $policy);
774 }
775 $this->emit('SET', $a[0], $policy['text']);
776 break;
777 case 'BATCH':
778 $index = $this->stageIndex($a[0]);
779 $lot = (int) $a[1];
780 if ($lot < 1 || $lot > 100) {
781 fail('invalid batch size ' . $a[1]);
782 }
783 $s->stages[$index]->lot = $lot;
784 $this->emit('BATCH', $a[0], $lot);
785 break;
786 case 'CAPACITY':
787 $index = $a[0] === 'PLANT' ? null : $this->stageIndex($a[0]);
788 $capacity = $a[1] === 'OFF' ? null : (int) $a[1];
789 if ($capacity !== null && $capacity > 10000) {
790 fail('invalid capacity ' . $a[1]);
791 }
792 if ($index === null) {
793 $s->plantCapacity = $capacity;
794 } else {
795 $s->stages[$index]->capacity = $capacity;
796 }
797 $this->emit('CAPACITY', $a[0], $a[1]);
798 break;
799 case 'COST':
800 $index = $this->stageIndex($a[0]);
801 $holding = (int) $a[1];
802 $backlog = (int) $a[2];
803 if ($holding > 1000 || $backlog > 1000) {
804 fail('invalid cost');
805 }
806 $s->stages[$index]->holding = $holding;
807 $s->stages[$index]->backlogRate = $backlog;
808 $this->emit('COST', $a[0], 'holding', $holding, 'backlog', $backlog);
809 break;
810 case 'VISIBILITY':
811 if (!in_array($a[0], ['ON', 'OFF'], true)) {
812 fail('unknown visibility ' . $a[0]);
813 }
814 $s->visibility = $a[0];
815 $this->emit('VISIBILITY', $a[0]);
816 break;
817 case 'SHOCK':
818 $week = (int) $a[0];
819 $demand = (int) $a[1];
820 if ($week < 1 || $week > count($s->demands)) {
821 fail('week ' . $a[0] . ' outside horizon');
822 }
823 if ($week <= $s->week) {
824 fail('week ' . $a[0] . ' already simulated');
825 }
826 $old = $s->demands[$week - 1];
827 $s->demands[$week - 1] = $demand;
828 $this->emit('SHOCK', 'week', $week, $old, '->', $demand);
829 break;
830 case 'STATUS':
831 $this->showStatus();
832 break;
833 case 'SETTINGS':
834 $this->emit('SETTINGS');
835 foreach ($s->stages as $index => $stage) {
836 $initial = $this->initials[$index];
837 $this->emit(
838 STAGE_NAMES[$index], 'holding', $stage->holding,
839 'backlog', $stage->backlogRate, 'batch', $stage->lot,
840 'capacity', $stage->capacity ?? 'OFF',
841 'delays', $initial['shipping'], $initial['orders']
842 );
843 }
844 $this->emit('PLANT', 'capacity', $s->plantCapacity ?? 'OFF');
845 $this->emit('VISIBILITY', $s->visibility);
846 break;
847 case 'WEEK':
848 $this->requireWeeks();
849 $week = (int) $a[0];
850 if ($week < 1 || $week > $s->week) {
851 fail('bad week ' . $a[0]);
852 }
853 $this->showWeek($week);
854 break;
855 case 'TABLE':
856 $index = $this->stageIndex($a[0]);
857 $this->requireWeeks();
858 [$first, $last] = $this->rangeCheck(array_slice($a, 1), $s->week);
859 $this->showTable($index, $first, $last);
860 break;
861 case 'CHAIN':
862 $this->requireWeeks();
863 [$first, $last] = $this->rangeCheck($a, $s->week);
864 $this->emit('CHAIN', $first . '-' . $last);
865 $header = [
866 str_pad('week', 4, ' ', STR_PAD_LEFT),
867 str_pad('demand', 6, ' ', STR_PAD_LEFT),
868 ];
869 foreach (STAGE_NAMES as $stage) {
870 $header[] = str_pad($stage, 11, ' ', STR_PAD_LEFT);
871 }
872 $this->emit(implode(' ', $header));
873 for ($week = $first; $week <= $last; ++$week) {
874 $fields = [
875 str_pad((string) $week, 4, ' ', STR_PAD_LEFT),
876 str_pad((string) $s->demands[$week - 1], 6, ' ', STR_PAD_LEFT),
877 ];
878 foreach ($s->stages as $stage) {
879 $fields[] = str_pad((string) $stage->records[$week - 1]['order'],
880 11, ' ', STR_PAD_LEFT);
881 }
882 $this->emit(implode(' ', $fields));
883 }
884 break;
885 case 'DEMANDS':
886 [$first, $last] = $this->rangeCheck($a, count($s->demands));
887 $values = array_slice($s->demands, $first - 1, $last - $first + 1);
888 $this->emit(
889 'DEMANDS', $first . '-' . $last,
890 'total', array_sum($values), 'min', min($values), 'max', max($values)
891 );
892 $this->emit(implode(' ', $values));
893 break;
894 case 'COSTS':
895 $this->showCosts();
896 break;
897 case 'SUMMARY':
898 $this->requireWeeks();
899 $this->emit('SUMMARY', 'weeks', '1-' . $s->week);
900 foreach ($s->stages as $index => $stage) {
901 $inventories = array_column($stage->records, 'inventory');
902 $backlogs = array_column($stage->records, 'backlog');
903 $orders = array_column($stage->records, 'order');
904 $this->emit(
905 STAGE_NAMES[$index],
906 'avg_inv', decimal(array_sum($inventories), $s->week, 2),
907 'avg_back', decimal(array_sum($backlogs), $s->week, 2),
908 'orders', array_sum($orders), 'min', min($orders),
909 'max', max($orders), 'zero_weeks', count(array_filter(
910 $orders, static fn(int $v): bool => $v === 0
911 ))
912 );
913 }
914 break;
915 case 'BALANCE':
916 $this->emit('BALANCE', 'week', $s->week);
917 $initialGoods = 0;
918 $produced = 0;
919 $delivered = 0;
920 $stock = 0;
921 $transit = 0;
922 foreach ($s->stages as $index => $stage) {
923 $initial = $this->initials[$index];
924 $received = array_sum(array_column($stage->records, 'received'));
925 $shipped = array_sum(array_column($stage->records, 'shipped'));
926 $this->emit(
927 STAGE_NAMES[$index], 'start', $initial['inventory'],
928 'received', $received, 'shipped', $shipped,
929 'end', $stage->inventory, 'owed', $stage->backlog
930 );
931 $initialGoods += $initial['inventory'] +
932 $initial['quantity'] * $initial['shipping'];
933 if ($index === 0) {
934 $delivered = $shipped;
935 }
936 $stock += $stage->inventory;
937 $transit += array_sum($stage->shipping);
938 }
939 $produced = array_sum(array_column($s->plantRecords, 'produced'));
940 $this->emit(
941 'SYSTEM', 'initial', $initialGoods, 'produced', $produced,
942 'delivered', $delivered, 'stock', $stock, 'transit', $transit
943 );
944 break;
945 case 'FILLRATE':
946 $this->requireWeeks();
947 $this->emit('FILLRATE', 'weeks', '1-' . $s->week);
948 foreach ($s->stages as $index => $stage) {
949 [$ordered, $ontime, $fill] = self::fillText($stage->records);
950 $stockouts = count(array_filter(
951 $stage->records,
952 static fn(array $r): bool => $r['backlog'] > 0
953 ));
954 $this->emit(
955 STAGE_NAMES[$index], 'fill', $fill, 'ontime', $ontime,
956 'ordered', $ordered, 'stockout_weeks', $stockouts,
957 'service', decimal(($s->week - $stockouts) * 100, $s->week, 2) . '%'
958 );
959 }
960 break;
961 case 'BULLWHIP':
962 $this->requireWeeks();
963 $this->showBullwhip();
964 break;
965 case 'PEAKS':
966 $this->requireWeeks();
967 $this->showPeaks();
968 break;
969 case 'LEADTIME':
970 $index = $this->stageIndex($a[0]);
971 $this->requireWeeks();
972 $this->showLeadtime($index);
973 break;
974 case 'PLOT':
975 case 'HISTOGRAM':
976 $index = $this->stageIndex($a[0]);
977 $this->modeCheck($a[1]);
978 if ($command === 'HISTOGRAM') {
979 $width = (int) $a[2];
980 if ($width < 1 || $width > 100) {
981 fail('invalid width ' . $a[2]);
982 }
983 }
984 $this->requireWeeks();
985 if ($command === 'PLOT') {
986 [$first, $last] = $this->rangeCheck(array_slice($a, 2), $s->week);
987 $this->showPlot($index, $a[1], $first, $last);
988 } else {
989 $this->showHistogram($index, $a[1], $width);
990 }
991 break;
992 case 'COMPARE':
993 $policies = [];
994 $start = 0;
995 for ($end = 0; $end <= count($a); ++$end) {
996 if ($end === count($a) || $a[$end] === '/') {
997 $policies[] = parsePolicy(array_slice($a, $start, $end - $start));
998 $start = $end + 1;
999 }
1000 }
1001 $rows = [];
1002 foreach ($policies as $index => $policy) {
1003 $trial = $this->trial(array_fill(0, 4, $policy));
1004 $cost = array_sum($trial->costs());
1005 [, , $fill] = self::fillText($trial->stages[0]->records);
1006 $demandVariation = variation($trial->demands);
1007 $factoryVariation = variation(array_column($trial->stages[3]->records, 'order'));
1008 $ratio = $demandVariation === 0
1009 ? 'n/a' : decimal($factoryVariation, $demandVariation, 4);
1010 $rows[] = [
1011 'cost' => $cost, 'index' => $index, 'policy' => $policy['text'],
1012 'fill' => $fill, 'ratio' => $ratio,
1013 ];
1014 }
1015 usort($rows, static fn(array $a, array $b): int =>
1016 ($a['cost'] <=> $b['cost']) ?: ($a['index'] <=> $b['index']));
1017 $this->emit('COMPARE', count($policies), 'weeks', count($s->demands));
1018 foreach ($rows as $rank => $row) {
1019 $this->emit(
1020 $rank + 1, $row['policy'], 'total', money($row['cost']),
1021 'fill', $row['fill'], 'bullwhip', $row['ratio']
1022 );
1023 }
1024 break;
1025 case 'OPTIMIZE':
1026 $indices = $this->target($a[0]);
1027 $low = (int) $a[1];
1028 $high = (int) $a[2];
1029 if ($low > $high) {
1030 fail('bad range ' . $a[1] . ' ' . $a[2]);
1031 }
1032 if ($high - $low > 200) {
1033 fail('range too large');
1034 }
1035 $best = null;
1036 for ($target = $low; $target <= $high; ++$target) {
1037 $policies = array_map(
1038 static fn(Stage $stage): array => $stage->policy, $s->stages
1039 );
1040 $replacement = [
1041 'name' => 'BASESTOCK', 'args' => [$target],
1042 'text' => 'BASESTOCK ' . $target,
1043 ];
1044 foreach ($indices as $index) {
1045 $policies[$index] = $replacement;
1046 }
1047 $trial = $this->trial($policies);
1048 $costs = $trial->costs();
1049 $objective = 0;
1050 foreach ($indices as $index) {
1051 $objective += $costs[$index];
1052 }
1053 if ($best === null || $objective < $best[0]) {
1054 $best = [$objective, $target, array_sum($costs)];
1055 }
1056 }
1057 $this->emit(
1058 'OPTIMIZE', $a[0], 'BASESTOCK', $best[1],
1059 'cost', money($best[0]), 'total', money($best[2])
1060 );
1061 break;
1062 default:
1063 fail('unknown command ' . $command);
1064 }
1065 }

1068 {
1069 $parts = explode(' ', $line);
1070 $command = array_shift($parts);
1071 try {
1072 $this->command($command, $parts);
1073 } catch (CommandError $error) {
1074 $this->emit('ERROR', $error->getMessage());
1075 }
1076 }
