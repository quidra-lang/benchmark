40string decimal(Number numerator, Number denominator, int places) {
41 Number whole = numerator / denominator;
42 Number remainder = numerator % denominator;
43 string answer = to_string(whole) + ".";
44 for (int digit = 0; digit < places; ++digit) {
45 remainder *= 10;
46 answer += static_cast<char>('0' + remainder / denominator);
47 remainder %= denominator;
48 }
49 return answer;
50}

52Number total(const vector<Number>& values) {
53 return accumulate(values.begin(), values.end(), Number{0});

56Number variation(const vector<Number>& values) {
57 Number squares = 0;
58 for (Number value : values) squares += value * value;
59 return static_cast<Number>(values.size()) * squares - total(values) * total(values);
60}

93Policy parse_policy(const vector<string>& tokens) {
94 const string& name = tokens[0];
95 auto it = PARAMETERS.find(name);
96 if (it == PARAMETERS.end()) throw runtime_error("unknown policy " + name);
97 int count = it->second;
98 if (static_cast<int>(tokens.size()) - 1 != count)
99 throw runtime_error(name + " needs " + to_string(count) + " parameters");
100 Policy result{name, {}, combined(tokens)};
101 for (int i = 0; i < count; ++i) {
102 string token = tokens[i + 1];
103 if (name == "STERMAN" && i < 3) {
104 token.erase(remove(token.begin(), token.end(), '.'), token.end());
105 }
106 result.args.push_back(number(token));
107 }
108 const auto& a = result.args;
109 bool valid = false;
110 if (name == "FIXED") valid = a[0] <= 1000;
111 else if (name == "PASS") valid = true;
112 else if (name == "BASESTOCK") valid = a[0] <= 10000;
113 else if (name == "MINMAX") valid = a[0] < a[1] && a[1] <= 10000;
114 else if (name == "MOVAVG") valid = 1 <= a[0] && a[0] <= 52 && a[1] <= 10000;
115 else if (name == "SMOOTH") valid = 1 <= a[0] && a[0] <= 10 && a[1] <= 10000;
116 else if (name == "STERMAN")
117 valid = a[0] <= 100 && a[1] <= 100 && a[2] <= 100 && a[3] <= 10000;
118 if (!valid) throw runtime_error("invalid parameter for " + name);
119 return result;
120}

186class Simulation {
187 vector<Setting> initials;
188
189public:
190 int week = 0;
191 vector<Number> demands;
192 array<Stage, 4> stages;
193 Number plant_backlog = 0;
194 optional<Number> plant_capacity;
195 vector<PlantRecord> plant_records;
196 string visibility = "OFF";
197
198 Simulation() = default;
199
200 Simulation(vector<Setting> settings, vector<Number> demand)
201 : initials(move(settings)), demands(move(demand)) {
202 for (int index = 0; index < 4; ++index) stages[index] = Stage(initials[index]);
203 }
204
205 Number signal(int index, int queried_week,
206 const array<Number, 4>* current = nullptr) const {
207 if (visibility == "ON") return demands[queried_week - 1];
208 if (current && queried_week == week + 1) return (*current)[index];
209 return stages[index].records[queried_week - 1].incoming;
210 }
211
212 void set_policy(int index, const Policy& policy) {
213 Stage& stage = stages[index];
214 stage.policy = policy;
215 if (policy.name == "STERMAN")
216 stage.expected = 100 * (week ? signal(index, week) : initials[index].quantity);
217 }
218
219 Pipeline positions() const {
220 Pipeline result;
221 for (int index = 0; index < 4; ++index) {
222 const Stage& stage = stages[index];
223 Number upstream_backlog = index < 3 ? stages[index + 1].backlog : plant_backlog;
224 Number line = upstream_backlog;
225 for (Number quantity : stage.orders) line += quantity;
226 for (Number quantity : stage.shipping) line += quantity;
227 result.supply[index] = line;
228 result.position[index] = stage.inventory + line - stage.backlog;
229 }
230 return result;
231 }
232
233 Number policy_order(int index, const array<Number, 4>& incoming,
234 Number line, Number position) {
235 Stage& stage = stages[index];
236 const Policy& policy = stage.policy;
237 const auto& a = policy.args;
238 int t = week + 1;
239 Number demand = signal(index, t, &incoming);
240 if (policy.name == "FIXED") return a[0];
241 if (policy.name == "PASS") return incoming[index];
242 if (policy.name == "BASESTOCK") return max(Number{0}, a[0] - position);
243 if (policy.name == "MINMAX") return position <= a[0] ? a[1] - position : 0;
244 if (policy.name == "MOVAVG") {
245 int count = static_cast<int>(min(a[0], static_cast<Number>(t)));
246 Number demand_sum = 0;
247 for (int u = t - count + 1; u <= t; ++u)
248 demand_sum += signal(index, u, &incoming);
249 Number forecast = demand_sum / count;
250 Number delay = initials[index].shipping + initials[index].orders;
251 return max(Number{0}, a[1] + forecast * delay - position);
252 }
253 if (policy.name == "SMOOTH")
254 return max(Number{0}, demand + static_cast<Number>(floor_div(a[1] - position, a[0])));
255
256 Number theta = a[0], alpha = a[1], beta = a[2], target = a[3];
257 stage.expected = (100 * theta * demand +
258 (100 - theta) * stage.expected) / 100;
259 Number net = stage.inventory - stage.backlog;
260 return max(Number{0}, floor_div(
261 100 * stage.expected + alpha * (100 * (target - net) - beta * line), 10000));
262 }
263
264 array<Record, 4> step() {
265 array<Number, 4> received{};
266 for (int index = 0; index < 4; ++index) {
267 Stage& stage = stages[index];
268 Number quantity = stage.shipping.front();
269 stage.shipping.pop_front();
270 stage.inventory += quantity;
271 received[index] = quantity;
272 }
273 array<Number, 4> incoming{
274 demands[week], stages[0].orders.front(),
275 stages[1].orders.front(), stages[2].orders.front()
276 };
277 Number plant_incoming = stages[3].orders.front();
278 for (Stage& stage : stages) stage.orders.pop_front();
279
280 array<Record, 4> records{};
281 for (int index = 0; index < 4; ++index) {
282 Stage& stage = stages[index];
283 Number before = stage.backlog;
284 Number requested = before + incoming[index];
285 Number shipped = min(stage.inventory, requested);
286 if (stage.capacity) shipped = min(shipped, *stage.capacity);
287 stage.inventory -= shipped;
288 stage.backlog = requested - shipped;
289 if (index) stages[index - 1].shipping.push_back(shipped);
290 Record& record = records[index];
291 record.received = received[index];
292 record.incoming = incoming[index];
293 record.shipped = shipped;
294 record.before = before;
295 record.ontime = min(max(Number{0}, shipped - before), incoming[index]);
296 record.inventory = stage.inventory;
297 record.backlog = stage.backlog;
298 }
299
300 Number plant_requested = plant_backlog + plant_incoming;
301 Number produced = plant_capacity ? min(plant_requested, *plant_capacity) : plant_requested;
302 plant_backlog = plant_requested - produced;
303 stages[3].shipping.push_back(produced);
304
305 Pipeline pipeline = positions();
306 array<Number, 4> orders{};
307 for (int index = 0; index < 4; ++index)
308 orders[index] = policy_order(index, incoming,
309 pipeline.supply[index], pipeline.position[index]);
310 for (int index = 0; index < 4; ++index) {
311 Stage& stage = stages[index];
312 Number quantity = (orders[index] + stage.lot - 1) / stage.lot * stage.lot;
313 stage.orders.push_back(quantity);
314 Record& record = records[index];
315 record.order = quantity;
316 record.holding = stage.holding * stage.inventory;
317 record.backlog_cost = stage.backlog_rate * stage.backlog;
318 record.cost = record.holding + record.backlog_cost;
319 stage.records.push_back(record);
320 }
321 plant_records.push_back({plant_incoming, produced, plant_backlog});
322 ++week;
323 return records;
324 }
325
326 array<Number, 4> costs() const {
327 array<Number, 4> result{};
328 for (int index = 0; index < 4; ++index)
329 result[index] = stages[index].cost();
330 return result;
331 }
332
333private:
334 static Number floor_div(Number numerator, Number denominator) {
335 Number quotient = numerator / denominator;
336 return quotient - (numerator % denominator < 0);
337 }

426 void generate(const vector<string>& a) {
427 int weeks = stoi(a[0]);
428 Number baseline = number(a[1]), spread = number(a[2]), seed = number(a[3]);
429 struct Change {
430 string kind;
431 int week;
432 Number amount;
433 };
434 vector<Change> changes;
435 for (size_t i = 4; i < a.size(); i += 3) {
436 Change change{a[i], stoi(a[i + 1]), number(a[i + 2])};
437 if (change.week < 1 || change.week > weeks)
438 fail("week " + to_string(change.week) + " outside horizon");
439 changes.push_back(move(change));
440 }
441 vector<Number> levels;
442 for (int t = 1; t <= weeks; ++t) {
443 Number level = baseline;
444 for (const Change& change : changes) {
445 if (change.kind == "STEP" && t >= change.week) level += change.amount;
446 else if (change.kind == "PULSE" && t == change.week) level += change.amount;
447 else if (change.kind == "RAMP" && t >= change.week)
448 level += change.amount * (t - change.week + 1);
449 }
450 if (level < 0) fail("negative level at week " + to_string(t));
451 levels.push_back(level);
452 }
453 vector<Number> generated;
454 for (Number level : levels) {
455 seed = 48271 * seed % 2147483647;
456 generated.push_back(level + seed % spread);
457 }
458 demand(generated);
459 }

606 void show_leadtime(int index) {
607 const Stage& stage = simulation.stages[index];
608 const Setting& initial = initials[index];
609 int nominal = initial.shipping + initial.orders;
610 vector<Number> cumulative;
611 Number received = 0;
612 for (const Record& record : stage.records) {
613 received += record.received;
614 cumulative.push_back(received - nominal * initial.quantity);
615 }
616 Number ordered = 0;
617 int count = 0;
618 vector<Number> leads;
619 for (int t = 0; t < static_cast<int>(stage.records.size()); ++t) {
620 const Record& record = stage.records[t];
621 ordered += record.order;
622 if (record.order == 0) continue;
623 ++count;
624 for (int u = t + 1; u < static_cast<int>(stage.records.size()); ++u) {
625 if (cumulative[u] >= ordered) {
626 leads.push_back(u - t);
627 break;
628 }
629 }
630 }
631 emit("LEADTIME", NAMES[index], "nominal", nominal,
632 "orders", count, "completed", leads.size(),
633 "open", count - static_cast<int>(leads.size()),
634 "min", leads.empty() ? "-" : integer(*min_element(leads.begin(), leads.end())),
635 "avg", leads.empty() ? "-" : decimal(total(leads), leads.size(), 2),
636 "max", leads.empty() ? "-" : integer(*max_element(leads.begin(), leads.end())));
637 }

639 void show_plot(int index, const string& mode, int first, int last) {
640 vector<Number> all = values(index, mode);
641 vector<Number> v(all.begin() + first - 1, all.begin() + last);
642 Number maximum = *max_element(v.begin(), v.end());
643 string period = to_string(first) + "-" + to_string(last);
644 if (mode == "NET") {
645 emit("PLOT", NAMES[index], mode, period,
646 "min", *min_element(v.begin(), v.end()), "max", maximum);
647 Number scale = 0;
648 for (Number value : v) scale = max(scale, abs(value));
649 for (int week = first; week <= last; ++week) {
650 Number value = v[week - first];
651 int width = scale ? static_cast<int>(abs(value) * 25 / scale) : 0;
652 string left = value < 0
653 ? string(25 - width, ' ') + string(width, '-')
654 : string(25, ' ');
655 string right = value > 0 ? string(width, '+') : "";
656 emit(aligned(to_string(week), 3) + " " + left + "|" +
657 right + " " + integer(value));
658 }
659 } else {
660 emit("PLOT", NAMES[index], mode, period, "max", maximum);
661 for (int week = first; week <= last; ++week) {
662 Number value = v[week - first];
663 int width = maximum ? static_cast<int>(value * 50 / maximum) : 0;
664 emit(aligned(to_string(week), 3) + " |" +
665 string(width, '#') + " " + integer(value));
666 }
667 }
668 }

670 void show_histogram(int index, const string& mode, Number width) {
671 vector<Number> observed = values(index, mode);
672 map<Number, int> counts;
673 for (Number value : observed) {
674 Number bucket = value / width - (value % width < 0);
675 ++counts[bucket];
676 }
677 emit("HISTOGRAM", NAMES[index], mode, "width", width,
678 "weeks", "1-" + to_string(simulation.week));
679 for (Number bucket = counts.begin()->first; bucket <= counts.rbegin()->first; ++bucket) {
680 int count = counts[bucket];
681 string text = integer(bucket * width) + ".." +
682 integer(bucket * width + width - 1) + " " + to_string(count);
683 if (count) text += " " + string(count, '*');
684 emit(text);
685 }
686 }

688 void execute(const string& command, const vector<string>& a) {
689 if (command == "DEMAND") {
690 vector<Number> data;
691 for (const string& value : a) data.push_back(number(value));
692 demand(data);
693 } else if (command == "GENERATE") {
694 generate(a);
695 } else if (command == "RUN") {
696 if (simulation.week == static_cast<int>(simulation.demands.size()))
697 fail("horizon reached at week " + to_string(simulation.demands.size()));
698 int first = simulation.week + 1;
699 int count = min(stoi(a[0]), static_cast<int>(simulation.demands.size()) - simulation.week);
700 Number cost = 0;
701 for (int t = 0; t < count; ++t) {
702 auto records = simulation.step();
703 for (int index = 0; index < 4; ++index) {
704 const Record& r = records[index];
705 cost += r.cost;
706 if (r.before == 0 && r.backlog > 0)
707 emit("STOCKOUT", NAMES[index], "week", simulation.week,
708 "backlog", r.backlog);
709 else if (r.before > 0 && r.backlog == 0)
710 emit("RECOVER", NAMES[index], "week", simulation.week);
711 }
712 }
713 Number demand_total = 0;
714 for (int t = first; t <= simulation.week; ++t)
715 demand_total += simulation.demands[t - 1];
716 emit("RUN", to_string(first) + "-" + to_string(simulation.week),
717 "demand", demand_total, "cost", money(cost));
718 if (simulation.week == static_cast<int>(simulation.demands.size()))
719 emit("HORIZON", simulation.week);
720 } else if (command == "RESET") {
721 initialize();
722 emit("RESET", "week", 0);
723 } else if (command == "SAVE") {
724 snapshots[a[0]] = simulation;
725 emit("SAVE", a[0], "week", simulation.week);
726 } else if (command == "LOAD") {
727 auto it = snapshots.find(a[0]);
728 if (it == snapshots.end()) fail("no snapshot " + a[0]);
729 simulation = it->second;
730 emit("LOAD", a[0], "week", simulation.week);
731 } else if (command == "SET") {
732 vector<int> indices = target(a[0]);
733 Policy policy = parse_policy(vector<string>(a.begin() + 1, a.end()));
734 for (int index : indices) simulation.set_policy(index, policy);
735 emit("SET", a[0], policy.text);
736 } else if (command == "BATCH") {
737 int index = stage_index(a[0]), lot = stoi(a[1]);
738 if (lot < 1 || lot > 100) fail("invalid batch size " + a[1]);
739 simulation.stages[index].lot = lot;
740 emit("BATCH", a[0], lot);
741 } else if (command == "CAPACITY") {
742 int index = a[0] == "PLANT" ? -1 : stage_index(a[0]);
743 optional<Number> capacity = a[1] == "OFF" ? nullopt :
744 optional<Number>(number(a[1]));
745 if (capacity && *capacity > 10000) fail("invalid capacity " + a[1]);
746 if (index == -1) simulation.plant_capacity = capacity;
747 else simulation.stages[index].capacity = capacity;
748 emit("CAPACITY", a[0], a[1]);
749 } else if (command == "COST") {
750 int index = stage_index(a[0]);
751 Number holding = number(a[1]), backlog = number(a[2]);
752 if (holding > 1000 || backlog > 1000) fail("invalid cost");
753 simulation.stages[index].holding = holding;
754 simulation.stages[index].backlog_rate = backlog;
755 emit("COST", a[0], "holding", holding, "backlog", backlog);
756 } else if (command == "VISIBILITY") {
757 if (a[0] != "ON" && a[0] != "OFF")
758 fail("unknown visibility " + a[0]);
759 simulation.visibility = a[0];
760 emit("VISIBILITY", a[0]);
761 } else if (command == "SHOCK") {
762 int week = stoi(a[0]);
763 Number demand = number(a[1]);
764 if (week < 1 || week > static_cast<int>(simulation.demands.size()))
765 fail("week " + a[0] + " outside horizon");
766 if (week <= simulation.week) fail("week " + a[0] + " already simulated");
767 Number old = simulation.demands[week - 1];
768 simulation.demands[week - 1] = demand;
769 emit("SHOCK", "week", week, old, "->", demand);
770 } else if (command == "STATUS") {
771 show_status();
772 } else if (command == "SETTINGS") {
773 emit("SETTINGS");
774 for (int index = 0; index < 4; ++index) {
775 const Stage& stage = simulation.stages[index];
776 const Setting& initial = initials[index];
777 emit(NAMES[index], "holding", stage.holding,
778 "backlog", stage.backlog_rate, "batch", stage.lot,
779 "capacity", stage.capacity ? integer(*stage.capacity) : "OFF",
780 "delays", initial.shipping, initial.orders);
781 }
782 emit("PLANT", "capacity", simulation.plant_capacity
783 ? integer(*simulation.plant_capacity) : "OFF");
784 emit("VISIBILITY", simulation.visibility);
785 } else if (command == "WEEK") {
786 require_weeks();
787 int week = stoi(a[0]);
788 if (week < 1 || week > simulation.week) fail("bad week " + a[0]);
789 show_week(week);
790 } else if (command == "TABLE") {
791 int index = stage_index(a[0]);
792 require_weeks();
793 auto [first, last] = range_check(
794 vector<string>(a.begin() + 1, a.end()), simulation.week);
795 show_table(index, first, last);
796 } else if (command == "CHAIN") {
797 require_weeks();
798 auto [first, last] = range_check(a, simulation.week);
799 emit("CHAIN", to_string(first) + "-" + to_string(last));
800 vector<string> header{aligned("week", 4), aligned("demand", 6)};
801 for (const char* stage : NAMES) header.push_back(aligned(stage, 11));
802 emit(combined(header));
803 for (int week = first; week <= last; ++week) {
804 vector<string> values{
805 aligned(to_string(week), 4),
806 aligned(integer(simulation.demands[week - 1]), 6)
807 };
808 for (const Stage& stage : simulation.stages)
809 values.push_back(aligned(integer(stage.records[week - 1].order), 11));
810 emit(combined(values));
811 }
812 } else if (command == "DEMANDS") {
813 auto [first, last] = range_check(a, simulation.demands.size());
814 vector<Number> values(simulation.demands.begin() + first - 1,
815 simulation.demands.begin() + last);
816 emit("DEMANDS", to_string(first) + "-" + to_string(last),
817 "total", total(values),
818 "min", *min_element(values.begin(), values.end()),
819 "max", *max_element(values.begin(), values.end()));
820 vector<string> parts;
821 for (Number value : values) parts.push_back(integer(value));
822 emit(combined(parts));
823 } else if (command == "COSTS") {
824 show_costs();
825 } else if (command == "SUMMARY") {
826 require_weeks();
827 emit("SUMMARY", "weeks", "1-" + to_string(simulation.week));
828 for (int index = 0; index < 4; ++index) {
829 Number inventory = 0, backlog = 0, order_total = 0;
830 Number smallest = numeric_limits<Number>::max(), largest = 0;
831 int zero_weeks = 0;
832 for (const Record& r : simulation.stages[index].records) {
833 inventory += r.inventory;
834 backlog += r.backlog;
835 order_total += r.order;
836 smallest = min(smallest, r.order);
837 largest = max(largest, r.order);
838 zero_weeks += r.order == 0;
839 }
840 emit(NAMES[index], "avg_inv", decimal(inventory, simulation.week, 2),
841 "avg_back", decimal(backlog, simulation.week, 2),
842 "orders", order_total, "min", smallest, "max", largest,
843 "zero_weeks", zero_weeks);
844 }
845 } else if (command == "BALANCE") {
846 emit("BALANCE", "week", simulation.week);
847 Number initial_goods = 0, produced = 0, delivered = 0, stock = 0, transit = 0;
848 for (int index = 0; index < 4; ++index) {
849 const Stage& stage = simulation.stages[index];
850 Number received = 0, shipped = 0;
851 for (const Record& r : stage.records) {
852 received += r.received;
853 shipped += r.shipped;
854 }
855 emit(NAMES[index], "start", initials[index].inventory,
856 "received", received, "shipped", shipped,
857 "end", stage.inventory, "owed", stage.backlog);
858 initial_goods += initials[index].inventory +
859 initials[index].quantity * initials[index].shipping;
860 if (index == 0) delivered = shipped;
861 stock += stage.inventory;
862 for (Number amount : stage.shipping) transit += amount;
863 }
864 for (const PlantRecord& record : simulation.plant_records)
865 produced += record.produced;
866 emit("SYSTEM", "initial", initial_goods, "produced", produced,
867 "delivered", delivered, "stock", stock, "transit", transit);
868 } else if (command == "FILLRATE") {
869 require_weeks();
870 emit("FILLRATE", "weeks", "1-" + to_string(simulation.week));
871 for (int index = 0; index < 4; ++index) {
872 const auto& records = simulation.stages[index].records;
873 Fill f = fill_text(records);
874 int stockouts = 0;
875 for (const Record& record : records) stockouts += record.backlog > 0;
876 emit(NAMES[index], "fill", f.fill,
877 "ontime", f.ontime, "ordered", f.ordered,
878 "stockout_weeks", stockouts,
879 "service", decimal(100LL * (simulation.week - stockouts),
880 simulation.week, 2) + "%");
881 }
882 } else if (command == "BULLWHIP") {
883 require_weeks();
884 show_bullwhip();
885 } else if (command == "PEAKS") {
886 require_weeks();
887 show_peaks();
888 } else if (command == "LEADTIME") {
889 int index = stage_index(a[0]);
890 require_weeks();
891 show_leadtime(index);
892 } else if (command == "PLOT" || command == "HISTOGRAM") {
893 int index = stage_index(a[0]);
894 mode_check(a[1]);
895 Number width = 0;
896 if (command == "HISTOGRAM") {
897 width = number(a[2]);
898 if (width < 1 || width > 100)
899 fail("invalid width " + a[2]);
900 }
901 require_weeks();
902 if (command == "PLOT") {
903 auto [first, last] = range_check(
904 vector<string>(a.begin() + 2, a.end()), simulation.week);
905 show_plot(index, a[1], first, last);
906 } else {
907 show_histogram(index, a[1], width);
908 }
909 } else if (command == "COMPARE") {
910 vector<Policy> policies;
911 size_t start = 0;
912 for (size_t end = 0; end <= a.size(); ++end) {
913 if (end == a.size() || a[end] == "/") {
914 policies.push_back(parse_policy(
915 vector<string>(a.begin() + start, a.begin() + end)));
916 start = end + 1;
917 }
918 }
919 struct Row {
920 Number cost;
921 size_t index;
922 string policy;
923 string fill;
924 string ratio;
925 };
926 vector<Row> rows;
927 for (size_t i = 0; i < policies.size(); ++i) {
928 array<Policy, 4> selected;
929 selected.fill(policies[i]);
930 Simulation s = trial(selected);
931 auto costs = s.costs();
932 Number cost = accumulate(costs.begin(), costs.end(), Number{0});
933 Fill f = fill_text(s.stages[0].records);
934 Number demand_var = variation(s.demands);
935 vector<Number> factory_orders;
936 for (const Record& r : s.stages[3].records)
937 factory_orders.push_back(r.order);
938 Number factory_var = variation(factory_orders);
939 string ratio = demand_var == 0 ? "n/a" : decimal(factory_var, demand_var, 4);
940 rows.push_back({cost, i, policies[i].text, f.fill, ratio});
941 }
942 sort(rows.begin(), rows.end(), [](const Row& a, const Row& b) {
943 return tie(a.cost, a.index) < tie(b.cost, b.index);
944 });
945 emit("COMPARE", policies.size(), "weeks", simulation.demands.size());
946 for (size_t i = 0; i < rows.size(); ++i)
947 emit(i + 1, rows[i].policy, "total", money(rows[i].cost),
948 "fill", rows[i].fill, "bullwhip", rows[i].ratio);
949 } else if (command == "OPTIMIZE") {
950 vector<int> indices = target(a[0]);
951 int low = stoi(a[1]), high = stoi(a[2]);
952 if (low > high) fail("bad range " + a[1] + " " + a[2]);
953 if (high - low > 200) fail("range too large");
954 Number best_cost = numeric_limits<Number>::max(), best_total = 0;
955 int best_target = low;
956 for (int value = low; value <= high; ++value) {
957 array<Policy, 4> policies;
958 for (int i = 0; i < 4; ++i) policies[i] = simulation.stages[i].policy;
959 Policy replacement{"BASESTOCK", {value}, "BASESTOCK " + to_string(value)};
960 for (int index : indices) policies[index] = replacement;
961 Simulation s = trial(policies);
962 auto costs = s.costs();
963 Number objective = 0, chain_total = 0;
964 for (int index : indices) objective += costs[index];
965 for (Number cost : costs) chain_total += cost;
966 if (objective < best_cost) {
967 best_cost = objective;
968 best_target = value;
969 best_total = chain_total;
970 }
971 }
972 emit("OPTIMIZE", a[0], "BASESTOCK", best_target,
973 "cost", money(best_cost), "total", money(best_total));
974 } else {
975 fail("unknown command " + command);
976 }
977 }

980 explicit Game(const vector<vector<string>>& settings) {
981 for (const vector<string>& tokens : settings) {
982 Policy policy = parse_policy(vector<string>(tokens.begin() + 7, tokens.end()));
983 initials.push_back({
984 number(tokens[1]), number(tokens[2]),
985 stoi(tokens[3]), stoi(tokens[4]),
986 number(tokens[5]), number(tokens[6]), move(policy)
987 });
988 }
989 initialize();
990 }

993 istringstream input(line);
994 string command, word;
995 input >> command;
996 vector<string> args;
997 while (input >> word) args.push_back(word);
998 try {
999 execute(command, args);
1000 } catch (const runtime_error& error) {
1001 emit("ERROR", error.what());
1002 }
1003 }
