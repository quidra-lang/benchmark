24func money(value int64) string {

28func decimal(numerator, denominator int64, places int) string {
29 whole, remainder := numerator/denominator, numerator%denominator
30 var result strings.Builder
31 fmt.Fprintf(&result, "%d.", whole)
32 for i := 0; i < places; i++ {
33 remainder *= 10
34 result.WriteByte('0' + byte(remainder/denominator))
35 remainder %= denominator
36 }
37 return result.String()
38}

40func variation(values []int64) int64 {
41 var sum, squares int64
42 for _, value := range values {
43 sum += value
44 squares += value * value
45 }
46 return int64(len(values))*squares - sum*sum
47}

49func sum(values []int64) int64 {
50 var result int64
51 for _, value := range values {
52 result += value
53 }
54 return result
55}

57func floorDiv(numerator, denominator int64) int64 {
58 result := numerator / denominator
59 if numerator%denominator < 0 {
60 result--
61 }
62 return result
63}

79func parsePolicy(tokens []string) (Policy, error) {
80 name := tokens[0]
81 count, known := policyParameters[name]
82 if !known {
83 return Policy{}, fmt.Errorf("unknown policy %s", name)
84 }
85 if len(tokens)-1 != count {
86 return Policy{}, fmt.Errorf("%s needs %d parameters", name, count)
87 }
88 args := make([]int64, count)
89 for i, token := range tokens[1:] {
90 if name == "STERMAN" && i < 3 {
91 token = strings.ReplaceAll(token, ".", "")
92 }
93 args[i] = parse(token)
94 }
95 valid := false
96 switch name {
97 case "FIXED":
98 valid = args[0] <= 1000
99 case "PASS":
100 valid = true
101 case "BASESTOCK":
102 valid = args[0] <= 10000
103 case "MINMAX":
104 valid = args[0] < args[1] && args[1] <= 10000
105 case "MOVAVG":
106 valid = 1 <= args[0] && args[0] <= 52 && args[1] <= 10000
107 case "SMOOTH":
108 valid = 1 <= args[0] && args[0] <= 10 && args[1] <= 10000
109 case "STERMAN":
110 valid = args[0] <= 100 && args[1] <= 100 && args[2] <= 100 && args[3] <= 10000
111 }
112 if !valid {
113 return Policy{}, fmt.Errorf("invalid parameter for %s", name)
114 }
115 return Policy{name, args, strings.Join(tokens, " ")}, nil
116}

227func (sim *Simulation) signal(index int, week int, current *[4]int64) int64 {
228 if sim.visibility == "ON" {
229 return sim.demands[week-1]
230 }
231 if current != nil && week == sim.week+1 {
232 return current[index]

234 return sim.stages[index].records[week-1].incoming

237func (sim *Simulation) setPolicy(index int, policy Policy) {
238 stage := &sim.stages[index]
239 stage.policy = policy
240 if policy.name == "STERMAN" {
241 previous := sim.initials[index].quantity
242 if sim.week > 0 {
243 previous = sim.signal(index, sim.week, nil)
244 }
245 stage.expected = 100 * previous
246 }
247}

249func (sim *Simulation) positions() ([4]int64, [4]int64) {
250 var supply, position [4]int64
251 for i, stage := range sim.stages {
252 upstream := sim.plantBacklog
253 if i < 3 {
254 upstream = sim.stages[i+1].backlog
255 }
256 line := sum(stage.orders) + upstream + sum(stage.shipping)
257 supply[i] = line
258 position[i] = stage.inventory + line - stage.backlog
259 }
260 return supply, position
261}

263func (sim *Simulation) policyOrder(index int, incoming [4]int64, line, position int64) int64 {
264 stage := &sim.stages[index]
265 name, args := stage.policy.name, stage.policy.args
266 week := sim.week + 1
267 demand := sim.signal(index, week, &incoming)
268 switch name {
269 case "FIXED":
270 return args[0]
271 case "PASS":
272 return incoming[index]
273 case "BASESTOCK":
274 if args[0] > position {
275 return args[0] - position
276 }
277 return 0
278 case "MINMAX":
279 if position <= args[0] {
280 return args[1] - position
281 }
282 return 0
283 case "MOVAVG":
284 count := int(args[0])
285 if count > week {
286 count = week
287 }
288 var total int64
289 for u := week - count + 1; u <= week; u++ {
290 total += sim.signal(index, u, &incoming)
291 }
292 forecast := total / int64(count)
293 delay := sim.initials[index].shipping + sim.initials[index].orders
294 value := args[1] + forecast*int64(delay) - position
295 if value > 0 {
296 return value
297 }
298 return 0
299 case "SMOOTH":
300 value := demand + floorDiv(args[1]-position, args[0])
301 if value > 0 {
302 return value
303 }
304 return 0
305 case "STERMAN":
306 theta, alpha, beta, target := args[0], args[1], args[2], args[3]
307 stage.expected = floorDiv(
308 100*theta*demand+(100-theta)*stage.expected, 100)
309 net := stage.inventory - stage.backlog
310 value := floorDiv(100*stage.expected+
311 alpha*(100*(target-net)-beta*line), 10000)
312 if value > 0 {
313 return value
314 }
315 return 0
316 }
317 return 0
318}

320func (sim *Simulation) step() [4]Record {
321 var received [4]int64
322 for i := range sim.stages {
323 stage := &sim.stages[i]
324 received[i] = stage.shipping[0]
325 stage.shipping = stage.shipping[1:]
326 stage.inventory += received[i]
327 }
328 incoming := [4]int64{
329 sim.demands[sim.week], sim.stages[0].orders[0],
330 sim.stages[1].orders[0], sim.stages[2].orders[0],
331 }
332 plantIncoming := sim.stages[3].orders[0]
333 for i := range sim.stages {
334 stage := &sim.stages[i]
335 stage.orders = stage.orders[1:]
336 }
337
338 var records [4]Record
339 for i := range sim.stages {
340 stage := &sim.stages[i]
341 before := stage.backlog
342 requested := before + incoming[i]
343 shipped := stage.inventory
344 if requested < shipped {
345 shipped = requested
346 }
347 if stage.capacity != nil && *stage.capacity < shipped {
348 shipped = *stage.capacity
349 }
350 stage.inventory -= shipped
351 stage.backlog = requested - shipped
352 if i > 0 {
353 sim.stages[i-1].shipping = append(sim.stages[i-1].shipping, shipped)
354 }
355 ontime := shipped - before
356 if ontime < 0 {
357 ontime = 0
358 }
359 if ontime > incoming[i] {
360 ontime = incoming[i]
361 }
362 records[i] = Record{
363 received: received[i], incoming: incoming[i], shipped: shipped,
364 before: before, ontime: ontime, inventory: stage.inventory,
365 backlog: stage.backlog,
366 }
367 }
368 requested := sim.plantBacklog + plantIncoming
369 produced := requested
370 if sim.plantCapacity != nil && *sim.plantCapacity < produced {
371 produced = *sim.plantCapacity
372 }
373 sim.plantBacklog = requested - produced
374 sim.stages[3].shipping = append(sim.stages[3].shipping, produced)
375
376 supply, positions := sim.positions()
377 var orders [4]int64
378 for i := range sim.stages {
379 orders[i] = sim.policyOrder(i, incoming, supply[i], positions[i])
380 }
381 for i := range sim.stages {
382 stage := &sim.stages[i]
383 quantity := (orders[i] + stage.lot - 1) / stage.lot * stage.lot
384 stage.orders = append(stage.orders, quantity)
385 record := &records[i]
386 record.order = quantity
387 record.holding = stage.holding * stage.inventory
388 record.backlogCost = stage.backlogRate * stage.backlog
389 record.cost = record.holding + record.backlogCost
390 stage.records = append(stage.records, *record)
391 }
392 sim.plantRecords = append(sim.plantRecords,
393 PlantRecord{plantIncoming, produced, sim.plantBacklog})
394 sim.week++
395 return records
396}

463func rangeCheck(args []string, limit int) (int, int, error) {
464 if len(args) == 0 {
465 return 1, limit, nil
466 }
467 first, last := int(parse(args[0])), int(parse(args[1]))
468 if first < 1 || first > last || last > limit {
469 return 0, 0, fmt.Errorf("bad range %s %s", args[0], args[1])
470 }
471 return first, last, nil
472}

506func (game *Game) trial(policies [4]Policy) *Simulation {
507 current := game.sim
508 trial := newSimulation(game.initials, current.demands)
509 trial.visibility = current.visibility
510 trial.plantCapacity = current.plantCapacity
511 for i := range trial.stages {
512 stage := &trial.stages[i]
513 previous := &current.stages[i]
514 stage.lot = previous.lot
515 stage.capacity = previous.capacity
516 stage.holding = previous.holding
517 stage.backlogRate = previous.backlogRate
518 trial.setPolicy(i, policies[i])
519 }
520 for trial.week < len(trial.demands) {
521 trial.step()
522 }
523 return trial
524}

541func (game *Game) generate(args []string) error {
542 weeks := int(parse(args[0]))
543 base, spread, seed := parse(args[1]), parse(args[2]), parse(args[3])
544 type Change struct {
545 kind string
546 week int
547 amount int64
548 }
549 changes := []Change{}
550 for i := 4; i < len(args); i += 3 {
551 change := Change{args[i], int(parse(args[i+1])), parse(args[i+2])}
552 if change.week < 1 || change.week > weeks {
553 return fmt.Errorf("week %d outside horizon", change.week)
554 }
555 changes = append(changes, change)
556 }
557 levels := make([]int64, weeks)
558 for week := 1; week <= weeks; week++ {
559 level := base
560 for _, change := range changes {
561 switch change.kind {
562 case "STEP":
563 if week >= change.week {
564 level += change.amount
565 }
566 case "PULSE":
567 if week == change.week {
568 level += change.amount
569 }
570 case "RAMP":
571 if week >= change.week {
572 level += change.amount * int64(week-change.week+1)
573 }
574 }
575 }
576 if level < 0 {
577 return fmt.Errorf("negative level at week %d", week)
578 }
579 levels[week-1] = level
580 }
581 values := make([]int64, weeks)
582 for i, level := range levels {
583 seed = 48271 * seed % 2147483647
584 values[i] = level + seed%spread
585 }
586 game.demand(values)
587 return nil
588}

679func (game *Game) bullwhip() {
680 sim := game.sim
681 n := sim.week
682 demands := sim.demands[:n]
683 demandVar := variation(demands)
684 game.emit("BULLWHIP", "weeks", fmt.Sprintf("1-%d", n), "demand",
685 "mean", decimal(sum(demands), int64(n), 4),
686 "var", decimal(demandVar, int64(n*n), 4))
687 previous := demandVar
688 for i, stage := range sim.stages {
689 values := make([]int64, len(stage.records))
690 for j, record := range stage.records {
691 values[j] = record.order
692 }
693 current := variation(values)
694 ratio := "n/a"
695 step := "n/a"
696 if demandVar != 0 {
697 ratio = decimal(current, demandVar, 4)
698 }
699 if previous != 0 {
700 step = decimal(current, previous, 4)
701 }
702 game.emit(stageNames[i], "mean", decimal(sum(values), int64(n), 4),
703 "var", decimal(current, int64(n*n), 4), "ratio", ratio, "step", step)
704 previous = current
705 }
706}

708func (game *Game) peaks() {
709 sim := game.sim
710 game.emit("PEAKS", "weeks", fmt.Sprintf("1-%d", sim.week))
711 for i, stage := range sim.stages {
712 maxInventory, maxBacklog := int64(0), int64(0)
713 inventoryWeek, backlogWeek := 0, 0
714 longest, run, first, bestFirst, bestLast := 0, 0, 0, 0, 0
715 for week, record := range stage.records {
716 t := week + 1
717 if record.inventory > maxInventory {
718 maxInventory = record.inventory
719 inventoryWeek = t
720 }
721 if record.backlog > maxBacklog {
722 maxBacklog = record.backlog
723 backlogWeek = t
724 }
725 if record.backlog > 0 {
726 if run == 0 {
727 first = t
728 }
729 run++
730 if run > longest {
731 longest, bestFirst, bestLast = run, first, t
732 }
733 } else {
734 run = 0
735 }
736 }
737 inventoryText, backlogText, longestText := "-", "-", "0"
738 if inventoryWeek != 0 {
739 inventoryText = fmt.Sprint(inventoryWeek)
740 }
741 if backlogWeek != 0 {
742 backlogText = fmt.Sprint(backlogWeek)
743 }
744 if longest != 0 {
745 longestText = fmt.Sprintf("%d weeks %d-%d", longest, bestFirst, bestLast)
746 }
747 game.emit(stageNames[i], "max_inv", maxInventory, "week", inventoryText,
748 "max_back", maxBacklog, "week", backlogText,
749 "longest_stockout", longestText)
750 }
751}

753func (game *Game) leadtime(index int) {
754 sim := game.sim
755 stage := &sim.stages[index]
756 initial := game.initials[index]
757 nominal := initial.shipping + initial.orders
758 cumulative := make([]int64, len(stage.records))
759 var received int64
760 for i, record := range stage.records {
761 received += record.received
762 cumulative[i] = received - int64(nominal)*initial.quantity
763 }
764 var ordered int64
765 count := 0
766 leads := []int64{}
767 for t, record := range stage.records {
768 ordered += record.order
769 if record.order == 0 {
770 continue
771 }
772 count++
773 for u := t + 1; u < len(stage.records); u++ {
774 if cumulative[u] >= ordered {
775 leads = append(leads, int64(u-t))
776 break
777 }
778 }
779 }
780 minimum, average, maximum := "-", "-", "-"
781 if len(leads) > 0 {
782 min, max := leads[0], leads[0]
783 for _, value := range leads {
784 if value < min {
785 min = value
786 }
787 if value > max {
788 max = value
789 }
790 }
791 minimum, maximum = fmt.Sprint(min), fmt.Sprint(max)
792 average = decimal(sum(leads), int64(len(leads)), 2)
793 }
794 game.emit("LEADTIME", stageNames[index], "nominal", nominal,
795 "orders", count, "completed", len(leads), "open", count-len(leads),
796 "min", minimum, "avg", average, "max", maximum)
797}

799func (game *Game) plot(index int, mode string, first, last int) {
800 values := game.values(index, mode)[first-1 : last]
801 maximum, minimum := values[0], values[0]
802 for _, value := range values {
803 if value > maximum {
804 maximum = value
805 }
806 if value < minimum {
807 minimum = value
808 }
809 }
810 if mode == "NET" {
811 game.emit("PLOT", stageNames[index], mode,
812 fmt.Sprintf("%d-%d", first, last), "min", minimum, "max", maximum)
813 var scale int64
814 for _, value := range values {
815 if value < 0 {
816 value = -value
817 }
818 if value > scale {
819 scale = value
820 }
821 }
822 for i, value := range values {
823 size := 0
824 if scale != 0 {
825 abs := value
826 if abs < 0 {
827 abs = -abs
828 }
829 size = int(abs * 25 / scale)
830 }
831 left := strings.Repeat(" ", 25)
832 rightBar := ""
833 if value < 0 {
834 left = strings.Repeat(" ", 25-size) + strings.Repeat("-", size)
835 } else if value > 0 {
836 rightBar = strings.Repeat("+", size)
837 }
838 game.emit(fmt.Sprintf("%3d %s|%s %d", first+i, left, rightBar, value))
839 }
840 } else {
841 game.emit("PLOT", stageNames[index], mode,
842 fmt.Sprintf("%d-%d", first, last), "max", maximum)
843 for i, value := range values {
844 size := 0
845 if maximum != 0 {
846 size = int(value * 50 / maximum)
847 }
848 game.emit(fmt.Sprintf("%3d |%s %d",
849 first+i, strings.Repeat("#", size), value))
850 }
851 }
852}

854func (game *Game) histogram(index int, mode string, width int64) {
855 values := game.values(index, mode)
856 bins := make(map[int64]int)
857 var min, max int64
858 for i, value := range values {
859 bucket := floorDiv(value, width)
860 bins[bucket]++
861 if i == 0 || bucket < min {
862 min = bucket
863 }
864 if i == 0 || bucket > max {
865 max = bucket
866 }
867 }
868 game.emit("HISTOGRAM", stageNames[index], mode, "width", width,
869 "weeks", fmt.Sprintf("1-%d", game.sim.week))
870 for bucket := min; bucket <= max; bucket++ {
871 count := bins[bucket]
872 text := fmt.Sprintf("%d..%d %d", bucket*width, bucket*width+width-1, count)
873 if count != 0 {
874 text += " " + strings.Repeat("*", count)
875 }
876 game.emit(text)
877 }

880func (game *Game) command(words []string) error {
881 command, args := words[0], words[1:]
882 sim := game.sim
883 switch command {
884 case "DEMAND":
885 values := make([]int64, len(args))
886 for i, token := range args {
887 values[i] = parse(token)
888 }
889 game.demand(values)
890 case "GENERATE":
891 return game.generate(args)
892 case "RUN":
893 if sim.week == len(sim.demands) {
894 return fmt.Errorf("horizon reached at week %d", len(sim.demands))
895 }
896 first := sim.week + 1
897 count := int(parse(args[0]))
898 if count > len(sim.demands)-sim.week {
899 count = len(sim.demands) - sim.week
900 }
901 var cost int64
902 for t := 0; t < count; t++ {
903 records := sim.step()
904 for i, record := range records {
905 cost += record.cost
906 if record.before == 0 && record.backlog > 0 {
907 game.emit("STOCKOUT", stageNames[i], "week", sim.week,
908 "backlog", record.backlog)
909 } else if record.before > 0 && record.backlog == 0 {
910 game.emit("RECOVER", stageNames[i], "week", sim.week)
911 }
912 }
913 }
914 game.emit("RUN", fmt.Sprintf("%d-%d", first, sim.week),
915 "demand", sum(sim.demands[first-1:sim.week]), "cost", money(cost))
916 if sim.week == len(sim.demands) {
917 game.emit("HORIZON", sim.week)
918 }
919 case "RESET":
920 game.initialize(nil)
921 game.emit("RESET", "week", 0)
922 case "SAVE":
923 game.saved[args[0]] = sim.clone()
924 game.emit("SAVE", args[0], "week", sim.week)
925 case "LOAD":
926 snapshot := game.saved[args[0]]
927 if snapshot == nil {
928 return fmt.Errorf("no snapshot %s", args[0])
929 }
930 game.sim = snapshot.clone()
931 game.emit("LOAD", args[0], "week", game.sim.week)
932 case "SET":
933 indices, err := target(args[0])
934 if err != nil {
935 return err
936 }
937 policy, err := parsePolicy(args[1:])
938 if err != nil {
939 return err
940 }
941 for _, index := range indices {
942 sim.setPolicy(index, policy)
943 }
944 game.emit("SET", args[0], policy.text)
945 case "BATCH":
946 index, err := stageIndex(args[0])
947 if err != nil {
948 return err
949 }
950 lot := parse(args[1])
951 if lot < 1 || lot > 100 {
952 return fmt.Errorf("invalid batch size %s", args[1])
953 }
954 sim.stages[index].lot = lot
955 game.emit("BATCH", args[0], lot)
956 case "CAPACITY":
957 index := -1
958 if args[0] != "PLANT" {
959 var err error
960 index, err = stageIndex(args[0])
961 if err != nil {
962 return err
963 }
964 }
965 var capacity *int64
966 if args[1] != "OFF" {
967 value := parse(args[1])
968 if value > 10000 {
969 return fmt.Errorf("invalid capacity %s", args[1])
970 }
971 capacity = &value
972 }
973 if index < 0 {
974 sim.plantCapacity = capacity
975 } else {
976 sim.stages[index].capacity = capacity
977 }
978 game.emit("CAPACITY", args[0], args[1])
979 case "COST":
980 index, err := stageIndex(args[0])
981 if err != nil {
982 return err
983 }
984 holding, backlog := parse(args[1]), parse(args[2])
985 if holding > 1000 || backlog > 1000 {
986 return errors.New("invalid cost")
987 }
988 sim.stages[index].holding = holding
989 sim.stages[index].backlogRate = backlog
990 game.emit("COST", args[0], "holding", holding, "backlog", backlog)
991 case "VISIBILITY":
992 if args[0] != "ON" && args[0] != "OFF" {
993 return fmt.Errorf("unknown visibility %s", args[0])
994 }
995 sim.visibility = args[0]
996 game.emit("VISIBILITY", args[0])
997 case "SHOCK":
998 week := int(parse(args[0]))
999 value := parse(args[1])
1000 if week < 1 || week > len(sim.demands) {
1001 return fmt.Errorf("week %s outside horizon", args[0])
1002 }
1003 if week <= sim.week {
1004 return fmt.Errorf("week %s already simulated", args[0])
1005 }
1006 old := sim.demands[week-1]
1007 sim.demands[week-1] = value
1008 game.emit("SHOCK", "week", week, old, "->", value)
1009 case "STATUS":
1010 game.showStatus()
1011 case "SETTINGS":
1012 game.emit("SETTINGS")
1013 for i, stage := range sim.stages {
1014 capacity := "OFF"
1015 if stage.capacity != nil {
1016 capacity = fmt.Sprint(*stage.capacity)
1017 }
1018 initial := game.initials[i]
1019 game.emit(stageNames[i], "holding", stage.holding,
1020 "backlog", stage.backlogRate, "batch", stage.lot,
1021 "capacity", capacity, "delays", initial.shipping, initial.orders)
1022 }
1023 capacity := "OFF"
1024 if sim.plantCapacity != nil {
1025 capacity = fmt.Sprint(*sim.plantCapacity)
1026 }
1027 game.emit("PLANT", "capacity", capacity)
1028 game.emit("VISIBILITY", sim.visibility)
1029 case "WEEK":
1030 if err := game.requireWeeks(); err != nil {
1031 return err
1032 }
1033 week := int(parse(args[0]))
1034 if week < 1 || week > sim.week {
1035 return fmt.Errorf("bad week %s", args[0])
1036 }
1037 game.showWeek(week)
1038 case "TABLE":
1039 index, err := stageIndex(args[0])
1040 if err != nil {
1041 return err
1042 }
1043 if err := game.requireWeeks(); err != nil {
1044 return err
1045 }
1046 first, last, err := rangeCheck(args[1:], sim.week)
1047 if err != nil {
1048 return err
1049 }
1050 game.showTable(index, first, last)
1051 case "CHAIN":
1052 if err := game.requireWeeks(); err != nil {
1053 return err
1054 }
1055 first, last, err := rangeCheck(args, sim.week)
1056 if err != nil {
1057 return err
1058 }
1059 game.emit("CHAIN", fmt.Sprintf("%d-%d", first, last))
1060 header := []string{right("week", 4), right("demand", 6)}
1061 for _, stage := range stageNames {
1062 header = append(header, right(stage, 11))
1063 }
1064 game.emit(strings.Join(header, " "))
1065 for week := first; week <= last; week++ {
1066 fields := []string{right(fmt.Sprint(week), 4),
1067 right(fmt.Sprint(sim.demands[week-1]), 6)}
1068 for _, stage := range sim.stages {
1069 fields = append(fields,
1070 right(fmt.Sprint(stage.records[week-1].order), 11))
1071 }
1072 game.emit(strings.Join(fields, " "))
1073 }
1074 case "DEMANDS":
1075 first, last, err := rangeCheck(args, len(sim.demands))
1076 if err != nil {
1077 return err
1078 }
1079 values := sim.demands[first-1 : last]
1080 minimum, maximum := values[0], values[0]
1081 for _, value := range values {
1082 if value < minimum {
1083 minimum = value
1084 }
1085 if value > maximum {
1086 maximum = value
1087 }
1088 }
1089 game.emit("DEMANDS", fmt.Sprintf("%d-%d", first, last),
1090 "total", sum(values), "min", minimum, "max", maximum)
1091 parts := make([]string, len(values))
1092 for i, value := range values {
1093 parts[i] = fmt.Sprint(value)
1094 }
1095 game.emit(strings.Join(parts, " "))
1096 case "COSTS":
1097 game.showCosts()
1098 case "SUMMARY":
1099 if err := game.requireWeeks(); err != nil {
1100 return err
1101 }
1102 game.emit("SUMMARY", "weeks", fmt.Sprintf("1-%d", sim.week))
1103 for i, stage := range sim.stages {
1104 var inv, back, orders int64
1105 minimum, maximum := stage.records[0].order, stage.records[0].order
1106 zeroWeeks := 0
1107 for _, record := range stage.records {
1108 inv += record.inventory
1109 back += record.backlog
1110 orders += record.order
1111 if record.order < minimum {
1112 minimum = record.order
1113 }
1114 if record.order > maximum {
1115 maximum = record.order
1116 }
1117 if record.order == 0 {
1118 zeroWeeks++
1119 }
1120 }
1121 game.emit(stageNames[i],
1122 "avg_inv", decimal(inv, int64(sim.week), 2),
1123 "avg_back", decimal(back, int64(sim.week), 2),
1124 "orders", orders, "min", minimum, "max", maximum,
1125 "zero_weeks", zeroWeeks)
1126 }
1127 case "BALANCE":
1128 game.emit("BALANCE", "week", sim.week)
1129 var initialGoods, produced, delivered, stock, transit int64
1130 for i, stage := range sim.stages {
1131 var received, shipped int64
1132 for _, record := range stage.records {
1133 received += record.received
1134 shipped += record.shipped
1135 }
1136 game.emit(stageNames[i], "start", game.initials[i].inventory,
1137 "received", received, "shipped", shipped,
1138 "end", stage.inventory, "owed", stage.backlog)
1139 initial := game.initials[i]
1140 initialGoods += initial.inventory + initial.quantity*int64(initial.shipping)
1141 if i == 0 {
1142 delivered = shipped
1143 }
1144 stock += stage.inventory
1145 transit += sum(stage.shipping)
1146 }
1147 for _, record := range sim.plantRecords {
1148 produced += record.produced
1149 }
1150 game.emit("SYSTEM", "initial", initialGoods, "produced", produced,
1151 "delivered", delivered, "stock", stock, "transit", transit)
1152 case "FILLRATE":
1153 if err := game.requireWeeks(); err != nil {
1154 return err
1155 }
1156 game.emit("FILLRATE", "weeks", fmt.Sprintf("1-%d", sim.week))
1157 for i, stage := range sim.stages {
1158 ordered, ontime, fill := fillText(stage.records)
1159 stockouts := 0
1160 for _, record := range stage.records {
1161 if record.backlog > 0 {
1162 stockouts++
1163 }
1164 }
1165 service := decimal(int64(sim.week-stockouts)*100, int64(sim.week), 2) + "%"
1166 game.emit(stageNames[i], "fill", fill, "ontime", ontime,
1167 "ordered", ordered, "stockout_weeks", stockouts, "service", service)
1168 }
1169 case "BULLWHIP":
1170 if err := game.requireWeeks(); err != nil {
1171 return err
1172 }
1173 game.bullwhip()
1174 case "PEAKS":
1175 if err := game.requireWeeks(); err != nil {
1176 return err
1177 }
1178 game.peaks()
1179 case "LEADTIME":
1180 index, err := stageIndex(args[0])
1181 if err != nil {
1182 return err
1183 }
1184 if err := game.requireWeeks(); err != nil {
1185 return err
1186 }
1187 game.leadtime(index)
1188 case "PLOT", "HISTOGRAM":
1189 index, err := stageIndex(args[0])
1190 if err != nil {
1191 return err
1192 }
1193 if err := modeCheck(args[1]); err != nil {
1194 return err
1195 }
1196 width := int64(0)
1197 if command == "HISTOGRAM" {
1198 width = parse(args[2])
1199 if width < 1 || width > 100 {
1200 return fmt.Errorf("invalid width %s", args[2])
1201 }
1202 }
1203 if err := game.requireWeeks(); err != nil {
1204 return err
1205 }
1206 if command == "PLOT" {
1207 first, last, err := rangeCheck(args[2:], sim.week)
1208 if err != nil {
1209 return err
1210 }
1211 game.plot(index, args[1], first, last)
1212 } else {
1213 game.histogram(index, args[1], width)
1214 }
1215 case "COMPARE":
1216 policies := []Policy{}
1217 start := 0
1218 for end := 0; end <= len(args); end++ {
1219 if end == len(args) || args[end] == "/" {
1220 policy, err := parsePolicy(args[start:end])
1221 if err != nil {
1222 return err
1223 }
1224 policies = append(policies, policy)
1225 start = end + 1
1226 }
1227 }
1228 type Row struct {
1229 cost, index int64
1230 policy string
1231 fill string
1232 ratio string
1233 }
1234 rows := []Row{}
1235 for i, policy := range policies {
1236 selected := [4]Policy{policy, policy, policy, policy}
1237 trial := game.trial(selected)
1238 costs := trial.costs()
1239 var cost int64
1240 for _, value := range costs {
1241 cost += value
1242 }
1243 _, _, fill := fillText(trial.stages[0].records)
1244 demandVar := variation(trial.demands)
1245 factoryOrders := make([]int64, len(trial.stages[3].records))
1246 for j, record := range trial.stages[3].records {
1247 factoryOrders[j] = record.order
1248 }
1249 factoryVar := variation(factoryOrders)
1250 ratio := "n/a"
1251 if demandVar != 0 {
1252 ratio = decimal(factoryVar, demandVar, 4)
1253 }
1254 rows = append(rows, Row{cost, int64(i), policy.text, fill, ratio})
1255 }
1256 sort.Slice(rows, func(i, j int) bool {
1257 if rows[i].cost != rows[j].cost {
1258 return rows[i].cost < rows[j].cost
1259 }
1260 return rows[i].index < rows[j].index
1261 })
1262 game.emit("COMPARE", len(policies), "weeks", len(sim.demands))
1263 for i, row := range rows {
1264 game.emit(i+1, row.policy, "total", money(row.cost),
1265 "fill", row.fill, "bullwhip", row.ratio)
1266 }
1267 case "OPTIMIZE":
1268 indices, err := target(args[0])
1269 if err != nil {
1270 return err
1271 }
1272 low, high := int(parse(args[1])), int(parse(args[2]))
1273 if low > high {
1274 return fmt.Errorf("bad range %s %s", args[1], args[2])
1275 }
1276 if high-low > 200 {
1277 return errors.New("range too large")
1278 }
1279 var bestCost, bestTotal int64
1280 bestTarget := low
1281 first := true
1282 for value := low; value <= high; value++ {
1283 var policies [4]Policy
1284 for i, stage := range sim.stages {
1285 policies[i] = stage.policy
1286 }
1287 replacement := Policy{"BASESTOCK", []int64{int64(value)},
1288 fmt.Sprintf("BASESTOCK %d", value)}
1289 for _, index := range indices {
1290 policies[index] = replacement
1291 }
1292 trial := game.trial(policies)
1293 costs := trial.costs()
1294 var objective, totalCost int64
1295 for _, index := range indices {
1296 objective += costs[index]
1297 }
1298 for _, value := range costs {
1299 totalCost += value
1300 }
1301 if first || objective < bestCost {
1302 bestCost, bestTotal, bestTarget = objective, totalCost, value
1303 first = false
1304 }
1305 }
1306 game.emit("OPTIMIZE", args[0], "BASESTOCK", bestTarget,
1307 "cost", money(bestCost), "total", money(bestTotal))
1308 default:
1309 return fmt.Errorf("unknown command %s", command)
1310 }
1311 return nil
1312}
