10fn money(value: i64) -> String {

14fn decimal(numerator: i64, denominator: i64, places: usize) -> String {
15 let scale = 10_i64.pow(places as u32);
16 let value = numerator * scale / denominator;
17 format!("{}.{:0width$}", value / scale, value % scale, width = places)
18}

20fn ratio(numerator: i64, denominator: i64, places: usize) -> String {
21 if denominator == 0 { "n/a".into() } else { decimal(numerator, denominator, places) }

40fn range(args: &[&str], limit: usize) -> Result<(usize, usize), String> {
41 if args.is_empty() { return Ok((1, limit)); }
42 let (from, to) = (number(args[0]) as usize, number(args[1]) as usize);
43 if from < 1 || from > to || to > limit {
44 Err(format!("bad range {} {}", args[0], args[1]))
45 } else {
46 Ok((from, to))
47 }

50fn variation(values: &[i64]) -> i64 {
51 let sum: i64 = values.iter().sum();
52 values.len() as i64 * values.iter().map(|value| value * value).sum::<i64>() - sum * sum
53}

55#[derive(Clone, Copy)]
56enum Rule {
57 Fixed(i64),
58 Pass,
59 BaseStock(i64),
60 MinMax(i64, i64),
61 MovingAverage(usize, i64),
62 Smooth(i64, i64),
63 Sterman(i64, i64, i64, i64),
64}
65
66#[derive(Clone)]
67struct Policy {
68 rule: Rule,
69 text: String,
70}
71
72impl Policy {
73 fn parse(tokens: &[&str]) -> Result<Self, String> {
74 let name = tokens[0];
75 let count = match name {
76 "PASS" => 0,
77 "FIXED" | "BASESTOCK" => 1,
78 "MINMAX" | "MOVAVG" | "SMOOTH" => 2,
79 "STERMAN" => 4,
80 _ => return Err(format!("unknown policy {name}")),
81 };
82 if tokens.len() != count + 1 { return Err(format!("{name} needs {count} parameters")); }
83 let values: Vec<i64> = tokens[1..].iter().enumerate().map(|(index, text)| {
84 if name == "STERMAN" && index < 3 {
85 let (whole, fraction) = text.split_once('.').unwrap();
86 number(whole) * 100 + number(fraction)
87 } else { number(text) }
88 }).collect();
89 let valid = match name {
90 "FIXED" => values[0] <= 1000,
91 "BASESTOCK" => values[0] <= 10000,
92 "MINMAX" => values[0] < values[1] && values[1] <= 10000,
93 "MOVAVG" => (1..=52).contains(&values[0]) && values[1] <= 10000,
94 "SMOOTH" => (1..=10).contains(&values[0]) && values[1] <= 10000,
95 "STERMAN" => values[..3].iter().all(|value| *value <= 100) && values[3] <= 10000,
96 _ => true,
97 };
98 if !valid { return Err(format!("invalid parameter for {name}")); }
99 let rule = match name {
100 "FIXED" => Rule::Fixed(values[0]),
101 "PASS" => Rule::Pass,
102 "BASESTOCK" => Rule::BaseStock(values[0]),
103 "MINMAX" => Rule::MinMax(values[0], values[1]),
104 "MOVAVG" => Rule::MovingAverage(values[0] as usize, values[1]),
105 "SMOOTH" => Rule::Smooth(values[0], values[1]),
106 _ => Rule::Sterman(values[0], values[1], values[2], values[3]),
107 };
108 Ok(Self { rule, text: tokens.join(" ") })
109 }
110
111 fn base_stock(level: i64) -> Self {
112 Self { rule: Rule::BaseStock(level), text: format!("BASESTOCK {level}") }
113 }
114}

225 fn position(&self, index: usize) -> (i64, i64) {
226 let stage = &self.stages[index];
227 let upstream = if index == 3 { self.plant.backlog } else { self.stages[index + 1].backlog };
228 let supply = stage.orders.iter().sum::<i64>() + upstream + stage.shipping.iter().sum::<i64>();
229 (supply, stage.inventory + supply - stage.backlog)
230 }

232 fn step(&mut self) {
233 let mut records = [Record::default(); 4];
234 for (stage, record) in self.stages.iter_mut().zip(&mut records) {
235 record.received = stage.shipping.pop_front().unwrap();
236 stage.inventory += record.received;
237 }
238 let mut incoming = [0; 5];
239 incoming[0] = self.demand[self.week];
240 for (index, stage) in self.stages.iter_mut().enumerate() {
241 incoming[index + 1] = stage.orders.pop_front().unwrap();
242 }
243 for index in 0..4 {
244 let stage = &mut self.stages[index];
245 let record = &mut records[index];
246 record.incoming = incoming[index];
247 record.previous_backlog = stage.backlog;
248 let requested = stage.backlog + record.incoming;
249 record.shipped = stage.inventory.min(requested).min(stage.capacity.unwrap_or(i64::MAX));
250 record.on_time = (record.shipped - stage.backlog).max(0).min(record.incoming);
251 stage.inventory -= record.shipped;
252 stage.backlog = requested - record.shipped;
253 record.inventory = stage.inventory;
254 record.backlog = stage.backlog;
255 if index > 0 { self.stages[index - 1].shipping.push_back(record.shipped); }
256 }
257 let requested = self.plant.backlog + incoming[4];
258 let produced = requested.min(self.plant.capacity.unwrap_or(i64::MAX));
259 self.plant.backlog = requested - produced;
260 self.plant.records.push((incoming[4], produced, self.plant.backlog));
261 self.stages[3].shipping.push_back(produced);
262
263 let positions: Vec<(i64, i64)> = (0..4).map(|index| self.position(index)).collect();
264 for index in 0..4 {
265 let (supply, position) = positions[index];
266 let signal = if self.visible { incoming[0] } else { incoming[index] };
267 let stage = &self.stages[index];
268 let order = match stage.policy.rule {
269 Rule::Fixed(quantity) => quantity,
270 Rule::Pass => incoming[index],
271 Rule::BaseStock(level) => (level - position).max(0),
272 Rule::MinMax(low, high) => if position <= low { high - position } else { 0 },
273 Rule::MovingAverage(window, level) => {
274 let count = window.min(self.week + 1);
275 let sum = signal + (self.week + 2 - count..=self.week)
276 .map(|week| self.signal(index, week)).sum::<i64>();
277 let forecast = sum / count as i64;
278 (level + forecast * (stage.shipping_delay + stage.order_delay) as i64 - position).max(0)
279 }
280 Rule::Smooth(window, level) => (signal + (level - position).div_euclid(window)).max(0),
281 Rule::Sterman(theta, alpha, beta, target) => {
282 let expected = (100 * theta * signal + (100 - theta) * stage.expected) / 100;
283 let net = stage.inventory - stage.backlog;
284 let order = (100 * expected + alpha * (100 * (target - net) - beta * supply))
285 .div_euclid(10000).max(0);
286 self.stages[index].expected = expected;
287 order
288 }
289 };
290 let stage = &mut self.stages[index];
291 let record = &mut records[index];
292 record.order = (order + stage.lot - 1) / stage.lot * stage.lot;
293 stage.orders.push_back(record.order);
294 record.holding_cost = stage.holding * stage.inventory;
295 record.backlog_cost = stage.backlog_rate * stage.backlog;
296 record.cost = record.holding_cost + record.backlog_cost;
297 }
298 for (stage, record) in self.stages.iter_mut().zip(records) { stage.records.push(record); }
299 self.week += 1;
300 }

375 fn bullwhip(&self) {
376 let n = self.week as i64;
377 let demand = &self.demand[..self.week];
378 let demand_variation = variation(demand);
379 let mut previous = demand_variation;
380 println!("BULLWHIP weeks 1-{n} demand mean {} var {}", decimal(demand.iter().sum(), n, 4),
381 decimal(demand_variation, n * n, 4));
382 for (name, stage) in STAGES.iter().zip(&self.stages) {
383 let orders: Vec<i64> = stage.records.iter().map(|record| record.order).collect();
384 let current = variation(&orders);
385 println!("{name} mean {} var {} ratio {} step {}", decimal(orders.iter().sum(), n, 4),
386 decimal(current, n * n, 4), ratio(current, demand_variation, 4), ratio(current, previous, 4));
387 previous = current;
388 }

391 fn peaks(&self) {
392 println!("PEAKS weeks 1-{}", self.week);
393 for (name, stage) in STAGES.iter().zip(&self.stages) {
394 let (mut max_inventory, mut inventory_week, mut max_backlog, mut backlog_week) = (0, 0, 0, 0);
395 let (mut run, mut longest, mut end) = (0, 0, 0);
396 for (index, record) in stage.records.iter().enumerate() {
397 if record.inventory > max_inventory { max_inventory = record.inventory; inventory_week = index + 1; }
398 if record.backlog > max_backlog { max_backlog = record.backlog; backlog_week = index + 1; }
399 run = if record.backlog > 0 { run + 1 } else { 0 };
400 if run > longest { longest = run; end = index + 1; }
401 }
402 println!("{name} max_inv {max_inventory} week {} max_back {max_backlog} week {} longest_stockout {}",
403 if inventory_week == 0 { "-".into() } else { inventory_week.to_string() },
404 if backlog_week == 0 { "-".into() } else { backlog_week.to_string() },
405 if longest == 0 { "0".into() } else { format!("{longest} weeks {}-{end}", end + 1 - longest) });
406 }

463 fn lead_time(&self, index: usize) {
464 let stage = &self.current.stages[index];
465 let nominal = stage.shipping_delay + stage.order_delay;
466 let mut received = -(nominal as i64 * self.configs[index].quantity);
467 let cumulative: Vec<i64> = stage.records.iter().map(|record| { received += record.received; received }).collect();
468 let mut ordered = 0;
469 let mut count = 0;
470 let mut lead_times = Vec::new();
471 for (week, record) in stage.records.iter().enumerate() {
472 ordered += record.order;
473 if record.order == 0 { continue; }
474 count += 1;
475 if let Some(arrival) = (week + 1..self.current.week).find(|arrival| cumulative[*arrival] >= ordered) {
476 lead_times.push((arrival - week) as i64);
477 }
478 }
479 let statistics = if lead_times.is_empty() { "min - avg - max -".into() } else {
480 format!("min {} avg {} max {}", lead_times.iter().min().unwrap(),
481 decimal(lead_times.iter().sum(), lead_times.len() as i64, 2), lead_times.iter().max().unwrap())
482 };
483 println!("LEADTIME {} nominal {nominal} orders {count} completed {} open {} {statistics}",
484 STAGES[index], lead_times.len(), count - lead_times.len());
485 }

511 fn command(&mut self, tokens: &[&str]) -> Result<(), String> {
512 let a = &tokens[1..];
513 match tokens[0] {
514 "DEMAND" => self.set_demand(a.iter().map(|text| number(text)).collect()),
515 "GENERATE" => {
516 let weeks = number(a[0]);
517 let (base, spread, mut random) = (number(a[1]), number(a[2]), number(a[3]));
518 let changes: Vec<(&str, i64, i64)> = a[4..].chunks(3)
519 .map(|change| (change[0], number(change[1]), number(change[2]))).collect();
520 for (_, week, _) in &changes {
521 if *week < 1 || *week > weeks { return Err(format!("week {week} outside horizon")); }
522 }
523 let mut demand = Vec::new();
524 for week in 1..=weeks {
525 let mut level = base;
526 for (kind, start, amount) in &changes {
527 level += match *kind {
528 "STEP" if week >= *start => *amount,
529 "PULSE" if week == *start => *amount,
530 "RAMP" if week >= *start => amount * (week - start + 1),
531 _ => 0,
532 };
533 }
534 if level < 0 { return Err(format!("negative level at week {week}")); }
535 random = random * 48271 % 2147483647;
536 demand.push(level + random % spread);
537 }
538 self.set_demand(demand);
539 }
540 "RUN" => {
541 let simulation = &mut self.current;
542 let horizon = simulation.demand.len();
543 if simulation.week == horizon { return Err(format!("horizon reached at week {horizon}")); }
544 let from = simulation.week + 1;
545 let to = horizon.min(simulation.week + number(a[0]) as usize);
546 let mut cost = 0;
547 while simulation.week < to {
548 simulation.step();
549 for (name, stage) in STAGES.iter().zip(&simulation.stages) {
550 let record = stage.records.last().unwrap();
551 cost += record.cost;
552 if record.previous_backlog == 0 && record.backlog > 0 {
553 println!("STOCKOUT {name} week {} backlog {}", simulation.week, record.backlog);
554 } else if record.previous_backlog > 0 && record.backlog == 0 {
555 println!("RECOVER {name} week {}", simulation.week);
556 }
557 }
558 }
559 println!("RUN {from}-{to} demand {} cost {}", simulation.demand[from - 1..to].iter().sum::<i64>(), money(cost));
560 if simulation.week == horizon { println!("HORIZON {horizon}"); }
561 }
562 "RESET" => { self.reset(); println!("RESET week 0"); }
563 "SAVE" => {
564 self.saved.insert(a[0].into(), self.current.clone());
565 println!("SAVE {} week {}", a[0], self.current.week);
566 }
567 "LOAD" => {
568 let snapshot = self.saved.get(a[0]).ok_or_else(|| format!("no snapshot {}", a[0]))?;
569 self.current = snapshot.clone();
570 println!("LOAD {} week {}", a[0], self.current.week);
571 }
572 "SET" => {
573 let selected = targets(a[0])?;
574 let policy = Policy::parse(&a[1..])?;
575 for index in selected {
576 let expected = if self.current.week == 0 { self.configs[index].quantity }
577 else { self.current.signal(index, self.current.week) } * 100;
578 let stage = &mut self.current.stages[index];
579 stage.policy = policy.clone();
580 if matches!(policy.rule, Rule::Sterman(..)) { stage.expected = expected; }
581 }
582 println!("SET {} {}", a[0], policy.text);
583 }
584 "BATCH" => {
585 let index = stage_index(a[0])?;
586 let lot = number(a[1]);
587 if !(1..=100).contains(&lot) { return Err(format!("invalid batch size {}", a[1])); }
588 self.current.stages[index].lot = lot;
589 println!("BATCH {} {lot}", a[0]);
590 }
591 "CAPACITY" => {
592 let index = if a[0] == "PLANT" { None } else { Some(stage_index(a[0])?) };
593 let capacity = if a[1] == "OFF" { None } else { Some(number(a[1])) };
594 if capacity.is_some_and(|value| value > 10000) { return Err(format!("invalid capacity {}", a[1])); }
595 if let Some(index) = index { self.current.stages[index].capacity = capacity; }
596 else { self.current.plant.capacity = capacity; }
597 println!("CAPACITY {} {}", a[0], a[1]);
598 }
599 "COST" => {
600 let index = stage_index(a[0])?;
601 let (holding, backlog) = (number(a[1]), number(a[2]));
602 if holding > 1000 || backlog > 1000 { return Err("invalid cost".into()); }
603 self.current.stages[index].holding = holding;
604 self.current.stages[index].backlog_rate = backlog;
605 println!("COST {} holding {holding} backlog {backlog}", a[0]);
606 }
607 "VISIBILITY" => {
608 if !["ON", "OFF"].contains(&a[0]) { return Err(format!("unknown visibility {}", a[0])); }
609 self.current.visible = a[0] == "ON";
610 println!("VISIBILITY {}", a[0]);
611 }
612 "SHOCK" => {
613 let week = number(a[0]) as usize;
614 if week < 1 || week > self.current.demand.len() { return Err(format!("week {} outside horizon", a[0])); }
615 if week <= self.current.week { return Err(format!("week {} already simulated", a[0])); }
616 let demand = number(a[1]);
617 println!("SHOCK week {week} {} -> {demand}", self.current.demand[week - 1]);
618 self.current.demand[week - 1] = demand;
619 }
620 "STATUS" => self.current.status(),
621 "SETTINGS" => self.current.settings(),
622 "WEEK" => {
623 self.current.require_weeks()?;
624 let week = number(a[0]) as usize;
625 if week < 1 || week > self.current.week { return Err(format!("bad week {}", a[0])); }
626 self.current.show_week(week);
627 }
628 "TABLE" => {
629 let index = stage_index(a[0])?;
630 self.current.require_weeks()?;
631 let (from, to) = range(&a[1..], self.current.week)?;
632 println!("TABLE {} {from}-{to}", a[0]);
633 println!("week in ship recv inv back order cost");
634 for week in from..=to {
635 let record = self.current.stages[index].records[week - 1];
636 println!("{week:>4} {:>6} {:>6} {:>6} {:>6} {:>6} {:>6} {:>9}", record.incoming,
637 record.shipped, record.received, record.inventory, record.backlog, record.order, money(record.cost));
638 }
639 }
640 "CHAIN" => {
641 self.current.require_weeks()?;
642 let (from, to) = range(a, self.current.week)?;
643 println!("CHAIN {from}-{to}");
644 println!("{:>4} {:>6} {:>11} {:>11} {:>11} {:>11}", "week", "demand", STAGES[0], STAGES[1], STAGES[2], STAGES[3]);
645 for week in from..=to {
646 println!("{week:>4} {:>6} {:>11} {:>11} {:>11} {:>11}", self.current.demand[week - 1],
647 self.current.stages[0].records[week - 1].order, self.current.stages[1].records[week - 1].order,
648 self.current.stages[2].records[week - 1].order, self.current.stages[3].records[week - 1].order);
649 }
650 }
651 "DEMANDS" => {
652 let (from, to) = range(a, self.current.demand.len())?;
653 let demand = &self.current.demand[from - 1..to];
654 println!("DEMANDS {from}-{to} total {} min {} max {}", demand.iter().sum::<i64>(),
655 demand.iter().min().unwrap(), demand.iter().max().unwrap());
656 println!("{}", demand.iter().map(ToString::to_string).collect::<Vec<_>>().join(" "));
657 }
658 "COSTS" => self.current.costs(),
659 "SUMMARY" => { self.current.require_weeks()?; self.current.summary(); }
660 "BALANCE" => self.balance(),
661 "FILLRATE" => { self.current.require_weeks()?; self.current.fill_rates(); }
662 "BULLWHIP" => { self.current.require_weeks()?; self.current.bullwhip(); }
663 "PEAKS" => { self.current.require_weeks()?; self.current.peaks(); }
664 "LEADTIME" => {
665 let index = stage_index(a[0])?;
666 self.current.require_weeks()?;
667 self.lead_time(index);
668 }
669 "PLOT" | "HISTOGRAM" => {
670 let index = stage_index(a[0])?;
671 let kind = a[1];
672 if !["ORDERS", "INV", "BACK", "NET"].contains(&kind) { return Err(format!("unknown mode {kind}")); }
673 if tokens[0] == "PLOT" {
674 self.current.require_weeks()?;
675 let (from, to) = range(&a[2..], self.current.week)?;
676 self.plot(index, kind, from, to);
677 } else {
678 let width = number(a[2]);
679 if !(1..=100).contains(&width) { return Err(format!("invalid width {}", a[2])); }
680 self.current.require_weeks()?;
681 let mut bins: BTreeMap<i64, usize> = BTreeMap::new();
682 for record in &self.current.stages[index].records {
683 *bins.entry(record.value(kind).div_euclid(width)).or_default() += 1;
684 }
685 println!("HISTOGRAM {} {kind} width {width} weeks 1-{}", a[0], self.current.week);
686 for bin in *bins.first_key_value().unwrap().0..=*bins.last_key_value().unwrap().0 {
687 let count = bins.get(&bin).copied().unwrap_or(0);
688 println!("{}..{} {count}{}", bin * width, bin * width + width - 1,
689 if count == 0 { String::new() } else { format!(" {}", "*".repeat(count)) });
690 }
691 }
692 }
693 "COMPARE" => {
694 let policies: Vec<Policy> = a.split(|token| *token == "/").map(Policy::parse).collect::<Result<_, _>>()?;
695 let demand_variation = variation(&self.current.demand);
696 let mut results = Vec::new();
697 for (index, policy) in policies.iter().enumerate() {
698 let trial = self.trial(&vec![policy.clone(); 4]);
699 let orders: Vec<i64> = trial.stages[3].records.iter().map(|record| record.order).collect();
700 results.push((trial.total_cost(), index, trial.stages[0].fill_rate(), ratio(variation(&orders), demand_variation, 4)));
701 }
702 results.sort_by_key(|row| (row.0, row.1));
703 println!("COMPARE {} weeks {}", policies.len(), self.current.demand.len());
704 for (rank, (cost, index, fill, bullwhip)) in results.iter().enumerate() {
705 println!("{} {} total {} fill {fill} bullwhip {bullwhip}", rank + 1, policies[*index].text, money(*cost));
706 }
707 }
708 "OPTIMIZE" => {
709 let selected = targets(a[0])?;
710 let (low, high) = (number(a[1]), number(a[2]));
711 if low > high { return Err(format!("bad range {} {}", a[1], a[2])); }
712 if high - low > 200 { return Err("range too large".into()); }
713 let mut best = None;
714 for level in low..=high {
715 let mut policies: Vec<Policy> = self.current.stages.iter().map(|stage| stage.policy.clone()).collect();
716 for index in &selected { policies[*index] = Policy::base_stock(level); }
717 let trial = self.trial(&policies);
718 let total = trial.total_cost();
719 let objective = if selected.len() == 4 { total } else { trial.stages[selected[0]].costs()[2] };
720 if best.is_none_or(|(cost, _, _)| objective < cost) { best = Some((objective, level, total)); }
721 }
722 let (objective, level, total) = best.unwrap();
723 println!("OPTIMIZE {} BASESTOCK {level} cost {} total {}", a[0], money(objective), money(total));
724 }
725 _ => return Err(format!("unknown command {}", tokens[0])),
726 }
727 Ok(())
728 }
729}

731fn main() {
732 let mut input = String::new();
733 io::stdin().read_to_string(&mut input).unwrap();
734 let mut lines = input.lines();
735 let configs: Vec<Configuration> = (0..4).map(|_| {
736 let parts: Vec<&str> = lines.next().unwrap().split_ascii_whitespace().collect();
737 Configuration {
738 inventory: number(parts[1]), quantity: number(parts[2]),
739 shipping_delay: number(parts[3]) as usize, order_delay: number(parts[4]) as usize,
740 holding: number(parts[5]), backlog_rate: number(parts[6]), policy: Policy::parse(&parts[7..]).unwrap(),
741 }
742 }).collect();
743 let current = Simulation::new(&configs, Vec::new());
744 let mut game = Game { configs, current, base_demand: Vec::new(), saved: BTreeMap::new() };
745 for line in lines {
746 let tokens: Vec<&str> = line.split_ascii_whitespace().collect();
747 if tokens[0] == "END" { break; }
748 if let Err(error) = game.command(&tokens) { println!("ERROR {error}"); }
749 }
