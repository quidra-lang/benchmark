63 static String quoted(String text) {
64 return text == null ? "(nil)" : "\"" + text.replace("\\", "\\\\").replace("\"", "\\\"") + "\"";
65 }
66 static String integerReply(int value) { return "(integer) " + value; }
67 static String arrayReply(Collection<String> values) {
68 if (values.isEmpty()) return "(empty array)";
69 int width = Integer.toString(values.size()).length(), index = 0;
70 var result = new StringBuilder();
71 for (String value : values) {
72 if (index > 0) result.append('\n');
73 String number = Integer.toString(++index);
74 result.append(" ".repeat(width - number.length())).append(number).append(") ").append(quoted(value));
75 }
76 return result.toString();
77 }

79 static List<String> splitLine(String line) {
80 var words = new ArrayList<String>();
81 int position = 0;
82 while (position < line.length()) {
83 if (line.charAt(position) == ' ') { position++; continue; }
84 if (line.charAt(position) != '"') {
85 int start = position;
86 while (position < line.length() && line.charAt(position) != ' ') position++;
87 words.add(line.substring(start, position));
88 continue;
89 }
90 position++;
91 var word = new StringBuilder();
92 while (position < line.length() && line.charAt(position) != '"') {
93 char c = line.charAt(position++);
94 if (c == '\\') {
95 if (position == line.length()) return null;
96 c = line.charAt(position++);
97 if (c != '\\' && c != '"') return null;
98 }
99 word.append(c);
100 }
101 if (position == line.length()) return null;
102 position++;
103 if (position < line.length() && line.charAt(position) != ' ') return null;
104 words.add(word.toString());
105 }
106 return words;
107 }

109 static Integer parseInteger(String text) {
110 if (text.equals("0")) return 0;
111 int start = text.startsWith("-") ? 1 : 0;
112 int length = text.length() - start;
113 if (length < 1 || length > 9 || text.charAt(start) < '1' || text.charAt(start) > '9') return null;
114 for (int i = start; i < text.length(); i++) if (text.charAt(i) < '0' || text.charAt(i) > '9') return null;
115 return Integer.valueOf(text);
116 }
117 static int number(String text) {
118 Integer value = parseInteger(text);
119 if (value == null) throw new CommandError(NOT_INTEGER);
120 return value;
121 }
122 static int increment(int old, int delta) {
123 int value = old + delta;
124 if (value < -999_999_999 || value > 999_999_999)
125 throw new CommandError("ERR increment or decrement would overflow");
126 return value;
127 }
128 static Range range(int length, int start, int stop) {
129 if (start < 0) start += length;
130 if (stop < 0) stop += length;
131 start = Math.max(0, start);
132 stop = Math.min(length - 1, stop);
133 return start <= stop ? new Range(start, stop + 1) : new Range(0, 0);
134 }

136 sealed interface Bound permits Infinity, FiniteBound { boolean accepts(int score, boolean lower); }
137 enum Infinity implements Bound {
138 NEGATIVE, POSITIVE;
139 public boolean accepts(int score, boolean lower) { return lower == (this == NEGATIVE); }
140 }
141 record FiniteBound(int value, boolean exclusive) implements Bound {
142 public boolean accepts(int score, boolean lower) {
143 return (lower ? score > value : score < value) || (!exclusive && score == value);
144 }
145 }
146 static Bound bound(String text) {
147 if (text.equals("-inf")) return Infinity.NEGATIVE;
148 if (text.equals("+inf")) return Infinity.POSITIVE;
149 boolean exclusive = text.startsWith("(");
150 Integer value = parseInteger(exclusive ? text.substring(1) : text);
151 if (value == null) throw new CommandError("ERR min or max is not a float");
152 return new FiniteBound(value, exclusive);
153 }

155 sealed interface PatternPart permits Literal, Wildcard, CharacterClass { boolean accepts(char c); }
156 record Literal(char value) implements PatternPart {
157 public boolean accepts(char c) { return value == c; }
158 }
159 enum Wildcard implements PatternPart {
160 ANY, STAR;
161 public boolean accepts(char c) { return true; }
162 }
163 record CharacterClass(boolean[] members, boolean negated) implements PatternPart {
164 public boolean accepts(char c) { return members[c] != negated; }
165 }
166 static List<PatternPart> pattern(String text) {
167 var parts = new ArrayList<PatternPart>();
168 for (int i = 0; i < text.length(); i++) {
169 char c = text.charAt(i);
170 if (c == '\\') {
171 if (++i == text.length()) throw new CommandError("ERR invalid pattern");
172 parts.add(new Literal(text.charAt(i)));
173 } else if (c == '[') {
174 int end = text.indexOf(']', i + 1);
175 if (end < 0) throw new CommandError("ERR invalid pattern");
176 i++;
177 boolean negated = i < end && text.charAt(i) == '^';
178 if (negated) i++;
179 if (i == end) throw new CommandError("ERR invalid pattern");
180 var members = new boolean[128];
181 while (i < end) {
182 int first = text.charAt(i);
183 if (i + 2 < end && text.charAt(i + 1) == '-') {
184 int last = text.charAt(i + 2);
185 Arrays.fill(members, Math.min(first, last), Math.max(first, last) + 1, true);
186 i += 3;
187 } else { members[first] = true; i++; }
188 }
189 parts.add(new CharacterClass(members, negated));
190 } else if (c == '*') parts.add(Wildcard.STAR);
191 else if (c == '?') parts.add(Wildcard.ANY);
192 else parts.add(new Literal(c));
193 }
194 return parts;
195 }
196 static boolean matches(List<PatternPart> parts, String text) {
197 var previous = new boolean[text.length() + 1];
198 previous[0] = true;
199 for (var part : parts) {
200 var next = new boolean[previous.length];
201 if (part == Wildcard.STAR) {
202 next[0] = previous[0];
203 for (int i = 1; i < next.length; i++) next[i] = previous[i] || next[i - 1];
204 } else {
205 for (int i = 0; i < text.length(); i++) next[i + 1] = previous[i] && part.accepts(text.charAt(i));
206 }
207 previous = next;
208 }
209 return previous[text.length()];
210 }

212 record Arity(int minimum, int maximum, int parity) {
213 Arity(int count) { this(count, count, -1); }
214 Arity(int minimum, int maximum) { this(minimum, maximum, -1); }
215 boolean accepts(int count) {
216 return count >= minimum && (maximum < 0 || count <= maximum) && (parity < 0 || count % 2 == parity);
217 }
218 }

244 <T extends Value> T read(String key, Class<T> type, T empty) {
245 var entry = database.get(key);
246 if (entry == null) return empty;
247 if (!type.isInstance(entry.value)) throw new CommandError(WRONG_TYPE);
248 return type.cast(entry.value);
249 }
250 String text(String key) { return read(key, TextValue.class, new TextValue("")).text; }
251 List<String> list(String key) { return read(key, ListValue.class, new ListValue(List.of())).items; }
252 Map<String, String> hash(String key) { return read(key, HashValue.class, new HashValue(Map.of())).fields; }
253 Set<String> set(String key) { return read(key, SetValue.class, new SetValue(Set.of())).members; }
254 Map<String, Integer> sortedSet(String key) { return read(key, SortedSetValue.class, new SortedSetValue(Map.of())).scores; }
255 void replace(String key, Value value, Integer expiry) {
256 if (value.empty()) database.remove(key);
257 else database.put(key, new Entry(value, expiry));
258 }
259 void update(String key, Value value) {
260 var old = database.get(key);
261 replace(key, value, old == null ? null : old.expiry);
262 }
263 int purge() {
264 int before = database.size();
265 database.values().removeIf(entry -> entry.expiry != null && entry.expiry <= time);
266 return before - database.size();
267 }

269 String textCommand(String command, List<String> a) {
270 String key = a.getFirst();
271 switch (command) {
272 case "SET" -> {
273 var options = new TreeSet<String>();
274 String secondsText = null;
275 for (int i = 2; i < a.size(); i++) {
276 String option = a.get(i).toUpperCase(Locale.ROOT);
277 if (!Set.of("NX", "XX", "EX", "KEEPTTL", "GET").contains(option) || !options.add(option))
278 throw new CommandError("ERR syntax error");
279 if (option.equals("EX")) {
280 if (++i == a.size()) throw new CommandError("ERR syntax error");
281 secondsText = a.get(i);
282 }
283 }
284 if ((options.contains("NX") && options.contains("XX")) || (options.contains("EX") && options.contains("KEEPTTL")))
285 throw new CommandError("ERR syntax error");
286 Integer expiry = null;
287 if (secondsText != null) {
288 int seconds = number(secondsText);
289 if (seconds <= 0) throw new CommandError("ERR invalid expire time in 'set' command");
290 expiry = time + seconds;
291 }
292 var old = database.get(key);
293 String oldText = null;
294 if (options.contains("GET")) { String value = text(key); if (old != null) oldText = value; }
295 boolean blocked = (options.contains("NX") && old != null) || (options.contains("XX") && old == null);
296 if (!blocked) {
297 if (options.contains("KEEPTTL") && old != null) expiry = old.expiry;
298 replace(key, new TextValue(a.get(1)), expiry);
299 }
300 return options.contains("GET") ? quoted(oldText) : blocked ? "(nil)" : "OK";
301 }
302 case "MSET" -> {
303 for (int i = 0; i < a.size(); i += 2) replace(a.get(i), new TextValue(a.get(i + 1)), null);
304 return "OK";
305 }
306 case "MGET" -> {
307 var values = new ArrayList<String>();
308 for (String name : a) {
309 var entry = database.get(name);
310 values.add(entry != null && entry.value instanceof TextValue value ? value.text : null);
311 }
312 return arrayReply(values);
313 }
314 case "INCR", "DECR", "INCRBY", "DECRBY" -> {
315 int delta = a.size() == 2 ? number(a.get(1)) : 1;
316 if (command.startsWith("DECR")) delta = -delta;
317 String old = text(key);
318 int value = increment(database.containsKey(key) ? number(old) : 0, delta);
319 update(key, new TextValue(Integer.toString(value)));
320 return integerReply(value);
321 }
322 case "APPEND" -> {
323 String value = text(key) + a.get(1);
324 update(key, new TextValue(value));
325 return integerReply(value.length());
326 }
327 case "GETRANGE" -> {
328 int start = number(a.get(1)), stop = number(a.get(2));
329 String value = text(key);
330 var selected = range(value.length(), start, stop);
331 return quoted(value.substring(selected.start, selected.end));
332 }
333 case "STRLEN" -> { return integerReply(text(key).length()); }
334 default -> { String value = text(key); return quoted(database.containsKey(key) ? value : null); }
335 }
336 }

338 String listCommand(String command, List<String> a) {
339 String key = a.getFirst();
340 switch (command) {
341 case "LPUSH", "RPUSH" -> {
342 var values = new ArrayList<>(list(key));
343 for (String value : a.subList(1, a.size())) {
344 if (command.equals("LPUSH")) values.addFirst(value); else values.add(value);
345 }
346 update(key, new ListValue(values));
347 return integerReply(values.size());
348 }
349 case "LPOP", "RPOP" -> {
350 Integer count = null;
351 if (a.size() == 2) {
352 count = parseInteger(a.get(1));
353 if (count == null || count < 0) throw new CommandError(POSITIVE);
354 }
355 var values = new ArrayList<>(list(key));
356 if (!database.containsKey(key)) return "(nil)";
357 int amount = Math.min(count == null ? 1 : count, values.size());
358 var removed = new ArrayList<String>();
359 for (int i = 0; i < amount; i++) removed.add(command.equals("LPOP") ? values.removeFirst() : values.removeLast());
360 update(key, new ListValue(values));
361 return count == null ? quoted(removed.getFirst()) : arrayReply(removed);
362 }
363 case "LRANGE", "LTRIM" -> {
364 int start = number(a.get(1)), stop = number(a.get(2));
365 var values = list(key);
366 var selected = range(values.size(), start, stop);
367 var kept = values.subList(selected.start, selected.end);
368 if (command.equals("LRANGE")) return arrayReply(kept);
369 update(key, new ListValue(kept));
370 return "OK";
371 }
372 case "LLEN" -> { return integerReply(list(key).size()); }
373 case "LINDEX", "LSET" -> {
374 int index = number(a.get(1));
375 var values = list(key);
376 if (index < 0) index += values.size();
377 if (command.equals("LINDEX")) return quoted(index >= 0 && index < values.size() ? values.get(index) : null);
378 if (!database.containsKey(key)) throw new CommandError("ERR no such key");
379 if (index < 0 || index >= values.size()) throw new CommandError("ERR index out of range");
380 var changed = new ArrayList<>(values);
381 changed.set(index, a.get(2));
382 update(key, new ListValue(changed));
383 return "OK";
384 }
385 case "LREM" -> {
386 int count = number(a.get(1));
387 var values = new ArrayList<>(list(key));
388 if (count < 0) Collections.reverse(values);
389 var kept = new ArrayList<String>();
390 int removed = 0;
391 for (String value : values) {
392 if (value.equals(a.get(2)) && (count == 0 || removed < Math.abs(count))) removed++;
393 else kept.add(value);
394 }
395 if (count < 0) Collections.reverse(kept);
396 update(key, new ListValue(kept));
397 return integerReply(removed);
398 }
399 default -> {
400 String where = a.get(1).toUpperCase(Locale.ROOT);
401 if (!where.equals("BEFORE") && !where.equals("AFTER")) throw new CommandError("ERR syntax error");
402 var values = new ArrayList<>(list(key));
403 if (!database.containsKey(key)) return integerReply(0);
404 int index = values.indexOf(a.get(2));
405 if (index < 0) return integerReply(-1);
406 values.add(index + (where.equals("AFTER") ? 1 : 0), a.get(3));
407 update(key, new ListValue(values));
408 return integerReply(values.size());
409 }
410 }
411 }

413 String hashCommand(String command, List<String> a) {
414 String key = a.getFirst();
415 if (command.equals("HINCRBY")) {
416 int delta = number(a.get(2));
417 var fields = new TreeMap<>(hash(key));
418 Integer old = 0;
419 if (fields.containsKey(a.get(1))) old = parseInteger(fields.get(a.get(1)));
420 if (old == null) throw new CommandError("ERR hash value is not an integer");
421 int value = increment(old, delta);
422 fields.put(a.get(1), Integer.toString(value));
423 update(key, new HashValue(fields));
424 return integerReply(value);
425 }
426 var fields = new TreeMap<>(hash(key));
427 switch (command) {
428 case "HSET" -> {
429 int added = 0;
430 for (int i = 1; i < a.size(); i += 2) if (fields.put(a.get(i), a.get(i + 1)) == null) added++;
431 update(key, new HashValue(fields));
432 return integerReply(added);
433 }
434 case "HDEL" -> {
435 int removed = 0;
436 for (String field : a.subList(1, a.size())) if (fields.remove(field) != null) removed++;
437 update(key, new HashValue(fields));
438 return integerReply(removed);
439 }
440 case "HGET" -> { return quoted(fields.get(a.get(1))); }
441 case "HEXISTS" -> { return integerReply(fields.containsKey(a.get(1)) ? 1 : 0); }
442 case "HLEN" -> { return integerReply(fields.size()); }
443 case "HKEYS" -> { return arrayReply(fields.keySet()); }
444 case "HVALS" -> { return arrayReply(fields.values()); }
445 default -> {
446 var result = new ArrayList<String>();
447 fields.forEach((field, value) -> { result.add(field); result.add(value); });
448 return arrayReply(result);
449 }
450 }

453 String setCommand(String command, List<String> a) {
454 String key = a.getFirst();
455 boolean store = command.endsWith("STORE");
456 if (command.startsWith("SINTER") || command.startsWith("SUNION") || command.startsWith("SDIFF")) {
457 var sources = new ArrayList<Set<String>>();
458 for (String source : a.subList(store ? 1 : 0, a.size())) sources.add(set(source));
459 var result = new TreeSet<>(sources.getFirst());
460 for (var source : sources.subList(1, sources.size())) {
461 if (command.startsWith("SINTER")) result.retainAll(source);
462 else if (command.startsWith("SUNION")) result.addAll(source);
463 else result.removeAll(source);
464 }
465 if (!store) return arrayReply(result);
466 replace(key, new SetValue(result), null);
467 return integerReply(result.size());
468 }
469 var members = new TreeSet<>(set(key));
470 return switch (command) {
471 case "SADD", "SREM" -> {
472 int changed = 0;
473 for (String member : a.subList(1, a.size()))
474 if (command.equals("SADD") ? members.add(member) : members.remove(member)) changed++;
475 update(key, new SetValue(members));
476 yield integerReply(changed);
477 }
478 case "SCARD" -> integerReply(members.size());
479 case "SISMEMBER" -> integerReply(members.contains(a.get(1)) ? 1 : 0);
480 default -> arrayReply(members);
481 };
482 }

484 String sortedSetCommand(String command, List<String> a) {
485 String key = a.getFirst();
486 if (command.equals("ZADD")) {
487 int[] values = new int[(a.size() - 1) / 2];
488 for (int i = 0; i < values.length; i++) values[i] = number(a.get(2 * i + 1));
489 var scores = new TreeMap<>(sortedSet(key));
490 int added = 0;
491 for (int i = 0; i < values.length; i++) if (scores.put(a.get(2 * i + 2), values[i]) == null) added++;
492 update(key, new SortedSetValue(scores));
493 return integerReply(added);
494 }
495 if (command.equals("ZINCRBY")) {
496 int delta = number(a.get(1));
497 var scores = new TreeMap<>(sortedSet(key));
498 int value = increment(scores.getOrDefault(a.get(2), 0), delta);
499 scores.put(a.get(2), value);
500 update(key, new SortedSetValue(scores));
501 return quoted(Integer.toString(value));
502 }
503 boolean withScores = false;
504 int start = 0, stop = 0;
505 Bound minimum = null, maximum = null;
506 if (command.equals("ZRANGE") || command.equals("ZRANGEBYSCORE")) {
507 withScores = a.size() == 4;
508 if (withScores && !a.get(3).equalsIgnoreCase("WITHSCORES")) throw new CommandError("ERR syntax error");
509 if (command.equals("ZRANGE")) { start = number(a.get(1)); stop = number(a.get(2)); }
510 else { minimum = bound(a.get(1)); maximum = bound(a.get(2)); }
511 }
512 var scores = new TreeMap<>(sortedSet(key));
513 if (command.equals("ZREM")) {
514 int removed = 0;
515 for (String member : a.subList(1, a.size())) if (scores.remove(member) != null) removed++;
516 update(key, new SortedSetValue(scores));
517 return integerReply(removed);
518 }
519 if (command.equals("ZCARD")) return integerReply(scores.size());
520 if (command.equals("ZSCORE")) {
521 Integer score = scores.get(a.get(1));
522 return quoted(score == null ? null : score.toString());
523 }
524 var ordered = new ArrayList<>(scores.entrySet());
525 ordered.sort(Comparator.<Map.Entry<String, Integer>>comparingInt(Map.Entry::getValue).thenComparing(Map.Entry::getKey));
526 if (command.equals("ZRANK")) {
527 for (int i = 0; i < ordered.size(); i++) if (ordered.get(i).getKey().equals(a.get(1))) return integerReply(i);
528 return "(nil)";
529 }
530 var selected = command.equals("ZRANGE") ? range(ordered.size(), start, stop) : new Range(0, ordered.size());
531 var result = new ArrayList<String>();
532 for (var item : ordered.subList(selected.start, selected.end)) {
533 if (minimum != null && (!minimum.accepts(item.getValue(), true) || !maximum.accepts(item.getValue(), false))) continue;
534 result.add(item.getKey());
535 if (withScores) result.add(item.getValue().toString());
536 }
537 return arrayReply(result);
538 }

540 String keyCommand(String command, List<String> a) {
541 switch (command) {
542 case "TIME" -> { return integerReply(time); }
543 case "DBSIZE" -> { return integerReply(database.size()); }
544 case "FLUSH" -> { database.clear(); return "OK"; }
545 case "INFO" -> {
546 var counts = new TreeMap<String, Integer>();
547 int expires = 0;
548 for (var entry : database.values()) {
549 counts.merge(entry.value.kind(), 1, Integer::sum);
550 if (entry.expiry != null) expires++;
551 }
552 return "keys=" + database.size() + ",expires=" + expires + ",string=" + counts.getOrDefault("string", 0)
553 + ",list=" + counts.getOrDefault("list", 0) + ",hash=" + counts.getOrDefault("hash", 0)
554 + ",set=" + counts.getOrDefault("set", 0) + ",zset=" + counts.getOrDefault("zset", 0)
555 + ",time=" + time + ",depth=" + stack.size() + ",snapshots=" + snapshots.size();
556 }
557 case "DEL", "EXISTS" -> {
558 int count = 0;
559 for (String key : a) if (command.equals("DEL") ? database.remove(key) != null : database.containsKey(key)) count++;
560 return integerReply(count);
561 }
562 case "KEYS" -> {
563 var parts = pattern(a.getFirst());
564 return arrayReply(database.keySet().stream().filter(key -> matches(parts, key)).toList());
565 }
566 case "TICK" -> {
567 int ticks = number(a.getFirst());
568 if (ticks <= 0) throw new CommandError(POSITIVE);
569 time += ticks;
570 return integerReply(purge());
571 }
572 }
573 String key = a.getFirst();
574 if (command.equals("EXPIRE")) {
575 String condition = a.size() == 3 ? a.get(2).toUpperCase(Locale.ROOT) : "";
576 if (a.size() == 3 && !Set.of("NX", "XX", "GT", "LT").contains(condition))
577 throw new CommandError("ERR Unsupported option " + a.get(2));
578 int seconds = number(a.get(1));
579 var entry = database.get(key);
580 if (entry == null) return integerReply(0);
581 int expiry = time + seconds;
582 boolean rejected = switch (condition) {
583 case "NX" -> entry.expiry != null;
584 case "XX" -> entry.expiry == null;
585 case "GT" -> entry.expiry == null || expiry <= entry.expiry;
586 case "LT" -> entry.expiry != null && expiry >= entry.expiry;
587 default -> false;
588 };
589 if (rejected) return integerReply(0);
590 if (seconds <= 0) database.remove(key);
591 else database.put(key, new Entry(entry.value, expiry));
592 return integerReply(1);
593 }
594 var entry = database.get(key);
595 switch (command) {
596 case "TYPE" -> { return entry == null ? "none" : entry.value.kind(); }
597 case "TTL" -> { return integerReply(entry == null ? -2 : entry.expiry == null ? -1 : entry.expiry - time); }
598 case "PERSIST" -> {
599 if (entry == null || entry.expiry == null) return integerReply(0);
600 database.put(key, new Entry(entry.value, null));
601 return integerReply(1);
602 }
603 default -> {
604 if (entry == null) throw new CommandError("ERR no such key");
605 String destination = a.get(1);
606 if (command.equals("RENAMENX") && database.containsKey(destination)) return integerReply(0);
607 if (!key.equals(destination)) { database.remove(key); database.put(destination, entry); }
608 return command.equals("RENAME") ? "OK" : integerReply(1);
609 }
610 }
611 }

613 String controlCommand(String command, List<String> a) {
614 switch (command) {
615 case "BEGIN" -> stack.addLast(new TreeMap<>(database));
616 case "COMMIT", "ROLLBACK" -> {
617 if (stack.isEmpty()) throw new CommandError("ERR no transaction");
618 var saved = stack.removeLast();
619 if (command.equals("ROLLBACK")) { database = saved; purge(); }
620 }
621 case "SNAPSHOT" -> snapshots.put(a.getFirst(), new TreeMap<>(database));
622 case "RESTORE" -> {
623 var saved = snapshots.get(a.getFirst());
624 if (saved == null) throw new CommandError("ERR no such snapshot");
625 database = new TreeMap<>(saved);
626 purge();
627 }
628 }
629 return "OK";
630 }

632 String execute(String command, List<String> args) {
633 return switch (command) {
634 case "SET", "GET", "MSET", "MGET", "INCR", "DECR", "INCRBY", "DECRBY", "APPEND", "STRLEN", "GETRANGE" -> textCommand(command, args);
635 case "LPUSH", "RPUSH", "LPOP", "RPOP", "LRANGE", "LLEN", "LINDEX", "LSET", "LREM", "LTRIM", "LINSERT" -> listCommand(command, args);
636 case "HSET", "HGET", "HDEL", "HGETALL", "HLEN", "HEXISTS", "HKEYS", "HVALS", "HINCRBY" -> hashCommand(command, args);
637 case "SADD", "SREM", "SISMEMBER", "SMEMBERS", "SCARD", "SINTER", "SUNION", "SDIFF", "SINTERSTORE", "SUNIONSTORE", "SDIFFSTORE" -> setCommand(command, args);
638 case "ZADD", "ZSCORE", "ZINCRBY", "ZREM", "ZCARD", "ZRANK", "ZRANGE", "ZRANGEBYSCORE" -> sortedSetCommand(command, args);
639 case "BEGIN", "COMMIT", "ROLLBACK", "SNAPSHOT", "RESTORE" -> controlCommand(command, args);
640 default -> keyCommand(command, args);
641 };
642 }

645 public static void main(String[] args) throws Exception {
646 var input = new BufferedReader(new InputStreamReader(System.in));
647 var output = new BufferedWriter(new OutputStreamWriter(System.out));
648 int count = Integer.parseInt(input.readLine());
649 var store = new Store();
650 for (int i = 0; i < count; i++) {
651 var words = splitLine(input.readLine());
652 String result;
653 if (words == null) result = "Invalid argument(s)";
654 else {
655 String command = words.getFirst().toUpperCase(Locale.ROOT);
656 var format = arity(command);
657 if (format == null) result = "(error) ERR unknown command '" + words.getFirst() + "'";
658 else if (!format.accepts(words.size() - 1))
659 result = "(error) ERR wrong number of arguments for '" + command.toLowerCase(Locale.ROOT) + "' command";
660 else {
661 try { result = store.execute(command, words.subList(1, words.size())); }
662 catch (CommandError error) { result = "(error) " + error.getMessage(); }
663 }
664 }
665 output.write(result);
666 output.write('\n');
667 }
668 output.flush();
669 }
