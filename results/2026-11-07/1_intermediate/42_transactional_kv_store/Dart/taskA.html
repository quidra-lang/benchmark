86int number(String text, [String error = notInteger]) {
87 if (text == '0') return 0;
88 final start = text.startsWith('-') ? 1 : 0;
89 final digits = text.length - start;
90 if (digits < 1 || digits > 9 || text.codeUnitAt(start) < 49 || text.codeUnitAt(start) > 57) {
91 throw CommandError(error);
92 }
93 var value = 0;
94 for (var i = start; i < text.length; i++) {
95 final digit = text.codeUnitAt(i) - 48;
96 if (digit < 0 || digit > 9) throw CommandError(error);
97 value = value * 10 + digit;
98 }
99 return start == 0 ? value : -value;
100}

102int increment(int value, int delta) {
103 final result = value + delta;
104 if (result < -999999999 || result > 999999999) throw const CommandError(overflow);
105 return result;
106}

108(int, int) normalizeRange(int length, int start, int stop) {
109 if (start < 0) start += length;
110 if (stop < 0) stop += length;
111 if (start < 0) start = 0;
112 if (stop >= length) stop = length - 1;
113 return start <= stop ? (start, stop + 1) : (0, 0);
114}

119List<String> arrayReply(Iterable<String?> source) {
120 final values = source.toList();
121 if (values.isEmpty) return ['(empty array)'];

123 return [
124 for (var i = 0; i < values.length; i++)
125 (i + 1).toString().padLeft(width) + ') ' + (values[i] == null ? '(nil)' : quote(values[i]!)),

129List<String>? splitArguments(String line) {
130 final words = <String>[];
131 var position = 0;
132 while (position < line.length) {
133 if (line[position] == ' ') { position++; continue; }
134 if (line[position] != '"') {
135 final start = position;
136 while (position < line.length && line[position] != ' ') { position++; }
137 words.add(line.substring(start, position));
138 continue;
139 }
140 position++;
141 final value = StringBuffer();
142 var closed = false;
143 while (position < line.length) {
144 var character = line[position++];
145 if (character == '"') { closed = true; break; }
146 if (character == r'\') {
147 if (position == line.length || (line[position] != '"' && line[position] != r'\')) return null;
148 character = line[position++];
149 }
150 value.write(character);
151 }
152 if (!closed || (position < line.length && line[position] != ' ')) return null;
153 words.add(value.toString());
154 }
155 return words;
156}

158enum PatternKind { literal, any, star, characterClass }
159final class PatternPart {
160 final PatternKind kind;
161 final int literal;
162 final Set<int> characters;
163 final bool negated;
164 const PatternPart(this.kind, {this.literal = 0, this.characters = const {}, this.negated = false});
165}
166
167List<PatternPart> parsePattern(String text) {
168 final parts = <PatternPart>[];
169 for (var i = 0; i < text.length; i++) {
170 final character = text.codeUnitAt(i);
171 if (character == 92) {
172 if (++i == text.length) throw const CommandError('ERR invalid pattern');
173 parts.add(PatternPart(PatternKind.literal, literal: text.codeUnitAt(i)));
174 } else if (character == 91) {
175 final end = text.indexOf(']', i + 1);
176 if (end < 0) throw const CommandError('ERR invalid pattern');
177 var start = i + 1;
178 final negated = start < end && text[start] == '^';
179 if (negated) start++;
180 if (start == end) throw const CommandError('ERR invalid pattern');
181 final characters = <int>{};
182 for (var j = start; j < end;) {
183 if (j + 2 < end && text[j + 1] == '-') {
184 final a = text.codeUnitAt(j), b = text.codeUnitAt(j + 2);
185 for (var code = a < b ? a : b; code <= (a > b ? a : b); code++) { characters.add(code); }
186 j += 3;
187 } else {
188 characters.add(text.codeUnitAt(j++));
189 }
190 }
191 parts.add(PatternPart(PatternKind.characterClass, characters: characters, negated: negated));
192 i = end;
193 } else {
194 parts.add(switch (character) {
195 42 => const PatternPart(PatternKind.star),
196 63 => const PatternPart(PatternKind.any),
197 _ => PatternPart(PatternKind.literal, literal: character),
198 });
199 }
200 }
201 return parts;
202}
203
204bool matches(List<PatternPart> parts, String value) {
205 var previous = List.filled(value.length + 1, false);
206 previous[0] = true;
207 for (final part in parts) {
208 final current = List.filled(previous.length, false);
209 if (part.kind == PatternKind.star) {
210 current[0] = previous[0];
211 for (var i = 1; i < current.length; i++) { current[i] = previous[i] || current[i - 1]; }
212 } else {
213 for (var i = 0; i < value.length; i++) {
214 final character = value.codeUnitAt(i);
215 final accepts = part.kind == PatternKind.any || (part.kind == PatternKind.literal
216 ? part.literal == character : part.characters.contains(character) != part.negated);
217 current[i + 1] = previous[i] && accepts;
218 }
219 }
220 previous = current;
221 }
222 return previous.last;
223}

225enum BoundKind { negativeInfinity, finite, positiveInfinity }
226final class Bound {
227 final BoundKind kind;
228 final int value;
229 final bool inclusive;
230 const Bound(this.kind, [this.value = 0, this.inclusive = true]);
231 bool accepts(int score, bool lower) => switch (kind) {
232 BoundKind.negativeInfinity => lower,
233 BoundKind.positiveInfinity => !lower,
234 BoundKind.finite => lower ? (inclusive ? score >= value : score > value) : (inclusive ? score <= value : score < value),
235 };
236}
237
238Bound scoreBound(String word) {
239 if (word == '-inf') return const Bound(BoundKind.negativeInfinity);
240 if (word == '+inf') return const Bound(BoundKind.positiveInfinity);
241 final inclusive = !word.startsWith('(');
242 return Bound(BoundKind.finite, number(inclusive ? word : word.substring(1), 'ERR min or max is not a float'), inclusive);
243}

325 List<String> textCommand(String command, List<String> args) {
326 final key = args[0];
327 if (command == 'SET') {
328 final options = <String>{};
329 String? secondsWord;
330 for (var i = 2; i < args.length; i++) {
331 final option = args[i].toUpperCase();
332 if (!['NX', 'XX', 'EX', 'KEEPTTL', 'GET'].contains(option) || !options.add(option)) {
333 throw const CommandError(syntaxError);
334 }
335 if (option == 'EX') {
336 if (++i == args.length) throw const CommandError(syntaxError);
337 secondsWord = args[i];
338 }
339 }
340 if (options.contains('NX') && options.contains('XX') || options.contains('EX') && options.contains('KEEPTTL')) {
341 throw const CommandError(syntaxError);
342 }
343 int? expiry;
344 if (secondsWord != null) {
345 final seconds = number(secondsWord);
346 if (seconds <= 0) throw const CommandError("ERR invalid expire time in 'set' command");
347 expiry = time + seconds;
348 }
349 final entry = database[key], exists = entry != null;
350 String? old;
351 if (options.contains('GET')) {
352 final value = read(key, const TextValue(''));
353 if (exists) old = value.text;
354 }
355 if (options.contains('KEEPTTL') && exists) expiry = entry!.expiry;
356 final changed = !(options.contains('NX') && exists) && !(options.contains('XX') && !exists);
357 if (changed) replace(key, TextValue(args[1]), expiry);
358 return options.contains('GET') ? textReply(old) : changed ? ['OK'] : textReply(null);
359 }
360 if (command == 'MSET') {
361 for (var i = 0; i < args.length; i += 2) { replace(args[i], TextValue(args[i + 1])); }
362 return ['OK'];
363 }
364 if (command == 'MGET') {
365 return arrayReply(args.map((key) => switch (database[key]?.value) {
366 TextValue(:final text) => text,
367 _ => null,
368 }));
369 }
370 if (['INCR', 'DECR', 'INCRBY', 'DECRBY'].contains(command)) {
371 var delta = args.length == 2 ? number(args[1]) : 1;
372 if (command == 'DECR' || command == 'DECRBY') delta = -delta;
373 final text = read(key, const TextValue('')).text;
374 final result = increment(database.containsKey(key) ? number(text) : 0, delta);
375 update(key, TextValue('$result'));
376 return integerReply(result);
377 }
378 if (command == 'GETRANGE') {
379 final start = number(args[1]), stop = number(args[2]);
380 final text = read(key, const TextValue('')).text;
381 final (first, end) = normalizeRange(text.length, start, stop);
382 return textReply(text.substring(first, end));
383 }
384 var text = read(key, const TextValue('')).text;
385 if (command == 'GET') return textReply(database.containsKey(key) ? text : null);
386 if (command == 'STRLEN') return integerReply(text.length);
387 text += args[1];
388 update(key, TextValue(text));
389 return integerReply(text.length);
390 }

392 List<String> listCommand(String command, List<String> args) {
393 final key = args[0];
394 if (command == 'LPOP' || command == 'RPOP') {
395 final counted = args.length == 2;
396 var count = counted ? number(args[1], positive) : 1;
397 if (count < 0) throw const CommandError(positive);
398 final value = read(key, ListValue([]));
399 if (!database.containsKey(key)) return textReply(null);
400 if (count > value.items.length) count = value.items.length;
401 final removed = <String>[];
402 for (var i = 0; i < count; i++) {
403 removed.add(value.items.removeAt(command == 'LPOP' ? 0 : value.items.length - 1));
404 }
405 update(key, value);
406 return counted ? arrayReply(removed) : textReply(removed.first);
407 }
408 if (command == 'LRANGE' || command == 'LTRIM') {
409 final start = number(args[1]), stop = number(args[2]);
410 final value = read(key, ListValue([]));
411 final (first, end) = normalizeRange(value.items.length, start, stop);
412 final kept = value.items.sublist(first, end);
413 if (command == 'LRANGE') return arrayReply(kept);
414 update(key, ListValue(kept));
415 return ['OK'];
416 }
417 if (command == 'LINDEX' || command == 'LSET') {
418 var index = number(args[1]);
419 final value = read(key, ListValue([]));
420 if (index < 0) index += value.items.length;
421 final inside = index >= 0 && index < value.items.length;
422 if (command == 'LINDEX') return textReply(inside ? value.items[index] : null);
423 if (!database.containsKey(key)) throw const CommandError(noKey);
424 if (!inside) throw const CommandError('ERR index out of range');
425 value.items[index] = args[2];
426 update(key, value);
427 return ['OK'];
428 }
429 if (command == 'LREM') {
430 final count = number(args[1]);
431 final value = read(key, ListValue([]));
432 final source = count < 0 ? value.items.reversed : value.items;
433 final limit = count == 0 ? value.items.length : count.abs();
434 var removed = 0;
435 final kept = <String>[];
436 for (final item in source) {
437 if (item == args[2] && removed < limit) { removed++; } else { kept.add(item); }
438 }
439 update(key, ListValue(count < 0 ? kept.reversed.toList() : kept));
440 return integerReply(removed);
441 }
442 if (command == 'LINSERT') {
443 final where = args[1].toUpperCase();
444 if (where != 'BEFORE' && where != 'AFTER') throw const CommandError(syntaxError);
445 final value = read(key, ListValue([]));
446 if (!database.containsKey(key)) return integerReply(0);
447 final pivot = value.items.indexOf(args[2]);
448 if (pivot < 0) return integerReply(-1);
449 value.items.insert(pivot + (where == 'AFTER' ? 1 : 0), args[3]);
450 update(key, value);
451 return integerReply(value.items.length);
452 }
453 final value = read(key, ListValue([]));
454 if (command == 'LLEN') return integerReply(value.items.length);
455 for (final element in args.skip(1)) {
456 if (command == 'LPUSH') { value.items.insert(0, element); } else { value.items.add(element); }
457 }
458 update(key, value);
459 return integerReply(value.items.length);
460 }

462 List<String> hashCommand(String command, List<String> args) {
463 final key = args[0];
464 if (command == 'HINCRBY') {
465 final delta = number(args[2]);
466 final value = read(key, HashValue({}));
467 final text = value.fields[args[1]];
468 final old = text == null ? 0 : number(text, 'ERR hash value is not an integer');
469 final result = increment(old, delta);
470 value.fields[args[1]] = '$result';
471 update(key, value);
472 return integerReply(result);
473 }
474 final value = read(key, HashValue({}));
475 if (command == 'HSET') {
476 var added = 0;
477 for (var i = 1; i < args.length; i += 2) {
478 if (!value.fields.containsKey(args[i])) added++;
479 value.fields[args[i]] = args[i + 1];
480 }
481 update(key, value);
482 return integerReply(added);
483 }
484 if (command == 'HGET') return textReply(value.fields[args[1]]);
485 if (command == 'HDEL') {
486 var removed = 0;
487 for (final field in args.skip(1)) { if (value.fields.remove(field) != null) removed++; }
488 update(key, value);
489 return integerReply(removed);
490 }
491 if (command == 'HLEN') return integerReply(value.fields.length);
492 if (command == 'HEXISTS') return integerReply(value.fields.containsKey(args[1]) ? 1 : 0);
493 final output = <String>[];
494 for (final field in ordered(value.fields.keys)) {
495 if (command != 'HVALS') output.add(field);
496 if (command != 'HKEYS') output.add(value.fields[field]!);
497 }
498 return arrayReply(output);

501 List<String> setCommand(String command, List<String> args) {
502 final storing = command.endsWith('STORE');
503 final operation = storing ? command.substring(0, command.length - 5) : command;
504 if (['SINTER', 'SUNION', 'SDIFF'].contains(operation)) {
505 final sources = [
506 for (final key in args.skip(storing ? 1 : 0)) read(key, SetValue({})).members,
507 ];
508 final result = sources.first;
509 for (final source in sources.skip(1)) {
510 if (operation == 'SUNION') {
511 result.addAll(source);
512 } else {
513 result.removeWhere((member) => operation == 'SINTER' ? !source.contains(member) : source.contains(member));
514 }
515 }
516 if (storing) {
517 replace(args[0], SetValue(result));
518 return integerReply(result.length);
519 }
520 return arrayReply(ordered(result));
521 }
522 final value = read(args[0], SetValue({}));
523 if (command == 'SCARD') return integerReply(value.members.length);
524 if (command == 'SMEMBERS') return arrayReply(ordered(value.members));
525 if (command == 'SISMEMBER') return integerReply(value.members.contains(args[1]) ? 1 : 0);
526 var changed = 0;
527 for (final member in args.skip(1)) {
528 if (command == 'SADD' ? value.members.add(member) : value.members.remove(member)) changed++;
529 }
530 update(args[0], value);
531 return integerReply(changed);
532 }

534 List<String> sortedSetCommand(String command, List<String> args) {
535 final key = args[0];
536 if (command == 'ZADD') {
537 final scores = [for (var i = 1; i < args.length; i += 2) number(args[i])];
538 final value = read(key, SortedSetValue({}));
539 var added = 0;
540 for (var i = 0; i < scores.length; i++) {
541 final member = args[2 * i + 2];
542 if (!value.scores.containsKey(member)) added++;
543 value.scores[member] = scores[i];
544 }
545 update(key, value);
546 return integerReply(added);
547 }
548 if (command == 'ZINCRBY') {
549 final delta = number(args[1]);
550 final value = read(key, SortedSetValue({}));
551 final result = increment(value.scores[args[2]] ?? 0, delta);
552 value.scores[args[2]] = result;
553 update(key, value);
554 return textReply('$result');
555 }
556 final byIndex = command == 'ZRANGE', byScore = command == 'ZRANGEBYSCORE', withScores = args.length == 4;
557 var start = 0, stop = 0;
558 var minimum = const Bound(BoundKind.negativeInfinity), maximum = const Bound(BoundKind.positiveInfinity);
559 if (byIndex || byScore) {
560 if (withScores && args[3].toUpperCase() != 'WITHSCORES') throw const CommandError(syntaxError);
561 if (byIndex) { start = number(args[1]); stop = number(args[2]); }
562 else { minimum = scoreBound(args[1]); maximum = scoreBound(args[2]); }
563 }
564 final value = read(key, SortedSetValue({}));
565 if (command == 'ZSCORE') return textReply(value.scores[args[1]]?.toString());
566 if (command == 'ZREM') {
567 var removed = 0;
568 for (final member in args.skip(1)) { if (value.scores.remove(member) != null) removed++; }
569 update(key, value);
570 return integerReply(removed);
571 }
572 if (command == 'ZCARD') return integerReply(value.scores.length);
573 var sorted = value.scores.entries.toList()..sort((left, right) {
574 final comparison = left.value.compareTo(right.value);
575 return comparison != 0 ? comparison : left.key.compareTo(right.key);
576 });
577 if (command == 'ZRANK') {
578 final rank = sorted.indexWhere((entry) => entry.key == args[1]);
579 return rank < 0 ? textReply(null) : integerReply(rank);
580 }
581 if (byIndex) {
582 final (first, end) = normalizeRange(sorted.length, start, stop);
583 sorted = sorted.sublist(first, end);
584 }
585 final output = <String>[];
586 for (final entry in sorted) {
587 if (byScore && (!minimum.accepts(entry.value, true) || !maximum.accepts(entry.value, false))) continue;
588 output.add(entry.key);
589 if (withScores) output.add(entry.value.toString());
590 }
591 return arrayReply(output);
592 }

594 List<String> keyCommand(String command, List<String> args) {
595 if (command == 'DEL' || command == 'EXISTS') {
596 var count = 0;
597 for (final key in args) {
598 if (command == 'DEL' ? database.remove(key) != null : database.containsKey(key)) count++;
599 }
600 return integerReply(count);
601 }
602 if (command == 'KEYS') {
603 final pattern = parsePattern(args[0]);
604 return arrayReply(ordered(database.keys.where((key) => matches(pattern, key))));
605 }
606 if (command == 'EXPIRE') {
607 final condition = args.length == 3 ? args[2].toUpperCase() : '';
608 if (args.length == 3 && !['NX', 'XX', 'GT', 'LT'].contains(condition)) {
609 throw CommandError('ERR Unsupported option ' + args[2]);
610 }
611 final seconds = number(args[1]);
612 final entry = database[args[0]];
613 if (entry == null) return integerReply(0);
614 final old = entry.expiry, expiry = time + seconds;
615 final rejected = switch (condition) {
616 'NX' => old != null,
617 'XX' => old == null,
618 'GT' => old == null || expiry <= old,
619 'LT' => old != null && expiry >= old,
620 _ => false,
621 };
622 if (rejected) return integerReply(0);
623 if (seconds <= 0) { database.remove(args[0]); }
624 else { database[args[0]] = Entry(entry.value, expiry); }
625 return integerReply(1);
626 }
627 if (command == 'TICK') {
628 final ticks = number(args[0]);
629 if (ticks <= 0) throw const CommandError(positive);
630 time += ticks;
631 return integerReply(purge());
632 }
633 if (command == 'TIME') return integerReply(time);
634 if (command == 'DBSIZE') return integerReply(database.length);
635 if (command == 'FLUSH') { database.clear(); return ['OK']; }
636 if (command == 'INFO') {
637 final keys = database.length, expires = database.values.where((entry) => entry.expiry != null).length;
638 final parts = ['keys=$keys', 'expires=$expires'];
639 for (final kind in ValueKind.values) {
640 final name = kind.name, count = database.values.where((entry) => entry.value.kind == kind).length;
641 parts.add('$name=$count');
642 }
643 final depth = transactions.length, snapshotCount = snapshots.length;
644 parts.addAll(['time=$time', 'depth=$depth', 'snapshots=$snapshotCount']);
645 return [parts.join(',')];
646 }
647 final key = args[0], entry = database[key];
648 if (command == 'TYPE') return [entry?.value.kind.name ?? 'none'];
649 if (command == 'TTL') return integerReply(entry == null ? -2 : entry.expiry == null ? -1 : entry.expiry! - time);
650 if (command == 'PERSIST') {
651 if (entry == null || entry.expiry == null) return integerReply(0);
652 database[key] = Entry(entry.value);
653 return integerReply(1);
654 }
655 if (entry == null) throw const CommandError(noKey);
656 if (command == 'RENAMENX' && database.containsKey(args[1])) return integerReply(0);
657 if (key != args[1]) { database[args[1]] = entry; database.remove(key); }
658 return command == 'RENAME' ? ['OK'] : integerReply(1);
659 }
660
661 List<String> controlCommand(String command, List<String> args) {
662 if (command == 'BEGIN') { transactions.add(copy(database)); return ['OK']; }
663 if (command == 'COMMIT' || command == 'ROLLBACK') {
664 if (transactions.isEmpty) throw const CommandError('ERR no transaction');
665 final saved = transactions.removeLast();
666 if (command == 'ROLLBACK') { database = saved; purge(); }
667 return ['OK'];
668 }
669 if (command == 'SNAPSHOT') { snapshots[args[0]] = copy(database); return ['OK']; }
670 final saved = snapshots[args[0]];
671 if (saved == null) throw const CommandError('ERR no such snapshot');
672 database = copy(saved);
673 purge();
674 return ['OK'];
675 }
676
677 List<String> execute(String line) {
678 final words = splitArguments(line);
679 if (words == null) return ['Invalid argument(s)'];
680 final original = words.first, command = original.toUpperCase();
681 final spec = commands[command];
682 if (spec == null) throw CommandError("ERR unknown command '$original'");
683 final args = words.sublist(1);
684 if (args.length < spec.minimum || spec.maximum >= 0 && args.length > spec.maximum ||
685 spec.parity >= 0 && args.length % 2 != spec.parity) {
686 final name = command.toLowerCase();
687 throw CommandError("ERR wrong number of arguments for '$name' command");
688 }
689 return switch (spec.group) {
690 Group.text => textCommand(command, args),
691 Group.list => listCommand(command, args),
692 Group.hash => hashCommand(command, args),
693 Group.set => setCommand(command, args),
694 Group.sortedSet => sortedSetCommand(command, args),
695 Group.key => keyCommand(command, args),
696 Group.control => controlCommand(command, args),
697 };
698 }
699}

708 } on CommandError catch (error) {
709 stdout.writeln('(error) ' + error.message);
710 }
