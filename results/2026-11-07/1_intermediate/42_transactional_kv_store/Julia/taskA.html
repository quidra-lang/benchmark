54function array_reply(values)
55 isempty(values) && return ["(empty array)"]
56 width = ndigits(length(values))
57 [lpad(string(i), width) * ") " * quoted(value) for (i, value) in enumerate(values)]

60function split_line(line::String)
61 words = String[]
62 i, n = 1, length(line)
63 while i <= n
64 if line[i] == ' '
65 i += 1
66 continue
67 end
68 if line[i] != '"'
69 start = i
70 while i <= n && line[i] != ' '
71 i += 1
72 end
73 push!(words, line[start:i-1])
74 continue
75 end
76 i += 1
77 text = IOBuffer()
78 closed = false
79 while i <= n
80 character = line[i]
81 i += 1
82 if character == '"'
83 closed = true
84 break
85 end
86 if character == '\\'
87 i > n && return nothing
88 character = line[i]
89 i += 1
90 character in ('\\', '"') || return nothing
91 end
92 print(text, character)
93 end
94 (!closed || (i <= n && line[i] != ' ')) && return nothing
95 push!(words, String(take!(text)))
96 end
97 words
98end

100valid_integer(text::String) = occursin(r"^(0|-?[1-9][0-9]{0,8})$", text)
101function number(text::String, message::String=NOT_INTEGER)
102 valid_integer(text) || fail(message)
103 parse(Int, text)
104end
105function incremented(previous::Int, delta::Int)
106 value = previous + delta
107 abs(value) <= 999999999 || fail("ERR increment or decrement would overflow")
108 value
109end
110function normalized_range(n::Int, first::Int, last::Int)
111 start = max(0, first < 0 ? first + n : first)
112 stop = min(n - 1, last < 0 ? last + n : last)
113 start <= stop ? (start+1:stop+1) : (1:0)
114end
115struct ScoreBound
116 infinity::Int
117 value::Int
118 exclusive::Bool
119end
120function score_bound(text::String)
121 text == "-inf" && return ScoreBound(-1, 0, false)
122 text == "+inf" && return ScoreBound(1, 0, false)
123 exclusive = startswith(text, "(")
124 value = number(exclusive ? text[2:end] : text, "ERR min or max is not a float")
125 ScoreBound(0, value, exclusive)
126end
127function within_bound(score::Int, bound::ScoreBound, lower::Bool)
128 bound.infinity != 0 && return lower == (bound.infinity < 0)
129 (lower ? score > bound.value : score < bound.value) || (!bound.exclusive && score == bound.value)
130end

132abstract type PatternPart end
133struct Literal <: PatternPart
134 value::Char
135end
136struct AnyCharacter <: PatternPart end
137struct AnySequence <: PatternPart end
138struct CharacterClass <: PatternPart
139 members::BitSet
140 negated::Bool
141end
142accepts(part::Literal, character::Char) = part.value == character
143accepts(::AnyCharacter, ::Char) = true
144accepts(part::CharacterClass, character::Char) = (Int(character) in part.members) != part.negated
145function parse_pattern(text::String)
146 parts = PatternPart[]
147 i, n = 1, length(text)
148 while i <= n
149 character = text[i]
150 i += 1
151 if character == '\\'
152 i <= n || fail("ERR invalid pattern")
153 push!(parts, Literal(text[i]))
154 i += 1
155 elseif character == '['
156 finish = findnext(']', text, i)
157 isnothing(finish) && fail("ERR invalid pattern")
158 negated = i < finish && text[i] == '^'
159 i += Int(negated)
160 i < finish || fail("ERR invalid pattern")
161 members = BitSet()
162 while i < finish
163 first = Int(text[i])
164 if i + 2 < finish && text[i+1] == '-'
165 last = Int(text[i+2])
166 union!(members, min(first, last):max(first, last))
167 i += 3
168 else
169 push!(members, first)
170 i += 1
171 end
172 end
173 push!(parts, CharacterClass(members, negated))
174 i = finish + 1
175 elseif character == '*'
176 push!(parts, AnySequence())
177 elseif character == '?'
178 push!(parts, AnyCharacter())
179 else
180 push!(parts, Literal(character))
181 end
182 end
183 parts
184end
185function matches_pattern(parts::Vector{PatternPart}, text::String)
186 previous = falses(length(text) + 1)
187 previous[1] = true
188 for part in parts
189 following = falses(length(previous))
190 if part isa AnySequence
191 following[1] = previous[1]
192 for i in eachindex(text)
193 following[i+1] = previous[i+1] || following[i]
194 end
195 else
196 for (i, character) in enumerate(text)
197 following[i+1] = previous[i] && accepts(part, character)
198 end
199 end
200 previous = following
201 end
202 last(previous)
203end

205function read_value(store::Store, key::String, ::Type{T}, empty::T) where {T<:Value}
206 entry = get(store.database, key, nothing)
207 isnothing(entry) && return empty
208 entry.value isa T || fail(WRONG_TYPE)
209 entry.value::T
210end
211text_value(store::Store, key::String) = read_value(store, key, TextValue, TextValue("")).text
212list_values(store::Store, key::String) = copy(read_value(store, key, ListValue, ListValue(String[])).items)
213hash_fields(store::Store, key::String) = copy(read_value(store, key, HashValue, HashValue(Dict{String,String}())).fields)
214set_members(store::Store, key::String) = copy(read_value(store, key, SetValue, SetValue(Set{String}())).members)
215zset_scores(store::Store, key::String) = copy(read_value(store, key, SortedSetValue, SortedSetValue(Dict{String,Int}())).scores)
216function replace_value!(store::Store, key::String, value::Value, expiry::Union{Nothing,Int}=nothing)
217 if empty_value(value)
218 delete!(store.database, key)
219 else
220 store.database[key] = Entry(value, expiry)
221 end
222end
223function update_value!(store::Store, key::String, value::Value)
224 old = get(store.database, key, nothing)
225 replace_value!(store, key, value, isnothing(old) ? nothing : old.expiry)
226end
227function purge!(store::Store)
228 before = length(store.database)
229 filter!(pair -> isnothing(pair.second.expiry) || pair.second.expiry > store.clock, store.database)
230 before - length(store.database)
231end

233struct CommandSpec
234 minimum::Int
235 maximum::Union{Nothing,Int}
236 parity::Union{Nothing,Int}
237end
238const COMMANDS = let commands = Dict{String,CommandSpec}()
239 function register!(names, minimum, maximum=minimum, parity=nothing)
240 for name in split(names)
241 commands[name] = CommandSpec(minimum, maximum, parity)
242 end
243 end
244 register!("TIME DBSIZE FLUSH INFO BEGIN COMMIT ROLLBACK", 0)
245 register!("GET INCR DECR STRLEN LLEN HGETALL HLEN HKEYS HVALS SMEMBERS SCARD ZCARD TYPE KEYS TTL PERSIST TICK SNAPSHOT RESTORE", 1)
246 register!("INCRBY DECRBY APPEND LINDEX HGET HEXISTS SISMEMBER ZSCORE ZRANK RENAME RENAMENX", 2)
247 register!("GETRANGE LRANGE LSET LREM LTRIM HINCRBY ZINCRBY", 3)
248 register!("LINSERT", 4)
249 register!("LPOP RPOP", 1, 2)
250 register!("EXPIRE", 2, 3)
251 register!("ZRANGE ZRANGEBYSCORE", 3, 4)
252 register!("MGET SINTER SUNION SDIFF DEL EXISTS", 1, nothing)
253 register!("SET LPUSH RPUSH HDEL SADD SREM ZREM SINTERSTORE SUNIONSTORE SDIFFSTORE", 2, nothing)
254 register!("MSET", 2, nothing, 0)
255 register!("HSET ZADD", 3, nothing, 1)
256 commands
257end
258function command_name(words::Vector{String})
259 name = uppercase(words[1])
260 spec = get(COMMANDS, name, nothing)
261 isnothing(spec) && fail("ERR unknown command '$(words[1])'")
262 n = length(words) - 1
263 if n < spec.minimum || (!isnothing(spec.maximum) && n > spec.maximum) ||
264 (!isnothing(spec.parity) && mod(n, 2) != spec.parity)
265 fail("ERR wrong number of arguments for '$(lowercase(name))' command")
266 end
267 name
268end

270function set_string!(store::Store, args::Vector{String})
271 key = args[1]
272 options = Set{String}()
273 seconds_text = nothing
274 i = 3
275 while i <= length(args)
276 option = uppercase(args[i])
277 i += 1
278 option in ("NX", "XX", "EX", "KEEPTTL", "GET") && !(option in options) || fail("ERR syntax error")
279 push!(options, option)
280 if option == "EX"
281 i <= length(args) || fail("ERR syntax error")
282 seconds_text = args[i]
283 i += 1
284 end
285 end
286 (("NX" in options && "XX" in options) || ("EX" in options && "KEEPTTL" in options)) && fail("ERR syntax error")
287 expiry = nothing
288 if "EX" in options
289 seconds = number(seconds_text)
290 seconds > 0 || fail("ERR invalid expire time in 'set' command")
291 expiry = store.clock + seconds
292 end
293 old = get(store.database, key, nothing)
294 old_text = nothing
295 if "GET" in options
296 text = text_value(store, key)
297 !isnothing(old) && (old_text = text)
298 end
299 blocked = ("NX" in options && !isnothing(old)) || ("XX" in options && isnothing(old))
300 if !blocked
301 "KEEPTTL" in options && !isnothing(old) && (expiry = old.expiry)
302 replace_value!(store, key, TextValue(args[2]), expiry)
303 end
304 "GET" in options ? quoted(old_text) : blocked ? "(nil)" : "OK"
305end
306function string_command!(store::Store, command::String, args::Vector{String})
307 key = args[1]
308 command == "SET" && return set_string!(store, args)
309 if command == "MSET"
310 for i in 1:2:length(args)
311 replace_value!(store, args[i], TextValue(args[i+1]))
312 end
313 return "OK"
314 elseif command == "MGET"
315 values = map(args) do name
316 entry = get(store.database, name, nothing)
317 !isnothing(entry) && entry.value isa TextValue ? entry.value.text : nothing
318 end
319 return array_reply(values)
320 elseif command in ("INCR", "DECR", "INCRBY", "DECRBY")
321 delta = length(args) == 2 ? number(args[2]) : 1
322 startswith(command, "DECR") && (delta = -delta)
323 text = text_value(store, key)
324 previous = haskey(store.database, key) ? number(text) : 0
325 value = incremented(previous, delta)
326 update_value!(store, key, TextValue(string(value)))
327 return integer_reply(value)
328 elseif command == "APPEND"
329 value = text_value(store, key) * args[2]
330 update_value!(store, key, TextValue(value))
331 return integer_reply(length(value))
332 elseif command == "GETRANGE"
333 first = number(args[2])
334 last = number(args[3])
335 value = text_value(store, key)
336 return quoted(value[normalized_range(length(value), first, last)])
337 end
338 value = text_value(store, key)
339 command == "STRLEN" ? integer_reply(length(value)) : quoted(haskey(store.database, key) ? value : nothing)
340end

342function list_command!(store::Store, command::String, args::Vector{String})
343 key = args[1]
344 if command in ("LPUSH", "RPUSH")
345 values = list_values(store, key)
346 for value in args[2:end]
347 command == "LPUSH" ? pushfirst!(values, value) : push!(values, value)
348 end
349 update_value!(store, key, ListValue(values))
350 return integer_reply(length(values))
351 elseif command in ("LPOP", "RPOP")
352 count = length(args) == 2 ? number(args[2], POSITIVE) : nothing
353 !isnothing(count) && count < 0 && fail(POSITIVE)
354 values = list_values(store, key)
355 haskey(store.database, key) || return "(nil)"
356 removed = String[]
357 for _ in 1:min(something(count, 1), length(values))
358 push!(removed, command == "LPOP" ? popfirst!(values) : pop!(values))
359 end
360 update_value!(store, key, ListValue(values))
361 return isnothing(count) ? quoted(removed[1]) : array_reply(removed)
362 elseif command in ("LRANGE", "LTRIM")
363 first = number(args[2])
364 last = number(args[3])
365 values = list_values(store, key)
366 kept = values[normalized_range(length(values), first, last)]
367 command == "LRANGE" && return array_reply(kept)
368 update_value!(store, key, ListValue(kept))
369 return "OK"
370 elseif command == "LLEN"
371 return integer_reply(length(list_values(store, key)))
372 elseif command in ("LINDEX", "LSET")
373 index = number(args[2])
374 values = list_values(store, key)
375 command == "LSET" && !haskey(store.database, key) && fail("ERR no such key")
376 index < 0 && (index += length(values))
377 if !(0 <= index < length(values))
378 command == "LINDEX" && return "(nil)"
379 fail("ERR index out of range")
380 end
381 command == "LINDEX" && return quoted(values[index+1])
382 values[index+1] = args[3]
383 update_value!(store, key, ListValue(values))
384 return "OK"
385 elseif command == "LREM"
386 count = number(args[2])
387 values = list_values(store, key)
388 positions = findall(==(args[3]), values)
389 if count > 0
390 positions = positions[1:min(count, length(positions))]
391 elseif count < 0
392 positions = positions[end-min(-count, length(positions))+1:end]
393 end
394 deleteat!(values, positions)
395 update_value!(store, key, ListValue(values))
396 return integer_reply(length(positions))
397 end
398 position = uppercase(args[2])
399 position in ("BEFORE", "AFTER") || fail("ERR syntax error")
400 values = list_values(store, key)
401 isempty(values) && return integer_reply(0)
402 index = findfirst(==(args[3]), values)
403 isnothing(index) && return integer_reply(-1)
404 insert!(values, index + Int(position == "AFTER"), args[4])
405 update_value!(store, key, ListValue(values))
406 integer_reply(length(values))
407end

409function hash_command!(store::Store, command::String, args::Vector{String})
410 key = args[1]
411 if command == "HINCRBY"
412 delta = number(args[3])
413 fields = hash_fields(store, key)
414 previous = haskey(fields, args[2]) ? number(fields[args[2]], "ERR hash value is not an integer") : 0
415 value = incremented(previous, delta)
416 fields[args[2]] = string(value)
417 update_value!(store, key, HashValue(fields))
418 return integer_reply(value)
419 end
420 fields = hash_fields(store, key)
421 if command == "HSET"
422 before = length(fields)
423 for i in 2:2:length(args)
424 fields[args[i]] = args[i+1]
425 end
426 update_value!(store, key, HashValue(fields))
427 return integer_reply(length(fields) - before)
428 elseif command == "HDEL"
429 before = length(fields)
430 for field in args[2:end]
431 delete!(fields, field)
432 end
433 update_value!(store, key, HashValue(fields))
434 return integer_reply(before - length(fields))
435 elseif command == "HGET"
436 return quoted(get(fields, args[2], nothing))
437 elseif command == "HEXISTS"
438 return integer_reply(Int(haskey(fields, args[2])))
439 elseif command == "HLEN"
440 return integer_reply(length(fields))
441 end
442 ordered = sort!(collect(keys(fields)))
443 command == "HKEYS" && return array_reply(ordered)
444 command == "HVALS" && return array_reply([fields[field] for field in ordered])
445 array_reply([item for field in ordered for item in (field, fields[field])])
446end

448function set_command!(store::Store, command::String, args::Vector{String})
449 key = args[1]
450 if startswith(command, "SINTER") || startswith(command, "SUNION") || startswith(command, "SDIFF")
451 saving = endswith(command, "STORE")
452 sources = [set_members(store, name) for name in (saving ? args[2:end] : args)]
453 operation = startswith(command, "SINTER") ? intersect : startswith(command, "SUNION") ? union : setdiff
454 result = foldl(operation, sources)
455 !saving && return array_reply(sort!(collect(result)))
456 replace_value!(store, key, SetValue(result))
457 return integer_reply(length(result))
458 end
459 values = set_members(store, key)
460 if command in ("SADD", "SREM")
461 before = length(values)
462 command == "SADD" ? union!(values, args[2:end]) : setdiff!(values, args[2:end])
463 update_value!(store, key, SetValue(values))
464 return integer_reply(abs(length(values) - before))
465 elseif command == "SCARD"
466 return integer_reply(length(values))
467 elseif command == "SISMEMBER"
468 return integer_reply(Int(args[2] in values))
469 end
470 array_reply(sort!(collect(values)))
471end

473function zset_command!(store::Store, command::String, args::Vector{String})
474 key = args[1]
475 if command == "ZADD"
476 numbers = [number(args[i]) for i in 2:2:length(args)]
477 scores = zset_scores(store, key)
478 before = length(scores)
479 for (i, score) in enumerate(numbers)
480 scores[args[2i+1]] = score
481 end
482 update_value!(store, key, SortedSetValue(scores))
483 return integer_reply(length(scores) - before)
484 elseif command == "ZINCRBY"
485 delta = number(args[2])
486 scores = zset_scores(store, key)
487 value = incremented(get(scores, args[3], 0), delta)
488 scores[args[3]] = value
489 update_value!(store, key, SortedSetValue(scores))
490 return quoted(string(value))
491 end
492 with_scores = false
493 first, last = 0, 0
494 minimum, maximum = nothing, nothing
495 if command in ("ZRANGE", "ZRANGEBYSCORE")
496 with_scores = length(args) == 4
497 with_scores && uppercase(args[4]) != "WITHSCORES" && fail("ERR syntax error")
498 if command == "ZRANGE"
499 first = number(args[2])
500 last = number(args[3])
501 else
502 minimum = score_bound(args[2])
503 maximum = score_bound(args[3])
504 end
505 end
506 scores = zset_scores(store, key)
507 if command == "ZREM"
508 before = length(scores)
509 for member in args[2:end]
510 delete!(scores, member)
511 end
512 update_value!(store, key, SortedSetValue(scores))
513 return integer_reply(before - length(scores))
514 elseif command == "ZCARD"
515 return integer_reply(length(scores))
516 elseif command == "ZSCORE"
517 return quoted(haskey(scores, args[2]) ? string(scores[args[2]]) : nothing)
518 end
519 ordered = sort!(collect(scores); by=pair -> (pair.second, pair.first))
520 if command == "ZRANK"
521 index = findfirst(pair -> pair.first == args[2], ordered)
522 return isnothing(index) ? "(nil)" : integer_reply(index - 1)
523 end
524 selected = command == "ZRANGE" ? ordered[normalized_range(length(ordered), first, last)] :
525 filter(pair -> within_bound(pair.second, minimum, true) && within_bound(pair.second, maximum, false), ordered)
526 items = with_scores ? [item for (member, score) in selected for item in (member, string(score))] :
527 [pair.first for pair in selected]
528 array_reply(items)
529end

531function key_command!(store::Store, command::String, args::Vector{String})
532 command == "TIME" && return integer_reply(store.clock)
533 command == "DBSIZE" && return integer_reply(length(store.database))
534 if command == "FLUSH"
535 empty!(store.database)
536 return "OK"
537 elseif command == "INFO"
538 counts = [count(entry -> kind(entry.value) == name, values(store.database)) for name in ("string", "list", "hash", "set", "zset")]
539 expires = count(entry -> !isnothing(entry.expiry), values(store.database))
540 return "keys=$(length(store.database)),expires=$expires,string=$(counts[1]),list=$(counts[2]),hash=$(counts[3])," *
541 "set=$(counts[4]),zset=$(counts[5]),time=$(store.clock),depth=$(length(store.stack)),snapshots=$(length(store.snapshots))"
542 elseif command == "DEL"
543 removed = 0
544 for key in args
545 removed += Int(!isnothing(pop!(store.database, key, nothing)))
546 end
547 return integer_reply(removed)
548 elseif command == "EXISTS"
549 return integer_reply(count(key -> haskey(store.database, key), args))
550 elseif command == "KEYS"
551 parts = parse_pattern(args[1])
552 return array_reply(sort!([key for key in keys(store.database) if matches_pattern(parts, key)]))
553 elseif command == "TICK"
554 ticks = number(args[1])
555 ticks > 0 || fail(POSITIVE)
556 store.clock += ticks
557 return integer_reply(purge!(store))
558 end
559 key = args[1]
560 if command == "EXPIRE"
561 condition = length(args) == 3 ? uppercase(args[3]) : nothing
562 !isnothing(condition) && !(condition in ("NX", "XX", "GT", "LT")) && fail("ERR Unsupported option $(args[3])")
563 seconds = number(args[2])
564 entry = get(store.database, key, nothing)
565 isnothing(entry) && return integer_reply(0)
566 deadline = store.clock + seconds
567 rejected = condition == "NX" ? !isnothing(entry.expiry) :
568 condition == "XX" ? isnothing(entry.expiry) :
569 condition == "GT" ? (isnothing(entry.expiry) || deadline <= entry.expiry) :
570 condition == "LT" ? (!isnothing(entry.expiry) && deadline >= entry.expiry) : false
571 rejected && return integer_reply(0)
572 if seconds <= 0
573 delete!(store.database, key)
574 else
575 store.database[key] = Entry(entry.value, deadline)
576 end
577 return integer_reply(1)
578 end
579 entry = get(store.database, key, nothing)
580 command == "TYPE" && return isnothing(entry) ? "none" : kind(entry.value)
581 command == "TTL" && return integer_reply(isnothing(entry) ? -2 : isnothing(entry.expiry) ? -1 : entry.expiry - store.clock)
582 if command == "PERSIST"
583 (isnothing(entry) || isnothing(entry.expiry)) && return integer_reply(0)
584 store.database[key] = Entry(entry.value, nothing)
585 return integer_reply(1)
586 end
587 isnothing(entry) && fail("ERR no such key")
588 destination = args[2]
589 command == "RENAMENX" && haskey(store.database, destination) && return integer_reply(0)
590 if key != destination
591 delete!(store.database, key)
592 store.database[destination] = entry
593 end
594 command == "RENAME" ? "OK" : integer_reply(1)
595end
596function control_command!(store::Store, command::String, args::Vector{String})
597 if command == "BEGIN"
598 push!(store.stack, copy(store.database))
599 elseif command in ("COMMIT", "ROLLBACK")
600 isempty(store.stack) && fail("ERR no transaction")
601 saved = pop!(store.stack)
602 if command == "ROLLBACK"
603 store.database = saved
604 purge!(store)
605 end
606 elseif command == "SNAPSHOT"
607 store.snapshots[args[1]] = copy(store.database)
608 else
609 saved = get(store.snapshots, args[1], nothing)
610 isnothing(saved) && fail("ERR no such snapshot")
611 store.database = copy(saved)
612 purge!(store)
613 end
614 "OK"
615end
616function execute!(store::Store, words::Vector{String})
617 command = command_name(words)
618 args = words[2:end]
619 command in ("SET", "GET", "MSET", "MGET", "INCR", "DECR", "INCRBY", "DECRBY", "APPEND", "STRLEN", "GETRANGE") &&
620 return string_command!(store, command, args)
621 command in ("LPUSH", "RPUSH", "LPOP", "RPOP", "LRANGE", "LLEN", "LINDEX", "LSET", "LREM", "LTRIM", "LINSERT") &&
622 return list_command!(store, command, args)
