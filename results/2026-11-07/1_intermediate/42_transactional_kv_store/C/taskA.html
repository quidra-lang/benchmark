51static bool equal_case(const char *a, const char *b) {
52 while (*a && *b) {
53 int x = *a++, y = *b++;
54 if (x >= 'a' && x <= 'z') x -= 'a' - 'A';
55 if (y >= 'a' && y <= 'z') y -= 'a' - 'A';
56 if (x != y) return false;
57 }
58 return *a == *b;
59}
60static bool split_line(const char *line, char *buffer, char **args, int *count) {
61 int input = 0, output = 0;
62 *count = 0;
63 while (line[input]) {
64 while (line[input] == ' ') ++input;
65 if (!line[input]) break;
66 args[(*count)++] = buffer + output;
67 if (line[input] == '"') {
68 ++input;
69 while (line[input] && line[input] != '"') {
70 if (line[input] == '\\') {
71 ++input;
72 if (line[input] != '\\' && line[input] != '"') return false;
73 }
74 buffer[output++] = line[input++];
75 }
76 if (line[input] != '"') return false;
77 ++input;
78 if (line[input] && line[input] != ' ') return false;
79 } else {
80 while (line[input] && line[input] != ' ') buffer[output++] = line[input++];
81 }
82 buffer[output++] = '\0';
83 }
84 return true;
85}
86static bool parse_integer(const char *text, int *value) {
87 if (!strcmp(text, "0")) { *value = 0; return true; }
88 bool negative = *text == '-';
89 if (negative) ++text;
90 size_t length = strlen(text);
91 if (length < 1 || length > 9 || *text < '1' || *text > '9') return false;
92 int result = 0;
93 for (; *text; ++text) {
94 if (*text < '0' || *text > '9') return false;
95 result = result * 10 + *text - '0';
96 }
97 *value = negative ? -result : result;
98 return true;
99}
100static int integer(const char *text) {
101 int value;
102 if (!parse_integer(text, &value)) fail("%s", not_integer);
103 return value;
104}
105static int incremented(int old, int change) {

108 return result;

110static Range normalize(int length, int start, int stop) {
111 if (start < 0) start += length;
112 if (stop < 0) stop += length;
113 if (start < 0) start = 0;
114 if (stop >= length) stop = length - 1;
115 return (Range){ start, stop };
116}
117static Bound bound(const char *text) {
118 if (!strcmp(text, "-inf")) return (Bound){ -1, 0, false };
119 if (!strcmp(text, "+inf")) return (Bound){ 1, 0, false };
120 bool exclusive = *text == '(';
121 int value;
122 if (!parse_integer(text + exclusive, &value)) fail("ERR min or max is not a float");
123 return (Bound){ 0, value, exclusive };
124}
125static void quoted(const char *text) {
126 if (!text) { fputs("(nil)", stdout); return; }
127 putchar('"');
128 for (; *text; ++text) {
129 if (*text == '"' || *text == '\\') putchar('\\');
130 putchar(*text);
131 }
132 putchar('"');
133}
134static void reply_string(const char *text) { quoted(text); putchar('\n'); }
135static void reply_integer(int value) { printf("(integer) %d\n", value); }
136static void append(Strings *strings, const char *text) {
137 if (strings->count == strings->capacity) {
138 strings->capacity = strings->capacity ? 2 * strings->capacity : 16;
139 const char **items = realloc(strings->items, (size_t)strings->capacity * sizeof(*items));
140 if (!items) exit(EXIT_FAILURE);
141 strings->items = items;
142 }
143 strings->items[strings->count++] = text;
144}
145static void reply_array(const Strings *strings) {
146 if (!strings->count) { puts("(empty array)"); return; }
147 int width = 1;
148 for (int n = strings->count; n >= 10; n /= 10) ++width;
149 for (int i = 0; i < strings->count; ++i) {
150 printf("%*d) ", width, i + 1);
151 quoted(strings->items[i]);
152 putchar('\n');
153 }
154}

170static Record *new_record(const char *key, Type type) {
171 Record *record = allocate(sizeof(*record));
172 *record = (Record){ .key = copy_string(key), .type = type, .expiry = -1 };
173 if (type == STRING) record->value.string = copy_string("");
174 else if (type == LIST) record->value.list = (List){0};
175 else record->value.members = (Collection){0};
176 return record;

178static Record *ensure(const char *key, Type type) {
179 Record *record = find(database, key);
180 if (!record) database->items[database->count++] = record = new_record(key, type);
181 return record;
182}
183static void free_record(Record *record) {
184 free(record->key);
185 if (record->type == STRING) free(record->value.string);
186 else if (record->type == LIST) {
187 for (int i = 0; i < record->value.list.count; ++i) free(record->value.list.items[i]);
188 } else {
189 for (int i = 0; i < record->value.members.count; ++i) {
190 free(record->value.members.items[i].name);
191 free(record->value.members.items[i].text);
192 }
193 }
194 free(record);
195}
196static bool delete_key(Database *db, const char *key) {
197 for (int i = 0; i < db->count; ++i) if (!strcmp(db->items[i]->key, key)) {
198 free_record(db->items[i]);
199 memmove(db->items + i, db->items + i + 1, (size_t)(--db->count - i) * sizeof(*db->items));
200 return true;
201 }
202 return false;
203}

219static Record *copy_record(const Record *source) {
220 Record *record = allocate(sizeof(*record));
221 *record = *source;
222 record->key = copy_string(source->key);
223 if (source->type == STRING) record->value.string = copy_string(source->value.string);
224 else if (source->type == LIST) {
225 for (int i = 0; i < source->value.list.count; ++i) record->value.list.items[i] = copy_string(source->value.list.items[i]);
226 } else {
227 for (int i = 0; i < source->value.members.count; ++i) {
228 Member *member = &record->value.members.items[i];
229 member->name = copy_string(member->name);
230 if (member->text) member->text = copy_string(member->text);
231 }
232 }
233 return record;
234}

256static int member_index(const Collection *members, const char *name) {
257 for (int i = 0; i < members->count; ++i) if (!strcmp(members->items[i].name, name)) return i;
258 return -1;

260static bool put_member(Collection *members, const char *name, const char *text, int score) {
261 int index = member_index(members, name);
262 bool added = index < 0;
263 if (added) {
264 index = members->count++;
265 members->items[index] = (Member){ .name = copy_string(name) };
266 }
267 Member *member = &members->items[index];
268 free(member->text);
269 member->text = text ? copy_string(text) : NULL;
270 member->score = score;
271 return added;
272}
273static bool remove_member(Collection *members, const char *name) {
274 int index = member_index(members, name);
275 if (index < 0) return false;
276 free(members->items[index].name);
277 free(members->items[index].text);
278 memmove(members->items + index, members->items + index + 1, (size_t)(--members->count - index) * sizeof(*members->items));
279 return true;
280}
281static int member_order(const void *a, const void *b) {
282 return strcmp((*(Member *const *)a)->name, (*(Member *const *)b)->name);
283}
284static int score_order(const void *a, const void *b) {
285 const Member *x = *(Member *const *)a, *y = *(Member *const *)b;
286 return x->score != y->score ? (x->score < y->score ? -1 : 1) : strcmp(x->name, y->name);
287}
288static int sorted_members(Record *record, Member **items, bool scores) {
289 int count = record ? record->value.members.count : 0;
290 for (int i = 0; i < count; ++i) items[i] = &record->value.members.items[i];
291 if (count > 1) qsort(items, (size_t)count, sizeof(*items), scores ? score_order : member_order);
292 return count;
293}

309typedef enum { LITERAL, ANY, STAR, CLASS } PartKind;
310typedef struct { PartKind kind; char literal; bool characters[128], negated; } Part;
311typedef struct { Part parts[201]; int count; } Pattern;
312static Pattern parse_pattern(const char *text) {
313 Pattern pattern = {0};
314 for (int i = 0; text[i]; ++i) {
315 Part *part = &pattern.parts[pattern.count++];
316 char character = text[i];
317 if (character == '\\') {
318 if (!text[++i]) fail("ERR invalid pattern");
319 part->literal = text[i];
320 } else if (character == '*') part->kind = STAR;
321 else if (character == '?') part->kind = ANY;
322 else if (character == '[') {
323 part->kind = CLASS;
324 int end = i + 1;
325 while (text[end] && text[end] != ']') ++end;
326 if (!text[end]) fail("ERR invalid pattern");
327 ++i;
328 if (text[i] == '^') { part->negated = true; ++i; }
329 if (i == end) fail("ERR invalid pattern");
330 while (i < end) {
331 int first = (unsigned char)text[i], last = first;
332 if (i + 2 < end && text[i + 1] == '-') { last = (unsigned char)text[i + 2]; i += 3; }
333 else ++i;
334 int low = first < last ? first : last, high = first > last ? first : last;
335 for (int c = low; c <= high; ++c) part->characters[c] = true;
336 }
337 } else part->literal = character;
338 }
339 return pattern;
340}
341static bool matches(const Pattern *pattern, const char *text) {
342 bool previous[201] = { true }, next[201];
343 int length = (int)strlen(text);
344 for (int i = 0; i < pattern->count; ++i) {
345 const Part *part = &pattern->parts[i];
346 memset(next, 0, sizeof(next));
347 if (part->kind == STAR) {
348 next[0] = previous[0];
349 for (int j = 1; j <= length; ++j) next[j] = previous[j] || next[j - 1];
350 } else {
351 for (int j = 0; j < length; ++j) {
352 bool accepts = part->kind == ANY || (part->kind == LITERAL && part->literal == text[j]) ||
353 (part->kind == CLASS && part->characters[(unsigned char)text[j]] != part->negated);
354 next[j + 1] = previous[j] && accepts;
355 }
356 }
357 memcpy(previous, next, sizeof(previous));
358 }
359 return previous[length];
360}

394static void execute(Command command, int count, char **args) {
395 Strings reply = {0};
396 char number[16];
397 switch (command) {
398 case C_SET: {
399 enum { NX = 1, XX = 2, EX = 4, KEEP = 8, GET = 16 };
400 int options = 0, seconds = 0;
401 const char *seconds_text = NULL;
402 for (int i = 2; i < count; ++i) {
403 int option = equal_case(args[i], "NX") ? NX : equal_case(args[i], "XX") ? XX :
404 equal_case(args[i], "EX") ? EX : equal_case(args[i], "KEEPTTL") ? KEEP : equal_case(args[i], "GET") ? GET : 0;
405 if (!option || (options & option)) fail("ERR syntax error");
406 options |= option;
407 if (option == EX) {
408 if (++i == count) fail("ERR syntax error");
409 seconds_text = args[i];
410 }
411 }
412 if (((options & NX) && (options & XX)) || ((options & EX) && (options & KEEP))) fail("ERR syntax error");
413 if (options & EX) {
414 seconds = integer(seconds_text);
415 if (seconds <= 0) fail("ERR invalid expire time in 'set' command");
416 }
417 Record *old = options & GET ? checked(args[0], STRING) : find(database, args[0]);
418 bool change = !((options & NX) && old) && !((options & XX) && !old);
419 char *previous = (options & GET) && old ? copy_string(old->value.string) : NULL;
420 if (change) {
421 int expiry = options & EX ? current_time + seconds : ((options & KEEP) && old ? old->expiry : -1);
422 replace_string(args[0], args[1], expiry);
423 }
424 if (options & GET) reply_string(previous);
425 else if (change) puts("OK");
426 else reply_string(NULL);
427 free(previous);
428 break;
429 }
430 case C_GET: case C_STRLEN: {
431 Record *record = checked(args[0], STRING);
432 if (command == C_GET) reply_string(record ? record->value.string : NULL);
433 else reply_integer(record ? (int)strlen(record->value.string) : 0);
434 break;
435 }
436 case C_MSET:
437 for (int i = 0; i < count; i += 2) replace_string(args[i], args[i + 1], -1);
438 puts("OK");
439 break;
440 case C_MGET:
441 for (int i = 0; i < count; ++i) {
442 Record *record = find(database, args[i]);
443 append(&reply, record && record->type == STRING ? record->value.string : NULL);
444 }
445 reply_array(&reply);
446 break;
447 case C_INCR: case C_DECR: case C_INCRBY: case C_DECRBY: {
448 int change = command == C_INCR ? 1 : command == C_DECR ? -1 : integer(args[1]);
449 if (command == C_DECRBY) change = -change;
450 Record *record = checked(args[0], STRING);
451 int old = record ? integer(record->value.string) : 0, result = incremented(old, change);
452 snprintf(number, sizeof(number), "%d", result);
453 write_string(ensure(args[0], STRING), number);
454 reply_integer(result);
455 break;
456 }
457 case C_APPEND: {
458 Record *record = checked(args[0], STRING);
459 size_t old_length = record ? strlen(record->value.string) : 0, added = strlen(args[1]);
460 char *text = allocate(old_length + added + 1);
461 if (record) memcpy(text, record->value.string, old_length);
462 memcpy(text + old_length, args[1], added + 1);
463 record = ensure(args[0], STRING);
464 free(record->value.string);
465 record->value.string = text;
466 reply_integer((int)(old_length + added));
467 break;
468 }
469 case C_GETRANGE: {
470 int start = integer(args[1]), stop = integer(args[2]);
471 Record *record = checked(args[0], STRING);
472 const char *text = record ? record->value.string : "";
473 Range range = normalize((int)strlen(text), start, stop);
474 int length = range.first <= range.last ? range.last - range.first + 1 : 0;
475 char result[1001];
476 if (length) memcpy(result, text + range.first, (size_t)length);
477 result[length] = '\0';
478 reply_string(result);
479 break;
480 }
481 case C_LPUSH: case C_RPUSH: {
482 checked(args[0], LIST);
483 List *list = &ensure(args[0], LIST)->value.list;
484 for (int i = 1; i < count; ++i) insert_element(list, command == C_LPUSH ? 0 : list->count, args[i]);
485 reply_integer(list->count);
486 break;
487 }
488 case C_LPOP: case C_RPOP: {
489 int wanted = 1;
490 if (count == 2 && (!parse_integer(args[1], &wanted) || wanted < 0)) fail("%s", positive);
491 Record *record = checked(args[0], LIST);
492 if (!record) { reply_string(NULL); break; }
493 List *list = &record->value.list;
494 if (wanted > list->count) wanted = list->count;
495 for (int i = 0; i < wanted; ++i) append(&reply, remove_element(list, command == C_LPOP ? 0 : list->count - 1));
496 remove_empty(record);
497 if (count == 1) reply_string(reply.items[0]);
498 else reply_array(&reply);
499 for (int i = 0; i < reply.count; ++i) free((void *)reply.items[i]);
500 break;
501 }
502 case C_LRANGE: case C_LTRIM: {
503 int start = integer(args[1]), stop = integer(args[2]);
504 Record *record = checked(args[0], LIST);
505 List *list = record ? &record->value.list : NULL;
506 Range range = normalize(list ? list->count : 0, start, stop);
507 if (command == C_LRANGE) {
508 for (int i = range.first; i <= range.last; ++i) append(&reply, list->items[i]);
509 reply_array(&reply);
510 } else {
511 if (list) {
512 int kept = 0;
513 for (int i = 0; i < list->count; ++i) {
514 if (i >= range.first && i <= range.last) list->items[kept++] = list->items[i];
515 else free(list->items[i]);
516 }
517 list->count = kept;
518 remove_empty(record);
519 }
520 puts("OK");
521 }
522 break;
523 }
524 case C_LLEN: {
525 Record *record = checked(args[0], LIST);
526 reply_integer(record ? record->value.list.count : 0);
527 break;
528 }
529 case C_LINDEX: case C_LSET: {
530 int index = integer(args[1]);
531 Record *record = checked(args[0], LIST);
532 if (command == C_LSET && !record) fail("ERR no such key");
533 int length = record ? record->value.list.count : 0;
534 if (index < 0) index += length;
535 bool valid = index >= 0 && index < length;
536 if (command == C_LINDEX) reply_string(valid ? record->value.list.items[index] : NULL);
537 else {
538 if (!valid) fail("ERR index out of range");
539 free(record->value.list.items[index]);
540 record->value.list.items[index] = copy_string(args[2]);
541 puts("OK");
542 }
543 break;
544 }
545 case C_LREM: {
546 int wanted = integer(args[1]);
547 Record *record = checked(args[0], LIST);
548 int removed = 0;
549 if (record) {
550 List *list = &record->value.list;
551 int limit = wanted < 0 ? -wanted : wanted;
552 for (int i = wanted < 0 ? list->count - 1 : 0;
553 i >= 0 && i < list->count && (!limit || removed < limit);) {
554 if (!strcmp(list->items[i], args[2])) {
555 free(remove_element(list, i));
556 ++removed;
557 if (wanted < 0) --i;
558 } else i += wanted < 0 ? -1 : 1;
559 }
560 remove_empty(record);
561 }
562 reply_integer(removed);
563 break;
564 }
565 case C_LINSERT: {
566 bool before = equal_case(args[1], "BEFORE");
567 if (!before && !equal_case(args[1], "AFTER")) fail("ERR syntax error");
568 Record *record = checked(args[0], LIST);
569 if (!record) { reply_integer(0); break; }
570 List *list = &record->value.list;
571 int index = 0;
572 while (index < list->count && strcmp(list->items[index], args[2])) ++index;
573 if (index == list->count) reply_integer(-1);
574 else {
575 insert_element(list, index + !before, args[3]);
576 reply_integer(list->count);
577 }
578 break;
579 }
580 case C_HSET: case C_SADD: case C_ZADD: {
581 int scores[100];
582 if (command == C_ZADD) for (int i = 1; i < count; i += 2) scores[i] = integer(args[i]);
583 Type type = command == C_HSET ? HASH : command == C_SADD ? SET : ZSET;
584 checked(args[0], type);
585 Collection *members = &ensure(args[0], type)->value.members;
586 int added = 0;
587 for (int i = 1; i < count; i += command == C_SADD ? 1 : 2) {
588 const char *name = command == C_ZADD ? args[i + 1] : args[i];
589 added += put_member(members, name, command == C_HSET ? args[i + 1] : NULL, command == C_ZADD ? scores[i] : 0);
590 }
591 reply_integer(added);
592 break;
593 }
594 case C_HGET: case C_HEXISTS: case C_SISMEMBER: case C_ZSCORE: {
595 Type type = command == C_SISMEMBER ? SET : command == C_ZSCORE ? ZSET : HASH;
596 Record *record = checked(args[0], type);
597 int index = record ? member_index(&record->value.members, args[1]) : -1;
598 if (command == C_HEXISTS || command == C_SISMEMBER) reply_integer(index >= 0);
599 else if (index < 0) reply_string(NULL);
600 else if (command == C_HGET) reply_string(record->value.members.items[index].text);
601 else {
602 snprintf(number, sizeof(number), "%d", record->value.members.items[index].score);
603 reply_string(number);
604 }
605 break;
606 }
607 case C_HDEL: case C_SREM: case C_ZREM: {
608 Type type = command == C_HDEL ? HASH : command == C_SREM ? SET : ZSET;
609 Record *record = checked(args[0], type);
610 int removed = 0;
611 if (record) {
612 for (int i = 1; i < count; ++i) removed += remove_member(&record->value.members, args[i]);
613 remove_empty(record);
614 }
615 reply_integer(removed);
616 break;
617 }
618 case C_HGETALL: case C_HKEYS: case C_HVALS: case C_SMEMBERS: {
619 Record *record = checked(args[0], command == C_SMEMBERS ? SET : HASH);
620 Member *members[100];
621 int length = sorted_members(record, members, false);
622 for (int i = 0; i < length; ++i) {
623 if (command != C_HVALS) append(&reply, members[i]->name);
624 if (command == C_HVALS || command == C_HGETALL) append(&reply, members[i]->text);
625 }
626 reply_array(&reply);
627 break;
628 }
629 case C_HLEN: case C_SCARD: case C_ZCARD: {
630 Record *record = checked(args[0], command == C_HLEN ? HASH : command == C_SCARD ? SET : ZSET);
631 reply_integer(record ? record->value.members.count : 0);
632 break;
633 }
634 case C_HINCRBY: case C_ZINCRBY: {
635 bool hash = command == C_HINCRBY;
636 int change = integer(args[hash ? 2 : 1]), old = 0;
637 const char *name = args[hash ? 1 : 2];
638 Record *record = checked(args[0], hash ? HASH : ZSET);
639 int index = record ? member_index(&record->value.members, name) : -1;
640 if (index >= 0) {
641 Member *member = &record->value.members.items[index];
642 if (hash) {
643 if (!parse_integer(member->text, &old)) fail("ERR hash value is not an integer");
644 } else old = member->score;
645 }
646 int result = incremented(old, change);
647 snprintf(number, sizeof(number), "%d", result);
648 put_member(&ensure(args[0], hash ? HASH : ZSET)->value.members, name, hash ? number : NULL, hash ? 0 : result);
649 if (hash) reply_integer(result);
650 else reply_string(number);
651 break;
652 }
653 case C_SINTER: case C_SUNION: case C_SDIFF:
654 case C_SINTERSTORE: case C_SUNIONSTORE: case C_SDIFFSTORE: {
655 bool store = command >= C_SINTERSTORE;
656 int first = store ? 1 : 0;
657 bool unite = command == C_SUNION || command == C_SUNIONSTORE;
658 bool intersect = command == C_SINTER || command == C_SINTERSTORE;
659 for (int i = first; i < count; ++i) checked(args[i], SET);
660 for (int i = first; i < (unite ? count : first + 1); ++i) {
661 Record *record = find(database, args[i]);
662 if (!record) continue;
663 for (int j = 0; j < record->value.members.count; ++j) {
664 const char *name = record->value.members.items[j].name;
665 bool keep = true;
666 if (!unite) for (int k = first + 1; k < count; ++k) {
667 Record *other = find(database, args[k]);
668 bool present = other && member_index(&other->value.members, name) >= 0;
669 if (intersect != present) { keep = false; break; }
670 }
671 if (keep) append(&reply, name);
672 }
673 }
674 sort_strings(&reply);
675 int unique = 0;
676 for (int i = 0; i < reply.count; ++i)
677 if (!unique || strcmp(reply.items[i], reply.items[unique - 1])) reply.items[unique++] = reply.items[i];
678 reply.count = unique;
679 if (store) {
680 if (!unique) delete_key(database, args[0]);
681 else {
682 Record *result = new_record(args[0], SET);
683 for (int i = 0; i < unique; ++i) put_member(&result->value.members, reply.items[i], NULL, 0);
684 replace_record(result);
685 }
686 reply_integer(unique);
687 } else reply_array(&reply);
688 break;
689 }
690 case C_ZRANK: {
691 Record *record = checked(args[0], ZSET);
692 Member *members[100];
693 int length = sorted_members(record, members, true), rank = 0;
694 while (rank < length && strcmp(members[rank]->name, args[1])) ++rank;
695 if (rank == length) reply_string(NULL);
696 else reply_integer(rank);
697 break;
698 }
699 case C_ZRANGE: case C_ZRANGEBYSCORE: {
700 bool scores = count == 4;
701 if (scores && !equal_case(args[3], "WITHSCORES")) fail("ERR syntax error");
702 int start = 0, stop = -1;
703 Bound minimum = {0}, maximum = {0};
704 if (command == C_ZRANGE) { start = integer(args[1]); stop = integer(args[2]); }
705 else { minimum = bound(args[1]); maximum = bound(args[2]); }
706 Record *record = checked(args[0], ZSET);
707 Member *members[100];
708 char score_text[100][16];
709 int length = sorted_members(record, members, true);
710 Range range = normalize(length, start, stop);
711 for (int i = 0; i < length; ++i) {
712 int score = members[i]->score;
713 bool keep = command == C_ZRANGE ? i >= range.first && i <= range.last :
714 (minimum.infinity < 0 || (!minimum.infinity && (score > minimum.value || (!minimum.exclusive && score == minimum.value)))) &&
715 (maximum.infinity > 0 || (!maximum.infinity && (score < maximum.value || (!maximum.exclusive && score == maximum.value))));
716 if (!keep) continue;
717 append(&reply, members[i]->name);
718 if (scores) {
719 snprintf(score_text[i], sizeof(score_text[i]), "%d", score);
720 append(&reply, score_text[i]);
721 }
722 }
723 reply_array(&reply);
724 break;
725 }
726 case C_DEL: case C_EXISTS: {
727 int result = 0;
728 for (int i = 0; i < count; ++i) result += command == C_DEL ? delete_key(database, args[i]) : find(database, args[i]) != NULL;
729 reply_integer(result);
730 break;
731 }
732 case C_TYPE: {
733 Record *record = find(database, args[0]);
734 puts(record ? type_names[record->type] : "none");
735 break;
736 }
737 case C_RENAME: case C_RENAMENX: {
738 Record *record = find(database, args[0]);
739 if (!record) fail("ERR no such key");
740 bool blocked = command == C_RENAMENX && find(database, args[1]);
741 if (!blocked && strcmp(args[0], args[1])) {
742 Record *moved = copy_record(record);
743 free(moved->key);
744 moved->key = copy_string(args[1]);
745 delete_key(database, args[0]);
746 replace_record(moved);
747 }
748 if (command == C_RENAME) puts("OK");
749 else reply_integer(!blocked);
750 break;
751 }
752 case C_KEYS: {
753 Pattern pattern = parse_pattern(args[0]);
754 for (int i = 0; i < database->count; ++i)
755 if (matches(&pattern, database->items[i]->key)) append(&reply, database->items[i]->key);
756 sort_strings(&reply);
757 reply_array(&reply);
758 break;
759 }
760 case C_EXPIRE: {
761 const char *condition = count == 3 ? args[2] : NULL;
762 if (condition && !equal_case(condition, "NX") && !equal_case(condition, "XX") && !equal_case(condition, "GT") && !equal_case(condition, "LT"))
763 fail("ERR Unsupported option %s", condition);
764 int seconds = integer(args[1]), expiry = current_time + seconds;
765 Record *record = find(database, args[0]);
766 bool apply = record != NULL;
767 if (record && condition) {
768 int old = record->expiry;
769 if (equal_case(condition, "NX")) apply = old < 0;
770 else if (equal_case(condition, "XX")) apply = old >= 0;
771 else if (equal_case(condition, "GT")) apply = old >= 0 && expiry > old;
772 else apply = old < 0 || expiry < old;
773 }
774 if (apply) {
775 if (seconds <= 0) delete_key(database, args[0]);
776 else record->expiry = expiry;
777 }
778 reply_integer(apply);
779 break;
780 }
781 case C_TTL: case C_PERSIST: {
782 Record *record = find(database, args[0]);
783 if (command == C_TTL) reply_integer(!record ? -2 : record->expiry < 0 ? -1 : record->expiry - current_time);
784 else {
785 bool changed = record && record->expiry >= 0;
786 if (changed) record->expiry = -1;
787 reply_integer(changed);
788 }
789 break;
790 }
791 case C_TICK: {
792 int ticks = integer(args[0]);
793 if (ticks <= 0) fail("%s", positive);
794 current_time += ticks;
795 reply_integer(expire_keys());
796 break;
797 }
798 case C_TIME: reply_integer(current_time); break;
799 case C_DBSIZE: reply_integer(database->count); break;
800 case C_FLUSH:
801 for (int i = 0; i < database->count; ++i) free_record(database->items[i]);
802 database->count = 0;
803 puts("OK");
804 break;
805 case C_INFO: {
806 int types[5] = {0}, expires = 0;
807 for (int i = 0; i < database->count; ++i) {
808 ++types[database->items[i]->type];
809 expires += database->items[i]->expiry >= 0;
810 }
811 printf("keys=%d,expires=%d,string=%d,list=%d,hash=%d,set=%d,zset=%d,time=%d,depth=%d,snapshots=%d\n",
812 database->count, expires, types[STRING], types[LIST], types[HASH], types[SET], types[ZSET], current_time, depth, snapshot_count);
813 break;
814 }
815 case C_BEGIN:
816 stack[depth++] = copy_database(database);
817 puts("OK");
818 break;
819 case C_COMMIT: case C_ROLLBACK:
820 if (!depth) fail("ERR no transaction");
821 if (command == C_COMMIT) free_database(stack[--depth]);
822 else {
823 free_database(database);
824 database = stack[--depth];
825 expire_keys();
826 }
827 puts("OK");
828 break;
829 case C_SNAPSHOT: case C_RESTORE: {
830 int index = 0;
831 while (index < snapshot_count && strcmp(snapshots[index].name, args[0])) ++index;
832 if (command == C_RESTORE) {
833 if (index == snapshot_count) fail("ERR no such snapshot");
834 Database *restored = copy_database(snapshots[index].database);
835 free_database(database);
836 database = restored;
837 expire_keys();
838 } else {
839 Database *saved = copy_database(database);
840 if (index == snapshot_count) {
841 snapshots[index].name = copy_string(args[0]);
842 ++snapshot_count;
843 } else free_database(snapshots[index].database);
844 snapshots[index].database = saved;
845 }
846 puts("OK");
847 break;
848 }
849 case COMMAND_COUNT: break;
850 }
851 free(reply.items);
852}

854static void process_line(const char *line) {
855 char buffer[202], *args[101];
856 int count;
857 if (!split_line(line, buffer, args, &count)) { puts("Invalid argument(s)"); return; }
858 int command = 0;
859 while (command < COMMAND_COUNT && !equal_case(args[0], formats[command].name)) ++command;
860 if (command == COMMAND_COUNT) fail("ERR unknown command '%s'", args[0]);
861 const Format *format = &formats[command];
862 --count;
863 if (count < format->minimum || count > format->maximum || (format->parity >= 0 && count % 2 != format->parity)) {
864 char name[20];
865 size_t i = 0;
866 for (; format->name[i]; ++i) name[i] = (char)(format->name[i] + 'a' - 'A');
867 name[i] = '\0';
868 fail("ERR wrong number of arguments for '%s' command", name);
869 }
870 execute((Command)command, count, args + 1);
871}
872int main(void) {
873 char line[205];
874 if (!fgets(line, sizeof(line), stdin)) return 0;
875 int commands = atoi(line);
876 database = allocate(sizeof(*database));
877 database->count = 0;
878 for (int i = 0; i < commands && fgets(line, sizeof(line), stdin); ++i) {
879 line[strcspn(line, "\r\n")] = '\0';
880 if (setjmp(command_error) == 0) process_line(line);
881 }
882 free_database(database);
883 for (int i = 0; i < depth; ++i) free_database(stack[i]);
884 for (int i = 0; i < snapshot_count; ++i) {
885 free(snapshots[i].name);
886 free_database(snapshots[i].database);
887 }
888 return 0;
889}
