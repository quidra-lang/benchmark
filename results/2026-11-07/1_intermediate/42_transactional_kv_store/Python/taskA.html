56def arity_matches(arity, count):

58 if count < minimum or (maximum is not None and count > maximum):

63def read_quoted(line, position):
64 content = []
65 while position < len(line):
66 character = line[position]
67 if character == '"':
68 return "".join(content), position + 1
69 if character == "\\":
70 if position + 1 == len(line) or line[position + 1] not in ('"', "\\"):
71 return None
72 position += 1
73 character = line[position]
74 content.append(character)
75 position += 1
76 return None

79def split_line(line):
80 arguments = []
81 position = 0
82 while position < len(line):
83 if line[position] == " ":
84 position += 1
85 elif line[position] == '"':
86 quoted = read_quoted(line, position + 1)
87 if quoted is None:
88 return None
89 argument, position = quoted
90 if position < len(line) and line[position] != " ":
91 return None
92 arguments.append(argument)
93 else:
94 end = line.find(" ", position)
95 if end < 0:
96 end = len(line)
97 arguments.append(line[position:end])
98 position = end
99 return arguments

119def normalize_range(length, start, stop):
120 if start < 0:
121 start += length
122 if stop < 0:
123 stop += length
124 start = max(start, 0)
125 stop = min(stop, length - 1)
126 if start > stop:
127 return 0, 0
128 return start, stop + 1

154class PatternPart:
155 def __init__(self, kind, characters=frozenset(), negated=False):
156 self.kind = kind
157 self.characters = characters
158 self.negated = negated

162 return True
163 return (character in self.characters) != self.negated

166def character_set(body):
167 characters = set()
168 index = 0
169 while index < len(body):
170 if index + 2 < len(body) and body[index + 1] == "-":
171 low, high = sorted((ord(body[index]), ord(body[index + 2])))
172 for code in range(low, high + 1):
173 characters.add(chr(code))
174 index += 3
175 else:
176 characters.add(body[index])
177 index += 1
178 return characters

181def parse_pattern(pattern):
182 parts = []
183 index = 0
184 while index < len(pattern):
185 character = pattern[index]
186 if character == "\\":
187 if index + 1 == len(pattern):
188 return None
189 parts.append(PatternPart(CHARACTER_SET, {pattern[index + 1]}))
190 index += 2
191 elif character == "[":
192 close = pattern.find("]", index + 1)
193 if close < 0:
194 return None
195 body = pattern[index + 1:close]
196 negated = body.startswith("^")
197 if negated:
198 body = body[1:]
199 if not body:
200 return None
201 parts.append(PatternPart(CHARACTER_SET, character_set(body), negated))
202 index = close + 1
203 elif character in (STAR, ANY_CHARACTER):
204 parts.append(PatternPart(character))
205 index += 1
206 else:
207 parts.append(PatternPart(CHARACTER_SET, {character}))
208 index += 1
209 return parts

212def pattern_matches(parts, text):
213 part_index = 0
214 position = 0
215 star_index = -1
216 star_position = 0
217 while position < len(text):
218 if part_index < len(parts) and parts[part_index].kind == STAR:
219 star_index = part_index
220 star_position = position
221 part_index += 1
222 elif part_index < len(parts) and parts[part_index].accepts(text[position]):
223 part_index += 1
224 position += 1
225 elif star_index >= 0:
226 star_position += 1
227 position = star_position
228 part_index = star_index + 1
229 else:
230 return False
231 while part_index < len(parts) and parts[part_index].kind == STAR:
232 part_index += 1
233 return part_index == len(parts)

258def parse_set_options(options):
259 seen = set()
260 seconds = None
261 index = 0
262 while index < len(options):
263 option = options[index].upper()
264 if option not in SET_OPTIONS or option in seen:
265 raise CommandError(SYNTAX)
266 seen.add(option)
267 if option == "EX":
268 if index + 1 == len(options):
269 raise CommandError(SYNTAX)
270 index += 1
271 seconds = options[index]
272 index += 1
273 if {"NX", "XX"} <= seen or {"EX", "KEEPTTL"} <= seen:
274 raise CommandError(SYNTAX)
275 return seen, seconds

422 def execute(self, line):
423 arguments = split_line(line)
424 if arguments is None:
425 return INVALID_LINE
426 name = arguments[0].upper()
427 if name not in ARITIES:
428 return error_reply(f"ERR unknown command '{arguments[0]}'")
429 if not arity_matches(ARITIES[name], len(arguments) - 1):
430 return error_reply(f"ERR wrong number of arguments for '{name.lower()}' command")
431 try:
432 return self.handlers[name](arguments[1:])
433 except CommandError as error:
434 return error_reply(str(error))

468 def set_value(self, arguments):
469 key, value = arguments[0], arguments[1]
470 options, seconds_text = parse_set_options(arguments[2:])
471 expiry = None
472 if "EX" in options:
473 seconds = integer_argument(seconds_text)
474 if seconds <= 0:
475 raise CommandError("ERR invalid expire time in 'set' command")
476 expiry = self.now + seconds
477 if "GET" in options:
478 previous = self.lookup(key, "string")
479 else:
480 previous = self.database.get(key)
481 exists = previous is not None
482 skipped = ("NX" in options and exists) or ("XX" in options and not exists)
483 if not skipped:
484 if "KEEPTTL" in options and exists:
485 expiry = previous.expiry
486 self.database[key] = Entry("string", value, expiry)
487 if "GET" in options:
488 return string_reply(previous.value) if exists else NIL
489 return NIL if skipped else OK

550 def pop(self, arguments, from_head):
551 key = arguments[0]
552 count = None
553 if len(arguments) == 2:
554 if not is_integer(arguments[1]) or int(arguments[1]) < 0:
555 raise CommandError(POSITIVE)
556 count = int(arguments[1])
557 entry = self.lookup(key, "list")
558 if entry is None:
559 return NIL
560 index = 0 if from_head else -1
561 if count is None:
562 element = entry.value.pop(index)
563 self.discard_if_empty(key)
564 return string_reply(element)
565 removed = []
566 while len(removed) < count and entry.value:
567 removed.append(entry.value.pop(index))
568 self.discard_if_empty(key)
569 return array_reply(removed)

604 def list_remove(self, arguments):
605 key, count_text, element = arguments
606 count = integer_argument(count_text)
607 entry = self.lookup(key, "list")
608 if entry is None:
609 return integer_reply(0)
610 scanned = entry.value if count >= 0 else reversed(entry.value)
611 kept = []
612 removed = 0
613 for value in scanned:
614 if value == element and (count == 0 or removed < abs(count)):
615 removed += 1
616 else:
617 kept.append(value)
618 if count < 0:
619 kept.reverse()
620 entry.value = kept
621 self.discard_if_empty(key)
622 return integer_reply(removed)

870 def expire(self, arguments):

872 condition = None
873 if len(arguments) == 3:
874 condition = arguments[2].upper()
875 if condition not in EXPIRE_CONDITIONS:
876 raise CommandError(f"ERR Unsupported option {arguments[2]}")
877 seconds = integer_argument(arguments[1])
878 entry = self.database.get(key)
879 if entry is None:
880 return integer_reply(0)
881 deadline = self.now + seconds
882 if condition is not None and expire_blocked(condition, entry.expiry, deadline):
883 return integer_reply(0)
884 if seconds <= 0:
885 del self.database[key]
