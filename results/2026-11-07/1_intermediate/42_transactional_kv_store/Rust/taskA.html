6type CommandResult<T> = Result<T, String>;
7const WRONG_TYPE: &str = "WRONGTYPE Operation against a key holding the wrong kind of value";
8const NOT_INTEGER: &str = "ERR value is not an integer or out of range";
9const POSITIVE: &str = "ERR value is out of range, must be positive";
10
11#[derive(Clone)]
12enum Value {
13 Text(String),
14 List(Vec<String>),
15 Hash(BTreeMap<String, String>),
16 Set(BTreeSet<String>),
17 SortedSet(BTreeMap<String, i32>),
18}
19impl Value {
20 fn kind(&self) -> &'static str {
21 match self {
22 Self::Text(_) => "string", Self::List(_) => "list", Self::Hash(_) => "hash",
23 Self::Set(_) => "set", Self::SortedSet(_) => "zset",
24 }
25 }
26 fn empty(&self) -> bool {
27 match self {
28 Self::Text(_) => false, Self::List(v) => v.is_empty(), Self::Hash(v) => v.is_empty(),
29 Self::Set(v) => v.is_empty(), Self::SortedSet(v) => v.is_empty(),
30 }
31 }
32}
33#[derive(Clone)]
34struct Entry { value: Value, expiry: Option<i32> }
35type Database = BTreeMap<String, Entry>;

37enum Reply { Status(String), Integer(i32), Text(Option<String>), Array(Vec<Option<String>>) }
38fn quote(text: &str) -> String { format!("\"{}\"", text.replace('\\', "\\\\").replace('"', "\\\"")) }
39impl Display for Reply {
40 fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
41 match self {
42 Self::Status(text) => f.write_str(text),
43 Self::Integer(value) => write!(f, "(integer) {value}"),
44 Self::Text(value) => f.write_str(&value.as_deref().map(quote).unwrap_or_else(|| "(nil)".into())),
45 Self::Array(values) => {
46 if values.is_empty() { return f.write_str("(empty array)"); }
47 let width = values.len().to_string().len();
48 for (index, value) in values.iter().enumerate() {
49 if index > 0 { writeln!(f)?; }
50 let item = value.as_deref().map(quote).unwrap_or_else(|| "(nil)".into());
51 write!(f, "{:>width$}) {item}", index + 1)?;
52 }
53 Ok(())
54 }
55 }
56 }
57}
58fn status(text: &str) -> Reply { Reply::Status(text.into()) }
59fn array(values: impl IntoIterator<Item = String>) -> Reply { Reply::Array(values.into_iter().map(Some).collect()) }

61fn split_line(line: &str) -> Option<Vec<String>> {
62 let bytes = line.as_bytes();
63 let mut args = Vec::new();
64 let mut position = 0;
65 while position < bytes.len() {
66 if bytes[position] == b' ' { position += 1; continue; }
67 if bytes[position] != b'"' {
68 let start = position;
69 while position < bytes.len() && bytes[position] != b' ' { position += 1; }
70 args.push(line[start..position].to_owned());
71 continue;
72 }
73 position += 1;
74 let mut text = Vec::new();
75 while position < bytes.len() && bytes[position] != b'"' {
76 if bytes[position] == b'\\' {
77 position += 1;
78 if position == bytes.len() || !matches!(bytes[position], b'\\' | b'"') { return None; }
79 }
80 text.push(bytes[position]);
81 position += 1;
82 }
83 if position == bytes.len() { return None; }
84 position += 1;
85 if position < bytes.len() && bytes[position] != b' ' { return None; }
86 args.push(String::from_utf8(text).unwrap());
87 }
88 Some(args)
89}
90fn parse_integer(text: &str) -> Option<i32> {
91 if text == "0" { return Some(0); }
92 let digits = text.strip_prefix('-').unwrap_or(text).as_bytes();
93 if !(1..=9).contains(&digits.len()) || !(b'1'..=b'9').contains(&digits[0]) || !digits.iter().all(u8::is_ascii_digit) {
94 return None;
95 }
96 text.parse().ok()
97}
98fn integer(text: &str) -> CommandResult<i32> { parse_integer(text).ok_or_else(|| NOT_INTEGER.into()) }
99fn add(old: i32, delta: i32) -> CommandResult<i32> {
100 let result = old + delta;
101 if !(-999_999_999..=999_999_999).contains(&result) { return Err("ERR increment or decrement would overflow".into()); }
102 Ok(result)
103}
104fn range(length: usize, mut start: i32, mut stop: i32) -> Range<usize> {
105 if start < 0 { start += length as i32; }
106 if stop < 0 { stop += length as i32; }
107 start = start.max(0);
108 stop = stop.min(length as i32 - 1);
109 if start > stop { 0..0 } else { start as usize..stop as usize + 1 }
110}

112enum Bound { NegativeInfinity, Finite(i32, bool), PositiveInfinity }
113impl Bound {
114 fn parse(text: &str) -> CommandResult<Self> {
115 match text {
116 "-inf" => Ok(Self::NegativeInfinity), "+inf" => Ok(Self::PositiveInfinity),
117 _ => {
118 let exclusive = text.starts_with('(');
119 let value = parse_integer(if exclusive { &text[1..] } else { text })
120 .ok_or_else(|| "ERR min or max is not a float".to_owned())?;
121 Ok(Self::Finite(value, exclusive))
122 }
123 }
124 }
125 fn accepts(&self, score: i32, lower: bool) -> bool {
126 match *self {
127 Self::NegativeInfinity => lower, Self::PositiveInfinity => !lower,
128 Self::Finite(value, exclusive) => if lower { score > value || (!exclusive && score == value) }
129 else { score < value || (!exclusive && score == value) },
130 }
131 }
132}
133enum Part { Literal(u8), Any, Star, Class([bool; 128], bool) }
134fn pattern(text: &str) -> CommandResult<Vec<Part>> {
135 let bytes = text.as_bytes();
136 let mut parts = Vec::new();
137 let mut i = 0;
138 while i < bytes.len() {
139 match bytes[i] {
140 b'\\' => {
141 i += 1;
142 if i == bytes.len() { return Err("ERR invalid pattern".into()); }
143 parts.push(Part::Literal(bytes[i]));
144 }
145 b'[' => {
146 let end = (i + 1..bytes.len()).find(|&j| bytes[j] == b']').ok_or_else(|| "ERR invalid pattern".to_owned())?;
147 i += 1;
148 let negated = i < end && bytes[i] == b'^';
149 if negated { i += 1; }
150 if i == end { return Err("ERR invalid pattern".into()); }
151 let mut characters = [false; 128];
152 while i < end {
153 let first = bytes[i] as usize;
154 if i + 2 < end && bytes[i + 1] == b'-' {
155 let last = bytes[i + 2] as usize;
156 characters[first.min(last)..=first.max(last)].fill(true);
157 i += 3;
158 } else { characters[first] = true; i += 1; }
159 }
160 parts.push(Part::Class(characters, negated));
161 }
162 b'*' => parts.push(Part::Star), b'?' => parts.push(Part::Any), c => parts.push(Part::Literal(c)),
163 }
164 i += 1;
165 }
166 Ok(parts)
167}
168fn matches_pattern(parts: &[Part], text: &str) -> bool {
169 let mut previous = vec![false; text.len() + 1];
170 previous[0] = true;
171 for part in parts {
172 let mut next = vec![false; previous.len()];
173 if matches!(part, Part::Star) {
174 next[0] = previous[0];
175 for i in 1..next.len() { next[i] = previous[i] || next[i - 1]; }
176 } else {
177 for (i, byte) in text.bytes().enumerate() {
178 let accepts = match part {
179 Part::Any => true, Part::Literal(c) => *c == byte,
180 Part::Class(characters, negated) => characters[byte as usize] != *negated, Part::Star => unreachable!(),
181 };
182 next[i + 1] = previous[i] && accepts;
183 }
184 }
185 previous = next;
186 }
187 previous[text.len()]
188}

190enum Arity { Exact(usize), Between(usize, usize), AtLeast(usize), Parity(usize, usize) }
191impl Arity {
192 fn accepts(&self, count: usize) -> bool {
193 match *self {
194 Self::Exact(n) => count == n, Self::Between(a, b) => (a..=b).contains(&count),
195 Self::AtLeast(n) => count >= n, Self::Parity(n, parity) => count >= n && count % 2 == parity,
196 }
197 }
198}
199fn arity(command: &str) -> Option<Arity> {
200 Some(match command {
201 "TIME" | "DBSIZE" | "FLUSH" | "INFO" | "BEGIN" | "COMMIT" | "ROLLBACK" => Arity::Exact(0),
202 "GET" | "INCR" | "DECR" | "STRLEN" | "LLEN" | "HGETALL" | "HLEN" | "HKEYS" | "HVALS" |
203 "SMEMBERS" | "SCARD" | "ZCARD" | "TYPE" | "KEYS" | "TTL" | "PERSIST" | "TICK" | "SNAPSHOT" | "RESTORE" => Arity::Exact(1),
204 "INCRBY" | "DECRBY" | "APPEND" | "LINDEX" | "HGET" | "HEXISTS" | "SISMEMBER" | "ZSCORE" | "ZRANK" | "RENAME" | "RENAMENX" => Arity::Exact(2),
205 "GETRANGE" | "LRANGE" | "LSET" | "LREM" | "LTRIM" | "HINCRBY" | "ZINCRBY" => Arity::Exact(3),
206 "LINSERT" => Arity::Exact(4), "LPOP" | "RPOP" => Arity::Between(1, 2), "EXPIRE" => Arity::Between(2, 3),
207 "ZRANGE" | "ZRANGEBYSCORE" => Arity::Between(3, 4),
208 "MGET" | "SINTER" | "SUNION" | "SDIFF" | "DEL" | "EXISTS" => Arity::AtLeast(1),
209 "SET" | "LPUSH" | "RPUSH" | "HDEL" | "SADD" | "SREM" | "ZREM" | "SINTERSTORE" | "SUNIONSTORE" | "SDIFFSTORE" => Arity::AtLeast(2),
210 "MSET" => Arity::Parity(2, 0), "HSET" | "ZADD" => Arity::Parity(3, 1), _ => return None,
211 })
212}

214#[derive(Default)]
215struct Store { database: Database, time: i32, stack: Vec<Database>, snapshots: BTreeMap<String, Database> }
216impl Store {
217 fn text(&self, key: &str) -> CommandResult<String> {
218 match self.database.get(key).map(|entry| &entry.value) {
219 None => Ok(String::new()), Some(Value::Text(text)) => Ok(text.clone()), _ => Err(WRONG_TYPE.into()),
220 }
221 }
222 fn list(&self, key: &str) -> CommandResult<Vec<String>> {
223 match self.database.get(key).map(|entry| &entry.value) {
224 None => Ok(Vec::new()), Some(Value::List(items)) => Ok(items.clone()), _ => Err(WRONG_TYPE.into()),
225 }
226 }
227 fn hash(&self, key: &str) -> CommandResult<BTreeMap<String, String>> {
228 match self.database.get(key).map(|entry| &entry.value) {
229 None => Ok(BTreeMap::new()), Some(Value::Hash(fields)) => Ok(fields.clone()), _ => Err(WRONG_TYPE.into()),
230 }
231 }
232 fn set(&self, key: &str) -> CommandResult<BTreeSet<String>> {
233 match self.database.get(key).map(|entry| &entry.value) {
234 None => Ok(BTreeSet::new()), Some(Value::Set(members)) => Ok(members.clone()), _ => Err(WRONG_TYPE.into()),
235 }
236 }
237 fn sorted_set(&self, key: &str) -> CommandResult<BTreeMap<String, i32>> {
238 match self.database.get(key).map(|entry| &entry.value) {
239 None => Ok(BTreeMap::new()), Some(Value::SortedSet(scores)) => Ok(scores.clone()), _ => Err(WRONG_TYPE.into()),
240 }
241 }
242 fn replace(&mut self, key: &str, value: Value, expiry: Option<i32>) {
243 if value.empty() { self.database.remove(key); }
244 else { self.database.insert(key.to_owned(), Entry { value, expiry }); }
245 }
246 fn update(&mut self, key: &str, value: Value) {
247 let expiry = self.database.get(key).and_then(|entry| entry.expiry);
248 self.replace(key, value, expiry);
249 }
250 fn purge(&mut self) -> i32 {
251 let before = self.database.len();
252 let time = self.time;
253 self.database.retain(|_, entry| entry.expiry.is_none_or(|expiry| expiry > time));
254 (before - self.database.len()) as i32
255 }
256 fn execute(&mut self, command: &str, a: &[String]) -> CommandResult<Reply> {
257 Ok(match command {
258 "SET" => {
259 let mut options = BTreeSet::new();
260 let mut seconds_text = None;
261 let mut i = 2;
262 while i < a.len() {
263 let option = a[i].to_ascii_uppercase();
264 if !matches!(option.as_str(), "NX" | "XX" | "EX" | "KEEPTTL" | "GET") || !options.insert(option.clone()) {
265 return Err("ERR syntax error".into());
266 }
267 if option == "EX" {
268 i += 1;
269 if i == a.len() { return Err("ERR syntax error".into()); }
270 seconds_text = Some(&a[i]);
271 }
272 i += 1;
273 }
274 if (options.contains("NX") && options.contains("XX")) || (options.contains("EX") && options.contains("KEEPTTL")) {
275 return Err("ERR syntax error".into());
276 }
277 let mut expiry = None;
278 if let Some(text) = seconds_text {
279 let seconds = integer(text)?;
280 if seconds <= 0 { return Err("ERR invalid expire time in 'set' command".into()); }
281 expiry = Some(self.time + seconds);
282 }
283 let exists = self.database.contains_key(&a[0]);
284 let old = if options.contains("GET") { let text = self.text(&a[0])?; exists.then_some(text) } else { None };
285 if options.contains("KEEPTTL") { expiry = self.database.get(&a[0]).and_then(|entry| entry.expiry); }
286 let changed = !(options.contains("NX") && exists) && !(options.contains("XX") && !exists);
287 if changed { self.replace(&a[0], Value::Text(a[1].clone()), expiry); }
288 if options.contains("GET") { Reply::Text(old) } else if changed { status("OK") } else { Reply::Text(None) }
289 }
290 "GET" | "STRLEN" => {
291 let text = self.text(&a[0])?;
292 if command == "GET" { Reply::Text(self.database.contains_key(&a[0]).then_some(text)) }
293 else { Reply::Integer(text.len() as i32) }
294 }
295 "MSET" => {
296 for pair in a.chunks_exact(2) { self.replace(&pair[0], Value::Text(pair[1].clone()), None); }
297 status("OK")
298 }
299 "MGET" => Reply::Array(a.iter().map(|key| match self.database.get(key).map(|entry| &entry.value) {
300 Some(Value::Text(text)) => Some(text.clone()), _ => None,
301 }).collect()),
302 "INCR" | "DECR" | "INCRBY" | "DECRBY" => {
303 let mut delta = if a.len() == 2 { integer(&a[1])? } else { 1 };
304 if matches!(command, "DECR" | "DECRBY") { delta = -delta; }
305 let text = self.text(&a[0])?;
306 let old = if self.database.contains_key(&a[0]) { integer(&text)? } else { 0 };
307 let result = add(old, delta)?;
308 self.update(&a[0], Value::Text(result.to_string()));
309 Reply::Integer(result)
310 }
311 "APPEND" => {
312 let mut text = self.text(&a[0])?;
313 text.push_str(&a[1]);
314 let length = text.len();
315 self.update(&a[0], Value::Text(text));
316 Reply::Integer(length as i32)
317 }
318 "GETRANGE" => {
319 let start = integer(&a[1])?;
320 let stop = integer(&a[2])?;
321 let text = self.text(&a[0])?;
322 Reply::Text(Some(text[range(text.len(), start, stop)].to_owned()))
323 }
324 "LPUSH" | "RPUSH" => {
325 let mut list = self.list(&a[0])?;
326 for element in &a[1..] {
327 if command == "LPUSH" { list.insert(0, element.clone()); } else { list.push(element.clone()); }
328 }
329 let length = list.len();
330 self.update(&a[0], Value::List(list));
331 Reply::Integer(length as i32)
332 }
333 "LPOP" | "RPOP" => {
334 let count = if a.len() == 2 { parse_integer(&a[1]).filter(|&n| n >= 0).ok_or_else(|| POSITIVE.to_owned())? } else { 1 };
335 let mut list = self.list(&a[0])?;
336 if !self.database.contains_key(&a[0]) { return Ok(Reply::Text(None)); }
337 let mut removed = Vec::new();
338 for _ in 0..(count as usize).min(list.len()) {
339 removed.push(if command == "LPOP" { list.remove(0) } else { list.pop().unwrap() });
340 }
341 self.update(&a[0], Value::List(list));
342 if a.len() == 1 { Reply::Text(removed.pop()) } else { array(removed) }
343 }
344 "LRANGE" | "LTRIM" => {
345 let start = integer(&a[1])?;
346 let stop = integer(&a[2])?;
347 let list = self.list(&a[0])?;
348 let kept = list[range(list.len(), start, stop)].to_vec();
349 if command == "LRANGE" { array(kept) }
350 else { self.update(&a[0], Value::List(kept)); status("OK") }
351 }
352 "LLEN" => Reply::Integer(self.list(&a[0])?.len() as i32),
353 "LINDEX" | "LSET" => {
354 let mut index = integer(&a[1])?;
355 let mut list = self.list(&a[0])?;
356 if command == "LSET" && !self.database.contains_key(&a[0]) { return Err("ERR no such key".into()); }
357 if index < 0 { index += list.len() as i32; }
358 let valid = (0..list.len() as i32).contains(&index);
359 if command == "LINDEX" { Reply::Text(if valid { Some(list[index as usize].clone()) } else { None }) }
360 else {
361 if !valid { return Err("ERR index out of range".into()); }
362 list[index as usize] = a[2].clone();
363 self.update(&a[0], Value::List(list));
364 status("OK")
365 }
366 }
367 "LREM" => {
368 let count = integer(&a[1])?;
369 let mut list = self.list(&a[0])?;
370 if count < 0 { list.reverse(); }
371 let limit = if count == 0 { list.len() } else { count.unsigned_abs() as usize };
372 let mut removed = 0;
373 list.retain(|element| {
374 if element == &a[2] && removed < limit { removed += 1; false } else { true }
375 });
376 if count < 0 { list.reverse(); }
377 self.update(&a[0], Value::List(list));
378 Reply::Integer(removed as i32)
379 }
380 "LINSERT" => {
381 let after = a[1].eq_ignore_ascii_case("AFTER");
382 if !after && !a[1].eq_ignore_ascii_case("BEFORE") { return Err("ERR syntax error".into()); }
383 let mut list = self.list(&a[0])?;
384 if !self.database.contains_key(&a[0]) { Reply::Integer(0) }
385 else if let Some(index) = list.iter().position(|element| element == &a[2]) {
386 list.insert(index + usize::from(after), a[3].clone());
387 let length = list.len();
388 self.update(&a[0], Value::List(list));
389 Reply::Integer(length as i32)
390 } else { Reply::Integer(-1) }
391 }
392 "HSET" | "HGET" | "HDEL" | "HGETALL" | "HLEN" | "HEXISTS" | "HKEYS" | "HVALS" | "HINCRBY" => {
393 let delta = if command == "HINCRBY" { integer(&a[2])? } else { 0 };
394 let mut fields = self.hash(&a[0])?;
395 match command {
396 "HSET" => {
397 let mut added = 0;
398 for pair in a[1..].chunks_exact(2) { added += i32::from(fields.insert(pair[0].clone(), pair[1].clone()).is_none()); }
399 self.update(&a[0], Value::Hash(fields));
400 Reply::Integer(added)
401 }
402 "HGET" => Reply::Text(fields.get(&a[1]).cloned()),
403 "HDEL" => {
404 let removed = a[1..].iter().filter(|field| fields.remove(*field).is_some()).count();
405 self.update(&a[0], Value::Hash(fields));
406 Reply::Integer(removed as i32)
407 }
408 "HLEN" => Reply::Integer(fields.len() as i32),
409 "HEXISTS" => Reply::Integer(i32::from(fields.contains_key(&a[1]))),
410 "HINCRBY" => {
411 let old = if let Some(text) = fields.get(&a[1]) {
412 parse_integer(text).ok_or_else(|| "ERR hash value is not an integer".to_owned())?
413 } else { 0 };
414 let result = add(old, delta)?;
415 fields.insert(a[1].clone(), result.to_string());
416 self.update(&a[0], Value::Hash(fields));
417 Reply::Integer(result)
418 }
419 _ => {
420 let mut result = Vec::new();
421 for (field, value) in fields {
422 if command != "HVALS" { result.push(field); }
423 if command != "HKEYS" { result.push(value); }
424 }
425 array(result)
426 }
427 }
428 }
429 "SINTER" | "SUNION" | "SDIFF" | "SINTERSTORE" | "SUNIONSTORE" | "SDIFFSTORE" => {
430 let storing = command.ends_with("STORE");
431 let operation = command.strip_suffix("STORE").unwrap_or(command);
432 let sources: Vec<BTreeSet<String>> = a[usize::from(storing)..].iter().map(|key| self.set(key)).collect::<CommandResult<_>>()?;
433 let mut result = sources[0].clone();
434 for source in &sources[1..] {
435 match operation {
436 "SINTER" => result.retain(|member| source.contains(member)),
437 "SUNION" => result.extend(source.iter().cloned()),
438 _ => result.retain(|member| !source.contains(member)),
439 }
440 }
441 if storing {
442 let length = result.len();
443 self.replace(&a[0], Value::Set(result), None);
444 Reply::Integer(length as i32)
445 } else { array(result) }
446 }
447 "SADD" | "SREM" | "SISMEMBER" | "SMEMBERS" | "SCARD" => {
448 let mut members = self.set(&a[0])?;
449 match command {
450 "SISMEMBER" => Reply::Integer(i32::from(members.contains(&a[1]))),
451 "SMEMBERS" => array(members), "SCARD" => Reply::Integer(members.len() as i32),
452 _ => {
453 let mut changed = 0;
454 for member in &a[1..] { changed += i32::from(if command == "SADD" { members.insert(member.clone()) } else { members.remove(member) }); }
455 self.update(&a[0], Value::Set(members));
456 Reply::Integer(changed)
457 }
458 }
459 }
460 "ZADD" => {
461 let scores: Vec<i32> = a[1..].chunks_exact(2).map(|pair| integer(&pair[0])).collect::<CommandResult<_>>()?;
462 let mut members = self.sorted_set(&a[0])?;
463 let mut added = 0;
464 for (pair, score) in a[1..].chunks_exact(2).zip(scores) {
465 added += i32::from(members.insert(pair[1].clone(), score).is_none());
466 }
467 self.update(&a[0], Value::SortedSet(members));
468 Reply::Integer(added)
469 }
470 "ZINCRBY" => {
471 let delta = integer(&a[1])?;
472 let mut members = self.sorted_set(&a[0])?;
473 let result = add(members.get(&a[2]).copied().unwrap_or(0), delta)?;
474 members.insert(a[2].clone(), result);
475 self.update(&a[0], Value::SortedSet(members));
476 Reply::Text(Some(result.to_string()))
477 }
478 "ZSCORE" | "ZREM" | "ZCARD" | "ZRANK" | "ZRANGE" | "ZRANGEBYSCORE" => {
479 let by_index = command == "ZRANGE";
480 let by_score = command == "ZRANGEBYSCORE";
481 let with_scores = a.len() == 4;
482 let (mut start, mut stop) = (0, 0);
483 let (mut minimum, mut maximum) = (Bound::NegativeInfinity, Bound::PositiveInfinity);
484 if by_index || by_score {
485 if with_scores && !a[3].eq_ignore_ascii_case("WITHSCORES") { return Err("ERR syntax error".into()); }
486 if by_index { start = integer(&a[1])?; stop = integer(&a[2])?; }
487 else { minimum = Bound::parse(&a[1])?; maximum = Bound::parse(&a[2])?; }
488 }
489 let mut members = self.sorted_set(&a[0])?;
490 match command {
491 "ZSCORE" => Reply::Text(members.get(&a[1]).map(i32::to_string)),
492 "ZREM" => {
493 let removed = a[1..].iter().filter(|member| members.remove(*member).is_some()).count();
494 self.update(&a[0], Value::SortedSet(members));
495 Reply::Integer(removed as i32)
496 }
497 "ZCARD" => Reply::Integer(members.len() as i32),
498 _ => {
499 let mut ordered: Vec<_> = members.into_iter().collect();
500 ordered.sort_by(|(a, x), (b, y)| x.cmp(y).then_with(|| a.cmp(b)));
501 if command == "ZRANK" {
502 match ordered.iter().position(|(member, _)| member == &a[1]) {
503 Some(rank) => Reply::Integer(rank as i32), None => Reply::Text(None),
504 }
505 } else {
506 let selection = if by_index { range(ordered.len(), start, stop) } else { 0..ordered.len() };
507 let mut result = Vec::new();
508 for (member, score) in &ordered[selection] {
509 if by_score && (!minimum.accepts(*score, true) || !maximum.accepts(*score, false)) { continue; }
510 result.push(member.clone());
511 if with_scores { result.push(score.to_string()); }
512 }
513 array(result)
514 }
515 }
516 }
517 }
518 "DEL" | "EXISTS" => {
519 let mut count = 0;
520 for key in a { count += i32::from(if command == "DEL" { self.database.remove(key).is_some() } else { self.database.contains_key(key) }); }
521 Reply::Integer(count)
522 }
523 "TYPE" => status(self.database.get(&a[0]).map_or("none", |entry| entry.value.kind())),
524 "RENAME" | "RENAMENX" => {
525 if !self.database.contains_key(&a[0]) { return Err("ERR no such key".into()); }
526 if command == "RENAMENX" && self.database.contains_key(&a[1]) { Reply::Integer(0) }
527 else {
528 if a[0] != a[1] { let entry = self.database.remove(&a[0]).unwrap(); self.database.insert(a[1].clone(), entry); }
529 if command == "RENAME" { status("OK") } else { Reply::Integer(1) }
530 }
531 }
532 "KEYS" => {
533 let parts = pattern(&a[0])?;
534 array(self.database.keys().filter(|key| matches_pattern(&parts, key)).cloned())
535 }
536 "EXPIRE" => {
537 let condition = a.get(2).map(|text| text.to_ascii_uppercase());
538 if condition.as_deref().is_some_and(|text| !matches!(text, "NX" | "XX" | "GT" | "LT")) {
539 return Err(format!("ERR Unsupported option {}", a[2]));
540 }
541 let seconds = integer(&a[1])?;
542 let expiry = self.time + seconds;
543 if let Some(entry) = self.database.get_mut(&a[0]) {
544 let rejected = match condition.as_deref() {
545 Some("NX") => entry.expiry.is_some(), Some("XX") => entry.expiry.is_none(),
546 Some("GT") => entry.expiry.is_none_or(|old| expiry <= old),
547 Some("LT") => entry.expiry.is_some_and(|old| expiry >= old), _ => false,
548 };
549 if rejected { Reply::Integer(0) }
550 else {
551 if seconds <= 0 { self.database.remove(&a[0]); } else { entry.expiry = Some(expiry); }
552 Reply::Integer(1)
553 }
554 } else { Reply::Integer(0) }
555 }
556 "TTL" => Reply::Integer(self.database.get(&a[0]).map_or(-2, |entry| entry.expiry.map_or(-1, |expiry| expiry - self.time))),
557 "PERSIST" => {
558 let changed = self.database.get_mut(&a[0]).is_some_and(|entry| entry.expiry.take().is_some());
559 Reply::Integer(i32::from(changed))
560 }
561 "TICK" => {
562 let ticks = integer(&a[0])?;
563 if ticks <= 0 { return Err(POSITIVE.into()); }
564 self.time += ticks;
565 Reply::Integer(self.purge())
566 }
567 "TIME" => Reply::Integer(self.time), "DBSIZE" => Reply::Integer(self.database.len() as i32),
568 "FLUSH" => { self.database.clear(); status("OK") }
569 "INFO" => {
570 let count = |kind: &str| self.database.values().filter(|entry| entry.value.kind() == kind).count();
571 Reply::Status(format!("keys={},expires={},string={},list={},hash={},set={},zset={},time={},depth={},snapshots={}",
572 self.database.len(), self.database.values().filter(|entry| entry.expiry.is_some()).count(),
573 count("string"), count("list"), count("hash"), count("set"), count("zset"), self.time, self.stack.len(), self.snapshots.len()))
574 }
575 "BEGIN" => { self.stack.push(self.database.clone()); status("OK") }
576 "COMMIT" | "ROLLBACK" => {
577 let saved = self.stack.pop().ok_or_else(|| "ERR no transaction".to_owned())?;
578 if command == "ROLLBACK" { self.database = saved; self.purge(); }
579 status("OK")
580 }
581 "SNAPSHOT" => { self.snapshots.insert(a[0].clone(), self.database.clone()); status("OK") }
582 "RESTORE" => {
583 let saved = self.snapshots.get(&a[0]).ok_or_else(|| "ERR no such snapshot".to_owned())?;
584 self.database = saved.clone();
585 self.purge();
586 status("OK")
587 }
588 _ => unreachable!(),
589 })
590 }
591 fn process(&mut self, line: &str) -> CommandResult<Reply> {
592 let Some(words) = split_line(line) else { return Ok(status("Invalid argument(s)")); };
593 let command = words[0].to_ascii_uppercase();
594 let spec = arity(&command).ok_or_else(|| format!("ERR unknown command '{}'", words[0]))?;
595 if !spec.accepts(words.len() - 1) {
596 return Err(format!("ERR wrong number of arguments for '{}' command", command.to_ascii_lowercase()));
597 }
598 self.execute(&command, &words[1..])
599 }
600}

602fn main() {
603 let mut input = String::new();
604 io::stdin().read_to_string(&mut input).unwrap();
605 let mut lines = input.lines();
606 let Some(first) = lines.next() else { return; };
607 let count: usize = first.parse().unwrap();
608 let mut store = Store::default();
609 let mut output = io::BufWriter::new(io::stdout().lock());
610 for line in lines.take(count) {
611 match store.process(line) {
612 Ok(reply) => writeln!(output, "{reply}").unwrap(),
613 Err(message) => writeln!(output, "(error) {message}").unwrap(),
614 }
615 }
