17split_line <- function(line) {
18 chars <- strsplit(line, "", fixed = TRUE)[[1L]]
19 n <- length(chars)
20 i <- 1L
21 words <- character()
22 while (i <= n) {
23 if (chars[i] == " ") { i <- i + 1L; next }
24 if (chars[i] != '"') {
25 start <- i
26 while (i <= n && chars[i] != " ") i <- i + 1L
27 words <- c(words, paste(chars[seq.int(start, i - 1L)], collapse = ""))
28 next
29 }
30 i <- i + 1L
31 text <- character()
32 closed <- FALSE
33 while (i <= n) {
34 value <- chars[i]
35 i <- i + 1L
36 if (value == '"') { closed <- TRUE; break }
37 if (value == "\\") {
38 if (i > n) return(NULL)
39 value <- chars[i]
40 i <- i + 1L
41 if (!value %in% c("\\", '"')) return(NULL)
42 }
43 text <- c(text, value)
44 }
45 if (!closed || (i <= n && chars[i] != " ")) return(NULL)
46 words <- c(words, paste(text, collapse = ""))
47 }
48 words
49}

65score_bound <- function(text) {
66 if (text == "-inf") return(list(infinity = -1L, value = 0L, exclusive = FALSE))
67 if (text == "+inf") return(list(infinity = 1L, value = 0L, exclusive = FALSE))
68 exclusive <- startsWith(text, "(")
69 if (exclusive) text <- substring(text, 2L)
70 if (!valid_integer(text)) fail("ERR min or max is not a float")
71 list(infinity = 0L, value = as.integer(text), exclusive = exclusive)
72}
73within_bound <- function(scores, bound, lower) {
74 if (bound$infinity) return(rep(lower == (bound$infinity < 0L), length(scores)))
75 (if (lower) scores > bound$value else scores < bound$value) | (!bound$exclusive & scores == bound$value)

77parse_pattern <- function(text) {
78 chars <- strsplit(text, "", fixed = TRUE)[[1L]]
79 parts <- list()
80 i <- 1L
81 while (i <= length(chars)) {
82 value <- chars[i]
83 i <- i + 1L
84 if (value == "\\") {
85 if (i > length(chars)) fail("ERR invalid pattern")
86 part <- list(kind = "literal", value = chars[i])
87 i <- i + 1L
88 } else if (value == "[") {
89 end <- i
90 while (end <= length(chars) && chars[end] != "]") end <- end + 1L
91 if (end > length(chars)) fail("ERR invalid pattern")
92 negated <- i < end && chars[i] == "^"
93 if (negated) i <- i + 1L
94 if (i == end) fail("ERR invalid pattern")
95 members <- rep(FALSE, 128L)
96 while (i < end) {
97 first <- utf8ToInt(chars[i])
98 if (i + 2L < end && chars[i + 1L] == "-") {
99 last <- utf8ToInt(chars[i + 2L])
100 members[seq.int(min(first, last), max(first, last)) + 1L] <- TRUE
101 i <- i + 3L
102 } else {
103 members[first + 1L] <- TRUE
104 i <- i + 1L
105 }
106 }
107 part <- list(kind = "class", members = members, negated = negated)
108 i <- end + 1L
109 } else part <- list(kind = if (value %in% c("*", "?")) value else "literal", value = value)
110 parts <- c(parts, list(part))
111 }
112 parts
113}
114matches <- function(parts, text) {
115 chars <- strsplit(text, "", fixed = TRUE)[[1L]]
116 previous <- c(TRUE, rep(FALSE, length(chars)))
117 for (part in parts) {
118 following <- rep(FALSE, length(previous))
119 if (part$kind == "*") {
120 following[1L] <- previous[1L]
121 for (i in seq_along(chars)) following[i + 1L] <- previous[i + 1L] || following[i]
122 } else {
123 accepted <- switch(part$kind, "?" = rep(TRUE, length(chars)), literal = chars == part$value,
124 class = part$members[utf8ToInt(text) + 1L] != part$negated)
125 following <- c(FALSE, head(previous, -1L) & accepted)
126 }
127 previous <- following
128 }
129 tail(previous, 1L)
130}

209set_string <- function(state, a) {
210 key <- a[1L]
211 options <- character()
212 seconds_text <- NULL
213 i <- 3L
214 while (i <= length(a)) {
215 option <- upper(a[i])
216 i <- i + 1L
217 if (!option %in% c("NX", "XX", "EX", "KEEPTTL", "GET") || option %in% options) fail("ERR syntax error")
218 options <- c(options, option)
219 if (option == "EX") {
220 if (i > length(a)) fail("ERR syntax error")
221 seconds_text <- a[i]
222 i <- i + 1L
223 }
224 }
225 if (all(c("NX", "XX") %in% options) || all(c("EX", "KEEPTTL") %in% options)) fail("ERR syntax error")
226 expiry <- NULL
227 if ("EX" %in% options) {
228 seconds <- number(seconds_text)
229 if (seconds <= 0L) fail("ERR invalid expire time in 'set' command")
230 expiry <- state$clock + seconds
231 }
232 old <- db_get(state$db, key)
233 old_text <- NULL
234 if ("GET" %in% options) {
235 text <- read_value(state, key, "string")
236 if (!is.null(old)) old_text <- text
237 }
238 blocked <- ("NX" %in% options && !is.null(old)) || ("XX" %in% options && is.null(old))
239 if (!blocked) {
240 if ("KEEPTTL" %in% options && !is.null(old)) expiry <- old$expiry
241 replace_value(state, key, "string", a[2L], expiry)
242 }
243 if ("GET" %in% options) quoted(old_text) else if (blocked) "(nil)" else "OK"
244}
245string_command <- function(state, command, a) {
246 key <- a[1L]
247 if (command == "SET") return(set_string(state, a))
248 if (command == "MSET") {
249 for (i in seq.int(1L, length(a), by = 2L)) replace_value(state, a[i], "string", a[i + 1L])
250 return("OK")
251 }
252 if (command == "MGET") return(array_reply(lapply(a, function(name) {
253 entry <- db_get(state$db, name)
254 if (!is.null(entry) && entry$kind == "string") entry$value else NULL
255 })))

257 delta <- if (length(a) == 2L) number(a[2L]) else 1L
258 if (startsWith(command, "DECR")) delta <- -delta
259 text <- read_value(state, key, "string")
260 previous <- if (is.null(db_get(state$db, key))) 0L else number(text)
261 value <- incremented(previous, delta)
262 update_value(state, key, "string", as.character(value))
263 return(integer_reply(value))
264 }

266 value <- paste0(read_value(state, key, "string"), a[2L])

268 return(integer_reply(nchar(value)))
269 }
270 if (command == "GETRANGE") {
271 first <- number(a[2L])
272 last <- number(a[3L])
273 chars <- strsplit(read_value(state, key, "string"), "", fixed = TRUE)[[1L]]
274 return(quoted(paste(chars[selected_indices(length(chars), first, last)], collapse = "")))
275 }
276 value <- read_value(state, key, "string")
277 if (command == "STRLEN") integer_reply(nchar(value)) else quoted(if (is.null(db_get(state$db, key))) NULL else value)

279list_command <- function(state, command, a) {
280 key <- a[1L]
281 if (command %in% c("LPUSH", "RPUSH")) {
282 value <- read_value(state, key, "list")
283 value <- if (command == "LPUSH") c(rev(a[-1L]), value) else c(value, a[-1L])
284 update_value(state, key, "list", value)
285 return(integer_reply(length(value)))
286 }
287 if (command %in% c("LPOP", "RPOP")) {
288 count <- 1L
289 explicit <- length(a) == 2L
290 if (explicit) {
291 if (!valid_integer(a[2L])) fail(positive)
292 count <- as.integer(a[2L])
293 if (count < 0L) fail(positive)
294 }
295 value <- read_value(state, key, "list")
296 if (is.null(db_get(state$db, key))) return("(nil)")
297 count <- min(count, length(value))
298 if (command == "LPOP") {
299 removed <- head(value, count)
300 kept <- tail(value, length(value) - count)
301 } else {
302 removed <- rev(tail(value, count))
303 kept <- head(value, length(value) - count)
304 }
305 update_value(state, key, "list", kept)
306 return(if (explicit) array_reply(removed) else quoted(removed[1L]))
307 }
308 if (command %in% c("LRANGE", "LTRIM")) {
309 first <- number(a[2L])
310 last <- number(a[3L])
311 value <- read_value(state, key, "list")
312 kept <- value[selected_indices(length(value), first, last)]
313 if (command == "LRANGE") return(array_reply(kept))
314 update_value(state, key, "list", kept)
315 return("OK")
316 }
317 if (command == "LLEN") return(integer_reply(length(read_value(state, key, "list"))))
318 if (command %in% c("LINDEX", "LSET")) {
319 index <- number(a[2L])
320 value <- read_value(state, key, "list")
321 if (command == "LSET" && is.null(db_get(state$db, key))) fail("ERR no such key")
322 if (index < 0L) index <- index + length(value)
323 if (index < 0L || index >= length(value)) {
324 if (command == "LINDEX") return("(nil)")
325 fail("ERR index out of range")
326 }
327 if (command == "LINDEX") return(quoted(value[index + 1L]))
328 value[index + 1L] <- a[3L]
329 update_value(state, key, "list", value)
330 return("OK")
331 }
332 if (command == "LREM") {
333 count <- number(a[2L])
334 value <- read_value(state, key, "list")
335 remove <- which(value == a[3L])
336 if (count > 0L) remove <- head(remove, min(count, length(remove)))
337 if (count < 0L) remove <- tail(remove, min(-count, length(remove)))
338 update_value(state, key, "list", value[!seq_along(value) %in% remove])
339 return(integer_reply(length(remove)))
340 }
341 where <- upper(a[2L])
342 if (!where %in% c("BEFORE", "AFTER")) fail("ERR syntax error")
343 value <- read_value(state, key, "list")
344 if (!length(value)) return(integer_reply(0L))
345 at <- match(a[3L], value, nomatch = 0L)
346 if (!at) return(integer_reply(-1L))
347 before <- at - as.integer(where == "BEFORE")
348 value <- c(head(value, before), a[4L], tail(value, length(value) - before))
349 update_value(state, key, "list", value)
350 integer_reply(length(value))
351}
352hash_command <- function(state, command, a) {
353 key <- a[1L]
354 if (command == "HINCRBY") delta <- number(a[3L])
355 value <- read_value(state, key, "hash")
356 if (command == "HINCRBY") {
357 at <- match(a[2L], value$keys, nomatch = 0L)
358 previous <- 0L
359 if (at) {
360 if (!valid_integer(value$values[at])) fail("ERR hash value is not an integer")
361 previous <- as.integer(value$values[at])
362 }
363 result <- incremented(previous, delta)
364 update_value(state, key, "hash", assoc_put(value, a[2L], as.character(result)))
365 return(integer_reply(result))
366 }
367 if (command == "HSET") {
368 old <- length(value$keys)
369 for (i in seq.int(2L, length(a), by = 2L)) value <- assoc_put(value, a[i], a[i + 1L])
370 update_value(state, key, "hash", value)
371 return(integer_reply(length(value$keys) - old))
372 }
373 if (command == "HDEL") {
374 kept <- assoc_remove(value, a[-1L])
375 update_value(state, key, "hash", kept)
376 return(integer_reply(length(value$keys) - length(kept$keys)))
377 }
378 if (command == "HGET") {
379 at <- match(a[2L], value$keys, nomatch = 0L)
380 return(quoted(if (at) value$values[at] else NULL))
381 }
382 if (command == "HEXISTS") return(integer_reply(as.integer(a[2L] %in% value$keys)))
383 if (command == "HLEN") return(integer_reply(length(value$keys)))
384 order <- order(value$keys, method = "radix")
385 if (command == "HKEYS") return(array_reply(value$keys[order]))
386 if (command == "HVALS") return(array_reply(value$values[order]))
387 array_reply(as.vector(rbind(value$keys[order], value$values[order])))
388}
389set_command <- function(state, command, a) {
390 key <- a[1L]
391 if (startsWith(command, "SINTER") || startsWith(command, "SUNION") || startsWith(command, "SDIFF")) {
392 saving <- endsWith(command, "STORE")
393 sources <- lapply(if (saving) a[-1L] else a, function(name) read_value(state, name, "set"))
394 operation <- if (startsWith(command, "SINTER")) intersect else if (startsWith(command, "SUNION")) union else setdiff
395 result <- Reduce(operation, sources)
396 if (!saving) return(array_reply(sort(result, method = "radix")))
397 replace_value(state, key, "set", result)
398 return(integer_reply(length(result)))
399 }
400 value <- read_value(state, key, "set")
401 if (command %in% c("SADD", "SREM")) {
402 result <- if (command == "SADD") union(value, a[-1L]) else setdiff(value, a[-1L])
403 changed <- abs(length(result) - length(value))
404 update_value(state, key, "set", result)
405 return(integer_reply(changed))
406 }
407 if (command == "SCARD") return(integer_reply(length(value)))
408 if (command == "SISMEMBER") return(integer_reply(as.integer(a[2L] %in% value)))
409 array_reply(sort(value, method = "radix"))
410}
411zset_command <- function(state, command, a) {
412 key <- a[1L]
413 if (command == "ZADD") {
414 indices <- seq.int(2L, length(a), by = 2L)
415 scores <- vapply(a[indices], number, integer(1L), USE.NAMES = FALSE)
416 value <- read_value(state, key, "zset")
417 old <- length(value$keys)
418 for (i in seq_along(scores)) value <- assoc_put(value, a[indices[i] + 1L], scores[i])
419 update_value(state, key, "zset", value)
420 return(integer_reply(length(value$keys) - old))
421 }
422 if (command == "ZINCRBY") {
423 delta <- number(a[2L])
424 value <- read_value(state, key, "zset")
425 at <- match(a[3L], value$keys, nomatch = 0L)
426 result <- incremented(if (at) value$values[at] else 0L, delta)
427 update_value(state, key, "zset", assoc_put(value, a[3L], result))
428 return(quoted(as.character(result)))
429 }
430 with_scores <- FALSE
431 if (command %in% c("ZRANGE", "ZRANGEBYSCORE")) {
432 with_scores <- length(a) == 4L
433 if (with_scores && upper(a[4L]) != "WITHSCORES") fail("ERR syntax error")
434 if (command == "ZRANGE") {
435 first <- number(a[2L])
436 last <- number(a[3L])
437 } else {
438 minimum <- score_bound(a[2L])
439 maximum <- score_bound(a[3L])
440 }
441 }
442 value <- read_value(state, key, "zset")
443 if (command == "ZREM") {
444 kept <- assoc_remove(value, a[-1L])
445 update_value(state, key, "zset", kept)
446 return(integer_reply(length(value$keys) - length(kept$keys)))
447 }
448 if (command == "ZCARD") return(integer_reply(length(value$keys)))
449 if (command == "ZSCORE") {
450 at <- match(a[2L], value$keys, nomatch = 0L)
451 return(quoted(if (at) as.character(value$values[at]) else NULL))
452 }
453 positions <- order(value$values, value$keys, method = "radix")
454 if (command == "ZRANK") {
455 at <- match(a[2L], value$keys[positions], nomatch = 0L)
456 return(if (at) integer_reply(at - 1L) else "(nil)")
457 }
458 if (command == "ZRANGE") positions <- positions[selected_indices(length(positions), first, last)] else
459 positions <- positions[within_bound(value$values[positions], minimum, TRUE) & within_bound(value$values[positions], maximum, FALSE)]
460 items <- value$keys[positions]
461 if (with_scores) items <- as.vector(rbind(items, as.character(value$values[positions])))
462 array_reply(items)
463}
464key_command <- function(state, command, a) {
465 if (command == "TIME") return(integer_reply(state$clock))
466 if (command == "DBSIZE") return(integer_reply(length(state$db)))
467 if (command == "FLUSH") { state$db <- list(); return("OK") }
468 if (command == "INFO") {
469 kinds <- c("string", "list", "hash", "set", "zset")
470 counts <- vapply(kinds, function(kind) sum(vapply(state$db, function(entry) entry$kind == kind, logical(1L))), integer(1L))
471 expires <- sum(vapply(state$db, function(entry) !is.null(entry$expiry), logical(1L)))
472 return(paste0("keys=", length(state$db), ",expires=", expires, ",string=", counts[1L], ",list=", counts[2L],
473 ",hash=", counts[3L], ",set=", counts[4L], ",zset=", counts[5L], ",time=", state$clock,
474 ",depth=", length(state$stack), ",snapshots=", length(state$snapshots)))
475 }
476 if (command == "DEL") return(integer_reply(sum(vapply(a, function(key) erase(state, key), logical(1L)))))
477 if (command == "EXISTS") return(integer_reply(sum(a %in% names(state$db))))
478 if (command == "KEYS") {
479 parts <- parse_pattern(a[1L])
480 keys <- names(state$db)
481 found <- keys[vapply(keys, function(key) matches(parts, key), logical(1L))]
482 return(array_reply(sort(found, method = "radix")))
483 }
484 if (command == "TICK") {
485 ticks <- number(a[1L])
486 if (ticks <= 0L) fail(positive)
487 state$clock <- state$clock + ticks
488 return(integer_reply(purge(state)))
489 }
490 key <- a[1L]
491 if (command == "EXPIRE") {
492 condition <- if (length(a) == 3L) upper(a[3L]) else ""
493 if (length(a) == 3L && !condition %in% c("NX", "XX", "GT", "LT")) fail(paste0("ERR Unsupported option ", a[3L]))
494 seconds <- number(a[2L])
495 entry <- db_get(state$db, key)
496 if (is.null(entry)) return(integer_reply(0L))
497 deadline <- state$clock + seconds
498 rejected <- switch(condition, NX = !is.null(entry$expiry), XX = is.null(entry$expiry),
499 GT = is.null(entry$expiry) || deadline <= entry$expiry,
500 LT = !is.null(entry$expiry) && deadline >= entry$expiry, FALSE)
501 if (rejected) return(integer_reply(0L))
502 if (seconds <= 0L) erase(state, key) else {
503 entry$expiry <- deadline
504 state$db <- db_put(state$db, key, entry)
505 }
506 return(integer_reply(1L))
507 }
508 entry <- db_get(state$db, key)
509 if (command == "TYPE") return(if (is.null(entry)) "none" else entry$kind)
510 if (command == "TTL") return(integer_reply(if (is.null(entry)) -2L else if (is.null(entry$expiry)) -1L else entry$expiry - state$clock))
511 if (command == "PERSIST") {
512 if (is.null(entry$expiry)) return(integer_reply(0L))
513 entry$expiry <- NULL
514 state$db <- db_put(state$db, key, entry)
515 return(integer_reply(1L))
516 }
517 if (is.null(entry)) fail("ERR no such key")
518 destination <- a[2L]
519 if (command == "RENAMENX" && !is.null(db_get(state$db, destination))) return(integer_reply(0L))
520 if (key != destination) {
521 erase(state, key)
522 state$db <- db_put(state$db, destination, entry)
523 }
524 if (command == "RENAME") "OK" else integer_reply(1L)
525}
526control_command <- function(state, command, a) {
527 if (command == "BEGIN") state$stack <- c(state$stack, list(state$db)) else if (command %in% c("COMMIT", "ROLLBACK")) {
528 if (!length(state$stack)) fail("ERR no transaction")
529 if (command == "ROLLBACK") {
530 state$db <- state$stack[[length(state$stack)]]
531 purge(state)
532 }
533 state$stack <- head(state$stack, -1L)
534 } else if (command == "SNAPSHOT") state$snapshots <- db_put(state$snapshots, a[1L], state$db) else {
535 saved <- db_get(state$snapshots, a[1L])
536 if (is.null(saved)) fail("ERR no such snapshot")
537 state$db <- saved
538 purge(state)
539 }
540 "OK"
541}
542execute <- function(state, words) {
543 command <- check_arity(words[1L], length(words) - 1L)
544 a <- words[-1L]
545 if (command %in% strsplit("SET GET MSET MGET INCR DECR INCRBY DECRBY APPEND STRLEN GETRANGE", " ")[[1L]])
546 return(string_command(state, command, a))
547 if (command %in% strsplit("LPUSH RPUSH LPOP RPOP LRANGE LLEN LINDEX LSET LREM LTRIM LINSERT", " ")[[1L]])
548 return(list_command(state, command, a))
549 if (command %in% strsplit("HSET HGET HDEL HGETALL HLEN HEXISTS HKEYS HVALS HINCRBY", " ")[[1L]])
550 return(hash_command(state, command, a))
551 if (command %in% strsplit("SADD SREM SISMEMBER SMEMBERS SCARD SINTER SUNION SDIFF SINTERSTORE SUNIONSTORE SDIFFSTORE", " ")[[1L]])
552 return(set_command(state, command, a))
553 if (command %in% strsplit("ZADD ZSCORE ZINCRBY ZREM ZCARD ZRANK ZRANGE ZRANGEBYSCORE", " ")[[1L]])
554 return(zset_command(state, command, a))
555 if (command %in% c("BEGIN", "COMMIT", "ROLLBACK", "SNAPSHOT", "RESTORE"))
556 return(control_command(state, command, a))
557 key_command(state, command, a)
558}
