50function quoteText(string $value): string { return '"' . str_replace(['\\', '"'], ['\\\\', '\\"'], $value) . '"'; }
51function integerReply(int $value): array { return ['(integer) ' . $value]; }
52function textReply(?string $value): array { return [$value === null ? '(nil)' : quoteText($value)]; }
53function arrayReply(array $values): array
54{
55 if ($values === []) return ['(empty array)'];
56 $width = strlen((string)count($values));
57 $output = [];
58 foreach (array_values($values) as $i => $value)
59 $output[] = str_pad((string)($i + 1), $width, ' ', STR_PAD_LEFT) . ') ' . ($value === null ? '(nil)' : quoteText($value));
60 return $output;
61}
62function parseNumber(string $word, string $error = NOT_INTEGER): int
63{
64 if ($word === '0') return 0;
65 $start = str_starts_with($word, '-') ? 1 : 0;
66 $length = strlen($word) - $start;
67 if ($length < 1 || $length > 9 || $word[$start] < '1' || $word[$start] > '9') throw new CommandError($error);
68 for ($i = $start; $i < strlen($word); $i++)
69 if ($word[$i] < '0' || $word[$i] > '9') throw new CommandError($error);
70 return (int)$word;
71}
72function checkedAdd(int $value, int $delta): int
73{
74 $result = $value + $delta;
75 if ($result < -999999999 || $result > 999999999) throw new CommandError(OVERFLOW);
76 return $result;
77}
78function normalizeRange(int $length, int $start, int $stop): array
79{
80 if ($start < 0) $start += $length;
81 if ($stop < 0) $stop += $length;
82 $start = max(0, $start);
83 $stop = min($length - 1, $stop);
84 return $start <= $stop ? [$start, $stop - $start + 1] : [0, 0];
85}
86function splitArguments(string $line): ?array
87{
88 $words = [];
89 $position = 0;
90 $length = strlen($line);
91 while ($position < $length) {
92 if ($line[$position] === ' ') { $position++; continue; }
93 if ($line[$position] !== '"') {
94 $start = $position;
95 while ($position < $length && $line[$position] !== ' ') $position++;
96 $words[] = substr($line, $start, $position - $start);
97 continue;
98 }
99 $position++;
100 $value = '';
101 $closed = false;
102 while ($position < $length) {
103 $character = $line[$position++];
104 if ($character === '"') { $closed = true; break; }
105 if ($character === '\\') {
106 if ($position === $length || ($line[$position] !== '"' && $line[$position] !== '\\')) return null;
107 $character = $line[$position++];
108 }
109 $value .= $character;
110 }
111 if (!$closed || ($position < $length && $line[$position] !== ' ')) return null;
112 $words[] = $value;
113 }
114 return $words;
115}
116function parsePattern(string $pattern): array
117{
118 $parts = [];
119 for ($i = 0; $i < strlen($pattern); $i++) {
120 $character = $pattern[$i];
121 if ($character === '\\') {
122 if (++$i === strlen($pattern)) throw new CommandError('ERR invalid pattern');
123 $parts[] = new PatternPart(PatternKind::Literal, $pattern[$i]);
124 } elseif ($character === '[') {
125 $end = strpos($pattern, ']', $i + 1);
126 if ($end === false) throw new CommandError('ERR invalid pattern');
127 $start = $i + 1;
128 $negated = $start < $end && $pattern[$start] === '^';
129 if ($negated) $start++;
130 if ($start === $end) throw new CommandError('ERR invalid pattern');
131 $characters = [];
132 for ($j = $start; $j < $end;) {
133 if ($j + 2 < $end && $pattern[$j + 1] === '-') {
134 $first = min(ord($pattern[$j]), ord($pattern[$j + 2]));
135 $last = max(ord($pattern[$j]), ord($pattern[$j + 2]));
136 for ($code = $first; $code <= $last; $code++) $characters[$code] = true;
137 $j += 3;
138 } else $characters[ord($pattern[$j++])] = true;
139 }
140 $parts[] = new PatternPart(PatternKind::CharacterClass, characters: $characters, negated: $negated);
141 $i = $end;
142 } else {
143 $kind = match ($character) { '*' => PatternKind::Star, '?' => PatternKind::Any, default => PatternKind::Literal };
144 $parts[] = new PatternPart($kind, $character);
145 }
146 }
147 return $parts;
148}
149function matchesPattern(array $parts, string $value): bool
150{
151 $previous = array_fill(0, strlen($value) + 1, false);
152 $previous[0] = true;
153 foreach ($parts as $part) {
154 $current = array_fill(0, count($previous), false);
155 if ($part->kind === PatternKind::Star) {
156 $current[0] = $previous[0];
157 for ($i = 1; $i < count($current); $i++) $current[$i] = $previous[$i] || $current[$i - 1];
158 } else {
159 for ($i = 0; $i < strlen($value); $i++) {
160 $accepts = match ($part->kind) {
161 PatternKind::Any => true,
162 PatternKind::Literal => $part->literal === $value[$i],
163 default => isset($part->characters[ord($value[$i])]) !== $part->negated,
164 };
165 $current[$i + 1] = $previous[$i] && $accepts;
166 }
167 }
168 $previous = $current;
169 }
170 return $previous[strlen($value)];
171}
172function parseBound(string $word): Bound
173{
174 if ($word === '-inf') return new Bound(infinity: -1);
175 if ($word === '+inf') return new Bound(infinity: 1);
176 $inclusive = !str_starts_with($word, '(');
177 return new Bound(parseNumber($inclusive ? $word : substr($word, 1), 'ERR min or max is not a float'), $inclusive);
178}

246 $removed = 0;

249 return $removed;

251 public function execute(array $words, array $specs): array
252 {
253 $command = strtoupper($words[0]);
254 $spec = $specs[$command] ?? null;
255 if ($spec === null) throw new CommandError("ERR unknown command '" . $words[0] . "'");
256 $args = array_slice($words, 1);
257 $count = count($args);
258 if ($count < $spec->minimum || ($spec->maximum >= 0 && $count > $spec->maximum)
259 || ($spec->parity >= 0 && $count % 2 !== $spec->parity))
260 throw new CommandError("ERR wrong number of arguments for '" . strtolower($command) . "' command");
261 return match ($spec->group) {
262 Group::Text => $this->textCommand($command, $args),
263 Group::Sequence => $this->listCommand($command, $args),
264 Group::Hash => $this->hashCommand($command, $args),
265 Group::Members => $this->setCommand($command, $args),
266 Group::Scores => $this->sortedSetCommand($command, $args),
267 Group::Key => $this->keyCommand($command, $args),
268 Group::Control => $this->controlCommand($command, $args),
269 };

271 private function textCommand(string $command, array $args): array
272 {
273 $key = $args[0];
274 if ($command === 'SET') {
275 $options = [];
276 $secondsWord = null;
277 for ($i = 2; $i < count($args); $i++) {
278 $option = strtoupper($args[$i]);
279 if (!in_array($option, ['NX', 'XX', 'EX', 'KEEPTTL', 'GET'], true) || isset($options[$option])) throw new CommandError(BAD_SYNTAX);
280 $options[$option] = true;
281 if ($option === 'EX') {
282 if (++$i === count($args)) throw new CommandError(BAD_SYNTAX);
283 $secondsWord = $args[$i];
284 }
285 }
286 if ((isset($options['NX']) && isset($options['XX'])) || (isset($options['EX']) && isset($options['KEEPTTL'])))
287 throw new CommandError(BAD_SYNTAX);
288 $seconds = $secondsWord === null ? null : parseNumber($secondsWord);
289 if ($seconds !== null && $seconds <= 0) throw new CommandError("ERR invalid expire time in 'set' command");
290 $old = $this->entry($key);
291 $previous = isset($options['GET']) ? $this->value($key, Kind::Text)?->data : null;
292 $changed = !(isset($options['NX']) && $old !== null) && !(isset($options['XX']) && $old === null);
293 if ($changed) {
294 $expiry = $seconds !== null ? $this->time + $seconds : (isset($options['KEEPTTL']) ? $old?->expiry : null);
295 $this->put($key, new Value(Kind::Text, $args[1]), $expiry);
296 }
297 return isset($options['GET']) ? textReply($previous) : ($changed ? ['OK'] : textReply(null));
298 }
299 if ($command === 'MSET') {
300 for ($i = 0; $i < count($args); $i += 2) $this->put($args[$i], new Value(Kind::Text, $args[$i + 1]));
301 return ['OK'];
302 }
303 if ($command === 'MGET') {
304 $result = [];
305 foreach ($args as $name) {
306 $entry = $this->entry($name);
307 $result[] = $entry?->value->kind === Kind::Text ? $entry->value->data : null;
308 }
309 return arrayReply($result);
310 }
311 if (in_array($command, ['INCR', 'DECR', 'INCRBY', 'DECRBY'], true)) {
312 $delta = count($args) === 2 ? parseNumber($args[1]) : 1;
313 if ($command === 'DECR' || $command === 'DECRBY') $delta = -$delta;
314 $old = $this->value($key, Kind::Text);
315 $result = checkedAdd($old === null ? 0 : parseNumber($old->data), $delta);
316 $this->update($key, Kind::Text, (string)$result);
317 return integerReply($result);
318 }
319 $start = $stop = 0;
320 if ($command === 'GETRANGE') { $start = parseNumber($args[1]); $stop = parseNumber($args[2]); }
321 $old = $this->value($key, Kind::Text);
322 if ($command === 'GET') return textReply($old?->data);
323 $text = $old?->data ?? '';
324 if ($command === 'STRLEN') return integerReply(strlen($text));
325 if ($command === 'APPEND') {
326 $text .= $args[1];
327 $this->update($key, Kind::Text, $text);
328 return integerReply(strlen($text));
329 }
330 [$start, $count] = normalizeRange(strlen($text), $start, $stop);
331 return textReply(substr($text, $start, $count));
332 }
333 private function listCommand(string $command, array $args): array
334 {
335 $key = $args[0];
336 $first = $second = 0;
337 if ($command === 'LPOP' || $command === 'RPOP') {
338 if (count($args) === 2) {
339 $first = parseNumber($args[1], POSITIVE);
340 if ($first < 0) throw new CommandError(POSITIVE);
341 }
342 } elseif (in_array($command, ['LINDEX', 'LSET', 'LREM', 'LRANGE', 'LTRIM'], true)) {
343 $first = parseNumber($args[1]);
344 if ($command === 'LRANGE' || $command === 'LTRIM') $second = parseNumber($args[2]);
345 } elseif ($command === 'LINSERT' && !in_array(strtoupper($args[1]), ['BEFORE', 'AFTER'], true)) {
346 throw new CommandError(BAD_SYNTAX);
347 }
348 $old = $this->value($key, Kind::Sequence);
349 $items = $old?->data ?? [];
350 if ($command === 'LPUSH' || $command === 'RPUSH') {
351 foreach (array_slice($args, 1) as $item)
352 if ($command === 'LPUSH') array_unshift($items, $item); else $items[] = $item;
353 $this->update($key, Kind::Sequence, $items);
354 return integerReply(count($items));
355 }
356 if ($command === 'LPOP' || $command === 'RPOP') {
357 if ($old === null) return textReply(null);
358 $count = count($args) === 1 ? 1 : min($first, count($items));
359 $removed = [];
360 for ($i = 0; $i < $count; $i++) $removed[] = $command === 'LPOP' ? array_shift($items) : array_pop($items);
361 $this->update($key, Kind::Sequence, $items);
362 return count($args) === 1 ? textReply($removed[0]) : arrayReply($removed);
363 }
364 if ($command === 'LLEN') return integerReply(count($items));
365 if ($command === 'LINDEX' || $command === 'LSET') {
366 $index = $first < 0 ? $first + count($items) : $first;
367 $valid = $index >= 0 && $index < count($items);
368 if ($command === 'LINDEX') return textReply($valid ? $items[$index] : null);
369 if ($old === null) throw new CommandError(NO_KEY);
370 if (!$valid) throw new CommandError('ERR index out of range');
371 $items[$index] = $args[2];
372 $this->update($key, Kind::Sequence, $items);
373 return ['OK'];
374 }
375 if ($command === 'LRANGE' || $command === 'LTRIM') {
376 [$start, $count] = normalizeRange(count($items), $first, $second);
377 $items = array_slice($items, $start, $count);
378 if ($command === 'LRANGE') return arrayReply($items);
379 $this->update($key, Kind::Sequence, $items);
380 return ['OK'];
381 }
382 if ($command === 'LREM') {
383 $removed = 0;
384 $indices = array_keys($items);
385 if ($first < 0) $indices = array_reverse($indices);
386 foreach ($indices as $index)
387 if ($items[$index] === $args[2] && ($first === 0 || $removed < abs($first))) {
388 unset($items[$index]);
389 $removed++;
390 }
391 $this->update($key, Kind::Sequence, array_values($items));
392 return integerReply($removed);
393 }
394 if ($old === null) return integerReply(0);
395 $index = array_search($args[2], $items, true);
396 if ($index === false) return integerReply(-1);
397 if (strtoupper($args[1]) === 'AFTER') $index++;
398 array_splice($items, $index, 0, [$args[3]]);
399 $this->update($key, Kind::Sequence, $items);
400 return integerReply(count($items));
401 }
402 private function hashCommand(string $command, array $args): array
403 {
404 $key = $args[0];
405 $delta = $command === 'HINCRBY' ? parseNumber($args[2]) : 0;
406 $fields = $this->value($key, Kind::Hash)?->data ?? [];
407 if ($command === 'HSET') {
408 $added = 0;
409 for ($i = 1; $i < count($args); $i += 2) {
410 $field = ':' . $args[$i];
411 if (!isset($fields[$field])) $added++;
412 $fields[$field] = $args[$i + 1];
413 }
414 $this->update($key, Kind::Hash, $fields);
415 return integerReply($added);
416 }
417 if ($command === 'HGET') return textReply($fields[':' . $args[1]] ?? null);
418 if ($command === 'HEXISTS') return integerReply(isset($fields[':' . $args[1]]) ? 1 : 0);
419 if ($command === 'HLEN') return integerReply(count($fields));
420 if ($command === 'HDEL') {
421 $removed = 0;
422 foreach (array_slice($args, 1) as $name) {
423 $field = ':' . $name;
424 if (isset($fields[$field])) { unset($fields[$field]); $removed++; }
425 }
426 $this->update($key, Kind::Hash, $fields);
427 return integerReply($removed);
428 }
429 if ($command === 'HINCRBY') {
430 $field = ':' . $args[1];
431 $old = isset($fields[$field]) ? parseNumber($fields[$field], 'ERR hash value is not an integer') : 0;
432 $result = checkedAdd($old, $delta);
433 $fields[$field] = (string)$result;
434 $this->update($key, Kind::Hash, $fields);
435 return integerReply($result);
436 }
437 ksort($fields, SORT_STRING);
438 $result = [];
439 foreach ($fields as $field => $value) {
440 if ($command !== 'HVALS') $result[] = substr($field, 1);
441 if ($command !== 'HKEYS') $result[] = $value;
442 }
443 return arrayReply($result);
444 }
445 private function setCommand(string $command, array $args): array
446 {
447 $storing = str_ends_with($command, 'STORE');
448 if (str_starts_with($command, 'SINTER') || str_starts_with($command, 'SUNION') || str_starts_with($command, 'SDIFF')) {
449 $sources = [];
450 foreach (array_slice($args, $storing ? 1 : 0) as $key) $sources[] = $this->value($key, Kind::Members)?->data ?? [];
451 $members = $sources[0];
452 foreach (array_slice($sources, 1) as $source)
453 if (str_starts_with($command, 'SINTER')) $members = array_intersect_key($members, $source);
454 elseif (str_starts_with($command, 'SUNION')) $members += $source;
455 else $members = array_diff_key($members, $source);
456 if ($storing) {
457 if ($members === []) unset($this->database[':' . $args[0]]);
458 else $this->put($args[0], new Value(Kind::Members, $members));
459 return integerReply(count($members));
460 }
461 } else {
462 $key = $args[0];
463 $members = $this->value($key, Kind::Members)?->data ?? [];
464 if ($command === 'SADD' || $command === 'SREM') {
465 $changed = 0;
466 foreach (array_slice($args, 1) as $name) {
467 $member = ':' . $name;
468 if ($command === 'SADD' && !isset($members[$member])) { $members[$member] = true; $changed++; }
469 elseif ($command === 'SREM' && isset($members[$member])) { unset($members[$member]); $changed++; }
470 }
471 $this->update($key, Kind::Members, $members);
472 return integerReply($changed);
473 }
474 if ($command === 'SISMEMBER') return integerReply(isset($members[':' . $args[1]]) ? 1 : 0);
475 if ($command === 'SCARD') return integerReply(count($members));
476 }
477 $names = array_keys($members);
478 sort($names, SORT_STRING);
479 return arrayReply(array_map(static fn(string $name): string => substr($name, 1), $names));
480 }
481 private function sortedSetCommand(string $command, array $args): array
482 {
483 $key = $args[0];
484 $numbers = [];
485 $lower = $upper = null;
486 $withScores = count($args) === 4 && ($command === 'ZRANGE' || $command === 'ZRANGEBYSCORE');
487 if ($withScores && strtoupper($args[3]) !== 'WITHSCORES') throw new CommandError(BAD_SYNTAX);
488 if ($command === 'ZADD')
489 for ($i = 1; $i < count($args); $i += 2) $numbers[] = parseNumber($args[$i]);
490 elseif ($command === 'ZINCRBY') $numbers[] = parseNumber($args[1]);
491 elseif ($command === 'ZRANGE') { $numbers[] = parseNumber($args[1]); $numbers[] = parseNumber($args[2]); }
492 elseif ($command === 'ZRANGEBYSCORE') { $lower = parseBound($args[1]); $upper = parseBound($args[2]); }
493 $scores = $this->value($key, Kind::Scores)?->data ?? [];
494 if ($command === 'ZADD') {
495 $added = 0;
496 foreach ($numbers as $i => $score) {
497 $member = ':' . $args[2 * $i + 2];
498 if (!isset($scores[$member])) $added++;
499 $scores[$member] = $score;
500 }
501 $this->update($key, Kind::Scores, $scores);
502 return integerReply($added);
503 }
504 if ($command === 'ZSCORE') {
505 $score = $scores[':' . $args[1]] ?? null;
506 return textReply($score === null ? null : (string)$score);
507 }
508 if ($command === 'ZINCRBY') {
509 $member = ':' . $args[2];
510 $result = checkedAdd($scores[$member] ?? 0, $numbers[0]);
511 $scores[$member] = $result;
512 $this->update($key, Kind::Scores, $scores);
513 return textReply((string)$result);
514 }
515 if ($command === 'ZREM') {
516 $removed = 0;
517 foreach (array_slice($args, 1) as $name) {
518 $member = ':' . $name;
519 if (isset($scores[$member])) { unset($scores[$member]); $removed++; }
520 }
521 $this->update($key, Kind::Scores, $scores);
522 return integerReply($removed);
523 }
524 if ($command === 'ZCARD') return integerReply(count($scores));
525 $members = array_keys($scores);
526 usort($members, static fn(string $a, string $b): int => ($scores[$a] <=> $scores[$b]) ?: strcmp($a, $b));
527 if ($command === 'ZRANK') {
528 $rank = array_search(':' . $args[1], $members, true);
529 return $rank === false ? textReply(null) : integerReply($rank);
530 }
531 if ($command === 'ZRANGE') {
532 [$start, $count] = normalizeRange(count($members), $numbers[0], $numbers[1]);
533 $members = array_slice($members, $start, $count);
534 } else $members = array_values(array_filter($members,
535 static fn(string $member): bool => $lower->accepts($scores[$member], true) && $upper->accepts($scores[$member], false)));
536 $result = [];
537 foreach ($members as $member) {
538 $result[] = substr($member, 1);
539 if ($withScores) $result[] = (string)$scores[$member];
540 }
541 return arrayReply($result);
542 }
543 private function keyCommand(string $command, array $args): array
544 {
545 if ($command === 'DEL' || $command === 'EXISTS') {
546 $count = 0;
547 foreach ($args as $key)
548 if ($this->entry($key) !== null) {
549 $count++;
550 if ($command === 'DEL') unset($this->database[':' . $key]);
551 }
552 return integerReply($count);
553 }
554 if ($command === 'KEYS') {
555 $pattern = parsePattern($args[0]);
556 $keys = [];
557 foreach ($this->database as $name => $entry) {
558 $key = substr($name, 1);
559 if (matchesPattern($pattern, $key)) $keys[] = $key;
560 }
561 sort($keys, SORT_STRING);
562 return arrayReply($keys);
563 }
564 if ($command === 'TICK') {
565 $amount = parseNumber($args[0]);
566 if ($amount <= 0) throw new CommandError(POSITIVE);
567 $this->time += $amount;
568 return integerReply($this->purge());
569 }
570 $key = $args[0];
571 $entry = $this->entry($key);
572 if ($command === 'TYPE') return [$entry?->value->kind->value ?? 'none'];
573 if ($command === 'TTL') return integerReply($entry === null ? -2 : ($entry->expiry === null ? -1 : $entry->expiry - $this->time));
574 if ($command === 'PERSIST') {
575 if ($entry === null || $entry->expiry === null) return integerReply(0);
576 $this->put($key, $entry->value);
577 return integerReply(1);
578 }
579 if ($command === 'RENAME' || $command === 'RENAMENX') {
580 if ($entry === null) throw new CommandError(NO_KEY);
581 $destination = $args[1];
582 if ($command === 'RENAMENX' && $this->entry($destination) !== null) return integerReply(0);
583 if ($key !== $destination) {
584 $this->database[':' . $destination] = $entry;
585 unset($this->database[':' . $key]);
586 }
587 return $command === 'RENAME' ? ['OK'] : integerReply(1);
588 }
589 $condition = count($args) === 3 ? strtoupper($args[2]) : '';
590 if (count($args) === 3 && !in_array($condition, ['NX', 'XX', 'GT', 'LT'], true))
591 throw new CommandError('ERR Unsupported option ' . $args[2]);
592 $seconds = parseNumber($args[1]);
593 if ($entry === null) return integerReply(0);
594 $expiry = $this->time + $seconds;
595 $old = $entry->expiry;
596 $accepted = match ($condition) {
597 'NX' => $old === null,
598 'XX' => $old !== null,
599 'GT' => $old !== null && $expiry > $old,
600 'LT' => $old === null || $expiry < $old,
601 default => true,
602 };
603 if (!$accepted) return integerReply(0);
604 if ($seconds <= 0) unset($this->database[':' . $key]);
605 else $this->put($key, $entry->value, $expiry);
606 return integerReply(1);
607 }
608 private function controlCommand(string $command, array $args): array
609 {
610 if ($command === 'TIME') return integerReply($this->time);
611 if ($command === 'DBSIZE') return integerReply(count($this->database));
612 if ($command === 'FLUSH') { $this->database = []; return ['OK']; }
613 if ($command === 'INFO') {
614 $counts = array_fill_keys(['string', 'list', 'hash', 'set', 'zset'], 0);
615 $expires = 0;
616 foreach ($this->database as $entry) {
617 $counts[$entry->value->kind->value]++;
618 if ($entry->expiry !== null) $expires++;
619 }
620 return ['keys=' . count($this->database) . ',expires=' . $expires . ',string=' . $counts['string']
621 . ',list=' . $counts['list'] . ',hash=' . $counts['hash'] . ',set=' . $counts['set'] . ',zset=' . $counts['zset']
622 . ',time=' . $this->time . ',depth=' . count($this->transactions) . ',snapshots=' . count($this->snapshots)];
623 }
624 if ($command === 'BEGIN') { $this->transactions[] = $this->database; return ['OK']; }
625 if ($command === 'COMMIT' || $command === 'ROLLBACK') {
626 if ($this->transactions === []) throw new CommandError('ERR no transaction');
627 $saved = array_pop($this->transactions);
628 if ($command === 'ROLLBACK') { $this->database = $saved; $this->purge(); }
629 return ['OK'];
630 }
631 $name = ':' . $args[0];
632 if ($command === 'SNAPSHOT') { $this->snapshots[$name] = $this->database; return ['OK']; }
633 if (!isset($this->snapshots[$name])) throw new CommandError('ERR no such snapshot');
634 $this->database = $this->snapshots[$name];
635 $this->purge();
636 return ['OK'];
637 }
638}
