3const messages = {
4 wrongType: 'WRONGTYPE Operation against a key holding the wrong kind of value',
5 notInteger: 'ERR value is not an integer or out of range',
6 overflow: 'ERR increment or decrement would overflow',
7 syntax: 'ERR syntax error',

11class CommandError extends Error {}
12function fail(message) { throw new CommandError(message); }
13const integerNotation = /^(?:0|-?[1-9][0-9]{0,8})$/;
14function number(value, message = messages.notInteger) {
15 if (!integerNotation.test(value)) fail(message);
16 return Number(value);
17}
18function increment(value, delta) {
19 const result = value + delta;
20 if (Math.abs(result) > 999999999) fail(messages.overflow);
21 return result;
22}
23const status = value => [value];
24const integer = value => [`(integer) ${value}`];
25const string = value => [value === null ? '(nil)' : JSON.stringify(value)];
26function array(values) {
27 if (values.length === 0) return ['(empty array)'];
28 const width = String(values.length).length;
29 return values.map((value, index) => `${String(index + 1).padStart(width)}) ${string(value)[0]}`);
30}
31const compare = (left, right) => left < right ? -1 : left > right ? 1 : 0;
32function range(sequence, start, stop) {
33 if (start < 0) start += sequence.length;
34 if (stop < 0) stop += sequence.length;
35 start = Math.max(0, start); stop = Math.min(sequence.length - 1, stop);
36 return start <= stop ? sequence.slice(start, stop + 1) : sequence.slice(0, 0);
37}

39function splitArguments(line) {
40 const argumentsList = [];
41 let position = 0;
42 while (position < line.length) {
43 if (line[position] === ' ') { ++position; continue; }
44 if (line[position] !== '"') {
45 const start = position;
46 while (position < line.length && line[position] !== ' ') ++position;
47 argumentsList.push(line.slice(start, position));
48 continue;
49 }
50 ++position;
51 let value = '', closed = false;
52 while (position < line.length) {
53 const character = line[position++];
54 if (character === '"') { closed = true; break; }
55 if (character === '\\') {
56 if (position === line.length || !['"', '\\'].includes(line[position])) return null;
57 value += line[position++];
58 } else value += character;
59 }
60 if (!closed || (position < line.length && line[position] !== ' ')) return null;
61 argumentsList.push(value);
62 }
63 return argumentsList;
64}

66function parsePattern(pattern) {
67 const parts = [];
68 for (let position = 0; position < pattern.length; ++position) {
69 const character = pattern[position];
70 if (character === '\\') {
71 if (++position === pattern.length) fail('ERR invalid pattern');
72 parts.push({ kind: 'literal', value: pattern[position] });
73 } else if (character === '[') {
74 const end = pattern.indexOf(']', position + 1);
75 if (end < 0) fail('ERR invalid pattern');
76 let content = pattern.slice(position + 1, end);
77 const negated = content.startsWith('^');
78 if (negated) content = content.slice(1);
79 if (content.length === 0) fail('ERR invalid pattern');
80 const characters = new Set();
81 for (let index = 0; index < content.length;) {
82 if (index + 2 < content.length && content[index + 1] === '-') {
83 const first = content.charCodeAt(index), last = content.charCodeAt(index + 2);
84 for (let code = Math.min(first, last); code <= Math.max(first, last); ++code) characters.add(String.fromCharCode(code));
85 index += 3;
86 } else characters.add(content[index++]);
87 }
88 parts.push({ kind: 'class', characters, negated }); position = end;
89 } else if (character === '*') parts.push({ kind: 'star' });
90 else if (character === '?') parts.push({ kind: 'any' });
91 else parts.push({ kind: 'literal', value: character });
92 }
93 return parts;
94}
95function matches(parts, value) {
96 let previous = Array(value.length + 1).fill(false);
97 previous[0] = true;
98 for (const part of parts) {
99 const current = Array(value.length + 1).fill(false);
100 if (part.kind === 'star') {
101 current[0] = previous[0];
102 for (let index = 1; index <= value.length; ++index) current[index] = previous[index] || current[index - 1];
103 } else {
104 for (let index = 0; index < value.length; ++index) {
105 const accepts = part.kind === 'any' || (part.kind === 'literal' ? part.value === value[index]
106 : part.characters.has(value[index]) !== part.negated);
107 current[index + 1] = previous[index] && accepts;
108 }
109 }
110 previous = current;
111 }
112 return previous[value.length];
113}
114function scoreBound(word) {
115 if (word === '-inf' || word === '+inf') return { value: word === '-inf' ? -Infinity : Infinity, inclusive: true };
116 const inclusive = !word.startsWith('(');
117 return { value: number(inclusive ? word : word.slice(1), 'ERR min or max is not a float'), inclusive };
118}

160class Store {
161 constructor() {
162 this.database = new Map(); this.time = 0; this.stack = []; this.snapshots = new Map();
163 }
164 data(key, type, fallback) {
165 const entry = this.database.get(key);
166 if (entry !== undefined && entry.type !== type) fail(messages.wrongType);
167 return entry === undefined ? fallback : entry.value;
168 }
169 write(key, type, value, expiry = undefined) {
170 const empty = type !== 'string' && (type === 'list' ? value.length : value.size) === 0;
171 if (empty) this.database.delete(key);
172 else this.database.set(key, { type, value, expiry: expiry === undefined ? (this.database.get(key)?.expiry ?? null) : expiry });
173 }
174 purge() {
175 let removed = 0;
176 for (const [key, entry] of this.database) if (entry.expiry !== null && entry.expiry <= this.time) {
177 this.database.delete(key); ++removed;
178 }
179 return removed;
180 }
181 command(line) {
182 const words = splitArguments(line);
183 if (words === null) return status('Invalid argument(s)');
184 const verb = words[0].toUpperCase(), args = words.slice(1), specification = commands.get(verb);
185 try {
186 if (specification === undefined) fail(`ERR unknown command '${words[0]}'`);
187 const { minimum, maximum, parity, group } = specification;
188 if (args.length < minimum || args.length > maximum || (parity !== null && args.length % 2 !== parity)) {
189 fail(`ERR wrong number of arguments for '${verb.toLowerCase()}' command`);
190 }
191 return this[group](verb, args);
192 } catch (error) {
193 if (!(error instanceof CommandError)) throw error;
194 return [`(error) ${error.message}`];
195 }
196 }
197 strings(verb, args) {
198 const key = args[0];
199 switch (verb) {
200 case 'SET': {
201 const options = new Map();
202 for (let index = 2; index < args.length; ++index) {
203 const option = args[index].toUpperCase();
204 if (!['NX', 'XX', 'EX', 'KEEPTTL', 'GET'].includes(option) || options.has(option)) fail(messages.syntax);
205 if (option === 'EX') {
206 if (++index === args.length) fail(messages.syntax);
207 options.set(option, args[index]);
208 } else options.set(option, true);
209 }
210 if ((options.has('NX') && options.has('XX')) || (options.has('EX') && options.has('KEEPTTL'))) fail(messages.syntax);
211 let expiry = null;
212 if (options.has('EX')) {
213 const seconds = number(options.get('EX'));
214 if (seconds <= 0) fail("ERR invalid expire time in 'set' command");
215 expiry = this.time + seconds;
216 }
217 const previous = this.database.get(key);
218 if (options.has('GET')) this.data(key, 'string', null);
219 const changed = !(options.has('NX') && previous !== undefined) && !(options.has('XX') && previous === undefined);
220 if (options.has('KEEPTTL') && previous !== undefined) expiry = previous.expiry;
221 if (changed) this.write(key, 'string', args[1], expiry);
222 return options.has('GET') ? string(previous?.value ?? null) : changed ? status('OK') : string(null);
223 }
224 case 'GET': return string(this.data(key, 'string', null));
225 case 'MSET':
226 for (let index = 0; index < args.length; index += 2) this.write(args[index], 'string', args[index + 1], null);
227 return status('OK');
228 case 'MGET':
229 return array(args.map(key => { const entry = this.database.get(key); return entry?.type === 'string' ? entry.value : null; }));
230 case 'INCR': case 'DECR': case 'INCRBY': case 'DECRBY': {
231 let delta = verb === 'INCR' ? 1 : verb === 'DECR' ? -1 : number(args[1]);
232 if (verb === 'DECRBY') delta = -delta;
233 const result = increment(number(this.data(key, 'string', '0')), delta);
234 this.write(key, 'string', String(result));
235 return integer(result);
236 }
237 case 'APPEND': {
238 const value = this.data(key, 'string', '') + args[1];
239 this.write(key, 'string', value);
240 return integer(value.length);
241 }
242 case 'STRLEN': return integer(this.data(key, 'string', '').length);
243 case 'GETRANGE': {
244 const start = number(args[1]), stop = number(args[2]);
245 return string(range(this.data(key, 'string', ''), start, stop));
246 }
247 }
248 }
249 lists(verb, args) {
250 const key = args[0];
251 let first, second;
252 if (verb === 'LPOP' || verb === 'RPOP') {
253 first = args.length === 2 ? number(args[1], messages.positive) : 1;
254 if (first < 0) fail(messages.positive);
255 } else if (['LRANGE', 'LINDEX', 'LSET', 'LREM', 'LTRIM'].includes(verb)) {
256 first = number(args[1]);
257 if (verb === 'LRANGE' || verb === 'LTRIM') second = number(args[2]);
258 } else if (verb === 'LINSERT' && !['BEFORE', 'AFTER'].includes(args[1].toUpperCase())) fail(messages.syntax);
259 const list = this.data(key, 'list', []);
260 switch (verb) {
261 case 'LPUSH': case 'RPUSH':
262 for (const element of args.slice(1)) {
263 if (verb === 'LPUSH') list.unshift(element); else list.push(element);
264 }
265 this.write(key, 'list', list);
266 return integer(list.length);
267 case 'LPOP': case 'RPOP': {
268 if (!this.database.has(key)) return string(null);
269 const removed = [];
270 while (removed.length < first && list.length > 0) removed.push(verb === 'LPOP' ? list.shift() : list.pop());
271 this.write(key, 'list', list);
272 return args.length === 2 ? array(removed) : string(removed[0]);
273 }
274 case 'LRANGE': return array(range(list, first, second));
275 case 'LLEN': return integer(list.length);
276 case 'LINDEX': case 'LSET': {
277 if (verb === 'LSET' && !this.database.has(key)) fail(messages.noKey);
278 const index = first < 0 ? first + list.length : first;
279 if (verb === 'LINDEX') return string(index >= 0 && index < list.length ? list[index] : null);
280 if (index < 0 || index >= list.length) fail('ERR index out of range');
281 list[index] = args[2]; this.write(key, 'list', list);
282 return status('OK');
283 }
284 case 'LREM': {
285 let removed = 0;
286 const ordered = first < 0 ? [...list].reverse() : list;
287 const remaining = ordered.filter(element => {
288 if (element === args[2] && (first === 0 || removed < Math.abs(first))) { ++removed; return false; }
289 return true;
290 });
291 if (first < 0) remaining.reverse();
292 this.write(key, 'list', remaining);
293 return integer(removed);
294 }
295 case 'LTRIM': this.write(key, 'list', range(list, first, second)); return status('OK');
296 case 'LINSERT': {
297 if (!this.database.has(key)) return integer(0);
298 const index = list.indexOf(args[2]);
299 if (index < 0) return integer(-1);
300 list.splice(index + (args[1].toUpperCase() === 'AFTER' ? 1 : 0), 0, args[3]);
301 this.write(key, 'list', list);
302 return integer(list.length);
303 }
304 }
305 }
306 hashes(verb, args) {
307 const key = args[0], delta = verb === 'HINCRBY' ? number(args[2]) : 0;
308 const hash = this.data(key, 'hash', new Map());
309 switch (verb) {
310 case 'HSET': {
311 let added = 0;
312 for (let index = 1; index < args.length; index += 2) {
313 if (!hash.has(args[index])) ++added;
314 hash.set(args[index], args[index + 1]);
315 }
316 this.write(key, 'hash', hash);
317 return integer(added);
318 }
319 case 'HGET': return string(hash.get(args[1]) ?? null);
320 case 'HDEL': {
321 let removed = 0;
322 for (const field of args.slice(1)) if (hash.delete(field)) ++removed;
323 this.write(key, 'hash', hash);
324 return integer(removed);
325 }
326 case 'HLEN': return integer(hash.size);
327 case 'HEXISTS': return integer(hash.has(args[1]) ? 1 : 0);
328 case 'HINCRBY': {
329 const result = increment(number(hash.get(args[1]) ?? '0', 'ERR hash value is not an integer'), delta);
330 hash.set(args[1], String(result)); this.write(key, 'hash', hash);
331 return integer(result);
332 }
333 case 'HGETALL': case 'HKEYS': case 'HVALS': {
334 const fields = [...hash.keys()].sort();
335 return array(verb === 'HKEYS' ? fields : verb === 'HVALS' ? fields.map(field => hash.get(field))
336 : fields.flatMap(field => [field, hash.get(field)]));
337 }
338 }
339 }
340 sets(verb, args) {
341 const storing = verb.endsWith('STORE'), operation = storing ? verb.slice(0, -5) : verb;
342 if (['SINTER', 'SUNION', 'SDIFF'].includes(operation)) {
343 const sources = (storing ? args.slice(1) : args).map(key => this.data(key, 'set', new Set()));
344 const result = new Set(sources[0]);
345 for (const source of sources.slice(1)) {
346 if (operation === 'SUNION') for (const member of source) result.add(member);
347 else if (operation === 'SINTER') { for (const member of result) if (!source.has(member)) result.delete(member); }
348 else for (const member of source) result.delete(member);
349 }
350 if (storing) { this.write(args[0], 'set', result, null); return integer(result.size); }
351 return array([...result].sort());
352 }
353 const key = args[0], members = this.data(key, 'set', new Set());
354 switch (verb) {
355 case 'SADD': {
356 let added = 0;
357 for (const member of args.slice(1)) if (!members.has(member)) { members.add(member); ++added; }
358 this.write(key, 'set', members);
359 return integer(added);
360 }
361 case 'SREM': {
362 let removed = 0;
363 for (const member of args.slice(1)) if (members.delete(member)) ++removed;
364 this.write(key, 'set', members);
365 return integer(removed);
366 }
367 case 'SISMEMBER': return integer(members.has(args[1]) ? 1 : 0);
368 case 'SMEMBERS': return array([...members].sort());
369 case 'SCARD': return integer(members.size);
370 }
371 }
372 sortedSets(verb, args) {
373 const key = args[0], scores = [];
374 let delta, low, high;
375 if (verb === 'ZADD') for (let index = 1; index < args.length; index += 2) scores.push(number(args[index]));
376 if (verb === 'ZINCRBY') delta = number(args[1]);
377 if (verb === 'ZRANGE' || verb === 'ZRANGEBYSCORE') {
378 if (args.length === 4 && args[3].toUpperCase() !== 'WITHSCORES') fail(messages.syntax);
379 if (verb === 'ZRANGE') { low = number(args[1]); high = number(args[2]); }
380 else { low = scoreBound(args[1]); high = scoreBound(args[2]); }
381 }
382 const members = this.data(key, 'zset', new Map());
383 switch (verb) {
384 case 'ZADD': {
385 let added = 0;
386 for (let index = 0; index < scores.length; ++index) {
387 const member = args[2 * index + 2];
388 if (!members.has(member)) ++added;
389 members.set(member, scores[index]);
390 }
391 this.write(key, 'zset', members);
392 return integer(added);
393 }
394 case 'ZSCORE': return string(members.has(args[1]) ? String(members.get(args[1])) : null);
395 case 'ZINCRBY': {
396 const score = increment(members.get(args[2]) ?? 0, delta);
397 members.set(args[2], score); this.write(key, 'zset', members);
398 return string(String(score));
399 }
400 case 'ZREM': {
401 let removed = 0;
402 for (const member of args.slice(1)) if (members.delete(member)) ++removed;
403 this.write(key, 'zset', members);
404 return integer(removed);
405 }
406 case 'ZCARD': return integer(members.size);
407 case 'ZRANK': case 'ZRANGE': case 'ZRANGEBYSCORE': {
408 const ordered = [...members].sort((left, right) => left[1] - right[1] || compare(left[0], right[0]));
409 if (verb === 'ZRANK') {
410 const rank = ordered.findIndex(([member]) => member === args[1]);
411 return rank < 0 ? string(null) : integer(rank);
412 }
413 const selected = verb === 'ZRANGE' ? range(ordered, low, high) : ordered.filter(([, score]) =>
414 (score > low.value || (score === low.value && low.inclusive)) && (score < high.value || (score === high.value && high.inclusive)));
415 return array(args.length === 4 ? selected.flatMap(([member, score]) => [member, String(score)]) : selected.map(([member]) => member));
416 }
417 }
418 }
419 keys(verb, args) {
420 const key = args[0];
421 switch (verb) {
422 case 'DEL': {
423 let removed = 0;
424 for (const key of args) if (this.database.delete(key)) ++removed;
425 return integer(removed);
426 }
427 case 'EXISTS': return integer(args.filter(key => this.database.has(key)).length);
428 case 'TYPE': return status(this.database.get(key)?.type ?? 'none');
429 case 'RENAME': case 'RENAMENX': {
430 const entry = this.database.get(key);
431 if (entry === undefined) fail(messages.noKey);
432 if (verb === 'RENAMENX' && this.database.has(args[1])) return integer(0);
433 if (key !== args[1]) { this.database.set(args[1], entry); this.database.delete(key); }
434 return verb === 'RENAMENX' ? integer(1) : status('OK');
435 }
436 case 'KEYS': {
437 const parts = parsePattern(key);
438 return array([...this.database.keys()].filter(key => matches(parts, key)).sort());
439 }
440 case 'EXPIRE': {
441 const condition = args.length === 3 ? args[2].toUpperCase() : null;
442 if (condition !== null && !['NX', 'XX', 'GT', 'LT'].includes(condition)) fail(`ERR Unsupported option ${args[2]}`);
443 const seconds = number(args[1]), entry = this.database.get(key);
444 if (entry === undefined) return integer(0);
445 const expiry = this.time + seconds, previous = entry.expiry;
446 if ((condition === 'NX' && previous !== null) || (condition === 'XX' && previous === null)
447 || (condition === 'GT' && (previous === null || expiry <= previous))
448 || (condition === 'LT' && previous !== null && expiry >= previous)) return integer(0);
449 if (seconds <= 0) this.database.delete(key); else entry.expiry = expiry;
450 return integer(1);
451 }
452 case 'TTL': {
453 const entry = this.database.get(key);
454 return integer(entry === undefined ? -2 : entry.expiry === null ? -1 : entry.expiry - this.time);
455 }
456 case 'PERSIST': {
457 const entry = this.database.get(key);
458 if (entry === undefined || entry.expiry === null) return integer(0);
459 entry.expiry = null;
460 return integer(1);
461 }
462 case 'TICK': {
463 const ticks = number(key);
464 if (ticks <= 0) fail(messages.positive);
465 this.time += ticks;
466 return integer(this.purge());
467 }
468 case 'TIME': return integer(this.time);
469 case 'DBSIZE': return integer(this.database.size);
470 case 'FLUSH': this.database.clear(); return status('OK');
471 case 'INFO': {
472 const counts = { string: 0, list: 0, hash: 0, set: 0, zset: 0 };
473 let expires = 0;
474 for (const entry of this.database.values()) { ++counts[entry.type]; if (entry.expiry !== null) ++expires; }
475 return status(`keys=${this.database.size},expires=${expires},string=${counts.string},list=${counts.list},hash=${counts.hash},set=${counts.set},zset=${counts.zset},time=${this.time},depth=${this.stack.length},snapshots=${this.snapshots.size}`);
476 }
477 }
478 }
479 control(verb, args) {
480 switch (verb) {
481 case 'BEGIN': this.stack.push(structuredClone(this.database)); break;
482 case 'COMMIT': case 'ROLLBACK':
483 if (this.stack.length === 0) fail('ERR no transaction');
484 if (verb === 'ROLLBACK') { this.database = this.stack.pop(); this.purge(); }
485 else this.stack.pop();
486 break;
487 case 'SNAPSHOT': this.snapshots.set(args[0], structuredClone(this.database)); break;
488 case 'RESTORE':
489 if (!this.snapshots.has(args[0])) fail('ERR no such snapshot');
490 this.database = structuredClone(this.snapshots.get(args[0])); this.purge();
491 break;
492 }
493 return status('OK');
494 }
