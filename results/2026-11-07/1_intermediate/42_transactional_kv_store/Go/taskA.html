68func readValue[T value](db database, key string, empty T) (T, error) {
69 item, exists := db[key]
70 if !exists {
71 return empty, nil
72 }
73 typed, ok := item.value.(T)
74 if !ok {
75 return empty, errWrongType
76 }
77 return typed.clone().(T), nil
78}

89func parseInteger(text string) (int, bool) {
90 if text == "0" {
91 return 0, true
92 }
93 start := 0
94 if strings.HasPrefix(text, "-") {
95 start = 1
96 }
97 if len(text)-start < 1 || len(text)-start > 9 || text[start] < '1' || text[start] > '9' {
98 return 0, false
99 }
100 number := 0
101 for i := start; i < len(text); i++ {
102 if text[i] < '0' || text[i] > '9' {
103 return 0, false
104 }
105 number = number*10 + int(text[i]-'0')
106 }
107 if start == 1 {
108 number = -number
109 }
110 return number, true
111}
112func integer(text string) (int, error) {
113 number, ok := parseInteger(text)
114 if !ok {
115 return 0, errNotInt
116 }
117 return number, nil
118}
119func increment(old, delta int) (int, error) {
120 result := old + delta
121 if result < -999999999 || result > 999999999 {
122 return 0, errOverflow
123 }
124 return result, nil
125}
126func normalize(length, start, stop int) (int, int) {
127 if start < 0 {
128 start += length
129 }
130 if stop < 0 {
131 stop += length
132 }
133 start = max(0, start)
134 stop = min(length-1, stop)
135 if start > stop {
136 return 0, 0
137 }
138 return start, stop + 1
139}
140func indices(a, b string) (int, int, error) {
141 start, err := integer(a)
142 if err != nil {
143 return 0, 0, err
144 }
145 stop, err := integer(b)
146 return start, stop, err
147}
148func splitLine(line string) ([]string, bool) {
149 var words []string
150 for position := 0; position < len(line); {
151 if line[position] == ' ' {
152 position++
153 continue
154 }
155 if line[position] != '"' {
156 start := position
157 for position < len(line) && line[position] != ' ' {
158 position++
159 }
160 words = append(words, line[start:position])
161 continue
162 }
163 position++
164 var text strings.Builder
165 for position < len(line) && line[position] != '"' {
166 if line[position] == '\\' {
167 position++
168 if position == len(line) || (line[position] != '\\' && line[position] != '"') {
169 return nil, false
170 }
171 }
172 text.WriteByte(line[position])
173 position++
174 }
175 if position == len(line) {
176 return nil, false
177 }
178 position++
179 if position < len(line) && line[position] != ' ' {
180 return nil, false
181 }
182 words = append(words, text.String())
183 }
184 return words, true
185}

194func integerReply(number int) string { return fmt.Sprintf("(integer) %d", number) }
195func textReply(text string) string { return strconv.Quote(text) }
196func arrayReply(items []optionalText) string {
197 if len(items) == 0 {
198 return "(empty array)"
199 }
200 width := len(strconv.Itoa(len(items)))
201 lines := make([]string, len(items))
202 for i, item := range items {
203 text := nilReply
204 if item.present {
205 text = textReply(item.text)
206 }
207 lines[i] = fmt.Sprintf("%*d) %s", width, i+1, text)
208 }
209 return strings.Join(lines, "\n")
210}
211func stringArray(items []string) string {
212 values := make([]optionalText, len(items))
213 for i, text := range items {
214 values[i] = optionalText{text, true}
215 }
216 return arrayReply(values)
217}

220type partKind int

222const (
223 literalPart partKind = iota
224 anyPart
225 starPart
226 classPart

229type patternPart struct {
230 kind partKind
231 literal byte
232 characters [128]bool
233 negated bool
234}
235
236func parsePattern(text string) ([]patternPart, error) {
237 var parts []patternPart
238 for i := 0; i < len(text); i++ {
239 part := patternPart{}
240 switch text[i] {
241 case '\\':
242 i++
243 if i == len(text) {
244 return nil, errors.New("ERR invalid pattern")
245 }
246 part.literal = text[i]
247 case '*':
248 part.kind = starPart
249 case '?':
250 part.kind = anyPart
251 case '[':
252 end := i + 1
253 for end < len(text) && text[end] != ']' {
254 end++
255 }
256 if end == len(text) {
257 return nil, errors.New("ERR invalid pattern")
258 }
259 part.kind = classPart
260 i++
261 if i < end && text[i] == '^' {
262 part.negated = true
263 i++
264 }
265 if i == end {
266 return nil, errors.New("ERR invalid pattern")
267 }
268 for i < end {
269 first, last := int(text[i]), int(text[i])
270 if i+2 < end && text[i+1] == '-' {
271 last = int(text[i+2])
272 i += 3
273 } else {
274 i++
275 }
276 for c := min(first, last); c <= max(first, last); c++ {
277 part.characters[c] = true
278 }
279 }
280 default:
281 part.literal = text[i]
282 }
283 parts = append(parts, part)
284 }
285 return parts, nil
286}
287func matchesPattern(parts []patternPart, text string) bool {
288 previous := make([]bool, len(text)+1)
289 previous[0] = true
290 for _, part := range parts {
291 next := make([]bool, len(previous))
292 if part.kind == starPart {
293 next[0] = previous[0]
294 for i := 1; i < len(next); i++ {
295 next[i] = previous[i] || next[i-1]
296 }
297 } else {
298 for i := range len(text) {
299 accepts := part.kind == anyPart || (part.kind == literalPart && part.literal == text[i]) ||
300 (part.kind == classPart && part.characters[text[i]] != part.negated)
301 next[i+1] = previous[i] && accepts
302 }
303 }
304 previous = next
305 }
306 return previous[len(text)]
307}

309type bound struct {
310 infinity int
311 number int
312 exclusive bool
313}
314
315func scoreBound(text string) (bound, error) {
316 if text == "-inf" {
317 return bound{infinity: -1}, nil
318 }
319 if text == "+inf" {
320 return bound{infinity: 1}, nil
321 }
322 exclusive := strings.HasPrefix(text, "(")
323 if exclusive {
324 text = text[1:]
325 }
326 number, ok := parseInteger(text)
327 if !ok {
328 return bound{}, errors.New("ERR min or max is not a float")
329 }
330 return bound{number: number, exclusive: exclusive}, nil
331}
332func (b bound) accepts(score int, lower bool) bool {
333 if b.infinity != 0 {
334 return (b.infinity < 0) == lower
335 }
336 if lower {
337 return score > b.number || (!b.exclusive && score == b.number)
338 }
339 return score < b.number || (!b.exclusive && score == b.number)
340}

413func (s *store) replace(key string, data value, expiry expiration) {
414 if data.empty() {
415 delete(s.database, key)
416 } else {
417 s.database[key] = entry{data, expiry}
418 }
419}
420func (s *store) update(key string, data value) { s.replace(key, data, s.database[key].expiry) }
421func (s *store) purge() int {
422 removed := 0
423 for key, item := range s.database {
424 if item.expiry.set && item.expiry.at <= s.time {
425 delete(s.database, key)
426 removed++
427 }
428 }
429 return removed
430}
431func (s *store) textCommand(command string, a []string) (string, error) {
432 key := a[0]
433 if command == "SET" {
434 options := make(map[string]bool)
435 secondsText := ""
436 for i := 2; i < len(a); i++ {
437 option := strings.ToUpper(a[i])
438 if !slices.Contains([]string{"NX", "XX", "EX", "KEEPTTL", "GET"}, option) || options[option] {
439 return "", errSyntax
440 }
441 options[option] = true
442 if option == "EX" {
443 i++
444 if i == len(a) {
445 return "", errSyntax
446 }
447 secondsText = a[i]
448 }
449 }
450 if (options["NX"] && options["XX"]) || (options["EX"] && options["KEEPTTL"]) {
451 return "", errSyntax
452 }
453 expiry := expiration{}
454 if options["EX"] {
455 seconds, err := integer(secondsText)
456 if err != nil {
457 return "", err
458 }
459 if seconds <= 0 {
460 return "", errors.New("ERR invalid expire time in 'set' command")
461 }
462 expiry = expiration{s.time + seconds, true}
463 }
464 old, exists := s.database[key]
465 previous := nilReply
466 if options["GET"] {
467 text, err := readValue(s.database, key, textValue(""))
468 if err != nil {
469 return "", err
470 }
471 if exists {
472 previous = textReply(string(text))
473 }
474 }
475 if options["KEEPTTL"] && exists {
476 expiry = old.expiry
477 }
478 changed := !(options["NX"] && exists) && !(options["XX"] && !exists)
479 if changed {
480 s.replace(key, textValue(a[1]), expiry)
481 }
482 if options["GET"] {
483 return previous, nil
484 }
485 if changed {
486 return "OK", nil
487 }
488 return nilReply, nil
489 }
490 if command == "MSET" {
491 for i := 0; i < len(a); i += 2 {
492 s.replace(a[i], textValue(a[i+1]), expiration{})
493 }
494 return "OK", nil
495 }
496 if command == "MGET" {
497 items := make([]optionalText, len(a))
498 for i, key := range a {
499 if text, ok := s.database[key].value.(textValue); ok {
500 items[i] = optionalText{string(text), true}
501 }
502 }
503 return arrayReply(items), nil
504 }
505 if command == "INCR" || command == "DECR" || command == "INCRBY" || command == "DECRBY" {
506 delta := 1
507 var err error
508 if len(a) == 2 {
509 delta, err = integer(a[1])
510 if err != nil {
511 return "", err
512 }
513 }
514 if command == "DECR" || command == "DECRBY" {
515 delta = -delta
516 }
517 text, err := readValue(s.database, key, textValue(""))
518 if err != nil {
519 return "", err
520 }
521 old := 0
522 if _, exists := s.database[key]; exists {
523 old, err = integer(string(text))
524 if err != nil {
525 return "", err
526 }
527 }
528 result, err := increment(old, delta)
529 if err != nil {
530 return "", err
531 }
532 s.update(key, textValue(strconv.Itoa(result)))
533 return integerReply(result), nil
534 }
535 if command == "GETRANGE" {
536 start, stop, err := indices(a[1], a[2])
537 if err != nil {
538 return "", err
539 }
540 text, err := readValue(s.database, key, textValue(""))
541 if err != nil {
542 return "", err
543 }
544 first, end := normalize(len(text), start, stop)
545 return textReply(string(text[first:end])), nil
546 }
547 text, err := readValue(s.database, key, textValue(""))
548 if err != nil {
549 return "", err
550 }
551 if command == "GET" {
552 if _, exists := s.database[key]; !exists {
553 return nilReply, nil
554 }
555 return textReply(string(text)), nil
556 }
557 if command == "STRLEN" {
558 return integerReply(len(text)), nil
559 }
560 text += textValue(a[1])
561 s.update(key, text)
562 return integerReply(len(text)), nil
563}

565func (s *store) listCommand(command string, a []string) (string, error) {
566 key := a[0]
567 switch command {
568 case "LPOP", "RPOP":
569 count := 1
570 if len(a) == 2 {
571 var ok bool
572 count, ok = parseInteger(a[1])
573 if !ok || count < 0 {
574 return "", errPositive
575 }
576 }
577 list, err := readValue(s.database, key, listValue{})
578 if err != nil {
579 return "", err
580 }
581 if _, exists := s.database[key]; !exists {
582 return nilReply, nil
583 }
584 var removed []string
585 count = min(count, len(list))
586 for range count {
587 index := 0
588 if command == "RPOP" {
589 index = len(list) - 1
590 }
591 removed = append(removed, list[index])
592 list = slices.Delete(list, index, index+1)
593 }
594 s.update(key, list)
595 if len(a) == 1 {
596 return textReply(removed[0]), nil
597 }
598 return stringArray(removed), nil
599 case "LRANGE", "LTRIM":
600 start, stop, err := indices(a[1], a[2])
601 if err != nil {
602 return "", err
603 }
604 list, err := readValue(s.database, key, listValue{})
605 if err != nil {
606 return "", err
607 }
608 first, end := normalize(len(list), start, stop)
609 kept := slices.Clone(list[first:end])
610 if command == "LRANGE" {
611 return stringArray(kept), nil
612 }
613 s.update(key, kept)
614 return "OK", nil
615 case "LINDEX", "LSET":
616 index, err := integer(a[1])
617 if err != nil {
618 return "", err
619 }
620 list, err := readValue(s.database, key, listValue{})
621 if err != nil {
622 return "", err
623 }
624 if command == "LSET" {
625 if _, exists := s.database[key]; !exists {
626 return "", errNoKey
627 }
628 }
629 if index < 0 {
630 index += len(list)
631 }
632 inside := index >= 0 && index < len(list)
633 if command == "LINDEX" {
634 if !inside {
635 return nilReply, nil
636 }
637 return textReply(list[index]), nil
638 }
639 if !inside {
640 return "", errors.New("ERR index out of range")
641 }
642 list[index] = a[2]
643 s.update(key, list)
644 return "OK", nil
645 case "LREM":
646 count, err := integer(a[1])
647 if err != nil {
648 return "", err
649 }
650 list, err := readValue(s.database, key, listValue{})
651 if err != nil {
652 return "", err
653 }
654 limit := count
655 if count < 0 {
656 limit = -count
657 slices.Reverse(list)
658 }
659 if count == 0 {
660 limit = len(list)
661 }
662 kept, removed := list[:0], 0
663 for _, element := range list {
664 if element == a[2] && removed < limit {
665 removed++
666 } else {
667 kept = append(kept, element)
668 }
669 }
670 if count < 0 {
671 slices.Reverse(kept)
672 }
673 s.update(key, kept)
674 return integerReply(removed), nil
675 case "LINSERT":
676 where := strings.ToUpper(a[1])
677 if where != "BEFORE" && where != "AFTER" {
678 return "", errSyntax
679 }
680 list, err := readValue(s.database, key, listValue{})
681 if err != nil {
682 return "", err
683 }
684 if _, exists := s.database[key]; !exists {
685 return integerReply(0), nil
686 }
687 index := slices.Index(list, a[2])
688 if index < 0 {
689 return integerReply(-1), nil
690 }
691 if where == "AFTER" {
692 index++
693 }
694 list = slices.Insert(list, index, a[3])
695 s.update(key, list)
696 return integerReply(len(list)), nil
697 }
698 list, err := readValue(s.database, key, listValue{})
699 if err != nil {
700 return "", err
701 }
702 if command == "LLEN" {
703 return integerReply(len(list)), nil
704 }
705 for _, element := range a[1:] {
706 if command == "LPUSH" {
707 list = slices.Insert(list, 0, element)
708 } else {
709 list = append(list, element)
710 }
711 }
712 s.update(key, list)
713 return integerReply(len(list)), nil
714}

716func (s *store) hashCommand(command string, a []string) (string, error) {
717 key, delta := a[0], 0
718 var err error
719 if command == "HINCRBY" {
720 delta, err = integer(a[2])
721 if err != nil {
722 return "", err
723 }
724 }
725 fields, err := readValue(s.database, key, hashValue{})
726 if err != nil {
727 return "", err
728 }
729 switch command {
730 case "HSET":
731 added := 0
732 for i := 1; i < len(a); i += 2 {
733 if _, exists := fields[a[i]]; !exists {
734 added++
735 }
736 fields[a[i]] = a[i+1]
737 }
738 s.update(key, fields)
739 return integerReply(added), nil
740 case "HGET":
741 if text, exists := fields[a[1]]; exists {
742 return textReply(text), nil
743 }
744 return nilReply, nil
745 case "HDEL":
746 removed := 0
747 for _, field := range a[1:] {
748 if _, exists := fields[field]; exists {
749 delete(fields, field)
750 removed++
751 }
752 }
753 s.update(key, fields)
754 return integerReply(removed), nil
755 case "HLEN":
756 return integerReply(len(fields)), nil
757 case "HEXISTS":
758 if _, exists := fields[a[1]]; exists {
759 return integerReply(1), nil
760 }
761 return integerReply(0), nil
762 case "HINCRBY":
763 old := 0
764 if text, exists := fields[a[1]]; exists {
765 var valid bool
766 old, valid = parseInteger(text)
767 if !valid {
768 return "", errors.New("ERR hash value is not an integer")
769 }
770 }
771 result, err := increment(old, delta)
772 if err != nil {
773 return "", err
774 }
775 fields[a[1]] = strconv.Itoa(result)
776 s.update(key, fields)
777 return integerReply(result), nil
778 }
779 var output []string
780 for _, field := range sortedKeys(fields) {
781 if command != "HVALS" {
782 output = append(output, field)
783 }
784 if command != "HKEYS" {
785 output = append(output, fields[field])
786 }
787 }
788 return stringArray(output), nil
789}

791func (s *store) setCommand(command string, a []string) (string, error) {
792 storing := strings.HasSuffix(command, "STORE")
793 operation := strings.TrimSuffix(command, "STORE")
794 if operation == "SINTER" || operation == "SUNION" || operation == "SDIFF" {
795 first := 0
796 if storing {
797 first = 1
798 }
799 var sources []setValue
800 for _, key := range a[first:] {
801 members, err := readValue(s.database, key, setValue{})
802 if err != nil {
803 return "", err
804 }
805 sources = append(sources, members)
806 }
807 result := sources[0]
808 for _, source := range sources[1:] {
809 if operation == "SUNION" {
810 maps.Copy(result, source)
811 } else {
812 for member := range result {
813 _, present := source[member]
814 if (operation == "SINTER") != present {
815 delete(result, member)
816 }
817 }
818 }
819 }
820 if storing {
821 s.replace(a[0], result, expiration{})
822 return integerReply(len(result)), nil
823 }
824 return stringArray(sortedKeys(result)), nil
825 }
826 members, err := readValue(s.database, a[0], setValue{})
827 if err != nil {
828 return "", err
829 }
830 switch command {
831 case "SCARD":
832 return integerReply(len(members)), nil
833 case "SMEMBERS":
834 return stringArray(sortedKeys(members)), nil
835 case "SISMEMBER":
836 if _, present := members[a[1]]; present {
837 return integerReply(1), nil
838 }
839 return integerReply(0), nil
840 }
841 changed := 0
842 for _, member := range a[1:] {
843 _, present := members[member]
844 if command == "SADD" && !present {
845 members[member] = struct{}{}
846 changed++
847 } else if command == "SREM" && present {
848 delete(members, member)
849 changed++
850 }
851 }
852 s.update(a[0], members)
853 return integerReply(changed), nil
854}

861func (s *store) zsetCommand(command string, a []string) (string, error) {
862 key := a[0]
863 if command == "ZADD" {
864 var scores []int
865 for i := 1; i < len(a); i += 2 {
866 score, err := integer(a[i])
867 if err != nil {
868 return "", err
869 }
870 scores = append(scores, score)
871 }
872 members, err := readValue(s.database, key, zsetValue{})
873 if err != nil {
874 return "", err
875 }
876 added := 0
877 for i, score := range scores {
878 member := a[2*i+2]
879 if _, present := members[member]; !present {
880 added++
881 }
882 members[member] = score
883 }
884 s.update(key, members)
885 return integerReply(added), nil
886 }
887 if command == "ZINCRBY" {
888 delta, err := integer(a[1])
889 if err != nil {
890 return "", err
891 }
892 members, err := readValue(s.database, key, zsetValue{})
893 if err != nil {
894 return "", err
895 }
896 result, err := increment(members[a[2]], delta)
897 if err != nil {
898 return "", err
899 }
900 members[a[2]] = result
901 s.update(key, members)
902 return textReply(strconv.Itoa(result)), nil
903 }
904 byIndex, byScore, withScores := command == "ZRANGE", command == "ZRANGEBYSCORE", len(a) == 4
905 start, stop := 0, 0
906 minimum, maximum := bound{infinity: -1}, bound{infinity: 1}
907 if byIndex || byScore {
908 if withScores && !strings.EqualFold(a[3], "WITHSCORES") {
909 return "", errSyntax
910 }
911 var err error
912 if byIndex {
913 start, stop, err = indices(a[1], a[2])
914 } else {
915 minimum, err = scoreBound(a[1])
916 if err == nil {
917 maximum, err = scoreBound(a[2])
918 }
919 }
920 if err != nil {
921 return "", err
922 }
923 }
924 members, err := readValue(s.database, key, zsetValue{})
925 if err != nil {
926 return "", err
927 }
928 switch command {
929 case "ZSCORE":
930 if score, exists := members[a[1]]; exists {
931 return textReply(strconv.Itoa(score)), nil
932 }
933 return nilReply, nil
934 case "ZREM":
935 removed := 0
936 for _, member := range a[1:] {
937 if _, exists := members[member]; exists {
938 delete(members, member)
939 removed++
940 }
941 }
942 s.update(key, members)
943 return integerReply(removed), nil
944 case "ZCARD":
945 return integerReply(len(members)), nil
946 }
947 var ordered []scoredMember
948 for member, score := range members {
949 ordered = append(ordered, scoredMember{member, score})
950 }
951 slices.SortFunc(ordered, func(a, b scoredMember) int {
952 if a.score != b.score {
953 return a.score - b.score
954 }
955 return strings.Compare(a.member, b.member)
956 })
957 if command == "ZRANK" {
958 for i, item := range ordered {
959 if item.member == a[1] {
960 return integerReply(i), nil
961 }
962 }
963 return nilReply, nil
964 }
965 if byIndex {
966 first, end := normalize(len(ordered), start, stop)
967 ordered = ordered[first:end]
968 }
969 var output []string
970 for _, item := range ordered {
971 if byScore && (!minimum.accepts(item.score, true) || !maximum.accepts(item.score, false)) {
972 continue
973 }
974 output = append(output, item.member)
975 if withScores {
976 output = append(output, strconv.Itoa(item.score))
977 }
978 }
979 return stringArray(output), nil
980}

982func (s *store) keyCommand(command string, a []string) (string, error) {
983 switch command {
984 case "DEL", "EXISTS":
985 count := 0
986 for _, key := range a {
987 if _, exists := s.database[key]; exists {
988 count++
989 if command == "DEL" {
990 delete(s.database, key)
991 }
992 }
993 }
994 return integerReply(count), nil
995 case "KEYS":
996 pattern, err := parsePattern(a[0])
997 if err != nil {
998 return "", err
999 }
1000 var result []string
1001 for _, key := range sortedKeys(s.database) {
1002 if matchesPattern(pattern, key) {
1003 result = append(result, key)
1004 }
1005 }
1006 return stringArray(result), nil
1007 case "EXPIRE":
1008 condition := ""
1009 if len(a) == 3 {
1010 condition = strings.ToUpper(a[2])
1011 if !slices.Contains([]string{"NX", "XX", "GT", "LT"}, condition) {
1012 return "", fmt.Errorf("ERR Unsupported option %s", a[2])
1013 }
1014 }
1015 seconds, err := integer(a[1])
1016 if err != nil {
1017 return "", err
1018 }
1019 item, exists := s.database[a[0]]
1020 if !exists {
1021 return integerReply(0), nil
1022 }
1023 expiry, old := s.time+seconds, item.expiry
1024 rejected := false
1025 switch condition {
1026 case "NX":
1027 rejected = old.set
1028 case "XX":
1029 rejected = !old.set
1030 case "GT":
1031 rejected = !old.set || expiry <= old.at
1032 case "LT":
1033 rejected = old.set && expiry >= old.at
1034 }
1035 if rejected {
1036 return integerReply(0), nil
1037 }
1038 if seconds <= 0 {
1039 delete(s.database, a[0])
1040 } else {
1041 item.expiry = expiration{expiry, true}
1042 s.database[a[0]] = item
1043 }
1044 return integerReply(1), nil
1045 case "TICK":
1046 ticks, err := integer(a[0])
1047 if err != nil {
1048 return "", err
1049 }
1050 if ticks <= 0 {
1051 return "", errPositive
1052 }
1053 s.time += ticks
1054 return integerReply(s.purge()), nil
1055 case "TIME":
1056 return integerReply(s.time), nil
1057 case "DBSIZE":
1058 return integerReply(len(s.database)), nil
1059 case "FLUSH":
1060 clear(s.database)
1061 return "OK", nil
1062 case "INFO":
1063 counts, expires := make(map[valueKind]int), 0
1064 for _, item := range s.database {
1065 counts[item.value.kind()]++
1066 if item.expiry.set {
1067 expires++
1068 }
1069 }
1070 return fmt.Sprintf("keys=%d,expires=%d,string=%d,list=%d,hash=%d,set=%d,zset=%d,time=%d,depth=%d,snapshots=%d",
1071 len(s.database), expires, counts[stringKind], counts[listKind], counts[hashKind], counts[setKind], counts[zsetKind],
1072 s.time, len(s.stack), len(s.snapshots)), nil
1073 }
1074 key := a[0]
1075 item, exists := s.database[key]
1076 switch command {
1077 case "TYPE":
1078 if !exists {
1079 return "none", nil
1080 }
1081 return string(item.value.kind()), nil
1082 case "TTL":
1083 if !exists {
1084 return integerReply(-2), nil
1085 }
1086 if !item.expiry.set {
1087 return integerReply(-1), nil
1088 }
1089 return integerReply(item.expiry.at - s.time), nil
1090 case "PERSIST":
1091 if !exists || !item.expiry.set {
1092 return integerReply(0), nil
1093 }
1094 item.expiry = expiration{}
1095 s.database[key] = item
1096 return integerReply(1), nil
1097 }
1098 if !exists {
1099 return "", errNoKey
1100 }
1101 if command == "RENAMENX" {
1102 if _, exists := s.database[a[1]]; exists {
1103 return integerReply(0), nil
1104 }
1105 }
1106 if key != a[1] {
1107 s.database[a[1]] = item
1108 delete(s.database, key)
1109 }
1110 if command == "RENAME" {
1111 return "OK", nil
1112 }
1113 return integerReply(1), nil
1114}

1116func (s *store) controlCommand(command string, a []string) (string, error) {
1117 switch command {
1118 case "BEGIN":
1119 s.stack = append(s.stack, cloneDatabase(s.database))
1120 case "COMMIT", "ROLLBACK":
1121 if len(s.stack) == 0 {
1122 return "", errors.New("ERR no transaction")
1123 }
1124 last := len(s.stack) - 1
1125 saved := s.stack[last]
1126 s.stack[last] = nil
1127 s.stack = s.stack[:last]
1128 if command == "ROLLBACK" {
1129 s.database = saved
1130 s.purge()
1131 }
1132 case "SNAPSHOT":
1133 s.snapshots[a[0]] = cloneDatabase(s.database)
1134 case "RESTORE":
1135 saved, exists := s.snapshots[a[0]]
1136 if !exists {
1137 return "", errors.New("ERR no such snapshot")
1138 }
1139 s.database = cloneDatabase(saved)
1140 s.purge()
1141 }
1142 return "OK", nil
1143}

1145func (s *store) execute(line string) (string, error) {
1146 words, valid := splitLine(line)
1147 if !valid {
1148 return "Invalid argument(s)", nil
1149 }
1150 command := strings.ToUpper(words[0])
1151 spec, known := commands[command]
1152 if !known {
1153 return "", fmt.Errorf("ERR unknown command '%s'", words[0])
1154 }
1155 a := words[1:]
1156 if len(a) < spec.minimum || (spec.maximum >= 0 && len(a) > spec.maximum) || (spec.parity >= 0 && len(a)%2 != spec.parity) {
1157 return "", fmt.Errorf("ERR wrong number of arguments for '%s' command", strings.ToLower(command))
1158 }
1159 switch spec.group {
1160 case textGroup:
1161 return s.textCommand(command, a)
1162 case listGroup:
1163 return s.listCommand(command, a)
1164 case hashGroup:
1165 return s.hashCommand(command, a)
1166 case setGroup:
1167 return s.setCommand(command, a)
1168 case zsetGroup:
1169 return s.zsetCommand(command, a)
1170 case keyGroup:
1171 return s.keyCommand(command, a)
1172 default:
1173 return s.controlCommand(command, a)
1174 }
1175}
