34std::string upper(std::string value) {
35 for (char& c : value) if ('a' <= c && c <= 'z') c = static_cast<char>(c - 'a' + 'A');
36 return value;
37}
38std::string lower(std::string value) {
39 for (char& c : value) if ('A' <= c && c <= 'Z') c = static_cast<char>(c - 'A' + 'a');
40 return value;
41}
42int number(const std::string& word, const std::string& message = notInteger) {
43 if (word == "0") return 0;
44 if (word.empty()) fail(message);
45 std::size_t start = word[0] == '-' ? 1 : 0, digits = word.size() - start;
46 if (digits < 1 || digits > 9 || word[start] < '1' || word[start] > '9') fail(message);
47 int result = 0;
48 for (std::size_t index = start; index < word.size(); ++index) {
49 if (word[index] < '0' || word[index] > '9') fail(message);
50 result = result * 10 + word[index] - '0';
51 }
52 return start == 1 ? -result : result;
53}
54int increment(int value, int delta) {
55 long long result = static_cast<long long>(value) + delta;
56 if (result < -999999999 || result > 999999999) fail(overflow);
57 return static_cast<int>(result);
58}
59std::string quote(const std::string& value) {
60 std::string result = "\"";
61 for (char c : value) { if (c == '"' || c == '\\') result += '\\'; result += c; }
62 return result + '"';
63}
64Reply status(const std::string& value) { return {value}; }
65Reply integer(long long value) { return {"(integer) " + std::to_string(value)}; }
66Reply stringReply(const std::optional<std::string>& value) { return {value ? quote(*value) : "(nil)"}; }
67Reply arrayReply(const std::vector<std::optional<std::string>>& values) {
68 if (values.empty()) return {"(empty array)"};
69 Reply result;
70 auto width = std::to_string(values.size()).size();
71 for (std::size_t index = 0; index < values.size(); ++index) {
72 auto label = std::to_string(index + 1);
73 result.push_back(std::string(width - label.size(), ' ') + label + ") " + stringReply(values[index])[0]);
74 }
75 return result;
76}
77Reply arrayReply(const Strings& values) {
78 return arrayReply(std::vector<std::optional<std::string>>(values.begin(), values.end()));
79}
80template<class Sequence>
81Sequence range(const Sequence& sequence, int start, int stop) {
82 int size = static_cast<int>(sequence.size());
83 if (start < 0) start += size;
84 if (stop < 0) stop += size;
85 start = std::max(0, start); stop = std::min(size - 1, stop);
86 return start <= stop ? Sequence(sequence.begin() + start, sequence.begin() + stop + 1) : Sequence{};
87}

89std::optional<Strings> splitArguments(const std::string& line) {
90 Strings arguments;
91 std::size_t position = 0;
92 while (position < line.size()) {
93 if (line[position] == ' ') { ++position; continue; }
94 if (line[position] != '"') {
95 auto start = position;
96 while (position < line.size() && line[position] != ' ') ++position;
97 arguments.push_back(line.substr(start, position - start));
98 continue;
99 }
100 ++position;
101 std::string value;
102 bool closed = false;
103 while (position < line.size()) {
104 char character = line[position++];
105 if (character == '"') { closed = true; break; }
106 if (character == '\\') {
107 if (position == line.size() || (line[position] != '"' && line[position] != '\\')) return std::nullopt;
108 value += line[position++];
109 } else value += character;
110 }
111 if (!closed || (position < line.size() && line[position] != ' ')) return std::nullopt;
112 arguments.push_back(value);
113 }
114 return arguments;
115}

117enum class PatternKind { Literal, Any, Star, CharacterClass };
118struct PatternPart {
119 PatternKind kind;
120 char literal = 0;
121 std::bitset<128> characters;
122 bool negated = false;
123};
124std::vector<PatternPart> parsePattern(const std::string& pattern) {
125 std::vector<PatternPart> parts;
126 for (std::size_t position = 0; position < pattern.size(); ++position) {
127 char character = pattern[position];
128 if (character == '\\') {
129 if (++position == pattern.size()) fail("ERR invalid pattern");
130 parts.push_back({PatternKind::Literal, pattern[position]});
131 } else if (character == '[') {
132 auto end = pattern.find(']', position + 1);
133 if (end == std::string::npos) fail("ERR invalid pattern");
134 auto content = pattern.substr(position + 1, end - position - 1);
135 PatternPart part{PatternKind::CharacterClass};
136 part.negated = !content.empty() && content[0] == '^';
137 if (part.negated) content.erase(content.begin());
138 if (content.empty()) fail("ERR invalid pattern");
139 for (std::size_t index = 0; index < content.size();) {
140 if (index + 2 < content.size() && content[index + 1] == '-') {
141 int first = content[index], last = content[index + 2];
142 for (int code = std::min(first, last); code <= std::max(first, last); ++code) part.characters.set(code);
143 index += 3;
144 } else part.characters.set(static_cast<unsigned char>(content[index++]));
145 }
146 parts.push_back(part); position = end;
147 } else if (character == '*') parts.push_back({PatternKind::Star});
148 else if (character == '?') parts.push_back({PatternKind::Any});
149 else parts.push_back({PatternKind::Literal, character});
150 }
151 return parts;
152}
153bool matches(const std::vector<PatternPart>& parts, const std::string& value) {
154 std::vector<bool> previous(value.size() + 1, false);
155 previous[0] = true;
156 for (const auto& part : parts) {
157 std::vector<bool> current(value.size() + 1, false);
158 if (part.kind == PatternKind::Star) {
159 current[0] = previous[0];
160 for (std::size_t index = 1; index <= value.size(); ++index) current[index] = previous[index] || current[index - 1];
161 } else {
162 for (std::size_t index = 0; index < value.size(); ++index) {
163 bool accepts = part.kind == PatternKind::Any || (part.kind == PatternKind::Literal ? part.literal == value[index]
164 : part.characters.test(static_cast<unsigned char>(value[index])) != part.negated);
165 current[index + 1] = previous[index] && accepts;
166 }
167 }
168 previous = std::move(current);
169 }
170 return previous.back();
171}

173enum class BoundKind { NegativeInfinity, Finite, PositiveInfinity };
174struct Bound {
175 BoundKind kind;
176 int value = 0;
177 bool inclusive = true;
178 bool accepts(int score, bool lowerBound) const {
179 if (kind == BoundKind::NegativeInfinity) return lowerBound;
180 if (kind == BoundKind::PositiveInfinity) return !lowerBound;
181 return lowerBound ? (inclusive ? score >= value : score > value) : (inclusive ? score <= value : score < value);
182 }
183};
184Bound scoreBound(const std::string& word) {
185 if (word == "-inf") return {BoundKind::NegativeInfinity};
186 if (word == "+inf") return {BoundKind::PositiveInfinity};
187 bool inclusive = word.empty() || word[0] != '(';
188 return {BoundKind::Finite, number(inclusive ? word : word.substr(1), "ERR min or max is not a float"), inclusive};
189}

243 template<class T> T readValue(const std::string& key) const {
244 auto found = database.find(key);
245 if (found == database.end()) return T{};
246 auto value = std::get_if<T>(&found->second.value);
247 if (!value) fail(wrongType);
248 return *value;
249 }
250 void replace(const std::string& key, Value value, std::optional<int> expiry = std::nullopt) {
251 bool emptyCollection = std::visit([](const auto& content) {
252 using T = std::decay_t<decltype(content)>;
253 if constexpr (std::is_same_v<T, std::string>) return false;
254 else return content.empty();
255 }, value);
256 if (emptyCollection) database.erase(key);
257 else database.insert_or_assign(key, Record{std::move(value), expiry});
258 }
259 void update(const std::string& key, Value value) {
260 auto found = database.find(key);
261 replace(key, std::move(value), found == database.end() ? std::nullopt : found->second.expiry);
262 }
263 int purge() {
264 int removed = 0;
265 for (auto iterator = database.begin(); iterator != database.end();) {
266 if (iterator->second.expiry && *iterator->second.expiry <= time) {
267 iterator = database.erase(iterator); ++removed;
268 } else ++iterator;
269 }
270 return removed;
271 }
272 Reply textCommand(const std::string& command, const Strings& args) {
273 const auto& key = args[0];
274 if (command == "SET") {
275 Members options;
276 std::optional<std::string> secondsWord;
277 for (std::size_t index = 2; index < args.size(); ++index) {
278 auto option = upper(args[index]);
279 if (option != "NX" && option != "XX" && option != "EX" && option != "KEEPTTL" && option != "GET") fail(syntax);
280 if (!options.insert(option).second) fail(syntax);
281 if (option == "EX") {
282 if (++index == args.size()) fail(syntax);
283 secondsWord = args[index];
284 }
285 }
286 if ((options.count("NX") && options.count("XX")) || (options.count("EX") && options.count("KEEPTTL"))) fail(syntax);
287 std::optional<int> expiry;
288 if (secondsWord) {
289 int seconds = number(*secondsWord);
290 if (seconds <= 0) fail("ERR invalid expire time in 'set' command");
291 expiry = time + seconds;
292 }
293 auto found = database.find(key);
294 bool exists = found != database.end();
295 std::optional<std::string> old;
296 if (options.count("GET")) {
297 auto value = readValue<std::string>(key);
298 if (exists) old = value;
299 }
300 if (options.count("KEEPTTL") && exists) expiry = found->second.expiry;
301 bool changed = !(options.count("NX") && exists) && !(options.count("XX") && !exists);
302 if (changed) replace(key, args[1], expiry);
303 if (options.count("GET")) return stringReply(old);
304 return changed ? status("OK") : stringReply(std::nullopt);
305 }
306 if (command == "MSET") {
307 for (std::size_t index = 0; index < args.size(); index += 2) replace(args[index], args[index + 1]);
308 return status("OK");
309 }
310 if (command == "MGET") {
311 std::vector<std::optional<std::string>> output;
312 for (const auto& name : args) {
313 auto found = database.find(name);
314 const auto* value = found == database.end() ? nullptr : std::get_if<std::string>(&found->second.value);
315 output.push_back(value ? std::optional<std::string>(*value) : std::nullopt);
316 }
317 return arrayReply(output);
318 }
319 if (command == "INCR" || command == "DECR" || command == "INCRBY" || command == "DECRBY") {
320 int delta = args.size() == 2 ? number(args[1]) : 1;
321 if (command == "DECR" || command == "DECRBY") delta = -delta;
322 auto value = readValue<std::string>(key);
323 int result = increment(database.count(key) ? number(value) : 0, delta);
324 update(key, std::to_string(result));
325 return integer(result);
326 }
327 if (command == "GETRANGE") {
328 int start = number(args[1]), stop = number(args[2]);
329 return stringReply(range(readValue<std::string>(key), start, stop));
330 }
331 auto value = readValue<std::string>(key);
332 if (command == "GET") return stringReply(database.count(key) ? std::optional<std::string>(value) : std::nullopt);
333 if (command == "STRLEN") return integer(value.size());
334 value += args[1];
335 update(key, value);
336 return integer(value.size());
337 }
338 Reply listCommand(const std::string& command, const Strings& args) {
339 const auto& key = args[0];
340 if (command == "LPOP" || command == "RPOP") {
341 bool counted = args.size() == 2;
342 int count = counted ? number(args[1], positive) : 1;
343 if (count < 0) fail(positive);
344 auto values = readValue<Strings>(key);
345 if (!database.count(key)) return stringReply(std::nullopt);
346 count = std::min(count, static_cast<int>(values.size()));
347 Strings removed;
348 for (int index = 0; index < count; ++index) {
349 if (command == "LPOP") { removed.push_back(values.front()); values.erase(values.begin()); }
350 else { removed.push_back(values.back()); values.pop_back(); }
351 }
352 update(key, values);
353 return counted ? arrayReply(removed) : stringReply(removed[0]);
354 }
355 if (command == "LRANGE" || command == "LTRIM") {
356 int start = number(args[1]), stop = number(args[2]);
357 auto values = range(readValue<Strings>(key), start, stop);
358 if (command == "LRANGE") return arrayReply(values);
359 update(key, values);
360 return status("OK");
361 }
362 if (command == "LINDEX" || command == "LSET") {
363 int index = number(args[1]);
364 auto values = readValue<Strings>(key);
365 int size = static_cast<int>(values.size());
366 if (index < 0) index += size;
367 bool inside = 0 <= index && index < size;
368 if (command == "LINDEX") return stringReply(inside ? std::optional<std::string>(values[index]) : std::nullopt);
369 if (!database.count(key)) fail(noKey);
370 if (!inside) fail("ERR index out of range");
371 values[index] = args[2];
372 update(key, values);
373 return status("OK");
374 }
375 if (command == "LREM") {
376 int count = number(args[1]);
377 auto values = readValue<Strings>(key);
378 if (count < 0) std::reverse(values.begin(), values.end());
379 int limit = count == 0 ? static_cast<int>(values.size()) : (count < 0 ? -count : count);
380 int removed = 0;
381 Strings kept;
382 for (const auto& value : values) {
383 if (value == args[2] && removed < limit) ++removed;
384 else kept.push_back(value);
385 }
386 if (count < 0) std::reverse(kept.begin(), kept.end());
387 update(key, kept);
388 return integer(removed);
389 }
390 if (command == "LINSERT") {
391 auto where = upper(args[1]);
392 if (where != "BEFORE" && where != "AFTER") fail(syntax);
393 auto values = readValue<Strings>(key);
394 if (!database.count(key)) return integer(0);
395 auto pivot = std::find(values.begin(), values.end(), args[2]);
396 if (pivot == values.end()) return integer(-1);
397 values.insert(pivot + (where == "AFTER" ? 1 : 0), args[3]);
398 update(key, values);
399 return integer(values.size());
400 }
401 auto values = readValue<Strings>(key);
402 if (command == "LLEN") return integer(values.size());
403 for (std::size_t index = 1; index < args.size(); ++index) {
404 if (command == "LPUSH") values.insert(values.begin(), args[index]);
405 else values.push_back(args[index]);
406 }
407 update(key, values);
408 return integer(values.size());
409 }
410 Reply hashCommand(const std::string& command, const Strings& args) {
411 const auto& key = args[0];
412 if (command == "HINCRBY") {
413 int delta = number(args[2]);
414 auto fields = readValue<Hash>(key);
415 auto found = fields.find(args[1]);
416 int old = found == fields.end() ? 0 : number(found->second, "ERR hash value is not an integer");
417 int result = increment(old, delta);
418 fields[args[1]] = std::to_string(result);
419 update(key, fields);
420 return integer(result);
421 }
422 auto fields = readValue<Hash>(key);
423 if (command == "HSET") {
424 int added = 0;
425 for (std::size_t index = 1; index < args.size(); index += 2) {
426 if (!fields.count(args[index])) ++added;
427 fields[args[index]] = args[index + 1];
428 }
429 update(key, fields);
430 return integer(added);
431 }
432 if (command == "HGET") {
433 auto found = fields.find(args[1]);
434 return stringReply(found == fields.end() ? std::nullopt : std::optional<std::string>(found->second));
435 }
436 if (command == "HDEL") {
437 int removed = 0;
438 for (std::size_t index = 1; index < args.size(); ++index) removed += static_cast<int>(fields.erase(args[index]));
439 update(key, fields);
440 return integer(removed);
441 }
442 if (command == "HLEN") return integer(fields.size());
443 if (command == "HEXISTS") return integer(fields.count(args[1]));
444 Strings output;
445 for (const auto& [field, value] : fields) {
446 if (command != "HVALS") output.push_back(field);
447 if (command != "HKEYS") output.push_back(value);
448 }
449 return arrayReply(output);
450 }
451 Reply setCommand(const std::string& command, const Strings& args) {
452 bool storing = command.find("STORE") != std::string::npos;
453 std::string operation = storing ? command.substr(0, command.size() - 5) : command;
454 if (operation == "SINTER" || operation == "SUNION" || operation == "SDIFF") {
455 std::vector<Members> sources;
456 for (std::size_t index = storing ? 1 : 0; index < args.size(); ++index) sources.push_back(readValue<Members>(args[index]));
457 Members result = sources[0];
458 for (std::size_t index = 1; index < sources.size(); ++index) {
459 if (operation == "SUNION") result.insert(sources[index].begin(), sources[index].end());
460 else {
461 for (auto member = result.begin(); member != result.end();) {
462 bool present = sources[index].count(*member) != 0;
463 if ((operation == "SINTER" && !present) || (operation == "SDIFF" && present)) member = result.erase(member);
464 else ++member;
465 }
466 }
467 }
468 if (storing) { replace(args[0], result); return integer(result.size()); }
469 return arrayReply(Strings(result.begin(), result.end()));
470 }
471 auto members = readValue<Members>(args[0]);
472 if (command == "SCARD") return integer(members.size());
473 if (command == "SMEMBERS") return arrayReply(Strings(members.begin(), members.end()));
474 if (command == "SISMEMBER") return integer(members.count(args[1]));
475 int changed = 0;
476 for (std::size_t index = 1; index < args.size(); ++index) {
477 if (command == "SADD") changed += members.insert(args[index]).second ? 1 : 0;
478 else changed += static_cast<int>(members.erase(args[index]));
479 }
480 update(args[0], members);
481 return integer(changed);
482 }
483 Reply sortedSetCommand(const std::string& command, const Strings& args) {
484 const auto& key = args[0];
485 if (command == "ZADD") {
486 std::vector<int> scores;
487 for (std::size_t index = 1; index < args.size(); index += 2) scores.push_back(number(args[index]));
488 auto members = readValue<SortedSet>(key);
489 int added = 0;
490 for (std::size_t index = 0; index < scores.size(); ++index) {
491 const auto& member = args[2 * index + 2];
492 if (!members.count(member)) ++added;
493 members[member] = scores[index];
494 }
495 update(key, members);
496 return integer(added);
497 }
498 if (command == "ZINCRBY") {
499 int delta = number(args[1]);
500 auto members = readValue<SortedSet>(key);
501 int result = increment(members.count(args[2]) ? members.at(args[2]) : 0, delta);
502 members[args[2]] = result;
503 update(key, members);
504 return stringReply(std::to_string(result));
505 }
506 bool byIndex = command == "ZRANGE", byScore = command == "ZRANGEBYSCORE";
507 bool withScores = args.size() == 4;
508 int start = 0, stop = 0;
509 Bound minimum{BoundKind::NegativeInfinity}, maximum{BoundKind::PositiveInfinity};
510 if (byIndex || byScore) {
511 if (withScores && upper(args[3]) != "WITHSCORES") fail(syntax);
512 if (byIndex) { start = number(args[1]); stop = number(args[2]); }
513 else { minimum = scoreBound(args[1]); maximum = scoreBound(args[2]); }
514 }
515 auto members = readValue<SortedSet>(key);
516 if (command == "ZSCORE") {
517 auto found = members.find(args[1]);
518 return stringReply(found == members.end() ? std::nullopt : std::optional<std::string>(std::to_string(found->second)));
519 }
520 if (command == "ZREM") {
521 int removed = 0;
522 for (std::size_t index = 1; index < args.size(); ++index) removed += static_cast<int>(members.erase(args[index]));
523 update(key, members);
524 return integer(removed);
525 }
526 if (command == "ZCARD") return integer(members.size());
527 std::vector<std::pair<std::string, int>> ordered(members.begin(), members.end());
528 std::sort(ordered.begin(), ordered.end(), [](const auto& left, const auto& right) {
529 return left.second != right.second ? left.second < right.second : left.first < right.first;
530 });
531 if (command == "ZRANK") {
532 for (std::size_t index = 0; index < ordered.size(); ++index) if (ordered[index].first == args[1]) return integer(index);
533 return stringReply(std::nullopt);
534 }
535 if (byIndex) ordered = range(ordered, start, stop);
536 Strings output;
537 for (const auto& [member, score] : ordered) {
538 if (byScore && (!minimum.accepts(score, true) || !maximum.accepts(score, false))) continue;
539 output.push_back(member);
540 if (withScores) output.push_back(std::to_string(score));
541 }
542 return arrayReply(output);
543 }
544 Reply keyCommand(const std::string& command, const Strings& args) {
545 if (command == "DEL" || command == "EXISTS") {
546 int count = 0;
547 for (const auto& key : args) count += static_cast<int>(command == "DEL" ? database.erase(key) : database.count(key));
548 return integer(count);
549 }
550 if (command == "KEYS") {
551 auto pattern = parsePattern(args[0]);
552 Strings output;
553 for (const auto& [key, record] : database) if (matches(pattern, key)) output.push_back(key);
554 return arrayReply(output);
555 }
556 if (command == "EXPIRE") {
557 std::string condition = args.size() == 3 ? upper(args[2]) : "";
558 if (args.size() == 3 && condition != "NX" && condition != "XX" && condition != "GT" && condition != "LT") fail("ERR Unsupported option " + args[2]);
559 int seconds = number(args[1]);
560 auto found = database.find(args[0]);
561 if (found == database.end()) return integer(0);
562 auto old = found->second.expiry;
563 int expiry = time + seconds;
564 bool rejected = (condition == "NX" && old) || (condition == "XX" && !old)
565 || (condition == "GT" && (!old || expiry <= *old)) || (condition == "LT" && old && expiry >= *old);
566 if (rejected) return integer(0);
567 if (seconds <= 0) database.erase(found);
568 else found->second.expiry = expiry;
569 return integer(1);
570 }
571 if (command == "TICK") {
572 int seconds = number(args[0]);
573 if (seconds <= 0) fail(positive);
574 time += seconds;
575 return integer(purge());
576 }
577 if (command == "TIME") return integer(time);
578 if (command == "DBSIZE") return integer(database.size());
579 if (command == "FLUSH") { database.clear(); return status("OK"); }
580 if (command == "INFO") {
581 std::array<int, 5> counts{};
582 int expires = 0;
583 for (const auto& [key, record] : database) { ++counts[record.value.index()]; if (record.expiry) ++expires; }
584 return status("keys=" + std::to_string(database.size()) + ",expires=" + std::to_string(expires)
585 + ",string=" + std::to_string(counts[0]) + ",list=" + std::to_string(counts[1])
586 + ",hash=" + std::to_string(counts[2]) + ",set=" + std::to_string(counts[3])
587 + ",zset=" + std::to_string(counts[4]) + ",time=" + std::to_string(time)
588 + ",depth=" + std::to_string(transactions.size()) + ",snapshots=" + std::to_string(snapshots.size()));
589 }
590 auto found = database.find(args[0]);
591 if (command == "TYPE") {
592 const std::array<std::string, 5> names{"string", "list", "hash", "set", "zset"};
593 return status(found == database.end() ? "none" : names[found->second.value.index()]);
594 }
595 if (command == "TTL") return integer(found == database.end() ? -2 : (found->second.expiry ? *found->second.expiry - time : -1));
596 if (command == "PERSIST") {
597 if (found == database.end() || !found->second.expiry) return integer(0);
598 found->second.expiry.reset();
599 return integer(1);
600 }
601 if (found == database.end()) fail(noKey);
602 if (command == "RENAMENX" && database.count(args[1])) return integer(0);
603 if (args[0] != args[1]) {
604 database.insert_or_assign(args[1], found->second);
605 database.erase(found);
606 }
607 return command == "RENAME" ? status("OK") : integer(1);
608 }
609 Reply controlCommand(const std::string& command, const Strings& args) {
610 if (command == "BEGIN") { transactions.push_back(database); return status("OK"); }
611 if (command == "COMMIT" || command == "ROLLBACK") {
612 if (transactions.empty()) fail("ERR no transaction");
613 if (command == "ROLLBACK") database = std::move(transactions.back());
614 transactions.pop_back();
615 if (command == "ROLLBACK") purge();
616 return status("OK");
617 }
618 if (command == "SNAPSHOT") { snapshots.insert_or_assign(args[0], database); return status("OK"); }
619 auto found = snapshots.find(args[0]);
620 if (found == snapshots.end()) fail("ERR no such snapshot");
621 database = found->second;
622 purge();
623 return status("OK");
624 }
625
626public:
627 Reply execute(const std::string& line) {
628 auto parsed = splitArguments(line);
629 if (!parsed || parsed->empty()) return status("Invalid argument(s)");
630 auto command = upper((*parsed)[0]);
631 auto found = commands.find(command);
632 if (found == commands.end()) fail("ERR unknown command '" + (*parsed)[0] + "'");
633 Strings args(parsed->begin() + 1, parsed->end());
634 const auto& specification = found->second;
635 int count = static_cast<int>(args.size());
636 if (count < specification.minimum || (specification.maximum >= 0 && count > specification.maximum)
637 || (specification.parity >= 0 && count % 2 != specification.parity)) {
638 fail("ERR wrong number of arguments for '" + lower(command) + "' command");
639 }
640 switch (specification.group) {
641 case Group::Text: return textCommand(command, args);
642 case Group::Lists: return listCommand(command, args);
643 case Group::Hashes: return hashCommand(command, args);
644 case Group::Sets: return setCommand(command, args);
645 case Group::SortedSets: return sortedSetCommand(command, args);
646 case Group::Keys: return keyCommand(command, args);
647 case Group::Control: return controlCommand(command, args);
648 }
649 throw std::logic_error("invalid command group");
650 }
651};
652
653int main() {
654 std::ios::sync_with_stdio(false);
655 std::cin.tie(nullptr);
656 std::string line;
657 if (!std::getline(std::cin, line)) return 0;
658 int count = std::stoi(line);
659 Store store;
660 for (int index = 0; index < count; ++index) {
661 std::getline(std::cin, line);
662 try {
663 for (const auto& output : store.execute(line)) std::cout << output << '\n';
664 } catch (const CommandError& error) {
665 std::cout << "(error) " << error.what() << '\n';
666 }
667 }
