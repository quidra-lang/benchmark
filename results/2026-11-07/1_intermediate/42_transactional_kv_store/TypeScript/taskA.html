32const integerNotation = /^(?:0|-?[1-9][0-9]{0,8})$/;
33function number(value: string, message: string = messages.notInteger): number {
34 if (!integerNotation.test(value)) fail(message);

37function increment(value: number, delta: number): number {
38 const result = value + delta;
39 if (Math.abs(result) > 999999999) fail(messages.overflow);
40 return result;

42const status = (value: string): Reply => [value];
43const integer = (value: number): Reply => [`(integer) ${value}`];
44const string = (value: string | null): Reply => [value === null ? '(nil)' : JSON.stringify(value)];
45function array(values: readonly (string | null)[]): Reply {
46 if (values.length === 0) return ['(empty array)'];

50const compare = (left: string, right: string): number => left < right ? -1 : left > right ? 1 : 0;

52function range<T>(sequence: readonly T[], start: number, stop: number): T[];
53function range(sequence: string | readonly unknown[], start: number, stop: number): string | unknown[] {
54 if (start < 0) start += sequence.length;
55 if (stop < 0) stop += sequence.length;
56 start = Math.max(0, start); stop = Math.min(sequence.length - 1, stop);
57 return start <= stop ? sequence.slice(start, stop + 1) : sequence.slice(0, 0);
58}

60function splitArguments(line: string): string[] | null {
61 const argumentsList: string[] = [];
62 let position = 0;
63 while (position < line.length) {
64 if (line[position] === ' ') { ++position; continue; }
65 if (line[position] !== '"') {
66 const start = position;
67 while (position < line.length && line[position] !== ' ') ++position;
68 argumentsList.push(line.slice(start, position));
69 continue;
70 }
71 ++position;
72 let value = '', closed = false;
73 while (position < line.length) {
74 const character = line[position++];
75 if (character === '"') { closed = true; break; }
76 if (character === '\\') {
77 if (position === line.length || !['"', '\\'].includes(line[position])) return null;
78 value += line[position++];
79 } else value += character;
80 }
81 if (!closed || (position < line.length && line[position] !== ' ')) return null;
82 argumentsList.push(value);
83 }
84 return argumentsList;
85}

87function parsePattern(pattern: string): PatternPart[] {
88 const parts: PatternPart[] = [];
89 for (let position = 0; position < pattern.length; ++position) {
90 const character = pattern[position];
91 if (character === '\\') {
92 if (++position === pattern.length) fail('ERR invalid pattern');
93 parts.push({ kind: 'literal', value: pattern[position] });
94 } else if (character === '[') {
95 const end = pattern.indexOf(']', position + 1);
96 if (end < 0) fail('ERR invalid pattern');
97 let content = pattern.slice(position + 1, end);
98 const negated = content.startsWith('^');
99 if (negated) content = content.slice(1);
100 if (content.length === 0) fail('ERR invalid pattern');
101 const characters = new Set<string>();
102 for (let index = 0; index < content.length;) {
103 if (index + 2 < content.length && content[index + 1] === '-') {
104 const first = content.charCodeAt(index), last = content.charCodeAt(index + 2);
105 for (let code = Math.min(first, last); code <= Math.max(first, last); ++code) characters.add(String.fromCharCode(code));
106 index += 3;
107 } else characters.add(content[index++]);
108 }
109 parts.push({ kind: 'class', characters, negated }); position = end;
110 } else if (character === '*') parts.push({ kind: 'star' });
111 else if (character === '?') parts.push({ kind: 'any' });
112 else parts.push({ kind: 'literal', value: character });
113 }
114 return parts;
115}
116function matches(parts: readonly PatternPart[], value: string): boolean {
117 let previous = new Array<boolean>(value.length + 1).fill(false);
118 previous[0] = true;
119 for (const part of parts) {
120 const current = new Array<boolean>(value.length + 1).fill(false);
121 if (part.kind === 'star') {
122 current[0] = previous[0];
123 for (let index = 1; index <= value.length; ++index) current[index] = previous[index] || current[index - 1];
124 } else {
125 for (let index = 0; index < value.length; ++index) {
126 const accepts = part.kind === 'any' || (part.kind === 'literal' ? part.value === value[index]
127 : part.characters.has(value[index]) !== part.negated);
128 current[index + 1] = previous[index] && accepts;
129 }
130 }
131 previous = current;
132 }
133 return previous[value.length];
134}
135function scoreBound(word: string): ScoreBound {
136 if (word === '-inf' || word === '+inf') return { value: word === '-inf' ? -Infinity : Infinity, inclusive: true };
137 const inclusive = !word.startsWith('(');
138 return { value: number(inclusive ? word : word.slice(1), 'ERR min or max is not a float'), inclusive };
139}

181class Store {
182 private database: Database = new Map();
183 private time = 0;
184 private stack: Database[] = [];
185 private snapshots = new Map<string, Database>();
186 private data<Kind extends ValueKind>(key: string, type: Kind, fallback: ValueTypes[Kind]): ValueTypes[Kind];
187 private data<Kind extends ValueKind>(key: string, type: Kind, fallback: null): ValueTypes[Kind] | null;
188 private data(key: string, type: ValueKind, fallback: ValueTypes[ValueKind] | null): ValueTypes[ValueKind] | null {
189 const entry = this.database.get(key);
190 if (entry !== undefined && entry.type !== type) fail(messages.wrongType);
191 return entry === undefined ? fallback : entry.value;
192 }
193 private write(key: string, value: Value, expiry?: number | null): void {
194 const empty = value.type !== 'string' && (value.type === 'list' ? value.value.length : value.value.size) === 0;
195 if (empty) this.database.delete(key);
196 else this.database.set(key, { ...value, expiry: expiry === undefined ? (this.database.get(key)?.expiry ?? null) : expiry });
197 }
198 private purge(): number {
199 let removed = 0;
200 for (const [key, entry] of this.database) if (entry.expiry !== null && entry.expiry <= this.time) {
201 this.database.delete(key); ++removed;
202 }
203 return removed;
204 }
205 command(line: string): Reply {
206 const words = splitArguments(line);
207 if (words === null) return status('Invalid argument(s)');
208 const verb = words[0].toUpperCase(), args = words.slice(1), specification = commands.get(verb);
209 try {
210 if (specification === undefined) fail(`ERR unknown command '${words[0]}'`);
211 const { minimum, maximum, parity, group } = specification;
212 if (args.length < minimum || args.length > maximum || (parity !== null && args.length % 2 !== parity)) {
213 fail(`ERR wrong number of arguments for '${verb.toLowerCase()}' command`);
214 }
215 return this[group](verb, args);
216 } catch (error) {
217 if (!(error instanceof CommandError)) throw error;
218 return [`(error) ${error.message}`];
219 }
220 }
221 private strings(verb: string, args: readonly string[]): Reply {
222 const key = args[0];
223 switch (verb) {
224 case 'SET': {
225 const options = new Map<string, string>();
226 for (let index = 2; index < args.length; ++index) {
227 const option = args[index].toUpperCase();
228 if (!['NX', 'XX', 'EX', 'KEEPTTL', 'GET'].includes(option) || options.has(option)) fail(messages.syntax);
229 if (option === 'EX') {
230 if (++index === args.length) fail(messages.syntax);
231 options.set(option, args[index]);
232 } else options.set(option, '');
233 }
234 if ((options.has('NX') && options.has('XX')) || (options.has('EX') && options.has('KEEPTTL'))) fail(messages.syntax);
235 let expiry: number | null = null;
236 if (options.has('EX')) {
237 const seconds = number(options.get('EX')!);
238 if (seconds <= 0) fail("ERR invalid expire time in 'set' command");
239 expiry = this.time + seconds;
240 }
241 const previous = this.database.get(key);
242 const oldValue = options.has('GET') ? this.data(key, 'string', null) : null;
243 const changed = !(options.has('NX') && previous !== undefined) && !(options.has('XX') && previous === undefined);
244 if (options.has('KEEPTTL') && previous !== undefined) expiry = previous.expiry;
245 if (changed) this.write(key, { type: 'string', value: args[1] }, expiry);
246 return options.has('GET') ? string(oldValue) : changed ? status('OK') : string(null);
247 }
248 case 'GET': return string(this.data(key, 'string', null));
249 case 'MSET':
250 for (let index = 0; index < args.length; index += 2) this.write(args[index], { type: 'string', value: args[index + 1] }, null);
251 return status('OK');
252 case 'MGET':
253 return array(args.map(key => { const entry = this.database.get(key); return entry?.type === 'string' ? entry.value : null; }));
254 case 'INCR': case 'DECR': case 'INCRBY': case 'DECRBY': {
255 let delta = verb === 'INCR' ? 1 : verb === 'DECR' ? -1 : number(args[1]);
256 if (verb === 'DECRBY') delta = -delta;
257 const result = increment(number(this.data(key, 'string', '0')), delta);
258 this.write(key, { type: 'string', value: String(result) });
259 return integer(result);
260 }
261 case 'APPEND': {
262 const value = this.data(key, 'string', '') + args[1];
263 this.write(key, { type: 'string', value });
264 return integer(value.length);
265 }
266 case 'STRLEN': return integer(this.data(key, 'string', '').length);
267 case 'GETRANGE': {
268 const start = number(args[1]), stop = number(args[2]);
269 return string(range(this.data(key, 'string', ''), start, stop));
270 }
271 }
272 throw new Error('Unreachable command');
273 }
274 private lists(verb: string, args: readonly string[]): Reply {
275 const key = args[0];
276 let first = 0, second = 0;
277 if (verb === 'LPOP' || verb === 'RPOP') {
278 first = args.length === 2 ? number(args[1], messages.positive) : 1;
279 if (first < 0) fail(messages.positive);
280 } else if (['LRANGE', 'LINDEX', 'LSET', 'LREM', 'LTRIM'].includes(verb)) {
281 first = number(args[1]);
282 if (verb === 'LRANGE' || verb === 'LTRIM') second = number(args[2]);
283 } else if (verb === 'LINSERT' && !['BEFORE', 'AFTER'].includes(args[1].toUpperCase())) fail(messages.syntax);
284 const list = this.data(key, 'list', []);
285 switch (verb) {
286 case 'LPUSH': case 'RPUSH':
287 for (const element of args.slice(1)) {
288 if (verb === 'LPUSH') list.unshift(element); else list.push(element);
289 }
290 this.write(key, { type: 'list', value: list });
291 return integer(list.length);
292 case 'LPOP': case 'RPOP': {
293 if (!this.database.has(key)) return string(null);
294 const removed: string[] = [];
295 while (removed.length < first && list.length > 0) removed.push(verb === 'LPOP' ? list.shift()! : list.pop()!);
296 this.write(key, { type: 'list', value: list });
297 return args.length === 2 ? array(removed) : string(removed[0]);
298 }
299 case 'LRANGE': return array(range(list, first, second));
300 case 'LLEN': return integer(list.length);
301 case 'LINDEX': case 'LSET': {
302 if (verb === 'LSET' && !this.database.has(key)) fail(messages.noKey);
303 const index = first < 0 ? first + list.length : first;
304 if (verb === 'LINDEX') return string(index >= 0 && index < list.length ? list[index] : null);
305 if (index < 0 || index >= list.length) fail('ERR index out of range');
306 list[index] = args[2]; this.write(key, { type: 'list', value: list });
307 return status('OK');
308 }
309 case 'LREM': {
310 let removed = 0;
311 const ordered = first < 0 ? [...list].reverse() : list;
312 const remaining = ordered.filter(element => {
313 if (element === args[2] && (first === 0 || removed < Math.abs(first))) { ++removed; return false; }
314 return true;
315 });
316 if (first < 0) remaining.reverse();
317 this.write(key, { type: 'list', value: remaining });
318 return integer(removed);
319 }
320 case 'LTRIM': this.write(key, { type: 'list', value: range(list, first, second) }); return status('OK');
321 case 'LINSERT': {
322 if (!this.database.has(key)) return integer(0);
323 const index = list.indexOf(args[2]);
324 if (index < 0) return integer(-1);
325 list.splice(index + (args[1].toUpperCase() === 'AFTER' ? 1 : 0), 0, args[3]);
326 this.write(key, { type: 'list', value: list });
327 return integer(list.length);
328 }
329 }
330 throw new Error('Unreachable command');
331 }
332 private hashes(verb: string, args: readonly string[]): Reply {
333 const key = args[0], delta = verb === 'HINCRBY' ? number(args[2]) : 0;
334 const hash = this.data(key, 'hash', new Map<string, string>());
335 switch (verb) {
336 case 'HSET': {
337 let added = 0;
338 for (let index = 1; index < args.length; index += 2) {
339 if (!hash.has(args[index])) ++added;
340 hash.set(args[index], args[index + 1]);
341 }
342 this.write(key, { type: 'hash', value: hash });
343 return integer(added);
344 }
345 case 'HGET': return string(hash.get(args[1]) ?? null);
346 case 'HDEL': {
347 let removed = 0;
348 for (const field of args.slice(1)) if (hash.delete(field)) ++removed;
349 this.write(key, { type: 'hash', value: hash });
350 return integer(removed);
351 }
352 case 'HLEN': return integer(hash.size);
353 case 'HEXISTS': return integer(hash.has(args[1]) ? 1 : 0);
354 case 'HINCRBY': {
355 const result = increment(number(hash.get(args[1]) ?? '0', 'ERR hash value is not an integer'), delta);
356 hash.set(args[1], String(result)); this.write(key, { type: 'hash', value: hash });
357 return integer(result);
358 }
359 case 'HGETALL': case 'HKEYS': case 'HVALS': {
360 const fields = [...hash.keys()].sort();
361 return array(verb === 'HKEYS' ? fields : verb === 'HVALS' ? fields.map(field => hash.get(field)!)
362 : fields.flatMap(field => [field, hash.get(field)!]));
363 }
364 }
365 throw new Error('Unreachable command');
366 }
367 private sets(verb: string, args: readonly string[]): Reply {
368 const storing = verb.endsWith('STORE'), operation = storing ? verb.slice(0, -5) : verb;
369 if (['SINTER', 'SUNION', 'SDIFF'].includes(operation)) {
370 const sources = (storing ? args.slice(1) : args).map(key => this.data(key, 'set', new Set<string>()));
371 const result = new Set(sources[0]);
372 for (const source of sources.slice(1)) {
373 if (operation === 'SUNION') for (const member of source) result.add(member);
374 else if (operation === 'SINTER') { for (const member of result) if (!source.has(member)) result.delete(member); }
375 else for (const member of source) result.delete(member);
376 }
377 if (storing) { this.write(args[0], { type: 'set', value: result }, null); return integer(result.size); }
378 return array([...result].sort());
379 }
380 const key = args[0], members = this.data(key, 'set', new Set<string>());
381 switch (verb) {
382 case 'SADD': {
383 let added = 0;
384 for (const member of args.slice(1)) if (!members.has(member)) { members.add(member); ++added; }
385 this.write(key, { type: 'set', value: members });
386 return integer(added);
387 }
388 case 'SREM': {
389 let removed = 0;
390 for (const member of args.slice(1)) if (members.delete(member)) ++removed;
391 this.write(key, { type: 'set', value: members });
392 return integer(removed);
393 }
394 case 'SISMEMBER': return integer(members.has(args[1]) ? 1 : 0);
395 case 'SMEMBERS': return array([...members].sort());
396 case 'SCARD': return integer(members.size);
397 }
398 throw new Error('Unreachable command');
399 }
400 private sortedSets(verb: string, args: readonly string[]): Reply {
401 const key = args[0], scores: number[] = [];
402 let delta = 0;
403 let query: RangeQuery | null = null;
404 if (verb === 'ZADD') for (let index = 1; index < args.length; index += 2) scores.push(number(args[index]));
405 if (verb === 'ZINCRBY') delta = number(args[1]);
406 if (verb === 'ZRANGE' || verb === 'ZRANGEBYSCORE') {
407 if (args.length === 4 && args[3].toUpperCase() !== 'WITHSCORES') fail(messages.syntax);
408 query = verb === 'ZRANGE' ? { kind: 'rank', start: number(args[1]), stop: number(args[2]) }
409 : { kind: 'score', min: scoreBound(args[1]), max: scoreBound(args[2]) };
410 }
411 const members = this.data(key, 'zset', new Map<string, number>());
412 switch (verb) {
413 case 'ZADD': {
414 let added = 0;
415 for (let index = 0; index < scores.length; ++index) {
416 const member = args[2 * index + 2];
417 if (!members.has(member)) ++added;
418 members.set(member, scores[index]);
419 }
420 this.write(key, { type: 'zset', value: members });
421 return integer(added);
422 }
423 case 'ZSCORE': return string(members.has(args[1]) ? String(members.get(args[1])) : null);
424 case 'ZINCRBY': {
425 const score = increment(members.get(args[2]) ?? 0, delta);
426 members.set(args[2], score); this.write(key, { type: 'zset', value: members });
427 return string(String(score));
428 }
429 case 'ZREM': {
430 let removed = 0;
431 for (const member of args.slice(1)) if (members.delete(member)) ++removed;
432 this.write(key, { type: 'zset', value: members });
433 return integer(removed);
434 }
435 case 'ZCARD': return integer(members.size);
436 case 'ZRANK': case 'ZRANGE': case 'ZRANGEBYSCORE': {
437 const ordered = [...members].sort((left, right) => left[1] - right[1] || compare(left[0], right[0]));
438 if (verb === 'ZRANK') {
439 const rank = ordered.findIndex(([member]) => member === args[1]);
440 return rank < 0 ? string(null) : integer(rank);
441 }
442 const bounds = query;
443 if (bounds === null) throw new Error('Missing range query');
444 const selected = bounds.kind === 'rank' ? range(ordered, bounds.start, bounds.stop) : ordered.filter(([, score]) =>
445 (score > bounds.min.value || (score === bounds.min.value && bounds.min.inclusive))
446 && (score < bounds.max.value || (score === bounds.max.value && bounds.max.inclusive)));
447 return array(args.length === 4 ? selected.flatMap(([member, score]) => [member, String(score)]) : selected.map(([member]) => member));
448 }
449 }
450 throw new Error('Unreachable command');
451 }
452 private keys(verb: string, args: readonly string[]): Reply {
453 const key = args[0];
454 switch (verb) {
455 case 'DEL': {
456 let removed = 0;
457 for (const key of args) if (this.database.delete(key)) ++removed;
458 return integer(removed);
459 }
460 case 'EXISTS': return integer(args.filter(key => this.database.has(key)).length);
461 case 'TYPE': return status(this.database.get(key)?.type ?? 'none');
462 case 'RENAME': case 'RENAMENX': {
463 const entry = this.database.get(key);
464 if (entry === undefined) fail(messages.noKey);
465 if (verb === 'RENAMENX' && this.database.has(args[1])) return integer(0);
466 if (key !== args[1]) { this.database.set(args[1], entry); this.database.delete(key); }
467 return verb === 'RENAMENX' ? integer(1) : status('OK');
468 }
469 case 'KEYS': {
470 const parts = parsePattern(key);
471 return array([...this.database.keys()].filter(key => matches(parts, key)).sort());
472 }
473 case 'EXPIRE': {
474 const condition = args.length === 3 ? args[2].toUpperCase() : null;
475 if (condition !== null && !['NX', 'XX', 'GT', 'LT'].includes(condition)) fail(`ERR Unsupported option ${args[2]}`);
476 const seconds = number(args[1]), entry = this.database.get(key);
477 if (entry === undefined) return integer(0);
478 const expiry = this.time + seconds, previous = entry.expiry;
479 if ((condition === 'NX' && previous !== null) || (condition === 'XX' && previous === null)
480 || (condition === 'GT' && (previous === null || expiry <= previous))
481 || (condition === 'LT' && previous !== null && expiry >= previous)) return integer(0);
482 if (seconds <= 0) this.database.delete(key); else entry.expiry = expiry;
483 return integer(1);
484 }
485 case 'TTL': {
486 const entry = this.database.get(key);
487 return integer(entry === undefined ? -2 : entry.expiry === null ? -1 : entry.expiry - this.time);
488 }
489 case 'PERSIST': {
490 const entry = this.database.get(key);
491 if (entry === undefined || entry.expiry === null) return integer(0);
492 entry.expiry = null;
493 return integer(1);
494 }
495 case 'TICK': {
496 const ticks = number(key);
497 if (ticks <= 0) fail(messages.positive);
498 this.time += ticks;
499 return integer(this.purge());
500 }
501 case 'TIME': return integer(this.time);
502 case 'DBSIZE': return integer(this.database.size);
503 case 'FLUSH': this.database.clear(); return status('OK');
504 case 'INFO': {
505 const counts = { string: 0, list: 0, hash: 0, set: 0, zset: 0 };
506 let expires = 0;
507 for (const entry of this.database.values()) { ++counts[entry.type]; if (entry.expiry !== null) ++expires; }
508 return status(`keys=${this.database.size},expires=${expires},string=${counts.string},list=${counts.list},hash=${counts.hash},set=${counts.set},zset=${counts.zset},time=${this.time},depth=${this.stack.length},snapshots=${this.snapshots.size}`);
509 }
510 }
511 throw new Error('Unreachable command');
512 }
513 private control(verb: string, args: readonly string[]): Reply {
514 switch (verb) {
515 case 'BEGIN': this.stack.push(structuredClone(this.database)); break;
516 case 'COMMIT': case 'ROLLBACK':
517 if (this.stack.length === 0) fail('ERR no transaction');
518 if (verb === 'ROLLBACK') { this.database = this.stack.pop()!; this.purge(); }
519 else this.stack.pop();
520 break;
521 case 'SNAPSHOT': this.snapshots.set(args[0], structuredClone(this.database)); break;
522 case 'RESTORE': {
523 const snapshot = this.snapshots.get(args[0]);
524 if (snapshot === undefined) fail('ERR no such snapshot');
525 this.database = structuredClone(snapshot); this.purge();
526 break;
527 }
528 }
529 return status('OK');
530 }
