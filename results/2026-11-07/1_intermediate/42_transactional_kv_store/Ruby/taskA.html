37def quoted(text)
38 text.nil? ? '(nil)' : '"' + text.gsub(/["\\]/) { |c| '\\' + c } + '"'
39end
40def integer_reply(value) = "(integer) #{value}"
41def array_reply(values)
42 return '(empty array)' if values.empty?
43 width = values.length.to_s.length
44 values.each_with_index.map { |value, i| "#{(i + 1).to_s.rjust(width)}) #{quoted(value)}" }.join("\n")
45end

47def split_line(line)
48 words = []
49 position = 0
50 while position < line.length
51 if line[position] == ' '
52 position += 1
53 next
54 end
55 if line[position] != '"'
56 start = position
57 position += 1 while position < line.length && line[position] != ' '
58 words << line[start...position].freeze
59 next
60 end
61 position += 1
62 word = +''
63 while position < line.length && line[position] != '"'
64 character = line[position]
65 position += 1
66 if character == '\\'
67 return nil if position == line.length
68 character = line[position]
69 position += 1
70 return nil unless character == '\\' || character == '"'
71 end
72 word << character
73 end
74 return nil if position == line.length
75 position += 1
76 return nil if position < line.length && line[position] != ' '
77 words << word.freeze
78 end
79 words
80end

82def parse_integer(text)
83 text.match?(/\A(?:0|-?[1-9][0-9]{0,8})\z/) ? text.to_i(10) : nil
84end
85def number(text, message = NOT_INTEGER)
86 parse_integer(text) || raise(CommandError, message)
87end
88def increment(old, delta)
89 value = old + delta
90 raise CommandError, 'ERR increment or decrement would overflow' unless (-999_999_999..999_999_999).cover?(value)
91 value
92end
93def normalized_range(length, start, stop)
94 start += length if start.negative?
95 stop += length if stop.negative?
96 start = [0, start].max
97 stop = [length - 1, stop].min
98 start <= stop ? (start..stop) : (0...0)
99end

101ScoreBound = Data.define(:value, :exclusive) do
102 def accepts?(score, lower)
103 return lower if value == :negative_infinity
104 return !lower if value == :positive_infinity
105 (lower ? score > value : score < value) || (!exclusive && score == value)
106 end
107end
108def score_bound(text)
109 return ScoreBound.new(:negative_infinity, false) if text == '-inf'
110 return ScoreBound.new(:positive_infinity, false) if text == '+inf'
111 exclusive = text.start_with?('(')
112 value = number(exclusive ? text[1..] : text, 'ERR min or max is not a float')
113 ScoreBound.new(value, exclusive)
114end

116PatternPart = Data.define(:kind, :value, :negated) do
117 def accepts?(character)
118 case kind
119 when :any then true
120 when :literal then value == character
121 else value.include?(character.ord) != negated
122 end
123 end
124end
125def parse_pattern(text)
126 parts = []
127 i = 0
128 while i < text.length
129 case (character = text[i])
130 when '\\'
131 i += 1
132 raise CommandError, 'ERR invalid pattern' if i == text.length
133 parts << PatternPart.new(:literal, text[i], false)
134 when '['
135 finish = text.index(']', i + 1)
136 raise CommandError, 'ERR invalid pattern' unless finish
137 i += 1
138 negated = i < finish && text[i] == '^'
139 i += 1 if negated
140 raise CommandError, 'ERR invalid pattern' if i == finish
141 characters = Set.new
142 while i < finish
143 first = text[i].ord
144 if i + 2 < finish && text[i + 1] == '-'
145 last = text[i + 2].ord
146 ([first, last].min..[first, last].max).each { |code| characters.add(code) }
147 i += 3
148 else
149 characters.add(first)
150 i += 1
151 end
152 end
153 parts << PatternPart.new(:class, characters, negated)
154 when '*' then parts << PatternPart.new(:star, nil, false)
155 when '?' then parts << PatternPart.new(:any, nil, false)
156 else parts << PatternPart.new(:literal, character, false)
157 end
158 i += 1
159 end
160 parts
161end
162def matches_pattern?(parts, text)
163 previous = Array.new(text.length + 1, false)
164 previous[0] = true
165 parts.each do |part|
166 current = Array.new(previous.length, false)
167 if part.kind == :star
168 current[0] = previous[0]
169 (1...current.length).each { |i| current[i] = previous[i] || current[i - 1] }
170 else
171 text.each_char.with_index { |character, i| current[i + 1] = previous[i] && part.accepts?(character) }
172 end
173 previous = current
174 end
175 previous[text.length]
176end

178CommandSpec = Data.define(:group, :minimum, :maximum, :parity) do
179 def accepts?(count)
180 count >= minimum && (maximum.nil? || count <= maximum) && (parity.nil? || count % 2 == parity)
181 end
182end

231 def execute(words)
232 command = words.first.upcase
233 spec = COMMANDS[command]
234 raise CommandError, "ERR unknown command '#{words.first}'" unless spec
235 args = words.drop(1)
236 raise CommandError, "ERR wrong number of arguments for '#{command.downcase}' command" unless spec.accepts?(args.length)
237 send(spec.group, command, args)
238 end

242 def read(key, type, empty)
243 entry = @database[key]
244 return empty unless entry
245 raise CommandError, WRONG_TYPE unless entry.value.is_a?(type)
246 entry.value
247 end
248 def text(key) = read(key, TextValue, TextValue.new('')).text
249 def list(key) = read(key, ListValue, ListValue.new([])).items.dup
250 def hash(key) = read(key, HashValue, HashValue.new({})).fields.dup
251 def members(key) = read(key, SetValue, SetValue.new(Set.new)).members.dup
252 def sorted_set(key) = read(key, SortedSetValue, SortedSetValue.new({})).scores.dup
253 def replace(key, value, expiry = nil)
254 if value.empty? then @database.delete(key)
255 else @database[key] = Entry.new(value, expiry)
256 end
257 end
258 def update(key, value) = replace(key, value, @database[key]&.expiry)
259 def purge
260 before = @database.length
261 @database.delete_if { |_, entry| !entry.expiry.nil? && entry.expiry <= @time }
262 before - @database.length
263 end

265 def text_command(command, a)
266 key = a.first
267 case command
268 when 'SET'
269 options = Set.new
270 seconds_text = nil
271 i = 2
272 while i < a.length
273 option = a[i].upcase
274 i += 1
275 raise CommandError, 'ERR syntax error' unless %w[NX XX EX KEEPTTL GET].include?(option) && options.add?(option)
276 if option == 'EX'
277 raise CommandError, 'ERR syntax error' if i == a.length
278 seconds_text = a[i]
279 i += 1
280 end
281 end
282 if (options.include?('NX') && options.include?('XX')) || (options.include?('EX') && options.include?('KEEPTTL'))
283 raise CommandError, 'ERR syntax error'
284 end
285 expiry = nil
286 unless seconds_text.nil?
287 seconds = number(seconds_text)
288 raise CommandError, "ERR invalid expire time in 'set' command" unless seconds.positive?
289 expiry = @time + seconds
290 end
291 old = @database[key]
292 old_text = nil
293 if options.include?('GET')
294 value = text(key)
295 old_text = value if old
296 end
297 blocked = (options.include?('NX') && old) || (options.include?('XX') && !old)
298 unless blocked
299 expiry = old.expiry if options.include?('KEEPTTL') && old
300 replace(key, TextValue.new(a[1]), expiry)
301 end
302 options.include?('GET') ? quoted(old_text) : blocked ? '(nil)' : 'OK'
303 when 'MSET'
304 a.each_slice(2) { |name, value| replace(name, TextValue.new(value)) }
305 'OK'
306 when 'MGET'
307 array_reply(a.map { |name| (entry = @database[name]) && entry.value.is_a?(TextValue) ? entry.value.text : nil })
308 when 'INCR', 'DECR', 'INCRBY', 'DECRBY'
309 delta = a.length == 2 ? number(a[1]) : 1
310 delta = -delta if command.start_with?('DECR')
311 old = text(key)
312 value = increment(@database.key?(key) ? number(old) : 0, delta)
313 update(key, TextValue.new(value.to_s))
314 integer_reply(value)
315 when 'APPEND'
316 value = text(key) + a[1]
317 update(key, TextValue.new(value))
318 integer_reply(value.length)
319 when 'GETRANGE'
320 start = number(a[1])
321 stop = number(a[2])
322 value = text(key)
323 quoted(value[normalized_range(value.length, start, stop)])
324 when 'STRLEN' then integer_reply(text(key).length)
325 else
326 value = text(key)
327 quoted(@database.key?(key) ? value : nil)
328 end
329 end

331 def list_command(command, a)
332 key = a.first
333 case command
334 when 'LPUSH', 'RPUSH'
335 values = list(key)
336 a.drop(1).each { |value| command == 'LPUSH' ? values.unshift(value) : values.push(value) }
337 update(key, ListValue.new(values))
338 integer_reply(values.length)
339 when 'LPOP', 'RPOP'
340 count = a.length == 2 ? number(a[1], POSITIVE) : nil
341 raise CommandError, POSITIVE if count && count.negative?
342 values = list(key)
343 return '(nil)' unless @database.key?(key)
344 removed = Array.new([count || 1, values.length].min) { command == 'LPOP' ? values.shift : values.pop }
345 update(key, ListValue.new(values))
346 count.nil? ? quoted(removed.first) : array_reply(removed)
347 when 'LRANGE', 'LTRIM'
348 start = number(a[1])
349 stop = number(a[2])
350 values = list(key)
351 kept = values[normalized_range(values.length, start, stop)]
352 return array_reply(kept) if command == 'LRANGE'
353 update(key, ListValue.new(kept))
354 'OK'
355 when 'LLEN' then integer_reply(list(key).length)
356 when 'LINDEX', 'LSET'
357 index = number(a[1])
358 values = list(key)
359 index += values.length if index.negative?
360 valid = index >= 0 && index < values.length
361 return quoted(valid ? values[index] : nil) if command == 'LINDEX'
362 raise CommandError, 'ERR no such key' unless @database.key?(key)
363 raise CommandError, 'ERR index out of range' unless valid
364 values[index] = a[2]
365 update(key, ListValue.new(values))
366 'OK'
367 when 'LREM'
368 count = number(a[1])
369 values = list(key)
370 values.reverse! if count.negative?
371 removed = 0
372 kept = values.reject do |value|
373 if value == a[2] && (count.zero? || removed < count.abs) then removed += 1; true
374 else false
375 end
376 end
377 kept.reverse! if count.negative?
378 update(key, ListValue.new(kept))
379 integer_reply(removed)
380 else
381 where = a[1].upcase
382 raise CommandError, 'ERR syntax error' unless %w[BEFORE AFTER].include?(where)
383 values = list(key)
384 return integer_reply(0) unless @database.key?(key)
385 index = values.index(a[2])
386 return integer_reply(-1) unless index
387 values.insert(index + (where == 'AFTER' ? 1 : 0), a[3])
388 update(key, ListValue.new(values))
389 integer_reply(values.length)
390 end
391 end

393 def hash_command(command, a)
394 key = a.first
395 if command == 'HINCRBY'
396 delta = number(a[2])
397 fields = hash(key)
398 old = fields.key?(a[1]) ? number(fields[a[1]], 'ERR hash value is not an integer') : 0
399 value = increment(old, delta)
400 fields[a[1]] = value.to_s
401 update(key, HashValue.new(fields))
402 return integer_reply(value)
403 end
404 fields = hash(key)
405 case command
406 when 'HSET'
407 added = 0
408 a.drop(1).each_slice(2) do |field, value|
409 added += 1 unless fields.key?(field)
410 fields[field] = value
411 end
412 update(key, HashValue.new(fields))
413 integer_reply(added)
414 when 'HDEL'
415 removed = a.drop(1).count { |field| fields.delete(field) }
416 update(key, HashValue.new(fields))
417 integer_reply(removed)
418 when 'HGET' then quoted(fields[a[1]])
419 when 'HEXISTS' then integer_reply(fields.key?(a[1]) ? 1 : 0)
420 when 'HLEN' then integer_reply(fields.length)
421 when 'HKEYS' then array_reply(fields.keys.sort)
422 when 'HVALS' then array_reply(fields.keys.sort.map { |field| fields[field] })
423 else array_reply(fields.keys.sort.flat_map { |field| [field, fields[field]] })
424 end
425 end

427 def set_command(command, a)
428 key = a.first
429 storing = command.end_with?('STORE')
430 if command.start_with?('SINTER', 'SUNION', 'SDIFF')
431 sources = a.drop(storing ? 1 : 0).map { |source| members(source) }
432 result = sources.first.dup
433 sources.drop(1).each do |source|
434 result = if command.start_with?('SINTER') then result & source
435 elsif command.start_with?('SUNION') then result | source
436 else result - source
437 end
438 end
439 return array_reply(result.to_a.sort) unless storing
440 replace(key, SetValue.new(result))
441 return integer_reply(result.length)
442 end
443 values = members(key)
444 case command
445 when 'SADD', 'SREM'
446 changed = a.drop(1).count { |member| command == 'SADD' ? values.add?(member) : values.delete?(member) }
447 update(key, SetValue.new(values))
448 integer_reply(changed)
449 when 'SCARD' then integer_reply(values.length)
450 when 'SISMEMBER' then integer_reply(values.include?(a[1]) ? 1 : 0)
451 else array_reply(values.to_a.sort)
452 end
453 end

455 def sorted_set_command(command, a)
456 key = a.first
457 if command == 'ZADD'
458 numbers = a.drop(1).each_slice(2).map { |score, _| number(score) }
459 scores = sorted_set(key)
460 added = 0
461 numbers.each_with_index do |score, i|
462 member = a[2 * i + 2]
463 added += 1 unless scores.key?(member)
464 scores[member] = score
465 end
466 update(key, SortedSetValue.new(scores))
467 return integer_reply(added)
468 end
469 if command == 'ZINCRBY'
470 delta = number(a[1])
471 scores = sorted_set(key)
472 value = increment(scores.fetch(a[2], 0), delta)
473 scores[a[2]] = value
474 update(key, SortedSetValue.new(scores))
475 return quoted(value.to_s)
476 end
477 with_scores = false
478 start = stop = 0
479 minimum = maximum = nil
480 if command == 'ZRANGE' || command == 'ZRANGEBYSCORE'
481 with_scores = a.length == 4
482 raise CommandError, 'ERR syntax error' if with_scores && a[3].upcase != 'WITHSCORES'
483 if command == 'ZRANGE'
484 start = number(a[1])
485 stop = number(a[2])
486 else
487 minimum = score_bound(a[1])
488 maximum = score_bound(a[2])
489 end
490 end
491 scores = sorted_set(key)
492 case command
493 when 'ZREM'
494 removed = a.drop(1).count { |member| scores.delete(member) }
495 update(key, SortedSetValue.new(scores))
496 return integer_reply(removed)
497 when 'ZCARD' then return integer_reply(scores.length)
498 when 'ZSCORE' then return quoted(scores[a[1]]&.to_s)
499 end
500 ordered = scores.sort_by { |member, score| [score, member] }
501 if command == 'ZRANK'
502 index = ordered.index { |member, _| member == a[1] }
503 return index.nil? ? '(nil)' : integer_reply(index)
504 end
505 selected = if command == 'ZRANGE'
506 ordered[normalized_range(ordered.length, start, stop)]
507 else
508 ordered.select { |_, score| minimum.accepts?(score, true) && maximum.accepts?(score, false) }
509 end
510 array_reply(with_scores ? selected.flat_map { |member, score| [member, score.to_s] } : selected.map(&:first))
511 end

513 def key_command(command, a)
514 case command
515 when 'TIME' then return integer_reply(@time)
516 when 'DBSIZE' then return integer_reply(@database.length)
517 when 'FLUSH' then @database.clear; return 'OK'
518 when 'INFO'
519 counts = Hash.new(0)
520 @database.each_value { |entry| counts[entry.value.kind] += 1 }
521 expires = @database.count { |_, entry| !entry.expiry.nil? }
522 return "keys=#{@database.length},expires=#{expires},string=#{counts[:string]},list=#{counts[:list]},hash=#{counts[:hash]}," \
523 "set=#{counts[:set]},zset=#{counts[:zset]},time=#{@time},depth=#{@stack.length},snapshots=#{@snapshots.length}"
524 when 'DEL' then return integer_reply(a.count { |key| @database.delete(key) })
525 when 'EXISTS' then return integer_reply(a.count { |key| @database.key?(key) })
526 when 'KEYS'
527 parts = parse_pattern(a.first)
528 return array_reply(@database.keys.sort.select { |key| matches_pattern?(parts, key) })
529 when 'TICK'
530 ticks = number(a.first)
531 raise CommandError, POSITIVE unless ticks.positive?
532 @time += ticks
533 return integer_reply(purge)
534 end
535 key = a.first
536 if command == 'EXPIRE'
537 condition = a.length == 3 ? a[2].upcase : nil
538 if condition && !%w[NX XX GT LT].include?(condition)
539 raise CommandError, "ERR Unsupported option #{a[2]}"
540 end
541 seconds = number(a[1])
542 entry = @database[key]
543 return integer_reply(0) unless entry
544 expiry = @time + seconds
545 rejected = case condition
546 when 'NX' then !entry.expiry.nil?
547 when 'XX' then entry.expiry.nil?
548 when 'GT' then entry.expiry.nil? || expiry <= entry.expiry
549 when 'LT' then !entry.expiry.nil? && expiry >= entry.expiry
550 else false
551 end
552 return integer_reply(0) if rejected
553 if seconds <= 0 then @database.delete(key)
554 else @database[key] = Entry.new(entry.value, expiry)
555 end
556 return integer_reply(1)
557 end
558 entry = @database[key]
559 case command
560 when 'TYPE' then entry ? entry.value.kind.to_s : 'none'
561 when 'TTL' then integer_reply(entry.nil? ? -2 : entry.expiry.nil? ? -1 : entry.expiry - @time)
562 when 'PERSIST'
563 return integer_reply(0) if entry.nil? || entry.expiry.nil?
564 @database[key] = Entry.new(entry.value, nil)
565 integer_reply(1)
566 else
567 raise CommandError, 'ERR no such key' unless entry
568 destination = a[1]
569 return integer_reply(0) if command == 'RENAMENX' && @database.key?(destination)
570 if key != destination
571 @database.delete(key)
572 @database[destination] = entry
573 end
574 command == 'RENAME' ? 'OK' : integer_reply(1)
575 end
576 end

578 def control_command(command, a)
579 case command
580 when 'BEGIN' then @stack << @database.dup
581 when 'COMMIT', 'ROLLBACK'
582 raise CommandError, 'ERR no transaction' if @stack.empty?
583 saved = @stack.pop
584 if command == 'ROLLBACK'
585 @database = saved
586 purge
587 end
588 when 'SNAPSHOT' then @snapshots[a.first] = @database.dup
589 when 'RESTORE'
590 saved = @snapshots[a.first]
591 raise CommandError, 'ERR no such snapshot' unless saved
592 @database = saved.dup
593 purge
594 end
595 'OK'
596 end
597end
