33func lower(c byte) bool { return c >= 'a' && c <= 'z' }
34func upper(c byte) bool { return c >= 'A' && c <= 'Z' }
35func letter(c byte) bool { return lower(c) || upper(c) }
36func digit(c byte) bool { return c >= '0' && c <= '9' }
37func lowercase(c byte) byte {
38 if upper(c) { return c + 'a' - 'A' }
39 return c
40}
41func uppercase(c byte) byte {
42 if lower(c) { return c + 'A' - 'a' }
43 return c
44}
45func plain(text string) string { return strings.NewReplacer("{", "", "}", "").Replace(text) }
46func letters(text string) string {
47 var result strings.Builder
48 for i := range len(text) { if letter(text[i]) { result.WriteByte(lowercase(text[i])) } }
49 return result.String()
50}
51func allDigits(text string) bool {
52 if text == "" { return false }
53 for i := range len(text) { if !digit(text[i]) { return false } }
54 return true
55}
56func splitDepth(text, separator string) []string {
57 var result []string
58 start, i, depth := 0, 0, 0
59 for i < len(text) {
60 if depth == 0 && strings.HasPrefix(text[i:], separator) {
61 result = append(result, text[start:i]); i += len(separator); start = i
62 } else {
63 if text[i] == '{' { depth++ }
64 if text[i] == '}' { depth-- }
65 i++
66 }
67 }
68 return append(result, text[start:])
69}
70func evaluate(expression string, macros map[string]string) (string, error) {
71 depth := 0
72 for i := range len(expression) {
73 if expression[i] == '{' { depth++ }
74 if expression[i] == '}' { depth-- }
75 if depth < 0 { return "", fmt.Errorf("unbalanced braces") }
76 }
77 if depth != 0 { return "", fmt.Errorf("unbalanced braces") }
78 var result strings.Builder
79 for _, term := range strings.Split(expression, " # ") {
80 if term[0] == '{' || term[0] == '"' { result.WriteString(term[1:len(term)-1])
81 } else if digit(term[0]) { result.WriteString(term)
82 } else if value, ok := macros[term]; ok { result.WriteString(value)
83 } else { return "", fmt.Errorf("unknown macro '%s'", term) }
84 }
85 return result.String(), nil
86}
87func accent(text string, at int) (byte, int) {
88 if at+1 >= len(text) || text[at] != '\\' { return 0, 0 }
89 command := text[at+1]
90 if command == '&' { return '&', 2 }
91 symbol := strings.ContainsRune("'\"`^~=.", rune(command))
92 if symbol && at+2 < len(text) && letter(text[at+2]) { return text[at+2], 3 }
93 if (symbol || strings.ContainsRune("cuvH", rune(command))) && at+4 < len(text) && text[at+2] == '{' && letter(text[at+3]) && text[at+4] == '}' { return text[at+3], 5 }
94 return 0, 0
95}
96func normalize(value string) string {
97 var result strings.Builder
98 for i := 0; i < len(value); {
99 if value[i] == '{' {
100 c, width := accent(value, i+1)
101 if width > 0 && i+width+1 < len(value) && value[i+width+1] == '}' { result.WriteByte(c); i += width+2; continue }
102 }
103 if c, width := accent(value, i); width > 0 { result.WriteByte(c); i += width
104 } else { result.WriteByte(value[i]); i++ }
105 }
106 return strings.Join(strings.Fields(strings.ReplaceAll(result.String(), "~", " ")), " ")
107}

112func parseName(text string) Name {
113 parts := splitDepth(text, ",")
114 for i := range parts { parts[i] = strings.Trim(parts[i], " ") }
115 words := splitDepth(parts[0], " ")
116 firstLower, lastLower := -1, -1
117 for i := 0; i+1 < len(words); i++ {
118 if lower(words[i][0]) {
119 if firstLower < 0 { firstLower = i }
120 lastLower = i
121 }
122 }
123 var name Name
124 if len(parts) == 1 {
125 if firstLower < 0 { name.First = strings.Join(words[:len(words)-1], " "); name.Last = words[len(words)-1]
126 } else { name.First = strings.Join(words[:firstLower], " "); name.Von = strings.Join(words[firstLower:lastLower+1], " "); name.Last = strings.Join(words[lastLower+1:], " ") }
127 } else {
128 if lastLower < 0 { name.Last = strings.Join(words, " ")
129 } else { name.Von = strings.Join(words[:lastLower+1], " "); name.Last = strings.Join(words[lastLower+1:], " ") }
130 name.First = parts[len(parts)-1]
131 if len(parts) == 3 { name.Jr = parts[1] }
132 }
133 return name
134}
135func parseNames(text string) *NameList {
136 result := &NameList{}
137 for _, name := range splitDepth(text, " and ") {
138 if name == "others" { result.EtAl = true } else { result.Names = append(result.Names, parseName(name)) }
139 }
140 return result
141}
142func (a *NameList) Equal(b *NameList) bool {
143 if len(a.Names) != len(b.Names) || a.EtAl != b.EtAl { return false }
144 for i := range a.Names { if a.Names[i] != b.Names[i] { return false } }
145 return true
146}
147func (a *NameList) Single() bool { return len(a.Names) == 1 && !a.EtAl }
148func (name Name) Surname() string {
149 if name.Von == "" { return plain(name.Last) }
150 return plain(name.Von + " " + name.Last)
151}
152func (name Name) Initials() string {
153 var words []string
154 for _, word := range strings.Fields(plain(name.First)) {
155 var initials []string
156 for _, piece := range strings.Split(word, "-") {
157 for i := range len(piece) { if letter(piece[i]) { initials = append(initials, string(uppercase(piece[i]))+"."); break } }
158 }
159 words = append(words, strings.Join(initials, "-"))
160 }
161 return strings.Join(words, " ")
162}
163func (name Name) Text(inverted, abbreviated bool) string {
164 given, result := plain(name.First), name.Surname()
165 if abbreviated { given = name.Initials() }
166 if name.First != "" {
167 if inverted { result += ", " + given } else { result = given + " " + result }
168 }
169 if name.Jr != "" { result += ", " + plain(name.Jr) }
170 return result
171}

175 APA Style = iota
176 IEEE

180func titleCase(title string, style Style) string {
181 if style == IEEE { return title }
182 if style == APA {
183 result, depth := []byte(title), 0
184 for i, c := range result {
185 switch {
186 case c == '{': depth++
187 case c == '}': depth--
188 case depth == 0 && letter(c):
189 if i == 0 || i >= 2 && title[i-2:i] == ": " { result[i] = uppercase(c) } else { result[i] = lowercase(c) }
190 }
191 }
192 return string(result)
193 }
194 words := splitDepth(title, " ")
195 for i, word := range words {
196 if word[0] == '{' { continue }
197 makeLower := i > 0 && i+1 < len(words) && !strings.HasSuffix(words[i-1], ":") && indexOf(smallWords, letters(word)) >= 0
198 result, depth, firstLetter := []byte(word), 0, true
199 for j, c := range result {
200 switch {
201 case c == '{': depth++
202 case c == '}': depth--
203 case letter(c):
204 if depth == 0 && makeLower { result[j] = lowercase(c) } else if depth == 0 && firstLetter { result[j] = uppercase(c) }
205 firstLetter = false
206 }
207 }
208 words[i] = string(result)
209 }
210 return strings.Join(words, " ")
211}
212func listed(names []string, conjunction string, listB bool) string {
213 if len(names) == 1 { return names[0] }
214 if listB && len(names) == 2 { return names[0]+" "+conjunction+" "+names[1] }
215 return strings.Join(names[:len(names)-1], ", ")+", "+conjunction+" "+names[len(names)-1]
216}
217func renderNames(list *NameList, style Style, editor bool) string {
218 names := make([]string, len(list.Names))
219 for i, name := range list.Names { names[i] = name.Text(!editor && (style == APA || style != IEEE && i == 0), style == APA || style == IEEE) }
220 if editor {
221 if list.EtAl { return names[0]+" et al." }
222 conjunction := "and"
223 if style == APA { conjunction = "&" }
224 return listed(names, conjunction, true)
225 }
226 switch style {
227 case APA:
228 if list.EtAl { return strings.Join(names, ", ")+", et al." }
229 if len(names) >= 21 { return strings.Join(names[:19], ", ")+", . . . "+names[len(names)-1] }
230 return listed(names, "&", false)
231 case IEEE:
232 if list.EtAl || len(names) >= 7 { return names[0]+" et al." }
233 return listed(names, "and", true)
234 case Chicago:
235 if list.EtAl || len(names) >= 11 { return strings.Join(names[:min(len(names), 7)], ", ")+", et al." }
236 return listed(names, "and", false)
237 default:
238 if list.EtAl || len(names) >= 3 { return names[0]+", et al." }
239 return listed(names, "and", false)
240 }

255func newEntry(key string, kind int, fields map[string]string) *Entry {
256 entry := &Entry{Key: key, Kind: kind, Fields: fields}
257 if value, ok := fields["author"]; ok { entry.Authors = parseNames(value) }
258 if value, ok := fields["editor"]; ok { entry.Editors = parseNames(value) }
259 lead := entry.Primary().Names[0]
260 entry.SortLast = strings.ToLower(plain(lead.Last))
261 entry.Year = plain(fields["year"])
262 entry.PlainTitle = plain(fields["title"])
263 entry.LowerTitle = strings.ToLower(entry.PlainTitle)
264 entry.SortTitle = entry.LowerTitle
265 for _, prefix := range []string{"a ", "an ", "the "} {
266 if strings.HasPrefix(entry.SortTitle, prefix) { entry.SortTitle = strings.TrimPrefix(entry.SortTitle, prefix); break }
267 }
268 var base strings.Builder
269 base.WriteString(letters(plain(lead.Last)))
270 for _, c := range fields["year"] { if c >= '0' && c <= '9' { base.WriteRune(c) } }
271 for _, word := range strings.Split(entry.PlainTitle, " ") {
272 word = letters(word)
273 if word != "" && indexOf(smallWords, word) < 0 { base.WriteString(word); break }
274 }
275 entry.BaseKey = base.String()
276 return entry
277}

282 return a.Key < b.Key

336func optionalEnd(pattern string, start int) int {
337 depth := 1
338 for i := start+1; ; i++ {
339 if pattern[i] == '<' { depth++ }
340 if pattern[i] == '>' { depth-- }
341 if depth == 0 { return i }
342 }
343}
344func optionalPresent(pattern string, start, end int, values map[string]string) bool {
345 for i := start; i < end; i++ {
346 if pattern[i] == '<' { i = optionalEnd(pattern, i)
347 } else if pattern[i] == '{' {
348 close := i+strings.IndexByte(pattern[i:], '}')
349 if values[pattern[i+1:close]] == "" { return false }
350 i = close
351 }
352 }
353 return true
354}
355func renderTemplate(pattern string, start, end int, values map[string]string, result *strings.Builder) {
356 for i := start; i < end; i++ {
357 switch pattern[i] {
358 case '<':
359 close := optionalEnd(pattern, i)
360 if optionalPresent(pattern, i+1, close, values) { renderTemplate(pattern, i+1, close, values, result) }
361 i = close
362 case '{':
363 close := i+strings.IndexByte(pattern[i:], '}')
364 result.WriteString(values[pattern[i+1:close]]); i = close
365 default:
366 if pattern[i] != '.' || result.Len() == 0 || !strings.ContainsRune(".?!", rune(result.String()[result.Len()-1])) { result.WriteByte(pattern[i]) }
367 }
368 }
369}
370func (entry *Entry) Reference(style Style, number int, repeated bool) string {
371 values := make(map[string]string)
372 for _, name := range fieldOrder { values[name] = plain(entry.Fields[name]) }
373 list := entry.Primary()
374 if repeated { values["author"] = "---" } else { values["author"] = renderNames(list, style, false) }
375 if entry.Authors == nil {
376 annotations := [4][2]string{{" (Ed.)", " (Eds.)"}, {", Ed.", ", Eds."}, {", ed.", ", eds."}, {", editor", ", editors"}}
377 index := 1
378 if list.Single() { index = 0 }
379 values["author"] += annotations[style][index]
380 }
381 if entry.Editors != nil {
382 values["editor"] = renderNames(entry.Editors, style, true)
383 values["eds"] = "Eds."
384 if entry.Editors.Single() { values["eds"] = "Ed." }
385 }
386 values["title"] = plain(titleCase(entry.Fields["title"], style))
387 values["year"] = entry.YearLabel(style)
388 if pages, ok := entry.Fields["pages"]; ok {
389 values["pages"] = strings.ReplaceAll(plain(pages), "--", "-")
390 prefix := "p. "
391 if strings.Contains(values["pages"], "-") { prefix = "pp. " }
392 values["pp"] = prefix+values["pages"]
393 }
394 if edition, ok := entry.Fields["edition"]; ok {
395 value := plain(edition)
396 if allDigits(value) {
397 n, _ := strconv.Atoi(value)
398 suffix := "th"
399 if n%100 < 11 || n%100 > 13 {
400 switch n%10 { case 1: suffix = "st"; case 2: suffix = "nd"; case 3: suffix = "rd" }
401 }
402 value = fmt.Sprintf("%d%s", n, suffix)
403 }
404 values["edition"] = value+" ed."
405 }
406 if month, ok := entry.Fields["month"]; ok {
407 value := plain(month)
408 if i := indexOf(months, value); i >= 0 { value = monthShort[i] }
409 values["mon"] = value
410 }
411 kind := entry.Kind
412 if kind == 3 { kind = 2 } else if kind > 3 { kind-- }
413 pattern := templates[style][kind]
414 var result strings.Builder
415 if style == IEEE { fmt.Fprintf(&result, "[%d] ", number) }
416 renderTemplate(pattern, 0, len(pattern), values, &result)
417 return result.String()
418}

420type Database struct {

426type LoadError struct { Line int; Reason string }
427func readDatabase(lines []string) *Database {
428 database := &Database{Lookup: make(map[string]*Entry)}
429 macros := make(map[string]string)
430 for i, name := range monthMacros { macros[name] = months[i] }
431 macroCount, errors := 0, 0
432 for i := 0; i < len(lines); {
433 line := lines[i]
434 if line == "" || line[0] == '%' { i++; continue }
435 if strings.HasPrefix(line, "@string{") {
436 name, expression, _ := strings.Cut(line[8:len(line)-1], " = ")
437 value, err := evaluate(expression, macros)
438 if _, exists := macros[name]; err == nil && exists { err = fmt.Errorf("duplicate macro") }
439 if err != nil { database.Output = append(database.Output, fmt.Sprintf("ERROR line %d: %s: %s", i+1, name, err)); errors++
440 } else { macros[name] = value; macroCount++ }
441 i++; continue
442 }
443 kindName, key, _ := strings.Cut(line[1:len(line)-1], "{")
444 kind := indexOf(typeNames, kindName)
445 var loadError *LoadError
446 if kind < 0 { loadError = &LoadError{i+1, fmt.Sprintf("unknown type '%s'", kindName)}
447 } else if _, exists := database.Lookup[key]; exists { loadError = &LoadError{i+1, "duplicate key"} }
448 fields := make(map[string]string)
449 i++
450 for lines[i] != "}" {
451 if loadError == nil {
452 name, expression, _ := strings.Cut(lines[i][2:], " = ")
453 value, err := evaluate(strings.TrimSuffix(expression, ","), macros)
454 if _, exists := fields[name]; err == nil && exists { err = fmt.Errorf("duplicate field '%s'", name) }
455 if err != nil { loadError = &LoadError{i+1, err.Error()} } else { fields[name] = normalize(value) }
456 }
457 i++
458 }
459 if loadError == nil {
460 for _, name := range required[kind] {
461 _, present := fields[name]
462 _, editor := fields["editor"]
463 if !present && !(kind == 1 && name == "author" && editor) { loadError = &LoadError{i+1, fmt.Sprintf("missing field '%s'", name)}; break }
464 }
465 }
466 if loadError != nil { database.Output = append(database.Output, fmt.Sprintf("ERROR line %d: %s: %s", loadError.Line, key, loadError.Reason)); errors++
467 } else { entry := newEntry(key, kind, fields); database.Entries = append(database.Entries, entry); database.Lookup[key] = entry }
468 i++
469 }
470 database.Output = append(database.Output, fmt.Sprintf("LOADED entries=%d macros=%d errors=%d", len(database.Entries), macroCount, errors))
471 assigned := make(map[*Entry]bool)
472 for i, entry := range database.Entries {
473 if assigned[entry] { continue }
474 var group []*Entry
475 for _, other := range database.Entries[i:] {
476 if entry.Primary().Equal(other.Primary()) && entry.Fields["year"] == other.Fields["year"] { group = append(group, other); assigned[other] = true }
477 }
478 if len(group) >= 2 {
479 sort.Slice(group, func(i, j int) bool {
480 if group[i].SortTitle != group[j].SortTitle { return group[i].SortTitle < group[j].SortTitle }
481 return group[i].Key < group[j].Key
482 })
483 for j, other := range group { other.Suffix = string(rune('a'+j)) }
484 }
485 }
486 return database
487}
488func (database *Database) IEEEOrder() ([]*Entry, map[*Entry]int) {
489 cited := make(map[*Entry]bool)
490 order := append([]*Entry(nil), database.Record...)
491 for _, entry := range order { cited[entry] = true }
492 var rest []*Entry
493 for _, entry := range database.Entries { if !cited[entry] { rest = append(rest, entry) } }
494 sort.Slice(rest, func(i, j int) bool { return rest[i].Key < rest[j].Key })
495 order = append(order, rest...)
496 numbers := make(map[*Entry]int)
497 for i, entry := range order { numbers[entry] = i+1 }
498 return order, numbers
499}

501type CitationKind int
502const (
503 Parenthetical CitationKind = iota
504 Narrative
505 Author
506 Year
507)
508type Citation struct { Start, End int; Kind CitationKind; Entries []*Entry }
509func citationText(citation Citation, style Style, numbers map[*Entry]int) string {
510 parenthetical := citation.Kind == Parenthetical
511 var parts []string
512 if citation.Kind == Author || citation.Kind == Year || style == MLA || style == IEEE && !parenthetical {
513 for _, entry := range citation.Entries {
514 value := entry.ShortName(style, false)
515 if citation.Kind == Year { value = entry.YearLabel(style)
516 } else if style == IEEE && citation.Kind == Narrative { value = fmt.Sprintf("%s [%d]", value, numbers[entry]) }
517 parts = append(parts, value)
518 }
519 separator := "; "
520 if citation.Kind == Year { separator = ", " }
521 text := strings.Join(parts, separator)
522 if style == MLA && parenthetical { text = "("+text+")" }
523 return text
524 }
525 if style == IEEE {
526 var sorted []int
527 for _, entry := range citation.Entries { sorted = append(sorted, numbers[entry]) }
528 sort.Ints(sorted)
529 for i := 0; i < len(sorted); {
530 end := i+1
531 for end < len(sorted) && sorted[end] == sorted[end-1]+1 { end++ }
532 if end-i >= 3 { parts = append(parts, fmt.Sprintf("[%d]-[%d]", sorted[i], sorted[end-1]))
533 } else { for _, number := range sorted[i:end] { parts = append(parts, fmt.Sprintf("[%d]", number)) } }
534 i = end
535 }
536 return strings.Join(parts, ", ")
537 }
538 entries := append([]*Entry(nil), citation.Entries...)
539 if parenthetical { sort.Slice(entries, func(i, j int) bool { return bibliographyLess(entries[i], entries[j]) }) }
540 for i := 0; i < len(entries); {
541 name := entries[i].ShortName(style, parenthetical)
542 end := i+1
543 for end < len(entries) && entries[end].ShortName(style, parenthetical) == name { end++ }
544 var years []string
545 for _, entry := range entries[i:end] { years = append(years, entry.YearLabel(style)) }
546 if parenthetical {
547 separator := " "
548 if style == APA { separator = ", " }
549 parts = append(parts, name+separator+strings.Join(years, ", "))
550 } else { parts = append(parts, name+" ("+strings.Join(years, ", ")+")") }
551 i = end
552 }
553 text := strings.Join(parts, "; ")
554 if parenthetical { text = "("+text+")" }
555 return text
556}
557func (database *Database) Cite(style Style, text string) (string, error) {
558 var citations []Citation
559 for position := 0; position < len(text); {
560 offset := strings.IndexByte(text[position:], '\\')
561 if offset < 0 { break }
562 start := position+offset
563 brace := start+strings.IndexByte(text[start:], '{')
564 end := brace+strings.IndexByte(text[brace:], '}')
565 citation := Citation{Start: start, End: end+1}
566 switch text[start+1:brace] { case "citet": citation.Kind = Narrative; case "citeauthor": citation.Kind = Author; case "citeyear": citation.Kind = Year }
567 seen := make(map[*Entry]bool)
568 for _, key := range strings.Split(text[brace+1:end], ",") {
569 entry, exists := database.Lookup[key]
570 if !exists { return "", fmt.Errorf("unknown key '%s'", key) }
571 if !seen[entry] { citation.Entries = append(citation.Entries, entry); seen[entry] = true }
572 }
573 citations = append(citations, citation); position = end+1
574 }
575 recorded := make(map[*Entry]bool)
576 for _, entry := range database.Record { recorded[entry] = true }
577 for _, citation := range citations { for _, entry := range citation.Entries { if !recorded[entry] { database.Record = append(database.Record, entry); recorded[entry] = true } } }
578 _, numbers := database.IEEEOrder()
579 var result strings.Builder
580 position := 0
581 for _, citation := range citations {
582 result.WriteString(text[position:citation.Start]); result.WriteString(citationText(citation, style, numbers)); position = citation.End
583 }
584 result.WriteString(text[position:])
585 return result.String(), nil
586}
587func (database *Database) Keygen() []string {
588 counts, used := make(map[string]int), make(map[string]int)
589 for _, entry := range database.Entries { counts[entry.BaseKey]++ }
590 var result []string
591 for _, entry := range database.Entries {
592 key := entry.BaseKey
593 if counts[key] >= 2 { key += string(rune('a'+used[entry.BaseKey])); used[entry.BaseKey]++ }
594 result = append(result, entry.Key+" -> "+key)
595 }
596 return result
597}
598func (database *Database) Check() []string {
599 var result []string
600 for i, entry := range database.Entries {
601 var warnings []string
602 if pages, exists := entry.Fields["pages"]; exists {
603 pages = plain(pages)
604 if strings.Contains(pages, "-") && !strings.Contains(pages, "--") { warnings = append(warnings, "pages use a single hyphen") }
605 parts := strings.FieldsFunc(pages, func(c rune) bool { return c == '-' })
606 if len(parts) == 2 && allDigits(parts[0]) && allDigits(parts[1]) {
607 a, _ := strconv.Atoi(parts[0]); b, _ := strconv.Atoi(parts[1])
608 if a > b { warnings = append(warnings, "pages are reversed") }
609 }
610 }
611 if len(entry.Year) != 4 || !allDigits(entry.Year) { warnings = append(warnings, "year is not 4 digits") }
612 lowerCount, upperCount := 0, 0
613 for j := range len(entry.PlainTitle) { if lower(entry.PlainTitle[j]) { lowerCount++ }; if upper(entry.PlainTitle[j]) { upperCount++ } }
614 if lowerCount == 0 && upperCount >= 2 { warnings = append(warnings, "title is all caps") }
615 if _, exists := entry.Fields["doi"]; entry.Kind == 0 && !exists { warnings = append(warnings, "article has no doi") }
616 for _, earlier := range database.Entries[:i] { if entry.LowerTitle == earlier.LowerTitle { warnings = append(warnings, "same title as "+earlier.Key); break } }
617 for _, name := range entry.UnknownFields() { warnings = append(warnings, fmt.Sprintf("unknown field '%s'", name)) }
618 for _, warning := range warnings { result = append(result, "WARN "+entry.Key+": "+warning) }
619 }
620 return append(result, fmt.Sprintf("CHECK warnings=%d", len(result)))
621}
622func (database *Database) Export(key string) ([]string, error) {
623 entry, exists := database.Lookup[key]
624 if !exists { return nil, fmt.Errorf("unknown key '%s'", key) }
625 var names []string
626 for _, name := range fieldOrder { if _, exists := entry.Fields[name]; exists { names = append(names, name) } }
627 names = append(names, entry.UnknownFields()...)
628 result := []string{"@"+typeNames[entry.Kind]+"{"+key+","}
629 for i, name := range names {
630 line := " "+name+" = {"+entry.Fields[name]+"}"
631 if i+1 < len(names) { line += "," }
632 result = append(result, line)
633 }
634 return append(result, "}"), nil
635}
636func sortedKeys[V any](values map[string]V) []string {
637 keys := make([]string, 0, len(values))
638 for key := range values { keys = append(keys, key) }
639 sort.Strings(keys)
640 return keys
641}
642type Pair struct { A, B string }
643func (database *Database) Stats() []string {
644 types, years := make(map[string]int), make(map[string]int)
645 authors := make(map[string]bool)
646 pairs := make(map[Pair]int)
647 for _, entry := range database.Entries {
648 types[typeNames[entry.Kind]]++; years[entry.Year]++
649 if entry.Authors != nil {
650 distinct := make(map[string]bool)
651 for _, name := range entry.Authors.Names { text := name.Text(false, false); distinct[text] = true; authors[text] = true }
652 names := sortedKeys(distinct)
653 for a := range names { for b := a+1; b < len(names); b++ { pairs[Pair{names[a], names[b]}]++ } }
654 }
655 }
656 result := []string{fmt.Sprintf("ENTRIES %d", len(database.Entries))}
657 for _, kind := range sortedKeys(types) { result = append(result, fmt.Sprintf("TYPE %s %d", kind, types[kind])) }
658 for _, year := range sortedKeys(years) { result = append(result, fmt.Sprintf("YEAR %s %d", year, years[year])) }
659 result = append(result, fmt.Sprintf("AUTHORS %d", len(authors)))
660 var best Pair
661 bestCount := 0
662 for pair, count := range pairs {
663 if count > bestCount || count == bestCount && (pair.A < best.A || pair.A == best.A && pair.B < best.B) { best, bestCount = pair, count }
664 }
665 if bestCount == 0 { return append(result, "PAIR none") }
666 return append(result, fmt.Sprintf("PAIR %s & %s %d", best.A, best.B, bestCount))
667}
668func (database *Database) Command(line string) ([]string, error) {
669 command, _, _ := strings.Cut(line, " ")
670 words := strings.Split(line, " ")
671 if command == "CITE" || command == "SEARCH" { words = strings.SplitN(line, " ", 3) }
672 commandNames := []string{"FORMAT", "BIBLIOGRAPHY", "CITE", "KEYGEN", "CHECK", "SEARCH", "EXPORT", "STATS"}
673 kind := indexOf(commandNames, command)
674 if kind < 0 { return nil, fmt.Errorf("unknown command '%s'", command) }
675 arities := []int{-1, 1, 2, 0, 0, 2, 1, 0}
676 if kind == 0 && len(words) < 3 || kind != 0 && len(words) != arities[kind]+1 { return nil, fmt.Errorf("wrong number of arguments") }
677 style := APA
678 if kind <= 2 {
679 index := indexOf([]string{"APA", "IEEE", "CHICAGO", "MLA"}, words[1])
680 if index < 0 { return nil, fmt.Errorf("unknown style '%s'", words[1]) }
681 style = Style(index)
682 }
683 switch command {
684 case "FORMAT":
685 _, numbers := database.IEEEOrder()
686 var result []string
687 for _, key := range words[2:] {
688 if entry, exists := database.Lookup[key]; exists { result = append(result, entry.Reference(style, numbers[entry], false))
689 } else { result = append(result, fmt.Sprintf("ERROR unknown key '%s'", key)) }
690 }
691 return result, nil
692 case "BIBLIOGRAPHY":
693 order, numbers := database.IEEEOrder()
694 if style != IEEE { sort.Slice(order, func(i, j int) bool { return bibliographyLess(order[i], order[j]) }) }
695 heading := "References"
696 if style == MLA { heading = "Works Cited" }
697 result := []string{heading}
698 for i, entry := range order {
699 repeated := i > 0 && (style == Chicago || style == MLA) && entry.Primary().Equal(order[i-1].Primary())
700 result = append(result, entry.Reference(style, numbers[entry], repeated))
701 }
702 return result, nil
703 case "CITE":
704 text, err := database.Cite(style, words[2]); return []string{text}, err
705 case "KEYGEN": return database.Keygen(), nil
706 case "CHECK": return database.Check(), nil
707 case "EXPORT": return database.Export(words[1])
708 case "STATS": return database.Stats(), nil
709 case "SEARCH":
710 var matches []string
711 needle := strings.ToLower(words[2])
712 for _, entry := range database.Entries { if value, exists := entry.Fields[words[1]]; exists && strings.Contains(strings.ToLower(plain(value)), needle) { matches = append(matches, entry.Key) } }
713 if len(matches) == 0 { return []string{"MATCHES 0"}, nil }
714 return []string{fmt.Sprintf("MATCHES %d: %s", len(matches), strings.Join(matches, " "))}, nil
715 }
716 panic("unreachable command")
717}
718func main() {
719 scanner := bufio.NewScanner(os.Stdin)
720 var lines []string
721 for scanner.Scan() { if scanner.Text() == "%%" { break }; lines = append(lines, scanner.Text()) }
722 database := readDatabase(lines)
723 output := bufio.NewWriter(os.Stdout)
724 defer output.Flush()
725 for _, line := range database.Output { fmt.Fprintln(output, line) }
726 scanner.Scan()
727 count, _ := strconv.Atoi(scanner.Text())
728 for range count {
729 scanner.Scan()
730 result, err := database.Command(scanner.Text())
731 if err != nil { fmt.Fprintln(output, "ERROR "+err.Error())
732 } else { for _, line := range result { fmt.Fprintln(output, line) } }
733 }
734}
