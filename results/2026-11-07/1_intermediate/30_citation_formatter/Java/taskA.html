32 static List<String> splitDepth(String text, String separator) {
33 var result = new ArrayList<String>();
34 int start = 0, i = 0, depth = 0;
35 while (i < text.length()) {
36 if (depth == 0 && text.startsWith(separator, i)) { result.add(text.substring(start, i)); i += separator.length(); start = i; }
37 else { if (text.charAt(i) == '{') depth++; if (text.charAt(i) == '}') depth--; i++; }
38 }
39 result.add(text.substring(start)); return result;

46 static String evaluate(String expression, Map<String, String> macros, int line) {
47 int depth = 0;
48 for (char c : expression.toCharArray()) {
49 if (c == '{') depth++; if (c == '}') depth--;
50 if (depth < 0) throw new LoadError(line, "unbalanced braces");
51 }
52 if (depth != 0) throw new LoadError(line, "unbalanced braces");
53 var result = new StringBuilder();
54 for (String term : expression.split(" # ", -1)) {
55 char first = term.charAt(0);
56 if (first == '{' || first == '"') result.append(term, 1, term.length() - 1);
57 else if (digit(first)) result.append(term);
58 else {
59 String value = macros.get(term);
60 if (value == null) throw new LoadError(line, "unknown macro '" + term + "'");
61 result.append(value);
62 }
63 }
64 return result.toString();

67 static Accent accent(String text, int at) {

73 if ((symbol || "cuvH".indexOf(command) >= 0) && at + 4 < text.length() && text.charAt(at + 2) == '{' && letter(text.charAt(at + 3)) && text.charAt(at + 4) == '}') return new Accent(text.charAt(at + 3), 5);

76 static String normalize(String text) {
77 var accents = new StringBuilder();
78 for (int i = 0; i < text.length();) {
79 Accent wrapped = text.charAt(i) == '{' ? accent(text, i + 1) : null;
80 if (wrapped != null && i + wrapped.width + 1 < text.length() && text.charAt(i + wrapped.width + 1) == '}') { accents.append(wrapped.value); i += wrapped.width + 2; }
81 else {
82 Accent current = accent(text, i);
83 if (current == null) accents.append(text.charAt(i++));
84 else { accents.append(current.value); i += current.width; }
85 }
86 }
87 return Arrays.stream(accents.toString().replace('~', ' ').split(" ", -1)).filter(s -> !s.isEmpty()).collect(Collectors.joining(" "));

128 static String titleCase(String title, Style style) {
129 if (style == Style.IEEE) return title;
130 if (style == Style.APA) {
131 char[] result = title.toCharArray(); int depth = 0;
132 for (int i = 0; i < result.length; i++) {
133 char c = result[i];
134 if (c == '{') depth++; else if (c == '}') depth--;
135 else if (depth == 0 && letter(c)) result[i] = i == 0 || i >= 2 && title.substring(i - 2, i).equals(": ") ? uppercase(c) : lowercase(c);
136 }
137 return new String(result);
138 }
139 List<String> words = splitDepth(title, " ");
140 for (int i = 0; i < words.size(); i++) {
141 String word = words.get(i);
142 if (word.startsWith("{")) continue;
143 boolean makeLower = i > 0 && i + 1 < words.size() && !words.get(i - 1).endsWith(":") && SMALL_WORDS.contains(letters(word));
144 char[] result = word.toCharArray(); int depth = 0; boolean firstLetter = true;
145 for (int j = 0; j < result.length; j++) {
146 char c = result[j];
147 if (c == '{') depth++; else if (c == '}') depth--;
148 else if (letter(c)) {
149 if (depth == 0 && makeLower) result[j] = lowercase(c); else if (depth == 0 && firstLetter) result[j] = uppercase(c);
150 firstLetter = false;
151 }
152 }
153 words.set(i, new String(result));
154 }
155 return String.join(" ", words);

238 static int optionalEnd(String pattern, int start) {
239 int depth = 1;
240 for (int i = start + 1;; i++) {
241 if (pattern.charAt(i) == '<') depth++; if (pattern.charAt(i) == '>') depth--;
242 if (depth == 0) return i;
243 }
244 }
245 static boolean optionalPresent(String pattern, int start, int end, Map<String, String> values) {
246 for (int i = start; i < end; i++) {
247 if (pattern.charAt(i) == '<') i = optionalEnd(pattern, i);
248 else if (pattern.charAt(i) == '{') {
249 int close = pattern.indexOf('}', i);
250 if (values.getOrDefault(pattern.substring(i + 1, close), "").isEmpty()) return false;
251 i = close;
252 }
253 }
254 return true;
255 }
256 static void renderTemplate(String pattern, int start, int end, Map<String, String> values, StringBuilder result) {
257 for (int i = start; i < end; i++) {
258 char c = pattern.charAt(i);
259 if (c == '<') {
260 int close = optionalEnd(pattern, i);
261 if (optionalPresent(pattern, i + 1, close, values)) renderTemplate(pattern, i + 1, close, values, result);
262 i = close;
263 } else if (c == '{') {
264 int close = pattern.indexOf('}', i);
265 result.append(values.getOrDefault(pattern.substring(i + 1, close), "")); i = close;
266 } else if (c != '.' || result.isEmpty() || ".?!".indexOf(result.charAt(result.length() - 1)) < 0) result.append(c);
267 }

426 List<String> check() {
427 var result = new ArrayList<String>(); var titles = new HashMap<String, String>();
428 for (Entry entry : entries) {
429 var warnings = new ArrayList<String>();
430 if (entry.fields.containsKey("pages")) {
431 String pages = plain(entry.fields.get("pages"));
432 if (pages.contains("-") && !pages.contains("--")) warnings.add("pages use a single hyphen");
433 List<String> parts = Arrays.stream(pages.split("-", -1)).filter(part -> !part.isEmpty()).toList();
434 if (parts.size() == 2 && parts.stream().allMatch(Main::digits) && Integer.parseInt(parts.getFirst()) > Integer.parseInt(parts.getLast())) warnings.add("pages are reversed");
435 }
436 if (entry.year.length() != 4 || !digits(entry.year)) warnings.add("year is not 4 digits");
437 if (entry.plainTitle.chars().noneMatch(Main::lower) && entry.plainTitle.chars().filter(Main::upper).count() >= 2) warnings.add("title is all caps");
438 if (entry.kind == 0 && !entry.fields.containsKey("doi")) warnings.add("article has no doi");
439 if (titles.containsKey(entry.lowerTitle)) warnings.add("same title as " + titles.get(entry.lowerTitle));
440 else titles.put(entry.lowerTitle, entry.key);
441 for (String field : entry.unknownFields()) warnings.add("unknown field '" + field + "'");
442 for (String warning : warnings) result.add("WARN " + entry.key + ": " + warning);
443 }

456 List<String> stats() {
457 var types = new TreeMap<String, Integer>(); var years = new TreeMap<String, Integer>();
458 var authors = new HashSet<String>(); var pairs = new TreeMap<Pair, Integer>();
459 for (Entry entry : entries) {
460 types.merge(TYPES.get(entry.kind), 1, Integer::sum); years.merge(entry.year, 1, Integer::sum);

470 years.forEach((year, count) -> result.add("YEAR " + year + " " + count));
471 result.add("AUTHORS " + authors.size());

476 List<String> command(String line) {
477 String command = line.split(" ", 2)[0];
478 String[] words = line.split(" ", command.equals("CITE") || command.equals("SEARCH") ? 3 : -1);
479 int arity = switch (command) {
480 case "FORMAT" -> -1; case "BIBLIOGRAPHY", "EXPORT" -> 1; case "CITE", "SEARCH" -> 2; case "KEYGEN", "CHECK", "STATS" -> 0;
481 default -> throw new Problem("unknown command '" + command + "'");
482 };
483 if (arity == -1 ? words.length < 3 : words.length != arity + 1) throw new Problem("wrong number of arguments");
484 Style style = Style.APA;
485 if (command.equals("FORMAT") || command.equals("BIBLIOGRAPHY") || command.equals("CITE")) {
486 style = switch (words[1]) { case "APA" -> Style.APA; case "IEEE" -> Style.IEEE; case "CHICAGO" -> Style.CHICAGO; case "MLA" -> Style.MLA; default -> throw new Problem("unknown style '" + words[1] + "'"); };
487 }
488 var result = new ArrayList<String>();
489 switch (command) {
490 case "FORMAT" -> {
491 Numbering numbering = numbering();
492 for (int i = 2; i < words.length; i++) {
493 Entry entry = lookup.get(words[i]);
494 result.add(entry == null ? "ERROR unknown key '" + words[i] + "'" : reference(entry, style, numbering.numbers.get(entry), false));
495 }
496 }
497 case "BIBLIOGRAPHY" -> {
498 Numbering numbering = numbering(); List<Entry> order = numbering.order;
499 if (style != Style.IEEE) order.sort(BIBLIOGRAPHY);
500 result.add(style == Style.MLA ? "Works Cited" : "References");
501 for (int i = 0; i < order.size(); i++) {
502 Entry entry = order.get(i);
503 boolean repeated = i > 0 && (style == Style.CHICAGO || style == Style.MLA) && entry.primary().equals(order.get(i - 1).primary());
504 result.add(reference(entry, style, numbering.numbers.get(entry), repeated));
505 }
506 }
507 case "CITE" -> result.add(cite(style, words[2]));
508 case "KEYGEN" -> result.addAll(keygen()); case "CHECK" -> result.addAll(check()); case "EXPORT" -> result.addAll(export(words[1])); case "STATS" -> result.addAll(stats());
509 case "SEARCH" -> {
510 String needle = lowerText(words[2]); var matches = new ArrayList<String>();
511 for (Entry entry : entries) if (entry.fields.containsKey(words[1]) && lowerText(plain(entry.fields.get(words[1]))).contains(needle)) matches.add(entry.key);
512 result.add(matches.isEmpty() ? "MATCHES 0" : "MATCHES " + matches.size() + ": " + String.join(" ", matches));
513 }
514 default -> throw new AssertionError();
515 }
516 return result;
517 }
