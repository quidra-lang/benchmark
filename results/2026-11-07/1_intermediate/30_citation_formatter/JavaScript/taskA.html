32function compareTuple(a, b) {
33 for (let i = 0; i < a.length; i++) {
34 const order = compare(a[i], b[i]);
35 if (order !== 0) return order;
36 }
37 return 0;

40function splitAtDepthZero(text, separator) {
41 const parts = [];
42 let depth = 0, start = 0, i = 0;
43 while (i < text.length) {
44 if (depth === 0 && text.startsWith(separator, i)) {
45 parts.push(text.slice(start, i));
46 i += separator.length;
47 start = i;
48 } else {
49 if (text[i] === "{") depth++;
50 if (text[i] === "}") depth--;
51 i++;
52 }
53 }
54 parts.push(text.slice(start));
55 return parts;
56}

58function accentAt(text, i) {
59 if (text[i] !== "\\") return null;
60 if (text[i + 1] === "&") return { char: "&", length: 2 };
61 const symbol = text[i + 1];
62 if ("'\"`^~=.".includes(symbol) && isLetter(text[i + 2])) {
63 return { char: text[i + 2], length: 3 };
64 }
65 if ("'\"`^~=.cuvH".includes(symbol) && text[i + 2] === "{" &&
66 isLetter(text[i + 3]) && text[i + 4] === "}") {
67 return { char: text[i + 3], length: 5 };
68 }
69 return null;
70}

72function normalize(text) {
73 let result = "";
74 for (let i = 0; i < text.length;) {
75 const enclosed = text[i] === "{" ? accentAt(text, i + 1) : null;
76 if (enclosed && text[i + 1 + enclosed.length] === "}") {
77 result += enclosed.char;
78 i += enclosed.length + 2;
79 continue;
80 }
81 const accent = accentAt(text, i);
82 if (accent) {
83 result += accent.char;
84 i += accent.length;
85 } else {
86 result += text[i++];
87 }
88 }
89 return result.replace(/~/g, " ").replace(/ +/g, " ").trim();
90}

92function evaluateExpression(expression, macros) {
93 let depth = 0;
94 for (const char of expression) {
95 if (char === "{") depth++;
96 if (char === "}") depth--;
97 if (depth < 0) throw new Error("unbalanced braces");
98 }
99 if (depth !== 0) throw new Error("unbalanced braces");
100 return expression.split(" # ").map(term => {
101 if (term[0] === "{" || term[0] === '"') return term.slice(1, -1);
102 if (/^[0-9]+$/.test(term)) return term;
103 if (!macros.has(term)) throw new Error(`unknown macro '${term}'`);
104 return macros.get(term);
105 }).join("");
106}

108class Person {
109 constructor(text) {
110 const parts = splitAtDepthZero(text, ",").map(part => part.trim());
111 const words = splitAtDepthZero(parts[0], " ");
112 const lower = words.flatMap((word, i) =>
113 i < words.length - 1 && /^[a-z]/.test(word) ? [i] : []);
114 this.first = "";
115 this.von = "";
116 this.jr = "";
117 if (parts.length === 1) {
118 const start = lower.length ? lower[0] : words.length - 1;
119 const end = lower.length ? lower.at(-1) + 1 : words.length - 1;
120 this.first = words.slice(0, start).join(" ");
121 this.von = words.slice(start, end).join(" ");
122 this.last = words.slice(end).join(" ");
123 } else {
124 const end = lower.length ? lower.at(-1) + 1 : 0;
125 this.von = words.slice(0, end).join(" ");
126 this.last = words.slice(end).join(" ");
127 this.first = parts.at(-1);
128 if (parts.length === 3) this.jr = parts[1];
129 }
130 }
131
132 surname() {
133 return plain(this.von ? `${this.von} ${this.last}` : this.last);
134 }
135
136 initials() {
137 const given = plain(this.first);
138 if (!given) return "";
139 return given.split(" ").map(word => word.split("-").map(piece =>
140 piece.match(/[A-Za-z]/)[0].toUpperCase() + ".").join("-")).join(" ");
141 }
142
143 format(inverted, initials) {
144 const given = initials ? this.initials() : plain(this.first);
145 let result = this.surname();
146 if (this.first) result = inverted ? `${result}, ${given}` : `${given} ${result}`;
147 if (this.jr) result += `, ${plain(this.jr)}`;
148 return result;
149 }
150}

193class Database {
194 constructor(lines, output) {
195 this.entries = [];
196 this.byKey = new Map();
197 const macros = new Map(monthNames.map(month => [month.slice(0, 3).toLowerCase(), month]));
198 let errorCount = 0;
199 const report = (line, name, reason) => {
200 output.push(`ERROR line ${line}: ${name}: ${reason}`);
201 errorCount++;
202 };
203 for (let i = 0; i < lines.length; i++) {
204 const line = lines[i];
205 if (!line || line.startsWith("%")) continue;
206 if (line.startsWith("@string{")) {
207 const [, name, expression] = line.match(/^@string\{([a-z][a-z0-9]*) = (.*)\}$/);
208 try {
209 const value = evaluateExpression(expression, macros);
210 if (macros.has(name)) throw new Error("duplicate macro");
211 macros.set(name, value);
212 } catch (error) {
213 report(i + 1, name, error.message);
214 }
215 continue;
216 }
217 const [, type, key] = line.match(/^@([a-z]+)\{([^,]+),$/);
218 const fields = new Map();
219 let failed = false;
220 const fail = reason => {
221 report(i + 1, key, reason);
222 failed = true;
223 };
224 if (!requiredFields.has(type)) fail(`unknown type '${type}'`);
225 else if (this.byKey.has(key)) fail("duplicate key");
226 while (lines[++i] !== "}") {
227 if (failed) continue;
228 const [, name, raw] = lines[i].match(/^ ([a-z]+) = (.*)$/);
229 try {
230 const value = evaluateExpression(raw.endsWith(",") ? raw.slice(0, -1) : raw, macros);
231 if (fields.has(name)) throw new Error(`duplicate field '${name}'`);
232 fields.set(name, normalize(value));
233 } catch (error) {
234 fail(error.message);
235 }
236 }
237 if (failed) continue;
238 for (const field of requiredFields.get(type)) {
239 if (!fields.has(field) && !(type === "book" && field === "author" && fields.has("editor"))) {
240 fail(`missing field '${field}'`);
241 break;
242 }
243 }
244 if (!failed) {
245 const entry = new Entry(key, type, fields);
246 this.entries.push(entry);
247 this.byKey.set(key, entry);
248 }
249 }
250 output.push(`LOADED entries=${this.entries.length} macros=${macros.size - 12} errors=${errorCount}`);
251 const groups = new Map();
252 for (const entry of this.entries) {
253 const groupKey = JSON.stringify([entry.primary.signature, entry.fields.get("year")]);
254 if (!groups.has(groupKey)) groups.set(groupKey, []);
255 groups.get(groupKey).push(entry);
256 }
257 for (const group of groups.values()) {
258 if (group.length < 2) continue;
259 group.sort((a, b) => compareTuple([a.sortTitle, a.key], [b.sortTitle, b.key]));
260 group.forEach((entry, i) => { entry.yearSuffix = String.fromCharCode(97 + i); });
261 }
262 this.bibliography = [...this.entries].sort((a, b) => compareTuple(a.sortKey, b.sortKey));
263 }
264}

272function authorValue(entry, style, repeated) {
273 const { names, etAl } = entry.primary;
274 let result = "---";
275 if (!repeated) {
276 if (style === "APA") {
277 const formatted = names.map(name => name.format(true, true));
278 result = etAl ? formatted.join(", ") + ", et al." : names.length <= 20
279 ? listNames(formatted, "&") : formatted.slice(0, 19).join(", ") + ", . . . " + formatted.at(-1);
280 } else if (style === "IEEE") {
281 const formatted = names.map(name => name.format(false, true));
282 result = etAl || names.length >= 7 ? formatted[0] + " et al." : listNames(formatted, "and", true);
283 } else {
284 const formatted = names.map((name, i) => name.format(i === 0, false));
285 if (style === "CHICAGO" && (etAl || names.length >= 11)) result = formatted.slice(0, 7).join(", ") + ", et al.";
286 else if (style === "MLA" && (etAl || names.length >= 3)) result = formatted[0] + ", et al.";
287 else result = listNames(formatted, "and");
288 }
289 }
290 if (!entry.author) {
291 const suffixes = { APA: [" (Ed.)", " (Eds.)"], IEEE: [", Ed.", ", Eds."],
292 CHICAGO: [", ed.", ", eds."], MLA: [", editor", ", editors"] };
293 result += suffixes[style][entry.primary.singular() ? 0 : 1];
294 }
295 return result;
296}

304function convertLetters(text, transform) {
305 let depth = 0, result = "";
306 for (let i = 0; i < text.length; i++) {
307 const char = text[i];
308 if (char === "{") depth++;
309 if (char === "}") depth--;
310 result += isLetter(char) ? transform(char, i, depth) : char;
311 }
312 return result;
313}

315function titleValue(title, style) {
316 if (style === "IEEE") return plain(title);
317 if (style === "APA") {
318 return plain(convertLetters(title, (char, i, depth) => depth ? char :
319 i === 0 || title.slice(Math.max(0, i - 2), i) === ": " ? char.toUpperCase() : char.toLowerCase()));
320 }
321 const words = splitAtDepthZero(title, " ");
322 return plain(words.map((word, i) => {
323 if (word.startsWith("{")) return word;
324 if (i > 0 && i < words.length - 1 && !words[i - 1].endsWith(":") && smallWords.has(letters(word))) {
325 return convertLetters(word, (char, _index, depth) => depth ? char : char.toLowerCase());
326 }
327 let first = true;
328 return convertLetters(word, (char, _index, depth) => {
329 const result = first && depth === 0 ? char.toUpperCase() : char;
330 first = false;
331 return result;
332 });
333 }).join(" "));
334}

336function ordinal(value) {
337 const number = Number(value);
338 const lastTwo = number % 100;
339 const suffix = lastTwo >= 11 && lastTwo <= 13 ? "th" : ({ 1: "st", 2: "nd", 3: "rd" }[number % 10] ?? "th");
340 return value + suffix;
341}

343function placeholderValues(entry, style, repeated) {
344 const values = new Map([...entry.fields].map(([field, value]) => [field, plain(value)]));
345 values.set("author", authorValue(entry, style, repeated));
346 values.set("editor", editorValue(entry.editor, style));
347 values.set("eds", entry.editor ? entry.editor.singular() ? "Ed." : "Eds." : "");
348 values.set("title", titleValue(entry.fields.get("title"), style));
349 values.set("year", entry.yearLabel(style));
350 const pages = entry.value("pages").replace(/--/g, "-");
351 values.set("pages", pages);
352 values.set("pp", entry.fields.has("pages") ? (pages.includes("-") ? "pp. " : "p. ") + pages : "");
353 const edition = entry.value("edition");
354 values.set("edition", entry.fields.has("edition") ? (/^[0-9]+$/.test(edition) ? ordinal(edition) : edition) + " ed." : "");
355 const month = entry.value("month");
356 values.set("mon", monthAbbreviations.get(month) ?? month);
357 return values;
358}

360function parseTemplate(source) {
361 let position = 0;
362 function sequence() {
363 const nodes = [];
364 while (position < source.length && source[position] !== ">") {
365 if (source[position] === "<") {
366 position++;
367 nodes.push({ kind: "optional", children: sequence() });
368 position++;
369 } else if (source[position] === "{") {
370 const end = source.indexOf("}", position);
371 nodes.push({ kind: "value", name: source.slice(position + 1, end) });
372 position = end + 1;
373 } else {
374 const start = position++;
375 while (position < source.length && !"<{>".includes(source[position])) position++;
376 nodes.push({ kind: "literal", text: source.slice(start, position) });
377 }
378 }
379 return nodes;
380 }
381 return sequence();
382}
383
384function renderTemplate(nodes, values) {
385 let result = "";
386 function visit(sequence) {
387 for (const node of sequence) {
388 if (node.kind === "value") result += values.get(node.name) ?? "";
389 else if (node.kind === "optional") {
390 if (node.children.every(child => child.kind !== "value" || (values.get(child.name) ?? "") !== "")) visit(node.children);
391 } else {
392 for (const char of node.text) {
393 if (char !== "." || !/[.?!]$/.test(result)) result += char;
394 }
395 }
396 }
397 }
398 visit(nodes);
399 return result;
400}

467 citation(kind, entries, style, numbers) {
468 const parenthetical = kind === "cite" || kind === "citep";
469 const name = entry => shortName(entry, style, parenthetical);
470 if (kind === "citeauthor") return entries.map(name).join("; ");
471 if (kind === "citeyear") return entries.map(entry => entry.yearLabel(style)).join(", ");
472 if (style === "MLA") {
473 const result = entries.map(name).join("; ");
474 return parenthetical ? `(${result})` : result;
475 }
476 if (style === "IEEE") {
477 if (!parenthetical) return entries.map(entry => `${name(entry)} [${numbers.get(entry.key)}]`).join("; ");
478 const sorted = entries.map(entry => numbers.get(entry.key)).sort((a, b) => a - b);
479 const pieces = [];
480 for (let i = 0; i < sorted.length;) {
481 let end = i + 1;
482 while (end < sorted.length && sorted[end] === sorted[end - 1] + 1) end++;
483 if (end - i >= 3) pieces.push(`[${sorted[i]}]-[${sorted[end - 1]}]`);
484 else for (let j = i; j < end; j++) pieces.push(`[${sorted[j]}]`);
485 i = end;
486 }
487 return pieces.join(", ");
488 }
489 const ordered = parenthetical ? [...entries].sort((a, b) => compareTuple(a.sortKey, b.sortKey)) : entries;
490 const groups = [];
491 for (const entry of ordered) {
492 const currentName = name(entry);
493 if (!groups.length || groups.at(-1).name !== currentName) groups.push({ name: currentName, years: [] });
494 groups.at(-1).years.push(entry.yearLabel(style));
495 }
496 const result = groups.map(group => parenthetical
497 ? group.name + (style === "APA" ? ", " : " ") + group.years.join(", ")
498 : `${group.name} (${group.years.join(", ")})`).join("; ");
499 return parenthetical ? `(${result})` : result;
500 }

503 const citations = [...text.matchAll(/\\(citeauthor|citeyear|citep|citet|cite)\{([^}]+)\}/g)].map(match => ({
504 start: match.index, length: match[0].length, kind: match[1], keys: [...new Set(match[2].split(","))],
505 }));
506 for (const citation of citations) {
507 for (const key of citation.keys) {
508 if (!this.database.byKey.has(key)) {
509 this.output.push(`ERROR unknown key '${key}'`);
510 return;
511 }
512 }
513 }
514 for (const citation of citations) {
515 for (const key of citation.keys) {
516 if (!this.cited.has(key)) {
517 this.cited.add(key);
518 this.citationRecord.push(key);
519 }
520 }
521 }
522 const numbers = this.ieeeNumbers();
523 let result = "", position = 0;
524 for (const citation of citations) {
525 result += text.slice(position, citation.start);
526 result += this.citation(citation.kind, citation.keys.map(key => this.database.byKey.get(key)), style, numbers);
527 position = citation.start + citation.length;
528 }
529 this.output.push(result + text.slice(position));
530 }

532 keygen() {
533 const bases = this.database.entries.map(entry => {
534 const titleWord = entry.value("title").split(" ").map(letters).find(word => word && !smallWords.has(word)) ?? "";
535 return letters(entry.primary.names[0].last) + entry.fields.get("year").replace(/[^0-9]/g, "") + titleWord;

547 check() {
548 let warnings = 0;
549 const titles = new Map();
550 for (const entry of this.database.entries) {
551 const warn = message => { this.output.push(`WARN ${entry.key}: ${message}`); warnings++; };
552 const pages = entry.value("pages"), title = entry.value("title");
553 if (pages.includes("-") && !pages.includes("--")) warn("pages use a single hyphen");
554 const pageParts = pages.split("-").filter(part => part !== "");
555 if (pageParts.length === 2 && pageParts.every(part => /^[0-9]+$/.test(part)) && Number(pageParts[0]) > Number(pageParts[1])) warn("pages are reversed");
556 if (!/^[0-9]{4}$/.test(entry.value("year"))) warn("year is not 4 digits");
557 if (!/[a-z]/.test(title) && (title.match(/[A-Z]/g)?.length ?? 0) >= 2) warn("title is all caps");
558 if (entry.type === "article" && !entry.fields.has("doi")) warn("article has no doi");
559 const titleKey = title.toLowerCase();
560 if (titles.has(titleKey)) warn(`same title as ${titles.get(titleKey)}`);
561 else titles.set(titleKey, entry.key);
562 for (const field of entry.unknownFields()) warn(`unknown field '${field}'`);
563 }
564 this.output.push(`CHECK warnings=${warnings}`);
565 }
566
567 export(key) {
568 const entry = this.database.byKey.get(key);
569 if (!entry) {
570 this.output.push(`ERROR unknown key '${key}'`);
571 return;
572 }
573 this.output.push(`@${entry.type}{${key},`);
574 const fields = [...fieldOrder.filter(field => entry.fields.has(field)), ...entry.unknownFields()];
575 fields.forEach((field, i) => this.output.push(` ${field} = {${entry.fields.get(field)}}${i + 1 < fields.length ? "," : ""}`));
576 this.output.push("}");
577 }

579 stats() {
580 const types = new Map(), years = new Map(), authors = new Set(), pairs = new Map();
581 for (const entry of this.database.entries) {
582 types.set(entry.type, (types.get(entry.type) ?? 0) + 1);
583 const year = entry.value("year");
584 years.set(year, (years.get(year) ?? 0) + 1);
585 if (!entry.author) continue;
586 const names = [...new Set(entry.author.names.map(name => name.format(false, false)))].sort(compare);
587 for (const name of names) authors.add(name);
588 for (let i = 0; i < names.length; i++) {
589 for (let j = i + 1; j < names.length; j++) {
590 const key = JSON.stringify([names[i], names[j]]);
591 if (!pairs.has(key)) pairs.set(key, { a: names[i], b: names[j], count: 0 });
592 pairs.get(key).count++;
593 }
594 }
595 }
596 this.output.push(`ENTRIES ${this.database.entries.length}`);
597 for (const type of [...types.keys()].sort(compare)) this.output.push(`TYPE ${type} ${types.get(type)}`);
598 for (const year of [...years.keys()].sort(compare)) this.output.push(`YEAR ${year} ${years.get(year)}`);
599 this.output.push(`AUTHORS ${authors.size}`);
600 const best = [...pairs.values()].sort((a, b) => b.count - a.count || compareTuple([a.a, a.b], [b.a, b.b]))[0];
601 this.output.push(best ? `PAIR ${best.a} & ${best.b} ${best.count}` : "PAIR none");
602 }

604 command(line) {
605 const firstSpace = line.indexOf(" ");
606 const command = firstSpace < 0 ? line : line.slice(0, firstSpace);
607 let parts;
608 if (command === "CITE" || command === "SEARCH") {
609 const secondSpace = firstSpace < 0 ? -1 : line.indexOf(" ", firstSpace + 1);
610 parts = firstSpace < 0 ? [line] : secondSpace < 0 ? [command, line.slice(firstSpace + 1)]
611 : [command, line.slice(firstSpace + 1, secondSpace), line.slice(secondSpace + 1)];
612 } else parts = line.split(" ");
613 const args = parts.slice(1);
614 if (!commandArity.has(command)) this.output.push(`ERROR unknown command '${command}'`);
615 else if (command === "FORMAT" ? args.length < 2 : args.length !== commandArity.get(command)) this.output.push("ERROR wrong number of arguments");
616 else if (["FORMAT", "BIBLIOGRAPHY", "CITE"].includes(command) && !styles.has(args[0])) this.output.push(`ERROR unknown style '${args[0]}'`);
617 else if (command === "FORMAT") {
618 const numbers = this.ieeeNumbers();
619 for (const key of args.slice(1)) {
620 const entry = this.database.byKey.get(key);
621 this.output.push(entry ? this.reference(entry, args[0], numbers) : `ERROR unknown key '${key}'`);
622 }
623 } else if (command === "BIBLIOGRAPHY") {
624 const style = args[0], numbers = this.ieeeNumbers();
625 const entries = style === "IEEE" ? this.ieeeOrder().map(key => this.database.byKey.get(key)) : this.database.bibliography;
626 this.output.push(style === "MLA" ? "Works Cited" : "References");
627 let previous = null;
628 for (const entry of entries) {
629 const repeated = (style === "CHICAGO" || style === "MLA") && previous === entry.primary.signature;
630 this.output.push(this.reference(entry, style, numbers, repeated));
631 previous = entry.primary.signature;
632 }
633 } else if (command === "CITE") this.cite(args[0], args[1]);
634 else if (command === "KEYGEN") this.keygen();
635 else if (command === "CHECK") this.check();
636 else if (command === "EXPORT") this.export(args[0]);
637 else if (command === "STATS") this.stats();
638 else if (command === "SEARCH") {
639 const keys = this.database.entries.filter(entry => entry.fields.has(args[0]) && entry.value(args[0]).toLowerCase().includes(args[1].toLowerCase())).map(entry => entry.key);
640 this.output.push(keys.length ? `MATCHES ${keys.length}: ${keys.join(" ")}` : "MATCHES 0");
641 }
642 }
