54fn letter(c: u8) bool {
55 return c >= 'A' and c <= 'Z' or c >= 'a' and c <= 'z';
56}
57fn digit(c: u8) bool {
58 return c >= '0' and c <= '9';
59}
60fn lowercase(c: u8) bool {
61 return c >= 'a' and c <= 'z';
62}
63fn uppercase(c: u8) bool {
64 return c >= 'A' and c <= 'Z';
65}
66fn lowerByte(c: u8) u8 {
67 return if (uppercase(c)) c + 32 else c;
68}
69fn upperByte(c: u8) u8 {
70 return if (lowercase(c)) c - 32 else c;
71}
72fn digits(text: []const u8) bool {
73 if (text.len == 0) return false;
74 for (text) |c| if (!digit(c)) return false;
75 return true;
76}

105fn lower(gpa: Allocator, text: []const u8) Allocator.Error![]const u8 {
106 const result = try gpa.dupe(u8, text);
107 for (result) |*c| c.* = lowerByte(c.*);
108 return result;
109}
110fn plain(gpa: Allocator, text: []const u8) Allocator.Error![]const u8 {
111 var result: Text = .{ .gpa = gpa };
112 for (text) |c| if (c != '{' and c != '}') try result.byte(c);
113 return result.result();
114}
115fn letters(gpa: Allocator, text: []const u8) Allocator.Error![]const u8 {
116 var result: Text = .{ .gpa = gpa };
117 for (text) |c| if (letter(c)) try result.byte(lowerByte(c));
118 return result.result();
119}
120fn splitTop(gpa: Allocator, text: []const u8, separator: []const u8) Allocator.Error![][]const u8 {
121 var parts: std.ArrayList([]const u8) = .empty;
122 var depth: i32 = 0;
123 var start: usize = 0;
124 var i: usize = 0;
125 while (i < text.len) {
126 if (depth == 0 and std.mem.startsWith(u8, text[i..], separator)) {
127 try parts.append(gpa, text[start..i]);
128 i += separator.len;
129 start = i;
130 continue;
131 }
132 if (text[i] == '{') depth += 1;
133 if (text[i] == '}') depth -= 1;
134 i += 1;
135 }
136 try parts.append(gpa, text[start..]);
137 return parts.items;
138}
139const Accent = struct { value: u8, length: usize };
140fn accent(text: []const u8, i: usize) ?Accent {
141 if (i + 1 >= text.len or text[i] != '\\') return null;
142 const command = text[i + 1];
143 if (command == '&') return .{ .value = '&', .length = 2 };
144 const symbol = std.mem.findScalar(u8, "'\"`^~=.", command) != null;
145 if (i + 4 < text.len and (symbol or std.mem.findScalar(u8, "cuvH", command) != null) and
146 text[i + 2] == '{' and letter(text[i + 3]) and text[i + 4] == '}')
147 {
148 return .{ .value = text[i + 3], .length = 5 };
149 }
150 if (symbol and i + 2 < text.len and letter(text[i + 2])) return .{ .value = text[i + 2], .length = 3 };
151 return null;
152}
153fn normalize(gpa: Allocator, text: []const u8) Allocator.Error![]const u8 {
154 var result: Text = .{ .gpa = gpa };
155 var pending_space = false;
156 var i: usize = 0;
157 while (i < text.len) {
158 var value = text[i];
159 var length: usize = 1;
160 const wrapped = if (text[i] == '{') accent(text, i + 1) else null;
161 if (wrapped != null and i + wrapped.?.length + 1 < text.len and text[i + wrapped.?.length + 1] == '}') {
162 value = wrapped.?.value;
163 length = wrapped.?.length + 2;
164 } else if (accent(text, i)) |direct| {
165 value = direct.value;
166 length = direct.length;
167 }
168 i += length;
169 if (value == '~' or value == ' ') {
170 pending_space = result.result().len != 0;
171 } else {
172 if (pending_space) try result.byte(' ');
173 pending_space = false;
174 try result.byte(value);
175 }
176 }
177 return result.result();
178}
179fn titleCase(gpa: Allocator, text: []const u8, style: Style) Allocator.Error![]const u8 {
180 if (style == .IEEE) return text;
181 if (style == .APA) {
182 const result = try gpa.dupe(u8, text);
183 var depth: i32 = 0;
184 for (result, 0..) |*c, i| {
185 if (c.* == '{') {
186 depth += 1;
187 } else if (c.* == '}') {
188 depth -= 1;
189 } else if (depth == 0 and letter(c.*)) {
190 c.* = if (i == 0 or i >= 2 and text[i - 2] == ':' and text[i - 1] == ' ') upperByte(c.*) else lowerByte(c.*);
191 }
192 }
193 return result;
194 }
195 const words = try splitTop(gpa, text, " ");
196 for (words, 0..) |word, i| {
197 if (word[0] == '{') continue;
198 const make_lower = i > 0 and i + 1 < words.len and !std.mem.endsWith(u8, words[i - 1], ":") and
199 contains(&small_words, try letters(gpa, word));
200 const changed = try gpa.dupe(u8, word);
201 var depth: i32 = 0;
202 for (changed) |*c| {
203 if (c.* == '{') {
204 depth += 1;
205 } else if (c.* == '}') {
206 depth -= 1;
207 } else if (letter(c.*)) {
208 if (make_lower) {
209 if (depth == 0) c.* = lowerByte(c.*);
210 } else {
211 if (depth == 0) c.* = upperByte(c.*);
212 break;
213 }
214 }
215 }
216 words[i] = changed;
217 }
218 return join(gpa, " ", words);
219}
220fn listNames(gpa: Allocator, names: []const []const u8, conjunction: []const u8, list_b: bool) Allocator.Error![]const u8 {
221 if (names.len == 1) return names[0];
222 if (list_b and names.len == 2) return join(gpa, " ", &.{ names[0], conjunction, names[1] });
223 return format(gpa, "{s}, {s} {s}", .{ try join(gpa, ", ", names[0 .. names.len - 1]), conjunction, names[names.len - 1] });
224}

226const Person = struct {
227 first: []const u8,
228 von: []const u8,
229 last: []const u8,
230 jr: []const u8,
231
232 fn parse(gpa: Allocator, text: []const u8) Allocator.Error!Person {
233 const parts = try splitTop(gpa, text, ",");
234 for (parts) |*part| part.* = std.mem.trim(u8, part.*, " ");
235 const words = try splitTop(gpa, parts[0], " ");
236 var first_lower: ?usize = null;
237 var after_lower: usize = 0;
238 for (words[0 .. words.len - 1], 0..) |word, i| {
239 if (lowercase(word[0])) {
240 if (first_lower == null) first_lower = i;
241 after_lower = i + 1;
242 }
243 }
244 if (parts.len == 1) {
245 if (first_lower) |first| return .{
246 .first = try join(gpa, " ", words[0..first]),
247 .von = try join(gpa, " ", words[first..after_lower]),
248 .last = try join(gpa, " ", words[after_lower..]),
249 .jr = "",
250 };
251 return .{ .first = try join(gpa, " ", words[0 .. words.len - 1]), .von = "", .last = words[words.len - 1], .jr = "" };
252 }
253 return .{
254 .first = parts[parts.len - 1],
255 .von = try join(gpa, " ", words[0..after_lower]),
256 .last = try join(gpa, " ", words[after_lower..]),
257 .jr = if (parts.len == 3) parts[1] else "",
258 };
259 }
260 fn same(a: Person, b: Person) bool {
261 return equal(a.first, b.first) and equal(a.von, b.von) and equal(a.last, b.last) and equal(a.jr, b.jr);
262 }
263 fn surname(person: Person, gpa: Allocator) Allocator.Error![]const u8 {
264 return plain(gpa, if (person.von.len == 0) person.last else try join(gpa, " ", &.{ person.von, person.last }));
265 }
266 fn initials(person: Person, gpa: Allocator) Allocator.Error![]const u8 {
267 var result: Text = .{ .gpa = gpa };
268 var words = std.mem.tokenizeScalar(u8, try plain(gpa, person.first), ' ');
269 while (words.next()) |word| {
270 if (result.result().len > 0) try result.byte(' ');
271 var pieces = std.mem.splitScalar(u8, word, '-');
272 var first = true;
273 while (pieces.next()) |piece| {
274 if (!first) try result.byte('-');
275 first = false;
276 for (piece) |c| {
277 if (letter(c)) {
278 try result.byte(upperByte(c));
279 try result.byte('.');
280 break;
281 }
282 }
283 }
284 }
285 return result.result();
286 }
287 fn notation(person: Person, gpa: Allocator, inverted: bool, use_initials: bool) Allocator.Error![]const u8 {
288 const given = if (use_initials) try person.initials(gpa) else try plain(gpa, person.first);
289 var result = try person.surname(gpa);
290 if (person.first.len != 0) result = if (inverted)
291 try join(gpa, ", ", &.{ result, given })
292 else
293 try join(gpa, " ", &.{ given, result });
294 if (person.jr.len != 0) result = try join(gpa, ", ", &.{ result, try plain(gpa, person.jr) });
295 return result;
296 }

298const NameList = struct {
299 names: []const Person,
300 et_al: bool,
301
302 fn parse(gpa: Allocator, text: []const u8) Allocator.Error!NameList {
303 const parts = try splitTop(gpa, text, " and ");
304 const et_al = equal(parts[parts.len - 1], "others");
305 const names = try gpa.alloc(Person, parts.len - @intFromBool(et_al));
306 for (names, parts[0..names.len]) |*name, part| name.* = try Person.parse(gpa, part);
307 return .{ .names = names, .et_al = et_al };
308 }
309 fn single(list: NameList) bool {
310 return list.names.len == 1 and !list.et_al;
311 }
312 fn same(a: NameList, b: NameList) bool {
313 if (a.et_al != b.et_al or a.names.len != b.names.len) return false;
314 for (a.names, b.names) |left, right| if (!left.same(right)) return false;
315 return true;
316 }
317 fn authors(list: NameList, gpa: Allocator, style: Style) Allocator.Error![]const u8 {
318 const values = try gpa.alloc([]const u8, list.names.len);
319 for (list.names, 0..) |name, i| values[i] = try name.notation(gpa, style == .APA or style != .IEEE and i == 0, style == .APA or style == .IEEE);
320 return switch (style) {
321 .APA => if (list.et_al) format(gpa, "{s}, et al.", .{try join(gpa, ", ", values)}) else if (values.len > 20) format(gpa, "{s}, . . . {s}", .{ try join(gpa, ", ", values[0..19]), values[values.len - 1] }) else listNames(gpa, values, "&", false),
322 .IEEE => if (list.et_al or values.len >= 7) format(gpa, "{s} et al.", .{values[0]}) else listNames(gpa, values, "and", true),
323 .CHICAGO => if (list.et_al or values.len >= 11) format(gpa, "{s}, et al.", .{try join(gpa, ", ", values[0..@min(7, values.len)])}) else listNames(gpa, values, "and", false),
324 .MLA => if (list.et_al or values.len >= 3) format(gpa, "{s}, et al.", .{values[0]}) else listNames(gpa, values, "and", false),
325 };
326 }
327 fn editors(list: NameList, gpa: Allocator, style: Style) Allocator.Error![]const u8 {
328 const values = try gpa.alloc([]const u8, list.names.len);
329 for (list.names, 0..) |name, i| values[i] = try name.notation(gpa, false, style == .APA or style == .IEEE);
330 return if (list.et_al) format(gpa, "{s} et al.", .{values[0]}) else listNames(gpa, values, if (style == .APA) "&" else "and", true);
331 }

349 fn primary(entry: Entry) NameList {

361 fn year(entry: Entry, gpa: Allocator, style: Style) Allocator.Error![]const u8 {
362 return if ((style == .APA or style == .CHICAGO) and entry.suffix != 0)

367 fn keyLess(_: void, a: *Entry, b: *Entry) bool {
368 return less({}, a.key, b.key);
369 }
370 fn titleLess(_: void, a: *Entry, b: *Entry) bool {
371 const order = std.mem.order(u8, a.sort_title, b.sort_title);
372 return if (order == .eq) keyLess({}, a, b) else order == .lt;
373 }
374 fn bibLess(_: void, a: *Entry, b: *Entry) bool {
375 const left = [_][]const u8{ a.sort_last, a.value("year"), a.sort_title, a.key };
376 const right = [_][]const u8{ b.sort_last, b.value("year"), b.sort_title, b.key };
377 for (left, right) |x, y| {
378 const order = std.mem.order(u8, x, y);
379 if (order != .eq) return order == .lt;
380 }
381 return false;
382 }

395const Database = struct {
396 gpa: Allocator,
397 macros: std.StringHashMapUnmanaged([]const u8) = .empty,
398 entries: std.ArrayList(*Entry) = .empty,
399 by_key: std.StringHashMapUnmanaged(*Entry) = .empty,
400 error_line: usize = 0,
401 problem: []const u8 = "",
402
403 fn fail(db: *Database, comptime pattern: []const u8, args: anytype) Error {
404 db.problem = format(db.gpa, pattern, args) catch return error.OutOfMemory;
405 return error.Invalid;
406 }
407 fn evaluate(db: *Database, expression: []const u8) Error![]const u8 {
408 var depth: i32 = 0;
409 for (expression) |c| {
410 if (c == '{') depth += 1;
411 if (c == '}') depth -= 1;
412 if (depth < 0) return db.fail("unbalanced braces", .{});
413 }
414 if (depth != 0) return db.fail("unbalanced braces", .{});
415 var result: Text = .{ .gpa = db.gpa };
416 var terms = std.mem.splitSequence(u8, expression, " # ");
417 while (terms.next()) |term| {
418 if (term[0] == '{' or term[0] == '"') {
419 try result.add(term[1 .. term.len - 1]);
420 } else if (digit(term[0])) {
421 try result.add(term);
422 } else {
423 try result.add(db.macros.get(term) orelse return db.fail("unknown macro '{s}'", .{term}));
424 }
425 }
426 return result.result();
427 }
428 fn readMacro(db: *Database, text: []const u8, separator: usize, name: []const u8) Error!void {
429 const value = try db.evaluate(text[separator + 3 .. text.len - 1]);
430 if (db.macros.contains(name)) return db.fail("duplicate macro", .{});
431 try db.macros.put(db.gpa, name, value);
432 }
433 fn readEntry(db: *Database, lines: []const []const u8, start: usize, end: usize, kind_text: []const u8, key: []const u8) Error!void {
434 db.error_line = start + 1;
435 const kind = std.meta.stringToEnum(Kind, kind_text) orelse return db.fail("unknown type '{s}'", .{kind_text});
436 if (db.by_key.contains(key)) return db.fail("duplicate key", .{});
437 var fields: std.ArrayList(Field) = .empty;
438 for (start + 1..end) |i| {
439 db.error_line = i + 1;
440 const separator = std.mem.find(u8, lines[i], " = ").?;
441 const name = lines[i][2..separator];
442 var expression = lines[i][separator + 3 ..];
443 if (std.mem.endsWith(u8, expression, ",")) expression = expression[0 .. expression.len - 1];
444 const evaluated = try db.evaluate(expression);
445 if (findField(fields.items, name) != null) return db.fail("duplicate field '{s}'", .{name});
446 const normalized = try normalize(db.gpa, evaluated);
447 try fields.append(db.gpa, .{ .name = name, .raw = normalized, .value = try plain(db.gpa, normalized) });
448 }
449 db.error_line = end + 1;
450 for (requiredFields(kind)) |name| {
451 if (findField(fields.items, name) == null and
452 !(kind == .book and equal(name, "author") and findField(fields.items, "editor") != null))
453 return db.fail("missing field '{s}'", .{name});
454 }
455 const authors: ?NameList = if (findField(fields.items, "author")) |field| try NameList.parse(db.gpa, field.raw) else null;
456 const editors: ?NameList = if (findField(fields.items, "editor")) |field| try NameList.parse(db.gpa, field.raw) else null;
457 const primary = authors orelse editors.?;
458 var sort_title = try lower(db.gpa, findField(fields.items, "title").?.value);
459 for ([_][]const u8{ "a ", "an ", "the " }) |prefix| {
460 if (std.mem.startsWith(u8, sort_title, prefix)) {
461 sort_title = sort_title[prefix.len..];
462 break;
463 }
464 }
465 const entry = try db.gpa.create(Entry);
466 entry.* = .{
467 .key = key,
468 .kind = kind,
469 .fields = fields.items,
470 .authors = authors,
471 .editors = editors,
472 .sort_last = try lower(db.gpa, try plain(db.gpa, primary.names[0].last)),
473 .sort_title = sort_title,
474 };
475 try db.entries.append(db.gpa, entry);
476 try db.by_key.put(db.gpa, key, entry);
477 }
478 fn load(db: *Database, lines: []const []const u8, out: *Writer) Error!void {
479 const month_keys = [_][]const u8{ "jan", "feb", "mar", "apr", "may", "jun", "jul", "aug", "sep", "oct", "nov", "dec" };
480 for (month_keys, months) |key, value| try db.macros.put(db.gpa, key, value);
481 var errors: usize = 0;
482 var i: usize = 0;
483 while (i < lines.len) : (i += 1) {
484 const text = lines[i];
485 if (text.len == 0 or text[0] == '%') continue;
486 if (std.mem.startsWith(u8, text, "@string{")) {
487 db.error_line = i + 1;
488 const separator = std.mem.find(u8, text, " = ").?;
489 const name = text[8..separator];
490 db.readMacro(text, separator, name) catch |err| {
491 if (err != error.Invalid) return err;
492 errors += 1;
493 try line(out, "ERROR line {d}: {s}: {s}", .{ db.error_line, name, db.problem });
494 };
495 } else {
496 var end = i + 1;
497 while (!equal(lines[end], "}")) end += 1;
498 const brace = std.mem.findScalar(u8, text, '{').?;
499 const key = text[brace + 1 .. text.len - 1];
500 db.readEntry(lines, i, end, text[1..brace], key) catch |err| {
501 if (err != error.Invalid) return err;
502 errors += 1;
503 try line(out, "ERROR line {d}: {s}: {s}", .{ db.error_line, key, db.problem });
504 };
505 i = end;
506 }
507 }
508 try db.assignSuffixes();
509 try line(out, "LOADED entries={d} macros={d} errors={d}", .{ db.entries.items.len, db.macros.count() - 12, errors });
510 }
511 fn assignSuffixes(db: *Database) Allocator.Error!void {
512 for (db.entries.items) |entry| {
513 if (entry.suffix != 0) continue;
514 var group: std.ArrayList(*Entry) = .empty;
515 for (db.entries.items) |other| {
516 if (entry.primary().same(other.primary()) and equal(entry.raw("year"), other.raw("year"))) try group.append(db.gpa, other);
517 }
518 if (group.items.len < 2) continue;
519 std.sort.heap(*Entry, group.items, {}, Entry.titleLess);

569const ReferenceFormatter = struct {
570 parsed: [4][6][]const Part,
571
572 fn init(gpa: Allocator) Allocator.Error!ReferenceFormatter {
573 var result: ReferenceFormatter = undefined;
574 for (templates, 0..) |style, i| {
575 for (style, 0..) |template, j| {
576 var position: usize = 0;
577 result.parsed[i][j] = try parse(gpa, template, &position);
578 }

582 fn parse(gpa: Allocator, text: []const u8, i: *usize) Allocator.Error![]const Part {
583 var parts: std.ArrayList(Part) = .empty;
584 while (i.* < text.len and text[i.*] != '>') {
585 switch (text[i.*]) {
586 '{' => {
587 const end = i.* + std.mem.findScalar(u8, text[i.*..], '}').?;
588 const name = std.meta.stringToEnum(Placeholder, text[i.* + 1 .. end]).?;
589 try parts.append(gpa, .{ .field = name });
590 i.* = end + 1;
591 },
592 '<' => {
593 i.* += 1;
594 try parts.append(gpa, .{ .optional = try parse(gpa, text, i) });
595 i.* += 1;
596 },
597 else => {
598 try parts.append(gpa, .{ .literal = text[i.*] });
599 i.* += 1;
600 },
601 }
602 }
603 return parts.items;
604 }
605 fn editorSuffix(entry: Entry, style: Style) []const u8 {
606 if (entry.authors != null) return "";
607 const single = entry.primary().single();
608 return switch (style) {
609 .APA => if (single) " (Ed.)" else " (Eds.)",
610 .IEEE => if (single) ", Ed." else ", Eds.",
611 .CHICAGO => if (single) ", ed." else ", eds.",
612 .MLA => if (single) ", editor" else ", editors",
613 };
614 }
615 fn edition(gpa: Allocator, text: []const u8) Allocator.Error![]const u8 {
616 if (!digits(text)) return format(gpa, "{s} ed.", .{text});
617 const number = integer(text);
618 const suffix: []const u8 = if (number % 100 >= 11 and number % 100 <= 13) "th" else switch (number % 10) {
619 1 => "st",
620 2 => "nd",
621 3 => "rd",
622 else => "th",
623 };
624 return format(gpa, "{s}{s} ed.", .{ text, suffix });
625 }
626 fn values(gpa: Allocator, entry: Entry, style: Style, repeated: bool) Allocator.Error!Values {
627 var result: Values = .{};
628 for (known_fields) |name| result.set(std.meta.stringToEnum(Placeholder, name).?, entry.value(name));
629 result.set(.author, try join(gpa, "", &.{ if (repeated) "---" else try entry.primary().authors(gpa, style), editorSuffix(entry, style) }));
630 result.set(.editor, if (entry.editors) |names| try names.editors(gpa, style) else "");
631 result.set(.eds, if (entry.editors) |names| if (names.single()) "Ed." else "Eds." else "");
632 result.set(.title, try plain(gpa, try titleCase(gpa, entry.raw("title"), style)));
633 result.set(.year, try entry.year(gpa, style));
634 if (entry.has("pages")) {
635 var pages: Text = .{ .gpa = gpa };
636 const text = entry.value("pages");
637 var i: usize = 0;
638 while (i < text.len) {
639 try pages.byte(text[i]);
640 i += if (std.mem.startsWith(u8, text[i..], "--")) @as(usize, 2) else 1;
641 }
642 result.set(.pages, pages.result());
643 result.set(.pp, try format(gpa, "{s} {s}", .{
644 @as([]const u8, if (std.mem.findScalar(u8, pages.result(), '-') != null) "pp." else "p."), pages.result(),
645 }));
646 }
647 if (entry.has("edition")) result.set(.edition, try edition(gpa, entry.value("edition")));
648 const month = entry.value("month");
649 result.set(.mon, month);
650 for (months, month_short) |full, short| {
651 if (equal(month, full)) {
652 result.set(.mon, short);
653 break;
654 }
655 }
656 return result;
657 }
658 fn render(text: *Text, parts: []const Part, fields: Values) Allocator.Error!void {
659 for (parts) |part| {
660 switch (part) {
661 .field => |name| try text.add(fields.get(name)),
662 .optional => |children| {
663 var visible = true;
664 for (children) |child| {
665 const empty = switch (child) {
666 .field => |name| fields.get(name).len == 0,
667 else => false,
668 };
669 if (empty) {
670 visible = false;
671 break;
672 }
673 }
674 if (visible) try render(text, children, fields);
675 },
676 .literal => |c| {
677 const current = text.result();
678 if (c != '.' or current.len == 0 or std.mem.findScalar(u8, ".?!", current[current.len - 1]) == null) try text.byte(c);
679 },
680 }
681 }
682 }

695 return result.result();

722fn shortName(gpa: Allocator, entry: Entry, style: Style, parenthetical: bool) Allocator.Error![]const u8 {
723 const primary = entry.primary();
724 const first = try primary.names[0].surname(gpa);

729 3 => format(gpa, "{s}, {s}, and {s}", .{ first, try primary.names[1].surname(gpa), try primary.names[2].surname(gpa) }),
730 else => unreachable,

733fn numberLess(_: void, a: usize, b: usize) bool {
734 return a < b;
735}
736fn numberRanges(gpa: Allocator, numbers: []usize) Allocator.Error![]const u8 {
737 std.sort.heap(usize, numbers, {}, numberLess);
738 var parts: std.ArrayList([]const u8) = .empty;
739 var i: usize = 0;
740 while (i < numbers.len) {
741 var end = i;
742 while (end + 1 < numbers.len and numbers[end + 1] == numbers[end] + 1) end += 1;
743 if (end - i >= 2) {
744 try parts.append(gpa, try format(gpa, "[{d}]-[{d}]", .{ numbers[i], numbers[end] }));
745 } else {
746 for (numbers[i .. end + 1]) |number| try parts.append(gpa, try format(gpa, "[{d}]", .{number}));
747 }
748 i = end + 1;
749 }
750 return join(gpa, ", ", parts.items);
751}
752fn citationText(gpa: Allocator, citation: Citation, style: Style, order: []const *Entry) Allocator.Error![]const u8 {
753 var parts: std.ArrayList([]const u8) = .empty;
754 if (citation.kind == .citeauthor or citation.kind == .citeyear) {
755 for (citation.entries) |entry| try parts.append(gpa, if (citation.kind == .citeauthor)
756 try shortName(gpa, entry.*, style, false)
757 else
758 try entry.year(gpa, style));
759 return join(gpa, if (citation.kind == .citeauthor) "; " else ", ", parts.items);
760 }
761 const parenthetical = citation.kind != .citet;
762 if (style == .IEEE) {
763 if (parenthetical) {
764 const numbers = try gpa.alloc(usize, citation.entries.len);
765 for (citation.entries, 0..) |entry, i| numbers[i] = ieeeNumber(order, entry);
766 return numberRanges(gpa, numbers);
767 }
768 for (citation.entries) |entry| try parts.append(gpa, try format(gpa, "{s} [{d}]", .{
769 try shortName(gpa, entry.*, style, false), ieeeNumber(order, entry),
770 }));
771 return join(gpa, "; ", parts.items);
772 }
773 if (style == .MLA) {
774 for (citation.entries) |entry| try parts.append(gpa, try shortName(gpa, entry.*, style, parenthetical));
775 const text = try join(gpa, "; ", parts.items);
776 return if (parenthetical) format(gpa, "({s})", .{text}) else text;
777 }
778 const entries = try gpa.dupe(*Entry, citation.entries);
779 if (parenthetical) std.sort.heap(*Entry, entries, {}, Entry.bibLess);
780 var i: usize = 0;
781 while (i < entries.len) {
782 const name = try shortName(gpa, entries[i].*, style, parenthetical);
783 var end = i + 1;
784 while (end < entries.len and equal(name, try shortName(gpa, entries[end].*, style, parenthetical))) end += 1;
785 var years: std.ArrayList([]const u8) = .empty;
786 for (entries[i..end]) |entry| try years.append(gpa, try entry.year(gpa, style));
787 const year_text = try join(gpa, ", ", years.items);
788 try parts.append(gpa, if (parenthetical)
789 try format(gpa, "{s}{s}{s}", .{ name, @as([]const u8, if (style == .APA) ", " else " "), year_text })
790 else
791 try format(gpa, "{s} ({s})", .{ name, year_text }));
792 i = end;
793 }
794 const text = try join(gpa, "; ", parts.items);
795 return if (parenthetical) format(gpa, "({s})", .{text}) else text;
796}

843 fn cite(session: *Session, text: []const u8, style: Style) Error!void {
844 var citations: std.ArrayList(Citation) = .empty;
845 var position: usize = 0;
846 while (std.mem.findScalar(u8, text[position..], '\\')) |offset| {
847 const start = position + offset;
848 const brace = start + std.mem.findScalar(u8, text[start..], '{').?;
849 const end = brace + std.mem.findScalar(u8, text[brace..], '}').?;
850 var entries: std.ArrayList(*Entry) = .empty;
851 var keys = std.mem.splitScalar(u8, text[brace + 1 .. end], ',');
852 while (keys.next()) |key| {
853 const entry = try session.lookup(key);
854 var seen = false;
855 for (entries.items) |other| {
856 if (other == entry) {
857 seen = true;
858 break;
859 }
860 }
861 if (!seen) try entries.append(session.gpa, entry);
862 }
863 try citations.append(session.gpa, .{
864 .start = start,
865 .end = end + 1,
866 .kind = std.meta.stringToEnum(CitationKind, text[start + 1 .. brace]).?,
867 .entries = entries.items,
868 });
869 position = end + 1;
870 }
871 for (citations.items) |citation| {
872 for (citation.entries) |entry| {
873 if (!session.manager.recorded(entry)) try session.manager.record.append(session.manager.db.gpa, entry);
874 }
875 }
876 const order = try session.manager.ieeeOrder(session.gpa);
877 position = 0;
878 for (citations.items) |citation| {
879 try session.out.writeAll(text[position..citation.start]);
880 try session.out.writeAll(try citationText(session.gpa, citation, style, order));
881 position = citation.end;
882 }
883 try line(session.out, "{s}", .{text[position..]});
884 }
885 fn keygen(session: *Session) Error!void {

891 var ordinal: usize = 0;
892 for (bases, 0..) |base, j| {
893 if (equal(base, bases[i])) {

896 }
897 }
898 try session.out.print("{s} -> {s}", .{ entry.key, bases[i] });
899 if (count > 1) try session.out.writeByte('a' + @as(u8, @intCast(ordinal)));
900 try session.out.writeByte('\n');
901 }

908 fn check(session: *Session) Error!void {
909 var titles: std.StringHashMapUnmanaged([]const u8) = .empty;
910 for (session.manager.db.entries.items) |entry| {
911 if (entry.has("pages")) {
912 const pages = entry.value("pages");
913 if (std.mem.findScalar(u8, pages, '-') != null and std.mem.find(u8, pages, "--") == null)
914 try session.warn(entry.key, "pages use a single hyphen", .{});
915 var parts: std.ArrayList([]const u8) = .empty;
916 var tokens = std.mem.tokenizeScalar(u8, pages, '-');
917 while (tokens.next()) |part| try parts.append(session.gpa, part);
918 if (parts.items.len == 2 and digits(parts.items[0]) and digits(parts.items[1]) and integer(parts.items[0]) > integer(parts.items[1]))
919 try session.warn(entry.key, "pages are reversed", .{});
920 }
921 const year = entry.value("year");
922 if (year.len != 4 or !digits(year)) try session.warn(entry.key, "year is not 4 digits", .{});
923 const title = entry.value("title");
924 var has_lower = false;
925 var upper_count: usize = 0;
926 for (title) |c| {
927 if (lowercase(c)) has_lower = true;
928 if (uppercase(c)) upper_count += 1;
929 }
930 if (!has_lower and upper_count >= 2) try session.warn(entry.key, "title is all caps", .{});
931 if (entry.kind == .article and !entry.has("doi")) try session.warn(entry.key, "article has no doi", .{});
932 const lowered = try lower(session.gpa, title);
933 if (titles.get(lowered)) |first| {
934 try session.warn(entry.key, "same title as {s}", .{first});
935 } else {
936 try titles.put(session.gpa, lowered, entry.key);
937 }
938 for (try unknownFields(session.gpa, entry.*)) |name| try session.warn(entry.key, "unknown field '{s}'", .{name});
939 }
940 try line(session.out, "CHECK warnings={d}", .{session.warning_count});
941 }

952 fn stats(session: *Session) Error!void {
953 const Pair = struct { first: usize, second: usize };
954 var types: Counts = .empty;
955 var years: Counts = .empty;
956 var authors: Counts = .empty;
957 var author_names: std.ArrayList([]const u8) = .empty;
958 var pairs: std.AutoHashMapUnmanaged(Pair, usize) = .empty;
959 for (session.manager.db.entries.items) |entry| {
960 try countValue(session.gpa, &types, @tagName(entry.kind));
961 try countValue(session.gpa, &years, entry.value("year"));
962 const list = entry.authors orelse continue;
963 const names = try session.gpa.alloc([]const u8, list.names.len);
964 for (list.names, 0..) |name, i| names[i] = try name.notation(session.gpa, false, false);
965 std.sort.heap([]const u8, names, {}, less);
966 var unique: std.ArrayList(usize) = .empty;
967 for (names, 0..) |name, i| {
968 if (i > 0 and equal(names[i - 1], name)) continue;
969 const id = authors.get(name) orelse blk: {
970 const next = author_names.items.len;
971 try authors.put(session.gpa, name, next);
972 try author_names.append(session.gpa, name);
973 break :blk next;
974 };
975 try unique.append(session.gpa, id);
976 }
977 for (unique.items, 0..) |a, i| {
978 for (unique.items[i + 1 ..]) |b| {
979 const pair: Pair = .{ .first = a, .second = b };
980 try pairs.put(session.gpa, pair, (pairs.get(pair) orelse 0) + 1);
981 }
982 }
983 }
984 try line(session.out, "ENTRIES {d}", .{session.manager.db.entries.items.len});
985 try printCounts(session.gpa, session.out, "TYPE", types);
986 try printCounts(session.gpa, session.out, "YEAR", years);
987 try line(session.out, "AUTHORS {d}", .{author_names.items.len});
988 var best: ?Pair = null;
989 var best_count: usize = 0;
990 var iterator = pairs.iterator();
991 while (iterator.next()) |item| {
992 const pair = item.key_ptr.*;
993 const count = item.value_ptr.*;
994 var take = count > best_count;
995 if (count == best_count and best != null) {
996 const first_order = std.mem.order(u8, author_names.items[pair.first], author_names.items[best.?.first]);
997 take = first_order == .lt or first_order == .eq and less({}, author_names.items[pair.second], author_names.items[best.?.second]);
998 }
999 if (take) {
1000 best = pair;
1001 best_count = count;
1002 }
1003 }
1004 if (best) |pair| {
1005 try line(session.out, "PAIR {s} & {s} {d}", .{ author_names.items[pair.first], author_names.items[pair.second], best_count });
1006 } else {
1007 try line(session.out, "PAIR none", .{});
1008 }
1009 }
1010 fn execute(session: *Session, text: []const u8) Error!void {
1011 const first_space = std.mem.findScalar(u8, text, ' ');
1012 const command_text = text[0 .. first_space orelse text.len];
1013 const command = std.meta.stringToEnum(Command, command_text) orelse return session.fail("unknown command '{s}'", .{command_text});
1014 var words: std.ArrayList([]const u8) = .empty;
1015 if ((command == .CITE or command == .SEARCH) and first_space != null) {
1016 const first = first_space.?;
1017 try words.append(session.gpa, command_text);
1018 if (std.mem.findScalar(u8, text[first + 1 ..], ' ')) |offset| {
1019 const second = first + 1 + offset;
1020 try words.append(session.gpa, text[first + 1 .. second]);
1021 try words.append(session.gpa, text[second + 1 ..]);
1022 } else {
1023 try words.append(session.gpa, text[first + 1 ..]);
1024 }
1025 } else {
1026 var tokens = std.mem.splitScalar(u8, text, ' ');
1027 while (tokens.next()) |word| try words.append(session.gpa, word);
1028 }
1029 const expected: usize = switch (command) {
1030 .FORMAT, .CITE, .SEARCH => 2,
1031 .BIBLIOGRAPHY, .EXPORT => 1,
1032 .KEYGEN, .CHECK, .STATS => 0,
1033 };
1034 const arguments = words.items[1..];
1035 if (if (command == .FORMAT) arguments.len < expected else arguments.len != expected)
1036 return session.fail("wrong number of arguments", .{});
1037 var style: Style = .APA;
1038 if (command == .FORMAT or command == .BIBLIOGRAPHY or command == .CITE)
1039 style = std.meta.stringToEnum(Style, arguments[0]) orelse return session.fail("unknown style '{s}'", .{arguments[0]});
1040 const manager = session.manager;
1041 switch (command) {
1042 .FORMAT => {
1043 const order = try manager.ieeeOrder(session.gpa);
1044 for (arguments[1..]) |key| {
1045 if (manager.db.by_key.get(key)) |entry| {
1046 try line(session.out, "{s}", .{try manager.formatter.reference(session.gpa, entry.*, style, ieeeNumber(order, entry), false)});
1047 } else {
1048 try line(session.out, "ERROR unknown key '{s}'", .{key});
1049 }
1050 }
1051 },
1052 .BIBLIOGRAPHY => {
1053 try line(session.out, "{s}", .{@as([]const u8, if (style == .MLA) "Works Cited" else "References")});
1054 const ieee_order = try manager.ieeeOrder(session.gpa);
1055 const order = try session.gpa.dupe(*Entry, ieee_order);
1056 if (style != .IEEE) std.sort.heap(*Entry, order, {}, Entry.bibLess);
1057 var previous: ?NameList = null;
1058 for (order) |entry| {
1059 const repeated = (style == .CHICAGO or style == .MLA) and previous != null and previous.?.same(entry.primary());
1060 try line(session.out, "{s}", .{try manager.formatter.reference(session.gpa, entry.*, style, ieeeNumber(ieee_order, entry), repeated)});
1061 previous = entry.primary();
1062 }
1063 },
1064 .CITE => try session.cite(arguments[1], style),
1065 .KEYGEN => try session.keygen(),
1066 .CHECK => try session.check(),
1067 .SEARCH => {
1068 const needle = try lower(session.gpa, arguments[1]);
1069 var matches: std.ArrayList([]const u8) = .empty;
1070 for (manager.db.entries.items) |entry| {
1071 if (entry.has(arguments[0]) and std.mem.find(u8, try lower(session.gpa, entry.value(arguments[0])), needle) != null)
1072 try matches.append(session.gpa, entry.key);
1073 }
1074 if (matches.items.len == 0) {
1075 try line(session.out, "MATCHES 0", .{});
1076 } else {
1077 try line(session.out, "MATCHES {d}: {s}", .{ matches.items.len, try join(session.gpa, " ", matches.items) });
1078 }
1079 },
1080 .EXPORT => try session.exportEntry((try session.lookup(arguments[0])).*),
1081 .STATS => try session.stats(),
1082 }
1083 }
1084};

1086fn readLine(gpa: Allocator, input: *std.Io.Reader) ![]const u8 {
1087 var buffer: Writer.Allocating = .init(gpa);
1088 defer buffer.deinit();
1089 _ = try input.streamDelimiterEnding(&buffer.writer, '\n');
1090 if (input.seek < input.end) input.toss(1) else if (buffer.written().len == 0) return error.EndOfStream;
1091 const text = try buffer.toOwnedSlice();
1092 return if (text.len > 0 and text[text.len - 1] == '\r') text[0 .. text.len - 1] else text;
1093}
