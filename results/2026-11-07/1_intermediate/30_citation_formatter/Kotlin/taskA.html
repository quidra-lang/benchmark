12private fun lower(c: Char) = c in 'a'..'z'
13private fun upper(c: Char) = c in 'A'..'Z'
14private fun letter(c: Char) = lower(c) || upper(c)
15private fun digit(c: Char) = c in '0'..'9'
16private fun lowercase(c: Char) = if (upper(c)) c + 32 else c
17private fun uppercase(c: Char) = if (lower(c)) c - 32 else c
18private fun lowerText(text: String) = text.map(::lowercase).joinToString("")
19private fun plain(text: String) = text.replace("{", "").replace("}", "")
20private fun letters(text: String) = lowerText(text.filter(::letter))
21private fun digits(text: String) = text.isNotEmpty() && text.all(::digit)

23private fun splitDepth(text: String, separator: String): MutableList<String> {
24 val result = mutableListOf<String>()
25 var start = 0
26 var i = 0
27 var depth = 0
28 while (i < text.length) {
29 if (depth == 0 && text.startsWith(separator, i)) {
30 result += text.substring(start, i)
31 i += separator.length
32 start = i
33 } else {
34 if (text[i] == '{') depth++
35 if (text[i] == '}') depth--
36 i++
37 }
38 }
39 result += text.substring(start)
40 return result
41}

46private fun evaluate(expression: String, macros: Map<String, String>, line: Int): String {
47 var depth = 0
48 for (c in expression) {
49 if (c == '{') depth++
50 if (c == '}') depth--
51 if (depth < 0) throw LoadProblem(line, "unbalanced braces")
52 }
53 if (depth != 0) throw LoadProblem(line, "unbalanced braces")
54 return buildString {
55 for (term in expression.split(" # ")) when {
56 term[0] == '{' || term[0] == '"' -> append(term.substring(1, term.length - 1))
57 digit(term[0]) -> append(term)
58 else -> append(macros[term] ?: throw LoadProblem(line, "unknown macro '$term'"))
59 }
60 }
61}

63private data class Accent(val value: Char, val width: Int)
64private fun accent(text: String, at: Int): Accent? {
65 if (at + 1 >= text.length || text[at] != '\\') return null
66 val command = text[at + 1]
67 if (command == '&') return Accent('&', 2)
68 val symbol = command in "'\"`^~=."
69 if (symbol && at + 2 < text.length && letter(text[at + 2])) return Accent(text[at + 2], 3)
70 if ((symbol || command in "cuvH") && at + 4 < text.length && text[at + 2] == '{' && letter(text[at + 3]) && text[at + 4] == '}') return Accent(text[at + 3], 5)
71 return null
72}

74private fun normalize(text: String): String {
75 val result = buildString {
76 var i = 0
77 while (i < text.length) {
78 val wrapped = if (text[i] == '{') accent(text, i + 1) else null
79 if (wrapped != null && i + wrapped.width + 1 < text.length && text[i + wrapped.width + 1] == '}') {
80 append(wrapped.value)
81 i += wrapped.width + 2
82 } else {
83 val current = accent(text, i)
84 if (current == null) append(text[i++]) else {
85 append(current.value)
86 i += current.width
87 }
88 }
89 }
90 }
91 return result.replace('~', ' ').split(' ').filter { it.isNotEmpty() }.joinToString(" ")
92}

94private data class Name(val first: String, val von: String, val last: String, val jr: String) {
95 companion object {
96 fun parse(text: String): Name {
97 val parts = splitDepth(text, ",").map { it.trim() }
98 val words = splitDepth(parts.first(), " ")
99 val lowerWords = (0 until words.lastIndex).filter { lower(words[it][0]) }
100 val firstLower = lowerWords.firstOrNull() ?: -1
101 val lastLower = lowerWords.lastOrNull() ?: -1
102 fun joined(from: Int, until: Int) = words.subList(from, until).joinToString(" ")
103 if (parts.size == 1) {
104 if (firstLower < 0) return Name(joined(0, words.lastIndex), "", words.last(), "")
105 return Name(joined(0, firstLower), joined(firstLower, lastLower + 1), joined(lastLower + 1, words.size), "")
106 }
107 return Name(parts.last(), if (lastLower < 0) "" else joined(0, lastLower + 1), joined(lastLower + 1, words.size), if (parts.size == 3) parts[1] else "")
108 }
109 }
110 fun surname() = plain(if (von.isEmpty()) last else "$von $last")
111 fun initials() = plain(first).split(' ').filter { it.isNotEmpty() }.joinToString(" ") { word ->
112 word.split('-').joinToString("-") { piece -> "${uppercase(piece.first(::letter))}." }
113 }
114 fun text(inverted: Boolean, abbreviated: Boolean): String {
115 val given = if (abbreviated) initials() else plain(first)
116 var result = surname()
117 if (first.isNotEmpty()) result = if (inverted) "$result, $given" else "$given $result"
118 if (jr.isNotEmpty()) result += ", ${plain(jr)}"
119 return result
120 }
121}

134private enum class Style { APA, IEEE, CHICAGO, MLA }
135private fun titleCase(title: String, style: Style): String {
136 if (style == Style.IEEE) return title
137 if (style == Style.APA) {
138 val result = title.toCharArray()
139 var depth = 0
140 for (i in result.indices) {
141 val c = result[i]
142 when {
143 c == '{' -> depth++
144 c == '}' -> depth--
145 depth == 0 && letter(c) -> result[i] = if (i == 0 || i >= 2 && title.substring(i - 2, i) == ": ") uppercase(c) else lowercase(c)
146 }
147 }
148 return result.concatToString()
149 }
150 val words = splitDepth(title, " ")
151 for (i in words.indices) {
152 val word = words[i]
153 if (word.startsWith('{')) continue
154 val makeLower = i > 0 && i < words.lastIndex && !words[i - 1].endsWith(':') && letters(word) in smallWords
155 val result = word.toCharArray()
156 var depth = 0
157 var firstLetter = true
158 for (j in result.indices) {
159 val c = result[j]
160 when {
161 c == '{' -> depth++
162 c == '}' -> depth--
163 letter(c) -> {
164 if (depth == 0 && makeLower) result[j] = lowercase(c)
165 else if (depth == 0 && firstLetter) result[j] = uppercase(c)
166 firstLetter = false
167 }
168 }
169 }
170 words[i] = result.concatToString()
171 }
172 return words.joinToString(" ")
173}

181private fun renderNames(list: NameList, style: Style, editor: Boolean): String {
182 val names = list.names.mapIndexed { i, name -> name.text(!editor && (style == Style.APA || style != Style.IEEE && i == 0), style == Style.APA || style == Style.IEEE) }
183 if (editor) return if (list.etAl) "${names.first()} et al." else listed(names, if (style == Style.APA) "&" else "and", true)

192 Style.MLA -> if (list.etAl || names.size >= 3) "${names.first()}, et al." else listed(names, "and", false)

250private fun optionalEnd(pattern: String, start: Int): Int {
251 var depth = 1
252 var i = start
253 while (depth > 0) {
254 i++
255 if (pattern[i] == '<') depth++
256 if (pattern[i] == '>') depth--
257 }
258 return i
259}

261private fun optionalPresent(pattern: String, start: Int, end: Int, values: Map<String, String>): Boolean {
262 var i = start
263 while (i < end) {
264 when (pattern[i]) {
265 '<' -> i = optionalEnd(pattern, i)
266 '{' -> {
267 val close = pattern.indexOf('}', i)
268 if (values[pattern.substring(i + 1, close)].isNullOrEmpty()) return false
269 i = close
270 }
271 }
272 i++
273 }
274 return true
275}

277private fun renderTemplate(pattern: String, start: Int, end: Int, values: Map<String, String>, result: StringBuilder) {
278 var i = start
279 while (i < end) {
280 val c = pattern[i]
281 when (c) {
282 '<' -> {
283 val close = optionalEnd(pattern, i)
284 if (optionalPresent(pattern, i + 1, close, values)) renderTemplate(pattern, i + 1, close, values, result)
285 i = close
286 }
287 '{' -> {
288 val close = pattern.indexOf('}', i)
289 result.append(values[pattern.substring(i + 1, close)] ?: "")
290 i = close
291 }
292 else -> if (c != '.' || result.isEmpty() || result.last() !in ".?!") result.append(c)
293 }
294 i++
295 }
296}

298private fun reference(entry: Entry, style: Style, number: Int, repeated: Boolean): String {
299 val values = fields.associateWith { plain(entry.values[it] ?: "") }.toMutableMap()
300 var author = if (repeated) "---" else renderNames(entry.primary, style, false)
301 if (entry.authors == null) {
302 val annotations = listOf(listOf(" (Ed.)", " (Eds.)"), listOf(", Ed.", ", Eds."), listOf(", ed.", ", eds."), listOf(", editor", ", editors"))
303 author += annotations[style.ordinal][if (entry.primary.single()) 0 else 1]
304 }
305 values["author"] = author
306 entry.editors?.let {
307 values["editor"] = renderNames(it, style, true)
308 values["eds"] = if (it.single()) "Ed." else "Eds."
309 }
310 values["title"] = plain(titleCase(entry.values.getValue("title"), style))
311 values["year"] = entry.yearLabel(style)
312 entry.values["pages"]?.let {
313 val pages = plain(it).replace("--", "-")
314 values["pages"] = pages
315 values["pp"] = (if ('-' in pages) "pp. " else "p. ") + pages
316 }
317 entry.values["edition"]?.let {
318 var edition = plain(it)
319 if (digits(edition)) {
320 val n = edition.toInt()
321 val suffix = if (n % 100 in 11..13) "th" else when (n % 10) { 1 -> "st"; 2 -> "nd"; 3 -> "rd"; else -> "th" }
322 edition = "$n$suffix"
323 }
324 values["edition"] = "$edition ed."
325 }
326 entry.values["month"]?.let {
327 val month = plain(it)
328 val index = months.indexOf(month)
329 values["mon"] = if (index < 0) month else monthShort[index]
330 }
331 val kind = when { entry.kind == 3 -> 2; entry.kind > 3 -> entry.kind - 1; else -> entry.kind }
332 val pattern = templates[style.ordinal][kind]
333 val result = StringBuilder(if (style == Style.IEEE) "[$number] " else "")
334 renderTemplate(pattern, 0, pattern.length, values, result)
335 return result.toString()
336}

342private class Database(lines: List<String>) {
343 val entries = mutableListOf<Entry>()
344 val record = mutableListOf<Entry>()
345 val lookup = sortedMapOf<String, Entry>()
346 val macros = monthMacros.zip(months).toMap().toMutableMap()
347 val output = mutableListOf<String>()
348 var macroCount = 0
349
350 init {
351 var i = 0
352 while (i < lines.size) {
353 val line = lines[i]
354 when {
355 line.isEmpty() || line.startsWith('%') -> Unit
356 line.startsWith("@string{") -> {
357 val equals = line.indexOf(" = ")
358 val name = line.substring(8, equals)
359 try {
360 val value = evaluate(line.substring(equals + 3, line.length - 1), macros, i + 1)
361 if (name in macros) throw LoadProblem(i + 1, "duplicate macro")
362 macros[name] = value
363 macroCount++
364 } catch (error: LoadProblem) { report(name, error) }
365 }
366 else -> {
367 var end = i + 1
368 while (lines[end] != "}") end++
369 val brace = line.indexOf('{')
370 val kind = line.substring(1, brace)
371 val key = line.substring(brace + 1, line.length - 1)
372 try { readEntry(lines, i, end, kind, key) } catch (error: LoadProblem) { report(key, error) }
373 i = end
374 }
375 }
376 i++
377 }
378 output += "LOADED entries=${entries.size} macros=$macroCount errors=${output.size}"
379 val assigned = mutableSetOf<Entry>()
380 for (entry in entries) {
381 if (entry in assigned) continue
382 val group = entries.filter { it.primary == entry.primary && it.values["year"] == entry.values["year"] }
383 .sortedWith(compareBy({ it.sortTitle }, { it.key }))
384 assigned.addAll(group)
385 if (group.size >= 2) group.forEachIndexed { index, item -> item.suffix = ('a' + index).toString() }
386 }

388 private fun report(name: String, error: LoadProblem) { output += "ERROR line ${error.line}: $name: ${error.message}" }
389 private fun readEntry(lines: List<String>, start: Int, end: Int, kindName: String, key: String) {
390 val kind = types.indexOf(kindName)
391 if (kind < 0) throw LoadProblem(start + 1, "unknown type '$kindName'")
392 if (key in lookup) throw LoadProblem(start + 1, "duplicate key")
393 val values = mutableMapOf<String, String>()
394 for (i in start + 1 until end) {
395 val equals = lines[i].indexOf(" = ")
396 val name = lines[i].substring(2, equals)
397 val expression = lines[i].substring(equals + 3).removeSuffix(",")
398 val value = evaluate(expression, macros, i + 1)
399 if (name in values) throw LoadProblem(i + 1, "duplicate field '$name'")
400 values[name] = normalize(value)
401 }
402 for (name in required[kind]) if (name !in values && !(kind == 1 && name == "author" && "editor" in values)) throw LoadProblem(end + 1, "missing field '$name'")
403 val entry = Entry(key, kind, values)
404 entries += entry
405 lookup[key] = entry
406 }

412 private fun citationText(citation: Citation, style: Style, numbers: Map<Entry, Int>): String {
413 val parenthetical = citation.kind == CitationKind.PARENTHETICAL
414 if (citation.kind == CitationKind.AUTHOR || citation.kind == CitationKind.YEAR || style == Style.MLA || style == Style.IEEE && !parenthetical) {
415 val text = citation.entries.joinToString(if (citation.kind == CitationKind.YEAR) ", " else "; ") { entry ->
416 if (citation.kind == CitationKind.YEAR) entry.yearLabel(style)
417 else entry.shortName(style, false) + if (style == Style.IEEE && citation.kind == CitationKind.NARRATIVE) " [${numbers.getValue(entry)}]" else ""
418 }
419 return if (style == Style.MLA && parenthetical) "($text)" else text
420 }
421 val parts = mutableListOf<String>()
422 if (style == Style.IEEE) {
423 val sorted = citation.entries.map { numbers.getValue(it) }.sorted()
424 var i = 0
425 while (i < sorted.size) {
426 var end = i + 1
427 while (end < sorted.size && sorted[end] == sorted[end - 1] + 1) end++
428 if (end - i >= 3) parts += "[${sorted[i]}]-[${sorted[end - 1]}]"
429 else for (j in i until end) parts += "[${sorted[j]}]"
430 i = end
431 }
432 return parts.joinToString(", ")
433 }
434 val entries = if (parenthetical) citation.entries.sortedWith(bibliography) else citation.entries
435 var i = 0
436 while (i < entries.size) {
437 val name = entries[i].shortName(style, parenthetical)
438 var end = i + 1
439 while (end < entries.size && entries[end].shortName(style, parenthetical) == name) end++
440 val years = entries.subList(i, end).joinToString(", ") { it.yearLabel(style) }
441 parts += if (parenthetical) name + (if (style == Style.APA) ", " else " ") + years else "$name ($years)"
442 i = end
443 }
444 val text = parts.joinToString("; ")
445 return if (parenthetical) "($text)" else text
446 }
447 private fun cite(style: Style, text: String): String {
448 val citations = mutableListOf<Citation>()
449 var start = text.indexOf('\\')
450 while (start >= 0) {
451 val brace = text.indexOf('{', start)
452 val end = text.indexOf('}', brace)
453 val kind = when (text.substring(start + 1, brace)) {
454 "citet" -> CitationKind.NARRATIVE
455 "citeauthor" -> CitationKind.AUTHOR
456 "citeyear" -> CitationKind.YEAR
457 else -> CitationKind.PARENTHETICAL
458 }
459 val seen = linkedSetOf<Entry>()
460 for (key in text.substring(brace + 1, end).split(',')) seen += lookup[key] ?: throw Problem("unknown key '$key'")
461 citations += Citation(start, end + 1, kind, seen.toList())
462 start = text.indexOf('\\', end + 1)
463 }
464 val recorded = record.toMutableSet()
465 for (citation in citations) for (entry in citation.entries) if (recorded.add(entry)) record += entry
466 val numbers = numbering().numbers
467 return buildString {
468 var position = 0
469 for (citation in citations) {
470 append(text, position, citation.start)
471 append(citationText(citation, style, numbers))
472 position = citation.end
473 }
474 append(text, position, text.length)
475 }

490 private fun check(): List<String> {
491 val result = mutableListOf<String>()
492 val titles = mutableMapOf<String, String>()
493 for (entry in entries) {
494 val warnings = mutableListOf<String>()
495 entry.values["pages"]?.let {
496 val pages = plain(it)

498 val parts = pages.split('-').filter(String::isNotEmpty)

500 }

502 if (entry.plainTitle.none(::lower) && entry.plainTitle.count(::upper) >= 2) warnings += "title is all caps"
503 if (entry.kind == 0 && "doi" !in entry.values) warnings += "article has no doi"
504 val earlier = titles[entry.lowerTitle]
505 if (earlier != null) warnings += "same title as $earlier" else titles[entry.lowerTitle] = entry.key
506 for (field in entry.unknownFields()) warnings += "unknown field '$field'"
507 result += warnings.map { "WARN ${entry.key}: $it" }
508 }
509 result += "CHECK warnings=${result.size}"
510 return result

521 val yearCounts = entries.groupingBy { it.year }.eachCount().toSortedMap()
522 val authors = mutableSetOf<String>()
523 val pairs = sortedMapOf<Pair<String, String>, Int>(compareBy({ it.first }, { it.second }))
524 for (entry in entries) entry.authors?.let { list ->
525 val names = list.names.map { it.text(false, false) }.distinct().sorted()
526 authors.addAll(names)
527 for (a in names.indices) for (b in a + 1 until names.size) {
528 val pair = names[a] to names[b]
529 pairs[pair] = (pairs[pair] ?: 0) + 1
530 }
531 }
532 val result = mutableListOf("ENTRIES ${entries.size}")
533 result += typeCounts.map { (type, count) -> "TYPE $type $count" }
534 result += yearCounts.map { (year, count) -> "YEAR $year $count" }
535 result += "AUTHORS ${authors.size}"
536 val best = pairs.entries.maxByOrNull { it.value }
537 result += if (best == null) "PAIR none" else "PAIR ${best.key.first} & ${best.key.second} ${best.value}"
538 return result
539 }
540 fun command(line: String): List<String> {
541 val command = line.substringBefore(' ')
542 val words = line.split(' ', limit = if (command == "CITE" || command == "SEARCH") 3 else 0)
543 val arity = when (command) {
544 "FORMAT" -> -1
545 "BIBLIOGRAPHY", "EXPORT" -> 1
546 "CITE", "SEARCH" -> 2
547 "KEYGEN", "CHECK", "STATS" -> 0
548 else -> throw Problem("unknown command '$command'")
549 }
550 if (if (arity == -1) words.size < 3 else words.size != arity + 1) throw Problem("wrong number of arguments")
551 val style = if (command in listOf("FORMAT", "BIBLIOGRAPHY", "CITE"))
552 Style.entries.firstOrNull { it.name == words[1] } ?: throw Problem("unknown style '${words[1]}'") else Style.APA
553 return when (command) {
554 "FORMAT" -> {
555 val numbers = numbering().numbers
556 words.drop(2).map { key -> lookup[key]?.let { reference(it, style, numbers.getValue(it), false) } ?: "ERROR unknown key '$key'" }
557 }
558 "BIBLIOGRAPHY" -> {
559 val numbering = numbering()
560 val order = if (style == Style.IEEE) numbering.order else numbering.order.sortedWith(bibliography)
561 listOf(if (style == Style.MLA) "Works Cited" else "References") + order.mapIndexed { index, entry ->
562 val repeated = index > 0 && (style == Style.CHICAGO || style == Style.MLA) && entry.primary == order[index - 1].primary
563 reference(entry, style, numbering.numbers.getValue(entry), repeated)
564 }
565 }
566 "CITE" -> listOf(cite(style, words[2]))
567 "KEYGEN" -> keygen()
568 "CHECK" -> check()
569 "EXPORT" -> export(words[1])
570 "STATS" -> stats()
571 else -> {
572 val needle = lowerText(words[2])
573 val matches = entries.filter { words[1] in it.values && needle in lowerText(plain(it.values.getValue(words[1]))) }.map { it.key }
574 listOf(if (matches.isEmpty()) "MATCHES 0" else "MATCHES ${matches.size}: ${matches.joinToString(" ")}")
575 }
576 }
577 }
578}

580fun main() {
581 val reader = System.`in`.bufferedReader(Charsets.US_ASCII)
582 val lines = mutableListOf<String>()
583 while (true) {
584 val line = reader.readLine()
585 if (line == "%%") break
586 lines += line
587 }
588 val database = Database(lines)
589 val writer = System.out.bufferedWriter(Charsets.UTF_8)
590 fun write(line: String) { writer.write(line); writer.write("\n") }
591 database.output.forEach(::write)
592 repeat(reader.readLine().toInt()) {
593 try { database.command(reader.readLine()).forEach(::write) }
594 catch (error: Problem) { write("ERROR ${error.message}") }
595 }
596 writer.flush()
597}
