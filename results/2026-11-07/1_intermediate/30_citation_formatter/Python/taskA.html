99def depths(text):
100 result = []
101 depth = 0
102 for char in text:
103 result.append(depth)
104 if char == "{":
105 depth += 1
106 elif char == "}":
107 depth -= 1
108 return result

111def split_top(text, separator):
112 levels = depths(text)
113 parts = []
114 start = 0
115 index = 0
116 while index < len(text):
117 if levels[index] == 0 and text.startswith(separator, index):
118 parts.append(text[start:index])
119 index += len(separator)
120 start = index
121 else:
122 index += 1
123 parts.append(text[start:])
124 return parts

135def evaluate(expression, macros):
136 depth = 0
137 for char in expression:
138 if char == "{":
139 depth += 1
140 elif char == "}":
141 depth -= 1
142 if depth < 0:
143 raise LoadError("unbalanced braces")
144 if depth != 0:
145 raise LoadError("unbalanced braces")
146 parts = []
147 for term in expression.split(" # "):
148 if term.startswith(("{", '"')):
149 parts.append(term[1:-1])
150 elif term.isdigit():
151 parts.append(term)
152 elif term in macros:
153 parts.append(macros[term])
154 else:
155 raise LoadError(f"unknown macro '{term}'")
156 return "".join(parts)

159def accent_at(text, index):
160 if text.startswith("\\&", index):
161 return "&", 2
162 command = text[index:index + 5]
163 if len(command) < 3 or command[0] != "\\":
164 return None
165 if command[1] in ACCENT_SYMBOLS and command[2].isalpha():
166 return command[2], 3
167 if (len(command) == 5 and command[1] in ACCENT_COMMANDS and command[2] == "{"
168 and command[3].isalpha() and command[4] == "}"):
169 return command[3], 5
170 return None

173def remove_accents(text):
174 result = []
175 index = 0
176 while index < len(text):
177 if text[index] == "{":
178 accent = accent_at(text, index + 1)
179 if accent is not None and text.startswith("}", index + 1 + accent[1]):
180 result.append(accent[0])
181 index += accent[1] + 2
182 continue
183 accent = accent_at(text, index)
184 if accent is None:
185 result.append(text[index])
186 index += 1
187 else:
188 result.append(accent[0])
189 index += accent[1]
190 return "".join(result)

262def parse_name(text):
263 parts = [part.strip() for part in split_top(text, ",")]
264 words = split_top(parts[0], " ")
265 lowercase = lowercase_positions(words)
266 if len(parts) == 1:
267 if not lowercase:
268 return Name(" ".join(words[:-1]), "", words[-1], "")
269 start, end = lowercase[0], lowercase[-1] + 1
270 return Name(" ".join(words[:start]), " ".join(words[start:end]), " ".join(words[end:]), "")
271 end = lowercase[-1] + 1 if lowercase else 0
272 von = " ".join(words[:end])
273 last = " ".join(words[end:])
274 if len(parts) == 2:
275 return Name(parts[1], von, last, "")
276 return Name(parts[2], von, last, parts[1])

290def sentence_case(title):
291 chars = list(title)
292 for index, depth in enumerate(depths(title)):
293 if depth == 0 and chars[index].isalpha():
294 starts = index == 0 or (index >= 2 and title[index - 2:index] == ": ")
295 chars[index] = chars[index].upper() if starts else chars[index].lower()
296 return "".join(chars)

301 char.lower() if depth == 0 else char

306def capitalize_first(word):
307 levels = depths(word)
308 for index, char in enumerate(word):
309 if char.isalpha():
310 if levels[index] == 0:
311 return word[:index] + char.upper() + word[index + 1:]
312 return word

316def title_case(title):
317 words = split_top(title, " ")
318 converted = []
319 for index, word in enumerate(words):
320 small = letters_of(word).lower() in SMALL_WORDS
321 inner = 0 < index < len(words) - 1
322 if word.startswith("{"):
323 converted.append(word)
324 elif small and inner and not words[index - 1].endswith(":"):
325 converted.append(lowercase_top(word))
326 else:
327 converted.append(capitalize_first(word))
328 return " ".join(converted)

376def parse_elements(template, index):
377 elements = []
378 while index < len(template) and template[index] != ">":
379 char = template[index]
380 if char == "{":
381 end = template.index("}", index)
382 elements.append(Element("field", template[index + 1:end]))
383 index = end + 1
384 elif char == "<":
385 children, index = parse_elements(template, index + 1)
386 elements.append(Element("optional", children=children))
387 index += 1
388 else:
389 elements.append(Element("text", char))
390 index += 1
391 return elements, index

394def parse_template(template):
395 elements, _ = parse_elements(template, 0)
396 return elements

399def render(elements, values, output=""):
400 for element in elements:
401 if element.kind == "field":
402 output += values.get(element.text, "")
403 elif element.kind == "optional":
404 fields = [child.text for child in element.children if child.kind == "field"]
405 if all(values.get(name, "") for name in fields):
406 output = render(element.children, values, output)
407 elif element.text != "." or not output.endswith((".", "?", "!")):
408 output += element.text
409 return output

424def author_names(name_list, style):
425 names = name_list.names
426 if style == "APA":
427 forms = [name.inverted(False) for name in names]
428 if name_list.et_al:
429 return ", ".join(forms) + ", et al."
430 if len(forms) <= 20:
431 return list_a(forms, "&")
432 return ", ".join(forms[:19]) + ", . . . " + forms[-1]
433 if style == "IEEE":
434 forms = [name.direct(False) for name in names]
435 if name_list.et_al or len(forms) >= 7:
436 return forms[0] + " et al."
437 return list_b(forms, "and")
438 forms = [names[0].inverted(True)] + [name.direct(True) for name in names[1:]]
439 if style == "CHICAGO":
440 if name_list.et_al or len(forms) >= 11:
441 return ", ".join(forms[:7]) + ", et al."
442 return list_a(forms, "and")
443 if name_list.et_al or len(forms) >= 3:
444 return forms[0] + ", et al."
445 return list_a(forms, "and")

457 if not edition.isdigit():
458 return edition + " ed."
459 number = int(edition)
460 if number % 100 in (11, 12, 13):
461 suffix = "th"
462 elif number % 10 == 1:
463 suffix = "st"
464 elif number % 10 == 2:
465 suffix = "nd"
466 elif number % 10 == 3:
467 suffix = "rd"

469 suffix = "th"

486def number_ranges(numbers):
487 parts = []
488 start = 0
489 while start < len(numbers):
490 end = start
491 while end + 1 < len(numbers) and numbers[end + 1] == numbers[end] + 1:
492 end += 1
493 if end - start >= 2:
494 parts.append(f"[{numbers[start]}]-[{numbers[end]}]")
495 else:
496 parts.extend(f"[{number}]" for number in numbers[start:end + 1])
497 start = end + 1
498 return ", ".join(parts)

533def entry_warnings(entry, titles):
534 messages = []
535 if "pages" in entry.fields:
536 pages = entry.plain_field("pages")
537 if "-" in pages and "--" not in pages:
538 messages.append("pages use a single hyphen")
539 numbers = [part for part in pages.split("-") if part]
540 if len(numbers) == 2 and all(part.isdigit() for part in numbers):
541 if int(numbers[0]) > int(numbers[1]):
542 messages.append("pages are reversed")
543 year = entry.plain_field("year")
544 if len(year) != 4 or not year.isdigit():
545 messages.append("year is not 4 digits")
546 title = entry.plain_field("title")
547 lowercase = sum(char.islower() for char in title)
548 uppercase = sum(char.isupper() for char in title)
549 if lowercase == 0 and uppercase >= 2:
550 messages.append("title is all caps")
551 if entry.kind == "article" and "doi" not in entry.fields:
552 messages.append("article has no doi")
553 lowered = title.lower()
554 if lowered in titles:
555 messages.append(f"same title as {titles[lowered]}")
556 else:
557 titles[lowered] = entry.key
558 for name in sorted(entry.fields):
559 if name not in KNOWN_FIELDS:
560 messages.append(f"unknown field '{name}'")
561 return messages

564class Database:
565 def __init__(self):
566 self.macros = dict(MONTH_MACROS)
567 self.macro_count = 0
568 self.entries = []
569 self.by_key = {}
570 self.messages = []
571
572 def report(self, index, name, reason):
573 self.messages.append(f"ERROR line {index + 1}: {name}: {reason}")
574
575 def load(self, lines):
576 index = 0
577 while lines[index] != "%%":
578 line = lines[index]
579 if line.startswith("@string{"):
580 self.define_macro(line, index)
581 elif line.startswith("@"):
582 end = index + 1
583 while lines[end] != "}":
584 end += 1
585 self.read_entry(lines, index, end)
586 index = end
587 index += 1
588 self.assign_suffixes()
589 return index
590
591 def define_macro(self, line, index):
592 name, expression = line[len("@string{"):-1].split(" = ", 1)
593 try:
594 value = evaluate(expression, self.macros)
595 except LoadError as error:
596 self.report(index, name, error)
597 return
598 if name in self.macros:
599 self.report(index, name, "duplicate macro")
600 return
601 self.macros[name] = value
602 self.macro_count += 1
603
604 def read_entry(self, lines, start, end):
605 header = lines[start]
606 brace = header.index("{")
607 kind = header[1:brace]
608 key = header[brace + 1:-1]
609 if kind not in REQUIRED_FIELDS:
610 self.report(start, key, f"unknown type '{kind}'")
611 return
612 if key in self.by_key:
613 self.report(start, key, "duplicate key")
614 return
615 fields = {}
616 for index in range(start + 1, end):
617 name, expression = lines[index][2:].split(" = ", 1)
618 if expression.endswith(","):
619 expression = expression[:-1]
620 try:
621 value = normalize(evaluate(expression, self.macros))
622 except LoadError as error:
623 self.report(index, key, error)
624 return
625 if name in fields:
626 self.report(index, key, f"duplicate field '{name}'")
627 return
628 fields[name] = value
629 for name in REQUIRED_FIELDS[kind]:
630 substituted = name == "author" and kind == "book" and "editor" in fields
631 if name not in fields and not substituted:
632 self.report(end, key, f"missing field '{name}'")
633 return
634 entry = Entry(key, kind, fields)
635 self.entries.append(entry)
636 self.by_key[key] = entry
637
638 def assign_suffixes(self):
639 for entry in self.entries:
640 group = [
641 other for other in self.entries
642 if other.primary == entry.primary and other.fields["year"] == entry.fields["year"]
643 ]
644 if len(group) > 1:
645 group.sort(key=lambda other: (other.sort_title(), other.key))
646 entry.suffix = chr(ord("a") + group.index(entry))
647
648 def summary(self):
649 return (
650 f"LOADED entries={len(self.entries)} macros={self.macro_count} "
651 f"errors={len(self.messages)}"
652 )

690 values["pp"] = ""

692 pages = values["pages"].replace("--", "-")
693 values["pages"] = pages
694 values["pp"] = ("pp. " if "-" in pages else "p. ") + pages
695 if "edition" in entry.fields:
696 values["edition"] = edition_text(values["edition"])
697 month = values.get("month", "")
698 values["mon"] = MONTH_ABBREVIATIONS.get(month, month)
699 return values

701 def reference(self, entry, style, repeated=False):
702 kind = "inproceedings" if entry.kind == "incollection" else entry.kind
703 template = parse_template(TEMPLATES[style][kind])
704 text = render(template, self.template_values(entry, style, repeated))
705 if style == "IEEE":
706 return f"[{self.ieee_numbers()[entry.key]}] {text}"
707 return text

717 def bibliography(self, arguments):
718 style = arguments[0]
719 self.emit(HEADINGS[style])
720 if style == "IEEE":
721 entries = [self.database.by_key[key] for key in self.ieee_order()]
722 else:
723 entries = sorted(self.database.entries, key=Entry.order_key)
724 previous = None
725 for entry in entries:
726 repeated = (
727 style in ("CHICAGO", "MLA") and previous is not None
728 and entry.primary == previous.primary
729 )
730 self.emit(self.reference(entry, style, repeated))

733 def author_year(self, entries, style, parenthetical):
734 groups = []
735 for entry in entries:
736 name = short_name(entry, style, style == "APA" and parenthetical)
737 label = entry.year_label(style)
738 if groups and groups[-1][0] == name:
739 groups[-1][1].append(label)
740 else:
741 groups.append((name, [label]))
742 if not parenthetical:
743 return "; ".join(f"{name} ({', '.join(labels)})" for name, labels in groups)
744 separator = ", " if style == "APA" else " "
745 return "(" + "; ".join(name + separator + ", ".join(labels) for name, labels in groups) + ")"

747 def citation_text(self, command, entries, style):
748 if command == "citeauthor":
749 return "; ".join(short_name(entry, style, False) for entry in entries)
750 if command == "citeyear":
751 return ", ".join(entry.year_label(style) for entry in entries)
752 parenthetical = command in ("cite", "citep")
753 if style == "IEEE":
754 numbers = self.ieee_numbers()
755 if parenthetical:
756 return number_ranges(sorted(numbers[entry.key] for entry in entries))
757 return "; ".join(
758 f"{short_name(entry, style, False)} [{numbers[entry.key]}]" for entry in entries
759 )
760 if style == "MLA":
761 names = "; ".join(short_name(entry, style, False) for entry in entries)
762 return f"({names})" if parenthetical else names
763 if parenthetical:
764 entries = sorted(entries, key=Entry.order_key)
765 return self.author_year(entries, style, parenthetical)

767 def cite(self, arguments):
768 style, text = arguments
769 citations = find_citations(text)
770 cited = [[self.entry(key) for key in citation.keys] for citation in citations]
771 for entries in cited:
772 for entry in entries:
773 if entry.key not in self.record:
774 self.record.append(entry.key)
775 pieces = []
776 position = 0
777 for citation, entries in zip(citations, cited):
778 pieces.append(text[position:citation.start])
779 pieces.append(self.citation_text(citation.command, entries, style))
780 position = citation.end
781 pieces.append(text[position:])
782 self.emit("".join(pieces))

784 def keygen(self, arguments):
785 bases = [base_key(entry) for entry in self.database.entries]
786 counts = Counter(bases)
787 used = Counter()
788 for entry, base in zip(self.database.entries, bases):
789 generated = base
790 if counts[base] > 1:
791 generated = base + chr(ord("a") + used[base])
792 used[base] += 1
793 self.emit(f"{entry.key} -> {generated}")

811 ]

817 def export(self, arguments):
818 entry = self.entry(arguments[0])
819 known = [name for name in KNOWN_FIELDS if name in entry.fields]
820 unknown = sorted(name for name in entry.fields if name not in KNOWN_FIELDS)
821 names = known + unknown
822 self.emit(f"@{entry.kind}{{{entry.key},")
823 for index, name in enumerate(names):
824 ending = "," if index < len(names) - 1 else ""
825 self.emit(f" {name} = {{{entry.fields[name]}}}{ending}")
826 self.emit("}")

828 def stats(self, arguments):
829 entries = self.database.entries
830 self.emit(f"ENTRIES {len(entries)}")
831 types = Counter(entry.kind for entry in entries)
832 for kind in sorted(types):
833 self.emit(f"TYPE {kind} {types[kind]}")
834 years = Counter(entry.plain_field("year") for entry in entries)
835 for year in sorted(years):
836 self.emit(f"YEAR {year} {years[year]}")
837 authors = set()
838 pairs = Counter()
839 for entry in entries:
840 if entry.authors is None:
841 continue
842 forms = sorted({name.direct(True) for name in entry.authors.names})
843 authors.update(forms)
844 for pair in combinations(forms, 2):
845 pairs[pair] += 1
846 self.emit(f"AUTHORS {len(authors)}")
847 if not pairs:
848 self.emit("PAIR none")
849 return
850 (first, second), count = min(pairs.items(), key=lambda item: (-item[1], item[0]))
851 self.emit(f"PAIR {first} & {second} {count}")

853 def execute(self, line):
854 if line.split(" ", 1)[0] in ("CITE", "SEARCH"):
855 command, *arguments = line.split(" ", 2)
856 else:
857 command, *arguments = line.split(" ")
858 handlers = {
859 "FORMAT": self.format,
860 "BIBLIOGRAPHY": self.bibliography,
861 "CITE": self.cite,
862 "KEYGEN": self.keygen,
863 "CHECK": self.check,
864 "SEARCH": self.search,
865 "EXPORT": self.export,
866 "STATS": self.stats,
867 }
868 try:
869 if command not in handlers:
870 raise CommandError(f"unknown command '{command}'")
871 expected = ARGUMENT_COUNTS[command]
872 if command == "FORMAT":
873 valid = len(arguments) >= expected
874 else:
875 valid = len(arguments) == expected
876 if not valid:
877 raise CommandError("wrong number of arguments")
878 if command in STYLED_COMMANDS and arguments[0] not in STYLES:
879 raise CommandError(f"unknown style '{arguments[0]}'")
880 handlers[command](arguments)
881 except CommandError as error:
882 self.emit(f"ERROR {error}")
