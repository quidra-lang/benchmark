28 def lower?(c) = ('a'..'z').cover?(c)
29 def upper?(c) = ('A'..'Z').cover?(c)
30 def letter?(c) = lower?(c) || upper?(c)
31 def digit?(c) = ('0'..'9').cover?(c)
32 def lower(text) = text.tr('A-Z', 'a-z')
33 def upper(text) = text.tr('a-z', 'A-Z')
34 def plain(text) = text.delete('{}')
35 def letters(text) = lower(text.delete('^A-Za-z'))
36 def digits?(text) = !text.empty? && text.each_char.all? { |c| digit?(c) }

38 def split_depth(text, separator)
39 result = []
40 start = i = depth = 0
41 while i < text.length
42 if depth.zero? && text[i, separator.length] == separator
43 result << text[start...i]
44 i += separator.length
45 start = i
46 else
47 depth += 1 if text[i] == '{'
48 depth -= 1 if text[i] == '}'
49 i += 1
50 end
51 end
52 result << text[start..]
53 end
54
55 def evaluate(expression, macros, line)
56 depth = 0
57 expression.each_char do |c|
58 depth += 1 if c == '{'
59 depth -= 1 if c == '}'
60 raise LoadProblem.new(line, 'unbalanced braces') if depth.negative?
61 end
62 raise LoadProblem.new(line, 'unbalanced braces') unless depth.zero?
63 expression.split(' # ', -1).map do |term|
64 if ['{', '"'].include?(term[0])
65 term[1...-1]
66 elsif digit?(term[0])
67 term
68 else
69 macros.fetch(term) { raise LoadProblem.new(line, "unknown macro '#{term}'") }
70 end
71 end.join
72 end

74 def accent(text, at)
75 return nil if at + 1 >= text.length || text[at] != '\\'
76 command = text[at + 1]
77 return ['&', 2] if command == '&'
78 symbol = "'\"`^~=.".include?(command)
79 return [text[at + 2], 3] if symbol && at + 2 < text.length && letter?(text[at + 2])
80 if (symbol || 'cuvH'.include?(command)) && at + 4 < text.length && text[at + 2] == '{' && letter?(text[at + 3]) && text[at + 4] == '}'
81 return [text[at + 3], 5]
82 end
83 nil
84 end
85
86 def normalize(text)
87 result = +''
88 i = 0
89 while i < text.length
90 wrapped = text[i] == '{' ? accent(text, i + 1) : nil
91 if wrapped && text[i + wrapped[1] + 1] == '}'
92 result << wrapped[0]
93 i += wrapped[1] + 2
94 else
95 current = accent(text, i)
96 result << (current ? current[0] : text[i])
97 i += current ? current[1] : 1
98 end
99 end
100 result.tr('~', ' ').split(' ').join(' ')
101 end

103 def title_case(title, style)
104 return title if style == 'IEEE'
105 if style == 'APA'
106 result = title.dup
107 depth = 0
108 title.each_char.with_index do |c, i|
109 if c == '{'
110 depth += 1
111 elsif c == '}'
112 depth -= 1
113 elsif depth.zero? && letter?(c)
114 result[i] = i.zero? || i >= 2 && title[i - 2, 2] == ': ' ? upper(c) : lower(c)
115 end
116 end
117 return result
118 end
119 words = split_depth(title, ' ')
120 words.each_index do |i|
121 word = words[i]
122 next if word.start_with?('{')
123 make_lower = i.positive? && i + 1 < words.length && !words[i - 1].end_with?(':') && SMALL_WORDS.include?(letters(word))
124 result = word.dup
125 depth = 0
126 first_letter = true
127 word.each_char.with_index do |c, j|
128 if c == '{'
129 depth += 1
130 elsif c == '}'
131 depth -= 1
132 elsif letter?(c)
133 result[j] = lower(c) if depth.zero? && make_lower
134 result[j] = upper(c) if depth.zero? && !make_lower && first_letter
135 first_letter = false
136 end
137 end
138 words[i] = result
139 end
140 words.join(' ')
141 end

145 return "#{names.first} #{conjunction} #{names.last}" if list_b && names.length == 2
146 "#{names[0...-1].join(', ')}, #{conjunction} #{names.last}"

150Name = Data.define(:first, :von, :last, :jr) do
151 def self.parse(text)
152 parts = Text.split_depth(text, ',').map(&:strip)
153 words = Text.split_depth(parts.first, ' ')
154 lower_words = (0...(words.length - 1)).select { |i| Text.lower?(words[i][0]) }
155 first_lower = lower_words.first
156 last_lower = lower_words.last
157 if parts.length == 1
158 return new(words[0...-1].join(' '), '', words.last, '') unless first_lower
159 return new(words[0...first_lower].join(' '), words[first_lower..last_lower].join(' '), words[last_lower + 1..].join(' '), '')
160 end
161 von = last_lower ? words[0..last_lower].join(' ') : ''
162 last = words[(last_lower ? last_lower + 1 : 0)..].join(' ')
163 new(parts.last, von, last, parts.length == 3 ? parts[1] : '')
164 end

166 def surname = Text.plain(von.empty? ? last : "#{von} #{last}")
167 def initials
168 Text.plain(first).split(' ').map do |word|
169 word.split('-', -1).map { |piece| "#{Text.upper(piece.each_char.find { |c| Text.letter?(c) })}." }.join('-')
170 end.join(' ')
171 end

177 result += ", #{Text.plain(jr)}" unless jr.empty?
178 result

214 def initialize(key, kind, fields)

275 def optional_end(pattern, start)
276 depth = 1
277 i = start
278 until depth.zero?
279 i += 1
280 depth += 1 if pattern[i] == '<'
281 depth -= 1 if pattern[i] == '>'
282 end
283 i
284 end
285
286 def optional_present?(pattern, start, finish, values)
287 i = start
288 while i < finish
289 if pattern[i] == '<'
290 i = optional_end(pattern, i)
291 elsif pattern[i] == '{'
292 close = pattern.index('}', i)
293 return false if values.fetch(pattern[i + 1...close], '').empty?
294 i = close
295 end
296 i += 1
297 end
298 true
299 end
300
301 def render(pattern, start, finish, values, result)
302 i = start
303 while i < finish
304 c = pattern[i]
305 if c == '<'
306 close = optional_end(pattern, i)
307 render(pattern, i + 1, close, values, result) if optional_present?(pattern, i + 1, close, values)
308 i = close
309 elsif c == '{'
310 close = pattern.index('}', i)
311 result << values.fetch(pattern[i + 1...close], '')
312 i = close
313 elsif c != '.' || result.empty? || !'.?!'.include?(result[-1])
314 result << c
315 end
316 i += 1
317 end
318 end

320 def format(entry, style, number, repeated = false)
321 values = FIELDS.to_h { |field| [field, Text.plain(entry.fields.fetch(field, ''))] }
322 author = repeated ? '---' : entry.primary.format(style)
323 author += ANNOTATIONS[STYLES.index(style)][entry.primary.single? ? 0 : 1] unless entry.authors
324 values['author'] = author
325 if entry.editors
326 values['editor'] = entry.editors.format(style, true)
327 values['eds'] = entry.editors.single? ? 'Ed.' : 'Eds.'
328 end
329 values['title'] = Text.plain(Text.title_case(entry.fields.fetch('title'), style))
330 values['year'] = entry.year_label(style)
331 if entry.fields.key?('pages')
332 pages = Text.plain(entry.fields['pages']).gsub('--', '-')
333 values['pages'] = pages
334 values['pp'] = (pages.include?('-') ? 'pp. ' : 'p. ') + pages
335 end
336 if entry.fields.key?('edition')
337 edition = Text.plain(entry.fields['edition'])
338 if Text.digits?(edition)
339 n = edition.to_i
340 suffix = (11..13).cover?(n % 100) ? 'th' : { 1 => 'st', 2 => 'nd', 3 => 'rd' }.fetch(n % 10, 'th')
341 edition = "#{n}#{suffix}"
342 end
343 values['edition'] = "#{edition} ed."
344 end
345 if entry.fields.key?('month')
346 month = Text.plain(entry.fields['month'])
347 index = MONTHS.index(month)
348 values['mon'] = index ? MONTH_SHORT[index] : month
349 end
350 kind = entry.kind == 3 ? 2 : entry.kind > 3 ? entry.kind - 1 : entry.kind
351 pattern = TEMPLATES[STYLES.index(style)][kind]
352 result = style == 'IEEE' ? "[#{number}] " : +''
353 render(pattern, 0, pattern.length, values, result)
354 result
355 end

360class Database

363 def initialize(lines)
364 @entries, @record, @lookup, @output = [], [], {}, []
365 @macros = MONTH_MACROS.zip(MONTHS).to_h
366 @macro_count = 0
367 i = 0
368 while i < lines.length
369 line = lines[i]
370 if line.empty? || line.start_with?('%')
371 i += 1
372 next
373 end
374 if line.start_with?('@string{')
375 equals = line.index(' = ')
376 name = line[8...equals]
377 begin
378 value = Text.evaluate(line[equals + 3...-1], @macros, i + 1)
379 raise LoadProblem.new(i + 1, 'duplicate macro') if @macros.key?(name)
380 @macros[name] = value.freeze
381 @macro_count += 1
382 rescue LoadProblem => error
383 report(name, error)
384 end
385 else
386 finish = i + 1
387 finish += 1 until lines[finish] == '}'
388 brace = line.index('{')
389 kind, key = line[1...brace], line[brace + 1...-1]
390 begin
391 read_entry(lines, i, finish, kind, key)
392 rescue LoadProblem => error
393 report(key, error)
394 end
395 i = finish
396 end
397 i += 1
398 end
399 output << "LOADED entries=#{entries.length} macros=#{@macro_count} errors=#{output.length}"
400 entries.group_by { |entry| [entry.primary, entry.fields['year']] }.each_value do |group|
401 next if group.length == 1
402 group.sort_by { |entry| [entry.sort_title, entry.key] }.each_with_index { |entry, index| entry.suffix = (97 + index).chr }
403 end
404 end

406 def report(name, error) = output << "ERROR line #{error.line}: #{name}: #{error.message}"

408 def read_entry(lines, start, finish, kind_name, key)
409 kind = TYPES.index(kind_name)
410 raise LoadProblem.new(start + 1, "unknown type '#{kind_name}'") unless kind
411 raise LoadProblem.new(start + 1, 'duplicate key') if lookup.key?(key)
412 fields = {}
413 (start + 1...finish).each do |i|
414 equals = lines[i].index(' = ')
415 name = lines[i][2...equals]
416 expression = lines[i][equals + 3..].delete_suffix(',')
417 value = Text.evaluate(expression, @macros, i + 1)
418 raise LoadProblem.new(i + 1, "duplicate field '#{name}'") if fields.key?(name)
419 fields[name] = Text.normalize(value).freeze
420 end
421 REQUIRED[kind].each do |field|
422 unless fields.key?(field) || kind == 1 && field == 'author' && fields.key?('editor')
423 raise LoadProblem.new(finish + 1, "missing field '#{field}'")
424 end
425 end
426 entry = Entry.new(key, kind, fields)
427 entries << entry
428 lookup[key] = entry
429 end

437 def citation_text(citation, style, numbers)
438 parenthetical = %w[cite citep].include?(citation.kind)
439 if %w[citeauthor citeyear].include?(citation.kind) || style == 'MLA' || style == 'IEEE' && !parenthetical
440 text = citation.entries.map do |entry|
441 if citation.kind == 'citeyear'
442 entry.year_label(style)
443 else
444 entry.short_name(style, false) + (style == 'IEEE' && citation.kind == 'citet' ? " [#{numbers.fetch(entry)}]" : '')
445 end
446 end.join(citation.kind == 'citeyear' ? ', ' : '; ')
447 return style == 'MLA' && parenthetical ? "(#{text})" : text
448 end
449 parts = []
450 if style == 'IEEE'
451 sorted = citation.entries.map { |entry| numbers.fetch(entry) }.sort
452 i = 0
453 while i < sorted.length
454 finish = i + 1
455 finish += 1 while finish < sorted.length && sorted[finish] == sorted[finish - 1] + 1
456 if finish - i >= 3
457 parts << "[#{sorted[i]}]-[#{sorted[finish - 1]}]"
458 else
459 (i...finish).each { |j| parts << "[#{sorted[j]}]" }
460 end
461 i = finish
462 end
463 return parts.join(', ')
464 end
465 ordered = parenthetical ? citation.entries.sort_by(&:sort_key) : citation.entries
466 i = 0
467 while i < ordered.length
468 name = ordered[i].short_name(style, parenthetical)
469 finish = i + 1
470 finish += 1 while finish < ordered.length && ordered[finish].short_name(style, parenthetical) == name
471 years = ordered[i...finish].map { |entry| entry.year_label(style) }.join(', ')
472 parts << (parenthetical ? name + (style == 'APA' ? ', ' : ' ') + years : "#{name} (#{years})")
473 i = finish
474 end
475 text = parts.join('; ')
476 parenthetical ? "(#{text})" : text
477 end

479 def cite(style, text)
480 citations = []
481 start = text.index('\\')
482 while start
483 brace = text.index('{', start)
484 finish = text.index('}', brace)
485 seen = Set.new
486 text[brace + 1...finish].split(',').each do |key|
487 seen << lookup.fetch(key) { raise Problem, "unknown key '#{key}'" }
488 end
489 citations << Citation.new(start, finish + 1, text[start + 1...brace], seen.to_a.freeze)
490 start = text.index('\\', finish + 1)
491 end
492 recorded = record.to_set
493 citations.each { |citation| citation.entries.each { |entry| record << entry if recorded.add?(entry) } }
494 numbers = numbering[1]
495 result = +''
496 position = 0
497 citations.each do |citation|
498 result << text[position...citation.start] << citation_text(citation, style, numbers)
499 position = citation.finish
500 end
501 result << text[position..]
502 end

518 def check
519 result = []
520 titles = {}
521 entries.each do |entry|
522 warnings = []
523 if entry.fields.key?('pages')
524 pages = Text.plain(entry.fields['pages'])
525 warnings << 'pages use a single hyphen' if pages.include?('-') && !pages.include?('--')
526 parts = pages.split('-', -1).reject(&:empty?)
527 warnings << 'pages are reversed' if parts.length == 2 && parts.all? { |part| Text.digits?(part) } && parts[0].to_i > parts[1].to_i
528 end
529 warnings << 'year is not 4 digits' unless entry.year.length == 4 && Text.digits?(entry.year)
530 warnings << 'title is all caps' if entry.plain_title.each_char.none? { |c| Text.lower?(c) } && entry.plain_title.each_char.count { |c| Text.upper?(c) } >= 2
531 warnings << 'article has no doi' if entry.kind.zero? && !entry.fields.key?('doi')
532 if titles.key?(entry.lower_title)
533 warnings << "same title as #{titles[entry.lower_title]}"

535 titles[entry.lower_title] = entry.key
536 end
537 entry.unknown_fields.each { |field| warnings << "unknown field '#{field}'" }
538 warnings.each { |warning| result << "WARN #{entry.key}: #{warning}" }
539 end
540 result << "CHECK warnings=#{result.length}"

572 first = line.index(' ')
573 command = first ? line[0...first] : line
574 if %w[CITE SEARCH].include?(command) && first
575 second = line.index(' ', first + 1)
576 words = second ? [command, line[first + 1...second], line[second + 1..]] : [command, line[first + 1..]]
577 else
578 words = line.split(' ')
579 end
580 arity = { 'FORMAT' => -1, 'BIBLIOGRAPHY' => 1, 'CITE' => 2, 'KEYGEN' => 0, 'CHECK' => 0, 'SEARCH' => 2, 'EXPORT' => 1, 'STATS' => 0 }
581 .fetch(command) { raise Problem, "unknown command '#{command}'" }
582 raise Problem, 'wrong number of arguments' if (arity == -1 ? words.length < 3 : words.length != arity + 1)
583 style = 'APA'
584 if %w[FORMAT BIBLIOGRAPHY CITE].include?(command)
585 style = words[1]
586 raise Problem, "unknown style '#{style}'" unless STYLES.include?(style)
587 end
588 case command
589 when 'FORMAT'
590 numbers = numbering[1]
591 words.drop(2).map { |key| lookup.key?(key) ? References.format(lookup[key], style, numbers.fetch(lookup[key])) : "ERROR unknown key '#{key}'" }
592 when 'BIBLIOGRAPHY'
593 order, numbers = numbering
594 order = order.sort_by(&:sort_key) unless style == 'IEEE'
595 rows = order.each_with_index.map do |entry, index|
596 repeated = index.positive? && %w[CHICAGO MLA].include?(style) && entry.primary == order[index - 1].primary
597 References.format(entry, style, numbers.fetch(entry), repeated)
598 end
599 [style == 'MLA' ? 'Works Cited' : 'References'] + rows
600 when 'CITE' then [cite(style, words[2])]
601 when 'KEYGEN' then keygen
602 when 'CHECK' then check
603 when 'EXPORT' then export(words[1])
604 when 'STATS' then stats
605 when 'SEARCH'
606 needle = Text.lower(words[2])
607 matches = entries.select { |entry| entry.fields.key?(words[1]) && Text.lower(Text.plain(entry.fields[words[1]])).include?(needle) }.map(&:key)
608 [matches.empty? ? 'MATCHES 0' : "MATCHES #{matches.length}: #{matches.join(' ')}"]
609 end
610 end
611end
