19fn plain(value: &str) -> String { value.chars().filter(|&c| c != '{' && c != '}').collect() }
20fn letters(value: &str) -> String { value.chars().filter(char::is_ascii_alphabetic).map(|c| c.to_ascii_lowercase()).collect() }
21fn digits(value: &str) -> bool { !value.is_empty() && value.bytes().all(|c| c.is_ascii_digit()) }
22fn split_depth<'a>(text: &'a str, delimiter: &str) -> Vec<&'a str> {
23 let mut result = Vec::new();
24 let (mut start, mut i, mut depth) = (0, 0, 0);
25 while i < text.len() {
26 if depth == 0 && text[i..].starts_with(delimiter) {
27 result.push(&text[start..i]); i += delimiter.len(); start = i;
28 } else {
29 match text.as_bytes()[i] { b'{' => depth += 1, b'}' => depth -= 1, _ => {} }
30 i += 1;
31 }
32 }
33 result.push(&text[start..]); result
34}
35fn evaluate(expression: &str, macros: &BTreeMap<String, String>) -> Result<String, String> {
36 let mut depth = 0;
37 for c in expression.bytes() {
38 match c { b'{' => depth += 1, b'}' => depth -= 1, _ => {} }
39 if depth < 0 { return Err("unbalanced braces".into()); }
40 }
41 if depth != 0 { return Err("unbalanced braces".into()); }
42 let mut result = String::new();
43 for term in expression.split(" # ") {
44 match term.as_bytes()[0] {
45 b'{' | b'"' => result.push_str(&term[1..term.len() - 1]),
46 b'0'..=b'9' => result.push_str(term),
47 _ => result.push_str(macros.get(term).ok_or_else(|| format!("unknown macro '{term}'"))?),
48 }
49 }
50 Ok(result)
51}
52fn accent(text: &[u8], at: usize) -> Option<(u8, usize)> {
53 if text.get(at) != Some(&b'\\') { return None; }
54 let &command = text.get(at + 1)?;
55 if command == b'&' { return Some((b'&', 2)); }
56 let symbol = b"'\"`^~=.".contains(&command);
57 if symbol && text.get(at + 2).is_some_and(u8::is_ascii_alphabetic) { return Some((text[at + 2], 3)); }
58 if (symbol || b"cuvH".contains(&command)) && text.get(at + 2) == Some(&b'{')
59 && text.get(at + 3).is_some_and(u8::is_ascii_alphabetic) && text.get(at + 4) == Some(&b'}') {
60 return Some((text[at + 3], 5));
61 }
62 None
63}
64fn normalize(value: &str) -> String {
65 let bytes = value.as_bytes();
66 let mut accents = String::new();
67 let mut i = 0;
68 while i < bytes.len() {
69 let wrapped = if bytes[i] == b'{' { accent(bytes, i + 1).filter(|&(_, width)| bytes.get(i + width + 1) == Some(&b'}')) } else { None };
70 if let Some((c, width)) = wrapped { accents.push(c as char); i += width + 2; }
71 else if let Some((c, width)) = accent(bytes, i) { accents.push(c as char); i += width; }
72 else { accents.push(bytes[i] as char); i += 1; }
73 }
74 accents.replace('~', " ").split(' ').filter(|s| !s.is_empty()).collect::<Vec<_>>().join(" ")
75}

77#[derive(Clone, PartialEq, Eq)]
78struct Name { first: String, von: String, last: String, jr: String }
79impl Name {
80 fn parse(text: &str) -> Self {
81 let parts: Vec<_> = split_depth(text, ",").into_iter().map(|p| p.trim_matches(' ')).collect();
82 let words = split_depth(parts[0], " ");
83 let lowercase: Vec<_> = (0..words.len() - 1).filter(|&i| words[i].as_bytes()[0].is_ascii_lowercase()).collect();
84 let mut name = Self { first: String::new(), von: String::new(), last: String::new(), jr: String::new() };
85 if parts.len() == 1 {
86 if let (Some(&first), Some(&last)) = (lowercase.first(), lowercase.last()) {
87 name.first = words[..first].join(" "); name.von = words[first..=last].join(" "); name.last = words[last + 1..].join(" ");
88 } else { name.first = words[..words.len() - 1].join(" "); name.last = words[words.len() - 1].into(); }
89 } else {
90 if let Some(&last) = lowercase.last() { name.von = words[..=last].join(" "); name.last = words[last + 1..].join(" "); }
91 else { name.last = words.join(" "); }
92 name.first = parts[parts.len() - 1].into();
93 if parts.len() == 3 { name.jr = parts[1].into(); }
94 }
95 name
96 }
97 fn surname(&self) -> String { plain(&if self.von.is_empty() { self.last.clone() } else { format!("{} {}", self.von, self.last) }) }
98 fn initials(&self) -> String {
99 plain(&self.first).split(' ').filter(|word| !word.is_empty()).map(|word| {
100 word.split('-').map(|part| format!("{}.", part.chars().find(char::is_ascii_alphabetic).unwrap().to_ascii_uppercase())).collect::<Vec<_>>().join("-")
101 }).collect::<Vec<_>>().join(" ")
102 }
103 fn text(&self, inverted: bool, abbreviated: bool) -> String {
104 let given = if abbreviated { self.initials() } else { plain(&self.first) };
105 let mut result = self.surname();
106 if !self.first.is_empty() { result = if inverted { format!("{result}, {given}") } else { format!("{given} {result}") }; }
107 if !self.jr.is_empty() { result.push_str(&format!(", {}", plain(&self.jr))); }
108 result
109 }
110}

112struct NameList { names: Vec<Name>, et_al: bool }
113impl NameList {
114 fn parse(text: &str) -> Self {
115 let mut names = split_depth(text, " and ");
116 let et_al = names.last() == Some(&"others");
117 if et_al { names.pop(); }
118 Self { names: names.into_iter().map(Name::parse).collect(), et_al }
119 }
120 fn single(&self) -> bool { self.names.len() == 1 && !self.et_al }

122#[derive(Clone, Copy, PartialEq, Eq)]
123enum Style { Apa, Ieee, Chicago, Mla }
124impl Style {
125 fn parse(text: &str) -> Option<Self> { match text { "APA" => Some(Self::Apa), "IEEE" => Some(Self::Ieee), "CHICAGO" => Some(Self::Chicago), "MLA" => Some(Self::Mla), _ => None } }

127fn title_case(title: &str, style: Style) -> String {
128 if style == Style::Ieee { return title.into(); }
129 if style == Style::Apa {
130 let mut result = title.as_bytes().to_vec();
131 let mut depth = 0;
132 for i in 0..result.len() {
133 match result[i] {
134 b'{' => depth += 1, b'}' => depth -= 1,
135 c if depth == 0 && c.is_ascii_alphabetic() => result[i] = if i == 0 || (i >= 2 && &title[i - 2..i] == ": ") { c.to_ascii_uppercase() } else { c.to_ascii_lowercase() },
136 _ => {}
137 }
138 }
139 return String::from_utf8(result).unwrap();
140 }
141 let words = split_depth(title, " ");
142 words.iter().enumerate().map(|(i, word)| {
143 if word.starts_with('{') { return (*word).to_owned(); }
144 let lowercase = i > 0 && i + 1 < words.len() && !words[i - 1].ends_with(':') && SMALL_WORDS.contains(&letters(word).as_str());
145 let (mut depth, mut first_letter) = (0, true);
146 word.bytes().map(|c| {
147 let mut result = c;
148 match c {
149 b'{' => depth += 1, b'}' => depth -= 1,
150 _ if c.is_ascii_alphabetic() => {
151 if depth == 0 && lowercase { result = c.to_ascii_lowercase(); }
152 else if depth == 0 && first_letter { result = c.to_ascii_uppercase(); }
153 first_letter = false;
154 }, _ => {}
155 }
156 result as char
157 }).collect::<String>()
158 }).collect::<Vec<_>>().join(" ")
159}
160fn listed(names: &[String], conjunction: &str, list_b: bool) -> String {
161 if names.len() == 1 { return names[0].clone(); }
162 if names.len() == 2 && list_b { return format!("{} {conjunction} {}", names[0], names[1]); }
163 format!("{}, {conjunction} {}", names[..names.len() - 1].join(", "), names.last().unwrap())
164}
165fn render_names(list: &NameList, style: Style, editor: bool) -> String {
166 let names: Vec<_> = list.names.iter().enumerate().map(|(i, name)| {
167 name.text(!editor && (style == Style::Apa || (style != Style::Ieee && i == 0)), style == Style::Apa || style == Style::Ieee)
168 }).collect();
169 if editor { return if list.et_al { format!("{} et al.", names[0]) } else { listed(&names, if style == Style::Apa { "&" } else { "and" }, true) }; }
170 match style {
171 Style::Apa if list.et_al => format!("{}, et al.", names.join(", ")),
172 Style::Apa if names.len() >= 21 => format!("{}, . . . {}", names[..19].join(", "), names.last().unwrap()),
173 Style::Apa => listed(&names, "&", false),
174 Style::Ieee if list.et_al || names.len() >= 7 => format!("{} et al.", names[0]),
175 Style::Ieee => listed(&names, "and", true),
176 Style::Chicago if list.et_al || names.len() >= 11 => format!("{}, et al.", names[..names.len().min(7)].join(", ")),
177 Style::Chicago => listed(&names, "and", false),
178 Style::Mla if list.et_al || names.len() >= 3 => format!("{}, et al.", names[0]),
179 Style::Mla => listed(&names, "and", false),
180 }
181}

188impl Entry {
189 fn new(key: String, kind: usize, fields: BTreeMap<String, String>) -> Self {
190 let authors = fields.get("author").map(|s| NameList::parse(s));
191 let editors = fields.get("editor").map(|s| NameList::parse(s));
192 let lead = &authors.as_ref().or(editors.as_ref()).unwrap().names[0];
193 let sort_last = plain(&lead.last).to_ascii_lowercase();
194 let year = plain(&fields["year"]);
195 let plain_title = plain(&fields["title"]);

197 let sort_title = ["a ", "an ", "the "].iter().find_map(|prefix| lower_title.strip_prefix(*prefix)).unwrap_or(&lower_title).to_owned();
198 let title_word = plain_title.split(' ').map(letters).find(|s| !s.is_empty() && !SMALL_WORDS.contains(&s.as_str())).unwrap_or_default();
199 let year_digits: String = fields["year"].chars().filter(char::is_ascii_digit).collect();
200 let base_key = format!("{}{year_digits}{title_word}", letters(&plain(&lead.last)));

212 }
213 fn unknown_fields(&self) -> Vec<&str> { self.fields.keys().map(String::as_str).filter(|name| !FIELDS.contains(name)).collect() }

247fn placeholder_index(name: &str) -> usize {
248 FIELDS.iter().position(|&field| field == name).unwrap_or(match name { "eds" => 18, "pp" => 19, _ => 20 })
249}
250fn optional_end(pattern: &str, start: usize) -> usize {
251 let mut depth = 1;
252 let mut i = start + 1;
253 loop {
254 match pattern.as_bytes()[i] { b'<' => depth += 1, b'>' => depth -= 1, _ => {} }
255 if depth == 0 { return i; }
256 i += 1;
257 }
258}
259fn optional_present(pattern: &str, start: usize, end: usize, values: &[String; 21]) -> bool {
260 let mut i = start;
261 while i < end {
262 if pattern.as_bytes()[i] == b'<' { i = optional_end(pattern, i); }
263 else if pattern.as_bytes()[i] == b'{' {
264 let close = i + pattern[i..].find('}').unwrap();
265 if values[placeholder_index(&pattern[i + 1..close])].is_empty() { return false; }
266 i = close;
267 }
268 i += 1;
269 }
270 true
271}
272fn render_template(pattern: &str, start: usize, end: usize, values: &[String; 21], result: &mut String) {
273 let mut i = start;
274 while i < end {
275 match pattern.as_bytes()[i] {
276 b'<' => {
277 let close = optional_end(pattern, i);
278 if optional_present(pattern, i + 1, close, values) { render_template(pattern, i + 1, close, values, result); }
279 i = close;
280 },
281 b'{' => {
282 let close = i + pattern[i..].find('}').unwrap();
283 result.push_str(&values[placeholder_index(&pattern[i + 1..close])]); i = close;
284 },
285 b'.' if result.as_bytes().last().is_some_and(|c| b".?!".contains(c)) => {},
286 c => result.push(c as char),
287 }
288 i += 1;
289 }

291impl Entry {
292 fn reference(&self, style: Style, number: usize, repeated: bool) -> String {
293 let mut values: [String; 21] = std::array::from_fn(|_| String::new());
294 for (i, name) in FIELDS.iter().enumerate() { if let Some(value) = self.fields.get(*name) { values[i] = plain(value); } }
295 let names = self.primary();
296 values[0] = if repeated { "---".into() } else { render_names(names, style, false) };
297 if self.authors.is_none() {
298 let annotations = [[" (Ed.)", " (Eds.)"], [", Ed.", ", Eds."], [", ed.", ", eds."], [", editor", ", editors"]];
299 values[0].push_str(annotations[style as usize][usize::from(!names.single())]);
300 }
301 if let Some(editors) = &self.editors {
302 values[1] = render_names(editors, style, true);
303 values[18] = if editors.single() { "Ed." } else { "Eds." }.into();
304 }
305 values[2] = plain(&title_case(&self.fields["title"], style)); values[15] = self.year_label(style);
306 if let Some(pages) = self.fields.get("pages") {
307 values[8] = plain(pages).replace("--", "-");
308 values[19] = format!("{} {}", if values[8].contains('-') { "pp." } else { "p." }, values[8]);
309 }
310 if let Some(edition) = self.fields.get("edition") {
311 let value = plain(edition);
312 values[5] = if digits(&value) {
313 let n: u32 = value.parse().unwrap();
314 let suffix = if (11..=13).contains(&(n % 100)) { "th" } else { match n % 10 { 1 => "st", 2 => "nd", 3 => "rd", _ => "th" } };
315 format!("{n}{suffix} ed.")
316 } else { format!("{value} ed.") };
317 }
318 if let Some(month) = self.fields.get("month") {
319 let value = plain(month);
320 values[20] = MONTHS.iter().position(|&m| m == value).map(|i| MONTH_SHORT[i].to_owned()).unwrap_or(value);
321 }
322 let kind = if self.kind == 3 { 2 } else if self.kind > 3 { self.kind - 1 } else { self.kind };
323 let pattern = TEMPLATES[style as usize][kind];
324 let mut result = if style == Style::Ieee { format!("[{number}] ") } else { String::new() };
325 render_template(pattern, 0, pattern.len(), &values, &mut result); result
326 }

329struct Database { entries: Vec<Entry>, lookup: BTreeMap<String, usize>, record: Vec<usize>, output: Vec<String> }
330impl Database {
331 fn read(lines: &[&str]) -> Self {
332 let mut database = Self { entries: Vec::new(), lookup: BTreeMap::new(), record: Vec::new(), output: Vec::new() };
333 let mut macros: BTreeMap<String, String> = MONTH_MACROS.into_iter().zip(MONTHS).map(|(a, b)| (a.into(), b.into())).collect();
334 let (mut macro_count, mut errors, mut i) = (0, 0, 0);
335 while i < lines.len() {
336 let line = lines[i];
337 if line.is_empty() || line.starts_with('%') { i += 1; continue; }
338 if let Some(definition) = line.strip_prefix("@string{") {
339 let (name, expression) = definition[..definition.len() - 1].split_once(" = ").unwrap();
340 let result = evaluate(expression, &macros).and_then(|value| if macros.contains_key(name) { Err("duplicate macro".into()) } else { Ok(value) });
341 match result {
342 Ok(value) => { macros.insert(name.into(), value); macro_count += 1; },
343 Err(reason) => { database.output.push(format!("ERROR line {}: {name}: {reason}", i + 1)); errors += 1; },
344 }
345 i += 1; continue;
346 }
347 let (kind_name, key) = line[1..line.len() - 1].split_once('{').unwrap();
348 let kind = TYPES.iter().position(|&value| value == kind_name);
349 let mut error = if kind.is_none() { Some((i + 1, format!("unknown type '{kind_name}'"))) }
350 else if database.lookup.contains_key(key) { Some((i + 1, "duplicate key".into())) } else { None };
351 let mut fields = BTreeMap::new();
352 i += 1;
353 while lines[i] != "}" {
354 if error.is_none() {
355 let (name, expression) = lines[i][2..].split_once(" = ").unwrap();
356 let expression = expression.strip_suffix(',').unwrap_or(expression);
357 let result = evaluate(expression, &macros).and_then(|value| if fields.contains_key(name) { Err(format!("duplicate field '{name}'")) } else { Ok(value) });
358 match result {
359 Ok(value) => { fields.insert(name.to_owned(), normalize(&value)); },
360 Err(reason) => error = Some((i + 1, reason)),
361 }
362 }
363 i += 1;
364 }
365 if error.is_none() {
366 for &name in REQUIRED[kind.unwrap()] {
367 if !fields.contains_key(name) && !(kind == Some(1) && name == "author" && fields.contains_key("editor")) {
368 error = Some((i + 1, format!("missing field '{name}'"))); break;
369 }
370 }
371 }
372 if let Some((line_number, reason)) = error { database.output.push(format!("ERROR line {line_number}: {key}: {reason}")); errors += 1; }
373 else {
374 database.lookup.insert(key.into(), database.entries.len());
375 database.entries.push(Entry::new(key.into(), kind.unwrap(), fields));
376 }
377 i += 1;
378 }
379 database.output.push(format!("LOADED entries={} macros={macro_count} errors={errors}", database.entries.len()));
380 let mut assigned = vec![false; database.entries.len()];
381 for i in 0..database.entries.len() {
382 if assigned[i] { continue; }
383 let mut group: Vec<_> = (i..database.entries.len()).filter(|&j| database.entries[i].primary() == database.entries[j].primary() && database.entries[i].fields["year"] == database.entries[j].fields["year"]).collect();
384 group.sort_by(|&a, &b| (&database.entries[a].sort_title, &database.entries[a].key).cmp(&(&database.entries[b].sort_title, &database.entries[b].key)));
385 for (index, &j) in group.iter().enumerate() { assigned[j] = true; if group.len() >= 2 { database.entries[j].suffix = Some((b'a' + index as u8) as char); } }
386 }
387 database
388 }
389 fn ieee_order(&self) -> (Vec<usize>, Vec<usize>) {
390 let cited: BTreeSet<_> = self.record.iter().copied().collect();
391 let mut order = self.record.clone();
392 order.extend(self.lookup.values().copied().filter(|index| !cited.contains(index)));
393 let mut numbers = vec![0; self.entries.len()];
394 for (i, &index) in order.iter().enumerate() { numbers[index] = i + 1; }
395 (order, numbers)
396 }

400enum CitationKind { Parenthetical, Narrative, Author, Year }
401struct Citation { start: usize, end: usize, kind: CitationKind, keys: Vec<usize> }
402impl Database {
403 fn citation_text(&self, citation: &Citation, style: Style, numbers: &[usize]) -> String {
404 let parenthetical = citation.kind == CitationKind::Parenthetical;
405 if citation.kind == CitationKind::Author || citation.kind == CitationKind::Year || style == Style::Mla || (style == Style::Ieee && !parenthetical) {
406 let parts: Vec<_> = citation.keys.iter().map(|&index| {
407 let entry = &self.entries[index];
408 if citation.kind == CitationKind::Year { entry.year_label(style) }
409 else if style == Style::Ieee && citation.kind == CitationKind::Narrative { format!("{} [{}]", entry.short_name(style, false), numbers[index]) }
410 else { entry.short_name(style, false) }
411 }).collect();
412 let text = parts.join(if citation.kind == CitationKind::Year { ", " } else { "; " });
413 return if style == Style::Mla && parenthetical { format!("({text})") } else { text };
414 }
415 let mut parts = Vec::new();
416 if style == Style::Ieee {
417 let mut sorted: Vec<_> = citation.keys.iter().map(|&i| numbers[i]).collect();
418 sorted.sort_unstable();
419 let mut i = 0;
420 while i < sorted.len() {
421 let mut end = i + 1;
422 while end < sorted.len() && sorted[end] == sorted[end - 1] + 1 { end += 1; }
423 if end - i >= 3 { parts.push(format!("[{}]-[{}]", sorted[i], sorted[end - 1])); }
424 else { parts.extend(sorted[i..end].iter().map(|n| format!("[{n}]"))); }
425 i = end;
426 }
427 return parts.join(", ");
428 }
429 let mut keys = citation.keys.clone();
430 if parenthetical { keys.sort_by(|&a, &b| self.entries[a].sort_key().cmp(&self.entries[b].sort_key())); }
431 let mut i = 0;
432 while i < keys.len() {
433 let name = self.entries[keys[i]].short_name(style, parenthetical);
434 let mut end = i + 1;
435 while end < keys.len() && self.entries[keys[end]].short_name(style, parenthetical) == name { end += 1; }
436 let years = keys[i..end].iter().map(|&j| self.entries[j].year_label(style)).collect::<Vec<_>>().join(", ");
437 parts.push(if parenthetical { format!("{name}{}{years}", if style == Style::Apa { ", " } else { " " }) } else { format!("{name} ({years})") });
438 i = end;
439 }
440 let text = parts.join("; ");
441 if parenthetical { format!("({text})") } else { text }
442 }
443 fn cite(&mut self, style: Style, text: &str) -> Result<String, String> {
444 let mut citations = Vec::new();
445 let mut position = 0;
446 while let Some(offset) = text[position..].find('\\') {
447 let start = position + offset;
448 let brace = start + text[start..].find('{').unwrap();
449 let end = brace + text[brace..].find('}').unwrap();
450 let kind = match &text[start + 1..brace] { "citet" => CitationKind::Narrative, "citeauthor" => CitationKind::Author, "citeyear" => CitationKind::Year, _ => CitationKind::Parenthetical };
451 let mut keys = Vec::new();
452 let mut seen = BTreeSet::new();
453 for key in text[brace + 1..end].split(',') {
454 let &index = self.lookup.get(key).ok_or_else(|| format!("unknown key '{key}'"))?;
455 if seen.insert(index) { keys.push(index); }
456 }
457 citations.push(Citation { start, end: end + 1, kind, keys }); position = end + 1;
458 }
459 let mut recorded: BTreeSet<_> = self.record.iter().copied().collect();
460 for citation in &citations { for &index in &citation.keys { if recorded.insert(index) { self.record.push(index); } } }
461 let (_, numbers) = self.ieee_order();
462 let mut result = String::new(); position = 0;
463 for citation in &citations {
464 result.push_str(&text[position..citation.start]); result.push_str(&self.citation_text(citation, style, &numbers)); position = citation.end;
465 }
466 result.push_str(&text[position..]); Ok(result)
467 }
468 fn keygen(&self) -> Vec<String> {
469 let mut counts = BTreeMap::new();
470 let mut used = BTreeMap::new();
471 for entry in &self.entries { *counts.entry(entry.base_key.as_str()).or_insert(0) += 1; }
472 self.entries.iter().map(|entry| {
473 let mut key = entry.base_key.clone();
474 if counts[entry.base_key.as_str()] >= 2 {
475 let index = used.entry(entry.base_key.as_str()).or_insert(0u32);
476 key.push(char::from_u32('a' as u32 + *index).unwrap()); *index += 1;
477 }
478 format!("{} -> {key}", entry.key)
479 }).collect()
480 }
481 fn check(&self) -> Vec<String> {
482 let mut result = Vec::new();
483 for (i, entry) in self.entries.iter().enumerate() {
484 let mut warnings = Vec::new();
485 if let Some(pages) = entry.fields.get("pages") {
486 let pages = plain(pages);
487 if pages.contains('-') && !pages.contains("--") { warnings.push("pages use a single hyphen".into()); }
488 let parts: Vec<_> = pages.split('-').filter(|part| !part.is_empty()).collect();
489 if parts.len() == 2 && parts.iter().all(|part| digits(part)) && parts[0].parse::<u32>().unwrap() > parts[1].parse::<u32>().unwrap() { warnings.push("pages are reversed".into()); }
490 }
491 if entry.year.len() != 4 || !digits(&entry.year) { warnings.push("year is not 4 digits".into()); }
492 if !entry.plain_title.bytes().any(|c| c.is_ascii_lowercase()) && entry.plain_title.bytes().filter(u8::is_ascii_uppercase).count() >= 2 { warnings.push("title is all caps".into()); }
493 if entry.kind == 0 && !entry.fields.contains_key("doi") { warnings.push("article has no doi".into()); }
494 if let Some(previous) = self.entries[..i].iter().find(|other| other.lower_title == entry.lower_title) { warnings.push(format!("same title as {}", previous.key)); }
495 warnings.extend(entry.unknown_fields().into_iter().map(|name| format!("unknown field '{name}'")));
496 result.extend(warnings.into_iter().map(|warning| format!("WARN {}: {warning}", entry.key)));
497 }
498 result.push(format!("CHECK warnings={}", result.len())); result
499 }
500 fn export(&self, key: &str) -> Result<Vec<String>, String> {
501 let &index = self.lookup.get(key).ok_or_else(|| format!("unknown key '{key}'"))?;
502 let entry = &self.entries[index];
503 let mut names: Vec<_> = FIELDS.into_iter().filter(|name| entry.fields.contains_key(*name)).collect();
504 names.extend(entry.unknown_fields());
505 let mut result = vec![format!("@{}{{{key},", TYPES[entry.kind])];
506 for (i, name) in names.iter().enumerate() { result.push(format!(" {name} = {{{}}}{}", entry.fields[*name], if i + 1 == names.len() { "" } else { "," })); }
507 result.push("}".into()); Ok(result)
508 }
509 fn stats(&self) -> Vec<String> {
510 let mut types = BTreeMap::new();
511 let mut years = BTreeMap::new();
512 let mut authors = BTreeSet::new();
513 let mut pairs = BTreeMap::new();
514 for entry in &self.entries {
515 *types.entry(TYPES[entry.kind]).or_insert(0usize) += 1;
516 *years.entry(entry.year.as_str()).or_insert(0usize) += 1;
517 if let Some(list) = &entry.authors {
518 let names: BTreeSet<_> = list.names.iter().map(|name| name.text(false, false)).collect();
519 authors.extend(names.iter().cloned());
520 let names: Vec<_> = names.into_iter().collect();
521 for a in 0..names.len() { for b in a + 1..names.len() { *pairs.entry((names[a].clone(), names[b].clone())).or_insert(0usize) += 1; } }
522 }
523 }
524 let mut result = vec![format!("ENTRIES {}", self.entries.len())];
525 result.extend(types.iter().map(|(kind, count)| format!("TYPE {kind} {count}")));
526 result.extend(years.iter().map(|(year, count)| format!("YEAR {year} {count}")));
527 result.push(format!("AUTHORS {}", authors.len()));
528 let mut best_count = 0;
529 let mut best = None;
530 for (pair, &count) in &pairs { if count > best_count { best = Some(pair); best_count = count; } }
531 result.push(match best { Some((a, b)) => format!("PAIR {a} & {b} {best_count}"), None => "PAIR none".into() }); result
532 }
533 fn command(&mut self, line: &str) -> Result<Vec<String>, String> {
534 let command = line.split(' ').next().unwrap();
535 let words: Vec<_> = if command == "CITE" || command == "SEARCH" { line.splitn(3, ' ').collect() } else { line.split(' ').collect() };
536 let arity = match command { "FORMAT" => None, "BIBLIOGRAPHY" | "EXPORT" => Some(1), "CITE" | "SEARCH" => Some(2), "KEYGEN" | "CHECK" | "STATS" => Some(0), _ => return Err(format!("unknown command '{command}'")) };
537 if arity.map_or(words.len() < 3, |n| words.len() != n + 1) { return Err("wrong number of arguments".into()); }
538 let style = if matches!(command, "FORMAT" | "BIBLIOGRAPHY" | "CITE") { Style::parse(words[1]).ok_or_else(|| format!("unknown style '{}'", words[1]))? } else { Style::Apa };
539 match command {
540 "FORMAT" => {
541 let (_, numbers) = self.ieee_order();
542 Ok(words[2..].iter().map(|key| match self.lookup.get(*key) {
543 Some(&index) => self.entries[index].reference(style, numbers[index], false),
544 None => format!("ERROR unknown key '{key}'"),
545 }).collect())
546 },
547 "BIBLIOGRAPHY" => {
548 let (mut order, numbers) = self.ieee_order();
549 if style != Style::Ieee { order.sort_by(|&a, &b| self.entries[a].sort_key().cmp(&self.entries[b].sort_key())); }
550 let mut result = vec![if style == Style::Mla { "Works Cited" } else { "References" }.into()];
551 for (i, &index) in order.iter().enumerate() {
552 let repeated = i > 0 && matches!(style, Style::Chicago | Style::Mla) && self.entries[index].primary() == self.entries[order[i - 1]].primary();
553 result.push(self.entries[index].reference(style, numbers[index], repeated));
554 }
555 Ok(result)
556 },
557 "CITE" => Ok(vec![self.cite(style, words[2])?]),
558 "KEYGEN" => Ok(self.keygen()), "CHECK" => Ok(self.check()), "EXPORT" => self.export(words[1]), "STATS" => Ok(self.stats()),
559 "SEARCH" => {
560 let needle = words[2].to_ascii_lowercase();
561 let matches: Vec<_> = self.entries.iter().filter(|entry| entry.fields.get(words[1]).is_some_and(|value| plain(value).to_ascii_lowercase().contains(&needle))).map(|entry| entry.key.as_str()).collect();
562 Ok(vec![if matches.is_empty() { "MATCHES 0".into() } else { format!("MATCHES {}: {}", matches.len(), matches.join(" ")) }])
563 }, _ => unreachable!(),
564 }
565 }
566}
567fn main() {
568 let mut input = String::new(); io::stdin().read_to_string(&mut input).unwrap();
569 let lines: Vec<_> = input.lines().collect();
570 let separator = lines.iter().position(|&line| line == "%%").unwrap();
571 let mut database = Database::read(&lines[..separator]);
572 let count: usize = lines[separator + 1].parse().unwrap();
573 for line in &lines[separator + 2..separator + 2 + count] {
574 match database.command(line) { Ok(output) => database.output.extend(output), Err(error) => database.output.push(format!("ERROR {error}")) }
575 }
576 println!("{}", database.output.join("\n"));
577}
