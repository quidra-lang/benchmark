26 public static bool Letter(char c) => c is >= 'a' and <= 'z' or >= 'A' and <= 'Z';
27 public static bool Digit(char c) => c is >= '0' and <= '9';
28 public static bool Digits(string text) => text.Length > 0 && text.All(Digit);
29 public static char Lower(char c) => c is >= 'A' and <= 'Z' ? (char)(c + 32) : c;
30 public static char Upper(char c) => c is >= 'a' and <= 'z' ? (char)(c - 32) : c;
31 public static string Lower(string text) => new(text.Select(Lower).ToArray());
32 public static string Plain(string text) => text.Replace("{", "").Replace("}", "");
33 public static string Letters(string text) => new(text.Where(Letter).ToArray());

35 public static List<string> SplitTop(string text, string separator)
36 {
37 List<string> result = [];
38 int depth = 0, start = 0, i = 0;
39 while (i < text.Length)
40 {
41 if (depth == 0 && text.AsSpan(i).StartsWith(separator, StringComparison.Ordinal))
42 {
43 result.Add(text[start..i]);
44 i += separator.Length;
45 start = i;
46 continue;
47 }
48 if (text[i] == '{') depth++;
49 if (text[i] == '}') depth--;
50 i++;
51 }
52 result.Add(text[start..]);
53 return result;
54 }
55
56 static bool Accent(string text, int i, out char value, out int length)
57 {
58 value = default;
59 length = 0;
60 if (i + 1 >= text.Length || text[i] != '\\') return false;
61 char command = text[i + 1];
62 if (command == '&') { value = '&'; length = 2; return true; }
63 bool symbol = "'\"`^~=.".Contains(command);
64 if (i + 4 < text.Length && (symbol || "cuvH".Contains(command)) &&
65 text[i + 2] == '{' && Letter(text[i + 3]) && text[i + 4] == '}')
66 {
67 value = text[i + 3]; length = 5; return true;
68 }
69 if (symbol && i + 2 < text.Length && Letter(text[i + 2]))
70 {
71 value = text[i + 2]; length = 3; return true;
72 }
73 return false;
74 }

76 public static string Normalize(string text)
77 {
78 StringBuilder accents = new();
79 for (int i = 0; i < text.Length;)
80 {
81 if (text[i] == '{' && Accent(text, i + 1, out char wrapped, out int count) &&
82 i + count + 1 < text.Length && text[i + count + 1] == '}')
83 {
84 accents.Append(wrapped);
85 i += count + 2;
86 }
87 else if (Accent(text, i, out char value, out int length))
88 {
89 accents.Append(value);
90 i += length;
91 }
92 else accents.Append(text[i++]);
93 }
94 StringBuilder result = new();
95 bool space = false;
96 foreach (char c in accents.ToString())
97 {
98 if (c is ' ' or '~') space = result.Length > 0;
99 else
100 {
101 if (space) result.Append(' ');
102 result.Append(c);
103 space = false;
104 }
105 }
106 return result.ToString();
107 }

109 static string LowerOutside(string word)
110 {
111 char[] result = word.ToCharArray();
112 int depth = 0;
113 for (int i = 0; i < result.Length; i++)
114 {
115 if (result[i] == '{') depth++;
116 else if (result[i] == '}') depth--;
117 else if (depth == 0) result[i] = Lower(result[i]);
118 }
119 return new string(result);
120 }

122 public static string Title(string text, Style style)
123 {
124 if (style == Style.IEEE) return text;
125 if (style == Style.APA)
126 {
127 char[] result = text.ToCharArray();
128 int depth = 0;
129 for (int i = 0; i < result.Length; i++)
130 {
131 if (text[i] == '{') depth++;
132 else if (text[i] == '}') depth--;
133 else if (depth == 0 && Letter(text[i]))
134 result[i] = i == 0 || i >= 2 && text[i - 2] == ':' && text[i - 1] == ' ' ? Upper(text[i]) : Lower(text[i]);
135 }
136 return new string(result);
137 }
138 List<string> words = SplitTop(text, " ");
139 for (int i = 0; i < words.Count; i++)
140 {
141 string word = words[i];
142 if (word.StartsWith('{')) continue;
143 if (i > 0 && i + 1 < words.Count && !words[i - 1].EndsWith(':') && SmallWords.Contains(Lower(Letters(word))))
144 {
145 words[i] = LowerOutside(word);
146 continue;
147 }
148 int depth = 0;
149 for (int j = 0; j < word.Length; j++)
150 {
151 if (word[j] == '{') depth++;
152 else if (word[j] == '}') depth--;
153 else if (Letter(word[j]))
154 {
155 if (depth == 0) words[i] = word[..j] + Upper(word[j]) + word[(j + 1)..];
156 break;
157 }
158 }
159 }
160 return string.Join(' ', words);
161 }
162
163 public static string ListNames(string[] names, string conjunction, bool listB = false)
164 {
165 if (names.Length == 1) return names[0];
166 if (listB && names.Length == 2) return names[0] + " " + conjunction + " " + names[1];
167 return string.Join(", ", names[..^1]) + ", " + conjunction + " " + names[^1];
168 }

171sealed record Person(string First, string Von, string Last, string Jr)
172{
173 public string Surname => Plain(Von.Length == 0 ? Last : Von + " " + Last);
174 public string Initials => string.Join(' ', Plain(First).Split(' ', StringSplitOptions.RemoveEmptyEntries)
175 .Select(word => string.Join('-', word.Split('-').Select(piece => Upper(piece.First(Letter)) + "."))));
176
177 public string Format(bool inverted, bool initials)
178 {
179 string given = initials ? Initials : Plain(First);
180 string result = Surname;
181 if (First.Length > 0) result = inverted ? result + ", " + given : given + " " + result;
182 if (Jr.Length > 0) result += ", " + Plain(Jr);
183 return result;
184 }
185
186 public static Person Parse(string text)
187 {
188 string[] parts = SplitTop(text, ",").Select(part => part.Trim(' ')).ToArray();
189 List<string> words = SplitTop(parts[0], " ");
190 int firstLower = -1, lastLower = -1;
191 for (int i = 0; i + 1 < words.Count; i++)
192 if (words[i][0] is >= 'a' and <= 'z')
193 {
194 if (firstLower < 0) firstLower = i;
195 lastLower = i;
196 }
197 string Join(int first, int end) => string.Join(' ', words.Skip(first).Take(end - first));
198 if (parts.Length == 1)
199 return firstLower < 0
200 ? new Person(Join(0, words.Count - 1), "", words[^1], "")
201 : new Person(Join(0, firstLower), Join(firstLower, lastLower + 1), Join(lastLower + 1, words.Count), "");
202 string von = lastLower < 0 ? "" : Join(0, lastLower + 1);
203 string last = Join(lastLower + 1, words.Count);
204 return new Person(parts[^1], von, last, parts.Length == 3 ? parts[1] : "");
205 }
206}

209{
210 public bool Single => Names.Length == 1 && !EtAl;

212 public static NameList Parse(string text)
213 {
214 List<string> parts = SplitTop(text, " and ");
215 bool etAl = parts[^1] == "others";
216 if (etAl) parts.RemoveAt(parts.Count - 1);
217 return new NameList(parts.Select(Person.Parse).ToArray(), etAl);

220 public string Authors(Style style)
221 {
222 bool initials = style is Style.APA or Style.IEEE;
223 string[] values = Names.Select((name, i) => name.Format(style == Style.APA || style != Style.IEEE && i == 0, initials)).ToArray();
224 return style switch
225 {
226 Style.APA when EtAl => string.Join(", ", values) + ", et al.",
227 Style.APA when values.Length > 20 => string.Join(", ", values[..19]) + ", . . . " + values[^1],
228 Style.APA => ListNames(values, "&"),
229 Style.IEEE when EtAl || values.Length >= 7 => values[0] + " et al.",
230 Style.IEEE => ListNames(values, "and", true),
231 Style.CHICAGO when EtAl || values.Length >= 11 => string.Join(", ", values.Take(7)) + ", et al.",
232 Style.CHICAGO => ListNames(values, "and"),
233 Style.MLA when EtAl || values.Length >= 3 => values[0] + ", et al.",
234 _ => ListNames(values, "and")
235 };
236 }
237
238 public string Editors(Style style)
239 {
240 string[] values = Names.Select(name => name.Format(false, style is Style.APA or Style.IEEE)).ToArray();
241 return EtAl ? values[0] + " et al." : ListNames(values, style == Style.APA ? "&" : "and", true);
242 }

257 get

259 string title = Lower(Value("title"));

269 string[] right = [Lower(Plain(b.Primary.Names[0].Last)), b.Value("year"), b.SortTitle, b.Key];
270 for (int i = 0; i < left.Length; i++)
271 {
272 int order = string.CompareOrdinal(left[i], right[i]);

280sealed class Database
281{
282 static readonly Dictionary<string, string[]> Required = new(StringComparer.Ordinal)
283 {
284 ["article"] = ["author", "title", "journal", "year"],
285 ["book"] = ["author", "title", "publisher", "year"],
286 ["inproceedings"] = ["author", "title", "booktitle", "year"],
287 ["incollection"] = ["author", "title", "booktitle", "publisher", "year"],
288 ["techreport"] = ["author", "title", "institution", "year"],
289 ["phdthesis"] = ["author", "title", "school", "year"],
290 ["misc"] = ["author", "title", "year"]
291 };
292 readonly Dictionary<string, string> macros = new(StringComparer.Ordinal);
293 public List<Entry> Entries { get; } = [];
294 public Dictionary<string, Entry> ByKey { get; } = new(StringComparer.Ordinal);
295 public List<string> Messages { get; } = [];
296 int macroCount;
297
298 public Database(IReadOnlyList<string> lines)
299 {
300 string[] monthKeys = "jan feb mar apr may jun jul aug sep oct nov dec".Split(' ');
301 for (int i = 0; i < monthKeys.Length; i++) macros[monthKeys[i]] = MonthNames[i];
302 for (int i = 0; i < lines.Count; i++)
303 {
304 string line = lines[i];
305 if (line.Length == 0 || line.StartsWith('%')) continue;
306 if (line.StartsWith("@string{", StringComparison.Ordinal))
307 {
308 int separator = line.IndexOf(" = ", StringComparison.Ordinal);
309 string name = line[8..separator];
310 try
311 {
312 string value = Evaluate(line[(separator + 3)..^1], i + 1);
313 if (macros.ContainsKey(name)) throw new LoadError(i + 1, "duplicate macro");
314 macros[name] = value;
315 macroCount++;
316 }
317 catch (LoadError error) { Report(name, error); }
318 }
319 else
320 {
321 int end = i + 1;
322 while (lines[end] != "}") end++;
323 int brace = line.IndexOf('{');
324 string type = line[1..brace], key = line[(brace + 1)..^1];
325 try { ReadEntry(lines, i, end, type, key); }
326 catch (LoadError error) { Report(key, error); }
327 i = end;
328 }
329 }
330 AssignSuffixes();
331 Messages.Add($"LOADED entries={Entries.Count} macros={macroCount} errors={Messages.Count}");
332 }
333
334 void Report(string name, LoadError error) => Messages.Add($"ERROR line {error.Line}: {name}: {error.Message}");
335
336 string Evaluate(string expression, int line)
337 {
338 int depth = 0;
339 foreach (char c in expression)
340 {
341 if (c == '{') depth++;
342 else if (c == '}') depth--;
343 if (depth < 0) throw new LoadError(line, "unbalanced braces");
344 }
345 if (depth != 0) throw new LoadError(line, "unbalanced braces");
346 StringBuilder result = new();
347 foreach (string term in expression.Split(" # ", StringSplitOptions.None))
348 {
349 if (term[0] is '{' or '"') result.Append(term[1..^1]);
350 else if (Digit(term[0])) result.Append(term);
351 else if (macros.TryGetValue(term, out string? value)) result.Append(value);
352 else throw new LoadError(line, $"unknown macro '{term}'");
353 }
354 return result.ToString();
355 }
356
357 void ReadEntry(IReadOnlyList<string> lines, int start, int end, string type, string key)
358 {
359 if (!Required.TryGetValue(type, out string[]? required)) throw new LoadError(start + 1, $"unknown type '{type}'");
360 if (ByKey.ContainsKey(key)) throw new LoadError(start + 1, "duplicate key");
361 Dictionary<string, string> fields = new(StringComparer.Ordinal);
362 for (int i = start + 1; i < end; i++)
363 {
364 string line = lines[i];
365 int separator = line.IndexOf(" = ", StringComparison.Ordinal);
366 string name = line[2..separator], expression = line[(separator + 3)..];
367 if (expression.EndsWith(',')) expression = expression[..^1];
368 string value = Evaluate(expression, i + 1);
369 if (fields.ContainsKey(name)) throw new LoadError(i + 1, $"duplicate field '{name}'");
370 fields[name] = Normalize(value);
371 }
372 foreach (string field in required)
373 if (!fields.ContainsKey(field) && !(type == "book" && field == "author" && fields.ContainsKey("editor")))
374 throw new LoadError(end + 1, $"missing field '{field}'");
375 Entry entry = new(type, key, fields);
376 Entries.Add(entry);
377 ByKey[key] = entry;
378 }
379
380 void AssignSuffixes()
381 {
382 HashSet<Entry> done = [];
383 foreach (Entry entry in Entries)
384 {
385 if (done.Contains(entry)) continue;
386 Entry[] group = Entries.Where(other => other.Primary.Same(entry.Primary) && other.Fields["year"] == entry.Fields["year"])
387 .OrderBy(other => other.SortTitle, StringComparer.Ordinal).ThenBy(other => other.Key, StringComparer.Ordinal).ToArray();
388 for (int i = 0; i < group.Length; i++)
389 {
390 done.Add(group[i]);
391 if (group.Length > 1) group[i].Suffix = ((char)('a' + i)).ToString();
392 }
393 }
394 }
395}

441 static TemplatePart[] Parse(string text)
442 {
443 int i = 0;
444 TemplatePart[] Read()
445 {
446 List<TemplatePart> parts = [];
447 while (i < text.Length && text[i] != '>')
448 {
449 if (text[i] == '{')
450 {
451 int end = text.IndexOf('}', i);
452 parts.Add(new FieldPart(text[(i + 1)..end]));
453 i = end + 1;
454 }
455 else if (text[i] == '<')
456 {
457 i++;
458 parts.Add(new OptionalPart(Read()));
459 i++;
460 }
461 else parts.Add(new LiteralPart(text[i++]));
462 }
463 return parts.ToArray();
464 }
465 return Read();
466 }

468 static string EditorSuffix(Entry entry, Style style)
469 {
470 if (entry.Authors is not null) return "";
471 bool single = entry.Primary.Single;
472 return style switch
473 {
474 Style.APA => single ? " (Ed.)" : " (Eds.)",
475 Style.IEEE => single ? ", Ed." : ", Eds.",
476 Style.CHICAGO => single ? ", ed." : ", eds.",
477 _ => single ? ", editor" : ", editors"

481 static string Edition(string value)
482 {
483 if (!Digits(value)) return value + " ed.";
484 int number = int.Parse(value);
485 string suffix = number % 100 is 11 or 12 or 13 ? "th" : (number % 10) switch { 1 => "st", 2 => "nd", 3 => "rd", _ => "th" };
486 return value + suffix + " ed.";
487 }

489 static Dictionary<string, string> Values(Entry entry, Style style, bool repeated)
490 {
491 Dictionary<string, string> result = entry.Fields.ToDictionary(pair => pair.Key, pair => Plain(pair.Value), StringComparer.Ordinal);
492 result["author"] = (repeated ? "---" : entry.Primary.Authors(style)) + EditorSuffix(entry, style);
493 result["editor"] = entry.Editors?.Editors(style) ?? "";
494 result["eds"] = entry.Editors is null ? "" : entry.Editors.Single ? "Ed." : "Eds.";
495 result["title"] = Plain(Title(entry.Fields["title"], style));
496 result["year"] = entry.Year(style);
497 result["pp"] = "";
498 if (entry.Fields.ContainsKey("pages"))
499 {
500 result["pages"] = entry.Value("pages").Replace("--", "-");
501 result["pp"] = (result["pages"].Contains('-') ? "pp. " : "p. ") + result["pages"];
502 }
503 if (entry.Fields.ContainsKey("edition")) result["edition"] = Edition(entry.Value("edition"));
504 string month = entry.Value("month");
505 int index = Array.IndexOf(MonthNames, month);
506 result["mon"] = index < 0 ? month : MonthShort[index];
507 return result;
508 }

510 public static string Format(Entry entry, Style style, int number = 0, bool repeated = false)
511 {
512 Dictionary<string, string> values = Values(entry, style, repeated);
513 StringBuilder result = new();
514 void Render(TemplatePart[] parts)
515 {
516 foreach (TemplatePart part in parts)
517 switch (part)
518 {
519 case FieldPart field:
520 result.Append(values.GetValueOrDefault(field.Name, ""));
521 break;
522 case OptionalPart optional:
523 if (optional.Parts.OfType<FieldPart>().All(placeholder => values.GetValueOrDefault(placeholder.Name, "").Length > 0)) Render(optional.Parts);
524 break;
525 case LiteralPart literal:
526 if (literal.Character != '.' || result.Length == 0 || result[^1] is not ('.' or '?' or '!')) result.Append(literal.Character);
527 break;
528 }
529 }
530 int type = entry.Type switch { "article" => 0, "book" => 1, "inproceedings" or "incollection" => 2, "techreport" => 3, "phdthesis" => 4, _ => 5 };
531 Render(Parsed[(int)style][type]);
532 return (style == Style.IEEE ? $"[{number}] " : "") + result;
533 }

560 static string NumberRanges(int[] numbers)
561 {
562 Array.Sort(numbers);
563 List<string> parts = [];
564 for (int i = 0; i < numbers.Length;)
565 {
566 int end = i;
567 while (end + 1 < numbers.Length && numbers[end + 1] == numbers[end] + 1) end++;
568 if (end - i >= 2) parts.Add($"[{numbers[i]}]-[{numbers[end]}]");
569 else for (int j = i; j <= end; j++) parts.Add($"[{numbers[j]}]");
570 i = end + 1;
571 }
572 return string.Join(", ", parts);
573 }

575 static string CitationText(Citation citation, Style style, Dictionary<Entry, int> numbers)
576 {
577 Entry[] entries = citation.Entries;
578 if (citation.Kind == "citeauthor") return string.Join("; ", entries.Select(entry => ShortName(entry, style, false)));
579 if (citation.Kind == "citeyear") return string.Join(", ", entries.Select(entry => entry.Year(style)));
580 bool parenthetical = citation.Kind != "citet";
581 if (style == Style.IEEE)
582 return parenthetical ? NumberRanges(entries.Select(entry => numbers[entry]).ToArray())
583 : string.Join("; ", entries.Select(entry => ShortName(entry, style, false) + $" [{numbers[entry]}]"));
584 if (style == Style.MLA)
585 {
586 string text = string.Join("; ", entries.Select(entry => ShortName(entry, style, parenthetical)));
587 return parenthetical ? "(" + text + ")" : text;
588 }
589 if (parenthetical) entries = entries.OrderBy(entry => entry, Entry.BibliographyComparer).ToArray();
590 List<string> groups = [];
591 for (int i = 0; i < entries.Length;)
592 {
593 string name = ShortName(entries[i], style, parenthetical);
594 int end = i + 1;
595 while (end < entries.Length && ShortName(entries[end], style, parenthetical) == name) end++;
596 string years = string.Join(", ", entries[i..end].Select(entry => entry.Year(style)));
597 groups.Add(parenthetical ? name + (style == Style.APA ? ", " : " ") + years : name + " (" + years + ")");
598 i = end;
599 }
600 string joined = string.Join("; ", groups);
601 return parenthetical ? "(" + joined + ")" : joined;
602 }

604 string Cite(string text, Style style)
605 {
606 List<Citation> citations = [];
607 for (int start = text.IndexOf('\\'); start >= 0;)
608 {
609 int brace = text.IndexOf('{', start), end = text.IndexOf('}', brace);
610 HashSet<string> seen = new(StringComparer.Ordinal);
611 List<Entry> entries = [];
612 foreach (string key in text[(brace + 1)..end].Split(','))
613 if (seen.Add(key)) entries.Add(Lookup(key));
614 citations.Add(new Citation(start, end + 1, text[(start + 1)..brace], entries.ToArray()));
615 start = text.IndexOf('\\', end + 1);
616 }
617 foreach (Citation citation in citations)
618 foreach (Entry entry in citation.Entries)
619 if (recorded.Add(entry)) record.Add(entry);
620 Dictionary<Entry, int> numbers = IeeeNumbers();
621 StringBuilder result = new();
622 int position = 0;
623 foreach (Citation citation in citations)
624 {
625 result.Append(text.AsSpan(position, citation.Start - position));
626 result.Append(CitationText(citation, style, numbers));
627 position = citation.End;
628 }
629 result.Append(text.AsSpan(position));
630 return result.ToString();
631 }

633 static string BaseKey(Entry entry)
634 {
635 string surname = Lower(Letters(Plain(entry.Primary.Names[0].Last)));
636 string year = new(entry.Fields["year"].Where(Digit).ToArray());
637 string word = entry.Value("title").Split(' ', StringSplitOptions.RemoveEmptyEntries).Select(part => Lower(Letters(part)))
638 .FirstOrDefault(part => part.Length > 0 && !SmallWords.Contains(part)) ?? "";
639 return surname + year + word;
640 }

642 IEnumerable<string> Keygen()
643 {
644 string[] bases = database.Entries.Select(BaseKey).ToArray();
645 Dictionary<string, int> counts = bases.GroupBy(value => value, StringComparer.Ordinal).ToDictionary(group => group.Key, group => group.Count(), StringComparer.Ordinal);
646 Dictionary<string, int> used = new(StringComparer.Ordinal);
647 for (int i = 0; i < bases.Length; i++)
648 {
649 string generated = bases[i];
650 if (counts[generated] > 1)
651 {
652 int index = used.GetValueOrDefault(generated);
653 used[generated] = index + 1;
654 generated += (char)('a' + index);
655 }
656 yield return database.Entries[i].Key + " -> " + generated;
657 }
658 }

660 static IEnumerable<string> Warnings(Entry entry, Dictionary<string, string> titles)
661 {
662 if (entry.Fields.ContainsKey("pages"))
663 {
664 string pages = entry.Value("pages");
665 if (pages.Contains('-') && !pages.Contains("--", StringComparison.Ordinal)) yield return "pages use a single hyphen";
666 string[] parts = pages.Split('-', StringSplitOptions.RemoveEmptyEntries);
667 if (parts.Length == 2 && parts.All(Digits) && int.Parse(parts[0]) > int.Parse(parts[1])) yield return "pages are reversed";
668 }
669 string year = entry.Value("year");
670 if (year.Length != 4 || !Digits(year)) yield return "year is not 4 digits";
671 string title = entry.Value("title");
672 if (!title.Any(c => c is >= 'a' and <= 'z') && title.Count(c => c is >= 'A' and <= 'Z') >= 2) yield return "title is all caps";
673 if (entry.Type == "article" && !entry.Fields.ContainsKey("doi")) yield return "article has no doi";
674 string lowered = Lower(title);
675 if (titles.TryGetValue(lowered, out string? key)) yield return "same title as " + key;
676 else titles[lowered] = entry.Key;
677 foreach (string field in entry.Fields.Keys.Where(field => !KnownFields.Contains(field)).OrderBy(field => field, StringComparer.Ordinal))
678 yield return $"unknown field '{field}'";
679 }

682 {
683 Dictionary<string, string> titles = new(StringComparer.Ordinal);
684 int count = 0;
685 foreach (Entry entry in database.Entries)
686 foreach (string warning in Warnings(entry, titles))
687 {
688 count++;
689 yield return $"WARN {entry.Key}: {warning}";
690 }
691 yield return $"CHECK warnings={count}";
692 }

695 {
696 yield return "@" + entry.Type + "{" + entry.Key + ",";
697 string[] fields = KnownFields.Where(entry.Fields.ContainsKey).Concat(entry.Fields.Keys.Where(field => !KnownFields.Contains(field))
698 .OrderBy(field => field, StringComparer.Ordinal)).ToArray();
699 for (int i = 0; i < fields.Length; i++)
700 yield return " " + fields[i] + " = {" + entry.Fields[fields[i]] + "}" + (i + 1 == fields.Length ? "" : ",");
701 yield return "}";

704 IEnumerable<string> Stats()
705 {
706 yield return $"ENTRIES {database.Entries.Count}";
707 foreach (IGrouping<string, Entry> group in database.Entries.GroupBy(entry => entry.Type, StringComparer.Ordinal).OrderBy(group => group.Key, StringComparer.Ordinal))
708 yield return $"TYPE {group.Key} {group.Count()}";
709 foreach (IGrouping<string, Entry> group in database.Entries.GroupBy(entry => entry.Value("year"), StringComparer.Ordinal).OrderBy(group => group.Key, StringComparer.Ordinal))
710 yield return $"YEAR {group.Key} {group.Count()}";
711 HashSet<string> authors = new(StringComparer.Ordinal);
712 Dictionary<(string First, string Second), int> pairs = [];
713 foreach (Entry entry in database.Entries)
714 {
715 if (entry.Authors is null) continue;
716 string[] names = entry.Authors.Names.Select(name => name.Format(false, false)).Distinct(StringComparer.Ordinal)
717 .OrderBy(name => name, StringComparer.Ordinal).ToArray();
718 authors.UnionWith(names);
719 for (int i = 0; i < names.Length; i++)
720 for (int j = i + 1; j < names.Length; j++)
721 {
722 (string, string) pair = (names[i], names[j]);
723 pairs[pair] = pairs.GetValueOrDefault(pair) + 1;
724 }
725 }
726 yield return $"AUTHORS {authors.Count}";
727 if (pairs.Count == 0) yield return "PAIR none";
728 else
729 {
730 KeyValuePair<(string First, string Second), int> best = pairs.OrderByDescending(pair => pair.Value)
731 .ThenBy(pair => pair.Key.First, StringComparer.Ordinal).ThenBy(pair => pair.Key.Second, StringComparer.Ordinal).First();
732 yield return $"PAIR {best.Key.First} & {best.Key.Second} {best.Value}";
733 }

736 public List<string> Execute(string line)
737 {
738 int firstSpace = line.IndexOf(' ');
739 string command = firstSpace < 0 ? line : line[..firstSpace];
740 string[] words = command is "CITE" or "SEARCH" ? line.Split(' ', 3, StringSplitOptions.None) : line.Split(' ');
741 int expected = command switch
742 {
743 "FORMAT" or "CITE" or "SEARCH" => 2,
744 "BIBLIOGRAPHY" or "EXPORT" => 1,
745 "KEYGEN" or "CHECK" or "STATS" => 0,
746 _ => throw new CommandError($"unknown command '{command}'")
747 };
748 if (command == "FORMAT" ? words.Length - 1 < expected : words.Length - 1 != expected)
749 throw new CommandError("wrong number of arguments");
750 Style style = Style.APA;
751 if (command is "FORMAT" or "BIBLIOGRAPHY" or "CITE")
752 style = words[1] switch
753 {
754 "APA" => Style.APA, "IEEE" => Style.IEEE, "CHICAGO" => Style.CHICAGO, "MLA" => Style.MLA,
755 _ => throw new CommandError($"unknown style '{words[1]}'")
756 };
757 List<string> output = [];
758 switch (command)
759 {
760 case "FORMAT":
761 {
762 Dictionary<Entry, int> numbers = IeeeNumbers();
763 foreach (string key in words[2..])
764 output.Add(database.ByKey.TryGetValue(key, out Entry? entry)
765 ? ReferenceFormatter.Format(entry, style, numbers[entry]) : $"ERROR unknown key '{key}'");
766 break;
767 }
768 case "BIBLIOGRAPHY":
769 {
770 output.Add(style == Style.MLA ? "Works Cited" : "References");
771 Dictionary<Entry, int> numbers = IeeeNumbers();
772 Entry[] entries = style == Style.IEEE ? IeeeOrder() : database.Entries.OrderBy(entry => entry, Entry.BibliographyComparer).ToArray();
773 NameList? previous = null;
774 foreach (Entry entry in entries)
775 {
776 bool repeated = (style is Style.CHICAGO or Style.MLA) && previous?.Same(entry.Primary) == true;
777 output.Add(ReferenceFormatter.Format(entry, style, numbers[entry], repeated));
778 previous = entry.Primary;
779 }
780 break;
781 }
782 case "CITE": output.Add(Cite(words[2], style)); break;
783 case "KEYGEN": output.AddRange(Keygen()); break;
784 case "CHECK": output.AddRange(Check()); break;
785 case "SEARCH":
786 {
787 string needle = Lower(words[2]);
788 string[] matches = database.Entries.Where(entry => entry.Fields.ContainsKey(words[1]) && Lower(entry.Value(words[1])).Contains(needle, StringComparison.Ordinal))
789 .Select(entry => entry.Key).ToArray();
790 output.Add(matches.Length == 0 ? "MATCHES 0" : $"MATCHES {matches.Length}: " + string.Join(' ', matches));
791 break;
792 }
793 case "EXPORT": output.AddRange(Export(Lookup(words[1]))); break;
794 case "STATS": output.AddRange(Stats()); break;
795 }
796 return output;
797 }
798}
