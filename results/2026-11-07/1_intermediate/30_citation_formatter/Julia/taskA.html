26ascii_lower(c::Char) = 'A' <= c <= 'Z' ? Char(Int(c) + 32) : c

36function split_depth(text::AbstractString, separator::String)
37 result = String[]
38 start = position = 1
39 depth = 0
40 while position <= lastindex(text)
41 if depth == 0 && startswith(SubString(text, position), separator)
42 push!(result, part(text, start, position - 1))
43 position += length(separator)
44 start = position
45 else
46 text[position] == '{' && (depth += 1)
47 text[position] == '}' && (depth -= 1)
48 position += 1
49 end
50 end
51 push!(result, part(text, start))
52end
53function evaluate(expression::AbstractString, macros::Dict{String,String}, line::Int)
54 depth = 0
55 for c in expression
56 c == '{' && (depth += 1)
57 c == '}' && (depth -= 1)
58 depth < 0 && load_fail(line, "unbalanced braces")
59 end
60 depth != 0 && load_fail(line, "unbalanced braces")
61 output = IOBuffer()
62 for term in split(expression, " # ")
63 if first(term) in ('{', '"')
64 print(output, part(term, 2, lastindex(term) - 1))
65 elseif all_digits(term)
66 print(output, term)
67 else
68 haskey(macros, term) || load_fail(line, "unknown macro '$term'")
69 print(output, macros[term])
70 end
71 end
72 String(take!(output))
73end
74function accent_at(text::AbstractString, position::Int)
75 position + 1 > lastindex(text) && return nothing
76 text[position] != '\\' && return nothing
77 command = text[position + 1]
78 command == '&' && return ('&', 2)
79 symbol = command in ('\'', '"', '`', '^', '~', '=', '.')
80 if symbol && position + 2 <= lastindex(text) && ascii_letter(text[position + 2])
81 return text[position + 2], 3
82 end
83 if (symbol || command in ('c', 'u', 'v', 'H')) && position + 4 <= lastindex(text) &&
84 text[position + 2] == '{' && ascii_letter(text[position + 3]) && text[position + 4] == '}'
85 return text[position + 3], 5
86 end
87 nothing
88end
89function normalize(text::AbstractString)
90 output = IOBuffer()
91 position = 1
92 while position <= lastindex(text)
93 wrapped = text[position] == '{' ? accent_at(text, position + 1) : nothing
94 if wrapped !== nothing && position + wrapped[2] + 1 <= lastindex(text) && text[position + wrapped[2] + 1] == '}'
95 print(output, wrapped[1])
96 position += wrapped[2] + 2
97 else
98 accent = accent_at(text, position)
99 print(output, accent === nothing ? text[position] : accent[1])
100 position += accent === nothing ? 1 : accent[2]
101 end
102 end
103 join(split(replace(String(take!(output)), '~' => ' ')), " ")
104end

118function parse_person(text::AbstractString)
119 parts = String.(strip.(split_depth(text, ",")))
120 words = split_depth(parts[1], " ")
121 lower = [i for i in 1:length(words)-1 if 'a' <= first(words[i]) <= 'z']
122 if length(parts) == 1
123 isempty(lower) && return Person(join(words[1:end-1], " "), "", words[end], "")
124 left, right = first(lower), last(lower)
125 return Person(join(words[1:left-1], " "), join(words[left:right], " "), join(words[right+1:end], " "), "")
126 end
127 right = isempty(lower) ? 0 : last(lower)
128 Person(parts[end], join(words[1:right], " "), join(words[right+1:end], " "), length(parts) == 3 ? parts[2] : "")
129end
130function parse_names(text::AbstractString)
131 parts = split_depth(text, " and ")
132 etal = last(parts) == "others"
133 etal && pop!(parts)
134 NameList(parse_person.(parts), etal)
135end

184function read_entry(lines::Vector{String}, start::Int, finish::Int, kind::String, key::String, lookup::Dict{String,Int}, macros::Dict{String,String})
185 haskey(REQUIRED_FIELDS, kind) || load_fail(start, "unknown type '$kind'")
186 haskey(lookup, key) && load_fail(start, "duplicate key")
187 fields = Dict{String,String}()
188 for i in start+1:finish-1
189 line = lines[i]
190 separator = first(findfirst(" = ", line))
191 name = part(line, 3, separator - 1)
192 expression = part(line, separator + 3, lastindex(line) - Int(endswith(line, ',')))
193 value = evaluate(expression, macros, i)
194 haskey(fields, name) && load_fail(i, "duplicate field '$name'")
195 fields[name] = normalize(value)
196 end
197 for required in REQUIRED_FIELDS[kind]
198 if !haskey(fields, required) && !(kind == "book" && required == "author" && haskey(fields, "editor"))
199 load_fail(finish, "missing field '$required'")
200 end
201 end
202 Entry(key, kind, fields)
203end
204function read_database(lines::Vector{String})
205 entries = Entry[]
206 lookup = Dict{String,Int}()
207 macros = Dict{String,String}(ascii_lower(month[1:3]) => month for month in MONTHS)
208 errors = 0
209 position = 1
210 while position <= length(lines)
211 line = lines[position]
212 if isempty(line) || startswith(line, '%')
213 position += 1
214 continue
215 end
216 if startswith(line, "@string{")
217 separator = first(findfirst(" = ", line))
218 name = part(line, 9, separator - 1)
219 try
220 value = evaluate(part(line, separator + 3, lastindex(line) - 1), macros, position)
221 haskey(macros, name) && load_fail(position, "duplicate macro")
222 macros[name] = value
223 catch problem
224 problem isa DatabaseError || rethrow()
225 errors += 1
226 println("ERROR line $(problem.line): $name: $(problem.message)")
227 end
228 else
229 opening = findfirst('{', line)
230 kind, key = part(line, 2, opening - 1), part(line, opening + 1, lastindex(line) - 1)
231 finish = position + 1
232 while lines[finish] != "}"
233 finish += 1
234 end
235 try
236 entry = read_entry(lines, position, finish, kind, key, lookup, macros)
237 push!(entries, entry)
238 lookup[key] = length(entries)
239 catch problem
240 problem isa DatabaseError || rethrow()
241 errors += 1
242 println("ERROR line $(problem.line): $key: $(problem.message)")
243 end
244 position = finish
245 end
246 position += 1
247 end
248 println("LOADED entries=$(length(entries)) macros=$(length(macros)-12) errors=$errors")
249 assigned = falses(length(entries))
250 for i in eachindex(entries)
251 assigned[i] && continue
252 entry = entries[i]
253 group = findall(other -> other.primary == entry.primary && field(other, "year") == field(entry, "year"), entries)
254 assigned[group] .= true
255 if length(group) >= 2
256 sort!(group; by = j -> (entries[j].sort_title, entries[j].key))
257 for (ordinal, j) in enumerate(group)
258 entries[j].suffix = string(Char(Int('a') + ordinal - 1))
259 end
260 end
261 end
262 Database(entries, lookup, sortperm(entries; by = entry -> entry.sort_key), Int[])
263end

270function author_names(names::NameList, style::Style)
271 forms = [name_text(name, style == APA || (style in (CHICAGO, MLA) && i == 1), style in (APA, IEEE)) for (i, name) in enumerate(names.names)]
272 if style == APA
273 names.etal && return join(forms, ", ") * ", et al."
274 length(forms) >= 21 && return join(forms[1:19], ", ") * ", . . . " * forms[end]
275 return list_names(forms, "&")
276 elseif style == IEEE
277 return names.etal || length(forms) >= 7 ? forms[1] * " et al." : list_names(forms, "and"; list_b = true)
278 elseif style == CHICAGO
279 return names.etal || length(forms) >= 11 ? join(forms[1:min(7, end)], ", ") * ", et al." : list_names(forms, "and")
280 end
281 names.etal || length(forms) >= 3 ? forms[1] * ", et al." : list_names(forms, "and")
282end

289function convert_case(text::String, mode::Symbol)
290 result = collect(text)
291 depth = 0
292 first_letter = true
293 for i in eachindex(result)
294 c = result[i]
295 if c == '{'
296 depth += 1
297 elseif c == '}'
298 depth -= 1
299 elseif ascii_letter(c)
300 if depth == 0
301 if mode == :lower
302 result[i] = ascii_lower(c)
303 elseif mode == :sentence
304 result[i] = i == 1 || i >= 3 && text[i-2:i-1] == ": " ? ascii_upper(c) : ascii_lower(c)
305 elseif first_letter
306 result[i] = ascii_upper(c)
307 end
308 end
309 first_letter = false
310 end
311 end
312 join(result)
313end
314function title_text(title::String, style::Style)
315 style == IEEE && return plain(title)
316 style == APA && return plain(convert_case(title, :sentence))
317 words = split_depth(title, " ")
318 for i in eachindex(words)
319 word = words[i]
320 startswith(word, '{') && continue
321 small = 1 < i < length(words) && !endswith(words[i-1], ':') && letters_only(word) in SMALL_WORDS
322 words[i] = convert_case(word, small ? :lower : :title)
323 end
324 plain(join(words, " "))
325end

361function optional_end(pattern::String, start::Int)
362 position, depth = start, 1
363 while depth > 0
364 position += 1
365 pattern[position] == '<' && (depth += 1)
366 pattern[position] == '>' && (depth -= 1)
367 end
368 position
369end
370function optional_present(pattern::String, start::Int, finish::Int, values::Dict{String,String})
371 position = start
372 while position < finish
373 if pattern[position] == '<'
374 position = optional_end(pattern, position)
375 elseif pattern[position] == '{'
376 closing = findnext('}', pattern, position)
377 isempty(values[part(pattern, position + 1, closing - 1)]) && return false
378 position = closing
379 end
380 position += 1
381 end
382 true
383end
384function render_template!(output::IOBuffer, last_output::Ref{Char}, pattern::String, start::Int, finish::Int, values::Dict{String,String})
385 position = start
386 while position < finish
387 c = pattern[position]
388 if c == '<'
389 closing = optional_end(pattern, position)
390 if optional_present(pattern, position + 1, closing, values)
391 render_template!(output, last_output, pattern, position + 1, closing, values)
392 end
393 position = closing
394 elseif c == '{'
395 closing = findnext('}', pattern, position)
396 value = values[part(pattern, position + 1, closing - 1)]
397 print(output, value)
398 !isempty(value) && (last_output[] = last(value))
399 position = closing
400 elseif c != '.' || !(last_output[] in ('.', '?', '!'))
401 print(output, c)
402 last_output[] = c
403 end
404 position += 1
405 end
406end
407function placeholder_values(entry::Entry, style::Style, repeated::Bool)
408 values = Dict{String,String}(name => plain(field(entry, name)) for name in KNOWN_FIELDS)
409 for name in ("eds", "pp", "mon")
410 values[name] = ""
411 end
412 author = repeated ? "---" : author_names(entry.primary, style)
413 if !haskey(entry.fields, "author")
414 labels = style == APA ? (" (Ed.)", " (Eds.)") : style == IEEE ? (", Ed.", ", Eds.") :
415 style == CHICAGO ? (", ed.", ", eds.") : (", editor", ", editors")
416 author *= labels[singular(entry.primary) ? 1 : 2]
417 end
418 values["author"] = author
419 values["editor"] = editor_names(entry.editors, style)
420 if entry.editors !== nothing
421 values["eds"] = singular(entry.editors) ? "Ed." : "Eds."
422 end
423 values["title"] = title_text(field(entry, "title"), style)
424 values["year"] = year_label(entry, style)
425 values["pages"] = replace(values["pages"], "--" => "-")
426 if haskey(entry.fields, "pages")
427 values["pp"] = (occursin('-', values["pages"]) ? "pp. " : "p. ") * values["pages"]
428 end
429 if haskey(entry.fields, "edition")
430 edition = values["edition"]
431 if all_digits(edition)
432 n = parse(Int, edition)
433 suffix = n % 100 in 11:13 ? "th" : n % 10 == 1 ? "st" : n % 10 == 2 ? "nd" : n % 10 == 3 ? "rd" : "th"
434 edition *= suffix
435 end
436 values["edition"] = edition * " ed."
437 end
438 if haskey(entry.fields, "month")
439 index = findfirst(==(values["month"]), MONTHS)
440 values["mon"] = index === nothing ? values["month"] : MONTH_SHORT[index]
441 end
442 values
443end

476function citation_text(database::Database, citation::Citation, style::Style, numbers::Vector{Int})
477 indices = citation.entries
478 parenthetical = citation.kind == Parenthetical
479 if citation.kind in (AuthorOnly, YearOnly) || style == MLA || style == IEEE && !parenthetical
480 values = [citation.kind == YearOnly ? year_label(database.entries[i], style) :
481 short_name(database.entries[i], style) * (style == IEEE && citation.kind == Narrative ? " [$(numbers[i])]" : "") for i in indices]
482 result = join(values, citation.kind == YearOnly ? ", " : "; ")
483 return style == MLA && parenthetical ? "($result)" : result
484 end
485 groups = String[]
486 if style == IEEE
487 ordered = sort(numbers[indices])
488 position = 1
489 while position <= length(ordered)
490 finish = position + 1
491 while finish <= length(ordered) && ordered[finish] == ordered[finish-1] + 1
492 finish += 1
493 end
494 if finish - position >= 3
495 push!(groups, "[$(ordered[position])]-[$(ordered[finish-1])]")
496 else
497 append!(groups, ["[$(ordered[i])]" for i in position:finish-1])
498 end
499 position = finish
500 end
501 return join(groups, ", ")
502 end
503 ordered = parenthetical ? sort(indices; by = i -> database.entries[i].sort_key) : indices
504 position = 1
505 while position <= length(ordered)
506 name = short_name(database.entries[ordered[position]], style, parenthetical)
507 finish = position + 1
508 while finish <= length(ordered) && short_name(database.entries[ordered[finish]], style, parenthetical) == name
509 finish += 1
510 end
511 years = join((year_label(database.entries[ordered[i]], style) for i in position:finish-1), ", ")
512 push!(groups, parenthetical ? name * (style == APA ? ", " : " ") * years : "$name ($years)")
513 position = finish
514 end
515 result = join(groups, "; ")
516 parenthetical ? "($result)" : result
517end
518function cite!(database::Database, style::Style, text::AbstractString)
519 citations = Citation[]
520 start = findfirst('\\', text)
521 while start !== nothing
522 opening = findnext('{', text, start)
523 finish = findnext('}', text, opening)
524 kind_text = part(text, start + 1, opening - 1)
525 kind = kind_text == "citet" ? Narrative : kind_text == "citeauthor" ? AuthorOnly : kind_text == "citeyear" ? YearOnly : Parenthetical
526 indices = Int[]
527 for key in unique(split(part(text, opening + 1, finish - 1), ','))
528 haskey(database.lookup, key) || fail("unknown key '$key'")
529 push!(indices, database.lookup[key])
530 end
531 push!(citations, Citation(start, finish, kind, indices))
532 start = findnext('\\', text, finish + 1)
533 end
534 recorded = Set(database.record)
535 for citation in citations, index in citation.entries
536 if !(index in recorded)
537 push!(recorded, index)
538 push!(database.record, index)
539 end
540 end
541 numbers = ieee_numbers(database)
542 output = IOBuffer()
543 position = 1
544 for citation in citations
545 print(output, part(text, position, citation.start - 1), citation_text(database, citation, style, numbers))
546 position = citation.finish + 1
547 end
548 print(output, part(text, position))
549 String(take!(output))
550end

562 end
563 seen = Dict{String,Int}()
564 for (i, base) in enumerate(bases)
565 ordinal = get(seen, base, 0)
566 suffix = totals[base] >= 2 ? string(Char(Int('a') + ordinal)) : ""
567 seen[base] = ordinal + 1
568 println("$(database.entries[i].key) -> $base$suffix")
569 end
570end
571unknown_fields(entry::Entry) = sort(setdiff(collect(keys(entry.fields)), KNOWN_FIELDS))
572function check_entries(database::Database)
573 warnings = 0
574 titles = Dict{String,String}()
575 for entry in database.entries
576 messages = String[]
577 if haskey(entry.fields, "pages")
578 pages = plain(field(entry, "pages"))
579 occursin('-', pages) && !occursin("--", pages) && push!(messages, "pages use a single hyphen")
580 parts = split(pages, '-'; keepempty = false)
581 if length(parts) == 2 && all(all_digits, parts) && parse(Int, parts[1]) > parse(Int, parts[2])
582 push!(messages, "pages are reversed")
583 end
584 end
585 !(length(entry.year) == 4 && all_digits(entry.year)) && push!(messages, "year is not 4 digits")
586 title = plain(field(entry, "title"))
587 if !any(c -> 'a' <= c <= 'z', title) && count(c -> 'A' <= c <= 'Z', title) >= 2
588 push!(messages, "title is all caps")
589 end
590 entry.kind == "article" && !haskey(entry.fields, "doi") && push!(messages, "article has no doi")
591 lowered = ascii_lower(title)
592 if haskey(titles, lowered)
593 push!(messages, "same title as $(titles[lowered])")
594 else
595 titles[lowered] = entry.key
596 end
597 append!(messages, ["unknown field '$name'" for name in unknown_fields(entry)])
598 for message in messages
599 println("WARN $(entry.key): $message")
600 end
601 warnings += length(messages)
602 end
603 println("CHECK warnings=$warnings")
604end
605function export_entry(entry::Entry)
606 fields = vcat(filter(name -> haskey(entry.fields, name), KNOWN_FIELDS), unknown_fields(entry))
607 println("@$(entry.kind){$(entry.key),")
608 for (i, name) in enumerate(fields)
609 println(" $name = {$(field(entry, name))}", i < length(fields) ? "," : "")
610 end
611 println('}')
612end
613function statistics(database::Database)
614 println("ENTRIES $(length(database.entries))")
615 for property in (:kind, :year)
616 counts = Dict{String,Int}()
617 for entry in database.entries
618 value = getfield(entry, property)
619 counts[value] = get(counts, value, 0) + 1
620 end
621 label = property == :kind ? "TYPE" : "YEAR"
622 for value in sort(collect(keys(counts)))
623 println("$label $value $(counts[value])")
624 end
625 end
626 authors = Set{String}()
627 pairs = Dict{Tuple{String,String},Int}()
628 for entry in database.entries
629 haskey(entry.fields, "author") || continue
630 names = sort(unique([name_text(person, false, false) for person in entry.primary.names]))
631 union!(authors, names)
632 for i in eachindex(names), j in i+1:length(names)
633 pair = (names[i], names[j])
634 pairs[pair] = get(pairs, pair, 0) + 1
635 end
636 end
637 println("AUTHORS $(length(authors))")
638 if isempty(pairs)
639 println("PAIR none")
640 else
641 best = first(sort(collect(keys(pairs)); by = pair -> (-pairs[pair], pair...)))
642 println("PAIR $(best[1]) & $(best[2]) $(pairs[best])")
643 end
644end
645function execute!(database::Database, line::String)
646 command = first(split(line, ' '; limit = 2))
647 commands = ("FORMAT", "BIBLIOGRAPHY", "CITE", "KEYGEN", "CHECK", "SEARCH", "EXPORT", "STATS")
648 command in commands || fail("unknown command '$command'")
649 words = split(line, ' '; limit = command in ("CITE", "SEARCH") ? 3 : 0, keepempty = true)
650 arguments = length(words) - 1
651 arity = command in ("BIBLIOGRAPHY", "EXPORT") ? 1 : command in ("CITE", "SEARCH") ? 2 : 0
652 (command == "FORMAT" ? arguments < 2 : arguments != arity) && fail("wrong number of arguments")
653 style = APA
654 if command in ("FORMAT", "BIBLIOGRAPHY", "CITE")
655 haskey(STYLES, words[2]) || fail("unknown style '$(words[2])'")
656 style = STYLES[words[2]]
657 end
658 if command == "FORMAT"
659 numbers = ieee_numbers(database)
660 for key in words[3:end]
661 if haskey(database.lookup, key)
662 index = database.lookup[key]
663 println(reference(database.entries[index], style, numbers[index]))
664 else
665 println("ERROR unknown key '$key'")
666 end
667 end
668 elseif command == "BIBLIOGRAPHY"
669 println(style == MLA ? "Works Cited" : "References")
670 order = style == IEEE ? ieee_order(database) : database.bibliography
671 numbers = ieee_numbers(database)
672 previous = nothing
673 for index in order
674 entry = database.entries[index]
675 repeated = style in (CHICAGO, MLA) && previous !== nothing && entry.primary == previous
676 println(reference(entry, style, numbers[index]; repeated = repeated))
677 previous = entry.primary
678 end
679 elseif command == "CITE"
680 println(cite!(database, style, words[3]))
681 elseif command == "KEYGEN"
682 keygen(database)
683 elseif command == "CHECK"
684 check_entries(database)
685 elseif command == "EXPORT"
686 haskey(database.lookup, words[2]) || fail("unknown key '$(words[2])'")
687 export_entry(database.entries[database.lookup[words[2]]])
688 elseif command == "SEARCH"
689 needle = ascii_lower(words[3])
690 matches = [entry.key for entry in database.entries if haskey(entry.fields, words[2]) && occursin(needle, ascii_lower(plain(field(entry, words[2]))))]
691 println(isempty(matches) ? "MATCHES 0" : "MATCHES $(length(matches)): " * join(matches, " "))
692 else
693 statistics(database)
694 end
695end
