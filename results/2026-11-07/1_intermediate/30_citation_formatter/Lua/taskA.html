9local function split(text, separator)
10 local parts, start = {}, 1
11 while true do
12 local position = text:find(separator, start, true)
13 if not position then break end
14 parts[#parts + 1] = text:sub(start, position - 1)
15 start = position + #separator
16 end
17 parts[#parts + 1] = text:sub(start)
18 return parts
19end

29local function ascii_less(a, b)
30 for i = 1, math.min(#a, #b) do
31 local x, y = a:byte(i), b:byte(i)
32 if x ~= y then return x < y end
33 end
34 return #a < #b
35end

46end

48local function upper(text)

52local function plain(text) return (text:gsub("[{}]", "")) end
53local function letters(text) return lower((text:gsub("[^A-Za-z]", ""))) end
54local function digits(text) return text:match("^[0-9]+$") ~= nil end
55local function is_letter(c) return c:match("^[A-Za-z]$") ~= nil end
56local function trim(text) return (text:gsub("^ +", ""):gsub(" +$", "")) end

58local function split_top(text, separator)
59 local parts, depth, start, i = {}, 0, 1, 1
60 while i <= #text do
61 if depth == 0 and text:sub(i, i + #separator - 1) == separator then
62 parts[#parts + 1] = text:sub(start, i - 1)
63 i = i + #separator
64 start = i
65 else
66 local c = text:sub(i, i)
67 if c == "{" then depth = depth + 1 end
68 if c == "}" then depth = depth - 1 end
69 i = i + 1
70 end
71 end
72 parts[#parts + 1] = text:sub(start)
73 return parts
74end

76local function accent(text, i)
77 if i + 1 > #text or text:sub(i, i) ~= "\\" then return nil end
78 local command = text:sub(i + 1, i + 1)
79 if command == "&" then return {value = "&", length = 2} end
80 local symbol = ("'\"`^~=."):find(command, 1, true) ~= nil
81 if i + 4 <= #text and (symbol or ("cuvH"):find(command, 1, true)) and
82 text:sub(i + 2, i + 2) == "{" and is_letter(text:sub(i + 3, i + 3)) and text:sub(i + 4, i + 4) == "}" then
83 return {value = text:sub(i + 3, i + 3), length = 5}
84 end
85 if symbol and i + 2 <= #text and is_letter(text:sub(i + 2, i + 2)) then
86 return {value = text:sub(i + 2, i + 2), length = 3}
87 end
88end

90local function normalize(text)
91 local result, i = {}, 1
92 while i <= #text do
93 local wrapped = text:sub(i, i) == "{" and accent(text, i + 1) or nil
94 if wrapped and text:sub(i + wrapped.length + 1, i + wrapped.length + 1) == "}" then
95 result[#result + 1] = wrapped.value
96 i = i + wrapped.length + 2
97 else
98 local direct = accent(text, i)
99 if direct then
100 result[#result + 1] = direct.value
101 i = i + direct.length
102 else
103 result[#result + 1] = text:sub(i, i)
104 i = i + 1
105 end
106 end
107 end
108 return trim((table.concat(result):gsub("~", " "):gsub(" +", " ")))
109end

111local function title_case(text, style)
112 if style == "IEEE" then return text end
113 if style == "APA" then
114 local result, depth = {}, 0
115 for i = 1, #text do
116 local c = text:sub(i, i)
117 if c == "{" then depth = depth + 1
118 elseif c == "}" then depth = depth - 1
119 elseif depth == 0 and is_letter(c) then
120 c = (i == 1 or i >= 3 and text:sub(i - 2, i - 1) == ": ") and upper(c) or lower(c)
121 end
122 result[#result + 1] = c
123 end
124 return table.concat(result)
125 end
126 local words = split_top(text, " ")
127 for i, word in ipairs(words) do
128 if word:sub(1, 1) ~= "{" then
129 local make_lower = i > 1 and i < #words and words[i - 1]:sub(-1) ~= ":" and small_words[letters(word)]
130 local result, depth, found_letter = {}, 0, false
131 for j = 1, #word do
132 local c = word:sub(j, j)
133 if c == "{" then depth = depth + 1
134 elseif c == "}" then depth = depth - 1
135 elseif is_letter(c) then
136 if make_lower then
137 if depth == 0 then c = lower(c) end
138 elseif not found_letter then
139 if depth == 0 then c = upper(c) end
140 found_letter = true
141 end
142 end
143 result[#result + 1] = c
144 end
145 words[i] = table.concat(result)
146 end
147 end
148 return table.concat(words, " ")
149end

151local function list_names(names, conjunction, list_b)
152 if #names == 1 then return names[1] end
153 if list_b and #names == 2 then return names[1] .. " " .. conjunction .. " " .. names[2] end
154 return table.concat(names, ", ", 1, #names - 1) .. ", " .. conjunction .. " " .. names[#names]
155end

160function Person.parse(text)
161 local parts = split_top(text, ",")
162 for i, part in ipairs(parts) do parts[i] = trim(part) end
163 local words = split_top(parts[1], " ")
164 local first_lower, last_lower = nil, 0
165 for i = 1, #words - 1 do
166 if words[i]:match("^[a-z]") then
167 first_lower = first_lower or i
168 last_lower = i
169 end
170 end
171 local first, von, last, jr = "", "", "", ""
172 if #parts == 1 then
173 if first_lower then
174 first = table.concat(words, " ", 1, first_lower - 1)
175 von = table.concat(words, " ", first_lower, last_lower)
176 last = table.concat(words, " ", last_lower + 1)
177 else
178 first = table.concat(words, " ", 1, #words - 1)
179 last = words[#words]
180 end
181 else
182 first = parts[#parts]
183 von = table.concat(words, " ", 1, last_lower)
184 last = table.concat(words, " ", last_lower + 1)
185 if #parts == 3 then jr = parts[2] end
186 end
187 return setmetatable({first = first, von = von, last = last, jr = jr}, Person)
188end

233function NameList:authors(style)
234 local values = {}
235 for i, name in ipairs(self.names) do
236 values[i] = name:notation(style == "APA" or style ~= "IEEE" and i == 1, style == "APA" or style == "IEEE")

246 if self.et_al or #values >= 11 then return table.concat(values, ", ", 1, math.min(7, #values)) .. ", et al." end
247 return list_names(values, "and")
248 end
249 if self.et_al or #values >= 3 then return values[1] .. ", et al." end
250 return list_names(values, "and")
251end

279function Entry.less(a, b)
280 local left = {a.sort_last, a:value("year"), a.sort_title, a.key}
281 local right = {b.sort_last, b:value("year"), b.sort_title, b.key}

294function Database:evaluate(expression, line)
295 local depth = 0
296 for c in expression:gmatch(".") do
297 if c == "{" then depth = depth + 1 end
298 if c == "}" then depth = depth - 1 end
299 if depth < 0 then load_error(line, "unbalanced braces") end
300 end
301 if depth ~= 0 then load_error(line, "unbalanced braces") end
302 local values = {}
303 for _, term in ipairs(split(expression, " # ")) do
304 local first = term:sub(1, 1)
305 if first == "{" or first == '"' then values[#values + 1] = term:sub(2, -2)
306 elseif digits(term) then values[#values + 1] = term
307 elseif self.macros[term] ~= nil then values[#values + 1] = self.macros[term]
308 else load_error(line, "unknown macro '" .. term .. "'") end
309 end
310 return table.concat(values)
311end
312
313function Database:read_entry(lines, start, finish, kind, key)
314 local required = required_fields[kind]
315 if not required then load_error(start, "unknown type '" .. kind .. "'") end
316 if self.by_key[key] then load_error(start, "duplicate key") end
317 local fields = {}
318 for i = start + 1, finish - 1 do
319 local separator = lines[i]:find(" = ", 1, true)
320 local name = lines[i]:sub(3, separator - 1)
321 local expression = lines[i]:sub(separator + 3)
322 if expression:sub(-1) == "," then expression = expression:sub(1, -2) end
323 local value = self:evaluate(expression, i)
324 if fields[name] ~= nil then load_error(i, "duplicate field '" .. name .. "'") end
325 fields[name] = normalize(value)
326 end
327 for _, name in ipairs(required) do
328 if fields[name] == nil and not (kind == "book" and name == "author" and fields.editor ~= nil) then
329 load_error(finish, "missing field '" .. name .. "'")
330 end
331 end
332 local entry = Entry.new(kind, key, fields)
333 self.entries[#self.entries + 1], self.by_key[key] = entry, entry
334end

336function Database:assign_suffixes()
337 local done = {}
338 for _, entry in ipairs(self.entries) do
339 if not done[entry] then
340 local group = {}
341 for _, other in ipairs(self.entries) do
342 if entry:primary():same(other:primary()) and entry.fields.year == other.fields.year then group[#group + 1] = other end
343 end
344 table.sort(group, function(a, b)
345 if a.sort_title ~= b.sort_title then return ascii_less(a.sort_title, b.sort_title) end
346 return ascii_less(a.key, b.key)
347 end)
348 for i, other in ipairs(group) do
349 done[other] = true
350 if #group > 1 then other.suffix = string.char(96 + i) end
351 end
352 end
353 end
354end

356function Database.new(lines)
357 local db = setmetatable({entries = {}, by_key = {}, macros = {}, messages = {}}, Database)
358 local month_keys = split("jan feb mar apr may jun jul aug sep oct nov dec", " ")
359 for i, key in ipairs(month_keys) do db.macros[key] = month_names[i] end
360 local i, macro_count, errors = 1, 0, 0
361 while i <= #lines do
362 local text = lines[i]
363 if text ~= "" and text:sub(1, 1) ~= "%" then
364 local name, okay, problem
365 if text:sub(1, 8) == "@string{" then
366 local separator = text:find(" = ", 1, true)
367 name = text:sub(9, separator - 1)
368 okay, problem = pcall(function()
369 local value = db:evaluate(text:sub(separator + 3, -2), i)
370 if db.macros[name] ~= nil then load_error(i, "duplicate macro") end
371 db.macros[name] = value
372 macro_count = macro_count + 1
373 end)
374 else
375 local finish = i + 1
376 while lines[finish] ~= "}" do finish = finish + 1 end
377 local brace = text:find("{", 1, true)
378 name = text:sub(brace + 1, -2)
379 okay, problem = pcall(function() db:read_entry(lines, i, finish, text:sub(2, brace - 1), name) end)
380 i = finish
381 end
382 if not okay then
383 if type(problem) ~= "table" or not problem.load_error then error(problem, 0) end
384 errors = errors + 1
385 db.messages[#db.messages + 1] = "ERROR line " .. problem.line .. ": " .. name .. ": " .. problem.message
386 end
387 end
388 i = i + 1
389 end
390 db:assign_suffixes()
391 db.messages[#db.messages + 1] = "LOADED entries=" .. #db.entries .. " macros=" .. macro_count .. " errors=" .. errors
392 return db
393end

430local function parse_template(text)
431 local i = 1
432 local function read()
433 local parts = {}
434 while i <= #text and text:sub(i, i) ~= ">" do
435 local c = text:sub(i, i)
436 if c == "{" then
437 local finish = text:find("}", i, true)
438 parts[#parts + 1] = {kind = "field", name = text:sub(i + 1, finish - 1)}
439 i = finish + 1
440 elseif c == "<" then
441 i = i + 1
442 parts[#parts + 1] = {kind = "optional", children = read()}
443 i = i + 1
444 else
445 parts[#parts + 1] = {kind = "literal", text = c}
446 i = i + 1
447 end
448 end
449 return parts
450 end
451 return read()
452end

466local function edition(text)
467 if not digits(text) then return text .. " ed." end
468 local number = tonumber(text, 10)
469 local suffix = "th"
470 if number % 100 < 11 or number % 100 > 13 then suffix = ({[1] = "st", [2] = "nd", [3] = "rd"})[number % 10] or "th" end
471 return text .. suffix .. " ed."

492local function reference(entry, style, number, repeated)
493 local values, output, last = placeholder_values(entry, style, repeated), {}, ""
494 local function append(text)
495 if text ~= "" then output[#output + 1], last = text, text:sub(-1) end
496 end
497 local function render(parts)
498 for _, part in ipairs(parts) do
499 if part.kind == "field" then append(values[part.name] or "")
500 elseif part.kind == "optional" then
501 local visible = true
502 for _, child in ipairs(part.children) do
503 if child.kind == "field" and (values[child.name] or "") == "" then visible = false; break end
504 end
505 if visible then render(part.children) end
506 elseif part.text ~= "." or last ~= "." and last ~= "?" and last ~= "!" then append(part.text) end
507 end
508 end
509 render(templates[style][entry.kind == "incollection" and "inproceedings" or entry.kind])
510 return (style == "IEEE" and "[" .. number .. "] " or "") .. table.concat(output)
511end

557 local entries, parts = {}, {}
558 for i, entry in ipairs(citation.entries) do entries[i] = entry end
559 if citation.kind == "citeauthor" or citation.kind == "citeyear" then
560 for i, entry in ipairs(entries) do parts[i] = citation.kind == "citeauthor" and short_name(entry, style, false) or entry:year(style) end
561 return table.concat(parts, citation.kind == "citeauthor" and "; " or ", ")
562 end
563 local parenthetical = citation.kind ~= "citet"
564 if style == "IEEE" then
565 for i, entry in ipairs(entries) do parts[i] = parenthetical and numbers[entry] or short_name(entry, style, false) .. " [" .. numbers[entry] .. "]" end
566 return parenthetical and number_ranges(parts) or table.concat(parts, "; ")
567 elseif style == "MLA" then
568 for i, entry in ipairs(entries) do parts[i] = short_name(entry, style, parenthetical) end
569 local text = table.concat(parts, "; ")
570 return parenthetical and "(" .. text .. ")" or text
571 end
572 if parenthetical then table.sort(entries, Entry.less) end
573 local i = 1
574 while i <= #entries do
575 local name, finish = short_name(entries[i], style, parenthetical), i + 1
576 while finish <= #entries and short_name(entries[finish], style, parenthetical) == name do finish = finish + 1 end
577 local years = {}
578 for j = i, finish - 1 do years[#years + 1] = entries[j]:year(style) end
579 local joined = table.concat(years, ", ")
580 parts[#parts + 1] = parenthetical and name .. (style == "APA" and ", " or " ") .. joined or name .. " (" .. joined .. ")"
581 i = finish
582 end
583 local text = table.concat(parts, "; ")
584 return parenthetical and "(" .. text .. ")" or text
585end

587function Manager:cite(text, style)
588 local citations, start = {}, text:find("\\", 1, true)
589 while start do
590 local brace = text:find("{", start, true)
591 local finish = text:find("}", brace, true)
592 local entries, seen = {}, {}
593 for _, key in ipairs(split(text:sub(brace + 1, finish - 1), ",")) do
594 local entry = self:lookup(key)
595 if not seen[entry] then seen[entry] = true; entries[#entries + 1] = entry end
596 end
597 citations[#citations + 1] = {start = start, finish = finish, kind = text:sub(start + 1, brace - 1), entries = entries}
598 start = text:find("\\", finish + 1, true)
599 end
600 for _, citation in ipairs(citations) do
601 for _, entry in ipairs(citation.entries) do
602 if not self.recorded[entry] then self.recorded[entry] = true; self.record[#self.record + 1] = entry end
603 end
604 end
605 local numbers, output, position = self:ieee_numbers(), {}, 1
606 for _, citation in ipairs(citations) do
607 output[#output + 1] = text:sub(position, citation.start - 1)
608 output[#output + 1] = citation_text(citation, style, numbers)
609 position = citation.finish + 1
610 end
611 output[#output + 1] = text:sub(position)
612 return table.concat(output)
613end

615local function base_key(entry)
616 local result = letters(plain(entry:primary().names[1].last)) .. entry.fields.year:gsub("[^0-9]", "")
617 for _, word in ipairs(split(entry:value("title"), " ")) do
618 local candidate = letters(word)

620 end

624function Manager:keygen()
625 local bases, counts, used, output = {}, {}, {}, {}
626 for i, entry in ipairs(self.database.entries) do
627 local base = base_key(entry)
628 bases[i], counts[base] = base, (counts[base] or 0) + 1
629 end
630 for i, entry in ipairs(self.database.entries) do
631 local base, suffix = bases[i], ""
632 if counts[base] > 1 then
633 suffix = string.char(97 + (used[base] or 0))
634 used[base] = (used[base] or 0) + 1
635 end
636 output[#output + 1] = entry.key .. " -> " .. base .. suffix
637 end
638 return output
639end

648function Manager:check()
649 local output, titles = {}, {}
650 for _, entry in ipairs(self.database.entries) do
651 local warnings = {}
652 if entry.fields.pages ~= nil then
653 local pages, pieces = entry:value("pages"), {}
654 if pages:find("-", 1, true) and not pages:find("--", 1, true) then warnings[#warnings + 1] = "pages use a single hyphen" end
655 for _, piece in ipairs(split(pages, "-")) do if piece ~= "" then pieces[#pieces + 1] = piece end end
656 if #pieces == 2 and digits(pieces[1]) and digits(pieces[2]) and tonumber(pieces[1], 10) > tonumber(pieces[2], 10) then
657 warnings[#warnings + 1] = "pages are reversed"
658 end
659 end
660 local year, title = entry:value("year"), entry:value("title")
661 if #year ~= 4 or not digits(year) then warnings[#warnings + 1] = "year is not 4 digits" end
662 local _, upper_count = title:gsub("[A-Z]", "")
663 if not title:find("[a-z]") and upper_count >= 2 then warnings[#warnings + 1] = "title is all caps" end
664 if entry.kind == "article" and entry.fields.doi == nil then warnings[#warnings + 1] = "article has no doi" end
665 local lowered = lower(title)
666 if titles[lowered] then warnings[#warnings + 1] = "same title as " .. titles[lowered] else titles[lowered] = entry.key end
667 for _, field in ipairs(unknown_fields(entry)) do warnings[#warnings + 1] = "unknown field '" .. field .. "'" end
668 for _, warning in ipairs(warnings) do output[#output + 1] = "WARN " .. entry.key .. ": " .. warning end
669 end
670 output[#output + 1] = "CHECK warnings=" .. #output
671 return output
672end

676 for _, field in ipairs(known_fields) do if entry.fields[field] ~= nil then fields[#fields + 1] = field end end

680 end
681 output[#output + 1] = "}"

685function Manager:stats()
686 local types, years, authors, pair_counts = {}, {}, {}, {}
687 for _, entry in ipairs(self.database.entries) do
688 types[entry.kind] = (types[entry.kind] or 0) + 1
689 local year = entry:value("year")
690 years[year] = (years[year] or 0) + 1
691 if entry.authors then
692 local distinct = {}
693 for _, name in ipairs(entry.authors.names) do
694 local text = name:notation(false, false)
695 distinct[text], authors[text] = true, true
696 end
697 local names = sorted_keys(distinct)
698 for i = 1, #names do
699 for j = i + 1, #names do
700 local a, b = names[i], names[j]
701 pair_counts[a] = pair_counts[a] or {}
702 pair_counts[a][b] = (pair_counts[a][b] or 0) + 1
703 end
704 end
705 end
706 end
707 local output = {"ENTRIES " .. #self.database.entries}
708 for _, kind in ipairs(sorted_keys(types)) do output[#output + 1] = "TYPE " .. kind .. " " .. types[kind] end
709 for _, year in ipairs(sorted_keys(years)) do output[#output + 1] = "YEAR " .. year .. " " .. years[year] end
710 output[#output + 1] = "AUTHORS " .. #sorted_keys(authors)
711 local best
712 for a, seconds in pairs(pair_counts) do
713 for b, count in pairs(seconds) do
714 if not best or count > best.count or count == best.count and
715 (ascii_less(a, best.a) or a == best.a and ascii_less(b, best.b)) then best = {a = a, b = b, count = count} end
716 end
717 end
718 output[#output + 1] = best and "PAIR " .. best.a .. " & " .. best.b .. " " .. best.count or "PAIR none"
719 return output
720end

722function Manager:execute(text)
723 local command = text:match("^[^ ]+")
724 local arity = {FORMAT = 2, BIBLIOGRAPHY = 1, CITE = 2, KEYGEN = 0, CHECK = 0, SEARCH = 2, EXPORT = 1, STATS = 0}
725 local expected = arity[command]
726 if expected == nil then fail("unknown command '" .. command .. "'") end
727 local words
728 if command == "CITE" or command == "SEARCH" then
729 words = {command}
730 local first = text:find(" ", 1, true)
731 if first then
732 local second = text:find(" ", first + 1, true)
733 if second then words[2], words[3] = text:sub(first + 1, second - 1), text:sub(second + 1)
734 else words[2] = text:sub(first + 1) end
735 end
736 else words = split(text, " ") end
737 if command == "FORMAT" and #words - 1 < 2 or command ~= "FORMAT" and #words - 1 ~= expected then fail("wrong number of arguments") end
738 local style = words[2]
739 if (command == "FORMAT" or command == "BIBLIOGRAPHY" or command == "CITE") and templates[style] == nil then
740 fail("unknown style '" .. style .. "'")
741 end
742 local output = {}
743 if command == "FORMAT" then
744 local numbers = self:ieee_numbers()
745 for i = 3, #words do
746 local key = words[i]
747 local entry = self.database.by_key[key]
748 output[#output + 1] = entry and reference(entry, style, numbers[entry], false) or "ERROR unknown key '" .. key .. "'"
749 end
750 elseif command == "BIBLIOGRAPHY" then
751 output[1] = style == "MLA" and "Works Cited" or "References"
752 local numbers, entries = self:ieee_numbers(), self:ieee_order()
753 if style ~= "IEEE" then table.sort(entries, Entry.less) end
754 local previous
755 for _, entry in ipairs(entries) do
756 local repeated = (style == "CHICAGO" or style == "MLA") and previous and previous:same(entry:primary())
757 output[#output + 1] = reference(entry, style, numbers[entry], repeated)
758 previous = entry:primary()
759 end
760 elseif command == "CITE" then return {self:cite(words[3], style)}
761 elseif command == "KEYGEN" then return self:keygen()
762 elseif command == "CHECK" then return self:check()
763 elseif command == "SEARCH" then
764 local matches, needle = {}, lower(words[3])
765 for _, entry in ipairs(self.database.entries) do
766 if entry.fields[words[2]] ~= nil and lower(entry:value(words[2])):find(needle, 1, true) then matches[#matches + 1] = entry.key end
767 end
768 return {#matches == 0 and "MATCHES 0" or "MATCHES " .. #matches .. ": " .. table.concat(matches, " ")}
769 elseif command == "EXPORT" then return export_entry(self:lookup(words[2]))
770 else return self:stats() end
771 return output
772end
