52func ascii<C: Collection>(_ bytes: C) -> String where C.Element == UInt8 {
53 String(decoding: bytes, as: UTF8.self)
54}
55func isLetter(_ byte: UInt8) -> Bool {
56 (65...90).contains(byte) || (97...122).contains(byte)
57}
58func isDigit(_ byte: UInt8) -> Bool {
59 (48...57).contains(byte)
60}
61func allDigits(_ text: String) -> Bool {
62 !text.isEmpty && text.utf8.allSatisfy(isDigit)
63}
64func lowerByte(_ byte: UInt8) -> UInt8 {
65 (65...90).contains(byte) ? byte + 32 : byte
66}
67func upperByte(_ byte: UInt8) -> UInt8 {
68 (97...122).contains(byte) ? byte - 32 : byte
69}
70func lower(_ text: String) -> String {
71 ascii(text.utf8.map(lowerByte))
72}
73func plain(_ text: String) -> String {
74 ascii(text.utf8.filter { $0 != 123 && $0 != 125 })
75}
76func letters(_ text: String) -> String {
77 ascii(text.utf8.filter(isLetter).map(lowerByte))
78}
79func trimmed(_ text: String) -> String {
80 text.trimmingCharacters(in: .whitespaces)
81}

83func splitDepthZero(_ text: String, at separator: String) -> [String] {
84 let bytes = Array(text.utf8), delimiter = Array(separator.utf8)
85 var result: [String] = []
86 var depth = 0, start = 0, position = 0
87 while position < bytes.count {
88 if depth == 0 && position + delimiter.count <= bytes.count &&
89 bytes[position..<(position + delimiter.count)].elementsEqual(delimiter)
90 {
91 result.append(ascii(bytes[start..<position]))
92 position += delimiter.count
93 start = position
94 } else {
95 if bytes[position] == 123 {
96 depth += 1
97 }
98 if bytes[position] == 125 {
99 depth -= 1
100 }
101 position += 1
102 }
103 }
104 result.append(ascii(bytes[start...]))
105 return result
106}

108func accentAt(_ bytes: [UInt8], _ position: Int) -> (character: UInt8, length: Int)? {
109 guard position + 1 < bytes.count, bytes[position] == 92 else {
110 return nil
111 }
112 let command = bytes[position + 1]
113 if command == 38 {
114 return (38, 2)
115 }
116 let symbol = Array("'\"`^~=.".utf8).contains(command)
117 if symbol && position + 2 < bytes.count && isLetter(bytes[position + 2]) {
118 return (bytes[position + 2], 3)
119 }
120 if (symbol || Array("cuvH".utf8).contains(command)) && position + 4 < bytes.count &&
121 bytes[position + 2] == 123 && isLetter(bytes[position + 3]) && bytes[position + 4] == 125
122 {
123 return (bytes[position + 3], 5)
124 }
125 return nil
126}

128func normalize(_ text: String) -> String {
129 let bytes = Array(text.utf8)
130 var result: [UInt8] = [], position = 0
131 while position < bytes.count {
132 if bytes[position] == 123, let enclosed = accentAt(bytes, position + 1),
133 position + enclosed.length + 1 < bytes.count, bytes[position + enclosed.length + 1] == 125
134 {
135 result.append(enclosed.character)
136 position += enclosed.length + 2
137 } else if let accent = accentAt(bytes, position) {
138 result.append(accent.character)
139 position += accent.length
140 } else {
141 result.append(bytes[position] == 126 ? 32 : bytes[position])
142 position += 1
143 }
144 }
145 return ascii(result).components(separatedBy: " ").filter { !$0.isEmpty }.joined(separator: " ")
146}

148func evaluateExpression(_ expression: String, macros: [String: String]) throws -> String {
149 var depth = 0
150 for byte in expression.utf8 {
151 if byte == 123 {
152 depth += 1
153 }
154 if byte == 125 {
155 depth -= 1
156 }
157 if depth < 0 {
158 throw BibliographyError.invalid("unbalanced braces")
159 }
160 }
161 guard depth == 0 else {
162 throw BibliographyError.invalid("unbalanced braces")
163 }
164 var result = ""
165 for term in expression.components(separatedBy: " # ") {
166 if term.hasPrefix("{") || term.hasPrefix("\"") {
167 result += String(term.dropFirst().dropLast())
168 } else if allDigits(term) {
169 result += term
170 } else {
171 guard let value = macros[term] else {
172 throw BibliographyError.invalid("unknown macro '\(term)'")
173 }
174 result += value
175 }
176 }
177 return result
178}

180struct Person: Hashable {
181 let first: String
182 let von: String
183 let last: String
184 let junior: String
185
186 init(_ text: String) {
187 let parts = splitDepthZero(text, at: ",").map(trimmed)
188 let words = splitDepthZero(parts[0], at: " ")
189 let lowercase = words.indices.dropLast().filter { index in
190 guard let byte = words[index].utf8.first else {
191 return false
192 }
193 return (97...122).contains(byte)
194 }
195 if parts.count == 1 {
196 let start = lowercase.first ?? (words.count - 1)
197 let end = lowercase.last.map { $0 + 1 } ?? (words.count - 1)
198 first = words[..<start].joined(separator: " ")
199 von = words[start..<end].joined(separator: " ")
200 last = words[end...].joined(separator: " ")
201 junior = ""
202 } else {
203 let end = lowercase.last.map { $0 + 1 } ?? 0
204 von = words[..<end].joined(separator: " ")
205 last = words[end...].joined(separator: " ")
206 first = parts.last!
207 junior = parts.count == 3 ? parts[1] : ""
208 }
209 }
210
211 var surname: String {
212 plain(von.isEmpty ? last : von + " " + last)
213 }
214 var initials: String {
215 let given = plain(first)
216 if given.isEmpty {
217 return ""
218 }
219 return given.components(separatedBy: " ").map { word in
220 word.components(separatedBy: "-").map { piece in
221 ascii([upperByte(piece.utf8.first(where: isLetter)!)]) + "."
222 }.joined(separator: "-")
223 }.joined(separator: " ")
224 }
225 func format(inverted: Bool, abbreviated: Bool) -> String {
226 let given = abbreviated ? initials : plain(first)
227 var result = surname
228 if !first.isEmpty {
229 result = inverted ? result + ", " + given : given + " " + result
230 }
231 if !junior.isEmpty {
232 result += ", " + plain(junior)
233 }
234 return result
235 }
236}

300final class Database {
301 let entries: [Entry]
302 let byKey: [String: Int]
303 let bibliography: [Int]
304
305 init(lines: [String], output: inout [String]) {
306 var entries: [Entry] = [], byKey: [String: Int] = [:]
307 var macros = Dictionary(uniqueKeysWithValues: monthNames.map { (lower(String($0.prefix(3))), $0) })
308 var errors = 0, position = 0
309 func report(_ line: Int, _ name: String, _ reason: String) {
310 output.append("ERROR line \(line): \(name): \(reason)")
311 errors += 1
312 }
313 while position < lines.count {
314 let line = lines[position]
315 if line.isEmpty || line.hasPrefix("%") {
316 position += 1
317 continue
318 }
319 if line.hasPrefix("@string{") {
320 let separator = line.range(of: " = ")!
321 let name = String(line[..<separator.lowerBound].dropFirst(8))
322 let expression = String(line[separator.upperBound...].dropLast())
323 do {
324 let value = try evaluateExpression(expression, macros: macros)
325 if macros[name] != nil {
326 throw BibliographyError.invalid("duplicate macro")
327 }
328 macros[name] = value
329 } catch BibliographyError.invalid(let reason) { report(position + 1, name, reason) }
330 catch { preconditionFailure("Unexpected database error") }
331 position += 1
332 continue
333 }
334 let opening = line.firstIndex(of: "{")!
335 let type = String(line[..<opening].dropFirst())
336 let key = String(line[line.index(after: opening)...].dropLast())
337 var fields: [String: String] = [:]
338 var failure: (line: Int, reason: String)?
339 if requiredFields[type] == nil {
340 failure = (position + 1, "unknown type '\(type)'")
341 } else if byKey[key] != nil {
342 failure = (position + 1, "duplicate key")
343 }
344 position += 1
345 while lines[position] != "}" {
346 if failure == nil {
347 let fieldLine = lines[position]
348 let separator = fieldLine.range(of: " = ")!
349 let name = String(fieldLine[..<separator.lowerBound].dropFirst(2))
350 var expression = String(fieldLine[separator.upperBound...])
351 if expression.hasSuffix(",") {
352 expression.removeLast()
353 }
354 do {
355 let value = try evaluateExpression(expression, macros: macros)
356 if fields[name] != nil {
357 throw BibliographyError.invalid("duplicate field '\(name)'")
358 }
359 fields[name] = normalize(value)
360 } catch BibliographyError.invalid(let reason) { failure = (position + 1, reason) }
361 catch { preconditionFailure("Unexpected database error") }
362 }
363 position += 1
364 }
365 if failure == nil {
366 for name in requiredFields[type]! where fields[name] == nil {
367 if type == "book" && name == "author" && fields["editor"] != nil {
368 continue
369 }
370 failure = (position + 1, "missing field '\(name)'")
371 break
372 }
373 }
374 if let failure {
375 report(failure.line, key, failure.reason)
376 } else {
377 byKey[key] = entries.count
378 entries.append(Entry(key: key, type: type, fields: fields))
379 }
380 position += 1
381 }
382 output.append("LOADED entries=\(entries.count) macros=\(macros.count - 12) errors=\(errors)")
383 var groups: [YearGroup: [Int]] = [:]
384 for index in entries.indices {
385 let group = YearGroup(names: entries[index].primary, year: entries[index].fields["year"]!)
386 groups[group, default: []].append(index)
387 }
388 for group in groups.values where group.count > 1 {
389 let sorted = group.sorted { a, b in
390 [entries[a].sortTitle, entries[a].key].lexicographicallyPrecedes([entries[b].sortTitle, entries[b].key])
391 }
392 for (suffix, index) in sorted.enumerated() {
393 entries[index].suffix = String(UnicodeScalar(97 + suffix)!)
394 }
395 }
396 self.entries = entries
397 self.byKey = byKey
398 bibliography = entries.indices.sorted { entries[$0].sortKey.lexicographicallyPrecedes(entries[$1].sortKey) }
399 }
400}

412func authorValue(_ entry: Entry, style: Style, repeated: Bool) -> String {
413 let list = entry.primary
414 var result = "---"
415 if !repeated {
416 let names = list.names.enumerated().map { index, person in
417 person.format(inverted: style == .apa || (style != .ieee && index == 0),
418 abbreviated: style == .apa || style == .ieee)
419 }
420 switch style {
421 case .apa:
422 if list.etAl {
423 result = names.joined(separator: ", ") + ", et al."
424 } else if names.count <= 20 {
425 result = listNames(names, conjunction: "&")
426 } else {
427 result = names.prefix(19).joined(separator: ", ") + ", . . . " + names.last!
428 }
429 case .ieee:
430 result = list.etAl || names.count >= 7 ? names[0] + " et al." : listNames(
431 names,
432 conjunction: "and",
433 listB: true
434 )
435 case .chicago:
436 result = list.etAl || names.count >= 11 ? names.prefix(7).joined(separator: ", ") + ", et al." : listNames(
437 names,
438 conjunction: "and"
439 )
440 case .mla:
441 result = list.etAl || names.count >= 3 ? names[0] + ", et al." : listNames(names, conjunction: "and")
442 }
443 }
444 if entry.author == nil {
445 switch style {
446 case .apa:
447 result += list.singular ? " (Ed.)" : " (Eds.)"
448 case .ieee:
449 result += list.singular ? ", Ed." : ", Eds."
450 case .chicago:
451 result += list.singular ? ", ed." : ", eds."
452 case .mla:
453 result += list.singular ? ", editor" : ", editors"
454 }
455 }
456 return result
457}

467func titleValue(_ title: String, style: Style) -> String {
468 if style == .ieee {
469 return plain(title)
470 }
471 if style == .apa {
472 let original = Array(title.utf8)
473 var result = original, depth = 0
474 for index in original.indices {
475 let byte = original[index]
476 if byte == 123 {
477 depth += 1
478 }
479 if byte == 125 {
480 depth -= 1
481 }
482 if depth == 0 && isLetter(byte) {
483 let upper = index == 0 || (index >= 2 && original[index - 2] == 58 && original[index - 1] == 32)
484 result[index] = upper ? upperByte(byte) : lowerByte(byte)
485 }
486 }
487 return plain(ascii(result))
488 }
489 let words = splitDepthZero(title, at: " ")
490 let result = words.enumerated().map { index, word -> String in
491 if word.hasPrefix("{") {
492 return word
493 }
494 let makeLower = index > 0 && index + 1 < words.count && !words[index - 1].hasSuffix(":") && smallWords
495 .contains(letters(word))
496 var bytes = Array(word.utf8), depth = 0, firstLetter = true
497 for i in bytes.indices {
498 let byte = bytes[i]
499 if byte == 123 {
500 depth += 1
501 }
502 if byte == 125 {
503 depth -= 1
504 }
505 if !isLetter(byte) {
506 continue
507 }
508 if depth == 0 {
509 if makeLower {
510 bytes[i] = lowerByte(byte)
511 } else if firstLetter {
512 bytes[i] = upperByte(byte)
513 }
514 }
515 firstLetter = false
516 }
517 return ascii(bytes)
518 }
519 return plain(result.joined(separator: " "))
520}

522func ordinal(_ text: String) -> String {
523 let number = Int(text)!
524 let lastTwo = number % 100
525 let suffix = (11...13).contains(lastTwo) ? "th" : ([1: "st", 2: "nd", 3: "rd"][number % 10] ?? "th")
526 return text + suffix
527}

529func placeholderValues(_ entry: Entry, style: Style, repeated: Bool) -> [String: String] {
530 var values = entry.fields.mapValues(plain)
531 values["author"] = authorValue(entry, style: style, repeated: repeated)
532 values["editor"] = editorValue(entry.editor, style: style)
533 values["eds"] = entry.editor.map { $0.singular ? "Ed." : "Eds." } ?? ""
534 values["title"] = titleValue(entry.fields["title"]!, style: style)
535 values["year"] = entry.yearLabel(style)
536 let pages = entry.value("pages").components(separatedBy: "--").joined(separator: "-")
537 values["pages"] = pages
538 values["pp"] = entry.fields["pages"] == nil ? "" : (pages.contains("-") ? "pp. " : "p. ") + pages
539 let edition = entry.value("edition")
540 values["edition"] = entry.fields["edition"] == nil ? "" : (allDigits(edition) ? ordinal(edition) : edition) + " ed."
541 let month = entry.value("month")
542 values["mon"] = monthAbbreviations[month] ?? month
543 return values
544}

546indirect enum TemplateNode {
547 case literal(String)
548 case value(String)
549 case optional([TemplateNode])
550}
551
552func parseTemplate(_ source: String) -> [TemplateNode] {
553 let bytes = Array(source.utf8)
554 var position = 0
555 func sequence() -> [TemplateNode] {
556 var nodes: [TemplateNode] = []
557 while position < bytes.count && bytes[position] != 62 {
558 if bytes[position] == 60 {
559 position += 1
560 nodes.append(.optional(sequence()))
561 position += 1
562 } else if bytes[position] == 123 {
563 let end = bytes[position...].firstIndex(of: 125)!
564 nodes.append(.value(ascii(bytes[(position + 1)..<end])))
565 position = end + 1
566 } else {
567 let start = position
568 position += 1
569 while position < bytes.count && ![60, 62, 123].contains(bytes[position]) {
570 position += 1
571 }
572 nodes.append(.literal(ascii(bytes[start..<position])))
573 }
574 }
575 return nodes
576 }
577 return sequence()
578}

580func renderTemplate(_ nodes: [TemplateNode], values: [String: String]) -> String {
581 var result = ""
582 func visit(_ nodes: [TemplateNode]) {
583 for node in nodes {
584 switch node {
585 case .value(let name):
586 result += values[name] ?? ""
587 case .optional(let children):
588 let present = children.allSatisfy { child in
589 if case .value(let name) = child {
590 return !(values[name] ?? "").isEmpty
591 }
592 return true
593 }
594 if present {
595 visit(children)
596 }
597 case .literal(let text):
598 for character in text {
599 if character != "." ||
600 (result.last.map { !".?!".contains($0) } ?? true)
601 {
602 result.append(character)
603 }
604 }
605 }
606 }
607 }
608 visit(nodes)
609 return result
610}

673final class Formatter {
674 private let database: Database
675 private var citationRecord: [String] = []
676 private var cited: Set<String> = []
677 private(set) var output: [String]
678
679 init(database: Database, output: [String]) {
680 self.database = database
681 self.output = output
682 }
683 private func ieeeOrder() -> [String] {
684 citationRecord + database.byKey.keys.filter { !cited.contains($0) }.sorted()
685 }
686 private func ieeeNumbers() -> [String: Int] {
687 Dictionary(uniqueKeysWithValues: ieeeOrder().enumerated().map { ($0.element, $0.offset + 1) })
688 }
689 private func reference(_ entry: Entry, style: Style, numbers: [String: Int], repeated: Bool = false) -> String {
690 let type = entry.type == "incollection" ? "inproceedings" : entry.type
691 let text = renderTemplate(
692 templates[style]![type]!,
693 values: placeholderValues(entry, style: style, repeated: repeated)
694 )
695 return (style == .ieee ? "[\(numbers[entry.key]!)] " : "") + text
696 }
697 private func citationText(_ kind: CitationKind, entries: [Entry], style: Style, numbers: [String: Int]) -> String {
698 let parenthetical = kind.parenthetical
699 func name(_ entry: Entry) -> String {
700 shortName(entry, style: style, parenthetical: parenthetical)
701 }
702 if kind == .citeauthor {
703 return entries.map(name).joined(separator: "; ")
704 }
705 if kind == .citeyear {
706 return entries.map { $0.yearLabel(style) }.joined(separator: ", ")
707 }
708 if style == .mla {
709 let text = entries.map(name).joined(separator: "; ")
710 return parenthetical ? "(\(text))" : text
711 }
712 if style == .ieee {
713 if !parenthetical {
714 return entries.map { "\(name($0)) [\(numbers[$0.key]!)]" }.joined(separator: "; ")
715 }
716 let sorted = entries.map { numbers[$0.key]! }.sorted()
717 var pieces: [String] = [], position = 0
718 while position < sorted.count {
719 var end = position + 1
720 while end < sorted.count && sorted[end] == sorted[end - 1] + 1 {
721 end += 1
722 }
723 if end - position >= 3 {
724 pieces.append("[\(sorted[position])]-[\(sorted[end - 1])]")
725 } else {
726 pieces.append(contentsOf: sorted[position..<end].map { "[\($0)]" })
727 }
728 position = end
729 }
730 return pieces.joined(separator: ", ")
731 }
732 let ordered = parenthetical ? entries.sorted { $0.sortKey.lexicographicallyPrecedes($1.sortKey) } : entries
733 var groups: [(name: String, years: [String])] = []
734 for entry in ordered {
735 let current = name(entry)
736 if groups.last?.name != current {
737 groups.append((current, []))
738 }
739 groups[groups.count - 1].years.append(entry.yearLabel(style))
740 }
741 let text = groups.map { group in
742 parenthetical ? group.name + (style == .apa ? ", " : " ") + group.years.joined(separator: ", ")
743 : group.name + " (" + group.years.joined(separator: ", ") + ")"
744 }.joined(separator: "; ")
745 return parenthetical ? "(\(text))" : text
746 }
747 private func cite(style: Style, text: String) {
748 let bytes = Array(text.utf8)
749 var citations: [Citation] = [], position = 0
750 while position < bytes.count {
751 guard let start = bytes[position...].firstIndex(of: 92) else {
752 break
753 }
754 let opening = bytes[start...].firstIndex(of: 123)!
755 let closing = bytes[opening...].firstIndex(of: 125)!
756 let kind = CitationKind(rawValue: ascii(bytes[(start + 1)..<opening]))!
757 var seen: Set<String> = []
758 let keys = ascii(bytes[(opening + 1)..<closing]).components(separatedBy: ",")
759 .filter { seen.insert($0).inserted }
760 citations.append(Citation(start: start, end: closing + 1, kind: kind, keys: keys))
761 position = closing + 1
762 }
763 for citation in citations {
764 for key in citation.keys where database.byKey[key] == nil {
765 output.append("ERROR unknown key '\(key)'")
766 return
767 }
768 }
769 for citation in citations {
770 for key in citation.keys where cited.insert(key).inserted {
771 citationRecord.append(key)
772 }
773 }
774 let numbers = ieeeNumbers()
775 var result = ""
776 position = 0
777 for citation in citations {
778 let entries = citation.keys.map { database.entries[database.byKey[$0]!] }
779 result += ascii(bytes[position..<citation.start])
780 result += citationText(citation.kind, entries: entries, style: style, numbers: numbers)
781 position = citation.end
782 }
783 output.append(result + ascii(bytes[position...]))
784 }
785 private func keygen() {
786 let bases = database.entries.map { entry -> String in
787 let titleWord = entry.value("title").components(separatedBy: " ").map(letters)
788 .first { !$0.isEmpty && !smallWords.contains($0) } ?? ""
789 return letters(entry.primary.names[0].last) + ascii(entry.fields["year"]!.utf8.filter(isDigit)) + titleWord
790 }
791 var counts: [String: Int] = [:], used: [String: Int] = [:]
792 for base in bases {
793 counts[base, default: 0] += 1
794 }
795 for (index, base) in bases.enumerated() {
796 let suffix = counts[base]! > 1 ? String(UnicodeScalar(97 + used[base, default: 0])!) : ""
797 used[base, default: 0] += 1
798 output.append("\(database.entries[index].key) -> \(base)\(suffix)")
799 }
800 }
801 private func check() {
802 var warnings = 0, firstTitles: [String: String] = [:]
803 for entry in database.entries {
804 func warn(_ message: String) {
805 output.append("WARN \(entry.key): \(message)")
806 warnings += 1
807 }
808 let pages = entry.value("pages"), title = entry.value("title")
809 if pages.contains("-") && !pages.contains("--") {
810 warn("pages use a single hyphen")
811 }
812 let parts = pages.components(separatedBy: "-").filter { !$0.isEmpty }
813 if parts.count == 2 && parts
814 .allSatisfy(allDigits) && Int(parts[0])! > Int(parts[1])!
815 {
816 warn("pages are reversed")
817 }
818 let year = entry.value("year")
819 if year.count != 4 || !allDigits(year) {
820 warn("year is not 4 digits")
821 }
822 if !title.utf8.contains(where: { (97...122).contains($0) }) && title.utf8.filter({ (65...90).contains($0) })
823 .count >= 2
824 {
825 warn("title is all caps")
826 }
827 if entry.type == "article" && entry.fields["doi"] == nil {
828 warn("article has no doi")
829 }
830 let titleKey = lower(title)
831 if let previous = firstTitles[titleKey] {
832 warn("same title as \(previous)")
833 } else {
834 firstTitles[titleKey] = entry.key
835 }
836 for name in entry.unknownFields {
837 warn("unknown field '\(name)'")
838 }
839 }
840 output.append("CHECK warnings=\(warnings)")
841 }
842 private func exportEntry(_ key: String) {
843 guard let index = database.byKey[key] else {
844 output.append("ERROR unknown key '\(key)'")
845 return
846 }
847 let entry = database.entries[index]
848 output.append("@\(entry.type){\(key),")
849 let fields = fieldOrder.filter { entry.fields[$0] != nil } + entry.unknownFields
850 for (index, field) in fields.enumerated() {
851 output.append(" \(field) = {\(entry.fields[field]!)}" + (index + 1 < fields.count ? "," : ""))
852 }
853 output.append("}")
854 }
855 private func stats() {
856 var types: [String: Int] = [:], years: [String: Int] = [:], pairs: [AuthorPair: Int] = [:]
857 var authors: Set<String> = []
858 for entry in database.entries {
859 types[entry.type, default: 0] += 1
860 years[entry.value("year"), default: 0] += 1
861 guard let author = entry.author else {
862 continue
863 }
864 let names = Set(author.names.map { $0.format(inverted: false, abbreviated: false) }).sorted()
865 authors.formUnion(names)
866 for first in names.indices {
867 for second in (first + 1)..<names.count {
868 pairs[AuthorPair(a: names[first], b: names[second]), default: 0] += 1
869 }
870 }
871 }
872 output.append("ENTRIES \(database.entries.count)")
873 for type in types.keys.sorted() {
874 output.append("TYPE \(type) \(types[type]!)")
875 }
876 for year in years.keys.sorted() {
877 output.append("YEAR \(year) \(years[year]!)")
878 }
879 output.append("AUTHORS \(authors.count)")
880 let ranked = pairs.sorted { left, right in
881 if left.value != right.value {
882 return left.value > right.value
883 }
884 return [left.key.a, left.key.b].lexicographicallyPrecedes([right.key.a, right.key.b])
885 }
886 if let best = ranked.first {
887 output.append("PAIR \(best.key.a) & \(best.key.b) \(best.value)")
888 } else {
889 output.append("PAIR none")
890 }
891 }

893 func command(_ line: String) {
894 let arity = [
895 "FORMAT": -1,
896 "BIBLIOGRAPHY": 1,
897 "CITE": 2,
898 "KEYGEN": 0,
899 "CHECK": 0,
900 "SEARCH": 2,
901 "EXPORT": 1,
902 "STATS": 0
903 ]
904 let firstSpace = line.firstIndex(of: " ")
905 let name = firstSpace.map { String(line[..<$0]) } ?? line
906 var parts: [String]
907 if (name == "CITE" || name == "SEARCH"), let firstSpace {
908 let start = line.index(after: firstSpace)
909 if let secondSpace = line[start...].firstIndex(of: " ") {
910 parts = [name, String(line[start..<secondSpace]), String(line[line.index(after: secondSpace)...])]
911 } else {
912 parts = [name, String(line[start...])]
913 }
914 } else {
915 parts = line.components(separatedBy: " ")
916 }
917 let args = Array(parts.dropFirst())
918 guard let required = arity[name] else {
919 output.append("ERROR unknown command '\(name)'")
920 return
921 }
922 if name == "FORMAT" ? args.count < 2 : args.count != required {
923 output.append("ERROR wrong number of arguments")
924 return
925 }
926 var style = Style.apa
927 if ["FORMAT", "BIBLIOGRAPHY", "CITE"].contains(name) {
928 guard let parsed = Style(rawValue: args[0]) else {
929 output.append("ERROR unknown style '\(args[0])'")
930 return
931 }
932 style = parsed
933 }
934 switch name {
935 case "FORMAT":
936 let numbers = ieeeNumbers()
937 for key in args.dropFirst() {
938 if let index = database.byKey[key] {
939 output.append(reference(
940 database.entries[index],
941 style: style,
942 numbers: numbers
943 ))
944 } else {
945 output.append("ERROR unknown key '\(key)'")
946 }
947 }
948 case "BIBLIOGRAPHY":
949 let numbers = ieeeNumbers()
950 let order = style == .ieee ? ieeeOrder().map { database.byKey[$0]! } : database.bibliography
951 output.append(style == .mla ? "Works Cited" : "References")
952 var previous: NameList?
953 for index in order {
954 let entry = database.entries[index]
955 let repeated = (style == .chicago || style == .mla) && previous == entry.primary
956 output.append(reference(entry, style: style, numbers: numbers, repeated: repeated))
957 previous = entry.primary
958 }
959 case "CITE":
960 cite(style: style, text: args[1])
961 case "KEYGEN":
962 keygen()
963 case "CHECK":
964 check()
965 case "EXPORT":
966 exportEntry(args[0])
967 case "STATS":
968 stats()
969 case "SEARCH":
970 let needle = lower(args[1])
971 let keys = database.entries
972 .filter { $0.fields[args[0]] != nil && lower($0.value(args[0])).contains(needle) }.map(\.key)
973 output.append(keys.isEmpty ? "MATCHES 0" : "MATCHES \(keys.count): " + keys.joined(separator: " "))
974 default:
975 break
976 }
977 }
978}
