39bool isLetter(char c) { return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z'); }
40bool isDigit(char c) { return c >= '0' && c <= '9'; }
41bool allDigits(const string& s) { return !s.empty() && all_of(s.begin(), s.end(), isDigit); }
42char lowerChar(char c) { return c >= 'A' && c <= 'Z' ? c + ('a' - 'A') : c; }
43char upperChar(char c) { return c >= 'a' && c <= 'z' ? c - ('a' - 'A') : c; }
44string lower(string s) { transform(s.begin(), s.end(), s.begin(), lowerChar); return s; }
45string plain(string s) {
46 s.erase(remove_if(s.begin(), s.end(), [](char c) { return c == '{' || c == '}'; }), s.end());
47 return s;
48}
49string letters(const string& s) {
50 string result;
51 for (char c : s) if (isLetter(c)) result += lowerChar(c);
52 return result;

54string trim(const string& s) {
55 auto begin = s.find_first_not_of(' ');
56 return begin == string::npos ? "" : s.substr(begin, s.find_last_not_of(' ') - begin + 1);
57}
58string join(const vector<string>& parts, const string& separator) {
59 string result;
60 for (size_t i = 0; i < parts.size(); ++i) {
61 if (i) result += separator;
62 result += parts[i];
63 }
64 return result;

66vector<string> split(const string& s, const string& delimiter) {
67 vector<string> result;
68 size_t start = 0, end;
69 while ((end = s.find(delimiter, start)) != string::npos) {
70 result.push_back(s.substr(start, end - start));
71 start = end + delimiter.size();
72 }
73 result.push_back(s.substr(start));
74 return result;
75}
76vector<string> splitDepthZero(const string& s, const string& delimiter) {
77 vector<string> result;
78 int depth = 0;
79 size_t start = 0, i = 0;
80 while (i < s.size()) {
81 if (depth == 0 && s.compare(i, delimiter.size(), delimiter) == 0) {
82 result.push_back(s.substr(start, i - start));
83 i += delimiter.size();
84 start = i;
85 } else {
86 if (s[i] == '{') ++depth;
87 if (s[i] == '}') --depth;
88 ++i;
89 }
90 }
91 result.push_back(s.substr(start));
92 return result;
93}
94string joinRange(const vector<string>& words, size_t begin, size_t end) {
95 return join(vector<string>(words.begin() + begin, words.begin() + end), " ");

98struct Accent { char result; size_t length; };
99optional<Accent> accentAt(const string& s, size_t i) {
100 if (i + 1 >= s.size() || s[i] != '\\') return nullopt;
101 if (s[i + 1] == '&') return Accent{'&', 2};
102 if (i + 2 < s.size() && string("'\"`^~=.").find(s[i + 1]) != string::npos && isLetter(s[i + 2]))
103 return Accent{s[i + 2], 3};
104 if (i + 4 < s.size() && string("'\"`^~=.cuvH").find(s[i + 1]) != string::npos &&
105 s[i + 2] == '{' && isLetter(s[i + 3]) && s[i + 4] == '}') return Accent{s[i + 3], 5};
106 return nullopt;
107}
108string normalize(const string& s) {
109 string accents;
110 for (size_t i = 0; i < s.size();) {
111 auto enclosed = s[i] == '{' ? accentAt(s, i + 1) : nullopt;
112 if (enclosed && i + enclosed->length + 1 < s.size() && s[i + enclosed->length + 1] == '}') {
113 accents += enclosed->result;
114 i += enclosed->length + 2;
115 } else if (auto accent = accentAt(s, i)) {
116 accents += accent->result;
117 i += accent->length;
118 } else accents += s[i++];
119 }
120 string result;
121 bool space = false;
122 for (char c : accents) {
123 if (c == '~' || c == ' ') space = !result.empty();
124 else {
125 if (space) result += ' ';
126 result += c;
127 space = false;
128 }
129 }
130 return result;
131}
132string evaluateExpression(const string& expression, const Fields& macros) {
133 int depth = 0;
134 for (char c : expression) {
135 if (c == '{') ++depth;
136 if (c == '}') --depth;
137 if (depth < 0) throw runtime_error("unbalanced braces");
138 }
139 if (depth != 0) throw runtime_error("unbalanced braces");
140 string result;
141 for (const string& term : split(expression, " # ")) {
142 if (term.front() == '{' || term.front() == '"') result += term.substr(1, term.size() - 2);
143 else if (allDigits(term)) result += term;
144 else {
145 auto found = macros.find(term);
146 if (found == macros.end()) throw runtime_error("unknown macro '" + term + "'");
147 result += found->second;
148 }
149 }
150 return result;
151}

153struct Person {
154 string first, von, last, jr;
155
156 explicit Person(const string& text) {
157 vector<string> parts = splitDepthZero(text, ",");
158 for (auto& part : parts) part = trim(part);
159 vector<string> words = splitDepthZero(parts[0], " ");
160 vector<size_t> lowercase;
161 for (size_t i = 0; i + 1 < words.size(); ++i)
162 if (!words[i].empty() && words[i][0] >= 'a' && words[i][0] <= 'z') lowercase.push_back(i);
163 if (parts.size() == 1) {
164 size_t start = lowercase.empty() ? words.size() - 1 : lowercase.front();
165 size_t end = lowercase.empty() ? words.size() - 1 : lowercase.back() + 1;
166 first = joinRange(words, 0, start);
167 von = joinRange(words, start, end);
168 last = joinRange(words, end, words.size());
169 } else {
170 size_t end = lowercase.empty() ? 0 : lowercase.back() + 1;
171 von = joinRange(words, 0, end);
172 last = joinRange(words, end, words.size());
173 first = parts.back();
174 if (parts.size() == 3) jr = parts[1];
175 }
176 }
177 bool operator==(const Person& other) const {
178 return tie(first, von, last, jr) == tie(other.first, other.von, other.last, other.jr);
179 }
180 string surname() const { return plain(von.empty() ? last : von + " " + last); }
181 string initials() const {
182 string given = plain(first);
183 if (given.empty()) return "";
184 vector<string> words;
185 for (const string& word : split(given, " ")) {
186 vector<string> pieces;
187 for (const string& piece : split(word, "-")) {
188 auto letter = find_if(piece.begin(), piece.end(), isLetter);
189 pieces.push_back(string(1, upperChar(*letter)) + ".");
190 }
191 words.push_back(join(pieces, "-"));
192 }
193 return join(words, " ");
194 }
195 string format(bool inverted, bool useInitials) const {
196 string given = useInitials ? initials() : plain(first), result = surname();
197 if (!first.empty()) result = inverted ? result + ", " + given : given + " " + result;
198 if (!jr.empty()) result += ", " + plain(jr);
199 return result;
200 }

202struct NameList {
203 vector<Person> names;
204 bool etAl;
205 explicit NameList(const string& text) {
206 auto parts = splitDepthZero(text, " and ");
207 etAl = parts.back() == "others";
208 if (etAl) parts.pop_back();
209 for (const string& part : parts) names.emplace_back(part);
210 }
211 bool singular() const { return names.size() == 1 && !etAl; }
212 bool operator==(const NameList& other) const { return etAl == other.etAl && names == other.names; }

214struct Entry {
215 string key, type;
216 Fields fields;
217 optional<NameList> author, editor;
218 string sortTitle, suffix;
219 Entry(string key, string type, Fields fields) : key(move(key)), type(move(type)), fields(move(fields)) {
220 if (this->fields.count("author")) author.emplace(this->fields.at("author"));
221 if (this->fields.count("editor")) editor.emplace(this->fields.at("editor"));
222 sortTitle = lower(value("title"));
223 for (const string& prefix : {"a ", "an ", "the "}) {
224 if (sortTitle.compare(0, prefix.size(), prefix) == 0) { sortTitle.erase(0, prefix.size()); break; }
225 }
226 }
227 string value(const string& name) const {
228 auto found = fields.find(name);
229 return found == fields.end() ? "" : plain(found->second);
230 }
231 const NameList& primary() const { return author ? *author : *editor; }
232 auto sortKey() const { return make_tuple(lower(plain(primary().names[0].last)), value("year"), sortTitle, key); }
233 string yearLabel(Style style) const { return value("year") + (style == Style::APA || style == Style::Chicago ? suffix : ""); }
234 vector<string> unknownFields() const {
235 vector<string> result;
236 for (const auto& [name, value] : fields) if (!knownFields.count(name)) result.push_back(name);
237 return result;
238 }

243public:
244 vector<Entry> entries;

248 Database(const vector<string>& lines, vector<string>& output) {
249 Fields macros;
250 for (const string& month : monthNames) macros[lower(month.substr(0, 3))] = month;
251 int errors = 0;
252 auto report = [&](size_t line, const string& name, const string& reason) {
253 output.push_back("ERROR line " + to_string(line) + ": " + name + ": " + reason);
254 ++errors;
255 };
256 for (size_t i = 0; i < lines.size(); ++i) {
257 const string& line = lines[i];
258 if (line.empty() || line.front() == '%') continue;
259 if (line.compare(0, 8, "@string{") == 0) {
260 size_t equal = line.find(" = ");
261 string name = line.substr(8, equal - 8), expression = line.substr(equal + 3, line.size() - equal - 4);
262 try {
263 string value = evaluateExpression(expression, macros);
264 if (macros.count(name)) throw runtime_error("duplicate macro");
265 macros[name] = move(value);
266 } catch (const runtime_error& error) { report(i + 1, name, error.what()); }
267 continue;
268 }
269 size_t brace = line.find('{');
270 string type = line.substr(1, brace - 1), key = line.substr(brace + 1, line.size() - brace - 2);
271 Fields fields;
272 bool failed = false;
273 auto fail = [&](const string& reason) { report(i + 1, key, reason); failed = true; };
274 if (!requiredFields.count(type)) fail("unknown type '" + type + "'");
275 else if (byKey.count(key)) fail("duplicate key");
276 while (lines[++i] != "}") {
277 if (failed) continue;
278 size_t equal = lines[i].find(" = ");
279 string name = lines[i].substr(2, equal - 2), expression = lines[i].substr(equal + 3);
280 if (expression.back() == ',') expression.pop_back();
281 try {
282 string value = evaluateExpression(expression, macros);
283 if (fields.count(name)) throw runtime_error("duplicate field '" + name + "'");
284 fields[name] = normalize(value);
285 } catch (const runtime_error& error) { fail(error.what()); }
286 }
287 if (failed) continue;
288 for (const string& field : requiredFields.at(type)) {
289 if (!fields.count(field) && !(type == "book" && field == "author" && fields.count("editor"))) {
290 fail("missing field '" + field + "'");
291 break;
292 }
293 }
294 if (!failed) {
295 byKey[key] = entries.size();
296 entries.emplace_back(key, type, move(fields));
297 }
298 }
299 output.push_back("LOADED entries=" + to_string(entries.size()) + " macros=" + to_string(macros.size() - 12) + " errors=" + to_string(errors));
300 vector<bool> assigned(entries.size(), false);
301 for (size_t i = 0; i < entries.size(); ++i) {
302 if (assigned[i]) continue;
303 vector<size_t> group;
304 for (size_t j = i; j < entries.size(); ++j) {
305 if (entries[i].primary() == entries[j].primary() && entries[i].fields.at("year") == entries[j].fields.at("year")) {
306 assigned[j] = true;
307 group.push_back(j);
308 }
309 }
310 if (group.size() < 2) continue;
311 sort(group.begin(), group.end(), [&](size_t a, size_t b) {
312 return tie(entries[a].sortTitle, entries[a].key) < tie(entries[b].sortTitle, entries[b].key);
313 });
314 for (size_t j = 0; j < group.size(); ++j) entries[group[j]].suffix = string(1, char('a' + j));
315 }
316 for (const Entry& entry : entries) bibliography.push_back(&entry);
317 sort(bibliography.begin(), bibliography.end(), bibliographyLess);
318 }
319 const Entry* find(const string& key) const {
320 auto found = byKey.find(key);
321 return found == byKey.end() ? nullptr : &entries[found->second];
322 }

325string listNames(const vector<string>& names, const string& conjunction, bool listB = false) {
326 if (names.size() == 1) return names[0];
327 if (listB && names.size() == 2) return names[0] + " " + conjunction + " " + names[1];
328 return join(vector<string>(names.begin(), names.end() - 1), ", ") + ", " + conjunction + " " + names.back();

330string authorValue(const Entry& entry, Style style, bool repeated) {
331 const NameList& list = entry.primary();
332 string result = "---";
333 if (!repeated) {
334 vector<string> names;
335 for (size_t i = 0; i < list.names.size(); ++i) {
336 bool inverted = style == Style::APA || (style != Style::IEEE && i == 0);
337 names.push_back(list.names[i].format(inverted, style == Style::APA || style == Style::IEEE));
338 }
339 if (style == Style::APA) {
340 if (list.etAl) result = join(names, ", ") + ", et al.";
341 else if (names.size() <= 20) result = listNames(names, "&");
342 else result = join(vector<string>(names.begin(), names.begin() + 19), ", ") + ", . . . " + names.back();
343 } else if (style == Style::IEEE) {
344 result = list.etAl || names.size() >= 7 ? names[0] + " et al." : listNames(names, "and", true);
345 } else if (style == Style::Chicago && (list.etAl || names.size() >= 11)) {
346 names.resize(min<size_t>(7, names.size()));
347 result = join(names, ", ") + ", et al.";
348 } else if (style == Style::MLA && (list.etAl || names.size() >= 3)) result = names[0] + ", et al.";
349 else result = listNames(names, "and");
350 }
351 if (!entry.author) {
352 bool singular = list.singular();
353 switch (style) {
354 case Style::APA: result += singular ? " (Ed.)" : " (Eds.)"; break;
355 case Style::IEEE: result += singular ? ", Ed." : ", Eds."; break;
356 case Style::Chicago: result += singular ? ", ed." : ", eds."; break;
357 case Style::MLA: result += singular ? ", editor" : ", editors"; break;
358 }
359 }
360 return result;
361}
362string editorValue(const optional<NameList>& editor, Style style) {
363 if (!editor) return "";
364 vector<string> names;
365 for (const Person& person : editor->names) names.push_back(person.format(false, style == Style::APA || style == Style::IEEE));
366 return editor->etAl ? names[0] + " et al." : listNames(names, style == Style::APA ? "&" : "and", true);
367}
368string titleValue(const string& title, Style style) {
369 if (style == Style::IEEE) return plain(title);
370 if (style == Style::APA) {
371 string result = title;
372 int depth = 0;
373 for (size_t i = 0; i < title.size(); ++i) {
374 if (title[i] == '{') ++depth;
375 if (title[i] == '}') --depth;
376 if (depth == 0 && isLetter(title[i])) {
377 bool capital = i == 0 || (i >= 2 && title.compare(i - 2, 2, ": ") == 0);
378 result[i] = capital ? upperChar(title[i]) : lowerChar(title[i]);
379 }
380 }
381 return plain(result);
382 }
383 vector<string> words = splitDepthZero(title, " "), result;
384 for (size_t i = 0; i < words.size(); ++i) {
385 string word = words[i];
386 if (!word.empty() && word.front() == '{') { result.push_back(word); continue; }
387 bool makeLower = i > 0 && i + 1 < words.size() && words[i - 1].back() != ':' && smallWords.count(letters(word));
388 int depth = 0;
389 bool firstLetter = true;
390 for (char& c : word) {
391 if (c == '{') ++depth;
392 if (c == '}') --depth;
393 if (!isLetter(c)) continue;
394 if (depth == 0) {
395 if (makeLower) c = lowerChar(c);
396 else if (firstLetter) c = upperChar(c);
397 }
398 firstLetter = false;
399 }
400 result.push_back(word);
401 }
402 return plain(join(result, " "));
403}
404string ordinal(const string& value) {
405 int number = stoi(value), lastTwo = number % 100;
406 string suffix = "th";
407 if (lastTwo < 11 || lastTwo > 13) {
408 if (number % 10 == 1) suffix = "st";
409 if (number % 10 == 2) suffix = "nd";
410 if (number % 10 == 3) suffix = "rd";
411 }
412 return value + suffix;
413}
414Fields placeholderValues(const Entry& entry, Style style, bool repeated) {
415 Fields values;
416 for (const auto& [name, value] : entry.fields) values[name] = plain(value);
417 values["author"] = authorValue(entry, style, repeated);
418 values["editor"] = editorValue(entry.editor, style);
419 values["eds"] = entry.editor ? (entry.editor->singular() ? "Ed." : "Eds.") : "";
420 values["title"] = titleValue(entry.fields.at("title"), style);
421 values["year"] = entry.yearLabel(style);
422 values["pages"] = join(split(entry.value("pages"), "--"), "-");
423 values["pp"] = entry.fields.count("pages") ? (values["pages"].find('-') == string::npos ? "p. " : "pp. ") + values["pages"] : "";
424 string edition = entry.value("edition");
425 values["edition"] = entry.fields.count("edition") ? (allDigits(edition) ? ordinal(edition) : edition) + " ed." : "";
426 string month = entry.value("month");
427 auto found = find(monthNames.begin(), monthNames.end(), month);
428 values["mon"] = found == monthNames.end() ? month : monthAbbreviations[found - monthNames.begin()];
429 return values;
430}

432enum class TemplateKind { Literal, Value, Optional };
433struct TemplateNode {
434 TemplateKind kind;
435 string text;
436 vector<TemplateNode> children;
437};
438vector<TemplateNode> parseTemplate(const string& source) {
439 size_t position = 0;
440 function<vector<TemplateNode>()> sequence = [&]() {
441 vector<TemplateNode> nodes;
442 while (position < source.size() && source[position] != '>') {
443 if (source[position] == '<') {
444 ++position;
445 nodes.push_back({TemplateKind::Optional, "", sequence()});
446 ++position;
447 } else if (source[position] == '{') {
448 size_t end = source.find('}', position);
449 nodes.push_back({TemplateKind::Value, source.substr(position + 1, end - position - 1), {}});
450 position = end + 1;
451 } else {
452 size_t start = position++;
453 while (position < source.size() && string("<{>").find(source[position]) == string::npos) ++position;
454 nodes.push_back({TemplateKind::Literal, source.substr(start, position - start), {}});
455 }
456 }
457 return nodes;
458 };
459 return sequence();
460}
461string renderTemplate(const vector<TemplateNode>& nodes, const Fields& values) {
462 string result;
463 auto value = [&](const string& name) {
464 auto found = values.find(name);
465 return found == values.end() ? "" : found->second;
466 };
467 function<void(const vector<TemplateNode>&)> visit = [&](const vector<TemplateNode>& sequence) {
468 for (const TemplateNode& node : sequence) {
469 if (node.kind == TemplateKind::Value) result += value(node.text);
470 else if (node.kind == TemplateKind::Optional) {
471 bool include = all_of(node.children.begin(), node.children.end(), [&](const TemplateNode& child) {
472 return child.kind != TemplateKind::Value || !value(child.text).empty();
473 });
474 if (include) visit(node.children);
475 } else for (char c : node.text) {
476 if (c != '.' || result.empty() || string(".?!").find(result.back()) == string::npos) result += c;
477 }
478 }
479 };
480 visit(nodes);
481 return result;
482}

518 map<Style, map<string, vector<TemplateNode>>> result;

520 for (const auto& [type, source] : sources) result[style][type] = parseTemplate(source);
521 return result;
522}();
523
524string shortName(const Entry& entry, Style style, bool parenthetical) {
525 const NameList& list = entry.primary();
526 if (list.etAl || list.names.size() >= (style == Style::Chicago ? 4u : 3u)) return list.names[0].surname() + " et al.";
527 vector<string> names;
528 for (const Person& person : list.names) names.push_back(person.surname());
529 return listNames(names, style == Style::APA && parenthetical ? "&" : "and", true);
530}

532class Formatter {
533 const Database& database;
534 vector<string>& output;
535 vector<string> citationRecord;
536 set<string> cited;
537
538 vector<string> ieeeOrder() const {
539 vector<string> result = citationRecord;
540 for (const auto& [key, index] : database.byKey) if (!cited.count(key)) result.push_back(key);
541 return result;
542 }
543 map<string, int> ieeeNumbers() const {
544 map<string, int> result;
545 auto order = ieeeOrder();
546 for (size_t i = 0; i < order.size(); ++i) result[order[i]] = int(i) + 1;
547 return result;
548 }
549 string reference(const Entry& entry, Style style, const map<string, int>& numbers, bool repeated = false) const {
550 string type = entry.type == "incollection" ? "inproceedings" : entry.type;
551 string result = renderTemplate(templates.at(style).at(type), placeholderValues(entry, style, repeated));
552 return (style == Style::IEEE ? "[" + to_string(numbers.at(entry.key)) + "] " : "") + result;
553 }
554 string citationText(const string& kind, vector<const Entry*> entries, Style style, const map<string, int>& numbers) const {
555 bool parenthetical = kind == "cite" || kind == "citep";
556 auto name = [&](const Entry* entry) { return shortName(*entry, style, parenthetical); };
557 vector<string> pieces;
558 if (kind == "citeauthor" || kind == "citeyear") {
559 for (const Entry* entry : entries) pieces.push_back(kind == "citeauthor" ? name(entry) : entry->yearLabel(style));
560 return join(pieces, kind == "citeauthor" ? "; " : ", ");
561 }
562 if (style == Style::MLA) {
563 for (const Entry* entry : entries) pieces.push_back(name(entry));
564 string result = join(pieces, "; ");
565 return parenthetical ? "(" + result + ")" : result;
566 }
567 if (style == Style::IEEE) {
568 if (!parenthetical) {
569 for (const Entry* entry : entries) pieces.push_back(name(entry) + " [" + to_string(numbers.at(entry->key)) + "]");
570 return join(pieces, "; ");
571 }
572 vector<int> sorted;
573 for (const Entry* entry : entries) sorted.push_back(numbers.at(entry->key));
574 sort(sorted.begin(), sorted.end());
575 for (size_t i = 0; i < sorted.size();) {
576 size_t end = i + 1;
577 while (end < sorted.size() && sorted[end] == sorted[end - 1] + 1) ++end;
578 if (end - i >= 3) pieces.push_back("[" + to_string(sorted[i]) + "]-[" + to_string(sorted[end - 1]) + "]");
579 else for (size_t j = i; j < end; ++j) pieces.push_back("[" + to_string(sorted[j]) + "]");
580 i = end;
581 }
582 return join(pieces, ", ");
583 }
584 if (parenthetical) sort(entries.begin(), entries.end(), bibliographyLess);
585 vector<pair<string, vector<string>>> groups;
586 for (const Entry* entry : entries) {
587 string current = name(entry);
588 if (groups.empty() || groups.back().first != current) groups.push_back({current, {}});
589 groups.back().second.push_back(entry->yearLabel(style));
590 }
591 for (const auto& [name, years] : groups) {
592 if (parenthetical) pieces.push_back(name + (style == Style::APA ? ", " : " ") + join(years, ", "));
593 else pieces.push_back(name + " (" + join(years, ", ") + ")");
594 }
595 string result = join(pieces, "; ");
596 return parenthetical ? "(" + result + ")" : result;
597 }
598 void cite(Style style, const string& text) {
599 struct Citation { size_t start, length; string kind; vector<string> keys; };
600 static const regex pattern(R"(\\(citeauthor|citeyear|citep|citet|cite)\{([^}]+)\})");
601 vector<Citation> citations;
602 for (sregex_iterator it(text.begin(), text.end(), pattern), end; it != end; ++it) {
603 Citation citation{size_t(it->position()), size_t(it->length()), (*it)[1].str(), {}};
604 set<string> seen;
605 for (const string& key : split((*it)[2].str(), ",")) if (seen.insert(key).second) citation.keys.push_back(key);
606 citations.push_back(move(citation));
607 }
608 for (const Citation& citation : citations) {
609 for (const string& key : citation.keys) {
610 if (!database.find(key)) { output.push_back("ERROR unknown key '" + key + "'"); return; }
611 }
612 }
613 for (const Citation& citation : citations)
614 for (const string& key : citation.keys) if (cited.insert(key).second) citationRecord.push_back(key);
615 auto numbers = ieeeNumbers();
616 string result;
617 size_t position = 0;
618 for (const Citation& citation : citations) {
619 result += text.substr(position, citation.start - position);
620 vector<const Entry*> entries;
621 for (const string& key : citation.keys) entries.push_back(database.find(key));
622 result += citationText(citation.kind, move(entries), style, numbers);
623 position = citation.start + citation.length;
624 }
625 output.push_back(result + text.substr(position));
626 }
627 void keygen() {
628 vector<string> bases;
629 map<string, size_t> counts, used;
630 for (const Entry& entry : database.entries) {
631 string base = letters(entry.primary().names[0].last);
632 for (char c : entry.fields.at("year")) if (isDigit(c)) base += c;
633 for (const string& word : split(entry.value("title"), " ")) {
634 string part = letters(word);
635 if (!part.empty() && !smallWords.count(part)) { base += part; break; }
636 }
637 bases.push_back(base);
638 ++counts[base];
639 }
640 for (size_t i = 0; i < bases.size(); ++i) {
641 const string& base = bases[i];
642 string suffix = counts[base] > 1 ? string(1, char('a' + used[base])) : "";
643 ++used[base];
644 output.push_back(database.entries[i].key + " -> " + base + suffix);
645 }
646 }
647 void check() {
648 int warnings = 0;
649 map<string, string> firstTitles;
650 for (const Entry& entry : database.entries) {
651 auto warn = [&](const string& message) { output.push_back("WARN " + entry.key + ": " + message); ++warnings; };
652 string pages = entry.value("pages"), title = entry.value("title");
653 if (pages.find('-') != string::npos && pages.find("--") == string::npos) warn("pages use a single hyphen");
654 vector<string> parts;
655 for (const string& part : split(pages, "-")) if (!part.empty()) parts.push_back(part);
656 if (parts.size() == 2 && allDigits(parts[0]) && allDigits(parts[1]) && stoll(parts[0]) > stoll(parts[1])) warn("pages are reversed");
657 string year = entry.value("year");
658 if (year.size() != 4 || !allDigits(year)) warn("year is not 4 digits");
659 bool hasLower = any_of(title.begin(), title.end(), [](char c) { return c >= 'a' && c <= 'z'; });
660 auto uppercase = count_if(title.begin(), title.end(), [](char c) { return c >= 'A' && c <= 'Z'; });
661 if (!hasLower && uppercase >= 2) warn("title is all caps");
662 if (entry.type == "article" && !entry.fields.count("doi")) warn("article has no doi");
663 string titleKey = lower(title);
664 if (firstTitles.count(titleKey)) warn("same title as " + firstTitles.at(titleKey));
665 else firstTitles[titleKey] = entry.key;
666 for (const string& field : entry.unknownFields()) warn("unknown field '" + field + "'");
667 }
668 output.push_back("CHECK warnings=" + to_string(warnings));
669 }
670 void exportEntry(const string& key) {
671 const Entry* entry = database.find(key);
672 if (!entry) { output.push_back("ERROR unknown key '" + key + "'"); return; }
673 output.push_back("@" + entry->type + "{" + key + ",");
674 vector<string> names;
675 for (const string& field : fieldOrder) if (entry->fields.count(field)) names.push_back(field);
676 auto unknown = entry->unknownFields();
677 names.insert(names.end(), unknown.begin(), unknown.end());
678 for (size_t i = 0; i < names.size(); ++i)
679 output.push_back(" " + names[i] + " = {" + entry->fields.at(names[i]) + "}" + (i + 1 < names.size() ? "," : ""));
680 output.push_back("}");
681 }
682 void stats() {
683 map<string, int> types, years;
684 set<string> authors;
685 map<pair<string, string>, int> pairs;
686 for (const Entry& entry : database.entries) {
687 ++types[entry.type];
688 ++years[entry.value("year")];
689 if (!entry.author) continue;
690 set<string> distinct;
691 for (const Person& person : entry.author->names) distinct.insert(person.format(false, false));
692 authors.insert(distinct.begin(), distinct.end());
693 vector<string> names(distinct.begin(), distinct.end());
694 for (size_t i = 0; i < names.size(); ++i)
695 for (size_t j = i + 1; j < names.size(); ++j) ++pairs[{names[i], names[j]}];
696 }
697 output.push_back("ENTRIES " + to_string(database.entries.size()));
698 for (const auto& [type, count] : types) output.push_back("TYPE " + type + " " + to_string(count));
699 for (const auto& [year, count] : years) output.push_back("YEAR " + year + " " + to_string(count));
700 output.push_back("AUTHORS " + to_string(authors.size()));
701 int maximum = 0;
702 pair<string, string> best;
703 for (const auto& [pair, count] : pairs) if (count > maximum) { best = pair; maximum = count; }
704 output.push_back(maximum ? "PAIR " + best.first + " & " + best.second + " " + to_string(maximum) : "PAIR none");
705 }

707public:
708 Formatter(const Database& database, vector<string>& output) : database(database), output(output) {}
709 void command(const string& line) {
710 static const map<string, int> arity = {{"FORMAT", -1}, {"BIBLIOGRAPHY", 1}, {"CITE", 2},
711 {"KEYGEN", 0}, {"CHECK", 0}, {"SEARCH", 2}, {"EXPORT", 1}, {"STATS", 0}};
712 static const map<string, Style> styles = {{"APA", Style::APA}, {"IEEE", Style::IEEE},
713 {"CHICAGO", Style::Chicago}, {"MLA", Style::MLA}};
714 size_t firstSpace = line.find(' ');
715 string command = line.substr(0, firstSpace);
716 vector<string> parts;
717 if (command == "CITE" || command == "SEARCH") {
718 parts.push_back(command);
719 if (firstSpace != string::npos) {
720 size_t secondSpace = line.find(' ', firstSpace + 1);
721 parts.push_back(line.substr(firstSpace + 1, secondSpace == string::npos ? string::npos : secondSpace - firstSpace - 1));
722 if (secondSpace != string::npos) parts.push_back(line.substr(secondSpace + 1));
723 }
724 } else parts = split(line, " ");
725 vector<string> args(parts.begin() + 1, parts.end());
726 if (!arity.count(command)) { output.push_back("ERROR unknown command '" + command + "'"); return; }
727 if (command == "FORMAT" ? args.size() < 2 : args.size() != size_t(arity.at(command))) {
728 output.push_back("ERROR wrong number of arguments"); return;
729 }
730 bool hasStyle = command == "FORMAT" || command == "BIBLIOGRAPHY" || command == "CITE";
731 if (hasStyle && !styles.count(args[0])) { output.push_back("ERROR unknown style '" + args[0] + "'"); return; }
732 Style style = hasStyle ? styles.at(args[0]) : Style::APA;
733 if (command == "FORMAT") {
734 auto numbers = ieeeNumbers();
735 for (size_t i = 1; i < args.size(); ++i) {
736 const Entry* entry = database.find(args[i]);
737 output.push_back(entry ? reference(*entry, style, numbers) : "ERROR unknown key '" + args[i] + "'");
738 }
739 } else if (command == "BIBLIOGRAPHY") {
740 auto numbers = ieeeNumbers();
741 vector<const Entry*> entries = database.bibliography;
742 if (style == Style::IEEE) {
743 entries.clear();
744 for (const string& key : ieeeOrder()) entries.push_back(database.find(key));
745 }
746 output.push_back(style == Style::MLA ? "Works Cited" : "References");
747 const NameList* previous = nullptr;
748 for (const Entry* entry : entries) {
749 bool repeated = (style == Style::Chicago || style == Style::MLA) && previous && *previous == entry->primary();
750 output.push_back(reference(*entry, style, numbers, repeated));
751 previous = &entry->primary();
752 }
753 } else if (command == "CITE") cite(style, args[1]);
754 else if (command == "KEYGEN") keygen();
755 else if (command == "CHECK") check();
756 else if (command == "EXPORT") exportEntry(args[0]);
757 else if (command == "STATS") stats();
758 else if (command == "SEARCH") {
759 vector<string> keys;
760 string needle = lower(args[1]);
761 for (const Entry& entry : database.entries)
762 if (entry.fields.count(args[0]) && lower(entry.value(args[0])).find(needle) != string::npos) keys.push_back(entry.key);
763 output.push_back(keys.empty() ? "MATCHES 0" : "MATCHES " + to_string(keys.size()) + ": " + join(keys, " "));
764 }
765 }
766};
