31depth_before <- function(text) {
32 characters <- chars(text)
33 c(0L, head(cumsum((characters == "{") - (characters == "}")), -1L))
34}
35split_depth <- function(text, separator) {
36 positions <- gregexpr(separator, text, fixed = TRUE)[[1L]]
37 positions <- positions[positions > 0L]
38 positions <- positions[depth_before(text)[positions] == 0L]
39 substring(text, c(1L, positions + nchar(separator)), c(positions - 1L, nchar(text)))
40}
41evaluate <- function(expression, macros) {
42 characters <- chars(expression)
43 depths <- cumsum((characters == "{") - (characters == "}"))
44 if (any(depths < 0L) || tail(depths, 1L) != 0L) fail("unbalanced braces")
45 terms <- split_fixed(expression, " # ")
46 values <- vapply(terms, function(term) {
47 if (substr(term, 1L, 1L) %in% c("{", '"')) return(substr(term, 2L, nchar(term) - 1L))
48 if (all_digits(term)) return(term)
49 value <- get0(term, envir = macros, inherits = FALSE)
50 if (is.null(value)) fail(paste0("unknown macro '", term, "'"))
51 value
52 }, character(1L), USE.NAMES = FALSE)
53 paste0(values, collapse = "")
54}
55accent_at <- function(characters, position) {
56 size <- length(characters)
57 if (position + 1L > size || characters[position] != "\\") return(NULL)
58 command <- characters[position + 1L]
59 if (command == "&") return(list(value = "&", width = 2L))
60 symbol <- command %in% c("'", '"', "`", "^", "~", "=", ".")
61 if (symbol && position + 2L <= size && grepl("^[A-Za-z]$", characters[position + 2L])) {
62 return(list(value = characters[position + 2L], width = 3L))
63 }
64 if ((symbol || command %in% c("c", "u", "v", "H")) && position + 4L <= size &&
65 characters[position + 2L] == "{" && grepl("^[A-Za-z]$", characters[position + 3L]) && characters[position + 4L] == "}") {
66 return(list(value = characters[position + 3L], width = 5L))
67 }
68 NULL
69}
70normalize <- function(text) {
71 characters <- chars(text)
72 output <- character(length(characters))
73 position <- 1L
74 used <- 0L
75 while (position <= length(characters)) {
76 wrapped <- if (characters[position] == "{") accent_at(characters, position + 1L) else NULL
77 if (!is.null(wrapped) && position + wrapped$width + 1L <= length(characters) &&
78 characters[position + wrapped$width + 1L] == "}") {
79 used <- used + 1L
80 output[used] <- wrapped$value
81 position <- position + wrapped$width + 2L
82 next
83 }
84 found <- accent_at(characters, position)
85 used <- used + 1L
86 if (is.null(found)) {
87 output[used] <- characters[position]
88 position <- position + 1L
89 } else {
90 output[used] <- found$value
91 position <- position + found$width
92 }
93 }
94 value <- gsub("~", " ", paste0(output[seq_len(used)], collapse = ""), fixed = TRUE)
95 trim_spaces(gsub(" +", " ", value))
96}

98parse_person <- function(text) {
99 parts <- trim_spaces(split_depth(text, ","))
100 words <- split_depth(parts[1L], " ")
101 lowercase_words <- which(grepl("^[a-z]", head(words, -1L)))
102 person <- list(first = "", von = "", last = "", junior = "")
103 if (length(parts) == 1L) {
104 if (!length(lowercase_words)) {
105 person$first <- paste(head(words, -1L), collapse = " ")
106 person$last <- tail(words, 1L)
107 } else {
108 start <- lowercase_words[1L]
109 finish <- tail(lowercase_words, 1L)
110 person$first <- paste(head(words, start - 1L), collapse = " ")
111 person$von <- paste(words[seq.int(start, finish)], collapse = " ")
112 person$last <- paste(words[seq.int(finish + 1L, length(words))], collapse = " ")
113 }
114 } else {
115 finish <- if (length(lowercase_words)) tail(lowercase_words, 1L) else 0L
116 person$von <- paste(head(words, finish), collapse = " ")
117 person$last <- paste(words[seq.int(finish + 1L, length(words))], collapse = " ")
118 person$first <- tail(parts, 1L)
119 if (length(parts) == 3L) person$junior <- parts[2L]
120 }
121 person
122}

166read_database <- function(lines) {
167 database <- new.env(parent = emptyenv())
168 database$entries <- list()
169 database$keys <- character()
170 database$record <- integer()
171 macros <- new.env(hash = TRUE, parent = emptyenv())
172 for (i in seq_along(month_names)) assign(substr(lowercase(month_names[i]), 1L, 3L), month_names[i], envir = macros)
173 errors <- 0L
174 position <- 1L
175 while (position <= length(lines)) {
176 line <- lines[position]
177 if (!nzchar(line) || startsWith(line, "%")) {
178 position <- position + 1L
179 next
180 }
181 reason <- ""
182 error_line <- position
183 if (startsWith(line, "@string{")) {
184 separator <- regexpr(" = ", line, fixed = TRUE)[1L]
185 name <- substr(line, 9L, separator - 1L)
186 expression <- substr(line, separator + 3L, nchar(line) - 1L)
187 tryCatch({
188 value <- evaluate(expression, macros)
189 if (exists(name, envir = macros, inherits = FALSE)) fail("duplicate macro")
190 assign(name, value, envir = macros)
191 }, bib_error = function(problem) reason <<- conditionMessage(problem))
192 } else {
193 opening <- regexpr("{", line, fixed = TRUE)[1L]
194 kind <- substr(line, 2L, opening - 1L)
195 name <- substr(line, opening + 1L, nchar(line) - 1L)
196 fields <- list()
197 if (!(kind %in% names(required_fields))) reason <- paste0("unknown type '", kind, "'") else
198 if (name %in% database$keys) reason <- "duplicate key"
199 position <- position + 1L
200 while (lines[position] != "}") {
201 if (!nzchar(reason)) {
202 field_line <- lines[position]
203 separator <- regexpr(" = ", field_line, fixed = TRUE)[1L]
204 field_name <- substr(field_line, 3L, separator - 1L)
205 expression <- substr(field_line, separator + 3L, nchar(field_line) - as.integer(endsWith(field_line, ",")))
206 tryCatch({
207 value <- evaluate(expression, macros)
208 if (field_name %in% names(fields)) fail(paste0("duplicate field '", field_name, "'"))
209 fields[[field_name]] <- normalize(value)
210 }, bib_error = function(problem) {
211 reason <<- conditionMessage(problem)
212 error_line <<- position
213 })
214 }
215 position <- position + 1L
216 }
217 if (!nzchar(reason)) {
218 for (required in required_fields[[kind]]) {
219 if (required %in% names(fields) || (kind == "book" && required == "author" && "editor" %in% names(fields))) next
220 reason <- paste0("missing field '", required, "'")
221 error_line <- position
222 break
223 }
224 }
225 if (!nzchar(reason)) {
226 database$keys <- c(database$keys, name)
227 database$entries[[length(database$entries) + 1L]] <- make_entry(name, kind, fields)
228 }
229 }
230 if (nzchar(reason)) {
231 errors <- errors + 1L
232 emit(paste0("ERROR line ", error_line, ": ", name, ": ", reason))
233 }
234 position <- position + 1L
235 }
236 emit(paste0("LOADED entries=", length(database$entries), " macros=", length(ls(macros)) - 12L, " errors=", errors))
237 database$bibliography <- ordered_entries(database$entries)
238 assigned <- rep(FALSE, length(database$entries))
239 for (i in seq_along(database$entries)) {
240 if (assigned[i]) next
241 entry <- database$entries[[i]]
242 group <- which(vapply(database$entries, function(other) identical(entry$primary, other$primary) &&
243 identical(field(entry, "year"), field(other, "year")), logical(1L)))
244 assigned[group] <- TRUE
245 if (length(group) >= 2L) {
246 group <- ordered_entries(database$entries, group, TRUE)
247 for (j in seq_along(group)) database$entries[[group[j]]]$suffix <- substr(lower_letters, j, j)
248 }
249 }
250 database
251}

277convert_outside <- function(text, mode) {
278 characters <- chars(text)
279 depths <- depth_before(text)
280 letters <- which(grepl("^[A-Za-z]$", characters))
281 outside <- letters[depths[letters] == 0L]
282 if (mode %in% c(0L, 1L)) characters[outside] <- lowercase(characters[outside])
283 if (mode == 1L) {
284 colons <- gregexpr(": ", text, fixed = TRUE)[[1L]]
285 capitals <- intersect(outside, c(1L, colons[colons > 0L] + 2L))
286 characters[capitals] <- uppercase(characters[capitals])
287 } else if (mode == 2L && length(letters) && depths[letters[1L]] == 0L) {
288 characters[letters[1L]] <- uppercase(characters[letters[1L]])
289 }
290 paste0(characters, collapse = "")
291}
292title_text <- function(title, style) {
293 if (style == "IEEE") return(plain(title))
294 if (style == "APA") return(plain(convert_outside(title, 1L)))
295 words <- split_depth(title, " ")
296 converted <- vapply(seq_along(words), function(i) {
297 word <- words[i]
298 if (startsWith(word, "{")) return(word)
299 small <- i > 1L && i < length(words) && !endsWith(words[i - 1L], ":") && letters_only(word) %in% small_words
300 convert_outside(word, if (small) 0L else 2L)
301 }, character(1L))
302 plain(paste(converted, collapse = " "))
303}

339parse_template <- function(source) {
340 characters <- chars(source)
341 position <- 1L
342 parse <- function() {
343 nodes <- list()
344 while (position <= length(characters) && characters[position] != ">") {
345 character <- characters[position]
346 position <<- position + 1L
347 if (character == "{") {
348 start <- position
349 while (characters[position] != "}") position <<- position + 1L
350 node <- list(kind = "value", text = substr(source, start, position - 1L))
351 position <<- position + 1L
352 } else if (character == "<") {
353 node <- list(kind = "optional", children = parse())
354 position <<- position + 1L
355 } else {
356 node <- list(kind = "literal", text = character)
357 }
358 nodes[[length(nodes) + 1L]] <- node
359 }
360 nodes
361 }
362 parse()
363}
364templates <- lapply(template_sources, function(style) lapply(style, parse_template))
365render_template <- function(nodes, values) {
366 pieces <- character()
367 last <- ""
368 append <- function(text) {
369 if (!nzchar(text)) return(invisible(NULL))
370 pieces <<- c(pieces, text)
371 last <<- substr(text, nchar(text), nchar(text))
372 }
373 render <- function(nodes) {
374 for (node in nodes) {
375 if (node$kind == "value") {
376 append(values[[node$text]])
377 } else if (node$kind == "optional") {
378 direct <- Filter(function(child) child$kind == "value", node$children)
379 if (all(vapply(direct, function(child) nzchar(values[[child$text]]), logical(1L)))) render(node$children)
380 } else if (node$text != "." || !(last %in% c(".", "?", "!"))) {
381 append(node$text)
382 }
383 }
384 }
385 render(nodes)
386 paste0(pieces, collapse = "")
387}

439citation_text <- function(database, citation, style, numbers) {
440 indices <- citation$indices
441 kind <- citation$kind
442 parenthetical <- kind %in% c("cite", "citep")
443 if (parenthetical && style %in% c("APA", "CHICAGO")) indices <- ordered_entries(database$entries, indices)
444 entries <- database$entries[indices]
445 names <- vapply(entries, short_name, character(1L), style = style, parenthetical = parenthetical)
446 years <- vapply(entries, year_label, character(1L), style = style)
447 if (kind == "citeauthor") return(paste(names, collapse = "; "))
448 if (kind == "citeyear") return(paste(years, collapse = ", "))
449 if (style == "MLA") {
450 result <- paste(names, collapse = "; ")
451 return(if (parenthetical) paste0("(", result, ")") else result)
452 }
453 if (style == "IEEE") {
454 if (!parenthetical) return(paste0(names, " [", numbers[indices], "]", collapse = "; "))
455 ordered <- sort(numbers[indices])
456 ranges <- split(ordered, cumsum(c(TRUE, diff(ordered) != 1L)))
457 parts <- lapply(ranges, function(range) if (length(range) >= 3L) paste0("[", range[1L], "]-[", tail(range, 1L), "]") else paste0("[", range, "]"))
458 return(paste(unlist(parts, use.names = FALSE), collapse = ", "))
459 }
460 lengths <- rle(names)$lengths
461 finishes <- cumsum(lengths)
462 starts <- c(1L, head(finishes, -1L) + 1L)
463 groups <- vapply(seq_along(starts), function(i) {
464 years_text <- paste(years[seq.int(starts[i], finishes[i])], collapse = ", ")
465 if (!parenthetical) paste0(names[starts[i]], " (", years_text, ")") else
466 paste0(names[starts[i]], if (style == "APA") ", " else " ", years_text)
467 }, character(1L))
468 result <- paste(groups, collapse = "; ")
469 if (parenthetical) paste0("(", result, ")") else result
470}
471cite <- function(database, style, text) {
472 starts <- gregexpr("\\", text, fixed = TRUE)[[1L]]
473 starts <- starts[starts > 0L]
474 citations <- lapply(starts, function(start) {
475 opening <- start - 1L + regexpr("{", substring(text, start), fixed = TRUE)[1L]
476 finish <- opening - 1L + regexpr("}", substring(text, opening), fixed = TRUE)[1L]
477 keys <- unique(split_fixed(substr(text, opening + 1L, finish - 1L), ","))
478 indices <- match(keys, database$keys)
479 missing <- which(is.na(indices))
480 if (length(missing)) fail(paste0("unknown key '", keys[missing[1L]], "'"))
481 list(start = start, finish = finish, kind = substr(text, start + 1L, opening - 1L), indices = indices)
482 })
483 database$record <- unique(c(database$record, unlist(lapply(citations, `[[`, "indices"), use.names = FALSE)))
484 numbers <- ieee_numbers(database)
485 pieces <- character()
486 position <- 1L
487 for (citation in citations) {
488 pieces <- c(pieces, substr(text, position, citation$start - 1L), citation_text(database, citation, style, numbers))
489 position <- citation$finish + 1L
490 }
491 paste0(c(pieces, substring(text, position)), collapse = "")
492}

494keygen <- function(database) {
495 bases <- vapply(database$entries, function(entry) {
496 words <- letters_only(split_fixed(plain(field(entry, "title")), " "))
497 words <- words[nzchar(words) & !(words %in% small_words)]
498 paste0(letters_only(plain(entry$primary$names[[1L]]$last)), gsub("[^0-9]", "", field(entry, "year")),
499 if (length(words)) words[1L] else "")
500 }, character(1L))
501 groups <- match(bases, unique(bases))
502 totals <- tabulate(groups)
503 seen <- integer(length(totals))
504 for (i in seq_along(bases)) {
505 group <- groups[i]
506 suffix <- if (totals[group] >= 2L) intToUtf8(97L + seen[group]) else ""
507 seen[group] <- seen[group] + 1L
508 emit(paste0(database$keys[i], " -> ", bases[i], suffix))
509 }
510}
511check_entries <- function(database) {
512 warnings <- 0L
513 titles <- character()
514 for (i in seq_along(database$entries)) {
515 entry <- database$entries[[i]]
516 messages <- character()
517 pages <- plain(field(entry, "pages"))
518 if (has_field(entry, "pages")) {
519 if (grepl("-", pages, fixed = TRUE) && !grepl("--", pages, fixed = TRUE)) messages <- c(messages, "pages use a single hyphen")
520 parts <- split_fixed(pages, "-")
521 parts <- parts[nzchar(parts)]
522 if (length(parts) == 2L && all(all_digits(parts)) && as.integer(parts[1L]) > as.integer(parts[2L])) messages <- c(messages, "pages are reversed")
523 }
524 if (!grepl("^[0-9]{4}$", entry$year)) messages <- c(messages, "year is not 4 digits")
525 title <- plain(field(entry, "title"))
526 if (!grepl("[a-z]", title) && sum(grepl("^[A-Z]$", chars(title))) >= 2L) messages <- c(messages, "title is all caps")
527 if (entry$kind == "article" && !has_field(entry, "doi")) messages <- c(messages, "article has no doi")
528 previous <- match(lowercase(title), titles)
529 if (!is.na(previous)) messages <- c(messages, paste0("same title as ", database$keys[previous]))
530 titles <- c(titles, lowercase(title))
531 unknown <- sort(setdiff(names(entry$fields), known_fields), method = "radix")
532 if (length(unknown)) messages <- c(messages, paste0("unknown field '", unknown, "'"))
533 for (message in messages) emit(paste0("WARN ", entry$key, ": ", message))
534 warnings <- warnings + length(messages)
535 }
536 emit(paste0("CHECK warnings=", warnings))
537}
538export_entry <- function(entry) {
539 fields <- c(known_fields[known_fields %in% names(entry$fields)], sort(setdiff(names(entry$fields), known_fields), method = "radix"))
540 emit(paste0("@", entry$kind, "{", entry$key, ","))
541 for (i in seq_along(fields)) emit(paste0(" ", fields[i], " = {", field(entry, fields[i]), "}", if (i < length(fields)) "," else ""))
542 emit("}")
543}
544statistics <- function(database) {
545 entries <- database$entries
546 emit(paste0("ENTRIES ", length(entries)))
547 for (property in c("kind", "year")) {
548 values <- vapply(entries, `[[`, character(1L), property)
549 for (value in sort(unique(values), method = "radix")) emit(paste(if (property == "kind") "TYPE" else "YEAR", value, sum(values == value)))
550 }
551 authors <- character()
552 pairs <- list()
553 for (entry in entries) {
554 if (!has_field(entry, "author")) next
555 names <- sort(unique(vapply(entry$primary$names, name_text, character(1L), inverted = FALSE, abbreviated = FALSE)), method = "radix")
556 authors <- union(authors, names)
557 if (length(names) >= 2L) {
558 for (i in seq_len(length(names) - 1L)) {
559 for (j in seq.int(i + 1L, length(names))) pairs[[length(pairs) + 1L]] <- c(names[i], names[j])
560 }
561 }
562 }
563 emit(paste0("AUTHORS ", length(authors)))
564 if (!length(pairs)) {
565 emit("PAIR none")
566 return(invisible(NULL))
567 }
568 a <- vapply(pairs, `[`, character(1L), 1L)
569 b <- vapply(pairs, `[`, character(1L), 2L)
570 keys <- paste(a, b, sep = "\t")
571 counts <- tabulate(match(keys, unique(keys)))
572 first <- !duplicated(keys)
573 choice <- order(-counts, a[first], b[first], method = "radix")[1L]
574 emit(paste0("PAIR ", a[first][choice], " & ", b[first][choice], " ", counts[choice]))
575}

577execute <- function(database, line) {
578 words <- split_fixed(line, " ")
579 command <- words[1L]
580 if (!(command %in% c("FORMAT", "BIBLIOGRAPHY", "CITE", "KEYGEN", "CHECK", "SEARCH", "EXPORT", "STATS"))) fail(paste0("unknown command '", command, "'"))
581 if (command %in% c("CITE", "SEARCH")) {
582 spaces <- gregexpr(" ", line, fixed = TRUE)[[1L]]
583 spaces <- head(spaces[spaces > 0L], 2L)
584 words <- substring(line, c(1L, spaces + 1L), c(spaces - 1L, nchar(line)))
585 }
586 count <- length(words) - 1L
587 expected <- switch(command, BIBLIOGRAPHY = 1L, CITE = 2L, SEARCH = 2L, EXPORT = 1L, 0L)
588 if (if (command == "FORMAT") count < 2L else count != expected) fail("wrong number of arguments")
589 if (command %in% c("FORMAT", "BIBLIOGRAPHY", "CITE") && !(words[2L] %in% styles)) fail(paste0("unknown style '", words[2L], "'"))
590 if (command == "FORMAT") {
591 numbers <- ieee_numbers(database)
592 for (key in words[-c(1L, 2L)]) {
593 index <- match(key, database$keys)
594 if (is.na(index)) emit(paste0("ERROR unknown key '", key, "'")) else emit(reference(database$entries[[index]], words[2L], numbers[index]))
595 }
596 } else if (command == "BIBLIOGRAPHY") {
597 style <- words[2L]
598 emit(if (style == "MLA") "Works Cited" else "References")
599 order <- if (style == "IEEE") ieee_order(database) else database$bibliography
600 numbers <- ieee_numbers(database)
601 previous <- NULL
602 for (index in order) {
603 entry <- database$entries[[index]]
604 repeated <- style %in% c("CHICAGO", "MLA") && identical(previous, entry$primary)
605 emit(reference(entry, style, numbers[index], repeated))
606 previous <- entry$primary
607 }
608 } else if (command == "CITE") {
609 emit(cite(database, words[2L], words[3L]))
610 } else if (command == "KEYGEN") {
611 keygen(database)
612 } else if (command == "CHECK") {
613 check_entries(database)
614 } else if (command == "SEARCH") {
615 query <- lowercase(words[3L])
616 matches <- which(vapply(database$entries, function(entry) has_field(entry, words[2L]) && grepl(query, lowercase(plain(field(entry, words[2L]))), fixed = TRUE), logical(1L)))
617 emit(if (length(matches)) paste0("MATCHES ", length(matches), ": ", paste(database$keys[matches], collapse = " ")) else "MATCHES 0")
618 } else if (command == "EXPORT") {
619 index <- match(words[2L], database$keys)
620 if (is.na(index)) fail(paste0("unknown key '", words[2L], "'"))
621 export_entry(database$entries[[index]])
622 } else {
623 statistics(database)
624 }
625 invisible(NULL)
626}
