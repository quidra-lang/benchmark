17function isLetter(string $c): bool { return isLower($c) || isUpper($c); }
18function isLower(string $c): bool { return ord($c) >= 97 && ord($c) <= 122; }
19function isUpper(string $c): bool { return ord($c) >= 65 && ord($c) <= 90; }
20function isDigit(string $c): bool { return ord($c) >= 48 && ord($c) <= 57; }
21function digits(string $text): bool { return $text !== '' && strspn($text, '0123456789') === strlen($text); }
22function lower(string $text): string { return strtr($text, 'ABCDEFGHIJKLMNOPQRSTUVWXYZ', 'abcdefghijklmnopqrstuvwxyz'); }
23function upper(string $text): string { return strtr($text, 'abcdefghijklmnopqrstuvwxyz', 'ABCDEFGHIJKLMNOPQRSTUVWXYZ'); }
24function plain(string $text): string { return str_replace(['{', '}'], '', $text); }
25function letters(string $text): string {
26 $result = '';
27 for ($i = 0; $i < strlen($text); $i++) if (isLetter($text[$i])) $result .= lower($text[$i]);
28 return $result;
29}
30function splitTop(string $text, string $separator): array {
31 $parts = [];
32 $depth = 0; $start = 0; $i = 0;
33 while ($i < strlen($text)) {
34 if ($depth === 0 && substr($text, $i, strlen($separator)) === $separator) {
35 $parts[] = substr($text, $start, $i - $start);
36 $i += strlen($separator);
37 $start = $i;
38 continue;
39 }
40 if ($text[$i] === '{') $depth++;
41 if ($text[$i] === '}') $depth--;
42 $i++;
43 }
44 $parts[] = substr($text, $start);
45 return $parts;
46}
47function accent(string $text, int $i): ?array {
48 if ($i + 1 >= strlen($text) || $text[$i] !== '\\') return null;
49 $command = $text[$i + 1];
50 if ($command === '&') return ['&', 2];
51 $symbol = str_contains("'\"`^~=.", $command);
52 if ($i + 4 < strlen($text) && ($symbol || str_contains('cuvH', $command)) &&
53 $text[$i + 2] === '{' && isLetter($text[$i + 3]) && $text[$i + 4] === '}') return [$text[$i + 3], 5];
54 return $symbol && $i + 2 < strlen($text) && isLetter($text[$i + 2]) ? [$text[$i + 2], 3] : null;
55}
56function normalize(string $text): string {
57 $result = '';
58 for ($i = 0; $i < strlen($text);) {
59 $wrapped = $text[$i] === '{' ? accent($text, $i + 1) : null;
60 if ($wrapped !== null && $i + $wrapped[1] + 1 < strlen($text) && $text[$i + $wrapped[1] + 1] === '}') {
61 $result .= $wrapped[0];
62 $i += $wrapped[1] + 2;
63 } else {
64 $direct = accent($text, $i);
65 if ($direct === null) $result .= $text[$i++];
66 else { $result .= $direct[0]; $i += $direct[1]; }
67 }
68 }
69 return trim(preg_replace('/ +/', ' ', str_replace('~', ' ', $result)), ' ');
70}
71function titleCase(string $text, Style $style): string {
72 if ($style === Style::IEEE) return $text;
73 if ($style === Style::APA) {
74 $result = $text; $depth = 0;
75 for ($i = 0; $i < strlen($text); $i++) {
76 if ($text[$i] === '{') $depth++;
77 elseif ($text[$i] === '}') $depth--;
78 elseif ($depth === 0 && isLetter($text[$i])) {
79 $result[$i] = $i === 0 || $i >= 2 && substr($text, $i - 2, 2) === ': ' ? upper($text[$i]) : lower($text[$i]);
80 }
81 }
82 return $result;
83 }
84 $words = splitTop($text, ' ');
85 foreach ($words as $i => $word) {
86 if (str_starts_with($word, '{')) continue;
87 $makeLower = $i > 0 && $i + 1 < count($words) && !str_ends_with($words[$i - 1], ':') && in_array(letters($word), SMALL_WORDS, true);
88 $depth = 0;
89 for ($j = 0; $j < strlen($word); $j++) {
90 if ($word[$j] === '{') $depth++;
91 elseif ($word[$j] === '}') $depth--;
92 elseif (isLetter($word[$j])) {
93 if ($makeLower) {
94 if ($depth === 0) $word[$j] = lower($word[$j]);
95 } else {
96 if ($depth === 0) $word[$j] = upper($word[$j]);
97 break;
98 }
99 }
100 }
101 $words[$i] = $word;
102 }
103 return implode(' ', $words);
104}
105function listNames(array $names, string $conjunction, bool $listB = false): string {
106 if (count($names) === 1) return $names[0];
107 if ($listB && count($names) === 2) return $names[0] . ' ' . $conjunction . ' ' . $names[1];
108 return implode(', ', array_slice($names, 0, -1)) . ', ' . $conjunction . ' ' . $names[count($names) - 1];
109}

111final readonly class Person {
112 public function __construct(public string $first, public string $von, public string $last, public string $jr) {}
113
114 public static function parse(string $text): self {
115 $parts = array_map(static fn(string $part): string => trim($part, ' '), splitTop($text, ','));
116 $words = splitTop($parts[0], ' ');
117 $firstLower = -1; $lastLower = -1;
118 for ($i = 0; $i + 1 < count($words); $i++) {
119 if (isLower($words[$i][0])) {
120 if ($firstLower < 0) $firstLower = $i;
121 $lastLower = $i;
122 }
123 }
124 $join = static fn(int $start, int $end): string => implode(' ', array_slice($words, $start, $end - $start));
125 if (count($parts) === 1) {
126 return $firstLower < 0 ? new self($join(0, count($words) - 1), '', $words[count($words) - 1], '')
127 : new self($join(0, $firstLower), $join($firstLower, $lastLower + 1), $join($lastLower + 1, count($words)), '');
128 }
129 return new self($parts[count($parts) - 1], $join(0, $lastLower + 1), $join($lastLower + 1, count($words)), count($parts) === 3 ? $parts[1] : '');
130 }
131 public function same(self $other): bool {
132 return $this->first === $other->first && $this->von === $other->von && $this->last === $other->last && $this->jr === $other->jr;
133 }
134 public function surname(): string { return plain($this->von === '' ? $this->last : $this->von . ' ' . $this->last); }
135 public function initials(): string {
136 $words = [];
137 foreach (explode(' ', plain($this->first)) as $word) {
138 if ($word === '') continue;
139 $pieces = [];
140 foreach (explode('-', $word) as $piece) {
141 for ($i = 0; $i < strlen($piece); $i++) {
142 if (isLetter($piece[$i])) { $pieces[] = upper($piece[$i]) . '.'; break; }
143 }
144 }
145 $words[] = implode('-', $pieces);
146 }
147 return implode(' ', $words);
148 }
149 public function notation(bool $inverted, bool $initials): string {
150 $given = $initials ? $this->initials() : plain($this->first);
151 $result = $this->surname();
152 if ($this->first !== '') $result = $inverted ? $result . ', ' . $given : $given . ' ' . $result;
153 if ($this->jr !== '') $result .= ', ' . plain($this->jr);
154 return $result;
155 }

157final readonly class NameList {
158 public function __construct(public array $names, public bool $etAl) {}
159 public static function parse(string $text): self {
160 $parts = splitTop($text, ' and ');
161 $etAl = $parts[count($parts) - 1] === 'others';
162 if ($etAl) array_pop($parts);
163 return new self(array_map(Person::parse(...), $parts), $etAl);
164 }
165 public function single(): bool { return count($this->names) === 1 && !$this->etAl; }
166 public function same(self $other): bool {
167 if ($this->etAl !== $other->etAl || count($this->names) !== count($other->names)) return false;
168 foreach ($this->names as $i => $name) if (!$name->same($other->names[$i])) return false;
169 return true;
170 }
171 public function authors(Style $style): string {
172 $values = [];
173 foreach ($this->names as $i => $name) $values[] = $name->notation(
174 $style === Style::APA || $style !== Style::IEEE && $i === 0, $style === Style::APA || $style === Style::IEEE);
175 return match ($style) {
176 Style::APA => $this->etAl ? implode(', ', $values) . ', et al.'
177 : (count($values) > 20 ? implode(', ', array_slice($values, 0, 19)) . ', . . . ' . $values[count($values) - 1] : listNames($values, '&')),
178 Style::IEEE => $this->etAl || count($values) >= 7 ? $values[0] . ' et al.' : listNames($values, 'and', true),
179 Style::CHICAGO => $this->etAl || count($values) >= 11 ? implode(', ', array_slice($values, 0, 7)) . ', et al.' : listNames($values, 'and'),
180 Style::MLA => $this->etAl || count($values) >= 3 ? $values[0] . ', et al.' : listNames($values, 'and'),
181 };
182 }
183 public function editors(Style $style): string {
184 $values = array_map(static fn(Person $name): string => $name->notation(false, $style === Style::APA || $style === Style::IEEE), $this->names);
185 return $this->etAl ? $values[0] . ' et al.' : listNames($values, $style === Style::APA ? '&' : 'and', true);
186 }

188final class Entry {
189 public readonly ?NameList $authors;
190 public readonly ?NameList $editors;
191 public readonly string $sortLast;
192 public readonly string $sortTitle;
193 public string $suffix = '';
194
195 public function __construct(public readonly int $index, public readonly string $type, public readonly string $key, public readonly array $fields) {
196 $this->authors = isset($fields['author']) ? NameList::parse($fields['author']) : null;
197 $this->editors = isset($fields['editor']) ? NameList::parse($fields['editor']) : null;
198 $this->sortLast = lower(plain($this->primary()->names[0]->last));
199 $title = lower($this->value('title'));
200 foreach (['a ', 'an ', 'the '] as $prefix) {
201 if (str_starts_with($title, $prefix)) { $title = substr($title, strlen($prefix)); break; }
202 }
203 $this->sortTitle = $title;
204 }
205 public function primary(): NameList { return $this->authors ?? $this->editors; }
206 public function value(string $field): string { return plain($this->fields[$field] ?? ''); }
207 public function year(Style $style): string { return $this->value('year') . ($style === Style::APA || $style === Style::CHICAGO ? $this->suffix : ''); }
208 public static function compare(self $a, self $b): int {
209 foreach ([[$a->sortLast, $b->sortLast], [$a->value('year'), $b->value('year')], [$a->sortTitle, $b->sortTitle], [$a->key, $b->key]] as [$x, $y]) {
210 $order = strcmp($x, $y);
211 if ($order !== 0) return $order;
212 }
213 return 0;
214 }

217final class Database {
218 private const REQUIRED = [
219 'article' => ['author', 'title', 'journal', 'year'], 'book' => ['author', 'title', 'publisher', 'year'],
220 'inproceedings' => ['author', 'title', 'booktitle', 'year'], 'incollection' => ['author', 'title', 'booktitle', 'publisher', 'year'],
221 'techreport' => ['author', 'title', 'institution', 'year'], 'phdthesis' => ['author', 'title', 'school', 'year'], 'misc' => ['author', 'title', 'year'],
222 ];
223 private array $macros = [];
224 public array $entries = [];
225 public array $byKey = [];
226 public array $messages = [];
227
228 public function __construct(array $lines) {
229 $monthKeys = ['jan', 'feb', 'mar', 'apr', 'may', 'jun', 'jul', 'aug', 'sep', 'oct', 'nov', 'dec'];
230 foreach ($monthKeys as $i => $key) $this->macros[$key] = MONTHS[$i];
231 for ($i = 0; $i < count($lines); $i++) {
232 $text = $lines[$i];
233 if ($text === '' || str_starts_with($text, '%')) continue;
234 if (str_starts_with($text, '@string{')) {
235 $separator = strpos($text, ' = ');
236 $name = substr($text, 8, $separator - 8);
237 try {
238 $value = $this->evaluate(substr($text, $separator + 3, -1), $i + 1);
239 if (isset($this->macros[$name])) throw new LoadError($i + 1, 'duplicate macro');
240 $this->macros[$name] = $value;
241 } catch (LoadError $error) { $this->report($name, $error); }
242 } else {
243 $end = $i + 1;
244 while ($lines[$end] !== '}') $end++;
245 $brace = strpos($text, '{');
246 $type = substr($text, 1, $brace - 1);
247 $key = substr($text, $brace + 1, -1);
248 try { $this->readEntry($lines, $i, $end, $type, $key); }
249 catch (LoadError $error) { $this->report($key, $error); }
250 $i = $end;
251 }
252 }
253 $this->assignSuffixes();
254 $this->messages[] = 'LOADED entries=' . count($this->entries) . ' macros=' . (count($this->macros) - 12) . ' errors=' . count($this->messages);
255 }
256 private function report(string $name, LoadError $error): void {
257 $this->messages[] = "ERROR line {$error->sourceLine}: $name: " . $error->getMessage();
258 }
259 private function evaluate(string $expression, int $line): string {
260 $depth = 0;
261 for ($i = 0; $i < strlen($expression); $i++) {
262 if ($expression[$i] === '{') $depth++;
263 if ($expression[$i] === '}') $depth--;
264 if ($depth < 0) throw new LoadError($line, 'unbalanced braces');
265 }
266 if ($depth !== 0) throw new LoadError($line, 'unbalanced braces');
267 $result = '';
268 foreach (explode(' # ', $expression) as $term) {
269 if ($term[0] === '{' || $term[0] === '"') $result .= substr($term, 1, -1);
270 elseif (isDigit($term[0])) $result .= $term;
271 elseif (isset($this->macros[$term])) $result .= $this->macros[$term];
272 else throw new LoadError($line, "unknown macro '$term'");
273 }
274 return $result;
275 }
276 private function readEntry(array $lines, int $start, int $end, string $type, string $key): void {
277 if (!array_key_exists($type, self::REQUIRED)) throw new LoadError($start + 1, "unknown type '$type'");
278 if (isset($this->byKey[':' . $key])) throw new LoadError($start + 1, 'duplicate key');
279 $fields = [];
280 for ($i = $start + 1; $i < $end; $i++) {
281 $separator = strpos($lines[$i], ' = ');
282 $name = substr($lines[$i], 2, $separator - 2);
283 $expression = substr($lines[$i], $separator + 3);
284 if (str_ends_with($expression, ',')) $expression = substr($expression, 0, -1);
285 $value = $this->evaluate($expression, $i + 1);
286 if (isset($fields[$name])) throw new LoadError($i + 1, "duplicate field '$name'");
287 $fields[$name] = normalize($value);
288 }
289 foreach (self::REQUIRED[$type] as $name) {
290 if (!isset($fields[$name]) && !($type === 'book' && $name === 'author' && isset($fields['editor']))) {
291 throw new LoadError($end + 1, "missing field '$name'");
292 }
293 }
294 $entry = new Entry(count($this->entries), $type, $key, $fields);
295 $this->entries[] = $entry;
296 $this->byKey[':' . $key] = $entry;
297 }
298 private function assignSuffixes(): void {
299 $done = [];
300 foreach ($this->entries as $entry) {
301 if (isset($done[$entry->index])) continue;
302 $group = array_values(array_filter($this->entries, static fn(Entry $other): bool =>
303 $entry->primary()->same($other->primary()) && $entry->fields['year'] === $other->fields['year']));
304 usort($group, static fn(Entry $a, Entry $b): int => strcmp($a->sortTitle, $b->sortTitle) ?: strcmp($a->key, $b->key));
305 foreach ($group as $i => $other) {
306 $done[$other->index] = true;
307 if (count($group) > 1) $other->suffix = chr(97 + $i);
308 }
309 }
310 }

353 public function __construct() {
354 foreach (self::TEMPLATES as $style => $templates) {

356 $position = 0;
357 $this->parsed[$style][] = self::parse($text, $position);

361 private static function parse(string $text, int &$i): array {
362 $parts = [];
363 while ($i < strlen($text) && $text[$i] !== '>') {
364 if ($text[$i] === '{') {
365 $end = strpos($text, '}', $i);
366 $parts[] = new TemplatePart(PartKind::Field, substr($text, $i + 1, $end - $i - 1));
367 $i = $end + 1;
368 } elseif ($text[$i] === '<') {
369 $i++;
370 $parts[] = new TemplatePart(PartKind::Optional, children: self::parse($text, $i));
371 $i++;
372 } else $parts[] = new TemplatePart(PartKind::Literal, $text[$i++]);
373 }
374 return $parts;
375 }
376 private static function editorSuffix(Entry $entry, Style $style): string {
377 if ($entry->authors !== null) return '';
378 $single = $entry->primary()->single();
379 return match ($style) {
380 Style::APA => $single ? ' (Ed.)' : ' (Eds.)', Style::IEEE => $single ? ', Ed.' : ', Eds.',
381 Style::CHICAGO => $single ? ', ed.' : ', eds.', Style::MLA => $single ? ', editor' : ', editors',
382 };
383 }
384 private static function edition(string $value): string {
385 if (!digits($value)) return $value . ' ed.';
386 $number = (int)$value;
387 $suffix = in_array($number % 100, [11, 12, 13], true) ? 'th' : match ($number % 10) { 1 => 'st', 2 => 'nd', 3 => 'rd', default => 'th' };
388 return $value . $suffix . ' ed.';
389 }
390 private static function values(Entry $entry, Style $style, bool $repeated): array {
391 $result = array_map(plain(...), $entry->fields);
392 $result['author'] = ($repeated ? '---' : $entry->primary()->authors($style)) . self::editorSuffix($entry, $style);
393 $result['editor'] = $entry->editors?->editors($style) ?? '';
394 $result['eds'] = $entry->editors === null ? '' : ($entry->editors->single() ? 'Ed.' : 'Eds.');
395 $result['title'] = plain(titleCase($entry->fields['title'], $style));
396 $result['year'] = $entry->year($style);
397 $result['pp'] = '';
398 if (isset($entry->fields['pages'])) {
399 $result['pages'] = str_replace('--', '-', $entry->value('pages'));
400 $result['pp'] = (str_contains($result['pages'], '-') ? 'pp. ' : 'p. ') . $result['pages'];
401 }
402 if (isset($entry->fields['edition'])) $result['edition'] = self::edition($entry->value('edition'));
403 $month = $entry->value('month');
404 $index = array_search($month, MONTHS, true);
405 $result['mon'] = $index === false ? $month : MONTH_SHORT[$index];
406 return $result;
407 }
408 private static function render(array $parts, array $fields, string &$result): void {
409 foreach ($parts as $part) {
410 if ($part->kind === PartKind::Field) {
411 $result .= $fields[$part->text] ?? '';
412 } elseif ($part->kind === PartKind::Optional) {
413 $visible = true;
414 foreach ($part->children as $child) {
415 if ($child->kind === PartKind::Field && ($fields[$child->text] ?? '') === '') { $visible = false; break; }
416 }
417 if ($visible) self::render($part->children, $fields, $result);
418 } elseif ($part->text !== '.' || $result === '' || !str_contains('.?!', $result[strlen($result) - 1])) $result .= $part->text;
419 }
420 }
421 public function reference(Entry $entry, Style $style, int $number, bool $repeated = false): string {
422 $type = match ($entry->type) { 'article' => 0, 'book' => 1, 'inproceedings', 'incollection' => 2, 'techreport' => 3, 'phdthesis' => 4, default => 5 };
423 $result = '';
424 self::render($this->parsed[$style->value][$type], self::values($entry, $style, $repeated), $result);
425 return ($style === Style::IEEE ? "[$number] " : '') . $result;
426 }

456 private static function numberRanges(array $numbers): string {
457 sort($numbers, SORT_NUMERIC);
458 $parts = [];
459 for ($i = 0; $i < count($numbers);) {
460 $end = $i;
461 while ($end + 1 < count($numbers) && $numbers[$end + 1] === $numbers[$end] + 1) $end++;
462 if ($end - $i >= 2) $parts[] = "[{$numbers[$i]}]-[{$numbers[$end]}]";
463 else for ($j = $i; $j <= $end; $j++) $parts[] = "[{$numbers[$j]}]";
464 $i = $end + 1;
465 }
466 return implode(', ', $parts);
467 }
468 private static function citationText(Citation $citation, Style $style, array $numbers): string {
469 $entries = $citation->entries;
470 if ($citation->kind === 'citeauthor') return implode('; ', array_map(static fn(Entry $entry): string => self::shortName($entry, $style, false), $entries));
471 if ($citation->kind === 'citeyear') return implode(', ', array_map(static fn(Entry $entry): string => $entry->year($style), $entries));
472 $parenthetical = $citation->kind !== 'citet';
473 if ($style === Style::IEEE) {
474 return $parenthetical ? self::numberRanges(array_map(static fn(Entry $entry): int => $numbers[$entry->index], $entries))
475 : implode('; ', array_map(static fn(Entry $entry): string => self::shortName($entry, $style, false) . ' [' . $numbers[$entry->index] . ']', $entries));
476 }
477 if ($style === Style::MLA) {
478 $text = implode('; ', array_map(static fn(Entry $entry): string => self::shortName($entry, $style, $parenthetical), $entries));
479 return $parenthetical ? '(' . $text . ')' : $text;
480 }
481 if ($parenthetical) usort($entries, Entry::compare(...));
482 $groups = [];
483 for ($i = 0; $i < count($entries);) {
484 $name = self::shortName($entries[$i], $style, $parenthetical);
485 $end = $i + 1;
486 while ($end < count($entries) && self::shortName($entries[$end], $style, $parenthetical) === $name) $end++;
487 $years = implode(', ', array_map(static fn(Entry $entry): string => $entry->year($style), array_slice($entries, $i, $end - $i)));
488 $groups[] = $parenthetical ? $name . ($style === Style::APA ? ', ' : ' ') . $years : $name . ' (' . $years . ')';
489 $i = $end;
490 }
491 $text = implode('; ', $groups);
492 return $parenthetical ? '(' . $text . ')' : $text;
493 }
494 private function cite(string $text, Style $style): string {
495 $citations = [];
496 for ($start = strpos($text, '\\'); $start !== false;) {
497 $brace = strpos($text, '{', $start); $end = strpos($text, '}', $brace);
498 $seen = []; $entries = [];
499 foreach (explode(',', substr($text, $brace + 1, $end - $brace - 1)) as $key) {
500 $entry = $this->lookup($key);
501 if (!isset($seen[$entry->index])) { $seen[$entry->index] = true; $entries[] = $entry; }
502 }
503 $citations[] = new Citation($start, $end + 1, substr($text, $start + 1, $brace - $start - 1), $entries);
504 $start = strpos($text, '\\', $end + 1);
505 }
506 foreach ($citations as $citation) {
507 foreach ($citation->entries as $entry) {
508 if (!isset($this->recorded[$entry->index])) { $this->recorded[$entry->index] = true; $this->record[] = $entry; }
509 }
510 }
511 $numbers = $this->ieeeNumbers();
512 $result = ''; $position = 0;
513 foreach ($citations as $citation) {
514 $result .= substr($text, $position, $citation->start - $position) . self::citationText($citation, $style, $numbers);
515 $position = $citation->end;
516 }
517 return $result . substr($text, $position);
518 }
519 private static function baseKey(Entry $entry): string {
520 $result = letters(plain($entry->primary()->names[0]->last));
521 foreach (str_split($entry->fields['year']) as $c) if (isDigit($c)) $result .= $c;
522 foreach (explode(' ', $entry->value('title')) as $word) {
523 $candidate = letters($word);
524 if ($candidate !== '' && !in_array($candidate, SMALL_WORDS, true)) { $result .= $candidate; break; }
525 }
526 return $result;
527 }

529 $bases = array_map(self::baseKey(...), $this->database->entries);
530 $counts = []; $used = []; $output = [];

540 }

548 private function check(): array {
549 $output = []; $titles = [];
550 foreach ($this->database->entries as $entry) {
551 $warnings = [];
552 if (isset($entry->fields['pages'])) {
553 $pages = $entry->value('pages');
554 if (str_contains($pages, '-') && !str_contains($pages, '--')) $warnings[] = 'pages use a single hyphen';
555 $parts = array_values(array_filter(explode('-', $pages), static fn(string $part): bool => $part !== ''));
556 if (count($parts) === 2 && digits($parts[0]) && digits($parts[1]) && (int)$parts[0] > (int)$parts[1]) $warnings[] = 'pages are reversed';
557 }
558 $year = $entry->value('year');
559 if (strlen($year) !== 4 || !digits($year)) $warnings[] = 'year is not 4 digits';
560 $title = $entry->value('title');
561 $hasLower = false; $upperCount = 0;
562 foreach (str_split($title) as $c) { if (isLower($c)) $hasLower = true; if (isUpper($c)) $upperCount++; }
563 if (!$hasLower && $upperCount >= 2) $warnings[] = 'title is all caps';
564 if ($entry->type === 'article' && !isset($entry->fields['doi'])) $warnings[] = 'article has no doi';
565 $titleKey = ':' . lower($title);
566 if (isset($titles[$titleKey])) $warnings[] = 'same title as ' . $titles[$titleKey];
567 else $titles[$titleKey] = $entry->key;
568 foreach (self::unknownFields($entry) as $field) $warnings[] = "unknown field '$field'";
569 foreach ($warnings as $warning) $output[] = "WARN {$entry->key}: $warning";
570 }
571 $output[] = 'CHECK warnings=' . count($output);
572 return $output;
573 }
574 private static function exportEntry(Entry $entry): array {
575 $fields = array_values(array_filter(KNOWN_FIELDS, static fn(string $field): bool => isset($entry->fields[$field])));
576 $fields = array_merge($fields, self::unknownFields($entry));
577 $output = ['@' . $entry->type . '{' . $entry->key . ','];
578 foreach ($fields as $i => $field) $output[] = ' ' . $field . ' = {' . $entry->fields[$field] . '}' . ($i + 1 < count($fields) ? ',' : '');
579 $output[] = '}';
580 return $output;
581 }
582 private function stats(): array {
583 $types = []; $years = []; $authors = []; $pairs = [];
584 foreach ($this->database->entries as $entry) {
585 $types[$entry->type] = ($types[$entry->type] ?? 0) + 1;
586 $year = ':' . $entry->value('year');
587 $years[$year] = ($years[$year] ?? 0) + 1;
588 if ($entry->authors === null) continue;
589 $names = [];
590 foreach ($entry->authors->names as $name) {
591 $full = $name->notation(false, false);
592 $names[':' . $full] = $full;
593 $authors[':' . $full] = true;
594 }
595 $names = array_values($names);
596 sort($names, SORT_STRING);
597 for ($i = 0; $i < count($names); $i++) {
598 for ($j = $i + 1; $j < count($names); $j++) {
599 $a = ':' . $names[$i]; $b = ':' . $names[$j];
600 $pairs[$a][$b] = ($pairs[$a][$b] ?? 0) + 1;
601 }
602 }
603 }
604 $output = ['ENTRIES ' . count($this->database->entries)];
605 ksort($types, SORT_STRING); ksort($years, SORT_STRING);
606 foreach ($types as $type => $count) $output[] = "TYPE $type $count";
607 foreach ($years as $year => $count) $output[] = 'YEAR ' . substr($year, 1) . ' ' . $count;
608 $output[] = 'AUTHORS ' . count($authors);
609 $best = null;
610 foreach ($pairs as $a => $seconds) {
611 foreach ($seconds as $b => $count) {
612 if ($best === null || $count > $best[2] || $count === $best[2] &&
613 (strcmp($a, $best[0]) < 0 || $a === $best[0] && strcmp($b, $best[1]) < 0)) $best = [$a, $b, $count];
614 }
615 }
616 $output[] = $best === null ? 'PAIR none' : 'PAIR ' . substr($best[0], 1) . ' & ' . substr($best[1], 1) . ' ' . $best[2];
617 return $output;
618 }
619 public function execute(string $line): array {
620 $command = explode(' ', $line, 2)[0];
621 $expected = match ($command) {
622 'FORMAT', 'CITE', 'SEARCH' => 2, 'BIBLIOGRAPHY', 'EXPORT' => 1, 'KEYGEN', 'CHECK', 'STATS' => 0,
623 default => throw new CommandError("unknown command '$command'"),
624 };
625 $words = in_array($command, ['CITE', 'SEARCH'], true) ? explode(' ', $line, 3) : explode(' ', $line);
626 if ($command === 'FORMAT' ? count($words) - 1 < $expected : count($words) - 1 !== $expected) throw new CommandError('wrong number of arguments');
627 $style = Style::APA;
628 if (in_array($command, ['FORMAT', 'BIBLIOGRAPHY', 'CITE'], true)) {
629 $style = Style::tryFrom($words[1]) ?? throw new CommandError("unknown style '{$words[1]}'");
630 }
631 $output = [];
632 switch ($command) {
633 case 'FORMAT':
634 $numbers = $this->ieeeNumbers();
635 foreach (array_slice($words, 2) as $key) {
636 $entry = $this->database->byKey[':' . $key] ?? null;
637 $output[] = $entry === null ? "ERROR unknown key '$key'" : $this->formatter->reference($entry, $style, $numbers[$entry->index]);
638 }
639 break;
640 case 'BIBLIOGRAPHY':
641 $output[] = $style === Style::MLA ? 'Works Cited' : 'References';
642 $numbers = $this->ieeeNumbers();
643 $entries = $style === Style::IEEE ? $this->ieeeOrder() : $this->database->entries;
644 if ($style !== Style::IEEE) usort($entries, Entry::compare(...));
645 $previous = null;
646 foreach ($entries as $entry) {
647 $repeated = ($style === Style::CHICAGO || $style === Style::MLA) && $previous !== null && $previous->same($entry->primary());
648 $output[] = $this->formatter->reference($entry, $style, $numbers[$entry->index], $repeated);
649 $previous = $entry->primary();
650 }
651 break;
652 case 'CITE': return [$this->cite($words[2], $style)];
653 case 'KEYGEN': return $this->keygen();
654 case 'CHECK': return $this->check();
655 case 'SEARCH':
656 $matches = []; $needle = lower($words[2]);
657 foreach ($this->database->entries as $entry) {
658 if (isset($entry->fields[$words[1]]) && str_contains(lower($entry->value($words[1])), $needle)) $matches[] = $entry->key;
659 }
660 return $matches === [] ? ['MATCHES 0'] : ['MATCHES ' . count($matches) . ': ' . implode(' ', $matches)];
661 case 'EXPORT': return self::exportEntry($this->lookup($words[1]));
662 case 'STATS': return $this->stats();
663 }
664 return $output;
665 }
666}
