61function parseCommand(line: string): Command {
62 const firstSpace = line.indexOf(" ");
63 const name = firstSpace < 0 ? line : line.slice(0, firstSpace);
64 if (!isCommandName(name)) throw new BibError(`unknown command '${name}'`);
65 let parts: string[];
66 if (name === "CITE" || name === "SEARCH") {
67 const secondSpace = firstSpace < 0 ? -1 : line.indexOf(" ", firstSpace + 1);
68 parts = firstSpace < 0 ? [line] : secondSpace < 0 ? [name, line.slice(firstSpace + 1)]
69 : [name, line.slice(firstSpace + 1, secondSpace), line.slice(secondSpace + 1)];
70 } else parts = line.split(" ");
71 const args = parts.slice(1);
72 if (name === "FORMAT" ? args.length < 2 : args.length !== commandArity[name]) {
73 throw new BibError("wrong number of arguments");
74 }
75 if (name === "FORMAT" || name === "BIBLIOGRAPHY" || name === "CITE") {
76 const style = args[0];
77 if (!isStyle(style)) throw new BibError(`unknown style '${style}'`);
78 if (name === "FORMAT") return { kind: name, style, keys: args.slice(1) };
79 if (name === "BIBLIOGRAPHY") return { kind: name, style };
80 return { kind: name, style, text: args[1] };
81 }
82 if (name === "SEARCH") return { kind: name, field: args[0], query: args[1] };
83 if (name === "EXPORT") return { kind: name, key: args[0] };
84 return { kind: name };
85}

87function compareTuple(a: readonly string[], b: readonly string[]): number {
88 for (let i = 0; i < a.length; i++) {
89 const order = compare(a[i], b[i]);
90 if (order !== 0) return order;
91 }
92 return 0;

95function splitAtDepthZero(text: string, separator: string): string[] {
96 const parts: string[] = [];
97 let depth = 0, start = 0, i = 0;
98 while (i < text.length) {
99 if (depth === 0 && text.startsWith(separator, i)) {
100 parts.push(text.slice(start, i));
101 i += separator.length;
102 start = i;
103 } else {
104 if (text[i] === "{") depth++;
105 if (text[i] === "}") depth--;
106 i++;
107 }
108 }
109 parts.push(text.slice(start));
110 return parts;
111}

113function accentAt(text: string, i: number): { char: string; length: number } | null {
114 if (text[i] !== "\\") return null;
115 if (text[i + 1] === "&") return { char: "&", length: 2 };
116 const symbol = text[i + 1];
117 if ("'\"`^~=.".includes(symbol) && isLetter(text[i + 2])) {
118 return { char: text[i + 2], length: 3 };
119 }
120 if ("'\"`^~=.cuvH".includes(symbol) && text[i + 2] === "{" &&
121 isLetter(text[i + 3]) && text[i + 4] === "}") {
122 return { char: text[i + 3], length: 5 };
123 }
124 return null;
125}

127function normalize(text: string): string {
128 let result = "";
129 for (let i = 0; i < text.length;) {
130 const enclosed = text[i] === "{" ? accentAt(text, i + 1) : null;
131 if (enclosed && text[i + 1 + enclosed.length] === "}") {
132 result += enclosed.char;
133 i += enclosed.length + 2;
134 continue;
135 }
136 const accent = accentAt(text, i);
137 if (accent) {
138 result += accent.char;
139 i += accent.length;
140 } else {
141 result += text[i++];
142 }
143 }
144 return result.replace(/~/g, " ").replace(/ +/g, " ").trim();
145}

147function evaluateExpression(expression: string, macros: ReadonlyMap<string, string>): string {
148 let depth = 0;
149 for (const char of expression) {
150 if (char === "{") depth++;
151 if (char === "}") depth--;
152 if (depth < 0) throw new BibError("unbalanced braces");
153 }
154 if (depth !== 0) throw new BibError("unbalanced braces");
155 return expression.split(" # ").map(term => {
156 if (term[0] === "{" || term[0] === '"') return term.slice(1, -1);
157 if (/^[0-9]+$/.test(term)) return term;
158 const value = macros.get(term);
159 if (value === undefined) throw new BibError(`unknown macro '${term}'`);
160 return value;
161 }).join("");
162}

164class Person {
165 readonly first: string;
166 readonly von: string;
167 readonly last: string;
168 readonly jr: string;
169
170 constructor(text: string) {
171 const parts = splitAtDepthZero(text, ",").map(part => part.trim());
172 const words = splitAtDepthZero(parts[0], " ");
173 const lower = words.flatMap((word, i) =>
174 i < words.length - 1 && /^[a-z]/.test(word) ? [i] : []);
175 this.first = "";
176 this.von = "";
177 this.jr = "";
178 if (parts.length === 1) {
179 const start = lower.length ? lower[0] : words.length - 1;
180 const end = lower.length ? lower.at(-1)! + 1 : words.length - 1;
181 this.first = words.slice(0, start).join(" ");
182 this.von = words.slice(start, end).join(" ");
183 this.last = words.slice(end).join(" ");
184 } else {
185 const end = lower.length ? lower.at(-1)! + 1 : 0;
186 this.von = words.slice(0, end).join(" ");
187 this.last = words.slice(end).join(" ");
188 this.first = parts.at(-1)!;
189 if (parts.length === 3) this.jr = parts[1];
190 }
191 }
192
193 surname(): string {
194 return plain(this.von ? `${this.von} ${this.last}` : this.last);
195 }
196
197 initials(): string {
198 const given = plain(this.first);
199 if (!given) return "";
200 return given.split(" ").map(word => word.split("-").map(piece =>
201 piece.match(/[A-Za-z]/)![0].toUpperCase() + ".").join("-")).join(" ");
202 }
203
204 format(inverted: boolean, initials: boolean): string {
205 const given = initials ? this.initials() : plain(this.first);
206 let result = this.surname();
207 if (this.first) result = inverted ? `${result}, ${given}` : `${given} ${result}`;
208 if (this.jr) result += `, ${plain(this.jr)}`;
209 return result;
210 }
211}

213class NameList {
214 readonly names: readonly Person[];
215 readonly etAl: boolean;
216 readonly signature: string;
217
218 constructor(text: string) {
219 const parts = splitAtDepthZero(text, " and ");
220 this.etAl = parts.at(-1) === "others";
221 this.names = (this.etAl ? parts.slice(0, -1) : parts).map(name => new Person(name));
222 this.signature = JSON.stringify([
223 this.names.map(name => [name.first, name.von, name.last, name.jr]), this.etAl,

263class Database {
264 readonly entries: readonly Entry[];
265 readonly byKey: ReadonlyMap<string, Entry>;
266 readonly bibliography: readonly Entry[];
267
268 constructor(lines: readonly string[], output: string[]) {
269 const entries: Entry[] = [];
270 const byKey = new Map<string, Entry>();
271 this.entries = entries;
272 this.byKey = byKey;
273 const macros = new Map(monthNames.map(month => [month.slice(0, 3).toLowerCase(), month]));
274 let errorCount = 0;
275 const report = (line: number, name: string, reason: string): void => {
276 output.push(`ERROR line ${line}: ${name}: ${reason}`);
277 errorCount++;
278 };
279 for (let i = 0; i < lines.length; i++) {
280 const line = lines[i];
281 if (!line || line.startsWith("%")) continue;
282 if (line.startsWith("@string{")) {
283 const [, name, expression] = line.match(/^@string\{([a-z][a-z0-9]*) = (.*)\}$/)!;
284 try {
285 const value = evaluateExpression(expression, macros);
286 if (macros.has(name)) throw new BibError("duplicate macro");
287 macros.set(name, value);
288 } catch (error) {
289 if (!(error instanceof BibError)) throw error;
290 report(i + 1, name, error.message);
291 }
292 continue;
293 }
294 const [, type, key] = line.match(/^@([a-z]+)\{([^,]+),$/)!;
295 const fields = new Map<string, string>();
296 let failed = false;
297 const fail = (reason: string): void => {
298 report(i + 1, key, reason);
299 failed = true;
300 };
301 if (!isEntryType(type)) fail(`unknown type '${type}'`);
302 else if (this.byKey.has(key)) fail("duplicate key");
303 while (lines[++i] !== "}") {
304 if (failed) continue;
305 const [, name, raw] = lines[i].match(/^ ([a-z]+) = (.*)$/)!;
306 try {
307 const value = evaluateExpression(raw.endsWith(",") ? raw.slice(0, -1) : raw, macros);
308 if (fields.has(name)) throw new BibError(`duplicate field '${name}'`);
309 fields.set(name, normalize(value));
310 } catch (error) {
311 if (!(error instanceof BibError)) throw error;
312 fail(error.message);
313 }
314 }
315 if (failed || !isEntryType(type)) continue;
316 for (const field of requiredFields.get(type)!) {
317 if (!fields.has(field) && !(type === "book" && field === "author" && fields.has("editor"))) {
318 fail(`missing field '${field}'`);
319 break;
320 }
321 }
322 if (!failed) {
323 const entry = new Entry(key, type, fields);
324 entries.push(entry);
325 byKey.set(key, entry);
326 }
327 }
328 output.push(`LOADED entries=${this.entries.length} macros=${macros.size - 12} errors=${errorCount}`);
329 const groups = new Map<string, Entry[]>();
330 for (const entry of this.entries) {
331 const groupKey = JSON.stringify([entry.primary.signature, entry.fields.get("year")]);
332 const group = groups.get(groupKey) ?? [];
333 group.push(entry);
334 groups.set(groupKey, group);
335 }
336 for (const group of groups.values()) {
337 if (group.length < 2) continue;
338 group.sort((a, b) => compareTuple([a.sortTitle, a.key], [b.sortTitle, b.key]));
339 group.forEach((entry, i) => { entry.yearSuffix = String.fromCharCode(97 + i); });
340 }
341 this.bibliography = [...this.entries].sort((a, b) => compareTuple(a.sortKey, b.sortKey));
342 }
343}

345function listNames(names: readonly string[], conjunction: string, useListB = false): string {
346 if (names.length === 1) return names[0];
347 if (useListB && names.length === 2) return names.join(` ${conjunction} `);
348 return `${names.slice(0, -1).join(", ")}, ${conjunction} ${names.at(-1)}`;

351function authorValue(entry: Entry, style: Style, repeated: boolean): string {
352 const { names, etAl } = entry.primary;
353 let result = "---";
354 if (!repeated) {
355 if (style === "APA") {
356 const formatted = names.map(name => name.format(true, true));
357 result = etAl ? formatted.join(", ") + ", et al." : names.length <= 20
358 ? listNames(formatted, "&") : formatted.slice(0, 19).join(", ") + ", . . . " + formatted.at(-1);
359 } else if (style === "IEEE") {
360 const formatted = names.map(name => name.format(false, true));
361 result = etAl || names.length >= 7 ? formatted[0] + " et al." : listNames(formatted, "and", true);
362 } else {
363 const formatted = names.map((name, i) => name.format(i === 0, false));
364 if (style === "CHICAGO" && (etAl || names.length >= 11)) result = formatted.slice(0, 7).join(", ") + ", et al.";
365 else if (style === "MLA" && (etAl || names.length >= 3)) result = formatted[0] + ", et al.";
366 else result = listNames(formatted, "and");
367 }
368 }
369 if (!entry.author) {
370 const suffixes = { APA: [" (Ed.)", " (Eds.)"], IEEE: [", Ed.", ", Eds."],
371 CHICAGO: [", ed.", ", eds."], MLA: [", editor", ", editors"] };
372 result += suffixes[style][entry.primary.singular() ? 0 : 1];
373 }
374 return result;
375}

377function editorValue(list: NameList | null, style: Style): string {
378 if (!list) return "";
379 const names = list.names.map(name => name.format(false, style === "APA" || style === "IEEE"));
380 return list.etAl ? names[0] + " et al." : listNames(names, style === "APA" ? "&" : "and", true);

383function convertLetters(text: string, transform: (char: string, index: number, depth: number) => string): string {
384 let depth = 0, result = "";
385 for (let i = 0; i < text.length; i++) {
386 const char = text[i];
387 if (char === "{") depth++;
388 if (char === "}") depth--;
389 result += isLetter(char) ? transform(char, i, depth) : char;
390 }
391 return result;
392}

394function titleValue(title: string, style: Style): string {
395 if (style === "IEEE") return plain(title);
396 if (style === "APA") {
397 return plain(convertLetters(title, (char, i, depth) => depth ? char :
398 i === 0 || title.slice(Math.max(0, i - 2), i) === ": " ? char.toUpperCase() : char.toLowerCase()));
399 }
400 const words = splitAtDepthZero(title, " ");
401 return plain(words.map((word, i) => {
402 if (word.startsWith("{")) return word;
403 if (i > 0 && i < words.length - 1 && !words[i - 1].endsWith(":") && smallWords.has(letters(word))) {
404 return convertLetters(word, (char, _index, depth) => depth ? char : char.toLowerCase());
405 }
406 let first = true;
407 return convertLetters(word, (char, _index, depth) => {
408 const result = first && depth === 0 ? char.toUpperCase() : char;
409 first = false;
410 return result;
411 });
412 }).join(" "));
413}

415function ordinal(value: string): string {
416 const number = Number(value);
417 const lastTwo = number % 100;
418 const suffix = lastTwo >= 11 && lastTwo <= 13 ? "th" : number % 10 === 1 ? "st"
419 : number % 10 === 2 ? "nd" : number % 10 === 3 ? "rd" : "th";
420 return value + suffix;
421}

423function placeholderValues(entry: Entry, style: Style, repeated: boolean): ReadonlyMap<string, string> {
424 const values = new Map([...entry.fields].map(([field, value]) => [field, plain(value)]));
425 values.set("author", authorValue(entry, style, repeated));
426 values.set("editor", editorValue(entry.editor, style));
427 values.set("eds", entry.editor ? entry.editor.singular() ? "Ed." : "Eds." : "");
428 values.set("title", titleValue(entry.fields.get("title")!, style));
429 values.set("year", entry.yearLabel(style));
430 const pages = entry.value("pages").replace(/--/g, "-");
431 values.set("pages", pages);
432 values.set("pp", entry.fields.has("pages") ? (pages.includes("-") ? "pp. " : "p. ") + pages : "");
433 const edition = entry.value("edition");
434 values.set("edition", entry.fields.has("edition") ? (/^[0-9]+$/.test(edition) ? ordinal(edition) : edition) + " ed." : "");
435 const month = entry.value("month");
436 values.set("mon", monthAbbreviations.get(month) ?? month);
437 return values;
438}

440function parseTemplate(source: string): readonly TemplateNode[] {
441 let position = 0;
442 function sequence(): TemplateNode[] {
443 const nodes: TemplateNode[] = [];
444 while (position < source.length && source[position] !== ">") {
445 if (source[position] === "<") {
446 position++;
447 nodes.push({ kind: "optional", children: sequence() });
448 position++;
449 } else if (source[position] === "{") {
450 const end = source.indexOf("}", position);
451 nodes.push({ kind: "value", name: source.slice(position + 1, end) });
452 position = end + 1;
453 } else {
454 const start = position++;
455 while (position < source.length && !"<{>".includes(source[position])) position++;
456 nodes.push({ kind: "literal", text: source.slice(start, position) });
457 }
458 }
459 return nodes;
460 }
461 return sequence();
462}

464function renderTemplate(nodes: readonly TemplateNode[], values: ReadonlyMap<string, string>): string {
465 let result = "";
466 function visit(sequence: readonly TemplateNode[]): void {
467 for (const node of sequence) {
468 if (node.kind === "value") result += values.get(node.name) ?? "";
469 else if (node.kind === "optional") {
470 if (node.children.every(child => child.kind !== "value" || (values.get(child.name) ?? "") !== "")) visit(node.children);
471 } else {
472 for (const char of node.text) {
473 if (char !== "." || !/[.?!]$/.test(result)) result += char;
474 }
475 }
476 }
477 }
478 visit(nodes);
479 return result;
480}

540 private reference(entry: Entry, style: Style, numbers: ReadonlyMap<string, number>, repeated = false): string {
541 const type = entry.type === "incollection" ? "inproceedings" : entry.type;
542 const reference = renderTemplate(templates.get(style)!.get(type)!, placeholderValues(entry, style, repeated));
543 return (style === "IEEE" ? `[${numbers.get(entry.key)}] ` : "") + reference;
544 }

546 private citation(kind: CitationKind, entries: readonly Entry[], style: Style, numbers: ReadonlyMap<string, number>): string {
547 const parenthetical = kind === "cite" || kind === "citep";
548 const name = (entry: Entry): string => shortName(entry, style, parenthetical);
549 if (kind === "citeauthor") return entries.map(name).join("; ");
550 if (kind === "citeyear") return entries.map(entry => entry.yearLabel(style)).join(", ");
551 if (style === "MLA") {
552 const result = entries.map(name).join("; ");
553 return parenthetical ? `(${result})` : result;
554 }
555 if (style === "IEEE") {
556 if (!parenthetical) return entries.map(entry => `${name(entry)} [${numbers.get(entry.key)}]`).join("; ");
557 const sorted = entries.map(entry => numbers.get(entry.key)!).sort((a, b) => a - b);
558 const pieces: string[] = [];
559 for (let i = 0; i < sorted.length;) {
560 let end = i + 1;
561 while (end < sorted.length && sorted[end] === sorted[end - 1] + 1) end++;
562 if (end - i >= 3) pieces.push(`[${sorted[i]}]-[${sorted[end - 1]}]`);
563 else for (let j = i; j < end; j++) pieces.push(`[${sorted[j]}]`);
564 i = end;
565 }
566 return pieces.join(", ");
567 }
568 const ordered = parenthetical ? [...entries].sort((a, b) => compareTuple(a.sortKey, b.sortKey)) : entries;
569 const groups: { name: string; years: string[] }[] = [];
570 for (const entry of ordered) {
571 const currentName = name(entry);
572 if (!groups.length || groups.at(-1)!.name !== currentName) groups.push({ name: currentName, years: [] });
573 groups.at(-1)!.years.push(entry.yearLabel(style));
574 }
575 const result = groups.map(group => parenthetical
576 ? group.name + (style === "APA" ? ", " : " ") + group.years.join(", ")
577 : `${group.name} (${group.years.join(", ")})`).join("; ");
578 return parenthetical ? `(${result})` : result;
579 }

581 private cite(style: Style, text: string): void {
582 const citations: Citation[] = [...text.matchAll(/\\(citeauthor|citeyear|citep|citet|cite)\{([^}]+)\}/g)].map(match => ({
583 start: match.index, length: match[0].length, kind: match[1] as CitationKind, keys: [...new Set(match[2].split(","))],
584 }));
585 for (const citation of citations) {
586 for (const key of citation.keys) {
587 if (!this.database.byKey.has(key)) {
588 this.output.push(`ERROR unknown key '${key}'`);
589 return;
590 }
591 }
592 }
593 for (const citation of citations) {
594 for (const key of citation.keys) {
595 if (!this.cited.has(key)) {
596 this.cited.add(key);
597 this.citationRecord.push(key);
598 }
599 }
600 }
601 const numbers = this.ieeeNumbers();
602 let result = "", position = 0;
603 for (const citation of citations) {
604 result += text.slice(position, citation.start);
605 result += this.citation(citation.kind, citation.keys.map(key => this.database.byKey.get(key)!), style, numbers);
606 position = citation.start + citation.length;
607 }
608 this.output.push(result + text.slice(position));
609 }

611 private keygen(): void {
612 const bases = this.database.entries.map(entry => {
613 const titleWord = entry.value("title").split(" ").map(letters).find(word => word && !smallWords.has(word)) ?? "";
614 return letters(entry.primary.names[0].last) + entry.fields.get("year")!.replace(/[^0-9]/g, "") + titleWord;
615 });
616 const counts = new Map<string, number>(), used = new Map<string, number>();
617 for (const base of bases) counts.set(base, (counts.get(base) ?? 0) + 1);
618 this.database.entries.forEach((entry, i) => {
619 const base = bases[i];
620 const suffix = counts.get(base)! > 1 ? String.fromCharCode(97 + (used.get(base) ?? 0)) : "";
621 used.set(base, (used.get(base) ?? 0) + 1);
622 this.output.push(`${entry.key} -> ${base}${suffix}`);
623 });
624 }

626 private check(): void {
627 let warnings = 0;
628 const titles = new Map<string, string>();
629 for (const entry of this.database.entries) {
630 const warn = (message: string): void => { this.output.push(`WARN ${entry.key}: ${message}`); warnings++; };
631 const pages = entry.value("pages"), title = entry.value("title");
632 if (pages.includes("-") && !pages.includes("--")) warn("pages use a single hyphen");
633 const pageParts = pages.split("-").filter(part => part !== "");
634 if (pageParts.length === 2 && pageParts.every(part => /^[0-9]+$/.test(part)) && Number(pageParts[0]) > Number(pageParts[1])) warn("pages are reversed");
635 if (!/^[0-9]{4}$/.test(entry.value("year"))) warn("year is not 4 digits");
636 if (!/[a-z]/.test(title) && (title.match(/[A-Z]/g)?.length ?? 0) >= 2) warn("title is all caps");
637 if (entry.type === "article" && !entry.fields.has("doi")) warn("article has no doi");
638 const titleKey = title.toLowerCase();
639 if (titles.has(titleKey)) warn(`same title as ${titles.get(titleKey)}`);
640 else titles.set(titleKey, entry.key);
641 for (const field of entry.unknownFields()) warn(`unknown field '${field}'`);
642 }
643 this.output.push(`CHECK warnings=${warnings}`);
644 }

646 private export(key: string): void {
647 const entry = this.database.byKey.get(key);
648 if (!entry) {
649 this.output.push(`ERROR unknown key '${key}'`);
650 return;
651 }
652 this.output.push(`@${entry.type}{${key},`);

654 fields.forEach((field, i) => this.output.push(` ${field} = {${entry.fields.get(field)}}${i + 1 < fields.length ? "," : ""}`));
655 this.output.push("}");
656 }

658 private stats(): void {
659 const types = new Map<EntryType, number>(), years = new Map<string, number>();
660 const authors = new Set<string>(), pairs = new Map<string, AuthorPair>();
661 for (const entry of this.database.entries) {
662 types.set(entry.type, (types.get(entry.type) ?? 0) + 1);
663 const year = entry.value("year");
664 years.set(year, (years.get(year) ?? 0) + 1);
665 if (!entry.author) continue;
666 const names = [...new Set(entry.author.names.map(name => name.format(false, false)))].sort(compare);
667 for (const name of names) authors.add(name);
668 for (let i = 0; i < names.length; i++) {
669 for (let j = i + 1; j < names.length; j++) {
670 const key = JSON.stringify([names[i], names[j]]);
671 if (!pairs.has(key)) pairs.set(key, { a: names[i], b: names[j], count: 0 });
672 pairs.get(key)!.count++;
673 }
674 }
675 }
676 this.output.push(`ENTRIES ${this.database.entries.length}`);
677 for (const type of [...types.keys()].sort(compare)) this.output.push(`TYPE ${type} ${types.get(type)}`);
678 for (const year of [...years.keys()].sort(compare)) this.output.push(`YEAR ${year} ${years.get(year)}`);
679 this.output.push(`AUTHORS ${authors.size}`);
680 const best = [...pairs.values()].sort((a, b) => b.count - a.count || compareTuple([a.a, a.b], [b.a, b.b]))[0];
681 this.output.push(best ? `PAIR ${best.a} & ${best.b} ${best.count}` : "PAIR none");
682 }

684 command(command: Command): void {
685 switch (command.kind) {
686 case "FORMAT": {
687 const numbers = this.ieeeNumbers();
688 for (const key of command.keys) {
689 const entry = this.database.byKey.get(key);
690 this.output.push(entry ? this.reference(entry, command.style, numbers) : `ERROR unknown key '${key}'`);
691 }
692 break;
693 }
694 case "BIBLIOGRAPHY": {
695 const { style } = command;
696 const numbers = this.ieeeNumbers();
697 const entries = style === "IEEE" ? this.ieeeOrder().map(key => this.database.byKey.get(key)!) : this.database.bibliography;
698 this.output.push(style === "MLA" ? "Works Cited" : "References");
699 let previous: string | null = null;
700 for (const entry of entries) {
701 const repeated = (style === "CHICAGO" || style === "MLA") && previous === entry.primary.signature;
702 this.output.push(this.reference(entry, style, numbers, repeated));
703 previous = entry.primary.signature;
704 }
705 break;
706 }
707 case "CITE": this.cite(command.style, command.text); break;
708 case "KEYGEN": this.keygen(); break;
709 case "CHECK": this.check(); break;
710 case "EXPORT": this.export(command.key); break;
711 case "STATS": this.stats(); break;
712 case "SEARCH": {
713 const query = command.query.toLowerCase();
714 const keys = this.database.entries.filter(entry => entry.fields.has(command.field) &&
715 entry.value(command.field).toLowerCase().includes(query)).map(entry => entry.key);
716 this.output.push(keys.length ? `MATCHES ${keys.length}: ${keys.join(" ")}` : "MATCHES 0");
717 break;
718 }
719 }
720 }
721}
