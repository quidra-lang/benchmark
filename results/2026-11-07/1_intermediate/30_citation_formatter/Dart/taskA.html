21bool isLetter(int c) => c >= 65 && c <= 90 || c >= 97 && c <= 122;
22bool isDigit(int c) => c >= 48 && c <= 57;
23bool isLower(int c) => c >= 97 && c <= 122;
24bool isUpper(int c) => c >= 65 && c <= 90;
25bool digits(String text) => text.isNotEmpty && text.codeUnits.every(isDigit);
26int lowerCode(int c) => isUpper(c) ? c + 32 : c;
27int upperCode(int c) => isLower(c) ? c - 32 : c;
28String lower(String text) => String.fromCharCodes(text.codeUnits.map(lowerCode));
29String plain(String text) => text.replaceAll('{', '').replaceAll('}', '');
30String letters(String text) => String.fromCharCodes(text.codeUnits.where(isLetter));

32List<String> splitTop(String text, String separator) {
33 final result = <String>[];
34 var depth = 0, start = 0, i = 0;
35 while (i < text.length) {
36 if (depth == 0 && text.startsWith(separator, i)) {
37 result.add(text.substring(start, i));
38 i += separator.length;
39 start = i;
40 continue;
41 }
42 if (text[i] == '{') depth++;
43 if (text[i] == '}') depth--;
44 i++;
45 }
46 result.add(text.substring(start));
47 return result;
48}

50(String, int)? accent(String text, int i) {
51 if (i + 1 >= text.length || text[i] != '\\') return null;
52 final command = text[i + 1];
53 if (command == '&') return ('&', 2);
54 final symbol = "'\"`^~=.".contains(command);
55 if (i + 4 < text.length && (symbol || 'cuvH'.contains(command)) &&
56 text[i + 2] == '{' && isLetter(text.codeUnitAt(i + 3)) && text[i + 4] == '}') {
57 return (text[i + 3], 5);
58 }
59 if (symbol && i + 2 < text.length && isLetter(text.codeUnitAt(i + 2))) return (text[i + 2], 3);
60 return null;
61}

63String normalize(String text) {
64 final result = StringBuffer();
65 for (var i = 0; i < text.length;) {
66 final wrapped = text[i] == '{' ? accent(text, i + 1) : null;
67 if (wrapped != null && i + wrapped.$2 + 1 < text.length && text[i + wrapped.$2 + 1] == '}') {
68 result.write(wrapped.$1);
69 i += wrapped.$2 + 2;
70 } else {
71 final direct = accent(text, i);
72 if (direct == null) {
73 result.write(text[i++]);
74 } else {
75 result.write(direct.$1);
76 i += direct.$2;
77 }
78 }
79 }
80 return result.toString().replaceAll('~', ' ').replaceAll(RegExp(' +'), ' ').trim();
81}

83String lowerOutside(String word) {
84 final result = word.codeUnits.toList();
85 var depth = 0;
86 for (var i = 0; i < word.length; i++) {
87 if (word[i] == '{') {
88 depth++;
89 } else if (word[i] == '}') {
90 depth--;
91 } else if (depth == 0) {
92 result[i] = lowerCode(result[i]);
93 }
94 }
95 return String.fromCharCodes(result);
96}

98String titleCase(String text, Style style) {
99 if (style == Style.ieee) return text;
100 if (style == Style.apa) {
101 final result = text.codeUnits.toList();
102 var depth = 0;
103 for (var i = 0; i < text.length; i++) {
104 if (text[i] == '{') {
105 depth++;
106 } else if (text[i] == '}') {
107 depth--;
108 } else if (depth == 0 && isLetter(result[i])) {
109 result[i] = i == 0 || i >= 2 && text[i - 2] == ':' && text[i - 1] == ' '
110 ? upperCode(result[i]) : lowerCode(result[i]);
111 }
112 }
113 return String.fromCharCodes(result);
114 }
115 final words = splitTop(text, ' ');
116 for (var i = 0; i < words.length; i++) {
117 final word = words[i];
118 if (word.startsWith('{')) continue;
119 if (i > 0 && i + 1 < words.length && !words[i - 1].endsWith(':') && smallWords.contains(lower(letters(word)))) {
120 words[i] = lowerOutside(word);
121 continue;
122 }
123 var depth = 0;
124 for (var j = 0; j < word.length; j++) {
125 if (word[j] == '{') {
126 depth++;
127 } else if (word[j] == '}') {
128 depth--;
129 } else if (isLetter(word.codeUnitAt(j))) {
130 if (depth == 0) words[i] = word.substring(0, j) + String.fromCharCode(upperCode(word.codeUnitAt(j))) + word.substring(j + 1);
131 break;
132 }
133 }
134 }
135 return words.join(' ');
136}

139 if (names.length == 1) return names[0];
140 if (listB && names.length == 2) return '${names[0]} $conjunction ${names[1]}';
141 return '${names.sublist(0, names.length - 1).join(', ')}, $conjunction ${names.last}';

144class Person {
145 final String first, von, last, jr;
146 const Person(this.first, this.von, this.last, this.jr);
147
148 factory Person.parse(String text) {
149 final parts = splitTop(text, ',').map((part) => part.trim()).toList();
150 final words = splitTop(parts[0], ' ');
151 var firstLower = -1, lastLower = -1;
152 for (var i = 0; i + 1 < words.length; i++) {
153 if (isLower(words[i].codeUnitAt(0))) {
154 if (firstLower < 0) firstLower = i;
155 lastLower = i;
156 }
157 }
158 String join(int start, int end) => words.sublist(start, end).join(' ');
159 if (parts.length == 1) {
160 return firstLower < 0
161 ? Person(join(0, words.length - 1), '', words.last, '')
162 : Person(join(0, firstLower), join(firstLower, lastLower + 1), join(lastLower + 1, words.length), '');
163 }
164 return Person(parts.last, lastLower < 0 ? '' : join(0, lastLower + 1), join(lastLower + 1, words.length), parts.length == 3 ? parts[1] : '');
165 }
166
167 bool same(Person other) => first == other.first && von == other.von && last == other.last && jr == other.jr;
168 String get surname => plain(von.isEmpty ? last : '$von $last');
169 String get initials => plain(first).split(' ').where((word) => word.isNotEmpty).map((word) => word.split('-')
170 .map((piece) => '${String.fromCharCode(upperCode(piece.codeUnits.firstWhere(isLetter)))}.').join('-')).join(' ');
171
172 String format({required bool inverted, required bool useInitials}) {
173 final given = useInitials ? initials : plain(first);
174 var result = surname;
175 if (first.isNotEmpty) result = inverted ? '$result, $given' : '$given $result';
176 if (jr.isNotEmpty) result += ', ${plain(jr)}';
177 return result;
178 }

239 String get sortTitle {
240 final title = lower(value('title'));
241 for (final prefix in ['a ', 'an ', 'the ']) {
242 if (title.startsWith(prefix)) return title.substring(prefix.length);

248 final left = [lower(plain(a.primary.names[0].last)), a.value('year'), a.sortTitle, a.key];
249 final right = [lower(plain(b.primary.names[0].last)), b.value('year'), b.sortTitle, b.key];
250 for (var i = 0; i < left.length; i++) {
251 final order = left[i].compareTo(right[i]);

258class Database {
259 static const requiredFields = {
260 'article': ['author', 'title', 'journal', 'year'],
261 'book': ['author', 'title', 'publisher', 'year'],
262 'inproceedings': ['author', 'title', 'booktitle', 'year'],
263 'incollection': ['author', 'title', 'booktitle', 'publisher', 'year'],
264 'techreport': ['author', 'title', 'institution', 'year'],
265 'phdthesis': ['author', 'title', 'school', 'year'],
266 'misc': ['author', 'title', 'year'],
267 };
268 final macros = <String, String>{};
269 final entries = <Entry>[];
270 final byKey = <String, Entry>{};
271 final messages = <String>[];
272 var macroCount = 0;
273
274 Database(List<String> lines) {
275 final monthKeys = 'jan feb mar apr may jun jul aug sep oct nov dec'.split(' ');
276 for (var i = 0; i < monthKeys.length; i++) {
277 macros[monthKeys[i]] = monthNames[i];
278 }
279 for (var i = 0; i < lines.length; i++) {
280 final line = lines[i];
281 if (line.isEmpty || line.startsWith('%')) continue;
282 if (line.startsWith('@string{')) {
283 final separator = line.indexOf(' = ');
284 final name = line.substring(8, separator);
285 try {
286 final value = evaluate(line.substring(separator + 3, line.length - 1), i + 1);
287 if (macros.containsKey(name)) throw LoadError(i + 1, 'duplicate macro');
288 macros[name] = value;
289 macroCount++;
290 } on LoadError catch (error) {
291 report(name, error);
292 }
293 } else {
294 var end = i + 1;
295 while (lines[end] != '}') {
296 end++;
297 }
298 final brace = line.indexOf('{');
299 final type = line.substring(1, brace), key = line.substring(brace + 1, line.length - 1);
300 try {
301 readEntry(lines, i, end, type, key);
302 } on LoadError catch (error) {
303 report(key, error);
304 }
305 i = end;
306 }
307 }
308 assignSuffixes();
309 messages.add('LOADED entries=${entries.length} macros=$macroCount errors=${messages.length}');
310 }
311
312 void report(String name, LoadError error) => messages.add('ERROR line ${error.line}: $name: ${error.message}');
313
314 String evaluate(String expression, int line) {
315 var depth = 0;
316 for (final c in expression.codeUnits) {
317 if (c == 123) depth++;
318 if (c == 125) depth--;
319 if (depth < 0) throw LoadError(line, 'unbalanced braces');
320 }
321 if (depth != 0) throw LoadError(line, 'unbalanced braces');
322 final result = StringBuffer();
323 for (final term in expression.split(' # ')) {
324 if (term.startsWith('{') || term.startsWith('"')) {
325 result.write(term.substring(1, term.length - 1));
326 } else if (isDigit(term.codeUnitAt(0))) {
327 result.write(term);
328 } else if (macros.containsKey(term)) {
329 result.write(macros[term]);
330 } else {
331 throw LoadError(line, "unknown macro '$term'");
332 }
333 }
334 return result.toString();
335 }
336
337 void readEntry(List<String> lines, int start, int end, String type, String key) {
338 final required = requiredFields[type];
339 if (required == null) throw LoadError(start + 1, "unknown type '$type'");
340 if (byKey.containsKey(key)) throw LoadError(start + 1, 'duplicate key');
341 final fields = <String, String>{};
342 for (var i = start + 1; i < end; i++) {
343 final separator = lines[i].indexOf(' = ');
344 final name = lines[i].substring(2, separator);
345 var expression = lines[i].substring(separator + 3);
346 if (expression.endsWith(',')) expression = expression.substring(0, expression.length - 1);
347 final value = evaluate(expression, i + 1);
348 if (fields.containsKey(name)) throw LoadError(i + 1, "duplicate field '$name'");
349 fields[name] = normalize(value);
350 }
351 for (final field in required) {
352 if (!fields.containsKey(field) && !(type == 'book' && field == 'author' && fields.containsKey('editor'))) {
353 throw LoadError(end + 1, "missing field '$field'");
354 }
355 }
356 final entry = Entry(type, key, fields);
357 entries.add(entry);
358 byKey[key] = entry;
359 }
360
361 void assignSuffixes() {
362 final done = <Entry>{};
363 for (final entry in entries) {
364 if (done.contains(entry)) continue;
365 final group = entries.where((other) => other.primary.same(entry.primary) && other.fields['year'] == entry.fields['year']).toList()
366 ..sort((a, b) {
367 final order = a.sortTitle.compareTo(b.sortTitle);
368 return order == 0 ? a.key.compareTo(b.key) : order;
369 });
370 for (var i = 0; i < group.length; i++) {
371 done.add(group[i]);
372 if (group.length > 1) group[i].suffix = String.fromCharCode(97 + i);
373 }
374 }
375 }

434 static List<TemplatePart> parse(String text) {
435 var i = 0;
436 List<TemplatePart> read() {
437 final parts = <TemplatePart>[];
438 while (i < text.length && text[i] != '>') {
439 if (text[i] == '{') {
440 final end = text.indexOf('}', i);
441 parts.add(FieldPart(text.substring(i + 1, end)));
442 i = end + 1;
443 } else if (text[i] == '<') {
444 i++;
445 parts.add(OptionalPart(read()));
446 i++;
447 } else {
448 parts.add(LiteralPart(text[i++]));
449 }
450 }
451 return parts;
452 }
453 return read();
454 }

467 static String edition(String value) {
468 if (!digits(value)) return '$value ed.';
469 final number = int.parse(value);
470 final suffix = [11, 12, 13].contains(number % 100) ? 'th' : switch (number % 10) { 1 => 'st', 2 => 'nd', 3 => 'rd', _ => 'th' };
471 return '$value$suffix ed.';

474 static Map<String, String> values(Entry entry, Style style, bool repeated) {
475 final result = entry.fields.map((key, value) => MapEntry(key, plain(value)));
476 result['author'] = (repeated ? '---' : entry.primary.authors(style)) + editorSuffix(entry, style);
477 result['editor'] = entry.editors?.editors(style) ?? '';
478 result['eds'] = entry.editors == null ? '' : entry.editors!.single ? 'Ed.' : 'Eds.';
479 result['title'] = plain(titleCase(entry.fields['title']!, style));
480 result['year'] = entry.year(style);
481 result['pp'] = '';
482 if (entry.fields.containsKey('pages')) {
483 result['pages'] = entry.value('pages').replaceAll('--', '-');
484 result['pp'] = (result['pages']!.contains('-') ? 'pp. ' : 'p. ') + result['pages']!;

486 if (entry.fields.containsKey('edition')) result['edition'] = edition(entry.value('edition'));
487 final month = entry.value('month');
488 final index = monthNames.indexOf(month);
489 result['mon'] = index < 0 ? month : monthShort[index];
490 return result;

493 static String format(Entry entry, Style style, {int number = 0, bool repeated = false}) {
494 final fields = values(entry, style, repeated);
495 final result = StringBuffer();
496 var last = '';
497 void write(String text) {
498 result.write(text);
499 if (text.isNotEmpty) last = text[text.length - 1];
500 }
501 void render(List<TemplatePart> parts) {
502 for (final part in parts) {
503 switch (part) {
504 case FieldPart(:final name):
505 write(fields[name] ?? '');
506 case OptionalPart(:final parts):
507 if (parts.whereType<FieldPart>().every((field) => (fields[field.name] ?? '').isNotEmpty)) render(parts);
508 case LiteralPart(:final text):
509 if (text != '.' || last.isEmpty || !'.?!'.contains(last)) write(text);
510 }
511 }
512 }
513 final type = switch (entry.type) {
514 'article' => 0, 'book' => 1, 'inproceedings' || 'incollection' => 2,
515 'techreport' => 3, 'phdthesis' => 4, _ => 5,
516 };
517 render(parsed[style.index][type]);
518 return (style == Style.ieee ? '[$number] ' : '') + result.toString();
519 }

553 static String numberRanges(List<int> numbers) {
554 numbers.sort();
555 final parts = <String>[];
556 for (var i = 0; i < numbers.length;) {
557 var end = i;
558 while (end + 1 < numbers.length && numbers[end + 1] == numbers[end] + 1) {
559 end++;
560 }
561 if (end - i >= 2) {
562 parts.add('[${numbers[i]}]-[${numbers[end]}]');
563 } else {
564 for (var j = i; j <= end; j++) {
565 parts.add('[${numbers[j]}]');
566 }
567 }
568 i = end + 1;
569 }
570 return parts.join(', ');
571 }

573 static String citationText(Citation citation, Style style, Map<Entry, int> numbers) {
574 var entries = citation.entries;
575 if (citation.kind == 'citeauthor') return entries.map((entry) => shortName(entry, style, false)).join('; ');
576 if (citation.kind == 'citeyear') return entries.map((entry) => entry.year(style)).join(', ');
577 final parenthetical = citation.kind != 'citet';
578 if (style == Style.ieee) {
579 return parenthetical ? numberRanges(entries.map((entry) => numbers[entry]!).toList())
580 : entries.map((entry) => '${shortName(entry, style, false)} [${numbers[entry]}]').join('; ');
581 }
582 if (style == Style.mla) {
583 final text = entries.map((entry) => shortName(entry, style, parenthetical)).join('; ');
584 return parenthetical ? '($text)' : text;
585 }
586 if (parenthetical) entries = entries.toList()..sort(Entry.compare);
587 final groups = <String>[];
588 for (var i = 0; i < entries.length;) {
589 final name = shortName(entries[i], style, parenthetical);
590 var end = i + 1;
591 while (end < entries.length && shortName(entries[end], style, parenthetical) == name) {
592 end++;
593 }
594 final years = entries.sublist(i, end).map((entry) => entry.year(style)).join(', ');
595 groups.add(parenthetical ? name + (style == Style.apa ? ', ' : ' ') + years : '$name ($years)');
596 i = end;
597 }
598 final joined = groups.join('; ');
599 return parenthetical ? '($joined)' : joined;
600 }

602 String cite(String text, Style style) {
603 final citations = <Citation>[];
604 for (var start = text.indexOf('\\'); start >= 0;) {
605 final brace = text.indexOf('{', start);
606 final end = text.indexOf('}', brace);
607 final seen = <String>{};
608 final entries = <Entry>[];
609 for (final key in text.substring(brace + 1, end).split(',')) {
610 if (seen.add(key)) entries.add(lookup(key));
611 }
612 citations.add(Citation(start, end + 1, text.substring(start + 1, brace), entries));
613 start = text.indexOf('\\', end + 1);
614 }
615 for (final citation in citations) {
616 for (final entry in citation.entries) {
617 if (recorded.add(entry)) record.add(entry);
618 }
619 }
620 final numbers = ieeeNumbers();
621 final result = StringBuffer();
622 var position = 0;
623 for (final citation in citations) {
624 result.write(text.substring(position, citation.start));
625 result.write(citationText(citation, style, numbers));
626 position = citation.end;
627 }
628 result.write(text.substring(position));
629 return result.toString();
630 }

632 static String baseKey(Entry entry) {
633 final surname = lower(letters(plain(entry.primary.names[0].last)));
634 final year = String.fromCharCodes(entry.fields['year']!.codeUnits.where(isDigit));
635 var word = '';
636 for (final part in entry.value('title').split(' ')) {
637 final candidate = lower(letters(part));
638 if (candidate.isNotEmpty && !smallWords.contains(candidate)) {
639 word = candidate;
640 break;
641 }
642 }
643 return surname + year + word;
644 }

646 Iterable<String> keygen() sync* {
647 final bases = database.entries.map(baseKey).toList();
648 final counts = <String, int>{}, used = <String, int>{};
649 for (final base in bases) {
650 counts[base] = (counts[base] ?? 0) + 1;
651 }
652 for (var i = 0; i < bases.length; i++) {
653 var generated = bases[i];
654 if (counts[generated]! > 1) {
655 final index = used[generated] ?? 0;
656 used[generated] = index + 1;
657 generated += String.fromCharCode(97 + index);
658 }
659 yield '${database.entries[i].key} -> $generated';
660 }

663 static Iterable<String> warnings(Entry entry, Map<String, String> titles) sync* {
664 if (entry.fields.containsKey('pages')) {
665 final pages = entry.value('pages');
666 if (pages.contains('-') && !pages.contains('--')) yield 'pages use a single hyphen';
667 final parts = pages.split('-').where((part) => part.isNotEmpty).toList();
668 if (parts.length == 2 && parts.every(digits) && int.parse(parts[0]) > int.parse(parts[1])) yield 'pages are reversed';
669 }
670 final year = entry.value('year');
671 if (year.length != 4 || !digits(year)) yield 'year is not 4 digits';
672 final title = entry.value('title');
673 if (!title.codeUnits.any(isLower) && title.codeUnits.where(isUpper).length >= 2) yield 'title is all caps';
674 if (entry.type == 'article' && !entry.fields.containsKey('doi')) yield 'article has no doi';
675 final lowered = lower(title);
676 if (titles.containsKey(lowered)) {
677 yield 'same title as ${titles[lowered]}';
678 } else {
679 titles[lowered] = entry.key;
680 }
681 final unknown = entry.fields.keys.where((field) => !knownFields.contains(field)).toList()..sort();
682 for (final field in unknown) {
683 yield "unknown field '$field'";
684 }
685 }

700 yield '@${entry.type}{${entry.key},';
701 final unknown = entry.fields.keys.where((field) => !knownFields.contains(field)).toList()..sort();
702 final fields = [...knownFields.where(entry.fields.containsKey), ...unknown];

709 Iterable<String> stats() sync* {
710 yield 'ENTRIES ${database.entries.length}';
711 final types = <String, int>{}, years = <String, int>{};
712 final authors = <String>{};
713 final pairs = <(String, String), int>{};
714 for (final entry in database.entries) {
715 types[entry.type] = (types[entry.type] ?? 0) + 1;
716 final year = entry.value('year');
717 years[year] = (years[year] ?? 0) + 1;
718 if (entry.authors == null) continue;
719 final names = entry.authors!.names.map((name) => name.format(inverted: false, useInitials: false)).toSet().toList()..sort();
720 authors.addAll(names);
721 for (var i = 0; i < names.length; i++) {
722 for (var j = i + 1; j < names.length; j++) {
723 final pair = (names[i], names[j]);
724 pairs[pair] = (pairs[pair] ?? 0) + 1;
725 }
726 }
727 }
728 for (final type in types.keys.toList()..sort()) {
729 yield 'TYPE $type ${types[type]}';
730 }
731 for (final year in years.keys.toList()..sort()) {
732 yield 'YEAR $year ${years[year]}';
733 }
734 yield 'AUTHORS ${authors.length}';
735 if (pairs.isEmpty) {
736 yield 'PAIR none';
737 } else {
738 final ranked = pairs.entries.toList()..sort((a, b) {
739 final count = b.value.compareTo(a.value);
740 if (count != 0) return count;
741 final first = a.key.$1.compareTo(b.key.$1);
742 return first == 0 ? a.key.$2.compareTo(b.key.$2) : first;
743 });
744 final best = ranked[0];
745 yield 'PAIR ${best.key.$1} & ${best.key.$2} ${best.value}';
746 }
747 }

749 List<String> execute(String line) {
750 final firstSpace = line.indexOf(' ');
751 final command = firstSpace < 0 ? line : line.substring(0, firstSpace);
752 List<String> words;
753 if ((command == 'CITE' || command == 'SEARCH') && firstSpace >= 0) {
754 final secondSpace = line.indexOf(' ', firstSpace + 1);
755 words = secondSpace < 0 ? [command, line.substring(firstSpace + 1)]
756 : [command, line.substring(firstSpace + 1, secondSpace), line.substring(secondSpace + 1)];
757 } else {
758 words = line.split(' ');
759 }
760 final expected = switch (command) {
761 'FORMAT' || 'CITE' || 'SEARCH' => 2,
762 'BIBLIOGRAPHY' || 'EXPORT' => 1,
763 'KEYGEN' || 'CHECK' || 'STATS' => 0,
764 _ => throw CommandError("unknown command '$command'"),
765 };
766 if (command == 'FORMAT' ? words.length - 1 < expected : words.length - 1 != expected) {
767 throw const CommandError('wrong number of arguments');
768 }
769 var style = Style.apa;
770 if (command == 'FORMAT' || command == 'BIBLIOGRAPHY' || command == 'CITE') {
771 style = switch (words[1]) {
772 'APA' => Style.apa, 'IEEE' => Style.ieee, 'CHICAGO' => Style.chicago, 'MLA' => Style.mla,
773 _ => throw CommandError("unknown style '${words[1]}'"),
774 };
775 }
776 final output = <String>[];
777 switch (command) {
778 case 'FORMAT':
779 final numbers = ieeeNumbers();
780 for (final key in words.skip(2)) {
781 final entry = database.byKey[key];
782 output.add(entry == null ? "ERROR unknown key '$key'" : ReferenceFormatter.format(entry, style, number: numbers[entry]!));
783 }
784 case 'BIBLIOGRAPHY':
785 output.add(style == Style.mla ? 'Works Cited' : 'References');
786 final numbers = ieeeNumbers();
787 final entries = style == Style.ieee ? ieeeOrder() : (database.entries.toList()..sort(Entry.compare));
788 NameList? previous;
789 for (final entry in entries) {
790 final repeated = (style == Style.chicago || style == Style.mla) && (previous?.same(entry.primary) ?? false);
791 output.add(ReferenceFormatter.format(entry, style, number: numbers[entry]!, repeated: repeated));
792 previous = entry.primary;
793 }
794 case 'CITE':
795 output.add(cite(words[2], style));
796 case 'KEYGEN':
797 output.addAll(keygen());
798 case 'CHECK':
799 output.addAll(check());
800 case 'SEARCH':
801 final needle = lower(words[2]);
802 final matches = database.entries.where((entry) => entry.fields.containsKey(words[1]) && lower(entry.value(words[1])).contains(needle))
803 .map((entry) => entry.key).toList();
804 output.add(matches.isEmpty ? 'MATCHES 0' : 'MATCHES ${matches.length}: ${matches.join(' ')}');
805 case 'EXPORT':
806 output.addAll(export(lookup(words[1])));
807 case 'STATS':
808 output.addAll(stats());
809 }
810 return output;
811 }
