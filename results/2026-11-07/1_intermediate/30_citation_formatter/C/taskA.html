34typedef struct { char *data; size_t count, capacity; } Text;

36static void append_n(Text *text, const char *value, size_t length) {
37 while (text->count + length + 1 > text->capacity) text->data = grow(text->data, &text->capacity, 1);
38 memcpy(text->data + text->count, value, length);
39 text->count += length; text->data[text->count] = '\0';
40}
41static void append(Text *text, const char *value) { append_n(text, value, strlen(value)); }
42static void append_char(Text *text, char value) { append_n(text, &value, 1); }
43static char *finish(Text *text) { if (!text->data) append(text, ""); return text->data; }

60static bool lower(char c) { return c >= 'a' && c <= 'z'; }
61static bool upper(char c) { return c >= 'A' && c <= 'Z'; }
62static bool letter(char c) { return lower(c) || upper(c); }
63static bool digit(char c) { return c >= '0' && c <= '9'; }
64static char lowercase(char c) { return upper(c) ? (char)(c + 'a' - 'A') : c; }
65static char uppercase(char c) { return lower(c) ? (char)(c + 'A' - 'a') : c; }

76static char *trim(const char *text) {
77 size_t first = 0, last = strlen(text);
78 while (first < last && text[first] == ' ') ++first;
79 while (last > first && text[last - 1] == ' ') --last;
80 return slice(text, first, last);
81}
82static Strings split(const char *text, const char *delimiter, bool at_depth_zero, bool skip_empty) {
83 Strings result = {0};
84 size_t start = 0, i = 0, length = strlen(text), width = strlen(delimiter);
85 int depth = 0;
86 while (i < length) {
87 if ((!at_depth_zero || depth == 0) && strncmp(text + i, delimiter, width) == 0) {
88 if (i > start || !skip_empty) PUSH(&result, slice(text, start, i));
89 i += width; start = i; continue;
90 }
91 if (text[i] == '{') ++depth;
92 if (text[i] == '}') --depth;
93 ++i;
94 }
95 if (i > start || !skip_empty) PUSH(&result, slice(text, start, i));
96 return result;
97}
98static char *join_range(Strings values, size_t start, size_t end, const char *separator) {
99 Text result = {0};
100 for (size_t i = start; i < end; ++i) { if (i > start) append(&result, separator); append(&result, values.data[i]); }
101 return finish(&result);
102}

152static Name parse_name(const char *text) {
153 Strings parts = split(text, ",", true, false);
154 for (size_t i = 0; i < parts.count; ++i) parts.data[i] = trim(parts.data[i]);
155 Strings words = split(parts.data[0], " ", true, true);
156 size_t first_lower = words.count, last_lower = words.count;
157 for (size_t i = 0; i + 1 < words.count; ++i) if (lower(words.data[i][0])) {
158 if (first_lower == words.count) first_lower = i;
159 last_lower = i;
160 }
161 Name name = {"", "", "", ""};
162 if (parts.count == 1) {
163 if (first_lower == words.count) {
164 name.first = join_range(words, 0, words.count - 1, " ");
165 name.last = words.data[words.count - 1];
166 } else {
167 name.first = join_range(words, 0, first_lower, " ");
168 name.von = join_range(words, first_lower, last_lower + 1, " ");
169 name.last = join_range(words, last_lower + 1, words.count, " ");
170 }
171 } else {
172 if (last_lower == words.count) name.last = join(words, " ");
173 else { name.von = join_range(words, 0, last_lower + 1, " "); name.last = join_range(words, last_lower + 1, words.count, " "); }
174 name.first = parts.data[parts.count - 1];
175 if (parts.count == 3) name.jr = parts.data[1];
176 }
177 return name;
178}
179static NameList parse_names(const char *text) {

182 for (size_t i = 0; i < people.count; ++i) {
183 if (!strcmp(people.data[i], "others")) result.et_al = true;
184 else PUSH(&result, parse_name(people.data[i]));
185 }
186 return result;
187}
188static bool same_names(const NameList *a, const NameList *b) {
189 if (a->count != b->count || a->et_al != b->et_al) return false;
190 for (size_t i = 0; i < a->count; ++i) {
191 Name x = a->data[i], y = b->data[i];
192 if (strcmp(x.first, y.first) || strcmp(x.von, y.von) || strcmp(x.last, y.last) || strcmp(x.jr, y.jr)) return false;
193 }
194 return true;
195}

197static char *initials(Name name) {
198 Strings words = split(plain(name.first), " ", false, true), result = {0};
199 for (size_t i = 0; i < words.count; ++i) {
200 Strings pieces = split(words.data[i], "-", false, false), parts = {0};
201 for (size_t j = 0; j < pieces.count; ++j) {
202 const char *piece = pieces.data[j];
203 while (!letter(*piece)) ++piece;
204 PUSH(&parts, format("%c.", uppercase(*piece)));
205 }
206 PUSH(&result, join(parts, "-"));
207 }
208 return join(result, " ");

252static char *title_case(const char *title, Style style) {
253 char *result = copy_string(title);
254 if (style == IEEE) return result;
255 if (style == APA) {
256 int depth = 0;
257 for (size_t i = 0; result[i]; ++i) {
258 if (result[i] == '{') ++depth;
259 else if (result[i] == '}') --depth;
260 else if (!depth && letter(result[i])) result[i] = i == 0 || (i >= 2 && title[i - 2] == ':' && title[i - 1] == ' ') ? uppercase(result[i]) : lowercase(result[i]);
261 }
262 return result;
263 }
264 Strings words = split(title, " ", true, true);
265 for (size_t i = 0; i < words.count; ++i) {
266 char *word = words.data[i];
267 if (word[0] == '{') continue;
268 bool make_lower = i > 0 && i + 1 < words.count && words.data[i - 1][strlen(words.data[i - 1]) - 1] != ':' && small_word(letters_only(word));
269 int depth = 0;
270 bool found_letter = false;
271 for (size_t j = 0; word[j]; ++j) {
272 if (word[j] == '{') ++depth;
273 else if (word[j] == '}') --depth;
274 else if (letter(word[j])) {
275 if (!depth && make_lower) word[j] = lowercase(word[j]);
276 else if (!depth && !found_letter) word[j] = uppercase(word[j]);
277 found_letter = true;
278 }
279 }
280 }
281 return join(words, " ");
282}

284static char *evaluate(const char *expression, char **error) {
285 int depth = 0;
286 for (const char *p = expression; *p; ++p) {
287 if (*p == '{') ++depth;
288 if (*p == '}') --depth;
289 if (depth < 0) { *error = "unbalanced braces"; return NULL; }
290 }
291 if (depth) { *error = "unbalanced braces"; return NULL; }
292 Strings terms = split(expression, " # ", false, false);
293 Text result = {0};
294 for (size_t i = 0; i < terms.count; ++i) {
295 char *term = terms.data[i];
296 if (*term == '{' || *term == '"') append_n(&result, term + 1, strlen(term) - 2);
297 else if (digit(*term)) append(&result, term);
298 else {
299 char *value = field(macros, term);
300 if (!value) { *error = format("unknown macro '%s'", term); return NULL; }
301 append(&result, value);
302 }
303 }
304 return finish(&result);
305}
306static size_t accent(const char *text, size_t length, size_t at, char *result) {
307 if (at >= length || text[at] != '\\' || at + 1 >= length) return 0;
308 char command = text[at + 1];
309 if (command == '&') { *result = '&'; return 2; }
310 bool symbol = strchr("'\"`^~=.", command) != NULL;
311 if (symbol && at + 2 < length && letter(text[at + 2])) { *result = text[at + 2]; return 3; }
312 if ((symbol || strchr("cuvH", command)) && at + 4 < length && text[at + 2] == '{' && letter(text[at + 3]) && text[at + 4] == '}') {
313 *result = text[at + 3]; return 5;
314 }
315 return 0;
316}
317static char *normalize(const char *value) {
318 Text accents = {0}, result = {0};
319 size_t length = strlen(value);
320 for (size_t i = 0; i < length;) {
321 char replacement;
322 size_t width = value[i] == '{' ? accent(value, length, i + 1, &replacement) : 0;
323 if (width && i + width + 1 < length && value[i + width + 1] == '}') {
324 append_char(&accents, replacement); i += width + 2;
325 } else if ((width = accent(value, length, i, &replacement))) {
326 append_char(&accents, replacement); i += width;
327 } else append_char(&accents, value[i++]);
328 }
329 bool space = false;
330 for (const char *p = finish(&accents); *p; ++p) {
331 if (*p == '~' || *p == ' ') space = result.count > 0;
332 else {
333 if (space) append_char(&result, ' ');
334 append_char(&result, *p); space = false;
335 }
336 }
337 return finish(&result);
338}

348static void read_database(void) {
349 for (int i = 0; i < 12; ++i) PUSH(&macros, ((Field){copy_string(month_macros[i]), copy_string(months[i])}));
350 char line[604];
351 int line_number = 0;
352 Entry draft = {0};
353 bool in_entry = false, invalid = false;
354 while (read_line(line, sizeof line) && strcmp(line, "%%")) {
355 ++line_number;
356 if (in_entry) {
357 if (!strcmp(line, "}")) {
358 if (!invalid) {
359 for (int i = 0; i < 5 && required[draft.type][i]; ++i) {
360 const char *name = required[draft.type][i];
361 if (field(draft.fields, name) || (draft.type == 1 && !strcmp(name, "author") && field(draft.fields, "editor"))) continue;
362 database_error(line_number, draft.key, format("missing field '%s'", name)); invalid = true; break;
363 }
364 if (!invalid) entries[entry_count++] = draft;
365 }
366 in_entry = false;
367 } else if (!invalid) {
368 char *equals = strstr(line, " = "), *name = slice(line, 2, (size_t)(equals - line));
369 size_t end = strlen(line);
370 if (line[end - 1] == ',') --end;
371 char *error = NULL, *value = evaluate(slice(line, (size_t)(equals - line) + 3, end), &error);
372 if (!error && field(draft.fields, name)) error = format("duplicate field '%s'", name);
373 if (error) { database_error(line_number, draft.key, error); invalid = true; }
374 else PUSH(&draft.fields, ((Field){name, normalize(value)}));
375 }
376 } else if (!*line || *line == '%') continue;
377 else if (!strncmp(line, "@string{", 8)) {
378 char *equals = strstr(line, " = "), *name = slice(line, 8, (size_t)(equals - line));
379 char *error = NULL, *value = evaluate(slice(line, (size_t)(equals - line) + 3, strlen(line) - 1), &error);
380 if (!error && field(macros, name)) error = "duplicate macro";
381 if (error) database_error(line_number, name, error);
382 else { PUSH(&macros, ((Field){name, value})); ++defined_macros; }
383 } else {
384 char *brace = strchr(line, '{'), *type = slice(line, 1, (size_t)(brace - line));
385 draft = (Entry){.key = slice(line, (size_t)(brace - line) + 1, strlen(line) - 1), .type = -1};
386 for (int i = 0; i < 7; ++i) if (!strcmp(type, type_names[i])) draft.type = i;
387 char *error = draft.type < 0 ? format("unknown type '%s'", type) : find_entry(draft.key) >= 0 ? "duplicate key" : NULL;
388 invalid = error != NULL; in_entry = true;
389 if (error) database_error(line_number, draft.key, error);
390 }
391 }
392 printf("LOADED entries=%d macros=%d errors=%d\n", entry_count, defined_macros, database_errors);
393}
394static void prepare_entries(void) {
395 for (int i = 0; i < entry_count; ++i) {
396 Entry *entry = &entries[i];
397 char *author = field(entry->fields, "author"), *editor = field(entry->fields, "editor");
398 entry->has_author = author != NULL; entry->has_editor = editor != NULL;
399 if (author) entry->authors = parse_names(author);
400 if (editor) entry->editors = parse_names(editor);
401 Name lead = primary(entry)->data[0];
402 entry->sort_last = lower_string(plain(lead.last));
403 entry->year = plain(field(entry->fields, "year"));
404 entry->plain_title = plain(field(entry->fields, "title"));
405 entry->lower_title = lower_string(entry->plain_title);
406 entry->sort_title = entry->lower_title;
407 if (!strncmp(entry->sort_title, "a ", 2)) entry->sort_title += 2;
408 else if (!strncmp(entry->sort_title, "an ", 3)) entry->sort_title += 3;
409 else if (!strncmp(entry->sort_title, "the ", 4)) entry->sort_title += 4;
410 Text base = {0};
411 append(&base, letters_only(plain(lead.last)));
412 for (const char *p = field(entry->fields, "year"); *p; ++p) if (digit(*p)) append_char(&base, *p);
413 Strings words = split(entry->plain_title, " ", false, true);
414 for (size_t j = 0; j < words.count; ++j) {
415 char *word = letters_only(words.data[j]);
416 if (*word && !small_word(word)) { append(&base, word); break; }
417 }
418 entry->base_key = finish(&base);
419 }
420 bool assigned[400] = {0};
421 for (int i = 0; i < entry_count; ++i) if (!assigned[i]) {
422 int group[400], count = 0;
423 for (int j = i; j < entry_count; ++j) if (same_names(primary(&entries[i]), primary(&entries[j])) && !strcmp(field(entries[i].fields, "year"), field(entries[j].fields, "year"))) {
424 group[count++] = j; assigned[j] = true;
425 }
426 if (count > 1) {
427 qsort(group, (size_t)count, sizeof *group, title_compare);
428 for (int j = 0; j < count; ++j) entries[group[j]].suffix = (char)('a' + j);
429 }
430 }
431}

438static char *render_names(NameList *names, Style style, bool editor) {
439 Strings forms = {0};
440 for (size_t i = 0; i < names->count; ++i) {
441 bool inverted = !editor && (style == APA || (style != IEEE && i == 0));
442 PUSH(&forms, name_text(names->data[i], inverted, style == APA || style == IEEE));
443 }
444 if (editor) return names->et_al ? format("%s et al.", forms.data[0]) : list_names(forms, style == APA ? "&" : "and", true);
445 if (style == APA) {
446 if (names->et_al) return format("%s, et al.", join(forms, ", "));
447 if (names->count >= 21) return format("%s, . . . %s", join_range(forms, 0, 19, ", "), forms.data[forms.count - 1]);
448 return list_names(forms, "&", false);
449 }
450 if (style == IEEE) return names->et_al || names->count >= 7 ? format("%s et al.", forms.data[0]) : list_names(forms, "and", true);
451 if (style == CHICAGO) return names->et_al || names->count >= 11 ? format("%s, et al.", join_range(forms, 0, forms.count < 7 ? forms.count : 7, ", ")) : list_names(forms, "and", false);
452 return names->et_al || names->count >= 3 ? format("%s, et al.", forms.data[0]) : list_names(forms, "and", false);
453}

471 return format("%d%s ed.", number, suffix);
472}

504static int placeholder_index(const char *name) {
505 int index = known_index(name);
506 return index >= 0 ? index : !strcmp(name, "eds") ? 18 : !strcmp(name, "pp") ? 19 : 20;

508static size_t optional_end(const char *pattern, size_t start) {
509 int depth = 1;
510 size_t end = start + 1;
511 for (; depth; ++end) { if (pattern[end] == '<') ++depth; if (pattern[end] == '>') --depth; }
512 return end - 1;
513}
514static bool optional_present(const char *pattern, size_t start, size_t end, char **values) {
515 for (size_t i = start; i < end; ++i) {
516 if (pattern[i] == '<') i = optional_end(pattern, i);
517 else if (pattern[i] == '{') {
518 size_t close = (size_t)(strchr(pattern + i, '}') - pattern);
519 if (!*values[placeholder_index(slice(pattern, i + 1, close))]) return false;
520 i = close;
521 }
522 }
523 return true;
524}
525static void render_template(const char *pattern, size_t start, size_t end, char **values, Text *output) {
526 for (size_t i = start; i < end; ++i) {
527 if (pattern[i] == '<') {
528 size_t close = optional_end(pattern, i);
529 if (optional_present(pattern, i + 1, close, values)) render_template(pattern, i + 1, close, values, output);
530 i = close;
531 } else if (pattern[i] == '{') {
532 size_t close = (size_t)(strchr(pattern + i, '}') - pattern);
533 append(output, values[placeholder_index(slice(pattern, i + 1, close))]); i = close;
534 } else if (pattern[i] != '.' || !output->count || !strchr(".?!", output->data[output->count - 1])) append_char(output, pattern[i]);
535 }
536}
537static char *reference(Entry *entry, Style style, int number, bool repeated) {
538 char *values[21];
539 for (int i = 0; i < 21; ++i) values[i] = "";
540 for (int i = 0; i < 18; ++i) {
541 char *value = field(entry->fields, known_fields[i]);
542 if (value) values[i] = plain(value);
543 }
544 NameList *names = primary(entry);
545 values[0] = repeated ? "---" : render_names(names, style, false);
546 if (!entry->has_author) {
547 static const char *const annotations[4][2] = {{" (Ed.)", " (Eds.)"}, {", Ed.", ", Eds."}, {", ed.", ", eds."}, {", editor", ", editors"}};
548 values[0] = format("%s%s", values[0], annotations[style][names->count != 1 || names->et_al]);
549 }
550 if (entry->has_editor) {
551 values[1] = render_names(&entry->editors, style, true);
552 values[18] = entry->editors.count == 1 && !entry->editors.et_al ? "Ed." : "Eds.";
553 }
554 values[2] = plain(title_case(field(entry->fields, "title"), style));
555 values[15] = year_label(entry, style);
556 char *pages = field(entry->fields, "pages"), *edition = field(entry->fields, "edition"), *month = field(entry->fields, "month");
557 if (pages) { values[8] = page_text(pages); values[19] = format("%s %s", strchr(values[8], '-') ? "pp." : "p.", values[8]); }
558 if (edition) values[5] = edition_text(edition);
559 if (month) {
560 values[20] = plain(month);
561 for (int i = 0; i < 12; ++i) if (!strcmp(values[20], months[i])) { values[20] = (char *)month_short[i]; break; }
562 }
563 int type = entry->type == 3 ? 2 : entry->type > 3 ? entry->type - 1 : entry->type;
564 const char *pattern = templates[style][type];
565 Text result = {0};
566 if (style == IEEE) append(&result, format("[%d] ", number));
567 render_template(pattern, 0, strlen(pattern), values, &result);
568 return finish(&result);
569}

571static void ieee_order(int *order, int *numbers) {
572 bool cited[400] = {0};
573 for (int i = 0; i < citation_count; ++i) { order[i] = citation_record[i]; cited[order[i]] = true; }
574 int count = citation_count;
575 for (int i = 0; i < entry_count; ++i) if (!cited[i]) order[count++] = i;
576 qsort(order + citation_count, (size_t)(count - citation_count), sizeof *order, key_compare);
577 for (int i = 0; i < count; ++i) numbers[order[i]] = i + 1;
578}
579static char *short_name(Entry *entry, Style style, bool parenthetical) {
580 NameList *names = primary(entry);
581 if (names->et_al || names->count >= (style == CHICAGO ? 4u : 3u)) return format("%s et al.", surname(names->data[0]));
582 Strings forms = {0};
583 for (size_t i = 0; i < names->count; ++i) PUSH(&forms, surname(names->data[i]));
584 return list_names(forms, style == APA && parenthetical ? "&" : "and", true);
585}
586typedef struct { int *data; size_t count, capacity; } Indices;
587typedef enum { PARENTHETICAL, NARRATIVE, AUTHOR, YEAR } CitationKind;
588typedef struct { size_t start, end; CitationKind kind; Indices keys; } Citation;
589typedef struct { Citation *data; size_t count, capacity; } Citations;
590static int number_compare(const void *a, const void *b) { return *(const int *)a - *(const int *)b; }
591static char *citation_text(Citation citation, Style style, const int *numbers) {
592 bool parenthetical = citation.kind == PARENTHETICAL;
593 Strings parts = {0};
594 if (citation.kind == AUTHOR || citation.kind == YEAR || style == MLA || (style == IEEE && !parenthetical)) {
595 for (size_t i = 0; i < citation.keys.count; ++i) {
596 int index = citation.keys.data[i];
597 Entry *entry = &entries[index];
598 char *value = citation.kind == YEAR ? year_label(entry, style) : short_name(entry, style, false);
599 if (style == IEEE && citation.kind == NARRATIVE) value = format("%s [%d]", value, numbers[index]);
600 PUSH(&parts, value);
601 }
602 char *text = join(parts, citation.kind == YEAR ? ", " : "; ");
603 return style == MLA && parenthetical ? format("(%s)", text) : text;
604 }
605 if (style == IEEE) {
606 int sorted[400];
607 for (size_t i = 0; i < citation.keys.count; ++i) sorted[i] = numbers[citation.keys.data[i]];
608 qsort(sorted, citation.keys.count, sizeof *sorted, number_compare);
609 for (size_t i = 0; i < citation.keys.count;) {
610 size_t end = i + 1;
611 while (end < citation.keys.count && sorted[end] == sorted[end - 1] + 1) ++end;
612 if (end - i >= 3) PUSH(&parts, format("[%d]-[%d]", sorted[i], sorted[end - 1]));
613 else for (size_t j = i; j < end; ++j) PUSH(&parts, format("[%d]", sorted[j]));
614 i = end;
615 }
616 return join(parts, ", ");
617 }
618 if (parenthetical) qsort(citation.keys.data, citation.keys.count, sizeof *citation.keys.data, bibliography_compare);
619 for (size_t i = 0; i < citation.keys.count;) {
620 char *name = short_name(&entries[citation.keys.data[i]], style, parenthetical);
621 Strings years = {0};
622 size_t end = i;
623 do {
624 PUSH(&years, year_label(&entries[citation.keys.data[end]], style)); ++end;
625 } while (end < citation.keys.count && !strcmp(name, short_name(&entries[citation.keys.data[end]], style, parenthetical)));
626 char *value = parenthetical ? format("%s%s%s", name, style == APA ? ", " : " ", join(years, ", ")) : format("%s (%s)", name, join(years, ", "));
627 PUSH(&parts, value); i = end;
628 }
629 char *text = join(parts, "; ");
630 return parenthetical ? format("(%s)", text) : text;
631}
632static void cite(Style style, const char *text) {
633 Citations citations = {0};
634 for (size_t i = 0; text[i]; ++i) if (text[i] == '\\') {
635 size_t brace = (size_t)(strchr(text + i, '{') - text), end = (size_t)(strchr(text + brace, '}') - text);
636 char *command = slice(text, i + 1, brace);
637 Citation citation = {.start = i, .end = end + 1, .kind = !strcmp(command, "citet") ? NARRATIVE : !strcmp(command, "citeauthor") ? AUTHOR : !strcmp(command, "citeyear") ? YEAR : PARENTHETICAL};
638 Strings keys = split(slice(text, brace + 1, end), ",", false, false);
639 bool seen[400] = {0};
640 for (size_t j = 0; j < keys.count; ++j) {
641 int index = find_entry(keys.data[j]);
642 if (index < 0) { printf("ERROR unknown key '%s'\n", keys.data[j]); return; }
643 if (!seen[index]) { PUSH(&citation.keys, index); seen[index] = true; }
644 }
645 PUSH(&citations, citation); i = end;
646 }
647 bool recorded[400] = {0};
648 for (int i = 0; i < citation_count; ++i) recorded[citation_record[i]] = true;
649 for (size_t i = 0; i < citations.count; ++i) for (size_t j = 0; j < citations.data[i].keys.count; ++j) {
650 int index = citations.data[i].keys.data[j];
651 if (!recorded[index]) { citation_record[citation_count++] = index; recorded[index] = true; }
652 }
653 int order[400], numbers[400];
654 ieee_order(order, numbers);
655 Text result = {0};
656 size_t position = 0;
657 for (size_t i = 0; i < citations.count; ++i) {
658 Citation citation = citations.data[i];
659 append_n(&result, text + position, citation.start - position);
660 append(&result, citation_text(citation, style, numbers)); position = citation.end;
661 }
662 append(&result, text + position); puts(finish(&result));
663}

667 if (result.count) qsort(result.data, result.count, sizeof *result.data, string_compare);
668 return result;
669}

679static void check(void) {
680 size_t count = 0;
681 for (int i = 0; i < entry_count; ++i) {
682 Entry *entry = &entries[i];
683 Strings warnings = {0};
684 char *pages = field(entry->fields, "pages");
685 if (pages) {
686 pages = plain(pages);
687 if (strchr(pages, '-') && !strstr(pages, "--")) PUSH(&warnings, "pages use a single hyphen");
688 Strings parts = split(pages, "-", false, true);
689 if (parts.count == 2 && all_digits(parts.data[0]) && all_digits(parts.data[1]) && strtol(parts.data[0], NULL, 10) > strtol(parts.data[1], NULL, 10)) PUSH(&warnings, "pages are reversed");
690 }
691 if (strlen(entry->year) != 4 || !all_digits(entry->year)) PUSH(&warnings, "year is not 4 digits");
692 size_t lower_count = 0, upper_count = 0;
693 for (const char *p = entry->plain_title; *p; ++p) { lower_count += lower(*p); upper_count += upper(*p); }
694 if (!lower_count && upper_count >= 2) PUSH(&warnings, "title is all caps");
695 if (entry->type == 0 && !field(entry->fields, "doi")) PUSH(&warnings, "article has no doi");
696 for (int j = 0; j < i; ++j) if (!strcmp(entry->lower_title, entries[j].lower_title)) { PUSH(&warnings, format("same title as %s", entries[j].key)); break; }
697 Strings unknown = unknown_fields(entry);
698 for (size_t j = 0; j < unknown.count; ++j) PUSH(&warnings, format("unknown field '%s'", unknown.data[j]));
699 for (size_t j = 0; j < warnings.count; ++j) printf("WARN %s: %s\n", entry->key, warnings.data[j]);
700 count += warnings.count;
701 }
702 printf("CHECK warnings=%zu\n", count);
703}
704static void export_entry(Entry *entry) {
705 Strings names = {0};
706 for (int i = 0; i < 18; ++i) if (field(entry->fields, known_fields[i])) PUSH(&names, (char *)known_fields[i]);
707 Strings unknown = unknown_fields(entry);
708 for (size_t i = 0; i < unknown.count; ++i) PUSH(&names, unknown.data[i]);
709 printf("@%s{%s,\n", type_names[entry->type], entry->key);
710 for (size_t i = 0; i < names.count; ++i) printf(" %s = {%s}%s\n", names.data[i], field(entry->fields, names.data[i]), i + 1 == names.count ? "" : ",");
711 puts("}");
712}
713typedef struct { char *a, *b; } Pair;
714typedef struct { Pair *data; size_t count, capacity; } Pairs;
715static int pair_compare(const void *left, const void *right) {
716 const Pair *a = left, *b = right;
717 int comparison = strcmp(a->a, b->a);
718 return comparison ? comparison : strcmp(a->b, b->b);
719}
720static void stats(void) {
721 printf("ENTRIES %d\n", entry_count);
722 Strings types = {0}, years = {0}, authors = {0};
723 Pairs pairs = {0};
724 for (int i = 0; i < entry_count; ++i) {
725 Entry *entry = &entries[i];
726 add_unique(&types, (char *)type_names[entry->type]); add_unique(&years, entry->year);
727 if (entry->has_author) {
728 Strings names = {0};
729 for (size_t j = 0; j < entry->authors.count; ++j) {
730 char *name = name_text(entry->authors.data[j], false, false);
731 add_unique(&names, name); add_unique(&authors, name);
732 }
733 qsort(names.data, names.count, sizeof *names.data, string_compare);
734 for (size_t a = 0; a < names.count; ++a) for (size_t b = a + 1; b < names.count; ++b) PUSH(&pairs, ((Pair){names.data[a], names.data[b]}));
735 }
736 }
737 if (types.count) qsort(types.data, types.count, sizeof *types.data, string_compare);
738 if (years.count) qsort(years.data, years.count, sizeof *years.data, string_compare);
739 for (size_t i = 0; i < types.count; ++i) {
740 int count = 0;
741 for (int j = 0; j < entry_count; ++j) count += !strcmp(types.data[i], type_names[entries[j].type]);
742 printf("TYPE %s %d\n", types.data[i], count);
743 }
744 for (size_t i = 0; i < years.count; ++i) {
745 int count = 0;
746 for (int j = 0; j < entry_count; ++j) count += !strcmp(years.data[i], entries[j].year);
747 printf("YEAR %s %d\n", years.data[i], count);
748 }
749 printf("AUTHORS %zu\n", authors.count);
750 if (!pairs.count) { puts("PAIR none"); return; }
751 qsort(pairs.data, pairs.count, sizeof *pairs.data, pair_compare);
752 size_t best = 0, best_count = 0;
753 for (size_t i = 0; i < pairs.count;) {
754 size_t end = i + 1;
755 while (end < pairs.count && !pair_compare(&pairs.data[i], &pairs.data[end])) ++end;
756 if (end - i > best_count) { best = i; best_count = end - i; }
757 i = end;
758 }
759 printf("PAIR %s & %s %zu\n", pairs.data[best].a, pairs.data[best].b, best_count);
760}
761static Strings command_words(const char *line) {
762 size_t length = strlen(line), first = strcspn(line, " ");
763 char *command = slice(line, 0, first);
764 if (strcmp(command, "CITE") && strcmp(command, "SEARCH")) return split(line, " ", false, false);
765 Strings words = {0}; PUSH(&words, command);
766 if (first == length) return words;
767 size_t second = first + 1 + strcspn(line + first + 1, " ");
768 PUSH(&words, slice(line, first + 1, second));
769 if (second < length) PUSH(&words, slice(line, second + 1, length));
770 return words;
771}
772static void command(const char *line) {
773 static const char *const commands[] = {"FORMAT", "BIBLIOGRAPHY", "CITE", "KEYGEN", "CHECK", "SEARCH", "EXPORT", "STATS"};
774 static const int arities[] = {-1, 1, 2, 0, 0, 2, 1, 0};
775 static const char *const styles[] = {"APA", "IEEE", "CHICAGO", "MLA"};
776 Strings words = command_words(line);
777 int kind = -1;
778 for (int i = 0; i < 8; ++i) if (!strcmp(words.data[0], commands[i])) kind = i;
779 if (kind < 0) { printf("ERROR unknown command '%s'\n", words.data[0]); return; }
780 if (kind == 0 ? words.count < 3 : words.count != (size_t)arities[kind] + 1) { puts("ERROR wrong number of arguments"); return; }
781 Style style = APA;
782 if (kind <= 2) {
783 int index = -1;
784 for (int i = 0; i < 4; ++i) if (!strcmp(words.data[1], styles[i])) index = i;
785 if (index < 0) { printf("ERROR unknown style '%s'\n", words.data[1]); return; }
786 style = (Style)index;
787 }
788 if (kind == 0 || kind == 1) {
789 int order[400], numbers[400]; ieee_order(order, numbers);
790 if (kind == 0) {
791 for (size_t i = 2; i < words.count; ++i) {
792 int index = find_entry(words.data[i]);
793 if (index < 0) printf("ERROR unknown key '%s'\n", words.data[i]);
794 else puts(reference(&entries[index], style, numbers[index], false));
795 }
796 } else {
797 puts(style == MLA ? "Works Cited" : "References");
798 if (style != IEEE) qsort(order, (size_t)entry_count, sizeof *order, bibliography_compare);
799 for (int i = 0; i < entry_count; ++i) {
800 bool repeated = i > 0 && (style == CHICAGO || style == MLA) && same_names(primary(&entries[order[i]]), primary(&entries[order[i - 1]]));
801 puts(reference(&entries[order[i]], style, numbers[order[i]], repeated));
802 }
803 }
804 } else if (kind == 2) cite(style, words.data[2]);
805 else if (kind == 3) keygen();
806 else if (kind == 4) check();
807 else if (kind == 5) {
808 Strings matches = {0};
809 char *needle = lower_string(words.data[2]);
810 for (int i = 0; i < entry_count; ++i) {
811 char *value = field(entries[i].fields, words.data[1]);
812 if (value && strstr(lower_string(plain(value)), needle)) PUSH(&matches, entries[i].key);
813 }
814 if (matches.count) printf("MATCHES %zu: %s\n", matches.count, join(matches, " "));
815 else puts("MATCHES 0");
816 } else if (kind == 6) {
817 int index = find_entry(words.data[1]);
818 if (index < 0) printf("ERROR unknown key '%s'\n", words.data[1]);
819 else export_entry(&entries[index]);
820 } else stats();
821}
