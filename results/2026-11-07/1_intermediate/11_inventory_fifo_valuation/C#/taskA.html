1using System;

6class Program {
7 static readonly string[] CATEGORIES={"received","returned","adjusted-in","adjusted-out","shipped","written-down","expired","recalled"};
8 class Fault : Exception { public Fault(string reason) : base(reason) {} }
9 static void error(bool condition, string reason) { if (condition) throw new Fault(reason); }
10 static long parseMoney(string s) {
11 return long.Parse(s.Replace(".",""));
12 }
13 static string money(long n) {
14 long a=Math.Abs(n);
15 return (n<0 ? "-" : "")+a/100+"."+(a%100).ToString("D2",CultureInfo.InvariantCulture);
16 }
17 static string signed(long n) { return (n>0 ? "+" : "")+n; }
18 static string signedMoney(long n) { return (n>0 ? "+" : "")+money(n); }
19 static string margin(long profit, long revenue) {
20 if (revenue==0) return "n/a";
21 long q=Math.Abs(profit)*1000/revenue;
22 return (profit<0 && q>0 ? "-" : "")+q/10+"."+q%10+"%";
23 }
24 record Card(long day, string kind, string warehouse, long quantity, long value);
25 class Item {
26 public string code,name;
27 public long level;
28 public long? lastCost;
29 public List<Card> cards=new();
30 public Item(string code, long level, string name) {
31 this.code=code;
32 this.level=level;
33 this.name=name;
34 }
35 }
36 class Warehouse {
37 public string code;
38 public Dictionary<string,Dictionary<int,Stock>> stock=new();
39 public Warehouse(string code) { this.code=code; }
40 }
41 class Lot {
42 public int number;
43 public Item item;
44 public long day,initialQuantity,initialCost,removed;
45 public string warehouse,kind;
46 public Lot(int number, Item item, long day, long quantity, long cost, string warehouse, string kind) {
47 this.number=number;
48 this.item=item;
49 this.day=day;
50 initialQuantity=quantity;
51 initialCost=cost;
52 this.warehouse=warehouse;
53 this.kind=kind;
54 }
55 }
56 class Stock {
57 public Lot lot;
58 public long quantity,cost;
59 public Stock(Lot lot, long quantity, long cost) {
60 this.lot=lot;
61 this.quantity=quantity;
62 this.cost=cost;
63 }
64 public long value() { return quantity*cost; }
65 }
66 record Part(Lot lot, long quantity, long cost);
67 record Issue(List<Part> parts, long cost);
68 class Order {
69 public int number;
70 public Warehouse warehouse;
71 public Item item;
72 public long day,price,ordered,shipped,backorder,cancelled,returned,revenue,cogs;
73 public List<Part> parts=new();
74 public Order(int number, Warehouse warehouse, Item item, long day, long quantity, long price) {
75 this.number=number;
76 this.warehouse=warehouse;
77 this.item=item;
78 this.day=day;
79 ordered=quantity;
80 backorder=quantity;
81 this.price=price;
82 }
83 }
84 class Inventory {
85 public long day=1;
86 public SortedDictionary<string,Warehouse> warehouses=new(StringComparer.Ordinal);
87 public SortedDictionary<string,Item> items=new(StringComparer.Ordinal);
88 public List<Lot> lots=new();
89 public List<Order> orders=new();
90 public long[][] totals=Enumerable.Range(0,8).Select(_ => new long[2]).ToArray();
91 public List<string> output;
92 public Warehouse warehouse(string code) {
93 Warehouse w=warehouses.GetValueOrDefault(code);
94 error(w==null,"unknown warehouse "+code);
95 return w;
96 }
97 public Item item(string code) {
98 Item s=items.GetValueOrDefault(code);
99 error(s==null,"unknown item "+code);
100 return s;
101 }
102 public Order order(string raw) {
103 long n=long.Parse(raw);
104 error(n<1 || n>orders.Count,"unknown order "+raw);
105 return orders[(int)n-1];
106 }
107 public Lot lookupLot(string raw) {
108 long n=long.Parse(raw);
109 error(n<1 || n>lots.Count,"unknown lot "+raw);
110 return lots[(int)n-1];
111 }
112 public List<Stock> stock(Warehouse w, Item s) {
113 Dictionary<int,Stock> map=w.stock.GetValueOrDefault(s.code);
114 return map==null ? new() : new(map.Values);
115 }
116 public List<Stock> fifo(Warehouse w, Item s) {
117 return stock(w,s).OrderBy(a => a.lot.day).ThenBy(a => a.lot.number).ToList();
118 }
119 public long onhand(Warehouse w, Item s) { return stock(w,s).Sum(a => a.quantity); }
120 public long stockValue(Warehouse w, Item s) { return stock(w,s).Sum(a => a.value()); }
121 public long onhand(Item s) { return warehouses.Values.Sum(w => onhand(w,s)); }
122 public long backorder(Item s) { return orders.Where(o => o.item==s).Sum(o => o.backorder); }
123 public List<Stock> selectedStock(Item s, Warehouse selected) {
124 return warehouses.Values.Where(w => selected==null || w==selected).SelectMany(w => stock(w,s)).ToList();
125 }
126 public void card(Item s, string kind, Warehouse w, long q, long v) { s.cards.Add(new Card(day,kind,w.code,q,v)); }
127 public void total(int category, long q, long value) {
128 totals[category][0]+=q;
129 totals[category][1]+=value;
130 }
131 public void add(Warehouse w, Item s, Lot lot, long q, long cost) {
132 Dictionary<int,Stock> map=w.stock.GetValueOrDefault(s.code);
133 if (map==null) {
134 map=new();
135 w.stock[s.code]=map;
136 }
137 Stock old=map.GetValueOrDefault(lot.number);
138 if (old==null) map[lot.number]=new Stock(lot,q,cost);
139 else old.quantity+=q;
140 }
141 public Lot create(Warehouse w, Item s, long q, long cost, long date, string kind) {
142 Lot lot=new Lot(lots.Count+1,s,date,q,cost,w.code,kind);
143 lots.Add(lot);
144 add(w,s,lot,q,cost);
145 return lot;
146 }
147 public Issue issue(Warehouse w, Item s, long quantity, bool removed) {
148 List<Part> parts=new();
149 long cost=0;
150 foreach (Stock a in fifo(w,s)) {
151 if (quantity==0) break;
152 long q=Math.Min(quantity,a.quantity);
153 parts.Add(new Part(a.lot,q,a.cost));
154 cost+=q*a.cost;
155 a.quantity-=q;
156 quantity-=q;
157 if (removed) a.lot.removed+=q;
158 if (a.quantity==0) w.stock.GetValueOrDefault(s.code).Remove(a.lot.number);
159 }
160 return new Issue(parts,cost);
161 }
162 public Issue fill(Order o, long q, string kind) {
163 Issue issued=issue(o.warehouse,o.item,q,false);
164 o.shipped+=q;
165 o.backorder-=q;
166 o.revenue+=q*o.price;
167 o.cogs+=issued.cost;
168 o.parts.AddRange(issued.parts);
169 if (q>0) {
170 total(4,q,issued.cost);
171 card(o.item,kind,o.warehouse,-q,-issued.cost);
172 }
173 return issued;
174 }
175 public void allocate(Warehouse w, Item s) {
176 long available=onhand(w,s);
177 foreach (Order o in orders) {
178 if (available==0) break;
179 if (o.warehouse==w && o.item==s && o.backorder>0) {
180 long q=Math.Min(available,o.backorder);
181 Issue issued=fill(o,q,"fill");
182 available-=q;
183 output.Add(" FILL order "+o.number+" qty "+q+" revenue "+money(q*o.price)+" cogs "+money(issued.cost)+
184 " profit "+money(q*o.price-issued.cost)+" backorder "+o.backorder);
185 }
186 }
187 }
188 public void insufficient(Warehouse w, Item s, long q) {
189 long h=onhand(w,s);
190 error(h<q,"insufficient stock "+w.code+" "+s.code+" "+h);
191 }
192 public long[] sales(Item s) {
193 List<Order> selected=orders.Where(o => s==null || o.item==s).ToList();
194 return new long[]{selected.Sum(o => o.shipped-o.returned),selected.Sum(o => o.revenue),
195 selected.Sum(o => o.cogs),selected.Count};
196 }
197 public string salesLine(string label, long[] a) {
198 return " "+label+" sold "+a[0]+" revenue "+money(a[1])+" cogs "+money(a[2])+" profit "+money(a[1]-a[2])+" margin "+margin(a[1]-a[2],a[1]);
199 }
200 public string buckets(string label, long[] a, bool monetary) {
201 string result=" "+label;
202 string[] labels={"0-30","31-60","61-90","91+"};
203 for (int i=0;i<4;i++) result+=" "+labels[i]+" "+(monetary ? money(a[i]) : Convert.ToString(a[i]));
204 return result;
205 }
206 public void execute(string[] t) {
207 switch (t[0]) {
208 case "DAY": {
209 long d=long.Parse(t[1]);
210 error(d<=day,"invalid day "+t[1]);
211 day=d;
212 output.Add("DAY "+day);
213 break;
214 }
215 case "WAREHOUSE":
216 error(warehouses.ContainsKey(t[1]),"duplicate warehouse "+t[1]);
217 warehouses[t[1]]=new Warehouse(t[1]);
218 output.Add("WAREHOUSE "+t[1]);
219 break;
220 case "ITEM": {
221 bool exists=items.ContainsKey(t[1]);
222 Item s=exists ? items.GetValueOrDefault(t[1]) : new Item(t[1],long.Parse(t[2]),t[3]);
223 s.level=long.Parse(t[2]);
224 s.name=t[3];
225 items[s.code]=s;
226 output.Add("ITEM "+s.code+(exists ? " updated " : " added ")+s.name+" level "+s.level);
227 break;
228 }
229 case "RECEIVE":
230 case "SHIP":
231 case "ADJUST":
232 case "COUNT": {
233 Warehouse w=warehouse(t[1]);
234 Item s=item(t[2]);
235 long q=t[0].Equals("ADJUST") ? long.Parse(t[3].Substring(1)) : long.Parse(t[3]);
236 if (!t[0].Equals("COUNT")) error(q==0,"invalid quantity");
237 long h=onhand(w,s);
238 if (t[0].Equals("RECEIVE")) {
239 long date=long.Parse(t[5]),cost=parseMoney(t[4]);
240 error(date>day,"future day "+t[5]);
241 Lot lot=create(w,s,q,cost,date,"receive");
242 s.lastCost=cost;
243 total(0,q,q*cost);
244 card(s,"receive",w,q,q*cost);
245 output.Add("RECEIVE "+w.code+" "+s.code+" lot "+lot.number+" qty "+q+" cost "+money(cost)+" day "+date+" value "+money(q*cost));
246 allocate(w,s);
247 } else if (t[0].Equals("SHIP")) {
248 Order o=new Order(orders.Count+1,w,s,day,q,parseMoney(t[4]));
249 orders.Add(o);
250 long f=Math.Min(q,h);
251 Issue issued=fill(o,f,"ship");
252 output.Add("SHIP order "+o.number+" "+w.code+" "+s.code+" shipped "+f+" revenue "+money(o.revenue)+" cogs "+money(issued.cost)+
253 " profit "+money(o.revenue-issued.cost)+" backorder "+o.backorder);
254 } else if (t[0].Equals("ADJUST")) {
255 if (t[3][0]=='+') {
256 long cost=parseMoney(t[4]);
257 Lot lot=create(w,s,q,cost,day,"adjust");
258 total(2,q,q*cost);
259 card(s,"adjust",w,q,q*cost);
260 output.Add("ADJUST "+w.code+" "+s.code+" +"+q+" lot "+lot.number+" value "+money(q*cost));
261 allocate(w,s);
262 } else {
263 insufficient(w,s,q);
264 Issue issued=issue(w,s,q,true);
265 total(3,q,issued.cost);
266 card(s,"adjust",w,-q,-issued.cost);
267 output.Add("ADJUST "+w.code+" "+s.code+" -"+q+" value "+money(issued.cost));
268 }
269 } else {
270 if (q==h) output.Add("COUNT "+w.code+" "+s.code+" qty "+q+" ok");
271 else if (q<h) {
272 Issue issued=issue(w,s,h-q,true);
273 total(3,h-q,issued.cost);
274 card(s,"count",w,q-h,-issued.cost);
275 output.Add("COUNT "+w.code+" "+s.code+" qty "+h+" -> "+q+" shortage "+(h-q)+" value "+money(issued.cost));
276 } else {
277 long cost=s.lastCost ?? 0;
278 Lot lot=create(w,s,q-h,cost,day,"count");
279 total(2,q-h,(q-h)*cost);
280 card(s,"count",w,q-h,(q-h)*cost);
281 output.Add("COUNT "+w.code+" "+s.code+" qty "+h+" -> "+q+" surplus "+(q-h)+" lot "+lot.number+" value "+money((q-h)*cost));
282 allocate(w,s);
283 }
284 }
285 break;
286 }
287 case "CANCEL": {
288 Order o=order(t[1]);
289 error(o.backorder==0,"no backorder "+t[1]);
290 o.cancelled+=o.backorder;
291 o.backorder=0;
292 output.Add("CANCEL order "+o.number+" qty "+o.cancelled);
293 break;
294 }
295 case "TRANSFER": {
296 Warehouse a=warehouse(t[1]),b=warehouse(t[2]);
297 error(a==b,"same warehouse "+t[1]);
298 Item s=item(t[3]);
299 long q=long.Parse(t[4]);
300 error(q==0,"invalid quantity");
301 insufficient(a,s,q);
302 Issue issued=issue(a,s,q,false);
303 foreach (Part p in issued.parts) add(b,s,p.lot,p.quantity,p.cost);
304 card(s,"transfer-out",a,-q,-issued.cost);
305 card(s,"transfer-in",b,q,issued.cost);
306 output.Add("TRANSFER "+a.code+" -> "+b.code+" "+s.code+" qty "+q+" value "+money(issued.cost)+" lots "+issued.parts.Count);
307 allocate(b,s);
308 break;
309 }
310 case "RETURN": {
311 Order o=order(t[1]);
312 long q=long.Parse(t[2]),cost=parseMoney(t[3]);
313 error(q==0,"invalid quantity");
314 error(q>o.shipped-o.returned,"return exceeds shipped "+t[1]+" "+(o.shipped-o.returned));
315 Lot lot=create(o.warehouse,o.item,q,cost,day,"return");
316 o.returned+=q;
317 o.revenue-=q*o.price;
318 o.cogs-=q*cost;
319 total(1,q,q*cost);
320 card(o.item,"return",o.warehouse,q,q*cost);
321 output.Add("RETURN order "+o.number+" "+o.warehouse.code+" "+o.item.code+" qty "+q+" lot "+lot.number+" refund "+money(q*o.price)+" value "+money(q*cost));
322 allocate(o.warehouse,o.item);
323 break;
324 }
325 case "MARKET": {
326 Item s=item(t[1]);
327 long price=parseMoney(t[2]),quantity=0,loss=0,count=0;
328 foreach (Warehouse w in warehouses.Values) {
329 long subtotal=0;
330 foreach (Stock a in stock(w,s)) {
331 if (a.cost>price) {
332 long amount=a.quantity*(a.cost-price);
333 subtotal+=amount;
334 quantity+=a.quantity;
335 count++;
336 a.cost=price;
337 }
338 }
339 if (subtotal>0) card(s,"writedown",w,0,-subtotal);
340 loss+=subtotal;
341 }
342 total(5,quantity,loss);
343 output.Add("MARKET "+s.code+" price "+money(price)+" lots "+count+" qty "+quantity+" loss "+money(loss));
344 break;
345 }
346 case "RECALL": {
347 Lot lot=lookupLot(t[1]);
348 long quantity=0,value=0;
349 List<string> details=new();
350 foreach (Warehouse w in warehouses.Values) {
351 foreach (Stock a in stock(w,lot.item)) if (a.lot==lot) quantity+=a.quantity;
352 }
353 error(quantity==0,"nothing to recall "+t[1]);
354 foreach (Warehouse w in warehouses.Values) {
355 foreach (Stock a in stock(w,lot.item)) {
356 if (a.lot==lot) {
357 value+=a.value();
358 card(lot.item,"recall",w,-a.quantity,-a.value());
359 details.Add(" "+w.code+" qty "+a.quantity+" value "+money(a.value()));
360 w.stock.GetValueOrDefault(lot.item.code).Remove(lot.number);
361 }
362 }
363 }
364 lot.removed+=quantity;
365 total(7,quantity,value);
366 output.Add("RECALL lot "+lot.number+" "+lot.item.code+" qty "+quantity+" value "+money(value));
367 output.AddRange(details);
368 break;
369 }
370 case "EXPIRE": {
371 long age=long.Parse(t[1]),quantity=0,value=0;
372 output.Add("EXPIRE "+age);
373 foreach (Item s in items.Values) {
374 foreach (Warehouse w in warehouses.Values) {
375 long q=0,v=0,count=0;
376 foreach (Stock a in stock(w,s)) {
377 if (day-a.lot.day>age) {
378 q+=a.quantity;
379 v+=a.value();
380 count++;
381 a.lot.removed+=a.quantity;
382 w.stock.GetValueOrDefault(s.code).Remove(a.lot.number);
383 }
384 }
385 if (count>0) {
386 total(6,q,v);
387 card(s,"expire",w,-q,-v);
388 output.Add(" "+s.code+" "+w.code+" lots "+count+" qty "+q+" value "+money(v));
389 quantity+=q;
390 value+=v;
391 }
392 }
393 }
394 output.Add(" total qty "+quantity+" value "+money(value));
395 break;
396 }
397 case "VALUE":
398 case "AGING": {
399 Warehouse selected=t.Length==2 ? warehouse(t[1]) : null;
400 if (t[0].Equals("VALUE")) {
401 output.Add("VALUE"+(selected==null ? "" : " "+selected.code));
402 long quantity=0,value=0;
403 foreach (Item s in items.Values) {
404 List<Stock> target=selectedStock(s,selected);
405 long q=target.Sum(a => a.quantity),v=target.Sum(a => a.value());
406 if (q==0) continue;
407 output.Add(" "+s.code+" qty "+q+" value "+money(v)+" avg "+money(v/q));
408 if (selected==null) {
409 foreach (Warehouse w in warehouses.Values) if (onhand(w,s)>0) output.Add(" "+w.code+" qty "+onhand(w,s)+" value "+money(stockValue(w,s)));
410 }
411 quantity+=q;
412 value+=v;
413 }
414 output.Add(" total qty "+quantity+" value "+money(value));
415 } else {
416 output.Add("AGING"+(selected==null ? "" : " "+selected.code)+" day "+day);
417 long[] quantities=new long[4],values=new long[4];
418 foreach (Item s in items.Values) {
419 long[] q=new long[4];
420 foreach (Stock a in selectedStock(s,selected)) {
421 long age=day-a.lot.day;
422 int b=age<=30 ? 0 : age<=60 ? 1 : age<=90 ? 2 : 3;
423 q[b]+=a.quantity;
424 quantities[b]+=a.quantity;
425 values[b]+=a.value();
426 }
427 if (q.Sum()>0) output.Add(buckets(s.code,q,false));
428 }
429 output.Add(buckets("total",quantities,false));
430 output.Add(buckets("value",values,true));
431 }
432 break;
433 }
434 case "LOTS": {
435 Warehouse w=warehouse(t[1]);
436 Item s=item(t[2]);
437 output.Add("LOTS "+w.code+" "+s.code);
438 foreach (Stock a in fifo(w,s)) output.Add(" lot "+a.lot.number+" day "+a.lot.day+" qty "+a.quantity+" cost "+money(a.cost)+" value "+money(a.value())+" age "+(day-a.lot.day));
439 foreach (Order o in orders) if (o.warehouse==w && o.item==s && o.backorder>0) output.Add(" backorder order "+o.number+" qty "+o.backorder);
440 break;
441 }
442 case "REORDER":
443 case "ITEMS":
444 output.Add(t[0]);
445 foreach (Item s in items.Values) {
446 long h=onhand(s),b=backorder(s);
447 if (t[0].Equals("ITEMS")) output.Add(" "+s.code+" "+s.name+" level "+s.level+" onhand "+h+" backorder "+b+" cost "+(s.lastCost==null ? "none" : money(s.lastCost.Value)));
448 else if (h-b<=s.level) output.Add(" "+s.code+" "+s.name+" onhand "+h+" backorder "+b+" level "+s.level);
449 }
450 break;
451 case "MARGIN":
452 case "TOP": {
453 Dictionary<Item,long[]> summaries=new();
454 foreach (Item s in items.Values) {
455 long[] a=sales(s);
456 if (a[3]>0) summaries[s]=a;
457 }
458 if (t[0].Equals("MARGIN")) {
459 output.Add("MARGIN");
460 foreach (Item s in items.Values) if (summaries.ContainsKey(s)) output.Add(salesLine(s.code,summaries[s]));
461 output.Add(salesLine("total",sales(null)));
462 } else {
463 int k=(int)long.Parse(t[1]);
464 output.Add("TOP "+k);
465 List<Item> ranked=summaries.Keys.OrderByDescending(s => summaries[s][1]-summaries[s][2])
466 .ThenByDescending(s => summaries[s][1]).ThenBy(s => s.code,StringComparer.Ordinal).Take(k).ToList();
467 for (int i=0;i<ranked.Count;i++) {
468 Item s=ranked[i];
469 long[] a=summaries[s];
470 output.Add(" "+(i+1)+" "+s.code+" profit "+money(a[1]-a[2])+" revenue "+money(a[1])+" margin "+margin(a[1]-a[2],a[1]));
471 }
472 }
473 break;
474 }
475 case "BACKORDERS":
476 output.Add("BACKORDERS");
477 foreach (Order o in orders) if (o.backorder>0) output.Add(" order "+o.number+" "+o.warehouse.code+" "+o.item.code+" qty "+o.backorder+" day "+o.day+" price "+money(o.price));
478 break;
479 case "ORDER": {
480 Order o=order(t[1]);
481 output.Add("ORDER "+o.number+" "+o.warehouse.code+" "+o.item.code+" day "+o.day+" price "+money(o.price));
482 output.Add(" ordered "+o.ordered+" shipped "+o.shipped+" backorder "+o.backorder+" cancelled "+o.cancelled+" returned "+o.returned);
483 foreach (Part p in o.parts) output.Add(" lot "+p.lot.number+" qty "+p.quantity+" cost "+money(p.cost));
484 output.Add(" revenue "+money(o.revenue)+" cogs "+money(o.cogs)+" profit "+money(o.revenue-o.cogs));
485 break;
486 }
487 case "TRACE": {
488 Lot lot=lookupLot(t[1]);
489 output.Add("TRACE lot "+lot.number+" "+lot.item.code+" "+lot.kind+" "+lot.warehouse+" day "+lot.day+" qty "+lot.initialQuantity+" cost "+money(lot.initialCost));
490 foreach (Warehouse w in warehouses.Values) {
491 foreach (Stock a in stock(w,lot.item)) if (a.lot==lot) output.Add(" held "+w.code+" qty "+a.quantity+" cost "+money(a.cost));
492 }
493 foreach (Order o in orders) {
494 long q=o.parts.Where(p => p.lot==lot).Sum(p => p.quantity);
495 if (q>0) output.Add(" order "+o.number+" qty "+q);
496 }
497 if (lot.removed>0) output.Add(" removed qty "+lot.removed);
498 break;
499 }
500 case "CARD": {
501 Item s=item(t[1]);
502 output.Add("CARD "+s.code+" "+s.name);
503 long quantity=0,value=0;
504 foreach (Card c in s.cards) {
505 quantity+=c.quantity;
506 value+=c.value;
507 output.Add(" day "+c.day+" "+c.kind+" "+c.warehouse+" "+signed(c.quantity)+" "+signedMoney(c.value)+" balance "+quantity+" "+money(value));
508 }
509 break;
510 }
511 case "STATUS":
512 output.Add("STATUS");
513 foreach (Warehouse w in warehouses.Values) {
514 List<Stock> held=w.stock.Values.SelectMany(map => map.Values).ToList();
515 long q=held.Sum(a => a.quantity),v=held.Sum(a => a.value());
516 int skus=w.stock.Values.Count(map => map.Count>0);
517 long backorder=orders.Where(o => o.warehouse==w).Sum(o => o.backorder);
518 output.Add(" "+w.code+" skus "+skus+" qty "+q+" value "+money(v)+" backorder "+backorder);
519 }
520 break;
521 case "SUMMARY": {
522 output.Add("SUMMARY");
523 for (int i=0;i<8;i++) output.Add(" "+CATEGORIES[i]+" qty "+totals[i][0]+" value "+money(totals[i][1]));
524 long q=items.Values.Sum(s => onhand(s));
525 long v=items.Values.Sum(s => warehouses.Values.Sum(w => stockValue(w,s)));
526 output.Add(" on-hand qty "+q+" value "+money(v));
527 output.Add(" orders "+orders.Count+" open "+orders.Count(o => o.backorder>0)+" backorder "+orders.Sum(o => o.backorder));
528 break;
529 }
530 }
531 if (output.Count==1 && new[]{"LOTS","REORDER","ITEMS","TOP","BACKORDERS","CARD","STATUS"}.Contains(t[0])) output.Add(" (none)");
532 }
533 public List<string> process(string line) {
534 output=new();
535 try { execute(line.Split(' ')); }
536 catch (Fault f) { output.Add("ERROR "+f.Message); }
537 return output;
538 }
539 }
540 static void Main() {
541 CultureInfo.CurrentCulture=CultureInfo.InvariantCulture;
