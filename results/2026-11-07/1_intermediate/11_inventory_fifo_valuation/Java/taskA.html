5 static final String[] CATEGORIES={"received","returned","adjusted-in","adjusted-out","shipped","written-down","expired","recalled"};
6 static class Fault extends RuntimeException { Fault(String reason) { super(reason); } }
7 static void error(boolean condition, String reason) { if (condition) throw new Fault(reason); }
8 static long parseMoney(String s) {
9 return Long.parseLong(s.replace(".",""));
10 }
11 static String money(long n) {
12 long a=Math.abs(n);
13 return String.format(Locale.ROOT,"%s%d.%02d",n<0 ? "-" : "",a/100,a%100);
14 }
15 static String signed(long n) { return (n>0 ? "+" : "")+n; }
16 static String signedMoney(long n) { return (n>0 ? "+" : "")+money(n); }
17 static String margin(long profit, long revenue) {
18 if (revenue==0) return "n/a";
19 long q=Math.abs(profit)*1000/revenue;
20 return (profit<0 && q>0 ? "-" : "")+q/10+"."+q%10+"%";
21 }

82 static class Inventory {
83 long day=1;
84 TreeMap<String,Warehouse> warehouses=new TreeMap<>();
85 TreeMap<String,Item> items=new TreeMap<>();
86 List<Lot> lots=new ArrayList<>();
87 List<Order> orders=new ArrayList<>();
88 long[][] totals=new long[8][2];
89 List<String> output;
90 Warehouse warehouse(String code) {
91 Warehouse w=warehouses.get(code);
92 error(w==null,"unknown warehouse "+code);
93 return w;
94 }
95 Item item(String code) {
96 Item s=items.get(code);
97 error(s==null,"unknown item "+code);
98 return s;
99 }
100 Order order(String raw) {
101 long n=Long.parseLong(raw);
102 error(n<1 || n>orders.size(),"unknown order "+raw);
103 return orders.get((int)n-1);
104 }
105 Lot lot(String raw) {
106 long n=Long.parseLong(raw);
107 error(n<1 || n>lots.size(),"unknown lot "+raw);
108 return lots.get((int)n-1);
109 }
110 List<Stock> stock(Warehouse w, Item s) {
111 Map<Integer,Stock> map=w.stock.get(s.code);
112 return map==null ? new ArrayList<>() : new ArrayList<>(map.values());
113 }
114 List<Stock> fifo(Warehouse w, Item s) {
115 List<Stock> result=stock(w,s);
116 result.sort(Comparator.comparingLong((Stock a) -> a.lot.day).thenComparingInt(a -> a.lot.number));
117 return result;
118 }
119 long onhand(Warehouse w, Item s) { return stock(w,s).stream().mapToLong(a -> a.quantity).sum(); }
120 long value(Warehouse w, Item s) { return stock(w,s).stream().mapToLong(Stock::value).sum(); }
121 long onhand(Item s) { return warehouses.values().stream().mapToLong(w -> onhand(w,s)).sum(); }
122 long backorder(Item s) { return orders.stream().filter(o -> o.item==s).mapToLong(o -> o.backorder).sum(); }
123 List<Stock> target(Item s, Warehouse selected) {
124 return warehouses.values().stream().filter(w -> selected==null || w==selected)
125 .flatMap(w -> stock(w,s).stream()).toList();
126 }
127 void card(Item s, String kind, Warehouse w, long q, long v) { s.cards.add(new Card(day,kind,w.code,q,v)); }
128 void total(int category, long q, long value) {
129 totals[category][0]+=q;
130 totals[category][1]+=value;
131 }
132 void add(Warehouse w, Item s, Lot lot, long q, long cost) {
133 Map<Integer,Stock> map=w.stock.computeIfAbsent(s.code,k -> new HashMap<>());
134 Stock old=map.get(lot.number);
135 if (old==null) map.put(lot.number,new Stock(lot,q,cost));
136 else old.quantity+=q;
137 }
138 Lot create(Warehouse w, Item s, long q, long cost, long date, String kind) {
139 Lot lot=new Lot(lots.size()+1,s,date,q,cost,w.code,kind);
140 lots.add(lot);
141 add(w,s,lot,q,cost);
142 return lot;
143 }
144 Issue issue(Warehouse w, Item s, long quantity, boolean removed) {
145 List<Part> parts=new ArrayList<>();
146 long cost=0;
147 for (Stock a:fifo(w,s)) {
148 if (quantity==0) break;
149 long q=Math.min(quantity,a.quantity);
150 parts.add(new Part(a.lot,q,a.cost));
151 cost+=q*a.cost;
152 a.quantity-=q;
153 quantity-=q;
154 if (removed) a.lot.removed+=q;
155 if (a.quantity==0) w.stock.get(s.code).remove(a.lot.number);
156 }
157 return new Issue(parts,cost);
158 }
159 Issue fill(Order o, long q, String kind) {
160 Issue issued=issue(o.warehouse,o.item,q,false);
161 o.shipped+=q;
162 o.backorder-=q;
163 o.revenue+=q*o.price;
164 o.cogs+=issued.cost();
165 o.parts.addAll(issued.parts());
166 if (q>0) {
167 total(4,q,issued.cost());
168 card(o.item,kind,o.warehouse,-q,-issued.cost());
169 }
170 return issued;
171 }
172 void allocate(Warehouse w, Item s) {
173 long available=onhand(w,s);
174 for (Order o:orders) {
175 if (available==0) break;
176 if (o.warehouse==w && o.item==s && o.backorder>0) {
177 long q=Math.min(available,o.backorder);
178 Issue issued=fill(o,q,"fill");
179 available-=q;
180 output.add(" FILL order "+o.number+" qty "+q+" revenue "+money(q*o.price)+" cogs "+money(issued.cost())+
181 " profit "+money(q*o.price-issued.cost())+" backorder "+o.backorder);
182 }
183 }
184 }
185 void insufficient(Warehouse w, Item s, long q) {
186 long h=onhand(w,s);
187 error(h<q,"insufficient stock "+w.code+" "+s.code+" "+h);
188 }
189 long[] sales(Item s) {
190 List<Order> selected=orders.stream().filter(o -> s==null || o.item==s).toList();
191 return new long[]{selected.stream().mapToLong(o -> o.shipped-o.returned).sum(),
192 selected.stream().mapToLong(o -> o.revenue).sum(),
193 selected.stream().mapToLong(o -> o.cogs).sum(),selected.size()};
194 }
195 String salesLine(String label, long[] a) {
196 return " "+label+" sold "+a[0]+" revenue "+money(a[1])+" cogs "+money(a[2])+" profit "+money(a[1]-a[2])+" margin "+margin(a[1]-a[2],a[1]);
197 }
198 String buckets(String label, long[] a, boolean monetary) {
199 String result=" "+label;
200 String[] labels={"0-30","31-60","61-90","91+"};
201 for (int i=0;i<4;i++) result+=" "+labels[i]+" "+(monetary ? money(a[i]) : Long.toString(a[i]));
202 return result;
203 }
204 void execute(String[] t) {
205 switch (t[0]) {
206 case "DAY": {
207 long d=Long.parseLong(t[1]);
208 error(d<=day,"invalid day "+t[1]);
209 day=d;
210 output.add("DAY "+day);
211 break;
212 }
213 case "WAREHOUSE":
214 error(warehouses.containsKey(t[1]),"duplicate warehouse "+t[1]);
215 warehouses.put(t[1],new Warehouse(t[1]));
216 output.add("WAREHOUSE "+t[1]);
217 break;
218 case "ITEM": {
219 boolean exists=items.containsKey(t[1]);
220 Item s=exists ? items.get(t[1]) : new Item(t[1],Long.parseLong(t[2]),t[3]);
221 s.level=Long.parseLong(t[2]);
222 s.name=t[3];
223 items.put(s.code,s);
224 output.add("ITEM "+s.code+(exists ? " updated " : " added ")+s.name+" level "+s.level);
225 break;
226 }
227 case "RECEIVE", "SHIP", "ADJUST", "COUNT": {
228 Warehouse w=warehouse(t[1]);
229 Item s=item(t[2]);
230 long q=t[0].equals("ADJUST") ? Long.parseLong(t[3].substring(1)) : Long.parseLong(t[3]);
231 if (!t[0].equals("COUNT")) error(q==0,"invalid quantity");
232 long h=onhand(w,s);
233 if (t[0].equals("RECEIVE")) {
234 long date=Long.parseLong(t[5]),cost=parseMoney(t[4]);
235 error(date>day,"future day "+t[5]);
236 Lot lot=create(w,s,q,cost,date,"receive");
237 s.lastCost=cost;
238 total(0,q,q*cost);
239 card(s,"receive",w,q,q*cost);
240 output.add("RECEIVE "+w.code+" "+s.code+" lot "+lot.number+" qty "+q+" cost "+money(cost)+" day "+date+" value "+money(q*cost));
241 allocate(w,s);
242 } else if (t[0].equals("SHIP")) {
243 Order o=new Order(orders.size()+1,w,s,day,q,parseMoney(t[4]));
244 orders.add(o);
245 long f=Math.min(q,h);
246 Issue issued=fill(o,f,"ship");
247 output.add("SHIP order "+o.number+" "+w.code+" "+s.code+" shipped "+f+" revenue "+money(o.revenue)+" cogs "+money(issued.cost())+
248 " profit "+money(o.revenue-issued.cost())+" backorder "+o.backorder);
249 } else if (t[0].equals("ADJUST")) {
250 if (t[3].charAt(0)=='+') {
251 long cost=parseMoney(t[4]);
252 Lot lot=create(w,s,q,cost,day,"adjust");
253 total(2,q,q*cost);
254 card(s,"adjust",w,q,q*cost);
255 output.add("ADJUST "+w.code+" "+s.code+" +"+q+" lot "+lot.number+" value "+money(q*cost));
256 allocate(w,s);
257 } else {
258 insufficient(w,s,q);
259 Issue issued=issue(w,s,q,true);
260 total(3,q,issued.cost());
261 card(s,"adjust",w,-q,-issued.cost());
262 output.add("ADJUST "+w.code+" "+s.code+" -"+q+" value "+money(issued.cost()));
263 }
264 } else {
265 if (q==h) output.add("COUNT "+w.code+" "+s.code+" qty "+q+" ok");
266 else if (q<h) {
267 Issue issued=issue(w,s,h-q,true);
268 total(3,h-q,issued.cost());
269 card(s,"count",w,q-h,-issued.cost());
270 output.add("COUNT "+w.code+" "+s.code+" qty "+h+" -> "+q+" shortage "+(h-q)+" value "+money(issued.cost()));
271 } else {
272 long cost=s.lastCost==null ? 0 : s.lastCost;
273 Lot lot=create(w,s,q-h,cost,day,"count");
274 total(2,q-h,(q-h)*cost);
275 card(s,"count",w,q-h,(q-h)*cost);
276 output.add("COUNT "+w.code+" "+s.code+" qty "+h+" -> "+q+" surplus "+(q-h)+" lot "+lot.number+" value "+money((q-h)*cost));
277 allocate(w,s);
278 }
279 }
280 break;
281 }
282 case "CANCEL": {
283 Order o=order(t[1]);
284 error(o.backorder==0,"no backorder "+t[1]);
285 o.cancelled+=o.backorder;
286 o.backorder=0;
287 output.add("CANCEL order "+o.number+" qty "+o.cancelled);
288 break;
289 }
290 case "TRANSFER": {
291 Warehouse a=warehouse(t[1]),b=warehouse(t[2]);
292 error(a==b,"same warehouse "+t[1]);
293 Item s=item(t[3]);
294 long q=Long.parseLong(t[4]);
295 error(q==0,"invalid quantity");
296 insufficient(a,s,q);
297 Issue issued=issue(a,s,q,false);
298 for (Part p:issued.parts()) add(b,s,p.lot(),p.quantity(),p.cost());
299 card(s,"transfer-out",a,-q,-issued.cost());
300 card(s,"transfer-in",b,q,issued.cost());
301 output.add("TRANSFER "+a.code+" -> "+b.code+" "+s.code+" qty "+q+" value "+money(issued.cost())+" lots "+issued.parts().size());
302 allocate(b,s);
303 break;
304 }
305 case "RETURN": {
306 Order o=order(t[1]);
307 long q=Long.parseLong(t[2]),cost=parseMoney(t[3]);
308 error(q==0,"invalid quantity");
309 error(q>o.shipped-o.returned,"return exceeds shipped "+t[1]+" "+(o.shipped-o.returned));
310 Lot lot=create(o.warehouse,o.item,q,cost,day,"return");
311 o.returned+=q;
312 o.revenue-=q*o.price;
313 o.cogs-=q*cost;
314 total(1,q,q*cost);
315 card(o.item,"return",o.warehouse,q,q*cost);
316 output.add("RETURN order "+o.number+" "+o.warehouse.code+" "+o.item.code+" qty "+q+" lot "+lot.number+" refund "+money(q*o.price)+" value "+money(q*cost));
317 allocate(o.warehouse,o.item);
318 break;
319 }
320 case "MARKET": {
321 Item s=item(t[1]);
322 long price=parseMoney(t[2]),quantity=0,loss=0,count=0;
323 for (Warehouse w:warehouses.values()) {
324 long subtotal=0;
325 for (Stock a:stock(w,s)) {
326 if (a.cost>price) {
327 long amount=a.quantity*(a.cost-price);
328 subtotal+=amount;
329 quantity+=a.quantity;
330 count++;
331 a.cost=price;
332 }
333 }
334 if (subtotal>0) card(s,"writedown",w,0,-subtotal);
335 loss+=subtotal;
336 }
337 total(5,quantity,loss);
338 output.add("MARKET "+s.code+" price "+money(price)+" lots "+count+" qty "+quantity+" loss "+money(loss));
339 break;
340 }
341 case "RECALL": {
342 Lot lot=lot(t[1]);
343 long quantity=0,value=0;
344 List<String> details=new ArrayList<>();
345 for (Warehouse w:warehouses.values()) {
346 for (Stock a:stock(w,lot.item)) if (a.lot==lot) quantity+=a.quantity;
347 }
348 error(quantity==0,"nothing to recall "+t[1]);
349 for (Warehouse w:warehouses.values()) {
350 for (Stock a:stock(w,lot.item)) {
351 if (a.lot==lot) {
352 value+=a.value();
353 card(lot.item,"recall",w,-a.quantity,-a.value());
354 details.add(" "+w.code+" qty "+a.quantity+" value "+money(a.value()));
355 w.stock.get(lot.item.code).remove(lot.number);
356 }
357 }
358 }
359 lot.removed+=quantity;
360 total(7,quantity,value);
361 output.add("RECALL lot "+lot.number+" "+lot.item.code+" qty "+quantity+" value "+money(value));
362 output.addAll(details);
363 break;
364 }
365 case "EXPIRE": {
366 long age=Long.parseLong(t[1]),quantity=0,value=0;
367 output.add("EXPIRE "+age);
368 for (Item s:items.values()) {
369 for (Warehouse w:warehouses.values()) {
370 long q=0,v=0,count=0;
371 for (Stock a:stock(w,s)) {
372 if (day-a.lot.day>age) {
373 q+=a.quantity;
374 v+=a.value();
375 count++;
376 a.lot.removed+=a.quantity;
377 w.stock.get(s.code).remove(a.lot.number);
378 }
379 }
380 if (count>0) {
381 total(6,q,v);
382 card(s,"expire",w,-q,-v);
383 output.add(" "+s.code+" "+w.code+" lots "+count+" qty "+q+" value "+money(v));
384 quantity+=q;
385 value+=v;
386 }
387 }
388 }
389 output.add(" total qty "+quantity+" value "+money(value));
390 break;
391 }
392 case "VALUE", "AGING": {
393 Warehouse selected=t.length==2 ? warehouse(t[1]) : null;
394 if (t[0].equals("VALUE")) {
395 output.add("VALUE"+(selected==null ? "" : " "+selected.code));
396 long quantity=0,value=0;
397 for (Item s:items.values()) {
398 List<Stock> target=target(s,selected);
399 long q=target.stream().mapToLong(a -> a.quantity).sum(),v=target.stream().mapToLong(Stock::value).sum();
400 if (q==0) continue;
401 output.add(" "+s.code+" qty "+q+" value "+money(v)+" avg "+money(v/q));
402 if (selected==null) {
403 for (Warehouse w:warehouses.values()) if (onhand(w,s)>0) output.add(" "+w.code+" qty "+onhand(w,s)+" value "+money(value(w,s)));
404 }
405 quantity+=q;
406 value+=v;
407 }
408 output.add(" total qty "+quantity+" value "+money(value));
409 } else {
410 output.add("AGING"+(selected==null ? "" : " "+selected.code)+" day "+day);
411 long[] quantities=new long[4],values=new long[4];
412 for (Item s:items.values()) {
413 long[] q=new long[4];
414 for (Stock a:target(s,selected)) {
415 long age=day-a.lot.day;
416 int b=age<=30 ? 0 : age<=60 ? 1 : age<=90 ? 2 : 3;
417 q[b]+=a.quantity;
418 quantities[b]+=a.quantity;
419 values[b]+=a.value();
420 }
421 if (Arrays.stream(q).sum()>0) output.add(buckets(s.code,q,false));
422 }
423 output.add(buckets("total",quantities,false));
424 output.add(buckets("value",values,true));
425 }
426 break;
427 }
428 case "LOTS": {
429 Warehouse w=warehouse(t[1]);
430 Item s=item(t[2]);
431 output.add("LOTS "+w.code+" "+s.code);
432 for (Stock a:fifo(w,s)) output.add(" lot "+a.lot.number+" day "+a.lot.day+" qty "+a.quantity+" cost "+money(a.cost)+" value "+money(a.value())+" age "+(day-a.lot.day));
433 for (Order o:orders) if (o.warehouse==w && o.item==s && o.backorder>0) output.add(" backorder order "+o.number+" qty "+o.backorder);
434 break;
435 }
436 case "REORDER", "ITEMS":
437 output.add(t[0]);
438 for (Item s:items.values()) {
439 long h=onhand(s),b=backorder(s);
440 if (t[0].equals("ITEMS")) output.add(" "+s.code+" "+s.name+" level "+s.level+" onhand "+h+" backorder "+b+" cost "+(s.lastCost==null ? "none" : money(s.lastCost)));
441 else if (h-b<=s.level) output.add(" "+s.code+" "+s.name+" onhand "+h+" backorder "+b+" level "+s.level);
442 }
443 break;
444 case "MARGIN", "TOP": {
445 Map<Item,long[]> summaries=new HashMap<>();
446 for (Item s:items.values()) {
447 long[] a=sales(s);
448 if (a[3]>0) summaries.put(s,a);
449 }
450 if (t[0].equals("MARGIN")) {
451 output.add("MARGIN");
452 for (Item s:items.values()) if (summaries.containsKey(s)) output.add(salesLine(s.code,summaries.get(s)));
453 output.add(salesLine("total",sales(null)));
454 } else {
455 int k=(int)Long.parseLong(t[1]);
456 output.add("TOP "+k);
457 List<Item> ranked=new ArrayList<>(summaries.keySet());
458 ranked.sort(Comparator.comparingLong((Item s) -> -(summaries.get(s)[1]-summaries.get(s)[2])).thenComparingLong(s -> -summaries.get(s)[1]).thenComparing(s -> s.code));
459 for (int i=0;i<Math.min(k,ranked.size());i++) {
460 Item s=ranked.get(i);
461 long[] a=summaries.get(s);
462 output.add(" "+(i+1)+" "+s.code+" profit "+money(a[1]-a[2])+" revenue "+money(a[1])+" margin "+margin(a[1]-a[2],a[1]));
463 }
464 }
465 break;
466 }
467 case "BACKORDERS":
468 output.add("BACKORDERS");
469 for (Order o:orders) if (o.backorder>0) output.add(" order "+o.number+" "+o.warehouse.code+" "+o.item.code+" qty "+o.backorder+" day "+o.day+" price "+money(o.price));
470 break;
471 case "ORDER": {
472 Order o=order(t[1]);
473 output.add("ORDER "+o.number+" "+o.warehouse.code+" "+o.item.code+" day "+o.day+" price "+money(o.price));
474 output.add(" ordered "+o.ordered+" shipped "+o.shipped+" backorder "+o.backorder+" cancelled "+o.cancelled+" returned "+o.returned);
475 for (Part p:o.parts) output.add(" lot "+p.lot().number+" qty "+p.quantity()+" cost "+money(p.cost()));
476 output.add(" revenue "+money(o.revenue)+" cogs "+money(o.cogs)+" profit "+money(o.revenue-o.cogs));
477 break;
478 }
479 case "TRACE": {
480 Lot lot=lot(t[1]);
481 output.add("TRACE lot "+lot.number+" "+lot.item.code+" "+lot.kind+" "+lot.warehouse+" day "+lot.day+" qty "+lot.initialQuantity+" cost "+money(lot.initialCost));
482 for (Warehouse w:warehouses.values()) {
483 for (Stock a:stock(w,lot.item)) if (a.lot==lot) output.add(" held "+w.code+" qty "+a.quantity+" cost "+money(a.cost));
484 }
485 for (Order o:orders) {
486 long q=o.parts.stream().filter(p -> p.lot()==lot).mapToLong(Part::quantity).sum();
487 if (q>0) output.add(" order "+o.number+" qty "+q);
488 }
489 if (lot.removed>0) output.add(" removed qty "+lot.removed);
490 break;
491 }
492 case "CARD": {
493 Item s=item(t[1]);
494 output.add("CARD "+s.code+" "+s.name);
495 long quantity=0,value=0;
496 for (Card c:s.cards) {
497 quantity+=c.quantity();
498 value+=c.value();
499 output.add(" day "+c.day()+" "+c.kind()+" "+c.warehouse()+" "+signed(c.quantity())+" "+signedMoney(c.value())+" balance "+quantity+" "+money(value));
500 }
501 break;
502 }
503 case "STATUS":
504 output.add("STATUS");
505 for (Warehouse w:warehouses.values()) {
506 List<Stock> held=w.stock.values().stream().flatMap(map -> map.values().stream()).toList();
507 long q=held.stream().mapToLong(a -> a.quantity).sum();
508 long v=held.stream().mapToLong(Stock::value).sum();
509 long skus=w.stock.values().stream().filter(map -> !map.isEmpty()).count();
510 long backorder=orders.stream().filter(o -> o.warehouse==w).mapToLong(o -> o.backorder).sum();
511 output.add(" "+w.code+" skus "+skus+" qty "+q+" value "+money(v)+" backorder "+backorder);
512 }
513 break;
514 case "SUMMARY": {
515 output.add("SUMMARY");
516 for (int i=0;i<8;i++) output.add(" "+CATEGORIES[i]+" qty "+totals[i][0]+" value "+money(totals[i][1]));
517 long q=items.values().stream().mapToLong(this::onhand).sum();
518 long v=items.values().stream().mapToLong(s -> warehouses.values().stream().mapToLong(w -> value(w,s)).sum()).sum();
519 output.add(" on-hand qty "+q+" value "+money(v));
520 output.add(" orders "+orders.size()+" open "+orders.stream().filter(o -> o.backorder>0).count()+" backorder "+orders.stream().mapToLong(o -> o.backorder).sum());
521 break;
522 }
523 }
524 if (output.size()==1 && Set.of("LOTS","REORDER","ITEMS","TOP","BACKORDERS","CARD","STATUS").contains(t[0])) output.add(" (none)");
525 }
526 List<String> process(String line) {
527 output=new ArrayList<>();
528 try { execute(line.split(" ")); }
529 catch (Fault f) { output.add("ERROR "+f.getMessage()); }
530 return output;
