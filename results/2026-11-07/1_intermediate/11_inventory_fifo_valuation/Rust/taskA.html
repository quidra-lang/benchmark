4const CATEGORIES: [&str; 8] = [
5 "received", "returned", "adjusted-in", "adjusted-out",
6 "shipped", "written-down", "expired", "recalled",
7];
8
9fn number(text: &str) -> i64 {
10 text.parse().unwrap()
11}
12
13fn cents(text: &str) -> i64 {
14 text.replace('.', "").parse().unwrap()
15}
16
17fn money(mut amount: i64) -> String {
18 let sign = if amount < 0 {
19 amount = -amount;
20 "-"
21 } else {
22 ""
23 };
24 format!("{sign}{}.{:02}", amount / 100, amount % 100)
25}
26
27fn signed_quantity(value: i64) -> String {
28 format!("{}{value}", if value > 0 { "+" } else { "" })
29}
30
31fn signed_money(value: i64) -> String {
32 format!("{}{}", if value > 0 { "+" } else { "" }, money(value))
33}
34
35fn margin(profit: i64, revenue: i64) -> String {
36 if revenue == 0 {
37 return "n/a".into();
38 }
39 let tenths = profit.abs() * 1000 / revenue;
40 format!("{}{}.{:01}%",
41 if profit < 0 && tenths > 0 { "-" } else { "" },
42 tenths / 10, tenths % 10)
43}
44
45macro_rules! line {
46 ($inv:expr, $indent:expr, $($part:expr),+ $(,)?) => {{
47 let parts = [$($part.to_string(),)+];
48 $inv.output.push(format!("{}{}", " ".repeat($indent), parts.join(" ")));
49 }};
50}

121struct Inventory {
122 day: i64,
123 stock: BTreeMap<String, BTreeMap<String, BTreeMap<usize, StockLot>>>,
124 items: BTreeMap<String, Item>,
125 lots: Vec<Lot>,

129}

131impl Inventory {
132 fn new() -> Self {
133 let totals = CATEGORIES.iter().map(|&name| (name, Total::default())).collect();
134 Self {
135 day: 1,
136 stock: BTreeMap::new(),
137 items: BTreeMap::new(),
138 lots: Vec::new(),
139 orders: Vec::new(),
140 totals,
141 output: Vec::new(),
142 }

144
145 fn warehouse(&self, code: &str) -> Result<(), String> {
146 if self.stock.contains_key(code) {
147 Ok(())
148 } else {
149 Err(format!("unknown warehouse {code}"))
150 }
151 }
152
153 fn item(&self, code: &str) -> Result<(), String> {
154 if self.items.contains_key(code) {
155 Ok(())
156 } else {
157 Err(format!("unknown item {code}"))
158 }
159 }
160
161 fn pair(&self, warehouse: &str, code: &str) -> Result<(), String> {
162 self.warehouse(warehouse)?;
163 self.item(code)
164 }
165
166 fn order_index(&self, text: &str) -> Result<usize, String> {
167 let id = number(text);
168 if id < 1 || id > self.orders.len() as i64 {
169 Err(format!("unknown order {text}"))
170 } else {
171 Ok(id as usize - 1)
172 }
173 }
174
175 fn lot_index(&self, text: &str) -> Result<usize, String> {
176 let id = number(text);
177 if id < 1 || id > self.lots.len() as i64 {
178 Err(format!("unknown lot {text}"))
179 } else {
180 Ok(id as usize - 1)
181 }
182 }
183
184 fn stock_lots(&self, w: &str, code: &str) -> Option<&BTreeMap<usize, StockLot>> {
185 self.stock.get(w).and_then(|items| items.get(code))
186 }
187
188 fn fifo(&self, w: &str, code: &str) -> Vec<usize> {
189 let mut ids = self.stock_lots(w, code)
190 .map(|stocks| stocks.keys().copied().collect::<Vec<_>>())
191 .unwrap_or_default();
192 ids.sort_by_key(|&id| (self.lots[id - 1].day, id));
193 ids
194 }
195
196 fn quantity(&self, w: &str, code: &str) -> i64 {
197 self.stock_lots(w, code)
198 .map(|stocks| stocks.values().map(|s| s.quantity).sum())
199 .unwrap_or(0)
200 }
201
202 fn total_quantity(&self, code: &str) -> i64 {
203 self.stock.keys().map(|w| self.quantity(w, code)).sum()
204 }
205
206 fn backorder(&self, code: &str) -> i64 {
207 self.orders.iter().filter(|o| o.item == code).map(|o| o.backorder).sum()
208 }
209
210 fn all_stocks(&self) -> Total {
211 let mut result = Total::default();
212 for items in self.stock.values() {
213 for lots in items.values() {
214 for stock in lots.values() {
215 result.quantity += stock.quantity;
216 result.value += stock.quantity * stock.cost;
217 }
218 }
219 }
220 result
221 }
222
223 fn require_stock(&self, w: &str, code: &str, q: i64) -> Result<(), String> {
224 let held = self.quantity(w, code);
225 if held < q {
226 Err(format!("insufficient stock {w} {code} {held}"))
227 } else {
228 Ok(())
229 }
230 }
231
232 fn card_entry(&mut self, code: &str, kind: &str, w: &str, q: i64, value: i64) {
233 self.items.get_mut(code).unwrap().card.push(CardEntry {
234 day: self.day,
235 kind: kind.into(),
236 warehouse: w.into(),
237 quantity: q,
238 value,
239 });
240 }
241
242 fn accumulate(&mut self, category: &'static str, q: i64, value: i64) {
243 let total = self.totals.get_mut(category).unwrap();
244 total.quantity += q;
245 total.value += value;
246 }
247
248 fn add_stock(&mut self, w: &str, id: usize, q: i64, cost: i64) {
249 let code = self.lots[id - 1].item.clone();
250 let stocks = self.stock.get_mut(w).unwrap().entry(code).or_default();
251 if let Some(stock) = stocks.get_mut(&id) {
252 stock.quantity += q;
253 } else {
254 stocks.insert(id, StockLot { quantity: q, cost });
255 }
256 }
257
258 fn create_lot(&mut self, w: &str, code: &str,
259 day: i64, q: i64, cost: i64, kind: &str) -> usize {
260 let id = self.lots.len() + 1;
261 self.lots.push(Lot {
262 number: id,
263 item: code.into(),
264 warehouse: w.into(),
265 kind: kind.into(),
266 day,
267 initial_quantity: q,
268 initial_cost: cost,
269 removed: 0,
270 });
271 self.add_stock(w, id, q, cost);
272 id
273 }
274
275 fn issue(&mut self, w: &str, code: &str, mut q: i64) -> (Vec<Part>, i64) {
276 let mut parts = Vec::new();
277 let mut cost = 0;
278 for id in self.fifo(w, code) {
279 if q == 0 {
280 break;
281 }
282 let stocks = self.stock.get_mut(w).unwrap().get_mut(code).unwrap();
283 let current = stocks.get_mut(&id).unwrap();
284 let taken = q.min(current.quantity);
285 parts.push(Part { lot: id, quantity: taken, cost: current.cost });
286 cost += taken * current.cost;
287 current.quantity -= taken;
288 q -= taken;
289 if current.quantity == 0 {
290 stocks.remove(&id);
291 }
292 }
293 (parts, cost)
294 }
295
296 fn fulfil(&mut self, index: usize, q: i64, kind: &str) -> i64 {
297 if q == 0 {
298 return 0;
299 }
300 let (w, code, price) = {
301 let order = &self.orders[index];
302 (order.warehouse.clone(), order.item.clone(), order.price)
303 };
304 let (parts, cost) = self.issue(&w, &code, q);
305 {
306 let order = &mut self.orders[index];
307 order.shipped += q;
308 order.backorder -= q;
309 order.revenue += q * price;
310 order.cogs += cost;
311 order.issues.extend(parts);
312 }
313 self.accumulate("shipped", q, cost);
314 self.card_entry(&code, kind, &w, -q, -cost);
315 cost
316 }
317
318 fn allocate(&mut self, w: &str, code: &str) {
319 let mut available = self.quantity(w, code);
320 for index in 0..self.orders.len() {
321 if available == 0 {
322 break;
323 }
324 let current = &self.orders[index];
325 if current.warehouse != w || current.item != code || current.backorder == 0 {
326 continue;
327 }
328 let q = available.min(current.backorder);
329 let revenue = q * current.price;
330 let cost = self.fulfil(index, q, "fill");
331 available -= q;
332 line!(self, 2, "FILL", "order", self.orders[index].number, "qty", q,
333 "revenue", money(revenue), "cogs", money(cost),
334 "profit", money(revenue - cost), "backorder", self.orders[index].backorder);
335 }
336 }
337
338 fn remove_parts(&mut self, parts: &[Part]) {
339 for part in parts {
340 self.lots[part.lot - 1].removed += part.quantity;
341 }
342 }
343
344 fn sales(&self) -> BTreeMap<String, Sale> {
345 let mut rows = BTreeMap::<String, Sale>::new();
346 for order in &self.orders {
347 let row = rows.entry(order.item.clone()).or_default();
348 row.quantity += order.shipped - order.returned;
349 row.revenue += order.revenue;
350 row.cogs += order.cogs;
351 }
352 rows
353 }
354
355 fn margin_line(&mut self, label: &str, row: &Sale) {
356 let profit = row.revenue - row.cogs;
357 line!(self, 2, label, "sold", row.quantity, "revenue", money(row.revenue),
358 "cogs", money(row.cogs), "profit", money(profit),
359 "margin", margin(profit, row.revenue));
360 }
361
362 fn aging_line(&mut self, label: &str, buckets: &[i64; 4], monetary: bool) {
363 let labels = ["0-30", "31-60", "61-90", "91+"];
364 let mut text = label.to_string();
365 for i in 0..4 {
366 let value = if monetary { money(buckets[i]) } else { buckets[i].to_string() };
367 text.push_str(&format!(" {} {}", labels[i], value));
368 }
369 line!(self, 2, text);
370 }

372 fn command(&mut self, cmd: &str, a: &[&str]) -> Result<(), String> {
373 match cmd {
374 "DAY" => {
375 let day = number(a[0]);
376 if day <= self.day {
377 return Err(format!("invalid day {}", a[0]));
378 }
379 self.day = day;
380 line!(self, 0, "DAY", day);
381 }
382 "WAREHOUSE" => {
383 let w = a[0];
384 if self.stock.contains_key(w) {
385 return Err(format!("duplicate warehouse {w}"));
386 }
387 self.stock.insert(w.into(), BTreeMap::new());
388 line!(self, 0, "WAREHOUSE", w);
389 }
390 "ITEM" => {
391 let (code, level, name) = (a[0], number(a[1]), a[2]);
392 let action;
393 if let Some(entry) = self.items.get_mut(code) {
394 entry.level = level;
395 entry.name = name.into();
396 action = "updated";
397 } else {
398 self.items.insert(code.into(), Item {
399 code: code.into(),
400 name: name.into(),
401 level,
402 last_cost: None,
403 card: Vec::new(),
404 });
405 action = "added";
406 }
407 line!(self, 0, "ITEM", code, action, name, "level", level);
408 }
409 "RECEIVE" => {
410 let (w, code) = (a[0], a[1]);
411 self.pair(w, code)?;
412 let (q, cost, day) = (number(a[2]), cents(a[3]), number(a[4]));
413 if q == 0 {
414 return Err("invalid quantity".into());
415 }
416 if day > self.day {
417 return Err(format!("future day {}", a[4]));
418 }
419 let id = self.create_lot(w, code, day, q, cost, "receive");
420 self.items.get_mut(code).unwrap().last_cost = Some(cost);
421 self.accumulate("received", q, q * cost);
422 self.card_entry(code, "receive", w, q, q * cost);
423 line!(self, 0, "RECEIVE", w, code, "lot", id, "qty", q, "cost",
424 money(cost), "day", day, "value", money(q * cost));
425 self.allocate(w, code);
426 }
427 "SHIP" => {
428 let (w, code) = (a[0], a[1]);
429 self.pair(w, code)?;
430 let q = number(a[2]);
431 if q == 0 {
432 return Err("invalid quantity".into());
433 }
434 let id = self.orders.len() + 1;
435 self.orders.push(Order {
436 number: id,
437 warehouse: w.into(),
438 item: code.into(),
439 day: self.day,
440 price: cents(a[3]),
441 ordered: q,
442 shipped: 0,
443 backorder: q,
444 cancelled: 0,
445 returned: 0,
446 revenue: 0,
447 cogs: 0,
448 issues: Vec::new(),
449 });
450 let shipped = q.min(self.quantity(w, code));
451 let cost = self.fulfil(id - 1, shipped, "ship");
452 let current = &self.orders[id - 1];
453 line!(self, 0, "SHIP", "order", id, w, code, "shipped", shipped,
454 "revenue", money(current.revenue), "cogs", money(cost),
455 "profit", money(current.revenue - cost), "backorder", current.backorder);
456 }
457 "CANCEL" => {
458 let index = self.order_index(a[0])?;
459 let current = &mut self.orders[index];
460 if current.backorder == 0 {
461 return Err(format!("no backorder {}", a[0]));
462 }
463 current.cancelled += current.backorder;
464 current.backorder = 0;
465 line!(self, 0, "CANCEL", "order", current.number, "qty", current.cancelled);
466 }
467 "TRANSFER" => {
468 let (source, destination, code) = (a[0], a[1], a[2]);
469 self.warehouse(source)?;
470 self.warehouse(destination)?;
471 if source == destination {
472 return Err(format!("same warehouse {source}"));
473 }
474 self.item(code)?;
475 let q = number(a[3]);
476 if q == 0 {
477 return Err("invalid quantity".into());
478 }
479 self.require_stock(source, code, q)?;
480 let (parts, cost) = self.issue(source, code, q);
481 for part in &parts {
482 self.add_stock(destination, part.lot, part.quantity, part.cost);
483 }
484 self.card_entry(code, "transfer-out", source, -q, -cost);
485 self.card_entry(code, "transfer-in", destination, q, cost);
486 line!(self, 0, "TRANSFER", source, "->", destination, code,
487 "qty", q, "value", money(cost), "lots", parts.len());
488 self.allocate(destination, code);
489 }
490 "ADJUST" => {
491 let (w, code, change) = (a[0], a[1], a[2]);
492 self.pair(w, code)?;
493 let q = number(&change[1..]);
494 if q == 0 {
495 return Err("invalid quantity".into());
496 }
497 if change.starts_with('+') {
498 let cost = cents(a[3]);
499 let id = self.create_lot(w, code, self.day, q, cost, "adjust");
500 self.accumulate("adjusted-in", q, q * cost);
501 self.card_entry(code, "adjust", w, q, q * cost);
502 line!(self, 0, "ADJUST", w, code, format!("+{q}"),
503 "lot", id, "value", money(q * cost));
504 self.allocate(w, code);
505 } else {
506 self.require_stock(w, code, q)?;
507 let (parts, cost) = self.issue(w, code, q);
508 self.remove_parts(&parts);
509 self.accumulate("adjusted-out", q, cost);
510 self.card_entry(code, "adjust", w, -q, -cost);
511 line!(self, 0, "ADJUST", w, code, format!("-{q}"), "value", money(cost));
512 }
513 }
514 "RETURN" => {
515 let index = self.order_index(a[0])?;
516 let (q, cost) = (number(a[1]), cents(a[2]));
517 if q == 0 {
518 return Err("invalid quantity".into());
519 }
520 let current = &self.orders[index];
521 let available = current.shipped - current.returned;
522 if q > available {
523 return Err(format!("return exceeds shipped {} {available}", a[0]));
524 }
525 let (w, code, price, id) = (
526 current.warehouse.clone(), current.item.clone(),
527 current.price, current.number,
528 );
529 let lot = self.create_lot(&w, &code, self.day, q, cost, "return");
530 let refund = q * price;
531 {
532 let order = &mut self.orders[index];
533 order.returned += q;
534 order.revenue -= refund;
535 order.cogs -= q * cost;
536 }
537 self.accumulate("returned", q, q * cost);
538 self.card_entry(&code, "return", &w, q, q * cost);
539 line!(self, 0, "RETURN", "order", id, &w, &code, "qty", q,
540 "lot", lot, "refund", money(refund), "value", money(q * cost));
541 self.allocate(&w, &code);
542 }
543 "COUNT" => {
544 let (w, code) = (a[0], a[1]);
545 self.pair(w, code)?;
546 let (actual, held) = (number(a[2]), self.quantity(w, code));
547 if actual == held {
548 line!(self, 0, "COUNT", w, code, "qty", actual, "ok");
549 } else if actual < held {
550 let q = held - actual;
551 let (parts, cost) = self.issue(w, code, q);
552 self.remove_parts(&parts);
553 self.accumulate("adjusted-out", q, cost);
554 self.card_entry(code, "count", w, -q, -cost);
555 line!(self, 0, "COUNT", w, code, "qty", held, "->", actual,
556 "shortage", q, "value", money(cost));
557 } else {
558 let q = actual - held;
559 let cost = self.items[code].last_cost.unwrap_or(0);
560 let lot = self.create_lot(w, code, self.day, q, cost, "count");
561 self.accumulate("adjusted-in", q, q * cost);
562 self.card_entry(code, "count", w, q, q * cost);
563 line!(self, 0, "COUNT", w, code, "qty", held, "->", actual,
564 "surplus", q, "lot", lot, "value", money(q * cost));
565 self.allocate(w, code);
566 }
567 }
568 "MARKET" => {
569 let code = a[0];
570 self.item(code)?;
571 let price = cents(a[1]);
572 let mut changed = 0;
573 let mut q = 0;
574 let mut loss = 0;
575 let warehouses: Vec<String> = self.stock.keys().cloned().collect();
576 for w in warehouses {
577 let mut warehouse_loss = 0;
578 if let Some(stocks) = self.stock.get_mut(&w).and_then(|v| v.get_mut(code)) {
579 for stock in stocks.values_mut() {
580 if stock.cost > price {
581 warehouse_loss += stock.quantity * (stock.cost - price);
582 changed += 1;
583 q += stock.quantity;
584 stock.cost = price;
585 }
586 }
587 }
588 if warehouse_loss > 0 {
589 self.card_entry(code, "writedown", &w, 0, -warehouse_loss);
590 }
591 loss += warehouse_loss;
592 }
593 self.accumulate("written-down", q, loss);
594 line!(self, 0, "MARKET", code, "price", money(price),
595 "lots", changed, "qty", q, "loss", money(loss));
596 }
597 "RECALL" => {
598 let index = self.lot_index(a[0])?;
599 let (id, code) = (self.lots[index].number, self.lots[index].item.clone());
600 let mut held = Vec::<(String, i64, i64)>::new();
601 for w in self.stock.keys() {
602 if let Some(stock) = self.stock_lots(w, &code).and_then(|s| s.get(&id)) {
603 held.push((w.clone(), stock.quantity, stock.cost));
604 }
605 }
606 if held.is_empty() {
607 return Err(format!("nothing to recall {}", a[0]));
608 }
609 let mut q = 0;
610 let mut value = 0;
611 for (w, quantity, cost) in &held {
612 q += quantity;
613 value += quantity * cost;
614 self.stock.get_mut(w).unwrap().get_mut(&code).unwrap().remove(&id);
615 self.card_entry(&code, "recall", w, -*quantity, -(quantity * cost));
616 }
617 self.lots[index].removed += q;
618 self.accumulate("recalled", q, value);
619 line!(self, 0, "RECALL", "lot", id, &code, "qty", q, "value", money(value));
620 for (w, quantity, cost) in held {
621 line!(self, 2, w, "qty", quantity, "value", money(quantity * cost));
622 }
623 }
624 "EXPIRE" => {
625 let days = number(a[0]);
626 line!(self, 0, "EXPIRE", days);
627 let mut total_q = 0;
628 let mut total_value = 0;
629 let codes: Vec<String> = self.items.keys().cloned().collect();
630 let warehouses: Vec<String> = self.stock.keys().cloned().collect();
631 for code in codes {
632 for w in &warehouses {
633 let expired: Vec<usize> = self.stock_lots(w, &code)
634 .map(|stocks| stocks.keys().copied()
635 .filter(|&id| self.day - self.lots[id - 1].day > days).collect())
636 .unwrap_or_default();
637 if expired.is_empty() {
638 continue;
639 }
640 let mut q = 0;
641 let mut value = 0;
642 for id in &expired {
643 let stock = self.stock.get_mut(w).unwrap()
644 .get_mut(&code).unwrap().remove(id).unwrap();
645 q += stock.quantity;
646 value += stock.quantity * stock.cost;
647 self.lots[id - 1].removed += stock.quantity;
648 }
649 self.accumulate("expired", q, value);
650 self.card_entry(&code, "expire", w, -q, -value);
651 total_q += q;
652 total_value += value;
653 line!(self, 2, &code, w, "lots", expired.len(), "qty",
654 q, "value", money(value));
655 }
656 }
657 line!(self, 2, "total", "qty", total_q, "value", money(total_value));
658 }
659 "VALUE" | "AGING" => {
660 let warehouses: Vec<String> = if a.is_empty() {
661 self.stock.keys().cloned().collect()
662 } else {
663 self.warehouse(a[0])?;
664 vec![a[0].to_string()]
665 };
666 if cmd == "VALUE" {
667 if a.is_empty() {
668 line!(self, 0, "VALUE");
669 } else {
670 line!(self, 0, "VALUE", a[0]);
671 }
672 } else if a.is_empty() {
673 line!(self, 0, "AGING", "day", self.day);
674 } else {
675 line!(self, 0, "AGING", a[0], "day", self.day);
676 }
677 let mut total_q = 0;
678 let mut total_value = 0;
679 let mut bucket_q = [0i64; 4];
680 let mut bucket_v = [0i64; 4];
681 let codes: Vec<String> = self.items.keys().cloned().collect();
682 for code in codes {
683 let mut q = 0;
684 let mut value = 0;
685 let mut values = Vec::<(i64, i64)>::new();
686 let mut item_buckets = [0i64; 4];
687 for w in &warehouses {
688 let mut held_q = 0;
689 let mut held_value = 0;
690 if let Some(stocks) = self.stock_lots(w, &code) {
691 for (&id, stock) in stocks {
692 held_q += stock.quantity;
693 held_value += stock.quantity * stock.cost;
694 if cmd == "AGING" {
695 let age = self.day - self.lots[id - 1].day;
696 let bucket = if age <= 30 { 0 } else if age <= 60 { 1 }
697 else if age <= 90 { 2 } else { 3 };
698 item_buckets[bucket] += stock.quantity;
699 bucket_q[bucket] += stock.quantity;
700 bucket_v[bucket] += stock.quantity * stock.cost;
701 }
702 }
703 }
704 q += held_q;
705 value += held_value;
706 values.push((held_q, held_value));
707 }
708 if q == 0 {
709 continue;
710 }
711 if cmd == "VALUE" {
712 line!(self, 2, &code, "qty", q, "value", money(value),
713 "avg", money(value / q));
714 if a.is_empty() {
715 for (i, (held_q, held_value)) in values.iter().enumerate() {
716 if *held_q > 0 {
717 line!(self, 4, &warehouses[i], "qty", held_q,
718 "value", money(*held_value));
719 }
720 }
721 }
722 } else {
723 self.aging_line(&code, &item_buckets, false);
724 }
725 total_q += q;
726 total_value += value;
727 }
728 if cmd == "VALUE" {
729 line!(self, 2, "total", "qty", total_q, "value", money(total_value));
730 } else {
731 self.aging_line("total", &bucket_q, false);
732 self.aging_line("value", &bucket_v, true);
733 }
734 }
735 "LOTS" => {
736 let (w, code) = (a[0], a[1]);
737 self.pair(w, code)?;
738 line!(self, 0, "LOTS", w, code);
739 let ids = self.fifo(w, code);
740 let pending: Vec<usize> = self.orders.iter().enumerate()
741 .filter(|(_, o)| o.warehouse == w && o.item == code && o.backorder > 0)
742 .map(|(i, _)| i).collect();
743 for id in &ids {
744 let stock = &self.stock_lots(w, code).unwrap()[id];
745 line!(self, 2, "lot", id, "day", self.lots[id - 1].day,
746 "qty", stock.quantity, "cost", money(stock.cost),
747 "value", money(stock.quantity * stock.cost),
748 "age", self.day - self.lots[id - 1].day);
749 }
750 for index in &pending {
751 let order = &self.orders[*index];
752 line!(self, 2, "backorder", "order", order.number, "qty", order.backorder);
753 }
754 if ids.is_empty() && pending.is_empty() {
755 line!(self, 2, "(none)");
756 }
757 }
758 "REORDER" | "ITEMS" => {
759 line!(self, 0, cmd);
760 let mut found = false;
761 let codes: Vec<String> = self.items.keys().cloned().collect();
762 for code in codes {
763 let entry = &self.items[&code];
764 let q = self.total_quantity(&code);
765 let pending = self.backorder(&code);
766 if cmd == "REORDER" {
767 if q - pending > entry.level {
768 continue;
769 }
770 line!(self, 2, &code, &entry.name, "onhand", q,
771 "backorder", pending, "level", entry.level);
772 } else {
773 let cost = entry.last_cost.map(money).unwrap_or_else(|| "none".into());
774 line!(self, 2, &code, &entry.name, "level", entry.level,
775 "onhand", q, "backorder", pending, "cost", cost);
776 }
777 found = true;
778 }
779 if !found {
780 line!(self, 2, "(none)");
781 }
782 }
783 "MARGIN" | "TOP" => {
784 let rows = self.sales();
785 if cmd == "MARGIN" {
786 line!(self, 0, "MARGIN");
787 let mut total = Sale::default();
788 for (code, row) in &rows {
789 self.margin_line(code, row);
790 total.quantity += row.quantity;
791 total.revenue += row.revenue;
792 total.cogs += row.cogs;
793 }
794 self.margin_line("total", &total);
795 } else {
796 let k = number(a[0]) as usize;
797 line!(self, 0, "TOP", k);
798 let mut codes: Vec<&String> = rows.keys().collect();
799 codes.sort_by(|a, b| {
800 let x = &rows[*a];
801 let y = &rows[*b];
802 (y.revenue - y.cogs).cmp(&(x.revenue - x.cogs))
803 .then(y.revenue.cmp(&x.revenue))
804 .then(a.cmp(b))
805 });
806 for (rank, &code) in codes.iter().take(k).enumerate() {
807 let row = &rows[code];
808 let profit = row.revenue - row.cogs;
809 line!(self, 2, rank + 1, code, "profit", money(profit),
810 "revenue", money(row.revenue), "margin", margin(profit, row.revenue));
811 }
812 if codes.is_empty() {
813 line!(self, 2, "(none)");
814 }
815 }
816 }
817 "BACKORDERS" => {
818 line!(self, 0, "BACKORDERS");
819 let mut found = false;
820 for order in &self.orders {
821 if order.backorder > 0 {
822 line!(self, 2, "order", order.number, &order.warehouse, &order.item,
823 "qty", order.backorder, "day", order.day, "price", money(order.price));
824 found = true;
825 }
826 }
827 if !found {
828 line!(self, 2, "(none)");
829 }
830 }
831 "ORDER" => {
832 let index = self.order_index(a[0])?;
833 let order = &self.orders[index];
834 line!(self, 0, "ORDER", order.number, &order.warehouse, &order.item,
835 "day", order.day, "price", money(order.price));
836 line!(self, 2, "ordered", order.ordered, "shipped", order.shipped,
837 "backorder", order.backorder, "cancelled", order.cancelled,
838 "returned", order.returned);
839 for part in &order.issues {
840 line!(self, 2, "lot", part.lot, "qty", part.quantity, "cost", money(part.cost));
841 }
842 line!(self, 2, "revenue", money(order.revenue), "cogs",
843 money(order.cogs), "profit", money(order.revenue - order.cogs));
844 }
845 "TRACE" => {
846 let index = self.lot_index(a[0])?;
847 let lot = &self.lots[index];
848 line!(self, 0, "TRACE", "lot", lot.number, &lot.item, &lot.kind,
849 &lot.warehouse, "day", lot.day, "qty", lot.initial_quantity,
850 "cost", money(lot.initial_cost));
851 for w in self.stock.keys() {
852 if let Some(stock) = self.stock_lots(w, &lot.item).and_then(|s| s.get(&lot.number)) {
853 line!(self, 2, "held", w, "qty", stock.quantity, "cost", money(stock.cost));
854 }
855 }
856 for order in &self.orders {
857 let q: i64 = order.issues.iter().filter(|p| p.lot == lot.number)
858 .map(|p| p.quantity).sum();
859 if q > 0 {
860 line!(self, 2, "order", order.number, "qty", q);
861 }
862 }
863 if lot.removed > 0 {
864 line!(self, 2, "removed", "qty", lot.removed);
865 }
866 }
867 "CARD" => {
868 self.item(a[0])?;
869 let entry = &self.items[a[0]];
870 line!(self, 0, "CARD", &entry.code, &entry.name);
871 let mut q = 0;
872 let mut value = 0;
873 for card in &entry.card {
874 q += card.quantity;
875 value += card.value;
876 line!(self, 2, "day", card.day, &card.kind, &card.warehouse,
877 signed_quantity(card.quantity), signed_money(card.value),
878 "balance", q, money(value));
879 }
880 if entry.card.is_empty() {
881 line!(self, 2, "(none)");
882 }
883 }
884 "STATUS" => {
885 line!(self, 0, "STATUS");
886 for (w, contents) in &self.stock {
887 let mut skus = 0;
888 let mut q = 0;
889 let mut value = 0;
890 for stocks in contents.values() {
891 if !stocks.is_empty() {
892 skus += 1;
893 }
894 for stock in stocks.values() {
895 q += stock.quantity;
896 value += stock.quantity * stock.cost;
897 }
898 }
899 let pending: i64 = self.orders.iter()
900 .filter(|o| o.warehouse == *w).map(|o| o.backorder).sum();
901 line!(self, 2, w, "skus", skus, "qty", q,
902 "value", money(value), "backorder", pending);
903 }
904 if self.stock.is_empty() {
905 line!(self, 2, "(none)");
906 }
907 }
908 "SUMMARY" => {
909 line!(self, 0, "SUMMARY");
910 for category in CATEGORIES {
911 let total = &self.totals[category];
912 line!(self, 2, category, "qty", total.quantity, "value", money(total.value));
913 }
914 let total = self.all_stocks();
915 line!(self, 2, "on-hand", "qty", total.quantity, "value", money(total.value));
916 let open = self.orders.iter().filter(|o| o.backorder > 0).count();
917 let pending: i64 = self.orders.iter().map(|o| o.backorder).sum();
918 line!(self, 2, "orders", self.orders.len(), "open", open, "backorder", pending);
919 }
920 _ => {}
921 }
922 Ok(())
923 }

925 fn execute(&mut self, line_text: &str) {
926 let mut parts = line_text.split_whitespace();
927 let cmd = parts.next().unwrap_or("");
928 let arguments: Vec<&str> = parts.collect();
929 if let Err(reason) = self.command(cmd, &arguments) {
930 line!(self, 0, "ERROR", reason);
931 }
932 }
933}

935fn main() {
936 let mut input = String::new();
937 io::stdin().read_to_string(&mut input).unwrap();
938 let mut lines = input.lines();
939 let n: usize = lines.next().unwrap_or("0").parse().unwrap();
940 let mut inventory = Inventory::new();
941 for line in lines.take(n) {
942 inventory.execute(line);
943 }
944 if !inventory.output.is_empty() {
945 println!("{}", inventory.output.join("\n"));
946 }
947}
