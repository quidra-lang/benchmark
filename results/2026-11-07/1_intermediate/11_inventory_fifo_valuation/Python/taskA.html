10def cents(text):
11 whole, fraction = text.split(".")
12 return 100 * int(whole) + int(fraction)

15def money(value):
16 sign = "-" if value < 0 else ""
17 value = abs(value)
18 return f"{sign}{value // 100}.{value % 100:02d}"

21def signed_quantity(value):
22 return "+" + str(value) if value > 0 else str(value)

25def signed_money(value):
26 return "+" + money(value) if value > 0 else money(value)

29def margin(profit, revenue):
30 if revenue == 0:
31 return "n/a"
32 tenths = abs(profit) * 1000 // revenue
33 sign = "-" if profit < 0 and tenths else ""
34 return f"{sign}{tenths // 10}.{tenths % 10}%"

172 def fifo(self, warehouse, item):
173 return sorted(self.holdings(warehouse, item).values(), key=lambda stock: (stock.lot.day, stock.lot.number))

175 def issue(self, warehouse, item, quantity):
176 parts = []
177 cost = 0
178 holdings = self.holdings(warehouse, item)
179 for stock in self.fifo(warehouse, item):
180 if quantity == 0:
181 break
182 taken = min(quantity, stock.quantity)
183 parts.append((stock.lot, taken, stock.cost))
184 cost += taken * stock.cost
185 stock.quantity -= taken
186 quantity -= taken
187 if stock.quantity == 0:
188 del holdings[stock.lot.number]
189 return parts, cost

191 def fulfil(self, order, quantity, kind):
192 if quantity == 0:
193 return 0
194 parts, cost = self.issue(order.warehouse, order.item, quantity)
195 order.shipped += quantity
196 order.backorder -= quantity
197 order.revenue += quantity * order.price
198 order.cogs += cost
199 order.issues.extend((lot.number, taken, unit_cost) for lot, taken, unit_cost in parts)
200 self.accumulate("shipped", quantity, cost)
201 self.card_entry(order.item, kind, order.warehouse, -quantity, -cost)
202 return cost

204 def allocate(self, warehouse, item):
205 available = self.quantity(warehouse, item)
206 for order in self.orders:
207 if available == 0:
208 break
209 if order.warehouse != warehouse or order.item != item or order.backorder == 0:
210 continue
211 filled = min(available, order.backorder)
212 cost = self.fulfil(order, filled, "fill")
213 revenue = filled * order.price
214 available -= filled
215 self.emit(
216 "FILL", "order", order.number, "qty", filled,
217 "revenue", money(revenue), "cogs", money(cost),
218 "profit", money(revenue - cost), "backorder", order.backorder, indent=2,
219 )

251 def receive(self, arguments):
252 warehouse, item, quantity, cost, day = arguments
253 product = self.pair(warehouse, item)
254 quantity, cost, day = int(quantity), cents(cost), int(day)
255 self.require_quantity(quantity)
256 if day > self.day:
257 raise CommandError(f"future day {arguments[4]}")
258 lot = self.create_lot(warehouse, item, day, quantity, cost, "receive")
259 product.last_cost = cost
260 self.accumulate("received", quantity, quantity * cost)
261 self.card_entry(item, "receive", warehouse, quantity, quantity * cost)
262 self.emit(
263 "RECEIVE", warehouse, item, "lot", lot.number, "qty", quantity,
264 "cost", money(cost), "day", day, "value", money(quantity * cost),
265 )
266 self.allocate(warehouse, item)

291 def transfer(self, arguments):
292 source, destination, item, quantity = arguments
293 self.warehouse(source)
294 self.warehouse(destination)
295 if source == destination:
296 raise CommandError(f"same warehouse {source}")
297 self.item(item)
298 quantity = int(quantity)
299 self.require_quantity(quantity)
300 self.require_stock(source, item, quantity)
301 parts, cost = self.issue(source, item, quantity)
302 for lot, taken, unit_cost in parts:
303 self.add_stock(destination, lot, taken, unit_cost)
304 self.card_entry(item, "transfer-out", source, -quantity, -cost)
305 self.card_entry(item, "transfer-in", destination, quantity, cost)
306 self.emit(
307 "TRANSFER", source, "->", destination, item, "qty", quantity,
308 "value", money(cost), "lots", len(parts),
309 )
310 self.allocate(destination, item)

312 def adjust(self, arguments):
313 warehouse, item, change = arguments[:3]
314 self.pair(warehouse, item)
315 quantity = int(change[1:])
316 self.require_quantity(quantity)
317 if change[0] == "+":
318 cost = cents(arguments[3])
319 lot = self.create_lot(warehouse, item, self.day, quantity, cost, "adjust")
320 self.accumulate("adjusted-in", quantity, quantity * cost)
321 self.card_entry(item, "adjust", warehouse, quantity, quantity * cost)
322 self.emit("ADJUST", warehouse, item, "+" + str(quantity), "lot", lot.number, "value", money(quantity * cost))
323 self.allocate(warehouse, item)
324 else:
325 self.require_stock(warehouse, item, quantity)
326 parts, cost = self.issue(warehouse, item, quantity)
327 self.remove_parts(parts)
328 self.accumulate("adjusted-out", quantity, cost)
329 self.card_entry(item, "adjust", warehouse, -quantity, -cost)
330 self.emit("ADJUST", warehouse, item, "-" + str(quantity), "value", money(cost))

332 def return_stock(self, arguments):
333 order = self.order(arguments[0])
334 quantity, cost = int(arguments[1]), cents(arguments[2])
335 self.require_quantity(quantity)
336 available = order.shipped - order.returned
337 if quantity > available:
338 raise CommandError(f"return exceeds shipped {arguments[0]} {available}")

340 lot = self.create_lot(warehouse, item, self.day, quantity, cost, "return")
341 order.returned += quantity
342 order.revenue -= quantity * order.price
343 order.cogs -= quantity * cost
344 self.accumulate("returned", quantity, quantity * cost)
345 self.card_entry(item, "return", warehouse, quantity, quantity * cost)
346 self.emit(
347 "RETURN", "order", order.number, warehouse, item, "qty", quantity,
348 "lot", lot.number, "refund", money(quantity * order.price), "value", money(quantity * cost),
349 )
350 self.allocate(warehouse, item)

352 def count(self, arguments):
353 warehouse, item, actual = arguments
354 product = self.pair(warehouse, item)
355 actual = int(actual)
356 held = self.quantity(warehouse, item)
357 if actual == held:
358 self.emit("COUNT", warehouse, item, "qty", actual, "ok")
359 elif actual < held:
360 shortage = held - actual
361 parts, cost = self.issue(warehouse, item, shortage)
362 self.remove_parts(parts)
363 self.accumulate("adjusted-out", shortage, cost)
364 self.card_entry(item, "count", warehouse, -shortage, -cost)
365 self.emit("COUNT", warehouse, item, "qty", held, "->", actual, "shortage", shortage, "value", money(cost))
366 else:
367 surplus = actual - held
368 cost = product.last_cost if product.last_cost is not None else 0
369 lot = self.create_lot(warehouse, item, self.day, surplus, cost, "count")
370 self.accumulate("adjusted-in", surplus, surplus * cost)
371 self.card_entry(item, "count", warehouse, surplus, surplus * cost)
372 self.emit(
373 "COUNT", warehouse, item, "qty", held, "->", actual,
374 "surplus", surplus, "lot", lot.number, "value", money(surplus * cost),
375 )
376 self.allocate(warehouse, item)

378 def market(self, arguments):
379 item, price = arguments
380 self.item(item)
381 price = cents(price)
382 lots_changed = 0
383 quantity = 0
384 loss = 0
385 for warehouse in sorted(self.stock):
386 warehouse_loss = 0
387 for stock in self.holdings(warehouse, item).values():
388 if stock.cost > price:
389 decrease = stock.quantity * (stock.cost - price)
390 warehouse_loss += decrease
391 lots_changed += 1
392 quantity += stock.quantity
393 stock.cost = price
394 if warehouse_loss:
395 self.card_entry(item, "writedown", warehouse, 0, -warehouse_loss)
396 loss += warehouse_loss
397 self.accumulate("written-down", quantity, loss)
398 self.emit("MARKET", item, "price", money(price), "lots", lots_changed, "qty", quantity, "loss", money(loss))

400 def recall(self, arguments):
401 lot = self.lot(arguments[0])
402 held = [
403 (warehouse, self.holdings(warehouse, lot.item)[lot.number])
404 for warehouse in sorted(self.stock) if lot.number in self.holdings(warehouse, lot.item)
405 ]
406 if not held:
407 raise CommandError(f"nothing to recall {arguments[0]}")
408 quantity = sum(stock.quantity for _, stock in held)
409 value = sum(stock.value() for _, stock in held)
410 for warehouse, stock in held:
411 del self.holdings(warehouse, lot.item)[lot.number]
412 self.card_entry(lot.item, "recall", warehouse, -stock.quantity, -stock.value())
413 lot.removed += quantity
414 self.accumulate("recalled", quantity, value)
415 self.emit("RECALL", "lot", lot.number, lot.item, "qty", quantity, "value", money(value))
416 for warehouse, stock in held:
417 self.emit(warehouse, "qty", stock.quantity, "value", money(stock.value()), indent=2)

419 def expire(self, arguments):
420 days = int(arguments[0])
421 self.emit("EXPIRE", days)
422 total_quantity = 0
423 total_value = 0
424 for item in sorted(self.items):
425 for warehouse in sorted(self.stock):
426 holdings = self.holdings(warehouse, item)
427 expired = [stock for stock in holdings.values() if self.day - stock.lot.day > days]
428 if not expired:
429 continue
430 quantity = sum(stock.quantity for stock in expired)
431 value = sum(stock.value() for stock in expired)
432 for stock in expired:
433 stock.lot.removed += stock.quantity
434 del holdings[stock.lot.number]
435 self.accumulate("expired", quantity, value)
436 self.card_entry(item, "expire", warehouse, -quantity, -value)
437 total_quantity += quantity
438 total_value += value
439 self.emit(item, warehouse, "lots", len(expired), "qty", quantity, "value", money(value), indent=2)
440 self.emit("total", "qty", total_quantity, "value", money(total_value), indent=2)

554 total_quantities = [0, 0, 0, 0]

557 quantities = [0, 0, 0, 0]

564 total_values[bucket] += stock.value()
565 if sum(quantities):
566 self.aging_line(item, quantities)

568 self.aging_line("value", total_values, True)

638 self.emit("SUMMARY")
639 for category in CATEGORIES:
640 quantity, value = self.totals[category]
641 self.emit(category, "qty", quantity, "value", money(value), indent=2)

651 handlers = {
