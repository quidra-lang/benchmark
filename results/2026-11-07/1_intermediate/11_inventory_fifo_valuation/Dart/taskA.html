1import 'dart:io';
2import 'dart:math' as math;

4int amount(String s) {
5 final p = s.split('.');
6 return int.parse(p[0]) * 100 + int.parse(p[1]);
7}
8String money(int c) => '${c < 0 ? '-' : ''}${c.abs() ~/ 100}.${(c.abs() % 100).toString().padLeft(2, '0')}';
9String signed(int n) => '${n > 0 ? '+' : ''}$n';
10String signedMoney(int n) => '${n > 0 ? '+' : ''}${money(n)}';
11String margin(int profit, int revenue) {
12 if (revenue == 0) return 'n/a';
13 final q = profit.abs() * 1000 ~/ revenue;
14 return '${profit < 0 && q > 0 ? '-' : ''}${q ~/ 10}.${q % 10}%';
15}
16const categories = ['received', 'returned', 'adjusted-in', 'adjusted-out', 'shipped', 'written-down', 'expired', 'recalled'];
17class InventoryError implements Exception {
18 final String reason;
19 InventoryError(this.reason);
20}
21Never fail(String s) => throw InventoryError(s);

33class Item {
34 final String code;
35 String name;
36 int level, orderCount = 0, sold = 0, revenue = 0, cogs = 0;
37 int? lastCost;
38 final lots = <Lot>[];
39 final card = <CardEntry>[];
40 Item(this.code, this.level, this.name);
41}
42class Lot {
43 final int number, day, initialQuantity, initialCost;
44 final Item item;
45 final Warehouse warehouse;
46 final String kind;
47 int cost, removed = 0;
48 final held = List.filled(20, 0);
49 Lot(this.number, this.item, this.day, this.initialQuantity, this.initialCost, this.warehouse, this.kind) : cost = initialCost;
50}

54 Part(this.lot, this.quantity, this.cost);
55}
56class Issue {

58 final int cost;
59 Issue(this.parts, this.cost);

61class Order {
62 final int number, day, price, ordered;
63 final Warehouse warehouse;
64 final Item item;
65 int shipped = 0, backorder, cancelled = 0, returned = 0, revenue = 0, cogs = 0;
66 final issues = <Part>[];
67 Order(this.number, this.warehouse, this.item, this.day, this.price, this.ordered) : backorder = ordered;
68}
69class Inventory {
70 int day = 1;
71 final warehouses = <String, Warehouse>{};
72 final items = <String, Item>{};
73 final lots = <Lot>[];
74 final orders = <Order>[];
75 final quantities = List.filled(8, 0), values = List.filled(8, 0);
76 List<Warehouse> warehouseList() => warehouses.values.toList()..sort((a, b) => a.code.compareTo(b.code));
77 List<Item> itemList() => items.values.toList()..sort((a, b) => a.code.compareTo(b.code));
78 Warehouse warehouse(String s) => warehouses[s] ?? fail('unknown warehouse $s');
79 Item item(String s) => items[s] ?? fail('unknown item $s');
80 Order order(String s) {
81 final n = int.parse(s);
82 if (n < 1 || n > orders.length) fail('unknown order $s');
83 return orders[n - 1];
84 }
85 Lot lot(String s) {
86 final n = int.parse(s);
87 if (n < 1 || n > lots.length) fail('unknown lot $s');
88 return lots[n - 1];
89 }
90 (int, int) stock(Item item, [Warehouse? w]) {
91 var q = 0, v = 0;
92 for (final lot in item.lots) {
93 final held = w == null ? lot.held.fold(0, (a, b) => a + b) : lot.held[w.index];
94 q += held;
95 v += held * lot.cost;
96 }
97 return (q, v);
98 }
99 int backorder(Item item, [Warehouse? w]) => orders.where((o) => o.item == item && (w == null || o.warehouse == w)).fold(0, (n, o) => n + o.backorder);
100 void tally(int category, int quantity, int value) {
101 quantities[category] += quantity;
102 values[category] += value;
103 }
104 void card(Item item, String kind, Warehouse w, int q, int v) => item.card.add(CardEntry(day, kind, w, q, v));
105 Lot create(Item item, Warehouse w, int q, int cost, int lotDay, String kind) {
106 final lot = Lot(lots.length + 1, item, lotDay, q, cost, w, kind);
107 lot.held[w.index] = q;
108 lots.add(lot);
109 item.lots.add(lot);
110 item.lots.sort((a, b) {
111 final c = a.day.compareTo(b.day);
112 return c != 0 ? c : a.number.compareTo(b.number);
113 });
114 return lot;
115 }
116 Issue issue(Item item, Warehouse w, int q, {bool removed = false}) {
117 var cost = 0;
118 final parts = <Part>[];
119 for (final lot in item.lots) {
120 final take = math.min(q, lot.held[w.index]);
121 if (take == 0) continue;
122 parts.add(Part(lot, take, lot.cost));
123 lot.held[w.index] -= take;
124 q -= take;
125 cost += take * lot.cost;
126 if (removed) lot.removed += take;
127 if (q == 0) break;
128 }
129 return Issue(parts, cost);
130 }
131 void ship(Order o, int q, Issue issued) {
132 o.shipped += q;
133 o.backorder -= q;
134 o.revenue += q * o.price;
135 o.cogs += issued.cost;
136 o.issues.addAll(issued.parts);
137 o.item.sold += q;
138 o.item.revenue += q * o.price;
139 o.item.cogs += issued.cost;
140 if (q > 0) tally(4, q, issued.cost);
141 }
142 void allocate(Warehouse w, Item item, StringBuffer out) {
143 for (final o in orders) {
144 if (o.warehouse != w || o.item != item || o.backorder == 0) continue;
145 final q = math.min(o.backorder, stock(item, w).$1);
146 if (q == 0) break;
147 final issued = issue(item, w, q);
148 ship(o, q, issued);
149 card(item, 'fill', w, -q, -issued.cost);
150 out.writeln(' FILL order ${o.number} qty $q revenue ${money(q * o.price)} cogs ${money(issued.cost)} profit ${money(q * o.price - issued.cost)} backorder ${o.backorder}');
151 }
152 }
153 void requireQuantity(int q) {
154 if (q == 0) fail('invalid quantity');
155 }
156 void requireStock(Item item, Warehouse w, int q) {
157 final h = stock(item, w).$1;
158 if (h < q) fail('insufficient stock ${w.code} ${item.code} $h');
159 }
160 void execute(List<String> a, StringBuffer out) {
161 final op = a[0], id = a.length > 1 ? a[1] : '';
162 if (op == 'DAY') {
163 final d = int.parse(id);
164 if (d <= day) fail('invalid day $id');
165 day = d;
166 out.writeln('DAY $d');
167 } else if (op == 'WAREHOUSE') {
168 if (warehouses.containsKey(id)) fail('duplicate warehouse $id');
169 warehouses[id] = Warehouse(id, warehouses.length);
170 out.writeln('WAREHOUSE $id');
171 } else if (op == 'ITEM') {
172 final added = !items.containsKey(id), level = int.parse(a[2]);
173 final s = items.putIfAbsent(id, () => Item(id, level, a[3]));
174 s.level = level;
175 s.name = a[3];
176 out.writeln('ITEM $id ${added ? 'added' : 'updated'} ${s.name} level $level');
177 } else if (['RECEIVE', 'SHIP', 'ADJUST', 'COUNT'].contains(op)) {
178 final w = warehouse(id);
179 final s = item(a[2]);
180 final prefix = '$op $id ${a[2]}';
181 if (op == 'RECEIVE') {
182 final q = int.parse(a[3]), c = amount(a[4]), d = int.parse(a[5]);
183 requireQuantity(q);
184 if (d > day) fail('future day ${a[5]}');
185 final l = create(s, w, q, c, d, 'receive');
186 s.lastCost = c;
187 tally(0, q, q * c);
188 card(s, 'receive', w, q, q * c);
189 out.writeln('$prefix lot ${l.number} qty $q cost ${money(c)} day $d value ${money(q * c)}');
190 allocate(w, s, out);
191 } else if (op == 'SHIP') {
192 final q = int.parse(a[3]), price = amount(a[4]);
193 requireQuantity(q);
194 final o = Order(orders.length + 1, w, s, day, price, q);
195 orders.add(o);
196 s.orderCount++;
197 final f = math.min(q, stock(s, w).$1);
198 final issued = issue(s, w, f);
199 ship(o, f, issued);
200 if (f > 0) card(s, 'ship', w, -f, -issued.cost);
201 out.writeln('SHIP order ${o.number} $id ${s.code} shipped $f revenue ${money(o.revenue)} cogs ${money(o.cogs)} profit ${money(o.revenue - o.cogs)} backorder ${o.backorder}');
202 } else if (op == 'ADJUST') {
203 final positive = a[3][0] == '+', q = int.parse(a[3].substring(1));
204 requireQuantity(q);
205 if (positive) {
206 final c = amount(a[4]);
207 final l = create(s, w, q, c, day, 'adjust');
208 tally(2, q, q * c);
209 card(s, 'adjust', w, q, q * c);
210 out.writeln('$prefix +$q lot ${l.number} value ${money(q * c)}');
211 allocate(w, s, out);
212 } else {
213 requireStock(s, w, q);
214 final issued = issue(s, w, q, removed: true);
215 tally(3, q, issued.cost);
216 card(s, 'adjust', w, -q, -issued.cost);
217 out.writeln('$prefix -$q value ${money(issued.cost)}');
218 }
219 } else {
220 final actual = int.parse(a[3]), h = stock(s, w).$1;
221 if (actual == h) {
222 out.writeln('$prefix qty $actual ok');
223 } else if (actual < h) {
224 final q = h - actual;
225 final issued = issue(s, w, q, removed: true);
226 tally(3, q, issued.cost);
227 card(s, 'count', w, -q, -issued.cost);
228 out.writeln('$prefix qty $h -> $actual shortage $q value ${money(issued.cost)}');
229 } else {
230 final q = actual - h, c = s.lastCost ?? 0;
231 final l = create(s, w, q, c, day, 'count');
232 tally(2, q, q * c);
233 card(s, 'count', w, q, q * c);
234 out.writeln('$prefix qty $h -> $actual surplus $q lot ${l.number} value ${money(q * c)}');
235 allocate(w, s, out);
236 }
237 }
238 } else if (op == 'CANCEL') {
239 final o = order(id);
240 if (o.backorder == 0) fail('no backorder $id');
241 final q = o.backorder;
242 o.cancelled += q;
243 o.backorder = 0;
244 out.writeln('CANCEL order ${o.number} qty $q');
245 } else if (op == 'TRANSFER') {
246 final from = warehouse(id);
247 final to = warehouse(a[2]);
248 if (from == to) fail('same warehouse $id');
249 final s = item(a[3]), q = int.parse(a[4]);
250 requireQuantity(q);
251 requireStock(s, from, q);
252 final issued = issue(s, from, q);
253 for (final p in issued.parts) p.lot.held[to.index] += p.quantity;
254 card(s, 'transfer-out', from, -q, -issued.cost);
255 card(s, 'transfer-in', to, q, issued.cost);
256 out.writeln('TRANSFER $id -> ${to.code} ${s.code} qty $q value ${money(issued.cost)} lots ${issued.parts.length}');
257 allocate(to, s, out);
258 } else if (op == 'RETURN') {
259 final o = order(id), q = int.parse(a[2]);
260 requireQuantity(q);
261 if (q > o.shipped - o.returned) fail('return exceeds shipped $id ${o.shipped - o.returned}');
262 final c = amount(a[3]);
263 final l = create(o.item, o.warehouse, q, c, day, 'return');
264 o.returned += q;
265 o.revenue -= q * o.price;
266 o.cogs -= q * c;
267 o.item.sold -= q;
268 o.item.revenue -= q * o.price;
269 o.item.cogs -= q * c;
270 tally(1, q, q * c);
271 card(o.item, 'return', o.warehouse, q, q * c);
272 out.writeln('RETURN order ${o.number} ${o.warehouse.code} ${o.item.code} qty $q lot ${l.number} refund ${money(q * o.price)} value ${money(q * c)}');
273 allocate(o.warehouse, o.item, out);
274 } else if (op == 'MARKET') {
275 final s = item(id), price = amount(a[2]);
276 var changed = 0, q = 0, loss = 0;
277 for (final w in warehouseList()) {
278 var warehouseLoss = 0;
279 for (final l in s.lots) {
280 if (l.cost <= price || l.held[w.index] == 0) continue;
281 final h = l.held[w.index], value = h * (l.cost - price);
282 changed++;
283 q += h;
284 loss += value;
285 warehouseLoss += value;
286 }
287 if (warehouseLoss > 0) card(s, 'writedown', w, 0, -warehouseLoss);
288 }
289 for (final l in s.lots) if (l.cost > price && l.held.any((q) => q > 0)) l.cost = price;
290 tally(5, q, loss);
291 out.writeln('MARKET $id price ${money(price)} lots $changed qty $q loss ${money(loss)}');
292 } else if (op == 'RECALL') {
293 final l = lot(id), held = warehouseList().where((w) => l.held[w.index] > 0).toList();
294 if (held.isEmpty) fail('nothing to recall $id');
295 final q = held.fold(0, (s, w) => s + l.held[w.index]);
296 out.writeln('RECALL lot ${l.number} ${l.item.code} qty $q value ${money(q * l.cost)}');
297 for (final w in held) {
298 final h = l.held[w.index];
299 card(l.item, 'recall', w, -h, -h * l.cost);
300 out.writeln(' ${w.code} qty $h value ${money(h * l.cost)}');
301 l.held[w.index] = 0;
302 }
303 l.removed += q;
304 tally(7, q, q * l.cost);
305 } else if (op == 'EXPIRE') {
306 final age = int.parse(id);
307 var totalQ = 0, totalV = 0;
308 out.writeln('EXPIRE $age');
309 final ws = warehouseList();
310 for (final s in itemList()) {
311 for (final w in ws) {
312 var count = 0, q = 0, value = 0;
313 for (final l in s.lots) {
314 if (l.held[w.index] == 0 || day - l.day <= age) continue;
315 final h = l.held[w.index];
316 count++;
317 q += h;
318 value += h * l.cost;
319 l.removed += h;
320 l.held[w.index] = 0;
321 }
322 if (count > 0) {
323 tally(6, q, value);
324 card(s, 'expire', w, -q, -value);
325 totalQ += q;
326 totalV += value;
327 out.writeln(' ${s.code} ${w.code} lots $count qty $q value ${money(value)}');
328 }
329 }
330 }
331 out.writeln(' total qty $totalQ value ${money(totalV)}');
332 } else if (op == 'VALUE' || op == 'AGING') {
333 final w = a.length > 1 ? warehouse(id) : null;
334 final ws = warehouseList();
335 if (op == 'VALUE') {
336 out.writeln('VALUE${w == null ? '' : ' ${w.code}'}');
337 var totalQ = 0, totalV = 0;
338 for (final s in itemList()) {
339 final (q, value) = stock(s, w);
340 if (q == 0) continue;
341 out.writeln(' ${s.code} qty $q value ${money(value)} avg ${money(value ~/ q)}');
342 totalQ += q;
343 totalV += value;
344 if (w == null) {
345 for (final wh in ws) {
346 final h = stock(s, wh);
347 if (h.$1 > 0) out.writeln(' ${wh.code} qty ${h.$1} value ${money(h.$2)}');
348 }
349 }
350 }
351 out.writeln(' total qty $totalQ value ${money(totalV)}');
352 } else {
353 out.writeln('AGING${w == null ? '' : ' ${w.code}'} day $day');
354 final totalQ = List.filled(4, 0), totalV = List.filled(4, 0);
355 const labels = ['0-30', '31-60', '61-90', '91+'];
356 for (final s in itemList()) {
357 final qs = List.filled(4, 0);
358 for (final l in s.lots) {
359 final h = w == null ? l.held.fold(0, (a, b) => a + b) : l.held[w.index];
360 final age = day - l.day;
361 final bucket = age <= 30 ? 0 : age <= 60 ? 1 : age <= 90 ? 2 : 3;
362 qs[bucket] += h;
363 totalQ[bucket] += h;
364 totalV[bucket] += h * l.cost;
365 }
366 if (qs.any((q) => q > 0)) out.writeln(' ${s.code} ${List.generate(4, (k) => '${labels[k]} ${qs[k]}').join(' ')}');
367 }
368 out.writeln(' total ${List.generate(4, (k) => '${labels[k]} ${totalQ[k]}').join(' ')}');
369 out.writeln(' value ${List.generate(4, (k) => '${labels[k]} ${money(totalV[k])}').join(' ')}');
370 }
371 } else if (op == 'LOTS') {
372 final w = warehouse(id);
373 final s = item(a[2]);
374 var count = 0;
375 out.writeln('LOTS $id ${s.code}');
376 for (final l in s.lots)
377 if (l.held[w.index] > 0) {
378 final q = l.held[w.index];
379 count++;
380 out.writeln(' lot ${l.number} day ${l.day} qty $q cost ${money(l.cost)} value ${money(q * l.cost)} age ${day - l.day}');
381 }
382 for (final o in orders)
383 if (o.warehouse == w && o.item == s && o.backorder > 0) {
384 count++;
385 out.writeln(' backorder order ${o.number} qty ${o.backorder}');
386 }
387 if (count == 0) out.writeln(' (none)');
388 } else if (op == 'REORDER' || op == 'ITEMS') {
389 out.writeln(op);
390 var count = 0;
391 for (final s in itemList()) {
392 final h = stock(s).$1, b = backorder(s);
393 if (op == 'REORDER') {
394 if (h - b > s.level) continue;
395 out.writeln(' ${s.code} ${s.name} onhand $h backorder $b level ${s.level}');
396 } else {
397 out.writeln(' ${s.code} ${s.name} level ${s.level} onhand $h backorder $b cost ${s.lastCost == null ? 'none' : money(s.lastCost!)}');
398 }
399 count++;
400 }
401 if (count == 0) out.writeln(' (none)');
402 } else if (op == 'MARGIN' || op == 'TOP') {
403 final selected = itemList().where((s) => s.orderCount > 0).toList();
404 if (op == 'MARGIN') {
405 out.writeln('MARGIN');
406 var sold = 0, revenue = 0, cogs = 0;
407 for (final s in selected) {
408 out.writeln(' ${s.code} sold ${s.sold} revenue ${money(s.revenue)} cogs ${money(s.cogs)} profit ${money(s.revenue - s.cogs)} margin ${margin(s.revenue - s.cogs, s.revenue)}');
409 sold += s.sold;
410 revenue += s.revenue;
411 cogs += s.cogs;
412 }
413 out.writeln(' total sold $sold revenue ${money(revenue)} cogs ${money(cogs)} profit ${money(revenue - cogs)} margin ${margin(revenue - cogs, revenue)}');
414 } else {
415 final k = int.parse(id);
416 selected.sort((a, b) {
417 var c = (b.revenue - b.cogs).compareTo(a.revenue - a.cogs);
418 if (c == 0) c = b.revenue.compareTo(a.revenue);
419 return c != 0 ? c : a.code.compareTo(b.code);
420 });
421 out.writeln('TOP $k');
422 for (var i = 0; i < math.min(k, selected.length); i++) {
423 final s = selected[i];
424 out.writeln(' ${i + 1} ${s.code} profit ${money(s.revenue - s.cogs)} revenue ${money(s.revenue)} margin ${margin(s.revenue - s.cogs, s.revenue)}');
425 }
426 if (selected.isEmpty) out.writeln(' (none)');
427 }
428 } else if (op == 'BACKORDERS') {
429 out.writeln('BACKORDERS');
430 var count = 0;
431 for (final o in orders)
432 if (o.backorder > 0) {
433 count++;
434 out.writeln(' order ${o.number} ${o.warehouse.code} ${o.item.code} qty ${o.backorder} day ${o.day} price ${money(o.price)}');
435 }
436 if (count == 0) out.writeln(' (none)');
437 } else if (op == 'ORDER') {
438 final o = order(id);
439 out.writeln('ORDER ${o.number} ${o.warehouse.code} ${o.item.code} day ${o.day} price ${money(o.price)}');
440 out.writeln(' ordered ${o.ordered} shipped ${o.shipped} backorder ${o.backorder} cancelled ${o.cancelled} returned ${o.returned}');
441 for (final p in o.issues) out.writeln(' lot ${p.lot.number} qty ${p.quantity} cost ${money(p.cost)}');
442 out.writeln(' revenue ${money(o.revenue)} cogs ${money(o.cogs)} profit ${money(o.revenue - o.cogs)}');
443 } else if (op == 'TRACE') {
444 final l = lot(id);
445 out.writeln('TRACE lot ${l.number} ${l.item.code} ${l.kind} ${l.warehouse.code} day ${l.day} qty ${l.initialQuantity} cost ${money(l.initialCost)}');
446 for (final w in warehouseList()) if (l.held[w.index] > 0) out.writeln(' held ${w.code} qty ${l.held[w.index]} cost ${money(l.cost)}');
447 for (final o in orders) {
448 final q = o.issues.where((p) => p.lot == l).fold(0, (s, p) => s + p.quantity);
449 if (q > 0) out.writeln(' order ${o.number} qty $q');
450 }
451 if (l.removed > 0) out.writeln(' removed qty ${l.removed}');
452 } else if (op == 'CARD') {
453 final s = item(id);
454 out.writeln('CARD $id ${s.name}');
455 var q = 0, value = 0;
456 for (final e in s.card) {
457 q += e.quantity;
458 value += e.value;
459 out.writeln(' day ${e.day} ${e.kind} ${e.warehouse.code} ${signed(e.quantity)} ${signedMoney(e.value)} balance $q ${money(value)}');
460 }
461 if (s.card.isEmpty) out.writeln(' (none)');
462 } else if (op == 'STATUS') {
463 out.writeln('STATUS');
464 for (final w in warehouseList()) {
465 var skus = 0, q = 0, value = 0, b = 0;
466 for (final s in items.values) {
467 final h = stock(s, w);
468 if (h.$1 > 0) skus++;
469 q += h.$1;
470 value += h.$2;
471 }
472 for (final o in orders) if (o.warehouse == w) b += o.backorder;
473 out.writeln(' ${w.code} skus $skus qty $q value ${money(value)} backorder $b');
474 }
475 if (warehouses.isEmpty) out.writeln(' (none)');
476 } else if (op == 'SUMMARY') {
477 out.writeln('SUMMARY');
478 for (var i = 0; i < 8; i++) out.writeln(' ${categories[i]} qty ${quantities[i]} value ${money(values[i])}');
479 var q = 0, value = 0, open = 0, b = 0;
480 for (final s in items.values) {
481 final h = stock(s);
482 q += h.$1;
483 value += h.$2;
484 }
485 for (final o in orders) {
486 if (o.backorder > 0) open++;
487 b += o.backorder;
488 }
489 out.writeln(' on-hand qty $q value ${money(value)}');
490 out.writeln(' orders ${orders.length} open $open backorder $b');
491 }
492 }
493}
494void main() {
495 final inventory = Inventory(), n = int.parse(stdin.readLineSync()!);
496 for (var i = 0; i < n; i++) {
497 final draft = StringBuffer(), a = stdin.readLineSync()!.split(' ');
498 try {
499 inventory.execute(a, draft);
500 stdout.write(draft);
501 } on InventoryError catch (e) {
502 print('ERROR ${e.reason}');
503 }
504 }
505}
