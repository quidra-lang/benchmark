195issue <- function(inv, w, code, q) {
196 parts <- list()
197 cost <- 0
198 for (stock in fifo(inv, w, code)) {
199 if (q == 0) break
200 taken <- min(q, stock$quantity)
201 parts[[length(parts) + 1]] <- list(
202 lot = stock$lot, quantity = taken, cost = stock$cost
203 )
204 cost <- cost + taken * stock$cost
205 stock$quantity <- stock$quantity - taken
206 q <- q - taken
207 if (stock$quantity == 0) {
208 rm(list = as.character(stock$lot$number), envir = inv$stock[[w]][[code]])
209 }
210 }
211 list(parts = parts, cost = cost)
212}

214fulfil <- function(inv, order, q, kind) {
215 if (q == 0) return(0)
216 issued <- issue(inv, order$warehouse, order$item, q)
217 order$shipped <- order$shipped + q
218 order$backorder <- order$backorder - q
219 order$revenue <- order$revenue + q * order$price
220 order$cogs <- order$cogs + issued$cost
221 order$issues <- c(order$issues, issued$parts)
222 accumulate(inv, "shipped", q, issued$cost)
223 card_entry(inv, order$item, kind, order$warehouse, -q, -issued$cost)
224 issued$cost
225}

227allocate <- function(inv, w, code) {
228 available <- quantity(inv, w, code)
229 for (order in inv$orders) {
230 if (available == 0) break
231 if (order$warehouse != w || order$item != code || order$backorder == 0) next
232 q <- min(available, order$backorder)
233 cost <- fulfil(inv, order, q, "fill")
234 revenue <- q * order$price
235 available <- available - q
236 emit(inv, 2, "FILL", "order", digits(order$number), "qty", digits(q),
237 "revenue", money(revenue), "cogs", money(cost),
238 "profit", money(revenue - cost), "backorder", digits(order$backorder))
239 }
240}

278command <- function(inv, cmd, a) {
279 if (cmd == "DAY") {
280 day <- number(a[1])
281 if (day <= inv$day) fail(paste("invalid day", a[1]))
282 inv$day <- day
283 emit(inv, 0, "DAY", digits(day))
284
285 } else if (cmd == "WAREHOUSE") {
286 w <- a[1]
287 if (has(inv$stock, w)) fail(paste("duplicate warehouse", w))
288 inv$stock[[w]] <- new.env(parent = emptyenv())
289 emit(inv, 0, "WAREHOUSE", w)
290
291 } else if (cmd == "ITEM") {
292 code <- a[1]
293 level <- number(a[2])
294 name <- a[3]
295 action <- "added"
296 if (has(inv$items, code)) {
297 entry <- inv$items[[code]]
298 entry$level <- level
299 entry$name <- name
300 action <- "updated"
301 } else {
302 inv$items[[code]] <- object(
303 code = code, name = name, level = level, last_cost = NULL, card = list()
304 )
305 }
306 emit(inv, 0, "ITEM", code, action, name, "level", digits(level))
307
308 } else if (cmd == "RECEIVE") {
309 w <- a[1]
310 code <- a[2]
311 entry <- pair(inv, w, code)
312 q <- number(a[3])
313 cost <- cents(a[4])
314 day <- number(a[5])
315 require_quantity(q)
316 if (day > inv$day) fail(paste("future day", a[5]))
317 lot <- create_lot(inv, w, code, day, q, cost, "receive")
318 entry$last_cost <- cost
319 accumulate(inv, "received", q, q * cost)
320 card_entry(inv, code, "receive", w, q, q * cost)
321 emit(inv, 0, "RECEIVE", w, code, "lot", digits(lot$number),
322 "qty", digits(q), "cost", money(cost), "day", digits(day),
323 "value", money(q * cost))
324 allocate(inv, w, code)
325
326 } else if (cmd == "SHIP") {
327 w <- a[1]
328 code <- a[2]
329 pair(inv, w, code)
330 q <- number(a[3])
331 require_quantity(q)
332 order <- object(
333 number = length(inv$orders) + 1, warehouse = w, item = code,
334 day = inv$day, price = cents(a[4]), ordered = q, shipped = 0,
335 backorder = q, cancelled = 0, returned = 0, revenue = 0, cogs = 0,
336 issues = list()
337 )
338 inv$orders[[length(inv$orders) + 1]] <- order
339 shipped <- min(q, quantity(inv, w, code))
340 cost <- fulfil(inv, order, shipped, "ship")
341 emit(inv, 0, "SHIP", "order", digits(order$number), w, code,
342 "shipped", digits(shipped), "revenue", money(order$revenue),
343 "cogs", money(cost), "profit", money(order$revenue - cost),
344 "backorder", digits(order$backorder))
345
346 } else if (cmd == "CANCEL") {
347 order <- lookup_order(inv, a[1])
348 if (order$backorder == 0) fail(paste("no backorder", a[1]))
349 order$cancelled <- order$cancelled + order$backorder
350 order$backorder <- 0
351 emit(inv, 0, "CANCEL", "order", digits(order$number),
352 "qty", digits(order$cancelled))
353
354 } else if (cmd == "TRANSFER") {
355 source <- a[1]
356 destination <- a[2]
357 code <- a[3]
358 warehouse(inv, source)
359 warehouse(inv, destination)
360 if (source == destination) fail(paste("same warehouse", source))
361 item(inv, code)
362 q <- number(a[4])
363 require_quantity(q)
364 require_stock(inv, source, code, q)
365 issued <- issue(inv, source, code, q)
366 for (part in issued$parts) {
367 add_stock(inv, destination, part$lot, part$quantity, part$cost)
368 }
369 card_entry(inv, code, "transfer-out", source, -q, -issued$cost)
370 card_entry(inv, code, "transfer-in", destination, q, issued$cost)
371 emit(inv, 0, "TRANSFER", source, "->", destination, code,
372 "qty", digits(q), "value", money(issued$cost),
373 "lots", length(issued$parts))
374 allocate(inv, destination, code)
375
376 } else if (cmd == "ADJUST") {
377 w <- a[1]
378 code <- a[2]
379 change <- a[3]
380 pair(inv, w, code)
381 q <- number(substr(change, 2, nchar(change)))
382 require_quantity(q)
383 if (substr(change, 1, 1) == "+") {
384 cost <- cents(a[4])
385 lot <- create_lot(inv, w, code, inv$day, q, cost, "adjust")
386 accumulate(inv, "adjusted-in", q, q * cost)
387 card_entry(inv, code, "adjust", w, q, q * cost)
388 emit(inv, 0, "ADJUST", w, code, paste0("+", digits(q)),
389 "lot", digits(lot$number), "value", money(q * cost))
390 allocate(inv, w, code)
391 } else {
392 require_stock(inv, w, code, q)
393 issued <- issue(inv, w, code, q)
394 remove_parts(issued$parts)
395 accumulate(inv, "adjusted-out", q, issued$cost)
396 card_entry(inv, code, "adjust", w, -q, -issued$cost)
397 emit(inv, 0, "ADJUST", w, code, paste0("-", digits(q)),
398 "value", money(issued$cost))
399 }
400
401 } else if (cmd == "RETURN") {
402 order <- lookup_order(inv, a[1])
403 q <- number(a[2])
404 cost <- cents(a[3])
405 require_quantity(q)
406 available <- order$shipped - order$returned
407 if (q > available) {
408 fail(paste("return exceeds shipped", a[1], digits(available)))
409 }
410 lot <- create_lot(inv, order$warehouse, order$item, inv$day, q, cost, "return")
411 refund <- q * order$price
412 order$returned <- order$returned + q
413 order$revenue <- order$revenue - refund
414 order$cogs <- order$cogs - q * cost
415 accumulate(inv, "returned", q, q * cost)
416 card_entry(inv, order$item, "return", order$warehouse, q, q * cost)
417 emit(inv, 0, "RETURN", "order", digits(order$number),
418 order$warehouse, order$item, "qty", digits(q),
419 "lot", digits(lot$number), "refund", money(refund),
420 "value", money(q * cost))
421 allocate(inv, order$warehouse, order$item)
422
423 } else if (cmd == "COUNT") {
424 w <- a[1]
425 code <- a[2]
426 entry <- pair(inv, w, code)
427 actual <- number(a[3])
428 held <- quantity(inv, w, code)
429 if (actual == held) {
430 emit(inv, 0, "COUNT", w, code, "qty", digits(actual), "ok")
431 } else if (actual < held) {
432 q <- held - actual
433 issued <- issue(inv, w, code, q)
434 remove_parts(issued$parts)
435 accumulate(inv, "adjusted-out", q, issued$cost)
436 card_entry(inv, code, "count", w, -q, -issued$cost)
437 emit(inv, 0, "COUNT", w, code, "qty", digits(held), "->",
438 digits(actual), "shortage", digits(q), "value", money(issued$cost))
439 } else {
440 q <- actual - held
441 cost <- if (is.null(entry$last_cost)) 0 else entry$last_cost
442 lot <- create_lot(inv, w, code, inv$day, q, cost, "count")
443 accumulate(inv, "adjusted-in", q, q * cost)
444 card_entry(inv, code, "count", w, q, q * cost)
445 emit(inv, 0, "COUNT", w, code, "qty", digits(held), "->",
446 digits(actual), "surplus", digits(q), "lot", digits(lot$number),
447 "value", money(q * cost))
448 allocate(inv, w, code)
449 }
450
451 } else if (cmd == "MARKET") {
452 code <- a[1]
453 item(inv, code)
454 price <- cents(a[2])
455 changed <- 0
456 q <- 0
457 loss <- 0
458 for (w in names_sorted(inv$stock)) {
459 warehouse_loss <- 0
460 for (stock in holdings(inv, w, code)) {
461 if (stock$cost <= price) next
462 warehouse_loss <- warehouse_loss + stock$quantity * (stock$cost - price)
463 changed <- changed + 1
464 q <- q + stock$quantity
465 stock$cost <- price
466 }
467 if (warehouse_loss > 0) {
468 card_entry(inv, code, "writedown", w, 0, -warehouse_loss)
469 }
470 loss <- loss + warehouse_loss
471 }
472 accumulate(inv, "written-down", q, loss)
473 emit(inv, 0, "MARKET", code, "price", money(price), "lots",
474 digits(changed), "qty", digits(q), "loss", money(loss))
475
476 } else if (cmd == "RECALL") {
477 lot <- lookup_lot(inv, a[1])
478 held <- list()
479 for (w in names_sorted(inv$stock)) {
480 stock <- holdings(inv, w, lot$item)[[as.character(lot$number)]]
481 if (!is.null(stock)) {
482 held[[length(held) + 1]] <- list(warehouse = w, stock = stock)
483 }
484 }
485 if (length(held) == 0) fail(paste("nothing to recall", a[1]))
486 q <- 0
487 value <- 0
488 for (entry in held) {
489 stock <- entry$stock
490 q <- q + stock$quantity
491 value <- value + stock$quantity * stock$cost
492 rm(list = as.character(lot$number), envir = inv$stock[[entry$warehouse]][[lot$item]])
493 card_entry(inv, lot$item, "recall", entry$warehouse,
494 -stock$quantity, -stock$quantity * stock$cost)
495 }
496 lot$removed <- lot$removed + q
497 accumulate(inv, "recalled", q, value)
498 emit(inv, 0, "RECALL", "lot", digits(lot$number), lot$item,
499 "qty", digits(q), "value", money(value))
500 for (entry in held) {
501 emit(inv, 2, entry$warehouse, "qty", digits(entry$stock$quantity),
502 "value", money(entry$stock$quantity * entry$stock$cost))
503 }
504
505 } else if (cmd == "EXPIRE") {
506 days <- number(a[1])
507 emit(inv, 0, "EXPIRE", digits(days))
508 total_q <- 0
509 total_value <- 0
510 for (code in names_sorted(inv$items)) {
511 for (w in names_sorted(inv$stock)) {
512 expired <- Filter(
513 function(stock) inv$day - stock$lot$day > days,
514 holdings(inv, w, code)
515 )
516 if (length(expired) == 0) next
517 q <- stock_quantity(expired)
518 value <- stock_value(expired)
519 for (stock in expired) {
520 stock$lot$removed <- stock$lot$removed + stock$quantity
521 rm(list = as.character(stock$lot$number), envir = inv$stock[[w]][[code]])
522 }
523 accumulate(inv, "expired", q, value)
524 card_entry(inv, code, "expire", w, -q, -value)
525 total_q <- total_q + q
526 total_value <- total_value + value
527 emit(inv, 2, code, w, "lots", length(expired), "qty", digits(q),
528 "value", money(value))
529 }
530 }
531 emit(inv, 2, "total", "qty", digits(total_q), "value", money(total_value))
532
533 } else if (cmd == "VALUE" || cmd == "AGING") {
534 warehouses <- if (length(a) > 0) warehouse(inv, a[1]) else names_sorted(inv$stock)
535 if (cmd == "VALUE") {
536 emit(inv, 0, c("VALUE", a))
537 } else {
538 emit(inv, 0, c("AGING", a, "day", digits(inv$day)))
539 }
540 total_q <- 0
541 total_value <- 0
542 bucket_q <- numeric(4)
543 bucket_v <- numeric(4)
544 for (code in names_sorted(inv$items)) {
545 rows <- lapply(warehouses, function(w) holdings(inv, w, code))
546 selected <- unlist(rows, recursive = FALSE, use.names = FALSE)
547 q <- stock_quantity(selected)
548 if (q == 0) next
549 value <- stock_value(selected)
550 if (cmd == "VALUE") {
551 emit(inv, 2, code, "qty", digits(q), "value", money(value),
552 "avg", money(floor(value / q)))
553 if (length(a) == 0) {
554 for (i in seq_along(warehouses)) {
555 held <- stock_quantity(rows[[i]])
556 if (held > 0) {
557 emit(inv, 4, warehouses[i], "qty", digits(held),
558 "value", money(stock_value(rows[[i]])))
559 }
560 }
561 }
562 } else {
563 item_buckets <- numeric(4)
564 for (stock in selected) {
565 age <- inv$day - stock$lot$day
566 bucket <- if (age <= 30) 1 else if (age <= 60) 2 else if (age <= 90) 3 else 4
567 item_buckets[bucket] <- item_buckets[bucket] + stock$quantity
568 bucket_q[bucket] <- bucket_q[bucket] + stock$quantity
569 bucket_v[bucket] <- bucket_v[bucket] + stock$quantity * stock$cost
570 }
571 aging_line(inv, code, item_buckets)
572 }
573 total_q <- total_q + q
574 total_value <- total_value + value
575 }
576 if (cmd == "VALUE") {
577 emit(inv, 2, "total", "qty", digits(total_q), "value", money(total_value))
578 } else {
579 aging_line(inv, "total", bucket_q)
580 aging_line(inv, "value", bucket_v, TRUE)
581 }
582
583 } else if (cmd == "LOTS") {
584 w <- a[1]
585 code <- a[2]
586 pair(inv, w, code)
587 emit(inv, 0, "LOTS", w, code)
588 stocks <- fifo(inv, w, code)
589 pending <- Filter(
590 function(o) o$warehouse == w && o$item == code && o$backorder > 0,
591 inv$orders
592 )
593 for (stock in stocks) {
594 emit(inv, 2, "lot", digits(stock$lot$number), "day", digits(stock$lot$day),
595 "qty", digits(stock$quantity), "cost", money(stock$cost),
596 "value", money(stock$quantity * stock$cost),
597 "age", digits(inv$day - stock$lot$day))
598 }
599 for (order in pending) {
600 emit(inv, 2, "backorder", "order", digits(order$number),
601 "qty", digits(order$backorder))
602 }
603 if (length(stocks) == 0 && length(pending) == 0) emit(inv, 2, "(none)")
604
605 } else if (cmd == "REORDER" || cmd == "ITEMS") {
606 emit(inv, 0, cmd)
607 found <- FALSE
608 for (code in names_sorted(inv$items)) {
609 entry <- inv$items[[code]]
610 q <- total_quantity(inv, code)
611 pending <- backorder(inv, code)
612 if (cmd == "REORDER") {
613 if (q - pending > entry$level) next
614 emit(inv, 2, code, entry$name, "onhand", digits(q),
615 "backorder", digits(pending), "level", digits(entry$level))
616 } else {
617 emit(inv, 2, code, entry$name, "level", digits(entry$level),
618 "onhand", digits(q), "backorder", digits(pending), "cost",
619 if (is.null(entry$last_cost)) "none" else money(entry$last_cost))
620 }
621 found <- TRUE
622 }
623 if (!found) emit(inv, 2, "(none)")
624
625 } else if (cmd == "MARGIN" || cmd == "TOP") {
626 rows <- sales(inv)
627 codes <- names_sorted(rows)
628 if (cmd == "MARGIN") {
629 emit(inv, 0, "MARGIN")
630 total_q <- 0
631 revenue <- 0
632 cogs <- 0
633 for (code in codes) {
634 entry <- rows[[code]]
635 margin_line(inv, code, entry)
636 total_q <- total_q + entry$quantity
637 revenue <- revenue + entry$revenue
638 cogs <- cogs + entry$cogs
639 }
640 margin_line(inv, "total", object(quantity = total_q, revenue = revenue, cogs = cogs))
641 } else {
642 k <- number(a[1])
643 emit(inv, 0, "TOP", digits(k))
644 if (length(codes) > 0) {
645 revenue <- vapply(codes, function(c) rows[[c]]$revenue, numeric(1))
646 profit <- vapply(codes, function(c) rows[[c]]$revenue - rows[[c]]$cogs, numeric(1))
647 codes <- codes[order(-profit, -revenue, codes)]
648 if (k > 0) {
649 for (i in seq_len(min(k, length(codes)))) {
650 entry <- rows[[codes[i]]]
651 gain <- entry$revenue - entry$cogs
652 emit(inv, 2, digits(i), codes[i], "profit", money(gain),
653 "revenue", money(entry$revenue), "margin", margin(gain, entry$revenue))
654 }
655 }
656 } else {
657 emit(inv, 2, "(none)")
658 }
659 }
660
661 } else if (cmd == "BACKORDERS") {
662 emit(inv, 0, "BACKORDERS")
663 pending <- Filter(function(o) o$backorder > 0, inv$orders)
664 for (order in pending) {
665 emit(inv, 2, "order", digits(order$number), order$warehouse, order$item,
666 "qty", digits(order$backorder), "day", digits(order$day),
667 "price", money(order$price))
668 }
669 if (length(pending) == 0) emit(inv, 2, "(none)")
670
671 } else if (cmd == "ORDER") {
672 order <- lookup_order(inv, a[1])
673 emit(inv, 0, "ORDER", digits(order$number), order$warehouse, order$item,
674 "day", digits(order$day), "price", money(order$price))
675 emit(inv, 2, "ordered", digits(order$ordered), "shipped", digits(order$shipped),
676 "backorder", digits(order$backorder), "cancelled", digits(order$cancelled),
677 "returned", digits(order$returned))
678 for (part in order$issues) {
679 emit(inv, 2, "lot", digits(part$lot$number), "qty", digits(part$quantity),
680 "cost", money(part$cost))
681 }
682 emit(inv, 2, "revenue", money(order$revenue), "cogs", money(order$cogs),
683 "profit", money(order$revenue - order$cogs))
684
685 } else if (cmd == "TRACE") {
686 lot <- lookup_lot(inv, a[1])
687 emit(inv, 0, "TRACE", "lot", digits(lot$number), lot$item, lot$kind,
688 lot$warehouse, "day", digits(lot$day), "qty", digits(lot$initial_quantity),
689 "cost", money(lot$initial_cost))
690 for (w in names_sorted(inv$stock)) {
691 stock <- holdings(inv, w, lot$item)[[as.character(lot$number)]]
692 if (!is.null(stock)) {
693 emit(inv, 2, "held", w, "qty", digits(stock$quantity), "cost", money(stock$cost))
694 }
695 }
696 for (order in inv$orders) {
697 q <- 0
698 for (part in order$issues) {
699 if (part$lot$number == lot$number) q <- q + part$quantity
700 }
701 if (q > 0) emit(inv, 2, "order", digits(order$number), "qty", digits(q))
702 }
703 if (lot$removed > 0) emit(inv, 2, "removed", "qty", digits(lot$removed))
704
705 } else if (cmd == "CARD") {
706 entry <- item(inv, a[1])
707 emit(inv, 0, "CARD", entry$code, entry$name)
708 q <- 0
709 value <- 0
710 for (card in entry$card) {
711 q <- q + card$quantity
712 value <- value + card$value
713 emit(inv, 2, "day", digits(card$day), card$kind, card$warehouse,
714 signed_quantity(card$quantity), signed_money(card$value),
715 "balance", digits(q), money(value))
716 }
717 if (length(entry$card) == 0) emit(inv, 2, "(none)")
718
719 } else if (cmd == "STATUS") {
720 emit(inv, 0, "STATUS")
721 for (w in names_sorted(inv$stock)) {
722 skus <- 0
723 stocks <- list()
724 for (code in names_sorted(inv$stock[[w]])) {
725 current <- holdings(inv, w, code)
726 if (length(current) > 0) skus <- skus + 1
727 stocks <- c(stocks, current)
728 }
729 pending <- 0
730 for (order in inv$orders) {
731 if (order$warehouse == w) pending <- pending + order$backorder
732 }
733 emit(inv, 2, w, "skus", digits(skus), "qty", digits(stock_quantity(stocks)),
734 "value", money(stock_value(stocks)), "backorder", digits(pending))
735 }
736 if (length(names_sorted(inv$stock)) == 0) emit(inv, 2, "(none)")
737
738 } else if (cmd == "SUMMARY") {
739 emit(inv, 0, "SUMMARY")
740 for (category in categories) {
741 total <- inv$totals[[category]]
742 emit(inv, 2, category, "qty", digits(total$quantity),
743 "value", money(total$value))
744 }
745 stocks <- all_stocks(inv)
746 emit(inv, 2, "on-hand", "qty", digits(stock_quantity(stocks)),
747 "value", money(stock_value(stocks)))
748 open <- 0
749 pending <- 0
750 for (order in inv$orders) {
751 if (order$backorder > 0) open <- open + 1
752 pending <- pending + order$backorder
753 }
754 emit(inv, 2, "orders", digits(length(inv$orders)),
755 "open", digits(open), "backorder", digits(pending))
756 }
757}

759execute <- function(inv, line) {
760 tokens <- strsplit(line, " ", fixed = TRUE)[[1]]
761 cmd <- tokens[1]
762 args <- tokens[-1]
763 tryCatch(
764 command(inv, cmd, args),
765 command_error = function(err) emit(inv, 0, "ERROR", err$message)
766 )
767}
768
769input <- readLines(stdin(), warn = FALSE)
770n <- as.integer(input[1])
771inv <- inventory()
772for (i in seq_len(n)) {
773 execute(inv, input[i + 1])
774}
775cat(paste(inv$output, collapse = "\n"), "\n", sep = "")
