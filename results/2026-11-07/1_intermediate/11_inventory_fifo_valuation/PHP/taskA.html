4const CATEGORIES = [
5 'received', 'returned', 'adjusted-in', 'adjusted-out',

9function cents(string $text): int {
10 return (int)str_replace('.', '', $text);
11}
12
13function money(int $cents): string {
14 $sign = $cents < 0 ? '-' : '';
15 $value = abs($cents);
16 return $sign . intdiv($value, 100) . '.' . str_pad((string)($value % 100), 2, '0', STR_PAD_LEFT);
17}
18
19function signedQuantity(int $quantity): string {
20 return ($quantity > 0 ? '+' : '') . $quantity;
21}
22
23function signedMoney(int $amount): string {
24 return ($amount > 0 ? '+' : '') . money($amount);
25}
26
27function marginText(int $profit, int $revenue): string {
28 if ($revenue === 0) {
29 return 'n/a';
30 }
31 $tenths = intdiv(abs($profit) * 1000, $revenue);
32 return ($profit < 0 && $tenths !== 0 ? '-' : '')
33 . intdiv($tenths, 10) . '.' . ($tenths % 10) . '%';
34}

36function stockQuantity(array $stocks): int {
37 $total = 0;
38 foreach ($stocks as $stock) {
39 $total += $stock->quantity;

41 return $total;

44function stockValue(array $stocks): int {
45 $total = 0;
46 foreach ($stocks as $stock) {
47 $total += $stock->quantity * $stock->cost;
48 }
49 return $total;

118 private function fifo(string $warehouse, string $code): array {
119 $stocks = array_values($this->holdings($warehouse, $code));
120 usort($stocks, static fn($a, $b) =>
121 ($a->lot->day <=> $b->lot->day) ?: ($a->lot->number <=> $b->lot->number));
122 return $stocks;
123 }

176 private function addStock(string $warehouse, object $lot, int $quantity, int $cost): void {
177 $this->stock[$warehouse][$lot->item] ??= [];
178 if (isset($this->stock[$warehouse][$lot->item][$lot->number])) {
179 $this->stock[$warehouse][$lot->item][$lot->number]->quantity += $quantity;

186
187 private function createLot(
188 string $warehouse, string $code, int $day, int $quantity, int $cost, string $kind
189 ): object {
190 $lot = (object)[
191 'number' => count($this->lots) + 1, 'item' => $code,
192 'day' => $day, 'initialQuantity' => $quantity, 'initialCost' => $cost,
193 'warehouse' => $warehouse, 'kind' => $kind, 'removed' => 0,
194 ];
195 $this->lots[] = $lot;
196 $this->addStock($warehouse, $lot, $quantity, $cost);
197 return $lot;
198 }
199
200 private function issue(string $warehouse, string $code, int $quantity): array {
201 $parts = [];
202 $cost = 0;
203 foreach ($this->fifo($warehouse, $code) as $stock) {
204 if ($quantity === 0) {
205 break;
206 }
207 $taken = min($quantity, $stock->quantity);
208 $parts[] = (object)['lot' => $stock->lot, 'quantity' => $taken, 'cost' => $stock->cost];
209 $cost += $taken * $stock->cost;
210 $stock->quantity -= $taken;
211 $quantity -= $taken;
212 if ($stock->quantity === 0) {
213 unset($this->stock[$warehouse][$code][$stock->lot->number]);
214 }
215 }
216 return [$parts, $cost];
217 }
218
219 private function fulfil(object $order, int $quantity, string $kind): int {
220 if ($quantity === 0) {
221 return 0;
222 }
223 [$parts, $cost] = $this->issue($order->warehouse, $order->item, $quantity);
224 $order->shipped += $quantity;
225 $order->backorder -= $quantity;
226 $order->revenue += $quantity * $order->price;
227 $order->cogs += $cost;
228 array_push($order->issues, ...$parts);
229 $this->accumulate('shipped', $quantity, $cost);
230 $this->cardEntry($order->item, $kind, $order->warehouse, -$quantity, -$cost);
231 return $cost;
232 }
233
234 private function allocate(string $warehouse, string $code): void {
235 $available = $this->quantity($warehouse, $code);
236 foreach ($this->orders as $order) {
237 if ($available === 0) {
238 break;
239 }
240 if ($order->warehouse !== $warehouse || $order->item !== $code || $order->backorder === 0) {
241 continue;
242 }
243 $quantity = min($available, $order->backorder);
244 $cost = $this->fulfil($order, $quantity, 'fill');
245 $revenue = $quantity * $order->price;
246 $available -= $quantity;
247 $this->emit(2, 'FILL', 'order', $order->number, 'qty', $quantity,
248 'revenue', money($revenue), 'cogs', money($cost), 'profit',
249 money($revenue - $cost), 'backorder', $order->backorder);
250 }
251 }
252
253 private function removeParts(array $parts): void {
254 foreach ($parts as $part) {
255 $part->lot->removed += $part->quantity;
256 }
257 }
258
259 private function sales(): array {
260 $rows = [];
261 foreach ($this->orders as $order) {
262 if (!isset($rows[$order->item])) {
263 $rows[$order->item] = (object)['quantity' => 0, 'revenue' => 0, 'cogs' => 0];
264 }
265 $row = $rows[$order->item];
266 $row->quantity += $order->shipped - $order->returned;
267 $row->revenue += $order->revenue;
268 $row->cogs += $order->cogs;
269 }
270 return $rows;
271 }

273 private function marginLine(string $label, object $row): void {

275 $this->emit(2, $label, 'sold', $row->quantity, 'revenue', money($row->revenue),
276 'cogs', money($row->cogs), 'profit', money($profit),
277 'margin', marginText($profit, $row->revenue));
278 }

280 private function agingLine(string $label, array $buckets, bool $monetary = false): void {
281 $names = ['0-30', '31-60', '61-90', '91+'];
282 $parts = [$label];
283 for ($i = 0; $i < 4; ++$i) {
284 $parts[] = $names[$i];
285 $parts[] = $monetary ? money($buckets[$i]) : $buckets[$i];
286 }
287 $this->emit(2, ...$parts);
288 }

301
302 private function command(string $command, array $a): void {
303 switch ($command) {
304 case 'DAY':
305 $day = (int)$a[0];
306 if ($day <= $this->day) {
307 throw new CommandError("invalid day $a[0]");
308 }
309 $this->day = $day;
310 $this->emit(0, 'DAY', $day);
311 return;
312
313 case 'WAREHOUSE':
314 $warehouse = $a[0];
315 if (array_key_exists($warehouse, $this->stock)) {
316 throw new CommandError("duplicate warehouse $warehouse");
317 }
318 $this->stock[$warehouse] = [];
319 $this->emit(0, 'WAREHOUSE', $warehouse);
320 return;
321
322 case 'ITEM':
323 [$code, $level, $name] = $a;
324 if (isset($this->items[$code])) {
325 $item = $this->items[$code];
326 $item->level = (int)$level;
327 $item->name = $name;
328 $action = 'updated';
329 } else {
330 $this->items[$code] = (object)[
331 'code' => $code, 'level' => (int)$level, 'name' => $name,
332 'lastCost' => null, 'card' => [],
333 ];
334 $action = 'added';
335 }
336 $this->emit(0, 'ITEM', $code, $action, $name, 'level', (int)$level);
337 return;
338
339 case 'RECEIVE':
340 [$warehouse, $code] = $a;
341 $item = $this->pair($warehouse, $code);
342 $quantity = (int)$a[2];
343 $cost = cents($a[3]);
344 $day = (int)$a[4];
345 $this->requireQuantity($quantity);
346 if ($day > $this->day) {
347 throw new CommandError("future day $a[4]");
348 }
349 $lot = $this->createLot($warehouse, $code, $day, $quantity, $cost, 'receive');
350 $item->lastCost = $cost;
351 $amount = $quantity * $cost;
352 $this->accumulate('received', $quantity, $amount);
353 $this->cardEntry($code, 'receive', $warehouse, $quantity, $amount);
354 $this->emit(0, 'RECEIVE', $warehouse, $code, 'lot', $lot->number, 'qty',
355 $quantity, 'cost', money($cost), 'day', $day, 'value', money($amount));
356 $this->allocate($warehouse, $code);
357 return;
358
359 case 'SHIP':
360 [$warehouse, $code] = $a;
361 $this->pair($warehouse, $code);
362 $quantity = (int)$a[2];
363 $this->requireQuantity($quantity);
364 $order = (object)[
365 'number' => count($this->orders) + 1, 'warehouse' => $warehouse,
366 'item' => $code, 'day' => $this->day, 'price' => cents($a[3]),
367 'ordered' => $quantity, 'shipped' => 0, 'backorder' => $quantity,
368 'cancelled' => 0, 'returned' => 0, 'revenue' => 0, 'cogs' => 0,
369 'issues' => [],
370 ];
371 $this->orders[] = $order;
372 $filled = min($quantity, $this->quantity($warehouse, $code));
373 $cost = $this->fulfil($order, $filled, 'ship');
374 $this->emit(0, 'SHIP', 'order', $order->number, $warehouse, $code, 'shipped',
375 $filled, 'revenue', money($order->revenue), 'cogs', money($cost),
376 'profit', money($order->revenue - $cost), 'backorder', $order->backorder);
377 return;
378
379 case 'CANCEL':
380 $order = $this->order($a[0]);
381 if ($order->backorder === 0) {
382 throw new CommandError("no backorder $a[0]");
383 }
384 $order->cancelled += $order->backorder;
385 $order->backorder = 0;
386 $this->emit(0, 'CANCEL', 'order', $order->number, 'qty', $order->cancelled);
387 return;
388
389 case 'TRANSFER':
390 [$source, $destination, $code] = $a;
391 $this->warehouse($source);
392 $this->warehouse($destination);
393 if ($source === $destination) {
394 throw new CommandError("same warehouse $source");
395 }
396 $this->item($code);
397 $quantity = (int)$a[3];
398 $this->requireQuantity($quantity);
399 $this->requireStock($source, $code, $quantity);
400 [$parts, $cost] = $this->issue($source, $code, $quantity);
401 foreach ($parts as $part) {
402 $this->addStock($destination, $part->lot, $part->quantity, $part->cost);
403 }
404 $this->cardEntry($code, 'transfer-out', $source, -$quantity, -$cost);
405 $this->cardEntry($code, 'transfer-in', $destination, $quantity, $cost);
406 $this->emit(0, 'TRANSFER', $source, '->', $destination, $code, 'qty',
407 $quantity, 'value', money($cost), 'lots', count($parts));
408 $this->allocate($destination, $code);
409 return;
410
411 case 'ADJUST':
412 [$warehouse, $code, $change] = $a;
413 $this->pair($warehouse, $code);
414 $quantity = (int)substr($change, 1);
415 $this->requireQuantity($quantity);
416 if ($change[0] === '+') {
417 $cost = cents($a[3]);
418 $amount = $quantity * $cost;
419 $lot = $this->createLot($warehouse, $code, $this->day, $quantity, $cost, 'adjust');
420 $this->accumulate('adjusted-in', $quantity, $amount);
421 $this->cardEntry($code, 'adjust', $warehouse, $quantity, $amount);
422 $this->emit(0, 'ADJUST', $warehouse, $code, "+$quantity",
423 'lot', $lot->number, 'value', money($amount));
424 $this->allocate($warehouse, $code);
425 } else {
426 $this->requireStock($warehouse, $code, $quantity);
427 [$parts, $cost] = $this->issue($warehouse, $code, $quantity);
428 $this->removeParts($parts);
429 $this->accumulate('adjusted-out', $quantity, $cost);
430 $this->cardEntry($code, 'adjust', $warehouse, -$quantity, -$cost);
431 $this->emit(0, 'ADJUST', $warehouse, $code, "-$quantity", 'value', money($cost));
432 }
433 return;
434
435 case 'RETURN':
436 $order = $this->order($a[0]);
437 $quantity = (int)$a[1];
438 $cost = cents($a[2]);
439 $this->requireQuantity($quantity);
440 $available = $order->shipped - $order->returned;
441 if ($quantity > $available) {
442 throw new CommandError("return exceeds shipped $a[0] $available");
443 }
444 $lot = $this->createLot($order->warehouse, $order->item,
445 $this->day, $quantity, $cost, 'return');
446 $amount = $quantity * $cost;
447 $refund = $quantity * $order->price;
448 $order->returned += $quantity;
449 $order->revenue -= $refund;
450 $order->cogs -= $amount;
451 $this->accumulate('returned', $quantity, $amount);
452 $this->cardEntry($order->item, 'return', $order->warehouse, $quantity, $amount);
453 $this->emit(0, 'RETURN', 'order', $order->number, $order->warehouse, $order->item,
454 'qty', $quantity, 'lot', $lot->number, 'refund', money($refund), 'value', money($amount));
455 $this->allocate($order->warehouse, $order->item);
456 return;
457
458 case 'COUNT':
459 [$warehouse, $code] = $a;
460 $item = $this->pair($warehouse, $code);
461 $actual = (int)$a[2];
462 $held = $this->quantity($warehouse, $code);
463 if ($actual === $held) {
464 $this->emit(0, 'COUNT', $warehouse, $code, 'qty', $actual, 'ok');
465 } elseif ($actual < $held) {
466 $quantity = $held - $actual;
467 [$parts, $cost] = $this->issue($warehouse, $code, $quantity);
468 $this->removeParts($parts);
469 $this->accumulate('adjusted-out', $quantity, $cost);
470 $this->cardEntry($code, 'count', $warehouse, -$quantity, -$cost);
471 $this->emit(0, 'COUNT', $warehouse, $code, 'qty', $held, '->', $actual,
472 'shortage', $quantity, 'value', money($cost));
473 } else {
474 $quantity = $actual - $held;
475 $cost = $item->lastCost ?? 0;
476 $amount = $quantity * $cost;
477 $lot = $this->createLot($warehouse, $code, $this->day, $quantity, $cost, 'count');
478 $this->accumulate('adjusted-in', $quantity, $amount);
479 $this->cardEntry($code, 'count', $warehouse, $quantity, $amount);
480 $this->emit(0, 'COUNT', $warehouse, $code, 'qty', $held, '->', $actual,
481 'surplus', $quantity, 'lot', $lot->number, 'value', money($amount));
482 $this->allocate($warehouse, $code);
483 }
484 return;
485
486 case 'MARKET':
487 $this->item($a[0]);
488 $price = cents($a[1]);
489 $changed = 0;
490 $quantity = 0;
491 $loss = 0;
492 foreach ($this->sortedWarehouses() as $warehouse) {
493 $warehouseLoss = 0;
494 foreach ($this->holdings($warehouse, $a[0]) as $stock) {
495 if ($stock->cost <= $price) {
496 continue;
497 }
498 $warehouseLoss += $stock->quantity * ($stock->cost - $price);
499 ++$changed;
500 $quantity += $stock->quantity;
501 $stock->cost = $price;
502 }
503 if ($warehouseLoss > 0) {
504 $this->cardEntry($a[0], 'writedown', $warehouse, 0, -$warehouseLoss);
505 }
506 $loss += $warehouseLoss;
507 }
508 $this->accumulate('written-down', $quantity, $loss);
509 $this->emit(0, 'MARKET', $a[0], 'price', money($price),
510 'lots', $changed, 'qty', $quantity, 'loss', money($loss));
511 return;
512
513 case 'RECALL':
514 $lot = $this->lot($a[0]);
515 $held = [];
516 foreach ($this->sortedWarehouses() as $warehouse) {
517 $stock = $this->stock[$warehouse][$lot->item][$lot->number] ?? null;
518 if ($stock !== null) {
519 $held[] = (object)['warehouse' => $warehouse, 'stock' => $stock];
520 }
521 }
522 if (!$held) {
523 throw new CommandError("nothing to recall $a[0]");
524 }
525 $quantity = 0;
526 $amount = 0;
527 foreach ($held as $entry) {
528 $quantity += $entry->stock->quantity;
529 $amount += $entry->stock->quantity * $entry->stock->cost;
530 unset($this->stock[$entry->warehouse][$lot->item][$lot->number]);
531 $this->cardEntry($lot->item, 'recall', $entry->warehouse,
532 -$entry->stock->quantity, -$entry->stock->quantity * $entry->stock->cost);
533 }
534 $lot->removed += $quantity;
535 $this->accumulate('recalled', $quantity, $amount);
536 $this->emit(0, 'RECALL', 'lot', $lot->number, $lot->item, 'qty', $quantity, 'value', money($amount));
537 foreach ($held as $entry) {
538 $this->emit(2, $entry->warehouse, 'qty', $entry->stock->quantity,
539 'value', money($entry->stock->quantity * $entry->stock->cost));
540 }
541 return;
542
543 case 'EXPIRE':
544 $days = (int)$a[0];
545 $this->emit(0, 'EXPIRE', $days);
546 $totalQuantity = 0;
547 $totalValue = 0;
548 foreach ($this->sortedItems() as $code) {
549 foreach ($this->sortedWarehouses() as $warehouse) {
550 $expired = array_filter($this->holdings($warehouse, $code),
551 fn($stock) => $this->day - $stock->lot->day > $days);
552 if (!$expired) {
553 continue;
554 }
555 $quantity = stockQuantity($expired);
556 $value = stockValue($expired);
557 foreach ($expired as $stock) {
558 $stock->lot->removed += $stock->quantity;
559 unset($this->stock[$warehouse][$code][$stock->lot->number]);
560 }
561 $this->accumulate('expired', $quantity, $value);
562 $this->cardEntry($code, 'expire', $warehouse, -$quantity, -$value);
563 $totalQuantity += $quantity;
564 $totalValue += $value;
565 $this->emit(2, $code, $warehouse, 'lots', count($expired),
566 'qty', $quantity, 'value', money($value));
567 }
568 }
569 $this->emit(2, 'total', 'qty', $totalQuantity, 'value', money($totalValue));
570 return;
571
572 case 'VALUE':
573 case 'AGING':
574 $warehouses = $a ? [$this->warehouse($a[0])] : $this->sortedWarehouses();
575 if ($command === 'VALUE') {
576 $this->emit(0, 'VALUE', ...$a);
577 } else {
578 $this->emit(0, 'AGING', ...array_merge($a, ['day', $this->day]));
579 }
580 $totalQuantity = 0;
581 $totalValue = 0;
582 $bucketQuantity = [0, 0, 0, 0];
583 $bucketValue = [0, 0, 0, 0];
584 foreach ($this->sortedItems() as $code) {
585 $stocks = [];
586 $rows = [];
587 foreach ($warehouses as $warehouse) {
588 $held = array_values($this->holdings($warehouse, $code));
589 $rows[$warehouse] = $held;
590 array_push($stocks, ...$held);
591 }
592 $quantity = stockQuantity($stocks);
593 if ($quantity === 0) {
594 continue;
595 }
596 $value = stockValue($stocks);
597 if ($command === 'VALUE') {
598 $this->emit(2, $code, 'qty', $quantity, 'value', money($value),
599 'avg', money(intdiv($value, $quantity)));
600 if (!$a) {
601 foreach ($rows as $warehouse => $held) {
602 $warehouseQuantity = stockQuantity($held);
603 if ($warehouseQuantity !== 0) {
604 $this->emit(4, $warehouse, 'qty', $warehouseQuantity,
605 'value', money(stockValue($held)));
606 }
607 }
608 }
609 } else {
610 $itemBuckets = [0, 0, 0, 0];
611 foreach ($stocks as $stock) {
612 $age = $this->day - $stock->lot->day;
613 $bucket = $age <= 30 ? 0 : ($age <= 60 ? 1 : ($age <= 90 ? 2 : 3));
614 $itemBuckets[$bucket] += $stock->quantity;
615 $bucketQuantity[$bucket] += $stock->quantity;
616 $bucketValue[$bucket] += $stock->quantity * $stock->cost;
617 }
618 $this->agingLine($code, $itemBuckets);
619 }
620 $totalQuantity += $quantity;
621 $totalValue += $value;
622 }
623 if ($command === 'VALUE') {
624 $this->emit(2, 'total', 'qty', $totalQuantity, 'value', money($totalValue));
625 } else {
626 $this->agingLine('total', $bucketQuantity);
627 $this->agingLine('value', $bucketValue, true);
628 }
629 return;
630
631 case 'LOTS':
632 [$warehouse, $code] = $a;
633 $this->pair($warehouse, $code);
634 $this->emit(0, 'LOTS', $warehouse, $code);
635 $stocks = $this->fifo($warehouse, $code);
636 $backorders = array_filter($this->orders,
637 fn($order) => $order->warehouse === $warehouse && $order->item === $code && $order->backorder > 0);
638 foreach ($stocks as $stock) {
639 $this->emit(2, 'lot', $stock->lot->number, 'day', $stock->lot->day,
640 'qty', $stock->quantity, 'cost', money($stock->cost),
641 'value', money($stock->quantity * $stock->cost),
642 'age', $this->day - $stock->lot->day);
643 }
644 foreach ($backorders as $order) {
645 $this->emit(2, 'backorder', 'order', $order->number, 'qty', $order->backorder);
646 }
647 if (!$stocks && !$backorders) {
648 $this->emit(2, '(none)');
649 }
650 return;
651
652 case 'REORDER':
653 case 'ITEMS':
654 $this->emit(0, $command);
655 $found = false;
656 foreach ($this->sortedItems() as $code) {
657 $item = $this->items[$code];
658 $quantity = $this->totalQuantity($code);
659 $backorder = $this->backorder($code);
660 if ($command === 'REORDER') {
661 if ($quantity - $backorder > $item->level) {
662 continue;
663 }
664 $this->emit(2, $code, $item->name, 'onhand', $quantity,
665 'backorder', $backorder, 'level', $item->level);
666 } else {
667 $this->emit(2, $code, $item->name, 'level', $item->level,
668 'onhand', $quantity, 'backorder', $backorder, 'cost',
669 $item->lastCost === null ? 'none' : money($item->lastCost));
670 }
671 $found = true;
672 }
673 if (!$found) {
674 $this->emit(2, '(none)');
675 }
676 return;
677
678 case 'MARGIN':
679 case 'TOP':
680 $rows = $this->sales();
681 if ($command === 'MARGIN') {
682 $this->emit(0, 'MARGIN');
683 $total = (object)['quantity' => 0, 'revenue' => 0, 'cogs' => 0];
684 $codes = array_keys($rows);
685 sort($codes, SORT_STRING);
686 foreach ($codes as $code) {
687 $row = $rows[$code];
688 $this->marginLine($code, $row);
689 $total->quantity += $row->quantity;
690 $total->revenue += $row->revenue;
691 $total->cogs += $row->cogs;
692 }
693 $this->marginLine('total', $total);
694 } else {
695 $this->emit(0, 'TOP', (int)$a[0]);
696 $codes = array_keys($rows);
697 usort($codes, static function ($a, $b) use ($rows) {
698 $left = $rows[$a];
699 $right = $rows[$b];
700 return (($right->revenue - $right->cogs) <=> ($left->revenue - $left->cogs))
701 ?: ($right->revenue <=> $left->revenue) ?: strcmp($a, $b);
702 });
703 $limit = min(count($codes), (int)$a[0]);
704 for ($i = 0; $i < $limit; ++$i) {
705 $row = $rows[$codes[$i]];
706 $profit = $row->revenue - $row->cogs;
707 $this->emit(2, $i + 1, $codes[$i], 'profit', money($profit),
708 'revenue', money($row->revenue), 'margin', marginText($profit, $row->revenue));
709 }
710 if (!$codes) {
711 $this->emit(2, '(none)');
712 }
713 }
714 return;
715
716 case 'BACKORDERS':
717 $this->emit(0, 'BACKORDERS');
718 $backorders = array_filter($this->orders, static fn($order) => $order->backorder > 0);
719 foreach ($backorders as $order) {
720 $this->emit(2, 'order', $order->number, $order->warehouse, $order->item,
721 'qty', $order->backorder, 'day', $order->day, 'price', money($order->price));
722 }
723 if (!$backorders) {
724 $this->emit(2, '(none)');
725 }
726 return;
727
728 case 'ORDER':
729 $order = $this->order($a[0]);
730 $this->emit(0, 'ORDER', $order->number, $order->warehouse, $order->item,
731 'day', $order->day, 'price', money($order->price));
732 $this->emit(2, 'ordered', $order->ordered, 'shipped', $order->shipped,
733 'backorder', $order->backorder, 'cancelled', $order->cancelled, 'returned', $order->returned);
734 foreach ($order->issues as $part) {
735 $this->emit(2, 'lot', $part->lot->number, 'qty', $part->quantity, 'cost', money($part->cost));
736 }
737 $this->emit(2, 'revenue', money($order->revenue), 'cogs', money($order->cogs),
738 'profit', money($order->revenue - $order->cogs));
739 return;
740
741 case 'TRACE':
742 $lot = $this->lot($a[0]);
743 $this->emit(0, 'TRACE', 'lot', $lot->number, $lot->item, $lot->kind,
744 $lot->warehouse, 'day', $lot->day, 'qty', $lot->initialQuantity,
745 'cost', money($lot->initialCost));
746 foreach ($this->sortedWarehouses() as $warehouse) {
747 $stock = $this->stock[$warehouse][$lot->item][$lot->number] ?? null;
748 if ($stock !== null) {
749 $this->emit(2, 'held', $warehouse, 'qty', $stock->quantity, 'cost', money($stock->cost));
750 }
751 }
752 foreach ($this->orders as $order) {
753 $quantity = 0;
754 foreach ($order->issues as $part) {
755 if ($part->lot->number === $lot->number) {
756 $quantity += $part->quantity;
757 }
758 }
759 if ($quantity > 0) {
760 $this->emit(2, 'order', $order->number, 'qty', $quantity);
761 }
762 }
763 if ($lot->removed !== 0) {
764 $this->emit(2, 'removed', 'qty', $lot->removed);
765 }
766 return;
767
768 case 'CARD':
769 $item = $this->item($a[0]);
770 $this->emit(0, 'CARD', $item->code, $item->name);
771 $quantity = 0;
772 $value = 0;
773 foreach ($item->card as $entry) {
774 $quantity += $entry->quantity;
775 $value += $entry->value;
776 $this->emit(2, 'day', $entry->day, $entry->kind, $entry->warehouse,
777 signedQuantity($entry->quantity), signedMoney($entry->value),
778 'balance', $quantity, money($value));
779 }
780 if (!$item->card) {
781 $this->emit(2, '(none)');
782 }
783 return;
784
785 case 'STATUS':
786 $this->emit(0, 'STATUS');
787 foreach ($this->sortedWarehouses() as $warehouse) {
788 $skus = 0;
789 $stocks = [];
790 foreach ($this->stock[$warehouse] as $holdings) {
791 if ($holdings) {
792 ++$skus;
793 }
794 array_push($stocks, ...array_values($holdings));
795 }
796 $backorder = 0;
797 foreach ($this->orders as $order) {
798 if ($order->warehouse === $warehouse) {
799 $backorder += $order->backorder;
800 }
801 }
802 $this->emit(2, $warehouse, 'skus', $skus, 'qty', stockQuantity($stocks),
803 'value', money(stockValue($stocks)), 'backorder', $backorder);
804 }
805 if (!$this->stock) {
806 $this->emit(2, '(none)');
807 }
808 return;
809
810 case 'SUMMARY':
811 $this->emit(0, 'SUMMARY');
812 foreach (CATEGORIES as $category) {
813 $total = $this->totals[$category];
814 $this->emit(2, $category, 'qty', $total->quantity, 'value', money($total->value));
815 }
816 $stocks = $this->allStocks();
817 $this->emit(2, 'on-hand', 'qty', stockQuantity($stocks), 'value', money(stockValue($stocks)));
818 $open = 0;
819 $backorder = 0;
820 foreach ($this->orders as $order) {
821 if ($order->backorder > 0) {
822 ++$open;
823 }
824 $backorder += $order->backorder;
825 }
826 $this->emit(2, 'orders', count($this->orders), 'open', $open, 'backorder', $backorder);
827 return;
828 }
829 }

831 public function execute(string $line): void {
832 $parts = explode(' ', $line);
833 $command = array_shift($parts);
834 try {
835 $this->command($command, $parts);
836 } catch (CommandError $error) {
837 $this->emit(0, 'ERROR', $error->getMessage());
838 }
839 }

841 public function result(): string {
842 return implode("\n", $this->output) . "\n";
843 }
