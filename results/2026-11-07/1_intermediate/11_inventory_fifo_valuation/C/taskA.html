127static void *grow(void *data, size_t *capacity, size_t length, size_t bytes) {
128 if (length >= *capacity) {
129 size_t target = *capacity == 0 ? 32 : 2 * *capacity;
130 void *next = realloc(data, bytes * target);
131 if (!next) {
132 fputs("out of memory\n", stderr);
133 exit(EXIT_FAILURE);
134 }
135 data = next;
136 *capacity = target;
137 }
138 return data;
139}

141static const char *money(Amount value) {
142 static char buffers[16][64];
143 static unsigned int cursor;
144 char *text = buffers[(cursor++) % 16];
145 const char *sign = value < 0 ? "-" : "";
146 Amount absolute = value < 0 ? -value : value;
147 snprintf(text, 64, "%s%" PRId64 ".%02" PRId64,
148 sign, absolute / 100, absolute % 100);
149 return text;
150}

156 if (value > 0) snprintf(text, 48, "+%" PRId64, value);
157 else snprintf(text, 48, "%" PRId64, value);
158 return text;
159}

166 else snprintf(text, 64, "%s", money(value));

170static const char *margin(Amount profit, Amount revenue) {
171 static char buffers[16][64];
172 static unsigned int cursor;
173 char *text = buffers[(cursor++) % 16];
174 Amount absolute = profit < 0 ? -profit : profit;
175 Amount tenths;
176 if (revenue == 0) {
177 strcpy(text, "n/a");
178 return text;
179 }
180 tenths = absolute * 1000 / revenue;
181 snprintf(text, 64, "%s%" PRId64 ".%" PRId64 "%%",
182 profit < 0 && tenths != 0 ? "-" : "", tenths / 10, tenths % 10);
183 return text;
184}

188 return atoll(text) * 100 + atoll(point + 1);

312static void add_stock(int warehouse, int lot, Amount quantity, Amount cost) {
313 size_t i;
314 for (i = 0; i < stock_count; ++i)

333static int earlier_stock(const Stock *first, const Stock *second) {
334 const Lot *a = &lots[first->lot], *b = &lots[second->lot];
335 return a->day < b->day || (a->day == b->day && first->lot < second->lot);

338static Amount issue(int warehouse, int item, Amount quantity, PartList *parts) {
339 Amount cost = 0;
340 while (quantity > 0) {
341 Stock *chosen = NULL;
342 size_t i;
343 for (i = 0; i < stock_count; ++i) {
344 Stock *candidate = &stocks[i];
345 if (!candidate->active || candidate->warehouse != warehouse ||
346 candidate->item != item)
347 continue;
348 if (!chosen || earlier_stock(candidate, chosen)) chosen = candidate;
349 }
350 {
351 Amount taken = chosen->quantity < quantity ? chosen->quantity : quantity;
352 append_part(parts, (Part){chosen->lot, taken, chosen->cost});
353 cost += taken * chosen->cost;
354 chosen->quantity -= taken;
355 quantity -= taken;
356 if (chosen->quantity == 0) chosen->active = 0;
357 }
358 }
359 return cost;
360}

362static void remove_parts(const PartList *parts) {
363 size_t i;

365 lots[parts->entries[i].lot].removed += parts->entries[i].quantity;

368static Amount fulfil(Order *order, Amount quantity, int kind) {
369 PartList parts = {0};
370 Amount cost;
371 size_t i;
372 if (quantity == 0) return 0;
373 cost = issue(order->warehouse, order->item, quantity, &parts);
374 for (i = 0; i < parts.length; ++i)
375 append_part(&order->issues, parts.entries[i]);
376 order->shipped += quantity;
377 order->backorder -= quantity;
378 order->revenue += quantity * order->price;
379 order->cogs += cost;
380 accumulate(SHIPPED, quantity, cost);
381 append_card(order->item, kind, order->warehouse, -quantity, -cost);
382 free(parts.entries);
383 return cost;
384}

386static void allocate_backorders(int warehouse, int item) {
387 Amount available = onhand(warehouse, item);
388 int i;
389 for (i = 0; i < order_count && available > 0; ++i) {
390 Order *order = &orders[i];
391 Amount filled, cost, revenue;
392 if (order->warehouse != warehouse || order->item != item ||
393 order->backorder == 0)
394 continue;
395 filled = available < order->backorder ? available : order->backorder;
396 cost = fulfil(order, filled, CARD_FILL);
397 revenue = filled * order->price;
398 available -= filled;
399 printf(" FILL order %d qty %" PRId64 " revenue %s cogs %s profit %s backorder %" PRId64 "\n",
400 i + 1, filled, money(revenue), money(cost), money(revenue - cost),
401 order->backorder);
402 }
403}

458static void receive_command(char **a) {
459 int warehouse, item, lot;
460 Amount quantity, cost;
461 int day;
462 if (!require_pair(a[0], a[1], &warehouse, &item)) return;
463 quantity = atoll(a[2]);
464 cost = parse_money(a[3]);
465 day = atoi(a[4]);
466 if (!require_quantity(quantity)) return;
467 if (day > current_day) {
468 printf("ERROR future day %s\n", a[4]);
469 return;
470 }
471 lot = create_lot(warehouse, item, day, quantity, cost, 0);
472 items[item].last_cost = cost;
473 items[item].received_cost = 1;
474 accumulate(RECEIVED, quantity, quantity * cost);
475 append_card(item, CARD_RECEIVE, warehouse, quantity, quantity * cost);
476 printf("RECEIVE %s %s lot %d qty %" PRId64 " cost %s day %d value %s\n",
477 a[0], a[1], lot + 1, quantity, money(cost), day, money(quantity * cost));
478 allocate_backorders(warehouse, item);
479}

481static void ship_command(char **a) {
482 int warehouse, item, index;
483 Amount quantity, price, filled, cost;
484 Order *order;
485 if (!require_pair(a[0], a[1], &warehouse, &item)) return;
486 quantity = atoll(a[2]);
487 price = parse_money(a[3]);
488 if (!require_quantity(quantity)) return;
489 index = order_count++;
490 order = &orders[index];
491 memset(order, 0, sizeof(*order));
492 order->warehouse = warehouse;
493 order->item = item;
494 order->day = current_day;
495 order->price = price;
496 order->ordered = quantity;
497 order->backorder = quantity;
498 filled = onhand(warehouse, item);
499 if (filled > quantity) filled = quantity;
500 cost = fulfil(order, filled, CARD_SHIP);
501 printf("SHIP order %d %s %s shipped %" PRId64 " revenue %s cogs %s profit %s backorder %" PRId64 "\n",
502 index + 1, a[0], a[1], filled, money(order->revenue),
503 money(cost), money(order->revenue - cost), order->backorder);
504}

519static void transfer_command(char **a) {
520 int source = require_warehouse(a[0]), destination, item;
521 Amount quantity, cost;
522 PartList parts = {0};
523 size_t i;
524 if (source < 0) return;
525 destination = require_warehouse(a[1]);
526 if (destination < 0) return;
527 if (source == destination) {
528 printf("ERROR same warehouse %s\n", a[0]);
529 return;
530 }
531 item = require_item(a[2]);
532 if (item < 0) return;
533 quantity = atoll(a[3]);
534 if (!require_quantity(quantity) || !require_stock(source, item, quantity)) return;
535 cost = issue(source, item, quantity, &parts);
536 for (i = 0; i < parts.length; ++i)
537 add_stock(destination, parts.entries[i].lot, parts.entries[i].quantity,
538 parts.entries[i].cost);
539 append_card(item, CARD_TRANSFER_OUT, source, -quantity, -cost);
540 append_card(item, CARD_TRANSFER_IN, destination, quantity, cost);
541 printf("TRANSFER %s -> %s %s qty %" PRId64 " value %s lots %zu\n",
542 a[0], a[1], a[2], quantity, money(cost), parts.length);
543 free(parts.entries);
544 allocate_backorders(destination, item);
545}

547static void adjust_command(char **a) {
548 int warehouse, item;
549 Amount quantity, cost;
550 PartList parts = {0};
551 if (!require_pair(a[0], a[1], &warehouse, &item)) return;
552 quantity = atoll(a[2] + 1);
553 if (!require_quantity(quantity)) return;
554 if (a[2][0] == '+') {
555 int lot = create_lot(warehouse, item, current_day, quantity, parse_money(a[3]), 1);
556 cost = lots[lot].cost;
557 accumulate(ADJUSTED_IN, quantity, quantity * cost);
558 append_card(item, CARD_ADJUST, warehouse, quantity, quantity * cost);
559 printf("ADJUST %s %s +%" PRId64 " lot %d value %s\n",
560 a[0], a[1], quantity, lot + 1, money(quantity * cost));
561 allocate_backorders(warehouse, item);
562 } else {
563 if (!require_stock(warehouse, item, quantity)) return;
564 cost = issue(warehouse, item, quantity, &parts);
565 remove_parts(&parts);
566 accumulate(ADJUSTED_OUT, quantity, cost);
567 append_card(item, CARD_ADJUST, warehouse, -quantity, -cost);
568 printf("ADJUST %s %s -%" PRId64 " value %s\n",
569 a[0], a[1], quantity, money(cost));
570 free(parts.entries);
571 }
572}

574static void return_command(char **a) {
575 Order *order = require_order(a[0]);
576 Amount quantity, cost, available;
577 int lot;
578 if (!order) return;
579 quantity = atoll(a[1]);
580 cost = parse_money(a[2]);
581 if (!require_quantity(quantity)) return;
582 available = order->shipped - order->returned;
583 if (quantity > available) {
584 printf("ERROR return exceeds shipped %s %" PRId64 "\n", a[0], available);
585 return;
586 }
587 lot = create_lot(order->warehouse, order->item, current_day, quantity, cost, 2);
588 order->returned += quantity;
589 order->revenue -= quantity * order->price;
590 order->cogs -= quantity * cost;
591 accumulate(RETURNED, quantity, quantity * cost);
592 append_card(order->item, CARD_RETURN, order->warehouse, quantity, quantity * cost);
593 printf("RETURN order %ld %s %s qty %" PRId64 " lot %d refund %s value %s\n",
594 (long)(order - orders) + 1, warehouses[order->warehouse].code,
595 items[order->item].code, quantity, lot + 1, money(quantity * order->price),
596 money(quantity * cost));
597 allocate_backorders(order->warehouse, order->item);
598}

600static void count_command(char **a) {
601 int warehouse, item;
602 Amount actual, held, delta, cost;
603 PartList parts = {0};
604 if (!require_pair(a[0], a[1], &warehouse, &item)) return;
605 actual = atoll(a[2]);
606 held = onhand(warehouse, item);
607 if (actual == held) {
608 printf("COUNT %s %s qty %" PRId64 " ok\n", a[0], a[1], actual);
609 } else if (actual < held) {
610 delta = held - actual;
611 cost = issue(warehouse, item, delta, &parts);
612 remove_parts(&parts);
613 accumulate(ADJUSTED_OUT, delta, cost);
614 append_card(item, CARD_COUNT, warehouse, -delta, -cost);
615 printf("COUNT %s %s qty %" PRId64 " -> %" PRId64 " shortage %" PRId64 " value %s\n",
616 a[0], a[1], held, actual, delta, money(cost));
617 free(parts.entries);
618 } else {
619 int lot;
620 delta = actual - held;
621 cost = items[item].received_cost ? items[item].last_cost : 0;
622 lot = create_lot(warehouse, item, current_day, delta, cost, 3);
623 accumulate(ADJUSTED_IN, delta, delta * cost);
624 append_card(item, CARD_COUNT, warehouse, delta, delta * cost);
625 printf("COUNT %s %s qty %" PRId64 " -> %" PRId64 " surplus %" PRId64 " lot %d value %s\n",
626 a[0], a[1], held, actual, delta, lot + 1, money(delta * cost));
627 allocate_backorders(warehouse, item);
628 }
629}

631static void market_command(char **a) {
632 int item = require_item(a[0]);
633 Amount price, quantity = 0, loss = 0;
634 int count = 0, order[MAX_WAREHOUSES], k;
635 size_t i;
636 if (item < 0) return;
637 price = parse_money(a[1]);
638 warehouse_order(order);
639 for (k = 0; k < warehouse_count; ++k) {
640 int warehouse = order[k];
641 Amount warehouse_loss = 0;
642 for (i = 0; i < stock_count; ++i) {
643 Stock *stock = &stocks[i];
644 if (stock->active && stock->warehouse == warehouse &&
645 stock->item == item && stock->cost > price) {
646 Amount decrease = stock->quantity * (stock->cost - price);
647 warehouse_loss += decrease;
648 ++count;
649 quantity += stock->quantity;
650 stock->cost = price;
651 }
652 }
653 if (warehouse_loss > 0)
654 append_card(item, CARD_WRITEDOWN, warehouse, 0, -warehouse_loss);
655 loss += warehouse_loss;
656 }
657 accumulate(WRITTEN_DOWN, quantity, loss);
658 printf("MARKET %s price %s lots %d qty %" PRId64 " loss %s\n",
659 a[0], money(price), count, quantity, money(loss));
660}

662static void recall_command(const char *text) {
663 Lot *lot = require_lot(text);
664 int warehouse_ordered[MAX_WAREHOUSES], k;
665 int number, item;
666 Amount quantity = 0, value = 0;
667 size_t i;
668 if (!lot) return;
669 number = (int)(lot - lots);
670 item = lot->item;
671 for (i = 0; i < stock_count; ++i)
672 if (stocks[i].active && stocks[i].lot == number) {
673 quantity += stocks[i].quantity;
674 value += stocks[i].quantity * stocks[i].cost;
675 }
676 if (!quantity) {
677 printf("ERROR nothing to recall %s\n", text);
678 return;
679 }
680 lot->removed += quantity;
681 accumulate(RECALLED, quantity, value);
682 printf("RECALL lot %d %s qty %" PRId64 " value %s\n",
683 number + 1, items[item].code, quantity, money(value));
684 warehouse_order(warehouse_ordered);
685 for (k = 0; k < warehouse_count; ++k) {
686 int warehouse = warehouse_ordered[k];
687 Amount warehouse_qty = 0, warehouse_value = 0;
688 for (i = 0; i < stock_count; ++i) {
689 Stock *stock = &stocks[i];
690 if (stock->active && stock->lot == number && stock->warehouse == warehouse) {
691 warehouse_qty += stock->quantity;
692 warehouse_value += stock->quantity * stock->cost;
693 stock->active = 0;
694 }
695 }
696 if (warehouse_qty) {

782static int compare_fifo_stock(const void *left, const void *right) {
783 int a = *(const int *)left, b = *(const int *)right;
784 return earlier_stock(&stocks[a], &stocks[b]) ? -1 :
785 earlier_stock(&stocks[b], &stocks[a]) ? 1 : 0;

788static void lots_command(char **a) {

800 if (!indices) exit(EXIT_FAILURE);

812 ++printed;
813 }
814 free(indices);
815 for (order = 0; order < order_count; ++order) {
816 Order *entry = &orders[order];
817 if (entry->warehouse == warehouse && entry->item == item && entry->backorder) {
818 printf(" backorder order %d qty %" PRId64 "\n", order + 1, entry->backorder);
819 ++printed;
820 }
821 }
822 if (!printed) puts(" (none)");
823}

899static Sales top_values[MAX_ITEMS];

901static int top_comparator(const void *left, const void *right) {
902 int a = *(const int *)left, b = *(const int *)right;
903 Amount a_profit = top_values[a].revenue - top_values[a].cogs;
904 Amount b_profit = top_values[b].revenue - top_values[b].cogs;
905 if (a_profit != b_profit) return a_profit > b_profit ? -1 : 1;
906 if (top_values[a].revenue != top_values[b].revenue)
907 return top_values[a].revenue > top_values[b].revenue ? -1 : 1;
908 return strcmp(items[a].code, items[b].code);
909}

911static void top_command(int count) {
912 int indices[MAX_ITEMS], used = 0, i;
913 summaries(top_values);
914 for (i = 0; i < item_count; ++i)
915 if (top_values[i].has_order) indices[used++] = i;
916 qsort(indices, (size_t)used, sizeof(int), top_comparator);
917 printf("TOP %d\n", count);
918 for (i = 0; i < count && i < used; ++i) {
919 int item = indices[i];
920 Amount revenue = top_values[item].revenue;
921 Amount profit = revenue - top_values[item].cogs;
922 printf(" %d %s profit %s revenue %s margin %s\n",
923 i + 1, items[item].code, money(profit),
924 money(revenue), margin(profit, revenue));
925 }
926 if (!used) puts(" (none)");
927}

940static void aging_command(int selected_warehouse) {
941 int item_indices[MAX_ITEMS], i, item, bucket;
942 int warehouse_indices[MAX_WAREHOUSES], warehouse, w, count = warehouse_count;
943 Amount total_quantities[4] = {0}, total_values[4] = {0};
944 size_t s;
945 item_order(item_indices);
946 if (selected_warehouse < 0) {
947 warehouse_order(warehouse_indices);
948 printf("AGING day %d\n", current_day);
949 } else {
950 warehouse_indices[0] = selected_warehouse;
951 count = 1;
952 printf("AGING %s day %d\n",
953 warehouses[selected_warehouse].code, current_day);
954 }
955 for (i = 0; i < item_count; ++i) {
956 Amount quantities[4] = {0};
957 item = item_indices[i];

959 warehouse = warehouse_indices[w];
960 for (s = 0; s < stock_count; ++s) {

963 if (!stock->active || stock->item != item ||
964 stock->warehouse != warehouse) continue;

966 bucket = age <= 30 ? 0 : age <= 60 ? 1 : age <= 90 ? 2 : 3;
967 quantities[bucket] += stock->quantity;
968 total_quantities[bucket] += stock->quantity;
969 total_values[bucket] += stock->quantity * stock->cost;
970 }
971 }
972 if (quantities[0] || quantities[1] || quantities[2] || quantities[3])
973 aging_line(items[item].code, quantities, 0);
974 }
975 aging_line("total", total_quantities, 0);
976 aging_line("value", total_values, 1);
977}

1116static void execute(char *line) {
1117 char *tokens[8];
1118 char *command;
1119 int count = 0;
1120 char *token = strtok(line, " ");
1121 if (!token) return;
1122 command = token;
1123 while ((token = strtok(NULL, " ")) != NULL)
1124 tokens[count++] = token;
1125 if (strcmp(command, "DAY") == 0) day_command(tokens[0]);
1126 else if (strcmp(command, "WAREHOUSE") == 0) warehouse_command(tokens[0]);
1127 else if (strcmp(command, "ITEM") == 0)
1128 item_command(tokens[0], tokens[1], tokens[2]);
1129 else if (strcmp(command, "RECEIVE") == 0) receive_command(tokens);
1130 else if (strcmp(command, "SHIP") == 0) ship_command(tokens);
1131 else if (strcmp(command, "CANCEL") == 0) cancel_command(tokens[0]);
1132 else if (strcmp(command, "TRANSFER") == 0) transfer_command(tokens);
1133 else if (strcmp(command, "ADJUST") == 0) adjust_command(tokens);
1134 else if (strcmp(command, "RETURN") == 0) return_command(tokens);
1135 else if (strcmp(command, "COUNT") == 0) count_command(tokens);
1136 else if (strcmp(command, "MARKET") == 0) market_command(tokens);
1137 else if (strcmp(command, "RECALL") == 0) recall_command(tokens[0]);
1138 else if (strcmp(command, "EXPIRE") == 0) expire_command(tokens[0]);
1139 else if (strcmp(command, "VALUE") == 0) {
1140 int warehouse = -1;
1141 if (count && (warehouse = require_warehouse(tokens[0])) < 0) return;
1142 value_command(warehouse);
1143 } else if (strcmp(command, "LOTS") == 0) lots_command(tokens);
1144 else if (strcmp(command, "REORDER") == 0) reorder_command();
1145 else if (strcmp(command, "ITEMS") == 0) items_command();
1146 else if (strcmp(command, "MARGIN") == 0) margin_command();
1147 else if (strcmp(command, "TOP") == 0) top_command(atoi(tokens[0]));
1148 else if (strcmp(command, "AGING") == 0) {
1149 int warehouse = -1;
1150 if (count && (warehouse = require_warehouse(tokens[0])) < 0) return;
1151 aging_command(warehouse);
1152 } else if (strcmp(command, "BACKORDERS") == 0) backorders_command();
1153 else if (strcmp(command, "ORDER") == 0) order_command(tokens[0]);
1154 else if (strcmp(command, "TRACE") == 0) trace_command(tokens[0]);
1155 else if (strcmp(command, "CARD") == 0) card_command(tokens[0]);
1156 else if (strcmp(command, "STATUS") == 0) status_command();
1157 else if (strcmp(command, "SUMMARY") == 0) summary_command();
1158}

1160int main(void) {
1161 char line[1024];
1162 int count, i;
1163 if (!fgets(line, sizeof(line), stdin)) return 0;
1164 count = atoi(line);

1166 if (!fgets(line, sizeof(line), stdin)) break;

1169 }
1170 for (i = 0; i < order_count; ++i) free(orders[i].issues.entries);
1171 free(cards);
1172 free(stocks);
1173 return 0;
