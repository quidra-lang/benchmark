85 quantity := 0

90 value += amount * state.lots[k].cost
91 return [quantity, value]

114void card_entry(State &state, int item, string kind, int warehouse, int quantity, int value)
115 if state.card_used == int(#state.cards)
116 capacity := if state.card_used == 0 then 16 else 2 * state.card_used

131int create_lot(State &state, int warehouse, int item, int quantity, int cost, int day, string kind)
132 index := state.lot_used
133 state.lot_used += 1
134 state.lots[index] = Lot(item, day, quantity, cost, warehouse, kind, cost, 0)
135 state.stock[index][warehouse] = quantity
136 state.item_lots[item][state.item_lot_used[item]] = index
137 state.item_lot_used[item] += 1
138 position := index
139 while position > 0 and state.lots[state.fifo[position - 1]].day > day
140 state.fifo[position] = state.fifo[position - 1]
141 position -= 1
142 state.fifo[position] = index
143 return index

145Part[] issue_stock(State &state, int warehouse, int item, int quantity)
146 Part[] result := []
147 for: 0 <= k < state.lot_used
148 if quantity == 0
149 break
150 int index := state.fifo[k]
151 if state.lots[index].item != item or state.stock[index][warehouse] == 0
152 continue
153 taken := math.min(quantity, state.stock[index][warehouse])
154 state.stock[index][warehouse] -= taken
155 quantity -= taken
156 result = result.append(Part(index, taken, state.lots[index].cost))
157 return result

159int parts_cost(const Part[] parts)
160 costs := [for part := parts[] then part.quantity * part.cost]

164 for part := parts[]

166 capacity := if state.issue_used == 0 then 16 else 2 * state.issue_used

174 tail := state.orders[order].issue_tail
175 if tail < 0
176 state.orders[order].issue_head = index
177 else
178 state.issues[tail].next = index
179 state.orders[order].issue_tail = index

185void allocate(State &state, int warehouse, int item)
186 for: 0 <= k < state.order_used
187 if state.orders[k].warehouse != warehouse or state.orders[k].item != item or state.orders[k].backorder == 0
188 continue
189 available := holding(state, warehouse, item)[0]
190 if available == 0
191 return
192 quantity := math.min(available, state.orders[k].backorder)
193 parts := issue_stock(&state, warehouse, item, quantity)
194 cost := parts_cost(parts)
195 revenue := quantity * state.orders[k].price
196 state.orders[k].shipped += quantity
197 state.orders[k].backorder -= quantity
198 state.orders[k].revenue += revenue
199 state.orders[k].cogs += cost
200 remember_issues(&state, k, parts)
201 accumulate(&state, 4, quantity, cost)
202 card_entry(&state, item, "fill", warehouse, -quantity, -cost)
203 emit(" FILL order " + string(k + 1) + " qty " + string(quantity) + " revenue " + money(revenue) + " cogs " + money(cost) + " profit " + money(revenue - cost) + " backorder " + string(state.orders[k].backorder))

216void command(State &state, string line)
217 words := line.split(" ")
218 action := words[0]
219 if action == "DAY"
220 day := integer_value(words[1])
221 if day <= state.day
222 fail("invalid day " + words[1])
223 else
224 state.day = day
225 emit("DAY " + string(day))
226 return
227 if action == "WAREHOUSE"
228 if warehouse_id(state, words[1]) >= 0
229 fail("duplicate warehouse " + words[1])
230 else
231 state.warehouses[state.warehouse_used] = words[1]
232 state.warehouse_used += 1
233 emit("WAREHOUSE " + words[1])
234 return
235 if action == "ITEM"
236 item := item_id(state, words[1])
237 added := item < 0
238 if added
239 item = state.item_used
240 state.item_used += 1
241 state.items[item] = Item(words[1], words[3], integer_value(words[2]), -1, -1, -1)
242 else
243 state.items[item].name = words[3]
244 state.items[item].level = integer_value(words[2])
245 emit("ITEM " + words[1] + (if added then " added " else " updated ") + words[3] + " level " + words[2])
246 return
247 if action == "CANCEL" or action == "RETURN" or action == "ORDER"
248 order := integer_value(words[1]) - 1
249 if order < 0 or order >= state.order_used
250 fail("unknown order " + words[1])
251 return
252 entry := state.orders[order]
253 string warehouse := state.warehouses[entry.warehouse]
254 item := state.items[entry.item].code
255 if action == "CANCEL"
256 if entry.backorder == 0
257 fail("no backorder " + words[1])
258 else
259 state.orders[order].cancelled += entry.backorder
260 state.orders[order].backorder = 0
261 emit("CANCEL order " + words[1] + " qty " + string(entry.backorder))
262 elif action == "RETURN"
263 quantity := integer_value(words[2])
264 cost := cents(words[3])
265 if quantity == 0
266 fail("invalid quantity")
267 elif quantity > entry.shipped - entry.returned
268 fail("return exceeds shipped " + words[1] + " " + string(entry.shipped - entry.returned))
269 else
270 lot := create_lot(&state, entry.warehouse, entry.item, quantity, cost, state.day, "return")
271 state.orders[order].returned += quantity
272 state.orders[order].revenue -= quantity * entry.price
273 state.orders[order].cogs -= quantity * cost
274 accumulate(&state, 1, quantity, quantity * cost)
275 card_entry(&state, entry.item, "return", entry.warehouse, quantity, quantity * cost)
276 emit("RETURN order " + words[1] + " " + warehouse + " " + item + " qty " + string(quantity) + " lot " + string(lot + 1) + " refund " + money(quantity * entry.price) + " value " + money(quantity * cost))
277 allocate(&state, entry.warehouse, entry.item)
278 else
279 emit("ORDER " + words[1] + " " + warehouse + " " + item + " day " + string(entry.day) + " price " + money(entry.price))
280 emit(" ordered " + string(entry.ordered) + " shipped " + string(entry.shipped) + " backorder " + string(entry.backorder) + " cancelled " + string(entry.cancelled) + " returned " + string(entry.returned))
281 cursor := entry.issue_head
282 while cursor >= 0
283 part := state.issues[cursor]
284 emit(" lot " + string(part.lot + 1) + " qty " + string(part.quantity) + " cost " + money(part.cost))
285 cursor = part.next
286 emit(" revenue " + money(entry.revenue) + " cogs " + money(entry.cogs) + " profit " + money(entry.revenue - entry.cogs))
287 return
288 if action == "RECEIVE" or action == "SHIP" or action == "TRANSFER" or action == "ADJUST" or action == "COUNT" or action == "LOTS"
289 warehouse := warehouse_id(state, words[1])
290 if warehouse < 0
291 fail("unknown warehouse " + words[1])
292 return
293 destination := -1
294 item_position := 2
295 if action == "TRANSFER"
296 destination = warehouse_id(state, words[2])
297 if destination < 0
298 fail("unknown warehouse " + words[2])
299 return
300 if destination == warehouse
301 fail("same warehouse " + words[1])
302 return
303 item_position = 3
304 item := item_id(state, words[item_position])
305 if item < 0
306 fail("unknown item " + words[item_position])
307 return
308 string pair := state.warehouses[warehouse] + " " + state.items[item].code
309 available := holding(state, warehouse, item)[0]
310 if action == "LOTS"
311 emit("LOTS " + pair)
312 shown := 0
313 for: 0 <= k < state.lot_used
314 int lot := state.fifo[k]
315 if state.lots[lot].item == item and state.stock[lot][warehouse] > 0
316 entry := state.lots[lot]
317 quantity := state.stock[lot][warehouse]
318 emit(" lot " + string(lot + 1) + " day " + string(entry.day) + " qty " + string(quantity) + " cost " + money(entry.cost) + " value " + money(quantity * entry.cost) + " age " + string(state.day - entry.day))
319 shown += 1
320 for: 0 <= k < state.order_used
321 entry := state.orders[k]
322 if entry.warehouse == warehouse and entry.item == item and entry.backorder > 0
323 emit(" backorder order " + string(k + 1) + " qty " + string(entry.backorder))
324 shown += 1
325 if shown == 0
326 emit(" (none)")
327 return
328 quantity := 0
329 positive := false
330 if action == "ADJUST"
331 positive = words[3].starts_with("+")
332 quantity = integer_value(words[3][1:])
333 else
334 quantity = integer_value(words[item_position + 1])
335 if quantity == 0 and action != "COUNT"
336 fail("invalid quantity")
337 return
338 if (action == "TRANSFER" or (action == "ADJUST" and not positive)) and available < quantity
339 fail("insufficient stock " + pair + " " + string(available))
340 return
341 if action == "RECEIVE"
342 cost := cents(words[4])
343 day := integer_value(words[5])
344 if day > state.day
345 fail("future day " + words[5])
346 return
347 lot := create_lot(&state, warehouse, item, quantity, cost, day, "receive")
348 state.items[item].last_cost = cost
349 accumulate(&state, 0, quantity, quantity * cost)
350 card_entry(&state, item, "receive", warehouse, quantity, quantity * cost)
351 emit("RECEIVE " + pair + " lot " + string(lot + 1) + " qty " + string(quantity) + " cost " + money(cost) + " day " + string(day) + " value " + money(quantity * cost))
352 allocate(&state, warehouse, item)
353 elif action == "SHIP"
354 price := cents(words[4])
355 filled := math.min(quantity, available)
356 parts := issue_stock(&state, warehouse, item, filled)
357 cost := parts_cost(parts)
358 order := state.order_used
359 state.order_used += 1
360 state.orders[order] = Order(warehouse, item, state.day, price, quantity, filled, quantity - filled, 0, 0, filled * price, cost, -1, -1)
361 remember_issues(&state, order, parts)
362 if filled > 0
363 accumulate(&state, 4, filled, cost)
364 card_entry(&state, item, "ship", warehouse, -filled, -cost)
365 emit("SHIP order " + string(order + 1) + " " + pair + " shipped " + string(filled) + " revenue " + money(filled * price) + " cogs " + money(cost) + " profit " + money(filled * price - cost) + " backorder " + string(quantity - filled))
366 elif action == "TRANSFER"
367 parts := issue_stock(&state, warehouse, item, quantity)
368 cost := parts_cost(parts)
369 for part := parts[]
370 state.stock[part.lot][destination] += part.quantity
371 card_entry(&state, item, "transfer-out", warehouse, -quantity, -cost)
372 card_entry(&state, item, "transfer-in", destination, quantity, cost)
373 emit("TRANSFER " + state.warehouses[warehouse] + " -> " + state.warehouses[destination] + " " + state.items[item].code + " qty " + string(quantity) + " value " + money(cost) + " lots " + string(#parts))
374 allocate(&state, destination, item)
375 elif action == "ADJUST" and positive
376 cost := cents(words[4])
377 lot := create_lot(&state, warehouse, item, quantity, cost, state.day, "adjust")
378 accumulate(&state, 2, quantity, quantity * cost)
379 card_entry(&state, item, "adjust", warehouse, quantity, quantity * cost)
380 emit("ADJUST " + pair + " +" + string(quantity) + " lot " + string(lot + 1) + " value " + money(quantity * cost))
381 allocate(&state, warehouse, item)
382 elif action == "ADJUST" or quantity < available
383 removed := if action == "ADJUST" then quantity else available - quantity
384 parts := issue_stock(&state, warehouse, item, removed)
385 cost := parts_cost(parts)
386 remove_parts(&state, parts)
387 accumulate(&state, 3, removed, cost)
388 card_entry(&state, item, if action == "ADJUST" then "adjust" else "count", warehouse, -removed, -cost)
389 emit((if action == "ADJUST" then "ADJUST " + pair + " -" + string(removed) else "COUNT " + pair + " qty " + string(available) + " -> " + string(quantity) + " shortage " + string(removed)) + " value " + money(cost))
390 elif quantity == available
391 emit("COUNT " + pair + " qty " + string(quantity) + " ok")
392 else
393 cost := if state.items[item].last_cost < 0 then 0 else state.items[item].last_cost
394 added := quantity - available
395 lot := create_lot(&state, warehouse, item, added, cost, state.day, "count")
396 accumulate(&state, 2, added, added * cost)
397 card_entry(&state, item, "count", warehouse, added, added * cost)
398 emit("COUNT " + pair + " qty " + string(available) + " -> " + string(quantity) + " surplus " + string(added) + " lot " + string(lot + 1) + " value " + money(added * cost))
399 allocate(&state, warehouse, item)
400 return
401 if action == "MARKET" or action == "CARD"
402 item := item_id(state, words[1])
403 if item < 0
404 fail("unknown item " + words[1])
405 return
406 if action == "CARD"
407 emit("CARD " + words[1] + " " + state.items[item].name)
408 cursor := state.items[item].card_head
409 quantity := 0
410 value := 0
411 if cursor < 0
412 emit(" (none)")
413 while cursor >= 0
414 entry := state.cards[cursor]
415 quantity += entry.quantity
416 value += entry.value
417 emit(" day " + string(entry.day) + " " + entry.kind + " " + state.warehouses[entry.warehouse] + " " + signed_quantity(entry.quantity) + " " + signed_money(entry.value) + " balance " + string(quantity) + " " + money(value))
418 cursor = entry.next
419 else
420 price := cents(words[2])
421 count := 0
422 quantity := 0
423 loss := 0
424 for warehouse := warehouse_order(state)[]
425 local_loss := 0
426 for lot := lot_ids(state, item)[]
427 entry := state.lots[lot]
428 held := state.stock[lot][warehouse]
429 if entry.item == item and entry.cost > price and held > 0
430 count += 1
431 quantity += held
432 difference := held * (entry.cost - price)
433 loss += difference
434 local_loss += difference
435 if local_loss > 0
436 card_entry(&state, item, "writedown", warehouse, 0, -local_loss)
437 for lot := lot_ids(state, item)[]
438 if state.lots[lot].cost > price
439 state.lots[lot].cost = price
440 accumulate(&state, 5, quantity, loss)
441 emit("MARKET " + words[1] + " price " + money(price) + " lots " + string(count) + " qty " + string(quantity) + " loss " + money(loss))
442 return
443 if action == "RECALL" or action == "TRACE"
444 lot := integer_value(words[1]) - 1
445 if lot < 0 or lot >= state.lot_used
446 fail("unknown lot " + words[1])
447 return
448 entry := state.lots[lot]
449 total := state.stock[lot][:<state.warehouse_used].sum()
450 if action == "RECALL" and total == 0
451 fail("nothing to recall " + words[1])
452 return
453 if action == "TRACE"
454 emit("TRACE lot " + words[1] + " " + state.items[entry.item].code + " " + entry.kind + " " + state.warehouses[entry.warehouse] + " day " + string(entry.day) + " qty " + string(entry.initial_quantity) + " cost " + money(entry.initial_cost))
455 else
456 emit("RECALL lot " + words[1] + " " + state.items[entry.item].code + " qty " + string(total) + " value " + money(total * entry.cost))
457 for warehouse := warehouse_order(state)[]
458 held := state.stock[lot][warehouse]
459 if held > 0
460 if action == "TRACE"
461 emit(" held " + state.warehouses[warehouse] + " qty " + string(held) + " cost " + money(entry.cost))
462 else
463 emit(" " + state.warehouses[warehouse] + " qty " + string(held) + " value " + money(held * entry.cost))
464 state.stock[lot][warehouse] = 0
465 card_entry(&state, entry.item, "recall", warehouse, -held, -held * entry.cost)
466 if action == "RECALL"
467 state.lots[lot].removed += total
468 accumulate(&state, 7, total, total * entry.cost)
469 else
470 issued := array(nat(state.order_used), fill = 0)
471 for: 0 <= k < state.issue_used
472 part := state.issues[k]
473 if part.lot == lot
474 issued[part.order] += part.quantity
475 for: 0 <= k < state.order_used
476 if issued[k] > 0
477 emit(" order " + string(k + 1) + " qty " + string(issued[k]))
478 if entry.removed > 0
479 emit(" removed qty " + string(entry.removed))
480 return
481 if action == "EXPIRE"
482 days := integer_value(words[1])
483 emit("EXPIRE " + string(days))
484 all_quantity := 0
485 all_value := 0
486 for item := item_order(state)[]
487 for warehouse := warehouse_order(state)[]
488 count := 0
489 quantity := 0
490 value := 0
491 for lot := lot_ids(state, item)[]
492 if state.day - state.lots[lot].day > days and state.stock[lot][warehouse] > 0
493 held := state.stock[lot][warehouse]
494 count += 1
495 quantity += held
496 value += held * state.lots[lot].cost
497 state.stock[lot][warehouse] = 0
498 state.lots[lot].removed += held
499 if count > 0
500 card_entry(&state, item, "expire", warehouse, -quantity, -value)
501 accumulate(&state, 6, quantity, value)
502 all_quantity += quantity
503 all_value += value
504 emit(" " + state.items[item].code + " " + state.warehouses[warehouse] + " lots " + string(count) + " qty " + string(quantity) + " value " + money(value))
505 emit(" total qty " + string(all_quantity) + " value " + money(all_value))
506 return
507 if action == "VALUE" or action == "AGING"
508 warehouse := -1
509 if #words == 2
510 warehouse = warehouse_id(state, words[1])
511 if warehouse < 0
512 fail("unknown warehouse " + words[1])
513 return
514 if action == "VALUE"
515 emit("VALUE" + (if warehouse < 0 then "" else " " + words[1]))
516 for item := item_order(state)[]
517 values := holding(state, warehouse, item)
518 if values[0] > 0
519 emit(" " + state.items[item].code + " qty " + string(values[0]) + " value " + money(values[1]) + " avg " + money(values[1] / values[0]))
520 if warehouse < 0
521 for location := warehouse_order(state)[]
522 local_values := holding(state, location, item)
523 if local_values[0] > 0
524 emit(" " + state.warehouses[location] + " qty " + string(local_values[0]) + " value " + money(local_values[1]))
525 total := holding(state, warehouse)
526 emit(" total qty " + string(total[0]) + " value " + money(total[1]))
527 else
528 emit("AGING" + (if warehouse < 0 then "" else " " + words[1]) + " day " + string(state.day))
529 total_quantity := array(4, fill = 0)
530 total_value := array(4, fill = 0)
531 for item := item_order(state)[]
532 quantities := array(4, fill = 0)
533 count := 0
534 for lot := lot_ids(state, item)[]
535 for: 0 <= location < state.warehouse_used
536 if warehouse < 0 or location == warehouse
537 held := state.stock[lot][location]
538 bucket := age_bucket(state.day - state.lots[lot].day)
539 quantities[bucket] += held
540 total_quantity[bucket] += held
541 total_value[bucket] += held * state.lots[lot].cost
542 count += held
543 if count > 0
544 emit(" " + state.items[item].code + " " + aging_fields(quantities, false))
545 emit(" total " + aging_fields(total_quantity, false))
