4const CATEGORIES = ["received", "returned", "adjusted-in", "adjusted-out", "shipped", "written-down", "expired", "recalled"];
5const compare = (a, b) => a < b ? -1 : a > b ? 1 : 0;
6const cents = (s) => BigInt(s.replace(".", ""));
7const value = (s) => BigInt(s.quantity) * s.cost;
8const sumQuantity = (stocks) => stocks.reduce((q, s) => q + s.quantity, 0);
9const sumValue = (stocks) => stocks.reduce((v, s) => v + value(s), 0n);
10const signedQuantity = (q) => q > 0 ? "+" + q : String(q);
11
12function money(v) {
13 const sign = v < 0n ? "-" : "";
14 const magnitude = v < 0n ? -v : v;
15 return sign + magnitude / 100n + "." + String(magnitude % 100n).padStart(2, "0");
16}
17
18function signedMoney(v) {
19 return (v > 0n ? "+" : "") + money(v);
20}
21
22function margin(profit, revenue) {
23 if (revenue === 0n) return "n/a";
24 const magnitude = profit < 0n ? -profit : profit;
25 const q = magnitude * 1000n / revenue;
26 return (profit < 0n && q > 0n ? "-" : "") + q / 10n + "." + q % 10n + "%";
27}

115 addStock(w, lot, quantity, cost) {
116 const pairs = this.stock.get(w);
117 if (!pairs.has(lot.item)) pairs.set(lot.item, new Map());
118 const holdings = pairs.get(lot.item);
119 const existing = holdings.get(lot.number);
120 if (existing) existing.quantity += quantity;
121 else holdings.set(lot.number, {lot, quantity, cost});

124 createLot(w, item, day, quantity, cost, kind) {
125 const lot = {number: this.lots.length + 1, item, day, initialQuantity: quantity, initialCost: cost, warehouse: w, kind, removed: 0};
126 this.lots.push(lot);
127 this.addStock(w, lot, quantity, cost);

131 fifo(w, code) {
132 return this.stocks(w, code).sort((a, b) => a.lot.day - b.lot.day || a.lot.number - b.lot.number);
133 }
134
135 issue(w, code, q) {
136 const parts = [];
137 let cost = 0n;
138 for (const s of this.fifo(w, code)) {
139 if (q === 0) break;
140 const taken = Math.min(q, s.quantity);
141 parts.push({lot: s.lot, quantity: taken, cost: s.cost});
142 cost += BigInt(taken) * s.cost;
143 s.quantity -= taken;
144 q -= taken;
145 if (s.quantity === 0) this.holdings(w, code).delete(s.lot.number);
146 }
147 return {parts, cost};
148 }
149
150 fulfil(o, q, kind) {
151 if (q === 0) return 0n;
152 const {parts, cost} = this.issue(o.warehouse, o.item, q);
153 o.shipped += q;
154 o.backorder -= q;
155 o.revenue += BigInt(q) * o.price;
156 o.cogs += cost;
157 o.issues.push(...parts);
158 this.accumulate("shipped", q, cost);
159 this.cardEntry(o.item, kind, o.warehouse, -q, -cost);
160 return cost;
161 }
162
163 allocate(w, code) {
164 let available = this.quantity(w, code);
165 for (const o of this.orders) {
166 if (available === 0) break;
167 if (o.warehouse !== w || o.item !== code || o.backorder === 0) continue;
168 const q = Math.min(available, o.backorder);
169 const cost = this.fulfil(o, q, "fill");
170 const revenue = BigInt(q) * o.price;
171 available -= q;
172 this.emit(2, "FILL", "order", o.number, "qty", q, "revenue", money(revenue),
173 "cogs", money(cost), "profit", money(revenue - cost), "backorder", o.backorder);
174 }
175 }
176
177 removeParts(parts) {
178 for (const p of parts) p.lot.removed += p.quantity;
179 }
180
181 sales() {
182 const rows = new Map();
183 for (const o of this.orders) {
184 if (!rows.has(o.item)) rows.set(o.item, {quantity: 0, revenue: 0n, cogs: 0n});
185 const row = rows.get(o.item);
186 row.quantity += o.shipped - o.returned;
187 row.revenue += o.revenue;
188 row.cogs += o.cogs;
189 }
190 return rows;
191 }

193 marginLine(label, row) {
194 this.emit(2, label, "sold", row.quantity, "revenue", money(row.revenue), "cogs", money(row.cogs),
195 "profit", money(row.revenue - row.cogs), "margin", margin(row.revenue - row.cogs, row.revenue));
196 }

198 agingLine(label, buckets, monetary = false) {
199 const parts = [label];
200 for (let i = 0; i < 4; i++) parts.push(["0-30", "31-60", "61-90", "91+"][i], monetary ? money(buckets[i]) : Number(buckets[i]));
201 this.emit(2, ...parts);
202 }

204 command(cmd, a) {
205 switch (cmd) {
206 case "DAY": {
207 const day = Number(a[0]);
208 if (day <= this.day) throw new CommandError("invalid day " + a[0]);
209 this.day = day;
210 this.emit(0, "DAY", day);
211 break;
212 }
213 case "WAREHOUSE": {
214 if (this.stock.has(a[0])) throw new CommandError("duplicate warehouse " + a[0]);
215 this.stock.set(a[0], new Map());
216 this.emit(0, "WAREHOUSE", a[0]);
217 break;
218 }
219 case "ITEM": {
220 const item = this.items.get(a[0]);
221 const action = item ? "updated" : "added";
222 if (item) {
223 item.level = Number(a[1]);
224 item.name = a[2];
225 } else {
226 this.items.set(a[0], {code: a[0], level: Number(a[1]), name: a[2], lastCost: null, card: []});
227 }
228 this.emit(0, "ITEM", a[0], action, a[2], "level", Number(a[1]));
229 break;
230 }
231 case "RECEIVE": {
232 const [w, code] = a;
233 const item = this.pair(w, code);
234 const q = Number(a[2]);
235 const cost = cents(a[3]);
236 const day = Number(a[4]);
237 this.requireQuantity(q);
238 if (day > this.day) throw new CommandError("future day " + a[4]);
239 const lot = this.createLot(w, code, day, q, cost, "receive");
240 const amount = BigInt(q) * cost;
241 item.lastCost = cost;
242 this.accumulate("received", q, amount);
243 this.cardEntry(code, "receive", w, q, amount);
244 this.emit(0, "RECEIVE", w, code, "lot", lot.number, "qty", q, "cost", money(cost), "day", day, "value", money(amount));
245 this.allocate(w, code);
246 break;
247 }
248 case "SHIP": {
249 const [w, code] = a;
250 this.pair(w, code);
251 const q = Number(a[2]);
252 this.requireQuantity(q);
253 const o = {number: this.orders.length + 1, warehouse: w, item: code, day: this.day, price: cents(a[3]),
254 ordered: q, shipped: 0, backorder: q, cancelled: 0, returned: 0, revenue: 0n, cogs: 0n, issues: []};
255 this.orders.push(o);
256 const filled = Math.min(q, this.quantity(w, code));
257 const cost = this.fulfil(o, filled, "ship");
258 this.emit(0, "SHIP", "order", o.number, w, code, "shipped", filled, "revenue", money(o.revenue),
259 "cogs", money(cost), "profit", money(o.revenue - cost), "backorder", o.backorder);
260 break;
261 }
262 case "CANCEL": {
263 const o = this.order(a[0]);
264 if (o.backorder === 0) throw new CommandError("no backorder " + a[0]);
265 o.cancelled += o.backorder;
266 o.backorder = 0;
267 this.emit(0, "CANCEL", "order", o.number, "qty", o.cancelled);
268 break;
269 }
270 case "TRANSFER": {
271 const [source, destination, code] = a;
272 this.warehouse(source);
273 this.warehouse(destination);
274 if (source === destination) throw new CommandError("same warehouse " + source);
275 this.item(code);
276 const q = Number(a[3]);
277 this.requireQuantity(q);
278 this.requireStock(source, code, q);
279 const {parts, cost} = this.issue(source, code, q);
280 for (const p of parts) this.addStock(destination, p.lot, p.quantity, p.cost);
281 this.cardEntry(code, "transfer-out", source, -q, -cost);
282 this.cardEntry(code, "transfer-in", destination, q, cost);
283 this.emit(0, "TRANSFER", source, "->", destination, code, "qty", q, "value", money(cost), "lots", parts.length);
284 this.allocate(destination, code);
285 break;
286 }
287 case "ADJUST": {
288 const [w, code, change] = a;
289 this.pair(w, code);
290 const q = Number(change.slice(1));
291 this.requireQuantity(q);
292 if (change[0] === "+") {
293 const cost = cents(a[3]);
294 const amount = BigInt(q) * cost;
295 const lot = this.createLot(w, code, this.day, q, cost, "adjust");
296 this.accumulate("adjusted-in", q, amount);
297 this.cardEntry(code, "adjust", w, q, amount);
298 this.emit(0, "ADJUST", w, code, "+" + q, "lot", lot.number, "value", money(amount));
299 this.allocate(w, code);
300 } else {
301 this.requireStock(w, code, q);
302 const {parts, cost} = this.issue(w, code, q);
303 this.removeParts(parts);
304 this.accumulate("adjusted-out", q, cost);
305 this.cardEntry(code, "adjust", w, -q, -cost);
306 this.emit(0, "ADJUST", w, code, "-" + q, "value", money(cost));
307 }
308 break;
309 }
310 case "RETURN": {
311 const o = this.order(a[0]);
312 const q = Number(a[1]);
313 const cost = cents(a[2]);
314 this.requireQuantity(q);
315 const available = o.shipped - o.returned;
316 if (q > available) throw new CommandError("return exceeds shipped " + a[0] + " " + available);
317 const lot = this.createLot(o.warehouse, o.item, this.day, q, cost, "return");
318 const amount = BigInt(q) * cost;
319 const refund = BigInt(q) * o.price;
320 o.returned += q;
321 o.revenue -= refund;
322 o.cogs -= amount;
323 this.accumulate("returned", q, amount);
324 this.cardEntry(o.item, "return", o.warehouse, q, amount);
325 this.emit(0, "RETURN", "order", o.number, o.warehouse, o.item, "qty", q, "lot", lot.number, "refund", money(refund), "value", money(amount));
326 this.allocate(o.warehouse, o.item);
327 break;
328 }
329 case "COUNT": {
330 const [w, code] = a;
331 const item = this.pair(w, code);
332 const actual = Number(a[2]);
333 const held = this.quantity(w, code);
334 if (actual === held) {
335 this.emit(0, "COUNT", w, code, "qty", actual, "ok");
336 } else if (actual < held) {
337 const q = held - actual;
338 const {parts, cost} = this.issue(w, code, q);
339 this.removeParts(parts);
340 this.accumulate("adjusted-out", q, cost);
341 this.cardEntry(code, "count", w, -q, -cost);
342 this.emit(0, "COUNT", w, code, "qty", held, "->", actual, "shortage", q, "value", money(cost));
343 } else {
344 const q = actual - held;
345 const cost = item.lastCost ?? 0n;
346 const amount = BigInt(q) * cost;
347 const lot = this.createLot(w, code, this.day, q, cost, "count");
348 this.accumulate("adjusted-in", q, amount);
349 this.cardEntry(code, "count", w, q, amount);
350 this.emit(0, "COUNT", w, code, "qty", held, "->", actual, "surplus", q, "lot", lot.number, "value", money(amount));
351 this.allocate(w, code);
352 }
353 break;
354 }
355 case "MARKET": {
356 this.item(a[0]);
357 const price = cents(a[1]);
358 let changed = 0;
359 let q = 0;
360 let loss = 0n;
361 for (const w of [...this.stock.keys()].sort()) {
362 let warehouseLoss = 0n;
363 for (const s of this.stocks(w, a[0])) {
364 if (s.cost <= price) continue;
365 warehouseLoss += BigInt(s.quantity) * (s.cost - price);
366 changed++;
367 q += s.quantity;
368 s.cost = price;
369 }
370 if (warehouseLoss > 0n) this.cardEntry(a[0], "writedown", w, 0, -warehouseLoss);
371 loss += warehouseLoss;
372 }
373 this.accumulate("written-down", q, loss);
374 this.emit(0, "MARKET", a[0], "price", money(price), "lots", changed, "qty", q, "loss", money(loss));
375 break;
376 }
377 case "RECALL": {
378 const lot = this.lot(a[0]);
379 const held = [];
380 for (const w of [...this.stock.keys()].sort()) {
381 const s = this.holdings(w, lot.item).get(lot.number);
382 if (s) held.push({warehouse: w, stock: s});
383 }
384 if (!held.length) throw new CommandError("nothing to recall " + a[0]);
385 const q = sumQuantity(held.map(r => r.stock));
386 const amount = sumValue(held.map(r => r.stock));
387 for (const r of held) {
388 this.holdings(r.warehouse, lot.item).delete(lot.number);
389 this.cardEntry(lot.item, "recall", r.warehouse, -r.stock.quantity, -value(r.stock));
390 }
391 lot.removed += q;
392 this.accumulate("recalled", q, amount);
393 this.emit(0, "RECALL", "lot", lot.number, lot.item, "qty", q, "value", money(amount));
394 for (const r of held) this.emit(2, r.warehouse, "qty", r.stock.quantity, "value", money(value(r.stock)));
395 break;
396 }
397 case "EXPIRE": {
398 const days = Number(a[0]);
399 this.emit(0, "EXPIRE", days);
400 let totalQ = 0;
401 let totalV = 0n;
402 for (const code of [...this.items.keys()].sort()) {
403 for (const w of [...this.stock.keys()].sort()) {
404 const expired = this.stocks(w, code).filter(s => this.day - s.lot.day > days);
405 if (!expired.length) continue;
406 const q = sumQuantity(expired);
407 const amount = sumValue(expired);
408 for (const s of expired) {
409 s.lot.removed += s.quantity;
410 this.holdings(w, code).delete(s.lot.number);
411 }
412 this.accumulate("expired", q, amount);
413 this.cardEntry(code, "expire", w, -q, -amount);
414 totalQ += q;
415 totalV += amount;
416 this.emit(2, code, w, "lots", expired.length, "qty", q, "value", money(amount));
417 }
418 }
419 this.emit(2, "total", "qty", totalQ, "value", money(totalV));
420 break;
421 }
422 case "VALUE":
423 case "AGING": {
424 const warehouses = a.length ? [this.warehouse(a[0])] : [...this.stock.keys()].sort();
425 if (cmd === "VALUE") this.emit(0, "VALUE", ...a);
426 else this.emit(0, "AGING", ...a, "day", this.day);
427 let totalQ = 0;
428 let totalV = 0n;
429 const bucketQ = [0, 0, 0, 0];
430 const bucketV = [0n, 0n, 0n, 0n];
431 for (const code of [...this.items.keys()].sort()) {
432 const rows = warehouses.map(w => ({w, stocks: this.stocks(w, code)}));
433 const stocks = rows.flatMap(r => r.stocks);
434 const q = sumQuantity(stocks);
435 const amount = sumValue(stocks);
436 if (!q) continue;
437 if (cmd === "VALUE") {
438 this.emit(2, code, "qty", q, "value", money(amount), "avg", money(amount / BigInt(q)));
439 if (!a.length) {
440 for (const r of rows) {
441 const quantity = sumQuantity(r.stocks);
442 if (quantity) this.emit(4, r.w, "qty", quantity, "value", money(sumValue(r.stocks)));
443 }
444 }
445 } else {
446 const itemBuckets = [0, 0, 0, 0];
447 for (const s of stocks) {
448 const age = this.day - s.lot.day;
449 const bucket = age <= 30 ? 0 : age <= 60 ? 1 : age <= 90 ? 2 : 3;
450 itemBuckets[bucket] += s.quantity;
451 bucketQ[bucket] += s.quantity;
452 bucketV[bucket] += value(s);
453 }
454 this.agingLine(code, itemBuckets);
455 }
456 totalQ += q;
457 totalV += amount;
458 }
459 if (cmd === "VALUE") this.emit(2, "total", "qty", totalQ, "value", money(totalV));
460 else {
461 this.agingLine("total", bucketQ);
462 this.agingLine("value", bucketV, true);
463 }
464 break;
465 }
466 case "LOTS": {
467 const [w, code] = a;
468 this.pair(w, code);
469 this.emit(0, "LOTS", w, code);
470 const stocks = this.fifo(w, code);
471 const orders = this.orders.filter(o => o.warehouse === w && o.item === code && o.backorder > 0);
472 for (const s of stocks) this.emit(2, "lot", s.lot.number, "day", s.lot.day, "qty", s.quantity, "cost", money(s.cost), "value", money(value(s)), "age", this.day - s.lot.day);
473 for (const o of orders) this.emit(2, "backorder", "order", o.number, "qty", o.backorder);
474 if (!stocks.length && !orders.length) this.emit(2, "(none)");
475 break;
476 }
477 case "REORDER":
478 case "ITEMS": {
479 this.emit(0, cmd);
480 let found = false;
481 for (const code of [...this.items.keys()].sort()) {
482 const item = this.items.get(code);
483 const q = this.totalQuantity(code);
484 const backorder = this.backorder(code);
485 if (cmd === "REORDER") {
486 if (q - backorder > item.level) continue;
487 this.emit(2, code, item.name, "onhand", q, "backorder", backorder, "level", item.level);
488 } else {
489 this.emit(2, code, item.name, "level", item.level, "onhand", q, "backorder", backorder,
490 "cost", item.lastCost === null ? "none" : money(item.lastCost));
491 }
492 found = true;
493 }
494 if (!found) this.emit(2, "(none)");
495 break;
496 }
497 case "MARGIN":
498 case "TOP": {
499 const rows = this.sales();
500 if (cmd === "MARGIN") {
501 this.emit(0, "MARGIN");
502 const total = {quantity: 0, revenue: 0n, cogs: 0n};
503 for (const code of [...rows.keys()].sort()) {
504 const row = rows.get(code);
505 this.marginLine(code, row);
506 total.quantity += row.quantity;
507 total.revenue += row.revenue;
508 total.cogs += row.cogs;
509 }
510 this.marginLine("total", total);
511 } else {
512 this.emit(0, "TOP", Number(a[0]));
513 const ids = [...rows.keys()].sort((a, b) => {
514 const x = rows.get(a);
515 const y = rows.get(b);
516 return compare(y.revenue - y.cogs, x.revenue - x.cogs) || compare(y.revenue, x.revenue) || compare(a, b);
517 });
518 for (let i = 0; i < Math.min(ids.length, Number(a[0])); i++) {
519 const row = rows.get(ids[i]);
520 this.emit(2, i + 1, ids[i], "profit", money(row.revenue - row.cogs), "revenue", money(row.revenue), "margin", margin(row.revenue - row.cogs, row.revenue));
521 }
522 if (!ids.length) this.emit(2, "(none)");
523 }
524 break;
525 }
526 case "BACKORDERS": {
527 this.emit(0, "BACKORDERS");
528 const orders = this.orders.filter(o => o.backorder > 0);
529 for (const o of orders) this.emit(2, "order", o.number, o.warehouse, o.item, "qty", o.backorder, "day", o.day, "price", money(o.price));
530 if (!orders.length) this.emit(2, "(none)");
531 break;
532 }
533 case "ORDER": {
534 const o = this.order(a[0]);
535 this.emit(0, "ORDER", o.number, o.warehouse, o.item, "day", o.day, "price", money(o.price));
536 this.emit(2, "ordered", o.ordered, "shipped", o.shipped, "backorder", o.backorder, "cancelled", o.cancelled, "returned", o.returned);
537 for (const p of o.issues) this.emit(2, "lot", p.lot.number, "qty", p.quantity, "cost", money(p.cost));
538 this.emit(2, "revenue", money(o.revenue), "cogs", money(o.cogs), "profit", money(o.revenue - o.cogs));
539 break;
540 }
541 case "TRACE": {
542 const lot = this.lot(a[0]);
543 this.emit(0, "TRACE", "lot", lot.number, lot.item, lot.kind, lot.warehouse, "day", lot.day, "qty", lot.initialQuantity, "cost", money(lot.initialCost));
544 for (const w of [...this.stock.keys()].sort()) {
545 const s = this.holdings(w, lot.item).get(lot.number);
546 if (s) this.emit(2, "held", w, "qty", s.quantity, "cost", money(s.cost));
547 }
548 for (const o of this.orders) {
549 const q = o.issues.filter(p => p.lot.number === lot.number).reduce((q, p) => q + p.quantity, 0);
550 if (q) this.emit(2, "order", o.number, "qty", q);
551 }
552 if (lot.removed) this.emit(2, "removed", "qty", lot.removed);
553 break;
554 }
555 case "CARD": {
556 const item = this.item(a[0]);
557 this.emit(0, "CARD", item.code, item.name);
558 let q = 0;
559 let amount = 0n;
560 for (const e of item.card) {
561 q += e.quantity;
562 amount += e.value;
563 this.emit(2, "day", e.day, e.kind, e.warehouse, signedQuantity(e.quantity), signedMoney(e.value), "balance", q, money(amount));
564 }
565 if (!item.card.length) this.emit(2, "(none)");
566 break;
567 }
568 case "STATUS": {
569 this.emit(0, "STATUS");
570 for (const w of [...this.stock.keys()].sort()) {
571 const pairs = this.stock.get(w);
572 const stocks = [...pairs.values()].flatMap(lots => [...lots.values()]);
573 const skus = [...pairs.values()].filter(lots => lots.size > 0).length;
574 const backorder = this.orders.filter(o => o.warehouse === w).reduce((q, o) => q + o.backorder, 0);
575 this.emit(2, w, "skus", skus, "qty", sumQuantity(stocks), "value", money(sumValue(stocks)), "backorder", backorder);
576 }
577 if (!this.stock.size) this.emit(2, "(none)");
578 break;
579 }
580 case "SUMMARY": {
581 this.emit(0, "SUMMARY");
582 for (const category of CATEGORIES) {
583 const total = this.totals.get(category);
584 this.emit(2, category, "qty", total.quantity, "value", money(total.value));
585 }
586 const stocks = this.allStocks();
587 this.emit(2, "on-hand", "qty", sumQuantity(stocks), "value", money(sumValue(stocks)));
588 this.emit(2, "orders", this.orders.length, "open", this.orders.filter(o => o.backorder > 0).length,
589 "backorder", this.orders.reduce((q, o) => q + o.backorder, 0));
590 break;
591 }
592 }
593 }
