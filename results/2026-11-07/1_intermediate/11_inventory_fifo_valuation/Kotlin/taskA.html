1import kotlin.math.abs
2import kotlin.math.min
3
4private val categories = listOf("received","returned","adjusted-in","adjusted-out","shipped","written-down","expired","recalled")
5private class Fault(reason: String) : RuntimeException(reason)
6private fun error(condition: Boolean, reason: String) { if (condition) throw Fault(reason) }
7private fun parseMoney(s: String): Long {
8 return s.replace(".","").toLong()
9}
10private fun money(n: Long) = (if (n < 0) "-" else "")+"${abs(n)/100}.${(abs(n)%100).toString().padStart(2,'0')}"
11private fun signed(n: Long) = (if (n > 0) "+" else "")+n
12private fun signedMoney(n: Long) = (if (n > 0) "+" else "")+money(n)
13private fun margin(profit: Long, revenue: Long): String {
14 if (revenue == 0L) return "n/a"
15 val q = abs(profit)*1000/revenue
16 return (if (profit < 0 && q > 0) "-" else "")+"${q/10}.${q%10}%"
17}
18private data class Card(val day: Long, val kind: String, val warehouse: String, val quantity: Long, val value: Long)
19private class Item(val code: String, var level: Long, var name: String) {
20 var lastCost: Long? = null
21 val cards = mutableListOf<Card>()
22}
23private class Warehouse(val code: String) {
24 val stock = mutableMapOf<String,MutableMap<Int,Stock>>()
25}
26private class Lot(val number: Int, val item: Item, val day: Long, val initialQuantity: Long,
27 val initialCost: Long, val warehouse: String, val kind: String) {
28 var removed = 0L
29}
30private class Stock(val lot: Lot, var quantity: Long, var cost: Long) {
31 fun value() = quantity*cost
32}
33private data class Part(val lot: Lot, val quantity: Long, val cost: Long)
34private data class Issue(val parts: List<Part>, val cost: Long)
35private class Order(val number: Int, val warehouse: Warehouse, val item: Item, val day: Long, val ordered: Long, val price: Long) {
36 var shipped = 0L
37 var backorder = ordered
38 var cancelled = 0L
39 var returned = 0L
40 var revenue = 0L
41 var cogs = 0L
42 val parts = mutableListOf<Part>()
43}
44private class Inventory {
45 var day = 1L
46 val warehouses = sortedMapOf<String,Warehouse>()
47 val items = sortedMapOf<String,Item>()
48 val lots = mutableListOf<Lot>()
49 val orders = mutableListOf<Order>()
50 val totals = Array(8) { LongArray(2) }
51 var output = mutableListOf<String>()
52 fun warehouse(code: String) = warehouses[code] ?: throw Fault("unknown warehouse $code")
53 fun item(code: String) = items[code] ?: throw Fault("unknown item $code")
54 fun order(raw: String): Order {
55 val n = raw.toLong()
56 error(n < 1 || n > orders.size,"unknown order $raw")
57 return orders[n.toInt()-1]
58 }
59 fun lot(raw: String): Lot {
60 val n = raw.toLong()
61 error(n < 1 || n > lots.size,"unknown lot $raw")
62 return lots[n.toInt()-1]
63 }
64 fun stock(w: Warehouse, s: Item) = w.stock[s.code]?.values?.toList() ?: emptyList()
65 fun fifo(w: Warehouse, s: Item) = stock(w,s).sortedWith(compareBy<Stock> { it.lot.day }.thenBy { it.lot.number })
66 fun onhand(w: Warehouse, s: Item) = stock(w,s).sumOf { it.quantity }
67 fun value(w: Warehouse, s: Item) = stock(w,s).sumOf { it.value() }
68 fun onhand(s: Item) = warehouses.values.sumOf { onhand(it,s) }
69 fun backorder(s: Item) = orders.filter { it.item === s }.sumOf { it.backorder }
70 fun target(s: Item, selected: Warehouse?) = warehouses.values.filter { selected == null || it === selected }.flatMap { stock(it,s) }
71 fun card(s: Item, kind: String, w: Warehouse, q: Long, v: Long) { s.cards.add(Card(day,kind,w.code,q,v)) }
72 fun total(category: Int, q: Long, value: Long) {
73 totals[category][0] += q
74 totals[category][1] += value
75 }
76 fun add(w: Warehouse, s: Item, lot: Lot, q: Long, cost: Long) {
77 val map = w.stock.getOrPut(s.code) { mutableMapOf() }
78 val old = map[lot.number]
79 if (old == null) map[lot.number] = Stock(lot,q,cost) else old.quantity += q
80 }
81 fun create(w: Warehouse, s: Item, q: Long, cost: Long, date: Long, kind: String): Lot {
82 val lot = Lot(lots.size+1,s,date,q,cost,w.code,kind)
83 lots.add(lot)
84 add(w,s,lot,q,cost)
85 return lot
86 }
87 fun issue(w: Warehouse, s: Item, quantity: Long, removed: Boolean): Issue {
88 val parts = mutableListOf<Part>()
89 var left = quantity
90 var cost = 0L
91 for (a in fifo(w,s)) {
92 if (left == 0L) break
93 val q = min(left,a.quantity)
94 parts.add(Part(a.lot,q,a.cost))
95 cost += q*a.cost
96 a.quantity -= q
97 left -= q
98 if (removed) a.lot.removed += q
99 if (a.quantity == 0L) w.stock.getValue(s.code).remove(a.lot.number)
100 }
101 return Issue(parts,cost)
102 }
103 fun fill(o: Order, q: Long, kind: String): Issue {
104 val issued = issue(o.warehouse,o.item,q,false)
105 o.shipped += q
106 o.backorder -= q
107 o.revenue += q*o.price
108 o.cogs += issued.cost
109 o.parts.addAll(issued.parts)
110 if (q > 0) {
111 total(4,q,issued.cost)
112 card(o.item,kind,o.warehouse,-q,-issued.cost)
113 }
114 return issued
115 }
116 fun allocate(w: Warehouse, s: Item) {
117 var available = onhand(w,s)
118 for (o in orders) {
119 if (available == 0L) break
120 if (o.warehouse === w && o.item === s && o.backorder > 0) {
121 val q = min(available,o.backorder)
122 val issued = fill(o,q,"fill")
123 available -= q
124 output.add(" FILL order ${o.number} qty $q revenue ${money(q*o.price)} cogs ${money(issued.cost)} profit ${money(q*o.price-issued.cost)} backorder ${o.backorder}")
125 }
126 }
127 }
128 fun insufficient(w: Warehouse, s: Item, q: Long) {
129 val h = onhand(w,s)
130 error(h < q,"insufficient stock ${w.code} ${s.code} $h")
131 }
132 fun sales(s: Item?): LongArray {
133 val selected = orders.filter { s == null || it.item === s }
134 return longArrayOf(selected.sumOf { it.shipped-it.returned },selected.sumOf { it.revenue },
135 selected.sumOf { it.cogs },selected.size.toLong())
136 }
137 fun salesLine(label: String, a: LongArray) = " $label sold ${a[0]} revenue ${money(a[1])} cogs ${money(a[2])} profit ${money(a[1]-a[2])} margin ${margin(a[1]-a[2],a[1])}"
138 fun buckets(label: String, a: LongArray, monetary: Boolean): String {
139 val labels = listOf("0-30","31-60","61-90","91+")
140 return " $label"+(0..3).joinToString("") { " ${labels[it]} "+(if (monetary) money(a[it]) else a[it].toString()) }
141 }
142 fun stockCommand(t: List<String>) {
143 val w = warehouse(t[1])
144 val s = item(t[2])
145 val q = (if (t[0] == "ADJUST") t[3].drop(1) else t[3]).toLong()
146 if (t[0] != "COUNT") error(q == 0L,"invalid quantity")
147 val h = onhand(w,s)
148 when (t[0]) {
149 "RECEIVE" -> {
150 val date = t[5].toLong()
151 val cost = parseMoney(t[4])
152 error(date > day,"future day ${t[5]}")
153 val lot = create(w,s,q,cost,date,"receive")
154 s.lastCost = cost
155 total(0,q,q*cost)
156 card(s,"receive",w,q,q*cost)
157 output.add("RECEIVE ${w.code} ${s.code} lot ${lot.number} qty $q cost ${money(cost)} day $date value ${money(q*cost)}")
158 allocate(w,s)
159 }
160 "SHIP" -> {
161 val o = Order(orders.size+1,w,s,day,q,parseMoney(t[4]))
162 orders.add(o)
163 val f = min(q,h)
164 val issued = fill(o,f,"ship")
165 output.add("SHIP order ${o.number} ${w.code} ${s.code} shipped $f revenue ${money(o.revenue)} cogs ${money(issued.cost)} profit ${money(o.revenue-issued.cost)} backorder ${o.backorder}")
166 }
167 "ADJUST" -> {
168 if (t[3][0] == '+') {
169 val cost = parseMoney(t[4])
170 val lot = create(w,s,q,cost,day,"adjust")
171 total(2,q,q*cost)
172 card(s,"adjust",w,q,q*cost)
173 output.add("ADJUST ${w.code} ${s.code} +$q lot ${lot.number} value ${money(q*cost)}")
174 allocate(w,s)
175 } else {
176 insufficient(w,s,q)
177 val issued = issue(w,s,q,true)
178 total(3,q,issued.cost)
179 card(s,"adjust",w,-q,-issued.cost)
180 output.add("ADJUST ${w.code} ${s.code} -$q value ${money(issued.cost)}")
181 }
182 }
183 "COUNT" -> {
184 if (q == h) output.add("COUNT ${w.code} ${s.code} qty $q ok")
185 else if (q < h) {
186 val issued = issue(w,s,h-q,true)
187 total(3,h-q,issued.cost)
188 card(s,"count",w,q-h,-issued.cost)
189 output.add("COUNT ${w.code} ${s.code} qty $h -> $q shortage ${h-q} value ${money(issued.cost)}")
190 } else {
191 val cost = s.lastCost ?: 0L
192 val lot = create(w,s,q-h,cost,day,"count")
193 total(2,q-h,(q-h)*cost)
194 card(s,"count",w,q-h,(q-h)*cost)
195 output.add("COUNT ${w.code} ${s.code} qty $h -> $q surplus ${q-h} lot ${lot.number} value ${money((q-h)*cost)}")
196 allocate(w,s)
197 }
198 }
199 }
200 }
201 fun execute(t: List<String>) {
202 when (t[0]) {
203 "DAY" -> {
204 val d = t[1].toLong()
205 error(d <= day,"invalid day ${t[1]}")
206 day = d
207 output.add("DAY $day")
208 }
209 "WAREHOUSE" -> {
210 error(t[1] in warehouses,"duplicate warehouse ${t[1]}")
211 warehouses[t[1]] = Warehouse(t[1])
212 output.add("WAREHOUSE ${t[1]}")
213 }
214 "ITEM" -> {
215 val existing = items[t[1]]
216 val s = existing ?: Item(t[1],t[2].toLong(),t[3])
217 s.level = t[2].toLong()
218 s.name = t[3]
219 items[s.code] = s
220 output.add("ITEM ${s.code}"+(if (existing == null) " added " else " updated ")+"${s.name} level ${s.level}")
221 }
222 "RECEIVE", "SHIP", "ADJUST", "COUNT" -> stockCommand(t)
223 "CANCEL" -> {
224 val o = order(t[1])
225 error(o.backorder == 0L,"no backorder ${t[1]}")
226 o.cancelled += o.backorder
227 o.backorder = 0
228 output.add("CANCEL order ${o.number} qty ${o.cancelled}")
229 }
230 "TRANSFER" -> {
231 val a = warehouse(t[1])
232 val b = warehouse(t[2])
233 error(a === b,"same warehouse ${t[1]}")
234 val s = item(t[3])
235 val q = t[4].toLong()
236 error(q == 0L,"invalid quantity")
237 insufficient(a,s,q)
238 val issued = issue(a,s,q,false)
239 for (p in issued.parts) add(b,s,p.lot,p.quantity,p.cost)
240 card(s,"transfer-out",a,-q,-issued.cost)
241 card(s,"transfer-in",b,q,issued.cost)
242 output.add("TRANSFER ${a.code} -> ${b.code} ${s.code} qty $q value ${money(issued.cost)} lots ${issued.parts.size}")
243 allocate(b,s)
244 }
245 "RETURN" -> {
246 val o = order(t[1])
247 val q = t[2].toLong()
248 val cost = parseMoney(t[3])
249 error(q == 0L,"invalid quantity")
250 error(q > o.shipped-o.returned,"return exceeds shipped ${t[1]} ${o.shipped-o.returned}")
251 val lot = create(o.warehouse,o.item,q,cost,day,"return")
252 o.returned += q
253 o.revenue -= q*o.price
254 o.cogs -= q*cost
255 total(1,q,q*cost)
256 card(o.item,"return",o.warehouse,q,q*cost)
257 output.add("RETURN order ${o.number} ${o.warehouse.code} ${o.item.code} qty $q lot ${lot.number} refund ${money(q*o.price)} value ${money(q*cost)}")
258 allocate(o.warehouse,o.item)
259 }
260 "MARKET" -> {
261 val s = item(t[1])
262 val price = parseMoney(t[2])
263 var quantity = 0L
264 var loss = 0L
265 var count = 0
266 for (w in warehouses.values) {
267 var subtotal = 0L
268 for (a in stock(w,s)) {
269 if (a.cost > price) {
270 subtotal += a.quantity*(a.cost-price)
271 quantity += a.quantity
272 count++
273 a.cost = price
274 }
275 }
276 if (subtotal > 0) card(s,"writedown",w,0,-subtotal)
277 loss += subtotal
278 }
279 total(5,quantity,loss)
280 output.add("MARKET ${s.code} price ${money(price)} lots $count qty $quantity loss ${money(loss)}")
281 }
282 "RECALL" -> {
283 val lot = lot(t[1])
284 val held = warehouses.values.mapNotNull { w -> stock(w,lot.item).firstOrNull { it.lot === lot }?.let { w to it } }
285 error(held.isEmpty(),"nothing to recall ${t[1]}")
286 val quantity = held.sumOf { it.second.quantity }
287 val value = held.sumOf { it.second.value() }
288 for ((w,a) in held) {
289 card(lot.item,"recall",w,-a.quantity,-a.value())
290 w.stock.getValue(lot.item.code).remove(lot.number)
291 }
292 lot.removed += quantity
293 total(7,quantity,value)
294 output.add("RECALL lot ${lot.number} ${lot.item.code} qty $quantity value ${money(value)}")
295 for ((w,a) in held) output.add(" ${w.code} qty ${a.quantity} value ${money(a.value())}")
296 }
297 "EXPIRE" -> {
298 val age = t[1].toLong()
299 var quantity = 0L
300 var value = 0L
301 output.add("EXPIRE $age")
302 for (s in items.values) {
303 for (w in warehouses.values) {
304 val expired = stock(w,s).filter { day-it.lot.day > age }
305 if (expired.isEmpty()) continue
306 val q = expired.sumOf { it.quantity }
307 val v = expired.sumOf { it.value() }
308 for (a in expired) {
309 a.lot.removed += a.quantity
310 w.stock.getValue(s.code).remove(a.lot.number)
311 }
312 total(6,q,v)
313 card(s,"expire",w,-q,-v)
314 output.add(" ${s.code} ${w.code} lots ${expired.size} qty $q value ${money(v)}")
315 quantity += q
316 value += v
317 }
318 }
319 output.add(" total qty $quantity value ${money(value)}")
320 }
321 "VALUE", "AGING" -> {
322 val selected = if (t.size == 2) warehouse(t[1]) else null
323 if (t[0] == "VALUE") {
324 output.add("VALUE"+(if (selected == null) "" else " ${selected.code}"))
325 var quantity = 0L
326 var value = 0L
327 for (s in items.values) {
328 val target = target(s,selected)
329 val q = target.sumOf { it.quantity }
330 val v = target.sumOf { it.value() }
331 if (q == 0L) continue
332 output.add(" ${s.code} qty $q value ${money(v)} avg ${money(v/q)}")
333 if (selected == null) {
334 for (w in warehouses.values) if (onhand(w,s) > 0) output.add(" ${w.code} qty ${onhand(w,s)} value ${money(value(w,s))}")
335 }
336 quantity += q
337 value += v
338 }
339 output.add(" total qty $quantity value ${money(value)}")
340 } else {
341 output.add("AGING"+(if (selected == null) "" else " ${selected.code}")+" day $day")
342 val quantities = LongArray(4)
343 val values = LongArray(4)
344 for (s in items.values) {
345 val q = LongArray(4)
346 for (a in target(s,selected)) {
347 val age = day-a.lot.day
348 val b = if (age <= 30) 0 else if (age <= 60) 1 else if (age <= 90) 2 else 3
349 q[b] += a.quantity
350 quantities[b] += a.quantity
351 values[b] += a.value()
352 }
353 if (q.sum() > 0) output.add(buckets(s.code,q,false))
354 }
355 output.add(buckets("total",quantities,false))
356 output.add(buckets("value",values,true))
357 }
358 }
359 "LOTS" -> {
360 val w = warehouse(t[1])
361 val s = item(t[2])
362 output.add("LOTS ${w.code} ${s.code}")
363 for (a in fifo(w,s)) output.add(" lot ${a.lot.number} day ${a.lot.day} qty ${a.quantity} cost ${money(a.cost)} value ${money(a.value())} age ${day-a.lot.day}")
364 for (o in orders) if (o.warehouse === w && o.item === s && o.backorder > 0) output.add(" backorder order ${o.number} qty ${o.backorder}")
365 }
366 "REORDER", "ITEMS" -> {
367 output.add(t[0])
368 for (s in items.values) {
369 val h = onhand(s)
370 val b = backorder(s)
371 if (t[0] == "ITEMS") output.add(" ${s.code} ${s.name} level ${s.level} onhand $h backorder $b cost "+(s.lastCost?.let { money(it) } ?: "none"))
372 else if (h-b <= s.level) output.add(" ${s.code} ${s.name} onhand $h backorder $b level ${s.level}")
373 }
374 }
375 "MARGIN", "TOP" -> {
376 val summaries = items.values.associateWith { sales(it) }.filterValues { it[3] > 0 }
377 if (t[0] == "MARGIN") {
378 output.add("MARGIN")
379 for (s in items.values) if (s in summaries) output.add(salesLine(s.code,summaries.getValue(s)))
380 output.add(salesLine("total",sales(null)))
381 } else {
382 val k = t[1].toInt()
383 output.add("TOP $k")
384 val ranked = summaries.keys.sortedWith(compareByDescending<Item> { summaries.getValue(it)[1]-summaries.getValue(it)[2] }
385 .thenByDescending { summaries.getValue(it)[1] }.thenBy { it.code }).take(k)
386 for ((i,s) in ranked.withIndex()) {
387 val a = summaries.getValue(s)
388 output.add(" ${i+1} ${s.code} profit ${money(a[1]-a[2])} revenue ${money(a[1])} margin ${margin(a[1]-a[2],a[1])}")
389 }
390 }
391 }
392 "BACKORDERS" -> {
393 output.add("BACKORDERS")
394 for (o in orders) if (o.backorder > 0) output.add(" order ${o.number} ${o.warehouse.code} ${o.item.code} qty ${o.backorder} day ${o.day} price ${money(o.price)}")
395 }
396 "ORDER" -> {
397 val o = order(t[1])
398 output.add("ORDER ${o.number} ${o.warehouse.code} ${o.item.code} day ${o.day} price ${money(o.price)}")
399 output.add(" ordered ${o.ordered} shipped ${o.shipped} backorder ${o.backorder} cancelled ${o.cancelled} returned ${o.returned}")
400 for (p in o.parts) output.add(" lot ${p.lot.number} qty ${p.quantity} cost ${money(p.cost)}")
401 output.add(" revenue ${money(o.revenue)} cogs ${money(o.cogs)} profit ${money(o.revenue-o.cogs)}")
402 }
403 "TRACE" -> {
404 val lot = lot(t[1])
405 output.add("TRACE lot ${lot.number} ${lot.item.code} ${lot.kind} ${lot.warehouse} day ${lot.day} qty ${lot.initialQuantity} cost ${money(lot.initialCost)}")
406 for (w in warehouses.values) for (a in stock(w,lot.item)) if (a.lot === lot) output.add(" held ${w.code} qty ${a.quantity} cost ${money(a.cost)}")
407 for (o in orders) {
408 val q = o.parts.filter { it.lot === lot }.sumOf { it.quantity }
409 if (q > 0) output.add(" order ${o.number} qty $q")
410 }
411 if (lot.removed > 0) output.add(" removed qty ${lot.removed}")
412 }
413 "CARD" -> {
414 val s = item(t[1])
415 output.add("CARD ${s.code} ${s.name}")
416 var quantity = 0L
417 var value = 0L
418 for (c in s.cards) {
419 quantity += c.quantity
420 value += c.value
421 output.add(" day ${c.day} ${c.kind} ${c.warehouse} ${signed(c.quantity)} ${signedMoney(c.value)} balance $quantity ${money(value)}")
422 }
423 }
424 "STATUS" -> {
425 output.add("STATUS")
426 for (w in warehouses.values) {
427 val q = items.values.sumOf { onhand(w,it) }
428 val v = items.values.sumOf { value(w,it) }
429 val skus = items.values.count { onhand(w,it) > 0 }
430 val b = orders.filter { it.warehouse === w }.sumOf { it.backorder }
431 output.add(" ${w.code} skus $skus qty $q value ${money(v)} backorder $b")
432 }
433 }
434 "SUMMARY" -> {
435 output.add("SUMMARY")
436 for (i in 0..7) output.add(" ${categories[i]} qty ${totals[i][0]} value ${money(totals[i][1])}")
437 val q = items.values.sumOf { onhand(it) }
438 val v = items.values.sumOf { s -> warehouses.values.sumOf { value(it,s) } }
439 output.add(" on-hand qty $q value ${money(v)}")
440 output.add(" orders ${orders.size} open ${orders.count { it.backorder > 0 }} backorder ${orders.sumOf { it.backorder }}")
441 }
442 }
443 if (output.size == 1 && t[0] in listOf("LOTS","REORDER","ITEMS","TOP","BACKORDERS","CARD","STATUS")) output.add(" (none)")
444 }
445 fun process(line: String): List<String> {
446 output = mutableListOf()
447 try { execute(line.split(' ')) }
448 catch (f: Fault) { output.add("ERROR ${f.message}") }
449 return output
450 }
451}
452fun main() {
453 val input = System.`in`.bufferedReader()
454 val output = System.out.bufferedWriter()
455 val inventory = Inventory()
456 repeat(input.readLine().toInt()) {
457 for (line in inventory.process(input.readLine())) {
458 output.write(line)
459 output.newLine()
460 }
461 }
462 output.flush()
463}
