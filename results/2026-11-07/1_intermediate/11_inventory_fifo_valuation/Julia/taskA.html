6cents(text::String) = parse(Int, replace(text, "." => ""))
7
8function money(amount::Int)
9 magnitude = abs(amount)
10 return (amount < 0 ? "-" : "") * string(div(magnitude, 100)) *
11 "." * lpad(string(mod(magnitude, 100)), 2, '0')
12end
13
14signed_quantity(quantity::Int) = (quantity > 0 ? "+" : "") * string(quantity)
15signed_money(amount::Int) = (amount > 0 ? "+" : "") * money(amount)
16
17function margin(profit::Int, revenue::Int)
18 revenue == 0 && return "n/a"
19 tenths = div(abs(profit) * 1000, revenue)
20 return (profit < 0 && tenths > 0 ? "-" : "") *
21 string(div(tenths, 10)) * "." * string(mod(tenths, 10)) * "%"
22end

184function accumulate!(inv::Inventory, category::String, q::Int, value::Int)

190function add_stock!(inv::Inventory, w::String, lot::Lot, q::Int, cost::Int)
191 bucket = get!(inv.stock[w], lot.item) do
192 Dict{Int, StockLot}()
193 end
194 if haskey(bucket, lot.number)
195 bucket[lot.number].quantity += q
196 else
197 bucket[lot.number] = StockLot(lot, q, cost)
198 end
199end

201function create_lot!(inv::Inventory, w::String, code::String,
202 day::Int, q::Int, cost::Int, kind::String)
203 lot = Lot(length(inv.lots) + 1, code, w, kind, day, q, cost, 0)
204 push!(inv.lots, lot)
205 add_stock!(inv, w, lot, q, cost)
206 return lot
207end
208
209function issue!(inv::Inventory, w::String, code::String, q::Int)
210 parts = Part[]
211 cost = 0
212 for stock in fifo(inv, w, code)
213 q == 0 && break
214 taken = min(q, stock.quantity)
215 push!(parts, Part(stock.lot, taken, stock.cost))
216 cost += taken * stock.cost
217 stock.quantity -= taken
218 q -= taken
219 if stock.quantity == 0
220 delete!(inv.stock[w][code], stock.lot.number)
221 end
222 end
223 return parts, cost
224end
225
226function fulfil!(inv::Inventory, order::Order, q::Int, kind::String)
227 q == 0 && return 0
228 parts, cost = issue!(inv, order.warehouse, order.item, q)
229 order.shipped += q
230 order.backorder -= q
231 order.revenue += q * order.price
232 order.cogs += cost
233 append!(order.issues, parts)
234 accumulate!(inv, "shipped", q, cost)
235 card_entry!(inv, order.item, kind, order.warehouse, -q, -cost)
236 return cost
237end
238
239function allocate!(inv::Inventory, w::String, code::String)
240 available = quantity(inv, w, code)
241 for order in inv.orders
242 available == 0 && break
243 if order.warehouse != w || order.item != code || order.backorder == 0
244 continue
245 end
246 q = min(available, order.backorder)
247 cost = fulfil!(inv, order, q, "fill")
248 revenue = q * order.price
249 available -= q
250 emit!(inv, 2, "FILL", "order", order.number, "qty", q, "revenue",
251 money(revenue), "cogs", money(cost), "profit", money(revenue - cost),
252 "backorder", order.backorder)
253 end
254end
255
256function remove_parts!(parts::Vector{Part})
257 for part in parts
258 part.lot.removed += part.quantity
259 end
260end
261
262function sales(inv::Inventory)
263 rows = Dict{String, Total}()
264 cogs = Dict{String, Int}()
265 for order in inv.orders
266 row = get!(rows, order.item) do
267 Total(0, 0)
268 end
269 row.quantity += order.shipped - order.returned
270 row.value += order.revenue
271 cogs[order.item] = get(cogs, order.item, 0) + order.cogs
272 end
273 return rows, cogs
274end
275
276function margin_line!(inv::Inventory, label::String, sold::Int, revenue::Int, cogs::Int)
277 profit = revenue - cogs
278 emit!(inv, 2, label, "sold", sold, "revenue", money(revenue), "cogs",
279 money(cogs), "profit", money(profit), "margin", margin(profit, revenue))
280end
281
282function aging_line!(inv::Inventory, label::String, buckets, monetary::Bool)
283 parts = Any[label]
284 for (index, name) in enumerate(("0-30", "31-60", "61-90", "91+"))
285 push!(parts, name, monetary ? money(buckets[index]) : buckets[index])
286 end
287 emit!(inv, 2, parts...)
288end
289
290function command!(inv::Inventory, cmd::String, a::Vector{String})
291 if cmd == "DAY"
292 day = parse(Int, a[1])
293 day > inv.day || fail("invalid day " * a[1])
294 inv.day = day
295 emit!(inv, 0, "DAY", day)
296
297 elseif cmd == "WAREHOUSE"
298 w = a[1]
299 !haskey(inv.stock, w) || fail("duplicate warehouse " * w)
300 inv.stock[w] = Dict{String, Dict{Int, StockLot}}()
301 emit!(inv, 0, "WAREHOUSE", w)
302
303 elseif cmd == "ITEM"
304 code, level, name = a[1], parse(Int, a[2]), a[3]
305 action = "added"
306 if haskey(inv.items, code)
307 entry = inv.items[code]
308 entry.level, entry.name = level, name
309 action = "updated"
310 else
311 inv.items[code] = Item(code, name, level, nothing, CardEntry[])
312 end
313 emit!(inv, 0, "ITEM", code, action, name, "level", level)
314
315 elseif cmd == "RECEIVE"
316 w, code = a[1], a[2]
317 entry = pair(inv, w, code)
318 q, cost, day = parse(Int, a[3]), cents(a[4]), parse(Int, a[5])
319 require_quantity(q)
320 day <= inv.day || fail("future day " * a[5])
321 lot = create_lot!(inv, w, code, day, q, cost, "receive")
322 entry.last_cost = cost
323 accumulate!(inv, "received", q, q * cost)
324 card_entry!(inv, code, "receive", w, q, q * cost)
325 emit!(inv, 0, "RECEIVE", w, code, "lot", lot.number, "qty", q,
326 "cost", money(cost), "day", day, "value", money(q * cost))
327 allocate!(inv, w, code)
328
329 elseif cmd == "SHIP"
330 w, code = a[1], a[2]
331 pair(inv, w, code)
332 q = parse(Int, a[3])
333 require_quantity(q)
334 order = Order(length(inv.orders) + 1, w, code, inv.day, cents(a[4]), q,
335 0, q, 0, 0, 0, 0, Part[])
336 push!(inv.orders, order)
337 shipped = min(q, quantity(inv, w, code))
338 cost = fulfil!(inv, order, shipped, "ship")
339 emit!(inv, 0, "SHIP", "order", order.number, w, code, "shipped", shipped,
340 "revenue", money(order.revenue), "cogs", money(cost),
341 "profit", money(order.revenue - cost), "backorder", order.backorder)
342
343 elseif cmd == "CANCEL"
344 order = lookup_order(inv, a[1])
345 order.backorder > 0 || fail("no backorder " * a[1])
346 order.cancelled += order.backorder
347 order.backorder = 0
348 emit!(inv, 0, "CANCEL", "order", order.number, "qty", order.cancelled)
349
350 elseif cmd == "TRANSFER"
351 source, destination, code = a[1], a[2], a[3]
352 warehouse(inv, source)
353 warehouse(inv, destination)
354 source != destination || fail("same warehouse " * source)
355 item(inv, code)
356 q = parse(Int, a[4])
357 require_quantity(q)
358 require_stock(inv, source, code, q)
359 parts, cost = issue!(inv, source, code, q)
360 for part in parts
361 add_stock!(inv, destination, part.lot, part.quantity, part.cost)
362 end
363 card_entry!(inv, code, "transfer-out", source, -q, -cost)
364 card_entry!(inv, code, "transfer-in", destination, q, cost)
365 emit!(inv, 0, "TRANSFER", source, "->", destination, code, "qty",
366 q, "value", money(cost), "lots", length(parts))
367 allocate!(inv, destination, code)
368
369 elseif cmd == "ADJUST"
370 w, code, change = a[1], a[2], a[3]
371 pair(inv, w, code)
372 q = parse(Int, change[2:end])
373 require_quantity(q)
374 if startswith(change, "+")
375 cost = cents(a[4])
376 lot = create_lot!(inv, w, code, inv.day, q, cost, "adjust")
377 accumulate!(inv, "adjusted-in", q, q * cost)
378 card_entry!(inv, code, "adjust", w, q, q * cost)
379 emit!(inv, 0, "ADJUST", w, code, "+" * string(q),
380 "lot", lot.number, "value", money(q * cost))
381 allocate!(inv, w, code)
382 else
383 require_stock(inv, w, code, q)
384 parts, cost = issue!(inv, w, code, q)
385 remove_parts!(parts)
386 accumulate!(inv, "adjusted-out", q, cost)
387 card_entry!(inv, code, "adjust", w, -q, -cost)
388 emit!(inv, 0, "ADJUST", w, code, "-" * string(q), "value", money(cost))
389 end
390
391 elseif cmd == "RETURN"
392 order = lookup_order(inv, a[1])
393 q, cost = parse(Int, a[2]), cents(a[3])
394 require_quantity(q)
395 available = order.shipped - order.returned
396 q <= available || fail("return exceeds shipped " * a[1] * " " * string(available))
397 lot = create_lot!(inv, order.warehouse, order.item, inv.day, q, cost, "return")
398 refund = q * order.price
399 order.returned += q
400 order.revenue -= refund
401 order.cogs -= q * cost
402 accumulate!(inv, "returned", q, q * cost)
403 card_entry!(inv, order.item, "return", order.warehouse, q, q * cost)
404 emit!(inv, 0, "RETURN", "order", order.number, order.warehouse, order.item,
405 "qty", q, "lot", lot.number, "refund", money(refund), "value", money(q * cost))
406 allocate!(inv, order.warehouse, order.item)
407
408 elseif cmd == "COUNT"
409 w, code = a[1], a[2]
410 entry = pair(inv, w, code)
411 actual, held = parse(Int, a[3]), quantity(inv, w, code)
412 if actual == held
413 emit!(inv, 0, "COUNT", w, code, "qty", actual, "ok")
414 elseif actual < held
415 q = held - actual
416 parts, cost = issue!(inv, w, code, q)
417 remove_parts!(parts)
418 accumulate!(inv, "adjusted-out", q, cost)
419 card_entry!(inv, code, "count", w, -q, -cost)
420 emit!(inv, 0, "COUNT", w, code, "qty", held, "->", actual,
421 "shortage", q, "value", money(cost))
422 else
423 q = actual - held
424 cost = isnothing(entry.last_cost) ? 0 : entry.last_cost
425 lot = create_lot!(inv, w, code, inv.day, q, cost, "count")
426 accumulate!(inv, "adjusted-in", q, q * cost)
427 card_entry!(inv, code, "count", w, q, q * cost)
428 emit!(inv, 0, "COUNT", w, code, "qty", held, "->", actual,
429 "surplus", q, "lot", lot.number, "value", money(q * cost))
430 allocate!(inv, w, code)
431 end
432
433 elseif cmd == "MARKET"
434 item(inv, a[1])
435 price = cents(a[2])
436 changed, q, loss = 0, 0, 0
437 for w in ordered_names(inv.stock)
438 warehouse_loss = 0
439 for stock in values(holdings(inv, w, a[1]))
440 if stock.cost > price
441 warehouse_loss += stock.quantity * (stock.cost - price)
442 changed += 1
443 q += stock.quantity
444 stock.cost = price
445 end
446 end
447 if warehouse_loss > 0
448 card_entry!(inv, a[1], "writedown", w, 0, -warehouse_loss)
449 end
450 loss += warehouse_loss
451 end
452 accumulate!(inv, "written-down", q, loss)
453 emit!(inv, 0, "MARKET", a[1], "price", money(price),
454 "lots", changed, "qty", q, "loss", money(loss))
455
456 elseif cmd == "RECALL"
457 lot = lookup_lot(inv, a[1])
458 held = Tuple{String, StockLot}[]
459 for w in ordered_names(inv.stock)
460 stock = get(holdings(inv, w, lot.item), lot.number, nothing)
461 if !isnothing(stock)
462 push!(held, (w, stock))
463 end
464 end
465 isempty(held) && fail("nothing to recall " * a[1])
466 q, value = 0, 0
467 for (w, stock) in held
468 q += stock.quantity
469 value += stock.quantity * stock.cost
470 delete!(inv.stock[w][lot.item], lot.number)
471 card_entry!(inv, lot.item, "recall", w, -stock.quantity, -stock.quantity * stock.cost)
472 end
473 lot.removed += q
474 accumulate!(inv, "recalled", q, value)
475 emit!(inv, 0, "RECALL", "lot", lot.number, lot.item, "qty", q, "value", money(value))
476 for (w, stock) in held
477 emit!(inv, 2, w, "qty", stock.quantity, "value", money(stock.quantity * stock.cost))
478 end
479
480 elseif cmd == "EXPIRE"
481 days = parse(Int, a[1])
482 emit!(inv, 0, "EXPIRE", days)
483 total_q, total_value = 0, 0
484 for code in ordered_names(inv.items)
485 for w in ordered_names(inv.stock)
486 expired = filter(stock -> inv.day - stock.lot.day > days, stocks(inv, w, code))
487 isempty(expired) && continue
488 q, value = sum_quantity(expired), sum_value(expired)
489 for stock in expired
490 stock.lot.removed += stock.quantity
491 delete!(inv.stock[w][code], stock.lot.number)
492 end
493 accumulate!(inv, "expired", q, value)
494 card_entry!(inv, code, "expire", w, -q, -value)
495 total_q += q
496 total_value += value
497 emit!(inv, 2, code, w, "lots", length(expired), "qty", q, "value", money(value))
498 end
499 end
500 emit!(inv, 2, "total", "qty", total_q, "value", money(total_value))
501
502 elseif cmd == "VALUE" || cmd == "AGING"
503 warehouses = isempty(a) ? ordered_names(inv.stock) : [warehouse(inv, a[1])]
504 if cmd == "VALUE"
505 emit!(inv, 0, "VALUE", a...)
506 else
507 emit!(inv, 0, "AGING", a..., "day", inv.day)
508 end
509 total_q, total_value = 0, 0
510 bucket_q, bucket_v = zeros(Int, 4), zeros(Int, 4)
511 for code in ordered_names(inv.items)
512 rows = [stocks(inv, w, code) for w in warehouses]
513 selected = StockLot[]
514 for row in rows
515 append!(selected, row)
516 end
517 q = sum_quantity(selected)
518 q == 0 && continue
519 value = sum_value(selected)
520 if cmd == "VALUE"
521 emit!(inv, 2, code, "qty", q, "value", money(value), "avg", money(div(value, q)))
522 if isempty(a)
523 for (w, row) in zip(warehouses, rows)
524 held = sum_quantity(row)
525 if held > 0
526 emit!(inv, 4, w, "qty", held, "value", money(sum_value(row)))
527 end
528 end
529 end
530 else
531 item_buckets = zeros(Int, 4)
532 for stock in selected
533 age = inv.day - stock.lot.day
534 bucket = age <= 30 ? 1 : (age <= 60 ? 2 : (age <= 90 ? 3 : 4))
535 item_buckets[bucket] += stock.quantity
536 bucket_q[bucket] += stock.quantity
537 bucket_v[bucket] += stock.quantity * stock.cost
538 end
539 aging_line!(inv, code, item_buckets, false)
540 end
541 total_q += q
542 total_value += value
543 end
544 if cmd == "VALUE"
545 emit!(inv, 2, "total", "qty", total_q, "value", money(total_value))
546 else
547 aging_line!(inv, "total", bucket_q, false)
548 aging_line!(inv, "value", bucket_v, true)
549 end
550
551 elseif cmd == "LOTS"
552 w, code = a[1], a[2]
553 pair(inv, w, code)
554 emit!(inv, 0, "LOTS", w, code)
555 entries = fifo(inv, w, code)
556 backorders = filter(o -> o.warehouse == w && o.item == code && o.backorder > 0, inv.orders)
557 for stock in entries
558 emit!(inv, 2, "lot", stock.lot.number, "day", stock.lot.day, "qty", stock.quantity,
559 "cost", money(stock.cost), "value", money(stock.quantity * stock.cost),
560 "age", inv.day - stock.lot.day)
561 end
562 for order in backorders
563 emit!(inv, 2, "backorder", "order", order.number, "qty", order.backorder)
564 end
565 if isempty(entries) && isempty(backorders)
566 emit!(inv, 2, "(none)")
567 end
568
569 elseif cmd == "REORDER" || cmd == "ITEMS"
570 emit!(inv, 0, cmd)
571 found = false
572 for code in ordered_names(inv.items)
573 entry = inv.items[code]
574 q, pending = total_quantity(inv, code), backorder(inv, code)
575 if cmd == "REORDER"
576 q - pending > entry.level && continue
577 emit!(inv, 2, code, entry.name, "onhand", q,
578 "backorder", pending, "level", entry.level)
579 else
580 emit!(inv, 2, code, entry.name, "level", entry.level,
581 "onhand", q, "backorder", pending, "cost",
582 isnothing(entry.last_cost) ? "none" : money(entry.last_cost))
583 end
584 found = true
585 end
586 !found && emit!(inv, 2, "(none)")
587
588 elseif cmd == "MARGIN" || cmd == "TOP"
589 rows, cogs = sales(inv)
590 if cmd == "MARGIN"
591 emit!(inv, 0, "MARGIN")
592 total_q, revenue, total_cogs = 0, 0, 0
593 for code in ordered_names(rows)
594 row = rows[code]
595 margin_line!(inv, code, row.quantity, row.value, cogs[code])
596 total_q += row.quantity
597 revenue += row.value
598 total_cogs += cogs[code]
599 end
600 margin_line!(inv, "total", total_q, revenue, total_cogs)
601 else
602 k = parse(Int, a[1])
603 emit!(inv, 0, "TOP", k)
604 codes = ordered_names(rows)
605 sort!(codes, by = code -> (-(rows[code].value - cogs[code]),
606 -rows[code].value, code))
607 for (rank, code) in enumerate(codes[1:min(k, length(codes))])
608 row = rows[code]
609 profit = row.value - cogs[code]
610 emit!(inv, 2, rank, code, "profit", money(profit), "revenue",
611 money(row.value), "margin", margin(profit, row.value))
612 end
613 isempty(codes) && emit!(inv, 2, "(none)")
614 end
615
616 elseif cmd == "BACKORDERS"
617 emit!(inv, 0, "BACKORDERS")
618 open = filter(o -> o.backorder > 0, inv.orders)
619 for order in open
620 emit!(inv, 2, "order", order.number, order.warehouse, order.item, "qty",
621 order.backorder, "day", order.day, "price", money(order.price))
622 end
623 isempty(open) && emit!(inv, 2, "(none)")
624
625 elseif cmd == "ORDER"
626 order = lookup_order(inv, a[1])
627 emit!(inv, 0, "ORDER", order.number, order.warehouse, order.item,
628 "day", order.day, "price", money(order.price))
629 emit!(inv, 2, "ordered", order.ordered, "shipped", order.shipped,
630 "backorder", order.backorder, "cancelled", order.cancelled,
631 "returned", order.returned)
632 for part in order.issues
633 emit!(inv, 2, "lot", part.lot.number, "qty", part.quantity, "cost", money(part.cost))
634 end
635 emit!(inv, 2, "revenue", money(order.revenue), "cogs",
636 money(order.cogs), "profit", money(order.revenue - order.cogs))
637
638 elseif cmd == "TRACE"
639 lot = lookup_lot(inv, a[1])
640 emit!(inv, 0, "TRACE", "lot", lot.number, lot.item, lot.kind, lot.warehouse,
641 "day", lot.day, "qty", lot.initial_quantity, "cost", money(lot.initial_cost))
642 for w in ordered_names(inv.stock)
643 stock = get(holdings(inv, w, lot.item), lot.number, nothing)
644 if !isnothing(stock)
645 emit!(inv, 2, "held", w, "qty", stock.quantity, "cost", money(stock.cost))
646 end
647 end
648 for order in inv.orders
649 q = sum((part.quantity for part in order.issues if part.lot.number == lot.number); init=0)
650 q > 0 && emit!(inv, 2, "order", order.number, "qty", q)
651 end
652 lot.removed > 0 && emit!(inv, 2, "removed", "qty", lot.removed)
653
654 elseif cmd == "CARD"
655 entry = item(inv, a[1])
656 emit!(inv, 0, "CARD", entry.code, entry.name)
657 q, value = 0, 0
658 for card in entry.card
659 q += card.quantity
660 value += card.value
661 emit!(inv, 2, "day", card.day, card.kind, card.warehouse,
662 signed_quantity(card.quantity), signed_money(card.value),
663 "balance", q, money(value))
664 end
665 isempty(entry.card) && emit!(inv, 2, "(none)")
666
667 elseif cmd == "STATUS"
668 emit!(inv, 0, "STATUS")
669 for w in ordered_names(inv.stock)
670 current = inv.stock[w]
671 skus = count(x -> !isempty(x), values(current))
672 selected = StockLot[]
673 for holdings in values(current)
674 append!(selected, values(holdings))
675 end
676 pending = sum((o.backorder for o in inv.orders if o.warehouse == w); init=0)
677 emit!(inv, 2, w, "skus", skus, "qty", sum_quantity(selected),
678 "value", money(sum_value(selected)), "backorder", pending)
679 end
680 isempty(inv.stock) && emit!(inv, 2, "(none)")
681
682 elseif cmd == "SUMMARY"
683 emit!(inv, 0, "SUMMARY")
684 for category in CATEGORIES
685 total = inv.totals[category]
686 emit!(inv, 2, category, "qty", total.quantity, "value", money(total.value))
687 end
688 selected = all_stocks(inv)
689 emit!(inv, 2, "on-hand", "qty", sum_quantity(selected), "value", money(sum_value(selected)))
690 open = count(o -> o.backorder > 0, inv.orders)
691 pending = sum((o.backorder for o in inv.orders); init=0)
692 emit!(inv, 2, "orders", length(inv.orders), "open", open, "backorder", pending)
693 end
694end

696function execute!(inv::Inventory, line::String)
697 tokens = String.(split(line, ' '))
698 cmd, args = tokens[1], tokens[2:end]
699 try
700 command!(inv, cmd, args)
701 catch error
702 if error isa CommandError
703 emit!(inv, 0, "ERROR", error.reason)
704 else
705 rethrow()
706 end
707 end
708end
709
710n = parse(Int, readline())
711inventory = Inventory()
712for _ in 1:n
713 execute!(inventory, readline())
714end
715println(join(inventory.output, "\n"))
