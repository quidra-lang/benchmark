1const std = @import("std");
2const Allocator = std.mem.Allocator;
3const Error = error{BadCommand} || Allocator.Error || std.Io.Writer.Error;
4const categories = [_][]const u8{ "received", "returned", "adjusted-in", "adjusted-out", "shipped", "written-down", "expired", "recalled" };
5fn integer(s: []const u8) i64 {
6 return std.fmt.parseInt(i64, s, 10) catch unreachable;
7}
8fn amount(s: []const u8) i64 {
9 const dot = std.mem.findScalar(u8, s, '.').?;
10 return integer(s[0..dot]) * 100 + integer(s[dot + 1 ..]);
11}
12const Warehouse = struct {
13 code: []const u8,
14 index: usize,
15};
16const CardEntry = struct {
17 day: i64,
18 kind: []const u8,
19 warehouse: *Warehouse,
20 quantity: i64,
21 value: i64,
22};
23const Item = struct {
24 code: []const u8,
25 name: []const u8,
26 level: i64,
27 last_cost: ?i64 = null,
28 order_count: usize = 0,
29 sold: i64 = 0,
30 revenue: i64 = 0,
31 cogs: i64 = 0,
32 lots: std.ArrayList(*Lot) = .empty,
33 card: std.ArrayList(CardEntry) = .empty,
34};
35const Lot = struct {
36 number: usize,
37 item: *Item,
38 day: i64,
39 initial_quantity: i64,
40 initial_cost: i64,
41 warehouse: *Warehouse,
42 kind: []const u8,
43 cost: i64,
44 removed: i64 = 0,
45 held: [20]i64 = @splat(0),
46};
47const Part = struct {
48 lot: *Lot,
49 quantity: i64,
50 cost: i64,
51};

53 parts: []Part,
54 cost: i64,
55};
56const Order = struct {
57 number: usize,
58 warehouse: *Warehouse,
59 item: *Item,
60 day: i64,
61 price: i64,
62 ordered: i64,
63 shipped: i64 = 0,
64 backorder: i64,
65 cancelled: i64 = 0,
66 returned: i64 = 0,
67 revenue: i64 = 0,
68 cogs: i64 = 0,
69 issues: std.ArrayList(Part) = .empty,
70};
71const Stock = struct {
72 quantity: i64 = 0,
73 value: i64 = 0,
74};
75fn held(lot: *Lot, w: ?*Warehouse) i64 {
76 if (w) |warehouse| return lot.held[warehouse.index];
77 var total: i64 = 0;
78 for (lot.held) |q| total += q;
79 return total;
80}
81fn stock(item: *Item, w: ?*Warehouse) Stock {
82 var result: Stock = .{};
83 for (item.lots.items) |lot| {
84 const q = held(lot, w);
85 result.quantity += q;
86 result.value += q * lot.cost;
87 }
88 return result;
89}
90const Inventory = struct {
91 gpa: Allocator,
92 out: *std.Io.Writer,
93 error_message: []const u8 = "",
94 day: i64 = 1,
95 warehouses: std.StringArrayHashMapUnmanaged(*Warehouse) = .empty,
96 items: std.StringArrayHashMapUnmanaged(*Item) = .empty,
97 lots: std.ArrayList(*Lot) = .empty,
98 orders: std.ArrayList(*Order) = .empty,
99 quantities: [8]i64 = @splat(0),
100 values: [8]i64 = @splat(0),
101 fn fail(self: *Inventory, comptime fmt: []const u8, args: anytype) Error {
102 self.error_message = std.fmt.allocPrint(self.gpa, fmt, args) catch return error.OutOfMemory;
103 return error.BadCommand;
104 }
105 fn money(self: *Inventory, c: i64) Allocator.Error![]const u8 {
106 const u: u64 = @intCast(if (c < 0) -c else c);
107 return std.fmt.allocPrint(self.gpa, "{s}{d}.{d:0>2}", .{ if (c < 0) "-" else "", u / 100, u % 100 });
108 }
109 fn margin(self: *Inventory, profit: i64, revenue: i64) Allocator.Error![]const u8 {
110 if (revenue == 0) return "n/a";
111 const q = @divTrunc(@as(i64, @intCast(@abs(profit))) * 1000, revenue);
112 return std.fmt.allocPrint(self.gpa, "{s}{d}.{d}%", .{ if (profit < 0 and q > 0) "-" else "", @divTrunc(q, 10), @mod(q, 10) });
113 }
114 fn warehouse(self: *Inventory, code: []const u8) Error!*Warehouse {
115 return self.warehouses.get(code) orelse return self.fail("unknown warehouse {s}", .{code});
116 }
117 fn item(self: *Inventory, code: []const u8) Error!*Item {
118 return self.items.get(code) orelse return self.fail("unknown item {s}", .{code});
119 }
120 fn order(self: *Inventory, token: []const u8) Error!*Order {
121 const n = integer(token);
122 if (n < 1 or n > @as(i64, @intCast(self.orders.items.len))) return self.fail("unknown order {s}", .{token});
123 return self.orders.items[@intCast(n - 1)];
124 }
125 fn lot(self: *Inventory, token: []const u8) Error!*Lot {
126 const n = integer(token);
127 if (n < 1 or n > @as(i64, @intCast(self.lots.items.len))) return self.fail("unknown lot {s}", .{token});
128 return self.lots.items[@intCast(n - 1)];
129 }
130 fn warehouseList(self: *Inventory) Error![]*Warehouse {
131 const result = try self.gpa.dupe(*Warehouse, self.warehouses.values());
132 std.mem.sort(*Warehouse, result, {}, struct {
133 fn less(_: void, left: *Warehouse, right: *Warehouse) bool {
134 return std.mem.order(u8, left.code, right.code) == .lt;
135 }
136 }.less);
137 return result;
138 }
139 fn itemList(self: *Inventory) Error![]*Item {
140 const result = try self.gpa.dupe(*Item, self.items.values());
141 std.mem.sort(*Item, result, {}, struct {
142 fn less(_: void, left: *Item, right: *Item) bool {
143 return std.mem.order(u8, left.code, right.code) == .lt;
144 }
145 }.less);
146 return result;
147 }
148 fn backorder(self: *Inventory, item_ptr: *Item, w: ?*Warehouse) i64 {
149 var result: i64 = 0;
150 for (self.orders.items) |o| {
151 if (o.item == item_ptr and (w == null or o.warehouse == w.?)) result += o.backorder;
152 }
153 return result;
154 }
155 fn tally(self: *Inventory, category: usize, quantity: i64, value: i64) void {
156 self.quantities[category] += quantity;
157 self.values[category] += value;
158 }
159 fn card(self: *Inventory, item_ptr: *Item, kind: []const u8, w: *Warehouse, quantity: i64, value: i64) Error!void {
160 try item_ptr.card.append(self.gpa, .{ .day = self.day, .kind = kind, .warehouse = w, .quantity = quantity, .value = value });
161 }
162 fn create(self: *Inventory, item_ptr: *Item, w: *Warehouse, q: i64, cost: i64, day: i64, kind: []const u8) Error!*Lot {
163 const l = try self.gpa.create(Lot);
164 l.* = .{ .number = self.lots.items.len + 1, .item = item_ptr, .day = day, .initial_quantity = q, .initial_cost = cost, .warehouse = w, .kind = kind, .cost = cost };
165 l.held[w.index] = q;
166 try self.lots.append(self.gpa, l);
167 try item_ptr.lots.append(self.gpa, l);
168 std.mem.sort(*Lot, item_ptr.lots.items, {}, struct {
169 fn less(_: void, left: *Lot, right: *Lot) bool {
170 return if (left.day != right.day) left.day < right.day else left.number < right.number;
171 }
172 }.less);
173 return l;
174 }
175 fn issue(self: *Inventory, item_ptr: *Item, w: *Warehouse, quantity: i64, removed: bool) Error!Issue {
176 var remaining = quantity;
177 var cost: i64 = 0;
178 var parts: std.ArrayList(Part) = .empty;
179 for (item_ptr.lots.items) |l| {
180 const take = @min(remaining, l.held[w.index]);
181 if (take == 0) continue;
182 try parts.append(self.gpa, .{ .lot = l, .quantity = take, .cost = l.cost });
183 l.held[w.index] -= take;
184 remaining -= take;
185 cost += take * l.cost;
186 if (removed) l.removed += take;
187 if (remaining == 0) break;
188 }
189 return .{ .parts = parts.items, .cost = cost };
190 }
191 fn ship(self: *Inventory, o: *Order, quantity: i64, issued: Issue) Error!void {
192 o.shipped += quantity;
193 o.backorder -= quantity;
194 o.revenue += quantity * o.price;
195 o.cogs += issued.cost;
196 try o.issues.appendSlice(self.gpa, issued.parts);
197 o.item.sold += quantity;
198 o.item.revenue += quantity * o.price;
199 o.item.cogs += issued.cost;
200 if (quantity > 0) self.tally(4, quantity, issued.cost);
201 }
202 fn allocate(self: *Inventory, w: *Warehouse, item_ptr: *Item) Error!void {
203 for (self.orders.items) |o| {
204 if (o.warehouse != w or o.item != item_ptr or o.backorder == 0) continue;
205 const q = @min(o.backorder, stock(item_ptr, w).quantity);
206 if (q == 0) break;
207 const issued = try self.issue(item_ptr, w, q, false);
208 try self.ship(o, q, issued);
209 try self.card(item_ptr, "fill", w, -q, -issued.cost);
210 try self.out.print(" FILL order {d} qty {d} revenue {s} cogs {s} profit {s} backorder {d}\n", .{ o.number, q, try self.money(q * o.price), try self.money(issued.cost), try self.money(q * o.price - issued.cost), o.backorder });
211 }
212 }
213 fn requireQuantity(self: *Inventory, q: i64) Error!void {
214 if (q == 0) return self.fail("invalid quantity", .{});
215 }
216 fn requireStock(self: *Inventory, item_ptr: *Item, w: *Warehouse, q: i64) Error!void {
217 const h = stock(item_ptr, w).quantity;
218 if (h < q) return self.fail("insufficient stock {s} {s} {d}", .{ w.code, item_ptr.code, h });
219 }
220 fn execute(self: *Inventory, a: []const []const u8) Error!void {
221 const op = a[0];
222 const id: []const u8 = if (a.len > 1) a[1] else "";
223 if (std.mem.eql(u8, op, "DAY")) {
224 const d = integer(id);
225 if (d <= self.day) return self.fail("invalid day {s}", .{id});
226 self.day = d;
227 try self.out.print("DAY {d}\n", .{d});
228 } else if (std.mem.eql(u8, op, "WAREHOUSE")) {
229 if (self.warehouses.contains(id)) return self.fail("duplicate warehouse {s}", .{id});
230 const w = try self.gpa.create(Warehouse);
231 w.* = .{ .code = id, .index = self.warehouses.count() };
232 try self.warehouses.put(self.gpa, id, w);
233 try self.out.print("WAREHOUSE {s}\n", .{id});
234 } else if (std.mem.eql(u8, op, "ITEM")) {
235 const old = self.items.get(id);
236 const s = old orelse try self.gpa.create(Item);
237 if (old == null) {
238 s.* = .{ .code = id, .name = a[3], .level = integer(a[2]) };
239 try self.items.put(self.gpa, id, s);
240 } else {
241 s.name = a[3];
242 s.level = integer(a[2]);
243 }
244 try self.out.print("ITEM {s} {s} {s} level {d}\n", .{ id, if (old == null) "added" else "updated", s.name, s.level });
245 } else if (std.mem.eql(u8, op, "RECEIVE") or std.mem.eql(u8, op, "SHIP") or std.mem.eql(u8, op, "ADJUST") or std.mem.eql(u8, op, "COUNT")) {
246 const w = try self.warehouse(id);
247 const s = try self.item(a[2]);
248 if (std.mem.eql(u8, op, "RECEIVE")) {
249 const q = integer(a[3]);
250 const c = amount(a[4]);
251 const d = integer(a[5]);
252 try self.requireQuantity(q);
253 if (d > self.day) return self.fail("future day {s}", .{a[5]});
254 const l = try self.create(s, w, q, c, d, "receive");
255 s.last_cost = c;
256 self.tally(0, q, q * c);
257 try self.card(s, "receive", w, q, q * c);
258 try self.out.print("RECEIVE {s} {s} lot {d} qty {d} cost {s} day {d} value {s}\n", .{ id, s.code, l.number, q, try self.money(c), d, try self.money(q * c) });
259 try self.allocate(w, s);
260 } else if (std.mem.eql(u8, op, "SHIP")) {
261 const q = integer(a[3]);
262 const price = amount(a[4]);
263 try self.requireQuantity(q);
264 const o = try self.gpa.create(Order);
265 o.* = .{ .number = self.orders.items.len + 1, .warehouse = w, .item = s, .day = self.day, .price = price, .ordered = q, .backorder = q };
266 try self.orders.append(self.gpa, o);
267 s.order_count += 1;
268 const f = @min(q, stock(s, w).quantity);
269 const issued = try self.issue(s, w, f, false);
270 try self.ship(o, f, issued);
271 if (f > 0) try self.card(s, "ship", w, -f, -issued.cost);
272 try self.out.print("SHIP order {d} {s} {s} shipped {d} revenue {s} cogs {s} profit {s} backorder {d}\n", .{ o.number, id, s.code, f, try self.money(o.revenue), try self.money(o.cogs), try self.money(o.revenue - o.cogs), o.backorder });
273 } else if (std.mem.eql(u8, op, "ADJUST")) {
274 const positive = a[3][0] == '+';
275 const q = integer(a[3][1..]);
276 try self.requireQuantity(q);
277 if (positive) {
278 const c = amount(a[4]);
279 const l = try self.create(s, w, q, c, self.day, "adjust");
280 self.tally(2, q, q * c);
281 try self.card(s, "adjust", w, q, q * c);
282 try self.out.print("ADJUST {s} {s} +{d} lot {d} value {s}\n", .{ id, s.code, q, l.number, try self.money(q * c) });
283 try self.allocate(w, s);
284 } else {
285 try self.requireStock(s, w, q);
286 const issued = try self.issue(s, w, q, true);
287 self.tally(3, q, issued.cost);
288 try self.card(s, "adjust", w, -q, -issued.cost);
289 try self.out.print("ADJUST {s} {s} -{d} value {s}\n", .{ id, s.code, q, try self.money(issued.cost) });
290 }
291 } else {
292 const actual = integer(a[3]);
293 const h = stock(s, w).quantity;
294 if (actual == h) {
295 try self.out.print("COUNT {s} {s} qty {d} ok\n", .{ id, s.code, actual });
296 } else if (actual < h) {
297 const q = h - actual;
298 const issued = try self.issue(s, w, q, true);
299 self.tally(3, q, issued.cost);
300 try self.card(s, "count", w, -q, -issued.cost);
301 try self.out.print("COUNT {s} {s} qty {d} -> {d} shortage {d} value {s}\n", .{ id, s.code, h, actual, q, try self.money(issued.cost) });
302 } else {
303 const q = actual - h;
304 const c = s.last_cost orelse 0;
305 const l = try self.create(s, w, q, c, self.day, "count");
306 self.tally(2, q, q * c);
307 try self.card(s, "count", w, q, q * c);
308 try self.out.print("COUNT {s} {s} qty {d} -> {d} surplus {d} lot {d} value {s}\n", .{ id, s.code, h, actual, q, l.number, try self.money(q * c) });
309 try self.allocate(w, s);
310 }
311 }
312 } else if (std.mem.eql(u8, op, "CANCEL")) {
313 const o = try self.order(id);
314 if (o.backorder == 0) return self.fail("no backorder {s}", .{id});
315 const q = o.backorder;
316 o.cancelled += q;
317 o.backorder = 0;
318 try self.out.print("CANCEL order {d} qty {d}\n", .{ o.number, q });
319 } else if (std.mem.eql(u8, op, "TRANSFER")) {
320 const from = try self.warehouse(id);
321 const to = try self.warehouse(a[2]);
322 if (from == to) return self.fail("same warehouse {s}", .{id});
323 const s = try self.item(a[3]);
324 const q = integer(a[4]);
325 try self.requireQuantity(q);
326 try self.requireStock(s, from, q);
327 const issued = try self.issue(s, from, q, false);
328 for (issued.parts) |p| p.lot.held[to.index] += p.quantity;
329 try self.card(s, "transfer-out", from, -q, -issued.cost);
330 try self.card(s, "transfer-in", to, q, issued.cost);
331 try self.out.print("TRANSFER {s} -> {s} {s} qty {d} value {s} lots {d}\n", .{ id, to.code, s.code, q, try self.money(issued.cost), issued.parts.len });
332 try self.allocate(to, s);
333 } else if (std.mem.eql(u8, op, "RETURN")) {
334 const o = try self.order(id);
335 const q = integer(a[2]);
336 try self.requireQuantity(q);
337 if (q > o.shipped - o.returned) return self.fail("return exceeds shipped {s} {d}", .{ id, o.shipped - o.returned });
338 const c = amount(a[3]);
339 const l = try self.create(o.item, o.warehouse, q, c, self.day, "return");
340 o.returned += q;
341 o.revenue -= q * o.price;
342 o.cogs -= q * c;
343 o.item.sold -= q;
344 o.item.revenue -= q * o.price;
345 o.item.cogs -= q * c;
346 self.tally(1, q, q * c);
347 try self.card(o.item, "return", o.warehouse, q, q * c);
348 try self.out.print("RETURN order {d} {s} {s} qty {d} lot {d} refund {s} value {s}\n", .{ o.number, o.warehouse.code, o.item.code, q, l.number, try self.money(q * o.price), try self.money(q * c) });
349 try self.allocate(o.warehouse, o.item);
350 } else if (std.mem.eql(u8, op, "MARKET")) {
351 const s = try self.item(id);
352 const price = amount(a[2]);
353 var changed: usize = 0;
354 var q: i64 = 0;
355 var loss: i64 = 0;
356 for (try self.warehouseList()) |w| {
357 var warehouse_loss: i64 = 0;
358 for (s.lots.items) |l| {
359 if (l.cost > price and l.held[w.index] > 0) {
360 const h = l.held[w.index];
361 const value = h * (l.cost - price);
362 changed += 1;
363 q += h;
364 loss += value;
365 warehouse_loss += value;
366 }
367 }
368 if (warehouse_loss > 0) try self.card(s, "writedown", w, 0, -warehouse_loss);
369 }
370 for (s.lots.items) |l| {
371 if (l.cost > price and held(l, null) > 0) l.cost = price;
372 }
373 self.tally(5, q, loss);
374 try self.out.print("MARKET {s} price {s} lots {d} qty {d} loss {s}\n", .{ id, try self.money(price), changed, q, try self.money(loss) });
375 } else if (std.mem.eql(u8, op, "RECALL")) {
376 const l = try self.lot(id);
377 const q = held(l, null);
378 if (q == 0) return self.fail("nothing to recall {s}", .{id});
379 try self.out.print("RECALL lot {d} {s} qty {d} value {s}\n", .{ l.number, l.item.code, q, try self.money(q * l.cost) });
380 for (try self.warehouseList()) |w| {
381 const h = l.held[w.index];
382 if (h == 0) continue;
383 try self.card(l.item, "recall", w, -h, -h * l.cost);
384 try self.out.print(" {s} qty {d} value {s}\n", .{ w.code, h, try self.money(h * l.cost) });
385 l.held[w.index] = 0;
386 }
387 l.removed += q;
388 self.tally(7, q, q * l.cost);
389 } else if (std.mem.eql(u8, op, "EXPIRE")) {
390 const age = integer(id);
391 var total_q: i64 = 0;
392 var total_v: i64 = 0;
393 try self.out.print("EXPIRE {d}\n", .{age});
394 const ws = try self.warehouseList();
395 for (try self.itemList()) |s| {
396 for (ws) |w| {
397 var count: usize = 0;
398 var q: i64 = 0;
399 var value: i64 = 0;
400 for (s.lots.items) |l| {
401 if (l.held[w.index] > 0 and self.day - l.day > age) {
402 const h = l.held[w.index];
403 count += 1;
404 q += h;
405 value += h * l.cost;
406 l.removed += h;
407 l.held[w.index] = 0;
408 }
409 }
410 if (count > 0) {
411 self.tally(6, q, value);
412 try self.card(s, "expire", w, -q, -value);
413 total_q += q;
414 total_v += value;
415 try self.out.print(" {s} {s} lots {d} qty {d} value {s}\n", .{ s.code, w.code, count, q, try self.money(value) });
416 }
417 }
418 }
419 try self.out.print(" total qty {d} value {s}\n", .{ total_q, try self.money(total_v) });
420 } else if (std.mem.eql(u8, op, "VALUE") or std.mem.eql(u8, op, "AGING")) {
421 const w: ?*Warehouse = if (a.len > 1) try self.warehouse(id) else null;
422 const ws = try self.warehouseList();
423 if (std.mem.eql(u8, op, "VALUE")) {
424 try self.out.print("VALUE{s}\n", .{if (w) |wh| try std.fmt.allocPrint(self.gpa, " {s}", .{wh.code}) else ""});
425 var total_q: i64 = 0;
426 var total_v: i64 = 0;
427 for (try self.itemList()) |s| {
428 const h = stock(s, w);
429 if (h.quantity == 0) continue;
430 total_q += h.quantity;
431 total_v += h.value;
432 try self.out.print(" {s} qty {d} value {s} avg {s}\n", .{ s.code, h.quantity, try self.money(h.value), try self.money(@divTrunc(h.value, h.quantity)) });
433 if (w == null) {
434 for (ws) |wh| {
435 const whs = stock(s, wh);
436 if (whs.quantity > 0) try self.out.print(" {s} qty {d} value {s}\n", .{ wh.code, whs.quantity, try self.money(whs.value) });
437 }
438 }
439 }
440 try self.out.print(" total qty {d} value {s}\n", .{ total_q, try self.money(total_v) });
441 } else {
442 try self.out.print("AGING{s} day {d}\n", .{ if (w) |wh| try std.fmt.allocPrint(self.gpa, " {s}", .{wh.code}) else "", self.day });
443 const labels = [_][]const u8{ "0-30", "31-60", "61-90", "91+" };
444 var total_q: [4]i64 = @splat(0);
445 var total_v: [4]i64 = @splat(0);
446 for (try self.itemList()) |s| {
447 var qs: [4]i64 = @splat(0);
448 var quantity: i64 = 0;
449 for (s.lots.items) |l| {
450 const h = held(l, w);
451 const age = self.day - l.day;
452 const bucket: usize = if (age <= 30) 0 else if (age <= 60) 1 else if (age <= 90) 2 else 3;
453 qs[bucket] += h;
454 total_q[bucket] += h;
455 total_v[bucket] += h * l.cost;
456 quantity += h;
457 }
458 if (quantity > 0) {
459 try self.out.print(" {s}", .{s.code});
460 for (0..4) |k| try self.out.print(" {s} {d}", .{ labels[k], qs[k] });
461 try self.out.writeByte('\n');
462 }
463 }
464 try self.out.writeAll(" total");
465 for (0..4) |k| try self.out.print(" {s} {d}", .{ labels[k], total_q[k] });
466 try self.out.writeAll("\n value");
467 for (0..4) |k| try self.out.print(" {s} {s}", .{ labels[k], try self.money(total_v[k]) });
468 try self.out.writeByte('\n');
469 }
470 } else if (std.mem.eql(u8, op, "LOTS")) {
471 const w = try self.warehouse(id);
472 const s = try self.item(a[2]);
473 var count: usize = 0;
474 try self.out.print("LOTS {s} {s}\n", .{ id, s.code });
475 for (s.lots.items) |l| {
476 const q = l.held[w.index];
477 if (q > 0) {
478 count += 1;
479 try self.out.print(" lot {d} day {d} qty {d} cost {s} value {s} age {d}\n", .{ l.number, l.day, q, try self.money(l.cost), try self.money(q * l.cost), self.day - l.day });
480 }
481 }
482 for (self.orders.items) |o| {
483 if (o.warehouse == w and o.item == s and o.backorder > 0) {
484 count += 1;
485 try self.out.print(" backorder order {d} qty {d}\n", .{ o.number, o.backorder });
486 }
487 }
488 if (count == 0) try self.out.writeAll(" (none)\n");
489 } else if (std.mem.eql(u8, op, "REORDER") or std.mem.eql(u8, op, "ITEMS")) {
490 try self.out.print("{s}\n", .{op});
491 var count: usize = 0;
492 for (try self.itemList()) |s| {
493 const h = stock(s, null).quantity;
494 const b = self.backorder(s, null);
495 if (std.mem.eql(u8, op, "REORDER")) {
496 if (h - b > s.level) continue;
497 try self.out.print(" {s} {s} onhand {d} backorder {d} level {d}\n", .{ s.code, s.name, h, b, s.level });
498 } else try self.out.print(" {s} {s} level {d} onhand {d} backorder {d} cost {s}\n", .{ s.code, s.name, s.level, h, b, if (s.last_cost) |c| try self.money(c) else "none" });
499 count += 1;
500 }
501 if (count == 0) try self.out.writeAll(" (none)\n");
502 } else if (std.mem.eql(u8, op, "MARGIN") or std.mem.eql(u8, op, "TOP")) {
503 var selected: std.ArrayList(*Item) = .empty;
504 for (try self.itemList()) |s| {
505 if (s.order_count > 0) try selected.append(self.gpa, s);
506 }
507 if (std.mem.eql(u8, op, "MARGIN")) {
508 try self.out.writeAll("MARGIN\n");
509 var sold: i64 = 0;
510 var revenue: i64 = 0;
511 var cogs: i64 = 0;
512 for (selected.items) |s| {
513 try self.out.print(" {s} sold {d} revenue {s} cogs {s} profit {s} margin {s}\n", .{ s.code, s.sold, try self.money(s.revenue), try self.money(s.cogs), try self.money(s.revenue - s.cogs), try self.margin(s.revenue - s.cogs, s.revenue) });
514 sold += s.sold;
515 revenue += s.revenue;
516 cogs += s.cogs;
517 }
518 try self.out.print(" total sold {d} revenue {s} cogs {s} profit {s} margin {s}\n", .{ sold, try self.money(revenue), try self.money(cogs), try self.money(revenue - cogs), try self.margin(revenue - cogs, revenue) });
519 } else {
520 std.mem.sort(*Item, selected.items, {}, struct {
521 fn less(_: void, left: *Item, right: *Item) bool {
522 const ap = left.revenue - left.cogs;
523 const bp = right.revenue - right.cogs;
524 if (ap != bp) return ap > bp;
525 if (left.revenue != right.revenue) return left.revenue > right.revenue;
526 return std.mem.order(u8, left.code, right.code) == .lt;
527 }
528 }.less);
529 const k: usize = @intCast(integer(id));
530 try self.out.print("TOP {d}\n", .{k});
531 for (selected.items[0..@min(k, selected.items.len)], 0..) |s, i| try self.out.print(" {d} {s} profit {s} revenue {s} margin {s}\n", .{ i + 1, s.code, try self.money(s.revenue - s.cogs), try self.money(s.revenue), try self.margin(s.revenue - s.cogs, s.revenue) });
532 if (selected.items.len == 0) try self.out.writeAll(" (none)\n");
533 }
534 } else if (std.mem.eql(u8, op, "BACKORDERS")) {
535 try self.out.writeAll("BACKORDERS\n");
536 var count: usize = 0;
537 for (self.orders.items) |o| {
538 if (o.backorder > 0) {
539 count += 1;
540 try self.out.print(" order {d} {s} {s} qty {d} day {d} price {s}\n", .{ o.number, o.warehouse.code, o.item.code, o.backorder, o.day, try self.money(o.price) });
541 }
542 }
543 if (count == 0) try self.out.writeAll(" (none)\n");
544 } else if (std.mem.eql(u8, op, "ORDER")) {
545 const o = try self.order(id);
546 try self.out.print("ORDER {d} {s} {s} day {d} price {s}\n", .{ o.number, o.warehouse.code, o.item.code, o.day, try self.money(o.price) });
547 try self.out.print(" ordered {d} shipped {d} backorder {d} cancelled {d} returned {d}\n", .{ o.ordered, o.shipped, o.backorder, o.cancelled, o.returned });
548 for (o.issues.items) |p| try self.out.print(" lot {d} qty {d} cost {s}\n", .{ p.lot.number, p.quantity, try self.money(p.cost) });
549 try self.out.print(" revenue {s} cogs {s} profit {s}\n", .{ try self.money(o.revenue), try self.money(o.cogs), try self.money(o.revenue - o.cogs) });
550 } else if (std.mem.eql(u8, op, "TRACE")) {
551 const l = try self.lot(id);
552 try self.out.print("TRACE lot {d} {s} {s} {s} day {d} qty {d} cost {s}\n", .{ l.number, l.item.code, l.kind, l.warehouse.code, l.day, l.initial_quantity, try self.money(l.initial_cost) });
553 for (try self.warehouseList()) |w| {
554 if (l.held[w.index] > 0) try self.out.print(" held {s} qty {d} cost {s}\n", .{ w.code, l.held[w.index], try self.money(l.cost) });
555 }
556 for (self.orders.items) |o| {
557 var q: i64 = 0;
558 for (o.issues.items) |p| {
559 if (p.lot == l) q += p.quantity;
560 }
561 if (q > 0) try self.out.print(" order {d} qty {d}\n", .{ o.number, q });
562 }
563 if (l.removed > 0) try self.out.print(" removed qty {d}\n", .{l.removed});
564 } else if (std.mem.eql(u8, op, "CARD")) {
565 const s = try self.item(id);
566 try self.out.print("CARD {s} {s}\n", .{ id, s.name });
567 var q: i64 = 0;
568 var value: i64 = 0;
569 for (s.card.items) |e| {
570 q += e.quantity;
571 value += e.value;
572 try self.out.print(" day {d} {s} {s} {s}{d} {s}{s} balance {d} {s}\n", .{ e.day, e.kind, e.warehouse.code, if (e.quantity > 0) "+" else "", e.quantity, if (e.value > 0) "+" else "", try self.money(e.value), q, try self.money(value) });
573 }
574 if (s.card.items.len == 0) try self.out.writeAll(" (none)\n");
575 } else if (std.mem.eql(u8, op, "STATUS")) {
576 try self.out.writeAll("STATUS\n");
577 for (try self.warehouseList()) |w| {
578 var skus: usize = 0;
579 var q: i64 = 0;
580 var value: i64 = 0;
581 var b: i64 = 0;
582 for (self.items.values()) |s| {
583 const h = stock(s, w);
584 if (h.quantity > 0) skus += 1;
585 q += h.quantity;
586 value += h.value;
587 }
588 for (self.orders.items) |o| {
589 if (o.warehouse == w) b += o.backorder;
590 }
591 try self.out.print(" {s} skus {d} qty {d} value {s} backorder {d}\n", .{ w.code, skus, q, try self.money(value), b });
592 }
593 if (self.warehouses.count() == 0) try self.out.writeAll(" (none)\n");
594 } else if (std.mem.eql(u8, op, "SUMMARY")) {
595 try self.out.writeAll("SUMMARY\n");
596 for (0..8) |i| try self.out.print(" {s} qty {d} value {s}\n", .{ categories[i], self.quantities[i], try self.money(self.values[i]) });
597 var q: i64 = 0;
598 var value: i64 = 0;
599 var open: usize = 0;
600 var b: i64 = 0;
601 for (self.items.values()) |s| {
602 const h = stock(s, null);
603 q += h.quantity;
604 value += h.value;
605 }
606 for (self.orders.items) |o| {
607 if (o.backorder > 0) open += 1;
608 b += o.backorder;
609 }
610 try self.out.print(" on-hand qty {d} value {s}\n orders {d} open {d} backorder {d}\n", .{ q, try self.money(value), self.orders.items.len, open, b });
611 }
612 }
613};
614pub fn main(init: std.process.Init) !void {
615 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
616 defer arena.deinit();
617 const gpa = arena.allocator();
618 var ib: [4096]u8 = undefined;
619 var input = std.Io.File.stdin().readerStreaming(init.io, &ib);
620 var ob: [8192]u8 = undefined;
621 var output = std.Io.File.stdout().writerStreaming(init.io, &ob);
622 var inventory = Inventory{ .gpa = gpa, .out = &output.interface };
623 const n: usize = @intCast(integer((try input.interface.takeDelimiter('\n')).?));
624 for (0..n) |_| {
625 const line = try gpa.dupe(u8, (try input.interface.takeDelimiter('\n')).?);
626 var a: std.ArrayList([]const u8) = .empty;
627 var it = std.mem.splitScalar(u8, line, ' ');
628 while (it.next()) |part| try a.append(gpa, part);
629 var draft: std.Io.Writer.Allocating = .init(gpa);
630 inventory.out = &draft.writer;
631 inventory.execute(a.items) catch |err| {
632 if (err == error.BadCommand) {
633 try output.interface.print("ERROR {s}\n", .{inventory.error_message});
634 continue;
635 }
636 return err;
637 };
638 try output.interface.writeAll(draft.writer.buffered());
639 }
640 try output.interface.flush();
641}
