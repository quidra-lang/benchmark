30string money(Amount amount) {
31 ostringstream out;
32 if (amount < 0) {
33 out << '-';
34 amount = -amount;
35 }
36 out << amount / 100 << '.' << setfill('0') << setw(2) << amount % 100;
37 return out.str();
38}

48string margin(Amount profit, Amount revenue) {
49 if (revenue == 0) {
50 return "n/a";
51 }
52 Amount tenths = (profit < 0 ? -profit : profit) * 1000 / revenue;
53 return (profit < 0 && tenths > 0 ? "-" : "") +
54 to_string(tenths / 10) + "." + to_string(tenths % 10) + "%";
55}

132 template<class... Args>
133 void emit(int indent, const Args&... parts) {
134 ostringstream out;
135 out << string(indent, ' ');
136 bool first = true;
137 ((out << (exchange(first, false) ? "" : " ") << parts), ...);
138 output.push_back(out.str());
139 }

186 const map<int, StockLot>& holdings(const string& warehouseCode, const string& code) const {
187 static const map<int, StockLot> empty;
188 auto w = stock.find(warehouseCode);
189 if (w == stock.end()) {
190 return empty;
191 }
192 auto entry = w->second.find(code);
193 return entry == w->second.end() ? empty : entry->second;
194 }
195
196 vector<int> fifo(const string& warehouseCode, const string& code) const {
197 vector<int> ids;
198 for (const auto& [id, current] : holdings(warehouseCode, code)) {
199 ids.push_back(id);
200 }
201 sort(ids.begin(), ids.end(), [&](int a, int b) {
202 if (lots[a - 1].day != lots[b - 1].day) {
203 return lots[a - 1].day < lots[b - 1].day;
204 }
205 return a < b;
206 });
207 return ids;
208 }
209
210 Amount quantity(const string& warehouseCode, const string& code) const {
211 Amount result = 0;
212 for (const auto& [id, current] : holdings(warehouseCode, code)) {
213 result += current.quantity;
214 }
215 return result;
216 }

236 Total allStocks() const {

275 int createLot(const string& w, const string& code, Amount receiptDay,
276 Amount q, Amount cost, const string& kind) {
277 int id = static_cast<int>(lots.size()) + 1;
278 lots.push_back({id, code, w, kind, receiptDay, q, cost, 0});
279 addStock(w, id, q, cost);
280 return id;
281 }
282
283 std::pair<vector<Part>, Amount> issue(const string& w, const string& code, Amount q) {
284 vector<Part> parts;
285 Amount cost = 0;
286 for (int id : fifo(w, code)) {
287 if (q == 0) {
288 break;
289 }
290 StockLot& current = stock.at(w).at(code).at(id);
291 Amount taken = min(q, current.quantity);
292 parts.push_back({id, taken, current.cost});
293 cost += taken * current.cost;
294 current.quantity -= taken;
295 q -= taken;
296 if (current.quantity == 0) {
297 stock.at(w).at(code).erase(id);
298 }
299 }
300 return {parts, cost};
301 }
302
303 Amount fulfil(Order& current, Amount q, const string& kind) {
304 if (q == 0) {
305 return 0;
306 }
307 auto [parts, cost] = issue(current.warehouse, current.item, q);
308 current.shipped += q;
309 current.backorder -= q;
310 current.revenue += q * current.price;
311 current.cogs += cost;
312 current.issues.insert(current.issues.end(), parts.begin(), parts.end());
313 accumulate("shipped", q, cost);
314 cardEntry(current.item, kind, current.warehouse, -q, -cost);
315 return cost;
316 }
317
318 void allocate(const string& w, const string& code) {
319 Amount available = quantity(w, code);
320 for (Order& current : orders) {
321 if (available == 0) {
322 break;
323 }
324 if (current.warehouse != w || current.item != code || current.backorder == 0) {
325 continue;
326 }
327 Amount q = min(available, current.backorder);
328 Amount cost = fulfil(current, q, "fill");
329 Amount revenue = q * current.price;
330 available -= q;
331 emit(2, "FILL", "order", current.number, "qty", q,
332 "revenue", money(revenue), "cogs", money(cost),
333 "profit", money(revenue - cost), "backorder", current.backorder);
334 }
335 }
336
337 void removeParts(const vector<Part>& parts) {
338 for (const Part& part : parts) {
339 lots[part.lot - 1].removed += part.quantity;
340 }
341 }

343 map<string, Sale> sales() const {
344 map<string, Sale> rows;
345 for (const Order& current : orders) {
346 Sale& row = rows[current.item];
347 row.quantity += current.shipped - current.returned;
348 row.revenue += current.revenue;
349 row.cogs += current.cogs;
350 }
351 return rows;
352 }

354 void marginLine(const string& label, const Sale& row) {
355 Amount profit = row.revenue - row.cogs;
356 emit(2, label, "sold", row.quantity, "revenue", money(row.revenue),
357 "cogs", money(row.cogs), "profit", money(profit),
358 "margin", margin(profit, row.revenue));
359 }
360
361 void agingLine(const string& label, const vector<Amount>& buckets, bool monetary) {
362 ostringstream text;
363 text << label;
364 const vector<string> labels = {"0-30", "31-60", "61-90", "91+"};
365 for (size_t i = 0; i < buckets.size(); ++i) {
366 text << ' ' << labels[i] << ' '
367 << (monetary ? money(buckets[i]) : to_string(buckets[i]));
368 }
369 emit(2, text.str());
370 }

372 void command(const string& cmd, const vector<string>& a) {
373 if (cmd == "DAY") {
374 Amount next = number(a[0]);
375 if (next <= day) {
376 error("invalid day " + a[0]);
377 }
378 day = next;
379 emit(0, "DAY", day);
380
381 } else if (cmd == "WAREHOUSE") {
382 const string& w = a[0];
383 if (stock.find(w) != stock.end()) {
384 error("duplicate warehouse " + w);
385 }
386 stock.emplace(w, map<string, map<int, StockLot>>{});
387 emit(0, "WAREHOUSE", w);
388
389 } else if (cmd == "ITEM") {
390 const string& code = a[0];
391 Amount level = number(a[1]);
392 string action = "added";
393 auto found = items.find(code);
394 if (found != items.end()) {
395 found->second.level = level;
396 found->second.name = a[2];
397 action = "updated";
398 } else {
399 items.emplace(code, Item{code, a[2], level, nullopt, {}});
400 }
401 emit(0, "ITEM", code, action, a[2], "level", level);
402
403 } else if (cmd == "RECEIVE") {
404 const string& w = a[0];
405 const string& code = a[1];
406 Item& entry = pair(w, code);
407 Amount q = number(a[2]);
408 Amount cost = cents(a[3]);
409 Amount receiptDay = number(a[4]);
410 requireQuantity(q);
411 if (receiptDay > day) {
412 error("future day " + a[4]);
413 }
414 int id = createLot(w, code, receiptDay, q, cost, "receive");
415 entry.lastCost = cost;
416 accumulate("received", q, q * cost);
417 cardEntry(code, "receive", w, q, q * cost);
418 emit(0, "RECEIVE", w, code, "lot", id, "qty", q, "cost",
419 money(cost), "day", receiptDay, "value", money(q * cost));
420 allocate(w, code);
421
422 } else if (cmd == "SHIP") {
423 const string& w = a[0];
424 const string& code = a[1];
425 pair(w, code);
426 Amount q = number(a[2]);
427 requireQuantity(q);
428 int id = static_cast<int>(orders.size()) + 1;
429 orders.push_back({id, w, code, day, cents(a[3]), q, 0, q});
430 Order& current = orders.back();
431 Amount shipped = min(q, quantity(w, code));
432 Amount cost = fulfil(current, shipped, "ship");
433 emit(0, "SHIP", "order", current.number, w, code, "shipped", shipped,
434 "revenue", money(current.revenue), "cogs", money(cost),
435 "profit", money(current.revenue - cost), "backorder", current.backorder);
436
437 } else if (cmd == "CANCEL") {
438 Order& current = order(a[0]);
439 if (current.backorder == 0) {
440 error("no backorder " + a[0]);
441 }
442 current.cancelled += current.backorder;
443 current.backorder = 0;
444 emit(0, "CANCEL", "order", current.number, "qty", current.cancelled);
445
446 } else if (cmd == "TRANSFER") {
447 const string& source = a[0];
448 const string& destination = a[1];
449 const string& code = a[2];
450 warehouse(source);
451 warehouse(destination);
452 if (source == destination) {
453 error("same warehouse " + source);
454 }
455 item(code);
456 Amount q = number(a[3]);
457 requireQuantity(q);
458 requireStock(source, code, q);
459 auto [parts, cost] = issue(source, code, q);
460 for (const Part& part : parts) {
461 addStock(destination, part.lot, part.quantity, part.cost);
462 }
463 cardEntry(code, "transfer-out", source, -q, -cost);
464 cardEntry(code, "transfer-in", destination, q, cost);
465 emit(0, "TRANSFER", source, "->", destination, code,
466 "qty", q, "value", money(cost), "lots", parts.size());
467 allocate(destination, code);
468
469 } else if (cmd == "ADJUST") {
470 const string& w = a[0];
471 const string& code = a[1];
472 pair(w, code);
473 Amount q = number(a[2].substr(1));
474 requireQuantity(q);
475 if (a[2][0] == '+') {
476 Amount cost = cents(a[3]);
477 int id = createLot(w, code, day, q, cost, "adjust");
478 accumulate("adjusted-in", q, q * cost);
479 cardEntry(code, "adjust", w, q, q * cost);
480 emit(0, "ADJUST", w, code, "+" + to_string(q),
481 "lot", id, "value", money(q * cost));
482 allocate(w, code);
483 } else {
484 requireStock(w, code, q);
485 auto [parts, cost] = issue(w, code, q);
486 removeParts(parts);
487 accumulate("adjusted-out", q, cost);
488 cardEntry(code, "adjust", w, -q, -cost);
489 emit(0, "ADJUST", w, code, "-" + to_string(q), "value", money(cost));
490 }
491
492 } else if (cmd == "RETURN") {
493 Order& current = order(a[0]);
494 Amount q = number(a[1]);
495 Amount cost = cents(a[2]);
496 requireQuantity(q);
497 Amount available = current.shipped - current.returned;
498 if (q > available) {
499 error("return exceeds shipped " + a[0] + " " + to_string(available));
500 }
501 int id = createLot(current.warehouse, current.item, day, q, cost, "return");
502 Amount refund = q * current.price;
503 current.returned += q;
504 current.revenue -= refund;
505 current.cogs -= q * cost;
506 accumulate("returned", q, q * cost);
507 cardEntry(current.item, "return", current.warehouse, q, q * cost);
508 emit(0, "RETURN", "order", current.number, current.warehouse, current.item,
509 "qty", q, "lot", id, "refund", money(refund), "value", money(q * cost));
510 allocate(current.warehouse, current.item);
511
512 } else if (cmd == "COUNT") {
513 const string& w = a[0];
514 const string& code = a[1];
515 Item& entry = pair(w, code);
516 Amount actual = number(a[2]);
517 Amount held = quantity(w, code);
518 if (actual == held) {
519 emit(0, "COUNT", w, code, "qty", actual, "ok");
520 } else if (actual < held) {
521 Amount q = held - actual;
522 auto [parts, cost] = issue(w, code, q);
523 removeParts(parts);
524 accumulate("adjusted-out", q, cost);
525 cardEntry(code, "count", w, -q, -cost);
526 emit(0, "COUNT", w, code, "qty", held, "->", actual,
527 "shortage", q, "value", money(cost));
528 } else {
529 Amount q = actual - held;
530 Amount cost = entry.lastCost.value_or(0);
531 int id = createLot(w, code, day, q, cost, "count");
532 accumulate("adjusted-in", q, q * cost);
533 cardEntry(code, "count", w, q, q * cost);
534 emit(0, "COUNT", w, code, "qty", held, "->", actual,
535 "surplus", q, "lot", id, "value", money(q * cost));
536 allocate(w, code);
537 }
538
539 } else if (cmd == "MARKET") {
540 const string& code = a[0];
541 item(code);
542 Amount price = cents(a[1]);
543 Amount changed = 0;
544 Amount q = 0;
545 Amount loss = 0;
546 for (auto& [w, contents] : stock) {
547 Amount warehouseLoss = 0;
548 auto it = contents.find(code);
549 if (it != contents.end()) {
550 for (auto& [id, current] : it->second) {
551 if (current.cost > price) {
552 warehouseLoss += current.quantity * (current.cost - price);
553 changed++;
554 q += current.quantity;
555 current.cost = price;
556 }
557 }
558 }
559 if (warehouseLoss > 0) {
560 cardEntry(code, "writedown", w, 0, -warehouseLoss);
561 }
562 loss += warehouseLoss;
563 }
564 accumulate("written-down", q, loss);
565 emit(0, "MARKET", code, "price", money(price),
566 "lots", changed, "qty", q, "loss", money(loss));
567
568 } else if (cmd == "RECALL") {
569 Lot& current = lot(a[0]);
570 struct Holding {
571 string warehouse;
572 Amount quantity;
573 Amount cost;
574 };
575 vector<Holding> held;
576 for (const auto& [w, contents] : stock) {
577 auto it = holdings(w, current.item).find(current.number);
578 if (it != holdings(w, current.item).end()) {
579 held.push_back({w, it->second.quantity, it->second.cost});
580 }
581 }
582 if (held.empty()) {
583 error("nothing to recall " + a[0]);
584 }
585 Amount q = 0;
586 Amount value = 0;
587 for (const Holding& entry : held) {
588 q += entry.quantity;
589 value += entry.quantity * entry.cost;
590 stock.at(entry.warehouse).at(current.item).erase(current.number);
591 cardEntry(current.item, "recall", entry.warehouse,
592 -entry.quantity, -entry.quantity * entry.cost);
593 }
594 current.removed += q;
595 accumulate("recalled", q, value);
596 emit(0, "RECALL", "lot", current.number, current.item,
597 "qty", q, "value", money(value));
598 for (const Holding& entry : held) {
599 emit(2, entry.warehouse, "qty", entry.quantity,
600 "value", money(entry.quantity * entry.cost));
601 }
602
603 } else if (cmd == "EXPIRE") {
604 Amount days = number(a[0]);
605 emit(0, "EXPIRE", days);
606 Amount totalQ = 0;
607 Amount totalValue = 0;
608 for (const auto& [code, entry] : items) {
609 for (auto& [w, contents] : stock) {
610 auto it = contents.find(code);
611 if (it == contents.end()) {
612 continue;
613 }
614 vector<int> expired;
615 Amount q = 0;
616 Amount value = 0;
617 for (const auto& [id, current] : it->second) {
618 if (day - lots[id - 1].day > days) {
619 expired.push_back(id);
620 q += current.quantity;
621 value += current.quantity * current.cost;
622 }
623 }
624 if (expired.empty()) {
625 continue;
626 }
627 for (int id : expired) {
628 lots[id - 1].removed += it->second.at(id).quantity;
629 it->second.erase(id);
630 }
631 accumulate("expired", q, value);
632 cardEntry(code, "expire", w, -q, -value);
633 totalQ += q;
634 totalValue += value;
635 emit(2, code, w, "lots", expired.size(), "qty", q, "value", money(value));
636 }
637 }
638 emit(2, "total", "qty", totalQ, "value", money(totalValue));
639
640 } else if (cmd == "VALUE" || cmd == "AGING") {
641 vector<string> warehouses;
642 if (a.empty()) {
643 for (const auto& [w, contents] : stock) {
644 warehouses.push_back(w);
645 }
646 } else {
647 warehouse(a[0]);
648 warehouses.push_back(a[0]);
649 }
650 if (cmd == "VALUE") {
651 if (a.empty()) {
652 emit(0, "VALUE");
653 } else {
654 emit(0, "VALUE", a[0]);
655 }
656 } else if (a.empty()) {
657 emit(0, "AGING", "day", day);
658 } else {
659 emit(0, "AGING", a[0], "day", day);
660 }
661 Amount totalQ = 0;
662 Amount totalValue = 0;
663 vector<Amount> bucketQ(4, 0), bucketV(4, 0);
664 for (const auto& [code, entry] : items) {
665 Amount q = 0;
666 Amount value = 0;
667 vector<std::pair<Amount, Amount>> warehouseValues;
668 vector<Amount> itemBuckets(4, 0);
669 for (const string& w : warehouses) {
670 Amount heldQ = 0;
671 Amount heldValue = 0;
672 for (const auto& [id, current] : holdings(w, code)) {
673 heldQ += current.quantity;
674 heldValue += current.quantity * current.cost;
675 if (cmd == "AGING") {
676 Amount age = day - lots[id - 1].day;
677 size_t bucket = age <= 30 ? 0 : age <= 60 ? 1 : age <= 90 ? 2 : 3;
678 itemBuckets[bucket] += current.quantity;
679 bucketQ[bucket] += current.quantity;
680 bucketV[bucket] += current.quantity * current.cost;
681 }
682 }
683 q += heldQ;
684 value += heldValue;
685 warehouseValues.push_back({heldQ, heldValue});
686 }
687 if (q == 0) {
688 continue;
689 }
690 if (cmd == "VALUE") {
691 emit(2, code, "qty", q, "value", money(value), "avg", money(value / q));
692 if (a.empty()) {
693 for (size_t i = 0; i < warehouses.size(); ++i) {
694 if (warehouseValues[i].first > 0) {
695 emit(4, warehouses[i], "qty", warehouseValues[i].first,
696 "value", money(warehouseValues[i].second));
697 }
698 }
699 }
700 } else {
701 agingLine(code, itemBuckets, false);
702 }
703 totalQ += q;
704 totalValue += value;
705 }
706 if (cmd == "VALUE") {
707 emit(2, "total", "qty", totalQ, "value", money(totalValue));
708 } else {
709 agingLine("total", bucketQ, false);
710 agingLine("value", bucketV, true);
711 }
712
713 } else if (cmd == "LOTS") {
714 const string& w = a[0];
715 const string& code = a[1];
716 pair(w, code);
717 emit(0, "LOTS", w, code);
718 vector<int> ids = fifo(w, code);
719 vector<const Order*> pending;
720 for (const Order& current : orders) {
721 if (current.warehouse == w && current.item == code && current.backorder > 0) {
722 pending.push_back(&current);
723 }
724 }
725 for (int id : ids) {
726 const StockLot& current = holdings(w, code).at(id);
727 emit(2, "lot", id, "day", lots[id - 1].day,
728 "qty", current.quantity, "cost", money(current.cost),
729 "value", money(current.quantity * current.cost),
730 "age", day - lots[id - 1].day);
731 }
732 for (const Order* current : pending) {
733 emit(2, "backorder", "order", current->number, "qty", current->backorder);
734 }
735 if (ids.empty() && pending.empty()) {
736 emit(2, "(none)");
737 }
738
739 } else if (cmd == "REORDER" || cmd == "ITEMS") {
740 emit(0, cmd);
741 bool found = false;
742 for (const auto& [code, entry] : items) {
743 Amount q = totalQuantity(code);
744 Amount pending = backorder(code);
745 if (cmd == "REORDER") {
746 if (q - pending > entry.level) {
747 continue;
748 }
749 emit(2, code, entry.name, "onhand", q,
750 "backorder", pending, "level", entry.level);
751 } else {
752 emit(2, code, entry.name, "level", entry.level,
753 "onhand", q, "backorder", pending,
754 "cost", entry.lastCost ? money(*entry.lastCost) : "none");
755 }
756 found = true;
757 }
758 if (!found) {
759 emit(2, "(none)");
760 }
761
762 } else if (cmd == "MARGIN" || cmd == "TOP") {
763 auto rows = sales();
764 if (cmd == "MARGIN") {
765 emit(0, "MARGIN");
766 Sale total;
767 for (const auto& [code, row] : rows) {
768 marginLine(code, row);
769 total.quantity += row.quantity;
770 total.revenue += row.revenue;
771 total.cogs += row.cogs;
772 }
773 marginLine("total", total);
774 } else {
775 Amount k = number(a[0]);
776 emit(0, "TOP", k);
777 vector<string> codes;
778 for (const auto& [code, row] : rows) {
779 codes.push_back(code);
780 }
781 sort(codes.begin(), codes.end(), [&](const string& left, const string& right) {
782 const Sale& a = rows.at(left);
783 const Sale& b = rows.at(right);
784 Amount leftProfit = a.revenue - a.cogs;
785 Amount rightProfit = b.revenue - b.cogs;
786 if (leftProfit != rightProfit) {
787 return leftProfit > rightProfit;
788 }
789 if (a.revenue != b.revenue) {
790 return a.revenue > b.revenue;
791 }
792 return left < right;
793 });
794 for (size_t i = 0; i < codes.size() && i < static_cast<size_t>(k); ++i) {
795 const Sale& row = rows.at(codes[i]);
796 Amount profit = row.revenue - row.cogs;
797 emit(2, i + 1, codes[i], "profit", money(profit),
798 "revenue", money(row.revenue), "margin", margin(profit, row.revenue));
799 }
800 if (codes.empty()) {
801 emit(2, "(none)");
802 }
803 }
804
805 } else if (cmd == "BACKORDERS") {
806 emit(0, "BACKORDERS");
807 bool found = false;
808 for (const Order& current : orders) {
809 if (current.backorder == 0) {
810 continue;
811 }
812 emit(2, "order", current.number, current.warehouse, current.item, "qty",
813 current.backorder, "day", current.day, "price", money(current.price));
814 found = true;
815 }
816 if (!found) {
817 emit(2, "(none)");
818 }
819
820 } else if (cmd == "ORDER") {
821 Order& current = order(a[0]);
822 emit(0, "ORDER", current.number, current.warehouse, current.item,
823 "day", current.day, "price", money(current.price));
824 emit(2, "ordered", current.ordered, "shipped", current.shipped,
825 "backorder", current.backorder, "cancelled", current.cancelled,
826 "returned", current.returned);
827 for (const Part& part : current.issues) {
828 emit(2, "lot", part.lot, "qty", part.quantity, "cost", money(part.cost));
829 }
830 emit(2, "revenue", money(current.revenue), "cogs",
831 money(current.cogs), "profit", money(current.revenue - current.cogs));
832
833 } else if (cmd == "TRACE") {
834 Lot& current = lot(a[0]);
835 emit(0, "TRACE", "lot", current.number, current.item, current.kind,
836 current.warehouse, "day", current.day, "qty", current.initialQuantity,
837 "cost", money(current.initialCost));
838 for (const auto& [w, contents] : stock) {
839 auto entry = holdings(w, current.item).find(current.number);
840 if (entry != holdings(w, current.item).end()) {
841 emit(2, "held", w, "qty", entry->second.quantity,
842 "cost", money(entry->second.cost));
843 }
844 }
845 for (const Order& currentOrder : orders) {
846 Amount q = 0;
847 for (const Part& part : currentOrder.issues) {
848 if (part.lot == current.number) {
849 q += part.quantity;
850 }
851 }
852 if (q > 0) {
853 emit(2, "order", currentOrder.number, "qty", q);
854 }
855 }
856 if (current.removed > 0) {
857 emit(2, "removed", "qty", current.removed);
858 }
859
860 } else if (cmd == "CARD") {
861 Item& entry = item(a[0]);
862 emit(0, "CARD", entry.code, entry.name);
863 Amount q = 0;
864 Amount value = 0;
865 for (const CardEntry& card : entry.card) {
866 q += card.quantity;
867 value += card.value;
868 emit(2, "day", card.day, card.kind, card.warehouse,
869 signedQuantity(card.quantity), signedMoney(card.value),
870 "balance", q, money(value));
871 }
872 if (entry.card.empty()) {
873 emit(2, "(none)");
874 }
875
876 } else if (cmd == "STATUS") {
877 emit(0, "STATUS");
878 for (const auto& [w, contents] : stock) {
879 Amount skus = 0;
880 Amount q = 0;
881 Amount value = 0;
882 for (const auto& [code, held] : contents) {
883 if (!held.empty()) {
884 skus++;
885 }
886 for (const auto& [id, current] : held) {
887 q += current.quantity;
888 value += current.quantity * current.cost;
889 }
890 }
891 Amount pending = 0;
892 for (const Order& current : orders) {
893 if (current.warehouse == w) {
894 pending += current.backorder;
895 }
896 }
897 emit(2, w, "skus", skus, "qty", q, "value", money(value),
898 "backorder", pending);
899 }
900 if (stock.empty()) {
901 emit(2, "(none)");
902 }
903
904 } else if (cmd == "SUMMARY") {
905 emit(0, "SUMMARY");
906 for (const string& category : categories) {
907 const Total& total = totals[category];
908 emit(2, category, "qty", total.quantity, "value", money(total.value));
909 }
910 Total total = allStocks();
911 emit(2, "on-hand", "qty", total.quantity, "value", money(total.value));
912 Amount open = 0;
913 Amount pending = 0;
914 for (const Order& current : orders) {
915 if (current.backorder > 0) {
916 open++;
917 }
918 pending += current.backorder;
919 }
920 emit(2, "orders", orders.size(), "open", open, "backorder", pending);
921 }
922 }

924public:
925 Inventory() {
926 for (const string& category : categories) {
927 totals[category] = {};
928 }
929 }

931 void execute(const string& line) {
932 istringstream input(line);
933 string cmd;
934 input >> cmd;
935 vector<string> args;
936 string token;
937 while (input >> token) {
938 args.push_back(token);
939 }
940 try {
941 command(cmd, args);
942 } catch (const runtime_error& exception) {
943 emit(0, "ERROR", exception.what());
944 }
945 }
946
947 void print() const {
948 for (const string& line : output) {
949 cout << line << '\n';
950 }
951 }
952};
