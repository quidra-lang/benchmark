4enum Fault: Error { case reason(String) }
5func error(_ condition: Bool, _ reason: String) throws {
6 if condition { throw Fault.reason(reason) }
7}
8func parseMoney(_ s: String) -> Int {
9 Int(s.replacingOccurrences(of:".",with:""))!
10}
11func money(_ n: Int) -> String {
12 String(format:"%@%ld.%02ld",locale:Locale(identifier:"en_US_POSIX"),n < 0 ? "-" : "",abs(n)/100,abs(n)%100)
13}
14func signed(_ n: Int) -> String { (n > 0 ? "+" : "")+String(n) }
15func signedMoney(_ n: Int) -> String { (n > 0 ? "+" : "")+money(n) }
16func margin(_ profit: Int, _ revenue: Int) -> String {
17 if revenue == 0 { return "n/a" }
18 let q = abs(profit)*1000/revenue
19 return (profit < 0 && q > 0 ? "-" : "")+"\(q/10).\(q%10)%"
20}

108final class Inventory {
109 var day = 1
110 var warehouses: [String:Warehouse] = [:]
111 var items: [String:Item] = [:]
112 var lots: [Lot] = []
113 var orders: [Order] = []
114 var totals = Array(repeating:[0,0],count:8)
115 var output: [String] = []
116 var sortedWarehouses: [Warehouse] { warehouses.keys.sorted().map { warehouses[$0]! } }
117 var sortedItems: [Item] { items.keys.sorted().map { items[$0]! } }
118 func warehouse(_ code: String) throws -> Warehouse {
119 guard let w = warehouses[code] else { throw Fault.reason("unknown warehouse "+code) }
120 return w
121 }
122 func item(_ code: String) throws -> Item {
123 guard let s = items[code] else { throw Fault.reason("unknown item "+code) }
124 return s
125 }
126 func order(_ raw: String) throws -> Order {
127 let n = Int(raw)!
128 try error(n < 1 || n > orders.count,"unknown order "+raw)
129 return orders[n-1]
130 }
131 func lookupLot(_ raw: String) throws -> Lot {
132 let n = Int(raw)!
133 try error(n < 1 || n > lots.count,"unknown lot "+raw)
134 return lots[n-1]
135 }
136 func stock(_ w: Warehouse, _ s: Item) -> [Stock] { w.stock[s.code].map { Array($0.values) } ?? [] }
137 func fifo(_ w: Warehouse, _ s: Item) -> [Stock] {
138 stock(w,s).sorted { $0.lot.day == $1.lot.day ? $0.lot.number < $1.lot.number : $0.lot.day < $1.lot.day }
139 }
140 func onhand(_ w: Warehouse, _ s: Item) -> Int { stock(w,s).reduce(0) { $0+$1.quantity } }
141 func stockValue(_ w: Warehouse, _ s: Item) -> Int { stock(w,s).reduce(0) { $0+$1.value } }
142 func onhand(_ s: Item) -> Int { warehouses.values.reduce(0) { $0+onhand($1,s) } }
143 func backorder(_ s: Item) -> Int { orders.filter { $0.item === s }.reduce(0) { $0+$1.backorder } }
144 func target(_ s: Item, _ selected: Warehouse?) -> [Stock] {
145 sortedWarehouses.filter { selected == nil || $0 === selected! }.flatMap { stock($0,s) }
146 }
147 func card(_ s: Item, _ kind: String, _ w: Warehouse, _ q: Int, _ v: Int) {
148 s.cards.append(Card(day:day,kind:kind,warehouse:w.code,quantity:q,value:v))
149 }
150 func total(_ category: Int, _ q: Int, _ value: Int) {
151 totals[category][0] += q
152 totals[category][1] += value
153 }
154 func add(_ w: Warehouse, _ s: Item, _ lot: Lot, _ q: Int, _ cost: Int) {
155 if let old = w.stock[s.code]?[lot.number] { old.quantity += q }
156 else { w.stock[s.code,default:[:]][lot.number] = Stock(lot,q,cost) }
157 }
158 func create(_ w: Warehouse, _ s: Item, _ q: Int, _ cost: Int, _ date: Int, _ kind: String) -> Lot {
159 let lot = Lot(lots.count+1,s,date,q,cost,w.code,kind)
160 lots.append(lot)
161 add(w,s,lot,q,cost)
162 return lot
163 }
164 func issue(_ w: Warehouse, _ s: Item, _ quantity: Int, _ removed: Bool) -> Issue {
165 var parts: [Part] = []
166 var left = quantity
167 var cost = 0
168 for a in fifo(w,s) {
169 if left == 0 { break }
170 let q = min(left,a.quantity)
171 parts.append(Part(lot:a.lot,quantity:q,cost:a.cost))
172 cost += q*a.cost
173 a.quantity -= q
174 left -= q
175 if removed { a.lot.removed += q }
176 if a.quantity == 0 { w.stock[s.code]?.removeValue(forKey:a.lot.number) }
177 }
178 return Issue(parts:parts,cost:cost)
179 }
180 func fill(_ o: Order, _ q: Int, _ kind: String) -> Issue {
181 let issued = issue(o.warehouse,o.item,q,false)
182 o.shipped += q
183 o.backorder -= q
184 o.revenue += q*o.price
185 o.cogs += issued.cost
186 o.parts.append(contentsOf:issued.parts)
187 if q > 0 {
188 total(4,q,issued.cost)
189 card(o.item,kind,o.warehouse,-q,-issued.cost)
190 }
191 return issued
192 }
193 func allocate(_ w: Warehouse, _ s: Item) {
194 var available = onhand(w,s)
195 for o in orders {
196 if available == 0 { break }
197 if o.warehouse === w && o.item === s && o.backorder > 0 {
198 let q = min(available,o.backorder)
199 let issued = fill(o,q,"fill")
200 available -= q
201 output.append(" FILL order \(o.number) qty \(q) revenue \(money(q*o.price)) cogs \(money(issued.cost)) profit \(money(q*o.price-issued.cost)) backorder \(o.backorder)")
202 }
203 }
204 }
205 func insufficient(_ w: Warehouse, _ s: Item, _ q: Int) throws {
206 let h = onhand(w,s)
207 try error(h < q,"insufficient stock \(w.code) \(s.code) \(h)")
208 }
209 func sales(_ s: Item?) -> [Int] {
210 let selected = orders.filter { s == nil || $0.item === s! }
211 return [selected.reduce(0) { $0+$1.shipped-$1.returned },selected.reduce(0) { $0+$1.revenue },selected.reduce(0) { $0+$1.cogs },selected.count]
212 }
213 func salesLine(_ label: String, _ a: [Int]) -> String {
214 " \(label) sold \(a[0]) revenue \(money(a[1])) cogs \(money(a[2])) profit \(money(a[1]-a[2])) margin \(margin(a[1]-a[2],a[1]))"
215 }
216 func buckets(_ label: String, _ a: [Int], _ monetary: Bool) -> String {
217 let labels = ["0-30","31-60","61-90","91+"]
218 return " "+label+(0..<4).map { " \(labels[$0]) "+(monetary ? money(a[$0]) : String(a[$0])) }.joined()
219 }
220 func stockCommand(_ t: [String]) throws {
221 let w = try warehouse(t[1])
222 let s = try item(t[2])
223 let q = Int(t[0] == "ADJUST" ? String(t[3].dropFirst()) : t[3])!
224 if t[0] != "COUNT" { try error(q == 0,"invalid quantity") }
225 let h = onhand(w,s)
226 switch t[0] {
227 case "RECEIVE":
228 let date = Int(t[5])!
229 let cost = parseMoney(t[4])
230 try error(date > day,"future day "+t[5])
231 let lot = create(w,s,q,cost,date,"receive")
232 s.lastCost = cost
233 total(0,q,q*cost)
234 card(s,"receive",w,q,q*cost)
235 output.append("RECEIVE \(w.code) \(s.code) lot \(lot.number) qty \(q) cost \(money(cost)) day \(date) value \(money(q*cost))")
236 allocate(w,s)
237 case "SHIP":
238 let o = Order(orders.count+1,w,s,day,q,parseMoney(t[4]))
239 orders.append(o)
240 let f = min(q,h)
241 let issued = fill(o,f,"ship")
242 output.append("SHIP order \(o.number) \(w.code) \(s.code) shipped \(f) revenue \(money(o.revenue)) cogs \(money(issued.cost)) profit \(money(o.revenue-issued.cost)) backorder \(o.backorder)")
243 case "ADJUST":
244 if t[3].first == "+" {
245 let cost = parseMoney(t[4])
246 let lot = create(w,s,q,cost,day,"adjust")
247 total(2,q,q*cost)
248 card(s,"adjust",w,q,q*cost)
249 output.append("ADJUST \(w.code) \(s.code) +\(q) lot \(lot.number) value \(money(q*cost))")
250 allocate(w,s)
251 } else {
252 try insufficient(w,s,q)
253 let issued = issue(w,s,q,true)
254 total(3,q,issued.cost)
255 card(s,"adjust",w,-q,-issued.cost)
256 output.append("ADJUST \(w.code) \(s.code) -\(q) value \(money(issued.cost))")
257 }
258 case "COUNT":
259 if q == h { output.append("COUNT \(w.code) \(s.code) qty \(q) ok") }
260 else if q < h {
261 let issued = issue(w,s,h-q,true)
262 total(3,h-q,issued.cost)
263 card(s,"count",w,q-h,-issued.cost)
264 output.append("COUNT \(w.code) \(s.code) qty \(h) -> \(q) shortage \(h-q) value \(money(issued.cost))")
265 } else {
266 let cost = s.lastCost ?? 0
267 let lot = create(w,s,q-h,cost,day,"count")
268 total(2,q-h,(q-h)*cost)
269 card(s,"count",w,q-h,(q-h)*cost)
270 output.append("COUNT \(w.code) \(s.code) qty \(h) -> \(q) surplus \(q-h) lot \(lot.number) value \(money((q-h)*cost))")
271 allocate(w,s)
272 }
273 default: break
274 }
275 }
276 func execute(_ t: [String]) throws {
277 switch t[0] {
278 case "DAY":
279 let d = Int(t[1])!
280 try error(d <= day,"invalid day "+t[1])
281 day = d
282 output.append("DAY \(day)")
283 case "WAREHOUSE":
284 try error(warehouses[t[1]] != nil,"duplicate warehouse "+t[1])
285 warehouses[t[1]] = Warehouse(t[1])
286 output.append("WAREHOUSE "+t[1])
287 case "ITEM":
288 let existing = items[t[1]]
289 let s = existing ?? Item(t[1],Int(t[2])!,t[3])
290 s.level = Int(t[2])!
291 s.name = t[3]
292 items[s.code] = s
293 output.append("ITEM \(s.code)"+(existing == nil ? " added " : " updated ")+"\(s.name) level \(s.level)")
294 case "RECEIVE", "SHIP", "ADJUST", "COUNT": try stockCommand(t)
295 case "CANCEL":
296 let o = try order(t[1])
297 try error(o.backorder == 0,"no backorder "+t[1])
298 o.cancelled += o.backorder
299 o.backorder = 0
300 output.append("CANCEL order \(o.number) qty \(o.cancelled)")
301 case "TRANSFER":
302 let a = try warehouse(t[1])
303 let b = try warehouse(t[2])
304 try error(a === b,"same warehouse "+t[1])
305 let s = try item(t[3])
306 let q = Int(t[4])!
307 try error(q == 0,"invalid quantity")
308 try insufficient(a,s,q)
309 let issued = issue(a,s,q,false)
310 for p in issued.parts { add(b,s,p.lot,p.quantity,p.cost) }
311 card(s,"transfer-out",a,-q,-issued.cost)
312 card(s,"transfer-in",b,q,issued.cost)
313 output.append("TRANSFER \(a.code) -> \(b.code) \(s.code) qty \(q) value \(money(issued.cost)) lots \(issued.parts.count)")
314 allocate(b,s)
315 case "RETURN":
316 let o = try order(t[1])
317 let q = Int(t[2])!
318 let cost = parseMoney(t[3])
319 try error(q == 0,"invalid quantity")
320 try error(q > o.shipped-o.returned,"return exceeds shipped \(t[1]) \(o.shipped-o.returned)")
321 let lot = create(o.warehouse,o.item,q,cost,day,"return")
322 o.returned += q
323 o.revenue -= q*o.price
324 o.cogs -= q*cost
325 total(1,q,q*cost)
326 card(o.item,"return",o.warehouse,q,q*cost)
327 output.append("RETURN order \(o.number) \(o.warehouse.code) \(o.item.code) qty \(q) lot \(lot.number) refund \(money(q*o.price)) value \(money(q*cost))")
328 allocate(o.warehouse,o.item)
329 case "MARKET":
330 let s = try item(t[1])
331 let price = parseMoney(t[2])
332 var quantity = 0
333 var loss = 0
334 var count = 0
335 for w in sortedWarehouses {
336 var subtotal = 0
337 for a in stock(w,s) where a.cost > price {
338 subtotal += a.quantity*(a.cost-price)
339 quantity += a.quantity
340 count += 1
341 a.cost = price
342 }
343 if subtotal > 0 { card(s,"writedown",w,0,-subtotal) }
344 loss += subtotal
345 }
346 total(5,quantity,loss)
347 output.append("MARKET \(s.code) price \(money(price)) lots \(count) qty \(quantity) loss \(money(loss))")
348 case "RECALL":
349 let lot = try lookupLot(t[1])
350 let held = sortedWarehouses.compactMap { w in stock(w,lot.item).first { $0.lot === lot }.map { (w,$0) } }
351 try error(held.isEmpty,"nothing to recall "+t[1])
352 let quantity = held.reduce(0) { $0+$1.1.quantity }
353 let value = held.reduce(0) { $0+$1.1.value }
354 for (w,a) in held {
355 card(lot.item,"recall",w,-a.quantity,-a.value)
356 w.stock[lot.item.code]?.removeValue(forKey:lot.number)
357 }
358 lot.removed += quantity
359 total(7,quantity,value)
360 output.append("RECALL lot \(lot.number) \(lot.item.code) qty \(quantity) value \(money(value))")
361 for (w,a) in held { output.append(" \(w.code) qty \(a.quantity) value \(money(a.value))") }
362 case "EXPIRE":
363 let age = Int(t[1])!
364 var quantity = 0
365 var value = 0
366 output.append("EXPIRE \(age)")
367 for s in sortedItems {
368 for w in sortedWarehouses {
369 let expired = stock(w,s).filter { day-$0.lot.day > age }
370 if expired.isEmpty { continue }
371 let q = expired.reduce(0) { $0+$1.quantity }
372 let v = expired.reduce(0) { $0+$1.value }
373 for a in expired {
374 a.lot.removed += a.quantity
375 w.stock[s.code]?.removeValue(forKey:a.lot.number)
376 }
377 total(6,q,v)
378 card(s,"expire",w,-q,-v)
379 output.append(" \(s.code) \(w.code) lots \(expired.count) qty \(q) value \(money(v))")
380 quantity += q
381 value += v
382 }
383 }
384 output.append(" total qty \(quantity) value \(money(value))")
385 case "VALUE", "AGING":
386 let selected = try t.count == 2 ? warehouse(t[1]) : nil
387 if t[0] == "VALUE" {
388 output.append("VALUE"+(selected.map { " "+$0.code } ?? ""))
389 var quantity = 0
390 var value = 0
391 for s in sortedItems {
392 let stock = target(s,selected)
393 let q = stock.reduce(0) { $0+$1.quantity }
394 let v = stock.reduce(0) { $0+$1.value }
395 if q == 0 { continue }
396 output.append(" \(s.code) qty \(q) value \(money(v)) avg \(money(v/q))")
397 if selected == nil {
398 for w in sortedWarehouses where onhand(w,s) > 0 { output.append(" \(w.code) qty \(onhand(w,s)) value \(money(stockValue(w,s)))") }
399 }
400 quantity += q
401 value += v
402 }
403 output.append(" total qty \(quantity) value \(money(value))")
404 } else {
405 output.append("AGING"+(selected.map { " "+$0.code } ?? "")+" day \(day)")
406 var quantities = Array(repeating:0,count:4)
407 var values = Array(repeating:0,count:4)
408 for s in sortedItems {
409 var q = Array(repeating:0,count:4)
410 for a in target(s,selected) {
411 let age = day-a.lot.day
412 let b = age <= 30 ? 0 : age <= 60 ? 1 : age <= 90 ? 2 : 3
413 q[b] += a.quantity
414 quantities[b] += a.quantity
415 values[b] += a.value
416 }
417 if q.reduce(0,+) > 0 { output.append(buckets(s.code,q,false)) }
418 }
419 output.append(buckets("total",quantities,false))
420 output.append(buckets("value",values,true))
421 }
422 case "LOTS":
423 let w = try warehouse(t[1])
424 let s = try item(t[2])
425 output.append("LOTS \(w.code) \(s.code)")
426 for a in fifo(w,s) { output.append(" lot \(a.lot.number) day \(a.lot.day) qty \(a.quantity) cost \(money(a.cost)) value \(money(a.value)) age \(day-a.lot.day)") }
427 for o in orders where o.warehouse === w && o.item === s && o.backorder > 0 { output.append(" backorder order \(o.number) qty \(o.backorder)") }
428 case "REORDER", "ITEMS":
429 output.append(t[0])
430 for s in sortedItems {
431 let h = onhand(s)
432 let b = backorder(s)
433 if t[0] == "ITEMS" { output.append(" \(s.code) \(s.name) level \(s.level) onhand \(h) backorder \(b) cost "+(s.lastCost.map(money) ?? "none")) }
434 else if h-b <= s.level { output.append(" \(s.code) \(s.name) onhand \(h) backorder \(b) level \(s.level)") }
435 }
436 case "MARGIN", "TOP":
437 let summaries = Dictionary(uniqueKeysWithValues:sortedItems.map { ($0.code,sales($0)) }).filter { $0.value[3] > 0 }
438 if t[0] == "MARGIN" {
439 output.append("MARGIN")
440 for s in sortedItems where summaries[s.code] != nil { output.append(salesLine(s.code,summaries[s.code]!)) }
441 output.append(salesLine("total",sales(nil)))
442 } else {
443 let k = Int(t[1])!
444 output.append("TOP \(k)")
445 let ranked = summaries.keys.sorted { a,b in
446 let x = summaries[a]!
447 let y = summaries[b]!
448 if x[1]-x[2] != y[1]-y[2] { return x[1]-x[2] > y[1]-y[2] }
449 return x[1] == y[1] ? a < b : x[1] > y[1]
450 }.prefix(k)
451 for (i,code) in ranked.enumerated() {
452 let a = summaries[code]!
453 output.append(" \(i+1) \(code) profit \(money(a[1]-a[2])) revenue \(money(a[1])) margin \(margin(a[1]-a[2],a[1]))")
454 }
455 }
456 case "BACKORDERS":
457 output.append("BACKORDERS")
458 for o in orders where o.backorder > 0 { output.append(" order \(o.number) \(o.warehouse.code) \(o.item.code) qty \(o.backorder) day \(o.day) price \(money(o.price))") }
459 case "ORDER":
460 let o = try order(t[1])
461 output.append("ORDER \(o.number) \(o.warehouse.code) \(o.item.code) day \(o.day) price \(money(o.price))")
462 output.append(" ordered \(o.ordered) shipped \(o.shipped) backorder \(o.backorder) cancelled \(o.cancelled) returned \(o.returned)")
463 for p in o.parts { output.append(" lot \(p.lot.number) qty \(p.quantity) cost \(money(p.cost))") }
464 output.append(" revenue \(money(o.revenue)) cogs \(money(o.cogs)) profit \(money(o.revenue-o.cogs))")
465 case "TRACE":
466 let lot = try lookupLot(t[1])
467 output.append("TRACE lot \(lot.number) \(lot.item.code) \(lot.kind) \(lot.warehouse) day \(lot.day) qty \(lot.initialQuantity) cost \(money(lot.initialCost))")
468 for w in sortedWarehouses {
469 for a in stock(w,lot.item) where a.lot === lot { output.append(" held \(w.code) qty \(a.quantity) cost \(money(a.cost))") }
470 }
471 for o in orders {
472 let q = o.parts.filter { $0.lot === lot }.reduce(0) { $0+$1.quantity }
473 if q > 0 { output.append(" order \(o.number) qty \(q)") }
474 }
475 if lot.removed > 0 { output.append(" removed qty \(lot.removed)") }
476 case "CARD":
477 let s = try item(t[1])
478 output.append("CARD \(s.code) \(s.name)")
479 var quantity = 0
480 var value = 0
481 for c in s.cards {
482 quantity += c.quantity
483 value += c.value
484 output.append(" day \(c.day) \(c.kind) \(c.warehouse) \(signed(c.quantity)) \(signedMoney(c.value)) balance \(quantity) \(money(value))")
485 }
486 case "STATUS":
487 output.append("STATUS")
488 for w in sortedWarehouses {
489 let q = items.values.reduce(0) { $0+onhand(w,$1) }
490 let v = items.values.reduce(0) { $0+stockValue(w,$1) }
491 let skus = items.values.filter { onhand(w,$0) > 0 }.count
492 let b = orders.filter { $0.warehouse === w }.reduce(0) { $0+$1.backorder }
493 output.append(" \(w.code) skus \(skus) qty \(q) value \(money(v)) backorder \(b)")
494 }
495 case "SUMMARY":
496 output.append("SUMMARY")
497 for i in 0..<8 { output.append(" \(categories[i]) qty \(totals[i][0]) value \(money(totals[i][1]))") }
498 let q = items.values.reduce(0) { $0+onhand($1) }
499 let v = items.values.reduce(0) { a,s in a+warehouses.values.reduce(0) { $0+stockValue($1,s) } }
500 output.append(" on-hand qty \(q) value \(money(v))")
501 output.append(" orders \(orders.count) open \(orders.filter { $0.backorder > 0 }.count) backorder \(orders.reduce(0) { $0+$1.backorder })")
502 default: break
503 }
504 if output.count == 1 && ["LOTS","REORDER","ITEMS","TOP","BACKORDERS","CARD","STATUS"].contains(t[0]) { output.append(" (none)") }
505 }
506 func process(_ line: String) -> [String] {
507 output = []
508 do { try execute(line.split(separator:" ").map(String.init)) }
509 catch Fault.reason(let reason) { output.append("ERROR "+reason) }
510 catch { fatalError("Unexpected error") }
511 return output
512 }
