97def escaped_read(text, position, stops):
98 chars = []
99 while position < len(text) and text[position] not in stops:
100 if text[position] == "\\" and text[position + 1:position + 2] in PUNCTUATION:
101 position += 1
102 chars.append(text[position])
103 position += 1
104 return "".join(chars), position

107def opening_fence(line):
108 match = FENCE.match(line)
109 if match is None:
110 return None
111 fence = match.group()
112 info = line[len(fence):].strip(" ")
113 if fence[0] == "`" and "`" in info:
114 return None
115 return fence, info

118def atx_heading(line):
119 match = ATX_HEADING.match(line)
120 if match is None:
121 return None
122 level = len(match.group(1))
123 text = line[level:].strip(" ")
124 trimmed = text.rstrip("#")
125 if trimmed == "" or trimmed.endswith(" "):
126 text = trimmed.strip(" ")
127 return Heading(level, text)

130def is_rule(line):
131 return RULE.fullmatch(line.replace(" ", "")) is not None

134def list_marker(line):
135 match = LIST_ITEM.match(line)
136 if match is None or is_rule(line):
137 return None
138 bullet, digits, delimiter = match.groups()
139 number = int(digits) if digits else None

143def interrupts_paragraph(line):
144 return not line.startswith(" ") and (

147 or is_rule(line)
148 or line.startswith(">")

153def indented_code(lines, index):
154 end = index
155 while end < len(lines) and (lines[end] == "" or lines[end].startswith(" ")):
156 end += 1
157 while lines[end - 1] == "":
158 end -= 1
159 return CodeBlock("", [line[4:] for line in lines[index:end]]), end

162def fenced_code(lines, index, fence, info):
163 end = index + 1
164 while end < len(lines):
165 line = lines[end]
166 if len(line) >= len(fence) and line.strip(fence[0]) == "":
167 break
168 end += 1
169 language = info.partition(" ")[0]
170 return CodeBlock(language, lines[index + 1:end]), end + 1

173def split_cells(line):
174 tokens = CELL_TOKEN.findall(line.strip(" ").removeprefix("|"))
175 if tokens and tokens[-1] == "|":
176 tokens.pop()
177 cells = [""]
178 for token in tokens:
179 if token == "|":
180 cells.append("")
181 else:
182 cells[-1] += "|" if token == "\\|" else token
183 return [cell.strip(" ") for cell in cells]

186def parse_table(lines, index):
187 if index + 1 >= len(lines) or "|" not in lines[index] or "|" not in lines[index + 1]:
188 return None
189 header = split_cells(lines[index])
190 delimiters = split_cells(lines[index + 1])
191 if len(header) != len(delimiters):
192 return None
193 if not all(DELIMITER_CELL.fullmatch(cell) for cell in delimiters):
194 return None
195 alignments = [ALIGNMENTS[cell.startswith(":"), cell.endswith(":")] for cell in delimiters]
196 columns = len(delimiters)
197 rows = []
198 end = index + 2
199 while end < len(lines) and "|" in lines[end]:
200 cells = split_cells(lines[end])[:columns]
201 rows.append(cells + [""] * (columns - len(cells)))
202 end += 1
203 return Table(alignments, header, rows), end
204
205
206def paragraph(lines, index):
207 end = index + 1
208 level = None
209 while end < len(lines) and lines[end] != "":
210 line = lines[end]
211 if UNDERLINE.fullmatch(line):
212 level = 1 if line[0] == "=" else 2
213 break
214 if interrupts_paragraph(line):
215 break
216 end += 1
217 text = "\n".join(line.lstrip(" ") for line in lines[index:end])
218 if level is not None:
219 return Heading(level, text), end + 1
220 if text == "[TOC]":
221 return TableOfContents(), end
222 return Paragraph(text), end

225class BlockParser:
226 def __init__(self):
227 self.references = {}
228 self.footnotes = {}
229
230 def parse(self, lines):
231 blocks = []
232 index = 0
233 while index < len(lines):
234 if lines[index] == "":
235 index += 1
236 continue
237 block, index = self.next_block(lines, index)
238 if block is not None:
239 blocks.append(block)
240 return blocks
241
242 def next_block(self, lines, index):
243 line = lines[index]
244 if line.startswith(" "):
245 return indented_code(lines, index)
246 if line.startswith(" "):
247 return paragraph(lines, index)
248 if fence := opening_fence(line):
249 return fenced_code(lines, index, *fence)
250 if heading := atx_heading(line):
251 return heading, index + 1
252 if is_rule(line):
253 return HorizontalRule(), index + 1
254 if line.startswith(">"):
255 return self.quote(lines, index)
256 if marker := list_marker(line):
257 return self.list_block(lines, index, marker)
258 if self.footnote_definition(line) or self.link_definition(line):
259 return None, index + 1
260 return parse_table(lines, index) or paragraph(lines, index)
261
262 def quote(self, lines, index):
263 content = []
264 while index < len(lines) and lines[index].startswith(">"):
265 content.append(lines[index][1:].removeprefix(" "))
266 index += 1
267 if alert := ALERT.fullmatch(content[0]):
268 return Alert(alert.group(1), self.parse(content[1:])), index
269 return Quote(self.parse(content)), index
270
271 def list_block(self, lines, index, marker):
272 kind = marker.kind
273 start = marker.number
274 items = []
275 while True:
276 content = [marker.content]
277 index += 1
278 while index < len(lines) and lines[index].startswith(" "):
279 content.append(lines[index][2:])
280 index += 1
281 items.append(self.parse(content))
282 following = index
283 while following < len(lines) and lines[following] == "":
284 following += 1
285 marker = list_marker(lines[following]) if following < len(lines) else None
286 if marker is None or marker.kind != kind:
287 return ListBlock(start is not None, start, items), index
288 index = following

290 def footnote_definition(self, line):
291 match = FOOTNOTE_DEFINITION.match(line)
292 if match is None:
293 return False
294 body = match.group(2).strip(" ")
295 if body == "":
296 return False
297 self.footnotes.setdefault(match.group(1).lower(), body)
298 return True
299
300 def link_definition(self, line):
301 if not line.startswith("["):
302 return False
303 close = line.find("]")
304 if close < 0 or not line.startswith(":", close + 1):
305 return False
306 label = line[1:close]
307 if "[" in label or label.strip(" ") == "" or label.startswith("^"):
308 return False
309 url, end = escaped_read(line, skip_whitespace(line, close + 2), " ")
310 if url == "":
311 return False
312 position = skip_whitespace(line, end)
313 title = None
314 if position < len(line):
315 if position == end or line[position] not in "\"'":
316 return False
317 title, end = escaped_read(line, position + 1, line[position])
318 if end != len(line) - 1:
319 return False
320 self.references.setdefault(normalize_label(label), Reference(url, title))
321 return True

324def backtick_closer(text, start, length):
325 for match in BACKTICKS.finditer(text, start):
326 if len(match.group()) == length:
327 return match.start()
328 return -1
329
330
331def backtick_length(text, position):
332 return BACKTICKS.match(text, position).end() - position
333
334
335def code_span(text, position):
336 length = backtick_length(text, position)
337 closer = backtick_closer(text, position + length, length)
338 if closer < 0:
339 return "`" * length, position + length
340 code = text[position + length:closer].replace("\n", " ")
341 if code.startswith(" ") and code.endswith(" ") and code.strip(" "):
342 code = code[1:-1]
343 return f"<code>{escape_html(code)}</code>", closer + length

346def autolink(text, position):
347 match = AUTOLINK.match(text, position)
348 if match is None:
349 return None
350 target = match.group(1)
351 if ANGLE_URL.match(target):
352 href = target
353 elif EMAIL.fullmatch(target):
354 href = "mailto:" + target
355 else:
356 return None
357 return f'<a href="{escape_html(href)}">{escape_html(target)}</a>', match.end()
358
359
360def url_autolink(text, position):
361 if position > 0 and text[position - 1] not in URL_PRECEDERS:
362 return None
363 match = BARE_URL.match(text, position)
364 if match is None:
365 return None
366 url = match.group().rstrip(URL_TRAILERS)
367 if url in ("http://", "https://"):
368 return None
369 return f'<a href="{escape_html(url)}">{escape_html(url)}</a>', position + len(url)

372def delimiter(text, position):
373 char = text[position]
374 end = position
375 while end < len(text) and text[end] == char:
376 end += 1
377 length = end - position
378 before = text[position - 1] if position > 0 else " "
379 after = text[end] if end < len(text) else " "
380 can_open = after not in WHITESPACE
381 can_close = before not in WHITESPACE
382 if char == "_" and before.isalnum() and after.isalnum():
383 can_open = can_close = False
384 if (char != "~" and length <= 3) or (char == "~" and length == 2):
385 return Delimiter(char, length, can_open, can_close), end
386 return char * length, end
387
388
389def find_closing_bracket(text, opening):
390 depth = 0
391 position = opening
392 while position < len(text):
393 char = text[position]
394 if char == "\\":
395 position += 2
396 continue
397 if char == "`":
398 length = backtick_length(text, position)
399 closer = backtick_closer(text, position + length, length)
400 position = position + length if closer < 0 else closer + length
401 continue
402 if char == "[":
403 depth += 1
404 elif char == "]":
405 depth -= 1
406 if depth == 0:
407 return position
408 position += 1
409 return -1

412def find_emphasis_closer(pieces, opener, end):
413 opening = pieces[opener]
414 for index in range(opener + 1, end):
415 piece = pieces[index]
416 if (
417 isinstance(piece, Delimiter)
418 and piece.can_close
419 and piece.char == opening.char
420 and piece.length == opening.length
421 ):
422 return index
423 return None
424
425
426def build_html(pieces, start, end):
427 parts = []
428 index = start
429 while index < end:
430 piece = pieces[index]
431 if isinstance(piece, str):
432 parts.append(piece)
433 elif piece.can_open and (closer := find_emphasis_closer(pieces, index, end)) is not None:
434 tags = STRIKETHROUGH_TAGS if piece.char == "~" else EMPHASIS_TAGS[piece.length]
435 parts.append(tags[0] + build_html(pieces, index + 1, closer) + tags[1])
436 index = closer
437 else:
438 parts.append(piece.char * piece.length)
439 index += 1
440 return "".join(parts)

443class InlineParser:
444 def __init__(self, renderer, text, links, footnotes):
445 self.renderer = renderer
446 self.text = text
447 self.links = links
448 self.footnotes = footnotes
449
450 def convert(self):
451 pieces = []
452 position = 0
453 while position < len(self.text):
454 piece, position = self.next_piece(position)
455 pieces.append(piece)
456 return build_html(pieces, 0, len(pieces))
457
458 def next_piece(self, position):
459 text = self.text
460 char = text[position]
461 following = text[position + 1:position + 2]
462 if char == "\\":
463 if following in PUNCTUATION:
464 return escape_html(following), position + 2
465 if following == "\n":
466 return "<br />\n", position + 2
467 return "\\", position + 1
468 if char == "`":
469 return code_span(text, position)
470 if self.links:
471 if char == "<" and (link := autolink(text, position)):
472 return link
473 if link := url_autolink(text, position):
474 return link
475 if char == "[" or (char == "!" and following == "["):
476 return self.bracket(position) or (char, position + 1)
477 if char in "*_~":
478 return delimiter(text, position)
479 return escape_html(char), position + 1
480
481 def bracket(self, position):
482 text = self.text
483 renderer = self.renderer
484 if self.footnotes and text.startswith("[^", position):
485 close = text.find("]", position + 2)
486 if close >= 0 and (name := text[position + 2:close].lower()) in renderer.footnotes:
487 return renderer.footnote_reference(name), close + 1
488 image = text[position] == "!"
489 if not image and not self.links:
490 return None
491 opening = position + 1 if image else position
492 closing = find_closing_bracket(text, opening)
493 if closing < 0:
494 return None
495 label = text[opening + 1:closing]
496 destination = self.destination(closing + 1, label)
497 if destination is None:
498 return None
499 reference, end = destination
500 url = escape_html(reference.url)
501 title = "" if reference.title is None else f' title="{escape_html(reference.title)}"'
502 content = renderer.inline(label, links=False, footnotes=False)
503 if image:
504 return f'<img src="{url}" alt="{strip_tags(content)}"{title} />', end
505 return f'<a href="{url}"{title}>{content}</a>', end
506
507 def destination(self, position, label):
508 text = self.text
509 references = self.renderer.references
510 if text.startswith("[", position):
511 close = text.find("]", position + 1)
512 if close >= 0 and "[" not in text[position + 1:close]:
513 name = normalize_label(text[position + 1:close] or label)
514 if name not in references:
515 return None
516 return references[name], close + 1
517 elif text.startswith("(", position) and (link := self.inline_destination(position + 1)):
518 return link
519 name = normalize_label(label)
520 if name in references:
521 return references[name], position
522 return None
523
524 def inline_destination(self, position):
525 text = self.text
526 url, end = escaped_read(text, skip_whitespace(text, position), WHITESPACE + ")")
527 position = skip_whitespace(text, end)
528 title = None
529 if position > end and text.startswith(('"', "'"), position):
530 title, end = escaped_read(text, position + 1, text[position])
531 if end == len(text):
532 return None
533 position = skip_whitespace(text, end + 1)
534 if text.startswith(")", position):
535 return Reference(url, title), position + 1
536 return None

559class Renderer:
560 def __init__(self, references, footnotes):
561 self.references = references
562 self.footnotes = footnotes
563 self.used_ids = set()
564 self.headings = []
565 self.footnote_numbers = {}
566 self.footnote_counts = {}
567
568 def inline(self, text, links=True, footnotes=True):
569 return InlineParser(self, text, links, footnotes).convert()
570
571 def render_document(self, blocks):
572 html = self.render_blocks(blocks).replace(TOC_MARKER, self.table_of_contents())
573 if self.footnote_numbers:
574 html += self.footnote_list()
575 return html
576
577 def render_blocks(self, blocks):
578 return "".join(self.render_block(block) + "\n" for block in blocks)
579
580 def render_block(self, block):
581 match block:
582 case Heading(level, text):
583 return self.render_heading(level, text)
584 case Paragraph(text):
585 return f"<p>{self.inline(text)}</p>"
586 case HorizontalRule():
587 return "<hr />"
588 case CodeBlock(language, lines):
589 attribute = f' class="language-{escape_html(language)}"' if language else ""
590 code = "".join(escape_html(line) + "\n" for line in lines)
591 return f"<pre><code{attribute}>{code}</code></pre>"
592 case Quote(blocks):
593 return f"<blockquote>\n{self.render_blocks(blocks)}</blockquote>"
594 case Alert(kind, blocks):
595 return (
596 f'<div class="alert alert-{kind.lower()}">\n'
597 f'<p class="alert-title">{kind.capitalize()}</p>\n'
598 f"{self.render_blocks(blocks)}</div>"
599 )
600 case ListBlock():
601 return self.render_list(block)
602 case Table():
603 return self.render_table(block)
604 case TableOfContents():
605 return TOC_MARKER
606
607 def render_heading(self, level, text):
608 start = text.rfind(" {#")
609 if text.endswith("}") and start >= 0 and LABEL.fullmatch(text[start + 3:-1]):
610 identifier = text[start + 3:-1]
611 text = text[:start].rstrip(" ")
612 else:
613 identifier = self.unique_id(heading_slug(text))
614 self.used_ids.add(identifier)
615 html = self.inline(text)
616 self.headings.append(TocEntry(level, identifier, strip_tags(html)))
617 return f'<h{level} id="{identifier}">{html}</h{level}>'
618
619 def unique_id(self, base):
620 identifier = base
621 suffix = 1
622 while identifier in self.used_ids:
623 identifier = f"{base}-{suffix}"
624 suffix += 1
625 return identifier
626
627 def render_list(self, block):
628 if not block.ordered:
629 opening, closing = "<ul>", "</ul>"
630 elif block.start == 1:
631 opening, closing = "<ol>", "</ol>"
632 else:
633 opening, closing = f'<ol start="{block.start}">', "</ol>"
634 items = "".join(self.render_item(item) + "\n" for item in block.items)
635 return f"{opening}\n{items}{closing}"
636
637 def render_item(self, blocks):
638 html = "<li>"
639 for position, block in enumerate(blocks):
640 if isinstance(block, Paragraph):
641 text = block.text
642 checkbox = TASK_BOXES.get(text[:4]) if position == 0 and len(text) > 4 else None
643 if checkbox:
644 html += f"{checkbox} {self.inline(text[4:])}"
645 else:
646 html += self.inline(text)
647 else:
648 if not html.endswith("\n"):
649 html += "\n"
650 html += self.render_block(block) + "\n"
651 return html + "</li>"
652
653 def render_table(self, table):
654 lines = ["<table>", "<thead>", "<tr>"]
655 lines += self.table_cells("th", table.alignments, table.header)
656 lines += ["</tr>", "</thead>"]
657 if table.rows:
658 lines.append("<tbody>")
659 for row in table.rows:
660 lines.append("<tr>")
661 lines += self.table_cells("td", table.alignments, row)
662 lines.append("</tr>")
663 lines.append("</tbody>")
664 lines.append("</table>")
665 return "\n".join(lines)
666
667 def table_cells(self, tag, alignments, cells):
668 html = []
669 for alignment, cell in zip(alignments, cells):
670 style = f' style="text-align:{alignment}"' if alignment else ""
671 html.append(f"<{tag}{style}>{self.inline(cell)}</{tag}>")
672 return html
673
674 def table_of_contents(self):
675 if not self.headings:
676 return '<nav class="toc">\n</nav>'
677 roots = []
678 parents = []
679 for heading in self.headings:
680 while parents and parents[-1].level >= heading.level:
681 parents.pop()
682 (parents[-1].children if parents else roots).append(heading)
683 parents.append(heading)
684 return f'<nav class="toc">\n{toc_list(roots)}\n</nav>'
685
686 def footnote_reference(self, name):
687 if name not in self.footnote_numbers:
688 self.footnote_numbers[name] = len(self.footnote_numbers) + 1
689 self.footnote_counts[name] = 0
690 number = self.footnote_numbers[name]
691 self.footnote_counts[name] += 1
692 reference_id = footnote_reference_id(number, self.footnote_counts[name])
693 return f'<sup id="{reference_id}"><a href="#fn-{number}">{number}</a></sup>'
694
695 def footnote_list(self):
696 lines = ['<section class="footnotes">', "<ol>"]
697 for name, number in self.footnote_numbers.items():
698 body = self.inline(self.footnotes[name], footnotes=False)
699 backlinks = " ".join(
700 f'<a href="#{footnote_reference_id(number, count)}">&#8617;</a>'
701 for count in range(1, self.footnote_counts[name] + 1)
702 )
703 lines.append(f'<li id="fn-{number}">{body} {backlinks}</li>')
704 lines += ["</ol>", "</section>"]
705 return "".join(line + "\n" for line in lines)
