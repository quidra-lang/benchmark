3PUNCTUATION = %q(!"#$%&'()*+,-./:;<=>?@[\]^_`{|}~)

9def escape_html(text)
10 text.gsub(/[&<>"]/, '&' => '&amp;', '<' => '&lt;', '>' => '&gt;', '"' => '&quot;')
11end
12
13def strip_tags(text) = text.gsub(/<[^>]*>/, '')
14def normalize(text) = text.downcase.gsub(/[ \n]+/, ' ').strip
15
16def skip_space(text, at)
17 at += 1 while at < text.size && WHITESPACE.include?(text[at])
18 at
19end

21def escaped_read(text, at, stops)
22 result = +''
23 while at < text.size && !stops.include?(text[at])
24 at += 1 if text[at] == '\\' && at + 1 < text.size && PUNCTUATION.include?(text[at + 1])
25 result << text[at]
26 at += 1
27 end
28 [result, at]
29end

31def opening_fence(line)
32 match = /\A(`{3,}|~{3,})/.match(line)
33 return nil unless match
34
35 fence = match[0]
36 info = line[fence.size..].strip
37 return nil if fence[0] == '`' && info.include?('`')
38
39 [fence, info]
40end

42def atx_heading(line)
43 match = /\A(\#{1,6})(?: |\z)/.match(line)
44 return nil unless match
45
46 text = line[match[1].size..].strip
47 trimmed = text.sub(/\#+\z/, '')
48 text = trimmed.strip if trimmed.empty? || trimmed.end_with?(' ')
49 Block.new(kind: :heading, text: text, number: match[1].size)
50end
51
52def rule?(line) = /\A(?:-{3,}|\*{3,}|_{3,})\z/.match?(line.delete(' '))
53
54def list_marker(line)
55 return nil if rule?(line)
56
57 match = /\A(?:([-*+])|([0-9]{1,9})([.)])) +/.match(line)
58 return nil unless match
59
60 [match[1] || match[3], !match[2].nil?, match[2]&.to_i, line[match[0].size..]]
61end
62
63def interrupts?(line)
64 !line.start_with?(' ') && (opening_fence(line) || atx_heading(line) || rule?(line) || line.start_with?('>') || list_marker(line))
65end

67def split_cells(line)
68 text = line.strip.delete_prefix('|')
69 cells = [+'']
70 at = 0
71 separator = false
72 while at < text.size
73 char = text[at]
74 at += 1
75 separator = false
76 if char == '\\' && at < text.size
77 cells[-1] << '\\' unless text[at] == '|'
78 cells[-1] << text[at]
79 at += 1
80 elsif char == '|'
81 cells << +''
82 separator = true
83 else
84 cells[-1] << char
85 end
86 end
87 cells.pop if separator
88 cells.map(&:strip)
89end

99 def parse(lines)
100 blocks = []
101 at = 0
102 while at < lines.size
103 line = lines[at]
104 if line.empty?
105 at += 1
106 next
107 end
108 if line.start_with?(' ')
109 finish = at
110 finish += 1 while finish < lines.size && (lines[finish].empty? || lines[finish].start_with?(' '))
111 finish -= 1 while lines[finish - 1].empty?
112 blocks << Block.new(kind: :code, text: '', lines: lines[at...finish].map { |row| row[4..] || '' })
113 at = finish
114 next
115 end
116 unless line.start_with?(' ')
117 if (fence = opening_fence(line))
118 marker, info = fence
119 code = []
120 at += 1
121 while at < lines.size
122 row = lines[at]
123 at += 1
124 break if row.size >= marker.size && row.delete(marker[0]).empty?
125
126 code << row
127 end
128 blocks << Block.new(kind: :code, text: info.split(' ').first || '', lines: code)
129 next
130 end
131 if (heading = atx_heading(line))
132 blocks << heading
133 at += 1
134 next
135 end
136 if rule?(line)
137 blocks << Block.new(kind: :rule)
138 at += 1
139 next
140 end
141 if line.start_with?('>')
142 content = []
143 while at < lines.size && lines[at].start_with?('>')
144 content << lines[at][1..].delete_prefix(' ')
145 at += 1
146 end
147 alert = /\A\[!(NOTE|TIP|IMPORTANT|WARNING|CAUTION)\]\z/.match(content.first)
148 content.shift if alert
149 blocks << Block.new(kind: alert ? :alert : :quote, text: alert&.[](1), children: parse(content))
150 next
151 end
152 if (marker = list_marker(line))
153 kind, ordered, number, = marker
154 items = []
155 loop do
156 content = [marker[3]]
157 at += 1
158 while at < lines.size && lines[at].start_with?(' ')
159 content << lines[at][2..]
160 at += 1
161 end
162 items << parse(content)
163 following = at
164 following += 1 while following < lines.size && lines[following].empty?
165 marker = following < lines.size ? list_marker(lines[following]) : nil
166 break unless marker && marker[0] == kind && marker[1] == ordered
167
168 at = following
169 end
170 blocks << Block.new(kind: :list, ordered: ordered, number: number, items: items)
171 next
172 end
173 if definition?(line)
174 at += 1
175 next
176 end
177 if at + 1 < lines.size && line.include?('|') && lines[at + 1].include?('|')
178 header = split_cells(line)
179 delimiters = split_cells(lines[at + 1])
180 if header.size == delimiters.size && delimiters.all? { |cell| /\A:?-+:?\z/.match?(cell) }
181 alignments = delimiters.map do |cell|
182 cell.start_with?(':') ? (cell.end_with?(':') ? 'center' : 'left') : (cell.end_with?(':') ? 'right' : nil)
183 end
184 rows = []
185 at += 2
186 while at < lines.size && lines[at].include?('|')
187 row = split_cells(lines[at]).take(header.size)
188 rows << row + Array.new(header.size - row.size, '')
189 at += 1
190 end
191 blocks << Block.new(kind: :table, alignments: alignments, lines: header, rows: rows)
192 next
193 end
194 end
195 end
196 text = [line.lstrip]
197 at += 1
198 level = nil
199 while at < lines.size && !lines[at].empty?
200 if /\A(?:={3,}|-{3,})\z/.match?(lines[at])
201 level = lines[at][0] == '=' ? 1 : 2
202 at += 1
203 break
204 end
205 break if interrupts?(lines[at])
206
207 text << lines[at].lstrip
208 at += 1
209 end
210 text = text.join("\n")
211 blocks << Block.new(kind: level ? :heading : (text == '[TOC]' ? :toc : :paragraph), text: text, number: level)
212 end
213 blocks
214 end

216 def definition?(line)
217 if (match = /\A\[\^([A-Za-z0-9_-]+)\]:(.*)\z/.match(line)) && !match[2].strip.empty?
218 @footnotes[match[1].downcase] ||= match[2].strip
219 return true
220 end
221 return false unless line.start_with?('[')
222
223 close = line.index(']')
224 return false unless close && line[close + 1] == ':'
225
226 label = line[1...close]
227 return false if label.include?('[') || label.strip.empty? || label.start_with?('^')
228
229 url, finish = escaped_read(line, skip_space(line, close + 2), ' ')
230 return false if url.empty?
231
232 at = skip_space(line, finish)
233 title = nil
234 if at < line.size
235 return false if at == finish || !"\"'".include?(line[at])
236
237 title, finish = escaped_read(line, at + 1, line[at])
238 return false unless finish == line.size - 1
239 end
240 @references[normalize(label)] ||= [url, title]
241 true
242 end
243end

245def run_end(text, at)
246 char = text[at]
247 at += 1
248 at += 1 while at < text.size && text[at] == char
249 at
250end
251
252def backtick_closer(text, at, length)
253 while (at = text.index('`', at))
254 finish = run_end(text, at)
255 return at if finish - at == length
256
257 at = finish
258 end
259 nil
260end
261
262def closing_bracket(text, at)
263 depth = 0
264 while at < text.size
265 char = text[at]
266 if char == '\\'
267 at += 2
268 next
269 end
270 if char == '`'
271 finish = run_end(text, at)
272 close = backtick_closer(text, finish, finish - at)
273 at = close ? close + finish - at : finish
274 next
275 end
276 depth += 1 if char == '['
277 if char == ']'
278 depth -= 1
279 return at if depth.zero?
280 end
281 at += 1
282 end
283 nil
284end

286def emphasis(pieces, first = 0, finish = pieces.size)
287 html = +''
288 at = first
289 while at < finish
290 piece = pieces[at]
291 if piece.is_a?(String)
292 html << piece
293 else
294 close = piece.can_open && ((at + 1)...finish).find do |index|
295 other = pieces[index]
296 other.is_a?(Delimiter) && other.char == piece.char && other.length == piece.length && other.can_close
297 end
298 if close
299 opening, closing = piece.char == '~' ? ['<del>', '</del>'] :
300 { 1 => ['<em>', '</em>'], 2 => ['<strong>', '</strong>'], 3 => ['<em><strong>', '</strong></em>'] }.fetch(piece.length)
301 html << opening << emphasis(pieces, at + 1, close) << closing
302 at = close
303 else
304 html << piece.char * piece.length
305 end
306 end
307 at += 1
308 end
309 html
310end

312class Renderer
313 def initialize(references, footnotes)
314 @references, @footnotes = references, footnotes
315 @used_ids = {}
316 @headings = []
317 @note_numbers = {}
318 @note_counts = Hash.new(0)
319 end

321 def inline(text, links: true, notes: true)
322 pieces = []
323 at = 0
324 while at < text.size
325 char = text[at]
326 following = text[at + 1]
327 if char == '\\'
328 if following && PUNCTUATION.include?(following)
329 pieces << escape_html(following)
330 at += 2
331 elsif following == "\n"
332 pieces << "<br />\n"
333 at += 2
334 else
335 pieces << '\\'
336 at += 1
337 end
338 next
339 end
340 if char == '`'
341 finish = run_end(text, at)
342 close = backtick_closer(text, finish, finish - at)
343 if close
344 code = text[finish...close].tr("\n", ' ')
345 code = code[1...-1] if code.start_with?(' ') && code.end_with?(' ') && !code.strip.empty?
346 pieces << "<code>#{escape_html(code)}</code>"
347 at = close + finish - at
348 else
349 pieces << text[at...finish]
350 at = finish
351 end
352 next
353 end
354 if links && char == '<' && (close = text.index('>', at + 1))
355 target = text[(at + 1)...close]
356 href = nil
357 if !target.match?(/[ \n<]/) && /\Ahttps?:\/\/.+/m.match?(target)
358 href = target
359 elsif /\A[A-Za-z0-9.+_-]+@[A-Za-z0-9-]+(?:\.[A-Za-z0-9-]+)+\z/.match?(target)
360 href = "mailto:#{target}"
361 end
362 if href
363 pieces << %(<a href="#{escape_html(href)}">#{escape_html(target)}</a>)
364 at = close + 1
365 next
366 end
367 end
368 if links && (at.zero? || " \n*_~(".include?(text[at - 1])) && (match = /\Ahttps?:\/\/[^ \n<]*/.match(text[at..]))
369 url = match[0].sub(/[.,:;!?*_~)]+\z/, '')
370 unless %w[http:// https://].include?(url)
371 pieces << %(<a href="#{escape_html(url)}">#{escape_html(url)}</a>)
372 at += url.size
373 next
374 end
375 end
376 if char == '[' || (char == '!' && following == '[')
377 if links && notes && text[at, 2] == '[^' && (close = text.index(']', at + 2))
378 name = text[(at + 2)...close].downcase
379 if @footnotes.key?(name)
380 pieces << footnote_reference(name)
381 at = close + 1
382 next
383 end
384 end
385 image = char == '!'
386 opening = at + (image ? 1 : 0)
387 if (image || links) && (close = closing_bracket(text, opening))
388 label = text[(opening + 1)...close]
389 if (dest = destination(text, close + 1, label))
390 reference, finish = dest
391 url, title = reference
392 attribute = title.nil? ? '' : %( title="#{escape_html(title)}")
393 content = inline(label, links: false, notes: false)
394 pieces << if image
395 %(<img src="#{escape_html(url)}" alt="#{strip_tags(content)}"#{attribute} />)
396 else
397 %(<a href="#{escape_html(url)}"#{attribute}>#{content}</a>)
398 end
399 at = finish
400 next
401 end
402 end
403 end
404 if '*_~'.include?(char)
405 finish = run_end(text, at)
406 length = finish - at
407 before = at.zero? ? ' ' : text[at - 1]
408 after = text[finish] || ' '
409 can_open, can_close = !WHITESPACE.include?(after), !WHITESPACE.include?(before)
410 can_open = can_close = false if char == '_' && /[A-Za-z0-9]/.match?(before) && /[A-Za-z0-9]/.match?(after)
411 pieces << if (char != '~' && length <= 3) || (char == '~' && length == 2)
412 Delimiter.new(char, length, can_open, can_close)
413 else char * length
414 end
415 at = finish
416 else
417 pieces << escape_html(char)
418 at += 1
419 end
420 end
421 emphasis(pieces)
422 end

424 def destination(text, at, label)
425 if text[at] == '['
426 close = text.index(']', at + 1)
427 if close && !text[(at + 1)...close].include?('[')
428 name = text[(at + 1)...close]
429 reference = @references[normalize(name.empty? ? label : name)]
430 return reference ? [reference, close + 1] : nil
431 end
432 elsif text[at] == '('
433 url, finish = escaped_read(text, skip_space(text, at + 1), " \n)")
434 position = skip_space(text, finish)
435 title = nil
436 valid = true
437 if position > finish && position < text.size && "\"'".include?(text[position])
438 title, finish = escaped_read(text, position + 1, text[position])
439 valid = finish < text.size
440 position = skip_space(text, finish + 1) if valid
441 end
442 return [[url, title], position + 1] if valid && text[position] == ')'
443 end
444 reference = @references[normalize(label)]
445 reference && [reference, at]
446 end

448 def document(blocks)
449 html = render_blocks(blocks)
450 contents = table_of_contents
451 html = html.gsub(TOC_MARKER) { contents }
452 html << footnote_list unless @note_numbers.empty?
453 html
454 end

456 def render_blocks(blocks) = blocks.map { |block| render(block) + "\n" }.join
457
458 def render(block)
459 case block.kind
460 when :heading then heading(block.number, block.text)
461 when :paragraph then "<p>#{inline(block.text)}</p>"
462 when :rule then '<hr />'
463 when :code
464 attribute = block.text.empty? ? '' : %( class="language-#{escape_html(block.text)}")
465 "<pre><code#{attribute}>#{block.lines.map { |line| escape_html(line) + "\n" }.join}</code></pre>"
466 when :quote then "<blockquote>\n#{render_blocks(block.children)}</blockquote>"
467 when :alert
468 %(<div class="alert alert-#{block.text.downcase}">\n<p class="alert-title">#{block.text.capitalize}</p>\n#{render_blocks(block.children)}</div>)
469 when :list
470 tag = block.ordered ? 'ol' : 'ul'
471 attribute = block.ordered && block.number != 1 ? %( start="#{block.number}") : ''
472 "<#{tag}#{attribute}>\n#{block.items.map { |item| render_item(item) + "\n" }.join}</#{tag}>"
473 when :table
474 html = "<table>\n<thead>\n<tr>\n#{table_cells('th', block, block.lines)}</tr>\n</thead>\n"
475 unless block.rows.empty?
476 html << "<tbody>\n"
477 block.rows.each { |row| html << "<tr>\n#{table_cells('td', block, row)}</tr>\n" }
478 html << "</tbody>\n"
479 end
480 html << '</table>'
481 when :toc then TOC_MARKER
482 end
483 end

485 def heading(level, text)
486 start = text.rindex(' {#')
487 candidate = start && text.end_with?('}') ? text[(start + 3)...-1] : nil
488 if candidate && /\A[A-Za-z0-9_-]+\z/.match?(candidate)
489 id = candidate
490 text = text[...start].sub(/ +\z/, '')
491 else
492 base = text.downcase.gsub(/[^a-z0-9_ \n-]/, '').tr(" \n", '--')
493 base = 'heading' if base.empty?
494 id = base
495 suffix = 1
496 while @used_ids.key?(id)
497 id = "#{base}-#{suffix}"
498 suffix += 1
499 end
500 end
501 @used_ids[id] = true
502 html = inline(text)
503 @headings << TocEntry.new(level, id, strip_tags(html), [])
504 %(<h#{level} id="#{id}">#{html}</h#{level}>)
505 end

507 def render_item(blocks)
508 html = +'<li>'
509 blocks.each_with_index do |block, index|
510 if block.kind == :paragraph
511 text = block.text
512 if index.zero? && text.size > 4 && ['[ ] ', '[x] ', '[X] '].include?(text[0, 4])
513 checked = text[1] == ' ' ? '' : ' checked=""'
514 html << %(<input type="checkbox"#{checked} disabled="" /> )
515 text = text[4..]
516 end
517 html << inline(text)
518 else
519 html << "\n" unless html.end_with?("\n")
520 html << render(block) << "\n"
521 end
522 end
523 html << '</li>'
524 end

526 def table_cells(tag, table, cells)
527 cells.zip(table.alignments).map do |cell, alignment|
528 style = alignment ? %( style="text-align:#{alignment}") : ''
529 "<#{tag}#{style}>#{inline(cell)}</#{tag}>\n"
530 end.join
531 end

533 def toc_list(entries)
534 html = +"<ul>\n"
535 entries.each do |entry|
536 html << %(<li><a href="##{entry.id}">#{entry.label}</a>)
537 html << "\n#{toc_list(entry.children)}\n" unless entry.children.empty?
538 html << "</li>\n"
539 end
540 html << '</ul>'
541 end

543 def table_of_contents
544 return "<nav class=\"toc\">\n</nav>" if @headings.empty?
545
546 roots = []
547 parents = []
548 @headings.each do |entry|
549 parents.pop while !parents.empty? && parents.last.level >= entry.level
550 (parents.empty? ? roots : parents.last.children) << entry
551 parents << entry
552 end
553 %(<nav class="toc">\n#{toc_list(roots)}\n</nav>)

556 def reference_id(number, count) = "fnref-#{number}#{count == 1 ? '' : "-#{count}"}"
557
558 def footnote_reference(name)
559 @note_numbers[name] ||= @note_numbers.size + 1
560 number = @note_numbers.fetch(name)
561 @note_counts[name] += 1
562 %(<sup id="#{reference_id(number, @note_counts[name])}"><a href="#fn-#{number}">#{number}</a></sup>)
563 end
564
565 def footnote_list
566 html = +"<section class=\"footnotes\">\n<ol>\n"
567 @note_numbers.each do |name, number|
568 body = inline(@footnotes.fetch(name), notes: false)
569 backlinks = (1..@note_counts[name]).map { |count| %(<a href="##{reference_id(number, count)}">&#8617;</a>) }.join(' ')
570 html << %(<li id="fn-#{number}">#{body} #{backlinks}</li>\n)
571 end
572 html << "</ol>\n</section>\n"
573 end
