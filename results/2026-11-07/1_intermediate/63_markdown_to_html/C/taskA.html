8static void *allocate(size_t bytes) {
9 void *p = calloc(1, bytes ? bytes : 1);
10 if (!p)
11 exit(EXIT_FAILURE);
12 if (allocation_count == allocation_capacity) {
13 allocation_capacity = allocation_capacity ? allocation_capacity * 2 : 128;
14 void **next = realloc(allocations, allocation_capacity * sizeof(void *));
15 if (!next)
16 exit(EXIT_FAILURE);
17 allocations = next;
18 }
19 allocations[allocation_count++] = p;
20 return p;
21}

44static void reserve(String *s, size_t length) {
45 if (length + 1 <= s->capacity)
46 return;
47 size_t capacity = s->capacity ? s->capacity : 32;
48 while (capacity < length + 1)
49 capacity *= 2;
50 char *next = allocate(capacity);
51 if (s->length)
52 memcpy(next, s->data, s->length);
53 s->data = next;
54 s->capacity = capacity;
55}
56static void append_n(String *s, const char *text, size_t length) {
57 reserve(s, s->length + length);
58 memcpy(s->data + s->length, text, length);
59 s->length += length;
60 s->data[s->length] = 0;
61}

126static void escape_into(String *html, const char *text) {
127 for (; *text; text++) {
128 switch (*text) {
129 case '&':
130 append(html, "&amp;");
131 break;
132 case '<':
133 append(html, "&lt;");
134 break;
135 case '>':
136 append(html, "&gt;");
137 break;
138 case '"':
139 append(html, "&quot;");
140 break;
141 default:
142 character(html, *text);
143 break;
144 }
145 }
146}

152static char *strip_tags(const char *text) {
153 String result = {0};
154 for (size_t at = 0; text[at];) {
155 const char *end = text[at] == '<' ? strchr(text + at, '>') : NULL;
156 if (end)
157 at = (size_t)(end - text) + 1;
158 else
159 character(&result, text[at++]);
160 }
161 return finish(result);
162}
163static char *normalize(const char *text) {
164 String result = {0};
165 int space = 0;
166 for (; *text; text++) {
167 if (whitespace(*text))
168 space = result.length != 0;
169 else {
170 if (space)
171 character(&result, ' ');
172 character(&result, (char)tolower((unsigned char)*text));
173 space = 0;
174 }
175 }
176 return finish(result);
177}

248static Fence opening_fence(const char *line) {
249 Fence fence = {0};
250 if (!has("`~", line[0]))
251 return fence;
252 size_t end = 1;
253 while (line[end] == line[0])
254 end++;
255 if (end < 3)
256 return fence;
257 char *info = trim(line + end);
258 if (line[0] == '`' && strchr(info, '`'))
259 return fence;
260 return (Fence){line[0], end, info};

262static Block *atx_heading(const char *line) {
263 size_t n = 0;
264 while (line[n] == '#')
265 n++;
266 if (!n || n > 6 || (line[n] && line[n] != ' '))
267 return NULL;
268 char *text = trim(line + n);
269 size_t end = strlen(text);
270 while (end && text[end - 1] == '#')
271 end--;
272 if (!end || text[end - 1] == ' ')
273 text = trim(slice(text, 0, end));
274 return block(HEADING, text, (int)n);
275}
276static int is_rule(const char *line) {
277 char ch = 0;
278 size_t n = 0;
279 for (; *line; line++)
280 if (*line != ' ') {
281 if (!ch)
282 ch = *line;
283 if (!has("-*_", ch) || *line != ch)
284 return 0;
285 n++;
286 }
287 return n >= 3;
288}
289typedef struct {
290 char marker;
291 int ordered, number;
292 char *content;
293} ListMarker;
294static ListMarker list_marker(const char *line) {
295 ListMarker result = {0};
296 if (is_rule(line))
297 return result;
298 size_t at = 0;
299 int number = 0;
300 if (has("-*+", line[0]))
301 at = 1;
302 else {
303 while (line[at] >= '0' && line[at] <= '9') {
304 if (at == 9)
305 return result;
306 number = number * 10 + line[at++] - '0';
307 }
308 if (!at || !has(".)", line[at]))
309 return result;
310 result.ordered = 1;
311 at++;
312 }
313 if (line[at] != ' ')
314 return (ListMarker){0};
315 result.marker = line[at - 1];
316 result.number = result.ordered ? number : 1;
317 while (line[at] == ' ')
318 at++;
319 result.content = (char *)line + at;
320 return result;
321}
322static int interrupts(const char *line) {
323 return line[0] && line[0] != ' ' &&
324 (opening_fence(line).ch || atx_heading(line) || is_rule(line) || line[0] == '>' ||
325 list_marker(line).marker);
326}
327static Strings split_cells(const char *line) {
328 char *text = trim(line);
329 size_t at = text[0] == '|' ? 1 : 0;
330 Strings cells = {0};
331 String current = {0};
332 int final_separator = 0;
333 while (text[at]) {
334 char ch = text[at++];
335 final_separator = 0;
336 if (ch == '\\' && text[at]) {
337 if (text[at] != '|')
338 character(&current, '\\');
339 character(&current, text[at++]);
340 } else if (ch == '|') {
341 ADD(cells, trim(finish(current)));
342 current = (String){0};
343 final_separator = 1;
344 } else
345 character(&current, ch);
346 }
347 if (!final_separator)
348 ADD(cells, trim(finish(current)));
349 return cells;
350}
351static int delimiter_cell(const char *cell) {
352 size_t at = cell[0] == ':' ? 1 : 0, first = at;
353 while (cell[at] == '-')
354 at++;
355 if (at == first)
356 return 0;
357 if (cell[at] == ':')
358 at++;
359 return !cell[at];
360}
361static int definition(Parser *p, const char *line) {
362 if (starts(line, "[^")) {
363 size_t at = 2;
364 while (label_character(line[at]))
365 at++;
366 if (at > 2 && line[at] == ']' && line[at + 1] == ':') {
367 char *body = trim(line + at + 2);
368 if (*body) {
369 char *name = lower(slice(line, 2, at));
370 if (!find_footnote(p, name))
371 ADD(p->footnotes, ((Footnote){name, body, 0, 0}));
372 return 1;
373 }
374 }
375 }
376 if (line[0] != '[')
377 return 0;
378 const char *close = strchr(line, ']');
379 if (!close || close[1] != ':')
380 return 0;
381 char *label = slice(line, 1, (size_t)(close - line));
382 if (strchr(label, '[') || !*trim(label) || label[0] == '^')
383 return 0;
384 Read url = escaped_read(line, skip_space(line, (size_t)(close - line) + 2), " ");
385 if (!*url.text)
386 return 0;
387 size_t at = skip_space(line, url.end);
388 char *title = NULL;
389 if (line[at]) {
390 if (at == url.end || !has("\"'", line[at]))
391 return 0;
392 char stop[2] = {line[at], 0};
393 Read quoted = escaped_read(line, at + 1, stop);
394 if (quoted.end != strlen(line) - 1)
395 return 0;
396 title = quoted.text;
397 }
398 char *name = normalize(label);
399 if (!find_reference(p, name))
400 ADD(p->references, ((Reference){name, url.text, title}));
401 return 1;
402}
403static Blocks parse(Parser *p, Strings lines) {
404 Blocks blocks = {0};
405 size_t at = 0;
406 while (at < lines.count) {
407 char *line = lines.data[at];
408 if (!*line) {
409 at++;
410 continue;
411 }
412 if (starts(line, " ")) {
413 Block *code = block(CODE, NULL, 0);
414 size_t end = at;
415 while (end < lines.count && (!*lines.data[end] || starts(lines.data[end], " ")))
416 end++;
417 while (end > at && !*lines.data[end - 1])
418 end--;
419 while (at < end) {
420 char *row = lines.data[at++];
421 ADD(code->lines, *row ? row + 4 : "");
422 }
423 ADD(blocks, code);
424 continue;
425 }
426 if (line[0] != ' ') {
427 Fence fence = opening_fence(line);
428 if (fence.ch) {
429 char *space = strchr(fence.info, ' ');
430 Block *code = block(
431 CODE, space ? slice(fence.info, 0, (size_t)(space - fence.info)) : fence.info,
432 0);
433 at++;
434 while (at < lines.count) {
435 char *row = lines.data[at++];
436 size_t n = strspn(row, fence.ch == '`' ? "`" : "~");
437 if (n >= fence.length && !row[n])
438 break;
439 ADD(code->lines, row);
440 }
441 ADD(blocks, code);
442 continue;
443 }
444 Block *heading = atx_heading(line);
445 if (heading) {
446 ADD(blocks, heading);
447 at++;
448 continue;
449 }
450 if (is_rule(line)) {
451 ADD(blocks, block(RULE, NULL, 0));
452 at++;
453 continue;
454 }
455 if (line[0] == '>') {
456 Strings content = {0};
457 while (at < lines.count && lines.data[at][0] == '>') {
458 char *row = lines.data[at++] + 1;
459 if (*row == ' ')
460 row++;
461 ADD(content, row);
462 }
463 Block *quote = block(QUOTE, NULL, 0);
464 const char *types[] = {"NOTE", "TIP", "IMPORTANT", "WARNING", "CAUTION"};
465 for (size_t i = 0; i < 5; i++) {
466 String marker = {0};
467 append(&marker, "[!");
468 append(&marker, types[i]);
469 character(&marker, ']');
470 if (!strcmp(content.data[0], finish(marker))) {
471 quote->kind = ALERT;
472 quote->text = (char *)types[i];
473 content.data++;
474 content.count--;
475 break;
476 }
477 }
478 quote->children = parse(p, content);
479 ADD(blocks, quote);
480 continue;
481 }
482 ListMarker marker = list_marker(line);
483 if (marker.marker) {
484 Block *list = block(LIST, NULL, marker.number);
485 list->ordered = marker.ordered;
486 char kind = marker.marker;
487 while (1) {
488 Strings content = {0};
489 ADD(content, marker.content);
490 at++;
491 while (at < lines.count && starts(lines.data[at], " ")) {
492 ADD(content, lines.data[at] + 2);
493 at++;
494 }
495 ADD(list->items, parse(p, content));
496 size_t next = at;
497 while (next < lines.count && !*lines.data[next])
498 next++;
499 marker = next < lines.count ? list_marker(lines.data[next]) : (ListMarker){0};
500 if (!marker.marker || marker.marker != kind || marker.ordered != list->ordered)
501 break;
502 at = next;
503 }
504 ADD(blocks, list);
505 continue;
506 }
507 if (definition(p, line)) {
508 at++;
509 continue;
510 }
511 if (at + 1 < lines.count && strchr(line, '|') && strchr(lines.data[at + 1], '|')) {
512 Strings header = split_cells(line), delimiters = split_cells(lines.data[at + 1]);
513 int valid = header.count == delimiters.count;
514 for (size_t i = 0; i < delimiters.count; i++)
515 if (!delimiter_cell(delimiters.data[i]))
516 valid = 0;
517 if (valid) {
518 Block *table = block(TABLE, NULL, 0);
519 table->lines = header;
520 for (size_t i = 0; i < delimiters.count; i++) {
521 char *cell = delimiters.data[i];
522 int left = cell[0] == ':', right = cell[strlen(cell) - 1] == ':';
523 ADD(table->alignments,
524 left ? (right ? "center" : "left") : (right ? "right" : ""));
525 }
526 at += 2;
527 while (at < lines.count && strchr(lines.data[at], '|')) {
528 Strings row = split_cells(lines.data[at++]);
529 while (row.count < delimiters.count)
530 ADD(row, "");
531 row.count = delimiters.count;
532 ADD(table->rows, row);
533 }
534 ADD(blocks, table);
535 continue;
536 }
537 }
538 }
539 Strings text = {0};
540 ADD(text, left_trim(line));
541 at++;
542 int level = 0;
543 while (at < lines.count && *lines.data[at]) {
544 char *row = lines.data[at];
545 size_t n = strlen(row);
546 if (n >= 3 && has("=-", row[0]) && strspn(row, row[0] == '=' ? "=" : "-") == n) {
547 level = row[0] == '=' ? 1 : 2;
548 at++;
549 break;
550 }
551 if (interrupts(row))
552 break;
553 ADD(text, left_trim(row));
554 at++;
555 }
556 char *combined = join(text, "\n");
557 ADD(blocks, block(level ? HEADING
558 : !strcmp(combined, "[TOC]") ? TOC
559 : PARAGRAPH,
560 combined, level));
561 }
562 return blocks;
563}

565static size_t run_end(const char *text, size_t at) {
566 char ch = text[at++];
567 while (text[at] == ch)
568 at++;
569 return at;

571static size_t backtick_closer(const char *text, size_t at, size_t length) {
572 while (text[at]) {
573 if (text[at] != '`') {
574 at++;
575 continue;
576 }
577 size_t end = run_end(text, at);
578 if (end - at == length)
579 return at;
580 at = end;
581 }
582 return (size_t)-1;
583}
584static size_t closing_bracket(const char *text, size_t at) {
585 int depth = 0;
586 while (text[at]) {
587 char ch = text[at];
588 if (ch == '\\') {
589 at += text[at + 1] ? 2 : 1;
590 continue;
591 }
592 if (ch == '`') {
593 size_t end = run_end(text, at), closer = backtick_closer(text, end, end - at);
594 at = closer == (size_t)-1 ? end : closer + end - at;
595 continue;
596 }
597 if (ch == '[')
598 depth++;
599 if (ch == ']' && --depth == 0)
600 return at;
601 at++;
602 }
603 return (size_t)-1;
604}

612static void emphasis_into(String *html, Pieces pieces, size_t begin, size_t end) {
613 for (size_t at = begin; at < end; at++) {
614 Piece piece = pieces.data[at];
615 if (!piece.delimiter) {
616 append(html, piece.html);
617 continue;
618 }
619 size_t close = at + 1;
620 if (piece.can_open) {
621 while (close < end &&
622 !(pieces.data[close].delimiter == piece.delimiter &&
623 pieces.data[close].length == piece.length && pieces.data[close].can_close))
624 close++;
625 } else
626 close = end;
627 if (close == end) {
628 for (size_t i = 0; i < piece.length; i++)
629 character(html, piece.delimiter);
630 continue;
631 }
632 const char *opening, *closing;
633 if (piece.delimiter == '~') {
634 opening = "<del>";
635 closing = "</del>";
636 } else if (piece.length == 1) {
637 opening = "<em>";
638 closing = "</em>";
639 } else if (piece.length == 2) {
640 opening = "<strong>";
641 closing = "</strong>";
642 } else {
643 opening = "<em><strong>";
644 closing = "</strong></em>";
645 }
646 append(html, opening);
647 emphasis_into(html, pieces, at + 1, close);
648 append(html, closing);
649 at = close;
650 }

735static int email_address(const char *text) {
736 const char *at = strchr(text, '@');
737 if (!at || at == text || strchr(at + 1, '@'))
738 return 0;
739 for (const char *p = text; p < at; p++)
740 if (!alphanumeric(*p) && !has(".+-_", *p))
741 return 0;
742 int dots = 0, length = 0;
743 for (const char *p = at + 1;; p++) {
744 if (*p == '.' || !*p) {
745 if (!length)
746 return 0;
747 if (!*p)
748 break;
749 dots++;
750 length = 0;
751 } else {
752 if (!alphanumeric(*p) && *p != '-')
753 return 0;
754 length++;
755 }
756 }
757 return dots >= 1;
758}
759static char *inline_html(Renderer *r, const char *text, int links, int notes) {
760 Pieces pieces = {0};
761 size_t at = 0;
762 while (text[at]) {
763 char ch = text[at];
764 if (ch == '\\') {
765 if (has(punctuation, text[at + 1])) {
766 ADD(pieces, ((Piece){escape_html(slice(text, at + 1, at + 2)), 0, 0, 0, 0}));
767 at += 2;
768 } else if (text[at + 1] == '\n') {
769 ADD(pieces, ((Piece){"<br />\n", 0, 0, 0, 0}));
770 at += 2;
771 } else {
772 ADD(pieces, ((Piece){"\\", 0, 0, 0, 0}));
773 at++;
774 }
775 continue;
776 }
777 if (ch == '`') {
778 size_t end = run_end(text, at), closer = backtick_closer(text, end, end - at);
779 if (closer == (size_t)-1) {
780 ADD(pieces, ((Piece){slice(text, at, end), 0, 0, 0, 0}));
781 at = end;
782 } else {
783 char *code = slice(text, end, closer);
784 size_t length = strlen(code);
785 for (size_t i = 0; i < length; i++)
786 if (code[i] == '\n')
787 code[i] = ' ';
788 if (length && code[0] == ' ' && code[length - 1] == ' ' && *trim(code))
789 code = slice(code, 1, length - 1);
790 String html = {0};
791 append(&html, "<code>");
792 escape_into(&html, code);
793 append(&html, "</code>");
794 ADD(pieces, ((Piece){finish(html), 0, 0, 0, 0}));
795 at = closer + end - at;
796 }
797 continue;
798 }
799 if (links && ch == '<') {
800 const char *close = strchr(text + at + 1, '>');
801 if (close) {
802 char *target = slice(text, at + 1, (size_t)(close - text));
803 char *href = NULL;
804 if (!strpbrk(target, " \n<") &&
805 ((starts(target, "http://") && strlen(target) > 7) ||
806 (starts(target, "https://") && strlen(target) > 8)))
807 href = target;
808 if (!href && email_address(target)) {
809 String url = {0};
810 append(&url, "mailto:");
811 append(&url, target);
812 href = finish(url);
813 }
814 if (href) {
815 String html = {0};
816 append(&html, "<a href=\"");
817 escape_into(&html, href);
818 append(&html, "\">");
819 escape_into(&html, target);
820 append(&html, "</a>");
821 ADD(pieces, ((Piece){finish(html), 0, 0, 0, 0}));
822 at = (size_t)(close - text) + 1;
823 continue;
824 }
825 }
826 }
827 if (links && (starts(text + at, "http://") || starts(text + at, "https://")) &&
828 (!at || has(" \n*_~(", text[at - 1]))) {
829 size_t end = at;
830 while (text[end] && !has(" \n<", text[end]))
831 end++;
832 while (end > at && has(".,:;!?*_~)", text[end - 1]))
833 end--;
834 char *url = slice(text, at, end);
835 if (strcmp(url, "http://") && strcmp(url, "https://")) {
836 String html = {0};
837 append(&html, "<a href=\"");
838 escape_into(&html, url);
839 append(&html, "\">");
840 escape_into(&html, url);
841 append(&html, "</a>");
842 ADD(pieces, ((Piece){finish(html), 0, 0, 0, 0}));
843 at = end;
844 continue;
845 }
846 }
847 if (ch == '[' || (ch == '!' && text[at + 1] == '[')) {
848 if (links && notes && starts(text + at, "[^")) {
849 const char *close = strchr(text + at + 2, ']');
850 if (close) {
851 Footnote *note = find_footnote(
852 r->parser, lower(slice(text, at + 2, (size_t)(close - text))));
853 if (note) {
854 ADD(pieces, ((Piece){footnote_reference(r, note), 0, 0, 0, 0}));
855 at = (size_t)(close - text) + 1;
856 continue;
857 }
858 }
859 }
860 int image = ch == '!';
861 size_t opening = at + (image ? 1 : 0), close = closing_bracket(text, opening);
862 if ((image || links) && close != (size_t)-1) {
863 char *label = slice(text, opening + 1, close);
864 Destination dest = destination(r, text, close + 1, label);
865 if (dest.reference) {
866 char *content = inline_html(r, label, 0, 0);
867 String html = {0};
868 append(&html, image ? "<img src=\"" : "<a href=\"");
869 escape_into(&html, dest.reference->url);
870 character(&html, '"');
871 if (image) {
872 append(&html, " alt=\"");
873 append(&html, strip_tags(content));
874 character(&html, '"');
875 }
876 if (dest.reference->title) {
877 append(&html, " title=\"");
878 escape_into(&html, dest.reference->title);
879 character(&html, '"');
880 }
881 if (image)
882 append(&html, " />");
883 else {
884 character(&html, '>');
885 append(&html, content);
886 append(&html, "</a>");
887 }
888 ADD(pieces, ((Piece){finish(html), 0, 0, 0, 0}));
889 at = dest.end;
890 continue;
891 }
892 }
893 }
894 if (has("*_~", ch)) {
895 size_t end = run_end(text, at), length = end - at;
896 char before = at ? text[at - 1] : ' ', after = text[end] ? text[end] : ' ';
897 int open = !whitespace(after), close = !whitespace(before);
898 if (ch == '_' && alphanumeric(before) && alphanumeric(after))
899 open = close = 0;
900 if ((ch != '~' && length <= 3) || (ch == '~' && length == 2))
901 ADD(pieces, ((Piece){NULL, ch, length, open, close}));
902 else
903 ADD(pieces, ((Piece){slice(text, at, end), 0, 0, 0, 0}));
904 at = end;
905 continue;
906 }
907 ADD(pieces, ((Piece){escape_html(slice(text, at, at + 1)), 0, 0, 0, 0}));
908 at++;
909 }
910 String html = {0};
911 emphasis_into(&html, pieces, 0, pieces.count);
912 return finish(html);
913}

923static char *render_heading(Renderer *r, Block *b) {
924 char *text = b->text, *id = NULL;
925 const char *last = NULL, *cursor = text;
926 while ((cursor = strstr(cursor, " {#"))) {
927 last = cursor;
928 cursor++;
929 }
930 size_t length = strlen(text);
931 if (length && text[length - 1] == '}' && last) {
932 char *candidate = slice(last, 3, strlen(last) - 1);
933 int valid = *candidate != 0;
934 for (size_t i = 0; candidate[i]; i++)
935 if (!label_character(candidate[i]))
936 valid = 0;
937 if (valid) {
938 id = candidate;
939 text = slice(text, 0, (size_t)(last - text));
940 size_t end = strlen(text);
941 while (end && text[end - 1] == ' ')
942 text[--end] = 0;
943 }
944 }
945 if (!id) {
946 String slug = {0};
947 for (char *p = lower(text); *p; p++) {
948 if (label_character(*p))
949 character(&slug, *p);
950 else if (whitespace(*p))
951 character(&slug, '-');
952 }
953 char *base = slug.length ? finish(slug) : "heading";
954 id = base;
955 for (size_t suffix = 1; string_in(r->used_ids, id); suffix++) {
956 String value = {0};
957 append(&value, base);
958 character(&value, '-');
959 append(&value, decimal(suffix));
960 id = finish(value);
961 }
962 }
963 ADD(r->used_ids, id);
964 char *content = inline_html(r, text, 1, 1);
965 Heading *heading = allocate(sizeof(Heading));
966 heading->level = b->number;
967 heading->id = id;
968 heading->label = strip_tags(content);
969 ADD(r->headings, heading);
970 String html = {0};
971 append(&html, "<h");
972 append(&html, decimal((size_t)b->number));
973 append(&html, " id=\"");
974 append(&html, id);
975 append(&html, "\">");
976 append(&html, content);
977 append(&html, "</h");
978 append(&html, decimal((size_t)b->number));
979 character(&html, '>');
980 return finish(html);
981}
982static char *render_item(Renderer *r, Blocks blocks) {
983 String html = {0};
984 append(&html, "<li>");
985 for (size_t i = 0; i < blocks.count; i++) {
986 Block *b = blocks.data[i];
987 if (b->kind == PARAGRAPH) {
988 char *text = b->text;
989 if (!i && strlen(text) > 4 &&
990 (starts(text, "[ ] ") || starts(text, "[x] ") || starts(text, "[X] "))) {
991 append(&html, text[1] == ' '
992 ? "<input type=\"checkbox\" disabled=\"\" /> "
993 : "<input type=\"checkbox\" checked=\"\" disabled=\"\" /> ");
994 text += 4;
995 }
996 append(&html, inline_html(r, text, 1, 1));
997 } else {
998 if (html.data[html.length - 1] != '\n')
999 character(&html, '\n');
1000 append(&html, render(r, b));
1001 character(&html, '\n');
1002 }
1003 }
1004 append(&html, "</li>");
1005 return finish(html);
1006}

1023static char *render(Renderer *r, Block *b) {
1024 String html = {0};
1025 switch (b->kind) {
1026 case HEADING:
1027 return render_heading(r, b);
1028 case PARAGRAPH:
1029 append(&html, "<p>");
1030 append(&html, inline_html(r, b->text, 1, 1));
1031 append(&html, "</p>");
1032 break;
1033 case RULE:
1034 return "<hr />";
1035 case CODE:
1036 append(&html, "<pre><code");
1037 if (*b->text) {
1038 append(&html, " class=\"language-");
1039 escape_into(&html, b->text);
1040 character(&html, '"');
1041 }
1042 character(&html, '>');
1043 for (size_t i = 0; i < b->lines.count; i++) {
1044 escape_into(&html, b->lines.data[i]);
1045 character(&html, '\n');
1046 }
1047 append(&html, "</code></pre>");
1048 break;
1049 case QUOTE:
1050 append(&html, "<blockquote>\n");
1051 append(&html, render_blocks(r, b->children));
1052 append(&html, "</blockquote>");
1053 break;
1054 case ALERT: {
1055 char *name = lower(b->text), *title = slice(name, 0, strlen(name));
1056 title[0] = (char)toupper((unsigned char)title[0]);
1057 append(&html, "<div class=\"alert alert-");
1058 append(&html, name);
1059 append(&html, "\">\n<p class=\"alert-title\">");
1060 append(&html, title);
1061 append(&html, "</p>\n");
1062 append(&html, render_blocks(r, b->children));
1063 append(&html, "</div>");
1064 break;
1065 }
1066 case LIST: {
1067 char *tag = b->ordered ? "ol" : "ul";
1068 character(&html, '<');
1069 append(&html, tag);
1070 if (b->ordered && b->number != 1) {
1071 append(&html, " start=\"");
1072 append(&html, decimal((size_t)b->number));
1073 character(&html, '"');
1074 }
1075 append(&html, ">\n");
1076 for (size_t i = 0; i < b->items.count; i++) {
1077 append(&html, render_item(r, b->items.data[i]));
1078 character(&html, '\n');
1079 }
1080 append(&html, "</");
1081 append(&html, tag);
1082 character(&html, '>');
1083 break;
1084 }
1085 case TABLE:
1086 append(&html, "<table>\n<thead>\n<tr>\n");
1087 table_cells(&html, r, b, "th", b->lines);
1088 append(&html, "</tr>\n</thead>\n");
1089 if (b->rows.count) {
1090 append(&html, "<tbody>\n");
1091 for (size_t i = 0; i < b->rows.count; i++) {
1092 append(&html, "<tr>\n");
1093 table_cells(&html, r, b, "td", b->rows.data[i]);
1094 append(&html, "</tr>\n");
1095 }
1096 append(&html, "</tbody>\n");
1097 }
1098 append(&html, "</table>");
1099 break;
1100 case TOC:
1101 return (char *)toc_marker;
1102 }
1103 return finish(html);
1104}

1126 if (!r->headings.count)

1132 parents.count--;

1164static char *document(Renderer *r, Blocks blocks) {
1165 char *body = render_blocks(r, blocks);
1166 char *toc = table_of_contents(r);
1167 String html = {0};
1168 size_t marker_length = strlen(toc_marker);
1169 while (*body) {
1170 const char *found = strstr(body, toc_marker);
1171 if (!found) {
1172 append(&html, body);
1173 break;
1174 }
1175 append_n(&html, body, (size_t)(found - body));
1176 append(&html, toc);
1177 body = (char *)found + marker_length;
1178 }
1179 if (r->notes.count)
1180 append(&html, footnote_list(r));
1181 return finish(html);
1182}
