21escape_html <- function(text) {
22 for (pair in list(c("&", "&amp;"), c("<", "&lt;"), c(">", "&gt;"), c('"', "&quot;"))) {
23 text <- gsub(pair[1L], pair[2L], text, fixed = TRUE)

25 text

34escaped_read <- function(text, at, stops) {

43opening_fence <- function(line) {
44 found <- match_parts("\\A(`{3,}|~{3,})", line)
45 if (is.null(found)) return(NULL)
46 fence <- found[1L]
47 info <- trim_spaces(slice(line, nchar(fence) + 1L))
48 if (char_at(fence, 1L) == "`" && grepl("`", info, fixed = TRUE)) return(NULL)
49 list(fence, info)

51atx_heading <- function(line) {
52 found <- match_parts("\\A(#{1,6})(?: |\\z)", line)
53 if (is.null(found)) return(NULL)
54 level <- nchar(found[2L])
55 text <- trim_spaces(slice(line, level + 1L))
56 trimmed <- sub("#+\\z", "", text, perl = TRUE)
57 if (!nzchar(trimmed) || endsWith(trimmed, " ")) text <- trim_spaces(trimmed)
58 list(kind = "heading", text = text, number = level)
59}
60is_rule <- function(line) whole("-{3,}|\\*{3,}|_{3,}", gsub(" ", "", line, fixed = TRUE))
61list_marker <- function(line) {
62 if (is_rule(line)) return(NULL)
63 found <- match_parts("\\A(?:([-*+])|([0-9]{1,9})([.)])) +", line)
64 if (is.null(found)) return(NULL)
65 ordered <- nzchar(found[3L])
66 list(kind = if (ordered) found[4L] else found[2L], ordered = ordered,
67 number = if (ordered) as.integer(found[3L]) else 1L,
68 content = slice(line, nchar(found[1L]) + 1L))
69}
70interrupts_paragraph <- function(line) {
71 !startsWith(line, " ") && (!is.null(opening_fence(line)) || !is.null(atx_heading(line)) ||
72 is_rule(line) || startsWith(line, ">") || !is.null(list_marker(line)))
73}
74split_cells <- function(line) {
75 text <- trim_spaces(line)
76 at <- if (startsWith(text, "|")) 2L else 1L
77 cells <- ""
78 separator <- FALSE
79 while (at <= nchar(text)) {
80 char <- char_at(text, at)
81 at <- at + 1L
82 separator <- FALSE
83 last <- length(cells)
84 if (char == "\\" && at <= nchar(text)) {
85 if (char_at(text, at) != "|") cells[last] <- paste0(cells[last], "\\")
86 cells[last] <- paste0(cells[last], char_at(text, at))
87 at <- at + 1L
88 } else if (char == "|") {
89 cells <- c(cells, "")
90 separator <- TRUE
91 } else cells[last] <- paste0(cells[last], char)
92 }
93 if (separator) cells <- head(cells, -1L)
94 vapply(cells, trim_spaces, character(1L), USE.NAMES = FALSE)
95}

102 definition <- function(line) {
103 found <- match_parts("\\A\\[\\^([A-Za-z0-9_-]+)\\]:(.*)\\z", line)
104 if (!is.null(found) && nzchar(trim_spaces(found[3L]))) {
105 name <- tolower(found[2L])
106 if (is.null(state$footnotes[[name]])) state$footnotes[[name]] <- trim_spaces(found[3L])
107 return(TRUE)
108 }
109 if (!startsWith(line, "[")) return(FALSE)
110 close <- regexpr("]", line, fixed = TRUE)[1L]
111 if (close < 1L || char_at(line, close + 1L) != ":") return(FALSE)
112 label <- slice(line, 2L, close)
113 if (grepl("[", label, fixed = TRUE) || !nzchar(trim_spaces(label)) || startsWith(label, "^")) return(FALSE)
114 read <- escaped_read(line, skip_space(line, close + 2L), " ")
115 url <- read[[1L]]
116 end <- read[[2L]]
117 if (!nzchar(url)) return(FALSE)
118 at <- skip_space(line, end)
119 title <- NULL
120 if (at <= nchar(line)) {
121 if (at == end || !has("\"'", char_at(line, at))) return(FALSE)
122 read <- escaped_read(line, at + 1L, char_at(line, at))
123 if (read[[2L]] != nchar(line)) return(FALSE)
124 title <- read[[1L]]
125 }
126 name <- normalize_label(label)
127 if (is.null(state$references[[name]])) state$references[[name]] <- list(url = url, title = title)
128 TRUE
129 }

131 parse <- function(lines) {
132 blocks <- list()
133 at <- 1L
134 while (at <= length(lines)) {
135 line <- lines[at]
136 if (!nzchar(line)) {
137 at <- at + 1L
138 next
139 }
140 if (startsWith(line, " ")) {
141 end <- at
142 while (end <= length(lines) && (!nzchar(lines[end]) || startsWith(lines[end], " "))) end <- end + 1L
143 while (!nzchar(lines[end - 1L])) end <- end - 1L
144 code <- vapply(lines[seq.int(at, end - 1L)], function(row) slice(row, 5L), character(1L))
145 blocks[[length(blocks) + 1L]] <- list(kind = "code", text = "", lines = code)
146 at <- end
147 next
148 }
149 if (!startsWith(line, " ")) {
150 fence <- opening_fence(line)
151 if (!is.null(fence)) {
152 code <- character()
153 at <- at + 1L
154 while (at <= length(lines)) {
155 row <- lines[at]
156 at <- at + 1L
157 if (nchar(row) >= nchar(fence[[1L]]) && !nzchar(gsub(char_at(fence[[1L]], 1L), "", row, fixed = TRUE))) break
158 code <- c(code, row)
159 }
160 language <- sub(" .*\\z", "", fence[[2L]], perl = TRUE)
161 blocks[[length(blocks) + 1L]] <- list(kind = "code", text = language, lines = code)
162 next
163 }
164 heading <- atx_heading(line)
165 if (!is.null(heading)) {
166 blocks[[length(blocks) + 1L]] <- heading
167 at <- at + 1L
168 next
169 }
170 if (is_rule(line)) {
171 blocks[[length(blocks) + 1L]] <- list(kind = "rule")
172 at <- at + 1L
173 next
174 }
175 if (startsWith(line, ">")) {
176 content <- character()
177 while (at <= length(lines) && startsWith(lines[at], ">")) {
178 row <- slice(lines[at], 2L)
179 if (startsWith(row, " ")) row <- slice(row, 2L)
180 content <- c(content, row)
181 at <- at + 1L
182 }
183 alert <- match_parts("\\A\\[!(NOTE|TIP|IMPORTANT|WARNING|CAUTION)\\]\\z", content[1L])
184 if (!is.null(alert)) content <- content[-1L]
185 blocks[[length(blocks) + 1L]] <- list(kind = if (is.null(alert)) "quote" else "alert",
186 text = if (is.null(alert)) "" else alert[2L], children = parse(content))
187 next
188 }
189 marker <- list_marker(line)
190 if (!is.null(marker)) {
191 kind <- marker$kind
192 ordered <- marker$ordered
193 number <- marker$number
194 items <- list()
195 repeat {
196 content <- marker$content
197 at <- at + 1L
198 while (at <= length(lines) && startsWith(lines[at], " ")) {
199 content <- c(content, slice(lines[at], 3L))
200 at <- at + 1L
201 }
202 items[[length(items) + 1L]] <- parse(content)
203 following <- at
204 while (following <= length(lines) && !nzchar(lines[following])) following <- following + 1L
205 marker <- if (following <= length(lines)) list_marker(lines[following]) else NULL
206 if (is.null(marker) || marker$kind != kind || marker$ordered != ordered) break
207 at <- following
208 }
209 blocks[[length(blocks) + 1L]] <- list(kind = "list", ordered = ordered, number = number, items = items)
210 next
211 }
212 if (definition(line)) {
213 at <- at + 1L
214 next
215 }
216 if (at < length(lines) && grepl("|", line, fixed = TRUE) && grepl("|", lines[at + 1L], fixed = TRUE)) {
217 header <- split_cells(line)
218 delimiters <- split_cells(lines[at + 1L])
219 if (length(header) == length(delimiters) && all(vapply(delimiters, function(cell) whole(":?-+:?", cell), logical(1L)))) {
220 alignments <- vapply(delimiters, function(cell) {
221 if (startsWith(cell, ":")) {
222 if (endsWith(cell, ":")) "center" else "left"
223 } else if (endsWith(cell, ":")) "right" else ""
224 }, character(1L))
225 rows <- list()
226 at <- at + 2L
227 while (at <= length(lines) && grepl("|", lines[at], fixed = TRUE)) {
228 row <- head(split_cells(lines[at]), length(header))
229 rows[[length(rows) + 1L]] <- c(row, rep("", length(header) - length(row)))
230 at <- at + 1L
231 }
232 blocks[[length(blocks) + 1L]] <- list(kind = "table", alignments = alignments, lines = header, rows = rows)
233 next
234 }
235 }
236 }
237 text <- left_trim(line)
238 at <- at + 1L
239 level <- NULL
240 while (at <= length(lines) && nzchar(lines[at])) {
241 if (whole("={3,}|-{3,}", lines[at])) {
242 level <- if (char_at(lines[at], 1L) == "=") 1L else 2L
243 at <- at + 1L
244 break
245 }
246 if (interrupts_paragraph(lines[at])) break
247 text <- c(text, left_trim(lines[at]))
248 at <- at + 1L
249 }
250 text <- paste(text, collapse = "\n")
251 blocks[[length(blocks) + 1L]] <- list(kind = if (!is.null(level)) "heading" else if (text == "[TOC]") "toc" else "paragraph",
252 text = text, number = level)
253 }
254 blocks
255 }
256 list(parse = parse, state = state)
257}

259run_end <- function(text, at) {
260 char <- char_at(text, at)
261 at <- at + 1L
262 while (at <= nchar(text) && char_at(text, at) == char) at <- at + 1L

265backtick_closer <- function(text, at, length) {
266 while (at <= nchar(text)) {
267 offset <- regexpr("`", slice(text, at), fixed = TRUE)[1L]
268 if (offset < 1L) return(NULL)
269 at <- at + offset - 1L
270 end <- run_end(text, at)
271 if (end - at == length) return(at)
272 at <- end
273 }
274 NULL
275}
276closing_bracket <- function(text, at) {
277 depth <- 0L
278 while (at <= nchar(text)) {
279 char <- char_at(text, at)
280 if (char == "\\") {
281 at <- at + 2L
282 next
283 }
284 if (char == "`") {
285 end <- run_end(text, at)
286 close <- backtick_closer(text, end, end - at)
287 at <- if (is.null(close)) end else close + end - at
288 next
289 }
290 if (char == "[") depth <- depth + 1L
291 if (char == "]") {
292 depth <- depth - 1L
293 if (depth == 0L) return(at)
294 }
295 at <- at + 1L
296 }
297 NULL
298}
299emphasis <- function(pieces, first = 1L, end = length(pieces) + 1L) {
300 html <- character()
301 at <- first
302 while (at < end) {
303 piece <- pieces[[at]]
304 if (is.character(piece)) html <- c(html, piece) else {
305 close <- end
306 if (piece$can_open && at + 1L < end) {
307 for (index in seq.int(at + 1L, end - 1L)) {
308 other <- pieces[[index]]
309 if (is.list(other) && other$char == piece$char && other$length == piece$length && other$can_close) {
310 close <- index
311 break
312 }
313 }
314 }
315 if (close == end) html <- c(html, strrep(piece$char, piece$length)) else {
316 tags <- if (piece$char == "~") c("<del>", "</del>") else switch(as.character(piece$length),
317 "1" = c("<em>", "</em>"), "2" = c("<strong>", "</strong>"), "3" = c("<em><strong>", "</strong></em>"))
318 html <- c(html, tags[1L], emphasis(pieces, at + 1L, close), tags[2L])
319 at <- close
320 }
321 }
322 at <- at + 1L
323 }
324 paste0(html, collapse = "")
325}

345 destination <- function(text, at, label) {
346 if (char_at(text, at) == "[") {
347 offset <- regexpr("]", slice(text, at + 1L), fixed = TRUE)[1L]
348 if (offset > 0L) {
349 close <- at + offset
350 name <- slice(text, at + 1L, close)
351 if (!grepl("[", name, fixed = TRUE)) {
352 reference <- lookup(references, normalize_label(if (nzchar(name)) name else label))
353 return(if (is.null(reference)) NULL else list(reference, close + 1L))
354 }
355 }
356 } else if (char_at(text, at) == "(") {
357 read <- escaped_read(text, skip_space(text, at + 1L), " \n)")
358 url <- read[[1L]]
359 end <- read[[2L]]
360 position <- skip_space(text, end)
361 title <- NULL
362 valid <- TRUE
363 if (position > end && has("\"'", char_at(text, position))) {
364 read <- escaped_read(text, position + 1L, char_at(text, position))
365 title <- read[[1L]]
366 valid <- read[[2L]] <= nchar(text)
367 if (valid) position <- skip_space(text, read[[2L]] + 1L)
368 }
369 if (valid && char_at(text, position) == ")") return(list(list(url = url, title = title), position + 1L))
370 }
371 reference <- lookup(references, normalize_label(label))
372 if (is.null(reference)) NULL else list(reference, at)
373 }

375 inline <- function(text, links = TRUE, notes = TRUE) {
376 pieces <- list()
377 at <- 1L
378 while (at <= nchar(text)) {
379 char <- char_at(text, at)
380 following <- char_at(text, at + 1L)
381 if (char == "\\") {
382 if (has(punctuation, following)) {
383 piece <- escape_html(following)
384 at <- at + 2L
385 } else if (following == "\n") {
386 piece <- "<br />\n"
387 at <- at + 2L
388 } else {
389 piece <- "\\"
390 at <- at + 1L
391 }
392 pieces[[length(pieces) + 1L]] <- piece
393 next
394 }
395 if (char == "`") {
396 end <- run_end(text, at)
397 close <- backtick_closer(text, end, end - at)
398 if (is.null(close)) {
399 piece <- slice(text, at, end)
400 at <- end
401 } else {
402 code <- gsub("\n", " ", slice(text, end, close), fixed = TRUE)
403 if (startsWith(code, " ") && endsWith(code, " ") && nzchar(trim_spaces(code))) code <- slice(code, 2L, nchar(code))
404 piece <- paste0("<code>", escape_html(code), "</code>")
405 at <- close + end - at
406 }
407 pieces[[length(pieces) + 1L]] <- piece
408 next
409 }
410 if (links && char == "<") {
411 offset <- regexpr(">", slice(text, at + 1L), fixed = TRUE)[1L]
412 if (offset > 0L) {
413 close <- at + offset
414 target <- slice(text, at + 1L, close)
415 href <- NULL
416 if (!grepl("[ \n<]", target) && grepl("\\Ahttps?://.+", target, perl = TRUE)) href <- target else
417 if (whole("[A-Za-z0-9.+_-]+@[A-Za-z0-9-]+(?:\\.[A-Za-z0-9-]+)+", target)) href <- paste0("mailto:", target)
418 if (!is.null(href)) {
419 pieces[[length(pieces) + 1L]] <- paste0('<a href="', escape_html(href), '">', escape_html(target), '</a>')
420 at <- close + 1L
421 next
422 }
423 }
424 }
425 if (links && (at == 1L || has(" \n*_~(", char_at(text, at - 1L)))) {
426 found <- match_parts("\\Ahttps?://[^ \n<]*", slice(text, at))
427 if (!is.null(found)) {
428 url <- sub("[.,:;!?*_~)]+\\z", "", found[1L], perl = TRUE)
429 if (!(url %in% c("http://", "https://"))) {
430 pieces[[length(pieces) + 1L]] <- paste0('<a href="', escape_html(url), '">', escape_html(url), '</a>')
431 at <- at + nchar(url)
432 next
433 }
434 }
435 }
436 if (char == "[" || (char == "!" && following == "[")) {
437 if (links && notes && starts(text, "[^", at)) {
438 offset <- regexpr("]", slice(text, at + 2L), fixed = TRUE)[1L]
439 if (offset > 0L) {
440 close <- at + offset + 1L
441 name <- tolower(slice(text, at + 2L, close))
442 if (!is.null(lookup(footnotes, name))) {
443 pieces[[length(pieces) + 1L]] <- footnote_reference(name)
444 at <- close + 1L
445 next
446 }
447 }
448 }
449 image <- char == "!"
450 opening <- at + as.integer(image)
451 close <- if (image || links) closing_bracket(text, opening) else NULL
452 if (!is.null(close)) {
453 label <- slice(text, opening + 1L, close)
454 dest <- destination(text, close + 1L, label)
455 if (!is.null(dest)) {
456 reference <- dest[[1L]]
457 title <- if (is.null(reference$title)) "" else paste0(' title="', escape_html(reference$title), '"')
458 content <- inline(label, FALSE, FALSE)
459 pieces[[length(pieces) + 1L]] <- if (image)
460 paste0('<img src="', escape_html(reference$url), '" alt="', strip_tags(content), '"', title, ' />') else
461 paste0('<a href="', escape_html(reference$url), '"', title, '>', content, '</a>')
462 at <- dest[[2L]]
463 next
464 }
465 }
466 }
467 if (has("*_~", char)) {
468 end <- run_end(text, at)
469 length <- end - at
470 before <- if (at == 1L) " " else char_at(text, at - 1L)
471 after <- if (end > nchar(text)) " " else char_at(text, end)
472 can_open <- !has(whitespace, after)
473 can_close <- !has(whitespace, before)
474 if (char == "_" && whole("[A-Za-z0-9]", before) && whole("[A-Za-z0-9]", after)) can_open <- can_close <- FALSE
475 piece <- if ((char != "~" && length <= 3L) || (char == "~" && length == 2L))
476 list(char = char, length = length, can_open = can_open, can_close = can_close) else strrep(char, length)
477 pieces[[length(pieces) + 1L]] <- piece
478 at <- end
479 } else {
480 pieces[[length(pieces) + 1L]] <- escape_html(char)
481 at <- at + 1L
482 }
483 }
484 emphasis(pieces)
485 }

487 heading <- function(level, text) {

489 start <- tail(matches, 1L)

491 if (whole("[A-Za-z0-9_-]+", candidate)) {
492 id <- candidate
493 text <- right_trim(slice(text, 1L, start))
494 } else {
495 base <- gsub("[^a-z0-9_ \n-]", "", tolower(text), perl = TRUE)
496 base <- gsub("[ \n]", "-", base)

499 suffix <- 1L
500 while (!is.null(state$used_ids[[id]])) {
501 id <- paste0(base, "-", suffix)
502 suffix <- suffix + 1L

504 }
505 state$used_ids[[id]] <- TRUE
506 html <- inline(text)
507 state$headings[[length(state$headings) + 1L]] <- list(level = level, id = id, label = strip_tags(html), children = integer())
508 paste0("<h", level, ' id="', id, '">', html, "</h", level, ">")
509 }

537 render <- function(block) {
538 switch(block$kind,
539 heading = heading(block$number, block$text),
540 paragraph = paste0("<p>", inline(block$text), "</p>"),
541 rule = "<hr />",
542 code = {
543 attribute <- if (nzchar(block$text)) paste0(' class="language-', escape_html(block$text), '"') else ""
544 code <- paste0(vapply(block$lines, function(line) paste0(escape_html(line), "\n"), character(1L)), collapse = "")
545 paste0("<pre><code", attribute, ">", code, "</code></pre>")
546 },
547 quote = paste0("<blockquote>\n", render_blocks(block$children), "</blockquote>"),
548 alert = {
549 kind <- tolower(block$text)
550 title <- paste0(toupper(char_at(kind, 1L)), slice(kind, 2L))
551 paste0('<div class="alert alert-', kind, '">\n<p class="alert-title">', title, "</p>\n", render_blocks(block$children), "</div>")
552 },
553 list = {
554 tag <- if (block$ordered) "ol" else "ul"
555 attribute <- if (block$ordered && block$number != 1L) paste0(' start="', block$number, '"') else ""
556 items <- paste0(vapply(block$items, function(item) paste0(render_item(item), "\n"), character(1L)), collapse = "")
557 paste0("<", tag, attribute, ">\n", items, "</", tag, ">")
558 },
559 table = {
560 html <- paste0("<table>\n<thead>\n<tr>\n", table_cells("th", block, block$lines), "</tr>\n</thead>\n")
561 if (length(block$rows)) {

571 toc_list <- function(indices) {
572 html <- "<ul>\n"
573 for (index in indices) {
574 entry <- state$headings[[index]]
575 html <- paste0(html, '<li><a href="#', entry$id, '">', entry$label, '</a>')
576 if (length(entry$children)) html <- paste0(html, "\n", toc_list(entry$children), "\n")
577 html <- paste0(html, "</li>\n")
578 }
579 paste0(html, "</ul>")

581 table_of_contents <- function() {
582 if (!length(state$headings)) return('<nav class="toc">\n</nav>')
583 roots <- integer()
584 parents <- integer()
585 for (index in seq_along(state$headings)) {
586 entry <- state$headings[[index]]
587 while (length(parents) && state$headings[[tail(parents, 1L)]]$level >= entry$level) parents <- head(parents, -1L)
588 if (!length(parents)) roots <- c(roots, index) else {
589 parent <- tail(parents, 1L)
590 state$headings[[parent]]$children <- c(state$headings[[parent]]$children, index)
591 }
592 parents <- c(parents, index)
593 }
594 paste0('<nav class="toc">\n', toc_list(roots), "\n</nav>")
595 }

605 }
606 paste0(html, "</ol>\n</section>\n")
607 }
