12const punctuation = "!\"#$%&'()*+,-./:;<=>?@[\\]^_`{|}~"
13const whitespace = " \n"

16var htmlEscaper = strings.NewReplacer("&", "&amp;", "<", "&lt;", ">", "&gt;", "\"", "&quot;")
17var listPattern = regexp.MustCompile(`^(?:([-*+])|([0-9]{1,9})([.)])) +`)
18var footnotePattern = regexp.MustCompile(`^\[\^([A-Za-z0-9_-]+)\]:(.*)$`)
19var delimiterPattern = regexp.MustCompile(`^:?-+:?$`)
20var labelPattern = regexp.MustCompile(`^[A-Za-z0-9_-]+$`)
21var emailPattern = regexp.MustCompile(`^[A-Za-z0-9.+_-]+@[A-Za-z0-9-]+(?:\.[A-Za-z0-9-]+)+$`)

23func escape(text string) string {
24 return htmlEscaper.Replace(text)
25}

27func trim(text string) string {
28 return strings.Trim(text, " ")
29}

31func alphanumeric(c byte) bool {
32 return c >= 'a' && c <= 'z' || c >= 'A' && c <= 'Z' || c >= '0' && c <= '9'
33}

35func normalize(text string) string {
36 return strings.Join(strings.Fields(strings.ToLower(text)), " ")
37}

39func skipSpace(text string, at int) int {
40 for at < len(text) && strings.ContainsRune(whitespace, rune(text[at])) {
41 at++
42 }
43 return at
44}

46func escapedRead(text string, at int, stops string) (string, int) {
47 var out strings.Builder
48 for at < len(text) && !strings.ContainsRune(stops, rune(text[at])) {
49 if text[at] == '\\' && at+1 < len(text) && strings.ContainsRune(punctuation, rune(text[at+1])) {
50 at++
51 }
52 out.WriteByte(text[at])
53 at++
54 }
55 return out.String(), at
56}

58func stripTags(text string) string {
59 var out strings.Builder
60 for len(text) > 0 {
61 start := strings.IndexByte(text, '<')
62 if start < 0 {
63 out.WriteString(text)
64 break
65 }
66 end := strings.IndexByte(text[start:], '>')
67 if end < 0 {
68 out.WriteString(text)
69 break
70 }
71 out.WriteString(text[:start])
72 text = text[start+end+1:]
73 }
74 return out.String()
75}

77func runEnd(text string, at int) int {
78 end := at + 1
79 for end < len(text) && text[end] == text[at] {
80 end++
81 }
82 return end
83}
84
85func backtickCloser(text string, at, length int) int {
86 for at < len(text) {
87 next := strings.IndexByte(text[at:], '`')
88 if next < 0 {
89 return -1
90 }
91 at += next
92 end := runEnd(text, at)
93 if end-at == length {
94 return at
95 }
96 at = end
97 }
98 return -1
99}
100
101func closingBracket(text string, at int) int {
102 depth := 0
103 for at < len(text) {
104 switch text[at] {
105 case '\\':
106 at += 2
107 continue
108 case '`':
109 end := runEnd(text, at)
110 close := backtickCloser(text, end, end-at)
111 if close < 0 {
112 at = end
113 } else {
114 at = close + end - at
115 }
116 continue
117 case '[':
118 depth++
119 case ']':
120 depth--
121 if depth == 0 {
122 return at
123 }
124 }
125 at++
126 }
127 return -1
128}

153func openingFence(line string) (string, string, bool) {
154 if len(line) == 0 || line[0] != '`' && line[0] != '~' {
155 return "", "", false
156 }
157 end := runEnd(line, 0)
158 if end < 3 {
159 return "", "", false
160 }
161 info := trim(line[end:])
162 if line[0] == '`' && strings.ContainsRune(info, '`') {
163 return "", "", false
164 }
165 return line[:end], info, true
166}

168func atxHeading(line string) *block {
169 n := 0
170 for n < len(line) && line[n] == '#' {
171 n++
172 }
173 if n < 1 || n > 6 || n < len(line) && line[n] != ' ' {
174 return nil
175 }
176 text := trim(line[n:])
177 stripped := strings.TrimRight(text, "#")
178 if stripped == "" || strings.HasSuffix(stripped, " ") {
179 text = trim(stripped)
180 }
181 return &block{kind: "heading", text: text, number: n}
182}

184func isRule(line string) bool {
185 text := strings.ReplaceAll(line, " ", "")
186 return len(text) >= 3 && strings.ContainsRune("-*_", rune(text[0])) && strings.Trim(text, text[:1]) == ""
187}
188
189func listMarker(line string) *marker {
190 if isRule(line) {
191 return nil
192 }
193 match := listPattern.FindStringSubmatch(line)
194 if match == nil {
195 return nil
196 }
197 m := &marker{kind: match[1], number: 1, content: line[len(match[0]):]}
198 if match[2] != "" {
199 m.kind = match[3]
200 m.ordered = true
201 m.number, _ = strconv.Atoi(match[2])
202 }
203 return m
204}

206func interrupts(line string) bool {
207 if line == "" || line[0] == ' ' {
208 return false
209 }
210 _, _, fence := openingFence(line)
211 return fence || atxHeading(line) != nil || isRule(line) || line[0] == '>' || listMarker(line) != nil
212}

214func splitCells(line string) []string {
215 text := trim(line)
216 if strings.HasPrefix(text, "|") {
217 text = text[1:]
218 }
219 cells := []string{}
220 var cell strings.Builder
221 finalSeparator := false
222 for at := 0; at < len(text); at++ {
223 c := text[at]
224 finalSeparator = false
225 if c == '\\' && at+1 < len(text) {
226 at++
227 if text[at] != '|' {
228 cell.WriteByte('\\')
229 }
230 cell.WriteByte(text[at])
231 } else if c == '|' {
232 cells = append(cells, trim(cell.String()))
233 cell.Reset()
234 finalSeparator = true
235 } else {
236 cell.WriteByte(c)
237 }
238 }
239 if !finalSeparator {
240 cells = append(cells, trim(cell.String()))
241 }
242 return cells
243}

250func (p *parser) definition(line string) bool {
251 if match := footnotePattern.FindStringSubmatch(line); match != nil && trim(match[2]) != "" {
252 name := strings.ToLower(match[1])
253 if _, exists := p.footnotes[name]; !exists {
254 p.footnotes[name] = trim(match[2])
255 }
256 return true
257 }
258 if !strings.HasPrefix(line, "[") {
259 return false
260 }
261 close := strings.IndexByte(line, ']')
262 if close < 0 || close+1 >= len(line) || line[close+1] != ':' {
263 return false
264 }
265 label := line[1:close]
266 if strings.Contains(label, "[") || trim(label) == "" || strings.HasPrefix(label, "^") {
267 return false
268 }
269 url, end := escapedRead(line, skipSpace(line, close+2), " ")
270 if url == "" {
271 return false
272 }
273 at := skipSpace(line, end)
274 var title *string
275 if at < len(line) {
276 if at == end || !strings.ContainsRune("\"'", rune(line[at])) {
277 return false
278 }
279 value, stop := escapedRead(line, at+1, line[at:at+1])
280 if stop != len(line)-1 {
281 return false
282 }
283 title = &value
284 }
285 key := normalize(label)
286 if _, exists := p.references[key]; !exists {
287 p.references[key] = reference{url, title}
288 }
289 return true
290}

292func (p *parser) parse(lines []string) []*block {
293 var blocks []*block
294 for at := 0; at < len(lines); {
295 line := lines[at]
296 if line == "" {
297 at++
298 continue
299 }
300 if strings.HasPrefix(line, " ") {
301 end := at
302 for end < len(lines) && (lines[end] == "" || strings.HasPrefix(lines[end], " ")) {
303 end++
304 }
305 for lines[end-1] == "" {
306 end--
307 }
308 b := &block{kind: "code"}
309 for at < end {
310 row := lines[at]
311 if row != "" {
312 row = row[4:]
313 }
314 b.lines = append(b.lines, row)
315 at++
316 }
317 blocks = append(blocks, b)
318 continue
319 }
320 if line[0] != ' ' {
321 if fence, info, ok := openingFence(line); ok {
322 language := strings.SplitN(info, " ", 2)[0]
323 b := &block{kind: "code", text: language}
324 at++
325 for at < len(lines) {
326 row := lines[at]
327 at++
328 if len(row) >= len(fence) && strings.Trim(row, fence[:1]) == "" {
329 break
330 }
331 b.lines = append(b.lines, row)
332 }
333 blocks = append(blocks, b)
334 continue
335 }
336 if heading := atxHeading(line); heading != nil {
337 blocks = append(blocks, heading)
338 at++
339 continue
340 }
341 if isRule(line) {
342 blocks = append(blocks, &block{kind: "rule"})
343 at++
344 continue
345 }
346 if line[0] == '>' {
347 var content []string
348 for at < len(lines) && strings.HasPrefix(lines[at], ">") {
349 row := lines[at][1:]
350 if strings.HasPrefix(row, " ") {
351 row = row[1:]
352 }
353 content = append(content, row)
354 at++
355 }
356 b := &block{kind: "quote"}
357 for _, kind := range []string{"NOTE", "TIP", "IMPORTANT", "WARNING", "CAUTION"} {
358 if content[0] == "[!"+kind+"]" {
359 b.kind = "alert"
360 b.text = kind
361 content = content[1:]
362 break
363 }
364 }
365 b.children = p.parse(content)
366 blocks = append(blocks, b)
367 continue
368 }
369 if m := listMarker(line); m != nil {
370 b := &block{kind: "list", number: m.number, ordered: m.ordered}
371 kind := m.kind
372 for {
373 content := []string{m.content}
374 at++
375 for at < len(lines) && strings.HasPrefix(lines[at], " ") {
376 content = append(content, lines[at][2:])
377 at++
378 }
379 b.items = append(b.items, p.parse(content))
380 next := at
381 for next < len(lines) && lines[next] == "" {
382 next++
383 }
384 if next >= len(lines) {
385 break
386 }
387 m = listMarker(lines[next])
388 if m == nil || m.kind != kind || m.ordered != b.ordered {
389 break
390 }
391 at = next
392 }
393 blocks = append(blocks, b)
394 continue
395 }
396 if p.definition(line) {
397 at++
398 continue
399 }
400 if at+1 < len(lines) && strings.Contains(line, "|") && strings.Contains(lines[at+1], "|") {
401 header, delimiters := splitCells(line), splitCells(lines[at+1])
402 valid := len(header) == len(delimiters)
403 for _, cell := range delimiters {
404 if !delimiterPattern.MatchString(cell) {
405 valid = false
406 }
407 }
408 if valid {
409 b := &block{kind: "table", lines: header}
410 for _, cell := range delimiters {
411 left, right := strings.HasPrefix(cell, ":"), strings.HasSuffix(cell, ":")
412 alignment := ""
413 if left && right {
414 alignment = "center"
415 } else if left {
416 alignment = "left"
417 } else if right {
418 alignment = "right"
419 }
420 b.alignments = append(b.alignments, alignment)
421 }
422 at += 2
423 for at < len(lines) && lines[at] != "" && strings.Contains(lines[at], "|") {
424 cells := splitCells(lines[at])
425 row := make([]string, len(delimiters))
426 copy(row, cells)
427 b.rows = append(b.rows, row)
428 at++
429 }
430 blocks = append(blocks, b)
431 continue
432 }
433 }
434 }
435 text := []string{strings.TrimLeft(line, " ")}
436 at++
437 level := 0
438 for at < len(lines) && lines[at] != "" {
439 row := lines[at]
440 if len(row) >= 3 && (row[0] == '=' || row[0] == '-') && strings.Trim(row, row[:1]) == "" {
441 level = 2
442 if row[0] == '=' {
443 level = 1
444 }
445 at++
446 break
447 }
448 if interrupts(row) {
449 break
450 }
451 text = append(text, strings.TrimLeft(row, " "))
452 at++
453 }
454 b := &block{kind: "paragraph", text: strings.Join(text, "\n"), number: level}
455 if level != 0 {
456 b.kind = "heading"
457 } else if b.text == "[TOC]" {
458 b.kind = "toc"
459 }
460 blocks = append(blocks, b)
461 }
462 return blocks
463}

465type piece struct {
466 html string
467 delimiter byte
468 length int
469 open, close bool
470}
471
472func emphasis(pieces []piece) string {
473 var out strings.Builder
474 for at := 0; at < len(pieces); at++ {
475 p := pieces[at]
476 if p.delimiter == 0 {
477 out.WriteString(p.html)
478 continue
479 }
480 end := len(pieces)
481 if p.open {
482 for next := at + 1; next < len(pieces); next++ {
483 q := pieces[next]
484 if q.delimiter == p.delimiter && q.length == p.length && q.close {
485 end = next
486 break
487 }
488 }
489 }
490 if end == len(pieces) {
491 out.WriteString(strings.Repeat(string(p.delimiter), p.length))
492 continue
493 }
494 opening, closing := "<del>", "</del>"
495 if p.delimiter != '~' {
496 switch p.length {
497 case 1:
498 opening, closing = "<em>", "</em>"
499 case 2:
500 opening, closing = "<strong>", "</strong>"
501 case 3:
502 opening, closing = "<em><strong>", "</strong></em>"
503 }
504 }
505 out.WriteString(opening)
506 out.WriteString(emphasis(pieces[at+1 : end]))
507 out.WriteString(closing)
508 at = end
509 }
510 return out.String()
511}

529func (r *renderer) destination(text string, at int, label string) (reference, int, bool) {
530 if at < len(text) && text[at] == '[' {
531 relative := strings.IndexByte(text[at+1:], ']')
532 if relative >= 0 {
533 close := at + 1 + relative
534 key := text[at+1 : close]
535 if !strings.Contains(key, "[") {
536 if key == "" {
537 key = label
538 }
539 ref, found := r.references[normalize(key)]
540 return ref, close + 1, found
541 }
542 }
543 } else if at < len(text) && text[at] == '(' {
544 url, end := escapedRead(text, skipSpace(text, at+1), " \n)")
545 next := skipSpace(text, end)
546 var title *string
547 valid := true
548 if next > end && next < len(text) && strings.ContainsRune("\"'", rune(text[next])) {
549 value, stop := escapedRead(text, next+1, text[next:next+1])
550 if stop == len(text) {
551 valid = false
552 } else {
553 title = &value
554 next = skipSpace(text, stop+1)
555 }
556 }
557 if valid && next < len(text) && text[next] == ')' {
558 return reference{url, title}, next + 1, true
559 }
560 }
561 ref, found := r.references[normalize(label)]
562 return ref, at, found
563}

573func (r *renderer) footnoteReference(name string) string {
574 number, exists := r.footnoteNumbers[name]
575 if !exists {
576 number = len(r.footnoteNames) + 1
577 r.footnoteNumbers[name] = number
578 r.footnoteNames = append(r.footnoteNames, name)
579 r.footnoteCounts = append(r.footnoteCounts, 0)
580 }
581 r.footnoteCounts[number-1]++
582 id := referenceID(number, r.footnoteCounts[number-1])
583 value := strconv.Itoa(number)
584 return `<sup id="` + id + `"><a href="#fn-` + value + `">` + value + `</a></sup>`
585}

587func (r *renderer) inline(text string, links, notes bool) string {
588 var pieces []piece
589 for at := 0; at < len(text); {
590 ch := text[at]
591 if ch == '\\' {
592 if at+1 < len(text) && strings.ContainsRune(punctuation, rune(text[at+1])) {
593 pieces = append(pieces, piece{html: escape(text[at+1 : at+2])})
594 at += 2
595 } else if at+1 < len(text) && text[at+1] == '\n' {
596 pieces = append(pieces, piece{html: "<br />\n"})
597 at += 2
598 } else {
599 pieces = append(pieces, piece{html: "\\"})
600 at++
601 }
602 continue
603 }
604 if ch == '`' {
605 end := runEnd(text, at)
606 close := backtickCloser(text, end, end-at)
607 if close < 0 {
608 pieces = append(pieces, piece{html: text[at:end]})
609 at = end
610 } else {
611 code := strings.ReplaceAll(text[end:close], "\n", " ")
612 if strings.HasPrefix(code, " ") && strings.HasSuffix(code, " ") && strings.Trim(code, " ") != "" {
613 code = code[1 : len(code)-1]
614 }
615 pieces = append(pieces, piece{html: "<code>" + escape(code) + "</code>"})
616 at = close + end - at
617 }
618 continue
619 }
620 if links && ch == '<' {
621 if offset := strings.IndexByte(text[at+1:], '>'); offset >= 0 {
622 close := at + 1 + offset
623 target := text[at+1 : close]
624 href := ""
625 if !strings.ContainsAny(target, " \n<") && (strings.HasPrefix(target, "http://") && len(target) > 7 || strings.HasPrefix(target, "https://") && len(target) > 8) {
626 href = target
627 } else if emailPattern.MatchString(target) {
628 href = "mailto:" + target
629 }
630 if href != "" {
631 pieces = append(pieces, piece{html: `<a href="` + escape(href) + `">` + escape(target) + `</a>`})
632 at = close + 1
633 continue
634 }
635 }
636 }
637 if links && (strings.HasPrefix(text[at:], "http://") || strings.HasPrefix(text[at:], "https://")) && (at == 0 || strings.ContainsRune(" \n*_~(", rune(text[at-1]))) {
638 end := at
639 for end < len(text) && !strings.ContainsRune(" \n<", rune(text[end])) {
640 end++
641 }
642 for end > at && strings.ContainsRune(".,:;!?*_~)", rune(text[end-1])) {
643 end--
644 }
645 url := text[at:end]
646 if url != "http://" && url != "https://" {
647 value := escape(url)
648 pieces = append(pieces, piece{html: `<a href="` + value + `">` + value + `</a>`})
649 at = end
650 continue
651 }
652 }
653 if ch == '[' || ch == '!' && at+1 < len(text) && text[at+1] == '[' {
654 if links && notes && strings.HasPrefix(text[at:], "[^") {
655 if offset := strings.IndexByte(text[at+2:], ']'); offset >= 0 {
656 close := at + 2 + offset
657 name := strings.ToLower(text[at+2 : close])
658 if _, found := r.footnotes[name]; found {
659 pieces = append(pieces, piece{html: r.footnoteReference(name)})
660 at = close + 1
661 continue
662 }
663 }
664 }
665 image := ch == '!'
666 opening := at
667 if image {
668 opening++
669 }
670 close := closingBracket(text, opening)
671 if (image || links) && close >= 0 {
672 label := text[opening+1 : close]
673 if ref, end, ok := r.destination(text, close+1, label); ok {
674 content := r.inline(label, false, false)
675 title := ""
676 if ref.title != nil {
677 title = ` title="` + escape(*ref.title) + `"`
678 }
679 var html string
680 if image {
681 html = `<img src="` + escape(ref.url) + `" alt="` + stripTags(content) + `"` + title + ` />`
682 } else {
683 html = `<a href="` + escape(ref.url) + `"` + title + `>` + content + `</a>`
684 }
685 pieces = append(pieces, piece{html: html})
686 at = end
687 continue
688 }
689 }
690 pieces = append(pieces, piece{html: string(ch)})
691 at++
692 continue
693 }
694 if strings.ContainsRune("*_~", rune(ch)) {
695 end := runEnd(text, at)
696 length := end - at
697 before, after := byte(' '), byte(' ')
698 if at > 0 {
699 before = text[at-1]
700 }
701 if end < len(text) {
702 after = text[end]
703 }
704 open, close := !strings.ContainsRune(whitespace, rune(after)), !strings.ContainsRune(whitespace, rune(before))
705 if ch == '_' && alphanumeric(before) && alphanumeric(after) {
706 open, close = false, false
707 }
708 if ch != '~' && length <= 3 || ch == '~' && length == 2 {
709 pieces = append(pieces, piece{delimiter: ch, length: length, open: open, close: close})
710 } else {
711 pieces = append(pieces, piece{html: text[at:end]})
712 }
713 at = end
714 continue
715 }
716 pieces = append(pieces, piece{html: escape(text[at : at+1])})
717 at++
718 }
719 return emphasis(pieces)
720}

731func (r *renderer) heading(level int, text string) string {
732 id := ""
733 start := strings.LastIndex(text, " {#")
734 if strings.HasSuffix(text, "}") && start >= 0 && labelPattern.MatchString(text[start+3:len(text)-1]) {
735 id = text[start+3 : len(text)-1]
736 text = strings.TrimRight(text[:start], " ")
737 }
738 if id == "" {
739 var slug strings.Builder
740 for _, c := range strings.ToLower(text) {
741 if c >= 'a' && c <= 'z' || c >= '0' && c <= '9' || c == '-' || c == '_' {
742 slug.WriteRune(c)
743 } else if c == ' ' || c == '\n' {
744 slug.WriteByte('-')
745 }
746 }
747 base := slug.String()
748 if base == "" {
749 base = "heading"
750 }
751 id = base
752 for suffix := 1; r.usedIDs[id]; suffix++ {
753 id = base + "-" + strconv.Itoa(suffix)
754 }
755 }
756 r.usedIDs[id] = true
757 html := r.inline(text, true, true)
758 r.headings = append(r.headings, &headingEntry{level: level, id: id, label: stripTags(html)})
759 n := strconv.Itoa(level)
760 return `<h` + n + ` id="` + id + `">` + html + `</h` + n + `>`
761}

763func (r *renderer) item(blocks []*block) string {
764 var out strings.Builder
765 out.WriteString("<li>")
766 newline := false
767 for index, b := range blocks {
768 if b.kind == "paragraph" {
769 text := b.text
770 if index == 0 && len(text) > 4 && (strings.HasPrefix(text, "[ ] ") || strings.HasPrefix(text, "[x] ") || strings.HasPrefix(text, "[X] ")) {
771 if text[1] == ' ' {
772 out.WriteString(`<input type="checkbox" disabled="" /> `)
773 } else {
774 out.WriteString(`<input type="checkbox" checked="" disabled="" /> `)
775 }
776 text = text[4:]
777 }
778 html := r.inline(text, true, true)
779 out.WriteString(html)
780 if html != "" {
781 newline = strings.HasSuffix(html, "\n")
782 }
783 } else {
784 if !newline {
785 out.WriteByte('\n')
786 }
787 out.WriteString(r.render(b))
788 out.WriteByte('\n')
789 newline = true
790 }
791 }
792 out.WriteString("</li>")
793 return out.String()
794}

796func (r *renderer) tableCells(tag string, b *block, cells []string) string {
797 var out strings.Builder
798 for index, cell := range cells {
799 out.WriteString("<" + tag)
800 if b.alignments[index] != "" {
801 out.WriteString(` style="text-align:` + b.alignments[index] + `"`)
802 }
803 out.WriteString(">" + r.inline(cell, true, true) + "</" + tag + ">\n")
804 }
805 return out.String()
806}

808func (r *renderer) render(b *block) string {
809 switch b.kind {
810 case "heading":
811 return r.heading(b.number, b.text)
812 case "paragraph":
813 return "<p>" + r.inline(b.text, true, true) + "</p>"
814 case "rule":
815 return "<hr />"
816 case "code":
817 var out strings.Builder
818 out.WriteString("<pre><code")
819 if b.text != "" {
820 out.WriteString(` class="language-` + escape(b.text) + `"`)
821 }
822 out.WriteByte('>')
823 for _, line := range b.lines {
824 out.WriteString(escape(line))
825 out.WriteByte('\n')
826 }
827 out.WriteString("</code></pre>")
828 return out.String()
829 case "quote":
830 return "<blockquote>\n" + r.blocks(b.children) + "</blockquote>"
831 case "alert":
832 kind := strings.ToLower(b.text)
833 title := strings.ToUpper(kind[:1]) + kind[1:]
834 return `<div class="alert alert-` + kind + `">` + "\n" + `<p class="alert-title">` + title + "</p>\n" + r.blocks(b.children) + "</div>"
835 case "list":
836 tag := "ul"
837 if b.ordered {
838 tag = "ol"
839 }
840 var out strings.Builder
841 out.WriteString("<" + tag)
842 if b.ordered && b.number != 1 {
843 out.WriteString(` start="` + strconv.Itoa(b.number) + `"`)
844 }
845 out.WriteString(">\n")
846 for _, item := range b.items {
847 out.WriteString(r.item(item))
848 out.WriteByte('\n')
849 }
850 out.WriteString("</" + tag + ">")
851 return out.String()
852 case "table":
853 var out strings.Builder
854 out.WriteString("<table>\n<thead>\n<tr>\n" + r.tableCells("th", b, b.lines) + "</tr>\n</thead>\n")
855 if len(b.rows) > 0 {
856 out.WriteString("<tbody>\n")
857 for _, row := range b.rows {
858 out.WriteString("<tr>\n" + r.tableCells("td", b, row) + "</tr>\n")
859 }
860 out.WriteString("</tbody>\n")
861 }
862 out.WriteString("</table>")
863 return out.String()
864 case "toc":
865 return tocMarker
866 }
867 panic("unknown block")
868}

870func tocList(entries []*headingEntry) string {
871 var out strings.Builder
872 out.WriteString("<ul>\n")
873 for _, h := range entries {
874 out.WriteString(`<li><a href="#` + h.id + `">` + h.label + `</a>`)
875 if len(h.children) > 0 {
876 out.WriteByte('\n')
877 out.WriteString(tocList(h.children))
878 out.WriteByte('\n')
879 }
880 out.WriteString("</li>\n")
881 }
882 out.WriteString("</ul>")
883 return out.String()
884}

886func (r *renderer) toc() string {
887 if len(r.headings) == 0 {
888 return "<nav class=\"toc\">\n</nav>"
889 }
890 var roots, parents []*headingEntry
891 for _, h := range r.headings {
892 for len(parents) > 0 && parents[len(parents)-1].level >= h.level {
893 parents = parents[:len(parents)-1]
894 }
895 if len(parents) == 0 {
896 roots = append(roots, h)
897 } else {
898 parent := parents[len(parents)-1]
899 parent.children = append(parent.children, h)
900 }
901 parents = append(parents, h)
902 }
903 return "<nav class=\"toc\">\n" + tocList(roots) + "\n</nav>"
904}

906func (r *renderer) footnoteList() string {
907 var out strings.Builder
908 out.WriteString("<section class=\"footnotes\">\n<ol>\n")
909 for index, name := range r.footnoteNames {
910 out.WriteString(`<li id="fn-` + strconv.Itoa(index+1) + `">` + r.inline(r.footnotes[name], true, false))
911 for count := 1; count <= r.footnoteCounts[index]; count++ {
912 out.WriteString(` <a href="#` + referenceID(index+1, count) + `">&#8617;</a>`)
913 }
914 out.WriteString("</li>\n")
915 }
916 out.WriteString("</ol>\n</section>\n")
917 return out.String()
918}

920func main() {
921 scanner := bufio.NewScanner(os.Stdin)
922 scanner.Buffer(make([]byte, 4096), 4096)
923 var lines []string
924 for scanner.Scan() {
925 line := scanner.Text()
926 if line == "@@END@@" {
927 break
928 }
929 lines = append(lines, line)
930 }
931 if err := scanner.Err(); err != nil {
932 panic(err)
933 }
934 p := &parser{references: make(map[string]reference), footnotes: make(map[string]string)}
935 blocks := p.parse(lines)
936 r := &renderer{references: p.references, footnotes: p.footnotes, usedIDs: make(map[string]bool), footnoteNumbers: make(map[string]int)}
937 html := r.blocks(blocks)
938 html = strings.ReplaceAll(html, tocMarker, r.toc())
939 if len(r.footnoteNames) > 0 {
940 html += r.footnoteList()
941 }
942 fmt.Print(html)
943}
