1local WHITESPACE = " \n"
2local PUNCTUATION = "!\"#$%&'()*+,-./:;<=>?@[\\]^_`{|}~"
3local TOC_MARKER = "\0toc\0"
4local HTML_ESCAPES = { ["&"] = "&amp;", ["<"] = "&lt;", [">"] = "&gt;", ['"'] = "&quot;" }
5local ALERT_TYPES = { NOTE = true, TIP = true, IMPORTANT = true, WARNING = true, CAUTION = true }
6
7local function starts(text, prefix, at)
8 at = at or 1
9 return text:sub(at, at + #prefix - 1) == prefix
10end
11
12local function contains(text, character)
13 return character ~= "" and text:find(character, 1, true) ~= nil
14end
15
16local function trim(text)
17 return (text:gsub("^ +", ""):gsub(" +$", ""))
18end
19
20local function escape(text)
21 return (text:gsub('[&<>"]', HTML_ESCAPES))
22end
23
24local function strip_tags(text)
25 return (text:gsub("<[^>]*>", ""))
26end
27
28local function normalize(text)
29 return (text:lower():gsub("[ \n]+", " "):gsub("^ ", ""):gsub(" $", ""))
30end
31
32local function skip_space(text, at)
33 while at <= #text and contains(WHITESPACE, text:sub(at, at)) do at = at + 1 end
34 return at
35end
36
37local function escaped_read(text, at, stops)
38 local parts = {}
39 while at <= #text and not contains(stops, text:sub(at, at)) do
40 if text:sub(at, at) == "\\" and contains(PUNCTUATION, text:sub(at + 1, at + 1)) then at = at + 1 end
41 parts[#parts + 1] = text:sub(at, at)
42 at = at + 1
43 end
44 return table.concat(parts), at
45end
46
47local function run_end(text, at)
48 local stop = at + 1
49 while stop <= #text and text:sub(stop, stop) == text:sub(at, at) do stop = stop + 1 end
50 return stop
51end
52
53local function backtick_closer(text, at, length)
54 while at <= #text do
55 at = text:find("`", at, true)
56 if not at then return nil end
57 local stop = run_end(text, at)
58 if stop - at == length then return at end
59 at = stop
60 end
61end
62
63local function closing_bracket(text, at)
64 local depth = 0
65 while at <= #text do
66 local character = text:sub(at, at)
67 if character == "\\" then at = at + 2
68 elseif character == "`" then
69 local stop = run_end(text, at)
70 local close = backtick_closer(text, stop, stop - at)
71 at = close and close + stop - at or stop
72 else
73 if character == "[" then depth = depth + 1
74 elseif character == "]" then
75 depth = depth - 1
76 if depth == 0 then return at end
77 end
78 at = at + 1
79 end
80 end
81end
82
83local function opening_fence(line)
84 if not contains("`~", line:sub(1, 1)) then return nil end
85 local stop = run_end(line, 1)
86 if stop < 4 then return nil end
87 local info = trim(line:sub(stop))
88 if line:sub(1, 1) == "`" and contains(info, "`") then return nil end
89 return line:sub(1, stop - 1), info
90end

92local function atx_heading(line)
93 local hashes = line:match("^(#+)")
94 if not hashes or #hashes > 6 or (#line > #hashes and line:sub(#hashes + 1, #hashes + 1) ~= " ") then return nil end
95 local text = trim(line:sub(#hashes + 1))
96 local stripped = text:gsub("#+$", "")
97 if stripped == "" or stripped:sub(-1) == " " then text = trim(stripped) end
98 return { kind = "heading", number = #hashes, text = text }
99end
100
101local function is_rule(line)
102 local compact = line:gsub(" ", "")
103 return #compact >= 3 and contains("-*_", compact:sub(1, 1))
104 and compact == compact:sub(1, 1):rep(#compact)
105end
106
107local function list_marker(line)
108 if is_rule(line) then return nil end
109 local marker, spaces, content = line:match("^([%-%*+])( +)(.*)$")
110 if marker then return { marker = marker, ordered = false, number = 1, content = content } end
111 local digits
112 digits, marker, spaces, content = line:match("^(%d+)([%.%)])( +)(.*)$")
113 if digits and #digits <= 9 then
114 return { marker = marker, ordered = true, number = tonumber(digits), content = content }
115 end
116end
117
118local function interrupts(line)
119 if line == "" or starts(line, " ") then return false end
120 return opening_fence(line) ~= nil or atx_heading(line) ~= nil or is_rule(line)
121 or starts(line, ">") or list_marker(line) ~= nil
122end
123
124local function split_cells(line)
125 local text = trim(line)
126 if starts(text, "|") then text = text:sub(2) end
127 local cells, parts, at, final_separator = {}, {}, 1, false
128 while at <= #text do
129 local character = text:sub(at, at)
130 final_separator = false
131 if character == "\\" and at < #text then
132 local following = text:sub(at + 1, at + 1)
133 parts[#parts + 1] = following == "|" and "|" or "\\" .. following
134 at = at + 2
135 elseif character == "|" then
136 cells[#cells + 1] = trim(table.concat(parts))
137 parts, final_separator, at = {}, true, at + 1
138 else parts[#parts + 1], at = character, at + 1 end
139 end
140 if not final_separator then cells[#cells + 1] = trim(table.concat(parts)) end
141 return cells
142end

147function Parser:definition(line)
148 local name, body = line:match("^%[%^([A-Za-z0-9_-]+)%]:(.*)$")
149 if name and trim(body) ~= "" then
150 name = name:lower()
151 if self.footnotes[name] == nil then self.footnotes[name] = trim(body) end
152 return true
153 end
154 if not starts(line, "[") then return false end
155 local close = line:find("]", 1, true)
156 if not close or line:sub(close + 1, close + 1) ~= ":" then return false end
157 local label = line:sub(2, close - 1)
158 if contains(label, "[") or trim(label) == "" or starts(label, "^") then return false end
159 local url, stop = escaped_read(line, skip_space(line, close + 2), " ")
160 if url == "" then return false end
161 local at, title = skip_space(line, stop), nil
162 if at <= #line then
163 local quote = line:sub(at, at)
164 if at == stop or not contains("\"'", quote) then return false end
165 local finish
166 title, finish = escaped_read(line, at + 1, quote)
167 if finish ~= #line then return false end
168 end
169 name = normalize(label)
170 if self.references[name] == nil then self.references[name] = { url = url, title = title } end
171 return true
172end

174function Parser:parse(lines)
175 local blocks, at = {}, 1
176 while at <= #lines do
177 local line, block = lines[at], nil
178 if line == "" then at = at + 1
179 elseif starts(line, " ") then
180 local stop = at
181 while stop <= #lines and (lines[stop] == "" or starts(lines[stop], " ")) do stop = stop + 1 end
182 while lines[stop - 1] == "" do stop = stop - 1 end
183 block = { kind = "code", text = "", lines = {} }
184 while at < stop do
185 block.lines[#block.lines + 1] = lines[at] == "" and "" or lines[at]:sub(5)
186 at = at + 1
187 end
188 else
189 if not starts(line, " ") then
190 local fence, info = opening_fence(line)
191 local heading = atx_heading(line)
192 local marker = list_marker(line)
193 if fence then
194 block = { kind = "code", text = info:match("^[^ ]*") or "", lines = {} }
195 at = at + 1
196 while at <= #lines do
197 local row = lines[at]
198 at = at + 1
199 if #row >= #fence and row == fence:sub(1, 1):rep(#row) then break end
200 block.lines[#block.lines + 1] = row
201 end
202 elseif heading then block, at = heading, at + 1
203 elseif is_rule(line) then block, at = { kind = "rule" }, at + 1
204 elseif starts(line, ">") then
205 local content = {}
206 while at <= #lines and starts(lines[at], ">") do
207 local row = lines[at]:sub(2)
208 if starts(row, " ") then row = row:sub(2) end
209 content[#content + 1], at = row, at + 1
210 end
211 block = { kind = "quote" }
212 local alert = content[1]:match("^%[!(%u+)%]$")
213 if ALERT_TYPES[alert] then block.kind, block.text = "alert", alert; table.remove(content, 1) end
214 block.children = self:parse(content)
215 elseif marker then
216 block = { kind = "list", ordered = marker.ordered, number = marker.number, items = {} }
217 local kind = marker.marker
218 while true do
219 local content = { marker.content }
220 at = at + 1
221 while at <= #lines and starts(lines[at], " ") do content[#content + 1], at = lines[at]:sub(3), at + 1 end
222 block.items[#block.items + 1] = self:parse(content)
223 local following = at
224 while following <= #lines and lines[following] == "" do following = following + 1 end
225 marker = following <= #lines and list_marker(lines[following]) or nil
226 if not marker or marker.marker ~= kind or marker.ordered ~= block.ordered then break end
227 at = following
228 end
229 elseif self:definition(line) then at = at + 1; block = false
230 elseif at < #lines and contains(line, "|") and contains(lines[at + 1], "|") then
231 local header, delimiters = split_cells(line), split_cells(lines[at + 1])
232 local valid = #header == #delimiters
233 for _, cell in ipairs(delimiters) do if not cell:match("^:?-+:?$") then valid = false end end
234 if valid then
235 block = { kind = "table", lines = header, alignments = {}, rows = {} }
236 for index, cell in ipairs(delimiters) do
237 local left, right = starts(cell, ":"), cell:sub(-1) == ":"
238 block.alignments[index] = left and (right and "center" or "left") or (right and "right" or "")
239 end
240 at = at + 2
241 while at <= #lines and lines[at] ~= "" and contains(lines[at], "|") do
242 local cells, row = split_cells(lines[at]), {}
243 for index = 1, #delimiters do row[index] = cells[index] or "" end
244 block.rows[#block.rows + 1], at = row, at + 1
245 end
246 end
247 end
248 end
249 if block == nil then
250 local text, level = { (line:gsub("^ +", "")) }, nil
251 at = at + 1
252 while at <= #lines and lines[at] ~= "" do
253 local row = lines[at]
254 if row:match("^===+$") or row:match("^%-%-%-+$") then
255 level, at = starts(row, "=") and 1 or 2, at + 1
256 break
257 end
258 if interrupts(row) then break end
259 text[#text + 1], at = row:gsub("^ +", ""), at + 1
260 end
261 local combined = table.concat(text, "\n")
262 block = { kind = level and "heading" or (combined == "[TOC]" and "toc" or "paragraph"), text = combined, number = level }
263 end
264 end
265 if block then blocks[#blocks + 1] = block end
266 end
267 return blocks
268end

270local function emphasis(pieces, start, stop)
271 local parts, at = {}, start
272 while at <= stop do
273 local piece = pieces[at]
274 if type(piece) == "string" then parts[#parts + 1] = piece
275 else
276 local close
277 if piece.open then
278 for following = at + 1, stop do
279 local other = pieces[following]
280 if type(other) == "table" and other.character == piece.character and other.length == piece.length and other.close then close = following; break end
281 end
282 end
283 if close then
284 local opening, closing = "<del>", "</del>"
285 if piece.character ~= "~" then
286 if piece.length == 1 then opening, closing = "<em>", "</em>"
287 elseif piece.length == 2 then opening, closing = "<strong>", "</strong>"
288 else opening, closing = "<em><strong>", "</strong></em>" end
289 end
290 parts[#parts + 1] = opening .. emphasis(pieces, at + 1, close - 1) .. closing
291 at = close
292 else parts[#parts + 1] = piece.character:rep(piece.length) end
293 end
294 at = at + 1
295 end
296 return table.concat(parts)
297end

299local function email(text)
300 local local_part, domain = text:match("^([^@]+)@([^@]+)$")
301 if not local_part or not local_part:match("^[A-Za-z0-9%.+_-]+$") then return false end
302 local parts = 0
303 for part in (domain .. "."):gmatch("(.-)%.") do
304 if not part:match("^[A-Za-z0-9-]+$") then return false end
305 parts = parts + 1
306 end
307 return parts >= 2
308end

313function Renderer:destination(text, at, label)
314 if text:sub(at, at) == "[" then
315 local close = text:find("]", at + 1, true)
316 if close then
317 local key = text:sub(at + 1, close - 1)
318 if not contains(key, "[") then
319 local reference = self.references[normalize(key == "" and label or key)]
320 if not reference then return nil end
321 return reference, close + 1
322 end
323 end
324 elseif text:sub(at, at) == "(" then
325 local url, stop = escaped_read(text, skip_space(text, at + 1), " \n)")
326 local following, title, valid = skip_space(text, stop), nil, true
327 local quote = text:sub(following, following)
328 if following > stop and contains("\"'", quote) then
329 local finish
330 title, finish = escaped_read(text, following + 1, quote)
331 if finish > #text then valid = false else following = skip_space(text, finish + 1) end
332 end
333 if valid and text:sub(following, following) == ")" then return { url = url, title = title }, following + 1 end
334 end
335 local reference = self.references[normalize(label)]
336 if reference then return reference, at end
337end

340 return "fnref-" .. number .. (count > 1 and "-" .. count or "")

343function Renderer:footnote_reference(name)
344 local number = self.footnote_numbers[name]
345 if not number then
346 number = #self.footnote_names + 1
347 self.footnote_numbers[name], self.footnote_names[number], self.footnote_counts[number] = number, name, 0
348 end
349 self.footnote_counts[number] = self.footnote_counts[number] + 1
350 return '<sup id="' .. reference_id(number, self.footnote_counts[number]) .. '"><a href="#fn-' .. number .. '">' .. number .. '</a></sup>'

353function Renderer:bracket(text, at, links, notes)
354 if links and notes and starts(text, "[^", at) then
355 local close = text:find("]", at + 2, true)
356 if close then
357 local name = text:sub(at + 2, close - 1):lower()
358 if self.footnotes[name] then return self:footnote_reference(name), close + 1 end
359 end
360 end
361 local image = text:sub(at, at) == "!"
362 if not image and not links then return nil end
363 local opening = image and at + 1 or at
364 local closing = closing_bracket(text, opening)
365 if not closing then return nil end
366 local label = text:sub(opening + 1, closing - 1)
367 local reference, stop = self:destination(text, closing + 1, label)
368 if not reference then return nil end
369 local content = self:inline(label, false, false)
370 local title = reference.title and ' title="' .. escape(reference.title) .. '"' or ""
371 if image then return '<img src="' .. escape(reference.url) .. '" alt="' .. strip_tags(content) .. '"' .. title .. ' />', stop end
372 return '<a href="' .. escape(reference.url) .. '"' .. title .. '>' .. content .. '</a>', stop
373end

375function Renderer:next_piece(text, at, links, notes)
376 local character, following = text:sub(at, at), text:sub(at + 1, at + 1)
377 if character == "\\" then
378 if contains(PUNCTUATION, following) then return escape(following), at + 2 end
379 if following == "\n" then return "<br />\n", at + 2 end
380 return "\\", at + 1
381 end
382 if character == "`" then
383 local stop = run_end(text, at)
384 local close = backtick_closer(text, stop, stop - at)
385 if not close then return text:sub(at, stop - 1), stop end
386 local code = text:sub(stop, close - 1):gsub("\n", " ")
387 if starts(code, " ") and code:sub(-1) == " " and code:find("[^ ]") then code = code:sub(2, -2) end
388 return "<code>" .. escape(code) .. "</code>", close + stop - at
389 end
390 if links and character == "<" then
391 local close = text:find(">", at + 1, true)
392 if close then
393 local target, href = text:sub(at + 1, close - 1), nil
394 if not target:find("[ \n<]") and ((starts(target, "http://") and #target > 7) or (starts(target, "https://") and #target > 8)) then href = target
395 elseif email(target) then href = "mailto:" .. target end
396 if href then return '<a href="' .. escape(href) .. '">' .. escape(target) .. '</a>', close + 1 end
397 end
398 end
399 if links and (starts(text, "http://", at) or starts(text, "https://", at))
400 and (at == 1 or contains(" \n*_~(", text:sub(at - 1, at - 1))) then
401 local stop = at
402 while stop <= #text and not contains(" \n<", text:sub(stop, stop)) do stop = stop + 1 end
403 while stop > at and contains(".,:;!?*_~)", text:sub(stop - 1, stop - 1)) do stop = stop - 1 end
404 local url = text:sub(at, stop - 1)
405 if url ~= "http://" and url ~= "https://" then
406 local escaped = escape(url)
407 return '<a href="' .. escaped .. '">' .. escaped .. '</a>', stop
408 end
409 end
410 if character == "[" or (character == "!" and following == "[") then
411 local html, stop = self:bracket(text, at, links, notes)
412 return html or character, stop or at + 1
413 end
414 if contains("*_~", character) then
415 local stop = run_end(text, at)
416 local length = stop - at
417 local before = at == 1 and " " or text:sub(at - 1, at - 1)
418 local after = stop > #text and " " or text:sub(stop, stop)
419 local can_open, can_close = not contains(WHITESPACE, after), not contains(WHITESPACE, before)
420 if character == "_" and before:match("^[A-Za-z0-9]$") and after:match("^[A-Za-z0-9]$") then can_open, can_close = false, false end
421 if (character ~= "~" and length <= 3) or (character == "~" and length == 2) then
422 return { character = character, length = length, open = can_open, close = can_close }, stop
423 end
424 return character:rep(length), stop
425 end
426 return escape(character), at + 1
427end

429function Renderer:inline(text, links, notes)
430 local pieces, at = {}, 1
431 while at <= #text do
432 local piece
433 piece, at = self:next_piece(text, at, links, notes)
434 pieces[#pieces + 1] = piece
435 end
436 return emphasis(pieces, 1, #pieces)
437end

445function Renderer:heading(level, text)
446 local identifier, start = nil, nil
447 local at = 1
448 while true do
449 local found = text:find(" {#", at, true)
450 if not found then break end
451 start, at = found, found + 1
452 end
453 if text:sub(-1) == "}" and start then
454 local candidate = text:sub(start + 3, -2)
455 if candidate:match("^[A-Za-z0-9_-]+$") then identifier, text = candidate, (text:sub(1, start - 1):gsub(" +$", "")) end
456 end
457 if not identifier then
458 local base = text:lower():gsub("[^a-z0-9_ \n-]", ""):gsub("[ \n]", "-")
459 if base == "" then base = "heading" end
460 identifier = base
461 local suffix = 1
462 while self.used_ids[identifier] do identifier, suffix = base .. "-" .. suffix, suffix + 1 end
463 end
464 self.used_ids[identifier] = true
465 local html = self:inline(text, true, true)
466 self.headings[#self.headings + 1] = { level = level, id = identifier, label = strip_tags(html), children = {} }
467 return '<h' .. level .. ' id="' .. identifier .. '">' .. html .. '</h' .. level .. '>'
468end

470function Renderer:item(blocks)
471 local parts, newline = { "<li>" }, false
472 for index, block in ipairs(blocks) do
473 if block.kind == "paragraph" then
474 local text = block.text
475 if index == 1 and #text > 4 and (starts(text, "[ ] ") or starts(text, "[x] ") or starts(text, "[X] ")) then
476 parts[#parts + 1] = text:sub(2, 2) == " " and '<input type="checkbox" disabled="" /> ' or '<input type="checkbox" checked="" disabled="" /> '
477 text = text:sub(5)
478 end
479 local html = self:inline(text, true, true)
480 parts[#parts + 1] = html
481 if html ~= "" then newline = html:sub(-1) == "\n" end
482 else
483 if not newline then parts[#parts + 1] = "\n" end
484 parts[#parts + 1], newline = self:render(block) .. "\n", true
485 end
486 end
487 parts[#parts + 1] = "</li>"
488 return table.concat(parts)
489end

491function Renderer:table_cells(tag, block, cells)
492 local parts = {}

494 local alignment = block.alignments[index]
495 local style = alignment ~= "" and ' style="text-align:' .. alignment .. '"' or ""

501function Renderer:render(block)
502 local kind = block.kind
503 if kind == "heading" then return self:heading(block.number, block.text)
504 elseif kind == "paragraph" then return "<p>" .. self:inline(block.text, true, true) .. "</p>"
505 elseif kind == "rule" then return "<hr />"
506 elseif kind == "code" then
507 local attribute = block.text ~= "" and ' class="language-' .. escape(block.text) .. '"' or ""
508 local parts = { "<pre><code" .. attribute .. ">" }
509 for _, line in ipairs(block.lines) do parts[#parts + 1] = escape(line) .. "\n" end
510 parts[#parts + 1] = "</code></pre>"
511 return table.concat(parts)
512 elseif kind == "quote" then return "<blockquote>\n" .. self:blocks(block.children) .. "</blockquote>"
513 elseif kind == "alert" then
514 local name = block.text:lower()
515 local title = name:sub(1, 1):upper() .. name:sub(2)
516 return '<div class="alert alert-' .. name .. '">\n<p class="alert-title">' .. title .. '</p>\n' .. self:blocks(block.children) .. '</div>'
517 elseif kind == "list" then
518 local tag = block.ordered and "ol" or "ul"
519 local attribute = block.ordered and block.number ~= 1 and ' start="' .. block.number .. '"' or ""
520 local parts = { '<' .. tag .. attribute .. '>\n' }
521 for _, item in ipairs(block.items) do parts[#parts + 1] = self:item(item) .. "\n" end
522 parts[#parts + 1] = '</' .. tag .. '>'
523 return table.concat(parts)
524 elseif kind == "table" then
525 local parts = { "<table>\n<thead>\n<tr>\n", self:table_cells("th", block, block.lines), "</tr>\n</thead>\n" }
526 if #block.rows > 0 then
527 parts[#parts + 1] = "<tbody>\n"
528 for _, row in ipairs(block.rows) do parts[#parts + 1] = "<tr>\n" .. self:table_cells("td", block, row) .. "</tr>\n" end
529 parts[#parts + 1] = "</tbody>\n"
530 end
531 parts[#parts + 1] = "</table>"
532 return table.concat(parts)
533 elseif kind == "toc" then return TOC_MARKER end
534end

536local function toc_list(entries)
537 local parts = { "<ul>\n" }
538 for _, entry in ipairs(entries) do
539 parts[#parts + 1] = '<li><a href="#' .. entry.id .. '">' .. entry.label .. '</a>'
540 if #entry.children > 0 then parts[#parts + 1] = "\n" .. toc_list(entry.children) .. "\n" end
541 parts[#parts + 1] = "</li>\n"
542 end
543 parts[#parts + 1] = "</ul>"
544 return table.concat(parts)
545end

547function Renderer:toc()
548 if #self.headings == 0 then return '<nav class="toc">\n</nav>' end
549 local roots, parents = {}, {}
550 for _, heading in ipairs(self.headings) do
551 while #parents > 0 and parents[#parents].level >= heading.level do table.remove(parents) end
552 local siblings = #parents > 0 and parents[#parents].children or roots
553 siblings[#siblings + 1], parents[#parents + 1] = heading, heading
554 end
555 return '<nav class="toc">\n' .. toc_list(roots) .. '\n</nav>'
556end

558function Renderer:footnote_list()
559 local parts = { '<section class="footnotes">\n<ol>\n' }
560 for number, name in ipairs(self.footnote_names) do
561 parts[#parts + 1] = '<li id="fn-' .. number .. '">' .. self:inline(self.footnotes[name], true, false)
562 for count = 1, self.footnote_counts[number] do parts[#parts + 1] = ' <a href="#' .. reference_id(number, count) .. '">&#8617;</a>' end
563 parts[#parts + 1] = '</li>\n'
564 end
565 parts[#parts + 1] = '</ol>\n</section>\n'
566 return table.concat(parts)

569local function main()
570 local lines = {}
571 for line in io.lines() do
