16func white(_ c: UInt8) -> Bool {
17 c == 32 || c == 10

20func alnum(_ c: UInt8) -> Bool {
21 (48...57).contains(c) || (65...90).contains(c) || (97...122).contains(c)

24func label(_ c: UInt8) -> Bool {
25 alnum(c) || c == 45 || c == 95

28func punctuation(_ c: UInt8) -> Bool {
29 (33...47).contains(c) || (58...64).contains(c) || (91...96).contains(c) || (123...126).contains(c)

43 String(s.reversed().drop(while: { $0 == " " }).reversed())

50func escape(_ s: String) -> String {
51 var html = ""
52 for c in s.utf8 {
53 switch c {
54 case 38: html += "&amp;"
55 case 60: html += "&lt;"
56 case 62: html += "&gt;"
57 case 34: html += "&quot;"
58 default: html += string([c])
59 }
60 }
61 return html
62}

73func starts(_ b: [UInt8], _ at: Int, _ s: String) -> Bool {
74 let prefix = ascii(s)
75 return at + prefix.count <= b.count && b[at..<(at + prefix.count)].elementsEqual(prefix)

78func stripTags(_ s: String) -> String {
79 let b = ascii(s)
80 var html = "", at = 0
81 while at < b.count {
82 if b[at] == 60, let end = first(b,62,at) {
83 at = end + 1
84 continue
85 }
86 html += string([b[at]])
87 at += 1
88 }
89 return html

93 s.lowercased().split(whereSeparator: {

98func skip(_ b: [UInt8], _ position: Int) -> Int {
99 var at = position
100 while at < b.count && white(b[at]) {
101 at += 1

106func escapedRead(_ b: [UInt8], _ position: Int, _ stops: [UInt8]) -> (String, Int) {
107 var at = position, result: [UInt8] = []
108 while at < b.count && !stops.contains(b[at]) {
109 if b[at] == 92 && at + 1 < b.count && punctuation(b[at+1]) {
110 at += 1
111 }
112 result.append(b[at])
113 at += 1
114 }
115 return (string(result),at)
116}

118func runEnd(_ b: [UInt8], _ position: Int) -> Int {
119 var at = position
120 let c = b[at]
121 while at < b.count && b[at] == c {
122 at += 1
123 }
124 return at
125}

127func tickCloser(_ b: [UInt8], _ position: Int, _ length: Int) -> Int? {
128 var at = position
129 while let next = first(b,96,at) {
130 let end = runEnd(b,next)
131 if end-next == length {
132 return next
133 }
134 at = end
135 }
136 return nil
137}
138
139func closingBracket(_ b: [UInt8], _ position: Int) -> Int? {
140 var at = position, depth = 0
141 while at < b.count {
142 if b[at] == 92 {
143 at += 2
144 continue
145 }
146 if b[at] == 96 {
147 let end = runEnd(b,at)
148 at = tickCloser(b,end,end-at).map {
149 $0 + end - at
150 } ?? end
151 continue
152 }
153 if b[at] == 91 {
154 depth += 1
155 }
156 if b[at] == 93 {
157 depth -= 1
158 if depth == 0 {
159 return at
160 }
161 }
162 at += 1
163 }
164 return nil
165}

190func fence(_ text: String) -> Fence? {
191 let b = ascii(text)
192 guard let c = b.first, c == 96 || c == 126 else {
193 return nil
194 }
195 let n = runEnd(b,0), info = trim(part(b,runEnd(b,0)))
196 guard n >= 3 && (c != 96 || !info.contains("`")) else {
197 return nil
198 }
199 return Fence(char: c,length: n,language: info.components(separatedBy: " ")[0])
200}

202func atx(_ line: String) -> Block? {
203 let b = ascii(line), n = line.prefix(while: {
204 $0 == "#"
205 }).count
206 guard n >= 1 && n <= 6 && (n == b.count || b[n] == 32) else {
207 return nil
208 }
209 var text = trim(part(b,n))
210 let unmarked = String(text.reversed().drop(while: {
211 $0 == "#"
212 }).reversed())
213 if unmarked.isEmpty || unmarked.hasSuffix(" ") {
214 text = trim(unmarked)
215 }
216 return .heading(n,text)
217}

219func rule(_ text: String) -> Bool {
220 let b = ascii(text).filter {
221 $0 != 32
222 }
223 return b.count >= 3 && [UInt8(45),42,95].contains(b[0]) && b.allSatisfy {
224 $0 == b[0]
225 }
226}
227
228func marker(_ text: String) -> Marker? {
229 let b = ascii(text)
230 guard let c = b.first, !rule(text) else {
231 return nil
232 }
233 let kind: UInt8, number: Int?
234 var at: Int
235 if [UInt8(45),42,43].contains(c) {
236 kind = c
237 number = nil
238 at = 1
239 } else {
240 let n = b.prefix(while: {
241 (48...57).contains($0)
242 }).count
243 guard n >= 1 && n <= 9 && n < b.count && [UInt8(46),41].contains(b[n]) else {
244 return nil
245 }
246 kind = b[n]
247 number = Int(part(b,0,n))!
248 at = n+1
249 }
250 guard at < b.count && b[at] == 32 else {
251 return nil
252 }
253 while at < b.count && b[at] == 32 {
254 at += 1
255 }
256 return Marker(kind: kind,number: number,content: part(b,at))
257}

259func interrupt(_ text: String) -> Bool {
260 !text.hasPrefix(" ") && (fence(text) != nil || atx(text) != nil || rule(text) || text.hasPrefix(">") || marker(text) != nil)

263func cells(_ line: String) -> [String] {
264 var b = ascii(trim(line))
265 if b.first == 124 {
266 b.removeFirst()
267 }
268 var result: [[UInt8]] = [[]], at = 0, separator = false
269 while at < b.count {
270 let c = b[at]
271 at += 1
272 separator = false
273 if c == 92 && at < b.count {
274 if b[at] != 124 {
275 result[result.count-1].append(92)
276 }
277 result[result.count-1].append(b[at])
278 at += 1
279 } else if c == 124 {
280 result.append([])
281 separator = true
282 } else {
283 result[result.count-1].append(c)
284 }
285 }
286 if separator {
287 result.removeLast()
288 }
289 return result.map {
290 trim(string($0))
291 }

294func alignment(_ text: String) -> String? {
295 var core = text
296 if core.hasPrefix(":") {
297 core.removeFirst()
298 }
299 if core.hasSuffix(":") {
300 core.removeLast()
301 }
302 guard !core.isEmpty && core.allSatisfy({
303 $0 == "-"
304 }) else {
305 return nil
306 }
307 if text.hasPrefix(":") {
308 return text.hasSuffix(":") ? "center" : "left"
309 }
310 return text.hasSuffix(":") ? "right" : ""

313final class Parser {
314 var references: [String:Reference] = [:]
315 var footnotes: [String:String] = [:]

317 func definition(_ text: String) -> Bool {
318 let b = ascii(text)
319 if text.hasPrefix("[^"), let end = first(b,93,2), starts(b,end,"]:") {
320 let name = part(b,2,end), body = trim(part(b,end+2))
321 if !name.isEmpty && name.utf8.allSatisfy(label) && !body.isEmpty {
322 if footnotes[lower(name)] == nil {
323 footnotes[lower(name)] = body
324 }
325 return true
326 }
327 }
328 guard text.hasPrefix("["), let close = first(b,93,0), starts(b,close+1,":") else {
329 return false
330 }
331 let name = part(b,1,close)
332 guard !name.contains("[") && !trim(name).isEmpty && !name.hasPrefix("^") else {
333 return false
334 }
335 let (url,end) = escapedRead(b,skip(b,close+2),[32])
336 guard !url.isEmpty else {
337 return false
338 }
339 let at = skip(b,end)
340 var title: String? = nil
341 if at < b.count {
342 guard at > end && [UInt8(34),39].contains(b[at]) else {
343 return false
344 }
345 let (value,stop) = escapedRead(b,at+1,[b[at]])
346 guard stop == b.count-1 else {
347 return false
348 }
349 title = value
350 }
351 if references[normalize(name)] == nil {
352 references[normalize(name)] = Reference(url: url,title: title)
353 }
354 return true
355 }

357 func parse(_ lines: [String]) -> [Block] {
358 var blocks: [Block] = [], at = 0
359 while at < lines.count {
360 let line = lines[at]
361 if line.isEmpty {
362 at += 1
363 continue
364 }
365 if line.hasPrefix(" ") {
366 let start = at
367 while at < lines.count && (lines[at].isEmpty || lines[at].hasPrefix(" ")) {
368 at += 1
369 }
370 while lines[at-1].isEmpty {
371 at -= 1
372 }
373 blocks.append(.code("",lines[start..<at].map {
374 $0.isEmpty ? "" : String($0.dropFirst(4))
375 }))
376 continue
377 }
378 if !line.hasPrefix(" ") {
379 if let fence = fence(line) {
380 at += 1
381 var code: [String] = []
382 while at < lines.count {
383 let row = lines[at]
384 at += 1
385 if row.utf8.count >= fence.length && row.utf8.allSatisfy({
386 $0 == fence.char
387 }) {
388 break
389 }
390 code.append(row)
391 }
392 blocks.append(.code(fence.language,code))
393 continue
394 }
395 if let heading = atx(line) {
396 blocks.append(heading)
397 at += 1
398 continue
399 }
400 if rule(line) {
401 blocks.append(.rule)
402 at += 1
403 continue
404 }
405 if line.hasPrefix(">") {
406 var content: [String] = []
407 while at < lines.count && lines[at].hasPrefix(">") {
408 var row = String(lines[at].dropFirst())
409 if row.hasPrefix(" ") {
410 row.removeFirst()
411 }
412 content.append(row)
413 at += 1
414 }
415 let kind = content[0].hasPrefix("[!") && content[0].hasSuffix("]") ? String(content[0].dropFirst(2).dropLast()) : ""
416 if ["NOTE","TIP","IMPORTANT","WARNING","CAUTION"].contains(kind) {
417 blocks.append(.alert(kind,parse(Array(content.dropFirst()))))
418 } else {
419 blocks.append(.quote(parse(content)))
420 }
421 continue
422 }
423 if let first = marker(line) {
424 var item = first, items: [[Block]] = []
425 while true {
426 var content = [item.content]
427 at += 1
428 while at < lines.count && lines[at].hasPrefix(" ") {
429 content.append(String(lines[at].dropFirst(2)))
430 at += 1
431 }
432 items.append(parse(content))
433 var next = at
434 while next < lines.count && lines[next].isEmpty {
435 next += 1
436 }
437 guard next < lines.count, let following = marker(lines[next]), following.kind == first.kind, (following.number != nil) == (first.number != nil) else {
438 break
439 }
440 item = following
441 at = next
442 }
443 blocks.append(.list(first.number,items))
444 continue
445 }
446 if definition(line) {
447 at += 1
448 continue
449 }
450 if at+1 < lines.count && line.contains("|") && lines[at+1].contains("|") {
451 let header = cells(line), delimiters = cells(lines[at+1]), align = delimiters.map(alignment)
452 if header.count == delimiters.count && align.allSatisfy({
453 $0 != nil
454 }) {
455 at += 2
456 var rows: [[String]] = []
457 while at < lines.count && lines[at].contains("|") {
458 var row = Array(cells(lines[at]).prefix(header.count))
459 at += 1
460 while row.count < header.count {
461 row.append("")
462 }
463 rows.append(row)
464 }
465 blocks.append(.table(align.map {
466 $0!
467 },header,rows))
468 continue
469 }
470 }
471 }
472 var text = [leftTrim(line)], level = 0
473 at += 1
474 while at < lines.count && !lines[at].isEmpty {
475 let row = lines[at], b = ascii(lines[at])
476 if b.count >= 3 && [UInt8(61),45].contains(b[0]) && b.allSatisfy({
477 $0 == b[0]
478 }) {
479 level = b[0] == 61 ? 1 : 2
480 at += 1
481 break
482 }
483 if interrupt(row) {
484 break
485 }
486 text.append(leftTrim(row))
487 at += 1
488 }
489 let combined = text.joined(separator: "\n")
490 blocks.append(level > 0 ? .heading(level,combined) : (combined == "[TOC]" ? .toc : .paragraph(combined)))
491 }
492 return blocks
493 }
494}

496struct Piece {
497 var html: String = ""
498 var delimiter: UInt8 = 0
499 var length: Int = 0
500 var open: Bool = false
501 var close: Bool = false
502}

504func emphasis(_ pieces: [Piece], _ start: Int = 0, _ end: Int? = nil) -> String {
505 let end = end ?? pieces.count
506 var html = "", at = start
507 while at < end {
508 let piece = pieces[at]
509 if piece.delimiter == 0 {
510 html += piece.html
511 at += 1
512 continue
513 }
514 var closer: Int? = nil
515 if piece.open {
516 closer = (at+1..<end).first {
517 pieces[$0].delimiter == piece.delimiter && pieces[$0].length == piece.length && pieces[$0].close
518 }
519 }
520 if let close = closer {
521 let tags: (String,String)
522 if piece.delimiter == 126 {
523 tags = ("<del>","</del>")
524 } else if piece.length == 1 {
525 tags = ("<em>","</em>")
526 } else if piece.length == 2 {
527 tags = ("<strong>","</strong>")
528 } else {
529 tags = ("<em><strong>","</strong></em>")
530 }
531 html += tags.0 + emphasis(pieces,at+1,close) + tags.1
532 at = close+1
533 } else {
534 html += string(Array(repeating: piece.delimiter,count: piece.length))
535 at += 1
536 }
537 }
538 return html
539}

541func email(_ text: String) -> Bool {
542 let parts = text.components(separatedBy: "@")
543 guard parts.count == 2 && !parts[0].isEmpty && parts[0].utf8.allSatisfy({
544 alnum($0) || [UInt8(46),43,45,95].contains($0)
545 }) else {
546 return false
547 }
548 let domain = parts[1].components(separatedBy: ".")
549 return domain.count >= 2 && domain.allSatisfy {
550 !$0.isEmpty && $0.utf8.allSatisfy {
551 alnum($0) || $0 == 45
552 }
553 }

571 func destination(_ b: [UInt8], _ at: Int, _ content: String) -> (Reference,Int)? {
572 if starts(b,at,"[") {
573 if let end = first(b,93,at+1) {
574 let name = part(b,at+1,end)
575 if !name.contains("[") {
576 guard let reference = references[normalize(name.isEmpty ? content : name)] else {
577 return nil
578 }
579 return (reference,end+1)
580 }
581 }
582 } else if starts(b,at,"(") {
583 let (url,end) = escapedRead(b,skip(b,at+1),[32,10,41])
584 var next = skip(b,end), title: String? = nil, valid = true
585 if next > end && next < b.count && [UInt8(34),39].contains(b[next]) {
586 let (value,stop) = escapedRead(b,next+1,[b[next]])
587 if stop == b.count {
588 valid = false
589 } else {
590 title = value
591 next = skip(b,stop+1)
592 }
593 }
594 if valid && starts(b,next,")") {
595 return (Reference(url: url,title: title),next+1)
596 }
597 }
598 return references[normalize(content)].map {
599 ($0,at)
600 }
601 }

603 func inline(_ text: String, links: Bool = true, notes: Bool = true) -> String {
604 let b = ascii(text)
605 var pieces: [Piece] = [], at = 0
606 while at < b.count {
607 let c = b[at]
608 if c == 92 {
609 if at+1 < b.count && punctuation(b[at+1]) {
610 pieces.append(Piece(html: escape(part(b,at+1,at+2))))
611 at += 2
612 } else if at+1 < b.count && b[at+1] == 10 {
613 pieces.append(Piece(html: "<br />\n"))
614 at += 2
615 } else {
616 pieces.append(Piece(html: "\\"))
617 at += 1
618 }
619 continue
620 }
621 if c == 96 {
622 let end = runEnd(b,at)
623 if let close = tickCloser(b,end,end-at) {
624 var code = part(b,end,close).replacingOccurrences(of: "\n",with: " ")
625 if code.hasPrefix(" ") && code.hasSuffix(" ") && code.contains(where: {
626 $0 != " "
627 }) {
628 code = String(code.dropFirst().dropLast())
629 }
630 pieces.append(Piece(html: "<code>\(escape(code))</code>"))
631 at = close+end-at
632 } else {
633 pieces.append(Piece(html: part(b,at,end)))
634 at = end
635 }
636 continue
637 }
638 if links && c == 60, let close = first(b,62,at+1) {
639 let target = part(b,at+1,close), clean = !b[at+1..<close].contains {
640 white($0) || $0 == 60
641 }
642 var href: String? = nil
643 if clean && ((target.hasPrefix("http://") && target.utf8.count > 7) || (target.hasPrefix("https://") && target.utf8.count > 8)) {
644 href = target
645 } else if email(target) {
646 href = "mailto:" + target
647 }
648 if let href {
649 pieces.append(Piece(html: "<a href=\"\(escape(href))\">\(escape(target))</a>"))
650 at = close+1
651 continue
652 }
653 }
654 if links && (starts(b,at,"http://") || starts(b,at,"https://")) && (at == 0 || [UInt8(32),10,42,95,126,40].contains(b[at-1])) {
655 var end = at
656 while end < b.count && ![UInt8(32),10,60].contains(b[end]) {
657 end += 1
658 }
659 while end > at && ascii(".,:;!?*_~)").contains(b[end-1]) {
660 end -= 1
661 }
662 let url = part(b,at,end)
663 if url != "http://" && url != "https://" {
664 pieces.append(Piece(html: "<a href=\"\(escape(url))\">\(escape(url))</a>"))
665 at = end
666 continue
667 }
668 }
669 if c == 91 || (c == 33 && starts(b,at+1,"[")) {
670 if links && notes && starts(b,at,"[^"), let close = first(b,93,at+2) {
671 let name = lower(part(b,at+2,close))
672 if footnotes[name] != nil {
673 pieces.append(Piece(html: noteReference(name)))
674 at = close+1
675 continue
676 }
677 }
678 let image = c == 33, opening = at + (c == 33 ? 1 : 0)
679 if image || links, let close = closingBracket(b,opening) {
680 let content = part(b,opening+1,close)
681 if let (reference,end) = destination(b,close+1,content) {
682 let body = inline(content,links: false,notes: false), url = escape(reference.url)
683 let title = reference.title.map {
684 " title=\"\(escape($0))\""
685 } ?? ""
686 let html = image ? "<img src=\"\(url)\" alt=\"\(stripTags(body))\"\(title) />" : "<a href=\"\(url)\"\(title)>\(body)</a>"
687 pieces.append(Piece(html: html))
688 at = end
689 continue
690 }
691 }
692 }
693 if [UInt8(42),95,126].contains(c) {
694 let end = runEnd(b,at), n = end-at, before: UInt8 = at == 0 ? 32 : b[at-1], after: UInt8 = end == b.count ? 32 : b[end]
695 var open = !white(after), close = !white(before)
696 if c == 95 && alnum(before) && alnum(after) {
697 open = false
698 close = false
699 }
700 if (c != 126 && n <= 3) || (c == 126 && n == 2) {
701 pieces.append(Piece(delimiter: c,length: n,open: open,close: close))
702 } else {
703 pieces.append(Piece(html: part(b,at,end)))
704 }
705 at = end
706 continue
707 }
708 pieces.append(Piece(html: escape(string([c]))))
709 at += 1
710 }
711 return emphasis(pieces)
712 }

715 var html = ""
716 for block in blocks {
717 html += render(block) + "\n"
718 }
719 return html

722 func render(_ block: Block) -> String {
723 switch block {
724 case let .heading(level,text): return heading(level,text)
725 case let .paragraph(text): return "<p>\(inline(text))</p>"
726 case .rule: return "<hr />"
727 case .toc: return tocMarker
728 case let .code(language,lines):
729 let attribute = language.isEmpty ? "" : " class=\"language-\(escape(language))\""
730 return "<pre><code\(attribute)>" + lines.map {
731 escape($0) + "\n"
732 }.joined() + "</code></pre>"
733 case let .quote(content): return "<blockquote>\n\(blocks(content))</blockquote>"
734 case let .alert(kind,content):
735 let name = lower(kind), title = String(kind.prefix(1)) + String(name.dropFirst())
736 return "<div class=\"alert alert-\(name)\">\n<p class=\"alert-title\">\(title)</p>\n\(blocks(content))</div>"
737 case let .list(start,items):
738 let tag = start == nil ? "ul" : "ol", attribute = start.flatMap {
739 $0 == 1 ? nil : " start=\"\($0)\""
740 } ?? ""
741 return "<\(tag)\(attribute)>\n" + items.map {
742 item($0) + "\n"
743 }.joined() + "</\(tag)>"
744 case let .table(align,header,rows):
745 var html = "<table>\n<thead>\n<tr>\n" + tableCells("th",align,header) + "</tr>\n</thead>\n"
746 if !rows.isEmpty {
747 html += "<tbody>\n"
748 for row in rows {
749 html += "<tr>\n" + tableCells("td",align,row) + "</tr>\n"
750 }
751 html += "</tbody>\n"
752 }
753 return html + "</table>"
754 }
755 }

757 func item(_ blocks: [Block]) -> String {
758 var html = "<li>"
759 for (i,block) in blocks.enumerated() {
760 if case let .paragraph(text) = block {
761 let task = i == 0 && text.utf8.count > 4 && ["[ ] ","[x] ","[X] "].contains {
762 text.hasPrefix($0)
763 }
764 if task {
765 html += text.hasPrefix("[ ] ") ? "<input type=\"checkbox\" disabled=\"\" /> " : "<input type=\"checkbox\" checked=\"\" disabled=\"\" /> "
766 }
767 html += inline(task ? String(text.dropFirst(4)) : text)
768 } else {
769 if !html.hasSuffix("\n") {
770 html += "\n"
771 }
772 html += render(block) + "\n"
773 }
774 }
775 return html + "</li>"
776 }

787 func heading(_ level: Int, _ source: String) -> String {
788 var text = source, id: String? = nil
789 if source.hasSuffix("}"), let range = source.range(of: " {#",options: .backwards) {
790 let candidate = String(source[range.upperBound..<source.index(before: source.endIndex)])
791 if !candidate.isEmpty && candidate.utf8.allSatisfy(label) {
792 id = candidate
793 text = rightTrim(String(source[..<range.lowerBound]))
794 }
795 }
796 if id == nil {
797 var base = string(ascii(lower(source)).compactMap {
798 label($0) ? $0 : (white($0) ? UInt8(45) : nil)
799 })
800 if base.isEmpty {
801 base = "heading"
802 }
803 var candidate = base, suffix = 1
804 while usedIds.contains(candidate) {
805 candidate = "\(base)-\(suffix)"
806 suffix += 1
807 }
808 id = candidate
809 }
810 let identifier = id!
811 usedIds.insert(identifier)
812 let html = inline(text)
813 headings.append(HeadingEntry(level: level,id: identifier,text: stripTags(html)))
814 return "<h\(level) id=\"\(identifier)\">\(html)</h\(level)>"
815 }

818 var html = "<ul>\n"
819 for i in entries {
820 let heading = headings[i]
821 html += "<li><a href=\"#\(heading.id)\">\(heading.text)</a>"
822 if !children[i].isEmpty {
823 html += "\n" + tocList(children,children[i]) + "\n"
824 }
825 html += "</li>\n"
826 }

830 func toc() -> String {
831 if headings.isEmpty {
832 return "<nav class=\"toc\">\n</nav>"
833 }
834 var roots: [Int] = [], parents: [Int] = [], children = Array(repeating: [Int](),count: headings.count)
835 for (i,heading) in headings.enumerated() {
836 while let p = parents.last, headings[p].level >= heading.level {
837 parents.removeLast()
838 }
839 if let p = parents.last {
840 children[p].append(i)
841 } else {
842 roots.append(i)
843 }
844 parents.append(i)
845 }
846 return "<nav class=\"toc\">\n" + tocList(children,roots) + "\n</nav>"
847 }
