19bool contains(const string& text, char ch) { return text.find(ch) != string::npos; }
20bool starts(const string& text, const string& prefix, std::size_t at = 0) {
21 return text.compare(at, prefix.size(), prefix) == 0;
22}
23bool alphanumeric(char ch) {
24 return (ch >= 'a' && ch <= 'z') || (ch >= 'A' && ch <= 'Z') || (ch >= '0' && ch <= '9');
25}
26bool labelCharacter(char ch) { return alphanumeric(ch) || ch == '-' || ch == '_'; }
27string lower(string text) {
28 for (char& ch : text) ch = static_cast<char>(std::tolower(static_cast<unsigned char>(ch)));
29 return text;
30}
31string trim(const string& text) {
32 std::size_t first = text.find_first_not_of(' ');
33 if (first == string::npos) return "";
34 return text.substr(first, text.find_last_not_of(' ') - first + 1);
35}
36string leftTrim(const string& text) {
37 std::size_t first = text.find_first_not_of(' ');
38 return first == string::npos ? "" : text.substr(first);
39}
40string join(const Strings& values, const string& separator) {
41 string result;
42 for (std::size_t i = 0; i < values.size(); ++i) {
43 if (i) result += separator;
44 result += values[i];
45 }
46 return result;
47}
48string escapeHtml(const string& text) {
49 string html;
50 for (char ch : text) {
51 switch (ch) {
52 case '&': html += "&amp;"; break;
53 case '<': html += "&lt;"; break;
54 case '>': html += "&gt;"; break;
55 case '"': html += "&quot;"; break;
56 default: html += ch;
57 }
58 }
59 return html;
60}
61string stripTags(const string& text) {
62 static const std::regex tag("<[^>]*>");
63 return std::regex_replace(text, tag, "");
64}
65string normalize(const string& text) {
66 string result;
67 bool space = false;
68 for (char ch : lower(text)) {
69 if (contains(whitespace, ch)) {
70 space = !result.empty();
71 } else {
72 if (space) result += ' ';
73 result += ch;
74 space = false;
75 }
76 }
77 return result;
78}
79std::size_t skipSpace(const string& text, std::size_t at) {
80 while (at < text.size() && contains(whitespace, text[at])) ++at;
81 return at;
82}
83std::pair<string, std::size_t> escapedRead(const string& text, std::size_t at, const string& stops) {
84 string result;
85 while (at < text.size() && !contains(stops, text[at])) {
86 if (text[at] == '\\' && at + 1 < text.size() && contains(punctuation, text[at + 1])) ++at;
87 result += text[at++];
88 }
89 return {result, at};
90}

120std::optional<std::pair<string, string>> openingFence(const string& line) {
121 if (line.empty() || (line[0] != '`' && line[0] != '~')) return std::nullopt;
122 std::size_t end = line.find_first_not_of(line[0]);
123 if (end == string::npos) end = line.size();
124 if (end < 3) return std::nullopt;
125 string info = trim(line.substr(end));
126 if (line[0] == '`' && contains(info, '`')) return std::nullopt;
127 return std::make_pair(line.substr(0, end), info);
128}
129std::optional<Block> atxHeading(const string& line) {
130 std::size_t n = 0;
131 while (n < line.size() && line[n] == '#') ++n;
132 if (n < 1 || n > 6 || (n < line.size() && line[n] != ' ')) return std::nullopt;
133 string text = trim(line.substr(n));
134 std::size_t last = text.find_last_not_of('#');
135 if (last == string::npos) text.clear();
136 else if (text[last] == ' ') text = trim(text.substr(0, last + 1));
137 return Block(Kind::Heading, text, static_cast<int>(n));
138}
139bool isRule(const string& line) {
140 string compact;
141 for (char ch : line) if (ch != ' ') compact += ch;
142 return compact.size() >= 3 && contains("-*_", compact[0])
143 && compact.find_first_not_of(compact[0]) == string::npos;
144}
145std::optional<ListMarker> listMarker(const string& line) {
146 if (isRule(line)) return std::nullopt;
147 static const std::regex item(R"((?:([-*+])|([0-9]{1,9})([.)])) +)");
148 std::smatch match;
149 if (!std::regex_search(line, match, item, std::regex_constants::match_continuous)) return std::nullopt;
150 bool ordered = match[2].matched;
151 return ListMarker{ordered ? match.str(3)[0] : match.str(1)[0], ordered,
152 ordered ? std::stoi(match.str(2)) : 1, line.substr(match.length())};
153}
154bool interruptsParagraph(const string& line) {
155 return !line.empty() && line[0] != ' ' &&
156 (openingFence(line).has_value() || atxHeading(line).has_value() || isRule(line)
157 || line[0] == '>' || listMarker(line).has_value());
158}
159Strings splitCells(const string& line) {
160 string text = trim(line);
161 std::size_t at = !text.empty() && text[0] == '|' ? 1 : 0;
162 Strings cells(1);
163 bool finalSeparator = false;
164 while (at < text.size()) {
165 char ch = text[at++];
166 finalSeparator = false;
167 if (ch == '\\' && at < text.size()) {
168 if (text[at] != '|') cells.back() += '\\';
169 cells.back() += text[at++];
170 } else if (ch == '|') {
171 cells.emplace_back();
172 finalSeparator = true;
173 } else cells.back() += ch;
174 }
175 if (finalSeparator) cells.pop_back();
176 for (string& cell : cells) cell = trim(cell);
177 return cells;
178}

180class BlockParser {
181public:
182 std::map<string, Reference> references;

185 std::vector<Block> parse(const Strings& lines) {
186 std::vector<Block> blocks;
187 std::size_t at = 0;
188 while (at < lines.size()) {
189 const string& line = lines[at];
190 if (line.empty()) { ++at; continue; }
191 if (starts(line, " ")) {
192 Block code(Kind::Code);
193 std::size_t end = at;
194 while (end < lines.size() && (lines[end].empty() || starts(lines[end], " "))) ++end;
195 while (end > at && lines[end - 1].empty()) --end;
196 while (at < end) {
197 code.lines.push_back(lines[at].empty() ? "" : lines[at].substr(4));
198 ++at;
199 }
200 blocks.push_back(std::move(code));
201 continue;
202 }
203 if (line[0] != ' ') {
204 if (std::optional<std::pair<string, string>> fence = openingFence(line)) {
205 Block code(Kind::Code, fence->second.substr(0, fence->second.find(' ')));
206 ++at;
207 while (at < lines.size()) {
208 const string& row = lines[at++];
209 if (row.size() >= fence->first.size() && row.find_first_not_of(fence->first[0]) == string::npos) break;
210 code.lines.push_back(row);
211 }
212 blocks.push_back(std::move(code));
213 continue;
214 }
215 if (std::optional<Block> heading = atxHeading(line)) {
216 blocks.push_back(std::move(*heading)); ++at; continue;
217 }
218 if (isRule(line)) { blocks.emplace_back(Kind::Rule); ++at; continue; }
219 if (line[0] == '>') {
220 Strings content;
221 while (at < lines.size() && starts(lines[at], ">")) {
222 string row = lines[at++].substr(1);
223 if (starts(row, " ")) row.erase(0, 1);
224 content.push_back(std::move(row));
225 }
226 static const std::regex alert(R"(\[!(NOTE|TIP|IMPORTANT|WARNING|CAUTION)\])");
227 std::smatch match;
228 Block quote(Kind::Quote);
229 if (std::regex_match(content.front(), match, alert)) {
230 quote.kind = Kind::Alert;
231 quote.text = match.str(1);
232 content.erase(content.begin());
233 }
234 quote.children = parse(content);
235 blocks.push_back(std::move(quote));
236 continue;
237 }
238 if (std::optional<ListMarker> marker = listMarker(line)) {
239 Block list(Kind::List, "", marker->number);
240 list.ordered = marker->ordered;
241 const char kind = marker->marker;
242 while (true) {
243 Strings content{marker->content};
244 ++at;
245 while (at < lines.size() && starts(lines[at], " ")) content.push_back(lines[at++].substr(2));
246 list.items.push_back(parse(content));
247 std::size_t next = at;
248 while (next < lines.size() && lines[next].empty()) ++next;
249 marker = next < lines.size() ? listMarker(lines[next]) : std::nullopt;
250 if (!marker || marker->marker != kind || marker->ordered != list.ordered) break;
251 at = next;
252 }
253 blocks.push_back(std::move(list));
254 continue;
255 }
256 if (definition(line)) { ++at; continue; }
257 if (at + 1 < lines.size() && contains(line, '|') && contains(lines[at + 1], '|')) {
258 Strings header = splitCells(line), delimiters = splitCells(lines[at + 1]);
259 static const std::regex delimiter(R"(:?-+:?)");
260 if (header.size() == delimiters.size() && std::all_of(delimiters.begin(), delimiters.end(),
261 [&](const string& cell) { return std::regex_match(cell, delimiter); })) {
262 Block table(Kind::Table);
263 table.lines = std::move(header);
264 for (const string& cell : delimiters) {
265 bool left = cell.front() == ':', right = cell.back() == ':';
266 table.alignments.push_back(left ? (right ? "center" : "left") : (right ? "right" : ""));
267 }
268 at += 2;
269 while (at < lines.size() && contains(lines[at], '|')) {
270 Strings row = splitCells(lines[at++]);
271 row.resize(delimiters.size());
272 table.rows.push_back(std::move(row));
273 }
274 blocks.push_back(std::move(table));
275 continue;
276 }
277 }
278 }
279 Strings text{leftTrim(line)};
280 ++at;
281 int level = 0;
282 while (at < lines.size() && !lines[at].empty()) {
283 const string& row = lines[at];
284 if (row.size() >= 3 && (row[0] == '=' || row[0] == '-') && row.find_first_not_of(row[0]) == string::npos) {
285 level = row[0] == '=' ? 1 : 2;
286 ++at;
287 break;
288 }
289 if (interruptsParagraph(row)) break;
290 text.push_back(leftTrim(row));
291 ++at;
292 }
293 string combined = join(text, "\n");
294 blocks.emplace_back(level ? Kind::Heading : (combined == "[TOC]" ? Kind::Toc : Kind::Paragraph), combined, level);
295 }
296 return blocks;
297 }

299private:
300 bool definition(const string& line) {
301 static const std::regex footnote(R"(\[\^([A-Za-z0-9_-]+)\]:(.*))");
302 std::smatch match;
303 if (std::regex_match(line, match, footnote) && !trim(match.str(2)).empty()) {
304 footnotes.emplace(lower(match.str(1)), trim(match.str(2)));
305 return true;
306 }
307 if (line[0] != '[') return false;
308 std::size_t close = line.find(']');
309 if (close == string::npos || close + 1 >= line.size() || line[close + 1] != ':') return false;
310 string label = line.substr(1, close - 1);
311 if (contains(label, '[') || trim(label).empty() || label[0] == '^') return false;
312 std::pair<string, std::size_t> url = escapedRead(line, skipSpace(line, close + 2), " ");
313 if (url.first.empty()) return false;
314 std::size_t at = skipSpace(line, url.second);
315 std::optional<string> title;
316 if (at < line.size()) {
317 if (at == url.second || !contains("\"'", line[at])) return false;
318 std::pair<string, std::size_t> quoted = escapedRead(line, at + 1, string(1, line[at]));
319 if (quoted.second != line.size() - 1) return false;
320 title = quoted.first;
321 }
322 references.emplace(normalize(label), Reference{url.first, title});
323 return true;
324 }
325};

327std::size_t runEnd(const string& text, std::size_t at) {
328 char ch = text[at++];
329 while (at < text.size() && text[at] == ch) ++at;
330 return at;
331}
332std::size_t backtickCloser(const string& text, std::size_t at, std::size_t length) {
333 while ((at = text.find('`', at)) != string::npos) {
334 std::size_t end = runEnd(text, at);
335 if (end - at == length) return at;
336 at = end;
337 }
338 return string::npos;
339}
340std::size_t closingBracket(const string& text, std::size_t at) {
341 int depth = 0;
342 while (at < text.size()) {
343 char ch = text[at];
344 if (ch == '\\') { at += 2; continue; }
345 if (ch == '`') {
346 std::size_t end = runEnd(text, at), close = backtickCloser(text, end, end - at);
347 at = close == string::npos ? end : close + end - at;
348 continue;
349 }
350 if (ch == '[') ++depth;
351 if (ch == ']' && --depth == 0) return at;
352 ++at;
353 }
354 return string::npos;
355}

357struct Piece {
358 string html;
359 char delimiter = 0;
360 std::size_t length = 0;
361 bool canOpen = false, canClose = false;
362};
363string emphasis(const std::vector<Piece>& pieces, std::size_t begin, std::size_t end) {
364 string html;
365 for (std::size_t at = begin; at < end; ++at) {
366 const Piece& piece = pieces[at];
367 if (!piece.delimiter) { html += piece.html; continue; }
368 std::size_t close = at + 1;
369 if (piece.canOpen) {
370 while (close < end && !(pieces[close].delimiter == piece.delimiter &&
371 pieces[close].length == piece.length && pieces[close].canClose)) ++close;
372 } else close = end;
373 if (close == end) { html += string(piece.length, piece.delimiter); continue; }
374 string opening, closing;
375 if (piece.delimiter == '~') { opening = "<del>"; closing = "</del>"; }
376 else if (piece.length == 1) { opening = "<em>"; closing = "</em>"; }
377 else if (piece.length == 2) { opening = "<strong>"; closing = "</strong>"; }
378 else { opening = "<em><strong>"; closing = "</strong></em>"; }
379 html += opening + emphasis(pieces, at + 1, close) + closing;
380 at = close;
381 }
382 return html;
383}

414 string inlineHtml(const string& text, bool links = true, bool notes = true) {
415 std::vector<Piece> pieces;
416 std::size_t at = 0;
417 while (at < text.size()) {
418 char ch = text[at];
419 if (ch == '\\') {
420 if (at + 1 < text.size() && contains(punctuation, text[at + 1])) {
421 pieces.push_back(Piece{escapeHtml(text.substr(at + 1, 1))}); at += 2;
422 } else if (at + 1 < text.size() && text[at + 1] == '\n') {
423 pieces.push_back(Piece{"<br />\n"}); at += 2;
424 } else { pieces.push_back(Piece{"\\"}); ++at; }
425 continue;
426 }
427 if (ch == '`') {
428 std::size_t end = runEnd(text, at), close = backtickCloser(text, end, end - at);
429 if (close == string::npos) { pieces.push_back(Piece{text.substr(at, end - at)}); at = end; }
430 else {
431 string code = text.substr(end, close - end);
432 std::replace(code.begin(), code.end(), '\n', ' ');
433 if (!code.empty() && code.front() == ' ' && code.back() == ' ' && code.find_first_not_of(' ') != string::npos)
434 code = code.substr(1, code.size() - 2);
435 pieces.push_back(Piece{"<code>" + escapeHtml(code) + "</code>"});
436 at = close + end - at;
437 }
438 continue;
439 }
440 if (links && ch == '<') {
441 std::size_t close = text.find('>', at + 1);
442 if (close != string::npos) {
443 string target = text.substr(at + 1, close - at - 1), href;
444 bool clean = target.find_first_of(" \n<") == string::npos;
445 if (clean && ((starts(target, "http://") && target.size() > 7) ||
446 (starts(target, "https://") && target.size() > 8))) href = target;
447 static const std::regex email(R"([A-Za-z0-9.+_-]+@[A-Za-z0-9-]+(?:\.[A-Za-z0-9-]+)+)");
448 if (href.empty() && std::regex_match(target, email)) href = "mailto:" + target;
449 if (!href.empty()) {
450 pieces.push_back(Piece{"<a href=\"" + escapeHtml(href) + "\">" + escapeHtml(target) + "</a>"});
451 at = close + 1; continue;
452 }
453 }
454 }
455 if (links && (starts(text, "http://", at) || starts(text, "https://", at)) &&
456 (at == 0 || contains(" \n*_~(", text[at - 1]))) {
457 std::size_t end = text.find_first_of(" \n<", at);
458 if (end == string::npos) end = text.size();
459 while (end > at && contains(".,:;!?*_~)", text[end - 1])) --end;
460 string url = text.substr(at, end - at);
461 if (url != "http://" && url != "https://") {
462 string escaped = escapeHtml(url);
463 pieces.push_back(Piece{"<a href=\"" + escaped + "\">" + escaped + "</a>"});
464 at = end; continue;
465 }
466 }
467 if (ch == '[' || (ch == '!' && at + 1 < text.size() && text[at + 1] == '[')) {
468 if (links && notes && starts(text, "[^", at)) {
469 std::size_t close = text.find(']', at + 2);
470 if (close != string::npos) {
471 string name = lower(text.substr(at + 2, close - at - 2));
472 if (footnotes.count(name)) {
473 pieces.push_back(Piece{footnoteReference(name)}); at = close + 1; continue;
474 }
475 }
476 }
477 bool image = ch == '!';
478 std::size_t opening = at + (image ? 1 : 0), close = closingBracket(text, opening);
479 if ((image || links) && close != string::npos) {
480 string label = text.substr(opening + 1, close - opening - 1);
481 if (std::optional<Destination> dest = destination(text, close + 1, label)) {
482 string content = inlineHtml(label, false, false);
483 string url = escapeHtml(dest->reference.url);
484 string title = dest->reference.title ? " title=\"" + escapeHtml(*dest->reference.title) + "\"" : "";
485 string html = image ? "<img src=\"" + url + "\" alt=\"" + stripTags(content) + "\"" + title + " />"
486 : "<a href=\"" + url + "\"" + title + ">" + content + "</a>";
487 pieces.push_back(Piece{html}); at = dest->end; continue;
488 }
489 }
490 }
491 if (contains("*_~", ch)) {
492 std::size_t end = runEnd(text, at), length = end - at;
493 char before = at ? text[at - 1] : ' ', after = end < text.size() ? text[end] : ' ';
494 bool open = !contains(whitespace, after), close = !contains(whitespace, before);
495 if (ch == '_' && alphanumeric(before) && alphanumeric(after)) open = close = false;
496 if ((ch != '~' && length <= 3) || (ch == '~' && length == 2))
497 pieces.push_back(Piece{"", ch, length, open, close});
498 else pieces.push_back(Piece{string(length, ch)});
499 at = end; continue;
500 }
501 pieces.push_back(Piece{escapeHtml(string(1, ch))});
502 ++at;
503 }
504 return emphasis(pieces, 0, pieces.size());
505 }

507 std::optional<Destination> destination(const string& text, std::size_t at, const string& label) const {
508 if (at < text.size() && text[at] == '[') {
509 std::size_t close = text.find(']', at + 1);
510 if (close != string::npos && text.substr(at + 1, close - at - 1).find('[') == string::npos) {
511 string key = text.substr(at + 1, close - at - 1);
512 key = normalize(key.empty() ? label : key);
513 std::map<string, Reference>::const_iterator found = references.find(key);
514 if (found == references.end()) return std::nullopt;
515 return Destination{found->second, close + 1};
516 }
517 } else if (at < text.size() && text[at] == '(') {
518 std::pair<string, std::size_t> url = escapedRead(text, skipSpace(text, at + 1), " \n)");
519 std::size_t next = skipSpace(text, url.second);
520 std::optional<string> title;
521 bool valid = true;
522 if (next > url.second && next < text.size() && contains("\"'", text[next])) {
523 std::pair<string, std::size_t> quoted = escapedRead(text, next + 1, string(1, text[next]));
524 if (quoted.second == text.size()) valid = false;
525 else { title = quoted.first; next = skipSpace(text, quoted.second + 1); }
526 }
527 if (valid && next < text.size() && text[next] == ')') return Destination{Reference{url.first, title}, next + 1};
528 }
529 std::map<string, Reference>::const_iterator found = references.find(normalize(label));
530 if (found == references.end()) return std::nullopt;
531 return Destination{found->second, at};
532 }

534 string renderBlocks(const std::vector<Block>& blocks) {
535 string html;
536 for (const Block& block : blocks) html += render(block) + "\n";
537 return html;
538 }
539 string render(const Block& block) {
540 switch (block.kind) {
541 case Kind::Heading: return heading(block.number, block.text);
542 case Kind::Paragraph: return "<p>" + inlineHtml(block.text) + "</p>";
543 case Kind::Rule: return "<hr />";
544 case Kind::Code: {
545 string html = "<pre><code";
546 if (!block.text.empty()) html += " class=\"language-" + escapeHtml(block.text) + "\"";
547 html += '>';
548 for (const string& line : block.lines) html += escapeHtml(line) + "\n";
549 return html + "</code></pre>";
550 }
551 case Kind::Quote: return "<blockquote>\n" + renderBlocks(block.children) + "</blockquote>";
552 case Kind::Alert: {
553 string name = lower(block.text), title = name;
554 title[0] = static_cast<char>(std::toupper(static_cast<unsigned char>(title[0])));
555 return "<div class=\"alert alert-" + name + "\">\n<p class=\"alert-title\">" + title + "</p>\n"
556 + renderBlocks(block.children) + "</div>";
557 }
558 case Kind::List: {
559 string tag = block.ordered ? "ol" : "ul", html = "<" + tag;
560 if (block.ordered && block.number != 1) html += " start=\"" + std::to_string(block.number) + "\"";
561 html += ">\n";
562 for (const std::vector<Block>& item : block.items) html += renderItem(item) + "\n";
563 return html + "</" + tag + ">";
564 }
565 case Kind::Table: {
566 string html = "<table>\n<thead>\n<tr>\n" + tableCells("th", block, block.lines) + "</tr>\n</thead>\n";
567 if (!block.rows.empty()) {
568 html += "<tbody>\n";
569 for (const Strings& row : block.rows) html += "<tr>\n" + tableCells("td", block, row) + "</tr>\n";
570 html += "</tbody>\n";
571 }
572 return html + "</table>";
573 }
574 case Kind::Toc: return tocMarker;
575 }
576 return "";
577 }
578 string heading(int level, string text) {
579 string id;
580 std::size_t explicitId = text.rfind(" {#");
581 if (!text.empty() && text.back() == '}' && explicitId != string::npos) {
582 string candidate = text.substr(explicitId + 3, text.size() - explicitId - 4);
583 if (!candidate.empty() && std::all_of(candidate.begin(), candidate.end(), labelCharacter)) {
584 id = candidate;
585 text = text.substr(0, explicitId);
586 std::size_t last = text.find_last_not_of(' ');
587 text = last == string::npos ? "" : text.substr(0, last + 1);
588 }
589 }
590 if (id.empty()) {
591 string base;
592 for (char ch : lower(text)) {
593 if (labelCharacter(ch)) base += ch;
594 else if (contains(whitespace, ch)) base += '-';
595 }
596 if (base.empty()) base = "heading";
597 id = base;
598 for (int suffix = 1; usedIds.count(id); ++suffix) id = base + "-" + std::to_string(suffix);
599 }
600 usedIds.insert(id);
601 string html = inlineHtml(text), number = std::to_string(level);
602 headings.push_back(HeadingEntry{level, id, stripTags(html), {}});
603 return "<h" + number + " id=\"" + id + "\">" + html + "</h" + number + ">";
604 }
605 string renderItem(const std::vector<Block>& blocks) {
606 string html = "<li>";
607 for (std::size_t i = 0; i < blocks.size(); ++i) {
608 const Block& block = blocks[i];
609 if (block.kind == Kind::Paragraph) {
610 string text = block.text;
611 if (i == 0 && text.size() > 4 && (starts(text, "[ ] ") || starts(text, "[x] ") || starts(text, "[X] "))) {
612 html += text[1] == ' ' ? "<input type=\"checkbox\" disabled=\"\" /> "
613 : "<input type=\"checkbox\" checked=\"\" disabled=\"\" /> ";
614 text.erase(0, 4);
615 }
616 html += inlineHtml(text);
617 } else {
618 if (html.back() != '\n') html += '\n';
619 html += render(block) + "\n";
620 }
621 }
622 return html + "</li>";
623 }
624 string tableCells(const string& tag, const Block& table, const Strings& cells) {
625 string html;
626 for (std::size_t i = 0; i < cells.size(); ++i) {
627 string style = table.alignments[i].empty() ? "" : " style=\"text-align:" + table.alignments[i] + "\"";
628 html += "<" + tag + style + ">" + inlineHtml(cells[i]) + "</" + tag + ">\n";
629 }
630 return html;
631 }
632 string tocList(const std::vector<std::size_t>& entries) const {
633 string html = "<ul>\n";
634 for (std::size_t index : entries) {
635 const HeadingEntry& entry = headings[index];
636 html += "<li><a href=\"#" + entry.id + "\">" + entry.label + "</a>";
637 if (!entry.children.empty()) html += "\n" + tocList(entry.children) + "\n";
638 html += "</li>\n";
639 }
640 return html + "</ul>";
641 }
642 string tableOfContents() {
643 if (headings.empty()) return "<nav class=\"toc\">\n</nav>";
644 std::vector<std::size_t> roots, parents;
645 for (std::size_t i = 0; i < headings.size(); ++i) {
646 while (!parents.empty() && headings[parents.back()].level >= headings[i].level) parents.pop_back();
647 if (parents.empty()) roots.push_back(i);
648 else headings[parents.back()].children.push_back(i);
649 parents.push_back(i);
650 }
651 return "<nav class=\"toc\">\n" + tocList(roots) + "\n</nav>";
652 }
653 string referenceId(std::size_t index, int count) const {
654 return "fnref-" + std::to_string(index + 1) + (count == 1 ? "" : "-" + std::to_string(count));
655 }
656 string footnoteReference(const string& name) {
657 if (!footnoteNumbers.count(name)) {
658 footnoteNumbers[name] = footnoteNames.size();
659 footnoteNames.push_back(name);
660 footnoteCounts.push_back(0);
661 }
662 std::size_t index = footnoteNumbers.at(name);
663 string id = referenceId(index, ++footnoteCounts[index]), number = std::to_string(index + 1);
664 return "<sup id=\"" + id + "\"><a href=\"#fn-" + number + "\">" + number + "</a></sup>";
665 }
666 string footnoteList() {
667 string html = "<section class=\"footnotes\">\n<ol>\n";
668 for (std::size_t i = 0; i < footnoteNames.size(); ++i) {
669 string body = inlineHtml(footnotes.at(footnoteNames[i]), true, false);
670 html += "<li id=\"fn-" + std::to_string(i + 1) + "\">" + body;
671 for (int count = 1; count <= footnoteCounts[i]; ++count)
672 html += " <a href=\"#" + referenceId(i, count) + "\">&#8617;</a>";
673 html += "</li>\n";
674 }
675 return html + "</ol>\n</section>\n";
676 }
677};
