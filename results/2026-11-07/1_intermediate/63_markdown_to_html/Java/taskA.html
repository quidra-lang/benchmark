31 static String trim(String s) {
32 int first = 0, end = s.length();
33 while (first < end && s.charAt(first) == ' ')
34 first++;
35 while (end > first && s.charAt(end - 1) == ' ')
36 end--;

51 static String escapeHtml(String s) {
52 var html = new StringBuilder();
53 for (int i = 0; i < s.length(); i++) {
54 switch (s.charAt(i)) {
55 case '&':
56 html.append("&amp;");
57 break;
58 case '<':
59 html.append("&lt;");
60 break;
61 case '>':
62 html.append("&gt;");
63 break;
64 case '"':
65 html.append("&quot;");
66 break;
67 default:
68 html.append(s.charAt(i));
69 }
70 }
71 return html.toString();
72 }
73 static String stripTags(String s) {
74 return s.replaceAll("<[^>]*>", "");

76 static String normalize(String s) {
77 var result = new StringBuilder();
78 boolean space = false;
79 for (int i = 0; i < s.length(); i++) {
80 char ch = s.charAt(i);
81 if (whitespace(ch))
82 space = result.length() > 0;
83 else {
84 if (space)
85 result.append(' ');
86 result.append(Character.toLowerCase(ch));
87 space = false;
88 }
89 }
90 return result.toString();
91 }
92 static int skipSpace(String s, int at) {
93 while (at < s.length() && whitespace(s.charAt(at)))

98 static Read escapedRead(String s, int at, String stops) {
99 var result = new StringBuilder();
100 while (at < s.length() && !has(stops, s.charAt(at))) {

104 }

129 static Fence openingFence(String line) {
130 if (line.isEmpty() || !has("`~", line.charAt(0)))
131 return null;
132 int end = 1;
133 while (end < line.length() && line.charAt(end) == line.charAt(0))
134 end++;
135 if (end < 3)
136 return null;
137 String info = trim(line.substring(end));
138 if (line.charAt(0) == '`' && info.indexOf('`') >= 0)
139 return null;
140 return new Fence(line.charAt(0), end, info);
141 }
142 static Block atxHeading(String line) {
143 int n = 0;
144 while (n < line.length() && line.charAt(n) == '#')
145 n++;
146 if (n < 1 || n > 6 || n < line.length() && line.charAt(n) != ' ')
147 return null;
148 String text = trim(line.substring(n));
149 int end = text.length();
150 while (end > 0 && text.charAt(end - 1) == '#')
151 end--;
152 if (end == 0 || text.charAt(end - 1) == ' ')
153 text = trim(text.substring(0, end));
154 return new Block(Kind.HEADING, text, n);
155 }
156 static boolean isRule(String line) {
157 String compact = line.replace(" ", "");
158 if (compact.length() < 3 || !has("-*_", compact.charAt(0)))
159 return false;
160 for (int i = 1; i < compact.length(); i++)
161 if (compact.charAt(i) != compact.charAt(0))
162 return false;
163 return true;
164 }
165 static final Pattern LIST_ITEM = Pattern.compile("(?:([-*+])|([0-9]{1,9})([.)])) +");
166 static ListMarker listMarker(String line) {
167 if (isRule(line))
168 return null;
169 Matcher m = LIST_ITEM.matcher(line);
170 if (!m.lookingAt())
171 return null;
172 boolean ordered = m.group(2) != null;
173 return new ListMarker((ordered ? m.group(3) : m.group(1)).charAt(0), ordered,
174 ordered ? Integer.parseInt(m.group(2)) : 1, line.substring(m.end()));
175 }
176 static boolean interrupts(String line) {
177 return !line.isEmpty() && line.charAt(0) != ' '
178 && (openingFence(line) != null || atxHeading(line) != null || isRule(line)
179 || line.charAt(0) == '>' || listMarker(line) != null);

181 static List<String> splitCells(String line) {
182 String text = trim(line);
183 int at = text.startsWith("|") ? 1 : 0;
184 var cells = new ArrayList<String>();
185 var cell = new StringBuilder();
186 boolean finalSeparator = false;
187 while (at < text.length()) {
188 char ch = text.charAt(at++);
189 finalSeparator = false;
190 if (ch == '\\' && at < text.length()) {
191 if (text.charAt(at) != '|')
192 cell.append('\\');
193 cell.append(text.charAt(at++));
194 } else if (ch == '|') {
195 cells.add(trim(cell.toString()));
196 cell.setLength(0);
197 finalSeparator = true;
198 } else
199 cell.append(ch);
200 }
201 if (!finalSeparator)
202 cells.add(trim(cell.toString()));
203 return cells;
204 }

208 boolean definition(String line) {
209 Matcher footnote = Pattern.compile("\\[\\^([A-Za-z0-9_-]+)\\]:(.*)").matcher(line);
210 if (footnote.matches() && !trim(footnote.group(2)).isEmpty()) {
211 footnotes.putIfAbsent(lower(footnote.group(1)), trim(footnote.group(2)));
212 return true;
213 }
214 if (!line.startsWith("["))
215 return false;
216 int close = line.indexOf(']');
217 if (close < 0 || close + 1 >= line.length() || line.charAt(close + 1) != ':')
218 return false;
219 String label = line.substring(1, close);
220 if (label.indexOf('[') >= 0 || trim(label).isEmpty() || label.startsWith("^"))
221 return false;
222 Read url = escapedRead(line, skipSpace(line, close + 2), " ");
223 if (url.text.isEmpty())
224 return false;
225 int at = skipSpace(line, url.end);
226 String title = null;
227 if (at < line.length()) {
228 if (at == url.end || !has("\"'", line.charAt(at)))
229 return false;
230 Read quoted = escapedRead(line, at + 1, String.valueOf(line.charAt(at)));
231 if (quoted.end != line.length() - 1)
232 return false;
233 title = quoted.text;
234 }
235 references.putIfAbsent(normalize(label), new Reference(url.text, title));
236 return true;
237 }
238 List<Block> parse(List<String> lines) {
239 var blocks = new ArrayList<Block>();
240 int at = 0;
241 while (at < lines.size()) {
242 String line = lines.get(at);
243 if (line.isEmpty()) {
244 at++;
245 continue;
246 }
247 if (line.startsWith(" ")) {
248 Block code = new Block(Kind.CODE);
249 int end = at;
250 while (end < lines.size()
251 && (lines.get(end).isEmpty() || lines.get(end).startsWith(" ")))
252 end++;
253 while (end > at && lines.get(end - 1).isEmpty())
254 end--;
255 while (at < end) {
256 String row = lines.get(at++);
257 code.lines.add(row.isEmpty() ? "" : row.substring(4));
258 }
259 blocks.add(code);
260 continue;
261 }
262 if (line.charAt(0) != ' ') {
263 Fence fence = openingFence(line);
264 if (fence != null) {
265 String language = fence.info.split(" ", 2)[0];
266 Block code = new Block(Kind.CODE, language, 0);
267 at++;
268 while (at < lines.size()) {
269 String row = lines.get(at++);
270 boolean closing = row.length() >= fence.length;
271 for (int i = 0; closing && i < row.length(); i++)
272 if (row.charAt(i) != fence.character)
273 closing = false;
274 if (closing)
275 break;
276 code.lines.add(row);
277 }
278 blocks.add(code);
279 continue;
280 }
281 Block heading = atxHeading(line);
282 if (heading != null) {
283 blocks.add(heading);
284 at++;
285 continue;
286 }
287 if (isRule(line)) {
288 blocks.add(new Block(Kind.RULE));
289 at++;
290 continue;
291 }
292 if (line.startsWith(">")) {
293 var content = new ArrayList<String>();
294 while (at < lines.size() && lines.get(at).startsWith(">")) {
295 String row = lines.get(at++).substring(1);
296 if (row.startsWith(" "))
297 row = row.substring(1);
298 content.add(row);
299 }
300 Block quote = new Block(Kind.QUOTE);
301 if (content.get(0).matches("\\[!(NOTE|TIP|IMPORTANT|WARNING|CAUTION)\\]")) {
302 quote.kind = Kind.ALERT;
303 quote.text = content.remove(0);
304 quote.text = quote.text.substring(2, quote.text.length() - 1);
305 }
306 quote.children = parse(content);
307 blocks.add(quote);
308 continue;
309 }
310 ListMarker marker = listMarker(line);
311 if (marker != null) {
312 Block list = new Block(Kind.LIST, "", marker.number);
313 list.ordered = marker.ordered;
314 char kind = marker.marker;
315 while (true) {
316 var content = new ArrayList<String>();
317 content.add(marker.content);
318 at++;
319 while (at < lines.size() && lines.get(at).startsWith(" "))
320 content.add(lines.get(at++).substring(2));
321 list.items.add(parse(content));
322 int next = at;
323 while (next < lines.size() && lines.get(next).isEmpty())
324 next++;
325 marker = next < lines.size() ? listMarker(lines.get(next)) : null;
326 if (marker == null || marker.marker != kind
327 || marker.ordered != list.ordered)
328 break;
329 at = next;
330 }
331 blocks.add(list);
332 continue;
333 }
334 if (definition(line)) {
335 at++;
336 continue;
337 }
338 if (at + 1 < lines.size() && line.indexOf('|') >= 0
339 && lines.get(at + 1).indexOf('|') >= 0) {
340 List<String> header = splitCells(line),
341 delimiters = splitCells(lines.get(at + 1));
342 if (header.size() == delimiters.size()
343 && delimiters.stream().allMatch(c -> c.matches(":?-+:?"))) {
344 Block table = new Block(Kind.TABLE);
345 table.lines = header;
346 for (String cell : delimiters) {
347 boolean left = cell.startsWith(":"), right = cell.endsWith(":");
348 table.alignments.add(
349 left ? (right ? "center" : "left") : (right ? "right" : ""));
350 }
351 at += 2;
352 while (at < lines.size() && lines.get(at).indexOf('|') >= 0) {
353 List<String> row = splitCells(lines.get(at++));
354 while (row.size() < delimiters.size())
355 row.add("");
356 table.rows.add(new ArrayList<>(row.subList(0, delimiters.size())));
357 }
358 blocks.add(table);
359 continue;
360 }
361 }
362 }
363 var text = new ArrayList<String>();
364 text.add(leftTrim(line));
365 at++;
366 int level = 0;
367 while (at < lines.size() && !lines.get(at).isEmpty()) {
368 String row = lines.get(at);
369 if (row.matches("={3,}|-{3,}")) {
370 level = row.charAt(0) == '=' ? 1 : 2;
371 at++;
372 break;
373 }
374 if (interrupts(row))
375 break;
376 text.add(leftTrim(row));
377 at++;
378 }
379 String combined = String.join("\n", text);
380 blocks.add(new Block(level != 0 ? Kind.HEADING
381 : combined.equals("[TOC]") ? Kind.TOC
382 : Kind.PARAGRAPH,
383 combined, level));
384 }
385 return blocks;
386 }

388 static int runEnd(String text, int at) {
389 char ch = text.charAt(at++);
390 while (at < text.length() && text.charAt(at) == ch)
391 at++;
392 return at;
393 }
394 static int backtickCloser(String text, int at, int length) {
395 while ((at = text.indexOf('`', at)) >= 0) {
396 int end = runEnd(text, at);
397 if (end - at == length)
398 return at;
399 at = end;
400 }
401 return -1;
402 }
403 static int closingBracket(String text, int at) {
404 int depth = 0;
405 while (at < text.length()) {
406 char ch = text.charAt(at);
407 if (ch == '\\') {
408 at += 2;
409 continue;
410 }
411 if (ch == '`') {
412 int end = runEnd(text, at), close = backtickCloser(text, end, end - at);
413 at = close < 0 ? end : close + end - at;
414 continue;
415 }
416 if (ch == '[')
417 depth++;
418 if (ch == ']' && --depth == 0)
419 return at;
420 at++;
421 }
422 return -1;

429 static void emphasis(StringBuilder html, List<Piece> pieces, int begin, int end) {
430 for (int at = begin; at < end; at++) {
431 Piece p = pieces.get(at);
432 if (p.delimiter == 0) {
433 html.append(p.html);
434 continue;
435 }
436 int close = at + 1;
437 if (p.canOpen) {
438 while (close < end
439 && !(pieces.get(close).delimiter == p.delimiter
440 && pieces.get(close).length == p.length && pieces.get(close).canClose))
441 close++;
442 } else
443 close = end;
444 if (close == end) {
445 html.append(String.valueOf(p.delimiter).repeat(p.length));
446 continue;
447 }
448 String open, shut;
449 if (p.delimiter == '~') {
450 open = "<del>";
451 shut = "</del>";
452 } else if (p.length == 1) {
453 open = "<em>";
454 shut = "</em>";
455 } else if (p.length == 2) {
456 open = "<strong>";
457 shut = "</strong>";
458 } else {
459 open = "<em><strong>";
460 shut = "</strong></em>";
461 }
462 html.append(open);
463 emphasis(html, pieces, at + 1, close);
464 html.append(shut);
465 at = close;
466 }

509 Destination destination(String text, int at, String label) {
510 if (at < text.length() && text.charAt(at) == '[') {
511 int close = text.indexOf(']', at + 1);
512 if (close >= 0 && text.substring(at + 1, close).indexOf('[') < 0) {

533 return new Destination(new Reference(url.text, title), next + 1);
534 }

536 return reference == null ? null : new Destination(reference, at);

541 String inline(String text, boolean links, boolean footnotes) {
542 var pieces = new ArrayList<Piece>();
543 int at = 0;
544 while (at < text.length()) {
545 char ch = text.charAt(at);
546 if (ch == '\\') {
547 if (at + 1 < text.length() && has(PUNCTUATION, text.charAt(at + 1))) {
548 pieces.add(new Piece(escapeHtml(text.substring(at + 1, at + 2))));
549 at += 2;
550 } else if (at + 1 < text.length() && text.charAt(at + 1) == '\n') {
551 pieces.add(new Piece("<br />\n"));
552 at += 2;
553 } else {
554 pieces.add(new Piece("\\"));
555 at++;
556 }
557 continue;
558 }
559 if (ch == '`') {
560 int end = runEnd(text, at), close = backtickCloser(text, end, end - at);
561 if (close < 0) {
562 pieces.add(new Piece(text.substring(at, end)));
563 at = end;
564 } else {
565 String code = text.substring(end, close).replace('\n', ' ');
566 if (code.startsWith(" ") && code.endsWith(" ") && !trim(code).isEmpty())
567 code = code.substring(1, code.length() - 1);
568 pieces.add(new Piece("<code>" + escapeHtml(code) + "</code>"));
569 at = close + end - at;
570 }
571 continue;
572 }
573 if (links && ch == '<') {
574 int close = text.indexOf('>', at + 1);
575 if (close >= 0) {
576 String target = text.substring(at + 1, close), href = null;
577 if (target.indexOf(' ') < 0 && target.indexOf('\n') < 0
578 && target.indexOf('<') < 0
579 && (target.startsWith("http://") && target.length() > 7
580 || target.startsWith("https://") && target.length() > 8))
581 href = target;
582 if (href == null
583 && target.matches(
584 "[A-Za-z0-9.+_-]+@[A-Za-z0-9-]+(?:\\.[A-Za-z0-9-]+)+"))
585 href = "mailto:" + target;
586 if (href != null) {
587 pieces.add(new Piece("<a href=\"" + escapeHtml(href) + "\">"
588 + escapeHtml(target) + "</a>"));
589 at = close + 1;
590 continue;
591 }
592 }
593 }
594 if (links && (text.startsWith("http://", at) || text.startsWith("https://", at))
595 && (at == 0 || has(" \n*_~(", text.charAt(at - 1)))) {
596 int end = at;
597 while (end < text.length() && !has(" \n<", text.charAt(end)))
598 end++;
599 while (end > at && has(".,:;!?*_~)", text.charAt(end - 1)))
600 end--;
601 String url = text.substring(at, end);
602 if (!url.equals("http://") && !url.equals("https://")) {
603 String escaped = escapeHtml(url);
604 pieces.add(new Piece("<a href=\"" + escaped + "\">" + escaped + "</a>"));
605 at = end;
606 continue;
607 }
608 }
609 if (ch == '['
610 || ch == '!' && at + 1 < text.length() && text.charAt(at + 1) == '[') {
611 if (links && footnotes && text.startsWith("[^", at)) {
612 int close = text.indexOf(']', at + 2);
613 if (close >= 0) {
614 String name = lower(text.substring(at + 2, close));
615 if (parser.footnotes.containsKey(name)) {
616 pieces.add(new Piece(footnoteReference(name)));
617 at = close + 1;
618 continue;
619 }
620 }
621 }
622 boolean image = ch == '!';
623 int opening = at + (image ? 1 : 0), close = closingBracket(text, opening);
624 if ((image || links) && close >= 0) {
625 String label = text.substring(opening + 1, close);
626 Destination dest = destination(text, close + 1, label);
627 if (dest != null) {
628 String content = inline(label, false, false),
629 url = escapeHtml(dest.reference.url);
630 String title = dest.reference.title == null
631 ? ""
632 : " title=\"" + escapeHtml(dest.reference.title) + "\"";
633 pieces.add(new Piece(image
634 ? "<img src=\"" + url + "\" alt=\"" + stripTags(content) + "\""
635 + title + " />"
636 : "<a href=\"" + url + "\"" + title + ">" + content + "</a>"));
637 at = dest.end;
638 continue;
639 }
640 }
641 }
642 if (has("*_~", ch)) {
643 int end = runEnd(text, at), length = end - at;
644 char before = at == 0 ? ' ' : text.charAt(at - 1),
645 after = end == text.length() ? ' ' : text.charAt(end);
646 boolean open = !whitespace(after), close = !whitespace(before);
647 if (ch == '_' && alphanumeric(before) && alphanumeric(after))
648 open = close = false;
649 pieces.add(ch != '~' && length <= 3 || ch == '~' && length == 2
650 ? new Piece("", ch, length, open, close)
651 : new Piece(text.substring(at, end)));
652 at = end;
653 continue;
654 }
655 pieces.add(new Piece(escapeHtml(text.substring(at, at + 1))));
656 at++;
657 }
658 var html = new StringBuilder();
659 emphasis(html, pieces, 0, pieces.size());
660 return html.toString();
661 }

668 String heading(Block block) {
669 String text = block.text, id = null;
670 int explicit = text.lastIndexOf(" {#");
671 if (text.endsWith("}") && explicit >= 0) {
672 String candidate = text.substring(explicit + 3, text.length() - 1);
673 if (!candidate.isEmpty()
674 && candidate.chars().allMatch(c -> labelCharacter((char) c))) {
675 id = candidate;
676 text = rightTrim(text.substring(0, explicit));
677 }
678 }
679 if (id == null) {
680 var slug = new StringBuilder();
681 for (int i = 0; i < text.length(); i++) {
682 char c = Character.toLowerCase(text.charAt(i));
683 if (labelCharacter(c))
684 slug.append(c);
685 else if (whitespace(c))
686 slug.append('-');
687 }
688 String base = slug.isEmpty() ? "heading" : slug.toString();
689 id = base;
690 for (int suffix = 1; usedIds.contains(id); suffix++)
691 id = base + "-" + suffix;
692 }
693 usedIds.add(id);
694 String html = inline(text);
695 headings.add(new Heading(block.number, id, stripTags(html)));
696 return "<h" + block.number + " id=\"" + id + "\">" + html + "</h" + block.number + ">";
697 }
698 String renderItem(List<Block> blocks) {
699 var html = new StringBuilder("<li>");
700 for (int i = 0; i < blocks.size(); i++) {
701 Block block = blocks.get(i);
702 if (block.kind == Kind.PARAGRAPH) {
703 String text = block.text;
704 if (i == 0 && text.length() > 4
705 && (text.startsWith("[ ] ") || text.startsWith("[x] ")
706 || text.startsWith("[X] "))) {
707 html.append(text.charAt(1) == ' '
708 ? "<input type=\"checkbox\" disabled=\"\" /> "
709 : "<input type=\"checkbox\" checked=\"\" disabled=\"\" /> ");
710 text = text.substring(4);
711 }
712 html.append(inline(text));
713 } else {
714 if (html.charAt(html.length() - 1) != '\n')
715 html.append('\n');
716 html.append(render(block)).append('\n');
717 }
718 }
719 return html.append("</li>").toString();
720 }
721 String tableCells(String tag, Block table, List<String> cells) {
722 var html = new StringBuilder();
723 for (int i = 0; i < cells.size(); i++) {
724 String alignment = table.alignments.get(i);
725 html.append('<').append(tag);
726 if (!alignment.isEmpty())
727 html.append(" style=\"text-align:").append(alignment).append('"');
728 html.append('>')
729 .append(inline(cells.get(i)))
730 .append("</")
731 .append(tag)
732 .append(">\n");
733 }
734 return html.toString();
735 }
736 String render(Block block) {
737 switch (block.kind) {
738 case HEADING:
739 return heading(block);
740 case PARAGRAPH:
741 return "<p>" + inline(block.text) + "</p>";
742 case RULE:
743 return "<hr />";
744 case CODE: {
745 var html = new StringBuilder("<pre><code");
746 if (!block.text.isEmpty())
747 html.append(" class=\"language-")
748 .append(escapeHtml(block.text))
749 .append('"');
750 html.append('>');
751 for (String line : block.lines)
752 html.append(escapeHtml(line)).append('\n');
753 return html.append("</code></pre>").toString();
754 }
755 case QUOTE:
756 return "<blockquote>\n" + renderBlocks(block.children) + "</blockquote>";
757 case ALERT: {
758 String name = lower(block.text),
759 title = Character.toUpperCase(name.charAt(0)) + name.substring(1);
760 return "<div class=\"alert alert-" + name + "\">\n<p class=\"alert-title\">"
761 + title + "</p>\n" + renderBlocks(block.children) + "</div>";
762 }
763 case LIST: {
764 String tag = block.ordered ? "ol" : "ul";
765 var html = new StringBuilder("<" + tag);
766 if (block.ordered && block.number != 1)
767 html.append(" start=\"").append(block.number).append('"');
768 html.append(">\n");
769 for (List<Block> item : block.items)
770 html.append(renderItem(item)).append('\n');
771 return html.append("</").append(tag).append('>').toString();
772 }
773 case TABLE: {
774 var html = new StringBuilder("<table>\n<thead>\n<tr>\n");
775 html.append(tableCells("th", block, block.lines)).append("</tr>\n</thead>\n");
776 if (!block.rows.isEmpty()) {
777 html.append("<tbody>\n");
778 for (List<String> row : block.rows)
779 html.append("<tr>\n")
780 .append(tableCells("td", block, row))
781 .append("</tr>\n");
782 html.append("</tbody>\n");
783 }
784 return html.append("</table>").toString();
785 }
786 case TOC:
787 return TOC_MARKER;
788 }
789 throw new IllegalStateException();
790 }
791 String tocList(List<Heading> entries) {
792 var html = new StringBuilder("<ul>\n");
793 for (Heading entry : entries) {
794 html.append("<li><a href=\"#")
795 .append(entry.id)
796 .append("\">")
797 .append(entry.label)
798 .append("</a>");
799 if (!entry.children.isEmpty())
800 html.append('\n').append(tocList(entry.children)).append('\n');
801 html.append("</li>\n");
802 }
803 return html.append("</ul>").toString();
804 }
805 String tableOfContents() {
806 if (headings.isEmpty())
807 return "<nav class=\"toc\">\n</nav>";
808 var roots = new ArrayList<Heading>();
809 var parents = new ArrayList<Heading>();
810 for (Heading heading : headings) {
811 while (!parents.isEmpty() && parents.get(parents.size() - 1).level >= heading.level)
812 parents.remove(parents.size() - 1);
813 if (parents.isEmpty())
814 roots.add(heading);
815 else
816 parents.get(parents.size() - 1).children.add(heading);
817 parents.add(heading);
818 }
819 return "<nav class=\"toc\">\n" + tocList(roots) + "\n</nav>";
820 }
821 String footnoteList() {
822 var html = new StringBuilder("<section class=\"footnotes\">\n<ol>\n");
823 for (int i = 0; i < notes.size(); i++) {
824 Note note = notes.get(i);
825 int number = i + 1;
826 html.append("<li id=\"fn-")
827 .append(number)
828 .append("\">")
829 .append(inline(parser.footnotes.get(note.name), true, false));
830 for (int count = 1; count <= note.count; count++)
831 html.append(" <a href=\"#")
832 .append(referenceId(number, count))
833 .append("\">&#8617;</a>");
834 html.append("</li>\n");
835 }
836 return html.append("</ol>\n</section>\n").toString();
837 }
838 String document(List<Block> blocks) {
839 String html = renderBlocks(blocks);
840 html = html.replace(TOC_MARKER, tableOfContents());
841 return notes.isEmpty() ? html : html + footnoteList();
842 }
