3type Alignment = '' | 'left' | 'center' | 'right';
4interface Heading { kind: 'heading'; level: number; text: string; }
5interface Paragraph { kind: 'paragraph'; text: string; }
6interface CodeBlock { kind: 'code'; language: string; lines: string[]; }
7interface Quote { kind: 'quote'; blocks: Block[]; }
8interface Alert { kind: 'alert'; alert: string; blocks: Block[]; }
9interface ListBlock { kind: 'list'; ordered: boolean; start: number; items: Block[][]; }
10interface Table { kind: 'table'; alignments: Alignment[]; header: string[]; rows: string[][]; }
11type Block = Heading | Paragraph | CodeBlock | Quote | Alert | ListBlock | Table | {kind: 'rule' | 'toc'};
12interface Reference { url: string; title: string | null; }
13interface ListMarker { marker: string; ordered: boolean; start: number; content: string; }
14interface Fence { fence: string; info: string; }
15interface ReadResult { value: string; end: number; }
16interface Destination { reference: Reference; end: number; }
17interface InlineResult { html: string; end: number; }
18interface Delimiter { char: string; length: number; canOpen: boolean; canClose: boolean; }
19type Piece = string | Delimiter;
20interface TocEntry { level: number; id: string; label: string; children: TocEntry[]; }
21interface NumberedNote { number: number; count: number; }

23const PUNCTUATION = '!"#$%&\'()*+,-./:;<=>?@[\\]^_`{|}~';
24const WHITESPACE = ' \n';
25const TOC_MARKER = '\x00toc\x00';
26const htmlEscapes: Readonly<Record<string, string>> = {'&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;'};
27function escapeHtml(text: string): string { return text.replace(/[&<>"]/g, char => htmlEscapes[char]); }
28function stripTags(text: string): string { return text.replace(/<[^>]*>/g, ''); }
29function normalize(text: string): string { return text.toLowerCase().replace(/[ \n]+/g, ' ').trim(); }
30function skipSpace(text: string, at: number): number {
31 while (at < text.length && WHITESPACE.includes(text[at])) at++;
32 return at;
33}
34function escapedRead(text: string, at: number, stops: string): ReadResult {
35 let value = '';
36 while (at < text.length && !stops.includes(text[at])) {
37 if (text[at] === '\\' && at + 1 < text.length && PUNCTUATION.includes(text[at + 1])) at++;
38 value += text[at++];
39 }
40 return {value, end: at};
41}
42function runEnd(text: string, at: number): number {
43 const char = text[at++];
44 while (at < text.length && text[at] === char) at++;
45 return at;
46}
47function openingFence(line: string): Fence | null {
48 if (!line.startsWith('`') && !line.startsWith('~')) return null;
49 const end = runEnd(line, 0), fence = line.slice(0, end), info = line.slice(end).trim();
50 if (end < 3 || (line[0] === '`' && info.includes('`'))) return null;
51 return {fence, info};
52}
53function atxHeading(line: string): Heading | null {
54 const match = /^(#{1,6})(?: |$)/.exec(line);
55 if (!match) return null;
56 let text = line.slice(match[1].length).trim();
57 const trimmed = text.replace(/#+$/, '');
58 if (!trimmed || trimmed.endsWith(' ')) text = trimmed.trim();
59 return {kind: 'heading', level: match[1].length, text};
60}
61function isRule(line: string): boolean { return /^(?:-{3,}|\*{3,}|_{3,})$/.test(line.replace(/ /g, '')); }
62function listMarker(line: string): ListMarker | null {
63 if (isRule(line)) return null;
64 const match = /^(?:([-*+])|([0-9]{1,9})([.)])) +/.exec(line);
65 if (!match) return null;
66 return {marker: match[1] || match[3], ordered: match[2] !== undefined,
67 start: match[2] === undefined ? 1 : Number(match[2]), content: line.slice(match[0].length)};
68}
69function interruptsParagraph(line: string): boolean {
70 return !line.startsWith(' ') && (openingFence(line) !== null || atxHeading(line) !== null
71 || isRule(line) || line.startsWith('>') || listMarker(line) !== null);
72}
73function splitCells(line: string): string[] {
74 let text = line.trim();
75 if (text.startsWith('|')) text = text.slice(1);
76 const cells = [''];
77 let finalSeparator = false;
78 for (let at = 0; at < text.length;) {
79 const char = text[at++];
80 finalSeparator = false;
81 if (char === '\\' && at < text.length) {
82 cells[cells.length - 1] += (text[at] === '|' ? '' : '\\') + text[at++];
83 } else if (char === '|') {
84 cells.push('');
85 finalSeparator = true;
86 } else cells[cells.length - 1] += char;
87 }
88 if (finalSeparator) cells.pop();
89 return cells.map(cell => cell.trim());
90}

92class BlockParser {
93 readonly references: Map<string, Reference>;
94 readonly footnotes: Map<string, string>;
95 constructor() {
96 this.references = new Map<string, Reference>();
97 this.footnotes = new Map<string, string>();
98 }
99 definition(line: string): boolean {
100 const footnote = /^\[\^([A-Za-z0-9_-]+)\]:(.*)$/.exec(line);
101 if (footnote && footnote[2].trim()) {
102 const name = footnote[1].toLowerCase();
103 if (!this.footnotes.has(name)) this.footnotes.set(name, footnote[2].trim());
104 return true;
105 }
106 if (!line.startsWith('[')) return false;
107 const close = line.indexOf(']');
108 if (close < 0 || line[close + 1] !== ':') return false;
109 const label = line.slice(1, close);
110 if (label.includes('[') || !label.trim() || label.startsWith('^')) return false;
111 const url = escapedRead(line, skipSpace(line, close + 2), ' ');
112 if (!url.value) return false;
113 const at = skipSpace(line, url.end);
114 let title: string | null = null;
115 if (at < line.length) {
116 if (at === url.end || !'"\''.includes(line[at])) return false;
117 const quoted = escapedRead(line, at + 1, line[at]);
118 if (quoted.end !== line.length - 1) return false;
119 title = quoted.value;
120 }
121 const name = normalize(label);
122 if (!this.references.has(name)) this.references.set(name, {url: url.value, title});
123 return true;
124 }
125 parse(lines: readonly string[]): Block[] {
126 const blocks: Block[] = [];
127 let at = 0;
128 while (at < lines.length) {
129 const line = lines[at];
130 if (line === '') { at++; continue; }
131 if (line.startsWith(' ')) {
132 let end = at;
133 while (end < lines.length && (lines[end] === '' || lines[end].startsWith(' '))) end++;
134 while (lines[end - 1] === '') end--;
135 blocks.push({kind: 'code', language: '', lines: lines.slice(at, end).map(row => row.slice(4))});
136 at = end;
137 continue;
138 }
139 if (!line.startsWith(' ')) {
140 const fence = openingFence(line);
141 if (fence !== null) {
142 const code: string[] = [];
143 at++;
144 while (at < lines.length) {
145 const row = lines[at++];
146 if (row.length >= fence.fence.length && [...row].every(char => char === fence.fence[0])) break;
147 code.push(row);
148 }
149 blocks.push({kind: 'code', language: fence.info.split(' ')[0], lines: code});
150 continue;
151 }
152 const heading = atxHeading(line);
153 if (heading !== null) { blocks.push(heading); at++; continue; }
154 if (isRule(line)) { blocks.push({kind: 'rule'}); at++; continue; }
155 if (line.startsWith('>')) {
156 const content: string[] = [];
157 while (at < lines.length && lines[at].startsWith('>')) content.push(lines[at++].slice(1).replace(/^ /, ''));
158 const alert = /^\[!(NOTE|TIP|IMPORTANT|WARNING|CAUTION)\]$/.exec(content[0]);
159 blocks.push(alert ? {kind: 'alert', alert: alert[1], blocks: this.parse(content.slice(1))}
160 : {kind: 'quote', blocks: this.parse(content)});
161 continue;
162 }
163 let marker = listMarker(line);
164 if (marker !== null) {
165 const first = marker, items: Block[][] = [];
166 while (true) {
167 const content = [marker.content];
168 at++;
169 while (at < lines.length && lines[at].startsWith(' ')) content.push(lines[at++].slice(2));
170 items.push(this.parse(content));
171 let next = at;
172 while (next < lines.length && lines[next] === '') next++;
173 const following = next < lines.length ? listMarker(lines[next]) : null;
174 if (following === null || following.marker !== first.marker || following.ordered !== first.ordered) break;
175 marker = following;
176 at = next;
177 }
178 blocks.push({kind: 'list', ordered: first.ordered, start: first.start, items});
179 continue;
180 }
181 if (this.definition(line)) { at++; continue; }
182 if (at + 1 < lines.length && line.includes('|') && lines[at + 1].includes('|')) {
183 const header = splitCells(line), delimiters = splitCells(lines[at + 1]);
184 if (header.length === delimiters.length && delimiters.every(cell => /^:?-+:?$/.test(cell))) {
185 const alignments: Alignment[] = delimiters.map(cell => cell.startsWith(':')
186 ? (cell.endsWith(':') ? 'center' : 'left') : cell.endsWith(':') ? 'right' : '');
187 const rows: string[][] = [];
188 at += 2;
189 while (at < lines.length && lines[at] !== '' && lines[at].includes('|')) {
190 const cells = splitCells(lines[at++]).slice(0, header.length);
191 while (cells.length < header.length) cells.push('');
192 rows.push(cells);
193 }
194 blocks.push({kind: 'table', alignments, header, rows});
195 continue;
196 }
197 }
198 }
199 const text = [line.replace(/^ +/, '')];
200 let level = 0;
201 at++;
202 while (at < lines.length && lines[at] !== '') {
203 const row = lines[at];
204 if (/^(?:={3,}|-{3,})$/.test(row)) { level = row[0] === '=' ? 1 : 2; at++; break; }
205 if (interruptsParagraph(row)) break;
206 text.push(row.replace(/^ +/, ''));
207 at++;
208 }
209 const combined = text.join('\n');
210 blocks.push(level ? {kind: 'heading', level, text: combined}
211 : combined === '[TOC]' ? {kind: 'toc'} : {kind: 'paragraph', text: combined});
212 }
213 return blocks;
214 }
215}

217function backtickCloser(text: string, at: number, length: number): number {
218 while ((at = text.indexOf('`', at)) !== -1) {
219 const end = runEnd(text, at);
220 if (end - at === length) return at;
221 at = end;
222 }
223 return -1;
224}
225function closingBracket(text: string, at: number): number {
226 let depth = 0;
227 while (at < text.length) {
228 const char = text[at];
229 if (char === '\\') { at += 2; continue; }
230 if (char === '`') {
231 const end = runEnd(text, at), close = backtickCloser(text, end, end - at);
232 at = close === -1 ? end : close + end - at;
233 continue;
234 }
235 if (char === '[') depth++;
236 else if (char === ']' && --depth === 0) return at;
237 at++;
238 }
239 return -1;
240}
241function emphasis(pieces: readonly Piece[], start: number, end: number): string {
242 let html = '';
243 for (let at = start; at < end; at++) {
244 const piece = pieces[at];
245 if (typeof piece === 'string') { html += piece; continue; }
246 let close = at + 1;
247 if (piece.canOpen) {
248 while (close < end) {
249 const candidate = pieces[close];
250 if (typeof candidate !== 'string' && candidate.char === piece.char
251 && candidate.length === piece.length && candidate.canClose) break;
252 close++;
253 }
254 } else close = end;
255 if (close === end) { html += piece.char.repeat(piece.length); continue; }
256 const tags = piece.char === '~' ? ['<del>', '</del>']
257 : piece.length === 1 ? ['<em>', '</em>'] : piece.length === 2 ? ['<strong>', '</strong>']
258 : ['<em><strong>', '</strong></em>'];
259 html += tags[0] + emphasis(pieces, at + 1, close) + tags[1];
260 at = close;
261 }
262 return html;
263}

265class Renderer {
266 readonly references: ReadonlyMap<string, Reference>;
267 readonly footnotes: ReadonlyMap<string, string>;
268 private readonly usedIds: Set<string>;
269 private readonly headings: TocEntry[];
270 private readonly numberedNotes: Map<string, NumberedNote>;
271 constructor(references: ReadonlyMap<string, Reference>, footnotes: ReadonlyMap<string, string>) {
272 this.references = references;
273 this.footnotes = footnotes;
274 this.usedIds = new Set<string>();
275 this.headings = [];
276 this.numberedNotes = new Map<string, NumberedNote>();
277 }
278 destination(text: string, at: number, label: string): Destination | null {
279 if (text[at] === '[') {
280 const close = text.indexOf(']', at + 1);
281 if (close !== -1 && !text.slice(at + 1, close).includes('[')) {
282 const name = normalize(text.slice(at + 1, close) || label), reference = this.references.get(name);
283 return reference === undefined ? null : {reference, end: close + 1};
284 }
285 } else if (text[at] === '(') {
286 const url = escapedRead(text, skipSpace(text, at + 1), ' \n)');
287 let next = skipSpace(text, url.end), title: string | null = null, valid = true;
288 if (next > url.end && next < text.length && '"\''.includes(text[next])) {
289 const quoted = escapedRead(text, next + 1, text[next]);
290 if (quoted.end === text.length) valid = false;
291 else { title = quoted.value; next = skipSpace(text, quoted.end + 1); }
292 }
293 if (valid && text[next] === ')') return {reference: {url: url.value, title}, end: next + 1};
294 }
295 const reference = this.references.get(normalize(label));
296 return reference === undefined ? null : {reference, end: at};
297 }
298 bracket(text: string, at: number, links: boolean, notes: boolean): InlineResult | null {
299 if (links && notes && text.startsWith('[^', at)) {
300 const close = text.indexOf(']', at + 2), name = text.slice(at + 2, close).toLowerCase();
301 if (close !== -1 && this.footnotes.has(name)) return {html: this.footnoteReference(name), end: close + 1};
302 }
303 const image = text[at] === '!';
304 if (!image && !links) return null;
305 const opening = at + (image ? 1 : 0), close = closingBracket(text, opening);
306 if (close === -1) return null;
307 const label = text.slice(opening + 1, close), destination = this.destination(text, close + 1, label);
308 if (destination === null) return null;
309 const reference = destination.reference, content = this.inline(label, false, false);
310 const title = reference.title === null ? '' : ` title="${escapeHtml(reference.title)}"`;
311 const url = escapeHtml(reference.url);
312 const html = image ? `<img src="${url}" alt="${stripTags(content)}"${title} />`
313 : `<a href="${url}"${title}>${content}</a>`;
314 return {html, end: destination.end};
315 }
316 inline(text: string, links = true, notes = true): string {
317 const pieces: Piece[] = [];
318 let at = 0;
319 while (at < text.length) {
320 const char = text[at], following = text[at + 1];
321 if (char === '\\') {
322 if (following !== undefined && PUNCTUATION.includes(following)) { pieces.push(escapeHtml(following)); at += 2; }
323 else if (following === '\n') { pieces.push('<br />\n'); at += 2; }
324 else { pieces.push('\\'); at++; }
325 continue;
326 }
327 if (char === '`') {
328 const end = runEnd(text, at), close = backtickCloser(text, end, end - at);
329 if (close === -1) { pieces.push(text.slice(at, end)); at = end; }
330 else {
331 let code = text.slice(end, close).replace(/\n/g, ' ');
332 if (code.startsWith(' ') && code.endsWith(' ') && /[^ ]/.test(code)) code = code.slice(1, -1);
333 pieces.push(`<code>${escapeHtml(code)}</code>`);
334 at = close + end - at;
335 }
336 continue;
337 }
338 if (links && char === '<') {
339 const close = text.indexOf('>', at + 1);
340 if (close !== -1) {
341 const target = text.slice(at + 1, close);
342 let href = '';
343 const clean = !/[ \n<]/.test(target);
344 if (clean && /^(?:http:\/\/.+|https:\/\/.+)$/.test(target)) href = target;
345 else if (clean && /^[A-Za-z0-9.+_-]+@[A-Za-z0-9-]+(?:\.[A-Za-z0-9-]+)+$/.test(target)) href = 'mailto:' + target;
346 if (href) { pieces.push(`<a href="${escapeHtml(href)}">${escapeHtml(target)}</a>`); at = close + 1; continue; }
347 }
348 }
349 if (links && (text.startsWith('http://', at) || text.startsWith('https://', at))
350 && (at === 0 || ' \n*_~('.includes(text[at - 1]))) {
351 let end = at;
352 while (end < text.length && !' \n<'.includes(text[end])) end++;
353 while (end > at && '.,:;!?*_~)'.includes(text[end - 1])) end--;
354 const url = text.slice(at, end);
355 if (url !== 'http://' && url !== 'https://') { pieces.push(`<a href="${escapeHtml(url)}">${escapeHtml(url)}</a>`); at = end; continue; }
356 }
357 if (char === '[' || (char === '!' && following === '[')) {
358 const result = this.bracket(text, at, links, notes);
359 if (result !== null) { pieces.push(result.html); at = result.end; continue; }
360 pieces.push(char); at++; continue;
361 }
362 if ('*_~'.includes(char)) {
363 const end = runEnd(text, at), length = end - at;
364 const before = at === 0 ? ' ' : text[at - 1], after = end === text.length ? ' ' : text[end];
365 let canOpen = !WHITESPACE.includes(after), canClose = !WHITESPACE.includes(before);
366 if (char === '_' && /^[A-Za-z0-9]$/.test(before) && /^[A-Za-z0-9]$/.test(after)) canOpen = canClose = false;
367 if ((char !== '~' && length <= 3) || (char === '~' && length === 2)) pieces.push({char, length, canOpen, canClose});
368 else pieces.push(char.repeat(length));
369 at = end;
370 continue;
371 }
372 pieces.push(escapeHtml(char)); at++;
373 }
374 return emphasis(pieces, 0, pieces.length);
375 }
376 blocks(blocks: readonly Block[]): string { return blocks.map(block => this.block(block) + '\n').join(''); }
377 block(block: Block): string {
378 switch (block.kind) {
379 case 'heading': return this.heading(block.level, block.text);
380 case 'paragraph': return `<p>${this.inline(block.text)}</p>`;
381 case 'rule': return '<hr />';
382 case 'toc': return TOC_MARKER;
383 case 'code': {
384 const attribute = block.language ? ` class="language-${escapeHtml(block.language)}"` : '';
385 return `<pre><code${attribute}>${block.lines.map(line => escapeHtml(line) + '\n').join('')}</code></pre>`;
386 }
387 case 'quote': return `<blockquote>\n${this.blocks(block.blocks)}</blockquote>`;
388 case 'alert': {
389 const kind = block.alert.toLowerCase(), title = kind[0].toUpperCase() + kind.slice(1);
390 return `<div class="alert alert-${kind}">\n<p class="alert-title">${title}</p>\n${this.blocks(block.blocks)}</div>`;
391 }
392 case 'list': {
393 const tag = block.ordered ? 'ol' : 'ul', attribute = block.ordered && block.start !== 1 ? ` start="${block.start}"` : '';
394 return `<${tag}${attribute}>\n${block.items.map(item => this.item(item) + '\n').join('')}</${tag}>`;
395 }
396 case 'table': {
397 let html = '<table>\n<thead>\n<tr>\n' + this.tableCells('th', block.alignments, block.header) + '</tr>\n</thead>\n';
398 if (block.rows.length) html += '<tbody>\n' + block.rows.map(row => '<tr>\n' + this.tableCells('td', block.alignments, row) + '</tr>\n').join('') + '</tbody>\n';
399 return html + '</table>';
400 }
401 }
402 }
403 heading(level: number, text: string): string {
404 const start = text.lastIndexOf(' {#');
405 let id: string;
406 if (text.endsWith('}') && start !== -1 && !text.slice(start + 3, -1).includes('\n') && /^[A-Za-z0-9_-]+$/.test(text.slice(start + 3, -1))) {
407 id = text.slice(start + 3, -1);
408 text = text.slice(0, start).replace(/ +$/, '');
409 } else {
410 const base = text.toLowerCase().replace(/[^a-z0-9_ \n-]/g, '').replace(/[ \n]/g, '-') || 'heading';
411 id = base;
412 for (let suffix = 1; this.usedIds.has(id); suffix++) id = `${base}-${suffix}`;
413 }
414 this.usedIds.add(id);
415 const html = this.inline(text);
416 this.headings.push({level, id, label: stripTags(html), children: []});
417 return `<h${level} id="${id}">${html}</h${level}>`;
418 }
419 item(blocks: readonly Block[]): string {
420 let html = '<li>';
421 blocks.forEach((block, index) => {
422 if (block.kind === 'paragraph') {
423 let text = block.text;
424 if (index === 0 && text.length > 4 && /^(?:\[ \]|\[x\]|\[X\]) /.test(text)) {
425 html += text[1] === ' ' ? '<input type="checkbox" disabled="" /> ' : '<input type="checkbox" checked="" disabled="" /> ';
426 text = text.slice(4);
427 }
428 html += this.inline(text);
429 } else {
430 if (!html.endsWith('\n')) html += '\n';
431 html += this.block(block) + '\n';
432 }
433 });
434 return html + '</li>';
435 }
436 tableCells(tag: string, alignments: readonly Alignment[], cells: readonly string[]): string {
437 return cells.map((cell, index) => {
438 const style = alignments[index] ? ` style="text-align:${alignments[index]}"` : '';
439 return `<${tag}${style}>${this.inline(cell)}</${tag}>\n`;
440 }).join('');
441 }
442 tocList(entries: readonly TocEntry[]): string {
443 return '<ul>\n' + entries.map(entry => `<li><a href="#${entry.id}">${entry.label}</a>`
444 + (entry.children.length ? '\n' + this.tocList(entry.children) + '\n' : '') + '</li>\n').join('') + '</ul>';
445 }
446 tableOfContents(): string {
447 if (!this.headings.length) return '<nav class="toc">\n</nav>';
448 const roots: TocEntry[] = [], parents: TocEntry[] = [];
449 for (const heading of this.headings) {
450 while (parents.length && parents[parents.length - 1].level >= heading.level) parents.pop();
451 (parents.length ? parents[parents.length - 1].children : roots).push(heading);
452 parents.push(heading);
453 }
454 return '<nav class="toc">\n' + this.tocList(roots) + '\n</nav>';
455 }
456 referenceId(number: number, count: number): string { return `fnref-${number}` + (count === 1 ? '' : `-${count}`); }
457 footnoteReference(name: string): string {
458 if (!this.numberedNotes.has(name)) this.numberedNotes.set(name, {number: this.numberedNotes.size + 1, count: 0});
459 const note = this.numberedNotes.get(name)!;
460 note.count++;
461 return `<sup id="${this.referenceId(note.number, note.count)}"><a href="#fn-${note.number}">${note.number}</a></sup>`;
462 }
463 footnoteList(): string {
464 let html = '<section class="footnotes">\n<ol>\n';
465 for (const [name, note] of this.numberedNotes) {
466 const body = this.inline(this.footnotes.get(name)!, true, false), backlinks: string[] = [];
467 for (let count = 1; count <= note.count; count++) backlinks.push(`<a href="#${this.referenceId(note.number, count)}">&#8617;</a>`);
468 html += `<li id="fn-${note.number}">${body} ${backlinks.join(' ')}</li>\n`;
469 }
470 return html + '</ol>\n</section>\n';
471 }
472 document(blocks: readonly Block[]): string {
473 const html = this.blocks(blocks), toc = this.tableOfContents();
474 return html.split(TOC_MARKER).join(toc) + (this.numberedNotes.size ? this.footnoteList() : '');
475 }
