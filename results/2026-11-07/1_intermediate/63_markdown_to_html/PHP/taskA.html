5const WHITESPACE = " \n";
6const PUNCTUATION = '!"#$%&\'()*+,-./:;<=>?@[\\]^_`{|}~';
7const TOC_MARKER = "\0toc\0";

9function escapeHtml(string $text): string
10{
11 return strtr($text, ['&' => '&amp;', '<' => '&lt;', '>' => '&gt;', '"' => '&quot;']);
12}
13
14function stripTags(string $text): string
15{
16 return preg_replace('/<[^>]*>/', '', $text);
17}
18
19function normalizeLabel(string $text): string
20{
21 return trim(preg_replace('/[ \n]+/', ' ', strtolower($text)), ' ');
22}
23
24function skipSpace(string $text, int $at): int
25{
26 while ($at < strlen($text) && str_contains(WHITESPACE, $text[$at])) {
27 $at++;
28 }
29 return $at;
30}
31
32function escapedRead(string $text, int $at, string $stops): array
33{
34 $result = '';
35 while ($at < strlen($text) && !str_contains($stops, $text[$at])) {
36 if ($text[$at] === '\\' && $at + 1 < strlen($text) && str_contains(PUNCTUATION, $text[$at + 1])) {
37 $at++;
38 }
39 $result .= $text[$at++];
40 }
41 return [$result, $at];
42}
43
44function openingFence(string $line): ?array
45{
46 if (!preg_match('/\A(`{3,}|~{3,})/', $line, $match)) {
47 return null;
48 }
49 $fence = $match[0];
50 $info = trim(substr($line, strlen($fence)), ' ');
51 if ($fence[0] === '`' && str_contains($info, '`')) {
52 return null;
53 }
54 return [$fence, $info];
55}
56
57function atxHeading(string $line): ?array
58{
59 if (!preg_match('/\A(#{1,6})(?: |\z)/', $line, $match)) {
60 return null;
61 }
62 $text = trim(substr($line, strlen($match[1])), ' ');
63 $trimmed = rtrim($text, '#');
64 if ($trimmed === '' || str_ends_with($trimmed, ' ')) {
65 $text = trim($trimmed, ' ');
66 }
67 return ['kind' => 'heading', 'text' => $text, 'number' => strlen($match[1])];
68}
69
70function isRule(string $line): bool
71{
72 return preg_match('/\A(?:-{3,}|\*{3,}|_{3,})\z/', str_replace(' ', '', $line)) === 1;
73}
74
75function listMarker(string $line): ?array
76{
77 if (isRule($line) || !preg_match('/\A(?:([-*+])|([0-9]{1,9})([.)])) +/', $line, $match)) {
78 return null;
79 }
80 $ordered = ($match[2] ?? '') !== '';
81 return [
82 $ordered ? $match[3] : $match[1], $ordered,
83 $ordered ? (int) $match[2] : 1, substr($line, strlen($match[0])),
84 ];
85}
86
87function interruptsParagraph(string $line): bool
88{
89 return !str_starts_with($line, ' ') && (
90 openingFence($line) !== null || atxHeading($line) !== null || isRule($line)
91 || str_starts_with($line, '>') || listMarker($line) !== null
92 );
93}
94
95function splitCells(string $line): array
96{
97 $text = trim($line, ' ');
98 $at = str_starts_with($text, '|') ? 1 : 0;
99 $cells = [''];
100 $separator = false;
101 while ($at < strlen($text)) {
102 $char = $text[$at++];
103 $separator = false;
104 $last = array_key_last($cells);
105 if ($char === '\\' && $at < strlen($text)) {
106 if ($text[$at] !== '|') {
107 $cells[$last] .= '\\';
108 }
109 $cells[$last] .= $text[$at++];
110 } elseif ($char === '|') {
111 $cells[] = '';
112 $separator = true;
113 } else {
114 $cells[$last] .= $char;
115 }
116 }
117 if ($separator) {
118 array_pop($cells);
119 }
120 return array_map(fn(string $cell): string => trim($cell, ' '), $cells);
121}

123final class BlockParser
124{
125 public array $references = [];
126 public array $footnotes = [];
127
128 public function parse(array $lines): array
129 {
130 $blocks = [];
131 $at = 0;
132 while ($at < count($lines)) {
133 $line = $lines[$at];
134 if ($line === '') {
135 $at++;
136 continue;
137 }
138 if (str_starts_with($line, ' ')) {
139 $end = $at;
140 while ($end < count($lines) && ($lines[$end] === '' || str_starts_with($lines[$end], ' '))) {
141 $end++;
142 }
143 while ($lines[$end - 1] === '') {
144 $end--;
145 }
146 $blocks[] = ['kind' => 'code', 'text' => '', 'lines' => array_map(
147 fn(string $row): string => substr($row, 4), array_slice($lines, $at, $end - $at),
148 )];
149 $at = $end;
150 continue;
151 }
152 if (!str_starts_with($line, ' ')) {
153 if (($fence = openingFence($line)) !== null) {
154 [$marker, $info] = $fence;
155 $code = [];
156 $at++;
157 while ($at < count($lines)) {
158 $row = $lines[$at++];
159 if (strlen($row) >= strlen($marker) && strspn($row, $marker[0]) === strlen($row)) {
160 break;
161 }
162 $code[] = $row;
163 }
164 $blocks[] = ['kind' => 'code', 'text' => explode(' ', $info)[0], 'lines' => $code];
165 continue;
166 }
167 if (($heading = atxHeading($line)) !== null) {
168 $blocks[] = $heading;
169 $at++;
170 continue;
171 }
172 if (isRule($line)) {
173 $blocks[] = ['kind' => 'rule'];
174 $at++;
175 continue;
176 }
177 if (str_starts_with($line, '>')) {
178 $content = [];
179 while ($at < count($lines) && str_starts_with($lines[$at], '>')) {
180 $row = substr($lines[$at++], 1);
181 $content[] = str_starts_with($row, ' ') ? substr($row, 1) : $row;
182 }
183 $alert = preg_match('/\A\[!(NOTE|TIP|IMPORTANT|WARNING|CAUTION)\]\z/', $content[0], $match) === 1;
184 if ($alert) {
185 array_shift($content);
186 }
187 $blocks[] = ['kind' => $alert ? 'alert' : 'quote', 'text' => $alert ? $match[1] : '', 'children' => $this->parse($content)];
188 continue;
189 }
190 if (($marker = listMarker($line)) !== null) {
191 [$kind, $ordered, $number] = $marker;
192 $items = [];
193 while (true) {
194 $content = [$marker[3]];
195 $at++;
196 while ($at < count($lines) && str_starts_with($lines[$at], ' ')) {
197 $content[] = substr($lines[$at++], 2);
198 }
199 $items[] = $this->parse($content);
200 $following = $at;
201 while ($following < count($lines) && $lines[$following] === '') {
202 $following++;
203 }
204 $marker = $following < count($lines) ? listMarker($lines[$following]) : null;
205 if ($marker === null || $marker[0] !== $kind || $marker[1] !== $ordered) {
206 break;
207 }
208 $at = $following;
209 }
210 $blocks[] = ['kind' => 'list', 'ordered' => $ordered, 'number' => $number, 'items' => $items];
211 continue;
212 }
213 if ($this->definition($line)) {
214 $at++;
215 continue;
216 }
217 if ($at + 1 < count($lines) && str_contains($line, '|') && str_contains($lines[$at + 1], '|')) {
218 $header = splitCells($line);
219 $delimiters = splitCells($lines[$at + 1]);
220 if (count($header) === count($delimiters) && array_filter($delimiters, fn(string $cell): bool => preg_match('/\A:?-+:?\z/', $cell) !== 1) === []) {
221 $alignments = array_map(
222 fn(string $cell): ?string => str_starts_with($cell, ':')
223 ? (str_ends_with($cell, ':') ? 'center' : 'left') : (str_ends_with($cell, ':') ? 'right' : null),
224 $delimiters,
225 );
226 $rows = [];
227 $at += 2;
228 while ($at < count($lines) && str_contains($lines[$at], '|')) {
229 $rows[] = array_pad(array_slice(splitCells($lines[$at++]), 0, count($header)), count($header), '');
230 }
231 $blocks[] = ['kind' => 'table', 'alignments' => $alignments, 'lines' => $header, 'rows' => $rows];
232 continue;
233 }
234 }
235 }
236 $text = [ltrim($line, ' ')];
237 $at++;
238 $level = null;
239 while ($at < count($lines) && $lines[$at] !== '') {
240 if (preg_match('/\A(?:={3,}|-{3,})\z/', $lines[$at])) {
241 $level = $lines[$at++][0] === '=' ? 1 : 2;
242 break;
243 }
244 if (interruptsParagraph($lines[$at])) {
245 break;
246 }
247 $text[] = ltrim($lines[$at++], ' ');
248 }
249 $text = implode("\n", $text);
250 $blocks[] = ['kind' => $level !== null ? 'heading' : ($text === '[TOC]' ? 'toc' : 'paragraph'), 'text' => $text, 'number' => $level];
251 }
252 return $blocks;
253 }

255 private function definition(string $line): bool
256 {
257 if (preg_match('/\A\[\^([A-Za-z0-9_-]+)\]:(.*)\z/', $line, $match) && trim($match[2], ' ') !== '') {
258 $name = strtolower($match[1]);
259 $this->footnotes[$name] ??= trim($match[2], ' ');
260 return true;
261 }
262 if (!str_starts_with($line, '[')) {
263 return false;
264 }
265 $close = strpos($line, ']');
266 if ($close === false || ($line[$close + 1] ?? '') !== ':') {
267 return false;
268 }
269 $label = substr($line, 1, $close - 1);
270 if (str_contains($label, '[') || trim($label, ' ') === '' || str_starts_with($label, '^')) {
271 return false;
272 }
273 [$url, $end] = escapedRead($line, skipSpace($line, $close + 2), ' ');
274 if ($url === '') {
275 return false;
276 }
277 $at = skipSpace($line, $end);
278 $title = null;
279 if ($at < strlen($line)) {
280 if ($at === $end || !str_contains('"\'', $line[$at])) {
281 return false;
282 }
283 [$title, $end] = escapedRead($line, $at + 1, $line[$at]);
284 if ($end !== strlen($line) - 1) {
285 return false;
286 }
287 }
288 $this->references[normalizeLabel($label)] ??= [$url, $title];
289 return true;
290 }
291}

293function runEnd(string $text, int $at): int
294{
295 $char = $text[$at++];
296 while ($at < strlen($text) && $text[$at] === $char) {
297 $at++;
298 }
299 return $at;
300}
301
302function backtickCloser(string $text, int $at, int $length): ?int
303{
304 while (($at = strpos($text, '`', $at)) !== false) {
305 $end = runEnd($text, $at);
306 if ($end - $at === $length) {
307 return $at;
308 }
309 $at = $end;
310 }
311 return null;
312}
313
314function closingBracket(string $text, int $at): ?int
315{
316 $depth = 0;
317 while ($at < strlen($text)) {
318 $char = $text[$at];
319 if ($char === '\\') {
320 $at += 2;
321 continue;
322 }
323 if ($char === '`') {
324 $end = runEnd($text, $at);
325 $close = backtickCloser($text, $end, $end - $at);
326 $at = $close === null ? $end : $close + $end - $at;
327 continue;
328 }
329 if ($char === '[') {
330 $depth++;
331 } elseif ($char === ']' && --$depth === 0) {
332 return $at;
333 }
334 $at++;
335 }
336 return null;
337}

341 public function __construct(public string $char, public int $length, public bool $canOpen, public bool $canClose) {}

343
344function emphasis(array $pieces, int $start = 0, ?int $end = null): string
345{
346 $end ??= count($pieces);
347 $html = '';
348 for ($at = $start; $at < $end; $at++) {
349 $piece = $pieces[$at];
350 if (is_string($piece)) {
351 $html .= $piece;
352 continue;
353 }
354 $close = $end;
355 if ($piece->canOpen) {
356 for ($index = $at + 1; $index < $end; $index++) {
357 $other = $pieces[$index];
358 if ($other instanceof Delimiter && $other->char === $piece->char && $other->length === $piece->length && $other->canClose) {
359 $close = $index;
360 break;
361 }
362 }
363 }
364 if ($close === $end) {
365 $html .= str_repeat($piece->char, $piece->length);
366 } else {
367 [$opening, $closing] = $piece->char === '~' ? ['<del>', '</del>'] : match ($piece->length) {
368 1 => ['<em>', '</em>'], 2 => ['<strong>', '</strong>'], 3 => ['<em><strong>', '</strong></em>'],
369 };
370 $html .= $opening . emphasis($pieces, $at + 1, $close) . $closing;
371 $at = $close;
372 }
373 }
374 return $html;
375}

377final class Renderer
378{
379 private array $usedIds = [];
380 private array $headings = [];
381 private array $noteNumbers = [];
382 private array $noteCounts = [];
383
384 public function __construct(private array $references, private array $footnotes) {}

386 private function inline(string $text, bool $links = true, bool $notes = true): string
387 {
388 $pieces = [];
389 $at = 0;
390 while ($at < strlen($text)) {
391 $char = $text[$at];
392 $following = $text[$at + 1] ?? '';
393 if ($char === '\\') {
394 if ($following !== '' && str_contains(PUNCTUATION, $following)) {
395 $pieces[] = escapeHtml($following);
396 $at += 2;
397 } elseif ($following === "\n") {
398 $pieces[] = "<br />\n";
399 $at += 2;
400 } else {
401 $pieces[] = '\\';
402 $at++;
403 }
404 continue;
405 }
406 if ($char === '`') {
407 $end = runEnd($text, $at);
408 $close = backtickCloser($text, $end, $end - $at);
409 if ($close === null) {
410 $pieces[] = substr($text, $at, $end - $at);
411 $at = $end;
412 } else {
413 $code = str_replace("\n", ' ', substr($text, $end, $close - $end));
414 if (str_starts_with($code, ' ') && str_ends_with($code, ' ') && trim($code, ' ') !== '') {
415 $code = substr($code, 1, -1);
416 }
417 $pieces[] = '<code>' . escapeHtml($code) . '</code>';
418 $at = $close + $end - $at;
419 }
420 continue;
421 }
422 if ($links && $char === '<' && ($close = strpos($text, '>', $at + 1)) !== false) {
423 $target = substr($text, $at + 1, $close - $at - 1);
424 $href = null;
425 if (!preg_match('/[ \n<]/', $target) && preg_match('~\Ahttps?://.+~s', $target)) {
426 $href = $target;
427 } elseif (preg_match('/\A[A-Za-z0-9.+_-]+@[A-Za-z0-9-]+(?:\.[A-Za-z0-9-]+)+\z/', $target)) {
428 $href = 'mailto:' . $target;
429 }
430 if ($href !== null) {
431 $pieces[] = '<a href="' . escapeHtml($href) . '">' . escapeHtml($target) . '</a>';
432 $at = $close + 1;
433 continue;
434 }
435 }
436 if ($links && ($at === 0 || str_contains(" \n*_~(", $text[$at - 1])) && preg_match('~\Ahttps?://[^ \n<]*~', substr($text, $at), $match)) {
437 $url = rtrim($match[0], '.,:;!?*_~)');
438 if ($url !== 'http://' && $url !== 'https://') {
439 $pieces[] = '<a href="' . escapeHtml($url) . '">' . escapeHtml($url) . '</a>';
440 $at += strlen($url);
441 continue;
442 }
443 }
444 if ($char === '[' || ($char === '!' && $following === '[')) {
445 if ($links && $notes && substr($text, $at, 2) === '[^' && ($close = strpos($text, ']', $at + 2)) !== false) {
446 $name = strtolower(substr($text, $at + 2, $close - $at - 2));
447 if (array_key_exists($name, $this->footnotes)) {
448 $pieces[] = $this->footnoteReference($name);
449 $at = $close + 1;
450 continue;
451 }
452 }
453 $image = $char === '!';
454 $opening = $at + ($image ? 1 : 0);
455 if (($image || $links) && ($close = closingBracket($text, $opening)) !== null) {
456 $label = substr($text, $opening + 1, $close - $opening - 1);
457 if (($destination = $this->destination($text, $close + 1, $label)) !== null) {
458 [[$url, $title], $end] = $destination;
459 $attribute = $title === null ? '' : ' title="' . escapeHtml($title) . '"';
460 $content = $this->inline($label, false, false);
461 $pieces[] = $image
462 ? '<img src="' . escapeHtml($url) . '" alt="' . stripTags($content) . '"' . $attribute . ' />'
463 : '<a href="' . escapeHtml($url) . '"' . $attribute . '>' . $content . '</a>';
464 $at = $end;
465 continue;
466 }
467 }
468 }
469 if (str_contains('*_~', $char)) {
470 $end = runEnd($text, $at);
471 $length = $end - $at;
472 $before = $at === 0 ? ' ' : $text[$at - 1];
473 $after = $text[$end] ?? ' ';
474 $canOpen = !str_contains(WHITESPACE, $after);
475 $canClose = !str_contains(WHITESPACE, $before);
476 if ($char === '_' && ctype_alnum($before) && ctype_alnum($after)) {
477 $canOpen = $canClose = false;
478 }
479 $pieces[] = ($char !== '~' && $length <= 3) || ($char === '~' && $length === 2)
480 ? new Delimiter($char, $length, $canOpen, $canClose) : str_repeat($char, $length);
481 $at = $end;
482 } else {
483 $pieces[] = escapeHtml($char);
484 $at++;
485 }
486 }
487 return emphasis($pieces);
488 }

490 private function destination(string $text, int $at, string $label): ?array
491 {
492 if (($text[$at] ?? '') === '[') {
493 $close = strpos($text, ']', $at + 1);
494 if ($close !== false && !str_contains(substr($text, $at + 1, $close - $at - 1), '[')) {
495 $name = substr($text, $at + 1, $close - $at - 1);
496 $reference = $this->references[normalizeLabel($name === '' ? $label : $name)] ?? null;
497 return $reference === null ? null : [$reference, $close + 1];
498 }
499 } elseif (($text[$at] ?? '') === '(') {
500 [$url, $end] = escapedRead($text, skipSpace($text, $at + 1), " \n)");
501 $position = skipSpace($text, $end);
502 $title = null;
503 $valid = true;
504 if ($position > $end && $position < strlen($text) && str_contains('"\'', $text[$position])) {
505 [$title, $end] = escapedRead($text, $position + 1, $text[$position]);
506 $valid = $end < strlen($text);
507 if ($valid) {
508 $position = skipSpace($text, $end + 1);
509 }
510 }
511 if ($valid && ($text[$position] ?? '') === ')') {
512 return [[$url, $title], $position + 1];
513 }
514 }
515 $reference = $this->references[normalizeLabel($label)] ?? null;
516 return $reference === null ? null : [$reference, $at];
517 }

519 public function document(array $blocks): string
520 {
521 $html = $this->renderBlocks($blocks);
522 $html = str_replace(TOC_MARKER, $this->tableOfContents(), $html);
523 return $html . ($this->noteNumbers === [] ? '' : $this->footnoteList());

526 private function renderBlocks(array $blocks): string
527 {
528 return implode('', array_map(fn(array $block): string => $this->render($block) . "\n", $blocks));
529 }
530
531 private function render(array $block): string
532 {
533 switch ($block['kind']) {
534 case 'heading': return $this->heading($block['number'], $block['text']);
535 case 'paragraph': return '<p>' . $this->inline($block['text']) . '</p>';
536 case 'rule': return '<hr />';
537 case 'code':
538 $attribute = $block['text'] === '' ? '' : ' class="language-' . escapeHtml($block['text']) . '"';
539 $code = implode('', array_map(fn(string $line): string => escapeHtml($line) . "\n", $block['lines']));
540 return "<pre><code$attribute>$code</code></pre>";
541 case 'quote': return "<blockquote>\n" . $this->renderBlocks($block['children']) . '</blockquote>';
542 case 'alert':
543 $kind = strtolower($block['text']);
544 return '<div class="alert alert-' . $kind . '">' . "\n" . '<p class="alert-title">' . ucfirst($kind) . "</p>\n" . $this->renderBlocks($block['children']) . '</div>';
545 case 'list':
546 $tag = $block['ordered'] ? 'ol' : 'ul';
547 $attribute = $block['ordered'] && $block['number'] !== 1 ? ' start="' . $block['number'] . '"' : '';
548 $items = implode('', array_map(fn(array $item): string => $this->renderItem($item) . "\n", $block['items']));
549 return "<$tag$attribute>\n$items</$tag>";
550 case 'table':
551 $html = "<table>\n<thead>\n<tr>\n" . $this->tableCells('th', $block, $block['lines']) . "</tr>\n</thead>\n";
552 if ($block['rows'] !== []) {
553 $html .= "<tbody>\n";
554 foreach ($block['rows'] as $row) {
555 $html .= "<tr>\n" . $this->tableCells('td', $block, $row) . "</tr>\n";
556 }
557 $html .= "</tbody>\n";
558 }
559 return $html . '</table>';
560 case 'toc': return TOC_MARKER;
561 }
562 return '';
563 }

565 private function heading(int $level, string $text): string
566 {
567 $start = strrpos($text, ' {#');
568 $candidate = $start !== false && str_ends_with($text, '}') ? substr($text, $start + 3, -1) : '';
569 if (preg_match('/\A[A-Za-z0-9_-]+\z/', $candidate)) {
570 $id = $candidate;
571 $text = rtrim(substr($text, 0, $start), ' ');
572 } else {
573 $base = str_replace([' ', "\n"], '-', preg_replace('/[^a-z0-9_ \n-]/', '', strtolower($text)));
574 if ($base === '') {
575 $base = 'heading';
576 }
577 $id = $base;
578 for ($suffix = 1; isset($this->usedIds[$id]); $suffix++) {
579 $id = "$base-$suffix";
580 }
581 }
582 $this->usedIds[$id] = true;
583 $html = $this->inline($text);
584 $this->headings[] = ['level' => $level, 'id' => $id, 'label' => stripTags($html), 'children' => []];
585 return "<h$level id=\"$id\">$html</h$level>";
586 }

588 private function renderItem(array $blocks): string
589 {
590 $html = '<li>';
591 foreach ($blocks as $index => $block) {
592 if ($block['kind'] === 'paragraph') {
593 $text = $block['text'];
594 if ($index === 0 && strlen($text) > 4 && in_array(substr($text, 0, 4), ['[ ] ', '[x] ', '[X] '], true)) {
595 $checked = $text[1] === ' ' ? '' : ' checked=""';
596 $html .= '<input type="checkbox"' . $checked . ' disabled="" /> ';
597 $text = substr($text, 4);
598 }
599 $html .= $this->inline($text);
600 } else {
601 if (!str_ends_with($html, "\n")) {
602 $html .= "\n";
603 }
604 $html .= $this->render($block) . "\n";
605 }
606 }
607 return $html . '</li>';
608 }

635 private function tableOfContents(): string
636 {
637 if ($this->headings === []) {
638 return "<nav class=\"toc\">\n</nav>";
639 }
640 $roots = [];
641 $parents = [];
642 foreach ($this->headings as $index => $entry) {
643 while ($parents !== [] && $this->headings[$parents[array_key_last($parents)]]['level'] >= $entry['level']) {
644 array_pop($parents);
645 }
646 if ($parents === []) {
647 $roots[] = $index;
648 } else {
649 $this->headings[$parents[array_key_last($parents)]]['children'][] = $index;
650 }
651 $parents[] = $index;
652 }
653 return "<nav class=\"toc\">\n" . $this->tocList($roots) . "\n</nav>";
654 }

656 private function referenceId(int $number, int $count): string
657 {
658 return "fnref-$number" . ($count === 1 ? '' : "-$count");
659 }

661 private function footnoteReference(string $name): string
662 {
663 if (!isset($this->noteNumbers[$name])) {
664 $this->noteNumbers[$name] = count($this->noteNumbers) + 1;
665 $this->noteCounts[$name] = 0;
666 }
667 $number = $this->noteNumbers[$name];
668 $id = $this->referenceId($number, ++$this->noteCounts[$name]);
669 return "<sup id=\"$id\"><a href=\"#fn-$number\">$number</a></sup>";
670 }

672 private function footnoteList(): string
673 {
674 $html = "<section class=\"footnotes\">\n<ol>\n";
675 foreach ($this->noteNumbers as $name => $number) {
676 $body = $this->inline($this->footnotes[$name], notes: false);
677 $backlinks = [];
678 for ($count = 1; $count <= $this->noteCounts[$name]; $count++) {
679 $backlinks[] = '<a href="#' . $this->referenceId($number, $count) . '">&#8617;</a>';
680 }
681 $html .= "<li id=\"fn-$number\">$body " . implode(' ', $backlinks) . "</li>\n";
682 }
683 return $html . "</ol>\n</section>\n";
684 }
