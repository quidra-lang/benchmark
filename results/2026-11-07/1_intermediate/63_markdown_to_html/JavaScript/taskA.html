1'use strict';
2const fs = require('node:fs');
3const PUNCTUATION = '!"#$%&\'()*+,-./:;<=>?@[\\]^_`{|}~';
4const WHITESPACE = ' \n';
5const TOC_MARKER = '\x00toc\x00';
6const htmlEscapes = {'&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;'};
7function escapeHtml(text) { return text.replace(/[&<>"]/g, char => htmlEscapes[char]); }
8function stripTags(text) { return text.replace(/<[^>]*>/g, ''); }
9function normalize(text) { return text.toLowerCase().replace(/[ \n]+/g, ' ').trim(); }
10function skipSpace(text, at) {
11 while (at < text.length && WHITESPACE.includes(text[at])) at++;
12 return at;
13}
14function escapedRead(text, at, stops) {
15 let value = '';
16 while (at < text.length && !stops.includes(text[at])) {
17 if (text[at] === '\\' && at + 1 < text.length && PUNCTUATION.includes(text[at + 1])) at++;
18 value += text[at++];
19 }
20 return {value, end: at};
21}
22function runEnd(text, at) {
23 const char = text[at++];
24 while (at < text.length && text[at] === char) at++;
25 return at;
26}
27function openingFence(line) {
28 if (!line.startsWith('`') && !line.startsWith('~')) return null;
29 const end = runEnd(line, 0), fence = line.slice(0, end), info = line.slice(end).trim();
30 if (end < 3 || (line[0] === '`' && info.includes('`'))) return null;
31 return {fence, info};
32}
33function atxHeading(line) {
34 const match = /^(#{1,6})(?: |$)/.exec(line);
35 if (!match) return null;
36 let text = line.slice(match[1].length).trim();
37 const trimmed = text.replace(/#+$/, '');
38 if (!trimmed || trimmed.endsWith(' ')) text = trimmed.trim();
39 return {kind: 'heading', level: match[1].length, text};
40}
41function isRule(line) { return /^(?:-{3,}|\*{3,}|_{3,})$/.test(line.replace(/ /g, '')); }
42function listMarker(line) {
43 if (isRule(line)) return null;
44 const match = /^(?:([-*+])|([0-9]{1,9})([.)])) +/.exec(line);
45 if (!match) return null;
46 return {marker: match[1] || match[3], ordered: match[2] !== undefined,
47 start: match[2] === undefined ? 1 : Number(match[2]), content: line.slice(match[0].length)};
48}
49function interruptsParagraph(line) {
50 return !line.startsWith(' ') && (openingFence(line) !== null || atxHeading(line) !== null
51 || isRule(line) || line.startsWith('>') || listMarker(line) !== null);
52}
53function splitCells(line) {
54 let text = line.trim();
55 if (text.startsWith('|')) text = text.slice(1);
56 const cells = [''];
57 let finalSeparator = false;
58 for (let at = 0; at < text.length;) {
59 const char = text[at++];
60 finalSeparator = false;
61 if (char === '\\' && at < text.length) {
62 cells[cells.length - 1] += (text[at] === '|' ? '' : '\\') + text[at++];
63 } else if (char === '|') {
64 cells.push('');
65 finalSeparator = true;
66 } else cells[cells.length - 1] += char;
67 }
68 if (finalSeparator) cells.pop();
69 return cells.map(cell => cell.trim());
70}

72class BlockParser {
73 constructor() {
74 this.references = new Map();
75 this.footnotes = new Map();
76 }
77 definition(line) {
78 const footnote = /^\[\^([A-Za-z0-9_-]+)\]:(.*)$/.exec(line);
79 if (footnote && footnote[2].trim()) {
80 const name = footnote[1].toLowerCase();
81 if (!this.footnotes.has(name)) this.footnotes.set(name, footnote[2].trim());
82 return true;
83 }
84 if (!line.startsWith('[')) return false;
85 const close = line.indexOf(']');
86 if (close < 0 || line[close + 1] !== ':') return false;
87 const label = line.slice(1, close);
88 if (label.includes('[') || !label.trim() || label.startsWith('^')) return false;
89 const url = escapedRead(line, skipSpace(line, close + 2), ' ');
90 if (!url.value) return false;
91 const at = skipSpace(line, url.end);
92 let title = null;
93 if (at < line.length) {
94 if (at === url.end || !'"\''.includes(line[at])) return false;
95 const quoted = escapedRead(line, at + 1, line[at]);
96 if (quoted.end !== line.length - 1) return false;
97 title = quoted.value;
98 }
99 const name = normalize(label);
100 if (!this.references.has(name)) this.references.set(name, {url: url.value, title});
101 return true;
102 }
103 parse(lines) {
104 const blocks = [];
105 let at = 0;
106 while (at < lines.length) {
107 const line = lines[at];
108 if (line === '') { at++; continue; }
109 if (line.startsWith(' ')) {
110 let end = at;
111 while (end < lines.length && (lines[end] === '' || lines[end].startsWith(' '))) end++;
112 while (lines[end - 1] === '') end--;
113 blocks.push({kind: 'code', language: '', lines: lines.slice(at, end).map(row => row.slice(4))});
114 at = end;
115 continue;
116 }
117 if (!line.startsWith(' ')) {
118 const fence = openingFence(line);
119 if (fence !== null) {
120 const code = [];
121 at++;
122 while (at < lines.length) {
123 const row = lines[at++];
124 if (row.length >= fence.fence.length && [...row].every(char => char === fence.fence[0])) break;
125 code.push(row);
126 }
127 blocks.push({kind: 'code', language: fence.info.split(' ')[0], lines: code});
128 continue;
129 }
130 const heading = atxHeading(line);
131 if (heading !== null) { blocks.push(heading); at++; continue; }
132 if (isRule(line)) { blocks.push({kind: 'rule'}); at++; continue; }
133 if (line.startsWith('>')) {
134 const content = [];
135 while (at < lines.length && lines[at].startsWith('>')) content.push(lines[at++].slice(1).replace(/^ /, ''));
136 const alert = /^\[!(NOTE|TIP|IMPORTANT|WARNING|CAUTION)\]$/.exec(content[0]);
137 blocks.push(alert ? {kind: 'alert', alert: alert[1], blocks: this.parse(content.slice(1))}
138 : {kind: 'quote', blocks: this.parse(content)});
139 continue;
140 }
141 let marker = listMarker(line);
142 if (marker !== null) {
143 const first = marker, items = [];
144 while (true) {
145 const content = [marker.content];
146 at++;
147 while (at < lines.length && lines[at].startsWith(' ')) content.push(lines[at++].slice(2));
148 items.push(this.parse(content));
149 let next = at;
150 while (next < lines.length && lines[next] === '') next++;
151 const following = next < lines.length ? listMarker(lines[next]) : null;
152 if (following === null || following.marker !== first.marker || following.ordered !== first.ordered) break;
153 marker = following;
154 at = next;
155 }
156 blocks.push({kind: 'list', ordered: first.ordered, start: first.start, items});
157 continue;
158 }
159 if (this.definition(line)) { at++; continue; }
160 if (at + 1 < lines.length && line.includes('|') && lines[at + 1].includes('|')) {
161 const header = splitCells(line), delimiters = splitCells(lines[at + 1]);
162 if (header.length === delimiters.length && delimiters.every(cell => /^:?-+:?$/.test(cell))) {
163 const alignments = delimiters.map(cell => cell.startsWith(':')
164 ? (cell.endsWith(':') ? 'center' : 'left') : cell.endsWith(':') ? 'right' : '');
165 const rows = [];
166 at += 2;
167 while (at < lines.length && lines[at] !== '' && lines[at].includes('|')) {
168 const cells = splitCells(lines[at++]).slice(0, header.length);
169 while (cells.length < header.length) cells.push('');
170 rows.push(cells);
171 }
172 blocks.push({kind: 'table', alignments, header, rows});
173 continue;
174 }
175 }
176 }
177 const text = [line.replace(/^ +/, '')];
178 let level = 0;
179 at++;
180 while (at < lines.length && lines[at] !== '') {
181 const row = lines[at];
182 if (/^(?:={3,}|-{3,})$/.test(row)) { level = row[0] === '=' ? 1 : 2; at++; break; }
183 if (interruptsParagraph(row)) break;
184 text.push(row.replace(/^ +/, ''));
185 at++;
186 }
187 const combined = text.join('\n');
188 blocks.push(level ? {kind: 'heading', level, text: combined}
189 : combined === '[TOC]' ? {kind: 'toc'} : {kind: 'paragraph', text: combined});
190 }
191 return blocks;
192 }
193}

195function backtickCloser(text, at, length) {
196 while ((at = text.indexOf('`', at)) !== -1) {
197 const end = runEnd(text, at);
198 if (end - at === length) return at;
199 at = end;
200 }
201 return -1;
202}
203function closingBracket(text, at) {
204 let depth = 0;
205 while (at < text.length) {
206 const char = text[at];
207 if (char === '\\') { at += 2; continue; }
208 if (char === '`') {
209 const end = runEnd(text, at), close = backtickCloser(text, end, end - at);
210 at = close === -1 ? end : close + end - at;
211 continue;
212 }
213 if (char === '[') depth++;
214 else if (char === ']' && --depth === 0) return at;
215 at++;
216 }
217 return -1;
218}
219function emphasis(pieces, start, end) {
220 let html = '';
221 for (let at = start; at < end; at++) {
222 const piece = pieces[at];
223 if (typeof piece === 'string') { html += piece; continue; }
224 let close = at + 1;
225 if (piece.canOpen) {
226 while (close < end) {
227 const candidate = pieces[close];
228 if (typeof candidate !== 'string' && candidate.char === piece.char
229 && candidate.length === piece.length && candidate.canClose) break;
230 close++;
231 }
232 } else close = end;
233 if (close === end) { html += piece.char.repeat(piece.length); continue; }
234 const tags = piece.char === '~' ? ['<del>', '</del>']
235 : piece.length === 1 ? ['<em>', '</em>'] : piece.length === 2 ? ['<strong>', '</strong>']
236 : ['<em><strong>', '</strong></em>'];
237 html += tags[0] + emphasis(pieces, at + 1, close) + tags[1];
238 at = close;
239 }
240 return html;
241}

243class Renderer {
244 constructor(references, footnotes) {
245 this.references = references;
246 this.footnotes = footnotes;
247 this.usedIds = new Set();
248 this.headings = [];
249 this.numberedNotes = new Map();
250 }
251 destination(text, at, label) {
252 if (text[at] === '[') {
253 const close = text.indexOf(']', at + 1);
254 if (close !== -1 && !text.slice(at + 1, close).includes('[')) {
255 const name = normalize(text.slice(at + 1, close) || label), reference = this.references.get(name);
256 return reference === undefined ? null : {reference, end: close + 1};
257 }
258 } else if (text[at] === '(') {
259 const url = escapedRead(text, skipSpace(text, at + 1), ' \n)');
260 let next = skipSpace(text, url.end), title = null, valid = true;
261 if (next > url.end && next < text.length && '"\''.includes(text[next])) {
262 const quoted = escapedRead(text, next + 1, text[next]);
263 if (quoted.end === text.length) valid = false;
264 else { title = quoted.value; next = skipSpace(text, quoted.end + 1); }
265 }
266 if (valid && text[next] === ')') return {reference: {url: url.value, title}, end: next + 1};
267 }
268 const reference = this.references.get(normalize(label));
269 return reference === undefined ? null : {reference, end: at};
270 }
271 bracket(text, at, links, notes) {
272 if (links && notes && text.startsWith('[^', at)) {
273 const close = text.indexOf(']', at + 2), name = text.slice(at + 2, close).toLowerCase();
274 if (close !== -1 && this.footnotes.has(name)) return {html: this.footnoteReference(name), end: close + 1};
275 }
276 const image = text[at] === '!';
277 if (!image && !links) return null;
278 const opening = at + (image ? 1 : 0), close = closingBracket(text, opening);
279 if (close === -1) return null;
280 const label = text.slice(opening + 1, close), destination = this.destination(text, close + 1, label);
281 if (destination === null) return null;
282 const reference = destination.reference, content = this.inline(label, false, false);
283 const title = reference.title === null ? '' : ` title="${escapeHtml(reference.title)}"`;
284 const url = escapeHtml(reference.url);
285 const html = image ? `<img src="${url}" alt="${stripTags(content)}"${title} />`
286 : `<a href="${url}"${title}>${content}</a>`;
287 return {html, end: destination.end};
288 }
289 inline(text, links = true, notes = true) {
290 const pieces = [];
291 let at = 0;
292 while (at < text.length) {
293 const char = text[at], following = text[at + 1];
294 if (char === '\\') {
295 if (following !== undefined && PUNCTUATION.includes(following)) { pieces.push(escapeHtml(following)); at += 2; }
296 else if (following === '\n') { pieces.push('<br />\n'); at += 2; }
297 else { pieces.push('\\'); at++; }
298 continue;
299 }
300 if (char === '`') {
301 const end = runEnd(text, at), close = backtickCloser(text, end, end - at);
302 if (close === -1) { pieces.push(text.slice(at, end)); at = end; }
303 else {
304 let code = text.slice(end, close).replace(/\n/g, ' ');
305 if (code.startsWith(' ') && code.endsWith(' ') && /[^ ]/.test(code)) code = code.slice(1, -1);
306 pieces.push(`<code>${escapeHtml(code)}</code>`);
307 at = close + end - at;
308 }
309 continue;
310 }
311 if (links && char === '<') {
312 const close = text.indexOf('>', at + 1);
313 if (close !== -1) {
314 const target = text.slice(at + 1, close);
315 let href = '';
316 const clean = !/[ \n<]/.test(target);
317 if (clean && /^(?:http:\/\/.+|https:\/\/.+)$/.test(target)) href = target;
318 else if (clean && /^[A-Za-z0-9.+_-]+@[A-Za-z0-9-]+(?:\.[A-Za-z0-9-]+)+$/.test(target)) href = 'mailto:' + target;
319 if (href) { pieces.push(`<a href="${escapeHtml(href)}">${escapeHtml(target)}</a>`); at = close + 1; continue; }
320 }
321 }
322 if (links && (text.startsWith('http://', at) || text.startsWith('https://', at))
323 && (at === 0 || ' \n*_~('.includes(text[at - 1]))) {
324 let end = at;
325 while (end < text.length && !' \n<'.includes(text[end])) end++;
326 while (end > at && '.,:;!?*_~)'.includes(text[end - 1])) end--;
327 const url = text.slice(at, end);
328 if (url !== 'http://' && url !== 'https://') { pieces.push(`<a href="${escapeHtml(url)}">${escapeHtml(url)}</a>`); at = end; continue; }
329 }
330 if (char === '[' || (char === '!' && following === '[')) {
331 const result = this.bracket(text, at, links, notes);
332 if (result !== null) { pieces.push(result.html); at = result.end; continue; }
333 pieces.push(char); at++; continue;
334 }
335 if ('*_~'.includes(char)) {
336 const end = runEnd(text, at), length = end - at;
337 const before = at === 0 ? ' ' : text[at - 1], after = end === text.length ? ' ' : text[end];
338 let canOpen = !WHITESPACE.includes(after), canClose = !WHITESPACE.includes(before);
339 if (char === '_' && /^[A-Za-z0-9]$/.test(before) && /^[A-Za-z0-9]$/.test(after)) canOpen = canClose = false;
340 if ((char !== '~' && length <= 3) || (char === '~' && length === 2)) pieces.push({char, length, canOpen, canClose});
341 else pieces.push(char.repeat(length));
342 at = end;
343 continue;
344 }
345 pieces.push(escapeHtml(char)); at++;
346 }
347 return emphasis(pieces, 0, pieces.length);
348 }
349 blocks(blocks) { return blocks.map(block => this.block(block) + '\n').join(''); }
350 block(block) {
351 switch (block.kind) {
352 case 'heading': return this.heading(block.level, block.text);
353 case 'paragraph': return `<p>${this.inline(block.text)}</p>`;
354 case 'rule': return '<hr />';
355 case 'toc': return TOC_MARKER;
356 case 'code': {
357 const attribute = block.language ? ` class="language-${escapeHtml(block.language)}"` : '';
358 return `<pre><code${attribute}>${block.lines.map(line => escapeHtml(line) + '\n').join('')}</code></pre>`;
359 }
360 case 'quote': return `<blockquote>\n${this.blocks(block.blocks)}</blockquote>`;
361 case 'alert': {
362 const kind = block.alert.toLowerCase(), title = kind[0].toUpperCase() + kind.slice(1);
363 return `<div class="alert alert-${kind}">\n<p class="alert-title">${title}</p>\n${this.blocks(block.blocks)}</div>`;
364 }
365 case 'list': {
366 const tag = block.ordered ? 'ol' : 'ul', attribute = block.ordered && block.start !== 1 ? ` start="${block.start}"` : '';
367 return `<${tag}${attribute}>\n${block.items.map(item => this.item(item) + '\n').join('')}</${tag}>`;
368 }
369 case 'table': {
370 let html = '<table>\n<thead>\n<tr>\n' + this.tableCells('th', block.alignments, block.header) + '</tr>\n</thead>\n';
371 if (block.rows.length) html += '<tbody>\n' + block.rows.map(row => '<tr>\n' + this.tableCells('td', block.alignments, row) + '</tr>\n').join('') + '</tbody>\n';
372 return html + '</table>';
373 }
374 }
375 }
376 heading(level, text) {
377 const start = text.lastIndexOf(' {#');
378 let id;
379 if (text.endsWith('}') && start !== -1 && !text.slice(start + 3, -1).includes('\n') && /^[A-Za-z0-9_-]+$/.test(text.slice(start + 3, -1))) {
380 id = text.slice(start + 3, -1);
381 text = text.slice(0, start).replace(/ +$/, '');
382 } else {
383 const base = text.toLowerCase().replace(/[^a-z0-9_ \n-]/g, '').replace(/[ \n]/g, '-') || 'heading';
384 id = base;
385 for (let suffix = 1; this.usedIds.has(id); suffix++) id = `${base}-${suffix}`;
386 }
387 this.usedIds.add(id);
388 const html = this.inline(text);
389 this.headings.push({level, id, label: stripTags(html), children: []});
390 return `<h${level} id="${id}">${html}</h${level}>`;
391 }
392 item(blocks) {
393 let html = '<li>';
394 blocks.forEach((block, index) => {
395 if (block.kind === 'paragraph') {
396 let text = block.text;
397 if (index === 0 && text.length > 4 && /^(?:\[ \]|\[x\]|\[X\]) /.test(text)) {
398 html += text[1] === ' ' ? '<input type="checkbox" disabled="" /> ' : '<input type="checkbox" checked="" disabled="" /> ';
399 text = text.slice(4);
400 }
401 html += this.inline(text);
402 } else {
403 if (!html.endsWith('\n')) html += '\n';
404 html += this.block(block) + '\n';
405 }
406 });
407 return html + '</li>';
408 }
409 tableCells(tag, alignments, cells) {
410 return cells.map((cell, index) => {
411 const style = alignments[index] ? ` style="text-align:${alignments[index]}"` : '';
412 return `<${tag}${style}>${this.inline(cell)}</${tag}>\n`;
413 }).join('');
414 }
415 tocList(entries) {
416 return '<ul>\n' + entries.map(entry => `<li><a href="#${entry.id}">${entry.label}</a>`
417 + (entry.children.length ? '\n' + this.tocList(entry.children) + '\n' : '') + '</li>\n').join('') + '</ul>';
418 }
419 tableOfContents() {
420 if (!this.headings.length) return '<nav class="toc">\n</nav>';
421 const roots = [], parents = [];
422 for (const heading of this.headings) {
423 while (parents.length && parents[parents.length - 1].level >= heading.level) parents.pop();
424 (parents.length ? parents[parents.length - 1].children : roots).push(heading);
425 parents.push(heading);
426 }
427 return '<nav class="toc">\n' + this.tocList(roots) + '\n</nav>';
428 }
429 referenceId(number, count) { return `fnref-${number}` + (count === 1 ? '' : `-${count}`); }
430 footnoteReference(name) {
431 if (!this.numberedNotes.has(name)) this.numberedNotes.set(name, {number: this.numberedNotes.size + 1, count: 0});
432 const note = this.numberedNotes.get(name);
433 note.count++;
434 return `<sup id="${this.referenceId(note.number, note.count)}"><a href="#fn-${note.number}">${note.number}</a></sup>`;
435 }
436 footnoteList() {
437 let html = '<section class="footnotes">\n<ol>\n';
438 for (const [name, note] of this.numberedNotes) {
439 const body = this.inline(this.footnotes.get(name), true, false), backlinks = [];
440 for (let count = 1; count <= note.count; count++) backlinks.push(`<a href="#${this.referenceId(note.number, count)}">&#8617;</a>`);
441 html += `<li id="fn-${note.number}">${body} ${backlinks.join(' ')}</li>\n`;
442 }
443 return html + '</ol>\n</section>\n';
444 }
445 document(blocks) {
446 const html = this.blocks(blocks), toc = this.tableOfContents();
447 return html.split(TOC_MARKER).join(toc) + (this.numberedNotes.size ? this.footnoteList() : '');
448 }
449}
