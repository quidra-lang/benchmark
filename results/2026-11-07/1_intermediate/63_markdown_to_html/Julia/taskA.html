1const WHITE = " \n"
2const PUNCT = "!\"#\$%&'()*+,-./:;<=>?@[\\]^_`{|}~"
3const TOC = "\x01toc\x02"
4part(s, a, b = lastindex(s)) = a > b ? "" : String(s[a:b])
5starts(s, prefix, at = 1) = startswith(part(s, at), prefix)
6escapehtml(s) = replace(s, '&'=>"&amp;", '<'=>"&lt;", '>'=>"&gt;", '"'=>"&quot;")
7striptags(s) = replace(s, r"<[^>]*>"=>"")
8normalize(s) = join(split(lowercase(s)), " ")
9trim(s) = String(strip(s, ' '))
10ltrim(s) = String(lstrip(s, ' '))
11function skipspace(s, at)
12 while at <= length(s) && s[at] in WHITE
13 at += 1
14 end
15 at
16end
17function escapedread(s, at, stops)
18 out = IOBuffer()
19 while at <= length(s) && !(s[at] in stops)
20 if s[at] == '\\' && at < length(s) && s[at+1] in PUNCT
21 at += 1
22 end
23 print(out, s[at])
24 at += 1
25 end
26 String(take!(out)), at
27end

50function fence(line)
51 m = match(r"^(`{3,}|~{3,})", line)
52 m === nothing && return nothing
53 run = String(m.match)
54 info = trim(part(line, length(run)+1))
55 first(run) == '`' && occursin('`', info) && return nothing
56 run, info
57end
58function atx(line)
59 m = match(r"^(#{1,6})(?: |$)", line)
60 m === nothing && return nothing
61 n = length(m.captures[1])
62 text = trim(part(line, n+1))
63 shorter = String(rstrip(text, '#'))
64 if isempty(shorter) || endswith(shorter, " ")
65 text = trim(shorter)
66 end
67 Block(:heading, text, n)
68end
69rule(line) = occursin(r"^(?:-{3,}|\*{3,}|_{3,})$", replace(line, " "=>""))
70function marker(line)
71 rule(line) && return nothing
72 m = match(r"^(?:([-*+])|([0-9]{1,9})([.)])) +", line)
73 m === nothing && return nothing
74 ordered = m.captures[2] !== nothing
75 kind = String(ordered ? m.captures[3] : m.captures[1])
76 kind, ordered, ordered ? parse(Int, m.captures[2]) : 1, part(line, length(m.match)+1)
77end
78interrupts(line) =
79 !startswith(line, " ") && (
80 fence(line)!==nothing ||
81 atx(line)!==nothing ||
82 rule(line) ||
83 startswith(line, ">") ||
84 marker(line)!==nothing
85 )
86function cells(line)
87 text = trim(line)
88 at = startswith(text, "|") ? 2 : 1
89 result = String[]
90 out = IOBuffer()
91 final_separator = false
92 while at <= length(text)
93 ch = text[at]
94 at += 1
95 final_separator = false
96 if ch == '\\' && at <= length(text)
97 text[at] != '|' && print(out, '\\')
98 print(out, text[at])
99 at += 1
100 elseif ch == '|'
101 push!(result, trim(String(take!(out))))
102 final_separator = true
103 else
104 print(out, ch)
105 end
106 end
107 !final_separator && push!(result, trim(String(take!(out))))
108 result
109end
110function definition!(p, line)
111 m = match(r"^\[\^([A-Za-z0-9_-]+)\]:(.*)$", line)
112 if m !== nothing && !isempty(trim(m.captures[2]))
113 get!(p.footnotes, lowercase(m.captures[1]), trim(m.captures[2]))
114 return true
115 end
116 startswith(line, "[") || return false
117 close = findfirst(']', line)
118 (close === nothing || !starts(line, ":", close+1)) && return false
119 label = part(line, 2, close-1)
120 (occursin('[', label) || isempty(trim(label)) || startswith(label, "^")) && return false
121 url, finish = escapedread(line, skipspace(line, close+2), " ")
122 isempty(url) && return false
123 at = skipspace(line, finish)
124 title = nothing
125 if at <= length(line)
126 (at == finish || !(line[at] in "\"'")) && return false
127 title, finish = escapedread(line, at+1, string(line[at]))
128 finish == length(line) || return false
129 end
130 get!(p.references, normalize(label), Reference(url, title))
131 true
132end
133function parseblocks!(p, lines)
134 blocks = Block[]
135 at = 1
136 while at <= length(lines)
137 line = lines[at]
138 if isempty(line)
139 at += 1
140 continue
141 elseif startswith(line, " ")
142 finish = at
143 while finish <= length(lines) &&
144 (isempty(lines[finish]) || startswith(lines[finish], " "))
145 finish += 1
146 end
147 while isempty(lines[finish-1])
148 finish -= 1
149 end
150 b = Block(:code)
151 b.lines = [part(s, 5) for s in lines[at:(finish-1)]]
152 push!(blocks, b)
153 at = finish
154 continue
155 end
156 if !startswith(line, " ")
157 f = fence(line)
158 if f !== nothing
159 run, info = f
160 b = Block(:code, isempty(info) ? "" : String(first(split(info, ' '))))
161 at += 1
162 while at <= length(lines)
163 row = lines[at]
164 at += 1
165 length(row)>=length(run) && all(==(first(run)), row) && break
166 push!(b.lines, row)
167 end
168 push!(blocks, b)
169 continue
170 end
171 h = atx(line)
172 if h !== nothing
173 push!(blocks, h)
174 at += 1
175 continue
176 end
177 if rule(line)
178 push!(blocks, Block(:rule))
179 at += 1
180 continue
181 end
182 if startswith(line, ">")
183 content = String[]
184 while at <= length(lines) && startswith(lines[at], ">")
185 row = part(lines[at], 2)
186 push!(content, starts(row, " ") ? part(row, 2) : row)
187 at += 1
188 end
189 alert = match(r"^\[!(NOTE|TIP|IMPORTANT|WARNING|CAUTION)\]$", content[1])
190 b = alert === nothing ? Block(:quote) : Block(:alert, String(alert.captures[1]))
191 b.children = parseblocks!(p, alert === nothing ? content : content[2:end])
192 push!(blocks, b)
193 continue
194 end
195 m = marker(line)
196 if m !== nothing
197 kind, ordered, number, body = m
198 b = Block(:list, "", number)
199 b.ordered = ordered
200 while true
201 content = [body]
202 at += 1
203 while at <= length(lines) && startswith(lines[at], " ")
204 push!(content, part(lines[at], 3))
205 at += 1
206 end
207 push!(b.items, parseblocks!(p, content))
208 following = at
209 while following<=length(lines) && isempty(lines[following])
210 following += 1
211 end
212 next = following<=length(lines) ? marker(lines[following]) : nothing
213 (next===nothing || next[1]!=kind || next[2]!=ordered) && break
214 at = following
215 body = next[4]
216 end
217 push!(blocks, b)
218 continue
219 end
220 if definition!(p, line)
221 at += 1
222 continue
223 end
224 if at<length(lines) && occursin('|', line) && occursin('|', lines[at+1])
225 header = cells(line)
226 delimiters = cells(lines[at+1])
227 if length(header)==length(delimiters) &&
228 all(c->occursin(r"^:?-+:?$", c), delimiters)
229 b = Block(:table)
230 b.lines = header
231 b.alignments = [
232 startswith(c, ":") ? (endswith(c, ":") ? "center" : "left") :
233 (endswith(c, ":") ? "right" : "") for c in delimiters
234 ]
235 at += 2
236 while at<=length(lines) && occursin('|', lines[at])
237 row = cells(lines[at])
238 resize!(row, min(length(row), length(header)))
239 while length(row)<length(header)
240 push!(row, "")
241 end
242 push!(b.rows, row)
243 at += 1
244 end
245 push!(blocks, b)
246 continue
247 end
248 end
249 end
250 finish = at+1
251 level = 0
252 while finish<=length(lines) && !isempty(lines[finish])
253 if occursin(r"^(?:={3,}|-{3,})$", lines[finish])
254 level = first(lines[finish])=='=' ? 1 : 2
255 break
256 end
257 interrupts(lines[finish]) && break
258 finish += 1
259 end
260 text = join(ltrim.(lines[at:(finish-1)]), "\n")
261 push!(blocks, Block(level>0 ? :heading : text=="[TOC]" ? :toc : :paragraph, text, level))
262 at = finish+(level>0)
263 end
264 blocks
265end

269 count::Int

271 closes::Bool
272end
273Piece(html::String) = Piece(html, ' ', 0, false, false)
274function emphasis(pieces, first = 1, last = length(pieces))
275 out = IOBuffer()
276 at = first
277 while at<=last
278 p = pieces[at]
279 close = 0
280 if p.opens
281 for i in (at+1):last
282 q = pieces[i]
283 if q.closes && q.char==p.char && q.count==p.count
284 close=i
285 break
286 end
287 end
288 end
289 if close>0
290 open, shut =
291 p.char=='~' ? ("<del>", "</del>") :
292 p.count==1 ? ("<em>", "</em>") :
293 p.count==2 ? ("<strong>", "</strong>") : ("<em><strong>", "</strong></em>")
294 print(out, open, emphasis(pieces, at+1, close-1), shut)
295 at=close+1
296 else
297 print(out, p.count==0 ? p.html : repeat(string(p.char), p.count))
298 at+=1
299 end
300 end
301 String(take!(out))
302end
303runend(s, at) = begin
304 finish=at+1
305 while finish<=length(s) && s[finish]==s[at]
306 finish+=1
307 end
308 finish
309end
310function tickclose(s, at, count)
311 while at<=length(s)
312 if s[at]=='`'
313 finish=runend(s, at)
314 finish-at==count && return at
315 at=finish
316 else
317 at+=1
318 end
319 end
320 0
321end
322function closingbracket(s, at)
323 depth=0
324 while at<=length(s)
325 if s[at]=='\\'
326 at+=2
327 continue
328 elseif s[at]=='`'
329 finish=runend(s, at)
330 close=tickclose(s, finish, finish-at)
331 at=close==0 ? finish : close+finish-at
332 continue
333 elseif s[at]=='['
334 depth+=1
335 elseif s[at]==']'
336 depth-=1
337 depth==0 && return at
338 end
339 at+=1
340 end
341 0
342end

353function destination(r, s, at, label)
354 if starts(s, "[", at)
355 close=findnext(']', s, at+1)
356 if close!==nothing && !occursin('[', part(s, at+1, close-1))
357 raw=part(s, at+1, close-1)
358 name=normalize(isempty(raw) ? label : raw)
359 return haskey(r.references, name) ? (r.references[name], close+1) : nothing
360 end
361 elseif starts(s, "(", at)
362 url, finish=escapedread(s, skipspace(s, at+1), " \n)")
363 next=skipspace(s, finish)
364 title=nothing
365 valid=true
366 if next>finish && next<=length(s) && s[next] in "\"'"
367 title, finish=escapedread(s, next+1, string(s[next]))
368 valid=finish<=length(s)
369 next=skipspace(s, finish+1)
370 end
371 valid && starts(s, ")", next) && return (Reference(url, title), next+1)
372 end
373 name=normalize(label)
374 haskey(r.references, name) ? (r.references[name], at) : nothing
375end
376refid(n, count) = count==1 ? "fnref-$n" : "fnref-$n-$count"
377function noteref!(r, name)
378 n=findfirst(==(name), r.notes)
379 if n===nothing
380 push!(r.notes, name)
381 push!(r.counts, 0)
382 n=length(r.notes)
383 end
384 r.counts[n]+=1
385 id=refid(n, r.counts[n])
386 "<sup id=\"$id\"><a href=\"#fn-$n\">$n</a></sup>"
387end
388function inline!(r, s; links = true, notes = true)
389 pieces=Piece[]
390 at=1
391 while at<=length(s)
392 ch=s[at]
393 if ch=='\\'
394 if at<length(s) && s[at+1] in PUNCT
395 push!(pieces, Piece(escapehtml(string(s[at+1]))))
396 at+=2
397 elseif at<length(s) && s[at+1]=='\n'
398 push!(pieces, Piece("<br />\n"))
399 at+=2
400 else
401 push!(pieces, Piece("\\"))
402 at+=1
403 end
404 continue
405 elseif ch=='`'
406 finish=runend(s, at)
407 count=finish-at
408 close=tickclose(s, finish, count)
409 if close==0
410 push!(pieces, Piece(repeat("`", count)))
411 at=finish
412 else
413 code=replace(part(s, finish, close-1), "\n"=>" ")
414 if startswith(code, " ") && endswith(code, " ") && !isempty(trim(code))
415 code=part(code, 2, length(code)-1)
416 end
417 push!(pieces, Piece("<code>$(escapehtml(code))</code>"))
418 at=close+count
419 end
420 continue
421 end
422 if links
423 if ch=='<'
424 close=findnext('>', s, at+1)
425 if close!==nothing
426 target=part(s, at+1, close-1)
427 url=occursin(r"^https?://[^ \n<]+\z", target)
428 email=occursin(r"^[A-Za-z0-9.+_-]+@[A-Za-z0-9-]+(?:\.[A-Za-z0-9-]+)+\z", target)
429 if url || email
430 href=escapehtml((email && !url ? "mailto:" : "")*target)
431 push!(pieces, Piece("<a href=\"$href\">$(escapehtml(target))</a>"))
432 at=close+1
433 continue
434 end
435 end
436 end
437 if (at==1 || s[at-1] in " \n*_~(") &&
438 (starts(s, "http://", at) || starts(s, "https://", at))
439 finish=at
440 while finish<=length(s) && !(s[finish] in " \n<")
441 finish+=1
442 end
443 while finish>at && s[finish-1] in ".,:;!?*_~)"
444 finish-=1
445 end
446 url=part(s, at, finish-1)
447 if !(url in ("http://", "https://"))
448 escaped=escapehtml(url)
449 push!(pieces, Piece("<a href=\"$escaped\">$escaped</a>"))
450 at=finish
451 continue
452 end
453 end
454 end
455 if ch=='[' || (ch=='!' && starts(s, "[", at+1))
456 if links && notes && starts(s, "[^", at)
457 close=findnext(']', s, at+2)
458 if close!==nothing
459 name=lowercase(part(s, at+2, close-1))
460 if haskey(r.definitions, name)
461 push!(pieces, Piece(noteref!(r, name)))
462 at=close+1
463 continue
464 end
465 end
466 end
467 image=ch=='!'
468 if image || links
469 opening=at+image
470 close=closingbracket(s, opening)
471 if close>0
472 label=part(s, opening+1, close-1)
473 dest=destination(r, s, close+1, label)
474 if dest!==nothing
475 reference, finish=dest
476 content=inline!(r, label; links = false, notes = false)
477 title=reference.title===nothing ? "" :
478 " title=\"$(escapehtml(reference.title))\""
479 url=escapehtml(reference.url)
480 html=image ? "<img src=\"$url\" alt=\"$(striptags(content))\"$title />" :
481 "<a href=\"$url\"$title>$content</a>"
482 push!(pieces, Piece(html))
483 at=finish
484 continue
485 end
486 end
487 end
488 end
489 if ch in "*_~"
490 finish=runend(s, at)
491 count=finish-at
492 before=at==1 ? ' ' : s[at-1]
493 after=finish>length(s) ? ' ' : s[finish]
494 opens=!(after in WHITE)
495 closes=!(before in WHITE)
496 if ch=='_' && isletterordigit(before) && isletterordigit(after)
497 opens=closes=false
498 end
499 if (ch!='~' && count<=3) || (ch=='~' && count==2)
500 push!(pieces, Piece("", ch, count, opens, closes))
501 else
502 push!(pieces, Piece(repeat(string(ch), count)))
503 end
504 at=finish
505 continue
506 end
507 push!(pieces, Piece(escapehtml(string(ch))))
508 at+=1
509 end
510 emphasis(pieces)
511end
512isletterordigit(c) = ('a'<=c<='z') || ('A'<=c<='Z') || ('0'<=c<='9')
513function renderheading!(r, b)
514 text=b.text
515 m=match(r"^(.*) \{#([A-Za-z0-9_-]+)\}$"s, text)
516 if m!==nothing
517 text=String(rstrip(m.captures[1], ' '))
518 id=String(m.captures[2])
519 else
520 base=replace(replace(lowercase(text), r"[^a-z0-9_ \n-]"=>""), r"[ \n]"=>"-")
521 isempty(base) && (base="heading")
522 id=base
523 suffix=1
524 while id in r.ids
525 id="$base-$suffix"
526 suffix+=1
527 end
528 end
529 push!(r.ids, id)
530 html=inline!(r, text)
531 push!(r.headings, (b.number, id, striptags(html)))
532 "<h$(b.number) id=\"$id\">$html</h$(b.number)>"
533end
534renderblocks!(r, blocks) = join((renderblock!(r, b)*"\n" for b in blocks))
535function renderitem!(r, blocks)
536 html="<li>"
537 for (i, b) in enumerate(blocks)
538 if b.kind==:paragraph
539 text=b.text
540 if i==1 && length(text)>4 && part(text, 1, 4) in ("[ ] ", "[x] ", "[X] ")
541 checked=text[2]==' ' ? "" : " checked=\"\""
542 html*="<input type=\"checkbox\"$checked disabled=\"\" /> "
543 text=part(text, 5)
544 end
545 html*=inline!(r, text)
546 else
547 !endswith(html, "\n") && (html*="\n")
548 html*=renderblock!(r, b)*"\n"
549 end
550 end
551 html*"</li>"
552end
553function renderblock!(r, b)
554 if b.kind==:heading
555 return renderheading!(r, b)
556 elseif b.kind==:paragraph
557 return "<p>$(inline!(r,b.text))</p>"
558 elseif b.kind==:rule
559 return "<hr />"
560 elseif b.kind==:toc
561 return TOC
562 elseif b.kind==:code
563 attr=isempty(b.text) ? "" : " class=\"language-$(escapehtml(b.text))\""
564 return "<pre><code$attr>"*join(escapehtml(s)*"\n" for s in b.lines)*"</code></pre>"
565 elseif b.kind==:quote
566 return "<blockquote>\n"*renderblocks!(r, b.children)*"</blockquote>"
567 elseif b.kind==:alert
568 title=uppercasefirst(lowercase(b.text))
569 return "<div class=\"alert alert-$(lowercase(b.text))\">\n<p class=\"alert-title\">$title</p>\n"*renderblocks!(
570 r,
571 b.children,
572 )*"</div>"
573 elseif b.kind==:list
574 tag=b.ordered ? "ol" : "ul"
575 attr=b.ordered && b.number!=1 ? " start=\"$(b.number)\"" : ""
576 return "<$tag$attr>\n"*join(renderitem!(r, item)*"\n" for item in b.items)*"</$tag>"
577 end
578 lines=["<table>", "<thead>", "<tr>"]
579 function row!(tag, row)
580 for (align, cell) in zip(b.alignments, row)
581 style=isempty(align) ? "" : " style=\"text-align:$align\""
582 push!(lines, "<$tag$style>$(inline!(r,cell))</$tag>")
583 end
584 end
585 row!("th", b.lines)
586 append!(lines, ["</tr>", "</thead>"])
587 if !isempty(b.rows)
588 push!(lines, "<tbody>")
589 for row in b.rows
590 push!(lines, "<tr>")
591 row!("td", row)
592 push!(lines, "</tr>")
593 end
594 push!(lines, "</tbody>")
595 end
596 push!(lines, "</table>")
597 join(lines, "\n")
598end
599function toc(r)
600 parents=Int[]
601 stack=Int[]
602 for (i, (level, id, label)) in enumerate(r.headings)
603 while !isempty(stack) && r.headings[last(stack)][1]>=level
604 pop!(stack)
605 end
606 push!(parents, isempty(stack) ? 0 : last(stack))
607 push!(stack, i)
608 end
609 function listing(parent)
610 html="<ul>\n"
611 for i in findall(==(parent), parents)
612 _, id, label=r.headings[i]
613 html*="<li><a href=\"#$id\">$label</a>"
614 if i in parents
615 html*="\n"*listing(i)*"\n"
616 end
617 html*="</li>\n"
618 end
619 html*"</ul>"
620 end
621 "<nav class=\"toc\">\n"*(isempty(parents) ? "" : listing(0)*"\n")*"</nav>"
622end
623function footnotes!(r)
624 isempty(r.notes) && return ""
625 html="<section class=\"footnotes\">\n<ol>\n"
626 for (n, name) in enumerate(r.notes)
627 body=inline!(r, r.definitions[name]; notes = false)
628 links=join(("<a href=\"#$(refid(n,j))\">&#8617;</a>" for j in 1:r.counts[n]), " ")
629 html*="<li id=\"fn-$n\">$body $links</li>\n"
630 end
631 html*"</ol>\n</section>\n"
632end
633function main()
634 lines=String[]
635 for line in eachline(stdin)
636 line=="@@END@@" && break
637 push!(lines, line)
638 end
639 parser=Parser()
640 blocks=parseblocks!(parser, lines)
641 renderer=Renderer(parser)

643 print(replace(html, TOC=>toc(renderer)), footnotes!(renderer))
644end
