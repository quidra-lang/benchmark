1private const val TOC = "\u0001toc\u0002"
2private fun white(c: Char) = c == ' ' || c == '\n'
3private fun alnum(c: Char) = c in 'a'..'z' || c in 'A'..'Z' || c in '0'..'9'
4private fun label(c: Char) = alnum(c) || c == '-' || c == '_'
5private fun punctuation(c: Char) = c in '!'..'/' || c in ':'..'@' || c in '['..'`' || c in '{'..'~'
6private fun escape(text: String): String = buildString {
7 for (c in text) append(when (c) {
8 '&' -> "&amp;"
9 '<' -> "&lt;"
10 '>' -> "&gt;"
11 '"' -> "&quot;"
12 else -> c.toString()
13 })
14}
15private fun stripTags(text: String): String = buildString {
16 var at = 0
17 while (at < text.length) {
18 if (text[at] == '<') {
19 val end = text.indexOf('>', at)
20 if (end >= 0) { at = end + 1; continue }
21 }
22 append(text[at++])
23 }
24}
25private fun normalize(text: String) = text.lowercase().split(' ', '\n').filter { it.isNotEmpty() }.joinToString(" ")
26private fun skip(text: String, position: Int): Int {
27 var at = position
28 while (at < text.length && white(text[at])) at++
29 return at
30}
31private fun escapedRead(text: String, position: Int, stops: String): Pair<String, Int> {
32 var at = position
33 val result = StringBuilder()
34 while (at < text.length && text[at] !in stops) {
35 if (text[at] == '\\' && at + 1 < text.length && punctuation(text[at + 1])) at++
36 result.append(text[at++])
37 }
38 return result.toString() to at
39}
40private fun runEnd(text: String, position: Int): Int {
41 var at = position
42 val c = text[at]
43 while (at < text.length && text[at] == c) at++
44 return at
45}
46private fun tickCloser(text: String, position: Int, length: Int): Int? {
47 var at = position
48 while (at < text.length) {
49 at = text.indexOf('`', at)
50 if (at < 0) return null
51 val end = runEnd(text, at)
52 if (end - at == length) return at
53 at = end
54 }
55 return null
56}
57private fun closingBracket(text: String, position: Int): Int? {
58 var at = position
59 var depth = 0
60 while (at < text.length) {
61 when (text[at]) {
62 '\\' -> { at += 2; continue }
63 '`' -> {
64 val end = runEnd(text, at)
65 at = tickCloser(text, end, end - at)?.plus(end - at) ?: end
66 continue
67 }
68 '[' -> depth++
69 ']' -> { depth--; if (depth == 0) return at }
70 }
71 at++
72 }
73 return null
74}

88private data class Fence(val char: Char, val length: Int, val language: String)
89private fun fence(text: String): Fence? {
90 if (text.isEmpty() || text[0] !in "`~") return null
91 val n = runEnd(text, 0)
92 val info = text.substring(n).trim(' ')
93 if (n < 3 || (text[0] == '`' && '`' in info)) return null
94 return Fence(text[0], n, info.substringBefore(' '))
95}
96private fun atx(line: String): Heading? {
97 val n = line.takeWhile { it == '#' }.length
98 if (n !in 1..6 || (n < line.length && line[n] != ' ')) return null
99 var text = line.substring(n).trim(' ')
100 val unmarked = text.trimEnd('#')
101 if (unmarked.isEmpty() || unmarked.endsWith(' ')) text = unmarked.trim(' ')
102 return Heading(n, text)
103}
104private fun rule(text: String): Boolean {
105 val compact = text.filter { it != ' ' }
106 return compact.length >= 3 && compact[0] in "-*_" && compact.all { it == compact[0] }
107}
108private fun marker(text: String): Marker? {
109 if (text.isEmpty() || rule(text)) return null
110 val kind: Char
111 val number: Int?
112 var at: Int
113 if (text[0] in "-*+") {
114 kind = text[0]
115 number = null
116 at = 1
117 } else {
118 val n = text.takeWhile { it in '0'..'9' }.length
119 if (n !in 1..9 || n == text.length || text[n] !in ".)") return null
120 kind = text[n]
121 number = text.substring(0, n).toInt()
122 at = n + 1
123 }
124 if (at == text.length || text[at] != ' ') return null
125 while (at < text.length && text[at] == ' ') at++
126 return Marker(kind, number, text.substring(at))
127}
128private fun interrupt(text: String) = !text.startsWith(' ') &&
129 (fence(text) != null || atx(text) != null || rule(text) || text.startsWith('>') || marker(text) != null)
130private fun cells(line: String): List<String> {
131 val text = line.trim(' ').removePrefix("|")
132 val result = mutableListOf(StringBuilder())
133 var at = 0
134 var separator = false
135 while (at < text.length) {
136 val c = text[at++]
137 separator = false
138 if (c == '\\' && at < text.length) {
139 if (text[at] != '|') result.last().append('\\')
140 result.last().append(text[at++])
141 } else if (c == '|') {
142 result.add(StringBuilder())
143 separator = true
144 } else result.last().append(c)
145 }
146 if (separator) result.removeAt(result.lastIndex)
147 return result.map { it.toString().trim(' ') }
148}
149private fun alignment(text: String): String? {
150 val core = text.removePrefix(":").removeSuffix(":")
151 if (core.isEmpty() || core.any { it != '-' }) return null
152 return when {
153 text.startsWith(':') && text.endsWith(':') -> "center"
154 text.startsWith(':') -> "left"
155 text.endsWith(':') -> "right"
156 else -> ""
157 }
158}

160 val references = mutableMapOf<String, Reference>()
161 val footnotes = mutableMapOf<String, String>()

163 private fun definition(text: String): Boolean {
164 if (text.startsWith("[^")) {
165 val end = text.indexOf("]:", 2)
166 if (end >= 0) {
167 val name = text.substring(2, end)
168 val body = text.substring(end + 2).trim(' ')
169 if (name.isNotEmpty() && name.all(::label) && body.isNotEmpty()) {
170 footnotes.putIfAbsent(name.lowercase(), body)
171 return true
172 }
173 }
174 }
175 if (!text.startsWith('[')) return false
176 val close = text.indexOf(']')
177 if (close < 0 || !text.startsWith(":", close + 1)) return false
178 val name = text.substring(1, close)
179 if ('[' in name || name.trim(' ').isEmpty() || name.startsWith('^')) return false
180 val (url, end) = escapedRead(text, skip(text, close + 2), " ")
181 if (url.isEmpty()) return false
182 val at = skip(text, end)
183 var title: String? = null
184 if (at < text.length) {
185 if (at == end || text[at] !in "\"'") return false
186 val (value, stop) = escapedRead(text, at + 1, text[at].toString())
187 if (stop != text.length - 1) return false
188 title = value
189 }
190 references.putIfAbsent(normalize(name), Reference(url, title))
191 return true
192 }

194 fun parse(lines: List<String>): List<Block> {
195 val blocks = mutableListOf<Block>()
196 var at = 0
197 while (at < lines.size) {
198 val line = lines[at]
199 if (line.isEmpty()) { at++; continue }
200 if (line.startsWith(" ")) {
201 val start = at
202 while (at < lines.size && (lines[at].isEmpty() || lines[at].startsWith(" "))) at++
203 while (lines[at - 1].isEmpty()) at--
204 blocks.add(Code("", lines.subList(start, at).map { if (it.isEmpty()) "" else it.substring(4) }))
205 continue
206 }
207 if (!line.startsWith(' ')) {
208 val fence = fence(line)
209 if (fence != null) {
210 at++
211 val code = mutableListOf<String>()
212 while (at < lines.size) {
213 val row = lines[at++]
214 if (row.length >= fence.length && row.all { it == fence.char }) break
215 code.add(row)
216 }
217 blocks.add(Code(fence.language, code))
218 continue
219 }
220 val heading = atx(line)
221 if (heading != null) { blocks.add(heading); at++; continue }
222 if (rule(line)) { blocks.add(Rule); at++; continue }
223 if (line.startsWith('>')) {
224 val content = mutableListOf<String>()
225 while (at < lines.size && lines[at].startsWith('>')) content.add(lines[at++].substring(1).removePrefix(" "))
226 val kind = content[0].removePrefix("[!").removeSuffix("]")
227 blocks.add(if (content[0] == "[!$kind]" && kind in listOf("NOTE", "TIP", "IMPORTANT", "WARNING", "CAUTION"))
228 Alert(kind, parse(content.drop(1))) else Quote(parse(content)))
229 continue
230 }
231 val first = marker(line)
232 if (first != null) {
233 var item = first
234 val items = mutableListOf<List<Block>>()
235 while (true) {
236 val content = mutableListOf(item.content)
237 at++
238 while (at < lines.size && lines[at].startsWith(" ")) content.add(lines[at++].substring(2))
239 items.add(parse(content))
240 var next = at
241 while (next < lines.size && lines[next].isEmpty()) next++
242 val following = lines.getOrNull(next)?.let(::marker)
243 if (following == null || following.kind != first.kind || (following.number != null) != (first.number != null)) break
244 item = following
245 at = next
246 }
247 blocks.add(ListBlock(first.number, items))
248 continue
249 }
250 if (definition(line)) { at++; continue }
251 if (at + 1 < lines.size && '|' in line && '|' in lines[at + 1]) {
252 val header = cells(line)
253 val delimiters = cells(lines[at + 1])
254 val align = delimiters.map(::alignment)
255 if (header.size == delimiters.size && align.all { it != null }) {
256 at += 2
257 val rows = mutableListOf<List<String>>()
258 while (at < lines.size && '|' in lines[at]) {
259 val row = cells(lines[at++]).take(header.size).toMutableList()
260 while (row.size < header.size) row.add("")
261 rows.add(row)
262 }
263 blocks.add(Table(align.map { it!! }, header, rows))
264 continue
265 }
266 }
267 }
268 val text = mutableListOf(line.trimStart(' '))
269 at++
270 var level = 0
271 while (at < lines.size && lines[at].isNotEmpty()) {
272 val row = lines[at]
273 if (row.length >= 3 && row[0] in "=-" && row.all { it == row[0] }) {
274 level = if (row[0] == '=') 1 else 2
275 at++
276 break
277 }
278 if (interrupt(row)) break
279 text.add(row.trimStart(' '))
280 at++
281 }
282 val combined = text.joinToString("\n")
283 blocks.add(if (level > 0) Heading(level, combined) else if (combined == "[TOC]") Contents else Paragraph(combined))
284 }
285 return blocks
286 }
287}

289private data class Piece(val html: String = "", val delimiter: Char = '\u0000', val length: Int = 0, val open: Boolean = false, val close: Boolean = false)
290private fun emphasis(pieces: List<Piece>, start: Int = 0, end: Int = pieces.size): String = buildString {
291 var at = start
292 while (at < end) {
293 val piece = pieces[at]
294 if (piece.delimiter == '\u0000') { append(piece.html); at++; continue }
295 val close = if (piece.open) (at + 1 until end).firstOrNull {
296 pieces[it].delimiter == piece.delimiter && pieces[it].length == piece.length && pieces[it].close
297 } else null
298 if (close == null) {
299 append(piece.delimiter.toString().repeat(piece.length))
300 at++
301 } else {
302 val tags = when {
303 piece.delimiter == '~' -> "<del>" to "</del>"
304 piece.length == 1 -> "<em>" to "</em>"
305 piece.length == 2 -> "<strong>" to "</strong>"
306 else -> "<em><strong>" to "</strong></em>"
307 }
308 append(tags.first).append(emphasis(pieces, at + 1, close)).append(tags.second)
309 at = close + 1
310 }
311 }
312}
313private fun email(text: String): Boolean {
314 val at = text.indexOf('@')
315 if (at <= 0 || text.indexOf('@', at + 1) >= 0) return false
316 val local = text.substring(0, at)
317 val parts = text.substring(at + 1).split('.')
318 return local.all { alnum(it) || it in ".+-_" } && parts.size >= 2 &&
319 parts.all { part -> part.isNotEmpty() && part.all { alnum(it) || it == '-' } }
320}

324 private val headings = mutableListOf<HeadingEntry>()

329 private fun destination(text: String, at: Int, content: String): Pair<Reference, Int>? {
330 if (text.startsWith("[", at)) {
331 val end = text.indexOf(']', at + 1)
332 if (end >= 0 && '[' !in text.substring(at + 1, end)) {
333 val name = text.substring(at + 1, end).ifEmpty { content }
334 return references[normalize(name)]?.let { it to end + 1 }
335 }
336 } else if (text.startsWith("(", at)) {
337 val (url, end) = escapedRead(text, skip(text, at + 1), " \n)")
338 var next = skip(text, end)
339 var title: String? = null
340 var valid = true
341 if (next > end && next < text.length && text[next] in "\"'") {
342 val (value, stop) = escapedRead(text, next + 1, text[next].toString())
343 if (stop == text.length) valid = false else {
344 title = value
345 next = skip(text, stop + 1)
346 }
347 }
348 if (valid && text.startsWith(")", next)) return Reference(url, title) to next + 1
349 }
350 return references[normalize(content)]?.let { it to at }
351 }

353 private fun inline(text: String, links: Boolean = true, notes: Boolean = true): String {
354 val pieces = mutableListOf<Piece>()
355 var at = 0
356 while (at < text.length) {
357 val c = text[at]
358 if (c == '\\') {
359 when {
360 at + 1 < text.length && punctuation(text[at + 1]) -> { pieces.add(Piece(escape(text.substring(at + 1, at + 2)))); at += 2 }
361 at + 1 < text.length && text[at + 1] == '\n' -> { pieces.add(Piece("<br />\n")); at += 2 }
362 else -> { pieces.add(Piece("\\")); at++ }
363 }
364 continue
365 }
366 if (c == '`') {
367 val end = runEnd(text, at)
368 val close = tickCloser(text, end, end - at)
369 if (close == null) { pieces.add(Piece(text.substring(at, end))); at = end }
370 else {
371 var code = text.substring(end, close).replace('\n', ' ')
372 if (code.startsWith(' ') && code.endsWith(' ') && code.any { it != ' ' }) code = code.substring(1, code.length - 1)
373 pieces.add(Piece("<code>${escape(code)}</code>"))
374 at = close + end - at
375 }
376 continue
377 }
378 if (links && c == '<') {
379 val close = text.indexOf('>', at + 1)
380 if (close >= 0) {
381 val target = text.substring(at + 1, close)
382 val clean = target.none { white(it) || it == '<' }
383 val href = when {
384 clean && ((target.startsWith("http://") && target.length > 7) || (target.startsWith("https://") && target.length > 8)) -> target
385 email(target) -> "mailto:$target"
386 else -> null
387 }
388 if (href != null) {
389 pieces.add(Piece("<a href=\"${escape(href)}\">${escape(target)}</a>"))
390 at = close + 1
391 continue
392 }
393 }
394 }
395 if (links && (text.startsWith("http://", at) || text.startsWith("https://", at)) && (at == 0 || text[at - 1] in " \n*_~(")) {
396 var end = at
397 while (end < text.length && text[end] !in " \n<") end++
398 while (end > at && text[end - 1] in ".,:;!?*_~)") end--
399 val url = text.substring(at, end)
400 if (url != "http://" && url != "https://") {
401 pieces.add(Piece("<a href=\"${escape(url)}\">${escape(url)}</a>"))
402 at = end
403 continue
404 }
405 }
406 if (c == '[' || (c == '!' && text.getOrNull(at + 1) == '[')) {
407 if (links && notes && text.startsWith("[^", at)) {
408 val close = text.indexOf(']', at + 2)
409 if (close >= 0) {
410 val name = text.substring(at + 2, close).lowercase()
411 if (name in footnotes) { pieces.add(Piece(noteReference(name))); at = close + 1; continue }
412 }
413 }
414 val image = c == '!'
415 val opening = at + if (image) 1 else 0
416 if (image || links) {
417 val close = closingBracket(text, opening)
418 if (close != null) {
419 val content = text.substring(opening + 1, close)
420 val dest = destination(text, close + 1, content)
421 if (dest != null) {
422 val (reference, end) = dest
423 val body = inline(content, false, false)
424 val title = reference.title?.let { " title=\"${escape(it)}\"" } ?: ""
425 val url = escape(reference.url)
426 val html = if (image) "<img src=\"$url\" alt=\"${stripTags(body)}\"$title />" else "<a href=\"$url\"$title>$body</a>"
427 pieces.add(Piece(html))
428 at = end
429 continue
430 }
431 }
432 }
433 }
434 if (c in "*_~") {
435 val end = runEnd(text, at)
436 val n = end - at
437 val before = text.getOrNull(at - 1) ?: ' '
438 val after = text.getOrNull(end) ?: ' '
439 var open = !white(after)
440 var close = !white(before)
441 if (c == '_' && alnum(before) && alnum(after)) { open = false; close = false }
442 pieces.add(if ((c != '~' && n <= 3) || (c == '~' && n == 2)) Piece(delimiter = c, length = n, open = open, close = close) else Piece(text.substring(at, end)))
443 at = end
444 continue
445 }
446 pieces.add(Piece(escape(c.toString())))
447 at++
448 }
449 return emphasis(pieces)
450 }

452 private fun blocks(blocks: List<Block>): String = buildString {
453 for (block in blocks) append(render(block)).append('\n')
454 }
455 private fun render(block: Block): String = when (block) {
456 is Heading -> heading(block.level, block.text)
457 is Paragraph -> "<p>${inline(block.text)}</p>"
458 Rule -> "<hr />"
459 Contents -> TOC
460 is Code -> {
461 val attribute = if (block.language.isEmpty()) "" else " class=\"language-${escape(block.language)}\""
462 "<pre><code$attribute>" + block.lines.joinToString("") { escape(it) + "\n" } + "</code></pre>"
463 }
464 is Quote -> "<blockquote>\n${blocks(block.blocks)}</blockquote>"
465 is Alert -> {
466 val name = block.kind.lowercase()
467 val title = name.replaceFirstChar { it.uppercase() }
468 "<div class=\"alert alert-$name\">\n<p class=\"alert-title\">$title</p>\n${blocks(block.blocks)}</div>"
469 }
470 is ListBlock -> {
471 val tag = if (block.start == null) "ul" else "ol"
472 val attribute = if (block.start != null && block.start != 1) " start=\"${block.start}\"" else ""
473 "<$tag$attribute>\n" + block.items.joinToString("") { item(it) + "\n" } + "</$tag>"
474 }
475 is Table -> buildString {
476 append("<table>\n<thead>\n<tr>\n").append(tableCells("th", block.align, block.header)).append("</tr>\n</thead>\n")
477 if (block.rows.isNotEmpty()) {
478 append("<tbody>\n")
479 for (row in block.rows) append("<tr>\n").append(tableCells("td", block.align, row)).append("</tr>\n")
480 append("</tbody>\n")
481 }
482 append("</table>")
483 }
484 }
485 private fun item(blocks: List<Block>): String = buildString {
486 append("<li>")
487 for ((index, block) in blocks.withIndex()) {
488 if (block is Paragraph) {
489 val task = index == 0 && block.text.length > 4 && listOf("[ ] ", "[x] ", "[X] ").any { block.text.startsWith(it) }
490 if (task) append(if (block.text[1] == ' ') "<input type=\"checkbox\" disabled=\"\" /> " else "<input type=\"checkbox\" checked=\"\" disabled=\"\" /> ")
491 append(inline(if (task) block.text.substring(4) else block.text))
492 } else {
493 if (last() != '\n') append('\n')
494 append(render(block)).append('\n')
495 }
496 }
497 append("</li>")
498 }
499 private fun tableCells(tag: String, align: List<String>, cells: List<String>): String = buildString {
500 for ((i, cell) in cells.withIndex()) {
501 val style = if (align[i].isEmpty()) "" else " style=\"text-align:${align[i]}\""
502 append("<$tag$style>${inline(cell)}</$tag>\n")
503 }
504 }
505 private fun heading(level: Int, source: String): String {
506 var text = source
507 var id: String? = null
508 val at = source.lastIndexOf(" {#")
509 if (at >= 0 && source.endsWith('}')) {
510 val candidate = source.substring(at + 3, source.length - 1)
511 if (candidate.isNotEmpty() && candidate.all(::label)) {
512 id = candidate
513 text = source.substring(0, at).trimEnd(' ')
514 }
515 }
516 if (id == null) {
517 val base = buildString {
518 for (c in source.lowercase()) {
519 if (label(c)) append(c) else if (white(c)) append('-')
520 }
521 }.ifEmpty { "heading" }
522 id = base
523 var suffix = 1
524 while (id in usedIds) id = "$base-${suffix++}"
525 }
526 usedIds.add(id)
527 val html = inline(text)
528 headings.add(HeadingEntry(level, id, stripTags(html)))
529 return "<h$level id=\"$id\">$html</h$level>"
530 }
531 private fun tocList(children: List<List<Int>>, entries: List<Int>): String = buildString {
532 append("<ul>\n")
533 for (i in entries) {
534 val heading = headings[i]
535 append("<li><a href=\"#${heading.id}\">${heading.text}</a>")
536 if (children[i].isNotEmpty()) append('\n').append(tocList(children, children[i])).append('\n')
537 append("</li>\n")
538 }
539 append("</ul>")
540 }
541 private fun toc(): String {
542 if (headings.isEmpty()) return "<nav class=\"toc\">\n</nav>"
543 val roots = mutableListOf<Int>()
544 val parents = mutableListOf<Int>()
545 val children = List(headings.size) { mutableListOf<Int>() }
546 for ((i, heading) in headings.withIndex()) {
547 while (parents.isNotEmpty() && headings[parents.last()].level >= heading.level) parents.removeAt(parents.lastIndex)
548 if (parents.isEmpty()) roots.add(i) else children[parents.last()].add(i)
549 parents.add(i)
550 }
551 return "<nav class=\"toc\">\n${tocList(children, roots)}\n</nav>"
552 }
553 private fun referenceId(index: Int, count: Int) = "fnref-${index + 1}" + if (count == 1) "" else "-$count"
554 private fun noteReference(name: String): String {
555 val index = noteNumbers.getOrPut(name) {
556 noteNames.add(name)
557 noteCounts.add(0)
558 noteNames.lastIndex
559 }
560 noteCounts[index]++
561 return "<sup id=\"${referenceId(index, noteCounts[index])}\"><a href=\"#fn-${index + 1}\">${index + 1}</a></sup>"
562 }
563 private fun noteList(): String = buildString {
564 append("<section class=\"footnotes\">\n<ol>\n")
565 for (i in noteNames.indices) {
566 val body = inline(footnotes.getValue(noteNames[i]), notes = false)
567 append("<li id=\"fn-${i + 1}\">$body")
568 for (count in 1..noteCounts[i]) append(" <a href=\"#${referenceId(i, count)}\">&#8617;</a>")
569 append("</li>\n")
570 }
571 append("</ol>\n</section>\n")
572 }
573 fun document(blocks: List<Block>): String {
574 val html = this.blocks(blocks).replace(TOC, toc())
575 return html + if (noteNames.isEmpty()) "" else noteList()
576 }
