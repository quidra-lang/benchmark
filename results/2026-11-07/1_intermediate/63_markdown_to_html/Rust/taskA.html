5fn trim(s: &str) -> &str {
6 s.trim_matches(' ')
7}
8
9fn white(c: u8) -> bool {
10 c == b' ' || c == b'\n'
11}
12
13fn label(c: u8) -> bool {
14 c.is_ascii_alphanumeric() || c == b'-' || c == b'_'
15}

17fn escape(s: &str) -> String {
18 let mut result = String::new();
19 for c in s.chars() {
20 match c {
21 '&' => result.push_str("&amp;"), '<' => result.push_str("&lt;"), '>' => result.push_str("&gt;"), '"' => result.push_str("&quot;"), _ => result.push(c),
22 }
23 }
24 result
25}
26
27fn strip_tags(s: &str) -> String {
28 let mut result = String::new();
29 let mut at = 0;
30 while at < s.len() {
31 if s.as_bytes()[at] == b'<' {
32 if let Some(end) = s[at..].find('>') {
33 at += end + 1;
34 continue;
35 }
36 }
37 result.push(s.as_bytes()[at] as char);
38 at += 1;
39 }
40 result
41}
42
43fn normalize(s: &str) -> String {
44 s.split([' ', '\n']).filter(|p| !p.is_empty()).collect::<Vec<_>>().join(" ").to_ascii_lowercase()
45}
46
47fn skip(s: &str, mut at: usize) -> usize {
48 while at < s.len() && white(s.as_bytes()[at]) {
49 at += 1;
50 }
51 at
52}
53
54fn escaped_read(s: &str, mut at: usize, stops: &[u8]) -> (String, usize) {
55 let b = s.as_bytes();
56 let mut result = String::new();
57 while at < b.len() && !stops.contains(&b[at]) {
58 if b[at] == b'\\' && at + 1 < b.len() && b[at + 1].is_ascii_punctuation() {
59 at += 1;
60 }
61 result.push(b[at] as char);
62 at += 1;
63 }
64 (result, at)
65}
66
67fn run_end(s: &str, mut at: usize) -> usize {
68 let c = s.as_bytes()[at];
69 while at < s.len() && s.as_bytes()[at] == c {
70 at += 1;
71 }
72 at
73}
74
75fn tick_closer(s: &str, mut at: usize, length: usize) -> Option<usize> {
76 while at < s.len() {
77 at += s[at..].find('`')?;
78 let end = run_end(s, at);
79 if end - at == length {
80 return Some(at);
81 }
82 at = end;
83 }
84 None
85}
86
87fn closing_bracket(s: &str, mut at: usize) -> Option<usize> {
88 let mut depth = 0;
89 while at < s.len() {
90 match s.as_bytes()[at] {
91 b'\\' => {
92 at += 2;
93 continue;
94 }
95 b'`' => {
96 let end = run_end(s, at);
97 at = tick_closer(s, end, end - at).map_or(end, |c| c + end - at);
98 continue;
99 }
100 b'[' => depth += 1,
101 b']' => {
102 depth -= 1;
103 if depth == 0 {
104 return Some(at);
105 }
106 }
107 _ => {
108 }
109 }
110 at += 1;
111 }
112 None
113}

133struct Marker {
134 kind: u8,
135 number: Option<usize>,
136 content: String,

138
139fn fence(s: &str) -> Option<(u8, usize, String)> {
140 let c = *s.as_bytes().first()?;
141 if c != b'`' && c != b'~' {
142 return None;
143 }
144 let n = run_end(s, 0);
145 let info = trim(&s[n..]);
146 if n < 3 || (c == b'`' && info.contains('`')) {
147 return None;
148 }
149 Some((c, n, info.split(' ').next().unwrap().into()))
150}
151
152fn atx(s: &str) -> Option<Block> {
153 let n = s.bytes().take_while(|&c| c == b'#').count();
154 if n == 0 || n > 6 || (n < s.len() && s.as_bytes()[n] != b' ') {
155 return None;
156 }
157 let mut text = trim(&s[n..]);
158 let unmarked = text.trim_end_matches('#');
159 if unmarked.is_empty() || unmarked.ends_with(' ') {
160 text = trim(unmarked);
161 }
162 Some(Block::Heading(n, text.into()))
163}
164
165fn rule(s: &str) -> bool {
166 let b: Vec<u8> = s.bytes().filter(|&c| c != b' ').collect();
167 b.len() >= 3 && b"-*_".contains(&b[0]) && b.iter().all(|&c| c == b[0])
168}
169
170fn marker(s: &str) -> Option<Marker> {
171 if rule(s) {
172 return None;
173 }
174 let b = s.as_bytes();
175 let first = *b.first()?;
176 let (kind, number, mut at) = if b"-*+".contains(&first) {
177 (first, None, 1)
178 } else {
179 let n = b.iter().take_while(|c| c.is_ascii_digit()).count();
180 if n == 0 || n > 9 || n == b.len() || !b".)".contains(&b[n]) {
181 return None;
182 }
183 (b[n], Some(s[..n].parse().unwrap()), n + 1)
184 };
185 if at >= b.len() || b[at] != b' ' {
186 return None;
187 }
188 while at < b.len() && b[at] == b' ' {
189 at += 1;
190 }
191 Some(Marker {
192 kind, number, content: s[at..].into()
193 })
194}
195
196fn interrupt(s: &str) -> bool {
197 !s.starts_with(' ') && (fence(s).is_some() || atx(s).is_some() || rule(s) || s.starts_with('>') || marker(s).is_some())
198}
199
200fn cells(s: &str) -> Vec<String> {
201 let text = trim(s).strip_prefix('|').unwrap_or(trim(s));
202 let b = text.as_bytes();
203 let mut result = vec![String::new()];
204 let mut at = 0;
205 let mut separator = false;
206 while at < b.len() {
207 let c = b[at];
208 at += 1;
209 separator = false;
210 if c == b'\\' && at < b.len() {
211 if b[at] != b'|' {
212 result.last_mut().unwrap().push('\\');
213 }
214 result.last_mut().unwrap().push(b[at] as char);
215 at += 1;
216 } else if c == b'|' {
217 result.push(String::new());
218 separator = true;
219 } else {
220 result.last_mut().unwrap().push(c as char);
221 }
222 }
223 if separator {
224 result.pop();
225 }
226 result.into_iter().map(|s| trim(&s).into()).collect()
227}
228
229fn alignment(s: &str) -> Option<String> {
230 let core = s.strip_prefix(':').unwrap_or(s);
231 let core = core.strip_suffix(':').unwrap_or(core);
232 if core.is_empty() || !core.bytes().all(|c| c == b'-') {
233 return None;
234 }
235 Some(match (s.starts_with(':'), s.ends_with(':')) {
236 (true, true) => "center", (true, false) => "left", (false, true) => "right", _ => ""
237 }.into())
238}

240#[derive(Default)]
241struct Parser {
242 references: HashMap<String, Reference>,
243 footnotes: HashMap<String, String>,
244}

246impl Parser {
247 fn definition(&mut self, s: &str) -> bool {
248 if let Some(rest) = s.strip_prefix("[^") {
249 if let Some(end) = rest.find("]:") {
250 let name = &rest[..end];
251 let body = trim(&rest[end + 2..]);
252 if !name.is_empty() && name.bytes().all(label) && !body.is_empty() {
253 self.footnotes.entry(name.to_ascii_lowercase()).or_insert(body.into());
254 return true;
255 }
256 }
257 }
258 if !s.starts_with('[') {
259 return false;
260 }
261 let Some(close) = s.find(']') else {
262 return false;
263 };
264 if !s[close + 1..].starts_with(':') {
265 return false;
266 }
267 let name = &s[1..close];
268 if name.contains('[') || trim(name).is_empty() || name.starts_with('^') {
269 return false;
270 }
271 let (url, end) = escaped_read(s, skip(s, close + 2), b" ");
272 if url.is_empty() {
273 return false;
274 }
275 let at = skip(s, end);
276 let mut title = None;
277 if at < s.len() {
278 if at == end || !b"\"'".contains(&s.as_bytes()[at]) {
279 return false;
280 }
281 let (value, end) = escaped_read(s, at + 1, &[s.as_bytes()[at]]);
282 if end != s.len() - 1 {
283 return false;
284 }
285 title = Some(value);
286 }
287 self.references.entry(normalize(name)).or_insert(Reference {
288 url, title
289 });
290 true
291 }

293 fn parse(&mut self, lines: &[String]) -> Vec<Block> {
294 let mut blocks = Vec::new();
295 let mut at = 0;
296 while at < lines.len() {
297 let line = &lines[at];
298 if line.is_empty() {
299 at += 1;
300 continue;
301 }
302 if line.starts_with(" ") {
303 let start = at;
304 while at < lines.len() && (lines[at].is_empty() || lines[at].starts_with(" ")) {
305 at += 1;
306 }
307 while lines[at - 1].is_empty() {
308 at -= 1;
309 }
310 blocks.push(Block::Code(String::new(), lines[start..at].iter().map(|s| if s.is_empty() {
311 String::new()
312 } else {
313 s[4..].into()
314 }).collect()));
315 continue;
316 }
317 if !line.starts_with(' ') {
318 if let Some((c, n, language)) = fence(line) {
319 at += 1;
320 let mut code = Vec::new();
321 while at < lines.len() {
322 let row = &lines[at];
323 at += 1;
324 if row.len() >= n && row.bytes().all(|v| v == c) {
325 break;
326 }
327 code.push(row.clone());
328 }
329 blocks.push(Block::Code(language, code));
330 continue;
331 }
332 if let Some(heading) = atx(line) {
333 blocks.push(heading);
334 at += 1;
335 continue;
336 }
337 if rule(line) {
338 blocks.push(Block::Rule);
339 at += 1;
340 continue;
341 }
342 if line.starts_with('>') {
343 let mut content = Vec::new();
344 while at < lines.len() && lines[at].starts_with('>') {
345 let row = &lines[at][1..];
346 content.push(row.strip_prefix(' ').unwrap_or(row).to_owned());
347 at += 1;
348 }
349 let kind = content[0].strip_prefix("[!").and_then(|s| s.strip_suffix(']'));
350 if let Some(kind) = kind.filter(|k| ["NOTE", "TIP", "IMPORTANT", "WARNING", "CAUTION"].contains(k)) {
351 blocks.push(Block::Alert(kind.into(), self.parse(&content[1..])));
352 } else {
353 blocks.push(Block::Quote(self.parse(&content)));
354 }
355 continue;
356 }
357 if let Some(mut item) = marker(line) {
358 let kind = item.kind;
359 let start = item.number;
360 let mut items = Vec::new();
361 loop {
362 let mut content = vec![item.content];
363 at += 1;
364 while at < lines.len() && lines[at].starts_with(" ") {
365 content.push(lines[at][2..].into());
366 at += 1;
367 }
368 items.push(self.parse(&content));
369 let mut next = at;
370 while next < lines.len() && lines[next].is_empty() {
371 next += 1;
372 }
373 let following = lines.get(next).and_then(|s| marker(s));
374 match following {
375 Some(m) if m.kind == kind && m.number.is_some() == start.is_some() => {
376 item = m;
377 at = next;
378 }
379 _ => break,
380 }
381 }
382 blocks.push(Block::List(start, items));
383 continue;
384 }
385 if self.definition(line) {
386 at += 1;
387 continue;
388 }
389 if at + 1 < lines.len() && line.contains('|') && lines[at + 1].contains('|') {
390 let header = cells(line);
391 let delimiters = cells(&lines[at + 1]);
392 let align: Option<Vec<String>> = delimiters.iter().map(|s| alignment(s)).collect();
393 if let Some(align) = align.filter(|_| header.len() == delimiters.len()) {
394 let columns = header.len();
395 let mut rows = Vec::new();
396 at += 2;
397 while at < lines.len() && lines[at].contains('|') {
398 let mut row = cells(&lines[at]);
399 row.resize(columns, String::new());
400 row.truncate(columns);
401 rows.push(row);
402 at += 1;
403 }
404 blocks.push(Block::Table(align, header, rows));
405 continue;
406 }
407 }
408 }
409 let mut text = vec![line.trim_start_matches(' ').to_owned()];
410 at += 1;
411 let mut level = 0;
412 while at < lines.len() && !lines[at].is_empty() {
413 let row = &lines[at];
414 if row.len() >= 3 && (row.starts_with('=') || row.starts_with('-')) && row.bytes().all(|c| c == row.as_bytes()[0]) {
415 level = if row.starts_with('=') {
416 1
417 } else {
418 2
419 };
420 at += 1;
421 break;
422 }
423 if interrupt(row) {
424 break;
425 }
426 text.push(row.trim_start_matches(' ').into());
427 at += 1;
428 }
429 let text = text.join("\n");
430 blocks.push(if level > 0 {
431 Block::Heading(level, text)
432 } else if text == "[TOC]" {
433 Block::Toc
434 } else {
435 Block::Paragraph(text)
436 });
437 }
438 blocks
439 }
440}

443 html: String,
444 delimiter: u8,
445 length: usize,
446 open: bool,
447 close: bool,
448}

450impl Piece {
451 fn text(html: String) -> Self {
452 Self {
453 html, delimiter: 0, length: 0, open: false, close: false

458fn emphasis(pieces: &[Piece]) -> String {
459 let mut html = String::new();
460 let mut at = 0;
461 while at < pieces.len() {
462 let piece = &pieces[at];
463 if piece.delimiter == 0 {
464 html.push_str(&piece.html);
465 at += 1;
466 continue;
467 }
468 let close = if piece.open {
469 (at + 1..pieces.len()).find(|&i| pieces[i].delimiter == piece.delimiter && pieces[i].length == piece.length && pieces[i].close)
470 } else {
471 None
472 };
473 if let Some(end) = close {
474 let (opening, closing) = match (piece.delimiter, piece.length) {
475 (b'~', _) => ("<del>", "</del>"), (_, 1) => ("<em>", "</em>"), (_, 2) => ("<strong>", "</strong>"), _ => ("<em><strong>", "</strong></em>"),
476 };
477 html.push_str(opening);
478 html.push_str(&emphasis(&pieces[at + 1..end]));
479 html.push_str(closing);
480 at = end + 1;
481 } else {
482 html.push_str(&String::from_utf8(vec![piece.delimiter; piece.length]).unwrap());
483 at += 1;
484 }
485 }
486 html
487}

489fn email(s: &str) -> bool {
490 let Some((local, domain)) = s.split_once('@') else {
491 return false;
492 };
493 if local.is_empty() || !local.bytes().all(|c| c.is_ascii_alphanumeric() || b".+-_".contains(&c)) {
494 return false;
495 }
496 let parts: Vec<&str> = domain.split('.').collect();
497 parts.len() >= 2 && parts.iter().all(|p| !p.is_empty() && p.bytes().all(|c| c.is_ascii_alphanumeric() || c == b'-'))
498}

523 fn destination(&self, s: &str, at: usize, text: &str) -> Option<(Reference, usize)> {
524 if s[at..].starts_with('[') {
525 if let Some(end) = s[at + 1..].find(']').map(|p| p + at + 1) {
526 let name = &s[at + 1..end];
527 if !name.contains('[') {
528 let key = normalize(if name.is_empty() {
529 text
530 } else {
531 name
532 });
533 return self.references.get(&key).cloned().map(|r| (r, end + 1));
534 }
535 }
536 } else if s[at..].starts_with('(') {
537 let (url, end) = escaped_read(s, skip(s, at + 1), b" \n)");
538 let mut next = skip(s, end);
539 let mut title = None;
540 let mut valid = true;
541 if next > end && next < s.len() && b"\"'".contains(&s.as_bytes()[next]) {
542 let (value, end) = escaped_read(s, next + 1, &[s.as_bytes()[next]]);
543 if end == s.len() {
544 valid = false;
545 } else {
546 title = Some(value);
547 next = skip(s, end + 1);
548 }
549 }
550 if valid && s[next..].starts_with(')') {
551 return Some((Reference {
552 url, title
553 }, next + 1));
554 }
555 }
556 self.references.get(&normalize(text)).cloned().map(|r| (r, at))
557 }

559 fn inline(&mut self, s: &str, links: bool, notes: bool) -> String {
560 let b = s.as_bytes();
561 let mut pieces = Vec::new();
562 let mut at = 0;
563 while at < s.len() {
564 let c = b[at];
565 if c == b'\\' {
566 if at + 1 < b.len() && b[at + 1].is_ascii_punctuation() {
567 pieces.push(Piece::text(escape(&s[at + 1..at + 2])));
568 at += 2;
569 } else if at + 1 < b.len() && b[at + 1] == b'\n' {
570 pieces.push(Piece::text("<br />\n".into()));
571 at += 2;
572 } else {
573 pieces.push(Piece::text("\\".into()));
574 at += 1;
575 }
576 continue;
577 }
578 if c == b'`' {
579 let end = run_end(s, at);
580 if let Some(close) = tick_closer(s, end, end - at) {
581 let mut code = s[end..close].replace('\n', " ");
582 if code.starts_with(' ') && code.ends_with(' ') && code.bytes().any(|c| c != b' ') {
583 code = code[1..code.len() - 1].into();
584 }
585 pieces.push(Piece::text(format!("<code>{}</code>", escape(&code))));
586 at = close + end - at;
587 } else {
588 pieces.push(Piece::text(s[at..end].into()));
589 at = end;
590 }
591 continue;
592 }
593 if links && c == b'<' {
594 if let Some(close) = s[at + 1..].find('>').map(|p| p + at + 1) {
595 let target = &s[at + 1..close];
596 let clean = !target.bytes().any(|c| white(c) || c == b'<');
597 let href = if clean && ((target.starts_with("http://") && target.len() > 7) || (target.starts_with("https://") && target.len() > 8)) {
598 Some(target.to_owned())
599 } else if email(target) {
600 Some(format!("mailto:{}", target))
601 } else {
602 None
603 };
604 if let Some(href) = href {
605 pieces.push(Piece::text(format!("<a href=\"{}\">{}</a>", escape(&href), escape(target))));
606 at = close + 1;
607 continue;
608 }
609 }
610 }
611 if links && (s[at..].starts_with("http://") || s[at..].starts_with("https://")) && (at == 0 || b" \n*_~(".contains(&b[at - 1])) {
612 let mut end = at;
613 while end < b.len() && !b" \n<".contains(&b[end]) {
614 end += 1;
615 }
616 while end > at && b".,:;!?*_~)".contains(&b[end - 1]) {
617 end -= 1;
618 }
619 let url = &s[at..end];
620 if url != "http://" && url != "https://" {
621 let url = escape(url);
622 pieces.push(Piece::text(format!("<a href=\"{}\">{}</a>", url, url)));
623 at = end;
624 continue;
625 }
626 }
627 if c == b'[' || (c == b'!' && b.get(at + 1) == Some(&b'[')) {
628 if links && notes && s[at..].starts_with("[^") {
629 if let Some(close) = s[at + 2..].find(']').map(|p| p + at + 2) {
630 let name = s[at + 2..close].to_ascii_lowercase();
631 if self.footnotes.contains_key(&name) {
632 pieces.push(Piece::text(self.note_reference(&name)));
633 at = close + 1;
634 continue;
635 }
636 }
637 }
638 let image = c == b'!';
639 let opening = at + usize::from(image);
640 if image || links {
641 if let Some(close) = closing_bracket(s, opening) {
642 let text = &s[opening + 1..close];
643 if let Some((reference, end)) = self.destination(s, close + 1, text) {
644 let content = self.inline(text, false, false);
645 let title = reference.title.map_or(String::new(), |t| format!(" title=\"{}\"", escape(&t)));
646 let url = escape(&reference.url);
647 let html = if image {
648 format!("<img src=\"{}\" alt=\"{}\"{} />", url, strip_tags(&content), title)
649 } else {
650 format!("<a href=\"{}\"{}>{}</a>", url, title, content)
651 };
652 pieces.push(Piece::text(html));
653 at = end;
654 continue;
655 }
656 }
657 }
658 }
659 if b"*_~".contains(&c) {
660 let end = run_end(s, at);
661 let length = end - at;
662 let before = if at == 0 {
663 b' '
664 } else {
665 b[at - 1]
666 };
667 let after = b.get(end).copied().unwrap_or(b' ');
668 let mut open = !white(after);
669 let mut close = !white(before);
670 if c == b'_' && before.is_ascii_alphanumeric() && after.is_ascii_alphanumeric() {
671 open = false;
672 close = false;
673 }
674 if (c != b'~' && length <= 3) || (c == b'~' && length == 2) {
675 pieces.push(Piece {
676 html: String::new(), delimiter: c, length, open, close
677 });
678 } else {
679 pieces.push(Piece::text(s[at..end].into()));
680 }
681 at = end;
682 continue;
683 }
684 pieces.push(Piece::text(escape(&s[at..at + 1])));
685 at += 1;
686 }
687 emphasis(&pieces)
688 }

693 html.push_str(&self.render(block));
694 html.push('\n');

699 fn render(&mut self, block: &Block) -> String {
700 match block {
701 Block::Heading(level, text) => self.heading(*level, text), Block::Paragraph(text) => format!("<p>{}</p>", self.inline(text, true, true)), Block::Rule => "<hr />".into(), Block::Code(language, lines) => {
702 let attribute = if language.is_empty() {
703 String::new()
704 } else {
705 format!(" class=\"language-{}\"", escape(language))
706 };
707 format!("<pre><code{}>{}</code></pre>", attribute, lines.iter().map(|s| format!("{}\n", escape(s))).collect::<String>())
708 }
709 Block::Quote(blocks) => format!("<blockquote>\n{}</blockquote>", self.blocks(blocks)), Block::Alert(kind, blocks) => {
710 let name = kind.to_ascii_lowercase();
711 let title = format!("{}{}", &kind[..1], &name[1..]);
712 format!("<div class=\"alert alert-{}\">\n<p class=\"alert-title\">{}</p>\n{}</div>", name, title, self.blocks(blocks))
713 }
714 Block::List(start, items) => {
715 let tag = if start.is_some() {
716 "ol"
717 } else {
718 "ul"
719 };
720 let attribute = start.filter(|&n| n != 1).map_or(String::new(), |n| format!(" start=\"{}\"", n));
721 let mut html = format!("<{}{}>\n", tag, attribute);
722 for item in items {
723 html.push_str(&self.item(item));
724 html.push('\n');
725 }
726 format!("{}</{}>", html, tag)
727 }
728 Block::Table(align, header, rows) => {
729 let mut html = format!("<table>\n<thead>\n<tr>\n{}</tr>\n</thead>\n", self.table_cells("th", align, header));
730 if !rows.is_empty() {
731 html.push_str("<tbody>\n");
732 for row in rows {
733 html.push_str(&format!("<tr>\n{}</tr>\n", self.table_cells("td", align, row)));
734 }
735 html.push_str("</tbody>\n");
736 }
737 html + "</table>"
738 }
739 Block::Toc => TOC.into(),
740 }
741 }

743 fn item(&mut self, blocks: &[Block]) -> String {
744 let mut html = "<li>".to_owned();
745 for (i, block) in blocks.iter().enumerate() {
746 if let Block::Paragraph(text) = block {
747 let task = i == 0 && text.len() > 4 && ["[ ] ", "[x] ", "[X] "].iter().any(|p| text.starts_with(p));
748 if task {
749 html.push_str(if text.as_bytes()[1] == b' ' {
750 "<input type=\"checkbox\" disabled=\"\" /> "
751 } else {
752 "<input type=\"checkbox\" checked=\"\" disabled=\"\" /> "
753 });
754 }
755 html.push_str(&self.inline(if task {
756 &text[4..]
757 } else {
758 text
759 }, true, true));
760 } else {
761 if !html.ends_with('\n') {
762 html.push('\n');
763 }
764 html.push_str(&self.render(block));
765 html.push('\n');
766 }
767 }
768 html + "</li>"
769 }

784 fn heading(&mut self, level: usize, source: &str) -> String {
785 let mut text = source;
786 let explicit = source.rfind(" {#").filter(|_| source.ends_with('}')).and_then(|at| {
787 let name = &source[at + 3..source.len() - 1];
788 if !name.is_empty() && name.bytes().all(label) {
789 Some((at, name))
790 } else {
791 None
792 }
793 });
794 let id = if let Some((at, name)) = explicit {
795 text = source[..at].trim_end_matches(' ');
796 name.into()
797 } else {
798 let mut base: String = source.to_ascii_lowercase().bytes().filter_map(|c| if label(c) {
799 Some(c as char)
800 } else if white(c) {
801 Some('-')
802 } else {
803 None
804 }).collect();
805 if base.is_empty() {
806 base = "heading".into();
807 }
808 let mut id = base.clone();
809 let mut suffix = 1;
810 while self.used_ids.contains(&id) {
811 id = format!("{}-{}", base, suffix);
812 suffix += 1;
813 }
814 id
815 };
816 self.used_ids.insert(id.clone());
817 let html = self.inline(text, true, true);
818 self.headings.push(Heading {
819 level, id: id.clone(), text: strip_tags(&html)
820 });
821 format!("<h{} id=\"{}\">{}</h{}>", level, id, html, level)
822 }

829 if !children[i].is_empty() {
830 html.push_str(&format!("\n{}\n", self.toc_list(children, &children[i])));
831 }
832 html.push_str("</li>\n");
833 }

837 fn toc(&self) -> String {
838 if self.headings.is_empty() {
839 return "<nav class=\"toc\">\n</nav>".into();
840 }
841 let mut roots = Vec::new();
842 let mut parents: Vec<usize> = Vec::new();
843 let mut children = vec![Vec::new(); self.headings.len()];
844 for (i, heading) in self.headings.iter().enumerate() {
845 while parents.last().is_some_and(|&p| self.headings[p].level >= heading.level) {
846 parents.pop();
847 }
848 if let Some(&p) = parents.last() {
849 children[p].push(i);
850 } else {
851 roots.push(i);
852 }
853 parents.push(i);
854 }
855 format!("<nav class=\"toc\">\n{}\n</nav>", self.toc_list(&children, &roots))
856 }
