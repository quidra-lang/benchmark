9const Builder = struct {
10 gpa: Allocator,
11 bytes: std.ArrayList(u8) = .empty,
12
13 fn add(self: *Builder, text: Text) Allocator.Error!void {
14 try self.bytes.appendSlice(self.gpa, text);
15 }
16
17 fn byte(self: *Builder, value: u8) Allocator.Error!void {
18 try self.bytes.append(self.gpa, value);
19 }
20
21 fn print(self: *Builder, comptime pattern: Text, args: anytype) Allocator.Error!void {
22 try self.add(try std.fmt.allocPrint(self.gpa, pattern, args));
23 }
24
25 fn result(self: *const Builder) Text {
26 return self.bytes.items;
27 }

55fn lower(gpa: Allocator, text: Text) Allocator.Error!Text {
56 const result = try gpa.dupe(u8, text);

62 return std.ascii.isAlphanumeric(letter) or letter == '-' or letter == '_';

66 if (text.len == 0) return false;
67 for (text) |letter| if (!labelCharacter(letter)) return false;

71fn escape(gpa: Allocator, text: Text) Allocator.Error!Text {
72 var out: Builder = .{ .gpa = gpa };
73 for (text) |letter| {
74 switch (letter) {
75 '&' => try out.add("&amp;"),
76 '<' => try out.add("&lt;"),
77 '>' => try out.add("&gt;"),
78 '"' => try out.add("&quot;"),
79 else => try out.byte(letter),
80 }
81 }
82 return out.result();
83}

85fn stripTags(gpa: Allocator, text: Text) Allocator.Error!Text {
86 var out: Builder = .{ .gpa = gpa };
87 var at: usize = 0;
88 while (find(text, "<", at)) |opening| {
89 const close = find(text, ">", opening + 1) orelse break;
90 try out.add(text[at..opening]);
91 at = close + 1;
92 }
93 try out.add(text[at..]);
94 return out.result();
95}

97fn normalize(gpa: Allocator, text: Text) Allocator.Error!Text {
98 var out: Builder = .{ .gpa = gpa };
99 var space = false;
100 for (text) |letter| {
101 if (contains(whitespace, letter)) {
102 space = out.bytes.items.len > 0;
103 } else {
104 if (space) try out.byte(' ');
105 try out.byte(std.ascii.toLower(letter));
106 space = false;
107 }
108 }
109 return out.result();
110}

112fn skipSpace(text: Text, start: usize) usize {
113 var at = start;
114 while (at < text.len and contains(whitespace, text[at])) : (at += 1) {}
115 return at;
116}

118const Read = struct { text: Text, end: usize };
119
120fn escapedRead(gpa: Allocator, text: Text, start: usize, stops: Text) Allocator.Error!Read {
121 var out: Builder = .{ .gpa = gpa };
122 var at = start;
123 while (at < text.len and !contains(stops, text[at])) : (at += 1) {
124 if (text[at] == '\\' and at + 1 < text.len and contains(punctuation, text[at + 1])) at += 1;
125 try out.byte(text[at]);
126 }
127 return .{ .text = out.result(), .end = at };
128}
129
130fn runEnd(text: Text, at: usize) usize {
131 var end = at + 1;
132 while (end < text.len and text[end] == text[at]) : (end += 1) {}
133 return end;
134}
135
136fn backtickCloser(text: Text, start: usize, length: usize) ?usize {
137 var at = start;
138 while (find(text, "`", at)) |next| {
139 const end = runEnd(text, next);
140 if (end - next == length) return next;
141 at = end;
142 }
143 return null;
144}
145
146fn closingBracket(text: Text, start: usize) ?usize {
147 var at = start;
148 var depth: usize = 0;
149 while (at < text.len) {
150 if (text[at] == '\\') {
151 at += 2;
152 continue;
153 }
154 if (text[at] == '`') {
155 const end = runEnd(text, at);
156 const close = backtickCloser(text, end, end - at);
157 at = if (close) |position| position + end - at else end;
158 continue;
159 }
160 if (text[at] == '[') depth += 1;
161 if (text[at] == ']') {
162 depth -= 1;
163 if (depth == 0) return at;
164 }
165 at += 1;
166 }
167 return null;
168}

170const BlockKind = enum { heading, paragraph, rule, code, quote, alert, list, table, toc };
171const Block = struct {
172 kind: BlockKind,
173 text: Text = "",
174 number: usize = 0,
175 ordered: bool = false,
176 lines: Texts = &.{},
177 children: []const *Block = &.{},
178 items: []const []const *Block = &.{},

182const Reference = struct { url: Text, title: ?Text = null };
183const Marker = struct { marker: u8, ordered: bool, number: usize, content: Text };
184const Fence = struct { fence: Text, info: Text };
185const Heading = struct { text: Text, level: usize };

187fn openingFence(line: Text) ?Fence {
188 if (line.len == 0 or !contains("`~", line[0])) return null;
189 const end = runEnd(line, 0);
190 if (end < 3) return null;
191 const info = trim(line[end..]);
192 if (line[0] == '`' and contains(info, '`')) return null;
193 return .{ .fence = line[0..end], .info = info };
194}
195
196fn atxHeading(line: Text) ?Heading {
197 var count: usize = 0;
198 while (count < line.len and line[count] == '#') : (count += 1) {}
199 if (count == 0 or count > 6 or (count < line.len and line[count] != ' ')) return null;
200 var text = trim(line[count..]);
201 var end = text.len;
202 while (end > 0 and text[end - 1] == '#') : (end -= 1) {}
203 if (end == 0 or text[end - 1] == ' ') text = trim(text[0..end]);
204 return .{ .text = text, .level = count };
205}
206
207fn isRule(line: Text) bool {
208 var count: usize = 0;
209 var character: u8 = 0;
210 for (line) |letter| {
211 if (letter == ' ') continue;
212 if (!contains("-*_", letter)) return false;
213 if (character != 0 and letter != character) return false;
214 character = letter;
215 count += 1;
216 }
217 return count >= 3;
218}
219
220fn listMarker(line: Text) ?Marker {
221 if (line.len < 2 or isRule(line)) return null;
222 var end: usize = 0;
223 var ordered = false;
224 var number: usize = 1;
225 var marker: u8 = undefined;
226 if (contains("-*+", line[0])) {
227 marker = line[0];
228 end = 1;
229 } else {
230 while (end < line.len and std.ascii.isDigit(line[end])) : (end += 1) {}
231 if (end == 0 or end > 9 or end == line.len or !contains(".)", line[end])) return null;
232 ordered = true;
233 number = std.fmt.parseInt(usize, line[0..end], 10) catch unreachable;
234 marker = line[end];
235 end += 1;
236 }
237 if (end == line.len or line[end] != ' ') return null;
238 while (end < line.len and line[end] == ' ') : (end += 1) {}
239 return .{ .marker = marker, .ordered = ordered, .number = number, .content = line[end..] };
240}
241
242fn interrupts(line: Text) bool {
243 if (line.len == 0 or line[0] == ' ') return false;
244 return openingFence(line) != null or atxHeading(line) != null or isRule(line) or line[0] == '>' or listMarker(line) != null;
245}
246
247fn underline(line: Text) ?usize {
248 if (line.len < 3 or !contains("=-", line[0])) return null;
249 for (line) |letter| if (letter != line[0]) return null;
250 return if (line[0] == '=') 1 else 2;
251}

253fn splitCells(gpa: Allocator, line: Text) Allocator.Error!Texts {
254 var text = trim(line);
255 if (starts(text, "|")) text = text[1..];
256 var cells: std.ArrayList(Text) = .empty;
257 var cell: Builder = .{ .gpa = gpa };
258 var final_separator = false;
259 var at: usize = 0;
260 while (at < text.len) {
261 const letter = text[at];
262 final_separator = false;
263 if (letter == '\\' and at + 1 < text.len) {
264 if (text[at + 1] != '|') try cell.byte('\\');
265 try cell.byte(text[at + 1]);
266 at += 2;
267 } else if (letter == '|') {
268 try cells.append(gpa, trim(cell.result()));
269 cell = .{ .gpa = gpa };
270 final_separator = true;
271 at += 1;
272 } else {
273 try cell.byte(letter);
274 at += 1;
275 }
276 }
277 if (!final_separator) try cells.append(gpa, trim(cell.result()));
278 return cells.items;
279}
280
281fn delimiterCell(cell: Text) bool {
282 var text = cell;
283 if (starts(text, ":")) text = text[1..];
284 if (ends(text, ":")) text = text[0 .. text.len - 1];
285 if (text.len == 0) return false;
286 for (text) |letter| if (letter != '-') return false;
287 return true;
288}

290const Parser = struct {
291 gpa: Allocator,
292 references: std.StringHashMapUnmanaged(Reference) = .empty,
293 footnotes: std.StringHashMapUnmanaged(Text) = .empty,

295 fn block(self: *Parser, kind: BlockKind, text: Text, number: usize) Allocator.Error!*Block {
296 const result = try self.gpa.create(Block);
297 result.* = .{ .kind = kind, .text = text, .number = number };
298 return result;
299 }
300
301 fn definition(self: *Parser, line: Text) Allocator.Error!bool {
302 if (!starts(line, "[")) return false;
303 const close = find(line, "]", 1) orelse return false;
304 if (close + 1 == line.len or line[close + 1] != ':') return false;
305 const label = line[1..close];
306 const body = trim(line[close + 2 ..]);
307 if (starts(label, "^") and isLabel(label[1..]) and body.len > 0) {
308 const key = try lower(self.gpa, label[1..]);
309 const entry = try self.footnotes.getOrPut(self.gpa, key);
310 if (!entry.found_existing) entry.value_ptr.* = body;
311 return true;
312 }
313 if (contains(label, '[') or trim(label).len == 0 or starts(label, "^")) return false;
314 const url = try escapedRead(self.gpa, line, skipSpace(line, close + 2), " ");
315 if (url.text.len == 0) return false;
316 const at = skipSpace(line, url.end);
317 var title: ?Text = null;
318 if (at < line.len) {
319 if (at == url.end or !contains("\"'", line[at])) return false;
320 const quoted = try escapedRead(self.gpa, line, at + 1, line[at .. at + 1]);
321 if (quoted.end != line.len - 1) return false;
322 title = quoted.text;
323 }
324 const key = try normalize(self.gpa, label);
325 const entry = try self.references.getOrPut(self.gpa, key);
326 if (!entry.found_existing) entry.value_ptr.* = .{ .url = url.text, .title = title };
327 return true;
328 }

330 fn parse(self: *Parser, lines: Texts) Allocator.Error![]const *Block {
331 var blocks: std.ArrayList(*Block) = .empty;
332 var at: usize = 0;
333 while (at < lines.len) {
334 const line = lines[at];
335 if (line.len == 0) {
336 at += 1;
337 continue;
338 }
339 if (starts(line, " ")) {
340 var end = at;
341 while (end < lines.len and (lines[end].len == 0 or starts(lines[end], " "))) : (end += 1) {}
342 while (lines[end - 1].len == 0) : (end -= 1) {}
343 const code = try self.block(.code, "", 0);
344 var code_lines: std.ArrayList(Text) = .empty;
345 while (at < end) : (at += 1) {
346 try code_lines.append(self.gpa, if (lines[at].len == 0) "" else lines[at][4..]);
347 }
348 code.lines = code_lines.items;
349 try blocks.append(self.gpa, code);
350 continue;
351 }
352 if (line[0] != ' ') {
353 if (openingFence(line)) |fence| {
354 const space = std.mem.findScalar(u8, fence.info, ' ') orelse fence.info.len;
355 const code = try self.block(.code, fence.info[0..space], 0);
356 var code_lines: std.ArrayList(Text) = .empty;
357 at += 1;
358 while (at < lines.len) {
359 const row = lines[at];
360 at += 1;
361 var closing = row.len >= fence.fence.len;
362 for (row) |letter| {
363 if (letter != fence.fence[0]) closing = false;
364 }
365 if (closing) break;
366 try code_lines.append(self.gpa, row);
367 }
368 code.lines = code_lines.items;
369 try blocks.append(self.gpa, code);
370 continue;
371 }
372 if (atxHeading(line)) |heading| {
373 try blocks.append(self.gpa, try self.block(.heading, heading.text, heading.level));
374 at += 1;
375 continue;
376 }
377 if (isRule(line)) {
378 try blocks.append(self.gpa, try self.block(.rule, "", 0));
379 at += 1;
380 continue;
381 }
382 if (line[0] == '>') {
383 var content: std.ArrayList(Text) = .empty;
384 while (at < lines.len and starts(lines[at], ">")) : (at += 1) {
385 var row = lines[at][1..];
386 if (starts(row, " ")) row = row[1..];
387 try content.append(self.gpa, row);
388 }
389 const quote = try self.block(.quote, "", 0);
390 var first: usize = 0;
391 for ([_]Text{ "NOTE", "TIP", "IMPORTANT", "WARNING", "CAUTION" }) |kind| {
392 const marker = try std.fmt.allocPrint(self.gpa, "[!{s}]", .{kind});
393 if (equal(content.items[0], marker)) {
394 quote.kind = .alert;
395 quote.text = kind;
396 first = 1;
397 break;
398 }
399 }
400 quote.children = try self.parse(content.items[first..]);
401 try blocks.append(self.gpa, quote);
402 continue;
403 }
404 if (listMarker(line)) |first_marker| {
405 const list = try self.block(.list, "", first_marker.number);
406 list.ordered = first_marker.ordered;
407 var items: std.ArrayList([]const *Block) = .empty;
408 var marker = first_marker;
409 while (true) {
410 var content: std.ArrayList(Text) = .empty;
411 try content.append(self.gpa, marker.content);
412 at += 1;
413 while (at < lines.len and starts(lines[at], " ")) : (at += 1) {
414 try content.append(self.gpa, lines[at][2..]);
415 }
416 try items.append(self.gpa, try self.parse(content.items));
417 var next = at;
418 while (next < lines.len and lines[next].len == 0) : (next += 1) {}
419 if (next == lines.len) break;
420 marker = listMarker(lines[next]) orelse break;
421 if (marker.marker != first_marker.marker or marker.ordered != first_marker.ordered) break;
422 at = next;
423 }
424 list.items = items.items;
425 try blocks.append(self.gpa, list);
426 continue;
427 }
428 if (try self.definition(line)) {
429 at += 1;
430 continue;
431 }
432 if (at + 1 < lines.len and contains(line, '|') and contains(lines[at + 1], '|')) {
433 const header = try splitCells(self.gpa, line);
434 const delimiters = try splitCells(self.gpa, lines[at + 1]);
435 var valid = header.len == delimiters.len;
436 for (delimiters) |cell| {
437 if (!delimiterCell(cell)) valid = false;
438 }
439 if (valid) {
440 const table = try self.block(.table, "", 0);
441 table.lines = header;
442 const alignments = try self.gpa.alloc(Text, delimiters.len);
443 for (delimiters, 0..) |cell, index| {
444 const left = starts(cell, ":");
445 const right = ends(cell, ":");
446 alignments[index] = if (left) (if (right) "center" else "left") else (if (right) "right" else "");
447 }
448 table.alignments = alignments;
449 var rows: std.ArrayList(Texts) = .empty;
450 at += 2;
451 while (at < lines.len and lines[at].len > 0 and contains(lines[at], '|')) : (at += 1) {
452 const cells = try splitCells(self.gpa, lines[at]);
453 const row = try self.gpa.alloc(Text, delimiters.len);
454 for (row, 0..) |*cell, index| cell.* = if (index < cells.len) cells[index] else "";
455 try rows.append(self.gpa, row);
456 }
457 table.rows = rows.items;
458 try blocks.append(self.gpa, table);
459 continue;
460 }
461 }
462 }
463 var text_lines: std.ArrayList(Text) = .empty;
464 try text_lines.append(self.gpa, std.mem.trimStart(u8, line, " "));
465 at += 1;
466 var level: usize = 0;
467 while (at < lines.len and lines[at].len > 0) {
468 if (underline(lines[at])) |heading_level| {
469 level = heading_level;
470 at += 1;
471 break;
472 }
473 if (interrupts(lines[at])) break;
474 try text_lines.append(self.gpa, std.mem.trimStart(u8, lines[at], " "));
475 at += 1;
476 }
477 const text = try std.mem.join(self.gpa, "\n", text_lines.items);
478 const kind: BlockKind = if (level > 0) .heading else if (equal(text, "[TOC]")) .toc else .paragraph;
479 try blocks.append(self.gpa, try self.block(kind, text, level));
480 }
481 return blocks.items;
482 }
483};

485const Piece = struct {
486 html: Text = "",
487 delimiter: u8 = 0,
488 length: usize = 0,
489 can_open: bool = false,
490 can_close: bool = false,
491};
492
493fn emphasis(gpa: Allocator, pieces: []const Piece) Allocator.Error!Text {
494 var out: Builder = .{ .gpa = gpa };
495 var at: usize = 0;
496 while (at < pieces.len) : (at += 1) {
497 const piece = pieces[at];
498 if (piece.delimiter == 0) {
499 try out.add(piece.html);
500 continue;
501 }
502 var closer: ?usize = null;
503 if (piece.can_open) {
504 for (pieces[at + 1 ..], at + 1..) |other, index| {
505 if (other.delimiter == piece.delimiter and other.length == piece.length and other.can_close) {
506 closer = index;
507 break;
508 }
509 }
510 }
511 if (closer) |close| {
512 const opening: Text = if (piece.delimiter == '~') "<del>" else switch (piece.length) {
513 1 => "<em>",
514 2 => "<strong>",
515 3 => "<em><strong>",
516 else => unreachable,
517 };
518 const closing: Text = if (piece.delimiter == '~') "</del>" else switch (piece.length) {
519 1 => "</em>",
520 2 => "</strong>",
521 3 => "</strong></em>",
522 else => unreachable,
523 };
524 try out.add(opening);
525 try out.add(try emphasis(gpa, pieces[at + 1 .. close]));
526 try out.add(closing);
527 at = close;
528 } else {
529 for (0..piece.length) |_| try out.byte(piece.delimiter);
530 }
531 }
532 return out.result();
533}

535fn isEmail(text: Text) bool {
536 const at = std.mem.findScalar(u8, text, '@') orelse return false;
537 if (at == 0) return false;
538 for (text[0..at]) |letter| {
539 if (!std.ascii.isAlphanumeric(letter) and !contains(".+-_", letter)) return false;
540 }
541 var parts = std.mem.splitScalar(u8, text[at + 1 ..], '.');
542 var count: usize = 0;
543 while (parts.next()) |part| {
544 if (part.len == 0) return false;
545 for (part) |letter| if (!std.ascii.isAlphanumeric(letter) and letter != '-') return false;
546 count += 1;
547 }
548 return count >= 2;
549}

559const Renderer = struct {
560 gpa: Allocator,
561 references: std.StringHashMapUnmanaged(Reference),
562 footnotes: std.StringHashMapUnmanaged(Text),
563 used_ids: std.StringHashMapUnmanaged(void) = .empty,

569 fn destination(self: *Renderer, text: Text, at: usize, label: Text) Allocator.Error!?Destination {
570 if (at < text.len and text[at] == '[') {
571 if (find(text, "]", at + 1)) |close| {
572 const key = text[at + 1 .. close];
573 if (!contains(key, '[')) {
574 const name = try normalize(self.gpa, if (key.len == 0) label else key);
575 const reference = self.references.get(name) orelse return null;
576 return .{ .reference = reference, .end = close + 1 };
577 }
578 }
579 } else if (at < text.len and text[at] == '(') {
580 const url = try escapedRead(self.gpa, text, skipSpace(text, at + 1), " \n)");
581 var next = skipSpace(text, url.end);
582 var title: ?Text = null;
583 var valid = true;
584 if (next > url.end and next < text.len and contains("\"'", text[next])) {
585 const quoted = try escapedRead(self.gpa, text, next + 1, text[next .. next + 1]);
586 if (quoted.end == text.len) valid = false else {
587 title = quoted.text;
588 next = skipSpace(text, quoted.end + 1);
589 }
590 }
591 if (valid and next < text.len and text[next] == ')') return .{
592 .reference = .{ .url = url.text, .title = title },
593 .end = next + 1,
594 };
595 }
596 const reference = self.references.get(try normalize(self.gpa, label)) orelse return null;
597 return .{ .reference = reference, .end = at };
598 }

608 entry.value_ptr.* = self.footnote_names.items.len;

613 self.footnote_counts.items[index] += 1;

616 }

618 fn bracket(self: *Renderer, text: Text, at: usize, links: bool, notes: bool) Allocator.Error!?Bracket {
619 if (links and notes and starts(text[at..], "[^")) {
620 if (find(text, "]", at + 2)) |close| {
621 const name = try lower(self.gpa, text[at + 2 .. close]);
622 if (self.footnotes.contains(name)) return .{ .html = try self.footnoteReference(name), .end = close + 1 };
623 }
624 }
625 const image = text[at] == '!';
626 if (!image and !links) return null;
627 const opening = if (image) at + 1 else at;
628 const close = closingBracket(text, opening) orelse return null;
629 const label = text[opening + 1 .. close];
630 const dest = (try self.destination(text, close + 1, label)) orelse return null;
631 const content = try self.inlineHtml(label, false, false);
632 var out: Builder = .{ .gpa = self.gpa };
633 const url = try escape(self.gpa, dest.reference.url);
634 if (image) try out.print("<img src=\"{s}\" alt=\"{s}\"", .{ url, try stripTags(self.gpa, content) }) else try out.print("<a href=\"{s}\"", .{url});
635 if (dest.reference.title) |title| try out.print(" title=\"{s}\"", .{try escape(self.gpa, title)});
636 if (image) try out.add(" />") else try out.print(">{s}</a>", .{content});
637 return .{ .html = out.result(), .end = dest.end };
638 }

640 fn inlineHtml(self: *Renderer, text: Text, links: bool, notes: bool) Allocator.Error!Text {
641 var pieces: std.ArrayList(Piece) = .empty;
642 var at: usize = 0;
643 while (at < text.len) {
644 const letter = text[at];
645 if (letter == '\\') {
646 if (at + 1 < text.len and contains(punctuation, text[at + 1])) {
647 try pieces.append(self.gpa, .{ .html = try escape(self.gpa, text[at + 1 .. at + 2]) });
648 at += 2;
649 } else if (at + 1 < text.len and text[at + 1] == '\n') {
650 try pieces.append(self.gpa, .{ .html = "<br />\n" });
651 at += 2;
652 } else {
653 try pieces.append(self.gpa, .{ .html = "\\" });
654 at += 1;
655 }
656 continue;
657 }
658 if (letter == '`') {
659 const end = runEnd(text, at);
660 if (backtickCloser(text, end, end - at)) |close| {
661 const code = try self.gpa.dupe(u8, text[end..close]);
662 for (code) |*character| {
663 if (character.* == '\n') character.* = ' ';
664 }
665 var value: Text = code;
666 if (starts(value, " ") and ends(value, " ") and trim(value).len > 0) value = value[1 .. value.len - 1];
667 try pieces.append(self.gpa, .{ .html = try std.fmt.allocPrint(self.gpa, "<code>{s}</code>", .{try escape(self.gpa, value)}) });
668 at = close + end - at;
669 } else {
670 try pieces.append(self.gpa, .{ .html = text[at..end] });
671 at = end;
672 }
673 continue;
674 }
675 if (links and letter == '<') {
676 if (find(text, ">", at + 1)) |close| {
677 const target = text[at + 1 .. close];
678 var clean = true;
679 for (target) |character| {
680 if (contains(" \n<", character)) clean = false;
681 }
682 var href: ?Text = null;
683 if (clean and ((starts(target, "http://") and target.len > 7) or (starts(target, "https://") and target.len > 8))) {
684 href = target;
685 } else if (isEmail(target)) {
686 href = try std.mem.concat(self.gpa, u8, &.{ "mailto:", target });
687 }
688 if (href) |url| {
689 try pieces.append(self.gpa, .{ .html = try std.fmt.allocPrint(self.gpa, "<a href=\"{s}\">{s}</a>", .{
690 try escape(self.gpa, url), try escape(self.gpa, target),
691 }) });
692 at = close + 1;
693 continue;
694 }
695 }
696 }
697 if (links and (starts(text[at..], "http://") or starts(text[at..], "https://")) and
698 (at == 0 or contains(" \n*_~(", text[at - 1])))
699 {
700 var end = at;
701 while (end < text.len and !contains(" \n<", text[end])) : (end += 1) {}
702 while (end > at and contains(".,:;!?*_~)", text[end - 1])) : (end -= 1) {}
703 const url = text[at..end];
704 if (!equal(url, "http://") and !equal(url, "https://")) {
705 const escaped = try escape(self.gpa, url);
706 try pieces.append(self.gpa, .{ .html = try std.fmt.allocPrint(self.gpa, "<a href=\"{s}\">{s}</a>", .{ escaped, escaped }) });
707 at = end;
708 continue;
709 }
710 }
711 if (letter == '[' or (letter == '!' and at + 1 < text.len and text[at + 1] == '[')) {
712 if (try self.bracket(text, at, links, notes)) |result| {
713 try pieces.append(self.gpa, .{ .html = result.html });
714 at = result.end;
715 } else {
716 try pieces.append(self.gpa, .{ .html = text[at .. at + 1] });
717 at += 1;
718 }
719 continue;
720 }
721 if (contains("*_~", letter)) {
722 const end = runEnd(text, at);
723 const length = end - at;
724 const before: u8 = if (at == 0) ' ' else text[at - 1];
725 const after: u8 = if (end == text.len) ' ' else text[end];
726 var can_open = !contains(whitespace, after);
727 var can_close = !contains(whitespace, before);
728 if (letter == '_' and std.ascii.isAlphanumeric(before) and std.ascii.isAlphanumeric(after)) {
729 can_open = false;
730 can_close = false;
731 }
732 if ((letter != '~' and length <= 3) or (letter == '~' and length == 2)) {
733 try pieces.append(self.gpa, .{ .delimiter = letter, .length = length, .can_open = can_open, .can_close = can_close });
734 } else try pieces.append(self.gpa, .{ .html = text[at..end] });
735 at = end;
736 continue;
737 }
738 try pieces.append(self.gpa, .{ .html = try escape(self.gpa, text[at .. at + 1]) });
739 at += 1;
740 }
741 return emphasis(self.gpa, pieces.items);
742 }

753 fn heading(self: *Renderer, level: usize, original: Text) Allocator.Error!Text {
754 var text = original;
755 var explicit: ?usize = null;
756 var at: usize = 0;
757 while (find(text, " {#", at)) |position| {
758 explicit = position;
759 at = position + 1;
760 }
761 var identifier: ?Text = null;
762 if (ends(text, "}") and explicit != null) {
763 const start = explicit.?;
764 const candidate = text[start + 3 .. text.len - 1];
765 if (isLabel(candidate)) {
766 identifier = candidate;
767 text = std.mem.trimEnd(u8, text[0..start], " ");
768 }
769 }
770 if (identifier == null) {
771 var slug: Builder = .{ .gpa = self.gpa };
772 for (text) |character| {
773 const letter = std.ascii.toLower(character);
774 if (labelCharacter(letter)) try slug.byte(letter) else if (contains(whitespace, letter)) try slug.byte('-');
775 }
776 const base: Text = if (slug.result().len == 0) "heading" else slug.result();
777 identifier = base;
778 var suffix: usize = 1;
779 while (self.used_ids.contains(identifier.?)) : (suffix += 1) {
780 identifier = try std.fmt.allocPrint(self.gpa, "{s}-{d}", .{ base, suffix });
781 }
782 }
783 const id = identifier.?;
784 try self.used_ids.put(self.gpa, id, {});
785 const html = try self.inlineHtml(text, true, true);
786 try self.headings.append(self.gpa, .{ .level = level, .id = id, .label = try stripTags(self.gpa, html) });
787 return std.fmt.allocPrint(self.gpa, "<h{d} id=\"{s}\">{s}</h{d}>", .{ level, id, html, level });
788 }

790 fn item(self: *Renderer, content: []const *Block) Allocator.Error!Text {
791 var out: Builder = .{ .gpa = self.gpa };
792 try out.add("<li>");
793 for (content, 0..) |block, index| {
794 if (block.kind == .paragraph) {
795 var text = block.text;
796 if (index == 0 and text.len > 4 and (starts(text, "[ ] ") or starts(text, "[x] ") or starts(text, "[X] "))) {
797 try out.add(if (text[1] == ' ') "<input type=\"checkbox\" disabled=\"\" /> " else "<input type=\"checkbox\" checked=\"\" disabled=\"\" /> ");
798 text = text[4..];
799 }
800 try out.add(try self.inlineHtml(text, true, true));
801 } else {
802 if (!ends(out.result(), "\n")) try out.byte('\n');
803 try out.add(try self.render(block));
804 try out.byte('\n');
805 }
806 }
807 try out.add("</li>");
808 return out.result();
809 }

816 try out.print(">{s}</{s}>\n", .{ try self.inlineHtml(cell, true, true), tag });

821 fn render(self: *Renderer, block: *const Block) Allocator.Error!Text {
822 var out: Builder = .{ .gpa = self.gpa };
823 switch (block.kind) {
824 .heading => return self.heading(block.number, block.text),
825 .paragraph => return std.fmt.allocPrint(self.gpa, "<p>{s}</p>", .{try self.inlineHtml(block.text, true, true)}),
826 .rule => return "<hr />",
827 .code => {
828 try out.add("<pre><code");
829 if (block.text.len > 0) try out.print(" class=\"language-{s}\"", .{try escape(self.gpa, block.text)});
830 try out.byte('>');
831 for (block.lines) |line| {
832 try out.add(try escape(self.gpa, line));
833 try out.byte('\n');
834 }
835 try out.add("</code></pre>");
836 },
837 .quote => try out.print("<blockquote>\n{s}</blockquote>", .{try self.blocks(block.children)}),
838 .alert => {
839 const kind = try lower(self.gpa, block.text);
840 const title = try self.gpa.dupe(u8, kind);
841 title[0] = std.ascii.toUpper(title[0]);
842 try out.print("<div class=\"alert alert-{s}\">\n<p class=\"alert-title\">{s}</p>\n{s}</div>", .{
843 kind, title, try self.blocks(block.children),
844 });
845 },
846 .list => {
847 const tag: Text = if (block.ordered) "ol" else "ul";
848 try out.print("<{s}", .{tag});
849 if (block.ordered and block.number != 1) try out.print(" start=\"{d}\"", .{block.number});
850 try out.add(">\n");
851 for (block.items) |content| {
852 try out.add(try self.item(content));
853 try out.byte('\n');
854 }
855 try out.print("</{s}>", .{tag});
856 },
857 .table => {
858 try out.print("<table>\n<thead>\n<tr>\n{s}</tr>\n</thead>\n", .{try self.tableCells("th", block, block.lines)});
859 if (block.rows.len > 0) {
860 try out.add("<tbody>\n");
861 for (block.rows) |row| try out.print("<tr>\n{s}</tr>\n", .{try self.tableCells("td", block, row)});
862 try out.add("</tbody>\n");
863 }
864 try out.add("</table>");
865 },
866 .toc => return toc_marker,
867 }
868 return out.result();
869 }

884 fn toc(self: *Renderer) Allocator.Error!Text {
885 if (self.headings.items.len == 0) return "<nav class=\"toc\">\n</nav>";
886 var roots: std.ArrayList(usize) = .empty;
887 var parents: std.ArrayList(usize) = .empty;
888 for (0..self.headings.items.len) |index| {
889 while (parents.items.len > 0 and self.headings.items[parents.items[parents.items.len - 1]].level >= self.headings.items[index].level) {
890 parents.items.len -= 1;
891 }
892 if (parents.items.len == 0) try roots.append(self.gpa, index) else try self.headings.items[parents.items[parents.items.len - 1]].children.append(self.gpa, index);
893 try parents.append(self.gpa, index);
894 }
895 return std.fmt.allocPrint(self.gpa, "<nav class=\"toc\">\n{s}\n</nav>", .{try self.tocList(roots.items)});
896 }

913 fn document(self: *Renderer, content: []const *Block) Allocator.Error!Text {
914 const html = try self.blocks(content);
915 const contents = try self.toc();
916 var out: Builder = .{ .gpa = self.gpa };
917 var at: usize = 0;
918 while (find(html, toc_marker, at)) |position| {
919 try out.add(html[at..position]);
920 try out.add(contents);
921 at = position + toc_marker.len;
922 }
923 try out.add(html[at..]);
924 if (self.footnote_names.items.len > 0) try out.add(try self.footnoteList());
925 return out.result();

929pub fn main(init: std.process.Init) !void {
930 var storage = std.heap.ArenaAllocator.init(std.heap.page_allocator);
931 defer storage.deinit();
932 const gpa = storage.allocator();
933 var input_buffer: [4096]u8 = undefined;
934 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
935 var output_buffer: [4096]u8 = undefined;
936 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
937 const text = try input.interface.allocRemaining(gpa, .unlimited);
938 var lines: std.ArrayList(Text) = .empty;
939 var pieces = std.mem.splitScalar(u8, text, '\n');
940 while (pieces.next()) |line| {
941 if (equal(line, "@@END@@")) break;
942 try lines.append(gpa, line);
943 }
944 var parser: Parser = .{ .gpa = gpa };
945 const blocks = try parser.parse(lines.items);
946 var renderer: Renderer = .{ .gpa = gpa, .references = parser.references, .footnotes = parser.footnotes };
947 try output.interface.writeAll(try renderer.document(blocks));
948 try output.interface.flush();
949}
