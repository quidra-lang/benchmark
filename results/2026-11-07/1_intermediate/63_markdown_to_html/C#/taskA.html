9 const string Punctuation = "!\"#$%&'()*+,-./:;<=>?@[\\]^_`{|}~";
10 const string TocMarker = "\u0001toc\u0002";
11 static bool Has(string set, char c) => set.Contains(c);
12 static bool Alphanumeric(char c) =>
13 c >= 'a' && c <= 'z' || c >= 'A' && c <= 'Z' || c >= '0' && c <= '9';
14 static bool LabelCharacter(char c) => Alphanumeric(c) || c == '-' || c == '_';
15 static bool Whitespace(char c) => c == ' ' || c == '\n';
16 static bool Matches(string text, string pattern) => Regex.IsMatch(text,
17 "\\A(?:" + pattern + ")\\z");
18 static string Trim(string text) => text.Trim(' ');
19 static string EscapeHtml(string text) => text.Replace("&", "&amp;")
20 .Replace("<", "&lt;")
21 .Replace(">", "&gt;")
22 .Replace("\"", "&quot;");
23 static string StripTags(string text) => Regex.Replace(text, "<[^>]*>", "");
24 static string Normalize(string text)
25 {
26 var result = new StringBuilder();
27 bool space = false;
28 foreach (char ch in text)
29 {
30 if (Whitespace(ch))
31 space = result.Length > 0;
32 else
33 {
34 if (space)
35 result.Append(' ');
36 result.Append(char.ToLowerInvariant(ch));
37 space = false;
38 }
39 }
40 return result.ToString();
41 }
42 static int SkipSpace(string text, int at)
43 {
44 while (at < text.Length && Whitespace(text[at]))
45 at++;
46 return at;
47 }
48 sealed record Read(string Text, int End);
49 static Read EscapedRead(string text, int at, string stops)
50 {
51 var result = new StringBuilder();
52 while (at < text.Length && !Has(stops, text[at]))
53 {
54 if (text[at] == '\\' && at + 1 < text.Length && Has(Punctuation, text[at + 1]))
55 at++;
56 result.Append(text[at++]);
57 }
58 return new(result.ToString(), at);
59 }

92 static Fence? OpeningFence(string line)
93 {
94 if (line.Length == 0 || !Has("`~", line[0]))
95 return null;
96 int end = 1;
97 while (end < line.Length && line[end] == line[0])
98 end++;
99 if (end < 3)
100 return null;
101 string info = Trim(line[end..]);
102 if (line[0] == '`' && info.Contains('`'))
103 return null;
104 return new(line[0], end, info);
105 }
106 static Block? AtxHeading(string line)
107 {
108 int n = 0;
109 while (n < line.Length && line[n] == '#')
110 n++;
111 if (n < 1 || n > 6 || n < line.Length && line[n] != ' ')
112 return null;
113 string text = Trim(line[n..]);
114 int end = text.Length;
115 while (end > 0 && text[end - 1] == '#')
116 end--;
117 if (end == 0 || text[end - 1] == ' ')
118 text = Trim(text[..end]);
119 return new(Kind.Heading, text, n);
120 }
121 static bool IsRule(string line)
122 {
123 string compact = line.Replace(" ", "");
124 return compact.Length >= 3 && Has("-*_", compact[0]) && compact.All(c => c == compact[0]);
125 }
126 static readonly Regex ListItem = new("\\A(?:([-*+])|([0-9]{1,9})([.)])) +");
127 static ListMarker? Marker(string line)
128 {
129 if (IsRule(line))
130 return null;
131 Match match = ListItem.Match(line);
132 if (!match.Success)
133 return null;
134 bool ordered = match.Groups[2].Success;
135 return new((ordered ? match.Groups[3].Value : match.Groups[1].Value)[0], ordered,
136 ordered ? int.Parse(match.Groups[2].Value) : 1, line[match.Length..]);
137 }
138 static bool Interrupts(string line) => line.Length > 0 && line[0] != ' ' &&
139 (OpeningFence(line) != null ||
140 AtxHeading(line) != null || IsRule(line) ||
141 line[0] == '>' || Marker(line) != null);
142 static List<string> SplitCells(string line)
143 {
144 string text = Trim(line);
145 int at = text.StartsWith('|') ? 1 : 0;
146 var cells = new List<string>();
147 var cell = new StringBuilder();
148 bool finalSeparator = false;
149 while (at < text.Length)
150 {
151 char ch = text[at++];
152 finalSeparator = false;
153 if (ch == '\\' && at < text.Length)
154 {
155 if (text[at] != '|')
156 cell.Append('\\');
157 cell.Append(text[at++]);
158 }
159 else if (ch == '|')
160 {
161 cells.Add(Trim(cell.ToString()));
162 cell.Clear();
163 finalSeparator = true;
164 }
165 else
166 cell.Append(ch);
167 }
168 if (!finalSeparator)
169 cells.Add(Trim(cell.ToString()));
170 return cells;
171 }

176 bool Definition(string line)
177 {
178 Match footnote = Regex.Match(line, "\\A\\[\\^([A-Za-z0-9_-]+)\\]:(.*)\\z");
179 if (footnote.Success && Trim(footnote.Groups[2].Value).Length > 0)
180 {
181 Footnotes.TryAdd(footnote.Groups[1].Value.ToLowerInvariant(),
182 Trim(footnote.Groups[2].Value));
183 return true;
184 }
185 if (!line.StartsWith('['))
186 return false;
187 int close = line.IndexOf(']');
188 if (close < 0 || close + 1 >= line.Length || line[close + 1] != ':')
189 return false;
190 string label = line[1..close];
191 if (label.Contains('[') || Trim(label).Length == 0 || label.StartsWith('^'))
192 return false;
193 Read url = EscapedRead(line, SkipSpace(line, close + 2), " ");
194 if (url.Text.Length == 0)
195 return false;
196 int at = SkipSpace(line, url.End);
197 string? title = null;
198 if (at < line.Length)
199 {
200 if (at == url.End || !Has("\"'", line[at]))
201 return false;
202 Read quoted = EscapedRead(line, at + 1, line[at].ToString());
203 if (quoted.End != line.Length - 1)
204 return false;
205 title = quoted.Text;
206 }
207 References.TryAdd(Normalize(label), new(url.Text, title));
208 return true;
209 }
210 public List<Block> Parse(List<string> lines)
211 {
212 var blocks = new List<Block>();
213 int at = 0;
214 while (at < lines.Count)
215 {
216 string line = lines[at];
217 if (line.Length == 0)
218 {
219 at++;
220 continue;
221 }
222 if (line.StartsWith(" "))
223 {
224 var code = new Block(Kind.Code);
225 int end = at;
226 while (end < lines.Count &&
227 (lines[end].Length == 0 || lines[end].StartsWith(" ")))
228 end++;
229 while (end > at && lines[end - 1].Length == 0)
230 end--;
231 while (at < end)
232 {
233 string row = lines[at++];
234 code.Lines.Add(row.Length == 0 ? "" : row[4..]);
235 }
236 blocks.Add(code);
237 continue;
238 }
239 if (line[0] != ' ')
240 {
241 Fence? fence = OpeningFence(line);
242 if (fence != null)
243 {
244 var code = new Block(Kind.Code, fence.Info.Split(' ')[0]);
245 at++;
246 while (at < lines.Count)
247 {
248 string row = lines[at++];
249 if (row.Length >= fence.Length && row.All(c => c == fence.Character))
250 break;
251 code.Lines.Add(row);
252 }
253 blocks.Add(code);
254 continue;
255 }
256 Block? heading = AtxHeading(line);
257 if (heading != null)
258 {
259 blocks.Add(heading);
260 at++;
261 continue;
262 }
263 if (IsRule(line))
264 {
265 blocks.Add(new(Kind.Rule));
266 at++;
267 continue;
268 }
269 if (line.StartsWith('>'))
270 {
271 var content = new List<string>();
272 while (at < lines.Count && lines[at].StartsWith('>'))
273 {
274 string row = lines[at++][1..];
275 if (row.StartsWith(' '))
276 row = row[1..];
277 content.Add(row);
278 }
279 var quote = new Block(Kind.Quote);
280 if (Matches(content[0], "\\[!(NOTE|TIP|IMPORTANT|WARNING|CAUTION)\\]"))
281 {
282 quote.Kind = Kind.Alert;
283 quote.Text = content[0][2.. ^ 1];
284 content.RemoveAt(0);
285 }
286 quote.Children = Parse(content);
287 blocks.Add(quote);
288 continue;
289 }
290 ListMarker? marker = Marker(line);
291 if (marker != null)
292 {
293 var list = new Block(Kind.List, "", marker.Number);
294 list.Ordered = marker.Ordered;
295 char kind = marker.Marker;
296 while (true)
297 {
298 var content = new List<string> { marker.Content };
299 at++;
300 while (at < lines.Count && lines[at].StartsWith(" "))
301 content.Add(lines[at++][2..]);
302 list.Items.Add(Parse(content));
303 int next = at;
304 while (next < lines.Count && lines[next].Length == 0)
305 next++;
306 marker = next < lines.Count ? Marker(lines[next]) : null;
307 if (marker == null || marker.Marker != kind ||
308 marker.Ordered != list.Ordered)
309 break;
310 at = next;
311 }
312 blocks.Add(list);
313 continue;
314 }
315 if (Definition(line))
316 {
317 at++;
318 continue;
319 }
320 if (at + 1 < lines.Count && line.Contains('|') && lines[at + 1].Contains('|'))
321 {
322 List<string> header = SplitCells(line),
323 delimiters = SplitCells(lines[at + 1]);
324 if (header.Count == delimiters.Count &&
325 delimiters.All(c => Matches(c, ":?-+:?")))
326 {
327 var table = new Block(Kind.Table);
328 table.Lines = header;
329 foreach (string cell in delimiters)
330 {
331 bool left = cell.StartsWith(':'), right = cell.EndsWith(':');
332 table.Alignments.Add(left ? (right ? "center" : "left")
333 : (right ? "right" : ""));
334 }
335 at += 2;
336 while (at < lines.Count && lines[at].Contains('|'))
337 {
338 List<string> row = SplitCells(lines[at++]);
339 while (row.Count < delimiters.Count)
340 row.Add("");
341 table.Rows.Add(row.Take(delimiters.Count).ToList());
342 }
343 blocks.Add(table);
344 continue;
345 }
346 }
347 }
348 var text = new List<string> { line.TrimStart(' ') };
349 at++;
350 int level = 0;
351 while (at < lines.Count && lines[at].Length > 0)
352 {
353 string row = lines[at];
354 if (Matches(row, "={3,}|-{3,}"))
355 {
356 level = row[0] == '=' ? 1 : 2;
357 at++;
358 break;
359 }
360 if (Interrupts(row))
361 break;
362 text.Add(row.TrimStart(' '));
363 at++;
364 }
365 string combined = string.Join("\n", text);
366 blocks.Add(new(level != 0 ? Kind.Heading
367 : combined == "[TOC]" ? Kind.Toc
368 : Kind.Paragraph,
369 combined, level));
370 }
371 return blocks;
372 }
373 }
374 static int RunEnd(string text, int at)
375 {
376 char ch = text[at++];
377 while (at < text.Length && text[at] == ch)
378 at++;
379 return at;
380 }
381 static int BacktickCloser(string text, int at, int length)
382 {
383 while ((at = text.IndexOf('`', at)) >= 0)
384 {
385 int end = RunEnd(text, at);
386 if (end - at == length)
387 return at;
388 at = end;
389 }
390 return -1;
391 }
392 static int ClosingBracket(string text, int at)
393 {
394 int depth = 0;
395 while (at < text.Length)
396 {
397 char ch = text[at];
398 if (ch == '\\')
399 {
400 at += 2;
401 continue;
402 }
403 if (ch == '`')
404 {
405 int end = RunEnd(text, at), close = BacktickCloser(text, end, end - at);
406 at = close < 0 ? end : close + end - at;
407 continue;
408 }
409 if (ch == '[')
410 depth++;
411 if (ch == ']' && --depth == 0)
412 return at;
413 at++;
414 }
415 return -1;
416 }
417 sealed record Piece(string Html, char Delimiter = '\0', int Length = 0, bool CanOpen = false,
418 bool CanClose = false);
419 static void Emphasis(StringBuilder html, List<Piece> pieces, int begin, int end)
420 {
421 for (int at = begin; at < end; at++)
422 {
423 Piece piece = pieces[at];
424 if (piece.Delimiter == 0)
425 {
426 html.Append(piece.Html);
427 continue;
428 }
429 int close = at + 1;
430 if (piece.CanOpen)
431 {
432 while (close < end &&
433 !(pieces[close].Delimiter == piece.Delimiter &&
434 pieces[close].Length == piece.Length && pieces[close].CanClose))
435 close++;
436 }
437 else
438 close = end;
439 if (close == end)
440 {
441 html.Append(piece.Delimiter, piece.Length);
442 continue;
443 }
444 string open, shut;
445 if (piece.Delimiter == '~')
446 {
447 open = "<del>";
448 shut = "</del>";
449 }
450 else if (piece.Length == 1)
451 {
452 open = "<em>";
453 shut = "</em>";
454 }
455 else if (piece.Length == 2)
456 {
457 open = "<strong>";
458 shut = "</strong>";
459 }
460 else
461 {
462 open = "<em><strong>";
463 shut = "</strong></em>";
464 }
465 html.Append(open);
466 Emphasis(html, pieces, at + 1, close);
467 html.Append(shut);
468 at = close;
469 }

504 static string ReferenceId(int number, int count) => "fnref-" + number +
505 (count == 1 ? "" : "-" + count);
506 string FootnoteReference(string name)
507 {
508 if (!noteNumbers.ContainsKey(name))
509 {
510 noteNumbers[name] = notes.Count;
511 notes.Add(new(name));
512 }
513 int index = noteNumbers[name], number = index + 1;
514 Note note = notes[index];
515 return "<sup id=\"" + ReferenceId(number, ++note.Count) + "\"><a href=\"#fn-" + number +
516 "\">" + number + "</a></sup>";
517 }
518 Destination? GetDestination(string text, int at, string label)
519 {
520 if (at < text.Length && text[at] == '[')
521 {
522 int close = text.IndexOf(']', at + 1);
523 if (close >= 0 && !text[(at + 1)..close].Contains('['))
524 {
525 string key = text[(at + 1)..close];
526 key = Normalize(key.Length == 0 ? label : key);
527 if (parser.References.TryGetValue(key, out Reference? reference))
528 return new(reference, close + 1);
529 return null;
530 }
531 }
532 else if (at < text.Length && text[at] == '(')
533 {
534 Read url = EscapedRead(text, SkipSpace(text, at + 1), " \n)");
535 int next = SkipSpace(text, url.End);
536 string? title = null;
537 bool valid = true;
538 if (next > url.End && next < text.Length && Has("\"'", text[next]))
539 {
540 Read quoted = EscapedRead(text, next + 1, text[next].ToString());
541 if (quoted.End == text.Length)
542 valid = false;
543 else
544 {
545 title = quoted.Text;
546 next = SkipSpace(text, quoted.End + 1);
547 }
548 }
549 if (valid && next < text.Length && text[next] == ')')
550 return new(new(url.Text, title), next + 1);
551 }
552 if (parser.References.TryGetValue(Normalize(label), out Reference? shortcut))
553 return new(shortcut, at);
554 return null;
555 }
556 string Inline(string text, bool links = true, bool footnotes = true)
557 {
558 var pieces = new List<Piece>();
559 int at = 0;
560 while (at < text.Length)
561 {
562 char ch = text[at];
563 if (ch == '\\')
564 {
565 if (at + 1 < text.Length && Has(Punctuation, text[at + 1]))
566 {
567 pieces.Add(new(EscapeHtml(text.Substring(at + 1, 1))));
568 at += 2;
569 }
570 else if (at + 1 < text.Length && text[at + 1] == '\n')
571 {
572 pieces.Add(new("<br />\n"));
573 at += 2;
574 }
575 else
576 {
577 pieces.Add(new("\\"));
578 at++;
579 }
580 continue;
581 }
582 if (ch == '`')
583 {
584 int end = RunEnd(text, at), close = BacktickCloser(text, end, end - at);
585 if (close < 0)
586 {
587 pieces.Add(new(text[at..end]));
588 at = end;
589 }
590 else
591 {
592 string code = text[end..close].Replace('\n', ' ');
593 if (code.StartsWith(' ') && code.EndsWith(' ') && Trim(code).Length > 0)
594 code = code[1.. ^ 1];
595 pieces.Add(new("<code>" + EscapeHtml(code) + "</code>"));
596 at = close + end - at;
597 }
598 continue;
599 }
600 if (links && ch == '<')
601 {
602 int close = text.IndexOf('>', at + 1);
603 if (close >= 0)
604 {
605 string target = text[(at + 1)..close];
606 string? href = null;
607 if (!target.Any(c => Has(" \n<", c)) &&
608 (target.StartsWith("http://", StringComparison.Ordinal) &&
609 target.Length > 7 ||
610 target.StartsWith("https://", StringComparison.Ordinal) &&
611 target.Length > 8))
612 href = target;
613 if (href == null &&
614 Matches(target, "[A-Za-z0-9.+_-]+@[A-Za-z0-9-]+(?:\\.[A-Za-z0-9-]+)+"))
615 href = "mailto:" + target;
616 if (href != null)
617 {
618 pieces.Add(new("<a href=\"" + EscapeHtml(href) + "\">" +
619 EscapeHtml(target) + "</a>"));
620 at = close + 1;
621 continue;
622 }
623 }
624 }
625 if (links &&
626 (text.AsSpan(at).StartsWith("http://", StringComparison.Ordinal) ||
627 text.AsSpan(at).StartsWith("https://", StringComparison.Ordinal)) &&
628 (at == 0 || Has(" \n*_~(", text[at - 1])))
629 {
630 int end = at;
631 while (end < text.Length && !Has(" \n<", text[end]))
632 end++;
633 while (end > at && Has(".,:;!?*_~)", text[end - 1]))
634 end--;
635 string url = text[at..end];
636 if (url != "http://" && url != "https://")
637 {
638 string escaped = EscapeHtml(url);
639 pieces.Add(new("<a href=\"" + escaped + "\">" + escaped + "</a>"));
640 at = end;
641 continue;
642 }
643 }
644 if (ch == '[' || ch == '!' && at + 1 < text.Length && text[at + 1] == '[')
645 {
646 if (links && footnotes &&
647 text.AsSpan(at).StartsWith("[^", StringComparison.Ordinal))
648 {
649 int close = text.IndexOf(']', at + 2);
650 if (close >= 0)
651 {
652 string name = text[(at + 2)..close].ToLowerInvariant();
653 if (parser.Footnotes.ContainsKey(name))
654 {
655 pieces.Add(new(FootnoteReference(name)));
656 at = close + 1;
657 continue;
658 }
659 }
660 }
661 bool image = ch == '!';
662 int opening = at + (image ? 1 : 0), closing = ClosingBracket(text, opening);
663 if ((image || links) && closing >= 0)
664 {
665 string label = text[(opening + 1)..closing];
666 Destination? dest = GetDestination(text, closing + 1, label);
667 if (dest != null)
668 {
669 string content = Inline(label, false, false),
670 url = EscapeHtml(dest.Reference.Url);
671 string title =
672 dest.Reference.Title == null
673 ? ""
674 : " title=\"" + EscapeHtml(dest.Reference.Title) + "\"";
675 pieces.Add(new(image ? "<img src=\"" + url + "\" alt=\"" +
676 StripTags(content) + "\"" + title + " />"
677 : "<a href=\"" + url + "\"" + title + ">" +
678 content + "</a>"));
679 at = dest.End;
680 continue;
681 }
682 }
683 }
684 if (Has("*_~", ch))
685 {
686 int end = RunEnd(text, at), length = end - at;
687 char before = at == 0 ? ' ' : text[at - 1],
688 after = end == text.Length ? ' ' : text[end];
689 bool open = !Whitespace(after), close = !Whitespace(before);
690 if (ch == '_' && Alphanumeric(before) && Alphanumeric(after))
691 open = close = false;
692 pieces.Add(ch != '~' && length <= 3 || ch == '~' && length == 2
693 ? new("", ch, length, open, close)
694 : new(text[at..end]));
695 at = end;
696 continue;
697 }
698 pieces.Add(new(EscapeHtml(text.Substring(at, 1))));
699 at++;
700 }
701 var html = new StringBuilder();
702 Emphasis(html, pieces, 0, pieces.Count);
703 return html.ToString();
704 }

712 string RenderHeading(Block block)
713 {
714 string text = block.Text;
715 string? id = null;
716 int explicitId = text.LastIndexOf(" {#", StringComparison.Ordinal);
717 if (text.EndsWith('}') && explicitId >= 0)
718 {
719 string candidate = text[(explicitId + 3)..^ 1];
720 if (candidate.Length > 0 && candidate.All(LabelCharacter))
721 {
722 id = candidate;
723 text = text[..explicitId].TrimEnd(' ');
724 }
725 }
726 if (id == null)
727 {
728 var slug = new StringBuilder();
729 foreach (char original in text)
730 {
731 char c = char.ToLowerInvariant(original);
732 if (LabelCharacter(c))
733 slug.Append(c);
734 else if (Whitespace(c))
735 slug.Append('-');
736 }
737 string baseId = slug.Length == 0 ? "heading" : slug.ToString();
738 id = baseId;
739 for (int suffix = 1; usedIds.Contains(id); suffix++)
740 id = baseId + "-" + suffix;
741 }
742 usedIds.Add(id);
743 string html = Inline(text);
744 headings.Add(new(block.Number, id, StripTags(html)));
745 return "<h" + block.Number + " id=\"" + id + "\">" + html + "</h" + block.Number + ">";
746 }
747 string RenderItem(List<Block> blocks)
748 {
749 var html = new StringBuilder("<li>");
750 for (int i = 0; i < blocks.Count; i++)
751 {
752 Block block = blocks[i];
753 if (block.Kind == Kind.Paragraph)
754 {
755 string text = block.Text;
756 if (i == 0 && text.Length > 4 &&
757 (text.StartsWith("[ ] ") || text.StartsWith("[x] ") ||
758 text.StartsWith("[X] ")))
759 {
760 html.Append(
761 text[1] == ' '
762 ? "<input type=\"checkbox\" disabled=\"\" /> "
763 : "<input type=\"checkbox\" checked=\"\" disabled=\"\" /> ");
764 text = text[4..];
765 }
766 html.Append(Inline(text));
767 }
768 else
769 {
770 if (html[html.Length - 1] != '\n')
771 html.Append('\n');
772 html.Append(Render(block)).Append('\n');
773 }
774 }
775 return html.Append("</li>").ToString();

777 string TableCells(string tag, Block table, List<string> cells)
778 {
779 var html = new StringBuilder();
780 for (int i = 0; i < cells.Count; i++)
781 {
782 string alignment = table.Alignments[i];
783 html.Append('<').Append(tag);
784 if (alignment.Length > 0)
785 html.Append(" style=\"text-align:").Append(alignment).Append('"');
786 html.Append('>').Append(Inline(cells[i])).Append("</").Append(tag).Append(">\n");
787 }
788 return html.ToString();
789 }
790 string Render(Block block)
791 {
792 switch (block.Kind)
793 {
794 case Kind.Heading:
795 return RenderHeading(block);
796 case Kind.Paragraph:
797 return "<p>" + Inline(block.Text) + "</p>";
798 case Kind.Rule:
799 return "<hr />";
800 case Kind.Code: {
801 var html = new StringBuilder("<pre><code");
802 if (block.Text.Length > 0)
803 html.Append(" class=\"language-").Append(EscapeHtml(block.Text)).Append('"');
804 html.Append('>');
805 foreach (string line in block.Lines)
806 html.Append(EscapeHtml(line)).Append('\n');
807 return html.Append("</code></pre>").ToString();
808 }
809 case Kind.Quote:
810 return "<blockquote>\n" + RenderBlocks(block.Children) + "</blockquote>";
811 case Kind.Alert: {
812 string name = block.Text.ToLowerInvariant(),
813 title = char.ToUpperInvariant(name[0]) + name[1..];
814 return "<div class=\"alert alert-" + name + "\">\n<p class=\"alert-title\">" +
815 title + "</p>\n" + RenderBlocks(block.Children) + "</div>";
816 }
817 case Kind.List: {
818 string tag = block.Ordered ? "ol" : "ul";
819 var html = new StringBuilder("<" + tag);
820 if (block.Ordered && block.Number != 1)
821 html.Append(" start=\"").Append(block.Number).Append('"');
822 html.Append(">\n");
823 foreach (List<Block> item in block.Items)
824 html.Append(RenderItem(item)).Append('\n');
825 return html.Append("</").Append(tag).Append('>').ToString();
826 }
827 case Kind.Table: {
828 var html = new StringBuilder("<table>\n<thead>\n<tr>\n");
829 html.Append(TableCells("th", block, block.Lines)).Append("</tr>\n</thead>\n");
830 if (block.Rows.Count > 0)
831 {
832 html.Append("<tbody>\n");
833 foreach (List<string> row in block.Rows)
834 html.Append("<tr>\n")
835 .Append(TableCells("td", block, row))
836 .Append("</tr>\n");
837 html.Append("</tbody>\n");
838 }
839 return html.Append("</table>").ToString();
840 }
841 case Kind.Toc:
842 return TocMarker;
843 default:
844 throw new InvalidOperationException();
845 }
846 }
847 string TocList(List<Heading> entries)
848 {
849 var html = new StringBuilder("<ul>\n");
850 foreach (Heading entry in entries)
851 {
852 html.Append("<li><a href=\"#")
853 .Append(entry.Id)
854 .Append("\">")
855 .Append(entry.Label)
856 .Append("</a>");
857 if (entry.Children.Count > 0)
858 html.Append('\n').Append(TocList(entry.Children)).Append('\n');
859 html.Append("</li>\n");
860 }
861 return html.Append("</ul>").ToString();
862 }
863 string TableOfContents()
864 {
865 if (headings.Count == 0)
866 return "<nav class=\"toc\">\n</nav>";
867 var roots = new List<Heading>();
868 var parents = new List<Heading>();
869 foreach (Heading heading in headings)
870 {
871 while (parents.Count > 0 && parents[^1].Level >= heading.Level)
872 parents.RemoveAt(parents.Count - 1);
873 if (parents.Count == 0)
874 roots.Add(heading);
875 else
876 parents[^1].Children.Add(heading);
877 parents.Add(heading);
878 }
879 return "<nav class=\"toc\">\n" + TocList(roots) + "\n</nav>";
880 }
881 string FootnoteList()
882 {
883 var html = new StringBuilder("<section class=\"footnotes\">\n<ol>\n");
884 for (int i = 0; i < notes.Count; i++)
885 {
886 Note note = notes[i];
887 int number = i + 1;
888 html.Append("<li id=\"fn-")
889 .Append(number)
890 .Append("\">")
891 .Append(Inline(parser.Footnotes[note.Name], true, false));
892 for (int count = 1; count <= note.Count; count++)
893 html.Append(" <a href=\"#")
894 .Append(ReferenceId(number, count))
895 .Append("\">&#8617;</a>");
896 html.Append("</li>\n");
897 }
898 return html.Append("</ol>\n</section>\n").ToString();
899 }
900 public string Document(List<Block> blocks)
901 {
902 string html = RenderBlocks(blocks);
903 html = html.Replace(TocMarker, TableOfContents());
904 return notes.Count == 0 ? html : html + FootnoteList();
905 }
