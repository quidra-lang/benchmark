5function dec(int $n): string {
6 return intdiv($n,100).'.'.str_pad((string)($n%100),2,'0',STR_PAD_LEFT);

8final class Order {
9 public int $remaining, $quantity, $visible=0, $filled=0, $value=0, $priority=0;
10 public string $status='INCOMING';
11 public array $history=['ACCEPT'];
12 public function __construct(public string $id, public string $trader, public string $kind, public string $side,
13 int $q, public ?int $price=null, public ?int $trigger=null, public ?int $peak=null, public ?string $condition=null) {
14 $this->quantity=$this->remaining=$q;
15 }
16 public function active(): bool {
17 return in_array($this->status,['PENDING','OPEN','PARTIAL'],true);
18 }
19 public function resting(): bool {
20 return in_array($this->status,['OPEN','PARTIAL'],true);
21 }
22 public function matches(int $p): bool {
23 return $this->price===null || ($this->side==='BUY' ? $p<=$this->price : $p>=$this->price);
24 }
25 public function triggered(int $p): bool {
26 return $this->side==='BUY' ? $p >= $this->trigger : $p <= $this->trigger;
27 }
28 public function line(): string {
29 return "ORDER {$this->id} {$this->status} {$this->filled}/{$this->quantity}";
30 }
31 public function details(): string {
32 $p=$this->price??'-';
33 $avg=$this->filled===0?'-':dec(intdiv($this->value*100,$this->filled));
34 return "{$this->id} {$this->trader} {$this->kind} {$this->side} $p {$this->status} {$this->filled}/{$this->quantity} avg $avg".($this->trigger===null?'':" trigger {$this->trigger}");
35 }

56 public function __construct(public int $number, public int $price, public int $quantity, public Order $buy, public Order $sell, public Order $taker, public int $buyFee, public int $sellFee) {
57 }

62final class Engine {
63 private array $orders=[],$traders=[],$trades=[];
64 private int $maker=0,$taker=0,$band=0,$priority=0,$tick=1,$lot=1;
65 private bool $halted=false;
66 private function emit(mixed ...$parts): void {
67 echo implode(' ',$parts),"\n";
68 }
69 private function check(bool $bad,string $reason): void {
70 if($bad) throw new CommandError($reason);
71 }
72 private function opposite(string $side): string {
73 return $side==='BUY'?'SELL':'BUY';
74 }
75 private function resting(string $side): array {
76 return array_values(array_filter($this->orders,fn(Order $o)=>$o->side===$side&&$o->resting()));
77 }
78 private function best(string $side): ?Order {
79 $os=$this->resting($side);
80 usort($os,fn(Order $a,Order $b)=>($side==='BUY'?$b->price<=>$a->price:$a->price<=>$b->price)?:$a->priority<=>$b->priority);
81 return $os[0]??null;
82 }
83 private function active(string $name): array {
84 return array_values(array_filter($this->orders,fn(Order $o)=>$o->trader===$name&&$o->active()));
85 }
86 private function register(string $name): Trader {
87 return $this->traders[$name]??=new Trader($name);
88 }
89 private function order(string $id): Order {
90 return $this->orders[$id]??throw new CommandError('unknown order');
91 }
92 private function trader(string $name): Trader {
93 return $this->traders[$name]??throw new CommandError('unknown trader');
94 }
95 private function last(): ?int {
96 return $this->trades===[]?null:$this->trades[array_key_last($this->trades)]->price;
97 }
98 private function exposure(string $name,string $side,?Order $excluded=null): int {
99 $v=$this->traders[$name]->net()*($side==='BUY'?1:-1);
100 foreach($this->active($name) as $o) if($o!==$excluded&&$o->side===$side) $v+=$o->remaining;
101 return $v;
102 }
103 private function risk(string $name,string $side,int $q,?Order $excluded=null): void {
104 $t=$this->traders[$name]??null;
105 if($t!==null&&$t->limit!==null) $this->check($this->exposure($name,$side,$excluded)+$q>$t->limit,'risk limit');
106 }
107 private function checkBand(int $p): void {
108 $l=$this->last();
109 $this->check($this->band>0&&$l!==null&&abs($p-$l)>$this->band,'outside band');
110 }
111 private function levels(string $side): array {
112 $ls=[];
113 foreach($this->resting($side) as $o) {
114 $ls[$o->price]??=[$o->price,0,0];
115 $ls[$o->price][1]+=$o->visible;
116 $ls[$o->price][2]++;
117 }
118 if($side==='BUY') krsort($ls,SORT_NUMERIC);
119 else ksort($ls,SORT_NUMERIC);
120 return array_values($ls);
121 }
122 private function cancel(Order $o): void {
123 $o->history[]="CANCEL {$o->remaining}";
124 $o->remaining=$o->visible=0;
125 $o->status='CANCELLED';
126 $this->emit($o->line());
127 }
128 private function match(Order $o): void {
129 while($o->remaining>0) {
130 $y=$this->best($this->opposite($o->side));
131 if($y===null||!$o->matches($y->price)) break;
132 if($y->trader===$o->trader) {
133 $this->emit('STP',$y->id,$y->remaining);
134 $y->history[]="STP {$y->remaining}";
135 $y->remaining=$y->visible=0;
136 $y->status='CANCELLED';
137 continue;
138 }
139 $q=min($o->remaining,$y->visible);
140 $p=$y->price;
141 [$buy,$sell]=$o->side==='BUY'?[$o,$y]:[$y,$o];
142 [$bf,$sf]=$o->side==='BUY'?[$q*$this->taker,$q*$this->maker]:[$q*$this->maker,$q*$this->taker];
143 $tr=new Trade(count($this->trades)+1,$p,$q,$buy,$sell,$o,$bf,$sf);
144 $this->trades[]=$tr;
145 foreach([$o,$y] as $x) {
146 $x->remaining-=$q;
147 $x->filled+=$q;
148 $x->value+=$q*$p;
149 $x->history[]="FILL {$tr->number} $p $q";
150 }
151 $y->visible-=$q;
152 $bt=$this->traders[$buy->trader];
153 $st=$this->traders[$sell->trader];
154 $bt->bought+=$q;
155 $bt->cash-=$q*$p+$bf;
156 $bt->fees+=$bf;
157 $st->sold+=$q;
158 $st->cash+=$q*$p-$sf;
159 $st->fees+=$sf;
160 $this->emit($tr->line());
161 $y->status=$y->remaining===0?'FILLED':'PARTIAL';
162 if($y->remaining>0&&$y->visible===0) {
163 $y->priority=++$this->priority;
164 $y->visible=min($y->peak,$y->remaining);
165 $h=$y->remaining-$y->visible;
166 $y->history[]="REFRESH {$y->visible} $h";
167 $this->emit('REFRESH',$y->id,$y->visible,$h);
168 }
169 }
170 }
171 private function finish(Order $o,bool $force=false): void {
172 if($o->remaining===0) $o->status='FILLED';
173 elseif($force||($o->kind==='LIMIT'&&$o->condition!=='IOC')||in_array($o->kind,['ICEBERG','STOPLIMIT'],true)) {
174 $o->priority=++$this->priority;
175 $o->visible=min($o->peak??$o->remaining,$o->remaining);
176 $o->status=$o->filled===0?'OPEN':'PARTIAL';
177 $o->history[]="REST {$o->price} {$o->visible}";
178 } else {
179 $o->history[]="EXPIRE {$o->remaining}";
180 $o->remaining=0;
181 $o->status='CANCELLED';
182 }
183 $this->emit($o->line());
184 }
185 private function triggers(): void {
186 while($this->last()!==null) {
187 $last=$this->last();
188 $selected=array_filter($this->orders,fn(Order $o)=>$o->status==='PENDING'&&$o->triggered($last));
189 if($selected===[]) break;
190 foreach($selected as $o) $o->status='TRIGGERING';
191 foreach($selected as $o) {
192 $this->emit('TRIGGER',$o->id);
193 $o->history[]='TRIGGER';
194 $this->match($o);
195 $this->finish($o);
196 }
197 }
198 }
199 private function submit(string $c,array $a): void {
200 [$id,$name,$side]=$a;
201 $p=$r=$peak=$cond=null;
202 if($c==='MARKET') $q=(int)$a[3];
203 elseif($c==='STOP') {
204 $r=(int)$a[3];
205 $q=(int)$a[4];
206 } elseif($c==='STOPLIMIT') {
207 $r=(int)$a[3];
208 $p=(int)$a[4];
209 $q=(int)$a[5];
210 } else {
211 $p=(int)$a[3];
212 $q=(int)$a[4];
213 if($c==='ICEBERG') $peak=(int)$a[5];
214 if($c==='LIMIT') $cond=$a[5]??null;
215 }
216 $prices=array_filter([$r,$p],fn($x)=>$x!==null);
217 $this->check($this->halted,'halted');
218 $this->check(isset($this->orders[$id]),'duplicate id');
219 $this->check(in_array(0,$prices,true),'invalid price');
220 $this->check($q===0,'invalid quantity');
221 foreach($prices as $price) $this->check($price%$this->tick!==0,'invalid tick');
222 $this->check($q%$this->lot!==0,'invalid lot');
223 $o=new Order($id,$name,$c,$side,$q,$p,$r,$peak,$cond);
224 $this->check($peak!==null&&($peak===0||$peak>$q||$peak%$this->lot!==0),'invalid peak');
225 $this->check($r!==null&&$this->last()!==null&&$o->triggered($this->last()),'invalid trigger');
226 if($p!==null) $this->checkBand($p);
227 $this->risk($name,$side,$q);
228 $y=$this->best($this->opposite($side));
229 $this->check($cond==='POSTONLY'&&$y!==null&&$o->matches($y->price),'would cross');
230 $v=0;
231 foreach($this->resting($this->opposite($side)) as $x) if($x->trader!==$name&&$o->matches($x->price)) $v+=$x->remaining;
232 $this->check($cond==='FOK'&&$v<$q,'unfillable');
233 $this->register($name);
234 $this->orders[$id]=$o;
235 if($r!==null) {
236 $o->status='PENDING';
237 $this->emit($o->line());
238 } else {
239 if($cond!=='POSTONLY') $this->match($o);
240 $this->finish($o);
241 $this->triggers();
242 }
243 }
244 private function replace(array $a): void {
245 [$id,$p,$q]=[$a[0],(int)$a[1],(int)$a[2]];
246 $this->check($this->halted,'halted');
247 $o=$this->order($id);
248 $this->check(!$o->resting(),'not open');
249 $this->check($p===0,'invalid price');
250 $this->check($q===0,'invalid quantity');
251 $moved=$p!==$o->price;
252 $this->check($moved&&$p%$this->tick!==0,'invalid tick');
253 $this->check($q%$this->lot!==0,'invalid lot');
254 $this->check(!$moved&&$q===$o->remaining,'no change');
255 if($moved) $this->checkBand($p);
256 if($q>$o->remaining) $this->risk($o->trader,$o->side,$q,$o);
257 $kept=!$moved&&$q<$o->remaining;
258 $o->quantity=$o->filled+$q;
259 $o->remaining=$q;
260 $word=$kept?'kept':'lost';
261 $o->history[]="REPLACE $p $q $word";
262 $this->emit('REPLACE',$id,$word);
263 if($kept) {
264 $o->visible=min($o->visible,$q);
265 $this->emit($o->line());
266 } else {
267 $o->price=$p;
268 $o->status='INCOMING';
269 $this->match($o);
270 $this->finish($o,true);
271 $this->triggers();
272 }
273 }
274 private function bar(array $ts): string {
275 $ps=array_map(fn(Trade $t)=>$t->price,$ts);
276 $v=array_sum(array_map(fn(Trade $t)=>$t->quantity,$ts));
277 return 'open '.$ps[0].' high '.max($ps).' low '.min($ps).' close '.$ps[count($ps)-1].' volume '.$v;
278 }
279 public function execute(string $line): void {
280 $a=explode(' ',trim($line));
281 $c=array_shift($a);
282 try {
283 if(in_array($c,['LIMIT','MARKET','STOP','STOPLIMIT','ICEBERG'],true)) {
284 $this->submit($c,$a);
285 return;
286 }
287 switch($c) {
288 case 'CANCEL':
289 $o=$this->order($a[0]);
290 $this->check(!$o->active(),'not open');
291 $this->cancel($o);
292 break;
293 case 'REPLACE':
294 $this->replace($a);
295 break;
296 case 'CANCELALL':
297 $this->trader($a[0]);
298 $os=$this->active($a[0]);
299 $this->emit($c,$a[0],count($os));
300 foreach($os as $o) $this->cancel($o);
301 break;
302 case 'HALT':
303 case 'RESUME':
304 $this->check($c==='HALT'?$this->halted:!$this->halted,$c==='HALT'?'already halted':'not halted');
305 $this->halted=$c==='HALT';
306 $this->emit($c);
307 break;
308 case 'RISK':
309 $this->register($a[0])->limit=(int)$a[1];
310 $this->emit($c,...[...$a,'exposure buy',$this->exposure($a[0],'BUY'),'sell',$this->exposure($a[0],'SELL')]);
311 break;
312 case 'FEE':
313 [$m,$t]=array_map('intval',$a);
314 $this->check($m+$t<0,'invalid fee');
315 $this->maker=$m;
316 $this->taker=$t;
317 $this->emit($c,'maker',$m,'taker',$t);
318 break;
319 case 'BAND':
320 $this->band=(int)$a[0];
321 $this->emit($c,$this->band);
322 break;
323 case 'TICK':
324 case 'LOT':
325 $n=(int)$a[0];
326 $this->check($n===0,'invalid size');
327 if($c==='TICK') $this->tick=$n;
328 else $this->lot=$n;
329 $this->emit($c,$n);
330 break;
331 case 'BOOK':
332 $n=(int)$a[0];
333 $this->check($n===0,'invalid depth');
334 $bs=$this->levels('BUY');
335 $ss=$this->levels('SELL');
336 $this->emit($c,'bids',count($bs),'asks',count($ss));
337 foreach(array_reverse(array_slice($ss,0,$n)) as $v) $this->emit('ASK',...$v);
338 foreach(array_slice($bs,0,$n) as $v) $this->emit('BID',...$v);
339 break;
340 case 'ORDERS':
341 $this->trader($a[0]);
342 $os=$this->active($a[0]);
343 $this->emit($c,$a[0],count($os));
344 foreach($os as $o) $this->emit($o->details());
345 break;
346 case 'STATUS':
347 $this->emit($c,$this->order($a[0])->details());
348 break;
349 case 'HISTORY':
350 $o=$this->order($a[0]);
351 $this->emit($c,$o->id,count($o->history));
352 foreach($o->history as $h) $this->emit($h);
353 break;
354 case 'POSITION':
355 $this->emit($this->trader($a[0])->line($this->last()??0));
356 break;
357 case 'FILLS':
358 $this->trader($a[0]);
359 $ts=array_filter($this->trades,fn(Trade $t)=>$t->buy->trader===$a[0]||$t->sell->trader===$a[0]);
360 $this->emit($c,$a[0],count($ts));
361 $net=0;
362 foreach($ts as $t) {
363 $b=$t->buy->trader===$a[0];
364 $o=$b?$t->buy:$t->sell;
365 $net+=$b?$t->quantity:-$t->quantity;
366 $this->emit($t->number,$o->side,$t->price,$t->quantity,$o->id,$t->taker===$o?'TAKER':'MAKER','fee',$b?$t->buyFee:$t->sellFee,'net',$net);
367 }
368 break;
369 case 'LAST':
370 if($this->trades===[]) $this->emit($c,'none');
371 else {
372 $t=$this->trades[array_key_last($this->trades)];
373 $this->emit($c,$t->price,$t->quantity);
374 }
375 break;
376 case 'VWAP':
377 if($this->trades===[]) $this->emit($c,'none');
378 else {
379 $v=$amount=0;
380 foreach($this->trades as $t) {
381 $v+=$t->quantity;
382 $amount+=$t->price*$t->quantity;
383 }
384 $this->emit($c,dec(intdiv($amount*100,$v)),'volume',$v,'trades',count($this->trades));
385 }
386 break;
387 case 'SPREAD':
388 $b=$this->best('BUY');
389 $s=$this->best('SELL');
390 $text='SPREAD bid '.($b?->price??'none').' ask '.($s?->price??'none');
391 if($b!==null&&$s!==null) {
392 $n=$b->price+$s->price;
393 $text.=' spread '.($s->price-$b->price).' mid '.intdiv($n,2).'.'.($n%2?5:0);
394 }
395 $this->emit($text);
396 break;
397 case 'IMPACT':
398 $side=$a[0];
399 $q=(int)$a[1];
400 $this->check($q===0,'invalid quantity');
401 $ls=$this->levels($this->opposite($side));
402 $r=$q;
403 $cost=$used=$worst=$total=0;
404 foreach($ls as [$p,$v]) {
405 $total+=$v;
406 if($r===0) continue;
407 $u=min($r,$v);
408 $r-=$u;
409 $cost+=$u*$p;
410 $worst=$p;
411 $used++;
412 }
413 $this->emit($c,$side,$q,...($r>0?['insufficient',$total]:['avg',dec(intdiv($cost*100,$q)),'worst',$worst,'levels',$used]));
414 break;
415 case 'TRADES':
416 case 'BARS':
417 $k=(int)$a[0];
418 $this->check($k===0,'invalid count');
419 if($c==='TRADES') {
420 $ts=array_slice($this->trades,-$k);
421 $this->emit($c,count($ts),'of',count($this->trades));
422 foreach($ts as $t) $this->emit($t->line());
423 } else {
424 $gs=array_chunk($this->trades,$k);
425 $this->emit($c,count($gs));
426 foreach($gs as $i=>$g) $this->emit($i+1,$this->bar($g));
427 }
428 break;
429 case 'RANK':
430 $ts=array_values($this->traders);
431 $mark=$this->last()??0;
432 usort($ts,fn(Trader $x,Trader $y)=>($y->pnl($mark)<=>$x->pnl($mark))?:($y->volume()<=>$x->volume())?:strcmp($x->name,$y->name));
433 $this->emit($c,count($ts));
434 foreach($ts as $i=>$t) $this->emit($i+1,$t->name,'pnl',$t->pnl($mark),'volume',$t->volume());
435 break;
436 case 'FLOW':
437 $b=$s=0;
438 foreach($this->trades as $t) {
439 if($t->taker->side==='BUY') $b+=$t->quantity;
440 else $s+=$t->quantity;
441 }
442 $this->emit($c,'buy',$b,'sell',$s,'net',$b-$s);
443 break;
444 case 'SUMMARY':
445 if($this->trades===[]) $this->emit($c,'trades',0);
446 else {
447 $ps=[];
448 $v=$fees=0;
449 foreach($this->trades as $t) {
450 $ps[]=$t->price;
451 $v+=$t->quantity;
452 $fees+=$t->buyFee+$t->sellFee;
453 }
454 $this->emit($c,'trades',count($ps),'volume',$v,'open',$ps[0],'high',max($ps),'low',min($ps),'close',$ps[count($ps)-1],'fees',$fees);
455 }
456 $names=array_keys($this->traders);
457 sort($names,SORT_STRING);
458 foreach($names as $name) $this->emit($this->traders[$name]->line($this->last()??0));
459 break;
460 }
461 } catch(CommandError $e) {
462 $prefix=in_array($c,['LIMIT','MARKET','STOP','STOPLIMIT','ICEBERG','CANCEL','REPLACE'],true)?['REJECT',$a[0]]:['ERROR'];
463 $this->emit(...[...$prefix,$e->getMessage()]);
464 }
465 }
466}
467$engine=new Engine();
468$n=(int)fgets(STDIN);
469for($i=0;$i<$n;$i++) $engine->execute(fgets(STDIN));
