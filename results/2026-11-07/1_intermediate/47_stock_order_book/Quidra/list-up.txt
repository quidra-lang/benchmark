int minimum(int a, int b)
    return if a < b then a else b
bool resting(const Order order)
    return order.status == "OPEN" or order.status == "PARTIAL"
bool matchable(bool sell, int price, int other)
    return price < 0 or (if sell then other >= price else other <= price)
int best(const Engine engine, bool sell)
    result := -1
    for: index < #engine.orders
        order := engine.orders[index]
        if not resting(order) or order.sell != sell
            continue
        if result < 0
            result = int(index)
        else
            previous := engine.orders[result]
            if (if sell then order.price < previous.price else order.price > previous.price) or (order.price == previous.price and order.priority < previous.priority)
                result = int(index)
    return result
void emit(Engine &engine, string text)
    engine.output = engine.output.append(text)
string trade_line(const Engine engine, int index)
    trade := engine.trades[index]
    buy := engine.orders[trade.buy]
    sell := engine.orders[trade.sell]
    return "TRADE {index + 1} {trade.price} {trade.quantity} {buy.id} {engine.traders[buy.trader].name} {sell.id} {engine.traders[sell.trader].name}"
void execute_order(Engine &engine, int index)
    while engine.orders[index].remaining > 0
        incoming := engine.orders[index]
        opposite := best(engine, not incoming.sell)
        if opposite < 0 or not matchable(incoming.sell, incoming.price, engine.orders[opposite].price)
            break
        maker_order := engine.orders[opposite]
        if incoming.trader == maker_order.trader
            emit(&engine, "STP {maker_order.id} {maker_order.remaining}")
            engine.orders[opposite].remaining = 0
            engine.orders[opposite].visible = 0
            engine.orders[opposite].status = "CANCELLED"
            continue
        quantity := minimum(incoming.remaining, maker_order.visible)
        price := maker_order.price
        buy := if incoming.sell then opposite else index
        sell := if incoming.sell then index else opposite
        buy_fee := quantity * (if buy == index then engine.taker else engine.maker)
        sell_fee := quantity * (if sell == index then engine.taker else engine.maker)
        engine.trades = engine.trades.append(Trade(price, quantity, buy, sell, index, buy_fee, sell_fee))
        number := int(#engine.trades)
        for affected := [index, opposite][]
            engine.orders[affected].remaining -= quantity
        engine.orders[opposite].visible -= quantity
        buyer := engine.orders[buy].trader
        seller := engine.orders[sell].trader
        engine.traders[buyer].fees += buy_fee
        engine.traders[seller].fees += sell_fee
        emit(&engine, trade_line(engine, number - 1))
        if engine.orders[opposite].remaining == 0
            engine.orders[opposite].status = "FILLED"
        else
            engine.orders[opposite].status = "PARTIAL"
            if engine.orders[opposite].visible == 0
                engine.priority += 1
                engine.orders[opposite].priority = engine.priority
                visible := minimum(maker_order.peak, engine.orders[opposite].remaining)
                hidden := engine.orders[opposite].remaining - visible
                engine.orders[opposite].visible = visible
                emit(&engine, "REFRESH {maker_order.id} {visible} {hidden}")
