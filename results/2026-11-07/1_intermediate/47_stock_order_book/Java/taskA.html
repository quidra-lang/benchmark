12 static String decimal(long n) {
13 return n / 100 + "." + String.format(Locale.ROOT, "%02d", n % 100);

15 static class Order {
16 String id, trader, type, side, condition, status = "OPEN";
17 long price, trigger, peak, quantity, left, visible, filled, value, priority;
18 List<String> history = new ArrayList<>(List.of("ACCEPT"));
19 Order(String id, String trader, String type, String side, long q) {
20 this.id=id;
21 this.trader=trader;
22 this.type=type;
23 this.side=side;
24 quantity=left=q;
25 }
26 boolean active() {
27 return resting() || status.equals("PENDING");
28 }
29 boolean resting() {
30 return status.equals("OPEN") || status.equals("PARTIAL");
31 }
32 boolean matches(long p) {
33 return price == 0 || (side.equals("BUY") ? p <= price : p >= price);
34 }
35 boolean triggered(long p) {
36 return side.equals("BUY") ? p >= trigger : p <= trigger;
37 }
38 String line() {
39 return "ORDER "+id+" "+status+" "+filled+"/"+quantity;
40 }
41 String details() {
42 return id+" "+trader+" "+type+" "+side+" "+(price==0?"-":price)+" "+status+" "+filled+"/"+quantity+
43 " avg "+(filled==0?"-":decimal(value*100/filled))+(trigger==0?"":" trigger "+trigger);
44 }
45 void fill(int number, long p, long q) {
46 left-=q;
47 filled+=q;
48 value+=p*q;
49 history.add("FILL "+number+" "+p+" "+q);
50 }
51 }
52 static class Trader {
53 String name;
54 long bought, sold, cash, fees;
55 Long limit;
56 Trader(String name) {
57 this.name=name;
58 }
59 long net() {
60 return bought-sold;
61 }
62 long volume() {
63 return bought+sold;
64 }
65 long pnl(long mark) {
66 return cash+net()*mark;
67 }
68 String position(long mark) {
69 return "POSITION "+name+" bought "+bought+" sold "+sold+" net "+net()+" cash "+cash+" fees "+fees+" pnl "+pnl(mark);
70 }
71 }
72 record Trade(int number, long price, long quantity, Order buy, Order sell, Order taker, long buyFee, long sellFee) {
73 String line() {
74 return "TRADE "+number+" "+price+" "+quantity+" "+buy.id+" "+buy.trader+" "+sell.id+" "+sell.trader;
75 }

84 static class Engine {
85 Map<String,Order> orders = new LinkedHashMap<>();
86 Map<String,Trader> traders = new TreeMap<>();
87 List<Order> pending = new ArrayList<>(), buys = new ArrayList<>(), sells = new ArrayList<>();
88 List<Trade> trades = new ArrayList<>();
89 StringBuilder out = new StringBuilder();
90 long makerFee, takerFee, band, tick=1, lot=1, priority;
91 boolean halted;
92 void emit(Object... parts) {
93 for(int i=0;i<parts.length;i++) {
94 if(i>0) out.append(' ');
95 out.append(parts[i]);
96 }
97 out.append('\n');
98 }
99 List<Order> book(String side) {
100 return side.equals("BUY")?buys:sells;
101 }
102 List<Order> opposite(String side) {
103 return side.equals("BUY")?sells:buys;
104 }
105 Comparator<Order> comparator(String side) {
106 return Comparator.<Order>comparingLong(o -> side.equals("BUY")?-o.price:o.price).thenComparingLong(o -> o.priority);
107 }
108 Order best(List<Order> b) {
109 return b.stream().min(comparator(b==buys?"BUY":"SELL")).orElse(null);
110 }
111 long last() {
112 return trades.isEmpty()?0:trades.getLast().price;
113 }
114 Trader register(String name) {
115 return traders.computeIfAbsent(name, Trader::new);
116 }
117 Trader trader(String name) {
118 check(!traders.containsKey(name),"unknown trader");
119 return traders.get(name);
120 }
121 Order order(String id) {
122 check(!orders.containsKey(id),"unknown order");
123 return orders.get(id);
124 }
125 List<Order> active(String name) {
126 return orders.values().stream().filter(o -> o.trader.equals(name)&&o.active()).toList();
127 }
128 long exposure(String name,String side,Order excluded) {
129 Trader t=traders.get(name);
130 long x=side.equals("BUY")?t.net():-t.net();
131 for(Order o:active(name)) if(o!=excluded&&o.side.equals(side)) x+=o.left;
132 return x;
133 }
134 void risk(String name,String side,long q,Order excluded) {
135 Trader t=traders.get(name);
136 check(t!=null&&t.limit!=null&&exposure(name,side,excluded)+q>t.limit,"risk limit");
137 }
138 void band(long p) {
139 check(band>0&&last()!=0&&Math.abs(p-last())>band,"outside band");
140 }
141 void accept(Order o) {
142 register(o.trader);
143 orders.put(o.id,o);
144 }
145 void match(Order x) {
146 List<Order> b=opposite(x.side);
147 while(x.left>0) {
148 Order y=best(b);
149 if(y==null||!x.matches(y.price)) break;
150 if(y.trader.equals(x.trader)) {
151 emit("STP",y.id,y.left);
152 y.history.add("STP "+y.left);
153 b.remove(y);
154 y.left=0;
155 y.status="CANCELLED";
156 continue;
157 }
158 long q=Math.min(x.left,y.visible),p=y.price;
159 Order buy=x.side.equals("BUY")?x:y,sell=buy==x?y:x;
160 long xf=q*takerFee,yf=q*makerFee;
161 Trade t=new Trade(trades.size()+1,p,q,buy,sell,x,buy==x?xf:yf,sell==x?xf:yf);
162 trades.add(t);
163 x.fill(t.number,p,q);
164 y.fill(t.number,p,q);
165 y.visible-=q;
166 Trader bt=traders.get(buy.trader),st=traders.get(sell.trader);
167 bt.bought+=q;
168 bt.cash-=p*q+t.buyFee;
169 bt.fees+=t.buyFee;
170 st.sold+=q;
171 st.cash+=p*q-t.sellFee;
172 st.fees+=t.sellFee;
173 emit(t.line());
174 if(y.left==0) {
175 b.remove(y);
176 y.status="FILLED";
177 }
178 else {
179 y.status="PARTIAL";
180 if(y.visible==0) {
181 y.priority=++priority;
182 y.visible=Math.min(y.peak,y.left);
183 y.history.add("REFRESH "+y.visible+" "+(y.left-y.visible));
184 emit("REFRESH",y.id,y.visible,y.left-y.visible);
185 }
186 }
187 }
188 }
189 void post(Order o,boolean forceRest) {
190 if(o.left==0) o.status="FILLED";
191 else if(forceRest||o.type.equals("ICEBERG")||o.type.equals("STOPLIMIT")||(o.type.equals("LIMIT")&&!"IOC".equals(o.condition))) {
192 o.priority=++priority;
193 o.visible=o.peak==0?o.left:Math.min(o.peak,o.left);
194 book(o.side).add(o);
195 o.status=o.filled>0?"PARTIAL":"OPEN";
196 o.history.add("REST "+o.price+" "+o.visible);
197 } else {
198 o.history.add("EXPIRE "+o.left);
199 o.left=0;
200 o.status="CANCELLED";
201 }
202 emit(o.line());
203 }
204 void trigger() {
205 while(!trades.isEmpty()) {
206 long p=last();
207 List<Order> batch=pending.stream().filter(o -> o.triggered(p)).toList();
208 if(batch.isEmpty()) return;
209 pending.removeAll(batch);
210 for(Order o:batch) {
211 emit("TRIGGER",o.id);
212 o.history.add("TRIGGER");
213 match(o);
214 post(o,false);
215 }
216 }
217 }
218 void cancel(Order o) {
219 if(o.status.equals("PENDING")) pending.remove(o);
220 else book(o.side).remove(o);
221 o.history.add("CANCEL "+o.left);
222 o.left=0;
223 o.status="CANCELLED";
224 emit(o.line());
225 }
226 void newOrder(String[] a) {
227 String kind=a[0];
228 boolean market=kind.equals("MARKET"),stop=kind.equals("STOP"),sl=kind.equals("STOPLIMIT"),ice=kind.equals("ICEBERG");
229 long p=market||stop?0:Long.parseLong(a[sl?5:4]);
230 long tr=stop||sl?Long.parseLong(a[4]):0;
231 long q=Long.parseLong(a[market?4:sl?6:5]);
232 Order o=new Order(a[1],a[2],kind,a[3],q);
233 o.price=p;
234 o.trigger=tr;
235 if(ice) o.peak=Long.parseLong(a[6]);
236 if(kind.equals("LIMIT")&&a.length==7) o.condition=a[6];
237 check(halted,"halted");
238 check(orders.containsKey(o.id),"duplicate id");
239 check((!market&&!stop&&p==0)||((stop||sl)&&tr==0),"invalid price");
240 check(q==0,"invalid quantity");
241 check(p%tick!=0||tr%tick!=0,"invalid tick");
242 check(q%lot!=0,"invalid lot");
243 if(ice) check(o.peak==0||o.peak>q||o.peak%lot!=0,"invalid peak");
244 if(stop||sl) check(last()!=0&&o.triggered(last()),"invalid trigger");
245 if(p!=0) band(p);
246 risk(o.trader,o.side,q,null);
247 Order b=best(opposite(o.side));
248 if("POSTONLY".equals(o.condition)) check(b!=null&&o.matches(b.price),"would cross");
249 if("FOK".equals(o.condition)) {
250 long total=0;
251 for(Order y:opposite(o.side)) if(!y.trader.equals(o.trader)&&o.matches(y.price)) total+=y.left;
252 check(total<q,"unfillable");
253 }
254 accept(o);
255 if(stop||sl) {
256 o.status="PENDING";
257 pending.add(o);
258 emit(o.line());
259 }
260 else {
261 if(!"POSTONLY".equals(o.condition)) match(o);
262 post(o,false);
263 trigger();
264 }
265 }
266 List<Level> levels(String side) {
267 List<Order> sorted=new ArrayList<>(book(side));
268 sorted.sort(comparator(side));
269 List<Level> result=new ArrayList<>();
270 for(Order o:sorted) {
271 if(result.isEmpty()||result.getLast().price!=o.price) result.add(new Level(o.price));
272 Level l=result.getLast();
273 l.quantity+=o.visible;
274 l.count++;
275 }
276 return result;
277 }
278 long positive(String s,String reason) {
279 long n=Long.parseLong(s);
280 check(n==0,reason);
281 return n;
282 }
283 String bar(List<Trade> ts) {
284 long high=0,low=Long.MAX_VALUE,volume=0;
285 for(Trade t:ts) {
286 high=Math.max(high,t.price);
287 low=Math.min(low,t.price);
288 volume+=t.quantity;
289 }
290 return "open "+ts.getFirst().price+" high "+high+" low "+low+" close "+ts.getLast().price+" volume "+volume;
291 }
292 void execute(String line) {
293 String[] a=line.split(" ");
294 String c=a[0];
295 try {
296 switch(c) {
297 case "LIMIT","MARKET","STOP","STOPLIMIT","ICEBERG" -> newOrder(a);
298 case "CANCEL" -> {
299 Order o=order(a[1]);
300 check(!o.active(),"not open");
301 cancel(o);
302 }
303 case "REPLACE" -> {
304 check(halted,"halted");
305 Order o=order(a[1]);
306 check(!o.resting(),"not open");
307 long p=Long.parseLong(a[2]),q=Long.parseLong(a[3]);
308 boolean moved=p!=o.price;
309 check(p==0,"invalid price");
310 check(q==0,"invalid quantity");
311 check(moved&&p%tick!=0,"invalid tick");
312 check(q%lot!=0,"invalid lot");
313 check(!moved&&q==o.left,"no change");
314 if(moved) band(p);
315 if(q>o.left) risk(o.trader,o.side,q,o);
316 o.quantity=o.filled+q;
317 boolean kept=!moved&&q<o.left;
318 o.history.add("REPLACE "+p+" "+q+" "+(kept?"kept":"lost"));
319 emit("REPLACE",o.id,kept?"kept":"lost");
320 if(kept) {
321 o.left=q;
322 o.visible=Math.min(o.visible,q);
323 emit(o.line());
324 }
325 else {
326 book(o.side).remove(o);
327 o.price=p;
328 o.left=q;
329 match(o);
330 post(o,true);
331 trigger();
332 }
333 }
334 case "CANCELALL" -> {
335 trader(a[1]);
336 List<Order> xs=active(a[1]);
337 emit("CANCELALL",a[1],xs.size());
338 for(Order o:xs) cancel(o);
339 }
340 case "HALT" -> {
341 check(halted,"already halted");
342 halted=true;
343 emit(c);
344 }
345 case "RESUME" -> {
346 check(!halted,"not halted");
347 halted=false;
348 emit(c);
349 }
350 case "RISK" -> {
351 register(a[1]).limit=Long.parseLong(a[2]);
352 emit(c,a[1],a[2],"exposure buy",exposure(a[1],"BUY",null),"sell",exposure(a[1],"SELL",null));
353 }
354 case "FEE" -> {
355 long m=Long.parseLong(a[1]),t=Long.parseLong(a[2]);
356 check(m+t<0,"invalid fee");
357 makerFee=m;
358 takerFee=t;
359 emit("FEE maker",m,"taker",t);
360 }
361 case "BAND" -> {
362 band=Long.parseLong(a[1]);
363 emit(c,band);
364 }
365 case "TICK","LOT" -> {
366 long v=positive(a[1],"invalid size");
367 if(c.equals("TICK")) tick=v;
368 else lot=v;
369 emit(c,v);
370 }
371 case "BOOK" -> {
372 long n=positive(a[1],"invalid depth");
373 List<Level> bid=levels("BUY"),ask=levels("SELL");
374 emit("BOOK bids",bid.size(),"asks",ask.size());
375 for(int i=(int)Math.min(n,ask.size())-1;i>=0;i--) {
376 Level l=ask.get(i);
377 emit("ASK",l.price,l.quantity,l.count);
378 }
379 for(int i=0;i<Math.min(n,bid.size());i++) {
380 Level l=bid.get(i);
381 emit("BID",l.price,l.quantity,l.count);
382 }
383 }
384 case "ORDERS" -> {
385 trader(a[1]);
386 List<Order> xs=active(a[1]);
387 emit(c,a[1],xs.size());
388 for(Order o:xs) emit(o.details());
389 }
390 case "STATUS" -> emit(c,order(a[1]).details());
391 case "HISTORY" -> {
392 Order o=order(a[1]);
393 emit(c,o.id,o.history.size());
394 for(String s:o.history) emit(s);
395 }
396 case "POSITION" -> emit(trader(a[1]).position(last()));
397 case "FILLS" -> {
398 trader(a[1]);
399 List<Trade> ts=trades.stream().filter(t -> t.buy.trader.equals(a[1])||t.sell.trader.equals(a[1])).toList();
400 emit(c,a[1],ts.size());
401 long net=0;
402 for(Trade t:ts) {
403 boolean buy=t.buy.trader.equals(a[1]);
404 Order o=buy?t.buy:t.sell;
405 net+=buy?t.quantity:-t.quantity;
406 emit(t.number,o.side,t.price,t.quantity,o.id,t.taker==o?"TAKER":"MAKER","fee",buy?t.buyFee:t.sellFee,"net",net);
407 }
408 }
409 case "LAST" -> {
410 if(trades.isEmpty()) emit(c,"none");
411 else emit(c,last(),trades.getLast().quantity);
412 }
413 case "VWAP" -> {
414 long v=0,value=0;
415 for(Trade t:trades) {
416 v+=t.quantity;
417 value+=t.price*t.quantity;
418 }
419 if(v==0) emit(c,"none");
420 else emit(c,decimal(value*100/v),"volume",v,"trades",trades.size());
421 }
422 case "SPREAD" -> {
423 Order b=best(buys),s=best(sells);
424 if(b==null||s==null) emit("SPREAD bid",b==null?"none":b.price,"ask",s==null?"none":s.price);
425 else {
426 long sum=b.price+s.price;
427 emit("SPREAD bid",b.price,"ask",s.price,"spread",s.price-b.price,"mid",sum/2+"."+(sum%2==0?0:5));
428 }
429 }
430 case "IMPACT" -> {
431 long q=positive(a[2],"invalid quantity"),left=q,cost=0,worst=0,total=0;
432 int used=0;
433 for(Level l:levels(a[1].equals("BUY")?"SELL":"BUY")) {
434 total+=l.quantity;
435 if(left>0) {
436 long u=Math.min(left,l.quantity);
437 left-=u;
438 cost+=u*l.price;
439 worst=l.price;
440 used++;
441 }
442 }
443 if(left>0) emit(c,a[1],q,"insufficient",total);
444 else emit(c,a[1],q,"avg",decimal(cost*100/q),"worst",worst,"levels",used);
445 }
446 case "TRADES" -> {
447 int n=(int)positive(a[1],"invalid count"),m=Math.min(n,trades.size());
448 emit(c,m,"of",trades.size());
449 for(Trade t:trades.subList(trades.size()-m,trades.size())) emit(t.line());
450 }
451 case "BARS" -> {
452 int k=(int)positive(a[1],"invalid count");
453 emit(c,(trades.size()+k-1)/k);
454 for(int i=0;i<trades.size();i+=k) emit(i/k+1,bar(trades.subList(i,Math.min(i+k,trades.size()))));
455 }
456 case "RANK" -> {
457 List<Trader> ts=new ArrayList<>(traders.values());
458 ts.sort(Comparator.<Trader>comparingLong(t -> -t.pnl(last())).thenComparingLong(t -> -t.volume()).thenComparing(t -> t.name));
459 emit(c,ts.size());
460 for(int i=0;i<ts.size();i++) {
461 Trader t=ts.get(i);
462 emit(i+1,t.name,"pnl",t.pnl(last()),"volume",t.volume());
463 }
464 }
465 case "FLOW" -> {
466 long buy=0,sell=0;
467 for(Trade t:trades) if(t.taker==t.buy) buy+=t.quantity;
468 else sell+=t.quantity;
469 emit("FLOW buy",buy,"sell",sell,"net",buy-sell);
470 }
471 case "SUMMARY" -> {
472 if(trades.isEmpty()) emit("SUMMARY trades 0");
473 else {
474 long volume=0,fees=0;
475 for(Trade t:trades) {
476 volume+=t.quantity;
477 fees+=t.buyFee+t.sellFee;
478 }
479 String b=bar(trades);
480 emit("SUMMARY trades",trades.size(),"volume",volume,b.substring(0,b.lastIndexOf(" volume")),"fees",fees);
481 }
482 for(Trader t:traders.values()) emit(t.position(last()));
483 }
484 }
485 } catch(Rejection e) {
486 if(Set.of("LIMIT","MARKET","STOP","STOPLIMIT","ICEBERG","CANCEL","REPLACE").contains(c)) emit("REJECT",a[1],e.getMessage());
487 else emit("ERROR",e.getMessage());
488 }
489 }
490 }
491 public static void main(String[] args) throws IOException {
492 BufferedReader in=new BufferedReader(new InputStreamReader(System.in));
493 Engine e=new Engine();
494 int n=Integer.parseInt(in.readLine());
495 for(int i=0;i<n;i++) e.execute(in.readLine());
496 System.out.print(e.out);
497 }
