3fn check(good: bool, reason: &str) -> Result<(), String> {
4 if good {
5 Ok(())
6 } else {
7 Err(reason.into())
8 }
9}
10fn decimal(x: i64) -> String {
11 format!("{}.{:02}", x / 100, x % 100)
12}
13fn opposite(s: &str) -> &str {
14 if s == "BUY" {
15 "SELL"
16 } else {
17 "BUY"
18 }
19}
20#[derive(Clone)]
21struct Order {
22 id: String,
23 trader: String,
24 kind: String,
25 side: String,
26 condition: String,
27 status: String,
28 price: Option<i64>,
29 trigger: Option<i64>,
30 peak: Option<i64>,
31 quantity: i64,
32 remaining: i64,
33 visible: i64,
34 filled: i64,
35 value: i64,
36 priority: i64,
37 history: Vec<String>,
38}
39impl Order {
40 fn active(&self) -> bool {
41 matches!(self.status.as_str(), "OPEN" | "PARTIAL" | "PENDING")
42 }
43 fn resting(&self) -> bool {
44 matches!(self.status.as_str(), "OPEN" | "PARTIAL")
45 }
46 fn matches(&self, p: i64) -> bool {
47 self.price.is_none_or(|x| if self.side == "BUY" {
48 p <= x
49 } else {
50 p >= x
51 })
52 }
53 fn triggered(&self, p: i64) -> bool {
54 if self.side == "BUY" {
55 p >= self.trigger.unwrap()
56 } else {
57 p <= self.trigger.unwrap()
58 }
59 }
60 fn line(&self) -> String {
61 format!("ORDER {} {} {}/{}", self.id, self.status, self.filled, self.quantity)
62 }
63 fn details(&self) -> String {
64 let mut s = format!("{} {} {} {} {} {} {}/{} avg {}", self.id, self.trader, self.kind, self.side, self.price.map_or("-".into(), |p| p.to_string()), self.status, self.filled, self.quantity, if self.filled == 0 {
65 "-".into()
66 } else {
67 decimal(self.value * 100 / self.filled)
68 });
69 if let Some(r) = self.trigger {
70 s += &format!(" trigger {}", r);
71 }
72 s
73 }
74 fn fill(&mut self, n: usize, p: i64, q: i64) {
75 self.remaining -= q;
76 self.filled += q;
77 self.value += p * q;
78 self.history.push(format!("FILL {} {} {}", n, p, q));
79 }
80}
81#[derive(Default)]
82struct Trader {
83 bought: i64,
84 sold: i64,
85 cash: i64,
86 fees: i64,
87 limit: Option<i64>,
88}
89impl Trader {
90 fn net(&self) -> i64 {
91 self.bought - self.sold
92 }
93 fn volume(&self) -> i64 {
94 self.bought + self.sold
95 }
96 fn pnl(&self, p: i64) -> i64 {
97 self.cash + self.net() * p
98 }
99 fn position(&self, name: &str, p: i64) -> String {
100 format!("POSITION {} bought {} sold {} net {} cash {} fees {} pnl {}", name, self.bought, self.sold, self.net(), self.cash, self.fees, self.pnl(p))
101 }
102}
103#[derive(Clone)]
104struct Trade {
105 number: usize,
106 price: i64,
107 quantity: i64,
108 buy: usize,
109 sell: usize,
110 taker: usize,
111 buy_fee: i64,

113}
114struct Engine {
115 orders: Vec<Order>,
116 ids: HashMap<String, usize>,
117 traders: BTreeMap<String, Trader>,
118 book: Vec<usize>,
119 pending: Vec<usize>,
120 trades: Vec<Trade>,
121 output: Vec<String>,
122 maker: i64,
123 taker: i64,
124 band: i64,
125 tick: i64,
126 lot: i64,
127 priority: i64,
128 halted: bool,
129}
130impl Engine {
131 fn new() -> Self {
132 Self {
133 orders: vec![], ids: HashMap::new(), traders: BTreeMap::new(), book: vec![], pending: vec![], trades: vec![], output: vec![], maker: 0, taker: 0, band: 0, tick: 1, lot: 1, priority: 0, halted: false
134 }
135 }
136 fn mark(&self) -> i64 {
137 self.trades.last().map_or(0, |t| t.price)
138 }
139 fn order(&self, id: &str) -> Result<usize, String> {
140 self.ids.get(id).copied().ok_or("unknown order".into())
141 }
142 fn trader(&self, name: &str) -> Result<&Trader, String> {
143 self.traders.get(name).ok_or("unknown trader".into())
144 }
145 fn active(&self, name: &str) -> Vec<usize> {
146 self.orders.iter().enumerate().filter(|(_, o)| o.trader == name && o.active()).map(|(i, _)| i).collect()
147 }
148 fn exposure(&self, name: &str, side: &str, exclude: Option<usize>) -> i64 {
149 (if side == "BUY" {
150 self.traders[name].net()
151 } else {
152 -self.traders[name].net()
153 }
154 ) + self.active(name).into_iter().filter(|&i| Some(i) != exclude && self.orders[i].side == side).map(|i| self.orders[i].remaining).sum::<i64>()
155 }
156 fn risk(&self, name: &str, side: &str, q: i64, exclude: Option<usize>) -> Result<(), String> {
157 if let Some(m) = self.traders.get(name).and_then(|t| t.limit) {
158 check(self.exposure(name, side, exclude) + q <= m, "risk limit")?;
159 }
160 Ok(())
161 }
162 fn band(&self, p: i64) -> Result<(), String> {
163 check(self.band == 0 || self.trades.is_empty() || (p - self.mark()).abs() <= self.band, "outside band")
164 }
165 fn sorted(&self, side: &str) -> Vec<usize> {
166 let mut v: Vec<usize> = self.book.iter().copied().filter(|&i| self.orders[i].side == side).collect();
167 v.sort_by_key(|&i| {
168 let o = &self.orders[i];
169 (if side == "BUY" {
170 -o.price.unwrap()
171 } else {
172 o.price.unwrap()
173 }, o.priority)
174 });
175 v
176 }
177 fn levels(&self, side: &str) -> Vec<(i64, i64, usize)> {
178 let mut v: Vec<(i64, i64, usize)> = vec![];
179 for i in self.sorted(side) {
180 let o = &self.orders[i];
181 let p = o.price.unwrap();
182 if v.last().is_none_or(|l| l.0 != p) {
183 v.push((p, 0, 0));
184 }
185 let l = v.last_mut().unwrap();
186 l.1 += o.visible;
187 l.2 += 1;
188 }
189 v
190 }
191 fn trade_line(&self, t: &Trade) -> String {
192 let b = &self.orders[t.buy];
193 let s = &self.orders[t.sell];
194 format!("TRADE {} {} {} {} {} {} {}", t.number, t.price, t.quantity, b.id, b.trader, s.id, s.trader)
195 }
196 fn match_order(&mut self, x: usize) {
197 while self.orders[x].remaining > 0 {
198 let Some(y) = self.sorted(opposite(&self.orders[x].side)).first().copied() else {
199 break;
200 };
201 let p = self.orders[y].price.unwrap();
202 if !self.orders[x].matches(p) {
203 break;
204 }
205 if self.orders[x].trader == self.orders[y].trader {
206 let o = &mut self.orders[y];
207 self.output.push(format!("STP {} {}", o.id, o.remaining));
208 o.history.push(format!("STP {}", o.remaining));
209 o.remaining = 0;
210 o.status = "CANCELLED".into();
211 self.book.retain(|&i| i != y);
212 continue;
213 }
214 let q = self.orders[x].remaining.min(self.orders[y].visible);
215 let (buy, sell) = if self.orders[x].side == "BUY" {
216 (x, y)
217 } else {
218 (y, x)
219 };
220 let bf = q * if buy == x {
221 self.taker
222 } else {
223 self.maker
224 };
225 let sf = q * if sell == x {
226 self.taker
227 } else {
228 self.maker
229 };
230 let t = Trade {
231 number: self.trades.len() + 1, price: p, quantity: q, buy, sell, taker: x, buy_fee: bf, sell_fee: sf
232 };
233 self.orders[x].fill(t.number, p, q);
234 self.orders[y].fill(t.number, p, q);
235 self.orders[y].visible -= q;
236 let b = self.traders.get_mut(&self.orders[buy].trader).unwrap();
237 b.bought += q;
238 b.cash -= p * q + bf;
239 b.fees += bf;
240 let s = self.traders.get_mut(&self.orders[sell].trader).unwrap();
241 s.sold += q;
242 s.cash += p * q - sf;
243 s.fees += sf;
244 self.output.push(self.trade_line(&t));
245 self.trades.push(t);
246 let o = &mut self.orders[y];
247 if o.remaining == 0 {
248 self.book.retain(|&i| i != y);
249 o.status = "FILLED".into();
250 } else {
251 o.status = "PARTIAL".into();
252 if o.visible == 0 {
253 self.priority += 1;
254 o.priority = self.priority;
255 o.visible = o.peak.unwrap().min(o.remaining);
256 o.history.push(format!("REFRESH {} {}", o.visible, o.remaining - o.visible));
257 self.output.push(format!("REFRESH {} {} {}", o.id, o.visible, o.remaining - o.visible));
258 }
259 }
260 }
261 }
262 fn post(&mut self, i: usize, force_rest: bool) {
263 let o = &mut self.orders[i];
264 if o.remaining == 0 {
265 o.status = "FILLED".into();
266 } else if force_rest || matches!(o.kind.as_str(), "ICEBERG" | "STOPLIMIT") || o.kind == "LIMIT" && o.condition != "IOC" {
267 self.priority += 1;
268 o.priority = self.priority;
269 o.visible = o.peak.unwrap_or(o.remaining).min(o.remaining);
270 o.status = if o.filled == 0 {
271 "OPEN"
272 } else {
273 "PARTIAL"
274 }
275 .into();
276 self.book.push(i);
277 o.history.push(format!("REST {} {}", o.price.unwrap(), o.visible));
278 } else {
279 o.history.push(format!("EXPIRE {}", o.remaining));
280 o.remaining = 0;
281 o.status = "CANCELLED".into();
282 }
283 self.output.push(o.line());
284 }
285 fn trigger(&mut self) {
286 while !self.trades.is_empty() {
287 let selected: Vec<usize> = self.pending.iter().copied().filter(|&i| self.orders[i].triggered(self.mark())).collect();
288 if selected.is_empty() {
289 break;
290 }
291 self.pending.retain(|i| !selected.contains(i));
292 for i in selected {
293 self.output.push(format!("TRIGGER {}", self.orders[i].id));
294 self.orders[i].history.push("TRIGGER".into());
295 self.match_order(i);
296 self.post(i, false);
297 }
298 }
299 }
300 fn cancel(&mut self, i: usize) {
301 self.book.retain(|&j| i != j);
302 self.pending.retain(|&j| i != j);
303 let o = &mut self.orders[i];
304 o.history.push(format!("CANCEL {}", o.remaining));
305 o.remaining = 0;
306 o.status = "CANCELLED".into();
307 self.output.push(o.line());
308 }
309 fn bar(&self, ts: &[Trade]) -> String {
310 format!("open {} high {} low {} close {} volume {}", ts[0].price, ts.iter().map(|t| t.price).max().unwrap(), ts.iter().map(|t| t.price).min().unwrap(), ts.last().unwrap().price, ts.iter().map(|t| t.quantity).sum::<i64>())
311 }
312 fn command(&mut self, w: &[&str]) -> Result<(), String> {
313 let c = w[0];
314 let num = |i: usize| w[i].parse::<i64>().unwrap();
315 if matches!(c, "LIMIT" | "MARKET" | "STOP" | "STOPLIMIT" | "ICEBERG") {
316 let q = num(if c == "MARKET" {
317 4
318 } else if c == "STOPLIMIT" {
319 6
320 } else {
321 5
322 });
323 let mut o = Order {
324 id: w[1].into(), trader: w[2].into(), kind: c.into(), side: w[3].into(), condition: if c == "LIMIT" && w.len() == 7 {
325 w[6].into()
326 } else {
327 String::new()
328 }, status: "OPEN".into(), price: None, trigger: None, peak: None, quantity: q, remaining: q, visible: 0, filled: 0, value: 0, priority: 0, history: vec!["ACCEPT".into()]
329 };
330 if matches!(c, "LIMIT" | "ICEBERG") {
331 o.price = Some(num(4));
332 }
333 if matches!(c, "STOP" | "STOPLIMIT") {
334 o.trigger = Some(num(4));
335 }
336 if c == "STOPLIMIT" {
337 o.price = Some(num(5));
338 }
339 if c == "ICEBERG" {
340 o.peak = Some(num(6));
341 }
342 let prices: Vec<i64> = [o.price, o.trigger].into_iter().flatten().collect();
343 check(!self.halted, "halted")?;
344 check(!self.ids.contains_key(&o.id), "duplicate id")?;
345 check(prices.iter().all(|&p| p > 0), "invalid price")?;
346 check(q > 0, "invalid quantity")?;
347 check(prices.iter().all(|p| p % self.tick == 0), "invalid tick")?;
348 check(q % self.lot == 0, "invalid lot")?;
349 if let Some(k) = o.peak {
350 check(k > 0 && k <= q && k % self.lot == 0, "invalid peak")?;
351 }
352 if o.trigger.is_some() {
353 check(self.trades.is_empty() || !o.triggered(self.mark()), "invalid trigger")?;
354 }
355 if let Some(p) = o.price {
356 self.band(p)?;
357 }
358 self.risk(&o.trader, &o.side, q, None)?;
359 let opposite = self.sorted(opposite(&o.side));
360 if o.condition == "POSTONLY" {
361 check(opposite.first().is_none_or(|&i| !o.matches(self.orders[i].price.unwrap())), "would cross")?;
362 }
363 if o.condition == "FOK" {
364 check(opposite.iter().filter(|&&i| self.orders[i].trader != o.trader && o.matches(self.orders[i].price.unwrap())).map(|&i| self.orders[i].remaining).sum::<i64>() >= q, "unfillable")?;
365 }
366 let i = self.orders.len();
367 self.traders.entry(o.trader.clone()).or_default();
368 self.ids.insert(o.id.clone(), i);
369 self.orders.push(o);
370 if self.orders[i].trigger.is_some() {
371 self.orders[i].status = "PENDING".into();
372 self.pending.push(i);
373 self.output.push(self.orders[i].line());
374 } else {
375 if self.orders[i].condition != "POSTONLY" {
376 self.match_order(i);
377 }
378 self.post(i, false);
379 self.trigger();
380 }
381 return Ok(());
382 }
383 match c {
384 "CANCEL" => {
385 let i = self.order(w[1])?;
386 check(self.orders[i].active(), "not open")?;
387 self.cancel(i);
388 }
389 "REPLACE" => {
390 check(!self.halted, "halted")?;
391 let i = self.order(w[1])?;
392 let o = &self.orders[i];
393 check(o.resting(), "not open")?;
394 let p = num(2);
395 let q = num(3);
396 let moved = Some(p) != o.price;
397 check(p > 0, "invalid price")?;
398 check(q > 0, "invalid quantity")?;
399 check(!moved || p % self.tick == 0, "invalid tick")?;
400 check(q % self.lot == 0, "invalid lot")?;
401 check(moved || q != o.remaining, "no change")?;
402 if moved {
403 self.band(p)?;
404 }
405 if q > o.remaining {
406 self.risk(&o.trader, &o.side, q, Some(i))?;
407 }
408 let kept = !moved && q < o.remaining;
409 let o = &mut self.orders[i];
410 o.quantity = o.filled + q;
411 o.remaining = q;
412 if kept {
413 o.visible = o.visible.min(q);
414 o.history.push(format!("REPLACE {} {} kept", p, q));
415 self.output.push(format!("REPLACE {} kept", o.id));
416 self.output.push(o.line());
417 } else {
418 o.price = Some(p);
419 o.history.push(format!("REPLACE {} {} lost", p, q));
420 self.output.push(format!("REPLACE {} lost", o.id));
421 self.book.retain(|&j| j != i);
422 self.match_order(i);
423 self.post(i, true);
424 self.trigger();
425 }
426 }
427 "CANCELALL" => {
428 self.trader(w[1])?;
429 let v = self.active(w[1]);
430 self.output.push(format!("CANCELALL {} {}", w[1], v.len()));
431 for i in v {
432 self.cancel(i);
433 }
434 }
435 "HALT" => {
436 check(!self.halted, "already halted")?;
437 self.halted = true;
438 self.output.push(c.into());
439 }
440 "RESUME" => {
441 check(self.halted, "not halted")?;
442 self.halted = false;
443 self.output.push(c.into());
444 }
445 "RISK" => {
446 self.traders.entry(w[1].into()).or_default().limit = Some(num(2));
447 self.output.push(format!("RISK {} {} exposure buy {} sell {}", w[1], num(2), self.exposure(w[1], "BUY", None), self.exposure(w[1], "SELL", None)));
448 }
449 "FEE" => {
450 check(num(1) + num(2) >= 0, "invalid fee")?;
451 self.maker = num(1);
452 self.taker = num(2);
453 self.output.push(format!("FEE maker {} taker {}", self.maker, self.taker));
454 }
455 "BAND" => {
456 self.band = num(1);
457 self.output.push(format!("BAND {}", self.band));
458 }
459 "TICK" | "LOT" => {
460 check(num(1) > 0, "invalid size")?;
461 if c == "TICK" {
462 self.tick = num(1);
463 } else {
464 self.lot = num(1);
465 }
466 self.output.push(format!("{} {}", c, num(1)));
467 }
468 "BOOK" => {
469 check(num(1) > 0, "invalid depth")?;
470 let n = num(1) as usize;
471 let bids = self.levels("BUY");
472 let asks = self.levels("SELL");
473 self.output.push(format!("BOOK bids {} asks {}", bids.len(), asks.len()));
474 for &(p, q, k) in asks.iter().take(n).collect::<Vec<_>>().into_iter().rev() {
475 self.output.push(format!("ASK {} {} {}", p, q, k));
476 }
477 for (p, q, k) in bids.into_iter().take(n) {
478 self.output.push(format!("BID {} {} {}", p, q, k));
479 }
480 }
481 "ORDERS" => {
482 self.trader(w[1])?;
483 let v = self.active(w[1]);
484 self.output.push(format!("ORDERS {} {}", w[1], v.len()));
485 for i in v {
486 self.output.push(self.orders[i].details());
487 }
488 }
489 "STATUS" => {
490 let i = self.order(w[1])?;
491 self.output.push(format!("STATUS {}", self.orders[i].details()));
492 }
493 "HISTORY" => {
494 let i = self.order(w[1])?;
495 let o = &self.orders[i];
496 self.output.push(format!("HISTORY {} {}", o.id, o.history.len()));
497 self.output.extend(o.history.clone());
498 }
499 "POSITION" => {
500 let s = self.trader(w[1])?.position(w[1], self.mark());
501 self.output.push(s);
502 }
503 "FILLS" => {
504 self.trader(w[1])?;
505 let ts: Vec<Trade> = self.trades.iter().filter(|t| self.orders[t.buy].trader == w[1] || self.orders[t.sell].trader == w[1]).cloned().collect();
506 self.output.push(format!("FILLS {} {}", w[1], ts.len()));
507 let mut net = 0;
508 for t in ts {
509 let buy = self.orders[t.buy].trader == w[1];
510 let i = if buy {
511 t.buy
512 } else {
513 t.sell
514 };
515 net += if buy {
516 t.quantity
517 } else {
518 -t.quantity
519 };
520 self.output.push(format!("{} {} {} {} {} {} fee {} net {}", t.number, if buy {
521 "BUY"
522 } else {
523 "SELL"
524 }, t.price, t.quantity, self.orders[i].id, if t.taker == i {
525 "TAKER"
526 } else {
527 "MAKER"
528 }, if buy {
529 t.buy_fee
530 } else {
531 t.sell_fee
532 }, net));
533 }
534 }
535 "LAST" => self.output.push(self.trades.last().map_or("LAST none".into(), |t| format!("LAST {} {}", t.price, t.quantity))),
536 "VWAP" => {
537 let v: i64 = self.trades.iter().map(|t| t.quantity).sum();
538 let a: i64 = self.trades.iter().map(|t| t.quantity * t.price).sum();
539 self.output.push(if v == 0 {
540 "VWAP none".into()
541 } else {
542 format!("VWAP {} volume {} trades {}", decimal(a * 100 / v), v, self.trades.len())
543 });
544 }
545 "SPREAD" => {
546 let b = self.sorted("BUY").first().map(|&i| self.orders[i].price.unwrap());
547 let a = self.sorted("SELL").first().map(|&i| self.orders[i].price.unwrap());
548 let mut s = format!("SPREAD bid {} ask {}", b.map_or("none".into(), |p| p.to_string()), a.map_or("none".into(), |p| p.to_string()));
549 if let (Some(b), Some(a)) = (b, a) {
550 s += &format!(" spread {} mid {}.{}", a - b, (a + b) / 2, if (a + b) % 2 == 0 {
551 0
552 } else {
553 5
554 });
555 }
556 self.output.push(s);
557 }
558 "IMPACT" => {
559 let q = num(2);
560 check(q > 0, "invalid quantity")?;
561 let ls = self.levels(opposite(w[1]));
562 let (mut rem, mut cost, mut worst, mut used) = (q, 0, 0, 0);
563 for &(p, v, _) in &ls {
564 if rem == 0 {
565 break;
566 }
567 let u = rem.min(v);
568 rem -= u;
569 cost += u * p;
570 worst = p;
571 used += 1;
572 }
573 self.output.push(if rem > 0 {
574 format!("IMPACT {} {} insufficient {}", w[1], q, ls.iter().map(|l| l.1).sum::<i64>())
575 } else {
576 format!("IMPACT {} {} avg {} worst {} levels {}", w[1], q, decimal(cost * 100 / q), worst, used)
577 });
578 }
579 "TRADES" => {
580 check(num(1) > 0, "invalid count")?;
581 let n = (num(1) as usize).min(self.trades.len());
582 self.output.push(format!("TRADES {} of {}", n, self.trades.len()));
583 for t in &self.trades[self.trades.len()-n..] {
584 self.output.push(self.trade_line(t));
585 }
586 }
587 "BARS" => {
588 check(num(1) > 0, "invalid count")?;
589 let k = num(1) as usize;
590 self.output.push(format!("BARS {}", self.trades.len().div_ceil(k)));
591 for (i, ts) in self.trades.chunks(k).enumerate() {
592 self.output.push(format!("{} {}", i + 1, self.bar(ts)));
593 }
594 }
595 "RANK" => {
596 let mut names: Vec<String> = self.traders.keys().cloned().collect();
597 names.sort_by_key(|n| (-self.traders[n].pnl(self.mark()), -self.traders[n].volume(), n.clone()));
598 self.output.push(format!("RANK {}", names.len()));
599 for (i, n) in names.iter().enumerate() {
600 self.output.push(format!("{} {} pnl {} volume {}", i + 1, n, self.traders[n].pnl(self.mark()), self.traders[n].volume()));
601 }
602 }
603 "FLOW" => {
604 let b: i64 = self.trades.iter().filter(|t| t.taker == t.buy).map(|t| t.quantity).sum();
605 let s: i64 = self.trades.iter().filter(|t| t.taker == t.sell).map(|t| t.quantity).sum();
606 self.output.push(format!("FLOW buy {} sell {} net {}", b, s, b - s));
607 }
608 "SUMMARY" => {
609 self.output.push(if self.trades.is_empty() {
610 "SUMMARY trades 0".into()
611 } else {
612 format!("SUMMARY trades {} volume {} open {} high {} low {} close {} fees {}", self.trades.len(), self.trades.iter().map(|t| t.quantity).sum::<i64>(), self.trades[0].price, self.trades.iter().map(|t| t.price).max().unwrap(), self.trades.iter().map(|t| t.price).min().unwrap(), self.mark(), self.trades.iter().map(|t| t.buy_fee + t.sell_fee).sum::<i64>())
613 });
614 for (n, t) in &self.traders {
615 self.output.push(t.position(n, self.mark()));
616 }
617 }
618 _ => unreachable!(),
619 }
620 Ok(())
621 }
622}
623fn main() {
624 let mut input = String::new();
625 io::stdin().read_to_string(&mut input).unwrap();
626 let mut lines = input.lines();
627 let n: usize = lines.next().unwrap().parse().unwrap();
628 let mut e = Engine::new();
629 for line in lines.take(n) {
630 let w: Vec<&str> = line.split(' ').collect();
631 if let Err(reason) = e.command(&w) {
632 let prefix = if matches!(w[0], "LIMIT" | "MARKET" | "STOP" | "STOPLIMIT" | "ICEBERG" | "CANCEL" | "REPLACE") {
633 format!("REJECT {} ", w[1])
634 } else {
635 "ERROR ".into()
636 };
637 e.output.push(prefix + &reason);
638 }
639 }
640 println!("{}", e.output.join("\n"));
641}
