12string num(Integer n) {
13 return to_string(n);

15string average(Integer value, Integer quantity) {
16 Integer c = value * 100 / quantity;
17 return num(c / 100) + "." + (c % 100 < 10 ? "0" : "") + num(c % 100);
18}
19void require(bool bad, const string &message) {
20 if (bad)
21 throw runtime_error(message);
22}
23struct Order {
24 string id, trader, type, side, condition, status = "PENDING";
25 Integer quantity, remaining, price = -1, trigger = -1, peak = -1, visible = 0, filled = 0,
26 value = 0, priority = 0;
27 vector<string> history{"ACCEPT"};
28 bool resting() const {
29 return status == "OPEN" || status == "PARTIAL";
30 }
31 bool active() const {
32 return resting() || status == "PENDING";
33 }
34 bool matches(Integer p) const {
35 return price < 0 || (side == "BUY" ? p <= price : p >= price);
36 }
37 bool triggered(Integer p) const {
38 return side == "BUY" ? p >= trigger : p <= trigger;
39 }
40 string line() const {
41 return "ORDER " + id + " " + status + " " + num(filled) + "/" + num(quantity);
42 }
43 string details() const {
44 return id + " " + trader + " " + type + " " + side + " " + (price < 0 ? "-" : num(price)) +
45 " " + status + " " + num(filled) + "/" + num(quantity) + " avg " +
46 (filled ? average(value, filled) : "-") +
47 (trigger < 0 ? "" : " trigger " + num(trigger));
48 }
49};
50struct Trader {
51 Integer bought = 0, sold = 0, cash = 0, fees = 0, limit = -1;
52 Integer net() const {
53 return bought - sold;
54 }
55 Integer volume() const {
56 return bought + sold;
57 }
58 Integer pnl(Integer mark) const {
59 return cash + net() * mark;
60 }
61 string position(const string &name, Integer mark) const {
62 return "POSITION " + name + " bought " + num(bought) + " sold " + num(sold) + " net " +
63 num(net()) + " cash " + num(cash) + " fees " + num(fees) + " pnl " + num(pnl(mark));
64 }
65};
66struct Trade {
67 Integer price, quantity, buyFee, sellFee;
68 int buy, sell, taker;
69};
70struct Level {
71 Integer price, quantity;
72 int count;
73};
74class Engine {
75 vector<Order> orders;
76 map<string, int> ids;
77 map<string, Trader> traders;
78 vector<Trade> trades;
79 Integer priority = 1, band = 0, tick = 1, lot = 1, makerFee = 0, takerFee = 0;
80 bool halted = false;
81 string other(const string &side) const {
82 return side == "BUY" ? "SELL" : "BUY";
83 }
84 Integer last() const {
85 return trades.empty() ? -1 : trades.back().price;
86 }
87 Integer mark() const {
88 return max<Integer>(0, last());
89 }
90 int order(const string &id) const {
91 auto it = ids.find(id);
92 require(it == ids.end(), "unknown order");
93 return it->second;
94 }
95 Trader &trader(const string &name) {
96 auto it = traders.find(name);
97 require(it == traders.end(), "unknown trader");
98 return it->second;
99 }
100 vector<int> book(const string &side) const {
101 vector<int> selected;
102 for (int i = 0; i < (int)orders.size(); ++i)
103 if (orders[i].side == side && orders[i].resting())
104 selected.push_back(i);
105 sort(selected.begin(), selected.end(), [&](int a, int b) {
106 if (orders[a].price != orders[b].price)
107 return side == "BUY" ? orders[a].price > orders[b].price
108 : orders[a].price < orders[b].price;
109 return orders[a].priority < orders[b].priority;
110 });
111 return selected;
112 }
113 vector<Level> levels(const string &side) const {
114 vector<Level> result;
115 for (int i : book(side)) {
116 const Order &o = orders[i];
117 if (!result.empty() && result.back().price == o.price) {
118 result.back().quantity += o.visible;
119 ++result.back().count;
120 } else
121 result.push_back({o.price, o.visible, 1});
122 }
123 return result;
124 }
125 Integer exposure(const string &name, const string &side, int excluded = -1) {
126 Integer n = side == "BUY" ? trader(name).net() : -trader(name).net();
127 for (int i = 0; i < (int)orders.size(); ++i)
128 if (i != excluded && orders[i].trader == name && orders[i].side == side &&
129 orders[i].active())
130 n += orders[i].remaining;
131 return n;
132 }
133 void risk(const string &name, const string &side, Integer quantity, int excluded = -1) {
134 auto it = traders.find(name);
135 if (it != traders.end() && it->second.limit >= 0)
136 require(exposure(name, side, excluded) + quantity > it->second.limit, "risk limit");
137 }
138 void checkBand(Integer price) const {
139 require(band > 0 && last() >= 0 && abs(price - last()) > band, "outside band");
140 }
141 string tradeLine(int i) const {
142 const Trade &t = trades[i];
143 const Order &b = orders[t.buy];
144 const Order &s = orders[t.sell];
145 return "TRADE " + num(i + 1) + " " + num(t.price) + " " + num(t.quantity) + " " + b.id +
146 " " + b.trader + " " + s.id + " " + s.trader;
147 }
148 void match(int index) {
149 Order &o = orders[index];
150 while (o.remaining) {
151 vector<int> opposite = book(other(o.side));
152 if (opposite.empty())
153 break;
154 int mi = opposite.front();
155 Order &m = orders[mi];
156 if (!o.matches(m.price))
157 break;
158 if (o.trader == m.trader) {
159 cout << "STP " << m.id << ' ' << m.remaining << '\n';
160 m.history.push_back("STP " + num(m.remaining));
161 m.remaining = 0;
162 m.status = "CANCELLED";
163 continue;
164 }
165 Integer q = min(o.remaining, m.visible), p = m.price, tf = q * takerFee,
166 mf = q * makerFee;
167 int buy = o.side == "BUY" ? index : mi, sell = o.side == "SELL" ? index : mi;
168 Trade t{p, q, o.side == "BUY" ? tf : mf, o.side == "SELL" ? tf : mf, buy, sell, index};
169 trades.push_back(t);
170 for (int i : {index, mi}) {
171 Order &f = orders[i];
172 f.remaining -= q;
173 f.filled += q;
174 f.value += p * q;
175 f.history.push_back("FILL " + num(trades.size()) + " " + num(p) + " " + num(q));
176 }
177 m.visible -= q;
178 Trader &bt = traders[orders[buy].trader];
179 Trader &st = traders[orders[sell].trader];
180 bt.bought += q;
181 bt.cash -= p * q + t.buyFee;
182 bt.fees += t.buyFee;
183 st.sold += q;
184 st.cash += p * q - t.sellFee;
185 st.fees += t.sellFee;
186 cout << tradeLine((int)trades.size() - 1) << '\n';
187 m.status = m.remaining ? "PARTIAL" : "FILLED";
188 if (m.remaining && !m.visible) {
189 m.priority = priority++;
190 m.visible = min(m.peak, m.remaining);
191 string h = num(m.visible) + " " + num(m.remaining - m.visible);
192 m.history.push_back("REFRESH " + h);
193 cout << "REFRESH " << m.id << ' ' << h << '\n';
194 }
195 }
196 }
197 void post(int i, bool replaced = false) {
198 Order &o = orders[i];
199 if (!o.remaining)
200 o.status = "FILLED";
201 else if (replaced || o.type == "ICEBERG" || o.type == "STOPLIMIT" ||
202 (o.type == "LIMIT" && o.condition != "IOC")) {
203 o.priority = priority++;
204 o.visible = o.peak < 0 ? o.remaining : min(o.peak, o.remaining);
205 o.status = o.filled ? "PARTIAL" : "OPEN";
206 o.history.push_back("REST " + num(o.price) + " " + num(o.visible));
207 } else {
208 o.history.push_back("EXPIRE " + num(o.remaining));
209 o.remaining = 0;
210 o.status = "CANCELLED";
211 }
212 cout << o.line() << '\n';
213 }
214 void trigger() {
215 while (!trades.empty()) {
216 vector<int> selected;
217 for (int i = 0; i < (int)orders.size(); ++i)
218 if (orders[i].status == "PENDING" && orders[i].triggered(last()))
219 selected.push_back(i);
220 if (selected.empty())
221 break;
222 for (int i : selected)
223 orders[i].status = "TRIGGERING";
224 for (int i : selected) {
225 cout << "TRIGGER " << orders[i].id << '\n';
226 orders[i].history.push_back("TRIGGER");
227 match(i);
228 post(i);
229 }
230 }
231 }
232 void cancel(int i) {
233 Order &o = orders[i];
234 o.history.push_back("CANCEL " + num(o.remaining));
235 o.remaining = 0;
236 o.status = "CANCELLED";
237 cout << o.line() << '\n';
238 }
239 string bar(int first, int end) const {
240 Integer high = trades[first].price, low = high, volume = 0;
241 for (int i = first; i < end; ++i) {
242 high = max(high, trades[i].price);
243 low = min(low, trades[i].price);
244 volume += trades[i].quantity;
245 }
246 return "open " + num(trades[first].price) + " high " + num(high) + " low " + num(low) +
247 " close " + num(trades[end - 1].price) + " volume " + num(volume);
248 }

250 public:
251 void execute(const string &line) {
252 istringstream input(line);
253 vector<string> a;
254 string c, word;
255 input >> c;
256 while (input >> word)
257 a.push_back(word);
258 try {
259 command(c, a);
260 } catch (const runtime_error &e) {
261 if (c == "LIMIT" || c == "MARKET" || c == "STOP" || c == "STOPLIMIT" ||
262 c == "ICEBERG" || c == "CANCEL" || c == "REPLACE")
263 cout << "REJECT " << a[0] << ' ' << e.what() << '\n';
264 else
265 cout << "ERROR " << e.what() << '\n';
266 }
267 }
268 void command(const string &c, const vector<string> &a) {
269 if (c == "LIMIT" || c == "MARKET" || c == "STOP" || c == "STOPLIMIT" || c == "ICEBERG") {
270 Order o;
271 o.id = a[0];
272 o.trader = a[1];
273 o.type = c;
274 o.side = a[2];
275 o.price = c == "MARKET" || c == "STOP" ? -1 : stoll(a[c == "STOPLIMIT" ? 4 : 3]);
276 o.trigger = c == "STOP" || c == "STOPLIMIT" ? stoll(a[3]) : -1;
277 o.quantity = o.remaining = stoll(a[c == "MARKET" ? 3 : c == "STOPLIMIT" ? 5 : 4]);
278 if (c == "ICEBERG")
279 o.peak = stoll(a[5]);
280 if (c == "LIMIT" && a.size() == 6)
281 o.condition = a[5];
282 require(halted, "halted");
283 require(ids.count(o.id), "duplicate id");
284 require(o.price == 0 || o.trigger == 0, "invalid price");
285 require(!o.quantity, "invalid quantity");
286 require((o.price >= 0 && o.price % tick) || (o.trigger >= 0 && o.trigger % tick),
287 "invalid tick");
288 require(o.quantity % lot, "invalid lot");
289 if (c == "ICEBERG")
290 require(o.peak == 0 || o.peak > o.quantity || o.peak % lot, "invalid peak");
291 if (o.trigger >= 0)
292 require(last() >= 0 && o.triggered(last()), "invalid trigger");
293 if (o.price >= 0)
294 checkBand(o.price);
295 risk(o.trader, o.side, o.quantity);
296 vector<int> opposite = book(other(o.side));
297 require(o.condition == "POSTONLY" && !opposite.empty() &&
298 o.matches(orders[opposite[0]].price),
299 "would cross");
300 if (o.condition == "FOK") {
301 Integer available = 0;
302 for (int i : opposite)
303 if (orders[i].trader != o.trader && o.matches(orders[i].price))
304 available += orders[i].remaining;
305 require(available < o.quantity, "unfillable");
306 }
307 traders.try_emplace(o.trader);
308 int i = (int)orders.size();
309 ids[o.id] = i;
310 orders.push_back(o);
311 if (o.trigger >= 0)
312 cout << o.line() << '\n';
313 else {
314 if (o.condition != "POSTONLY")
315 match(i);
316 post(i);
317 trigger();
318 }
319 return;
320 }
321 const string id = a.empty() ? "" : a[0];
322 if (c == "CANCEL") {
323 int i = order(id);
324 require(!orders[i].active(), "not open");
325 cancel(i);
326 } else if (c == "REPLACE") {
327 require(halted, "halted");
328 int i = order(id);
329 Order &o = orders[i];
330 require(!o.resting(), "not open");
331 Integer p = stoll(a[1]), q = stoll(a[2]);
332 bool moved = p != o.price;
333 require(!p, "invalid price");
334 require(!q, "invalid quantity");
335 require(moved && p % tick, "invalid tick");
336 require(q % lot, "invalid lot");
337 require(!moved && q == o.remaining, "no change");
338 if (moved)
339 checkBand(p);
340 if (q > o.remaining)
341 risk(o.trader, o.side, q, i);
342 bool kept = !moved && q < o.remaining;
343 o.quantity = o.filled + q;
344 o.remaining = q;
345 o.price = p;
346 o.history.push_back("REPLACE " + num(p) + " " + num(q) + (kept ? " kept" : " lost"));
347 cout << "REPLACE " << id << (kept ? " kept\n" : " lost\n");
348 if (kept) {
349 o.visible = min(o.visible, q);
350 cout << o.line() << '\n';
351 } else {
352 o.status = "EXECUTING";
353 match(i);
354 post(i, true);
355 trigger();
356 }
357 } else if (c == "CANCELALL") {
358 trader(id);
359 vector<int> selected;
360 for (int i = 0; i < (int)orders.size(); ++i)
361 if (orders[i].trader == id && orders[i].active())
362 selected.push_back(i);
363 cout << c << ' ' << id << ' ' << selected.size() << '\n';
364 for (int i : selected)
365 cancel(i);
366 } else if (c == "HALT") {
367 require(halted, "already halted");
368 halted = true;
369 cout << c << '\n';
370 } else if (c == "RESUME") {
371 require(!halted, "not halted");
372 halted = false;
373 cout << c << '\n';
374 } else if (c == "RISK") {
375 traders[id].limit = stoll(a[1]);
376 cout << c << ' ' << id << ' ' << a[1] << " exposure buy " << exposure(id, "BUY")
377 << " sell " << exposure(id, "SELL") << '\n';
378 } else if (c == "FEE") {
379 Integer m = stoll(a[0]), t = stoll(a[1]);
380 require(m + t < 0, "invalid fee");
381 makerFee = m;
382 takerFee = t;
383 cout << "FEE maker " << m << " taker " << t << '\n';
384 } else if (c == "BAND") {
385 band = stoll(id);
386 cout << c << ' ' << id << '\n';
387 } else if (c == "TICK" || c == "LOT") {
388 Integer value = stoll(id);
389 require(!value, "invalid size");
390 (c == "TICK" ? tick : lot) = value;
391 cout << c << ' ' << value << '\n';
392 } else if (c == "BOOK") {
393 int n = stoi(id);
394 require(!n, "invalid depth");
395 vector<Level> bids = levels("BUY"), asks = levels("SELL");
396 cout << "BOOK bids " << bids.size() << " asks " << asks.size() << '\n';
397 for (int i = min(n, (int)asks.size()) - 1; i >= 0; --i)
398 cout << "ASK " << asks[i].price << ' ' << asks[i].quantity << ' ' << asks[i].count
399 << '\n';
400 for (int i = 0; i < min(n, (int)bids.size()); ++i)
401 cout << "BID " << bids[i].price << ' ' << bids[i].quantity << ' ' << bids[i].count
402 << '\n';
403 } else if (c == "ORDERS") {
404 trader(id);
405 vector<int> selected;
406 for (int i = 0; i < (int)orders.size(); ++i)
407 if (orders[i].trader == id && orders[i].active())
408 selected.push_back(i);
409 cout << c << ' ' << id << ' ' << selected.size() << '\n';
410 for (int i : selected)
411 cout << orders[i].details() << '\n';
412 } else if (c == "STATUS") {
413 const Order &current = orders[order(id)];
414 cout << "STATUS " << current.details() << '\n';
415 } else if (c == "HISTORY") {
416 const Order &o = orders[order(id)];
417 cout << c << ' ' << id << ' ' << o.history.size() << '\n';
418 for (const string &h : o.history)
419 cout << h << '\n';
420 } else if (c == "POSITION")
421 cout << trader(id).position(id, mark()) << '\n';
422 else if (c == "FILLS") {
423 trader(id);
424 vector<int> selected;
425 for (int i = 0; i < (int)trades.size(); ++i)
426 if (orders[trades[i].buy].trader == id || orders[trades[i].sell].trader == id)
427 selected.push_back(i);
428 cout << c << ' ' << id << ' ' << selected.size() << '\n';
429 Integer net = 0;
430 for (int i : selected) {
431 const Trade &t = trades[i];
432 bool buy = orders[t.buy].trader == id;
433 int oi = buy ? t.buy : t.sell;
434 net += buy ? t.quantity : -t.quantity;
435 cout << i + 1 << ' ' << (buy ? "BUY" : "SELL") << ' ' << t.price << ' '
436 << t.quantity << ' ' << orders[oi].id << ' '
437 << (oi == t.taker ? "TAKER" : "MAKER") << " fee "
438 << (buy ? t.buyFee : t.sellFee) << " net " << net << '\n';
439 }
440 } else if (c == "LAST") {
441 if (trades.empty())
442 cout << "LAST none\n";
443 else
444 cout << "LAST " << last() << ' ' << trades.back().quantity << '\n';
445 } else if (c == "VWAP") {
446 if (trades.empty())
447 cout << "VWAP none\n";
448 else {
449 Integer volume = 0, value = 0;
450 for (const Trade &t : trades) {
451 volume += t.quantity;
452 value += t.quantity * t.price;
453 }
454 cout << "VWAP " << average(value, volume) << " volume " << volume << " trades "
455 << trades.size() << '\n';
456 }
457 } else if (c == "SPREAD") {
458 vector<int> b = book("BUY"), s = book("SELL");
459 cout << "SPREAD bid " << (b.empty() ? "none" : num(orders[b[0]].price)) << " ask "
460 << (s.empty() ? "none" : num(orders[s[0]].price));
461 if (!b.empty() && !s.empty()) {
462 Integer bp = orders[b[0]].price, sp = orders[s[0]].price;
463 cout << " spread " << sp - bp << " mid " << (sp + bp) / 2
464 << ((sp + bp) % 2 ? ".5" : ".0");
465 }
466 cout << '\n';
467 } else if (c == "IMPACT") {
468 Integer q = stoll(a[1]);
469 require(!q, "invalid quantity");
470 vector<Level> ls = levels(other(id));
471 Integer r = q, value = 0, worst = 0, total = 0;
472 int used = 0;
473 for (const Level &l : ls) {
474 total += l.quantity;
475 if (!r)
476 continue;
477 Integer n = min(r, l.quantity);
478 r -= n;
479 value += n * l.price;
480 worst = l.price;
481 ++used;
482 }
483 cout << c << ' ' << id << ' ' << q;
484 if (r)
485 cout << " insufficient " << total;
486 else
487 cout << " avg " << average(value, q) << " worst " << worst << " levels " << used;
488 cout << '\n';
489 } else if (c == "TRADES") {
490 int k = stoi(id);
491 require(!k, "invalid count");
492 int n = min(k, (int)trades.size());
493 cout << "TRADES " << n << " of " << trades.size() << '\n';
494 for (int i = (int)trades.size() - n; i < (int)trades.size(); ++i)
495 cout << tradeLine(i) << '\n';
496 } else if (c == "BARS") {
497 int k = stoi(id);
498 require(!k, "invalid count");
499 cout << "BARS " << (trades.size() + k - 1) / k << '\n';
500 for (int i = 0; i < (int)trades.size(); i += k)
501 cout << i / k + 1 << ' ' << bar(i, min(i + k, (int)trades.size())) << '\n';
502 } else if (c == "RANK") {
503 vector<string> names;
504 for (const auto &entry : traders)
505 names.push_back(entry.first);
506 sort(names.begin(), names.end(), [&](const string &a, const string &b) {
507 const Trader &x = traders[a];
508 const Trader &y = traders[b];
509 if (x.pnl(mark()) != y.pnl(mark()))
510 return x.pnl(mark()) > y.pnl(mark());
511 if (x.volume() != y.volume())
512 return x.volume() > y.volume();
513 return a < b;
514 });
515 cout << "RANK " << names.size() << '\n';
516 for (int i = 0; i < (int)names.size(); ++i)
517 cout << i + 1 << ' ' << names[i] << " pnl " << traders[names[i]].pnl(mark())
518 << " volume " << traders[names[i]].volume() << '\n';
519 } else if (c == "FLOW") {
520 Integer b = 0, s = 0;
521 for (const Trade &t : trades)
522 (orders[t.taker].side == "BUY" ? b : s) += t.quantity;
523 cout << "FLOW buy " << b << " sell " << s << " net " << b - s << '\n';
524 } else if (c == "SUMMARY") {
525 if (trades.empty())
526 cout << "SUMMARY trades 0\n";
527 else {
528 Integer fees = 0;
529 for (const Trade &t : trades)
530 fees += t.buyFee + t.sellFee;
531 string values = bar(0, (int)trades.size());
532 size_t at = values.rfind(" volume ");
533 cout << "SUMMARY trades " << trades.size() << " volume " << values.substr(at + 8)
534 << ' ' << values.substr(0, at) << " fees " << fees << '\n';
535 }
536 for (const auto &entry : traders)
537 cout << entry.second.position(entry.first, mark()) << '\n';
538 }
539 }
540};
