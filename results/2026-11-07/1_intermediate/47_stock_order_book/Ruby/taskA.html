1class CommandError < StandardError
2end
3Order = Struct.new(:id, :trader, :kind, :side, :price, :trigger, :peak, :condition,
4 :quantity, :remaining, :visible, :filled, :value, :priority, :status, :history, keyword_init: true) do
5 def active? = %w[PENDING OPEN PARTIAL].include?(status)
6 def resting? = %w[OPEN PARTIAL].include?(status)
7 def matches?(p) = price.nil? || (side == 'BUY' ? p <= price : p >= price)
8 def triggered?(p) = side == 'BUY' ? p >= trigger : p <= trigger
9 def line = "ORDER #{id} #{status} #{filled}/#{quantity}"
10 def details
11 text = "#{id} #{trader} #{kind} #{side} #{price || '-'} #{status} #{filled}/#{quantity} avg #{filled.zero? ? '-' : decimals(value * 100 / filled)}"
12 trigger ? "#{text} trigger #{trigger}" : text
13 end
14end
15Trader = Struct.new(:name, :bought, :sold, :cash, :fees, :limit) do
16 def net = bought - sold
17 def volume = bought + sold
18 def pnl(mark) = cash + net * mark
19 def line(mark) = "POSITION #{name} bought #{bought} sold #{sold} net #{net} cash #{cash} fees #{fees} pnl #{pnl(mark)}"
20end
21Trade = Struct.new(:number, :price, :quantity, :buy, :sell, :taker, :buy_fee, :sell_fee) do
22 def line = "TRADE #{number} #{price} #{quantity} #{buy.id} #{buy.trader} #{sell.id} #{sell.trader}"
23end
24def decimals(n) = "#{n / 100}.#{(n % 100).to_s.rjust(2, '0')}"
25class Engine
26 def initialize
27 @orders, @traders, @trades = {}, {}, []
28 @maker = @taker = @band = @priority = 0
29 @tick = @lot = 1
30 @halted = false
31 end
32 def emit(*parts) = puts(parts.join(' '))
33 def fail_if(condition, reason) = (raise CommandError, reason if condition)
34 def opposite(side) = side == 'BUY' ? 'SELL' : 'BUY'
35 def resting(side) = @orders.values.select { |o| o.side == side && o.resting? }
36 def best(side) = resting(side).min_by { |o| [side == 'BUY' ? -o.price : o.price, o.priority] }
37 def active(name) = @orders.values.select { |o| o.trader == name && o.active? }
38 def register(name) = @traders[name] ||= Trader.new(name, 0, 0, 0, 0, nil)
39 def order(id) = @orders.fetch(id) { raise CommandError, 'unknown order' }
40 def trader(name) = @traders.fetch(name) { raise CommandError, 'unknown trader' }
41 def last = @trades.last&.price
42 def exposure(name, side, excluded = nil)
43 t = @traders[name]
44 (side == 'BUY' ? t.net : -t.net) + active(name).sum { |o| o.side == side && o != excluded ? o.remaining : 0 }
45 end
46 def risk(name, side, quantity, excluded = nil)
47 t = @traders[name]
48 fail_if(t && t.limit && exposure(name, side, excluded) + quantity > t.limit, 'risk limit')
49 end
50 def band(price) = fail_if(@band > 0 && last && (price - last).abs > @band, 'outside band')
51 def levels(side)
52 resting(side).group_by(&:price).sort_by { |p, _| side == 'BUY' ? -p : p }.map { |p, orders| [p, orders.sum(&:visible), orders.length] }
53 end
54 def cancel(o)
55 o.history << "CANCEL #{o.remaining}"
56 o.remaining, o.visible, o.status = 0, 0, 'CANCELLED'
57 emit(o.line)
58 end
59 def match(o)
60 while o.remaining > 0
61 y = best(opposite(o.side))
62 break unless y && o.matches?(y.price)
63 if y.trader == o.trader
64 emit('STP', y.id, y.remaining)
65 y.history << "STP #{y.remaining}"
66 y.remaining, y.visible, y.status = 0, 0, 'CANCELLED'
67 next
68 end
69 q, p = [o.remaining, y.visible].min, y.price
70 buy, sell = o.side == 'BUY' ? [o, y] : [y, o]
71 bf, sf = o.side == 'BUY' ? [q * @taker, q * @maker] : [q * @maker, q * @taker]
72 tr = Trade.new(@trades.length + 1, p, q, buy, sell, o, bf, sf)
73 @trades << tr
74 [o, y].each do |x|
75 x.remaining -= q
76 x.filled += q
77 x.value += q * p
78 x.history << "FILL #{tr.number} #{p} #{q}"
79 end
80 y.visible -= q
81 bt, st = @traders[buy.trader], @traders[sell.trader]
82 bt.bought += q
83 bt.cash -= q * p + bf
84 bt.fees += bf
85 st.sold += q
86 st.cash += q * p - sf
87 st.fees += sf
88 emit(tr.line)
89 y.status = y.remaining.zero? ? 'FILLED' : 'PARTIAL'
90 if y.remaining > 0 && y.visible.zero?
91 @priority += 1
92 y.priority = @priority
93 y.visible = [y.peak, y.remaining].min
94 h = y.remaining - y.visible
95 y.history << "REFRESH #{y.visible} #{h}"
96 emit('REFRESH', y.id, y.visible, h)
97 end
98 end
99 end
100 def finish(o, force = false)
101 if o.remaining.zero?
102 o.status = 'FILLED'
103 elsif force || (o.kind == 'LIMIT' && o.condition != 'IOC') || %w[ICEBERG STOPLIMIT].include?(o.kind)
104 @priority += 1
105 o.priority = @priority
106 o.visible = [o.peak || o.remaining, o.remaining].min
107 o.status = o.filled.zero? ? 'OPEN' : 'PARTIAL'
108 o.history << "REST #{o.price} #{o.visible}"
109 else
110 o.history << "EXPIRE #{o.remaining}"
111 o.remaining, o.status = 0, 'CANCELLED'
112 end
113 emit(o.line)
114 end
115 def triggers
116 loop do
117 break unless last
118 selected = @orders.values.select { |o| o.status == 'PENDING' && o.triggered?(last) }
119 break if selected.empty?
120 selected.each { |o| o.status = 'TRIGGERING' }
121 selected.each do |o|
122 emit('TRIGGER', o.id)
123 o.history << 'TRIGGER'
124 match(o)
125 finish(o)
126 end
127 end
128 end
129 def submit(command, a)
130 id, name, side = a[0, 3]
131 p = r = peak = cond = nil
132 case command
133 when 'MARKET'
134 q = a[3].to_i
135 when 'STOP'
136 r, q = a[3, 2].map(&:to_i)
137 when 'STOPLIMIT'
138 r, p, q = a[3, 3].map(&:to_i)
139 else
140 p, q = a[3, 2].map(&:to_i)
141 peak = a[5].to_i if command == 'ICEBERG'
142 cond = a[5] if command == 'LIMIT'
143 end
144 prices = [r, p].compact
145 fail_if(@halted, 'halted')
146 fail_if(@orders.key?(id), 'duplicate id')
147 fail_if(prices.include?(0), 'invalid price')
148 fail_if(q.zero?, 'invalid quantity')
149 fail_if(prices.any? { |x| x % @tick != 0 }, 'invalid tick')
150 fail_if(q % @lot != 0, 'invalid lot')
151 o = Order.new(id: id, trader: name, kind: command, side: side, price: p, trigger: r, peak: peak, condition: cond,
152 quantity: q, remaining: q, visible: 0, filled: 0, value: 0, priority: 0, status: 'INCOMING', history: ['ACCEPT'])
153 fail_if(peak && (peak.zero? || peak > q || peak % @lot != 0), 'invalid peak')
154 fail_if(r && last && o.triggered?(last), 'invalid trigger')
155 band(p) if p
156 risk(name, side, q)
157 y = best(opposite(side))
158 fail_if(cond == 'POSTONLY' && y && o.matches?(y.price), 'would cross')
159 available = resting(opposite(side)).sum { |x| x.trader != name && o.matches?(x.price) ? x.remaining : 0 }
160 fail_if(cond == 'FOK' && available < q, 'unfillable')
161 register(name)
162 @orders[id] = o
163 if r
164 o.status = 'PENDING'
165 emit(o.line)
166 else
167 match(o) unless cond == 'POSTONLY'
168 finish(o)
169 triggers
170 end
171 end
172 def replace(a)
173 id, p, q = a[0], a[1].to_i, a[2].to_i
174 fail_if(@halted, 'halted')
175 o = order(id)
176 fail_if(!o.resting?, 'not open')
177 fail_if(p.zero?, 'invalid price')
178 fail_if(q.zero?, 'invalid quantity')
179 moved = p != o.price
180 fail_if(moved && p % @tick != 0, 'invalid tick')
181 fail_if(q % @lot != 0, 'invalid lot')
182 fail_if(!moved && q == o.remaining, 'no change')
183 band(p) if moved
184 risk(o.trader, o.side, q, o) if q > o.remaining
185 kept = !moved && q < o.remaining
186 o.quantity = o.filled + q
187 o.remaining = q
188 o.history << "REPLACE #{p} #{q} #{kept ? 'kept' : 'lost'}"
189 emit('REPLACE', id, kept ? 'kept' : 'lost')
190 if kept
191 o.visible = [o.visible, q].min
192 emit(o.line)
193 else
194 o.price = p
195 o.status = 'INCOMING'
196 match(o)
197 finish(o, true)
198 triggers
199 end
200 end
201 def bar(group)
202 ps = group.map(&:price)
203 "open #{ps.first} high #{ps.max} low #{ps.min} close #{ps.last} volume #{group.sum(&:quantity)}"
204 end
205 def execute(line)
206 c, *a = line.split
207 if %w[LIMIT MARKET STOP STOPLIMIT ICEBERG].include?(c)
208 submit(c, a)
209 return
210 end
211 case c
212 when 'CANCEL'
213 o = order(a[0])
214 fail_if(!o.active?, 'not open')
215 cancel(o)
216 when 'REPLACE'
217 replace(a)
218 when 'CANCELALL'
219 trader(a[0])
220 os = active(a[0])
221 emit(c, a[0], os.length)
222 os.each { |o| cancel(o) }
223 when 'HALT', 'RESUME'
224 fail_if(c == 'HALT' ? @halted : !@halted, c == 'HALT' ? 'already halted' : 'not halted')
225 @halted = c == 'HALT'
226 emit(c)
227 when 'RISK'
228 register(a[0]).limit = a[1].to_i
229 emit(c, *a, 'exposure buy', exposure(a[0], 'BUY'), 'sell', exposure(a[0], 'SELL'))
230 when 'FEE'
231 m, t = a.map(&:to_i)
232 fail_if(m + t < 0, 'invalid fee')
233 @maker, @taker = m, t
234 emit(c, 'maker', m, 'taker', t)
235 when 'BAND'
236 @band = a[0].to_i
237 emit(c, @band)
238 when 'TICK', 'LOT'
239 n = a[0].to_i
240 fail_if(n.zero?, 'invalid size')
241 c == 'TICK' ? @tick = n : @lot = n
242 emit(c, n)
243 when 'BOOK'
244 n = a[0].to_i
245 fail_if(n.zero?, 'invalid depth')
246 bids, asks = levels('BUY'), levels('SELL')
247 emit(c, 'bids', bids.length, 'asks', asks.length)
248 asks.take(n).reverse_each { |v| emit('ASK', *v) }
249 bids.take(n).each { |v| emit('BID', *v) }
250 when 'ORDERS'
251 trader(a[0])
252 os = active(a[0])
253 emit(c, a[0], os.length)
254 os.each { |o| emit(o.details) }
255 when 'STATUS'
256 emit(c, order(a[0]).details)
257 when 'HISTORY'
258 o = order(a[0])
259 emit(c, o.id, o.history.length)
260 o.history.each { |h| emit(h) }
261 when 'POSITION'
262 emit(trader(a[0]).line(last || 0))
263 when 'FILLS'
264 trader(a[0])
265 ts = @trades.select { |t| [t.buy.trader, t.sell.trader].include?(a[0]) }
266 emit(c, a[0], ts.length)
267 net = 0
268 ts.each do |t|
269 buy = t.buy.trader == a[0]
270 o = buy ? t.buy : t.sell
271 net += buy ? t.quantity : -t.quantity
272 emit(t.number, o.side, t.price, t.quantity, o.id, t.taker == o ? 'TAKER' : 'MAKER', 'fee', buy ? t.buy_fee : t.sell_fee, 'net', net)
273 end
274 when 'LAST'
275 emit(c, *(@trades.empty? ? ['none'] : [last, @trades.last.quantity]))
276 when 'VWAP'
277 if @trades.empty?
278 emit(c, 'none')
279 else
280 v = @trades.sum(&:quantity)
281 emit(c, decimals(@trades.sum { |t| t.price * t.quantity } * 100 / v), 'volume', v, 'trades', @trades.length)
282 end
283 when 'SPREAD'
284 b, s = best('BUY'), best('SELL')
285 text = "SPREAD bid #{b&.price || 'none'} ask #{s&.price || 'none'}"
286 text += " spread #{s.price - b.price} mid #{(b.price + s.price) / 2}.#{(b.price + s.price).odd? ? 5 : 0}" if b && s
287 emit(text)
288 when 'IMPACT'
289 side, q = a[0], a[1].to_i
290 fail_if(q.zero?, 'invalid quantity')
291 ls = levels(opposite(side))
292 r, cost, used, worst = q, 0, 0, 0
293 ls.each do |p, v, _|
294 break if r.zero?
295 u = [r, v].min
296 r -= u
297 cost += u * p
298 used += 1
299 worst = p
300 end
301 emit(c, side, q, *(r > 0 ? ['insufficient', ls.sum { |v| v[1] }] : ['avg', decimals(cost * 100 / q), 'worst', worst, 'levels', used]))
302 when 'TRADES', 'BARS'
303 k = a[0].to_i
304 fail_if(k.zero?, 'invalid count')
305 if c == 'TRADES'
306 ts = @trades.last(k)
307 emit(c, ts.length, 'of', @trades.length)
308 ts.each { |t| emit(t.line) }
309 else
310 groups = @trades.each_slice(k).to_a
311 emit(c, groups.length)
312 groups.each_with_index { |g, i| emit(i + 1, bar(g)) }
313 end
314 when 'RANK'
315 ts = @traders.values.sort_by { |t| [-t.pnl(last || 0), -t.volume, t.name] }
316 emit(c, ts.length)
317 ts.each_with_index { |t, i| emit(i + 1, t.name, 'pnl', t.pnl(last || 0), 'volume', t.volume) }
318 when 'FLOW'
319 b = @trades.sum { |t| t.taker.side == 'BUY' ? t.quantity : 0 }
320 s = @trades.sum { |t| t.taker.side == 'SELL' ? t.quantity : 0 }
321 emit(c, 'buy', b, 'sell', s, 'net', b - s)
322 when 'SUMMARY'
323 if @trades.empty?
324 emit(c, 'trades', 0)
325 else
326 ps = @trades.map(&:price)
327 emit(c, 'trades', @trades.length, 'volume', @trades.sum(&:quantity), 'open', ps.first, 'high', ps.max, 'low', ps.min, 'close', ps.last, 'fees', @trades.sum { |t| t.buy_fee + t.sell_fee })
328 end
329 @traders.keys.sort.each { |name| emit(@traders[name].line(last || 0)) }
330 end
331 rescue CommandError => e
332 emit(*(%w[LIMIT MARKET STOP STOPLIMIT ICEBERG CANCEL REPLACE].include?(c) ? ['REJECT', a[0]] : ['ERROR']), e.message)
333 end
334end
335engine = Engine.new
336STDIN.gets.to_i.times { engine.execute(STDIN.gets) }
