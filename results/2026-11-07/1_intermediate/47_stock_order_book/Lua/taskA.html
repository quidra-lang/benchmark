1local function emit(...)
2 local a={...}
3 for i,v in ipairs(a) do
4 a[i]=tostring(v)
5 end
6 io.write(table.concat(a,' '),'\n')
7end

9 if bad then
10 error(reason,0)
11 end
12end
13local function dec(n)
14 return string.format('%d.%02d',n//100,n%100)
15end
16local function opposite(s)
17 return s=='BUY' and 'SELL' or 'BUY'
18end
19local orders, sequence, traders, trades = {}, {}, {}, {}
20local maker,taker,band,priority,tick,lot,halted=0,0,0,0,1,1,false
21local function active(o)
22 return o.status=='PENDING' or o.status=='OPEN' or o.status=='PARTIAL'
23end
24local function resting(o)
25 return o.status=='OPEN' or o.status=='PARTIAL'
26end
27local function matches(o,p)
28 return o.price==nil or (o.side=='BUY' and p<=o.price) or (o.side=='SELL' and p>=o.price)
29end
30local function triggered(o,p)
31 return (o.side=='BUY' and p>=o.trigger) or (o.side=='SELL' and p<=o.trigger)
32end
33local function last()
34 return #trades>0 and trades[#trades].price or nil
35end
36local function register(name)
37 if not traders[name] then
38 traders[name]={name=name,bought=0,sold=0,cash=0,fees=0}
39 end
40 return traders[name]
41end
42local function get_order(id)
43 check(not orders[id],'unknown order')
44 return orders[id]
45end
46local function get_trader(name)
47 check(not traders[name],'unknown trader')
48 return traders[name]
49end
50local function best(side)
51 local b=nil
52 for _,o in ipairs(sequence) do
53 if resting(o) and o.side==side and (not b or (o.price==b.price and o.priority<b.priority) or (side=='BUY' and o.price>b.price) or (side=='SELL' and o.price<b.price)) then
54 b=o
55 end
56 end
57 return b
58end
59local function active_orders(name)
60 local a={}
61 for _,o in ipairs(sequence) do
62 if o.trader==name and active(o) then
63 a[#a+1]=o
64 end
65 end
66 return a
67end
68local function exposure(name,side,excluded)
69 local t=traders[name]
70 local total=(t.bought-t.sold)*(side=='BUY' and 1 or -1)
71 for _,o in ipairs(active_orders(name)) do
72 if o.side==side and o~=excluded then
73 total=total+o.remaining
74 end
75 end
76 return total
77end
78local function risk(name,side,q,excluded)
79 local t=traders[name]
80 if t and t.limit then
81 check(exposure(name,side,excluded)+q>t.limit,'risk limit')
82 end
83end
84local function check_band(p)
85 local l=last()
86 check(band>0 and l and math.abs(p-l)>band,'outside band')
87end
88local function order_line(o)
89 return 'ORDER '..o.id..' '..o.status..' '..o.filled..'/'..o.quantity
90end
91local function details(o)
92 local text=table.concat({o.id,o.trader,o.kind,o.side,o.price or '-',o.status,o.filled..'/'..o.quantity,'avg',o.filled==0 and '-' or dec(o.value*100//o.filled)},' ')
93 return text..(o.trigger and ' trigger '..o.trigger or '')
94end
95local function pnl(t)
96 return t.cash+(t.bought-t.sold)*(last() or 0)
97end
98local function position(t)
99 emit('POSITION',t.name,'bought',t.bought,'sold',t.sold,'net',t.bought-t.sold,'cash',t.cash,'fees',t.fees,'pnl',pnl(t))
100end
101local function trade_line(t)
102 return table.concat({'TRADE',t.number,t.price,t.quantity,t.buy.id,t.buy.trader,t.sell.id,t.sell.trader},' ')
103end
104local function levels(side)
105 local map,ls={},{}
106 for _,o in ipairs(sequence) do
107 if resting(o) and o.side==side then
108 if not map[o.price] then
109 map[o.price]={o.price,0,0}
110 ls[#ls+1]=map[o.price]
111 end
112 local l=map[o.price]
113 l[2]=l[2]+o.visible
114 l[3]=l[3]+1
115 end
116 end
117 table.sort(ls,function(a,b)
118 if side=='BUY' then
119 return a[1]>b[1]
120 end
121 return a[1]<b[1]
122 end)
123 return ls
124end
125local function cancel(o)
126 o.history[#o.history+1]='CANCEL '..o.remaining
127 o.remaining=0
128 o.visible=0
129 o.status='CANCELLED'
130 emit(order_line(o))
131end
132local function match(o)
133 while o.remaining>0 do
134 local y=best(opposite(o.side))
135 if not y or not matches(o,y.price) then
136 break
137 end
138 if y.trader==o.trader then
139 emit('STP',y.id,y.remaining)
140 y.history[#y.history+1]='STP '..y.remaining
141 y.remaining=0
142 y.visible=0
143 y.status='CANCELLED'
144 else
145 local q,p=math.min(o.remaining,y.visible),y.price
146 local buy,sell,bf,sf
147 if o.side=='BUY' then
148 buy,sell,bf,sf=o,y,q*taker,q*maker
149 else
150 buy,sell,bf,sf=y,o,q*maker,q*taker
151 end
152 local tr={number=#trades+1,price=p,quantity=q,buy=buy,sell=sell,taker=o,buy_fee=bf,sell_fee=sf}
153 trades[#trades+1]=tr
154 for _,x in ipairs({o,y}) do
155 x.remaining=x.remaining-q
156 x.filled=x.filled+q
157 x.value=x.value+q*p
158 x.history[#x.history+1]='FILL '..tr.number..' '..p..' '..q
159 end
160 y.visible=y.visible-q
161 local bt,st=traders[buy.trader],traders[sell.trader]
162 bt.bought=bt.bought+q
163 bt.cash=bt.cash-q*p-bf
164 bt.fees=bt.fees+bf
165 st.sold=st.sold+q
166 st.cash=st.cash+q*p-sf
167 st.fees=st.fees+sf
168 emit(trade_line(tr))
169 y.status=y.remaining==0 and 'FILLED' or 'PARTIAL'
170 if y.remaining>0 and y.visible==0 then
171 priority=priority+1
172 y.priority=priority
173 y.visible=math.min(y.peak,y.remaining)
174 local h=y.remaining-y.visible
175 y.history[#y.history+1]='REFRESH '..y.visible..' '..h
176 emit('REFRESH',y.id,y.visible,h)
177 end
178 end
179 end
180end
181local function finish(o,force)
182 if o.remaining==0 then
183 o.status='FILLED'
184 elseif force or (o.kind=='LIMIT' and o.condition~='IOC') or o.kind=='ICEBERG' or o.kind=='STOPLIMIT' then
185 priority=priority+1
186 o.priority=priority
187 o.visible=math.min(o.peak or o.remaining,o.remaining)
188 o.status=o.filled==0 and 'OPEN' or 'PARTIAL'
189 o.history[#o.history+1]='REST '..o.price..' '..o.visible
190 else
191 o.history[#o.history+1]='EXPIRE '..o.remaining
192 o.remaining=0
193 o.status='CANCELLED'
194 end
195 emit(order_line(o))
196end
197local function triggers()
198 while last() do
199 local selected,l={},last()
200 for _,o in ipairs(sequence) do
201 if o.status=='PENDING' and triggered(o,l) then
202 selected[#selected+1]=o
203 o.status='TRIGGERING'
204 end
205 end
206 if #selected==0 then
207 break
208 end
209 for _,o in ipairs(selected) do
210 emit('TRIGGER',o.id)
211 o.history[#o.history+1]='TRIGGER'
212 match(o)
213 finish(o)
214 end
215 end
216end
217local function submit(c,a)
218 local id,name,side=a[1],a[2],a[3]
219 local p,r,q,peak,cond
220 if c=='MARKET' then
221 q=tonumber(a[4])
222 elseif c=='STOP' then
223 r,q=tonumber(a[4]),tonumber(a[5])
224 elseif c=='STOPLIMIT' then
225 r,p,q=tonumber(a[4]),tonumber(a[5]),tonumber(a[6])
226 else
227 p,q=tonumber(a[4]),tonumber(a[5])
228 if c=='ICEBERG' then
229 peak=tonumber(a[6])
230 elseif c=='LIMIT' then
231 cond=a[6]
232 end
233 end
234 check(halted,'halted')
235 check(orders[id]~=nil,'duplicate id')
236 check(p==0 or r==0,'invalid price')
237 check(q==0,'invalid quantity')
238 check((p and p%tick~=0) or (r and r%tick~=0),'invalid tick')
239 check(q%lot~=0,'invalid lot')
240 local o={id=id,trader=name,kind=c,side=side,price=p,trigger=r,peak=peak,condition=cond,quantity=q,remaining=q,visible=0,filled=0,value=0,priority=0,status='INCOMING',history={'ACCEPT'}}
241 check(peak and (peak==0 or peak>q or peak%lot~=0),'invalid peak')
242 check(r and last() and triggered(o,last()),'invalid trigger')
243 if p then
244 check_band(p)
245 end
246 risk(name,side,q)
247 local y=best(opposite(side))
248 check(cond=='POSTONLY' and y and matches(o,y.price),'would cross')
249 local available=0
250 for _,x in ipairs(sequence) do
251 if resting(x) and x.side==opposite(side) and x.trader~=name and matches(o,x.price) then
252 available=available+x.remaining
253 end
254 end
255 check(cond=='FOK' and available<q,'unfillable')
256 register(name)
257 orders[id]=o
258 sequence[#sequence+1]=o
259 if r then
260 o.status='PENDING'
261 emit(order_line(o))
262 else
263 if cond~='POSTONLY' then
264 match(o)
265 end
266 finish(o)
267 triggers()
268 end
269end
270local function replace(a)
271 local id,p,q=a[1],tonumber(a[2]),tonumber(a[3])
272 check(halted,'halted')
273 local o=get_order(id)
274 check(not resting(o),'not open')
275 check(p==0,'invalid price')
276 check(q==0,'invalid quantity')
277 local moved=p~=o.price
278 check(moved and p%tick~=0,'invalid tick')
279 check(q%lot~=0,'invalid lot')
280 check(not moved and q==o.remaining,'no change')
281 if moved then
282 check_band(p)
283 end
284 if q>o.remaining then
285 risk(o.trader,o.side,q,o)
286 end
287 local kept=not moved and q<o.remaining
288 local word=kept and 'kept' or 'lost'
289 o.quantity=o.filled+q
290 o.remaining=q
291 o.history[#o.history+1]='REPLACE '..p..' '..q..' '..word
292 emit('REPLACE',id,word)
293 if kept then
294 o.visible=math.min(o.visible,q)
295 emit(order_line(o))
296 else
297 o.price=p
298 o.status='INCOMING'
299 match(o)
300 finish(o,true)
301 triggers()
302 end
303end
304local function bar(ts)
305 local high,low,vol=ts[1].price,ts[1].price,0
306 for _,t in ipairs(ts) do
307 high=math.max(high,t.price)
308 low=math.min(low,t.price)
309 vol=vol+t.quantity
310 end
311 return 'open '..ts[1].price..' high '..high..' low '..low..' close '..ts[#ts].price..' volume '..vol
312end
313local function execute(c,a)
314 if c=='LIMIT' or c=='MARKET' or c=='STOP' or c=='STOPLIMIT' or c=='ICEBERG' then
315 submit(c,a)
316 elseif c=='CANCEL' then
317 local o=get_order(a[1])
318 check(not active(o),'not open')
319 cancel(o)
320 elseif c=='REPLACE' then
321 replace(a)
322 elseif c=='CANCELALL' then
323 get_trader(a[1])
324 local os=active_orders(a[1])
325 emit(c,a[1],#os)
326 for _,o in ipairs(os) do
327 cancel(o)
328 end
329 elseif c=='HALT' then
330 check(halted,'already halted')
331 halted=true
332 emit(c)
333 elseif c=='RESUME' then
334 check(not halted,'not halted')
335 halted=false
336 emit(c)
337 elseif c=='RISK' then
338 register(a[1]).limit=tonumber(a[2])
339 emit(c,a[1],a[2],'exposure buy',exposure(a[1],'BUY'),'sell',exposure(a[1],'SELL'))
340 elseif c=='FEE' then
341 local m,t=tonumber(a[1]),tonumber(a[2])
342 check(m+t<0,'invalid fee')
343 maker,taker=m,t
344 emit(c,'maker',m,'taker',t)
345 elseif c=='BAND' then
346 band=tonumber(a[1])
347 emit(c,band)
348 elseif c=='TICK' or c=='LOT' then
349 local n=tonumber(a[1])
350 check(n==0,'invalid size')
351 if c=='TICK' then
352 tick=n
353 else
354 lot=n
355 end
356 emit(c,n)
357 elseif c=='BOOK' then
358 local n=tonumber(a[1])
359 check(n==0,'invalid depth')
360 local bs,ss=levels('BUY'),levels('SELL')
361 emit(c,'bids',#bs,'asks',#ss)
362 for i=math.min(n,#ss),1,-1 do
363 emit('ASK',table.unpack(ss[i]))
364 end
365 for i=1,math.min(n,#bs) do
366 emit('BID',table.unpack(bs[i]))
367 end
368 elseif c=='ORDERS' then
369 get_trader(a[1])
370 local os=active_orders(a[1])
371 emit(c,a[1],#os)
372 for _,o in ipairs(os) do
373 emit(details(o))
374 end
375 elseif c=='STATUS' then
376 emit(c,details(get_order(a[1])))
377 elseif c=='HISTORY' then
378 local o=get_order(a[1])
379 emit(c,o.id,#o.history)
380 for _,h in ipairs(o.history) do
381 emit(h)
382 end
383 elseif c=='POSITION' then
384 position(get_trader(a[1]))
385 elseif c=='FILLS' then
386 get_trader(a[1])
387 local ts={}
388 for _,t in ipairs(trades) do
389 if t.buy.trader==a[1] or t.sell.trader==a[1] then
390 ts[#ts+1]=t
391 end
392 end
393 emit(c,a[1],#ts)
394 local net=0
395 for _,t in ipairs(ts) do
396 local buy=t.buy.trader==a[1]
397 local o=buy and t.buy or t.sell
398 net=net+(buy and t.quantity or -t.quantity)
399 emit(t.number,o.side,t.price,t.quantity,o.id,t.taker==o and 'TAKER' or 'MAKER','fee',buy and t.buy_fee or t.sell_fee,'net',net)
400 end
401 elseif c=='LAST' then
402 if #trades==0 then
403 emit(c,'none')
404 else
405 emit(c,last(),trades[#trades].quantity)
406 end
407 elseif c=='VWAP' then
408 if #trades==0 then
409 emit(c,'none')
410 else
411 local v,cost=0,0
412 for _,t in ipairs(trades) do
413 v=v+t.quantity
414 cost=cost+t.price*t.quantity
415 end
416 emit(c,dec(cost*100//v),'volume',v,'trades',#trades)
417 end
418 elseif c=='SPREAD' then
419 local b,s=best('BUY'),best('SELL')
420 local text='SPREAD bid '..(b and b.price or 'none')..' ask '..(s and s.price or 'none')
421 if b and s then
422 local n=b.price+s.price
423 text=text..' spread '..(s.price-b.price)..' mid '..(n//2)..'.'..(n%2==1 and 5 or 0)
424 end
425 emit(text)
426 elseif c=='IMPACT' then
427 local side,q=a[1],tonumber(a[2])
428 check(q==0,'invalid quantity')
429 local r,cost,worst,used,total=q,0,0,0,0
430 for _,l in ipairs(levels(opposite(side))) do
431 total=total+l[2]
432 if r>0 then
433 local u=math.min(r,l[2])
434 r=r-u
435 cost=cost+u*l[1]
436 worst=l[1]
437 used=used+1
438 end
439 end
440 if r>0 then
441 emit(c,side,q,'insufficient',total)
442 else
443 emit(c,side,q,'avg',dec(cost*100//q),'worst',worst,'levels',used)
444 end
445 elseif c=='TRADES' or c=='BARS' then
446 local k=tonumber(a[1])
447 check(k==0,'invalid count')
448 if c=='TRADES' then
449 local m=math.min(k,#trades)
450 emit(c,m,'of',#trades)
451 for i=#trades-m+1,#trades do
452 emit(trade_line(trades[i]))
453 end
454 else
455 emit(c,(#trades+k-1)//k)
456 local number=0
457 for i=1,#trades,k do
458 number=number+1
459 local ts={}
460 for j=i,math.min(i+k-1,#trades) do
461 ts[#ts+1]=trades[j]
462 end
463 emit(number,bar(ts))
464 end
465 end
466 elseif c=='RANK' then
467 local ts={}
468 for _,t in pairs(traders) do
469 ts[#ts+1]=t
470 end
471 table.sort(ts,function(x,y)
472 if pnl(x)~=pnl(y) then
473 return pnl(x)>pnl(y)
474 end
475 if x.bought+x.sold~=y.bought+y.sold then
476 return x.bought+x.sold>y.bought+y.sold
477 end
478 return x.name<y.name
479 end)
480 emit(c,#ts)
481 for i,t in ipairs(ts) do
482 emit(i,t.name,'pnl',pnl(t),'volume',t.bought+t.sold)
483 end
484 elseif c=='FLOW' then
485 local b,s=0,0
486 for _,t in ipairs(trades) do
487 if t.taker.side=='BUY' then
488 b=b+t.quantity
489 else
490 s=s+t.quantity
491 end
492 end
493 emit(c,'buy',b,'sell',s,'net',b-s)
494 elseif c=='SUMMARY' then
495 if #trades==0 then
496 emit(c,'trades',0)
497 else
498 local hi,lo,v,fees=trades[1].price,trades[1].price,0,0
499 for _,t in ipairs(trades) do
500 hi=math.max(hi,t.price)
501 lo=math.min(lo,t.price)
502 v=v+t.quantity
503 fees=fees+t.buy_fee+t.sell_fee
504 end
505 emit(c,'trades',#trades,'volume',v,'open',trades[1].price,'high',hi,'low',lo,'close',last(),'fees',fees)
506 end
507 local names={}
508 for n in pairs(traders) do
509 names[#names+1]=n
510 end
511 table.sort(names)
512 for _,n in ipairs(names) do
513 position(traders[n])
514 end
515 end
516end
517local count=tonumber(io.read())
518for _=1,count do
519 local words={}
520 for w in io.read():gmatch('%S+') do
521 words[#words+1]=w
522 end
523 local c=table.remove(words,1)
524 local ok,reason=pcall(execute,c,words)
525 if not ok then
526 if c=='LIMIT' or c=='MARKET' or c=='STOP' or c=='STOPLIMIT' or c=='ICEBERG' or c=='CANCEL' or c=='REPLACE' then
527 emit('REJECT',words[1],reason)
528 else
529 emit('ERROR',reason)
530 end
531 end
532end
