2class CommandError extends Error {
3}
4const opposite = (side: string) => side === 'BUY' ? 'SELL' : 'BUY';
5const fixed = (c: number) => `${Math.floor(c / 100)}.${String(c % 100).padStart(2, '0')}`;
6const average = (value: number, quantity: number) => fixed(Math.floor(value * 100 / quantity));
7const check = (bad: boolean, message: string) => {
8 if (bad) throw new CommandError(message);
9};
10class Order {
11 filled = 0;
12 value = 0;
13 visible = 0;
14 priority = 0;
15 status = 'PENDING';
16 history = ['ACCEPT'];
17 id: string;
18 trader: string;
19 type: string;
20 side: string;
21 quantity: number;
22 remaining: number;
23 price: number;
24 trigger: number;
25 peak: number;
26 condition: string;
27 constructor(id: string, trader: string, type: string, side: string, quantity: number, price = -1, trigger = -1, peak = -1,
28 condition = '') {
29 this.id = id;
30 this.trader = trader;
31 this.type = type;
32 this.side = side;
33 this.quantity = this.remaining = quantity;
34 this.price = price;
35 this.trigger = trigger;
36 this.peak = peak;
37 this.condition = condition;
38 }
39 resting() {
40 return this.status === 'OPEN' || this.status === 'PARTIAL';
41 }
42 active() {
43 return this.resting() || this.status === 'PENDING';
44 }
45 matches(price: number) {
46 return this.price < 0 || (this.side === 'BUY' ? price <= this.price : price >= this.price);
47 }
48 triggered(price: number) {
49 return this.side === 'BUY' ? price >= this.trigger : price <= this.trigger;
50 }
51 line() {
52 return `ORDER ${this.id} ${this.status} ${this.filled}/${this.quantity}`;
53 }
54 details() {
55 return `${this.id} ${this.trader} ${this.type} ${this.side} ${this.price < 0 ? '-' : this.price} ${this.status} ${this.filled}/${this.quantity} avg ${this.filled ? average(this.value, this.filled) : '-'}` +
56 (this.trigger < 0 ? '' : ` trigger ${this.trigger}`);
57 }
58}
59class Trader {
60 bought = 0;
61 sold = 0;
62 cash = 0;
63 fees = 0;
64 limit = -1;
65 name: string;
66 constructor(name: string) {
67 this.name = name;
68 }
69 net() {
70 return this.bought - this.sold;
71 }
72 volume() {
73 return this.bought + this.sold;
74 }
75 pnl(mark: number) {
76 return this.cash + this.net() * mark;
77 }
78 position(mark: number) {
79 return `POSITION ${this.name} bought ${this.bought} sold ${this.sold} net ${this.net()} cash ${this.cash} fees ${this.fees} pnl ${this.pnl(mark)}`;
80 }
81}
82interface Trade {
83 number: number;
84 price: number;
85 quantity: number;
86 buy: Order;
87 sell: Order;
88 taker: Order;
89 buyFee: number;
90 sellFee: number;
91}
92interface Level {
93 price: number;
94 quantity: number;
95 count: number;
96}
97class Engine {
98 orders = new Map<string, Order>();
99 traders = new Map<string, Trader>();
100 trades: Trade[] = [];
101 output: string[] = [];
102 priority = 1;
103 halted = false;
104 band = 0;
105 tick = 1;
106 lot = 1;
107 makerFee = 0;
108 takerFee = 0;
109 emit(...parts: (string | number)[]) {
110 this.output.push(parts.join(' '));
111 }
112 all() {
113 return [...this.orders.values()];
114 }
115 register(name: string) {
116 let t = this.traders.get(name);
117 if (!t) {
118 t = new Trader(name);
119 this.traders.set(name, t);
120 }
121 return t;
122 }
123 trader(name: string) {
124 const t = this.traders.get(name);
125 if (!t) throw new CommandError('unknown trader');
126 return t;
127 }
128 order(id: string) {
129 const o = this.orders.get(id);
130 if (!o) throw new CommandError('unknown order');
131 return o;
132 }
133 book(side: string) {
134 return this.all().filter(o => o.side === side && o.resting()).sort((a, b) => (side === 'BUY' ? b.price - a.price : a.price - b.price) || a.priority - b.priority);
135 }
136 levels(side: string) {
137 const levels: Level[] = [];
138 for (const o of this.book(side)) {
139 const last = levels[levels.length - 1];
140 if (last && last.price === o.price) {
141 last.quantity += o.visible;
142 ++last.count;
143 }
144 else levels.push({
145 price: o.price, quantity: o.visible, count: 1
146 });
147 }
148 return levels;
149 }
150 last() {
151 return this.trades.length ? this.trades[this.trades.length - 1].price : -1;
152 }
153 exposure(name: string, side: string, excluded: Order | null = null) {
154 const t = this.trader(name);
155 return (side === 'BUY' ? t.net() : -t.net()) + this.all().filter(o => o.trader === name && o.side === side && o.active() && o !== excluded).reduce((n,
156 o) => n + o.remaining, 0);
157 }
158 risk(name: string, side: string, quantity: number, excluded: Order | null = null) {
159 const t = this.traders.get(name);
160 check(!!t && t.limit >= 0 && this.exposure(name, side, excluded) + quantity > t.limit, 'risk limit');
161 }
162 checkBand(price: number) {
163 check(this.band > 0 && this.last() >= 0 && Math.abs(price - this.last()) > this.band, 'outside band');
164 }
165 tradeLine(t: Trade) {
166 return `TRADE ${t.number} ${t.price} ${t.quantity} ${t.buy.id} ${t.buy.trader} ${t.sell.id} ${t.sell.trader}`;
167 }
168 match(o: Order) {
169 while (o.remaining) {
170 const m = this.book(opposite(o.side))[0];
171 if (!m || !o.matches(m.price)) break;
172 if (m.trader === o.trader) {
173 this.emit('STP', m.id, m.remaining);
174 m.history.push(`STP ${m.remaining}`);
175 m.remaining = 0;
176 m.status = 'CANCELLED';
177 continue;
178 }
179 const q = Math.min(o.remaining, m.visible), p = m.price;
180 const buy = o.side === 'BUY' ? o : m, sell = o.side === 'SELL' ? o : m;
181 const tf = q * this.takerFee, mf = q * this.makerFee;
182 const t = {
183 number: this.trades.length + 1, price: p, quantity: q, buy, sell, taker: o,
184 buyFee: o.side === 'BUY' ? tf : mf, sellFee: o.side === 'SELL' ? tf : mf
185 };
186 this.trades.push(t);
187 for (const a of [o, m]) {
188 a.remaining -= q;
189 a.filled += q;
190 a.value += p * q;
191 a.history.push(`FILL ${t.number} ${p} ${q}`);
192 }
193 m.visible -= q;
194 const bt = this.trader(buy.trader), st = this.trader(sell.trader);
195 bt.bought += q;
196 bt.cash -= p * q + t.buyFee;
197 bt.fees += t.buyFee;
198 st.sold += q;
199 st.cash += p * q - t.sellFee;
200 st.fees += t.sellFee;
201 this.emit(this.tradeLine(t));
202 m.status = m.remaining ? 'PARTIAL' : 'FILLED';
203 if (m.remaining && !m.visible) {
204 m.priority = this.priority++;
205 m.visible = Math.min(m.peak, m.remaining);
206 m.history.push(`REFRESH ${m.visible} ${m.remaining - m.visible}`);
207 this.emit('REFRESH', m.id, m.visible, m.remaining - m.visible);
208 }
209 }
210 }
211 post(o: Order, replaced = false) {
212 if (!o.remaining) o.status = 'FILLED';
213 else if (replaced || o.type === 'ICEBERG' || o.type === 'STOPLIMIT' || (o.type === 'LIMIT' && o.condition !== 'IOC')) {
214 o.priority = this.priority++;
215 o.visible = o.peak < 0 ? o.remaining : Math.min(o.peak, o.remaining);
216 o.status = o.filled ? 'PARTIAL' : 'OPEN';
217 o.history.push(`REST ${o.price} ${o.visible}`);
218 } else {
219 o.history.push(`EXPIRE ${o.remaining}`);
220 o.remaining = 0;
221 o.status = 'CANCELLED';
222 }
223 this.emit(o.line());
224 }
225 trigger() {
226 while (this.trades.length) {
227 const selected = this.all().filter(o => o.status === 'PENDING' && o.triggered(this.last()));
228 if (!selected.length) break;
229 for (const o of selected) o.status = 'TRIGGERING';
230 for (const o of selected) {
231 this.emit('TRIGGER', o.id);
232 o.history.push('TRIGGER');
233 this.match(o);
234 this.post(o);
235 }
236 }
237 }
238 cancel(o: Order) {
239 o.history.push(`CANCEL ${o.remaining}`);
240 o.remaining = 0;
241 o.status = 'CANCELLED';
242 this.emit(o.line());
243 }
244 bar(trades: Trade[]) {
245 return `open ${trades[0].price} high ${trades.reduce((best, t) => Math.max(best, t.price), trades[0].price)} low ${trades.reduce((best, t) => Math.min(best, t.price), trades[0].price)} close ${trades[trades.length - 1].price} volume ${trades.reduce((n, t) => n + t.quantity, 0)}`;
246 }
247 execute(line: string) {
248 const [c, ...a] = line.split(' ');
249 try {
250 this.command(c, a);
251 }
252 catch (e) {
253 if (!(e instanceof CommandError)) throw e;
254 this.emit(...(['LIMIT', 'MARKET', 'STOP', 'STOPLIMIT', 'ICEBERG', 'CANCEL', 'REPLACE'].includes(c) ? ['REJECT', a[0],
255 e.message] : ['ERROR', e.message]));
256 }
257 }
258 command(c: string, a: string[]) {
259 const [id, name, side] = a;
260 if (['LIMIT', 'MARKET', 'STOP', 'STOPLIMIT', 'ICEBERG'].includes(c)) {
261 const p = c === 'MARKET' || c === 'STOP' ? -1 : Number(a[c === 'STOPLIMIT' ? 4 : 3]);
262 const r = c === 'STOP' || c === 'STOPLIMIT' ? Number(a[3]) : -1;
263 const q = Number(a[c === 'MARKET' ? 3 : c === 'STOPLIMIT' ? 5 : 4]);
264 const peak = c === 'ICEBERG' ? Number(a[5]) : -1, condition = c === 'LIMIT' ? a[5] ?? '' : '';
265 const prices = [p, r].filter(x => x >= 0);
266 check(this.halted, 'halted');
267 check(this.orders.has(id), 'duplicate id');
268 check(prices.includes(0), 'invalid price');
269 check(!q, 'invalid quantity');
270 check(prices.some(x => x % this.tick !== 0), 'invalid tick');
271 check(q % this.lot !== 0, 'invalid lot');
272 if (c === 'ICEBERG') check(peak === 0 || peak > q || peak % this.lot !== 0, 'invalid peak');
273 const o = new Order(id, name, c, side, q, p, r, peak, condition);
274 if (r >= 0) check(this.last() >= 0 && o.triggered(this.last()), 'invalid trigger');
275 if (p >= 0) this.checkBand(p);
276 this.risk(name, side, q);
277 const book = this.book(opposite(side));
278 check(condition === 'POSTONLY' && !!book[0] && o.matches(book[0].price), 'would cross');
279 check(condition === 'FOK' && book.filter(m => m.trader !== name && o.matches(m.price)).reduce((n, m) => n + m.remaining,
280 0) < q, 'unfillable');
281 this.register(name);
282 this.orders.set(id, o);
283 if (r >= 0) this.emit(o.line());
284 else {
285 if (condition !== 'POSTONLY') this.match(o);
286 this.post(o);
287 this.trigger();
288 }
289 return;
290 }
291 switch (c) {
292 case 'CANCEL': {
293 const o = this.order(id);
294 check(!o.active(), 'not open');
295 this.cancel(o);
296 break;
297 }
298 case 'REPLACE': {
299 check(this.halted, 'halted');
300 const o = this.order(id);
301 check(!o.resting(), 'not open');
302 const p = Number(name), q = Number(side), moved = p !== o.price;
303 check(!p, 'invalid price');
304 check(!q, 'invalid quantity');
305 check(moved && p % this.tick !== 0, 'invalid tick');
306 check(q % this.lot !== 0, 'invalid lot');
307 check(!moved && q === o.remaining, 'no change');
308 if (moved) this.checkBand(p);
309 if (q > o.remaining) this.risk(o.trader, o.side, q, o);
310 const kept = !moved && q < o.remaining;
311 o.quantity = o.filled + q;
312 o.remaining = q;
313 o.price = p;
314 o.history.push(`REPLACE ${p} ${q} ${kept ? 'kept' : 'lost'}`);
315 this.emit('REPLACE', id, kept ? 'kept' : 'lost');
316 if (kept) {
317 o.visible = Math.min(o.visible, q);
318 this.emit(o.line());
319 }
320 else {
321 o.status = 'EXECUTING';
322 this.match(o);
323 this.post(o, true);
324 this.trigger();
325 }
326 break;
327 }
328 case 'CANCELALL': {
329 this.trader(id);
330 const os = this.all().filter(o => o.trader === id && o.active());
331 this.emit(c, id, os.length);
332 for (const o of os) this.cancel(o);
333 break;
334 }
335 case 'HALT': check(this.halted, 'already halted');
336 this.halted = true;
337 this.emit(c);
338 break;
339 case 'RESUME': check(!this.halted, 'not halted');
340 this.halted = false;
341 this.emit(c);
342 break;
343 case 'RISK': this.register(id).limit = Number(name);
344 this.emit(c, id, name, 'exposure buy', this.exposure(id, 'BUY'), 'sell', this.exposure(id, 'SELL'));
345 break;
346 case 'FEE': {
347 const maker = Number(id), taker = Number(name);
348 check(maker + taker < 0, 'invalid fee');
349 this.makerFee = maker;
350 this.takerFee = taker;
351 this.emit(c, 'maker', maker, 'taker', taker);
352 break;
353 }
354 case 'BAND': this.band = Number(id);
355 this.emit(c, id);
356 break;
357 case 'TICK': case 'LOT': check(Number(id) === 0, 'invalid size');
358 if (c === 'TICK') this.tick = Number(id);
359 else this.lot = Number(id);
360 this.emit(c, id);
361 break;
362 case 'BOOK': {
363 const n = Number(id);
364 check(!n, 'invalid depth');
365 const bids = this.levels('BUY'), asks = this.levels('SELL');
366 this.emit(c, 'bids', bids.length, 'asks', asks.length);
367 for (const l of asks.slice(0, n).reverse()) this.emit('ASK', l.price, l.quantity, l.count);
368 for (const l of bids.slice(0, n)) this.emit('BID', l.price, l.quantity, l.count);
369 break;
370 }
371 case 'ORDERS': {
372 this.trader(id);
373 const os = this.all().filter(o => o.trader === id && o.active());
374 this.emit(c, id, os.length);
375 for (const o of os) this.emit(o.details());
376 break;
377 }
378 case 'STATUS': this.emit(c, this.order(id).details());
379 break;
380 case 'HISTORY': {
381 const o = this.order(id);
382 this.emit(c, id, o.history.length);
383 for (const h of o.history) this.emit(h);
384 break;
385 }
386 case 'POSITION': this.emit(this.trader(id).position(Math.max(0, this.last())));
387 break;
388 case 'FILLS': {
389 this.trader(id);
390 const ts = this.trades.filter(t => t.buy.trader === id || t.sell.trader === id);
391 this.emit(c, id, ts.length);
392 let net = 0;
393 for (const t of ts) {
394 const buy = t.buy.trader === id, o = buy ? t.buy : t.sell;
395 net += buy ? t.quantity : -t.quantity;
396 this.emit(t.number, buy ? 'BUY' : 'SELL', t.price, t.quantity, o.id, o === t.taker ? 'TAKER' : 'MAKER', 'fee', buy ? t.buyFee : t.sellFee,
397 'net', net);
398 }
399 break;
400 }
401 case 'LAST': {
402 const t = this.trades[this.trades.length - 1];
403 if (t) this.emit(c, t.price, t.quantity);
404 else this.emit(c, 'none');
405 break;
406 }
407 case 'VWAP': if (!this.trades.length) this.emit(c, 'none');
408 else {
409 const v = this.trades.reduce((n, t) => n + t.quantity, 0);
410 this.emit(c, average(this.trades.reduce((n, t) => n + t.price * t.quantity, 0), v), 'volume', v, 'trades', this.trades.length);
411 }
412 break;
413 case 'SPREAD': {
414 const b = this.book('BUY')[0], s = this.book('SELL')[0];
415 if (b && s) this.emit(c, 'bid', b.price, 'ask', s.price, 'spread', s.price - b.price, 'mid', `${Math.floor((b.price + s.price) / 2)}.${(b.price + s.price) % 2 ? 5 : 0}`);
416 else this.emit(c, 'bid', b ? b.price : 'none', 'ask', s ? s.price : 'none');
417 break;
418 }
419 case 'IMPACT': {
420 const q = Number(name);
421 check(!q, 'invalid quantity');
422 const ls = this.levels(opposite(id));
423 let r = q, cost = 0, worst = 0, used = 0;
424 for (const l of ls) {
425 if (!r) break;
426 const n = Math.min(r, l.quantity);
427 r -= n;
428 cost += n * l.price;
429 worst = l.price;
430 ++used;
431 }
432 if (r) this.emit(c, id, q, 'insufficient', ls.reduce((n, l) => n + l.quantity, 0));
433 else this.emit(c, id, q, 'avg', average(cost, q), 'worst', worst, 'levels', used);
434 break;
435 }
436 case 'TRADES': {
437 const n = Number(id);
438 check(!n, 'invalid count');
439 const ts = this.trades.slice(Math.max(0, this.trades.length - n));
440 this.emit(c, ts.length, 'of', this.trades.length);
441 for (const t of ts) this.emit(this.tradeLine(t));
442 break;
443 }
444 case 'BARS': {
445 const n = Number(id);
446 check(!n, 'invalid count');
447 this.emit(c, Math.ceil(this.trades.length / n));
448 for (let i = 0; i < this.trades.length; i += n) this.emit(i / n + 1, this.bar(this.trades.slice(i, i + n)));
449 break;
450 }
451 case 'RANK': {
452 const mark = Math.max(0, this.last()), ts = [...this.traders.values()].sort((a, b) => b.pnl(mark) - a.pnl(mark) || b.volume() - a.volume() || (a.name < b.name ? -1 : a.name > b.name ? 1 : 0));
453 this.emit(c, ts.length);
454 ts.forEach((t, i) => this.emit(i + 1, t.name, 'pnl', t.pnl(mark), 'volume', t.volume()));
455 break;
456 }
457 case 'FLOW': {
458 const b = this.trades.filter(t => t.taker.side === 'BUY').reduce((n, t) => n + t.quantity, 0), s = this.trades.filter(t => t.taker.side === 'SELL').reduce((n,
459 t) => n + t.quantity, 0);
460 this.emit(c, 'buy', b, 'sell', s, 'net', b - s);
461 break;
462 }
463 case 'SUMMARY': if (this.trades.length) this.emit(c, 'trades', this.trades.length, 'volume', this.trades.reduce((n,
464 t) => n + t.quantity, 0), this.bar(this.trades).split(' ').slice(0, -2).join(' '), 'fees', this.trades.reduce((n,
465 t) => n + t.buyFee + t.sellFee, 0));
466 else this.emit(c, 'trades', 0);
467 for (const name of [...this.traders.keys()].sort()) this.emit(this.trader(name).position(Math.max(0, this.last())));
468 break;
469 }
470 }
471}
472const lines = readFileSync(0, 'utf8').trimEnd().split('\n');
473const engine = new Engine();
474for (let i = 1; i <= Number(lines[0]); ++i) engine.execute(lines[i].replace(/\r$/, ''));
475process.stdout.write(engine.output.join('\n') + '\n');
