2using System;
3using System.Collections.Generic;

5sealed class Order
6{
7 public string Id, Trader, Kind, Side, Condition = "", Status = "OPEN";
8 public int? Price, Trigger, Peak;
9 public int Quantity, Remaining, Visible, Filled, Value, Priority, Number;
10 public List<string> History = new() {
11 "ACCEPT"
12 };
13 public Order(string id, string trader, string kind, string side, int quantity)
14 {
15 Id = id;
16 Trader = trader;
17 Kind = kind;
18 Side = side;
19 Quantity = Remaining = quantity;
20 }
21 public bool Active => Status is "OPEN" or "PARTIAL" or "PENDING";
22 public bool Resting => Status is "OPEN" or "PARTIAL";
23 public bool Matches(int p) => Price == null || (Side == "BUY" ? p <= Price : p >= Price);
24 public bool Triggered(int p) => Side == "BUY" ? p >= Trigger : p <= Trigger;
25 public string Line => $"ORDER {Id} {Status} {Filled}/{Quantity}";
26 public string Details => $"{Id} {Trader} {Kind} {Side} {Price?.ToString() ?? "-"} {Status} {Filled}/{Quantity} avg {(Filled == 0 ? "-" : Engine.Decimal(Value * 100 / Filled))}" + (Trigger is int r ? $" trigger {r}" : "");
27 public void Fill(int n, int p, int q) {
28 Remaining -= q;
29 Filled += q;
30 Value += p * q;
31 History.Add($"FILL {n} {p} {q}");
32 }
33}
34sealed class Trader
35{
36 public string Name;
37 public int Bought, Sold, Cash, Fees;
38 public int? Limit;
39 public Trader(string name) {
40 Name = name;
41 }
42 public int Net => Bought - Sold;
43 public int Volume => Bought + Sold;
44 public int Pnl(int p) => Cash + Net * p;
45 public string Position(int p) => $"POSITION {Name} bought {Bought} sold {Sold} net {Net} cash {Cash} fees {Fees} pnl {Pnl(p)}";
46}
47sealed record Trade(int Number, int Price, int Quantity, Order Buy, Order Sell, string Taker, int BuyFee, int SellFee)
48{
49 public string Line => $"TRADE {Number} {Price} {Quantity} {Buy.Id} {Buy.Trader} {Sell.Id} {Sell.Trader}";
50}
51sealed class CommandError : Exception {
52 public CommandError(string message) : base(message) {
53 }
54}
55sealed class Engine
56{
57 readonly Dictionary<string, Order> orders = new(StringComparer.Ordinal);
58 readonly Dictionary<string, Trader> traders = new(StringComparer.Ordinal);
59 readonly List<Trade> trades = new();
60 readonly HashSet<Order> book = new();
61 readonly List<Order> pending = new();
62 readonly List<string> output = new();
63 int maker, taker, band, tick = 1, lot = 1, priority;
64 bool halted;
65 int Mark => trades.Count == 0 ? 0 : trades[^1].Price;
66 public static string Decimal(int x) => $"{x / 100}.{x % 100:D2}";
67 static void Require(bool good, string reason) {
68 if (!good) throw new CommandError(reason);
69 }
70 Trader Register(string name) {
71 if (!traders.ContainsKey(name)) traders[name] = new(name);
72 return traders[name];
73 }
74 Trader RequireTrader(string name) {
75 Require(traders.ContainsKey(name), "unknown trader");
76 return traders[name];
77 }
78 Order RequireOrder(string id) {
79 Require(orders.ContainsKey(id), "unknown order");
80 return orders[id];
81 }
82 List<Order> Active(string name) => orders.Values.Where(o => o.Trader == name && o.Active).OrderBy(o => o.Number).ToList();
83 int Exposure(string name, string side, Order? exclude = null) => (side == "BUY" ? traders[name].Net : -traders[name].Net) + Active(name).Where(o => o.Side == side && o != exclude).Sum(o => o.Remaining);
84 void Risk(string name, string side, int q, Order? exclude = null) {
85 if (traders.TryGetValue(name, out Trader? t) && t.Limit is int m) Require(Exposure(name, side, exclude) + q <= m, "risk limit");
86 }
87 void Band(int p) => Require(band == 0 || trades.Count == 0 || Math.Abs(p - Mark) <= band, "outside band");
88 List<Order> Sorted(string side) => book.Where(o => o.Side == side).OrderBy(o => side == "BUY" ? -o.Price!.Value : o.Price!.Value).ThenBy(o => o.Priority).ToList();
89 Order? Best(string side) => Sorted(side).FirstOrDefault();
90 List<(int Price, int Quantity, int Count)> Levels(string side) => Sorted(side).GroupBy(o => o.Price!.Value).Select(g => (g.Key, g.Sum(o => o.Visible), g.Count())).ToList();
91 static string Opposite(string side) => side == "BUY" ? "SELL" : "BUY";
92 void Match(Order x)
93 {
94 while (x.Remaining > 0)
95 {
96 Order? y = Best(Opposite(x.Side));
97 if (y == null || !x.Matches(y.Price!.Value)) break;
98 if (x.Trader == y.Trader)
99 {
100 output.Add($"STP {y.Id} {y.Remaining}");
101 y.History.Add($"STP {y.Remaining}");
102 book.Remove(y);
103 y.Remaining = 0;
104 y.Status = "CANCELLED";
105 continue;
106 }
107 int q = Math.Min(x.Remaining, y.Visible), p = y.Price.Value;
108 Order buy = x.Side == "BUY" ? x : y, sell = x.Side == "SELL" ? x : y;
109 int bf = q * (buy == x ? taker : maker), sf = q * (sell == x ? taker : maker);
110 var trade = new Trade(trades.Count + 1, p, q, buy, sell, x.Id, bf, sf);
111 trades.Add(trade);
112 x.Fill(trade.Number, p, q);
113 y.Fill(trade.Number, p, q);
114 y.Visible -= q;
115 Trader b = traders[buy.Trader], s = traders[sell.Trader];
116 b.Bought += q;
117 b.Cash -= p * q + bf;
118 b.Fees += bf;
119 s.Sold += q;
120 s.Cash += p * q - sf;
121 s.Fees += sf;
122 output.Add(trade.Line);
123 if (y.Remaining == 0) {
124 book.Remove(y);
125 y.Status = "FILLED";
126 } else
127 {
128 y.Status = "PARTIAL";
129 if (y.Visible == 0)
130 {
131 y.Priority = ++priority;
132 y.Visible = Math.Min(y.Peak!.Value, y.Remaining);
133 string refresh = $"REFRESH {y.Visible} {y.Remaining - y.Visible}";
134 y.History.Add(refresh);
135 output.Add($"REFRESH {y.Id} {y.Visible} {y.Remaining - y.Visible}");
136 }
137 }
138 }
139 }
140 void Post(Order o, bool forceRest = false)
141 {
142 if (o.Remaining == 0) o.Status = "FILLED";
143 else if (forceRest || o.Kind is "ICEBERG" or "STOPLIMIT" || o.Kind == "LIMIT" && o.Condition != "IOC")
144 {
145 o.Priority = ++priority;
146 o.Visible = Math.Min(o.Peak ?? o.Remaining, o.Remaining);
147 o.Status = o.Filled == 0 ? "OPEN" : "PARTIAL";
148 book.Add(o);
149 o.History.Add($"REST {o.Price} {o.Visible}");
150 } else {
151 o.History.Add($"EXPIRE {o.Remaining}");
152 o.Remaining = 0;
153 o.Status = "CANCELLED";
154 }
155 output.Add(o.Line);
156 }
157 void Trigger()
158 {
159 while (trades.Count > 0)
160 {
161 List<Order> selected = pending.Where(o => o.Triggered(Mark)).OrderBy(o => o.Number).ToList();
162 if (selected.Count == 0) break;
163 foreach (Order o in selected) pending.Remove(o);
164 foreach (Order o in selected) {
165 output.Add($"TRIGGER {o.Id}");
166 o.History.Add("TRIGGER");
167 Match(o);
168 Post(o);
169 }
170 }
171 }
172 void Cancel(Order o) {
173 book.Remove(o);
174 pending.Remove(o);
175 o.History.Add($"CANCEL {o.Remaining}");
176 o.Remaining = 0;
177 o.Status = "CANCELLED";
178 output.Add(o.Line);
179 }
180 string Bars(IEnumerable<Trade> source)
181 {
182 List<Trade> ts = source.ToList();
183 return $"open {ts[0].Price} high {ts.Max(t => t.Price)} low {ts.Min(t => t.Price)} close {ts[^1].Price} volume {ts.Sum(t => t.Quantity)}";
184 }
185 void Command(string[] w)
186 {
187 string c = w[0];
188 int Num(int i) => int.Parse(w[i]);
189 if (c is "LIMIT" or "MARKET" or "STOP" or "STOPLIMIT" or "ICEBERG")
190 {
191 int qi = c == "MARKET" ? 4 : c == "STOPLIMIT" ? 6 : 5;
192 var o = new Order(w[1], w[2], c, w[3], Num(qi));
193 if (c is "LIMIT" or "ICEBERG") o.Price = Num(4);
194 if (c is "STOP" or "STOPLIMIT") o.Trigger = Num(4);
195 if (c == "STOPLIMIT") o.Price = Num(5);
196 if (c == "ICEBERG") o.Peak = Num(6);
197 if (c == "LIMIT" && w.Length == 7) o.Condition = w[6];
198 int[] prices = new[] {
199 o.Price, o.Trigger
200 }
201 .Where(p => p != null).Select(p => p!.Value).ToArray();
202 Require(!halted, "halted");
203 Require(!orders.ContainsKey(o.Id), "duplicate id");
204 Require(prices.All(p => p != 0), "invalid price");
205 Require(o.Quantity != 0, "invalid quantity");
206 Require(prices.All(p => p % tick == 0), "invalid tick");
207 Require(o.Quantity % lot == 0, "invalid lot");
208 if (o.Peak is int k) Require(k > 0 && k <= o.Quantity && k % lot == 0, "invalid peak");
209 if (o.Trigger != null) Require(trades.Count == 0 || !o.Triggered(Mark), "invalid trigger");
210 if (o.Price is int p) Band(p);
211 Risk(o.Trader, o.Side, o.Quantity);
212 Order? best = Best(Opposite(o.Side));
213 if (o.Condition == "POSTONLY") Require(best == null || !o.Matches(best.Price!.Value), "would cross");
214 if (o.Condition == "FOK") Require(Sorted(Opposite(o.Side)).Where(a => a.Trader != o.Trader && o.Matches(a.Price!.Value)).Sum(a => a.Remaining) >= o.Quantity, "unfillable");
215 Register(o.Trader);
216 o.Number = orders.Count + 1;
217 orders.Add(o.Id, o);
218 if (o.Trigger != null) {
219 o.Status = "PENDING";
220 pending.Add(o);
221 output.Add(o.Line);
222 } else {
223 if (o.Condition != "POSTONLY") Match(o);
224 Post(o);
225 Trigger();
226 }
227 return;
228 }
229 switch (c)
230 {
231 case "CANCEL": {
232 Order o = RequireOrder(w[1]);
233 Require(o.Active, "not open");
234 Cancel(o);
235 break;
236 }
237 case "REPLACE":
238 {
239 Require(!halted, "halted");
240 Order o = RequireOrder(w[1]);
241 Require(o.Resting, "not open");
242 int p = Num(2), q = Num(3);
243 bool moved = p != o.Price;
244 Require(p > 0, "invalid price");
245 Require(q > 0, "invalid quantity");
246 Require(!moved || p % tick == 0, "invalid tick");
247 Require(q % lot == 0, "invalid lot");
248 Require(moved || q != o.Remaining, "no change");
249 if (moved) Band(p);
250 if (q > o.Remaining) Risk(o.Trader, o.Side, q, o);
251 o.Quantity = o.Filled + q;
252 if (!moved && q < o.Remaining)
253 {
254 o.Remaining = q;
255 o.Visible = Math.Min(o.Visible, q);
256 o.History.Add($"REPLACE {p} {q} kept");
257 output.Add($"REPLACE {o.Id} kept");
258 output.Add(o.Line);
259 } else
260 {
261 book.Remove(o);
262 o.Price = p;
263 o.Remaining = q;
264 o.History.Add($"REPLACE {p} {q} lost");
265 output.Add($"REPLACE {o.Id} lost");
266 Match(o);
267 Post(o, true);
268 Trigger();
269 }
270 break;
271 }
272 case "CANCELALL": {
273 RequireTrader(w[1]);
274 List<Order> active = Active(w[1]);
275 output.Add($"CANCELALL {w[1]} {active.Count}");
276 foreach (Order o in active) Cancel(o);
277 break;
278 }
279 case "HALT": Require(!halted, "already halted");
280 halted = true;
281 output.Add(c);
282 break;
283 case "RESUME": Require(halted, "not halted");
284 halted = false;
285 output.Add(c);
286 break;
287 case "RISK": Register(w[1]).Limit = Num(2);
288 output.Add($"RISK {w[1]} {w[2]} exposure buy {Exposure(w[1], "BUY")} sell {Exposure(w[1], "SELL")}");
289 break;
290 case "FEE": Require(Num(1) + Num(2) >= 0, "invalid fee");
291 maker = Num(1);
292 taker = Num(2);
293 output.Add($"FEE maker {maker} taker {taker}");
294 break;
295 case "BAND": band = Num(1);
296 output.Add($"BAND {band}");
297 break;
298 case "TICK": case "LOT": Require(Num(1) > 0, "invalid size");
299 if (c == "TICK") tick = Num(1);
300 else lot = Num(1);
301 output.Add($"{c} {w[1]}");
302 break;
303 case "BOOK":
304 {
305 int n = Num(1);
306 Require(n > 0, "invalid depth");
307 var bids = Levels("BUY");
308 var asks = Levels("SELL");
309 output.Add($"BOOK bids {bids.Count} asks {asks.Count}");
310 foreach (var l in asks.Take(n).Reverse()) output.Add($"ASK {l.Price} {l.Quantity} {l.Count}");
311 foreach (var l in bids.Take(n)) output.Add($"BID {l.Price} {l.Quantity} {l.Count}");
312 break;
313 }
314 case "ORDERS": {
315 RequireTrader(w[1]);
316 var active = Active(w[1]);
317 output.Add($"ORDERS {w[1]} {active.Count}");
318 output.AddRange(active.Select(o => o.Details));
319 break;
320 }
321 case "STATUS": output.Add("STATUS " + RequireOrder(w[1]).Details);
322 break;
323 case "HISTORY": {
324 Order o = RequireOrder(w[1]);
325 output.Add($"HISTORY {o.Id} {o.History.Count}");
326 output.AddRange(o.History);
327 break;
328 }
329 case "POSITION": output.Add(RequireTrader(w[1]).Position(Mark));
330 break;
331 case "FILLS":
332 {
333 string name = w[1];
334 RequireTrader(name);
335 var ts = trades.Where(t => t.Buy.Trader == name || t.Sell.Trader == name).ToList();
336 output.Add($"FILLS {name} {ts.Count}");
337 int net = 0;
338 foreach (Trade t in ts) {
339 bool buy = t.Buy.Trader == name;
340 string id = buy ? t.Buy.Id : t.Sell.Id;
341 net += buy ? t.Quantity : -t.Quantity;
342 output.Add($"{t.Number} {(buy ? "BUY" : "SELL")} {t.Price} {t.Quantity} {id} {(t.Taker == id ? "TAKER" : "MAKER")} fee {(buy ? t.BuyFee : t.SellFee)} net {net}");
343 }
344 break;
345 }
346 case "LAST": output.Add(trades.Count == 0 ? "LAST none" : $"LAST {Mark} {trades[^1].Quantity}");
347 break;
348 case "VWAP": output.Add(trades.Count == 0 ? "VWAP none" : $"VWAP {Decimal(trades.Sum(t => t.Price * t.Quantity) * 100 / trades.Sum(t => t.Quantity))} volume {trades.Sum(t => t.Quantity)} trades {trades.Count}");
349 break;
350 case "SPREAD":
351 {
352 Order? b = Best("BUY"), a = Best("SELL");
353 string line = $"SPREAD bid {b?.Price?.ToString() ?? "none"} ask {a?.Price?.ToString() ?? "none"}";
354 if (a != null && b != null) {
355 int sum = a.Price!.Value + b.Price!.Value;
356 line += $" spread {a.Price - b.Price} mid {sum / 2}.{(sum % 2 == 0 ? 0 : 5)}";
357 }
358 output.Add(line);
359 break;
360 }
361 case "IMPACT":
362 {
363 int q = Num(2);
364 Require(q > 0, "invalid quantity");
365 var levels = Levels(Opposite(w[1]));
366 int rem = q, cost = 0, worst = 0, used = 0;
367 foreach (var l in levels) {
368 if (rem == 0) break;
369 int u = Math.Min(rem, l.Quantity);
370 rem -= u;
371 cost += u * l.Price;
372 worst = l.Price;
373 used++;
374 }
375 output.Add(rem > 0 ? $"IMPACT {w[1]} {q} insufficient {levels.Sum(l => l.Quantity)}" : $"IMPACT {w[1]} {q} avg {Decimal(cost * 100 / q)} worst {worst} levels {used}");
376 break;
377 }
378 case "TRADES": {
379 int k = Num(1);
380 Require(k > 0, "invalid count");
381 output.Add($"TRADES {Math.Min(k, trades.Count)} of {trades.Count}");
382 output.AddRange(trades.Skip(Math.Max(0, trades.Count - k)).Select(t => t.Line));
383 break;
384 }
385 case "BARS": {
386 int k = Num(1);
387 Require(k > 0, "invalid count");
388 output.Add($"BARS {(trades.Count + k - 1) / k}");
389 for (int i = 0;
390 i < trades.Count;
391 i += k) output.Add($"{i / k + 1} {Bars(trades.Skip(i).Take(k))}");
392 break;
393 }
394 case "RANK": {
395 var ranked = traders.Values.OrderByDescending(t => t.Pnl(Mark)).ThenByDescending(t => t.Volume).ThenBy(t => t.Name, StringComparer.Ordinal).ToList();
396 output.Add($"RANK {ranked.Count}");
397 for (int i = 0;
398 i < ranked.Count;
399 i++) output.Add($"{i + 1} {ranked[i].Name} pnl {ranked[i].Pnl(Mark)} volume {ranked[i].Volume}");
400 break;
401 }
402 case "FLOW": {
403 int b = trades.Where(t => t.Taker == t.Buy.Id).Sum(t => t.Quantity), s = trades.Where(t => t.Taker == t.Sell.Id).Sum(t => t.Quantity);
404 output.Add($"FLOW buy {b} sell {s} net {b - s}");
405 break;
406 }
407 case "SUMMARY": output.Add(trades.Count == 0 ? "SUMMARY trades 0" : $"SUMMARY trades {trades.Count} volume {trades.Sum(t => t.Quantity)} open {trades[0].Price} high {trades.Max(t => t.Price)} low {trades.Min(t => t.Price)} close {Mark} fees {trades.Sum(t => t.BuyFee + t.SellFee)}");
408 foreach (Trader t in traders.Values.OrderBy(t => t.Name, StringComparer.Ordinal)) output.Add(t.Position(Mark));
409 break;
410 }
411 }
412 public void Execute(string line)
413 {
414 string[] w = line.Split(' ');
415 try {
416 Command(w);
417 } catch (CommandError e) {
418 output.Add((w[0] is "LIMIT" or "MARKET" or "STOP" or "STOPLIMIT" or "ICEBERG" or "CANCEL" or "REPLACE" ? $"REJECT {w[1]} " : "ERROR ") + e.Message);
419 }
420 }
421 public void Write() {
422 Console.Write(string.Join("\n", output));
423 Console.Write('\n');
424 }
425}

428 int n = int.Parse(Console.ReadLine()!);
429 var engine = new Engine();
430 for (int i = 0;
431 i < n;

433 engine.Write();
434 }
