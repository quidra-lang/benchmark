39mutable struct MatchingEngine
40 orders::Vector{BookOrder}
41 ids::Dict{String, BookOrder}
42 traders::Dict{String, BookTrader}
43 trades::Vector{BookTrade}
44 priority::Int
45 maker::Int
46 taker::Int
47 band::Int

49 lot::Int

53struct CommandError <: Exception
54 reason::String
55end
56fail(s) = throw(CommandError(s))
57emit(xs...) = println(join(xs, " "))
58dec(n) = string(n÷100, ".", lpad(n%100, 2, '0'))
59resting(o) = o.status in ("OPEN", "PARTIAL")
60active(o) = resting(o) || o.status == "PENDING"
61lastprice(e) = isempty(e.trades) ? 0 : e.trades[end].price
62register(e, t) = get!(()->BookTrader(0, 0, 0, 0, nothing), e.traders, t)
63matchable(o, p) = o.price == 0 || (o.side == "BUY" ? p <= o.price : p >= o.price)
64triggered(o, p) = o.side == "BUY" ? p >= o.trigger : p <= o.trigger

66function exposure(e, t, s, excluded = nothing)
67 x = get(e.traders, t, nothing)
68 n = isnothing(x) ? 0 : x.bought-x.sold
69 (s == "BUY" ? n : -n)+sum((o.remaining for o in e.orders if o !== excluded && o.trader == t && o.side == s && active(o));
70 init = 0)
71end
72
73function risk(e, t, s, q, excluded = nothing)
74 x = get(e.traders, t, nothing)
75 !isnothing(x) && !isnothing(x.limit) && exposure(e, t, s, excluded)+q>x.limit
76end
77outside(e, p) = e.band>0 && !isempty(e.trades) && abs(p-lastprice(e))>e.band
78
79function book(e, s)
80 sort([o for o in e.orders if resting(o) && o.side == s];
81 by = o->(s == "BUY" ? -o.price : o.price, o.priority))
82end
83best(e, s) = let b = book(e, s)
84 isempty(b) ? nothing : b[1]
85end
86orderline(o) = emit("ORDER", o.id, o.status, "$(o.filled)/$(o.quantity)")
87
88function details(o)
89 p = o.price == 0 ? "-" : string(o.price)
90 a = o.filled == 0 ? "-" : dec(o.value*100÷o.filled)
91 s = "$(o.id) $(o.trader) $(o.kind) $(o.side) $p $(o.status) $(o.filled)/$(o.quantity) avg $a"
92 o.trigger>0 ? s*" trigger $(o.trigger)" : s
93end

95function showposition(e, t)
96 x = e.traders[t]
97 n = x.bought-x.sold
98 emit("POSITION", t, "bought", x.bought, "sold", x.sold, "net", n, "cash", x.cash, "fees", x.fees, "pnl", x.cash+n*lastprice(e))
99end
100tradeline(e, i) = let x = e.trades[i]
101 emit("TRADE", i, x.price, x.quantity, x.buy.id, x.buy.trader, x.sell.id, x.sell.trader)
102end

104function execute!(e, o)
105 op = o.side == "BUY" ? "SELL" : "BUY"
106 while o.remaining>0
107 y = best(e, op)
108 (isnothing(y) || !matchable(o, y.price)) && break
109 if y.trader == o.trader
110 emit("STP", y.id, y.remaining)
111 push!(y.history, "STP $(y.remaining)")
112 y.remaining = 0
113 y.status = "CANCELLED"
114 continue
115 end
116 q = min(o.remaining, y.visible)
117 p = y.price
118 tf = q*e.taker
119 mf = q*e.maker
120 b, s, bf, sf = o.side == "BUY" ? (o, y, tf, mf) : (y, o, mf, tf)
121 push!(e.trades, BookTrade(p, q, b, s, o, bf, sf))
122 i = length(e.trades)
123 for x in (o, y)
124 x.remaining -= q
125 x.filled += q
126 x.value += q*p
127 push!(x.history, "FILL $i $p $q")
128 end
129 y.visible -= q
130 bt = e.traders[b.trader]
131 st = e.traders[s.trader]
132 bt.bought += q
133 bt.cash -= q*p+bf
134 bt.fees += bf
135 st.sold += q
136 st.cash += q*p-sf
137 st.fees += sf
138 tradeline(e, i)
139 if y.remaining == 0
140 y.status = "FILLED"
141 else
142 y.status = "PARTIAL"
143 if y.visible == 0
144 e.priority += 1
145 y.priority = e.priority
146 y.visible = min(y.peak, y.remaining)
147 h = y.remaining-y.visible
148 push!(y.history, "REFRESH $(y.visible) $h")
149 emit("REFRESH", y.id, y.visible, h)
150 end
151 end
152 end
153end
154
155function post!(e, o, force = false)
156 if o.remaining == 0
157 o.status = "FILLED"
158 elseif force || (o.kind == "LIMIT" && o.condition != "IOC") || o.kind in ("ICEBERG", "STOPLIMIT")
159 e.priority += 1
160 o.priority = e.priority
161 o.visible = o.peak>0 ? min(o.peak, o.remaining) : o.remaining
162 o.status = o.filled>0 ? "PARTIAL" : "OPEN"
163 push!(o.history, "REST $(o.price) $(o.visible)")
164 else
165 push!(o.history, "EXPIRE $(o.remaining)")
166 o.remaining = 0
167 o.status = "CANCELLED"
168 end
169 orderline(o)
170end

172function stops!(e)
173 while !isempty(e.trades)
174 p = lastprice(e)
175 selected = [o for o in e.orders if o.status == "PENDING" && triggered(o, p)]
176 isempty(selected) && return
177 for o in selected
178 o.status = "TRIGGERING"
179 end
180 for o in selected
181 emit("TRIGGER", o.id)
182 push!(o.history, "TRIGGER")
183 execute!(e, o)
184 post!(e, o)
185 end
186 end
187end
188
189function cancel!(o)
190 push!(o.history, "CANCEL $(o.remaining)")
191 o.remaining = 0
192 o.status = "CANCELLED"
193 orderline(o)
194end

196function levels(e, s)
197 d = Dict{Int, Tuple{Int, Int}}()
198 for o in book(e, s)
199 q, n = get(d, o.price, (0, 0))
200 d[o.price] = (q+o.visible, n+1)
201 end
202 [(p, d[p]...) for p in sort(collect(keys(d));
203 rev = s == "BUY")]
204end

206function command!(e, w)
207 c = w[1]
208 n(i) = parse(Int, w[i])
209 if c in ("LIMIT", "MARKET", "STOP", "STOPLIMIT", "ICEBERG")
210 p = r = k = 0
211 condition = ""
212 if c == "MARKET"
213 q = n(5)
214 elseif c == "STOP"
215 r = n(5)
216 q = n(6)
217 elseif c == "STOPLIMIT"
218 r = n(5)
219 p = n(6)
220 q = n(7)
221 else
222 p = n(5)
223 q = n(6)
224 if c == "ICEBERG"
225 k = n(7)
226 elseif length(w) == 7
227 condition = w[7]
228 end
229 end
230 o = BookOrder(w[2], w[3], c, w[4], condition, "ARRIVING", p, r, k, q, q, 0, 0, 0, 0, ["ACCEPT"])
231 prices = c == "MARKET" ? Int[] : c == "STOP" ? [r] : c == "STOPLIMIT" ? [r, p] : [p]
232 e.halted && fail("halted")
233 haskey(e.ids, o.id) && fail("duplicate id")
234 any(==(0), prices) && fail("invalid price")
235 q == 0 && fail("invalid quantity")
236 any(x->x%e.tick != 0, prices) && fail("invalid tick")
237 q%e.lot != 0 && fail("invalid lot")
238 r>0 && !isempty(e.trades) && triggered(o, lastprice(e)) && fail("invalid trigger")
239 c == "ICEBERG" && (k == 0 || k>q || k%e.lot != 0) && fail("invalid peak")
240 p>0 && outside(e, p) && fail("outside band")
241 risk(e, o.trader, o.side, q) && fail("risk limit")
242 b = book(e, o.side == "BUY" ? "SELL" : "BUY")
243 condition == "POSTONLY" && !isempty(b) && matchable(o, b[1].price) && fail("would cross")
244 condition == "FOK" && sum((x.remaining for x in b if x.trader != o.trader && matchable(o, x.price));
245 init = 0)<q && fail("unfillable")
246 register(e, o.trader)
247 push!(e.orders, o)
248 e.ids[o.id] = o
249 if r>0
250 o.status = "PENDING"
251 orderline(o)
252 else
253 condition != "POSTONLY" && execute!(e, o)
254 post!(e, o)
255 stops!(e)
256 end
257 return
258 end
259 if c in ("CANCEL", "REPLACE", "STATUS", "HISTORY")
260 c == "REPLACE" && e.halted && fail("halted")
261 haskey(e.ids, w[2]) || fail("unknown order")
262 o = e.ids[w[2]]
263 if c == "STATUS"
264 emit(c, details(o))
265 elseif c == "HISTORY"
266 emit(c, o.id, length(o.history))
267 foreach(emit, o.history)
268 elseif c == "CANCEL"
269 active(o) || fail("not open")
270 cancel!(o)
271 else
272 p = n(3)
273 q = n(4)
274 moved = p != o.price
275 resting(o) || fail("not open")
276 p == 0 && fail("invalid price")
277 q == 0 && fail("invalid quantity")
278 moved && p%e.tick != 0 && fail("invalid tick")
279 q%e.lot != 0 && fail("invalid lot")
280 !moved && q == o.remaining && fail("no change")
281 moved && outside(e, p) && fail("outside band")
282 q>o.remaining && risk(e, o.trader, o.side, q, o) && fail("risk limit")
283 kept = !moved && q<o.remaining
284 mode = kept ? "kept" : "lost"
285 o.quantity = o.filled+q
286 o.price = p
287 o.remaining = q
288 push!(o.history, "REPLACE $p $q $mode")
289 emit(c, o.id, mode)
290 if kept
291 o.visible = min(o.visible, q)
292 orderline(o)
293 else
294 o.status = "ARRIVING"
295 execute!(e, o)
296 post!(e, o, true)
297 stops!(e)
298 end
299 end
300 return
301 end
302 if c in ("CANCELALL", "ORDERS", "POSITION", "FILLS")
303 t = w[2]
304 haskey(e.traders, t) || fail("unknown trader")
305 if c in ("CANCELALL", "ORDERS")
306 a = [o for o in e.orders if o.trader == t && active(o)]
307 emit(c, t, length(a))
308 for o in a
309 c == "CANCELALL" ? cancel!(o) : emit(details(o))
310 end
311 elseif c == "POSITION"
312 showposition(e, t)
313 else
314 a = [i for i in eachindex(e.trades) if e.trades[i].buy.trader == t || e.trades[i].sell.trader == t]
315 emit(c, t, length(a))
316 net = 0
317 for i in a
318 x = e.trades[i]
319 buy = x.buy.trader == t
320 o = buy ? x.buy : x.sell
321 net += buy ? x.quantity : -x.quantity
322 emit(i, buy ? "BUY" : "SELL", x.price, x.quantity, o.id, x.taker === o ? "TAKER" : "MAKER", "fee", buy ? x.bf : x.sf, "net", net)
323 end
324 end
325 return
326 end
327 if c == "HALT"
328 e.halted && fail("already halted")
329 e.halted = true
330 emit(c)
331 elseif c == "RESUME"
332 !e.halted && fail("not halted")
333 e.halted = false
334 emit(c)
335 elseif c == "RISK"
336 t = w[2]
337 m = n(3)
338 register(e, t).limit = m
339 emit(c, t, m, "exposure buy", exposure(e, t, "BUY"), "sell", exposure(e, t, "SELL"))
340 elseif c == "FEE"
341 a = n(2)
342 b = n(3)
343 a+b<0 && fail("invalid fee")
344 e.maker = a
345 e.taker = b
346 emit(c, "maker", a, "taker", b)
347 elseif c == "BAND"
348 e.band = n(2)
349 emit(c, e.band)
350 elseif c in ("TICK", "LOT")
351 u = n(2)
352 u == 0 && fail("invalid size")
353 c == "TICK" ? (e.tick = u) : (e.lot = u)
354 emit(c, u)
355 elseif c == "BOOK"
356 k = n(2)
357 k == 0 && fail("invalid depth")
358 b = levels(e, "BUY")
359 a = levels(e, "SELL")
360 emit(c, "bids", length(b), "asks", length(a))
361 for x in reverse(a[1:min(k, length(a))])
362 emit("ASK", x...)
363 end
364 for x in b[1:min(k, length(b))]
365 emit("BID", x...)
366 end
367 elseif c == "LAST"
368 isempty(e.trades) ? emit(c, "none") : emit(c, e.trades[end].price, e.trades[end].quantity)
369 elseif c == "VWAP"
370 if isempty(e.trades)
371 emit(c, "none")
372 else
373 v = sum(x.quantity for x in e.trades)
374 a = sum(x.price*x.quantity for x in e.trades)
375 emit(c, dec(a*100÷v), "volume", v, "trades", length(e.trades))
376 end
377 elseif c == "SPREAD"
378 b = best(e, "BUY")
379 a = best(e, "SELL")
380 if isnothing(b) || isnothing(a)
381 emit(c, "bid", isnothing(b) ? "none" : b.price, "ask", isnothing(a) ? "none" : a.price)
382 else
383 s = b.price+a.price
384 emit(c, "bid", b.price, "ask", a.price, "spread", a.price-b.price, "mid", "$(s÷2).$(s%2*5)")
385 end
386 elseif c == "IMPACT"
387 s = w[2]
388 q = n(3)
389 q == 0 && fail("invalid quantity")
390 r = q
391 cost = worst = used = total = 0
392 for (p, v, _) in levels(e, s == "BUY" ? "SELL" : "BUY")
393 total += v
394 if r>0
395 u = min(r, v)
396 r -= u
397 cost += u*p
398 worst = p
399 used += 1
400 end
401 end
402 r>0 ? emit(c, s, q, "insufficient", total) : emit(c, s, q, "avg", dec(cost*100÷q), "worst", worst, "levels", used)
403 elseif c in ("TRADES", "BARS")
404 k = n(2)
405 k == 0 && fail("invalid count")
406 T = length(e.trades)
407 if c == "TRADES"
408 m = min(k, T)
409 emit(c, m, "of", T)
410 for i in T-m+1:T
411 tradeline(e, i)
412 end
413 else
414 emit(c, cld(T, k))
415 for i in 1:k:T
416 g = e.trades[i:min(i+k-1, T)]
417 p = [x.price for x in g]
418 emit((i-1)÷k+1, "open", p[1], "high", maximum(p), "low", minimum(p), "close", p[end], "volume", sum(x.quantity for x in g))
419 end
420 end
421 elseif c == "RANK"
422 pnl(t) = let x = e.traders[t]
423 x.cash+(x.bought-x.sold)*lastprice(e)
424 end
425 names = sort(collect(keys(e.traders));
426 by = t->(-pnl(t), -e.traders[t].bought-e.traders[t].sold, t))
427 emit(c, length(names))
428 for (i, t) in enumerate(names)
429 x = e.traders[t]
430 emit(i, t, "pnl", pnl(t), "volume", x.bought+x.sold)
431 end
432 elseif c == "FLOW"
433 b = sum((x.quantity for x in e.trades if x.taker.side == "BUY");
434 init = 0)
435 s = sum((x.quantity for x in e.trades if x.taker.side == "SELL");
436 init = 0)
437 emit(c, "buy", b, "sell", s, "net", b-s)
438 elseif c == "SUMMARY"
439 if isempty(e.trades)
440 emit(c, "trades", 0)
441 else
442 p = [x.price for x in e.trades]
443 emit(c, "trades", length(p), "volume", sum(x.quantity for x in e.trades), "open", p[1], "high", maximum(p), "low", minimum(p), "close", p[end], "fees", sum(x.bf+x.sf for x in e.trades))
444 end
445 for t in sort(collect(keys(e.traders)))
446 showposition(e, t)
447 end
448 end
449end

451function main()
452 e = MatchingEngine(BookOrder[], Dict{String, BookOrder}(), Dict{String, BookTrader}(), BookTrade[], 0, 0, 0, 0, 1, 1, false)
453 for _ in 1:parse(Int, readline())
454 w = String.(split(readline()))
455 try
456 command!(e, w)
457 catch err
458 err isa CommandError || rethrow()
459 w[1] in ("LIMIT", "MARKET", "STOP", "STOPLIMIT", "ICEBERG", "CANCEL", "REPLACE") ? emit("REJECT", w[2], err.reason) : emit("ERROR", err.reason)
460 end
461 end
462end
