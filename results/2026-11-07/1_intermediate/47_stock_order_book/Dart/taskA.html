1import 'dart:io';
2import 'dart:math';
3class Rejection implements Exception {
4 final String message;
5 Rejection(this.message);
6}
7void check(bool bad, String reason) {
8 if (bad) throw Rejection(reason);
9}
10String decimal(int n) => '${n ~/ 100}.${(n % 100).toString().padLeft(2, '0')}';
11class Order {
12 final String id, trader, type, side;
13 String condition = '', status = 'OPEN';
14 int price = 0, trigger = 0, peak = 0, quantity, left, visible = 0, filled = 0, value = 0, priority = 0;
15 final List<String> history = ['ACCEPT'];
16 Order(this.id, this.trader, this.type, this.side, this.quantity) : left = quantity;
17 bool get resting => status == 'OPEN' || status == 'PARTIAL';
18 bool get active => resting || status == 'PENDING';
19 bool matches(int p) => price == 0 || (side == 'BUY' ? p <= price : p >= price);
20 bool triggered(int p) => side == 'BUY' ? p >= trigger : p <= trigger;
21 String get line => 'ORDER $id $status $filled/$quantity';
22 String get details => '$id $trader $type $side ${price == 0 ? '-' : price} $status $filled/$quantity avg ${filled == 0 ? '-' : decimal(value * 100 ~/ filled)}${trigger == 0 ? '' : ' trigger $trigger'}';
23 void fill(int number, int p, int q) {
24 left -= q;
25 filled += q;
26 value += p * q;
27 history.add('FILL $number $p $q');
28 }
29}
30class Trader {
31 final String name;
32 int bought = 0, sold = 0, cash = 0, fees = 0;
33 int? limit;
34 Trader(this.name);
35 int get net => bought - sold;
36 int get volume => bought + sold;
37 int pnl(int mark) => cash + net * mark;
38 String position(int mark) => 'POSITION $name bought $bought sold $sold net $net cash $cash fees $fees pnl ${pnl(mark)}';
39}
40class Trade {
41 final int number, price, quantity, buyFee, sellFee;
42 final Order buy, sell, taker;
43 Trade(this.number, this.price, this.quantity, this.buy, this.sell, this.taker, this.buyFee, this.sellFee);
44 String get line => 'TRADE $number $price $quantity ${buy.id} ${buy.trader} ${sell.id} ${sell.trader}';
45}
46class Level {
47 final int price;
48 int quantity = 0, count = 0;
49 Level(this.price);
50}
51class Engine {
52 final Map<String, Order> orders = {};
53 final Map<String, Trader> traders = {};
54 final List<Order> pending = [], buys = [], sells = [];
55 final List<Trade> trades = [];
56 final StringBuffer output = StringBuffer();
57 int makerFee = 0, takerFee = 0, band = 0, tick = 1, lot = 1, priority = 0;
58 bool halted = false;
59 void emit(List<Object> parts) => output.writeln(parts.join(' '));
60 List<Order> book(String side) => side == 'BUY' ? buys : sells;
61 List<Order> opposite(String side) => side == 'BUY' ? sells : buys;
62 int compare(Order a, Order b, String side) {
63 final p = side == 'BUY' ? b.price.compareTo(a.price) : a.price.compareTo(b.price);
64 return p == 0 ? a.priority.compareTo(b.priority) : p;
65 }
66 Order? best(List<Order> b) {
67 Order? result;
68 for (final o in b) {
69 if (result == null || compare(o, result, identical(b, buys) ? 'BUY' : 'SELL') < 0) result = o;
70 }
71 return result;
72 }
73 int get last => trades.isEmpty ? 0 : trades.last.price;
74 Trader register(String name) => traders.putIfAbsent(name, () => Trader(name));
75 Trader trader(String name) {
76 check(!traders.containsKey(name), 'unknown trader');
77 return traders[name]!;
78 }
79 Order order(String id) {
80 check(!orders.containsKey(id), 'unknown order');
81 return orders[id]!;
82 }
83 List<Order> active(String name) => orders.values.where((o) => o.trader == name && o.active).toList();
84 int exposure(String name, String side, [Order? excluded]) {
85 int n = side == 'BUY' ? traders[name]!.net : -traders[name]!.net;
86 for (final o in active(name)) {
87 if (o != excluded && o.side == side) n += o.left;
88 }
89 return n;
90 }
91 void risk(String name, String side, int q, [Order? excluded]) {
92 final limit = traders[name]?.limit;
93 if (limit != null) check(exposure(name, side, excluded) + q > limit, 'risk limit');
94 }
95 void checkBand(int p) => check(band > 0 && last != 0 && (p - last).abs() > band, 'outside band');
96 void match(Order x) {
97 final b = opposite(x.side);
98 while (x.left > 0) {
99 final y = best(b);
100 if (y == null || !x.matches(y.price)) break;
101 if (x.trader == y.trader) {
102 emit(['STP', y.id, y.left]);
103 y.history.add('STP ${y.left}');
104 b.remove(y);
105 y.left = 0;
106 y.status = 'CANCELLED';
107 continue;
108 }
109 final q = min(x.left, y.visible), p = y.price;
110 final buy = x.side == 'BUY' ? x : y, sell = x.side == 'BUY' ? y : x;
111 final xf = q * takerFee, yf = q * makerFee;
112 final t = Trade(trades.length + 1, p, q, buy, sell, x, buy == x ? xf : yf, sell == x ? xf : yf);
113 trades.add(t);
114 x.fill(t.number, p, q);
115 y.fill(t.number, p, q);
116 y.visible -= q;
117 final bt = traders[buy.trader]!, st = traders[sell.trader]!;
118 bt.bought += q;
119 bt.cash -= p * q + t.buyFee;
120 bt.fees += t.buyFee;
121 st.sold += q;
122 st.cash += p * q - t.sellFee;
123 st.fees += t.sellFee;
124 emit([t.line]);
125 if (y.left == 0) {
126 b.remove(y);
127 y.status = 'FILLED';
128 }
129 else {
130 y.status = 'PARTIAL';
131 if (y.visible == 0) {
132 y.priority = ++priority;
133 y.visible = min(y.peak, y.left);
134 y.history.add('REFRESH ${y.visible} ${y.left - y.visible}');
135 emit(['REFRESH', y.id, y.visible, y.left - y.visible]);
136 }
137 }
138 }
139 }
140 void post(Order o, [bool forceRest = false]) {
141 if (o.left == 0) {
142 o.status = 'FILLED';
143 }
144 else if (forceRest || o.type == 'ICEBERG' || o.type == 'STOPLIMIT' || (o.type == 'LIMIT' && o.condition != 'IOC')) {
145 o.priority = ++priority;
146 o.visible = o.peak == 0 ? o.left : min(o.peak, o.left);
147 book(o.side).add(o);
148 o.status = o.filled > 0 ? 'PARTIAL' : 'OPEN';
149 o.history.add('REST ${o.price} ${o.visible}');
150 } else {
151 o.history.add('EXPIRE ${o.left}');
152 o.left = 0;
153 o.status = 'CANCELLED';
154 }
155 emit([o.line]);
156 }
157 void trigger() {
158 while (trades.isNotEmpty) {
159 final p = last, batch = pending.where((o) => o.triggered(last)).toList();
160 if (batch.isEmpty) return;
161 pending.removeWhere((o) => o.triggered(p));
162 for (final o in batch) {
163 emit(['TRIGGER', o.id]);
164 o.history.add('TRIGGER');
165 match(o);
166 post(o);
167 }
168 }
169 }
170 void cancel(Order o) {
171 if (o.status == 'PENDING') {
172 pending.remove(o);
173 } else {
174 book(o.side).remove(o);
175 }
176 o.history.add('CANCEL ${o.left}');
177 o.left = 0;
178 o.status = 'CANCELLED';
179 emit([o.line]);
180 }
181 void newOrder(List<String> a) {
182 final kind = a[0], market = a[0] == 'MARKET', stop = a[0] == 'STOP', sl = a[0] == 'STOPLIMIT', ice = a[0] == 'ICEBERG';
183 final p = market || stop ? 0 : int.parse(a[sl ? 5 : 4]), tr = stop || sl ? int.parse(a[4]) : 0;
184 final q = int.parse(a[market ? 4 : sl ? 6 : 5]);
185 final o = Order(a[1], a[2], kind, a[3], q)..price = p..trigger = tr;
186 if (ice) o.peak = int.parse(a[6]);
187 if (kind == 'LIMIT' && a.length == 7) o.condition = a[6];
188 check(halted, 'halted');
189 check(orders.containsKey(o.id), 'duplicate id');
190 check((!market && !stop && p == 0) || ((stop || sl) && tr == 0), 'invalid price');
191 check(q == 0, 'invalid quantity');
192 check(p % tick != 0 || tr % tick != 0, 'invalid tick');
193 check(q % lot != 0, 'invalid lot');
194 if (ice) check(o.peak == 0 || o.peak > q || o.peak % lot != 0, 'invalid peak');
195 if (stop || sl) check(last != 0 && o.triggered(last), 'invalid trigger');
196 if (p != 0) checkBand(p);
197 risk(o.trader, o.side, q);
198 final b = best(opposite(o.side));
199 if (o.condition == 'POSTONLY') check(b != null && o.matches(b.price), 'would cross');
200 if (o.condition == 'FOK') {
201 int total = 0;
202 for (final y in opposite(o.side)) {
203 if (y.trader != o.trader && o.matches(y.price)) total += y.left;
204 }
205 check(total < q, 'unfillable');
206 }
207 register(o.trader);
208 orders[o.id] = o;
209 if (stop || sl) {
210 o.status = 'PENDING';
211 pending.add(o);
212 emit([o.line]);
213 }
214 else {
215 if (o.condition != 'POSTONLY') match(o);
216 post(o);
217 trigger();
218 }
219 }
220 List<Level> levels(String side) {
221 final sorted = List<Order>.of(book(side))..sort((a, b) => compare(a, b, side));
222 final result = <Level>[];
223 for (final o in sorted) {
224 if (result.isEmpty || result.last.price != o.price) result.add(Level(o.price));
225 result.last.quantity += o.visible;
226 result.last.count++;
227 }
228 return result;
229 }
230 int positive(String text, String reason) {
231 final n = int.parse(text);
232 check(n == 0, reason);
233 return n;
234 }
235 String bar(List<Trade> ts) {
236 int high = ts.first.price, low = high, volume = 0;
237 for (final t in ts) {
238 high = max(high, t.price);
239 low = min(low, t.price);
240 volume += t.quantity;
241 }
242 return 'open ${ts.first.price} high $high low $low close ${ts.last.price} volume $volume';
243 }
244 void execute(String line) {
245 final a = line.split(' '), c = line.split(' ')[0];
246 try {
247 switch (c) {
248 case 'LIMIT':
249 case 'MARKET':
250 case 'STOP':
251 case 'STOPLIMIT':
252 case 'ICEBERG': {
253 newOrder(a);
254 }
255 case 'CANCEL': {
256 final o = order(a[1]);
257 check(!o.active, 'not open');
258 cancel(o);
259 }
260 case 'REPLACE': {
261 check(halted, 'halted');
262 final o = order(a[1]);
263 check(!o.resting, 'not open');
264 final p = int.parse(a[2]), q = int.parse(a[3]), moved = int.parse(a[2]) != o.price;
265 check(p == 0, 'invalid price');
266 check(q == 0, 'invalid quantity');
267 check(moved && p % tick != 0, 'invalid tick');
268 check(q % lot != 0, 'invalid lot');
269 check(!moved && q == o.left, 'no change');
270 if (moved) checkBand(p);
271 if (q > o.left) risk(o.trader, o.side, q, o);
272 o.quantity = o.filled + q;
273 final kept = !moved && q < o.left;
274 o.history.add('REPLACE $p $q ${kept ? 'kept' : 'lost'}');
275 emit(['REPLACE', o.id, kept ? 'kept' : 'lost']);
276 if (kept) {
277 o.left = q;
278 o.visible = min(o.visible, q);
279 emit([o.line]);
280 }
281 else {
282 book(o.side).remove(o);
283 o.price = p;
284 o.left = q;
285 match(o);
286 post(o, true);
287 trigger();
288 }
289 }
290 case 'CANCELALL': {
291 trader(a[1]);
292 final xs = active(a[1]);
293 emit([c, a[1], xs.length]);
294 for (final o in xs) {
295 cancel(o);
296 }
297 }
298 case 'HALT': {
299 check(halted, 'already halted');
300 halted = true;
301 emit([c]);
302 }
303 case 'RESUME': {
304 check(!halted, 'not halted');
305 halted = false;
306 emit([c]);
307 }
308 case 'RISK': {
309 register(a[1]).limit = int.parse(a[2]);
310 emit([c, a[1], a[2], 'exposure buy', exposure(a[1], 'BUY'), 'sell', exposure(a[1], 'SELL')]);
311 }
312 case 'FEE': {
313 final m = int.parse(a[1]), t = int.parse(a[2]);
314 check(m + t < 0, 'invalid fee');
315 makerFee = m;
316 takerFee = t;
317 emit(['FEE maker', m, 'taker', t]);
318 }
319 case 'BAND': {
320 band = int.parse(a[1]);
321 emit([c, band]);
322 }
323 case 'TICK':
324 case 'LOT': {
325 final v = positive(a[1], 'invalid size');
326 if (c == 'TICK') {
327 tick = v;
328 } else {
329 lot = v;
330 }
331 emit([c, v]);
332 }
333 case 'BOOK': {
334 final n = positive(a[1], 'invalid depth'), bids = levels('BUY'), asks = levels('SELL');
335 emit(['BOOK bids', bids.length, 'asks', asks.length]);
336 for (final l in asks.take(n).toList().reversed) {
337 emit(['ASK', l.price, l.quantity, l.count]);
338 }
339 for (final l in bids.take(n)) {
340 emit(['BID', l.price, l.quantity, l.count]);
341 }
342 }
343 case 'ORDERS': {
344 trader(a[1]);
345 final xs = active(a[1]);
346 emit([c, a[1], xs.length]);
347 for (final o in xs) {
348 emit([o.details]);
349 }
350 }
351 case 'STATUS': {
352 emit([c, order(a[1]).details]);
353 }
354 case 'HISTORY': {
355 final o = order(a[1]);
356 emit([c, o.id, o.history.length]);
357 for (final h in o.history) {
358 emit([h]);
359 }
360 }
361 case 'POSITION': {
362 emit([trader(a[1]).position(last)]);
363 }
364 case 'FILLS': {
365 trader(a[1]);
366 final ts = trades.where((t) => t.buy.trader == a[1] || t.sell.trader == a[1]).toList();
367 emit([c, a[1], ts.length]);
368 int net = 0;
369 for (final t in ts) {
370 final buy = t.buy.trader == a[1], o = t.buy.trader == a[1] ? t.buy : t.sell;
371 net += buy ? t.quantity : -t.quantity;
372 emit([t.number, o.side, t.price, t.quantity, o.id, t.taker == o ? 'TAKER' : 'MAKER', 'fee', buy ? t.buyFee : t.sellFee, 'net', net]);
373 }
374 }
375 case 'LAST': {
376 if (trades.isEmpty) {
377 emit([c, 'none']);
378 } else {
379 emit([c, last, trades.last.quantity]);
380 }
381 }
382 case 'VWAP': {
383 int v = 0, amount = 0;
384 for (final t in trades) {
385 v += t.quantity;
386 amount += t.price * t.quantity;
387 }
388 if (v == 0) {
389 emit([c, 'none']);
390 } else {
391 emit([c, decimal(amount * 100 ~/ v), 'volume', v, 'trades', trades.length]);
392 }
393 }
394 case 'SPREAD': {
395 final b = best(buys), s = best(sells);
396 if (b == null || s == null) {
397 emit(['SPREAD bid', b?.price ?? 'none', 'ask', s?.price ?? 'none']);
398 } else {
399 final sum = b.price + s.price;
400 emit(['SPREAD bid', b.price, 'ask', s.price, 'spread', s.price - b.price, 'mid', '${sum ~/ 2}.${sum % 2 == 0 ? 0 : 5}']);
401 }
402 }
403 case 'IMPACT': {
404 final q = positive(a[2], 'invalid quantity');
405 int left = q, cost = 0, worst = 0, used = 0, total = 0;
406 for (final l in levels(a[1] == 'BUY' ? 'SELL' : 'BUY')) {
407 total += l.quantity;
408 if (left > 0) {
409 final u = min(left, l.quantity);
410 left -= u;
411 cost += u * l.price;
412 worst = l.price;
413 used++;
414 }
415 }
416 if (left > 0) {
417 emit([c, a[1], q, 'insufficient', total]);
418 } else {
419 emit([c, a[1], q, 'avg', decimal(cost * 100 ~/ q), 'worst', worst, 'levels', used]);
420 }
421 }
422 case 'TRADES': {
423 final n = positive(a[1], 'invalid count'), m = min(n, trades.length);
424 emit([c, m, 'of', trades.length]);
425 for (final t in trades.skip(trades.length - m)) {
426 emit([t.line]);
427 }
428 }
429 case 'BARS': {
430 final k = positive(a[1], 'invalid count');
431 emit([c, (trades.length + k - 1) ~/ k]);
432 for (int i = 0; i < trades.length; i += k) {
433 emit([i ~/ k + 1, bar(trades.sublist(i, min(i + k, trades.length)))]);
434 }
435 }
436 case 'RANK': {
437 final ts = traders.values.toList()..sort((a, b) {
438 int r = b.pnl(last).compareTo(a.pnl(last));
439 if (r == 0) r = b.volume.compareTo(a.volume);
440 return r == 0 ? a.name.compareTo(b.name) : r;
441 });
442 emit([c, ts.length]);
443 for (int i = 0; i < ts.length; i++) {
444 final t = ts[i];
445 emit([i + 1, t.name, 'pnl', t.pnl(last), 'volume', t.volume]);
446 }
447 }
448 case 'FLOW': {
449 int buy = 0, sell = 0;
450 for (final t in trades) {
451 if (t.taker == t.buy) {
452 buy += t.quantity;
453 } else {
454 sell += t.quantity;
455 }
456 }
457 emit(['FLOW buy', buy, 'sell', sell, 'net', buy - sell]);
458 }
459 case 'SUMMARY': {
460 if (trades.isEmpty) {
461 emit(['SUMMARY trades 0']);
462 } else {
463 int volume = 0, fees = 0;
464 for (final t in trades) {
465 volume += t.quantity;
466 fees += t.buyFee + t.sellFee;
467 }
468 final b = bar(trades);
469 emit(['SUMMARY trades', trades.length, 'volume', volume, b.substring(0, b.lastIndexOf(' volume')), 'fees', fees]);
470 }
471 final names = traders.keys.toList()..sort();
472 for (final name in names) {
473 emit([traders[name]!.position(last)]);
474 }
475 }
476 }
477 }
478 on Rejection catch (e) {
479 emit([if (['LIMIT', 'MARKET', 'STOP', 'STOPLIMIT', 'ICEBERG', 'CANCEL', 'REPLACE'].contains(c)) 'REJECT ${a[1]}' else 'ERROR', e.message]);
480 }
481 }
482}
483void main() {
484 final e = Engine(), n = int.parse(stdin.readLineSync()!);
485 for (int i = 0; i < n; i++) {
486 e.execute(stdin.readLineSync()!);
487 }
488 stdout.write(e.output);
489}
