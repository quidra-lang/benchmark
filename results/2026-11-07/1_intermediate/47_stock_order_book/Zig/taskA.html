98fn eq(a: Text, b: Text) bool {
99 return std.mem.eql(u8, a, b);
100}
101
102fn number(s: Text) I {
103 return std.fmt.parseInt(I, s, 10) catch unreachable;
104}
105
106fn abs(x: I) I {
107 return if (x < 0) -x else x;
108}
109
110fn resting(o: Order) bool {
111 return o.status == .OPEN or o.status == .PARTIAL;
112}
113
114fn active(o: Order) bool {
115 return resting(o) or o.status == .PENDING;
116}

118fn matchable(sell: bool, price: I, other: I) bool {
119 return price < 0 or (if (sell) other >= price else other <= price);
120}

123 out: *std.Io.Writer,

135 fn format(e: *Engine, comptime f: Text, args: anytype) !Text {
136 return std.fmt.allocPrint(e.gpa, f, args);

143 fn history(e: *Engine, i: usize, comptime f: Text, args: anytype) !void {

148 fn findOrder(e: *const Engine, id: Text) ?usize {
149 for (e.orders.items, 0..) |o, i| if (eq(o.id, id)) return i;
150 return null;
151 }
152
153 fn findTrader(e: *const Engine, name: Text) ?usize {
154 for (e.traders.items, 0..) |t, i| if (eq(t.name, name)) return i;
155 return null;
156 }
157
158 fn register(e: *Engine, name: Text) !usize {
159 if (e.findTrader(name)) |i| return i;
160 const i = e.traders.items.len;
161 try e.traders.append(e.gpa, .{ .name = name });
162 return i;
163 }
164
165 fn last(e: *const Engine) I {
166 return if (e.trades.items.len == 0) 0 else e.trades.items[e.trades.items.len - 1].price;
167 }
168
169 fn net(e: *const Engine, t: usize) I {
170 return e.traders.items[t].bought - e.traders.items[t].sold;
171 }
172
173 fn pnl(e: *const Engine, t: usize) I {
174 return e.traders.items[t].cash + e.net(t) * e.last();
175 }
176
177 fn exposure(e: *const Engine, t: usize, sell: bool, exclude: ?usize) I {
178 var value = if (sell) -e.net(t) else e.net(t);
179 for (e.orders.items, 0..) |o, i| {
180 if (exclude != null and exclude.? == i) continue;
181 if (active(o) and o.trader == t and o.sell == sell) value += o.remaining;
182 }
183 return value;
184 }
185
186 fn risk(e: *const Engine, t: ?usize, sell: bool, q: I, exclude: ?usize) bool {
187 if (t) |i| {
188 if (e.traders.items[i].limit) |limit| return e.exposure(i, sell, exclude) + q > limit;
189 }
190 return false;
191 }
192
193 fn outside(e: *const Engine, p: I) bool {
194 return e.band > 0 and e.trades.items.len > 0 and abs(p - e.last()) > e.band;
195 }
196
197 fn best(e: *const Engine, sell: bool) ?usize {
198 var result: ?usize = null;
199 for (e.orders.items, 0..) |o, i| {
200 if (!resting(o) or o.sell != sell) continue;
201 if (result) |b| {
202 const previous = e.orders.items[b];
203 if ((if (sell) o.price < previous.price else o.price > previous.price) or (o.price == previous.price and o.priority < previous.priority)) result = i;
204 } else result = i;
205 }
206 return result;
207 }

209 fn decimal(e: *Engine, value: I) !Text {
210 return e.format("{d}.{d:0>2}", .{ @divTrunc(value, 100), @mod(value, 100) });
211 }

213 fn orderLine(e: *Engine, i: usize) !void {
214 const o = e.orders.items[i];
215 try e.line("ORDER {s} {s} {d}/{d}", .{ o.id, @tagName(o.status), o.filled, o.quantity });
216 }
217
218 fn details(e: *Engine, i: usize) !void {
219 const o = e.orders.items[i];
220 const p = if (o.price < 0) "-" else try e.format("{d}", .{o.price});
221 const avg = if (o.filled == 0) "-" else try e.decimal(@divTrunc(o.value * 100, o.filled));
222 try e.out.print("{s} {s} {s} {s} {s} {s} {d}/{d} avg {s}", .{ o.id, e.traders.items[o.trader].name, @tagName(o.kind), if (o.sell) "SELL" else "BUY", p, @tagName(o.status), o.filled, o.quantity, avg });
223 if (o.kind == .STOP or o.kind == .STOPLIMIT) try e.out.print(" trigger {d}", .{o.trigger});
224 try e.out.writeByte('\n');
225 }

227 fn position(e: *Engine, i: usize) !void {
228 const t = e.traders.items[i];
229 try e.line("POSITION {s} bought {d} sold {d} net {d} cash {d} fees {d} pnl {d}", .{ t.name, t.bought, t.sold, e.net(i), t.cash, t.fees, e.pnl(i) });
230 }

232 fn tradeLine(e: *Engine, i: usize) !void {
233 const x = e.trades.items[i];
234 const b = e.orders.items[x.buy];
235 const s = e.orders.items[x.sell];
236 try e.line("TRADE {d} {d} {d} {s} {s} {s} {s}", .{ i + 1, x.price, x.quantity, b.id, e.traders.items[b.trader].name, s.id, e.traders.items[s.trader].name });
237 }

239 fn executeOrder(e: *Engine, i: usize) !void {
240 while (e.orders.items[i].remaining > 0) {
241 const incoming = e.orders.items[i];
242 const j = e.best(!incoming.sell) orelse break;
243 const resting_order = e.orders.items[j];
244 if (!matchable(incoming.sell, incoming.price, resting_order.price)) break;
245 if (incoming.trader == resting_order.trader) {
246 try e.line("STP {s} {d}", .{ resting_order.id, resting_order.remaining });
247 try e.history(j, "STP {d}", .{resting_order.remaining});
248 e.orders.items[j].remaining = 0;
249 e.orders.items[j].visible = 0;
250 e.orders.items[j].status = .CANCELLED;
251 continue;
252 }
253 const q = @min(incoming.remaining, resting_order.visible);
254 const p = resting_order.price;
255 const bi = if (incoming.sell) j else i;
256 const si = if (incoming.sell) i else j;
257 const bf = q * (if (bi == i) e.taker else e.maker);
258 const sf = q * (if (si == i) e.taker else e.maker);
259 try e.trades.append(e.gpa, .{ .price = p, .quantity = q, .buy = bi, .sell = si, .taker = i, .buy_fee = bf, .sell_fee = sf });
260 for ([_]usize{ i, j }) |affected| {
261 e.orders.items[affected].remaining -= q;
262 e.orders.items[affected].filled += q;
263 e.orders.items[affected].value += q * p;
264 try e.history(affected, "FILL {d} {d} {d}", .{ e.trades.items.len, p, q });
265 }
266 e.orders.items[j].visible -= q;
267 const buyer = e.orders.items[bi].trader;
268 const seller = e.orders.items[si].trader;
269 e.traders.items[buyer].bought += q;
270 e.traders.items[buyer].cash -= q * p + bf;
271 e.traders.items[buyer].fees += bf;
272 e.traders.items[seller].sold += q;
273 e.traders.items[seller].cash += q * p - sf;
274 e.traders.items[seller].fees += sf;
275 try e.tradeLine(e.trades.items.len - 1);
276 if (e.orders.items[j].remaining == 0) e.orders.items[j].status = .FILLED else {
277 e.orders.items[j].status = .PARTIAL;
278 if (e.orders.items[j].visible == 0) {
279 e.priority += 1;
280 e.orders.items[j].priority = e.priority;
281 const visible = @min(resting_order.peak, e.orders.items[j].remaining);
282 const hidden = e.orders.items[j].remaining - visible;
283 e.orders.items[j].visible = visible;
284 try e.history(j, "REFRESH {d} {d}", .{ visible, hidden });
285 try e.line("REFRESH {s} {d} {d}", .{ resting_order.id, visible, hidden });
286 }
287 }
288 }
289 }

291 fn finish(e: *Engine, i: usize, stay: bool) !void {
292 if (e.orders.items[i].remaining == 0) e.orders.items[i].status = .FILLED else if (stay) {
293 e.priority += 1;
294 e.orders.items[i].priority = e.priority;
295 const o = e.orders.items[i];
296 const visible = if (o.kind == .ICEBERG) @min(o.peak, o.remaining) else o.remaining;
297 e.orders.items[i].visible = visible;
298 e.orders.items[i].status = if (o.filled > 0) .PARTIAL else .OPEN;
299 try e.history(i, "REST {d} {d}", .{ o.price, visible });
300 } else {
301 try e.history(i, "EXPIRE {d}", .{e.orders.items[i].remaining});
302 e.orders.items[i].remaining = 0;
303 e.orders.items[i].visible = 0;
304 e.orders.items[i].status = .CANCELLED;
305 }
306 try e.orderLine(i);
307 }

309 fn triggers(e: *Engine) !void {
310 while (e.trades.items.len > 0) {
311 var selected: [1000]usize = undefined;
312 var n: usize = 0;
313 const p = e.last();
314 for (e.orders.items, 0..) |o, i| if (o.status == .PENDING and (if (o.sell) p <= o.trigger else p >= o.trigger)) {
315 selected[n] = i;
316 n += 1;
317 };
318 if (n == 0) break;
319 for (selected[0..n]) |i| e.orders.items[i].status = .CANCELLED;
320 for (selected[0..n]) |i| {
321 try e.line("TRIGGER {s}", .{e.orders.items[i].id});
322 try e.history(i, "TRIGGER", .{});
323 try e.executeOrder(i);
324 try e.finish(i, e.orders.items[i].kind == .STOPLIMIT);
325 }
326 }
327 }

329 fn cancel(e: *Engine, i: usize) !void {
330 try e.history(i, "CANCEL {d}", .{e.orders.items[i].remaining});
331 e.orders.items[i].remaining = 0;
332 e.orders.items[i].visible = 0;
333 e.orders.items[i].status = .CANCELLED;
334 try e.orderLine(i);
335 }

337 fn levels(e: *Engine, sell: bool) ![]Level {
338 var result: std.ArrayList(Level) = .empty;
339 for (e.orders.items) |o| if (resting(o) and o.sell == sell) {
340 var index: ?usize = null;
341 for (result.items, 0..) |v, j| if (v.price == o.price) {
342 index = j;
343 break;
344 };
345 if (index == null) {
346 index = result.items.len;
347 try result.append(e.gpa, .{ .price = o.price });
348 }
349 result.items[index.?].visible += o.visible;
350 result.items[index.?].count += 1;
351 };
352 for (0..result.items.len) |i| {
353 const value = result.items[i];
354 var j = i;
355 while (j > 0 and result.items[j - 1].price > value.price) : (j -= 1) result.items[j] = result.items[j - 1];
356 result.items[j] = value;
357 }
358 return result.items;
359 }

369 fn newOrder(e: *Engine, w: []const Text, kind: Kind) !void {
370 const sell = eq(w[3], "SELL");
371 const t = e.findTrader(w[2]);
372 var p: I = -1;
373 var r: I = 0;
374 var q: I = 0;
375 var peak: I = 0;
376 var condition: Condition = .none;
377 switch (kind) {
378 .LIMIT, .ICEBERG => {
379 p = number(w[4]);
380 q = number(w[5]);
381 if (kind == .ICEBERG) peak = number(w[6]) else if (w.len == 7) condition = std.meta.stringToEnum(Condition, w[6]).?;
382 },
383 .MARKET => q = number(w[4]),
384 .STOP => {
385 r = number(w[4]);
386 q = number(w[5]);
387 },
388 .STOPLIMIT => {
389 r = number(w[4]);
390 p = number(w[5]);
391 q = number(w[6]);
392 },
393 }
394 const stop = kind == .STOP or kind == .STOPLIMIT;
395 var reason: ?Text = null;
396 if (e.halted) reason = "halted" else if (e.findOrder(w[1]) != null) reason = "duplicate id" else if (p == 0 or (stop and r == 0)) reason = "invalid price" else if (q == 0) reason = "invalid quantity" else if ((p >= 0 and @mod(p, e.tick) != 0) or (stop and @mod(r, e.tick) != 0)) reason = "invalid tick" else if (@mod(q, e.lot) != 0) reason = "invalid lot" else if (kind == .ICEBERG and (peak == 0 or peak > q or @mod(peak, e.lot) != 0)) reason = "invalid peak" else if (stop and e.trades.items.len > 0 and (if (sell) r >= e.last() else r <= e.last())) reason = "invalid trigger" else if (p >= 0 and e.outside(p)) reason = "outside band" else if (e.risk(t, sell, q, null)) reason = "risk limit";
397 if (reason == null and condition == .POSTONLY) {
398 if (e.best(!sell)) |j| if (matchable(sell, p, e.orders.items[j].price)) {
399 reason = "would cross";
400 };
401 }
402 if (reason == null and condition == .FOK) {
403 var available: I = 0;
404 for (e.orders.items) |o| if (resting(o) and o.sell != sell and (t == null or o.trader != t.?) and matchable(sell, p, o.price)) {
405 available += o.remaining;
406 };
407 if (available < q) reason = "unfillable";
408 }
409 if (reason) |text| return e.reject(w[1], text);
410 const trader = try e.register(w[2]);
411 const i = e.orders.items.len;
412 try e.orders.append(e.gpa, .{ .id = w[1], .trader = trader, .kind = kind, .sell = sell, .price = p, .trigger = r, .peak = peak, .quantity = q, .remaining = q, .condition = condition });
413 try e.history(i, "ACCEPT", .{});
414 if (stop) try e.orderLine(i) else {
415 if (condition != .POSTONLY) try e.executeOrder(i);
416 try e.finish(i, kind == .ICEBERG or (kind == .LIMIT and condition != .IOC));
417 try e.triggers();
418 }
419 }

421 fn command(e: *Engine, w: []const Text) !void {
422 const verb = std.meta.stringToEnum(Verb, w[0]).?;
423 switch (verb) {
424 .LIMIT, .MARKET, .STOP, .STOPLIMIT, .ICEBERG => return e.newOrder(w, std.meta.stringToEnum(Kind, w[0]).?),
425 .CANCEL, .REPLACE => {
426 const replace = verb == .REPLACE;
427 if (replace and e.halted) return e.reject(w[1], "halted");
428 const i = e.findOrder(w[1]) orelse return e.reject(w[1], "unknown order");
429 const o = e.orders.items[i];
430 if (if (replace) !resting(o) else !active(o)) return e.reject(w[1], "not open");
431 if (!replace) return e.cancel(i);
432 const p = number(w[2]);
433 const q = number(w[3]);
434 const reason: ?Text = if (p == 0) "invalid price" else if (q == 0) "invalid quantity" else if (p != o.price and @mod(p, e.tick) != 0) "invalid tick" else if (@mod(q, e.lot) != 0) "invalid lot" else if (p == o.price and q == o.remaining) "no change" else if (p != o.price and e.outside(p)) "outside band" else if (q > o.remaining and e.risk(o.trader, o.sell, q, i)) "risk limit" else null;
435 if (reason) |text| return e.reject(w[1], text);
436 const kept = p == o.price and q < o.remaining;
437 const mode: Text = if (kept) "kept" else "lost";
438 e.orders.items[i].quantity = o.filled + q;
439 e.orders.items[i].remaining = q;
440 try e.history(i, "REPLACE {d} {d} {s}", .{ p, q, mode });
441 try e.line("REPLACE {s} {s}", .{ o.id, mode });
442 if (kept) {
443 e.orders.items[i].visible = @min(o.visible, q);
444 try e.orderLine(i);
445 } else {
446 e.orders.items[i].status = .CANCELLED;
447 e.orders.items[i].price = p;
448 try e.executeOrder(i);
449 try e.finish(i, true);
450 try e.triggers();
451 }
452 },
453 .HALT, .RESUME => {
454 const h = verb == .HALT;
455 if (e.halted == h) return e.fail(if (h) "already halted" else "not halted");
456 e.halted = h;
457 try e.line("{s}", .{w[0]});
458 },
459 .RISK => {
460 const t = try e.register(w[1]);
461 const limit = number(w[2]);
462 e.traders.items[t].limit = limit;
463 try e.line("RISK {s} {d} exposure buy {d} sell {d}", .{ w[1], limit, e.exposure(t, false, null), e.exposure(t, true, null) });
464 },
465 .FEE => {
466 const a = number(w[1]);
467 const b = number(w[2]);
468 if (a + b < 0) return e.fail("invalid fee");
469 e.maker = a;
470 e.taker = b;
471 try e.line("FEE maker {d} taker {d}", .{ a, b });
472 },
473 .BAND, .TICK, .LOT => {
474 const v = number(w[1]);
475 if (verb != .BAND and v == 0) return e.fail("invalid size");
476 switch (verb) {
477 .BAND => e.band = v,
478 .TICK => e.tick = v,
479 else => e.lot = v,
480 }
481 try e.line("{s} {d}", .{ w[0], v });
482 },
483 .CANCELALL, .ORDERS, .POSITION, .FILLS => {
484 const t = e.findTrader(w[1]) orelse return e.fail("unknown trader");
485 if (verb == .POSITION) return e.position(t);
486 var count: usize = 0;
487 if (verb == .FILLS) {
488 for (e.trades.items) |x| if (e.orders.items[x.buy].trader == t or e.orders.items[x.sell].trader == t) {
489 count += 1;
490 };
491 try e.line("FILLS {s} {d}", .{ w[1], count });
492 var cumulative: I = 0;
493 for (e.trades.items, 0..) |x, j| {
494 const buy = e.orders.items[x.buy].trader == t;
495 const own = if (buy) x.buy else x.sell;
496 if (e.orders.items[own].trader != t) continue;
497 cumulative += if (buy) x.quantity else -x.quantity;
498 try e.line("{d} {s} {d} {d} {s} {s} fee {d} net {d}", .{ j + 1, if (buy) "BUY" else "SELL", x.price, x.quantity, e.orders.items[own].id, if (own == x.taker) "TAKER" else "MAKER", if (buy) x.buy_fee else x.sell_fee, cumulative });
499 }
500 } else {
501 for (e.orders.items) |o| if (active(o) and o.trader == t) {
502 count += 1;
503 };
504 try e.line("{s} {s} {d}", .{ w[0], w[1], count });
505 for (0..e.orders.items.len) |i| if (active(e.orders.items[i]) and e.orders.items[i].trader == t) {
506 if (verb == .CANCELALL) try e.cancel(i) else try e.details(i);
507 };
508 }
509 },
510 .STATUS, .HISTORY => {
511 const i = e.findOrder(w[1]) orelse return e.fail("unknown order");
512 if (verb == .STATUS) {
513 try e.out.writeAll("STATUS ");
514 try e.details(i);
515 } else {
516 try e.line("HISTORY {s} {d}", .{ w[1], e.orders.items[i].history.items.len });
517 for (e.orders.items[i].history.items) |text| try e.line("{s}", .{text});
518 }
519 },
520 .BOOK => {
521 const depth: usize = @intCast(number(w[1]));
522 if (depth == 0) return e.fail("invalid depth");
523 const bids = try e.levels(false);
524 const asks = try e.levels(true);
525 try e.line("BOOK bids {d} asks {d}", .{ bids.len, asks.len });
526 const n = @min(depth, asks.len);
527 for (0..n) |j| {
528 const v = asks[n - j - 1];
529 try e.line("ASK {d} {d} {d}", .{ v.price, v.visible, v.count });
530 }
531 for (0..@min(depth, bids.len)) |j| {
532 const v = bids[bids.len - j - 1];
533 try e.line("BID {d} {d} {d}", .{ v.price, v.visible, v.count });
534 }
535 },
536 .IMPACT => {
537 const sell = eq(w[1], "SELL");
538 const q = number(w[2]);
539 if (q == 0) return e.fail("invalid quantity");
540 const book = try e.levels(!sell);
541 var remainder = q;
542 var cost: I = 0;
543 var total: I = 0;
544 var worst: I = 0;
545 var used: usize = 0;
546 for (0..book.len) |j| {
547 const v = book[if (sell) book.len - j - 1 else j];
548 total += v.visible;
549 if (remainder > 0) {
550 const taken = @min(remainder, v.visible);
551 remainder -= taken;
552 cost += taken * v.price;
553 worst = v.price;
554 used += 1;
555 }
556 }
557 if (remainder > 0) try e.line("IMPACT {s} {d} insufficient {d}", .{ w[1], q, total }) else try e.line("IMPACT {s} {d} avg {s} worst {d} levels {d}", .{ w[1], q, try e.decimal(@divTrunc(cost * 100, q)), worst, used });
558 },
559 .LAST => {
560 if (e.trades.items.len == 0) try e.line("LAST none", .{}) else try e.line("LAST {d} {d}", .{ e.last(), e.trades.items[e.trades.items.len - 1].quantity });
561 },
562 .SPREAD => {
563 const bi = e.best(false);
564 const si = e.best(true);
565 const b = if (bi) |i| try e.format("{d}", .{e.orders.items[i].price}) else "none";
566 const a = if (si) |i| try e.format("{d}", .{e.orders.items[i].price}) else "none";
567 try e.out.print("SPREAD bid {s} ask {s}", .{ b, a });
568 if (bi != null and si != null) {
569 const bp = e.orders.items[bi.?].price;
570 const ap = e.orders.items[si.?].price;
571 try e.out.print(" spread {d} mid {d}.{d}", .{ ap - bp, @divTrunc(ap + bp, 2), if (@mod(ap + bp, 2) == 0) @as(I, 0) else @as(I, 5) });
572 }
573 try e.out.writeByte('\n');
574 },
575 .TRADES, .BARS => {
576 const k: usize = @intCast(number(w[1]));
577 if (k == 0) return e.fail("invalid count");
578 const n = e.trades.items.len;
579 if (verb == .TRADES) {
580 const shown = @min(k, n);
581 try e.line("TRADES {d} of {d}", .{ shown, n });
582 for (n - shown..n) |i| try e.tradeLine(i);
583 } else {
584 try e.line("BARS {d}", .{(n + k - 1) / k});
585 var start: usize = 0;
586 var group: usize = 1;
587 while (start < n) : (group += 1) {
588 const end = @min(start + k, n);
589 var high: I = 0;
590 var low = e.trades.items[start].price;
591 var volume: I = 0;
592 for (e.trades.items[start..end]) |x| {
593 high = @max(high, x.price);
594 low = @min(low, x.price);
595 volume += x.quantity;
596 }
597 try e.line("{d} open {d} high {d} low {d} close {d} volume {d}", .{ group, e.trades.items[start].price, high, low, e.trades.items[end - 1].price, volume });
598 start = end;
599 }
600 }
601 },
602 .FLOW => {
603 var buy: I = 0;
604 var sell: I = 0;
605 for (e.trades.items) |x| {
606 if (e.orders.items[x.taker].sell) sell += x.quantity else buy += x.quantity;
607 }
608 try e.line("FLOW buy {d} sell {d} net {d}", .{ buy, sell, buy - sell });
609 },
610 .VWAP, .SUMMARY => {
611 const n = e.trades.items.len;
612 var volume: I = 0;
613 var cost: I = 0;
614 var fees: I = 0;
615 var high: I = 0;
616 var low: I = if (n == 0) 0 else e.trades.items[0].price;
617 for (e.trades.items) |x| {
618 volume += x.quantity;
619 cost += x.price * x.quantity;
620 fees += x.buy_fee + x.sell_fee;
621 high = @max(high, x.price);
622 low = @min(low, x.price);
623 }
624 if (verb == .VWAP) {
625 if (n == 0) try e.line("VWAP none", .{}) else try e.line("VWAP {s} volume {d} trades {d}", .{ try e.decimal(@divTrunc(cost * 100, volume)), volume, n });
626 } else {
627 if (n == 0) try e.line("SUMMARY trades 0", .{}) else try e.line("SUMMARY trades {d} volume {d} open {d} high {d} low {d} close {d} fees {d}", .{ n, volume, e.trades.items[0].price, high, low, e.last(), fees });
628 try e.traderListing(false);
629 }
630 },
631 .RANK => try e.traderListing(true),
632 }
633 }

635 fn before(e: *Engine, a: usize, b: usize, ranked: bool) bool {
636 if (ranked) {
637 if (e.pnl(a) != e.pnl(b)) return e.pnl(a) > e.pnl(b);
638 const av = e.traders.items[a].bought + e.traders.items[a].sold;
639 const bv = e.traders.items[b].bought + e.traders.items[b].sold;
640 if (av != bv) return av > bv;
641 }
642 return std.mem.order(u8, e.traders.items[a].name, e.traders.items[b].name) == .lt;
643 }
644
645 fn traderListing(e: *Engine, ranked: bool) !void {
646 var ids: [2000]usize = undefined;
647 const n = e.traders.items.len;
648 for (0..n) |i| {
649 var j = i;
650 while (j > 0 and e.before(i, ids[j - 1], ranked)) : (j -= 1) ids[j] = ids[j - 1];
651 ids[j] = i;
652 }
653 if (ranked) try e.line("RANK {d}", .{n});
654 for (ids[0..n], 0..) |i, rank| {
655 if (ranked) try e.line("{d} {s} pnl {d} volume {d}", .{ rank + 1, e.traders.items[i].name, e.pnl(i), e.traders.items[i].bought + e.traders.items[i].sold }) else try e.position(i);
656 }
657 }

660fn readLine(reader: *std.Io.Reader, gpa: A) !?Text {
661 var buffer: std.Io.Writer.Allocating = .init(gpa);
662 defer buffer.deinit();
663 _ = try reader.streamDelimiterEnding(&buffer.writer, '\n');
664 if (reader.bufferedLen() == 0 and buffer.written().len == 0) return null;
665 if (reader.bufferedLen() > 0) reader.toss(1);
666 return try buffer.toOwnedSlice();
667}

669pub fn main(init: std.process.Init) !void {
670 var arena: std.heap.ArenaAllocator = .init(std.heap.page_allocator);
671 defer arena.deinit();
672 const gpa = arena.allocator();
673 var ib: [4096]u8 = undefined;
674 var ob: [4096]u8 = undefined;
675 var input = std.Io.File.stdin().readerStreaming(init.io, &ib);
676 var output = std.Io.File.stdout().writerStreaming(init.io, &ob);
677 var engine: Engine = .{ .gpa = gpa, .out = &output.interface };
678 const first = (try readLine(&input.interface, gpa)).?;
679 const n: usize = @intCast(number(std.mem.trim(u8, first, "\r")));
680 for (0..n) |_| {
681 const text = std.mem.trim(u8, (try readLine(&input.interface, gpa)).?, "\r");
682 var words: [10]Text = undefined;
683 var count: usize = 0;
684 var it = std.mem.splitScalar(u8, text, ' ');
685 while (it.next()) |word| {
686 words[count] = word;
687 count += 1;
688 }
689 try engine.command(words[0..count]);
690 }
691 try output.interface.flush();
692}
