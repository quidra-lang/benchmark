1import kotlin.math.abs
2import kotlin.math.min
3private class Rejection(message: String) : RuntimeException(message)
4private fun reject(bad: Boolean, reason: String) {
5 if (bad) throw Rejection(reason)
6}
7private fun decimal(n: Long) = "${n / 100}.${(n % 100).toString().padStart(2, '0')}"
8private class Order(val id: String, val trader: String, val type: String, val side: String, var quantity: Long) {
9 var condition = ""
10 var status = "OPEN"
11 var price = 0L
12 var trigger = 0L
13 var peak = 0L
14 var left = quantity
15 var visible = 0L
16 var filled = 0L
17 var value = 0L
18 var priority = 0L
19 val history = mutableListOf("ACCEPT")
20 fun resting() = status == "OPEN" || status == "PARTIAL"
21 fun active() = resting() || status == "PENDING"
22 fun matches(p: Long) = price == 0L || if (side == "BUY") p <= price else p >= price
23 fun triggered(p: Long) = if (side == "BUY") p >= trigger else p <= trigger
24 fun line() = "ORDER $id $status $filled/$quantity"
25 fun details(): String = "$id $trader $type $side ${if (price == 0L) "-" else price} $status $filled/$quantity avg ${if (filled == 0L) "-" else decimal(value * 100 / filled)}" + if (trigger == 0L) "" else " trigger $trigger"
26 fun fill(number: Int, p: Long, q: Long) {
27 left -= q
28 filled += q
29 value += p * q
30 history.add("FILL $number $p $q")
31 }
32}
33private class Trader(val name: String) {
34 var bought = 0L
35 var sold = 0L
36 var cash = 0L
37 var fees = 0L
38 var limit: Long? = null
39 fun net() = bought - sold
40 fun volume() = bought + sold
41 fun pnl(mark: Long) = cash + net() * mark
42 fun position(mark: Long) = "POSITION $name bought $bought sold $sold net ${net()} cash $cash fees $fees pnl ${pnl(mark)}"
43}
44private data class Trade(val number: Int, val price: Long, val quantity: Long, val buy: Order, val sell: Order, val taker: Order, val buyFee: Long, val sellFee: Long) {
45 fun line() = "TRADE $number $price $quantity ${buy.id} ${buy.trader} ${sell.id} ${sell.trader}"
46}
47private class Level(val price: Long, var quantity: Long = 0, var count: Int = 0)
48private class Engine {
49 val orders = linkedMapOf<String, Order>()
50 val traders = sortedMapOf<String, Trader>()
51 val pending = mutableListOf<Order>()
52 val buys = mutableListOf<Order>()
53 val sells = mutableListOf<Order>()
54 val trades = mutableListOf<Trade>()
55 val output = StringBuilder()
56 var makerFee = 0L
57 var takerFee = 0L
58 var band = 0L
59 var tick = 1L
60 var lot = 1L
61 var priority = 0L
62 var halted = false
63 fun emit(vararg parts: Any) {
64 output.append(parts.joinToString(" ")).append('\n')
65 }
66 fun book(side: String) = if (side == "BUY") buys else sells
67 fun opposite(side: String) = if (side == "BUY") sells else buys
68 fun comparator(side: String): Comparator<Order> = compareBy<Order> {
69 if (side == "BUY") -it.price else it.price
70 }.thenBy {
71 it.priority
72 }
73 fun best(book: List<Order>) = book.minWithOrNull(comparator(if (book === buys) "BUY" else "SELL"))
74 fun last() = trades.lastOrNull()?.price ?: 0L
75 fun register(name: String) = traders.getOrPut(name) {
76 Trader(name)
77 }
78 fun trader(name: String): Trader {
79 reject(name !in traders, "unknown trader")
80 return traders.getValue(name)
81 }
82 fun order(id: String): Order {
83 reject(id !in orders, "unknown order")
84 return orders.getValue(id)
85 }
86 fun active(name: String) = orders.values.filter {
87 it.trader == name && it.active()
88 }
89 fun exposure(name: String, side: String, excluded: Order? = null): Long = (if (side == "BUY") traders.getValue(name).net() else -traders.getValue(name).net()) + active(name).filter {
90 it !== excluded && it.side == side
91 }.sumOf {
92 it.left
93 }
94 fun risk(name: String, side: String, q: Long, excluded: Order? = null) {
95 val limit = traders[name]?.limit ?: return
96 reject(exposure(name, side, excluded) + q > limit, "risk limit")
97 }
98 fun band(p: Long) {
99 reject(band > 0 && last() != 0L && abs(p - last()) > band, "outside band")
100 }
101 fun match(x: Order) {
102 val b = opposite(x.side)
103 while (x.left > 0) {
104 val y = best(b) ?: break
105 if (!x.matches(y.price)) break
106 if (x.trader == y.trader) {
107 emit("STP", y.id, y.left)
108 y.history.add("STP ${y.left}")
109 b.remove(y)
110 y.left = 0
111 y.status = "CANCELLED"
112 continue
113 }
114 val q = min(x.left, y.visible)
115 val p = y.price
116 val buy = if (x.side == "BUY") x else y
117 val sell = if (buy === x) y else x
118 val xf = q * takerFee
119 val yf = q * makerFee
120 val t = Trade(trades.size + 1, p, q, buy, sell, x, if (buy === x) xf else yf, if (sell === x) xf else yf)
121 trades.add(t)
122 x.fill(t.number, p, q)
123 y.fill(t.number, p, q)
124 y.visible -= q
125 val bt = traders.getValue(buy.trader)
126 val st = traders.getValue(sell.trader)
127 bt.bought += q
128 bt.cash -= p * q + t.buyFee
129 bt.fees += t.buyFee
130 st.sold += q
131 st.cash += p * q - t.sellFee
132 st.fees += t.sellFee
133 emit(t.line())
134 if (y.left == 0L) {
135 b.remove(y)
136 y.status = "FILLED"
137 }
138 else {
139 y.status = "PARTIAL"
140 if (y.visible == 0L) {
141 y.priority = ++priority
142 y.visible = min(y.peak, y.left)
143 y.history.add("REFRESH ${y.visible} ${y.left - y.visible}")
144 emit("REFRESH", y.id, y.visible, y.left - y.visible)
145 }
146 }
147 }
148 }
149 fun post(o: Order, forceRest: Boolean = false) {
150 if (o.left == 0L) o.status = "FILLED"
151 else if (forceRest || o.type in listOf("ICEBERG", "STOPLIMIT") || o.type == "LIMIT" && o.condition != "IOC") {
152 o.priority = ++priority
153 o.visible = if (o.peak == 0L) o.left else min(o.peak, o.left)
154 book(o.side).add(o)
155 o.status = if (o.filled > 0) "PARTIAL" else "OPEN"
156 o.history.add("REST ${o.price} ${o.visible}")
157 } else {
158 o.history.add("EXPIRE ${o.left}")
159 o.left = 0
160 o.status = "CANCELLED"
161 }
162 emit(o.line())
163 }
164 fun trigger() {
165 while (trades.isNotEmpty()) {
166 val p = last()
167 val batch = pending.filter {
168 it.triggered(p)
169 }
170 if (batch.isEmpty()) return
171 pending.removeAll(batch.toSet())
172 for (o in batch) {
173 emit("TRIGGER", o.id)
174 o.history.add("TRIGGER")
175 match(o)
176 post(o)
177 }
178 }
179 }
180 fun cancel(o: Order) {
181 if (o.status == "PENDING") pending.remove(o) else book(o.side).remove(o)
182 o.history.add("CANCEL ${o.left}")
183 o.left = 0
184 o.status = "CANCELLED"
185 emit(o.line())
186 }
187 fun newOrder(a: List<String>) {
188 val kind = a[0]
189 val market = kind == "MARKET"
190 val stop = kind == "STOP"
191 val sl = kind == "STOPLIMIT"
192 val ice = kind == "ICEBERG"
193 val p = if (market || stop) 0L else a[if (sl) 5 else 4].toLong()
194 val tr = if (stop || sl) a[4].toLong() else 0L
195 val q = a[if (market) 4 else if (sl) 6 else 5].toLong()
196 val o = Order(a[1], a[2], kind, a[3], q)
197 o.price = p
198 o.trigger = tr
199 if (ice) o.peak = a[6].toLong()
200 if (kind == "LIMIT" && a.size == 7) o.condition = a[6]
201 reject(halted, "halted")
202 reject(o.id in orders, "duplicate id")
203 reject(!market && !stop && p == 0L || (stop || sl) && tr == 0L, "invalid price")
204 reject(q == 0L, "invalid quantity")
205 reject(p % tick != 0L || tr % tick != 0L, "invalid tick")
206 reject(q % lot != 0L, "invalid lot")
207 if (ice) reject(o.peak == 0L || o.peak > q || o.peak % lot != 0L, "invalid peak")
208 if (stop || sl) reject(last() != 0L && o.triggered(last()), "invalid trigger")
209 if (p != 0L) band(p)
210 risk(o.trader, o.side, q)
211 val b = best(opposite(o.side))
212 if (o.condition == "POSTONLY") reject(b != null && o.matches(b.price), "would cross")
213 if (o.condition == "FOK") reject(opposite(o.side).filter {
214 it.trader != o.trader && o.matches(it.price)
215 }.sumOf {
216 it.left
217 }
218 < q, "unfillable")
219 register(o.trader)
220 orders[o.id] = o
221 if (stop || sl) {
222 o.status = "PENDING"
223 pending.add(o)
224 emit(o.line())
225 }
226 else {
227 if (o.condition != "POSTONLY") match(o)
228 post(o)
229 trigger()
230 }
231 }
232 fun levels(side: String): List<Level> {
233 val result = mutableListOf<Level>()
234 for (o in book(side).sortedWith(comparator(side))) {
235 if (result.lastOrNull()?.price != o.price) result.add(Level(o.price))
236 result.last().quantity += o.visible
237 result.last().count++
238 }
239 return result
240 }
241 fun positive(text: String, reason: String): Long {
242 val n = text.toLong()
243 reject(n == 0L, reason)
244 return n
245 }
246 fun bar(ts: List<Trade>) = "open ${ts.first().price} high ${ts.maxOf { it.price }} low ${ts.minOf { it.price }} close ${ts.last().price} volume ${ts.sumOf { it.quantity }}"
247 fun execute(line: String) {
248 val a = line.split(' ')
249 val c = a[0]
250 try {
251 when (c) {
252 "LIMIT", "MARKET", "STOP", "STOPLIMIT", "ICEBERG" -> newOrder(a)
253 "CANCEL" -> {
254 val o = order(a[1])
255 reject(!o.active(), "not open")
256 cancel(o)
257 }
258 "REPLACE" -> {
259 reject(halted, "halted")
260 val o = order(a[1])
261 reject(!o.resting(), "not open")
262 val p = a[2].toLong()
263 val q = a[3].toLong()
264 val moved = p != o.price
265 reject(p == 0L, "invalid price")
266 reject(q == 0L, "invalid quantity")
267 reject(moved && p % tick != 0L, "invalid tick")
268 reject(q % lot != 0L, "invalid lot")
269 reject(!moved && q == o.left, "no change")
270 if (moved) band(p)
271 if (q > o.left) risk(o.trader, o.side, q, o)
272 o.quantity = o.filled + q
273 val kept = !moved && q < o.left
274 o.history.add("REPLACE $p $q ${if (kept) "kept" else "lost"}")
275 emit("REPLACE", o.id, if (kept) "kept" else "lost")
276 if (kept) {
277 o.left = q
278 o.visible = min(o.visible, q)
279 emit(o.line())
280 }
281 else {
282 book(o.side).remove(o)
283 o.price = p
284 o.left = q
285 match(o)
286 post(o, true)
287 trigger()
288 }
289 }
290 "CANCELALL" -> {
291 trader(a[1])
292 val xs = active(a[1])
293 emit(c, a[1], xs.size)
294 xs.forEach {
295 cancel(it)
296 }
297 }
298 "HALT" -> {
299 reject(halted, "already halted")
300 halted = true
301 emit(c)
302 }
303 "RESUME" -> {
304 reject(!halted, "not halted")
305 halted = false
306 emit(c)
307 }
308 "RISK" -> {
309 register(a[1]).limit = a[2].toLong()
310 emit(c, a[1], a[2], "exposure buy", exposure(a[1], "BUY"), "sell", exposure(a[1], "SELL"))
311 }
312 "FEE" -> {
313 val m = a[1].toLong()
314 val t = a[2].toLong()
315 reject(m + t < 0, "invalid fee")
316 makerFee = m
317 takerFee = t
318 emit("FEE maker", m, "taker", t)
319 }
320 "BAND" -> {
321 band = a[1].toLong()
322 emit(c, band)
323 }
324 "TICK", "LOT" -> {
325 val v = positive(a[1], "invalid size")
326 if (c == "TICK") tick = v else lot = v
327 emit(c, v)
328 }
329 "BOOK" -> {
330 val n = positive(a[1], "invalid depth").toInt()
331 val bids = levels("BUY")
332 val asks = levels("SELL")
333 emit("BOOK bids", bids.size, "asks", asks.size)
334 for (l in asks.take(n).reversed()) emit("ASK", l.price, l.quantity, l.count)
335 for (l in bids.take(n)) emit("BID", l.price, l.quantity, l.count)
336 }
337 "ORDERS" -> {
338 trader(a[1])
339 val xs = active(a[1])
340 emit(c, a[1], xs.size)
341 xs.forEach {
342 emit(it.details())
343 }
344 }
345 "STATUS" -> emit(c, order(a[1]).details())
346 "HISTORY" -> {
347 val o = order(a[1])
348 emit(c, o.id, o.history.size)
349 o.history.forEach {
350 emit(it)
351 }
352 }
353 "POSITION" -> emit(trader(a[1]).position(last()))
354 "FILLS" -> {
355 trader(a[1])
356 val ts = trades.filter {
357 it.buy.trader == a[1] || it.sell.trader == a[1]
358 }
359 emit(c, a[1], ts.size)
360 var net = 0L
361 for (t in ts) {
362 val buy = t.buy.trader == a[1]
363 val o = if (buy) t.buy else t.sell
364 net += if (buy) t.quantity else -t.quantity
365 emit(t.number, o.side, t.price, t.quantity, o.id, if (t.taker === o) "TAKER" else "MAKER", "fee", if (buy) t.buyFee else t.sellFee, "net", net)
366 }
367 }
368 "LAST" -> if (trades.isEmpty()) emit(c, "none") else emit(c, last(), trades.last().quantity)
369 "VWAP" -> if (trades.isEmpty()) emit(c, "none") else {
370 val v = trades.sumOf {
371 it.quantity
372 }
373 emit(c, decimal(trades.sumOf {
374 it.price * it.quantity
375 } * 100 / v), "volume", v, "trades", trades.size)
376 }
377 "SPREAD" -> {
378 val b = best(buys)
379 val s = best(sells)
380 if (b == null || s == null) emit("SPREAD bid", b?.price ?: "none", "ask", s?.price ?: "none") else {
381 val sum = b.price + s.price
382 emit("SPREAD bid", b.price, "ask", s.price, "spread", s.price - b.price, "mid", "${sum / 2}.${if (sum % 2 == 0L) 0 else 5}")
383 }
384 }
385 "IMPACT" -> {
386 val q = positive(a[2], "invalid quantity")
387 var left = q
388 var cost = 0L
389 var worst = 0L
390 var used = 0
391 val ls = levels(if (a[1] == "BUY") "SELL" else "BUY")
392 for (l in ls) if (left > 0) {
393 val u = min(left, l.quantity)
394 left -= u
395 cost += u * l.price
396 worst = l.price
397 used++
398 }
399 if (left > 0) emit(c, a[1], q, "insufficient", ls.sumOf {
400 it.quantity
401 }) else emit(c, a[1], q, "avg", decimal(cost * 100 / q), "worst", worst, "levels", used)
402 }
403 "TRADES" -> {
404 val n = positive(a[1], "invalid count").toInt()
405 val ts = trades.takeLast(min(n, trades.size))
406 emit(c, ts.size, "of", trades.size)
407 ts.forEach {
408 emit(it.line())
409 }
410 }
411 "BARS" -> {
412 val k = positive(a[1], "invalid count").toInt()
413 val groups = trades.chunked(k)
414 emit(c, groups.size)
415 groups.forEachIndexed {
416 i, ts -> emit(i + 1, bar(ts))
417 }
418 }
419 "RANK" -> {
420 val ts = traders.values.sortedWith(compareByDescending<Trader> {
421 it.pnl(last())
422 }.thenByDescending {
423 it.volume()
424 }.thenBy {
425 it.name
426 })
427 emit(c, ts.size)
428 ts.forEachIndexed {
429 i, t -> emit(i + 1, t.name, "pnl", t.pnl(last()), "volume", t.volume())
430 }
431 }
432 "FLOW" -> {
433 val buy = trades.filter {
434 it.taker === it.buy
435 }.sumOf {
436 it.quantity
437 }
438 val sell = trades.filter {
439 it.taker === it.sell
440 }.sumOf {
441 it.quantity
442 }
443 emit("FLOW buy", buy, "sell", sell, "net", buy - sell)
444 }
445 "SUMMARY" -> {
446 if (trades.isEmpty()) emit("SUMMARY trades 0") else emit("SUMMARY trades", trades.size, "volume", trades.sumOf {
447 it.quantity
448 }, bar(trades).substringBeforeLast(" volume"), "fees", trades.sumOf {
449 it.buyFee + it.sellFee
450 })
451 traders.values.forEach {
452 emit(it.position(last()))
453 }
454 }
455 }
456 } catch (e: Rejection) {
457 if (c in listOf("LIMIT", "MARKET", "STOP", "STOPLIMIT", "ICEBERG", "CANCEL", "REPLACE")) emit("REJECT", a[1], e.message!!) else emit("ERROR", e.message!!)
458 }
459 }
460}

463 val engine = Engine()

465 engine.execute(reader.readLine())
466 }
