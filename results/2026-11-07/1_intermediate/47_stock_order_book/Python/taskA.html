31class Order:
32 def __init__(
33 self, order_id, trader, kind, side, quantity,
34 price=None, trigger=None, peak=None, condition=None,
35 ):
36 self.order_id = order_id
37 self.trader = trader
38 self.kind = kind
39 self.side = side
40 self.condition = condition
41 self.price = price
42 self.trigger = trigger
43 self.peak = peak
44 self.quantity = quantity
45 self.remaining = quantity
46 self.visible = 0
47 self.filled = 0
48 self.value = 0
49 self.priority = 0
50 self.number = 0
51 self.status = "OPEN"
52 self.history = ["ACCEPT"]
53
54 def is_active(self):
55 return self.status in ACTIVE_STATUSES
56
57 def is_resting(self):
58 return self.status in RESTING_STATUSES
59
60 def stays_on_book(self):
61 if self.kind == "LIMIT":
62 return self.condition != "IOC"
63 return self.kind in ("ICEBERG", "STOPLIMIT")
64
65 def matches(self, price):
66 if self.price is None:
67 return True
68 if self.side == "BUY":
69 return price <= self.price
70 return price >= self.price
71
72 def triggered_by(self, last_price):
73 if self.side == "BUY":
74 return last_price >= self.trigger
75 return last_price <= self.trigger
76
77 def fill(self, number, price, quantity):
78 self.remaining -= quantity
79 self.filled += quantity
80 self.value += quantity * price
81 self.history.append(f"FILL {number} {price} {quantity}")
82
83 def order_line(self):
84 return f"ORDER {self.order_id} {self.status} {self.filled}/{self.quantity}"
85
86 def details(self):
87 price = "-" if self.price is None else self.price
88 average = "-" if self.filled == 0 else two_decimals(self.value * 100 // self.filled)
89 text = (
90 f"{self.order_id} {self.trader} {self.kind} {self.side} {price} {self.status} "
91 f"{self.filled}/{self.quantity} avg {average}"
92 )
93 if self.trigger is not None:
94 text += f" trigger {self.trigger}"
95 return text

170 def sort_key(self, order):

174 def best(self):

183 def levels(self):
184 levels = []
185 for order in sorted(self.orders, key=self.sort_key):

201 self.books = {"BUY": Book("BUY"), "SELL": Book("SELL")}

240 def exposure(self, name, side, excluded=None):
241 net = self.traders[name].net()

252 def require_order(self, order_id):

262 def require_new(self, order_id, prices, quantity):
263 if self.halted:
264 raise CommandError("halted")
265 if order_id in self.orders:
266 raise CommandError("duplicate id")
267 if any(price == 0 for price in prices):
268 raise CommandError("invalid price")
269 if quantity == 0:
270 raise CommandError("invalid quantity")
271 if any(price % self.tick != 0 for price in prices):
272 raise CommandError("invalid tick")
273 if quantity % self.lot != 0:
274 raise CommandError("invalid lot")

276 def require_band(self, price):

280 def require_risk(self, name, side, quantity, excluded=None):
281 trader = self.traders.get(name)
282 if trader is None or trader.limit is None:
283 return
284 if self.exposure(name, side, excluded) + quantity > trader.limit:
285 raise CommandError("risk limit")

292 def accept(self, order):
293 self.register(order.trader)

297 def trade(self, taker, maker):
298 quantity = min(taker.remaining, maker.visible)
299 price = maker.price
300 taker_fee = quantity * self.taker_fee
301 maker_fee = quantity * self.maker_fee
302 if taker.side == "BUY":
303 buy, sell = taker, maker
304 buy_fee, sell_fee = taker_fee, maker_fee
305 else:
306 buy, sell = maker, taker
307 buy_fee, sell_fee = maker_fee, taker_fee
308 trade = Trade(len(self.trades) + 1, price, quantity, buy, sell, taker, buy_fee, sell_fee)
309 self.trades.append(trade)
310 taker.fill(trade.number, price, quantity)
311 maker.fill(trade.number, price, quantity)
312 maker.visible -= quantity
313 self.traders[buy.trader].buy(quantity, price)
314 self.traders[sell.trader].sell(quantity, price)
315 self.traders[taker.trader].pay(taker_fee)
316 self.traders[maker.trader].pay(maker_fee)
317 self.emit(trade.line())

319 def refresh(self, order):
320 order.priority = self.next_priority()
321 order.visible = min(order.peak, order.remaining)
322 hidden = order.remaining - order.visible
323 order.history.append(f"REFRESH {order.visible} {hidden}")
324 self.emit("REFRESH", order.order_id, order.visible, hidden)

326 def match_order(self, order):
327 book = self.books[OPPOSITE[order.side]]
328 while order.remaining > 0:
329 resting = book.best()
330 if resting is None or not order.matches(resting.price):
331 return
332 if resting.trader == order.trader:
333 self.emit("STP", resting.order_id, resting.remaining)
334 resting.history.append(f"STP {resting.remaining}")
335 book.remove(resting)
336 resting.remaining = 0
337 resting.status = "CANCELLED"
338 continue
339 self.trade(order, resting)
340 if resting.remaining == 0:
341 book.remove(resting)
342 resting.status = "FILLED"
343 else:
344 resting.status = "PARTIAL"
345 if resting.visible == 0:
346 self.refresh(resting)

348 def place(self, order):
349 order.priority = self.next_priority()
350 if order.peak is None:
351 order.visible = order.remaining
352 else:
353 order.visible = min(order.peak, order.remaining)
354 self.books[order.side].add(order)
355 order.status = "PARTIAL" if order.filled > 0 else "OPEN"
356 order.history.append(f"REST {order.price} {order.visible}")
357
358 def post_process(self, order):
359 if order.remaining == 0:
360 order.status = "FILLED"
361 elif order.stays_on_book():
362 self.place(order)
363 else:
364 order.history.append(f"EXPIRE {order.remaining}")
365 order.remaining = 0
366 order.status = "CANCELLED"
367 self.emit(order.order_line())

369 def trigger_stops(self):
370 while self.trades:
371 last = self.last_price()
372 selected = [order for order in self.pending if order.triggered_by(last)]
373 if not selected:
374 return
375 self.pending = [order for order in self.pending if not order.triggered_by(last)]
376 for order in selected:
377 self.emit("TRIGGER", order.order_id)
378 order.history.append("TRIGGER")
379 self.match_order(order)
380 self.post_process(order)

382 def submit(self, order):
383 self.accept(order)
384 if order.condition != "POSTONLY":
385 self.match_order(order)
386 self.post_process(order)
387 self.trigger_stops()

389 def queue_stop(self, order):
390 self.accept(order)

392 self.pending.append(order)
393 self.emit(order.order_line())

395 def cancel_order(self, order):

401 order.remaining = 0

405 def limit(self, arguments):
406 order_id, trader, side = arguments[:3]
407 price, quantity = int(arguments[3]), int(arguments[4])
408 condition = arguments[5] if len(arguments) == 6 else None
409 order = Order(order_id, trader, "LIMIT", side, quantity, price=price, condition=condition)
410 self.require_new(order_id, (price,), quantity)
411 self.require_band(price)
412 self.require_risk(trader, side, quantity)
413 opposite = self.books[OPPOSITE[side]]
414 best = opposite.best()
415 if condition == "POSTONLY" and best is not None and order.matches(best.price):
416 raise CommandError("would cross")
417 if condition == "FOK" and opposite.available(order) < quantity:
418 raise CommandError("unfillable")
419 self.submit(order)

467 def replace(self, arguments):
468 order_id = arguments[0]
469 price, quantity = int(arguments[1]), int(arguments[2])
470 if self.halted:
471 raise CommandError("halted")
472 order = self.require_order(order_id)
473 if not order.is_resting():
474 raise CommandError("not open")
475 if price == 0:
476 raise CommandError("invalid price")
477 if quantity == 0:
478 raise CommandError("invalid quantity")
479 moved = price != order.price
480 if moved and price % self.tick != 0:
481 raise CommandError("invalid tick")
482 if quantity % self.lot != 0:
483 raise CommandError("invalid lot")
484 if not moved and quantity == order.remaining:
485 raise CommandError("no change")
486 if moved:
487 self.require_band(price)
488 if quantity > order.remaining:
489 self.require_risk(order.trader, order.side, quantity, order)
490 order.quantity = order.filled + quantity
491 if not moved and quantity < order.remaining:
492 order.remaining = quantity
493 order.visible = min(order.visible, quantity)
494 order.history.append(f"REPLACE {price} {quantity} kept")
495 self.emit("REPLACE", order_id, "kept")
496 self.emit(order.order_line())
497 return
498 self.books[order.side].remove(order)
499 order.price = price
500 order.remaining = quantity
501 order.history.append(f"REPLACE {price} {quantity} lost")
502 self.emit("REPLACE", order_id, "lost")
503 self.match_order(order)
504 self.post_process(order)
505 self.trigger_stops()

642 def impact(self, arguments):
643 side = arguments[0]
644 quantity = positive(arguments[1], "invalid quantity")
645 levels = self.books[OPPOSITE[side]].levels()
646 remainder = quantity
647 cost = 0
648 worst = 0
649 used = 0
650 for level in levels:
651 if remainder == 0:
652 break
653 taken = min(remainder, level.quantity)
654 remainder -= taken
655 cost += taken * level.price
656 worst = level.price
657 used += 1
658 if remainder > 0:
659 total = sum(level.quantity for level in levels)
660 self.emit("IMPACT", side, quantity, "insufficient", total)
661 else:
662 self.emit(
663 "IMPACT", side, quantity, "avg", two_decimals(cost * 100 // quantity),
664 "worst", worst, "levels", used,
665 )

714 def execute(self, line):
