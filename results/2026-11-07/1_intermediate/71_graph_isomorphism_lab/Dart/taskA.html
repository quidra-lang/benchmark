50int lex(List<int> a, List<int> b) {
51 for (var i = 0; i < math.min(a.length, b.length); i++) {
52 if (a[i] != b[i]) return a[i].compareTo(b[i]);
53 }
54 return a.length.compareTo(b.length);

56int popcount(int mask) {
57 var count = 0;
58 while (mask != 0) {
59 mask &= mask - 1;
60 count++;
61 }
62 return count;

64List<List<int>> cells(List<int> colors) {
65 final result = List.generate(colors.reduce(math.max) + 1, (_) => <int>[]);
66 for (var v = 0; v < colors.length; v++) {
67 result[colors[v]].add(v);
68 }

71Refinement refine(Graph graph, List<int> initial) {
72 var colors = List<int>.of(initial), rounds = 0;
73 while (true) {
74 final signatures = <List<int>>[];
75 for (var v = 0; v < graph.n; v++) {
76 final neighbors = graph.neighbors(v).map((u) => colors[u]).toList()..sort();
77 signatures.add([colors[v], ...neighbors]);
78 }
79 final order = List.generate(graph.n, (v) => v)
80 ..sort((a, b) => lex(signatures[a], signatures[b]));
81 final next = List<int>.filled(graph.n, 0);
82 var color = 0;
83 for (var i = 0; i < order.length; i++) {
84 if (i > 0 && lex(signatures[order[i - 1]], signatures[order[i]]) != 0) color++;
85 next[order[i]] = color;
86 }
87 if (color + 1 == colors.reduce(math.max) + 1) return Refinement(colors, rounds);
88 colors = next;
89 rounds++;
90 }
91}
92Refinement stable(Graph graph) {
93 if (graph.stable == null) {

95 final unique = degrees.toSet().toList()..sort();
96 graph.stable = refine(graph, degrees.map(unique.indexOf).toList());
97 }
98 return graph.stable!;

117Canonical canon(Graph graph) {
118 if (graph.canonical != null) return graph.canonical!;
119 final result = Canonical();
120 void visit(List<int> colors) {
121 result.nodes++;
122 final partition = cells(colors);
123 if (partition.length == graph.n) {
124 result.leaves++;
125 final string = leafString(graph, colors);
126 if (result.bestLeaves.isEmpty || string.compareTo(result.string) > 0) {
127 result.string = string;
128 result.labels = List.of(colors);
129 result.bestLeaves = [List.of(colors)];
130 } else if (string == result.string) {
131 result.bestLeaves.add(List.of(colors));
132 }
133 return;
134 }
135 final candidates = partition.where((cell) => cell.length > 1);
136 final chosen = candidates.reduce((best, cell) => cell.length < best.length ? cell : best);
137 for (final vertex in chosen) {
138 visit(refine(graph, individualize(colors, vertex)).colors);
139 }
140 }
141 visit(stable(graph).colors);
142 graph.canonical = result;
143 return result;
144}
145Invariants invariants(Graph graph) {
146 if (graph.invariants != null) return graph.invariants!;
147 final n = graph.n;
148 final degrees = graph.rows.map(popcount).toList()..sort((a, b) => b.compareTo(a));
149 final sizes = <int>[], seen = <int>{}, eccentricities = <int>[];
150 var triangles = 0;
151 int? shortest;
152 var bipartite = true;
153 for (var u = 0; u < n; u++) {
154 for (var v = u + 1; v < n; v++) {
155 for (var w = v + 1; w < n; w++) {
156 if (graph.adjacent(u, v) && graph.adjacent(u, w) && graph.adjacent(v, w)) triangles++;
157 }
158 }
159 }
160 for (var start = 0; start < n; start++) {
161 final distance = List<int>.filled(n, -1), parent = List<int>.filled(n, -1);
162 final queue = <int>[start];
163 distance[start] = 0;
164 for (var i = 0; i < queue.length; i++) {
165 final u = queue[i];
166 for (final v in graph.neighbors(u)) {
167 if (distance[v] < 0) {
168 distance[v] = distance[u] + 1;
169 parent[v] = u;
170 queue.add(v);
171 } else if (parent[u] != v) {
172 final cycle = distance[u] + distance[v] + 1;
173 shortest = shortest == null ? cycle : math.min(shortest!, cycle);
174 }
175 if (distance[v] % 2 == distance[u] % 2) bipartite = false;
176 }
177 }
178 eccentricities.add(distance.reduce(math.max));
179 if (!seen.contains(start)) {
180 sizes.add(queue.length);
181 seen.addAll(queue);
182 }
183 }
184 sizes.sort((a, b) => b.compareTo(a));
185 final result = Invariants(degrees, triangles, sizes, shortest?.toString() ?? 'none',
186 sizes.length > 1 ? 'INF' : eccentricities.reduce(math.max).toString(),
187 sizes.length > 1 ? 'INF' : eccentricities.reduce(math.min).toString(),
188 bipartite ? 'yes' : 'no');
189 graph.invariants = result;
190 return result;
191}
192List<int> clique(Graph graph) {
193 var best = <int>[];
194 for (var mask = 1; mask < (1 << graph.n); mask++) {
195 final vertices = [for (var v = 0; v < graph.n; v++) if ((mask & (1 << v)) != 0) v];
196 if (vertices.length < best.length) continue;
197 if (vertices.every((u) => vertices.every((v) => u == v || graph.adjacent(u, v))) &&
198 (vertices.length > best.length || lex(vertices, best) < 0)) best = vertices;
199 }
200 return best;
201}
202List<int> coloring(Graph graph, int lower) {
203 final colors = List<int>.filled(graph.n, -1);
204 bool search(int vertex, int count) {
205 if (vertex == graph.n) return true;
206 for (var color = 0; color < count; color++) {
207 var valid = true;
208 for (var u = 0; u < vertex; u++) {
209 if (colors[u] == color && graph.adjacent(u, vertex)) {
210 valid = false;
211 break;
212 }
213 }
214 if (!valid) continue;
215 colors[vertex] = color;
216 if (search(vertex + 1, count)) return true;
217 }
218 colors[vertex] = -1;
219 return false;
220 }
221 for (var count = lower; count <= graph.n; count++) {
222 if (search(0, count)) return List.of(colors);
223 }
224 return colors;
225}
226class DisjointSet {
227 final List<int> parent;
228 DisjointSet(int n) : parent = List.generate(n, (i) => i);
229 int find(int x) => parent[x] == x ? x : parent[x] = find(parent[x]);
230 void join(int a, int b) { parent[find(b)] = find(a); }
231 List<List<int>> groups() {
232 final groups = <int, List<int>>{};
233 for (var i = 0; i < parent.length; i++) {
234 groups.putIfAbsent(find(i), () => []).add(i);
235 }
236 return groups.values.toList()..sort((a, b) => a.first.compareTo(b.first));
237 }

286Graph define(String command, List<String> args) {
287 final n = int.tryParse(args.length > 1 ? args[1] : '') ?? 0;
288 final source = args.length > 1 ? registry[args[1]] : null;
289 Graph result;
290 switch (command) {
291 case 'GRAPH':
292 checkSize(n);
293 result = Graph(n);
294 for (final word in args.skip(2)) {
295 final pair = word.split('-').map(int.parse).toList();
296 final u = pair[0], v = pair[1];
297 if (u >= n || v >= n || u == v) fail('bad edge $word');
298 if (result.adjacent(u, v)) fail('duplicate edge $word');
299 result.add(u, v);
300 }
301 return result;
302 case 'CYCLE':
303 case 'PATH':
304 case 'COMPLETE':
305 checkSize(n, command == 'CYCLE' ? 3 : 1);
306 result = Graph(n);
307 for (var u = 0; u < n; u++) {
308 for (var v = u + 1; v < n; v++) {
309 if (command == 'COMPLETE' || v == u + 1 || command == 'CYCLE' && u == 0 && v == n - 1) result.add(u, v);
310 }
311 }
312 return result;
313 case 'BIPARTITE':
314 final b = int.parse(args[2]);
315 if (n == 0 || b == 0) fail('bad value');
316 checkSize(n + b);
317 result = Graph(n + b);
318 for (var u = 0; u < n; u++) {
319 for (var v = 0; v < b; v++) { result.add(u, n + v); }
320 }
321 return result;
322 case 'STAR':
323 case 'WHEEL':
324 if (n < (command == 'WHEEL' ? 3 : 1)) fail('bad value');
325 checkSize(n + 1);
326 result = Graph(n + 1);
327 for (var v = 1; v <= n; v++) { result.add(0, v); }
328 if (command == 'WHEEL') {
329 for (var v = 1; v <= n; v++) { result.add(v, v % n + 1); }
330 }
331 return result;
332 case 'PETERSEN':
333 result = Graph(10);
334 for (var i = 0; i < 5; i++) {
335 result.add(i, (i + 1) % 5);
336 result.add(i, i + 5);
337 result.add(i + 5, 5 + (i + 2) % 5);
338 }
339 return result;
340 case 'HYPERCUBE':
341 if (n == 0) fail('bad value');
342 if (n >= 5) fail('too large');
343 result = Graph(1 << n);
344 for (var u = 0; u < result.n; u++) {
345 for (var bit = 0; bit < n; bit++) { result.add(u, u ^ (1 << bit)); }
346 }
347 return result;
348 case 'CIRCULANT':
349 checkSize(n, 3);
350 final jumps = args.skip(2).map(int.parse).toList();
351 for (var i = 0; i < jumps.length; i++) {
352 if (jumps[i] == 0 || 2 * jumps[i] > n || i > 0 && jumps[i - 1] >= jumps[i]) fail('bad value');
353 }
354 result = Graph(n);
355 for (var u = 0; u < n; u++) {
356 for (final jump in jumps) { result.add(u, (u + jump) % n); }
357 }
358 return result;
359 case 'GRID':
360 final c = int.parse(args[2]);
361 if (n == 0 || c == 0) fail('bad value');
362 checkSize(n * c);
363 result = Graph(n * c);
364 for (var x = 0; x < n; x++) {
365 for (var y = 0; y < c; y++) {
366 if (y + 1 < c) result.add(x * c + y, x * c + y + 1);
367 if (x + 1 < n) result.add(x * c + y, (x + 1) * c + y);
368 }
369 }
370 return result;
371 case 'COMPLEMENT':
372 result = Graph(source!.n);
373 for (var u = 0; u < source.n; u++) {
374 for (var v = u + 1; v < source.n; v++) {
375 if (!source.adjacent(u, v)) result.add(u, v);
376 }
377 }
378 return result;
379 case 'LINE':
380 final edges = source!.edges();
381 if (edges.isEmpty) fail('no edges');
382 checkSize(edges.length);
383 result = Graph(edges.length);
384 for (var u = 0; u < edges.length; u++) {
385 for (var v = u + 1; v < edges.length; v++) {
386 final (a, b) = edges[u];
387 final (c, d) = edges[v];
388 if (a == c || a == d || b == c || b == d) result.add(u, v);
389 }
390 }
391 return result;
392 case 'PRODUCT':
393 case 'UNION':
394 final other = registry[args[2]]!, a = source!.n, b = other.n;
395 checkSize(command == 'PRODUCT' ? a * b : a + b);
396 result = Graph(command == 'PRODUCT' ? a * b : a + b);
397 if (command == 'UNION') {
398 for (final (u, v) in source.edges()) { result.add(u, v); }
399 for (final (u, v) in other.edges()) { result.add(u + a, v + a); }
400 } else {
401 for (var x = 0; x < a; x++) {
402 for (final (u, v) in other.edges()) { result.add(x * b + u, x * b + v); }
403 }
404 for (final (u, v) in source.edges()) {
405 for (var y = 0; y < b; y++) { result.add(u * b + y, v * b + y); }
406 }
407 }
408 return result;
409 case 'RELABEL':
410 final permutation = args.skip(2).map(int.parse).toList();
411 if (permutation.length != source!.n || permutation.toSet().length != source.n ||
412 permutation.any((v) => v >= source.n)) fail('bad permutation');
413 return relabel(source, permutation);
414 case 'INDUCED':
415 final vertices = args.skip(2).map(int.parse).toList();
416 if (vertices.toSet().length != vertices.length || vertices.any((v) => v >= source!.n)) fail('bad vertices');
417 result = Graph(vertices.length);
418 for (var u = 0; u < vertices.length; u++) {
419 for (var v = u + 1; v < vertices.length; v++) {
420 if (source!.adjacent(vertices[u], vertices[v])) result.add(u, v);
421 }
422 }
423 return result;
424 case 'CANONIZE':
425 return relabel(source!, canon(source).labels);
426 }
427 return Graph(1);
428}
429void iso(Graph a, Graph b) {
430 if (a.n != b.n) { output.add('not isomorphic vertices ${a.n} ${b.n}'); return; }
431 final ma = a.edges().length, mb = b.edges().length;
432 if (ma != mb) { output.add('not isomorphic edges $ma $mb'); return; }
433 final x = invariants(a), y = invariants(b);
434 if (lex(x.degrees, y.degrees) != 0) { output.add('not isomorphic degrees'); return; }
435 final checks = <(String, Object, Object)>[
436 ('triangles', x.triangles, y.triangles), ('components', x.sizes.length, y.sizes.length),
437 ('girth', x.girth, y.girth), ('diameter', x.diameter, y.diameter), ('bipartite', x.bipartite, y.bipartite)
438 ];
439 for (final (name, p, q) in checks) {
440 if (p != q) { output.add('not isomorphic $name $p $q'); return; }
441 }
442 final r = stable(a), s = stable(b);
443 if (r.rounds != s.rounds || lex(cells(r.colors).map((cell) => cell.length).toList(),
444 cells(s.colors).map((cell) => cell.length).toList()) != 0) {
445 output.add('not isomorphic refinement');
446 return;
447 }
448 final ca = canon(a), cb = canon(b);
449 if (ca.string != cb.string) { output.add('not isomorphic canonical'); return; }
450 final inverse = List<int>.filled(b.n, 0);
451 for (var v = 0; v < b.n; v++) { inverse[cb.labels[v]] = v; }
452 output.add('isomorphic');
453 output.add('mapping ${ca.labels.map((label) => inverse[label]).join(' ')}');
454}
455void automorphisms(String name, Graph graph) {
456 final canonical = canon(graph), edges = graph.edges();
457 final vertices = DisjointSet(graph.n), edgeOrbits = DisjointSet(edges.length);
458 final inverse = List<int>.filled(graph.n, 0);
459 for (var v = 0; v < graph.n; v++) { inverse[canonical.labels[v]] = v; }
460 final edgeIndices = <Edge, int>{for (var i = 0; i < edges.length; i++) edges[i]: i};
461 for (final labels in canonical.bestLeaves) {
462 final sigma = labels.map((label) => inverse[label]).toList();
463 for (var v = 0; v < graph.n; v++) { vertices.join(v, sigma[v]); }
464 for (var i = 0; i < edges.length; i++) {
465 final (u, v) = edges[i];
466 final a = math.min(sigma[u], sigma[v]), b = math.max(sigma[u], sigma[v]);
467 edgeOrbits.join(i, edgeIndices[(a, b)]!);
468 }
469 }
470 final vg = vertices.groups(), eg = edgeOrbits.groups();
471 output.add('aut $name order ${canonical.bestLeaves.length}');
472 output.add('vertex orbits ${vg.length}');
473 for (final group in vg) { output.add('orbit ${group.join(' ')}'); }
474 output.add('edge orbits ${eg.length}');
475 for (final group in eg) {
476 output.add('orbit ${group.map((i) => '${edges[i].$1}-${edges[i].$2}').join(' ')}');
477 }
478 output.add('transitive vertex ${vg.length == 1 ? 'yes' : 'no'} edge ${eg.length == 1 ? 'yes' : 'no'}');
479}
480void subiso(String nameA, String nameB, Graph pattern, Graph target) {
481 var embeddings = 0, induced = 0;
482 final first = <(List<int>, bool)>[], mapping = <int>[];
483 void search(int vertex, int used, bool isInduced) {
484 if (vertex == pattern.n) {
485 embeddings++;
486 if (isInduced) induced++;
487 if (first.length < 5) first.add((List.of(mapping), isInduced));
488 return;
489 }
490 for (var v = 0; v < target.n; v++) {
491 if ((used & (1 << v)) != 0) continue;
492 var valid = true, nextInduced = isInduced;
493 for (var u = 0; u < vertex; u++) {
494 if (pattern.adjacent(u, vertex) && !target.adjacent(mapping[u], v)) {
495 valid = false;
496 break;
497 }
498 if (!pattern.adjacent(u, vertex) && target.adjacent(mapping[u], v)) nextInduced = false;
499 }
500 if (valid) {
501 mapping.add(v);
502 search(vertex + 1, used | (1 << v), nextInduced);
503 mapping.removeLast();
504 }
505 }
506 }
507 if (pattern.n <= target.n) search(0, 0, true);
508 final order = canon(pattern).bestLeaves.length;
509 output.add('subiso $nameA $nameB embeddings $embeddings induced $induced');
510 output.add('copies ${embeddings ~/ order} induced ${induced ~/ order}');
511 for (final (mapping, isInduced) in first) {
512 output.add('embedding ${mapping.join(' ')}${isInduced ? ' induced' : ''}');
513 }
514}

516 if (n == 0) fail('bad value');
517 if (n > 5) fail('too large');

519 final max = n * (n - 1) ~/ 2;
520 for (var m = 0; m < max; m++) {
521 final next = <String>{};

540 }

542 for (var m = 0; m < levels.length; m++) {
543 final level = levels[m];
544 output.add('edges $m count ${level.length} ${level.map((string) => string.isEmpty ? '-' : string).join(' ')}');
545 }

547void execute(String line) {
548 final words = line.split(' '), command = words.first, args = words.skip(1).toList();
549 final format = formats[command];
550 if (format == null) fail('unknown command $command');
551 if (args.length < format.fixed.length || format.rest == null && args.length != format.fixed.length) fail('bad arguments');
552 for (var i = 0; i < args.length; i++) {
553 if (!valid(args[i], i < format.fixed.length ? format.fixed[i] : format.rest!)) fail('bad arguments');
554 }
555 for (var i = 0; i < format.fixed.length; i++) {
556 if (format.fixed[i] == 'graph' && !registry.containsKey(args[i])) fail('no graph ${args[i]}');
557 }
558 final name = args.isEmpty ? '' : args.first, graph = registry[args.isEmpty ? '' : args.first];
559 if (format.fixed.isNotEmpty && format.fixed.first == 'name') {
560 final result = define(command, args), redefined = registry.containsKey(name);
561 registry[name] = result;
562 output.add('${redefined ? 'redefined' : 'defined'} $name n ${result.n} m ${result.edges().length}');
563 return;
564 }
565 switch (command) {
566 case 'DELETE':
567 registry.remove(name);
568 output.add('deleted $name');
569 case 'LIST':
570 output.add('graphs ${registry.length}');
571 for (final key in registry.keys.toList()..sort()) {
572 final item = registry[key]!;
573 output.add('$key n ${item.n} m ${item.edges().length}');
574 }
575 case 'SHOW':
576 output.add('graph $name n ${graph!.n} m ${graph.edges().length}');
577 for (var v = 0; v < graph.n; v++) {
578 final neighbors = graph.neighbors(v);
579 output.add('$v:${neighbors.isEmpty ? '' : ' ${neighbors.join(' ')}'}');
580 }
581 case 'INVARIANTS':
582 final value = invariants(graph!);
583 output.addAll(['invariants $name', 'degrees ${value.degrees.join(' ')}',
584 'edges ${graph.edges().length} triangles ${value.triangles}',
585 'components ${value.sizes.length} sizes ${value.sizes.join(' ')}',
586 'girth ${value.girth} diameter ${value.diameter} radius ${value.radius}',
587 'bipartite ${value.bipartite}']);
588 if (graph.n > 12) {
589 output.addAll(['clique -', 'chromatic -']);
590 } else {
591 final largest = clique(graph), colors = coloring(graph, largest.length);
592 output.add('clique ${largest.length} ${largest.join(' ')}');
593 output.add('chromatic ${colors.reduce(math.max) + 1} ${colors.join(' ')}');
594 }
595 case 'REFINE':
596 final vertices = args.skip(1).map(int.parse).toList();
597 if (vertices.any((v) => v >= graph!.n)) fail('bad vertices');
598 var colors = stable(graph!).colors, rounds = stable(graph).rounds;
599 for (final vertex in vertices) {
600 if (colors.where((color) => color == colors[vertex]).length == 1) fail('singleton $vertex');
601 final next = refine(graph, individualize(colors, vertex));
602 colors = next.colors;
603 rounds += next.rounds;
604 }
605 final partition = cells(colors);
606 output.add('refine $name rounds $rounds cells ${partition.length}');
607 for (var color = 0; color < partition.length; color++) {
608 output.add('$color: ${partition[color].join(' ')}');
609 }
610 case 'CANON':
611 final result = canon(graph!);
612 output.addAll(['canon $name nodes ${result.nodes} leaves ${result.leaves}',
613 'labels ${result.labels.join(' ')}', 'string ${result.string.isEmpty ? '-' : result.string}']);
614 case 'ISO':
615 iso(graph!, registry[args[1]]!);
616 case 'AUT':
617 automorphisms(name, graph!);
618 case 'SUBISO':
619 subiso(name, args[1], graph!, registry[args[1]]!);
620 case 'CLASSIFY':
621 classify(int.parse(name));
622 }
623}
624void main() {
625 final count = int.parse(stdin.readLineSync()!);
626 for (var i = 0; i < count; i++) {
627 try {
628 execute(stdin.readLineSync()!);
629 } on GraphError catch (error) {
630 output.add(error.message);
631 }
632 }
633 if (output.isNotEmpty) stdout.write('${output.join('\n')}\n');
634}
