36fn split(gpa: Allocator, text: Text, separator: u8) Allocator.Error![]Text {
37 var result: std.ArrayList(Text) = .empty;
38 var start: usize = 0;
39 for (text, 0..) |c, i| if (c == separator) {
40 try result.append(gpa, text[start..i]);
41 start = i + 1;
42 };
43 try result.append(gpa, text[start..]);
44 return result.items;
45}

51fn numberWord(text: Text) bool {
52 if (text.len == 0 or text.len > 3 or (text.len > 1 and text[0] == '0')) return false;
53 for (text) |c| if (!std.ascii.isDigit(c)) return false;
54 return true;
55}

57fn nameWord(text: Text) bool {
58 if (text.len == 0 or text.len > 8 or text[0] < 'a' or text[0] > 'z') return false;
59 for (text) |c| if ((c < 'a' or c > 'z') and !std.ascii.isDigit(c)) return false;
60 return true;
61}

63fn edgeWord(text: Text) bool {
64 const at = std.mem.findScalar(u8, text, '-') orelse return false;
65 return numberWord(text[0..at]) and numberWord(text[at + 1 ..]);
66}

101fn compare(a: []const usize, b: []const usize) std.math.Order {
102 for (0..@min(a.len, b.len)) |i| {
103 const order = std.math.order(a[i], b[i]);
104 if (order != .eq) return order;
105 }
106 return std.math.order(a.len, b.len);
107}

128const Graph = struct {
129 n: usize,
130 adjacent: [16]u16 = @splat(0),
131 canonical: ?*Canon = null,
132 fn has(self: *const Graph, a: usize, b: usize) bool {
133 return self.adjacent[a] & (@as(u16, 1) << @as(u4, @intCast(b))) != 0;
134 }
135 fn join(self: *Graph, a: usize, b: usize) void {
136 self.adjacent[a] |= @as(u16, 1) << @as(u4, @intCast(b));
137 self.adjacent[b] |= @as(u16, 1) << @as(u4, @intCast(a));
138 }
139 fn edges(self: *const Graph, gpa: Allocator) Allocator.Error![]Edge {
140 var result: std.ArrayList(Edge) = .empty;
141 for (0..self.n) |a| for (a + 1..self.n) |b| if (self.has(a, b)) {
142 try result.append(gpa, .{ .a = a, .b = b });
143 };
144 return result.items;
145 }
146 fn leafText(self: *const Graph, gpa: Allocator, labels: []const usize) Allocator.Error!Text {
147 const reverse = try gpa.alloc(usize, self.n);
148 for (labels, 0..) |label, v| reverse[label] = v;
149 const text = try gpa.alloc(u8, self.n * (self.n - 1) / 2);
150 var at: usize = 0;
151 for (0..self.n) |a| for (a + 1..self.n) |b| {
152 text[at] = if (self.has(reverse[a], reverse[b])) '1' else '0';
153 at += 1;
154 };
155 return text;
156 }
157 fn canon(self: *Graph, gpa: Allocator) Error!*Canon {
158 if (self.canonical) |cached| return cached;
159 const result = try gpa.create(Canon);
160 result.* = .{};
161 const initial = try refine(gpa, self, try degreeColors(gpa, self));
162 try visitCanonical(gpa, self, result, initial.colors);
163 self.canonical = result;
164 return result;
165 }
166 fn invariants(self: *const Graph, gpa: Allocator) Error!Invariants {
167 const degrees = try gpa.alloc(usize, self.n);
168 for (self.adjacent[0..self.n], 0..) |mask, i| degrees[i] = @popCount(mask);
169 std.mem.sort(usize, degrees, {}, greaterNumber);
170 var triangles: usize = 0;
171 for (0..self.n) |a| for (a + 1..self.n) |b| for (b + 1..self.n) |c| if (self.has(a, b) and self.has(a, c) and self.has(b, c)) {
172 triangles += 1;
173 };
174 var sizes: std.ArrayList(usize) = .empty;
175 var reached: [16]bool = @splat(false);
176 var coloring: [16]i8 = @splat(0);
177 var shortest: usize = 0;
178 var diameter: usize = 0;
179 var radius = self.n;
180 var bipartite = true;
181 for (0..self.n) |start| {
182 var distance: [16]i32 = @splat(-1);
183 var parent: [16]i32 = @splat(-1);
184 var queue: std.ArrayList(usize) = .empty;
185 distance[start] = 0;
186 try queue.append(gpa, start);
187 var at: usize = 0;
188 var eccentricity: usize = 0;
189 while (at < queue.items.len) : (at += 1) {
190 const v = queue.items[at];
191 eccentricity = @max(eccentricity, @as(usize, @intCast(distance[v])));
192 for (0..self.n) |w| {
193 if (!self.has(v, w)) continue;
194 if (distance[w] < 0) {
195 distance[w] = distance[v] + 1;
196 parent[w] = @intCast(v);
197 try queue.append(gpa, w);
198 } else if (parent[v] != @as(i32, @intCast(w))) {
199 const length: usize = @intCast(distance[v] + distance[w] + 1);
200 if (shortest == 0 or length < shortest) shortest = length;
201 }
202 }
203 }
204 if (!reached[start]) {
205 try sizes.append(gpa, queue.items.len);
206 for (queue.items) |v| reached[v] = true;
207 coloring[start] = 1;
208 var pending: std.ArrayList(usize) = .empty;
209 try pending.append(gpa, start);
210 var position: usize = 0;
211 while (position < pending.items.len) : (position += 1) {
212 const v = pending.items[position];
213 for (0..self.n) |w| if (self.has(v, w)) {
214 if (coloring[w] == 0) {
215 coloring[w] = -coloring[v];
216 try pending.append(gpa, w);
217 } else if (coloring[w] == coloring[v]) bipartite = false;
218 };
219 }
220 }
221 diameter = @max(diameter, eccentricity);
222 radius = @min(radius, eccentricity);
223 }
224 std.mem.sort(usize, sizes.items, {}, greaterNumber);
225 return .{ .degrees = degrees, .sizes = sizes.items, .edges = (try self.edges(gpa)).len, .triangles = triangles, .components = sizes.items.len, .girth = if (shortest == 0) "none" else try std.fmt.allocPrint(gpa, "{d}", .{shortest}), .diameter = if (sizes.items.len == 1) try std.fmt.allocPrint(gpa, "{d}", .{diameter}) else "INF", .radius = if (sizes.items.len == 1) try std.fmt.allocPrint(gpa, "{d}", .{radius}) else "INF", .bipartite = bipartite };
226 }
227 fn clique(self: *const Graph, gpa: Allocator) Error![]usize {
228 var best: []usize = &.{};
229 const limit = @as(u32, 1) << @as(u5, @intCast(self.n));
230 var mask: u32 = 1;
231 while (mask < limit) : (mask += 1) {
232 if (@popCount(mask) < best.len) continue;
233 var vertices: std.ArrayList(usize) = .empty;
234 var valid = true;
235 for (0..self.n) |v| if (mask & (@as(u32, 1) << @as(u5, @intCast(v))) != 0) {
236 try vertices.append(gpa, v);
237 if (mask & ~(@as(u32, self.adjacent[v]) | (@as(u32, 1) << @as(u5, @intCast(v)))) != 0) {
238 valid = false;
239 break;
240 }
241 };
242 if (valid and (vertices.items.len > best.len or compare(vertices.items, best) == .lt)) best = vertices.items;
243 }
244 return best;
245 }
246 fn chromatic(self: *const Graph, gpa: Allocator) Error!struct { count: usize, colors: []usize } {
247 const colors = try gpa.alloc(usize, self.n);
248 for (1..self.n + 1) |limit| if (colorSearch(self, colors, 0, limit)) return .{ .count = limit, .colors = colors };
249 unreachable;
250 }

259fn degreeColors(gpa: Allocator, g: *const Graph) Allocator.Error![]usize {
260 const degrees = try gpa.alloc(usize, g.n);
261 const ordered = try gpa.alloc(usize, g.n);
262 for (g.adjacent[0..g.n], 0..) |mask, i| {
263 degrees[i] = @popCount(mask);
264 ordered[i] = degrees[i];
265 }
266 std.mem.sort(usize, ordered, {}, lessNumber);
267 var unique: std.ArrayList(usize) = .empty;
268 for (ordered) |degree| if (unique.items.len == 0 or unique.items[unique.items.len - 1] != degree) {
269 try unique.append(gpa, degree);
270 };
271 for (degrees) |*degree| for (unique.items, 0..) |value, i| {
272 if (value == degree.*) {
273 degree.* = i;
274 break;
275 }
276 };
277 return degrees;
278}

286fn refine(gpa: Allocator, g: *const Graph, initial: []const usize) Error!Refinement {
287 var current = try gpa.dupe(usize, initial);
288 var rounds: usize = 0;
289 while (true) {
290 const signatures = try gpa.alloc([]usize, g.n);
291 const vertices = try gpa.alloc(usize, g.n);
292 for (0..g.n) |v| {
293 var neighbors: std.ArrayList(usize) = .empty;
294 for (0..g.n) |w| if (g.has(v, w)) {
295 try neighbors.append(gpa, current[w]);
296 };
297 std.mem.sort(usize, neighbors.items, {}, lessNumber);
298 signatures[v] = try gpa.alloc(usize, neighbors.items.len + 1);
299 signatures[v][0] = current[v];
300 @memcpy(signatures[v][1..], neighbors.items);
301 vertices[v] = v;
302 }
303 std.mem.sort(usize, vertices, signatures, signatureLess);
304 const next = try gpa.alloc(usize, g.n);
305 var color: usize = 0;
306 for (vertices, 0..) |v, i| {
307 if (i > 0 and compare(signatures[v], signatures[vertices[i - 1]]) != .eq) color += 1;
308 next[v] = color;
309 }
310 if (color + 1 == colorCount(current)) return .{ .colors = current, .rounds = rounds };
311 current = next;
312 rounds += 1;
313 }
314}

317 const result = try gpa.dupe(usize, colors);
318 const selected = colors[vertex];

320 result[v] += 1;

325fn visitCanonical(gpa: Allocator, g: *const Graph, result: *Canon, colors: []const usize) Error!void {
326 result.nodes += 1;
327 const sizes = try cellSizes(gpa, colors);
328 if (sizes.len == g.n) {
329 result.leaves += 1;
330 const text = try g.leafText(gpa, colors);
331 if (result.labels == null or std.mem.order(u8, text, result.text) == .gt) {
332 result.text = text;
333 result.labels = try gpa.dupe(usize, colors);
334 result.best = .empty;
335 }
336 if (equal(text, result.text)) try result.best.append(gpa, try gpa.dupe(usize, colors));
337 return;
338 }
339 var selected: usize = 0;
340 var size = g.n + 1;
341 for (sizes, 0..) |count, color| if (count > 1 and count < size) {
342 selected = color;
343 size = count;
344 };
345 for (colors, 0..) |color, v| if (color == selected) {
346 const next = try refine(gpa, g, try individualize(gpa, colors, v));
347 try visitCanonical(gpa, g, result, next.colors);
348 };
349}

357fn colorSearch(g: *const Graph, colors: []usize, v: usize, limit: usize) bool {
358 if (v == g.n) return true;
359 for (0..limit) |color| {
360 var valid = true;
361 for (0..v) |w| if (g.has(v, w) and colors[w] == color) {
362 valid = false;
363 break;
364 };
365 if (valid) {
366 colors[v] = color;
367 if (colorSearch(g, colors, v + 1, limit)) return true;
368 }
369 }
370 return false;
371}

373const Dsu = struct {
374 parents: []usize,
375 fn init(gpa: Allocator, n: usize) Allocator.Error!Dsu {
376 const parents = try gpa.alloc(usize, n);
377 for (parents, 0..) |*parent, i| parent.* = i;
378 return .{ .parents = parents };
379 }
380 fn find(self: *Dsu, x: usize) usize {
381 if (self.parents[x] != x) self.parents[x] = self.find(self.parents[x]);
382 return self.parents[x];
383 }
384 fn unite(self: *Dsu, a: usize, b: usize) void {
385 const x = self.find(a);
386 const y = self.find(b);
387 if (x != y) self.parents[y] = x;
388 }
389 fn groups(self: *Dsu, gpa: Allocator) Allocator.Error![][]usize {
390 var lists: std.ArrayList(std.ArrayList(usize)) = .empty;
391 const where = try gpa.alloc(?usize, self.parents.len);
392 @memset(where, null);
393 for (0..self.parents.len) |i| {
394 const root = self.find(i);
395 if (where[root] == null) {
396 where[root] = lists.items.len;
397 try lists.append(gpa, .empty);
398 }
399 try lists.items[where[root].?].append(gpa, i);
400 }
401 const result = try gpa.alloc([]usize, lists.items.len);
402 for (lists.items, 0..) |list, i| result[i] = list.items;
403 return result;
404 }

409const Embeddings = struct {
410 gpa: Allocator,
411 pattern: *const Graph,
412 target: *const Graph,
413 mapping: []usize,
414 used: []bool,
415 count: usize = 0,
416 induced: usize = 0,
417 examples: std.ArrayList(Example) = .empty,
418 fn visit(self: *Embeddings, v: usize, is_induced: bool) Error!void {
419 if (v == self.pattern.n) {
420 self.count += 1;
421 if (is_induced) self.induced += 1;
422 if (self.examples.items.len < 5) try self.examples.append(self.gpa, .{ .mapping = try self.gpa.dupe(usize, self.mapping), .induced = is_induced });
423 return;
424 }
425 for (0..self.target.n) |w| {
426 if (self.used[w]) continue;
427 var valid = true;
428 var next_induced = is_induced;
429 for (0..v) |u| {
430 if (self.pattern.has(u, v) and !self.target.has(self.mapping[u], w)) {
431 valid = false;
432 break;
433 }
434 if (!self.pattern.has(u, v) and self.target.has(self.mapping[u], w)) next_induced = false;
435 }
436 if (valid) {
437 self.mapping[v] = w;
438 self.used[w] = true;
439 try self.visit(v + 1, next_induced);
440 self.used[w] = false;
441 }
442 }
443 }

450const Lab = struct {
451 gpa: Allocator,
452 out: Builder,
453 graphs: std.StringHashMapUnmanaged(*Graph) = .empty,
454 fn automorphisms(self: *Lab, name: Text, g: *Graph) Error!void {
455 const gpa = self.gpa;
456 const data = try g.canon(gpa);
457 const edges = try g.edges(gpa);
458 var vertices = try Dsu.init(gpa, g.n);
459 var edge_sets = try Dsu.init(gpa, edges.len);
460 const reverse = try gpa.alloc(usize, g.n);
461 for (data.labels.?, 0..) |label, v| reverse[label] = v;
462 for (data.best.items) |labeling| {
463 const sigma = try gpa.alloc(usize, g.n);
464 for (labeling, 0..) |label, v| {
465 sigma[v] = reverse[label];
466 vertices.unite(v, sigma[v]);
467 }
468 for (edges, 0..) |edge, i| {
469 const a = @min(sigma[edge.a], sigma[edge.b]);
470 const b = @max(sigma[edge.a], sigma[edge.b]);
471 for (edges, 0..) |mapped, j| if (a == mapped.a and b == mapped.b) {
472 edge_sets.unite(i, j);
473 break;
474 };
475 }
476 }
477 const vertex_orbits = try vertices.groups(gpa);
478 const edge_orbits = try edge_sets.groups(gpa);
479 try self.out.line("aut {s} order {d}", .{ name, data.best.items.len });
480 try self.out.line("vertex orbits {d}", .{vertex_orbits.len});
481 for (vertex_orbits) |orbit| try self.out.line("orbit {s}", .{try integers(gpa, orbit)});
482 try self.out.line("edge orbits {d}", .{edge_orbits.len});
483 for (edge_orbits) |orbit| {
484 try self.out.add("orbit");
485 for (orbit) |index| try self.out.print(" {d}-{d}", .{ edges[index].a, edges[index].b });
486 try self.out.add("\n");
487 }
488 try self.out.line("transitive vertex {s} edge {s}", .{ yes(vertex_orbits.len == 1), yes(edge_orbits.len == 1) });
489 }
490 fn subiso(self: *Lab, a_name: Text, b_name: Text, pattern: *Graph, target: *Graph) Error!void {
491 const gpa = self.gpa;
492 var search: Embeddings = .{ .gpa = gpa, .pattern = pattern, .target = target, .mapping = try gpa.alloc(usize, pattern.n), .used = try gpa.alloc(bool, target.n) };
493 @memset(search.used, false);
494 if (pattern.n <= target.n) try search.visit(0, true);
495 const group_order = (try pattern.canon(gpa)).best.items.len;
496 try self.out.line("subiso {s} {s} embeddings {d} induced {d}", .{ a_name, b_name, search.count, search.induced });
497 try self.out.line("copies {d} induced {d}", .{ search.count / group_order, search.induced / group_order });
498 for (search.examples.items) |example| try self.out.line("embedding {s}{s}", .{ try integers(gpa, example.mapping), if (example.induced) " induced" else "" });
499 }
500 fn classify(self: *Lab, n: usize) Error!void {
501 const gpa = self.gpa;
502 if (n == 0) return bad(gpa, "bad value", .{});
503 if (n > 5) return bad(gpa, "too large", .{});
504 var current: std.StringHashMapUnmanaged(void) = .empty;
505 const zero = try gpa.alloc(u8, n * (n - 1) / 2);
506 @memset(zero, '0');
507 try current.put(gpa, zero, {});
508 var levels: std.ArrayList([]Text) = .empty;
509 var total: usize = 0;
510 for (0..n * (n - 1) / 2 + 1) |_| {
511 const level = try sortedKeys(gpa, current);
512 try levels.append(gpa, level);
513 total += level.len;
514 var next: std.StringHashMapUnmanaged(void) = .empty;
515 for (level) |text| {
516 const source = try makeGraph(gpa, n);
517 var index: usize = 0;
518 for (0..n) |a| for (a + 1..n) |b| {
519 if (text[index] == '1') source.join(a, b);
520 index += 1;
521 };
522 for (0..n) |a| for (a + 1..n) |b| if (!source.has(a, b)) {
523 const changed = try makeGraph(gpa, n);
524 changed.adjacent = source.adjacent;
525 changed.join(a, b);
526 try next.put(gpa, (try changed.canon(gpa)).text, {});
527 };
528 }
529 current = next;
530 }
531 try self.out.line("classify {d} total {d}", .{ n, total });
532 for (levels.items, 0..) |level, m| {
533 try self.out.print("edges {d} count {d}", .{ m, level.len });
534 for (level) |text| try self.out.print(" {s}", .{if (text.len == 0) "-" else text});
535 try self.out.add("\n");
536 }
537 }
538 fn command(self: *Lab, words: []const Text) Error!void {
539 const gpa = self.gpa;
540 const name = words[0];
541 const args = words[1..];
542 var found: ?Command = null;
543 for (command_names, 0..) |candidate, i| if (equal(candidate, name)) {
544 found = @fromBackingInt(@intCast(i));
545 break;
546 };
547 const cmd = found orelse return bad(gpa, "unknown command {s}", .{name});
548 const defining = @backingInt(cmd) < 18;
549 var minimum: usize = 1;
550 var maximum: ?usize = 1;
551 switch (cmd) {
552 .list => {
553 minimum = 0;
554 maximum = 0;
555 },
556 .graph, .relabel, .refine => {
557 minimum = if (cmd == .refine) 1 else 2;
558 maximum = null;
559 },
560 .circulant, .induced => {
561 minimum = 3;
562 maximum = null;
563 },
564 .bipartite, .grid, .product, .union_graph => {
565 minimum = 3;
566 maximum = 3;
567 },
568 .cycle, .path, .complete, .star, .wheel, .hypercube, .complement, .line, .canonize, .iso, .subiso => {
569 minimum = 2;
570 maximum = 2;
571 },
572 else => {},
573 }
574 if (args.len < minimum or (maximum != null and args.len > maximum.?)) return bad(gpa, "bad arguments", .{});
575 const name_positions = try gpa.alloc(bool, args.len);
576 @memset(name_positions, false);
577 if (defining) name_positions[0] = true;
578 const existing: []const usize = switch (cmd) {
579 .complement, .line, .relabel, .induced, .canonize => &.{1},
580 .product, .union_graph => &.{ 1, 2 },
581 .delete, .show, .invariants, .refine, .canon, .aut => &.{0},
582 .iso, .subiso => &.{ 0, 1 },
583 else => &.{},
584 };
585 for (existing) |i| name_positions[i] = true;
586 for (args, 0..) |arg, i| {
587 const valid = if (name_positions[i]) nameWord(arg) else if (cmd == .graph and i > 1) edgeWord(arg) else numberWord(arg);
588 if (!valid) return bad(gpa, "bad arguments", .{});
589 }
590 for (existing) |i| if (!self.graphs.contains(args[i])) return bad(gpa, "no graph {s}", .{args[i]});
591 var g: *Graph = undefined;
592 if (defining) {
593 switch (cmd) {
594 .graph, .cycle, .path, .complete, .star, .wheel, .hypercube, .circulant => {
595 var n = number(args[1]);
596 const lower: usize = if (cmd == .cycle or cmd == .circulant or cmd == .wheel) 3 else 1;
597 const upper: usize = if (cmd == .wheel or cmd == .star) 15 else if (cmd == .hypercube) 4 else 16;
598 if (n < lower) return bad(gpa, "bad value", .{});
599 if (n > upper) return bad(gpa, "too large", .{});
600 if (cmd == .star or cmd == .wheel) n += 1 else if (cmd == .hypercube) n = @as(usize, 1) << @as(u6, @intCast(n));
601 g = try makeGraph(gpa, n);
602 switch (cmd) {
603 .graph => for (args[2..]) |word| {
604 const at = std.mem.findScalar(u8, word, '-').?;
605 const a = number(word[0..at]);
606 const b = number(word[at + 1 ..]);
607 if (a >= n or b >= n or a == b) return bad(gpa, "bad edge {s}", .{word});
608 if (g.has(a, b)) return bad(gpa, "duplicate edge {s}", .{word});
609 g.join(a, b);
610 },
611 .cycle => for (0..n) |v| g.join(v, (v + 1) % n),
612 .path => for (0..n - 1) |v| g.join(v, v + 1),
613 .complete => for (0..n) |a| {
614 for (a + 1..n) |b| g.join(a, b);
615 },
616 .star, .wheel => for (1..n) |v| {
617 g.join(0, v);
618 if (cmd == .wheel) g.join(v, if (v + 1 == n) 1 else v + 1);
619 },
620 .hypercube => for (0..n) |v| {
621 var bit: usize = 1;
622 while (bit < n) : (bit <<= 1) g.join(v, v ^ bit);
623 },
624 .circulant => {
625 var previous: usize = 0;
626 for (args[2..]) |word| {
627 const jump = number(word);
628 if (jump <= previous or 2 * jump > n) return bad(gpa, "bad value", .{});
629 for (0..n) |v| g.join(v, (v + jump) % n);
630 previous = jump;
631 }
632 },
633 else => unreachable,
634 }
635 },
636 .bipartite, .grid => {
637 const a = number(args[1]);
638 const b = number(args[2]);
639 if (a == 0 or b == 0) return bad(gpa, "bad value", .{});
640 const n = if (cmd == .grid) a * b else a + b;
641 if (n > 16) return bad(gpa, "too large", .{});
642 g = try makeGraph(gpa, n);
643 for (0..a) |x| for (0..b) |y| {
644 if (cmd == .bipartite) g.join(x, a + y) else {
645 if (y + 1 < b) g.join(x * b + y, x * b + y + 1);
646 if (x + 1 < a) g.join(x * b + y, (x + 1) * b + y);
647 }
648 };
649 },
650 .petersen => {
651 g = try makeGraph(gpa, 10);
652 for (0..5) |i| {
653 g.join(i, (i + 1) % 5);
654 g.join(i, i + 5);
655 g.join(i + 5, 5 + (i + 2) % 5);
656 }
657 },
658 .complement => {
659 const a = self.graphs.get(args[1]).?;
660 g = try makeGraph(gpa, a.n);
661 for (0..a.n) |x| for (x + 1..a.n) |y| if (!a.has(x, y)) {
662 g.join(x, y);
663 };
664 },
665 .line => {
666 const edges = try self.graphs.get(args[1]).?.edges(gpa);
667 if (edges.len == 0) return bad(gpa, "no edges", .{});
668 if (edges.len > 16) return bad(gpa, "too large", .{});
669 g = try makeGraph(gpa, edges.len);
670 for (edges, 0..) |a, i| for (i + 1..edges.len) |j| {
671 const b = edges[j];
672 if (a.a == b.a or a.a == b.b or a.b == b.a or a.b == b.b) g.join(i, j);
673 };
674 },
675 .product, .union_graph => {
676 const a = self.graphs.get(args[1]).?;
677 const b = self.graphs.get(args[2]).?;
678 const n = if (cmd == .product) a.n * b.n else a.n + b.n;
679 if (n > 16) return bad(gpa, "too large", .{});
680 g = try makeGraph(gpa, n);
681 if (cmd == .union_graph) {
682 for (try a.edges(gpa)) |edge| g.join(edge.a, edge.b);
683 for (try b.edges(gpa)) |edge| g.join(edge.a + a.n, edge.b + a.n);
684 } else {
685 for (0..a.n) |x| for (try b.edges(gpa)) |edge| {
686 g.join(x * b.n + edge.a, x * b.n + edge.b);
687 };
688 for (0..b.n) |y| for (try a.edges(gpa)) |edge| {
689 g.join(edge.a * b.n + y, edge.b * b.n + y);
690 };
691 }
692 },
693 .relabel, .induced => {
694 const a = self.graphs.get(args[1]).?;
695 if (cmd == .relabel and args.len - 2 != a.n) return bad(gpa, "bad permutation", .{});
696 const values = try gpa.alloc(usize, args.len - 2);
697 const seen = try gpa.alloc(bool, a.n);
698 @memset(seen, false);
699 for (args[2..], 0..) |word, i| {
700 const value = number(word);
701 if (value >= a.n or seen[value]) return bad(gpa, "{s}", .{if (cmd == .relabel) "bad permutation" else "bad vertices"});
702 seen[value] = true;
703 values[i] = value;
704 }
705 if (cmd == .relabel) g = try relabel(gpa, a, values) else {
706 g = try makeGraph(gpa, values.len);
707 for (values, 0..) |x, i| for (i + 1..values.len) |j| if (a.has(x, values[j])) {
708 g.join(i, j);
709 };
710 }
711 },
712 .canonize => {
713 const source = self.graphs.get(args[1]).?;
714 g = try relabel(gpa, source, (try source.canon(gpa)).labels.?);
715 },
716 else => unreachable,
717 }
718 const action: Text = if (self.graphs.contains(args[0])) "redefined" else "defined";
719 try self.graphs.put(gpa, args[0], g);
720 try self.out.line("{s} {s} n {d} m {d}", .{ action, args[0], g.n, (try g.edges(gpa)).len });
721 return;
722 }
723 if (existing.len > 0) g = self.graphs.get(args[0]).?;
724 switch (cmd) {
725 .delete => {
726 _ = self.graphs.remove(args[0]);
727 try self.out.line("deleted {s}", .{args[0]});
728 },
729 .list => {
730 const names = try sortedKeys(gpa, self.graphs);
731 try self.out.line("graphs {d}", .{names.len});
732 for (names) |graph_name| {
733 const value = self.graphs.get(graph_name).?;
734 try self.out.line("{s} n {d} m {d}", .{ graph_name, value.n, (try value.edges(gpa)).len });
735 }
736 },
737 .show => {
738 try self.out.line("graph {s} n {d} m {d}", .{ args[0], g.n, (try g.edges(gpa)).len });
739 for (0..g.n) |v| {
740 try self.out.print("{d}:", .{v});
741 for (0..g.n) |w| if (g.has(v, w)) {
742 try self.out.print(" {d}", .{w});
743 };
744 try self.out.add("\n");
745 }
746 },
747 .invariants => {
748 const data = try g.invariants(gpa);
749 try self.out.line("invariants {s}", .{args[0]});
750 try self.out.line("degrees {s}", .{try integers(gpa, data.degrees)});
751 try self.out.line("edges {d} triangles {d}", .{ data.edges, data.triangles });
752 try self.out.line("components {d} sizes {s}", .{ data.components, try integers(gpa, data.sizes) });
753 try self.out.line("girth {s} diameter {s} radius {s}", .{ data.girth, data.diameter, data.radius });
754 try self.out.line("bipartite {s}", .{yes(data.bipartite)});
755 if (g.n > 12) try self.out.add("clique -\nchromatic -\n") else {
756 const clique = try g.clique(gpa);
757 const chromatic = try g.chromatic(gpa);
758 try self.out.line("clique {d} {s}", .{ clique.len, try integers(gpa, clique) });
759 try self.out.line("chromatic {d} {s}", .{ chromatic.count, try integers(gpa, chromatic.colors) });
760 }
761 },
762 .refine => {
763 for (args[1..]) |word| if (number(word) >= g.n) return bad(gpa, "bad vertices", .{});
764 var state = try refine(gpa, g, try degreeColors(gpa, g));
765 for (args[1..]) |word| {
766 const v = number(word);
767 if ((try cellSizes(gpa, state.colors))[state.colors[v]] == 1) return bad(gpa, "singleton {d}", .{v});
768 const next = try refine(gpa, g, try individualize(gpa, state.colors, v));
769 state = .{ .colors = next.colors, .rounds = state.rounds + next.rounds };
770 }
771 try self.out.line("refine {s} rounds {d} cells {d}", .{ args[0], state.rounds, colorCount(state.colors) });
772 for (0..colorCount(state.colors)) |color| {
773 try self.out.print("{d}:", .{color});
774 for (state.colors, 0..) |c, v| if (c == color) {
775 try self.out.print(" {d}", .{v});
776 };
777 try self.out.add("\n");
778 }
779 },
780 .canon => {
781 const data = try g.canon(gpa);
782 try self.out.line("canon {s} nodes {d} leaves {d}", .{ args[0], data.nodes, data.leaves });
783 try self.out.line("labels {s}", .{try integers(gpa, data.labels.?)});
784 try self.out.line("string {s}", .{if (data.text.len == 0) "-" else data.text});
785 },
786 .iso => {
787 const b = self.graphs.get(args[1]).?;
788 const a_data = try g.invariants(gpa);
789 const b_data = try b.invariants(gpa);
790 var message: ?Text = null;
791 if (g.n != b.n) message = try std.fmt.allocPrint(gpa, "vertices {d} {d}", .{ g.n, b.n }) else if (a_data.edges != b_data.edges) message = try std.fmt.allocPrint(gpa, "edges {d} {d}", .{ a_data.edges, b_data.edges }) else if (compare(a_data.degrees, b_data.degrees) != .eq) message = "degrees" else if (a_data.triangles != b_data.triangles) message = try std.fmt.allocPrint(gpa, "triangles {d} {d}", .{ a_data.triangles, b_data.triangles }) else if (a_data.components != b_data.components) message = try std.fmt.allocPrint(gpa, "components {d} {d}", .{ a_data.components, b_data.components }) else if (!equal(a_data.girth, b_data.girth)) message = try std.fmt.allocPrint(gpa, "girth {s} {s}", .{ a_data.girth, b_data.girth }) else if (!equal(a_data.diameter, b_data.diameter)) message = try std.fmt.allocPrint(gpa, "diameter {s} {s}", .{ a_data.diameter, b_data.diameter }) else if (a_data.bipartite != b_data.bipartite) message = try std.fmt.allocPrint(gpa, "bipartite {s} {s}", .{ yes(a_data.bipartite), yes(b_data.bipartite) }) else {
792 const ac = try refine(gpa, g, try degreeColors(gpa, g));
793 const bc = try refine(gpa, b, try degreeColors(gpa, b));
794 if (ac.rounds != bc.rounds or compare(try cellSizes(gpa, ac.colors), try cellSizes(gpa, bc.colors)) != .eq) message = "refinement" else if (!equal((try g.canon(gpa)).text, (try b.canon(gpa)).text)) message = "canonical";
795 }
796 if (message) |text| try self.out.line("not isomorphic {s}", .{text}) else {
797 const a = try g.canon(gpa);
798 const other = try b.canon(gpa);
799 const reverse = try gpa.alloc(usize, b.n);
800 const mapping = try gpa.alloc(usize, g.n);
801 for (other.labels.?, 0..) |label, v| reverse[label] = v;
802 for (a.labels.?, 0..) |label, v| mapping[v] = reverse[label];
803 try self.out.add("isomorphic\n");
804 try self.out.line("mapping {s}", .{try integers(gpa, mapping)});
805 }
806 },
807 .aut => try self.automorphisms(args[0], g),
808 .subiso => try self.subiso(args[0], args[1], g, self.graphs.get(args[1]).?),
809 .classify => try self.classify(number(args[0])),
810 else => unreachable,
811 }
812 }
813};

815pub fn main(init: std.process.Init) !void {
816 var storage = std.heap.ArenaAllocator.init(std.heap.page_allocator);
817 defer storage.deinit();
818 const gpa = storage.allocator();
819 var input_buffer: [4096]u8 = undefined;
820 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
821 var output_buffer: [4096]u8 = undefined;
822 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
823 const text = try input.interface.allocRemaining(gpa, .unlimited);
824 const lines = try split(gpa, text, '\n');
825 const n = number(lines[0]);
826 var lab: Lab = .{ .gpa = gpa, .out = .{ .gpa = gpa } };
827 for (lines[1 .. n + 1]) |line| {
828 lab.command(try split(gpa, line, ' ')) catch |err| {
829 if (err != error.Invalid) return err;
830 try lab.out.line("error: {s}", .{failure});
831 };
832 }
833 try output.interface.writeAll(lab.out.result());
834 try output.interface.flush();
835}
