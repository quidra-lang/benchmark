3class Graph
4 attr_reader :n, :adj
5
6 def initialize(n)
7 @n = n
8 @adj = Array.new(n) { Array.new(n, false) }
9 end
10
11 def join(u, v)
12 @adj[u][v] = @adj[v][u] = true
13 end
14
15 def vertices
16 (0...@n).to_a
17 end
18
19 def neighbors(v)
20 vertices.select { |w| @adj[v][w] }
21 end
22
23 def edges
24 @edges ||= vertices.flat_map { |u| ((u + 1)...@n).select { |v| @adj[u][v] }.map { |v| [u, v] } }
25 end
26
27 def degrees
28 @degrees ||= vertices.map { |v| neighbors(v).length }
29 end
30
31 def refine(colors)
32 rounds = 0
33 loop do
34 signatures = vertices.map { |v| [colors[v], neighbors(v).map { |w| colors[w] }.sort] }
35 ordered = signatures.uniq.sort
36 fresh = signatures.map { |s| ordered.index(s) }
37 return [colors, rounds] if fresh.uniq.length == colors.uniq.length
38 colors = fresh
39 rounds += 1
40 end
41 end
42
43 def stable
44 @stable ||= refine(degrees.map { |d| degrees.uniq.sort.index(d) })
45 end
46
47 def individualize(colors, v)
48 colors.each_with_index.map { |c, w| c < colors[v] || w == v ? c : c + 1 }
49 end
50
51 def leaf_string(labels)
52 inverse = Array.new(@n)
53 labels.each_with_index { |label, v| inverse[label] = v }
54 vertices.flat_map { |u| ((u + 1)...@n).map { |v| @adj[inverse[u]][inverse[v]] ? '1' : '0' } }.join
55 end
56
57 def canonical
58 return @canonical if @canonical
59 nodes = 0
60 leaves = 0
61 best = nil
62 labels = nil
63 winners = []
64 visit = lambda do |colors|
65 nodes += 1
66 if colors.uniq.length == @n
67 leaves += 1
68 string = leaf_string(colors)
69 if best.nil? || string > best
70 best = string
71 labels = colors
72 winners = [colors]
73 elsif string == best
74 winners << colors
75 end
76 else
77 cells = colors.uniq.sort.map { |c| vertices.select { |v| colors[v] == c } }
78 cell = cells.select { |c| c.length > 1 }.min_by { |c| [c.length, colors[c.first]] }
79 cell.each { |v| visit.call(refine(individualize(colors, v))[0]) }
80 end
81 end
82 visit.call(stable[0])
83 @canonical = { nodes: nodes, leaves: leaves, string: best, labels: labels, winners: winners }
84 end
85
86 def invariants
87 return @invariants if @invariants
88 components = []
89 seen = []
90 vertices.each do |start|
91 next if seen.include?(start)
92 queue = [start]
93 seen << start
94 queue.each do |u|
95 neighbors(u).each do |v|
96 next if seen.include?(v)
97 seen << v
98 queue << v
99 end
100 end
101 components << queue.length
102 end
103 girth = nil
104 eccentricities = []
105 bipartite = true
106 vertices.each do |start|
107 distance = Array.new(@n)
108 parent = Array.new(@n)
109 distance[start] = 0
110 queue = [start]
111 queue.each do |u|
112 neighbors(u).each do |v|
113 if distance[v].nil?
114 distance[v] = distance[u] + 1
115 parent[v] = u
116 queue << v
117 elsif parent[u] != v
118 length = distance[u] + distance[v] + 1
119 girth = length if girth.nil? || length < girth
120 end
121 bipartite = false if !distance[v].nil? && distance[u] % 2 == distance[v] % 2
122 end
123 end
124 eccentricities << distance.compact.max
125 end
126 triangles = vertices.combination(3).count { |a, b, c| @adj[a][b] && @adj[a][c] && @adj[b][c] }
127 @invariants = {
128 degrees: degrees.sort.reverse, triangles: triangles, sizes: components.sort.reverse,
129 components: components.length, girth: girth || 'none',
130 diameter: components.length == 1 ? eccentricities.max : 'INF',
131 radius: components.length == 1 ? eccentricities.min : 'INF', bipartite: bipartite ? 'yes' : 'no'
132 }
133 end
134
135 def clique
136 @n.downto(1) do |size|
137 found = vertices.combination(size).find { |set| set.combination(2).all? { |u, v| @adj[u][v] } }
138 return found if found
139 end
140 end
141
142 def coloring
143 1.upto(@n) do |limit|
144 colors = Array.new(@n, -1)
145 search = lambda do |v, maximum|
146 return true if v == @n
147 0.upto([limit - 1, maximum + 1].min) do |c|
148 next if (0...v).any? { |w| @adj[v][w] && colors[w] == c }
149 colors[v] = c
150 return true if search.call(v + 1, [maximum, c].max)
151 end
152 colors[v] = -1
153 false
154 end
155 return [limit, colors] if search.call(0, -1)
156 end
157 end
158end

162class GraphLab
163 SCHEMAS = {
164 'GRAPH' => [2, nil, %i[name num], :edge], 'CYCLE' => [2, 2, %i[name num]],
165 'PATH' => [2, 2, %i[name num]], 'COMPLETE' => [2, 2, %i[name num]],
166 'BIPARTITE' => [3, 3, %i[name num num]], 'STAR' => [2, 2, %i[name num]],
167 'WHEEL' => [2, 2, %i[name num]], 'PETERSEN' => [1, 1, %i[name]],
168 'HYPERCUBE' => [2, 2, %i[name num]], 'CIRCULANT' => [3, nil, %i[name num], :num],
169 'GRID' => [3, 3, %i[name num num]], 'COMPLEMENT' => [2, 2, %i[name graph]],
170 'LINE' => [2, 2, %i[name graph]], 'PRODUCT' => [3, 3, %i[name graph graph]],
171 'UNION' => [3, 3, %i[name graph graph]], 'RELABEL' => [2, nil, %i[name graph], :num],
172 'INDUCED' => [3, nil, %i[name graph], :num], 'CANONIZE' => [2, 2, %i[name graph]],
173 'DELETE' => [1, 1, %i[graph]], 'LIST' => [0, 0, []], 'SHOW' => [1, 1, %i[graph]],
174 'INVARIANTS' => [1, 1, %i[graph]], 'REFINE' => [1, nil, %i[graph], :num],
175 'CANON' => [1, 1, %i[graph]], 'ISO' => [2, 2, %i[graph graph]],
176 'AUT' => [1, 1, %i[graph]], 'SUBISO' => [2, 2, %i[graph graph]],
177 'CLASSIFY' => [1, 1, %i[num]]
178 }.freeze
179 NAME = /\A[a-z][a-z0-9]{0,7}\z/
180 NUMBER = /\A(?:0|[1-9][0-9]{0,2})\z/
181 EDGE = /\A(?:0|[1-9][0-9]{0,2})-(?:0|[1-9][0-9]{0,2})\z/
182
183 def initialize
184 @graphs = {}
185 @classes = {}
186 end
187
188 def validate(command, args)
189 schema = SCHEMAS[command]
190 raise GraphError, "unknown command #{command}" unless schema
191 minimum, maximum, types, tail = schema
192 raise GraphError, 'bad arguments' if args.length < minimum || (maximum && args.length > maximum)
193 args.each_with_index do |word, i|
194 type = types[i] || tail
195 regex = type == :num ? NUMBER : (type == :edge ? EDGE : NAME)
196 raise GraphError, 'bad arguments' unless regex.match?(word)
197 end
198 types.each_with_index do |type, i|
199 raise GraphError, "no graph #{args[i]}" if type == :graph && !@graphs.key?(args[i])
200 end
201 end
202
203 def size_check(value, minimum = 1, maximum = 16)
204 raise GraphError, 'bad value' if value < minimum
205 raise GraphError, 'too large' if value > maximum
206 end
207
208 def define(command, args)
209 name = args[0]
210 number = args[1].to_i
211 source = @graphs[args[1]]
212 case command
213 when 'GRAPH'
214 size_check(number)
215 graph = Graph.new(number)
216 args.drop(2).each do |word|
217 u, v = word.split('-').map(&:to_i)
218 raise GraphError, "bad edge #{word}" if u >= number || v >= number || u == v
219 raise GraphError, "duplicate edge #{word}" if graph.adj[u][v]
220 graph.join(u, v)
221 end
222 when 'PATH', 'CYCLE', 'COMPLETE', 'STAR', 'WHEEL', 'CIRCULANT'
223 minimum = %w[CYCLE WHEEL CIRCULANT].include?(command) ? 3 : 1
224 size_check(number, minimum, %w[STAR WHEEL].include?(command) ? 15 : 16)
225 n = number + (%w[STAR WHEEL].include?(command) ? 1 : 0)
226 graph = Graph.new(n)
227 case command
228 when 'PATH', 'CYCLE'
229 (0...(n - 1)).each { |v| graph.join(v, v + 1) }
230 graph.join(n - 1, 0) if command == 'CYCLE'
231 when 'COMPLETE'
232 graph.vertices.combination(2) { |u, v| graph.join(u, v) }
233 when 'STAR', 'WHEEL'
234 (1...n).each { |v| graph.join(0, v) }
235 if command == 'WHEEL'
236 (1...(n - 1)).each { |v| graph.join(v, v + 1) }
237 graph.join(n - 1, 1)
238 end
239 when 'CIRCULANT'
240 jumps = args.drop(2).map(&:to_i)
241 if jumps.any? { |j| j.zero? || 2 * j > n } || jumps.each_cons(2).any? { |a, b| a >= b }
242 raise GraphError, 'bad value'
243 end
244 graph.vertices.each { |v| jumps.each { |j| graph.join(v, (v + j) % n) } }
245 end
246 when 'BIPARTITE', 'GRID'
247 a, b = args.drop(1).map(&:to_i)
248 raise GraphError, 'bad value' if a.zero? || b.zero?
249 size_check(command == 'GRID' ? a * b : a + b)
250 graph = Graph.new(command == 'GRID' ? a * b : a + b)
251 a.times do |x|
252 b.times do |y|
253 if command == 'BIPARTITE'
254 graph.join(x, a + y)
255 else
256 graph.join(x * b + y, x * b + y + 1) if y + 1 < b
257 graph.join(x * b + y, (x + 1) * b + y) if x + 1 < a
258 end
259 end
260 end
261 when 'PETERSEN'
262 graph = Graph.new(10)
263 5.times do |i|
264 graph.join(i, (i + 1) % 5)
265 graph.join(i, i + 5)
266 graph.join(5 + i, 5 + (i + 2) % 5)
267 end
268 when 'HYPERCUBE'
269 size_check(number, 1, 4)
270 graph = Graph.new(2**number)
271 graph.vertices.each { |v| number.times { |bit| graph.join(v, v ^ (1 << bit)) } }
272 when 'COMPLEMENT'
273 graph = Graph.new(source.n)
274 source.vertices.combination(2) { |u, v| graph.join(u, v) unless source.adj[u][v] }
275 when 'LINE'
276 raise GraphError, 'no edges' if source.edges.empty?
277 size_check(source.edges.length)
278 graph = Graph.new(source.edges.length)
279 graph.vertices.combination(2) { |u, v| graph.join(u, v) unless (source.edges[u] & source.edges[v]).empty? }
280 when 'PRODUCT', 'UNION'
281 other = @graphs[args[2]]
282 size_check(command == 'PRODUCT' ? source.n * other.n : source.n + other.n)
283 if command == 'PRODUCT'
284 graph = Graph.new(source.n * other.n)
285 source.vertices.each { |x| other.edges.each { |u, v| graph.join(x * other.n + u, x * other.n + v) } }
286 other.vertices.each { |y| source.edges.each { |u, v| graph.join(u * other.n + y, v * other.n + y) } }
287 else
288 graph = Graph.new(source.n + other.n)
289 source.edges.each { |u, v| graph.join(u, v) }
290 other.edges.each { |u, v| graph.join(source.n + u, source.n + v) }
291 end
292 when 'RELABEL', 'CANONIZE'
293 labels = command == 'CANONIZE' ? source.canonical[:labels] : args.drop(2).map(&:to_i)
294 raise GraphError, 'bad permutation' unless labels.sort == source.vertices
295 graph = Graph.new(source.n)
296 source.edges.each { |u, v| graph.join(labels[u], labels[v]) }
297 when 'INDUCED'
298 selected = args.drop(2).map(&:to_i)
299 raise GraphError, 'bad vertices' if selected.any? { |v| v >= source.n } || selected.uniq.length != selected.length
300 graph = Graph.new(selected.length)
301 graph.vertices.combination(2) { |u, v| graph.join(u, v) if source.adj[selected[u]][selected[v]] }
302 end
303 verb = @graphs.key?(name) ? 'redefined' : 'defined'
304 @graphs[name] = graph
305 puts "#{verb} #{name} n #{graph.n} m #{graph.edges.length}"
306 end

308 def show_invariants(name, graph)
309 inv = graph.invariants
310 puts "invariants #{name}"
311 puts "degrees #{inv[:degrees].join(' ')}"
312 puts "edges #{graph.edges.length} triangles #{inv[:triangles]}"
313 puts "components #{inv[:components]} sizes #{inv[:sizes].join(' ')}"
314 puts "girth #{inv[:girth]} diameter #{inv[:diameter]} radius #{inv[:radius]}"
315 puts "bipartite #{inv[:bipartite]}"
316 if graph.n > 12
317 puts 'clique -', 'chromatic -'
318 else
319 clique = graph.clique
320 count, colors = graph.coloring
321 puts "clique #{clique.length} #{clique.join(' ')}"
322 puts "chromatic #{count} #{colors.join(' ')}"
323 end
324 end

326 def isomorphism(a, b)
327 av = a.invariants
328 bv = b.invariants
329 tests = [[:vertices, a.n, b.n], [:edges, a.edges.length, b.edges.length],
330 [:degrees, av[:degrees], bv[:degrees]], [:triangles, av[:triangles], bv[:triangles]],
331 [:components, av[:components], bv[:components]], [:girth, av[:girth], bv[:girth]],
332 [:diameter, av[:diameter], bv[:diameter]], [:bipartite, av[:bipartite], bv[:bipartite]]]
333 tests.each do |kind, x, y|
334 next if x == y
335 puts "not isomorphic #{kind}#{kind == :degrees ? '' : " #{x} #{y}"}"
336 return
337 end
338 profile = ->(g) { [g.stable[1], g.stable[0].uniq.sort.map { |c| g.stable[0].count(c) }] }
339 if profile.call(a) != profile.call(b)
340 puts 'not isomorphic refinement'
341 elsif a.canonical[:string] != b.canonical[:string]
342 puts 'not isomorphic canonical'
343 else
344 inverse = Array.new(b.n)
345 b.canonical[:labels].each_with_index { |label, v| inverse[label] = v }
346 puts 'isomorphic'
347 puts "mapping #{a.canonical[:labels].map { |c| inverse[c] }.join(' ')}"
348 end
349 end

351 def automorphisms(name, graph)
352 canon = graph.canonical
353 inverse = Array.new(graph.n)
354 canon[:labels].each_with_index { |label, v| inverse[label] = v }
355 vertex_parent = graph.vertices
356 edge_parent = (0...graph.edges.length).to_a
357 find = lambda do |parents, x|
358 x = parents[x] while parents[x] != x
359 x
360 end
361 unite = lambda { |parents, x, y| parents[find.call(parents, x)] = find.call(parents, y) }
362 edge_index = graph.edges.each_with_index.to_h
363 canon[:winners].each do |labels|
364 mapping = labels.map { |label| inverse[label] }
365 graph.vertices.each { |v| unite.call(vertex_parent, v, mapping[v]) }
366 graph.edges.each_with_index do |(u, v), i|
367 unite.call(edge_parent, i, edge_index[[mapping[u], mapping[v]].sort])
368 end
369 end
370 vertex_orbits = graph.vertices.group_by { |v| find.call(vertex_parent, v) }.values.sort_by(&:first)
371 edge_orbits = (0...graph.edges.length).group_by { |i| find.call(edge_parent, i) }.values.sort_by(&:first)
372 puts "aut #{name} order #{canon[:winners].length}"
373 puts "vertex orbits #{vertex_orbits.length}"
374 vertex_orbits.each { |orbit| puts "orbit #{orbit.join(' ')}" }
375 puts "edge orbits #{edge_orbits.length}"
376 edge_orbits.each { |orbit| puts "orbit #{orbit.map { |i| graph.edges[i].join('-') }.join(' ')}" }
377 puts "transitive vertex #{vertex_orbits.length == 1 ? 'yes' : 'no'} edge #{edge_orbits.length == 1 ? 'yes' : 'no'}"
378 end

380 def subiso(pattern_name, target_name, pattern, target)
381 count = 0
382 induced_count = 0
383 examples = []
384 mapping = []
385 search = lambda do |depth, induced|
386 if depth == pattern.n
387 count += 1
388 induced_count += 1 if induced
389 examples << [mapping.dup, induced] if examples.length < 5
390 return
391 end
392 target.vertices.each do |v|
393 next if mapping.include?(v)
394 next unless (0...depth).all? { |u| !pattern.adj[depth][u] || target.adj[v][mapping[u]] }
395 still_induced = induced && (0...depth).all? { |u| pattern.adj[depth][u] == target.adj[v][mapping[u]] }
396 mapping << v
397 search.call(depth + 1, still_induced)
398 mapping.pop
399 end
400 end
401 search.call(0, true) if pattern.n <= target.n
402 order = pattern.canonical[:winners].length
403 puts "subiso #{pattern_name} #{target_name} embeddings #{count} induced #{induced_count}"
404 puts "copies #{count / order} induced #{induced_count / order}"
405 examples.each { |map, induced| puts "embedding #{map.join(' ')}#{induced ? ' induced' : ''}" }
406 end

408 def classify(n)
409 size_check(n, 1, 5)
410 unless @classes.key?(n)
411 levels = [Set['0' * (n * (n - 1) / 2)]]
412 (n * (n - 1) / 2).times do |m|
413 next_level = Set.new
414 levels[m].each do |string|
415 base_edges = (0...n).to_a.combination(2).to_a
416 string.each_char.with_index do |bit, extra|
417 next unless bit == '0'
418 graph = Graph.new(n)
419 base_edges.each_with_index { |(u, v), i| graph.join(u, v) if string[i] == '1' || i == extra }
420 next_level << graph.canonical[:string]
421 end
422 end
423 levels << next_level
424 end
425 @classes[n] = levels
426 end
427 levels = @classes[n]
428 puts "classify #{n} total #{levels.sum(&:length)}"
429 levels.each_with_index do |strings, m|
430 puts "edges #{m} count #{strings.length} #{strings.to_a.sort.map { |s| s.empty? ? '-' : s }.join(' ')}"
431 end
432 end

434 def execute(line)
435 command, *args = line.split(' ')
436 validate(command, args)
437 if SCHEMAS[command][2][0] == :name
438 define(command, args)
439 return
440 end
441 name = args[0]
442 graph = @graphs[name]
443 case command
444 when 'DELETE'
445 @graphs.delete(name)
446 puts "deleted #{name}"
447 when 'LIST'
448 puts "graphs #{@graphs.length}"
449 @graphs.keys.sort.each { |key| puts "#{key} n #{@graphs[key].n} m #{@graphs[key].edges.length}" }
450 when 'SHOW'
451 puts "graph #{name} n #{graph.n} m #{graph.edges.length}"
452 graph.vertices.each do |v|
453 neighbors = graph.neighbors(v)
454 puts "#{v}:#{neighbors.empty? ? '' : " #{neighbors.join(' ')}"}"
455 end
456 when 'INVARIANTS'
457 show_invariants(name, graph)
458 when 'REFINE'
459 selected = args.drop(1).map(&:to_i)
460 raise GraphError, 'bad vertices' if selected.any? { |v| v >= graph.n }
461 colors, rounds = graph.stable
462 selected.each do |v|
463 raise GraphError, "singleton #{v}" if colors.count(colors[v]) == 1
464 colors, added = graph.refine(graph.individualize(colors, v))
465 rounds += added
466 end
467 puts "refine #{name} rounds #{rounds} cells #{colors.uniq.length}"
468 colors.uniq.sort.each { |c| puts "#{c}: #{graph.vertices.select { |v| colors[v] == c }.join(' ')}" }
469 when 'CANON'
470 result = graph.canonical
471 puts "canon #{name} nodes #{result[:nodes]} leaves #{result[:leaves]}"
472 puts "labels #{result[:labels].join(' ')}"
473 puts "string #{result[:string].empty? ? '-' : result[:string]}"
474 when 'ISO'
475 isomorphism(graph, @graphs[args[1]])
476 when 'AUT'
477 automorphisms(name, graph)
478 when 'SUBISO'
479 subiso(name, args[1], graph, @graphs[args[1]])
480 when 'CLASSIFY'
481 classify(name.to_i)
482 end
483 rescue GraphError => error
484 puts "error: #{error.message}"
485 end
486end
