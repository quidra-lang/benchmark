nat[] unique_values(nat[] items)
    nat[] result := []
    for item := items.sorted()[]
        if #result == 0 or result[-1] != item
            result = result.append(item)
    return result

bool lex_less(nat[] first, nat[] second)
    for: i < #first
        if i == #second
            return false
        if first[i] != second[i]
            return first[i] < second[i]
    return #first < #second

                if adj[u][v]
nat[] degree_colors(Graph graph)
    degrees := graph.degrees()
    ordered := unique_values(degrees)
    return [for degree := degrees[] then #[for other := ordered[] if other < degree then other]]

class Refinement(nat[] colors, nat rounds)

Refinement refine(Graph graph, nat[] initial)
    colors := initial
    nat rounds := 0
    while true
        nat[][] signatures := []
        for: v < graph.n
            neighbors := [for w := (0:<nat(graph.n))[] if graph.adj[v][w] then colors[w]]
            signatures = signatures.append([colors[v]].concat(neighbors.sorted()))
        nat[] representatives := []
        for: v < graph.n
            found := false
            for w := representatives[]
                if signatures[v] == signatures[w]
                    found = true
                    break
            if not found
                representatives = representatives.append(v)
        nat[] next := []
        for: v < graph.n
            next = next.append(#[for w := representatives[] if lex_less(signatures[w], signatures[v]) then w])
        if #representatives == #unique_values(colors)
            return Refinement(colors, rounds)
        colors = next
        rounds += 1
Refinement stable(Graph graph)
    return refine(graph, degree_colors(graph))
