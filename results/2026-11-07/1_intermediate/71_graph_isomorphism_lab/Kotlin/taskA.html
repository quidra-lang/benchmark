1private class Problem(val text: String) : RuntimeException(text)
2private fun fail(reason: String): Nothing = throw Problem("error: $reason")
3private fun numbers(values: List<Int>) = values.joinToString(" ")
4private fun yes(value: Boolean) = if (value) "yes" else "no"
5private fun shown(value: Int?, absent: String) = value?.toString() ?: absent
6private fun colorCount(colors: List<Int>) = colors.max()+1
7private fun individualize(colors: List<Int>, v: Int) = colors.mapIndexed {
8 w, c -> if (c < colors[v] || w == v) c else c+1
9}
10private fun compareLists(a: List<Int>, b: List<Int>): Int {
11 for (i in 0 until minOf(a.size, b.size)) if (a[i] != b[i]) return a[i].compareTo(b[i])
12 return a.size.compareTo(b.size)
13}
14private data class Canonical(var nodes: Int = 0, var leaves: Int = 0, var text: String = "", var label: List<Int> = emptyList(), val winners: MutableList<List<Int>> = mutableListOf())
15private data class Invariants(val degrees: List<Int>, val triangles: Int, val sizes: List<Int>, val girth: Int?, val diameter: Int?, val radius: Int?, val bipartite: Boolean)
16private class Graph(val n: Int) {
17 val adjacent = IntArray(n)
18 fun has(u: Int, v: Int) = adjacent[u] and (1 shl v) != 0
19 fun join(u: Int, v: Int) {
20 adjacent[u] = adjacent[u] or (1 shl v)
21 adjacent[v] = adjacent[v] or (1 shl u)
22 }
23 fun edges(): List<Pair<Int, Int>> = buildList {
24 for (u in 0 until n) for (v in u+1 until n) if (has(u, v)) add(u to v)
25 }
26 fun degrees() = adjacent.map {
27 it.countOneBits()
28 }
29 fun refine(input: List<Int>): Pair<List<Int>, Int> {
30 var colors = input
31 var rounds = 0
32 while (true) {
33 val signatures = (0 until n).map {
34 v -> colors[v] to (0 until n).filter {
35 has(v, it)
36 }.map {
37 colors[it]
38 }.sorted()
39 }
40 val order = Comparator<Pair<Int, List<Int>>> {
41 a, b -> a.first.compareTo(b.first).takeIf {
42 it != 0
43 } ?: compareLists(a.second, b.second)
44 }
45 val unique = signatures.distinct().sortedWith(order)
46 if (unique.size == colorCount(colors)) return colors to rounds
47 colors = signatures.map {
48 unique.binarySearch(it, order)
49 }
50 rounds++
51 }
52 }
53 fun stable(): Pair<List<Int>, Int> {
54 val d = degrees()
55 val unique = d.distinct().sorted()
56 return refine(d.map {
57 unique.indexOf(it)
58 })
59 }
60 fun leafString(colors: List<Int>): String {
61 val inverse = IntArray(n)
62 colors.forEachIndexed {
63 v, c -> inverse[c] = v
64 }
65 return buildString {
66 for (i in 0 until n) for (j in i+1 until n) append(if (has(inverse[i], inverse[j])) '1' else '0')
67 }
68 }
69 fun canonical(): Canonical {
70 val result = Canonical()
71 fun visit(colors: List<Int>) {
72 result.nodes++
73 val k = colorCount(colors)
74 if (k == n) {
75 result.leaves++
76 val text = leafString(colors)
77 if (result.winners.isEmpty() || text > result.text) {
78 result.text = text
79 result.label = colors
80 result.winners.clear()
81 result.winners.add(colors)
82 }
83 else if (text == result.text) result.winners.add(colors)
84 return
85 }
86 val cell = (0 until k).map {
87 c -> colors.count {
88 it == c
89 } to c
90 }.filter {
91 it.first > 1
92 }.minWith(compareBy<Pair<Int, Int>> {
93 it.first
94 }.thenBy {
95 it.second
96 }).second
97 for (v in 0 until n) if (colors[v] == cell) visit(refine(individualize(colors, v)).first)
98 }
99 visit(stable().first)
100 return result
101 }
102 fun invariants(): Invariants {
103 val seen = BooleanArray(n)
104 val sizes = mutableListOf<Int>()
105 val eccentricities = mutableListOf<Int>()
106 var girth: Int? = null
107 var bipartite = true
108 for (start in 0 until n) {
109 val distance = IntArray(n) {
110 -1
111 }
112 val parent = IntArray(n) {
113 n
114 }
115 val queue = ArrayDeque<Int>()
116 distance[start] = 0
117 queue.addLast(start)
118 while (queue.isNotEmpty()) {
119 val u = queue.removeFirst()
120 for (v in 0 until n) if (has(u, v)) {
121 if (distance[v] < 0) {
122 distance[v] = distance[u]+1
123 parent[v] = u
124 queue.addLast(v)
125 }
126 else {
127 if (distance[u]%2 == distance[v]%2) bipartite = false
128 if (parent[u] != v && parent[v] != u) {
129 val length = distance[u]+distance[v]+1
130 girth = girth?.let {
131 minOf(it, length)
132 } ?: length
133 }
134 }
135 }
136 }
137 if (!seen[start]) {
138 sizes.add(distance.count {
139 it >= 0
140 })
141 for (v in 0 until n) if (distance[v] >= 0) seen[v] = true
142 }
143 eccentricities.add(distance.max())
144 }
145 var triangles = 0
146 for (u in 0 until n) for (v in u+1 until n) for (w in v+1 until n) if (has(u,v) && has(v,w) && has(u,w)) triangles++
147 return Invariants(degrees().sortedDescending(), triangles, sizes.sortedDescending(), girth, if (sizes.size == 1) eccentricities.max() else null, if (sizes.size == 1) eccentricities.min() else null, bipartite)
148 }
149 fun clique(): List<Int> {
150 var best = emptyList<Int>()
151 for (mask in 1 until (1 shl n)) {
152 val vertices = (0 until n).filter {
153 mask and (1 shl it) != 0
154 }
155 if (vertices.size < best.size) continue
156 if (vertices.all {
157 (mask xor (1 shl it)) and adjacent[it].inv() == 0
158 } && (vertices.size > best.size || compareLists(vertices,best) < 0)) best = vertices
159 }
160 return best
161 }
162 fun coloring(): Pair<Int, List<Int>> {
163 val colors = mutableListOf<Int>()
164 fun find(limit: Int, used: Int): Boolean {
165 val v = colors.size
166 if (v == n) return true
167 for (c in 0 until minOf(limit,used+1)) {
168 if ((0 until v).any {
169 has(it,v) && colors[it] == c
170 }) continue
171 colors.add(c)
172 if (find(limit,maxOf(used,c+1))) return true
173 colors.removeAt(colors.lastIndex)
174 }
175 return false
176 }
177 for (limit in 1..n) {
178 colors.clear()
179 if (find(limit,0)) return limit to colors.toList()
180 }
181 error("no coloring")
182 }
183}
184private class UnionFind(n: Int) {
185 private val parent = IntArray(n) {
186 it
187 }
188 fun root(v: Int): Int {
189 if (parent[v] != v) parent[v] = root(parent[v])
190 return parent[v]
191 }
192 fun join(u: Int, v: Int) {
193 parent[root(u)] = root(v)
194 }
195 fun orbits() = parent.indices.groupBy {
196 root(it)
197 }.values.sortedBy {
198 it[0]
199 }
200}
201private fun graphName(s: String) = s.matches(Regex("[a-z][a-z0-9]{0,7}"))
202private fun number(s: String): Int? = if (s.matches(Regex("0|[1-9][0-9]{0,2}"))) s.toInt() else null
203private fun edge(s: String): Pair<Int, Int>? {
204 val p = s.split('-')
205 if (p.size != 2) return null
206 return (number(p[0]) ?: return null) to (number(p[1]) ?: return null)
207}
208private class Lab {
209 val registry = sortedMapOf<String, Graph>()
210 fun command(w: List<String>): List<String> {
211 val c = w[0]
212 val a = w.drop(1)
213 val count = a.size
214 val defining = setOf("GRAPH","CYCLE","PATH","COMPLETE","BIPARTITE","STAR","WHEEL","PETERSEN","HYPERCUBE","CIRCULANT","GRID","COMPLEMENT","LINE","PRODUCT","UNION","RELABEL","INDUCED","CANONIZE")
215 val known = setOf("DELETE","LIST","SHOW","INVARIANTS","REFINE","CANON","ISO","AUT","SUBISO","CLASSIFY")
216 if (c !in defining && c !in known) fail("unknown command $c")
217 val shape = when(c) {
218 "GRAPH","RELABEL" -> count >= 2
219 "CIRCULANT","INDUCED" -> count >= 3
220 "REFINE" -> count >= 1
221 "LIST" -> count == 0
222 "PETERSEN","DELETE","SHOW","INVARIANTS","CANON","AUT","CLASSIFY" -> count == 1
223 "BIPARTITE","GRID","PRODUCT","UNION" -> count == 3
224 else -> count == 2
225 }
226 if (!shape) fail("bad arguments")
227 val references = when(c) {
228 "COMPLEMENT","LINE","RELABEL","INDUCED","CANONIZE" -> listOf(1)
229 "PRODUCT","UNION" -> listOf(1,2)
230 "ISO","SUBISO" -> listOf(0,1)
231 "DELETE","SHOW","INVARIANTS","REFINE","CANON","AUT" -> listOf(0)
232 else -> emptyList()
233 }
234 val numeric = when(c) {
235 "GRAPH","CYCLE","PATH","COMPLETE","STAR","WHEEL","HYPERCUBE" -> listOf(1)
236 "BIPARTITE","GRID" -> listOf(1,2)
237 "CIRCULANT" -> (1 until count).toList()
238 "RELABEL","INDUCED" -> (2 until count).toList()
239 "REFINE" -> (1 until count).toList()
240 "CLASSIFY" -> listOf(0)
241 else -> emptyList()
242 }
243 for ((i,s) in a.withIndex()) if (!(if (i in numeric) number(s) != null else if (c == "GRAPH" && i >= 2) edge(s) != null else graphName(s))) fail("bad arguments")
244 for (i in references) if (a[i] !in registry) fail("no graph ${a[i]}")
245 fun num(i: Int) = number(a[i])!!
246 fun graph(i: Int) = registry.getValue(a[i])
247 fun bounded(n: Int, minimum: Int = 1): Int {
248 if (n < minimum) fail("bad value")
249 if (n > 16) fail("too large")
250 return n
251 }
252 if (c in defining) {
253 val g = when(c) {
254 "GRAPH","CYCLE","PATH","COMPLETE","CIRCULANT" -> Graph(bounded(num(1),if (c == "CYCLE" || c == "CIRCULANT") 3 else 1))
255 "STAR","WHEEL" -> {
256 if (num(1) < if (c == "STAR") 1 else 3) fail("bad value")
257 Graph(bounded(num(1)+1))
258 }
259 "BIPARTITE","GRID" -> {
260 if (num(1) == 0 || num(2) == 0) fail("bad value")
261 Graph(bounded(if (c == "GRID") num(1)*num(2) else num(1)+num(2)))
262 }
263 "HYPERCUBE" -> {
264 if (num(1) == 0) fail("bad value")
265 if (num(1) > 4) fail("too large")
266 Graph(1 shl num(1))
267 }
268 "PETERSEN" -> Graph(10)
269 "COMPLEMENT","RELABEL","CANONIZE" -> Graph(graph(1).n)
270 "LINE" -> {
271 val m = graph(1).edges().size
272 if (m == 0) fail("no edges")
273 Graph(bounded(m))
274 }
275 "INDUCED" -> Graph(count-2)
276 "PRODUCT" -> Graph(bounded(graph(1).n*graph(2).n))
277 "UNION" -> Graph(bounded(graph(1).n+graph(2).n))
278 else -> error("unknown definition")
279 }
280 val n = g.n
281 when(c) {
282 "GRAPH" -> for (s in a.drop(2)) {
283 val (u,v) = edge(s)!!
284 if (u >= n || v >= n || u == v) fail("bad edge $s")
285 if (g.has(u,v)) fail("duplicate edge $s")
286 g.join(u,v)
287 }
288 "CYCLE","PATH" -> {
289 for (i in 0 until n-1) g.join(i,i+1)
290 if (c == "CYCLE") g.join(n-1,0)
291 }
292 "COMPLETE" -> for (i in 0 until n) for (j in i+1 until n) g.join(i,j)
293 "BIPARTITE" -> for (i in 0 until num(1)) for (j in num(1) until n) g.join(i,j)
294 "STAR","WHEEL" -> {
295 for (i in 1 until n) g.join(0,i)
296 if (c == "WHEEL") {
297 for (i in 1 until n-1) g.join(i,i+1)
298 g.join(n-1,1)
299 }
300 }
301 "PETERSEN" -> for (i in 0..4) {
302 g.join(i,(i+1)%5)
303 g.join(i,i+5)
304 g.join(i+5,5+(i+2)%5)
305 }
306 "HYPERCUBE" -> for (u in 0 until n) for (bit in 0 until num(1)) g.join(u,u xor (1 shl bit))
307 "CIRCULANT" -> {
308 val jumps = (2 until count).map(::num)
309 if (jumps.any {
310 it == 0 || it*2 > n
311 } || jumps.zipWithNext().any {
312 it.first >= it.second
313 }) fail("bad value")
314 for (u in 0 until n) for (j in jumps) g.join(u,(u+j)%n)
315 }
316 "GRID" -> for (x in 0 until num(1)) for (y in 0 until num(2)) {
317 val u = x*num(2)+y
318 if (y+1 < num(2)) g.join(u,u+1)
319 if (x+1 < num(1)) g.join(u,u+num(2))
320 }
321 "COMPLEMENT" -> {
322 val old = graph(1)
323 for (u in 0 until n) for (v in u+1 until n) if (!old.has(u,v)) g.join(u,v)
324 }
325 "LINE" -> {
326 val edges = graph(1).edges()
327 for (i in 0 until n) for (j in i+1 until n) if (edges[i].first == edges[j].first || edges[i].first == edges[j].second || edges[i].second == edges[j].first || edges[i].second == edges[j].second) g.join(i,j)
328 }
329 "PRODUCT" -> {
330 val x = graph(1)
331 val y = graph(2)
332 for (u in 0 until x.n) for ((v,w) in y.edges()) g.join(u*y.n+v,u*y.n+w)
333 for ((u,v) in x.edges()) for (w in 0 until y.n) g.join(u*y.n+w,v*y.n+w)
334 }
335 "UNION" -> {
336 val x = graph(1)
337 for ((u,v) in x.edges()) g.join(u,v)
338 for ((u,v) in graph(2).edges()) g.join(u+x.n,v+x.n)
339 }
340 "RELABEL","CANONIZE" -> {
341 val old = graph(1)
342 val p = if (c == "CANONIZE") old.canonical().label else (2 until count).map(::num)
343 if (p.size != n || p.toSet() != (0 until n).toSet()) fail("bad permutation")
344 for ((u,v) in old.edges()) g.join(p[u],p[v])
345 }
346 "INDUCED" -> {
347 val old = graph(1)
348 val p = (2 until count).map(::num)
349 if (p.any {
350 it >= old.n
351 } || p.toSet().size != p.size) fail("bad vertices")
352 for (u in 0 until n) for (v in u+1 until n) if (old.has(p[u],p[v])) g.join(u,v)
353 }
354 }
355 val output = "${if (a[0] in registry) "redefined" else "defined"} ${a[0]} n $n m ${g.edges().size}"
356 registry[a[0]] = g
357 return listOf(output)
358 }
359 if (c == "LIST") return listOf("graphs ${registry.size}")+registry.map {
360 (name,g) -> "$name n ${g.n} m ${g.edges().size}"
361 }
362 if (c == "DELETE") {
363 registry.remove(a[0])
364 return listOf("deleted ${a[0]}")
365 }
366 if (c == "CLASSIFY") {
367 val n = num(0)
368 if (n == 0) fail("bad value")
369 if (n > 5) fail("too large")
370 val pairs = n*(n-1)/2
371 val classes = List(pairs+1) {
372 sortedSetOf<String>()
373 }
374 for (mask in 0 until (1 shl pairs)) {
375 val g = Graph(n)
376 var bit = 0
377 for (u in 0 until n) for (v in u+1 until n) {
378 if (mask and (1 shl bit) != 0) g.join(u,v)
379 bit++
380 }
381 classes[mask.countOneBits()].add(g.canonical().text)
382 }
383 return listOf("classify $n total ${classes.sumOf { it.size }}")+classes.mapIndexed {
384 m,s -> "edges $m count ${s.size} ${s.joinToString(" ") { it.ifEmpty { "-" } }}"
385 }
386 }
387 val g = graph(0)
388 val n = g.n
389 when(c) {
390 "SHOW" -> return listOf("graph ${a[0]} n $n m ${g.edges().size}")+(0 until n).map {
391 u -> "$u:"+(0 until n).filter {
392 g.has(u,it)
393 }.joinToString("") {
394 " $it"
395 }
396 }
397 "INVARIANTS" -> {
398 val x = g.invariants()
399 val out = mutableListOf("invariants ${a[0]}","degrees ${numbers(x.degrees)}","edges ${g.edges().size} triangles ${x.triangles}","components ${x.sizes.size} sizes ${numbers(x.sizes)}","girth ${shown(x.girth,"none")} diameter ${shown(x.diameter,"INF")} radius ${shown(x.radius,"INF")}","bipartite ${yes(x.bipartite)}")
400 if (n > 12) out.addAll(listOf("clique -","chromatic -")) else {

402 val (chi,colors) = g.coloring()
403 out.add("clique ${clique.size} ${numbers(clique)}")
404 out.add("chromatic $chi ${numbers(colors)}")
405 }
406 return out
407 }
408 "REFINE" -> {
409 val vertices = (1 until count).map(::num)
410 if (vertices.any {
411 it >= n
412 }) fail("bad vertices")
413 var (colors,rounds) = g.stable()
414 for (v in vertices) {
415 if (colors.count {
416 it == colors[v]

427 "CANON" -> {
428 val x = g.canonical()
429 return listOf("canon ${a[0]} nodes ${x.nodes} leaves ${x.leaves}","labels ${numbers(x.label)}","string ${x.text.ifEmpty { "-" }}")

431 "ISO" -> {
432 val h = graph(1)
433 val x = g.invariants()
434 val y = h.invariants()
435 val difference = when {
436 n != h.n -> "vertices $n ${h.n}"
437 g.edges().size != h.edges().size -> "edges ${g.edges().size} ${h.edges().size}"
438 x.degrees != y.degrees -> "degrees"
439 x.triangles != y.triangles -> "triangles ${x.triangles} ${y.triangles}"
440 x.sizes.size != y.sizes.size -> "components ${x.sizes.size} ${y.sizes.size}"
441 x.girth != y.girth -> "girth ${shown(x.girth,"none")} ${shown(y.girth,"none")}"
442 x.diameter != y.diameter -> "diameter ${shown(x.diameter,"INF")} ${shown(y.diameter,"INF")}"
443 x.bipartite != y.bipartite -> "bipartite ${yes(x.bipartite)} ${yes(y.bipartite)}"
444 else -> null
445 }

447 val (p,r) = g.stable()

451 it == c

460 "AUT" -> {
461 val result = g.canonical()
462 val edges = g.edges()

464 val edgeSets = UnionFind(edges.size)
465 val inverse = IntArray(n)
466 result.label.forEachIndexed {
467 v,c -> inverse[c] = v
468 }
469 for (label in result.winners) {
470 val map = label.map {
471 inverse[it]
472 }
473 for (v in 0 until n) vertices.join(v,map[v])
474 for ((i,e) in edges.withIndex()) {
475 val u = map[e.first]
476 val v = map[e.second]
477 edgeSets.join(i,edges.indexOf(minOf(u,v) to maxOf(u,v)))
478 }
479 }
480 val vo = vertices.orbits()
481 val eo = edgeSets.orbits()
482 val out = mutableListOf("aut ${a[0]} order ${result.winners.size}","vertex orbits ${vo.size}")
483 out.addAll(vo.map {
484 "orbit ${numbers(it)}"
485 })
486 out.add("edge orbits ${eo.size}")
487 out.addAll(eo.map {
488 orbit -> "orbit ${orbit.joinToString(" ") { "${edges[it].first}-${edges[it].second}" }}"
489 })
490 out.add("transitive vertex ${yes(vo.size == 1)} edge ${yes(eo.size == 1)}")
491 return out
492 }
493 "SUBISO" -> {
494 val target = graph(1)
495 val mapping = mutableListOf<Int>()
496 var embeddings = 0
497 var inducedCount = 0
498 val first = mutableListOf<Pair<List<Int>,Boolean>>()
499 fun enumerate(used: Int, induced: Boolean) {
500 val v = mapping.size
501 if (v == n) {
502 embeddings++
503 if (induced) inducedCount++
504 if (first.size < 5) first.add(mapping.toList() to induced)
505 return
506 }
507 for (w in 0 until target.n) {
508 if (used and (1 shl w) != 0 || (0 until v).any {
509 g.has(it,v) && !target.has(mapping[it],w)
510 }) continue
511 val next = induced && (0 until v).all {
512 g.has(it,v) == target.has(mapping[it],w)
513 }
514 mapping.add(w)
515 enumerate(used or (1 shl w),next)
516 mapping.removeAt(mapping.lastIndex)
517 }
518 }
519 if (n <= target.n) enumerate(0,true)
520 val order = g.canonical().winners.size
521 return listOf("subiso ${a[0]} ${a[1]} embeddings $embeddings induced $inducedCount","copies ${embeddings/order} induced ${inducedCount/order}")+first.map {
522 (map,induced) -> "embedding ${numbers(map)}${if (induced) " induced" else ""}"
523 }
524 }
525 }
526 error("unknown command")
527 }
