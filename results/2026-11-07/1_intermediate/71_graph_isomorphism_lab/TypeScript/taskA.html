27function fail(message: string): never { throw new Error(`error: ${message}`); }
28function lex(a: number[], b: number[]): number {
29 for (let i = 0; i < Math.min(a.length, b.length); i++) if (a[i] !== b[i]) return a[i] - b[i];
30 return a.length - b.length;
31}
32function popcount(mask: number): number {
33 let count = 0;
34 while (mask) { mask &= mask - 1; count++; }
35 return count;
36}
37function cells(colors: number[]): number[][] {
38 const result = Array.from({ length: Math.max(...colors) + 1 }, () => [] as number[]);
39 colors.forEach((color, vertex) => result[color].push(vertex));
40 return result;
41}
42function refine(graph: Graph, initial: number[]): Refinement {
43 let colors = [...initial], rounds = 0;
44 while (true) {
45 const signatures = colors.map((color, vertex) => [color, ...graph.neighbors(vertex).map(v => colors[v]).sort((a, b) => a - b)]);
46 const order = Array.from({ length: graph.n }, (_, v) => v).sort((a, b) => lex(signatures[a], signatures[b]));
47 const next = Array<number>(graph.n).fill(0);
48 let color = 0;
49 for (let i = 0; i < order.length; i++) {
50 if (i && lex(signatures[order[i - 1]], signatures[order[i]]) !== 0) color++;
51 next[order[i]] = color;
52 }
53 if (color + 1 === Math.max(...colors) + 1) return { colors, rounds };
54 colors = next; rounds++;
55 }
56}
57function stable(graph: Graph): Refinement {
58 if (!graph.stable) {
59 const degrees = graph.rows.map(popcount), unique = [...new Set(degrees)].sort((a, b) => a - b);
60 graph.stable = refine(graph, degrees.map(degree => unique.indexOf(degree)));
61 }
62 return graph.stable;
63}
64function individualize(colors: number[], vertex: number): number[] {
65 const x = colors[vertex];
66 return colors.map((color, w) => color < x || w === vertex ? color : color + 1);
67}
68function leafString(graph: Graph, labels: number[]): string {
69 const inverse = Array<number>(graph.n);
70 labels.forEach((label, v) => inverse[label] = v);
71 let result = '';
72 for (let i = 0; i < graph.n; i++) for (let j = i + 1; j < graph.n; j++) result += graph.adjacent(inverse[i], inverse[j]) ? '1' : '0';
73 return result;
74}
75function canon(graph: Graph): Canonical {
76 if (graph.canonical) return graph.canonical;
77 const result: Canonical = { nodes: 0, leaves: 0, labels: [], string: '', bestLeaves: [] };
78 function visit(colors: number[]): void {
79 result.nodes++;
80 const partition = cells(colors);
81 if (partition.length === graph.n) {
82 result.leaves++;
83 const string = leafString(graph, colors);
84 if (!result.bestLeaves.length || string > result.string) {
85 result.string = string;
86 result.labels = [...colors];
87 result.bestLeaves = [[...colors]];
88 } else if (string === result.string) result.bestLeaves.push([...colors]);
89 return;
90 }
91 const chosen = partition.filter(cell => cell.length > 1).reduce((best, cell) => cell.length < best.length ? cell : best);
92 for (const vertex of chosen) visit(refine(graph, individualize(colors, vertex)).colors);
93 }
94 visit(stable(graph).colors);
95 graph.canonical = result;
96 return result;
97}
98function invariants(graph: Graph): Invariants {
99 if (graph.invariants) return graph.invariants;
100 const n = graph.n, degrees = graph.rows.map(popcount).sort((a, b) => b - a), sizes: number[] = [];
101 const seen = new Set<number>();
102 let triangles = 0, shortest = Infinity, bipartite = true;
103 const eccentricities: number[] = [];
104 for (let u = 0; u < n; u++) for (let v = u + 1; v < n; v++) for (let w = v + 1; w < n; w++) {
105 if (graph.adjacent(u, v) && graph.adjacent(u, w) && graph.adjacent(v, w)) triangles++;
106 }
107 for (let start = 0; start < n; start++) {
108 const distance = Array<number>(n).fill(-1), parent = Array<number>(n).fill(-1), queue = [start];
109 distance[start] = 0;
110 for (let i = 0; i < queue.length; i++) {
111 const u = queue[i];
112 for (const v of graph.neighbors(u)) {
113 if (distance[v] < 0) { distance[v] = distance[u] + 1; parent[v] = u; queue.push(v); }
114 else if (parent[u] !== v) shortest = Math.min(shortest, distance[u] + distance[v] + 1);
115 if (distance[v] % 2 === distance[u] % 2) bipartite = false;
116 }
117 }
118 eccentricities.push(Math.max(...distance));
119 if (!seen.has(start)) { sizes.push(queue.length); for (const v of queue) seen.add(v); }
120 }
121 sizes.sort((a, b) => b - a);
122 const result: Invariants = {
123 degrees, triangles, sizes, girth: shortest === Infinity ? 'none' : String(shortest),
124 diameter: sizes.length > 1 ? 'INF' : String(Math.max(...eccentricities)),
125 radius: sizes.length > 1 ? 'INF' : String(Math.min(...eccentricities)),
126 bipartite: bipartite ? 'yes' : 'no'
127 };
128 graph.invariants = result;
129 return result;
130}
131function clique(graph: Graph): number[] {
132 let best: number[] = [];
133 for (let mask = 1; mask < 1 << graph.n; mask++) {
134 const vertices = Array.from({ length: graph.n }, (_, v) => v).filter(v => (mask & (1 << v)) !== 0);
135 if (vertices.length < best.length) continue;
136 if (vertices.every(u => vertices.every(v => u === v || graph.adjacent(u, v))) &&
137 (vertices.length > best.length || lex(vertices, best) < 0)) best = vertices;
138 }
139 return best;
140}
141function coloring(graph: Graph, lower: number): number[] {
142 const colors = Array<number>(graph.n).fill(-1);
143 function search(vertex: number, count: number): boolean {
144 if (vertex === graph.n) return true;
145 for (let color = 0; color < count; color++) {
146 let valid = true;
147 for (let u = 0; u < vertex; u++) if (colors[u] === color && graph.adjacent(u, vertex)) { valid = false; break; }
148 if (!valid) continue;
149 colors[vertex] = color;
150 if (search(vertex + 1, count)) return true;
151 }
152 colors[vertex] = -1;
153 return false;
154 }
155 for (let count = lower; count <= graph.n; count++) if (search(0, count)) return [...colors];
156 return colors;
157}
158class DisjointSet {
159 readonly parent: number[];
160 constructor(n: number) { this.parent = Array.from({ length: n }, (_, i) => i); }
161 find(x: number): number { return this.parent[x] === x ? x : this.parent[x] = this.find(this.parent[x]); }
162 join(a: number, b: number): void { this.parent[this.find(b)] = this.find(a); }
163 groups(): number[][] {
164 const groups = new Map<number, number[]>();
165 for (let i = 0; i < this.parent.length; i++) {
166 const root = this.find(i);
167 if (!groups.has(root)) groups.set(root, []);
168 groups.get(root)!.push(i);
169 }
170 return [...groups.values()].sort((a, b) => a[0] - b[0]);
171 }

187const namePattern = /^[a-z][a-z0-9]{0,7}$/;

189function valid(word: string, kind: string): boolean {
190 if (kind === 'num') return numberPattern.test(word);
191 if (kind === 'edge') { const pair = word.split('-'); return pair.length === 2 && pair.every(value => numberPattern.test(value)); }
192 return namePattern.test(word);
193}

203function define(command: string, args: string[]): Graph {
204 const n = Number(args[1]), source = registry.get(args[1])!;
205 let result: Graph;
206 switch (command) {
207 case 'GRAPH': {
208 checkSize(n); result = new Graph(n);
209 for (const word of args.slice(2)) {
210 const [u, v] = word.split('-').map(Number);
211 if (u >= n || v >= n || u === v) fail(`bad edge ${word}`);
212 if (result.adjacent(u, v)) fail(`duplicate edge ${word}`);
213 result.add(u, v);
214 }
215 return result;
216 }
217 case 'CYCLE': case 'PATH': case 'COMPLETE':
218 checkSize(n, command === 'CYCLE' ? 3 : 1); result = new Graph(n);
219 for (let u = 0; u < n; u++) for (let v = u + 1; v < n; v++) {
220 if (command === 'COMPLETE' || v === u + 1 || command === 'CYCLE' && u === 0 && v === n - 1) result.add(u, v);
221 }
222 return result;
223 case 'BIPARTITE': {
224 const b = Number(args[2]);
225 if (!n || !b) fail('bad value');
226 checkSize(n + b); result = new Graph(n + b);
227 for (let u = 0; u < n; u++) for (let v = 0; v < b; v++) result.add(u, n + v);
228 return result;
229 }
230 case 'STAR': case 'WHEEL':
231 if (n < (command === 'WHEEL' ? 3 : 1)) fail('bad value');
232 checkSize(n + 1); result = new Graph(n + 1);
233 for (let v = 1; v <= n; v++) result.add(0, v);
234 if (command === 'WHEEL') for (let v = 1; v <= n; v++) result.add(v, v % n + 1);
235 return result;
236 case 'PETERSEN':
237 result = new Graph(10);
238 for (let i = 0; i < 5; i++) { result.add(i, (i + 1) % 5); result.add(i, i + 5); result.add(i + 5, 5 + (i + 2) % 5); }
239 return result;
240 case 'HYPERCUBE':
241 if (!n) fail('bad value');
242 if (n >= 5) fail('too large');
243 result = new Graph(2 ** n);
244 for (let u = 0; u < result.n; u++) for (let bit = 0; bit < n; bit++) result.add(u, u ^ (1 << bit));
245 return result;
246 case 'CIRCULANT': {
247 checkSize(n, 3);
248 const jumps = args.slice(2).map(Number);
249 if (jumps.some((jump, i) => jump === 0 || 2 * jump > n || i > 0 && jumps[i - 1] >= jump)) fail('bad value');
250 result = new Graph(n);
251 for (let u = 0; u < n; u++) for (const jump of jumps) result.add(u, (u + jump) % n);
252 return result;
253 }
254 case 'GRID': {
255 const c = Number(args[2]);
256 if (!n || !c) fail('bad value');
257 checkSize(n * c); result = new Graph(n * c);
258 for (let x = 0; x < n; x++) for (let y = 0; y < c; y++) {
259 if (y + 1 < c) result.add(x * c + y, x * c + y + 1);
260 if (x + 1 < n) result.add(x * c + y, (x + 1) * c + y);
261 }
262 return result;
263 }
264 case 'COMPLEMENT':
265 result = new Graph(source.n);
266 for (let u = 0; u < source.n; u++) for (let v = u + 1; v < source.n; v++) if (!source.adjacent(u, v)) result.add(u, v);
267 return result;
268 case 'LINE': {
269 const edges = source.edges();
270 if (!edges.length) fail('no edges');
271 checkSize(edges.length); result = new Graph(edges.length);
272 for (let u = 0; u < edges.length; u++) for (let v = u + 1; v < edges.length; v++) if (edges[u].some(vertex => edges[v].includes(vertex))) result.add(u, v);
273 return result;
274 }
275 case 'PRODUCT': case 'UNION': {
276 const other = registry.get(args[2])!, a = source.n, b = other.n;
277 checkSize(command === 'PRODUCT' ? a * b : a + b);
278 result = new Graph(command === 'PRODUCT' ? a * b : a + b);
279 if (command === 'UNION') {
280 for (const [u, v] of source.edges()) result.add(u, v);
281 for (const [u, v] of other.edges()) result.add(u + a, v + a);
282 } else {
283 for (let x = 0; x < a; x++) for (const [u, v] of other.edges()) result.add(x * b + u, x * b + v);
284 for (const [u, v] of source.edges()) for (let y = 0; y < b; y++) result.add(u * b + y, v * b + y);
285 }
286 return result;
287 }
288 case 'RELABEL': {
289 const permutation = args.slice(2).map(Number);
290 if (permutation.length !== source.n || new Set(permutation).size !== source.n || permutation.some(v => v >= source.n)) fail('bad permutation');
291 return relabel(source, permutation);
292 }
293 case 'INDUCED': {
294 const vertices = args.slice(2).map(Number);
295 if (new Set(vertices).size !== vertices.length || vertices.some(v => v >= source.n)) fail('bad vertices');
296 result = new Graph(vertices.length);
297 for (let u = 0; u < vertices.length; u++) for (let v = u + 1; v < vertices.length; v++) if (source.adjacent(vertices[u], vertices[v])) result.add(u, v);
298 return result;
299 }
300 case 'CANONIZE': return relabel(source, canon(source).labels);
301 }
302 return new Graph(1);
303}
304function iso(a: Graph, b: Graph): void {
305 if (a.n !== b.n) { output.push(`not isomorphic vertices ${a.n} ${b.n}`); return; }
306 const ma = a.edges().length, mb = b.edges().length;
307 if (ma !== mb) { output.push(`not isomorphic edges ${ma} ${mb}`); return; }
308 const x = invariants(a), y = invariants(b);
309 if (lex(x.degrees, y.degrees)) { output.push('not isomorphic degrees'); return; }
310 const checks: [string, string | number, string | number][] = [
311 ['triangles', x.triangles, y.triangles], ['components', x.sizes.length, y.sizes.length],
312 ['girth', x.girth, y.girth], ['diameter', x.diameter, y.diameter], ['bipartite', x.bipartite, y.bipartite]
313 ];
314 for (const [name, p, q] of checks) if (p !== q) { output.push(`not isomorphic ${name} ${p} ${q}`); return; }
315 const r = stable(a), s = stable(b);
316 if (r.rounds !== s.rounds || lex(cells(r.colors).map(cell => cell.length), cells(s.colors).map(cell => cell.length))) {
317 output.push('not isomorphic refinement'); return;
318 }
319 const ca = canon(a), cb = canon(b);
320 if (ca.string !== cb.string) { output.push('not isomorphic canonical'); return; }
321 const inverse = Array<number>(b.n);
322 cb.labels.forEach((label, v) => inverse[label] = v);
323 output.push('isomorphic', `mapping ${ca.labels.map(label => inverse[label]).join(' ')}`);
324}
325function automorphisms(name: string, graph: Graph): void {
326 const canonical = canon(graph), edges = graph.edges();
327 const vertices = new DisjointSet(graph.n), edgeOrbits = new DisjointSet(edges.length);
328 const inverse = Array<number>(graph.n);
329 canonical.labels.forEach((label, v) => inverse[label] = v);
330 const edgeIndices = new Map<string, number>(edges.map(([u, v], i): [string, number] => [`${u}-${v}`, i]));
331 for (const labels of canonical.bestLeaves) {
332 const sigma = labels.map(label => inverse[label]);
333 sigma.forEach((w, v) => vertices.join(v, w));
334 edges.forEach(([u, v], i) => {
335 const a = Math.min(sigma[u], sigma[v]), b = Math.max(sigma[u], sigma[v]);
336 edgeOrbits.join(i, edgeIndices.get(`${a}-${b}`)!);
337 });
338 }
339 const vg = vertices.groups(), eg = edgeOrbits.groups();
340 output.push(`aut ${name} order ${canonical.bestLeaves.length}`, `vertex orbits ${vg.length}`);
341 for (const group of vg) output.push(`orbit ${group.join(' ')}`);
342 output.push(`edge orbits ${eg.length}`);
343 for (const group of eg) output.push(`orbit ${group.map(i => edges[i].join('-')).join(' ')}`);
344 output.push(`transitive vertex ${vg.length === 1 ? 'yes' : 'no'} edge ${eg.length === 1 ? 'yes' : 'no'}`);
345}
346function subiso(nameA: string, nameB: string, pattern: Graph, target: Graph): void {
347 let embeddings = 0, induced = 0;
348 const first: { mapping: number[]; induced: boolean }[] = [], mapping: number[] = [];
349 function search(vertex: number, used: number, isInduced: boolean): void {
350 if (vertex === pattern.n) {
351 embeddings++; if (isInduced) induced++;
352 if (first.length < 5) first.push({ mapping: [...mapping], induced: isInduced });
353 return;
354 }
355 for (let v = 0; v < target.n; v++) if (!(used & (1 << v))) {
356 let valid = true, nextInduced = isInduced;
357 for (let u = 0; u < vertex; u++) {
358 if (pattern.adjacent(u, vertex) && !target.adjacent(mapping[u], v)) { valid = false; break; }
359 if (!pattern.adjacent(u, vertex) && target.adjacent(mapping[u], v)) nextInduced = false;
360 }
361 if (valid) { mapping.push(v); search(vertex + 1, used | (1 << v), nextInduced); mapping.pop(); }
362 }
363 }
364 if (pattern.n <= target.n) search(0, 0, true);
365 const order = canon(pattern).bestLeaves.length;
366 output.push(`subiso ${nameA} ${nameB} embeddings ${embeddings} induced ${induced}`, `copies ${embeddings / order} induced ${induced / order}`);
367 for (const item of first) output.push(`embedding ${item.mapping.join(' ')}${item.induced ? ' induced' : ''}`);
368}
369function classify(n: number): void {
370 if (!n) fail('bad value');
371 if (n > 5) fail('too large');
372 const levels: string[][] = [[canon(new Graph(n)).string]];
373 const max = n * (n - 1) / 2;
374 for (let m = 0; m < max; m++) {
375 const next = new Set<string>();
376 for (const string of levels[m]) {
377 const graph = new Graph(n);
378 let i = 0;
379 for (let u = 0; u < n; u++) for (let v = u + 1; v < n; v++) if (string[i++] === '1') graph.add(u, v);
380 for (let u = 0; u < n; u++) for (let v = u + 1; v < n; v++) if (!graph.adjacent(u, v)) {
381 const added = relabel(graph, Array.from({ length: n }, (_, vertex) => vertex));
382 added.add(u, v); next.add(canon(added).string);
383 }
384 }
385 levels.push([...next].sort());
386 }
387 output.push(`classify ${n} total ${levels.reduce((sum, level) => sum + level.length, 0)}`);
388 levels.forEach((level, m) => output.push(`edges ${m} count ${level.length} ${level.map(string => string || '-').join(' ')}`));
389}
390function execute(line: string): void {
391 const [command, ...args] = line.split(' '), format = formats.get(command);
392 if (!format) fail(`unknown command ${command}`);
393 if (args.length < format.fixed.length || !format.rest && args.length !== format.fixed.length) fail('bad arguments');
394 for (let i = 0; i < args.length; i++) if (!valid(args[i], format.fixed[i] ?? format.rest!)) fail('bad arguments');
395 format.fixed.forEach((kind, i) => { if (kind === 'graph' && !registry.has(args[i])) fail(`no graph ${args[i]}`); });
396 const name = args[0], graph = registry.get(name)!;
397 if (format.fixed[0] === 'name') {
398 const result = define(command, args), redefined = registry.has(name);
399 registry.set(name, result);
400 output.push(`${redefined ? 'redefined' : 'defined'} ${name} n ${result.n} m ${result.edges().length}`);
401 return;
402 }
403 switch (command) {
404 case 'DELETE': registry.delete(name); output.push(`deleted ${name}`); break;
405 case 'LIST':
406 output.push(`graphs ${registry.size}`);
407 for (const key of [...registry.keys()].sort()) {
408 const item = registry.get(key)!;
409 output.push(`${key} n ${item.n} m ${item.edges().length}`);
410 }
411 break;
412 case 'SHOW':
413 output.push(`graph ${name} n ${graph.n} m ${graph.edges().length}`);
414 for (let v = 0; v < graph.n; v++) { const neighbors = graph.neighbors(v); output.push(`${v}:${neighbors.length ? ` ${neighbors.join(' ')}` : ''}`); }
415 break;
416 case 'INVARIANTS': {
417 const value = invariants(graph);
418 output.push(`invariants ${name}`, `degrees ${value.degrees.join(' ')}`, `edges ${graph.edges().length} triangles ${value.triangles}`,
419 `components ${value.sizes.length} sizes ${value.sizes.join(' ')}`, `girth ${value.girth} diameter ${value.diameter} radius ${value.radius}`, `bipartite ${value.bipartite}`);
420 if (graph.n > 12) output.push('clique -', 'chromatic -');
421 else {
422 const largest = clique(graph), colors = coloring(graph, largest.length);
423 output.push(`clique ${largest.length} ${largest.join(' ')}`, `chromatic ${Math.max(...colors) + 1} ${colors.join(' ')}`);
424 }
425 break;
426 }
427 case 'REFINE': {
428 const vertices = args.slice(1).map(Number);
429 if (vertices.some(v => v >= graph.n)) fail('bad vertices');
430 let { colors, rounds } = stable(graph);
431 for (const vertex of vertices) {
432 if (colors.filter(color => color === colors[vertex]).length === 1) fail(`singleton ${vertex}`);
433 const next = refine(graph, individualize(colors, vertex));
434 colors = next.colors; rounds += next.rounds;
435 }
436 const partition = cells(colors);
437 output.push(`refine ${name} rounds ${rounds} cells ${partition.length}`);
438 partition.forEach((cell, color) => output.push(`${color}: ${cell.join(' ')}`));
439 break;
440 }
441 case 'CANON': {
442 const result = canon(graph);
443 output.push(`canon ${name} nodes ${result.nodes} leaves ${result.leaves}`, `labels ${result.labels.join(' ')}`, `string ${result.string || '-'}`);
444 break;
445 }
446 case 'ISO': iso(graph, registry.get(args[1])!); break;
447 case 'AUT': automorphisms(name, graph); break;
448 case 'SUBISO': subiso(name, args[1], graph, registry.get(args[1])!); break;
449 case 'CLASSIFY': classify(Number(name)); break;
450 }
451}
452const lines = fs.readFileSync(0, 'utf8').split(/\r?\n/), count = Number(lines[0]);
453for (const line of lines.slice(1, count + 1)) {
454 try { execute(line); }
