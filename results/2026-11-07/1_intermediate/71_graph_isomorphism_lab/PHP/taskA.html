2function compare_sequences(array $a, array $b): int
3{
4 for ($i = 0; $i < min(count($a), count($b)); $i++) {
5 if ($a[$i] !== $b[$i]) {
6 return $a[$i] <=> $b[$i];
7 }
8 }
9 return count($a) <=> count($b);
10}

62 public function refine(array $colors): array
63 {
64 $rounds = 0;
65 while (true) {
66 $signatures = [];
67 foreach ($this->vertices() as $v) {
68 $neighbors = array_map(fn($w) => $colors[$w], $this->neighbors($v));
69 sort($neighbors, SORT_NUMERIC);
70 $signatures[] = [$colors[$v], $neighbors];
71 }
72 $ordered = $signatures;
73 usort($ordered, fn($a, $b) => ($a[0] <=> $b[0]) ?: compare_sequences($a[1], $b[1]));
74 $unique = [];
75 foreach ($ordered as $signature) {
76 if (!$unique || $unique[count($unique) - 1] !== $signature) {
77 $unique[] = $signature;
78 }
79 }
80 $fresh = array_map(fn($s) => array_search($s, $unique, true), $signatures);
81 if (count($unique) === count(array_unique($colors))) {
82 return [$colors, $rounds];
83 }
84 $colors = $fresh;
85 $rounds++;
86 }

121 public function canonical(): array
122 {
123 if ($this->canonicalCache !== null) {
124 return $this->canonicalCache;
125 }
126 $nodes = 0;
127 $leaves = 0;
128 $best = null;
129 $labels = [];
130 $winners = [];
131 $visit = function (array $colors) use (&$visit, &$nodes, &$leaves, &$best, &$labels, &$winners): void {
132 $nodes++;
133 if (count(array_unique($colors)) === $this->n) {
134 $leaves++;
135 $string = $this->leafString($colors);
136 if ($best === null || strcmp($string, $best) > 0) {
137 $best = $string;
138 $labels = $colors;
139 $winners = [$colors];
140 } elseif ($string === $best) {
141 $winners[] = $colors;
142 }
143 return;
144 }
145 $cells = [];
146 foreach ($colors as $v => $color) {
147 $cells[$color][] = $v;
148 }
149 ksort($cells, SORT_NUMERIC);
150 $chosen = null;
151 foreach ($cells as $cell) {
152 if (count($cell) > 1 && ($chosen === null || count($cell) < count($chosen))) {
153 $chosen = $cell;
154 }
155 }
156 foreach ($chosen as $v) {
157 $visit($this->refine($this->individualize($colors, $v))[0]);
158 }
159 };
160 $visit($this->stable()[0]);
161 return $this->canonicalCache = compact('nodes', 'leaves', 'best', 'labels', 'winners');

164 public function invariants(): array
165 {
166 if ($this->invariantCache !== null) {
167 return $this->invariantCache;
168 }
169 $components = [];
170 $seen = [];
171 foreach ($this->vertices() as $start) {
172 if (isset($seen[$start])) {
173 continue;
174 }
175 $queue = [$start];
176 $seen[$start] = true;
177 for ($i = 0; $i < count($queue); $i++) {
178 foreach ($this->neighbors($queue[$i]) as $v) {
179 if (!isset($seen[$v])) {
180 $seen[$v] = true;
181 $queue[] = $v;
182 }
183 }
184 }
185 $components[] = count($queue);
186 }
187 $girth = null;
188 $eccentricities = [];
189 $bipartite = true;
190 foreach ($this->vertices() as $start) {
191 $distance = [$start => 0];
192 $parent = [];
193 $queue = [$start];
194 for ($i = 0; $i < count($queue); $i++) {
195 $u = $queue[$i];
196 foreach ($this->neighbors($u) as $v) {
197 if (!isset($distance[$v])) {
198 $distance[$v] = $distance[$u] + 1;
199 $parent[$v] = $u;
200 $queue[] = $v;
201 } elseif (($parent[$u] ?? null) !== $v) {
202 $length = $distance[$u] + $distance[$v] + 1;
203 $girth = $girth === null ? $length : min($girth, $length);
204 }
205 if ($distance[$u] % 2 === $distance[$v] % 2) {
206 $bipartite = false;
207 }
208 }
209 }
210 $eccentricities[] = max($distance);
211 }
212 $triangles = 0;
213 for ($a = 0; $a < $this->n; $a++) {
214 for ($b = $a + 1; $b < $this->n; $b++) {
215 for ($c = $b + 1; $c < $this->n; $c++) {
216 if ($this->adj[$a][$b] && $this->adj[$a][$c] && $this->adj[$b][$c]) {
217 $triangles++;
218 }
219 }
220 }
221 }
222 $degrees = $this->degrees();
223 rsort($degrees, SORT_NUMERIC);
224 rsort($components, SORT_NUMERIC);
225 return $this->invariantCache = [
226 'degrees' => $degrees, 'triangles' => $triangles, 'sizes' => $components,
227 'components' => count($components), 'girth' => $girth ?? 'none',
228 'diameter' => count($components) === 1 ? max($eccentricities) : 'INF',
229 'radius' => count($components) === 1 ? min($eccentricities) : 'INF',
230 'bipartite' => $bipartite ? 'yes' : 'no'
231 ];
232 }

234 public function clique(): array
235 {
236 $best = [];
237 for ($mask = 1; $mask < (1 << $this->n); $mask++) {
238 $set = [];
239 foreach ($this->vertices() as $v) {
240 if ($mask & (1 << $v)) {
241 foreach ($set as $w) {
242 if (!$this->adj[$v][$w]) {
243 continue 3;
244 }
245 }
246 $set[] = $v;
247 }
248 }
249 if (count($set) > count($best) || (count($set) === count($best) && compare_sequences($set, $best) < 0)) {
250 $best = $set;
251 }
252 }
253 return $best;
254 }

256 public function coloring(): array
257 {
258 for ($limit = 1; $limit <= $this->n; $limit++) {
259 $colors = array_fill(0, $this->n, -1);
260 $search = function (int $v, int $maximum) use (&$search, &$colors, $limit): bool {
261 if ($v === $this->n) {
262 return true;
263 }
264 for ($color = 0; $color <= min($limit - 1, $maximum + 1); $color++) {
265 for ($w = 0; $w < $v; $w++) {
266 if ($this->adj[$v][$w] && $colors[$w] === $color) {
267 continue 2;
268 }
269 }
270 $colors[$v] = $color;
271 if ($search($v + 1, max($maximum, $color))) {
272 return true;
273 }
274 }
275 $colors[$v] = -1;
276 return false;
277 };
278 if ($search(0, -1)) {
279 return [$limit, $colors];
280 }
281 }

308 private function validate(string $command, array $args): array
309 {
310 if (!isset(self::SCHEMAS[$command])) {
311 throw new GraphError("unknown command $command");
312 }
313 $schema = self::SCHEMAS[$command];
314 [$minimum, $maximum, $types] = $schema;
315 if (count($args) < $minimum || ($maximum !== null && count($args) > $maximum)) {
316 throw new GraphError('bad arguments');
317 }
318 foreach ($args as $i => $word) {
319 $type = $types[$i] ?? $schema[3];
320 $pattern = match ($type) {
321 'num' => '~\A(?:0|[1-9][0-9]{0,2})\z~',
322 'edge' => '~\A(?:0|[1-9][0-9]{0,2})-(?:0|[1-9][0-9]{0,2})\z~',
323 default => '~\A[a-z][a-z0-9]{0,7}\z~'
324 };
325 if (!preg_match($pattern, $word)) {
326 throw new GraphError('bad arguments');
327 }
328 }
329 foreach ($types as $i => $type) {
330 if ($type === 'graph' && !isset($this->graphs[$args[$i]])) {
331 throw new GraphError("no graph {$args[$i]}");
332 }
333 }
334 return $schema;

347 private function define(string $command, array $args): void
348 {
349 $name = $args[0];
350 $number = (int)($args[1] ?? 0);
351 $source = $this->graphs[$args[1] ?? ''] ?? null;
352 switch ($command) {
353 case 'GRAPH':
354 $this->sizeCheck($number);
355 $graph = new Graph($number);
356 foreach (array_slice($args, 2) as $word) {
357 [$u, $v] = array_map('intval', explode('-', $word));
358 if ($u >= $number || $v >= $number || $u === $v) {
359 throw new GraphError("bad edge $word");
360 }
361 if ($graph->adj[$u][$v]) {
362 throw new GraphError("duplicate edge $word");
363 }
364 $graph->join($u, $v);
365 }
366 break;
367 case 'PATH':
368 case 'CYCLE':
369 case 'COMPLETE':
370 case 'STAR':
371 case 'WHEEL':
372 case 'CIRCULANT':
373 $this->sizeCheck($number, in_array($command, ['CYCLE', 'WHEEL', 'CIRCULANT']) ? 3 : 1,
374 in_array($command, ['STAR', 'WHEEL']) ? 15 : 16);
375 $n = $number + (in_array($command, ['STAR', 'WHEEL']) ? 1 : 0);
376 $graph = new Graph($n);
377 if ($command === 'PATH' || $command === 'CYCLE') {
378 for ($v = 0; $v + 1 < $n; $v++) {
379 $graph->join($v, $v + 1);
380 }
381 if ($command === 'CYCLE') {
382 $graph->join($n - 1, 0);
383 }
384 } elseif ($command === 'COMPLETE') {
385 for ($u = 0; $u < $n; $u++) {
386 for ($v = $u + 1; $v < $n; $v++) {
387 $graph->join($u, $v);
388 }
389 }
390 } elseif ($command === 'STAR' || $command === 'WHEEL') {
391 for ($v = 1; $v < $n; $v++) {
392 $graph->join(0, $v);
393 }
394 if ($command === 'WHEEL') {
395 for ($v = 1; $v + 1 < $n; $v++) {
396 $graph->join($v, $v + 1);
397 }
398 $graph->join($n - 1, 1);
399 }
400 } else {
401 $jumps = array_map('intval', array_slice($args, 2));
402 $previous = 0;
403 foreach ($jumps as $jump) {
404 if ($jump <= $previous || 2 * $jump > $n) {
405 throw new GraphError('bad value');
406 }
407 $previous = $jump;
408 }
409 foreach ($graph->vertices() as $v) {
410 foreach ($jumps as $jump) {
411 $graph->join($v, ($v + $jump) % $n);
412 }
413 }
414 }
415 break;
416 case 'BIPARTITE':
417 case 'GRID':
418 $a = (int)$args[1];
419 $b = (int)$args[2];
420 if ($a === 0 || $b === 0) {
421 throw new GraphError('bad value');
422 }
423 $n = $command === 'GRID' ? $a * $b : $a + $b;
424 $this->sizeCheck($n);
425 $graph = new Graph($n);
426 for ($x = 0; $x < $a; $x++) {
427 for ($y = 0; $y < $b; $y++) {
428 if ($command === 'BIPARTITE') {
429 $graph->join($x, $a + $y);
430 } else {
431 if ($y + 1 < $b) {
432 $graph->join($x * $b + $y, $x * $b + $y + 1);
433 }
434 if ($x + 1 < $a) {
435 $graph->join($x * $b + $y, ($x + 1) * $b + $y);
436 }
437 }
438 }
439 }
440 break;
441 case 'PETERSEN':
442 $graph = new Graph(10);
443 for ($i = 0; $i < 5; $i++) {
444 $graph->join($i, ($i + 1) % 5);
445 $graph->join($i, $i + 5);
446 $graph->join(5 + $i, 5 + ($i + 2) % 5);
447 }
448 break;
449 case 'HYPERCUBE':
450 $this->sizeCheck($number, 1, 4);
451 $graph = new Graph(1 << $number);
452 foreach ($graph->vertices() as $v) {
453 for ($bit = 0; $bit < $number; $bit++) {
454 $graph->join($v, $v ^ (1 << $bit));
455 }
456 }
457 break;
458 case 'COMPLEMENT':
459 $graph = new Graph($source->n);
460 for ($u = 0; $u < $source->n; $u++) {
461 for ($v = $u + 1; $v < $source->n; $v++) {
462 if (!$source->adj[$u][$v]) {
463 $graph->join($u, $v);
464 }
465 }
466 }
467 break;
468 case 'LINE':
469 $edges = $source->edges();
470 if (!$edges) {
471 throw new GraphError('no edges');
472 }
473 $this->sizeCheck(count($edges));
474 $graph = new Graph(count($edges));
475 for ($u = 0; $u < count($edges); $u++) {
476 for ($v = $u + 1; $v < count($edges); $v++) {
477 if (array_intersect($edges[$u], $edges[$v])) {
478 $graph->join($u, $v);
479 }
480 }
481 }
482 break;
483 case 'PRODUCT':
484 case 'UNION':
485 $other = $this->graphs[$args[2]];
486 $this->sizeCheck($command === 'PRODUCT' ? $source->n * $other->n : $source->n + $other->n);
487 $graph = new Graph($command === 'PRODUCT' ? $source->n * $other->n : $source->n + $other->n);
488 if ($command === 'PRODUCT') {
489 foreach ($source->vertices() as $x) {
490 foreach ($other->edges() as [$u, $v]) {
491 $graph->join($x * $other->n + $u, $x * $other->n + $v);
492 }
493 }
494 foreach ($other->vertices() as $y) {
495 foreach ($source->edges() as [$u, $v]) {
496 $graph->join($u * $other->n + $y, $v * $other->n + $y);
497 }
498 }
499 } else {
500 foreach ($source->edges() as [$u, $v]) {
501 $graph->join($u, $v);
502 }
503 foreach ($other->edges() as [$u, $v]) {
504 $graph->join($source->n + $u, $source->n + $v);
505 }
506 }
507 break;
508 case 'RELABEL':
509 case 'CANONIZE':
510 $labels = $command === 'CANONIZE' ? $source->canonical()['labels'] : array_map('intval', array_slice($args, 2));
511 $ordered = $labels;
512 sort($ordered, SORT_NUMERIC);
513 if ($ordered !== $source->vertices()) {
514 throw new GraphError('bad permutation');
515 }
516 $graph = new Graph($source->n);
517 foreach ($source->edges() as [$u, $v]) {
518 $graph->join($labels[$u], $labels[$v]);
519 }
520 break;
521 case 'INDUCED':
522 $selected = array_map('intval', array_slice($args, 2));
523 if (count(array_unique($selected)) !== count($selected) || max($selected) >= $source->n) {
524 throw new GraphError('bad vertices');
525 }
526 $graph = new Graph(count($selected));
527 for ($u = 0; $u < count($selected); $u++) {
528 for ($v = $u + 1; $v < count($selected); $v++) {
529 if ($source->adj[$selected[$u]][$selected[$v]]) {
530 $graph->join($u, $v);
531 }
532 }
533 }
534 break;
535 }
536 $verb = isset($this->graphs[$name]) ? 'redefined' : 'defined';
537 $this->graphs[$name] = $graph;
538 echo "$verb $name n {$graph->n} m ", count($graph->edges()), "\n";
539 }

561 {
562 $av = $a->invariants();
563 $bv = $b->invariants();
564 $tests = [['vertices', $a->n, $b->n], ['edges', count($a->edges()), count($b->edges())],
565 ['degrees', $av['degrees'], $bv['degrees']], ['triangles', $av['triangles'], $bv['triangles']],

568 foreach ($tests as [$kind, $x, $y]) {
569 if ($x !== $y) {
570 echo "not isomorphic $kind", $kind === 'degrees' ? '' : " $x $y", "\n";

591 private function automorphisms(string $name, Graph $graph): void
592 {
593 $canon = $graph->canonical();
594 $inverse = array_flip($canon['labels']);
595 $vertexParent = $graph->vertices();
596 $edgeParent = [];
597 $edgeIndex = [];
598 foreach ($graph->edges() as $i => [$u, $v]) {
599 $edgeParent[] = $i;
600 $edgeIndex["$u-$v"] = $i;
601 }
602 $find = function (array $parents, int $x): int {
603 while ($parents[$x] !== $x) {
604 $x = $parents[$x];
605 }
606 return $x;
607 };
608 $unite = function (array &$parents, int $x, int $y) use ($find): void {
609 $parents[$find($parents, $x)] = $find($parents, $y);
610 };
611 foreach ($canon['winners'] as $labels) {
612 $mapping = array_map(fn($label) => $inverse[$label], $labels);
613 foreach ($graph->vertices() as $v) {
614 $unite($vertexParent, $v, $mapping[$v]);
615 }
616 foreach ($graph->edges() as $i => [$u, $v]) {
617 $a = min($mapping[$u], $mapping[$v]);
618 $b = max($mapping[$u], $mapping[$v]);
619 $unite($edgeParent, $i, $edgeIndex["$a-$b"]);
620 }
621 }
622 $vertexOrbits = [];
623 foreach ($graph->vertices() as $v) {
624 $vertexOrbits[$find($vertexParent, $v)][] = $v;
625 }
626 $edgeOrbits = [];
627 foreach ($graph->edges() as $i => $_) {
628 $edgeOrbits[$find($edgeParent, $i)][] = $i;
629 }
630 usort($vertexOrbits, fn($a, $b) => $a[0] <=> $b[0]);
631 usort($edgeOrbits, fn($a, $b) => $a[0] <=> $b[0]);
632 echo "aut $name order ", count($canon['winners']), "\nvertex orbits ", count($vertexOrbits), "\n";
633 foreach ($vertexOrbits as $orbit) {
634 echo 'orbit ', implode(' ', $orbit), "\n";
635 }
636 echo 'edge orbits ', count($edgeOrbits), "\n";
637 foreach ($edgeOrbits as $orbit) {
638 echo 'orbit ', implode(' ', array_map(fn($i) => implode('-', $graph->edges()[$i]), $orbit)), "\n";
639 }
640 echo 'transitive vertex ', count($vertexOrbits) === 1 ? 'yes' : 'no',
641 ' edge ', count($edgeOrbits) === 1 ? 'yes' : 'no', "\n";
642 }

644 private function subiso(string $pn, string $tn, Graph $pattern, Graph $target): void
645 {
646 $count = 0;
647 $inducedCount = 0;
648 $examples = [];
649 $mapping = [];
650 $search = function (int $depth, bool $induced) use (&$search, &$count, &$inducedCount, &$examples, &$mapping, $pattern, $target): void {
651 if ($depth === $pattern->n) {
652 $count++;
653 if ($induced) {
654 $inducedCount++;
655 }
656 if (count($examples) < 5) {
657 $examples[] = [$mapping, $induced];
658 }
659 return;
660 }
661 foreach ($target->vertices() as $v) {
662 if (in_array($v, $mapping, true)) {
663 continue;
664 }
665 $stillInduced = $induced;
666 for ($u = 0; $u < $depth; $u++) {
667 if ($pattern->adj[$depth][$u] && !$target->adj[$v][$mapping[$u]]) {
668 continue 2;
669 }
670 if ($pattern->adj[$depth][$u] !== $target->adj[$v][$mapping[$u]]) {
671 $stillInduced = false;
672 }
673 }
674 $mapping[] = $v;
675 $search($depth + 1, $stillInduced);
676 array_pop($mapping);
677 }
678 };
679 if ($pattern->n <= $target->n) {
680 $search(0, true);
681 }
682 $order = count($pattern->canonical()['winners']);
683 echo "subiso $pn $tn embeddings $count induced $inducedCount\n";
684 echo 'copies ', intdiv($count, $order), ' induced ', intdiv($inducedCount, $order), "\n";
685 foreach ($examples as [$map, $induced]) {
686 echo 'embedding ', implode(' ', $map), $induced ? ' induced' : '', "\n";
687 }
688 }

690 private function classify(int $n): void
691 {
692 $this->sizeCheck($n, 1, 5);
693 if (!isset($this->classes[$n])) {
694 $edgePairs = [];
695 for ($u = 0; $u < $n; $u++) {
696 for ($v = $u + 1; $v < $n; $v++) {
697 $edgePairs[] = [$u, $v];
698 }
699 }
700 $levels = [[str_repeat('0', count($edgePairs))]];
701 for ($m = 0; $m < count($edgePairs); $m++) {
702 $next = [];
703 foreach ($levels[$m] as $string) {
704 foreach ($edgePairs as $extra => $_) {
705 if ($string[$extra] !== '0') {
706 continue;
707 }
708 $graph = new Graph($n);
709 foreach ($edgePairs as $i => [$u, $v]) {
710 if ($string[$i] === '1' || $i === $extra) {
711 $graph->join($u, $v);
712 }
713 }
714 $value = $graph->canonical()['best'];
715 $next['s' . $value] = $value;
716 }
717 }
718 $levels[] = array_values($next);
719 }
720 $this->classes[$n] = $levels;
721 }
722 $levels = $this->classes[$n];
723 echo "classify $n total ", array_sum(array_map('count', $levels)), "\n";
724 foreach ($levels as $m => $strings) {
725 sort($strings, SORT_STRING);
726 echo "edges $m count ", count($strings), ' ',
727 implode(' ', array_map(fn($s) => $s === '' ? '-' : $s, $strings)), "\n";
728 }
729 }

731 public function execute(string $line): void
732 {
733 $words = explode(' ', $line);
734 $command = array_shift($words);
735 $args = $words;
736 try {
737 $schema = $this->validate($command, $args);
738 if (($schema[2][0] ?? null) === 'name') {
739 $this->define($command, $args);
740 return;
741 }
742 $name = $args[0] ?? null;
743 $graph = $this->graphs[$name ?? ''] ?? null;
744 switch ($command) {
745 case 'DELETE':
746 unset($this->graphs[$name]);
747 echo "deleted $name\n";
748 break;
749 case 'LIST':
750 echo 'graphs ', count($this->graphs), "\n";
751 $names = array_keys($this->graphs);
752 sort($names, SORT_STRING);
753 foreach ($names as $key) {
754 echo "$key n {$this->graphs[$key]->n} m ", count($this->graphs[$key]->edges()), "\n";
755 }
756 break;
757 case 'SHOW':
758 echo "graph $name n {$graph->n} m ", count($graph->edges()), "\n";
759 foreach ($graph->vertices() as $v) {
760 $neighbors = $graph->neighbors($v);
761 echo "$v:", $neighbors ? ' ' . implode(' ', $neighbors) : '', "\n";
762 }
763 break;
764 case 'INVARIANTS':
765 $this->showInvariants($name, $graph);
766 break;
767 case 'REFINE':
768 $selected = array_map('intval', array_slice($args, 1));
769 foreach ($selected as $v) {
770 if ($v >= $graph->n) {
771 throw new GraphError('bad vertices');
772 }
773 }
774 [$colors, $rounds] = $graph->stable();
775 foreach ($selected as $v) {
776 if (count(array_keys($colors, $colors[$v], true)) === 1) {
777 throw new GraphError("singleton $v");
778 }
779 [$colors, $added] = $graph->refine($graph->individualize($colors, $v));
780 $rounds += $added;
781 }
782 echo "refine $name rounds $rounds cells ", count(array_unique($colors)), "\n";
783 for ($c = 0; $c < count(array_unique($colors)); $c++) {
784 echo "$c: ", implode(' ', array_keys($colors, $c, true)), "\n";
785 }
786 break;
787 case 'CANON':
788 $result = $graph->canonical();
789 echo "canon $name nodes {$result['nodes']} leaves {$result['leaves']}\n";
790 echo 'labels ', implode(' ', $result['labels']), "\n";
791 echo 'string ', $result['best'] === '' ? '-' : $result['best'], "\n";
792 break;
793 case 'ISO':
794 $this->isomorphism($graph, $this->graphs[$args[1]]);
795 break;
796 case 'AUT':
797 $this->automorphisms($name, $graph);
798 break;
799 case 'SUBISO':
800 $this->subiso($name, $args[1], $graph, $this->graphs[$args[1]]);
801 break;
802 case 'CLASSIFY':
803 $this->classify((int)$name);
804 break;
805 }
806 } catch (GraphError $error) {
807 echo 'error: ', $error->getMessage(), "\n";
808 }
809 }
810}
