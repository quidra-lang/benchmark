22 boolean edge(int u, int v) {
23 return (adjacent[u] & 1 << v) != 0;

25 void join(int u, int v) {
26 adjacent[u] |= 1 << v;

34 for (int u = 0; u < n; u++) for (int v = u + 1; v < n; v++) if (edge(u, v)) result.add(new int[]{u, v});

96 static Refined refine(Graph graph, int[] initial) {
97 int[] colors = initial.clone();
98 int rounds = 0;
99 while (true) {
100 int[][] signatures = new int[graph.n][];
101 for (int v = 0; v < graph.n; v++) {
102 int[] neighbors = new int[graph.degree(v)];
103 int at = 0;
104 for (int w = 0; w < graph.n; w++) if (graph.edge(v, w)) neighbors[at++] = colors[w];
105 Arrays.sort(neighbors);
106 signatures[v] = new int[neighbors.length + 1];
107 signatures[v][0] = colors[v];
108 System.arraycopy(neighbors, 0, signatures[v], 1, neighbors.length);
109 }
110 Integer[] order = new Integer[graph.n];
111 for (int i = 0; i < graph.n; i++) order[i] = i;
112 Arrays.sort(order, (a, b) -> compare(signatures[a], signatures[b]));
113 int[] next = new int[graph.n];
114 int color = 0;
115 for (int i = 0; i < graph.n; i++) {
116 if (i > 0 && compare(signatures[order[i - 1]], signatures[order[i]]) != 0) color++;
117 next[order[i]] = color;
118 }
119 if (cells(next) == cells(colors)) return new Refined(colors, rounds);
120 colors = next;
121 rounds++;
122 }
123 }

139 static void canonicalSearch(Graph graph, int[] colors, Canonical result) {
140 result.nodes++;
141 if (cells(colors) == graph.n) {
142 result.leaves++;
143 String text = leafString(graph, colors);
144 int order = result.string == null ? 1 : text.compareTo(result.string);
145 if (order > 0) {
146 result.string = text;
147 result.labels = colors.clone();
148 result.maxima.clear();
149 }
150 if (order >= 0) result.maxima.add(colors.clone());
151 return;
152 }
153 int chosen = -1, smallest = graph.n + 1;
154 int[] sizes = new int[cells(colors)];
155 for (int color : colors) sizes[color]++;
156 for (int color = 0; color < sizes.length; color++) if (sizes[color] >= 2 && sizes[color] < smallest) {
157 chosen = color;
158 smallest = sizes[color];
159 }
160 for (int vertex = 0; vertex < graph.n; vertex++) if (colors[vertex] == chosen) canonicalSearch(graph, refine(graph, individualize(colors, vertex)).colors(), result);

169 static Invariants invariants(Graph graph) {
170 int[] degrees = new int[graph.n];
171 for (int i = 0; i < graph.n; i++) degrees[i] = graph.degree(i);
172 Arrays.sort(degrees);
173 for (int i = 0; i < graph.n / 2; i++) {
174 int x = degrees[i];
175 degrees[i] = degrees[graph.n - 1 - i];
176 degrees[graph.n - 1 - i] = x;
177 }
178 int triangles = 0;
179 for (int a = 0; a < graph.n; a++) for (int b = a + 1; b < graph.n; b++) for (int c = b + 1; c < graph.n; c++) if (graph.edge(a, b) && graph.edge(a, c) && graph.edge(b, c)) triangles++;
180 int[][] distances = new int[graph.n][graph.n];
181 boolean[] visited = new boolean[graph.n];
182 List<Integer> sizes = new ArrayList<>();
183 int girth = Integer.MAX_VALUE;
184 boolean bipartite = true;
185 for (int source = 0; source < graph.n; source++) {
186 Arrays.fill(distances[source], -1);
187 int[] parents = new int[graph.n];
188 Arrays.fill(parents, -1);
189 ArrayDeque<Integer> queue = new ArrayDeque<>();
190 queue.add(source);
191 distances[source][source] = 0;
192 int size = 0;
193 while (!queue.isEmpty()) {
194 int u = queue.remove();
195 size++;
196 for (int v = 0; v < graph.n; v++) if (graph.edge(u, v)) {
197 if (distances[source][v] < 0) {
198 distances[source][v] = distances[source][u] + 1;
199 parents[v] = u;
200 queue.add(v);
201 } else if (parents[u] != v && parents[v] != u) girth = Math.min(girth, distances[source][u] + distances[source][v] + 1);
202 if (distances[source][v] % 2 == distances[source][u] % 2) bipartite = false;
203 }
204 }
205 if (!visited[source]) {
206 sizes.add(size);
207 for (int i = 0; i < graph.n; i++) if (distances[source][i] >= 0) visited[i] = true;
208 }
209 }
210 sizes.sort(Comparator.reverseOrder());
211 int diameter = 0, radius = graph.n;
212 if (sizes.size() == 1) for (int[] row : distances) {
213 int eccentricity = Arrays.stream(row).max().orElse(0);
214 diameter = Math.max(diameter, eccentricity);
215 radius = Math.min(radius, eccentricity);
216 }
217 return new Invariants(degrees, graph.edges().size(), triangles, sizes, girth == Integer.MAX_VALUE ? "none" : "" + girth, sizes.size() == 1 ? "" + diameter : "INF", sizes.size() == 1 ? "" + radius : "INF", bipartite ? "yes" : "no");
218 }

220 static boolean colorSearch(Graph graph, int count, int vertex, int largest, int[] colors) {
221 if (vertex == graph.n) return true;
222 for (int color = 0; color < count && color <= largest + 1; color++) {
223 boolean good = true;
224 for (int other = 0; other < vertex; other++) if (graph.edge(vertex, other) && colors[other] == color) good = false;
225 if (good) {
226 colors[vertex] = color;
227 if (colorSearch(graph, count, vertex + 1, Math.max(largest, color), colors)) return true;
228 }
229 }
230 return false;
231 }

264 static class DisjointSet {
265 final int[] parent;
266 DisjointSet(int n) {
267 parent = new int[n];
268 for (int i = 0; i < n; i++) parent[i] = i;
269 }
270 int find(int v) {
271 return parent[v] == v ? v : (parent[v] = find(parent[v]));
272 }
273 void join(int a, int b) {
274 a = find(a);
275 b = find(b);
276 if (a != b) parent[Math.max(a, b)] = Math.min(a, b);
277 }
278 List<List<Integer>> groups() {
279 TreeMap<Integer, List<Integer>> groups = new TreeMap<>();
280 for (int i = 0; i < parent.length; i++) groups.computeIfAbsent(find(i), key -> new ArrayList<>()).add(i);
281 return new ArrayList<>(groups.values());
282 }
283 }

285 static void automorphisms(String name, Graph graph) {
286 Canonical data = canonical(graph);
287 List<int[]> edges = graph.edges();
288 DisjointSet vertices = new DisjointSet(graph.n), edgeSets = new DisjointSet(edges.size());
289 int[] inverse = new int[graph.n];
290 for (int i = 0; i < graph.n; i++) inverse[data.labels[i]] = i;
291 int[][] edgeIndex = new int[graph.n][graph.n];
292 for (int i = 0; i < edges.size(); i++) {
293 int[] edge = edges.get(i);
294 edgeIndex[edge[0]][edge[1]] = edgeIndex[edge[1]][edge[0]] = i;
295 }
296 for (int[] leaf : data.maxima) {
297 int[] permutation = new int[graph.n];
298 for (int i = 0; i < graph.n; i++) {
299 permutation[i] = inverse[leaf[i]];
300 vertices.join(i, permutation[i]);
301 }
302 for (int i = 0; i < edges.size(); i++) edgeSets.join(i, edgeIndex[permutation[edges.get(i)[0]]][permutation[edges.get(i)[1]]]);
303 }
304 var vertexGroups = vertices.groups();
305 var edgeGroups = edgeSets.groups();
306 System.out.println("aut " + name + " order " + data.maxima.size());
307 System.out.println("vertex orbits " + vertexGroups.size());
308 for (var group : vertexGroups) System.out.println("orbit " + numbers(group));
309 System.out.println("edge orbits " + edgeGroups.size());
310 for (var group : edgeGroups) System.out.println("orbit " + String.join(" ", group.stream().map(i -> edges.get(i)[0] + "-" + edges.get(i)[1]).toList()));
311 System.out.println("transitive vertex " + (vertexGroups.size() == 1 ? "yes" : "no") + " edge " + (edgeGroups.size() == 1 ? "yes" : "no"));
312 }

326 for (int before = 0; before < vertex; before++) {

330 }

332 map[vertex] = candidate;
333 search(pattern, target, vertex + 1, used | 1 << candidate, map, induced);

339 static void isomorphism(String leftName, Graph left, String rightName, Graph right) {
340 if (left.n != right.n) {
341 System.out.println("not isomorphic vertices " + left.n + " " + right.n);
342 return;
343 }
344 Invariants a = invariants(left), b = invariants(right);
345 String reason = null;
346 if (a.edges() != b.edges()) reason = "edges " + a.edges() + " " + b.edges();
347 else if (!Arrays.equals(a.degrees(), b.degrees())) reason = "degrees";
348 else if (a.triangles() != b.triangles()) reason = "triangles " + a.triangles() + " " + b.triangles();
349 else if (a.sizes().size() != b.sizes().size()) reason = "components " + a.sizes().size() + " " + b.sizes().size();
350 else if (!a.girth().equals(b.girth())) reason = "girth " + a.girth() + " " + b.girth();
351 else if (!a.diameter().equals(b.diameter())) reason = "diameter " + a.diameter() + " " + b.diameter();
352 else if (!a.bipartite().equals(b.bipartite())) reason = "bipartite " + a.bipartite() + " " + b.bipartite();
353 if (reason != null) {
354 System.out.println("not isomorphic " + reason);
355 return;
356 }
357 Refined x = refine(left, degreeColors(left)), y = refine(right, degreeColors(right));
358 int[] xs = new int[cells(x.colors())], ys = new int[cells(y.colors())];
359 for (int color : x.colors()) xs[color]++;
360 for (int color : y.colors()) ys[color]++;
361 if (x.rounds() != y.rounds() || !Arrays.equals(xs, ys)) {
362 System.out.println("not isomorphic refinement");
363 return;
364 }
365 Canonical c = canonical(left), d = canonical(right);
366 if (!c.string.equals(d.string)) {
367 System.out.println("not isomorphic canonical");
368 return;
369 }
370 int[] inverse = new int[right.n], mapping = new int[left.n];
371 for (int i = 0; i < right.n; i++) inverse[d.labels[i]] = i;
372 for (int i = 0; i < left.n; i++) mapping[i] = inverse[c.labels[i]];
373 System.out.println("isomorphic\nmapping " + numbers(mapping));
374 }

410 String command = all[0];
411 require(DEFINITIONS.contains(command) || OTHER.contains(command), "unknown command " + command);
412 int minimum, maximum, names;
413 switch (command) {

418 minimum = maximum = 1;
419 names = 0;
420 break;

423 break;
424 case "CYCLE", "PATH", "COMPLETE", "STAR", "WHEEL", "HYPERCUBE":
425 minimum = maximum = 2;

431 break;

433 minimum = maximum = names = 2;
434 break;
435 case "PRODUCT", "UNION":
436 minimum = maximum = names = 3;
437 break;

439 minimum = 2;
440 maximum = Integer.MAX_VALUE;
441 names = 1;
442 break;
443 case "CIRCULANT":
444 minimum = 3;
445 maximum = Integer.MAX_VALUE;
446 names = 1;
447 break;

449 minimum = names = 2;
450 maximum = Integer.MAX_VALUE;
451 break;
452 case "INDUCED":
453 minimum = 3;
454 names = 2;
455 maximum = Integer.MAX_VALUE;
456 break;

459 maximum = Integer.MAX_VALUE;

462 for (int i = 0; i < names; i++) require(name(args[i]), "bad arguments");
463 for (int i = names; i < args.length; i++) require(command.equals("GRAPH") && i >= 2 ? args[i].matches("(?:0|[1-9][0-9]{0,2})-(?:0|[1-9][0-9]{0,2})") : number(args[i]), "bad arguments");
464 int firstReference = DEFINITIONS.contains(command) ? (Set.of("COMPLEMENT", "LINE", "PRODUCT", "UNION", "RELABEL", "INDUCED", "CANONIZE").contains(command) ? 1 : names) : 0;
465 if (!command.equals("LIST") && !command.equals("CLASSIFY")) for (int i = firstReference; i < names; i++) require(graphs.containsKey(args[i]), "no graph " + args[i]);
466 String name = args.length == 0 ? "" : args[0];
467 Graph graph = null;
468 int n = args.length > names ? integer(args[names]) : 0;
469 if (DEFINITIONS.contains(command)) {
470 switch (command) {
471 case "GRAPH", "CYCLE", "PATH", "COMPLETE", "CIRCULANT":
472 require(n >= (command.equals("CYCLE") || command.equals("CIRCULANT") ? 3 : 1), "bad value");
473 require(n <= 16, "too large");

482 if (command.equals("PATH") || command.equals("CYCLE")) {
483 for (int i = 1; i < n; i++) graph.join(i - 1, i);
484 if (command.equals("CYCLE")) graph.join(n - 1, 0);
485 }
486 if (command.equals("COMPLETE")) for (int u = 0; u < n; u++) for (int v = u + 1; v < n; v++) graph.join(u, v);
487 if (command.equals("CIRCULANT")) {
488 int previous = 0;
489 for (int i = 2; i < args.length; i++) {
490 int jump = integer(args[i]);
491 require(jump > previous && jump * 2 <= n, "bad value");
492 previous = jump;
493 for (int v = 0; v < n; v++) graph.join(v, (v + jump) % n);
494 }
495 }
496 break;
497 case "STAR", "WHEEL":
498 require(n >= (command.equals("WHEEL") ? 3 : 1), "bad value");
499 require(n < 16, "too large");
500 graph = new Graph(n + 1);
501 for (int i = 1; i <= n; i++) graph.join(0, i);
502 if (command.equals("WHEEL")) for (int i = 1; i <= n; i++) graph.join(i, i == n ? 1 : i + 1);
503 break;
504 case "BIPARTITE", "GRID":
505 {
506 int b = integer(args[2]);
507 require(n > 0 && b > 0, "bad value");
508 int size = command.equals("GRID") ? n * b : n + b;
509 require(size <= 16, "too large");
510 graph = new Graph(size);
511 if (command.equals("BIPARTITE")) for (int u = 0; u < n; u++) for (int v = 0; v < b; v++) graph.join(u, n + v);
512 else for (int x = 0; x < n; x++) for (int y = 0; y < b; y++) {
513 if (x + 1 < n) graph.join(x * b + y, (x + 1) * b + y);
514 if (y + 1 < b) graph.join(x * b + y, x * b + y + 1);
515 }
516 break;
517 }
518 case "PETERSEN":
519 graph = new Graph(10);
520 for (int i = 0; i < 5; i++) {
521 graph.join(i, (i + 1) % 5);
522 graph.join(i, i + 5);
523 graph.join(i + 5, 5 + (i + 2) % 5);
524 }
525 break;
526 case "HYPERCUBE":
527 require(n > 0, "bad value");
528 require(n <= 4, "too large");
529 graph = new Graph(1 << n);
530 for (int u = 0; u < graph.n; u++) for (int bit = 0; bit < n; bit++) graph.join(u, u ^ 1 << bit);
531 break;
532 case "COMPLEMENT":
533 {
534 Graph source = graphs.get(args[1]);
535 graph = new Graph(source.n);
536 for (int u = 0; u < source.n; u++) for (int v = u + 1; v < source.n; v++) if (!source.edge(u, v)) graph.join(u, v);
537 break;
538 }
539 case "LINE":
540 {
541 List<int[]> edges = graphs.get(args[1]).edges();
542 require(!edges.isEmpty(), "no edges");
543 require(edges.size() <= 16, "too large");
544 graph = new Graph(edges.size());
545 for (int i = 0; i < edges.size(); i++) for (int j = i + 1; j < edges.size(); j++) if (edges.get(i)[0] == edges.get(j)[0] || edges.get(i)[0] == edges.get(j)[1] || edges.get(i)[1] == edges.get(j)[0] || edges.get(i)[1] == edges.get(j)[1]) graph.join(i, j);
546 break;
547 }
548 case "PRODUCT", "UNION":
549 {
550 Graph a = graphs.get(args[1]), b = graphs.get(args[2]);
551 boolean product = command.equals("PRODUCT");
552 int size = product ? a.n * b.n : a.n + b.n;
553 require(size <= 16, "too large");
554 graph = new Graph(size);
555 if (product) {
556 for (int x = 0; x < a.n; x++) for (int[] edge : b.edges()) graph.join(x * b.n + edge[0], x * b.n + edge[1]);
557 for (int y = 0; y < b.n; y++) for (int[] edge : a.edges()) graph.join(edge[0] * b.n + y, edge[1] * b.n + y);
558 } else {
559 for (int[] edge : a.edges()) graph.join(edge[0], edge[1]);
560 for (int[] edge : b.edges()) graph.join(a.n + edge[0], a.n + edge[1]);
561 }
562 break;
563 }
564 case "RELABEL", "INDUCED", "CANONIZE":
565 {
566 Graph source = graphs.get(args[1]);
567 int[] vertices = command.equals("CANONIZE") ? canonical(source).labels : Arrays.stream(args).skip(2).mapToInt(Main::integer).toArray();
568 if (command.equals("RELABEL")) {
569 require(vertices.length == source.n && Arrays.stream(vertices).distinct().count() == source.n && Arrays.stream(vertices).allMatch(v -> v < source.n), "bad permutation");
570 } else if (command.equals("INDUCED")) require(Arrays.stream(vertices).distinct().count() == vertices.length && Arrays.stream(vertices).allMatch(v -> v < source.n), "bad vertices");
571 graph = new Graph(command.equals("INDUCED") ? vertices.length : source.n);
572 if (command.equals("INDUCED")) {
573 for (int u = 0; u < vertices.length; u++) for (int v = u + 1; v < vertices.length; v++) if (source.edge(vertices[u], vertices[v])) graph.join(u, v);
574 } else for (int[] edge : source.edges()) graph.join(vertices[edge[0]], vertices[edge[1]]);
575 break;
576 }
577 }
578 boolean existed = graphs.containsKey(name);
579 graphs.put(name, graph);
580 System.out.println((existed ? "redefined " : "defined ") + name + " n " + graph.n + " m " + graph.edges().size());
581 return;
582 }
583 graph = graphs.get(name);
584 switch (command) {
585 case "DELETE":
586 graphs.remove(name);
587 System.out.println("deleted " + name);
588 break;
589 case "LIST":
590 System.out.println("graphs " + graphs.size());
591 graphs.forEach((key, value) -> System.out.println(key + " n " + value.n + " m " + value.edges().size()));
592 break;
593 case "SHOW":
594 System.out.println("graph " + name + " n " + graph.n + " m " + graph.edges().size());
595 for (int v = 0; v < graph.n; v++) {
596 StringBuilder row = new StringBuilder(v + ":");
597 for (int w = 0; w < graph.n; w++) if (graph.edge(v, w)) row.append(' ').append(w);
598 System.out.println(row);
599 }
600 break;
601 case "INVARIANTS":
602 printInvariants(name, graph);
603 break;
604 case "REFINE":
605 {
606 int[] vertices = Arrays.stream(args).skip(1).mapToInt(Main::integer).toArray();
607 Graph source = graph;
608 require(Arrays.stream(vertices).allMatch(v -> v < source.n), "bad vertices");
609 Refined result = refine(graph, degreeColors(graph));
610 int rounds = result.rounds();
611 int[] colors = result.colors();
612 for (int vertex : vertices) {
613 int size = 0;
614 for (int color : colors) if (color == colors[vertex]) size++;
615 require(size > 1, "singleton " + vertex);
616 result = refine(graph, individualize(colors, vertex));
617 colors = result.colors();
618 rounds += result.rounds();
619 }
620 System.out.println("refine " + name + " rounds " + rounds + " cells " + cells(colors));
621 for (int color = 0; color < cells(colors); color++) {
622 StringBuilder row = new StringBuilder(color + ":");
623 for (int v = 0; v < graph.n; v++) if (colors[v] == color) row.append(' ').append(v);
624 System.out.println(row);
625 }
626 break;
627 }
628 case "CANON":
629 {
630 Canonical result = canonical(graph);
631 System.out.println("canon " + name + " nodes " + result.nodes + " leaves " + result.leaves);
632 System.out.println("labels " + numbers(result.labels));
633 System.out.println("string " + (result.string.isEmpty() ? "-" : result.string));
634 break;
635 }
636 case "AUT":
637 automorphisms(name, graph);
638 break;
639 case "ISO":
640 isomorphism(name, graph, args[1], graphs.get(args[1]));
641 break;
642 case "SUBISO":
643 {
644 Graph destination = graphs.get(args[1]);
645 Embeddings result = new Embeddings();
646 if (graph.n <= destination.n) result.search(graph, destination, 0, 0, new int[graph.n], true);
647 int order = canonical(graph).maxima.size();
648 System.out.println("subiso " + name + " " + args[1] + " embeddings " + result.count + " induced " + result.induced);
649 System.out.println("copies " + result.count / order + " induced " + result.induced / order);
650 for (Embedding embedding : result.first) System.out.println("embedding " + numbers(embedding.map()) + (embedding.induced() ? " induced" : ""));
651 break;
652 }
653 case "CLASSIFY":
654 classify(integer(name));
655 break;
656 }
657 }
