1struct CommandError <: Exception
2 message::String
3end
4fail(s) = throw(CommandError(s))
5check(b, s) = b ? nothing : fail(s)
6yesno(b) = b ? "yes" : "no"
7numbers(xs) = join(xs, ' ')
8struct Graph
9 adj::BitMatrix
10end
11Graph(n::Int) = Graph(falses(n, n))
12vertices(g) = size(g.adj, 1)
13graphedges(g) = [(u, v) for u in 1:vertices(g) for v in (u+1):vertices(g) if g.adj[u, v]]
14edgecount(g) = div(count(g.adj), 2)
15function addedge!(g, u, v)
16 g.adj[u, v]=true
17 g.adj[v, u]=true
18end
19function sizecheck(n, minimum = 1, maximum = 16)
20 check(n>=minimum, "bad value")
21 check(n<=maximum, "too large")
22end
23function distances(g, start)
24 d=fill(-1, vertices(g))
25 d[start]=0
26 queue=[start]
27 at=1
28 while at<=length(queue)
29 u=queue[at]
30 at+=1
31 for v in 1:vertices(g)
32 if g.adj[u, v] && d[v]<0
33 d[v]=d[u]+1
34 push!(queue, v)
35 end
36 end
37 end
38 d
39end
40function graphinfo(g)
41 n=vertices(g)
42 seen=falses(n)
43 sizes=Int[]
44 bipartite=true
45 girth=typemax(Int)
46 eccentricities=Int[]
47 for root in 1:n
48 d=distances(g, root)
49 push!(eccentricities, maximum(d))
50 if !seen[root]
51 component=findall(>=(0), d)
52 seen[component].=true
53 push!(sizes, length(component))
54 end
55 for (u, v) in graphedges(g)
56 d[u]>=0 && d[v]>=0 && iseven(d[u]-d[v]) && (bipartite=false)
57 end
58 depth=fill(-1, n)
59 parents=zeros(Int, n)
60 depth[root]=0
61 queue=[root]
62 at=1
63 while at<=length(queue)
64 u=queue[at]
65 at+=1
66 for v in 1:n
67 g.adj[u, v] || continue
68 if depth[v]<0
69 depth[v]=depth[u]+1
70 parents[v]=u
71 push!(queue, v)
72 elseif v!=parents[u]
73 girth=min(girth, depth[u]+depth[v]+1)
74 end
75 end
76 end
77 end
78 triangles=sum(g.adj[u, w] && g.adj[v, w] for (u, v) in graphedges(g) for w in (v+1):n; init = 0)
79 degrees=sort!(vec(sum(g.adj; dims = 2)); rev = true)
80 sort!(sizes; rev = true)
81 (
82 n = n,
83 m = edgecount(g),
84 degrees = degrees,
85 triangles = triangles,
86 sizes = sizes,
87 girth = girth==typemax(Int) ? "none" : string(girth),
88 diameter = length(sizes)==1 ? string(maximum(eccentricities)) : "INF",
89 radius = length(sizes)==1 ? string(minimum(eccentricities)) : "INF",
90 bipartite = yesno(bipartite),
91 )
92end
93function maximumclique(g)
94 best=Int[]
95 function extend(clique, candidates)
96 length(clique)>length(best) && (best=clique)
97 for v in candidates
98 extend([clique; v], [w for w in candidates if w>v && g.adj[v, w]])
99 end
100 end
101 extend(Int[], collect(1:vertices(g)))
102 best
103end
104function colouring(g, limit)
105 colors=Int[]
106 function extend(used)
107 v=length(colors)+1
108 v>vertices(g) && return true
109 for color in 0:min(used, limit-1)
110 any(w->g.adj[v, w] && colors[w]==color, 1:(v-1)) && continue
111 push!(colors, color)
112 extend(max(used, color+1)) && return true
113 pop!(colors)
114 end
115 false
116 end
117 extend(0) ? colors : nothing
118end
119function rankkeys(keys)
120 ordered=sort!(unique(keys))
121 positions=Dict(key=>i-1 for (i, key) in enumerate(ordered))
122 [positions[key] for key in keys]
123end
124degreecolors(g) = rankkeys(vec(sum(g.adj; dims = 2)))
125function refine(g, colors)
126 rounds=0
127 while true
128 signatures=[
129 (colors[v], Tuple(sort!([colors[w] for w in 1:vertices(g) if g.adj[v, w]]))) for
130 v in 1:vertices(g)
131 ]
132 next=rankkeys(signatures)
133 maximum(next)==maximum(colors) && return colors, rounds
134 colors=next
135 rounds+=1
136 end
137end
138stable(g) = refine(g, degreecolors(g))
139cells(colors) = [findall(==(c), colors) for c in 0:maximum(colors)]
140individualize(colors, v) = [c<colors[v] || w==v ? c : c+1 for (w, c) in enumerate(colors)]
141function leafstring(g, labels)
142 inverse=sortperm(labels)
143 join(
144 g.adj[inverse[i], inverse[j]] ? '1' : '0' for i in 1:vertices(g) for j in (i+1):vertices(g)
145 )
146end
147struct Canonical
148 nodes::Int
149 leaves::Int
150 text::String
151 labels::Vector{Int}
152 best::Vector{Vector{Int}}
153end
154function canonical(g)
155 nodes=0
156 leaves=0
157 text=nothing
158 labels=Int[]
159 best=Vector{Int}[]
160 function visit(colors)
161 nodes+=1
162 if maximum(colors)==vertices(g)-1
163 leaves+=1
164 s=leafstring(g, colors)
165 if text===nothing || s>text
166 text=s
167 labels=colors
168 best=[colors]
169 elseif s==text
170 push!(best, colors)
171 end
172 return
173 end
174 choices=[cell for cell in cells(colors) if length(cell)>1]
175 chosen=choices[argmin(length.(choices))]
176 for v in chosen
177 next, _=refine(g, individualize(colors, v))
178 visit(next)
179 end
180 end
181 colors, _=stable(g)
182 visit(colors)
183 Canonical(nodes, leaves, text, labels, best)
184end
185function invariantreport(g, name)
186 info=graphinfo(g)
187 println("invariants $name\ndegrees ", numbers(info.degrees))
188 println("edges $(info.m) triangles $(info.triangles)")
189 println("components $(length(info.sizes)) sizes ", numbers(info.sizes))
190 println(
191 "girth $(info.girth) diameter $(info.diameter) radius $(info.radius)\nbipartite $(info.bipartite)",
192 )
193 if vertices(g)>=13
194 println("clique -\nchromatic -")
195 return
196 end
197 clique=maximumclique(g)
198 println("clique $(length(clique)) ", numbers(clique .- 1))
199 for limit in length(clique):vertices(g)
200 colors=colouring(g, limit)
201 if colors!==nothing
202 println("chromatic $limit ", numbers(colors))
203 return
204 end
205 end
206end
207function isoreport(a, b)
208 x=graphinfo(a)
209 y=graphinfo(b)
210 tests=(
211 ("vertices", x.n, y.n, true),
212 ("edges", x.m, y.m, true),
213 ("degrees", x.degrees, y.degrees, false),
214 ("triangles", x.triangles, y.triangles, true),
215 ("components", length(x.sizes), length(y.sizes), true),
216 ("girth", x.girth, y.girth, true),
217 ("diameter", x.diameter, y.diameter, true),
218 ("bipartite", x.bipartite, y.bipartite, true),
219 )
220 for (name, left, right, shown) in tests
221 if left!=right
222 println("not isomorphic $name", shown ? " $left $right" : "")
223 return
224 end
225 end
226 ac, ar=stable(a)
227 bc, br=stable(b)
228 if ar!=br || length.(cells(ac))!=length.(cells(bc))
229 println("not isomorphic refinement")
230 return
231 end
232 af=canonical(a)
233 bf=canonical(b)
234 if af.text!=bf.text
235 println("not isomorphic canonical")
236 return
237 end
238 inverse=sortperm(bf.labels)
239 println("isomorphic\nmapping ", numbers([inverse[label+1]-1 for label in af.labels]))
240end
241function automorphisms(g)
242 form=canonical(g)
243 inverse=sortperm(form.labels)
244 [[inverse[label+1] for label in leaf] for leaf in form.best]
245end
246function automorphismreport(g, name)
247 permutations=automorphisms(g)
248 seen=Set{Int}()
249 vorbits=Vector{Int}[]
250 for v in 1:vertices(g)
251 v in seen && continue
252 orbit=sort!(unique([p[v] for p in permutations]))
253 union!(seen, orbit)
254 push!(vorbits, orbit)
255 end
256 eseen=Set{Tuple{Int,Int}}()
257 eorbits=Vector{Tuple{Int,Int}}[]
258 for (u, v) in graphedges(g)
259 (u, v) in eseen && continue
260 orbit=sort!(unique([minmax(p[u], p[v]) for p in permutations]))
261 union!(eseen, orbit)
262 push!(eorbits, orbit)
263 end
264 println("aut $name order $(length(permutations))\nvertex orbits $(length(vorbits))")
265 for orbit in vorbits
266 println("orbit ", numbers(orbit .- 1))
267 end
268 println("edge orbits $(length(eorbits))")
269 for orbit in eorbits
270 println("orbit ", join(["$(u-1)-$(v-1)" for (u, v) in orbit], ' '))
271 end
272 println("transitive vertex $(yesno(length(vorbits)==1)) edge $(yesno(length(eorbits)==1))")
273end
274function subisoreport(pattern, target, pname, tname)
275 mapping=Int[]
276 total=0
277 induced=0
278 shown=Tuple{Vector{Int},Bool}[]
279 function extend()
280 v=length(mapping)+1
281 if v>vertices(pattern)
282 exact=all(
283 pattern.adj[u, w]==target.adj[mapping[u], mapping[w]] for u in 1:vertices(pattern)
284 for w in (u+1):vertices(pattern)
285 )
286 total+=1
287 induced+=exact
288 length(shown)<5 && push!(shown, (copy(mapping), exact))
289 return
290 end
291 for w in 1:vertices(target)
292 w in mapping && continue
293 all(u->!pattern.adj[v, u] || target.adj[w, mapping[u]], 1:(v-1)) || continue
294 push!(mapping, w)
295 extend()
296 pop!(mapping)
297 end
298 end
299 extend()
300 order=length(canonical(pattern).best)
301 println(
302 "subiso $pname $tname embeddings $total induced $induced\ncopies $(div(total,order)) induced $(div(induced,order))",
303 )
304 for (image, exact) in shown
305 println("embedding ", numbers(image .- 1), exact ? " induced" : "")
306 end
307end
308function classify(n)
309 sizecheck(n, 1, 5)
310 pairs=[(u, v) for u in 1:n for v in (u+1):n]
311 layers=[Set([canonical(Graph(n)).text])]
312 for _ in eachindex(pairs)
313 next=Set{String}()
314 for text in layers[end]
315 g=Graph(n)
316 for (i, (u, v)) in enumerate(pairs)
317 text[i]=='1' && addedge!(g, u, v)
318 end
319 for (i, (u, v)) in enumerate(pairs)
320 if text[i]=='0'
321 changed=Graph(copy(g.adj))
322 addedge!(changed, u, v)
323 push!(next, canonical(changed).text)
324 end
325 end
326 end
327 push!(layers, next)
328 end
329 println("classify $n total $(sum(length,layers))")
330 for (m, layer) in enumerate(layers)
331 strings=sort!(collect(layer))
332 println(
333 "edges $(m-1) count $(length(layer)) ",
334 join([isempty(s) ? "-" : s for s in strings], ' '),
335 )
336 end
337end
338const SCHEMAS=Dict(
339 "GRAPH"=>["name", "number", "edge*"],
340 "CYCLE"=>["name", "number"],
341 "PATH"=>["name", "number"],
342 "COMPLETE"=>["name", "number"],
343 "BIPARTITE"=>["name", "number", "number"],
344 "STAR"=>["name", "number"],
345 "WHEEL"=>["name", "number"],
346 "PETERSEN"=>["name"],
347 "HYPERCUBE"=>["name", "number"],
348 "CIRCULANT"=>["name", "number", "number+"],
349 "GRID"=>["name", "number", "number"],
350 "COMPLEMENT"=>["name", "graph"],
351 "LINE"=>["name", "graph"],
352 "PRODUCT"=>["name", "graph", "graph"],
353 "UNION"=>["name", "graph", "graph"],
354 "RELABEL"=>["name", "graph", "number*"],
355 "INDUCED"=>["name", "graph", "number+"],
356 "CANONIZE"=>["name", "graph"],
357 "DELETE"=>["graph"],
358 "LIST"=>String[],
359 "SHOW"=>["graph"],
360 "INVARIANTS"=>["graph"],
361 "REFINE"=>["graph", "number*"],
362 "CANON"=>["graph"],
363 "ISO"=>["graph", "graph"],
364 "AUT"=>["graph"],
365 "SUBISO"=>["graph", "graph"],
366 "CLASSIFY"=>["number"],
367)
368function arguments(command, words, registry)
369 check(haskey(SCHEMAS, command), "unknown command $command")
370 schema=SCHEMAS[command]
371 repeat=!isempty(schema) && last(schema)[end] in ('*', '+')
372 fixed=length(schema)-repeat
373 check(
374 repeat ? length(words)>=fixed+(last(schema)[end]=='+') : length(words)==fixed,
375 "bad arguments",
376 )
377 for (i, word) in enumerate(words)
378 kind=i<=fixed ? schema[i] : chop(last(schema))
379 valid=kind in ("name", "graph") ? occursin(r"\A[a-z][a-z0-9]{0,7}\z", word) :
380 kind=="number" ? occursin(r"\A(?:0|[1-9][0-9]{0,2})\z", word) :
381 occursin(r"\A(?:0|[1-9][0-9]{0,2})-(?:0|[1-9][0-9]{0,2})\z", word)
382 check(valid, "bad arguments")
383 end
384 for i in 1:fixed
385 schema[i]=="graph" && check(haskey(registry, words[i]), "no graph $(words[i])")
386 end
387 nothing
388end
389function buildgraph(command, args, registry)
390 if command in ("GRAPH", "CYCLE", "PATH", "COMPLETE", "CIRCULANT")
391 n=parse(Int, args[2])
392 sizecheck(n, command in ("CYCLE", "CIRCULANT") ? 3 : 1)
393 g=Graph(n)
394 if command=="GRAPH"
395 for word in args[3:end]
396 u, v=parse.(Int, split(word, '-'))
397 check(u<n && v<n && u!=v, "bad edge $word")
398 check(!g.adj[u+1, v+1], "duplicate edge $word")
399 addedge!(g, u+1, v+1)
400 end
401 elseif command=="CYCLE"
402 for u in 1:n
403 addedge!(g, u, mod1(u+1, n))
404 end
405 elseif command=="PATH"
406 for u in 1:(n-1)
407 addedge!(g, u, u+1)
408 end
409 elseif command=="COMPLETE"
410 for u in 1:n, v in (u+1):n
411 addedge!(g, u, v)
412 end
413 else
414 jumps=parse.(Int, args[3:end])
415 check(
416 all(j->j>0 && 2j<=n, jumps) && all(i->jumps[i]<jumps[i+1], 1:(length(jumps)-1)),
417 "bad value",
418 )
419 for u in 1:n, j in jumps
420 addedge!(g, u, mod1(u+j, n))
421 end
422 end
423 return g
424 elseif command in ("BIPARTITE", "GRID")
425 a, b=parse.(Int, args[2:3])
426 check(a>0 && b>0, "bad value")
427 n=command=="GRID" ? a*b : a+b
428 sizecheck(n)
429 g=Graph(n)
430 if command=="BIPARTITE"
431 for u in 1:a, v in (a+1):(a+b)
432 addedge!(g, u, v)
433 end
434 else
435 for x in 0:(a-1), y in 0:(b-1)
436 u=x*b+y+1
437 y+1<b && addedge!(g, u, u+1)
438 x+1<a && addedge!(g, u, u+b)
439 end
440 end
441 return g
442 elseif command in ("STAR", "WHEEL")
443 k=parse(Int, args[2])
444 sizecheck(k, command=="WHEEL" ? 3 : 1, 15)
445 g=Graph(k+1)
446 for v in 2:(k+1)
447 addedge!(g, 1, v)
448 end
449 if command=="WHEEL"
450 for u in 1:k
451 addedge!(g, u+1, mod(u, k)+2)
452 end
453 end
454 return g
455 elseif command=="PETERSEN"
456 g=Graph(10)
457 for i in 0:4
458 addedge!(g, i+1, mod(i+1, 5)+1)
459 addedge!(g, i+1, i+6)
460 addedge!(g, i+6, mod(i+2, 5)+6)
461 end
462 return g
463 elseif command=="HYPERCUBE"
464 d=parse(Int, args[2])
465 sizecheck(d, 1, 4)
466 n=2^d
467 g=Graph(n)
468 for v in 0:(n-1), bit in 0:(d-1)
469 addedge!(g, v+1, xor(v, 1<<bit)+1)
470 end
471 return g
472 end
473 source=registry[args[2]]
474 n=vertices(source)
475 if command=="COMPLEMENT"
476 g=Graph(n)
477 for u in 1:n, v in (u+1):n
478 !source.adj[u, v] && addedge!(g, u, v)
479 end
480 return g
481 elseif command=="LINE"
482 edges=graphedges(source)
483 check(!isempty(edges), "no edges")
484 sizecheck(length(edges))
485 g=Graph(length(edges))
486 for i in eachindex(edges), j in (i+1):length(edges)
487 any(x->x in edges[j], edges[i]) && addedge!(g, i, j)
488 end
489 return g
490 elseif command in ("PRODUCT", "UNION")
491 other=registry[args[3]]
492 m=vertices(other)
493 size=command=="PRODUCT" ? n*m : n+m
494 sizecheck(size)
495 g=Graph(size)
496 if command=="PRODUCT"
497 for (u, v) in graphedges(source), y in 1:m
498 addedge!(g, (u-1)*m+y, (v-1)*m+y)
499 end
500 for (u, v) in graphedges(other), x in 0:(n-1)
501 addedge!(g, x*m+u, x*m+v)
502 end
503 else
504 for (u, v) in graphedges(source)
505 addedge!(g, u, v)
506 end
507 for (u, v) in graphedges(other)
508 addedge!(g, u+n, v+n)
509 end
510 end
511 return g
512 elseif command in ("RELABEL", "CANONIZE")
513 permutation=command=="CANONIZE" ? canonical(source).labels : parse.(Int, args[3:end])
514 check(sort(permutation)==collect(0:(n-1)), "bad permutation")
515 g=Graph(n)
516 for (u, v) in graphedges(source)
517 addedge!(g, permutation[u]+1, permutation[v]+1)
518 end
519 return g
520 else
521 selected=parse.(Int, args[3:end])
522 check(all(v->v<n, selected) && length(unique(selected))==length(selected), "bad vertices")
523 g=Graph(length(selected))
524 for i in eachindex(selected), j in (i+1):length(selected)
525 source.adj[selected[i]+1, selected[j]+1] && addedge!(g, i, j)
526 end
527 return g
528 end
529end
530function execute!(registry, line)
531 words=String.(split(line, ' '))
532 command=words[1]
533 args=words[2:end]
534 arguments(command, args, registry)
535 schema=SCHEMAS[command]
536 if !isempty(schema) && schema[1]=="name"
537 g=buildgraph(command, args, registry)
538 name=args[1]
539 status=haskey(registry, name) ? "redefined" : "defined"
540 registry[name]=g
541 println("$status $name n $(vertices(g)) m $(edgecount(g))")
542 return
543 end
544 if command=="LIST"
545 println("graphs $(length(registry))")
546 for name in sort!(collect(keys(registry)))
547 g=registry[name]
548 println("$name n $(vertices(g)) m $(edgecount(g))")
549 end
550 elseif command=="CLASSIFY"
551 classify(parse(Int, args[1]))
552 elseif command=="DELETE"
553 delete!(registry, args[1])
554 println("deleted $(args[1])")
555 else
556 name=args[1]
557 g=registry[name]
558 if command=="SHOW"
559 println("graph $name n $(vertices(g)) m $(edgecount(g))")
560 for v in 1:vertices(g)
561 neighbors=[w-1 for w in 1:vertices(g) if g.adj[v, w]]
562 println("$(v-1):", isempty(neighbors) ? "" : " "*numbers(neighbors))
563 end
564 elseif command=="INVARIANTS"
565 invariantreport(g, name)
566 elseif command=="CANON"
567 form=canonical(g)
568 println(
569 "canon $name nodes $(form.nodes) leaves $(form.leaves)\nlabels ",
570 numbers(form.labels),
571 )
572 println("string ", isempty(form.text) ? "-" : form.text)
573 elseif command=="REFINE"
574 selected=parse.(Int, args[2:end])
575 check(all(v->v<vertices(g), selected), "bad vertices")
576 colors, rounds=stable(g)
577 for v in selected
578 check(count(==(colors[v+1]), colors)>1, "singleton $v")
579 colors, extra=refine(g, individualize(colors, v+1))
580 rounds+=extra
581 end
582 groups=cells(colors)
583 println("refine $name rounds $rounds cells $(length(groups))")
584 for (c, cell) in enumerate(groups)
585 println("$(c-1): ", numbers(cell .- 1))
586 end
587 elseif command=="ISO"
588 isoreport(g, registry[args[2]])
589 elseif command=="AUT"
590 automorphismreport(g, name)
591 elseif command=="SUBISO"
592 subisoreport(g, registry[args[2]], name, args[2])
593 end
594 end
595end
596function main()
597 registry=Dict{String,Graph}()
598 n=parse(Int, readline())
599 for _ in 1:n
600 line=readline()
601 try
602 execute!(registry, line)
603 catch e
604 e isa CommandError || rethrow()
605 println("error: $(e.message)")
606 end
607 end
608end
