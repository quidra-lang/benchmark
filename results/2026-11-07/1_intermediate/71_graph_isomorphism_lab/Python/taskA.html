27class Graph:
28 def __init__(self, n, edges=()):
29 self.n = n
30 self.adj = [set() for _ in range(n)]
31 for u, v in edges:
32 self.add_edge(u, v)
33
34 def add_edge(self, u, v):
35 self.adj[u].add(v)
36 self.adj[v].add(u)
37
38 def has_edge(self, u, v):
39 return v in self.adj[u]
40
41 def edges(self):
42 return [(u, v) for u in range(self.n) for v in sorted(self.adj[u]) if u < v]
43
44 def edge_count(self):
45 return sum(map(len, self.adj)) // 2

71def custom_graph(n, edge_words):
72 check_range(n, 1, MAX_VERTICES)
73 graph = Graph(n)
74 for word in edge_words:
75 u, v = map(int, word.split("-"))
76 if u >= n or v >= n or u == v:
77 raise CommandError(f"bad edge {word}")
78 if graph.has_edge(u, v):
79 raise CommandError(f"duplicate edge {word}")
80 graph.add_edge(u, v)
81 return graph

138def grid_graph(rows, columns):
139 if rows == 0 or columns == 0:
140 raise CommandError("bad value")
141 check_size(rows * columns)
142 edges = []
143 for x in range(rows):
144 for y in range(columns):
145 vertex = x * columns + y
146 if y + 1 < columns:
147 edges.append((vertex, vertex + 1))
148 if x + 1 < rows:
149 edges.append((vertex, vertex + columns))
150 return Graph(rows * columns, edges)

167def cartesian_product(first, second):
168 check_size(first.n * second.n)
169 width = second.n
170 edges = [(x * width + y, x * width + z) for y, z in second.edges() for x in range(first.n)]
171 edges += [(x * width + y, w * width + y) for x, w in first.edges() for y in range(width)]
172 return Graph(first.n * width, edges)

200def distances_from(graph, root):
201 distance = [None] * graph.n
202 distance[root] = 0
203 queue = deque([root])
204 while queue:
205 u = queue.popleft()
206 for w in graph.adj[u]:
207 if distance[w] is None:
208 distance[w] = distance[u] + 1
209 queue.append(w)
210 return distance

213def components(graph):
214 seen = set()
215 result = []
216 for root in range(graph.n):
217 if root not in seen:
218 distance = distances_from(graph, root)
219 component = [v for v in range(graph.n) if distance[v] is not None]
220 seen.update(component)
221 result.append(component)
222 return result

225def is_bipartite(graph):
226 for component in components(graph):
227 depth = distances_from(graph, component[0])
228 if any(depth[u] == depth[w] for u in component for w in graph.adj[u]):
229 return False
230 return True

233def girth(graph):
234 lengths = []
235 for root in range(graph.n):
236 distance = [None] * graph.n
237 parent = [None] * graph.n
238 distance[root] = 0
239 queue = deque([root])
240 while queue:
241 u = queue.popleft()
242 for w in graph.adj[u]:
243 if distance[w] is None:
244 distance[w] = distance[u] + 1
245 parent[w] = u
246 queue.append(w)
247 elif w != parent[u]:
248 lengths.append(distance[u] + distance[w] + 1)
249 return min(lengths, default=None)

252def count_triangles(graph):
253 return sum(1 for u, v in graph.edges() for w in graph.adj[u] & graph.adj[v] if w > v)

256def maximum_clique(graph):
257 best = []
258
259 def extend(clique, candidates):
260 nonlocal best
261 if len(clique) > len(best):
262 best = clique
263 for v in candidates:
264 extend(clique + [v], [w for w in candidates if w > v and graph.has_edge(v, w)])
265
266 extend([], list(range(graph.n)))
267 return best

270def colouring_with(graph, limit):
271 colours = []
272
273 def extend(used):
274 v = len(colours)
275 if v == graph.n:
276 return True
277 for colour in range(min(used + 1, limit)):
278 if all(colours[w] != colour for w in graph.adj[v] if w < v):
279 colours.append(colour)
280 if extend(max(used, colour + 1)):
281 return True
282 colours.pop()
283 return False
284
285 return colours if extend(0) else None

288def chromatic_colouring(graph, clique_size):
289 for limit in count(clique_size):
290 colours = colouring_with(graph, limit)

295class Invariants:
296 def __init__(self, graph):
297 self.vertices = graph.n
298 self.edges = graph.edge_count()
299 self.degrees = sorted(map(len, graph.adj), reverse=True)
300 self.triangles = count_triangles(graph)
301 self.sizes = sorted(map(len, components(graph)), reverse=True)
302 length = girth(graph)
303 self.girth = "none" if length is None else length
304 self.bipartite = yes_no(is_bipartite(graph))
305 if len(self.sizes) == 1:
306 eccentricities = [max(distances_from(graph, v)) for v in range(graph.n)]
307 self.diameter = max(eccentricities)
308 self.radius = min(eccentricities)
309 else:
310 self.diameter = self.radius = "INF"

313def rank(keys):
314 position = {key: index for index, key in enumerate(sorted(set(keys)))}
315 return [position[key] for key in keys]

318def degree_colouring(graph):
319 return rank([len(neighbours) for neighbours in graph.adj])

322def refine(graph, colours):
323 rounds = 0
324 while True:
325 signatures = [
326 (colours[v], tuple(sorted(colours[w] for w in graph.adj[v])))
327 for v in range(graph.n)
328 ]
329 refined = rank(signatures)
330 if max(refined) == max(colours):
331 return colours, rounds
332 colours = refined
333 rounds += 1

340def individualise(colours, vertex):
341 chosen = colours[vertex]
342 return [
343 colour if colour < chosen or other == vertex else colour + 1
344 for other, colour in enumerate(colours)

368class CanonicalForm:
369 def __init__(self, graph):
370 self.graph = graph
371 self.nodes = 0
372 self.leaves = 0
373 self.string = None
374 self.labels = None
375 self.best_leaves = []
376 self.visit(stable_colouring(graph)[0])
377
378 def visit(self, colours):
379 self.nodes += 1
380 if max(colours) == self.graph.n - 1:
381 self.record_leaf(colours)
382 return
383 cells = cells_of(colours)
384 target = min((cell for cell in cells if len(cell) > 1), key=len)
385 for vertex in target:
386 child, _ = refine(self.graph, individualise(colours, vertex))
387 self.visit(child)
388
389 def record_leaf(self, labels):
390 self.leaves += 1
391 string = leaf_string(self.graph, labels)
392 if self.string is None or string > self.string:
393 self.string = string
394 self.labels = labels
395 self.best_leaves = [labels]
396 elif string == self.string:
397 self.best_leaves.append(labels)
398
399 def automorphisms(self):
400 vertex_at = inverse(self.labels)
401 return [[vertex_at[label] for label in leaf] for leaf in self.best_leaves]

404def orbits(items, automorphisms, image):
405 result = []
406 seen = set()
407 for item in items:
408 if item not in seen:
409 orbit = sorted({image(sigma, item) for sigma in automorphisms})
410 seen.update(orbit)
411 result.append(orbit)
412 return result

415def embeddings(pattern, target):
416 earlier = [[u for u in pattern.adj[v] if u < v] for v in range(pattern.n)]
417 mapping = []
418
419 def extend():
420 v = len(mapping)
421 if v == pattern.n:
422 yield list(mapping)
423 return
424 for w in range(target.n):
425 if w not in mapping and all(target.has_edge(w, mapping[u]) for u in earlier[v]):
426 mapping.append(w)
427 yield from extend()
428 mapping.pop()
429
430 return extend()

433def is_induced(pattern, target, mapping):
434 return all(
435 target.has_edge(mapping[u], mapping[v]) == pattern.has_edge(u, v)
436 for u, v in combinations(range(pattern.n), 2)

440def parse_word(kind, word):
441 if not PATTERNS[kind].fullmatch(word):
442 raise CommandError("bad arguments")
443 return int(word) if kind == "number" else word

446def parse_arguments(kinds, words):
447 fixed = kinds
448 repeated = None
449 minimum = len(kinds)
450 if kinds and kinds[-1][-1] in "*+":
451 *fixed, repeated = kinds
452 minimum = len(fixed) + 1 if repeated.endswith("+") else len(fixed)
453 if len(words) < minimum or (repeated is None and len(words) > len(fixed)):
454 raise CommandError("bad arguments")
455 values = [parse_word(kind, word) for kind, word in zip(fixed, words)]
456 if repeated is not None:
457 values.append([parse_word(repeated[:-1], word) for word in words[len(fixed):]])
458 return values

461class GraphLab:
462 def __init__(self):
463 self.graphs = {}
464 self.commands = {
465 "GRAPH": (("name", "number", "edge*"), custom_graph),
466 "CYCLE": (("name", "number"), cycle_graph),
467 "PATH": (("name", "number"), path_graph),
468 "COMPLETE": (("name", "number"), complete_graph),
469 "BIPARTITE": (("name", "number", "number"), bipartite_graph),
470 "STAR": (("name", "number"), star_graph),
471 "WHEEL": (("name", "number"), wheel_graph),
472 "PETERSEN": (("name",), petersen_graph),
473 "HYPERCUBE": (("name", "number"), hypercube_graph),
474 "CIRCULANT": (("name", "number", "number+"), circulant_graph),
475 "GRID": (("name", "number", "number"), grid_graph),
476 "COMPLEMENT": (("name", "graph"), complement),
477 "LINE": (("name", "graph"), line_graph),
478 "PRODUCT": (("name", "graph", "graph"), cartesian_product),
479 "UNION": (("name", "graph", "graph"), disjoint_union),
480 "RELABEL": (("name", "graph", "number*"), relabel),
481 "INDUCED": (("name", "graph", "number+"), induced_subgraph),
482 "CANONIZE": (("name", "graph"), canonize),
483 "DELETE": (("graph",), self.delete),
484 "LIST": ((), self.list_graphs),
485 "SHOW": (("graph",), self.show),
486 "INVARIANTS": (("graph",), self.invariants),
487 "REFINE": (("graph", "number*"), self.refine_partition),
488 "CANON": (("graph",), self.canon),
489 "ISO": (("graph", "graph"), self.isomorphism),
490 "AUT": (("graph",), self.automorphism_group),
491 "SUBISO": (("graph", "graph"), self.subgraph_isomorphism),
492 "CLASSIFY": (("number",), self.classify),
493 }

495 def execute(self, line):
496 command, *words = line.split(" ")
497 try:
498 if command not in self.commands:
499 raise CommandError(f"unknown command {command}")
500 kinds, handler = self.commands[command]
501 arguments = parse_arguments(kinds, words)
502 for kind, value in zip(kinds, arguments):
503 if kind == "graph" and value not in self.graphs:
504 raise CommandError(f"no graph {value}")
505 if kinds[:1] == ("name",):
506 self.define(arguments[0], handler, kinds[1:], arguments[1:])
507 else:
508 handler(*arguments)
509 except CommandError as error:
510 print(f"error: {error}")

512 def define(self, name, build, kinds, arguments):

514 self.graphs[value] if kind == "graph" else value

517 graph = build(*inputs)
518 status = "redefined" if name in self.graphs else "defined"
519 self.graphs[name] = graph
520 print(status, name, "n", graph.n, "m", graph.edge_count())

556 def refine_partition(self, name, vertices):
557 graph = self.graphs[name]
558 if any(v >= graph.n for v in vertices):
559 raise CommandError("bad vertices")
560 colours, rounds = stable_colouring(graph)
561 for v in vertices:
562 if colours.count(colours[v]) == 1:
563 raise CommandError(f"singleton {v}")
564 colours, extra = refine(graph, individualise(colours, v))
565 rounds += extra
566 cells = cells_of(colours)
567 print("refine", name, "rounds", rounds, "cells", len(cells))
568 for colour, cell in enumerate(cells):
569 print(f"{colour}:", *cell)

577 def isomorphism(self, first_name, second_name):
578 first = self.graphs[first_name]
579 second = self.graphs[second_name]
580 a = Invariants(first)
581 b = Invariants(second)
582 checks = [
583 ("vertices", a.vertices, b.vertices, True),
584 ("edges", a.edges, b.edges, True),
585 ("degrees", a.degrees, b.degrees, False),
586 ("triangles", a.triangles, b.triangles, True),
587 ("components", len(a.sizes), len(b.sizes), True),
588 ("girth", a.girth, b.girth, True),
589 ("diameter", a.diameter, b.diameter, True),
590 ("bipartite", a.bipartite, b.bipartite, True),
591 ("refinement", refinement_profile(first), refinement_profile(second), False),
592 ]
593 for label, left, right, shown in checks:
594 if left != right:
595 values = (left, right) if shown else ()
596 print("not isomorphic", label, *values)
597 return
598 first_form = CanonicalForm(first)
599 second_form = CanonicalForm(second)
600 if first_form.string != second_form.string:
601 print("not isomorphic canonical")
602 return
603 vertex_at = inverse(second_form.labels)
604 print("isomorphic")
605 print("mapping", *(vertex_at[label] for label in first_form.labels))

607 def automorphism_group(self, name):
608 graph = self.graphs[name]
609 automorphisms = CanonicalForm(graph).automorphisms()
610 vertex_orbits = orbits(range(graph.n), automorphisms, lambda sigma, v: sigma[v])
611 edge_orbits = orbits(
612 graph.edges(),
613 automorphisms,
614 lambda sigma, edge: tuple(sorted(sigma[end] for end in edge)),
615 )
616 print("aut", name, "order", len(automorphisms))
617 print("vertex orbits", len(vertex_orbits))
618 for orbit in vertex_orbits:
619 print("orbit", *orbit)
620 print("edge orbits", len(edge_orbits))
621 for orbit in edge_orbits:
622 print("orbit", *(f"{u}-{v}" for u, v in orbit))
623 vertex_transitive = yes_no(len(vertex_orbits) == 1)
624 edge_transitive = yes_no(len(edge_orbits) == 1)
625 print("transitive vertex", vertex_transitive, "edge", edge_transitive)

627 def subgraph_isomorphism(self, pattern_name, target_name):
628 pattern = self.graphs[pattern_name]
629 target = self.graphs[target_name]
630 total = 0
631 induced_total = 0
632 shown = []
633 for mapping in embeddings(pattern, target):
634 induced = is_induced(pattern, target, mapping)
635 total += 1
636 induced_total += induced
637 if len(shown) < SHOWN_EMBEDDINGS:
638 shown.append((mapping, induced))
639 symmetries = len(CanonicalForm(pattern).best_leaves)
640 print("subiso", pattern_name, target_name, "embeddings", total, "induced", induced_total)
641 print("copies", total // symmetries, "induced", induced_total // symmetries)
642 for mapping, induced in shown:
643 suffix = ["induced"] if induced else []
644 print("embedding", *mapping, *suffix)

646 def classify(self, n):
647 check_range(n, 1, MAX_CLASSIFY_VERTICES)
648 pairs = list(combinations(range(n), 2))
649 layers = [{CanonicalForm(Graph(n)).string}]
650 for _ in range(len(pairs)):
651 following = set()
652 for string in layers[-1]:
653 edges = [pair for pair, bit in zip(pairs, string) if bit == "1"]
654 for pair, bit in zip(pairs, string):
655 if bit == "0":
656 following.add(CanonicalForm(Graph(n, edges + [pair])).string)
657 layers.append(following)
658 print("classify", n, "total", sum(map(len, layers)))
659 for m, layer in enumerate(layers):
660 strings = [string or "-" for string in sorted(layer)]
661 print("edges", m, "count", len(layer), *strings)
