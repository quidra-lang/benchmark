1use std::collections::{BTreeMap, BTreeSet, VecDeque};
2use std::io::{self, Read};
3#[derive(Clone)]
4struct Graph {
5 adjacent: Vec<u32>,
6}
7impl Graph {
8 fn new(n: usize) -> Self {
9 Self {
10 adjacent: vec![0; n]
11 }
12 }
13 fn n(&self) -> usize {
14 self.adjacent.len()
15 }
16 fn has(&self, u: usize, v: usize) -> bool {
17 self.adjacent[u] & (1 << v) != 0
18 }
19 fn join(&mut self, u: usize, v: usize) {
20 self.adjacent[u] |= 1 << v;
21 self.adjacent[v] |= 1 << u;
22 }
23 fn edges(&self) -> Vec<(usize, usize)> {
24 (0..self.n()).flat_map(|u| (u + 1..self.n()).filter_map(move |v| self.has(u, v).then_some((u, v)))).collect()
25 }
26 fn degrees(&self) -> Vec<usize> {
27 self.adjacent.iter().map(|a| a.count_ones() as usize).collect()
28 }
29 fn refine(&self, mut colors: Vec<usize>) -> (Vec<usize>, usize) {
30 let mut rounds = 0;
31 loop {
32 let signatures: Vec<_> = (0..self.n()).map(|v| {
33 let mut neighbors: Vec<_> = (0..self.n()).filter(|&w| self.has(v, w)).map(|w| colors[w]).collect();
34 neighbors.sort();
35 (colors[v], neighbors)
36 }).collect();
37 let unique: Vec<_> = signatures.iter().cloned().collect::<BTreeSet<_>>().into_iter().collect();
38 let next: Vec<_> = signatures.iter().map(|s| unique.binary_search(s).unwrap()).collect();
39 if color_count(&next) == color_count(&colors) {
40 return (colors, rounds);
41 }
42 colors = next;
43 rounds += 1;
44 }
45 }
46 fn stable(&self) -> (Vec<usize>, usize) {
47 let degrees = self.degrees();
48 let unique: Vec<_> = degrees.iter().copied().collect::<BTreeSet<_>>().into_iter().collect();
49 self.refine(degrees.iter().map(|d| unique.binary_search(d).unwrap()).collect())
50 }
51 fn leaf_string(&self, colors: &[usize]) -> String {
52 let mut inverse = vec![0; self.n()];
53 for (v, &c) in colors.iter().enumerate() {
54 inverse[c] = v;
55 }
56 (0..self.n()).flat_map(|i| (i + 1..self.n()).map(move |j| (i, j)))
57 .map(|(i, j)| if self.has(inverse[i], inverse[j]) {
58 '1'
59 } else {
60 '0'
61 }).collect()
62 }
63 fn canonical(&self) -> Canonical {
64 fn visit(g: &Graph, colors: Vec<usize>, result: &mut Canonical) {
65 result.nodes += 1;
66 let k = color_count(&colors);
67 if k == g.n() {
68 result.leaves += 1;
69 let text = g.leaf_string(&colors);
70 if result.winners.is_empty() || text > result.text {
71 result.text = text;
72 result.label = colors.clone();
73 result.winners.clear();
74 result.winners.push(colors);
75 } else if text == result.text {
76 result.winners.push(colors);
77 }
78 return;
79 }
80 let cell = (0..k).map(|c| (colors.iter().filter(|&&x| x == c).count(), c))
81 .filter(|&(size, _)| size > 1).min().unwrap().1;
82 for v in 0..g.n() {
83 if colors[v] == cell {
84 visit(g, g.refine(individualize(&colors, v)).0, result);
85 }
86 }
87 }
88 let mut result = Canonical {
89 nodes: 0, leaves: 0, text: String::new(), label: Vec::new(), winners: Vec::new()
90 };
91 visit(self, self.stable().0, &mut result);
92 result
93 }
94 fn invariants(&self) -> Invariants {
95 let n = self.n();
96 let mut sizes = Vec::new();
97 let mut seen = vec![false; n];
98 let mut eccentricities = Vec::new();
99 let mut girth: Option<usize> = None;
100 let mut bipartite = true;
101 for start in 0..n {
102 let mut distance = vec![None; n];
103 let mut parent = vec![n; n];
104 let mut queue = VecDeque::from([start]);
105 distance[start] = Some(0usize);
106 while let Some(u) = queue.pop_front() {
107 for v in 0..n {
108 if !self.has(u, v) {
109 continue;
110 }
111 if let Some(d) = distance[v] {
112 if d % 2 == distance[u].unwrap() % 2 {
113 bipartite = false;
114 }
115 if parent[u] != v && parent[v] != u {
116 let length = distance[u].unwrap() + d + 1;
117 girth = Some(girth.map_or(length, |old| old.min(length)));
118 }
119 } else {
120 distance[v] = Some(distance[u].unwrap() + 1);
121 parent[v] = u;
122 queue.push_back(v);
123 }
124 }
125 }
126 if !seen[start] {
127 sizes.push(distance.iter().filter(|d| d.is_some()).count());
128 for v in 0..n {
129 if distance[v].is_some() {
130 seen[v] = true;
131 }
132 }
133 }
134 eccentricities.push(distance.into_iter().flatten().max().unwrap());
135 }
136 sizes.sort_by(|a, b| b.cmp(a));
137 let mut degrees = self.degrees();
138 degrees.sort_by(|a, b| b.cmp(a));
139 let triangles = (0..n).map(|u| (u + 1..n).map(|v| (v + 1..n)
140 .filter(|&w| self.has(u, v) && self.has(v, w) && self.has(u, w)).count()).sum::<usize>()).sum();
141 Invariants {
142 degrees, triangles, sizes: sizes.clone(), girth,
143 diameter: (sizes.len() == 1).then(|| *eccentricities.iter().max().unwrap()),
144 radius: (sizes.len() == 1).then(|| *eccentricities.iter().min().unwrap()), bipartite,
145 }
146 }
147 fn clique(&self) -> Vec<usize> {
148 let mut best = Vec::new();
149 for mask in 1u32..1u32 << self.n() {
150 let vertices: Vec<_> = (0..self.n()).filter(|&v| mask & (1 << v) != 0).collect();
151 if vertices.len() < best.len() {
152 continue;
153 }
154 if vertices.iter().all(|&v| (mask ^ (1 << v)) & !self.adjacent[v] == 0)
155 && (vertices.len() > best.len() || vertices < best) {
156 best = vertices;
157 }
158 }
159 best
160 }
161 fn coloring(&self) -> (usize, Vec<usize>) {
162 fn find(g: &Graph, limit: usize, colors: &mut Vec<usize>, used: usize) -> bool {
163 let v = colors.len();
164 if v == g.n() {
165 return true;
166 }
167 for c in 0..limit.min(used + 1) {
168 if (0..v).any(|u| g.has(u, v) && colors[u] == c) {
169 continue;
170 }
171 colors.push(c);
172 if find(g, limit, colors, used.max(c + 1)) {
173 return true;
174 }
175 colors.pop();
176 }
177 false
178 }
179 for limit in 1..=self.n() {
180 let mut colors = Vec::new();
181 if find(self, limit, &mut colors, 0) {
182 return (limit, colors);
183 }
184 }
185 unreachable!()
186 }
187}
188fn color_count(colors: &[usize]) -> usize {
189 colors.iter().max().unwrap() + 1
190}
191fn individualize(colors: &[usize], v: usize) -> Vec<usize> {
192 colors.iter().enumerate().map(|(w, &c)| if c < colors[v] || w == v {
193 c
194 } else {
195 c + 1
196 }).collect()
197}

217struct UnionFind {
218 parent: Vec<usize>
219}
220impl UnionFind {
221 fn new(n: usize) -> Self {
222 Self {
223 parent: (0..n).collect()
224 }
225 }
226 fn root(&mut self, v: usize) -> usize {
227 if self.parent[v] != v {
228 self.parent[v] = self.root(self.parent[v]);
229 }
230 self.parent[v]
231 }
232 fn join(&mut self, u: usize, v: usize) {
233 let a = self.root(u);
234 let b = self.root(v);
235 self.parent[a] = b;
236 }
237 fn orbits(&mut self) -> Vec<Vec<usize>> {
238 let mut groups = BTreeMap::<usize, Vec<usize>>::new();
239 for v in 0..self.parent.len() {
240 groups.entry(self.root(v)).or_default().push(v);
241 }
242 let mut result: Vec<_> = groups.into_values().collect();
243 result.sort_by_key(|g| g[0]);
244 result
245 }
246}
247fn graph_name(s: &str) -> bool {
248 (1..=8).contains(&s.len()) && s.as_bytes()[0].is_ascii_lowercase() && s.bytes().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit())
249}
250fn number(s: &str) -> Option<usize> {
251 if !(1..=3).contains(&s.len()) || (s.len() > 1 && s.starts_with('0')) || !s.bytes().all(|c| c.is_ascii_digit()) {
252 None
253 } else {
254 s.parse().ok()
255 }
256}
257fn edge(s: &str) -> Option<(usize, usize)> {
258 let (a, b) = s.split_once('-')?;
259 Some((number(a)?, number(b)?))
260}

264fn command(registry: &mut BTreeMap<String, Graph>, w: &[&str]) -> Result<Vec<String>, String> {
265 let c = w[0];
266 let a = &w[1..];
267 let defining = ["GRAPH", "CYCLE", "PATH", "COMPLETE", "BIPARTITE", "STAR", "WHEEL", "PETERSEN", "HYPERCUBE", "CIRCULANT", "GRID", "COMPLEMENT", "LINE", "PRODUCT", "UNION", "RELABEL", "INDUCED", "CANONIZE"];
268 let known = ["DELETE", "LIST", "SHOW", "INVARIANTS", "REFINE", "CANON", "ISO", "AUT", "SUBISO", "CLASSIFY"];
269 if !defining.contains(&c) && !known.contains(&c) {
270 return Err(fail(&format!("unknown command {c}")));
271 }
272 let count = a.len();
273 let shape = match c {
274 "GRAPH" | "RELABEL" => count >= 2, "CIRCULANT" | "INDUCED" => count >= 3,
275 "REFINE" => count >= 1, "LIST" => count == 0, "PETERSEN" | "DELETE" | "SHOW" | "INVARIANTS" | "CANON" | "AUT" | "CLASSIFY" => count == 1,
276 "BIPARTITE" | "GRID" | "PRODUCT" | "UNION" => count == 3, _ => count == 2,
277 };
278 if !shape {
279 return Err(fail("bad arguments"));
280 }
281 let graph_positions: Vec<usize> = match c {
282 "COMPLEMENT" | "LINE" | "RELABEL" | "INDUCED" | "CANONIZE" => vec![1],
283 "PRODUCT" | "UNION" => vec![1, 2], "ISO" | "SUBISO" => vec![0, 1],
284 "DELETE" | "SHOW" | "INVARIANTS" | "REFINE" | "CANON" | "AUT" => vec![0], _ => Vec::new(),
285 };
286 let numeric_positions: Vec<usize> = match c {
287 "GRAPH" => vec![1], "CYCLE" | "PATH" | "COMPLETE" | "STAR" | "WHEEL" | "HYPERCUBE" => vec![1],
288 "BIPARTITE" | "GRID" => vec![1, 2], "CIRCULANT" => (1..count).collect(),
289 "RELABEL" | "INDUCED" => (2..count).collect(), "REFINE" => (1..count).collect(), "CLASSIFY" => vec![0], _ => Vec::new(),
290 };
291 for (i, word) in a.iter().enumerate() {
292 let valid = if numeric_positions.contains(&i) {
293 number(word).is_some()
294 }
295 else if c == "GRAPH" && i >= 2 {
296 edge(word).is_some()
297 } else {
298 graph_name(word)
299 };
300 if !valid {
301 return Err(fail("bad arguments"));
302 }
303 }
304 for &i in &graph_positions {
305 if !registry.contains_key(a[i]) {
306 return Err(fail(&format!("no graph {}", a[i])));
307 }
308 }
309 let num = |i: usize| number(a[i]).unwrap();
310 let graph = |i: usize| registry[a[i]].clone();
311 let bounded = |n: usize, minimum: usize| -> Result<usize, String> {
312 if n < minimum {
313 Err(fail("bad value"))
314 } else if n > 16 {
315 Err(fail("too large"))
316 } else {
317 Ok(n)
318 }
319 };
320 if defining.contains(&c) {
321 let mut g = match c {
322 "GRAPH" | "CYCLE" | "PATH" | "COMPLETE" | "CIRCULANT" => Graph::new(bounded(num(1), if c == "CYCLE" || c == "CIRCULANT" {
323 3
324 } else {
325 1
326 })?),
327 "STAR" | "WHEEL" => {
328 if num(1) < if c == "STAR" {
329 1
330 } else {
331 3
332 } {
333 return Err(fail("bad value"));
334 } Graph::new(bounded(num(1) + 1, 1)?)
335 },
336 "BIPARTITE" | "GRID" => {
337 if num(1) == 0 || num(2) == 0 {
338 return Err(fail("bad value"));
339 } Graph::new(bounded(if c == "GRID" {
340 num(1) * num(2)
341 } else {
342 num(1) + num(2)
343 }, 1)?)
344 },
345 "HYPERCUBE" => {
346 if num(1) == 0 {
347 return Err(fail("bad value"));
348 }
349 if num(1) > 4 {
350 return Err(fail("too large"));
351 } Graph::new(1 << num(1))
352 },
353 "PETERSEN" => Graph::new(10),
354 "COMPLEMENT" | "RELABEL" | "CANONIZE" => Graph::new(graph(1).n()),
355 "LINE" => {
356 let m = graph(1).edges().len();
357 if m == 0 {
358 return Err(fail("no edges"));
359 } Graph::new(bounded(m, 1)?)
360 },
361 "INDUCED" => Graph::new(count - 2),
362 "PRODUCT" => Graph::new(bounded(graph(1).n() * graph(2).n(), 1)?),
363 "UNION" => Graph::new(bounded(graph(1).n() + graph(2).n(), 1)?), _ => unreachable!(),
364 };
365 let n = g.n();
366 match c {
367 "GRAPH" => for &word in &a[2..] {
368 let (u, v) = edge(word).unwrap();
369 if u >= n || v >= n || u == v {
370 return Err(fail(&format!("bad edge {word}")));
371 }
372 if g.has(u, v) {
373 return Err(fail(&format!("duplicate edge {word}")));
374 }
375 g.join(u, v);
376 },
377 "CYCLE" | "PATH" => {
378 for i in 0..n - 1 {
379 g.join(i, i + 1);
380 }
381 if c == "CYCLE" {
382 g.join(n - 1, 0);
383 }
384 },
385 "COMPLETE" => for i in 0..n {
386 for j in i + 1..n {
387 g.join(i, j);
388 }
389 },
390 "BIPARTITE" => for i in 0..num(1) {
391 for j in num(1)..n {
392 g.join(i, j);
393 }
394 },
395 "STAR" | "WHEEL" => {
396 for i in 1..n {
397 g.join(0, i);
398 }
399 if c == "WHEEL" {
400 for i in 1..n - 1 {
401 g.join(i, i + 1);
402 } g.join(n - 1, 1);
403 }
404 },
405 "PETERSEN" => for i in 0..5 {
406 g.join(i, (i + 1) % 5);
407 g.join(i, i + 5);
408 g.join(i + 5, 5 + (i + 2) % 5);
409 },
410 "HYPERCUBE" => for u in 0..n {
411 for bit in 0..num(1) {
412 g.join(u, u ^ (1 << bit));
413 }
414 },
415 "CIRCULANT" => {
416 let jumps: Vec<_> = (2..count).map(num).collect();
417 if jumps.iter().any(|&j| j == 0 || 2 * j > n) || jumps.windows(2).any(|p| p[0] >= p[1]) {
418 return Err(fail("bad value"));
419 }
420 for i in 0..n {
421 for &j in &jumps {
422 g.join(i, (i + j) % n);
423 }
424 }
425 },
426 "GRID" => for x in 0..num(1) {
427 for y in 0..num(2) {
428 let u = x * num(2) + y;
429 if y + 1 < num(2) {
430 g.join(u, u + 1);
431 }
432 if x + 1 < num(1) {
433 g.join(u, u + num(2));
434 }
435 }
436 },
437 "COMPLEMENT" => {
438 let old = graph(1);
439 for u in 0..n {
440 for v in u + 1..n {
441 if !old.has(u, v) {
442 g.join(u, v);
443 }
444 }
445 }
446 },
447 "LINE" => {
448 let edges = graph(1).edges();
449 for i in 0..n {
450 for j in i + 1..n {

452 g.join(i, j);
453 }
454 }
455 }
456 },
457 "PRODUCT" => {
458 let x = graph(1);
459 let y = graph(2);
460 for u in 0..x.n() {
461 for (v, w) in y.edges() {
462 g.join(u * y.n() + v, u * y.n() + w);
463 }
464 }
465 for (u, v) in x.edges() {
466 for w in 0..y.n() {
467 g.join(u * y.n() + w, v * y.n() + w);
468 }
469 }
470 },
471 "UNION" => {
472 let x = graph(1);

474 g.join(u, v);

477 g.join(u + x.n(), v + x.n());
478 }
479 },
480 "RELABEL" | "CANONIZE" => {
481 let old = graph(1);
482 let p: Vec<_> = if c == "CANONIZE" {
483 old.canonical().label

485 (2..count).map(num).collect()
486 };

488 return Err(fail("bad permutation"));

492 }
493 },
494 "INDUCED" => {

496 let p: Vec<_> = (2..count).map(num).collect();
497 if p.iter().any(|&v| v >= old.n()) || p.iter().collect::<BTreeSet<_>>().len() != p.len() {
498 return Err(fail("bad vertices"));
499 }
500 for u in 0..n {
501 for v in u + 1..n {
502 if old.has(p[u], p[v]) {
503 g.join(u, v);
504 }
505 }
506 }
507 }, _ => unreachable!(),
508 }
509 let output = format!("{} {} n {} m {}", if registry.contains_key(a[0]) {
510 "redefined"
511 } else {
512 "defined"
513 }, a[0], n, g.edges().len());
514 registry.insert(a[0].into(), g);
515 return Ok(vec![output]);
516 }
517 if c == "LIST" {
518 let mut out = vec![format!("graphs {}", registry.len())];
519 out.extend(registry.iter().map(|(name, g)| format!("{name} n {} m {}", g.n(), g.edges().len())));
520 return Ok(out);
521 }
522 if c == "DELETE" {
523 registry.remove(a[0]);
524 return Ok(vec![format!("deleted {}", a[0])]);
525 }
526 if c == "CLASSIFY" {
527 let n = num(0);
528 if n == 0 {
529 return Err(fail("bad value"));
530 }
531 if n > 5 {
532 return Err(fail("too large"));
533 }
534 let pairs = Graph::new(n).n() * (n - 1) / 2;
535 let mut classes = vec![BTreeSet::new(); pairs + 1];
536 for mask in 0usize..1usize << pairs {
537 let mut g = Graph::new(n);
538 let mut bit = 0;
539 for u in 0..n {
540 for v in u + 1..n {
541 if mask & (1 << bit) != 0 {
542 g.join(u, v);
543 } bit += 1;
544 }
545 }
546 classes[mask.count_ones() as usize].insert(g.canonical().text);
547 }
548 let mut out = vec![format!("classify {n} total {}", classes.iter().map(BTreeSet::len).sum::<usize>())];
549 for (m, set) in classes.iter().enumerate() {
550 out.push(format!("edges {m} count {} {}", set.len(), set.iter().map(|s| if s.is_empty() {
551 "-"
552 } else {
553 s
554 }).collect::<Vec<_>>().join(" ")));
555 }
556 return Ok(out);
557 }
558 let g = graph(0);
559 let n = g.n();
560 match c {
561 "SHOW" => {
562 let mut out = vec![format!("graph {} n {n} m {}", a[0], g.edges().len())];
563 for u in 0..n {
564 let neighbors: Vec<_> = (0..n).filter(|&v| g.has(u, v)).collect();
565 out.push(format!("{u}:{}", if neighbors.is_empty() {
566 String::new()
567 } else {
568 " ".to_string() + &numbers(&neighbors)
569 }));
570 } Ok(out)
571 },
572 "INVARIANTS" => {
573 let inv = g.invariants();
574 let mut out = vec![format!("invariants {}", a[0]), format!("degrees {}", numbers(&inv.degrees)), format!("edges {} triangles {}", g.edges().len(), inv.triangles), format!("components {} sizes {}", inv.sizes.len(), numbers(&inv.sizes)), format!("girth {} diameter {} radius {}", shown(inv.girth, "none"), shown(inv.diameter, "INF"), shown(inv.radius, "INF")), format!("bipartite {}", yes(inv.bipartite))];
575 if n > 12 {
576 out.extend(["clique -".into(), "chromatic -".into()]);
577 } else {
578 let clique = g.clique();
579 let (chi, colors) = g.coloring();
580 out.push(format!("clique {} {}", clique.len(), numbers(&clique)));
581 out.push(format!("chromatic {chi} {}", numbers(&colors)));
582 } Ok(out)
583 },
584 "REFINE" => {
585 let vertices: Vec<_> = (1..count).map(num).collect();
586 if vertices.iter().any(|&v| v >= n) {
587 return Err(fail("bad vertices"));
588 }
589 let (mut colors, mut rounds) = g.stable();
590 for v in vertices {
591 if colors.iter().filter(|&&x| x == colors[v]).count() == 1 {
592 return Err(fail(&format!("singleton {v}")));
593 }
594 let (next, r) = g.refine(individualize(&colors, v));
595 colors = next;
596 rounds += r;
597 }
598 let k = color_count(&colors);
599 let mut out = vec![format!("refine {} rounds {rounds} cells {k}", a[0])];
600 for cell in 0..k {
601 out.push(format!("{cell}: {}", numbers(&(0..n).filter(|&v| colors[v] == cell).collect::<Vec<_>>())));
602 } Ok(out)
603 },
604 "CANON" => {
605 let result = g.canonical();
606 Ok(vec![format!("canon {} nodes {} leaves {}", a[0], result.nodes, result.leaves), format!("labels {}", numbers(&result.label)), format!("string {}", if result.text.is_empty() {
607 "-"
608 } else {
609 &result.text
610 })])
611 },
612 "ISO" => {
613 let h = graph(1);
614 let x = g.invariants();
615 let y = h.invariants();
616 let difference = if n != h.n() {
617 Some(format!("vertices {n} {}", h.n()))
618 } else if g.edges().len() != h.edges().len() {
619 Some(format!("edges {} {}", g.edges().len(), h.edges().len()))
620 } else if x.degrees != y.degrees {
621 Some("degrees".into())
622 } else if x.triangles != y.triangles {
623 Some(format!("triangles {} {}", x.triangles, y.triangles))
624 } else if x.sizes.len() != y.sizes.len() {
625 Some(format!("components {} {}", x.sizes.len(), y.sizes.len()))
626 } else if x.girth != y.girth {
627 Some(format!("girth {} {}", shown(x.girth, "none"), shown(y.girth, "none")))
628 } else if x.diameter != y.diameter {
629 Some(format!("diameter {} {}", shown(x.diameter, "INF"), shown(y.diameter, "INF")))
630 } else if x.bipartite != y.bipartite {
631 Some(format!("bipartite {} {}", yes(x.bipartite), yes(y.bipartite)))
632 } else {
633 None
634 };
635 if let Some(reason) = difference {
636 return Ok(vec![format!("not isomorphic {reason}")]);
637 }
638 let (p, r) = g.stable();
639 let (q, s) = h.stable();
640 let sizes = |colors: &[usize]| (0..color_count(colors)).map(|c| colors.iter().filter(|&&x| x == c).count()).collect::<Vec<_>>();
641 if r != s || sizes(&p) != sizes(&q) {
642 return Ok(vec!["not isomorphic refinement".into()]);
643 }
644 let x = g.canonical();
645 let y = h.canonical();
646 if x.text != y.text {
647 return Ok(vec!["not isomorphic canonical".into()]);
648 }
649 let mapping: Vec<_> = x.label.iter().map(|c| y.label.iter().position(|d| d == c).unwrap()).collect();
650 Ok(vec!["isomorphic".into(), format!("mapping {}", numbers(&mapping))])
651 },
652 "AUT" => {
653 let result = g.canonical();
654 let edges = g.edges();
655 let mut vertices = UnionFind::new(n);
656 let mut edge_sets = UnionFind::new(edges.len());
657 let mut inverse = vec![0; n];
658 for v in 0..n {
659 inverse[result.label[v]] = v;
660 }
661 for label in &result.winners {
662 let map: Vec<_> = label.iter().map(|&c| inverse[c]).collect();
663 for v in 0..n {
664 vertices.join(v, map[v]);
665 }
666 for (i, &(u, v)) in edges.iter().enumerate() {
667 let e = (map[u].min(map[v]), map[u].max(map[v]));
668 edge_sets.join(i, edges.binary_search(&e).unwrap());
669 }
670 }
671 let vo = vertices.orbits();
672 let eo = edge_sets.orbits();
673 let mut out = vec![format!("aut {} order {}", a[0], result.winners.len()), format!("vertex orbits {}", vo.len())];
674 out.extend(vo.iter().map(|o| format!("orbit {}", numbers(o))));
675 out.push(format!("edge orbits {}", eo.len()));
676 for orbit in &eo {
677 out.push(format!("orbit {}", orbit.iter().map(|&i| format!("{}-{}", edges[i].0, edges[i].1)).collect::<Vec<_>>().join(" ")));
678 }
679 out.push(format!("transitive vertex {} edge {}", yes(vo.len() == 1), yes(eo.len() == 1)));
680 Ok(out)
681 },
682 "SUBISO" => {
683 fn enumerate(p: &Graph, t: &Graph, mapping: &mut Vec<usize>, used: u32, induced: bool, counts: &mut [usize; 2], first: &mut Vec<(Vec<usize>, bool)>) {
684 let v = mapping.len();
685 if v == p.n() {
686 counts[0] += 1;
687 counts[1] += usize::from(induced);
688 if first.len() < 5 {
689 first.push((mapping.clone(), induced));
690 }
691 return;
692 }
693 for w in 0..t.n() {
694 if used & (1 << w) != 0 || (0..v).any(|u| p.has(u, v) && !t.has(mapping[u], w)) {
695 continue;
696 }
697 let next = induced && (0..v).all(|u| p.has(u, v) == t.has(mapping[u], w));
698 mapping.push(w);
699 enumerate(p, t, mapping, used | (1 << w), next, counts, first);
700 mapping.pop();
701 }
702 }
703 let target = graph(1);
704 let mut counts = [0; 2];
705 let mut first = Vec::new();
706 if n <= target.n() {
707 enumerate(&g, &target, &mut Vec::new(), 0, true, &mut counts, &mut first);
708 }
709 let order = g.canonical().winners.len();
710 let mut out = vec![format!("subiso {} {} embeddings {} induced {}", a[0], a[1], counts[0], counts[1]), format!("copies {} induced {}", counts[0] / order, counts[1] / order)];
711 for (mapping, induced) in first {
712 out.push(format!("embedding {}{}", numbers(&mapping), if induced {
713 " induced"
714 } else {
715 ""
716 }));
717 } Ok(out)
718 }, _ => unreachable!(),
719 }
720}
