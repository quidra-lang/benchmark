3class Graph {
4 constructor(n) { this.n = n; this.rows = Array(n).fill(0); }
5 add(u, v) { this.rows[u] |= 1 << v; this.rows[v] |= 1 << u; }
6 adjacent(u, v) { return (this.rows[u] & (1 << v)) !== 0; }
7 neighbors(u) { return Array.from({ length: this.n }, (_, v) => v).filter(v => this.adjacent(u, v)); }
8 edges() {
9 const result = [];
10 for (let u = 0; u < this.n; u++) for (let v = u + 1; v < this.n; v++) if (this.adjacent(u, v)) result.push([u, v]);
11 return result;
12 }
13}

17function lex(a, b) {
18 for (let i = 0; i < Math.min(a.length, b.length); i++) if (a[i] !== b[i]) return a[i] - b[i];
19 return a.length - b.length;
20}
21function popcount(mask) {
22 let count = 0;
23 while (mask) { mask &= mask - 1; count++; }
24 return count;
25}
26function cells(colors) {
27 const result = Array.from({ length: Math.max(...colors) + 1 }, () => []);
28 colors.forEach((color, vertex) => result[color].push(vertex));
29 return result;
30}
31function refine(graph, initial) {
32 let colors = [...initial], rounds = 0;
33 while (true) {
34 const signatures = colors.map((color, vertex) => [color, ...graph.neighbors(vertex).map(v => colors[v]).sort((a, b) => a - b)]);
35 const order = Array.from({ length: graph.n }, (_, v) => v).sort((a, b) => lex(signatures[a], signatures[b]));
36 const next = Array(graph.n).fill(0);
37 let color = 0;
38 for (let i = 0; i < order.length; i++) {
39 if (i && lex(signatures[order[i - 1]], signatures[order[i]]) !== 0) color++;
40 next[order[i]] = color;
41 }
42 if (color + 1 === Math.max(...colors) + 1) return { colors, rounds };
43 colors = next; rounds++;
44 }
45}
46function stable(graph) {
47 if (!graph.stable) {
48 const degrees = graph.rows.map(popcount), unique = [...new Set(degrees)].sort((a, b) => a - b);
49 graph.stable = refine(graph, degrees.map(degree => unique.indexOf(degree)));
50 }
51 return graph.stable;
52}
53function individualize(colors, vertex) {
54 const x = colors[vertex];
55 return colors.map((color, w) => color < x || w === vertex ? color : color + 1);
56}
57function leafString(graph, labels) {
58 const inverse = Array(graph.n);
59 labels.forEach((label, v) => inverse[label] = v);
60 let result = '';
61 for (let i = 0; i < graph.n; i++) for (let j = i + 1; j < graph.n; j++) result += graph.adjacent(inverse[i], inverse[j]) ? '1' : '0';
62 return result;
63}
64function canon(graph) {
65 if (graph.canonical) return graph.canonical;
66 const result = { nodes: 0, leaves: 0, labels: [], string: '', bestLeaves: [] };
67 function visit(colors) {
68 result.nodes++;
69 const partition = cells(colors);
70 if (partition.length === graph.n) {
71 result.leaves++;
72 const string = leafString(graph, colors);
73 if (!result.bestLeaves.length || string > result.string) {
74 result.string = string;
75 result.labels = [...colors];
76 result.bestLeaves = [[...colors]];
77 } else if (string === result.string) result.bestLeaves.push([...colors]);
78 return;
79 }
80 const chosen = partition.filter(cell => cell.length > 1).reduce((best, cell) => cell.length < best.length ? cell : best);
81 for (const vertex of chosen) visit(refine(graph, individualize(colors, vertex)).colors);
82 }
83 visit(stable(graph).colors);
84 graph.canonical = result;
85 return result;
86}
87function invariants(graph) {
88 if (graph.invariants) return graph.invariants;
89 const n = graph.n, degrees = graph.rows.map(popcount).sort((a, b) => b - a), sizes = [];
90 const seen = new Set();
91 let triangles = 0, shortest = Infinity, bipartite = true;
92 const eccentricities = [];
93 for (let u = 0; u < n; u++) for (let v = u + 1; v < n; v++) for (let w = v + 1; w < n; w++) {
94 if (graph.adjacent(u, v) && graph.adjacent(u, w) && graph.adjacent(v, w)) triangles++;
95 }
96 for (let start = 0; start < n; start++) {
97 const distance = Array(n).fill(-1), parent = Array(n).fill(-1), queue = [start];
98 distance[start] = 0;
99 for (let i = 0; i < queue.length; i++) {
100 const u = queue[i];
101 for (const v of graph.neighbors(u)) {
102 if (distance[v] < 0) { distance[v] = distance[u] + 1; parent[v] = u; queue.push(v); }
103 else if (parent[u] !== v) shortest = Math.min(shortest, distance[u] + distance[v] + 1);
104 if (distance[v] % 2 === distance[u] % 2) bipartite = false;
105 }
106 }
107 eccentricities.push(Math.max(...distance));
108 if (!seen.has(start)) { sizes.push(queue.length); for (const v of queue) seen.add(v); }
109 }
110 sizes.sort((a, b) => b - a);
111 const result = {
112 degrees, triangles, sizes, girth: shortest === Infinity ? 'none' : String(shortest),
113 diameter: sizes.length > 1 ? 'INF' : String(Math.max(...eccentricities)),
114 radius: sizes.length > 1 ? 'INF' : String(Math.min(...eccentricities)),
115 bipartite: bipartite ? 'yes' : 'no'
116 };
117 graph.invariants = result;
118 return result;
119}
120function clique(graph) {
121 let best = [];
122 for (let mask = 1; mask < 1 << graph.n; mask++) {
123 const vertices = Array.from({ length: graph.n }, (_, v) => v).filter(v => (mask & (1 << v)) !== 0);
124 if (vertices.length < best.length) continue;
125 if (vertices.every(u => vertices.every(v => u === v || graph.adjacent(u, v))) &&
126 (vertices.length > best.length || lex(vertices, best) < 0)) best = vertices;
127 }
128 return best;
129}
130function coloring(graph, lower) {
131 const colors = Array(graph.n).fill(-1);
132 function search(vertex, count) {
133 if (vertex === graph.n) return true;
134 for (let color = 0; color < count; color++) {
135 let valid = true;
136 for (let u = 0; u < vertex; u++) if (colors[u] === color && graph.adjacent(u, vertex)) { valid = false; break; }
137 if (!valid) continue;
138 colors[vertex] = color;
139 if (search(vertex + 1, count)) return true;
140 }
141 colors[vertex] = -1;
142 return false;
143 }
144 for (let count = lower; count <= graph.n; count++) if (search(0, count)) return [...colors];
145 return colors;
146}
147class DisjointSet {
148 constructor(n) { this.parent = Array.from({ length: n }, (_, i) => i); }
149 find(x) { return this.parent[x] === x ? x : this.parent[x] = this.find(this.parent[x]); }
150 join(a, b) { this.parent[this.find(b)] = this.find(a); }
151 groups() {
152 const groups = new Map();
153 for (let i = 0; i < this.parent.length; i++) {
154 const root = this.find(i);
155 if (!groups.has(root)) groups.set(root, []);
156 groups.get(root).push(i);
157 }
158 return [...groups.values()].sort((a, b) => a[0] - b[0]);
159 }
160}

163 ...['CYCLE', 'PATH', 'COMPLETE', 'STAR', 'WHEEL', 'HYPERCUBE'].map(command => [command, { fixed: ['name', 'num'] }]),

174const namePattern = /^[a-z][a-z0-9]{0,7}$/;
175const numberPattern = /^(0|[1-9][0-9]{0,2})$/;
176function valid(word, kind) {
177 if (kind === 'num') return numberPattern.test(word);
178 if (kind === 'edge') { const pair = word.split('-'); return pair.length === 2 && pair.every(value => numberPattern.test(value)); }
179 return namePattern.test(word);
180}

190function define(command, args) {
191 const n = Number(args[1]), source = registry.get(args[1]);
192 let result;
193 switch (command) {
194 case 'GRAPH': {
195 checkSize(n); result = new Graph(n);
196 for (const word of args.slice(2)) {
197 const [u, v] = word.split('-').map(Number);
198 if (u >= n || v >= n || u === v) fail(`bad edge ${word}`);
199 if (result.adjacent(u, v)) fail(`duplicate edge ${word}`);
200 result.add(u, v);
201 }
202 return result;
203 }
204 case 'CYCLE': case 'PATH': case 'COMPLETE':
205 checkSize(n, command === 'CYCLE' ? 3 : 1); result = new Graph(n);
206 for (let u = 0; u < n; u++) for (let v = u + 1; v < n; v++) {
207 if (command === 'COMPLETE' || v === u + 1 || command === 'CYCLE' && u === 0 && v === n - 1) result.add(u, v);
208 }
209 return result;
210 case 'BIPARTITE': {
211 const b = Number(args[2]);
212 if (!n || !b) fail('bad value');
213 checkSize(n + b); result = new Graph(n + b);
214 for (let u = 0; u < n; u++) for (let v = 0; v < b; v++) result.add(u, n + v);
215 return result;
216 }
217 case 'STAR': case 'WHEEL':
218 if (n < (command === 'WHEEL' ? 3 : 1)) fail('bad value');
219 checkSize(n + 1); result = new Graph(n + 1);
220 for (let v = 1; v <= n; v++) result.add(0, v);
221 if (command === 'WHEEL') for (let v = 1; v <= n; v++) result.add(v, v % n + 1);
222 return result;
223 case 'PETERSEN':
224 result = new Graph(10);
225 for (let i = 0; i < 5; i++) { result.add(i, (i + 1) % 5); result.add(i, i + 5); result.add(i + 5, 5 + (i + 2) % 5); }
226 return result;
227 case 'HYPERCUBE':
228 if (!n) fail('bad value');
229 if (n >= 5) fail('too large');
230 result = new Graph(2 ** n);
231 for (let u = 0; u < result.n; u++) for (let bit = 0; bit < n; bit++) result.add(u, u ^ (1 << bit));
232 return result;
233 case 'CIRCULANT': {
234 checkSize(n, 3);
235 const jumps = args.slice(2).map(Number);
236 if (jumps.some((jump, i) => jump === 0 || 2 * jump > n || i > 0 && jumps[i - 1] >= jump)) fail('bad value');
237 result = new Graph(n);
238 for (let u = 0; u < n; u++) for (const jump of jumps) result.add(u, (u + jump) % n);
239 return result;
240 }
241 case 'GRID': {
242 const c = Number(args[2]);
243 if (!n || !c) fail('bad value');
244 checkSize(n * c); result = new Graph(n * c);
245 for (let x = 0; x < n; x++) for (let y = 0; y < c; y++) {
246 if (y + 1 < c) result.add(x * c + y, x * c + y + 1);
247 if (x + 1 < n) result.add(x * c + y, (x + 1) * c + y);
248 }
249 return result;
250 }
251 case 'COMPLEMENT':
252 result = new Graph(source.n);
253 for (let u = 0; u < source.n; u++) for (let v = u + 1; v < source.n; v++) if (!source.adjacent(u, v)) result.add(u, v);
254 return result;
255 case 'LINE': {
256 const edges = source.edges();
257 if (!edges.length) fail('no edges');
258 checkSize(edges.length); result = new Graph(edges.length);
259 for (let u = 0; u < edges.length; u++) for (let v = u + 1; v < edges.length; v++) if (edges[u].some(vertex => edges[v].includes(vertex))) result.add(u, v);
260 return result;
261 }
262 case 'PRODUCT': case 'UNION': {
263 const other = registry.get(args[2]), a = source.n, b = other.n;
264 checkSize(command === 'PRODUCT' ? a * b : a + b);
265 result = new Graph(command === 'PRODUCT' ? a * b : a + b);
266 if (command === 'UNION') {
267 for (const [u, v] of source.edges()) result.add(u, v);
268 for (const [u, v] of other.edges()) result.add(u + a, v + a);
269 } else {
270 for (let x = 0; x < a; x++) for (const [u, v] of other.edges()) result.add(x * b + u, x * b + v);
271 for (const [u, v] of source.edges()) for (let y = 0; y < b; y++) result.add(u * b + y, v * b + y);
272 }
273 return result;
274 }
275 case 'RELABEL': {
276 const permutation = args.slice(2).map(Number);
277 if (permutation.length !== source.n || new Set(permutation).size !== source.n || permutation.some(v => v >= source.n)) fail('bad permutation');
278 return relabel(source, permutation);
279 }
280 case 'INDUCED': {
281 const vertices = args.slice(2).map(Number);
282 if (new Set(vertices).size !== vertices.length || vertices.some(v => v >= source.n)) fail('bad vertices');
283 result = new Graph(vertices.length);
284 for (let u = 0; u < vertices.length; u++) for (let v = u + 1; v < vertices.length; v++) if (source.adjacent(vertices[u], vertices[v])) result.add(u, v);
285 return result;
286 }
287 case 'CANONIZE': return relabel(source, canon(source).labels);
288 }
289 return new Graph(1);
290}
291function iso(a, b) {
292 if (a.n !== b.n) { output.push(`not isomorphic vertices ${a.n} ${b.n}`); return; }
293 const ma = a.edges().length, mb = b.edges().length;
294 if (ma !== mb) { output.push(`not isomorphic edges ${ma} ${mb}`); return; }
295 const x = invariants(a), y = invariants(b);
296 if (lex(x.degrees, y.degrees)) { output.push('not isomorphic degrees'); return; }
297 const checks = [
298 ['triangles', x.triangles, y.triangles], ['components', x.sizes.length, y.sizes.length],
299 ['girth', x.girth, y.girth], ['diameter', x.diameter, y.diameter], ['bipartite', x.bipartite, y.bipartite]
300 ];
301 for (const [name, p, q] of checks) if (p !== q) { output.push(`not isomorphic ${name} ${p} ${q}`); return; }
302 const r = stable(a), s = stable(b);
303 if (r.rounds !== s.rounds || lex(cells(r.colors).map(cell => cell.length), cells(s.colors).map(cell => cell.length))) {
304 output.push('not isomorphic refinement'); return;
305 }
306 const ca = canon(a), cb = canon(b);
307 if (ca.string !== cb.string) { output.push('not isomorphic canonical'); return; }
308 const inverse = Array(b.n);
309 cb.labels.forEach((label, v) => inverse[label] = v);
310 output.push('isomorphic', `mapping ${ca.labels.map(label => inverse[label]).join(' ')}`);
311}
312function automorphisms(name, graph) {
313 const canonical = canon(graph), edges = graph.edges();
314 const vertices = new DisjointSet(graph.n), edgeOrbits = new DisjointSet(edges.length);
315 const inverse = Array(graph.n);
316 canonical.labels.forEach((label, v) => inverse[label] = v);
317 const edgeIndices = new Map(edges.map(([u, v], i) => [`${u}-${v}`, i]));
318 for (const labels of canonical.bestLeaves) {
319 const sigma = labels.map(label => inverse[label]);
320 sigma.forEach((w, v) => vertices.join(v, w));
321 edges.forEach(([u, v], i) => {
322 const a = Math.min(sigma[u], sigma[v]), b = Math.max(sigma[u], sigma[v]);
323 edgeOrbits.join(i, edgeIndices.get(`${a}-${b}`));
324 });
325 }
326 const vg = vertices.groups(), eg = edgeOrbits.groups();
327 output.push(`aut ${name} order ${canonical.bestLeaves.length}`, `vertex orbits ${vg.length}`);
328 for (const group of vg) output.push(`orbit ${group.join(' ')}`);
329 output.push(`edge orbits ${eg.length}`);
330 for (const group of eg) output.push(`orbit ${group.map(i => edges[i].join('-')).join(' ')}`);
331 output.push(`transitive vertex ${vg.length === 1 ? 'yes' : 'no'} edge ${eg.length === 1 ? 'yes' : 'no'}`);
332}
333function subiso(nameA, nameB, pattern, target) {
334 let embeddings = 0, induced = 0;
335 const first = [], mapping = [];
336 function search(vertex, used, isInduced) {
337 if (vertex === pattern.n) {
338 embeddings++; if (isInduced) induced++;
339 if (first.length < 5) first.push({ mapping: [...mapping], induced: isInduced });
340 return;
341 }
342 for (let v = 0; v < target.n; v++) if (!(used & (1 << v))) {
343 let valid = true, nextInduced = isInduced;
344 for (let u = 0; u < vertex; u++) {
345 if (pattern.adjacent(u, vertex) && !target.adjacent(mapping[u], v)) { valid = false; break; }
346 if (!pattern.adjacent(u, vertex) && target.adjacent(mapping[u], v)) nextInduced = false;
347 }
348 if (valid) { mapping.push(v); search(vertex + 1, used | (1 << v), nextInduced); mapping.pop(); }
349 }
350 }
351 if (pattern.n <= target.n) search(0, 0, true);
352 const order = canon(pattern).bestLeaves.length;
353 output.push(`subiso ${nameA} ${nameB} embeddings ${embeddings} induced ${induced}`, `copies ${embeddings / order} induced ${induced / order}`);
354 for (const item of first) output.push(`embedding ${item.mapping.join(' ')}${item.induced ? ' induced' : ''}`);
355}
356function classify(n) {
357 if (!n) fail('bad value');
358 if (n > 5) fail('too large');
359 const levels = [[canon(new Graph(n)).string]];
360 const max = n * (n - 1) / 2;
361 for (let m = 0; m < max; m++) {
362 const next = new Set();
363 for (const string of levels[m]) {
364 const graph = new Graph(n);
365 let i = 0;
366 for (let u = 0; u < n; u++) for (let v = u + 1; v < n; v++) if (string[i++] === '1') graph.add(u, v);
367 for (let u = 0; u < n; u++) for (let v = u + 1; v < n; v++) if (!graph.adjacent(u, v)) {
368 const added = relabel(graph, Array.from({ length: n }, (_, vertex) => vertex));
369 added.add(u, v); next.add(canon(added).string);
370 }
371 }
372 levels.push([...next].sort());
373 }
374 output.push(`classify ${n} total ${levels.reduce((sum, level) => sum + level.length, 0)}`);
375 levels.forEach((level, m) => output.push(`edges ${m} count ${level.length} ${level.map(string => string || '-').join(' ')}`));
376}
377function execute(line) {
378 const [command, ...args] = line.split(' '), format = formats.get(command);
379 if (!format) fail(`unknown command ${command}`);
380 if (args.length < format.fixed.length || !format.rest && args.length !== format.fixed.length) fail('bad arguments');
381 for (let i = 0; i < args.length; i++) if (!valid(args[i], format.fixed[i] ?? format.rest)) fail('bad arguments');
382 format.fixed.forEach((kind, i) => { if (kind === 'graph' && !registry.has(args[i])) fail(`no graph ${args[i]}`); });
383 const name = args[0], graph = registry.get(name);
384 if (format.fixed[0] === 'name') {
385 const result = define(command, args), redefined = registry.has(name);
386 registry.set(name, result);
387 output.push(`${redefined ? 'redefined' : 'defined'} ${name} n ${result.n} m ${result.edges().length}`);
388 return;
389 }
390 switch (command) {
391 case 'DELETE': registry.delete(name); output.push(`deleted ${name}`); break;
392 case 'LIST':
393 output.push(`graphs ${registry.size}`);
394 for (const key of [...registry.keys()].sort()) {
395 const item = registry.get(key);
396 output.push(`${key} n ${item.n} m ${item.edges().length}`);
397 }
398 break;
399 case 'SHOW':
400 output.push(`graph ${name} n ${graph.n} m ${graph.edges().length}`);
401 for (let v = 0; v < graph.n; v++) { const neighbors = graph.neighbors(v); output.push(`${v}:${neighbors.length ? ` ${neighbors.join(' ')}` : ''}`); }
402 break;
403 case 'INVARIANTS': {
404 const value = invariants(graph);
405 output.push(`invariants ${name}`, `degrees ${value.degrees.join(' ')}`, `edges ${graph.edges().length} triangles ${value.triangles}`,
406 `components ${value.sizes.length} sizes ${value.sizes.join(' ')}`, `girth ${value.girth} diameter ${value.diameter} radius ${value.radius}`, `bipartite ${value.bipartite}`);
407 if (graph.n > 12) output.push('clique -', 'chromatic -');
408 else {
409 const largest = clique(graph), colors = coloring(graph, largest.length);
410 output.push(`clique ${largest.length} ${largest.join(' ')}`, `chromatic ${Math.max(...colors) + 1} ${colors.join(' ')}`);
411 }
412 break;
413 }
414 case 'REFINE': {
415 const vertices = args.slice(1).map(Number);
416 if (vertices.some(v => v >= graph.n)) fail('bad vertices');
417 let { colors, rounds } = stable(graph);
418 for (const vertex of vertices) {
419 if (colors.filter(color => color === colors[vertex]).length === 1) fail(`singleton ${vertex}`);
420 const next = refine(graph, individualize(colors, vertex));
421 colors = next.colors; rounds += next.rounds;
422 }
423 const partition = cells(colors);
424 output.push(`refine ${name} rounds ${rounds} cells ${partition.length}`);
425 partition.forEach((cell, color) => output.push(`${color}: ${cell.join(' ')}`));
426 break;
427 }
428 case 'CANON': {
429 const result = canon(graph);
430 output.push(`canon ${name} nodes ${result.nodes} leaves ${result.leaves}`, `labels ${result.labels.join(' ')}`, `string ${result.string || '-'}`);
431 break;
432 }
433 case 'ISO': iso(graph, registry.get(args[1])); break;
434 case 'AUT': automorphisms(name, graph); break;
435 case 'SUBISO': subiso(name, args[1], graph, registry.get(args[1])); break;
436 case 'CLASSIFY': classify(Number(name)); break;
437 }
438}
439const lines = fs.readFileSync(0, 'utf8').split(/\r?\n/), count = Number(lines[0]);
440for (const line of lines.slice(1, count + 1)) {
441 try { execute(line); }
442 catch (error) { output.push(error.message); }
443}
444if (output.length) process.stdout.write(output.join('\n') + '\n');
