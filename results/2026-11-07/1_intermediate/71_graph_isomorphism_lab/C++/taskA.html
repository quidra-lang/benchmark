14struct Graph {
15 int n;
16 std::vector<std::vector<bool>> adjacent;
17
18 explicit Graph(int vertices) : n(vertices), adjacent(n, std::vector<bool>(n, false)) {}
19
20 void add(int u, int v) {
21 adjacent[u][v] = adjacent[v][u] = true;
22 }
23
24 std::vector<std::pair<int, int>> edges() const {
25 std::vector<std::pair<int, int>> result;
26 for (int u = 0; u < n; ++u) {
27 for (int v = u + 1; v < n; ++v) {
28 if (adjacent[u][v]) result.emplace_back(u, v);
29 }
30 }
31 return result;
32 }
33
34 std::vector<int> degrees() const {
35 std::vector<int> result(n);
36 for (int v = 0; v < n; ++v) {
37 result[v] = static_cast<int>(std::count(adjacent[v].begin(), adjacent[v].end(), true));
38 }
39 return result;
40 }
41};

70static std::vector<int> degreeColors(const Graph& graph) {
71 std::vector<int> degrees = graph.degrees();
72 std::vector<int> unique = degrees;
73 std::sort(unique.begin(), unique.end());
74 unique.erase(std::unique(unique.begin(), unique.end()), unique.end());
75 for (int& degree : degrees) degree = static_cast<int>(std::lower_bound(unique.begin(), unique.end(), degree) - unique.begin());
76 return degrees;
77}

79struct Refinement {
80 std::vector<int> colors;
81 int rounds;
82};

84static Refinement refine(const Graph& graph, std::vector<int> colors) {
85 int rounds = 0;
86 using Signature = std::pair<int, std::vector<int>>;
87 while (true) {
88 std::vector<Signature> signatures;
89 for (int v = 0; v < graph.n; ++v) {
90 std::vector<int> neighbors;
91 for (int w = 0; w < graph.n; ++w) {
92 if (graph.adjacent[v][w]) neighbors.push_back(colors[w]);
93 }
94 std::sort(neighbors.begin(), neighbors.end());
95 signatures.emplace_back(colors[v], neighbors);
96 }
97 std::vector<Signature> unique = signatures;
98 std::sort(unique.begin(), unique.end());
99 unique.erase(std::unique(unique.begin(), unique.end()), unique.end());
100 if (static_cast<int>(unique.size()) == cellCount(colors)) return {colors, rounds};
101 for (int v = 0; v < graph.n; ++v) {
102 colors[v] = static_cast<int>(std::lower_bound(unique.begin(), unique.end(), signatures[v]) - unique.begin());
103 }
104 ++rounds;
105 }
106}

108static std::vector<int> individualize(std::vector<int> colors, int vertex) {
109 int chosen = colors[vertex];
110 for (std::size_t v = 0; v < colors.size(); ++v) {
111 if (static_cast<int>(v) != vertex && colors[v] >= chosen) ++colors[v];
112 }
113 return colors;
114}

116struct CanonicalForm {
117 long long nodes = 0;
118 long long leaves = 0;
119 std::string bits;
120 std::vector<int> labels;
121 std::vector<std::vector<int>> bestLeaves;
122
123 explicit CanonicalForm(const Graph& graph) {
124 visit(graph, refine(graph, degreeColors(graph)).colors);
125 }
126
127private:
128 void visit(const Graph& graph, const std::vector<int>& colors) {
129 ++nodes;
130 int count = cellCount(colors);
131 if (count == graph.n) {
132 ++leaves;
133 std::vector<int> vertex = inverse(colors);
134 std::string candidate;
135 for (int i = 0; i < graph.n; ++i) {
136 for (int j = i + 1; j < graph.n; ++j) candidate += graph.adjacent[vertex[i]][vertex[j]] ? '1' : '0';
137 }
138 if (labels.empty() || candidate > bits) {
139 bits = candidate;
140 labels = colors;
141 bestLeaves.clear();
142 }
143 if (candidate == bits) bestLeaves.push_back(colors);
144 return;
145 }
146 std::vector<std::vector<int>> cells(count);
147 for (int v = 0; v < graph.n; ++v) cells[colors[v]].push_back(v);
148 int chosen = -1;
149 for (int color = 0; color < count; ++color) {
150 if (cells[color].size() > 1 && (chosen < 0 || cells[color].size() < cells[chosen].size())) chosen = color;
151 }
152 for (int v : cells[chosen]) visit(graph, refine(graph, individualize(colors, v)).colors);
153 }
154};

173static Invariants invariants(const Graph& graph) {
174 Invariants result;
175 result.degrees = graph.degrees();
176 std::sort(result.degrees.rbegin(), result.degrees.rend());
177 result.edges = static_cast<int>(graph.edges().size());
178 for (int u = 0; u < graph.n; ++u) {
179 for (int v = u + 1; v < graph.n; ++v) {
180 for (int w = v + 1; w < graph.n; ++w) {
181 if (graph.adjacent[u][v] && graph.adjacent[v][w] && graph.adjacent[w][u]) ++result.triangles;
182 }
183 }
184 }
185 std::vector<int> colors(graph.n, -1);
186 for (int root = 0; root < graph.n; ++root) {
187 if (colors[root] >= 0) continue;
188 int size = 0;
189 std::queue<int> pending;
190 colors[root] = 0;
191 pending.push(root);
192 while (!pending.empty()) {
193 int u = pending.front();
194 pending.pop();
195 ++size;
196 for (int v = 0; v < graph.n; ++v) {
197 if (!graph.adjacent[u][v]) continue;
198 if (colors[v] < 0) {
199 colors[v] = 1 - colors[u];
200 pending.push(v);
201 } else if (colors[v] == colors[u]) result.bipartite = false;
202 }
203 }
204 result.componentSizes.push_back(size);
205 }
206 std::sort(result.componentSizes.rbegin(), result.componentSizes.rend());
207 result.radius = graph.n;
208 for (int root = 0; root < graph.n; ++root) {
209 std::vector<int> distance(graph.n, -1);
210 std::vector<int> parent(graph.n, -1);
211 std::queue<int> pending;
212 distance[root] = 0;
213 pending.push(root);
214 while (!pending.empty()) {
215 int u = pending.front();
216 pending.pop();
217 for (int v = 0; v < graph.n; ++v) {
218 if (!graph.adjacent[u][v]) continue;
219 if (distance[v] < 0) {
220 distance[v] = distance[u] + 1;
221 parent[v] = u;
222 pending.push(v);
223 } else if (parent[u] != v && parent[v] != u) {
224 int cycle = distance[u] + distance[v] + 1;
225 if (result.girth == 0 || cycle < result.girth) result.girth = cycle;
226 }
227 }
228 }
229 int eccentricity = *std::max_element(distance.begin(), distance.end());
230 result.radius = std::min(result.radius, eccentricity);
231 result.diameter = std::max(result.diameter, eccentricity);
232 }
233 if (result.componentSizes.size() > 1) result.diameter = result.radius = -1;
234 return result;
235}

237static std::vector<int> largestClique(const Graph& graph) {
238 std::vector<int> best;
239 for (int mask = 1; mask < (1 << graph.n); ++mask) {
240 std::vector<int> vertices;
241 bool valid = true;
242 for (int v = 0; v < graph.n && valid; ++v) {
243 if ((mask & (1 << v)) == 0) continue;
244 for (int u : vertices) {
245 if (!graph.adjacent[u][v]) valid = false;
246 }
247 vertices.push_back(v);
248 }
249 if (valid && (vertices.size() > best.size() || (vertices.size() == best.size() && vertices < best))) best = vertices;
250 }
251 return best;
252}

254static std::pair<int, std::vector<int>> coloring(const Graph& graph) {
255 std::vector<int> colors(graph.n, -1);
256 for (int limit = 1; limit <= graph.n; ++limit) {
257 std::function<bool(int, int)> assign = [&](int vertex, int highest) {
258 if (vertex == graph.n) return true;
259 for (int color = 0; color < limit && color <= highest + 1; ++color) {
260 bool valid = true;
261 for (int u = 0; u < vertex; ++u) {
262 if (graph.adjacent[u][vertex] && colors[u] == color) valid = false;
263 }
264 if (!valid) continue;
265 colors[vertex] = color;
266 if (assign(vertex + 1, std::max(highest, color))) return true;
267 colors[vertex] = -1;
268 }
269 return false;
270 };
271 if (assign(0, -1)) return {limit, colors};
272 }
273 return {graph.n, colors};
274}

310static void isomorphism(const Graph& a, const Graph& b) {
311 std::string difference;
312 if (a.n != b.n) difference = "vertices " + std::to_string(a.n) + ' ' + std::to_string(b.n);
313 else {
314 Invariants x = invariants(a);
315 Invariants y = invariants(b);
316 if (x.edges != y.edges) difference = "edges " + std::to_string(x.edges) + ' ' + std::to_string(y.edges);
317 else if (x.degrees != y.degrees) difference = "degrees";
318 else if (x.triangles != y.triangles) difference = "triangles " + std::to_string(x.triangles) + ' ' + std::to_string(y.triangles);
319 else if (x.componentSizes.size() != y.componentSizes.size()) difference = "components " + std::to_string(x.componentSizes.size()) + ' ' + std::to_string(y.componentSizes.size());
320 else if (x.girth != y.girth) difference = "girth " + girthText(x.girth) + ' ' + girthText(y.girth);
321 else if (x.diameter != y.diameter) difference = "diameter " + distanceText(x.diameter) + ' ' + distanceText(y.diameter);
322 else if (x.bipartite != y.bipartite) difference = "bipartite " + yesNo(x.bipartite) + ' ' + yesNo(y.bipartite);
323 else {
324 Refinement ra = refine(a, degreeColors(a));
325 Refinement rb = refine(b, degreeColors(b));
326 if (ra.rounds != rb.rounds || cellSizes(ra.colors) != cellSizes(rb.colors)) difference = "refinement";
327 else {
328 CanonicalForm ca(a);
329 CanonicalForm cb(b);
330 if (ca.bits != cb.bits) difference = "canonical";
331 else {
332 std::vector<int> vertex = inverse(cb.labels);
333 std::cout << "isomorphic\nmapping";
334 for (int label : ca.labels) std::cout << ' ' << vertex[label];
335 std::cout << '\n';
336 return;
337 }
338 }
339 }
340 }
341 std::cout << "not isomorphic " << difference << '\n';
342}

344class DisjointSet {
345 std::vector<int> parent;
346public:
347 explicit DisjointSet(int size) : parent(size) {
348 std::iota(parent.begin(), parent.end(), 0);
349 }
350 int find(int value) {
351 return parent[value] == value ? value : parent[value] = find(parent[value]);
352 }
353 void join(int a, int b) {
354 parent[find(b)] = find(a);
355 }
356 std::vector<std::vector<int>> groups() {
357 std::map<int, std::vector<int>> cells;
358 for (std::size_t i = 0; i < parent.size(); ++i) cells[find(static_cast<int>(i))].push_back(static_cast<int>(i));
359 std::vector<std::vector<int>> result;
360 for (const auto& [root, cell] : cells) result.push_back(cell);
361 std::sort(result.begin(), result.end());
362 return result;
363 }
364};

366static void automorphisms(const std::string& name, const Graph& graph) {
367 CanonicalForm canonical(graph);
368 std::vector<int> vertex = inverse(canonical.labels);
369 std::vector<std::pair<int, int>> edges = graph.edges();
370 std::map<std::pair<int, int>, int> edgeIds;
371 for (std::size_t i = 0; i < edges.size(); ++i) edgeIds[edges[i]] = static_cast<int>(i);
372 DisjointSet vertexSets(graph.n);
373 DisjointSet edgeSets(static_cast<int>(edges.size()));
374 for (const std::vector<int>& labels : canonical.bestLeaves) {
375 std::vector<int> permutation(graph.n);
376 for (int v = 0; v < graph.n; ++v) {
377 permutation[v] = vertex[labels[v]];
378 vertexSets.join(v, permutation[v]);
379 }
380 for (std::size_t i = 0; i < edges.size(); ++i) {
381 int a = permutation[edges[i].first];
382 int b = permutation[edges[i].second];
383 if (a > b) std::swap(a, b);
384 edgeSets.join(static_cast<int>(i), edgeIds.at({a, b}));
385 }
386 }
387 std::vector<std::vector<int>> vertices = vertexSets.groups();
388 std::vector<std::vector<int>> edgeGroups = edgeSets.groups();
389 std::cout << "aut " << name << " order " << canonical.bestLeaves.size() << "\nvertex orbits " << vertices.size() << '\n';
390 for (const std::vector<int>& group : vertices) {
391 std::cout << "orbit";
392 sequence(group);
393 std::cout << '\n';
394 }
395 std::cout << "edge orbits " << edgeGroups.size() << '\n';
396 for (const std::vector<int>& group : edgeGroups) {
397 std::cout << "orbit";
398 for (int edge : group) std::cout << ' ' << edges[edge].first << '-' << edges[edge].second;
399 std::cout << '\n';
400 }
401 std::cout << "transitive vertex " << yesNo(vertices.size() == 1) << " edge " << yesNo(edgeGroups.size() == 1) << '\n';
402}

404static void subisomorphism(const std::string& patternName, const std::string& targetName,
405 const Graph& pattern, const Graph& target) {
406 long long embeddings = 0;
407 long long induced = 0;
408 std::vector<std::pair<std::vector<int>, bool>> first;
409 std::vector<int> mapping(pattern.n, -1);
410 std::vector<bool> used(target.n, false);
411 std::function<void(int, bool)> search = [&](int vertex, bool exact) {
412 if (vertex == pattern.n) {
413 ++embeddings;
414 if (exact) ++induced;
415 if (first.size() < 5) first.emplace_back(mapping, exact);
416 return;
417 }
418 for (int v = 0; v < target.n; ++v) {
419 if (used[v]) continue;
420 bool valid = true;
421 bool nextExact = exact;
422 for (int u = 0; u < vertex; ++u) {
423 bool edge = target.adjacent[mapping[u]][v];
424 if (pattern.adjacent[u][vertex] && !edge) valid = false;
425 if (pattern.adjacent[u][vertex] != edge) nextExact = false;
426 }
427 if (!valid) continue;
428 mapping[vertex] = v;
429 used[v] = true;
430 search(vertex + 1, nextExact);
431 used[v] = false;
432 }
433 };
434 if (pattern.n <= target.n) search(0, true);
435 long long order = static_cast<long long>(CanonicalForm(pattern).bestLeaves.size());
436 std::cout << "subiso " << patternName << ' ' << targetName << " embeddings " << embeddings << " induced " << induced
437 << "\ncopies " << embeddings / order << " induced " << induced / order << '\n';
438 for (const auto& [values, exact] : first) {
439 std::cout << "embedding";
440 sequence(values);
441 if (exact) std::cout << " induced";
442 std::cout << '\n';
443 }
444}

446static void classify(int n) {
447 range(n, 1, 5);
448 int possible = n * (n - 1) / 2;
449 std::vector<std::set<std::string>> classes(possible + 1);
450 for (int mask = 0; mask < (1 << possible); ++mask) {
451 Graph graph(n);
452 int bit = 0;
453 int count = 0;
454 for (int u = 0; u < n; ++u) {
455 for (int v = u + 1; v < n; ++v, ++bit) {
456 if (mask & (1 << bit)) {
457 graph.add(u, v);
458 ++count;
459 }
460 }
461 }
462 classes[count].insert(CanonicalForm(graph).bits);
463 }
464 std::size_t total = 0;
465 for (const auto& group : classes) total += group.size();
466 std::cout << "classify " << n << " total " << total << '\n';
467 for (int m = 0; m <= possible; ++m) {
468 std::cout << "edges " << m << " count " << classes[m].size();
469 for (const std::string& bits : classes[m]) std::cout << ' ' << (bits.empty() ? "-" : bits);
470 std::cout << '\n';
471 }
472}

481static bool nameWord(const std::string& text) {
482 return !text.empty() && text.size() <= 8 && text[0] >= 'a' && text[0] <= 'z' &&
483 text.find_first_not_of("abcdefghijklmnopqrstuvwxyz0123456789") == std::string::npos;

486static bool numberWord(const std::string& text) {
487 return !text.empty() && text.size() <= 3 && (text.size() == 1 || text[0] != '0') &&
488 text.find_first_not_of("0123456789") == std::string::npos;
489}

491static std::pair<int, int> edgeWord(const std::string& text) {
492 std::size_t dash = text.find('-');
493 if (dash == std::string::npos || !numberWord(text.substr(0, dash)) || !numberWord(text.substr(dash + 1))) fail("bad arguments");
494 return {std::stoi(text.substr(0, dash)), std::stoi(text.substr(dash + 1))};
495}

512 Graph build(const std::string& command, const std::vector<std::string>& a) const {
513 if (command == "GRAPH") {
514 int n = std::stoi(a[1]);
515 range(n, 1, 16);
516 Graph graph(n);
517 for (std::size_t i = 2; i < a.size(); ++i) {
518 auto [u, v] = edgeWord(a[i]);
519 if (u >= n || v >= n || u == v) fail("bad edge " + a[i]);
520 if (graph.adjacent[u][v]) fail("duplicate edge " + a[i]);
521 graph.add(u, v);
522 }
523 return graph;
524 }
525 if (command == "CYCLE" || command == "PATH" || command == "COMPLETE" || command == "CIRCULANT") {
526 int n = std::stoi(a[1]);
527 range(n, command == "CYCLE" || command == "CIRCULANT" ? 3 : 1, 16);
528 Graph graph(n);
529 if (command == "COMPLETE") {
530 for (int u = 0; u < n; ++u) {
531 for (int v = u + 1; v < n; ++v) graph.add(u, v);
532 }
533 } else if (command == "CIRCULANT") {
534 int previous = 0;
535 for (std::size_t i = 2; i < a.size(); ++i) {
536 int jump = std::stoi(a[i]);
537 if (jump <= previous || 2 * jump > n) fail("bad value");
538 previous = jump;
539 for (int v = 0; v < n; ++v) graph.add(v, (v + jump) % n);
540 }
541 } else {
542 for (int v = 1; v < n; ++v) graph.add(v - 1, v);
543 if (command == "CYCLE") graph.add(n - 1, 0);
544 }
545 return graph;
546 }
547 if (command == "BIPARTITE" || command == "GRID") {
548 int first = std::stoi(a[1]);
549 int second = std::stoi(a[2]);
550 if (first == 0 || second == 0) fail("bad value");
551 int n = command == "GRID" ? first * second : first + second;
552 range(n, 1, 16);
553 Graph graph(n);
554 for (int x = 0; x < first; ++x) {
555 for (int y = 0; y < second; ++y) {
556 if (command == "BIPARTITE") graph.add(x, first + y);
557 else {
558 if (x + 1 < first) graph.add(x * second + y, (x + 1) * second + y);
559 if (y + 1 < second) graph.add(x * second + y, x * second + y + 1);
560 }
561 }
562 }
563 return graph;
564 }
565 if (command == "STAR" || command == "WHEEL") {
566 int k = std::stoi(a[1]);
567 range(k, command == "STAR" ? 1 : 3, 15);
568 Graph graph(k + 1);
569 for (int v = 1; v <= k; ++v) graph.add(0, v);
570 if (command == "WHEEL") {
571 for (int v = 1; v <= k; ++v) graph.add(v, v % k + 1);
572 }
573 return graph;
574 }
575 if (command == "PETERSEN") {
576 Graph graph(10);
577 for (int v = 0; v < 5; ++v) {
578 graph.add(v, (v + 1) % 5);
579 graph.add(v, v + 5);
580 graph.add(v + 5, (v + 2) % 5 + 5);
581 }
582 return graph;
583 }
584 if (command == "HYPERCUBE") {
585 int d = std::stoi(a[1]);
586 range(d, 1, 4);
587 Graph graph(1 << d);
588 for (int v = 0; v < graph.n; ++v) {
589 for (int bit = 0; bit < d; ++bit) graph.add(v, v ^ (1 << bit));
590 }
591 return graph;
592 }
593 const Graph& source = graphs.at(a[1]);
594 if (command == "COMPLEMENT") {
595 Graph graph(source.n);
596 for (int u = 0; u < source.n; ++u) {
597 for (int v = u + 1; v < source.n; ++v) {
598 if (!source.adjacent[u][v]) graph.add(u, v);
599 }
600 }
601 return graph;
602 }
603 if (command == "LINE") {
604 std::vector<std::pair<int, int>> edges = source.edges();
605 if (edges.empty()) fail("no edges");
606 range(static_cast<int>(edges.size()), 1, 16);
607 Graph graph(static_cast<int>(edges.size()));
608 for (int u = 0; u < graph.n; ++u) {
609 for (int v = u + 1; v < graph.n; ++v) {
610 if (edges[u].first == edges[v].first || edges[u].first == edges[v].second ||
611 edges[u].second == edges[v].first || edges[u].second == edges[v].second) graph.add(u, v);
612 }
613 }
614 return graph;
615 }
616 if (command == "PRODUCT" || command == "UNION") {
617 const Graph& second = graphs.at(a[2]);
618 int n = command == "PRODUCT" ? source.n * second.n : source.n + second.n;
619 range(n, 1, 16);
620 Graph graph(n);
621 if (command == "UNION") {
622 for (const auto& [u, v] : source.edges()) graph.add(u, v);
623 for (const auto& [u, v] : second.edges()) graph.add(u + source.n, v + source.n);
624 } else {
625 for (const auto& [u, v] : source.edges()) {
626 for (int y = 0; y < second.n; ++y) graph.add(u * second.n + y, v * second.n + y);
627 }
628 for (const auto& [u, v] : second.edges()) {
629 for (int x = 0; x < source.n; ++x) graph.add(x * second.n + u, x * second.n + v);
630 }
631 }
632 return graph;
633 }
634 if (command == "CANONIZE") return relabel(source, CanonicalForm(source).labels);
635 std::vector<int> values;
636 for (std::size_t i = 2; i < a.size(); ++i) values.push_back(std::stoi(a[i]));
637 if (command == "RELABEL") {
638 std::vector<int> sorted = values;
639 std::sort(sorted.begin(), sorted.end());
640 std::vector<int> expected(source.n);
641 std::iota(expected.begin(), expected.end(), 0);
642 if (sorted != expected) fail("bad permutation");
643 return relabel(source, values);
644 }
645 std::set<int> unique(values.begin(), values.end());
646 if (unique.size() != values.size() || *unique.rbegin() >= source.n) fail("bad vertices");
647 Graph graph(static_cast<int>(values.size()));
648 for (int u = 0; u < graph.n; ++u) {
649 for (int v = u + 1; v < graph.n; ++v) {
650 if (source.adjacent[values[u]][values[v]]) graph.add(u, v);
651 }
652 }
653 return graph;
654 }
655
656public:
657 void execute(const std::string& line) {
658 std::istringstream input(line);
659 std::string command;
660 input >> command;
661 if (!formats.count(command)) fail("unknown command " + command);
662 std::vector<std::string> arguments;
663 for (std::string word; input >> word;) arguments.push_back(word);
664 const Format& format = formats.at(command);
665 int count = static_cast<int>(arguments.size());
666 if (count < format.minimum || (format.maximum >= 0 && count > format.maximum)) fail("bad arguments");
667 for (std::size_t i = 0; i < arguments.size(); ++i) {
668 char kind = i < format.fixed.size() ? format.fixed[i] : format.tail;
669 if ((kind == 'n' || kind == 'g') && !nameWord(arguments[i])) fail("bad arguments");
670 if (kind == 'i' && !numberWord(arguments[i])) fail("bad arguments");
671 if (kind == 'e') edgeWord(arguments[i]);
672 }
673 for (std::size_t i = 0; i < format.fixed.size(); ++i) {
674 if (format.fixed[i] == 'g' && !graphs.count(arguments[i])) fail("no graph " + arguments[i]);
675 }
676 if (!format.fixed.empty() && format.fixed[0] == 'n') {
677 Graph graph = build(command, arguments);
678 bool existed = graphs.count(arguments[0]) != 0;
679 graphs.insert_or_assign(arguments[0], graph);
680 std::cout << (existed ? "redefined " : "defined ") << arguments[0] << " n " << graph.n << " m " << graph.edges().size() << '\n';
681 return;
682 }
683 if (command == "LIST") {
684 std::cout << "graphs " << graphs.size() << '\n';
685 for (const auto& [name, graph] : graphs) std::cout << name << " n " << graph.n << " m " << graph.edges().size() << '\n';
686 return;
687 }
688 if (command == "CLASSIFY") {
689 classify(std::stoi(arguments[0]));
690 return;
691 }
692 const std::string& name = arguments[0];
693 const Graph& graph = graphs.at(name);
694 if (command == "DELETE") {
695 graphs.erase(name);
696 std::cout << "deleted " << name << '\n';
697 } else if (command == "SHOW") {
698 std::cout << "graph " << name << " n " << graph.n << " m " << graph.edges().size() << '\n';
699 for (int v = 0; v < graph.n; ++v) {
700 std::cout << v << ':';
701 for (int w = 0; w < graph.n; ++w) {
702 if (graph.adjacent[v][w]) std::cout << ' ' << w;
703 }
704 std::cout << '\n';
705 }
706 } else if (command == "INVARIANTS") printInvariants(name, graph);
707 else if (command == "REFINE") {
708 for (std::size_t i = 1; i < arguments.size(); ++i) {
709 if (std::stoi(arguments[i]) >= graph.n) fail("bad vertices");
710 }
711 Refinement result = refine(graph, degreeColors(graph));
712 for (std::size_t i = 1; i < arguments.size(); ++i) {
713 int vertex = std::stoi(arguments[i]);
714 if (std::count(result.colors.begin(), result.colors.end(), result.colors[vertex]) == 1) fail("singleton " + arguments[i]);
715 Refinement next = refine(graph, individualize(result.colors, vertex));
716 result.colors = std::move(next.colors);
717 result.rounds += next.rounds;
718 }
719 std::cout << "refine " << name << " rounds " << result.rounds << " cells " << cellCount(result.colors) << '\n';
720 for (int color = 0; color < cellCount(result.colors); ++color) {
721 std::cout << color << ':';
722 for (int v = 0; v < graph.n; ++v) {
723 if (result.colors[v] == color) std::cout << ' ' << v;
724 }
725 std::cout << '\n';
726 }
727 } else if (command == "CANON") {
728 CanonicalForm result(graph);
729 std::cout << "canon " << name << " nodes " << result.nodes << " leaves " << result.leaves << "\nlabels";
730 sequence(result.labels);
731 std::cout << "\nstring " << (result.bits.empty() ? "-" : result.bits) << '\n';
732 } else if (command == "ISO") isomorphism(graph, graphs.at(arguments[1]));
733 else if (command == "AUT") automorphisms(name, graph);
734 else if (command == "SUBISO") subisomorphism(name, arguments[1], graph, graphs.at(arguments[1]));
735 }
736};
737
738int main() {
739 int count;
740 std::cin >> count;
741 std::string line;
742 std::getline(std::cin, line);
743 Laboratory laboratory;
744 while (count-- > 0 && std::getline(std::cin, line)) {
745 try {
746 laboratory.execute(line);
747 } catch (const std::runtime_error& error) {
748 std::cout << "error: " << error.what() << '\n';
749 }
750 }
