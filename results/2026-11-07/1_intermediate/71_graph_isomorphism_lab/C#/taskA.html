21 public Graph(Graph graph)
22 {
23 N = graph.N;
24 Adjacent = (int[])graph.Adjacent.Clone();
25 }
26 public bool Edge(int u, int v) => (Adjacent[u] & 1 << v) != 0;
27 public void Join(int u, int v)
28 {
29 Adjacent[u] |= 1 << v;
30 Adjacent[v] |= 1 << u;
31 }
32 public int Degree(int vertex) => BitOperations.PopCount((uint)Adjacent[vertex]);
33 public List<(int U, int V)> Edges()
34 {
35 List<(int, int)> result = [];
36 for (int u = 0; u < N; u++) for (int v = u + 1; v < N; v++) if (Edge(u, v)) result.Add((u, v));
37 return result;
38 }

91 static Refined Refine(Graph graph, int[] initial)
92 {
93 int[] colors = (int[])initial.Clone();
94 int rounds = 0;
95 while (true)
96 {
97 int[][] signatures = new int[graph.N][];
98 for (int v = 0; v < graph.N; v++)
99 {
100 List<int> neighbors = [];
101 for (int w = 0; w < graph.N; w++) if (graph.Edge(v, w)) neighbors.Add(colors[w]);
102 neighbors.Sort();
103 signatures[v] = [colors[v], .. neighbors];
104 }
105 int[] order = Enumerable.Range(0, graph.N).OrderBy(vertex => signatures[vertex], Comparer<int[]>.Create(Compare)).ToArray(), next = new int[graph.N];
106 int color = 0;
107 for (int i = 0; i < graph.N; i++)
108 {
109 if (i > 0 && Compare(signatures[order[i - 1]], signatures[order[i]]) != 0) color++;
110 next[order[i]] = color;
111 }
112 if (Cells(next) == Cells(colors)) return new(colors, rounds);
113 colors = next;
114 rounds++;
115 }
116 }

134 static void CanonicalSearch(Graph graph, int[] colors, Canonical result)
135 {
136 result.Nodes++;
137 if (Cells(colors) == graph.N)
138 {
139 result.Leaves++;

141 int order = result.Text is null ? 1 : StringComparer.Ordinal.Compare(text, result.Text);
142 if (order > 0)
143 {
144 result.Text = text;
145 result.Labels = (int[])colors.Clone();
146 result.Maxima.Clear();

148 if (order >= 0) result.Maxima.Add((int[])colors.Clone());
149 return;

156 chosen = color;
157 smallest = sizes[color];
158 }
159 for (int vertex = 0; vertex < graph.N; vertex++) if (colors[vertex] == chosen) CanonicalSearch(graph, Refine(graph, Individualize(colors, vertex)).Colors, result);

169 static Invariant Invariants(Graph graph)
170 {
171 int[] degrees = Enumerable.Range(0, graph.N).Select(graph.Degree).OrderDescending().ToArray();
172 int triangles = 0;
173 for (int a = 0; a < graph.N; a++) for (int b = a + 1; b < graph.N; b++) for (int c = b + 1; c < graph.N; c++) if (graph.Edge(a, b) && graph.Edge(a, c) && graph.Edge(b, c)) triangles++;
174 int[][] distances = new int[graph.N][];
175 bool[] visited = new bool[graph.N];
176 List<int> sizes = [];
177 int girth = int.MaxValue;
178 bool bipartite = true;
179 for (int source = 0; source < graph.N; source++)
180 {
181 int[] d = Enumerable.Repeat(-1, graph.N).ToArray(), parents = Enumerable.Repeat(-1, graph.N).ToArray();
182 distances[source] = d;
183 Queue<int> queue = new();
184 queue.Enqueue(source);
185 d[source] = 0;
186 int size = 0;
187 while (queue.Count > 0)
188 {
189 int u = queue.Dequeue();
190 size++;
191 for (int v = 0; v < graph.N; v++) if (graph.Edge(u, v))
192 {
193 if (d[v] < 0)
194 {
195 d[v] = d[u] + 1;
196 parents[v] = u;
197 queue.Enqueue(v);
198 }
199 else if (parents[u] != v && parents[v] != u) girth = Math.Min(girth, d[u] + d[v] + 1);
200 if (d[u] % 2 == d[v] % 2) bipartite = false;
201 }
202 }
203 if (!visited[source])
204 {
205 sizes.Add(size);
206 for (int i = 0; i < graph.N; i++) if (d[i] >= 0) visited[i] = true;
207 }
208 }
209 sizes.Sort((a, b) => b.CompareTo(a));
210 int diameter = 0, radius = graph.N;
211 if (sizes.Count == 1) foreach (int[] row in distances)
212 {
213 int eccentricity = row.Max();
214 diameter = Math.Max(diameter, eccentricity);
215 radius = Math.Min(radius, eccentricity);
216 }
217 return new(degrees, graph.Edges().Count, triangles, sizes, girth == int.MaxValue ? "none" : girth.ToString(), sizes.Count == 1 ? diameter.ToString() : "INF", sizes.Count == 1 ? radius.ToString() : "INF", bipartite ? "yes" : "no");
218 }

220 static bool ColorSearch(Graph graph, int count, int vertex, int largest, int[] colors)
221 {
222 if (vertex == graph.N) return true;
223 for (int color = 0; color < count && color <= largest + 1; color++)
224 {
225 bool good = true;
226 for (int other = 0; other < vertex; other++) if (graph.Edge(vertex, other) && colors[other] == color) good = false;
227 if (good)
228 {
229 colors[vertex] = color;
230 if (ColorSearch(graph, count, vertex + 1, Math.Max(largest, color), colors)) return true;
231 }
232 }
233 return false;
234 }

236 static void PrintInvariants(string name, Graph graph)
237 {
238 var data = Invariants(graph);
239 Console.WriteLine("invariants " + name);
240 Console.WriteLine("degrees " + Numbers(data.Degrees));
241 Console.WriteLine($"edges {data.Edges} triangles {data.Triangles}");
242 Console.WriteLine($"components {data.Sizes.Count} sizes {Numbers(data.Sizes)}");
243 Console.WriteLine($"girth {data.Girth} diameter {data.Diameter} radius {data.Radius}");
244 Console.WriteLine("bipartite " + data.Bipartite);
245 if (graph.N >= 13)
246 {
247 Console.WriteLine("clique -\nchromatic -");
248 return;
249 }
250 int[] best = [];
251 for (int mask = 1; mask < 1 << graph.N; mask++)
252 {
253 if (BitOperations.PopCount((uint)mask) < best.Length) continue;
254 bool good = true;
255 List<int> selected = [];
256 for (int u = 0; u < graph.N; u++) if ((mask & 1 << u) != 0)
257 {
258 selected.Add(u);
259 for (int v = u + 1; v < graph.N; v++) if ((mask & 1 << v) != 0 && !graph.Edge(u, v)) good = false;
260 }
261 int[] candidate = selected.ToArray();
262 if (good && (candidate.Length > best.Length || Compare(candidate, best) < 0)) best = candidate;
263 }
264 Console.WriteLine($"clique {best.Length} {Numbers(best)}");
265 int[] colors = new int[graph.N];
266 int count = 1;
267 while (!ColorSearch(graph, count, 0, -1, colors)) count++;
268 Console.WriteLine($"chromatic {count} {Numbers(colors)}");
269 }

271 class DisjointSet(int count)
272 {
273 readonly int[] parent = Enumerable.Range(0, count).ToArray();
274 public int Find(int vertex) => parent[vertex] == vertex ? vertex : parent[vertex] = Find(parent[vertex]);
275 public void Join(int left, int right)
276 {
277 left = Find(left);
278 right = Find(right);
279 if (left != right) parent[Math.Max(left, right)] = Math.Min(left, right);
280 }
281 public List<List<int>> Groups()
282 {
283 SortedDictionary<int, List<int>> groups = [];
284 for (int i = 0; i < parent.Length; i++)
285 {
286 int root = Find(i);
287 if (!groups.ContainsKey(root)) groups[root] = [];
288 groups[root].Add(i);
289 }
290 return groups.Values.ToList();
291 }
292 }

294 static void Automorphisms(string name, Graph graph)
295 {
296 Canonical data = Canon(graph);
297 var edges = graph.Edges();
298 DisjointSet vertices = new(graph.N), edgeSets = new(edges.Count);
299 int[] inverse = new int[graph.N];
300 for (int i = 0; i < graph.N; i++) inverse[data.Labels[i]] = i;
301 int[,] edgeIndex = new int[graph.N, graph.N];
302 for (int i = 0; i < edges.Count; i++) edgeIndex[edges[i].U, edges[i].V] = edgeIndex[edges[i].V, edges[i].U] = i;
303 foreach (int[] leaf in data.Maxima)
304 {
305 int[] permutation = new int[graph.N];
306 for (int i = 0; i < graph.N; i++)
307 {
308 permutation[i] = inverse[leaf[i]];
309 vertices.Join(i, permutation[i]);
310 }
311 for (int i = 0; i < edges.Count; i++) edgeSets.Join(i, edgeIndex[permutation[edges[i].U], permutation[edges[i].V]]);
312 }
313 var vertexGroups = vertices.Groups();
314 var edgeGroups = edgeSets.Groups();
315 Console.WriteLine($"aut {name} order {data.Maxima.Count}");
316 Console.WriteLine($"vertex orbits {vertexGroups.Count}");
317 foreach (var group in vertexGroups) Console.WriteLine("orbit " + Numbers(group));
318 Console.WriteLine($"edge orbits {edgeGroups.Count}");
319 foreach (var group in edgeGroups) Console.WriteLine("orbit " + string.Join(" ", group.Select(i => $"{edges[i].U}-{edges[i].V}")));
320 Console.WriteLine($"transitive vertex {(vertexGroups.Count == 1 ? "yes" : "no")} edge {(edgeGroups.Count == 1 ? "yes" : "no")}");
321 }

324 {
325 public long Count, Induced;
326 public List<Embedding> First = [];
327 public void Search(Graph pattern, Graph target, int vertex, int used, int[] map, bool stillInduced)
328 {
329 if (vertex == pattern.N)
330 {
331 Count++;
332 if (stillInduced) Induced++;
333 if (First.Count < 5) First.Add(new((int[])map.Clone(), stillInduced));
334 return;
335 }
336 for (int candidate = 0; candidate < target.N; candidate++) if ((used & 1 << candidate) == 0)
337 {
338 bool good = true, induced = stillInduced;
339 for (int before = 0; before < vertex; before++)
340 {
341 bool a = pattern.Edge(vertex, before), b = target.Edge(candidate, map[before]);
342 if (a && !b) good = false;
343 if (a != b) induced = false;
344 }
345 if (good)
346 {
347 map[vertex] = candidate;
348 Search(pattern, target, vertex + 1, used | 1 << candidate, map, induced);
349 }
350 }

354 static void Isomorphism(Graph left, Graph right)
355 {
356 if (left.N != right.N)
357 {
358 Console.WriteLine($"not isomorphic vertices {left.N} {right.N}");
359 return;
360 }
361 var a = Invariants(left);
362 var b = Invariants(right);
363 string? reason = null;
364 if (a.Edges != b.Edges) reason = $"edges {a.Edges} {b.Edges}";
365 else if (!a.Degrees.SequenceEqual(b.Degrees)) reason = "degrees";
366 else if (a.Triangles != b.Triangles) reason = $"triangles {a.Triangles} {b.Triangles}";
367 else if (a.Sizes.Count != b.Sizes.Count) reason = $"components {a.Sizes.Count} {b.Sizes.Count}";
368 else if (a.Girth != b.Girth) reason = $"girth {a.Girth} {b.Girth}";
369 else if (a.Diameter != b.Diameter) reason = $"diameter {a.Diameter} {b.Diameter}";
370 else if (a.Bipartite != b.Bipartite) reason = $"bipartite {a.Bipartite} {b.Bipartite}";
371 if (reason is not null)
372 {
373 Console.WriteLine("not isomorphic " + reason);
374 return;
375 }
376 var x = Refine(left, DegreeColors(left));
377 var y = Refine(right, DegreeColors(right));
378 int[] xs = new int[Cells(x.Colors)], ys = new int[Cells(y.Colors)];
379 foreach (int color in x.Colors) xs[color]++;
380 foreach (int color in y.Colors) ys[color]++;
381 if (x.Rounds != y.Rounds || !xs.SequenceEqual(ys))
382 {
383 Console.WriteLine("not isomorphic refinement");
384 return;
385 }
386 Canonical c = Canon(left), d = Canon(right);
387 if (c.Text != d.Text)
388 {
389 Console.WriteLine("not isomorphic canonical");
390 return;
391 }
392 int[] inverse = new int[right.N], mapping = new int[left.N];
393 for (int i = 0; i < right.N; i++) inverse[d.Labels[i]] = i;
394 for (int i = 0; i < left.N; i++) mapping[i] = inverse[c.Labels[i]];
395 Console.WriteLine("isomorphic\nmapping " + Numbers(mapping));
396 }

406 static void Classify(int n)
407 {
408 Require(n > 0, "bad value");
409 Require(n <= 5, "too large");
410 List<SortedSet<string>> classes = [];
411 SortedSet<string> current = new(StringComparer.Ordinal) { Canon(new(n)).Text! };
412 int total = 0;
413 for (int m = 0; m <= n * (n - 1) / 2; m++)
414 {
415 classes.Add(current);
416 total += current.Count;
417 SortedSet<string> next = new(StringComparer.Ordinal);
418 foreach (string text in current)
419 {
420 Graph graph = FromString(n, text);
421 for (int u = 0; u < n; u++) for (int v = u + 1; v < n; v++) if (!graph.Edge(u, v))
422 {
423 Graph added = new(graph);
424 added.Join(u, v);
425 next.Add(Canon(added).Text!);
426 }
427 }
428 current = next;
429 }
430 Console.WriteLine($"classify {n} total {total}");
431 for (int m = 0; m < classes.Count; m++) Console.WriteLine($"edges {m} count {classes[m].Count} " + string.Join(" ", classes[m].Select(text => text.Length == 0 ? "-" : text)));
432 }

435 {
436 string[] all = line.Split(' '), args = all[1..];
437 string command = all[0];
438 Require(Definitions.Contains(command) || Other.Contains(command), "unknown command " + command);
439 int minimum, maximum, names;
440 switch (command)
441 {
442 case "LIST":
443 minimum = maximum = names = 0;
444 break;
445 case "CLASSIFY":
446 minimum = maximum = 1;
447 names = 0;
448 break;
449 case "PETERSEN":
450 case "DELETE":
451 case "SHOW":
452 case "INVARIANTS":
453 case "CANON":
454 case "AUT":
455 minimum = maximum = names = 1;
456 break;
457 case "CYCLE":
458 case "PATH":
459 case "COMPLETE":
460 case "STAR":
461 case "WHEEL":
462 case "HYPERCUBE":
463 minimum = maximum = 2;
464 names = 1;
465 break;
466 case "BIPARTITE":
467 case "GRID":
468 minimum = maximum = 3;
469 names = 1;
470 break;
471 case "COMPLEMENT":
472 case "LINE":
473 case "CANONIZE":
474 case "ISO":
475 case "SUBISO":
476 minimum = maximum = names = 2;
477 break;
478 case "PRODUCT":
479 case "UNION":
480 minimum = maximum = names = 3;
481 break;
482 case "GRAPH":
483 minimum = 2;
484 maximum = int.MaxValue;
485 names = 1;
486 break;
487 case "CIRCULANT":
488 minimum = 3;
489 maximum = int.MaxValue;
490 names = 1;
491 break;
492 case "RELABEL":
493 minimum = names = 2;
494 maximum = int.MaxValue;
495 break;
496 case "INDUCED":
497 minimum = 3;
498 names = 2;
499 maximum = int.MaxValue;
500 break;
501 default:
502 minimum = names = 1;
503 maximum = int.MaxValue;
504 break;
505 }
506 Require(args.Length >= minimum && args.Length <= maximum, "bad arguments");
507 for (int i = 0; i < names; i++) Require(Name(args[i]), "bad arguments");
508 for (int i = names; i < args.Length; i++) Require(command == "GRAPH" && i >= 2 ? Matches(args[i], "(?:0|[1-9][0-9]{0,2})-(?:0|[1-9][0-9]{0,2})") : Number(args[i]), "bad arguments");
509 int firstReference = Definitions.Contains(command) ? (command is "COMPLEMENT" or "LINE" or "PRODUCT" or "UNION" or "RELABEL" or "INDUCED" or "CANONIZE" ? 1 : names) : 0;
510 if (command is not ("LIST" or "CLASSIFY")) for (int i = firstReference; i < names; i++) Require(Graphs.ContainsKey(args[i]), "no graph " + args[i]);
511 string name = args.Length == 0 ? "" : args[0];
512 Graph? graph = null;
513 int n = args.Length > names ? Integer(args[names]) : 0;
514 if (Definitions.Contains(command))
515 {
516 switch (command)
517 {
518 case "GRAPH":
519 case "CYCLE":
520 case "PATH":
521 case "COMPLETE":
522 case "CIRCULANT":
523 Require(n >= (command is "CYCLE" or "CIRCULANT" ? 3 : 1), "bad value");
524 Require(n <= 16, "too large");
525 graph = new(n);
526 if (command == "GRAPH") for (int i = 2; i < args.Length; i++)
527 {
528 string[] edge = args[i].Split('-');
529 int u = Integer(edge[0]), v = Integer(edge[1]);
530 Require(u < n && v < n && u != v, "bad edge " + args[i]);
531 Require(!graph.Edge(u, v), "duplicate edge " + args[i]);
532 graph.Join(u, v);
533 }
534 if (command is "PATH" or "CYCLE")
535 {
536 for (int i = 1; i < n; i++) graph.Join(i - 1, i);
537 if (command == "CYCLE") graph.Join(n - 1, 0);
538 }
539 if (command == "COMPLETE") for (int u = 0; u < n; u++) for (int v = u + 1; v < n; v++) graph.Join(u, v);
540 if (command == "CIRCULANT")
541 {
542 int previous = 0;
543 for (int i = 2; i < args.Length; i++)
544 {
545 int jump = Integer(args[i]);
546 Require(jump > previous && jump * 2 <= n, "bad value");
547 previous = jump;
548 for (int v = 0; v < n; v++) graph.Join(v, (v + jump) % n);
549 }
550 }
551 break;
552 case "STAR":
553 case "WHEEL":
554 Require(n >= (command == "WHEEL" ? 3 : 1), "bad value");
555 Require(n < 16, "too large");
556 graph = new(n + 1);
557 for (int i = 1; i <= n; i++) graph.Join(0, i);
558 if (command == "WHEEL") for (int i = 1; i <= n; i++) graph.Join(i, i == n ? 1 : i + 1);
559 break;
560 case "BIPARTITE":
561 case "GRID":
562 {
563 int b = Integer(args[2]);
564 Require(n > 0 && b > 0, "bad value");
565 int size = command == "GRID" ? n * b : n + b;
566 Require(size <= 16, "too large");
567 graph = new(size);
568 if (command == "BIPARTITE") for (int u = 0; u < n; u++) for (int v = 0; v < b; v++) graph.Join(u, n + v);
569 else for (int x = 0; x < n; x++) for (int y = 0; y < b; y++)
570 {
571 if (x + 1 < n) graph.Join(x * b + y, (x + 1) * b + y);
572 if (y + 1 < b) graph.Join(x * b + y, x * b + y + 1);
573 }
574 break;
575 }
576 case "PETERSEN":
577 graph = new(10);
578 for (int i = 0; i < 5; i++)
579 {
580 graph.Join(i, (i + 1) % 5);
581 graph.Join(i, i + 5);
582 graph.Join(i + 5, 5 + (i + 2) % 5);
583 }
584 break;
585 case "HYPERCUBE":
586 Require(n > 0, "bad value");
587 Require(n <= 4, "too large");
588 graph = new(1 << n);
589 for (int u = 0; u < graph.N; u++) for (int bit = 0; bit < n; bit++) graph.Join(u, u ^ 1 << bit);
590 break;
591 case "COMPLEMENT":
592 {
593 Graph source = Graphs[args[1]];
594 graph = new(source.N);
595 for (int u = 0; u < source.N; u++) for (int v = u + 1; v < source.N; v++) if (!source.Edge(u, v)) graph.Join(u, v);
596 break;
597 }
598 case "LINE":
599 {
600 var edges = Graphs[args[1]].Edges();
601 Require(edges.Count > 0, "no edges");
602 Require(edges.Count <= 16, "too large");
603 graph = new(edges.Count);
604 for (int i = 0; i < edges.Count; i++) for (int j = i + 1; j < edges.Count; j++) if (edges[i].U == edges[j].U || edges[i].U == edges[j].V || edges[i].V == edges[j].U || edges[i].V == edges[j].V) graph.Join(i, j);
605 break;
606 }
607 case "PRODUCT":
608 case "UNION":
609 {
610 Graph a = Graphs[args[1]], b = Graphs[args[2]];
611 bool product = command == "PRODUCT";
612 int size = product ? a.N * b.N : a.N + b.N;
613 Require(size <= 16, "too large");
614 graph = new(size);
615 if (product)
616 {
617 for (int x = 0; x < a.N; x++) foreach (var edge in b.Edges()) graph.Join(x * b.N + edge.U, x * b.N + edge.V);
618 for (int y = 0; y < b.N; y++) foreach (var edge in a.Edges()) graph.Join(edge.U * b.N + y, edge.V * b.N + y);
619 }
620 else
621 {
622 foreach (var edge in a.Edges()) graph.Join(edge.U, edge.V);
623 foreach (var edge in b.Edges()) graph.Join(a.N + edge.U, a.N + edge.V);
624 }
625 break;
626 }
627 case "RELABEL":
628 case "INDUCED":
629 case "CANONIZE":
630 {
631 Graph source = Graphs[args[1]];
632 int[] vertices = command == "CANONIZE" ? Canon(source).Labels : args.Skip(2).Select(Integer).ToArray();
633 if (command == "RELABEL") Require(vertices.Length == source.N && vertices.Distinct().Count() == source.N && vertices.All(v => v < source.N), "bad permutation");
634 else if (command == "INDUCED") Require(vertices.Distinct().Count() == vertices.Length && vertices.All(v => v < source.N), "bad vertices");
635 graph = new(command == "INDUCED" ? vertices.Length : source.N);
636 if (command == "INDUCED")
637 {
638 for (int u = 0; u < vertices.Length; u++) for (int v = u + 1; v < vertices.Length; v++) if (source.Edge(vertices[u], vertices[v])) graph.Join(u, v);
639 }
640 else foreach (var edge in source.Edges()) graph.Join(vertices[edge.U], vertices[edge.V]);
641 break;
642 }
643 }
644 bool existed = Graphs.ContainsKey(name);
645 Graphs[name] = graph!;
646 Console.WriteLine($"{(existed ? "redefined" : "defined")} {name} n {graph!.N} m {graph.Edges().Count}");
647 return;
648 }
649 graph = Graphs.GetValueOrDefault(name);
650 switch (command)
651 {
652 case "DELETE":
653 Graphs.Remove(name);
654 Console.WriteLine("deleted " + name);
655 break;
656 case "LIST":
657 Console.WriteLine($"graphs {Graphs.Count}");
658 foreach (var entry in Graphs) Console.WriteLine($"{entry.Key} n {entry.Value.N} m {entry.Value.Edges().Count}");
659 break;
660 case "SHOW":
661 Console.WriteLine($"graph {name} n {graph!.N} m {graph.Edges().Count}");
662 for (int v = 0; v < graph.N; v++)
663 {
664 StringBuilder row = new(v + ":");
665 for (int w = 0; w < graph.N; w++) if (graph.Edge(v, w)) row.Append(' ').Append(w);
666 Console.WriteLine(row);
667 }
668 break;
669 case "INVARIANTS":
670 PrintInvariants(name, graph!);
671 break;
672 case "REFINE":
673 {
674 Graph source = graph!;
675 int[] vertices = args.Skip(1).Select(Integer).ToArray();
676 Require(vertices.All(v => v < source.N), "bad vertices");
677 var result = Refine(source, DegreeColors(source));
678 int rounds = result.Rounds;
679 int[] colors = result.Colors;
680 foreach (int vertex in vertices)
681 {
682 Require(colors.Count(color => color == colors[vertex]) > 1, "singleton " + vertex);
683 result = Refine(source, Individualize(colors, vertex));
684 colors = result.Colors;
685 rounds += result.Rounds;
686 }
687 Console.WriteLine($"refine {name} rounds {rounds} cells {Cells(colors)}");
688 for (int color = 0; color < Cells(colors); color++)
689 {
690 StringBuilder row = new(color + ":");
691 for (int v = 0; v < source.N; v++) if (colors[v] == color) row.Append(' ').Append(v);
692 Console.WriteLine(row);
693 }
694 break;
695 }
696 case "CANON":
697 {
698 var result = Canon(graph!);
699 Console.WriteLine($"canon {name} nodes {result.Nodes} leaves {result.Leaves}");
700 Console.WriteLine("labels " + Numbers(result.Labels));
701 Console.WriteLine("string " + (result.Text!.Length == 0 ? "-" : result.Text));
702 break;
703 }
704 case "AUT":
705 Automorphisms(name, graph!);
706 break;
707 case "ISO":
708 Isomorphism(graph!, Graphs[args[1]]);
709 break;
710 case "SUBISO":
711 {
712 Graph destination = Graphs[args[1]];
713 Embeddings result = new();
714 if (graph!.N <= destination.N) result.Search(graph, destination, 0, 0, new int[graph.N], true);
715 int order = Canon(graph).Maxima.Count;
716 Console.WriteLine($"subiso {name} {args[1]} embeddings {result.Count} induced {result.Induced}");
717 Console.WriteLine($"copies {result.Count / order} induced {result.Induced / order}");
718 foreach (var embedding in result.First) Console.WriteLine("embedding " + Numbers(embedding.Map) + (embedding.Induced ? " induced" : ""));
719 break;
720 }
721 case "CLASSIFY":
722 Classify(Integer(name));
723 break;
724 }
