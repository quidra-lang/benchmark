32 for (i in seq_len(min(length(a), length(b)))) {

41refine <- function(g, colors) {
42 rounds <- 0L
43 repeat {
44 signatures <- lapply(vertices(g), function(v) {
45 list(color = colors[[v + 1L]], neighbors = sort(colors[neighbors(g, v) + 1L]))
46 })
47 ordered <- integer()
48 for (i in seq_along(signatures)) {
49 at <- length(ordered)
50 while (at > 0L && compare_signature(signatures[[i]], signatures[[ordered[[at]]]]) < 0L) at <- at - 1L
51 ordered <- append(ordered, i, after = at)
52 }
53 fresh <- integer(g$n)
54 color <- -1L
55 previous <- NULL
56 for (i in ordered) {
57 if (is.null(previous) || compare_signature(signatures[[i]], previous) != 0L) color <- color + 1L
58 fresh[[i]] <- color
59 previous <- signatures[[i]]
60 }
61 if (color + 1L == length(unique(colors))) return(list(colors = colors, rounds = rounds))
62 colors <- fresh
63 rounds <- rounds + 1L
64 }

86canonical <- function(g) {
87 if (!is.null(g$canonical)) return(g$canonical)
88 nodes <- 0L
89 leaves <- 0L
90 best <- NULL
91 labels <- integer()
92 winners <- list()
93 visit <- function(colors) {
94 nodes <<- nodes + 1L
95 if (length(unique(colors)) == g$n) {
96 leaves <<- leaves + 1L
97 string <- leaf_string(g, colors)
98 if (is.null(best) || string > best) {
99 best <<- string
100 labels <<- colors
101 winners <<- list(colors)
102 } else if (string == best) {
103 winners[[length(winners) + 1L]] <<- colors
104 }
105 return(invisible(NULL))
106 }
107 cells <- lapply(0:max(colors), function(c) which(colors == c) - 1L)
108 sizes <- lengths(cells)
109 sizes[sizes == 1L] <- g$n + 1L
110 cell <- cells[[which.min(sizes)]]
111 for (v in cell) visit(refine(g, individualize(colors, v))$colors)
112 }
113 visit(stable(g)$colors)
114 g$canonical <- list(nodes = nodes, leaves = leaves, string = best, labels = labels, winners = winners)
115 g$canonical
116}
117invariants <- function(g) {
118 if (!is.null(g$invariants)) return(g$invariants)
119 components <- integer()
120 seen <- rep(FALSE, g$n)
121 for (start in vertices(g)) {
122 if (seen[[start + 1L]]) next
123 queue <- start
124 seen[[start + 1L]] <- TRUE
125 at <- 1L
126 while (at <= length(queue)) {
127 u <- queue[[at]]
128 at <- at + 1L
129 for (v in neighbors(g, u)) {
130 if (!seen[[v + 1L]]) {
131 seen[[v + 1L]] <- TRUE
132 queue <- c(queue, v)
133 }
134 }
135 }
136 components <- c(components, length(queue))
137 }
138 girth <- Inf
139 eccentricities <- integer()
140 bipartite <- TRUE
141 for (start in vertices(g)) {
142 distance <- rep(-1L, g$n)
143 parent <- rep(-1L, g$n)
144 distance[[start + 1L]] <- 0L
145 queue <- start
146 at <- 1L
147 while (at <= length(queue)) {
148 u <- queue[[at]]
149 at <- at + 1L
150 for (v in neighbors(g, u)) {
151 if (distance[[v + 1L]] == -1L) {
152 distance[[v + 1L]] <- distance[[u + 1L]] + 1L
153 parent[[v + 1L]] <- u
154 queue <- c(queue, v)
155 } else if (parent[[u + 1L]] != v) {
156 girth <- min(girth, distance[[u + 1L]] + distance[[v + 1L]] + 1L)
157 }
158 if (distance[[u + 1L]] %% 2L == distance[[v + 1L]] %% 2L) bipartite <- FALSE
159 }
160 }
161 eccentricities <- c(eccentricities, max(distance))
162 }
163 triangles <- 0L
164 if (g$n >= 3L) {
165 for (a in 0:(g$n - 3L)) {
166 for (b in (a + 1L):(g$n - 2L)) {
167 for (c in (b + 1L):(g$n - 1L)) {
168 if (g$adj[a + 1L, b + 1L] && g$adj[a + 1L, c + 1L] && g$adj[b + 1L, c + 1L]) triangles <- triangles + 1L
169 }
170 }
171 }
172 }
173 g$invariants <- list(degrees = sort(degrees(g), decreasing = TRUE), triangles = triangles,
174 sizes = sort(components, decreasing = TRUE), components = length(components),
175 girth = if (is.infinite(girth)) "none" else girth,
176 diameter = if (length(components) == 1L) max(eccentricities) else "INF",
177 radius = if (length(components) == 1L) min(eccentricities) else "INF",
178 bipartite = if (bipartite) "yes" else "no")
179 g$invariants
180}
181clique <- function(g) {
182 best <- integer()
183 for (mask in seq_len(2^g$n - 1L)) {
184 set <- vertices(g)[bitwAnd(mask, bitwShiftL(1L, vertices(g))) != 0L]
185 valid <- all(vapply(pairs(length(set)), function(e) g$adj[set[[e[[1]] + 1L]] + 1L, set[[e[[2]] + 1L]] + 1L], logical(1)))
186 if (valid && (length(set) > length(best) || (length(set) == length(best) && compare_sequences(set, best) < 0L))) best <- set
187 }
188 best
189}
190coloring <- function(g) {
191 for (limit in seq_len(g$n)) {
192 colors <- rep(-1L, g$n)
193 search <- function(v, maximum) {
194 if (v == g$n) return(TRUE)
195 for (color in 0:min(limit - 1L, maximum + 1L)) {
196 previous <- seq_len(v)
197 if (any(g$adj[v + 1L, previous] & colors[previous] == color)) next
198 colors[[v + 1L]] <<- color
199 if (search(v + 1L, max(maximum, color))) return(TRUE)
200 }
201 colors[[v + 1L]] <<- -1L
202 FALSE
203 }
204 if (search(0L, -1L)) return(list(count = limit, colors = colors))
205 }

226validate <- function(command, args) {
227 schema <- schemas[[command]]
228 if (is.null(schema)) fail(paste("unknown command", command))
229 if (length(args) < schema[[1]] || length(args) > schema[[2]]) fail("bad arguments")
230 types <- schema[[3]]
231 for (i in seq_along(args)) {
232 type <- if (i <= length(types)) types[[i]] else schema[[4]]
233 pattern <- switch(type, num = "\\A(?:0|[1-9][0-9]{0,2})\\z",
234 edge = "\\A(?:0|[1-9][0-9]{0,2})-(?:0|[1-9][0-9]{0,2})\\z", "\\A[a-z][a-z0-9]{0,7}\\z")
235 if (!grepl(pattern, args[[i]], perl = TRUE)) fail("bad arguments")
236 }
237 for (i in seq_along(types)) {
238 if (types[[i]] == "graph" && is.null(get_graph(args[[i]]))) fail(paste("no graph", args[[i]]))
239 }
240 schema

246define_graph <- function(command, args) {
247 name <- args[[1]]
248 number <- if (length(args) >= 2L && grepl("^[0-9]+$", args[[2]])) as.integer(args[[2]]) else 0L
249 source <- if (length(args) >= 2L) get_graph(args[[2]]) else NULL
250 if (command == "GRAPH") {
251 size_check(number)
252 g <- new_graph(number)
253 for (word in args[-c(1L, 2L)]) {
254 e <- as.integer(strsplit(word, "-", fixed = TRUE)[[1]])
255 if (e[[1]] >= number || e[[2]] >= number || e[[1]] == e[[2]]) fail(paste("bad edge", word))
256 if (g$adj[e[[1]] + 1L, e[[2]] + 1L]) fail(paste("duplicate edge", word))
257 join(g, e[[1]], e[[2]])
258 }
259 } else if (command %in% c("PATH", "CYCLE", "COMPLETE", "STAR", "WHEEL", "CIRCULANT")) {
260 minimum <- if (command %in% c("CYCLE", "WHEEL", "CIRCULANT")) 3L else 1L
261 maximum <- if (command %in% c("STAR", "WHEEL")) 15L else 16L
262 size_check(number, minimum, maximum)
263 n <- number + as.integer(command %in% c("STAR", "WHEEL"))
264 g <- new_graph(n)
265 if (command %in% c("PATH", "CYCLE")) {
266 if (n >= 2L) for (v in 0:(n - 2L)) join(g, v, v + 1L)
267 if (command == "CYCLE") join(g, n - 1L, 0L)
268 } else if (command == "COMPLETE") {
269 for (e in pairs(n)) join(g, e[[1]], e[[2]])
270 } else if (command %in% c("STAR", "WHEEL")) {
271 for (v in seq_len(n - 1L)) join(g, 0L, v)
272 if (command == "WHEEL") {
273 for (v in seq_len(n - 2L)) join(g, v, v + 1L)
274 join(g, n - 1L, 1L)
275 }
276 } else {
277 jumps <- as.integer(args[-c(1L, 2L)])
278 if (any(jumps == 0L | 2L * jumps > n) || any(diff(jumps) <= 0L)) fail("bad value")
279 for (v in vertices(g)) for (j in jumps) join(g, v, (v + j) %% n)
280 }
281 } else if (command %in% c("BIPARTITE", "GRID")) {
282 a <- as.integer(args[[2]])
283 b <- as.integer(args[[3]])
284 if (a == 0L || b == 0L) fail("bad value")
285 n <- if (command == "GRID") a * b else a + b
286 size_check(n)
287 g <- new_graph(n)
288 for (x in 0:(a - 1L)) {
289 for (y in 0:(b - 1L)) {
290 if (command == "BIPARTITE") {
291 join(g, x, a + y)
292 } else {
293 if (y + 1L < b) join(g, x * b + y, x * b + y + 1L)
294 if (x + 1L < a) join(g, x * b + y, (x + 1L) * b + y)
295 }
296 }
297 }
298 } else if (command == "PETERSEN") {
299 g <- new_graph(10L)
300 for (i in 0:4) {
301 join(g, i, (i + 1L) %% 5L)
302 join(g, i, i + 5L)
303 join(g, i + 5L, 5L + (i + 2L) %% 5L)
304 }
305 } else if (command == "HYPERCUBE") {
306 size_check(number, 1L, 4L)
307 g <- new_graph(2^number)
308 for (v in vertices(g)) for (bit in 0:(number - 1L)) join(g, v, bitwXor(v, bitwShiftL(1L, bit)))
309 } else if (command == "COMPLEMENT") {
310 g <- new_graph(source$n)
311 for (e in pairs(source$n)) if (!source$adj[e[[1]] + 1L, e[[2]] + 1L]) join(g, e[[1]], e[[2]])
312 } else if (command == "LINE") {
313 edges <- edge_list(source)
314 if (!length(edges)) fail("no edges")
315 size_check(length(edges))
316 g <- new_graph(length(edges))
317 for (e in pairs(length(edges))) {
318 if (length(intersect(edges[[e[[1]] + 1L]], edges[[e[[2]] + 1L]]))) join(g, e[[1]], e[[2]])
319 }
320 } else if (command %in% c("PRODUCT", "UNION")) {
321 other <- get_graph(args[[3]])
322 n <- if (command == "PRODUCT") source$n * other$n else source$n + other$n
323 size_check(n)
324 g <- new_graph(n)
325 if (command == "PRODUCT") {
326 for (x in vertices(source)) for (e in edge_list(other)) join(g, x * other$n + e[[1]], x * other$n + e[[2]])
327 for (y in vertices(other)) for (e in edge_list(source)) join(g, e[[1]] * other$n + y, e[[2]] * other$n + y)
328 } else {
329 for (e in edge_list(source)) join(g, e[[1]], e[[2]])
330 for (e in edge_list(other)) join(g, source$n + e[[1]], source$n + e[[2]])
331 }
332 } else if (command %in% c("RELABEL", "CANONIZE")) {
333 labels <- if (command == "CANONIZE") canonical(source)$labels else as.integer(args[-c(1L, 2L)])
334 if (!identical(sort(labels), vertices(source))) fail("bad permutation")
335 g <- new_graph(source$n)
336 for (e in edge_list(source)) join(g, labels[[e[[1]] + 1L]], labels[[e[[2]] + 1L]])
337 } else if (command == "INDUCED") {
338 selected <- as.integer(args[-c(1L, 2L)])
339 if (any(selected >= source$n) || anyDuplicated(selected)) fail("bad vertices")
340 g <- new_graph(length(selected))
341 for (e in pairs(length(selected))) {
342 if (source$adj[selected[[e[[1]] + 1L]] + 1L, selected[[e[[2]] + 1L]] + 1L]) join(g, e[[1]], e[[2]])
343 }
344 }
345 verb <- if (is.null(get_graph(name))) "defined" else "redefined"
346 registry[[name]] <- g
347 say(verb, " ", name, " n ", g$n, " m ", length(edge_list(g)))
348}

368 av <- invariants(a)
369 bv <- invariants(b)
370 tests <- list(list("vertices", a$n, b$n), list("edges", length(edge_list(a)), length(edge_list(b))),
371 list("degrees", av$degrees, bv$degrees), list("triangles", av$triangles, bv$triangles),
372 list("components", av$components, bv$components), list("girth", av$girth, bv$girth),
373 list("diameter", av$diameter, bv$diameter), list("bipartite", av$bipartite, bv$bipartite))
374 for (test in tests) {
375 if (!identical(test[[2]], test[[3]])) {
376 say("not isomorphic ", test[[1]], if (test[[1]] == "degrees") "" else paste0(" ", test[[2]], " ", test[[3]]))

394automorphisms <- function(name, g) {
395 canon <- canonical(g)
396 inverse <- order(canon$labels) - 1L
397 vertex_parent <- seq_len(g$n)
398 edges <- edge_list(g)
399 edge_parent <- seq_along(edges)
400 find <- function(parents, x) {
401 while (parents[[x]] != x) x <- parents[[x]]
402 x
403 }
404 edge_keys <- vapply(edges, function(e) paste(e, collapse = "-"), character(1))
405 for (labels in canon$winners) {
406 mapping <- inverse[labels + 1L]
407 for (v in vertices(g)) vertex_parent[[find(vertex_parent, v + 1L)]] <- find(vertex_parent, mapping[[v + 1L]] + 1L)
408 for (i in seq_along(edges)) {
409 key <- paste(sort(mapping[edges[[i]] + 1L]), collapse = "-")
410 edge_parent[[find(edge_parent, i)]] <- find(edge_parent, match(key, edge_keys))
411 }
412 }
413 orbits <- function(parents) {
414 result <- list()
415 for (i in seq_along(parents)) {
416 root <- find(parents, i)
417 key <- as.character(root)
418 result[[key]] <- c(result[[key]], i)
419 }
420 result[order(vapply(result, function(x) x[[1]], integer(1)))]
421 }
422 vertex_orbits <- orbits(vertex_parent)
423 edge_orbits <- orbits(edge_parent)
424 say("aut ", name, " order ", length(canon$winners))
425 say("vertex orbits ", length(vertex_orbits))
426 for (orbit in vertex_orbits) say("orbit ", paste(orbit - 1L, collapse = " "))
427 say("edge orbits ", length(edge_orbits))
428 for (orbit in edge_orbits) say("orbit ", paste(edge_keys[orbit], collapse = " "))
429 say("transitive vertex ", if (length(vertex_orbits) == 1L) "yes" else "no",
430 " edge ", if (length(edge_orbits) == 1L) "yes" else "no")
431}
432subiso <- function(pn, tn, pattern, target) {
433 count <- 0L
434 induced_count <- 0L
435 examples <- list()
436 search <- function(mapping, induced) {
437 depth <- length(mapping)
438 if (depth == pattern$n) {
439 count <<- count + 1L
440 if (induced) induced_count <<- induced_count + 1L
441 if (length(examples) < 5L) examples[[length(examples) + 1L]] <<- list(mapping = mapping, induced = induced)
442 return(invisible(NULL))
443 }
444 for (v in vertices(target)) {
445 if (v %in% mapping) next
446 previous <- seq_len(depth)
447 pe <- pattern$adj[depth + 1L, previous]
448 te <- target$adj[v + 1L, mapping + 1L]
449 if (any(pe & !te)) next
450 search(c(mapping, v), induced && all(pe == te))
451 }
452 }
453 if (pattern$n <= target$n) search(integer(), TRUE)
454 group_order <- length(canonical(pattern)$winners)
455 say("subiso ", pn, " ", tn, " embeddings ", count, " induced ", induced_count)
456 say("copies ", count %/% group_order, " induced ", induced_count %/% group_order)
457 for (example in examples) say("embedding ", paste(example$mapping, collapse = " "), if (example$induced) " induced" else "")
458}
459classify <- function(n) {
460 size_check(n, 1L, 5L)
461 key <- as.character(n)
462 levels <- class_cache[[key]]
463 if (is.null(levels)) {
464 edge_pairs <- pairs(n)
465 levels <- list(paste(rep("0", length(edge_pairs)), collapse = ""))
466 for (m in seq_along(edge_pairs)) {
467 next_level <- character()
468 for (string in levels[[m]]) {
469 bits <- strsplit(string, "", fixed = TRUE)[[1]]
470 for (extra in which(bits == "0")) {
471 g <- new_graph(n)
472 for (i in seq_along(edge_pairs)) {
473 if (bits[[i]] == "1" || i == extra) join(g, edge_pairs[[i]][[1]], edge_pairs[[i]][[2]])
474 }
475 next_level <- union(next_level, canonical(g)$string)
476 }
477 }
478 levels[[length(levels) + 1L]] <- next_level
479 }
480 class_cache[[key]] <- levels
481 }
482 say("classify ", n, " total ", sum(lengths(levels)))
483 for (m in seq_along(levels)) {
484 strings <- sort(levels[[m]], method = "radix")
485 strings[strings == ""] <- "-"
486 say("edges ", m - 1L, " count ", length(strings), " ", paste(strings, collapse = " "))
487 }
488}
489execute <- function(line) {
490 words <- strsplit(line, " ", fixed = TRUE)[[1]]
491 command <- words[[1]]
492 args <- words[-1]
493 schema <- validate(command, args)
494 if (length(schema[[3]]) && schema[[3]][[1]] == "name") {
495 define_graph(command, args)
496 return(invisible(NULL))
497 }
498 name <- if (length(args)) args[[1]] else NULL
499 g <- if (!is.null(name)) get_graph(name) else NULL
500 switch(command,
501 DELETE = {
502 rm(list = name, envir = registry)
503 say("deleted ", name)
504 },
505 LIST = {
506 names <- sort(ls(registry), method = "radix")
507 say("graphs ", length(names))
508 for (name in names) say(name, " n ", get_graph(name)$n, " m ", length(edge_list(get_graph(name))))
509 },
510 SHOW = {
511 say("graph ", name, " n ", g$n, " m ", length(edge_list(g)))
512 for (v in vertices(g)) {
513 ns <- neighbors(g, v)
514 say(v, ":", if (length(ns)) paste0(" ", paste(ns, collapse = " ")) else "")
515 }
516 },
517 INVARIANTS = show_invariants(name, g),
518 REFINE = {
519 selected <- as.integer(args[-1])
520 if (any(selected >= g$n)) fail("bad vertices")
521 result <- stable(g)
