7class HttpRouter
8 VALID_NAME = /\A[a-z][a-z0-9_]{0,19}\z/
9 ROLE = /\A[a-z]{1,16}\z/
10 MEDIA = /\A[a-z0-9.+-]+\/[a-z0-9.+-]+\z/
11 SEGMENT = /\A[A-Za-z0-9._~-]+\z/
12 REASONS = { 200 => 'OK', 204 => 'No Content', 301 => 'Moved Permanently', 302 => 'Found', 304 => 'Not Modified',
13 307 => 'Temporary Redirect', 308 => 'Permanent Redirect', 400 => 'Bad Request', 401 => 'Unauthorized',
14 403 => 'Forbidden', 404 => 'Not Found', 405 => 'Method Not Allowed', 406 => 'Not Acceptable',
15 412 => 'Precondition Failed', 429 => 'Too Many Requests', 501 => 'Not Implemented' }.freeze
16 HEADERS = %w[Location Allow Retry-After WWW-Authenticate Access-Control-Allow-Origin Access-Control-Allow-Methods Vary ETag Content-Type Content-Length].freeze

18 def initialize
19 @routes = []
20 @tokens = {}
21 @redirects = []
22 @origins = []
23 @limit = nil
24 @errors = 0
25 @clients = {}
26 @statuses = Hash.new(0)
27 @requests = @url_ok = @url_error = 0
28 end

30 def segment?(text)
31 SEGMENT.match?(text) && !%w[. ..].include?(text)
32 end
33
34 def canonical?(text)
35 text == '/' || (text.start_with?('/') && text[1..].split('/', -1).all? { |part| segment?(part) })
36 end
37
38 def pattern(text)
39 return [] if text == '/'
40 return nil unless text.start_with?('/')
41 parts = text[1..].split('/', -1)
42 names = []
43 pieces = []
44 parts.each_with_index do |part, index|
45 if segment?(part)
46 pieces << [:literal, part]
47 else
48 match = /\A\{(\*?)([a-z][a-z0-9_]{0,19})(?::(int|slug))?\}\z/.match(part)
49 return nil unless match
50 star, name, type = match.captures
51 return nil if names.include?(name) || (star == '*' && (type || index != parts.length - 1))
52 names << name
53 pieces << [star == '*' ? :wildcard : (type ? type.to_sym : :parameter), name]
54 end
55 end
56 pieces
57 end

59 def config_error(message)
60 raise ConfigurationError, message
61 end

63 def configure(line, number)
64 command, *args = line.split(' ')
65 counts = { 'ROUTE' => [3, 5], 'TOKEN' => [2], 'REDIRECT' => [3], 'ORIGIN' => [1], 'LIMIT' => [2] }
66 config_error("unknown command #{command}") unless counts.key?(command) || command == 'TYPES'
67 correct_count = command == 'TYPES' ? args.length >= 2 : counts[command].include?(args.length)
68 config_error('wrong number of arguments') unless correct_count
69 case command
70 when 'ROUTE'
71 name, methods, path = args
72 config_error("bad route name #{name}") unless VALID_NAME.match?(name)
73 config_error("duplicate route #{name}") if @routes.any? { |route| route[:name] == name }
74 method_list = methods.split(',', -1)
75 config_error("bad methods #{methods}") unless method_list.all? { |method| %w[GET POST PUT PATCH DELETE].include?(method) } && method_list.uniq.length == method_list.length
76 pieces = pattern(path)
77 config_error("bad pattern #{path}") unless pieces
78 config_error("unknown option #{args[3]}") if args.length == 5 && args[3] != 'ROLE'
79 config_error("bad role #{args[4]}") if args.length == 5 && !ROLE.match?(args[4])
80 @routes << { name: name, methods: method_list, pattern: path, pieces: pieces, role: args.length == 5 ? args[4] : 'none', types: nil, hits: 0, number: @routes.length }
81 when 'TOKEN'
82 token, role = args
83 config_error("bad token #{token}") unless token.match?(/\A[A-Za-z0-9._-]{1,32}\z/)
84 config_error("duplicate token #{token}") if @tokens.key?(token)
85 config_error("bad role #{role}") unless ROLE.match?(role)
86 @tokens[token] = role
87 when 'REDIRECT'
88 from, to, code = args
89 [from, to].each do |path|
90 base = path.end_with?('/*') ? path[0...-2] : path
91 config_error("bad path #{path}") unless canonical?(base) && (!path.end_with?('/*') || base != '/')
92 end
93 config_error('wildcard mismatch') unless from.end_with?('/*') == to.end_with?('/*')
94 config_error("bad redirect code #{code}") unless %w[301 302 307 308].include?(code)
95 config_error("duplicate redirect #{from}") if @redirects.any? { |rule| rule[:from] == from }
96 star = from.end_with?('/*')
97 @redirects << { from: from, base: star ? from[0...-2] : from, to: star ? to[0...-2] : to, wildcard: star, code: code.to_i }
98 when 'ORIGIN'
99 origin = args[0]
100 config_error("bad origin #{origin}") unless origin.match?(/\Ahttps?:\/\/[a-z0-9.:-]+\z/)
101 config_error("duplicate origin #{origin}") if @origins.include?(origin)
102 @origins << origin
103 when 'LIMIT'
104 valid = args.all? { |word| word.match?(/\A[1-9][0-9]*\z/) } && args[0].to_i.between?(1, 100) && args[1].to_i.between?(1, 3600)
105 config_error('bad limit') unless valid
106 config_error('duplicate limit') if @limit
107 @limit = args.map(&:to_i)
108 when 'TYPES'
109 name, *types = args
110 route = @routes.find { |item| item[:name] == name }
111 config_error("unknown route #{name}") unless route
112 config_error("duplicate types #{name}") if route[:types]
113 seen = []
114 types.each do |type|
115 config_error("bad media type #{type}") unless MEDIA.match?(type)
116 config_error("duplicate media type #{type}") if seen.include?(type)
117 seen << type
118 end
119 route[:types] = types
120 end
121 rescue ConfigurationError => error
122 @errors += 1
123 puts "CONFIG ERROR #{number}: #{error.message}"
124 end

126 def priority(route)
127 pieces = route[:pieces]
128 [-pieces.count { |kind, _| kind == :literal }, -pieces.count { |kind, _| [:int, :slug].include?(kind) }, pieces.any? { |kind, _| kind == :wildcard } ? 1 : 0, route[:number]]
129 end

131 def decode(text)
132 text.gsub(/%([0-9A-Fa-f]{2})/) { Regexp.last_match(1).to_i(16).chr }
133 end
134
135 def encode(text)
136 text.each_char.map { |char| char.match?(/[A-Za-z0-9._~-]/) ? char : format('%%%02X', char.ord) }.join
137 end
138
139 def normalize(path)
140 path = path.gsub(/%([0-9A-Fa-f]{2})/) do
141 char = Regexp.last_match(1).to_i(16).chr
142 char.match?(/[A-Za-z0-9._~-]/) ? char : format('%%%02X', char.ord)
143 end
144 parts = []
145 path[1..].split('/', -1).each do |part|
146 if part == '..'
147 parts.pop
148 elsif part != '' && part != '.'
149 parts << part
150 end
151 end
152 '/' + parts.join('/')
153 end

155 def match_route(route, segments)
156 pieces = route[:pieces]
157 wildcard = !pieces.empty? && pieces[-1][0] == :wildcard
158 count = pieces.length - (wildcard ? 1 : 0)
159 length_fits = wildcard ? segments.length >= count : segments.length == count
160 return nil unless length_fits
161 parameters = {}
162 pieces.each_with_index do |(kind, name), index|
163 text = segments[index]
164 case kind
165 when :literal
166 return nil unless text == name
167 when :parameter
168 parameters[name] = decode(text)
169 when :int
170 return nil unless text.match?(/\A[0-9]{1,9}\z/)
171 parameters[name] = text.to_i.to_s
172 when :slug
173 return nil unless text.match?(/\A[a-z0-9-]+\z/)
174 parameters[name] = text
175 when :wildcard
176 parameters[name] = segments[count..].map { |part| decode(part) }.join('/')
177 end
178 end
179 parameters
180 end

182 def selected_type(types, accept)
183 return types[0] if accept.nil?
184 elements = []
185 accept.split(',', -1).each do |element|
186 range, *parameters = element.split(';', -1).map { |part| part.strip }
187 range = range.downcase
188 next unless range == '*/*' || range.match?(/\A[a-z0-9.+-]+\/\*\z/) || MEDIA.match?(range)
189 q = nil
190 parameters.each do |parameter|
191 next unless parameter.include?('=')
192 name, value = parameter.split('=', 2).map { |part| part.strip }
193 if name.downcase == 'q'
194 q = value
195 break
196 end
197 end
198 next if q && !q.match?(/\A(?:0(?:\.[0-9]{0,3})?|1(?:\.0{0,3})?)\z/)
199 weight = q ? (q.to_f * 1000).round : 1000
200 specificity = range == '*/*' ? 1 : (range.end_with?('/*') ? 2 : 3)
201 elements << [range, specificity, weight]
202 end
203 best = nil
204 best_weight = 0
205 types.each do |type|
206 specificity = 0
207 weight = 0
208 elements.each do |range, rank, candidate|
209 fits = range == '*/*' || range == type || (rank == 2 && range.split('/')[0] == type.split('/')[0])
210 if fits && rank > specificity
211 specificity = rank
212 weight = candidate
213 end
214 end
215 if weight > best_weight
216 best_weight = weight
217 best = type
218 end
219 end
220 best
221 end

223 def json_string(text)
224 '"' + text.gsub(/[\\"]/) { |char| '\\' + char } + '"'
225 end

227 def html(text)
228 text.gsub(/[&<>"]/) { |char| { '&' => '&amp;', '<' => '&lt;', '>' => '&gt;', '"' => '&quot;' }[char] }
229 end

231 def body_for(route, parameters, query, headers, type)
232 params = parameters.sort
233 queries = query.nil? ? [] : query.split('&', -1).reject(&:empty?).map do |part|
234 name, value = part.split('=', 2)
235 [decode(name.tr('+', ' ')), decode((value || '').tr('+', ' '))]
236 end
237 cookies = {}
238 if headers.key?('cookie')
239 headers['cookie'].split(';', -1).each do |part|
240 part = part.strip
241 next unless part.include?('=')
242 name, value = part.split('=', 2)
243 cookies[name] = value if name != '' && !cookies.key?(name)
244 end
245 end
246 cookie_pairs = cookies.sort
247 if type == 'application/json'
248 pairs = ->(list) { list.map { |name, value| json_string(name) + ':' + json_string(value) }.join(',') }
249 query_json = queries.map { |name, value| '[' + json_string(name) + ',' + json_string(value) + ']' }.join(',')
250 '{"route":' + json_string(route[:name]) + ',"params":{' + pairs.call(params) + '},"query":[' + query_json + '],"cookies":{' + pairs.call(cookie_pairs) + '}}'
251 elsif type == 'text/html'
252 output = '<dl><dt>route</dt><dd>' + html(route[:name]) + '</dd>'
253 [['param', params], ['query', queries], ['cookie', cookie_pairs]].each do |label, pairs|
254 pairs.each { |name, value| output += '<dt>' + label + ' ' + html(name) + '</dt><dd>' + html(value) + '</dd>' }
255 end
256 output + '</dl>'
257 else
258 pairs = ->(list) { list.empty? ? '-' : list.map { |name, value| encode(name) + '=' + encode(value) }.join('&') }
259 "route=#{route[:name]} params=#{pairs.call(params)} query=#{pairs.call(queries)} cookies=#{pairs.call(cookie_pairs)}"
260 end
261 end

263 def etag(body)
264 a = 1
265 b = 0
266 body.each_byte do |char|
267 a = (a + char) % 65521
268 b = (b + a) % 65521
269 end
270 format('"%04x%04x"', b, a)
271 end

273 def response(status, body = '', headers = {}, type = nil)
274 { status: status, body: body, headers: headers, type: type }
275 end

277 def handle(method, target, headers)
278 return response(501, "not implemented: #{method}") unless %w[GET HEAD POST PUT PATCH DELETE OPTIONS].include?(method)
279 return response(400, 'bad request: missing host') if !headers.key?('host') || headers['host'].empty?
280 effective = method
281 if headers.key?('x-http-method-override')
282 override = headers['x-http-method-override']
283 return response(400, 'bad request: method override') unless method == 'POST' && %w[PUT PATCH DELETE].include?(override)
284 effective = override
285 end
286 client = headers['x-client']
287 if client && !client.empty?
288 time = headers['x-time'].to_i
289 record = @clients[client] ||= { allowed: 0, limited: 0, level: @limit ? @limit[0] * @limit[1] : 0, last: time }
290 if @limit
291 capacity, period = @limit
292 record[:level] = [capacity * period, record[:level] + (time - record[:last]) * capacity].min
293 record[:last] = time
294 if record[:level] < period
295 record[:limited] += 1
296 wait = (period - record[:level] + capacity - 1) / capacity
297 return response(429, "too many requests: client #{client}", { 'Retry-After' => wait.to_s })
298 end
299 record[:level] -= period
300 end
301 record[:allowed] += 1
302 end
303 path, query = target.split('?', 2)
304 suffix = target.include?('?') ? '?' + (query || '') : ''
305 normalized = normalize(path)
306 return response(308, "redirect to #{normalized}#{suffix}", { 'Location' => normalized + suffix }) if normalized != path
307 @redirects.each do |rule|
308 if path == rule[:base] || (rule[:wildcard] && path.start_with?(rule[:base] + '/'))
309 destination = rule[:to] + path[rule[:base].length..] + suffix
310 return response(rule[:code], "redirect to #{destination}", { 'Location' => destination })
311 end
312 end
313 segments = path == '/' ? [] : path[1..].split('/', -1)
314 matches = @ordered.filter_map do |route|
315 params = match_route(route, segments)
316 params ? [route, params] : nil
317 end
318 return response(404, "not found: #{path}") if matches.empty?
319 allowed = matches.flat_map { |route, _| route[:methods] }.uniq + ['OPTIONS']
320 allowed << 'HEAD' if allowed.include?('GET')
321 allow = allowed.uniq.sort.join(', ')
322 return response(204, '', { 'Allow' => allow }) if effective == 'OPTIONS'
323 lookup = effective == 'HEAD' ? 'GET' : effective
324 selected = matches.find { |route, _| route[:methods].include?(lookup) }
325 return response(405, "method not allowed: #{effective}", { 'Allow' => allow }) unless selected
326 route, params = selected
327 route[:hits] += 1
328 if route[:role] != 'none'
329 return response(401, 'unauthorized: missing token', { 'WWW-Authenticate' => 'Bearer realm="api"' }) unless headers.key?('authorization')
330 scheme, _, token = headers['authorization'].partition(' ')
331 unless scheme.downcase == 'bearer' && @tokens.key?(token)
332 return response(401, 'unauthorized: invalid token', { 'WWW-Authenticate' => 'Bearer realm="api", error="invalid_token"' })
333 end
334 return response(403, "forbidden: requires role #{route[:role]}") unless [route[:role], 'admin'].include?(@tokens[token])
335 end
336 extra = route[:types].length >= 2 ? { 'Vary' => 'Accept' } : {}
337 type = selected_type(route[:types], headers['accept'])
338 return response(406, 'not acceptable: ' + route[:types].join(', '), extra) unless type
339 body = body_for(route, params, query, headers, type)
340 tag = etag(body)
341 if headers.key?('if-none-match')
342 found = headers['if-none-match'].split(',', -1).any? do |part|
343 part = part.strip.delete_prefix('W/')
344 part == '*' || part == tag
345 end
346 if found
347 return lookup == 'GET' ? response(304, '', extra.merge('ETag' => tag)) : response(412, "precondition failed: #{tag}", extra)
348 end
349 end
350 response(200, body, extra.merge('ETag' => tag), type)
351 end

353 def request(line, header_lines)
354 @requests += 1
355 head = false
356 cors = false
357 headers = {}
358 match = /\A([A-Z]{1,16}) (\/[^ #]*) HTTP\/1\.1\z/.match(line)
359 if !match
360 result = response(400, 'bad request: request line')
361 else
362 method, target = match.captures
363 head = method == 'HEAD'
364 valid_escapes = true
365 position = 0
366 while position < target.length
367 if target[position] == '%'
368 hex = target[position + 1, 2]
369 unless hex && hex.match?(/\A[0-9A-Fa-f]{2}\z/) && hex.to_i(16).between?(32, 126)
370 valid_escapes = false
371 break
372 end
373 position += 3
374 else
375 position += 1
376 end
377 end
378 if !valid_escapes
379 result = response(400, 'bad request: percent escape')
380 else
381 result = nil
382 header_lines.each_with_index do |text, index|
383 name, value = text.split(':', 2)
384 if !value || !name.match?(/\A[A-Za-z0-9-]+\z/)
385 result = response(400, "bad request: header line #{index + 1}")
386 break
387 end
388 name = name.downcase
389 if headers.key?(name)
390 result = response(400, "bad request: duplicate header #{name}")
391 break
392 end
393 headers[name] = value.strip
394 end
395 unless result
396 cors = true
397 result = handle(method, target, headers)
398 end
399 end
400 end
401 if cors && headers.key?('origin') && @origins.include?(headers['origin'])
402 result[:headers]['Access-Control-Allow-Origin'] = headers['origin']
403 if result[:status] == 204 && headers.key?('access-control-request-method')
404 result[:headers]['Access-Control-Allow-Methods'] = result[:headers]['Allow']
405 end
406 end
407 status = result[:status]
408 @statuses[status] += 1
409 unless [204, 304].include?(status)
410 result[:headers]['Content-Type'] = status == 200 ? result[:type] : 'text/plain'
411 result[:headers]['Content-Length'] = result[:body].length.to_s
412 end
413 puts "REQUEST #{@requests}"
414 puts "HTTP/1.1 #{status} #{REASONS[status]}"
415 HEADERS.each { |name| puts "#{name}: #{result[:headers][name]}" if result[:headers].key?(name) }
416 puts result[:body] unless head || [204, 304].include?(status)
417 end

419 def urlfor(line)
420 _, name, *arguments = line.split(' ')
421 route = @routes.find { |item| item[:name] == name }
422 raise UrlError, "unknown route #{name}" unless route
423 parameter_names = route[:pieces].reject { |kind, _| kind == :literal }.map(&:last)
424 values = {}
425 queries = []
426 arguments.each do |argument|
427 key, value = argument.split('=', 2)
428 raise UrlError, "bad argument #{argument}" if value.nil? || key.empty?
429 if parameter_names.include?(key)
430 raise UrlError, "duplicate parameter #{key}" if values.key?(key)
431 values[key] = value
432 else
433 queries << [key, value]
434 end
435 end
436 route[:pieces].each do |kind, key|
437 next if kind == :literal
438 raise UrlError, "missing parameter #{key}" if kind != :wildcard && !values.key?(key)
439 text = values.fetch(key, '')
440 valid = case kind
441 when :parameter then text != '' && !%w[. ..].include?(text)
442 when :int then text.match?(/\A[0-9]{1,9}\z/)
443 when :slug then text.match?(/\A[a-z0-9-]+\z/)
444 else text == '' || text.split('/', -1).all? { |part| part != '' && !%w[. ..].include?(part) }
445 end
446 raise UrlError, "bad value for #{key}" unless valid
447 end
448 parts = route[:pieces].flat_map do |kind, key|
449 if kind == :literal
450 [key]
451 elsif kind == :wildcard
452 text = values.fetch(key, '')
453 text.empty? ? [] : text.split('/', -1).map { |part| encode(part) }
454 else
455 [encode(values[key])]
456 end
457 end
458 url = '/' + parts.join('/')
459 url += '?' + queries.map { |key, value| encode(key) + '=' + encode(value) }.join('&') unless queries.empty?
460 @url_ok += 1
461 puts "URLFOR #{name} #{url}"
462 rescue UrlError => error
463 @url_error += 1
464 puts "URLFOR ERROR #{error.message}"
465 end

467 def run
468 input = STDIN.each_line.map { |line| line.chomp("\n").delete_suffix("\r") }
469 position = 0
470 while input[position] != 'REQUESTS'
471 configure(input[position], position + 1)
472 position += 1
473 end
474 position += 1
475 puts "CONFIG routes=#{@routes.length} tokens=#{@tokens.length} redirects=#{@redirects.length} origins=#{@origins.length} limit=#{@limit ? @limit.join('/') : 'none'} errors=#{@errors}"
476 @routes.each { |route| route[:types] ||= ['application/json'] }
477 @ordered = @routes.sort_by { |route| priority(route) }
478 @ordered.each_with_index do |route, index|
479 methods = route[:methods] + (route[:methods].include?('GET') ? ['HEAD'] : [])
480 puts "TABLE #{index + 1} #{route[:name]} #{methods.sort.join(',')} #{route[:pattern]} role=#{route[:role] == 'none' ? '-' : route[:role]} types=#{route[:types].join(',')}"
481 end
482 while input[position] != 'END'
483 line = input[position]
484 position += 1
485 if line.start_with?('URLFOR ')
486 urlfor(line)
487 else
488 headers = []
489 while input[position] != ''
490 headers << input[position]
491 position += 1
492 end
493 position += 1
494 request(line, headers)
495 end
496 end
497 puts "SUMMARY requests=#{@requests} urlfor_ok=#{@url_ok} urlfor_error=#{@url_error}"
498 @statuses.keys.sort.each { |status| puts "STATUS #{status} #{@statuses[status]}" }
499 @routes.each { |route| puts "ROUTE #{route[:name]} #{route[:hits]}" }
500 @clients.keys.sort.each { |client| puts "CLIENT #{client} allowed=#{@clients[client][:allowed]} limited=#{@clients[client][:limited]}" }
501 end
502end
