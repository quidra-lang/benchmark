7private fun unreserved(c: Char) = c in 'A'..'Z' || c in 'a'..'z' || c in '0'..'9' || c in "-._~"

9private fun segment(s: String) = s.isNotEmpty() && s != "." && s != ".." && s.all(::unreserved)

11private fun canonical(s: String) =

14private fun mediaPart(s: String) = s.matches(Regex("[a-z0-9.+-]+"))

21private fun intParameter(s: String) = s.matches(Regex("[0-9]{1,9}"))

25private fun splitFirst(s: String, c: Char): Pair<String, String>? {
26 val at = s.indexOf(c)

30private fun hex(c: Char): Int? =
31 when (c) {
32 in '0'..'9' -> c - '0'
33 in 'A'..'F' -> c - 'A' + 10
34 in 'a'..'f' -> c - 'a' + 10
35 else -> null
36 }

38private fun validEscapes(s: String): Boolean {
39 var at = 0
40 while (at < s.length) {
41 if (s[at] == '%') {
42 if (at + 2 >= s.length) return false
43 val a = hex(s[at + 1]) ?: return false
44 val b = hex(s[at + 2]) ?: return false
45 if (a * 16 + b !in 32..126) return false
46 at += 3
47 } else at++
48 }
49 return true
50}
51
52private fun decode(s: String): String = buildString {
53 var at = 0
54 while (at < s.length) {
55 if (s[at] == '%') {
56 append((hex(s[at + 1])!! * 16 + hex(s[at + 2])!!).toChar())
57 at += 3
58 } else append(s[at++])
59 }
60}
61
62private fun encode(s: String) =
63 s.map { if (unreserved(it)) it.toString() else "%%%02X".format(it.code) }.joinToString("")
64
65private fun normalize(path: String): String {
66 val text = buildString {
67 var at = 0
68 while (at < path.length) {
69 if (path[at] == '%') {
70 val c = (hex(path[at + 1])!! * 16 + hex(path[at + 2])!!).toChar()
71 append(if (unreserved(c)) c.toString() else "%%%02X".format(c.code))
72 at += 3
73 } else append(path[at++])
74 }
75 }
76 val parts = mutableListOf<String>()
77 for (p in text.substring(1).split('/')) when (p) {
78 "",
79 "." -> Unit
80 ".." -> if (parts.isNotEmpty()) parts.removeAt(parts.lastIndex)
81 else -> parts.add(p)
82 }
83 return "/" + parts.joinToString("/")
84}

87
88private fun parsePattern(s: String): List<Piece>? {
89 if (s == "/") return emptyList()
90 if (!s.startsWith('/')) return null
91 val parts = s.substring(1).split('/')
92 val result = mutableListOf<Piece>()
93 val names = mutableSetOf<String>()
94 for ((i, p) in parts.withIndex()) {
95 if (segment(p)) {
96 result.add(Piece('l', p))
97 continue
98 }
99 if (!p.startsWith('{') || !p.endsWith('}')) return null
100 val inner = p.substring(1, p.lastIndex)
101 val kind: Char
102 val value: String
103 if (inner.startsWith('*')) {
104 if (i != parts.lastIndex) return null
105 kind = 'w'
106 value = inner.substring(1)
107 } else {
108 val typed = splitFirst(inner, ':')
109 if (typed != null) {
110 value = typed.first
111 kind =
112 when (typed.second) {
113 "int" -> 'i'
114 "slug" -> 's'
115 else -> return null
116 }
117 } else {
118 kind = 'p'
119 value = inner
120 }
121 }
122 if (!validName(value) || !names.add(value)) return null
123 result.add(Piece(kind, value))
124 }
125 return result
126}

128private data class Route(
129 val name: String,
130 val methods: Set<String>,
131 val pattern: String,
132 val pieces: List<Piece>,
133 val role: String,
134 var types: List<String>? = null,
135 var count: Int = 0,
136) {
137 fun fits(parts: List<String>): Boolean {
138 val wildcard = pieces.lastOrNull()?.kind == 'w'
139 val m = pieces.size - if (wildcard) 1 else 0
140 if (if (wildcard) parts.size < m else parts.size != m) return false
141 return (0 until m).all { i ->
142 val p = pieces[i]
143 val x = parts[i]
144 when (p.kind) {
145 'l' -> p.value == x
146 'i' -> intParameter(x)
147 's' -> slug(x)
148 else -> true
149 }
150 }
151 }

153 fun parameters(parts: List<String>): List<Pair<String, String>> =
154 pieces
155 .mapIndexedNotNull { i, p ->
156 val value =
157 when (p.kind) {
158 'l' -> return@mapIndexedNotNull null
159 'w' -> parts.drop(i).joinToString("/", transform = ::decode)
160 'i' -> parts[i].toInt().toString()
161 's' -> parts[i]
162 else -> decode(parts[i])
163 }
164 p.value to value
165 }
166 .sortedBy { it.first }

192 fun config(w: List<String>) {
193 val command = w[0]
194 val args = w.size - 1
195 fun require(condition: Boolean, reason: String) {
196 if (!condition) throw Problem(reason)
197 }
198 require(
199 command in setOf("ROUTE", "TOKEN", "REDIRECT", "ORIGIN", "LIMIT", "TYPES"),
200 "unknown command $command",
201 )
202 val correct =
203 when (command) {
204 "ROUTE" -> args == 3 || args == 5
205 "TOKEN",
206 "LIMIT" -> args == 2
207 "REDIRECT" -> args == 3
208 "ORIGIN" -> args == 1
209 else -> args >= 2
210 }
211 require(correct, "wrong number of arguments")
212 when (command) {
213 "ROUTE" -> {
214 require(validName(w[1]), "bad route name ${w[1]}")
215 require(routes.none { it.name == w[1] }, "duplicate route ${w[1]}")
216 val methods = w[2].split(',')
217 require(
218 methods.all { it in setOf("GET", "POST", "PUT", "PATCH", "DELETE") } &&
219 methods.toSet().size == methods.size,
220 "bad methods ${w[2]}",
221 )
222 val pieces = parsePattern(w[3]) ?: throw Problem("bad pattern ${w[3]}")
223 if (args == 5) {
224 require(w[4] == "ROLE", "unknown option ${w[4]}")
225 require(role(w[5]), "bad role ${w[5]}")
226 }
227 routes.add(
228 Route(w[1], methods.toSet(), w[3], pieces, if (args == 5) w[5] else "none"),
229 )
230 }
231 "TOKEN" -> {
232 require(w[1].matches(Regex("[A-Za-z0-9._-]{1,32}")), "bad token ${w[1]}")
233 require(w[1] !in tokens, "duplicate token ${w[1]}")
234 require(role(w[2]), "bad role ${w[2]}")
235 tokens[w[1]] = w[2]
236 }
237 "REDIRECT" -> {
238 fun rule(s: String) =
239 canonical(s) ||
240 (s.endsWith("/*") && s.dropLast(2) != "/" && canonical(s.dropLast(2)))
241 require(rule(w[1]), "bad path ${w[1]}")
242 require(rule(w[2]), "bad path ${w[2]}")
243 val p = w[1].endsWith("/*")
244 require(p == w[2].endsWith("/*"), "wildcard mismatch")
245 require(w[3] in setOf("301", "302", "307", "308"), "bad redirect code ${w[3]}")
246 val from = if (p) w[1].dropLast(2) else w[1]
247 val to = if (p) w[2].dropLast(2) else w[2]
248 require(
249 redirects.none { it.prefix == p && it.from == from },
250 "duplicate redirect ${w[1]}",
251 )
252 redirects.add(Redirect(from, to, p, w[3].toInt()))
253 }
254 "ORIGIN" -> {
255 require(w[1].matches(Regex("https?://[a-z0-9.:-]+")), "bad origin ${w[1]}")
256 require(w[1] !in origins, "duplicate origin ${w[1]}")
257 origins.add(w[1])
258 }
259 "LIMIT" -> {
260 val n = w[1].toIntOrNull()
261 val s = w[2].toIntOrNull()
262 require(
263 w[1].matches(Regex("[1-9][0-9]*")) &&
264 w[2].matches(Regex("[1-9][0-9]*")) &&
265 n != null &&
266 n in 1..100 &&
267 s != null &&
268 s in 1..3600,
269 "bad limit",
270 )
271 require(limit == null, "duplicate limit")
272 limit = n!! to s!!
273 }
274 else -> {
275 val r =
276 routes.firstOrNull { it.name == w[1] } ?: throw Problem("unknown route ${w[1]}")
277 require(r.types == null, "duplicate types ${w[1]}")
278 val seen = mutableSetOf<String>()
279 for (t in w.drop(2)) {
280 require(media(t), "bad media type $t")
281 require(seen.add(t), "duplicate media type $t")
282 }
283 r.types = w.drop(2)
284 }
285 }
286 }

312 fun request(line: String, raw: List<String>): Response {
313 val w = line.split(' ')
314 if (
315 w.size != 3 ||
316 !w[0].matches(Regex("[A-Z]{1,16}")) ||
317 !w[1].startsWith('/') ||
318 '#' in w[1] ||
319 w[2] != "HTTP/1.1"
320 )
321 return Response(400, "bad request: request line")
322 val head = w[0] == "HEAD"
323 fun bad(body: String) = Response(400, body, head = head)
324 if (!validEscapes(w[1])) return bad("bad request: percent escape")
325 val headers = mutableMapOf<String, String>()
326 for ((i, row) in raw.withIndex()) {
327 val pair = splitFirst(row, ':') ?: return bad("bad request: header line ${i+1}")
328 if (!pair.first.matches(Regex("[A-Za-z0-9-]+")))
329 return bad("bad request: header line ${i+1}")
330 val name = pair.first.lowercase()
331 if (name in headers) return bad("bad request: duplicate header $name")
332 headers[name] = pair.second.trim(' ')
333 }
334 val response = validRequest(w[0], w[1], headers)
335 response.head = head
336 val origin = headers["origin"]
337 if (origin != null && origin in origins) {
338 response.headers["Access-Control-Allow-Origin"] = origin
339 if (response.status == 204 && "access-control-request-method" in headers)
340 response.headers["Access-Control-Allow-Methods"] =
341 response.headers.getValue("Allow")
342 }
343 return response
344 }

346 fun validRequest(method: String, target: String, h: Map<String, String>): Response {
347 if (method !in setOf("GET", "HEAD", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"))
348 return Response(501, "not implemented: $method")
349 if (h["host"].isNullOrEmpty()) return Response(400, "bad request: missing host")
350 var effective = method
351 h["x-http-method-override"]?.let {
352 if (method != "POST" || it !in setOf("PUT", "PATCH", "DELETE"))
353 return Response(400, "bad request: method override")
354 effective = it
355 }
356 val client = h["x-client"]
357 if (!client.isNullOrEmpty()) {
358 val time = h.getValue("x-time").toInt()
359 val (n, s) = limit ?: (0 to 0)
360 val c = clients.getOrPut(client) { Client(n * s, time) }
361 if (limit != null) {
362 c.level = minOf(n * s, c.level + (time - c.last) * n)
363 c.last = time
364 if (c.level < s) {
365 c.limited++
366 return Response(
367 429,
368 "too many requests: client $client",
369 mutableMapOf("Retry-After" to ((s - c.level + n - 1) / n).toString()),
370 )
371 }
372 c.level -= s
373 }
374 c.allowed++
375 }
376 val split = splitFirst(target, '?')
377 val path = split?.first ?: target
378 val query = split?.second
379 val suffix = query?.let { "?$it" } ?: ""
380 val normalized = normalize(path)
381 if (normalized != path) {
382 val dest = normalized + suffix
383 return Response(308, "redirect to $dest", mutableMapOf("Location" to dest))
384 }
385 for (r in redirects) if (path == r.from || (r.prefix && path.startsWith(r.from + "/"))) {
386 val dest = r.to + path.substring(r.from.length) + suffix
387 return Response(r.code, "redirect to $dest", mutableMapOf("Location" to dest))
388 }
389 val parts = if (path == "/") emptyList() else path.substring(1).split('/')
390 val matching = priority.filter { routes[it].fits(parts) }
391 if (matching.isEmpty()) return Response(404, "not found: $path")
392 val allowed = mutableSetOf("OPTIONS")
393 matching.forEach { allowed.addAll(routes[it].methods) }
394 if ("GET" in allowed) allowed.add("HEAD")
395 val allow = allowed.sorted().joinToString(", ")
396 if (effective == "OPTIONS") return Response(204, headers = mutableMapOf("Allow" to allow))
397 val lookup = if (effective == "HEAD") "GET" else effective
398 val i =
399 matching.firstOrNull { lookup in routes[it].methods }
400 ?: return Response(
401 405,
402 "method not allowed: $effective",
403 mutableMapOf("Allow" to allow),
404 )
405 val route = routes[i]
406 route.count++
407 if (route.role != "none") {
408 val auth =
409 h["authorization"]
410 ?: return Response(
411 401,
412 "unauthorized: missing token",
413 mutableMapOf("WWW-Authenticate" to "Bearer realm=\"api\""),
414 )
415 val (scheme, token) = splitFirst(auth, ' ') ?: (auth to "")
416 val tokenRole = tokens[token]
417 if (scheme.lowercase() != "bearer" || tokenRole == null)
418 return Response(
419 401,
420 "unauthorized: invalid token",
421 mutableMapOf(
422 "WWW-Authenticate" to "Bearer realm=\"api\", error=\"invalid_token\"",
423 ),
424 )
425 if (tokenRole != route.role && tokenRole != "admin")
426 return Response(403, "forbidden: requires role ${route.role}")
427 }
428 val types = route.types!!
429 fun vary(r: Response): Response {
430 if (types.size >= 2) r.headers["Vary"] = "Accept"
431 return r
432 }
433 val selected = if ("accept" in h) negotiate(types, h.getValue("accept")) else types[0]
434 if (selected == null)
435 return vary(Response(406, "not acceptable: ${types.joinToString(", ")}"))
436 val params = route.parameters(parts)
437 val queries =
438 query
439 ?.split('&')
440 ?.filter { it.isNotEmpty() }
441 ?.map {
442 val (a, b) = splitFirst(it, '=') ?: (it to "")
443 decode(a.replace('+', ' ')) to decode(b.replace('+', ' '))
444 } ?: emptyList()
445 val cookieMap = sortedMapOf<String, String>()
446 h["cookie"]
447 ?.split(';')
448 ?.map { it.trim(' ') }
449 ?.forEach { p ->
450 val pair = splitFirst(p, '=')
451 if (pair != null && pair.first.isNotEmpty())
452 cookieMap.putIfAbsent(pair.first, pair.second)
453 }
454 val cookies = cookieMap.entries.map { it.key to it.value }
455 val body = makeBody(route.name, selected, params, queries, cookies)
456 val tag = etag(body)
457 if (
458 h["if-none-match"]?.split(',')?.any {
459 val value = it.trim(' ').removePrefix("W/")
460 value == "*" || value == tag
461 } == true
462 ) {
463 return if (lookup == "GET") vary(Response(304, headers = mutableMapOf("ETag" to tag)))
464 else vary(Response(412, "precondition failed: $tag"))
465 }
466 return vary(Response(200, body, mutableMapOf("ETag" to tag), contentType = selected))
467 }

469 fun urlfor(w: List<String>): String {
470 val route = routes.firstOrNull { it.name == w[1] } ?: throw Problem("unknown route ${w[1]}")
471 val names = route.pieces.filter { it.kind != 'l' }.map { it.value }.toSet()
472 val args = mutableListOf<Pair<String, String>>()
473 val params = mutableMapOf<String, String>()
474 for (arg in w.drop(2)) {
475 val (name, value) = splitFirst(arg, '=') ?: throw Problem("bad argument $arg")
476 if (name.isEmpty()) throw Problem("bad argument $arg")
477 if (name in names) {
478 if (name in params) throw Problem("duplicate parameter $name")
479 params[name] = value
480 }
481 args.add(name to value)
482 }
483 for (p in route.pieces.filter { it.kind != 'l' }) {
484 val value = params[p.value]
485 if (p.kind != 'w' && value == null) throw Problem("missing parameter ${p.value}")
486 val good =
487 when (p.kind) {
488 'p' -> !value.isNullOrEmpty() && value != "." && value != ".."
489 'i' -> intParameter(value!!)
490 's' -> slug(value!!)
491 else ->
492 value.isNullOrEmpty() ||
493 value.split('/').all { it.isNotEmpty() && it != "." && it != ".." }
494 }
495 if (!good) throw Problem("bad value for ${p.value}")
496 }
497 val pieces = mutableListOf<String>()
498 for (p in route.pieces) if (p.kind == 'l') pieces.add(p.value)
499 else {
500 val value = params[p.value] ?: ""
501 if (p.kind == 'w') {
502 if (value.isNotEmpty()) pieces.addAll(value.split('/').map(::encode))
503 } else pieces.add(encode(value))
504 }
505 var url = "/" + pieces.joinToString("/")
506 val queries =
507 args.filter { it.first !in names }.map { encode(it.first) + "=" + encode(it.second) }
508 if (queries.isNotEmpty()) url += "?" + queries.joinToString("&")
509 return "URLFOR ${route.name} $url"
510 }
511}

513private fun negotiate(types: List<String>, accept: String): String? {
514 val elements = mutableListOf<Triple<String, Int, Int>>()
515 for (element in accept.split(',')) {
516 val parts = element.split(';').map { it.trim(' ') }
517 val range = parts[0].lowercase()
518 val specificity =
519 when {
520 range == "*/*" -> 1
521 range.endsWith("/*") && mediaPart(range.dropLast(2)) -> 2
522 media(range) -> 3
523 else -> continue
524 }
525 val q =
526 parts
527 .drop(1)
528 .mapNotNull { splitFirst(it, '=') }
529 .firstOrNull { it.first.trim(' ').lowercase() == "q" }
530 ?.second
531 ?.trim(' ')
532 val weight =
533 if (q == null) 1000
534 else {
535 if (!q.matches(Regex("(?:0(?:\\.[0-9]{0,3})?|1(?:\\.0{0,3})?)"))) continue
536 if (q.startsWith('1')) 1000
537 else (splitFirst(q, '.')?.second ?: "").padEnd(3, '0').toInt()
538 }
539 elements.add(Triple(range, specificity, weight))
540 }
541 var selected: String? = null
542 var best = 0
543 for (t in types) {
544 var specificity = 0
545 var weight = 0
546 for ((range, s, w) in elements) if (
547 s > specificity &&
548 (range == "*/*" ||
549 range == t ||
550 (range.endsWith("/*") && range.dropLast(2) == t.substringBefore('/')))
551 ) {
552 specificity = s
553 weight = w
554 }
555 if (weight > best) {
556 best = weight
557 selected = t
558 }
559 }
560 return selected
561}

577private fun makeBody(
578 route: String,
579 type: String,
580 params: List<Pair<String, String>>,
581 query: List<Pair<String, String>>,
582 cookies: List<Pair<String, String>>,
583): String {
584 if (type == "application/json") {
585 fun obj(pairs: List<Pair<String, String>>) =
586 pairs.joinToString(",") { json(it.first) + ":" + json(it.second) }
587 return "{\"route\":" +
588 json(route) +
589 ",\"params\":{" +
590 obj(params) +
591 "},\"query\":[" +
592 query.joinToString(",") { "[" + json(it.first) + "," + json(it.second) + "]" } +
593 "],\"cookies\":{" +
594 obj(cookies) +
595 "}}"
596 }
597 if (type == "text/html")
598 return buildString {
599 append("<dl><dt>route</dt><dd>").append(html(route)).append("</dd>")
600 for ((kind, list) in
601 listOf("param" to params, "query" to query, "cookie" to cookies)) for ((a, b) in
602 list) append("<dt>$kind ${html(a)}</dt><dd>${html(b)}</dd>")
603 append("</dl>")
604 }
605 fun list(pairs: List<Pair<String, String>>) =
606 if (pairs.isEmpty()) "-"
607 else pairs.joinToString("&") { encode(it.first) + "=" + encode(it.second) }
608 return "route=$route params=${list(params)} query=${list(query)} cookies=${list(cookies)}"
609}

611private fun etag(body: String): String {
612 var a = 1
613 var b = 0
614 for (c in body) {
615 a = (a + c.code) % 65521
616 b = (b + a) % 65521
617 }
618 return "\"%04x%04x\"".format(b, a)
619}

664fun main() {
665 val lines = generateSequence(::readLine).toList()
666 val router = Router()
667 var at = 0
668 while (lines[at] != "REQUESTS") {
669 try {
670 router.config(lines[at].split(' '))
671 } catch (p: Problem) {
672 router.configErrors++
673 println("CONFIG ERROR ${at+1}: ${p.reason}")
674 }
675 at++
676 }
677 at++
678 router.table()
679 var requests = 0
680 var urlOK = 0
681 var urlError = 0
682 while (lines[at] != "END") {
683 val line = lines[at++]
684 if (line.startsWith("URLFOR ")) {
685 try {
686 println(router.urlfor(line.split(' ')))
687 urlOK++
688 } catch (p: Problem) {
689 urlError++
690 println("URLFOR ERROR ${p.reason}")
691 }
692 } else {
693 requests++
694 val headers = mutableListOf<String>()
695 while (lines[at].isNotEmpty()) headers.add(lines[at++])
696 at++
697 val r = router.request(line, headers)
698 router.statuses[r.status] = (router.statuses[r.status] ?: 0) + 1
699 output(requests, r)
700 }
701 }
702 println("SUMMARY requests=$requests urlfor_ok=$urlOK urlfor_error=$urlError")
703 for ((status, count) in router.statuses) println("STATUS $status $count")
704 for (r in router.routes) println("ROUTE ${r.name} ${r.count}")
705 for ((name, c) in router.clients) println(
706 "CLIENT $name allowed=${c.allowed} limited=${c.limited}",
707 )
708}
