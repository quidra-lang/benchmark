44parse_pattern <- function(text) {
45 if (text == "/") return(list())
46 if (!startsWith(text, "/")) return(NULL)
47 parts <- split_keep(substring(text, 2), "/")
48 pieces <- list()
49 names <- character()
50 for (index in seq_along(parts)) {
51 part <- parts[index]
52 if (segment(part)) {
53 pieces[[length(pieces) + 1]] <- list(kind = "literal", name = part)
54 next
55 }
56 match <- regmatches(part, regexec("^\\{(\\*?)([a-z][a-z0-9_]{0,19})(:(int|slug))?\\}$", part))[[1]]
57 if (!length(match)) return(NULL)
58 star <- match[2] == "*"
59 name <- match[3]
60 type <- match[5]
61 if (name %in% names || (star && (type != "" || index != length(parts)))) return(NULL)
62 names <- c(names, name)
63 pieces[[length(pieces) + 1]] <- list(kind = if (star) "wildcard" else if (type == "") "parameter" else type, name = name)
64 }

143 output <- character()

145 while (position <= nchar(text)) {

147 if (char == "%") {
148 output <- c(output, intToUtf8(strtoi(substr(text, position + 1, position + 2), 16)))

181match_route <- function(route, segments) {
182 pieces <- route$pieces
183 wildcard <- length(pieces) > 0 && pieces[[length(pieces)]]$kind == "wildcard"
184 count <- length(pieces) - as.integer(wildcard)
185 if (if (wildcard) length(segments) < count else length(segments) != count) return(NULL)
186 parameters <- list()
187 for (index in seq_along(pieces)) {
188 piece <- pieces[[index]]
189 kind <- piece$kind
190 name <- piece$name
191 text <- if (index <= length(segments)) segments[index] else ""
192 if (kind == "literal") {
193 if (text != name) return(NULL)
194 } else if (kind == "parameter") {
195 parameters[name] <- list(decode(text))
196 } else if (kind == "int") {
197 if (!grepl("^[0-9]{1,9}$", text)) return(NULL)
198 parameters[name] <- list(sprintf("%.0f", as.numeric(text)))
199 } else if (kind == "slug") {
200 if (!grepl("^[a-z0-9-]+$", text)) return(NULL)
201 parameters[name] <- list(text)
202 } else {
203 rest <- if (length(segments) > count) segments[seq.int(count + 1, length(segments))] else character()
204 parameters[name] <- list(paste(vapply(rest, decode, character(1)), collapse = "/"))
205 }
206 }
207 parameters

209selected_type <- function(types, accept) {
210 if (is.null(accept)) return(types[1])
211 elements <- list()
212 for (element in split_keep(accept, ",")) {
213 parts <- vapply(split_keep(element, ";"), strip_spaces, character(1))
214 range <- tolower(parts[1])
215 if (!(range == "*/*" || grepl("^[a-z0-9.+-]+/\\*$", range) || grepl(media_pattern, range))) next
216 q <- NULL
217 for (parameter in parts[-1]) {
218 pair <- first_pair(parameter, "=")
219 if (pair$present && tolower(strip_spaces(pair$left)) == "q") {
220 q <- strip_spaces(pair$right)
221 break
222 }
223 }
224 if (!is.null(q) && !grepl("^(0(\\.[0-9]{0,3})?|1(\\.0{0,3})?)$", q)) next
225 weight <- if (is.null(q)) 1000 else round(as.numeric(q) * 1000)
226 specificity <- if (range == "*/*") 1 else if (endsWith(range, "/*")) 2 else 3
227 elements[[length(elements) + 1]] <- list(range = range, specificity = specificity, weight = weight)
228 }
229 best <- NULL
230 best_weight <- 0
231 for (type in types) {
232 specificity <- 0
233 weight <- 0
234 for (element in elements) {
235 range <- element$range
236 fits <- range == "*/*" || range == type || (element$specificity == 2 && first_pair(range, "/")$left == first_pair(type, "/")$left)
237 if (fits && element$specificity > specificity) {
238 specificity <- element$specificity
239 weight <- element$weight
240 }
241 }
242 if (weight > best_weight) {
243 best_weight <- weight
244 best <- type
245 }
246 }
247 best

285 }

292etag <- function(body) {
293 a <- 1
294 b <- 0
295 for (char in utf8ToInt(body)) {
296 a <- (a + char) %% 65521
297 b <- (b + a) %% 65521
298 }
299 sprintf('"%04x%04x"', as.integer(b), as.integer(a))
300}
301response <- function(status, body = "", headers = list(), type = NULL) list(status = status, body = body, headers = headers, type = type)
302handle <- function(method, target, headers) {
303 if (!(method %in% c("GET", "HEAD", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"))) return(response(501, paste0("not implemented: ", method)))
304 if (!("host" %in% names(headers)) || headers[["host"]] == "") return(response(400, "bad request: missing host"))
305 effective <- method
306 if ("x-http-method-override" %in% names(headers)) {
307 override <- headers[["x-http-method-override"]]
308 if (method != "POST" || !(override %in% c("PUT", "PATCH", "DELETE"))) return(response(400, "bad request: method override"))
309 effective <- override
310 }
311 client <- headers[["x-client"]]
312 if (!is.null(client) && client != "") {
313 time <- as.numeric(headers[["x-time"]])
314 if (!(client %in% names(clients))) clients[[client]] <<- object(allowed = 0, limited = 0, level = if (is.null(rate_limit)) 0 else prod(rate_limit), last = time)
315 record <- clients[[client]]
316 if (!is.null(rate_limit)) {
317 capacity <- rate_limit[1]
318 period <- rate_limit[2]
319 record$level <- min(capacity * period, record$level + (time - record$last) * capacity)
320 record$last <- time
321 if (record$level < period) {
322 record$limited <- record$limited + 1
323 wait <- ceiling((period - record$level) / capacity)
324 return(response(429, paste0("too many requests: client ", client), list("Retry-After" = as.character(wait))))
325 }
326 record$level <- record$level - period
327 }
328 record$allowed <- record$allowed + 1
329 }
330 target_parts <- first_pair(target, "?")
331 path <- target_parts$left
332 query <- if (target_parts$present) target_parts$right else NULL
333 suffix <- if (target_parts$present) paste0("?", query) else ""
334 normalized <- normalize(path)
335 if (normalized != path) return(response(308, paste0("redirect to ", normalized, suffix), list(Location = paste0(normalized, suffix))))
336 for (rule in redirects) {
337 if (path == rule$base || (rule$wildcard && startsWith(path, paste0(rule$base, "/")))) {
338 destination <- paste0(rule$to, text_slice(path, nchar(rule$base) + 1), suffix)
339 return(response(rule$code, paste0("redirect to ", destination), list(Location = destination)))
340 }
341 }
342 segments <- if (path == "/") character() else split_keep(substring(path, 2), "/")
343 matches <- list()
344 allowed <- "OPTIONS"
345 for (route in ordered_routes) {
346 parameters <- match_route(route, segments)
347 if (!is.null(parameters)) {
348 matches[[length(matches) + 1]] <- list(route = route, parameters = parameters)
349 allowed <- c(allowed, route$methods)
350 }
351 }
352 if (!length(matches)) return(response(404, paste0("not found: ", path)))
353 if ("GET" %in% allowed) allowed <- c(allowed, "HEAD")
354 allow <- paste(sort(unique(allowed), method = "radix"), collapse = ", ")
355 if (effective == "OPTIONS") return(response(204, headers = list(Allow = allow)))
356 lookup <- if (effective == "HEAD") "GET" else effective
357 selected <- NULL
358 for (candidate in matches) {
359 if (lookup %in% candidate$route$methods) {
360 selected <- candidate
361 break
362 }
363 }
364 if (is.null(selected)) return(response(405, paste0("method not allowed: ", effective), list(Allow = allow)))
365 route <- selected$route
366 parameters <- selected$parameters
367 route$hits <- route$hits + 1
368 if (route$role != "none") {
369 if (!("authorization" %in% names(headers))) return(response(401, "unauthorized: missing token", list("WWW-Authenticate" = 'Bearer realm="api"')))
370 authorization <- first_pair(headers[["authorization"]], " ")
371 token <- authorization$right
372 if (tolower(authorization$left) != "bearer" || !(token %in% names(tokens))) return(response(401, "unauthorized: invalid token", list("WWW-Authenticate" = 'Bearer realm="api", error="invalid_token"')))
373 if (!(tokens[[token]] %in% c(route$role, "admin"))) return(response(403, paste0("forbidden: requires role ", route$role)))
374 }
375 extra <- if (length(route$types) >= 2) list(Vary = "Accept") else list()
376 type <- selected_type(route$types, headers[["accept"]])
377 if (is.null(type)) return(response(406, paste0("not acceptable: ", paste(route$types, collapse = ", ")), extra))
378 body <- body_for(route, parameters, query, headers, type)
379 tag <- etag(body)
380 if ("if-none-match" %in% names(headers)) {
381 for (part in split_keep(headers[["if-none-match"]], ",")) {
382 part <- strip_spaces(part)
383 if (startsWith(part, "W/")) part <- substring(part, 3)
384 if (part == "*" || part == tag) {
385 if (lookup == "GET") return(response(304, headers = c(extra, list(ETag = tag))))
386 return(response(412, paste0("precondition failed: ", tag), extra))
387 }
388 }
389 }
390 response(200, body, c(extra, list(ETag = tag)), type)
391}
392request <- function(line, header_lines) {
393 request_count <<- request_count + 1
394 head <- FALSE
395 cors <- FALSE
396 headers <- list()
397 match <- regmatches(line, regexec("^([A-Z]{1,16}) (/[^ #]*) HTTP/1\\.1$", line))[[1]]
398 if (!length(match)) {
399 result <- response(400, "bad request: request line")
400 } else {
401 method <- match[2]
402 target <- match[3]
403 head <- method == "HEAD"
404 valid_escapes <- TRUE
405 position <- 1
406 while (position <= nchar(target)) {
407 if (substr(target, position, position) == "%") {
408 hex <- substr(target, position + 1, position + 2)
409 if (!grepl("^[0-9A-Fa-f]{2}$", hex) || strtoi(hex, 16) < 32 || strtoi(hex, 16) > 126) {
410 valid_escapes <- FALSE
411 break
412 }
413 position <- position + 3
414 } else position <- position + 1
415 }
416 if (!valid_escapes) {
417 result <- response(400, "bad request: percent escape")
418 } else {
419 result <- NULL
420 for (index in seq_along(header_lines)) {
421 pair <- first_pair(header_lines[index], ":")
422 if (!pair$present || !grepl("^[A-Za-z0-9-]+$", pair$left)) {
423 result <- response(400, paste0("bad request: header line ", index))
424 break
425 }
426 name <- tolower(pair$left)
427 if (name %in% names(headers)) {
428 result <- response(400, paste0("bad request: duplicate header ", name))
429 break
430 }
431 headers[name] <- list(strip_spaces(pair$right))
432 }
433 if (is.null(result)) {
434 cors <- TRUE
435 result <- handle(method, target, headers)
436 }
437 }
438 }
439 if (cors && "origin" %in% names(headers) && headers[["origin"]] %in% origins) {
440 result$headers[["Access-Control-Allow-Origin"]] <- headers[["origin"]]
441 if (result$status == 204 && "access-control-request-method" %in% names(headers)) result$headers[["Access-Control-Allow-Methods"]] <- result$headers[["Allow"]]
442 }
443 status <- result$status
444 key <- as.character(status)
445 statuses[key] <<- if (key %in% names(statuses)) statuses[key] + 1 else 1
446 if (!(status %in% c(204, 304))) {
447 result$headers[["Content-Type"]] <- if (status == 200) result$type else "text/plain"
448 result$headers[["Content-Length"]] <- as.character(nchar(result$body))
449 }
450 emit("REQUEST ", request_count)
451 emit("HTTP/1.1 ", status, " ", reasons[key])
452 for (name in header_order) if (name %in% names(result$headers)) emit(name, ": ", result$headers[[name]])
453 if (!head && !(status %in% c(204, 304))) emit(result$body)
454}
455urlfor <- function(line) {
456 words <- split_keep(line, " ")
457 name <- words[2]
458 arguments <- if (length(words) > 2) words[seq.int(3, length(words))] else character()
459 tryCatch({
460 route <- find_route(name)
461 if (is.null(route)) fail_url(paste0("unknown route ", name))
462 parameter_names <- vapply(Filter(function(piece) piece$kind != "literal", route$pieces), function(piece) piece$name, character(1))
463 values <- list()
464 queries <- list()
465 for (argument in arguments) {
466 pair <- first_pair(argument, "=")
467 key <- pair$left
468 if (!pair$present || key == "") fail_url(paste0("bad argument ", argument))
469 if (key %in% parameter_names) {
470 if (key %in% names(values)) fail_url(paste0("duplicate parameter ", key))
471 values[key] <- list(pair$right)
472 } else queries[[length(queries) + 1]] <- c(key, pair$right)
473 }
474 for (piece in route$pieces) {
475 kind <- piece$kind
476 key <- piece$name
477 if (kind == "literal") next
478 if (kind != "wildcard" && !(key %in% names(values))) fail_url(paste0("missing parameter ", key))
479 text <- if (key %in% names(values)) values[[key]] else ""
480 valid <- switch(kind, parameter = text != "" && !(text %in% c(".", "..")), int = grepl("^[0-9]{1,9}$", text), slug = grepl("^[a-z0-9-]+$", text),
481 wildcard = text == "" || all(vapply(split_keep(text, "/"), function(part) !(part %in% c("", ".", "..")), logical(1))))
482 if (!valid) fail_url(paste0("bad value for ", key))
483 }
484 parts <- character()
485 for (piece in route$pieces) {
486 kind <- piece$kind
487 key <- piece$name
