190 func fits(_ parts: [String]) -> Bool {

592 for p in route.pieces where p.kind != "l" {

599 good = value != "" && value != "." && value != ".."
600 case "i":
601 good = intParameter(value!)
602 case "s":
603 good = slug(value!)
604 default:
605 good = value == nil || value == "" || value!.components(separatedBy: "/")

607 if !good {
608 throw Problem(reason: "bad value for \(p.value)")
609 }
610 }
611 var pieces: [String] = []
612 for p in route.pieces {
613 if p.kind == "l" {
614 pieces.append(p.value)
615 } else {
616 let value = params[p.value] ?? ""
617 if p.kind == "w" {
618 if !value.isEmpty {
619 pieces += value.components(separatedBy: "/").map(encode)
620 }
621 } else {
622 pieces.append(encode(value))
623 }
624 }
625 }
626 var url = "/" + pieces.joined(separator: "/")

634func negotiate(_ types: [String], _ accept: String) -> String? {
635 var elements: [(String, Int, Int)] = []
636 for element in accept.components(separatedBy: ",") {
637 let parts = element.components(separatedBy: ";").map(trim), range = parts[0].lowercased(), specificity: Int
638 if range == "*/*" {
639 specificity = 1
640 } else if range.hasSuffix("/*") && mediaPart(String(range.dropLast(2))) {
641 specificity = 2
642 } else if media(range) {
643 specificity = 3
644 } else {
645 continue
646 }
647 var q: String? = nil
648 for p in parts
649 .dropFirst()
650 {
651 if let (name, value) = splitFirst(p, "="), trim(name).lowercased() == "q" {
652 q = trim(value)
653 break
654 }
655 }
656 let weight: Int
657 if let q {
658 guard matches(q, "(?:0(?:\\.[0-9]{0,3})?|1(?:\\.0{0,3})?)") else {
659 continue
660 }
661 if q.hasPrefix("1") {
662 weight = 1000
663 } else {
664 let f = splitFirst(q, ".")?.1 ?? ""
665 weight = Int(f + String(repeating: "0", count: 3 - f.count))!
666 }
667 } else {
668 weight = 1000
669 }
670 elements.append((range, specificity, weight))
671 }
672 var selected: String? = nil, best = 0
673 for t in types {
674 var specificity = 0, weight = 0
675 for (range, s, w) in elements where s > specificity {
676 if range == "*/*" || range == t ||
677 (range.hasSuffix("/*") && String(range.dropLast(2)) == t.components(separatedBy: "/")[0])
678 {
679 specificity = s
680 weight = w
681 }
682 }
683 if weight > best {
684 best = weight
685 selected = t
686 }
687 }
688 return selected
689}

698 return "&amp;"
699 case "<":
700 return "&lt;"
701 case ">":
702 return "&gt;"
703 case "\"":
704 return "&quot;"
705 default:
706 return String(c) } }.joined()
707}
708func makeBody(
709 _ route: String,
710 _ type: String,
711 _ params: [(String, String)],
712 _ query: [(String, String)],
713 _ cookies: [(String, String)]
714) -> String {
715 if type == "application/json" {
716 func object(_ pairs: [(String, String)]) -> String {
717 pairs.map { json($0.0) + ":" + json($0.1) }
718 .joined(separator: ",")
719 }
720 return "{\"route\":" + json(route) + ",\"params\":{" + object(params) + "},\"query\":[" + query
721 .map { "[" + json($0.0) + "," + json($0.1) + "]" }
722 .joined(separator: ",") + "],\"cookies\":{" + object(cookies) + "}}"
723 }
724 if type == "text/html" {
725 var result = "<dl><dt>route</dt><dd>" + html(route) + "</dd>"
726 for (kind, list) in [("param", params), ("query", query), ("cookie", cookies)] {
727 for (a, b) in list {
728 result += "<dt>\(kind) \(html(a))</dt><dd>\(html(b))</dd>"
729 }
730 }
731 return result + "</dl>"
732 }
733 func list(_ pairs: [(String, String)]) -> String {
734 pairs.isEmpty ? "-" : pairs
735 .map { encode($0.0) + "=" + encode($0.1) }.joined(separator: "&")
736 }
737 return "route=\(route) params=\(list(params)) query=\(list(query)) cookies=\(list(cookies))"
738}
739func etag(_ body: String) -> String {
740 var a = 1, b = 0
741 for c in body.utf8 {
742 a = (a + Int(c)) % 65521
743 b = (b + a) % 65521
744 }
745 return String(format: "\"%04x%04x\"", b, a)
746}
