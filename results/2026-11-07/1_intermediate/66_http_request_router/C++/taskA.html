19std::vector<std::string> split(const std::string& text, char separator) {
20 std::vector<std::string> result;
21 std::size_t start = 0;
22 for (std::size_t i = 0; i <= text.size(); ++i) {
23 if (i == text.size() || text[i] == separator) {
24 result.push_back(text.substr(start, i - start));
25 start = i + 1;
26 }
27 }
28 return result;
29}

31std::string trim(const std::string& text) {
32 std::size_t first = text.find_first_not_of(' ');
33 if (first == std::string::npos) return "";
34 return text.substr(first, text.find_last_not_of(' ') - first + 1);
35}

58bool valid_name(const std::string& name) { return matches(name, "[a-z][a-z0-9_]{0,19}"); }

60bool unreserved(char c) {
61 return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') ||
62 std::string("-._~").find(c) != std::string::npos;
63}

65bool segment(const std::string& text) {
66 return !text.empty() && text != "." && text != ".." &&
67 std::all_of(text.begin(), text.end(), unreserved);

70bool canonical_path(const std::string& path) {
71 if (path == "/") return true;
72 if (path.empty() || path[0] != '/') return false;
73 for (const std::string& part : split(path.substr(1), '/')) if (!segment(part)) return false;
74 return true;
75}

77bool decimal_between(const std::string& text, int low, int high) {
78 if (!matches(text, "0|[1-9][0-9]*")) return false;
79 int value = 0;
80 for (char c : text) {
81 if (value > high / 10 || (value == high / 10 && c - '0' > high % 10)) return false;
82 value = value * 10 + c - '0';
83 }
84 return value >= low;
85}

87int hex_digit(char c) {
88 if (c >= '0' && c <= '9') return c - '0';
89 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
90 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
91 return -1;
92}

94bool valid_escapes(const std::string& text) {
95 for (std::size_t i = 0; i < text.size(); ++i) {
96 if (text[i] != '%') continue;
97 if (i + 2 >= text.size() || hex_digit(text[i + 1]) < 0 || hex_digit(text[i + 2]) < 0) return false;
98 int code = hex_digit(text[i + 1]) * 16 + hex_digit(text[i + 2]);
99 if (code < 32 || code > 126) return false;
100 i += 2;
101 }
102 return true;
103}

105std::string decode(const std::string& text) {
106 std::string result;
107 for (std::size_t i = 0; i < text.size(); ++i) {
108 if (text[i] == '%') {
109 result += static_cast<char>(hex_digit(text[i + 1]) * 16 + hex_digit(text[i + 2]));
110 i += 2;
111 } else result += text[i];
112 }
113 return result;
114}

116std::string decode_form(std::string text) {
117 std::replace(text.begin(), text.end(), '+', ' ');
118 return decode(text);

121std::string encode(const std::string& text) {
122 static const char* digits = "0123456789ABCDEF";
123 std::string result;
124 for (unsigned char c : text) {
125 if (unreserved(static_cast<char>(c))) result += static_cast<char>(c);
126 else {
127 result += '%';
128 result += digits[c >> 4];
129 result += digits[c & 15];
130 }
131 }
132 return result;
133}

136 std::vector<std::string> values;
137 for (const auto& pair : pairs) values.push_back(encode(pair.first) + "=" + encode(pair.second));
138 return join(values, "&");

141std::string normalize(const std::string& path) {
142 std::string changed;
143 static const char* digits = "0123456789ABCDEF";
144 for (std::size_t i = 0; i < path.size(); ++i) {
145 if (path[i] != '%') {
146 changed += path[i];
147 continue;
148 }
149 int code = hex_digit(path[i + 1]) * 16 + hex_digit(path[i + 2]);
150 if (unreserved(static_cast<char>(code))) changed += static_cast<char>(code);
151 else {
152 changed += '%';
153 changed += digits[code >> 4];
154 changed += digits[code & 15];
155 }
156 i += 2;
157 }
158 std::vector<std::string> stack;
159 for (const std::string& part : split(changed.substr(1), '/')) {
160 if (part.empty() || part == ".") continue;
161 if (part == "..") { if (!stack.empty()) stack.pop_back(); }
162 else stack.push_back(part);
163 }
164 return "/" + join(stack, "/");
165}

167enum class PieceKind { Literal, Untyped, Int, Slug, Wildcard };
168struct Piece {
169 PieceKind kind;
170 std::string text;
171 bool fits(const std::string& value) const {
172 if (kind == PieceKind::Literal) return text == value;
173 if (kind == PieceKind::Int) return matches(value, "[0-9]{1,9}");
174 if (kind == PieceKind::Slug) return matches(value, "[a-z0-9-]+");
175 return true;
176 }
177 bool accepts(const std::string& value) const {
178 if (kind == PieceKind::Untyped) return !value.empty() && value != "." && value != "..";
179 if (kind == PieceKind::Wildcard) {
180 if (value.empty()) return true;
181 for (const std::string& part : split(value, '/'))
182 if (part.empty() || part == "." || part == "..") return false;
183 return true;
184 }
185 return fits(value);
186 }
187};

189std::optional<std::vector<Piece>> parse_pattern(const std::string& pattern) {
190 if (pattern.empty() || pattern[0] != '/') return std::nullopt;
191 std::vector<Piece> pieces;
192 if (pattern == "/") return pieces;
193 std::set<std::string> parameters;
194 std::vector<std::string> parts = split(pattern.substr(1), '/');
195 for (std::size_t i = 0; i < parts.size(); ++i) {
196 const std::string& text = parts[i];
197 if (segment(text)) {
198 pieces.push_back({PieceKind::Literal, text});
199 continue;
200 }
201 if (text.size() < 3 || text.front() != '{' || text.back() != '}') return std::nullopt;
202 std::string inner = text.substr(1, text.size() - 2), name = inner;
203 PieceKind kind = PieceKind::Untyped;
204 if (!inner.empty() && inner[0] == '*') {
205 if (i + 1 != parts.size()) return std::nullopt;
206 name = inner.substr(1);
207 kind = PieceKind::Wildcard;
208 } else {
209 std::size_t colon = inner.find(':');
210 if (colon != std::string::npos) {
211 name = inner.substr(0, colon);
212 std::string type = inner.substr(colon + 1);
213 if (type == "int") kind = PieceKind::Int;
214 else if (type == "slug") kind = PieceKind::Slug;
215 else return std::nullopt;
216 }
217 }
218 if (!valid_name(name) || !parameters.insert(name).second) return std::nullopt;
219 pieces.push_back({kind, name});
220 }
221 return pieces;
222}

224struct Route {
225 std::string name, pattern, role;
226 std::set<std::string> methods;
227 std::vector<Piece> pieces;
228 std::vector<std::string> types;
229 int selections = 0;
230
231 bool wildcard() const { return !pieces.empty() && pieces.back().kind == PieceKind::Wildcard; }
232 int literal_count() const {
233 return static_cast<int>(std::count_if(pieces.begin(), pieces.end(), [](const Piece& p) { return p.kind == PieceKind::Literal; }));
234 }
235 int typed_count() const {
236 return static_cast<int>(std::count_if(pieces.begin(), pieces.end(), [](const Piece& p) {
237 return p.kind == PieceKind::Int || p.kind == PieceKind::Slug;
238 }));
239 }
240 bool matches_path(const std::vector<std::string>& segments) const {
241 std::size_t count = pieces.size() - (wildcard() ? 1 : 0);
242 if (segments.size() < count || (!wildcard() && segments.size() != count)) return false;
243 for (std::size_t i = 0; i < count; ++i) if (!pieces[i].fits(segments[i])) return false;
244 return true;
245 }
246 Pairs parameters(const std::vector<std::string>& segments) const {
247 Pairs result;
248 for (std::size_t i = 0; i < pieces.size(); ++i) {
249 const Piece& piece = pieces[i];
250 if (piece.kind == PieceKind::Literal) continue;
251 std::string value;
252 if (piece.kind == PieceKind::Wildcard) {
253 std::vector<std::string> rest;
254 for (std::size_t j = i; j < segments.size(); ++j) rest.push_back(decode(segments[j]));
255 value = join(rest, "/");
256 } else if (piece.kind == PieceKind::Int) value = std::to_string(std::stoi(segments[i]));
257 else value = piece.kind == PieceKind::Untyped ? decode(segments[i]) : segments[i];
258 result.emplace_back(piece.text, value);
259 }
260 std::sort(result.begin(), result.end());
261 return result;
262 }
263};

317Pairs query_pairs(const std::optional<std::string>& query) {
318 Pairs pairs;
319 if (!query) return pairs;
320 for (const std::string& part : split(*query, '&')) {
321 if (part.empty()) continue;
322 std::size_t equals = part.find('=');
323 pairs.emplace_back(decode_form(part.substr(0, equals)),
324 equals == std::string::npos ? "" : decode_form(part.substr(equals + 1)));
325 }
326 return pairs;
327}

329Pairs cookie_pairs(const Headers& headers) {
330 auto found = headers.find("cookie");
331 if (found == headers.end()) return {};
332 std::map<std::string, std::string> values;
333 for (const std::string& raw : split(found->second, ';')) {
334 std::string part = trim(raw);
335 std::size_t equals = part.find('=');
336 if (equals == std::string::npos || equals == 0) continue;
337 values.emplace(part.substr(0, equals), part.substr(equals + 1));
338 }
339 return Pairs(values.begin(), values.end());
340}

342std::string json_pairs(const Pairs& pairs, bool array) {
343 std::vector<std::string> items;
344 for (const auto& pair : pairs) {
345 std::string item = quoted(pair.first) + (array ? "," : ":") + quoted(pair.second);
346 items.push_back(array ? "[" + item + "]" : item);

348 return std::string(array ? "[" : "{") + join(items, ",") + (array ? "]" : "}");

351std::string body_for(const std::string& type, const std::string& route, const Pairs& parameters,
352 const Pairs& query, const Pairs& cookies) {
353 if (type == "application/json")
354 return "{\"route\":" + quoted(route) + ",\"params\":" + json_pairs(parameters, false) +
355 ",\"query\":" + json_pairs(query, true) + ",\"cookies\":" + json_pairs(cookies, false) + "}";
356 if (type == "text/html") {
357 std::string body = "<dl><dt>route</dt><dd>" + html(route) + "</dd>";
358 auto append = [&](const Pairs& pairs, const std::string& kind) {
359 for (const auto& pair : pairs)
360 body += "<dt>" + kind + " " + html(pair.first) + "</dt><dd>" + html(pair.second) + "</dd>";
361 };
362 append(parameters, "param");
363 append(query, "query");
364 append(cookies, "cookie");
365 return body + "</dl>";
366 }
367 auto list = [](const Pairs& pairs) { return pairs.empty() ? "-" : encoded_pairs(pairs); };
368 return "route=" + route + " params=" + list(parameters) + " query=" + list(query) + " cookies=" + list(cookies);
369}

371std::string etag_for(const std::string& body) {
372 std::uint32_t a = 1, b = 0;
373 for (unsigned char c : body) {
374 a = (a + c) % 65521;
375 b = (b + a) % 65521;
376 }
377 std::ostringstream result;
378 result << '"' << std::hex << std::setfill('0') << std::setw(4) << b << std::setw(4) << a << '"';
379 return result.str();
380}

382std::optional<std::string> negotiate(const std::vector<std::string>& types, const Headers& headers) {
383 auto accept = headers.find("accept");
384 if (accept == headers.end()) return types.front();
385 struct Range {
386 std::string text;
387 int specificity, weight;
388 };
389 std::vector<Range> ranges;
390 for (const std::string& element : split(accept->second, ',')) {
391 std::vector<std::string> parts = split(element, ';');
392 std::string range = lower(trim(parts[0]));
393 int specificity = range == "*/*" ? 1 : matches(range, "[a-z0-9.+-]+/\\*") ? 2 : media_type(range) ? 3 : 0;
394 if (!specificity) continue;
395 int weight = 1000;
396 bool valid = true;
397 for (std::size_t i = 1; i < parts.size(); ++i) {
398 std::size_t equals = parts[i].find('=');
399 if (equals == std::string::npos || lower(trim(parts[i].substr(0, equals))) != "q") continue;
400 std::string q = trim(parts[i].substr(equals + 1));
401 if (!matches(q, "0(\\.[0-9]{0,3})?|1(\\.0{0,3})?")) valid = false;
402 else {
403 weight = (q[0] - '0') * 1000;
404 int place = 100;
405 for (std::size_t j = 2; j < q.size(); ++j) {
406 weight += (q[j] - '0') * place;
407 place /= 10;
408 }
409 }
410 break;
411 }
412 if (valid) ranges.push_back({range, specificity, weight});
413 }
414 int best_weight = 0;
415 std::optional<std::string> selected;
416 for (const std::string& type : types) {
417 int specificity = 0, weight = 0;
418 for (const Range& range : ranges) {
419 bool fits = range.specificity == 1 || range.text == type ||
420 (range.specificity == 2 && range.text.substr(0, range.text.size() - 1) ==
421 type.substr(0, type.find('/') + 1));
422 if (fits && range.specificity > specificity) {
423 specificity = range.specificity;
424 weight = range.weight;
425 }
426 }
427 if (weight > best_weight) {
428 best_weight = weight;
429 selected = type;
430 }
431 }
432 return selected;
433}

447 void configure(const std::vector<std::string>& w) {
448 const std::string& command = w[0];
449 if (command != "ROUTE" && command != "TOKEN" && command != "REDIRECT" && command != "ORIGIN" &&
450 command != "LIMIT" && command != "TYPES") throw std::runtime_error("unknown command " + command);
451 std::size_t count = w.size() - 1;
452 bool correct = command == "ROUTE" ? count == 3 || count == 5
453 : command == "TOKEN" || command == "LIMIT" ? count == 2
454 : command == "REDIRECT" ? count == 3 : command == "ORIGIN" ? count == 1 : count >= 2;
455 if (!correct) throw std::runtime_error("wrong number of arguments");
456 if (command == "ROUTE") {
457 if (!valid_name(w[1])) throw std::runtime_error("bad route name " + w[1]);
458 if (route_names.count(w[1])) throw std::runtime_error("duplicate route " + w[1]);
459 std::set<std::string> methods;
460 for (const std::string& method : split(w[2], ',')) {
461 if ((method != "GET" && method != "POST" && method != "PUT" && method != "PATCH" && method != "DELETE") ||
462 !methods.insert(method).second) throw std::runtime_error("bad methods " + w[2]);
463 }
464 std::optional<std::vector<Piece>> pieces = parse_pattern(w[3]);
465 if (!pieces) throw std::runtime_error("bad pattern " + w[3]);
466 if (count == 5 && w[4] != "ROLE") throw std::runtime_error("unknown option " + w[4]);
467 if (count == 5 && !matches(w[5], "[a-z]{1,16}")) throw std::runtime_error("bad role " + w[5]);
468 route_names.emplace(w[1], routes.size());
469 routes.push_back({w[1], w[3], count == 5 ? w[5] : "none", methods, *pieces, {}});
470 } else if (command == "TOKEN") {
471 if (!matches(w[1], "[A-Za-z0-9._-]{1,32}")) throw std::runtime_error("bad token " + w[1]);
472 if (tokens.count(w[1])) throw std::runtime_error("duplicate token " + w[1]);
473 if (!matches(w[2], "[a-z]{1,16}")) throw std::runtime_error("bad role " + w[2]);
474 tokens[w[1]] = w[2];
475 } else if (command == "REDIRECT") {
476 std::optional<RulePath> from = rule_path(w[1]);
477 if (!from) throw std::runtime_error("bad path " + w[1]);
478 std::optional<RulePath> to = rule_path(w[2]);
479 if (!to) throw std::runtime_error("bad path " + w[2]);
480 if (from->prefix != to->prefix) throw std::runtime_error("wildcard mismatch");
481 if (w[3] != "301" && w[3] != "302" && w[3] != "307" && w[3] != "308")
482 throw std::runtime_error("bad redirect code " + w[3]);
483 for (const Redirect& redirect : redirects)
484 if (redirect.from == w[1]) throw std::runtime_error("duplicate redirect " + w[1]);
485 redirects.push_back({w[1], from->base, to->base, from->prefix, std::stoi(w[3])});
486 } else if (command == "ORIGIN") {
487 if (!matches(w[1], "https?://[a-z0-9.:-]+")) throw std::runtime_error("bad origin " + w[1]);
488 if (origins.count(w[1])) throw std::runtime_error("duplicate origin " + w[1]);
489 origins.insert(w[1]);
490 } else if (command == "LIMIT") {
491 if (!decimal_between(w[1], 1, 100) || !decimal_between(w[2], 1, 3600)) throw std::runtime_error("bad limit");
492 if (limit) throw std::runtime_error("duplicate limit");
493 limit = std::make_pair(std::stoi(w[1]), std::stoi(w[2]));
494 } else {
495 auto found = route_names.find(w[1]);
496 if (found == route_names.end()) throw std::runtime_error("unknown route " + w[1]);
497 Route& route = routes[found->second];
498 if (!route.types.empty()) throw std::runtime_error("duplicate types " + w[1]);
499 std::set<std::string> seen;
500 for (std::size_t i = 2; i < w.size(); ++i) {
501 if (!media_type(w[i])) throw std::runtime_error("bad media type " + w[i]);
502 if (!seen.insert(w[i]).second) throw std::runtime_error("duplicate media type " + w[i]);
503 }
504 route.types.assign(w.begin() + 2, w.end());
505 }
506 }

508 Response route_request(const std::string& method, const std::string& path,
509 const std::optional<std::string>& query, const Headers& headers) {
510 std::vector<std::string> segments = path == "/" ? std::vector<std::string>{} : split(path.substr(1), '/');
511 std::vector<std::size_t> matching;
512 std::set<std::string> allowed{"OPTIONS"};
513 for (std::size_t index : ranked) {
514 if (!routes[index].matches_path(segments)) continue;
515 matching.push_back(index);
516 allowed.insert(routes[index].methods.begin(), routes[index].methods.end());
517 }
518 if (matching.empty()) return {404, "not found: " + path};
519 if (allowed.count("GET")) allowed.insert("HEAD");
520 std::string allow = join(allowed, ", ");
521 if (method == "OPTIONS") return {204, "", {{"Allow", allow}}};
522 std::string lookup = method == "HEAD" ? "GET" : method;
523 Route* selected = nullptr;
524 for (std::size_t index : matching) {
525 if (routes[index].methods.count(lookup)) {
526 selected = &routes[index];
527 break;
528 }
529 }
530 if (!selected) return {405, "method not allowed: " + method, {{"Allow", allow}}};
531 ++selected->selections;
532 if (selected->role != "none") {
533 auto auth = headers.find("authorization");
534 if (auth == headers.end())
535 return {401, "unauthorized: missing token", {{"WWW-Authenticate", "Bearer realm=\"api\""}}};
536 std::size_t space = auth->second.find(' ');
537 std::string scheme = lower(auth->second.substr(0, space));
538 std::string token = space == std::string::npos ? "" : auth->second.substr(space + 1);
539 auto found = tokens.find(token);
540 if (scheme != "bearer" || found == tokens.end())
541 return {401, "unauthorized: invalid token", {{"WWW-Authenticate", "Bearer realm=\"api\", error=\"invalid_token\""}}};
542 if (found->second != selected->role && found->second != "admin")
543 return {403, "forbidden: requires role " + selected->role};
544 }
545 Headers response_headers;
546 if (selected->types.size() > 1) response_headers["Vary"] = "Accept";
547 std::optional<std::string> type = negotiate(selected->types, headers);
548 if (!type) return {406, "not acceptable: " + join(selected->types, ", "), response_headers};
549 std::string body = body_for(*type, selected->name, selected->parameters(segments), query_pairs(query), cookie_pairs(headers));
550 std::string etag = etag_for(body);
551 auto conditional = headers.find("if-none-match");
552 if (conditional != headers.end()) {
553 for (std::string candidate : split(conditional->second, ',')) {
554 candidate = trim(candidate);
555 if (candidate.compare(0, 2, "W/") == 0) candidate = candidate.substr(2);
556 if (candidate != "*" && candidate != etag) continue;
557 if (lookup == "GET") {
558 response_headers["ETag"] = etag;
559 return {304, "", response_headers};
560 }
561 return {412, "precondition failed: " + etag, response_headers};
562 }
563 }
564 response_headers["ETag"] = etag;
565 return {200, body, response_headers, *type};
566 }
567
568 Response dispatch(const std::string& method, const std::string& target, const Headers& headers) {
569 static const std::set<std::string> known{"GET", "HEAD", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"};
570 if (!known.count(method)) return {501, "not implemented: " + method};
571 auto host = headers.find("host");
572 if (host == headers.end() || host->second.empty()) return {400, "bad request: missing host"};
573 std::string effective = method;
574 auto override = headers.find("x-http-method-override");
575 if (override != headers.end()) {
576 if (method != "POST" || (override->second != "PUT" && override->second != "PATCH" && override->second != "DELETE"))
577 return {400, "bad request: method override"};
578 effective = override->second;
579 }
580 auto client_header = headers.find("x-client");
581 if (client_header != headers.end() && !client_header->second.empty()) {
582 const std::string& name = client_header->second;
583 Client& client = clients[name];
584 if (limit) {
585 int n = limit->first, seconds = limit->second, time = std::stoi(headers.at("x-time"));
586 if (!client.initialized) {
587 client.level = n * seconds;
588 client.last_time = time;
589 client.initialized = true;
590 }
591 client.level = std::min(n * seconds, client.level + (time - client.last_time) * n);
592 client.last_time = time;
593 if (client.level < seconds) {
594 ++client.limited;
595 int wait = (seconds - client.level + n - 1) / n;
596 return {429, "too many requests: client " + name, {{"Retry-After", std::to_string(wait)}}};
597 }
598 client.level -= seconds;
599 }
600 ++client.allowed;
601 }
602 std::size_t question = target.find('?');
603 std::string path = target.substr(0, question);
604 std::optional<std::string> query;
605 if (question != std::string::npos) query = target.substr(question + 1);
606 std::string suffix = query ? "?" + *query : "";
607 auto redirect_response = [](int code, const std::string& destination) {
608 return Response(code, "redirect to " + destination, {{"Location", destination}});
609 };
610 std::string normalized = normalize(path);
611 if (normalized != path) return redirect_response(308, normalized + suffix);
612 for (const Redirect& redirect : redirects) {
613 bool applies = path == redirect.source ||
614 (redirect.prefix && path.compare(0, redirect.source.size() + 1, redirect.source + "/") == 0);
615 if (applies) return redirect_response(redirect.code, redirect.target + path.substr(redirect.source.size()) + suffix);
616 }
617 return route_request(effective, path, query, headers);
618 }
619
620 Response respond(const std::string& method, const std::string& target, const std::vector<std::string>& lines) {
621 if (!valid_escapes(target)) return {400, "bad request: percent escape"};
622 Headers headers;
623 for (std::size_t i = 0; i < lines.size(); ++i) {
624 std::size_t colon = lines[i].find(':');
625 if (colon == std::string::npos || !matches(lines[i].substr(0, colon), "[A-Za-z0-9-]+"))
626 return {400, "bad request: header line " + std::to_string(i + 1)};
627 std::string name = lower(lines[i].substr(0, colon));
628 if (headers.count(name)) return {400, "bad request: duplicate header " + name};
629 headers.emplace(name, trim(lines[i].substr(colon + 1)));
630 }
631 Response response = dispatch(method, target, headers);
632 auto origin = headers.find("origin");
633 if (origin != headers.end() && origins.count(origin->second)) {
634 response.headers["Access-Control-Allow-Origin"] = origin->second;
635 if (response.status == 204 && headers.count("access-control-request-method"))
636 response.headers["Access-Control-Allow-Methods"] = response.headers.at("Allow");
637 }
638 return response;
639 }

641 std::string build_url(const std::string& name, const std::vector<std::string>& arguments) const {
642 auto found = route_names.find(name);
643 if (found == route_names.end()) throw std::runtime_error("unknown route " + name);
644 const Route& route = routes[found->second];
645 std::set<std::string> parameter_names;
646 for (const Piece& piece : route.pieces) if (piece.kind != PieceKind::Literal) parameter_names.insert(piece.text);
647 std::map<std::string, std::string> values;
648 Pairs query;
649 for (const std::string& argument : arguments) {
650 std::size_t equals = argument.find('=');
651 if (equals == std::string::npos || equals == 0) throw std::runtime_error("bad argument " + argument);
652 std::string key = argument.substr(0, equals), value = argument.substr(equals + 1);
653 if (!parameter_names.count(key)) query.emplace_back(key, value);
654 else if (!values.emplace(key, value).second) throw std::runtime_error("duplicate parameter " + key);
655 }
656 std::vector<std::string> pieces;
657 for (const Piece& piece : route.pieces) {
658 if (piece.kind == PieceKind::Literal) {
659 pieces.push_back(piece.text);
660 continue;
661 }
662 auto argument = values.find(piece.text);
663 if (argument == values.end() && piece.kind != PieceKind::Wildcard)
664 throw std::runtime_error("missing parameter " + piece.text);
665 std::string value = argument == values.end() ? "" : argument->second;
666 if (!piece.accepts(value)) throw std::runtime_error("bad value for " + piece.text);
667 if (piece.kind == PieceKind::Wildcard) {
668 if (!value.empty()) for (const std::string& part : split(value, '/')) pieces.push_back(encode(part));
669 } else pieces.push_back(encode(value));
670 }
671 std::string url = "/" + join(pieces, "/");
672 if (!query.empty()) url += "?" + encoded_pairs(query);
673 return url;
674 }

676 static void write_response(Response response, bool head) {
677 static const std::map<int, std::string> reasons{
678 {200,"OK"}, {204,"No Content"}, {301,"Moved Permanently"}, {302,"Found"}, {304,"Not Modified"},
679 {307,"Temporary Redirect"}, {308,"Permanent Redirect"}, {400,"Bad Request"}, {401,"Unauthorized"},
680 {403,"Forbidden"}, {404,"Not Found"}, {405,"Method Not Allowed"}, {406,"Not Acceptable"},
681 {412,"Precondition Failed"}, {429,"Too Many Requests"}, {501,"Not Implemented"}
682 };
683 static const std::vector<std::string> header_order{
684 "Location", "Allow", "Retry-After", "WWW-Authenticate", "Access-Control-Allow-Origin",
685 "Access-Control-Allow-Methods", "Vary", "ETag", "Content-Type", "Content-Length"
686 };
687 bool has_body = response.status != 204 && response.status != 304;
688 if (has_body) {
689 response.headers["Content-Type"] = response.status == 200 ? response.type : "text/plain";
690 response.headers["Content-Length"] = std::to_string(response.body.size());
691 }
692 std::cout << "HTTP/1.1 " << response.status << ' ' << reasons.at(response.status) << '\n';
693 for (const std::string& name : header_order) {
694 auto header = response.headers.find(name);
695 if (header != response.headers.end()) std::cout << name << ": " << header->second << '\n';
696 }
697 if (has_body && !head) std::cout << response.body << '\n';
698 }

709 void finish_configuration() {
710 std::cout << "CONFIG routes=" << routes.size() << " tokens=" << tokens.size() << " redirects=" << redirects.size()
711 << " origins=" << origins.size() << " limit=";
712 if (limit) std::cout << limit->first << '/' << limit->second;
713 else std::cout << "none";
714 std::cout << " errors=" << config_errors << '\n';
715 for (std::size_t i = 0; i < routes.size(); ++i) {
716 if (routes[i].types.empty()) routes[i].types.push_back("application/json");
717 ranked.push_back(i);
718 }
719 std::stable_sort(ranked.begin(), ranked.end(), [&](std::size_t a, std::size_t b) {
720 return std::make_tuple(-routes[a].literal_count(), -routes[a].typed_count(), routes[a].wildcard()) <
721 std::make_tuple(-routes[b].literal_count(), -routes[b].typed_count(), routes[b].wildcard());
722 });
723 for (std::size_t rank = 0; rank < ranked.size(); ++rank) {
724 const Route& route = routes[ranked[rank]];
725 std::set<std::string> methods = route.methods;
726 if (methods.count("GET")) methods.insert("HEAD");
727 std::cout << "TABLE " << rank + 1 << ' ' << route.name << ' ' << join(methods, ",") << ' ' << route.pattern
728 << " role=" << (route.role == "none" ? "-" : route.role) << " types=" << join(route.types, ",") << '\n';
729 }

732 void request(const std::string& line, const std::vector<std::string>& headers) {
733 ++requests;
734 std::vector<std::string> parts = split(line, ' ');
735 bool valid = parts.size() == 3 && matches(parts[0], "[A-Z]{1,16}") && !parts[1].empty() && parts[1][0] == '/' &&
736 parts[1].find('#') == std::string::npos && parts[2] == "HTTP/1.1";
737 Response response = valid ? respond(parts[0], parts[1], headers) : Response(400, "bad request: request line");
738 ++statuses[response.status];
739 std::cout << "REQUEST " << requests << '\n';
740 write_response(std::move(response), valid && parts[0] == "HEAD");
741 }

743 void urlfor(const std::string& line) {
744 std::vector<std::string> parts = split(line, ' ');
745 try {
746 std::string url = build_url(parts[1], std::vector<std::string>(parts.begin() + 2, parts.end()));
747 ++url_ok;
748 std::cout << "URLFOR " << parts[1] << ' ' << url << '\n';
749 } catch (const std::runtime_error& error) {
750 ++url_error;
751 std::cout << "URLFOR ERROR " << error.what() << '\n';
752 }
753 }

758 for (const Route& route : routes) std::cout << "ROUTE " << route.name << ' ' << route.selections << '\n';

761 }
