            wait := 0
                level := clients[id].level + (timestamp - clients[id].timestamp) * capacity
                clients[id].level = if level > capacity * period then capacity * period else level
                clients[id].timestamp = timestamp
                if clients[id].level < period
                    wait = (period - clients[id].level + capacity - 1) / capacity
                else
                    clients[id].level -= period
            if wait > 0
                clients[id].limited += 1
                response := Response(429, "too many requests: client {client}")
                .response.headers.set("Retry-After", "{wait}")
                return response
