26func trim(s string) string {
27 return strings.Trim(s, " ")
28}

30func unreserved(c byte) bool {
31 return c >= 'a' && c <= 'z' || c >= 'A' && c <= 'Z' || c >= '0' && c <= '9' || strings.ContainsRune("-._~", rune(c))
32}
33
34func segment(s string) bool {
35 if s == "" || s == "." || s == ".." {
36 return false
37 }
38 for i := range s {
39 if !unreserved(s[i]) {
40 return false
41 }
42 }
43 return true
44}
45
46func canonical(s string) bool {
47 if s == "/" {
48 return true
49 }
50 if !strings.HasPrefix(s, "/") {
51 return false
52 }
53 for _, p := range strings.Split(s[1:], "/") {
54 if !segment(p) {
55 return false
56 }
57 }
58 return true
59}
60
61func media(s string) bool {
62 parts := strings.Split(s, "/")
63 return len(parts) == 2 && mediaPartPattern.MatchString(parts[0]) && mediaPartPattern.MatchString(parts[1])
64}
65
66func rulePath(s string) (string, bool, bool) {
67 if strings.HasSuffix(s, "/*") {
68 base := s[:len(s)-2]
69 return base, true, base != "/" && canonical(base)
70 }
71 return s, false, canonical(s)
72}

74func encode(s string) string {
75 var b strings.Builder
76 const hex = "0123456789ABCDEF"
77 for i := range s {
78 c := s[i]
79 if unreserved(c) {
80 b.WriteByte(c)
81 } else {
82 b.WriteByte('%')
83 b.WriteByte(hex[c>>4])
84 b.WriteByte(hex[c&15])
85 }
86 }
87 return b.String()
88}

90func hex(c byte) int {
91 switch {
92 case c >= '0' && c <= '9':
93 return int(c - '0')
94 case c >= 'a' && c <= 'f':
95 return int(c-'a') + 10
96 case c >= 'A' && c <= 'F':
97 return int(c-'A') + 10
98 }
99 return -1
100}

102func decode(s string) string {
103 var b strings.Builder
104 for i := 0; i < len(s); i++ {
105 if s[i] == '%' && i+2 < len(s) {
106 b.WriteByte(byte(hex(s[i+1])*16 + hex(s[i+2])))
107 i += 2
108 } else {
109 b.WriteByte(s[i])
110 }
111 }
112 return b.String()
113}
114
115func escapeCheck(s string) bool {
116 for i := 0; i < len(s); i++ {
117 if s[i] == '%' {
118 if i+2 >= len(s) || hex(s[i+1]) < 0 || hex(s[i+2]) < 0 {
119 return false
120 }
121 n := hex(s[i+1])*16 + hex(s[i+2])
122 if n < 32 || n > 126 {
123 return false
124 }
125 i += 2
126 }
127 }
128 return true
129}

131func normalize(path string) string {
132 var b strings.Builder
133 for i := 0; i < len(path); i++ {
134 if path[i] == '%' {
135 c := byte(hex(path[i+1])*16 + hex(path[i+2]))
136 if unreserved(c) {
137 b.WriteByte(c)
138 } else {
139 b.WriteString(strings.ToUpper(path[i : i+3]))
140 }
141 i += 2
142 } else {
143 b.WriteByte(path[i])
144 }
145 }
146 parts := []string{}
147 for _, p := range strings.Split(b.String()[1:], "/") {
148 if p == "" || p == "." {
149 continue
150 }
151 if p == ".." {
152 if len(parts) > 0 {
153 parts = parts[:len(parts)-1]
154 }
155 } else {
156 parts = append(parts, p)
157 }
158 }
159 return "/" + strings.Join(parts, "/")
160}

162type piece struct {
163 kind, name string
164}
165
166func parsePattern(pattern string) ([]piece, bool) {
167 if pattern == "/" {
168 return []piece{}, true
169 }
170 if !strings.HasPrefix(pattern, "/") {
171 return nil, false
172 }
173 parts := strings.Split(pattern[1:], "/")
174 pieces := []piece{}
175 names := map[string]bool{}
176 for i, p := range parts {
177 x := piece{kind: "literal", name: p}
178 if segment(p) {
179 pieces = append(pieces, x)
180 continue
181 }
182 if len(p) < 3 || p[0] != '{' || p[len(p)-1] != '}' {
183 return nil, false
184 }
185 inside := p[1 : len(p)-1]
186 x.kind = "param"
187 x.name = inside
188 if strings.HasPrefix(inside, "*") {
189 x.kind = "wild"
190 x.name = inside[1:]
191 if i != len(parts)-1 {
192 return nil, false
193 }
194 } else if name, t, ok := strings.Cut(inside, ":"); ok {
195 x.name = name
196 x.kind = t
197 if t != "int" && t != "slug" {
198 return nil, false
199 }
200 }
201 if !namePattern.MatchString(x.name) || names[x.name] {
202 return nil, false
203 }
204 names[x.name] = true
205 pieces = append(pieces, x)
206 }
207 return pieces, true
208}

219func higher(a, b *route) bool {
220 if a.literals != b.literals {
221 return a.literals > b.literals
222 }
223 if a.typed != b.typed {
224 return a.typed > b.typed
225 }
226 if a.wild != b.wild {
227 return !a.wild
228 }
229 return a.number < b.number
230}

257func (r *router) config(words []string) string {
258 cmd, args := words[0], words[1:]
259 n := len(args)
260 validCount := false
261 switch cmd {
262 case "ROUTE":
263 validCount = n == 3 || n == 5
264 case "TOKEN", "LIMIT":
265 validCount = n == 2
266 case "REDIRECT":
267 validCount = n == 3
268 case "ORIGIN":
269 validCount = n == 1
270 case "TYPES":
271 validCount = n >= 2
272 default:
273 return "unknown command " + cmd
274 }
275 if !validCount {
276 return "wrong number of arguments"
277 }
278 switch cmd {
279 case "ROUTE":
280 name := args[0]
281 if !namePattern.MatchString(name) {
282 return "bad route name " + name
283 }
284 if r.byName[name] != nil {
285 return "duplicate route " + name
286 }
287 methods := map[string]bool{}
288 for _, m := range strings.Split(args[1], ",") {
289 if !strings.Contains(" GET POST PUT PATCH DELETE ", " "+m+" ") || methods[m] {
290 return "bad methods " + args[1]
291 }
292 methods[m] = true
293 }
294 pieces, ok := parsePattern(args[2])
295 if !ok {
296 return "bad pattern " + args[2]
297 }
298 role := ""
299 if n == 5 {
300 if args[3] != "ROLE" {
301 return "unknown option " + args[3]
302 }
303 if !rolePattern.MatchString(args[4]) {
304 return "bad role " + args[4]
305 }
306 role = args[4]
307 if role == "none" {
308 role = ""
309 }
310 }
311 v := &route{name: name, pattern: args[2], role: role, methods: methods, pieces: pieces, number: len(r.routes)}
312 for _, p := range pieces {
313 switch p.kind {
314 case "literal":
315 v.literals++
316 case "int", "slug":
317 v.typed++
318 case "wild":
319 v.wild = true
320 }
321 }
322 r.routes = append(r.routes, v)
323 r.byName[name] = v
324 case "TOKEN":
325 if !tokenPattern.MatchString(args[0]) {
326 return "bad token " + args[0]
327 }
328 if _, ok := r.tokens[args[0]]; ok {
329 return "duplicate token " + args[0]
330 }
331 if !rolePattern.MatchString(args[1]) {
332 return "bad role " + args[1]
333 }
334 r.tokens[args[0]] = args[1]
335 case "REDIRECT":
336 from, prefix, ok := rulePath(args[0])
337 if !ok {
338 return "bad path " + args[0]
339 }
340 to, toPrefix, ok := rulePath(args[1])
341 if !ok {
342 return "bad path " + args[1]
343 }
344 if prefix != toPrefix {
345 return "wildcard mismatch"
346 }
347 if args[2] != "301" && args[2] != "302" && args[2] != "307" && args[2] != "308" {
348 return "bad redirect code " + args[2]
349 }
350 for _, d := range r.redirects {
351 if d.from == from && d.prefix == prefix {
352 return "duplicate redirect " + args[0]
353 }
354 }
355 code, _ := strconv.Atoi(args[2])
356 r.redirects = append(r.redirects, redirect{from, to, prefix, code})
357 case "ORIGIN":
358 if !originPattern.MatchString(args[0]) {
359 return "bad origin " + args[0]
360 }
361 if r.origins[args[0]] {
362 return "duplicate origin " + args[0]
363 }
364 r.origins[args[0]] = true
365 case "LIMIT":
366 n, errN := strconv.Atoi(args[0])
367 s, errS := strconv.Atoi(args[1])
368 if !decimalPattern.MatchString(args[0]) || !decimalPattern.MatchString(args[1]) || errN != nil || errS != nil || n < 1 || n > 100 || s < 1 || s > 3600 {
369 return "bad limit"
370 }
371 if r.capacity > 0 {
372 return "duplicate limit"
373 }
374 r.capacity, r.period = n, s
375 case "TYPES":
376 v := r.byName[args[0]]
377 if v == nil {
378 return "unknown route " + args[0]
379 }
380 if v.types != nil {
381 return "duplicate types " + args[0]
382 }
383 seen := map[string]bool{}
384 for _, t := range args[1:] {
385 if !media(t) {
386 return "bad media type " + t
387 }
388 if seen[t] {
389 return "duplicate media type " + t
390 }
391 seen[t] = true
392 }
393 v.types = append([]string{}, args[1:]...)
394 }
395 return ""
396}

398func methodList(methods map[string]bool, options bool) []string {
399 set := map[string]bool{}
400 for m := range methods {
401 set[m] = true
402 }
403 if set["GET"] {
404 set["HEAD"] = true
405 }
406 if options {
407 set["OPTIONS"] = true
408 }
409 out := []string{}
410 for m := range set {
411 out = append(out, m)
412 }
413 sort.Strings(out)
414 return out

417func (r *router) table(out *bufio.Writer) {
418 limit := "none"
419 if r.capacity > 0 {
420 limit = fmt.Sprintf("%d/%d", r.capacity, r.period)
421 }
422 fmt.Fprintf(out, "CONFIG routes=%d tokens=%d redirects=%d origins=%d limit=%s errors=%d\n", len(r.routes), len(r.tokens), len(r.redirects), len(r.origins), limit, r.errors)
423 r.priority = append([]*route{}, r.routes...)

425 return higher(r.priority[i], r.priority[j])

432 if role == "" {
433 role = "-"

436 }
437}

439type pair struct {
440 name, value string
441}
442
443func match(v *route, segments []string) ([]pair, bool) {
444 m := len(v.pieces)
445 if v.wild {
446 m--
447 }
448 if !v.wild && len(segments) != m || v.wild && len(segments) < m {
449 return nil, false
450 }
451 params := []pair{}
452 for j, p := range v.pieces {
453 if p.kind == "wild" {
454 values := []string{}
455 for _, s := range segments[j:] {
456 values = append(values, decode(s))
457 }
458 params = append(params, pair{p.name, strings.Join(values, "/")})
459 break
460 }
461 x := segments[j]
462 switch p.kind {
463 case "literal":
464 if p.name != x {
465 return nil, false
466 }
467 case "int":
468 if !intPattern.MatchString(x) {
469 return nil, false
470 }
471 n, _ := strconv.Atoi(x)
472 params = append(params, pair{p.name, strconv.Itoa(n)})
473 case "slug":
474 if !slugPattern.MatchString(x) {
475 return nil, false
476 }
477 params = append(params, pair{p.name, x})
478 case "param":
479 params = append(params, pair{p.name, decode(x)})
480 }
481 }
482 sort.Slice(params, func(i, j int) bool {
483 return params[i].name < params[j].name
484 })
485 return params, true
486}

488func negotiate(types []string, accept string, present bool) string {
489 if !present {
490 return types[0]
491 }
492 type element struct {
493 rangeText string
494 specificity, weight int
495 }
496 elements := []element{}
497 for _, text := range strings.Split(accept, ",") {
498 parts := strings.Split(text, ";")
499 rangeText := strings.ToLower(trim(parts[0]))
500 spec := 3
501 if rangeText == "*/*" {
502 spec = 1
503 } else if strings.HasSuffix(rangeText, "/*") && mediaPartPattern.MatchString(strings.TrimSuffix(rangeText, "/*")) {
504 spec = 2
505 } else if !media(rangeText) {
506 continue
507 }
508 weight := 1000
509 valid := true
510 for _, p := range parts[1:] {
511 name, v, ok := strings.Cut(trim(p), "=")
512 if ok && strings.ToLower(trim(name)) == "q" {
513 v = trim(v)
514 if !qPattern.MatchString(v) {
515 valid = false
516 } else {
517 whole, fraction, _ := strings.Cut(v, ".")
518 if whole == "1" && strings.Trim(fraction, "0") != "" {
519 valid = false
520 } else {
521 fraction += ("000")
522 weight, _ = strconv.Atoi(whole + fraction[:3])
523 }
524 }
525 break
526 }
527 }
528 if valid {
529 elements = append(elements, element{rangeText, spec, weight})
530 }
531 }
532 selected := ""
533 best := 0
534 for _, t := range types {
535 specificity, weight := 0, 0
536 main, _, _ := strings.Cut(t, "/")
537 for _, e := range elements {
538 fits := e.rangeText == "*/*" || e.rangeText == main+"/*" || e.rangeText == t
539 if fits && e.specificity > specificity {
540 specificity = e.specificity
541 weight = e.weight
542 }
543 }
544 if weight > best {
545 best = weight
546 selected = t
547 }
548 }
549 return selected
550}

552func quoted(s string) string {
553 return "\"" + strings.NewReplacer("\\", "\\\\", "\"", "\\\"").Replace(s) + "\""
554}
555
556var htmlEscaper = strings.NewReplacer("&", "&amp;", "<", "&lt;", ">", "&gt;", "\"", "&quot;")
557
558func bodyFor(v *route, params []pair, query string, headers map[string]string, selected string) string {
559 queries, cookies := []pair{}, []pair{}
560 for _, p := range strings.Split(query, "&") {
561 if p == "" {
562 continue
563 }
564 name, value, _ := strings.Cut(p, "=")
565 queries = append(queries, pair{decode(strings.ReplaceAll(name, "+", " ")), decode(strings.ReplaceAll(value, "+", " "))})
566 }
567 seen := map[string]bool{}
568 if cookie, ok := headers["cookie"]; ok {
569 for _, p := range strings.Split(cookie, ";") {
570 name, value, ok := strings.Cut(trim(p), "=")
571 if ok && name != "" && !seen[name] {
572 seen[name] = true
573 cookies = append(cookies, pair{name, value})
574 }
575 }
576 }
577 sort.Slice(cookies, func(i, j int) bool {
578 return cookies[i].name < cookies[j].name
579 })
580 if selected == "application/json" {
581 object := func(ps []pair) string {
582 parts := []string{}
583 for _, p := range ps {
584 parts = append(parts, quoted(p.name)+":"+quoted(p.value))
585 }
586 return "{" + strings.Join(parts, ",") + "}"
587 }
588 qp := []string{}
589 for _, p := range queries {
590 qp = append(qp, "["+quoted(p.name)+","+quoted(p.value)+"]")
591 }
592 return "{\"route\":" + quoted(v.name) + ",\"params\":" + object(params) + ",\"query\":[" + strings.Join(qp, ",") + "],\"cookies\":" + object(cookies) + "}"
593 }
594 if selected == "text/html" {
595 s := "<dl><dt>route</dt><dd>" + htmlEscaper.Replace(v.name) + "</dd>"
596 for i, ps := range [][]pair{params, queries, cookies} {
597 label := []string{"param", "query", "cookie"}[i]
598 for _, p := range ps {
599 s += "<dt>" + label + " " + htmlEscaper.Replace(p.name) + "</dt><dd>" + htmlEscaper.Replace(p.value) + "</dd>"
600 }
601 }
602 return s + "</dl>"
603 }
604 encoded := func(ps []pair) string {
605 if len(ps) == 0 {
606 return "-"
607 }
608 parts := []string{}
609 for _, p := range ps {
610 parts = append(parts, encode(p.name)+"="+encode(p.value))
611 }
612 return strings.Join(parts, "&")
613 }
614 return "route=" + v.name + " params=" + encoded(params) + " query=" + encoded(queries) + " cookies=" + encoded(cookies)
615}

617func etag(body string) string {
618 a, b := 1, 0
619 for i := range body {
620 a = (a + int(body[i])) % 65521
621 b = (b + a) % 65521
622 }
623 return fmt.Sprintf("\"%04x%04x\"", b, a)
624}

637func (r *router) dispatch(method, target string, h map[string]string) response {
638 if !strings.Contains(" GET HEAD POST PUT PATCH DELETE OPTIONS ", " "+method+" ") {
639 return reply(501, "not implemented: "+method)
640 }
641 if h["host"] == "" {
642 return reply(400, "bad request: missing host")
643 }
644 effective := method
645 if override, ok := h["x-http-method-override"]; ok {
646 if method == "POST" && (override == "PUT" || override == "PATCH" || override == "DELETE") {
647 effective = override
648 } else {
649 return reply(400, "bad request: method override")
650 }
651 }
652 if c := h["x-client"]; c != "" {
653 t, _ := strconv.Atoi(h["x-time"])
654 record := r.clients[c]
655 if record == nil {
656 record = &client{level: r.capacity * r.period, last: t}
657 r.clients[c] = record
658 }
659 if r.capacity > 0 {
660 record.level += (t - record.last) * r.capacity
661 if record.level > r.capacity*r.period {
662 record.level = r.capacity * r.period
663 }
664 record.last = t
665 if record.level < r.period {
666 record.limited++
667 w := (r.period - record.level + r.capacity - 1) / r.capacity
668 resp := reply(429, "too many requests: client "+c)
669 resp.headers["Retry-After"] = strconv.Itoa(w)
670 return resp
671 }
672 record.level -= r.period
673 }
674 record.allowed++
675 }
676 path, query, hasQuery := strings.Cut(target, "?")
677 suffix := ""
678 if hasQuery {
679 suffix = "?" + query
680 }
681 normalized := normalize(path)
682 if normalized != path {
683 location := normalized + suffix
684 resp := reply(308, "redirect to "+location)
685 resp.headers["Location"] = location
686 return resp
687 }
688 for _, d := range r.redirects {
689 if path == d.from || d.prefix && strings.HasPrefix(path, d.from+"/") {
690 location := d.to + path[len(d.from):] + suffix
691 resp := reply(d.code, "redirect to "+location)
692 resp.headers["Location"] = location
693 return resp
694 }
695 }
696 segments := []string{}
697 if path != "/" {
698 segments = strings.Split(path[1:], "/")
699 }
700 matching := []*route{}
701 values := map[*route][]pair{}
702 allowed := map[string]bool{}
703 for _, v := range r.priority {
704 ps, ok := match(v, segments)
705 if ok {
706 matching = append(matching, v)
707 values[v] = ps
708 for m := range v.methods {
709 allowed[m] = true
710 }
711 }
712 }
713 if len(matching) == 0 {
714 return reply(404, "not found: "+path)
715 }
716 allow := strings.Join(methodList(allowed, true), ", ")
717 if effective == "OPTIONS" {
718 resp := reply(204, "")
719 resp.headers["Allow"] = allow
720 return resp
721 }
722 lookup := effective
723 if lookup == "HEAD" {
724 lookup = "GET"
725 }
726 var selected *route
727 for _, v := range matching {
728 if v.methods[lookup] {
729 selected = v
730 break
731 }
732 }
733 if selected == nil {
734 resp := reply(405, "method not allowed: "+effective)
735 resp.headers["Allow"] = allow
736 return resp
737 }
738 selected.selections++
739 if selected.role != "" {
740 auth, ok := h["authorization"]
741 if !ok {
742 resp := reply(401, "unauthorized: missing token")
743 resp.headers["WWW-Authenticate"] = "Bearer realm=\"api\""
744 return resp
745 }
746 scheme, token, _ := strings.Cut(auth, " ")
747 role, exists := r.tokens[token]
748 if strings.ToLower(scheme) != "bearer" || !exists {
749 resp := reply(401, "unauthorized: invalid token")
750 resp.headers["WWW-Authenticate"] = "Bearer realm=\"api\", error=\"invalid_token\""
751 return resp
752 }
753 if role != selected.role && role != "admin" {
754 return reply(403, "forbidden: requires role "+selected.role)
755 }
756 }
757 accept, present := h["accept"]
758 selectedType := negotiate(selected.types, accept, present)
759 var resp response
760 if selectedType == "" {
761 resp = reply(406, "not acceptable: "+strings.Join(selected.types, ", "))
762 } else {
763 body := bodyFor(selected, values[selected], query, h, selectedType)
764 tag := etag(body)
765 conditional := false
766 if condition, ok := h["if-none-match"]; ok {
767 for _, p := range strings.Split(condition, ",") {
768 p = trim(p)
769 p = strings.TrimPrefix(p, "W/")
770 if p == "*" || p == tag {
771 conditional = true
772 break
773 }
774 }
775 }
776 if conditional {
777 if lookup == "GET" {
778 resp = reply(304, "")
779 resp.headers["ETag"] = tag
780 } else {
781 resp = reply(412, "precondition failed: "+tag)
782 }
783 } else {
784 resp = reply(200, body)
785 resp.contentType = selectedType
786 resp.headers["ETag"] = tag
787 }
788 }
789 if len(selected.types) > 1 {
790 resp.headers["Vary"] = "Accept"
791 }
792 return resp
793}

795func (r *router) request(line string, headerLines []string) response {
796 parts := strings.Split(line, " ")
797 if len(parts) != 3 || !methodPattern.MatchString(parts[0]) || !strings.HasPrefix(parts[1], "/") || strings.Contains(parts[1], "#") || parts[2] != "HTTP/1.1" {
798 return reply(400, "bad request: request line")
799 }
800 head := parts[0] == "HEAD"
801 early := func(body string) response {
802 resp := reply(400, body)
803 resp.head = head
804 return resp
805 }
806 if !escapeCheck(parts[1]) {
807 return early("bad request: percent escape")
808 }
809 h := map[string]string{}
810 for i, l := range headerLines {
811 name, v, ok := strings.Cut(l, ":")
812 if !ok || !headerPattern.MatchString(name) {
813 return early(fmt.Sprintf("bad request: header line %d", i+1))
814 }
815 name = strings.ToLower(name)
816 if _, exists := h[name]; exists {
817 return early("bad request: duplicate header " + name)
818 }
819 h[name] = trim(v)
820 }
821 resp := r.dispatch(parts[0], parts[1], h)
822 resp.head = head
823 if origin, ok := h["origin"]; ok && r.origins[origin] {
824 resp.headers["Access-Control-Allow-Origin"] = origin
825 if resp.status == 204 {
826 if _, ok := h["access-control-request-method"]; ok {
827 resp.headers["Access-Control-Allow-Methods"] = resp.headers["Allow"]
828 }
829 }
830 }
831 return resp
832}

834var phrases = map[int]string{200: "OK", 204: "No Content", 301: "Moved Permanently", 302: "Found", 304: "Not Modified", 307: "Temporary Redirect", 308: "Permanent Redirect", 400: "Bad Request", 401: "Unauthorized", 403: "Forbidden", 404: "Not Found", 405: "Method Not Allowed", 406: "Not Acceptable", 412: "Precondition Failed", 429: "Too Many Requests", 501: "Not Implemented"}
835var headerOrder = []string{"Location", "Allow", "Retry-After", "WWW-Authenticate", "Access-Control-Allow-Origin", "Access-Control-Allow-Methods", "Vary", "ETag", "Content-Type", "Content-Length"}

837func (r *router) printResponse(out *bufio.Writer, resp response) {
838 r.requests++
839 r.statuses[resp.status]++
840 if resp.status != 204 && resp.status != 304 {
841 contentType := "text/plain"
842 if resp.status == 200 {
843 contentType = resp.contentType
844 }
845 resp.headers["Content-Type"] = contentType
846 resp.headers["Content-Length"] = strconv.Itoa(len(resp.body))
847 }
848 fmt.Fprintf(out, "REQUEST %d\nHTTP/1.1 %d %s\n", r.requests, resp.status, phrases[resp.status])
849 for _, name := range headerOrder {
850 if v, ok := resp.headers[name]; ok {
851 fmt.Fprintf(out, "%s: %s\n", name, v)
852 }
853 }
854 if resp.status != 204 && resp.status != 304 && !resp.head {
855 fmt.Fprintln(out, resp.body)
856 }
857}

859func (r *router) urlfor(words []string) (string, string) {
860 name := words[1]
861 v := r.byName[name]
862 if v == nil {
863 return "", "unknown route " + name
864 }
865 parameters := map[string]bool{}
866 for _, p := range v.pieces {
867 if p.kind != "literal" {
868 parameters[p.name] = true
869 }
870 }
871 values := map[string]string{}
872 query := []pair{}
873 for _, a := range words[2:] {
874 key, value, ok := strings.Cut(a, "=")
875 if !ok || key == "" {
876 return "", "bad argument " + a
877 }
878 if parameters[key] {
879 if _, exists := values[key]; exists {
880 return "", "duplicate parameter " + key
881 }
882 values[key] = value
883 } else {
884 query = append(query, pair{key, value})
885 }
886 }
887 path := []string{}
888 for _, p := range v.pieces {
889 if p.kind == "literal" {
890 path = append(path, p.name)
891 continue
892 }
893 value, exists := values[p.name]
894 if p.kind != "wild" && !exists {
895 return "", "missing parameter " + p.name
896 }
897 valid := true
898 switch p.kind {
899 case "param":
900 valid = value != "" && value != "." && value != ".."
901 case "int":
902 valid = intPattern.MatchString(value)
903 case "slug":
904 valid = slugPattern.MatchString(value)
905 case "wild":
906 if value != "" {
907 for _, s := range strings.Split(value, "/") {
908 if s == "" || s == "." || s == ".." {
909 valid = false
910 break
911 }
912 }
913 }
914 }
915 if !valid {
916 return "", "bad value for " + p.name
917 }
918 if p.kind == "wild" {
919 if value != "" {
920 for _, s := range strings.Split(value, "/") {
921 path = append(path, encode(s))
922 }
923 }
924 } else {
925 path = append(path, encode(value))
926 }
927 }
928 url := "/" + strings.Join(path, "/")
929 if len(query) > 0 {
930 qs := []string{}
931 for _, p := range query {
932 qs = append(qs, encode(p.name)+"="+encode(p.value))
933 }
934 url += "?" + strings.Join(qs, "&")
935 }
936 return "URLFOR " + name + " " + url, ""
937}

939func (r *router) summary(out *bufio.Writer) {
940 fmt.Fprintf(out, "SUMMARY requests=%d urlfor_ok=%d urlfor_error=%d\n", r.requests, r.urlOK, r.urlError)
941 codes := []int{}
942 for c := range r.statuses {
943 codes = append(codes, c)
944 }
945 sort.Ints(codes)
946 for _, c := range codes {
947 fmt.Fprintf(out, "STATUS %d %d\n", c, r.statuses[c])
948 }
949 for _, v := range r.routes {
950 fmt.Fprintf(out, "ROUTE %s %d\n", v.name, v.selections)
951 }
952 names := []string{}
953 for c := range r.clients {
954 names = append(names, c)
955 }
956 sort.Strings(names)
957 for _, c := range names {
958 v := r.clients[c]
959 fmt.Fprintf(out, "CLIENT %s allowed=%d limited=%d\n", c, v.allowed, v.limited)
960 }
961}

963func main() {
964 data, _ := io.ReadAll(os.Stdin)
965 lines := strings.Split(string(data), "\n")
966 r := newRouter()
967 out := bufio.NewWriter(os.Stdout)
968 defer out.Flush()
969 pos := 0
970 for lines[pos] != "REQUESTS" {
971 reason := r.config(strings.Split(lines[pos], " "))
972 if reason != "" {
973 r.errors++
974 fmt.Fprintf(out, "CONFIG ERROR %d: %s\n", pos+1, reason)
975 }
976 pos++
977 }
978 pos++
979 r.table(out)
980 for lines[pos] != "END" {
981 line := lines[pos]
982 pos++
983 if strings.HasPrefix(line, "URLFOR ") {
984 text, err := r.urlfor(strings.Split(line, " "))
985 if err != "" {
986 r.urlError++
987 fmt.Fprintln(out, "URLFOR ERROR "+err)
988 } else {
989 r.urlOK++
990 fmt.Fprintln(out, text)
991 }
992 } else {
993 headers := []string{}
994 for lines[pos] != "" {
995 headers = append(headers, lines[pos])
996 pos++
997 }
998 pos++
999 r.printResponse(out, r.request(line, headers))
1000 }
1001 }
1002 r.summary(out)
1003}
