1local function split(s, separator)
2 local out, at = {}, 1
3 while true do
4 local p = s:find(separator, at, true)
5 if not p then
6 out[#out + 1] = s:sub(at)
7 return out
8 end
9 out[#out + 1] = s:sub(at, p - 1)
10 at = p + #separator
11 end
12end

107local function escape_check(s)
108 local at = 1
109 while at <= #s do
110 if s:sub(at, at) == "%" then
111 local h = s:sub(at + 1, at + 2)
112 if #h ~= 2 or not h:match("^%x%x$") or tonumber(h, 16) < 32 or tonumber(h, 16) > 126 then
113 return false
114 end
115 at = at + 2
116 end
117 at = at + 1
118 end
119 return true
120end

122local function normalize(path)
123 path = path:gsub("%%(%x%x)", function (h)
124 local c = string.char(tonumber(h, 16))
125 return unreserved(c) and c or "%" .. h:upper()
126 end)
127 local out = {}
128 for _, p in ipairs(split(path:sub(2), "/")) do
129 if p == ".." then
130 out[#out] = nil
131 elseif p ~= "" and p ~= "." then
132 out[#out + 1] = p
133 end
134 end
135 return "/" .. table.concat(out, "/")
136end

138local function rule_path(s)
139 if s:sub(- 2) == "/*" then
140 local base = s:sub(1, - 3)
141 return base, true, base ~= "/" and canonical(base)
142 end
143 return s, false, canonical(s)

146local function parse_pattern(pattern)
147 if pattern == "/" then
148 return {}
149 end
150 if not starts(pattern, "/") then
151 return nil
152 end
153 local parts, out, names = split(pattern:sub(2), "/"), {}, {}
154 for i, p in ipairs(parts) do
155 local piece = { kind = "literal", name = p }
156 if not segment(p) then
157 if #p < 3 or p:sub(1, 1) ~= "{" or p:sub(- 1) ~= "}" then
158 return nil
159 end
160 local inside = p:sub(2, - 2)
161 piece.kind, piece.name = "param", inside
162 if starts(inside, "*") then
163 piece.kind, piece.name = "wild", inside:sub(2)
164 if i ~= #parts then
165 return nil
166 end
167 else
168 local name, kind, colon = cut(inside, ":")
169 if colon then
170 if kind ~= "int" and kind ~= "slug" then
171 return nil
172 end
173 piece.kind, piece.name = kind, name
174 end
175 end
176 if not valid_name(piece.name) or names[piece.name] then
177 return nil
178 end
179 names[piece.name] = true
180 end
181 out[#out + 1] = piece
182 end
183 return out
184end

190local function config(w)
191 local cmd, n = w[1], #w - 1
192 if cmd ~= "ROUTE" and cmd ~= "TOKEN" and cmd ~= "REDIRECT" and cmd ~= "ORIGIN" and cmd ~= "LIMIT" and cmd ~= "TYPES" then
193 bad("unknown command %s", cmd)
194 end
195 local count_ok = cmd == "ROUTE" and (n == 3 or n == 5) or (cmd == "TOKEN" or cmd == "LIMIT") and n == 2 or cmd == "REDIRECT" and n == 3 or cmd == "ORIGIN" and n == 1 or cmd == "TYPES" and n >= 2
196 if not count_ok then
197 bad("wrong number of arguments")
198 end
199 if cmd == "ROUTE" then
200 if not valid_name(w[2]) then
201 bad("bad route name %s", w[2])
202 end
203 if by_name[w[2]] then
204 bad("duplicate route %s", w[2])
205 end
206 local methods = {}
207 for _, m in ipairs(split(w[3], ",")) do
208 if not BASE_METHODS[m] or methods[m] then
209 bad("bad methods %s", w[3])
210 end
211 methods[m] = true
212 end
213 local pieces = parse_pattern(w[4])
214 if not pieces then
215 bad("bad pattern %s", w[4])
216 end
217 local role
218 if n == 5 then
219 if w[5] ~= "ROLE" then
220 bad("unknown option %s", w[5])
221 end
222 if not role_name(w[6]) then
223 bad("bad role %s", w[6])
224 end
225 role = w[6]
226 if role == "none" then
227 role = nil
228 end
229 end
230 local r = { name = w[2], pattern = w[4], methods = methods, pieces = pieces, role = role, number = #routes + 1, literals = 0, typed = 0, wild = false, selections = 0 }
231 for _, p in ipairs(pieces) do
232 if p.kind == "literal" then
233 r.literals = r.literals + 1
234 elseif p.kind == "int" or p.kind == "slug" then
235 r.typed = r.typed + 1
236 elseif p.kind == "wild" then
237 r.wild = true
238 end
239 end
240 routes[#routes + 1], by_name[r.name] = r, r
241 elseif cmd == "TOKEN" then
242 if #w[2] < 1 or #w[2] > 32 or not w[2]:match("^[A-Za-z0-9%._%-]+$") then
243 bad("bad token %s", w[2])
244 end
245 if tokens[w[2]] then
246 bad("duplicate token %s", w[2])
247 end
248 if not role_name(w[3]) then
249 bad("bad role %s", w[3])
250 end
251 tokens[w[2]] = w[3]
252 elseif cmd == "REDIRECT" then
253 local from, wildcard, ok = rule_path(w[2])
254 if not ok then
255 bad("bad path %s", w[2])
256 end
257 local to, to_wildcard, valid = rule_path(w[3])
258 if not valid then
259 bad("bad path %s", w[3])
260 end
261 if wildcard ~= to_wildcard then
262 bad("wildcard mismatch")
263 end
264 if w[4] ~= "301" and w[4] ~= "302" and w[4] ~= "307" and w[4] ~= "308" then
265 bad("bad redirect code %s", w[4])
266 end
267 for _, r in ipairs(redirects) do
268 if r.from == from and r.wildcard == wildcard then
269 bad("duplicate redirect %s", w[2])
270 end
271 end
272 redirects[#redirects + 1] = { from = from, to = to, wildcard = wildcard, code = tonumber(w[4]) }
273 elseif cmd == "ORIGIN" then
274 local host = w[2]:match("^http://([a-z0-9%.%-%:]+)$") or w[2]:match("^https://([a-z0-9%.%-%:]+)$")
275 if not host then
276 bad("bad origin %s", w[2])
277 end
278 if origins[w[2]] then
279 bad("duplicate origin %s", w[2])
280 end
281 origins[w[2]] = true
282 elseif cmd == "LIMIT" then
283 local n, s = tonumber(w[2]), tonumber(w[3])
284 if not decimal(w[2]) or not decimal(w[3]) or n < 1 or n > 100 or s < 1 or s > 3600 then
285 bad("bad limit")
286 end
287 if capacity > 0 then
288 bad("duplicate limit")
289 end
290 capacity, period = n, s
291 elseif cmd == "TYPES" then
292 local r = by_name[w[2]]
293 if not r then
294 bad("unknown route %s", w[2])
295 end
296 if r.types then
297 bad("duplicate types %s", w[2])
298 end
299 local seen, types = {}, {}
300 for i = 3, #w do
301 local t = w[i]
302 if not media(t) then
303 bad("bad media type %s", t)
304 end
305 if seen[t] then
306 bad("duplicate media type %s", t)
307 end
308 seen[t] = true
309 types[#types + 1] = t
310 end
311 r.types = types
312 end
313end

355local function match(r, segments)
356 local m = #r.pieces -(r.wild and 1 or 0)
357 if not r.wild and #segments ~= m or r.wild and #segments < m then
358 return nil
359 end
360 local params = {}
361 for j, p in ipairs(r.pieces) do
362 local x = segments[j]
363 if p.kind == "literal" then
364 if p.name ~= x then
365 return nil
366 end
367 elseif p.kind == "int" then
368 if not int_value(x) then
369 return nil
370 end
371 params[#params + 1] = { p.name, tostring(tonumber(x)) }
372 elseif p.kind == "slug" then
373 if not slug(x) then
374 return nil
375 end
376 params[#params + 1] = { p.name, x }
377 elseif p.kind == "param" then
378 params[#params + 1] = { p.name, decode(x) }
379 else
380 local values = {}
381 for i = j, #segments do
382 values[#values + 1] = decode(segments[i])
383 end
384 params[#params + 1] = { p.name, table.concat(values, "/") }
385 end
386 end
387 table.sort(params, function (a, b)
388 return a[1] < b[1]
389 end)
390 return params
391end

393local function q_weight(s)
394 local whole, fraction, point = cut(s, ".")
395 if whole ~= "0" and whole ~= "1" then
396 return nil
397 end
398 if point and (#fraction > 3 or fraction ~= "" and not fraction:match("^[0-9]+$")) then
399 return nil
400 end
401 if whole == "1" and fraction:find("[1-9]") then
402 return nil
403 end
404 return tonumber(whole ..(fraction .. "000"):sub(1, 3))
405end

407local function negotiate(types, accept)
408 if accept == nil then
409 return types[1]
410 end
411 local elements = {}
412 for _, text in ipairs(split(accept, ",")) do
413 local fields = split(text, ";")
414 local range, specificity = trim(fields[1]):lower(), 3
415 if range == "*/*" then
416 specificity = 1
417 elseif range:sub(- 2) == "/*" and media_part(range:sub(1, - 3)) then
418 specificity = 2
419 elseif not media(range) then
420 specificity = 0
421 end
422 local weight = 1000
423 for i = 2, #fields do
424 local name, value, equal = cut(trim(fields[i]), "=")
425 if equal and trim(name):lower() == "q" then
426 weight = q_weight(trim(value))
427 break
428 end
429 end
430 if specificity > 0 and weight ~= nil then
431 elements[#elements + 1] = { range = range, specificity = specificity, weight = weight }
432 end
433 end
434 local selected, best = nil, 0
435 for _, t in ipairs(types) do
436 local specificity, weight = 0, 0
437 local main = cut(t, "/")
438 for _, e in ipairs(elements) do
439 if (e.range == "*/*" or e.range == main .. "/*" or e.range == t) and e.specificity > specificity then
440 specificity, weight = e.specificity, e.weight
441 end
442 end
443 if weight > best then
444 selected, best = t, weight
445 end
446 end
447 return selected
448end

459local function body_for(route, params, query, headers, selected)
460 local queries, cookies, seen = {}, {}, {}
461 for _, p in ipairs(split(query, "&")) do
462 if p ~= "" then
463 local name, value = cut(p, "=")
464 queries[#queries + 1] = { decode((name:gsub("%+", " "))), decode((value:gsub("%+", " "))) }
465 end
466 end
467 if headers.cookie then
468 for _, p in ipairs(split(headers.cookie, ";")) do
469 local name, value, equal = cut(trim(p), "=")
470 if equal and name ~= "" and not seen[name] then
471 seen[name] = true
472 cookies[#cookies + 1] = { name, value }
473 end
474 end
475 end
476 table.sort(cookies, function (a, b)
477 return a[1] < b[1]
478 end)
479 if selected == "application/json" then
480 local function object(ps)
481 local out = {}
482 for _, p in ipairs(ps) do
483 out[#out + 1] = quoted(p[1]) .. ":" .. quoted(p[2])
484 end
485 return "{" .. table.concat(out, ",") .. "}"
486 end
487 local qs = {}
488 for _, p in ipairs(queries) do
489 qs[#qs + 1] = "[" .. quoted(p[1]) .. "," .. quoted(p[2]) .. "]"
490 end
491 return '{"route":' .. quoted(route.name) .. ',"params":' .. object(params) .. ',"query":[' .. table.concat(qs, ",") .. '],"cookies":' .. object(cookies) .. "}"
492 elseif selected == "text/html" then
493 local out = { "<dl><dt>route</dt><dd>" .. html(route.name) .. "</dd>" }
494 for i, ps in ipairs({ params, queries, cookies }) do
495 local label =({ "param", "query", "cookie" })[i]
496 for _, p in ipairs(ps) do
497 out[#out + 1] = "<dt>" .. label .. " " .. html(p[1]) .. "</dt><dd>" .. html(p[2]) .. "</dd>"
498 end
499 end
500 out[#out + 1] = "</dl>"
501 return table.concat(out)
502 end

504 if #ps == 0 then

509 out[#out + 1] = encode(p[1]) .. "=" .. encode(p[2])

516local function etag(body)
517 local a, b = 1, 0
518 for i = 1, #body do
519 a =(a + body:byte(i)) % 65521
520 b =(b + a) % 65521
521 end
522 return string.format('"%04x%04x"', b, a)
523end

529local function dispatch(method, target, h)
530 if not METHODS[method] then
531 return reply(501, "not implemented: " .. method)
532 end
533 if not h.host or h.host == "" then
534 return reply(400, "bad request: missing host")
535 end
536 local effective = method
537 if h["x-http-method-override"] ~= nil then
538 local override = h["x-http-method-override"]
539 if method == "POST" and (override == "PUT" or override == "PATCH" or override == "DELETE") then
540 effective = override
541 else
542 return reply(400, "bad request: method override")
543 end
544 end
545 local c = h["x-client"]
546 if c and c ~= "" then
547 local t = tonumber(h["x-time"])
548 if not clients[c] then
549 clients[c] = { level = capacity * period, last = t, allowed = 0, limited = 0 }
550 end
551 local record = clients[c]
552 if capacity > 0 then
553 record.level, record.last = math.min(capacity * period, record.level +(t - record.last) * capacity), t
554 if record.level < period then
555 record.limited = record.limited + 1
556 return reply(429, "too many requests: client " .. c, { ["Retry-After"] = tostring(math.ceil((period - record.level) / capacity)) })
557 end
558 record.level = record.level - period
559 end
560 record.allowed = record.allowed + 1
561 end
562 local path, query, has_query = cut(target, "?")
563 local suffix = has_query and "?" .. query or ""
564 local normalized = normalize(path)
565 if normalized ~= path then
566 local location = normalized .. suffix
567 return reply(308, "redirect to " .. location, { Location = location })
568 end
569 for _, r in ipairs(redirects) do
570 if path == r.from or r.wildcard and starts(path, r.from .. "/") then
571 local location = r.to .. path:sub(#r.from + 1) .. suffix
572 return reply(r.code, "redirect to " .. location, { Location = location })
573 end
574 end
575 local segments = path == "/" and {} or split(path:sub(2), "/")
576 local matches, values, allowed = {}, {}, {}
577 for _, r in ipairs(priority) do
578 local params = match(r, segments)
579 if params then
580 matches[#matches + 1] = r
581 values[r] = params
582 for m in pairs(r.methods) do
583 allowed[m] = true
584 end
585 end
586 end
587 if #matches == 0 then
588 return reply(404, "not found: " .. path)
589 end
590 local allow = table.concat(method_list(allowed, true), ", ")
591 if effective == "OPTIONS" then
592 return reply(204, "", { Allow = allow })
593 end
594 local lookup = effective == "HEAD" and "GET" or effective
595 local selected
596 for _, r in ipairs(matches) do
597 if r.methods[lookup] then
598 selected = r
599 break
600 end
601 end
602 if not selected then
603 return reply(405, "method not allowed: " .. effective, { Allow = allow })
604 end
605 selected.selections = selected.selections + 1
606 if selected.role then
607 if h.authorization == nil then
608 return reply(401, "unauthorized: missing token", { ["WWW-Authenticate"] = 'Bearer realm="api"' })
609 end
610 local scheme, token = cut(h.authorization, " ")
611 if scheme:lower() ~= "bearer" or not tokens[token] then
612 return reply(401, "unauthorized: invalid token", { ["WWW-Authenticate"] = 'Bearer realm="api", error="invalid_token"' })
613 end
614 if tokens[token] ~= selected.role and tokens[token] ~= "admin" then
615 return reply(403, "forbidden: requires role " .. selected.role)
616 end
617 end
618 local selected_type = negotiate(selected.types, h.accept)
619 local response
620 if not selected_type then
621 response = reply(406, "not acceptable: " .. table.concat(selected.types, ", "))
622 else
623 local body = body_for(selected, values[selected], query, h, selected_type)
624 local tag, conditional = etag(body), false
625 if h["if-none-match"] ~= nil then
626 for _, p in ipairs(split(h["if-none-match"], ",")) do
627 p = trim(p)
628 if starts(p, "W/") then
629 p = p:sub(3)
630 end
631 if p == "*" or p == tag then
632 conditional = true
633 break
634 end
635 end
636 end
637 if conditional then
638 if lookup == "GET" then
639 response = reply(304, "", { ETag = tag })
640 else
641 response = reply(412, "precondition failed: " .. tag)
642 end
643 else
644 response = reply(200, body, { ETag = tag })
645 response.content_type = selected_type
646 end
647 end
648 if #selected.types > 1 then
649 response.headers.Vary = "Accept"
650 end
651 return response
652end

654local function request(line, header_lines)
655 local parts = split(line, " ")
656 if #parts ~= 3 or #parts[1] < 1 or #parts[1] > 16 or not parts[1]:match("^[A-Z]+$") or not starts(parts[2], "/") or parts[2]:find("#", 1, true) or parts[3] ~= "HTTP/1.1" then
657 return reply(400, "bad request: request line")
658 end
659 local head = parts[1] == "HEAD"
660 local function early(body)
661 local response = reply(400, body)
662 response.head = head
663 return response
664 end
665 if not escape_check(parts[2]) then
666 return early("bad request: percent escape")
667 end
668 local h = {}
669 for i, text in ipairs(header_lines) do
670 local name, value, colon = cut(text, ":")
671 if not colon or not name:match("^[A-Za-z0-9%-]+$") then
672 return early("bad request: header line " .. i)
673 end
674 name = name:lower()
675 if h[name] ~= nil then
676 return early("bad request: duplicate header " .. name)
677 end
678 h[name] = trim(value)
679 end
680 local response = dispatch(parts[1], parts[2], h)
681 response.head = head
682 if h.origin ~= nil and origins[h.origin] then
683 response.headers["Access-Control-Allow-Origin"] = h.origin
684 if response.status == 204 and h["access-control-request-method"] ~= nil then
685 response.headers["Access-Control-Allow-Methods"] = response.headers.Allow
686 end
687 end
688 return response
689end

693local function print_response(response)
694 requests = requests + 1
695 statuses[response.status] =(statuses[response.status] or 0) + 1

712local function urlfor(w)
713 local name, r = w[2], by_name[w[2]]
714 if not r then
715 bad("unknown route %s", name)
716 end
717 local parameters, values, query = {}, {}, {}
718 for _, p in ipairs(r.pieces) do
719 if p.kind ~= "literal" then
720 parameters[p.name] = true
721 end
722 end
723 for i = 3, #w do
724 local key, value, equal = cut(w[i], "=")
725 if not equal or key == "" then
726 bad("bad argument %s", w[i])
727 end
728 if parameters[key] then
729 if values[key] ~= nil then
730 bad("duplicate parameter %s", key)
731 end
732 values[key] = value
733 else
734 query[#query + 1] = { key, value }
735 end
736 end
737 local path = {}
738 for _, p in ipairs(r.pieces) do
739 if p.kind == "literal" then
740 path[#path + 1] = p.name
741 else
742 local value = values[p.name]
743 if p.kind ~= "wild" and value == nil then
744 bad("missing parameter %s", p.name)
745 end
746 value = value or ""
747 local valid = true
748 if p.kind == "param" then
749 valid = value ~= "" and value ~= "." and value ~= ".."
750 elseif p.kind == "int" then
751 valid = int_value(value)
752 elseif p.kind == "slug" then
753 valid = slug(value)
754 elseif value ~= "" then
755 for _, part in ipairs(split(value, "/")) do
756 if part == "" or part == "." or part == ".." then
757 valid = false
758 break
759 end
760 end
761 end
762 if not valid then
763 bad("bad value for %s", p.name)
764 end
765 if p.kind == "wild" then
766 if value ~= "" then
767 for _, part in ipairs(split(value, "/")) do
768 path[#path + 1] = encode(part)
769 end
770 end
771 else
772 path[#path + 1] = encode(value)
773 end
774 end
775 end
776 local url = "/" .. table.concat(path, "/")
777 if #query > 0 then
778 local parts = {}
779 for _, p in ipairs(query) do
780 parts[#parts + 1] = encode(p[1]) .. "=" .. encode(p[2])
781 end
782 url = url .. "?" .. table.concat(parts, "&")
783 end
784 return "URLFOR " .. name .. " " .. url
785end
