114class Piece:
115 def __init__(self, kind, text):
116 self.kind = kind
117 self.text = text
118
119 def fits(self, segment):
120 if self.kind == LITERAL:
121 return segment == self.text
122 if self.kind == INT:
123 return INT_VALUE.fullmatch(segment) is not None
124 if self.kind == SLUG:
125 return SLUG_VALUE.fullmatch(segment) is not None
126 return True
127
128 def accepts(self, value):
129 if self.kind == UNTYPED:
130 return value not in ("", ".", "..")
131 if self.kind == WILDCARD:
132 return value == "" or all(part not in ("", ".", "..") for part in value.split("/"))
133 return self.fits(value)
134
135
136def parse_piece(text):
137 if is_segment(text):
138 return Piece(LITERAL, text)
139 wildcard = WILDCARD_PIECE.fullmatch(text)
140 if wildcard is not None:
141 return Piece(WILDCARD, wildcard.group(1))
142 parameter = PARAMETER_PIECE.fullmatch(text)
143 if parameter is not None:
144 return Piece(PARAMETER_KINDS[parameter.group(2)], parameter.group(1))
145 return None
146
147
148def parse_pattern(pattern):
149 if not pattern.startswith("/"):
150 return None
151 if pattern == "/":
152 return []
153 pieces = []
154 for text in pattern[1:].split("/"):
155 piece = parse_piece(text)
156 if piece is None:
157 return None
158 pieces.append(piece)
159 if any(piece.kind == WILDCARD for piece in pieces[:-1]):
160 return None
161 names = [piece.text for piece in pieces if piece.kind != LITERAL]
162 if len(set(names)) != len(names):
163 return None
164 return pieces

167class Route:
168 def __init__(self, name, methods, pattern, pieces, role, number):
169 self.name = name
170 self.methods = methods
171 self.pattern = pattern
172 self.pieces = pieces
173 self.role = role
174 self.number = number
175 self.types = None
176 self.selections = 0
177 self.fixed = [piece for piece in pieces if piece.kind != WILDCARD]
178 self.has_wildcard = len(self.fixed) < len(pieces)
179
180 def priority(self):
181 literals = sum(piece.kind == LITERAL for piece in self.pieces)
182 typed = sum(piece.kind in (INT, SLUG) for piece in self.pieces)
183 return -literals, -typed, self.has_wildcard, self.number
184
185 def matches(self, segments):
186 count = len(self.fixed)
187 if len(segments) < count or (len(segments) > count and not self.has_wildcard):
188 return False
189 return all(piece.fits(segment) for piece, segment in zip(self.fixed, segments))
190
191 def parameters(self, segments):
192 values = {}
193 for piece, segment in zip(self.fixed, segments):
194 if piece.kind == UNTYPED:
195 values[piece.text] = decode(segment)
196 elif piece.kind == INT:
197 values[piece.text] = str(int(segment))
198 elif piece.kind == SLUG:
199 values[piece.text] = segment
200 if self.has_wildcard:
201 rest = segments[len(self.fixed):]
202 values[self.pieces[-1].text] = "/".join(decode(segment) for segment in rest)
203 return values
204
205 def build_path(self, values):
206 segments = []
207 for piece in self.pieces:
208 if piece.kind == LITERAL:
209 segments.append(piece.text)
210 continue
211 value = values.get(piece.text)
212 if value is None and piece.kind != WILDCARD:
213 raise UrlError(f"missing parameter {piece.text}")
214 if value is not None and not piece.accepts(value):
215 raise UrlError(f"bad value for {piece.text}")
216 if piece.kind != WILDCARD:
217 segments.append(encode(value))
218 elif value:
219 segments.extend(encode(part) for part in value.split("/"))
220 return "/" + "/".join(segments)

223def parse_rule_path(path):
224 if path.endswith("/*"):
225 base = path[:-2]
226 if base != "/" and is_canonical_path(base):
227 return base, True
228 return None
229 if is_canonical_path(path):
230 return path, False
231 return None

247class RateLimit:
248 def __init__(self, capacity, period):
249 self.capacity = capacity
250 self.period = period
251 self.buckets = {}
252
253 def consume(self, client, time):
254 full = self.capacity * self.period
255 level, last = self.buckets.get(client, (full, time))
256 level = min(full, level + (time - last) * self.capacity)
257 if level < self.period:
258 self.buckets[client] = (level, time)
259 return (self.period - level + self.capacity - 1) // self.capacity
260 self.buckets[client] = (level - self.period, time)
261 return None

312 segments = []

322def parse_weight(q_value):
323 if Q_VALUE.fullmatch(q_value) is None:
324 return None
325 whole, _, fraction = q_value.partition(".")
326 return int(whole) * 1000 + int(fraction.ljust(3, "0"))

329def range_specificity(media_range):
330 if media_range == "*/*":
331 return 1
332 if MAIN_TYPE_RANGE.fullmatch(media_range):
333 return 2
334 if MEDIA_TYPE.fullmatch(media_range):
335 return 3
336 return 0

339def parse_accept(value):
340 elements = []
341 for element in value.split(","):
342 first, *parameters = [part.strip() for part in element.split(";")]
343 media_range = first.lower()
344 specificity = range_specificity(media_range)
345 weight = 1000
346 for parameter in parameters:
347 name, equals, q_value = parameter.partition("=")
348 if equals and name.strip().lower() == "q":
349 weight = parse_weight(q_value.strip())
350 break
351 if specificity > 0 and weight is not None:
352 elements.append((media_range, specificity, weight))
353 return elements

356def range_fits(media_range, media_type):
357 main_type, _, subtype = media_range.partition("/")
358 if subtype == "*":
359 return main_type == "*" or media_type.startswith(main_type + "/")
360 return media_range == media_type

363def negotiate(types, accept):
364 if accept is None:
365 return types[0]
366 elements = parse_accept(accept)
367 selected, best_weight = None, 0
368 for media_type in types:
369 weight, best_specificity = 0, 0
370 for media_range, specificity, element_weight in elements:
371 if specificity > best_specificity and range_fits(media_range, media_type):
372 weight, best_specificity = element_weight, specificity
373 if weight > best_weight:
374 selected, best_weight = media_type, weight
375 return selected

389def parse_cookies(header):
390 cookies = {}
391 if header is None:
392 return []
393 for part in header.split(";"):
394 name, equals, value = part.strip().partition("=")
395 if equals and name and name not in cookies:
396 cookies[name] = value

459class Router:
460 def __init__(self):
461 self.routes = {}
462 self.ranked = []
463 self.tokens = {}
464 self.redirects = {}
465 self.origins = set()
466 self.limit = None
467 self.config_errors = 0
468 self.request_count = 0
469 self.urlfor_ok = 0
470 self.urlfor_errors = 0
471 self.status_counts = Counter()
472 self.clients = {}

474 def configure(self, number, line):
475 try:
476 self.run_command(line)
477 except ConfigError as error:
478 self.config_errors += 1
479 print(f"CONFIG ERROR {number}: {error}")
480
481 def run_command(self, line):
482 command, *arguments = line.split()
483 count = len(arguments)
484 commands = {
485 "ROUTE": (self.add_route, count in (3, 5)),
486 "TOKEN": (self.add_token, count == 2),
487 "REDIRECT": (self.add_redirect, count == 3),
488 "ORIGIN": (self.add_origin, count == 1),
489 "LIMIT": (self.set_limit, count == 2),
490 "TYPES": (self.set_types, count >= 2),
491 }
492 if command not in commands:
493 raise ConfigError(f"unknown command {command}")
494 handler, valid_count = commands[command]
495 if not valid_count:
496 raise ConfigError("wrong number of arguments")
497 handler(*arguments)

499 def add_route(self, name, methods_text, pattern, option=None, role=NO_ROLE):
500 if VALID_NAME.fullmatch(name) is None:
501 raise ConfigError(f"bad route name {name}")
502 if name in self.routes:
503 raise ConfigError(f"duplicate route {name}")
504 methods = methods_text.split(",")
505 if not set(methods) <= set(ROUTE_METHODS) or len(set(methods)) != len(methods):
506 raise ConfigError(f"bad methods {methods_text}")
507 pieces = parse_pattern(pattern)
508 if pieces is None:
509 raise ConfigError(f"bad pattern {pattern}")
510 if option is not None:
511 if option != "ROLE":
512 raise ConfigError(f"unknown option {option}")
513 if ROLE_NAME.fullmatch(role) is None:
514 raise ConfigError(f"bad role {role}")
515 self.routes[name] = Route(name, set(methods), pattern, pieces, role, len(self.routes))
516
517 def add_token(self, value, role):
518 if TOKEN_VALUE.fullmatch(value) is None:
519 raise ConfigError(f"bad token {value}")
520 if value in self.tokens:
521 raise ConfigError(f"duplicate token {value}")
522 if ROLE_NAME.fullmatch(role) is None:
523 raise ConfigError(f"bad role {role}")
524 self.tokens[value] = role
525
526 def add_redirect(self, source, target, code):
527 source_rule = parse_rule_path(source)
528 if source_rule is None:
529 raise ConfigError(f"bad path {source}")
530 target_rule = parse_rule_path(target)
531 if target_rule is None:
532 raise ConfigError(f"bad path {target}")
533 source_base, source_is_prefix = source_rule
534 target_base, target_is_prefix = target_rule
535 if source_is_prefix != target_is_prefix:
536 raise ConfigError("wildcard mismatch")
537 if code not in REDIRECT_CODES:
538 raise ConfigError(f"bad redirect code {code}")
539 if source in self.redirects:
540 raise ConfigError(f"duplicate redirect {source}")
541 self.redirects[source] = RedirectRule(source_base, source_is_prefix, target_base, int(code))
542
543 def add_origin(self, origin):
544 if ORIGIN.fullmatch(origin) is None:
545 raise ConfigError(f"bad origin {origin}")
546 if origin in self.origins:
547 raise ConfigError(f"duplicate origin {origin}")
548 self.origins.add(origin)

550 def set_limit(self, capacity, period):
551 if not (is_decimal_between(capacity, 1, 100) and is_decimal_between(period, 1, 3600)):
552 raise ConfigError("bad limit")
553 if self.limit is not None:
554 raise ConfigError("duplicate limit")
555 self.limit = RateLimit(int(capacity), int(period))
556
557 def set_types(self, name, *types):
558 route = self.routes.get(name)
559 if route is None:
560 raise ConfigError(f"unknown route {name}")
561 if route.types is not None:
562 raise ConfigError(f"duplicate types {name}")
563 for index, media_type in enumerate(types):
564 if MEDIA_TYPE.fullmatch(media_type) is None:
565 raise ConfigError(f"bad media type {media_type}")
566 if media_type in types[:index]:
567 raise ConfigError(f"duplicate media type {media_type}")
568 route.types = list(types)

570 def finish_configuration(self):
571 limit = f"{self.limit.capacity}/{self.limit.period}" if self.limit else "none"
572 print(
573 f"CONFIG routes={len(self.routes)} tokens={len(self.tokens)} "
574 f"redirects={len(self.redirects)} origins={len(self.origins)} "
575 f"limit={limit} errors={self.config_errors}"
576 )
577 for route in self.routes.values():
578 if route.types is None:
579 route.types = [DEFAULT_TYPE]
580 self.ranked = sorted(self.routes.values(), key=Route.priority)
581 for rank, route in enumerate(self.ranked, 1):
582 methods = ",".join(sorted(with_head(route.methods)))
583 role = "-" if route.role == NO_ROLE else route.role
584 print(
585 f"TABLE {rank} {route.name} {methods} {route.pattern} "
586 f"role={role} types={','.join(route.types)}"
587 )

589 def handle_request(self, request_line, header_lines):
590 self.request_count += 1
591 parts = request_line.split(" ")
592 if len(parts) == 3 and is_valid_request_line(*parts):
593 method, target, _ = parts
594 response = self.respond(method, target, header_lines)
595 show_body = method != "HEAD"
596 else:
597 response = Response(400, "bad request: request line")
598 show_body = True
599 self.status_counts[response.status] += 1
600 print(f"REQUEST {self.request_count}")
601 response.write(show_body)

603 def respond(self, method, target, header_lines):
604 if not has_valid_escapes(target):
605 return Response(400, "bad request: percent escape")
606 headers = {}
607 for number, line in enumerate(header_lines, 1):
608 name, colon, value = line.partition(":")
609 if not colon or HEADER_NAME.fullmatch(name) is None:
610 return Response(400, f"bad request: header line {number}")
611 name = name.lower()
612 if name in headers:
613 return Response(400, f"bad request: duplicate header {name}")
614 headers[name] = value.strip()
615 response = self.dispatch(method, target, headers)
616 origin = headers.get("origin")
617 if origin in self.origins:
618 response.headers["Access-Control-Allow-Origin"] = origin
619 if response.status == 204 and "access-control-request-method" in headers:
620 response.headers["Access-Control-Allow-Methods"] = response.headers["Allow"]
621 return response

623 def dispatch(self, method, target, headers):
624 if method not in KNOWN_METHODS:
625 return Response(501, f"not implemented: {method}")
626 if not headers.get("host"):
627 return Response(400, "bad request: missing host")
628 effective_method = method
629 override = headers.get("x-http-method-override")
630 if override is not None:
631 if method != "POST" or override not in OVERRIDE_METHODS:
632 return Response(400, "bad request: method override")
633 effective_method = override
634 client = headers.get("x-client")
635 if client:
636 wait = self.check_rate(client, int(headers["x-time"]))
637 if wait is not None:
638 return Response(429, f"too many requests: client {client}", {"Retry-After": wait})
639 path, question, query = target.partition("?")
640 query_suffix = question + query
641 normalized = normalize_path(path)
642 if normalized != path:
643 return redirect_response(308, normalized + query_suffix)
644 for rule in self.redirects.values():
645 destination = rule.destination(path)
646 if destination is not None:
647 return redirect_response(rule.code, destination + query_suffix)
648 return self.route_request(effective_method, path, query if question else None, headers)

650 def check_rate(self, client, time):
651 counts = self.clients.setdefault(client, Counter())
652 wait = self.limit.consume(client, time) if self.limit else None
653 if wait is None:
654 counts["allowed"] += 1
655 else:
656 counts["limited"] += 1
657 return wait

659 def route_request(self, method, path, query, headers):
660 segments = path[1:].split("/") if path != "/" else []
661 matching = [route for route in self.ranked if route.matches(segments)]
662 if not matching:
663 return Response(404, f"not found: {path}")
664 allowed = with_head({"OPTIONS"}.union(*(route.methods for route in matching)))
665 allow = ", ".join(sorted(allowed))
666 if method == "OPTIONS":
667 return Response(204, headers={"Allow": allow})
668 lookup = "GET" if method == "HEAD" else method
669 route = next((route for route in matching if lookup in route.methods), None)
670 if route is None:
671 return Response(405, f"method not allowed: {method}", {"Allow": allow})
672 route.selections += 1
673 params = sorted(route.parameters(segments).items())
674 denied = self.authorize(route.role, headers.get("authorization"))
675 if denied is not None:
676 return denied
677 vary = {"Vary": "Accept"} if len(route.types) > 1 else {}
678 media_type = negotiate(route.types, headers.get("accept"))
679 if media_type is None:
680 return Response(406, f"not acceptable: {', '.join(route.types)}", vary)
681 cookies = parse_cookies(headers.get("cookie"))
682 body = render_body(media_type, route.name, params, parse_query(query), cookies)
683 etag = etag_of(body)
684 if_none_match = headers.get("if-none-match")
685 if if_none_match is not None and etag_matches(if_none_match, etag):
686 if lookup == "GET":
687 return Response(304, headers=vary | {"ETag": etag})
688 return Response(412, f"precondition failed: {etag}", vary)
689 return Response(200, body, vary | {"ETag": etag}, media_type)

691 def authorize(self, role, authorization):
692 if role == NO_ROLE:
693 return None
694 if authorization is None:
695 challenge = 'Bearer realm="api"'
696 return Response(401, "unauthorized: missing token", {"WWW-Authenticate": challenge})
697 scheme, _, token = authorization.partition(" ")
698 if scheme.lower() != "bearer" or token not in self.tokens:
699 challenge = 'Bearer realm="api", error="invalid_token"'
700 return Response(401, "unauthorized: invalid token", {"WWW-Authenticate": challenge})
701 if self.tokens[token] not in (role, ADMIN_ROLE):
702 return Response(403, f"forbidden: requires role {role}")
703 return None

705 def url_for(self, name, *arguments):
706 try:
707 url = self.build_url(name, arguments)
708 except UrlError as error:
709 self.urlfor_errors += 1
710 print(f"URLFOR ERROR {error}")
711 else:
712 self.urlfor_ok += 1
713 print(f"URLFOR {name} {url}")
714
715 def build_url(self, name, arguments):
716 route = self.routes.get(name)
717 if route is None:
718 raise UrlError(f"unknown route {name}")
719 parameter_names = {piece.text for piece in route.pieces if piece.kind != LITERAL}
720 values = {}
721 query = []
722 for argument in arguments:
723 key, equals, value = argument.partition("=")
724 if not equals or not key:
725 raise UrlError(f"bad argument {argument}")
726 if key not in parameter_names:
727 query.append((key, value))
728 elif key in values:
729 raise UrlError(f"duplicate parameter {key}")
730 else:
731 values[key] = value
732 url = route.build_path(values)
733 if query:
734 url += "?" + join_encoded(query)
735 return url
