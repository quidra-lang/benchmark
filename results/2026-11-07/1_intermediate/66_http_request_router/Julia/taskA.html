32function partition(s, separator)
33 p=findfirst(separator, s)
34 p===nothing ? (String(s), false, "") : (part(s, 1, first(p)-1), true, part(s, last(p)+1))
35end
36segment(s) = !isempty(s) && all(c->c in UNRESERVED, s) && !(s in (".", ".."))
37canonical(s) =
38 s=="/" || startswith(s, "/") && all(segment, split(part(s, 2), '/'; keepempty = true))
39decimal(s, lo, hi) = occursin(r"^(?:0|[1-9][0-9]*)$", s) && lo<=parse(BigInt, s)<=hi
40decode(s) = replace(s, r"%[0-9A-Fa-f]{2}"=>x->string(Char(parse(Int, part(x, 2); base = 16))))
41encode(s) =
42 join(c in UNRESERVED ? string(c) : "%"*uppercase(string(Int(c); base = 16, pad = 2)) for c in s)
43encoded(pairs) = join((encode(k)*"="*encode(v) for (k, v) in pairs), "&")
44withhead(methods) = "GET" in methods ? union(Set(methods), Set(["HEAD"])) : Set(methods)
45struct Piece
46 kind::Symbol
47 name::String
48end
49function pattern(s)
50 startswith(s, "/") || return nothing
51 s=="/" && return Piece[]
52 pieces=Piece[]
53 names=Set{String}()
54 fields=split(part(s, 2), '/'; keepempty = true)
55 for (i, field) in enumerate(fields)
56 if segment(field)
57 push!(pieces, Piece(:literal, field))
58 continue
59 end
60 m=match(r"^\{([a-z][a-z0-9_]{0,19})(?::(int|slug))?\}$", field)
61 w=match(r"^\{\*([a-z][a-z0-9_]{0,19})\}$", field)
62 if w!==nothing && i==length(fields)
63 k=:wildcard
64 name=String(w.captures[1])
65 elseif m!==nothing
66 k=m.captures[2]===nothing ? :untyped : Symbol(m.captures[2])
67 name=String(m.captures[1])
68 else
69 return nothing
70 end
71 name in names && return nothing
72 push!(names, name)
73 push!(pieces, Piece(k, name))
74 end
75 pieces
76end
77fits(p, s) =
78 p.kind==:literal ? s==p.name :
79 p.kind==:int ? occursin(r"^[0-9]{1,9}$", s) :
80 p.kind==:slug ? occursin(r"^[a-z0-9-]+$", s) : true
81accepts(p, s) =
82 p.kind==:untyped ? !(s in ("", ".", "..")) :
83 p.kind==:wildcard ?
84 isempty(s) || all(x->!(x in ("", ".", "..")), split(s, '/'; keepempty = true)) : fits(p, s)

95wild(r) = !isempty(r.pieces) && last(r.pieces).kind==:wildcard
96priority(r) = (
97 -count(p->p.kind==:literal, r.pieces),
98 -count(p->p.kind in (:int, :slug), r.pieces),
99 wild(r),
100 r.number,

102function matches(r, segments)
103 n=length(r.pieces)-wild(r)
104 length(segments)>=n &&
105 (wild(r) || length(segments)==n) &&
106 all(i->fits(r.pieces[i], segments[i]), 1:n)
107end

109 if endswith(s, "/*")
110 base=part(s, 1, length(s)-2)
111 return base!="/" && canonical(base) ? (base, true) : nothing
112 end
113 canonical(s) ? (String(s), false) : nothing
114end

162function configure!(r, line)
163 words=String.(split(line, ' '))
164 cmd=words[1]
165 a=words[2:end]
166 n=length(a)
167 cmd in ("ROUTE", "TOKEN", "REDIRECT", "ORIGIN", "LIMIT", "TYPES") ||
168 fail("unknown command $cmd")
169 valid=cmd=="ROUTE" ? n in (3, 5) :
170 cmd=="TOKEN" ? n==2 :
171 cmd=="REDIRECT" ? n==3 : cmd=="ORIGIN" ? n==1 : cmd=="LIMIT" ? n==2 : n>=2
172 valid || fail("wrong number of arguments")
173 if cmd=="ROUTE"
174 name, methods, word=a[1:3]
175 occursin(NAME, name) || fail("bad route name $name")
176 findroute(r, name)!==nothing && fail("duplicate route $name")
177 ms=String.(split(methods, ','; keepempty = true))
178 all(m->m in ("GET", "POST", "PUT", "PATCH", "DELETE"), ms) &&
179 length(unique(ms))==length(ms) || fail("bad methods $methods")
180 pieces=pattern(word)
181 pieces===nothing && fail("bad pattern $word")
182 role="none"
183 if n==5
184 a[4]=="ROLE" || fail("unknown option $(a[4])")
185 occursin(ROLE, a[5]) || fail("bad role $(a[5])")
186 role=a[5]
187 end
188 push!(r.routes, Route(name, Set(ms), word, pieces, role, length(r.routes)+1, String[], 0))
189 elseif cmd=="TOKEN"
190 token, role=a
191 occursin(r"^[A-Za-z0-9._-]{1,32}$", token) || fail("bad token $token")
192 haskey(r.tokens, token) && fail("duplicate token $token")
193 occursin(ROLE, role) || fail("bad role $role")
194 r.tokens[token]=role
195 elseif cmd=="REDIRECT"
196 from, to, code=a
197 src=rulepath(from)
198 src===nothing && fail("bad path $from")
199 dst=rulepath(to)
200 dst===nothing && fail("bad path $to")
201 src[2]==dst[2] || fail("wildcard mismatch")
202 code in ("301", "302", "307", "308") || fail("bad redirect code $code")
203 any(x->x[1]==from, r.redirects) && fail("duplicate redirect $from")
204 push!(r.redirects, (from, src[1], src[2], dst[1], parse(Int, code)))
205 elseif cmd=="ORIGIN"
206 s=a[1]
207 occursin(r"^https?://[a-z0-9.:-]+$", s) || fail("bad origin $s")
208 s in r.origins && fail("duplicate origin $s")
209 push!(r.origins, s)
210 elseif cmd=="LIMIT"
211 decimal(a[1], 1, 100) && decimal(a[2], 1, 3600) || fail("bad limit")
212 r.capacity>0 && fail("duplicate limit")
213 r.capacity=parse(Int, a[1])
214 r.period=parse(Int, a[2])
215 else
216 name=a[1]
217 idx=findroute(r, name)
218 idx===nothing && fail("unknown route $name")
219 route=r.routes[idx]
220 !isempty(route.types) && fail("duplicate types $name")
221 seen=Set{String}()
222 for t in a[2:end]
223 occursin(MEDIA, t) || fail("bad media type $t")
224 t in seen && fail("duplicate media type $t")
225 push!(seen, t)
226 end
227 route.types=a[2:end]
228 end
229end
230function finish!(r)
231 limit=r.capacity==0 ? "none" : "$(r.capacity)/$(r.period)"
232 println(
233 "CONFIG routes=$(length(r.routes)) tokens=$(length(r.tokens)) redirects=$(length(r.redirects)) origins=$(length(r.origins)) limit=$limit errors=$(r.errors)",
234 )
235 for route in r.routes
236 isempty(route.types) && push!(route.types, "application/json")
237 end
238 r.ranked=sort(r.routes; by = priority)
239 for (i, route) in enumerate(r.ranked)
240 println(
241 "TABLE $i $(route.name) $(join(sort!(collect(withhead(route.methods))),',')) $(route.pattern) role=$(route.role=="none" ? "-" : route.role) types=$(join(route.types,','))",
242 )
243 end
244end
245function normalized(path)
246 escaped=replace(path, r"%[0-9A-Fa-f]{2}"=>s->begin
247 c=Char(parse(Int, part(s, 2); base = 16))
248 c in UNRESERVED ? string(c) : uppercase(s)
249 end)
250 pieces=String[]
251 for s in split(part(escaped, 2), '/'; keepempty = true)
252 if s==".."
253 !isempty(pieces) && pop!(pieces)
254 elseif !(s in ("", "."))
255 push!(pieces, s)
256 end
257 end
258 "/"*join(pieces, '/')
259end
260function negotiate(types, accept)
261 accept===nothing && return types[1]
262 elements=Tuple{String,Int,Int}[]
263 for raw in split(accept, ','; keepempty = true)
264 parts=trim.(split(raw, ';'; keepempty = true))
265 range=lowercase(parts[1])
266 specificity=range=="*/*" ? 1 :
267 occursin(r"^[a-z0-9.+-]+/\*$", range) ? 2 : occursin(MEDIA, range) ? 3 : 0
268 weight=1000
269 for parameter in parts[2:end]
270 key, equal, v=partition(parameter, "=")
271 if equal && lowercase(trim(key))=="q"
272 v=trim(v)
273 if !occursin(r"^(?:0(?:\.[0-9]{0,3})?|1(?:\.0{0,3})?)$", v)
274 weight=-1
275 else
276 whole, _, fraction=partition(v, ".")
277 weight=1000*parse(Int, whole)+parse(Int, rpad(fraction, 3, '0'))
278 end
279 break
280 end
281 end
282 specificity>0 && weight>=0 && push!(elements, (range, specificity, weight))
283 end
284 best=nothing
285 bestweight=0
286 for type in types
287 specificity=0
288 weight=0
289 for (range, rank, w) in elements
290 fits=range=="*/*" ||
291 endswith(range, "/*") && startswith(type, part(range, 1, length(range)-1)) ||
292 range==type
293 if fits && rank>specificity
294 specificity=rank
295 weight=w
296 end
297 end
298 if weight>bestweight
299 best=type
300 bestweight=weight
301 end
302 end
303 best
304end
305jsonstring(s) = "\""*replace(s, '\\'=>"\\\\", '"'=>"\\\"")*"\""
306jsonobject(pairs) = "{"*join((jsonstring(k)*":"*jsonstring(v) for (k, v) in pairs), ',')*"}"
307html(s) = replace(s, '&'=>"&amp;", '<'=>"&lt;", '>'=>"&gt;", '"'=>"&quot;")
308function body(type, name, params, query, cookies)
309 if type=="application/json"
310 q=join(("["*jsonstring(k)*","*jsonstring(v)*"]" for (k, v) in query), ',')
311 return "{\"route\":"*jsonstring(name)*",\"params\":"*jsonobject(params)*",\"query\":["*q*"],\"cookies\":"*jsonobject(
312 cookies,
313 )*"}"
314 elseif type=="text/html"
315 entries=[("route", name)]
316 for (prefix, pairs) in (("param", params), ("query", query), ("cookie", cookies))
317 append!(entries, [(prefix*" "*k, v) for (k, v) in pairs])
318 end
319 return "<dl>"*join("<dt>$(html(k))</dt><dd>$(html(v))</dd>" for (k, v) in entries)*"</dl>"
320 end
321 lists=[isempty(pairs) ? "-" : encoded(pairs) for pairs in (params, query, cookies)]
322 "route=$name params=$(lists[1]) query=$(lists[2]) cookies=$(lists[3])"
323end
324function etag(s)
325 a=1
326 b=0
327 for c in s
328 a=mod(a+Int(c), 65521)
329 b=mod(b+a, 65521)
330 end
331 "\""*string(b; base = 16, pad = 4)*string(a; base = 16, pad = 4)*"\""
332end
333function route_request!(r, method, path, query, h)
334 segments=path=="/" ? String[] : String.(split(part(path, 2), '/'))
335 matching=filter(route->matches(route, segments), r.ranked)
336 isempty(matching) && return Response(404, "not found: $path")
337 allowed=Set(["OPTIONS"])
338 for route in matching
339 union!(allowed, route.methods)
340 end
341 allow=join(sort!(collect(withhead(allowed))), ", ")
342 method=="OPTIONS" && return Response(204, nothing, Dict("Allow"=>allow))
343 lookup=method=="HEAD" ? "GET" : method
344 idx=findfirst(route->lookup in route.methods, matching)
345 idx===nothing && return Response(405, "method not allowed: $method", Dict("Allow"=>allow))
346 route=matching[idx]
347 route.selections+=1
348 if route.role!="none"
349 !haskey(h, "authorization") && return Response(
350 401,
351 "unauthorized: missing token",
352 Dict("WWW-Authenticate"=>"Bearer realm=\"api\""),
353 )
354 scheme, _, token=partition(h["authorization"], " ")
355 (lowercase(scheme)!="bearer" || !haskey(r.tokens, token)) && return Response(
356 401,
357 "unauthorized: invalid token",
358 Dict("WWW-Authenticate"=>"Bearer realm=\"api\", error=\"invalid_token\""),
359 )
360 r.tokens[token] in (route.role, "admin") ||
361 return Response(403, "forbidden: requires role $(route.role)")
362 end
363 vary=length(route.types)>1 ? Dict("Vary"=>"Accept") : Dict{String,String}()
364 type=negotiate(route.types, get(h, "accept", nothing))
365 type===nothing && return Response(406, "not acceptable: $(join(route.types,", "))", vary)
366 params=Tuple{String,String}[]
367 for (i, p) in enumerate(route.pieces)
368 p.kind==:literal && continue
369 v=p.kind==:wildcard ? join(decode.(segments[i:end]), '/') :
370 p.kind==:int ? string(parse(Int, segments[i])) :
371 p.kind==:untyped ? decode(segments[i]) : segments[i]
372 push!(params, (p.name, v))
373 end
374 sort!(params; by = first)
375 queries=Tuple{String,String}[]
376 if query!==nothing
377 for raw in split(query, '&'; keepempty = true)
378 isempty(raw) && continue
379 k, _, v=partition(raw, "=")
380 push!(queries, (decode(replace(k, '+'=>' ')), decode(replace(v, '+'=>' '))))
381 end
382 end
383 cookies=Dict{String,String}()
384 for raw in split(get(h, "cookie", ""), ';'; keepempty = true)
385 k, equal, v=partition(trim(raw), "=")
386 equal && !isempty(k) && !haskey(cookies, k) && (cookies[k]=v)
387 end
388 result=body(type, route.name, params, queries, sort!(collect(cookies); by = first))
389 tag=etag(result)
390 if haskey(h, "if-none-match")
391 hit=any(
392 raw->begin
393 x=trim(raw)
394 startswith(x, "W/") && (x=part(x, 3))
395 x in ("*", tag)
396 end,
397 split(h["if-none-match"], ','; keepempty = true),
398 )
399 if hit
400 if lookup=="GET"
401 vary["ETag"]=tag
402 return Response(304, nothing, vary)
403 else
404 return Response(412, "precondition failed: $tag", vary)
405 end
406 end
407 end
408 vary["ETag"]=tag
409 Response(200, result, vary, type)
410end
411function dispatch!(r, method, target, h)
412 method in ("GET", "HEAD", "POST", "PUT", "PATCH", "DELETE", "OPTIONS") ||
413 return Response(501, "not implemented: $method")
414 isempty(get(h, "host", "")) && return Response(400, "bad request: missing host")
415 effective=method
416 if haskey(h, "x-http-method-override")
417 override=h["x-http-method-override"]
418 method=="POST" && override in ("PUT", "PATCH", "DELETE") ||
419 return Response(400, "bad request: method override")
420 effective=override
421 end
422 client=get(h, "x-client", "")
423 if !isempty(client)
424 time=parse(Int, h["x-time"])
425 c=get!(r.clients, client, Client(r.capacity*r.period, time, 0, 0))
426 wait=0
427 if r.capacity>0
428 c.level=min(r.capacity*r.period, c.level+(time-c.time)*r.capacity)
429 c.time=time
430 if c.level<r.period
431 wait=cld(r.period-c.level, r.capacity)
432 else
433 c.level-=r.period
434 end
435 end
436 if wait>0
437 c.limited+=1
438 return Response(
439 429,
440 "too many requests: client $client",
441 Dict("Retry-After"=>string(wait)),
442 )
443 end
444 c.allowed+=1
445 end
446 path, question, query=partition(target, "?")
447 suffix=question ? "?"*query : ""
448 normal=normalized(path)
449 normal!=path && return redirect(308, normal*suffix)
450 for (_, base, prefix, to, code) in r.redirects
451 (path==base || prefix && startswith(path, base*"/")) &&
452 return redirect(code, to*part(path, length(base)+1)*suffix)
453 end
454 route_request!(r, effective, path, question ? query : nothing, h)
455end
456function respond!(r, line, headers)
457 fields=String.(split(line, ' '; keepempty = true))
458 valid=length(fields)==3 &&
459 occursin(r"^[A-Z]{1,16}$", fields[1]) &&
460 startswith(fields[2], "/") &&
461 !occursin('#', fields[2]) &&
462 fields[3]=="HTTP/1.1"
463 valid || return Response(400, "bad request: request line"), true
464 method, target, _=fields
465 showbody=method!="HEAD"
466 for i in eachindex(target)
467 if target[i]=='%'
468 raw=part(target, i+1, min(i+2, length(target)))
469 occursin(r"^[0-9A-Fa-f]{2}$", raw) && 32<=parse(Int, raw; base = 16)<=126 ||
470 return Response(400, "bad request: percent escape"), showbody
471 end
472 end
473 h=Dict{String,String}()
474 for (i, header) in enumerate(headers)
475 name, colon, v=partition(header, ":")
476 colon && occursin(r"^[A-Za-z0-9-]+$", name) ||
477 return Response(400, "bad request: header line $i"), showbody
478 name=lowercase(name)
479 haskey(h, name) && return Response(400, "bad request: duplicate header $name"), showbody
480 h[name]=trim(v)
481 end
482 response=dispatch!(r, method, target, h)
483 origin=get(h, "origin", nothing)
484 if origin in r.origins
485 response.headers["Access-Control-Allow-Origin"]=origin
486 response.status==204 &&
487 haskey(h, "access-control-request-method") &&
488 (response.headers["Access-Control-Allow-Methods"]=response.headers["Allow"])
489 end
490 response, showbody
491end
492function urlfor(r, name, args)
493 idx=findroute(r, name)
494 idx===nothing && fail("unknown route $name")
495 route=r.routes[idx]
496 names=Set(p.name for p in route.pieces if p.kind!=:literal)
497 values=Dict{String,String}()
498 query=Tuple{String,String}[]
499 for arg in args
500 k, equal, v=partition(arg, "=")
501 equal && !isempty(k) || fail("bad argument $arg")
502 if k in names
503 haskey(values, k) && fail("duplicate parameter $k")
504 values[k]=v
505 else
506 push!(query, (k, v))
507 end
508 end
509 segments=String[]
510 for p in route.pieces
511 if p.kind==:literal
512 push!(segments, p.name)
513 continue
514 end
515 v=get(values, p.name, nothing)
516 v===nothing && p.kind!=:wildcard && fail("missing parameter $(p.name)")
517 v!==nothing && !accepts(p, v) && fail("bad value for $(p.name)")
518 if p.kind!=:wildcard
519 push!(segments, encode(v))
520 elseif v!==nothing && !isempty(v)
521 append!(segments, encode.(split(v, '/')))
522 end
523 end
524 "/"*join(segments, '/')*(isempty(query) ? "" : "?"*encoded(query))
525end
526function main()
527 lines=readlines()
528 r=Router()
529 at=1
530 while lines[at]!="REQUESTS"
531 try
532 configure!(r, lines[at])
533 catch e
534 e isa InputError || rethrow()
535 r.errors+=1
536 println("CONFIG ERROR $at: $(e.message)")
537 end
538 at+=1
539 end
540 finish!(r)
541 at+=1
542 while lines[at]!="END"
543 if startswith(lines[at], "URLFOR ")
544 fields=String.(split(lines[at], ' '))
545 name=fields[2]
546 try
547 url=urlfor(r, name, fields[3:end])
548 r.urlok+=1
549 println("URLFOR $name $url")
550 catch e
551 e isa InputError || rethrow()
552 r.urlerror+=1
553 println("URLFOR ERROR $(e.message)")
554 end
555 at+=1
556 else
557 line=lines[at]
558 at+=1
559 headers=String[]
560 while !isempty(lines[at])
561 push!(headers, lines[at])
562 at+=1
563 end
564 at+=1
565 response, showbody=respond!(r, line, headers)
566 r.requests+=1
567 status=response.status
568 r.statuses[status]=get(r.statuses, status, 0)+1
569 println("REQUEST $(r.requests)\nHTTP/1.1 $status $(REASONS[status])")
570 if response.body!==nothing
571 response.headers["Content-Type"]=response.type
572 response.headers["Content-Length"]=string(length(response.body))
573 end
574 for name in HEADERORDER
575 haskey(response.headers, name) && println("$name: $(response.headers[name])")
576 end
577 response.body!==nothing && showbody && println(response.body)
578 end
579 end
580 println("SUMMARY requests=$(r.requests) urlfor_ok=$(r.urlok) urlfor_error=$(r.urlerror)")
581 for status in sort!(collect(keys(r.statuses)))
582 println("STATUS $status $(r.statuses[status])")
583 end
584 for route in r.routes
585 println("ROUTE $(route.name) $(route.selections)")
586 end
587 for name in sort!(collect(keys(r.clients)))
588 c=r.clients[name]
589 println("CLIENT $name allowed=$(c.allowed) limited=$(c.limited)")
590 end
591end
