4fn name(s: &str) -> bool {
5 (1..=20).contains(&s.len())
6 && s.as_bytes()[0].is_ascii_lowercase()
7 && s.bytes()
8 .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == b'_')
9}
10fn role(s: &str) -> bool {
11 (1..=16).contains(&s.len()) && s.bytes().all(|c| c.is_ascii_lowercase())
12}
13fn unreserved(c: u8) -> bool {
14 c.is_ascii_alphanumeric() || b"-._~".contains(&c)
15}
16fn segment(s: &str) -> bool {
17 !s.is_empty() && s != "." && s != ".." && s.bytes().all(unreserved)
18}
19fn canonical(s: &str) -> bool {
20 s == "/" || (s.starts_with('/') && s[1..].split('/').all(segment))
21}
22fn media_part(s: &str) -> bool {
23 !s.is_empty()
24 && s.bytes()
25 .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || b".+-".contains(&c))
26}
27fn media(s: &str) -> bool {
28 s.split_once('/')
29 .is_some_and(|(a, b)| media_part(a) && media_part(b))
30}
31fn int_parameter(s: &str) -> bool {
32 (1..=9).contains(&s.len()) && s.bytes().all(|c| c.is_ascii_digit())
33}
34fn slug(s: &str) -> bool {
35 !s.is_empty()
36 && s.bytes()
37 .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == b'-')
38}
39fn decimal(s: &str, max: i64) -> Option<i64> {
40 if s.is_empty() || (s.len() > 1 && s.starts_with('0')) || !s.bytes().all(|c| c.is_ascii_digit())
41 {
42 None
43 } else {
44 s.parse::<i64>().ok().filter(|&n| n >= 1 && n <= max)
45 }
46}
47fn hex(c: u8) -> Option<u8> {
48 match c {
49 b'0'..=b'9' => Some(c - b'0'),
50 b'a'..=b'f' => Some(c - b'a' + 10),
51 b'A'..=b'F' => Some(c - b'A' + 10),
52 _ => None,
53 }
54}
55fn decode(s: &str) -> String {
56 let b = s.as_bytes();
57 let mut at = 0;
58 let mut out = String::new();
59 while at < b.len() {
60 if b[at] == b'%' {
61 out.push((hex(b[at + 1]).unwrap() * 16 + hex(b[at + 2]).unwrap()) as char);
62 at += 3;
63 } else {
64 out.push(b[at] as char);
65 at += 1;
66 }
67 }
68 out
69}
70fn encode(s: &str) -> String {
71 s.bytes()
72 .map(|c| {
73 if unreserved(c) {
74 (c as char).to_string()
75 } else {
76 format!("%{c:02X}")
77 }
78 })
79 .collect()
80}
81fn valid_escapes(s: &str) -> bool {
82 let b = s.as_bytes();
83 let mut at = 0;
84 while at < b.len() {
85 if b[at] == b'%' {
86 if at + 2 >= b.len() {
87 return false;
88 }
89 let (Some(a), Some(c)) = (hex(b[at + 1]), hex(b[at + 2])) else {
90 return false;
91 };
92 if !(32..=126).contains(&(a * 16 + c)) {
93 return false;
94 }
95 at += 3;
96 } else {
97 at += 1;
98 }
99 }
100 true
101}
102fn normalize(path: &str) -> String {
103 let b = path.as_bytes();
104 let mut text = String::new();
105 let mut at = 0;
106 while at < b.len() {
107 if b[at] == b'%' {
108 let c = hex(b[at + 1]).unwrap() * 16 + hex(b[at + 2]).unwrap();
109 if unreserved(c) {
110 text.push(c as char);
111 } else {
112 text += &format!("%{c:02X}");
113 }
114 at += 3;
115 } else {
116 text.push(b[at] as char);
117 at += 1;
118 }
119 }
120 let mut parts = Vec::new();
121 for part in text[1..].split('/') {
122 match part {
123 "" | "." => {}
124 ".." => {
125 parts.pop();
126 }
127 _ => parts.push(part),
128 }
129 }
130 format!("/{}", parts.join("/"))
131}
132#[derive(Clone)]
133struct Piece {
134 kind: char,
135 value: String,
136}
137fn pattern(s: &str) -> Option<Vec<Piece>> {
138 if s == "/" {
139 return Some(Vec::new());
140 }
141 if !s.starts_with('/') {
142 return None;
143 }
144 let parts: Vec<_> = s[1..].split('/').collect();
145 let mut result = Vec::new();
146 let mut names = BTreeSet::new();
147 for (i, p) in parts.iter().enumerate() {
148 let piece = if segment(p) {
149 Piece {
150 kind: 'l',
151 value: (*p).into(),
152 }
153 } else if p.starts_with('{') && p.ends_with('}') {
154 let inner = &p[1..p.len() - 1];
155 let (kind, value) = if let Some(value) = inner.strip_prefix('*') {
156 if i + 1 != parts.len() {
157 return None;
158 }
159 ('w', value)
160 } else if let Some((value, t)) = inner.split_once(':') {
161 (
162 match t {
163 "int" => 'i',
164 "slug" => 's',
165 _ => return None,
166 },
167 value,
168 )
169 } else {
170 ('p', inner)
171 };
172 if !name(value) || !names.insert(value.to_string()) {
173 return None;
174 }
175 Piece {
176 kind,
177 value: value.into(),
178 }
179 } else {
180 return None;
181 };
182 result.push(piece);
183 }
184 Some(result)
185}
186#[derive(Clone)]
187struct Route {
188 name: String,
189 methods: BTreeSet<String>,
190 pattern: String,
191 pieces: Vec<Piece>,
192 role: String,
193 types: Option<Vec<String>>,
194 count: usize,
195}
196impl Route {
197 fn matches(&self, parts: &[&str]) -> bool {
198 let wildcard = self.pieces.last().is_some_and(|p| p.kind == 'w');
199 let m = self.pieces.len() - usize::from(wildcard);
200 (if wildcard {
201 parts.len() >= m
202 } else {
203 parts.len() == m
204 }) && self.pieces[..m]
205 .iter()
206 .zip(parts)
207 .all(|(p, x)| match p.kind {
208 'l' => p.value == *x,
209 'i' => int_parameter(x),
210 's' => slug(x),
211 _ => true,
212 })
213 }
214 fn parameters(&self, parts: &[&str]) -> Vec<(String, String)> {
215 let mut result = Vec::new();
216 for (i, p) in self.pieces.iter().enumerate() {
217 let value = match p.kind {
218 'l' => continue,
219 'w' => parts[i..]
220 .iter()
221 .map(|s| decode(s))
222 .collect::<Vec<_>>()
223 .join("/"),
224 'i' => parts[i].parse::<u32>().unwrap().to_string(),
225 's' => parts[i].to_string(),
226 _ => decode(parts[i]),
227 };
228 result.push((p.value.clone(), value));
229 }
230 result.sort_by(|a, b| a.0.cmp(&b.0));
231 result
232 }
233}

293 fn config(&mut self, w: &[&str]) -> Result<(), String> {
294 let command = w[0];
295 let args = w.len() - 1;
296 if !["ROUTE", "TOKEN", "REDIRECT", "ORIGIN", "LIMIT", "TYPES"].contains(&command) {
297 return Err(format!("unknown command {command}"));
298 }
299 let valid = match command {
300 "ROUTE" => args == 3 || args == 5,
301 "TOKEN" | "LIMIT" => args == 2,
302 "REDIRECT" => args == 3,
303 "ORIGIN" => args == 1,
304 _ => args >= 2,
305 };
306 if !valid {
307 return Err("wrong number of arguments".into());
308 }
309 match command {
310 "ROUTE" => {
311 if !name(w[1]) {
312 return Err(format!("bad route name {}", w[1]));
313 }
314 if self.routes.iter().any(|r| r.name == w[1]) {
315 return Err(format!("duplicate route {}", w[1]));
316 }
317 let methods: Vec<_> = w[2].split(',').collect();
318 let set: BTreeSet<_> = methods.iter().map(|m| m.to_string()).collect();
319 if methods
320 .iter()
321 .any(|m| !["GET", "POST", "PUT", "PATCH", "DELETE"].contains(m))
322 || methods.len() != set.len()
323 {
324 return Err(format!("bad methods {}", w[2]));
325 }
326 let pieces = pattern(w[3]).ok_or_else(|| format!("bad pattern {}", w[3]))?;
327 if args == 5 && w[4] != "ROLE" {
328 return Err(format!("unknown option {}", w[4]));
329 }
330 if args == 5 && !role(w[5]) {
331 return Err(format!("bad role {}", w[5]));
332 }
333 self.routes.push(Route {
334 name: w[1].into(),
335 methods: set,
336 pattern: w[3].into(),
337 pieces,
338 role: if args == 5 {
339 w[5].into()
340 } else {
341 "none".into()
342 },
343 types: None,
344 count: 0,
345 });
346 }
347 "TOKEN" => {
348 if !(1..=32).contains(&w[1].len())
349 || !w[1]
350 .bytes()
351 .all(|c| c.is_ascii_alphanumeric() || b"._-".contains(&c))
352 {
353 return Err(format!("bad token {}", w[1]));
354 }
355 if self.tokens.contains_key(w[1]) {
356 return Err(format!("duplicate token {}", w[1]));
357 }
358 if !role(w[2]) {
359 return Err(format!("bad role {}", w[2]));
360 }
361 self.tokens.insert(w[1].into(), w[2].into());
362 }
363 "REDIRECT" => {
364 let rule = |s: &str| {
365 canonical(s)
366 || s.strip_suffix("/*")
367 .is_some_and(|base| base != "/" && canonical(base))
368 };
369 if !rule(w[1]) {
370 return Err(format!("bad path {}", w[1]));
371 }
372 if !rule(w[2]) {
373 return Err(format!("bad path {}", w[2]));
374 }
375 let p = w[1].ends_with("/*");
376 if p != w[2].ends_with("/*") {
377 return Err("wildcard mismatch".into());
378 }
379 if !["301", "302", "307", "308"].contains(&w[3]) {
380 return Err(format!("bad redirect code {}", w[3]));
381 }
382 if self
383 .redirects
384 .iter()
385 .any(|r| r.prefix == p && r.from == w[1].strip_suffix("/*").unwrap_or(w[1]))
386 {
387 return Err(format!("duplicate redirect {}", w[1]));
388 }
389 self.redirects.push(Redirect {
390 from: w[1].strip_suffix("/*").unwrap_or(w[1]).into(),
391 to: w[2].strip_suffix("/*").unwrap_or(w[2]).into(),
392 prefix: p,
393 code: w[3].parse().unwrap(),
394 });
395 }
396 "ORIGIN" => {
397 let suffix = w[1]
398 .strip_prefix("http://")
399 .or_else(|| w[1].strip_prefix("https://"));
400 if !suffix.is_some_and(|s| {
401 !s.is_empty()
402 && s.bytes().all(|c| {
403 c.is_ascii_lowercase() || c.is_ascii_digit() || b".-:".contains(&c)
404 })
405 }) {
406 return Err(format!("bad origin {}", w[1]));
407 }
408 if self.origins.contains(w[1]) {
409 return Err(format!("duplicate origin {}", w[1]));
410 }
411 self.origins.insert(w[1].into());
412 }
413 "LIMIT" => {
414 let (Some(n), Some(s)) = (decimal(w[1], 100), decimal(w[2], 3600)) else {
415 return Err("bad limit".into());
416 };
417 if self.limit.is_some() {
418 return Err("duplicate limit".into());
419 }
420 self.limit = Some((n, s));
421 }
422 _ => {
423 let i = self
424 .routes
425 .iter()
426 .position(|r| r.name == w[1])
427 .ok_or_else(|| format!("unknown route {}", w[1]))?;
428 if self.routes[i].types.is_some() {
429 return Err(format!("duplicate types {}", w[1]));
430 }
431 let mut types = BTreeSet::new();
432 for t in &w[2..] {
433 if !media(t) {
434 return Err(format!("bad media type {t}"));
435 }
436 if !types.insert(*t) {
437 return Err(format!("duplicate media type {t}"));
438 }
439 }
440 self.routes[i].types = Some(w[2..].iter().map(|s| s.to_string()).collect());
441 }
442 }
443 Ok(())
444 }
445 fn table(&mut self) {
446 println!(
447 "CONFIG routes={} tokens={} redirects={} origins={} limit={} errors={}",
448 self.routes.len(),
449 self.tokens.len(),
450 self.redirects.len(),
451 self.origins.len(),
452 self.limit
453 .map_or("none".into(), |(n, s)| format!("{n}/{s}")),
454 self.config_errors
455 );
456 for route in &mut self.routes {
457 route
458 .types
459 .get_or_insert_with(|| vec!["application/json".into()]);
460 }
461 self.priority = (0..self.routes.len()).collect();
462 let routes = &self.routes;
463 self.priority.sort_by_key(|&i| {
464 let r = &routes[i];
465 (
466 std::cmp::Reverse(r.pieces.iter().filter(|p| p.kind == 'l').count()),
467 std::cmp::Reverse(
468 r.pieces

470 .filter(|p| p.kind == 'i' || p.kind == 's')
471 .count(),
472 ),
473 r.pieces.iter().any(|p| p.kind == 'w'),
474 i,
475 )
476 });
477 for (rank, &i) in self.priority.iter().enumerate() {
478 let r = &self.routes[i];
479 let mut methods = r.methods.clone();
480 if methods.contains("GET") {
481 methods.insert("HEAD".into());
482 }
483 println!(

485 rank + 1,
486 r.name,
487 methods.into_iter().collect::<Vec<_>>().join(","),
488 r.pattern,
489 if r.role == "none" { "-" } else { &r.role },
490 r.types.as_ref().unwrap().join(",")
491 );
492 }

494 fn request(&mut self, line: &str, raw: &[&str]) -> Response {
495 let w: Vec<_> = line.split(' ').collect();
496 if w.len() != 3
497 || !(1..=16).contains(&w[0].len())
498 || !w[0].bytes().all(|c| c.is_ascii_uppercase())
499 || !w[1].starts_with('/')
500 || w[1].contains('#')
501 || w[2] != "HTTP/1.1"
502 {
503 return Response::new(400, "bad request: request line");
504 }
505 let head = w[0] == "HEAD";
506 if !valid_escapes(w[1]) {
507 let mut r = Response::new(400, "bad request: percent escape");
508 r.head = head;
509 return r;
510 }
511 let mut headers = BTreeMap::new();
512 for (i, row) in raw.iter().enumerate() {
513 let Some((name, value)) = row.split_once(':') else {
514 let mut r = Response::new(400, format!("bad request: header line {}", i + 1));
515 r.head = head;
516 return r;
517 };
518 if name.is_empty() || !name.bytes().all(|c| c.is_ascii_alphanumeric() || c == b'-') {
519 let mut r = Response::new(400, format!("bad request: header line {}", i + 1));
520 r.head = head;
521 return r;
522 }
523 let name = name.to_ascii_lowercase();
524 if headers.contains_key(&name) {
525 let mut r = Response::new(400, format!("bad request: duplicate header {name}"));
526 r.head = head;
527 return r;
528 }
529 headers.insert(name, value.trim_matches(' ').to_string());
530 }
531 let mut response = self.valid_request(w[0], w[1], &headers);
532 response.head = head;
533 if let Some(origin) = headers.get("origin").filter(|s| self.origins.contains(*s)) {
534 response
535 .headers
536 .insert("Access-Control-Allow-Origin".into(), origin.clone());
537 if response.status == 204 && headers.contains_key("access-control-request-method") {
538 response.headers.insert(
539 "Access-Control-Allow-Methods".into(),
540 response.headers["Allow"].clone(),
541 );
542 }
543 }
544 response
545 }
546 fn valid_request(
547 &mut self,
548 method: &str,
549 target: &str,
550 h: &BTreeMap<String, String>,
551 ) -> Response {
552 if !["GET", "HEAD", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"].contains(&method) {
553 return Response::new(501, format!("not implemented: {method}"));
554 }
555 if h.get("host").is_none_or(|s| s.is_empty()) {
556 return Response::new(400, "bad request: missing host");
557 }
558 let mut effective = method;
559 if let Some(value) = h.get("x-http-method-override") {
560 if method == "POST" && ["PUT", "PATCH", "DELETE"].contains(&value.as_str()) {
561 effective = value;
562 } else {
563 return Response::new(400, "bad request: method override");
564 }
565 }
566 if let Some(client) = h.get("x-client").filter(|s| !s.is_empty()) {
567 let time = h["x-time"].parse::<i64>().unwrap();
568 let (n, s) = self.limit.unwrap_or((0, 0));
569 let c = self.clients.entry(client.clone()).or_insert(Client {
570 level: n * s,
571 last: time,
572 allowed: 0,
573 limited: 0,
574 });
575 if self.limit.is_some() {
576 c.level = (n * s).min(c.level + (time - c.last) * n);
577 c.last = time;
578 if c.level < s {
579 c.limited += 1;
580 return Response::new(429, format!("too many requests: client {client}"))
581 .header("Retry-After", ((s - c.level + n - 1) / n).to_string());
582 }
583 c.level -= s;
584 }
585 c.allowed += 1;
586 }
587 let (path, query) = target
588 .split_once('?')
589 .map_or((target, None), |(p, q)| (p, Some(q)));
590 let suffix = query.map_or(String::new(), |q| format!("?{q}"));
591 let normalized = normalize(path);
592 if normalized != path {
593 let dest = normalized + &suffix;
594 return Response::new(308, format!("redirect to {dest}")).header("Location", dest);
595 }
596 for rule in &self.redirects {
597 if path == rule.from || (rule.prefix && path.starts_with(&(rule.from.clone() + "/"))) {
598 let dest = rule.to.clone() + &path[rule.from.len()..] + &suffix;
599 return Response::new(rule.code, format!("redirect to {dest}"))
600 .header("Location", dest);
601 }
602 }
603 let parts: Vec<_> = if path == "/" {
604 Vec::new()
605 } else {
606 path[1..].split('/').collect()
607 };
608 let matching: Vec<_> = self
609 .priority
610 .iter()
611 .copied()
612 .filter(|&i| self.routes[i].matches(&parts))
613 .collect();
614 if matching.is_empty() {
615 return Response::new(404, format!("not found: {path}"));
616 }
617 let mut allow = BTreeSet::from(["OPTIONS".to_string()]);
618 for &i in &matching {
619 allow.extend(self.routes[i].methods.iter().cloned());
620 }
621 if allow.contains("GET") {
622 allow.insert("HEAD".into());
623 }
624 let allow = allow.into_iter().collect::<Vec<_>>().join(", ");
625 if effective == "OPTIONS" {
626 return Response::new(204, "").header("Allow", allow);
627 }
628 let lookup = if effective == "HEAD" {
629 "GET"
630 } else {
631 effective
632 };
633 let Some(i) = matching
634 .into_iter()
635 .find(|&i| self.routes[i].methods.contains(lookup))
636 else {
637 return Response::new(405, format!("method not allowed: {effective}"))
638 .header("Allow", allow);
639 };
640 self.routes[i].count += 1;
641 let route = self.routes[i].clone();
642 if route.role != "none" {
643 let Some(auth) = h.get("authorization") else {
644 return Response::new(401, "unauthorized: missing token")
645 .header("WWW-Authenticate", "Bearer realm=\"api\"");
646 };
647 let (scheme, token) = auth.split_once(' ').unwrap_or((auth, ""));
648 let token_role = self.tokens.get(token);
649 if !scheme.eq_ignore_ascii_case("bearer") || token_role.is_none() {
650 return Response::new(401, "unauthorized: invalid token").header(
651 "WWW-Authenticate",
652 "Bearer realm=\"api\", error=\"invalid_token\"",
653 );
654 }
655 if token_role.unwrap() != &route.role && token_role.unwrap() != "admin" {
656 return Response::new(403, format!("forbidden: requires role {}", route.role));
657 }
658 }
659 let types = route.types.as_ref().unwrap();
660 let vary = |mut r: Response| {
661 if types.len() >= 2 {
662 r.headers.insert("Vary".into(), "Accept".into());
663 }
664 r
665 };
666 let selected = if let Some(accept) = h.get("accept") {
667 negotiate(types, accept)
668 } else {
669 Some(types[0].clone())
670 };
671 let Some(selected) = selected else {
672 return vary(Response::new(
673 406,
674 format!("not acceptable: {}", types.join(", ")),
675 ));
676 };
677 let params = route.parameters(&parts);
678 let queries = query.map_or(Vec::new(), |q| {
679 q.split('&')
680 .filter(|s| !s.is_empty())
681 .map(|s| {
682 let (a, b) = s.split_once('=').unwrap_or((s, ""));
683 (decode(&a.replace('+', " ")), decode(&b.replace('+', " ")))
684 })
685 .collect()
686 });
687 let mut cookies = BTreeMap::new();
688 if let Some(cookie) = h.get("cookie") {
689 for p in cookie.split(';').map(|s| s.trim_matches(' ')) {
690 if let Some((a, b)) = p.split_once('=') {
691 if !a.is_empty() {
692 cookies.entry(a.to_string()).or_insert(b.to_string());
693 }
694 }
695 }
696 }
697 let cookies = cookies.into_iter().collect::<Vec<_>>();
698 let body = body(&route.name, &selected, &params, &queries, &cookies);
699 let etag = etag(&body);
700 if h.get("if-none-match").is_some_and(|s| {
701 s.split(',').any(|p| {
702 let p = p.trim_matches(' ');
703 let p = p.strip_prefix("W/").unwrap_or(p);
704 p == "*" || p == etag
705 })
706 }) {
707 return if lookup == "GET" {
708 vary(Response::new(304, "").header("ETag", etag))
709 } else {
710 vary(Response::new(412, format!("precondition failed: {etag}")))
711 };
712 }
713 let mut response = vary(Response::new(200, body).header("ETag", etag));
714 response.content_type = selected;
715 response
716 }
717 fn urlfor(&self, w: &[&str]) -> Result<String, String> {
718 let route = self
719 .routes
720 .iter()
721 .find(|r| r.name == w[1])
722 .ok_or_else(|| format!("unknown route {}", w[1]))?;
723 let names: BTreeSet<_> = route
724 .pieces
725 .iter()
726 .filter(|p| p.kind != 'l')
727 .map(|p| p.value.as_str())
728 .collect();
729 let mut args = Vec::new();
730 let mut params = BTreeMap::new();
731 for arg in &w[2..] {
732 let Some((name, value)) = arg.split_once('=') else {
733 return Err(format!("bad argument {arg}"));
734 };
735 if name.is_empty() {
736 return Err(format!("bad argument {arg}"));
737 }
738 if names.contains(name) && params.insert(name, value).is_some() {
739 return Err(format!("duplicate parameter {name}"));
740 }
741 args.push((name, value));
742 }
743 for p in route.pieces.iter().filter(|p| p.kind != 'l') {
744 let value = params.get(p.value.as_str()).copied();
745 if p.kind != 'w' && value.is_none() {
746 return Err(format!("missing parameter {}", p.value));
747 }
748 let valid = match p.kind {
749 'p' => value.is_some_and(|v| !v.is_empty() && v != "." && v != ".."),
750 'i' => value.is_some_and(int_parameter),
751 's' => value.is_some_and(slug),
752 _ => value.is_none_or(|v| {
753 v.is_empty() || v.split('/').all(|p| !p.is_empty() && p != "." && p != "..")
754 }),
755 };
756 if !valid {
757 return Err(format!("bad value for {}", p.value));
758 }
759 }
760 let mut pieces = Vec::new();
761 for p in &route.pieces {
762 if p.kind == 'l' {
763 pieces.push(p.value.clone());
764 } else {
765 let v = params.get(p.value.as_str()).copied().unwrap_or("");
766 if p.kind == 'w' {
767 if !v.is_empty() {
768 pieces.extend(v.split('/').map(encode));
769 }
770 } else {
771 pieces.push(encode(v));
772 }
773 }
774 }
775 let mut url = format!("/{}", pieces.join("/"));
776 let queries: Vec<_> = args
777 .into_iter()
778 .filter(|(n, _)| !names.contains(n))
779 .map(|(n, v)| format!("{}={}", encode(n), encode(v)))
780 .collect();
781 if !queries.is_empty() {
782 url.push('?');
783 url += &queries.join("&");
784 }
785 Ok(format!("URLFOR {} {url}", route.name))
786 }
787}
788fn negotiate(types: &[String], accept: &str) -> Option<String> {
789 let mut elements = Vec::new();
790 for element in accept.split(',') {
791 let parts: Vec<_> = element.split(';').map(|s| s.trim_matches(' ')).collect();
792 let range = parts[0].to_ascii_lowercase();
793 let specificity = if range == "*/*" {
794 1
795 } else if range.strip_suffix("/*").is_some_and(media_part) {
796 2
797 } else if media(&range) {
798 3
799 } else {
800 continue;
801 };
802 let q = parts[1..].iter().find_map(|p| {
803 p.split_once('=')
804 .filter(|(n, _)| n.trim_matches(' ').eq_ignore_ascii_case("q"))
805 .map(|(_, v)| v.trim_matches(' '))
806 });
807 let weight = if let Some(q) = q {
808 let (whole, fraction) = q.split_once('.').map_or((q, None), |(w, f)| (w, Some(f)));
809 if !["0", "1"].contains(&whole)
810 || fraction.is_some_and(|f| {
811 f.len() > 3
812 || !f.bytes().all(|c| c.is_ascii_digit())
813 || (whole == "1" && f.bytes().any(|c| c != b'0'))
814 })
815 {
816 continue;
817 }
818 if whole == "1" {
819 1000
820 } else {
821 fraction.map_or(0, |f| format!("{f:0<3}").parse::<i32>().unwrap())
822 }
823 } else {
824 1000
825 };
826 elements.push((range, specificity, weight));
827 }
828 let mut best = 0;
829 let mut selected = None;
830 for t in types {
831 let mut specificity = 0;
832 let mut weight = 0;
833 for (range, s, w) in &elements {
834 if *s > specificity
835 && (range == "*/*"
836 || range == t
837 || range
838 .strip_suffix("/*")
839 .is_some_and(|main| t.split_once('/').unwrap().0 == main))
840 {
841 specificity = *s;
842 weight = *w;
843 }
844 }
845 if weight > best {
846 best = weight;
847 selected = Some(t.clone());
848 }
849 }
850 selected
851}
852fn json(s: &str) -> String {
853 format!("\"{}\"", s.replace('\\', "\\\\").replace('"', "\\\""))
854}
855fn html(s: &str) -> String {
856 s.chars()
857 .map(|c| match c {
858 '&' => "&amp;".into(),
859 '<' => "&lt;".into(),
860 '>' => "&gt;".into(),
861 '"' => "&quot;".into(),
862 _ => c.to_string(),
863 })
864 .collect()
865}
866fn body(
867 route: &str,
868 t: &str,
869 params: &[(String, String)],
870 query: &[(String, String)],
871 cookies: &[(String, String)],
872) -> String {
873 if t == "application/json" {
874 let object = |pairs: &[(String, String)]| {
875 pairs
876 .iter()
877 .map(|(a, b)| format!("{}:{}", json(a), json(b)))
878 .collect::<Vec<_>>()
879 .join(",")
880 };
881 format!(
882 "{{\"route\":{},\"params\":{{{}}},\"query\":[{}],\"cookies\":{{{}}}}}",
883 json(route),
884 object(params),
885 query
886 .iter()
887 .map(|(a, b)| format!("[{},{}]", json(a), json(b)))
888 .collect::<Vec<_>>()
889 .join(","),
890 object(cookies)
891 )
892 } else if t == "text/html" {
893 let mut text = format!("<dl><dt>route</dt><dd>{}</dd>", html(route));
894 for (kind, list) in [("param", params), ("query", query), ("cookie", cookies)] {
895 for (a, b) in list {
896 text += &format!("<dt>{kind} {}</dt><dd>{}</dd>", html(a), html(b));
897 }
898 }
899 text + "</dl>"
900 } else {
901 let list = |pairs: &[(String, String)]| {
902 if pairs.is_empty() {
903 "-".into()
904 } else {
905 pairs
906 .iter()
907 .map(|(a, b)| format!("{}={}", encode(a), encode(b)))
908 .collect::<Vec<_>>()
909 .join("&")
910 }
911 };
912 format!(
913 "route={route} params={} query={} cookies={}",
914 list(params),
915 list(query),
916 list(cookies)
917 )
918 }
919}
920fn etag(body: &str) -> String {
921 let mut a = 1u32;
922 let mut b = 0u32;
923 for c in body.bytes() {
924 a = (a + c as u32) % 65521;
925 b = (b + a) % 65521;
926 }
927 format!("\"{b:04x}{a:04x}\"")
928}
929fn reason(status: u16) -> &'static str {
930 match status {
931 200 => "OK",
932 204 => "No Content",
933 301 => "Moved Permanently",
934 302 => "Found",
935 304 => "Not Modified",
936 307 => "Temporary Redirect",
937 308 => "Permanent Redirect",
938 400 => "Bad Request",
939 401 => "Unauthorized",
940 403 => "Forbidden",
941 404 => "Not Found",
942 405 => "Method Not Allowed",
943 406 => "Not Acceptable",
944 412 => "Precondition Failed",
945 429 => "Too Many Requests",
946 _ => "Not Implemented",
947 }
948}
949fn output(number: usize, mut r: Response) {
950 println!(
951 "REQUEST {number}\nHTTP/1.1 {} {}",
952 r.status,
953 reason(r.status)
954 );
955 if r.status != 204 && r.status != 304 {
956 r.headers.insert("Content-Type".into(), r.content_type);
957 r.headers
958 .insert("Content-Length".into(), r.body.len().to_string());
959 }
960 for name in [
961 "Location",
962 "Allow",
963 "Retry-After",
964 "WWW-Authenticate",
965 "Access-Control-Allow-Origin",
966 "Access-Control-Allow-Methods",
967 "Vary",
968 "ETag",
969 "Content-Type",
970 "Content-Length",
971 ] {
972 if let Some(value) = r.headers.get(name) {
973 println!("{name}: {value}");
974 }
975 }
976 if r.status != 204 && r.status != 304 && !r.head {
977 println!("{}", r.body);
978 }
979}
980fn main() {
981 let mut input = String::new();
982 io::stdin().read_to_string(&mut input).unwrap();
983 let lines: Vec<_> = input.lines().collect();
984 let mut at = 0;
985 let mut router = Router::new();
986 while lines[at] != "REQUESTS" {
987 if let Err(error) = router.config(&lines[at].split(' ').collect::<Vec<_>>()) {
988 router.config_errors += 1;
989 println!("CONFIG ERROR {}: {error}", at + 1);
990 }
991 at += 1;
992 }
993 at += 1;
994 router.table();
995 let mut requests = 0;
996 let mut url_ok = 0;
997 let mut url_error = 0;
998 while lines[at] != "END" {
999 let line = lines[at];
1000 at += 1;
1001 if line.starts_with("URLFOR ") {
1002 match router.urlfor(&line.split(' ').collect::<Vec<_>>()) {
1003 Ok(text) => {
1004 url_ok += 1;
1005 println!("{text}");
1006 }
1007 Err(error) => {
1008 url_error += 1;
1009 println!("URLFOR ERROR {error}");
1010 }
1011 }
1012 } else {
1013 requests += 1;
1014 let mut headers = Vec::new();
1015 while !lines[at].is_empty() {
1016 headers.push(lines[at]);
1017 at += 1;
1018 }
1019 at += 1;
1020 let response = router.request(line, &headers);
1021 *router.statuses.entry(response.status).or_default() += 1;
1022 output(requests, response);
1023 }
1024 }
1025 println!("SUMMARY requests={requests} urlfor_ok={url_ok} urlfor_error={url_error}");
1026 for (status, count) in router.statuses {
1027 println!("STATUS {status} {count}");
1028 }
1029 for route in router.routes {
1030 println!("ROUTE {} {}", route.name, route.count);
1031 }
1032 for (name, c) in router.clients {
1033 println!("CLIENT {name} allowed={} limited={}", c.allowed, c.limited);
1034 }
1035}
