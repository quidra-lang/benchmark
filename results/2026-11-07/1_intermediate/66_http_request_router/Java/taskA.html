102 static List<Piece> pattern(String text) {
103 if (!text.startsWith("/")) return null;
104 if (text.equals("/")) return new ArrayList<>();
105 String[] fields = text.substring(1).split("/", -1);
106 List<Piece> pieces = new ArrayList<>();
107 Set<String> names = new HashSet<>();
108 for (int i = 0; i < fields.length; i++) {
109 String field = fields[i];
110 Piece piece;
111 if (segment(field)) piece = new Piece('L', field);
112 else {
113 Matcher match = Pattern.compile("\\{(\\*?)(" + NAME + ")(?::(int|slug))?\\}").matcher(field);
114 if (!match.matches() || !match.group(1).isEmpty() && match.group(3) != null) return null;
115 char kind = !match.group(1).isEmpty() ? 'W' : match.group(3) == null ? 'U' : match.group(3).equals("int") ? 'I' : 'S';
116 if (kind == 'W' && i != fields.length - 1 || !names.add(match.group(2))) return null;
117 piece = new Piece(kind, match.group(2));
118 }
119 pieces.add(piece);
120 }
121 return pieces;
122 }

129 static void configure(String[] a) {
130 if (!Set.of("ROUTE", "TOKEN", "REDIRECT", "ORIGIN", "LIMIT", "TYPES").contains(a[0])) invalid("unknown command " + a[0]);
131 int n = a.length - 1;
132 boolean correct = switch (a[0]) {
133 case "ROUTE" -> n == 3 || n == 5;
134 case "TOKEN", "LIMIT" -> n == 2;
135 case "REDIRECT" -> n == 3;
136 case "ORIGIN" -> n == 1;
137 default -> n >= 2;
138 };
139 if (!correct) invalid("wrong number of arguments");
140 switch (a[0]) {
141 case "ROUTE": {
142 if (!a[1].matches(NAME)) invalid("bad route name " + a[1]);
143 if (find(a[1]) != null) invalid("duplicate route " + a[1]);
144 Set<String> methods = new HashSet<>();
145 for (String method : a[2].split(",", -1)) if (!ROUTE_METHODS.contains(method) || !methods.add(method)) invalid("bad methods " + a[2]);
146 List<Piece> pieces = pattern(a[3]);
147 if (pieces == null) invalid("bad pattern " + a[3]);
148 if (n == 5 && !a[4].equals("ROLE")) invalid("unknown option " + a[4]);
149 if (n == 5 && !a[5].matches("[a-z]{1,16}")) invalid("bad role " + a[5]);
150 Route route = new Route();
151 route.name = a[1];
152 route.methods = methods;
153 route.pattern = a[3];
154 route.pieces = pieces;
155 route.role = n == 5 ? a[5] : "none";
156 route.number = routes.size();
157 for (Piece p : pieces) {
158 if (p.kind() == 'L') route.literals++;
159 if (p.kind() == 'I' || p.kind() == 'S') route.typed++;
160 if (p.kind() == 'W') route.wildcard = true;
161 }
162 routes.add(route);
163 break;
164 }
165 case "TOKEN": if (!a[1].matches("[A-Za-z0-9._-]{1,32}")) invalid("bad token " + a[1]);
166 if (tokens.containsKey(a[1])) invalid("duplicate token " + a[1]);
167 if (!a[2].matches("[a-z]{1,16}")) invalid("bad role " + a[2]);
168 tokens.put(a[1], a[2]);
169 break;
170 case "REDIRECT": if (!rulePath(a[1])) invalid("bad path " + a[1]);
171 if (!rulePath(a[2])) invalid("bad path " + a[2]);
172 if (a[1].endsWith("/*") != a[2].endsWith("/*")) invalid("wildcard mismatch");
173 if (!Set.of("301", "302", "307", "308").contains(a[3])) invalid("bad redirect code " + a[3]);
174 for (Redirect redirect : redirects) if ((redirect.from() + (redirect.prefix() ? "/*" : "")).equals(a[1])) invalid("duplicate redirect " + a[1]);
175 redirects.add(new Redirect(base(a[1]), base(a[2]), a[1].endsWith("/*"), Integer.parseInt(a[3])));
176 break;
177 case "ORIGIN": if (!a[1].matches("https?://[a-z0-9.:-]+")) invalid("bad origin " + a[1]);
178 if (origins.contains(a[1])) invalid("duplicate origin " + a[1]);
179 origins.add(a[1]);
180 break;
181 case "LIMIT": if (!decimal(a[1], 100) || !decimal(a[2], 3600)) invalid("bad limit");
182 if (capacity != 0) invalid("duplicate limit");
183 capacity = Integer.parseInt(a[1]);
184 period = Integer.parseInt(a[2]);
185 break;
186 case "TYPES": Route route = find(a[1]);
187 if (route == null) invalid("unknown route " + a[1]);
188 if (route.types != null) invalid("duplicate types " + a[1]);
189 List<String> types = new ArrayList<>();
190 for (String type : Arrays.copyOfRange(a, 2, a.length)) {
191 if (!type.matches(MEDIA)) invalid("bad media type " + type);
192 if (types.contains(type)) invalid("duplicate media type " + type);
193 types.add(type);
194 }
195 route.types = types;
196 }

227 static String normalize(String path) {
228 StringBuilder text = new StringBuilder();
229 for (int i = 0; i < path.length(); i++) {
230 char c = path.charAt(i);
231 if (c == '%') {
232 char decoded = (char)Integer.parseInt(path.substring(i + 1, i + 3), 16);
233 text.append(unreserved(decoded) ? "" + decoded : "%" + path.substring(i + 1, i + 3).toUpperCase(Locale.ROOT));
234 i += 2;
235 } else text.append(c);
236 }
237 List<String> parts = new ArrayList<>();
238 for (String part : text.substring(1).split("/", -1)) if (part.equals("..")) {
239 if (!parts.isEmpty()) parts.removeLast();
240 } else if (!part.isEmpty() && !part.equals(".")) parts.add(part);
241 return "/" + String.join("/", parts);

244 static boolean matches(Route route, String[] segments) {
245 int count = route.pieces.size() - (route.wildcard ? 1 : 0);
246 if (route.wildcard ? segments.length < count : segments.length != count) return false;
247 for (int i = 0; i < count; i++) {
248 Piece p = route.pieces.get(i);
249 String s = segments[i];
250 if (p.kind() == 'L' && !p.text().equals(s) || p.kind() == 'I' && !s.matches("[0-9]{1,9}") || p.kind() == 'S' && !s.matches("[a-z0-9-]+")) return false;
251 }
252 return true;
253 }

255 static String negotiate(List<String> types, String accept) {
256 if (accept == null) return types.get(0);
257 String selected = null;
258 int bestWeight = 0;
259 for (String type : types) {
260 int specificity = 0, weight = 0;
261 for (String element : accept.split(",", -1)) {
262 String[] parts = element.split(";", -1);
263 String range = lower(strip(parts[0]));
264 int spec = range.equals("*/*") ? 1 : range.matches("[a-z0-9.+-]+/\\*") ? 2 : range.matches(MEDIA) ? 3 : 0;
265 if (spec == 0) continue;
266 String q = null;
267 for (int i = 1; i < parts.length; i++) {
268 int eq = parts[i].indexOf('=');
269 if (eq >= 0 && lower(strip(parts[i].substring(0, eq))).equals("q")) {
270 q = strip(parts[i].substring(eq + 1));
271 break;
272 }
273 }
274 if (q != null && !q.matches("(?:0(?:\\.[0-9]{0,3})?|1(?:\\.0{0,3})?)")) continue;
275 int w = 1000;
276 if (q != null) {
277 String digits = q.contains(".") ? q.substring(q.indexOf('.') + 1) : "";
278 w = (q.charAt(0) - '0') * 1000 + Integer.parseInt((digits + "000").substring(0, 3));
279 }
280 boolean fits = spec == 1 || spec == 2 && type.startsWith(range.substring(0, range.length() - 1)) || spec == 3 && type.equals(range);
281 if (fits && spec > specificity) {
282 specificity = spec;
283 weight = w;
284 }
285 }
286 if (weight > bestWeight) {
287 bestWeight = weight;
288 selected = type;
289 }
290 }
291 return selected;
292 }

343 static String etag(String body) {
344 int a = 1, b = 0;
345 for (char c : body.toCharArray()) {
346 a = (a + c) % 65521;
347 b = (b + a) % 65521;
348 }
349 return String.format(Locale.ROOT, "\"%04x%04x\"", b, a);
350 }

352 static Response process(Request request) {
353 String[] line = request.line.split(" ", -1);
354 if (line.length != 3 || !line[0].matches("[A-Z]{1,16}") || !line[1].startsWith("/") || line[1].contains("#") || !line[2].equals("HTTP/1.1")) return new Response(400, "bad request: request line");
355 String method = line[0], target = line[1];
356 request.head = method.equals("HEAD");
357 for (int i = 0; i < target.length(); i++) if (target.charAt(i) == '%') {
358 if (i + 2 >= target.length() || !target.substring(i + 1, i + 3).matches("[0-9a-fA-F]{2}")) return new Response(400, "bad request: percent escape");
359 int value = Integer.parseInt(target.substring(i + 1, i + 3), 16);
360 if (value < 32 || value > 126) return new Response(400, "bad request: percent escape");
361 i += 2;
362 }
363 Map<String, String> headers = request.headers;
364 int number = 0;
365 for (String header : request.headerLines) {
366 number++;
367 int colon = header.indexOf(':');
368 if (colon < 0 || !header.substring(0, colon).matches("[A-Za-z0-9-]+")) return new Response(400, "bad request: header line " + number);
369 String name = lower(header.substring(0, colon));
370 if (headers.containsKey(name)) return new Response(400, "bad request: duplicate header " + name);
371 headers.put(name, strip(header.substring(colon + 1)));
372 }
373 request.headersValid = true;
374 if (!REQUEST_METHODS.contains(method)) return new Response(501, "not implemented: " + method);
375 if (headers.getOrDefault("host", "").isEmpty()) return new Response(400, "bad request: missing host");
376 String effective = method;
377 if (headers.containsKey("x-http-method-override")) {
378 effective = headers.get("x-http-method-override");
379 if (!method.equals("POST") || !Set.of("PUT", "PATCH", "DELETE").contains(effective)) return new Response(400, "bad request: method override");
380 }
381 String clientName = headers.getOrDefault("x-client", "");
382 if (!clientName.isEmpty()) {
383 int time = Integer.parseInt(headers.get("x-time"));
384 Client client = clients.get(clientName);
385 if (client == null) {
386 client = new Client();
387 client.level = capacity * period;
388 client.last = time;
389 clients.put(clientName, client);
390 }
391 if (capacity != 0) {
392 client.level = Math.min(capacity * period, client.level + (time - client.last) * capacity);
393 client.last = time;
394 if (client.level < period) {
395 client.limited++;
396 return new Response(429, "too many requests: client " + clientName).header("Retry-After", "" + ((period - client.level + capacity - 1) / capacity));
397 }
398 client.level -= period;
399 }
400 client.allowed++;
401 }
402 int question = target.indexOf('?');
403 String path = question < 0 ? target : target.substring(0, question), query = question < 0 ? null : target.substring(question + 1), suffix = question < 0 ? "" : target.substring(question);
404 String normal = normalize(path);
405 if (!normal.equals(path)) {
406 String location = normal + suffix;
407 return new Response(308, "redirect to " + location).header("Location", location);
408 }
409 for (Redirect redirect : redirects) if (path.equals(redirect.from()) || redirect.prefix() && path.startsWith(redirect.from() + "/")) {
410 String location = redirect.to() + path.substring(redirect.from().length()) + suffix;
411 return new Response(redirect.code(), "redirect to " + location).header("Location", location);
412 }
413 String[] segments = path.equals("/") ? new String[0] : path.substring(1).split("/", -1);
414 List<Route> matching = priority.stream().filter(r -> matches(r, segments)).toList();
415 if (matching.isEmpty()) return new Response(404, "not found: " + path);
416 Set<String> allowed = new TreeSet<>();
417 allowed.add("OPTIONS");
418 for (Route route : matching) allowed.addAll(route.methods);
419 if (allowed.contains("GET")) allowed.add("HEAD");
420 String allow = String.join(", ", allowed);
421 if (effective.equals("OPTIONS")) return new Response(204, "").header("Allow", allow);
422 String lookup = effective.equals("HEAD") ? "GET" : effective;
423 Route route = null;
424 for (Route candidate : matching) if (candidate.methods.contains(lookup)) {
425 route = candidate;
426 break;
427 }
428 if (route == null) return new Response(405, "method not allowed: " + effective).header("Allow", allow);
429 route.selections++;
430 if (!route.role.equals("none")) {
431 if (!headers.containsKey("authorization")) return new Response(401, "unauthorized: missing token").header("WWW-Authenticate", "Bearer realm=\"api\"");
432 String auth = headers.get("authorization");
433 int space = auth.indexOf(' ');
434 String scheme = space < 0 ? auth : auth.substring(0, space), token = space < 0 ? "" : auth.substring(space + 1);
435 if (!lower(scheme).equals("bearer") || !tokens.containsKey(token)) return new Response(401, "unauthorized: invalid token").header("WWW-Authenticate", "Bearer realm=\"api\", error=\"invalid_token\"");
436 if (!tokens.get(token).equals(route.role) && !tokens.get(token).equals("admin")) return new Response(403, "forbidden: requires role " + route.role);
437 }
438 String type = negotiate(route.types, headers.get("accept"));
439 Response response;
440 if (type == null) response = new Response(406, "not acceptable: " + String.join(", ", route.types));
441 else {
442 String body = body(route, segments, query, headers, type), tag = etag(body);
443 boolean match = false;
444 if (headers.containsKey("if-none-match")) for (String item : headers.get("if-none-match").split(",", -1)) {
445 String value = strip(item);
446 if (value.startsWith("W/")) value = value.substring(2);
447 if (value.equals("*") || value.equals(tag)) match = true;
448 }
449 if (match && lookup.equals("GET")) response = new Response(304, "").header("ETag", tag);
450 else if (match) response = new Response(412, "precondition failed: " + tag);
451 else {
452 response = new Response(200, body).header("ETag", tag);
453 response.type = type;
454 }
455 }
456 if (route.types.size() >= 2) response.header("Vary", "Accept");
457 return response;
458 }

487 static void urlfor(String[] a) {
488 try {
489 Route route = find(a[1]);
490 if (route == null) invalid("unknown route " + a[1]);
491 Set<String> names = new HashSet<>();
492 for (Piece p : route.pieces) if (p.kind() != 'L') names.add(p.text());
493 Map<String, String> parameters = new HashMap<>();
494 List<Pair> query = new ArrayList<>();
495 for (String arg : Arrays.copyOfRange(a, 2, a.length)) {
496 int eq = arg.indexOf('=');
497 if (eq <= 0) invalid("bad argument " + arg);
498 String name = arg.substring(0, eq), value = arg.substring(eq + 1);
499 if (names.contains(name)) {
500 if (parameters.containsKey(name)) invalid("duplicate parameter " + name);
501 parameters.put(name, value);
502 } else query.add(new Pair(name, value));
503 }
504 for (Piece p : route.pieces) if (p.kind() != 'L') {
505 if (p.kind() != 'W' && !parameters.containsKey(p.text())) invalid("missing parameter " + p.text());
506 if (!validParameter(p, parameters.getOrDefault(p.text(), ""))) invalid("bad value for " + p.text());
507 }
508 List<String> pieces = new ArrayList<>();
509 for (Piece p : route.pieces) if (p.kind() == 'L') pieces.add(p.text());
510 else if (p.kind() == 'W') {
511 String value = parameters.getOrDefault(p.text(), "");
512 if (!value.isEmpty()) for (String part : value.split("/", -1)) pieces.add(encode(part));
513 } else pieces.add(encode(parameters.get(p.text())));
514 String url = "/" + String.join("/", pieces);
515 if (!query.isEmpty()) url += "?" + encodedPairs(query);
516 urlOk++;
517 System.out.println("URLFOR " + a[1] + " " + url);
518 } catch (Invalid ex) {
519 urlErrors++;
520 System.out.println("URLFOR ERROR " + ex.getMessage());
521 }
522 }
