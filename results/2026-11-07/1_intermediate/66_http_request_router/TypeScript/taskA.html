22function segment(text: string): boolean {
23 return /^[A-Za-z0-9._~-]+$/.test(text) && text !== '.' && text !== '..';
24}
25function canonicalPath(text: string): boolean {
26 return text === '/' || text.startsWith('/') && text.slice(1).split('/').every(segment);
27}
28function rulePath(text: string): { base: string; prefix: boolean } | null {
29 if (canonicalPath(text)) return { base: text, prefix: false };
30 if (text.endsWith('/*') && text.slice(0, -2) !== '/' && canonicalPath(text.slice(0, -2))) return { base: text.slice(0, -2), prefix: true };
31 return null;
32}
33function parsePattern(pattern: string): Piece[] | null {
34 if (pattern === '/') return [];
35 if (!pattern.startsWith('/')) return null;
36 const texts = pattern.slice(1).split('/');
37 const pieces: Piece[] = [];
38 const names = new Set<string>();
39 for (let index = 0; index < texts.length; index++) {
40 const text = texts[index];
41 if (segment(text)) {
42 pieces.push({ kind: 'literal', value: text });
43 continue;
44 }
45 const parameter = /^\{([a-z][a-z0-9_]{0,19})(?::(int|slug))?\}$/.exec(text);
46 const wildcard = /^\{\*([a-z][a-z0-9_]{0,19})\}$/.exec(text);
47 if (parameter === null && wildcard === null) return null;
48 const name = parameter === null ? wildcard![1] : parameter[1];
49 const kind = parameter === null ? 'wildcard' : parameter[2] ?? 'parameter';
50 if (names.has(name) || kind === 'wildcard' && index !== texts.length - 1) return null;
51 names.add(name);
52 pieces.push({ kind, value: name });
53 }
54 return pieces;
55}
56function compareRoutes(a: Route, b: Route): number {
57 const count = (route: Route, kinds: string[]): number => route.pieces.filter(piece => kinds.includes(piece.kind)).length;
58 return count(b, ['literal']) - count(a, ['literal']) || count(b, ['int', 'slug']) - count(a, ['int', 'slug']) || count(a, ['wildcard']) - count(b, ['wildcard']) || a.order - b.order;
59}
60function encode(text: string): string {
61 return [...text].map(char => unreserved.test(char) ? char : '%' + char.charCodeAt(0).toString(16).toUpperCase().padStart(2, '0')).join('');
62}
63function decode(text: string): string {
64 return text.replace(/%([0-9A-Fa-f]{2})/g, (_, digits: string) => String.fromCharCode(parseInt(digits, 16)));
65}
66function validEscapes(text: string): boolean {
67 for (let index = 0; index < text.length; index++) {
68 if (text[index] !== '%') continue;
69 const digits = text.slice(index + 1, index + 3);
70 if (!/^[0-9A-Fa-f]{2}$/.test(digits)) return false;
71 const code = parseInt(digits, 16);
72 if (code < 32 || code > 126) return false;
73 index += 2;
74 }
75 return true;
76}
77function normalizePath(path: string): string {
78 const decoded = path.replace(/%([0-9A-Fa-f]{2})/g, (escape, digits: string) => {
79 const char = String.fromCharCode(parseInt(digits, 16));
80 return unreserved.test(char) ? char : escape.toUpperCase();
81 });
82 const pieces: string[] = [];
83 for (const piece of decoded.slice(1).split('/')) {
84 if (piece === '' || piece === '.') continue;
85 if (piece === '..') pieces.pop();
86 else pieces.push(piece);
87 }
88 return '/' + pieces.join('/');
89}
90function matchRoute(route: Route, segments: string[]): boolean {
91 const wildcard = route.pieces.some(piece => piece.kind === 'wildcard');
92 const length = route.pieces.length - (wildcard ? 1 : 0);
93 if (wildcard ? segments.length < length : segments.length !== length) return false;
94 for (let index = 0; index < length; index++) {
95 const piece = route.pieces[index];
96 const value = segments[index];
97 if (piece.kind === 'literal' && piece.value !== value) return false;
98 if (piece.kind === 'int' && !/^[0-9]{1,9}$/.test(value)) return false;
99 if (piece.kind === 'slug' && !/^[a-z0-9-]+$/.test(value)) return false;
100 }
101 return true;
102}
103function parameterPairs(route: Route, segments: string[]): Pair[] {
104 const pairs: Pair[] = [];
105 route.pieces.forEach((piece, index) => {
106 if (piece.kind === 'literal') return;
107 const value = piece.kind === 'wildcard' ? segments.slice(index).map(decode).join('/') : piece.kind === 'int' ? String(Number(segments[index])) : piece.kind === 'slug' ? segments[index] : decode(segments[index]);
108 pairs.push([piece.value, value]);
109 });
110 return pairs.sort((a, b) => compareText(a[0], b[0]));
111}
112function queryPairs(query: string | null): Pair[] {
113 if (query === null) return [];
114 return query.split('&').filter(part => part !== '').map<Pair>(part => {
115 const separator = part.indexOf('=');
116 const name = separator < 0 ? part : part.slice(0, separator);
117 const value = separator < 0 ? '' : part.slice(separator + 1);
118 return [decode(name.replace(/\+/g, ' ')), decode(value.replace(/\+/g, ' '))];
119 });
120}
121function cookiePairs(cookie: string | undefined): Pair[] {
122 const pairs = new Map<string, string>();
123 if (cookie !== undefined) {
124 for (const part of cookie.split(';').map(strip)) {
125 const separator = part.indexOf('=');
126 if (separator <= 0) continue;
127 const name = part.slice(0, separator);
128 if (!pairs.has(name)) pairs.set(name, part.slice(separator + 1));
129 }
130 }
131 return [...pairs].sort((a, b) => compareText(a[0], b[0]));
132}
133function buildBody(route: Route, type: string, params: Pair[], query: Pair[], cookies: Pair[]): string {
134 const quote = (value: string): string => JSON.stringify(value);
135 if (type === 'application/json') {
136 const object = (pairs: Pair[]): string => '{' + pairs.map(([name, value]) => `${quote(name)}:${quote(value)}`).join(',') + '}';
137 return `{"route":${quote(route.name)},"params":${object(params)},"query":[${query.map(([name, value]) => `[${quote(name)},${quote(value)}]`).join(',')}],"cookies":${object(cookies)}}`;
138 }
139 if (type === 'text/html') {
140 const escapes: Record<string, string> = { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' };
141 const html = (value: string): string => value.replace(/[&<>"]/g, char => escapes[char]);
142 let result = `<dl><dt>route</dt><dd>${html(route.name)}</dd>`;
143 for (const [label, pairs] of [['param', params], ['query', query], ['cookie', cookies]] as [string, Pair[]][]) {
144 for (const [name, value] of pairs) result += `<dt>${label} ${html(name)}</dt><dd>${html(value)}</dd>`;
145 }
146 return result + '</dl>';
147 }
148 const list = (pairs: Pair[]): string => pairs.length === 0 ? '-' : pairs.map(([name, value]) => `${encode(name)}=${encode(value)}`).join('&');
149 return `route=${route.name} params=${list(params)} query=${list(query)} cookies=${list(cookies)}`;
150}
151function etag(body: string): string {
152 let a = 1;
153 let b = 0;
154 for (const char of body) {
155 a = (a + char.charCodeAt(0)) % 65521;
156 b = (b + a) % 65521;
157 }
158 return '"' + b.toString(16).padStart(4, '0') + a.toString(16).padStart(4, '0') + '"';
159}
160function negotiate(types: string[], accept: string | undefined): string | null {
161 if (accept === undefined) return types[0];
162 const elements: AcceptElement[] = [];
163 for (const element of accept.split(',')) {
164 const [first, ...parameters] = element.split(';').map(strip);
165 const range = first.toLowerCase();
166 let specificity: number;
167 if (range === '*/*') specificity = 1;
168 else if (/^[a-z0-9.+-]+\/\*$/.test(range)) specificity = 2;
169 else if (mediaType.test(range)) specificity = 3;
170 else continue;
171 let weight = 1000;
172 let valid = true;
173 for (const parameter of parameters) {
174 const separator = parameter.indexOf('=');
175 if (separator < 0 || strip(parameter.slice(0, separator)).toLowerCase() !== 'q') continue;
176 const q = strip(parameter.slice(separator + 1));
177 if (!/^(?:0(?:\.[0-9]{0,3})?|1(?:\.0{0,3})?)$/.test(q)) valid = false;
178 else weight = Math.round(Number(q) * 1000);
179 break;
180 }
181 if (valid) elements.push({ range, weight, specificity });
182 }
183 let best: string | null = null;
184 let bestWeight = 0;
185 for (const type of types) {
186 let specificity = 0;
187 let weight = 0;
188 for (const element of elements) {
189 const fits = element.range === '*/*' || element.range === type || element.specificity === 2 && element.range.slice(0, -1) === type.split('/')[0] + '/';
190 if (fits && element.specificity > specificity) {
191 specificity = element.specificity;
192 weight = element.weight;
193 }
194 }
195 if (weight > bestWeight) {
196 best = type;
197 bestWeight = weight;
198 }
199 }
200 return best;
201}

219 configure(words: string[]): void {
220 const [command, ...args] = words;
221 const arities: Record<string, number[]> = { ROUTE: [3, 5], TOKEN: [2], REDIRECT: [3], ORIGIN: [1], LIMIT: [2] };
222 if (!Object.hasOwn(arities, command) && command !== 'TYPES') throw new Error(`unknown command ${command}`);
223 if (command === 'TYPES' ? args.length < 2 : !arities[command].includes(args.length)) throw new Error('wrong number of arguments');
224 if (command === 'ROUTE') {
225 const [name, methodText, pattern] = args;
226 if (!validName.test(name)) throw new Error(`bad route name ${name}`);
227 if (this.byName.has(name)) throw new Error(`duplicate route ${name}`);
228 const methodList = methodText.split(',');
229 if (methodList.some(method => !methods.has(method)) || new Set(methodList).size !== methodList.length) throw new Error(`bad methods ${methodText}`);
230 const pieces = parsePattern(pattern);
231 if (pieces === null) throw new Error(`bad pattern ${pattern}`);
232 if (args.length === 5 && args[3] !== 'ROLE') throw new Error(`unknown option ${args[3]}`);
233 if (args.length === 5 && !roleName.test(args[4])) throw new Error(`bad role ${args[4]}`);
234 const route: Route = { name, methods: new Set(methodList), pattern, pieces, role: args.length === 5 ? args[4] : 'none', types: null, order: this.routes.length, selected: 0 };
235 this.routes.push(route);
236 this.byName.set(name, route);
237 } else if (command === 'TOKEN') {
238 const [value, role] = args;
239 if (!/^[A-Za-z0-9._-]{1,32}$/.test(value)) throw new Error(`bad token ${value}`);
240 if (this.tokens.has(value)) throw new Error(`duplicate token ${value}`);
241 if (!roleName.test(role)) throw new Error(`bad role ${role}`);
242 this.tokens.set(value, role);
243 } else if (command === 'REDIRECT') {
244 const from = rulePath(args[0]);
245 if (from === null) throw new Error(`bad path ${args[0]}`);
246 const to = rulePath(args[1]);
247 if (to === null) throw new Error(`bad path ${args[1]}`);
248 if (from.prefix !== to.prefix) throw new Error('wildcard mismatch');
249 if (!['301', '302', '307', '308'].includes(args[2])) throw new Error(`bad redirect code ${args[2]}`);
250 if (this.redirects.some(rule => rule.from === from.base && rule.prefix === from.prefix)) throw new Error(`duplicate redirect ${args[0]}`);
251 this.redirects.push({ from: from.base, to: to.base, prefix: from.prefix, code: Number(args[2]) });
252 } else if (command === 'ORIGIN') {
253 if (!/^https?:\/\/[a-z0-9.:-]+$/.test(args[0])) throw new Error(`bad origin ${args[0]}`);
254 if (this.origins.has(args[0])) throw new Error(`duplicate origin ${args[0]}`);
255 this.origins.add(args[0]);
256 } else if (command === 'LIMIT') {
257 if (!/^[1-9][0-9]*$/.test(args[0]) || Number(args[0]) > 100 || !/^[1-9][0-9]*$/.test(args[1]) || Number(args[1]) > 3600) throw new Error('bad limit');
258 if (this.limit !== null) throw new Error('duplicate limit');
259 this.limit = { capacity: Number(args[0]), seconds: Number(args[1]) };
260 } else {
261 const route = this.byName.get(args[0]);
262 if (route === undefined) throw new Error(`unknown route ${args[0]}`);
263 if (route.types !== null) throw new Error(`duplicate types ${args[0]}`);
264 const types: string[] = [];
265 for (const type of args.slice(1)) {
266 if (!mediaType.test(type)) throw new Error(`bad media type ${type}`);
267 if (types.includes(type)) throw new Error(`duplicate media type ${type}`);
268 types.push(type);
269 }
270 route.types = types;
271 }
272 }
273
274 finishConfiguration(): void {
275 this.output.push(`CONFIG routes=${this.routes.length} tokens=${this.tokens.size} redirects=${this.redirects.length} origins=${this.origins.size} limit=${this.limit === null ? 'none' : `${this.limit.capacity}/${this.limit.seconds}`} errors=${this.errors}`);
276 for (const route of this.routes) if (route.types === null) route.types = ['application/json'];
277 this.table = [...this.routes].sort(compareRoutes);
278 this.table.forEach((route, index) => {
279 const methodSet = new Set(route.methods);
280 if (methodSet.has('GET')) methodSet.add('HEAD');
281 this.output.push(`TABLE ${index + 1} ${route.name} ${[...methodSet].sort().join(',')} ${route.pattern} role=${route.role === 'none' ? '-' : route.role} types=${route.types!.join(',')}`);
282 });
283 }

285 request(line: string, headerLines: string[]): Response {
286 const headers = new Map<string, string>();
287 let head = false;
288 let cors = false;
289 const finish = (code: number, body = '', extra: Pair[] = [], type = 'text/plain'): Response => {
290 const responseHeaders = new Map(extra);
291 const origin = headers.get('origin');
292 if (cors && origin !== undefined && this.origins.has(origin)) {
293 responseHeaders.set('Access-Control-Allow-Origin', origin);
294 if (code === 204 && headers.has('access-control-request-method')) responseHeaders.set('Access-Control-Allow-Methods', responseHeaders.get('Allow')!);
295 }
296 if (code !== 204 && code !== 304) {
297 responseHeaders.set('Content-Type', code === 200 ? type : 'text/plain');
298 responseHeaders.set('Content-Length', String(body.length));
299 }
300 return { code, body, headers: responseHeaders, head };
301 };
302 const parts = line.split(' ');
303 if (parts.length !== 3 || !/^[A-Z]{1,16}$/.test(parts[0]) || !parts[1].startsWith('/') || parts[1].includes('#') || parts[2] !== 'HTTP/1.1') return finish(400, 'bad request: request line');
304 const [method, target] = parts;
305 head = method === 'HEAD';
306 if (!validEscapes(target)) return finish(400, 'bad request: percent escape');
307 for (let index = 0; index < headerLines.length; index++) {
308 const header = headerLines[index];
309 const separator = header.indexOf(':');
310 if (separator < 0 || !/^[A-Za-z0-9-]+$/.test(header.slice(0, separator))) return finish(400, `bad request: header line ${index + 1}`);
311 const name = header.slice(0, separator).toLowerCase();
312 if (headers.has(name)) return finish(400, `bad request: duplicate header ${name}`);
313 headers.set(name, strip(header.slice(separator + 1)));
314 }
315 cors = true;
316 if (!requestMethods.has(method)) return finish(501, `not implemented: ${method}`);
317 if (!headers.has('host') || headers.get('host') === '') return finish(400, 'bad request: missing host');
318 let effective = method;
319 if (headers.has('x-http-method-override')) {
320 const override = headers.get('x-http-method-override')!;
321 if (method !== 'POST' || !['PUT', 'PATCH', 'DELETE'].includes(override)) return finish(400, 'bad request: method override');
322 effective = override;
323 }
324 const clientName = headers.get('x-client');
325 if (clientName !== undefined && clientName !== '') {
326 const time = Number(headers.get('x-time'));
327 let client = this.clients.get(clientName);
328 if (client === undefined) {
329 client = { level: this.limit === null ? 0 : this.limit.capacity * this.limit.seconds, last: time, allowed: 0, limited: 0 };
330 this.clients.set(clientName, client);
331 }
332 if (this.limit !== null) {
333 const { capacity, seconds } = this.limit;
334 client.level = Math.min(capacity * seconds, client.level + (time - client.last) * capacity);
335 client.last = time;
336 if (client.level < seconds) {
337 client.limited++;
338 return finish(429, `too many requests: client ${clientName}`, [['Retry-After', String(Math.ceil((seconds - client.level) / capacity))]]);
339 }
340 client.level -= seconds;
341 }
342 client.allowed++;
343 }
344 const question = target.indexOf('?');
345 const path = question < 0 ? target : target.slice(0, question);
346 const query = question < 0 ? null : target.slice(question + 1);
347 const suffix = question < 0 ? '' : target.slice(question);
348 const normalized = normalizePath(path);
349 if (normalized !== path) return finish(308, `redirect to ${normalized}${suffix}`, [['Location', normalized + suffix]]);
350 for (const rule of this.redirects) {
351 if (path === rule.from || rule.prefix && path.startsWith(rule.from + '/')) {
352 const destination = rule.to + path.slice(rule.from.length) + suffix;
353 return finish(rule.code, `redirect to ${destination}`, [['Location', destination]]);
354 }
355 }
356 const segments = path === '/' ? [] : path.slice(1).split('/');
357 const matching = this.table.filter(route => matchRoute(route, segments));
358 if (matching.length === 0) return finish(404, `not found: ${path}`);
359 const allowed = new Set<string>(['OPTIONS']);
360 for (const route of matching) for (const method of route.methods) allowed.add(method);
361 if (allowed.has('GET')) allowed.add('HEAD');
362 const allow = [...allowed].sort().join(', ');
363 if (effective === 'OPTIONS') return finish(204, '', [['Allow', allow]]);
364 const lookup = effective === 'HEAD' ? 'GET' : effective;
365 const selected = matching.find(route => route.methods.has(lookup));
366 if (selected === undefined) return finish(405, `method not allowed: ${effective}`, [['Allow', allow]]);
367 selected.selected++;
368 if (selected.role !== 'none') {
369 if (!headers.has('authorization')) return finish(401, 'unauthorized: missing token', [['WWW-Authenticate', 'Bearer realm="api"']]);
370 const authorization = headers.get('authorization')!;
371 const space = authorization.indexOf(' ');
372 const scheme = space < 0 ? authorization : authorization.slice(0, space);
373 const token = space < 0 ? '' : authorization.slice(space + 1);
374 if (scheme.toLowerCase() !== 'bearer' || !this.tokens.has(token)) return finish(401, 'unauthorized: invalid token', [['WWW-Authenticate', 'Bearer realm="api", error="invalid_token"']]);
375 const role = this.tokens.get(token)!;
376 if (role !== selected.role && role !== 'admin') return finish(403, `forbidden: requires role ${selected.role}`);
377 }
378 const types = selected.types!;
379 const extra: Pair[] = types.length > 1 ? [['Vary', 'Accept']] : [];
380 const type = negotiate(types, headers.get('accept'));
381 if (type === null) return finish(406, `not acceptable: ${types.join(', ')}`, extra);
382 const body = buildBody(selected, type, parameterPairs(selected, segments), queryPairs(query), cookiePairs(headers.get('cookie')));
383 const tag = etag(body);
384 const condition = headers.get('if-none-match');
385 if (condition !== undefined && condition.split(',').map(strip).map(value => value.startsWith('W/') ? value.slice(2) : value).some(value => value === '*' || value === tag)) {
386 if (lookup === 'GET') return finish(304, '', [...extra, ['ETag', tag]]);
387 return finish(412, `precondition failed: ${tag}`, extra);
388 }
389 return finish(200, body, [...extra, ['ETag', tag]], type);
390 }

392 urlFor(words: string[]): string {
393 const name = words[1];
394 const route = this.byName.get(name);
395 if (route === undefined) throw new Error(`unknown route ${name}`);
396 const parameterNames = new Set(route.pieces.filter(piece => piece.kind !== 'literal').map(piece => piece.value));
397 const parameters = new Map<string, string>();
398 const query: Pair[] = [];
399 for (const argument of words.slice(2)) {
400 const separator = argument.indexOf('=');
401 if (separator <= 0) throw new Error(`bad argument ${argument}`);
402 const key = argument.slice(0, separator);
403 const value = argument.slice(separator + 1);
404 if (parameterNames.has(key)) {
405 if (parameters.has(key)) throw new Error(`duplicate parameter ${key}`);
406 parameters.set(key, value);
407 } else query.push([key, value]);
408 }
409 const segments: string[] = [];
410 for (const piece of route.pieces) {
411 if (piece.kind === 'literal') {
412 segments.push(piece.value);
413 continue;
414 }
415 const value = parameters.get(piece.value);
416 if (piece.kind !== 'wildcard' && value === undefined) throw new Error(`missing parameter ${piece.value}`);
417 const text = value ?? '';
418 let valid: boolean;
419 if (piece.kind === 'parameter') valid = text !== '' && text !== '.' && text !== '..';
420 else if (piece.kind === 'int') valid = /^[0-9]{1,9}$/.test(text);
421 else if (piece.kind === 'slug') valid = /^[a-z0-9-]+$/.test(text);
422 else valid = text === '' || text.split('/').every(part => part !== '' && part !== '.' && part !== '..');
423 if (!valid) throw new Error(`bad value for ${piece.value}`);
424 if (piece.kind === 'wildcard') {
425 if (text !== '') segments.push(...text.split('/').map(encode));
426 } else segments.push(encode(text));
427 }
428 const suffix = query.length === 0 ? '' : '?' + query.map(([key, value]) => `${encode(key)}=${encode(value)}`).join('&');
429 return `URLFOR ${name} /${segments.join('/')}${suffix}`;
430 }

432 emitResponse(response: Response): void {
433 this.requestCount++;
434 this.statuses.set(response.code, (this.statuses.get(response.code) ?? 0) + 1);
435 this.output.push(`REQUEST ${this.requestCount}`, `HTTP/1.1 ${response.code} ${reasons[response.code]}`);
436 for (const name of headerOrder) if (response.headers.has(name)) this.output.push(`${name}: ${response.headers.get(name)!}`);
437 if (!response.head && response.code !== 204 && response.code !== 304) this.output.push(response.body);
438 }

440 run(lines: string[]): void {
441 let position = 0;
442 while (lines[position] !== 'REQUESTS') {
443 try {
444 this.configure(lines[position].split(' '));
445 } catch (error) {
446 this.errors++;
447 this.output.push(`CONFIG ERROR ${position + 1}: ${(error as Error).message}`);
448 }
449 position++;
450 }
451 position++;
452 this.finishConfiguration();
453 while (lines[position] !== 'END') {
454 const line = lines[position++];
455 if (line.startsWith('URLFOR ')) {
456 try {
457 this.output.push(this.urlFor(line.split(' ')));
458 this.urlOk++;
459 } catch (error) {
460 this.urlError++;
461 this.output.push(`URLFOR ERROR ${(error as Error).message}`);
462 }
463 } else {
464 const headers: string[] = [];
465 while (lines[position] !== '') headers.push(lines[position++]);
466 position++;
467 this.emitResponse(this.request(line, headers));
468 }
469 }
470 this.output.push(`SUMMARY requests=${this.requestCount} urlfor_ok=${this.urlOk} urlfor_error=${this.urlError}`);
471 for (const code of [...this.statuses.keys()].sort((a, b) => a - b)) this.output.push(`STATUS ${code} ${this.statuses.get(code)!}`);
472 for (const route of this.routes) this.output.push(`ROUTE ${route.name} ${route.selected}`);
473 for (const name of [...this.clients.keys()].sort()) {
474 const client = this.clients.get(name)!;
475 this.output.push(`CLIENT ${name} allowed=${client.allowed} limited=${client.limited}`);
476 }
477 }
478}

480const router = new HttpRouter();
481router.run(fs.readFileSync(0, 'utf8').split('\n'));
482process.stdout.write(router.output.join('\n') + '\n');
