83List<Piece>? parsePattern(String pattern) {
84 if (pattern == '/') return [];
85 if (!pattern.startsWith('/')) return null;
86 final texts = pattern.substring(1).split('/');
87 final pieces = <Piece>[];
88 final names = <String>{};
89 for (var index = 0; index < texts.length; index++) {
90 final text = texts[index];
91 if (segment(text)) {
92 pieces.add(Piece('literal', text));
93 continue;
94 }
95 final parameter = RegExp(r'^\{([a-z][a-z0-9_]{0,19})(?::(int|slug))?\}$').firstMatch(text);
96 final wildcard = RegExp(r'^\{\*([a-z][a-z0-9_]{0,19})\}$').firstMatch(text);
97 if (parameter == null && wildcard == null) return null;
98 final name = parameter == null ? wildcard![1]! : parameter[1]!;
99 final kind = parameter == null ? 'wildcard' : parameter[2] ?? 'parameter';
100 if (names.contains(name) || kind == 'wildcard' && index != texts.length - 1) return null;
101 names.add(name);
102 pieces.add(Piece(kind, name));
103 }
104 return pieces;
105}

115String encode(String text) => text.split('').map((char) => unreserved.hasMatch(char) ? char : '%${char.codeUnitAt(0).toRadixString(16).toUpperCase().padLeft(2, '0')}').join();
116String decode(String text) => text.replaceAllMapped(RegExp(r'%([0-9A-Fa-f]{2})'), (match) => String.fromCharCode(int.parse(match[1]!, radix: 16)));

118bool validEscapes(String text) {
119 for (var index = 0; index < text.length; index++) {
120 if (text[index] != '%') continue;
121 if (index + 2 >= text.length) return false;
122 final digits = text.substring(index + 1, index + 3);
123 if (!RegExp(r'^[0-9A-Fa-f]{2}$').hasMatch(digits)) return false;
124 final code = int.parse(digits, radix: 16);
125 if (code < 32 || code > 126) return false;
126 index += 2;
127 }
128 return true;
129}

131String normalizePath(String path) {
132 final decoded = path.replaceAllMapped(RegExp(r'%([0-9A-Fa-f]{2})'), (match) {
133 final char = String.fromCharCode(int.parse(match[1]!, radix: 16));
134 return unreserved.hasMatch(char) ? char : match[0]!.toUpperCase();
135 });
136 final pieces = <String>[];
137 for (final piece in decoded.substring(1).split('/')) {
138 if (piece.isEmpty || piece == '.') continue;
139 if (piece == '..') {
140 if (pieces.isNotEmpty) pieces.removeLast();
141 } else {
142 pieces.add(piece);
143 }
144 }
145 return '/${pieces.join('/')}';
146}

148bool matchRoute(Route route, List<String> segments) {
149 final wildcard = route.pieces.any((piece) => piece.kind == 'wildcard');
150 final length = route.pieces.length - (wildcard ? 1 : 0);
151 if (wildcard ? segments.length < length : segments.length != length) return false;
152 for (var index = 0; index < length; index++) {
153 final piece = route.pieces[index];
154 final value = segments[index];
155 if (piece.kind == 'literal' && piece.value != value) return false;
156 if (piece.kind == 'int' && !RegExp(r'^[0-9]{1,9}$').hasMatch(value)) return false;
157 if (piece.kind == 'slug' && !RegExp(r'^[a-z0-9-]+$').hasMatch(value)) return false;
158 }
159 return true;
160}

162List<Pair> parameterPairs(Route route, List<String> segments) {
163 final pairs = <Pair>[];
164 for (var index = 0; index < route.pieces.length; index++) {
165 final piece = route.pieces[index];
166 if (piece.kind == 'literal') continue;
167 final value = piece.kind == 'wildcard' ? segments.skip(index).map(decode).join('/') : piece.kind == 'int' ? int.parse(segments[index]).toString() : piece.kind == 'slug' ? segments[index] : decode(segments[index]);
168 pairs.add((piece.value, value));
169 }
170 return pairs..sort((a, b) => a.$1.compareTo(b.$1));

173List<Pair> queryPairs(String? query) {
174 if (query == null) return [];
175 final pairs = <Pair>[];
176 for (final part in query.split('&')) {
177 if (part.isEmpty) continue;
178 final separator = part.indexOf('=');
179 final name = separator < 0 ? part : part.substring(0, separator);
180 final value = separator < 0 ? '' : part.substring(separator + 1);
181 pairs.add((decode(name.replaceAll('+', ' ')), decode(value.replaceAll('+', ' '))));
182 }
183 return pairs;

186List<Pair> cookiePairs(String? cookie) {
187 final pairs = <String, String>{};
188 if (cookie != null) {
189 for (final part in cookie.split(';').map(strip)) {
190 final separator = part.indexOf('=');
191 if (separator <= 0) continue;
192 final name = part.substring(0, separator);
193 pairs.putIfAbsent(name, () => part.substring(separator + 1));
194 }
195 }
196 return pairs.entries.map((entry) => (entry.key, entry.value)).toList()..sort((a, b) => a.$1.compareTo(b.$1));

199String buildBody(Route route, String type, List<Pair> params, List<Pair> query, List<Pair> cookies) {
200 String quote(String value) => jsonEncode(value);
201 if (type == 'application/json') {
202 String object(List<Pair> pairs) => '{${pairs.map((pair) => '${quote(pair.$1)}:${quote(pair.$2)}').join(',')}}';
203 return '{"route":${quote(route.name)},"params":${object(params)},"query":[${query.map((pair) => '[${quote(pair.$1)},${quote(pair.$2)}]').join(',')}],"cookies":${object(cookies)}}';
204 }
205 if (type == 'text/html') {
206 const escapes = {'&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;'};
207 String html(String value) => value.replaceAllMapped(RegExp('[&<>"]'), (match) => escapes[match[0]]!);
208 final result = StringBuffer('<dl><dt>route</dt><dd>${html(route.name)}</dd>');
209 for (final group in [('param', params), ('query', query), ('cookie', cookies)]) {
210 for (final pair in group.$2) result.write('<dt>${group.$1} ${html(pair.$1)}</dt><dd>${html(pair.$2)}</dd>');
211 }
212 return '${result.toString()}</dl>';
213 }
214 String list(List<Pair> pairs) => pairs.isEmpty ? '-' : pairs.map((pair) => '${encode(pair.$1)}=${encode(pair.$2)}').join('&');
215 return 'route=${route.name} params=${list(params)} query=${list(query)} cookies=${list(cookies)}';
216}

218String etag(String body) {
219 var a = 1;
220 var b = 0;
221 for (final code in body.codeUnits) {
222 a = (a + code) % 65521;
223 b = (b + a) % 65521;
224 }
225 return '"${b.toRadixString(16).padLeft(4, '0')}${a.toRadixString(16).padLeft(4, '0')}"';
226}

228String? negotiate(List<String> types, String? accept) {
229 if (accept == null) return types[0];
230 final elements = <AcceptElement>[];
231 for (final element in accept.split(',')) {
232 final parts = element.split(';').map(strip).toList();
233 final range = parts[0].toLowerCase();
234 final int specificity;
235 if (range == '*/*') {
236 specificity = 1;
237 } else if (RegExp(r'^[a-z0-9.+-]+\/\*$').hasMatch(range)) {
238 specificity = 2;
239 } else if (mediaType.hasMatch(range)) {
240 specificity = 3;
241 } else {
242 continue;
243 }
244 var weight = 1000;
245 var valid = true;
246 for (final parameter in parts.skip(1)) {
247 final separator = parameter.indexOf('=');
248 if (separator < 0 || strip(parameter.substring(0, separator)).toLowerCase() != 'q') continue;
249 final q = strip(parameter.substring(separator + 1));
250 if (!RegExp(r'^(?:0(?:\.[0-9]{0,3})?|1(?:\.0{0,3})?)$').hasMatch(q)) {
251 valid = false;
252 } else {
253 final decimal = q.indexOf('.');
254 weight = q.startsWith('1') ? 1000 : decimal < 0 ? 0 : int.parse(q.substring(decimal + 1).padRight(3, '0'));
255 }
256 break;
257 }
258 if (valid) elements.add((range: range, weight: weight, specificity: specificity));
259 }
260 String? best;
261 var bestWeight = 0;
262 for (final type in types) {
263 var specificity = 0;
264 var weight = 0;
265 for (final element in elements) {
266 final fits = element.range == '*/*' || element.range == type || element.specificity == 2 && element.range.substring(0, element.range.length - 1) == '${type.split('/')[0]}/';
267 if (fits && element.specificity > specificity) {
268 specificity = element.specificity;
269 weight = element.weight;
270 }
271 }
272 if (weight > bestWeight) {
273 best = type;
274 bestWeight = weight;
275 }
276 }
277 return best;
278}

296 void configure(List<String> words) {
297 final command = words[0];
298 final args = words.sublist(1);
299 const arities = {'ROUTE': [3, 5], 'TOKEN': [2], 'REDIRECT': [3], 'ORIGIN': [1], 'LIMIT': [2]};
300 if (!arities.containsKey(command) && command != 'TYPES') throw InputFailure('unknown command $command');
301 if (command == 'TYPES' ? args.length < 2 : !arities[command]!.contains(args.length)) throw InputFailure('wrong number of arguments');
302 if (command == 'ROUTE') {
303 final name = args[0];
304 final methodText = args[1];
305 final pattern = args[2];
306 if (!validName.hasMatch(name)) throw InputFailure('bad route name $name');
307 if (byName.containsKey(name)) throw InputFailure('duplicate route $name');
308 final methodList = methodText.split(',');
309 if (methodList.any((method) => !methods.contains(method)) || methodList.toSet().length != methodList.length) throw InputFailure('bad methods $methodText');
310 final pieces = parsePattern(pattern);
311 if (pieces == null) throw InputFailure('bad pattern $pattern');
312 if (args.length == 5 && args[3] != 'ROLE') throw InputFailure('unknown option ${args[3]}');
313 if (args.length == 5 && !roleName.hasMatch(args[4])) throw InputFailure('bad role ${args[4]}');
314 final route = Route(name, methodList.toSet(), pattern, pieces, args.length == 5 ? args[4] : 'none', routes.length);
315 routes.add(route);
316 byName[name] = route;
317 } else if (command == 'TOKEN') {
318 final value = args[0];
319 final role = args[1];
320 if (!RegExp(r'^[A-Za-z0-9._-]{1,32}$').hasMatch(value)) throw InputFailure('bad token $value');
321 if (tokens.containsKey(value)) throw InputFailure('duplicate token $value');
322 if (!roleName.hasMatch(role)) throw InputFailure('bad role $role');
323 tokens[value] = role;
324 } else if (command == 'REDIRECT') {
325 final from = rulePath(args[0]);
326 if (from == null) throw InputFailure('bad path ${args[0]}');
327 final to = rulePath(args[1]);
328 if (to == null) throw InputFailure('bad path ${args[1]}');
329 if (from.prefix != to.prefix) throw InputFailure('wildcard mismatch');
330 if (!['301', '302', '307', '308'].contains(args[2])) throw InputFailure('bad redirect code ${args[2]}');
331 if (redirects.any((rule) => rule.from == from.base && rule.prefix == from.prefix)) throw InputFailure('duplicate redirect ${args[0]}');
332 redirects.add(Redirect(from.base, to.base, from.prefix, int.parse(args[2])));
333 } else if (command == 'ORIGIN') {
334 if (!RegExp(r'^https?:\/\/[a-z0-9.:-]+$').hasMatch(args[0])) throw InputFailure('bad origin ${args[0]}');
335 if (origins.contains(args[0])) throw InputFailure('duplicate origin ${args[0]}');
336 origins.add(args[0]);
337 } else if (command == 'LIMIT') {
338 final n = int.tryParse(args[0]);
339 final seconds = int.tryParse(args[1]);
340 if (!RegExp(r'^[1-9][0-9]*$').hasMatch(args[0]) || n == null || n > 100 || !RegExp(r'^[1-9][0-9]*$').hasMatch(args[1]) || seconds == null || seconds > 3600) throw InputFailure('bad limit');
341 if (limit != null) throw InputFailure('duplicate limit');
342 limit = RateLimit(n, seconds);
343 } else {
344 final route = byName[args[0]];
345 if (route == null) throw InputFailure('unknown route ${args[0]}');
346 if (route.types != null) throw InputFailure('duplicate types ${args[0]}');
347 final types = <String>[];
348 for (final type in args.skip(1)) {
349 if (!mediaType.hasMatch(type)) throw InputFailure('bad media type $type');
350 if (types.contains(type)) throw InputFailure('duplicate media type $type');
351 types.add(type);
352 }
353 route.types = types;
354 }
355 }

370 RouterResponse request(String line, List<String> headerLines) {
371 final headers = <String, String>{};
372 var head = false;
373 var cors = false;
374 RouterResponse finish(int code, [String body = '', List<Pair> extra = const [], String type = 'text/plain']) {
375 final responseHeaders = Map<String, String>.fromEntries(extra.map((pair) => MapEntry(pair.$1, pair.$2)));
376 final origin = headers['origin'];
377 if (cors && origin != null && origins.contains(origin)) {
378 responseHeaders['Access-Control-Allow-Origin'] = origin;
379 if (code == 204 && headers.containsKey('access-control-request-method')) responseHeaders['Access-Control-Allow-Methods'] = responseHeaders['Allow']!;
380 }
381 if (code != 204 && code != 304) {
382 responseHeaders['Content-Type'] = code == 200 ? type : 'text/plain';
383 responseHeaders['Content-Length'] = '${body.length}';
384 }
385 return RouterResponse(code, body, responseHeaders, head);
386 }
387 final parts = line.split(' ');
388 if (parts.length != 3 || !RegExp(r'^[A-Z]{1,16}$').hasMatch(parts[0]) || !parts[1].startsWith('/') || parts[1].contains('#') || parts[2] != 'HTTP/1.1') return finish(400, 'bad request: request line');
389 final method = parts[0];
390 final target = parts[1];
391 head = method == 'HEAD';
392 if (!validEscapes(target)) return finish(400, 'bad request: percent escape');
393 for (var index = 0; index < headerLines.length; index++) {
394 final header = headerLines[index];
395 final separator = header.indexOf(':');
396 if (separator < 0 || !RegExp(r'^[A-Za-z0-9-]+$').hasMatch(header.substring(0, separator))) return finish(400, 'bad request: header line ${index + 1}');
397 final name = header.substring(0, separator).toLowerCase();
398 if (headers.containsKey(name)) return finish(400, 'bad request: duplicate header $name');
399 headers[name] = strip(header.substring(separator + 1));
400 }
401 cors = true;
402 if (!requestMethods.contains(method)) return finish(501, 'not implemented: $method');
403 if (!headers.containsKey('host') || headers['host'] == '') return finish(400, 'bad request: missing host');
404 var effective = method;
405 if (headers.containsKey('x-http-method-override')) {
406 final override = headers['x-http-method-override']!;
407 if (method != 'POST' || !['PUT', 'PATCH', 'DELETE'].contains(override)) return finish(400, 'bad request: method override');
408 effective = override;
409 }
410 final clientName = headers['x-client'];
411 if (clientName != null && clientName.isNotEmpty) {
412 final time = int.parse(headers['x-time']!);
413 final client = clients.putIfAbsent(clientName, () => Client(limit == null ? 0 : limit!.capacity * limit!.seconds, time));
414 final rate = limit;
415 if (rate != null) {
416 final refilled = client.level + (time - client.last) * rate.capacity;
417 final capacity = rate.capacity * rate.seconds;
418 client.level = refilled < capacity ? refilled : capacity;
419 client.last = time;
420 if (client.level < rate.seconds) {
421 client.limited++;
422 final wait = (rate.seconds - client.level + rate.capacity - 1) ~/ rate.capacity;
423 return finish(429, 'too many requests: client $clientName', [('Retry-After', '$wait')]);
424 }
425 client.level -= rate.seconds;
426 }
427 client.allowed++;
428 }
429 final question = target.indexOf('?');
430 final path = question < 0 ? target : target.substring(0, question);
431 final query = question < 0 ? null : target.substring(question + 1);
432 final suffix = question < 0 ? '' : target.substring(question);
433 final normalized = normalizePath(path);
434 if (normalized != path) return finish(308, 'redirect to $normalized$suffix', [('Location', normalized + suffix)]);
435 for (final rule in redirects) {
436 if (path == rule.from || rule.prefix && path.startsWith('${rule.from}/')) {
437 final destination = rule.to + path.substring(rule.from.length) + suffix;
438 return finish(rule.code, 'redirect to $destination', [('Location', destination)]);
439 }
440 }
441 final segments = path == '/' ? <String>[] : path.substring(1).split('/');
442 final matching = table.where((route) => matchRoute(route, segments)).toList();
443 if (matching.isEmpty) return finish(404, 'not found: $path');
444 final allowed = <String>{'OPTIONS'};
445 for (final route in matching) allowed.addAll(route.methods);
446 if (allowed.contains('GET')) allowed.add('HEAD');
447 final allowedList = allowed.toList()..sort();
448 final allow = allowedList.join(', ');
449 if (effective == 'OPTIONS') return finish(204, '', [('Allow', allow)]);
450 final lookup = effective == 'HEAD' ? 'GET' : effective;
451 Route? selected;
452 for (final route in matching) {
453 if (route.methods.contains(lookup)) {
454 selected = route;
455 break;
456 }
457 }
458 if (selected == null) return finish(405, 'method not allowed: $effective', [('Allow', allow)]);
459 selected.selected++;
460 if (selected.role != 'none') {
461 if (!headers.containsKey('authorization')) return finish(401, 'unauthorized: missing token', [('WWW-Authenticate', 'Bearer realm="api"')]);
462 final authorization = headers['authorization']!;
463 final space = authorization.indexOf(' ');
464 final scheme = space < 0 ? authorization : authorization.substring(0, space);
465 final token = space < 0 ? '' : authorization.substring(space + 1);
466 if (scheme.toLowerCase() != 'bearer' || !tokens.containsKey(token)) return finish(401, 'unauthorized: invalid token', [('WWW-Authenticate', 'Bearer realm="api", error="invalid_token"')]);
467 final role = tokens[token]!;
468 if (role != selected.role && role != 'admin') return finish(403, 'forbidden: requires role ${selected.role}');
469 }
470 final types = selected.types!;
471 final extra = <Pair>[if (types.length > 1) ('Vary', 'Accept')];
472 final type = negotiate(types, headers['accept']);
473 if (type == null) return finish(406, 'not acceptable: ${types.join(', ')}', extra);
474 final body = buildBody(selected, type, parameterPairs(selected, segments), queryPairs(query), cookiePairs(headers['cookie']));
475 final tag = etag(body);
476 final condition = headers['if-none-match'];
477 if (condition != null && condition.split(',').map(strip).map((value) => value.startsWith('W/') ? value.substring(2) : value).any((value) => value == '*' || value == tag)) {
478 if (lookup == 'GET') return finish(304, '', [...extra, ('ETag', tag)]);
479 return finish(412, 'precondition failed: $tag', extra);
480 }
481 return finish(200, body, [...extra, ('ETag', tag)], type);
482 }

484 String urlFor(List<String> words) {
485 final name = words[1];
486 final route = byName[name];
487 if (route == null) throw InputFailure('unknown route $name');
488 final parameterNames = route.pieces.where((piece) => piece.kind != 'literal').map((piece) => piece.value).toSet();
489 final parameters = <String, String>{};
490 final query = <Pair>[];
491 for (final argument in words.skip(2)) {
492 final separator = argument.indexOf('=');
493 if (separator <= 0) throw InputFailure('bad argument $argument');
494 final key = argument.substring(0, separator);
495 final value = argument.substring(separator + 1);
496 if (parameterNames.contains(key)) {
497 if (parameters.containsKey(key)) throw InputFailure('duplicate parameter $key');
498 parameters[key] = value;
499 } else {
500 query.add((key, value));
501 }
502 }
503 final segments = <String>[];
504 for (final piece in route.pieces) {
505 if (piece.kind == 'literal') {
506 segments.add(piece.value);
507 continue;
508 }
509 final value = parameters[piece.value];
510 if (piece.kind != 'wildcard' && value == null) throw InputFailure('missing parameter ${piece.value}');
511 final text = value ?? '';
512 final bool valid;
513 if (piece.kind == 'parameter') {
514 valid = text.isNotEmpty && text != '.' && text != '..';
515 } else if (piece.kind == 'int') {
516 valid = RegExp(r'^[0-9]{1,9}$').hasMatch(text);
517 } else if (piece.kind == 'slug') {
518 valid = RegExp(r'^[a-z0-9-]+$').hasMatch(text);
519 } else {
520 valid = text.isEmpty || text.split('/').every((part) => part.isNotEmpty && part != '.' && part != '..');
521 }
522 if (!valid) throw InputFailure('bad value for ${piece.value}');
523 if (piece.kind == 'wildcard') {
524 if (text.isNotEmpty) segments.addAll(text.split('/').map(encode));
525 } else {
526 segments.add(encode(text));
527 }
528 }
529 final suffix = query.isEmpty ? '' : '?${query.map((pair) => '${encode(pair.$1)}=${encode(pair.$2)}').join('&')}';
530 return 'URLFOR $name /${segments.join('/')}$suffix';
531 }
532
533 void emitResponse(RouterResponse response) {
534 requestCount++;
535 statuses[response.code] = (statuses[response.code] ?? 0) + 1;
536 output.add('REQUEST $requestCount');
537 output.add('HTTP/1.1 ${response.code} ${reasons[response.code]}');
538 for (final name in headerOrder) {
539 if (response.headers.containsKey(name)) output.add('$name: ${response.headers[name]}');
540 }
541 if (!response.head && response.code != 204 && response.code != 304) output.add(response.body);
542 }
543
544 void run(List<String> lines) {
545 var position = 0;
546 while (lines[position] != 'REQUESTS') {
547 try {
548 configure(lines[position].split(' '));
549 } on InputFailure catch (error) {
550 errors++;
551 output.add('CONFIG ERROR ${position + 1}: ${error.message}');
552 }
553 position++;
554 }
555 position++;
556 finishConfiguration();
557 while (lines[position] != 'END') {
558 final line = lines[position++];
559 if (line.startsWith('URLFOR ')) {
560 try {
561 output.add(urlFor(line.split(' ')));
562 urlOk++;
563 } on InputFailure catch (error) {
564 urlError++;
565 output.add('URLFOR ERROR ${error.message}');
566 }
567 } else {
568 final headers = <String>[];
569 while (lines[position].isNotEmpty) headers.add(lines[position++]);
570 position++;
571 emitResponse(request(line, headers));
572 }
573 }
574 output.add('SUMMARY requests=$requestCount urlfor_ok=$urlOk urlfor_error=$urlError');
575 final codes = statuses.keys.toList()..sort();
576 for (final code in codes) output.add('STATUS $code ${statuses[code]}');
577 for (final route in routes) output.add('ROUTE ${route.name} ${route.selected}');
578 final clientNames = clients.keys.toList()..sort();
579 for (final name in clientNames) {
580 final client = clients[name]!;
581 output.add('CLIENT $name allowed=${client.allowed} limited=${client.limited}');
582 }
583 }
584}
