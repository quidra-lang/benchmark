162static char **split(const char *s, char delimiter, size_t *count) {
163 char **parts = NULL;
164 size_t start = 0;
165 *count = 0;
166 for (size_t i = 0;; i++) if (!s[i] || s[i] == delimiter) {
167 APPEND(parts, *count, slice(s + start, i - start));
168 start = i + 1;
169 if (!s[i]) break;
170 }
171 return parts;

183static bool matches(const char *s, const char *pattern) {
184 regex_t regex;
185 char *full = format("^(%s)$", pattern);
186 if (regcomp(&regex, full, REG_EXTENDED | REG_NOSUB)) exit(EXIT_FAILURE);
187 bool result = !regexec(&regex, s, 0, NULL, 0);
188 regfree(&regex);
189 free(full);
190 return result;
191}

263static bool parse_pattern(Route *r, const char *text) {
264 if (*text != '/') return false;
265 if (!strcmp(text, "/")) return true;
266 size_t count;
267 char **fields = split(text + 1, '/', &count);
268 for (size_t i = 0; i < count; i++) {
269 char *field = fields[i];
270 Piece p = {'L', field};
271 if (!segment(field)) {
272 size_t n = strlen(field);
273 if (n < 3 || *field != '{' || field[n - 1] != '}') return false;
274 char *inside = slice(field + 1, n - 2);
275 p.kind = 'U';
276 p.text = inside;
277 if (*inside == '*') {
278 p.kind = 'W';
279 p.text++;
280 if (i + 1 != count) return false;
281 } else {
282 char *colon = strchr(inside, ':');
283 if (colon) {
284 p.text = slice(inside, (size_t)(colon - inside));
285 if (!strcmp(colon + 1, "int")) p.kind = 'I';
286 else if (!strcmp(colon + 1, "slug")) p.kind = 'S';
287 else return false;
288 }
289 }
290 if (!valid_name(p.text)) return false;
291 for (size_t j = 0; j < r->count; j++) if (r->pieces[j].kind != 'L' && !strcmp(r->pieces[j].text, p.text)) return false;
292 }
293 if (p.kind == 'L') r->literals++;
294 if (p.kind == 'I' || p.kind == 'S') r->typed++;
295 if (p.kind == 'W') r->wildcard = true;
296 APPEND(r->pieces, r->count, p);
297 }
298 return true;
299}

306static void configure(char **a, size_t argc) {
307 char *command = a[0];
308 size_t n = argc - 1;
309 if (strcmp(command, "ROUTE") && strcmp(command, "TOKEN") && strcmp(command, "REDIRECT") && strcmp(command, "ORIGIN") && strcmp(command, "LIMIT") && strcmp(command, "TYPES")) invalid("unknown command %s", command);
310 bool correct = !strcmp(command, "ROUTE") ? n == 3 || n == 5 : !strcmp(command, "TOKEN") || !strcmp(command, "LIMIT") ? n == 2 : !strcmp(command, "REDIRECT") ? n == 3 : !strcmp(command, "ORIGIN") ? n == 1 : n >= 2;
311 if (!correct) invalid("wrong number of arguments");
312 if (!strcmp(command, "ROUTE")) {
313 if (!valid_name(a[1])) invalid("bad route name %s", a[1]);
314 if (find_route(a[1])) invalid("duplicate route %s", a[1]);
315 Route *r = allocate(sizeof(*r));
316 size_t count;
317 char **methods = split(a[2], ',', &count);
318 for (size_t i = 0; i < count; i++) {
319 unsigned bit = method_bit(methods[i]);
320 if (!bit || bit == HEAD || bit == OPTIONS || (r->methods & bit)) invalid("bad methods %s", a[2]);
321 r->methods |= bit;
322 }
323 if (!parse_pattern(r, a[3])) invalid("bad pattern %s", a[3]);
324 if (n == 5 && strcmp(a[4], "ROLE")) invalid("unknown option %s", a[4]);
325 if (n == 5 && !matches(a[5], "[a-z]{1,16}")) invalid("bad role %s", a[5]);
326 r->name = a[1];
327 r->pattern = a[3];
328 r->role = n == 5 ? a[5] : "none";
329 r->number = (int)route_count;
330 APPEND(routes, route_count, r);
331 } else if (!strcmp(command, "TOKEN")) {
332 if (!matches(a[1], "[A-Za-z0-9._-]{1,32}")) invalid("bad token %s", a[1]);
333 if (pair_get(tokens, a[1])) invalid("duplicate token %s", a[1]);
334 if (!matches(a[2], "[a-z]{1,16}")) invalid("bad role %s", a[2]);
335 pair_add(&tokens, a[1], a[2]);
336 } else if (!strcmp(command, "REDIRECT")) {
337 if (!rule_path(a[1])) invalid("bad path %s", a[1]);
338 if (!rule_path(a[2])) invalid("bad path %s", a[2]);
339 bool prefix = ends(a[1], "/*");
340 if (prefix != ends(a[2], "/*")) invalid("wildcard mismatch");
341 if (strcmp(a[3], "301") && strcmp(a[3], "302") && strcmp(a[3], "307") && strcmp(a[3], "308")) invalid("bad redirect code %s", a[3]);
342 char *from = base(a[1]);
343 for (size_t i = 0; i < redirect_count; i++) if (redirects[i]->prefix == prefix && !strcmp(redirects[i]->from, from)) invalid("duplicate redirect %s", a[1]);
344 Redirect *r = allocate(sizeof(*r));
345 *r = (Redirect){from, base(a[2]), prefix, atoi(a[3])};
346 APPEND(redirects, redirect_count, r);
347 } else if (!strcmp(command, "ORIGIN")) {
348 if (!matches(a[1], "https?://[a-z0-9.:-]+")) invalid("bad origin %s", a[1]);
349 for (size_t i = 0; i < origin_count; i++) if (!strcmp(origins[i], a[1])) invalid("duplicate origin %s", a[1]);
350 APPEND(origins, origin_count, a[1]);
351 } else if (!strcmp(command, "LIMIT")) {
352 if (!decimal(a[1], 100) || !decimal(a[2], 3600)) invalid("bad limit");
353 if (capacity) invalid("duplicate limit");
354 capacity = atoi(a[1]);
355 period = atoi(a[2]);
356 } else {
357 Route *r = find_route(a[1]);
358 if (!r) invalid("unknown route %s", a[1]);
359 if (r->types) invalid("duplicate types %s", a[1]);
360 for (size_t i = 2; i < argc; i++) {
361 if (!media(a[i])) invalid("bad media type %s", a[i]);
362 for (size_t j = 2; j < i; j++) if (!strcmp(a[i], a[j])) invalid("duplicate media type %s", a[i]);
363 }
364 for (size_t i = 2; i < argc; i++) APPEND(r->types, r->type_count, a[i]);
365 }
366}

368static int route_compare(const void *a, const void *b) {
369 const Route *x = *(Route *const *)a, *y = *(Route *const *)b;
370 if (x->literals != y->literals) return y->literals - x->literals;
371 if (x->typed != y->typed) return y->typed - x->typed;
372 if (x->wildcard != y->wildcard) return x->wildcard ? 1 : -1;
373 return x->number - y->number;
374}

414static char *normalize(const char *path) {
415 Buffer text = {0};
416 for (size_t i = 0; path[i]; i++) if (path[i] == '%') {
417 char c = (char)(hex(path[i + 1]) * 16 + hex(path[i + 2]));
418 if (unreserved(c)) character(&text, c);
419 else append(&text, format("%%%02X", (unsigned char)c));
420 i += 2;
421 } else character(&text, path[i]);
422 size_t count, used = 0;
423 char **parts = split(finish(&text) + 1, '/', &count), **segments = NULL;
424 for (size_t i = 0; i < count; i++) if (!strcmp(parts[i], "..")) {
425 if (used) used--;
426 } else if (*parts[i] && strcmp(parts[i], ".")) APPEND(segments, used, parts[i]);
427 return format("/%s", join(segments, used, "/"));
428}

440static char *negotiate(Route *r, const char *accept) {
441 if (!accept) return r->types[0];
442 char *selected = NULL;
443 int best_weight = 0;
444 size_t count;
445 char **elements = split(accept, ',', &count);
446 for (size_t t = 0; t < r->type_count; t++) {
447 int specificity = 0, weight = 0;
448 for (size_t i = 0; i < count; i++) {
449 size_t fields;
450 char **parts = split(elements[i], ';', &fields);
451 char *range = lower(trim(parts[0]));
452 int spec = !strcmp(range, "*/*") ? 1 : matches(range, "[a-z0-9.+-]+/\\*") ? 2 : media(range) ? 3 : 0;
453 if (!spec) continue;
454 char *q = NULL;
455 for (size_t j = 1; j < fields; j++) {
456 char *eq = strchr(parts[j], '=');
457 if (eq && !strcmp(lower(trim(slice(parts[j], (size_t)(eq - parts[j])))), "q")) {
458 q = trim(eq + 1);
459 break;
460 }
461 }
462 int value = 1000;
463 if (q) {
464 if (!matches(q, "[01](\\.[0-9]{0,3})?")) continue;
465 bool valid = true;
466 for (size_t j = 2; q[1] && q[j]; j++) if (q[0] == '1' && q[j] != '0') valid = false;
467 if (!valid) continue;
468 value = (q[0] - '0') * 1000;
469 if (q[1]) {
470 int factor = 100;
471 for (size_t j = 2; q[j]; j++, factor /= 10) value += (q[j] - '0') * factor;
472 }
473 }
474 bool fits = spec == 1 || (spec == 2 && starts(r->types[t], slice(range, strlen(range) - 1))) || (spec == 3 && !strcmp(range, r->types[t]));
475 if (fits && spec > specificity) {
476 specificity = spec;
477 weight = value;
478 }
479 }
480 if (weight > best_weight) {
481 best_weight = weight;
482 selected = r->types[t];
483 }
484 }
485 return selected;
486}

537static char *body(Route *r, char **segments, size_t segment_count, const char *query, Pairs headers, const char *type) {
538 Pairs parameters = {0}, queries = {0}, cookies = {0};
539 for (size_t i = 0; i < r->count; i++) {
540 Piece p = r->pieces[i];
541 if (p.kind == 'L') continue;
542 char *value;
543 if (p.kind == 'W') {
544 Buffer b = {0};
545 for (size_t j = i; j < segment_count; j++) {
546 if (j > i) character(&b, '/');
547 append(&b, decode(segments[j]));
548 }
549 value = finish(&b);
550 } else value = p.kind == 'I' ? format("%d", atoi(segments[i])) : p.kind == 'U' ? decode(segments[i]) : segments[i];
551 pair_add(&parameters, p.text, value);
552 }
553 if (parameters.count) qsort(parameters.items, parameters.count, sizeof(*parameters.items), pair_compare);
554 if (query) {
555 size_t count;
556 char **parts = split(query, '&', &count);
557 for (size_t i = 0; i < count; i++) if (*parts[i]) {
558 char *eq = strchr(parts[i], '=');
559 pair_add(&queries, query_decode(eq ? slice(parts[i], (size_t)(eq - parts[i])) : parts[i]), query_decode(eq ? eq + 1 : ""));
560 }
561 }
562 char *cookie = pair_get(headers, "cookie");
563 if (cookie) {
564 size_t count;
565 char **parts = split(cookie, ';', &count);
566 for (size_t i = 0; i < count; i++) {
567 char *part = trim(parts[i]), *eq = strchr(part, '=');
568 if (eq && eq != part) {
569 char *name = slice(part, (size_t)(eq - part));
570 if (pair_index(cookies, name) < 0) pair_add(&cookies, name, eq + 1);
571 }
572 }
573 }
574 if (cookies.count) qsort(cookies.items, cookies.count, sizeof(*cookies.items), pair_compare);
575 if (!strcmp(type, "application/json")) return format("{\"route\":%s,\"params\":%s,\"query\":%s,\"cookies\":%s}", json(r->name), json_pairs(parameters, false), json_pairs(queries, true), json_pairs(cookies, false));
576 if (!strcmp(type, "text/html")) {
577 Buffer b = {0};
578 append(&b, format("<dl><dt>route</dt><dd>%s</dd>", html(r->name)));
579 Pairs groups[] = {parameters, queries, cookies};
580 char *labels[] = {"param", "query", "cookie"};
581 for (int i = 0; i < 3; i++) for (size_t j = 0; j < groups[i].count; j++) {
582 Pair p = groups[i].items[j];
583 append(&b, format("<dt>%s %s</dt><dd>%s</dd>", labels[i], html(p.name), html(p.value)));
584 }
585 append(&b, "</dl>");
586 return finish(&b);
587 }
588 return format("route=%s params=%s query=%s cookies=%s", r->name, encoded_pairs(parameters), encoded_pairs(queries), encoded_pairs(cookies));
589}

591static char *etag(const char *s) {
592 unsigned a = 1, b = 0;
593 for (; *s; s++) {
594 a = (a + (unsigned char)*s) % 65521;
595 b = (b + a) % 65521;
596 }
597 return format("\"%04x%04x\"", b, a);
598}

613static Response *process(Request *request) {
614 size_t count;
615 char **line = split(request->line, ' ', &count);
616 if (count != 3 || !matches(line[0], "[A-Z]{1,16}") || *line[1] != '/' || strchr(line[1], '#') || strcmp(line[2], "HTTP/1.1")) return response(400, "bad request: request line");
617 char *method = line[0], *target = line[1];
618 request->head = !strcmp(method, "HEAD");
619 for (size_t i = 0; target[i]; i++) if (target[i] == '%') {
620 if (!target[i + 1] || !target[i + 2] || hex(target[i + 1]) < 0 || hex(target[i + 2]) < 0) return response(400, "bad request: percent escape");
621 int value = hex(target[i + 1]) * 16 + hex(target[i + 2]);
622 if (value < 32 || value > 126) return response(400, "bad request: percent escape");
623 i += 2;
624 }
625 for (size_t i = 0; i < request->header_count; i++) {
626 char *header = request->header_lines[i], *colon = strchr(header, ':');
627 if (!colon || !matches(slice(header, (size_t)(colon - header)), "[A-Za-z0-9-]+")) return response(400, format("bad request: header line %zu", i + 1));
628 char *name = lower(slice(header, (size_t)(colon - header)));
629 if (pair_index(request->headers, name) >= 0) return response(400, format("bad request: duplicate header %s", name));
630 pair_add(&request->headers, name, trim(colon + 1));
631 }
632 request->headers_valid = true;
633 Pairs headers = request->headers;
634 if (!method_bit(method)) return response(501, format("not implemented: %s", method));
635 char *host = pair_get(headers, "host");
636 if (!host || !*host) return response(400, "bad request: missing host");
637 char *effective = method, *override = pair_get(headers, "x-http-method-override");
638 if (override) {
639 if (strcmp(method, "POST") || (strcmp(override, "PUT") && strcmp(override, "PATCH") && strcmp(override, "DELETE"))) return response(400, "bad request: method override");
640 effective = override;
641 }
642 char *client_name = pair_get(headers, "x-client");
643 if (client_name && *client_name) {
644 int time = atoi(pair_get(headers, "x-time"));
645 Client *client = NULL;
646 for (size_t i = 0; i < client_count; i++) if (!strcmp(clients[i]->name, client_name)) {
647 client = clients[i];
648 break;
649 }
650 if (!client) {
651 client = allocate(sizeof(*client));
652 client->name = client_name;
653 client->level = capacity * period;
654 client->last = time;
655 APPEND(clients, client_count, client);
656 }
657 if (capacity) {
658 int level = client->level + (time - client->last) * capacity;
659 client->level = level < capacity * period ? level : capacity * period;
660 client->last = time;
661 if (client->level < period) {
662 client->limited++;
663 return with_header(response(429, format("too many requests: client %s", client_name)), RETRY_AFTER, format("%d", (period - client->level + capacity - 1) / capacity));
664 }
665 client->level -= period;
666 }
667 client->allowed++;
668 }
669 char *question = strchr(target, '?'), *path = question ? slice(target, (size_t)(question - target)) : target, *query = question ? question + 1 : NULL, *suffix = question ? question : "";
670 char *normal = normalize(path);
671 if (strcmp(normal, path)) {
672 char *location = format("%s%s", normal, suffix);
673 return with_header(response(308, format("redirect to %s", location)), LOCATION, location);
674 }
675 for (size_t i = 0; i < redirect_count; i++) {
676 Redirect *r = redirects[i];
677 if (!strcmp(path, r->from) || (r->prefix && starts(path, format("%s/", r->from)))) {
678 char *location = format("%s%s%s", r->to, path + strlen(r->from), suffix);
679 return with_header(response(r->code, format("redirect to %s", location)), LOCATION, location);
680 }
681 }
682 size_t segment_count = 0;
683 char **segments = !strcmp(path, "/") ? NULL : split(path + 1, '/', &segment_count);
684 unsigned allowed = OPTIONS;
685 Route *selected = NULL;
686 bool any = false;
687 char *lookup = !strcmp(effective, "HEAD") ? "GET" : effective;
688 for (size_t i = 0; i < route_count; i++) if (route_matches(priority[i], segments, segment_count)) {
689 any = true;
690 allowed |= priority[i]->methods;
691 if (!selected && (priority[i]->methods & method_bit(lookup))) selected = priority[i];
692 }
693 if (!any) return response(404, format("not found: %s", path));
694 if (allowed & GET) allowed |= HEAD;
695 char *allow = method_list(allowed, ", ");
696 if (!strcmp(effective, "OPTIONS")) return with_header(response(204, ""), ALLOW, allow);
697 if (!selected) return with_header(response(405, format("method not allowed: %s", effective)), ALLOW, allow);
698 selected->selections++;
699 if (strcmp(selected->role, "none")) {
700 char *auth = pair_get(headers, "authorization");
701 if (!auth) return with_header(response(401, "unauthorized: missing token"), AUTHENTICATE, "Bearer realm=\"api\"");
702 char *space = strchr(auth, ' '), *scheme = space ? slice(auth, (size_t)(space - auth)) : auth, *token = space ? space + 1 : "", *role = pair_get(tokens, token);
703 if (strcmp(lower(scheme), "bearer") || !role) return with_header(response(401, "unauthorized: invalid token"), AUTHENTICATE, "Bearer realm=\"api\", error=\"invalid_token\"");
704 if (strcmp(role, selected->role) && strcmp(role, "admin")) return response(403, format("forbidden: requires role %s", selected->role));
705 }
706 char *type = negotiate(selected, pair_get(headers, "accept"));
707 Response *result;
708 if (!type) result = response(406, format("not acceptable: %s", join(selected->types, selected->type_count, ", ")));
709 else {
710 char *text = body(selected, segments, segment_count, query, headers, type), *tag = etag(text), *condition = pair_get(headers, "if-none-match");
711 bool matched = false;
712 if (condition) {
713 size_t n;
714 char **parts = split(condition, ',', &n);
715 for (size_t i = 0; i < n; i++) {
716 char *value = trim(parts[i]);
717 if (starts(value, "W/")) value += 2;
718 if (!strcmp(value, "*") || !strcmp(value, tag)) matched = true;
719 }
720 }
721 if (matched && !strcmp(lookup, "GET")) result = with_header(response(304, ""), ETAG, tag);
722 else if (matched) result = response(412, format("precondition failed: %s", tag));
723 else {
724 result = with_header(response(200, text), ETAG, tag);
725 result->type = type;
726 }
727 }
728 if (selected->type_count >= 2) result->headers[VARY] = "Accept";
729 return result;
730}

732static char *phrase(int status) {
733 switch (status) {
734 case 200: return "OK";
735 case 204: return "No Content";
736 case 301: return "Moved Permanently";
737 case 302: return "Found";
738 case 304: return "Not Modified";
739 case 307: return "Temporary Redirect";
740 case 308: return "Permanent Redirect";
741 case 400: return "Bad Request";
742 case 401: return "Unauthorized";
743 case 403: return "Forbidden";
744 case 404: return "Not Found";
745 case 405: return "Method Not Allowed";
746 case 406: return "Not Acceptable";
747 case 412: return "Precondition Failed";
748 case 429: return "Too Many Requests";
749 default: return "Not Implemented";
750 }
751}

753static void respond(Request request) {
754 Response *r = process(&request);
755 char *origin = pair_get(request.headers, "origin");
756 bool permitted = false;
757 if (origin) for (size_t i = 0; i < origin_count; i++) if (!strcmp(origins[i], origin)) permitted = true;
758 if (request.headers_valid && permitted) {
759 r->headers[CORS_ORIGIN] = origin;
760 if (r->status == 204 && pair_get(request.headers, "access-control-request-method")) r->headers[CORS_METHODS] = r->headers[ALLOW];
761 }
762 if (r->status != 204 && r->status != 304) {
763 r->headers[CONTENT_TYPE] = r->status == 200 ? r->type : "text/plain";
764 r->headers[CONTENT_LENGTH] = format("%zu", strlen(r->body));
765 }
766 statuses[r->status]++;
767 printf("REQUEST %d\nHTTP/1.1 %d %s\n", ++requests, r->status, phrase(r->status));
768 for (int i = 0; i < HEADER_COUNT; i++) if (r->headers[i]) printf("%s: %s\n", header_names[i], r->headers[i]);
769 if (r->status != 204 && r->status != 304 && !request.head) puts(r->body);
770}

772static bool valid_parameter(Piece p, const char *value) {
773 if (p.kind == 'I') return matches(value, "[0-9]{1,9}");
774 if (p.kind == 'S') return matches(value, "[a-z0-9-]+");
775 if (p.kind == 'W' && *value) {
776 size_t count;
777 char **parts = split(value, '/', &count);
778 for (size_t i = 0; i < count; i++) if (!*parts[i] || !strcmp(parts[i], ".") || !strcmp(parts[i], "..")) return false;
779 return true;
780 }
781 return p.kind == 'W' || (*value && strcmp(value, ".") && strcmp(value, ".."));
782}

785 for (size_t i = 0; i < r->count; i++) if (r->pieces[i].kind != 'L' && !strcmp(r->pieces[i].text, name)) return true;
786 return false;

789static void urlfor(char **a, size_t argc) {
790 if (setjmp(failure)) {
791 url_errors++;
792 printf("URLFOR ERROR %s\n", error_message);
793 return;
794 }
795 Route *r = find_route(a[1]);
796 if (!r) invalid("unknown route %s", a[1]);
797 Pairs parameters = {0}, query = {0};
798 for (size_t i = 2; i < argc; i++) {
799 char *eq = strchr(a[i], '=');
800 if (!eq || eq == a[i]) invalid("bad argument %s", a[i]);
801 char *name = slice(a[i], (size_t)(eq - a[i]));
802 if (parameter_name(r, name)) {
803 if (pair_index(parameters, name) >= 0) invalid("duplicate parameter %s", name);
804 pair_add(&parameters, name, eq + 1);
805 } else pair_add(&query, name, eq + 1);
806 }
807 for (size_t i = 0; i < r->count; i++) {
808 Piece p = r->pieces[i];
809 if (p.kind == 'L') continue;
810 char *value = pair_get(parameters, p.text);
811 if (!value && p.kind != 'W') invalid("missing parameter %s", p.text);
812 if (!valid_parameter(p, value ? value : "")) invalid("bad value for %s", p.text);
813 }
814 Buffer path = {0};
815 for (size_t i = 0; i < r->count; i++) {
816 Piece p = r->pieces[i];
817 char *value = p.kind == 'L' ? p.text : pair_get(parameters, p.text);
818 if (p.kind == 'W') {
819 if (value && *value) {
820 size_t count;
821 char **parts = split(value, '/', &count);
822 for (size_t j = 0; j < count; j++) {
823 character(&path, '/');
824 append(&path, encode(parts[j]));
825 }
826 }
827 } else {
828 character(&path, '/');
829 append(&path, p.kind == 'L' ? value : encode(value));
830 }
831 }
832 if (!path.length) character(&path, '/');
833 if (query.count) {
834 character(&path, '?');
835 append(&path, encoded_pairs(query));
836 }
837 url_ok++;
838 printf("URLFOR %s %s\n", a[1], finish(&path));
839}

841static char *read_line(void) {
842 Buffer b = {0};
843 int c;
844 while ((c = getchar()) != EOF && c != '\n') character(&b, (char)c);
845 if (c == EOF && !b.length) return NULL;
846 return finish(&b);
847}
