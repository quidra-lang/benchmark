55 static List<Piece>? ParsePattern(string text)
56 {
57 if (!text.StartsWith('/')) return null;
58 if (text == "/") return [];
59 string[] fields = text[1..].Split('/');
60 List<Piece> pieces = []; HashSet<string> names = new(StringComparer.Ordinal);
61 for (int i = 0; i < fields.Length; i++)
62 {
63 string field = fields[i];
64 if (Segment(field)) { pieces.Add(new('L', field)); continue; }
65 Match match = Regex.Match(field, @"\A\{(\*?)(" + NamePattern + @")(?::(int|slug))?\}\z");
66 if (!match.Success || match.Groups[1].Length > 0 && match.Groups[3].Success) return null;
67 char kind = match.Groups[1].Length > 0 ? 'W' : !match.Groups[3].Success ? 'U' : match.Groups[3].Value == "int" ? 'I' : 'S';
68 if (kind == 'W' && i != fields.Length - 1 || !names.Add(match.Groups[2].Value)) return null;
69 pieces.Add(new(kind, match.Groups[2].Value));
70 }
71 return pieces;
72 }

74 static void Configure(string[] args)
75 {

108 if (Origins.Contains(args[1])) throw new Invalid("duplicate origin " + args[1]);
109 Origins.Add(args[1]); break;
110 case "LIMIT":
111 if (!Decimal(args[1], 100) || !Decimal(args[2], 3600)) throw new Invalid("bad limit");
112 if (capacity != 0) throw new Invalid("duplicate limit");

117 if (route.Types is not null) throw new Invalid("duplicate types " + args[1]);

122 if (types.Contains(type)) throw new Invalid("duplicate media type " + type);
123 types.Add(type);

126 }

171 static bool RouteMatches(Route route, string[] segments)
172 {
173 int count = route.Pieces.Count - (route.Wildcard ? 1 : 0);
174 if (route.Wildcard ? segments.Length < count : segments.Length != count) return false;
175 for (int i = 0; i < count; i++)
176 {
177 Piece piece = route.Pieces[i]; string value = segments[i];
178 if (piece.Kind == 'L' && piece.Text != value || piece.Kind == 'I' && !Matches(value, "[0-9]{1,9}") || piece.Kind == 'S' && !Matches(value, "[a-z0-9-]+")) return false;
179 }

182 static string? Negotiate(List<string> types, string? accept)
183 {
184 if (accept is null) return types[0];
185 string? selected = null; int bestWeight = 0;
186 foreach (string type in types)
187 {
188 int specificity = 0, weight = 0;
189 foreach (string element in accept.Split(','))
190 {
191 string[] parts = element.Split(';'); string range = parts[0].Trim(' ').ToLowerInvariant();
192 int spec = range == "*/*" ? 1 : Matches(range, @"[a-z0-9.+-]+/\*") ? 2 : Matches(range, MediaPattern) ? 3 : 0;
193 if (spec == 0) continue;
194 string? q = null;
195 foreach (string parameter in parts.Skip(1))
196 {
197 int equal = parameter.IndexOf('=');
198 if (equal >= 0 && parameter[..equal].Trim(' ').ToLowerInvariant() == "q") { q = parameter[(equal + 1)..].Trim(' '); break; }
199 }
200 if (q is not null && !Matches(q, @"(?:0(?:\.[0-9]{0,3})?|1(?:\.0{0,3})?)")) continue;
201 int value = 1000;
202 if (q is not null)
203 {
204 string digits = q.Contains('.') ? q[(q.IndexOf('.') + 1)..] : "";
205 value = (q[0] - '0') * 1000 + int.Parse((digits + "000")[..3]);
206 }
207 bool fits = spec == 1 || spec == 2 && type.StartsWith(range[..^1], StringComparison.Ordinal) || spec == 3 && type == range;
208 if (fits && spec > specificity) { specificity = spec; weight = value; }
209 }
210 if (weight > bestWeight) { bestWeight = weight; selected = type; }
211 }
212 return selected;
213 }

253 static string Etag(string body)
254 {

256 foreach (char c in body) { a = (a + c) % 65521; b = (b + a) % 65521; }

258 }
259 static Response Process(Request request)
260 {
261 string[] line = request.Line.Split(' ');
262 if (line.Length != 3 || !Matches(line[0], "[A-Z]{1,16}") || !line[1].StartsWith('/') || line[1].Contains('#') || line[2] != "HTTP/1.1") return new(400, "bad request: request line");
263 string method = line[0], target = line[1]; request.Head = method == "HEAD";
264 for (int i = 0; i < target.Length; i++) if (target[i] == '%')
265 {
266 if (i + 2 >= target.Length || !Matches(target.Substring(i + 1, 2), "[0-9a-fA-F]{2}")) return new(400, "bad request: percent escape");
267 int value = int.Parse(target.AsSpan(i + 1, 2), NumberStyles.HexNumber, CultureInfo.InvariantCulture);
268 if (value is < 32 or > 126) return new(400, "bad request: percent escape"); i += 2;
269 }
270 var headers = request.Headers;
271 int number = 0;
272 foreach (string header in request.HeaderLines)
273 {
274 number++; int colon = header.IndexOf(':');
275 if (colon < 0 || !Matches(header[..colon], "[A-Za-z0-9-]+")) return new(400, "bad request: header line " + number);
276 string name = header[..colon].ToLowerInvariant();
277 if (headers.ContainsKey(name)) return new(400, "bad request: duplicate header " + name);
278 headers[name] = header[(colon + 1)..].Trim(' ');
279 }
280 request.HeadersValid = true;
281 if (!RequestMethods.Contains(method)) return new(501, "not implemented: " + method);
282 if (headers.GetValueOrDefault("host", "").Length == 0) return new(400, "bad request: missing host");
283 string effective = method;
284 if (headers.TryGetValue("x-http-method-override", out string? overridden))
285 {
286 if (method != "POST" || overridden is not ("PUT" or "PATCH" or "DELETE")) return new(400, "bad request: method override");
287 effective = overridden;
288 }
289 string clientName = headers.GetValueOrDefault("x-client", "");
290 if (clientName.Length > 0)
291 {
292 int time = int.Parse(headers["x-time"]);
293 if (!Clients.TryGetValue(clientName, out Client? client)) { client = new() { Level = capacity * period, Last = time }; Clients[clientName] = client; }
294 if (capacity > 0)
295 {
296 client.Level = Math.Min(capacity * period, client.Level + (time - client.Last) * capacity); client.Last = time;
297 if (client.Level < period) { client.Limited++; return new Response(429, "too many requests: client " + clientName).Header("Retry-After", ((period - client.Level + capacity - 1) / capacity).ToString(CultureInfo.InvariantCulture)); }
298 client.Level -= period;
299 }
300 client.Allowed++;
301 }
302 int question = target.IndexOf('?');
303 string path = question < 0 ? target : target[..question], suffix = question < 0 ? "" : target[question..];
304 string? query = question < 0 ? null : target[(question + 1)..];
305 string normalized = Normalize(path);
306 if (normalized != path) { string location = normalized + suffix; return new Response(308, "redirect to " + location).Header("Location", location); }
307 foreach (Redirect rule in Redirects)
308 if (path == rule.From || rule.Prefix && path.StartsWith(rule.From + "/", StringComparison.Ordinal))
309 { string location = rule.To + path[rule.From.Length..] + suffix; return new Response(rule.Code, "redirect to " + location).Header("Location", location); }
310 string[] segments = path == "/" ? [] : path[1..].Split('/');
311 List<Route> matching = priority.Where(matchedRoute => RouteMatches(matchedRoute, segments)).ToList();
312 if (matching.Count == 0) return new(404, "not found: " + path);
313 SortedSet<string> allowed = new(StringComparer.Ordinal) { "OPTIONS" };
314 foreach (Route candidate in matching) allowed.UnionWith(candidate.Methods);
315 if (allowed.Contains("GET")) allowed.Add("HEAD");
316 string allow = string.Join(", ", allowed);
317 if (effective == "OPTIONS") return new Response(204, "").Header("Allow", allow);
318 string lookup = effective == "HEAD" ? "GET" : effective;
319 Route? route = matching.Find(candidate => candidate.Methods.Contains(lookup));
320 if (route is null) return new Response(405, "method not allowed: " + effective).Header("Allow", allow);
321 route.Selections++;
322 if (route.Role != "none")
323 {
324 if (!headers.TryGetValue("authorization", out string? authorization)) return new Response(401, "unauthorized: missing token").Header("WWW-Authenticate", "Bearer realm=\"api\"");
325 int space = authorization.IndexOf(' '); string scheme = space < 0 ? authorization : authorization[..space], token = space < 0 ? "" : authorization[(space + 1)..];
326 if (scheme.ToLowerInvariant() != "bearer" || !Tokens.TryGetValue(token, out string? role)) return new Response(401, "unauthorized: invalid token").Header("WWW-Authenticate", "Bearer realm=\"api\", error=\"invalid_token\"");
327 if (role != route.Role && role != "admin") return new(403, "forbidden: requires role " + route.Role);
328 }
329 string? type = Negotiate(route.Types!, headers.GetValueOrDefault("accept"));
330 Response response;
331 if (type is null) response = new(406, "not acceptable: " + string.Join(", ", route.Types!));
332 else
333 {
334 string body = Body(route, segments, query, headers, type), tag = Etag(body);
335 bool matched = false;
336 if (headers.TryGetValue("if-none-match", out string? condition))
337 foreach (string item in condition.Split(','))
338 {
339 string value = item.Trim(' '); if (value.StartsWith("W/", StringComparison.Ordinal)) value = value[2..];
340 if (value == "*" || value == tag) matched = true;
341 }
342 if (matched && lookup == "GET") response = new Response(304, "").Header("ETag", tag);
343 else if (matched) response = new(412, "precondition failed: " + tag);
344 else { response = new Response(200, body).Header("ETag", tag); response.Type = type; }
345 }
346 if (route.Types!.Count >= 2) response.Header("Vary", "Accept");
347 return response;
348 }

398 }
