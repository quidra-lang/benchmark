17number_fits <- function(text, low, high) {
18 negative <- startsWith(text, "-")
19 digits <- if (negative) substring(text, 2) else text
20 digits <- sub("^0+", "", digits)
21 if (digits == "") return(TRUE)
22 limit <- if (negative) -low else high
23 if (limit < 0) return(FALSE)
24 bound <- decimal(limit)
25 nchar(digits) < nchar(bound) ||
26 (nchar(digits) == nchar(bound) && digits <= bound)
27}

69base64_text <- function(data) {
70 alphabet <- strsplit("ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/", "")[[1]]
71 result <- character()
72 if (length(data)) for (i in seq.int(1L, length(data), by = 3L)) {
73 size <- min(3L, length(data) - i + 1L)
74 block <- c(data[seq.int(i, i + size - 1L)], rep(0, 3L - size))
75 n <- block[1] * 65536 + block[2] * 256 + block[3]

77 floor(n / 64) %% 64, n %% 64) + 1L
78 chars <- alphabet[indices]
79 if (size < 3L) chars[seq.int(size + 2L, 4L)] <- "="
80 result <- c(result, chars)
81 }
82 paste(result, collapse = "")

248read_wire <- function(type, data, explaining = FALSE) {
249 lines <- character()
250 slice <- function(start, end) if (end > start) data[seq.int(start + 1L, end)] else integer()
251 read_varint <- function(start, end) {
252 value <- 0
253 position <- start
254 for (i in seq_len(8L)) {
255 if (position >= end) byte_error("truncated varint", start)
256 b <- data[position + 1L]
257 position <- position + 1L
258 if (i == 8L) {
259 if (b >= 128) byte_error("varint too long", start)
260 if (b >= 16) byte_error("varint too large", start)
261 }
262 value <- value + (b %% 128) * 128^(i - 1L)
263 if (b < 128) return(list(value = value, position = position))
264 }
265 }
266 read_fixed <- function(start, end, wire) {
267 size <- if (wire == 1) 8L else 4L
268 if (end - start < size) byte_error(paste("truncated", wire_names[decimal(wire)]), start)
269 list(value = slice(start, start + size), position = start + size)
270 }
271 scalar <- function(f, raw, position) {
272 if (f$kind == "string") {
273 bad <- which(raw >= 128)
274 if (length(bad)) byte_error("invalid string", position + bad[1] - 1L)
275 return(raw)
276 }
277 if (f$kind == "bytes") return(raw)
278 if (f$wire != 0) {
279 if (f$wire == 1 && (raw[8] != 0 || raw[7] >= 32)) byte_error("value out of range", position)
280 raw <- sum(raw * 256^(seq_along(raw) - 1L))
281 }
282 if (f$kind == "bool") {
283 if (raw > 1) byte_error("value out of range", position)
284 return(raw == 1)
285 }
286 if (f$kind == "enum") {
287 if (!(raw %in% enums[[f$type]])) byte_error("unknown enum number", position)
288 return(raw)
289 }
290 value <- raw
291 if (f$type %in% c("sint32", "sint64")) value <- unzigzag(raw)
292 if (f$type == "sfixed32" && value >= 2147483648) value <- value - 4294967296
293 bounds <- ranges[[f$type]]
294 if (value < bounds[1] || value > bounds[2]) byte_error("value out of range", position)
295 value
296 }
297 read_message <- function(type, start, end, depth) {
298 m <- new_message(type)
299 position <- start
300 while (position < end) {
301 record_start <- position
302 tag <- read_varint(position, end)
303 position <- tag$position
304 number <- floor(tag$value / 8)
305 wire <- tag$value %% 8
306 if (number == 0) byte_error("invalid field number", record_start)
307 if (!(wire %in% c(0, 1, 2, 5))) byte_error("invalid wire type", record_start)
308 f <- messages[[type]]$by_number[[decimal(number)]]
309 packed <- !is.null(f) && f$repeated && f$wire != 2 && wire == 2
310 if (!is.null(f) && wire != f$wire && !packed) byte_error("wire type mismatch", record_start)
311 value_start <- position
312 if (wire == 0) {
313 read <- read_varint(position, end)
314 raw <- read$value
315 position <- read$position
316 } else if (wire == 2) {
317 length_start <- position
318 read <- read_varint(position, end)
319 value_start <- read$position
320 if (read$value > end - value_start) byte_error("length past end", length_start)
321 position <- value_start + read$value
322 raw <- slice(value_start, position)
323 } else {
324 read <- read_fixed(position, end, wire)
325 raw <- read$value
326 position <- read$position
327 }
328 name <- if (is.null(f)) "?" else f$name
329 header <- paste0(strrep(" ", depth), decimal(record_start), " ", decimal(number), " ", name, " ", wire_names[decimal(wire)])
330 if (wire == 2) header <- paste(header, length(raw))
331 if (is.null(f)) {
332 u <- list(number = number, wire = wire, raw = raw)
333 m$unknown <- c(m$unknown, list(u))
334 if (explaining) lines <<- c(lines, paste(header, unknown_notation(u)))
335 } else {
336 if (f$kind == "message") {
337 if (explaining) lines <<- c(lines, header)
338 values <- list(read_message(f$type, value_start, position, depth + 1L))
339 } else if (packed) {
340 values <- list()
341 p <- value_start
342 while (p < position) {
343 element_start <- p
344 read <- if (f$wire == 0) read_varint(p, position) else read_fixed(p, position, f$wire)
345 p <- read$position
346 values <- c(values, list(scalar(f, read$value, element_start)))
347 }
348 if (explaining) {
349 notations <- vapply(values, function(v) value_notation(f, v), character(1))
350 lines <<- c(lines, paste(c(header, "packed", notations), collapse = " "))
351 }
352 } else {
353 values <- list(scalar(f, raw, value_start))
354 if (explaining) lines <<- c(lines, paste(header, value_notation(f, values[[1]])))
355 }
356 if (!explaining) m <- apply_values(m, f, values)
357 }
358 }
359 m
360 }
361 m <- read_message(type, 0, length(data), 0L)
362 if (explaining) return(if (length(lines)) lines else "(empty)")
363 check_required(m)
364 m
365}

367tokenize <- function(text) {
368 chars <- strsplit(text, "", fixed = TRUE)[[1]]
369 n <- length(chars)
370 result <- list()
371 p <- 1L
372 add <- function(kind, column, text = "", value = NULL) {
373 result[[length(result) + 1L]] <<- list(kind = kind, column = column, text = text, value = value)
374 }
375 while (p <= n) {
376 ch <- chars[p]
377 start <- p
378 if (ch == " ") {
379 p <- p + 1L
380 } else if (ch %in% c("{", "}", "[", "]", ":", ",")) {
381 add("symbol", p, ch)
382 p <- p + 1L
383 } else if (ch == '"') {
384 p <- p + 1L
385 bytes <- integer()
386 while (p <= n && chars[p] != '"') {
387 if (chars[p] != "\\") {
388 bytes <- c(bytes, utf8ToInt(chars[p]))
389 p <- p + 1L
390 } else {
391 if (p + 1L > n) column_error("syntax error", start)
392 escape <- chars[p + 1L]
393 escapes <- c('"' = 34L, "\\" = 92L, n = 10L, t = 9L)
394 if (escape %in% names(escapes)) {
395 bytes <- c(bytes, unname(escapes[escape]))
396 p <- p + 2L
397 } else if (escape == "x" && p + 3L <= n) {
398 code <- paste(chars[seq.int(p + 2L, p + 3L)], collapse = "")
399 if (!grepl("^[0-7][0-9a-f]$", code)) column_error("syntax error", start)
400 bytes <- c(bytes, strtoi(code, 16L))
401 p <- p + 4L
402 } else column_error("syntax error", start)
403 }
404 }
405 if (p > n) column_error("syntax error", start)
406 add("string", start, value = bytes)
407 p <- p + 1L
408 } else if (ch == "<") {
409 p <- p + 1L
410 first <- p
411 while (p <= n && chars[p] != ">") p <- p + 1L
412 if (p > n) column_error("syntax error", start)
413 code <- if (p > first) paste(chars[seq.int(first, p - 1L)], collapse = "") else ""
414 if (!grepl("^([0-9a-f]{2})*$", code)) column_error("syntax error", start)
415 add("bytes", start, value = hex_bytes(code))
416 p <- p + 1L
417 } else if (ch == "-" || grepl("^[0-9]$", ch)) {
418 if (ch == "-") p <- p + 1L
419 first <- p
420 while (p <= n && grepl("^[0-9]$", chars[p])) p <- p + 1L
421 if (p == first) column_error("syntax error", start)
422 add("integer", start, paste(chars[seq.int(start, p - 1L)], collapse = ""))
423 } else if (grepl("^[A-Za-z_]$", ch)) {
424 p <- p + 1L
425 while (p <= n && grepl("^[A-Za-z0-9_]$", chars[p])) p <- p + 1L
426 add("name", start, paste(chars[seq.int(start, p - 1L)], collapse = ""))
427 } else column_error("syntax error", p)
428 }
429 add("end", n + 1L)
430 result
431}

449 if (token$kind == "end") return("")
450 if (token$kind == "name") return(if (token$text %in% c("true", "false")) "boolean" else "name")

453 element <- function(f) {
454 token <- peek()
455 token_shape <- shape(token)
456 needed <- c(integer = "integer", bool = "boolean", enum = "name", string = "string", bytes = "bytes", message = "message")[[f$kind]]
457 if (token_shape == "") syntax()
458 if (token_shape != needed) syntax("type mismatch")
459 if (token_shape == "message") return(message(f$type))
460 take()
461 if (token_shape == "integer") {
462 bounds <- ranges[[f$type]]

464 return(as.numeric(token$text))
465 }
466 if (token_shape == "boolean") return(token$text == "true")
467 if (token_shape == "name") {

469 return(unname(enums[[f$type]][token$text]))

471 token$value
472 }
473 field_value <- function(f) {
474 if (!f$repeated) return(element(f))
475 token_shape <- shape(peek())
476 if (token_shape == "") syntax()
477 if (token_shape != "list") syntax("type mismatch")
478 take()
479 values <- list()
480 if (!at("]")) {
481 values <- list(element(f))
482 while (at(",")) {
483 take()

490 message <- function(type) {
491 expect("{")
492 m <- new_message(type)
493 seen <- character()
494 groups <- character()
495 if (!at("}")) repeat {
496 token <- peek()
497 if (token$kind != "name") syntax()

499 if (is.null(f)) syntax("unknown field")
500 if (f$name %in% seen) syntax("duplicate field")
501 if (nzchar(f$group) && f$group %in% groups) syntax("oneof conflict")
502 take()
503 seen <- c(seen, f$name)
504 if (nzchar(f$group)) groups <- c(groups, f$group)
505 expect(":")
506 value <- field_value(f)

508 if (!at(",")) break
509 take()
510 }
511 closing <- expect("}")
512 for (f in messages[[type]]$fields) if (f$required && !(f$name %in% seen)) column_error(paste("missing required field", f$name), closing$column)
513 m
514 }
515 value <- if (!is.null(type)) message(type) else if (whole) field_value(field) else element(field)
516 if (peek()$kind != "end") syntax()
517 value

521 if (!grepl("^[a-z][a-z0-9_]*(\\[[0-9]{1,6}\\])?(\\.[a-z][a-z0-9_]*(\\[[0-9]{1,6}\\])?)*$", text)) fail("invalid path")

528 if (is.null(f) || (!is.null(index) && !f$repeated)) fail("invalid path")
529 result[[i]] <- list(field = f, index = index)
530 if (i < length(parts)) {
531 if (f$kind != "message" || (f$repeated && is.null(index))) fail("invalid path")
532 type <- f$type
533 }
534 }
535 result
536}
537whole_segment <- function(segment) segment$field$repeated && is.null(segment$index)
538get_path <- function(m, path) {
539 last <- path[[length(path)]]
540 for (segment in head(path, -1L)) {
541 values <- items(m, segment$field)
542 index <- if (is.null(segment$index)) 1L else segment$index + 1L
543 if (index > length(values)) return(if (whole_segment(last)) "[]" else "(none)")
544 m <- values[[index]]
545 }
546 values <- items(m, last$field)
547 if (whole_segment(last)) return(list_notation(last$field, values))
548 index <- if (is.null(last$index)) 1L else last$index + 1L
549 if (index > length(values)) "(none)" else value_notation(last$field, values[[index]])
550}
551change_path <- function(m, path, value = NULL, clear = FALSE, level = 1L) {
552 segment <- path[[level]]
553 f <- segment$field
554 values <- items(m, f)
555 index <- if (is.null(segment$index)) 1L else segment$index + 1L
556 if (level == length(path)) {
557 if (clear) {
558 if (is.null(segment$index)) m$values[[f$name]] <- NULL
559 else if (index <= length(values)) m <- set_items(m, f, values[-index])
560 } else if (whole_segment(segment)) m <- set_items(m, f, value)
561 else if (f$repeated) {
562 if (index > length(values)) fail("index out of range")
563 values[index] <- list(value)
564 m <- set_items(m, f, values)
565 } else {
566 m <- remove_group(m, f)
567 m$values[f$name] <- list(value)
568 }
569 return(m)
570 }
571 if (index > length(values)) {
572 if (clear) return(m)
573 if (f$repeated) fail("index out of range")
574 m <- remove_group(m, f)
575 values <- list(new_message(f$type))
576 }
577 child <- change_path(values[[index]], path, value, clear, level + 1L)
578 if (f$repeated) {
579 values[index] <- list(child)
580 m <- set_items(m, f, values)
581 } else m$values[f$name] <- list(child)
582 m
583}
584compare_messages <- function(a, b, prefix = "") {
585 result <- character()
586 for (f in messages[[a$type]]$fields) {
587 left <- items(a, f)
588 right <- items(b, f)
589 for (i in seq_len(max(length(left), length(right)))) {
590 path <- paste0(prefix, f$name, if (f$repeated) paste0("[", i - 1L, "]") else "")
591 both <- i <= length(left) && i <= length(right)
592 if (both && f$kind == "message") result <- c(result, compare_messages(left[[i]], right[[i]], paste0(path, ".")))
593 else if (!both || !isTRUE(all.equal(left[[i]], right[[i]], tolerance = 0, check.attributes = FALSE))) {
594 old <- if (i <= length(left)) value_notation(f, left[[i]]) else "(none)"
595 new <- if (i <= length(right)) value_notation(f, right[[i]]) else "(none)"
596 result <- c(result, paste0(path, ": ", old, " -> ", new))
597 }
598 }
599 }
600 result
601}
602schema_fields <- function(type, chain, depth) {
603 result <- character()
604 for (f in messages[[type]]$fields) {
605 kind <- if (f$required) "required " else if (f$packed) "repeated packed " else if (f$repeated) "repeated " else if (nzchar(f$group)) paste0("oneof ", f$group, " ") else ""
606 line <- paste0(strrep(" ", depth), decimal(f$number), " ", f$name, ": ", kind, f$type)
607 if (f$kind == "enum") {
608 values <- sort(enums[[f$type]])
609 line <- paste0(line, " (", paste(paste0(names(values), "=", vapply(values, decimal, character(1))), collapse = ", "), ")")
610 }
611 recursive <- f$kind == "message" && f$type %in% chain
612 result <- c(result, paste0(line, if (recursive) " (recursive)" else ""))
613 if (f$kind == "message" && !recursive) result <- c(result, schema_fields(f$type, c(chain, f$type), depth + 1L))
614 }
615 result
616}
617word_counts <- c(VARINT = 2L, ZIGZAG = 2L, SCHEMA = 2L, ENCODE = 3L, SIZE = 3L,
618 DECODE = 3L, LITERAL = 3L, JSON = 3L, CANON = 3L, EXPLAIN = 3L,
619 GET = 4L, PATCH = 5L, CLEAR = 4L, MERGE = 4L, DIFF = 4L)
620text_after <- function(line, separator) {
621 p <- gregexpr(" ", line, fixed = TRUE)[[1]][separator]
622 substring(line, p + 1L)
623}
624execute <- function(line) {
625 words <- strsplit(line, " ", fixed = TRUE)[[1]]
626 command <- words[1]
627 if (!(command %in% names(word_counts))) fail("unknown command")
628 expected <- word_counts[[command]]
629 variable <- command %in% c("ENCODE", "SIZE", "PATCH")
630 if (length(words) < expected || (!variable && length(words) != expected)) fail("wrong number of arguments")
631 if (command %in% c("VARINT", "ZIGZAG")) {
632 bounds <- if (command == "VARINT") ranges$uint64 else ranges$sint64
633 if (!grepl("^-?[0-9]{1,18}$", words[2]) || !number_fits(words[2], bounds[1], bounds[2])) fail("invalid number")
634 value <- as.numeric(words[2])
635 if (command == "VARINT") return(hex_text(varint_bytes(value)))
636 value <- zigzag(value)
637 return(paste(decimal(value), hex_text(varint_bytes(value))))
638 }
639 type <- words[2]
640 if (!(type %in% names(messages))) fail("unknown message type")
641 data <- list()
642 if (!(command %in% c("SCHEMA", "ENCODE", "SIZE"))) {
643 data <- list(parse_hex(words[3]))
644 if (command %in% c("MERGE", "DIFF")) data[[2]] <- parse_hex(words[4])
645 }
646 if (command %in% c("GET", "PATCH", "CLEAR")) path <- parse_path(type, words[4])
647 if (command == "SCHEMA") return(c(type, schema_fields(type, type, 1L)))
648 if (command %in% c("ENCODE", "SIZE")) {
649 m <- parse_literal(text_after(line, 2L), type = type)
650 encoded <- encode_message(m)
651 if (command == "ENCODE") return(hex_text(encoded))
652 parts <- as.character(length(encoded))
653 for (f in messages[[type]]$fields) if (has_field(m, f)) parts <- c(parts, paste0(f$name, "=", length(field_records(f, items(m, f)))))
654 return(paste(parts, collapse = " "))
655 }
656 if (command == "EXPLAIN") return(read_wire(type, data[[1]], TRUE))
657 if (command == "MERGE") return(text_message(read_wire(type, c(data[[1]], data[[2]]))))
658 if (command == "DIFF") {
659 a <- tryCatch(read_wire(type, data[[1]]), command_error = function(e) fail(paste(e$message, "in first")))
660 b <- tryCatch(read_wire(type, data[[2]]), command_error = function(e) fail(paste(e$message, "in second")))
661 differences <- compare_messages(a, b)
662 return(if (length(differences)) differences else "SAME")
663 }
664 m <- read_wire(type, data[[1]])
665 if (command == "DECODE") return(text_message(m))
666 if (command == "LITERAL") return(message_notation(m))
667 if (command == "JSON") return(json_message(m))
668 if (command == "CANON") return(hex_text(encode_message(m)))
669 if (command == "GET") return(get_path(m, path))
670 if (command == "PATCH") {
671 last <- path[[length(path)]]
672 value <- parse_literal(text_after(line, 4L), field = last$field, whole = whole_segment(last))
673 m <- change_path(m, path, value)
674 } else if (command == "CLEAR") m <- change_path(m, path, clear = TRUE)
675 check_required(m)
676 hex_text(encode_message(m))
677}
678count <- as.integer(input[line_index])
679output <- character()
680for (i in seq_len(count)) {
681 line <- input[line_index + i]
682 result <- tryCatch(execute(line), command_error = function(e) paste("ERROR", e$message))
683 output <- c(output, result)
684}
685writeLines(output)
