87const maxUnsigned int64 = 9_007_199_254_740_991
88
89func (t *definition) contains(number int64) bool {
90 switch t.kind {
91 case uint32Kind, fixed32Kind:
92 return number >= 0 && number <= 4_294_967_295
93 case uint64Kind, fixed64Kind:
94 return number >= 0 && number <= maxUnsigned
95 case sint32Kind, sfixed32Kind:
96 return number >= -2_147_483_648 && number <= 2_147_483_647
97 case sint64Kind:
98 return number >= -4_503_599_627_370_496 && number <= 4_503_599_627_370_495
99 default:
100 return true
101 }

103func readSchema(input *bufio.Scanner) schema {
104 types := make(schema)
105 names := []string{"uint32", "uint64", "sint32", "sint64", "bool", "fixed32", "sfixed32", "fixed64", "string", "bytes"}
106 for i, name := range names { types[name] = &definition{name: name, kind: kind(i)} }
107 var current *definition
108 for input.Scan() {
109 if input.Text() == "END SCHEMA" { break }
110 words := strings.Split(input.Text(), " ")
111 switch words[0] {
112 case "ENUM", "MESSAGE":
113 current = &definition{name: words[1], kind: enumKind}
114 if words[0] == "MESSAGE" { current.kind = messageKind }
115 types[current.name] = current
116 case "VALUE":
117 number, _ := strconv.ParseInt(words[2], 10, 64)
118 current.members = append(current.members, enumMember{words[1], number})
119 case "FIELD":
120 number, _ := strconv.ParseInt(words[1], 10, 64)
121 field := &field{number: number, typeName: words[2], name: words[3]}
122 if len(words) > 4 {
123 field.required, field.repeated = words[4] == "REQUIRED", words[4] == "REPEATED"
124 field.packed = field.repeated && len(words) == 6
125 if words[4] == "ONEOF" { field.group = words[5] }
126 }
127 current.fields = append(current.fields, field)
128 }
129 }
130 for _, typ := range types {
131 slices.SortFunc(typ.fields, func(a, b *field) int { return cmp.Compare(a.number, b.number) })
132 slices.SortFunc(typ.members, func(a, b enumMember) int { return cmp.Compare(a.number, b.number) })
133 for _, field := range typ.fields { field.typ = types[field.typeName] }
134 }
135 return types
136}

152type message struct {
153 typ *definition
154 fields map[*field][]value
155 unknown []unknown
156}
157func newMessage(typ *definition) *message { return &message{typ: typ, fields: make(map[*field][]value)} }
158func (m *message) choose(field *field) {
159 if field.group == "" { return }
160 for _, other := range m.typ.fields {
161 if other != field && other.group == field.group { delete(m.fields, other) }
162 }
163}
164func (m *message) apply(field *field, incoming []value) {
165 if len(incoming) == 0 { return }
166 m.choose(field)
167 old := m.fields[field]
168 if field.repeated {
169 m.fields[field] = append(old, incoming...)
170 } else if field.typ.kind == messageKind && len(old) != 0 {
171 old[0].(*message).merge(incoming[0].(*message))
172 } else {
173 m.fields[field] = incoming
174 }
175}
176func (m *message) merge(incoming *message) {
177 for _, field := range m.typ.fields { m.apply(field, incoming.fields[field]) }
178 m.unknown = append(m.unknown, incoming.unknown...)
179}
180func (m *message) required(prefix string) {
181 for _, field := range m.typ.fields {
182 values, path := m.fields[field], prefix + field.name
183 if field.required && len(values) == 0 { fail("missing required field " + path) }
184 if field.typ.kind == messageKind {
185 for i, value := range values {
186 next := path
187 if field.repeated { next += fmt.Sprintf("[%d]", i) }
188 value.(*message).required(next + ".")
189 }
190 }
191 }
192}

194type codecError string
195func (e codecError) Error() string { return string(e) }
196func fail(reason string) { panic(codecError(reason)) }
197func byteError(reason string, offset int) { fail(fmt.Sprintf("%s at byte %d", reason, offset)) }
198func columnError(reason string, column int) { fail(fmt.Sprintf("%s at column %d", reason, column)) }

200func zigzag(number int64) int64 {
201 if number >= 0 { return number * 2 }
202 return -number * 2 - 1
203}
204func unzigzag(number int64) int64 {
205 if number % 2 == 0 { return number / 2 }
206 return -1 - number / 2
207}
208func putVarint(bytes []byte, number int64) []byte {
209 for number >= 128 { bytes = append(bytes, byte(number % 128 + 128)); number /= 128 }
210 return append(bytes, byte(number))
211}
212func encodeElement(bytes []byte, typ *definition, item value) []byte {
213 var number int64
214 switch item := item.(type) {
215 case *message:
216 payload := encode(item)
217 return append(putVarint(bytes, int64(len(payload))), payload...)
218 case binaryValue:
219 return append(putVarint(bytes, int64(len(item))), item...)
220 case booleanValue:
221 if item { number = 1 }
222 case numberValue:
223 number = int64(item)
224 }
225 if typ.kind == sint32Kind || typ.kind == sint64Kind { number = zigzag(number) }
226 if typ.kind == sfixed32Kind && number < 0 { number += 4_294_967_296 }
227 if typ.wire() == 0 { return putVarint(bytes, number) }
228 count := 4
229 if typ.wire() == 1 { count = 8 }
230 for range count { bytes = append(bytes, byte(number % 256)); number /= 256 }
231 return bytes
232}
233func encodeField(field *field, values []value) []byte {
234 var bytes []byte
235 if len(values) == 0 { return bytes }
236 if field.packed {
237 var payload []byte
238 for _, value := range values { payload = encodeElement(payload, field.typ, value) }
239 bytes = putVarint(bytes, field.number * 8 + 2)
240 bytes = append(putVarint(bytes, int64(len(payload))), payload...)
241 } else {
242 for _, value := range values {
243 bytes = putVarint(bytes, field.number * 8 + int64(field.typ.wire()))
244 bytes = encodeElement(bytes, field.typ, value)
245 }
246 }
247 return bytes
248}
249func encode(m *message) []byte {
250 var bytes []byte
251 for _, field := range m.typ.fields { bytes = append(bytes, encodeField(field, m.fields[field])...) }
252 for _, unknown := range m.unknown {
253 bytes = putVarint(bytes, unknown.number * 8 + int64(unknown.wire))
254 if unknown.wire == 0 {
255 bytes = putVarint(bytes, unknown.integer)
256 } else {
257 if unknown.wire == 2 { bytes = putVarint(bytes, int64(len(unknown.bytes))) }
258 bytes = append(bytes, unknown.bytes...)
259 }
260 }
261 return bytes
262}
263func hexNotation(bytes []byte) string {
264 if len(bytes) == 0 { return "-" }
265 return hex.EncodeToString(bytes)
266}
267func byteNotation(bytes []byte) string { return "<" + hex.EncodeToString(bytes) + ">" }
268func quote(bytes []byte, json bool) string {
269 var result strings.Builder
270 result.WriteByte('"')
271 for _, byte := range bytes {
272 switch byte {
273 case '"': result.WriteString(`\"`)
274 case '\\': result.WriteString(`\\`)
275 case '\n': result.WriteString(`\n`)
276 case '\t': result.WriteString(`\t`)
277 default:
278 if byte >= 32 && byte <= 126 { result.WriteByte(byte) } else {
279 if json { result.WriteString(`\u00`) } else { result.WriteString(`\x`) }
280 fmt.Fprintf(&result, "%02x", byte)
281 }
282 }
283 }
284 result.WriteByte('"')
285 return result.String()
286}
287func notation(typ *definition, item value, json bool) string {
288 switch item := item.(type) {
289 case numberValue:
290 text := strconv.FormatInt(int64(item), 10)
291 if typ.kind == enumKind {
292 text, _ = typ.enumName(int64(item))
293 if json { return `"` + text + `"` }
294 } else if json && (typ.kind == uint64Kind || typ.kind == sint64Kind || typ.kind == fixed64Kind) { return `"` + text + `"` }
295 return text
296 case booleanValue:
297 return strconv.FormatBool(bool(item))
298 case binaryValue:
299 if typ.kind == stringKind { return quote(item, json) }
300 if json { return `"` + base64.StdEncoding.EncodeToString(item) + `"` }
301 return byteNotation(item)
302 case *message:
303 return messageNotation(item, json)
304 default:
305 return "(none)"
306 }
307}
308func sequenceNotation(typ *definition, values []value, json bool) string {
309 items := make([]string, len(values))
310 for i, value := range values { items[i] = notation(typ, value, json) }
311 separator := ", "
312 if json { separator = "," }
313 return "[" + strings.Join(items, separator) + "]"
314}
315func messageNotation(m *message, json bool) string {
316 var items []string
317 separator := ", "
318 if json { separator = "," }
319 for _, field := range m.typ.fields {
320 values := m.fields[field]
321 if len(values) == 0 { continue }
322 key := field.name + ": "
323 if json {
324 var name strings.Builder
325 uppercase := false
326 for _, c := range []byte(field.name) {
327 if c == '_' { uppercase = true; continue }
328 if uppercase && c >= 'a' && c <= 'z' { c -= 'a' - 'A' }
329 name.WriteByte(c); uppercase = false
330 }
331 key = `"` + name.String() + `":`
332 }
333 if field.repeated { items = append(items, key + sequenceNotation(field.typ, values, json)) } else {
334 items = append(items, key + notation(field.typ, values[0], json))
335 }
336 }
337 return "{" + strings.Join(items, separator) + "}"
338}
339func wireName(wire int) string {
340 switch wire { case 0: return "varint"; case 1: return "i64"; case 2: return "len"; default: return "i32" }
341}
342func rawNotation(raw unknown) string {
343 if raw.wire == 0 { return strconv.FormatInt(raw.integer, 10) }
344 return byteNotation(raw.bytes)
345}
346func textContents(m *message, depth int, lines *[]string) {
347 indent := strings.Repeat(" ", depth)
348 for _, field := range m.typ.fields {
349 for _, value := range m.fields[field] {
350 if child, ok := value.(*message); ok {
351 *lines = append(*lines, indent + field.name + " {")
352 textContents(child, depth + 1, lines)
353 *lines = append(*lines, indent + "}")
354 } else { *lines = append(*lines, indent + field.name + ": " + notation(field.typ, value, false)) }
355 }
356 }
357 for _, unknown := range m.unknown { *lines = append(*lines, fmt.Sprintf("%s%d: %s %s", indent, unknown.number, wireName(unknown.wire), rawNotation(unknown))) }
358}
359func textMessage(m *message) []string {
360 lines := []string{m.typ.name + " {"}
361 textContents(m, 1, &lines)
362 return append(lines, "}")
363}

365type raw struct {
366 number int64
367 bytes []byte
368 position, payload int
369}
370type reader struct {
371 data []byte
372 position, end int
373}
374func (r *reader) varint() int64 {
375 start := r.position
376 var number int64
377 for i := range 8 {
378 if r.position >= r.end { byteError("truncated varint", start) }
379 byte := r.data[r.position]; r.position++
380 number += int64(byte & 127) << (7 * i)
381 if byte < 128 {
382 if number > maxUnsigned { byteError("varint too large", start) }
383 return number
384 }
385 }
386 byteError("varint too long", start)
387 return 0
388}
389func (r *reader) read(wire int) raw {
390 start := r.position
391 if wire == 0 { return raw{number: r.varint(), position: start, payload: start} }
392 length := int64(4)
393 if wire == 1 { length = 8 } else if wire == 2 { length = r.varint() }
394 if length > int64(r.end - r.position) {
395 reason := "truncated i32"
396 if wire == 1 { reason = "truncated i64" } else if wire == 2 { reason = "length past end" }
397 byteError(reason, start)
398 }
399 payload := r.position; r.position += int(length)
400 return raw{bytes: r.data[payload:r.position], position: start, payload: payload}
401}
402type decoder struct {
403 data []byte
404 explain bool
405 lines []string
406}
407func convert(typ *definition, raw raw) value {
408 if typ.kind == stringKind || typ.kind == bytesKind {
409 if typ.kind == stringKind {
410 for i, byte := range raw.bytes { if byte >= 128 { byteError("invalid string", raw.payload + i) } }
411 }
412 return binaryValue(raw.bytes)
413 }
414 number := raw.number
415 if typ.wire() == 1 || typ.wire() == 5 {
416 if typ.kind == fixed64Kind && (raw.bytes[7] != 0 || raw.bytes[6] >= 32) { byteError("value out of range", raw.position) }
417 number = 0
418 for i := len(raw.bytes) - 1; i >= 0; i-- { number = number * 256 + int64(raw.bytes[i]) }
419 }
420 if typ.kind == sint32Kind || typ.kind == sint64Kind { number = unzigzag(number) }
421 if typ.kind == sfixed32Kind && number >= 2_147_483_648 { number -= 4_294_967_296 }
422 if !typ.contains(number) { byteError("value out of range", raw.position) }
423 if typ.kind == boolKind {
424 if number > 1 { byteError("value out of range", raw.position) }
425 return booleanValue(number == 1)
426 }
427 if typ.kind == enumKind {
428 if _, ok := typ.enumName(number); !ok { byteError("unknown enum number", raw.position) }
429 }
430 return numberValue(number)
431}
432func (d *decoder) message(typ *definition, start, end, depth int) *message {
433 m := newMessage(typ)
434 r := reader{d.data, start, end}
435 for r.position < end {
436 offset := r.position; tag := r.varint()
437 number, wire := tag / 8, int(tag % 8)
438 if number == 0 { byteError("invalid field number", offset) }
439 if wire != 0 && wire != 1 && wire != 2 && wire != 5 { byteError("invalid wire type", offset) }
440 field := typ.byNumber(number)
441 packed := field != nil && field.repeated && field.typ.wire() != 2 && wire == 2
442 if field != nil && wire != field.typ.wire() && !packed { byteError("wire type mismatch", offset) }
443 raw := r.read(wire)
444 name := "?"
445 if field != nil { name = field.name }
446 header := fmt.Sprintf("%s%d %d %s %s", strings.Repeat(" ", depth), offset, number, name, wireName(wire))
447 if wire == 2 { header += fmt.Sprintf(" %d", len(raw.bytes)) }
448 if field == nil {
449 unknown := unknown{number, raw.number, wire, raw.bytes}
450 if d.explain { d.lines = append(d.lines, header + " " + rawNotation(unknown)) } else { m.unknown = append(m.unknown, unknown) }
451 continue
452 }
453 var values []value
454 if field.typ.kind == messageKind {
455 if d.explain { d.lines = append(d.lines, header) }
456 values = append(values, d.message(field.typ, raw.payload, raw.payload + len(raw.bytes), depth + 1))
457 } else {
458 if packed {
459 elements := reader{d.data, raw.payload, raw.payload + len(raw.bytes)}
460 for elements.position < elements.end { values = append(values, convert(field.typ, elements.read(field.typ.wire()))) }
461 } else { values = append(values, convert(field.typ, raw)) }
462 if d.explain {
463 if packed { header += " packed" }
464 for _, value := range values { header += " " + notation(field.typ, value, false) }
465 d.lines = append(d.lines, header)
466 }
467 }
468 if !d.explain { m.apply(field, values) }
469 }
470 return m
471}
472func decode(typ *definition, bytes []byte, suffix string) (m *message) {
473 defer func() {
474 if recovered := recover(); recovered != nil {
475 if problem, ok := recovered.(codecError); ok { panic(codecError(string(problem) + suffix)) }
476 panic(recovered)
477 }
478 }()
479 d := decoder{data: bytes}
480 m = d.message(typ, 0, len(bytes), 0)
481 m.required("")
482 return m
483}

485type tokenKind int
486const (
487 symbolToken tokenKind = iota
488 integerToken
489 nameToken
490 stringToken
491 bytesToken
492 endToken
493)
494type shape int
495const (
496 noShape shape = iota
497 messageShape
498 listShape
499 integerShape
500 booleanShape
501 nameShape
502 stringShape
503 bytesShape
504)
505type token struct {
506 kind tokenKind
507 column int
508 symbol byte
509 name string
510 number int64
511 bytes []byte
512}
513func (t token) shape() shape {
514 switch t.kind {
515 case integerToken: return integerShape
516 case stringToken: return stringShape
517 case bytesToken: return bytesShape
518 case nameToken:
519 if t.name == "true" || t.name == "false" { return booleanShape }
520 return nameShape
521 case symbolToken:
522 if t.symbol == '{' { return messageShape }
523 if t.symbol == '[' { return listShape }
524 }
525 return noShape
526}
527func digit(c byte) bool { return c >= '0' && c <= '9' }
528func letter(c byte) bool { return c >= 'a' && c <= 'z' || c >= 'A' && c <= 'Z' }
529func hexDigit(c byte) int {
530 if digit(c) { return int(c - '0') }
531 if c >= 'a' && c <= 'f' { return int(c - 'a') + 10 }
532 return -1
533}
534func tokenize(source string) []token {
535 var tokens []token
536 for at := 0; at < len(source); {
537 c := source[at]
538 if c == ' ' { at++; continue }
539 start := at; at++
540 t := token{kind: symbolToken, symbol: c, column: start + 1}
541 switch {
542 case strings.ContainsRune("{}[]:,", rune(c)):
543 case c == '"':
544 t.kind = stringToken
545 for at < len(source) && source[at] != '"' {
546 b := source[at]; at++
547 if b == '\\' {
548 if at == len(source) { columnError("syntax error", t.column) }
549 escaped := source[at]; at++
550 switch escaped {
551 case '"', '\\': b = escaped
552 case 'n': b = '\n'
553 case 't': b = '\t'
554 case 'x':
555 if at + 1 >= len(source) { columnError("syntax error", t.column) }
556 high, low := hexDigit(source[at]), hexDigit(source[at + 1])
557 if high < 0 || high > 7 || low < 0 { columnError("syntax error", t.column) }
558 b = byte(high * 16 + low); at += 2
559 default: columnError("syntax error", t.column)
560 }
561 }
562 t.bytes = append(t.bytes, b)
563 }
564 if at == len(source) { columnError("syntax error", t.column) }
565 at++
566 case c == '<':
567 t.kind = bytesToken
568 relative := strings.IndexByte(source[at:], '>')
569 if relative < 0 || relative % 2 != 0 { columnError("syntax error", t.column) }
570 end := at + relative
571 for at < end {
572 high, low := hexDigit(source[at]), hexDigit(source[at + 1])
573 if high < 0 || low < 0 { columnError("syntax error", t.column) }
574 t.bytes = append(t.bytes, byte(high * 16 + low)); at += 2
575 }
576 at++
577 case c == '-' || digit(c):
578 t.kind = integerToken
579 if c == '-' && (at == len(source) || !digit(source[at])) { columnError("syntax error", t.column) }
580 for at < len(source) && digit(source[at]) { at++ }
581 t.number, _ = strconv.ParseInt(source[start:at], 10, 64)
582 case letter(c) || c == '_':
583 t.kind = nameToken
584 for at < len(source) && (letter(source[at]) || digit(source[at]) || source[at] == '_') { at++ }
585 t.name = source[start:at]
586 default:
587 columnError("syntax error", t.column)
588 }
589 tokens = append(tokens, t)
590 }
591 return append(tokens, token{kind: endToken, column: len(source) + 1})
592}
593type literal struct {
594 tokens []token
595 position int
596}
597func newLiteral(source string) *literal { return &literal{tokens: tokenize(source)} }
598func (p *literal) next() token { return p.tokens[p.position] }
599func (p *literal) symbol(c byte) bool { return p.next().kind == symbolToken && p.next().symbol == c }
600func (p *literal) expect(c byte) {
601 if !p.symbol(c) { columnError("syntax error", p.next().column) }
602 p.position++
603}
604func (p *literal) end() {
605 if p.next().kind != endToken { columnError("syntax error", p.next().column) }
606}
607func (p *literal) require(wanted shape) {
608 if p.next().shape() == noShape { columnError("syntax error", p.next().column) }
609 if p.next().shape() != wanted { columnError("type mismatch", p.next().column) }
610}
611func (p *literal) element(typ *definition) value {
612 wanted := integerShape
613 switch typ.kind {
614 case boolKind: wanted = booleanShape
615 case stringKind: wanted = stringShape
616 case bytesKind: wanted = bytesShape
617 case enumKind: wanted = nameShape
618 case messageKind: wanted = messageShape
619 }
620 p.require(wanted)
621 if wanted == messageShape { return p.message(typ) }
622 t := p.next(); p.position++
623 switch wanted {
624 case integerShape:
625 if !typ.contains(t.number) { columnError("out of range", t.column) }
626 return numberValue(t.number)
627 case booleanShape:
628 return booleanValue(t.name == "true")
629 case nameShape:
630 number, ok := typ.enumNumber(t.name)
631 if !ok { columnError("unknown enum value", t.column) }
632 return numberValue(number)
633 default:
634 return binaryValue(t.bytes)
635 }
636}
637func (p *literal) fieldValue(field *field) []value {
638 if !field.repeated { return []value{p.element(field.typ)} }
639 p.require(listShape); p.expect('[')
640 var values []value
641 if !p.symbol(']') {
642 values = append(values, p.element(field.typ))
643 for p.symbol(',') { p.position++; values = append(values, p.element(field.typ)) }
644 }
645 p.expect(']')
646 return values
647}
648func (p *literal) message(typ *definition) *message {
649 p.expect('{')
650 m := newMessage(typ)
651 seen, groups := make(map[*field]bool), make(map[string]bool)
652 if !p.symbol('}') {
653 for {
654 t := p.next()
655 if t.kind != nameToken { columnError("syntax error", t.column) }
656 field := typ.byName(t.name)
657 if field == nil { columnError("unknown field", t.column) }
658 if seen[field] { columnError("duplicate field", t.column) }
659 if field.group != "" && groups[field.group] { columnError("oneof conflict", t.column) }
660 seen[field], groups[field.group] = true, true
661 p.position++; p.expect(':')
662 m.fields[field] = p.fieldValue(field)
663 if !p.symbol(',') { break }
664 p.position++
665 }
666 }
667 closing := p.next(); p.expect('}')
668 for _, field := range typ.fields {
669 if field.required && !seen[field] { columnError("missing required field " + field.name, closing.column) }
670 }
671 return m
672}

674type segment struct {
675 field *field
676 index int
677 indexed bool
678}
679func parsePath(typ *definition, source string) []segment {
680 var path []segment
681 parts := strings.Split(source, ".")
682 for i, part := range parts {
683 name, index, indexed := part, 0, false
684 if open := strings.IndexByte(part, '['); open >= 0 {
685 if !strings.HasSuffix(part, "]") { fail("invalid path") }
686 digits := part[open + 1:len(part) - 1]
687 if len(digits) == 0 || len(digits) > 6 { fail("invalid path") }
688 for j := range len(digits) { if !digit(digits[j]) { fail("invalid path") } }
689 index, _ = strconv.Atoi(digits); indexed = true; name = part[:open]
690 }
691 field := typ.byName(name)
692 if field == nil || indexed && !field.repeated { fail("invalid path") }
693 if i + 1 < len(parts) && (field.typ.kind != messageKind || field.repeated && !indexed) { fail("invalid path") }
694 path = append(path, segment{field, index, indexed}); typ = field.typ
695 }
696 return path
697}
698func getPath(m *message, path []segment) string {
699 for i, segment := range path {
700 var values []value
701 if m != nil { values = m.fields[segment.field] }
702 if i + 1 == len(path) && segment.field.repeated && !segment.indexed { return sequenceNotation(segment.field.typ, values, false) }
703 var selected value
704 if segment.index < len(values) { selected = values[segment.index] }
705 if i + 1 == len(path) { return notation(segment.field.typ, selected, false) }
706 if selected == nil { m = nil } else { m = selected.(*message) }
707 }
708 return "(none)"
709}
710func editPath(m *message, path []segment, replacement *[]value) {
711 for i, segment := range path {
712 field, values := segment.field, m.fields[segment.field]
713 if i + 1 == len(path) {
714 if segment.indexed {
715 if segment.index >= len(values) {
716 if replacement != nil { fail("index out of range") }
717 } else if replacement == nil {
718 m.fields[field] = slices.Delete(values, segment.index, segment.index + 1)
719 } else { values[segment.index] = (*replacement)[0] }
720 } else if replacement == nil { delete(m.fields, field) } else { m.choose(field); m.fields[field] = *replacement }
721 return
722 }
723 if segment.index >= len(values) {
724 if replacement == nil { return }
725 if segment.indexed { fail("index out of range") }
726 m.choose(field); values = []value{newMessage(field.typ)}; m.fields[field] = values
727 }
728 m = values[segment.index].(*message)
729 }
730}
731func equalValue(a, b value) bool {
732 switch a := a.(type) {
733 case numberValue: return a == b.(numberValue)
734 case booleanValue: return a == b.(booleanValue)
735 case binaryValue: return slices.Equal(a, b.(binaryValue))
736 default: return false
737 }
738}
739func differences(a, b *message, prefix string, lines *[]string) {
740 for _, field := range a.typ.fields {
741 x, y := a.fields[field], b.fields[field]
742 for i := range max(len(x), len(y)) {
743 path := prefix + field.name
744 if field.repeated { path += fmt.Sprintf("[%d]", i) }
745 if i < len(x) && i < len(y) && field.typ.kind == messageKind {
746 differences(x[i].(*message), y[i].(*message), path + ".", lines)
747 } else if i >= len(x) || i >= len(y) || !equalValue(x[i], y[i]) {
748 var left, right value
749 if i < len(x) { left = x[i] }
750 if i < len(y) { right = y[i] }
751 *lines = append(*lines, path + ": " + notation(field.typ, left, false) + " -> " + notation(field.typ, right, false))
752 }
753 }
754 }
755}
756func schemaFields(typ *definition, chain []*definition, lines *[]string) {
757 chain = append(chain, typ)
758 for _, field := range typ.fields {
759 mode := ""
760 if field.required { mode = "required " } else if field.packed { mode = "repeated packed " } else if field.repeated { mode = "repeated " } else if field.group != "" { mode = "oneof " + field.group + " " }
761 line := fmt.Sprintf("%s%d %s: %s%s", strings.Repeat(" ", len(chain)), field.number, field.name, mode, field.typ.name)
762 if field.typ.kind == enumKind {
763 var members []string
764 for _, member := range field.typ.members { members = append(members, fmt.Sprintf("%s=%d", member.name, member.number)) }
765 line += " (" + strings.Join(members, ", ") + ")"
766 }
767 recursive := field.typ.kind == messageKind && slices.Contains(chain, field.typ)
768 if recursive { line += " (recursive)" }
769 *lines = append(*lines, line)
770 if field.typ.kind == messageKind && !recursive { schemaFields(field.typ, chain, lines) }
771 }
772}
773func parseHex(source string) []byte {
774 if source == "-" { return nil }
775 if len(source) == 0 || len(source) % 2 != 0 { fail("invalid hex") }
776 for i := range len(source) { if hexDigit(source[i]) < 0 { fail("invalid hex") } }
777 bytes, _ := hex.DecodeString(source)
778 return bytes
779}
780func rest(source string, count int) string { return strings.SplitN(source, " ", count + 1)[count] }
781func (s schema) perform(source string) []string {
782 words := strings.Split(source, " ")
783 command, count := words[0], 0
784 switch command {
785 case "VARINT", "ZIGZAG", "SCHEMA": count = 2
786 case "ENCODE", "SIZE", "DECODE", "LITERAL", "JSON", "CANON", "EXPLAIN": count = 3
787 case "GET", "CLEAR", "MERGE", "DIFF": count = 4
788 case "PATCH": count = 5
789 default: fail("unknown command")
790 }
791 flexible := command == "ENCODE" || command == "SIZE" || command == "PATCH"
792 if len(words) < count || !flexible && len(words) != count { fail("wrong number of arguments") }
793 if command == "VARINT" || command == "ZIGZAG" {
794 digits := strings.TrimPrefix(words[1], "-")
795 if len(digits) == 0 || len(digits) > 18 { fail("invalid number") }
796 for i := range len(digits) { if !digit(digits[i]) { fail("invalid number") } }
797 number, _ := strconv.ParseInt(words[1], 10, 64)
798 if command == "VARINT" {
799 if number < 0 || number > maxUnsigned { fail("invalid number") }
800 return []string{hexNotation(putVarint(nil, number))}
801 }
802 if !s["sint64"].contains(number) { fail("invalid number") }
803 z := zigzag(number)
804 return []string{fmt.Sprintf("%d %s", z, hexNotation(putVarint(nil, z)))}
805 }
806 typ := s[words[1]]
807 if typ == nil || typ.kind != messageKind { fail("unknown message type") }
808 if command == "SCHEMA" {
809 lines := []string{typ.name}; schemaFields(typ, nil, &lines); return lines
810 }
811 if command == "ENCODE" || command == "SIZE" {
812 parser := newLiteral(rest(source, 2)); m := parser.message(typ); parser.end()
813 bytes := encode(m)
814 if command == "ENCODE" { return []string{hexNotation(bytes)} }
815 line := strconv.Itoa(len(bytes))
816 for _, field := range typ.fields {
817 if values := m.fields[field]; len(values) != 0 { line += fmt.Sprintf(" %s=%d", field.name, len(encodeField(field, values))) }
818 }
819 return []string{line}
820 }
821 first := parseHex(words[2])
822 var second []byte
823 if command == "MERGE" || command == "DIFF" { second = parseHex(words[3]) }
824 var path []segment
825 if command == "GET" || command == "PATCH" || command == "CLEAR" { path = parsePath(typ, words[3]) }
826 if command == "EXPLAIN" {
827 d := decoder{data: first, explain: true}; d.message(typ, 0, len(first), 0)
828 if len(first) == 0 { return []string{"(empty)"} }
829 return d.lines
830 }
831 if command == "DIFF" {
832 a := decode(typ, first, " in first"); b := decode(typ, second, " in second")
833 var lines []string; differences(a, b, "", &lines)
834 if len(lines) == 0 { return []string{"SAME"} }
835 return lines
836 }
837 if command == "MERGE" { first = append(first, second...) }
838 m := decode(typ, first, "")
839 switch command {
840 case "DECODE", "MERGE": return textMessage(m)
841 case "LITERAL", "JSON": return []string{messageNotation(m, command == "JSON")}
842 case "GET": return []string{getPath(m, path)}
843 case "PATCH":
844 parser := newLiteral(rest(source, 4)); last := path[len(path) - 1]
845 var replacement []value
846 if last.field.repeated && !last.indexed { replacement = parser.fieldValue(last.field) } else { replacement = []value{parser.element(last.field.typ)} }
847 parser.end(); editPath(m, path, &replacement); m.required("")
848 case "CLEAR":
849 editPath(m, path, nil); m.required("")
850 }
851 return []string{hexNotation(encode(m))}
852}
853func (s schema) execute(source string) (lines []string) {
854 defer func() {
855 if recovered := recover(); recovered != nil {
856 if problem, ok := recovered.(codecError); ok { lines = []string{"ERROR " + string(problem)} } else { panic(recovered) }
857 }
858 }()
859 return s.perform(source)
860}
861func main() {
862 input := bufio.NewScanner(os.Stdin)
863 types := readSchema(input)
864 input.Scan(); count, _ := strconv.Atoi(input.Text())
865 output := bufio.NewWriter(os.Stdout)
866 defer output.Flush()
867 for range count {
868 input.Scan()
869 for _, line := range types.execute(input.Text()) { fmt.Fprintln(output, line) }
870 }
871}
