63final class Message extends Value {

69 List<Value> values(Field field) => fields[field] ?? <Value>[];
70 void choose(Field field) {
71 if (field.group == null) return;
72 for (final other in type.fields) {
73 if (other != field && other.group == field.group) fields.remove(other);
74 }
75 }
76 void apply(Field field, List<Value> incoming) {
77 if (incoming.isEmpty) return;
78 choose(field);
79 if (field.isRepeated) {
80 (fields[field] ??= []).addAll(incoming);
81 } else if (field.type.kind == Kind.message && fields.containsKey(field)) {
82 (fields[field]!.first as Message).merge(incoming.first as Message);
83 } else {
84 fields[field] = incoming;
85 }
86 }
87 void merge(Message source) {
88 for (final field in type.fields) {
89 apply(field, source.values(field));
90 }
91 unknown.addAll(source.unknown);
92 }

100String hexDigits(List<int> bytes) => bytes.map((byte) => byte.toRadixString(16).padLeft(2, '0')).join();
101String hex(List<int> bytes) => bytes.isEmpty ? '-' : hexDigits(bytes);
102String byteNotation(List<int> bytes) => '<' + hexDigits(bytes) + '>';
103String wireName(int wire) => switch (wire) { 0 => 'varint', 1 => 'i64', 2 => 'len', _ => 'i32' };
104String rawNotation(Unknown value) => value.wire == 0 ? value.integer.toString() : byteNotation(value.bytes);
105String quote(List<int> bytes, bool json) {
106 final result = StringBuffer('"');
107 for (final byte in bytes) {
108 result.write(switch (byte) {
109 34 => r'\"', 92 => r'\\', 10 => r'\n', 9 => r'\t',
110 >= 32 && <= 126 => String.fromCharCode(byte),
111 _ => (json ? r'\u00' : r'\x') + byte.toRadixString(16).padLeft(2, '0'),
112 });
113 }
114 return (result..write('"')).toString();
115}
116String valueNotation(SchemaType type, Value value, {bool json = false}) {
117 if (value is NumberValue) {
118 final number = value.number;
119 if (type.kind == Kind.enumType) {
120 final name = type.numbers[number]!;
121 return json ? '"$name"' : name;
122 }
123 final wide = [Kind.uint64, Kind.sint64, Kind.fixed64].contains(type.kind);
124 return json && wide ? '"$number"' : '$number';
125 }
126 if (value is BooleanValue) return value.state ? 'true' : 'false';
127 if (value is BinaryValue) {
128 if (type.kind == Kind.string) return quote(value.bytes, json);
129 return json ? '"' + base64Encode(value.bytes) + '"' : byteNotation(value.bytes);
130 }
131 return messageNotation(value as Message, json: json);
132}
133String sequenceNotation(SchemaType type, List<Value> values, {bool json = false}) =>
134 '[' + values.map((value) => valueNotation(type, value, json: json)).join(json ? ',' : ', ') + ']';
135String jsonName(String name) => name.replaceAllMapped(RegExp(r'_([a-z0-9])'), (match) => match[1]!.toUpperCase());
136String messageNotation(Message message, {bool json = false}) {
137 final items = <String>[];
138 for (final field in message.type.fields) {
139 final values = message.values(field);
140 if (values.isEmpty) continue;
141 final key = json ? '"' + jsonName(field.name) + '":' : field.name + ': ';
142 items.add(key + (field.isRepeated ? sequenceNotation(field.type, values, json: json) :
143 valueNotation(field.type, values.first, json: json)));
144 }
145 return '{' + items.join(json ? ',' : ', ') + '}';
146}
147List<String> textNotation(Message message) {
148 final lines = [message.type.name + ' {'];
149 void contents(Message current, int depth) {
150 final indent = ' ' * depth;
151 for (final field in current.type.fields) {
152 for (final value in current.values(field)) {
153 if (value is Message) {
154 lines.add(indent + field.name + ' {');
155 contents(value, depth + 1);
156 lines.add('$indent}');
157 } else {
158 lines.add(indent + field.name + ': ' + valueNotation(field.type, value));
159 }
160 }
161 }
162 for (final value in current.unknown) {
163 lines.add(indent + value.number.toString() + ': ' + wireName(value.wire) + ' ' + rawNotation(value));
164 }
165 }
166 contents(message, 1);
167 lines.add('}');
168 return lines;
169}

171const maxUnsigned = 9007199254740991;
172int zigzag(int value) => value >= 0 ? 2 * value : -2 * value - 1;
173int unzigzag(int value) => value.isEven ? value ~/ 2 : -1 - value ~/ 2;
174bool inRange(SchemaType type, int value) => switch (type.kind) {
175 Kind.uint32 || Kind.fixed32 => value >= 0 && value <= 4294967295,
176 Kind.uint64 || Kind.fixed64 => value >= 0 && value <= maxUnsigned,
177 Kind.sint32 || Kind.sfixed32 => value >= -2147483648 && value <= 2147483647,
178 Kind.sint64 => value >= -4503599627370496 && value <= 4503599627370495,
179 _ => true,
180};
181void writeVarint(List<int> output, int value) {
182 while (value >= 128) {
183 output.add(value % 128 + 128);
184 value ~/= 128;
185 }
186 output.add(value);
187}
188void writeElement(List<int> output, SchemaType type, Value value) {
189 if (value is Message || value is BinaryValue) {
190 final payload = value is Message ? encode(value) : (value as BinaryValue).bytes;
191 writeVarint(output, payload.length);
192 output.addAll(payload);
193 return;
194 }
195 var number = value is BooleanValue ? (value.state ? 1 : 0) : (value as NumberValue).number;
196 if (type.kind == Kind.sint32 || type.kind == Kind.sint64) number = zigzag(number);
197 if (type.kind == Kind.sfixed32 && number < 0) number += 4294967296;
198 if (type.wire == 0) {
199 writeVarint(output, number);
200 } else {
201 for (var i = 0; i < (type.wire == 1 ? 8 : 4); i++) {
202 output.add(number % 256);
203 number ~/= 256;
204 }
205 }
206}
207List<int> encodeField(Field field, List<Value> values) {
208 final output = <int>[];
209 if (values.isEmpty) return output;
210 if (field.isPacked) {
211 final payload = <int>[];
212 for (final value in values) {
213 writeElement(payload, field.type, value);
214 }
215 writeVarint(output, field.number * 8 + 2);
216 writeVarint(output, payload.length);
217 output.addAll(payload);
218 } else {
219 for (final value in values) {
220 writeVarint(output, field.number * 8 + field.type.wire);
221 writeElement(output, field.type, value);
222 }
223 }
224 return output;
225}
226List<int> encode(Message message) {
227 final output = <int>[];
228 for (final field in message.type.fields) {
229 output.addAll(encodeField(field, message.values(field)));
230 }
231 for (final value in message.unknown) {
232 writeVarint(output, value.number * 8 + value.wire);
233 if (value.wire == 0) {
234 writeVarint(output, value.integer);
235 } else {
236 if (value.wire == 2) writeVarint(output, value.bytes.length);
237 output.addAll(value.bytes);
238 }
239 }
240 return output;
241}
242void checkRequired(Message message, [String prefix = '']) {
243 for (final field in message.type.fields) {
244 final values = message.values(field), path = prefix + field.name;
245 if (field.isRequired && values.isEmpty) throw CodecError('missing required field $path');
246 if (field.type.kind == Kind.message) {
247 for (var i = 0; i < values.length; i++) {
248 checkRequired(values[i] as Message, path + (field.isRepeated ? '[$i]' : '') + '.');
249 }
250 }
251 }

259CodecError byteError(String reason, int position) => CodecError('$reason at byte $position');
260final class Reader {
261 final List<int> data;
262 final int end;
263 int position;
264 Reader(this.data, this.position, this.end);
265
266 int varint() {
267 final start = position;
268 var value = 0;
269 for (var i = 0; i < 8; i++) {
270 if (position == end) throw byteError('truncated varint', start);
271 final byte = data[position++];
272 value += (byte & 127) << (7 * i);
273 if (byte < 128) {
274 if (value > maxUnsigned) throw byteError('varint too large', start);
275 return value;
276 }
277 }
278 throw byteError('varint too long', start);
279 }
280 Raw read(int wire) {
281 final start = position;
282 if (wire == 0) return Raw(varint(), [], start, start);
283 final length = wire == 2 ? varint() : wire == 1 ? 8 : 4;
284 if (length > end - position) {
285 throw byteError(wire == 2 ? 'length past end' : wire == 1 ? 'truncated i64' : 'truncated i32', start);
286 }
287 final payload = position;
288 position += length;
289 return Raw(0, data.sublist(payload, position), start, payload);
290 }
291}

299 Value convertValue(SchemaType type, Raw raw) {
300 if (type.kind == Kind.string) {
301 for (var i = 0; i < raw.bytes.length; i++) {
302 if (raw.bytes[i] >= 128) throw byteError('invalid string', raw.payload + i);
303 }
304 return BinaryValue(raw.bytes);
305 }
306 if (type.kind == Kind.bytes) return BinaryValue(raw.bytes);
307 var number = raw.number;
308 if (type.wire == 1 || type.wire == 5) {
309 if (type.kind == Kind.fixed64 && (raw.bytes[7] != 0 || raw.bytes[6] >= 32)) {
310 throw byteError('value out of range', raw.position);
311 }
312 number = 0;
313 for (var i = raw.bytes.length - 1; i >= 0; i--) {
314 number = number * 256 + raw.bytes[i];
315 }
316 }
317 if (type.kind == Kind.sint32 || type.kind == Kind.sint64) number = unzigzag(number);
318 if (type.kind == Kind.sfixed32 && number >= 2147483648) number -= 4294967296;
319 if (!inRange(type, number)) throw byteError('value out of range', raw.position);
320 if (type.kind == Kind.boolType) {
321 if (number > 1) throw byteError('value out of range', raw.position);
322 return BooleanValue(number == 1);
323 }
324 if (type.kind == Kind.enumType && !type.numbers.containsKey(number)) {
325 throw byteError('unknown enum number', raw.position);
326 }
327 return NumberValue(number);
328 }
329 Message readMessage(SchemaType type, int start, int end, int depth) {
330 final message = Message(type), reader = Reader(data, start, end);
331 while (reader.position < end) {
332 final offset = reader.position, tag = reader.varint();
333 final number = tag ~/ 8, wire = tag % 8;
334 if (number == 0) throw byteError('invalid field number', offset);
335 if (![0, 1, 2, 5].contains(wire)) throw byteError('invalid wire type', offset);
336 final field = type.byNumber(number);
337 final packed = field != null && field.isRepeated && field.type.packable && wire == 2;
338 if (field != null && wire != field.type.wire && !packed) throw byteError('wire type mismatch', offset);
339 final raw = reader.read(wire);
340 final header = ' ' * depth + '$offset $number ' + (field?.name ?? '?') + ' ' + wireName(wire) +
341 (wire == 2 ? ' ' + raw.bytes.length.toString() : '');
342 if (field == null) {
343 final unknown = Unknown(number, wire, raw.number, raw.bytes);
344 if (explain) {
345 lines.add('$header ' + rawNotation(unknown));
346 } else {
347 message.unknown.add(unknown);
348 }
349 continue;
350 }
351 final values = <Value>[];
352 if (field.type.kind == Kind.message) {
353 if (explain) lines.add(header);
354 values.add(readMessage(field.type, raw.payload, raw.payload + raw.bytes.length, depth + 1));
355 } else if (packed) {
356 final elements = Reader(data, raw.payload, raw.payload + raw.bytes.length);
357 while (elements.position < elements.end) {
358 values.add(convertValue(field.type, elements.read(field.type.wire)));
359 }
360 if (explain) lines.add('$header packed' + values.map((value) => ' ' + valueNotation(field.type, value)).join());
361 } else {
362 values.add(convertValue(field.type, raw));
363 if (explain) lines.add('$header ' + valueNotation(field.type, values.first));
364 }
365 if (!explain) message.apply(field, values);
366 }
367 return message;
368 }
369 Message decode(SchemaType type) {
370 final message = readMessage(type, 0, data.length, 0);
371 if (explain) {
372 if (lines.isEmpty) lines.add('(empty)');
373 } else {
374 checkRequired(message);
375 }
376 return message;
377 }
378}

380enum TokenKind { symbol, integer, name, string, bytes, end }
381enum Shape { none, message, list, integer, boolean, name, string, bytes }
382final class Token {
383 final TokenKind kind;
384 final String text;
385 final int column, number;
386 final List<int> data;
387 Token(this.kind, this.text, this.column, {this.number = 0, this.data = const []});
388 Shape get shape => switch (kind) {
389 TokenKind.integer => Shape.integer, TokenKind.string => Shape.string, TokenKind.bytes => Shape.bytes,
390 TokenKind.name => text == 'true' || text == 'false' ? Shape.boolean : Shape.name,
391 TokenKind.symbol when text == '{' => Shape.message,
392 TokenKind.symbol when text == '[' => Shape.list,
393 _ => Shape.none,
394 };
395}
396bool digit(int code) => code >= 48 && code <= 57;
397bool letter(int code) => code >= 65 && code <= 90 || code >= 97 && code <= 122;
398int hexDigit(int code) => digit(code) ? code - 48 : code >= 97 && code <= 102 ? code - 87 : -1;
399List<int> parseHexDigits(String text) => [for (var i = 0; i < text.length; i += 2) int.parse(text.substring(i, i + 2), radix: 16)];

401final class Literal {
402 final tokens = <Token>[];
403 int position = 0;
404 Token get next => tokens[position];
405 CodecError error(String reason, Token token) => CodecError('$reason at column ' + token.column.toString());
406 Literal(String text) {
407 var at = 0;
408 while (at < text.length) {
409 final character = text[at];
410 if (character == ' ') { at++; continue; }
411 final start = at++;
412 final token = Token(TokenKind.symbol, character, start + 1);
413 if ('{}[]:,'.contains(character)) {
414 tokens.add(token);
415 } else if (character == '"') {
416 final bytes = <int>[];
417 var closed = false;
418 while (at < text.length) {
419 final current = text[at++];
420 if (current == '"') { closed = true; break; }
421 if (current != '\\') { bytes.add(current.codeUnitAt(0)); continue; }
422 if (at == text.length) throw error('syntax error', token);
423 final escaped = text[at++];
424 if (escaped == '"' || escaped == '\\') {
425 bytes.add(escaped.codeUnitAt(0));
426 } else if (escaped == 'n') {
427 bytes.add(10);
428 } else if (escaped == 't') {
429 bytes.add(9);
430 } else if (escaped == 'x') {
431 if (at + 2 > text.length) throw error('syntax error', token);
432 final high = hexDigit(text.codeUnitAt(at)), low = hexDigit(text.codeUnitAt(at + 1));
433 if (high < 0 || high > 7 || low < 0) throw error('syntax error', token);
434 bytes.add(high * 16 + low);
435 at += 2;
436 } else {
437 throw error('syntax error', token);
438 }
439 }
440 if (!closed) throw error('syntax error', token);
441 tokens.add(Token(TokenKind.string, '', start + 1, data: bytes));
442 } else if (character == '<') {
443 final end = text.indexOf('>', at);
444 if (end < 0) throw error('syntax error', token);
445 final digits = text.substring(at, end);
446 if (digits.length.isOdd || digits.codeUnits.any((code) => hexDigit(code) < 0)) throw error('syntax error', token);
447 tokens.add(Token(TokenKind.bytes, '', start + 1, data: parseHexDigits(digits)));
448 at = end + 1;
449 } else if (character == '-' || digit(character.codeUnitAt(0))) {
450 if (character == '-' && (at == text.length || !digit(text.codeUnitAt(at)))) throw error('syntax error', token);
451 while (at < text.length && digit(text.codeUnitAt(at))) { at++; }
452 tokens.add(Token(TokenKind.integer, '', start + 1, number: int.parse(text.substring(start, at))));
453 } else if (letter(character.codeUnitAt(0)) || character == '_') {
454 while (at < text.length && (letter(text.codeUnitAt(at)) || digit(text.codeUnitAt(at)) || text[at] == '_')) { at++; }
455 tokens.add(Token(TokenKind.name, text.substring(start, at), start + 1));
456 } else {
457 throw error('syntax error', token);
458 }
459 }
460 tokens.add(Token(TokenKind.end, '', text.length + 1));
461 }
462 bool isSymbol(String symbol) => next.kind == TokenKind.symbol && next.text == symbol;
463 void expect(String symbol) {
464 if (!isSymbol(symbol)) throw error('syntax error', next);
465 position++;
466 }
467 void end() {
468 if (next.kind != TokenKind.end) throw error('syntax error', next);
469 }
470 void require(Shape shape) {
471 if (next.shape == Shape.none) throw error('syntax error', next);
472 if (next.shape != shape) throw error('type mismatch', next);
473 }
474 Message message(SchemaType type) {
475 expect('{');
476 final result = Message(type), seen = <Field>{}, groups = <String>{};
477 if (!isSymbol('}')) {
478 while (true) {
479 final name = next;
480 if (name.kind != TokenKind.name) throw error('syntax error', name);
481 final field = type.byName(name.text);
482 if (field == null) throw error('unknown field', name);
483 if (!seen.add(field)) throw error('duplicate field', name);
484 if (field.group != null && !groups.add(field.group!)) throw error('oneof conflict', name);
485 position++;
486 expect(':');
487 result.fields[field] = fieldValue(field);
488 if (!isSymbol(',')) break;
489 position++;
490 }
491 }
492 final closing = next;
493 expect('}');
494 for (final field in type.fields) {
495 if (field.isRequired && !seen.contains(field)) throw error('missing required field ' + field.name, closing);
496 }
497 return result;
498 }
499 List<Value> fieldValue(Field field) {
500 if (!field.isRepeated) return [element(field.type)];
501 require(Shape.list);
502 expect('[');
503 final values = <Value>[];
504 if (!isSymbol(']')) {
505 values.add(element(field.type));
506 while (isSymbol(',')) { position++; values.add(element(field.type)); }
507 }
508 expect(']');
509 return values;
510 }
511 Value element(SchemaType type) {
512 final shape = switch (type.kind) {
513 Kind.boolType => Shape.boolean, Kind.string => Shape.string, Kind.bytes => Shape.bytes,
514 Kind.enumType => Shape.name, Kind.message => Shape.message, _ => Shape.integer,
515 };
516 require(shape);
517 final token = next;
518 if (shape == Shape.message) return message(type);
519 position++;
520 if (shape == Shape.integer) {
521 if (!inRange(type, token.number)) throw error('out of range', token);
522 return NumberValue(token.number);
523 }
524 if (shape == Shape.boolean) return BooleanValue(token.text == 'true');
525 if (shape == Shape.name) {
526 final number = type.names[token.text];
527 if (number == null) throw error('unknown enum value', token);
528 return NumberValue(number);
529 }
530 return BinaryValue(token.data);
531 }
532}

543 Engine() {
544 const names = ['uint32', 'uint64', 'sint32', 'sint64', 'bool', 'fixed32', 'sfixed32', 'fixed64', 'string', 'bytes'];
545 for (var i = 0; i < names.length; i++) {
546 types[names[i]] = SchemaType(names[i], Kind.values[i]);

549 void readSchema() {
550 SchemaType? current;
551 while (true) {
552 final line = stdin.readLineSync()!;
553 if (line == 'END SCHEMA') break;
554 final words = line.split(' ');
555 if (words[0] == 'ENUM' || words[0] == 'MESSAGE') {
556 current = SchemaType(words[1], words[0] == 'ENUM' ? Kind.enumType : Kind.message);
557 types[current.name] = current;
558 } else if (words[0] == 'VALUE') {
559 final number = int.parse(words[2]);
560 final definition = current!;
561 definition.numbers[number] = words[1];
562 definition.names[words[1]] = number;
563 } else if (words[0] == 'FIELD') {
564 final mode = words.length > 4 ? words[4] : '';
565 current!.fields.add(Field(int.parse(words[1]), words[2], words[3], mode == 'REQUIRED',
566 mode == 'REPEATED', words.length > 5 && words[5] == 'PACKED', mode == 'ONEOF' ? words[5] : null));
567 }
568 }
569 for (final type in types.values) {
570 type.fields.sort((a, b) => a.number.compareTo(b.number));
571 for (final field in type.fields) {
572 field.type = types[field.typeName]!;
573 }
574 }
575 }
576 List<Segment> path(SchemaType type, String text) {
577 final segments = <Segment>[], words = text.split('.');
578 for (var i = 0; i < words.length; i++) {
579 final match = RegExp(r'^([a-z][a-z0-9_]*)(?:\[([0-9]{1,6})\])?$').firstMatch(words[i]);
580 if (match == null) throw CodecError('invalid path');
581 final field = type.byName(match[1]!);
582 final index = match[2] == null ? null : int.parse(match[2]!);
583 if (field == null || index != null && !field.isRepeated) throw CodecError('invalid path');
584 if (i + 1 < words.length && (field.type.kind != Kind.message || field.isRepeated && index == null)) {
585 throw CodecError('invalid path');
586 }
587 segments.add(Segment(field, index));
588 type = field.type;
589 }
590 return segments;
591 }
592 String get(Message message, List<Segment> segments) {
593 Message? current = message;
594 for (var step = 0; step < segments.length; step++) {
595 final segment = segments[step], values = current?.values(segments[step].field) ?? <Value>[];
596 final index = segment.index ?? 0;
597 if (step == segments.length - 1) {
598 if (segment.wholeList) return sequenceNotation(segment.field.type, values);
599 return index < values.length ? valueNotation(segment.field.type, values[index]) : '(none)';
600 }
601 current = index < values.length ? values[index] as Message : null;
602 }
603 throw StateError('Empty path');
604 }
605 void edit(Message message, List<Segment> segments, List<Value>? replacement) {
606 var current = message;
607 for (var i = 0; i < segments.length - 1; i++) {
608 final segment = segments[i], index = segments[i].index ?? 0;
609 var values = current.values(segment.field);
610 if (index >= values.length) {
611 if (replacement == null) return;
612 if (segment.index != null) throw CodecError('index out of range');
613 current.choose(segment.field);
614 values = [Message(segment.field.type)];
615 current.fields[segment.field] = values;
616 }
617 current = values[index] as Message;
618 }
619 final last = segments.last, index = segments.last.index;
620 if (index != null) {
621 final values = current.values(last.field);
622 if (index >= values.length) {
623 if (replacement != null) throw CodecError('index out of range');
624 } else if (replacement == null) {
625 values.removeAt(index);
626 } else {
627 values[index] = replacement.first;
628 }
629 } else if (replacement == null) {
630 current.fields.remove(last.field);
631 } else {
632 current.choose(last.field);
633 current.fields[last.field] = replacement;
634 }
635 }
636 bool equal(Value a, Value b) {
637 if (a is NumberValue && b is NumberValue) return a.number == b.number;
638 if (a is BooleanValue && b is BooleanValue) return a.state == b.state;
639 if (a is BinaryValue && b is BinaryValue) {
640 if (a.bytes.length != b.bytes.length) return false;
641 for (var i = 0; i < a.bytes.length; i++) {
642 if (a.bytes[i] != b.bytes[i]) return false;
643 }
644 return true;
645 }
646 return false;
647 }
648 void differences(Message a, Message b, String prefix, List<String> lines) {
649 for (final field in a.type.fields) {
650 final x = a.values(field), y = b.values(field);
651 final count = x.length > y.length ? x.length : y.length;
652 for (var i = 0; i < count; i++) {
653 final path = prefix + field.name + (field.isRepeated ? '[$i]' : '');
654 if (i < x.length && i < y.length && field.type.kind == Kind.message) {
655 differences(x[i] as Message, y[i] as Message, '$path.', lines);
656 } else if (i >= x.length || i >= y.length || !equal(x[i], y[i])) {
657 lines.add('$path: ' + (i < x.length ? valueNotation(field.type, x[i]) : '(none)') +
658 ' -> ' + (i < y.length ? valueNotation(field.type, y[i]) : '(none)'));
659 }
660 }
661 }
662 }
663 List<String> schema(SchemaType type) {
664 final lines = [type.name], chain = <SchemaType>{};
665 void visit(SchemaType current, int depth) {
666 chain.add(current);
667 for (final field in current.fields) {
668 final mode = field.isRequired ? 'required ' : field.isPacked ? 'repeated packed ' :
669 field.isRepeated ? 'repeated ' : field.group != null ? 'oneof ' + field.group! + ' ' : '';
670 var line = ' ' * depth + field.number.toString() + ' ' + field.name + ': ' + mode + field.type.name;
671 if (field.type.kind == Kind.enumType) {
672 line += ' (' + field.type.numbers.entries.map((entry) => entry.value + '=' + entry.key.toString()).join(', ') + ')';
673 }
674 final recursive = field.type.kind == Kind.message && chain.contains(field.type);
675 lines.add(line + (recursive ? ' (recursive)' : ''));
676 if (field.type.kind == Kind.message && !recursive) visit(field.type, depth + 1);
677 }
678 chain.remove(current);
679 }
680 visit(type, 1);
681 return lines;
682 }
683 List<int> parseHex(String text) {
684 if (text == '-') return [];
685 if (!RegExp(r'^(?:[0-9a-f]{2})+$').hasMatch(text)) throw CodecError('invalid hex');
686 return parseHexDigits(text);
687 }
688 String rest(String line, int separators) {
689 var at = -1;
690 for (var i = 0; i < separators; i++) {
691 at = line.indexOf(' ', at + 1);
692 }
693 return line.substring(at + 1);
694 }
695 Message decode(SchemaType type, List<int> data, [String suffix = '']) {
696 try {
697 return Decoder(data).decode(type);
698 } on CodecError catch (error) {
699 throw CodecError(error.message + suffix);
700 }
701 }
702 List<String> execute(String line) {
703 final words = line.split(' '), command = words.first;
704 final minimum = switch (command) {
705 'VARINT' || 'ZIGZAG' || 'SCHEMA' => 2,
706 'ENCODE' || 'SIZE' || 'DECODE' || 'LITERAL' || 'JSON' || 'CANON' || 'EXPLAIN' => 3,
707 'GET' || 'CLEAR' || 'MERGE' || 'DIFF' => 4, 'PATCH' => 5, _ => 0,
708 };
709 if (minimum == 0) throw CodecError('unknown command');
710 if (words.length < minimum || !['ENCODE', 'SIZE', 'PATCH'].contains(command) && words.length != minimum) {
711 throw CodecError('wrong number of arguments');
712 }
713 if (command == 'VARINT' || command == 'ZIGZAG') {
714 if (!RegExp(r'^-?[0-9]{1,18}$').hasMatch(words[1])) throw CodecError('invalid number');
715 final number = int.parse(words[1]);
716 if (command == 'VARINT' ? number < 0 || number > maxUnsigned :
717 number < -4503599627370496 || number > 4503599627370495) {
718 throw CodecError('invalid number');
719 }
720 final value = command == 'ZIGZAG' ? zigzag(number) : number;
721 final bytes = <int>[];
722 writeVarint(bytes, value);
723 return [(command == 'ZIGZAG' ? '$value ' : '') + hex(bytes)];
724 }
725 final type = types[words[1]];
726 if (type == null || type.kind != Kind.message) throw CodecError('unknown message type');
727 final data = ['SCHEMA', 'ENCODE', 'SIZE'].contains(command) ? <int>[] : parseHex(words[2]);
728 final second = command == 'MERGE' || command == 'DIFF' ? parseHex(words[3]) : <int>[];
729 final segments = ['GET', 'PATCH', 'CLEAR'].contains(command) ? path(type, words[3]) : <Segment>[];
730 if (command == 'SCHEMA') return schema(type);
731 if (command == 'ENCODE' || command == 'SIZE') {
732 final literal = Literal(rest(line, 2)), message = literal.message(type);
733 literal.end();
734 final encoded = encode(message);
735 if (command == 'ENCODE') return [hex(encoded)];
736 return [encoded.length.toString() + type.fields.where((field) => message.values(field).isNotEmpty)
737 .map((field) => ' ' + field.name + '=' + encodeField(field, message.values(field)).length.toString()).join()];
738 }
739 if (command == 'EXPLAIN') {
740 final decoder = Decoder(data, explain: true);
741 decoder.decode(type);
742 return decoder.lines;
743 }
744 if (command == 'MERGE') return textNotation(decode(type, [...data, ...second]));
745 if (command == 'DIFF') {
746 final a = decode(type, data, ' in first');
747 final b = decode(type, second, ' in second');
748 final lines = <String>[];
749 differences(a, b, '', lines);
750 return lines.isEmpty ? ['SAME'] : lines;
751 }
752 final message = decode(type, data);
753 if (command == 'DECODE') return textNotation(message);
754 if (command == 'LITERAL') return [messageNotation(message)];
755 if (command == 'JSON') return [messageNotation(message, json: true)];
756 if (command == 'GET') return [get(message, segments)];
757 if (command == 'PATCH') {
758 final literal = Literal(rest(line, 4)), last = segments.last;
759 final replacement = last.wholeList ? literal.fieldValue(last.field) : [literal.element(last.field.type)];
760 literal.end();
761 edit(message, segments, replacement);
762 checkRequired(message);
763 } else if (command == 'CLEAR') {
764 edit(message, segments, null);
765 checkRequired(message);
766 }
767 return [hex(encode(message))];
768 }
769}
