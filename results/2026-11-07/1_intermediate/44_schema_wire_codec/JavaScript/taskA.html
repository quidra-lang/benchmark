9class CommandError extends Error {}
10function fail(reason) { throw new CommandError(reason); }
11const byteError = (reason, offset) => fail(`${reason} at byte ${offset}`);
12const columnError = (reason, column) => fail(`${reason} at column ${column}`);
13const hex = data => data.length === 0 ? '-' : data.toString('hex');
14const rawText = raw => typeof raw === 'bigint' ? String(raw) : `<${raw.toString('hex')}>`;
15const zigzag = value => value < 0n ? -2n * value - 1n : 2n * value;

17class Schema {
18 constructor(lines) {
19 this.types = new Map();
20 const integers = [
21 ['uint32', 0, 0n, 4294967295n], ['uint64', 0, 0n, maximum],
22 ['sint32', 0, -2147483648n, 2147483647n], ['sint64', 0, signedMinimum, signedMaximum],
23 ['fixed32', 5, 0n, 4294967295n], ['sfixed32', 5, -2147483648n, 2147483647n],
24 ['fixed64', 1, 0n, maximum],
25 ];
26 for (const [name, wire, minimum, maximum] of integers) this.types.set(name, { kind: 'integer', name, wire, minimum, maximum });
27 for (const [name, wire] of [['bool', 0], ['string', 2], ['bytes', 2]]) this.types.set(name, { kind: name, name, wire });
28 let current = null;
29 this.nextLine = 0;
30 while (lines[this.nextLine] !== 'END SCHEMA') {
31 const words = lines[this.nextLine++].split(' ');
32 if (words[0] === 'ENUM') {
33 current = { kind: 'enum', name: words[1], wire: 0, values: [], byName: new Map(), byNumber: new Map() };
34 this.types.set(current.name, current);
35 } else if (words[0] === 'MESSAGE') {
36 current = { kind: 'message', name: words[1], wire: 2, fields: [], byName: new Map(), byNumber: new Map() };
37 this.types.set(current.name, current);
38 } else if (words[0] === 'VALUE') {
39 const name = words[1], number = BigInt(words[2]);
40 current.values.push({ name, number });
41 current.byName.set(name, number); current.byNumber.set(number, name);
42 } else if (words[0] === 'FIELD') {
43 current.fields.push({ number: Number(words[1]), typeName: words[2], name: words[3],
44 required: words[4] === 'REQUIRED', repeated: words[4] === 'REPEATED',
45 packed: words[5] === 'PACKED', group: words[4] === 'ONEOF' ? words[5] : null });
46 } else current = null;
47 }
48 this.nextLine++;
49 for (const type of this.types.values()) {
50 if (type.kind === 'enum') type.values.sort((a, b) => a.number < b.number ? -1 : a.number > b.number ? 1 : 0);
51 if (type.kind !== 'message') continue;
52 type.fields.sort((a, b) => a.number - b.number);
53 for (const field of type.fields) {
54 field.type = this.types.get(field.typeName);
55 type.byNumber.set(field.number, field); type.byName.set(field.name, field);
56 }
57 }
58 }
59 tree(type) {
60 const lines = [type.name];
61 const visit = (current, chain, depth) => {
62 for (const field of current.fields) {
63 const kind = field.required ? 'required ' : field.packed ? 'repeated packed '
64 : field.repeated ? 'repeated ' : field.group !== null ? `oneof ${field.group} ` : '';
65 let line = ' '.repeat(depth) + `${field.number} ${field.name}: ${kind}${field.type.name}`;
66 if (field.type.kind === 'enum') line += ` (${field.type.values.map(value => `${value.name}=${value.number}`).join(', ')})`;
67 const recursive = field.type.kind === 'message' && chain.includes(field.type);
68 if (recursive) line += ' (recursive)';
69 lines.push(line);
70 if (field.type.kind === 'message' && !recursive) visit(field.type, chain.concat(field.type), depth + 1);
71 }
72 };
73 visit(type, [type], 1);
74 return lines;
75 }
76}

78class Message {
79 constructor(type) { this.type = type; this.values = new Map(); this.unknown = []; }
80 replace(field, values) {
81 if (field.group !== null) {
82 for (const other of this.type.fields) if (other !== field && other.group === field.group) this.values.delete(other.number);
83 }
84 if (values.length) this.values.set(field.number, values);
85 else this.values.delete(field.number);
86 }
87 apply(field, values) {
88 const previous = this.values.get(field.number) ?? [];
89 if (field.repeated) this.replace(field, previous.concat(values));
90 else if (field.type.kind === 'message' && previous.length) {
91 previous[0].merge(values[0]); this.replace(field, previous);
92 } else this.replace(field, values);
93 }
94 merge(other) {
95 for (const field of this.type.fields) {
96 const values = other.values.get(field.number);
97 if (values) this.apply(field, values);
98 }
99 this.unknown = this.unknown.concat(other.unknown);
100 }
101 checkRequired(prefix = '') {
102 for (const field of this.type.fields) {
103 const values = this.values.get(field.number) ?? [];
104 const path = prefix + field.name;
105 if (field.required && values.length === 0) fail(`missing required field ${path}`);
106 if (field.type.kind === 'message') values.forEach((value, index) => value.checkRequired(path + (field.repeated ? `[${index}]` : '') + '.'));
107 }
108 }
109}

111function varint(value) {
112 const bytes = [];
113 do {
114 const low = Number(value % 128n); value /= 128n;
115 bytes.push(low + (value === 0n ? 0 : 128));
116 } while (value !== 0n);
117 return Buffer.from(bytes);
118}
119function valueBytes(type, value) {
120 if (value instanceof Message) return encode(value);
121 if (typeof value === 'boolean') return varint(value ? 1n : 0n);
122 if (typeof value === 'string') return Buffer.from(value, 'ascii');
123 if (typeof value !== 'bigint') return value;
124 if (type.name === 'sint32' || type.name === 'sint64') value = zigzag(value);
125 if (type.wire === 0) return varint(value);
126 if (type.name === 'sfixed32' && value < 0n) value += 4294967296n;
127 const result = Buffer.alloc(type.wire === 5 ? 4 : 8);
128 if (type.wire === 5) result.writeUInt32LE(Number(value));
129 else result.writeBigUInt64LE(value);
130 return result;
131}
132function record(number, wire, raw) {
133 const tag = varint(BigInt(number) * 8n + BigInt(wire));
134 if (wire === 0) return Buffer.concat([tag, varint(raw)]);
135 return Buffer.concat(wire === 2 ? [tag, varint(BigInt(raw.length)), raw] : [tag, raw]);
136}
137function fieldBytes(field, values) {
138 if (values.length === 0) return Buffer.alloc(0);
139 if (field.packed) return record(field.number, 2, Buffer.concat(values.map(value => valueBytes(field.type, value))));
140 const tag = varint(BigInt(field.number) * 8n + BigInt(field.type.wire));
141 return Buffer.concat(values.map(value => {
142 const bytes = valueBytes(field.type, value);
143 return Buffer.concat(field.type.wire === 2 ? [tag, varint(BigInt(bytes.length)), bytes] : [tag, bytes]);
144 }));
145}
146function encode(message) {
147 const chunks = [];
148 for (const field of message.type.fields) {
149 const values = message.values.get(field.number);
150 if (values) chunks.push(fieldBytes(field, values));
151 }
152 for (const unknown of message.unknown) chunks.push(record(unknown.number, unknown.wire, unknown.raw));
153 return Buffer.concat(chunks);
154}

156function quote(text, json = false) {
157 let result = '"';
158 for (const character of text) {
159 const code = character.charCodeAt(0);
160 if (character === '"' || character === '\\') result += '\\' + character;
161 else if (code === 10) result += '\\n';
162 else if (code === 9) result += '\\t';
163 else if (code >= 32 && code <= 126) result += character;
164 else result += (json ? '\\u00' : '\\x') + code.toString(16).padStart(2, '0');
165 }
166 return result + '"';
167}
168function notation(type, value, json = false) {
169 if (value instanceof Message) {
170 const items = [];
171 for (const field of type.fields) {
172 const values = value.values.get(field.number);
173 if (!values) continue;
174 const name = json ? quote(field.name.replace(/_([a-z0-9])/g, (_, character) => character.toUpperCase())) : field.name;
175 const text = field.repeated ? listNotation(field.type, values, json) : notation(field.type, values[0], json);
176 items.push(name + (json ? ':' : ': ') + text);
177 }
178 return '{' + items.join(json ? ',' : ', ') + '}';
179 }
180 if (typeof value === 'bigint') {
181 if (type.kind === 'enum') return json ? quote(type.byNumber.get(value)) : type.byNumber.get(value);
182 return json && type.name.endsWith('64') ? quote(String(value)) : String(value);
183 }
184 if (typeof value === 'boolean') return String(value);
185 if (typeof value === 'string') return quote(value, json);
186 return json ? quote(value.toString('base64')) : rawText(value);
187}
188function listNotation(type, values, json = false) { return '[' + values.map(value => notation(type, value, json)).join(json ? ',' : ', ') + ']'; }
189function textFormat(message) {
190 const lines = [`${message.type.name} {`];
191 const visit = (current, depth) => {
192 const indent = ' '.repeat(depth);
193 for (const field of current.type.fields) {
194 for (const value of current.values.get(field.number) ?? []) {
195 if (value instanceof Message) {
196 lines.push(`${indent}${field.name} {`); visit(value, depth + 1); lines.push(indent + '}');
197 } else lines.push(`${indent}${field.name}: ${notation(field.type, value)}`);
198 }
199 }
200 for (const unknown of current.unknown) lines.push(`${indent}${unknown.number}: ${wireNames.get(unknown.wire)} ${rawText(unknown.raw)}`);
201 };
202 visit(message, 1); lines.push('}');
203 return lines;
204}

206class Reader {
207 constructor(data, position = 0, end = data.length) { this.data = data; this.position = position; this.end = end; }
208 varint() {
209 const start = this.position;
210 let value = 0n, place = 1n;
211 for (let index = 0; index < 8; index++) {
212 if (this.position === this.end) byteError('truncated varint', start);
213 const byte = this.data[this.position++];
214 value += BigInt(byte % 128) * place;
215 if (byte < 128) {
216 if (value > maximum) byteError('varint too large', start);
217 return value;
218 }
219 if (index === 7) byteError('varint too long', start);
220 place *= 128n;
221 }
222 throw new Error('Unreachable varint');
223 }
224 fixed(length) {
225 const start = this.position;
226 if (this.end - start < length) byteError(length === 4 ? 'truncated i32' : 'truncated i64', start);
227 this.position += length;
228 return this.data.subarray(start, this.position);
229 }
230 raw(wire) {
231 const position = this.position;
232 if (wire === 0) return { raw: this.varint(), position, start: position };
233 if (wire !== 2) return { raw: this.fixed(wire === 5 ? 4 : 8), position, start: position };
234 const length = this.varint(), start = this.position;
235 if (length > BigInt(this.end - start)) byteError('length past end', position);
236 this.position += Number(length);
237 return { raw: this.data.subarray(start, this.position), position, start };
238 }
239}
240class Decoder {
241 constructor(data, explain = false) { this.data = data; this.explain = explain; this.lines = []; }
242 convert(type, raw, position, start, depth) {
243 if (type.kind === 'message') return this.message(type, start, start + raw.length, depth + 1);
244 if (type.kind === 'bytes') return raw;
245 if (type.kind === 'string') {
246 for (let index = 0; index < raw.length; index++) if (raw[index] >= 128) byteError('invalid string', start + index);
247 return raw.toString('ascii');
248 }
249 let value = typeof raw === 'bigint' ? raw : raw.length === 4 ? BigInt(raw.readUInt32LE()) : raw.readBigUInt64LE();
250 if (type.kind === 'bool') {
251 if (value !== 0n && value !== 1n) byteError('value out of range', position);
252 return value === 1n;
253 }
254 if (type.kind === 'enum') {
255 if (!type.byNumber.has(value)) byteError('unknown enum number', position);
256 return value;
257 }
258 if (type.name === 'sint32' || type.name === 'sint64') value = value % 2n === 0n ? value / 2n : -(value + 1n) / 2n;
259 if (type.name === 'sfixed32' && value >= 2147483648n) value -= 4294967296n;
260 if (value < type.minimum || value > type.maximum) byteError('value out of range', position);
261 return value;
262 }
263 message(type, start = 0, end = this.data.length, depth = 0) {
264 const result = new Message(type), reader = new Reader(this.data, start, end);
265 while (reader.position < end) {
266 const offset = reader.position, tag = reader.varint();
267 const number = Number(tag / 8n), wire = Number(tag % 8n);
268 if (number === 0) byteError('invalid field number', offset);
269 if (!wireNames.has(wire)) byteError('invalid wire type', offset);
270 const field = type.byNumber.get(number);
271 const packed = field && field.repeated && field.type.wire !== 2 && wire === 2;
272 if (field && wire !== field.type.wire && !packed) byteError('wire type mismatch', offset);
273 const value = reader.raw(wire);
274 let line = ' '.repeat(depth) + `${offset} ${number} ${field ? field.name : '?'} ${wireNames.get(wire)}`;
275 if (wire === 2) line += ` ${value.raw.length}`;
276 if (!field) {
277 if (this.explain) this.lines.push(line + ' ' + rawText(value.raw));
278 else result.unknown.push({ number, wire, raw: value.raw });
279 continue;
280 }
281 const values = [];
282 if (packed) {
283 const payload = new Reader(this.data, value.start, reader.position);
284 while (payload.position < payload.end) {
285 const element = payload.raw(field.type.wire);
286 values.push(this.convert(field.type, element.raw, element.position, element.start, depth));
287 }
288 line += ' packed' + values.map(item => ' ' + notation(field.type, item)).join('');
289 } else {
290 if (this.explain && field.type.kind === 'message') this.lines.push(line);
291 const item = this.convert(field.type, value.raw, value.position, value.start, depth);
292 values.push(item);
293 if (field.type.kind !== 'message') line += ' ' + notation(field.type, item);
294 }
295 if (this.explain) { if (field.type.kind !== 'message') this.lines.push(line); }
296 else result.apply(field, values);
297 }
298 return result;
299 }
300 decode(type) { const result = this.message(type); result.checkRequired(); return result; }
301}

303function tokenize(text) {
304 const tokens = [];
305 let position = 0;
306 while (position < text.length) {
307 if (text[position] === ' ') { position++; continue; }
308 const start = position, column = start + 1, character = text[position++];
309 if ('{}[]:,'.includes(character)) { tokens.push({ kind: character, column }); continue; }
310 if (character === '"') {
311 let value = '', closed = false;
312 while (position < text.length) {
313 let next = text[position++];
314 if (next === '"') { closed = true; break; }
315 if (next === '\\') {
316 if (position === text.length) columnError('syntax error', column);
317 const escape = text[position++];
318 if (escape === '"' || escape === '\\') next = escape;
319 else if (escape === 'n') next = '\n';
320 else if (escape === 't') next = '\t';
321 else if (escape === 'x') {
322 const pair = text.slice(position, position + 2);
323 if (!/^[0-7][0-9a-f]$/.test(pair)) columnError('syntax error', column);
324 next = String.fromCharCode(parseInt(pair, 16)); position += 2;
325 } else columnError('syntax error', column);
326 }
327 value += next;
328 }
329 if (!closed) columnError('syntax error', column);
330 tokens.push({ kind: 'string', value, column });
331 } else if (character === '<') {
332 const end = text.indexOf('>', position);
333 const digits = text.slice(position, end);
334 if (end < 0 || !/^(?:[0-9a-f]{2})*$/.test(digits)) columnError('syntax error', column);
335 tokens.push({ kind: 'bytes', value: Buffer.from(digits, 'hex'), column }); position = end + 1;
336 } else if (character === '-' || /[0-9]/.test(character)) {
337 if (character === '-' && (position === text.length || !/[0-9]/.test(text[position]))) columnError('syntax error', column);
338 while (position < text.length && /[0-9]/.test(text[position])) position++;
339 tokens.push({ kind: 'integer', value: BigInt(text.slice(start, position)), column });
340 } else if (/[A-Za-z_]/.test(character)) {
341 while (position < text.length && /[A-Za-z0-9_]/.test(text[position])) position++;
342 tokens.push({ kind: 'name', value: text.slice(start, position), column });
343 } else columnError('syntax error', column);
344 }
345 tokens.push({ kind: 'end', column: text.length + 1 });
346 return tokens;
347}
348function shape(token) {
349 if (token.kind === '{') return 'message';
350 if (token.kind === '[') return 'list';
351 if (token.kind === 'name') return token.value === 'true' || token.value === 'false' ? 'bool' : 'enum';
352 return ['integer', 'string', 'bytes'].includes(token.kind) ? token.kind : null;
353}
354class LiteralParser {
355 constructor(text) { this.tokens = tokenize(text); this.position = 0; }
356 get next() { return this.tokens[this.position]; }
357 expect(kind) {
358 const token = this.next;
359 if (token.kind !== kind) columnError('syntax error', token.column);
360 this.position++;
361 return token;
362 }
363 finish() { if (this.next.kind !== 'end') columnError('syntax error', this.next.column); }
364 message(type) {
365 this.expect('{');
366 const result = new Message(type), seen = new Set(), groups = new Set();
367 if (this.next.kind !== '}') {
368 while (true) {
369 const token = this.next;
370 if (token.kind !== 'name') columnError('syntax error', token.column);
371 const field = type.byName.get(token.value);
372 if (!field) columnError('unknown field', token.column);
373 if (seen.has(field.number)) columnError('duplicate field', token.column);
374 if (field.group !== null && groups.has(field.group)) columnError('oneof conflict', token.column);
375 seen.add(field.number); if (field.group !== null) groups.add(field.group);
376 this.position++; this.expect(':');
377 result.replace(field, this.field(field));
378 if (this.next.kind !== ',') break;
379 this.position++;
380 }
381 }
382 const closing = this.expect('}');
383 for (const field of type.fields) if (field.required && !seen.has(field.number)) columnError(`missing required field ${field.name}`, closing.column);
384 return result;
385 }
386 field(field) {
387 if (!field.repeated) return [this.element(field.type)];
388 const nextShape = shape(this.next);
389 if (nextShape === null) columnError('syntax error', this.next.column);
390 if (nextShape !== 'list') columnError('type mismatch', this.next.column);
391 this.expect('[');
392 const values = [];
393 if (this.next.kind !== ']') {
394 values.push(this.element(field.type));
395 while (this.next.kind === ',') { this.position++; values.push(this.element(field.type)); }
396 }
397 this.expect(']');
398 return values;
399 }
400 element(type) {
401 const token = this.next, nextShape = shape(token);
402 if (nextShape === null) columnError('syntax error', token.column);
403 if (nextShape !== type.kind) columnError('type mismatch', token.column);
404 if (type.kind === 'message') return this.message(type);
405 this.position++;
406 if (type.kind === 'integer') {
407 if (token.value < type.minimum || token.value > type.maximum) columnError('out of range', token.column);
408 } else if (type.kind === 'bool') return token.value === 'true';
409 else if (type.kind === 'enum') {
410 if (!type.byName.has(token.value)) columnError('unknown enum value', token.column);
411 return type.byName.get(token.value);
412 }
413 return token.value;
414 }
415}

417function parsePath(type, text) {
418 const parts = text.split('.'), segments = [];
419 let current = type;
420 for (let position = 0; position < parts.length; position++) {
421 const match = parts[position].match(/^([a-z][a-z0-9_]*)(?:\[([0-9]{1,6})\])?$/);
422 if (!match) fail('invalid path');
423 const field = current.byName.get(match[1]), index = match[2] === undefined ? null : Number(match[2]);
424 if (!field || (index !== null && !field.repeated)) fail('invalid path');
425 if (position + 1 < parts.length) {
426 if (field.type.kind !== 'message' || (field.repeated && index === null)) fail('invalid path');
427 current = field.type;
428 }
429 segments.push({ field, index });
430 }
431 return segments;
432}
433function parentAt(message, segments, create = false) {
434 let current = message;
435 for (const { field, index } of segments.slice(0, -1)) {
436 let values = current.values.get(field.number) ?? [];
437 if (index !== null && index >= values.length) {
438 if (create) fail('index out of range');
439 return null;
440 }
441 if (index === null && values.length === 0) {
442 if (!create) return null;
443 values = [new Message(field.type)]; current.replace(field, values);
444 }
445 current = values[index ?? 0];
446 }
447 return current;
448}
449function getPath(message, segments) {
450 const parent = parentAt(message, segments), { field, index } = segments.at(-1);
451 const values = parent?.values.get(field.number) ?? [];
452 if (field.repeated && index === null) return listNotation(field.type, values);
453 const value = values[index ?? 0];
454 return value === undefined ? '(none)' : notation(field.type, value);
455}
456function patch(message, segments, text) {
457 const { field, index } = segments.at(-1), parser = new LiteralParser(text);
458 const replacement = field.repeated && index === null ? parser.field(field) : [parser.element(field.type)];
459 parser.finish();
460 const parent = parentAt(message, segments, true);
461 if (index === null) parent.replace(field, replacement);
462 else {
463 const values = parent.values.get(field.number) ?? [];
464 if (index >= values.length) fail('index out of range');
465 values[index] = replacement[0];
466 }
467 message.checkRequired();
468}
469function clear(message, segments) {
470 const parent = parentAt(message, segments), { field, index } = segments.at(-1);
471 if (parent !== null) {
472 if (index === null) parent.values.delete(field.number);
473 else {
474 const values = parent.values.get(field.number) ?? [];
475 if (index < values.length) values.splice(index, 1);
476 if (values.length === 0) parent.values.delete(field.number);
477 }
478 }
479 message.checkRequired();
480}
481function differences(first, second) {
482 const lines = [];
483 const visit = (left, right, prefix) => {
484 for (const field of left.type.fields) {
485 const x = left.values.get(field.number) ?? [], y = right.values.get(field.number) ?? [];
486 for (let index = 0; index < Math.max(x.length, y.length); index++) {
487 const path = prefix + field.name + (field.repeated ? `[${index}]` : '');
488 const a = x[index], b = y[index];
489 if (a instanceof Message && b instanceof Message) visit(a, b, path + '.');
490 else if (a === undefined || b === undefined || !(Buffer.isBuffer(a) && Buffer.isBuffer(b) ? a.equals(b) : a === b)) {
491 lines.push(`${path}: ${a === undefined ? '(none)' : notation(field.type, a)} -> ${b === undefined ? '(none)' : notation(field.type, b)}`);
492 }
493 }
494 }
495 };
496 visit(first, second, '');
497 return lines.length ? lines : ['SAME'];
498}

517function execute(schema, line) {
518 const words = line.split(' '), command = words[0], arity = arities.get(command);
519 if (!arity) fail('unknown command');
520 if (words.length < arity[0] || words.length > arity[1]) fail('wrong number of arguments');
521 if (command === 'VARINT' || command === 'ZIGZAG') {
522 if (!/^-?[0-9]{1,18}$/.test(words[1])) fail('invalid number');
523 const value = BigInt(words[1]);
524 if (command === 'VARINT') {
525 if (value < 0n || value > maximum) fail('invalid number');
526 return [hex(varint(value))];
527 }
528 if (value < signedMinimum || value > signedMaximum) fail('invalid number');
529 const encoded = zigzag(value);
530 return [`${encoded} ${hex(varint(encoded))}`];
531 }
532 const type = schema.types.get(words[1]);
533 if (!type || type.kind !== 'message') fail('unknown message type');
534 if (command === 'SCHEMA') return schema.tree(type);
535 if (command === 'ENCODE' || command === 'SIZE') {
536 const parser = new LiteralParser(words.slice(2).join(' '));
537 const message = parser.message(type); parser.finish();
538 const bytes = encode(message);
539 if (command === 'ENCODE') return [hex(bytes)];
540 let line = String(bytes.length);
541 for (const field of type.fields) {
542 const values = message.values.get(field.number);
543 if (values) line += ` ${field.name}=${fieldBytes(field, values).length}`;
544 }
545 return [line];
546 }
547 const data = fromHex(words[2]);
548 if (command === 'MERGE' || command === 'DIFF') {
549 const second = fromHex(words[3]);
550 if (command === 'MERGE') return textFormat(new Decoder(Buffer.concat([data, second])).decode(type));
551 return differences(decodePart(type, data, 'first'), decodePart(type, second, 'second'));
552 }
553 const segments = ['GET', 'PATCH', 'CLEAR'].includes(command) ? parsePath(type, words[3]) : null;
554 if (command === 'EXPLAIN') {
555 const decoder = new Decoder(data, true); decoder.message(type);
556 return decoder.lines.length ? decoder.lines : ['(empty)'];
557 }
558 const message = new Decoder(data).decode(type);
559 switch (command) {
560 case 'DECODE': return textFormat(message);
561 case 'LITERAL': return [notation(type, message)];
562 case 'JSON': return [notation(type, message, true)];
563 case 'GET': return [getPath(message, segments)];
564 case 'PATCH': patch(message, segments, words.slice(4).join(' ')); break;
565 case 'CLEAR': clear(message, segments); break;
566 case 'CANON': break;
567 default: throw new Error('Unreachable command');
568 }
569 return [hex(encode(message))];
570}
571
572const lines = fs.readFileSync(0, 'utf8').split(/\r?\n/);
573const schema = new Schema(lines);
574const count = Number(lines[schema.nextLine]);
575for (let index = 0; index < count; index++) {
576 try {
577 for (const line of execute(schema, lines[schema.nextLine + 1 + index])) process.stdout.write(line + '\n');
578 } catch (error) {
579 if (!(error instanceof CommandError)) throw error;
580 process.stdout.write('ERROR ' + error.message + '\n');
581 }
582}
