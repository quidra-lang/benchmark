37 Kind::SInt32, Kind::SFixed32 => $value >= -2147483648 && $value <= 2147483647,

59final class Schema
60{
61 public array $types = [];
62
63 public function __construct()
64 {
65 $kinds = [
66 'uint32' => Kind::UInt32, 'uint64' => Kind::UInt64,
67 'sint32' => Kind::SInt32, 'sint64' => Kind::SInt64, 'bool' => Kind::Boolean,
68 'fixed32' => Kind::Fixed32, 'sfixed32' => Kind::SFixed32,
69 'fixed64' => Kind::Fixed64, 'string' => Kind::Text, 'bytes' => Kind::Bytes,
70 ];
71 foreach ($kinds as $name => $kind) $this->types[$name] = new SchemaType($name, $kind);
72 $current = null;
73 while (($line = inputLine()) !== 'END SCHEMA') {
74 $words = explode(' ', $line);
75 switch ($words[0]) {
76 case 'ENUM':
77 case 'MESSAGE':
78 $current = new SchemaType($words[1], $words[0] === 'ENUM' ? Kind::Enumeration : Kind::Message);
79 $this->types[$current->name] = $current;
80 break;
81 case 'VALUE':
82 $current->members[(int)$words[2]] = $words[1];
83 $current->numbers[$words[1]] = (int)$words[2];
84 break;
85 case 'FIELD':
86 $mode = $words[4] ?? '';
87 $field = new Field((int)$words[1], $words[2], $words[3], $mode === 'REQUIRED',
88 $mode === 'REPEATED', ($words[5] ?? '') === 'PACKED',
89 $mode === 'ONEOF' ? $words[5] : null);
90 $current->fields[$field->number] = $field;
91 $current->namedFields[$field->name] = $field;
92 break;
93 }
94 }
95 foreach ($this->types as $type) {
96 ksort($type->fields, SORT_NUMERIC);
97 ksort($type->members, SORT_NUMERIC);
98 foreach ($type->fields as $field) $field->type = $this->types[$field->typeName];
99 }
100 }

108final class Message
109{
110 public array $fields = [];
111 public array $unknown = [];
112
113 public function __construct(public readonly SchemaType $type) {}
114
115 public function values(Field $field): array
116 {
117 return $this->fields[$field->number] ?? [];
118 }
119
120 public function choose(Field $field): void
121 {
122 if ($field->group === null) return;
123 foreach ($this->type->fields as $other) {
124 if ($other !== $field && $other->group === $field->group) unset($this->fields[$other->number]);
125 }
126 }
127
128 public function apply(Field $field, array $values): void
129 {
130 if ($values === []) return;
131 $this->choose($field);
132 $old = $this->values($field);
133 if ($field->repeated) $this->fields[$field->number] = array_merge($old, $values);
134 elseif ($field->type->kind === Kind::Message && $old !== []) $old[0]->merge($values[0]);
135 else $this->fields[$field->number] = $values;
136 }
137
138 public function merge(Message $source): void
139 {
140 foreach ($this->type->fields as $field) $this->apply($field, $source->values($field));
141 $this->unknown = array_merge($this->unknown, $source->unknown);
142 }
143}

155function varint(int $value): string
156{
157 $bytes = '';
158 while ($value >= 128) {
159 $bytes .= chr($value % 128 + 128);
160 $value = intdiv($value, 128);
161 }
162 return $bytes . chr($value);

175function quoted(string $value, bool $json = false): string
176{
177 $text = '"';
178 for ($i = 0; $i < strlen($value); ++$i) {
179 $byte = ord($value[$i]);
180 $text .= match ($byte) {
181 34 => '\\"', 92 => '\\\\', 10 => '\\n', 9 => '\\t',
182 default => $byte >= 32 && $byte <= 126 ? $value[$i] :
183 ($json ? '\\u00' : '\\x') . sprintf('%02x', $byte),
184 };
185 }
186 return $text . '"';
187}

189function valueText(SchemaType $type, int|bool|string|Message $value, bool $json = false): string
190{
191 return match ($type->kind) {
192 Kind::Message => messageText($value, $json),
193 Kind::Boolean => $value ? 'true' : 'false',
194 Kind::Enumeration => $json ? quoted($type->members[$value]) : $type->members[$value],
195 Kind::Text => quoted($value, $json),
196 Kind::Bytes => $json ? '"' . base64_encode($value) . '"' : '<' . bin2hex($value) . '>',
197 Kind::UInt64, Kind::SInt64, Kind::Fixed64 => $json ? '"' . $value . '"' : (string)$value,
198 default => (string)$value,
199 };
200}

202function sequenceText(SchemaType $type, array $values, bool $json = false): string
203{
204 $items = array_map(fn($value) => valueText($type, $value, $json), $values);
205 return '[' . implode($json ? ',' : ', ', $items) . ']';
206}

208function messageText(Message $message, bool $json = false): string
209{
210 $items = [];
211 foreach ($message->type->fields as $field) {
212 $values = $message->values($field);
213 if ($values === []) continue;
214 $key = $field->name;
215 if ($json) {
216 $key = preg_replace_callback('/_([a-z0-9])/', fn($match) => strtoupper($match[1]), $key);
217 $key = quoted($key);
218 }
219 $value = $field->repeated ? sequenceText($field->type, $values, $json) :
220 valueText($field->type, $values[0], $json);
221 $items[] = $key . ($json ? ':' : ': ') . $value;
222 }
223 return '{' . implode($json ? ',' : ', ', $items) . '}';
224}

226function rawText(Unknown $unknown): string
227{
228 return $unknown->wire === 0 ? (string)$unknown->value : '<' . bin2hex($unknown->value) . '>';
229}

231function textContents(Message $message, int $depth, array &$lines): void
232{
233 $indent = str_repeat(' ', 2 * $depth);
234 foreach ($message->type->fields as $field) {
235 foreach ($message->values($field) as $value) {
236 if ($field->type->kind === Kind::Message) {
237 $lines[] = $indent . $field->name . ' {';
238 textContents($value, $depth + 1, $lines);
239 $lines[] = $indent . '}';
240 } else $lines[] = $indent . $field->name . ': ' . valueText($field->type, $value);
241 }
242 }
243 foreach ($message->unknown as $unknown)
244 $lines[] = $indent . $unknown->number . ': ' . wireName($unknown->wire) . ' ' . rawText($unknown);
245}

247function textMessage(Message $message): array
248{
249 $lines = [$message->type->name . ' {'];
250 textContents($message, 1, $lines);
251 $lines[] = '}';
252 return $lines;
253}

255function encodeValue(SchemaType $type, int|bool|string|Message $value): string
256{
257 if ($type->wire() === 2) {
258 $payload = $type->kind === Kind::Message ? encode($value) : $value;
259 return varint(strlen($payload)) . $payload;
260 }
261 if ($type->wire() === 0) {
262 $number = match ($type->kind) {
263 Kind::SInt32, Kind::SInt64 => zigzag($value),
264 Kind::Boolean => $value ? 1 : 0,
265 default => $value,
266 };
267 return varint($number);
268 }
269 if ($value < 0) $value += 4294967296;
270 $bytes = '';
271 $length = $type->wire() === 1 ? 8 : 4;
272 for ($i = 0; $i < $length; ++$i) {
273 $bytes .= chr($value % 256);
274 $value = intdiv($value, 256);
275 }
276 return $bytes;
277}
278
279function encodeField(Field $field, array $values): string
280{
281 if ($values === []) return '';
282 $bytes = '';
283 if ($field->packed) {
284 foreach ($values as $value) $bytes .= encodeValue($field->type, $value);
285 return varint($field->number * 8 + 2) . varint(strlen($bytes)) . $bytes;
286 }
287 foreach ($values as $value)
288 $bytes .= varint($field->number * 8 + $field->type->wire()) . encodeValue($field->type, $value);
289 return $bytes;
290}
291
292function encode(Message $message): string
293{
294 $bytes = '';
295 foreach ($message->type->fields as $field) $bytes .= encodeField($field, $message->values($field));
296 foreach ($message->unknown as $unknown) {
297 $bytes .= varint($unknown->number * 8 + $unknown->wire);
298 $bytes .= match ($unknown->wire) {
299 0 => varint($unknown->value),
300 2 => varint(strlen($unknown->value)) . $unknown->value,
301 default => $unknown->value,
302 };
303 }
304 return $bytes;
305}

312final class ByteReader
313{
314 public function __construct(public readonly string $data, public int $position, public readonly int $end) {}
315
316 public function varint(): int
317 {
318 $start = $this->position;
319 $value = 0;
320 for ($i = 0; $i < 8; ++$i) {
321 if ($this->position === $this->end) throw new CodecError("truncated varint at byte $start");
322 $byte = ord($this->data[$this->position++]);
323 $value |= ($byte & 127) << (7 * $i);
324 if ($byte < 128) {
325 if ($value > 9007199254740991) throw new CodecError("varint too large at byte $start");
326 return $value;
327 }
328 }
329 throw new CodecError("varint too long at byte $start");
330 }
331
332 public function read(int $wire): RawValue
333 {
334 $start = $this->position;
335 if ($wire === 0) return new RawValue($this->varint(), $start, $start);
336 $length = match ($wire) { 1 => 8, 5 => 4, 2 => $this->varint() };
337 $payload = $this->position;
338 if ($length > $this->end - $payload) {
339 $reason = $wire === 2 ? 'length past end' : 'truncated ' . wireName($wire);
340 throw new CodecError("$reason at byte $start");
341 }
342 $this->position += $length;
343 return new RawValue(substr($this->data, $payload, $length), $start, $payload);
344 }
345}

362final class Literal
363{
364 private array $tokens = [];
365 private int $position = 0;
366
367 public function __construct(string $source)
368 {
369 $length = strlen($source);
370 for ($i = 0; $i < $length;) {
371 $c = $source[$i];
372 if ($c === ' ') { ++$i; continue; }
373 $column = $i + 1;
374 if (str_contains('{}[]:,', $c)) {
375 $this->tokens[] = new Token($c, null, $column);
376 ++$i;
377 } elseif ($c === '"') {
378 ++$i;
379 $value = '';
380 while ($i < $length && $source[$i] !== '"') {
381 $c = $source[$i++];
382 if ($c !== '\\') { $value .= $c; continue; }
383 if ($i === $length) $this->problem('syntax error', $column);
384 $escape = $source[$i++];
385 if ($escape === '"' || $escape === '\\') $value .= $escape;
386 elseif ($escape === 'n') $value .= "\n";
387 elseif ($escape === 't') $value .= "\t";
388 elseif ($escape === 'x' && $i + 2 <= $length &&
389 preg_match('/\A[0-7][0-9a-f]\z/', substr($source, $i, 2))) {
390 $value .= chr(hexdec(substr($source, $i, 2)));
391 $i += 2;
392 } else $this->problem('syntax error', $column);
393 }
394 if ($i === $length) $this->problem('syntax error', $column);
395 ++$i;
396 $this->tokens[] = new Token('string', $value, $column);
397 } elseif ($c === '<') {
398 $end = strpos($source, '>', $i + 1);
399 if ($end === false) $this->problem('syntax error', $column);
400 $hex = substr($source, $i + 1, $end - $i - 1);
401 if (!validHex($hex)) $this->problem('syntax error', $column);
402 $this->tokens[] = new Token('bytes', hex2bin($hex), $column);
403 $i = $end + 1;
404 } elseif ($c === '-' || asciiDigit($c)) {
405 $start = $i;
406 if ($c === '-') ++$i;
407 $digits = $i;
408 while ($i < $length && asciiDigit($source[$i])) ++$i;
409 if ($digits === $i) $this->problem('syntax error', $column);
410 $this->tokens[] = new Token('integer', (int)substr($source, $start, $i - $start), $column);
411 } elseif (asciiLetter($c) || $c === '_') {
412 $start = $i++;
413 while ($i < $length && (asciiLetter($source[$i]) || asciiDigit($source[$i]) || $source[$i] === '_')) ++$i;
414 $this->tokens[] = new Token('name', substr($source, $start, $i - $start), $column);
415 } else $this->problem('syntax error', $column);
416 }
417 $this->tokens[] = new Token('end', null, $length + 1);
418 }

420 private function problem(string $reason, int $column): never
421 {
422 throw new CodecError("$reason at column $column");
423 }
424
425 private function expect(string $kind): Token
426 {
427 $token = $this->tokens[$this->position];
428 if ($token->kind !== $kind) $this->problem('syntax error', $token->column);
429 ++$this->position;
430 return $token;
431 }
432
433 public function finish(): void { $this->expect('end'); }
434
435 public function message(SchemaType $type): Message
436 {
437 $this->expect('{');
438 $message = new Message($type);
439 $seen = $groups = [];
440 if ($this->tokens[$this->position]->kind !== '}') {
441 while (true) {
442 $token = $this->tokens[$this->position];
443 if ($token->kind !== 'name') $this->problem('syntax error', $token->column);
444 $field = $type->namedFields[$token->value] ?? null;
445 if ($field === null) $this->problem('unknown field', $token->column);
446 if (isset($seen[$field->number])) $this->problem('duplicate field', $token->column);
447 if ($field->group !== null && isset($groups[$field->group])) $this->problem('oneof conflict', $token->column);
448 $seen[$field->number] = true;
449 if ($field->group !== null) $groups[$field->group] = true;
450 ++$this->position;
451 $this->expect(':');
452 $message->fields[$field->number] = $this->fieldValue($field);
453 if ($this->tokens[$this->position]->kind !== ',') break;
454 ++$this->position;
455 }
456 }
457 $closing = $this->expect('}');
458 foreach ($type->fields as $field)
459 if ($field->required && !isset($seen[$field->number]))
460 $this->problem('missing required field ' . $field->name, $closing->column);
461 return $message;
462 }

464 public function fieldValue(Field $field): array
465 {
466 if (!$field->repeated) return [$this->element($field->type)];
467 $token = $this->tokens[$this->position];
468 if ($token->shape() === '') $this->problem('syntax error', $token->column);
469 if ($token->shape() !== 'list') $this->problem('type mismatch', $token->column);
470 ++$this->position;
471 $values = [];
472 if ($this->tokens[$this->position]->kind !== ']') {
473 $values[] = $this->element($field->type);
474 while ($this->tokens[$this->position]->kind === ',') {
475 ++$this->position;
476 $values[] = $this->element($field->type);
477 }
478 }
479 $this->expect(']');
480 return $values;
481 }
482
483 public function element(SchemaType $type): int|bool|string|Message
484 {
485 $token = $this->tokens[$this->position];
486 $shape = match ($type->kind) {
487 Kind::Boolean => 'boolean', Kind::Text => 'string', Kind::Bytes => 'bytes',
488 Kind::Enumeration => 'name', Kind::Message => 'message', default => 'integer',
489 };
490 if ($token->shape() === '') $this->problem('syntax error', $token->column);
491 if ($token->shape() !== $shape) $this->problem('type mismatch', $token->column);
492 if ($shape === 'message') return $this->message($type);
493 ++$this->position;
494 if ($shape === 'integer' && !$type->inRange($token->value)) $this->problem('out of range', $token->column);
495 if ($shape === 'name') {
496 if (!isset($type->numbers[$token->value])) $this->problem('unknown enum value', $token->column);
497 return $type->numbers[$token->value];
498 }
499 return $shape === 'boolean' ? $token->value === 'true' : $token->value;
500 }

526final class Codec
527{
528 private array $explanation = [];
529
530 public function __construct(private readonly Schema $schema) {}
531
532 private function convert(SchemaType $type, RawValue $raw): int|bool|string
533 {
534 if ($type->kind === Kind::Text || $type->kind === Kind::Bytes) {
535 if ($type->kind === Kind::Text) {
536 for ($i = 0; $i < strlen($raw->value); ++$i)
537 if (ord($raw->value[$i]) >= 128)
538 throw new CodecError('invalid string at byte ' . ($raw->payload + $i));
539 }
540 return $raw->value;
541 }
542 if ($type->wire() === 0) $number = $raw->value;
543 else {
544 if ($type->kind === Kind::Fixed64 && (ord($raw->value[7]) !== 0 || ord($raw->value[6]) >= 32))
545 throw new CodecError('value out of range at byte ' . $raw->position);
546 $number = 0;
547 for ($i = strlen($raw->value) - 1; $i >= 0; --$i) $number = $number * 256 + ord($raw->value[$i]);
548 }
549 if ($type->kind === Kind::SInt32 || $type->kind === Kind::SInt64)
550 $number = $number % 2 === 0 ? intdiv($number, 2) : -intdiv($number, 2) - 1;
551 elseif ($type->kind === Kind::SFixed32 && $number >= 2147483648) $number -= 4294967296;
552 if (!$type->inRange($number) || ($type->kind === Kind::Boolean && $number > 1))
553 throw new CodecError('value out of range at byte ' . $raw->position);
554 if ($type->kind === Kind::Enumeration && !isset($type->members[$number]))
555 throw new CodecError('unknown enum number at byte ' . $raw->position);
556 return $type->kind === Kind::Boolean ? $number === 1 : $number;
557 }
558
559 private function readMessage(SchemaType $type, ByteReader $reader, bool $explain, int $depth): Message
560 {
561 $message = new Message($type);
562 while ($reader->position < $reader->end) {
563 $offset = $reader->position;
564 $tag = $reader->varint();
565 $number = intdiv($tag, 8);
566 $wire = $tag % 8;
567 if ($number === 0) throw new CodecError("invalid field number at byte $offset");
568 if (!in_array($wire, [0, 1, 2, 5], true)) throw new CodecError("invalid wire type at byte $offset");
569 $field = $type->fields[$number] ?? null;
570 $packed = $field !== null && $field->repeated && $field->type->wire() !== 2 && $wire === 2;
571 if ($field !== null && $wire !== $field->type->wire() && !$packed)
572 throw new CodecError("wire type mismatch at byte $offset");
573 $raw = $reader->read($wire);
574 $header = str_repeat(' ', 2 * $depth) . "$offset $number " . ($field?->name ?? '?') . ' ' . wireName($wire);
575 if ($wire === 2) $header .= ' ' . strlen($raw->value);
576 if ($field === null) {
577 $unknown = new Unknown($number, $wire, $raw->value);
578 if ($explain) $this->explanation[] = $header . ' ' . rawText($unknown);
579 else $message->unknown[] = $unknown;
580 continue;
581 }
582 $values = [];
583 if ($field->type->kind === Kind::Message) {
584 if ($explain) $this->explanation[] = $header;
585 $child = new ByteReader($reader->data, $raw->payload, $raw->payload + strlen($raw->value));
586 $values[] = $this->readMessage($field->type, $child, $explain, $depth + 1);
587 } elseif ($packed) {
588 $elements = new ByteReader($reader->data, $raw->payload, $raw->payload + strlen($raw->value));
589 $header .= ' packed';
590 while ($elements->position < $elements->end) {
591 $value = $this->convert($field->type, $elements->read($field->type->wire()));
592 $values[] = $value;
593 if ($explain) $header .= ' ' . valueText($field->type, $value);
594 }
595 if ($explain) $this->explanation[] = $header;
596 } else {
597 $values[] = $this->convert($field->type, $raw);
598 if ($explain) $this->explanation[] = $header . ' ' . valueText($field->type, $values[0]);
599 }
600 if (!$explain) $message->apply($field, $values);
601 }
602 return $message;
603 }

605 private function required(Message $message, string $prefix = ''): void
606 {
607 foreach ($message->type->fields as $field) {
608 $values = $message->values($field);
609 $path = $prefix . $field->name;
610 if ($field->required && $values === []) throw new CodecError("missing required field $path");
611 if ($field->type->kind === Kind::Message) {
612 foreach ($values as $i => $child)
613 $this->required($child, $path . ($field->repeated ? "[$i]" : '') . '.');
614 }
615 }
616 }

618 private function decode(SchemaType $type, string $bytes, bool $explain = false): Message
619 {
620 $message = $this->readMessage($type, new ByteReader($bytes, 0, strlen($bytes)), $explain, 0);
621 if (!$explain) $this->required($message);
622 elseif ($bytes === '') $this->explanation[] = '(empty)';
623 return $message;
624 }

629 catch (CodecError $error) { throw new CodecError($error->getMessage() . " in $side"); }

632 private function path(SchemaType $type, string $source): array
633 {
634 $parts = explode('.', $source);
635 $path = [];
636 foreach ($parts as $i => $part) {
637 if (!preg_match('/\A([a-z][a-z0-9]*(?:_[a-z0-9]+)*)(?:\[([0-9]{1,6})\])?\z/', $part, $match))
638 throw new CodecError('invalid path');
639 $field = $type->namedFields[$match[1]] ?? null;
640 $index = isset($match[2]) ? (int)$match[2] : null;
641 if ($field === null || ($index !== null && !$field->repeated))
642 throw new CodecError('invalid path');
643 if ($i + 1 < count($parts) && ($field->type->kind !== Kind::Message || ($field->repeated && $index === null)))
644 throw new CodecError('invalid path');
645 $path[] = new Segment($field, $index);
646 $type = $field->type;
647 }
648 return $path;
649 }

651 private function get(Message $message, array $path): string
652 {
653 $current = $message;
654 foreach ($path as $step => $segment) {
655 $values = $current?->values($segment->field) ?? [];
656 $index = $segment->index ?? 0;
657 if ($step + 1 === count($path)) {
658 if ($segment->wholeList()) return sequenceText($segment->field->type, $values);
659 return $index < count($values) ? valueText($segment->field->type, $values[$index]) : '(none)';
660 }
661 $current = $index < count($values) ? $values[$index] : null;
662 }
663 throw new LogicException();
664 }

666 private function edit(Message $message, array $path, ?array $replacement): void
667 {
668 $current = $message;
669 foreach (array_slice($path, 0, -1) as $segment) {
670 $values = $current->values($segment->field);
671 $index = $segment->index ?? 0;
672 if ($index >= count($values)) {
673 if ($replacement === null) return;
674 if ($segment->index !== null) throw new CodecError('index out of range');
675 $current->choose($segment->field);
676 $values = [new Message($segment->field->type)];
677 $current->fields[$segment->field->number] = $values;
678 }
679 $current = $values[$index];
680 }
681 $last = $path[count($path) - 1];
682 $number = $last->field->number;
683 if ($last->index !== null) {
684 $values = $current->values($last->field);
685 if ($last->index >= count($values)) {
686 if ($replacement !== null) throw new CodecError('index out of range');
687 } else {
688 if ($replacement === null) array_splice($values, $last->index, 1);
689 else $values[$last->index] = $replacement[0];
690 $current->fields[$number] = $values;
691 }
692 } elseif ($replacement === null) unset($current->fields[$number]);
693 else {
694 $current->choose($last->field);
695 $current->fields[$number] = $replacement;
696 }
697 }

699 private function differences(Message $a, Message $b, string $prefix, array &$lines): void
700 {
701 foreach ($a->type->fields as $field) {
702 $x = $a->values($field);
703 $y = $b->values($field);
704 for ($i = 0; $i < max(count($x), count($y)); ++$i) {
705 $path = $prefix . $field->name . ($field->repeated ? "[$i]" : '');
706 if ($i < count($x) && $i < count($y) && $field->type->kind === Kind::Message)
707 $this->differences($x[$i], $y[$i], $path . '.', $lines);
708 elseif ($i >= count($x) || $i >= count($y) || $x[$i] !== $y[$i]) {
709 $old = $i < count($x) ? valueText($field->type, $x[$i]) : '(none)';
710 $new = $i < count($y) ? valueText($field->type, $y[$i]) : '(none)';
711 $lines[] = "$path: $old -> $new";
712 }
713 }
714 }
715 }

717 private function schemaFields(SchemaType $type, array $chain, int $depth, array &$lines): void
718 {
719 foreach ($type->fields as $field) {
720 $kind = $field->required ? 'required ' : ($field->packed ? 'repeated packed ' :
721 ($field->repeated ? 'repeated ' : ($field->group !== null ? "oneof $field->group " : '')));
722 $line = str_repeat(' ', 2 * $depth) . "$field->number $field->name: $kind" . $field->type->name;
723 if ($field->type->kind === Kind::Enumeration) {
724 $members = [];
725 foreach ($field->type->members as $number => $name) $members[] = "$name=$number";
726 $line .= ' (' . implode(', ', $members) . ')';
727 }
728 $recursive = $field->type->kind === Kind::Message && in_array($field->type, $chain, true);
729 $lines[] = $line . ($recursive ? ' (recursive)' : '');
730 if ($field->type->kind === Kind::Message && !$recursive)
731 $this->schemaFields($field->type, [...$chain, $field->type], $depth + 1, $lines);
732 }
733 }

742 public function execute(string $source): array
743 {
744 $words = preg_split('/ +/', $source);
745 $command = $words[0];
746 $minimum = match ($command) {
747 'VARINT', 'ZIGZAG', 'SCHEMA' => 2,
748 'ENCODE', 'SIZE', 'DECODE', 'LITERAL', 'JSON', 'CANON', 'EXPLAIN' => 3,
749 'GET', 'CLEAR', 'MERGE', 'DIFF' => 4,
750 'PATCH' => 5,
751 default => throw new CodecError('unknown command'),
752 };
753 $flexible = in_array($command, ['ENCODE', 'SIZE', 'PATCH'], true);
754 if (count($words) < $minimum || (!$flexible && count($words) !== $minimum))
755 throw new CodecError('wrong number of arguments');
756 if ($command === 'VARINT' || $command === 'ZIGZAG') {
757 if (!preg_match('/\A-?[0-9]{1,18}\z/', $words[1])) throw new CodecError('invalid number');
758 $number = (int)$words[1];
759 $valid = $command === 'VARINT' ? $number >= 0 && $number <= 9007199254740991 :
760 $number >= -4503599627370496 && $number <= 4503599627370495;
761 if (!$valid) throw new CodecError('invalid number');
762 $value = $command === 'ZIGZAG' ? zigzag($number) : $number;
763 return [($command === 'ZIGZAG' ? "$value " : '') . hexText(varint($value))];
764 }
765 $type = $this->schema->types[$words[1]] ?? null;
766 if ($type === null || $type->kind !== Kind::Message) throw new CodecError('unknown message type');
767 $data = in_array($command, ['SCHEMA', 'ENCODE', 'SIZE'], true) ? '' : $this->hex($words[2]);
768 $second = $command === 'MERGE' || $command === 'DIFF' ? $this->hex($words[3]) : '';
769 $path = in_array($command, ['GET', 'PATCH', 'CLEAR'], true) ? $this->path($type, $words[3]) : [];
770 if ($command === 'SCHEMA') {
771 $lines = [$type->name];
772 $this->schemaFields($type, [$type], 1, $lines);
773 return $lines;
774 }
775 if ($command === 'ENCODE' || $command === 'SIZE') {
776 $literal = new Literal(rest($source, 2));
777 $message = $literal->message($type);
778 $literal->finish();
779 $encoded = encode($message);
780 if ($command === 'ENCODE') return [hexText($encoded)];
781 $line = (string)strlen($encoded);
782 foreach ($type->fields as $field) {
783 $values = $message->values($field);
784 if ($values !== []) $line .= ' ' . $field->name . '=' . strlen(encodeField($field, $values));
785 }
786 return [$line];
787 }
788 if ($command === 'EXPLAIN') {
789 $this->explanation = [];
790 $this->decode($type, $data, true);
791 return $this->explanation;
792 }
793 if ($command === 'MERGE') return textMessage($this->decode($type, $data . $second));
794 if ($command === 'DIFF') {
795 $a = $this->decodeSide($type, $data, 'first');
796 $b = $this->decodeSide($type, $second, 'second');
797 $lines = [];
798 $this->differences($a, $b, '', $lines);
799 return $lines === [] ? ['SAME'] : $lines;
800 }
801 $message = $this->decode($type, $data);
802 if ($command === 'DECODE') return textMessage($message);
803 if ($command === 'LITERAL' || $command === 'JSON') return [messageText($message, $command === 'JSON')];
804 if ($command === 'GET') return [$this->get($message, $path)];
805 if ($command === 'PATCH') {
806 $literal = new Literal(rest($source, 4));
807 $last = $path[count($path) - 1];
808 $replacement = $last->wholeList() ? $literal->fieldValue($last->field) : [$literal->element($last->field->type)];
809 $literal->finish();
810 $this->edit($message, $path, $replacement);
811 $this->required($message);
812 } elseif ($command === 'CLEAR') {
813 $this->edit($message, $path, null);
814 $this->required($message);
815 }
816 return [hexText(encode($message))];
817 }
818}
