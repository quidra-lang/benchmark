5type Result<T> = std::result::Result<T, String>;
6const MAX_UNSIGNED: i64 = 9_007_199_254_740_991;

8#[derive(Clone, Copy, PartialEq, Eq)]
9enum Scalar { Uint32, Uint64, Sint32, Sint64, Bool, Fixed32, Sfixed32, Fixed64, String, Bytes }
10
11impl Scalar {
12 fn wire(self) -> u8 {
13 match self {
14 Self::Fixed64 => 1,
15 Self::String | Self::Bytes => 2,
16 Self::Fixed32 | Self::Sfixed32 => 5,
17 _ => 0,
18 }
19 }
20 fn contains(self, number: i64) -> bool {
21 match self {
22 Self::Uint32 | Self::Fixed32 => (0..=4_294_967_295).contains(&number),
23 Self::Uint64 | Self::Fixed64 => (0..=MAX_UNSIGNED).contains(&number),
24 Self::Sint32 | Self::Sfixed32 => (-2_147_483_648..=2_147_483_647).contains(&number),
25 Self::Sint64 => (-4_503_599_627_370_496..=4_503_599_627_370_495).contains(&number),
26 _ => true,
27 }
28 }
29}

31struct Field {
32 number: i64,
33 name: String,
34 typ: usize,
35 required: bool,
36 repeated: bool,
37 packed: bool,
38 group: Option<String>,
39}
40enum Definition { Scalar(Scalar), Enum(BTreeMap<i64, String>), Message(Vec<Field>) }
41struct Type { name: String, definition: Definition }
42struct Schema { types: Vec<Type>, names: BTreeMap<String, usize> }
43
44impl Schema {
45 fn fields(&self, typ: usize) -> &[Field] {
46 match &self.types[typ].definition { Definition::Message(fields) => fields, _ => &[] }
47 }
48 fn is_message(&self, typ: usize) -> bool { matches!(&self.types[typ].definition, Definition::Message(_)) }
49 fn scalar(&self, typ: usize) -> Option<Scalar> {
50 match &self.types[typ].definition { Definition::Scalar(value) => Some(*value), _ => None }
51 }
52 fn wire(&self, typ: usize) -> u8 {
53 match &self.types[typ].definition { Definition::Scalar(value) => value.wire(), Definition::Enum(_) => 0, Definition::Message(_) => 2 }
54 }
55 fn enum_name(&self, typ: usize, number: i64) -> Option<&str> {
56 match &self.types[typ].definition { Definition::Enum(values) => values.get(&number).map(String::as_str), _ => None }
57 }
58 fn read(lines: &mut std::str::Lines<'_>) -> Self {
59 let mut schema = Self { types: Vec::new(), names: BTreeMap::new() };
60 for (name, scalar) in [
61 ("uint32", Scalar::Uint32), ("uint64", Scalar::Uint64), ("sint32", Scalar::Sint32),
62 ("sint64", Scalar::Sint64), ("bool", Scalar::Bool), ("fixed32", Scalar::Fixed32),
63 ("sfixed32", Scalar::Sfixed32), ("fixed64", Scalar::Fixed64), ("string", Scalar::String), ("bytes", Scalar::Bytes),
64 ] {
65 schema.names.insert(name.into(), schema.types.len());
66 schema.types.push(Type { name: name.into(), definition: Definition::Scalar(scalar) });
67 }
68 let mut pending = Vec::new();
69 let mut current = 0;
70 for line in lines.by_ref() {
71 if line == "END SCHEMA" { break; }
72 let words: Vec<_> = line.split(' ').collect();
73 match words[0] {
74 "ENUM" | "MESSAGE" => {
75 current = schema.types.len();
76 schema.names.insert(words[1].into(), current);
77 let definition = if words[0] == "ENUM" { Definition::Enum(BTreeMap::new()) } else { Definition::Message(Vec::new()) };
78 schema.types.push(Type { name: words[1].into(), definition });
79 }
80 "VALUE" => if let Definition::Enum(values) = &mut schema.types[current].definition {
81 values.insert(words[2].parse().unwrap(), words[1].into());
82 },
83 "FIELD" => if let Definition::Message(fields) = &mut schema.types[current].definition {
84 pending.push((current, fields.len(), words[2].to_owned()));
85 let mode = words.get(4).copied().unwrap_or("");
86 fields.push(Field {
87 number: words[1].parse().unwrap(), name: words[3].into(), typ: 0,
88 required: mode == "REQUIRED", repeated: mode == "REPEATED",
89 packed: mode == "REPEATED" && words.len() == 6,
90 group: (mode == "ONEOF").then(|| words[5].to_owned()),
91 });
92 },
93 _ => {},
94 }
95 }
96 for (owner, index, name) in pending {
97 if let Definition::Message(fields) = &mut schema.types[owner].definition { fields[index].typ = schema.names[&name]; }
98 }
99 for typ in &mut schema.types {
100 if let Definition::Message(fields) = &mut typ.definition { fields.sort_by_key(|field| field.number); }
101 }
102 schema
103 }
104}

106enum Value { Number(i64), Boolean(bool), Binary(Vec<u8>), Message(Message) }
107struct Unknown { number: i64, wire: u8, integer: i64, bytes: Vec<u8> }
108struct Message { typ: usize, fields: Vec<Vec<Value>>, unknown: Vec<Unknown> }
109
110impl Message {
111 fn new(typ: usize, schema: &Schema) -> Self {
112 Self { typ, fields: (0..schema.fields(typ).len()).map(|_| Vec::new()).collect(), unknown: Vec::new() }
113 }
114 fn choose(&mut self, index: usize, schema: &Schema) {
115 let fields = schema.fields(self.typ);
116 if let Some(group) = &fields[index].group {
117 for (i, field) in fields.iter().enumerate() {
118 if i != index && field.group.as_ref() == Some(group) { self.fields[i].clear(); }
119 }
120 }
121 }
122 fn apply(&mut self, index: usize, mut values: Vec<Value>, schema: &Schema) {
123 if values.is_empty() { return; }
124 self.choose(index, schema);
125 let field = &schema.fields(self.typ)[index];
126 let old = &mut self.fields[index];
127 if field.repeated { old.extend(values); }
128 else if schema.is_message(field.typ) && !old.is_empty() {
129 if let (Value::Message(existing), Value::Message(incoming)) = (&mut old[0], values.pop().unwrap()) { existing.merge(incoming, schema); }
130 } else { *old = values; }
131 }
132 fn merge(&mut self, incoming: Message, schema: &Schema) {
133 for (i, values) in incoming.fields.into_iter().enumerate() { self.apply(i, values, schema); }
134 self.unknown.extend(incoming.unknown);
135 }
136 fn required(&self, schema: &Schema, prefix: &str) -> Result<()> {
137 for (i, field) in schema.fields(self.typ).iter().enumerate() {
138 let values = &self.fields[i];
139 let path = format!("{prefix}{}", field.name);
140 if field.required && values.is_empty() { return Err(format!("missing required field {path}")); }
141 for (j, value) in values.iter().enumerate() {
142 if let Value::Message(child) = value {
143 let next = if field.repeated { format!("{path}[{j}].") } else { format!("{path}.") };
144 child.required(schema, &next)?;
145 }
146 }
147 }
148 Ok(())
149 }
150}

152fn zigzag(value: i64) -> i64 { if value >= 0 { value * 2 } else { -value * 2 - 1 } }
153fn unzigzag(value: i64) -> i64 { if value % 2 == 0 { value / 2 } else { -1 - value / 2 } }
154fn put_varint(output: &mut Vec<u8>, mut value: i64) {
155 while value >= 128 { output.push((value % 128 + 128) as u8); value /= 128; }
156 output.push(value as u8);
157}
158fn encode_element(output: &mut Vec<u8>, typ: usize, value: &Value, schema: &Schema) {
159 match value {
160 Value::Message(message) => {
161 let bytes = encode(message, schema); put_varint(output, bytes.len() as i64); output.extend(bytes);
162 }
163 Value::Binary(bytes) => { put_varint(output, bytes.len() as i64); output.extend(bytes); }
164 _ => {
165 let mut number = match value { Value::Number(n) => *n, Value::Boolean(b) => i64::from(*b), _ => unreachable!() };
166 if matches!(schema.scalar(typ), Some(Scalar::Sint32 | Scalar::Sint64)) { number = zigzag(number); }
167 if schema.scalar(typ) == Some(Scalar::Sfixed32) && number < 0 { number += 4_294_967_296; }
168 let wire = schema.wire(typ);
169 if wire == 0 { put_varint(output, number); }
170 else { output.extend_from_slice(&number.to_le_bytes()[..if wire == 1 { 8 } else { 4 }]); }
171 }
172 }
173}
174fn encode_field(field: &Field, values: &[Value], schema: &Schema) -> Vec<u8> {
175 let mut output = Vec::new();
176 if values.is_empty() { return output; }
177 if field.packed {
178 let mut payload = Vec::new();
179 for value in values { encode_element(&mut payload, field.typ, value, schema); }
180 put_varint(&mut output, field.number * 8 + 2); put_varint(&mut output, payload.len() as i64); output.extend(payload);
181 } else {
182 for value in values {
183 put_varint(&mut output, field.number * 8 + i64::from(schema.wire(field.typ)));
184 encode_element(&mut output, field.typ, value, schema);
185 }
186 }
187 output
188}
189fn encode(message: &Message, schema: &Schema) -> Vec<u8> {
190 let mut output = Vec::new();
191 for (i, field) in schema.fields(message.typ).iter().enumerate() { output.extend(encode_field(field, &message.fields[i], schema)); }
192 for unknown in &message.unknown {
193 put_varint(&mut output, unknown.number * 8 + i64::from(unknown.wire));
194 if unknown.wire == 0 { put_varint(&mut output, unknown.integer); }
195 else {
196 if unknown.wire == 2 { put_varint(&mut output, unknown.bytes.len() as i64); }
197 output.extend(&unknown.bytes);
198 }
199 }
200 output
201}
202fn hex_digits(bytes: &[u8]) -> String { bytes.iter().map(|byte| format!("{byte:02x}")).collect() }
203fn hex(bytes: &[u8]) -> String { if bytes.is_empty() { "-".into() } else { hex_digits(bytes) } }
204fn byte_notation(bytes: &[u8]) -> String { format!("<{}>", hex_digits(bytes)) }
205fn quote(bytes: &[u8], json: bool) -> String {
206 let mut output = String::from("\"");
207 for &byte in bytes {
208 match byte {
209 b'"' => output.push_str("\\\""), b'\\' => output.push_str("\\\\"), b'\n' => output.push_str("\\n"), b'\t' => output.push_str("\\t"),
210 32..=126 => output.push(char::from(byte)),
211 _ => { write!(output, "{}{:02x}", if json { "\\u00" } else { "\\x" }, byte).unwrap(); }
212 }
213 }
214 output.push('"'); output
215}
216fn base64(bytes: &[u8]) -> String {
217 const ALPHABET: &[u8] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
218 let mut result = String::new();
219 for chunk in bytes.chunks(3) {
220 let number = u32::from(chunk[0]) * 65536 + u32::from(*chunk.get(1).unwrap_or(&0)) * 256 + u32::from(*chunk.get(2).unwrap_or(&0));
221 result.push(char::from(ALPHABET[(number / 262144) as usize]));
222 result.push(char::from(ALPHABET[(number / 4096 % 64) as usize]));
223 result.push(if chunk.len() > 1 { char::from(ALPHABET[(number / 64 % 64) as usize]) } else { '=' });
224 result.push(if chunk.len() > 2 { char::from(ALPHABET[(number % 64) as usize]) } else { '=' });
225 }
226 result
227}
228fn notation(typ: usize, value: &Value, json: bool, schema: &Schema) -> String {
229 match value {
230 Value::Number(number) => {
231 if let Some(name) = schema.enum_name(typ, *number) { if json { format!("\"{name}\"") } else { name.into() } }
232 else if json && matches!(schema.scalar(typ), Some(Scalar::Uint64 | Scalar::Sint64 | Scalar::Fixed64)) { format!("\"{number}\"") }
233 else { number.to_string() }
234 }
235 Value::Boolean(state) => state.to_string(),
236 Value::Binary(bytes) => if schema.scalar(typ) == Some(Scalar::String) { quote(bytes, json) }
237 else if json { format!("\"{}\"", base64(bytes)) } else { byte_notation(bytes) },
238 Value::Message(message) => message_notation(message, json, schema),
239 }
240}
241fn sequence_notation(typ: usize, values: &[Value], json: bool, schema: &Schema) -> String {
242 format!("[{}]", values.iter().map(|value| notation(typ, value, json, schema)).collect::<Vec<_>>().join(if json { "," } else { ", " }))
243}
244fn message_notation(message: &Message, json: bool, schema: &Schema) -> String {
245 let mut items = Vec::new();
246 for (i, field) in schema.fields(message.typ).iter().enumerate() {
247 let values = &message.fields[i];
248 if values.is_empty() { continue; }
249 let key = if json {
250 let mut name = String::new(); let mut uppercase = false;
251 for c in field.name.chars() {
252 if c == '_' { uppercase = true; }
253 else { name.push(if uppercase { c.to_ascii_uppercase() } else { c }); uppercase = false; }
254 }
255 format!("\"{name}\":")
256 } else { format!("{}: ", field.name) };
257 let value = if field.repeated { sequence_notation(field.typ, values, json, schema) } else { notation(field.typ, &values[0], json, schema) };
258 items.push(key + &value);
259 }
260 format!("{{{}}}", items.join(if json { "," } else { ", " }))
261}
262fn wire_name(wire: u8) -> &'static str { match wire { 0 => "varint", 1 => "i64", 2 => "len", _ => "i32" } }
263fn text_contents(message: &Message, depth: usize, schema: &Schema, lines: &mut Vec<String>) {
264 let indent = " ".repeat(depth);
265 for (i, field) in schema.fields(message.typ).iter().enumerate() {
266 for value in &message.fields[i] {
267 if let Value::Message(child) = value {
268 lines.push(format!("{indent}{} {{", field.name)); text_contents(child, depth + 1, schema, lines); lines.push(format!("{indent}}}"));
269 } else { lines.push(format!("{indent}{}: {}", field.name, notation(field.typ, value, false, schema))); }
270 }
271 }
272 for unknown in &message.unknown {
273 let raw = if unknown.wire == 0 { unknown.integer.to_string() } else { byte_notation(&unknown.bytes) };
274 lines.push(format!("{indent}{}: {} {raw}", unknown.number, wire_name(unknown.wire)));
275 }
276}
277fn text_message(message: &Message, schema: &Schema) -> Vec<String> {
278 let mut lines = vec![format!("{} {{", schema.types[message.typ].name)];
279 text_contents(message, 1, schema, &mut lines); lines.push("}".into()); lines
280}

282struct Raw<'a> { integer: i64, bytes: &'a [u8], position: usize, payload: usize }
283struct Decoder<'a> { data: &'a [u8], schema: &'a Schema, explain: bool, lines: Vec<String> }
284fn byte_error(reason: &str, position: usize) -> String { format!("{reason} at byte {position}") }
285
286impl<'a> Decoder<'a> {
287 fn varint(&self, position: &mut usize, end: usize) -> Result<i64> {
288 let start = *position; let mut number = 0;
289 for i in 0..8 {
290 if *position >= end { return Err(byte_error("truncated varint", start)); }
291 let byte = self.data[*position]; *position += 1;
292 number += i64::from(byte & 127) << (7 * i);
293 if byte < 128 {
294 if number > MAX_UNSIGNED { return Err(byte_error("varint too large", start)); }
295 return Ok(number);
296 }
297 }
298 Err(byte_error("varint too long", start))
299 }
300 fn raw(&self, wire: u8, position: &mut usize, end: usize) -> Result<Raw<'a>> {
301 let start = *position;
302 if wire == 0 { return Ok(Raw { integer: self.varint(position, end)?, bytes: &[], position: start, payload: start }); }
303 let length = if wire == 2 { self.varint(position, end)? } else if wire == 1 { 8 } else { 4 };
304 if length > (end - *position) as i64 {
305 return Err(byte_error(match wire { 2 => "length past end", 1 => "truncated i64", _ => "truncated i32" }, start));
306 }
307 let payload = *position; *position += length as usize;
308 Ok(Raw { integer: 0, bytes: &self.data[payload..*position], position: start, payload })
309 }
310 fn convert(&self, typ: usize, raw: Raw<'_>) -> Result<Value> {
311 let scalar = self.schema.scalar(typ);
312 if matches!(scalar, Some(Scalar::String | Scalar::Bytes)) {
313 if scalar == Some(Scalar::String) {
314 if let Some(i) = raw.bytes.iter().position(|&byte| byte >= 128) { return Err(byte_error("invalid string", raw.payload + i)); }
315 }
316 return Ok(Value::Binary(raw.bytes.to_vec()));
317 }
318 let mut number = raw.integer;
319 if matches!(self.schema.wire(typ), 1 | 5) {
320 if scalar == Some(Scalar::Fixed64) && (raw.bytes[7] != 0 || raw.bytes[6] >= 32) { return Err(byte_error("value out of range", raw.position)); }
321 number = raw.bytes.iter().rev().fold(0, |value, &byte| value * 256 + i64::from(byte));
322 }
323 if matches!(scalar, Some(Scalar::Sint32 | Scalar::Sint64)) { number = unzigzag(number); }
324 if scalar == Some(Scalar::Sfixed32) && number >= 2_147_483_648 { number -= 4_294_967_296; }
325 if scalar.is_some_and(|kind| !kind.contains(number)) || (scalar == Some(Scalar::Bool) && number > 1) { return Err(byte_error("value out of range", raw.position)); }
326 if matches!(&self.schema.types[typ].definition, Definition::Enum(_)) && self.schema.enum_name(typ, number).is_none() { return Err(byte_error("unknown enum number", raw.position)); }
327 Ok(if scalar == Some(Scalar::Bool) { Value::Boolean(number != 0) } else { Value::Number(number) })
328 }
329 fn message(&mut self, typ: usize, mut position: usize, end: usize, depth: usize) -> Result<Message> {
330 let schema = self.schema;
331 let mut message = Message::new(typ, schema);
332 while position < end {
333 let start = position; let tag = self.varint(&mut position, end)?;
334 let number = tag / 8; let wire = (tag % 8) as u8;
335 if number == 0 { return Err(byte_error("invalid field number", start)); }
336 if !matches!(wire, 0 | 1 | 2 | 5) { return Err(byte_error("invalid wire type", start)); }
337 let index = schema.fields(typ).iter().position(|field| field.number == number);
338 let field = index.map(|i| &schema.fields(typ)[i]);
339 let packed = field.is_some_and(|field| field.repeated && schema.wire(field.typ) != 2 && wire == 2);
340 if field.is_some_and(|field| wire != schema.wire(field.typ) && !packed) { return Err(byte_error("wire type mismatch", start)); }
341 let raw = self.raw(wire, &mut position, end)?;
342 let mut header = format!("{}{start} {number} {} {}", " ".repeat(depth), field.map_or("?", |field| field.name.as_str()), wire_name(wire));
343 if wire == 2 { write!(header, " {}", raw.bytes.len()).unwrap(); }
344 let Some(field) = field else {
345 if self.explain {
346 let value = if wire == 0 { raw.integer.to_string() } else { byte_notation(raw.bytes) };
347 self.lines.push(format!("{header} {value}"));
348 } else { message.unknown.push(Unknown { number, wire, integer: raw.integer, bytes: raw.bytes.to_vec() }); }
349 continue;
350 };
351 let mut values = Vec::new();
352 if schema.is_message(field.typ) {
353 if self.explain { self.lines.push(header); }
354 values.push(Value::Message(self.message(field.typ, raw.payload, raw.payload + raw.bytes.len(), depth + 1)?));
355 } else {
356 if packed {
357 let mut element = raw.payload; let limit = raw.payload + raw.bytes.len();
358 while element < limit { values.push(self.convert(field.typ, self.raw(schema.wire(field.typ), &mut element, limit)?)?); }
359 } else { values.push(self.convert(field.typ, raw)?); }
360 if self.explain {
361 if packed { header.push_str(" packed"); }
362 for value in &values { header.push(' '); header.push_str(&notation(field.typ, value, false, schema)); }
363 self.lines.push(header);
364 }
365 }
366 if !self.explain { message.apply(index.unwrap(), values, schema); }
367 }
368 Ok(message)
369 }
370}
371fn decode(typ: usize, bytes: &[u8], schema: &Schema) -> Result<Message> {
372 let mut decoder = Decoder { data: bytes, schema, explain: false, lines: Vec::new() };
373 let message = decoder.message(typ, 0, bytes.len(), 0)?; message.required(schema, "")?; Ok(message)
374}

376#[derive(Clone)]
377enum TokenKind { Symbol(u8), Integer(i64), Name(String), String(Vec<u8>), Bytes(Vec<u8>), End }
378#[derive(Clone)]
379struct Token { kind: TokenKind, column: usize }
380#[derive(PartialEq, Eq)]
381enum Shape { None, Message, List, Integer, Boolean, Name, String, Bytes }
382impl Token {
383 fn shape(&self) -> Shape {
384 match &self.kind {
385 TokenKind::Symbol(b'{') => Shape::Message, TokenKind::Symbol(b'[') => Shape::List,
386 TokenKind::Integer(_) => Shape::Integer,
387 TokenKind::Name(name) => if name == "true" || name == "false" { Shape::Boolean } else { Shape::Name },
388 TokenKind::String(_) => Shape::String, TokenKind::Bytes(_) => Shape::Bytes, _ => Shape::None,
389 }
390 }
391}
392fn column_error(reason: &str, column: usize) -> String { format!("{reason} at column {column}") }
393fn hex_digit(byte: u8) -> Option<u8> {
394 match byte { b'0'..=b'9' => Some(byte - b'0'), b'a'..=b'f' => Some(byte - b'a' + 10), _ => None }
395}
396fn tokenize(source: &str) -> Result<Vec<Token>> {
397 let text = source.as_bytes(); let mut position = 0; let mut tokens = Vec::new();
398 while position < text.len() {
399 let c = text[position];
400 if c == b' ' { position += 1; continue; }
401 let start = position; position += 1;
402 let syntax = || column_error("syntax error", start + 1);
403 let kind = if b"{}[]:,".contains(&c) { TokenKind::Symbol(c) }
404 else if c == b'"' {
405 let mut bytes = Vec::new();
406 while position < text.len() && text[position] != b'"' {
407 let mut byte = text[position]; position += 1;
408 if byte == b'\\' {
409 let escaped = *text.get(position).ok_or_else(syntax)?; position += 1;
410 byte = match escaped {
411 b'"' | b'\\' => escaped, b'n' => b'\n', b't' => b'\t',
412 b'x' => {
413 let high = text.get(position).copied().and_then(hex_digit).ok_or_else(syntax)?;
414 let low = text.get(position + 1).copied().and_then(hex_digit).ok_or_else(syntax)?;
415 if high >= 8 { return Err(syntax()); }
416 position += 2; high * 16 + low
417 }
418 _ => return Err(syntax()),
419 };
420 }
421 bytes.push(byte);
422 }
423 if position == text.len() { return Err(syntax()); }
424 position += 1; TokenKind::String(bytes)
425 } else if c == b'<' {
426 let end = text[position..].iter().position(|&byte| byte == b'>').map(|i| position + i).ok_or_else(syntax)?;
427 if (end - position) % 2 != 0 { return Err(syntax()); }
428 let mut bytes = Vec::new();
429 for pair in text[position..end].chunks(2) { bytes.push(hex_digit(pair[0]).ok_or_else(syntax)? * 16 + hex_digit(pair[1]).ok_or_else(syntax)?); }
430 position = end + 1; TokenKind::Bytes(bytes)
431 } else if c == b'-' || c.is_ascii_digit() {
432 if c == b'-' && !text.get(position).is_some_and(u8::is_ascii_digit) { return Err(syntax()); }
433 while text.get(position).is_some_and(u8::is_ascii_digit) { position += 1; }
434 TokenKind::Integer(source[start..position].parse().unwrap())
435 } else if c.is_ascii_alphabetic() || c == b'_' {
436 while text.get(position).is_some_and(|byte| byte.is_ascii_alphanumeric() || *byte == b'_') { position += 1; }
437 TokenKind::Name(source[start..position].into())
438 } else { return Err(syntax()); };
439 tokens.push(Token { kind, column: start + 1 });
440 }
441 tokens.push(Token { kind: TokenKind::End, column: text.len() + 1 }); Ok(tokens)
442}
443struct Parser<'a> { tokens: Vec<Token>, position: usize, schema: &'a Schema }
444impl<'a> Parser<'a> {
445 fn new(source: &str, schema: &'a Schema) -> Result<Self> { Ok(Self { tokens: tokenize(source)?, position: 0, schema }) }
446 fn next(&self) -> &Token { &self.tokens[self.position] }
447 fn symbol(&self, c: u8) -> bool { matches!(self.next().kind, TokenKind::Symbol(byte) if byte == c) }
448 fn expect(&mut self, c: u8) -> Result<()> {
449 if !self.symbol(c) { return Err(column_error("syntax error", self.next().column)); }
450 self.position += 1; Ok(())
451 }
452 fn end(&self) -> Result<()> {
453 if matches!(self.next().kind, TokenKind::End) { Ok(()) } else { Err(column_error("syntax error", self.next().column)) }
454 }
455 fn shape(&self, wanted: Shape) -> Result<()> {
456 let actual = self.next().shape();
457 if actual == Shape::None { Err(column_error("syntax error", self.next().column)) }
458 else if actual != wanted { Err(column_error("type mismatch", self.next().column)) } else { Ok(()) }
459 }
460 fn element(&mut self, typ: usize) -> Result<Value> {
461 let wanted = match &self.schema.types[typ].definition {
462 Definition::Message(_) => Shape::Message, Definition::Enum(_) => Shape::Name,
463 Definition::Scalar(kind) => match kind { Scalar::Bool => Shape::Boolean, Scalar::String => Shape::String, Scalar::Bytes => Shape::Bytes, _ => Shape::Integer },
464 };
465 self.shape(wanted)?;
466 if self.schema.is_message(typ) { return Ok(Value::Message(self.message(typ)?)); }
467 let token = self.next().clone(); self.position += 1;
468 match token.kind {
469 TokenKind::Integer(number) => {
470 if !self.schema.scalar(typ).unwrap().contains(number) { return Err(column_error("out of range", token.column)); }
471 Ok(Value::Number(number))
472 }
473 TokenKind::Name(name) => {
474 if self.schema.scalar(typ) == Some(Scalar::Bool) { return Ok(Value::Boolean(name == "true")); }
475 if let Definition::Enum(values) = &self.schema.types[typ].definition {
476 if let Some((&number, _)) = values.iter().find(|(_, value)| *value == &name) { return Ok(Value::Number(number)); }
477 }
478 Err(column_error("unknown enum value", token.column))
479 }
480 TokenKind::String(bytes) | TokenKind::Bytes(bytes) => Ok(Value::Binary(bytes)),
481 _ => unreachable!(),
482 }
483 }
484 fn field(&mut self, field: &Field) -> Result<Vec<Value>> {
485 if !field.repeated { return Ok(vec![self.element(field.typ)?]); }
486 self.shape(Shape::List)?; self.expect(b'[')?;
487 let mut values = Vec::new();
488 if !self.symbol(b']') {
489 values.push(self.element(field.typ)?);
490 while self.symbol(b',') { self.position += 1; values.push(self.element(field.typ)?); }
491 }
492 self.expect(b']')?; Ok(values)
493 }
494 fn message(&mut self, typ: usize) -> Result<Message> {
495 self.expect(b'{')?;
496 let schema = self.schema; let fields = schema.fields(typ);
497 let mut message = Message::new(typ, schema); let mut seen = vec![false; fields.len()];
498 if !self.symbol(b'}') {
499 loop {
500 let token = self.next();
501 let TokenKind::Name(name) = &token.kind else { return Err(column_error("syntax error", token.column)); };
502 let index = fields.iter().position(|field| &field.name == name).ok_or_else(|| column_error("unknown field", token.column))?;
503 let field = &fields[index];
504 if seen[index] { return Err(column_error("duplicate field", token.column)); }
505 if let Some(group) = &field.group {
506 if fields.iter().enumerate().any(|(i, other)| seen[i] && other.group.as_ref() == Some(group)) { return Err(column_error("oneof conflict", token.column)); }
507 }
508 self.position += 1; seen[index] = true; self.expect(b':')?;
509 message.fields[index] = self.field(field)?;
510 if !self.symbol(b',') { break; }
511 self.position += 1;
512 }
513 }
514 let column = self.next().column; self.expect(b'}')?;
515 for (i, field) in fields.iter().enumerate() {
516 if field.required && !seen[i] { return Err(column_error(&format!("missing required field {}", field.name), column)); }
517 }
518 Ok(message)
519 }
520}
521
522struct Segment { owner: usize, field: usize, index: Option<usize> }
523impl Schema {
524 fn path(&self, mut typ: usize, source: &str) -> Result<Vec<Segment>> {
525 let parts: Vec<_> = source.split('.').collect(); let mut path = Vec::new();
526 for (i, part) in parts.iter().enumerate() {
527 let (name, index) = if let Some((name, suffix)) = part.split_once('[') {
528 let digits = suffix.strip_suffix(']').ok_or("invalid path")?;
529 if digits.is_empty() || digits.len() > 6 || !digits.bytes().all(|byte| byte.is_ascii_digit()) { return Err("invalid path".into()); }
530 (name, Some(digits.parse::<usize>().unwrap()))
531 } else { (*part, None) };
532 let index_of_field = self.fields(typ).iter().position(|field| field.name == name).ok_or("invalid path")?;
533 let field = &self.fields(typ)[index_of_field];
534 if index.is_some() && !field.repeated { return Err("invalid path".into()); }
535 if i + 1 < parts.len() && (!self.is_message(field.typ) || field.repeated && index.is_none()) { return Err("invalid path".into()); }
536 path.push(Segment { owner: typ, field: index_of_field, index }); typ = field.typ;
537 }
538 Ok(path)
539 }
540 fn tree(&self, typ: usize, chain: &mut Vec<usize>, lines: &mut Vec<String>) {
541 chain.push(typ);
542 for field in self.fields(typ) {
543 let mode = if field.required { "required ".into() } else if field.packed { "repeated packed ".into() }
544 else if field.repeated { "repeated ".into() } else if let Some(group) = &field.group { format!("oneof {group} ") } else { String::new() };
545 let mut line = format!("{}{} {}: {mode}{}", " ".repeat(chain.len()), field.number, field.name, self.types[field.typ].name);
546 if let Definition::Enum(values) = &self.types[field.typ].definition {
547 write!(line, " ({})", values.iter().map(|(n, name)| format!("{name}={n}")).collect::<Vec<_>>().join(", ")).unwrap();
548 }
549 let recursive = self.is_message(field.typ) && chain.contains(&field.typ);
550 if recursive { line.push_str(" (recursive)"); }
551 lines.push(line);
552 if self.is_message(field.typ) && !recursive { self.tree(field.typ, chain, lines); }
553 }
554 chain.pop();
555 }
556}
557fn get_path(message: &Message, path: &[Segment], schema: &Schema) -> String {
558 let mut current = Some(message);
559 for (i, segment) in path.iter().enumerate() {
560 let field = &schema.fields(segment.owner)[segment.field];
561 let values = current.map_or(&[][..], |message| message.fields[segment.field].as_slice());
562 if i + 1 == path.len() && field.repeated && segment.index.is_none() { return sequence_notation(field.typ, values, false, schema); }
563 let value = values.get(segment.index.unwrap_or(0));
564 if i + 1 == path.len() { return value.map_or_else(|| "(none)".into(), |value| notation(field.typ, value, false, schema)); }
565 current = match value { Some(Value::Message(child)) => Some(child), _ => None };
566 }
567 unreachable!()
568}
569fn edit_path(message: &mut Message, path: &[Segment], replacement: Option<Vec<Value>>, schema: &Schema) -> Result<()> {
570 let segment = &path[0]; let field = &schema.fields(message.typ)[segment.field];
571 let index = segment.index.unwrap_or(0);
572 if path.len() == 1 {
573 if segment.index.is_some() {
574 let values = &mut message.fields[segment.field];
575 if index >= values.len() { if replacement.is_some() { return Err("index out of range".into()); } }
576 else if let Some(mut replacement) = replacement { values[index] = replacement.pop().unwrap(); }
577 else { values.remove(index); }
578 } else if let Some(replacement) = replacement { message.choose(segment.field, schema); message.fields[segment.field] = replacement; }
579 else { message.fields[segment.field].clear(); }
580 return Ok(());
581 }
582 if index >= message.fields[segment.field].len() {
583 if replacement.is_none() { return Ok(()); }
584 if segment.index.is_some() { return Err("index out of range".into()); }
585 message.choose(segment.field, schema); message.fields[segment.field].push(Value::Message(Message::new(field.typ, schema)));
586 }
587 if let Value::Message(child) = &mut message.fields[segment.field][index] { edit_path(child, &path[1..], replacement, schema)?; }
588 Ok(())
589}
590fn equal(a: &Value, b: &Value) -> bool {
591 match (a, b) { (Value::Number(x), Value::Number(y)) => x == y, (Value::Boolean(x), Value::Boolean(y)) => x == y, (Value::Binary(x), Value::Binary(y)) => x == y, _ => false }
592}
593fn differences(a: &Message, b: &Message, prefix: &str, schema: &Schema, lines: &mut Vec<String>) {
594 for (i, field) in schema.fields(a.typ).iter().enumerate() {
595 let x = &a.fields[i]; let y = &b.fields[i];
596 for j in 0..x.len().max(y.len()) {
597 let path = if field.repeated { format!("{prefix}{}[{j}]", field.name) } else { format!("{prefix}{}", field.name) };
598 match (x.get(j), y.get(j)) {
599 (Some(Value::Message(left)), Some(Value::Message(right))) => differences(left, right, &(path + "."), schema, lines),
600 (Some(left), Some(right)) if equal(left, right) => {},
601 (left, right) => {
602 let render = |value: Option<&Value>| value.map_or_else(|| "(none)".into(), |value| notation(field.typ, value, false, schema));
603 lines.push(format!("{path}: {} -> {}", render(left), render(right)));
604 }
605 }
606 }
607 }
608}
609fn parse_hex(source: &str) -> Result<Vec<u8>> {
610 if source == "-" { return Ok(Vec::new()); }
611 if source.is_empty() || source.len() % 2 != 0 { return Err("invalid hex".into()); }
612 source.as_bytes().chunks(2).map(|pair| Ok(hex_digit(pair[0]).ok_or("invalid hex")? * 16 + hex_digit(pair[1]).ok_or("invalid hex")?)).collect()
613}
614fn rest(source: &str, count: usize) -> &str { source.splitn(count + 1, ' ').nth(count).unwrap() }
615
616impl Schema {
617 fn execute(&self, source: &str) -> Result<Vec<String>> {
618 let words: Vec<_> = source.split(' ').collect(); let command = words[0];
619 let count = match command {
620 "VARINT" | "ZIGZAG" | "SCHEMA" => 2,
621 "ENCODE" | "SIZE" | "DECODE" | "LITERAL" | "JSON" | "CANON" | "EXPLAIN" => 3,
622 "GET" | "CLEAR" | "MERGE" | "DIFF" => 4, "PATCH" => 5, _ => return Err("unknown command".into()),
623 };
624 if words.len() < count || (!matches!(command, "ENCODE" | "SIZE" | "PATCH") && words.len() != count) { return Err("wrong number of arguments".into()); }
625 if matches!(command, "VARINT" | "ZIGZAG") {
626 let digits = words[1].strip_prefix('-').unwrap_or(words[1]);
627 if digits.is_empty() || digits.len() > 18 || !digits.bytes().all(|byte| byte.is_ascii_digit()) { return Err("invalid number".into()); }
628 let number: i64 = words[1].parse().unwrap();
629 if (command == "VARINT" && !(0..=MAX_UNSIGNED).contains(&number)) || (command == "ZIGZAG" && !Scalar::Sint64.contains(number)) { return Err("invalid number".into()); }
630 let value = if command == "ZIGZAG" { zigzag(number) } else { number };
631 let mut bytes = Vec::new(); put_varint(&mut bytes, value);
632 return Ok(vec![if command == "ZIGZAG" { format!("{value} {}", hex(&bytes)) } else { hex(&bytes) }]);
633 }
634 let typ = *self.names.get(words[1]).filter(|&&typ| self.is_message(typ)).ok_or("unknown message type")?;
635 if command == "SCHEMA" {
636 let mut lines = vec![self.types[typ].name.clone()]; self.tree(typ, &mut Vec::new(), &mut lines); return Ok(lines);
637 }
638 if matches!(command, "ENCODE" | "SIZE") {
639 let mut parser = Parser::new(rest(source, 2), self)?; let message = parser.message(typ)?; parser.end()?;
640 let encoded = encode(&message, self);
641 if command == "ENCODE" { return Ok(vec![hex(&encoded)]); }
642 let mut line = encoded.len().to_string();
643 for (i, field) in self.fields(typ).iter().enumerate() {
644 if !message.fields[i].is_empty() { write!(line, " {}={}", field.name, encode_field(field, &message.fields[i], self).len()).unwrap(); }
645 }
646 return Ok(vec![line]);
647 }
648 let mut first = parse_hex(words[2])?;
649 let second = if matches!(command, "MERGE" | "DIFF") { parse_hex(words[3])? } else { Vec::new() };
650 let path = if matches!(command, "GET" | "PATCH" | "CLEAR") { self.path(typ, words[3])? } else { Vec::new() };
651 if command == "EXPLAIN" {
652 let mut decoder = Decoder { data: &first, schema: self, explain: true, lines: Vec::new() };
653 decoder.message(typ, 0, first.len(), 0)?;
654 if first.is_empty() { decoder.lines.push("(empty)".into()); }
655 return Ok(decoder.lines);
656 }
657 if command == "DIFF" {
658 let a = decode(typ, &first, self).map_err(|error| error + " in first")?;
659 let b = decode(typ, &second, self).map_err(|error| error + " in second")?;
660 let mut lines = Vec::new(); differences(&a, &b, "", self, &mut lines);
661 if lines.is_empty() { lines.push("SAME".into()); } return Ok(lines);
662 }
663 if command == "MERGE" { first.extend(second); }
664 let mut message = decode(typ, &first, self)?;
665 match command {
666 "DECODE" | "MERGE" => Ok(text_message(&message, self)),
667 "LITERAL" | "JSON" => Ok(vec![message_notation(&message, command == "JSON", self)]),
668 "GET" => Ok(vec![get_path(&message, &path, self)]),
669 _ => {
670 if command == "PATCH" {
671 let last = path.last().unwrap(); let field = &self.fields(last.owner)[last.field];
672 let mut parser = Parser::new(rest(source, 4), self)?;
673 let replacement = if field.repeated && last.index.is_none() { parser.field(field)? } else { vec![parser.element(field.typ)?] };
674 parser.end()?; edit_path(&mut message, &path, Some(replacement), self)?; message.required(self, "")?;
675 } else if command == "CLEAR" { edit_path(&mut message, &path, None, self)?; message.required(self, "")?; }
676 Ok(vec![hex(&encode(&message, self))])
677 }
678 }
679 }
680}

682fn main() {
683 let mut input = String::new(); io::stdin().read_to_string(&mut input).unwrap();
684 let mut lines = input.lines(); let schema = Schema::read(&mut lines);
685 let count: usize = lines.next().unwrap().parse().unwrap();
686 let mut output = io::BufWriter::new(io::stdout().lock());
687 for source in lines.take(count) {
688 match schema.execute(source) {
689 Ok(lines) => for line in lines { writeln!(output, "{line}").unwrap(); },
690 Err(error) => writeln!(output, "ERROR {error}").unwrap(),
691 }
692 }
