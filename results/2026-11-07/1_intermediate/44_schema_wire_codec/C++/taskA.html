30Lines split(const std::string& text, char separator = ' ') {
31 Lines result;
32 std::size_t start = 0;
33 while (true) {
34 std::size_t end = text.find(separator, start);
35 if (end == std::string::npos) { result.push_back(text.substr(start)); return result; }
36 result.push_back(text.substr(start, end - start)); start = end + 1;
37 }
38}

40 std::string result;
41 for (std::size_t index = 0; index < lines.size(); ++index) {
42 if (index) result += separator;
43 result += lines[index];
44 }
45 return result;
46}

53bool digit(char character) { return character >= '0' && character <= '9'; }
54bool letter(char character) { return (character >= 'a' && character <= 'z') || (character >= 'A' && character <= 'Z'); }
55Int decimal(const std::string& text) {
56 bool negative = text.front() == '-';
57 Int value = 0;
58 for (std::size_t index = negative ? 1 : 0; index < text.size(); ++index) value = value * 10 + text[index] - '0';
59 return negative ? -value : value;
60}

116struct Schema {
117 std::map<std::string, Type> types;
118 explicit Schema(std::istream& input) {
119 for (const Type& type : std::vector<Type>{
120 {"uint32", Wire::Varint, IntegerSpec{0, 4294967295LL}},
121 {"uint64", Wire::Varint, IntegerSpec{0, Int(maximum)}},
122 {"sint32", Wire::Varint, IntegerSpec{-2147483648LL, 2147483647LL}},
123 {"sint64", Wire::Varint, IntegerSpec{signedMinimum, signedMaximum}},
124 {"fixed32", Wire::I32, IntegerSpec{0, 4294967295LL}},
125 {"sfixed32", Wire::I32, IntegerSpec{-2147483648LL, 2147483647LL}},
126 {"fixed64", Wire::I64, IntegerSpec{0, Int(maximum)}},
127 {"bool", Wire::Varint, BooleanSpec{}}, {"string", Wire::Length, StringSpec{}}, {"bytes", Wire::Length, BytesSpec{}},
128 }) types.emplace(type.name, type);
129 Type* current = nullptr;
130 for (std::string line = readLine(input); line != "END SCHEMA"; line = readLine(input)) {
131 Lines words = split(line);
132 if (words[0] == "ENUM" || words[0] == "MESSAGE") {
133 TypeData data = words[0] == "ENUM" ? TypeData(EnumSpec{}) : TypeData(MessageSpec{});
134 Wire wire = words[0] == "ENUM" ? Wire::Varint : Wire::Length;
135 current = &types.emplace(words[1], Type{words[1], wire, std::move(data)}).first->second;
136 } else if (words[0] == "VALUE") {
137 EnumSpec& definition = std::get<EnumSpec>(current->data);
138 Int number = decimal(words[2]);
139 definition.byName.emplace(words[1], number); definition.byNumber.emplace(number, words[1]);
140 } else if (words[0] == "FIELD") {
141 Field field;
142 field.number = UInt(decimal(words[1])); field.typeName = words[2]; field.name = words[3];
143 std::string mode = words.size() > 4 ? words[4] : "";
144 field.required = mode == "REQUIRED"; field.repeated = mode == "REPEATED";
145 field.packed = words.size() > 5 && words[5] == "PACKED";
146 if (mode == "ONEOF") field.group = words[5];
147 std::get<MessageSpec>(current->data).fields.push_back(std::move(field));
148 } else current = nullptr;
149 }
150 for (auto& [name, type] : types) {
151 if (!type.message()) continue;
152 MessageSpec& definition = std::get<MessageSpec>(type.data);
153 std::sort(definition.fields.begin(), definition.fields.end(), [](const Field& left, const Field& right) { return left.number < right.number; });
154 for (std::size_t index = 0; index < definition.fields.size(); ++index) {
155 Field& field = definition.fields[index]; field.type = &types.at(field.typeName);
156 definition.byName.emplace(field.name, index); definition.byNumber.emplace(field.number, index);
157 }
158 }
159 }
160 Lines tree(const Type& type) const {
161 Lines result{type.name};
162 std::vector<const Type*> chain{&type};
163 std::function<void(const Type&, std::size_t)> visit = [&](const Type& current, std::size_t depth) {
164 for (const Field& field : current.fields()) {
165 std::string kind = field.required ? "required " : field.packed ? "repeated packed " : field.repeated ? "repeated "
166 : field.group ? "oneof " + *field.group + " " : "";
167 std::string line = std::string(depth * 2, ' ') + std::to_string(field.number) + " " + field.name + ": " + kind + field.type->name;
168 if (const EnumSpec* enumeration = std::get_if<EnumSpec>(&field.type->data)) {
169 Lines values;
170 for (const auto& [number, name] : enumeration->byNumber) values.push_back(name + "=" + std::to_string(number));
171 line += " (" + joined(values, ", ") + ")";
172 }
173 bool recursive = field.type->message() && std::find(chain.begin(), chain.end(), field.type) != chain.end();
174 if (recursive) line += " (recursive)";
175 result.push_back(line);
176 if (field.type->message() && !recursive) { chain.push_back(field.type); visit(*field.type, depth + 1); chain.pop_back(); }
177 }
178 };
179 visit(type, 1); return result;
180 }
181};

183struct Message;
184using MessagePtr = std::shared_ptr<Message>;
185using Value = std::variant<Int, bool, std::string, Bytes, MessagePtr>;
186using Values = std::vector<Value>;
187using Raw = std::variant<UInt, Bytes>;
188struct UnknownField { UInt number; Wire wire; Raw raw; };
189struct Message {
190 const Type* type;
191 std::map<UInt, Values> values;
192 std::vector<UnknownField> unknown;
193 explicit Message(const Type& type) : type(&type) {}
194 const Values& sequence(const Field& field) const {
195 static const Values empty;
196 std::map<UInt, Values>::const_iterator found = values.find(field.number);
197 return found == values.end() ? empty : found->second;
198 }
199 void replace(const Field& field, Values items) {
200 if (field.group) for (const Field& other : type->fields()) if (other.number != field.number && other.group == field.group) values.erase(other.number);
201 if (items.empty()) values.erase(field.number);
202 else values.insert_or_assign(field.number, std::move(items));
203 }
204 void apply(const Field& field, const Values& items) {
205 if (field.repeated) {
206 Values combined = sequence(field); combined.insert(combined.end(), items.begin(), items.end()); replace(field, std::move(combined));
207 } else if (field.type->message() && !sequence(field).empty()) {
208 Values previous = sequence(field);
209 std::get<MessagePtr>(previous.front())->merge(*std::get<MessagePtr>(items.front()));
210 replace(field, std::move(previous));
211 } else replace(field, items);
212 }
213 void merge(const Message& other) {
214 for (const Field& field : type->fields()) if (!other.sequence(field).empty()) apply(field, other.sequence(field));
215 unknown.insert(unknown.end(), other.unknown.begin(), other.unknown.end());
216 }
217 void checkRequired(const std::string& prefix = "") const {
218 for (const Field& field : type->fields()) {
219 const Values& items = sequence(field);
220 std::string path = prefix + field.name;
221 if (field.required && items.empty()) fail("missing required field " + path);
222 if (field.type->message()) for (std::size_t index = 0; index < items.size(); ++index) {
223 std::get<MessagePtr>(items[index])->checkRequired(path + (field.repeated ? "[" + std::to_string(index) + "]" : "") + ".");
224 }
225 }
226 }
227};

229Bytes varint(UInt value) {
230 Bytes result;
231 do { unsigned low = unsigned(value % 128); value /= 128; result.push_back(std::uint8_t(low + (value ? 128 : 0))); } while (value);
232 return result;
233}
234Bytes encode(const Message& message);
235Bytes valueBytes(const Type& type, const Value& value) {
236 if (const MessagePtr* message = std::get_if<MessagePtr>(&value)) return encode(**message);
237 if (const bool* boolean = std::get_if<bool>(&value)) return varint(*boolean ? 1 : 0);
238 if (const std::string* text = std::get_if<std::string>(&value)) return Bytes(text->begin(), text->end());
239 if (const Bytes* bytes = std::get_if<Bytes>(&value)) return *bytes;
240 Int integer = std::get<Int>(value);
241 UInt number = type.name == "sint32" || type.name == "sint64" ? zigzag(integer)
242 : type.name == "sfixed32" && integer < 0 ? UInt(integer + 4294967296LL) : UInt(integer);
243 if (type.wire == Wire::Varint) return varint(number);
244 Bytes result(type.wire == Wire::I32 ? 4 : 8);
245 for (std::uint8_t& byte : result) { byte = std::uint8_t(number % 256); number /= 256; }
246 return result;
247}
248Bytes record(UInt number, Wire wire, const Raw& raw) {
249 Bytes result = varint(number * 8 + UInt(wire));
250 if (wire == Wire::Varint) append(result, varint(std::get<UInt>(raw)));
251 else {
252 const Bytes& bytes = std::get<Bytes>(raw);
253 if (wire == Wire::Length) append(result, varint(bytes.size()));
254 append(result, bytes);
255 }
256 return result;
257}
258Bytes fieldBytes(const Field& field, const Values& values) {
259 Bytes result;
260 if (values.empty()) return result;
261 if (field.packed) {
262 for (const Value& value : values) append(result, valueBytes(*field.type, value));
263 return record(field.number, Wire::Length, result);
264 }
265 for (const Value& value : values) {
266 Bytes bytes = valueBytes(*field.type, value);
267 append(result, varint(field.number * 8 + UInt(field.type->wire)));
268 if (field.type->wire == Wire::Length) append(result, varint(bytes.size()));
269 append(result, bytes);
270 }
271 return result;
272}
273Bytes encode(const Message& message) {
274 Bytes result;
275 for (const Field& field : message.type->fields()) append(result, fieldBytes(field, message.sequence(field)));
276 for (const UnknownField& field : message.unknown) append(result, record(field.number, field.wire, field.raw));
277 return result;
278}

280std::string quote(const std::string& text, bool json = false) {
281 std::string result = "\"";
282 for (unsigned char byte : text) {
283 if (byte == '"' || byte == '\\') { result += '\\'; result += char(byte); }
284 else if (byte == '\n') result += "\\n";
285 else if (byte == '\t') result += "\\t";
286 else if (byte >= 32 && byte <= 126) result += char(byte);
287 else result += std::string(json ? "\\u00" : "\\x") + hexDigits(Bytes{byte});
288 }
289 return result + '"';
290}
291std::string camel(const std::string& name) {
292 std::string result;
293 bool upper = false;
294 for (char character : name) {
295 if (character == '_') upper = true;
296 else { result += upper && character >= 'a' && character <= 'z' ? char(character - 'a' + 'A') : character; upper = false; }
297 }
298 return result;
299}
300std::string base64(const Bytes& bytes) {
301 constexpr char alphabet[] = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
302 std::string result;
303 for (std::size_t index = 0; index < bytes.size(); index += 3) {
304 unsigned value = unsigned(bytes[index]) << 16;
305 if (index + 1 < bytes.size()) value += unsigned(bytes[index + 1]) << 8;
306 if (index + 2 < bytes.size()) value += bytes[index + 2];
307 result += alphabet[value / 262144]; result += alphabet[value / 4096 % 64];
308 result += index + 1 < bytes.size() ? alphabet[value / 64 % 64] : '=';
309 result += index + 2 < bytes.size() ? alphabet[value % 64] : '=';
310 }
311 return result;
312}

314std::string listNotation(const Type& type, const Values& values, bool json = false) {
315 Lines items;
316 for (const Value& value : values) items.push_back(notation(type, value, json));
317 return "[" + joined(items, json ? "," : ", ") + "]";
318}
319std::string notation(const Type& type, const Value& value, bool json) {
320 if (const MessagePtr* message = std::get_if<MessagePtr>(&value)) {
321 Lines items;
322 for (const Field& field : type.fields()) {
323 const Values& values = (*message)->sequence(field);
324 if (values.empty()) continue;
325 std::string key = json ? quote(camel(field.name), true) : field.name;
326 std::string text = field.repeated ? listNotation(*field.type, values, json) : notation(*field.type, values.front(), json);
327 items.push_back(key + (json ? ":" : ": ") + text);
328 }
329 return "{" + joined(items, json ? "," : ", ") + "}";
330 }
331 if (const std::string* text = std::get_if<std::string>(&value)) return quote(*text, json);
332 if (const Bytes* bytes = std::get_if<Bytes>(&value)) return json ? quote(base64(*bytes), true) : "<" + hexDigits(*bytes) + ">";
333 if (const bool* boolean = std::get_if<bool>(&value)) return *boolean ? "true" : "false";
334 Int integer = std::get<Int>(value);
335 if (const EnumSpec* enumeration = std::get_if<EnumSpec>(&type.data)) {
336 const std::string& name = enumeration->byNumber.at(integer);
337 return json ? quote(name, true) : name;
338 }
339 std::string text = std::to_string(integer);
340 return json && (type.name == "uint64" || type.name == "sint64" || type.name == "fixed64") ? quote(text, true) : text;
341}
342std::string rawText(const Raw& raw) {
343 if (const UInt* integer = std::get_if<UInt>(&raw)) return std::to_string(*integer);
344 return "<" + hexDigits(std::get<Bytes>(raw)) + ">";
345}
346Lines textFormat(const Message& message) {
347 Lines result{message.type->name + " {"};
348 std::function<void(const Message&, std::size_t)> visit = [&](const Message& current, std::size_t depth) {
349 std::string indent(depth * 2, ' ');
350 for (const Field& field : current.type->fields()) for (const Value& value : current.sequence(field)) {
351 if (field.type->message()) {
352 result.push_back(indent + field.name + " {"); visit(*std::get<MessagePtr>(value), depth + 1); result.push_back(indent + "}");
353 } else result.push_back(indent + field.name + ": " + notation(*field.type, value));
354 }
355 for (const UnknownField& field : current.unknown) result.push_back(indent + std::to_string(field.number) + ": " + wireName(field.wire) + " " + rawText(field.raw));
356 };
357 visit(message, 1); result.push_back("}"); return result;
358}

360struct ReadValue { Raw raw; std::size_t position, start; };
361struct Reader {
362 const Bytes& data;
363 std::size_t position, end;
364 UInt integer() {
365 std::size_t start = position;
366 UInt result = 0;
367 for (unsigned index = 0; index < 8; ++index) {
368 if (position == end) byteError("truncated varint", start);
369 unsigned byte = data[position++];
370 result += UInt(byte % 128) << (7 * index);
371 if (byte < 128) {
372 if (result > maximum) byteError("varint too large", start);
373 return result;
374 }
375 }
376 byteError("varint too long", start);
377 }
378 ReadValue raw(Wire wire) {
379 std::size_t valuePosition = position;
380 if (wire == Wire::Varint) return {integer(), valuePosition, valuePosition};
381 std::size_t count;
382 if (wire == Wire::Length) {
383 UInt length = integer();
384 if (length > end - position) byteError("length past end", valuePosition);
385 count = std::size_t(length);
386 } else {
387 count = wire == Wire::I32 ? 4 : 8;
388 if (end - position < count) byteError("truncated " + wireName(wire), valuePosition);
389 }
390 std::size_t start = position;
391 Bytes bytes(data.begin() + position, data.begin() + position + count);
392 position += count;
393 return {std::move(bytes), valuePosition, start};
394 }
395};
396struct Decoder {
397 const Bytes& data;
398 bool explain = false;
399 Lines lines;
400 Value convert(const Type& type, const ReadValue& input, std::size_t depth) {
401 if (type.wire == Wire::Length) {
402 const Bytes& bytes = std::get<Bytes>(input.raw);
403 if (type.message()) return message(type, input.start, input.start + bytes.size(), depth);
404 if (std::holds_alternative<BytesSpec>(type.data)) return bytes;
405 for (std::size_t index = 0; index < bytes.size(); ++index) if (bytes[index] >= 128) byteError("invalid string", input.start + index);
406 return std::string(bytes.begin(), bytes.end());
407 }
408 UInt number = 0;
409 if (const UInt* integer = std::get_if<UInt>(&input.raw)) number = *integer;
410 else {
411 const Bytes& bytes = std::get<Bytes>(input.raw);
412 for (std::size_t index = 0; index < bytes.size(); ++index) number |= UInt(bytes[index]) << (8 * index);
413 }
414 if (number > maximum) byteError("value out of range", input.position);
415 if (std::holds_alternative<BooleanSpec>(type.data)) {
416 if (number > 1) byteError("value out of range", input.position);
417 return number == 1;
418 }
419 if (const EnumSpec* enumeration = std::get_if<EnumSpec>(&type.data)) {
420 if (!enumeration->byNumber.count(Int(number))) byteError("unknown enum number", input.position);
421 return Int(number);
422 }
423 Int value = Int(number);
424 if (type.name == "sint32" || type.name == "sint64") value = number % 2 ? -Int((number + 1) / 2) : Int(number / 2);
425 else if (type.name == "sfixed32" && number >= 2147483648ULL) value -= 4294967296LL;
426 const IntegerSpec& range = std::get<IntegerSpec>(type.data);
427 if (value < range.minimum || value > range.maximum) byteError("value out of range", input.position);
428 return value;
429 }
430 MessagePtr message(const Type& type, std::size_t start, std::size_t end, std::size_t depth = 0) {
431 MessagePtr result = std::make_shared<Message>(type);
432 Reader reader{data, start, end};
433 while (reader.position < end) {
434 std::size_t offset = reader.position;
435 UInt tag = reader.integer(), number = tag / 8, wireNumber = tag % 8;
436 if (number == 0) byteError("invalid field number", offset);
437 if (wireNumber != 0 && wireNumber != 1 && wireNumber != 2 && wireNumber != 5) byteError("invalid wire type", offset);
438 Wire wire = Wire(wireNumber);
439 const Field* field = type.field(number);
440 bool packed = field && field->repeated && field->type->wire != Wire::Length && wire == Wire::Length;
441 if (field && wire != field->type->wire && !packed) byteError("wire type mismatch", offset);
442 ReadValue input = reader.raw(wire);
443 std::string line;
444 if (explain) {
445 line = std::string(depth * 2, ' ') + std::to_string(offset) + " " + std::to_string(number) + " " + (field ? field->name : "?") + " " + wireName(wire);
446 if (wire == Wire::Length) line += " " + std::to_string(std::get<Bytes>(input.raw).size());
447 }
448 if (!field) {
449 if (explain) lines.push_back(line + " " + rawText(input.raw));
450 else result->unknown.push_back({number, wire, std::move(input.raw)});
451 continue;
452 }
453 Values values;
454 if (packed) {
455 Reader elements{data, input.start, reader.position};
456 while (elements.position < elements.end) values.push_back(convert(*field->type, elements.raw(field->type->wire), depth + 1));
457 if (explain) {
458 line += " packed";
459 for (const Value& value : values) line += " " + notation(*field->type, value);
460 lines.push_back(line);
461 }
462 } else {
463 if (explain && field->type->message()) lines.push_back(line);
464 values.push_back(convert(*field->type, input, depth + 1));
465 if (explain && !field->type->message()) lines.push_back(line + " " + notation(*field->type, values.front()));
466 }
467 if (!explain) result->apply(*field, values);
468 }
469 return result;
470 }
471 MessagePtr decode(const Type& type) {
472 MessagePtr result = message(type, 0, data.size());
473 result->checkRequired(); return result;
474 }

477bool lowerHex(char character) { return digit(character) || (character >= 'a' && character <= 'f'); }
478Bytes fromHex(const std::string& text) {
479 Bytes result;
480 for (std::size_t index = 0; index < text.size(); index += 2) {
481 unsigned high = digit(text[index]) ? text[index] - '0' : text[index] - 'a' + 10;
482 unsigned low = digit(text[index + 1]) ? text[index + 1] - '0' : text[index + 1] - 'a' + 10;
483 result.push_back(std::uint8_t(high * 16 + low));
484 }
485 return result;
486}
487Bytes parseHex(const std::string& text) {
488 if (text == "-") return {};
489 if (text.empty() || text.size() % 2 || !std::all_of(text.begin(), text.end(), lowerHex)) fail("invalid hex");
490 return fromHex(text);
491}
492enum class TokenKind { Symbol, Integer, Name, String, Bytes, End };
493enum class Shape { None, Message, List, Integer, Boolean, Name, String, Bytes };
494struct Token {
495 TokenKind kind;
496 std::size_t column;
497 std::variant<std::monostate, char, Int, std::string, Bytes> value;
498 bool symbol(char expected) const { return kind == TokenKind::Symbol && std::get<char>(value) == expected; }
499 Shape shape() const {
500 switch (kind) {
501 case TokenKind::Symbol: return symbol('{') ? Shape::Message : symbol('[') ? Shape::List : Shape::None;
502 case TokenKind::Integer: return Shape::Integer;
503 case TokenKind::Name: {
504 const std::string& name = std::get<std::string>(value);
505 return name == "true" || name == "false" ? Shape::Boolean : Shape::Name;
506 }
507 case TokenKind::String: return Shape::String;
508 case TokenKind::Bytes: return Shape::Bytes;
509 case TokenKind::End: return Shape::None;
510 }
511 throw std::logic_error("Invalid token kind");
512 }
513};
514std::vector<Token> tokenize(const std::string& text) {
515 std::vector<Token> tokens;
516 std::size_t index = 0;
517 while (index < text.size()) {
518 char character = text[index];
519 if (character == ' ') { ++index; continue; }
520 std::size_t start = index++, column = start + 1;
521 if (std::string("{}[]:,").find(character) != std::string::npos) tokens.push_back({TokenKind::Symbol, column, character});
522 else if (character == '"') {
523 std::string value;
524 bool closed = false;
525 while (index < text.size()) {
526 char current = text[index++];
527 if (current == '"') { closed = true; break; }
528 if (current != '\\') { value += current; continue; }
529 if (index == text.size()) columnError("syntax error", column);
530 char escape = text[index++];
531 if (escape == '"' || escape == '\\') value += escape;
532 else if (escape == 'n') value += '\n';
533 else if (escape == 't') value += '\t';
534 else if (escape == 'x' && index + 2 <= text.size() && lowerHex(text[index]) && lowerHex(text[index + 1]) && text[index] <= '7') {
535 value += char(fromHex(text.substr(index, 2)).front()); index += 2;
536 } else columnError("syntax error", column);
537 }
538 if (!closed) columnError("syntax error", column);
539 tokens.push_back({TokenKind::String, column, std::move(value)});
540 } else if (character == '<') {
541 std::size_t end = text.find('>', index);
542 if (end == std::string::npos) columnError("syntax error", column);
543 std::string digits = text.substr(index, end - index);
544 if (digits.size() % 2 || !std::all_of(digits.begin(), digits.end(), lowerHex)) columnError("syntax error", column);
545 tokens.push_back({TokenKind::Bytes, column, fromHex(digits)}); index = end + 1;
546 } else if (character == '-' || digit(character)) {
547 if (character == '-' && (index == text.size() || !digit(text[index]))) columnError("syntax error", column);
548 while (index < text.size() && digit(text[index])) ++index;
549 tokens.push_back({TokenKind::Integer, column, decimal(text.substr(start, index - start))});
550 } else if (letter(character) || character == '_') {
551 while (index < text.size() && (letter(text[index]) || digit(text[index]) || text[index] == '_')) ++index;
552 tokens.push_back({TokenKind::Name, column, text.substr(start, index - start)});
553 } else columnError("syntax error", column);
554 }
555 tokens.push_back({TokenKind::End, text.size() + 1, std::monostate{}}); return tokens;
556}
557struct LiteralParser {
558 std::vector<Token> tokens;
559 std::size_t position = 0;
560 explicit LiteralParser(const std::string& text) : tokens(tokenize(text)) {}
561 const Token& peek() const { return tokens[position]; }
562 const Token& take() { return tokens[position++]; }
563 bool accept(char symbol) { if (!peek().symbol(symbol)) return false; ++position; return true; }
564 const Token& expect(char symbol) {
565 if (!peek().symbol(symbol)) columnError("syntax error", peek().column);
566 return take();
567 }
568 void finish() const { if (peek().kind != TokenKind::End) columnError("syntax error", peek().column); }
569 void requireShape(Shape expected) const {
570 Shape actual = peek().shape();
571 if (actual == Shape::None) columnError("syntax error", peek().column);
572 if (actual != expected) columnError("type mismatch", peek().column);
573 }
574 MessagePtr message(const Type& type) {
575 expect('{');
576 MessagePtr result = std::make_shared<Message>(type);
577 std::set<UInt> seen;
578 std::set<std::string> groups;
579 if (!peek().symbol('}')) while (true) {
580 const Token& token = peek();
581 if (token.kind != TokenKind::Name) columnError("syntax error", token.column);
582 const Field* field = type.field(std::get<std::string>(token.value));
583 if (!field) columnError("unknown field", token.column);
584 if (seen.count(field->number)) columnError("duplicate field", token.column);
585 if (field->group && groups.count(*field->group)) columnError("oneof conflict", token.column);
586 seen.insert(field->number); if (field->group) groups.insert(*field->group);
587 take(); expect(':'); result->replace(*field, fieldValue(*field));
588 if (!accept(',')) break;
589 }
590 const Token& closing = expect('}');
591 for (const Field& field : type.fields()) if (field.required && !seen.count(field.number)) columnError("missing required field " + field.name, closing.column);
592 return result;
593 }
594 Values fieldValue(const Field& field) {
595 if (!field.repeated) return {element(*field.type)};
596 requireShape(Shape::List); expect('[');
597 Values values;
598 if (!peek().symbol(']')) {
599 values.push_back(element(*field.type));
600 while (accept(',')) values.push_back(element(*field.type));
601 }
602 expect(']'); return values;
603 }
604 Value element(const Type& type) {
605 Shape expected = std::holds_alternative<IntegerSpec>(type.data) ? Shape::Integer
606 : std::holds_alternative<BooleanSpec>(type.data) ? Shape::Boolean
607 : std::holds_alternative<EnumSpec>(type.data) ? Shape::Name
608 : std::holds_alternative<StringSpec>(type.data) ? Shape::String
609 : std::holds_alternative<BytesSpec>(type.data) ? Shape::Bytes : Shape::Message;
610 requireShape(expected);
611 if (expected == Shape::Message) return message(type);
612 const Token& token = take();
613 if (expected == Shape::Integer) {
614 Int value = std::get<Int>(token.value);
615 const IntegerSpec& range = std::get<IntegerSpec>(type.data);
616 if (value < range.minimum || value > range.maximum) columnError("out of range", token.column);
617 return value;
618 }
619 if (expected == Shape::Boolean) return std::get<std::string>(token.value) == "true";
620 if (expected == Shape::Name) {
621 const EnumSpec& enumeration = std::get<EnumSpec>(type.data);
622 std::map<std::string, Int>::const_iterator found = enumeration.byName.find(std::get<std::string>(token.value));
623 if (found == enumeration.byName.end()) columnError("unknown enum value", token.column);
624 return found->second;
625 }
626 if (expected == Shape::String) return std::get<std::string>(token.value);
627 return std::get<Bytes>(token.value);
628 }
629};
630
631struct Segment { const Field* field; std::optional<std::size_t> index; };
632using Path = std::vector<Segment>;
633Path parsePath(const Type& type, const std::string& text) {
634 static const std::regex syntax(R"(^([a-z][a-z0-9]*(?:_[a-z0-9]+)*)(?:\[([0-9]{1,6})\])?$)");
635 Lines parts = split(text, '.');
636 const Type* current = &type;
637 Path path;
638 for (std::size_t index = 0; index < parts.size(); ++index) {
639 std::smatch match;
640 if (!std::regex_match(parts[index], match, syntax)) fail("invalid path");
641 const Field* field = current->field(match[1].str());
642 if (!field) fail("invalid path");
643 std::optional<std::size_t> element;
644 if (match[2].matched) element = std::size_t(decimal(match[2].str()));
645 if (element && !field->repeated) fail("invalid path");
646 if (index + 1 < parts.size() && (!field->type->message() || (field->repeated && !element))) fail("invalid path");
647 path.push_back({field, element}); current = field->type;
648 }
649 return path;
650}
651Message* parentAt(Message& message, const Path& path, bool create = false) {
652 Message* current = &message;
653 for (std::size_t index = 0; index + 1 < path.size(); ++index) {
654 const Segment& segment = path[index];
655 std::size_t element = segment.index.value_or(0);
656 if (current->sequence(*segment.field).size() <= element) {
657 if (!create) return nullptr;
658 if (segment.index) fail("index out of range");
659 current->replace(*segment.field, {std::make_shared<Message>(*segment.field->type)});
660 }
661 current = std::get<MessagePtr>(current->sequence(*segment.field)[element]).get();
662 }
663 return current;
664}
665std::string getAt(Message& message, const Path& path) {
666 const Segment& last = path.back();
667 Message* parent = parentAt(message, path);
668 bool whole = last.field->repeated && !last.index;
669 if (!parent) return whole ? "[]" : "(none)";
670 const Values& values = parent->sequence(*last.field);
671 if (whole) return listNotation(*last.field->type, values);
672 std::size_t index = last.index.value_or(0);
673 return index < values.size() ? notation(*last.field->type, values[index]) : "(none)";
674}
675void patchAt(Message& message, const Path& path, Values values) {
676 Message* parent = parentAt(message, path, true);
677 const Segment& last = path.back();
678 if (last.index) {
679 Values previous = parent->sequence(*last.field);
680 if (*last.index >= previous.size()) fail("index out of range");
681 previous[*last.index] = values.front(); parent->replace(*last.field, std::move(previous));
682 } else parent->replace(*last.field, std::move(values));
683}
684void clearAt(Message& message, const Path& path) {
685 Message* parent = parentAt(message, path);
686 if (!parent) return;
687 const Segment& last = path.back();
688 if (!last.index) parent->values.erase(last.field->number);
689 else {
690 Values values = parent->sequence(*last.field);
691 if (*last.index >= values.size()) return;
692 values.erase(values.begin() + *last.index); parent->replace(*last.field, std::move(values));
693 }
694}
695void differences(const Message& first, const Message& second, const std::string& prefix, Lines& lines) {
696 for (const Field& field : first.type->fields()) {
697 const Values& left = first.sequence(field);
698 const Values& right = second.sequence(field);
699 for (std::size_t index = 0; index < std::max(left.size(), right.size()); ++index) {
700 const Value* a = index < left.size() ? &left[index] : nullptr;
701 const Value* b = index < right.size() ? &right[index] : nullptr;
702 std::string path = prefix + field.name + (field.repeated ? "[" + std::to_string(index) + "]" : "");
703 if (a && b && field.type->message()) differences(*std::get<MessagePtr>(*a), *std::get<MessagePtr>(*b), path + ".", lines);
704 else if (!a || !b || *a != *b) lines.push_back(path + ": " + (a ? notation(*field.type, *a) : "(none)") + " -> " + (b ? notation(*field.type, *b) : "(none)"));
705 }
706 }
707}

709Lines execute(const Schema& schema, const std::string& line) {
710 static const std::map<std::string, std::pair<std::size_t, std::size_t>> arity{
711 {"VARINT", {2, 2}}, {"ZIGZAG", {2, 2}}, {"SCHEMA", {2, 2}},
712 {"ENCODE", {3, 0}}, {"SIZE", {3, 0}}, {"DECODE", {3, 3}}, {"LITERAL", {3, 3}},
713 {"JSON", {3, 3}}, {"CANON", {3, 3}}, {"EXPLAIN", {3, 3}},
714 {"GET", {4, 4}}, {"CLEAR", {4, 4}}, {"MERGE", {4, 4}}, {"DIFF", {4, 4}}, {"PATCH", {5, 0}},
715 };
716 Lines words = split(line);
717 const std::string& command = words[0];
718 if (!arity.count(command)) fail("unknown command");
719 const auto& [minimumWords, maximumWords] = arity.at(command);
720 if (words.size() < minimumWords || (maximumWords && words.size() != maximumWords)) fail("wrong number of arguments");
721 if (command == "VARINT" || command == "ZIGZAG") {
722 static const std::regex numberSyntax("-?[0-9]{1,18}");
723 if (!std::regex_match(words[1], numberSyntax)) fail("invalid number");
724 Int value = decimal(words[1]);
725 if (command == "VARINT") {
726 if (value < 0 || value > Int(maximum)) fail("invalid number");
727 return {hex(varint(UInt(value)))};
728 }
729 if (value < signedMinimum || value > signedMaximum) fail("invalid number");
730 UInt encoded = zigzag(value);
731 return {std::to_string(encoded) + " " + hex(varint(encoded))};
732 }
733 std::map<std::string, Type>::const_iterator found = schema.types.find(words[1]);
734 if (found == schema.types.end() || !found->second.message()) fail("unknown message type");
735 const Type& type = found->second;
736 if (command == "SCHEMA") return schema.tree(type);
737 if (command == "ENCODE" || command == "SIZE") {
738 LiteralParser parser(joined(Lines(words.begin() + 2, words.end()), " "));
739 MessagePtr message = parser.message(type); parser.finish();
740 Bytes bytes = encode(*message);
741 if (command == "ENCODE") return {hex(bytes)};
742 std::string result = std::to_string(bytes.size());
743 for (const Field& field : type.fields()) if (!message->sequence(field).empty()) {
744 result += " " + field.name + "=" + std::to_string(fieldBytes(field, message->sequence(field)).size());
745 }
746 return {result};
747 }
748 Bytes first = parseHex(words[2]), second;
749 if (command == "MERGE" || command == "DIFF") second = parseHex(words[3]);
750 Path path;
751 if (command == "GET" || command == "PATCH" || command == "CLEAR") path = parsePath(type, words[3]);
752 if (command == "EXPLAIN") {
753 Decoder decoder{first, true, {}};
754 decoder.message(type, 0, first.size());
755 return decoder.lines.empty() ? Lines{"(empty)"} : decoder.lines;
756 }
757 if (command == "DIFF") {
758 MessagePtr a, b;
759 try { a = Decoder{first, false, {}}.decode(type); }
760 catch (const CommandError& error) { fail(std::string(error.what()) + " in first"); }
761 try { b = Decoder{second, false, {}}.decode(type); }
762 catch (const CommandError& error) { fail(std::string(error.what()) + " in second"); }
763 Lines result;
764 differences(*a, *b, "", result);
765 return result.empty() ? Lines{"SAME"} : result;
766 }
767 if (command == "MERGE") append(first, second);
768 MessagePtr message = Decoder{first, false, {}}.decode(type);
769 if (command == "DECODE" || command == "MERGE") return textFormat(*message);
770 if (command == "LITERAL" || command == "JSON") return {notation(type, message, command == "JSON")};
771 if (command == "GET") return {getAt(*message, path)};
772 if (command == "PATCH") {
773 LiteralParser parser(joined(Lines(words.begin() + 4, words.end()), " "));
774 const Segment& last = path.back();
775 Values values = last.field->repeated && !last.index ? parser.fieldValue(*last.field) : Values{parser.element(*last.field->type)};
776 parser.finish(); patchAt(*message, path, std::move(values)); message->checkRequired();
777 } else if (command == "CLEAR") { clearAt(*message, path); message->checkRequired(); }
778 return {hex(encode(*message))};
779}

784 Schema schema(std::cin);
