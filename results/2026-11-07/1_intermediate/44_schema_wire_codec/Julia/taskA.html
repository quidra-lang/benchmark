4 "uint32" => 0, "uint64" => 0, "sint32" => 0, "sint64" => 0, "bool" => 0,

10 "sint64" => (-4503599627370496, 4503599627370495),

28struct CommandError <: Exception
29 reason::String
30end
31fail(reason) = throw(CommandError(reason))
32byte_error(reason, position) = fail("$reason at byte $position")
33column_error(reason, column) = fail("$reason at column $column")
34is_hex(text) = all(c -> c in HEX_DIGITS, text)
35is_digit(c) = '0' <= c <= '9'
36is_letter(c) = 'a' <= c <= 'z' || 'A' <= c <= 'Z'
37zigzag(n) = n >= 0 ? 2n : -2n - 1
38unzigzag(n) = iseven(n) ? n ÷ 2 : -((n + 1) ÷ 2)
39hex_text(data) = isempty(data) ? "-" : bytes2hex(data)
40bytes_notation(data) = "<" * bytes2hex(data) * ">"

42function varint_bytes(n)
43 result = UInt8[]
44 while n >= 128
45 push!(result, UInt8((n % 128) + 128))
46 n ÷= 128
47 end
48 push!(result, UInt8(n))
49 return result
50end
51function parse_hex(text)
52 text == "-" && return UInt8[]
53 (isempty(text) || isodd(length(text)) || !is_hex(text)) && fail("invalid hex")
54 return hex2bytes(text)
55end
56function quote_bytes(data, prefix)
57 io = IOBuffer()
58 print(io, '"')
59 for b in data
60 if b == 0x22
61 print(io, "\\\"")
62 elseif b == 0x5c
63 print(io, "\\\\")
64 elseif b == 0x0a
65 print(io, "\\n")
66 elseif b == 0x09
67 print(io, "\\t")
68 elseif 0x20 <= b <= 0x7e
69 print(io, Char(b))
70 else
71 print(io, prefix, string(b; base=16, pad=2))
72 end
73 end
74 print(io, '"')
75 return String(take!(io))
76end

107function read_schema(io)
108 schema = Schema()
109 current = ""
110 while true
111 words = String.(split(readline(io), ' '))
112 words == ["END", "SCHEMA"] && break
113 if words[1] == "ENUM"
114 current = words[2]
115 schema.enums[current] = EnumType()
116 elseif words[1] == "MESSAGE"
117 current = words[2]
118 schema.messages[current] = MessageType(current)
119 elseif words[1] == "VALUE"
120 name, number = words[2], parse(Int, words[3])
121 schema.enums[current].numbers[name] = number
122 schema.enums[current].names[number] = name
123 elseif words[1] == "FIELD"
124 options = words[5:end]
125 required = options == ["REQUIRED"]
126 repeated = !isempty(options) && options[1] == "REPEATED"
127 packed = options == ["REPEATED", "PACKED"]
128 group = !isempty(options) && options[1] == "ONEOF" ? options[2] : nothing
129 f = Field(parse(Int, words[2]), words[3], words[4], required, repeated,
130 packed, group, :unresolved, -1)
131 typ = schema.messages[current]
132 push!(typ.fields, f)
133 typ.by_number[f.number] = f
134 typ.by_name[f.name] = f
135 end
136 end
137 for typ in values(schema.messages)
138 sort!(typ.fields; by=f -> f.number)
139 for f in typ.fields
140 if haskey(WIRE_TYPES, f.type_name)
141 f.wire = WIRE_TYPES[f.type_name]
142 f.kind = haskey(INTEGER_RANGES, f.type_name) ? :integer : Symbol(f.type_name)
143 elseif haskey(schema.enums, f.type_name)
144 f.wire, f.kind = 0, :enum
145 else
146 f.wire, f.kind = 2, :message
147 end
148 end
149 end
150 return schema
151end

159mutable struct Message
160 typ::MessageType
161 values::Dict{Int,Any}
162 unknown::Vector{UnknownField}

164Message(typ) = Message(typ, Dict{Int,Any}(), UnknownField[])
165hasfieldvalue(m, f) = haskey(m.values, f.number)
166function field_values(m, f)
167 hasfieldvalue(m, f) || return Any[]
168 return f.repeated ? m.values[f.number] : Any[m.values[f.number]]
169end
170function set_items!(m, f, items)
171 if isempty(items)
172 delete!(m.values, f.number)
173 else
174 m.values[f.number] = items
175 end
176end
177function remove_group!(m, f)
178 isnothing(f.group) && return
179 for other in m.typ.fields
180 if other.number != f.number && other.group == f.group
181 delete!(m.values, other.number)
182 end
183 end
184end
185function apply_values!(m, f, items)
186 remove_group!(m, f)
187 if f.repeated
188 set_items!(m, f, vcat(field_values(m, f), items))
189 elseif f.kind == :message && hasfieldvalue(m, f)
190 merge_message!(m.values[f.number], items[1])
191 else
192 m.values[f.number] = items[1]
193 end
194end
195function merge_message!(target, source)
196 for f in source.typ.fields
197 hasfieldvalue(source, f) && apply_values!(target, f, field_values(source, f))
198 end
199 append!(target.unknown, source.unknown)
200end
201function check_required(m, prefix="")
202 for f in m.typ.fields
203 f.required && !hasfieldvalue(m, f) && fail("missing required field $prefix$(f.name)")
204 if f.kind == :message
205 for (i, value) in enumerate(field_values(m, f))
206 suffix = f.repeated ? "[$(i - 1)]" : ""
207 check_required(value, "$prefix$(f.name)$suffix.")
208 end
209 end
210 end
211end

213function element_bytes(f, value)
214 if f.wire == 2
215 payload = f.kind == :message ? encode_message(value) : value
216 return vcat(varint_bytes(length(payload)), payload)
217 elseif f.wire in (1, 5)
218 n = value < 0 ? value + 4294967296 : value
219 size = f.wire == 1 ? 8 : 4
220 return UInt8[(n >> (8i)) & 0xff for i in 0:size-1]
221 elseif f.kind == :bool
222 return varint_bytes(value ? 1 : 0)
223 else
224 return varint_bytes(f.type_name in ("sint32", "sint64") ? zigzag(value) : value)
225 end
226end
227function field_records(f, items)
228 result = UInt8[]
229 if f.packed
230 payload = UInt8[]
231 for item in items
232 append!(payload, element_bytes(f, item))
233 end
234 append!(result, varint_bytes(8 * f.number + 2))
235 append!(result, varint_bytes(length(payload)))
236 append!(result, payload)
237 else
238 tag = varint_bytes(8 * f.number + f.wire)
239 for item in items
240 append!(result, tag)
241 append!(result, element_bytes(f, item))
242 end
243 end
244 return result
245end
246function encode_message(m)
247 result = UInt8[]
248 for f in m.typ.fields
249 hasfieldvalue(m, f) && append!(result, field_records(f, field_values(m, f)))
250 end
251 for u in m.unknown
252 append!(result, varint_bytes(8 * u.number + u.wire))
253 if u.wire == 0
254 append!(result, varint_bytes(u.raw))
255 else
256 u.wire == 2 && append!(result, varint_bytes(length(u.raw)))
257 append!(result, u.raw)
258 end
259 end
260 return result
261end

263function value_notation(schema, f, value)
264 f.kind == :message && return message_notation(schema, value)
265 f.kind == :enum && return schema.enums[f.type_name].names[value]
266 f.kind == :bool && return value ? "true" : "false"
267 f.kind == :string && return quote_bytes(value, "\\x")
268 f.kind == :bytes && return bytes_notation(value)
269 return string(value)
270end
271list_notation(s, f, items) = "[" * join((value_notation(s, f, v) for v in items), ", ") * "]"
272function message_notation(schema, m)
273 items = String[]
274 for f in m.typ.fields
275 hasfieldvalue(m, f) || continue
276 vs = field_values(m, f)
277 text = f.repeated ? list_notation(schema, f, vs) : value_notation(schema, f, vs[1])
278 push!(items, "$(f.name): $text")
279 end
280 return "{" * join(items, ", ") * "}"
281end
282function json_value(schema, f, value)
283 f.kind == :message && return json_message(schema, value)
284 f.kind == :enum && return "\"" * schema.enums[f.type_name].names[value] * "\""
285 f.kind == :bool && return value ? "true" : "false"
286 f.kind == :string && return quote_bytes(value, "\\u00")
287 f.kind == :bytes && return "\"" * base64encode(value) * "\""
288 f.type_name in ("uint64", "sint64", "fixed64") && return "\"$value\""
289 return string(value)
290end
291function json_message(schema, m)
292 items = String[]
293 for f in m.typ.fields
294 hasfieldvalue(m, f) || continue
295 parts = split(f.name, '_')
296 key = parts[1] * join(uppercasefirst.(parts[2:end]))
297 vs = [json_value(schema, f, v) for v in field_values(m, f)]
298 text = f.repeated ? "[" * join(vs, ',') * "]" : vs[1]
299 push!(items, "\"$key\":$text")
300 end
301 return "{" * join(items, ',') * "}"
302end

304mutable struct Reader
305 schema::Schema
306 data::Vector{UInt8}
307 explaining::Bool
308 lines::Vector{String}

310function read_varint(r, start, finish)
311 value = 0
312 for i in 0:7
313 position = start + i
314 position >= finish && byte_error("truncated varint", start)
315 b = Int(r.data[position + 1])
316 value |= (b & 127) << (7i)
317 if b < 128
318 value >= 9007199254740992 && byte_error("varint too large", start)
319 return value, position + 1
320 end
321 i == 7 && byte_error("varint too long", start)
322 end
323 error("unreachable")
324end
325function read_fixed(r, start, finish, wire)
326 size = wire == 1 ? 8 : 4
327 finish - start < size && byte_error("truncated $(WIRE_NAMES[wire])", start)
328 return r.data[start+1:start+size], start + size
329end
330function little_number(data)
331 value = UInt64(0)
332 for (i, b) in enumerate(data)
333 value |= UInt64(b) << (8(i - 1))
334 end
335 return value
336end
337function scalar_value(schema, f, number, position)
338 if f.kind == :bool
339 number > 1 && byte_error("value out of range", position)
340 return number == 1
341 elseif f.kind == :enum
342 haskey(schema.enums[f.type_name].names, number) || byte_error("unknown enum number", position)
343 return Int(number)
344 end
345 if f.type_name == "fixed64"
346 number > UInt64(9007199254740991) && byte_error("value out of range", position)
347 end
348 value = Int(number)
349 if f.type_name in ("sint32", "sint64")
350 value = unzigzag(value)
351 elseif f.type_name == "sfixed32" && value >= 2147483648
352 value -= 4294967296
353 end
354 low, high = INTEGER_RANGES[f.type_name]
355 low <= value <= high || byte_error("value out of range", position)
356 return value
357end
358function convert_value(r, f, raw, position)
359 if f.kind == :string
360 for (i, b) in enumerate(raw)
361 b >= 0x80 && byte_error("invalid string", position + i - 1)
362 end
363 return raw
364 elseif f.kind == :bytes
365 return raw
366 end
367 number = f.wire == 0 ? raw : little_number(raw)
368 return scalar_value(r.schema, f, number, position)
369end
370function packed_values(r, f, start, finish)
371 values = Any[]
372 position = start
373 while position < finish
374 element_start = position
375 if f.wire == 0
376 number, position = read_varint(r, position, finish)
377 else
378 raw, position = read_fixed(r, position, finish, f.wire)
379 number = little_number(raw)
380 end
381 push!(values, scalar_value(r.schema, f, number, element_start))
382 end
383 return values
384end
385function read_message(r, typ, start, finish, depth)
386 m = Message(typ)
387 position = start
388 while position < finish
389 position = read_record!(r, m, position, finish, depth)
390 end
391 return m
392end
393function read_record!(r, m, start, finish, depth)
394 tag, position = read_varint(r, start, finish)
395 number, wire = tag ÷ 8, tag % 8
396 number == 0 && byte_error("invalid field number", start)
397 haskey(WIRE_NAMES, wire) || byte_error("invalid wire type", start)
398 f = get(m.typ.by_number, number, nothing)
399 packed = !isnothing(f) && f.repeated && f.wire != 2 && wire == 2
400 if !isnothing(f) && wire != f.wire && !packed
401 byte_error("wire type mismatch", start)
402 end
403 value_start = position
404 if wire == 0
405 raw, position = read_varint(r, position, finish)
406 elseif wire == 2
407 q = position
408 payload_length, value_start = read_varint(r, position, finish)
409 value_start + payload_length > finish && byte_error("length past end", q)
410 position = value_start + payload_length
411 raw = r.data[value_start+1:position]
412 else
413 raw, position = read_fixed(r, position, finish, wire)
414 end
415 name = isnothing(f) ? "?" : f.name
416 header = "$(repeat(" ", depth))$start $number $name $(WIRE_NAMES[wire])"
417 wire == 2 && (header *= " $(length(raw))")
418 if isnothing(f)
419 u = UnknownField(number, wire, raw)
420 push!(m.unknown, u)
421 r.explaining && push!(r.lines, "$header $(unknown_notation(u))")
422 return position
423 end
424 if f.kind == :message
425 r.explaining && push!(r.lines, header)
426 values = Any[read_message(r, r.schema.messages[f.type_name], value_start, position, depth + 1)]
427 elseif packed
428 values = packed_values(r, f, value_start, position)
429 if r.explaining
430 suffix = join((" " * value_notation(r.schema, f, v) for v in values))
431 push!(r.lines, "$header packed$suffix")
432 end
433 else
434 values = Any[convert_value(r, f, raw, value_start)]
435 r.explaining && push!(r.lines, "$header $(value_notation(r.schema, f, values[1]))")
436 end
437 !r.explaining && apply_values!(m, f, values)
438 return position
439end
440function decode(schema, typ, data)
441 r = Reader(schema, data, false, String[])
442 m = read_message(r, typ, 0, length(data), 0)
443 check_required(m)
444 return m
445end

447struct Token
448 kind::Symbol
449 column::Int
450 text::String
451 value::Any
452end
453Token(kind, column; text="", value=nothing) = Token(kind, column, text, value)
454function shape(t)
455 if t.kind == :symbol
456 return t.text == "{" ? :message : t.text == "[" ? :list : nothing
457 elseif t.kind == :name
458 return t.text in ("true", "false") ? :boolean : :name
459 end
460 return t.kind == :end ? nothing : t.kind
461end
462function tokenize(text)
463 tokens = Token[]
464 n, i = length(text), 1
465 while i <= n
466 c = text[i]
467 if c == ' '
468 i += 1
469 elseif c in "{}[]:,"
470 push!(tokens, Token(:symbol, i; text=string(c)))
471 i += 1
472 elseif c == '"'
473 start = i
474 i += 1
475 data = UInt8[]
476 while i <= n && text[i] != '"'
477 c = text[i]
478 if c != '\\'
479 push!(data, UInt8(c))
480 i += 1
481 else
482 i + 1 <= n || column_error("syntax error", start)
483 escape = text[i + 1]
484 if escape in ('"', '\\', 'n', 't')
485 b = escape == 'n' ? 0x0a : escape == 't' ? 0x09 : UInt8(escape)
486 push!(data, b)
487 i += 2
488 elseif escape == 'x' && i + 3 <= n && is_hex(text[i+2:i+3])
489 b = parse(Int, text[i+2:i+3]; base=16)
490 b < 128 || column_error("syntax error", start)
491 push!(data, UInt8(b))
492 i += 4
493 else
494 column_error("syntax error", start)
495 end
496 end
497 end
498 i <= n || column_error("syntax error", start)
499 push!(tokens, Token(:string, start; value=data))
500 i += 1
501 elseif c == '<'
502 start = i
503 close = findnext('>', text, i + 1)
504 isnothing(close) && column_error("syntax error", start)
505 digits = text[i+1:close-1]
506 (isodd(length(digits)) || !is_hex(digits)) && column_error("syntax error", start)
507 push!(tokens, Token(:bytes, start; value=hex2bytes(digits)))
508 i = close + 1
509 elseif c == '-' || is_digit(c)
510 start = i
511 c == '-' && (i += 1)
512 digits_start = i
513 while i <= n && is_digit(text[i])
514 i += 1
515 end
516 i == digits_start && column_error("syntax error", start)
517 push!(tokens, Token(:integer, start; value=parse(Int, text[start:i-1])))
518 elseif is_letter(c) || c == '_'
519 start = i
520 i += 1
521 while i <= n && (is_letter(text[i]) || is_digit(text[i]) || text[i] == '_')
522 i += 1
523 end
524 push!(tokens, Token(:name, start; text=String(text[start:i-1])))
525 else
526 column_error("syntax error", i)
527 end
528 end
529 push!(tokens, Token(:end, n + 1))
530 return tokens
531end
532mutable struct LiteralParser
533 schema::Schema
534 tokens::Vector{Token}
535 index::Int
536end
537LiteralParser(schema, text) = LiteralParser(schema, tokenize(text), 1)
538current_token(p) = p.tokens[p.index]
539function take_token!(p)
540 t = current_token(p)
541 p.index += 1
542 return t
543end
544at_symbol(p, sym) = current_token(p).kind == :symbol && current_token(p).text == sym
545parser_error(p, reason) = column_error(reason, current_token(p).column)
546function expect!(p, sym)
547 at_symbol(p, sym) || parser_error(p, "syntax error")
548 return take_token!(p)
549end
550finish_literal(p) = current_token(p).kind == :end || parser_error(p, "syntax error")
551function parse_message!(p, typ)
552 expect!(p, "{")
553 m = Message(typ)
554 seen = Set{Int}()
555 groups = Set{String}()
556 if !at_symbol(p, "}")
557 while true
558 t = current_token(p)
559 t.kind == :name || parser_error(p, "syntax error")
560 f = get(typ.by_name, t.text, nothing)
561 isnothing(f) && parser_error(p, "unknown field")
562 f.number in seen && parser_error(p, "duplicate field")
563 !isnothing(f.group) && f.group in groups && parser_error(p, "oneof conflict")
564 take_token!(p)
565 push!(seen, f.number)
566 !isnothing(f.group) && push!(groups, f.group)
567 expect!(p, ":")
568 if f.repeated
569 set_items!(m, f, parse_list!(p, f))
570 else
571 m.values[f.number] = parse_element!(p, f)
572 end
573 at_symbol(p, ",") || break
574 take_token!(p)
575 end
576 end
577 closing = expect!(p, "}")
578 for f in typ.fields
579 f.required && f.number ∉ seen && column_error("missing required field $(f.name)", closing.column)
580 end
581 return m
582end
583function parse_list!(p, f)
584 s = shape(current_token(p))
585 isnothing(s) && parser_error(p, "syntax error")
586 s == :list || parser_error(p, "type mismatch")
587 take_token!(p)
588 items = Any[]
589 if !at_symbol(p, "]")
590 push!(items, parse_element!(p, f))
591 while at_symbol(p, ",")
592 take_token!(p)
593 push!(items, parse_element!(p, f))
594 end
595 end
596 expect!(p, "]")
597 return items
598end
599function parse_element!(p, f)
600 t = current_token(p)
601 s = shape(t)
602 isnothing(s) && parser_error(p, "syntax error")
603 s == SHAPES[f.kind] || parser_error(p, "type mismatch")
604 s == :message && return parse_message!(p, p.schema.messages[f.type_name])
605 take_token!(p)
606 if s == :integer
607 low, high = INTEGER_RANGES[f.type_name]
608 low <= t.value <= high || column_error("out of range", t.column)
609 return t.value
610 elseif s == :boolean
611 return t.text == "true"
612 elseif s == :name
613 e = p.schema.enums[f.type_name]
614 haskey(e.numbers, t.text) || column_error("unknown enum value", t.column)
615 return e.numbers[t.text]
616 end
617 return t.value
618end

620struct PathSegment
621 field::Field
622 index::Union{Nothing,Int}
623end
624whole(seg) = seg.field.repeated && isnothing(seg.index)
625item_position(seg) = isnothing(seg.index) ? 1 : seg.index + 1
626function parse_path(schema, typ, text)
627 segments = PathSegment[]
628 parts = split(text, '.'; keepempty=true)
629 current = typ
630 for (i, part) in enumerate(parts)
631 m = match(r"^([a-z][a-z0-9_]*)(?:\[([0-9]{1,6})\])?$", part)
632 isnothing(m) && fail("invalid path")
633 name = m.captures[1]
634 index = isnothing(m.captures[2]) ? nothing : parse(Int, m.captures[2])
635 f = get(current.by_name, name, nothing)
636 (isnothing(f) || (!isnothing(index) && !f.repeated)) && fail("invalid path")
637 push!(segments, PathSegment(f, index))
638 if i < length(parts)
639 (f.kind != :message || (f.repeated && isnothing(index))) && fail("invalid path")
640 current = schema.messages[f.type_name]
641 end
642 end
643 return segments
644end
645function follow(m, segments)
646 for seg in segments
647 vs = field_values(m, seg.field)
648 i = item_position(seg)
649 i > length(vs) && return nothing
650 m = vs[i]
651 end
652 return m
653end
654function patch_child!(schema, m, seg)
655 f = seg.field
656 if f.repeated
657 vs = field_values(m, f)
658 i = item_position(seg)
659 i > length(vs) && fail("index out of range")
660 return vs[i]
661 end
662 if !hasfieldvalue(m, f)
663 remove_group!(m, f)
664 m.values[f.number] = Message(schema.messages[f.type_name])
665 end
666 return m.values[f.number]
667end
668
669function text_contents!(lines, schema, m, depth)
670 indent = repeat(" ", depth)
671 for f in m.typ.fields
672 for value in field_values(m, f)
673 if f.kind == :message
674 push!(lines, "$indent$(f.name) {")
675 text_contents!(lines, schema, value, depth + 1)
676 push!(lines, indent * "}")
677 else
678 push!(lines, "$indent$(f.name): $(value_notation(schema, f, value))")
679 end
680 end
681 end
682 for u in m.unknown
683 push!(lines, "$indent$(u.number): $(WIRE_NAMES[u.wire]) $(unknown_notation(u))")
684 end
685end
686function text_message(schema, m)
687 lines = [m.typ.name * " {"]
688 text_contents!(lines, schema, m, 1)
689 push!(lines, "}")
690 return lines
691end
692function schema_fields!(lines, schema, typ, chain, depth)
693 for f in typ.fields
694 kind = f.required ? "required " : f.packed ? "repeated packed " :
695 f.repeated ? "repeated " : isnothing(f.group) ? "" : "oneof $(f.group) "
696 line = "$(repeat(" ", depth))$(f.number) $(f.name): $kind$(f.type_name)"
697 if f.kind == :enum
698 names = schema.enums[f.type_name].names
699 line *= " (" * join(("$(names[n])=$n" for n in sort!(collect(keys(names)))), ", ") * ")"
700 end
701 recursive = f.kind == :message && f.type_name in chain
702 push!(lines, recursive ? line * " (recursive)" : line)
703 if f.kind == :message && !recursive
704 schema_fields!(lines, schema, schema.messages[f.type_name], vcat(chain, [f.type_name]), depth + 1)
705 end
706 end
707end
708function compare_messages!(lines, schema, a, b, prefix="")
709 for f in a.typ.fields
710 left, right = field_values(a, f), field_values(b, f)
711 for i in 1:max(length(left), length(right))
712 path = prefix * f.name * (f.repeated ? "[$(i - 1)]" : "")
713 both = i <= length(left) && i <= length(right)
714 if both && f.kind == :message
715 compare_messages!(lines, schema, left[i], right[i], path * ".")
716 elseif !both || left[i] != right[i]
717 old = i <= length(left) ? value_notation(schema, f, left[i]) : "(none)"
718 new = i <= length(right) ? value_notation(schema, f, right[i]) : "(none)"
719 push!(lines, "$path: $old -> $new")
720 end
721 end
722 end
723end

725function process_command(schema, line)
726 words = String.(split(line, ' '; keepempty=true))
727 cmd = words[1]
728 haskey(WORD_COUNTS, cmd) || fail("unknown command")
729 expected = WORD_COUNTS[cmd]
730 count = length(words)
731 variable = cmd in ("ENCODE", "SIZE", "PATCH")
732 (count < expected || (!variable && count != expected)) && fail("wrong number of arguments")
733 if cmd == "VARINT" || cmd == "ZIGZAG"
734 occursin(r"^-?[0-9]{1,18}$", words[2]) || fail("invalid number")
735 n = parse(Int, words[2])
736 low, high = cmd == "VARINT" ? INTEGER_RANGES["uint64"] : INTEGER_RANGES["sint64"]
737 low <= n <= high || fail("invalid number")
738 cmd == "VARINT" && return [hex_text(varint_bytes(n))]
739 z = zigzag(n)
740 return ["$z $(hex_text(varint_bytes(z)))"]
741 end
742 haskey(schema.messages, words[2]) || fail("unknown message type")
743 typ = schema.messages[words[2]]
744 data = [parse_hex(words[2 + i]) for i in 1:get(HEX_COUNTS, cmd, 0)]
745 segments = cmd in ("GET", "PATCH", "CLEAR") ? parse_path(schema, typ, words[4]) : PathSegment[]
746 if cmd == "SCHEMA"
747 lines = [typ.name]
748 schema_fields!(lines, schema, typ, [typ.name], 1)
749 return lines
750 elseif cmd == "ENCODE" || cmd == "SIZE"
751 text = split(line, ' '; limit=3, keepempty=true)[3]
752 p = LiteralParser(schema, text)
753 m = parse_message!(p, typ)
754 finish_literal(p)
755 bytes = encode_message(m)
756 cmd == "ENCODE" && return [hex_text(bytes)]
757 parts = [string(length(bytes))]
758 for f in typ.fields
759 hasfieldvalue(m, f) && push!(parts, "$(f.name)=$(length(field_records(f, field_values(m, f))))")
760 end
761 return [join(parts, ' ')]
762 elseif cmd == "EXPLAIN"
763 isempty(data[1]) && return ["(empty)"]
764 r = Reader(schema, data[1], true, String[])
765 read_message(r, typ, 0, length(data[1]), 0)
766 return r.lines
767 elseif cmd == "MERGE"
768 return text_message(schema, decode(schema, typ, vcat(data[1], data[2])))
769 elseif cmd == "DIFF"
770 messages = Message[]
771 for (bytes, label) in zip(data, ("first", "second"))
772 try
773 push!(messages, decode(schema, typ, bytes))
774 catch err
775 err isa CommandError || rethrow()
776 fail("$(err.reason) in $label")
777 end
778 end
779 lines = String[]
780 compare_messages!(lines, schema, messages[1], messages[2])
781 return isempty(lines) ? ["SAME"] : lines
782 end
783 m = decode(schema, typ, data[1])
784 cmd == "DECODE" && return text_message(schema, m)
785 cmd == "LITERAL" && return [message_notation(schema, m)]
786 cmd == "JSON" && return [json_message(schema, m)]
787 cmd == "CANON" && return [hex_text(encode_message(m))]
788 lastseg = last(segments)
789 f = lastseg.field
790 if cmd == "GET"
791 target = follow(m, segments[1:end-1])
792 vs = isnothing(target) ? Any[] : field_values(target, f)
793 whole(lastseg) && return [list_notation(schema, f, vs)]
794 i = item_position(lastseg)
795 return [i <= length(vs) ? value_notation(schema, f, vs[i]) : "(none)"]
796 elseif cmd == "PATCH"
797 text = split(line, ' '; limit=5, keepempty=true)[5]
798 p = LiteralParser(schema, text)
799 value = whole(lastseg) ? parse_list!(p, f) : parse_element!(p, f)
800 finish_literal(p)
801 target = m
802 for seg in segments[1:end-1]
803 target = patch_child!(schema, target, seg)
804 end
805 if whole(lastseg)
806 set_items!(target, f, value)
807 elseif f.repeated
808 vs = field_values(target, f)
809 i = item_position(lastseg)
810 i > length(vs) && fail("index out of range")
811 vs[i] = value
812 else
813 remove_group!(target, f)
814 target.values[f.number] = value
815 end
816 elseif cmd == "CLEAR"
817 target = follow(m, segments[1:end-1])
818 if !isnothing(target)
819 if isnothing(lastseg.index)
820 delete!(target.values, f.number)
821 else
822 vs = field_values(target, f)
823 i = item_position(lastseg)
824 if i <= length(vs)
825 deleteat!(vs, i)
826 set_items!(target, f, vs)
827 end
828 end
829 end
830 end
831 check_required(m)
832 return [hex_text(encode_message(m))]
833end
834function execute(schema, line)
835 try
836 return process_command(schema, line)
837 catch err
838 err isa CommandError || rethrow()
839 return ["ERROR $(err.reason)"]
840 end
841end

845 for _ in 1:n
846 for line in execute(schema, readline())
847 println(line)
848 end
849 end
