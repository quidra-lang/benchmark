5private const val VARINT_LIMIT = 9_007_199_254_740_992L
6private const val HEX_DIGITS = "0123456789abcdef"
7private val INTEGER_RANGES = mapOf(
8 "uint32" to (0L..4_294_967_295L),
9 "uint64" to (0L..9_007_199_254_740_991L),
10 "sint32" to (-2_147_483_648L..2_147_483_647L),
11 "sint64" to (-4_503_599_627_370_496L..4_503_599_627_370_495L),

30private fun zigzag(number: Long) = if (number >= 0) 2 * number else -2 * number - 1
31private fun unzigzag(number: Long) = if (number % 2 == 0L) number / 2 else -(number + 1) / 2
32private fun isHex(text: String) = text.all { it in HEX_DIGITS }
33private fun isLetter(char: Char) = char in 'a'..'z' || char in 'A'..'Z'

35private fun hexDigits(data: ByteArray) = buildString {
36 for (byte in data) {
37 val value = byte.toInt() and 255
38 append(HEX_DIGITS[value / 16])
39 append(HEX_DIGITS[value % 16])
40 }
41}
42
43private fun hexText(data: ByteArray) = if (data.isEmpty()) "-" else hexDigits(data)
44private fun bytesNotation(data: ByteArray) = "<${hexDigits(data)}>"
45
46private fun hexBytes(text: String) = ByteArray(text.length / 2) { index ->
47 (HEX_DIGITS.indexOf(text[2 * index]) * 16 + HEX_DIGITS.indexOf(text[2 * index + 1])).toByte()
48}
49
50private fun parseHex(text: String): ByteArray {
51 if (text == "-") return byteArrayOf()
52 if (text.isEmpty() || text.length % 2 != 0 || !isHex(text)) throw CommandError("invalid hex")
53 return hexBytes(text)
54}

56private fun writeVarint(output: ByteArrayOutputStream, value: Long) {
57 var remaining = value
58 while (remaining >= 128) {
59 output.write((remaining % 128 + 128).toInt())
60 remaining /= 128
61 }
62 output.write(remaining.toInt())
63}

65private fun varintBytes(value: Long): ByteArray {
66 val output = ByteArrayOutputStream()
67 writeVarint(output, value)
68 return output.toByteArray()
69}

71private fun quote(data: ByteArray, json: Boolean = false) = buildString {
72 append('"')
73 for (byte in data) {
74 val value = byte.toInt() and 255
75 when (value) {
76 0x22 -> append("\\\"")
77 0x5c -> append("\\\\")
78 0x0a -> append("\\n")
79 0x09 -> append("\\t")
80 in 0x20..0x7e -> append(value.toChar())
81 else -> {
82 append(if (json) "\\u00" else "\\x")
83 append(HEX_DIGITS[value / 16])
84 append(HEX_DIGITS[value % 16])
85 }
86 }
87 }
88 append('"')
89}

131private class Schema {
132 val enums = mutableMapOf<String, EnumType>()
133 val messages = mutableMapOf<String, MessageType>()

136private fun readSchema(input: BufferedReader): Schema {
137 val schema = Schema()
138 var currentEnum: EnumType? = null
139 var currentMessage: MessageType? = null
140 while (true) {
141 val line = input.readLine()
142 if (line == "END SCHEMA") break
143 val words = line.split(' ')
144 when (words[0]) {
145 "ENUM" -> {
146 currentEnum = EnumType(words[1])
147 schema.enums[words[1]] = currentEnum
148 currentMessage = null
149 }
150 "MESSAGE" -> {
151 currentMessage = MessageType(words[1])
152 schema.messages[words[1]] = currentMessage
153 currentEnum = null
154 }
155 "VALUE" -> {
156 val number = words[2].toLong()
157 val enumType = currentEnum!!
158 enumType.byName[words[1]] = number
159 enumType.byNumber[number] = words[1]
160 }
161 "FIELD" -> {
162 val options = words.drop(4)
163 currentMessage!!.add(Field(
164 words[1].toLong(), words[2], words[3],
165 options.firstOrNull() == "REQUIRED", options.firstOrNull() == "REPEATED",
166 options == listOf("REPEATED", "PACKED"),
167 if (options.firstOrNull() == "ONEOF") options[1] else null
168 ))
169 }
170 "END" -> {
171 currentEnum = null
172 currentMessage = null
173 }
174 }
175 }
176 for (type in schema.messages.values) {
177 type.fields.sortBy { it.number }
178 for (field in type.fields) {
179 field.enumType = schema.enums[field.typeName]
180 field.messageType = schema.messages[field.typeName]
181 field.wire = SCALAR_WIRES[field.typeName] ?: if (field.enumType != null) 0 else 2
182 }
183 }
184 return schema
185}

203private class Message(val type: MessageType) {
204 val values = mutableMapOf<Long, MutableList<Value>>()
205 val unknown = mutableListOf<UnknownField>()
206
207 fun elements(field: Field): List<Value> = values[field.number] ?: emptyList()
208 fun has(field: Field) = !values[field.number].isNullOrEmpty()
209
210 fun setElements(field: Field, elements: List<Value>) {
211 if (elements.isEmpty()) values.remove(field.number)
212 else values[field.number] = elements.toMutableList()
213 }
214
215 fun clearGroup(field: Field) {
216 if (field.group == null) return
217 for (other in type.fields) {
218 if (other.number != field.number && other.group == field.group) values.remove(other.number)
219 }
220 }
221
222 fun apply(field: Field, incoming: List<Value>) {
223 clearGroup(field)
224 when {
225 field.repeated -> {
226 if (incoming.isNotEmpty()) values.getOrPut(field.number) { mutableListOf() }.addAll(incoming)
227 }
228 field.messageType != null && has(field) -> {
229 val target = (elements(field).single() as NestedValue).message
230 merge(target, (incoming.single() as NestedValue).message)
231 }
232 else -> setElements(field, incoming)
233 }
234 }
235}
236
237private fun merge(target: Message, source: Message) {
238 for (field in source.type.fields) {
239 if (source.has(field)) target.apply(field, source.elements(field))
240 }
241 target.unknown.addAll(source.unknown)
242}

244private fun checkRequired(message: Message, prefix: String = "") {
245 for (field in message.type.fields) {
246 val path = prefix + field.name
247 if (field.required && !message.has(field)) throw CommandError("missing required field $path")
248 if (field.messageType != null) {
249 for ((index, value) in message.elements(field).withIndex()) {
250 val segment = path + if (field.repeated) "[$index]" else ""
251 checkRequired((value as NestedValue).message, "$segment.")
252 }
253 }
254 }
255}

257private fun valueBytes(field: Field, value: Value): ByteArray {
258 return when (value) {
259 is DataValue -> value.data
260 is NestedValue -> encode(value.message)
261 is BooleanValue -> varintBytes(if (value.boolean) 1 else 0)
262 is EnumValue -> varintBytes(value.number)
263 is IntegerValue -> {
264 if (field.wire == 0) {
265 varintBytes(if (field.typeName in setOf("sint32", "sint64")) zigzag(value.number) else value.number)
266 } else {
267 val number = if (field.typeName == "sfixed32" && value.number < 0)
268 value.number + 4_294_967_296L else value.number
269 ByteArray(if (field.wire == 5) 4 else 8) { index ->
270 (number ushr (8 * index)).toByte()
271 }
272 }
273 }
274 }
275}

277private fun encodeField(field: Field, elements: List<Value>): ByteArray {
278 val output = ByteArrayOutputStream()
279 if (elements.isEmpty()) return output.toByteArray()
280 if (field.packed) {
281 val payload = ByteArrayOutputStream()
282 for (value in elements) payload.write(valueBytes(field, value))
283 writeVarint(output, field.number * 8 + 2)
284 writeVarint(output, payload.size().toLong())
285 output.write(payload.toByteArray())
286 } else {
287 for (value in elements) {
288 writeVarint(output, field.number * 8 + field.wire)
289 val raw = valueBytes(field, value)
290 if (field.wire == 2) writeVarint(output, raw.size.toLong())
291 output.write(raw)
292 }
293 }
294 return output.toByteArray()
295}
296
297private fun encode(message: Message): ByteArray {
298 val output = ByteArrayOutputStream()
299 for (field in message.type.fields) output.write(encodeField(field, message.elements(field)))
300 for (unknown in message.unknown) {
301 writeVarint(output, unknown.number * 8 + unknown.wire)
302 when (unknown.wire) {
303 0 -> writeVarint(output, unknown.rawNumber)
304 2 -> {
305 writeVarint(output, unknown.rawBytes.size.toLong())
306 output.write(unknown.rawBytes)
307 }
308 else -> output.write(unknown.rawBytes)
309 }
310 }
311 return output.toByteArray()
312}

314private fun valueNotation(field: Field, value: Value): String = when (value) {
315 is IntegerValue -> value.number.toString()
316 is BooleanValue -> value.boolean.toString()
317 is EnumValue -> value.name
318 is DataValue -> if (field.typeName == "string") quote(value.data) else bytesNotation(value.data)
319 is NestedValue -> messageNotation(value.message)

322private fun sequenceNotation(field: Field, elements: List<Value>) =
323 elements.joinToString(", ", "[", "]") { valueNotation(field, it) }

325private fun messageNotation(message: Message) = message.type.fields.filter { message.has(it) }
326 .joinToString(", ", "{", "}") { field ->
327 val value = if (field.repeated) sequenceNotation(field, message.elements(field))
328 else valueNotation(field, message.elements(field).single())
329 "${field.name}: $value"
330 }

334 return buildString {
335 for (char in name) {
336 if (char == '_') uppercase = true else {

344private fun jsonValue(field: Field, value: Value): String = when (value) {
345 is IntegerValue -> if (field.typeName in setOf("uint64", "sint64", "fixed64"))
346 "\"${value.number}\"" else value.number.toString()
347 is BooleanValue -> value.boolean.toString()
348 is EnumValue -> "\"${value.name}\""
349 is DataValue -> if (field.typeName == "string") quote(value.data, true)
350 else "\"${Base64.getEncoder().encodeToString(value.data)}\""
351 is NestedValue -> jsonMessage(value.message)
352}

354private fun jsonMessage(message: Message) = message.type.fields.filter { message.has(it) }
355 .joinToString(",", "{", "}") { field ->
356 val value = if (field.repeated) message.elements(field).joinToString(",", "[", "]") {
357 jsonValue(field, it)
358 } else jsonValue(field, message.elements(field).single())
359 "\"${jsonKey(field.name)}\":$value"

362private fun textMessage(message: Message): List<String> {
363 val lines = mutableListOf("${message.type.name} {")
364 fun contents(current: Message, depth: Int) {
365 val indent = " ".repeat(depth)
366 for (field in current.type.fields) {
367 for (value in current.elements(field)) {
368 if (value is NestedValue) {
369 lines.add("$indent${field.name} {")
370 contents(value.message, depth + 1)
371 lines.add("$indent}")
372 } else lines.add("$indent${field.name}: ${valueNotation(field, value)}")
373 }
374 }
375 for (unknown in current.unknown) {
376 lines.add("$indent${unknown.number}: ${WIRE_NAMES.getValue(unknown.wire)} ${unknown.notation()}")
377 }
378 }
379 contents(message, 1)
380 lines.add("}")
381 return lines
382}

384private enum class TokenKind { SYMBOL, INTEGER, NAME, STRING, BYTES, END }
385
386private data class Token(
387 val kind: TokenKind,
388 val text: String,
389 val column: Int,
390 val number: Long = 0,
391 val data: ByteArray = byteArrayOf()
392) {
393 fun shape(): String? = when (kind) {
394 TokenKind.INTEGER -> "integer"
395 TokenKind.NAME -> if (text == "true" || text == "false") "boolean" else "name"
396 TokenKind.STRING -> "string"
397 TokenKind.BYTES -> "bytes"
398 TokenKind.SYMBOL -> when (text) { "{" -> "message"; "[" -> "list"; else -> null }
399 TokenKind.END -> null
400 }
401}
402
403private fun tokenize(text: String): List<Token> {
404 val tokens = mutableListOf<Token>()
405 var position = 0
406 while (position < text.length) {
407 val start = position
408 val char = text[position]
409 when {
410 char == ' ' -> position++
411 char in "{}[]:," -> {
412 tokens.add(Token(TokenKind.SYMBOL, char.toString(), start + 1))
413 position++
414 }
415 char == '"' -> {
416 position++
417 val bytes = ByteArrayOutputStream()
418 var closed = false
419 while (position < text.length) {
420 val next = text[position++]
421 if (next == '"') {
422 closed = true
423 break
424 }
425 if (next != '\\') bytes.write(next.code) else {
426 if (position == text.length) columnError("syntax error", start + 1)
427 when (val escape = text[position++]) {
428 '"', '\\' -> bytes.write(escape.code)
429 'n' -> bytes.write(10)
430 't' -> bytes.write(9)
431 'x' -> {
432 if (position + 2 > text.length) columnError("syntax error", start + 1)
433 val digits = text.substring(position, position + 2)
434 if (!isHex(digits) || digits.toInt(16) > 127) columnError("syntax error", start + 1)
435 bytes.write(digits.toInt(16))
436 position += 2
437 }
438 else -> columnError("syntax error", start + 1)
439 }
440 }
441 }
442 if (!closed) columnError("syntax error", start + 1)
443 tokens.add(Token(TokenKind.STRING, "", start + 1, data = bytes.toByteArray()))
444 }
445 char == '<' -> {
446 val end = text.indexOf('>', position + 1)
447 if (end < 0) columnError("syntax error", start + 1)
448 val digits = text.substring(position + 1, end)
449 if (digits.length % 2 != 0 || !isHex(digits)) columnError("syntax error", start + 1)
450 tokens.add(Token(TokenKind.BYTES, "", start + 1, data = hexBytes(digits)))
451 position = end + 1
452 }
453 char == '-' || char in '0'..'9' -> {
454 if (char == '-') position++
455 val digitStart = position
456 while (position < text.length && text[position] in '0'..'9') position++
457 if (position == digitStart) columnError("syntax error", start + 1)
458 val digits = text.substring(start, position)
459 tokens.add(Token(TokenKind.INTEGER, digits, start + 1, digits.toLong()))
460 }
461 isLetter(char) || char == '_' -> {
462 position++
463 while (position < text.length && (isLetter(text[position]) ||
464 text[position] in '0'..'9' || text[position] == '_')) position++
465 tokens.add(Token(TokenKind.NAME, text.substring(start, position), start + 1))
466 }
467 else -> columnError("syntax error", start + 1)
468 }
469 }
470 tokens.add(Token(TokenKind.END, "", text.length + 1))
471 return tokens
472}

474private class LiteralParser(text: String) {
475 private val tokens = tokenize(text)
476 private var position = 0
477 private val next get() = tokens[position]
478 private fun at(symbol: String) = next.kind == TokenKind.SYMBOL && next.text == symbol
479
480 private fun expect(symbol: String) {
481 if (!at(symbol)) columnError("syntax error", next.column)
482 position++
483 }
484
485 fun finish() {
486 if (next.kind != TokenKind.END) columnError("syntax error", next.column)
487 }
488
489 fun readMessage(type: MessageType): Message {
490 expect("{")
491 val message = Message(type)
492 val seen = mutableSetOf<Long>()
493 val groups = mutableSetOf<String>()
494 if (!at("}")) {
495 while (true) {
496 val token = next
497 if (token.kind != TokenKind.NAME) columnError("syntax error", token.column)
498 val field = type.byName[token.text] ?: columnError("unknown field", token.column)
499 if (!seen.add(field.number)) columnError("duplicate field", token.column)
500 if (field.group != null && !groups.add(field.group)) columnError("oneof conflict", token.column)
501 position++
502 expect(":")
503 message.setElements(field, readField(field))
504 if (at(",")) {
505 position++
506 continue
507 }
508 if (!at("}")) columnError("syntax error", next.column)
509 break
510 }
511 }
512 val closing = next
513 expect("}")
514 for (field in type.fields) {
515 if (field.required && field.number !in seen) {
516 columnError("missing required field ${field.name}", closing.column)
517 }
518 }
519 return message
520 }
521
522 fun readField(field: Field): List<Value> {
523 if (!field.repeated) return listOf(readElement(field))
524 val shape = next.shape() ?: columnError("syntax error", next.column)
525 if (shape != "list") columnError("type mismatch", next.column)
526 expect("[")
527 val values = mutableListOf<Value>()
528 if (!at("]")) {
529 while (true) {
530 values.add(readElement(field))
531 if (!at(",")) break
532 position++
533 }
534 }
535 expect("]")
536 return values
537 }
538
539 fun readElement(field: Field): Value {
540 val token = next
541 val shape = token.shape() ?: columnError("syntax error", token.column)
542 val expected = when {
543 field.typeName in INTEGER_RANGES -> "integer"
544 field.typeName == "bool" -> "boolean"
545 field.enumType != null -> "name"
546 field.messageType != null -> "message"
547 else -> field.typeName
548 }
549 if (shape != expected) columnError("type mismatch", token.column)
550 if (field.messageType != null) return NestedValue(readMessage(field.messageType!!))
551 position++
552 return when (expected) {
553 "integer" -> {
554 if (token.number !in INTEGER_RANGES.getValue(field.typeName)) columnError("out of range", token.column)
555 IntegerValue(token.number)
556 }
557 "boolean" -> BooleanValue(token.text == "true")
558 "name" -> {
559 val number = field.enumType!!.byName[token.text] ?: columnError("unknown enum value", token.column)
560 EnumValue(token.text, number)
561 }
562 else -> DataValue(token.data)
563 }
564 }
565}

573private class WireReader(private val data: ByteArray, private val explanation: MutableList<String>? = null) {
574 private var position = 0
575
576 private fun readVarint(end: Int): Long {
577 val start = position
578 var value = 0L
579 for (index in 0 until 8) {
580 if (position >= end) byteError("truncated varint", start)
581 val byte = data[position++].toInt() and 255
582 value = value or ((byte and 127).toLong() shl (7 * index))
583 if (byte < 128) {
584 if (value >= VARINT_LIMIT) byteError("varint too large", start)
585 return value
586 }
587 if (index == 7) byteError("varint too long", start)
588 }
589 error("unreachable varint state")
590 }
591
592 private fun readRaw(wire: Int, end: Int): RawValue {
593 val start = position
594 if (wire == 0) return RawValue(number = readVarint(end))
595 if (wire == 2) {
596 val length = readVarint(end)
597 val payloadStart = position
598 if (length > (end - position).toLong()) byteError("length past end", start)
599 position += length.toInt()
600 return RawValue(bytes = data.copyOfRange(payloadStart, position), payloadStart = payloadStart)
601 }
602 val size = if (wire == 5) 4 else 8
603 if (end - position < size) byteError(if (wire == 5) "truncated i32" else "truncated i64", start)
604 position += size
605 return RawValue(bytes = data.copyOfRange(start, position))
606 }
607
608 private fun convert(field: Field, raw: RawValue, valuePosition: Int): Value {
609 if (field.typeName == "string") {
610 for ((index, byte) in raw.bytes.withIndex()) {
611 if ((byte.toInt() and 255) >= 128) byteError("invalid string", raw.payloadStart + index)
612 }
613 return DataValue(raw.bytes)
614 }
615 if (field.typeName == "bytes") return DataValue(raw.bytes)
616 if (field.enumType != null) {
617 val name = field.enumType!!.byNumber[raw.number] ?: byteError("unknown enum number", valuePosition)
618 return EnumValue(name, raw.number)
619 }
620 if (field.typeName == "bool") {
621 if (raw.number !in 0L..1L) byteError("value out of range", valuePosition)
622 return BooleanValue(raw.number == 1L)
623 }
624 var number = raw.number
625 if (field.wire != 0) {
626 if (field.typeName == "fixed64" &&
627 (raw.bytes[7].toInt() != 0 || (raw.bytes[6].toInt() and 255) >= 32)) {
628 byteError("value out of range", valuePosition)
629 }
630 number = 0
631 for ((index, byte) in raw.bytes.withIndex()) {
632 number = number or ((byte.toInt() and 255).toLong() shl (8 * index))
633 }
634 }
635 when (field.typeName) {
636 "sint32", "sint64" -> number = unzigzag(number)
637 "sfixed32" -> if (number >= 2_147_483_648L) number -= 4_294_967_296L
638 }
639 if (number !in INTEGER_RANGES.getValue(field.typeName)) byteError("value out of range", valuePosition)
640 return IntegerValue(number)
641 }

643 fun readMessage(type: MessageType, end: Int = data.size, depth: Int = 0): Message {
644 val message = Message(type)
645 while (position < end) {
646 val recordStart = position
647 val tag = readVarint(end)
648 val number = tag / 8
649 val wire = (tag % 8).toInt()
650 if (number == 0L) byteError("invalid field number", recordStart)
651 if (wire !in WIRE_NAMES) byteError("invalid wire type", recordStart)
652 val field = type.byNumber[number]
653 val packed = field != null && field.repeated && field.packable && wire == 2
654 if (field != null && wire != field.wire && !packed) byteError("wire type mismatch", recordStart)
655 val valuePosition = position
656 val raw = readRaw(wire, end)
657 var heading = "${" ".repeat(depth)}$recordStart $number ${field?.name ?: "?"} ${WIRE_NAMES.getValue(wire)}"
658 if (wire == 2) heading += " ${raw.bytes.size}"
659 if (field == null) {
660 val unknown = UnknownField(number, wire, raw.number, raw.bytes)
661 if (explanation == null) message.unknown.add(unknown)
662 else explanation.add("$heading ${unknown.notation()}")
663 continue
664 }
665 val values: List<Value>
666 if (field.messageType != null) {
667 explanation?.add(heading)
668 position = raw.payloadStart
669 values = listOf(NestedValue(readMessage(field.messageType!!, raw.payloadStart + raw.bytes.size, depth + 1)))
670 } else if (packed) {
671 position = raw.payloadStart
672 val payloadEnd = raw.payloadStart + raw.bytes.size
673 val elements = mutableListOf<Value>()
674 while (position < payloadEnd) {
675 val elementPosition = position
676 val element = readRaw(field.wire, payloadEnd)
677 elements.add(convert(field, element, elementPosition))
678 }
679 values = elements
680 explanation?.add(heading + " packed" + elements.joinToString("") { " ${valueNotation(field, it)}" })
681 } else {
682 val value = convert(field, raw, valuePosition)
683 values = listOf(value)
684 explanation?.add("$heading ${valueNotation(field, value)}")
685 }
686 if (explanation == null) message.apply(field, values)
687 }
688 return message
689 }
690}

692private fun decode(type: MessageType, data: ByteArray): Message {
693 val message = WireReader(data).readMessage(type)
694 checkRequired(message)
695 return message

698private fun explain(type: MessageType, data: ByteArray): List<String> {
699 if (data.isEmpty()) return listOf("(empty)")
700 val lines = mutableListOf<String>()
701 WireReader(data, lines).readMessage(type)

705private data class PathSegment(val field: Field, val index: Int?)
706
707private fun parsePath(type: MessageType, text: String): List<PathSegment> {
708 val pattern = Regex("([a-z][a-z0-9_]*)(?:\\[([0-9]{1,6})\\])?")
709 val parts = text.split('.')
710 val result = mutableListOf<PathSegment>()
711 var currentType = type
712 for ((position, part) in parts.withIndex()) {
713 val match = pattern.matchEntire(part) ?: throw CommandError("invalid path")
714 val field = currentType.byName[match.groupValues[1]] ?: throw CommandError("invalid path")
715 val index = match.groupValues[2].takeIf { it.isNotEmpty() }?.toInt()
716 if (index != null && !field.repeated) throw CommandError("invalid path")
717 result.add(PathSegment(field, index))
718 if (position != parts.lastIndex) {
719 currentType = field.messageType ?: throw CommandError("invalid path")
720 if (field.repeated && index == null) throw CommandError("invalid path")
721 }
722 }
723 return result
724}
725
726private fun getPath(message: Message, path: List<PathSegment>): String {
727 var current: Message? = message
728 for ((position, segment) in path.withIndex()) {
729 val elements = current?.elements(segment.field) ?: emptyList()
730 if (position == path.lastIndex) {
731 if (segment.field.repeated && segment.index == null) return sequenceNotation(segment.field, elements)
732 val value = elements.getOrNull(segment.index ?: 0) ?: return "(none)"
733 return valueNotation(segment.field, value)
734 }
735 current = (elements.getOrNull(segment.index ?: 0) as? NestedValue)?.message
736 }
737 error("empty validated path")
738}
739
740private fun patchPath(message: Message, path: List<PathSegment>, replacement: List<Value>) {
741 var current = message
742 for (segment in path.dropLast(1)) {
743 val field = segment.field
744 if (segment.index != null) {
745 val value = current.elements(field).getOrNull(segment.index) ?: throw CommandError("index out of range")
746 current = (value as NestedValue).message
747 } else {
748 if (!current.has(field)) {
749 current.clearGroup(field)
750 current.setElements(field, listOf(NestedValue(Message(field.messageType!!))))
751 }
752 current = (current.elements(field).single() as NestedValue).message
753 }
754 }
755 val last = path.last()
756 if (last.index != null) {
757 val elements = current.values[last.field.number] ?: throw CommandError("index out of range")
758 if (last.index >= elements.size) throw CommandError("index out of range")
759 elements[last.index] = replacement.single()
760 } else {
761 current.clearGroup(last.field)
762 current.setElements(last.field, replacement)
763 }
764}
765
766private fun clearPath(message: Message, path: List<PathSegment>) {
767 var current = message
768 for (segment in path.dropLast(1)) {
769 val value = current.elements(segment.field).getOrNull(segment.index ?: 0) ?: return
770 current = (value as NestedValue).message
771 }
772 val last = path.last()
773 if (last.index == null) current.values.remove(last.field.number) else {
774 val elements = current.values[last.field.number] ?: return
775 if (last.index < elements.size) elements.removeAt(last.index)
776 if (elements.isEmpty()) current.values.remove(last.field.number)
777 }
778}

780private fun sameValue(first: Value, second: Value) =
781 if (first is DataValue && second is DataValue) first.data.contentEquals(second.data) else first == second
782
783private fun differences(first: Message, second: Message): List<String> {
784 val lines = mutableListOf<String>()
785 fun compare(a: Message, b: Message, prefix: String) {
786 for (field in a.type.fields) {
787 val x = a.elements(field)
788 val y = b.elements(field)
789 for (index in 0 until maxOf(x.size, y.size)) {
790 val path = prefix + field.name + if (field.repeated) "[$index]" else ""
791 val left = x.getOrNull(index)
792 val right = y.getOrNull(index)
793 if (left is NestedValue && right is NestedValue) compare(left.message, right.message, "$path.")
794 else if (left == null || right == null || !sameValue(left, right)) {
795 val before = left?.let { valueNotation(field, it) } ?: "(none)"
796 val after = right?.let { valueNotation(field, it) } ?: "(none)"
797 lines.add("$path: $before -> $after")
798 }
799 }
800 }
801 }
802 compare(first, second, "")
803 return lines.ifEmpty { listOf("SAME") }
804}
805
806private fun schemaTree(type: MessageType): List<String> {
807 val lines = mutableListOf(type.name)
808 fun fields(current: MessageType, chain: Set<MessageType>, depth: Int) {
809 for (field in current.fields) {
810 var line = "${" ".repeat(depth)}${field.number} ${field.name}: ${field.kindWord()}${field.typeName}"
811 field.enumType?.let { enumType ->
812 line += enumType.byNumber.toSortedMap().entries.joinToString(", ", " (", ")") {
813 "${it.value}=${it.key}"
814 }
815 }
816 val nested = field.messageType
817 if (nested != null && nested in chain) line += " (recursive)"
818 lines.add(line)
819 if (nested != null && nested !in chain) fields(nested, chain + nested, depth + 1)
820 }
821 }
822 fields(type, setOf(type), 1)
823 return lines
824}

826private fun parseNumber(text: String, range: LongRange): Long {
827 if (!Regex("-?[0-9]{1,18}").matches(text)) throw CommandError("invalid number")
828 val number = text.toLong()
829 if (number !in range) throw CommandError("invalid number")
830 return number
831}

833private class Commands(private val schema: Schema) {
834 private val counts = mapOf(
835 "VARINT" to 2, "ZIGZAG" to 2, "SCHEMA" to 2, "ENCODE" to 3, "SIZE" to 3,
836 "DECODE" to 3, "LITERAL" to 3, "JSON" to 3, "CANON" to 3, "EXPLAIN" to 3,
837 "GET" to 4, "PATCH" to 5, "CLEAR" to 4, "MERGE" to 4, "DIFF" to 4
838 )
839
840 fun execute(line: String): List<String> {
841 val words = line.split(' ')
842 val command = words[0]
843 val count = counts[command] ?: throw CommandError("unknown command")
844 val hasText = command in setOf("ENCODE", "SIZE", "PATCH")
845 if ((hasText && words.size < count) || (!hasText && words.size != count)) {
846 throw CommandError("wrong number of arguments")
847 }
848 if (command == "VARINT") {
849 val number = parseNumber(words[1], 0L until VARINT_LIMIT)
850 return listOf(hexText(varintBytes(number)))
851 }
852 if (command == "ZIGZAG") {
853 val number = parseNumber(words[1], INTEGER_RANGES.getValue("sint64"))
854 val value = zigzag(number)
855 return listOf("$value ${hexText(varintBytes(value))}")
856 }
857 val type = schema.messages[words[1]] ?: throw CommandError("unknown message type")
858 val hasHex = command !in setOf("SCHEMA", "ENCODE", "SIZE")
859 val first = if (hasHex) parseHex(words[2]) else byteArrayOf()
860 val second = if (command == "MERGE" || command == "DIFF") parseHex(words[3]) else byteArrayOf()
861 val path = if (command in setOf("GET", "PATCH", "CLEAR")) parsePath(type, words[3]) else emptyList()
862 return when (command) {
863 "SCHEMA" -> schemaTree(type)
864 "ENCODE", "SIZE" -> {
865 val parser = LiteralParser(line.split(' ', limit = 3)[2])
866 val message = parser.readMessage(type)
867 parser.finish()
868 if (command == "ENCODE") listOf(hexText(encode(message))) else {
869 val details = type.fields.filter { message.has(it) }.joinToString("") {
870 " ${it.name}=${encodeField(it, message.elements(it)).size}"
871 }
872 listOf("${encode(message).size}$details")
873 }
874 }
875 "EXPLAIN" -> explain(type, first)
876 "MERGE" -> textMessage(decode(type, first + second))
877 "DIFF" -> {
878 val a = try { decode(type, first) } catch (error: CommandError) {
879 throw CommandError("${error.message} in first")
880 }
881 val b = try { decode(type, second) } catch (error: CommandError) {
882 throw CommandError("${error.message} in second")
883 }
884 differences(a, b)
885 }
886 else -> {
887 val message = decode(type, first)
888 when (command) {
889 "DECODE" -> textMessage(message)
890 "LITERAL" -> listOf(messageNotation(message))
891 "JSON" -> listOf(jsonMessage(message))
892 "CANON" -> listOf(hexText(encode(message)))
893 "GET" -> listOf(getPath(message, path))
894 "PATCH" -> {
895 val parser = LiteralParser(line.split(' ', limit = 5)[4])
896 val last = path.last()
897 val replacement = if (last.field.repeated && last.index == null) parser.readField(last.field)
898 else listOf(parser.readElement(last.field))
899 parser.finish()
900 patchPath(message, path, replacement)
901 checkRequired(message)
902 listOf(hexText(encode(message)))
903 }
904 "CLEAR" -> {
905 clearPath(message, path)
906 checkRequired(message)
907 listOf(hexText(encode(message)))
908 }
909 else -> error("unreachable command")
910 }
911 }
912 }
913 }
914}

916fun main() {
917 val input = System.`in`.bufferedReader()
918 val output = System.out.bufferedWriter()
919 val commands = Commands(readSchema(input))
920 repeat(input.readLine().toInt()) {
921 val result = try {
922 commands.execute(input.readLine())
923 } catch (error: CommandError) {
924 listOf("ERROR ${error.message}")
925 }
926 for (line in result) {
927 output.write(line)
928 output.newLine()
929 }
930 }
931 output.flush()
932}
