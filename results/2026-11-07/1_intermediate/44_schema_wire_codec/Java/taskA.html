87 void choose(Field field) {

118 static long zigzag(long number) { return number >= 0 ? number * 2 : -number * 2 - 1; }
119 static long unzigzag(long number) { return number % 2 == 0 ? number / 2 : -1 - number / 2; }
120 static void putVarint(ByteArrayOutputStream bytes, long number) {
121 while (number >= 128) { bytes.write((int)(number % 128 + 128)); number /= 128; }
122 bytes.write((int)number);
123 }
124 static void encodeElement(ByteArrayOutputStream bytes, Type type, Value value) {
125 if (value instanceof Message || value instanceof BinaryValue) {
126 byte[] payload = value instanceof Message message ? encode(message) : ((BinaryValue)value).bytes;
127 putVarint(bytes, payload.length); bytes.writeBytes(payload); return;
128 }
129 long number = value instanceof BooleanValue b ? (b.state ? 1 : 0) : ((NumberValue)value).number;
130 if (type.kind == Kind.SINT32 || type.kind == Kind.SINT64) number = zigzag(number);
131 if (type.kind == Kind.SFIXED32 && number < 0) number += 4_294_967_296L;
132 if (type.wire() == 0) putVarint(bytes, number);
133 else for (int i = 0; i < (type.wire() == 1 ? 8 : 4); i++) { bytes.write((int)(number % 256)); number /= 256; }
134 }
135 static byte[] encodeField(Field field, List<Value> values) {
136 var bytes = new ByteArrayOutputStream();
137 if (!values.isEmpty()) {
138 if (field.packed) {
139 var payload = new ByteArrayOutputStream();
140 for (var value : values) encodeElement(payload, field.type, value);
141 putVarint(bytes, field.number * 8 + 2); putVarint(bytes, payload.size()); bytes.writeBytes(payload.toByteArray());
142 } else for (var value : values) {
143 putVarint(bytes, field.number * 8 + field.type.wire()); encodeElement(bytes, field.type, value);
144 }
145 }
146 return bytes.toByteArray();
147 }
148 static byte[] encode(Message message) {
149 var bytes = new ByteArrayOutputStream();
150 for (var field : message.type.fields) bytes.writeBytes(encodeField(field, message.values(field)));
151 for (var unknown : message.unknown) {
152 putVarint(bytes, unknown.number * 8 + unknown.wire);
153 if (unknown.wire == 0) putVarint(bytes, unknown.integer);
154 else { if (unknown.wire == 2) putVarint(bytes, unknown.bytes.length); bytes.writeBytes(unknown.bytes); }
155 }
156 return bytes.toByteArray();
157 }

160 static String quote(byte[] bytes, boolean json) {
161 var text = new StringBuilder("\"");
162 for (byte raw : bytes) {
163 int b = raw & 255;
164 switch (b) {
165 case 34 -> text.append("\\\""); case 92 -> text.append("\\\\");
166 case 10 -> text.append("\\n"); case 9 -> text.append("\\t");
167 default -> {
168 if (b >= 32 && b <= 126) text.append((char)b);
169 else text.append(json ? "\\u00" : "\\x").append(String.format(Locale.ROOT, "%02x", b));
170 }
171 }
172 }
173 return text.append('"').toString();
174 }
175 static String notation(Type type, Value value, boolean json) {
176 if (value == null) return "(none)";
177 return switch (value) {
178 case NumberValue n -> {
179 String text = type.kind == Kind.ENUM ? type.members.get(n.number) : Long.toString(n.number);
180 boolean quoted = json && (type.kind == Kind.ENUM || type.kind == Kind.UINT64 || type.kind == Kind.SINT64 || type.kind == Kind.FIXED64);
181 yield quoted ? "\"" + text + "\"" : text;
182 }
183 case BooleanValue b -> Boolean.toString(b.state);
184 case BinaryValue b -> type.kind == Kind.STRING ? quote(b.bytes, json) : json ? "\"" + Base64.getEncoder().encodeToString(b.bytes) + "\"" : byteNotation(b.bytes);

193 static String messageNotation(Message message, boolean json) {
194 var items = new ArrayList<String>();

196 var values = message.values(field);

206 }

227 static final class ByteReader {
228 final byte[] data;
229 final int end;
230 int position;
231 ByteReader(byte[] data, int position, int end) { this.data = data; this.position = position; this.end = end; }
232 long varint() {
233 int start = position; long number = 0;
234 for (int i = 0; i < 8; i++) {
235 if (position >= end) throw byteError("truncated varint", start);
236 int b = data[position++] & 255; number += (long)(b & 127) << (7 * i);
237 if (b < 128) { if (number > MAX_UNSIGNED) throw byteError("varint too large", start); return number; }
238 }
239 throw byteError("varint too long", start);
240 }
241 Raw read(int wire) {
242 int start = position;
243 if (wire == 0) return new Raw(varint(), new byte[0], start, start);
244 long length = wire == 2 ? varint() : wire == 1 ? 8 : 4;
245 if (length > end - position) throw byteError(wire == 2 ? "length past end" : wire == 1 ? "truncated i64" : "truncated i32", start);
246 int payload = position; position += (int)length;
247 return new Raw(0, Arrays.copyOfRange(data, payload, position), start, payload);
248 }
249 }
250 static Value convert(Type type, Raw raw) {
251 if (type.kind == Kind.STRING || type.kind == Kind.BYTES) {
252 if (type.kind == Kind.STRING) for (int i = 0; i < raw.bytes.length; i++)
253 if ((raw.bytes[i] & 255) >= 128) throw byteError("invalid string", raw.payload + i);
254 return new BinaryValue(raw.bytes);
255 }
256 long number = raw.number;
257 if (type.wire() == 1 || type.wire() == 5) {
258 if (type.kind == Kind.FIXED64 && (raw.bytes[7] != 0 || (raw.bytes[6] & 255) >= 32)) throw byteError("value out of range", raw.position);
259 number = 0;
260 for (int i = raw.bytes.length - 1; i >= 0; i--) number = number * 256 + (raw.bytes[i] & 255);
261 }
262 if (type.kind == Kind.SINT32 || type.kind == Kind.SINT64) number = unzigzag(number);
263 if (type.kind == Kind.SFIXED32 && number >= 2_147_483_648L) number -= 4_294_967_296L;
264 if (!type.contains(number) || type.kind == Kind.BOOL && number > 1) throw byteError("value out of range", raw.position);
265 if (type.kind == Kind.ENUM && !type.members.containsKey(number)) throw byteError("unknown enum number", raw.position);
266 return type.kind == Kind.BOOL ? new BooleanValue(number != 0) : new NumberValue(number);
267 }
268 static final class Decoder {
269 final byte[] data;
270 final boolean explain;
271 final List<String> lines = new ArrayList<>();
272 Decoder(byte[] data, boolean explain) { this.data = data; this.explain = explain; }
273 Message message(Type type, int start, int end, int depth) {
274 var message = new Message(type); var reader = new ByteReader(data, start, end);
275 while (reader.position < end) {
276 int offset = reader.position; long tag = reader.varint(), number = tag / 8; int wire = (int)(tag % 8);
277 if (number == 0) throw byteError("invalid field number", offset);
278 if (wire != 0 && wire != 1 && wire != 2 && wire != 5) throw byteError("invalid wire type", offset);
279 var field = type.byNumber(number);
280 boolean packed = field != null && field.repeated && field.type.wire() != 2 && wire == 2;
281 if (field != null && wire != field.type.wire() && !packed) throw byteError("wire type mismatch", offset);
282 var raw = reader.read(wire);
283 String header = " ".repeat(depth) + offset + " " + number + " " + (field == null ? "?" : field.name) + " " + wireName(wire);
284 if (wire == 2) header += " " + raw.bytes.length;
285 if (field == null) {
286 var unknown = new Unknown(number, wire, raw.number, raw.bytes);
287 if (explain) lines.add(header + " " + rawNotation(unknown)); else message.unknown.add(unknown);
288 continue;
289 }
290 var values = new ArrayList<Value>();
291 if (field.type.kind == Kind.MESSAGE) {
292 if (explain) lines.add(header);
293 values.add(message(field.type, raw.payload, raw.payload + raw.bytes.length, depth + 1));
294 } else {
295 if (packed) {
296 var elements = new ByteReader(data, raw.payload, raw.payload + raw.bytes.length);
297 while (elements.position < elements.end) values.add(convert(field.type, elements.read(field.type.wire())));
298 } else values.add(convert(field.type, raw));
299 if (explain) {
300 if (packed) header += " packed";
301 for (var value : values) header += " " + notation(field.type, value, false);
302 lines.add(header);
303 }
304 }
305 if (!explain) message.apply(field, values);
306 }
307 return message;
308 }

315 enum TokenKind { SYMBOL, INTEGER, NAME, STRING, BYTES, END }
316 enum Shape { NONE, MESSAGE, LIST, INTEGER, BOOLEAN, NAME, STRING, BYTES }
317 record Token(TokenKind kind, int column, String text, long number, byte[] bytes) {
318 Shape shape() {
319 return switch (kind) {
320 case INTEGER -> Shape.INTEGER; case STRING -> Shape.STRING; case BYTES -> Shape.BYTES;
321 case NAME -> text.equals("true") || text.equals("false") ? Shape.BOOLEAN : Shape.NAME;
322 case SYMBOL -> text.equals("{") ? Shape.MESSAGE : text.equals("[") ? Shape.LIST : Shape.NONE;
323 default -> Shape.NONE;
324 };
325 }
326 }
327 static boolean digit(char c) { return c >= '0' && c <= '9'; }
328 static boolean letter(char c) { return c >= 'a' && c <= 'z' || c >= 'A' && c <= 'Z'; }
329 static int hexDigit(char c) { return digit(c) ? c - '0' : c >= 'a' && c <= 'f' ? c - 'a' + 10 : -1; }
330 static List<Token> tokenize(String source) {
331 var tokens = new ArrayList<Token>();
332 for (int at = 0; at < source.length();) {
333 char c = source.charAt(at);
334 if (c == ' ') { at++; continue; }
335 int start = at++, column = start + 1;
336 Token token;
337 if ("{}[]:,".indexOf(c) >= 0) token = new Token(TokenKind.SYMBOL, column, Character.toString(c), 0, null);
338 else if (c == '"') {
339 var bytes = new ByteArrayOutputStream();
340 while (at < source.length() && source.charAt(at) != '"') {
341 int b = source.charAt(at++);
342 if (b == '\\') {
343 if (at == source.length()) throw columnError("syntax error", column);
344 char escaped = source.charAt(at++);
345 switch (escaped) {
346 case '"', '\\' -> b = escaped; case 'n' -> b = '\n'; case 't' -> b = '\t';
347 case 'x' -> {
348 if (at + 1 >= source.length()) throw columnError("syntax error", column);
349 int high = hexDigit(source.charAt(at)), low = hexDigit(source.charAt(at + 1));
350 if (high < 0 || high > 7 || low < 0) throw columnError("syntax error", column);
351 b = high * 16 + low; at += 2;
352 }
353 default -> throw columnError("syntax error", column);
354 }
355 }
356 bytes.write(b);
357 }
358 if (at == source.length()) throw columnError("syntax error", column);
359 at++; token = new Token(TokenKind.STRING, column, null, 0, bytes.toByteArray());
360 } else if (c == '<') {
361 int end = source.indexOf('>', at);
362 if (end < 0 || (end - at) % 2 != 0) throw columnError("syntax error", column);
363 var bytes = new ByteArrayOutputStream();
364 while (at < end) {
365 int high = hexDigit(source.charAt(at)), low = hexDigit(source.charAt(at + 1));
366 if (high < 0 || low < 0) throw columnError("syntax error", column);
367 bytes.write(high * 16 + low); at += 2;
368 }
369 at++; token = new Token(TokenKind.BYTES, column, null, 0, bytes.toByteArray());
370 } else if (c == '-' || digit(c)) {
371 if (c == '-' && (at == source.length() || !digit(source.charAt(at)))) throw columnError("syntax error", column);
372 while (at < source.length() && digit(source.charAt(at))) at++;
373 token = new Token(TokenKind.INTEGER, column, null, Long.parseLong(source.substring(start, at)), null);
374 } else if (letter(c) || c == '_') {
375 while (at < source.length() && (letter(source.charAt(at)) || digit(source.charAt(at)) || source.charAt(at) == '_')) at++;
376 token = new Token(TokenKind.NAME, column, source.substring(start, at), 0, null);
377 } else throw columnError("syntax error", column);
378 tokens.add(token);
379 }
380 tokens.add(new Token(TokenKind.END, source.length() + 1, null, 0, null)); return tokens;
381 }
382 static final class Literal {
383 final List<Token> tokens;
384 int position;
385 Literal(String source) { tokens = tokenize(source); }
386 Token next() { return tokens.get(position); }
387 boolean symbol(String text) { return next().kind == TokenKind.SYMBOL && next().text.equals(text); }
388 Token expect(String text) {
389 if (!symbol(text)) throw columnError("syntax error", next().column);
390 return tokens.get(position++);
391 }
392 void end() { if (next().kind != TokenKind.END) throw columnError("syntax error", next().column); }
393 void require(Shape shape) {
394 if (next().shape() == Shape.NONE) throw columnError("syntax error", next().column);
395 if (next().shape() != shape) throw columnError("type mismatch", next().column);
396 }
397 Value element(Type type) {
398 Shape shape = switch (type.kind) {
399 case BOOL -> Shape.BOOLEAN; case STRING -> Shape.STRING; case BYTES -> Shape.BYTES;
400 case ENUM -> Shape.NAME; case MESSAGE -> Shape.MESSAGE; default -> Shape.INTEGER;
401 };
402 require(shape);
403 if (shape == Shape.MESSAGE) return message(type);
404 Token token = tokens.get(position++);
405 return switch (shape) {
406 case INTEGER -> {
407 if (!type.contains(token.number)) throw columnError("out of range", token.column);
408 yield new NumberValue(token.number);
409 }
410 case BOOLEAN -> new BooleanValue(token.text.equals("true"));
411 case NAME -> {
412 Long number = null;
413 for (var entry : type.members.entrySet()) if (entry.getValue().equals(token.text)) { number = entry.getKey(); break; }
414 if (number == null) throw columnError("unknown enum value", token.column);
415 yield new NumberValue(number);
416 }
417 case STRING, BYTES -> new BinaryValue(token.bytes);
418 default -> throw new AssertionError();
419 };
420 }
421 List<Value> field(Field field) {
422 if (!field.repeated) return List.of(element(field.type));
423 require(Shape.LIST); expect("[");
424 var values = new ArrayList<Value>();
425 if (!symbol("]")) {
426 values.add(element(field.type));
427 while (symbol(",")) { position++; values.add(element(field.type)); }
428 }
429 expect("]"); return values;
430 }
431 Message message(Type type) {
432 expect("{");
433 var message = new Message(type); var seen = new HashSet<Field>(); var groups = new HashSet<String>();
434 if (!symbol("}")) {
435 while (true) {
436 Token token = next();
437 if (token.kind != TokenKind.NAME) throw columnError("syntax error", token.column);
438 var field = type.byName(token.text);
439 if (field == null) throw columnError("unknown field", token.column);
440 if (!seen.add(field)) throw columnError("duplicate field", token.column);
441 if (field.group != null && !groups.add(field.group)) throw columnError("oneof conflict", token.column);
442 position++; expect(":"); message.fields.put(field, field(field));
443 if (!symbol(",")) break;
444 position++;
445 }
446 }
447 Token closing = expect("}");
448 for (var field : type.fields) if (field.required && !seen.contains(field))
449 throw columnError("missing required field " + field.name, closing.column);
450 return message;
451 }

457 static List<Segment> path(Type type, String source) {
458 String[] parts = source.split("\\.", -1); var path = new ArrayList<Segment>();
459 for (int i = 0; i < parts.length; i++) {
460 String part = parts[i], name = part; Integer index = null;
461 int open = part.indexOf('[');
462 if (open >= 0) {
463 if (!part.endsWith("]")) throw new CodecException("invalid path");
464 String digits = part.substring(open + 1, part.length() - 1);
465 if (digits.isEmpty() || digits.length() > 6 || !digits.chars().allMatch(c -> c >= '0' && c <= '9')) throw new CodecException("invalid path");
466 index = Integer.parseInt(digits); name = part.substring(0, open);
467 }
468 var field = type.byName(name);
469 if (field == null || index != null && !field.repeated) throw new CodecException("invalid path");
470 if (i + 1 < parts.length && (field.type.kind != Kind.MESSAGE || field.repeated && index == null)) throw new CodecException("invalid path");
471 path.add(new Segment(field, index)); type = field.type;
472 }
473 return path;

475 static String get(Message message, List<Segment> path) {
476 for (int i = 0; i < path.size(); i++) {
477 var segment = path.get(i); List<Value> values = message == null ? List.of() : message.values(segment.field);
478 int index = segment.index == null ? 0 : segment.index;
479 if (i + 1 == path.size() && segment.wholeList()) return sequenceNotation(segment.field.type, values, false);
480 Value value = index < values.size() ? values.get(index) : null;
481 if (i + 1 == path.size()) return notation(segment.field.type, value, false);
482 message = (Message)value;
483 }
484 throw new AssertionError();

486 static void edit(Message message, List<Segment> path, List<Value> replacement) {
487 for (int i = 0; i < path.size(); i++) {
488 var segment = path.get(i); var field = segment.field; var values = message.values(field);
489 int index = segment.index == null ? 0 : segment.index;

495 } else if (replacement == null) message.fields.remove(field);

497 return;
498 }
499 if (index >= values.size()) {
500 if (replacement == null) return;
501 if (segment.index != null) throw new CodecException("index out of range");
502 message.choose(field); values = new ArrayList<>(List.of(new Message(field.type))); message.fields.put(field, values);
503 }
504 message = (Message)values.get(index);
505 }
506 }
507 static boolean equal(Value a, Value b) {
508 if (a == null || b == null) return a == b;
509 return switch (a) {
510 case NumberValue n -> n.number == ((NumberValue)b).number;
511 case BooleanValue v -> v.state == ((BooleanValue)b).state;
512 case BinaryValue v -> Arrays.equals(v.bytes, ((BinaryValue)b).bytes);
513 case Message _ -> false;
514 };
515 }
516 static void differences(Message a, Message b, String prefix, List<String> lines) {
517 for (var field : a.type.fields) {
518 var x = a.values(field); var y = b.values(field);
519 for (int i = 0; i < Math.max(x.size(), y.size()); i++) {
520 String path = prefix + field.name + (field.repeated ? "[" + i + "]" : "");
521 Value left = i < x.size() ? x.get(i) : null, right = i < y.size() ? y.get(i) : null;
522 if (left instanceof Message l && right instanceof Message r) differences(l, r, path + ".", lines);
523 else if (!equal(left, right)) lines.add(path + ": " + notation(field.type, left, false) + " -> " + notation(field.type, right, false));
524 }
525 }
526 }
527 static void schemaFields(Type type, List<Type> chain, List<String> lines) {
528 chain.add(type);
529 for (var field : type.fields) {
530 String mode = field.required ? "required " : field.packed ? "repeated packed " : field.repeated ? "repeated " : field.group != null ? "oneof " + field.group + " " : "";
531 String line = " ".repeat(chain.size()) + field.number + " " + field.name + ": " + mode + field.type.name;
532 if (field.type.kind == Kind.ENUM) {
533 var members = new ArrayList<String>(); field.type.members.forEach((number, name) -> members.add(name + "=" + number));
534 line += " (" + String.join(", ", members) + ")";
535 }
536 boolean recursive = field.type.kind == Kind.MESSAGE && chain.contains(field.type);
537 lines.add(line + (recursive ? " (recursive)" : ""));
538 if (field.type.kind == Kind.MESSAGE && !recursive) schemaFields(field.type, chain, lines);
539 }
540 chain.removeLast();
541 }
542 static byte[] parseHex(String source) {
543 if (source.equals("-")) return new byte[0];
544 if (!source.matches("(?:[0-9a-f]{2})+")) throw new CodecException("invalid hex");
545 return HexFormat.of().parseHex(source);
546 }
547 static String rest(String source, int count) { return source.split(" ", count + 1)[count]; }
548 static List<String> execute(Schema schema, String source) {
549 String[] words = source.split(" ", -1); String command = words[0];
550 int count = switch (command) {
551 case "VARINT", "ZIGZAG", "SCHEMA" -> 2;
552 case "ENCODE", "SIZE", "DECODE", "LITERAL", "JSON", "CANON", "EXPLAIN" -> 3;
553 case "GET", "CLEAR", "MERGE", "DIFF" -> 4;
554 case "PATCH" -> 5;
555 default -> throw new CodecException("unknown command");
556 };
557 boolean flexible = Set.of("ENCODE", "SIZE", "PATCH").contains(command);
558 if (words.length < count || !flexible && words.length != count) throw new CodecException("wrong number of arguments");
559 if (command.equals("VARINT") || command.equals("ZIGZAG")) {
560 if (!words[1].matches("-?[0-9]{1,18}")) throw new CodecException("invalid number");
561 long number = Long.parseLong(words[1]);
562 if (command.equals("VARINT") ? number < 0 || number > MAX_UNSIGNED : !schema.types.get("sint64").contains(number)) throw new CodecException("invalid number");
563 long value = command.equals("ZIGZAG") ? zigzag(number) : number;
564 var bytes = new ByteArrayOutputStream(); putVarint(bytes, value);
565 return List.of((command.equals("ZIGZAG") ? value + " " : "") + hex(bytes.toByteArray()));
566 }
567 var type = schema.types.get(words[1]);
568 if (type == null || type.kind != Kind.MESSAGE) throw new CodecException("unknown message type");
569 if (command.equals("SCHEMA")) {
570 var lines = new ArrayList<String>(); lines.add(type.name); schemaFields(type, new ArrayList<>(), lines); return lines;
571 }
572 if (command.equals("ENCODE") || command.equals("SIZE")) {
573 var literal = new Literal(rest(source, 2)); var message = literal.message(type); literal.end();
574 byte[] bytes = encode(message);
575 if (command.equals("ENCODE")) return List.of(hex(bytes));
576 var line = new StringBuilder(Integer.toString(bytes.length));
577 for (var field : type.fields) if (!message.values(field).isEmpty()) line.append(' ').append(field.name).append('=').append(encodeField(field, message.values(field)).length);
578 return List.of(line.toString());
579 }
580 byte[] first = parseHex(words[2]);
581 byte[] second = command.equals("MERGE") || command.equals("DIFF") ? parseHex(words[3]) : new byte[0];
582 var path = Set.of("GET", "PATCH", "CLEAR").contains(command) ? path(type, words[3]) : List.<Segment>of();
583 if (command.equals("EXPLAIN")) {
584 var decoder = new Decoder(first, true); decoder.message(type, 0, first.length, 0);
585 return first.length == 0 ? List.of("(empty)") : decoder.lines;
586 }
587 if (command.equals("DIFF")) {
588 var a = decode(type, first, " in first"); var b = decode(type, second, " in second");
589 var lines = new ArrayList<String>(); differences(a, b, "", lines); return lines.isEmpty() ? List.of("SAME") : lines;
590 }
591 if (command.equals("MERGE")) {
592 byte[] joined = Arrays.copyOf(first, first.length + second.length); System.arraycopy(second, 0, joined, first.length, second.length); first = joined;
593 }
594 var message = decode(type, first, "");
595 return switch (command) {
596 case "DECODE", "MERGE" -> textMessage(message);
597 case "LITERAL", "JSON" -> List.of(messageNotation(message, command.equals("JSON")));
598 case "GET" -> List.of(get(message, path));
599 default -> {
600 if (command.equals("PATCH")) {
601 var literal = new Literal(rest(source, 4)); var last = path.getLast();
602 var replacement = last.wholeList() ? literal.field(last.field) : List.of(literal.element(last.field.type));
603 literal.end(); edit(message, path, replacement); message.required("");
604 } else if (command.equals("CLEAR")) { edit(message, path, null); message.required(""); }
605 yield List.of(hex(encode(message)));
606 }
607 };
608 }
609 public static void main(String[] args) throws IOException {
610 var input = new BufferedReader(new InputStreamReader(System.in, StandardCharsets.US_ASCII));
611 var schema = new Schema(input); int count = Integer.parseInt(input.readLine());
612 var output = new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out, StandardCharsets.US_ASCII)));
613 for (int i = 0; i < count; i++) {
614 try { for (String line : execute(schema, input.readLine())) output.print(line + "\n"); }
615 catch (CodecException error) { output.print("ERROR " + error.getMessage() + "\n"); }
616 }
617 output.flush();
618 }
