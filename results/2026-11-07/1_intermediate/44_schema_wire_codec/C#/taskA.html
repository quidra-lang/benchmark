10sealed class SchemaType(string name, Kind kind)
11{
12 public string Name { get; } = name;
13 public Kind Kind { get; } = kind;
14 public List<Field> Fields { get; } = [];
15 public SortedDictionary<long, string> Numbers { get; } = [];
16 public Dictionary<string, long> Names { get; } = new(StringComparer.Ordinal);
17 public int Wire => Kind switch {
18 Kind.Fixed64 => 1, Kind.String or Kind.Bytes or Kind.Message => 2,
19 Kind.Fixed32 or Kind.SFixed32 => 5, _ => 0
20 };
21 public bool Packable => Wire != 2;
22 public Field? Find(string name) => Fields.Find(field => field.Name == name);
23 public Field? Find(long number) => Fields.Find(field => field.Number == number);
24}

38abstract record Value;
39sealed record NumberValue(long Number) : Value;
40sealed record BooleanValue(bool State) : Value;
41sealed record BinaryValue(byte[] Bytes) : Value;
42sealed record Unknown(long Number, int Wire, long Integer, byte[] Bytes);
43sealed record Message(SchemaType Type) : Value
44{
45 public Dictionary<Field, List<Value>> Fields { get; } = [];
46 public List<Unknown> Unknown { get; } = [];
47 public List<Value> Values(Field field) => Fields.GetValueOrDefault(field) ?? [];
48 public void Choose(Field field)
49 {
50 if (field.Group is null) return;
51 foreach (Field other in Type.Fields)
52 if (other != field && other.Group == field.Group) Fields.Remove(other);
53 }
54 public void Apply(Field field, List<Value> values)
55 {
56 if (values.Count == 0) return;
57 Choose(field);
58 if (field.Repeated) {
59 if (!Fields.TryGetValue(field, out List<Value>? old)) Fields[field] = old = [];
60 old.AddRange(values);
61 } else if (field.Type.Kind == Kind.Message && Fields.TryGetValue(field, out List<Value>? old)) {
62 ((Message)old[0]).Merge((Message)values[0]);
63 } else Fields[field] = values;
64 }
65 public void Merge(Message source)
66 {
67 foreach (Field field in Type.Fields) Apply(field, source.Values(field));
68 Unknown.AddRange(source.Unknown);
69 }
70}

74static class Format
75{
76 public static string Hex(byte[] bytes) => bytes.Length == 0 ? "-" : Convert.ToHexString(bytes).ToLowerInvariant();
77 public static string Bytes(byte[] bytes) => "<" + Convert.ToHexString(bytes).ToLowerInvariant() + ">";
78 public static string Wire(int wire) => wire switch { 0 => "varint", 1 => "i64", 2 => "len", _ => "i32" };
79 public static string Raw(Unknown value) => value.Wire == 0 ? value.Integer.ToString() : Bytes(value.Bytes);
80 public static string Quote(byte[] bytes, bool json)
81 {
82 StringBuilder result = new("\"");
83 foreach (byte value in bytes)
84 result.Append(value switch {
85 34 => "\\\"", 92 => "\\\\", 10 => "\\n", 9 => "\\t",
86 >= 32 and <= 126 => ((char)value).ToString(),
87 _ => (json ? "\\u00" : "\\x") + value.ToString("x2")
88 });
89 return result.Append('"').ToString();
90 }
91 public static string Value(SchemaType type, Value value, bool json = false) => value switch {
92 NumberValue number when type.Kind == Kind.Enum =>
93 json ? "\"" + type.Numbers[number.Number] + "\"" : type.Numbers[number.Number],
94 NumberValue number when json && type.Kind is Kind.UInt64 or Kind.SInt64 or Kind.Fixed64 =>
95 "\"" + number.Number + "\"",
96 NumberValue number => number.Number.ToString(),
97 BooleanValue boolean => boolean.State ? "true" : "false",
98 BinaryValue binary when type.Kind == Kind.String => Quote(binary.Bytes, json),
99 BinaryValue binary => json ? "\"" + Convert.ToBase64String(binary.Bytes) + "\"" : Bytes(binary.Bytes),
100 Message message => Message(message, json),
101 _ => throw new InvalidOperationException()
102 };
103 public static string Sequence(SchemaType type, IEnumerable<Value> values, bool json = false) =>
104 "[" + string.Join(json ? "," : ", ", values.Select(value => Value(type, value, json))) + "]";
105 static string JsonName(string name) => Regex.Replace(name, "_([a-z0-9])", match => match.Groups[1].Value.ToUpperInvariant());
106 public static string Message(Message message, bool json = false)
107 {
108 List<string> items = [];
109 foreach (Field field in message.Type.Fields) {
110 List<Value> values = message.Values(field);
111 if (values.Count == 0) continue;
112 string key = json ? "\"" + JsonName(field.Name) + "\":" : field.Name + ": ";
113 items.Add(key + (field.Repeated ? Sequence(field.Type, values, json) : Value(field.Type, values[0], json)));
114 }
115 return "{" + string.Join(json ? "," : ", ", items) + "}";
116 }
117 public static List<string> Text(Message message)
118 {
119 List<string> lines = [message.Type.Name + " {"];
120 Contents(message, 1, lines);
121 lines.Add("}");
122 return lines;
123 }
124 static void Contents(Message message, int depth, List<string> lines)
125 {
126 string indent = new(' ', depth * 2);
127 foreach (Field field in message.Type.Fields)
128 foreach (Value value in message.Values(field))
129 if (value is Message child) {
130 lines.Add(indent + field.Name + " {");
131 Contents(child, depth + 1, lines);
132 lines.Add(indent + "}");
133 } else lines.Add(indent + field.Name + ": " + Value(field.Type, value));
134 foreach (Unknown unknown in message.Unknown)
135 lines.Add(indent + unknown.Number + ": " + Wire(unknown.Wire) + " " + Raw(unknown));
136 }
137}

139static class Codec
140{
141 public const long MaxUnsigned = 9007199254740991;
142 public static long Zigzag(long value) => value >= 0 ? 2 * value : -2 * value - 1;
143 public static long Unzigzag(long value) => value % 2 == 0 ? value / 2 : -1 - value / 2;
144 public static bool InRange(SchemaType type, long value) => type.Kind switch {
145 Kind.UInt32 or Kind.Fixed32 => value is >= 0 and <= 4294967295,
146 Kind.UInt64 or Kind.Fixed64 => value is >= 0 and <= MaxUnsigned,
147 Kind.SInt32 or Kind.SFixed32 => value is >= -2147483648 and <= 2147483647,
148 Kind.SInt64 => value is >= -4503599627370496 and <= 4503599627370495,
149 _ => true
150 };
151 public static void Varint(List<byte> output, long value)
152 {
153 while (value >= 128) {
154 output.Add((byte)(value % 128 + 128));
155 value /= 128;
156 }
157 output.Add((byte)value);
158 }
159 static void Element(List<byte> output, SchemaType type, Value value)
160 {
161 if (value is Message message) {
162 byte[] payload = Encode(message);
163 Varint(output, payload.Length);
164 output.AddRange(payload);
165 } else if (value is BinaryValue binary) {
166 Varint(output, binary.Bytes.Length);
167 output.AddRange(binary.Bytes);
168 } else {
169 long number = value is BooleanValue boolean ? (boolean.State ? 1 : 0) : ((NumberValue)value).Number;
170 if (type.Kind is Kind.SInt32 or Kind.SInt64) number = Zigzag(number);
171 if (type.Kind == Kind.SFixed32 && number < 0) number += 4294967296;
172 if (type.Wire == 0) Varint(output, number);
173 else for (int i = 0; i < (type.Wire == 1 ? 8 : 4); i++) {
174 output.Add((byte)(number % 256));
175 number /= 256;
176 }
177 }
178 }
179 public static byte[] FieldBytes(Field field, List<Value> values)
180 {
181 List<byte> output = [];
182 if (values.Count == 0) return [];
183 if (field.Packed) {
184 List<byte> payload = [];
185 foreach (Value value in values) Element(payload, field.Type, value);
186 Varint(output, field.Number * 8 + 2);
187 Varint(output, payload.Count);
188 output.AddRange(payload);
189 } else foreach (Value value in values) {
190 Varint(output, field.Number * 8 + field.Type.Wire);
191 Element(output, field.Type, value);
192 }
193 return output.ToArray();
194 }
195 public static byte[] Encode(Message message)
196 {
197 List<byte> output = [];
198 foreach (Field field in message.Type.Fields) output.AddRange(FieldBytes(field, message.Values(field)));
199 foreach (Unknown unknown in message.Unknown) {
200 Varint(output, unknown.Number * 8 + unknown.Wire);
201 if (unknown.Wire == 0) Varint(output, unknown.Integer);
202 else {
203 if (unknown.Wire == 2) Varint(output, unknown.Bytes.Length);
204 output.AddRange(unknown.Bytes);
205 }
206 }
207 return output.ToArray();
208 }
209 public static void Required(Message message, string prefix = "")
210 {
211 foreach (Field field in message.Type.Fields) {
212 List<Value> values = message.Values(field);
213 string path = prefix + field.Name;
214 if (field.Required && values.Count == 0) throw new CodecError("missing required field " + path);
215 if (field.Type.Kind == Kind.Message)
216 for (int i = 0; i < values.Count; i++)
217 Required((Message)values[i], path + (field.Repeated ? "[" + i + "]" : "") + ".");
218 }
219 }
220}

224sealed class Decoder(byte[] data, bool explain = false)
225{
226 public List<string> Lines { get; } = [];
227 static CodecError Error(string reason, int offset) => new(reason + " at byte " + offset);
228 long Varint(ref int position, int end)
229 {
230 int start = position;
231 long value = 0;
232 for (int i = 0; i < 8; i++) {
233 if (position == end) throw Error("truncated varint", start);
234 byte current = data[position++];
235 value += (long)(current & 127) << (7 * i);
236 if (current < 128) {
237 if (value > Codec.MaxUnsigned) throw Error("varint too large", start);
238 return value;
239 }
240 }
241 throw Error("varint too long", start);
242 }
243 Raw Read(int wire, ref int position, int end)
244 {
245 int start = position;
246 if (wire == 0) return new Raw(Varint(ref position, end), [], start, start);
247 long length = wire == 2 ? Varint(ref position, end) : wire == 1 ? 8 : 4;
248 if (length > end - position)
249 throw Error(wire == 2 ? "length past end" : wire == 1 ? "truncated i64" : "truncated i32", start);
250 int payload = position;
251 position += (int)length;
252 return new Raw(0, data[payload..position], start, payload);
253 }
254 Value ConvertValue(SchemaType type, Raw raw)
255 {
256 if (type.Kind == Kind.String) {
257 for (int i = 0; i < raw.Bytes.Length; i++)
258 if (raw.Bytes[i] >= 128) throw Error("invalid string", raw.Payload + i);
259 return new BinaryValue(raw.Bytes);
260 }
261 if (type.Kind == Kind.Bytes) return new BinaryValue(raw.Bytes);
262 long number = raw.Number;
263 if (type.Wire is 1 or 5) {
264 if (type.Kind == Kind.Fixed64 && (raw.Bytes[7] != 0 || raw.Bytes[6] >= 32))
265 throw Error("value out of range", raw.Position);
266 number = 0;
267 for (int i = raw.Bytes.Length - 1; i >= 0; i--) number = number * 256 + raw.Bytes[i];
268 }
269 if (type.Kind is Kind.SInt32 or Kind.SInt64) number = Codec.Unzigzag(number);
270 if (type.Kind == Kind.SFixed32 && number >= 2147483648) number -= 4294967296;
271 if (!Codec.InRange(type, number)) throw Error("value out of range", raw.Position);
272 if (type.Kind == Kind.Bool) {
273 if (number > 1) throw Error("value out of range", raw.Position);
274 return new BooleanValue(number == 1);
275 }
276 if (type.Kind == Kind.Enum && !type.Numbers.ContainsKey(number)) throw Error("unknown enum number", raw.Position);
277 return new NumberValue(number);
278 }
279 Message ReadMessage(SchemaType type, int start, int end, int depth)
280 {
281 Message message = new(type);
282 int position = start;
283 while (position < end) {
284 int offset = position;
285 long tag = Varint(ref position, end), number = tag / 8;
286 int wire = (int)(tag % 8);
287 if (number == 0) throw Error("invalid field number", offset);
288 if (wire is not (0 or 1 or 2 or 5)) throw Error("invalid wire type", offset);
289 Field? field = type.Find(number);
290 bool packed = field is not null && field.Repeated && field.Type.Packable && wire == 2;
291 if (field is not null && wire != field.Type.Wire && !packed) throw Error("wire type mismatch", offset);
292 Raw raw = Read(wire, ref position, end);
293 string header = new string(' ', depth * 2) + offset + " " + number + " " +
294 (field?.Name ?? "?") + " " + Format.Wire(wire) + (wire == 2 ? " " + raw.Bytes.Length : "");
295 if (field is null) {
296 Unknown unknown = new(number, wire, raw.Number, raw.Bytes);
297 if (explain) Lines.Add(header + " " + Format.Raw(unknown));
298 else message.Unknown.Add(unknown);
299 continue;
300 }
301 List<Value> values = [];
302 if (field.Type.Kind == Kind.Message) {
303 if (explain) Lines.Add(header);
304 values.Add(ReadMessage(field.Type, raw.Payload, raw.Payload + raw.Bytes.Length, depth + 1));
305 } else if (packed) {
306 int at = raw.Payload, limit = at + raw.Bytes.Length;
307 while (at < limit) values.Add(ConvertValue(field.Type, Read(field.Type.Wire, ref at, limit)));
308 if (explain) Lines.Add(header + " packed" + string.Concat(values.Select(value => " " + Format.Value(field.Type, value))));
309 } else {
310 values.Add(ConvertValue(field.Type, raw));
311 if (explain) Lines.Add(header + " " + Format.Value(field.Type, values[0]));
312 }
313 if (!explain) message.Apply(field, values);
314 }
315 return message;
316 }
317 public Message Decode(SchemaType type)
318 {
319 Message message = ReadMessage(type, 0, data.Length, 0);
320 if (explain) { if (Lines.Count == 0) Lines.Add("(empty)"); }
321 else Codec.Required(message);
322 return message;
323 }
324}

326enum TokenKind { Symbol, Integer, Name, String, Bytes, End }
327enum Shape { None, Message, List, Integer, Boolean, Name, String, Bytes }
328readonly record struct Token(TokenKind Kind, string Text, int Column, long Number = 0, byte[]? Data = null)
329{
330 public Shape Shape => Kind switch {
331 TokenKind.Integer => Shape.Integer, TokenKind.String => Shape.String, TokenKind.Bytes => Shape.Bytes,
332 TokenKind.Name => Text is "true" or "false" ? Shape.Boolean : Shape.Name,
333 TokenKind.Symbol when Text == "{" => Shape.Message,
334 TokenKind.Symbol when Text == "[" => Shape.List, _ => Shape.None
335 };
336}

338sealed class Literal
339{
340 readonly List<Token> tokens = [];
341 int position;
342 Token Next => tokens[position];
343 static bool Letter(char value) => value is >= 'a' and <= 'z' or >= 'A' and <= 'Z';
344 static bool Digit(char value) => value is >= '0' and <= '9';
345 static int HexDigit(char value) => value is >= '0' and <= '9' ? value - '0' :
346 value is >= 'a' and <= 'f' ? value - 'a' + 10 : -1;
347 static CodecError Error(string reason, Token token) => new(reason + " at column " + token.Column);
348 public Literal(string text)
349 {
350 int at = 0;
351 while (at < text.Length) {
352 char value = text[at];
353 if (value == ' ') { at++; continue; }
354 int start = at++;
355 Token token = new(TokenKind.Symbol, value.ToString(), start + 1);
356 if ("{}[]:,".Contains(value)) tokens.Add(token);
357 else if (value == '"') {
358 List<byte> bytes = [];
359 bool closed = false;
360 while (at < text.Length) {
361 char current = text[at++];
362 if (current == '"') { closed = true; break; }
363 if (current != '\\') { bytes.Add((byte)current); continue; }
364 if (at == text.Length) throw Error("syntax error", token);
365 char escaped = text[at++];
366 if (escaped is '"' or '\\') bytes.Add((byte)escaped);
367 else if (escaped == 'n') bytes.Add(10);
368 else if (escaped == 't') bytes.Add(9);
369 else if (escaped == 'x') {
370 if (at + 2 > text.Length) throw Error("syntax error", token);
371 int high = HexDigit(text[at]), low = HexDigit(text[at + 1]);
372 if (high is < 0 or > 7 || low < 0) throw Error("syntax error", token);
373 bytes.Add((byte)(high * 16 + low));
374 at += 2;
375 } else throw Error("syntax error", token);
376 }
377 if (!closed) throw Error("syntax error", token);
378 tokens.Add(new(TokenKind.String, "", start + 1, Data: bytes.ToArray()));
379 } else if (value == '<') {
380 int end = text.IndexOf('>', at);
381 if (end < 0) throw Error("syntax error", token);
382 string digits = text[at..end];
383 if (digits.Length % 2 != 0 || digits.Any(character => HexDigit(character) < 0)) throw Error("syntax error", token);
384 tokens.Add(new(TokenKind.Bytes, "", start + 1, Data: Convert.FromHexString(digits)));
385 at = end + 1;
386 } else if (value == '-' || Digit(value)) {
387 if (value == '-' && (at == text.Length || !Digit(text[at]))) throw Error("syntax error", token);
388 while (at < text.Length && Digit(text[at])) at++;
389 tokens.Add(new(TokenKind.Integer, "", start + 1, long.Parse(text[start..at])));
390 } else if (Letter(value) || value == '_') {
391 while (at < text.Length && (Letter(text[at]) || Digit(text[at]) || text[at] == '_')) at++;
392 tokens.Add(new(TokenKind.Name, text[start..at], start + 1));
393 } else throw Error("syntax error", token);
394 }
395 tokens.Add(new(TokenKind.End, "", text.Length + 1));
396 }
397 bool Is(string symbol) => Next.Kind == TokenKind.Symbol && Next.Text == symbol;
398 void Expect(string symbol)
399 {
400 if (!Is(symbol)) throw Error("syntax error", Next);
401 position++;
402 }
403 public void End()
404 {
405 if (Next.Kind != TokenKind.End) throw Error("syntax error", Next);
406 }
407 public Message Message(SchemaType type)
408 {
409 Expect("{");
410 Message message = new(type);
411 HashSet<Field> seen = [];
412 HashSet<string> groups = new(StringComparer.Ordinal);
413 if (!Is("}")) while (true) {
414 Token name = Next;
415 if (name.Kind != TokenKind.Name) throw Error("syntax error", name);
416 Field field = type.Find(name.Text) ?? throw Error("unknown field", name);
417 if (!seen.Add(field)) throw Error("duplicate field", name);
418 if (field.Group is not null && !groups.Add(field.Group)) throw Error("oneof conflict", name);
419 position++;
420 Expect(":");
421 message.Fields[field] = FieldValue(field);
422 if (!Is(",")) break;
423 position++;
424 }
425 Token closing = Next;
426 Expect("}");
427 foreach (Field field in type.Fields)
428 if (field.Required && !seen.Contains(field)) throw Error("missing required field " + field.Name, closing);
429 return message;
430 }
431 void Require(Shape shape)
432 {
433 if (Next.Shape == Shape.None) throw Error("syntax error", Next);
434 if (Next.Shape != shape) throw Error("type mismatch", Next);
435 }
436 public List<Value> FieldValue(Field field)
437 {
438 if (!field.Repeated) return [Element(field.Type)];
439 Require(Shape.List);
440 Expect("[");
441 List<Value> values = [];
442 if (!Is("]")) {
443 values.Add(Element(field.Type));
444 while (Is(",")) { position++; values.Add(Element(field.Type)); }
445 }
446 Expect("]");
447 return values;
448 }
449 public Value Element(SchemaType type)
450 {
451 Shape shape = type.Kind switch {
452 Kind.Bool => Shape.Boolean, Kind.String => Shape.String, Kind.Bytes => Shape.Bytes,
453 Kind.Enum => Shape.Name, Kind.Message => Shape.Message, _ => Shape.Integer
454 };
455 Require(shape);
456 Token token = Next;
457 if (shape == Shape.Message) return Message(type);
458 position++;
459 if (shape == Shape.Integer) {
460 if (!Codec.InRange(type, token.Number)) throw Error("out of range", token);
461 return new NumberValue(token.Number);
462 }
463 if (shape == Shape.Boolean) return new BooleanValue(token.Text == "true");
464 if (shape == Shape.Name) {
465 if (!type.Names.TryGetValue(token.Text, out long number)) throw Error("unknown enum value", token);
466 return new NumberValue(number);
467 }
468 return new BinaryValue(token.Data!);
469 }
470}

477sealed class Engine
478{
479 readonly Dictionary<string, SchemaType> types = new(StringComparer.Ordinal);
480 public Engine()
481 {
482 string[] names = ["uint32", "uint64", "sint32", "sint64", "bool", "fixed32", "sfixed32", "fixed64", "string", "bytes"];
483 for (int i = 0; i < names.Length; i++) types[names[i]] = new(names[i], (Kind)i);
484 }
485 public void ReadSchema()
486 {
487 SchemaType? current = null;
488 while (Console.ReadLine() is string line && line != "END SCHEMA") {
489 string[] words = line.Split(' ');
490 if (words[0] is "ENUM" or "MESSAGE") {
491 current = new(words[1], words[0] == "ENUM" ? Kind.Enum : Kind.Message);
492 types.Add(current.Name, current);
493 } else if (words[0] == "VALUE") {
494 long number = long.Parse(words[2]);
495 current!.Numbers.Add(number, words[1]);
496 current.Names.Add(words[1], number);
497 } else if (words[0] == "FIELD") {
498 string mode = words.Length > 4 ? words[4] : "";
499 current!.Fields.Add(new(long.Parse(words[1]), words[2], words[3], mode == "REQUIRED",
500 mode == "REPEATED", words.Length > 5 && words[5] == "PACKED", mode == "ONEOF" ? words[5] : null));
501 }
502 }
503 foreach (SchemaType type in types.Values) {
504 type.Fields.Sort((a, b) => a.Number.CompareTo(b.Number));
505 foreach (Field field in type.Fields) field.Type = types[field.TypeName];
506 }
507 }
508 static List<Segment> Path(SchemaType type, string path)
509 {
510 List<Segment> segments = [];
511 string[] words = path.Split('.');
512 for (int i = 0; i < words.Length; i++) {
513 Match match = Regex.Match(words[i], @"^([a-z][a-z0-9_]*)(?:\[([0-9]{1,6})\])?$");
514 if (!match.Success) throw new CodecError("invalid path");
515 Field? field = type.Find(match.Groups[1].Value);
516 int? index = match.Groups[2].Success ? int.Parse(match.Groups[2].Value) : null;
517 if (field is null || (index is not null && !field.Repeated)) throw new CodecError("invalid path");
518 if (i + 1 < words.Length && (field.Type.Kind != Kind.Message || (field.Repeated && index is null)))
519 throw new CodecError("invalid path");
520 segments.Add(new(field, index));
521 type = field.Type;
522 }
523 return segments;
524 }
525 static string Get(Message message, List<Segment> path)
526 {
527 Message? current = message;
528 for (int step = 0; step < path.Count; step++) {
529 Segment segment = path[step];
530 List<Value> values = current?.Values(segment.Field) ?? [];
531 if (step == path.Count - 1) {
532 if (segment.WholeList) return Format.Sequence(segment.Field.Type, values);
533 int index = segment.Index ?? 0;
534 return index < values.Count ? Format.Value(segment.Field.Type, values[index]) : "(none)";
535 }
536 int at = segment.Index ?? 0;
537 current = at < values.Count ? (Message)values[at] : null;
538 }
539 throw new InvalidOperationException();
540 }
541 static void Edit(Message message, List<Segment> path, List<Value>? replacement)
542 {
543 Message current = message;
544 for (int i = 0; i < path.Count - 1; i++) {
545 Segment segment = path[i];
546 List<Value> values = current.Values(segment.Field);
547 int index = segment.Index ?? 0;
548 if (index >= values.Count) {
549 if (replacement is null) return;
550 if (segment.Index is not null) throw new CodecError("index out of range");
551 current.Choose(segment.Field);
552 values = [new Message(segment.Field.Type)];
553 current.Fields[segment.Field] = values;
554 }
555 current = (Message)values[index];
556 }
557 Segment last = path[^1];
558 if (last.Index is int at) {
559 List<Value> values = current.Values(last.Field);
560 if (at >= values.Count) {
561 if (replacement is not null) throw new CodecError("index out of range");
562 } else if (replacement is null) values.RemoveAt(at);
563 else values[at] = replacement[0];
564 } else if (replacement is null) current.Fields.Remove(last.Field);
565 else {
566 current.Choose(last.Field);
567 current.Fields[last.Field] = replacement;
568 }
569 }
570 static bool Equal(Value a, Value b) => (a, b) switch {
571 (NumberValue x, NumberValue y) => x.Number == y.Number,
572 (BooleanValue x, BooleanValue y) => x.State == y.State,
573 (BinaryValue x, BinaryValue y) => x.Bytes.SequenceEqual(y.Bytes),
574 _ => false
575 };
576 static void Differences(Message a, Message b, string prefix, List<string> lines)
577 {
578 foreach (Field field in a.Type.Fields) {
579 List<Value> x = a.Values(field), y = b.Values(field);
580 for (int i = 0; i < Math.Max(x.Count, y.Count); i++) {
581 string path = prefix + field.Name + (field.Repeated ? "[" + i + "]" : "");
582 if (i < x.Count && i < y.Count && field.Type.Kind == Kind.Message)
583 Differences((Message)x[i], (Message)y[i], path + ".", lines);
584 else if (i >= x.Count || i >= y.Count || !Equal(x[i], y[i]))
585 lines.Add(path + ": " + (i < x.Count ? Format.Value(field.Type, x[i]) : "(none)") +
586 " -> " + (i < y.Count ? Format.Value(field.Type, y[i]) : "(none)"));
587 }
588 }
589 }
590 static List<string> Schema(SchemaType type)
591 {
592 List<string> lines = [type.Name];
593 HashSet<SchemaType> chain = [];
594 void Visit(SchemaType current, int depth)
595 {
596 chain.Add(current);
597 foreach (Field field in current.Fields) {
598 string mode = field.Required ? "required " : field.Packed ? "repeated packed " : field.Repeated ? "repeated " :
599 field.Group is not null ? "oneof " + field.Group + " " : "";
600 string line = new string(' ', depth * 2) + field.Number + " " + field.Name + ": " + mode + field.Type.Name;
601 if (field.Type.Kind == Kind.Enum)
602 line += " (" + string.Join(", ", field.Type.Numbers.Select(pair => pair.Value + "=" + pair.Key)) + ")";
603 bool recursive = field.Type.Kind == Kind.Message && chain.Contains(field.Type);
604 lines.Add(line + (recursive ? " (recursive)" : ""));
605 if (field.Type.Kind == Kind.Message && !recursive) Visit(field.Type, depth + 1);
606 }
607 chain.Remove(current);
608 }
609 Visit(type, 1);
610 return lines;
611 }
612 static byte[] Hex(string text)
613 {
614 if (text == "-") return [];
615 if (text.Length == 0 || text.Length % 2 != 0 || text.Any(c => c is not (>= '0' and <= '9' or >= 'a' and <= 'f')))
616 throw new CodecError("invalid hex");
617 return Convert.FromHexString(text);
618 }
619 static string Rest(string line, int separators)
620 {
621 int at = -1;
622 for (int i = 0; i < separators; i++) at = line.IndexOf(' ', at + 1);
623 return line[(at + 1)..];
624 }
625 static Message Decode(SchemaType type, byte[] bytes, string suffix = "")
626 {
627 try { return new Decoder(bytes).Decode(type); }
628 catch (CodecError error) { throw new CodecError(error.Message + suffix); }
629 }
630 public List<string> Execute(string line)
631 {
632 string[] words = line.Split(' ');
633 string command = words[0];
634 int minimum = command switch {
635 "VARINT" or "ZIGZAG" or "SCHEMA" => 2,
636 "ENCODE" or "SIZE" or "DECODE" or "LITERAL" or "JSON" or "CANON" or "EXPLAIN" => 3,
637 "GET" or "CLEAR" or "MERGE" or "DIFF" => 4, "PATCH" => 5, _ => 0
638 };
639 if (minimum == 0) throw new CodecError("unknown command");
640 if (words.Length < minimum || (command is not ("ENCODE" or "SIZE" or "PATCH") && words.Length != minimum))
641 throw new CodecError("wrong number of arguments");
642 if (command is "VARINT" or "ZIGZAG") {
643 if (!Regex.IsMatch(words[1], @"^-?[0-9]{1,18}$")) throw new CodecError("invalid number");
644 long number = long.Parse(words[1]);
645 if (command == "VARINT" ? number < 0 || number > Codec.MaxUnsigned :
646 number < -4503599627370496 || number > 4503599627370495) throw new CodecError("invalid number");
647 long value = command == "ZIGZAG" ? Codec.Zigzag(number) : number;
648 List<byte> bytes = [];
649 Codec.Varint(bytes, value);
650 return [(command == "ZIGZAG" ? value + " " : "") + Format.Hex(bytes.ToArray())];
651 }
652 if (!types.TryGetValue(words[1], out SchemaType? type) || type.Kind != Kind.Message)
653 throw new CodecError("unknown message type");
654 byte[] data = [], second = [];
655 if (command is not ("SCHEMA" or "ENCODE" or "SIZE")) data = Hex(words[2]);
656 if (command is "MERGE" or "DIFF") second = Hex(words[3]);
657 List<Segment> path = command is "GET" or "PATCH" or "CLEAR" ? Path(type, words[3]) : [];
658 if (command == "SCHEMA") return Schema(type);
659 if (command is "ENCODE" or "SIZE") {
660 Literal literal = new(Rest(line, 2));
661 Message parsed = literal.Message(type);
662 literal.End();
663 byte[] encoded = Codec.Encode(parsed);
664 if (command == "ENCODE") return [Format.Hex(encoded)];
665 return [encoded.Length + string.Concat(type.Fields.Where(field => parsed.Values(field).Count > 0)
666 .Select(field => " " + field.Name + "=" + Codec.FieldBytes(field, parsed.Values(field)).Length))];
667 }
668 if (command == "EXPLAIN") {
669 Decoder decoder = new(data, true);
670 decoder.Decode(type);
671 return decoder.Lines;
672 }
673 if (command == "MERGE") return Format.Text(Decode(type, [.. data, .. second]));
674 if (command == "DIFF") {
675 Message firstMessage = Decode(type, data, " in first");
676 Message secondMessage = Decode(type, second, " in second");
677 List<string> lines = [];
678 Differences(firstMessage, secondMessage, "", lines);
679 return lines.Count == 0 ? ["SAME"] : lines;
680 }
681 Message message = Decode(type, data);
682 if (command == "DECODE") return Format.Text(message);
683 if (command == "LITERAL") return [Format.Message(message)];
684 if (command == "JSON") return [Format.Message(message, true)];
685 if (command == "GET") return [Get(message, path)];
686 if (command == "PATCH") {
687 Literal literal = new(Rest(line, 4));
688 Segment last = path[^1];
689 List<Value> replacement = last.WholeList ? literal.FieldValue(last.Field) : [literal.Element(last.Field.Type)];
690 literal.End();
691 Edit(message, path, replacement);
692 Codec.Required(message);
693 } else if (command == "CLEAR") {
694 Edit(message, path, null);
695 Codec.Required(message);
696 }
697 return [Format.Hex(Codec.Encode(message))];
698 }
699}
