34vecform(v) = "(" * join(v, ",") * ")"
35pname(id) = "P$id"
36plist(ids) = join(pname.(ids), " ")
37ids(s) = sort(collect(keys(s.processes)))
38need(p) = p.maximum - p.allocation
39blocked(s, id) = any(q -> q.id == id, s.queue)
40function request(s, id)
41 for q in s.queue
42 q.id == id && return q.value
43 end
44 zeros(Int, length(s.total))
45end
46function check(s, detect=false, trace=false)
47 work = copy(s.available)
48 remaining = Int[]
49 idle = Int[]
50 sequence = Int[]
51 for id in ids(s)
52 if detect && all(iszero, s.processes[id].allocation)
53 push!(idle, id)
54 else
55 push!(remaining, id)
56 end
57 end
58 tag = detect ? "[detect]" : "[safety]"
59 if trace
60 suffix = isempty(idle) ? "" : " idle " * plist(idle)
61 println(tag, " start work=", vecform(work), suffix)
62 end
63 while true
64 at = findfirst(remaining) do id
65 v = detect ? request(s, id) : need(s.processes[id])
66 all(v .<= work)
67 end
68 isnothing(at) && break
69 id = remaining[at]
70 work += s.processes[id].allocation
71 deleteat!(remaining, at)
72 push!(sequence, id)
73 trace && println(tag, " ", pname(id), " can finish, work=", vecform(work))
74 end
75 if trace && !isempty(remaining)
76 println(tag, " stuck work=", vecform(work), " remaining ", plist(remaining))
77 end
78 isempty(remaining), sequence, remaining
79end
80function decision(m, id, v; reject=true, trace=m.trace)
81 s = m.state
82 p = s.processes[id]
83 if reject
84 blocked(s, id) && return ("rejected", "process $(pname(id)) is blocked", Int[])
85 all(iszero, v) && return ("rejected", "zero vector", Int[])
86 any(v .> need(p)) && return ("rejected", "request exceeds need", Int[])
87 if m.mode == "ORDER"
88 last = findlast(>(0), p.allocation)
89 if !isnothing(last) && any(>(0), v[1:last])
90 return "rejected", "resource order violation", Int[]
91 end
92 end
93 end
94 any(v .> s.available) && return ("blocked", "insufficient resources", Int[])
95 if m.mode == "AVOID"
96 olda, oldp = s.available, p.allocation
97 s.available = olda - v
98 p.allocation = oldp + v
99 ok, seq, _ = check(s, false, trace)
100 s.available, p.allocation = olda, oldp
101 !ok && return ("blocked", "unsafe state", Int[])
102 return "granted", "", seq
103 end
104 "granted", "", Int[]
105end
106function record!(m, kind, id, v, wait=0)
107 push!(m.state.history, Event(m.time, kind, id, copy(v), wait))
108end
109function grant!(s, id, v)
110 s.available -= v
111 s.processes[id].allocation += v
112end
113function remove_wait!(s, id)
114 filter!(q -> q.id != id, s.queue)
115end
116function retry!(m)
117 for q in copy(m.state.queue)
118 result, _, _ = decision(m, q.id, q.value; reject=false)
119 if result == "granted"
120 grant!(m.state, q.id, q.value)
121 remove_wait!(m.state, q.id)
122 println("unblocked ", pname(q.id), " ", vecform(q.value))
123 record!(m, "unblocked", q.id, q.value, m.time - q.since)
124 end
125 end
126end
127function terminate!(m, id, kind; retry=true)
128 s = m.state
129 v = copy(s.processes[id].allocation)
130 s.available += v
131 delete!(s.processes, id)
132 remove_wait!(s, id)
133 println(kind, " ", pname(id), " released ", vecform(v))
134 record!(m, kind, id, v)
135 retry && retry!(m)
136end
137function victims(s)
138 _, _, dead = check(s, true)
139 choices = Vector{Int}[]
140 for mask in 1:(2^length(dead)-1)
141 push!(choices, [dead[i] for i in eachindex(dead) if mask & (1 << (i-1)) != 0])
142 end
143 sort!(choices; by=v -> (length(v), sum(sum(s.processes[id].allocation) for id in v), v))
144 for choice in choices
145 trial = deepcopy(s)
146 for id in choice
147 trial.available += trial.processes[id].allocation
148 delete!(trial.processes, id)
149 end
150 ok, _, _ = check(trial, true)
151 ok && return choice
152 end
153 Int[]
154end
155function graph(m, resource)
156 s = m.state
157 order = ids(s)
158 edges = Dict(id => Tuple{Int,Int}[] for id in order)
159 if resource
160 if any(!=(1), s.total)
161 println("error: not single-instance")
162 return
163 end
164 for j in eachindex(s.total), id in order
165 s.processes[id].allocation[j] > 0 && println("assign ", Char('A'+j-1), " -> ", pname(id))
166 end
167 for id in order
168 r = request(s, id)
169 for j in eachindex(r)
170 r[j] == 0 && continue
171 println("request ", pname(id), " -> ", Char('A'+j-1))
172 for to in order
173 s.processes[to].allocation[j] > 0 && push!(edges[id], (to, j))
174 end
175 end
176 end
177 for id in order
178 r = request(s, id)
179 for j in eachindex(r)
180 if need(s.processes[id])[j] > 0 && r[j] == 0
181 println("claim ", pname(id), " -> ", Char('A'+j-1))
182 end
183 end
184 end
185 else
186 count = 0
187 for id in order, to in order
188 id == to && continue
189 labels = [string(Char('A'+j-1)) for j in eachindex(s.total)
190 if request(s, id)[j] > s.available[j] && s.processes[to].allocation[j] > 0]
191 if !isempty(labels)
192 println(pname(id), " -> ", pname(to), " on ", join(labels, ","))
193 push!(edges[id], (to, 0))
194 count += 1
195 end
196 end
197 count == 0 && println("no wait edges")
198 end
199 cycles = Vector{Int}[]
200 for start in order
201 function visit(at, path, seen)
202 for (to, type) in edges[at]
203 next = resource ? [path; type; to] : [path; to]
204 if to == start
205 (resource || length(seen) >= 2) && push!(cycles, next)
206 elseif to > start && !(to in seen)
207 visit(to, next, union(seen, Set([to])))
208 end
209 end
210 end
211 visit(start, [start], Set([start]))
212 end
213 sort!(cycles)
214 for cycle in cycles
215 parts = [resource && iseven(i) ? string(Char('A'+x-1)) : pname(x)
216 for (i, x) in enumerate(cycle)]
217 println("cycle: ", join(parts, " -> "))
218 end
219 isempty(cycles) && println("no cycles")
220end
221function table(s)
222 widths = length.(string.(s.total))
223 names = join([lpad(string(Char('A'+i-1)), widths[i]) for i in eachindex(widths)], " ")
224 cw = max.([10, 3, 4, 7], length(names))
225 form(v) = join([lpad(string(v[i]), widths[i]) for i in eachindex(v)], " ")
226 function row(heading, cols)
227 line = rpad(heading, 9)
228 for i in 1:4
229 line *= " " * rpad(i <= length(cols) ? cols[i] : "", cw[i])
230 end
231 println(rstrip(line))
232 end
233 row("Process", ["Allocation", "Max", "Need", "Request"])
234 row("", fill(names, 4))
235 for id in ids(s)
236 p = s.processes[id]
237 row(pname(id), form.([p.allocation, p.maximum, need(p), request(s, id)]))
238 end
239 row("Available", [form(s.available)])
240 row("Total", [form(s.total)])
241 if isempty(s.queue)
242 println("blocked queue: empty")
243 else
244 println("blocked queue: ", join(["$(pname(q.id)) $(vecform(q.value)) since t=$(q.since)" for q in s.queue], ", "))
245 end
246end
247function execute!(m, words)
248 cmd = words[1]
249 args = words[2:end]
250 s = m.state
251 hasid = cmd in ["PROCESS", "CLAIM", "REQUEST", "WHATIF", "RELEASE", "FINISH", "ABORT", "CANCEL", "MAXGRANT"]
252 hasvector = cmd in ["PROCESS", "CLAIM", "REQUEST", "WHATIF", "RELEASE", "ADD", "REMOVE"]
253 id = hasid ? parse(Int, args[1]) : 0
254 v = hasvector ? parse.(Int, args[(hasid ? 2 : 1):end]) : Int[]
255 err = ""
256 if hasvector && length(v) != length(s.total)
257 err = "wrong vector length"
258 elseif hasid && cmd == "PROCESS" && haskey(s.processes, id)
259 err = "duplicate process $(pname(id))"
260 elseif hasid && cmd != "PROCESS" && !haskey(s.processes, id)
261 err = "unknown process $(pname(id))"
262 end
263 if !isempty(err)
264 println("error: ", err)
265 return
266 end
267 if cmd == "RESOURCES"
268 total = parse.(Int, args)
269 m.state = State(total, copy(total), Dict{Int,Process}(), Waiting[], Event[])
270 println("resources ", vecform(total))
271 elseif cmd in ["MODE", "POLICY", "TRACE", "AUTODETECT"]
272 if cmd == "MODE"
273 m.mode = args[1]
274 elseif cmd == "POLICY"
275 m.policy = args[1]
276 elseif cmd == "TRACE"
277 m.trace = args[1] == "ON"
278 else
279 m.automatic = args[1] == "ON"
280 end
281 println(lowercase(cmd), " ", args[1])
282 elseif cmd == "PROCESS"
283 if any(v .> s.total)
284 err = "max exceeds total"
285 else
286 s.processes[id] = Process(v, zeros(Int, length(v)), m.time)
287 println("process ", pname(id), " max ", vecform(v))
288 end
289 elseif cmd == "CLAIM"
290 p = s.processes[id]
291 if blocked(s, id)
292 err = "process $(pname(id)) is blocked"
293 elseif any(p.allocation .> v)
294 err = "claim below allocation"
295 elseif any(v .> s.total)
296 err = "max exceeds total"
297 else
298 old = p.maximum
299 p.maximum = v
300 if m.mode == "AVOID"
301 ok, _, _ = check(s, false, m.trace)
302 if !ok
303 p.maximum = old
304 err = "claim makes state unsafe"
305 end
306 end
307 if isempty(err)
308 println("claim ", pname(id), " max ", vecform(v))
309 retry!(m)
310 end
311 end
312 elseif cmd in ["ADD", "REMOVE"]
313 if all(iszero, v)
314 err = "zero vector"
315 elseif cmd == "ADD"
316 s.total += v
317 s.available += v
318 println("added ", vecform(v), " total ", vecform(s.total))
319 retry!(m)
320 elseif any(v .> s.available)
321 err = "remove exceeds available"
322 elseif any(iszero, s.total - v)
323 err = "total must stay positive"
324 elseif any(p -> any(p.maximum .> s.total - v), values(s.processes))
325 err = "max exceeds total"
326 else
327 old = s.available
328 s.available = old - v
329 if m.mode == "AVOID"
330 ok, _, _ = check(s, false, m.trace)
331 !ok && (err = "removal makes state unsafe")
332 end
333 if isempty(err)
334 s.total -= v
335 println("removed ", vecform(v), " total ", vecform(s.total))
336 else
337 s.available = old
338 end
339 end
340 elseif cmd in ["REQUEST", "WHATIF"]
341 result, reason, seq = decision(m, id, v)
342 if cmd == "WHATIF"
343 line = "whatif $(pname(id)) $(vecform(v)): $result"
344 if result != "granted"
345 line *= " ($reason)"
346 elseif m.mode == "AVOID"
347 line *= ", safe sequence " * plist(seq)
348 end
349 println(line)
350 elseif result == "rejected"
351 err = reason
352 else
353 if result == "granted"
354 grant!(s, id, v)
355 println("granted ", pname(id), " ", vecform(v))
356 else
357 push!(s.queue, Waiting(id, v, m.time))
358 println("blocked ", pname(id), " ", vecform(v), ": ", reason)
359 end
360 record!(m, result, id, v)
361 if result == "blocked" && m.automatic
362 _, _, dead = check(s, true, m.trace)
363 !isempty(dead) && println("deadlock detected: ", plist(dead))
364 end
365 end
366 elseif cmd == "RELEASE"
367 p = s.processes[id]
368 if blocked(s, id)
369 err = "process $(pname(id)) is blocked"
370 elseif all(iszero, v)
371 err = "zero vector"
372 elseif any(v .> p.allocation)
373 err = "release exceeds allocation"
374 else
375 p.allocation -= v
376 s.available += v
377 println("released ", pname(id), " ", vecform(v))
378 record!(m, "released", id, v)
379 retry!(m)
380 end
381 elseif cmd in ["FINISH", "ABORT"]
382 if cmd == "FINISH" && blocked(s, id)
383 err = "process $(pname(id)) is blocked"
384 else
385 terminate!(m, id, cmd == "FINISH" ? "finished" : "aborted")
386 end
387 elseif cmd == "CANCEL"
388 if !blocked(s, id)
389 err = "process $(pname(id)) is not blocked"
390 else
391 v = request(s, id)
392 remove_wait!(s, id)
393 println("cancelled ", pname(id), " ", vecform(v))
394 record!(m, "cancelled", id, v)
395 end
396 elseif cmd == "MAXGRANT"
397 if blocked(s, id)
398 err = "process $(pname(id)) is blocked"
399 else
400 result = zeros(Int, length(s.total))
401 for j in eachindex(result)
402 for x in need(s.processes[id])[j]:-1:1
403 v = zeros(Int, length(result))
404 v[j] = x
405 decision(m, id, v; trace=false)[1] == "granted" || continue
406 result[j] = x
407 break
408 end
409 end
410 println("maxgrant ", pname(id), ": ", vecform(result))
411 end
412 elseif cmd == "SAFE"
413 if isempty(s.processes)
414 println("safe: no processes")
415 else
416 ok, seq, left = check(s, false, m.trace)
417 println(ok ? "safe: " * plist(seq) : "unsafe: " * plist(left) * " cannot finish")
418 end
419 elseif cmd == "SEQUENCES"
420 if isempty(s.processes)
421 println("safe sequences: no processes")
422 else
423 count = 0
424 samples = String[]
425 k = parse(Int, args[1])
426 order = ids(s)
427 function search(work, path)
428 if length(path) == length(order)
429 count += 1
430 length(samples) < k && push!(samples, plist(path))
431 return
432 end
433 for id in order
434 p = s.processes[id]
435 if !(id in path) && all(need(p) .<= work)
436 search(work + p.allocation, [path; id])
437 end
438 end
439 end
440 search(s.available, Int[])
441 println("safe sequences: ", count)
442 foreach(println, samples)
443 end
444 elseif cmd == "DEFICIT"
445 result = String[]
446 for j in eachindex(s.total)
447 limit = sum((need(p)[j] for p in values(s.processes)); init=0)
448 value = "-"
449 for x in 0:limit
450 trial = deepcopy(s)
451 trial.available[j] += x
452 if check(trial)[1]
453 value = string(x)
454 break
455 end
456 end
457 push!(result, value)
458 end
459 println("deficit: ", vecform(result))
460 elseif cmd == "STATE"
461 table(s)
462 elseif cmd == "STATS"
463 for j in eachindex(s.total)
464 total = s.total[j]
465 used = total - s.available[j]
466 requested = sum((q.value[j] for q in s.queue); init=0)
467 pct = used * 1000 ÷ total
468 println(Char('A'+j-1), ": allocated $used/$total (", pct ÷ 10, ".", pct % 10, "%), requested ", requested)
469 end
470 kinds = ["granted", "blocked", "unblocked", "released", "finished", "aborted", "cancelled"]
471 counts = Dict(kind => Base.count(e -> e.kind == kind, s.history) for kind in kinds)
472 println("events: ", join(["$kind $(counts[kind])" for kind in kinds], ", "))
473 n = counts["unblocked"]
474 if n == 0
475 println("waits: none")
476 else
477 total = sum(e.wait for e in s.history if e.kind == "unblocked")
478 avg = total * 100 ÷ n
479 println("waits: $n unblocked, total $total, average ", avg ÷ 100, ".", lpad(string(avg % 100), 2, '0'))
480 end
481 elseif cmd == "DETECT"
482 _, _, dead = check(s, true, m.trace)
483 println(isempty(dead) ? "no deadlock" : "deadlocked: " * plist(dead))
484 elseif cmd == "VICTIMS"
485 choice = victims(s)
486 println(isempty(choice) ? "no deadlock" : "victims: " * plist(choice))
487 elseif cmd == "RECOVER"
488 count = 0
489 while true
490 _, _, dead = check(s, true, m.trace)
491 isempty(dead) && break
492 if !isempty(args)
493 for id in dead
494 terminate!(m, id, "aborted"; retry=false)
495 count += 1
496 end
497 retry!(m)
498 break
499 end
500 function priority(id)
501 p = s.processes[id]
502 m.policy == "ALLOC" && return (sum(p.allocation), -id)
503 m.policy == "NEED" && return (-sum(need(p)), -id)
504 (-p.created, -id)
505 end
506 sort!(dead; by=priority)
507 terminate!(m, dead[1], "aborted")
508 count += 1
509 end
510 println(count == 0 ? "no deadlock" : "recovered: $count aborted")
511 elseif cmd in ["WAITGRAPH", "GRAPH"]
512 graph(m, cmd == "GRAPH")
513 elseif cmd == "HISTORY"
514 found = false
515 for e in s.history
516 if !isempty(args) && args[1] != e.kind && args[1] != string(e.id)
517 continue
518 end
519 line = "t=$(e.time) $(e.kind) $(pname(e.id)) $(vecform(e.value))"
520 e.kind == "unblocked" && (line *= " waited $(e.wait)")
521 println(line)
522 found = true
523 end
524 !found && println("no events")
525 elseif cmd == "SNAPSHOT"
526 m.snapshots[args[1]] = deepcopy(s)
527 println("saved ", args[1])
528 elseif cmd == "RESTORE"
529 if !haskey(m.snapshots, args[1])
530 err = "unknown snapshot $(args[1])"
531 else
532 m.state = deepcopy(m.snapshots[args[1]])
533 println("restored ", args[1])
534 end
535 end
536 !isempty(err) && println("error: ", err)
537end
538function main()
539 m = Manager(State(Int[], Int[], Dict{Int,Process}(), Waiting[], Event[]), "AVOID", "ALLOC", false, false, 0, Dict{String,State}())
540 n = parse(Int, readline())
541 for t in 1:n
542 m.time = t
543 execute!(m, String.(split(readline())))
544 end
545end
