1class Manager
2 KINDS = %w[granted blocked unblocked released finished aborted cancelled].freeze

4 def initialize
5 @mode, @policy, @trace, @auto = 'AVOID', 'ALLOC', false, false
6 @snapshots = {}
7 @total, @available, @processes, @queue, @history = [], [], {}, [], []
8 end

10 def vector(v)
11 "(#{v.join(',')})"
12 end
13
14 def names(ids)
15 ids.map { |id| "P#{id}" }.join(' ')
16 end

18 def add(a, b)
19 a.zip(b).map { |x, y| x + y }
20 end
21
22 def sub(a, b)
23 a.zip(b).map { |x, y| x - y }
24 end

26 def fits(a, b)
27 a.zip(b).all? { |x, y| x <= y }
28 end

30 def zero
31 Array.new(@total.size, 0)
32 end

34 def need(id, ps = @processes)
35 sub(ps[id][:max], ps[id][:alloc])
36 end
37
38 def check(detect = false, trace = false, av = @available, ps = @processes, queue = @queue)
39 work = av.dup
40 idle = detect ? ps.keys.select { |id| ps[id][:alloc].all?(&:zero?) }.sort : []
41 remaining = ps.keys.sort - idle
42 sequence = []
43 label = detect ? 'detect' : 'safety'
44 if trace
45 suffix = idle.empty? ? '' : " idle #{names(idle)}"
46 puts "[#{label}] start work=#{vector(work)}#{suffix}"
47 end
48 loop do
49 id = remaining.find do |p|
50 request = queue.find { |q| q[0] == p }
51 required = detect ? (request ? request[1] : zero) : need(p, ps)
52 fits(required, work)
53 end
54 break unless id
55 work = add(work, ps[id][:alloc])
56 remaining.delete(id)
57 sequence << id
58 puts "[#{label}] P#{id} can finish, work=#{vector(work)}" if trace
59 end
60 if trace && !remaining.empty?
61 puts "[#{label}] stuck work=#{vector(work)} remaining #{names(remaining)}"
62 end
63 [remaining.empty?, sequence, remaining]
64 end

66 def blocked(id)
67 @queue.find { |q| q[0] == id }
68 end
69
70 def reject(id, v)
71 return "process P#{id} is blocked" if blocked(id)
72 return 'zero vector' if v.all?(&:zero?)
73 return 'request exceeds need' unless fits(v, need(id))
74 held = @processes[id][:alloc].each_index.select { |i| @processes[id][:alloc][i] > 0 }
75 if @mode == 'ORDER' && !held.empty? && v.each_index.any? { |i| v[i] > 0 && i <= held.max }
76 return 'resource order violation'
77 end
78 nil
79 end
80
81 def decision(id, v, trace = @trace)
82 return [false, 'insufficient resources', []] unless fits(v, @available)
83 if @mode == 'AVOID'
84 ps = Marshal.load(Marshal.dump(@processes))
85 ps[id][:alloc] = add(ps[id][:alloc], v)
86 safe, sequence, = check(false, trace, sub(@available, v), ps)
87 return [false, 'unsafe state', []] unless safe
88 return [true, '', sequence]
89 end
90 [true, '', []]
91 end

93 def record(kind, id, v, wait = nil)
94 @history << [@time, kind, id, v.dup, wait]
95 end
96
97 def grant(id, v)
98 @available = sub(@available, v)
99 @processes[id][:alloc] = add(@processes[id][:alloc], v)
100 end
101
102 def retry_queue
103 old = @queue
104 @queue = []
105 old.each do |id, v, since|
106 if decision(id, v)[0]
107 grant(id, v)
108 puts "unblocked P#{id} #{vector(v)}"
109 record('unblocked', id, v, @time - since)
110 else
111 @queue << [id, v, since]
112 end
113 end
114 end
115
116 def terminate(id, kind, retry_after = true)
117 allocation = @processes[id][:alloc]
118 @available = add(@available, allocation)
119 @processes.delete(id)
120 @queue.reject! { |q| q[0] == id }
121 puts "#{kind} P#{id} released #{vector(allocation)}"
122 record(kind, id, allocation)
123 retry_queue if retry_after
124 end

126 def fail_if(condition, message)
127 raise message if condition

130 def run(line, time)
131 @time = time
132 command, *args = line.split
133 id = args[0].to_i
134 vector_commands = %w[PROCESS CLAIM REQUEST WHATIF RELEASE]
135 v = (vector_commands.include?(command) ? args.drop(1) : args).map(&:to_i)
136 if (vector_commands + %w[ADD REMOVE]).include?(command)
137 fail_if(v.size != @total.size, 'wrong vector length')
138 end
139 if %w[CLAIM REQUEST WHATIF RELEASE FINISH ABORT CANCEL MAXGRANT].include?(command)
140 fail_if(!@processes.key?(id), "unknown process P#{id}")
141 end
142 case command
143 when 'RESOURCES'
144 @total, @available = v.dup, v.dup
145 @processes, @queue, @history = {}, [], []
146 puts "resources #{vector(v)}"
147 when 'MODE', 'POLICY', 'TRACE', 'AUTODETECT'
148 case command
149 when 'MODE' then @mode = args[0]
150 when 'POLICY' then @policy = args[0]
151 when 'TRACE' then @trace = args[0] == 'ON'
152 when 'AUTODETECT' then @auto = args[0] == 'ON'
153 end
154 puts "#{command.downcase} #{args[0]}"
155 when 'PROCESS'
156 fail_if(@processes.key?(id), "duplicate process P#{id}")
157 fail_if(!fits(v, @total), 'max exceeds total')
158 @processes[id] = {max: v, alloc: zero, born: @time}
159 puts "process P#{id} max #{vector(v)}"
160 when 'CLAIM'
161 fail_if(blocked(id), "process P#{id} is blocked")
162 fail_if(!fits(@processes[id][:alloc], v), 'claim below allocation')
163 fail_if(!fits(v, @total), 'max exceeds total')
164 ps = Marshal.load(Marshal.dump(@processes))
165 ps[id][:max] = v
166 fail_if(@mode == 'AVOID' && !check(false, @trace, @available, ps)[0], 'claim makes state unsafe')
167 @processes[id][:max] = v
168 puts "claim P#{id} max #{vector(v)}"
169 retry_queue
170 when 'ADD', 'REMOVE'
171 fail_if(v.all?(&:zero?), 'zero vector')
172 if command == 'ADD'
173 @total, @available = add(@total, v), add(@available, v)
174 puts "added #{vector(v)} total #{vector(@total)}"
175 retry_queue
176 else
177 fail_if(!fits(v, @available), 'remove exceeds available')
178 total = sub(@total, v)
179 fail_if(total.include?(0), 'total must stay positive')
180 fail_if(@processes.keys.any? { |p| !fits(@processes[p][:max], total) }, 'max exceeds total')
181 available = sub(@available, v)
182 fail_if(@mode == 'AVOID' && !check(false, @trace, available)[0], 'removal makes state unsafe')
183 @total, @available = total, available
184 puts "removed #{vector(v)} total #{vector(@total)}"
185 end
186 when 'REQUEST', 'WHATIF'
187 reason = reject(id, v)
188 if reason
189 raise reason if command == 'REQUEST'
190 puts "whatif P#{id} #{vector(v)}: rejected (#{reason})"
191 else
192 granted, reason, sequence = decision(id, v)
193 if command == 'WHATIF'
194 result = granted ? 'granted' : "blocked (#{reason})"
195 result += ", safe sequence #{names(sequence)}" if granted && @mode == 'AVOID'
196 puts "whatif P#{id} #{vector(v)}: #{result}"
197 elsif granted
198 grant(id, v)
199 puts "granted P#{id} #{vector(v)}"
200 record('granted', id, v)
201 else
202 @queue << [id, v, @time]
203 puts "blocked P#{id} #{vector(v)}: #{reason}"
204 record('blocked', id, v)
205 if @auto
206 dead = check(true, @trace)[2]
207 puts "deadlock detected: #{names(dead)}" unless dead.empty?
208 end
209 end
210 end
211 when 'RELEASE'
212 fail_if(blocked(id), "process P#{id} is blocked")
213 fail_if(v.all?(&:zero?), 'zero vector')
214 fail_if(!fits(v, @processes[id][:alloc]), 'release exceeds allocation')
215 @available = add(@available, v)
216 @processes[id][:alloc] = sub(@processes[id][:alloc], v)
217 puts "released P#{id} #{vector(v)}"
218 record('released', id, v)
219 retry_queue
220 when 'FINISH', 'ABORT'
221 fail_if(command == 'FINISH' && blocked(id), "process P#{id} is blocked")
222 terminate(id, command == 'FINISH' ? 'finished' : 'aborted')
223 when 'CANCEL'
224 request = blocked(id)
225 fail_if(!request, "process P#{id} is not blocked")
226 @queue.delete(request)
227 puts "cancelled P#{id} #{vector(request[1])}"
228 record('cancelled', id, request[1])
229 when 'MAXGRANT'
230 fail_if(blocked(id), "process P#{id} is blocked")
231 result = need(id).each_index.map do |i|
232 (0..need(id)[i]).to_a.reverse.find do |x|
233 w = zero
234 w[i] = x
235 x == 0 || (!reject(id, w) && decision(id, w, false)[0])
236 end
237 end
238 puts "maxgrant P#{id}: #{vector(result)}"
239 when 'SAFE'
240 if @processes.empty?
241 puts 'safe: no processes'
242 else
243 safe, sequence, remaining = check(false, @trace)
244 puts(safe ? "safe: #{names(sequence)}" : "unsafe: #{names(remaining)} cannot finish")
245 end
246 when 'SEQUENCES'
247 if @processes.empty?
248 puts 'safe sequences: no processes'
249 else
250 count, first = 0, []
251 visit = lambda do |work, remaining, path|
252 if remaining.empty?
253 count += 1
254 first << path if first.size < id
255 else
256 remaining.each do |p|
257 visit.call(add(work, @processes[p][:alloc]), remaining - [p], path + [p]) if fits(need(p), work)
258 end
259 end
260 end
261 visit.call(@available, @processes.keys.sort, [])
262 puts "safe sequences: #{count}"
263 first.each { |path| puts names(path) }
264 end
265 when 'DEFICIT'
266 result = @total.each_index.map do |i|
267 limit = @processes.keys.sum { |p| need(p)[i] }
268 (0..limit).find do |x|
269 work = @available.dup
270 work[i] += x
271 check(false, false, work)[0]
272 end || '-'
273 end
274 puts "deficit: #{vector(result)}"
275 when 'STATE'
276 widths = @total.map { |n| n.to_s.size }
277 table = ->(w) { w.each_with_index.map { |x, i| x.to_s.rjust(widths[i]) }.join(' ') }
278 name_row = table.call(@total.each_index.map { |i| (65 + i).chr })
279 columns = [10, 3, 4, 7].map { |w| [w, name_row.size].max }
280 row = lambda do |heading, content|
281 puts((heading.ljust(9) + content.each_with_index.map { |x, i| ' ' + x.ljust(columns[i]) }.join).rstrip)
282 end
283 row.call('Process', %w[Allocation Max Need Request])
284 row.call('', Array.new(4, name_row))
285 @processes.keys.sort.each do |p|
286 request = blocked(p)
287 row.call("P#{p}", [@processes[p][:alloc], @processes[p][:max], need(p), request ? request[1] : zero].map { |w| table.call(w) })
288 end
289 row.call('Available', [table.call(@available), '', '', ''])
290 row.call('Total', [table.call(@total), '', '', ''])
291 text = @queue.map { |p, w, s| "P#{p} #{vector(w)} since t=#{s}" }.join(', ')
292 puts "blocked queue: #{text.empty? ? 'empty' : text}"
293 when 'STATS'
294 @total.each_index do |i|
295 used = @total[i] - @available[i]
296 percent = used * 1000 / @total[i]
297 requested = @queue.sum { |q| q[1][i] }
298 puts "#{(65 + i).chr}: allocated #{used}/#{@total[i]} (#{percent / 10}.#{percent % 10}%), requested #{requested}"
299 end
300 puts 'events: ' + KINDS.map { |k| "#{k} #{@history.count { |e| e[1] == k }}" }.join(', ')
301 waits = @history.select { |e| e[1] == 'unblocked' }.map { |e| e[4] }
302 if waits.empty?
303 puts 'waits: none'
304 else
305 average = waits.sum * 100 / waits.size
306 puts "waits: #{waits.size} unblocked, total #{waits.sum}, average #{average / 100}.#{(average % 100).to_s.rjust(2, '0')}"
307 end
308 when 'DETECT'
309 dead = check(true, @trace)[2]
310 puts(dead.empty? ? 'no deadlock' : "deadlocked: #{names(dead)}")
311 when 'VICTIMS'
312 dead = check(true)[2]
313 if dead.empty?
314 puts 'no deadlock'
315 else
316 candidates = (1..dead.size).flat_map { |n| dead.combination(n).to_a }
317 candidates.sort_by! { |s| [s.size, s.sum { |p| @processes[p][:alloc].sum }, s] }
318 victims = candidates.find do |s|
319 ps = @processes.reject { |p, _| s.include?(p) }
320 work = s.reduce(@available) { |a, p| add(a, @processes[p][:alloc]) }
321 check(true, false, work, ps, @queue.reject { |q| s.include?(q[0]) })[0]
322 end
323 puts "victims: #{names(victims)}"
324 end
325 when 'RECOVER'
326 count = 0
327 loop do
328 dead = check(true, @trace)[2]
329 break if dead.empty?
330 if args[0] == 'ALL'
331 dead.each { |p| terminate(p, 'aborted', false) }
332 count += dead.size
333 retry_queue
334 break
335 end
336 p = dead.min_by do |q|
337 case @policy
338 when 'ALLOC' then [@processes[q][:alloc].sum, -q]
339 when 'NEED' then [-need(q).sum, -q]
340 else [-@processes[q][:born]]
341 end
342 end
343 terminate(p, 'aborted')
344 count += 1
345 end
346 puts(count == 0 ? 'no deadlock' : "recovered: #{count} aborted")
347 when 'WAITGRAPH', 'GRAPH'
348 graph(command == 'GRAPH')
349 when 'HISTORY'
350 events = @history.select { |e| args.empty? || e[1] == args[0] || (args[0].match?(/\A\d+\z/) && e[2] == id) }
351 puts 'no events' if events.empty?
352 events.each do |s, kind, p, w, wait|
353 text = "t=#{s} #{kind} P#{p} #{vector(w)}"
354 text += " waited #{wait}" if kind == 'unblocked'
355 puts text
356 end
357 when 'SNAPSHOT'
358 @snapshots[args[0]] = Marshal.dump([@total, @available, @processes, @queue, @history])
359 puts "saved #{args[0]}"
360 when 'RESTORE'
361 fail_if(!@snapshots.key?(args[0]), "unknown snapshot #{args[0]}")
362 @total, @available, @processes, @queue, @history = Marshal.load(@snapshots[args[0]])
363 puts "restored #{args[0]}"
364 end
365 rescue RuntimeError => e
366 puts "error: #{e.message}"
367 end

369 def graph(single)
370 fail_if(single && @total.any? { |x| x != 1 }, 'not single-instance')
371 ids = @processes.keys.sort
372 edges = ids.to_h { |p| [p, []] }
373 if single
374 @total.each_index do |i|
375 ids.each { |p| puts "assign #{(65 + i).chr} -> P#{p}" if @processes[p][:alloc][i] > 0 }
376 end
377 ids.each do |p|
378 q = blocked(p)
379 next unless q
380 @total.each_index do |i|
381 next if q[1][i] == 0
382 puts "request P#{p} -> #{(65 + i).chr}"
383 holder = ids.find { |j| @processes[j][:alloc][i] > 0 }
384 edges[p] << [i, holder] if holder
385 end
386 end
387 ids.each do |p|
388 q = blocked(p)
389 @total.each_index do |i|
390 puts "claim P#{p} -> #{(65 + i).chr}" if need(p)[i] > 0 && (!q || q[1][i] == 0)
391 end
392 end
393 else
394 ids.each do |p|
395 q = blocked(p)
396 next unless q
397 ids.each do |j|
398 next if p == j
399 labels = @total.each_index.select { |i| q[1][i] > @available[i] && @processes[j][:alloc][i] > 0 }
400 next if labels.empty?
401 edges[p] << [j, j]
402 puts "P#{p} -> P#{j} on #{labels.map { |i| (65 + i).chr }.join(',')}"
403 end
404 end
405 puts 'no wait edges' if edges.values.all?(&:empty?)
406 end
407 cycles = []
408 ids.each do |start|
409 visit = lambda do |p, seen, path|
410 edges[p].each do |label, next_p|
411 extension = single ? [label, next_p] : [next_p]
412 if next_p == start
413 cycles << path + extension if single || seen.size >= 2
414 elsif next_p > start && !seen.include?(next_p)
415 visit.call(next_p, seen + [next_p], path + extension)
416 end
417 end
418 end
419 visit.call(start, [start], [start])
420 end
421 cycles.sort.each do |path|
422 text = path.each_with_index.map { |x, i| single && i.odd? ? (65 + x).chr : "P#{x}" }.join(' -> ')
423 puts "cycle: #{text}"
424 end
425 puts 'no cycles' if cycles.empty?
426 end
427end
