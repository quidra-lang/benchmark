1import java.util.TreeMap

3private data class Process(val id: Int, var maximum: List<Int>, val created: Int, var allocation: List<Int>) {
4 fun need(): List<Int> = sub(maximum, allocation)
5 fun duplicate() = copy(maximum = maximum.toList(), allocation = allocation.toList())
6}
7private data class Waiting(val id: Int, val vector: List<Int>, val since: Int)
8private data class Event(val command: Int, val kind: String, val id: Int, val vector: List<Int>, val waited: Int = 0)
9private data class State(
10 var total: List<Int>, var available: List<Int> = total.toList(),
11 val processes: TreeMap<Int, Process> = TreeMap(),
12 var queue: MutableList<Waiting> = mutableListOf(),
13 val history: MutableList<Event> = mutableListOf()
14) {
15 fun duplicate(): State {
16 val s = State(total.toList(), available.toList())
17 processes.forEach { (id, p) -> s.processes[id] = p.duplicate() }
18 s.queue.addAll(queue.map { it.copy(vector = it.vector.toList()) })
19 s.history.addAll(history.map { it.copy(vector = it.vector.toList()) })
20 return s
21 }
22}
23private data class Check(val sequence: List<Int>, val remaining: List<Int>)
24private data class Decision(val reason: String?, val sequence: List<Int> = emptyList())
25private var state = State(emptyList())
26private val snapshots = mutableMapOf<String, State>()
27private var mode = "AVOID"
28private var policy = "ALLOC"
29private var tracing = false
30private var autodetect = false
31private var command = 0
32private val output = StringBuilder()
33private val kinds = listOf("granted", "blocked", "unblocked", "released", "finished", "aborted", "cancelled")
34private fun emit(line: String) { output.append(line.trimEnd()).append('\n') }
35private fun pn(id: Int) = "P$id"
36private fun pl(ids: List<Int>) = ids.joinToString(" ") { pn(it) }
37private fun vl(v: List<Int>) = v.joinToString(",", "(", ")")
38private fun type(i: Int) = ('A'.code + i).toChar().toString()
39private fun add(a: List<Int>, b: List<Int>) = a.indices.map { a[it] + b[it] }
40private fun sub(a: List<Int>, b: List<Int>) = a.indices.map { a[it] - b[it] }
41private fun fits(a: List<Int>, b: List<Int>) = a.indices.all { a[it] <= b[it] }
42private fun fail(message: String): Nothing = throw IllegalArgumentException(message)
43private fun waiting(id: Int) = state.queue.find { it.id == id }
44private fun requestOf(id: Int) = waiting(id)?.vector ?: List(state.total.size) { 0 }
45private fun requireProcess(id: Int, unblocked: Boolean = false): Process {
46 val p = state.processes[id] ?: fail("unknown process ${pn(id)}")
47 if (unblocked && waiting(id) != null) fail("process ${pn(id)} is blocked")
48 return p
49}
50private fun vector(words: List<String>, start: Int): List<Int> {
51 if (words.size - start != state.total.size) fail("wrong vector length")
52 return words.drop(start).map { it.toInt() }
53}
54private fun check(available: List<Int>, detecting: Boolean, traced: Boolean, excluded: Set<Int> = emptySet()): Check {
55 var work = available.toList()
56 val remaining = mutableListOf<Int>()
57 val idle = mutableListOf<Int>()
58 val sequence = mutableListOf<Int>()
59 for (p in state.processes.values) {
60 if (p.id in excluded) work = add(work, p.allocation)
61 else if (detecting && p.allocation.sum() == 0) idle.add(p.id)
62 else remaining.add(p.id)
63 }
64 val prefix = if (detecting) "[detect]" else "[safety]"
65 if (traced) emit("$prefix start work=${vl(work)}" + if (idle.isEmpty()) "" else " idle ${pl(idle)}")
66 while (true) {
67 val chosen = remaining.find { id -> fits(if (detecting) requestOf(id) else state.processes.getValue(id).need(), work) } ?: break
68 work = add(work, state.processes.getValue(chosen).allocation)
69 remaining.remove(chosen)
70 sequence.add(chosen)
71 if (traced) emit("$prefix ${pn(chosen)} can finish, work=${vl(work)}")
72 }
73 if (traced && remaining.isNotEmpty()) emit("$prefix stuck work=${vl(work)} remaining ${pl(remaining)}")
74 return Check(sequence, remaining)
75}
76private fun detect(traced: Boolean) = check(state.available, true, traced).remaining
77private fun rejection(p: Process, v: List<Int>): String? {
78 if (waiting(p.id) != null) return "process ${pn(p.id)} is blocked"
79 if (v.sum() == 0) return "zero vector"
80 if (!fits(v, p.need())) return "request exceeds need"
81 if (mode == "ORDER") {
82 val last = p.allocation.indexOfLast { it > 0 }
83 if ((0..last).any { v[it] > 0 }) return "resource order violation"
84 }
85 return null
86}
87private fun decision(p: Process, v: List<Int>, traced: Boolean): Decision {
88 if (!fits(v, state.available)) return Decision("insufficient resources")
89 if (mode != "AVOID") return Decision(null)
90 val old = p.allocation
91 p.allocation = add(old, v)
92 val result = check(sub(state.available, v), false, traced)
93 p.allocation = old
94 return Decision(if (result.remaining.isEmpty()) null else "unsafe state", result.sequence)
95}
96private fun allocate(p: Process, v: List<Int>) {
97 p.allocation = add(p.allocation, v)
98 state.available = sub(state.available, v)
99}
100private fun record(kind: String, id: Int, v: List<Int>, waited: Int = 0) {
101 state.history.add(Event(command, kind, id, v.toList(), waited))
102}
103private fun retry() {
104 val kept = mutableListOf<Waiting>()
105 for (w in state.queue) {
106 val p = state.processes.getValue(w.id)
107 if (decision(p, w.vector, tracing).reason == null) {
108 allocate(p, w.vector)
109 emit("unblocked ${pn(w.id)} ${vl(w.vector)}")
110 record("unblocked", w.id, w.vector, command - w.since)
111 } else kept.add(w)
112 }
113 state.queue = kept
114}
115private fun terminate(id: Int, kind: String) {
116 val p = state.processes.remove(id)!!
117 state.queue.removeAll { it.id == id }
118 state.available = add(state.available, p.allocation)
119 emit("$kind ${pn(id)} released ${vl(p.allocation)}")
120 record(kind, id, p.allocation)
121}
122private fun compareLists(a: List<Int>, b: List<Int>): Int {
123 for (i in 0 until minOf(a.size, b.size)) {
124 val c = a[i].compareTo(b[i])
125 if (c != 0) return c
126 }
127 return a.size.compareTo(b.size)
128}
129private data class Edge(val destination: Int, val resource: Int)
130private fun graph(resources: Boolean) {
131 val m = state.total.size
132 if (resources && state.total.any { it != 1 }) fail("not single-instance")
133 val edges = state.processes.keys.associateWith { mutableListOf<Edge>() }
134 var any = false
135 if (!resources) {
136 for (p in state.processes.values) {
137 val req = requestOf(p.id)
138 for (q in state.processes.values) {
139 if (p.id == q.id) continue
140 val labels = (0 until m).filter { req[it] > state.available[it] && q.allocation[it] > 0 }
141 if (labels.isNotEmpty()) {
142 edges.getValue(p.id).add(Edge(q.id, -1))
143 emit("${pn(p.id)} -> ${pn(q.id)} on ${labels.joinToString(",") { type(it) }}")
144 any = true
145 }
146 }
147 }
148 if (!any) emit("no wait edges")
149 } else {
150 val holders = IntArray(m) { -1 }
151 for (p in state.processes.values) for (i in 0 until m) if (p.allocation[i] > 0) holders[i] = p.id
152 for (i in 0 until m) if (holders[i] >= 0) emit("assign ${type(i)} -> ${pn(holders[i])}")
153 for (p in state.processes.values) {
154 val req = requestOf(p.id)
155 for (i in 0 until m) if (req[i] > 0) {
156 emit("request ${pn(p.id)} -> ${type(i)}")
157 if (holders[i] >= 0) edges.getValue(p.id).add(Edge(holders[i], i))
158 }
159 }
160 for (p in state.processes.values) {
161 val req = requestOf(p.id)
162 val need = p.need()
163 for (i in 0 until m) if (need[i] > 0 && req[i] == 0) emit("claim ${pn(p.id)} -> ${type(i)}")
164 }
165 }
166 val cycles = mutableListOf<Pair<List<Int>, String>>()
167 for (start in state.processes.keys) {
168 val visited = mutableSetOf(start)
169 fun search(current: Int, key: List<Int>, labels: List<String>) {
170 for (edge in edges.getValue(current)) {
171 val next = edge.destination
172 if (next < start) continue
173 val nextKey = key + (if (resources) listOf(edge.resource, next) else listOf(next))
174 val nextLabels = labels + (if (resources) listOf(type(edge.resource), pn(next)) else listOf(pn(next)))
175 if (next == start) {
176 if (resources || visited.size > 1) cycles.add(nextKey to "cycle: ${nextLabels.joinToString(" -> ")}")
177 } else if (next !in visited) {
178 visited.add(next)
179 search(next, nextKey, nextLabels)
180 visited.remove(next)
181 }
182 }
183 }
184 search(start, listOf(start), listOf(pn(start)))
185 }
186 cycles.sortWith { a, b -> compareLists(a.first, b.first) }
187 cycles.forEach { emit(it.second) }
188 if (cycles.isEmpty()) emit("no cycles")
189}
190private fun table(v: List<Int>) = v.indices.joinToString(" ") { v[it].toString().padStart(state.total[it].toString().length) }
191private fun showState() {
192 val names = state.total.indices.joinToString(" ") { type(it).padStart(state.total[it].toString().length) }
193 val headings = listOf("Allocation", "Max", "Need", "Request")
194 val widths = headings.map { maxOf(it.length, names.length) }
195 fun row(heading: String, fields: List<String>) {
196 emit(heading.padEnd(9) + fields.indices.joinToString("") { " " + fields[it].padEnd(widths[it]) })
197 }
198 row("Process", headings)
199 row("", List(4) { names })
200 for (p in state.processes.values) row(pn(p.id), listOf(table(p.allocation), table(p.maximum), table(p.need()), table(requestOf(p.id))))
201 row("Available", listOf(table(state.available), "", "", ""))
202 row("Total", listOf(table(state.total), "", "", ""))
203 emit("blocked queue: " + if (state.queue.isEmpty()) "empty" else state.queue.joinToString(", ") { "${pn(it.id)} ${vl(it.vector)} since t=${it.since}" })
204}
205private fun stats() {
206 for (i in state.total.indices) {
207 val total = state.total[i]
208 val used = total - state.available[i]
209 val requested = state.queue.sumOf { it.vector[i] }
210 val percent = used * 1000 / total
211 emit("${type(i)}: allocated $used/$total (${percent / 10}.${percent % 10}%), requested $requested")
212 }
213 emit("events: " + kinds.joinToString(", ") { kind -> "$kind ${state.history.count { it.kind == kind }}" })
214 val waits = state.history.filter { it.kind == "unblocked" }
215 if (waits.isEmpty()) emit("waits: none")
216 else {
217 val total = waits.sumOf { it.waited }
218 val average = total * 100 / waits.size
219 emit("waits: ${waits.size} unblocked, total $total, average ${average / 100}.${(average % 100).toString().padStart(2, '0')}")
220 }
221}
222private fun execute(w: List<String>) {
223 val c = w[0]
224 if (c in listOf("MODE", "POLICY", "TRACE", "AUTODETECT")) {
225 when (c) {
226 "MODE" -> mode = w[1]
227 "POLICY" -> policy = w[1]
228 "TRACE" -> tracing = w[1] == "ON"
229 "AUTODETECT" -> autodetect = w[1] == "ON"
230 }
231 emit("${c.lowercase()} ${w[1]}")
232 return
233 }
234 if (c == "RESOURCES") { state = State(w.drop(1).map { it.toInt() }); emit("resources ${vl(state.total)}"); return }
235 if (c in listOf("PROCESS", "CLAIM", "REQUEST", "WHATIF", "RELEASE")) {
236 val id = w[1].toInt()
237 val v = vector(w, 2)
238 val name = pn(id)
239 if (c == "PROCESS") {
240 if (id in state.processes) fail("duplicate process $name")
241 if (!fits(v, state.total)) fail("max exceeds total")
242 state.processes[id] = Process(id, v, command, List(v.size) { 0 })
243 emit("process $name max ${vl(v)}")
244 return
245 }
246 val p = requireProcess(id, c == "CLAIM" || c == "RELEASE")
247 if (c == "CLAIM") {
248 if (!fits(p.allocation, v)) fail("claim below allocation")
249 if (!fits(v, state.total)) fail("max exceeds total")
250 val old = p.maximum
251 p.maximum = v
252 val unsafe = mode == "AVOID" && check(state.available, false, tracing).remaining.isNotEmpty()
253 p.maximum = old
254 if (unsafe) fail("claim makes state unsafe")
255 p.maximum = v
256 emit("claim $name max ${vl(v)}")
257 retry()
258 return
259 }
260 if (c == "RELEASE") {
261 if (v.sum() == 0) fail("zero vector")
262 if (!fits(v, p.allocation)) fail("release exceeds allocation")
263 p.allocation = sub(p.allocation, v)
264 state.available = add(state.available, v)
265 emit("released $name ${vl(v)}")
266 record("released", id, v)
267 retry()
268 return
269 }
270 val reject = rejection(p, v)
271 val prefix = "whatif $name ${vl(v)}"
272 if (reject != null) {
273 if (c == "REQUEST") fail(reject)
274 emit("$prefix: rejected ($reject)")
275 return
276 }
277 val d = decision(p, v, tracing)
278 if (c == "WHATIF") {
279 if (d.reason != null) emit("$prefix: blocked (${d.reason})")
280 else emit("$prefix: granted" + if (mode == "AVOID") ", safe sequence ${pl(d.sequence)}" else "")
281 } else if (d.reason == null) {
282 allocate(p, v)
283 emit("granted $name ${vl(v)}")
284 record("granted", id, v)
285 } else {
286 state.queue.add(Waiting(id, v, command))
287 emit("blocked $name ${vl(v)}: ${d.reason}")
288 record("blocked", id, v)
289 if (autodetect) { val dead = detect(tracing); if (dead.isNotEmpty()) emit("deadlock detected: ${pl(dead)}") }
290 }
291 return
292 }
293 if (c == "ADD" || c == "REMOVE") {
294 val v = vector(w, 1)
295 if (v.sum() == 0) fail("zero vector")
296 if (c == "ADD") {
297 state.total = add(state.total, v)
298 state.available = add(state.available, v)
299 emit("added ${vl(v)} total ${vl(state.total)}")
300 retry()
301 } else {
302 if (!fits(v, state.available)) fail("remove exceeds available")
303 val total = sub(state.total, v)
304 val available = sub(state.available, v)
305 if (total.any { it == 0 }) fail("total must stay positive")
306 if (state.processes.values.any { !fits(it.maximum, total) }) fail("max exceeds total")
307 if (mode == "AVOID" && check(available, false, tracing).remaining.isNotEmpty()) fail("removal makes state unsafe")
308 state.total = total
309 state.available = available
310 emit("removed ${vl(v)} total ${vl(total)}")
311 }
312 return
313 }
314 when (c) {
315 "FINISH", "ABORT" -> {
316 val id = w[1].toInt()
317 requireProcess(id, c == "FINISH")
318 terminate(id, if (c == "FINISH") "finished" else "aborted")
319 retry()
320 }
321 "CANCEL" -> {
322 val id = w[1].toInt()
323 requireProcess(id)
324 val entry = waiting(id) ?: fail("process ${pn(id)} is not blocked")
325 state.queue.remove(entry)
326 emit("cancelled ${pn(id)} ${vl(entry.vector)}")
327 record("cancelled", id, entry.vector)
328 }
329 "MAXGRANT" -> {
330 val id = w[1].toInt()
331 val p = requireProcess(id, true)
332 val result = MutableList(state.total.size) { 0 }
333 for (i in result.indices) {
334 val v = MutableList(result.size) { 0 }
335 for (amount in p.need()[i] downTo 1) {
336 v[i] = amount
337 if (rejection(p, v) == null && decision(p, v, false).reason == null) { result[i] = amount; break }
338 }
339 }
340 emit("maxgrant ${pn(id)}: ${vl(result)}")
341 }
342 "SAFE" -> {
343 if (state.processes.isEmpty()) emit("safe: no processes")
344 else { val r = check(state.available, false, tracing); emit(if (r.remaining.isEmpty()) "safe: ${pl(r.sequence)}" else "unsafe: ${pl(r.remaining)} cannot finish") }
345 }
346 "SEQUENCES" -> {
347 if (state.processes.isEmpty()) { emit("safe sequences: no processes"); return }
348 val limit = w[1].toInt()
349 var count = 0
350 val path = mutableListOf<Int>()
351 val found = mutableListOf<String>()
352 fun search(work: List<Int>) {
353 if (path.size == state.processes.size) { count++; if (found.size < limit) found.add(pl(path)); return }
354 for (p in state.processes.values) if (p.id !in path && fits(p.need(), work)) {
355 path.add(p.id)
356 search(add(work, p.allocation))
357 path.removeAt(path.lastIndex)
358 }
359 }
360 search(state.available)
361 emit("safe sequences: $count")
362 found.forEach { emit(it) }
363 }
364 "DEFICIT" -> {
365 val results = mutableListOf<String>()
366 for (i in state.total.indices) {
367 val limit = state.processes.values.sumOf { it.need()[i] }
368 var result = "-"
369 for (x in 0..limit) {
370 val work = state.available.toMutableList()
371 work[i] += x
372 if (check(work, false, false).remaining.isEmpty()) { result = x.toString(); break }
373 }
374 results.add(result)
375 }
376 emit("deficit: ${results.joinToString(",", "(", ")")}")
377 }
378 "STATE" -> showState()
379 "STATS" -> stats()
380 "DETECT" -> { val dead = detect(tracing); emit(if (dead.isEmpty()) "no deadlock" else "deadlocked: ${pl(dead)}") }
381 "VICTIMS" -> {
382 val dead = detect(false)
383 if (dead.isEmpty()) { emit("no deadlock"); return }
384 val groups = (1 until (1 shl dead.size)).map { mask -> dead.indices.filter { (mask and (1 shl it)) != 0 }.map { dead[it] } }.toMutableList()
385 groups.sortWith { a, b ->
386 var result = a.size.compareTo(b.size)
387 if (result == 0) result = a.sumOf { state.processes.getValue(it).allocation.sum() }.compareTo(b.sumOf { state.processes.getValue(it).allocation.sum() })
388 if (result == 0) result = compareLists(a, b)
389 result
390 }
391 for (group in groups) if (check(state.available, true, false, group.toSet()).remaining.isEmpty()) { emit("victims: ${pl(group)}"); break }
392 }
393 "RECOVER" -> {
394 var dead = detect(tracing)
395 var count = 0
396 if (w.size > 1) { for (id in dead) { terminate(id, "aborted"); count++ }; if (count > 0) retry() }
397 else while (dead.isNotEmpty()) {
398 var victim = dead.first()
399 for (id in dead) {
400 val a = state.processes.getValue(id)
401 val b = state.processes.getValue(victim)
402 val better = when (policy) {
403 "ALLOC" -> a.allocation.sum() < b.allocation.sum() || (a.allocation.sum() == b.allocation.sum() && id > victim)
404 "NEED" -> a.need().sum() > b.need().sum() || (a.need().sum() == b.need().sum() && id > victim)
405 else -> a.created > b.created
406 }
407 if (better) victim = id
408 }
409 terminate(victim, "aborted")
410 retry()
411 count++
412 dead = detect(tracing)
413 }
414 emit(if (count == 0) "no deadlock" else "recovered: $count aborted")
415 }
416 "WAITGRAPH" -> graph(false)
417 "GRAPH" -> graph(true)
418 "HISTORY" -> {
419 var any = false
420 for (e in state.history) {
421 if (w.size > 1 && !(if (w[1].first().isDigit()) e.id == w[1].toInt() else e.kind == w[1])) continue
422 emit("t=${e.command} ${e.kind} ${pn(e.id)} ${vl(e.vector)}" + if (e.kind == "unblocked") " waited ${e.waited}" else "")
423 any = true
424 }
425 if (!any) emit("no events")
426 }
427 "SNAPSHOT" -> { snapshots[w[1]] = state.duplicate(); emit("saved ${w[1]}") }
428 "RESTORE" -> { state = snapshots[w[1]]?.duplicate() ?: fail("unknown snapshot ${w[1]}"); emit("restored ${w[1]}") }
429 }
430}
431fun main() {
432 val input = System.`in`.bufferedReader()
433 val n = input.readLine().toInt()
434 for (i in 1..n) {
435 command = i
436 try { execute(input.readLine().split(' ')) }
437 catch (error: IllegalArgumentException) { emit("error: ${error.message}") }
438 }
439 print(output)
440}
