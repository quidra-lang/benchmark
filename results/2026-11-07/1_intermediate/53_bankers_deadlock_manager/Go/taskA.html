59func vector(a []string) Vector {

61 for i, s := range a {

67func copyv(v Vector) Vector {

71func combine(a, b Vector, sign int) Vector {
72 c := copyv(a)
73 for i := range c {
74 c[i] += sign * b[i]
75 }
76 return c

79func le(a, b Vector) bool {
80 for i := range a {
81 if a[i] > b[i] {
82 return false
83 }
84 }
85 return true

88func sum(a Vector) int {
89 n := 0
90 for _, x := range a {
91 n += x

116func (s State) ids() []int {
117 ids := make([]int, 0, len(s.Processes))
118 for id := range s.Processes {
119 ids = append(ids, id)
120 }

125func (s State) clone() State {
126 c := State{copyv(s.Total), copyv(s.Available), make(map[int]*Process), append([]Waiting{}, s.Queue...), append([]Event{}, s.History...)}
127 for id, p := range s.Processes {
128 c.Processes[id] = &Process{copyv(p.Max), copyv(p.Allocation), p.Created}
129 }
130 for i := range c.Queue {
131 c.Queue[i].Value = copyv(c.Queue[i].Value)
132 }
133 for i := range c.History {
134 c.History[i].Value = copyv(c.History[i].Value)
135 }
136 return c
137}

140 for _, q := range s.Queue {

157func (s State) check(detect, trace bool) (bool, []int, []int) {
158 work := copyv(s.Available)
159 remaining, idle, sequence := []int{}, []int{}, []int{}
160 for _, id := range s.ids() {
161 if detect && sum(s.Processes[id].Allocation) == 0 {
162 idle = append(idle, id)
163 } else {
164 remaining = append(remaining, id)
165 }
166 }
167 tag := "[safety]"
168 if detect {
169 tag = "[detect]"
170 }
171 if trace {
172 line := tag + " start work=" + vf(work)
173 if len(idle) > 0 {
174 line += " idle " + pf(idle)
175 }
176 emit(line)
177 }
178 for {
179 chosen := -1
180 for i, id := range remaining {
181 p := s.Processes[id]
182 need := combine(p.Max, p.Allocation, -1)
183 if detect {
184 need = s.request(id)
185 }
186 if le(need, work) {
187 chosen = i
188 break
189 }
190 }
191 if chosen < 0 {
192 break
193 }
194 id := remaining[chosen]
195 work = combine(work, s.Processes[id].Allocation, 1)
196 sequence = append(sequence, id)
197 remaining = append(remaining[:chosen], remaining[chosen+1:]...)
198 if trace {
199 emit(tag + " " + pname(id) + " can finish, work=" + vf(work))
200 }
201 }
202 if trace && len(remaining) > 0 {
203 emit(tag + " stuck work=" + vf(work) + " remaining " + pf(remaining))
204 }
205 return len(remaining) == 0, sequence, remaining
206}

208func (m *Manager) decision(id int, v Vector, reject, trace bool) (string, string, []int) {
209 p := m.Processes[id]
210 if reject {
211 if m.blocked(id) {
212 return "rejected", "process " + pname(id) + " is blocked", nil
213 }
214 if sum(v) == 0 {
215 return "rejected", "zero vector", nil
216 }
217 if !le(v, combine(p.Max, p.Allocation, -1)) {
218 return "rejected", "request exceeds need", nil
219 }
220 if m.Mode == "ORDER" {
221 last := -1
222 for j, x := range p.Allocation {
223 if x > 0 {
224 last = j
225 }
226 }
227 for j, x := range v {
228 if x > 0 && j <= last {
229 return "rejected", "resource order violation", nil
230 }
231 }
232 }
233 }
234 if !le(v, m.Available) {
235 return "blocked", "insufficient resources", nil
236 }
237 if m.Mode == "AVOID" {
238 oldA, oldP := m.Available, p.Allocation
239 m.Available = combine(oldA, v, -1)
240 p.Allocation = combine(oldP, v, 1)
241 ok, seq, _ := m.State.check(false, trace)
242 m.Available, p.Allocation = oldA, oldP
243 if !ok {
244 return "blocked", "unsafe state", nil
245 }
246 return "granted", "", seq
247 }
248 return "granted", "", nil
249}

251func (m *Manager) event(kind string, id int, v Vector, wait int) {
252 m.History = append(m.History, Event{m.Time, kind, id, copyv(v), wait})
253}

255func (m *Manager) grant(id int, v Vector) {
256 p := m.Processes[id]
257 p.Allocation = combine(p.Allocation, v, 1)
258 m.Available = combine(m.Available, v, -1)
259}

261func (m *Manager) retry() {
262 queue := append([]Waiting{}, m.Queue...)
263 for _, q := range queue {
264 result, _, _ := m.decision(q.ID, q.Value, false, m.Trace)
265 if result == "granted" {
266 m.grant(q.ID, q.Value)
267 m.removeWait(q.ID)
268 emit("unblocked " + pname(q.ID) + " " + vf(q.Value))
269 m.event("unblocked", q.ID, q.Value, m.Time-q.Since)
270 }
271 }
272}

274func (m *Manager) removeWait(id int) {
275 for i, q := range m.Queue {
276 if q.ID == id {
277 m.Queue = append(m.Queue[:i], m.Queue[i+1:]...)
278 return
279 }
280 }

283func (m *Manager) terminate(id int, kind string, retry bool) {
284 v := copyv(m.Processes[id].Allocation)
285 m.Available = combine(m.Available, v, 1)
286 delete(m.Processes, id)
287 m.removeWait(id)
288 emit(kind + " " + pname(id) + " released " + vf(v))
289 m.event(kind, id, v, 0)
290 if retry {
291 m.retry()
292 }
293}

295func (s State) victims() []int {
296 _, _, dead := s.check(true, false)
297 type Candidate struct {
298 IDs []int
299 Cost int
300 }
301 choices := []Candidate{}
302 for mask := 1; mask < (1 << len(dead)); mask++ {
303 c := Candidate{}
304 for i, id := range dead {
305 if mask&(1<<i) != 0 {
306 c.IDs = append(c.IDs, id)
307 c.Cost += sum(s.Processes[id].Allocation)
308 }
309 }
310 choices = append(choices, c)
311 }
312 sort.Slice(choices, func(i, j int) bool {
313 a, b := choices[i], choices[j]
314 if len(a.IDs) != len(b.IDs) {
315 return len(a.IDs) < len(b.IDs)
316 }
317 if a.Cost != b.Cost {
318 return a.Cost < b.Cost
319 }
320 for k := range a.IDs {
321 if a.IDs[k] != b.IDs[k] {
322 return a.IDs[k] < b.IDs[k]
323 }
324 }
325 return false
326 })
327 for _, c := range choices {
328 trial := s.clone()
329 for _, id := range c.IDs {
330 trial.Available = combine(trial.Available, trial.Processes[id].Allocation, 1)
331 delete(trial.Processes, id)
332 }
333 ok, _, _ := trial.check(true, false)
334 if ok {
335 return c.IDs
336 }
337 }
338 return nil
339}

341func (m *Manager) graphs(resource bool) {
342 ids := m.ids()
343 n := len(m.Total)
344 type Edge struct {
345 To, Type int
346 }
347 edges := map[int][]Edge{}
348 if resource {
349 for _, x := range m.Total {
350 if x != 1 {
351 emit("error: not single-instance")
352 return
353 }
354 }
355 for j := 0; j < n; j++ {
356 for _, id := range ids {
357 if m.Processes[id].Allocation[j] > 0 {
358 emit(fmt.Sprintf("assign %c -> %s", 'A'+j, pname(id)))
359 }
360 }
361 }
362 for _, id := range ids {
363 r := m.request(id)
364 for j, x := range r {
365 if x > 0 {
366 emit(fmt.Sprintf("request %s -> %c", pname(id), 'A'+j))
367 for _, to := range ids {
368 if m.Processes[to].Allocation[j] > 0 {
369 edges[id] = append(edges[id], Edge{to, j})
370 }
371 }
372 }
373 }
374 }
375 for _, id := range ids {
376 p := m.Processes[id]
377 r := m.request(id)
378 for j, x := range p.Max {
379 if x > p.Allocation[j] && r[j] == 0 {
380 emit(fmt.Sprintf("claim %s -> %c", pname(id), 'A'+j))
381 }
382 }
383 }
384 } else {
385 count := 0
386 for _, id := range ids {
387 r := m.request(id)
388 for _, to := range ids {
389 if id == to {
390 continue
391 }
392 labels := []string{}
393 for j, x := range r {
394 if x > m.Available[j] && m.Processes[to].Allocation[j] > 0 {
395 labels = append(labels, string(rune('A'+j)))
396 }
397 }
398 if len(labels) > 0 {
399 emit(pname(id) + " -> " + pname(to) + " on " + strings.Join(labels, ","))
400 edges[id] = append(edges[id], Edge{to, 0})
401 count++
402 }
403 }
404 }
405 if count == 0 {
406 emit("no wait edges")
407 }
408 }
409 cycles := [][]int{}
410 for _, start := range ids {
411 var visit func(int, []int, map[int]bool)
412 visit = func(at int, path []int, seen map[int]bool) {
413 for _, e := range edges[at] {
414 next := append([]int{}, path...)
415 if resource {
416 next = append(next, e.Type)
417 }
418 next = append(next, e.To)
419 if e.To == start {
420 if resource || len(seen) >= 2 {
421 cycles = append(cycles, next)
422 }
423 continue
424 }
425 if e.To < start || seen[e.To] {
426 continue
427 }
428 seen[e.To] = true
429 visit(e.To, next, seen)
430 delete(seen, e.To)
431 }
432 }
433 visit(start, []int{start}, map[int]bool{start: true})
434 }
435 sort.Slice(cycles, func(i, j int) bool {
436 a, b := cycles[i], cycles[j]
437 for k := 0; k < len(a) && k < len(b); k++ {
438 if a[k] != b[k] {
439 return a[k] < b[k]
440 }
441 }
442 return len(a) < len(b)
443 })
444 for _, cycle := range cycles {
445 parts := []string{}
446 for i, x := range cycle {
447 if resource && i%2 == 1 {
448 parts = append(parts, string(rune('A'+x)))
449 } else {
450 parts = append(parts, pname(x))
451 }
452 }
453 emit("cycle: " + strings.Join(parts, " -> "))
454 }
455 if len(cycles) == 0 {
456 emit("no cycles")
457 }
458}

460func (m *Manager) table() {
461 widths := make([]int, len(m.Total))
462 names := make([]string, len(widths))
463 for i, x := range m.Total {
464 widths[i] = len(strconv.Itoa(x))
465 names[i] = fmt.Sprintf("%*s", widths[i], string(rune('A'+i)))
466 }
467 name := strings.Join(names, " ")
468 cw := []int{max(10, len(name)), max(3, len(name)), max(4, len(name)), max(7, len(name))}
469 row := func(heading string, cols []string) {
470 line := fmt.Sprintf("%-9s", heading)
471 for i := 0; i < 4; i++ {
472 value := ""
473 if i < len(cols) {
474 value = cols[i]
475 }
476 line += fmt.Sprintf(" %-*s", cw[i], value)
477 }
478 emit(strings.TrimRight(line, " "))
479 }
480 form := func(v Vector) string {
481 a := make([]string, len(v))
482 for i, x := range v {
483 a[i] = fmt.Sprintf("%*d", widths[i], x)
484 }
485 return strings.Join(a, " ")
486 }
487 row("Process", []string{"Allocation", "Max", "Need", "Request"})
488 row("", []string{name, name, name, name})
489 for _, id := range m.ids() {
490 p := m.Processes[id]
491 row(pname(id), []string{form(p.Allocation), form(p.Max), form(combine(p.Max, p.Allocation, -1)), form(m.request(id))})
492 }
493 row("Available", []string{form(m.Available)})
494 row("Total", []string{form(m.Total)})
495 if len(m.Queue) == 0 {
496 emit("blocked queue: empty")
497 } else {
498 a := []string{}
499 for _, q := range m.Queue {
500 a = append(a, fmt.Sprintf("%s %s since t=%d", pname(q.ID), vf(q.Value), q.Since))
501 }
502 emit("blocked queue: " + strings.Join(a, ", "))
503 }
504}

506func (m *Manager) execute(a []string) {
507 cmd := a[0]
508 args := a[1:]
509 err := ""
510 id := 0
511 var v Vector
512 hasID := strings.Contains(" PROCESS CLAIM REQUEST WHATIF RELEASE FINISH ABORT CANCEL MAXGRANT ", " "+cmd+" ")
513 hasVector := strings.Contains(" PROCESS CLAIM REQUEST WHATIF RELEASE ADD REMOVE ", " "+cmd+" ")
514 if hasID {
515 id = number(args[0])
516 }
517 if hasVector {
518 start := 0
519 if hasID {
520 start = 1
521 }
522 v = vector(args[start:])
523 if len(v) != len(m.Total) {
524 err = "wrong vector length"
525 }
526 }
527 if err == "" && hasID {
528 _, exists := m.Processes[id]
529 if cmd == "PROCESS" {
530 if exists {
531 err = "duplicate process " + pname(id)
532 }
533 } else if !exists {
534 err = "unknown process " + pname(id)
535 }
536 }
537 if err != "" {
538 emit("error: " + err)
539 return
540 }
541 switch cmd {
542 case "RESOURCES":
543 m.State = State{vector(args), vector(args), make(map[int]*Process), nil, nil}
544 emit("resources " + vf(m.Total))
545 case "MODE", "POLICY", "TRACE", "AUTODETECT":
546 x := args[0]
547 switch cmd {
548 case "MODE":
549 m.Mode = x
550 case "POLICY":
551 m.Policy = x
552 case "TRACE":
553 m.Trace = x == "ON"
554 case "AUTODETECT":
555 m.Auto = x == "ON"
556 }
557 emit(strings.ToLower(cmd) + " " + x)
558 case "PROCESS":
559 if !le(v, m.Total) {
560 err = "max exceeds total"
561 } else {
562 m.Processes[id] = &Process{v, make(Vector, len(v)), m.Time}
563 emit("process " + pname(id) + " max " + vf(v))
564 }
565 case "CLAIM":
566 p := m.Processes[id]
567 if m.blocked(id) {
568 err = "process " + pname(id) + " is blocked"
569 } else if !le(p.Allocation, v) {
570 err = "claim below allocation"
571 } else if !le(v, m.Total) {
572 err = "max exceeds total"
573 } else {
574 old := p.Max
575 p.Max = v
576 if m.Mode == "AVOID" {
577 ok, _, _ := m.State.check(false, m.Trace)
578 if !ok {
579 p.Max = old
580 err = "claim makes state unsafe"
581 }
582 }
583 if err == "" {
584 emit("claim " + pname(id) + " max " + vf(v))
585 m.retry()
586 }
587 }
588 case "ADD", "REMOVE":
589 if sum(v) == 0 {
590 err = "zero vector"
591 } else if cmd == "ADD" {
592 m.Total = combine(m.Total, v, 1)
593 m.Available = combine(m.Available, v, 1)
594 emit("added " + vf(v) + " total " + vf(m.Total))
595 m.retry()
596 } else {
597 total := combine(m.Total, v, -1)
598 if !le(v, m.Available) {
599 err = "remove exceeds available"
600 } else {
601 for _, x := range total {
602 if x == 0 {
603 err = "total must stay positive"
604 break
605 }
606 }
607 if err == "" {
608 for _, p := range m.Processes {
609 if !le(p.Max, total) {
610 err = "max exceeds total"
611 break
612 }
613 }
614 }
615 if err == "" {
616 old := m.Available
617 m.Available = combine(old, v, -1)
618 if m.Mode == "AVOID" {
619 ok, _, _ := m.State.check(false, m.Trace)
620 if !ok {
621 err = "removal makes state unsafe"
622 }
623 }
624 if err != "" {
625 m.Available = old
626 } else {
627 m.Total = total
628 emit("removed " + vf(v) + " total " + vf(total))
629 }
630 }
631 }
632 }
633 case "REQUEST", "WHATIF":
634 result, reason, seq := m.decision(id, v, true, m.Trace)
635 if cmd == "WHATIF" {
636 line := "whatif " + pname(id) + " " + vf(v) + ": " + result
637 if result != "granted" {
638 line += " (" + reason + ")"
639 } else if m.Mode == "AVOID" {
640 line += ", safe sequence " + pf(seq)
641 }
642 emit(line)
643 } else if result == "rejected" {
644 err = reason
645 } else {
646 if result == "granted" {
647 m.grant(id, v)
648 emit("granted " + pname(id) + " " + vf(v))
649 } else {
650 m.Queue = append(m.Queue, Waiting{id, v, m.Time})
651 emit("blocked " + pname(id) + " " + vf(v) + ": " + reason)
652 }
653 m.event(result, id, v, 0)
654 if result == "blocked" && m.Auto {
655 _, _, dead := m.State.check(true, m.Trace)
656 if len(dead) > 0 {
657 emit("deadlock detected: " + pf(dead))
658 }
659 }
660 }
661 case "RELEASE":
662 p := m.Processes[id]
663 if m.blocked(id) {
664 err = "process " + pname(id) + " is blocked"
665 } else if sum(v) == 0 {
666 err = "zero vector"
667 } else if !le(v, p.Allocation) {
668 err = "release exceeds allocation"
669 } else {
670 p.Allocation = combine(p.Allocation, v, -1)
671 m.Available = combine(m.Available, v, 1)
672 emit("released " + pname(id) + " " + vf(v))
673 m.event("released", id, v, 0)
674 m.retry()
675 }
676 case "FINISH", "ABORT":
677 if cmd == "FINISH" && m.blocked(id) {
678 err = "process " + pname(id) + " is blocked"
679 } else {
680 kind := "finished"
681 if cmd == "ABORT" {
682 kind = "aborted"
683 }
684 m.terminate(id, kind, true)
685 }
686 case "CANCEL":
687 if !m.blocked(id) {
688 err = "process " + pname(id) + " is not blocked"
689 } else {
690 value := m.request(id)
691 m.removeWait(id)
692 emit("cancelled " + pname(id) + " " + vf(value))
693 m.event("cancelled", id, value, 0)
694 }
695 case "MAXGRANT":
696 if m.blocked(id) {
697 err = "process " + pname(id) + " is blocked"
698 } else {
699 need := combine(m.Processes[id].Max, m.Processes[id].Allocation, -1)
700 values := make(Vector, len(need))
701 for j, limit := range need {
702 for x := limit; x > 0; x-- {
703 r := make(Vector, len(need))
704 r[j] = x
705 result, _, _ := m.decision(id, r, true, false)
706 if result == "granted" {
707 values[j] = x
708 break
709 }
710 }
711 }
712 emit("maxgrant " + pname(id) + ": " + vf(values))
713 }
714 case "SAFE":
715 if len(m.Processes) == 0 {
716 emit("safe: no processes")
717 } else {
718 ok, seq, left := m.State.check(false, m.Trace)
719 if ok {
720 emit("safe: " + pf(seq))
721 } else {
722 emit("unsafe: " + pf(left) + " cannot finish")
723 }
724 }
725 case "SEQUENCES":
726 if len(m.Processes) == 0 {
727 emit("safe sequences: no processes")
728 break
729 }
730 count, k := 0, number(args[0])
731 samples := []string{}
732 ids := m.ids()
733 var search func(Vector, []int, map[int]bool)
734 search = func(work Vector, path []int, seen map[int]bool) {
735 if len(path) == len(ids) {
736 count++
737 if len(samples) < k {
738 samples = append(samples, pf(path))
739 }
740 return
741 }
742 for _, id := range ids {
743 p := m.Processes[id]
744 if !seen[id] && le(combine(p.Max, p.Allocation, -1), work) {
745 seen[id] = true
746 search(combine(work, p.Allocation, 1), append(path, id), seen)
747 delete(seen, id)
748 }
749 }
750 }
751 search(m.Available, nil, map[int]bool{})
752 emit(fmt.Sprintf("safe sequences: %d", count))
753 for _, line := range samples {
754 emit(line)
755 }
756 case "DEFICIT":
757 values := make([]string, len(m.Total))
758 for j := range values {
759 limit := 0
760 for _, p := range m.Processes {
761 limit += p.Max[j] - p.Allocation[j]
762 }
763 values[j] = "-"
764 for x := 0; x <= limit; x++ {
765 s := m.State
766 s.Available = copyv(m.Available)
767 s.Available[j] += x
768 ok, _, _ := s.check(false, false)
769 if ok {
770 values[j] = strconv.Itoa(x)
771 break
772 }
773 }
774 }
775 emit("deficit: (" + strings.Join(values, ",") + ")")
776 case "STATE":
777 m.table()
778 case "STATS":
779 for j, total := range m.Total {
780 used := total - m.Available[j]
781 requested := 0
782 for _, q := range m.Queue {
783 requested += q.Value[j]
784 }
785 pct := used * 1000 / total
786 emit(fmt.Sprintf("%c: allocated %d/%d (%d.%d%%), requested %d", 'A'+j, used, total, pct/10, pct%10, requested))
787 }
788 kinds := []string{"granted", "blocked", "unblocked", "released", "finished", "aborted", "cancelled"}
789 counts := map[string]int{}
790 waits := 0
791 for _, e := range m.History {
792 counts[e.Kind]++
793 if e.Kind == "unblocked" {
794 waits += e.Wait
795 }
796 }
797 parts := []string{}
798 for _, kind := range kinds {
799 parts = append(parts, fmt.Sprintf("%s %d", kind, counts[kind]))
800 }
801 emit("events: " + strings.Join(parts, ", "))
802 if counts["unblocked"] == 0 {
803 emit("waits: none")
804 } else {
805 avg := waits * 100 / counts["unblocked"]
806 emit(fmt.Sprintf("waits: %d unblocked, total %d, average %d.%02d", counts["unblocked"], waits, avg/100, avg%100))
807 }
808 case "DETECT":
809 _, _, dead := m.State.check(true, m.Trace)
810 if len(dead) == 0 {
811 emit("no deadlock")
812 } else {
813 emit("deadlocked: " + pf(dead))
814 }
815 case "VICTIMS":
816 victims := m.State.victims()
817 if len(victims) == 0 {
818 emit("no deadlock")
819 } else {
820 emit("victims: " + pf(victims))
821 }
822 case "RECOVER":
823 count := 0
824 for {
825 _, _, dead := m.State.check(true, m.Trace)
826 if len(dead) == 0 {
827 break
828 }
829 if len(args) > 0 {
830 for _, id := range dead {
831 m.terminate(id, "aborted", false)
832 count++
833 }
834 m.retry()
835 break
836 }
837 chosen := dead[0]
838 for _, id := range dead[1:] {
839 p, q := m.Processes[id], m.Processes[chosen]
840 better := false
841 switch m.Policy {
842 case "ALLOC":
843 a, b := sum(p.Allocation), sum(q.Allocation)
844 better = a < b || a == b && id > chosen
845 case "NEED":
846 a, b := sum(combine(p.Max, p.Allocation, -1)), sum(combine(q.Max, q.Allocation, -1))
847 better = a > b || a == b && id > chosen
848 case "YOUNGEST":
849 better = p.Created > q.Created
850 }
851 if better {
852 chosen = id
853 }
854 }
855 m.terminate(chosen, "aborted", true)
856 count++
857 }
858 if count == 0 {
859 emit("no deadlock")
860 } else {
861 emit(fmt.Sprintf("recovered: %d aborted", count))
862 }
863 case "WAITGRAPH":
864 m.graphs(false)
865 case "GRAPH":
866 m.graphs(true)
867 case "HISTORY":
868 count := 0
869 for _, e := range m.History {
870 if len(args) > 0 && args[0] != e.Kind && args[0] != strconv.Itoa(e.ID) {
871 continue
872 }
873 line := fmt.Sprintf("t=%d %s %s %s", e.Time, e.Kind, pname(e.ID), vf(e.Value))
874 if e.Kind == "unblocked" {
875 line += fmt.Sprintf(" waited %d", e.Wait)
876 }
877 emit(line)
878 count++
879 }
880 if count == 0 {
881 emit("no events")
882 }
883 case "SNAPSHOT":
884 m.Snapshots[args[0]] = m.State.clone()
885 emit("saved " + args[0])
886 case "RESTORE":
887 s, ok := m.Snapshots[args[0]]
888 if !ok {
889 err = "unknown snapshot " + args[0]
890 } else {
891 m.State = s.clone()
892 emit("restored " + args[0])
893 }
894 }
895 if err != "" {
896 emit("error: " + err)
897 }
898}
