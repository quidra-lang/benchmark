125Vec parseVector(const vector<string> &words, size_t start) {
126 if (words.size() - start != state.total.size())
127 fail("wrong vector length");
128 Vec v;
129 for (size_t i = start; i < words.size(); ++i)
130 v.push_back(stoi(words[i]));

133Check check(Vec work, bool detecting, bool traced, const set<int> &excluded = {}) {
134 Vec remaining, idle, sequence;
135 for (const auto &item : state.processes) {
136 const Process &p = item.second;
137 if (excluded.count(p.id))
138 work = add(work, p.allocation);
139 else if (detecting && sum(p.allocation) == 0)
140 idle.push_back(p.id);
141 else
142 remaining.push_back(p.id);
143 }
144 string prefix = detecting ? "[detect]" : "[safety]";
145 if (traced)
146 emit(prefix + " start work=" + vl(work) + (idle.empty() ? "" : " idle " + pl(idle)));
147 while (true) {
148 int chosen = -1;
149 for (int id : remaining)
150 if (fits(detecting ? requestOf(id) : need(state.processes.at(id)), work)) {
151 chosen = id;
152 break;
153 }
154 if (chosen < 0)
155 break;
156 work = add(work, state.processes.at(chosen).allocation);
157 remaining.erase(find(remaining.begin(), remaining.end(), chosen));
158 sequence.push_back(chosen);
159 if (traced)
160 emit(prefix + " " + pn(chosen) + " can finish, work=" + vl(work));
161 }
162 if (traced && !remaining.empty())
163 emit(prefix + " stuck work=" + vl(work) + " remaining " + pl(remaining));
164 return {sequence, remaining};
165}
166Vec detect(bool traced) {
167 return check(state.available, true, traced).remaining;
168}
169string rejection(const Process &p, const Vec &v) {
170 if (waiting(p.id))
171 return "process " + pn(p.id) + " is blocked";
172 if (sum(v) == 0)
173 return "zero vector";
174 if (!fits(v, need(p)))
175 return "request exceeds need";
176 if (mode == "ORDER") {
177 int last = -1;
178 for (size_t i = 0; i < v.size(); ++i)
179 if (p.allocation[i] > 0)
180 last = int(i);
181 for (int i = 0; i <= last; ++i)
182 if (v[i] > 0)
183 return "resource order violation";
184 }
185 return "";
186}
187Decision decision(Process &p, const Vec &v, bool traced) {
188 if (!fits(v, state.available))
189 return {"insufficient resources", {}};
190 if (mode != "AVOID")
191 return {"", {}};
192 Vec old = p.allocation;
193 p.allocation = add(old, v);
194 Check result = check(sub(state.available, v), false, traced);
195 p.allocation = old;
196 return {result.remaining.empty() ? "" : "unsafe state", result.sequence};
197}
198void allocate(Process &p, const Vec &v) {
199 p.allocation = add(p.allocation, v);
200 state.available = sub(state.available, v);
201}
202void record(const string &kind, int id, const Vec &v, int waited = 0) {
203 state.history.push_back({commandNumber, kind, id, v, waited});
204}
205void retry() {
206 vector<Waiting> kept;
207 for (const Waiting &w : state.queue) {
208 Process &p = state.processes.at(w.id);
209 if (decision(p, w.value, tracing).reason.empty()) {
210 allocate(p, w.value);
211 emit("unblocked " + pn(w.id) + " " + vl(w.value));
212 record("unblocked", w.id, w.value, commandNumber - w.since);
213 } else
214 kept.push_back(w);
215 }
216 state.queue = kept;
217}
218void terminate(int id, const string &kind) {
219 Vec released = state.processes.at(id).allocation;
220 state.queue.erase(remove_if(state.queue.begin(), state.queue.end(),
221 [id](const Waiting &w) { return w.id == id; }),
222 state.queue.end());
223 state.available = add(state.available, released);
224 state.processes.erase(id);
225 emit(kind + " " + pn(id) + " released " + vl(released));
226 record(kind, id, released);
227}
228void graph(bool resources) {
229 int m = int(state.total.size());
230 if (resources && any_of(state.total.begin(), state.total.end(), [](int x) { return x != 1; }))
231 fail("not single-instance");
232 map<int, vector<pair<int, int>>> edges;
233 for (const auto &item : state.processes)
234 edges[item.first] = {};
235 bool any = false;
236 if (!resources) {
237 for (const auto &item : state.processes) {
238 int id = item.first;
239 Vec req = requestOf(id);
240 for (const auto &other : state.processes) {
241 int next = other.first;
242 if (id == next)
243 continue;
244 vector<string> labels;
245 for (int i = 0; i < m; ++i)
246 if (req[i] > state.available[i] && other.second.allocation[i] > 0)
247 labels.push_back(type(i));
248 if (!labels.empty()) {
249 edges[id].push_back({next, -1});
250 any = true;
251 emit(pn(id) + " -> " + pn(next) + " on " + join(labels, ","));
252 }
253 }
254 }
255 if (!any)
256 emit("no wait edges");
257 } else {
258 Vec holders(m, -1);
259 for (const auto &item : state.processes)
260 for (int i = 0; i < m; ++i)
261 if (item.second.allocation[i] > 0)
262 holders[i] = item.first;
263 for (int i = 0; i < m; ++i)
264 if (holders[i] >= 0)
265 emit("assign " + type(i) + " -> " + pn(holders[i]));
266 for (const auto &item : state.processes) {
267 int id = item.first;
268 Vec req = requestOf(id);
269 for (int i = 0; i < m; ++i)
270 if (req[i] > 0) {
271 emit("request " + pn(id) + " -> " + type(i));
272 if (holders[i] >= 0)
273 edges[id].push_back({holders[i], i});
274 }
275 }
276 for (const auto &item : state.processes) {
277 int id = item.first;
278 Vec req = requestOf(id), n = need(item.second);
279 for (int i = 0; i < m; ++i)
280 if (n[i] > 0 && req[i] == 0)
281 emit("claim " + pn(id) + " -> " + type(i));
282 }
283 }
284 map<Vec, string> cycles;
285 for (const auto &item : state.processes) {
286 int start = item.first;
287 set<int> visited{start};
288 function<void(int, Vec, vector<string>)> search = [&](int current, Vec key,
289 vector<string> labels) {
290 for (const pair<int, int> &edge : edges[current]) {
291 int next = edge.first;
292 if (next < start)
293 continue;
294 Vec nextKey = key;
295 vector<string> nextLabels = labels;
296 if (resources) {
297 nextKey.push_back(edge.second);
298 nextLabels.push_back(type(edge.second));
299 }
300 nextKey.push_back(next);
301 nextLabels.push_back(pn(next));
302 if (next == start) {
303 if (resources || visited.size() > 1)
304 cycles[nextKey] = "cycle: " + join(nextLabels, " -> ");
305 } else if (!visited.count(next)) {
306 visited.insert(next);
307 search(next, nextKey, nextLabels);
308 visited.erase(next);
309 }
310 }
311 };
312 search(start, {start}, {pn(start)});
313 }
314 for (const auto &cycle : cycles)
315 emit(cycle.second);
316 if (cycles.empty())
317 emit("no cycles");
318}

323 text << ' ';

328void showState() {
329 ostringstream names;
330 for (size_t i = 0; i < state.total.size(); ++i) {
331 if (i)
332 names << ' ';
333 names << setw(int(to_string(state.total[i]).size())) << type(int(i));
334 }
335 string name = names.str();
336 vector<string> headings = {"Allocation", "Max", "Need", "Request"};
337 Vec widths;
338 for (const string &h : headings)
339 widths.push_back(int(max(h.size(), name.size())));
340 auto row = [&](const string &heading, const vector<string> &fields) {
341 ostringstream line;
342 line << left << setw(9) << heading;
343 for (int i = 0; i < 4; ++i)
344 line << " " << setw(widths[i]) << fields[i];
345 emit(line.str());
346 };
347 row("Process", headings);
348 row("", {name, name, name, name});
349 for (const auto &item : state.processes) {
350 const Process &p = item.second;
351 row(pn(p.id),
352 {table(p.allocation), table(p.maximum), table(need(p)), table(requestOf(p.id))});
353 }
354 row("Available", {table(state.available), "", "", ""});
355 row("Total", {table(state.total), "", "", ""});
356 vector<string> queue;
357 for (const Waiting &w : state.queue)
358 queue.push_back(pn(w.id) + " " + vl(w.value) + " since t=" + to_string(w.since));
359 emit("blocked queue: " + (queue.empty() ? "empty" : join(queue, ", ")));
360}
361void stats() {
362 for (size_t i = 0; i < state.total.size(); ++i) {
363 int total = state.total[i], used = total - state.available[i], requested = 0;
364 for (const Waiting &w : state.queue)
365 requested += w.value[i];
366 int percent = used * 1000 / total;
367 emit(type(int(i)) + ": allocated " + to_string(used) + "/" + to_string(total) + " (" +
368 to_string(percent / 10) + "." + to_string(percent % 10) + "%), requested " +
369 to_string(requested));
370 }
371 vector<string> counts;
372 for (const string &kind : kinds) {
373 int n = 0;
374 for (const Event &e : state.history)
375 if (e.kind == kind)
376 ++n;
377 counts.push_back(kind + " " + to_string(n));
378 }
379 emit("events: " + join(counts, ", "));
380 int waits = 0, total = 0;
381 for (const Event &e : state.history)
382 if (e.kind == "unblocked") {
383 ++waits;
384 total += e.waited;
385 }
386 if (waits == 0)
387 emit("waits: none");
388 else {
389 int average = total * 100 / waits;
390 ostringstream line;
391 line << "waits: " << waits << " unblocked, total " << total << ", average " << average / 100
392 << '.' << setfill('0') << setw(2) << average % 100;
393 emit(line.str());
394 }
395}
396void execute(const vector<string> &w) {
397 string c = w[0];
398 if (c == "MODE" || c == "POLICY" || c == "TRACE" || c == "AUTODETECT") {
399 if (c == "MODE")
400 mode = w[1];
401 if (c == "POLICY")
402 policy = w[1];
403 if (c == "TRACE")
404 tracing = w[1] == "ON";
405 if (c == "AUTODETECT")
406 autodetect = w[1] == "ON";
407 transform(c.begin(), c.end(), c.begin(),
408 [](unsigned char ch) { return char(tolower(ch)); });
409 emit(c + " " + w[1]);
410 return;
411 }
412 if (c == "RESOURCES") {
413 Vec v;
414 for (size_t i = 1; i < w.size(); ++i)
415 v.push_back(stoi(w[i]));
416 state = State{};
417 state.total = state.available = v;
418 emit("resources " + vl(v));
419 return;
420 }
421 if (c == "PROCESS" || c == "CLAIM" || c == "REQUEST" || c == "WHATIF" || c == "RELEASE") {
422 int id = stoi(w[1]);
423 Vec v = parseVector(w, 2);
424 string name = pn(id);
425 if (c == "PROCESS") {
426 if (state.processes.count(id))
427 fail("duplicate process " + name);
428 if (!fits(v, state.total))
429 fail("max exceeds total");
430 state.processes[id] = {id, commandNumber, v, Vec(v.size())};
431 emit("process " + name + " max " + vl(v));
432 return;
433 }
434 Process &p = require(id, c == "CLAIM" || c == "RELEASE");
435 if (c == "CLAIM") {
436 if (!fits(p.allocation, v))
437 fail("claim below allocation");
438 if (!fits(v, state.total))
439 fail("max exceeds total");
440 Vec old = p.maximum;
441 p.maximum = v;
442 bool unsafe =
443 mode == "AVOID" && !check(state.available, false, tracing).remaining.empty();
444 p.maximum = old;
445 if (unsafe)
446 fail("claim makes state unsafe");
447 p.maximum = v;
448 emit("claim " + name + " max " + vl(v));
449 retry();
450 return;
451 }
452 if (c == "RELEASE") {
453 if (sum(v) == 0)
454 fail("zero vector");
455 if (!fits(v, p.allocation))
456 fail("release exceeds allocation");
457 p.allocation = sub(p.allocation, v);
458 state.available = add(state.available, v);
459 emit("released " + name + " " + vl(v));
460 record("released", id, v);
461 retry();
462 return;
463 }
464 string reject = rejection(p, v), prefix = "whatif " + name + " " + vl(v);
465 if (!reject.empty()) {
466 if (c == "REQUEST")
467 fail(reject);
468 emit(prefix + ": rejected (" + reject + ")");
469 return;
470 }
471 Decision d = decision(p, v, tracing);
472 if (c == "WHATIF")
473 emit(!d.reason.empty()
474 ? prefix + ": blocked (" + d.reason + ")"
475 : prefix + ": granted" +
476 (mode == "AVOID" ? ", safe sequence " + pl(d.sequence) : ""));
477 else if (d.reason.empty()) {
478 allocate(p, v);
479 emit("granted " + name + " " + vl(v));
480 record("granted", id, v);
481 } else {
482 state.queue.push_back({id, v, commandNumber});
483 emit("blocked " + name + " " + vl(v) + ": " + d.reason);
484 record("blocked", id, v);
485 if (autodetect) {
486 Vec dead = detect(tracing);
487 if (!dead.empty())
488 emit("deadlock detected: " + pl(dead));
489 }
490 }
491 return;
492 }
493 if (c == "ADD" || c == "REMOVE") {
494 Vec v = parseVector(w, 1);
495 if (sum(v) == 0)
496 fail("zero vector");
497 if (c == "ADD") {
498 state.total = add(state.total, v);
499 state.available = add(state.available, v);
500 emit("added " + vl(v) + " total " + vl(state.total));
501 retry();
502 } else {
503 if (!fits(v, state.available))
504 fail("remove exceeds available");
505 Vec total = sub(state.total, v), available = sub(state.available, v);
506 if (find(total.begin(), total.end(), 0) != total.end())
507 fail("total must stay positive");
508 for (const auto &item : state.processes)
509 if (!fits(item.second.maximum, total))
510 fail("max exceeds total");
511 if (mode == "AVOID" && !check(available, false, tracing).remaining.empty())
512 fail("removal makes state unsafe");
513 state.total = total;
514 state.available = available;
515 emit("removed " + vl(v) + " total " + vl(total));
516 }
517 return;
518 }
519 if (c == "FINISH" || c == "ABORT") {
520 int id = stoi(w[1]);
521 require(id, c == "FINISH");
522 terminate(id, c == "FINISH" ? "finished" : "aborted");
523 retry();
524 } else if (c == "CANCEL") {
525 int id = stoi(w[1]);
526 require(id, false);
527 const Waiting *entry = waiting(id);
528 if (!entry)
529 fail("process " + pn(id) + " is not blocked");
530 Vec value = entry->value;
531 state.queue.erase(remove_if(state.queue.begin(), state.queue.end(),
532 [id](const Waiting &r) { return r.id == id; }),
533 state.queue.end());
534 emit("cancelled " + pn(id) + " " + vl(value));
535 record("cancelled", id, value);
536 } else if (c == "MAXGRANT") {
537 int id = stoi(w[1]);
538 Process &p = require(id, true);
539 Vec result(state.total.size()), n = need(p);
540 for (size_t i = 0; i < result.size(); ++i) {
541 Vec v(result.size());
542 for (int amount = n[i]; amount > 0; --amount) {
543 v[i] = amount;
544 if (rejection(p, v).empty() && decision(p, v, false).reason.empty()) {
545 result[i] = amount;
546 break;
547 }
548 }
549 }
550 emit("maxgrant " + pn(id) + ": " + vl(result));
551 } else if (c == "SAFE") {
552 if (state.processes.empty())
553 emit("safe: no processes");
554 else {
555 Check r = check(state.available, false, tracing);
556 emit(r.remaining.empty() ? "safe: " + pl(r.sequence)
557 : "unsafe: " + pl(r.remaining) + " cannot finish");
558 }
559 } else if (c == "SEQUENCES") {
560 if (state.processes.empty()) {
561 emit("safe sequences: no processes");
562 return;
563 }
564 int count = 0, limit = stoi(w[1]);
565 Vec path;
566 vector<string> found;
567 set<int> used;
568 function<void(Vec)> search = [&](Vec work) {
569 if (path.size() == state.processes.size()) {
570 ++count;
571 if (int(found.size()) < limit)
572 found.push_back(pl(path));
573 return;
574 }
575 for (const auto &item : state.processes)
576 if (!used.count(item.first) && fits(need(item.second), work)) {
577 used.insert(item.first);
578 path.push_back(item.first);
579 search(add(work, item.second.allocation));
580 path.pop_back();
581 used.erase(item.first);
582 }
583 };
584 search(state.available);
585 emit("safe sequences: " + to_string(count));
586 for (const string &line : found)
587 emit(line);
588 } else if (c == "DEFICIT") {
589 vector<string> values;
590 for (size_t i = 0; i < state.total.size(); ++i) {
591 int limit = 0;
592 for (const auto &item : state.processes)
593 limit += need(item.second)[i];
594 string value = "-";
595 for (int x = 0; x <= limit; ++x) {
596 Vec work = state.available;
597 work[i] += x;
598 if (check(work, false, false).remaining.empty()) {
599 value = to_string(x);
600 break;
601 }
602 }
603 values.push_back(value);
604 }
605 emit("deficit: (" + join(values, ",") + ")");
606 } else if (c == "STATE")
607 showState();
608 else if (c == "STATS")
609 stats();
610 else if (c == "DETECT") {
611 Vec dead = detect(tracing);
612 emit(dead.empty() ? "no deadlock" : "deadlocked: " + pl(dead));
613 } else if (c == "VICTIMS") {
614 Vec dead = detect(false);
615 if (dead.empty()) {
616 emit("no deadlock");
617 return;
618 }
619 vector<Vec> groups;
620 for (int mask = 1; mask < (1 << dead.size()); ++mask) {
621 Vec group;
622 for (size_t i = 0; i < dead.size(); ++i)
623 if (mask & (1 << i))
624 group.push_back(dead[i]);
625 groups.push_back(group);
626 }
627 auto held = [](const Vec &ids) {
628 int n = 0;
629 for (int id : ids)
630 n += sum(state.processes.at(id).allocation);
631 return n;
632 };
633 sort(groups.begin(), groups.end(), [&](const Vec &a, const Vec &b) {
634 if (a.size() != b.size())
635 return a.size() < b.size();
636 int x = held(a), y = held(b);
637 return x != y ? x < y : a < b;
638 });
639 for (const Vec &group : groups)
640 if (check(state.available, true, false, set<int>(group.begin(), group.end()))
641 .remaining.empty()) {
642 emit("victims: " + pl(group));
643 break;
644 }
645 } else if (c == "RECOVER") {
646 Vec dead = detect(tracing);
647 int count = 0;
648 if (w.size() > 1) {
649 for (int id : dead) {
650 terminate(id, "aborted");
651 ++count;
652 }
653 if (count)
654 retry();
655 } else
656 while (!dead.empty()) {
657 int victim = dead.front();
658 for (int id : dead) {
659 const Process &a = state.processes.at(id);
660 const Process &b = state.processes.at(victim);
661 bool better;
662 if (policy == "ALLOC")
663 better = sum(a.allocation) < sum(b.allocation) ||
664 (sum(a.allocation) == sum(b.allocation) && id > victim);
665 else if (policy == "NEED")
666 better = sum(need(a)) > sum(need(b)) ||
667 (sum(need(a)) == sum(need(b)) && id > victim);
668 else
669 better = a.created > b.created;
670 if (better)
671 victim = id;
672 }
673 terminate(victim, "aborted");
674 retry();
675 ++count;
676 dead = detect(tracing);
677 }
678 emit(count ? "recovered: " + to_string(count) + " aborted" : "no deadlock");
679 } else if (c == "WAITGRAPH")
680 graph(false);
681 else if (c == "GRAPH")
682 graph(true);
683 else if (c == "HISTORY") {
684 bool any = false;
685 for (const Event &e : state.history) {
686 if (w.size() > 1 && !(isdigit(static_cast<unsigned char>(w[1][0])) ? e.id == stoi(w[1])
687 : e.kind == w[1]))
688 continue;
689 emit("t=" + to_string(e.command) + " " + e.kind + " " + pn(e.id) + " " + vl(e.value) +
690 (e.kind == "unblocked" ? " waited " + to_string(e.waited) : ""));
691 any = true;
692 }
693 if (!any)
694 emit("no events");
695 } else if (c == "SNAPSHOT") {
696 snapshots[w[1]] = state;
697 emit("saved " + w[1]);
698 } else if (c == "RESTORE") {
699 if (!snapshots.count(w[1]))
700 fail("unknown snapshot " + w[1]);
701 state = snapshots.at(w[1]);
702 emit("restored " + w[1]);
703 }
