14fn sum(v: Vec) I {
15 var answer: I = 0;
16 for (v) |item| answer += item;
17 return answer;
18}
19fn fits(a: Vec, b: Vec) bool {
20 for (a, b) |left, right| if (left > right) return false;
21 return true;
22}
23fn combine(a: Vec, b: Vec, sign: I) Vec {
24 var answer = a;
25 for (&answer, b) |*left, right| left.* += sign * right;
26 return answer;

114 fn inspect(s: *const Manager, model: Model, detection: bool, verbose: bool) !Check {
115 var work = model.available;
116 var done: [100]bool = undefined;
117 var idle: Ids = .{};
118 for (model.jobs, 0..) |job, who| {
119 done[who] = !job.exists;
120 if (job.exists and detection and sum(job.allocation) == 0) {
121 done[who] = true;
122 idle.add(who);
123 }
124 }
125 const tag: Text = if (detection) "detect" else "safety";
126 if (verbose) {
127 const suffix = if (idle.len > 0) try s.h.text(" idle {s}", .{try s.h.people(idle, " ")}) else "";
128 try s.h.line("[{s}] start work={s}{s}", .{ tag, try s.h.vector(work, model.m), suffix });
129 }
130 var result: Check = .{};
131 while (true) {
132 var chosen: ?usize = null;
133 for (0..100) |who| {
134 if (done[who]) continue;
135 var requirement = combine(model.jobs[who].maximum, model.jobs[who].allocation, -1);
136 if (detection) {
137 requirement = @splat(0);
138 for (model.queue.items) |waiting| {
139 if (waiting.who == who) requirement = waiting.amount;
140 }
141 }
142 if (fits(requirement, work)) {
143 chosen = who;
144 break;
145 }
146 }
147 const who = chosen orelse break;
148 done[who] = true;
149 work = combine(work, model.jobs[who].allocation, 1);
150 result.order.add(who);
151 if (verbose) try s.h.line("[{s}] P{d} can finish, work={s}", .{ tag, who, try s.h.vector(work, model.m) });
152 }
153 for (done, 0..) |finished, who| if (!finished) {
154 result.remaining.add(who);
155 };
156 if (verbose and result.remaining.len > 0) try s.h.line("[{s}] stuck work={s} remaining {s}", .{ tag, try s.h.vector(work, model.m), try s.h.people(result.remaining, " ") });
157 return result;
158 }
159 fn decide(s: *const Manager, who: usize, amount: Vec, reject: bool, verbose: bool) !Decision {
160 if (reject) {
161 if (s.queued(who) != null) return .{ .status = .rejected, .reason = try s.h.text("process P{d} is blocked", .{who}) };
162 if (sum(amount) == 0) return .{ .status = .rejected, .reason = "zero vector" };
163 if (!fits(amount, s.needOf(who))) return .{ .status = .rejected, .reason = "request exceeds need" };
164 if (eq(s.mode, "ORDER")) {
165 var last: ?usize = null;
166 for (s.data.jobs[who].allocation, 0..) |value, j| if (value > 0) {
167 last = j;
168 };
169 if (last) |end| for (amount[0 .. end + 1]) |value| {
170 if (value > 0) return .{ .status = .rejected, .reason = "resource order violation" };
171 };
172 }
173 }
174 if (!fits(amount, s.data.available)) return .{ .status = .blocked, .reason = "insufficient resources" };
175 if (!eq(s.mode, "AVOID")) return .{ .status = .granted };
176 var trial = s.data;
177 trial.available = combine(trial.available, amount, -1);
178 trial.jobs[who].allocation = combine(trial.jobs[who].allocation, amount, 1);
179 const checked = try s.inspect(trial, false, verbose);
180 return if (checked.remaining.len == 0) .{ .status = .granted, .order = checked.order } else .{ .status = .blocked, .reason = "unsafe state" };
181 }
182 fn record(s: *Manager, kind: Text, who: usize, amount: Vec, waited: I) !void {

185 fn grant(s: *Manager, who: usize, amount: Vec) void {
186 s.data.available = combine(s.data.available, amount, -1);
187 s.data.jobs[who].allocation = combine(s.data.jobs[who].allocation, amount, 1);
188 }
189 fn retryQueue(s: *Manager) !void {
190 var at: usize = 0;
191 while (at < s.data.queue.items.len) {
192 const waiting = s.data.queue.items[at];
193 const choice = try s.decide(waiting.who, waiting.amount, false, s.tracing);
194 if (choice.status == .granted) {
195 s.grant(waiting.who, waiting.amount);
196 _ = s.data.queue.orderedRemove(at);
197 try s.h.line("unblocked P{d} {s}", .{ waiting.who, try s.h.vector(waiting.amount, s.data.m) });
198 try s.record("unblocked", waiting.who, waiting.amount, s.stamp - waiting.since);
199 } else at += 1;
200 }
201 }
202 fn terminate(s: *Manager, who: usize, kind: Text, retry_after: bool) !void {
203 const released = s.data.jobs[who].allocation;
204 if (s.queued(who)) |at| _ = s.data.queue.orderedRemove(at);
205 s.data.available = combine(s.data.available, released, 1);
206 s.data.jobs[who].exists = false;
207 try s.h.line("{s} P{d} released {s}", .{ kind, who, try s.h.vector(released, s.data.m) });
208 try s.record(kind, who, released, 0);
209 if (retry_after) try s.retryQueue();
210 }
211 fn permutations(s: *const Manager, work: Vec, prefix: Ids, count: *usize, limit: usize, results: *std.ArrayList(Ids)) anyerror!void {
212 var complete = true;
213 for (s.data.jobs, 0..) |job, who| {
214 if (job.exists and !prefix.contains(who)) {
215 complete = false;
216 if (fits(s.needOf(who), work)) {
217 var extended = prefix;
218 extended.add(who);
219 try s.permutations(combine(work, job.allocation, 1), extended, count, limit, results);
220 }
221 }
222 }
223 if (complete) {
224 count.* += 1;
225 if (results.items.len < limit) try results.append(s.h.a, prefix);
226 }
227 }

250 const Cycle = struct { path: Ids, labels: [100]usize };
251 fn findCycles(s: *const Manager, resource_graph: bool, path: Ids, labels: [100]usize, results: *std.ArrayList(Cycle)) anyerror!void {
252 const current = path.values[path.len - 1];
253 const start = path.values[0];
254 const wanted = s.requestOf(current);
255 for (s.data.jobs, 0..) |job, next_id| {
256 if (!job.exists or (!resource_graph and current == next_id)) continue;
257 for (0..s.data.m) |j| {
258 if (job.allocation[j] == 0 or (if (resource_graph) wanted[j] <= 0 else wanted[j] <= s.data.available[j])) continue;
259 var next_labels = labels;
260 next_labels[path.len - 1] = j;
261 if (next_id == start) {
262 try results.append(s.h.a, .{ .path = path, .labels = next_labels });
263 } else if (next_id > start and !path.contains(next_id)) {
264 var extended = path;
265 extended.add(next_id);
266 try s.findCycles(resource_graph, extended, next_labels, results);
267 }
268 if (!resource_graph) break;
269 }
270 }
271 }
272 fn graph(s: *const Manager, resource_graph: bool) !void {
273 var edges: usize = 0;
274 if (resource_graph) {
275 for (s.data.total[0..s.data.m]) |value| if (value != 1) return s.h.line("error: not single-instance", .{});
276 for (0..s.data.m) |j| for (s.data.jobs, 0..) |job, who| {
277 if (job.exists and job.allocation[j] > 0) try s.h.line("assign {c} -> P{d}", .{ @as(u8, 'A') + @as(u8, @intCast(j)), who });
278 };
279 for (0..2) |stage| for (s.data.jobs, 0..) |job, who| {
280 if (!job.exists) continue;
281 const wanted = s.requestOf(who);
282 const needed = s.needOf(who);
283 for (0..s.data.m) |j| {
284 if ((stage == 0 and wanted[j] > 0) or (stage == 1 and needed[j] > 0 and wanted[j] == 0)) try s.h.line("{s} P{d} -> {c}", .{ if (stage == 0) @as(Text, "request") else "claim", who, @as(u8, 'A') + @as(u8, @intCast(j)) });
285 }
286 };
287 } else {
288 for (s.data.jobs, 0..) |job, who| {
289 if (!job.exists) continue;
290 const wanted = s.requestOf(who);
291 for (s.data.jobs, 0..) |other, next_id| {
292 if (!other.exists or who == next_id) continue;
293 var labels: std.Io.Writer.Allocating = .init(s.h.a);
294 for (0..s.data.m) |j| {
295 if (wanted[j] > s.data.available[j] and other.allocation[j] > 0) {
296 if (labels.written().len > 0) try labels.writer.writeByte(',');
297 try labels.writer.writeByte(@as(u8, 'A') + @as(u8, @intCast(j)));
298 }
299 }
300 if (labels.written().len > 0) {
301 edges += 1;
302 try s.h.line("P{d} -> P{d} on {s}", .{ who, next_id, labels.written() });
303 }
304 }
305 }
306 if (edges == 0) try s.h.line("no wait edges", .{});
307 }
308 var results: std.ArrayList(Cycle) = .empty;
309 for (s.data.jobs, 0..) |job, who| {
310 if (job.exists) {
311 var path: Ids = .{};
312 path.add(who);
313 try s.findCycles(resource_graph, path, undefined, &results);
314 }
315 }

317 fn less(resource: bool, a: Cycle, b: Cycle) bool {
318 const limit = @min(a.path.len, b.path.len);
319 for (0..limit) |j| {
320 if (a.path.values[j] != b.path.values[j]) return a.path.values[j] < b.path.values[j];
321 if (resource and a.labels[j] != b.labels[j]) return a.labels[j] < b.labels[j];
322 }
323 return a.path.len < b.path.len;
324 }
325 }.less);
326 if (results.items.len == 0) return s.h.line("no cycles", .{});
327 for (results.items) |cycle| {
328 try s.h.out.print("cycle: P{d}", .{cycle.path.values[0]});
329 for (0..cycle.path.len) |j| {
330 if (resource_graph) try s.h.out.print(" -> {c}", .{@as(u8, 'A') + @as(u8, @intCast(cycle.labels[j]))});
331 try s.h.out.print(" -> P{d}", .{cycle.path.values[(j + 1) % cycle.path.len]});
332 }
333 try s.h.out.writeByte('\n');
334 }

336 fn command(s: *Manager, tokens: []const Text) !void {
337 const op = tokens[0];
338 if (eq(op, "RESOURCES")) {
339 s.data = .{ .m = tokens.len - 1 };
340 for (tokens[1..], 0..) |token, j| s.data.total[j] = number(token);
341 s.data.available = s.data.total;
342 return s.h.line("resources {s}", .{try s.h.vector(s.data.total, s.data.m)});
343 }
344 if (eq(op, "MODE") or eq(op, "POLICY") or eq(op, "TRACE") or eq(op, "AUTODETECT")) {
345 var label: Text = undefined;
346 if (eq(op, "MODE")) {
347 s.mode = tokens[1];
348 label = "mode";
349 } else if (eq(op, "POLICY")) {
350 s.policy = tokens[1];
351 label = "policy";
352 } else if (eq(op, "TRACE")) {
353 s.tracing = eq(tokens[1], "ON");
354 label = "trace";
355 } else {
356 s.autodetect = eq(tokens[1], "ON");
357 label = "autodetect";
358 }
359 return s.h.line("{s} {s}", .{ label, tokens[1] });
360 }
361 if (eq(op, "SNAPSHOT") or eq(op, "RESTORE")) {
362 var at: ?usize = null;
363 for (s.snapshots.items, 0..) |saved, j| if (eq(saved.name, tokens[1])) {
364 at = j;
365 };
366 if (eq(op, "SNAPSHOT")) {
367 const saved = Snapshot{ .name = tokens[1], .data = try s.data.clone(s.h.a) };
368 if (at) |j| s.snapshots.items[j] = saved else try s.snapshots.append(s.h.a, saved);
369 return s.h.line("saved {s}", .{tokens[1]});
370 }
371 const j = at orelse return s.h.line("error: unknown snapshot {s}", .{tokens[1]});
372 s.data = try s.snapshots.items[j].data.clone(s.h.a);
373 return s.h.line("restored {s}", .{tokens[1]});
374 }
375 if (eq(op, "SAFE") or eq(op, "DETECT")) {
376 var any_jobs = false;
377 for (s.data.jobs) |job| any_jobs = any_jobs or job.exists;
378 if (eq(op, "SAFE") and !any_jobs) return s.h.line("safe: no processes", .{});
379 const checked = try s.inspect(s.data, eq(op, "DETECT"), s.tracing);
380 if (eq(op, "DETECT")) {
381 if (checked.remaining.len == 0) return s.h.line("no deadlock", .{});
382 return s.h.line("deadlocked: {s}", .{try s.h.people(checked.remaining, " ")});
383 }
384 if (checked.remaining.len == 0) return s.h.line("safe: {s}", .{try s.h.people(checked.order, " ")});
385 return s.h.line("unsafe: {s} cannot finish", .{try s.h.people(checked.remaining, " ")});
386 }
387 if (eq(op, "SEQUENCES")) {
388 var results: std.ArrayList(Ids) = .empty;
389 var total_count: usize = 0;
390 var any_jobs = false;
391 for (s.data.jobs) |job| any_jobs = any_jobs or job.exists;
392 if (!any_jobs) return s.h.line("safe sequences: no processes", .{});
393 try s.permutations(s.data.available, .{}, &total_count, index(number(tokens[1])), &results);
394 try s.h.line("safe sequences: {d}", .{total_count});
395 for (results.items) |sequence| try s.h.line("{s}", .{try s.h.people(sequence, " ")});
396 return;
397 }
398 if (eq(op, "DEFICIT")) {
399 try s.h.out.writeAll("deficit: (");
400 for (0..s.data.m) |j| {
401 if (j > 0) try s.h.out.writeByte(',');
402 var limit: I = 0;
403 for (s.data.jobs, 0..) |job, who| if (job.exists) {
404 limit += s.needOf(who)[j];
405 };
406 var extra: I = 0;
407 while (extra <= limit) : (extra += 1) {
408 var trial = s.data;
409 trial.available[j] += extra;
410 if ((try s.inspect(trial, false, false)).remaining.len == 0) break;
411 }
412 if (extra > limit) try s.h.out.writeByte('-') else try s.h.out.print("{d}", .{extra});
413 }
414 return s.h.line(")", .{});
415 }
416 if (eq(op, "VICTIMS")) {
417 const dead = (try s.inspect(s.data, true, false)).remaining;
418 if (dead.len == 0) return s.h.line("no deadlock", .{});
419 var best = dead;
420 var best_weight: I = std.math.maxInt(I);
421 const limit = @as(usize, 1) << @as(u6, @intCast(dead.len));
422 for (1..limit) |mask| {
423 var selected: Ids = .{};
424 var weight: I = 0;
425 var trial = s.data;
426 for (dead.items(), 0..) |who, j| {
427 if (mask & (@as(usize, 1) << @as(u6, @intCast(j))) != 0) {
428 selected.add(who);
429 weight += sum(s.data.jobs[who].allocation);
430 trial.available = combine(trial.available, trial.jobs[who].allocation, 1);
431 trial.jobs[who].exists = false;
432 }
433 }
434 if (selected.len > best.len) continue;
435 var preferred = selected.len < best.len or weight < best_weight;
436 if (selected.len == best.len and weight == best_weight) {
437 for (selected.items(), best.items()) |a, b| {
438 if (a != b) {
439 preferred = a < b;
440 break;
441 }
442 }
443 }
444 if (preferred and (try s.inspect(trial, true, false)).remaining.len == 0) {
445 best = selected;
446 best_weight = weight;
447 }
448 }
449 return s.h.line("victims: {s}", .{try s.h.people(best, " ")});
450 }
451 if (eq(op, "RECOVER")) {
452 var count_aborted: usize = 0;
453 while (true) {
454 const dead = (try s.inspect(s.data, true, s.tracing)).remaining;
455 if (dead.len == 0) break;
456 if (tokens.len == 2) {
457 for (dead.items()) |who| {
458 try s.terminate(who, "aborted", false);
459 count_aborted += 1;
460 }
461 try s.retryQueue();
462 break;
463 }
464 var chosen = dead.values[0];
465 for (dead.items()) |who| {
466 const a = if (eq(s.policy, "ALLOC")) -sum(s.data.jobs[chosen].allocation) else if (eq(s.policy, "NEED")) sum(s.needOf(chosen)) else s.data.jobs[chosen].created;
467 const b = if (eq(s.policy, "ALLOC")) -sum(s.data.jobs[who].allocation) else if (eq(s.policy, "NEED")) sum(s.needOf(who)) else s.data.jobs[who].created;
468 if (b > a or (b == a and who > chosen)) chosen = who;
469 }
470 try s.terminate(chosen, "aborted", true);
471 count_aborted += 1;
472 }
473 if (count_aborted == 0) return s.h.line("no deadlock", .{});
474 return s.h.line("recovered: {d} aborted", .{count_aborted});
475 }
476 if (eq(op, "STATE")) {
477 var names: std.Io.Writer.Allocating = .init(s.h.a);
478 for (0..s.data.m) |j| {
479 if (j > 0) try names.writer.writeByte(' ');
480 const width = (try s.h.text("{d}", .{s.data.total[j]})).len;
481 try names.writer.splatByteAll(' ', width - 1);
482 try names.writer.writeByte(@as(u8, 'A') + @as(u8, @intCast(j)));
483 }
484 const row = names.written();
485 const widths: [4]usize = .{ @max(10, row.len), @max(3, row.len), @max(4, row.len), @max(7, row.len) };
486 try s.tableRow("Process", &.{ "Allocation", "Max", "Need", "Request" }, widths);
487 try s.tableRow("", &.{ row, row, row, row }, widths);
488 for (s.data.jobs, 0..) |job, who| if (job.exists) {
489 try s.tableRow(try s.h.text("P{d}", .{who}), &.{ try s.tableVector(job.allocation), try s.tableVector(job.maximum), try s.tableVector(s.needOf(who)), try s.tableVector(s.requestOf(who)) }, widths);
490 };
491 try s.tableRow("Available", &.{try s.tableVector(s.data.available)}, widths);
492 try s.tableRow("Total", &.{try s.tableVector(s.data.total)}, widths);
493 if (s.data.queue.items.len == 0) return s.h.line("blocked queue: empty", .{});
494 try s.h.out.writeAll("blocked queue: ");
495 for (s.data.queue.items, 0..) |waiting, j| {
496 if (j > 0) try s.h.out.writeAll(", ");
497 try s.h.out.print("P{d} {s} since t={d}", .{ waiting.who, try s.h.vector(waiting.amount, s.data.m), waiting.since });
498 }
499 return s.h.out.writeByte('\n');
500 }
501 if (eq(op, "STATS")) {
502 for (0..s.data.m) |j| {
503 const used = s.data.total[j] - s.data.available[j];
504 const percent = @divTrunc(used * 1000, s.data.total[j]);
505 var wanted: I = 0;
506 for (s.data.queue.items) |waiting| wanted += waiting.amount[j];
507 try s.h.line("{c}: allocated {d}/{d} ({d}.{d}%), requested {d}", .{ @as(u8, 'A') + @as(u8, @intCast(j)), used, s.data.total[j], @divTrunc(percent, 10), @mod(percent, 10), wanted });
508 }
509 const kinds = [_]Text{ "granted", "blocked", "unblocked", "released", "finished", "aborted", "cancelled" };
510 var counts: [7]I = @splat(0);
511 var waited_sum: I = 0;
512 for (s.data.events.items) |event| {
513 for (kinds, 0..) |kind, j| if (eq(kind, event.kind)) {
514 counts[j] += 1;
515 };
516 if (eq(event.kind, "unblocked")) waited_sum += event.waited;
517 }
518 try s.h.out.writeAll("events: ");
519 for (kinds, counts, 0..) |kind, count, j| {
520 if (j > 0) try s.h.out.writeAll(", ");
521 try s.h.out.print("{s} {d}", .{ kind, count });
522 }
523 try s.h.out.writeByte('\n');
524 if (counts[2] == 0) return s.h.line("waits: none", .{});
525 const average = @divTrunc(waited_sum * 100, counts[2]);
526 return s.h.line("waits: {d} unblocked, total {d}, average {d}.{d:0>2}", .{ counts[2], waited_sum, @divTrunc(average, 100), @mod(average, 100) });
527 }
528 if (eq(op, "HISTORY")) {
529 var printed = false;
530 for (s.data.events.items) |event| {
531 if (tokens.len == 2 and !eq(tokens[1], event.kind) and !eq(tokens[1], try s.h.text("{d}", .{event.who}))) continue;
532 const suffix = if (eq(event.kind, "unblocked")) try s.h.text(" waited {d}", .{event.waited}) else "";
533 try s.h.line("t={d} {s} P{d} {s}{s}", .{ event.stamp, event.kind, event.who, try s.h.vector(event.amount, s.data.m), suffix });
534 printed = true;
535 }
536 if (!printed) try s.h.line("no events", .{});
537 return;
538 }
539 if (eq(op, "WAITGRAPH") or eq(op, "GRAPH")) return s.graph(eq(op, "GRAPH"));
540 const global_vector = eq(op, "ADD") or eq(op, "REMOVE");
541 const vector_command = global_vector or eq(op, "PROCESS") or eq(op, "CLAIM") or eq(op, "REQUEST") or eq(op, "WHATIF") or eq(op, "RELEASE");
542 const who: usize = if (global_vector) 0 else index(number(tokens[1]));
543 var amount: Vec = @splat(0);
544 if (vector_command) {
545 const beginning: usize = if (global_vector) 1 else 2;
546 if (tokens.len - beginning != s.data.m) return s.h.line("error: wrong vector length", .{});
547 for (tokens[beginning..], 0..) |token, j| amount[j] = number(token);
548 }
549 if (!global_vector and !eq(op, "PROCESS") and !s.data.jobs[who].exists) return s.h.line("error: unknown process P{d}", .{who});
550 if (eq(op, "PROCESS")) {
551 if (s.data.jobs[who].exists) return s.h.line("error: duplicate process P{d}", .{who});
552 if (!fits(amount, s.data.total)) return s.h.line("error: max exceeds total", .{});
553 s.data.jobs[who] = .{ .exists = true, .maximum = amount, .created = s.stamp };
554 return s.h.line("process P{d} max {s}", .{ who, try s.h.vector(amount, s.data.m) });
555 }
556 if (eq(op, "REQUEST") or eq(op, "WHATIF")) {
557 const choice = try s.decide(who, amount, true, s.tracing);
558 if (eq(op, "WHATIF")) {
559 var description: Text = if (choice.status == .granted) "granted" else try s.h.text("{s} ({s})", .{ @tagName(choice.status), choice.reason });
560 if (choice.status == .granted and eq(s.mode, "AVOID")) description = try s.h.text("{s}, safe sequence {s}", .{ description, try s.h.people(choice.order, " ") });
561 return s.h.line("whatif P{d} {s}: {s}", .{ who, try s.h.vector(amount, s.data.m), description });
562 }
563 if (choice.status == .rejected) return s.h.line("error: {s}", .{choice.reason});
564 if (choice.status == .granted) {
565 s.grant(who, amount);
566 try s.record("granted", who, amount, 0);
567 return s.h.line("granted P{d} {s}", .{ who, try s.h.vector(amount, s.data.m) });
568 }
569 try s.data.queue.append(s.h.a, .{ .who = who, .amount = amount, .since = s.stamp });
570 try s.record("blocked", who, amount, 0);
571 try s.h.line("blocked P{d} {s}: {s}", .{ who, try s.h.vector(amount, s.data.m), choice.reason });
572 if (s.autodetect) {
573 const dead = (try s.inspect(s.data, true, s.tracing)).remaining;
574 if (dead.len > 0) try s.h.line("deadlock detected: {s}", .{try s.h.people(dead, " ")});
575 }
576 return;
577 }
578 if (eq(op, "ABORT")) return s.terminate(who, "aborted", true);
579 if (eq(op, "CANCEL")) {
580 const at = s.queued(who) orelse return s.h.line("error: process P{d} is not blocked", .{who});
581 amount = s.data.queue.orderedRemove(at).amount;
582 try s.record("cancelled", who, amount, 0);
583 return s.h.line("cancelled P{d} {s}", .{ who, try s.h.vector(amount, s.data.m) });
584 }
585 if (!global_vector and s.queued(who) != null) return s.h.line("error: process P{d} is blocked", .{who});
586 if (eq(op, "FINISH")) return s.terminate(who, "finished", true);
587 if (eq(op, "MAXGRANT")) {
588 var answer: Vec = @splat(0);
589 const needed = s.needOf(who);
590 for (0..s.data.m) |j| {
591 var candidate: I = 1;
592 while (candidate <= needed[j]) : (candidate += 1) {
593 var trial: Vec = @splat(0);
594 trial[j] = candidate;
595 if ((try s.decide(who, trial, true, false)).status == .granted) answer[j] = candidate;
596 }
597 }
598 return s.h.line("maxgrant P{d}: {s}", .{ who, try s.h.vector(answer, s.data.m) });
599 }
600 if (eq(op, "CLAIM")) {
601 if (!fits(s.data.jobs[who].allocation, amount)) return s.h.line("error: claim below allocation", .{});
602 if (!fits(amount, s.data.total)) return s.h.line("error: max exceeds total", .{});
603 var trial = s.data;
604 trial.jobs[who].maximum = amount;
605 if (eq(s.mode, "AVOID") and (try s.inspect(trial, false, s.tracing)).remaining.len > 0) return s.h.line("error: claim makes state unsafe", .{});
606 s.data = trial;
607 try s.h.line("claim P{d} max {s}", .{ who, try s.h.vector(amount, s.data.m) });
608 return s.retryQueue();
609 }
610 if (sum(amount) == 0) return s.h.line("error: zero vector", .{});
611 if (eq(op, "RELEASE")) {
612 if (!fits(amount, s.data.jobs[who].allocation)) return s.h.line("error: release exceeds allocation", .{});
613 s.data.jobs[who].allocation = combine(s.data.jobs[who].allocation, amount, -1);
614 s.data.available = combine(s.data.available, amount, 1);
615 try s.record("released", who, amount, 0);
616 try s.h.line("released P{d} {s}", .{ who, try s.h.vector(amount, s.data.m) });
617 return s.retryQueue();
618 }
619 if (eq(op, "ADD")) {
620 s.data.total = combine(s.data.total, amount, 1);
621 s.data.available = combine(s.data.available, amount, 1);
622 try s.h.line("added {s} total {s}", .{ try s.h.vector(amount, s.data.m), try s.h.vector(s.data.total, s.data.m) });
623 return s.retryQueue();
624 }
625 if (!fits(amount, s.data.available)) return s.h.line("error: remove exceeds available", .{});
626 const reduced = combine(s.data.total, amount, -1);
627 for (reduced[0..s.data.m]) |item| if (item == 0) return s.h.line("error: total must stay positive", .{});
628 for (s.data.jobs) |job| if (job.exists and !fits(job.maximum, reduced)) return s.h.line("error: max exceeds total", .{});
629 var trial = s.data;
630 trial.total = reduced;
631 trial.available = combine(trial.available, amount, -1);
632 if (eq(s.mode, "AVOID") and (try s.inspect(trial, false, s.tracing)).remaining.len > 0) return s.h.line("error: removal makes state unsafe", .{});
633 s.data = trial;
634 try s.h.line("removed {s} total {s}", .{ try s.h.vector(amount, s.data.m), try s.h.vector(s.data.total, s.data.m) });
635 }
636};
637fn readLine(r: *std.Io.Reader, a: std.mem.Allocator) !?Text {
638 var b: std.Io.Writer.Allocating = .init(a);
639 defer b.deinit();
640 _ = try r.streamDelimiterEnding(&b.writer, '\n');
641 if (r.bufferedLen() == 0 and b.written().len == 0) return null;
642 if (r.bufferedLen() > 0) r.toss(1);
643 return b.toOwnedSlice();
644}
645pub fn main(init: std.process.Init) !void {
646 var arena: std.heap.ArenaAllocator = .init(std.heap.page_allocator);
647 defer arena.deinit();
648 const a = arena.allocator();
649 var ib: [4096]u8 = undefined;
650 var ob: [4096]u8 = undefined;
651 var input = std.Io.File.stdin().readerStreaming(init.io, &ib);
652 var output = std.Io.File.stdout().writerStreaming(init.io, &ob);
653 var manager: Manager = .{ .h = .{ .a = a, .out = &output.interface } };
654 const count = number(std.mem.trim(u8, (try readLine(&input.interface, a)).?, "\r"));
655 for (0..index(count)) |j| {
656 const line = std.mem.trim(u8, (try readLine(&input.interface, a)).?, "\r");
657 var tokens: std.ArrayList(Text) = .empty;
658 var it = std.mem.splitScalar(u8, line, ' ');
659 while (it.next()) |token| try tokens.append(a, token);
660 manager.stamp = @as(I, @intCast(j)) + 1;
661 try manager.command(tokens.items);
662 }
663 try output.interface.flush();
664}
