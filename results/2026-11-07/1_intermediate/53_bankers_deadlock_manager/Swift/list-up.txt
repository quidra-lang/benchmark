            let waits = state.history.filter { $0.kind == "unblocked" },
                total = state.history.filter { $0.kind == "unblocked" }.reduce(0) { $0 + $1.wait }
            if waits.isEmpty {
                print("waits: none")
            } else {
                let average = total * 100 / waits.count
                print(
                    "waits: \(waits.count) unblocked, total \(total), average \(average / 100).\(String(format: "%02d", average % 100))"
                )
