        $waits = [];
        foreach ($this->history as $e) {
            ++$counts[$e[1]];
            if ($e[1] === "unblocked") {
                $waits[] = $e[4];
            }
        if (!$waits) {
            $this->out("waits: none");
        } else {
            $n = count($waits);
            $sum = array_sum($waits);
            $avg = intdiv($sum * 100, $n);
            $this->out(
                "waits: $n unblocked, total $sum, average " .
                    intdiv($avg, 100) .
                    "." .
                    str_pad((string) ($avg % 100), 2, "0", STR_PAD_LEFT),
            );
