4class Process {
5 final int id, created;
6 List<int> maximum, allocation;
7 Process(this.id, this.maximum, this.created, this.allocation);
8 List<int> need() => sub(maximum, allocation);
9 Process copy() => Process(id, List.of(maximum), created, List.of(allocation));
10}
11class Waiting {
12 final int id, since;
13 final List<int> vector;
14 Waiting(this.id, this.vector, this.since);
15}
16class Event {
17 final int command, id, waited;
18 final String kind;
19 final List<int> vector;
20 Event(this.command, this.kind, this.id, this.vector, this.waited);
21}
22class State {
23 List<int> total, available;
24 final SplayTreeMap<int, Process> processes = SplayTreeMap();
25 List<Waiting> queue = [];
26 final List<Event> history = [];
27 State(List<int> values) : total = List.of(values), available = List.of(values);
28 State copy() {
29 final result = State(total);
30 result.available = List.of(available);
31 for (final p in processes.values) { result.processes[p.id] = p.copy(); }
32 result.queue = queue.map((w) => Waiting(w.id, List.of(w.vector), w.since)).toList();
33 result.history.addAll(history.map((e) => Event(e.command, e.kind, e.id, List.of(e.vector), e.waited)));
34 return result;
35 }
36}

57void emit(String line) => output.writeln(line.trimRight());
58String pn(int id) => 'P$id';
59String pl(List<int> ids) => ids.map(pn).join(' ');
60String vl(List<int> values) => '(${values.join(',')})';
61String type(int i) => String.fromCharCode(65 + i);
62List<int> add(List<int> a, List<int> b) => List.generate(a.length, (i) => a[i] + b[i]);
63List<int> sub(List<int> a, List<int> b) => List.generate(a.length, (i) => a[i] - b[i]);
64int sum(List<int> v) => v.fold(0, (a, b) => a + b);
65bool fits(List<int> a, List<int> b) {
66 for (int i = 0; i < a.length; i++) { if (a[i] > b[i]) return false; }
67 return true;
68}
69Never fail(String message) => throw CommandError(message);
70Waiting? waiting(int id) {
71 for (final w in state.queue) { if (w.id == id) return w; }
72 return null;
73}
74List<int> requestOf(int id) => waiting(id)?.vector ?? List.filled(state.total.length, 0);
75Process requireProcess(int id, [bool unblocked = false]) {
76 final p = state.processes[id];
77 if (p == null) fail('unknown process ${pn(id)}');
78 if (unblocked && waiting(id) != null) fail('process ${pn(id)} is blocked');
79 return p;
80}
81List<int> vector(List<String> words, int start) {
82 if (words.length - start != state.total.length) fail('wrong vector length');
83 return words.skip(start).map(int.parse).toList();
84}
85Check check(List<int> available, bool detecting, bool traced, [Set<int> excluded = const {}]) {
86 List<int> work = List.of(available);
87 final List<int> remaining = [], idle = [], sequence = [];
88 for (final p in state.processes.values) {
89 if (excluded.contains(p.id)) { work = add(work, p.allocation); }
90 else if (detecting && sum(p.allocation) == 0) { idle.add(p.id); }
91 else { remaining.add(p.id); }
92 }
93 final prefix = detecting ? '[detect]' : '[safety]';
94 if (traced) emit('$prefix start work=${vl(work)}${idle.isEmpty ? '' : ' idle ${pl(idle)}'}');
95 while (true) {
96 int? chosen;
97 for (final id in remaining) {
98 if (fits(detecting ? requestOf(id) : state.processes[id]!.need(), work)) { chosen = id; break; }
99 }
100 if (chosen == null) break;
101 work = add(work, state.processes[chosen]!.allocation);
102 remaining.remove(chosen);
103 sequence.add(chosen);
104 if (traced) emit('$prefix ${pn(chosen)} can finish, work=${vl(work)}');
105 }
106 if (traced && remaining.isNotEmpty) emit('$prefix stuck work=${vl(work)} remaining ${pl(remaining)}');
107 return Check(sequence, remaining);
108}
109List<int> detect(bool traced) => check(state.available, true, traced).remaining;
110String? rejection(Process p, List<int> v) {
111 if (waiting(p.id) != null) return 'process ${pn(p.id)} is blocked';
112 if (sum(v) == 0) return 'zero vector';
113 if (!fits(v, p.need())) return 'request exceeds need';
114 if (mode == 'ORDER') {
115 int last = -1;
116 for (int i = 0; i < v.length; i++) { if (p.allocation[i] > 0) last = i; }
117 for (int i = 0; i <= last; i++) { if (v[i] > 0) return 'resource order violation'; }
118 }
119 return null;
120}
121Decision decision(Process p, List<int> v, bool traced) {
122 if (!fits(v, state.available)) return Decision('insufficient resources');
123 if (mode != 'AVOID') return Decision(null);
124 final old = p.allocation;
125 p.allocation = add(old, v);
126 final result = check(sub(state.available, v), false, traced);
127 p.allocation = old;
128 return Decision(result.remaining.isEmpty ? null : 'unsafe state', result.sequence);
129}
130void allocate(Process p, List<int> v) {
131 p.allocation = add(p.allocation, v);
132 state.available = sub(state.available, v);
133}
134void record(String kind, int id, List<int> v, [int waited = 0]) {
135 state.history.add(Event(command, kind, id, List.of(v), waited));
136}
137void retry() {
138 final List<Waiting> kept = [];
139 for (final w in state.queue) {
140 final p = state.processes[w.id]!;
141 if (decision(p, w.vector, tracing).reason == null) {
142 allocate(p, w.vector);
143 emit('unblocked ${pn(w.id)} ${vl(w.vector)}');
144 record('unblocked', w.id, w.vector, command - w.since);
145 } else { kept.add(w); }
146 }
147 state.queue = kept;
148}
149void terminate(int id, String kind) {
150 final p = state.processes.remove(id)!;
151 state.queue.removeWhere((w) => w.id == id);
152 state.available = add(state.available, p.allocation);
153 emit('$kind ${pn(id)} released ${vl(p.allocation)}');
154 record(kind, id, p.allocation);
155}
156int compareLists(List<int> a, List<int> b) {
157 for (int i = 0; i < a.length && i < b.length; i++) {
158 final c = a[i].compareTo(b[i]);
159 if (c != 0) return c;
160 }
161 return a.length.compareTo(b.length);
162}

167void graph(bool resources) {
168 final m = state.total.length;
169 if (resources && state.total.any((x) => x != 1)) fail('not single-instance');
170 final Map<int, List<Edge>> edges = {for (final id in state.processes.keys) id: []};
171 bool any = false;
172 if (!resources) {
173 for (final p in state.processes.values) {
174 final req = requestOf(p.id);
175 for (final q in state.processes.values) {
176 if (p.id == q.id) continue;
177 final List<String> labels = [];
178 for (int i = 0; i < m; i++) { if (req[i] > state.available[i] && q.allocation[i] > 0) labels.add(type(i)); }
179 if (labels.isNotEmpty) {
180 edges[p.id]!.add(Edge(q.id, -1));
181 emit('${pn(p.id)} -> ${pn(q.id)} on ${labels.join(',')}');
182 any = true;
183 }
184 }
185 }
186 if (!any) emit('no wait edges');
187 } else {
188 final holders = List.filled(m, -1);
189 for (final p in state.processes.values) {
190 for (int i = 0; i < m; i++) { if (p.allocation[i] > 0) holders[i] = p.id; }
191 }
192 for (int i = 0; i < m; i++) { if (holders[i] >= 0) emit('assign ${type(i)} -> ${pn(holders[i])}'); }
193 for (final p in state.processes.values) {
194 final req = requestOf(p.id);
195 for (int i = 0; i < m; i++) {
196 if (req[i] > 0) {
197 emit('request ${pn(p.id)} -> ${type(i)}');
198 if (holders[i] >= 0) edges[p.id]!.add(Edge(holders[i], i));
199 }
200 }
201 }
202 for (final p in state.processes.values) {
203 final req = requestOf(p.id), need = p.need();
204 for (int i = 0; i < m; i++) { if (need[i] > 0 && req[i] == 0) emit('claim ${pn(p.id)} -> ${type(i)}'); }
205 }
206 }
207 final List<(List<int>, String)> cycles = [];
208 for (final start in state.processes.keys) {
209 final Set<int> visited = {start};
210 void search(int current, List<int> key, List<String> labels) {
211 for (final edge in edges[current]!) {
212 final next = edge.destination;
213 if (next < start) continue;
214 final nextKey = [...key, if (resources) edge.resource, next];
215 final nextLabels = [...labels, if (resources) type(edge.resource), pn(next)];
216 if (next == start) {
217 if (resources || visited.length > 1) cycles.add((nextKey, 'cycle: ${nextLabels.join(' -> ')}'));
218 } else if (!visited.contains(next)) {
219 visited.add(next);
220 search(next, nextKey, nextLabels);
221 visited.remove(next);
222 }
223 }
224 }
225 search(start, [start], [pn(start)]);
226 }
227 cycles.sort((a, b) => compareLists(a.$1, b.$1));
228 for (final cycle in cycles) { emit(cycle.$2); }
229 if (cycles.isEmpty) emit('no cycles');
230}
231String table(List<int> v) => List.generate(v.length, (i) => v[i].toString().padLeft(state.total[i].toString().length)).join(' ');
232void showState() {
233 final names = List.generate(state.total.length, (i) => type(i).padLeft(state.total[i].toString().length)).join(' ');
234 const headings = ['Allocation', 'Max', 'Need', 'Request'];
235 final widths = headings.map((h) => h.length > names.length ? h.length : names.length).toList();
236 void row(String heading, List<String> fields) {
237 emit(heading.padRight(9) + List.generate(4, (i) => ' ${fields[i].padRight(widths[i])}').join());
238 }
239 row('Process', headings);
240 row('', List.filled(4, names));
241 for (final p in state.processes.values) { row(pn(p.id), [table(p.allocation), table(p.maximum), table(p.need()), table(requestOf(p.id))]); }
242 row('Available', [table(state.available), '', '', '']);
243 row('Total', [table(state.total), '', '', '']);
244 emit('blocked queue: ${state.queue.isEmpty ? 'empty' : state.queue.map((w) => '${pn(w.id)} ${vl(w.vector)} since t=${w.since}').join(', ')}');
245}
246void stats() {
247 for (int i = 0; i < state.total.length; i++) {
248 final total = state.total[i], used = state.total[i] - state.available[i];
249 int requested = 0;
250 for (final w in state.queue) { requested += w.vector[i]; }
251 final percent = used * 1000 ~/ total;
252 emit('${type(i)}: allocated $used/$total (${percent ~/ 10}.${percent % 10}%), requested $requested');
253 }
254 emit('events: ${kinds.map((kind) => '$kind ${state.history.where((e) => e.kind == kind).length}').join(', ')}');
255 final waits = state.history.where((e) => e.kind == 'unblocked').toList();
256 if (waits.isEmpty) { emit('waits: none'); }
257 else {
258 int total = 0;
259 for (final e in waits) { total += e.waited; }
260 final average = total * 100 ~/ waits.length;
261 emit('waits: ${waits.length} unblocked, total $total, average ${average ~/ 100}.${(average % 100).toString().padLeft(2, '0')}');
262 }
263}
264void execute(List<String> w) {
265 final c = w[0];
266 if (['MODE', 'POLICY', 'TRACE', 'AUTODETECT'].contains(c)) {
267 if (c == 'MODE') mode = w[1];
268 if (c == 'POLICY') policy = w[1];
269 if (c == 'TRACE') tracing = w[1] == 'ON';
270 if (c == 'AUTODETECT') autodetect = w[1] == 'ON';
271 emit('${c.toLowerCase()} ${w[1]}');
272 return;
273 }
274 if (c == 'RESOURCES') { state = State(w.skip(1).map(int.parse).toList()); emit('resources ${vl(state.total)}'); return; }
275 if (['PROCESS', 'CLAIM', 'REQUEST', 'WHATIF', 'RELEASE'].contains(c)) {
276 final id = int.parse(w[1]), v = vector(w, 2), name = pn(int.parse(w[1]));
277 if (c == 'PROCESS') {
278 if (state.processes.containsKey(id)) fail('duplicate process $name');
279 if (!fits(v, state.total)) fail('max exceeds total');
280 state.processes[id] = Process(id, v, command, List.filled(v.length, 0));
281 emit('process $name max ${vl(v)}');
282 return;
283 }
284 final p = requireProcess(id, c == 'CLAIM' || c == 'RELEASE');
285 if (c == 'CLAIM') {
286 if (!fits(p.allocation, v)) fail('claim below allocation');
287 if (!fits(v, state.total)) fail('max exceeds total');
288 final old = p.maximum;
289 p.maximum = v;
290 final unsafe = mode == 'AVOID' && check(state.available, false, tracing).remaining.isNotEmpty;
291 p.maximum = old;
292 if (unsafe) fail('claim makes state unsafe');
293 p.maximum = v;
294 emit('claim $name max ${vl(v)}');
295 retry();
296 return;
297 }
298 if (c == 'RELEASE') {
299 if (sum(v) == 0) fail('zero vector');
300 if (!fits(v, p.allocation)) fail('release exceeds allocation');
301 p.allocation = sub(p.allocation, v);
302 state.available = add(state.available, v);
303 emit('released $name ${vl(v)}');
304 record('released', id, v);
305 retry();
306 return;
307 }
308 final reject = rejection(p, v), prefix = 'whatif $name ${vl(v)}';
309 if (reject != null) {
310 if (c == 'REQUEST') fail(reject);
311 emit('$prefix: rejected ($reject)');
312 return;
313 }
314 final d = decision(p, v, tracing);
315 if (c == 'WHATIF') {
316 if (d.reason != null) { emit('$prefix: blocked (${d.reason})'); }
317 else { emit('$prefix: granted${mode == 'AVOID' ? ', safe sequence ${pl(d.sequence)}' : ''}'); }
318 } else if (d.reason == null) {
319 allocate(p, v);
320 emit('granted $name ${vl(v)}');
321 record('granted', id, v);
322 } else {
323 state.queue.add(Waiting(id, v, command));
324 emit('blocked $name ${vl(v)}: ${d.reason}');
325 record('blocked', id, v);
326 if (autodetect) { final dead = detect(tracing); if (dead.isNotEmpty) emit('deadlock detected: ${pl(dead)}'); }
327 }
328 return;
329 }
330 if (c == 'ADD' || c == 'REMOVE') {
331 final v = vector(w, 1);
332 if (sum(v) == 0) fail('zero vector');
333 if (c == 'ADD') {
334 state.total = add(state.total, v);
335 state.available = add(state.available, v);
336 emit('added ${vl(v)} total ${vl(state.total)}');
337 retry();
338 } else {
339 if (!fits(v, state.available)) fail('remove exceeds available');
340 final total = sub(state.total, v), available = sub(state.available, v);
341 if (total.contains(0)) fail('total must stay positive');
342 if (state.processes.values.any((p) => !fits(p.maximum, total))) fail('max exceeds total');
343 if (mode == 'AVOID' && check(available, false, tracing).remaining.isNotEmpty) fail('removal makes state unsafe');
344 state.total = total;
345 state.available = available;
346 emit('removed ${vl(v)} total ${vl(total)}');
347 }
348 return;
349 }
350 switch (c) {
351 case 'FINISH':
352 case 'ABORT': {
353 final id = int.parse(w[1]);
354 requireProcess(id, c == 'FINISH');
355 terminate(id, c == 'FINISH' ? 'finished' : 'aborted');
356 retry();
357 }
358 case 'CANCEL': {
359 final id = int.parse(w[1]);
360 requireProcess(id);
361 final entry = waiting(id);
362 if (entry == null) fail('process ${pn(id)} is not blocked');
363 state.queue.remove(entry);
364 emit('cancelled ${pn(id)} ${vl(entry.vector)}');
365 record('cancelled', id, entry.vector);
366 }
367 case 'MAXGRANT': {
368 final id = int.parse(w[1]);
369 final p = requireProcess(id, true);
370 final result = List.filled(state.total.length, 0);
371 for (int i = 0; i < result.length; i++) {
372 final v = List.filled(result.length, 0);
373 for (int amount = p.need()[i]; amount > 0; amount--) {
374 v[i] = amount;
375 if (rejection(p, v) == null && decision(p, v, false).reason == null) { result[i] = amount; break; }
376 }
377 }
378 emit('maxgrant ${pn(id)}: ${vl(result)}');
379 }
380 case 'SAFE': {
381 if (state.processes.isEmpty) { emit('safe: no processes'); }
382 else { final r = check(state.available, false, tracing); emit(r.remaining.isEmpty ? 'safe: ${pl(r.sequence)}' : 'unsafe: ${pl(r.remaining)} cannot finish'); }
383 }
384 case 'SEQUENCES': {
385 if (state.processes.isEmpty) { emit('safe sequences: no processes'); return; }
386 final limit = int.parse(w[1]);
387 int count = 0;
388 final List<int> path = [];
389 final List<String> found = [];
390 void search(List<int> work) {
391 if (path.length == state.processes.length) { count++; if (found.length < limit) found.add(pl(path)); return; }
392 for (final p in state.processes.values) {
393 if (!path.contains(p.id) && fits(p.need(), work)) {
394 path.add(p.id);
395 search(add(work, p.allocation));
396 path.removeLast();
397 }
398 }
399 }
400 search(state.available);
401 emit('safe sequences: $count');
402 for (final line in found) { emit(line); }
403 }
404 case 'DEFICIT': {
405 final List<String> results = [];
406 for (int i = 0; i < state.total.length; i++) {
407 int limit = 0;
408 for (final p in state.processes.values) { limit += p.need()[i]; }
409 String result = '-';
410 for (int x = 0; x <= limit; x++) {
411 final work = List.of(state.available);
412 work[i] += x;
413 if (check(work, false, false).remaining.isEmpty) { result = x.toString(); break; }
414 }
415 results.add(result);
416 }
417 emit('deficit: (${results.join(',')})');
418 }
419 case 'STATE': {
420 showState();
421 }
422 case 'STATS': {
423 stats();
424 }
425 case 'DETECT': {
426 final dead = detect(tracing);
427 emit(dead.isEmpty ? 'no deadlock' : 'deadlocked: ${pl(dead)}');
428 }
429 case 'VICTIMS': {
430 final dead = detect(false);
431 if (dead.isEmpty) { emit('no deadlock'); return; }
432 final List<List<int>> groups = [];
433 for (int mask = 1; mask < (1 << dead.length); mask++) {
434 groups.add([for (int i = 0; i < dead.length; i++) if ((mask & (1 << i)) != 0) dead[i]]);
435 }
436 int held(List<int> ids) => ids.fold(0, (total, id) => total + sum(state.processes[id]!.allocation));
437 groups.sort((a, b) {
438 int result = a.length.compareTo(b.length);
439 if (result == 0) result = held(a).compareTo(held(b));
440 return result == 0 ? compareLists(a, b) : result;
441 });
442 for (final group in groups) {
443 if (check(state.available, true, false, group.toSet()).remaining.isEmpty) { emit('victims: ${pl(group)}'); break; }
444 }
445 }
446 case 'RECOVER': {
447 List<int> dead = detect(tracing);
448 int count = 0;
449 if (w.length > 1) {
450 for (final id in dead) { terminate(id, 'aborted'); count++; }
451 if (count > 0) retry();
452 } else {
453 while (dead.isNotEmpty) {
454 int victim = dead.first;
455 for (final id in dead) {
456 final a = state.processes[id]!, b = state.processes[victim]!;
457 final better = policy == 'ALLOC'
458 ? sum(a.allocation) < sum(b.allocation) || (sum(a.allocation) == sum(b.allocation) && id > victim)
459 : policy == 'NEED'
460 ? sum(a.need()) > sum(b.need()) || (sum(a.need()) == sum(b.need()) && id > victim)
461 : a.created > b.created;
462 if (better) victim = id;
463 }
464 terminate(victim, 'aborted');
465 retry();
466 count++;
467 dead = detect(tracing);
468 }
469 }
470 emit(count == 0 ? 'no deadlock' : 'recovered: $count aborted');
471 }
472 case 'WAITGRAPH': {
473 graph(false);
474 }
475 case 'GRAPH': {
476 graph(true);
477 }
478 case 'HISTORY': {
479 bool any = false;
480 for (final e in state.history) {
481 if (w.length > 1 && !(int.tryParse(w[1]) != null ? e.id == int.parse(w[1]) : e.kind == w[1])) continue;
482 emit('t=${e.command} ${e.kind} ${pn(e.id)} ${vl(e.vector)}${e.kind == 'unblocked' ? ' waited ${e.waited}' : ''}');
483 any = true;
484 }
485 if (!any) emit('no events');
486 }
487 case 'SNAPSHOT': {
488 snapshots[w[1]] = state.copy();
489 emit('saved ${w[1]}');
490 }
491 case 'RESTORE': {
492 final saved = snapshots[w[1]];
493 if (saved == null) fail('unknown snapshot ${w[1]}');
494 state = saved.copy();
495 emit('restored ${w[1]}');
496 }
497 }
498}
499void main() {
500 final n = int.parse(stdin.readLineSync()!);
501 for (command = 1; command <= n; command++) {
502 try { execute(stdin.readLineSync()!.split(' ')); }
503 on CommandError catch (error) { emit('error: ${error.message}'); }
504 }
505 stdout.write(output);
506}
