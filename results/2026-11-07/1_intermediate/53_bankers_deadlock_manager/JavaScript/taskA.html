28function check(detect = false, verbose = false) {
29 let work = state.available.slice();
30 const idle = detect
31 ? ordered()
32 .filter((p) => zero(p.alloc))
33 .map((p) => p.id)
34 : [];
35 let remaining = ordered().filter((p) => !idle.includes(p.id));
36 const sequence = [];
37 const tag = detect ? "detect" : "safety";
38 if (verbose)
39 emit(`[${tag}] start work=${vec(work)}${idle.length ? " idle " + names(idle) : ""}`);
40 while (true) {
41 const p = remaining.find((p) =>
42 le(detect ? (waiting(p.id)?.v ?? state.total.map(() => 0)) : need(p), work),
43 );
44 if (!p) break;
45 work = plus(work, p.alloc);
46 sequence.push(p.id);
47 remaining = remaining.filter((q) => q.id !== p.id);
48 if (verbose) emit(`[${tag}] P${p.id} can finish, work=${vec(work)}`);
49 }
50 const ids = remaining.map((p) => p.id);
51 if (verbose && ids.length) emit(`[${tag}] stuck work=${vec(work)} remaining ${names(ids)}`);
52 return { sequence, remaining: ids };
53}
54function decide(p, v, reject, verbose) {
55 let reason = "";
56 if (reject) {
57 if (waiting(p.id)) reason = `process P${p.id} is blocked`;
58 else if (zero(v)) reason = "zero vector";
59 else if (!le(v, need(p))) reason = "request exceeds need";
60 else if (mode === "ORDER" && !zero(p.alloc)) {
61 const last = p.alloc.reduce((last, x, i) => (x > 0 ? i : last), -1);
62 if (v.some((x, i) => x > 0 && i <= last)) reason = "resource order violation";
63 }
64 }
65 if (reason) return { kind: "rejected", reason, sequence: [] };
66 if (!le(v, state.available))
67 return { kind: "blocked", reason: "insufficient resources", sequence: [] };
68 let sequence = [];
69 if (mode === "AVOID") {
70 p.alloc = plus(p.alloc, v);
71 state.available = minus(state.available, v);
72 const result = check(false, verbose);
73 p.alloc = minus(p.alloc, v);
74 state.available = plus(state.available, v);
75 sequence = result.sequence;
76 if (result.remaining.length) return { kind: "blocked", reason: "unsafe state", sequence };
77 }
78 return { kind: "granted", reason: "", sequence };
79}
80function allocate(p, v) {
81 p.alloc = plus(p.alloc, v);
82 state.available = minus(state.available, v);

84function retry() {
85 const pending = state.queue.slice();
86 for (const q of pending) {
87 const p = state.processes.find((p) => p.id === q.id);
88 if (decide(p, q.v, false, trace).kind === "granted") {
89 allocate(p, q.v);
90 state.queue = state.queue.filter((w) => w.id !== q.id);
91 emit(`unblocked P${q.id} ${vec(q.v)}`);
92 record("unblocked", q.id, q.v, time - q.since);
93 }
94 }

96function terminate(p, kind, retries = true) {
97 state.available = plus(state.available, p.alloc);
98 state.processes = state.processes.filter((q) => q.id !== p.id);
99 state.queue = state.queue.filter((q) => q.id !== p.id);
100 emit(`${kind} P${p.id} released ${vec(p.alloc)}`);
101 record(kind, p.id, p.alloc);
102 if (retries) retry();
103}
104function cycles(graph, resource) {
105 const found = [];
106 function visit(start, p, path, seen) {
107 for (const edge of graph.get(p) ?? []) {
108 const next = resource ? path.concat(edge.label[0], edge.to) : path.concat(edge.to);
109 if (edge.to === start) found.push(next);
110 else if (edge.to > start && !seen.includes(edge.to))
111 visit(start, edge.to, next, seen.concat(edge.to));
112 }
113 }
114 for (const p of ordered()) visit(p.id, p.id, [p.id], [p.id]);
115 found.sort((a, b) => {
116 for (let i = 0; i < Math.min(a.length, b.length); i++)
117 if (a[i] !== b[i]) return a[i] - b[i];
118 return a.length - b.length;
119 });
120 for (const c of found)
121 emit(
122 "cycle: " +
123 c
124 .map((n, i) => (resource && i % 2 ? String.fromCharCode(65 + n) : `P${n}`))
125 .join(" -> "),
126 );
127 if (!found.length) emit("no cycles");
128}
129function execute(tokens) {
130 const [command, ...args] = tokens;
131 const id = Number(args[0]);
132 const p = state.processes.find((p) => p.id === id);
133 const hasId = ["PROCESS", "CLAIM", "REQUEST", "WHATIF", "RELEASE"].includes(command);
134 const v = args.slice(hasId ? 1 : 0).map(Number);
135 const error = (message) => emit("error: " + message);
136 if (
137 ["PROCESS", "CLAIM", "ADD", "REMOVE", "REQUEST", "WHATIF", "RELEASE"].includes(command) &&
138 v.length !== state.total.length
139 ) {
140 error("wrong vector length");
141 return;
142 }
143 if (
144 ["CLAIM", "REQUEST", "WHATIF", "RELEASE", "FINISH", "ABORT", "CANCEL", "MAXGRANT"].includes(
145 command,
146 ) &&
147 !p
148 ) {
149 error(`unknown process P${id}`);
150 return;
151 }
152 switch (command) {
153 case "RESOURCES":
154 state = {
155 total: v.slice(),
156 available: v.slice(),
157 processes: [],
158 queue: [],
159 history: [],
160 };
161 emit("resources " + vec(v));
162 break;
163 case "MODE":
164 mode = args[0];
165 emit("mode " + mode);
166 break;
167 case "POLICY":
168 policy = args[0];
169 emit("policy " + policy);
170 break;
171 case "TRACE":
172 trace = args[0] === "ON";
173 emit("trace " + args[0]);
174 break;
175 case "AUTODETECT":
176 auto = args[0] === "ON";
177 emit("autodetect " + args[0]);
178 break;
179 case "PROCESS":
180 if (p) error(`duplicate process P${id}`);
181 else if (!le(v, state.total)) error("max exceeds total");
182 else {
183 state.processes.push({ id, max: v.slice(), alloc: v.map(() => 0), born: time });
184 emit(`process P${id} max ${vec(v)}`);
185 }
186 break;
187 case "CLAIM": {
188 if (waiting(id)) {
189 error(`process P${id} is blocked`);
190 break;
191 }
192 if (!le(p.alloc, v)) {
193 error("claim below allocation");
194 break;
195 }
196 if (!le(v, state.total)) {
197 error("max exceeds total");
198 break;
199 }
200 const old = p.max;
201 p.max = v.slice();
202 if (mode === "AVOID" && check(false, trace).remaining.length) {
203 p.max = old;
204 error("claim makes state unsafe");
205 break;
206 }
207 emit(`claim P${id} max ${vec(v)}`);
208 retry();
209 break;
210 }
211 case "ADD":
212 if (zero(v)) {
213 error("zero vector");
214 break;
215 }
216 state.total = plus(state.total, v);
217 state.available = plus(state.available, v);
218 emit(`added ${vec(v)} total ${vec(state.total)}`);
219 retry();
220 break;
221 case "REMOVE": {
222 if (zero(v)) {
223 error("zero vector");
224 break;
225 }
226 if (!le(v, state.available)) {
227 error("remove exceeds available");
228 break;
229 }
230 const total = minus(state.total, v);
231 if (total.some((x) => x === 0)) {
232 error("total must stay positive");
233 break;
234 }
235 if (state.processes.some((p) => !le(p.max, total))) {
236 error("max exceeds total");
237 break;
238 }
239 state.available = minus(state.available, v);
240 if (mode === "AVOID" && check(false, trace).remaining.length) {
241 state.available = plus(state.available, v);
242 error("removal makes state unsafe");
243 break;
244 }
245 state.total = total;
246 emit(`removed ${vec(v)} total ${vec(total)}`);
247 break;
248 }
249 case "REQUEST":
250 case "WHATIF": {
251 const d = decide(p, v, true, trace);
252 if (command === "WHATIF") {
253 emit(
254 `whatif P${id} ${vec(v)}: ` +
255 (d.kind === "granted"
256 ? "granted" +
257 (mode === "AVOID" ? ", safe sequence " + names(d.sequence) : "")
258 : `${d.kind} (${d.reason})`),
259 );
260 } else if (d.kind === "rejected") error(d.reason);
261 else if (d.kind === "granted") {
262 allocate(p, v);
263 emit(`granted P${id} ${vec(v)}`);
264 record("granted", id, v);
265 } else {
266 state.queue.push({ id, v: v.slice(), since: time });
267 emit(`blocked P${id} ${vec(v)}: ${d.reason}`);
268 record("blocked", id, v);
269 if (auto) {
270 const dead = check(true, trace).remaining;
271 if (dead.length) emit("deadlock detected: " + names(dead));
272 }
273 }
274 break;
275 }
276 case "RELEASE":
277 if (waiting(id)) error(`process P${id} is blocked`);
278 else if (zero(v)) error("zero vector");
279 else if (!le(v, p.alloc)) error("release exceeds allocation");
280 else {
281 p.alloc = minus(p.alloc, v);
282 state.available = plus(state.available, v);
283 emit(`released P${id} ${vec(v)}`);
284 record("released", id, v);
285 retry();
286 }
287 break;
288 case "FINISH":
289 case "ABORT":
290 if (command === "FINISH" && waiting(id)) error(`process P${id} is blocked`);
291 else terminate(p, command === "FINISH" ? "finished" : "aborted");
292 break;
293 case "CANCEL": {
294 const q = waiting(id);
295 if (!q) error(`process P${id} is not blocked`);
296 else {
297 state.queue = state.queue.filter((q) => q.id !== id);
298 emit(`cancelled P${id} ${vec(q.v)}`);
299 record("cancelled", id, q.v);
300 }
301 break;
302 }
303 case "MAXGRANT": {
304 if (waiting(id)) {
305 error(`process P${id} is blocked`);
306 break;
307 }
308 const result = need(p).map((n, i) => {
309 let low = 0,
310 high = n;
311 while (low < high) {
312 const mid = Math.floor((low + high + 1) / 2),
313 q = state.total.map(() => 0);
314 q[i] = mid;
315 if (decide(p, q, true, false).kind === "granted") low = mid;
316 else high = mid - 1;
317 }
318 return low;
319 });
320 emit(`maxgrant P${id}: ${vec(result)}`);
321 break;
322 }
323 case "SAFE": {
324 if (!state.processes.length) {
325 emit("safe: no processes");
326 break;
327 }
328 const c = check(false, trace);
329 emit(
330 c.remaining.length
331 ? `unsafe: ${names(c.remaining)} cannot finish`
332 : "safe: " + names(c.sequence),
333 );
334 break;
335 }
336 case "SEQUENCES": {
337 if (!state.processes.length) {
338 emit("safe sequences: no processes");
339 break;
340 }
341 let count = 0;
342 const first = [];
343 function walk(rest, work, seq) {
344 if (!rest.length) {
345 count++;
346 if (first.length < id) first.push(seq);
347 return;
348 }
349 for (const q of rest)
350 if (le(need(q), work))
351 walk(
352 rest.filter((p) => p !== q),
353 plus(work, q.alloc),
354 seq.concat(q.id),
355 );
356 }
357 walk(ordered(), state.available, []);
358 emit("safe sequences: " + count);
359 first.forEach((s) => emit(names(s)));
360 break;
361 }
362 case "DEFICIT": {
363 const original = state.available.slice();
364 const deficits = state.total.map((_, i) => {
365 let low = 0,
366 high = state.processes.reduce((s, p) => s + need(p)[i], 0);
367 state.available = original.slice();
368 state.available[i] += high;
369 if (check().remaining.length) return "-";
370 while (low < high) {
371 const mid = Math.floor((low + high) / 2);
372 state.available = original.slice();
373 state.available[i] += mid;
374 if (check().remaining.length) low = mid + 1;
375 else high = mid;
376 }
377 return String(low);
378 });
379 state.available = original;
380 emit(`deficit: (${deficits.join(",")})`);
381 break;
382 }
383 case "STATE": {
384 const widths = state.total.map((t) => String(t).length);
385 const rowName = widths.map((w, i) => String.fromCharCode(65 + i).padStart(w)).join(" ");
386 const cols = [10, 3, 4, 7].map((w) => Math.max(w, rowName.length));
387 const table = (v) => v.map((x, i) => String(x).padStart(widths[i])).join(" ");
388 const row = (heading, values) =>
389 emit(
390 (
391 heading.padEnd(9) +
392 cols.map((w, i) => " " + (values[i] ?? "").padEnd(w)).join("")
393 ).trimEnd(),
394 );
395 row("Process", ["Allocation", "Max", "Need", "Request"]);
396 row(
397 "",
398 cols.map(() => rowName),
399 );
400 ordered().forEach((p) =>
401 row(
402 `P${p.id}`,
403 [p.alloc, p.max, need(p), waiting(p.id)?.v ?? state.total.map(() => 0)].map(
404 table,
405 ),
406 ),
407 );
408 row("Available", [table(state.available)]);
409 row("Total", [table(state.total)]);
410 emit(
411 "blocked queue: " +
412 (state.queue.length
413 ? state.queue
414 .map((q) => `P${q.id} ${vec(q.v)} since t=${q.since}`)
415 .join(", ")
416 : "empty"),
417 );
418 break;
419 }
420 case "STATS": {
421 state.total.forEach((t, i) => {
422 const u = t - state.available[i],
423 percent = Math.floor((u * 1000) / t);
424 emit(
425 `${String.fromCharCode(65 + i)}: allocated ${u}/${t} (${Math.floor(percent / 10)}.${percent % 10}%), requested ${state.queue.reduce((s, q) => s + q.v[i], 0)}`,
426 );
427 });
428 const kinds = [
429 "granted",
430 "blocked",
431 "unblocked",
432 "released",
433 "finished",
434 "aborted",
435 "cancelled",
436 ];
437 emit(
438 "events: " +
439 kinds
440 .map((k) => `${k} ${state.history.filter((e) => e.kind === k).length}`)
441 .join(", "),
442 );
443 const waits = state.history.filter((e) => e.kind === "unblocked"),
444 total = waits.reduce((s, e) => s + e.wait, 0);
445 const average = waits.length ? Math.floor((total * 100) / waits.length) : 0;
446 emit(
447 waits.length
448 ? `waits: ${waits.length} unblocked, total ${total}, average ${Math.floor(average / 100)}.${String(average % 100).padStart(2, "0")}`
449 : "waits: none",
450 );
451 break;
452 }
453 case "DETECT": {
454 const dead = check(true, trace).remaining;
455 emit(dead.length ? "deadlocked: " + names(dead) : "no deadlock");
456 break;
457 }
458 case "VICTIMS": {
459 const dead = check(true).remaining;
460 if (!dead.length) {
461 emit("no deadlock");
462 break;
463 }
464 const candidates = [];
465 for (let mask = 1; mask < 2 ** dead.length; mask++)
466 candidates.push(dead.filter((_, i) => mask & (1 << i)));
467 const cost = (ids) =>
468 ids.reduce((s, id) => s + sum(state.processes.find((p) => p.id === id).alloc), 0);
469 candidates.sort((a, b) => {
470 if (a.length !== b.length) return a.length - b.length;
471 if (cost(a) !== cost(b)) return cost(a) - cost(b);
472 for (let i = 0; i < a.length; i++) if (a[i] !== b[i]) return a[i] - b[i];
473 return 0;
474 });
475 const saved = copy(state);
476 for (const ids of candidates) {
477 state = copy(saved);
478 for (const id of ids) {
479 const p = state.processes.find((p) => p.id === id);
480 state.available = plus(state.available, p.alloc);
481 }
482 state.processes = state.processes.filter((p) => !ids.includes(p.id));
483 state.queue = state.queue.filter((q) => !ids.includes(q.id));
484 if (!check(true).remaining.length) {
485 emit("victims: " + names(ids));
486 break;
487 }
488 }
489 state = saved;
490 break;
491 }
492 case "RECOVER": {
493 let count = 0;
494 if (args[0] === "ALL") {
495 const dead = check(true, trace).remaining;
496 for (const id of dead) {
497 terminate(
498 state.processes.find((p) => p.id === id),
499 "aborted",
500 false,
501 );
502 count++;
503 }
504 if (count) retry();
505 } else {
506 while (true) {
507 const dead = check(true, trace).remaining;
508 if (!dead.length) break;
509 const choices = ordered().filter((p) => dead.includes(p.id));
510 choices.sort((a, b) =>
511 policy === "YOUNGEST"
512 ? b.born - a.born
513 : (policy === "ALLOC"
514 ? sum(a.alloc) - sum(b.alloc)
515 : sum(need(b)) - sum(need(a))) || b.id - a.id,
516 );
517 terminate(choices[0], "aborted");
518 count++;
519 }
520 }
521 emit(count ? `recovered: ${count} aborted` : "no deadlock");
522 break;
523 }
524 case "WAITGRAPH":
525 case "GRAPH": {
526 const resource = command === "GRAPH";
527 if (resource && state.total.some((x) => x !== 1)) {
528 error("not single-instance");
529 break;
530 }
531 const graph = new Map();
532 if (resource) {
533 state.total.forEach((_, i) =>
534 ordered().forEach((p) => {
535 if (p.alloc[i]) emit(`assign ${String.fromCharCode(65 + i)} -> P${p.id}`);
536 }),
537 );
538 for (const p of ordered()) {
539 const q = waiting(p.id);
540 if (!q) continue;
541 q.v.forEach((v, i) => {
542 if (!v) return;
543 emit(`request P${p.id} -> ${String.fromCharCode(65 + i)}`);
544 for (const holder of ordered())
545 if (holder.alloc[i]) {
546 const edges = graph.get(p.id) ?? [];
547 edges.push({ to: holder.id, label: [i] });
548 graph.set(p.id, edges);
549 }
550 });
551 }
552 for (const p of ordered())
553 need(p).forEach((v, i) => {
554 if (v && !(waiting(p.id)?.v[i] ?? 0))
555 emit(`claim P${p.id} -> ${String.fromCharCode(65 + i)}`);
556 });
557 } else {
558 let count = 0;
559 for (const p of ordered()) {
560 const q = waiting(p.id);
561 if (!q) continue;
562 for (const other of ordered())
563 if (p !== other) {
564 const label = q.v.flatMap((v, i) =>
565 v > state.available[i] && other.alloc[i] > 0 ? [i] : [],
566 );
567 if (!label.length) continue;
568 emit(
569 `P${p.id} -> P${other.id} on ${label.map((i) => String.fromCharCode(65 + i)).join(",")}`,
570 );
571 count++;
572 const edges = graph.get(p.id) ?? [];
573 edges.push({ to: other.id, label });
574 graph.set(p.id, edges);
575 }
576 }
577 if (!count) emit("no wait edges");
578 }
579 cycles(graph, resource);
580 break;
581 }
582 case "HISTORY": {
583 const events = state.history.filter(
584 (e) => !args.length || args[0] === String(e.id) || args[0] === e.kind,
585 );
586 events.forEach((e) =>
587 emit(
588 `t=${e.time} ${e.kind} P${e.id} ${vec(e.v)}${e.kind === "unblocked" ? " waited " + e.wait : ""}`,
589 ),
590 );
591 if (!events.length) emit("no events");
592 break;
593 }
594 case "SNAPSHOT":
595 snapshots.set(args[0], copy(state));
596 emit("saved " + args[0]);
597 break;
598 case "RESTORE": {
599 const saved = snapshots.get(args[0]);
600 if (!saved) error("unknown snapshot " + args[0]);
601 else {
602 state = copy(saved);
603 emit("restored " + args[0]);
604 }
605 break;
606 }
607 }
608}
