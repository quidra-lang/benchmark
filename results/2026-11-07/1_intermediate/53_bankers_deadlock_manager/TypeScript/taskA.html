41function check(detect = false, verbose = false): Check {
42 let work = state.available.slice();
43 const idle = detect
44 ? ordered()
45 .filter((p) => zero(p.alloc))
46 .map((p) => p.id)
47 : [];
48 let remaining = ordered().filter((p) => !idle.includes(p.id));
49 const sequence: number[] = [];
50 const tag = detect ? "detect" : "safety";
51 if (verbose)
52 emit(`[${tag}] start work=${vec(work)}${idle.length ? " idle " + names(idle) : ""}`);
53 while (true) {
54 const p = remaining.find((p) =>
55 le(detect ? (waiting(p.id)?.v ?? state.total.map(() => 0)) : need(p), work),
56 );
57 if (!p) break;
58 work = plus(work, p.alloc);
59 sequence.push(p.id);
60 remaining = remaining.filter((q) => q.id !== p.id);
61 if (verbose) emit(`[${tag}] P${p.id} can finish, work=${vec(work)}`);
62 }
63 const ids = remaining.map((p) => p.id);
64 if (verbose && ids.length) emit(`[${tag}] stuck work=${vec(work)} remaining ${names(ids)}`);
65 return { sequence, remaining: ids };
66}
67function decide(p: Process, v: Vector, reject: boolean, verbose: boolean): Decision {
68 let reason = "";
69 if (reject) {
70 if (waiting(p.id)) reason = `process P${p.id} is blocked`;
71 else if (zero(v)) reason = "zero vector";
72 else if (!le(v, need(p))) reason = "request exceeds need";
73 else if (mode === "ORDER" && !zero(p.alloc)) {
74 const last = p.alloc.reduce((last, x, i) => (x > 0 ? i : last), -1);
75 if (v.some((x, i) => x > 0 && i <= last)) reason = "resource order violation";
76 }
77 }
78 if (reason) return { kind: "rejected", reason, sequence: [] };
79 if (!le(v, state.available))
80 return { kind: "blocked", reason: "insufficient resources", sequence: [] };
81 let sequence: number[] = [];
82 if (mode === "AVOID") {
83 p.alloc = plus(p.alloc, v);
84 state.available = minus(state.available, v);
85 const result = check(false, verbose);
86 p.alloc = minus(p.alloc, v);
87 state.available = plus(state.available, v);
88 sequence = result.sequence;
89 if (result.remaining.length) return { kind: "blocked", reason: "unsafe state", sequence };
90 }
91 return { kind: "granted", reason: "", sequence };
92}
93function allocate(p: Process, v: Vector): void {
94 p.alloc = plus(p.alloc, v);
95 state.available = minus(state.available, v);

97function retry(): void {
98 const pending = state.queue.slice();
99 for (const q of pending) {
100 const p = state.processes.find((p) => p.id === q.id)!;
101 if (decide(p, q.v, false, trace).kind === "granted") {
102 allocate(p, q.v);
103 state.queue = state.queue.filter((w) => w.id !== q.id);
104 emit(`unblocked P${q.id} ${vec(q.v)}`);
105 record("unblocked", q.id, q.v, time - q.since);
106 }
107 }
108}
109function terminate(p: Process, kind: string, retries = true): void {
110 state.available = plus(state.available, p.alloc);
111 state.processes = state.processes.filter((q) => q.id !== p.id);
112 state.queue = state.queue.filter((q) => q.id !== p.id);
113 emit(`${kind} P${p.id} released ${vec(p.alloc)}`);
114 record(kind, p.id, p.alloc);
115 if (retries) retry();
116}
117function cycles(graph: Map<number, { to: number; label: number[] }[]>, resource: boolean): void {
118 const found: number[][] = [];
119 function visit(start: number, p: number, path: number[], seen: number[]): void {
120 for (const edge of graph.get(p) ?? []) {
121 const next = resource ? path.concat(edge.label[0], edge.to) : path.concat(edge.to);
122 if (edge.to === start) found.push(next);
123 else if (edge.to > start && !seen.includes(edge.to))
124 visit(start, edge.to, next, seen.concat(edge.to));
125 }
126 }
127 for (const p of ordered()) visit(p.id, p.id, [p.id], [p.id]);
128 found.sort((a, b) => {
129 for (let i = 0; i < Math.min(a.length, b.length); i++)
130 if (a[i] !== b[i]) return a[i] - b[i];
131 return a.length - b.length;
132 });
133 for (const c of found)
134 emit(
135 "cycle: " +
136 c
137 .map((n, i) => (resource && i % 2 ? String.fromCharCode(65 + n) : `P${n}`))
138 .join(" -> "),
139 );
140 if (!found.length) emit("no cycles");
141}
142function execute(tokens: string[]): void {
143 const [command, ...args] = tokens;
144 const id = Number(args[0]);
145 const p = state.processes.find((p) => p.id === id);
146 const hasId = ["PROCESS", "CLAIM", "REQUEST", "WHATIF", "RELEASE"].includes(command);
147 const v = args.slice(hasId ? 1 : 0).map(Number);
148 const error = (message: string): void => emit("error: " + message);
149 if (
150 ["PROCESS", "CLAIM", "ADD", "REMOVE", "REQUEST", "WHATIF", "RELEASE"].includes(command) &&
151 v.length !== state.total.length
152 ) {
153 error("wrong vector length");
154 return;
155 }
156 if (
157 ["CLAIM", "REQUEST", "WHATIF", "RELEASE", "FINISH", "ABORT", "CANCEL", "MAXGRANT"].includes(
158 command,
159 ) &&
160 !p
161 ) {
162 error(`unknown process P${id}`);
163 return;
164 }
165 switch (command) {
166 case "RESOURCES":
167 state = {
168 total: v.slice(),
169 available: v.slice(),
170 processes: [],
171 queue: [],
172 history: [],
173 };
174 emit("resources " + vec(v));
175 break;
176 case "MODE":
177 mode = args[0];
178 emit("mode " + mode);
179 break;
180 case "POLICY":
181 policy = args[0];
182 emit("policy " + policy);
183 break;
184 case "TRACE":
185 trace = args[0] === "ON";
186 emit("trace " + args[0]);
187 break;
188 case "AUTODETECT":
189 auto = args[0] === "ON";
190 emit("autodetect " + args[0]);
191 break;
192 case "PROCESS":
193 if (p) error(`duplicate process P${id}`);
194 else if (!le(v, state.total)) error("max exceeds total");
195 else {
196 state.processes.push({ id, max: v.slice(), alloc: v.map(() => 0), born: time });
197 emit(`process P${id} max ${vec(v)}`);
198 }
199 break;
200 case "CLAIM": {
201 if (waiting(id)) {
202 error(`process P${id} is blocked`);
203 break;
204 }
205 if (!le(p!.alloc, v)) {
206 error("claim below allocation");
207 break;
208 }
209 if (!le(v, state.total)) {
210 error("max exceeds total");
211 break;
212 }
213 const old = p!.max;
214 p!.max = v.slice();
215 if (mode === "AVOID" && check(false, trace).remaining.length) {
216 p!.max = old;
217 error("claim makes state unsafe");
218 break;
219 }
220 emit(`claim P${id} max ${vec(v)}`);
221 retry();
222 break;
223 }
224 case "ADD":
225 if (zero(v)) {
226 error("zero vector");
227 break;
228 }
229 state.total = plus(state.total, v);
230 state.available = plus(state.available, v);
231 emit(`added ${vec(v)} total ${vec(state.total)}`);
232 retry();
233 break;
234 case "REMOVE": {
235 if (zero(v)) {
236 error("zero vector");
237 break;
238 }
239 if (!le(v, state.available)) {
240 error("remove exceeds available");
241 break;
242 }
243 const total = minus(state.total, v);
244 if (total.some((x) => x === 0)) {
245 error("total must stay positive");
246 break;
247 }
248 if (state.processes.some((p) => !le(p.max, total))) {
249 error("max exceeds total");
250 break;
251 }
252 state.available = minus(state.available, v);
253 if (mode === "AVOID" && check(false, trace).remaining.length) {
254 state.available = plus(state.available, v);
255 error("removal makes state unsafe");
256 break;
257 }
258 state.total = total;
259 emit(`removed ${vec(v)} total ${vec(total)}`);
260 break;
261 }
262 case "REQUEST":
263 case "WHATIF": {
264 const d = decide(p!, v, true, trace);
265 if (command === "WHATIF") {
266 emit(
267 `whatif P${id} ${vec(v)}: ` +
268 (d.kind === "granted"
269 ? "granted" +
270 (mode === "AVOID" ? ", safe sequence " + names(d.sequence) : "")
271 : `${d.kind} (${d.reason})`),
272 );
273 } else if (d.kind === "rejected") error(d.reason);
274 else if (d.kind === "granted") {
275 allocate(p!, v);
276 emit(`granted P${id} ${vec(v)}`);
277 record("granted", id, v);
278 } else {
279 state.queue.push({ id, v: v.slice(), since: time });
280 emit(`blocked P${id} ${vec(v)}: ${d.reason}`);
281 record("blocked", id, v);
282 if (auto) {
283 const dead = check(true, trace).remaining;
284 if (dead.length) emit("deadlock detected: " + names(dead));
285 }
286 }
287 break;
288 }
289 case "RELEASE":
290 if (waiting(id)) error(`process P${id} is blocked`);
291 else if (zero(v)) error("zero vector");
292 else if (!le(v, p!.alloc)) error("release exceeds allocation");
293 else {
294 p!.alloc = minus(p!.alloc, v);
295 state.available = plus(state.available, v);
296 emit(`released P${id} ${vec(v)}`);
297 record("released", id, v);
298 retry();
299 }
300 break;
301 case "FINISH":
302 case "ABORT":
303 if (command === "FINISH" && waiting(id)) error(`process P${id} is blocked`);
304 else terminate(p!, command === "FINISH" ? "finished" : "aborted");
305 break;
306 case "CANCEL": {
307 const q = waiting(id);
308 if (!q) error(`process P${id} is not blocked`);
309 else {
310 state.queue = state.queue.filter((q) => q.id !== id);
311 emit(`cancelled P${id} ${vec(q.v)}`);
312 record("cancelled", id, q.v);
313 }
314 break;
315 }
316 case "MAXGRANT": {
317 if (waiting(id)) {
318 error(`process P${id} is blocked`);
319 break;
320 }
321 const result = need(p!).map((n, i) => {
322 let low = 0,
323 high = n;
324 while (low < high) {
325 const mid = Math.floor((low + high + 1) / 2),
326 q = state.total.map(() => 0);
327 q[i] = mid;
328 if (decide(p!, q, true, false).kind === "granted") low = mid;
329 else high = mid - 1;
330 }
331 return low;
332 });
333 emit(`maxgrant P${id}: ${vec(result)}`);
334 break;
335 }
336 case "SAFE": {
337 if (!state.processes.length) {
338 emit("safe: no processes");
339 break;
340 }
341 const c = check(false, trace);
342 emit(
343 c.remaining.length
344 ? `unsafe: ${names(c.remaining)} cannot finish`
345 : "safe: " + names(c.sequence),
346 );
347 break;
348 }
349 case "SEQUENCES": {
350 if (!state.processes.length) {
351 emit("safe sequences: no processes");
352 break;
353 }
354 let count = 0;
355 const first: number[][] = [];
356 function walk(rest: Process[], work: Vector, seq: number[]): void {
357 if (!rest.length) {
358 count++;
359 if (first.length < id) first.push(seq);
360 return;
361 }
362 for (const q of rest)
363 if (le(need(q), work))
364 walk(
365 rest.filter((p) => p !== q),
366 plus(work, q.alloc),
367 seq.concat(q.id),
368 );
369 }
370 walk(ordered(), state.available, []);
371 emit("safe sequences: " + count);
372 first.forEach((s) => emit(names(s)));
373 break;
374 }
375 case "DEFICIT": {
376 const original = state.available.slice();
377 const deficits = state.total.map((_, i) => {
378 let low = 0,
379 high = state.processes.reduce((s, p) => s + need(p)[i], 0);
380 state.available = original.slice();
381 state.available[i] += high;
382 if (check().remaining.length) return "-";
383 while (low < high) {
384 const mid = Math.floor((low + high) / 2);
385 state.available = original.slice();
386 state.available[i] += mid;
387 if (check().remaining.length) low = mid + 1;
388 else high = mid;
389 }
390 return String(low);
391 });
392 state.available = original;
393 emit(`deficit: (${deficits.join(",")})`);
394 break;
395 }
396 case "STATE": {
397 const widths = state.total.map((t) => String(t).length);
398 const rowName = widths.map((w, i) => String.fromCharCode(65 + i).padStart(w)).join(" ");
399 const cols = [10, 3, 4, 7].map((w) => Math.max(w, rowName.length));
400 const table = (v: Vector): string =>
401 v.map((x, i) => String(x).padStart(widths[i])).join(" ");
402 const row = (heading: string, values: string[]): void =>
403 emit(
404 (
405 heading.padEnd(9) +
406 cols.map((w, i) => " " + (values[i] ?? "").padEnd(w)).join("")
407 ).trimEnd(),
408 );
409 row("Process", ["Allocation", "Max", "Need", "Request"]);
410 row(
411 "",
412 cols.map(() => rowName),
413 );
414 ordered().forEach((p) =>
415 row(
416 `P${p.id}`,
417 [p.alloc, p.max, need(p), waiting(p.id)?.v ?? state.total.map(() => 0)].map(
418 table,
419 ),
420 ),
421 );
422 row("Available", [table(state.available)]);
423 row("Total", [table(state.total)]);
424 emit(
425 "blocked queue: " +
426 (state.queue.length
427 ? state.queue
428 .map((q) => `P${q.id} ${vec(q.v)} since t=${q.since}`)
429 .join(", ")
430 : "empty"),
431 );
432 break;
433 }
434 case "STATS": {
435 state.total.forEach((t, i) => {
436 const u = t - state.available[i],
437 percent = Math.floor((u * 1000) / t);
438 emit(
439 `${String.fromCharCode(65 + i)}: allocated ${u}/${t} (${Math.floor(percent / 10)}.${percent % 10}%), requested ${state.queue.reduce((s, q) => s + q.v[i], 0)}`,
440 );
441 });
442 const kinds = [
443 "granted",
444 "blocked",
445 "unblocked",
446 "released",
447 "finished",
448 "aborted",
449 "cancelled",
450 ];
451 emit(
452 "events: " +
453 kinds
454 .map((k) => `${k} ${state.history.filter((e) => e.kind === k).length}`)
455 .join(", "),
456 );
457 const waits = state.history.filter((e) => e.kind === "unblocked"),
458 total = waits.reduce((s, e) => s + e.wait, 0);
459 const average = waits.length ? Math.floor((total * 100) / waits.length) : 0;
460 emit(
461 waits.length
462 ? `waits: ${waits.length} unblocked, total ${total}, average ${Math.floor(average / 100)}.${String(average % 100).padStart(2, "0")}`
463 : "waits: none",
464 );
465 break;
466 }
467 case "DETECT": {
468 const dead = check(true, trace).remaining;
469 emit(dead.length ? "deadlocked: " + names(dead) : "no deadlock");
470 break;
471 }
472 case "VICTIMS": {
473 const dead = check(true).remaining;
474 if (!dead.length) {
475 emit("no deadlock");
476 break;
477 }
478 const candidates: number[][] = [];
479 for (let mask = 1; mask < 2 ** dead.length; mask++)
480 candidates.push(dead.filter((_, i) => mask & (1 << i)));
481 const cost = (ids: number[]): number =>
482 ids.reduce((s, id) => s + sum(state.processes.find((p) => p.id === id)!.alloc), 0);
483 candidates.sort((a, b) => {
484 if (a.length !== b.length) return a.length - b.length;
485 if (cost(a) !== cost(b)) return cost(a) - cost(b);
486 for (let i = 0; i < a.length; i++) if (a[i] !== b[i]) return a[i] - b[i];
487 return 0;
488 });
489 const saved = copy(state);
490 for (const ids of candidates) {
491 state = copy(saved);
492 for (const id of ids) {
493 const p = state.processes.find((p) => p.id === id)!;
494 state.available = plus(state.available, p.alloc);
495 }
496 state.processes = state.processes.filter((p) => !ids.includes(p.id));
497 state.queue = state.queue.filter((q) => !ids.includes(q.id));
498 if (!check(true).remaining.length) {
499 emit("victims: " + names(ids));
500 break;
501 }
502 }
503 state = saved;
504 break;
505 }
506 case "RECOVER": {
507 let count = 0;
508 if (args[0] === "ALL") {
509 const dead = check(true, trace).remaining;
510 for (const id of dead) {
511 terminate(
512 state.processes.find((p) => p.id === id)!,
513 "aborted",
514 false,
515 );
516 count++;
517 }
518 if (count) retry();
519 } else {
520 while (true) {
521 const dead = check(true, trace).remaining;
522 if (!dead.length) break;
523 const choices = ordered().filter((p) => dead.includes(p.id));
524 choices.sort((a, b) =>
525 policy === "YOUNGEST"
526 ? b.born - a.born
527 : (policy === "ALLOC"
528 ? sum(a.alloc) - sum(b.alloc)
529 : sum(need(b)) - sum(need(a))) || b.id - a.id,
530 );
531 terminate(choices[0], "aborted");
532 count++;
533 }
534 }
535 emit(count ? `recovered: ${count} aborted` : "no deadlock");
536 break;
537 }
538 case "WAITGRAPH":
539 case "GRAPH": {
540 const resource = command === "GRAPH";
541 if (resource && state.total.some((x) => x !== 1)) {
542 error("not single-instance");
543 break;
544 }
545 const graph = new Map<number, { to: number; label: number[] }[]>();
546 if (resource) {
547 state.total.forEach((_, i) =>
548 ordered().forEach((p) => {
549 if (p.alloc[i]) emit(`assign ${String.fromCharCode(65 + i)} -> P${p.id}`);
550 }),
551 );
552 for (const p of ordered()) {
553 const q = waiting(p.id);
554 if (!q) continue;
555 q.v.forEach((v, i) => {
556 if (!v) return;
557 emit(`request P${p.id} -> ${String.fromCharCode(65 + i)}`);
558 for (const holder of ordered())
559 if (holder.alloc[i]) {
560 const edges = graph.get(p.id) ?? [];
561 edges.push({ to: holder.id, label: [i] });
562 graph.set(p.id, edges);
563 }
564 });
565 }
566 for (const p of ordered())
567 need(p).forEach((v, i) => {
568 if (v && !(waiting(p.id)?.v[i] ?? 0))
569 emit(`claim P${p.id} -> ${String.fromCharCode(65 + i)}`);
570 });
571 } else {
572 let count = 0;
573 for (const p of ordered()) {
574 const q = waiting(p.id);
575 if (!q) continue;
576 for (const other of ordered())
577 if (p !== other) {
578 const label = q.v.flatMap((v, i) =>
579 v > state.available[i] && other.alloc[i] > 0 ? [i] : [],
580 );
581 if (!label.length) continue;
582 emit(
583 `P${p.id} -> P${other.id} on ${label.map((i) => String.fromCharCode(65 + i)).join(",")}`,
584 );
585 count++;
586 const edges = graph.get(p.id) ?? [];
587 edges.push({ to: other.id, label });
588 graph.set(p.id, edges);
589 }
590 }
591 if (!count) emit("no wait edges");
592 }
593 cycles(graph, resource);
594 break;
595 }
596 case "HISTORY": {
597 const events = state.history.filter(
598 (e) => !args.length || args[0] === String(e.id) || args[0] === e.kind,
599 );
600 events.forEach((e) =>
601 emit(
602 `t=${e.time} ${e.kind} P${e.id} ${vec(e.v)}${e.kind === "unblocked" ? " waited " + e.wait : ""}`,
603 ),
604 );
605 if (!events.length) emit("no events");
606 break;
607 }
608 case "SNAPSHOT":
609 snapshots.set(args[0], copy(state));
610 emit("saved " + args[0]);
611 break;
612 case "RESTORE": {
613 const saved = snapshots.get(args[0]);
614 if (!saved) error("unknown snapshot " + args[0]);
615 else {
616 state = copy(saved);
617 emit("restored " + args[0]);
618 }
619 break;
620 }
621 }
