52static int le(const int *a, const int *b) {
53 for (int j = 0; j < state.m; ++j)
54 if (a[j] > b[j])
55 return 0;
56 return 1;
57}
58static int zero(const int *v) {
59 for (int j = 0; j < state.m; ++j)
60 if (v[j])
61 return 0;
62 return 1;
63}
64static void change(int *a, const int *b, int sign) {
65 for (int j = 0; j < state.m; ++j)
66 a[j] += sign * b[j];
67}
68static int sum(const int *v) {
69 int s = 0;
70 for (int j = 0; j < state.m; ++j)
71 s += v[j];
72 return s;

74static void need(const Process *p, int *v) {
75 for (int j = 0; j < state.m; ++j)
76 v[j] = p->max[j] - p->allocation[j];
77}
78static int blocked(const State *s, int id) {
79 for (int i = 0; i < s->queued; ++i)
80 if (s->queue[i].id == id)
81 return i;
82 return -1;
83}
84static State copy_state(const State *s) {
85 State result = *s;
86 result.capacity = result.events;
87 result.history = result.events ? malloc(result.events * sizeof(Event)) : NULL;
88 if (result.events)
89 memcpy(result.history, s->history, result.events * sizeof(Event));
90 return result;
91}
92static void record(int kind, int id, const int *v, int wait) {
93 if (state.events == state.capacity) {
94 state.capacity = state.capacity ? state.capacity * 2 : 16;
95 state.history = realloc(state.history, state.capacity * sizeof(Event));
96 }
97 Event *e = &state.history[state.events++];
98 *e = (Event){.time = now, .kind = kind, .id = id, .wait = wait};
99 memcpy(e->vector, v, state.m * sizeof(int));
100}
101static Check check(const State *s, int detect, int verbose) {
102 Check c = {0};
103 int work[TYPES], done[IDS] = {0}, idle[IDS], idle_count = 0;
104 memcpy(work, s->available, sizeof(work));
105 for (int id = 0; id < IDS; ++id) {
106 if (!s->p[id].exists)
107 done[id] = 1;
108 else if (detect && zero(s->p[id].allocation)) {
109 done[id] = 1;
110 idle[idle_count++] = id;
111 }
112 }
113 const char *tag = detect ? "[detect]" : "[safety]";
114 if (verbose) {
115 printf("%s start work=", tag);
116 vector(work);
117 if (idle_count) {
118 printf(" idle ");
119 list(idle, idle_count);
120 }
121 putchar('\n');
122 }
123 for (;;) {
124 int chosen = -1;
125 for (int id = 0; id < IDS; ++id)
126 if (!done[id]) {
127 int v[TYPES] = {0}, q = blocked(s, id);
128 if (!detect)
129 need(&s->p[id], v);
130 else if (q >= 0)
131 memcpy(v, s->queue[q].vector, sizeof(v));
132 if (le(v, work)) {
133 chosen = id;
134 break;
135 }
136 }
137 if (chosen < 0)
138 break;
139 done[chosen] = 1;
140 c.sequence[c.count++] = chosen;
141 change(work, s->p[chosen].allocation, 1);
142 if (verbose) {
143 printf("%s P%d can finish, work=", tag, chosen);
144 vector(work);
145 putchar('\n');
146 }
147 }
148 for (int id = 0; id < IDS; ++id)
149 if (!done[id])
150 c.left[c.remaining++] = id;
151 if (verbose && c.remaining) {
152 printf("%s stuck work=", tag);
153 vector(work);
154 printf(" remaining ");
155 list(c.left, c.remaining);
156 putchar('\n');
157 }
158 return c;
159}
160static const char *rejection(int id, const int *v, char *message) {
161 if (blocked(&state, id) >= 0) {
162 sprintf(message, "process P%d is blocked", id);
163 return message;
164 }
165 if (zero(v))
166 return "zero vector";
167 int n[TYPES];
168 need(&state.p[id], n);
169 if (!le(v, n))
170 return "request exceeds need";
171 if (mode == 2) {
172 int last = -1;
173 for (int j = 0; j < state.m; ++j)
174 if (state.p[id].allocation[j])
175 last = j;
176 for (int j = 0; j <= last; ++j)
177 if (v[j])
178 return "resource order violation";
179 }
180 return NULL;
181}
182static const char *decision(int id, const int *v, int verbose, Check *result) {
183 if (!le(v, state.available))
184 return "insufficient resources";
185 if (mode != 0)
186 return NULL;
187 State tentative = state;
188 change(tentative.available, v, -1);
189 change(tentative.p[id].allocation, v, 1);
190 *result = check(&tentative, 0, verbose);
191 return result->remaining ? "unsafe state" : NULL;
192}
193static void allocate(int id, const int *v) {
194 change(state.available, v, -1);
195 change(state.p[id].allocation, v, 1);
196}
197static void remove_queue(int i) {
198 memmove(&state.queue[i], &state.queue[i + 1], (state.queued - i - 1) * sizeof(Waiting));
199 --state.queued;
200}
201static void retry(void) {
202 for (int i = 0; i < state.queued;) {
203 Waiting q = state.queue[i];
204 Check result;
205 if (decision(q.id, q.vector, tracing, &result)) {
206 ++i;
207 continue;
208 }
209 allocate(q.id, q.vector);
210 remove_queue(i);
211 printf("unblocked P%d ", q.id);
212 vector(q.vector);
213 putchar('\n');
214 record(2, q.id, q.vector, now - q.since);
215 }
216}
217static void terminate(int id, int kind, int do_retry) {
218 int v[TYPES];
219 memcpy(v, state.p[id].allocation, sizeof(v));
220 change(state.available, v, 1);
221 state.p[id].exists = 0;
222 int q = blocked(&state, id);
223 if (q >= 0)
224 remove_queue(q);
225 printf("%s P%d released ", kinds[kind], id);
226 vector(v);
227 putchar('\n');
228 record(kind, id, v, 0);
229 if (do_retry)
230 retry();
231}
232static int sequence_limit, sequence_count, sequence_rows[50][MAX_PROCESSES], sequence_saved,
233 sequence_size;
234static void permutations(int *path, int depth, int *used, int *work) {
235 if (depth == sequence_size) {
236 ++sequence_count;
237 if (sequence_saved < sequence_limit)
238 memcpy(sequence_rows[sequence_saved++], path, depth * sizeof(int));
239 return;
240 }
241 for (int id = 0; id < IDS; ++id)
242 if (state.p[id].exists && !used[id]) {
243 int n[TYPES];
244 need(&state.p[id], n);
245 if (!le(n, work))
246 continue;
247 used[id] = 1;
248 path[depth] = id;
249 change(work, state.p[id].allocation, 1);
250 permutations(path, depth + 1, used, work);
251 change(work, state.p[id].allocation, -1);
252 used[id] = 0;
253 }
254}
255static void sequences(int limit) {
256 sequence_size = 0;
257 for (int id = 0; id < IDS; ++id)
258 sequence_size += state.p[id].exists;
259 if (!sequence_size) {
260 puts("safe sequences: no processes");
261 return;
262 }
263 sequence_count = sequence_saved = 0;
264 sequence_limit = limit;
265 int path[MAX_PROCESSES], used[IDS] = {0}, work[TYPES];
266 memcpy(work, state.available, sizeof(work));
267 permutations(path, 0, used, work);
268 printf("safe sequences: %d\n", sequence_count);
269 for (int i = 0; i < sequence_saved; ++i) {
270 list(sequence_rows[i], sequence_size);
271 putchar('\n');
272 }
273}
274static void victims(void) {
275 Check dead = check(&state, 1, 0);
276 if (!dead.remaining) {
277 puts("no deadlock");
278 return;
279 }
280 int best = 0, best_count = IDS, best_sum = 0;
281 for (int mask = 1; mask < (1 << dead.remaining); ++mask) {
282 State candidate = state;
283 int count = 0, amount = 0;
284 for (int i = 0; i < dead.remaining; ++i)
285 if (mask & (1 << i)) {
286 int id = dead.left[i];
287 ++count;
288 amount += sum(candidate.p[id].allocation);
289 change(candidate.available, candidate.p[id].allocation, 1);
290 candidate.p[id].exists = 0;
291 int q = blocked(&candidate, id);
292 if (q >= 0) {
293 memmove(&candidate.queue[q], &candidate.queue[q + 1],
294 (candidate.queued - q - 1) * sizeof(Waiting));
295 --candidate.queued;
296 }
297 }
298 int preferred = count < best_count || (count == best_count && amount < best_sum);
299 if (count == best_count && amount == best_sum) {
300 for (int i = 0; i < dead.remaining; ++i)
301 if ((mask & (1 << i)) != (best & (1 << i))) {
302 preferred = (mask & (1 << i)) != 0;
303 break;
304 }
305 }
306 if (preferred && !check(&candidate, 1, 0).remaining) {
307 best = mask;
308 best_count = count;
309 best_sum = amount;
310 }
311 }
312 int ids[IDS], n = 0;
313 for (int i = 0; i < dead.remaining; ++i)
314 if (best & (1 << i))
315 ids[n++] = dead.left[i];
316 printf("victims: ");
317 list(ids, n);
318 putchar('\n');
319}
320static int wait_edges[IDS][IDS], resource_edges[IDS][TYPES], owner[TYPES], visited[IDS], path[IDS],
321 type_path[IDS], graph_resource, cycle_count;
322static void cycles(int start, int at, int depth) {
323 for (int element = 0; element < (graph_resource ? state.m : IDS); ++element) {
324 int next = graph_resource ? owner[element] : element;
325 if (graph_resource ? !resource_edges[at][element] || next < 0 : !wait_edges[at][next])
326 continue;
327 if (next == start && (graph_resource || depth >= 2)) {
328 printf("cycle: P%d", start);
329 for (int i = 1; i < depth; ++i) {
330 if (graph_resource)
331 printf(" -> %c", 'A' + type_path[i - 1]);
332 printf(" -> P%d", path[i]);
333 }
334 if (graph_resource)
335 printf(" -> %c", 'A' + element);
336 printf(" -> P%d\n", start);
337 ++cycle_count;
338 } else if (next > start && !visited[next]) {
339 visited[next] = 1;
340 path[depth] = next;
341 type_path[depth - 1] = element;
342 cycles(start, next, depth + 1);
343 visited[next] = 0;
344 }
345 }
346}
347static void graph(int resource) {
348 graph_resource = resource;
349 cycle_count = 0;
350 memset(wait_edges, 0, sizeof(wait_edges));
351 memset(resource_edges, 0, sizeof(resource_edges));
352 if (resource) {
353 for (int j = 0; j < state.m; ++j)
354 if (state.total[j] != 1) {
355 error("not single-instance");
356 return;
357 }
358 for (int j = 0; j < state.m; ++j) {
359 owner[j] = -1;
360 for (int id = 0; id < IDS; ++id)
361 if (state.p[id].exists && state.p[id].allocation[j]) {
362 owner[j] = id;
363 printf("assign %c -> P%d\n", 'A' + j, id);
364 }
365 }
366 for (int id = 0; id < IDS; ++id)
367 if (state.p[id].exists) {
368 int q = blocked(&state, id);
369 for (int j = 0; j < state.m; ++j)
370 if (q >= 0 && state.queue[q].vector[j]) {
371 resource_edges[id][j] = 1;
372 printf("request P%d -> %c\n", id, 'A' + j);
373 }
374 }
375 for (int id = 0; id < IDS; ++id)
376 if (state.p[id].exists) {
377 int n[TYPES];
378 need(&state.p[id], n);
379 for (int j = 0; j < state.m; ++j)
380 if (n[j] && !resource_edges[id][j])
381 printf("claim P%d -> %c\n", id, 'A' + j);
382 }
383 } else {
384 int edges = 0;
385 for (int id = 0; id < IDS; ++id) {
386 int q = blocked(&state, id);
387 if (q < 0)
388 continue;
389 for (int other = 0; other < IDS; ++other)
390 if (other != id && state.p[other].exists) {
391 int labels[TYPES], n = 0;
392 for (int j = 0; j < state.m; ++j)
393 if (state.queue[q].vector[j] > state.available[j] &&
394 state.p[other].allocation[j])
395 labels[n++] = j;
396 if (!n)
397 continue;
398 wait_edges[id][other] = 1;
399 ++edges;
400 printf("P%d -> P%d on ", id, other);
401 for (int i = 0; i < n; ++i)
402 printf("%s%c", i ? "," : "", 'A' + labels[i]);
403 putchar('\n');
404 }
405 }
406 if (!edges)
407 puts("no wait edges");
408 }
409 for (int id = 0; id < IDS; ++id)
410 if (state.p[id].exists) {
411 memset(visited, 0, sizeof(visited));
412 visited[id] = 1;
413 path[0] = id;
414 cycles(id, id, 1);
415 }
416 if (!cycle_count)
417 puts("no cycles");
418}
419static void table_form(char *out, const int *v, const int *widths) {
420 int offset = 0;
421 for (int j = 0; j < state.m; ++j)
422 offset += sprintf(out + offset, "%s%*d", j ? " " : "", widths[j], v[j]);
423}
424static void row(const char *heading, char cells[4][32], const int *widths) {
425 char line[256];
426 int offset = sprintf(line, "%-9s", heading);
427 for (int i = 0; i < 4; ++i)
428 offset += sprintf(line + offset, " %-*s", widths[i], cells[i]);
429 while (offset && line[offset - 1] == ' ')
430 --offset;
431 line[offset] = 0;
432 puts(line);
433}
434static void table(void) {
435 int widths[TYPES], columns[] = {10, 3, 4, 7}, length = 0;
436 char names[32] = "", cells[4][32];
437 for (int j = 0; j < state.m; ++j) {
438 widths[j] = state.total[j] < 10 ? 1 : 2;
439 length += sprintf(names + length, "%s%*c", j ? " " : "", widths[j], 'A' + j);
440 }
441 for (int i = 0; i < 4; ++i)
442 if (columns[i] < length)
443 columns[i] = length;
444 strcpy(cells[0], "Allocation");
445 strcpy(cells[1], "Max");
446 strcpy(cells[2], "Need");
447 strcpy(cells[3], "Request");
448 row("Process", cells, columns);
449 for (int i = 0; i < 4; ++i)
450 strcpy(cells[i], names);
451 row("", cells, columns);
452 for (int id = 0; id < IDS; ++id)
453 if (state.p[id].exists) {
454 int n[TYPES], z[TYPES] = {0}, q = blocked(&state, id);
455 char heading[16];
456 need(&state.p[id], n);
457 sprintf(heading, "P%d", id);
458 table_form(cells[0], state.p[id].allocation, widths);
459 table_form(cells[1], state.p[id].max, widths);
460 table_form(cells[2], n, widths);
461 table_form(cells[3], q >= 0 ? state.queue[q].vector : z, widths);
462 row(heading, cells, columns);
463 }
464 for (int i = 1; i < 4; ++i)
465 cells[i][0] = 0;
466 table_form(cells[0], state.available, widths);
467 row("Available", cells, columns);
468 table_form(cells[0], state.total, widths);
469 row("Total", cells, columns);
470 printf("blocked queue: ");
471 if (!state.queued)
472 printf("empty");
473 for (int i = 0; i < state.queued; ++i) {
474 Waiting q = state.queue[i];
475 printf("%sP%d ", i ? ", " : "", q.id);
476 vector(q.vector);
477 printf(" since t=%d", q.since);
478 }
479 putchar('\n');
480}
481int main(void) {
482 char line[256], *words[16];
483 int n;
484 if (!fgets(line, sizeof(line), stdin))
485 return 0;
486 n = atoi(line);
487 for (now = 1; now <= n; ++now) {
488 if (!fgets(line, sizeof(line), stdin))
489 break;
490 int argc = 0;
491 for (char *p = strtok(line, " \r\n"); p; p = strtok(NULL, " \r\n"))
492 words[argc++] = p;
493 char *cmd = words[0];
494 if (!strcmp(cmd, "RESOURCES")) {
495 free(state.history);
496 memset(&state, 0, sizeof(state));
497 state.m = argc - 1;
498 for (int j = 0; j < state.m; ++j)
499 state.total[j] = state.available[j] = atoi(words[j + 1]);
500 printf("resources ");
501 vector(state.total);
502 putchar('\n');
503 continue;
504 }
505 if (!strcmp(cmd, "MODE") || !strcmp(cmd, "POLICY") || !strcmp(cmd, "TRACE") ||
506 !strcmp(cmd, "AUTODETECT")) {
507 const char *label;
508 if (!strcmp(cmd, "MODE")) {
509 mode = !strcmp(words[1], "AVOID") ? 0 : !strcmp(words[1], "DETECT") ? 1 : 2;
510 label = "mode";
511 } else if (!strcmp(cmd, "POLICY")) {
512 policy = !strcmp(words[1], "ALLOC") ? 0 : !strcmp(words[1], "NEED") ? 1 : 2;
513 label = "policy";
514 } else if (!strcmp(cmd, "TRACE")) {
515 tracing = !strcmp(words[1], "ON");
516 label = "trace";
517 } else {
518 automatic = !strcmp(words[1], "ON");
519 label = "autodetect";
520 }
521 printf("%s %s\n", label, words[1]);
522 continue;
523 }
524 if (!strcmp(cmd, "SNAPSHOT") || !strcmp(cmd, "RESTORE")) {
525 Snapshot *snap = snapshots;
526 while (snap && strcmp(snap->name, words[1]))
527 snap = snap->next;
528 if (!strcmp(cmd, "SNAPSHOT")) {
529 if (!snap) {
530 snap = calloc(1, sizeof(*snap));
531 strcpy(snap->name, words[1]);
532 snap->next = snapshots;
533 snapshots = snap;
534 } else
535 free(snap->state.history);
536 snap->state = copy_state(&state);
537 printf("saved %s\n", words[1]);
538 } else if (!snap)
539 printf("error: unknown snapshot %s\n", words[1]);
540 else {
541 free(state.history);
542 state = copy_state(&snap->state);
543 printf("restored %s\n", words[1]);
544 }
545 continue;
546 }
547 if (!strcmp(cmd, "STATE")) {
548 table();
549 continue;
550 }
551 if (!strcmp(cmd, "GRAPH") || !strcmp(cmd, "WAITGRAPH")) {
552 graph(!strcmp(cmd, "GRAPH"));
553 continue;
554 }
555 if (!strcmp(cmd, "SEQUENCES")) {
556 sequences(atoi(words[1]));
557 continue;
558 }
559 if (!strcmp(cmd, "VICTIMS")) {
560 victims();
561 continue;
562 }
563 if (!strcmp(cmd, "SAFE") || !strcmp(cmd, "DETECT")) {
564 int count = 0, detect = !strcmp(cmd, "DETECT");
565 for (int id = 0; id < IDS; ++id)
566 count += state.p[id].exists;
567 if (!detect && !count) {
568 puts("safe: no processes");
569 continue;
570 }
571 Check c = check(&state, detect, tracing);
572 if (detect && !c.remaining)
573 puts("no deadlock");
574 else if (detect) {
575 printf("deadlocked: ");
576 list(c.left, c.remaining);
577 putchar('\n');
578 } else if (c.remaining) {
579 printf("unsafe: ");
580 list(c.left, c.remaining);
581 puts(" cannot finish");
582 } else {
583 printf("safe: ");
584 list(c.sequence, c.count);
585 putchar('\n');
586 }
587 continue;
588 }
589 if (!strcmp(cmd, "RECOVER")) {
590 int count = 0;
591 for (;;) {
592 Check c = check(&state, 1, tracing);
593 if (!c.remaining)
594 break;
595 if (argc > 1) {
596 for (int i = 0; i < c.remaining; ++i)
597 terminate(c.left[i], 5, 0);
598 count += c.remaining;
599 retry();
600 break;
601 }
602 int victim = c.left[0], best;
603 int v[TYPES];
604 need(&state.p[victim], v);
605 best = policy == 0 ? -sum(state.p[victim].allocation)
606 : policy == 1 ? sum(v)
607 : state.p[victim].created;
608 for (int i = 1; i < c.remaining; ++i) {
609 int id = c.left[i];
610 need(&state.p[id], v);
611 int rank = policy == 0 ? -sum(state.p[id].allocation)
612 : policy == 1 ? sum(v)
613 : state.p[id].created;
614 if (rank > best || (rank == best && id > victim)) {
615 best = rank;
616 victim = id;
617 }
618 }
619 terminate(victim, 5, 1);
620 ++count;
621 }
622 if (count)
623 printf("recovered: %d aborted\n", count);
624 else
625 puts("no deadlock");
626 continue;
627 }
628 if (!strcmp(cmd, "DEFICIT")) {
629 printf("deficit: (");
630 for (int j = 0; j < state.m; ++j) {
631 int bound = 0, found = -1;
632 for (int id = 0; id < IDS; ++id)
633 if (state.p[id].exists)
634 bound += state.p[id].max[j] - state.p[id].allocation[j];
635 for (int x = 0; x <= bound; ++x) {
636 State candidate = state;
637 candidate.available[j] += x;
638 if (!check(&candidate, 0, 0).remaining) {
639 found = x;
640 break;
641 }
642 }
643 if (j)
644 putchar(',');
645 if (found < 0)
646 putchar('-');
647 else
648 printf("%d", found);
649 }
650 puts(")");
651 continue;
652 }
653 if (!strcmp(cmd, "HISTORY")) {
654 int count = 0;
655 for (size_t i = 0; i < state.events; ++i) {
656 Event e = state.history[i];
657 if (argc > 1 && strcmp(words[1], kinds[e.kind]) &&
658 !(words[1][0] >= '0' && words[1][0] <= '9' && atoi(words[1]) == e.id))
659 continue;
660 printf("t=%d %s P%d ", e.time, kinds[e.kind], e.id);
661 vector(e.vector);
662 if (e.kind == 2)
663 printf(" waited %d", e.wait);
664 putchar('\n');
665 ++count;
666 }
667 if (!count)
668 puts("no events");
669 continue;
670 }
671 if (!strcmp(cmd, "STATS")) {
672 int counts[7] = {0}, waits = 0;
673 for (int j = 0; j < state.m; ++j) {
674 int used = state.total[j] - state.available[j],
675 percent = used * 1000 / state.total[j], requested = 0;
676 for (int i = 0; i < state.queued; ++i)
677 requested += state.queue[i].vector[j];
678 printf("%c: allocated %d/%d (%d.%d%%), requested %d\n", 'A' + j, used,
679 state.total[j], percent / 10, percent % 10, requested);
680 }
681 for (size_t i = 0; i < state.events; ++i) {
682 ++counts[state.history[i].kind];
683 if (state.history[i].kind == 2)
684 waits += state.history[i].wait;
685 }
686 printf("events: ");
687 for (int k = 0; k < 7; ++k)
688 printf("%s%s %d", k ? ", " : "", kinds[k], counts[k]);
689 putchar('\n');
690 if (!counts[2])
691 puts("waits: none");
692 else {
693 int average = waits * 100 / counts[2];
694 printf("waits: %d unblocked, total %d, average %d.%02d\n", counts[2], waits,
695 average / 100, average % 100);
696 }
697 continue;
698 }
699 int resource = !strcmp(cmd, "ADD") || !strcmp(cmd, "REMOVE");
700 int has_vector = resource || !strcmp(cmd, "PROCESS") || !strcmp(cmd, "CLAIM") ||
701 !strcmp(cmd, "REQUEST") || !strcmp(cmd, "WHATIF") ||
702 !strcmp(cmd, "RELEASE");
703 int id = resource ? 0 : atoi(words[1]), v[TYPES] = {0};
704 if (has_vector && argc - (resource ? 1 : 2) != state.m) {
705 error("wrong vector length");
706 continue;
707 }
708 if (has_vector)
709 for (int j = 0; j < state.m; ++j)
710 v[j] = atoi(words[j + (resource ? 1 : 2)]);
711 if (!resource && strcmp(cmd, "PROCESS") && !state.p[id].exists) {
712 printf("error: unknown process P%d\n", id);
713 continue;
714 }
715 if (!strcmp(cmd, "PROCESS")) {
716 if (state.p[id].exists)
717 printf("error: duplicate process P%d\n", id);
718 else if (!le(v, state.total))
719 error("max exceeds total");
720 else {
721 state.p[id] = (Process){.created = now, .exists = 1};
722 memcpy(state.p[id].max, v, sizeof(v));
723 printf("process P%d max ", id);
724 vector(v);
725 putchar('\n');
726 }
727 continue;
728 }
729 if (!strcmp(cmd, "REQUEST") || !strcmp(cmd, "WHATIF")) {
730 char message[64];
731 const char *reason = rejection(id, v, message);
732 Check c = {0};
733 if (reason) {
734 if (!strcmp(cmd, "REQUEST"))
735 error(reason);
736 else {
737 printf("whatif P%d ", id);
738 vector(v);
739 printf(": rejected (%s)\n", reason);
740 }
741 continue;
742 }
743 reason = decision(id, v, tracing, &c);
744 if (!strcmp(cmd, "WHATIF")) {
745 printf("whatif P%d ", id);
746 vector(v);
747 if (reason)
748 printf(": blocked (%s)\n", reason);
749 else {
750 printf(": granted");
751 if (!mode) {
752 printf(", safe sequence ");
753 list(c.sequence, c.count);
754 }
755 putchar('\n');
756 }
757 } else if (!reason) {
758 allocate(id, v);
759 printf("granted P%d ", id);
760 vector(v);
761 putchar('\n');
762 record(0, id, v, 0);
763 } else {
764 Waiting *q = &state.queue[state.queued++];
765 *q = (Waiting){.id = id, .since = now};
766 memcpy(q->vector, v, sizeof(v));
767 printf("blocked P%d ", id);
768 vector(v);
769 printf(": %s\n", reason);
770 record(1, id, v, 0);
771 if (automatic) {
772 Check d = check(&state, 1, tracing);
773 if (d.remaining) {
774 printf("deadlock detected: ");
775 list(d.left, d.remaining);
776 putchar('\n');
777 }
778 }
779 }
780 continue;
781 }
782 if (!resource && strcmp(cmd, "ABORT") && strcmp(cmd, "CANCEL") &&
783 blocked(&state, id) >= 0) {
784 printf("error: process P%d is blocked\n", id);
785 continue;
786 }
787 if (!strcmp(cmd, "FINISH") || !strcmp(cmd, "ABORT")) {
788 terminate(id, !strcmp(cmd, "FINISH") ? 4 : 5, 1);
789 continue;
790 }
791 if (!strcmp(cmd, "CANCEL")) {
792 int q = blocked(&state, id);
793 if (q < 0)
794 printf("error: process P%d is not blocked\n", id);
795 else {
796 Waiting w = state.queue[q];
797 remove_queue(q);
798 printf("cancelled P%d ", id);
799 vector(w.vector);
800 putchar('\n');
801 record(6, id, w.vector, 0);
802 }
803 continue;
804 }
805 if (!strcmp(cmd, "MAXGRANT")) {
806 int answer[TYPES] = {0}, nvec[TYPES];
807 need(&state.p[id], nvec);
808 for (int j = 0; j < state.m; ++j)
809 for (int x = nvec[j]; x > 0; --x) {
810 int request[TYPES] = {0};
811 request[j] = x;
812 char message[64];
813 Check c;
814 if (!rejection(id, request, message) && !decision(id, request, 0, &c)) {
815 answer[j] = x;
816 break;
817 }
818 }
819 printf("maxgrant P%d: ", id);
820 vector(answer);
821 putchar('\n');
822 continue;
823 }
824 if (!strcmp(cmd, "CLAIM")) {
825 if (!le(state.p[id].allocation, v)) {
826 error("claim below allocation");
827 continue;
828 }
829 if (!le(v, state.total)) {
830 error("max exceeds total");
831 continue;
832 }
833 State candidate = state;
834 memcpy(candidate.p[id].max, v, sizeof(v));
835 if (!mode && check(&candidate, 0, tracing).remaining) {
836 error("claim makes state unsafe");
837 continue;
838 }
839 memcpy(state.p[id].max, v, sizeof(v));
840 printf("claim P%d max ", id);
841 vector(v);
842 putchar('\n');
843 retry();
844 continue;
845 }
846 if (zero(v)) {
847 error("zero vector");
848 continue;
849 }
850 if (!strcmp(cmd, "RELEASE")) {
851 if (!le(v, state.p[id].allocation)) {
852 error("release exceeds allocation");
853 continue;
854 }
855 change(state.p[id].allocation, v, -1);
856 change(state.available, v, 1);
857 printf("released P%d ", id);
858 vector(v);
859 putchar('\n');
860 record(3, id, v, 0);
861 retry();
862 } else if (!strcmp(cmd, "ADD")) {
863 change(state.total, v, 1);
864 change(state.available, v, 1);
865 printf("added ");
866 vector(v);
867 printf(" total ");
868 vector(state.total);
869 putchar('\n');
870 retry();
871 } else {
872 if (!le(v, state.available)) {
873 error("remove exceeds available");
874 continue;
875 }
876 State candidate = state;
877 change(candidate.total, v, -1);
878 change(candidate.available, v, -1);
879 int bad = 0;
880 for (int j = 0; j < state.m; ++j)
881 if (!candidate.total[j])
882 bad = 1;
883 if (bad) {
884 error("total must stay positive");
885 continue;
886 }
887 for (int p = 0; p < IDS; ++p)
888 if (state.p[p].exists && !le(state.p[p].max, candidate.total))
889 bad = 1;
890 if (bad) {
891 error("max exceeds total");
892 continue;
893 }
894 if (!mode && check(&candidate, 0, tracing).remaining) {
895 error("removal makes state unsafe");
896 continue;
897 }
898 state = candidate;
899 printf("removed ");
900 vector(v);
901 printf(" total ");
902 vector(state.total);
903 putchar('\n');
904 }
905 }
906 free(state.history);
907 while (snapshots) {
908 Snapshot *next = snapshots->next;
909 free(snapshots->state.history);
910 free(snapshots);
911 snapshots = next;
912 }
913 return 0;
914}
