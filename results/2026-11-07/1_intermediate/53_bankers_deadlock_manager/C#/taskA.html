1using System;
2using System.Collections.Generic;
3using System.Linq;
4
5record Process(int Id, int[] Max, int[] Allocation, int Created);
6record Waiting(int Id, int[] Vector, int Since);
7record Event(int Time, string Kind, int Id, int[] Vector, int Wait = 0);
8class State {
9 public int[] Total = Array.Empty<int>(), Available = Array.Empty<int>();
10 public SortedDictionary<int, Process> Processes = new();
11 public List<Waiting> Queue = new();
12 public List<Event> History = new();
13 public State Copy() => new State {
14 Total = (int[])Total.Clone(), Available = (int[])Available.Clone(),
15 Processes = new(Processes.ToDictionary(
16 x => x.Key, x => x.Value with { Max = (int[])x.Value.Max.Clone(),
17 Allocation = (int[])x.Value.Allocation.Clone() })),
18 Queue = Queue.Select(x => x with { Vector = (int[])x.Vector.Clone() }).ToList(),
19 History = History.Select(x => x with { Vector = (int[])x.Vector.Clone() }).ToList()
20 };
21}
22class Program {
23 static State s = new();
24 static readonly Dictionary<string, State> snapshots = new();
25 static string mode = "AVOID", policy = "ALLOC";
26 static bool trace, auto;
27 static int time;
28 static readonly string[] kinds = { "granted", "blocked", "unblocked", "released",
29 "finished", "aborted", "cancelled" };
30 static string V(int[] v) => "(" + string.Join(",", v) + ")";
31 static string L(IEnumerable<int> ids) => string.Join(" ", ids.Select(x => "P" + x));
32 static bool Le(int[] a, int[] b) => a.Zip(b).All(x => x.First <= x.Second);
33 static int[] Add(int[] a, int[] b) => a.Zip(b, (x, y) => x + y).ToArray();
34 static int[] Sub(int[] a, int[] b) => a.Zip(b, (x, y) => x - y).ToArray();
35 static int[] Need(Process p) => Sub(p.Max, p.Allocation);
36 static int[] Zero() => new int[s.Total.Length];
37 static Waiting? Block(int id) => s.Queue.FirstOrDefault(x => x.Id == id);
38 static void Error(string message) => Console.WriteLine("error: " + message);
39 static (List<int> Sequence, List<int> Left) Check(State state, bool detect, bool verbose) {
40 int[] work = (int[])state.Available.Clone();
41 var idle = state.Processes.Values.Where(p => detect && p.Allocation.All(x => x == 0))
42 .Select(p => p.Id)
43 .ToList();
44 var left = state.Processes.Keys.Except(idle).ToList();
45 var sequence = new List<int>();
46 string tag = detect ? "[detect]" : "[safety]";
47 if (verbose)
48 Console.WriteLine(tag + " start work=" + V(work) +
49 (idle.Count > 0 ? " idle " + L(idle) : ""));
50 while (true) {
51 int index = left.FindIndex(
52 id => Le(detect ? state.Queue.FirstOrDefault(q => q.Id == id)?.Vector ?? new int
53 [work.Length]
54 : Need(state.Processes[id]),
55 work));
56 if (index < 0)
57 break;
58 int id = left[index];
59 left.RemoveAt(index);
60 sequence.Add(id);
61 work = Add(work, state.Processes[id].Allocation);
62 if (verbose)
63 Console.WriteLine(tag + " P" + id + " can finish, work=" + V(work));
64 }
65 if (verbose && left.Count > 0)
66 Console.WriteLine(tag + " stuck work=" + V(work) + " remaining " + L(left));
67 return (sequence, left);
68 }
69 static string Reject(int id, int[] v) {
70 Process p = s.Processes[id];
71 if (Block(id) != null)
72 return "process P" + id + " is blocked";
73 if (v.All(x => x == 0))
74 return "zero vector";
75 if (!Le(v, Need(p)))
76 return "request exceeds need";
77 int last = Array.FindLastIndex(p.Allocation, x => x > 0);
78 if (mode == "ORDER" && last >= 0 && v.Take(last + 1).Any(x => x > 0))
79 return "resource order violation";
80 return "";
81 }
82 static (string Reason, List<int> Sequence) Decision(int id, int[] v, bool verbose) {
83 if (!Le(v, s.Available))
84 return ("insufficient resources", new());
85 if (mode != "AVOID")
86 return ("", new());
87 State candidate = s.Copy();
88 candidate.Available = Sub(candidate.Available, v);
89 Process p = candidate.Processes[id];
90 candidate.Processes[id] = p with { Allocation = Add(p.Allocation, v) };
91 var check = Check(candidate, false, verbose);
92 return (check.Left.Count == 0 ? "" : "unsafe state", check.Sequence);
93 }
94 static void Allocate(int id, int[] v) {
95 Process p = s.Processes[id];
96 s.Processes[id] = p with { Allocation = Add(p.Allocation, v) };
97 s.Available = Sub(s.Available, v);
98 }
99 static void Record(string kind, int id, int[] v,
100 int wait = 0) => s.History.Add(new(time, kind, id, (int[])v.Clone(), wait));
101 static void Retry() {
102 foreach (Waiting q in s.Queue.ToArray()) {
103 if (Decision(q.Id, q.Vector, trace).Reason != "")
104 continue;
105 Allocate(q.Id, q.Vector);
106 s.Queue.Remove(q);
107 Console.WriteLine("unblocked P" + q.Id + " " + V(q.Vector));
108 Record("unblocked", q.Id, q.Vector, time - q.Since);
109 }
110 }
111 static void Terminate(int id, string kind, bool retry) {
112 int[] u = s.Processes[id].Allocation;
113 s.Available = Add(s.Available, u);
114 s.Processes.Remove(id);
115 s.Queue.RemoveAll(q => q.Id == id);
116 Console.WriteLine(kind + " P" + id + " released " + V(u));
117 Record(kind, id, u);
118 if (retry)
119 Retry();
120 }
121 static void Sequences(int limit) {
122 if (s.Processes.Count == 0) {
123 Console.WriteLine("safe sequences: no processes");
124 return;
125 }
126 var rows = new List<string>();
127 int count = 0;
128 void Search(List<int> path, List<int> left, int[] work) {
129 if (left.Count == 0) {
130 count++;
131 if (rows.Count < limit)
132 rows.Add(L(path));
133 return;
134 }
135 foreach (int id in left)
136 if (Le(Need(s.Processes[id]), work))
137 Search(path.Append(id).ToList(), left.Where(x => x != id).ToList(),
138 Add(work, s.Processes[id].Allocation));
139 }
140 Search(new(), s.Processes.Keys.ToList(), s.Available);
141 Console.WriteLine("safe sequences: " + count);
142 rows.ForEach(Console.WriteLine);
143 }
144 static void Victims() {
145 List<int> dead = Check(s, true, false).Left;
146 if (dead.Count == 0) {
147 Console.WriteLine("no deadlock");
148 return;
149 }
150 var sets = new List<List<int>>();
151 for (int mask = 1; mask < (1 << dead.Count); mask++)
152 sets.Add(dead.Where((id, i) => (mask & (1 << i)) != 0).ToList());
153 sets.Sort((a, b) => {
154 int c = a.Count.CompareTo(b.Count);
155 if (c != 0)
156 return c;
157 c = a.Sum(id => s.Processes[id].Allocation.Sum())
158 .CompareTo(b.Sum(id => s.Processes[id].Allocation.Sum()));
159 if (c != 0)
160 return c;
161 for (int i = 0; i < a.Count; i++)
162 if (a[i] != b[i])
163 return a[i].CompareTo(b[i]);
164 return 0;
165 });
166 foreach (var set in sets) {
167 State candidate = s.Copy();
168 foreach (int id in set) {
169 candidate.Available = Add(candidate.Available, candidate.Processes[id].Allocation);
170 candidate.Processes.Remove(id);
171 candidate.Queue.RemoveAll(q => q.Id == id);
172 }
173 if (Check(candidate, true, false).Left.Count == 0) {
174 Console.WriteLine("victims: " + L(set));
175 return;
176 }
177 }
178 }
179 static void Graph(bool resource) {
180 if (resource && s.Total.Any(x => x != 1)) {
181 Error("not single-instance");
182 return;
183 }
184 int m = s.Total.Length;
185 var edges = new SortedDictionary<int, List<(int Type, int To)>>();
186 foreach (int id in s.Processes.Keys)
187 edges[id] = new();
188 if (resource) {
189 for (int j = 0; j < m; j++)
190 foreach (Process p in s.Processes.Values)
191 if (p.Allocation[j] > 0)
192 Console.WriteLine("assign " + (char)('A' + j) + " -> P" + p.Id);
193 foreach (Process p in s.Processes.Values)
194 for (int j = 0; j < m; j++)
195 if ((Block(p.Id)?.Vector[j] ?? 0) > 0) {
196 Console.WriteLine("request P" + p.Id + " -> " + (char)('A' + j));
197 foreach (Process other in s.Processes.Values)
198 if (other.Allocation[j] > 0)
199 edges[p.Id].Add((j, other.Id));
200 }
201 foreach (Process p in s.Processes.Values)
202 for (int j = 0; j < m; j++)
203 if (Need(p)[j] > 0 && (Block(p.Id)?.Vector[j] ?? 0) == 0)
204 Console.WriteLine("claim P" + p.Id + " -> " + (char)('A' + j));
205 } else {
206 int count = 0;
207 foreach (Process p in s.Processes.Values) {
208 Waiting? q = Block(p.Id);
209 if (q == null)
210 continue;
211 foreach (Process other in s.Processes.Values) {
212 if (p.Id == other.Id)
213 continue;
214 var labels =
215 Enumerable.Range(0, m)
216 .Where(j => q.Vector[j] > s.Available[j] && other.Allocation[j] > 0)
217 .ToList();
218 if (labels.Count == 0)
219 continue;
220 count++;
221 edges[p.Id].Add((-1, other.Id));
222 Console.WriteLine("P" + p.Id + " -> P" + other.Id + " on " +
223 string.Join(",", labels.Select(j => (char)('A' + j))));
224 }
225 }
226 if (count == 0)
227 Console.WriteLine("no wait edges");
228 }
229 bool any = false;
230 foreach (int start in s.Processes.Keys) {
231 void Visit(int at, List<int> seen, List<string> path) {
232 foreach (var edge in edges[at]
233 .OrderBy(e => resource ? e.Type : e.To)
234 .ThenBy(e => e.To)) {
235 var next = new List<string>(path);
236 if (resource)
237 next.Add(((char)('A' + edge.Type)).ToString());
238 next.Add("P" + edge.To);
239 if (edge.To == start && (resource || seen.Count >= 2)) {
240 Console.WriteLine("cycle: " + string.Join(" -> ", next));
241 any = true;
242 } else if (edge.To > start && !seen.Contains(edge.To))
243 Visit(edge.To, seen.Append(edge.To).ToList(), next);
244 }
245 }
246 Visit(start, new() { start }, new() { "P" + start });
247 }
248 if (!any)
249 Console.WriteLine("no cycles");
250 }
251 static void Table() {
252 int[] widths = s.Total.Select(x => x.ToString().Length).ToArray();
253 string Form(int[] v) =>
254 string.Join(" ", v.Select((x, i) => x.ToString().PadLeft(widths[i])));
255 string names =
256 string.Join(" ", widths.Select((w, i) => ((char)('A' + i)).ToString().PadLeft(w)));
257 int[] columns = new[] { 10, 3, 4, 7 }.Select(w => Math.Max(w, names.Length)).ToArray();
258 void Row(string head, params string[] cells) {
259 string row = head.PadRight(9);
260 for (int i = 0; i < 4; i++)
261 row += " " + (i < cells.Length ? cells[i] : "").PadRight(columns[i]);
262 Console.WriteLine(row.TrimEnd());
263 }
264 Row("Process", "Allocation", "Max", "Need", "Request");
265 Row("", names, names, names, names);
266 foreach (Process p in s.Processes.Values)
267 Row("P" + p.Id, Form(p.Allocation), Form(p.Max), Form(Need(p)),
268 Form(Block(p.Id)?.Vector ?? Zero()));
269 Row("Available", Form(s.Available));
270 Row("Total", Form(s.Total));
271 Console.WriteLine(
272 "blocked queue: " +
273 (s.Queue.Count == 0
274 ? "empty"
275 : string.Join(", ", s.Queue.Select(q => "P" + q.Id + " " + V(q.Vector) +
276 " since t=" + q.Since))));
277 }
278 static void Main() {
279 int n = int.Parse(Console.ReadLine()!);
280 for (time = 1; time <= n; time++) {
281 string[] a = Console.ReadLine()!.Split(' ');
282 string cmd = a[0];
283 if (cmd == "RESOURCES") {
284 int[] v = a.Skip(1).Select(int.Parse).ToArray();
285 s = new State { Total = v, Available = (int[])v.Clone() };
286 Console.WriteLine("resources " + V(v));
287 continue;
288 }
289 if (cmd is "MODE" or "POLICY" or "TRACE" or "AUTODETECT") {
290 if (cmd == "MODE")
291 mode = a[1];
292 if (cmd == "POLICY")
293 policy = a[1];
294 if (cmd == "TRACE")
295 trace = a[1] == "ON";
296 if (cmd == "AUTODETECT")
297 auto = a[1] == "ON";
298 Console.WriteLine(cmd.ToLowerInvariant() + " " + a[1]);
299 continue;
300 }
301 if (cmd == "SNAPSHOT") {
302 snapshots[a[1]] = s.Copy();
303 Console.WriteLine("saved " + a[1]);
304 continue;
305 }
306 if (cmd == "RESTORE") {
307 if (!snapshots.ContainsKey(a[1]))
308 Error("unknown snapshot " + a[1]);
309 else {
310 s = snapshots[a[1]].Copy();
311 Console.WriteLine("restored " + a[1]);
312 }
313 continue;
314 }
315 if (cmd == "STATE") {
316 Table();
317 continue;
318 }
319 if (cmd is "GRAPH" or "WAITGRAPH") {
320 Graph(cmd == "GRAPH");
321 continue;
322 }
323 if (cmd == "SEQUENCES") {
324 Sequences(int.Parse(a[1]));
325 continue;
326 }
327 if (cmd == "VICTIMS") {
328 Victims();
329 continue;
330 }
331 if (cmd == "SAFE") {
332 if (s.Processes.Count == 0)
333 Console.WriteLine("safe: no processes");
334 else {
335 var c = Check(s, false, trace);
336 Console.WriteLine(c.Left.Count == 0
337 ? "safe: " + L(c.Sequence)
338 : "unsafe: " + L(c.Left) + " cannot finish");
339 }
340 continue;
341 }
342 if (cmd == "DETECT") {
343 var dead = Check(s, true, trace).Left;
344 Console.WriteLine(dead.Count == 0 ? "no deadlock" : "deadlocked: " + L(dead));
345 continue;
346 }
347 if (cmd == "RECOVER") {
348 int count = 0;
349 while (true) {
350 var dead = Check(s, true, trace).Left;
351 if (dead.Count == 0)
352 break;
353 if (a.Length > 1) {
354 foreach (int id in dead)
355 Terminate(id, "aborted", false);
356 count += dead.Count;
357 Retry();
358 break;
359 }
360 int victim = policy == "YOUNGEST" ? dead.MaxBy(id => s.Processes[id].Created)
361 : policy == "NEED"
362 ? dead.OrderByDescending(id => Need(s.Processes[id]).Sum())
363 .ThenByDescending(id => id)
364 .First()
365 : dead.OrderBy(id => s.Processes[id].Allocation.Sum())
366 .ThenByDescending(id => id)
367 .First();
368 Terminate(victim, "aborted", true);
369 count++;
370 }
371 Console.WriteLine(count == 0 ? "no deadlock" : "recovered: " + count + " aborted");
372 continue;
373 }
374 if (cmd == "DEFICIT") {
375 var values = new List<string>();
376 for (int j = 0; j < s.Total.Length; j++) {
377 int bound = s.Processes.Values.Sum(p => Need(p)[j]);
378 string value = "-";
379 for (int x = 0; x <= bound; x++) {
380 State candidate = s.Copy();
381 candidate.Available[j] += x;
382 if (Check(candidate, false, false).Left.Count == 0) {
383 value = x.ToString();
384 break;
385 }
386 }
387 values.Add(value);
388 }
389 Console.WriteLine("deficit: (" + string.Join(",", values) + ")");
390 continue;
391 }
392 if (cmd == "HISTORY") {
393 var events =
394 s.History.Where(e => a.Length == 1 || a[1] == e.Kind || a[1] == e.Id.ToString())
395 .ToList();
396 foreach (Event e in events)
397 Console.WriteLine("t=" + e.Time + " " + e.Kind + " P" + e.Id + " " +
398 V(e.Vector) +
399 (e.Kind == "unblocked" ? " waited " + e.Wait : ""));
400 if (events.Count == 0)
401 Console.WriteLine("no events");
402 continue;
403 }
404 if (cmd == "STATS") {
405 for (int j = 0; j < s.Total.Length; j++) {
406 int used = s.Total[j] - s.Available[j], tenths = used * 1000 / s.Total[j];
407 Console.WriteLine((char)('A' + j) + ": allocated " + used + "/" + s.Total[j] +
408 " (" + tenths / 10 + "." + tenths % 10 + "%), requested " +
409 s.Queue.Sum(q => q.Vector[j]));
410 }
411 Console.WriteLine(
412 "events: " +
413 string.Join(", ",
414 kinds.Select(k => k + " " + s.History.Count(e => e.Kind == k))));
415 var waits = s.History.Where(e => e.Kind == "unblocked").ToList();
416 int total = waits.Sum(e => e.Wait);
417 if (waits.Count == 0)
418 Console.WriteLine("waits: none");
419 else {
420 int hundredths = total * 100 / waits.Count;
421 Console.WriteLine("waits: " + waits.Count + " unblocked, total " + total +
422 ", average " + hundredths / 100 + "." +
423 (hundredths % 100).ToString("D2"));
424 }
425 continue;
426 }
427 bool vectorCommand = cmd is "PROCESS" or "CLAIM" or "REQUEST" or "WHATIF" or
428 "RELEASE" or "ADD" or "REMOVE";
429 bool resourceCommand = cmd is "ADD" or "REMOVE";
430 int idValue = resourceCommand ? 0 : int.Parse(a[1]);
431 int[] vector = vectorCommand
432 ? a.Skip(resourceCommand ? 1 : 2).Select(int.Parse).ToArray()
433 : Zero();
434 if (vectorCommand && vector.Length != s.Total.Length) {
435 Error("wrong vector length");
436 continue;
437 }
438 if (!resourceCommand && cmd != "PROCESS" && !s.Processes.ContainsKey(idValue)) {
439 Error("unknown process P" + idValue);
440 continue;
441 }
442 if (cmd == "PROCESS") {
443 if (s.Processes.ContainsKey(idValue))
444 Error("duplicate process P" + idValue);
445 else if (!Le(vector, s.Total))
446 Error("max exceeds total");
447 else {
448 s.Processes[idValue] = new(idValue, vector, Zero(), time);
449 Console.WriteLine("process P" + idValue + " max " + V(vector));
450 }
451 continue;
452 }
453 if (cmd is "REQUEST" or "WHATIF") {
454 string rejection = Reject(idValue, vector);
455 if (rejection != "") {
456 if (cmd == "REQUEST")
457 Error(rejection);
458 else
459 Console.WriteLine("whatif P" + idValue + " " + V(vector) + ": rejected (" +
460 rejection + ")");
461 continue;
462 }
463 var decision = Decision(idValue, vector, trace);
464 if (cmd == "WHATIF")
465 Console.WriteLine(
466 "whatif P" + idValue + " " + V(vector) +
467 (decision.Reason != ""
468 ? ": blocked (" + decision.Reason + ")"
469 : ": granted" + (mode == "AVOID"
470 ? ", safe sequence " + L(decision.Sequence)
471 : "")));
472 else if (decision.Reason == "") {
473 Allocate(idValue, vector);
474 Console.WriteLine("granted P" + idValue + " " + V(vector));
475 Record("granted", idValue, vector);
476 } else {
477 s.Queue.Add(new(idValue, vector, time));
478 Console.WriteLine("blocked P" + idValue + " " + V(vector) + ": " +
479 decision.Reason);
480 Record("blocked", idValue, vector);
481 if (auto) {
482 var dead = Check(s, true, trace).Left;
483 if (dead.Count > 0)
484 Console.WriteLine("deadlock detected: " + L(dead));
485 }
486 }
487 continue;
488 }
489 if (!resourceCommand && cmd != "ABORT" && cmd != "CANCEL" && Block(idValue) != null) {
490 Error("process P" + idValue + " is blocked");
491 continue;
492 }
493 if (cmd == "MAXGRANT") {
494 int[] answer = Zero();
495 for (int j = 0; j < answer.Length; j++)
496 for (int x = Need(s.Processes[idValue])[j]; x > 0; x--) {
497 int[] v = Zero();
498 v[j] = x;
499 if (Reject(idValue, v) == "" && Decision(idValue, v, false).Reason == "") {
500 answer[j] = x;
501 break;
502 }
503 }
504 Console.WriteLine("maxgrant P" + idValue + ": " + V(answer));
505 } else if (cmd == "CANCEL") {
506 Waiting? q = Block(idValue);
507 if (q == null)
508 Error("process P" + idValue + " is not blocked");
509 else {
510 s.Queue.Remove(q);
511 Console.WriteLine("cancelled P" + idValue + " " + V(q.Vector));
512 Record("cancelled", idValue, q.Vector);
513 }
514 } else if (cmd is "FINISH" or "ABORT")
515 Terminate(idValue, cmd == "FINISH" ? "finished" : "aborted", true);
516 else if (cmd == "CLAIM") {
517 Process p = s.Processes[idValue];
518 if (!Le(p.Allocation, vector)) {
519 Error("claim below allocation");
520 continue;
521 }
522 if (!Le(vector, s.Total)) {
523 Error("max exceeds total");
524 continue;
525 }
526 State candidate = s.Copy();
527 candidate.Processes[idValue] = p with { Max = vector };
528 if (mode == "AVOID" && Check(candidate, false, trace).Left.Count > 0) {
529 Error("claim makes state unsafe");
530 continue;
531 }
532 s.Processes[idValue] = p with { Max = vector };
533 Console.WriteLine("claim P" + idValue + " max " + V(vector));
534 Retry();
535 } else {
536 if (vector.All(x => x == 0)) {
537 Error("zero vector");
538 continue;
539 }
540 if (cmd == "RELEASE") {
541 Process p = s.Processes[idValue];
542 if (!Le(vector, p.Allocation)) {
543 Error("release exceeds allocation");
544 continue;
545 }
546 s.Processes[idValue] = p with { Allocation = Sub(p.Allocation, vector) };
547 s.Available = Add(s.Available, vector);
548 Console.WriteLine("released P" + idValue + " " + V(vector));
549 Record("released", idValue, vector);
550 Retry();
551 } else if (cmd == "ADD") {
552 s.Total = Add(s.Total, vector);
553 s.Available = Add(s.Available, vector);
554 Console.WriteLine("added " + V(vector) + " total " + V(s.Total));
555 Retry();
556 } else {
557 if (!Le(vector, s.Available)) {
558 Error("remove exceeds available");
559 continue;
560 }
561 int[] total = Sub(s.Total, vector);
562 if (total.Any(x => x == 0)) {
563 Error("total must stay positive");
564 continue;
565 }
566 if (s.Processes.Values.Any(p => !Le(p.Max, total))) {
567 Error("max exceeds total");
568 continue;
569 }
570 State candidate = s.Copy();
571 candidate.Available = Sub(candidate.Available, vector);
572 if (mode == "AVOID" && Check(candidate, false, trace).Left.Count > 0) {
573 Error("removal makes state unsafe");
574 continue;
575 }
576 s.Total = total;
577 s.Available = candidate.Available;
578 Console.WriteLine("removed " + V(vector) + " total " + V(total));
579 }
580 }
581 }
582 }
