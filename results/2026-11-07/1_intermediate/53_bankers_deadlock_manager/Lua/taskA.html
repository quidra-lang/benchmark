1local function copy(value)
2 if type(value) ~= 'table' then return value end
3 local result = {}
4 for k, v in pairs(value) do result[k] = copy(v) end
5 return result
6end
7local function vec(v) return '(' .. table.concat(v, ',') .. ')' end
8local function names(ids)
9 local result = {}
10 for _, id in ipairs(ids) do result[#result + 1] = 'P' .. id end
11 return table.concat(result, ' ')
12end
13local function add(a, b)
14 local result = {}
15 for i, x in ipairs(a) do result[i] = x + b[i] end
16 return result
17end
18local function sub(a, b)
19 local result = {}
20 for i, x in ipairs(a) do result[i] = x - b[i] end
21 return result
22end
23local function fits(a, b)
24 for i, x in ipairs(a) do if x > b[i] then return false end end
25 return true
26end
27local function sum(a)
28 local result = 0
29 for _, x in ipairs(a) do result = result + x end
30 return result
31end
32local function sortedKeys(a)
33 local result = {}
34 for k in pairs(a) do result[#result + 1] = k end
35 table.sort(result)
36 return result
37end
38local function lexLess(a, b)
39 for i = 1, math.min(#a, #b) do
40 if a[i] ~= b[i] then return a[i] < b[i] end
41 end
42 return #a < #b
43end
44local total, available, processes, queue, history = {}, {}, {}, {}, {}
45local mode, policy, trace, autodetect = 'AVOID', 'ALLOC', false, false
46local snapshots, time = {}, 0
47local kinds = {'granted','blocked','unblocked','released','finished','aborted','cancelled'}
48local function zero()
49 local result = {}
50 for i = 1, #total do result[i] = 0 end
51 return result
52end
53local function need(id, ps)
54 local p = (ps or processes)[id]
55 return sub(p.max, p.alloc)
56end
57local function blocked(id, qs)
58 for _, q in ipairs(qs or queue) do if q[1] == id then return q end end
59end
60local function check(detect, tracing, av, ps, qs)
61 ps, qs = ps or processes, qs or queue
62 local work = copy(av or available)
63 local remaining, idle, sequence = {}, {}, {}
64 for _, id in ipairs(sortedKeys(ps)) do
65 local dest = detect and sum(ps[id].alloc) == 0 and idle or remaining
66 dest[#dest + 1] = id
67 end
68 local label = detect and 'detect' or 'safety'
69 if tracing then
70 local suffix = #idle > 0 and ' idle ' .. names(idle) or ''
71 print('[' .. label .. '] start work=' .. vec(work) .. suffix)
72 end
73 while true do
74 local index
75 for i, id in ipairs(remaining) do
76 local q = blocked(id, qs)
77 local required = detect and (q and q[2] or zero()) or need(id, ps)
78 if fits(required, work) then index = i; break end
79 end
80 if not index then break end
81 local id = table.remove(remaining, index)
82 work = add(work, ps[id].alloc)
83 sequence[#sequence + 1] = id
84 if tracing then print('[' .. label .. '] P' .. id .. ' can finish, work=' .. vec(work)) end
85 end
86 if tracing and #remaining > 0 then
87 print('[' .. label .. '] stuck work=' .. vec(work) .. ' remaining ' .. names(remaining))
88 end
89 return #remaining == 0, sequence, remaining
90end
91local function reject(id, v)
92 if blocked(id) then return 'process P' .. id .. ' is blocked' end
93 if sum(v) == 0 then return 'zero vector' end
94 if not fits(v, need(id)) then return 'request exceeds need' end
95 local last = 0
96 for i, x in ipairs(processes[id].alloc) do if x > 0 then last = i end end
97 if mode == 'ORDER' then
98 for i, x in ipairs(v) do if x > 0 and i <= last then return 'resource order violation' end end
99 end
100end
101local function decision(id, v, tracing)
102 if not fits(v, available) then return false, 'insufficient resources', {} end
103 if mode == 'AVOID' then
104 local ps = copy(processes)
105 ps[id].alloc = add(ps[id].alloc, v)
106 local safe, sequence = check(false, tracing, sub(available, v), ps)
107 if not safe then return false, 'unsafe state', {} end
108 return true, '', sequence
109 end
110 return true, '', {}
111end
112local function record(kind, id, v, wait)
113 history[#history + 1] = {time, kind, id, copy(v), wait}
114end
115local function grant(id, v)
116 available = sub(available, v)
117 processes[id].alloc = add(processes[id].alloc, v)
118end
119local function retryQueue()
120 local old = queue
121 queue = {}
122 for _, q in ipairs(old) do
123 local id, v, since = q[1], q[2], q[3]
124 if decision(id, v, trace) then
125 grant(id, v)
126 print('unblocked P' .. id .. ' ' .. vec(v))
127 record('unblocked', id, v, time - since)
128 else queue[#queue + 1] = q end
129 end
130end
131local function removeRequest(id)
132 for i = #queue, 1, -1 do if queue[i][1] == id then table.remove(queue, i) end end
133end
134local function terminate(id, kind, retryAfter)
135 local allocation = processes[id].alloc
136 available = add(available, allocation)
137 processes[id] = nil
138 removeRequest(id)
139 print(kind .. ' P' .. id .. ' released ' .. vec(allocation))
140 record(kind, id, allocation)
141 if retryAfter then retryQueue() end
142end
143local function requireThat(valid, message)
144 if not valid then error(message, 0) end
145end
146local function graph(single)
147 if single then for _, x in ipairs(total) do requireThat(x == 1, 'not single-instance') end end
148 local ids, edges, edgeCount = sortedKeys(processes), {}, 0
149 for _, p in ipairs(ids) do edges[p] = {} end
150 if single then
151 for i = 1, #total do
152 for _, p in ipairs(ids) do
153 if processes[p].alloc[i] > 0 then print('assign ' .. string.char(64+i) .. ' -> P' .. p) end
154 end
155 end
156 for _, p in ipairs(ids) do
157 local q = blocked(p)
158 if q then
159 for i, x in ipairs(q[2]) do
160 if x > 0 then
161 print('request P' .. p .. ' -> ' .. string.char(64+i))
162 for _, j in ipairs(ids) do
163 if processes[j].alloc[i] > 0 then edges[p][#edges[p]+1] = {i,j} end
164 end
165 end
166 end
167 end
168 end
169 for _, p in ipairs(ids) do
170 local q = blocked(p)
171 for i, x in ipairs(need(p)) do
172 if x > 0 and (not q or q[2][i] == 0) then print('claim P' .. p .. ' -> ' .. string.char(64+i)) end
173 end
174 end
175 else
176 for _, p in ipairs(ids) do
177 local q = blocked(p)
178 if q then
179 for _, j in ipairs(ids) do
180 if j ~= p then
181 local labels = {}
182 for i, x in ipairs(q[2]) do
183 if x > available[i] and processes[j].alloc[i] > 0 then labels[#labels+1] = string.char(64+i) end
184 end
185 if #labels > 0 then
186 print('P' .. p .. ' -> P' .. j .. ' on ' .. table.concat(labels, ','))
187 edges[p][#edges[p]+1] = {j,j}
188 edgeCount = edgeCount + 1
189 end
190 end
191 end
192 end
193 end
194 if edgeCount == 0 then print('no wait edges') end
195 end
196 local cycles = {}
197 for _, start in ipairs(ids) do
198 local function visit(p, seen, path)
199 for _, e in ipairs(edges[p]) do
200 local label, nextId = e[1], e[2]
201 local nextPath = copy(path)
202 if single then nextPath[#nextPath+1] = label end
203 nextPath[#nextPath+1] = nextId
204 if nextId == start then
205 if single or #path >= 2 then cycles[#cycles+1] = nextPath end
206 elseif nextId > start and not seen[nextId] then
207 local nextSeen = copy(seen)
208 nextSeen[nextId] = true
209 visit(nextId, nextSeen, nextPath)
210 end
211 end
212 end
213 visit(start, {[start]=true}, {start})
214 end
215 table.sort(cycles, lexLess)
216 for _, path in ipairs(cycles) do
217 local text = {}
218 for i, x in ipairs(path) do text[i] = single and i%2 == 0 and string.char(64+x) or 'P' .. x end
219 print('cycle: ' .. table.concat(text, ' -> '))
220 end
221 if #cycles == 0 then print('no cycles') end
222end
223local function stateTable()
224 local widths = {}
225 for i, x in ipairs(total) do widths[i] = #tostring(x) end
226 local function tableForm(v)
227 local a = {}
228 for i, x in ipairs(v) do a[i] = string.format('%' .. widths[i] .. 's', tostring(x)) end
229 return table.concat(a, ' ')
230 end
231 local letters = {}
232 for i = 1, #total do letters[i] = string.char(64+i) end
233 local name = tableForm(letters)
234 local columns = {math.max(10,#name),math.max(3,#name),math.max(4,#name),math.max(7,#name)}
235 local function row(heading, content)
236 local s = string.format('%-9s', heading)
237 for i, x in ipairs(content) do s = s .. ' ' .. string.format('%-' .. columns[i] .. 's', x) end
238 print((s:gsub('%s+$', '')))
239 end
240 row('Process', {'Allocation','Max','Need','Request'})
241 row('', {name,name,name,name})
242 for _, p in ipairs(sortedKeys(processes)) do
243 local q = blocked(p)
244 row('P'..p, {tableForm(processes[p].alloc),tableForm(processes[p].max),tableForm(need(p)),tableForm(q and q[2] or zero())})
245 end
246 row('Available', {tableForm(available),'','',''})
247 row('Total', {tableForm(total),'','',''})
248 local requests = {}
249 for _, q in ipairs(queue) do requests[#requests+1] = 'P' .. q[1] .. ' ' .. vec(q[2]) .. ' since t=' .. q[3] end
250 print('blocked queue: ' .. (#requests > 0 and table.concat(requests, ', ') or 'empty'))
251end
252local function statistics()
253 for i, t in ipairs(total) do
254 local used, requested = t - available[i], 0
255 for _, q in ipairs(queue) do requested = requested + q[2][i] end
256 local percent = used * 1000 // t
257 print(string.char(64+i) .. ': allocated ' .. used .. '/' .. t .. ' (' .. percent//10 .. '.' .. percent%10 .. '%), requested ' .. requested)
258 end
259 local counts, waits, waitSum = {}, 0, 0
260 for _, k in ipairs(kinds) do counts[k] = 0 end
261 for _, e in ipairs(history) do
262 counts[e[2]] = counts[e[2]] + 1
263 if e[2] == 'unblocked' then waits, waitSum = waits+1, waitSum+e[5] end
264 end
265 local parts = {}
266 for _, k in ipairs(kinds) do parts[#parts+1] = k .. ' ' .. counts[k] end
267 print('events: ' .. table.concat(parts, ', '))
268 if waits == 0 then print('waits: none')
269 else
270 local avg = waitSum*100//waits
271 print('waits: ' .. waits .. ' unblocked, total ' .. waitSum .. ', average ' .. avg//100 .. '.' .. string.format('%02d',avg%100))
272 end
273end
274local function run(line)
275 local words = {}
276 for word in line:gmatch('%S+') do words[#words+1] = word end
277 local cmd = table.remove(words,1)
278 local id = tonumber(words[1]) or 0
279 local withId = cmd == 'PROCESS' or cmd == 'CLAIM' or cmd == 'REQUEST' or cmd == 'WHATIF' or cmd == 'RELEASE'
280 local v = {}
281 for i = withId and 2 or 1, #words do v[#v+1] = tonumber(words[i]) end
282 if withId or cmd == 'ADD' or cmd == 'REMOVE' then requireThat(#v == #total, 'wrong vector length') end
283 if cmd == 'CLAIM' or cmd == 'REQUEST' or cmd == 'WHATIF' or cmd == 'RELEASE' or cmd == 'FINISH' or cmd == 'ABORT' or cmd == 'CANCEL' or cmd == 'MAXGRANT' then
284 requireThat(processes[id] ~= nil, 'unknown process P' .. id)
285 end
286 if cmd == 'RESOURCES' then
287 total, available, processes, queue, history = copy(v), copy(v), {}, {}, {}
288 print('resources ' .. vec(v))
289 elseif cmd == 'MODE' or cmd == 'POLICY' or cmd == 'TRACE' or cmd == 'AUTODETECT' then
290 if cmd == 'MODE' then mode = words[1]
291 elseif cmd == 'POLICY' then policy = words[1]
292 elseif cmd == 'TRACE' then trace = words[1] == 'ON'
293 else autodetect = words[1] == 'ON' end
294 print(cmd:lower() .. ' ' .. words[1])
295 elseif cmd == 'PROCESS' then
296 requireThat(not processes[id], 'duplicate process P' .. id)
297 requireThat(fits(v,total), 'max exceeds total')
298 processes[id] = {max=v,alloc=zero(),born=time}
299 print('process P' .. id .. ' max ' .. vec(v))
300 elseif cmd == 'CLAIM' then
301 requireThat(not blocked(id), 'process P' .. id .. ' is blocked')
302 requireThat(fits(processes[id].alloc,v), 'claim below allocation')
303 requireThat(fits(v,total), 'max exceeds total')
304 if mode == 'AVOID' then
305 local ps = copy(processes)
306 ps[id].max = v
307 requireThat(check(false,trace,available,ps), 'claim makes state unsafe')
308 end
309 processes[id].max = v
310 print('claim P' .. id .. ' max ' .. vec(v))
311 retryQueue()
312 elseif cmd == 'ADD' or cmd == 'REMOVE' then
313 requireThat(sum(v) > 0, 'zero vector')
314 if cmd == 'ADD' then
315 total, available = add(total,v), add(available,v)
316 print('added ' .. vec(v) .. ' total ' .. vec(total))
317 retryQueue()
318 else
319 requireThat(fits(v,available), 'remove exceeds available')
320 local nextTotal, nextAvailable = sub(total,v), sub(available,v)
321 for _, x in ipairs(nextTotal) do requireThat(x ~= 0, 'total must stay positive') end
322 for _, p in pairs(processes) do requireThat(fits(p.max,nextTotal), 'max exceeds total') end
323 if mode == 'AVOID' then requireThat(check(false,trace,nextAvailable), 'removal makes state unsafe') end
324 total, available = nextTotal, nextAvailable
325 print('removed ' .. vec(v) .. ' total ' .. vec(total))
326 end
327 elseif cmd == 'REQUEST' or cmd == 'WHATIF' then
328 local reason = reject(id,v)
329 if reason then
330 if cmd == 'REQUEST' then error(reason,0) end
331 print('whatif P' .. id .. ' ' .. vec(v) .. ': rejected (' .. reason .. ')')
332 else
333 local granted, why, sequence = decision(id,v,trace)
334 if cmd == 'WHATIF' then
335 local result = granted and 'granted' or 'blocked (' .. why .. ')'
336 if granted and mode == 'AVOID' then result = result .. ', safe sequence ' .. names(sequence) end
337 print('whatif P' .. id .. ' ' .. vec(v) .. ': ' .. result)
338 elseif granted then
339 grant(id,v)
340 print('granted P' .. id .. ' ' .. vec(v))
341 record('granted',id,v)
342 else
343 queue[#queue+1] = {id,v,time}
344 print('blocked P' .. id .. ' ' .. vec(v) .. ': ' .. why)
345 record('blocked',id,v)
346 if autodetect then
347 local _, _, dead = check(true,trace)
348 if #dead > 0 then print('deadlock detected: ' .. names(dead)) end
349 end
350 end
351 end
352 elseif cmd == 'RELEASE' then
353 requireThat(not blocked(id), 'process P' .. id .. ' is blocked')
354 requireThat(sum(v) > 0, 'zero vector')
355 requireThat(fits(v,processes[id].alloc), 'release exceeds allocation')
356 available, processes[id].alloc = add(available,v), sub(processes[id].alloc,v)
357 print('released P' .. id .. ' ' .. vec(v))
358 record('released',id,v)
359 retryQueue()
360 elseif cmd == 'FINISH' or cmd == 'ABORT' then
361 requireThat(cmd ~= 'FINISH' or not blocked(id), 'process P' .. id .. ' is blocked')
362 terminate(id,cmd == 'FINISH' and 'finished' or 'aborted',true)
363 elseif cmd == 'CANCEL' then
364 local q = blocked(id)
365 requireThat(q ~= nil, 'process P' .. id .. ' is not blocked')
366 removeRequest(id)
367 print('cancelled P' .. id .. ' ' .. vec(q[2]))
368 record('cancelled',id,q[2])
369 elseif cmd == 'MAXGRANT' then
370 requireThat(not blocked(id), 'process P' .. id .. ' is blocked')
371 local result = zero()
372 for i, n in ipairs(need(id)) do
373 for x = n, 1, -1 do
374 local w = zero(); w[i] = x
375 if not reject(id,w) and decision(id,w,false) then result[i] = x; break end
376 end
377 end
378 print('maxgrant P' .. id .. ': ' .. vec(result))
379 elseif cmd == 'SAFE' then
380 if next(processes) == nil then print('safe: no processes')
381 else
382 local safe, sequence, remaining = check(false,trace)
383 print(safe and 'safe: ' .. names(sequence) or 'unsafe: ' .. names(remaining) .. ' cannot finish')
384 end
385 elseif cmd == 'SEQUENCES' then
386 if next(processes) == nil then print('safe sequences: no processes')
387 else
388 local count, first = 0, {}
389 local function visit(work, remaining, path)
390 if #remaining == 0 then
391 count = count + 1
392 if #first < id then first[#first+1] = path end
393 return
394 end
395 for i, p in ipairs(remaining) do
396 if fits(need(p),work) then
397 local rest, nextPath = copy(remaining), copy(path)
398 table.remove(rest,i); nextPath[#nextPath+1] = p
399 visit(add(work,processes[p].alloc),rest,nextPath)
400 end
401 end
402 end
403 visit(available,sortedKeys(processes),{})
404 print('safe sequences: ' .. count)
405 for _, path in ipairs(first) do print(names(path)) end
406 end
407 elseif cmd == 'DEFICIT' then
408 local result = {}
409 for i = 1, #total do
410 local limit = 0
411 for p in pairs(processes) do limit = limit + need(p)[i] end
412 result[i] = '-'
413 for x = 0, limit do
414 local av = copy(available); av[i] = av[i]+x
415 if check(false,false,av) then result[i] = x; break end
416 end
417 end
418 print('deficit: ' .. vec(result))
419 elseif cmd == 'STATE' then stateTable()
420 elseif cmd == 'STATS' then statistics()
421 elseif cmd == 'DETECT' then
422 local _, _, dead = check(true,trace)
423 print(#dead > 0 and 'deadlocked: ' .. names(dead) or 'no deadlock')
424 elseif cmd == 'VICTIMS' then
425 local _, _, dead = check(true,false)
426 if #dead == 0 then print('no deadlock')
427 else
428 local candidates = {}
429 for mask = 1, (1 << #dead)-1 do
430 local set, cost = {}, 0
431 for i, p in ipairs(dead) do
432 if mask & (1 << (i-1)) ~= 0 then set[#set+1] = p; cost = cost+sum(processes[p].alloc) end
433 end
434 candidates[#candidates+1] = {set,cost}
435 end
436 table.sort(candidates,function(a,b)
437 if #a[1] ~= #b[1] then return #a[1] < #b[1] end
438 if a[2] ~= b[2] then return a[2] < b[2] end
439 return lexLess(a[1],b[1])
440 end)
441 for _, candidate in ipairs(candidates) do
442 local set, ps, av, qs = candidate[1], copy(processes), copy(available), copy(queue)
443 local removed = {}
444 for _, p in ipairs(set) do av = add(av,ps[p].alloc); ps[p] = nil; removed[p] = true end
445 for i = #qs, 1, -1 do if removed[qs[i][1]] then table.remove(qs,i) end end
446 if check(true,false,av,ps,qs) then print('victims: ' .. names(set)); break end
447 end
448 end
449 elseif cmd == 'RECOVER' then
450 local count = 0
451 while true do
452 local _, _, dead = check(true,trace)
453 if #dead == 0 then break end
454 if words[1] == 'ALL' then
455 for _, p in ipairs(dead) do terminate(p,'aborted',false) end
456 count = count+#dead; retryQueue(); break
457 end
458 table.sort(dead,function(a,b)
459 if policy == 'YOUNGEST' then return processes[a].born > processes[b].born end
460 local x = policy == 'ALLOC' and sum(processes[a].alloc) or -sum(need(a))
461 local y = policy == 'ALLOC' and sum(processes[b].alloc) or -sum(need(b))
462 return x < y or (x == y and a > b)
463 end)
464 terminate(dead[1],'aborted',true)
465 count = count+1
466 end
467 print(count > 0 and 'recovered: ' .. count .. ' aborted' or 'no deadlock')
468 elseif cmd == 'WAITGRAPH' or cmd == 'GRAPH' then graph(cmd == 'GRAPH')
469 elseif cmd == 'HISTORY' then
470 local found = false
471 for _, e in ipairs(history) do
472 if not words[1] or e[2] == words[1] or (tonumber(words[1]) and e[3] == id) then
473 local text = 't=' .. e[1] .. ' ' .. e[2] .. ' P' .. e[3] .. ' ' .. vec(e[4])
474 if e[2] == 'unblocked' then text = text .. ' waited ' .. e[5] end
475 print(text); found = true
476 end
477 end
478 if not found then print('no events') end
479 elseif cmd == 'SNAPSHOT' then
480 snapshots[words[1]] = copy({total,available,processes,queue,history})
481 print('saved ' .. words[1])
482 elseif cmd == 'RESTORE' then
483 requireThat(snapshots[words[1]] ~= nil, 'unknown snapshot ' .. words[1])
484 local saved = copy(snapshots[words[1]])
485 total, available, processes, queue, history = table.unpack(saved)
486 print('restored ' .. words[1])
487 end
488end
489local n = tonumber(io.read())
490for i = 1, n do
491 time = i
492 local ok, message = pcall(run,io.read())
493 if not ok then print('error: ' .. message) end
494end
