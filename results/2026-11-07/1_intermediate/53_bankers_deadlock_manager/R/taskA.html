26check <- function(s, detect = FALSE, trace = FALSE) {
27 work <- s$available
28 remaining <- ids(s)
29 idle <- integer()
30 sequence <- integer()
31 if (detect) {
32 idle <- remaining[vapply(remaining, function(id) all(s$processes[[pn(id)]]$allocation == 0), logical(1))]
33 remaining <- setdiff(remaining, idle)
34 }
35 tag <- if (detect) "[detect]" else "[safety]"
36 if (trace) {
37 suffix <- if (length(idle)) paste0(" idle ", pf(idle)) else ""
38 line(tag, " start work=", vf(work), suffix)
39 }
40 repeat {
41 chosen <- NA_integer_
42 for (id in remaining) {
43 p <- s$processes[[pn(id)]]
44 v <- if (detect) request(s, id) else need(p)
45 if (all(v <= work)) {
46 chosen <- id
47 break
48 }
49 }
50 if (is.na(chosen)) break
51 work <- work + s$processes[[pn(chosen)]]$allocation
52 remaining <- remaining[remaining != chosen]
53 sequence <- c(sequence, chosen)
54 if (trace) line(tag, " ", pn(chosen), " can finish, work=", vf(work))
55 }
56 if (trace && length(remaining)) line(tag, " stuck work=", vf(work), " remaining ", pf(remaining))
57 list(ok = !length(remaining), sequence = sequence, remaining = remaining)
58}
59decision <- function(id, v, reject = TRUE, trace = m$trace) {
60 s <- m$s
61 p <- s$processes[[pn(id)]]
62 result <- function(status, reason = "", sequence = integer()) list(status = status, reason = reason, sequence = sequence)
63 if (reject) {
64 if (blocked(s, id)) return(result("rejected", paste0("process ", pn(id), " is blocked")))
65 if (all(v == 0)) return(result("rejected", "zero vector"))
66 if (any(v > need(p))) return(result("rejected", "request exceeds need"))
67 held <- which(p$allocation > 0)
68 if (m$mode == "ORDER" && length(held) && any(v[seq_len(max(held))] > 0)) {
69 return(result("rejected", "resource order violation"))
70 }
71 }
72 if (any(v > s$available)) return(result("blocked", "insufficient resources"))
73 if (m$mode == "AVOID") {
74 s$available <- s$available - v
75 s$processes[[pn(id)]]$allocation <- p$allocation + v
76 tested <- check(s, FALSE, trace)
77 if (!tested$ok) return(result("blocked", "unsafe state"))
78 return(result("granted", sequence = tested$sequence))
79 }
80 result("granted")
81}
82record <- function(kind, id, v, wait = 0) {
83 m$s$history[[length(m$s$history) + 1L]] <- list(time = m$time, kind = kind, id = id, value = v, wait = wait)
84}
85remove_wait <- function(id) {
86 m$s$queue <- Filter(function(q) q$id != id, m$s$queue)
87}
88grant <- function(id, v) {
89 m$s$available <- m$s$available - v
90 m$s$processes[[pn(id)]]$allocation <- m$s$processes[[pn(id)]]$allocation + v
91}
92retry <- function() {
93 queue <- m$s$queue
94 for (q in queue) {
95 if (decision(q$id, q$value, FALSE)$status == "granted") {
96 grant(q$id, q$value)
97 remove_wait(q$id)
98 line("unblocked ", pn(q$id), " ", vf(q$value))
99 record("unblocked", q$id, q$value, m$time - q$since)
100 }
101 }
102}
103terminate <- function(id, kind, retry_after = TRUE) {
104 v <- m$s$processes[[pn(id)]]$allocation
105 m$s$available <- m$s$available + v
106 m$s$processes[[pn(id)]] <- NULL
107 remove_wait(id)
108 line(kind, " ", pn(id), " released ", vf(v))
109 record(kind, id, v)
110 if (retry_after) retry()
111}
112lex_less <- function(a, b) {
113 for (i in seq_len(min(length(a), length(b)))) {
114 if (a[i] != b[i]) return(a[i] < b[i])
115 }
116 length(a) < length(b)
117}
118lex_sort <- function(items) {
119 if (length(items) < 2L) return(items)
120 for (i in 2:length(items)) {
121 j <- i
122 while (j > 1L && lex_less(items[[j]], items[[j - 1L]])) {
123 tmp <- items[[j]]
124 items[[j]] <- items[[j - 1L]]
125 items[[j - 1L]] <- tmp
126 j <- j - 1L
127 }
128 }
129 items
130}
131victims <- function(s) {
132 dead <- check(s, TRUE)$remaining
133 if (!length(dead)) return(integer())
134 for (size in seq_along(dead)) {
135 choices <- combn(dead, size, simplify = FALSE)
136 choices <- lex_sort(choices)
137 costs <- vapply(choices, function(choice) sum(vapply(choice, function(id) sum(s$processes[[pn(id)]]$allocation), numeric(1))), numeric(1))
138 choices <- choices[order(costs, seq_along(choices))]
139 for (choice in choices) {
140 trial <- s
141 for (id in choice) {
142 trial$available <- trial$available + trial$processes[[pn(id)]]$allocation
143 trial$processes[[pn(id)]] <- NULL
144 }
145 if (check(trial, TRUE)$ok) return(choice)
146 }
147 }
148 integer()

150graph <- function(resource) {
151 s <- m$s
152 order_ids <- ids(s)
153 edges <- setNames(lapply(order_ids, function(id) list()), pn(order_ids))
154 if (resource) {
155 if (any(s$total != 1)) {
156 line("error: not single-instance")
157 return()
158 }
159 for (j in seq_along(s$total)) {
160 for (id in order_ids) {
161 if (s$processes[[pn(id)]]$allocation[j] > 0) line("assign ", LETTERS[j], " -> ", pn(id))
162 }
163 }
164 for (id in order_ids) {
165 r <- request(s, id)
166 for (j in which(r > 0)) {
167 line("request ", pn(id), " -> ", LETTERS[j])
168 for (to in order_ids) {
169 if (s$processes[[pn(to)]]$allocation[j] > 0) {
170 edges[[pn(id)]][[length(edges[[pn(id)]]) + 1L]] <- c(to, j)
171 }
172 }
173 }
174 }
175 for (id in order_ids) {
176 for (j in which(need(s$processes[[pn(id)]]) > 0 & request(s, id) == 0)) {
177 line("claim ", pn(id), " -> ", LETTERS[j])
178 }
179 }
180 } else {
181 count <- 0L
182 for (id in order_ids) {
183 for (to in order_ids) {
184 if (id == to) next
185 labels <- which(request(s, id) > s$available & s$processes[[pn(to)]]$allocation > 0)
186 if (length(labels)) {
187 line(pn(id), " -> ", pn(to), " on ", paste(LETTERS[labels], collapse = ","))
188 edges[[pn(id)]][[length(edges[[pn(id)]]) + 1L]] <- c(to, 0)
189 count <- count + 1L
190 }
191 }
192 }
193 if (!count) line("no wait edges")
194 }
195 cycles <- list()
196 for (start in order_ids) {
197 visit <- function(at, path, seen) {
198 for (edge in edges[[pn(at)]]) {
199 to <- edge[1]
200 next_path <- if (resource) c(path, edge[2], to) else c(path, to)
201 if (to == start) {
202 if (resource || length(seen) >= 2L) cycles[[length(cycles) + 1L]] <<- next_path
203 } else if (to > start && !(to %in% seen)) {
204 visit(to, next_path, c(seen, to))
205 }
206 }
207 }
208 visit(start, start, start)
209 }
210 for (cycle in lex_sort(cycles)) {
211 parts <- vapply(seq_along(cycle), function(i) {
212 if (resource && i %% 2 == 0) LETTERS[cycle[i]] else pn(cycle[i])
213 }, character(1))
214 line("cycle: ", paste(parts, collapse = " -> "))
215 }
216 if (!length(cycles)) line("no cycles")
217}
218table <- function(s) {
219 widths <- nchar(s$total)
220 names_row <- paste(mapply(function(x, w) sprintf("%*s", w, x), LETTERS[seq_along(widths)], widths), collapse = " ")
221 cw <- pmax(c(10, 3, 4, 7), nchar(names_row))
222 form <- function(v) paste(mapply(function(x, w) sprintf("%*d", w, x), v, widths), collapse = " ")
223 row <- function(heading, cols) {
224 cols <- c(cols, rep("", 4L - length(cols)))
225 result <- sprintf("%-9s", heading)
226 for (i in 1:4) result <- paste0(result, " ", sprintf("%-*s", cw[i], cols[i]))
227 line(sub(" +$", "", result))
228 }
229 row("Process", c("Allocation", "Max", "Need", "Request"))
230 row("", rep(names_row, 4))
231 for (id in ids(s)) {
232 p <- s$processes[[pn(id)]]
233 row(pn(id), vapply(list(p$allocation, p$maximum, need(p), request(s, id)), form, character(1)))
234 }
235 row("Available", form(s$available))
236 row("Total", form(s$total))
237 if (!length(s$queue)) {
238 line("blocked queue: empty")
239 } else {
240 items <- vapply(s$queue, function(q) paste0(pn(q$id), " ", vf(q$value), " since t=", q$since), character(1))
241 line("blocked queue: ", paste(items, collapse = ", "))
242 }

244execute <- function(words) {
245 cmd <- words[1]
246 args <- words[-1]
247 has_id <- cmd %in% c("PROCESS", "CLAIM", "REQUEST", "WHATIF", "RELEASE", "FINISH", "ABORT", "CANCEL", "MAXGRANT")
248 has_vector <- cmd %in% c("PROCESS", "CLAIM", "REQUEST", "WHATIF", "RELEASE", "ADD", "REMOVE")
249 id <- if (has_id) as.integer(args[1]) else 0L
250 v <- if (has_vector) as.numeric(if (has_id) args[-1] else args) else numeric()
251 err <- ""
252 if (has_vector && length(v) != length(m$s$total)) {
253 err <- "wrong vector length"
254 } else if (has_id && cmd == "PROCESS" && !is.null(m$s$processes[[pn(id)]])) {
255 err <- paste0("duplicate process ", pn(id))
256 } else if (has_id && cmd != "PROCESS" && is.null(m$s$processes[[pn(id)]])) {
257 err <- paste0("unknown process ", pn(id))
258 }
259 if (nzchar(err)) {
260 line("error: ", err)
261 return()
262 }
263 if (cmd == "RESOURCES") {
264 m$s <- empty_state(as.numeric(args))
265 line("resources ", vf(m$s$total))
266 } else if (cmd %in% c("MODE", "POLICY", "TRACE", "AUTODETECT")) {
267 if (cmd == "MODE") m$mode <- args[1]
268 if (cmd == "POLICY") m$policy <- args[1]
269 if (cmd == "TRACE") m$trace <- args[1] == "ON"
270 if (cmd == "AUTODETECT") m$automatic <- args[1] == "ON"
271 line(tolower(cmd), " ", args[1])
272 } else if (cmd == "PROCESS") {
273 if (any(v > m$s$total)) {
274 err <- "max exceeds total"
275 } else {
276 m$s$processes[[pn(id)]] <- list(maximum = v, allocation = numeric(length(v)), created = m$time)
277 line("process ", pn(id), " max ", vf(v))
278 }
279 } else if (cmd == "CLAIM") {
280 p <- m$s$processes[[pn(id)]]
281 if (blocked(m$s, id)) {
282 err <- paste0("process ", pn(id), " is blocked")
283 } else if (any(p$allocation > v)) {
284 err <- "claim below allocation"
285 } else if (any(v > m$s$total)) {
286 err <- "max exceeds total"
287 } else {
288 trial <- m$s
289 trial$processes[[pn(id)]]$maximum <- v
290 if (m$mode == "AVOID" && !check(trial, FALSE, m$trace)$ok) {
291 err <- "claim makes state unsafe"
292 } else {
293 m$s <- trial
294 line("claim ", pn(id), " max ", vf(v))
295 retry()
296 }
297 }
298 } else if (cmd %in% c("ADD", "REMOVE")) {
299 if (all(v == 0)) {
300 err <- "zero vector"
301 } else if (cmd == "ADD") {
302 m$s$total <- m$s$total + v
303 m$s$available <- m$s$available + v
304 line("added ", vf(v), " total ", vf(m$s$total))
305 retry()
306 } else if (any(v > m$s$available)) {
307 err <- "remove exceeds available"
308 } else if (any(m$s$total - v == 0)) {
309 err <- "total must stay positive"
310 } else if (any(vapply(m$s$processes, function(p) any(p$maximum > m$s$total - v), logical(1)))) {
311 err <- "max exceeds total"
312 } else {
313 trial <- m$s
314 trial$available <- trial$available - v
315 if (m$mode == "AVOID" && !check(trial, FALSE, m$trace)$ok) {
316 err <- "removal makes state unsafe"
317 } else {
318 trial$total <- trial$total - v
319 m$s <- trial
320 line("removed ", vf(v), " total ", vf(m$s$total))
321 }
322 }
323 } else if (cmd %in% c("REQUEST", "WHATIF")) {
324 d <- decision(id, v)
325 if (cmd == "WHATIF") {
326 suffix <- if (d$status != "granted") paste0(" (", d$reason, ")") else if (m$mode == "AVOID") paste0(", safe sequence ", pf(d$sequence)) else ""
327 line("whatif ", pn(id), " ", vf(v), ": ", d$status, suffix)
328 } else if (d$status == "rejected") {
329 err <- d$reason
330 } else {
331 if (d$status == "granted") {
332 grant(id, v)
333 line("granted ", pn(id), " ", vf(v))
334 } else {
335 m$s$queue[[length(m$s$queue) + 1L]] <- list(id = id, value = v, since = m$time)
336 line("blocked ", pn(id), " ", vf(v), ": ", d$reason)
337 }
338 record(d$status, id, v)
339 if (d$status == "blocked" && m$automatic) {
340 dead <- check(m$s, TRUE, m$trace)$remaining
341 if (length(dead)) line("deadlock detected: ", pf(dead))
342 }
343 }
344 } else if (cmd == "RELEASE") {
345 p <- m$s$processes[[pn(id)]]
346 if (blocked(m$s, id)) {
347 err <- paste0("process ", pn(id), " is blocked")
348 } else if (all(v == 0)) {
349 err <- "zero vector"
350 } else if (any(v > p$allocation)) {
351 err <- "release exceeds allocation"
352 } else {
353 m$s$processes[[pn(id)]]$allocation <- p$allocation - v
354 m$s$available <- m$s$available + v
355 line("released ", pn(id), " ", vf(v))
356 record("released", id, v)
357 retry()
358 }
359 } else if (cmd %in% c("FINISH", "ABORT")) {
360 if (cmd == "FINISH" && blocked(m$s, id)) {
361 err <- paste0("process ", pn(id), " is blocked")
362 } else {
363 terminate(id, if (cmd == "FINISH") "finished" else "aborted")
364 }
365 } else if (cmd == "CANCEL") {
366 if (!blocked(m$s, id)) {
367 err <- paste0("process ", pn(id), " is not blocked")
368 } else {
369 v <- request(m$s, id)
370 remove_wait(id)
371 line("cancelled ", pn(id), " ", vf(v))
372 record("cancelled", id, v)
373 }
374 } else if (cmd == "MAXGRANT") {
375 if (blocked(m$s, id)) {
376 err <- paste0("process ", pn(id), " is blocked")
377 } else {
378 result <- numeric(length(m$s$total))
379 for (j in seq_along(result)) {
380 limit <- need(m$s$processes[[pn(id)]])[j]
381 if (!limit) next
382 for (x in seq(limit, 1)) {
383 value <- numeric(length(result))
384 value[j] <- x
385 if (decision(id, value, trace = FALSE)$status == "granted") {
386 result[j] <- x
387 break
388 }
389 }
390 }
391 line("maxgrant ", pn(id), ": ", vf(result))
392 }
393 } else if (cmd == "SAFE") {
394 if (!length(m$s$processes)) {
395 line("safe: no processes")
396 } else {
397 result <- check(m$s, FALSE, m$trace)
398 line(if (result$ok) paste0("safe: ", pf(result$sequence)) else paste0("unsafe: ", pf(result$remaining), " cannot finish"))
399 }
400 } else if (cmd == "SEQUENCES") {
401 if (!length(m$s$processes)) {
402 line("safe sequences: no processes")
403 } else {
404 total <- 0L
405 samples <- character()
406 search <- function(work, path) {
407 if (length(path) == length(m$s$processes)) {
408 total <<- total + 1L
409 if (length(samples) < as.integer(args[1])) samples <<- c(samples, pf(path))
410 return()
411 }
412 for (id in setdiff(ids(m$s), path)) {
413 p <- m$s$processes[[pn(id)]]
414 if (all(need(p) <= work)) search(work + p$allocation, c(path, id))
415 }
416 }
417 search(m$s$available, integer())
418 line("safe sequences: ", total)
419 for (sample in samples) line(sample)
420 }
421 } else if (cmd == "DEFICIT") {
422 result <- character(length(m$s$total))
423 for (j in seq_along(result)) {
424 limit <- sum(vapply(m$s$processes, function(p) need(p)[j], numeric(1)))
425 result[j] <- "-"
426 for (x in 0:limit) {
427 trial <- m$s
428 trial$available[j] <- trial$available[j] + x
429 if (check(trial)$ok) {
430 result[j] <- as.character(x)
431 break
432 }
433 }
434 }
435 line("deficit: ", vf(result))
436 } else if (cmd == "STATE") {
437 table(m$s)
438 } else if (cmd == "STATS") {
439 for (j in seq_along(m$s$total)) {
440 total <- m$s$total[j]
441 used <- total - m$s$available[j]
442 requested <- sum(vapply(m$s$queue, function(q) q$value[j], numeric(1)))
443 pct <- (used * 1000) %/% total
444 line(LETTERS[j], ": allocated ", used, "/", total, " (", pct %/% 10, ".", pct %% 10, "%), requested ", requested)
445 }
446 kinds <- c("granted", "blocked", "unblocked", "released", "finished", "aborted", "cancelled")
447 counts <- vapply(kinds, function(kind) sum(vapply(m$s$history, function(e) e$kind == kind, logical(1))), numeric(1))
448 line("events: ", paste(paste(kinds, counts), collapse = ", "))
449 n <- counts["unblocked"]
450 if (!n) {
451 line("waits: none")
452 } else {
453 total <- sum(vapply(m$s$history, function(e) if (e$kind == "unblocked") e$wait else 0, numeric(1)))
454 avg <- (total * 100) %/% n
455 line("waits: ", n, " unblocked, total ", total, ", average ", avg %/% 100, ".", sprintf("%02d", as.integer(avg %% 100)))
456 }
457 } else if (cmd == "DETECT") {
458 dead <- check(m$s, TRUE, m$trace)$remaining
459 line(if (length(dead)) paste0("deadlocked: ", pf(dead)) else "no deadlock")
460 } else if (cmd == "VICTIMS") {
461 choice <- victims(m$s)
462 line(if (length(choice)) paste0("victims: ", pf(choice)) else "no deadlock")
463 } else if (cmd == "RECOVER") {
464 count <- 0L
465 repeat {
466 dead <- check(m$s, TRUE, m$trace)$remaining
467 if (!length(dead)) break
468 if (length(args)) {
469 for (id in dead) terminate(id, "aborted", FALSE)
470 count <- count + length(dead)
471 retry()
472 break
473 }
474 priority <- vapply(dead, function(id) {
475 p <- m$s$processes[[pn(id)]]
476 if (m$policy == "ALLOC") sum(p$allocation) else if (m$policy == "NEED") -sum(need(p)) else -p$created
477 }, numeric(1))
478 chosen <- dead[order(priority, -dead)[1]]
479 terminate(chosen, "aborted")
480 count <- count + 1L
481 }
482 line(if (count) paste0("recovered: ", count, " aborted") else "no deadlock")
483 } else if (cmd %in% c("WAITGRAPH", "GRAPH")) {
484 graph(cmd == "GRAPH")
485 } else if (cmd == "HISTORY") {
486 found <- FALSE
487 for (e in m$s$history) {
488 if (length(args) && args[1] != e$kind && args[1] != as.character(e$id)) next
489 suffix <- if (e$kind == "unblocked") paste0(" waited ", e$wait) else ""
490 line("t=", e$time, " ", e$kind, " ", pn(e$id), " ", vf(e$value), suffix)
491 found <- TRUE
492 }
493 if (!found) line("no events")
494 } else if (cmd == "SNAPSHOT") {
495 m$snapshots[[args[1]]] <- m$s
496 line("saved ", args[1])
497 } else if (cmd == "RESTORE") {
498 if (is.null(m$snapshots[[args[1]]])) {
499 err <- paste0("unknown snapshot ", args[1])
500 } else {
501 m$s <- m$snapshots[[args[1]]]
502 line("restored ", args[1])
503 }
504 }
505 if (nzchar(err)) line("error: ", err)
506}
507input <- file("stdin", "r")
508n <- as.integer(readLines(input, n = 1L))
509for (t in seq_len(n)) {
510 m$time <- t
511 execute(strsplit(readLines(input, n = 1L), " ", fixed = TRUE)[[1]])
512}
513close(input)
