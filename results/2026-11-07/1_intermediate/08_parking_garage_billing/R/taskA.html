111interval_fee <- function(v,a,b,free) {
112 total <- b-a
113 if (valid_pass(v$plate,b) || total<=free) return(c(0L,0L))
114 rule <- rates[[v$type]]
115 parking <- i <- 0L
116 while (total>0L) {
117 minutes <- min(total,1440L)
118 billable <- minutes-(if (i==0L) free else 0L)
119 parking <- parking+min(((billable+29L)%/%30L)*rule[1L],rule[2L])
120 total <- total-minutes
121 i <- i+1L
122 }
123 c(parking,(date_of(b)-date_of(a))*fees[["NIGHT"]])
124}
125charges <- function(v,lost=FALSE) {
126 if (is.null(v$payment)) {
127 amounts <- interval_fee(v,v$entry,now,15L+v$validation)
128 loss <- if (lost) fees[["LOST"]] else 0L
129 total <- sum(amounts)+loss+v$cost
130 return(c(amounts,loss,v$cost,total,total))
131 }
132 extra <- interval_fee(v,v$payment$time,now,15L)
133 parking <- v$payment$parking+extra[1L]
134 night <- v$payment$night+extra[2L]
135 c(parking,night,0L,v$cost,parking+night+v$cost,sum(extra))
136}

153location_order <- function(values,entry=FALSE) {
154 rows <- vapply(values,function(v) v$spots[[1L]]$level,integer(1L))
155 numbers <- vapply(values,function(v) v$spots[[1L]]$number,integer(1L))
156 if (entry) {
157 times <- vapply(values,function(v) v$entry,integer(1L))
158 return(values[order(times,rows,numbers,method="radix")])
159 }
160 values[order(rows,numbers,method="radix")]
161}

166execute <- function(t) {
167 cmd <- t[1L]
168 out <- character()
169 emit <- function(lines) out <<- c(out,lines)
170 if (cmd=="LEVEL") {
171 level_no <- length(levels)+1L
172 level <- list()
173 for (index in seq_along(kinds)) {
174 kind <- kinds[index]
175 for (i in seq_len(as.integer(t[index+1L]))) {
176 number <- length(level)+1L
177 id <- paste0("L",level_no,"-",number)
178 s <- object(id=id,level=level_no,number=number,kind=kind,closed=FALSE,plate="")
179 spots[[id]] <- s
180 level[[number]] <- s
181 }
182 }
183 levels[[level_no]] <<- level
184 emit(paste("LEVEL",level_no,"spots",length(level),"MOTO",t[2L],"COMPACT",t[3L],"LARGE",t[4L],"EV",t[5L]))
185 } else if (cmd %in% c("CLOSE","OPEN")) {
186 s <- get_spot(t[2L])
187 if (cmd=="CLOSE") {
188 if (s$closed) fail(paste("spot",t[2L],"is already closed"))
189 if (s$plate!="") fail(paste("spot",t[2L],"is occupied by",s$plate))
190 s$closed <- TRUE
191 } else {
192 if (!s$closed) fail(paste("spot",t[2L],"is not closed"))
193 s$closed <- FALSE
194 }
195 emit(paste(cmd,s$id,s$kind))
196 } else if (cmd=="RATE") {
197 if (is.null(rates[[t[2L]]])) fail(paste("unknown type",t[2L]))
198 if (as.integer(t[4L])<as.integer(t[3L])) fail(paste("cap below rate for",t[2L]))
199 rates[[t[2L]]] <<- as.integer(t[3:4])
200 emit(paste("RATE",t[2L],t[3L],t[4L]))
201 } else if (cmd=="FEE") {
202 if (!(t[2L] %in% names(fees))) fail(paste("unknown fee",t[2L]))
203 fees[t[2L]] <<- as.integer(t[3L])
204 emit(paste("FEE",t[2L],t[3L]))
205 } else if (cmd=="ENTER") {
206 plate <- t[2L]
207 type <- t[3L]
208 if (is.null(rates[[type]])) fail(paste("unknown type",type))
209 if (!is.null(vehicles[[plate]])) fail(paste(plate,"is already parked at",location(vehicles[[plate]])))
210 for (w in queue) if (w$plate==plate) fail(paste(plate,"is already waiting"))
211 chosen <- assignment(type)
212 if (length(chosen)) {
213 v <- park_vehicle(plate,type,chosen)
214 emit(paste("ENTER",plate,type,location(v)))
215 } else if (length(queue)<3L) {
216 queue[[length(queue)+1L]] <<- list(plate=plate,type=type,time=now)
217 emit(paste("WAIT",plate,type,length(queue)))
218 } else {
219 add_stat(type,"refused",1L,date_of(now))
220 emit(paste("FULL",plate,type))
221 }
222 } else if (cmd=="LEAVE") {
223 found <- which(vapply(queue,function(w) w$plate==t[2L],logical(1L)))
224 if (!length(found)) fail(paste(t[2L],"is not waiting"))
225 w <- queue[[found]]
226 queue <<- queue[-found]
227 emit(paste("LEAVE",w$plate,w$type))
228 } else if (cmd %in% c("EXIT","PAY","QUOTE","CHARGE","VALIDATE","MOVE")) {
229 v <- get_vehicle(t[2L])
230 if (cmd=="EXIT") {
231 lost <- length(t)==3L
232 if (lost && !is.null(v$payment)) fail(paste(v$plate,"has already paid"))
233 f <- charges(v,lost)
234 emit(fee_line("EXIT",v,f))
235 record <- list(entry=v$entry,exit=now,location=location(v),type=v$type,total=f[5L],lost=lost)
236 records <- history[[v$plate]]
237 records[[length(records)+1L]] <- record
238 history[[v$plate]] <- records
239 for (s in v$spots) s$plate <- ""
240 rm(list=v$plate,envir=vehicles)
241 add_stat(v$type,"exits",1L,date_of(now))
242 add_stat(v$type,"stay",now-v$entry,date_of(now))
243 fields <- c("parking","night","lost","charge","total")
244 for (index in seq_along(fields)) add_stat(v$type,fields[index],f[index],date_of(now))
245 } else if (cmd=="PAY") {
246 if (!is.null(v$payment)) fail(paste(v$plate,"has already paid"))
247 f <- charges(v)
248 v$payment <- list(time=now,parking=f[1L],night=f[2L])
249 emit(paste("PAY",v$plate,"parking",f[1L],"night",f[2L],"charge",f[4L],"total",f[5L]))
250 } else if (cmd=="QUOTE") {
251 emit(fee_line("QUOTE",v,charges(v)))
252 } else if (cmd=="CHARGE") {
253 if (v$type!="ECAR") fail(paste(v$plate,"is not an electric car"))
254 if (v$spots[[1L]]$kind!="EV") fail(paste(v$plate,"is not at a charger"))
255 if (!is.null(v$payment)) fail(paste(v$plate,"has already paid"))
256 kwh <- as.integer(t[3L])
257 if (v$kwh+kwh>100L) fail(paste("charge limit exceeded for",v$plate))
258 cost <- kwh*fees[["KWH"]]
259 v$kwh <- v$kwh+kwh
260 v$cost <- v$cost+cost
261 emit(paste("CHARGE",v$plate,"kwh",kwh,"cost",cost,"stay_kwh",v$kwh,"stay_cost",v$cost))
262 } else if (cmd=="VALIDATE") {
263 if (!is.null(v$payment)) fail(paste(v$plate,"has already paid"))
264 minutes <- as.integer(t[3L])
265 if (v$validation+minutes>180L) fail(paste("validation limit exceeded for",v$plate))
266 v$validation <- v$validation+minutes
267 emit(paste("VALIDATE",v$plate,minutes,"total",v$validation))
268 } else {
269 if (v$type=="BUS") fail(paste("bus",v$plate,"cannot be moved"))
270 s <- get_spot(t[3L])
271 if (v$spots[[1L]]$id==s$id) fail(paste(v$plate,"is already at",s$id))
272 if (s$closed) fail(paste("spot",s$id,"is closed"))
273 if (s$plate!="") fail(paste("spot",s$id,"is occupied by",s$plate))
274 if (!(s$kind %in% prefs[[v$type]])) fail(paste(v$type,"cannot use",s$kind,"spot",s$id))
275 emit(move_vehicle(v,s))
276 }
277 } else if (cmd=="PASS") {
278 plate <- t[2L]
279 passes[[plate]] <- if (valid_pass(plate,now)) passes[[plate]]+as.integer(t[3L]) else date_of(now)+as.integer(t[3L])-1L
280 emit(paste("PASS",plate,"until",passes[[plate]]))
281 } else if (cmd=="UNPASS") {
282 if (!valid_pass(t[2L],now)) fail(paste(t[2L],"has no pass"))
283 rm(list=t[2L],envir=passes)
284 emit(paste("UNPASS",t[2L]))
285 } else if (cmd=="REBALANCE") {
286 targets <- location_order(Filter(function(v) v$type!="BUS",map_values(vehicles)))
287 moved <- 0L
288 for (v in targets) {
289 current <- match(v$spots[[1L]]$kind,prefs[[v$type]])
290 if (current==1L) next
291 chosen <- assignment(v$type)
292 if (length(chosen) && match(chosen[[1L]]$kind,prefs[[v$type]])<current) {
293 emit(move_vehicle(v,chosen[[1L]]))
294 moved <- moved+1L
295 }
296 }
297 emit(paste("REBALANCE",moved))
298 } else if (cmd=="WHERE") {
299 plate <- t[2L]
300 v <- vehicles[[plate]]
301 if (!is.null(v)) {
302 emit(paste0(paste("WHERE",plate,location(v),v$type,time_text(v$entry),"stay",duration(now-v$entry)),annotation(v)))
303 } else {
304 position <- which(vapply(queue,function(w) w$plate==plate,logical(1L)))
305 records <- history[[plate]]
306 if (length(position)) {
307 emit(paste("WHERE",plate,"waiting",position))
308 } else if (!is.null(records)) {
309 emit(paste("WHERE",plate,"gone",time_text(records[[length(records)]]$exit)))
310 } else {
311 fail(paste("unknown plate",plate))
312 }
313 }
314 } else if (cmd %in% c("LIST","OVERSTAY")) {
315 if (cmd=="LIST" && length(t)==2L && as.integer(t[2L])>length(levels)) fail(paste("unknown level",t[2L]))
316 targets <- map_values(vehicles)
317 if (cmd=="LIST" && length(t)==2L) targets <- Filter(function(v) v$spots[[1L]]$level==as.integer(t[2L]),targets)
318 if (cmd=="OVERSTAY") targets <- Filter(function(v) now-v$entry>=as.integer(t[2L])*60L,targets)
319 targets <- location_order(targets,cmd=="OVERSTAY")
320 header <- if (cmd=="OVERSTAY") paste(cmd,t[2L]) else cmd
321 emit(paste(header,length(targets)))
322 for (v in targets) {
323 text <- paste(location(v),v$plate,v$type,time_text(v$entry))
324 text <- if (cmd=="LIST") paste0(text,annotation(v)) else paste(text,"stay",duration(now-v$entry))
325 emit(text)
326 }
327 } else if (cmd=="MAP") {
328 emit(paste("MAP",length(levels)))
329 for (index in seq_along(levels)) {
330 symbols <- vapply(levels[[index]],function(s) {
331 if (s$closed) return("#")
332 initial <- substr(s$kind,1L,1L)
333 if (s$plate=="") tolower(initial) else initial
334 },character(1L))
335 starts <- seq.int(1L,length(symbols),by=10L)
336 chunks <- vapply(starts,function(i) paste(symbols[i:min(i+9L,length(symbols))],collapse=""),character(1L))
337 emit(paste0("L",index," ",paste(chunks,collapse=" ")))
338 }
339 } else if (cmd=="OCCUPANCY") {
340 emit(paste("OCCUPANCY vehicles",map_size(vehicles),"waiting",length(queue)))
341 total_used <- total_available <- setNames(rep(0L,4L),kinds)
342 total_closed <- 0L
343 for (index in seq_along(levels)) {
344 used <- available <- setNames(rep(0L,4L),kinds)
345 closed_count <- 0L
346 for (s in levels[[index]]) {
347 if (s$closed) closed_count <- closed_count+1L else available[s$kind] <- available[s$kind]+1L
348 if (s$plate!="") used[s$kind] <- used[s$kind]+1L
349 }
350 emit(occupancy_line(paste0("L",index),used,available,closed_count))
351 total_used <- total_used+used
352 total_available <- total_available+available
353 total_closed <- total_closed+closed_count
354 }
355 capacity <- sum(total_available)
356 usage <- if (capacity==0L) 0L else (sum(total_used)*1000L)%/%capacity
357 emit(paste0(occupancy_line("TOTAL",total_used,total_available,total_closed)," rate ",one_decimal(usage),"%"))
358 } else if (cmd=="SIGN") {
359 parts <- "SIGN"
360 for (type in types) {
361 count <- 0L
362 for (level in levels) {
363 run <- 0L
364 for (s in level) {
365 if (type=="BUS") {
366 if (s$kind=="LARGE" && free_spot(s)) {
367 run <- run+1L
368 } else {
369 count <- count+run%/%3L
370 run <- 0L
371 }
372 } else if (s$kind %in% prefs[[type]] && free_spot(s)) {
373 count <- count+1L
374 }
375 }
376 if (type=="BUS") count <- count+run%/%3L
377 }
378 parts <- c(parts,type,if (count==0L) "FULL" else as.character(count))
379 }
380 emit(paste(parts,collapse=" "))
381 } else if (cmd=="RATES") {
382 emit("RATES")
383 for (type in types) emit(paste(type,rates[[type]][1L],rates[[type]][2L]))
384 emit(paste("NIGHT",fees[["NIGHT"]],"LOST",fees[["LOST"]],"KWH",fees[["KWH"]]))
385 } else if (cmd=="PASSES") {
386 plates <- sort(Filter(function(plate) valid_pass(plate,now),ls(passes,all.names=TRUE)),method="radix")
387 emit(paste("PASSES",length(plates)))
388 for (plate in plates) emit(paste0(paste(plate,"until",passes[[plate]]),if (!is.null(vehicles[[plate]])) " IN" else ""))
389 } else if (cmd=="HISTORY") {
390 records <- history[[t[2L]]]
391 if (is.null(records)) fail(paste("no history for",t[2L]))
392 paid <- sum(vapply(records,function(r) r$total,integer(1L)))
393 emit(paste("HISTORY",t[2L],"visits",length(records),"paid",paid))
394 for (r in records) emit(paste0(paste(time_text(r$entry),time_text(r$exit),r$location,r$type,r$total),if (r$lost) " LOST" else ""))
395 } else if (cmd=="REVENUE") {
396 emit("REVENUE")
397 total <- stats[[types[1L]]]*0L
398 for (type in c(types,"TOTAL")) {
399 s <- if (type=="TOTAL") total else stats[[type]]
400 if (type!="TOTAL") total <- total+s
401 emit(paste(type,"exits",s[["exits"]],"parking",s[["parking"]],"night",s[["night"]],"lost",s[["lost"]],"charge",s[["charge"]],"total",s[["total"]]))
402 }
403 } else if (cmd=="DAILY") {
404 date <- as.integer(t[2L])
405 rows <- daily[[as.character(date)]]
406 if (is.null(rows)) rows <- setNames(lapply(types,function(k) c(entries=0L,exits=0L,refused=0L,revenue=0L)),types)
407 total <- c(entries=0L,exits=0L,refused=0L,revenue=0L)
408 lines <- character()
409 for (type in types) {
410 s <- rows[[type]]
411 total <- total+s
412 lines <- c(lines,paste(type,"entries",s[["entries"]],"exits",s[["exits"]],"refused",s[["refused"]],"revenue",s[["revenue"]]))
413 }
414 emit(paste("DAILY",date,"entries",total[["entries"]],"exits",total[["exits"]],"refused",total[["refused"]],"revenue",total[["revenue"]]))
415 emit(lines)
416 } else if (cmd=="STATS") {
417 emit(paste("STATS parked",map_size(vehicles),"peak",peak,"at",if (peak==0L) "-" else time_text(peak_time)))
418 total <- c(entries=0L,exits=0L,refused=0L,stay=0L)
419 for (type in c(types,"ALL")) {
420 s <- if (type=="ALL") total else stats[[type]][names(total)]
421 if (type!="ALL") total <- total+s
422 average <- if (s[["exits"]]==0L) 0L else (s[["stay"]]*10L)%/%s[["exits"]]
423 emit(paste(type,"entries",s[["entries"]],"exits",s[["exits"]],"refused",s[["refused"]],"avg_stay",one_decimal(average)))
424 }
425 }
426 if (cmd %in% c("LEVEL","OPEN","EXIT","MOVE","REBALANCE")) emit(process_queue())
427 out
428}
429input <- file("stdin",open="r")
430n <- as.integer(readLines(input,n=1L))
431for (i in seq_len(n)) {
432 tokens <- strsplit(readLines(input,n=1L)," ",fixed=TRUE)[[1L]]
433 clock <- as.integer(strsplit(tokens[2L],":",fixed=TRUE)[[1L]])
434 now <- (as.integer(tokens[1L])-1L)*1440L+clock[1L]*60L+clock[2L]
435 prefix <- giveups()
436 if (length(prefix)) cat(paste(prefix,collapse="\n"),"\n",sep="")
437 out <- tryCatch(execute(tokens[-c(1L,2L)]),error=function(error) paste("ERROR",conditionMessage(error)))
438 if (length(out)) cat(paste(out,collapse="\n"),"\n",sep="")
439}
