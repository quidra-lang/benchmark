1import 'dart:io';
2import 'dart:math' as math;
3
4const types = ['MOTORCYCLE', 'CAR', 'ECAR', 'VAN', 'BUS'];
5const kinds = ['MOTO', 'COMPACT', 'LARGE', 'EV'];
6const preferences = [[0, 1, 2], [1, 2], [3, 1, 2], [2], [2]];
7int date(int t) => t ~/ 1440 + 1;
8String duration(int t) => '${t ~/ 60}:${(t % 60).toString().padLeft(2, '0')}';
9String time(int t) => '${date(t)}-${duration(t % 1440).padLeft(5, '0')}';
10String decimal(int q) => '${q ~/ 10}.${q % 10}';
11class CommandError implements Exception {
12 final String message;
13 CommandError(this.message);
14}
15Never fail(String s) => throw CommandError(s);
16class Spot {
17 final int level, number, kind, order;
18 bool closed = false;
19 Vehicle? occupant;
20 Spot(this.level, this.number, this.kind, this.order);
21 bool get free => !closed && occupant == null;
22 String get label => 'L$level-$number';
23}
24class Vehicle {
25 final String plate;
26 final int type, entry;
27 List<Spot> spots;
28 int kwh = 0, charge = 0, validation = 0;
29 List<int>? payment;
30 Vehicle(this.plate, this.type, this.entry, this.spots);
31 String get location => type == 4 ? '${spots.first.label}..${spots.last.number}' : spots.first.label;
32}
33class Waiting {
34 final String plate;
35 final int type, joined;
36 Waiting(this.plate, this.type, this.joined);
37}
38class ExitRecord {
39 final String plate, location;
40 final int type, entry, exit;
41 final List<int> fees;
42 final bool lost;
43 ExitRecord(Vehicle v, this.exit, this.fees, this.lost)
44 : plate = v.plate, location = v.location, type = v.type, entry = v.entry;
45}
46class Daily {
47 final entries = List.filled(5, 0), exits = List.filled(5, 0), refused = List.filled(5, 0), revenue = List.filled(5, 0);
48}
49class Garage {
50 int now = 0, peak = 0;
51 int? peakTime;
52 final levels = <List<Spot>>[], spots = <Spot>[];
53 final vehicles = <String, Vehicle>{}, passes = <String, int>{};
54 final queue = <Waiting>[], records = <ExitRecord>[];
55 final rates = [100, 200, 200, 300, 600], caps = [800, 1600, 1600, 2400, 4800];
56 final fees = <String, int>{'NIGHT': 300, 'LOST': 2000, 'KWH': 30};
57 final daily = <int, Daily>{};
58 final entries = List.filled(5, 0), exits = List.filled(5, 0), refused = List.filled(5, 0), stays = List.filled(5, 0);
59 Daily day(int d) => daily.putIfAbsent(d, () => Daily());
60 bool pass(String p) => (passes[p] ?? 0) >= date(now);
61 Vehicle parked(String p) => vehicles[p] ?? fail('$p is not parked');
62 Spot findSpot(String s) {
63 final p = s.substring(1).split('-');
64 final l = int.parse(p[0]), n = int.parse(p[1]);
65 if (l > levels.length || n > levels[l - 1].length) fail('unknown spot $s');
66 return levels[l - 1][n - 1];
67 }
68 List<Spot>? assign(int type) {
69 if (type == 4) {
70 for (final level in levels) {
71 for (var i = 0; i + 2 < level.length; i++) {
72 final triple = level.sublist(i, i + 3);
73 if (triple.every((s) => s.kind == 2 && s.free)) return triple;
74 }
75 }
76 } else {
77 for (final kind in preferences[type]) {
78 for (final s in spots) {
79 if (s.kind == kind && s.free) return [s];
80 }
81 }
82 }
83 return null;
84 }
85 Vehicle park(String p, int type, List<Spot> place) {
86 final v = Vehicle(p, type, now, place);
87 vehicles[p] = v;
88 for (final s in place) s.occupant = v;
89 entries[type]++;
90 day(date(now)).entries[type]++;
91 if (vehicles.length > peak) {
92 peak = vehicles.length;
93 peakTime = now;
94 }
95 return v;
96 }
97 void refuse(int type, int t) {
98 refused[type]++;
99 day(date(t)).refused[type]++;
100 }
101 void giveup(StringBuffer out) {
102 for (final w in queue.toList()) {
103 if (now - w.joined >= 30) {
104 queue.remove(w);
105 refuse(w.type, w.joined + 30);
106 out.writeln('GIVEUP ${w.plate} ${types[w.type]} ${time(w.joined + 30)}');
107 }
108 }
109 }
110 void admit(StringBuffer out) {
111 for (final w in queue.toList()) {
112 final place = assign(w.type);
113 if (place != null) {
114 queue.remove(w);
115 final v = park(w.plate, w.type, place);
116 out.writeln('ADMIT ${w.plate} ${types[w.type]} ${v.location} waited ${duration(now - w.joined)}');
117 }
118 }
119 }
120 List<int> interval(Vehicle v, int a, int free) {
121 final total = now - a;
122 if (pass(v.plate) || total <= free) return [0, 0];
123 var parking = 0;
124 for (var offset = 0; offset < total; offset += 1440) {
125 final bill = math.min(total - offset, 1440) - (offset == 0 ? free : 0);
126 parking += math.min(((bill + 29) ~/ 30) * rates[v.type], caps[v.type]);
127 }
128 return [parking, (date(now) - date(a)) * fees['NIGHT']!];
129 }
130 List<int> charges(Vehicle v, bool lost) {
131 final payment = v.payment;
132 final f = interval(v, payment == null ? v.entry : payment[0], payment == null ? 15 + v.validation : 15);
133 final parking = f[0] + (payment == null ? 0 : payment[1]);
134 final night = f[1] + (payment == null ? 0 : payment[2]);
135 final loss = lost ? fees['LOST']! : 0;
136 final total = parking + night + loss + v.charge;
137 return [parking, night, loss, v.charge, total, payment == null ? total : f[0] + f[1]];
138 }
139 String annotation(Vehicle v) => '${pass(v.plate) ? ' PASS' : ''}${v.validation == 0 ? '' : ' VAL ${v.validation}'}${v.kwh == 0 ? '' : ' KWH ${v.kwh}'}${v.payment == null ? '' : ' PAID'}';
140 List<Vehicle> ordered() => vehicles.values.toList()..sort((a, b) => a.spots.first.order.compareTo(b.spots.first.order));
141 void feeLine(StringBuffer out, String label, Vehicle v, List<int> f) => out.writeln('$label ${v.plate} ${v.location} stay ${duration(now - v.entry)} parking ${f[0]} night ${f[1]} lost ${f[2]} charge ${f[3]} total ${f[4]} due ${f[5]}');
142 void move(Vehicle v, List<Spot> place, StringBuffer out) {
143 final old = v.location;
144 for (final s in v.spots) s.occupant = null;
145 v.spots = place;
146 for (final s in place) s.occupant = v;
147 out.writeln('MOVE ${v.plate} $old -> ${v.location}');
148 }
149 void execute(List<String> a, StringBuffer out) {
150 final op = a[2], args = a.sublist(3);
151 final p = args.isEmpty ? '' : args[0];
152 if (op == 'LEVEL') {
153 final counts = args.map(int.parse).toList(), level = <Spot>[];
154 final l = levels.length + 1;
155 for (var k = 0; k < 4; k++) {
156 for (var j = 0; j < counts[k]; j++) {
157 final s = Spot(l, level.length + 1, k, spots.length);
158 level.add(s);
159 spots.add(s);
160 }
161 }
162 levels.add(level);
163 out.writeln('LEVEL $l spots ${level.length} MOTO ${counts[0]} COMPACT ${counts[1]} LARGE ${counts[2]} EV ${counts[3]}');
164 admit(out);
165 } else if (op == 'CLOSE' || op == 'OPEN') {
166 final s = findSpot(p);
167 if (op == 'CLOSE') {
168 if (s.closed) fail('spot $p is already closed');
169 if (s.occupant != null) fail('spot $p is occupied by ${s.occupant!.plate}');
170 s.closed = true;
171 } else {
172 if (!s.closed) fail('spot $p is not closed');
173 s.closed = false;
174 }
175 out.writeln('$op $p ${kinds[s.kind]}');
176 if (op == 'OPEN') admit(out);
177 } else if (op == 'RATE') {
178 final k = types.indexOf(p);
179 if (k < 0) fail('unknown type $p');
180 final r = int.parse(args[1]), c = int.parse(args[2]);
181 if (c < r) fail('cap below rate for $p');
182 rates[k] = r;
183 caps[k] = c;
184 out.writeln('RATE $p $r $c');
185 } else if (op == 'FEE') {
186 if (!fees.containsKey(p)) fail('unknown fee $p');
187 fees[p] = int.parse(args[1]);
188 out.writeln('FEE $p ${fees[p]}');
189 } else if (op == 'ENTER') {
190 final k = types.indexOf(args[1]);
191 if (k < 0) fail('unknown type ${args[1]}');
192 final existing = vehicles[p];
193 if (existing != null) fail('$p is already parked at ${existing.location}');
194 if (queue.any((w) => w.plate == p)) fail('$p is already waiting');
195 final place = assign(k);
196 if (place != null) {
197 final v = park(p, k, place);
198 out.writeln('ENTER $p ${types[k]} ${v.location}');
199 } else if (queue.length < 3) {
200 queue.add(Waiting(p, k, now));
201 out.writeln('WAIT $p ${types[k]} ${queue.length}');
202 } else {
203 refuse(k, now);
204 out.writeln('FULL $p ${types[k]}');
205 }
206 } else if (op == 'LEAVE') {
207 final i = queue.indexWhere((w) => w.plate == p);
208 if (i < 0) fail('$p is not waiting');
209 final w = queue.removeAt(i);
210 out.writeln('LEAVE $p ${types[w.type]}');
211 } else if (['EXIT', 'PAY', 'QUOTE', 'CHARGE', 'VALIDATE'].contains(op)) {
212 final v = parked(p);
213 if (op == 'EXIT') {
214 final lost = args.length > 1;
215 if (lost && v.payment != null) fail('$p has already paid');
216 final f = charges(v, lost);
217 feeLine(out, op, v, f);
218 records.add(ExitRecord(v, now, f, lost));
219 exits[v.type]++;
220 stays[v.type] += now - v.entry;
221 day(date(now)).exits[v.type]++;
222 day(date(now)).revenue[v.type] += f[4];
223 for (final s in v.spots) s.occupant = null;
224 vehicles.remove(p);
225 admit(out);
226 } else if (op == 'QUOTE') {
227 feeLine(out, op, v, charges(v, false));
228 } else if (op == 'PAY') {
229 if (v.payment != null) fail('$p has already paid');
230 final f = charges(v, false);
231 v.payment = [now, f[0], f[1]];
232 out.writeln('PAY $p parking ${f[0]} night ${f[1]} charge ${f[3]} total ${f[4]}');
233 } else if (op == 'CHARGE') {
234 if (v.type != 2) fail('$p is not an electric car');
235 if (v.spots.first.kind != 3) fail('$p is not at a charger');
236 if (v.payment != null) fail('$p has already paid');
237 final kwh = int.parse(args[1]);
238 if (v.kwh + kwh > 100) fail('charge limit exceeded for $p');
239 final cost = kwh * fees['KWH']!;
240 v.kwh += kwh;
241 v.charge += cost;
242 out.writeln('CHARGE $p kwh $kwh cost $cost stay_kwh ${v.kwh} stay_cost ${v.charge}');
243 } else {
244 if (v.payment != null) fail('$p has already paid');
245 final n = int.parse(args[1]);
246 if (v.validation + n > 180) fail('validation limit exceeded for $p');
247 v.validation += n;
248 out.writeln('VALIDATE $p $n total ${v.validation}');
249 }
250 } else if (op == 'PASS') {
251 passes[p] = (pass(p) ? passes[p]! : date(now) - 1) + int.parse(args[1]);
252 out.writeln('PASS $p until ${passes[p]}');
253 } else if (op == 'UNPASS') {
254 if (!pass(p)) fail('$p has no pass');
255 passes.remove(p);
256 out.writeln('UNPASS $p');
257 } else if (op == 'MOVE') {
258 final v = parked(p);
259 if (v.type == 4) fail('bus $p cannot be moved');
260 final dest = args[1], s = findSpot(args[1]);
261 if (s == v.spots.first) fail('$p is already at $dest');
262 if (s.closed) fail('spot $dest is closed');
263 if (s.occupant != null) fail('spot $dest is occupied by ${s.occupant!.plate}');
264 if (!preferences[v.type].contains(s.kind)) fail('${types[v.type]} cannot use ${kinds[s.kind]} spot $dest');
265 move(v, [s], out);
266 admit(out);
267 } else if (op == 'REBALANCE') {
268 var moved = 0;
269 for (final v in ordered().where((v) => v.type != 4)) {
270 final pref = preferences[v.type], place = assign(v.type);
271 if (place != null && pref.indexOf(place.first.kind) < pref.indexOf(v.spots.first.kind)) {
272 move(v, place, out);
273 moved++;
274 }
275 }
276 out.writeln('REBALANCE $moved');
277 admit(out);
278 } else if (op == 'WHERE') {
279 final v = vehicles[p], i = queue.indexWhere((w) => w.plate == p);
280 final last = records.reversed.where((r) => r.plate == p).firstOrNull;
281 if (v != null) {
282 out.writeln('WHERE $p ${v.location} ${types[v.type]} ${time(v.entry)} stay ${duration(now - v.entry)}${annotation(v)}');
283 } else if (i >= 0) {
284 out.writeln('WHERE $p waiting ${i + 1}');
285 } else if (last != null) {
286 out.writeln('WHERE $p gone ${time(last.exit)}');
287 } else {
288 fail('unknown plate $p');
289 }
290 } else if (op == 'LIST') {
291 final l = args.isEmpty ? null : int.parse(p);
292 if (l != null && l > levels.length) fail('unknown level $l');
293 final list = ordered().where((v) => l == null || v.spots.first.level == l).toList();
294 out.writeln('LIST ${list.length}');
295 for (final v in list) out.writeln('${v.location} ${v.plate} ${types[v.type]} ${time(v.entry)}${annotation(v)}');
296 } else if (op == 'MAP') {
297 out.writeln('MAP ${levels.length}');
298 for (var i = 0; i < levels.length; i++) {
299 final symbols = <String>[];
300 for (var j = 0; j < levels[i].length; j++) {
301 if (j > 0 && j % 10 == 0) symbols.add(' ');
302 final s = levels[i][j];
303 symbols.add(s.closed ? '#' : s.occupant != null ? 'MCLE'[s.kind] : 'mcle'[s.kind]);
304 }
305 out.writeln('L${i + 1} ${symbols.join()}');
306 }
307 } else if (op == 'OCCUPANCY') {
308 final allUsed = List.filled(4, 0), allAvail = List.filled(4, 0);
309 var allClosed = 0;
310 out.writeln('OCCUPANCY vehicles ${vehicles.length} waiting ${queue.length}');
311 for (var i = 0; i < levels.length; i++) {
312 final used = List.filled(4, 0), avail = List.filled(4, 0);
313 var closed = 0;
314 for (final s in levels[i]) {
315 if (s.closed)
316 closed++;
317 else
318 avail[s.kind]++;
319 if (s.occupant != null) used[s.kind]++;
320 }
321 out.writeln('L${i + 1} ${List.generate(4, (k) => '${kinds[k]} ${used[k]}/${avail[k]}').join(' ')} closed $closed');
322 for (var k = 0; k < 4; k++) {
323 allUsed[k] += used[k];
324 allAvail[k] += avail[k];
325 }
326 allClosed += closed;
327 }
328 final av = allAvail.fold(0, (a, b) => a + b), us = allUsed.fold(0, (a, b) => a + b);
329 out.writeln('TOTAL ${List.generate(4, (k) => '${kinds[k]} ${allUsed[k]}/${allAvail[k]}').join(' ')} closed $allClosed rate ${decimal(av == 0 ? 0 : us * 1000 ~/ av)}%');
330 } else if (op == 'SIGN') {
331 final counts = List.filled(5, 0);
332 for (var k = 0; k < 4; k++) counts[k] = spots.where((s) => s.free && preferences[k].contains(s.kind)).length;
333 for (final level in levels) {
334 var run = 0;
335 for (final s in level) {
336 if (s.free && s.kind == 2)
337 run++;
338 else {
339 counts[4] += run ~/ 3;
340 run = 0;
341 }
342 }
343 counts[4] += run ~/ 3;
344 }
345 out.writeln('SIGN ${List.generate(5, (k) => '${types[k]} ${counts[k] == 0 ? 'FULL' : counts[k]}').join(' ')}');
346 } else if (op == 'RATES') {
347 out.writeln('RATES');
348 for (var k = 0; k < 5; k++) out.writeln('${types[k]} ${rates[k]} ${caps[k]}');
349 out.writeln('NIGHT ${fees['NIGHT']} LOST ${fees['LOST']} KWH ${fees['KWH']}');
350 } else if (op == 'PASSES') {
351 final ps = passes.keys.where(pass).toList()..sort();
352 out.writeln('PASSES ${ps.length}');
353 for (final plate in ps) out.writeln('$plate until ${passes[plate]}${vehicles.containsKey(plate) ? ' IN' : ''}');
354 } else if (op == 'OVERSTAY') {
355 final hours = int.parse(p);
356 final vs = vehicles.values.where((v) => now - v.entry >= hours * 60).toList();
357 vs.sort((a, b) {
358 final c = a.entry.compareTo(b.entry);
359 return c != 0 ? c : a.spots.first.order.compareTo(b.spots.first.order);
360 });
361 out.writeln('OVERSTAY $hours ${vs.length}');
362 for (final v in vs) out.writeln('${v.location} ${v.plate} ${types[v.type]} ${time(v.entry)} stay ${duration(now - v.entry)}');
363 } else if (op == 'HISTORY') {
364 final rs = records.where((r) => r.plate == p).toList();
365 if (rs.isEmpty) fail('no history for $p');
366 out.writeln('HISTORY $p visits ${rs.length} paid ${rs.fold(0, (s, r) => s + r.fees[4])}');
367 for (final r in rs) out.writeln('${time(r.entry)} ${time(r.exit)} ${r.location} ${types[r.type]} ${r.fees[4]}${r.lost ? ' LOST' : ''}');
368 } else if (op == 'REVENUE') {
369 final totals = List.generate(6, (_) => List.filled(6, 0));
370 for (final r in records) {
371 for (final k in [r.type, 5]) {
372 totals[k][0]++;
373 for (var j = 0; j < 5; j++) totals[k][j + 1] += r.fees[j];
374 }
375 }
376 out.writeln('REVENUE');
377 for (var k = 0; k < 6; k++) {
378 final f = totals[k];
379 out.writeln('${k == 5 ? 'TOTAL' : types[k]} exits ${f[0]} parking ${f[1]} night ${f[2]} lost ${f[3]} charge ${f[4]} total ${f[5]}');
380 }
381 } else if (op == 'DAILY') {
382 final d = int.parse(p), s = daily[d] ?? Daily();
383 for (final k in [-1, 0, 1, 2, 3, 4]) {
384 int value(List<int> xs) => k < 0 ? xs.fold(0, (a, b) => a + b) : xs[k];
385 out.writeln('${k < 0 ? 'DAILY $d' : types[k]} entries ${value(s.entries)} exits ${value(s.exits)} refused ${value(s.refused)} revenue ${value(s.revenue)}');
386 }
387 } else if (op == 'STATS') {
388 out.writeln('STATS parked ${vehicles.length} peak $peak at ${peakTime == null ? '-' : time(peakTime!)}');
389 for (var k = 0; k < 6; k++) {
390 int value(List<int> xs) => k == 5 ? xs.fold(0, (a, b) => a + b) : xs[k];
391 final e = value(exits), stay = value(stays);
392 out.writeln('${k == 5 ? 'ALL' : types[k]} entries ${value(entries)} exits $e refused ${value(refused)} avg_stay ${decimal(e == 0 ? 0 : stay * 10 ~/ e)}');
393 }
394 }
395 }
396}
397void main() {
398 final garage = Garage(), output = StringBuffer();
399 final n = int.parse(stdin.readLineSync()!);
400 for (var i = 0; i < n; i++) {
401 final a = stdin.readLineSync()!.split(' ');
402 final hhmm = a[1].split(':');
403 garage.now = (int.parse(a[0]) - 1) * 1440 + int.parse(hhmm[0]) * 60 + int.parse(hhmm[1]);
404 garage.giveup(output);
405 final draft = StringBuffer();
406 try {
407 garage.execute(a, draft);
408 output.write(draft);
409 } on CommandError catch (e) {
410 output.writeln('ERROR ${e.message}');
411 }
412 }
413 stdout.write(output);
414}
