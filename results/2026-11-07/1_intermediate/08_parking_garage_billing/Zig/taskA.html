1const std = @import("std");
2const Allocator = std.mem.Allocator;
3const Error = error{BadCommand} || Allocator.Error || std.Io.Writer.Error;
4const types = [_][]const u8{ "MOTORCYCLE", "CAR", "ECAR", "VAN", "BUS" };
5const kinds = [_][]const u8{ "MOTO", "COMPACT", "LARGE", "EV" };
6const preferences = [_][]const usize{ &.{ 0, 1, 2 }, &.{ 1, 2 }, &.{ 3, 1, 2 }, &.{2}, &.{2} };
7fn integer(s: []const u8) i64 {
8 return std.fmt.parseInt(i64, s, 10) catch unreachable;
9}
10fn index(names: []const []const u8, s: []const u8) ?usize {
11 for (names, 0..) |name, i| {
12 if (std.mem.eql(u8, name, s)) return i;
13 }
14 return null;
15}
16fn date(t: i64) i64 {
17 return @divTrunc(t, 1440) + 1;
18}

40const Waiting = struct {
41 plate: []const u8,
42 kind: usize,
43 joined: i64,
44};

46 plate: []const u8,

54const Stats = struct {
55 entries: [5]i64 = @splat(0),
56 exits: [5]i64 = @splat(0),
57 refused: [5]i64 = @splat(0),
58 revenue: [5]i64 = @splat(0),
59 stays: [5]i64 = @splat(0),

61const Garage = struct {
62 gpa: Allocator,
63 out: *std.Io.Writer,
64 error_message: []const u8 = "",
65 now: i64 = 0,
66 peak: usize = 0,
67 peak_time: ?i64 = null,
68 spots: [1080]Spot = undefined,
69 spot_count: usize = 0,
70 levels: [9]struct { first: usize, end: usize } = undefined,
71 level_count: usize = 0,
72 vehicles: std.StringArrayHashMapUnmanaged(*Vehicle) = .empty,
73 queue: std.ArrayList(Waiting) = .empty,
74 records: std.ArrayList(Record) = .empty,
75 passes: std.StringHashMapUnmanaged(i64) = .empty,
76 rates: [5]i64 = .{ 100, 200, 200, 300, 600 },
77 caps: [5]i64 = .{ 800, 1600, 1600, 2400, 4800 },
78 amounts: [3]i64 = .{ 300, 2000, 30 },
79 stats: Stats = .{},
80 daily: [1000]Stats = @splat(.{}),
81
82 fn fail(self: *Garage, comptime fmt: []const u8, args: anytype) Error {
83 self.error_message = std.fmt.allocPrint(self.gpa, fmt, args) catch return error.OutOfMemory;
84 return error.BadCommand;
85 }
86 fn duration(self: *Garage, t: i64) Allocator.Error![]const u8 {
87 const u: u64 = @intCast(t);
88 return std.fmt.allocPrint(self.gpa, "{d}:{d:0>2}", .{ u / 60, u % 60 });
89 }
90 fn time(self: *Garage, t: i64) Allocator.Error![]const u8 {
91 const u: u64 = @intCast(@mod(t, 1440));
92 return std.fmt.allocPrint(self.gpa, "{d}-{d:0>2}:{d:0>2}", .{ date(t), u / 60, u % 60 });
93 }
94 fn decimal(self: *Garage, q: i64) Allocator.Error![]const u8 {
95 return std.fmt.allocPrint(self.gpa, "{d}.{d}", .{ @divTrunc(q, 10), @mod(q, 10) });
96 }
97 fn pass(self: *Garage, plate: []const u8) bool {
98 return (self.passes.get(plate) orelse 0) >= date(self.now);
99 }
100 fn parked(self: *Garage, plate: []const u8) Error!*Vehicle {
101 return self.vehicles.get(plate) orelse return self.fail("{s} is not parked", .{plate});
102 }
103 fn waiting(self: *Garage, plate: []const u8) ?usize {
104 for (self.queue.items, 0..) |w, i| {
105 if (std.mem.eql(u8, w.plate, plate)) return i;
106 }
107 return null;
108 }
109 fn spotLabel(self: *Garage, start: usize) Allocator.Error![]const u8 {
110 const s = self.spots[start];
111 return std.fmt.allocPrint(self.gpa, "L{d}-{d}", .{ s.level, s.number });
112 }
113 fn location(self: *Garage, v: *const Vehicle) Allocator.Error![]const u8 {
114 const s = self.spots[v.start];
115 if (v.kind == 4) return std.fmt.allocPrint(self.gpa, "L{d}-{d}..{d}", .{ s.level, s.number, s.number + 2 });
116 return self.spotLabel(v.start);
117 }
118 fn findSpot(self: *Garage, s: []const u8) Error!usize {
119 const dash = std.mem.findScalar(u8, s, '-').?;
120 const l: usize = @intCast(integer(s[1..dash]));
121 const n: usize = @intCast(integer(s[dash + 1 ..]));
122 if (l > self.level_count) return self.fail("unknown spot {s}", .{s});
123 const level = self.levels[l - 1];
124 if (n > level.end - level.first) return self.fail("unknown spot {s}", .{s});
125 return level.first + n - 1;
126 }
127 fn free(self: *Garage, i: usize) bool {
128 return !self.spots[i].closed and self.spots[i].occupant == null;
129 }
130 fn assign(self: *Garage, kind: usize) ?usize {
131 if (kind == 4) {
132 for (self.levels[0..self.level_count]) |l| {
133 var i = l.first;
134 while (i + 2 < l.end) : (i += 1) {
135 var ok = true;
136 for (i..i + 3) |j| {
137 if (self.spots[j].kind != 2 or !self.free(j)) ok = false;
138 }
139 if (ok) return i;
140 }
141 }
142 } else {
143 for (preferences[kind]) |k| {
144 for (0..self.spot_count) |i| {
145 if (self.spots[i].kind == k and self.free(i)) return i;
146 }
147 }
148 }
149 return null;
150 }
151 fn park(self: *Garage, plate: []const u8, kind: usize, start: usize) Error!*Vehicle {
152 const v = try self.gpa.create(Vehicle);
153 v.* = .{ .plate = plate, .kind = kind, .entry = self.now, .start = start };
154 try self.vehicles.put(self.gpa, plate, v);
155 for (start..start + v.width()) |i| self.spots[i].occupant = v;
156 self.stats.entries[kind] += 1;
157 self.daily[@intCast(date(self.now))].entries[kind] += 1;
158 if (self.vehicles.count() > self.peak) {
159 self.peak = self.vehicles.count();
160 self.peak_time = self.now;
161 }
162 return v;
163 }
164 fn refuse(self: *Garage, kind: usize, t: i64) void {
165 self.stats.refused[kind] += 1;
166 self.daily[@intCast(date(t))].refused[kind] += 1;
167 }
168 fn giveup(self: *Garage) Error!void {
169 var i: usize = 0;
170 while (i < self.queue.items.len) {
171 const w = self.queue.items[i];
172 if (self.now - w.joined >= 30) {
173 _ = self.queue.orderedRemove(i);
174 self.refuse(w.kind, w.joined + 30);
175 try self.out.print("GIVEUP {s} {s} {s}\n", .{ w.plate, types[w.kind], try self.time(w.joined + 30) });
176 } else i += 1;
177 }
178 }
179 fn admit(self: *Garage) Error!void {
180 var i: usize = 0;
181 while (i < self.queue.items.len) {
182 const w = self.queue.items[i];
183 if (self.assign(w.kind)) |place| {
184 _ = self.queue.orderedRemove(i);
185 const v = try self.park(w.plate, w.kind, place);
186 try self.out.print("ADMIT {s} {s} {s} waited {s}\n", .{ w.plate, types[w.kind], try self.location(v), try self.duration(self.now - w.joined) });
187 } else i += 1;
188 }
189 }
190 fn interval(self: *Garage, v: *Vehicle, a: i64, free_time: i64) [2]i64 {
191 const total = self.now - a;
192 if (self.pass(v.plate) or total <= free_time) return .{ 0, 0 };
193 var parking: i64 = 0;
194 var offset: i64 = 0;
195 while (offset < total) : (offset += 1440) {
196 const bill = @min(total - offset, 1440) - (if (offset == 0) free_time else @as(i64, 0));
197 parking += @min(@divTrunc(bill + 29, 30) * self.rates[v.kind], self.caps[v.kind]);
198 }
199 return .{ parking, (date(self.now) - date(a)) * self.amounts[0] };
200 }
201 fn charges(self: *Garage, v: *Vehicle, lost: bool) [6]i64 {
202 var parking: i64 = 0;
203 var night: i64 = 0;
204 var due: i64 = 0;
205 const loss = if (lost) self.amounts[1] else 0;
206 if (v.payment) |p| {
207 const f = self.interval(v, p[0], 15);
208 parking = p[1] + f[0];
209 night = p[2] + f[1];
210 due = f[0] + f[1];
211 } else {
212 const f = self.interval(v, v.entry, 15 + v.validation);
213 parking = f[0];
214 night = f[1];
215 due = parking + night + loss + v.charge;
216 }
217 return .{ parking, night, loss, v.charge, parking + night + loss + v.charge, due };
218 }
219 fn annotation(self: *Garage, v: *Vehicle) Error!void {
220 if (self.pass(v.plate)) try self.out.writeAll(" PASS");
221 if (v.validation != 0) try self.out.print(" VAL {d}", .{v.validation});
222 if (v.kwh != 0) try self.out.print(" KWH {d}", .{v.kwh});
223 if (v.payment != null) try self.out.writeAll(" PAID");
224 }
225 fn ordered(self: *Garage) Error![]*Vehicle {
226 const vs = try self.gpa.dupe(*Vehicle, self.vehicles.values());
227 std.mem.sort(*Vehicle, vs, {}, locationLess);
228 return vs;
229 }
230 fn feeLine(self: *Garage, label: []const u8, v: *Vehicle, f: [6]i64) Error!void {
231 try self.out.print("{s} {s} {s} stay {s} parking {d} night {d} lost {d} charge {d} total {d} due {d}\n", .{
232 label, v.plate, try self.location(v), try self.duration(self.now - v.entry), f[0], f[1], f[2], f[3], f[4], f[5],
233 });
234 }
235 fn move(self: *Garage, v: *Vehicle, dest: usize) Error!void {
236 const old = try self.location(v);
237 self.spots[v.start].occupant = null;
238 v.start = dest;
239 self.spots[dest].occupant = v;
240 try self.out.print("MOVE {s} {s} -> {s}\n", .{ v.plate, old, try self.location(v) });
241 }
242 fn execute(self: *Garage, a: []const []const u8) Error!void {
243 const op = a[2];
244 const args = a[3..];
245 const p: []const u8 = if (args.len > 0) args[0] else "";
246 if (std.mem.eql(u8, op, "LEVEL")) {
247 var counts: [4]usize = undefined;
248 const first = self.spot_count;
249 for (0..4) |k| {
250 counts[k] = @intCast(integer(args[k]));
251 for (0..counts[k]) |_| {
252 self.spots[self.spot_count] = .{ .level = self.level_count + 1, .number = self.spot_count - first + 1, .kind = k };
253 self.spot_count += 1;
254 }
255 }
256 self.levels[self.level_count] = .{ .first = first, .end = self.spot_count };
257 self.level_count += 1;
258 try self.out.print("LEVEL {d} spots {d} MOTO {d} COMPACT {d} LARGE {d} EV {d}\n", .{ self.level_count, self.spot_count - first, counts[0], counts[1], counts[2], counts[3] });
259 try self.admit();
260 } else if (std.mem.eql(u8, op, "CLOSE") or std.mem.eql(u8, op, "OPEN")) {
261 const s = &self.spots[try self.findSpot(p)];
262 if (std.mem.eql(u8, op, "CLOSE")) {
263 if (s.closed) return self.fail("spot {s} is already closed", .{p});
264 if (s.occupant) |v| return self.fail("spot {s} is occupied by {s}", .{ p, v.plate });
265 s.closed = true;
266 } else {
267 if (!s.closed) return self.fail("spot {s} is not closed", .{p});
268 s.closed = false;
269 }
270 try self.out.print("{s} {s} {s}\n", .{ op, p, kinds[s.kind] });
271 if (std.mem.eql(u8, op, "OPEN")) try self.admit();
272 } else if (std.mem.eql(u8, op, "RATE")) {
273 const k = index(&types, p) orelse return self.fail("unknown type {s}", .{p});
274 const r = integer(args[1]);
275 const c = integer(args[2]);
276 if (c < r) return self.fail("cap below rate for {s}", .{p});
277 self.rates[k] = r;
278 self.caps[k] = c;
279 try self.out.print("RATE {s} {d} {d}\n", .{ p, r, c });
280 } else if (std.mem.eql(u8, op, "FEE")) {
281 const names = [_][]const u8{ "NIGHT", "LOST", "KWH" };
282 const k = index(&names, p) orelse return self.fail("unknown fee {s}", .{p});
283 self.amounts[k] = integer(args[1]);
284 try self.out.print("FEE {s} {d}\n", .{ p, self.amounts[k] });
285 } else if (std.mem.eql(u8, op, "ENTER")) {
286 const k = index(&types, args[1]) orelse return self.fail("unknown type {s}", .{args[1]});
287 if (self.vehicles.get(p)) |v| return self.fail("{s} is already parked at {s}", .{ p, try self.location(v) });
288 if (self.waiting(p) != null) return self.fail("{s} is already waiting", .{p});
289 if (self.assign(k)) |place| {
290 const v = try self.park(p, k, place);
291 try self.out.print("ENTER {s} {s} {s}\n", .{ p, types[k], try self.location(v) });
292 } else if (self.queue.items.len < 3) {
293 try self.queue.append(self.gpa, .{ .plate = p, .kind = k, .joined = self.now });
294 try self.out.print("WAIT {s} {s} {d}\n", .{ p, types[k], self.queue.items.len });
295 } else {
296 self.refuse(k, self.now);
297 try self.out.print("FULL {s} {s}\n", .{ p, types[k] });
298 }
299 } else if (std.mem.eql(u8, op, "LEAVE")) {
300 const i = self.waiting(p) orelse return self.fail("{s} is not waiting", .{p});
301 const w = self.queue.orderedRemove(i);
302 try self.out.print("LEAVE {s} {s}\n", .{ p, types[w.kind] });
303 } else if (std.mem.eql(u8, op, "EXIT") or std.mem.eql(u8, op, "PAY") or std.mem.eql(u8, op, "QUOTE") or std.mem.eql(u8, op, "CHARGE") or std.mem.eql(u8, op, "VALIDATE")) {
304 const v = try self.parked(p);
305 if (std.mem.eql(u8, op, "EXIT")) {
306 const lost = args.len > 1;
307 if (lost and v.payment != null) return self.fail("{s} has already paid", .{p});
308 const f = self.charges(v, lost);
309 try self.feeLine(op, v, f);
310 try self.records.append(self.gpa, .{ .plate = p, .location = try self.location(v), .kind = v.kind, .entry = v.entry, .exit = self.now, .fees = f, .lost = lost });
311 self.stats.exits[v.kind] += 1;
312 self.stats.stays[v.kind] += self.now - v.entry;
313 self.daily[@intCast(date(self.now))].exits[v.kind] += 1;
314 self.daily[@intCast(date(self.now))].revenue[v.kind] += f[4];
315 for (v.start..v.start + v.width()) |i| self.spots[i].occupant = null;
316 _ = self.vehicles.swapRemove(p);
317 try self.admit();
318 } else if (std.mem.eql(u8, op, "QUOTE")) {
319 try self.feeLine(op, v, self.charges(v, false));
320 } else if (std.mem.eql(u8, op, "PAY")) {
321 if (v.payment != null) return self.fail("{s} has already paid", .{p});
322 const f = self.charges(v, false);
323 v.payment = .{ self.now, f[0], f[1] };
324 try self.out.print("PAY {s} parking {d} night {d} charge {d} total {d}\n", .{ p, f[0], f[1], f[3], f[4] });
325 } else if (std.mem.eql(u8, op, "CHARGE")) {
326 if (v.kind != 2) return self.fail("{s} is not an electric car", .{p});
327 if (self.spots[v.start].kind != 3) return self.fail("{s} is not at a charger", .{p});
328 if (v.payment != null) return self.fail("{s} has already paid", .{p});
329 const kwh = integer(args[1]);
330 if (v.kwh + kwh > 100) return self.fail("charge limit exceeded for {s}", .{p});
331 const cost = kwh * self.amounts[2];
332 v.kwh += kwh;
333 v.charge += cost;
334 try self.out.print("CHARGE {s} kwh {d} cost {d} stay_kwh {d} stay_cost {d}\n", .{ p, kwh, cost, v.kwh, v.charge });
335 } else {
336 if (v.payment != null) return self.fail("{s} has already paid", .{p});
337 const n = integer(args[1]);
338 if (v.validation + n > 180) return self.fail("validation limit exceeded for {s}", .{p});
339 v.validation += n;
340 try self.out.print("VALIDATE {s} {d} total {d}\n", .{ p, n, v.validation });
341 }
342 } else if (std.mem.eql(u8, op, "PASS")) {
343 const last = (if (self.pass(p)) self.passes.get(p).? else date(self.now) - 1) + integer(args[1]);
344 try self.passes.put(self.gpa, p, last);
345 try self.out.print("PASS {s} until {d}\n", .{ p, last });
346 } else if (std.mem.eql(u8, op, "UNPASS")) {
347 if (!self.pass(p)) return self.fail("{s} has no pass", .{p});
348 _ = self.passes.remove(p);
349 try self.out.print("UNPASS {s}\n", .{p});
350 } else if (std.mem.eql(u8, op, "MOVE")) {
351 const v = try self.parked(p);
352 if (v.kind == 4) return self.fail("bus {s} cannot be moved", .{p});
353 const dest = try self.findSpot(args[1]);
354 const s = self.spots[dest];
355 if (dest == v.start) return self.fail("{s} is already at {s}", .{ p, args[1] });
356 if (s.closed) return self.fail("spot {s} is closed", .{args[1]});
357 if (s.occupant) |other| return self.fail("spot {s} is occupied by {s}", .{ args[1], other.plate });
358 if (std.mem.findScalar(usize, preferences[v.kind], s.kind) == null) return self.fail("{s} cannot use {s} spot {s}", .{ types[v.kind], kinds[s.kind], args[1] });
359 try self.move(v, dest);
360 try self.admit();
361 } else if (std.mem.eql(u8, op, "REBALANCE")) {
362 var moved: usize = 0;
363 for (try self.ordered()) |v| {
364 if (v.kind == 4) continue;
365 if (self.assign(v.kind)) |place| {
366 const pref = preferences[v.kind];
367 if (std.mem.findScalar(usize, pref, self.spots[place].kind).? < std.mem.findScalar(usize, pref, self.spots[v.start].kind).?) {
368 try self.move(v, place);
369 moved += 1;
370 }
371 }
372 }
373 try self.out.print("REBALANCE {d}\n", .{moved});
374 try self.admit();
375 } else if (std.mem.eql(u8, op, "WHERE")) {
376 if (self.vehicles.get(p)) |v| {
377 try self.out.print("WHERE {s} {s} {s} {s} stay {s}", .{ p, try self.location(v), types[v.kind], try self.time(v.entry), try self.duration(self.now - v.entry) });
378 try self.annotation(v);
379 try self.out.writeByte('\n');
380 } else if (self.waiting(p)) |i| {
381 try self.out.print("WHERE {s} waiting {d}\n", .{ p, i + 1 });
382 } else {
383 var last: ?i64 = null;
384 var history = std.mem.reverseIterator(self.records.items);
385 while (history.next()) |r| {
386 if (std.mem.eql(u8, r.plate, p)) {
387 last = r.exit;
388 break;
389 }
390 }
391 if (last) |t| {
392 try self.out.print("WHERE {s} gone {s}\n", .{ p, try self.time(t) });
393 } else return self.fail("unknown plate {s}", .{p});
394 }
395 } else if (std.mem.eql(u8, op, "LIST")) {
396 const l: usize = if (args.len == 0) 0 else @intCast(integer(p));
397 if (l > self.level_count) return self.fail("unknown level {d}", .{l});
398 const vs = try self.ordered();
399 var count: usize = 0;
400 for (vs) |v| {
401 if (l == 0 or self.spots[v.start].level == l) count += 1;
402 }
403 try self.out.print("LIST {d}\n", .{count});
404 for (vs) |v| {
405 if (l != 0 and self.spots[v.start].level != l) continue;
406 try self.out.print("{s} {s} {s} {s}", .{ try self.location(v), v.plate, types[v.kind], try self.time(v.entry) });
407 try self.annotation(v);
408 try self.out.writeByte('\n');
409 }
410 } else if (std.mem.eql(u8, op, "MAP")) {
411 try self.out.print("MAP {d}\n", .{self.level_count});
412 for (self.levels[0..self.level_count], 0..) |l, level| {
413 try self.out.print("L{d} ", .{level + 1});
414 for (l.first..l.end) |i| {
415 if (i > l.first and (i - l.first) % 10 == 0) try self.out.writeByte(' ');
416 const s = self.spots[i];
417 try self.out.writeByte(if (s.closed) '#' else if (s.occupant != null) "MCLE"[s.kind] else "mcle"[s.kind]);
418 }
419 try self.out.writeByte('\n');
420 }
421 } else if (std.mem.eql(u8, op, "OCCUPANCY")) {
422 var all_used: [4]i64 = @splat(0);
423 var all_available: [4]i64 = @splat(0);
424 var all_closed: i64 = 0;
425 try self.out.print("OCCUPANCY vehicles {d} waiting {d}\n", .{ self.vehicles.count(), self.queue.items.len });
426 for (self.levels[0..self.level_count], 0..) |l, level| {
427 var used: [4]i64 = @splat(0);
428 var available: [4]i64 = @splat(0);
429 var closed: i64 = 0;
430 for (self.spots[l.first..l.end]) |s| {
431 if (s.closed) closed += 1 else available[s.kind] += 1;
432 if (s.occupant != null) used[s.kind] += 1;
433 }
434 try self.out.print("L{d}", .{level + 1});
435 for (0..4) |k| {
436 try self.out.print(" {s} {d}/{d}", .{ kinds[k], used[k], available[k] });
437 all_used[k] += used[k];
438 all_available[k] += available[k];
439 }
440 try self.out.print(" closed {d}\n", .{closed});
441 all_closed += closed;
442 }
443 var us: i64 = 0;
444 var av: i64 = 0;
445 try self.out.writeAll("TOTAL");
446 for (0..4) |k| {
447 us += all_used[k];
448 av += all_available[k];
449 try self.out.print(" {s} {d}/{d}", .{ kinds[k], all_used[k], all_available[k] });
450 }
451 try self.out.print(" closed {d} rate {s}%\n", .{ all_closed, try self.decimal(if (av == 0) 0 else @divTrunc(us * 1000, av)) });
452 } else if (std.mem.eql(u8, op, "SIGN")) {
453 var counts: [5]usize = @splat(0);
454 for (0..4) |k| {
455 for (0..self.spot_count) |i| {
456 if (self.free(i) and std.mem.findScalar(usize, preferences[k], self.spots[i].kind) != null) counts[k] += 1;
457 }
458 }
459 for (self.levels[0..self.level_count]) |l| {
460 var run: usize = 0;
461 for (l.first..l.end) |i| {
462 if (self.free(i) and self.spots[i].kind == 2)
463 run += 1
464 else {
465 counts[4] += run / 3;
466 run = 0;
467 }
468 }
469 counts[4] += run / 3;
470 }
471 try self.out.writeAll("SIGN");
472 for (0..5) |k| {
473 try self.out.print(" {s} ", .{types[k]});
474 if (counts[k] == 0) try self.out.writeAll("FULL") else try self.out.print("{d}", .{counts[k]});
475 }
476 try self.out.writeByte('\n');
477 } else if (std.mem.eql(u8, op, "RATES")) {
478 try self.out.writeAll("RATES\n");
479 for (0..5) |k| try self.out.print("{s} {d} {d}\n", .{ types[k], self.rates[k], self.caps[k] });
480 try self.out.print("NIGHT {d} LOST {d} KWH {d}\n", .{ self.amounts[0], self.amounts[1], self.amounts[2] });
481 } else if (std.mem.eql(u8, op, "PASSES")) {
482 var ps: std.ArrayList([]const u8) = .empty;
483 var it = self.passes.iterator();
484 while (it.next()) |entry| {
485 if (self.pass(entry.key_ptr.*)) try ps.append(self.gpa, entry.key_ptr.*);
486 }
487 std.mem.sort([]const u8, ps.items, {}, plateLess);
488 try self.out.print("PASSES {d}\n", .{ps.items.len});
489 for (ps.items) |plate| try self.out.print("{s} until {d}{s}\n", .{ plate, self.passes.get(plate).?, if (self.vehicles.get(plate) != null) " IN" else "" });
490 } else if (std.mem.eql(u8, op, "OVERSTAY")) {
491 const hours = integer(p);
492 var vs: std.ArrayList(*Vehicle) = .empty;
493 for (self.vehicles.values()) |v| {
494 if (self.now - v.entry >= hours * 60) try vs.append(self.gpa, v);
495 }
496 std.mem.sort(*Vehicle, vs.items, {}, entryLess);
497 try self.out.print("OVERSTAY {d} {d}\n", .{ hours, vs.items.len });
498 for (vs.items) |v| try self.out.print("{s} {s} {s} {s} stay {s}\n", .{ try self.location(v), v.plate, types[v.kind], try self.time(v.entry), try self.duration(self.now - v.entry) });
499 } else if (std.mem.eql(u8, op, "HISTORY")) {
500 var count: usize = 0;
501 var paid: i64 = 0;
502 for (self.records.items) |r| {
503 if (std.mem.eql(u8, r.plate, p)) {
504 count += 1;
505 paid += r.fees[4];
506 }
507 }
508 if (count == 0) return self.fail("no history for {s}", .{p});
509 try self.out.print("HISTORY {s} visits {d} paid {d}\n", .{ p, count, paid });
510 for (self.records.items) |r| {
511 if (std.mem.eql(u8, r.plate, p)) try self.out.print("{s} {s} {s} {s} {d}{s}\n", .{ try self.time(r.entry), try self.time(r.exit), r.location, types[r.kind], r.fees[4], if (r.lost) " LOST" else "" });
512 }
513 } else if (std.mem.eql(u8, op, "REVENUE")) {
514 var totals: [6][6]i64 = @splat(@splat(0));
515 for (self.records.items) |r| {
516 for ([_]usize{ r.kind, 5 }) |k| {
517 totals[k][0] += 1;
518 for (0..5) |j| totals[k][j + 1] += r.fees[j];
519 }
520 }
521 try self.out.writeAll("REVENUE\n");
522 for (totals, 0..) |f, k| try self.out.print("{s} exits {d} parking {d} night {d} lost {d} charge {d} total {d}\n", .{ if (k == 5) "TOTAL" else types[k], f[0], f[1], f[2], f[3], f[4], f[5] });
523 } else if (std.mem.eql(u8, op, "DAILY")) {
524 const d: usize = @intCast(integer(p));
525 const s = self.daily[d];
526 for (0..6) |row| {
527 const k: ?usize = if (row == 0) null else row - 1;
528 if (row == 0) try self.out.print("DAILY {d}", .{d}) else try self.out.writeAll(types[k.?]);
529 try self.out.print(" entries {d} exits {d} refused {d} revenue {d}\n", .{ statTotal(s.entries, k), statTotal(s.exits, k), statTotal(s.refused, k), statTotal(s.revenue, k) });
530 }
531 } else if (std.mem.eql(u8, op, "STATS")) {
532 try self.out.print("STATS parked {d} peak {d} at {s}\n", .{ self.vehicles.count(), self.peak, if (self.peak_time) |t| try self.time(t) else "-" });
533 for (0..6) |row| {
534 const k: ?usize = if (row == 5) null else row;
535 const ex = statTotal(self.stats.exits, k);
536 try self.out.print("{s} entries {d} exits {d} refused {d} avg_stay {s}\n", .{ if (row == 5) "ALL" else types[row], statTotal(self.stats.entries, k), ex, statTotal(self.stats.refused, k), try self.decimal(if (ex == 0) 0 else @divTrunc(statTotal(self.stats.stays, k) * 10, ex)) });
537 }
538 }
539 }
540};
541fn locationLess(_: void, a: *Vehicle, b: *Vehicle) bool {
542 return a.start < b.start;
543}
544fn entryLess(_: void, a: *Vehicle, b: *Vehicle) bool {
545 return a.entry < b.entry or (a.entry == b.entry and locationLess({}, a, b));
546}
547fn plateLess(_: void, a: []const u8, b: []const u8) bool {
548 return std.mem.order(u8, a, b) == .lt;
549}
550fn statTotal(values: [5]i64, k: ?usize) i64 {
551 if (k) |i| return values[i];
552 var sum: i64 = 0;
553 for (values) |v| sum += v;
554 return sum;
555}
556pub fn main(init: std.process.Init) !void {
557 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
558 defer arena.deinit();
559 const gpa = arena.allocator();
560 var ib: [4096]u8 = undefined;
561 var input = std.Io.File.stdin().readerStreaming(init.io, &ib);
562 var ob: [8192]u8 = undefined;
563 var output = std.Io.File.stdout().writerStreaming(init.io, &ob);
564 const garage = try gpa.create(Garage);
565 garage.* = .{ .gpa = gpa, .out = &output.interface };
566 const n: usize = @intCast(integer((try input.interface.takeDelimiter('\n')).?));
567 for (0..n) |_| {
568 const line = try gpa.dupe(u8, (try input.interface.takeDelimiter('\n')).?);
569 var tokens: std.ArrayList([]const u8) = .empty;
570 var it = std.mem.splitScalar(u8, line, ' ');
571 while (it.next()) |part| try tokens.append(gpa, part);
572 const a = tokens.items;
573 garage.now = (integer(a[0]) - 1) * 1440 + integer(a[1][0..2]) * 60 + integer(a[1][3..5]);
574 try garage.giveup();
575 var draft: std.Io.Writer.Allocating = .init(gpa);
576 garage.out = &draft.writer;
577 garage.execute(a) catch |err| {
578 garage.out = &output.interface;
579 if (err == error.BadCommand) {
580 try output.interface.print("ERROR {s}\n", .{garage.error_message});
581 continue;
582 }
583 return err;
584 };
585 garage.out = &output.interface;
586 try output.interface.writeAll(draft.writer.buffered());
587 }
588 try output.interface.flush();
589}
