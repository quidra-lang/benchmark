4const TYPES = ["MOTORCYCLE", "CAR", "ECAR", "VAN", "BUS"];
5const KINDS = ["MOTO", "COMPACT", "LARGE", "EV"];
6const PREF = {
7 MOTORCYCLE: ["MOTO", "COMPACT", "LARGE"],
8 CAR: ["COMPACT", "LARGE"],
9 ECAR: ["EV", "COMPACT", "LARGE"],
10 VAN: ["LARGE"]
11};
12const FIELDS = ["entries", "exits", "refused", "stay", "parking", "night", "lost", "charge", "total"];
13const dateOf = (t) => Math.floor(t / 1440) + 1;
14const duration = (m) => Math.floor(m / 60) + ":" + String(m % 60).padStart(2, "0");
15const stamp = (t) => dateOf(t) + "-" + String(Math.floor(t % 1440 / 60)).padStart(2, "0") + ":" + String(t % 60).padStart(2, "0");
16const emptyTotals = () => Object.fromEntries(FIELDS.map(f => [f, 0]));
17const emptyTypes = () => Object.fromEntries(TYPES.map(k => [k, emptyTotals()]));
18const sumTotals = (rows) => Object.fromEntries(FIELDS.map(f => [f, rows.reduce((a, r) => a + r[f], 0)]));
19const free = (s) => !s.closed && s.vehicle === null;
20const location = (v) => v.spots[0].code + (v.kind === "BUS" ? ".." + v.spots[2].number : "");
21const order = (a, b) => a.spots[0].level - b.spots[0].level || a.spots[0].number - b.spots[0].number;
22class CommandError extends Error {}

24class Garage {
25 now = 0;
26 levels = [];
27 spots = [];
28 spotCodes = new Map();
29 vehicles = new Map();
30 queue = [];
31 passes = new Map();
32 history = new Map();
33 rates = {MOTORCYCLE: [100, 800], CAR: [200, 1600], ECAR: [200, 1600], VAN: [300, 2400], BUS: [600, 4800]};
34 fees = {NIGHT: 300, LOST: 2000, KWH: 30};
35 totals = emptyTypes();
36 daily = new Map();
37 peak = 0;
38 peakTime = null;
39 output = [];
40
41 emit(...parts) {
42 this.output.push(parts.join(" "));

45 vehicle(plate) {
46 const v = this.vehicles.get(plate);
47 if (!v) throw new CommandError(plate + " is not parked");
48 return v;
49 }
50
51 spot(code) {
52 const s = this.spotCodes.get(code);
53 if (!s) throw new CommandError("unknown spot " + code);
54 return s;
55 }
56
57 unpaid(v) {
58 if (v.payment !== null) throw new CommandError(v.plate + " has already paid");
59 }
60
61 hasPass(plate, time = this.now) {
62 return this.passes.has(plate) && dateOf(time) <= this.passes.get(plate);
63 }
64
65 dayTotals(day) {
66 if (!this.daily.has(day)) this.daily.set(day, emptyTypes());
67 return this.daily.get(day);
68 }
69
70 count(kind, field, day) {
71 this.totals[kind][field]++;
72 this.dayTotals(day)[kind][field]++;
73 }
74
75 assignment(kind) {
76 if (kind === "BUS") {
77 for (const level of this.levels) {
78 for (let i = 0; i + 2 < level.length; i++) {
79 const triple = level.slice(i, i + 3);
80 if (triple.every(s => s.kind === "LARGE" && free(s))) return triple;
81 }
82 }
83 } else {
84 for (const preferred of PREF[kind]) {
85 const s = this.spots.find(s => s.kind === preferred && free(s));
86 if (s) return [s];
87 }
88 }
89 return null;
90 }
91
92 park(plate, kind, spots) {
93 const v = {plate, kind, spots, entered: this.now, energy: 0, chargeCost: 0, validation: 0, payment: null};
94 this.vehicles.set(plate, v);
95 for (const s of spots) s.vehicle = v;
96 this.count(kind, "entries", dateOf(this.now));
97 if (this.vehicles.size > this.peak) {
98 this.peak = this.vehicles.size;
99 this.peakTime = this.now;
100 }
101 return v;
102 }
103
104 giveUp() {
105 const remaining = [];
106 for (const w of this.queue) {
107 const deadline = w.joined + 30;
108 if (deadline <= this.now) {
109 this.emit("GIVEUP", w.plate, w.kind, stamp(deadline));
110 this.count(w.kind, "refused", dateOf(deadline));
111 } else {
112 remaining.push(w);
113 }
114 }
115 this.queue = remaining;
116 }
117
118 processQueue() {
119 const remaining = [];
120 for (const w of this.queue) {
121 const spots = this.assignment(w.kind);
122 if (spots === null) {
123 remaining.push(w);
124 } else {
125 const v = this.park(w.plate, w.kind, spots);
126 this.emit("ADMIT", w.plate, w.kind, location(v), "waited", duration(this.now - w.joined));
127 }
128 }
129 this.queue = remaining;
130 }
131
132 interval(v, start, end, grace) {
133 const stay = end - start;
134 if (this.hasPass(v.plate, end) || stay <= grace) return [0, 0];
135 const [rate, cap] = this.rates[v.kind];
136 let parking = 0;
137 for (let offset = 0; offset < stay; offset += 1440) {
138 const billable = Math.min(stay - offset, 1440) - (offset === 0 ? grace : 0);
139 parking += Math.min(Math.floor((billable + 29) / 30) * rate, cap);
140 }
141 return [parking, (dateOf(end) - dateOf(start)) * this.fees.NIGHT];
142 }
143
144 charges(v, lost = false) {
145 let parking;
146 let night;
147 let due;
148 const lostFee = lost ? this.fees.LOST : 0;
149 if (v.payment === null) {
150 [parking, night] = this.interval(v, v.entered, this.now, 15 + v.validation);
151 due = parking + night + lostFee + v.chargeCost;
152 } else {
153 const [paidTime, paidParking, paidNight] = v.payment;
154 const [additional, extraNight] = this.interval(v, paidTime, this.now, 15);
155 parking = paidParking + additional;
156 night = paidNight + extraNight;
157 due = additional + extraNight;
158 }
159 return {parking, night, lost: lostFee, charge: v.chargeCost, total: parking + night + lostFee + v.chargeCost, due};
160 }
161
162 feeLine(label, v, c) {
163 this.emit(label, v.plate, location(v), "stay", duration(this.now - v.entered),
164 "parking", c.parking, "night", c.night, "lost", c.lost, "charge", c.charge, "total", c.total, "due", c.due);
165 }
166
167 annotation(v) {
168 const parts = [];
169 if (this.hasPass(v.plate)) parts.push("PASS");
170 if (v.validation) parts.push("VAL", v.validation);
171 if (v.energy) parts.push("KWH", v.energy);
172 if (v.payment !== null) parts.push("PAID");
173 return parts.length ? " " + parts.join(" ") : "";
174 }
175
176 moveVehicle(v, s) {
177 const old = location(v);
178 v.spots[0].vehicle = null;
179 v.spots = [s];
180 s.vehicle = v;
181 this.emit("MOVE", v.plate, old, "->", s.code);
182 }
183
184 revenueLine(label, t) {
185 const parts = [label, "exits", t.exits];
186 for (const f of ["parking", "night", "lost", "charge", "total"]) parts.push(f, t[f]);
187 this.emit(...parts);
188 }
189
190 dailyLine(prefix, t) {
191 this.emit(...prefix, "entries", t.entries, "exits", t.exits, "refused", t.refused, "revenue", t.total);
192 }
193
194 statsLine(label, t) {
195 const q = t.exits ? Math.floor(t.stay * 10 / t.exits) : 0;
196 this.emit(label, "entries", t.entries, "exits", t.exits, "refused", t.refused, "avg_stay", Math.floor(q / 10) + "." + q % 10);
197 }

199 command(cmd, a) {
200 const plate = a[0];
201 switch (cmd) {
202 case "LEVEL": {
203 const counts = a.map(Number);
204 const level = [];
205 const index = this.levels.length + 1;
206 for (let k = 0; k < KINDS.length; k++) {
207 for (let j = 0; j < counts[k]; j++) {
208 const number = level.length + 1;
209 const s = {level: index, number, kind: KINDS[k], code: "L" + index + "-" + number, closed: false, vehicle: null};
210 level.push(s);
211 this.spots.push(s);
212 this.spotCodes.set(s.code, s);
213 }
214 }
215 this.levels.push(level);
216 this.emit("LEVEL", index, "spots", level.length, "MOTO", counts[0], "COMPACT", counts[1], "LARGE", counts[2], "EV", counts[3]);
217 this.processQueue();
218 break;
219 }
220 case "CLOSE": {
221 const s = this.spot(a[0]);
222 if (s.closed) throw new CommandError("spot " + s.code + " is already closed");
223 if (s.vehicle) throw new CommandError("spot " + s.code + " is occupied by " + s.vehicle.plate);
224 s.closed = true;
225 this.emit("CLOSE", s.code, s.kind);
226 break;
227 }
228 case "OPEN": {
229 const s = this.spot(a[0]);
230 if (!s.closed) throw new CommandError("spot " + s.code + " is not closed");
231 s.closed = false;
232 this.emit("OPEN", s.code, s.kind);
233 this.processQueue();
234 break;
235 }
236 case "RATE": {
237 if (!TYPES.includes(a[0])) throw new CommandError("unknown type " + a[0]);
238 const rate = Number(a[1]);
239 const cap = Number(a[2]);
240 if (cap < rate) throw new CommandError("cap below rate for " + a[0]);
241 this.rates[a[0]] = [rate, cap];
242 this.emit("RATE", a[0], rate, cap);
243 break;
244 }
245 case "FEE": {
246 if (!(a[0] in this.fees)) throw new CommandError("unknown fee " + a[0]);
247 this.fees[a[0]] = Number(a[1]);
248 this.emit("FEE", a[0], Number(a[1]));
249 break;
250 }
251 case "ENTER": {
252 const kind = a[1];
253 if (!TYPES.includes(kind)) throw new CommandError("unknown type " + kind);
254 if (this.vehicles.has(plate)) throw new CommandError(plate + " is already parked at " + location(this.vehicles.get(plate)));
255 if (this.queue.some(w => w.plate === plate)) throw new CommandError(plate + " is already waiting");
256 const spots = this.assignment(kind);
257 if (spots !== null) {
258 const v = this.park(plate, kind, spots);
259 this.emit("ENTER", plate, kind, location(v));
260 } else if (this.queue.length < 3) {
261 this.queue.push({plate, kind, joined: this.now});
262 this.emit("WAIT", plate, kind, this.queue.length);
263 } else {
264 this.emit("FULL", plate, kind);
265 this.count(kind, "refused", dateOf(this.now));
266 }
267 break;
268 }
269 case "LEAVE": {
270 const i = this.queue.findIndex(w => w.plate === plate);
271 if (i < 0) throw new CommandError(plate + " is not waiting");
272 const [w] = this.queue.splice(i, 1);
273 this.emit("LEAVE", plate, w.kind);
274 break;
275 }
276 case "EXIT": {
277 const v = this.vehicle(plate);
278 const lost = a.length === 2;
279 if (lost) this.unpaid(v);
280 const c = this.charges(v, lost);
281 this.feeLine("EXIT", v, c);
282 if (!this.history.has(plate)) this.history.set(plate, []);
283 this.history.get(plate).push({entered: v.entered, exited: this.now, location: location(v), kind: v.kind, total: c.total, lost});
284 this.count(v.kind, "exits", dateOf(this.now));
285 const daily = this.dayTotals(dateOf(this.now))[v.kind];
286 for (const f of ["parking", "night", "lost", "charge", "total"]) {
287 this.totals[v.kind][f] += c[f];
288 daily[f] += c[f];
289 }
290 this.totals[v.kind].stay += this.now - v.entered;
291 daily.stay += this.now - v.entered;
292 for (const s of v.spots) s.vehicle = null;
293 this.vehicles.delete(plate);
294 this.processQueue();
295 break;
296 }
297 case "PAY": {
298 const v = this.vehicle(plate);
299 this.unpaid(v);
300 const c = this.charges(v);
301 v.payment = [this.now, c.parking, c.night];
302 this.emit("PAY", plate, "parking", c.parking, "night", c.night, "charge", c.charge, "total", c.total);
303 break;
304 }
305 case "QUOTE": {
306 const v = this.vehicle(plate);
307 this.feeLine("QUOTE", v, this.charges(v));
308 break;
309 }
310 case "CHARGE": {
311 const v = this.vehicle(plate);
312 const kwh = Number(a[1]);
313 if (v.kind !== "ECAR") throw new CommandError(plate + " is not an electric car");
314 if (v.spots[0].kind !== "EV") throw new CommandError(plate + " is not at a charger");
315 this.unpaid(v);
316 if (v.energy + kwh > 100) throw new CommandError("charge limit exceeded for " + plate);
317 const cost = kwh * this.fees.KWH;
318 v.energy += kwh;
319 v.chargeCost += cost;
320 this.emit("CHARGE", plate, "kwh", kwh, "cost", cost, "stay_kwh", v.energy, "stay_cost", v.chargeCost);
321 break;
322 }
323 case "VALIDATE": {
324 const v = this.vehicle(plate);
325 this.unpaid(v);
326 const minutes = Number(a[1]);
327 if (v.validation + minutes > 180) throw new CommandError("validation limit exceeded for " + plate);
328 v.validation += minutes;
329 this.emit("VALIDATE", plate, minutes, "total", v.validation);
330 break;
331 }
332 case "PASS": {
333 const last = this.hasPass(plate) ? this.passes.get(plate) : dateOf(this.now) - 1;
334 this.passes.set(plate, last + Number(a[1]));
335 this.emit("PASS", plate, "until", this.passes.get(plate));
336 break;
337 }
338 case "UNPASS": {
339 if (!this.hasPass(plate)) throw new CommandError(plate + " has no pass");
340 this.passes.delete(plate);
341 this.emit("UNPASS", plate);
342 break;
343 }
344 case "MOVE": {
345 const v = this.vehicle(plate);
346 if (v.kind === "BUS") throw new CommandError("bus " + plate + " cannot be moved");
347 const s = this.spot(a[1]);
348 if (s === v.spots[0]) throw new CommandError(plate + " is already at " + s.code);
349 if (s.closed) throw new CommandError("spot " + s.code + " is closed");
350 if (s.vehicle) throw new CommandError("spot " + s.code + " is occupied by " + s.vehicle.plate);
351 if (!PREF[v.kind].includes(s.kind)) throw new CommandError(v.kind + " cannot use " + s.kind + " spot " + s.code);
352 this.moveVehicle(v, s);
353 this.processQueue();
354 break;
355 }
356 case "REBALANCE": {
357 const targets = [...this.vehicles.values()].filter(v => v.kind !== "BUS").sort(order);
358 let moved = 0;
359 for (const v of targets) {
360 const rank = PREF[v.kind].indexOf(v.spots[0].kind);
361 if (rank === 0) continue;
362 const spots = this.assignment(v.kind);
363 if (spots && PREF[v.kind].indexOf(spots[0].kind) < rank) {
364 this.moveVehicle(v, spots[0]);
365 moved++;
366 }
367 }
368 this.emit("REBALANCE", moved);
369 this.processQueue();
370 break;
371 }
372 case "WHERE": {
373 const v = this.vehicles.get(plate);
374 const position = this.queue.findIndex(w => w.plate === plate);
375 if (v) {
376 this.emit("WHERE", plate, location(v), v.kind, stamp(v.entered), "stay", duration(this.now - v.entered) + this.annotation(v));
377 } else if (position >= 0) {
378 this.emit("WHERE", plate, "waiting", position + 1);
379 } else {
380 const records = this.history.get(plate);
381 if (!records) throw new CommandError("unknown plate " + plate);
382 this.emit("WHERE", plate, "gone", stamp(records[records.length - 1].exited));
383 }
384 break;
385 }
386 case "LIST":
387 case "OVERSTAY": {
388 const level = a.length ? Number(a[0]) : null;
389 if (cmd === "LIST" && level !== null && level > this.levels.length) throw new CommandError("unknown level " + level);
390 const targets = [...this.vehicles.values()].filter(v => cmd === "LIST" ? level === null || v.spots[0].level === level : this.now - v.entered >= level * 60);
391 targets.sort(cmd === "LIST" ? order : (a, b) => a.entered - b.entered || order(a, b));
392 if (cmd === "LIST") this.emit("LIST", targets.length);
393 else this.emit("OVERSTAY", level, targets.length);
394 for (const v of targets) {
395 if (cmd === "LIST") this.emit(location(v), v.plate, v.kind, stamp(v.entered) + this.annotation(v));
396 else this.emit(location(v), v.plate, v.kind, stamp(v.entered), "stay", duration(this.now - v.entered));
397 }
398 break;
399 }
400 case "MAP": {
401 this.emit("MAP", this.levels.length);
402 for (let i = 0; i < this.levels.length; i++) {
403 const symbols = this.levels[i].map(s => s.closed ? "#" : s.vehicle ? s.kind[0] : s.kind[0].toLowerCase()).join("");
404 this.emit("L" + (i + 1), ...symbols.match(/.{1,10}/g));
405 }
406 break;
407 }
408 case "OCCUPANCY": {
409 this.emit("OCCUPANCY", "vehicles", this.vehicles.size, "waiting", this.queue.length);
410 const allUsed = Object.fromEntries(KINDS.map(k => [k, 0]));
411 const allAvailable = {...allUsed};
412 let allClosed = 0;
413 for (let i = 0; i < this.levels.length; i++) {
414 const used = Object.fromEntries(KINDS.map(k => [k, 0]));
415 const available = {...used};
416 let closed = 0;
417 for (const s of this.levels[i]) {
418 if (s.closed) closed++;
419 else available[s.kind]++;
420 if (s.vehicle) used[s.kind]++;
421 }
422 const parts = ["L" + (i + 1)];
423 for (const k of KINDS) {
424 parts.push(k, used[k] + "/" + available[k]);
425 allUsed[k] += used[k];
426 allAvailable[k] += available[k];
427 }
428 this.emit(...parts, "closed", closed);
429 allClosed += closed;
430 }
431 const parts = ["TOTAL"];
432 for (const k of KINDS) parts.push(k, allUsed[k] + "/" + allAvailable[k]);
433 const available = Object.values(allAvailable).reduce((a, b) => a + b, 0);
434 const used = Object.values(allUsed).reduce((a, b) => a + b, 0);
435 const q = available ? Math.floor(used * 1000 / available) : 0;
436 this.emit(...parts, "closed", allClosed, "rate", Math.floor(q / 10) + "." + q % 10 + "%");
437 break;
438 }
439 case "SIGN": {
440 const parts = ["SIGN"];
441 for (const kind of TYPES) {
442 let count = 0;
443 if (kind === "BUS") {
444 for (const level of this.levels) {
445 let run = 0;
446 for (const s of level) {
447 if (s.kind === "LARGE" && free(s)) run++;
448 else {
449 count += Math.floor(run / 3);
450 run = 0;
451 }
452 }
453 count += Math.floor(run / 3);
454 }
455 } else {
456 count = this.spots.filter(s => free(s) && PREF[kind].includes(s.kind)).length;
457 }
458 parts.push(kind, count || "FULL");
459 }
460 this.emit(...parts);
461 break;
462 }
463 case "RATES": {
464 this.emit("RATES");
465 for (const kind of TYPES) this.emit(kind, ...this.rates[kind]);
466 this.emit("NIGHT", this.fees.NIGHT, "LOST", this.fees.LOST, "KWH", this.fees.KWH);
467 break;
468 }
469 case "PASSES": {
470 const plates = [...this.passes.keys()].filter(p => this.hasPass(p)).sort();
471 this.emit("PASSES", plates.length);
472 for (const p of plates) this.emit(p, "until", this.passes.get(p), ...(this.vehicles.has(p) ? ["IN"] : []));
473 break;
474 }
475 case "HISTORY": {
476 const records = this.history.get(plate);
477 if (!records) throw new CommandError("no history for " + plate);
478 this.emit("HISTORY", plate, "visits", records.length, "paid", records.reduce((a, r) => a + r.total, 0));
479 for (const r of records) this.emit(stamp(r.entered), stamp(r.exited), r.location, r.kind, r.total, ...(r.lost ? ["LOST"] : []));
480 break;
481 }
482 case "REVENUE": {
483 this.emit("REVENUE");
484 for (const kind of TYPES) this.revenueLine(kind, this.totals[kind]);
485 this.revenueLine("TOTAL", sumTotals(Object.values(this.totals)));
486 break;
487 }
488 case "DAILY": {
489 const day = Number(a[0]);
490 const totals = this.daily.get(day) ?? emptyTypes();
491 this.dailyLine(["DAILY", day], sumTotals(Object.values(totals)));
492 for (const kind of TYPES) this.dailyLine([kind], totals[kind]);
493 break;
494 }
495 case "STATS": {
496 this.emit("STATS", "parked", this.vehicles.size, "peak", this.peak, "at", this.peakTime === null ? "-" : stamp(this.peakTime));
497 for (const kind of TYPES) this.statsLine(kind, this.totals[kind]);
498 this.statsLine("ALL", sumTotals(Object.values(this.totals)));
499 break;
500 }
501 }
502 }
503
504 execute(line) {
505 const [day, clock, cmd, ...args] = line.split(" ");
506 const [hour, minute] = clock.split(":").map(Number);
507 this.now = (Number(day) - 1) * 1440 + hour * 60 + minute;
508 this.giveUp();
509 try {
510 this.command(cmd, args);
511 } catch (error) {
512 if (!(error instanceof CommandError)) throw error;
513 this.emit("ERROR", error.message);
514 }
515 }
516}
