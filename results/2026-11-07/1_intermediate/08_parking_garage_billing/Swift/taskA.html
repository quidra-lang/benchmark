3let types = ["MOTORCYCLE","CAR","ECAR","VAN","BUS"]
4let kinds = ["MOTO","COMPACT","LARGE","EV"]
5let prefs = [[0,1,2],[1,2],[3,1,2],[2],[2]]
6enum Fault: Error { case message(String) }
7func error(_ condition: Bool, _ message: String) throws {
8 if condition { throw Fault.message(message) }
9}
10func time(_ n: Int) -> String {
11 String(format: "%ld-%02ld:%02ld",locale: Locale(identifier: "en_US_POSIX"),n/1440+1,n/60%24,n%60)
12}
13func duration(_ n: Int) -> String {
14 String(format: "%ld:%02ld",locale: Locale(identifier: "en_US_POSIX"),n/60,n%60)
15}
16func tenths(_ n: Int) -> String { "\(n/10).\(n%10)" }
17final class Spot {
18 let level: Int
19 let number: Int
20 let kind: Int
21 let order: Int
22 var closed = false
23 weak var occupant: Vehicle?
24 init(_ level: Int, _ number: Int, _ kind: Int, _ order: Int) {
25 self.level = level
26 self.number = number
27 self.kind = kind
28 self.order = order
29 }
30 func id() -> String { "L\(level)-\(number)" }
31 func free() -> Bool { !closed && occupant == nil }
32}
33final class Vehicle {
34 let plate: String
35 let kind: Int
36 let entry: Int
37 var spots: [Spot]
38 var energy = 0
39 var charge = 0
40 var validation = 0
41 var paidAt = -1
42 var paidParking = 0
43 var paidNight = 0
44 init(_ plate: String, _ kind: Int, _ entry: Int, _ spots: [Spot]) {
45 self.plate = plate
46 self.kind = kind
47 self.entry = entry
48 self.spots = spots
49 }
50 func location() -> String { spots[0].id()+(kind == 4 ? "..\(spots[2].number)" : "") }
51}

53 let plate: String
54 let kind: Int
55 let joined: Int
56}

65final class Ledger { var rows = Array(repeating:Array(repeating:0,count:4),count:5) }
66final class Garage {
67 var now = 0
68 var peak = 0
69 var peakAt = 0
70 var levels: [[Spot]] = []
71 var spots: [Spot] = []
72 var parked: [String:Vehicle] = [:]
73 var queue: [Waiting] = []
74 var passes: [String:Int] = [:]
75 var history: [String:[Visit]] = [:]
76 var rates = [[100,800],[200,1600],[200,1600],[300,2400],[600,4800]]
77 var night = 300
78 var lost = 2000
79 var kwh = 30
80 var stats = Array(repeating:Array(repeating:0,count:9),count:5)
81 var days: [Int:Ledger] = [:]
82 var output: [String] = []
83 func daily(_ date: Int) -> Ledger {
84 if let existing = days[date] { return existing }
85 let result = Ledger()
86 days[date] = result
87 return result
88 }
89 func hasPass(_ plate: String) -> Bool { (passes[plate] ?? 0) >= now/1440+1 }
90 func vehicle(_ plate: String) throws -> Vehicle {
91 guard let v = parked[plate] else { throw Fault.message(plate+" is not parked") }
92 return v
93 }
94 func spot(_ id: String) throws -> Spot {
95 let p = id.dropFirst().split(separator:"-")
96 let l = Int(p[0])!
97 let n = Int(p[1])!
98 try error(l > levels.count || n > levels[l-1].count,"unknown spot "+id)
99 return levels[l-1][n-1]
100 }
101 func assign(_ kind: Int) -> [Spot]? {
102 if kind == 4 {
103 for level in levels where level.count >= 3 {
104 for i in 0..<(level.count-2) {
105 let triple = Array(level[i..<i+3])
106 if triple.allSatisfy({ $0.kind == 2 && $0.free() }) { return triple }
107 }
108 }
109 } else {
110 for wanted in prefs[kind] {
111 if let first = spots.first(where: { $0.kind == wanted && $0.free() }) { return [first] }
112 }
113 }
114 return nil
115 }
116 func occupy(_ v: Vehicle) { for s in v.spots { s.occupant = v } }
117 func free(_ v: Vehicle) { for s in v.spots { s.occupant = nil } }
118 func park(_ plate: String, _ kind: Int, _ assigned: [Spot]) -> Vehicle {
119 let v = Vehicle(plate,kind,now,assigned)
120 parked[plate] = v
121 occupy(v)
122 stats[kind][0] += 1
123 daily(now/1440+1).rows[kind][0] += 1
124 if parked.count > peak {
125 peak = parked.count
126 peakAt = now
127 }
128 return v
129 }
130 func refuse(_ kind: Int, _ at: Int) {
131 stats[kind][2] += 1
132 daily(at/1440+1).rows[kind][2] += 1
133 }
134 func giveup() {
135 for w in queue {
136 if now-w.joined >= 30 {
137 queue.removeAll { $0.plate == w.plate }
138 refuse(w.kind,w.joined+30)
139 output.append("GIVEUP \(w.plate) \(types[w.kind]) \(time(w.joined+30))")
140 }
141 }
142 }
143 func admit() {
144 for w in queue {
145 if let assigned = assign(w.kind) {
146 queue.removeAll { $0.plate == w.plate }
147 let v = park(w.plate,w.kind,assigned)
148 output.append("ADMIT \(v.plate) \(types[v.kind]) \(v.location()) waited \(duration(now-w.joined))")
149 }
150 }
151 }
152 func interval(_ v: Vehicle, _ start: Int, _ free: Int) -> [Int] {
153 let total = now-start
154 if hasPass(v.plate) || total <= free { return [0,0] }
155 var parking = 0
156 for offset in stride(from:0,to:total,by:1440) {
157 let billable = min(total-offset,1440)-(offset == 0 ? free : 0)
158 parking += min((billable+29)/30*rates[v.kind][0],rates[v.kind][1])
159 }
160 return [parking,(now/1440-start/1440)*night]
161 }
162 func fees(_ v: Vehicle, _ loss: Bool) -> [Int] {
163 let pair = interval(v,v.paidAt < 0 ? v.entry : v.paidAt,v.paidAt < 0 ? 15+v.validation : 15)
164 let parking = pair[0]+(v.paidAt < 0 ? 0 : v.paidParking)
165 let surcharge = pair[1]+(v.paidAt < 0 ? 0 : v.paidNight)
166 let ticket = loss ? lost : 0
167 let total = parking+surcharge+ticket+v.charge
168 return [parking,surcharge,ticket,v.charge,total,v.paidAt < 0 ? total : pair[0]+pair[1]]
169 }
170 func feeLine(_ label: String, _ v: Vehicle, _ f: [Int]) -> String {
171 "\(label) \(v.plate) \(v.location()) stay \(duration(now-v.entry)) parking \(f[0])"+
172 " night \(f[1]) lost \(f[2]) charge \(f[3]) total \(f[4]) due \(f[5])"
173 }
174 func annotation(_ v: Vehicle) -> String {
175 (hasPass(v.plate) ? " PASS" : "")+(v.validation > 0 ? " VAL \(v.validation)" : "")+
176 (v.energy > 0 ? " KWH \(v.energy)" : "")+(v.paidAt >= 0 ? " PAID" : "")
177 }
178 func ordered() -> [Vehicle] { parked.values.sorted { $0.spots[0].order < $1.spots[0].order } }
179 func move(_ v: Vehicle, _ target: [Spot]) {
180 let old = v.location()
181 free(v)
182 v.spots = target
183 occupy(v)
184 output.append("MOVE \(v.plate) \(old) -> \(v.location())")
185 }
186 func occupancy(_ target: [Spot]) -> [Int] {
187 var a = Array(repeating:0,count:9)
188 for s in target {
189 if s.closed { a[8] += 1 }
190 else { a[4+s.kind] += 1 }
191 if s.occupant != nil { a[s.kind] += 1 }
192 }
193 return a
194 }
195 func occupancyLine(_ label: String, _ a: [Int]) -> String {
196 label+(0..<4).map { " \(kinds[$0]) \(a[$0])/\(a[$0+4])" }.joined()+" closed \(a[8])"
197 }
198 func revenueLine(_ label: String, _ a: [Int]) -> String {
199 "\(label) exits \(a[1]) parking \(a[4]) night \(a[5]) lost \(a[6]) charge \(a[7]) total \(a[8])"
200 }
201 func statsLine(_ label: String, _ a: [Int]) -> String {
202 "\(label) entries \(a[0]) exits \(a[1]) refused \(a[2]) avg_stay \(tenths(a[1] == 0 ? 0 : a[3]*10/a[1]))"
203 }
204 func dailyLine(_ label: String, _ a: [Int]) -> String { "\(label) entries \(a[0]) exits \(a[1]) refused \(a[2]) revenue \(a[3])" }
205 func sum(_ a: [[Int]], _ width: Int) -> [Int] { (0..<width).map { i in a.reduce(0) { $0+$1[i] } } }
206 func execute(_ t: [String]) throws {
207 var trigger = false
208 let command = t[2]
209 switch command {
210 case "LEVEL":
211 var added: [Spot] = []
212 let counts = (0..<4).map { Int(t[$0+3])! }
213 for k in 0..<4 {
214 for _ in 0..<counts[k] {
215 let s = Spot(levels.count+1,added.count+1,k,spots.count)
216 added.append(s)
217 spots.append(s)
218 }
219 }
220 levels.append(added)
221 output.append("LEVEL \(levels.count) spots \(added.count) MOTO \(counts[0]) COMPACT \(counts[1]) LARGE \(counts[2]) EV \(counts[3])")
222 trigger = true
223 case "CLOSE", "OPEN":
224 let s = try spot(t[3])
225 if command == "CLOSE" {
226 try error(s.closed,"spot \(t[3]) is already closed")
227 try error(s.occupant != nil,"spot \(t[3]) is occupied by \(s.occupant?.plate ?? "")")
228 s.closed = true
229 } else {
230 try error(!s.closed,"spot \(t[3]) is not closed")
231 s.closed = false
232 trigger = true
233 }
234 output.append("\(command) \(s.id()) \(kinds[s.kind])")
235 case "RATE":
236 guard let k = types.firstIndex(of:t[3]) else { throw Fault.message("unknown type "+t[3]) }
237 let rate = Int(t[4])!
238 let cap = Int(t[5])!
239 try error(cap < rate,"cap below rate for "+t[3])
240 rates[k] = [rate,cap]
241 output.append("RATE \(t[3]) \(rate) \(cap)")
242 case "FEE":
243 let amount = Int(t[4])!
244 try error(!["NIGHT","LOST","KWH"].contains(t[3]),"unknown fee "+t[3])
245 switch t[3] {
246 case "NIGHT": night = amount
247 case "LOST": lost = amount
248 default: kwh = amount
249 }
250 output.append("FEE \(t[3]) \(amount)")
251 case "ENTER":
252 let plate = t[3]
253 guard let k = types.firstIndex(of:t[4]) else { throw Fault.message("unknown type "+t[4]) }
254 let existing = parked[plate]
255 try error(existing != nil,"\(plate) is already parked at \(existing?.location() ?? "")")
256 try error(queue.contains { $0.plate == plate },plate+" is already waiting")
257 if let assigned = assign(k) { output.append("ENTER \(plate) \(types[k]) \(park(plate,k,assigned).location())") }
258 else if queue.count < 3 {
259 queue.append(Waiting(plate:plate,kind:k,joined:now))
260 output.append("WAIT \(plate) \(types[k]) \(queue.count)")
261 } else {
262 refuse(k,now)
263 output.append("FULL \(plate) \(types[k])")
264 }
265 case "LEAVE":
266 guard let w = queue.first(where:{ $0.plate == t[3] }) else { throw Fault.message(t[3]+" is not waiting") }
267 queue.removeAll { $0.plate == w.plate }
268 output.append("LEAVE \(w.plate) \(types[w.kind])")
269 case "EXIT", "PAY", "QUOTE":
270 let v = try vehicle(t[3])
271 let loss = t.count == 5
272 try error(v.paidAt >= 0 && (command == "PAY" || loss),v.plate+" has already paid")
273 let f = fees(v,loss)
274 if command == "PAY" {
275 v.paidAt = now
276 v.paidParking = f[0]
277 v.paidNight = f[1]
278 output.append("PAY \(v.plate) parking \(f[0]) night \(f[1]) charge \(f[3]) total \(f[4])")
279 } else {
280 output.append(feeLine(command,v,f))
281 if command == "EXIT" {
282 history[v.plate,default:[]].append(Visit(entry:v.entry,exit:now,location:v.location(),kind:v.kind,total:f[4],loss:loss))
283 stats[v.kind][1] += 1
284 stats[v.kind][3] += now-v.entry
285 for i in 0..<5 { stats[v.kind][i+4] += f[i] }
286 let d = daily(now/1440+1)
287 d.rows[v.kind][1] += 1
288 d.rows[v.kind][3] += f[4]
289 free(v)
290 parked.removeValue(forKey:v.plate)
291 trigger = true
292 }
293 }
294 case "CHARGE":
295 let v = try vehicle(t[3])
296 try error(v.kind != 2,v.plate+" is not an electric car")
297 try error(v.spots[0].kind != 3,v.plate+" is not at a charger")
298 try error(v.paidAt >= 0,v.plate+" has already paid")
299 let energy = Int(t[4])!
300 try error(v.energy+energy > 100,"charge limit exceeded for "+v.plate)
301 let cost = energy*kwh
302 v.energy += energy
303 v.charge += cost
304 output.append("CHARGE \(v.plate) kwh \(energy) cost \(cost) stay_kwh \(v.energy) stay_cost \(v.charge)")
305 case "VALIDATE":
306 let v = try vehicle(t[3])
307 try error(v.paidAt >= 0,v.plate+" has already paid")
308 let minutes = Int(t[4])!
309 try error(v.validation+minutes > 180,"validation limit exceeded for "+v.plate)
310 v.validation += minutes
311 output.append("VALIDATE \(v.plate) \(minutes) total \(v.validation)")
312 case "PASS":
313 let end = (hasPass(t[3]) ? passes[t[3]]! : now/1440)+Int(t[4])!
314 passes[t[3]] = end
315 output.append("PASS \(t[3]) until \(end)")
316 case "UNPASS":
317 try error(!hasPass(t[3]),t[3]+" has no pass")
318 passes.removeValue(forKey:t[3])
319 output.append("UNPASS "+t[3])
320 case "MOVE":
321 let v = try vehicle(t[3])
322 try error(v.kind == 4,"bus \(v.plate) cannot be moved")
323 let s = try spot(t[4])
324 try error(v.spots[0] === s,"\(v.plate) is already at \(t[4])")
325 try error(s.closed,"spot \(t[4]) is closed")
326 try error(s.occupant != nil,"spot \(t[4]) is occupied by \(s.occupant?.plate ?? "")")
327 try error(!prefs[v.kind].contains(s.kind),"\(types[v.kind]) cannot use \(kinds[s.kind]) spot \(t[4])")
328 move(v,[s])
329 trigger = true
330 case "REBALANCE":
331 var moved = 0
332 for v in ordered() where v.kind != 4 {
333 if let target = assign(v.kind), prefs[v.kind].firstIndex(of:target[0].kind)! < prefs[v.kind].firstIndex(of:v.spots[0].kind)! {
334 move(v,target)
335 moved += 1
336 }
337 }
338 output.append("REBALANCE \(moved)")
339 trigger = true
340 case "WHERE":
341 let v = parked[t[3]]
342 let p = queue.firstIndex { $0.plate == t[3] }
343 let visits = history[t[3]]
344 try error(v == nil && p == nil && visits == nil,"unknown plate "+t[3])
345 if let v { output.append("WHERE \(v.plate) \(v.location()) \(types[v.kind]) \(time(v.entry)) stay \(duration(now-v.entry))\(annotation(v))") }
346 else if let p { output.append("WHERE \(t[3]) waiting \(p+1)") }
347 else { output.append("WHERE \(t[3]) gone \(time(visits!.last!.exit))") }
348 case "LIST":
349 let l = t.count == 4 ? Int(t[3])! : 0
350 try error(l > levels.count,"unknown level \(l)")
351 let found = ordered().filter { l == 0 || $0.spots[0].level == l }
352 output.append("LIST \(found.count)")
353 for v in found { output.append("\(v.location()) \(v.plate) \(types[v.kind]) \(time(v.entry))\(annotation(v))") }
354 case "MAP":
355 output.append("MAP \(levels.count)")
356 for (i,level) in levels.enumerated() {
357 let symbols = level.map { s -> Character in
358 if s.closed { return "#" }
359 let initial = kinds[s.kind].first!
360 return s.occupant != nil ? initial : Character(String(initial).lowercased())
361 }
362 let chunks = stride(from:0,to:symbols.count,by:10).map { String(symbols[$0..<min($0+10,symbols.count)]) }
363 output.append("L\(i+1) "+chunks.joined(separator:" "))
364 }
365 case "OCCUPANCY":
366 output.append("OCCUPANCY vehicles \(parked.count) waiting \(queue.count)")
367 for (i,level) in levels.enumerated() { output.append(occupancyLine("L\(i+1)",occupancy(level))) }
368 let a = occupancy(spots)
369 let used = a.prefix(4).reduce(0,+)
370 let available = a[4..<8].reduce(0,+)
371 output.append(occupancyLine("TOTAL",a)+" rate \(tenths(available == 0 ? 0 : used*1000/available))%")
372 case "SIGN":
373 var line = "SIGN"
374 for k in 0..<5 {
375 var count = 0
376 if k == 4 {
377 for level in levels {
378 var run = 0
379 for s in level {
380 if s.kind == 2 && s.free() {
381 run += 1
382 if run == 3 {
383 count += 1
384 run = 0
385 }
386 } else { run = 0 }
387 }
388 }
389 } else {
390 count = spots.filter { prefs[k].contains($0.kind) && $0.free() }.count
391 }
392 line += " \(types[k]) "+(count == 0 ? "FULL" : String(count))
393 }
394 output.append(line)
395 case "RATES":
396 output.append("RATES")
397 for k in 0..<5 { output.append("\(types[k]) \(rates[k][0]) \(rates[k][1])") }
398 output.append("NIGHT \(night) LOST \(lost) KWH \(kwh)")
399 case "PASSES":
400 let found = passes.keys.filter { hasPass($0) }.sorted()
401 output.append("PASSES \(found.count)")
402 for plate in found { output.append("\(plate) until \(passes[plate]!)"+(parked[plate] != nil ? " IN" : "")) }
403 case "OVERSTAY":
404 let hours = Int(t[3])!
405 let found = ordered().filter { now-$0.entry >= hours*60 }.sorted {
406 $0.entry == $1.entry ? $0.spots[0].order < $1.spots[0].order : $0.entry < $1.entry
407 }
408 output.append("OVERSTAY \(hours) \(found.count)")
409 for v in found { output.append("\(v.location()) \(v.plate) \(types[v.kind]) \(time(v.entry)) stay \(duration(now-v.entry))") }
410 case "HISTORY":
411 guard let visits = history[t[3]] else { throw Fault.message("no history for "+t[3]) }
412 output.append("HISTORY \(t[3]) visits \(visits.count) paid \(visits.reduce(0) { $0+$1.total })")
413 for v in visits { output.append("\(time(v.entry)) \(time(v.exit)) \(v.location) \(types[v.kind]) \(v.total)"+(v.loss ? " LOST" : "")) }
414 case "REVENUE":
415 output.append("REVENUE")
416 for k in 0..<5 { output.append(revenueLine(types[k],stats[k])) }
417 output.append(revenueLine("TOTAL",sum(stats,9)))
418 case "DAILY":
419 let d = Int(t[3])!
420 let a = daily(d).rows
421 output.append(dailyLine("DAILY \(d)",sum(a,4)))
422 for k in 0..<5 { output.append(dailyLine(types[k],a[k])) }
423 case "STATS":
424 output.append("STATS parked \(parked.count) peak \(peak) at "+(peak == 0 ? "-" : time(peakAt)))
425 for k in 0..<5 { output.append(statsLine(types[k],stats[k])) }
426 output.append(statsLine("ALL",sum(stats,9)))
427 default: break
428 }
429 if trigger { admit() }
430 }
431 func process(_ line: String) -> [String] {
432 let t = line.split(separator:" ").map(String.init)
433 let hm = t[1].split(separator:":")
434 now = (Int(t[0])!-1)*1440+Int(hm[0])!*60+Int(hm[1])!
435 output = []
436 giveup()
437 do { try execute(t) }
438 catch Fault.message(let message) { output.append("ERROR "+message) }
439 catch { fatalError("Unexpected error") }
440 return output
441 }
442}
443let garage = Garage()
444let commandCount = Int(readLine()!)!
445for _ in 0..<commandCount {
446 for line in garage.process(readLine()!) { print(line) }
447}
