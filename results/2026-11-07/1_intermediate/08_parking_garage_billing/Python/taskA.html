59class Level:
60 def __init__(self, number, counts):
61 self.number = number
62 self.spots = []
63 for kind, count in zip(SPOT_KINDS, counts):
64 for _ in range(count):
65 self.spots.append(Spot(number, len(self.spots) + 1, kind))
66
67 def bus_assignment(self):
68 for start in range(len(self.spots) - 2):
69 triple = self.spots[start:start + 3]
70 if all(spot.kind == "LARGE" and spot.free() for spot in triple):
71 return triple
72 return None
73
74 def bus_capacity(self):
75 count = 0
76 run = 0
77 for spot in self.spots:
78 if spot.kind == "LARGE" and spot.free():
79 run += 1
80 else:
81 count += run // 3
82 run = 0
83 return count + run // 3
84
85 def occupancy(self):
86 used = {kind: 0 for kind in SPOT_KINDS}
87 available = {kind: 0 for kind in SPOT_KINDS}
88 closed = 0
89 for spot in self.spots:
90 if spot.closed:
91 closed += 1
92 else:
93 available[spot.kind] += 1
94 if spot.vehicle is not None:
95 used[spot.kind] += 1
96 return used, available, closed

188 def assignment(self, kind):
189 if kind == "BUS":
190 for level in self.levels:
191 triple = level.bus_assignment()
192 if triple is not None:
193 return triple
194 return None
195 for preferred in PREFERENCES[kind]:

201 def park(self, plate, kind, spots):
202 vehicle = Vehicle(plate, kind, spots, self.now)
203 self.vehicles[plate] = vehicle
204 for spot in spots:
205 spot.vehicle = vehicle
206 self.add_count(kind, "entries", date_of(self.now))
207 if len(self.vehicles) > self.peak:
208 self.peak = len(self.vehicles)
209 self.peak_time = self.now
210 return vehicle

212 def give_up(self):
213 remaining = []
214 for waiting in self.queue:
215 deadline = waiting.joined + 30
216 if deadline <= self.now:
217 self.emit("GIVEUP", waiting.plate, waiting.kind, time_text(deadline))
218 self.add_count(waiting.kind, "refused", date_of(deadline))
219 else:
220 remaining.append(waiting)
221 self.queue = remaining

223 def process_queue(self):
224 remaining = []
225 for waiting in self.queue:
226 spots = self.assignment(waiting.kind)
227 if spots is None:
228 remaining.append(waiting)
229 else:
230 vehicle = self.park(waiting.plate, waiting.kind, spots)
231 self.emit(
232 "ADMIT", waiting.plate, waiting.kind, vehicle.location(),
233 "waited", duration_text(self.now - waiting.joined),
234 )
235 self.queue = remaining

237 def interval_fees(self, vehicle, start, end, free_minutes):
238 stay = end - start
239 if self.has_pass(vehicle.plate, end) or stay <= free_minutes:
240 return 0, 0
241 rate, cap = self.rates[vehicle.kind]
242 parking = 0
243 for offset in range(0, stay, 1440):
244 minutes = min(stay - offset, 1440)
245 billable = minutes - free_minutes if offset == 0 else minutes
246 parking += min(((billable + 29) // 30) * rate, cap)
247 night = (date_of(end) - date_of(start)) * self.fees["NIGHT"]
248 return parking, night
249
250 def exit_charges(self, vehicle, lost=False):
251 if vehicle.payment is None:
252 parking, night = self.interval_fees(
253 vehicle, vehicle.entered, self.now, 15 + vehicle.validation
254 )
255 lost_fee = self.fees["LOST"] if lost else 0
256 total = parking + night + lost_fee + vehicle.charge_cost
257 due = total
258 else:
259 paid_time, paid_parking, paid_night = vehicle.payment
260 additional, additional_night = self.interval_fees(vehicle, paid_time, self.now, 15)
261 parking = paid_parking + additional
262 night = paid_night + additional_night
263 lost_fee = 0
264 total = parking + night + vehicle.charge_cost
265 due = additional + additional_night
266 return {
267 "parking": parking, "night": night, "lost": lost_fee,
268 "charge": vehicle.charge_cost, "total": total, "due": due,
269 }

338 def enter(self, arguments):
339 plate, kind = arguments
340 if kind not in self.rates:
341 raise CommandError(f"unknown type {kind}")
342 if plate in self.vehicles:
343 location = self.vehicles[plate].location()
344 raise CommandError(f"{plate} is already parked at {location}")
345 if any(waiting.plate == plate for waiting in self.queue):
346 raise CommandError(f"{plate} is already waiting")
347 spots = self.assignment(kind)
348 if spots is not None:
349 vehicle = self.park(plate, kind, spots)
350 self.emit("ENTER", plate, kind, vehicle.location())
351 elif len(self.queue) < 3:
352 self.queue.append(WaitingVehicle(plate, kind, self.now))
353 self.emit("WAIT", plate, kind, len(self.queue))
354 else:
355 self.emit("FULL", plate, kind)
356 self.add_count(kind, "refused", date_of(self.now))

358 def leave(self, arguments):
359 plate = arguments[0]
360 for index, waiting in enumerate(self.queue):
361 if waiting.plate == plate:
362 del self.queue[index]
363 self.emit("LEAVE", plate, waiting.kind)
364 return
365 raise CommandError(f"{plate} is not waiting")

367 def exit(self, arguments):
368 vehicle = self.require_vehicle(arguments[0])
369 lost = len(arguments) == 2
370 if lost:
371 self.require_unpaid(vehicle)
372 charges = self.exit_charges(vehicle, lost)
373 self.fee_line("EXIT", vehicle, charges)
374 record = ExitRecord(vehicle, self.now, charges["total"], lost)
375 self.history.setdefault(vehicle.plate, []).append(record)
376 self.add_count(vehicle.kind, "exits", date_of(self.now))
377 totals = self.totals[vehicle.kind]
378 daily = self.day_totals(date_of(self.now))[vehicle.kind]
379 for field in ("parking", "night", "lost", "charge", "total"):
380 totals[field] += charges[field]
381 daily[field] += charges[field]
382 stay = self.now - vehicle.entered
383 totals["stay"] += stay
384 daily["stay"] += stay
385 for spot in vehicle.spots:
386 spot.vehicle = None
387 del self.vehicles[vehicle.plate]
388 self.process_queue()

405 def charge(self, arguments):
406 vehicle = self.require_vehicle(arguments[0])
407 energy = int(arguments[1])
408 if vehicle.kind != "ECAR":
409 raise CommandError(f"{vehicle.plate} is not an electric car")
410 if vehicle.spots[0].kind != "EV":
411 raise CommandError(f"{vehicle.plate} is not at a charger")
412 self.require_unpaid(vehicle)
413 if vehicle.energy + energy > 100:
414 raise CommandError(f"charge limit exceeded for {vehicle.plate}")
415 cost = energy * self.fees["KWH"]
416 vehicle.energy += energy
417 vehicle.charge_cost += cost
418 self.emit(
419 "CHARGE", vehicle.plate, "kwh", energy, "cost", cost,
420 "stay_kwh", vehicle.energy, "stay_cost", vehicle.charge_cost,
421 )

455 def move(self, arguments):
456 plate, code = arguments
457 vehicle = self.require_vehicle(plate)
458 if vehicle.kind == "BUS":
459 raise CommandError(f"bus {plate} cannot be moved")
460 spot = self.require_spot(code)
461 if spot is vehicle.spots[0]:
462 raise CommandError(f"{plate} is already at {code}")
463 if spot.closed:
464 raise CommandError(f"spot {code} is closed")
465 if spot.vehicle is not None:
466 raise CommandError(f"spot {code} is occupied by {spot.vehicle.plate}")

468 raise CommandError(f"{vehicle.kind} cannot use {spot.kind} spot {code}")
469 self.move_vehicle(vehicle, spot)

472 def rebalance(self, arguments):
473 targets = sorted(
474 (v for v in self.vehicles.values() if v.kind != "BUS"),
475 key=lambda vehicle: vehicle.order(),
476 )
477 moved = 0
478 for vehicle in targets:
479 preferences = PREFERENCES[vehicle.kind]
480 rank = preferences.index(vehicle.spots[0].kind)
481 if rank == 0:
482 continue
483 spots = self.assignment(vehicle.kind)
484 if spots is not None and preferences.index(spots[0].kind) < rank:
485 self.move_vehicle(vehicle, spots[0])
486 moved += 1
487 self.emit("REBALANCE", moved)
488 self.process_queue()

538 def occupancy(self, arguments):
539 self.emit("OCCUPANCY", "vehicles", len(self.vehicles), "waiting", len(self.queue))
540 total_used = {kind: 0 for kind in SPOT_KINDS}
541 total_available = {kind: 0 for kind in SPOT_KINDS}
542 total_closed = 0
543 for level in self.levels:
544 used, available, closed = level.occupancy()
545 parts = ["L" + str(level.number)]
546 for kind in SPOT_KINDS:
547 parts.extend((kind, f"{used[kind]}/{available[kind]}"))
548 total_used[kind] += used[kind]
549 total_available[kind] += available[kind]
550 parts.extend(("closed", closed))
551 total_closed += closed
552 self.emit(*parts)
553 parts = ["TOTAL"]
554 for kind in SPOT_KINDS:
555 parts.extend((kind, f"{total_used[kind]}/{total_available[kind]}"))
556 available_count = sum(total_available.values())
557 usage = sum(total_used.values()) * 1000 // available_count if available_count else 0
558 parts.extend(("closed", total_closed, "rate", f"{usage // 10}.{usage % 10}%"))
559 self.emit(*parts)
