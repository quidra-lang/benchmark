void refusal(State &state, int kind, int minute)
    state.stats[kind][2] += 1
    state.daily[minute / 1440 + 1][kind][2] += 1

void give_up(State &state)
    Waiting[] remaining := []
    for item := state.waiting[]
        if state.now - item.joined >= 30
            emit("GIVEUP " + item.plate + " " + type_name(item.kind) + " " + clock_text(item.joined + 30))
            refusal(&state, item.kind, item.joined + 30)
        else
            remaining = remaining.append(item)
    state.waiting = remaining
