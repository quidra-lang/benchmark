188int[] interval_fee(const State state, int index, int start, int finish, int free_minutes)
189 item := state.vehicles[index]
190 stay := finish - start
191 if valid_pass(state, item.plate, finish) or stay <= free_minutes
192 return [0, 0]
193 parking := 0
194 offset := 0
195 while offset < stay
196 minutes := math.min(1440, stay - offset)
197 if offset == 0
198 minutes -= free_minutes
199 parking += math.min((minutes + 29) / 30 * state.rates[item.kind], state.caps[item.kind])
200 offset += 1440
201 return [parking, (finish / 1440 - start / 1440) * state.fees[0]]

203int[] exit_fees(const State state, int index, bool lost)
204 item := state.vehicles[index]
205 amounts := array(6, fill = 0)
206 if item.paid_at < 0
207 partial := interval_fee(state, index, item.entered, state.now, 15 + item.validation)
208 amounts[0] = partial[0]
209 amounts[1] = partial[1]
210 amounts[2] = if lost then state.fees[1] else 0
211 amounts[5] = amounts[0] + amounts[1] + amounts[2] + item.charging
212 else
213 extra := interval_fee(state, index, item.paid_at, state.now, 15)
214 amounts[0] = item.paid_parking + extra[0]
215 amounts[1] = item.paid_night + extra[1]
216 amounts[5] = extra[0] + extra[1]
217 amounts[3] = item.charging
218 amounts[4] = amounts[:<4].sum()
219 return amounts

260void command(State &state, string line)
261 words := line.split(" ")
262 clock := words[1].split(":")
263 state.now = (integer_value(words[0]) - 1) * 1440 + integer_value(clock[0]) * 60 + integer_value(clock[1])
264 give_up(&state)
265 action := words[2]
266 if action == "LEVEL"
267 state.levels += 1
268 total := 0
269 result := ""
270 for: 0 <= kind < 4
271 count := integer_value(words[kind + 3])
272 result += " " + spot_kind(kind) + " " + string(count)
273 for: 0 <= k < count
274 total += 1
275 state.spots = state.spots.append(Spot(state.levels, total, kind, false, -1))
276 emit("LEVEL " + string(state.levels) + " spots " + string(total) + result)
277 process_queue(&state)
278 return
279 if action == "CLOSE" or action == "OPEN"
280 target := spot_id(state, words[3])
281 if target < 0
282 fail("unknown spot " + words[3])
283 elif action == "CLOSE" and state.spots[target].closed
284 fail("spot " + words[3] + " is already closed")
285 elif action == "OPEN" and not state.spots[target].closed
286 fail("spot " + words[3] + " is not closed")
287 elif action == "CLOSE" and state.spots[target].occupant >= 0
288 fail("spot " + words[3] + " is occupied by " + state.vehicles[state.spots[target].occupant].plate)
289 else
290 state.spots[target].closed = action == "CLOSE"
291 emit(action + " " + words[3] + " " + spot_kind(state.spots[target].kind))
292 if action == "OPEN"
293 process_queue(&state)
294 return
295 if action == "RATE" or action == "FEE"
296 if action == "RATE"
297 kind := type_id(words[3])
298 rate := integer_value(words[4])
299 cap := integer_value(words[5])
300 if kind < 0
301 fail("unknown type " + words[3])
302 elif cap < rate
303 fail("cap below rate for " + words[3])
304 else
305 state.rates[kind] = rate
306 state.caps[kind] = cap
307 emit("RATE " + words[3] + " " + string(rate) + " " + string(cap))
308 else
309 item := -1
310 kinds := ["NIGHT", "LOST", "KWH"]
311 for: 0 <= k < 3
312 if words[3] == kinds[k]
313 item = k
314 if item < 0
315 fail("unknown fee " + words[3])
316 else
317 state.fees[item] = integer_value(words[4])
318 emit("FEE " + words[3] + " " + string(state.fees[item]))
319 return
320 if action == "ENTER"
321 plate := words[3]
322 kind := type_id(words[4])
323 index := parked_id(state, plate)
324 queued := waiting_id(state, plate)
325 if kind < 0
326 fail("unknown type " + words[4])
327 elif index >= 0
328 fail(plate + " is already parked at " + location(state, index))
329 elif queued >= 0
330 fail(plate + " is already waiting")
331 else
332 target := assign_spot(state, kind)
333 if target >= 0
334 admitted := admit(&state, plate, kind, target)
335 emit("ENTER " + plate + " " + type_name(kind) + " " + location(state, admitted))
336 elif #state.waiting < 3
337 state.waiting = state.waiting.append(Waiting(plate, kind, state.now))
338 emit("WAIT " + plate + " " + type_name(kind) + " " + string(#state.waiting))
339 else
340 refusal(&state, kind, state.now)
341 emit("FULL " + plate + " " + type_name(kind))
342 return
343 if action == "LEAVE"
344 queued := waiting_id(state, words[3])
345 if queued < 0
346 fail(words[3] + " is not waiting")
347 else
348 emit("LEAVE " + words[3] + " " + type_name(state.waiting[queued].kind))
349 state.waiting = state.waiting[:<queued].concat(state.waiting[queued + 1:])
350 return
351 if action == "PASS" or action == "UNPASS"
352 index := pass_id(state, words[3])
353 valid := valid_pass(state, words[3], state.now)
354 if action == "UNPASS"
355 if not valid
356 fail(words[3] + " has no pass")
357 else
358 state.passes[index].last_day = 0
359 emit("UNPASS " + words[3])
360 else
361 days := integer_value(words[4])
362 deadline := (if valid then state.passes[index].last_day else state.now / 1440) + days
363 if index < 0
364 state.passes = state.passes.append(Pass(words[3], deadline))
365 else
366 state.passes[index].last_day = deadline
367 emit("PASS " + words[3] + " until " + string(deadline))
368 return
369 if action == "EXIT" or action == "PAY" or action == "QUOTE" or action == "CHARGE" or action == "VALIDATE" or action == "MOVE"
370 plate := words[3]
371 index := parked_id(state, plate)
372 if index < 0
373 fail(plate + " is not parked")
374 return
375 item := state.vehicles[index]
376 if action == "MOVE"
377 target := spot_id(state, words[4])
378 if item.kind == 4
379 fail("bus " + plate + " cannot be moved")
380 elif target < 0
381 fail("unknown spot " + words[4])
382 elif item.spot == target
383 fail(plate + " is already at " + words[4])
384 elif state.spots[target].closed
385 fail("spot " + words[4] + " is closed")
386 elif state.spots[target].occupant >= 0
387 fail("spot " + words[4] + " is occupied by " + state.vehicles[state.spots[target].occupant].plate)
388 elif preference_rank(item.kind, state.spots[target].kind) < 0
389 fail(type_name(item.kind) + " cannot use " + spot_kind(state.spots[target].kind) + " spot " + words[4])
390 else
391 move_vehicle(&state, index, target)
392 process_queue(&state)
393 return
394 if action == "CHARGE"
395 energy := integer_value(words[4])
396 if item.kind != 2
397 fail(plate + " is not an electric car")
398 elif state.spots[item.spot].kind != 3
399 fail(plate + " is not at a charger")
400 elif item.paid_at >= 0
401 fail(plate + " has already paid")
402 elif item.energy + energy > 100
403 fail("charge limit exceeded for " + plate)
404 else
405 int cost := energy * state.fees[2]
406 state.vehicles[index].energy += energy
407 state.vehicles[index].charging += cost
408 emit("CHARGE " + plate + " kwh " + string(energy) + " cost " + string(cost) + " stay_kwh " + string(state.vehicles[index].energy) + " stay_cost " + string(state.vehicles[index].charging))
409 return
410 if action == "VALIDATE"
411 minutes := integer_value(words[4])
412 if item.paid_at >= 0
413 fail(plate + " has already paid")
414 elif item.validation + minutes > 180
415 fail("validation limit exceeded for " + plate)
416 else
417 state.vehicles[index].validation += minutes
418 emit("VALIDATE " + plate + " " + string(minutes) + " total " + string(state.vehicles[index].validation))
419 return
420 lost := action == "EXIT" and #words == 5
421 if item.paid_at >= 0 and (action == "PAY" or lost)
422 fail(plate + " has already paid")
423 return
424 amounts := exit_fees(state, index, lost)
425 if action == "PAY"
426 state.vehicles[index].paid_at = state.now
427 state.vehicles[index].paid_parking = amounts[0]
428 state.vehicles[index].paid_night = amounts[1]
429 emit("PAY " + plate + " parking " + string(amounts[0]) + " night " + string(amounts[1]) + " charge " + string(amounts[3]) + " total " + string(amounts[4]))
430 else
431 fee_line(state, index, action, amounts)
432 if action == "EXIT"
433 state.vehicles[index].exit_location = location(state, index)
434 state.vehicles[index].exited = state.now
435 state.vehicles[index].total = amounts[4]
436 state.vehicles[index].lost = lost
437 state.vehicles[index].active = false
438 state.exits[state.exit_used] = index
439 state.exit_used += 1
440 vacate(&state, index)
441 state.parked -= 1
442 state.stats[item.kind][1] += 1
443 state.stats[item.kind][3] += state.now - item.entered
444 state.daily[state.now / 1440 + 1][item.kind][1] += 1
445 state.daily[state.now / 1440 + 1][item.kind][3] += amounts[4]
446 state.revenue[item.kind][0] += 1
447 for: 0 <= k < 5
448 state.revenue[item.kind][k + 1] += amounts[k]
449 process_queue(&state)
450 return
451 if action == "REBALANCE"
452 moved := 0
453 snapshot := location_order(state)
454 for index := snapshot[]
455 item := state.vehicles[index]
456 if item.kind != 4 and preference_rank(item.kind, state.spots[item.spot].kind) > 0
457 target := assign_spot(state, item.kind)
458 if target >= 0 and preference_rank(item.kind, state.spots[target].kind) < preference_rank(item.kind, state.spots[item.spot].kind)
459 move_vehicle(&state, index, target)
460 moved += 1
461 emit("REBALANCE " + string(moved))
462 process_queue(&state)
463 elif action == "WHERE"
464 plate := words[3]
465 index := parked_id(state, plate)
466 queued := waiting_id(state, plate)
467 last := -1
468 for: 0 <= k < state.exit_used
469 if state.vehicles[state.exits[k]].plate == plate
470 last = state.exits[k]
471 if index >= 0
472 item := state.vehicles[index]
473 emit("WHERE " + plate + " " + location(state, index) + " " + type_name(item.kind) + " " + clock_text(item.entered) + " stay " + duration(state.now - item.entered) + annotation(state, index))
474 elif queued >= 0
475 emit("WHERE " + plate + " waiting " + string(queued + 1))
476 elif last >= 0
477 emit("WHERE " + plate + " gone " + clock_text(state.vehicles[last].exited))
478 else
479 fail("unknown plate " + plate)
480 elif action == "LIST" or action == "OVERSTAY"
481 level := if action == "LIST" and #words == 4 then integer_value(words[3]) else 0
482 if level > state.levels
483 fail("unknown level " + string(level))
484 return
485 hours := if action == "OVERSTAY" then integer_value(words[3]) else 0
486 int[] found := []
487 for index := location_order(state)[]
488 item := state.vehicles[index]
489 if (level == 0 or state.spots[item.spot].level == level) and state.now - item.entered >= hours * 60
490 found = found.append(index)
491 if action == "OVERSTAY"
492 for: 1 <= k < int(#found)
493 index := found[k]
494 position := k
495 while position > 0 and state.vehicles[found[position - 1]].entered > state.vehicles[index].entered
496 found[position] = found[position - 1]
497 position -= 1
498 found[position] = index
499 emit(action + (if action == "OVERSTAY" then " " + string(hours) else "") + " " + string(#found))
500 for index := found[]
501 item := state.vehicles[index]
502 text := location(state, index) + " " + item.plate + " " + type_name(item.kind) + " " + clock_text(item.entered)
503 emit(text + (if action == "OVERSTAY" then " stay " + duration(state.now - item.entered) else annotation(state, index)))
504 elif action == "MAP"
505 emit("MAP " + string(state.levels))
506 for: 1 <= level <= state.levels
507 symbols := ""
508 count := 0
509 for item := state.spots[]
510 if item.level == level
511 if count > 0 and count % 10 == 0
512 symbols += " "
513 symbols += if item.closed then "#" elif item.occupant >= 0 then "MCLE"[item.kind] else "mcle"[item.kind]
514 count += 1
515 emit("L" + string(level) + " " + symbols)
516 elif action == "OCCUPANCY"
517 emit("OCCUPANCY vehicles " + string(state.parked) + " waiting " + string(#state.waiting))
518 all_used := array(4, fill = 0)
519 all_available := array(4, fill = 0)
520 all_closed := 0
521 for: 1 <= level <= state.levels
522 used := array(4, fill = 0)
523 available := array(4, fill = 0)
524 closed_count := 0
525 for item := state.spots[]
526 if item.level == level
527 if item.closed
528 closed_count += 1
529 else
530 available[item.kind] += 1
531 if item.occupant >= 0
532 used[item.kind] += 1

535 all_available += available
536 all_closed += closed_count
537 numerator := all_used.sum()
538 denominator := all_available.sum()
539 usage := if denominator == 0 then 0 else numerator * 1000 / denominator
540 emit(occupancy_line("TOTAL", all_used, all_available, all_closed) + " rate " + tenth(usage) + "%")
541 elif action == "SIGN"
542 spaces := array(5, fill = 0)
543 for: 0 <= kind < 4
544 for: 0 <= k < int(#state.spots)
545 if free_spot(state, k) and preference_rank(kind, state.spots[k].kind) >= 0
546 spaces[kind] += 1
547 run := 0
548 previous_level := 0
549 for: 0 <= k < int(#state.spots)
550 item := state.spots[k]
551 if item.level != previous_level or item.kind != 2 or not free_spot(state, k)
552 spaces[4] += run / 3
553 run = 0
554 if item.kind == 2 and free_spot(state, k)
555 run += 1
556 previous_level = item.level
557 spaces[4] += run / 3
558 text := "SIGN"
559 for: 0 <= k < 5
560 text += " " + type_name(k) + " " + (if spaces[k] == 0 then "FULL" else string(spaces[k]))
561 emit(text)
562 elif action == "RATES"
563 emit("RATES")
564 for: 0 <= k < 5
565 emit(type_name(k) + " " + string(state.rates[k]) + " " + string(state.caps[k]))
566 emit("NIGHT " + string(state.fees[0]) + " LOST " + string(state.fees[1]) + " KWH " + string(state.fees[2]))
567 elif action == "PASSES"
568 plates := [for item := state.passes[] if item.last_day >= state.now / 1440 + 1 then item.plate]
569 plates = plates.sorted()
570 emit("PASSES " + string(#plates))
571 for plate := plates[]
572 index := pass_id(state, plate)
573 emit(plate + " until " + string(state.passes[index].last_day) + (if parked_id(state, plate) >= 0 then " IN" else ""))
574 elif action == "HISTORY"
575 int[] found := []
576 total := 0
577 for: 0 <= k < state.exit_used
578 int index := state.exits[k]
579 if state.vehicles[index].plate == words[3]
580 found = found.append(index)
581 total += state.vehicles[index].total
582 if #found == 0
583 fail("no history for " + words[3])
584 else
585 emit("HISTORY " + words[3] + " visits " + string(#found) + " paid " + string(total))
586 for index := found[]
587 item := state.vehicles[index]
588 emit(clock_text(item.entered) + " " + clock_text(item.exited) + " " + item.exit_location + " " + type_name(item.kind) + " " + string(item.total) + (if item.lost then " LOST" else ""))
589 elif action == "REVENUE"
590 emit("REVENUE")
