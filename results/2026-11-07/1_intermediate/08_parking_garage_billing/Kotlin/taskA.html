1import kotlin.math.min
2
3private val types = arrayOf("MOTORCYCLE","CAR","ECAR","VAN","BUS")
4private val kinds = arrayOf("MOTO","COMPACT","LARGE","EV")
5private val prefs = arrayOf(intArrayOf(0,1,2),intArrayOf(1,2),intArrayOf(3,1,2),intArrayOf(2),intArrayOf(2))
6private class Fault(message: String) : RuntimeException(message)
7private fun error(condition: Boolean, message: String) { if (condition) throw Fault(message) }
8private fun two(n: Int) = n.toString().padStart(2,'0')
9private fun time(n: Int) = "${n/1440+1}-${two(n/60%24)}:${two(n%60)}"
10private fun duration(n: Int) = "${n/60}:${two(n%60)}"
11private fun tenths(n: Int) = "${n/10}.${n%10}"
12private class Spot(val level: Int, val number: Int, val kind: Int, val order: Int) {
13 var closed = false
14 var occupant: Vehicle? = null
15 fun id() = "L$level-$number"
16 fun free() = !closed && occupant == null
17}
18private class Vehicle(val plate: String, val kind: Int, val entry: Int, var spots: List<Spot>) {
19 var energy = 0
20 var charge = 0
21 var validation = 0
22 var paidAt = -1
23 var paidParking = 0
24 var paidNight = 0
25 fun location() = spots[0].id()+(if (kind == 4) "..${spots[2].number}" else "")
26}
27private data class Waiting(val plate: String, val kind: Int, val joined: Int)
28private data class Visit(val entry: Int, val exit: Int, val location: String, val kind: Int, val total: Int, val loss: Boolean)
29private class Garage {
30 var now = 0
31 var peak = 0
32 var peakAt = 0
33 val levels = mutableListOf<List<Spot>>()
34 val spots = mutableListOf<Spot>()
35 val parked = mutableMapOf<String,Vehicle>()
36 val queue = mutableListOf<Waiting>()
37 val passes = sortedMapOf<String,Int>()
38 val history = mutableMapOf<String,MutableList<Visit>>()
39 val rates = arrayOf(intArrayOf(100,800),intArrayOf(200,1600),intArrayOf(200,1600),intArrayOf(300,2400),intArrayOf(600,4800))
40 var night = 300
41 var lost = 2000
42 var kwh = 30
43 val stats = Array(5) { IntArray(9) }
44 val days = mutableMapOf<Int,Array<IntArray>>()
45 var output = mutableListOf<String>()
46 fun daily(date: Int) = days.getOrPut(date) { Array(5) { IntArray(4) } }
47 fun pass(plate: String) = (passes[plate] ?: 0) >= now/1440+1
48 fun vehicle(plate: String) = parked[plate] ?: throw Fault("$plate is not parked")
49 fun spot(id: String): Spot {
50 val p = id.drop(1).split('-')
51 val l = p[0].toInt()
52 val n = p[1].toInt()
53 error(l > levels.size || n > levels[l-1].size,"unknown spot $id")
54 return levels[l-1][n-1]
55 }
56 fun assign(kind: Int): List<Spot>? {
57 if (kind == 4) {
58 for (level in levels) {
59 for (i in 0 until level.size-2) {
60 val triple = level.subList(i,i+3)
61 if (triple.all { it.kind == 2 && it.free() }) return triple.toList()
62 }
63 }
64 } else {
65 for (wanted in prefs[kind]) {
66 spots.firstOrNull { it.kind == wanted && it.free() }?.let { return listOf(it) }
67 }
68 }
69 return null
70 }
71 fun occupy(v: Vehicle) { for (s in v.spots) s.occupant = v }
72 fun free(v: Vehicle) { for (s in v.spots) s.occupant = null }
73 fun park(plate: String, kind: Int, assigned: List<Spot>): Vehicle {
74 val v = Vehicle(plate,kind,now,assigned)
75 parked[plate] = v
76 occupy(v)
77 stats[kind][0]++
78 daily(now/1440+1)[kind][0]++
79 if (parked.size > peak) {
80 peak = parked.size
81 peakAt = now
82 }
83 return v
84 }
85 fun refuse(kind: Int, at: Int) {
86 stats[kind][2]++
87 daily(at/1440+1)[kind][2]++
88 }
89 fun giveup() {
90 for (w in queue.toList()) {
91 if (now-w.joined >= 30) {
92 queue.remove(w)
93 refuse(w.kind,w.joined+30)
94 output.add("GIVEUP ${w.plate} ${types[w.kind]} ${time(w.joined+30)}")
95 }
96 }
97 }
98 fun admit() {
99 for (w in queue.toList()) {
100 val assigned = assign(w.kind)
101 if (assigned != null) {
102 queue.remove(w)
103 val v = park(w.plate,w.kind,assigned)
104 output.add("ADMIT ${v.plate} ${types[v.kind]} ${v.location()} waited ${duration(now-w.joined)}")
105 }
106 }
107 }
108 fun interval(v: Vehicle, start: Int, free: Int): IntArray {
109 val total = now-start
110 if (pass(v.plate) || total <= free) return intArrayOf(0,0)
111 var parking = 0
112 for (offset in 0 until total step 1440) {
113 val billable = min(total-offset,1440)-(if (offset == 0) free else 0)
114 parking += min((billable+29)/30*rates[v.kind][0],rates[v.kind][1])
115 }
116 return intArrayOf(parking,(now/1440-start/1440)*night)
117 }
118 fun fees(v: Vehicle, loss: Boolean): IntArray {
119 val pair = interval(v,if (v.paidAt < 0) v.entry else v.paidAt,if (v.paidAt < 0) 15+v.validation else 15)
120 val parking = pair[0]+(if (v.paidAt < 0) 0 else v.paidParking)
121 val surcharge = pair[1]+(if (v.paidAt < 0) 0 else v.paidNight)
122 val ticket = if (loss) lost else 0
123 val total = parking+surcharge+ticket+v.charge
124 return intArrayOf(parking,surcharge,ticket,v.charge,total,if (v.paidAt < 0) total else pair[0]+pair[1])
125 }
126 fun feeLine(label: String, v: Vehicle, f: IntArray) = "$label ${v.plate} ${v.location()} stay ${duration(now-v.entry)}"+
127 " parking ${f[0]} night ${f[1]} lost ${f[2]} charge ${f[3]} total ${f[4]} due ${f[5]}"
128 fun annotation(v: Vehicle) = (if (pass(v.plate)) " PASS" else "")+(if (v.validation > 0) " VAL ${v.validation}" else "")+
129 (if (v.energy > 0) " KWH ${v.energy}" else "")+(if (v.paidAt >= 0) " PAID" else "")
130 fun ordered() = parked.values.sortedBy { it.spots[0].order }
131 fun move(v: Vehicle, target: List<Spot>) {
132 val old = v.location()
133 free(v)
134 v.spots = target
135 occupy(v)
136 output.add("MOVE ${v.plate} $old -> ${v.location()}")
137 }
138 fun occupancy(target: List<Spot>): IntArray {
139 val a = IntArray(9)
140 for (s in target) {
141 if (s.closed) a[8]++ else a[4+s.kind]++
142 if (s.occupant != null) a[s.kind]++
143 }
144 return a
145 }
146 fun occupancyLine(label: String, a: IntArray) = label+(0..3).joinToString("") { " ${kinds[it]} ${a[it]}/${a[it+4]}" }+" closed ${a[8]}"
147 fun revenueLine(label: String, a: IntArray) = "$label exits ${a[1]} parking ${a[4]} night ${a[5]} lost ${a[6]} charge ${a[7]} total ${a[8]}"
148 fun statsLine(label: String, a: IntArray) = "$label entries ${a[0]} exits ${a[1]} refused ${a[2]} avg_stay ${tenths(if (a[1] == 0) 0 else a[3]*10/a[1])}"
149 fun dailyLine(label: String, a: IntArray) = "$label entries ${a[0]} exits ${a[1]} refused ${a[2]} revenue ${a[3]}"
150 fun sum(a: Array<IntArray>, width: Int) = IntArray(width) { i -> a.sumOf { it[i] } }
151 fun execute(t: List<String>) {
152 var trigger = false
153 val command = t[2]
154 when (command) {
155 "LEVEL" -> {
156 val added = mutableListOf<Spot>()
157 val counts = IntArray(4) { t[it+3].toInt() }
158 for (k in 0..3) {
159 repeat(counts[k]) {
160 val s = Spot(levels.size+1,added.size+1,k,spots.size)
161 added.add(s)
162 spots.add(s)
163 }
164 }
165 levels.add(added)
166 output.add("LEVEL ${levels.size} spots ${added.size} MOTO ${counts[0]} COMPACT ${counts[1]} LARGE ${counts[2]} EV ${counts[3]}")
167 trigger = true
168 }
169 "CLOSE", "OPEN" -> {
170 val s = spot(t[3])
171 if (command == "CLOSE") {
172 error(s.closed,"spot ${t[3]} is already closed")
173 error(s.occupant != null,"spot ${t[3]} is occupied by ${s.occupant?.plate}")
174 s.closed = true
175 } else {
176 error(!s.closed,"spot ${t[3]} is not closed")
177 s.closed = false
178 trigger = true
179 }
180 output.add("$command ${s.id()} ${kinds[s.kind]}")
181 }
182 "RATE" -> {
183 val k = types.indexOf(t[3])
184 error(k < 0,"unknown type ${t[3]}")
185 val rate = t[4].toInt()
186 val cap = t[5].toInt()
187 error(cap < rate,"cap below rate for ${t[3]}")
188 rates[k] = intArrayOf(rate,cap)
189 output.add("RATE ${t[3]} $rate $cap")
190 }
191 "FEE" -> {
192 val amount = t[4].toInt()
193 error(t[3] !in listOf("NIGHT","LOST","KWH"),"unknown fee ${t[3]}")
194 when (t[3]) {
195 "NIGHT" -> night = amount
196 "LOST" -> lost = amount
197 else -> kwh = amount
198 }
199 output.add("FEE ${t[3]} $amount")
200 }
201 "ENTER" -> {
202 val plate = t[3]
203 val k = types.indexOf(t[4])
204 error(k < 0,"unknown type ${t[4]}")
205 val existing = parked[plate]
206 error(existing != null,"$plate is already parked at ${existing?.location()}")
207 error(queue.any { it.plate == plate },"$plate is already waiting")
208 val assigned = assign(k)
209 if (assigned != null) output.add("ENTER $plate ${types[k]} ${park(plate,k,assigned).location()}")
210 else if (queue.size < 3) {
211 queue.add(Waiting(plate,k,now))
212 output.add("WAIT $plate ${types[k]} ${queue.size}")
213 } else {
214 refuse(k,now)
215 output.add("FULL $plate ${types[k]}")
216 }
217 }
218 "LEAVE" -> {
219 val w = queue.firstOrNull { it.plate == t[3] } ?: throw Fault("${t[3]} is not waiting")
220 queue.remove(w)
221 output.add("LEAVE ${w.plate} ${types[w.kind]}")
222 }
223 "EXIT", "PAY", "QUOTE" -> {
224 val v = vehicle(t[3])
225 val loss = t.size == 5
226 error(v.paidAt >= 0 && (command == "PAY" || loss),"${v.plate} has already paid")
227 val f = fees(v,loss)
228 if (command == "PAY") {
229 v.paidAt = now
230 v.paidParking = f[0]
231 v.paidNight = f[1]
232 output.add("PAY ${v.plate} parking ${f[0]} night ${f[1]} charge ${f[3]} total ${f[4]}")
233 } else {
234 output.add(feeLine(command,v,f))
235 if (command == "EXIT") {
236 history.getOrPut(v.plate) { mutableListOf() }.add(Visit(v.entry,now,v.location(),v.kind,f[4],loss))
237 stats[v.kind][1]++
238 stats[v.kind][3] += now-v.entry
239 for (i in 0..4) stats[v.kind][i+4] += f[i]
240 val d = daily(now/1440+1)[v.kind]
241 d[1]++
242 d[3] += f[4]
243 free(v)
244 parked.remove(v.plate)
245 trigger = true
246 }
247 }
248 }
249 "CHARGE" -> {
250 val v = vehicle(t[3])
251 error(v.kind != 2,"${v.plate} is not an electric car")
252 error(v.spots[0].kind != 3,"${v.plate} is not at a charger")
253 error(v.paidAt >= 0,"${v.plate} has already paid")
254 val energy = t[4].toInt()
255 error(v.energy+energy > 100,"charge limit exceeded for ${v.plate}")
256 val cost = energy*kwh
257 v.energy += energy
258 v.charge += cost
259 output.add("CHARGE ${v.plate} kwh $energy cost $cost stay_kwh ${v.energy} stay_cost ${v.charge}")
260 }
261 "VALIDATE" -> {
262 val v = vehicle(t[3])
263 error(v.paidAt >= 0,"${v.plate} has already paid")
264 val minutes = t[4].toInt()
265 error(v.validation+minutes > 180,"validation limit exceeded for ${v.plate}")
266 v.validation += minutes
267 output.add("VALIDATE ${v.plate} $minutes total ${v.validation}")
268 }
269 "PASS" -> {
270 val end = (if (pass(t[3])) passes.getValue(t[3]) else now/1440)+t[4].toInt()
271 passes[t[3]] = end
272 output.add("PASS ${t[3]} until $end")
273 }
274 "UNPASS" -> {
275 error(!pass(t[3]),"${t[3]} has no pass")
276 passes.remove(t[3])
277 output.add("UNPASS ${t[3]}")
278 }
279 "MOVE" -> {
280 val v = vehicle(t[3])
281 error(v.kind == 4,"bus ${v.plate} cannot be moved")
282 val s = spot(t[4])
283 error(v.spots[0] === s,"${v.plate} is already at ${t[4]}")
284 error(s.closed,"spot ${t[4]} is closed")
285 error(s.occupant != null,"spot ${t[4]} is occupied by ${s.occupant?.plate}")
286 error(s.kind !in prefs[v.kind],"${types[v.kind]} cannot use ${kinds[s.kind]} spot ${t[4]}")
287 move(v,listOf(s))
288 trigger = true
289 }
290 "REBALANCE" -> {
291 var moved = 0
292 for (v in ordered()) {
293 if (v.kind == 4) continue
294 val target = assign(v.kind)
295 if (target != null && prefs[v.kind].indexOf(target[0].kind) < prefs[v.kind].indexOf(v.spots[0].kind)) {
296 move(v,target)
297 moved++
298 }
299 }
300 output.add("REBALANCE $moved")
301 trigger = true
302 }
303 "WHERE" -> {
304 val v = parked[t[3]]
305 val p = queue.indexOfFirst { it.plate == t[3] }
306 val visits = history[t[3]]
307 error(v == null && p < 0 && visits == null,"unknown plate ${t[3]}")
308 if (v != null) output.add("WHERE ${v.plate} ${v.location()} ${types[v.kind]} ${time(v.entry)} stay ${duration(now-v.entry)}${annotation(v)}")
309 else if (p >= 0) output.add("WHERE ${t[3]} waiting ${p+1}")
310 else output.add("WHERE ${t[3]} gone ${time(visits!!.last().exit)}")
311 }
312 "LIST" -> {
313 val l = if (t.size == 4) t[3].toInt() else 0
314 error(l > levels.size,"unknown level $l")
315 val found = ordered().filter { l == 0 || it.spots[0].level == l }
316 output.add("LIST ${found.size}")
317 for (v in found) output.add("${v.location()} ${v.plate} ${types[v.kind]} ${time(v.entry)}${annotation(v)}")
318 }
319 "MAP" -> {
320 output.add("MAP ${levels.size}")
321 for ((i,level) in levels.withIndex()) {
322 val symbols = level.map { s -> if (s.closed) '#' else if (s.occupant != null) kinds[s.kind][0] else kinds[s.kind][0].lowercaseChar() }.joinToString("")
323 output.add("L${i+1} "+symbols.chunked(10).joinToString(" "))
324 }
325 }
326 "OCCUPANCY" -> {
327 output.add("OCCUPANCY vehicles ${parked.size} waiting ${queue.size}")
328 for ((i,level) in levels.withIndex()) output.add(occupancyLine("L${i+1}",occupancy(level)))
329 val a = occupancy(spots)
330 val used = a.take(4).sum()
331 val available = a.drop(4).take(4).sum()
332 output.add(occupancyLine("TOTAL",a)+" rate ${tenths(if (available == 0) 0 else used*1000/available)}%")
333 }
334 "SIGN" -> {
335 var line = "SIGN"
336 for (k in 0..4) {
337 var count = 0
338 if (k == 4) {
339 for (level in levels) {
340 var run = 0
341 for (s in level) {
342 if (s.kind == 2 && s.free()) {
343 run++
344 if (run == 3) {
345 count++
346 run = 0
347 }
348 } else run = 0
349 }
350 }
351 } else {
352 count = spots.count { it.kind in prefs[k] && it.free() }
353 }
354 line += " ${types[k]} "+(if (count == 0) "FULL" else count.toString())
355 }
356 output.add(line)
357 }
358 "RATES" -> {
359 output.add("RATES")
360 for (k in 0..4) output.add("${types[k]} ${rates[k][0]} ${rates[k][1]}")
361 output.add("NIGHT $night LOST $lost KWH $kwh")
362 }
363 "PASSES" -> {
364 val found = passes.keys.filter { pass(it) }
365 output.add("PASSES ${found.size}")
366 for (plate in found) output.add("$plate until ${passes[plate]}"+(if (plate in parked) " IN" else ""))
367 }
368 "OVERSTAY" -> {
369 val hours = t[3].toInt()
370 val found = ordered().filter { now-it.entry >= hours*60 }.sortedBy { it.entry }
371 output.add("OVERSTAY $hours ${found.size}")
372 for (v in found) output.add("${v.location()} ${v.plate} ${types[v.kind]} ${time(v.entry)} stay ${duration(now-v.entry)}")
373 }
374 "HISTORY" -> {
375 val visits = history[t[3]] ?: throw Fault("no history for ${t[3]}")
376 output.add("HISTORY ${t[3]} visits ${visits.size} paid ${visits.sumOf { it.total }}")
377 for (v in visits) output.add("${time(v.entry)} ${time(v.exit)} ${v.location} ${types[v.kind]} ${v.total}"+(if (v.loss) " LOST" else ""))
378 }
379 "REVENUE" -> {
380 output.add("REVENUE")
381 for (k in 0..4) output.add(revenueLine(types[k],stats[k]))
382 output.add(revenueLine("TOTAL",sum(stats,9)))
383 }
384 "DAILY" -> {
385 val d = t[3].toInt()
386 val a = daily(d)
387 output.add(dailyLine("DAILY $d",sum(a,4)))
388 for (k in 0..4) output.add(dailyLine(types[k],a[k]))
389 }
390 "STATS" -> {
391 output.add("STATS parked ${parked.size} peak $peak at "+(if (peak == 0) "-" else time(peakAt)))
392 for (k in 0..4) output.add(statsLine(types[k],stats[k]))
393 output.add(statsLine("ALL",sum(stats,9)))
394 }
395 }
396 if (trigger) admit()
397 }
398 fun process(line: String): List<String> {
399 val t = line.split(' ')
400 val hm = t[1].split(':')
401 now = (t[0].toInt()-1)*1440+hm[0].toInt()*60+hm[1].toInt()
402 output = mutableListOf()
403 giveup()
404 try { execute(t) }
405 catch (f: Fault) { output.add("ERROR ${f.message}") }
406 return output
407 }
408}
409fun main() {
410 val input = System.`in`.bufferedReader()
411 val output = System.out.bufferedWriter()
412 val garage = Garage()
413 repeat(input.readLine().toInt()) {
414 for (line in garage.process(input.readLine())) {
415 output.write(line)
416 output.newLine()
417 }
418 }
419 output.flush()
420}
