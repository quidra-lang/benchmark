14var types = []string{"MOTORCYCLE", "CAR", "ECAR", "VAN", "BUS"}
15var kinds = []string{"MOTO", "COMPACT", "LARGE", "EV"}
16var preferences = [][]int{{0,1,2}, {1,2}, {3,1,2}, {2}, {2}}
17type spot struct { level, number, kind, vehicle int; closed bool }
18type vehicle struct {
19 plate string
20 kind, spot, entry, energy, charge, validation int
21 paid bool
22 payment, parking, night int
23}
24type waiting struct { plate string; kind, joined int }
25type bill struct { parking, night, lost, charge, total, due int }
26type record struct { entry, exit, kind, total int; location string; lost bool }
27type counts struct { entries, exits, refused, revenue, stay int }
28type revenue struct { exits, parking, night, lost, charge, total int }
29type garage struct {
30 now, peak, peakTime int
31 rates, caps [5]int
32 fees [3]int
33 spots []spot
34 levels [][]int
35 vehicles []*vehicle
36 queue []waiting
37 parked, passes map[string]int
38 histories map[string][]record
39 totals [5]counts
40 daily map[int][5]counts
41 revenue [5]revenue
42 output []string
43}
44func number(s string) int {
45 value,_ := strconv.Atoi(s)
46 return value
47}
48func timeText(t int) string { return fmt.Sprintf("%d-%02d:%02d",t/1440+1,t%1440/60,t%60) }
49func duration(t int) string { return fmt.Sprintf("%d:%02d",t/60,t%60) }
50func tenth(q int) string { return fmt.Sprintf("%d.%d",q/10,q%10) }
51func (g *garage) line(pattern string, args ...any) { g.output = append(g.output,fmt.Sprintf(pattern,args...)) }
52func (g *garage) error(condition bool, reason string) bool {
53 if condition { g.line("ERROR %s",reason) }
54 return condition
55}
56func (g *garage) label(s int) string { return fmt.Sprintf("L%d-%d",g.spots[s].level,g.spots[s].number) }
57func (g *garage) location(v *vehicle) string {
58 result := g.label(v.spot)
59 if v.kind == 4 { result += fmt.Sprintf("..%d",g.spots[v.spot].number+2) }
60 return result
61}
62func (g *garage) free(s int) bool { return !g.spots[s].closed && g.spots[s].vehicle < 0 }
63func (g *garage) validPass(p string) bool {
64 last,ok := g.passes[p]
65 return ok && g.now/1440+1 <= last
66}
67func (g *garage) spotID(s string) int {
68 levelText,numberText,_ := strings.Cut(s[1:],"-")
69 level,n := number(levelText),number(numberText)
70 if level > len(g.levels) || n > len(g.levels[level-1]) { return -1 }
71 return g.levels[level-1][n-1]
72}
73func (g *garage) getVehicle(p string) int {
74 id,ok := g.parked[p]
75 if g.error(!ok,p+" is not parked") { return -1 }
76 return id
77}
78func (g *garage) waiting(p string) int {
79 return slices.IndexFunc(g.queue,func(w waiting) bool { return w.plate == p })
80}
81func preference(kind, spotKind int) int {
82 return slices.Index(preferences[kind],spotKind)
83}
84func (g *garage) assign(kind int) int {
85 if kind == 4 {
86 for _,level := range g.levels {
87 for j:=0; j+2<len(level); j++ {
88 a,b,c := level[j],level[j+1],level[j+2]
89 if g.spots[a].kind == 2 && g.spots[b].kind == 2 && g.spots[c].kind == 2 && g.free(a) && g.free(b) && g.free(c) { return a }
90 }
91 }
92 } else {
93 for _,k := range preferences[kind] {
94 for i,s := range g.spots {
95 if s.kind == k && g.free(i) { return i }
96 }
97 }
98 }
99 return -1
100}
101func (g *garage) occupy(id, s int) {
102 v := g.vehicles[id]
103 v.spot = s
104 size := 1
105 if v.kind == 4 { size = 3 }
106 for i:=0; i<size; i++ { g.spots[s+i].vehicle = id }
107}
108func (g *garage) vacate(id int) {
109 v := g.vehicles[id]
110 size := 1
111 if v.kind == 4 { size = 3 }
112 for i:=0; i<size; i++ { g.spots[v.spot+i].vehicle = -1 }
113}
114func (g *garage) park(plate string, kind, s int) int {
115 id := len(g.vehicles)
116 g.vehicles = append(g.vehicles,&vehicle{plate:plate,kind:kind,spot:s,entry:g.now})
117 g.parked[plate] = id
118 g.occupy(id,s)
119 g.totals[kind].entries++
120 today := g.daily[g.now/1440+1]
121 today[kind].entries++
122 g.daily[g.now/1440+1] = today
123 if len(g.parked) > g.peak {
124 g.peak = len(g.parked)
125 g.peakTime = g.now
126 }
127 return id
128}
129func (g *garage) refuse(kind, t int) {
130 g.totals[kind].refused++
131 today := g.daily[t/1440+1]
132 today[kind].refused++
133 g.daily[t/1440+1] = today
134}
135func (g *garage) giveups() {
136 remaining := []waiting{}
137 for _,w := range g.queue {
138 if g.now-w.joined >= 30 {
139 g.line("GIVEUP %s %s %s",w.plate,types[w.kind],timeText(w.joined+30))
140 g.refuse(w.kind,w.joined+30)
141 } else { remaining = append(remaining,w) }
142 }
143 g.queue = remaining
144}
145func (g *garage) processQueue() {
146 remaining := []waiting{}
147 for _,w := range g.queue {
148 s := g.assign(w.kind)
149 if s < 0 { remaining = append(remaining,w) } else {
150 id := g.park(w.plate,w.kind,s)
151 g.line("ADMIT %s %s %s waited %s",w.plate,types[w.kind],g.location(g.vehicles[id]),duration(g.now-w.joined))
152 }
153 }
154 g.queue = remaining
155}
156func (g *garage) interval(v *vehicle, from, freeTime int) (int,int) {
157 length := g.now-from
158 if g.validPass(v.plate) || length <= freeTime { return 0,0 }
159 parking := 0
160 for i:=0; i*1440<length; i++ {
161 minutes := min(length-i*1440,1440)
162 if i == 0 { minutes -= freeTime }
163 parking += min((minutes+29)/30*g.rates[v.kind],g.caps[v.kind])
164 }
165 return parking,(g.now/1440-from/1440)*g.fees[0]
166}
167func (g *garage) bill(v *vehicle, loss bool) bill {
168 from,freeTime := v.entry,15+v.validation
169 if v.paid { from,freeTime = v.payment,15 }
170 p,n := g.interval(v,from,freeTime)
171 due := p+n
172 if v.paid {
173 p += v.parking
174 n += v.night
175 }
176 lost := 0
177 if loss { lost = g.fees[1] }
178 total := p+n+lost+v.charge
179 if !v.paid { due = total }
180 return bill{p,n,lost,v.charge,total,due}
181}
182func (g *garage) feeLine(heading string, v *vehicle, b bill) {
183 g.line("%s %s %s stay %s parking %d night %d lost %d charge %d total %d due %d",heading,v.plate,g.location(v),duration(g.now-v.entry),b.parking,b.night,b.lost,b.charge,b.total,b.due)
184}
185func (g *garage) annotation(v *vehicle) string {
186 s := ""
187 if g.validPass(v.plate) { s += " PASS" }
188 if v.validation > 0 { s += fmt.Sprintf(" VAL %d",v.validation) }
189 if v.energy > 0 { s += fmt.Sprintf(" KWH %d",v.energy) }
190 if v.paid { s += " PAID" }
191 return s
192}
193func (g *garage) orderedVehicles() []int {
194 ids := slices.Collect(maps.Values(g.parked))
195 sort.Slice(ids,func(i,j int) bool { return g.vehicles[ids[i]].spot < g.vehicles[ids[j]].spot })
196 return ids
197}
198func (g *garage) move(id,s int) {
199 before := g.location(g.vehicles[id])
200 g.vacate(id)
201 g.occupy(id,s)
202 g.line("MOVE %s %s -> %s",g.vehicles[id].plate,before,g.label(s))
203}
204func (g *garage) execute(a []string) {
205 cmd := a[0]
206 switch cmd {
207 case "LEVEL":
208 level := []int{}
209 for k:=0; k<4; k++ {
210 for range number(a[k+1]) {
211 level = append(level,len(g.spots))
212 g.spots = append(g.spots,spot{level:len(g.levels)+1,number:len(level),kind:k,vehicle:-1})
213 }
214 }
215 g.levels = append(g.levels,level)
216 g.line("LEVEL %d spots %d MOTO %s COMPACT %s LARGE %s EV %s",len(g.levels),len(level),a[1],a[2],a[3],a[4])
217 g.processQueue()
218 case "CLOSE","OPEN":
219 s := g.spotID(a[1])
220 if g.error(s < 0,"unknown spot "+a[1]) { return }
221 spot := &g.spots[s]
222 if cmd == "CLOSE" {
223 if g.error(spot.closed,"spot "+a[1]+" is already closed") { return }
224 if spot.vehicle >= 0 {
225 g.error(true,"spot "+a[1]+" is occupied by "+g.vehicles[spot.vehicle].plate)
226 return
227 }
228 spot.closed = true
229 } else {
230 if g.error(!spot.closed,"spot "+a[1]+" is not closed") { return }
231 spot.closed = false
232 }
233 g.line("%s %s %s",cmd,a[1],kinds[spot.kind])
234 if cmd == "OPEN" { g.processQueue() }
235 case "RATE":
236 kind := slices.Index(types,a[1])
237 if g.error(kind < 0,"unknown type "+a[1]) { return }
238 rate,cap := number(a[2]),number(a[3])
239 if g.error(cap < rate,"cap below rate for "+a[1]) { return }
240 g.rates[kind],g.caps[kind] = rate,cap
241 g.line("RATE %s %d %d",a[1],rate,cap)
242 case "FEE":
243 item := slices.Index([]string{"NIGHT","LOST","KWH"},a[1])
244 if g.error(item < 0,"unknown fee "+a[1]) { return }
245 g.fees[item] = number(a[2])
246 g.line("FEE %s %s",a[1],a[2])
247 case "ENTER":
248 kind := slices.Index(types,a[2])
249 if g.error(kind < 0,"unknown type "+a[2]) { return }
250 if id,ok := g.parked[a[1]]; ok {
251 g.error(true,a[1]+" is already parked at "+g.location(g.vehicles[id]))
252 return
253 }
254 if g.error(g.waiting(a[1]) >= 0,a[1]+" is already waiting") { return }
255 s := g.assign(kind)
256 if s >= 0 {
257 id := g.park(a[1],kind,s)
258 g.line("ENTER %s %s %s",a[1],a[2],g.location(g.vehicles[id]))
259 } else if len(g.queue) < 3 {
260 g.queue = append(g.queue,waiting{a[1],kind,g.now})
261 g.line("WAIT %s %s %d",a[1],a[2],len(g.queue))
262 } else {
263 g.refuse(kind,g.now)
264 g.line("FULL %s %s",a[1],a[2])
265 }
266 case "LEAVE":
267 i := g.waiting(a[1])
268 if g.error(i < 0,a[1]+" is not waiting") { return }
269 g.line("LEAVE %s %s",a[1],types[g.queue[i].kind])
270 g.queue = slices.Delete(g.queue,i,i+1)
271 case "EXIT","PAY","QUOTE","CHARGE","VALIDATE","MOVE":
272 id := g.getVehicle(a[1])
273 if id < 0 { return }
274 v := g.vehicles[id]
275 switch cmd {
276 case "EXIT":
277 loss := len(a) == 3
278 if g.error(loss && v.paid,a[1]+" has already paid") { return }
279 b := g.bill(v,loss)
280 g.feeLine("EXIT",v,b)
281 g.histories[v.plate] = append(g.histories[v.plate],record{v.entry,g.now,v.kind,b.total,g.location(v),loss})
282 all := &g.totals[v.kind]
283 all.exits++
284 all.stay += g.now-v.entry
285 all.revenue += b.total
286 today := g.daily[g.now/1440+1]
287 today[v.kind].exits++
288 today[v.kind].revenue += b.total
289 g.daily[g.now/1440+1] = today
290 r := &g.revenue[v.kind]
291 r.exits++
292 r.parking += b.parking
293 r.night += b.night
294 r.lost += b.lost
295 r.charge += b.charge
296 r.total += b.total
297 g.vacate(id)
298 delete(g.parked,v.plate)
299 g.processQueue()
300 case "QUOTE": g.feeLine("QUOTE",v,g.bill(v,false))
301 case "PAY":
302 if g.error(v.paid,a[1]+" has already paid") { return }
303 b := g.bill(v,false)
304 v.paid = true
305 v.payment,v.parking,v.night = g.now,b.parking,b.night
306 g.line("PAY %s parking %d night %d charge %d total %d",v.plate,b.parking,b.night,b.charge,b.total)
307 case "CHARGE":
308 if g.error(v.kind != 2,a[1]+" is not an electric car") || g.error(g.spots[v.spot].kind != 3,a[1]+" is not at a charger") || g.error(v.paid,a[1]+" has already paid") { return }
309 energy := number(a[2])
310 if g.error(v.energy+energy > 100,"charge limit exceeded for "+a[1]) { return }
311 cost := energy*g.fees[2]
312 v.energy += energy
313 v.charge += cost
314 g.line("CHARGE %s kwh %d cost %d stay_kwh %d stay_cost %d",v.plate,energy,cost,v.energy,v.charge)
315 case "VALIDATE":
316 if g.error(v.paid,a[1]+" has already paid") { return }
317 minutes := number(a[2])
318 if g.error(v.validation+minutes > 180,"validation limit exceeded for "+a[1]) { return }
319 v.validation += minutes
320 g.line("VALIDATE %s %d total %d",v.plate,minutes,v.validation)
321 case "MOVE":
322 if g.error(v.kind == 4,"bus "+a[1]+" cannot be moved") { return }
323 s := g.spotID(a[2])
324 if g.error(s < 0,"unknown spot "+a[2]) || g.error(s == v.spot,a[1]+" is already at "+a[2]) { return }
325 if g.error(g.spots[s].closed,"spot "+a[2]+" is closed") { return }
326 if g.spots[s].vehicle >= 0 {
327 g.error(true,"spot "+a[2]+" is occupied by "+g.vehicles[g.spots[s].vehicle].plate)
328 return
329 }
330 if g.error(preference(v.kind,g.spots[s].kind) < 0,types[v.kind]+" cannot use "+kinds[g.spots[s].kind]+" spot "+a[2]) { return }
331 g.move(id,s)
332 g.processQueue()
333 }
334 case "PASS":
335 days := number(a[2])
336 if g.validPass(a[1]) { g.passes[a[1]] += days } else { g.passes[a[1]] = g.now/1440+days }
337 g.line("PASS %s until %d",a[1],g.passes[a[1]])
338 case "UNPASS":
339 if g.error(!g.validPass(a[1]),a[1]+" has no pass") { return }
340 delete(g.passes,a[1])
341 g.line("UNPASS %s",a[1])
342 case "REBALANCE":
343 ids,moved := g.orderedVehicles(),0
344 for _,id := range ids {
345 v := g.vehicles[id]
346 if v.kind == 4 { continue }
347 current := preference(v.kind,g.spots[v.spot].kind)
348 if current == 0 { continue }
349 s := g.assign(v.kind)
350 if s >= 0 && preference(v.kind,g.spots[s].kind) < current {
351 g.move(id,s)
352 moved++
353 }
354 }
355 g.line("REBALANCE %d",moved)
356 g.processQueue()
357 case "WHERE":
358 if id,ok := g.parked[a[1]]; ok {
359 v := g.vehicles[id]
360 g.line("WHERE %s %s %s %s stay %s%s",a[1],g.location(v),types[v.kind],timeText(v.entry),duration(g.now-v.entry),g.annotation(v))
361 } else if q := g.waiting(a[1]); q >= 0 { g.line("WHERE %s waiting %d",a[1],q+1) } else if records := g.histories[a[1]]; len(records) > 0 { g.line("WHERE %s gone %s",a[1],timeText(records[len(records)-1].exit)) } else { g.error(true,"unknown plate "+a[1]) }
362 case "LIST","OVERSTAY":
363 level := 0
364 if cmd == "LIST" && len(a) == 2 { level = number(a[1]) }
365 if g.error(level > len(g.levels),fmt.Sprintf("unknown level %d",level)) { return }
366 ids := []int{}
367 for _,id := range g.orderedVehicles() {
368 v := g.vehicles[id]
369 if level != 0 && g.spots[v.spot].level != level { continue }
370 if cmd == "OVERSTAY" && g.now-v.entry < number(a[1])*60 { continue }
371 ids = append(ids,id)
372 }
373 if cmd == "OVERSTAY" {
374 sort.SliceStable(ids,func(i,j int) bool { return g.vehicles[ids[i]].entry < g.vehicles[ids[j]].entry })
375 g.line("OVERSTAY %s %d",a[1],len(ids))
376 } else { g.line("LIST %d",len(ids)) }
377 for _,id := range ids {
378 v := g.vehicles[id]
379 suffix := g.annotation(v)
380 if cmd == "OVERSTAY" { suffix = " stay "+duration(g.now-v.entry) }
381 g.line("%s %s %s %s%s",g.location(v),v.plate,types[v.kind],timeText(v.entry),suffix)
382 }
383 case "MAP":
384 g.line("MAP %d",len(g.levels))
385 for i,level := range g.levels {
386 symbols := ""
387 for j,id := range level {
388 if j > 0 && j%10 == 0 { symbols += " " }
389 s := g.spots[id]
390 c := "mcle"[s.kind]
391 if s.closed { c = '#' } else if s.vehicle >= 0 { c -= 'a'-'A' }
392 symbols += string(c)
393 }
394 g.line("L%d %s",i+1,symbols)
395 }
396 case "OCCUPANCY":
397 g.line("OCCUPANCY vehicles %d waiting %d",len(g.parked),len(g.queue))
398 allUsed,allAvailable := [4]int{},[4]int{}
399 allClosed := 0
400 for i,level := range g.levels {
401 used,available := [4]int{},[4]int{}
402 closed := 0
403 for _,id := range level {
404 s := g.spots[id]
405 if s.closed { closed++ } else { available[s.kind]++ }
406 if s.vehicle >= 0 { used[s.kind]++ }
407 }
408 s := fmt.Sprintf("L%d",i+1)
409 for k:=0; k<4; k++ {
410 s += fmt.Sprintf(" %s %d/%d",kinds[k],used[k],available[k])
411 allUsed[k] += used[k]
412 allAvailable[k] += available[k]
413 }
414 allClosed += closed
415 g.line("%s closed %d",s,closed)
416 }
417 s,used,available := "TOTAL",0,0
418 for k:=0; k<4; k++ {
419 s += fmt.Sprintf(" %s %d/%d",kinds[k],allUsed[k],allAvailable[k])
420 used += allUsed[k]
421 available += allAvailable[k]
422 }
423 q := 0
424 if available > 0 { q = used*1000/available }
425 g.line("%s closed %d rate %s%%",s,allClosed,tenth(q))
426 case "SIGN":
427 freeCounts,buses := [4]int{},0
428 for _,level := range g.levels {
429 run := 0
430 for _,id := range level {
431 if g.free(id) { freeCounts[g.spots[id].kind]++ }
432 if g.free(id) && g.spots[id].kind == 2 { run++ } else {
433 buses += run/3
434 run = 0
435 }
436 }
437 buses += run/3
438 }
439 s := "SIGN"
440 for k:=0; k<5; k++ {
441 count := buses
442 if k != 4 {
443 count = 0
444 for _,kind := range preferences[k] { count += freeCounts[kind] }
445 }
446 value := "FULL"
447 if count > 0 { value = strconv.Itoa(count) }
448 s += " "+types[k]+" "+value
449 }
450 g.line("%s",s)
451 case "RATES":
452 g.line("RATES")
453 for k:=0; k<5; k++ { g.line("%s %d %d",types[k],g.rates[k],g.caps[k]) }
454 g.line("NIGHT %d LOST %d KWH %d",g.fees[0],g.fees[1],g.fees[2])
455 case "PASSES":
456 plates := []string{}
457 for p,last := range g.passes {
458 if g.now/1440+1 <= last { plates = append(plates,p) }
459 }
460 sort.Strings(plates)
461 g.line("PASSES %d",len(plates))
462 for _,p := range plates {
463 suffix := ""
464 if _,ok := g.parked[p]; ok { suffix = " IN" }
465 g.line("%s until %d%s",p,g.passes[p],suffix)
466 }
467 case "HISTORY":
468 records := g.histories[a[1]]
469 if g.error(len(records) == 0,"no history for "+a[1]) { return }
470 total := 0
471 for _,r := range records { total += r.total }
472 g.line("HISTORY %s visits %d paid %d",a[1],len(records),total)
473 for _,r := range records {
474 suffix := ""
475 if r.lost { suffix = " LOST" }
476 g.line("%s %s %s %s %d%s",timeText(r.entry),timeText(r.exit),r.location,types[r.kind],r.total,suffix)
477 }
478 case "REVENUE":
479 g.line("REVENUE")
480 all := revenue{}
481 for k:=0; k<=5; k++ {
482 r,name := all,"TOTAL"
483 if k < 5 { r,name = g.revenue[k],types[k] }
484 g.line("%s exits %d parking %d night %d lost %d charge %d total %d",name,r.exits,r.parking,r.night,r.lost,r.charge,r.total)
485 if k < 5 {
486 all.exits += r.exits
487 all.parking += r.parking
488 all.night += r.night
489 all.lost += r.lost
490 all.charge += r.charge
491 all.total += r.total
492 }
493 }
494 case "DAILY","STATS":
495 rows := g.totals
496 if cmd == "DAILY" { rows = g.daily[number(a[1])] }
497 all := counts{}
498 for _,c := range rows {
499 all.entries += c.entries
500 all.exits += c.exits
501 all.refused += c.refused
502 all.revenue += c.revenue
503 all.stay += c.stay
504 }
505 if cmd == "DAILY" { g.line("DAILY %s entries %d exits %d refused %d revenue %d",a[1],all.entries,all.exits,all.refused,all.revenue) } else {
506 at := "-"
507 if g.peak > 0 { at = timeText(g.peakTime) }
508 g.line("STATS parked %d peak %d at %s",len(g.parked),g.peak,at)
509 }
510 limit := 6
511 if cmd == "DAILY" { limit = 5 }
512 for k:=0; k<limit; k++ {
513 c,name := all,"ALL"
514 if k < 5 { c,name = rows[k],types[k] }
515 if cmd == "DAILY" { g.line("%s entries %d exits %d refused %d revenue %d",name,c.entries,c.exits,c.refused,c.revenue) } else {
516 q := 0
517 if c.exits > 0 { q = c.stay*10/c.exits }
518 g.line("%s entries %d exits %d refused %d avg_stay %s",name,c.entries,c.exits,c.refused,tenth(q))
519 }
520 }
521 }
522}
523func main() {
524 g := garage{rates:[5]int{100,200,200,300,600},caps:[5]int{800,1600,1600,2400,4800},fees:[3]int{300,2000,30},parked:map[string]int{},passes:map[string]int{},histories:map[string][]record{},daily:map[int][5]counts{}}
525 scanner := bufio.NewScanner(os.Stdin)
526 scanner.Scan()
527 count := number(scanner.Text())
528 out := bufio.NewWriter(os.Stdout)
529 defer out.Flush()
530 for i:=0; i<count; i++ {
531 scanner.Scan()
