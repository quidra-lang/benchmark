1local TYPES = {'MOTORCYCLE','CAR','ECAR','VAN','BUS'}
2local KINDS = {'MOTO','COMPACT','LARGE','EV'}
3local PREFS = {MOTORCYCLE={'MOTO','COMPACT','LARGE'},CAR={'COMPACT','LARGE'},ECAR={'EV','COMPACT','LARGE'},VAN={'LARGE'}}
4local rates = {MOTORCYCLE={100,800},CAR={200,1600},ECAR={200,1600},VAN={300,2400},BUS={600,4800}}
5local fees = {NIGHT=300,LOST=2000,KWH=30}
6local levels,spots,vehicles,queue,passes,history,stats,daily = {},{},{},{},{},{},{},{}
7local now,peak,peak_time = 0,0,0
8local function fail(message) error(message,0) end
9local function zeros(fields)
10 local result = {}
11 for _,field in ipairs(fields) do result[field] = 0 end
12 return result
13end
14for _,kind in ipairs(TYPES) do stats[kind] = zeros({'entries','exits','refused','stay','parking','night','lost','charge','total'}) end
15local function size(map)
16 local total = 0
17 for _ in pairs(map) do total = total+1 end
18 return total
19end
20local function find(items,needle)
21 for index,value in ipairs(items) do
22 if value == needle then return index end
23 end
24 return nil
25end
26local function append(target,items)
27 table.move(items,1,#items,#target+1,target)
28end
29local function date_of(time) return time//1440+1 end
30local function time_text(time) return date_of(time) .. '-' .. string.format('%02d:%02d',(time%1440)//60,time%60) end
31local function duration(minutes) return string.format('%d:%02d',minutes//60,minutes%60) end
32local function one_decimal(q) return (q//10) .. '.' .. (q%10) end
33local function valid_pass(plate,time) return passes[plate] ~= nil and date_of(time) <= passes[plate] end
34local function get_spot(id)
35 if not spots[id] then fail('unknown spot ' .. id) end
36 return spots[id]
37end
38local function get_vehicle(plate)
39 if not vehicles[plate] then fail(plate .. ' is not parked') end
40 return vehicles[plate]
41end
42local function location(v)
43 local s = v.spots[1]
44 if v.type == 'BUS' then return 'L' .. s.level .. '-' .. s.number .. '..' .. (s.number+2) end
45 return s.id
46end
47local function location_less(a,b)
48 local x,y = a.spots[1],b.spots[1]
49 if x.level ~= y.level then return x.level < y.level end
50 return x.number < y.number
51end
52local function free_spot(s) return not s.closed and s.plate == '' end
53local function assignment(kind)
54 if kind == 'BUS' then
55 for _,level in ipairs(levels) do
56 for i = 1,#level-2 do
57 local triple,valid = {},true
58 for j = i,i+2 do
59 triple[#triple+1] = level[j]
60 if level[j].kind ~= 'LARGE' or not free_spot(level[j]) then valid = false end
61 end
62 if valid then return triple end
63 end
64 end
65 else
66 for _,preferred in ipairs(PREFS[kind]) do
67 for _,level in ipairs(levels) do
68 for _,s in ipairs(level) do
69 if s.kind == preferred and free_spot(s) then return {s} end
70 end
71 end
72 end
73 end
74 return {}
75end
76local function add_stat(kind,field,amount,date)
77 stats[kind][field] = stats[kind][field]+amount
78 if field == 'entries' or field == 'exits' or field == 'refused' or field == 'total' then
79 if not daily[date] then
80 daily[date] = {}
81 for _,k in ipairs(TYPES) do daily[date][k] = zeros({'entries','exits','refused','revenue'}) end
82 end
83 local daily_field = field == 'total' and 'revenue' or field
84 daily[date][kind][daily_field] = daily[date][kind][daily_field]+amount
85 end
86end
87local function park_vehicle(plate,kind,chosen)
88 local v = {plate=plate,type=kind,spots=chosen,entry=now,kwh=0,cost=0,validation=0,payment=nil}
89 for _,s in ipairs(chosen) do s.plate = plate end
90 vehicles[plate] = v
91 add_stat(kind,'entries',1,date_of(now))
92 if size(vehicles) > peak then peak,peak_time = size(vehicles),now end
93 return v
94end
95local function process_queue(out)
96 local remaining = {}
97 for _,waiting in ipairs(queue) do
98 local chosen = assignment(waiting.type)
99 if #chosen == 0 then
100 remaining[#remaining+1] = waiting
101 else
102 local v = park_vehicle(waiting.plate,waiting.type,chosen)
103 out[#out+1] = 'ADMIT ' .. v.plate .. ' ' .. v.type .. ' ' .. location(v) .. ' waited ' .. duration(now-waiting.time)
104 end
105 end
106 queue = remaining
107end
108local function giveups()
109 local remaining,out = {},{}
110 for _,waiting in ipairs(queue) do
111 if now-waiting.time >= 30 then
112 local time = waiting.time+30
113 out[#out+1] = 'GIVEUP ' .. waiting.plate .. ' ' .. waiting.type .. ' ' .. time_text(time)
114 add_stat(waiting.type,'refused',1,date_of(time))
115 else
116 remaining[#remaining+1] = waiting
117 end
118 end
119 queue = remaining
120 return out
121end
122local function interval_fee(v,a,b,free)
123 local total = b-a
124 if valid_pass(v.plate,b) or total <= free then return 0,0 end
125 local rate,cap = table.unpack(rates[v.type])
126 local parking,i = 0,0
127 while total > 0 do
128 local minutes = math.min(total,1440)
129 local billable = minutes-(i == 0 and free or 0)
130 parking = parking+math.min(((billable+29)//30)*rate,cap)
131 total,i = total-minutes,i+1
132 end
133 return parking,(date_of(b)-date_of(a))*fees.NIGHT
134end
135local function charges(v,lost)
136 if not v.payment then
137 local parking,night = interval_fee(v,v.entry,now,15+v.validation)
138 local loss = lost and fees.LOST or 0
139 local total = parking+night+loss+v.cost
140 return {parking,night,loss,v.cost,total,total}
141 end
142 local extra,extra_night = interval_fee(v,v.payment.time,now,15)
143 local parking,night = v.payment.parking+extra,v.payment.night+extra_night
144 return {parking,night,0,v.cost,parking+night+v.cost,extra+extra_night}
145end
146local function fee_line(label,v,f)
147 return label .. ' ' .. v.plate .. ' ' .. location(v) .. ' stay ' .. duration(now-v.entry) .. ' parking ' .. f[1] .. ' night ' .. f[2] .. ' lost ' .. f[3] .. ' charge ' .. f[4] .. ' total ' .. f[5] .. ' due ' .. f[6]
148end
149local function annotation(v)
150 local text = valid_pass(v.plate,now) and ' PASS' or ''
151 if v.validation ~= 0 then text = text .. ' VAL ' .. v.validation end
152 if v.kwh ~= 0 then text = text .. ' KWH ' .. v.kwh end
153 if v.payment then text = text .. ' PAID' end
154 return text
155end
156local function move_vehicle(v,spot)
157 local old = location(v)
158 v.spots[1].plate = ''
159 spot.plate = v.plate
160 v.spots = {spot}
161 return 'MOVE ' .. v.plate .. ' ' .. old .. ' -> ' .. spot.id
162end
163local function occupancy_line(label,used,available,closed_count)
164 local parts = {label}
165 for _,kind in ipairs(KINDS) do append(parts,{kind,used[kind] .. '/' .. available[kind]}) end
166 append(parts,{'closed',closed_count})
167 return table.concat(parts,' ')
168end
169local function execute(t)
170 local cmd,out = t[1],{}
171 local function emit(text) out[#out+1] = text end
172 if cmd == 'LEVEL' then
173 local level_no,level = #levels+1,{}
174 for index,kind in ipairs(KINDS) do
175 for _ = 1,tonumber(t[index+1]) do
176 local number = #level+1
177 local id = 'L' .. level_no .. '-' .. number
178 local s = {id=id,level=level_no,number=number,kind=kind,closed=false,plate=''}
179 spots[id] = s
180 level[#level+1] = s
181 end
182 end
183 levels[#levels+1] = level
184 emit('LEVEL ' .. level_no .. ' spots ' .. #level .. ' MOTO ' .. t[2] .. ' COMPACT ' .. t[3] .. ' LARGE ' .. t[4] .. ' EV ' .. t[5])
185 elseif cmd == 'CLOSE' or cmd == 'OPEN' then
186 local s = get_spot(t[2])
187 if cmd == 'CLOSE' then
188 if s.closed then fail('spot ' .. t[2] .. ' is already closed') end
189 if s.plate ~= '' then fail('spot ' .. t[2] .. ' is occupied by ' .. s.plate) end
190 s.closed = true
191 else
192 if not s.closed then fail('spot ' .. t[2] .. ' is not closed') end
193 s.closed = false
194 end
195 emit(cmd .. ' ' .. s.id .. ' ' .. s.kind)
196 elseif cmd == 'RATE' then
197 if not rates[t[2]] then fail('unknown type ' .. t[2]) end
198 if tonumber(t[4]) < tonumber(t[3]) then fail('cap below rate for ' .. t[2]) end
199 rates[t[2]] = {tonumber(t[3]),tonumber(t[4])}
200 emit('RATE ' .. t[2] .. ' ' .. t[3] .. ' ' .. t[4])
201 elseif cmd == 'FEE' then
202 if fees[t[2]] == nil then fail('unknown fee ' .. t[2]) end
203 fees[t[2]] = tonumber(t[3])
204 emit('FEE ' .. t[2] .. ' ' .. t[3])
205 elseif cmd == 'ENTER' then
206 local plate,kind = t[2],t[3]
207 if not rates[kind] then fail('unknown type ' .. kind) end
208 if vehicles[plate] then fail(plate .. ' is already parked at ' .. location(vehicles[plate])) end
209 for _,w in ipairs(queue) do
210 if w.plate == plate then fail(plate .. ' is already waiting') end
211 end
212 local chosen = assignment(kind)
213 if #chosen > 0 then
214 local v = park_vehicle(plate,kind,chosen)
215 emit('ENTER ' .. plate .. ' ' .. kind .. ' ' .. location(v))
216 elseif #queue < 3 then
217 queue[#queue+1] = {plate=plate,type=kind,time=now}
218 emit('WAIT ' .. plate .. ' ' .. kind .. ' ' .. #queue)
219 else
220 add_stat(kind,'refused',1,date_of(now))
221 emit('FULL ' .. plate .. ' ' .. kind)
222 end
223 elseif cmd == 'LEAVE' then
224 local index
225 for i,w in ipairs(queue) do
226 if w.plate == t[2] then index = i end
227 end
228 if not index then fail(t[2] .. ' is not waiting') end
229 local w = table.remove(queue,index)
230 emit('LEAVE ' .. w.plate .. ' ' .. w.type)
231 elseif cmd == 'EXIT' or cmd == 'PAY' or cmd == 'QUOTE' or cmd == 'CHARGE' or cmd == 'VALIDATE' or cmd == 'MOVE' then
232 local v = get_vehicle(t[2])
233 if cmd == 'EXIT' then
234 local lost = #t == 3
235 if lost and v.payment then fail(v.plate .. ' has already paid') end
236 local f = charges(v,lost)
237 emit(fee_line('EXIT',v,f))
238 local record = {entry=v.entry,exit=now,location=location(v),type=v.type,total=f[5],lost=lost}
239 if not history[v.plate] then history[v.plate] = {} end
240 local records = history[v.plate]
241 records[#records+1] = record
242 for _,s in ipairs(v.spots) do s.plate = '' end
243 vehicles[v.plate] = nil
244 add_stat(v.type,'exits',1,date_of(now))
245 add_stat(v.type,'stay',now-v.entry,date_of(now))
246 for index,field in ipairs({'parking','night','lost','charge','total'}) do add_stat(v.type,field,f[index],date_of(now)) end
247 elseif cmd == 'PAY' then
248 if v.payment then fail(v.plate .. ' has already paid') end
249 local f = charges(v)
250 v.payment = {time=now,parking=f[1],night=f[2]}
251 emit('PAY ' .. v.plate .. ' parking ' .. f[1] .. ' night ' .. f[2] .. ' charge ' .. f[4] .. ' total ' .. f[5])
252 elseif cmd == 'QUOTE' then
253 emit(fee_line('QUOTE',v,charges(v)))
254 elseif cmd == 'CHARGE' then
255 if v.type ~= 'ECAR' then fail(v.plate .. ' is not an electric car') end
256 if v.spots[1].kind ~= 'EV' then fail(v.plate .. ' is not at a charger') end
257 if v.payment then fail(v.plate .. ' has already paid') end
258 local kwh = tonumber(t[3])
259 if v.kwh+kwh > 100 then fail('charge limit exceeded for ' .. v.plate) end
260 local cost = kwh*fees.KWH
261 v.kwh,v.cost = v.kwh+kwh,v.cost+cost
262 emit('CHARGE ' .. v.plate .. ' kwh ' .. kwh .. ' cost ' .. cost .. ' stay_kwh ' .. v.kwh .. ' stay_cost ' .. v.cost)
263 elseif cmd == 'VALIDATE' then
264 if v.payment then fail(v.plate .. ' has already paid') end
265 local minutes = tonumber(t[3])
266 if v.validation+minutes > 180 then fail('validation limit exceeded for ' .. v.plate) end
267 v.validation = v.validation+minutes
268 emit('VALIDATE ' .. v.plate .. ' ' .. minutes .. ' total ' .. v.validation)
269 else
270 if v.type == 'BUS' then fail('bus ' .. v.plate .. ' cannot be moved') end
271 local s = get_spot(t[3])
272 if v.spots[1].id == s.id then fail(v.plate .. ' is already at ' .. s.id) end
273 if s.closed then fail('spot ' .. s.id .. ' is closed') end
274 if s.plate ~= '' then fail('spot ' .. s.id .. ' is occupied by ' .. s.plate) end
275 if not find(PREFS[v.type],s.kind) then fail(v.type .. ' cannot use ' .. s.kind .. ' spot ' .. s.id) end
276 emit(move_vehicle(v,s))
277 end
278 elseif cmd == 'PASS' then
279 local plate,days = t[2],tonumber(t[3])
280 passes[plate] = valid_pass(plate,now) and passes[plate]+days or date_of(now)+days-1
281 emit('PASS ' .. plate .. ' until ' .. passes[plate])
282 elseif cmd == 'UNPASS' then
283 if not valid_pass(t[2],now) then fail(t[2] .. ' has no pass') end
284 passes[t[2]] = nil
285 emit('UNPASS ' .. t[2])
286 elseif cmd == 'REBALANCE' then
287 local targets = {}
288 for _,v in pairs(vehicles) do
289 if v.type ~= 'BUS' then targets[#targets+1] = v end
290 end
291 table.sort(targets,location_less)
292 local moved = 0
293 for _,v in ipairs(targets) do
294 local current = find(PREFS[v.type],v.spots[1].kind)
295 if current ~= 1 then
296 local chosen = assignment(v.type)
297 if #chosen > 0 and find(PREFS[v.type],chosen[1].kind) < current then
298 emit(move_vehicle(v,chosen[1]))
299 moved = moved+1
300 end
301 end
302 end
303 emit('REBALANCE ' .. moved)
304 elseif cmd == 'WHERE' then
305 local plate = t[2]
306 if vehicles[plate] then
307 local v = vehicles[plate]
308 emit('WHERE ' .. plate .. ' ' .. location(v) .. ' ' .. v.type .. ' ' .. time_text(v.entry) .. ' stay ' .. duration(now-v.entry) .. annotation(v))
309 else
310 local position
311 for index,w in ipairs(queue) do
312 if w.plate == plate then position = index end
313 end
314 if position then
315 emit('WHERE ' .. plate .. ' waiting ' .. position)
316 elseif history[plate] then
317 local records = history[plate]
318 emit('WHERE ' .. plate .. ' gone ' .. time_text(records[#records].exit))
319 else
320 fail('unknown plate ' .. plate)
321 end
322 end
323 elseif cmd == 'LIST' or cmd == 'OVERSTAY' then
324 if cmd == 'LIST' and t[2] and not levels[tonumber(t[2])] then fail('unknown level ' .. t[2]) end
325 local targets = {}
326 for _,v in pairs(vehicles) do
327 local include = true
328 if cmd == 'LIST' and t[2] then include = v.spots[1].level == tonumber(t[2]) end
329 if cmd == 'OVERSTAY' then include = now-v.entry >= tonumber(t[2])*60 end
330 if include then targets[#targets+1] = v end
331 end
332 table.sort(targets,function(a,b)
333 if cmd == 'OVERSTAY' and a.entry ~= b.entry then return a.entry < b.entry end
334 return location_less(a,b)
335 end)
336 local header = cmd == 'OVERSTAY' and cmd .. ' ' .. t[2] or cmd
337 emit(header .. ' ' .. #targets)
338 for _,v in ipairs(targets) do
339 local text = location(v) .. ' ' .. v.plate .. ' ' .. v.type .. ' ' .. time_text(v.entry)
340 if cmd == 'LIST' then text = text .. annotation(v) else text = text .. ' stay ' .. duration(now-v.entry) end
341 emit(text)
342 end
343 elseif cmd == 'MAP' then
344 emit('MAP ' .. #levels)
345 for index,level in ipairs(levels) do
346 local symbols = ''
347 for _,s in ipairs(level) do
348 local char
349 if s.closed then char = '#'
350 elseif s.plate ~= '' then char = s.kind:sub(1,1)
351 else char = s.kind:sub(1,1):lower() end
352 symbols = symbols .. char
353 end
354 local chunks = {}
355 for i = 1,#symbols,10 do chunks[#chunks+1] = symbols:sub(i,i+9) end
356 emit('L' .. index .. ' ' .. table.concat(chunks,' '))
357 end
358 elseif cmd == 'OCCUPANCY' then
359 emit('OCCUPANCY vehicles ' .. size(vehicles) .. ' waiting ' .. #queue)
360 local total_used,total_available,total_closed = zeros(KINDS),zeros(KINDS),0
361 for index,level in ipairs(levels) do
362 local used,available,closed_count = zeros(KINDS),zeros(KINDS),0
363 for _,s in ipairs(level) do
364 if s.closed then closed_count = closed_count+1 else available[s.kind] = available[s.kind]+1 end
365 if s.plate ~= '' then used[s.kind] = used[s.kind]+1 end
366 end
367 emit(occupancy_line('L' .. index,used,available,closed_count))
368 for _,kind in ipairs(KINDS) do
369 total_used[kind] = total_used[kind]+used[kind]
370 total_available[kind] = total_available[kind]+available[kind]
371 end
372 total_closed = total_closed+closed_count
373 end
374 local used,capacity = 0,0
375 for _,kind in ipairs(KINDS) do used,capacity = used+total_used[kind],capacity+total_available[kind] end
376 local usage = capacity == 0 and 0 or (used*1000)//capacity
377 emit(occupancy_line('TOTAL',total_used,total_available,total_closed) .. ' rate ' .. one_decimal(usage) .. '%')
378 elseif cmd == 'SIGN' then
379 local parts = {'SIGN'}
380 for _,kind in ipairs(TYPES) do
381 local count = 0
382 for _,level in ipairs(levels) do
383 local run = 0
384 for _,s in ipairs(level) do
385 if kind == 'BUS' then
386 if s.kind == 'LARGE' and free_spot(s) then run = run+1
387 else count,run = count+run//3,0 end
388 elseif find(PREFS[kind],s.kind) and free_spot(s) then
389 count = count+1
390 end
391 end
392 if kind == 'BUS' then count = count+run//3 end
393 end
394 append(parts,{kind,count == 0 and 'FULL' or count})
395 end
396 emit(table.concat(parts,' '))
397 elseif cmd == 'RATES' then
398 emit('RATES')
399 for _,kind in ipairs(TYPES) do emit(kind .. ' ' .. rates[kind][1] .. ' ' .. rates[kind][2]) end
400 emit('NIGHT ' .. fees.NIGHT .. ' LOST ' .. fees.LOST .. ' KWH ' .. fees.KWH)
401 elseif cmd == 'PASSES' then
402 local plates = {}
403 for plate in pairs(passes) do
404 if valid_pass(plate,now) then plates[#plates+1] = plate end
405 end
406 table.sort(plates)
407 emit('PASSES ' .. #plates)
408 for _,plate in ipairs(plates) do emit(plate .. ' until ' .. passes[plate] .. (vehicles[plate] and ' IN' or '')) end
409 elseif cmd == 'HISTORY' then
410 local records = history[t[2]]
411 if not records then fail('no history for ' .. t[2]) end
412 local paid = 0
413 for _,r in ipairs(records) do paid = paid+r.total end
414 emit('HISTORY ' .. t[2] .. ' visits ' .. #records .. ' paid ' .. paid)
415 for _,r in ipairs(records) do emit(time_text(r.entry) .. ' ' .. time_text(r.exit) .. ' ' .. r.location .. ' ' .. r.type .. ' ' .. r.total .. (r.lost and ' LOST' or '')) end
416 elseif cmd == 'REVENUE' then
417 emit('REVENUE')
418 local total = zeros({'exits','parking','night','lost','charge','total'})
419 local labels = {table.unpack(TYPES)}
420 labels[#labels+1] = 'TOTAL'
421 for _,kind in ipairs(labels) do
422 local s = kind == 'TOTAL' and total or stats[kind]
423 if kind ~= 'TOTAL' then
424 for key in pairs(total) do total[key] = total[key]+s[key] end
425 end
426 emit(kind .. ' exits ' .. s.exits .. ' parking ' .. s.parking .. ' night ' .. s.night .. ' lost ' .. s.lost .. ' charge ' .. s.charge .. ' total ' .. s.total)
427 end
428 elseif cmd == 'DAILY' then
429 local date = tonumber(t[2])
430 local rows = daily[date] or {}
431 local total,lines = zeros({'entries','exits','refused','revenue'}),{}
432 for _,kind in ipairs(TYPES) do
433 local s = rows[kind] or zeros({'entries','exits','refused','revenue'})
434 for key in pairs(total) do total[key] = total[key]+s[key] end
435 lines[#lines+1] = kind .. ' entries ' .. s.entries .. ' exits ' .. s.exits .. ' refused ' .. s.refused .. ' revenue ' .. s.revenue
436 end
437 emit('DAILY ' .. date .. ' entries ' .. total.entries .. ' exits ' .. total.exits .. ' refused ' .. total.refused .. ' revenue ' .. total.revenue)
438 append(out,lines)
439 elseif cmd == 'STATS' then
440 emit('STATS parked ' .. size(vehicles) .. ' peak ' .. peak .. ' at ' .. (peak == 0 and '-' or time_text(peak_time)))
441 local total,labels = zeros({'entries','exits','refused','stay'}),{table.unpack(TYPES)}
442 labels[#labels+1] = 'ALL'
443 for _,kind in ipairs(labels) do
444 local s = kind == 'ALL' and total or stats[kind]
445 if kind ~= 'ALL' then
446 for key in pairs(total) do total[key] = total[key]+s[key] end
447 end
448 local average = s.exits == 0 and 0 or (s.stay*10)//s.exits
449 emit(kind .. ' entries ' .. s.entries .. ' exits ' .. s.exits .. ' refused ' .. s.refused .. ' avg_stay ' .. one_decimal(average))
450 end
451 end
452 if cmd == 'LEVEL' or cmd == 'OPEN' or cmd == 'EXIT' or cmd == 'MOVE' or cmd == 'REBALANCE' then process_queue(out) end
453 return out
454end
455local n = tonumber(io.read('*l'))
456for _ = 1,n do
457 local tokens = {}
458 for word in io.read('*l'):gmatch('%S+') do tokens[#tokens+1] = word end
459 local hour,minute = tokens[2]:match('^(%d%d):(%d%d)$')
460 now = (tonumber(tokens[1])-1)*1440+tonumber(hour)*60+tonumber(minute)
461 for _,line in ipairs(giveups()) do io.write(line,'\n') end
462 local command = {table.unpack(tokens,3)}
463 local ok,out = pcall(execute,command)
464 if ok then
465 for _,line in ipairs(out) do io.write(line,'\n') end
466 else
467 io.write('ERROR ',out,'\n')
468 end
469end
