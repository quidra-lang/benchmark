1<?php
2const TYPES = ['MOTORCYCLE','CAR','ECAR','VAN','BUS'];
3const KINDS = ['MOTO','COMPACT','LARGE','EV'];
4const PREFS = ['MOTORCYCLE'=>['MOTO','COMPACT','LARGE'],'CAR'=>['COMPACT','LARGE'],'ECAR'=>['EV','COMPACT','LARGE'],'VAN'=>['LARGE']];
5$rates = ['MOTORCYCLE'=>[100,800],'CAR'=>[200,1600],'ECAR'=>[200,1600],'VAN'=>[300,2400],'BUS'=>[600,4800]];
6$fees = ['NIGHT'=>300,'LOST'=>2000,'KWH'=>30];
7$levels = [];
8$spots = [];
9$vehicles = [];
10$queue = [];
11$passes = [];
12$history = [];
13$stats = array_fill_keys(TYPES,array_fill_keys(['entries','exits','refused','stay','parking','night','lost','charge','total'],0));
14$daily = [];
15$now = 0;

18function fail(string $message): never { throw new InvalidArgumentException($message); }
19function date_of(int $time): int { return intdiv($time,1440)+1; }
20function time_text(int $time): string { return date_of($time) . '-' . sprintf('%02d:%02d',intdiv($time%1440,60),$time%60); }
21function duration(int $minutes): string { return intdiv($minutes,60) . ':' . sprintf('%02d',$minutes%60); }
22function one_decimal(int $q): string { return intdiv($q,10) . '.' . $q%10; }
23function valid_pass(string $plate,int $time): bool
24{
25 global $passes;
26 return isset($passes[$plate]) && date_of($time)<=$passes[$plate];
27}
28function get_spot(string $id): object
29{
30 global $spots;
31 if (!isset($spots[$id])) fail("unknown spot $id");
32 return $spots[$id];
33}
34function get_vehicle(string $plate): object
35{
36 global $vehicles;
37 if (!isset($vehicles[$plate])) fail("$plate is not parked");
38 return $vehicles[$plate];
39}
40function location(object $v): string
41{
42 $s = $v->spots[0];
43 return $v->type==='BUS' ? "L$s->level-$s->number.." . ($s->number+2) : $s->id;
44}
45function location_key(object $v): array { return [$v->spots[0]->level,$v->spots[0]->number]; }
46function free_spot(object $s): bool { return !$s->closed && $s->plate===''; }
47function assignment(string $type): array
48{
49 global $levels;
50 if ($type==='BUS') {
51 foreach ($levels as $level) {
52 for ($i=0; $i+2<count($level); ++$i) {
53 $triple = array_slice($level,$i,3);
54 if (array_all($triple,fn($s)=>$s->kind==='LARGE' && free_spot($s))) return $triple;
55 }
56 }
57 } else {
58 foreach (PREFS[$type] as $kind) {
59 foreach ($levels as $level) {
60 foreach ($level as $s) if ($s->kind===$kind && free_spot($s)) return [$s];
61 }
62 }
63 }
64 return [];
65}
66function add_stat(string $type,string $field,int $amount,int $date): void
67{
68 global $stats,$daily;
69 $stats[$type][$field] += $amount;
70 if (in_array($field,['entries','exits','refused','total'],true)) {
71 if (!isset($daily[$date])) {
72 $daily[$date] = [];
73 foreach (TYPES as $k) $daily[$date][$k] = array_fill_keys(['entries','exits','refused','revenue'],0);
74 }
75 $daily[$date][$type][$field==='total' ? 'revenue' : $field] += $amount;
76 }
77}
78function park_vehicle(string $plate,string $type,array $chosen): object
79{
80 global $now,$vehicles,$peak,$peakTime;
81 $v = (object)['plate'=>$plate,'type'=>$type,'spots'=>$chosen,'entry'=>$now,'kwh'=>0,'cost'=>0,'validation'=>0,'payment'=>null];
82 foreach ($chosen as $s) $s->plate = $plate;
83 $vehicles[$plate] = $v;
84 add_stat($type,'entries',1,date_of($now));
85 if (count($vehicles)>$peak) {
86 $peak = count($vehicles);
87 $peakTime = $now;
88 }
89 return $v;
90}
91function process_queue(array &$out): void
92{
93 global $queue,$now;
94 $remaining = [];
95 foreach ($queue as $waiting) {
96 $chosen = assignment($waiting->type);
97 if ($chosen===[]) {
98 $remaining[] = $waiting;
99 } else {
100 $v = park_vehicle($waiting->plate,$waiting->type,$chosen);
101 $out[] = "ADMIT $v->plate $v->type " . location($v) . ' waited ' . duration($now-$waiting->time);
102 }
103 }
104 $queue = $remaining;
105}
106function giveups(): array
107{
108 global $queue,$now;
109 $remaining = [];
110 $out = [];
111 foreach ($queue as $waiting) {
112 if ($now-$waiting->time>=30) {
113 $time = $waiting->time+30;
114 $out[] = "GIVEUP $waiting->plate $waiting->type " . time_text($time);
115 add_stat($waiting->type,'refused',1,date_of($time));
116 } else {
117 $remaining[] = $waiting;
118 }
119 }
120 $queue = $remaining;
121 return $out;
122}
123function interval_fee(object $v,int $a,int $b,int $free): array
124{
125 global $rates,$fees;
126 $total = $b-$a;
127 if (valid_pass($v->plate,$b) || $total<=$free) return [0,0];
128 [$rate,$cap] = $rates[$v->type];
129 $parking = 0;
130 for ($i=0; $total>0; ++$i) {
131 $minutes = min($total,1440);
132 $billable = $minutes-($i===0 ? $free : 0);
133 $parking += min(intdiv($billable+29,30)*$rate,$cap);
134 $total -= $minutes;
135 }
136 return [$parking,(date_of($b)-date_of($a))*$fees['NIGHT']];
137}
138function charges(object $v,bool $lost=false): array
139{
140 global $now,$fees;
141 if ($v->payment===null) {
142 [$parking,$night] = interval_fee($v,$v->entry,$now,15+$v->validation);
143 $loss = $lost ? $fees['LOST'] : 0;
144 $total = $parking+$night+$loss+$v->cost;
145 return [$parking,$night,$loss,$v->cost,$total,$total];
146 }
147 [$extra,$extraNight] = interval_fee($v,$v->payment->time,$now,15);
148 $parking = $v->payment->parking+$extra;
149 $night = $v->payment->night+$extraNight;
150 return [$parking,$night,0,$v->cost,$parking+$night+$v->cost,$extra+$extraNight];
151}
152function fee_line(string $label,object $v,array $f): string
153{
154 global $now;
155 return "$label $v->plate " . location($v) . ' stay ' . duration($now-$v->entry) . " parking $f[0] night $f[1] lost $f[2] charge $f[3] total $f[4] due $f[5]";
156}
157function annotation(object $v): string
158{
159 global $now;
160 $text = valid_pass($v->plate,$now) ? ' PASS' : '';
161 if ($v->validation!==0) $text .= " VAL $v->validation";
162 if ($v->kwh!==0) $text .= " KWH $v->kwh";
163 if ($v->payment!==null) $text .= ' PAID';
164 return $text;
165}
166function move_vehicle(object $v,object $spot): string
167{
168 $old = location($v);
169 $v->spots[0]->plate = '';
170 $spot->plate = $v->plate;
171 $v->spots = [$spot];
172 return "MOVE $v->plate $old -> $spot->id";
173}
174function occupancy_line(string $label,array $used,array $available,int $closed): string
175{
176 $parts = [$label];
177 foreach (KINDS as $kind) $parts = [...$parts,$kind,"$used[$kind]/$available[$kind]"];
178 return implode(' ',$parts) . " closed $closed";
179}
180function execute(array $t): array
181{
182 global $now,$levels,$spots,$vehicles,$queue,$rates,$fees,$passes,$history,$stats,$daily,$peak,$peakTime;
183 $cmd = $t[0];
184 $out = [];
185 if ($cmd==='LEVEL') {
186 $levelNo = count($levels)+1;
187 $level = [];
188 foreach (KINDS as $index=>$kind) {
189 for ($i=0; $i<(int)$t[$index+1]; ++$i) {
190 $number = count($level)+1;
191 $id = "L$levelNo-$number";
192 $s = (object)['id'=>$id,'level'=>$levelNo,'number'=>$number,'kind'=>$kind,'closed'=>false,'plate'=>''];
193 $spots[$id] = $s;
194 $level[] = $s;
195 }
196 }
197 $levels[] = $level;
198 $out[] = "LEVEL $levelNo spots " . count($level) . " MOTO $t[1] COMPACT $t[2] LARGE $t[3] EV $t[4]";
199 } elseif ($cmd==='CLOSE' || $cmd==='OPEN') {
200 $s = get_spot($t[1]);
201 if ($cmd==='CLOSE') {
202 if ($s->closed) fail("spot $t[1] is already closed");
203 if ($s->plate!=='') fail("spot $t[1] is occupied by $s->plate");
204 $s->closed = true;
205 } else {
206 if (!$s->closed) fail("spot $t[1] is not closed");
207 $s->closed = false;
208 }
209 $out[] = "$cmd $s->id $s->kind";
210 } elseif ($cmd==='RATE') {
211 if (!isset($rates[$t[1]])) fail("unknown type $t[1]");
212 if ((int)$t[3]<(int)$t[2]) fail("cap below rate for $t[1]");
213 $rates[$t[1]] = [(int)$t[2],(int)$t[3]];
214 $out[] = "RATE $t[1] $t[2] $t[3]";
215 } elseif ($cmd==='FEE') {
216 if (!isset($fees[$t[1]])) fail("unknown fee $t[1]");
217 $fees[$t[1]] = (int)$t[2];
218 $out[] = "FEE $t[1] $t[2]";
219 } elseif ($cmd==='ENTER') {
220 [$plate,$type] = array_slice($t,1,2);
221 if (!isset($rates[$type])) fail("unknown type $type");
222 if (isset($vehicles[$plate])) fail("$plate is already parked at " . location($vehicles[$plate]));
223 if (array_any($queue,fn($w)=>$w->plate===$plate)) fail("$plate is already waiting");
224 $chosen = assignment($type);
225 if ($chosen!==[]) {
226 $v = park_vehicle($plate,$type,$chosen);
227 $out[] = "ENTER $plate $type " . location($v);
228 } elseif (count($queue)<3) {
229 $queue[] = (object)['plate'=>$plate,'type'=>$type,'time'=>$now];
230 $out[] = "WAIT $plate $type " . count($queue);
231 } else {
232 add_stat($type,'refused',1,date_of($now));
233 $out[] = "FULL $plate $type";
234 }
235 } elseif ($cmd==='LEAVE') {
236 $found = array_find_key($queue,fn($w)=>$w->plate===$t[1]);
237 if ($found===null) fail("$t[1] is not waiting");
238 $w = $queue[$found];
239 array_splice($queue,$found,1);
240 $out[] = "LEAVE $w->plate $w->type";
241 } elseif (in_array($cmd,['EXIT','PAY','QUOTE','CHARGE','VALIDATE','MOVE'],true)) {
242 $v = get_vehicle($t[1]);
243 if ($cmd==='EXIT') {
244 $lost = count($t)===3;
245 if ($lost && $v->payment!==null) fail("$v->plate has already paid");
246 $f = charges($v,$lost);
247 $out[] = fee_line('EXIT',$v,$f);
248 $record = (object)['entry'=>$v->entry,'exit'=>$now,'location'=>location($v),'type'=>$v->type,'total'=>$f[4],'lost'=>$lost];
249 $history[$v->plate][] = $record;
250 foreach ($v->spots as $s) $s->plate = '';
251 unset($vehicles[$v->plate]);
252 add_stat($v->type,'exits',1,date_of($now));
253 add_stat($v->type,'stay',$now-$v->entry,date_of($now));
254 foreach (['parking','night','lost','charge','total'] as $index=>$field) add_stat($v->type,$field,$f[$index],date_of($now));
255 } elseif ($cmd==='PAY') {
256 if ($v->payment!==null) fail("$v->plate has already paid");
257 $f = charges($v);
258 $v->payment = (object)['time'=>$now,'parking'=>$f[0],'night'=>$f[1]];
259 $out[] = "PAY $v->plate parking $f[0] night $f[1] charge $f[3] total $f[4]";
260 } elseif ($cmd==='QUOTE') {
261 $out[] = fee_line('QUOTE',$v,charges($v));
262 } elseif ($cmd==='CHARGE') {
263 if ($v->type!=='ECAR') fail("$v->plate is not an electric car");
264 if ($v->spots[0]->kind!=='EV') fail("$v->plate is not at a charger");
265 if ($v->payment!==null) fail("$v->plate has already paid");
266 $kwh = (int)$t[2];
267 if ($v->kwh+$kwh>100) fail("charge limit exceeded for $v->plate");
268 $cost = $kwh*$fees['KWH'];
269 $v->kwh += $kwh;
270 $v->cost += $cost;
271 $out[] = "CHARGE $v->plate kwh $kwh cost $cost stay_kwh $v->kwh stay_cost $v->cost";
272 } elseif ($cmd==='VALIDATE') {
273 if ($v->payment!==null) fail("$v->plate has already paid");
274 $minutes = (int)$t[2];
275 if ($v->validation+$minutes>180) fail("validation limit exceeded for $v->plate");
276 $v->validation += $minutes;
277 $out[] = "VALIDATE $v->plate $minutes total $v->validation";
278 } else {
279 if ($v->type==='BUS') fail("bus $v->plate cannot be moved");
280 $s = get_spot($t[2]);
281 if ($v->spots[0]->id===$s->id) fail("$v->plate is already at $s->id");
282 if ($s->closed) fail("spot $s->id is closed");
283 if ($s->plate!=='') fail("spot $s->id is occupied by $s->plate");
284 if (!in_array($s->kind,PREFS[$v->type],true)) fail("$v->type cannot use $s->kind spot $s->id");
285 $out[] = move_vehicle($v,$s);
286 }
287 } elseif ($cmd==='PASS') {
288 $plate = $t[1];
289 $passes[$plate] = valid_pass($plate,$now) ? $passes[$plate]+(int)$t[2] : date_of($now)+(int)$t[2]-1;
290 $out[] = "PASS $plate until $passes[$plate]";
291 } elseif ($cmd==='UNPASS') {
292 if (!valid_pass($t[1],$now)) fail("$t[1] has no pass");
293 unset($passes[$t[1]]);
294 $out[] = "UNPASS $t[1]";
295 } elseif ($cmd==='REBALANCE') {
296 $targets = array_values(array_filter($vehicles,fn($v)=>$v->type!=='BUS'));
297 usort($targets,fn($a,$b)=>location_key($a)<=>location_key($b));
298 $moved = 0;
299 foreach ($targets as $v) {
300 $current = array_search($v->spots[0]->kind,PREFS[$v->type],true);
301 if ($current===0) continue;
302 $chosen = assignment($v->type);
303 if ($chosen!==[] && array_search($chosen[0]->kind,PREFS[$v->type],true)<$current) {
304 $out[] = move_vehicle($v,$chosen[0]);
305 ++$moved;
306 }
307 }
308 $out[] = "REBALANCE $moved";
309 } elseif ($cmd==='WHERE') {
310 $plate = $t[1];
311 if (isset($vehicles[$plate])) {
312 $v = $vehicles[$plate];
313 $out[] = "WHERE $plate " . location($v) . " $v->type " . time_text($v->entry) . ' stay ' . duration($now-$v->entry) . annotation($v);
314 } else {
315 $position = array_find_key($queue,fn($w)=>$w->plate===$plate);
316 if ($position!==null) $out[] = "WHERE $plate waiting " . ($position+1);
317 elseif (isset($history[$plate])) $out[] = "WHERE $plate gone " . time_text($history[$plate][count($history[$plate])-1]->exit);
318 else fail("unknown plate $plate");
319 }
320 } elseif ($cmd==='LIST' || $cmd==='OVERSTAY') {
321 if ($cmd==='LIST' && isset($t[1]) && !isset($levels[(int)$t[1]-1])) fail("unknown level $t[1]");
322 $targets = array_values($vehicles);
323 if ($cmd==='LIST' && isset($t[1])) $targets = array_values(array_filter($targets,fn($v)=>$v->spots[0]->level===(int)$t[1]));
324 if ($cmd==='OVERSTAY') $targets = array_values(array_filter($targets,fn($v)=>$now-$v->entry>=(int)$t[1]*60));
325 usort($targets,function($a,$b) use ($cmd) {
326 if ($cmd==='OVERSTAY' && $a->entry!==$b->entry) return $a->entry<=>$b->entry;
327 return location_key($a)<=>location_key($b);
328 });
329 $out[] = $cmd . ($cmd==='OVERSTAY' ? " $t[1]" : '') . ' ' . count($targets);
330 foreach ($targets as $v) {
331 $text = location($v) . " $v->plate $v->type " . time_text($v->entry);
332 $text .= $cmd==='LIST' ? annotation($v) : ' stay ' . duration($now-$v->entry);
333 $out[] = $text;
334 }
335 } elseif ($cmd==='MAP') {
336 $out[] = 'MAP ' . count($levels);
337 foreach ($levels as $index=>$level) {
338 $symbols = '';
339 foreach ($level as $s) $symbols .= $s->closed ? '#' : ($s->plate!=='' ? $s->kind[0] : strtolower($s->kind[0]));
340 $out[] = 'L' . ($index+1) . ' ' . implode(' ',str_split($symbols,10));
341 }
342 } elseif ($cmd==='OCCUPANCY') {
343 $out[] = 'OCCUPANCY vehicles ' . count($vehicles) . ' waiting ' . count($queue);
344 $totalUsed = array_fill_keys(KINDS,0);
345 $totalAvailable = array_fill_keys(KINDS,0);
346 $totalClosed = 0;
347 foreach ($levels as $index=>$level) {
348 $used = array_fill_keys(KINDS,0);
349 $available = array_fill_keys(KINDS,0);
350 $closedCount = 0;
351 foreach ($level as $s) {
352 if ($s->closed) ++$closedCount;
353 else ++$available[$s->kind];
354 if ($s->plate!=='') ++$used[$s->kind];
355 }
356 $out[] = occupancy_line('L' . ($index+1),$used,$available,$closedCount);
357 foreach (KINDS as $kind) {
358 $totalUsed[$kind] += $used[$kind];
359 $totalAvailable[$kind] += $available[$kind];
360 }
361 $totalClosed += $closedCount;
362 }
363 $capacity = array_sum($totalAvailable);
364 $usage = $capacity===0 ? 0 : intdiv(array_sum($totalUsed)*1000,$capacity);
365 $out[] = occupancy_line('TOTAL',$totalUsed,$totalAvailable,$totalClosed) . ' rate ' . one_decimal($usage) . '%';
366 } elseif ($cmd==='SIGN') {
367 $parts = ['SIGN'];
368 foreach (TYPES as $type) {
369 $count = 0;
370 foreach ($levels as $level) {
371 $run = 0;
372 foreach ($level as $s) {
373 if ($type==='BUS') {
374 if ($s->kind==='LARGE' && free_spot($s)) ++$run;
375 else {
376 $count += intdiv($run,3);
377 $run = 0;
378 }
379 } elseif (in_array($s->kind,PREFS[$type],true) && free_spot($s)) ++$count;
380 }
381 if ($type==='BUS') $count += intdiv($run,3);
382 }
383 $parts = [...$parts,$type,$count===0 ? 'FULL' : $count];
384 }
385 $out[] = implode(' ',$parts);
386 } elseif ($cmd==='RATES') {
387 $out[] = 'RATES';
388 foreach (TYPES as $type) $out[] = "$type " . implode(' ',$rates[$type]);
389 $out[] = "NIGHT $fees[NIGHT] LOST $fees[LOST] KWH $fees[KWH]";
390 } elseif ($cmd==='PASSES') {
391 $plates = array_values(array_filter(array_keys($passes),fn($plate)=>valid_pass((string)$plate,$now)));
392 sort($plates,SORT_STRING);
393 $out[] = 'PASSES ' . count($plates);
394 foreach ($plates as $plate) $out[] = "$plate until $passes[$plate]" . (isset($vehicles[$plate]) ? ' IN' : '');
395 } elseif ($cmd==='HISTORY') {
396 if (!isset($history[$t[1]])) fail("no history for $t[1]");
397 $records = $history[$t[1]];
398 $out[] = "HISTORY $t[1] visits " . count($records) . ' paid ' . array_sum(array_column($records,'total'));
399 foreach ($records as $r) $out[] = time_text($r->entry) . ' ' . time_text($r->exit) . " $r->location $r->type $r->total" . ($r->lost ? ' LOST' : '');
400 } elseif ($cmd==='REVENUE') {
401 $out[] = 'REVENUE';
402 $total = array_fill_keys(array_keys($stats[TYPES[0]]),0);
403 foreach (TYPES as $type) {
404 $s = $stats[$type];
405 $out[] = "$type exits $s[exits] parking $s[parking] night $s[night] lost $s[lost] charge $s[charge] total $s[total]";
406 foreach ($s as $key=>$value) $total[$key] += $value;
407 }
408 $out[] = "TOTAL exits $total[exits] parking $total[parking] night $total[night] lost $total[lost] charge $total[charge] total $total[total]";
409 } elseif ($cmd==='DAILY') {
410 $date = (int)$t[1];
411 $rows = $daily[$date] ?? array_fill_keys(TYPES,array_fill_keys(['entries','exits','refused','revenue'],0));
412 $total = array_fill_keys(['entries','exits','refused','revenue'],0);
413 $lines = [];
414 foreach (TYPES as $type) {
415 $s = $rows[$type];
416 foreach ($s as $key=>$value) $total[$key] += $value;
417 $lines[] = "$type entries $s[entries] exits $s[exits] refused $s[refused] revenue $s[revenue]";
418 }
419 $out[] = "DAILY $date entries $total[entries] exits $total[exits] refused $total[refused] revenue $total[revenue]";
420 $out = [...$out,...$lines];
421 } elseif ($cmd==='STATS') {
422 $out[] = 'STATS parked ' . count($vehicles) . " peak $peak at " . ($peak===0 ? '-' : time_text($peakTime));
423 $total = array_fill_keys(['entries','exits','refused','stay'],0);
424 foreach ([...TYPES,'ALL'] as $type) {
425 $s = $type==='ALL' ? $total : $stats[$type];
426 if ($type!=='ALL') foreach (array_keys($total) as $key) $total[$key] += $s[$key];
427 $average = $s['exits']===0 ? 0 : intdiv($s['stay']*10,$s['exits']);
428 $out[] = "$type entries $s[entries] exits $s[exits] refused $s[refused] avg_stay " . one_decimal($average);
429 }
430 }
431 if (in_array($cmd,['LEVEL','OPEN','EXIT','MOVE','REBALANCE'],true)) process_queue($out);
432 return $out;
433}

435for ($i=0; $i<$n; ++$i) {

437 [$hour,$minute] = array_map('intval',explode(':',$tokens[1]));
438 $now = ((int)$tokens[0]-1)*1440+$hour*60+$minute;
439 foreach (giveups() as $output) echo $output,"\n";
440 try {
441 foreach (execute(array_slice($tokens,2)) as $output) echo $output,"\n";
442 } catch (InvalidArgumentException $error) {
443 echo 'ERROR ',$error->getMessage(),"\n";
