2import java.util.*;
3import java.util.stream.IntStream;

5public class Main {
6 static final String[] TYPES={"MOTORCYCLE","CAR","ECAR","VAN","BUS"};
7 static final String[] KINDS={"MOTO","COMPACT","LARGE","EV"};
8 static final List<List<Integer>> PREFS=List.of(List.of(0,1,2),List.of(1,2),List.of(3,1,2),List.of(2),List.of(2));
9 static class Fault extends RuntimeException { Fault(String message) { super(message); } }
10 static void error(boolean condition, String message) { if (condition) throw new Fault(message); }
11 static String time(int n) { return String.format(Locale.ROOT,"%d-%02d:%02d",n/1440+1,n/60%24,n%60); }
12 static String duration(int n) { return String.format(Locale.ROOT,"%d:%02d",n/60,n%60); }
13 static String decimal(int n) { return n/10+"."+n%10; }
14 static class Spot {
15 int level, number, kind, order;
16 boolean closed;
17 Vehicle occupant;
18 Spot(int level, int number, int kind, int order) {
19 this.level=level;
20 this.number=number;
21 this.kind=kind;
22 this.order=order;
23 }
24 String id() { return "L"+level+"-"+number; }
25 boolean free() { return !closed && occupant==null; }
26 }
27 static class Vehicle {
28 String plate;
29 int kind, entry, energy, charge, validation, paidAt=-1, paidParking, paidNight;
30 List<Spot> spots;
31 Vehicle(String plate, int kind, int entry, List<Spot> spots) {
32 this.plate=plate;
33 this.kind=kind;
34 this.entry=entry;
35 this.spots=spots;
36 }
37 String location() { return spots.get(0).id()+(kind==4 ? ".."+spots.get(2).number : ""); }
38 }
39 record Waiting(String plate, int kind, int joined) {}
40 record Visit(int entry, int exit, String location, int kind, int total, boolean loss) {}
41 static class Garage {
42 int now, peak, peakAt;
43 List<List<Spot>> levels=new ArrayList<>();
44 List<Spot> spots=new ArrayList<>();
45 Map<String,Vehicle> parked=new HashMap<>();
46 List<Waiting> queue=new ArrayList<>();
47 TreeMap<String,Integer> passes=new TreeMap<>();
48 Map<String,List<Visit>> history=new HashMap<>();
49 int[][] rates={{100,800},{200,1600},{200,1600},{300,2400},{600,4800}};
50 int night=300, lost=2000, kwh=30;
51 int[][] stats=new int[5][9];
52 Map<Integer,int[][]> days=new HashMap<>();
53 List<String> output;
54 int[][] daily(int date) { return days.computeIfAbsent(date,d -> new int[5][4]); }
55 boolean pass(String plate) { return passes.getOrDefault(plate,0)>=now/1440+1; }
56 Vehicle vehicle(String plate) {
57 Vehicle v=parked.get(plate);
58 error(v==null,plate+" is not parked");
59 return v;
60 }
61 Spot spot(String id) {
62 String[] p=id.substring(1).split("-");
63 int l=Integer.parseInt(p[0]), n=Integer.parseInt(p[1]);
64 error(l>levels.size() || n>levels.get(l-1).size(),"unknown spot "+id);
65 return levels.get(l-1).get(n-1);
66 }
67 List<Spot> assign(int kind) {
68 if (kind==4) {
69 for (List<Spot> level:levels) {
70 for (int i=0;i+2<level.size();i++) {
71 List<Spot> triple=level.subList(i,i+3);
72 if (triple.stream().allMatch(s -> s.kind==2 && s.free())) return new ArrayList<>(triple);
73 }
74 }
75 } else {
76 for (int wanted:PREFS.get(kind)) {
77 Spot first=spots.stream().filter(s -> s.kind==wanted && s.free()).findFirst().orElse(null);
78 if (first!=null) return List.of(first);
79 }
80 }
81 return null;
82 }
83 void occupy(Vehicle v) { for (Spot s:v.spots) s.occupant=v; }
84 void free(Vehicle v) { for (Spot s:v.spots) s.occupant=null; }
85 Vehicle park(String plate, int kind, List<Spot> assigned) {
86 Vehicle v=new Vehicle(plate,kind,now,assigned);
87 parked.put(plate,v);
88 occupy(v);
89 stats[kind][0]++;
90 daily(now/1440+1)[kind][0]++;
91 if (parked.size()>peak) {
92 peak=parked.size();
93 peakAt=now;
94 }
95 return v;
96 }
97 void refuse(int kind, int at) {
98 stats[kind][2]++;
99 daily(at/1440+1)[kind][2]++;
100 }
101 void giveup() {
102 for (Waiting w:new ArrayList<>(queue)) {
103 if (now-w.joined()>=30) {
104 queue.remove(w);
105 refuse(w.kind(),w.joined()+30);
106 output.add("GIVEUP "+w.plate()+" "+TYPES[w.kind()]+" "+time(w.joined()+30));
107 }
108 }
109 }
110 void admit() {
111 for (Waiting w:new ArrayList<>(queue)) {
112 List<Spot> assigned=assign(w.kind());
113 if (assigned!=null) {
114 queue.remove(w);
115 Vehicle v=park(w.plate(),w.kind(),assigned);
116 output.add("ADMIT "+v.plate+" "+TYPES[v.kind]+" "+v.location()+" waited "+duration(now-w.joined()));
117 }
118 }
119 }
120 int[] interval(Vehicle v, int start, int free) {
121 int total=now-start;
122 if (pass(v.plate) || total<=free) return new int[]{0,0};
123 int parking=0;
124 for (int offset=0;offset<total;offset+=1440) {
125 int billable=Math.min(total-offset,1440)-(offset==0 ? free : 0);
126 parking+=Math.min((billable+29)/30*rates[v.kind][0],rates[v.kind][1]);
127 }
128 return new int[]{parking,(now/1440-start/1440)*night};
129 }
130 int[] fees(Vehicle v, boolean loss) {
131 int[] pair=interval(v,v.paidAt<0 ? v.entry : v.paidAt,v.paidAt<0 ? 15+v.validation : 15);
132 int parking=pair[0]+(v.paidAt<0 ? 0 : v.paidParking);
133 int surcharge=pair[1]+(v.paidAt<0 ? 0 : v.paidNight);
134 int ticket=loss ? lost : 0;
135 int total=parking+surcharge+ticket+v.charge;
136 return new int[]{parking,surcharge,ticket,v.charge,total,v.paidAt<0 ? total : pair[0]+pair[1]};
137 }
138 String feeLine(String label, Vehicle v, int[] f) {
139 return label+" "+v.plate+" "+v.location()+" stay "+duration(now-v.entry)+" parking "+f[0]+
140 " night "+f[1]+" lost "+f[2]+" charge "+f[3]+" total "+f[4]+" due "+f[5];
141 }
142 String annotation(Vehicle v) {
143 return (pass(v.plate) ? " PASS" : "")+(v.validation>0 ? " VAL "+v.validation : "")+
144 (v.energy>0 ? " KWH "+v.energy : "")+(v.paidAt>=0 ? " PAID" : "");
145 }
146 List<Vehicle> ordered() {
147 return parked.values().stream().sorted(Comparator.comparingInt(v -> v.spots.get(0).order)).toList();
148 }
149 void move(Vehicle v, List<Spot> target) {
150 String old=v.location();
151 free(v);
152 v.spots=target;
153 occupy(v);
154 output.add("MOVE "+v.plate+" "+old+" -> "+v.location());
155 }
156 int[] occupancy(List<Spot> target) {
157 int[] result=new int[9];
158 for (Spot s:target) {
159 if (s.closed) result[8]++;
160 else result[4+s.kind]++;
161 if (s.occupant!=null) result[s.kind]++;
162 }
163 return result;
164 }
165 String occupancyLine(String label, int[] a) {
166 String result=label;
167 for (int k=0;k<4;k++) result+=" "+KINDS[k]+" "+a[k]+"/"+a[k+4];
168 return result+" closed "+a[8];
169 }
170 String revenueLine(String label, int[] a) {
171 return label+" exits "+a[1]+" parking "+a[4]+" night "+a[5]+" lost "+a[6]+" charge "+a[7]+" total "+a[8];
172 }
173 String statsLine(String label, int[] a) {
174 return label+" entries "+a[0]+" exits "+a[1]+" refused "+a[2]+" avg_stay "+decimal(a[1]==0 ? 0 : a[3]*10/a[1]);
175 }
176 String dailyLine(String label, int[] a) { return label+" entries "+a[0]+" exits "+a[1]+" refused "+a[2]+" revenue "+a[3]; }
177 int[] sum(int[][] a, int width) {
178 return IntStream.range(0,width).map(i -> Arrays.stream(a).mapToInt(row -> row[i]).sum()).toArray();
179 }
180 void execute(String[] t) {
181 boolean trigger=false;
182 String command=t[2];
183 switch (command) {
184 case "LEVEL": {
185 List<Spot> added=new ArrayList<>();
186 int[] counts=new int[4];
187 for (int k=0;k<4;k++) {
188 counts[k]=Integer.parseInt(t[k+3]);
189 for (int i=0;i<counts[k];i++) {
190 Spot s=new Spot(levels.size()+1,added.size()+1,k,spots.size());
191 added.add(s);
192 spots.add(s);
193 }
194 }
195 levels.add(added);
196 output.add("LEVEL "+levels.size()+" spots "+added.size()+" MOTO "+counts[0]+" COMPACT "+counts[1]+" LARGE "+counts[2]+" EV "+counts[3]);
197 trigger=true;
198 break;
199 }
200 case "CLOSE", "OPEN": {
201 Spot s=spot(t[3]);
202 if (command.equals("CLOSE")) {
203 error(s.closed,"spot "+t[3]+" is already closed");
204 error(s.occupant!=null,"spot "+t[3]+" is occupied by "+(s.occupant==null ? "" : s.occupant.plate));
205 s.closed=true;
206 } else {
207 error(!s.closed,"spot "+t[3]+" is not closed");
208 s.closed=false;
209 trigger=true;
210 }
211 output.add(command+" "+s.id()+" "+KINDS[s.kind]);
212 break;
213 }
214 case "RATE": {
215 int k=Arrays.asList(TYPES).indexOf(t[3]);
216 error(k<0,"unknown type "+t[3]);
217 int rate=Integer.parseInt(t[4]), cap=Integer.parseInt(t[5]);
218 error(cap<rate,"cap below rate for "+t[3]);
219 rates[k]=new int[]{rate,cap};
220 output.add("RATE "+t[3]+" "+rate+" "+cap);
221 break;
222 }
223 case "FEE": {
224 int amount=Integer.parseInt(t[4]);
225 error(!List.of("NIGHT","LOST","KWH").contains(t[3]),"unknown fee "+t[3]);
226 switch (t[3]) {
227 case "NIGHT":
228 night=amount;
229 break;
230 case "LOST":
231 lost=amount;
232 break;
233 default: kwh=amount;
234 }
235 output.add("FEE "+t[3]+" "+amount);
236 break;
237 }
238 case "ENTER": {
239 String plate=t[3];
240 int k=Arrays.asList(TYPES).indexOf(t[4]);
241 error(k<0,"unknown type "+t[4]);
242 Vehicle existing=parked.get(plate);
243 error(existing!=null,plate+" is already parked at "+(existing==null ? "" : existing.location()));
244 error(queue.stream().anyMatch(w -> w.plate().equals(plate)),plate+" is already waiting");
245 List<Spot> assigned=assign(k);
246 if (assigned!=null) output.add("ENTER "+plate+" "+TYPES[k]+" "+park(plate,k,assigned).location());
247 else if (queue.size()<3) {
248 queue.add(new Waiting(plate,k,now));
249 output.add("WAIT "+plate+" "+TYPES[k]+" "+queue.size());
250 } else {
251 refuse(k,now);
252 output.add("FULL "+plate+" "+TYPES[k]);
253 }
254 break;
255 }
256 case "LEAVE": {
257 Waiting w=queue.stream().filter(x -> x.plate().equals(t[3])).findFirst().orElse(null);
258 error(w==null,t[3]+" is not waiting");
259 queue.remove(w);
260 output.add("LEAVE "+w.plate()+" "+TYPES[w.kind()]);
261 break;
262 }
263 case "EXIT", "PAY", "QUOTE": {
264 Vehicle v=vehicle(t[3]);
265 boolean loss=t.length==5;
266 error(v.paidAt>=0 && (command.equals("PAY") || loss),v.plate+" has already paid");
267 int[] f=fees(v,loss);
268 if (command.equals("PAY")) {
269 v.paidAt=now;
270 v.paidParking=f[0];
271 v.paidNight=f[1];
272 output.add("PAY "+v.plate+" parking "+f[0]+" night "+f[1]+" charge "+f[3]+" total "+f[4]);
273 } else {
274 output.add(feeLine(command,v,f));
275 if (command.equals("EXIT")) {
276 history.computeIfAbsent(v.plate,p -> new ArrayList<>()).add(new Visit(v.entry,now,v.location(),v.kind,f[4],loss));
277 stats[v.kind][1]++;
278 stats[v.kind][3]+=now-v.entry;
279 for (int i=0;i<5;i++) stats[v.kind][i+4]+=f[i];
280 int[] d=daily(now/1440+1)[v.kind];
281 d[1]++;
282 d[3]+=f[4];
283 free(v);
284 parked.remove(v.plate);
285 trigger=true;
286 }
287 }
288 break;
289 }
290 case "CHARGE": {
291 Vehicle v=vehicle(t[3]);
292 error(v.kind!=2,v.plate+" is not an electric car");
293 error(v.spots.get(0).kind!=3,v.plate+" is not at a charger");
294 error(v.paidAt>=0,v.plate+" has already paid");
295 int energy=Integer.parseInt(t[4]);
296 error(v.energy+energy>100,"charge limit exceeded for "+v.plate);
297 int cost=energy*kwh;
298 v.energy+=energy;
299 v.charge+=cost;
300 output.add("CHARGE "+v.plate+" kwh "+energy+" cost "+cost+" stay_kwh "+v.energy+" stay_cost "+v.charge);
301 break;
302 }
303 case "VALIDATE": {
304 Vehicle v=vehicle(t[3]);
305 error(v.paidAt>=0,v.plate+" has already paid");
306 int minutes=Integer.parseInt(t[4]);
307 error(v.validation+minutes>180,"validation limit exceeded for "+v.plate);
308 v.validation+=minutes;
309 output.add("VALIDATE "+v.plate+" "+minutes+" total "+v.validation);
310 break;
311 }
312 case "PASS": {
313 int end=(pass(t[3]) ? passes.get(t[3]) : now/1440)+Integer.parseInt(t[4]);
314 passes.put(t[3],end);
315 output.add("PASS "+t[3]+" until "+end);
316 break;
317 }
318 case "UNPASS": {
319 error(!pass(t[3]),t[3]+" has no pass");
320 passes.remove(t[3]);
321 output.add("UNPASS "+t[3]);
322 break;
323 }
324 case "MOVE": {
325 Vehicle v=vehicle(t[3]);
326 error(v.kind==4,"bus "+v.plate+" cannot be moved");
327 Spot s=spot(t[4]);
328 error(v.spots.get(0)==s,v.plate+" is already at "+t[4]);
329 error(s.closed,"spot "+t[4]+" is closed");
330 error(s.occupant!=null,"spot "+t[4]+" is occupied by "+(s.occupant==null ? "" : s.occupant.plate));
331 error(!PREFS.get(v.kind).contains(s.kind),TYPES[v.kind]+" cannot use "+KINDS[s.kind]+" spot "+t[4]);
332 move(v,List.of(s));
333 trigger=true;
334 break;
335 }
336 case "REBALANCE": {
337 int moved=0;
338 for (Vehicle v:ordered()) {
339 if (v.kind==4) continue;
340 List<Spot> target=assign(v.kind);
341 if (target!=null && PREFS.get(v.kind).indexOf(target.get(0).kind)<
342 PREFS.get(v.kind).indexOf(v.spots.get(0).kind)) {
343 move(v,target);
344 moved++;
345 }
346 }
347 output.add("REBALANCE "+moved);
348 trigger=true;
349 break;
350 }
351 case "WHERE": {
352 Vehicle v=parked.get(t[3]);
353 int p=-1;
354 for (int i=0;i<queue.size();i++) if (queue.get(i).plate().equals(t[3])) p=i;
355 List<Visit> visits=history.get(t[3]);
356 error(v==null && p<0 && visits==null,"unknown plate "+t[3]);
357 if (v!=null) output.add("WHERE "+v.plate+" "+v.location()+" "+TYPES[v.kind]+" "+time(v.entry)+" stay "+duration(now-v.entry)+annotation(v));
358 else if (p>=0) output.add("WHERE "+t[3]+" waiting "+(p+1));
359 else output.add("WHERE "+t[3]+" gone "+time(visits.get(visits.size()-1).exit()));
360 break;
361 }
362 case "LIST": {
363 int l=t.length==4 ? Integer.parseInt(t[3]) : 0;
364 error(l>levels.size(),"unknown level "+l);
365 List<Vehicle> found=ordered().stream().filter(v -> l==0 || v.spots.get(0).level==l).toList();
366 output.add("LIST "+found.size());
367 for (Vehicle v:found) output.add(v.location()+" "+v.plate+" "+TYPES[v.kind]+" "+time(v.entry)+annotation(v));
368 break;
369 }
370 case "MAP": {
371 output.add("MAP "+levels.size());
372 for (int i=0;i<levels.size();i++) {
373 List<String> chunks=new ArrayList<>();
374 StringBuilder symbols=new StringBuilder();
375 for (Spot s:levels.get(i)) symbols.append(s.closed ? '#' : s.occupant!=null ? KINDS[s.kind].charAt(0) : Character.toLowerCase(KINDS[s.kind].charAt(0)));
376 for (int j=0;j<symbols.length();j+=10) chunks.add(symbols.substring(j,Math.min(j+10,symbols.length())));
377 output.add("L"+(i+1)+" "+String.join(" ",chunks));
378 }
379 break;
380 }
381 case "OCCUPANCY": {
382 output.add("OCCUPANCY vehicles "+parked.size()+" waiting "+queue.size());
383 for (int i=0;i<levels.size();i++) output.add(occupancyLine("L"+(i+1),occupancy(levels.get(i))));
384 int[] a=occupancy(spots);
385 int used=Arrays.stream(a,0,4).sum(), available=Arrays.stream(a,4,8).sum();
386 output.add(occupancyLine("TOTAL",a)+" rate "+decimal(available==0 ? 0 : used*1000/available)+"%");
387 break;
388 }
389 case "SIGN": {
390 String line="SIGN";
391 for (int k=0;k<5;k++) {
392 int count=0;
393 if (k==4) {
394 for (List<Spot> level:levels) {
395 int run=0;
396 for (Spot s:level) {
397 if (s.kind==2 && s.free()) {
398 run++;
399 if (run==3) {
400 count++;
401 run=0;
402 }
403 } else run=0;
404 }
405 }
406 } else {
407 List<Integer> allowed=PREFS.get(k);
408 count=(int)spots.stream().filter(s -> allowed.contains(s.kind) && s.free()).count();
409 }
410 line+=" "+TYPES[k]+" "+(count==0 ? "FULL" : Integer.toString(count));
411 }
412 output.add(line);
413 break;
414 }
415 case "RATES": {
416 output.add("RATES");
417 for (int k=0;k<5;k++) output.add(TYPES[k]+" "+rates[k][0]+" "+rates[k][1]);
418 output.add("NIGHT "+night+" LOST "+lost+" KWH "+kwh);
419 break;
420 }
421 case "PASSES": {
422 List<String> found=passes.keySet().stream().filter(this::pass).toList();
423 output.add("PASSES "+found.size());
424 for (String plate:found) output.add(plate+" until "+passes.get(plate)+(parked.containsKey(plate) ? " IN" : ""));
425 break;
426 }
427 case "OVERSTAY": {
428 int hours=Integer.parseInt(t[3]);
429 List<Vehicle> found=ordered().stream().filter(v -> now-v.entry>=hours*60).sorted(Comparator.comparingInt(v -> v.entry)).toList();
430 output.add("OVERSTAY "+hours+" "+found.size());
431 for (Vehicle v:found) output.add(v.location()+" "+v.plate+" "+TYPES[v.kind]+" "+time(v.entry)+" stay "+duration(now-v.entry));
432 break;
433 }
434 case "HISTORY": {
435 List<Visit> visits=history.get(t[3]);
436 error(visits==null,"no history for "+t[3]);
437 output.add("HISTORY "+t[3]+" visits "+visits.size()+" paid "+visits.stream().mapToInt(Visit::total).sum());
438 for (Visit v:visits) output.add(time(v.entry())+" "+time(v.exit())+" "+v.location()+" "+TYPES[v.kind()]+" "+v.total()+(v.loss() ? " LOST" : ""));
439 break;
440 }
441 case "REVENUE": {
442 output.add("REVENUE");
443 for (int k=0;k<5;k++) output.add(revenueLine(TYPES[k],stats[k]));
444 output.add(revenueLine("TOTAL",sum(stats,9)));
445 break;
446 }
447 case "DAILY": {
448 int d=Integer.parseInt(t[3]);
449 int[][] a=daily(d);
450 output.add(dailyLine("DAILY "+d,sum(a,4)));
451 for (int k=0;k<5;k++) output.add(dailyLine(TYPES[k],a[k]));
452 break;
453 }
454 case "STATS": {
455 output.add("STATS parked "+parked.size()+" peak "+peak+" at "+(peak==0 ? "-" : time(peakAt)));
456 for (int k=0;k<5;k++) output.add(statsLine(TYPES[k],stats[k]));
457 output.add(statsLine("ALL",sum(stats,9)));
458 break;
459 }
460 }
461 if (trigger) admit();
462 }
463 List<String> process(String line) {
464 String[] t=line.split(" "), hm=t[1].split(":");
465 now=(Integer.parseInt(t[0])-1)*1440+Integer.parseInt(hm[0])*60+Integer.parseInt(hm[1]);
466 output=new ArrayList<>();
467 giveup();
468 try { execute(t); }
469 catch (Fault f) { output.add("ERROR "+f.getMessage()); }
470 return output;
471 }
472 }
473 public static void main(String[] args) throws IOException {
474 BufferedReader input=new BufferedReader(new InputStreamReader(System.in));
475 PrintWriter output=new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));
476 Garage garage=new Garage();
477 int n=Integer.parseInt(input.readLine());
478 for (int i=0;i<n;i++) {
479 for (String line:garage.process(input.readLine())) output.println(line);
480 }
481 output.flush();
482 }
