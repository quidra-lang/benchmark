38static char *format(const char *pattern, ...) {
39 va_list args, measure;
40 va_start(args,pattern);
41 va_copy(measure,args);
42 int length = vsnprintf(NULL,0,pattern,measure);
43 va_end(measure);
44 if (length < 0) exit(EXIT_FAILURE);
45 Temp *t = malloc(sizeof(*t)+(size_t)length+1);
46 if (!t) exit(EXIT_FAILURE);
47 t->next = temps;
48 temps = t;
49 vsnprintf(t->data,(size_t)length+1,pattern,args);
50 va_end(args);
51 return t->data;
52}
53static void clear_temps(void) {
54 while (temps) {
55 Temp *next = temps->next;
56 free(temps);
57 temps = next;
58 }
59}

71static char *time_text(int t) { return format("%d-%02d:%02d",t/1440+1,t%1440/60,t%60); }
72static char *duration(int t) { return format("%d:%02d",t/60,t%60); }

74static int lookup(const char *const *names, int count, const char *s) {
75 for (int i = 0; i < count; ++i) if (!strcmp(names[i],s)) return i;
76 return -1;

78static char *label(int s) { return format("L%d-%d",spots[s].level,spots[s].number); }
79static char *location(const Vehicle *v) {
80 return v->type == 4 ? format("%s..%d",label(v->spot),spots[v->spot].number+2) : label(v->spot);
81}
82static bool free_spot(int s) { return !spots[s].closed && spots[s].vehicle < 0; }
83static int pass_id(const char *p) {
84 for (int i = 0; i < pass_count; ++i) if (!strcmp(passes[i].plate,p)) return i;
85 return -1;
86}
87static bool valid_pass(const char *p) {
88 int i = pass_id(p);
89 return i >= 0 && now/1440+1 <= passes[i].last;
90}
91static int spot_id(const char *s) {
92 int level, number;
93 sscanf(s,"L%d-%d",&level,&number);
94 if (level > level_count || number > sizes[level-1]) return -1;
95 return starts[level-1]+number-1;
96}
97static int parked(const char *plate) {
98 for (int i = 0; i < vehicle_count; ++i) if (vehicles[i].active && !strcmp(vehicles[i].plate,plate)) return i;
99 return -1;
100}
101static int vehicle(const char *plate) {
102 int i = parked(plate);
103 error(i < 0,format("%s is not parked",plate));
104 return i;
105}
106static int waiting(const char *plate) {
107 for (int i = 0; i < queue_count; ++i) if (!strcmp(queue[i].plate,plate)) return i;
108 return -1;
109}
110static int preference(int type, int kind) {
111 for (int i = 0; i < preference_counts[type]; ++i) if (preferences[type][i] == kind) return i;
112 return -1;
113}
114static int assign(int type) {
115 if (type == 4) {
116 for (int l = 0; l < level_count; ++l) {
117 for (int j = 0; j+2 < sizes[l]; ++j) {
118 int s = starts[l]+j;
119 if (spots[s].kind == 2 && spots[s+1].kind == 2 && spots[s+2].kind == 2 && free_spot(s) && free_spot(s+1) && free_spot(s+2)) return s;
120 }
121 }
122 } else {
123 for (int k = 0; k < preference_counts[type]; ++k) {
124 for (int i = 0; i < spot_count; ++i) if (spots[i].kind == preferences[type][k] && free_spot(i)) return i;
125 }
126 }
127 return -1;
128}
129static void occupy(int id, int s) {
130 Vehicle *v = &vehicles[id];
131 v->spot = s;
132 for (int i = 0; i < (v->type == 4 ? 3 : 1); ++i) spots[s+i].vehicle = id;
133}
134static void vacate(int id) {
135 Vehicle *v = &vehicles[id];
136 for (int i = 0; i < (v->type == 4 ? 3 : 1); ++i) spots[v->spot+i].vehicle = -1;
137}
138static int park(const char *plate, int type, int s) {
139 int id = vehicle_count++;
140 Vehicle *v = &vehicles[id];
141 strcpy(v->plate,plate);
142 v->type = type;
143 v->entry = now;
144 v->active = true;
145 occupy(id,s);
146 ++parked_count;
147 ++totals[type].entries;
148 ++daily[now/1440+1][type].entries;
149 if (parked_count > peak) {
150 peak = parked_count;
151 peak_time = now;
152 }
153 return id;
154}
155static void refuse(int type, int time) {
156 ++totals[type].refused;
157 ++daily[time/1440+1][type].refused;
158}
159static void giveups(void) {
160 int length = 0;
161 for (int i = 0; i < queue_count; ++i) {
162 Waiting w = queue[i];
163 if (now-w.joined >= 30) {
164 line("GIVEUP %s %s %s",w.plate,types[w.type],time_text(w.joined+30));
165 refuse(w.type,w.joined+30);
166 } else queue[length++] = w;
167 }
168 queue_count = length;
169}
170static void process_queue(void) {
171 int length = 0;
172 for (int i = 0; i < queue_count; ++i) {
173 Waiting w = queue[i];
174 int s = assign(w.type);
175 if (s < 0) queue[length++] = w;
176 else {
177 int id = park(w.plate,w.type,s);
178 line("ADMIT %s %s %s waited %s",w.plate,types[w.type],location(&vehicles[id]),duration(now-w.joined));
179 }
180 }
181 queue_count = length;
182}
183static void interval(const Vehicle *v, int from, int free_time, int *parking, int *night) {
184 int length = now-from;
185 *parking = 0;
186 *night = 0;
187 if (valid_pass(v->plate) || length <= free_time) return;
188 for (int i = 0; i*1440 < length; ++i) {
189 int minutes = length-i*1440;
190 if (minutes > 1440) minutes = 1440;
191 if (!i) minutes -= free_time;
192 int amount = (minutes+29)/30*rates[v->type];
193 *parking += amount < caps[v->type] ? amount : caps[v->type];
194 }
195 *night = (now/1440-from/1440)*fees[0];
196}
197static Bill bill(const Vehicle *v, bool loss) {
198 int p, n;
199 interval(v,v->paid ? v->payment : v->entry,v->paid ? 15 : 15+v->validation,&p,&n);
200 int due = p+n;
201 p += v->paid ? v->parking : 0;
202 n += v->paid ? v->night : 0;
203 int lost = loss ? fees[1] : 0;
204 int total = p+n+lost+v->charge;
205 return (Bill){p,n,lost,v->charge,total,v->paid ? due : total};
206}
207static void fee_line(const char *heading, const Vehicle *v, Bill b) {
208 line("%s %s %s stay %s parking %d night %d lost %d charge %d total %d due %d",heading,v->plate,location(v),duration(now-v->entry),b.parking,b.night,b.lost,b.charge,b.total,b.due);
209}
210static char *annotation(const Vehicle *v) {
211 char *s = valid_pass(v->plate) ? " PASS" : "";
212 if (v->validation) s = format("%s VAL %d",s,v->validation);
213 if (v->energy) s = format("%s KWH %d",s,v->energy);
214 return v->paid ? format("%s PAID",s) : s;
215}
216static int location_order(const void *left, const void *right) {
217 int a = vehicles[*(const int *)left].spot, b = vehicles[*(const int *)right].spot;
218 return a < b ? -1 : a > b ? 1 : 0;
219}
220static int entry_order(const void *left, const void *right) {
221 int a = vehicles[*(const int *)left].entry, b = vehicles[*(const int *)right].entry;
222 return a != b ? (a < b ? -1 : 1) : location_order(left,right);
223}
224static int ordered_vehicles(int *ids) {
225 int length = 0;
226 for (int i = 0; i < vehicle_count; ++i) if (vehicles[i].active) ids[length++] = i;
227 qsort(ids,(size_t)length,sizeof(*ids),location_order);
228 return length;
229}
230static void move_vehicle(int id, int s) {
231 char *before = location(&vehicles[id]);
232 vacate(id);
233 occupy(id,s);
234 line("MOVE %s %s -> %s",vehicles[id].plate,before,label(s));
235}
236static int pass_order(const void *left, const void *right) {
237 return strcmp(passes[*(const int *)left].plate,passes[*(const int *)right].plate);
238}
239static Counts add_counts(Counts a, Counts b) {
240 return (Counts){a.entries+b.entries,a.exits+b.exits,a.refused+b.refused,a.revenue+b.revenue,a.stay+b.stay};
241}
242static void execute(char **a, int count) {
243 const char *cmd = a[0];
244 if (!strcmp(cmd,"LEVEL")) {
245 int level = level_count++, start = spot_count;
246 starts[level] = start;
247 for (int k = 0; k < 4; ++k) {
248 int added = atoi(a[k+1]);
249 for (int i = 0; i < added; ++i) {
250 spots[spot_count] = (Spot){level+1,spot_count-start+1,k,-1,false};
251 ++spot_count;
252 }
253 }
254 sizes[level] = spot_count-start;
255 line("LEVEL %d spots %d MOTO %s COMPACT %s LARGE %s EV %s",level+1,sizes[level],a[1],a[2],a[3],a[4]);
256 process_queue();
257 } else if (!strcmp(cmd,"CLOSE") || !strcmp(cmd,"OPEN")) {
258 int s = spot_id(a[1]);
259 if (error(s < 0,format("unknown spot %s",a[1]))) return;
260 Spot *p = &spots[s];
261 if (!strcmp(cmd,"CLOSE")) {
262 if (error(p->closed,format("spot %s is already closed",a[1]))) return;
263 if (error(p->vehicle >= 0,format("spot %s is occupied by %s",a[1],p->vehicle >= 0 ? vehicles[p->vehicle].plate : ""))) return;
264 p->closed = true;
265 } else {
266 if (error(!p->closed,format("spot %s is not closed",a[1]))) return;
267 p->closed = false;
268 }
269 line("%s %s %s",cmd,a[1],kinds[p->kind]);
270 if (!strcmp(cmd,"OPEN")) process_queue();
271 } else if (!strcmp(cmd,"RATE")) {
272 int type = lookup(types,5,a[1]);
273 if (error(type < 0,format("unknown type %s",a[1]))) return;
274 int rate = atoi(a[2]), cap = atoi(a[3]);
275 if (error(cap < rate,format("cap below rate for %s",a[1]))) return;
276 rates[type] = rate;
277 caps[type] = cap;
278 line("RATE %s %d %d",a[1],rate,cap);
279 } else if (!strcmp(cmd,"FEE")) {
280 const char *items[] = {"NIGHT","LOST","KWH"};
281 int item = lookup(items,3,a[1]);
282 if (error(item < 0,format("unknown fee %s",a[1]))) return;
283 fees[item] = atoi(a[2]);
284 line("FEE %s %s",a[1],a[2]);
285 } else if (!strcmp(cmd,"ENTER")) {
286 int type = lookup(types,5,a[2]);
287 if (error(type < 0,format("unknown type %s",a[2]))) return;
288 int previous = parked(a[1]);
289 if (error(previous >= 0,format("%s is already parked at %s",a[1],previous >= 0 ? location(&vehicles[previous]) : ""))) return;
290 if (error(waiting(a[1]) >= 0,format("%s is already waiting",a[1]))) return;
291 int s = assign(type);
292 if (s >= 0) {
293 int id = park(a[1],type,s);
294 line("ENTER %s %s %s",a[1],a[2],location(&vehicles[id]));
295 } else if (queue_count < 3) {
296 Waiting *w = &queue[queue_count++];
297 strcpy(w->plate,a[1]);
298 w->type = type;
299 w->joined = now;
300 line("WAIT %s %s %d",a[1],a[2],queue_count);
301 } else {
302 refuse(type,now);
303 line("FULL %s %s",a[1],a[2]);
304 }
305 } else if (!strcmp(cmd,"LEAVE")) {
306 int i = waiting(a[1]);
307 if (error(i < 0,format("%s is not waiting",a[1]))) return;
308 line("LEAVE %s %s",a[1],types[queue[i].type]);
309 memmove(queue+i,queue+i+1,(size_t)(queue_count-i-1)*sizeof(*queue));
310 --queue_count;
311 } else if (!strcmp(cmd,"EXIT") || !strcmp(cmd,"PAY") || !strcmp(cmd,"QUOTE") || !strcmp(cmd,"CHARGE") || !strcmp(cmd,"VALIDATE") || !strcmp(cmd,"MOVE")) {
312 int id = vehicle(a[1]);
313 if (id < 0) return;
314 Vehicle *v = &vehicles[id];
315 if (!strcmp(cmd,"EXIT")) {
316 bool loss = count == 3;
317 if (error(loss && v->paid,format("%s has already paid",a[1]))) return;
318 Bill b = bill(v,loss);
319 fee_line("EXIT",v,b);
320 Record *r = &records[record_count++];
321 strcpy(r->plate,v->plate);
322 strcpy(r->location,location(v));
323 r->entry = v->entry;
324 r->exit = now;
325 r->type = v->type;
326 r->total = b.total;
327 r->lost = loss;
328 ++totals[v->type].exits;
329 totals[v->type].stay += now-v->entry;
330 totals[v->type].revenue += b.total;
331 ++daily[now/1440+1][v->type].exits;
332 daily[now/1440+1][v->type].revenue += b.total;
333 Revenue *sum = &revenue[v->type];
334 ++sum->exits;
335 sum->parking += b.parking;
336 sum->night += b.night;
337 sum->lost += b.lost;
338 sum->charge += b.charge;
339 sum->total += b.total;
340 vacate(id);
341 v->active = false;
342 --parked_count;
343 process_queue();
344 } else if (!strcmp(cmd,"QUOTE")) fee_line("QUOTE",v,bill(v,false));
345 else if (!strcmp(cmd,"PAY")) {
346 if (error(v->paid,format("%s has already paid",a[1]))) return;
347 Bill b = bill(v,false);
348 v->paid = true;
349 v->payment = now;
350 v->parking = b.parking;
351 v->night = b.night;
352 line("PAY %s parking %d night %d charge %d total %d",v->plate,b.parking,b.night,b.charge,b.total);
353 } else if (!strcmp(cmd,"CHARGE")) {
354 if (error(v->type != 2,format("%s is not an electric car",a[1])) || error(spots[v->spot].kind != 3,format("%s is not at a charger",a[1])) || error(v->paid,format("%s has already paid",a[1]))) return;
355 int energy = atoi(a[2]);
356 if (error(v->energy+energy > 100,format("charge limit exceeded for %s",a[1]))) return;
357 int cost = energy*fees[2];
358 v->energy += energy;
359 v->charge += cost;
360 line("CHARGE %s kwh %d cost %d stay_kwh %d stay_cost %d",v->plate,energy,cost,v->energy,v->charge);
361 } else if (!strcmp(cmd,"VALIDATE")) {
362 if (error(v->paid,format("%s has already paid",a[1]))) return;
363 int minutes = atoi(a[2]);
364 if (error(v->validation+minutes > 180,format("validation limit exceeded for %s",a[1]))) return;
365 v->validation += minutes;
366 line("VALIDATE %s %d total %d",v->plate,minutes,v->validation);
367 } else {
368 if (error(v->type == 4,format("bus %s cannot be moved",a[1]))) return;
369 int s = spot_id(a[2]);
370 if (error(s < 0,format("unknown spot %s",a[2])) || error(s == v->spot,format("%s is already at %s",a[1],a[2]))) return;
371 if (error(spots[s].closed,format("spot %s is closed",a[2]))) return;
372 if (error(spots[s].vehicle >= 0,format("spot %s is occupied by %s",a[2],spots[s].vehicle >= 0 ? vehicles[spots[s].vehicle].plate : ""))) return;
373 if (error(preference(v->type,spots[s].kind) < 0,format("%s cannot use %s spot %s",types[v->type],kinds[spots[s].kind],a[2]))) return;
374 move_vehicle(id,s);
375 process_queue();
376 }
377 } else if (!strcmp(cmd,"PASS")) {
378 int i = pass_id(a[1]), days = atoi(a[2]);
379 bool valid = i >= 0 && now/1440+1 <= passes[i].last;
380 if (i < 0) {
381 i = pass_count++;
382 strcpy(passes[i].plate,a[1]);
383 }
384 passes[i].last = valid ? passes[i].last+days : now/1440+days;
385 line("PASS %s until %d",a[1],passes[i].last);
386 } else if (!strcmp(cmd,"UNPASS")) {
387 if (error(!valid_pass(a[1]),format("%s has no pass",a[1]))) return;
388 passes[pass_id(a[1])].last = 0;
389 line("UNPASS %s",a[1]);
390 } else if (!strcmp(cmd,"REBALANCE")) {
391 int ids[1000], count_ids = ordered_vehicles(ids), moved = 0;
392 for (int j = 0; j < count_ids; ++j) {
393 int id = ids[j];
394 Vehicle *v = &vehicles[id];
395 if (v->type == 4) continue;
396 int current = preference(v->type,spots[v->spot].kind);
397 if (!current) continue;
398 int s = assign(v->type);
399 if (s >= 0 && preference(v->type,spots[s].kind) < current) {
400 move_vehicle(id,s);
401 ++moved;
402 }
403 }
404 line("REBALANCE %d",moved);
405 process_queue();
406 } else if (!strcmp(cmd,"WHERE")) {
407 int id = parked(a[1]), q = waiting(a[1]), last = -1;
408 if (id >= 0) {
409 Vehicle *v = &vehicles[id];
410 line("WHERE %s %s %s %s stay %s%s",a[1],location(v),types[v->type],time_text(v->entry),duration(now-v->entry),annotation(v));
411 } else if (q >= 0) line("WHERE %s waiting %d",a[1],q+1);
412 else {
413 for (int i = 0; i < record_count; ++i) if (!strcmp(records[i].plate,a[1])) last = i;
414 if (last >= 0) line("WHERE %s gone %s",a[1],time_text(records[last].exit));
415 else error(true,format("unknown plate %s",a[1]));
416 }
417 } else if (!strcmp(cmd,"LIST") || !strcmp(cmd,"OVERSTAY")) {
418 int level = !strcmp(cmd,"LIST") && count == 2 ? atoi(a[1]) : 0;
419 if (error(level > level_count,format("unknown level %d",level))) return;
420 int ids[1000], all_count = ordered_vehicles(ids), length = 0;
421 for (int i = 0; i < all_count; ++i) {
422 Vehicle *v = &vehicles[ids[i]];
423 if (level && spots[v->spot].level != level) continue;
424 if (!strcmp(cmd,"OVERSTAY") && now-v->entry < atoi(a[1])*60) continue;
425 ids[length++] = ids[i];
426 }
427 if (!strcmp(cmd,"OVERSTAY")) {
428 qsort(ids,(size_t)length,sizeof(*ids),entry_order);
429 line("OVERSTAY %s %d",a[1],length);
430 } else line("LIST %d",length);
431 for (int i = 0; i < length; ++i) {
432 Vehicle *v = &vehicles[ids[i]];
433 line("%s %s %s %s%s",location(v),v->plate,types[v->type],time_text(v->entry),!strcmp(cmd,"LIST") ? annotation(v) : format(" stay %s",duration(now-v->entry)));
434 }
435 } else if (!strcmp(cmd,"MAP")) {
436 line("MAP %d",level_count);
437 for (int l = 0; l < level_count; ++l) {
438 char symbols[140];
439 int length = 0;
440 for (int j = 0; j < sizes[l]; ++j) {
441 if (j && j%10 == 0) symbols[length++] = ' ';
442 Spot s = spots[starts[l]+j];
443 symbols[length++] = s.closed ? '#' : (s.vehicle >= 0 ? "MCLE" : "mcle")[s.kind];
444 }
445 symbols[length] = '\0';
446 line("L%d %s",l+1,symbols);
447 }
448 } else if (!strcmp(cmd,"OCCUPANCY")) {
449 line("OCCUPANCY vehicles %d waiting %d",parked_count,queue_count);
450 int all_used[4] = {0}, all_available[4] = {0}, all_closed = 0;
451 for (int l = 0; l < level_count; ++l) {
452 int used[4] = {0}, available[4] = {0}, closed = 0;
453 for (int j = 0; j < sizes[l]; ++j) {
454 Spot s = spots[starts[l]+j];
455 if (s.closed) ++closed;
456 else ++available[s.kind];
457 if (s.vehicle >= 0) ++used[s.kind];
458 }
459 printf("L%d",l+1);
460 for (int k = 0; k < 4; ++k) {
461 printf(" %s %d/%d",kinds[k],used[k],available[k]);
462 all_used[k] += used[k];
463 all_available[k] += available[k];
464 }
465 all_closed += closed;
466 line(" closed %d",closed);
467 }
468 printf("TOTAL");
469 int used = 0, available = 0;
470 for (int k = 0; k < 4; ++k) {
471 printf(" %s %d/%d",kinds[k],all_used[k],all_available[k]);
472 used += all_used[k];
473 available += all_available[k];
474 }
475 line(" closed %d rate %s%%",all_closed,tenth(available ? used*1000/available : 0));
476 } else if (!strcmp(cmd,"SIGN")) {
477 int free_counts[4] = {0}, buses = 0;
478 for (int l = 0; l < level_count; ++l) {
479 int run = 0;
480 for (int j = 0; j < sizes[l]; ++j) {
481 int s = starts[l]+j;
482 if (free_spot(s)) ++free_counts[spots[s].kind];
483 if (free_spot(s) && spots[s].kind == 2) ++run;
484 else {
485 buses += run/3;
486 run = 0;
487 }
488 }
489 buses += run/3;
490 }
491 printf("SIGN");
492 for (int k = 0; k < 5; ++k) {
493 int count = buses;
494 if (k != 4) {
495 count = 0;
496 for (int i = 0; i < preference_counts[k]; ++i) count += free_counts[preferences[k][i]];
497 }
498 printf(" %s %s",types[k],count ? format("%d",count) : "FULL");
499 }
500 putchar('\n');
501 } else if (!strcmp(cmd,"RATES")) {
502 line("RATES");
503 for (int k = 0; k < 5; ++k) line("%s %d %d",types[k],rates[k],caps[k]);
504 line("NIGHT %d LOST %d KWH %d",fees[0],fees[1],fees[2]);
505 } else if (!strcmp(cmd,"PASSES")) {
506 int ids[1000], length = 0;
507 for (int i = 0; i < pass_count; ++i) if (now/1440+1 <= passes[i].last) ids[length++] = i;
508 qsort(ids,(size_t)length,sizeof(*ids),pass_order);
509 line("PASSES %d",length);
510 for (int j = 0; j < length; ++j) {
511 Pass p = passes[ids[j]];
512 line("%s until %d%s",p.plate,p.last,parked(p.plate) >= 0 ? " IN" : "");
513 }
514 } else if (!strcmp(cmd,"HISTORY")) {
515 int length = 0, total = 0;
516 for (int i = 0; i < record_count; ++i) {
517 if (!strcmp(records[i].plate,a[1])) {
518 ++length;
519 total += records[i].total;
520 }
521 }
522 if (error(!length,format("no history for %s",a[1]))) return;
523 line("HISTORY %s visits %d paid %d",a[1],length,total);
524 for (int i = 0; i < record_count; ++i) {
525 Record r = records[i];
526 if (!strcmp(r.plate,a[1])) line("%s %s %s %s %d%s",time_text(r.entry),time_text(r.exit),r.location,types[r.type],r.total,r.lost ? " LOST" : "");
527 }
528 } else if (!strcmp(cmd,"REVENUE")) {
529 line("REVENUE");
530 Revenue all = {0};
531 for (int k = 0; k <= 5; ++k) {
532 Revenue r = k == 5 ? all : revenue[k];
533 line("%s exits %d parking %d night %d lost %d charge %d total %d",k == 5 ? "TOTAL" : types[k],r.exits,r.parking,r.night,r.lost,r.charge,r.total);
534 if (k < 5) {
535 all.exits += r.exits;
536 all.parking += r.parking;
537 all.night += r.night;
538 all.lost += r.lost;
539 all.charge += r.charge;
540 all.total += r.total;
541 }
542 }
543 } else if (!strcmp(cmd,"DAILY") || !strcmp(cmd,"STATS")) {
544 bool by_day = !strcmp(cmd,"DAILY");
545 Counts *rows = by_day ? daily[atoi(a[1])] : totals;
546 Counts all = {0};
547 for (int k = 0; k < 5; ++k) all = add_counts(all,rows[k]);
548 if (by_day) line("DAILY %s entries %d exits %d refused %d revenue %d",a[1],all.entries,all.exits,all.refused,all.revenue);
549 else line("STATS parked %d peak %d at %s",parked_count,peak,peak ? time_text(peak_time) : "-");
550 for (int k = 0; k < (by_day ? 5 : 6); ++k) {
551 Counts c = k == 5 ? all : rows[k];
552 const char *name = k == 5 ? "ALL" : types[k];
553 if (by_day) line("%s entries %d exits %d refused %d revenue %d",name,c.entries,c.exits,c.refused,c.revenue);
554 else line("%s entries %d exits %d refused %d avg_stay %s",name,c.entries,c.exits,c.refused,tenth(c.exits ? c.stay*10/c.exits : 0));
555 }
556 }
557}
558int main(void) {
559 char input[256];
560 if (!fgets(input,sizeof(input),stdin)) return 0;
561 int count = atoi(input);
562 for (int i = 0; i < count; ++i) {
563 if (!fgets(input,sizeof(input),stdin)) break;
564 input[strcspn(input,"\n")] = '\0';
565 char *a[12];
566 int length = 0;
567 for (char *word = strtok(input," "); word; word = strtok(NULL," ")) a[length++] = word;
568 int hour, minute;
569 sscanf(a[1],"%d:%d",&hour,&minute);
570 now = (atoi(a[0])-1)*1440+hour*60+minute;
571 giveups();
572 execute(a+2,length-2);
573 clear_temps();
574 }
