1using Printf
2const TYPES = ("MOTORCYCLE","CAR","ECAR","VAN","BUS")
3const KINDS = ("MOTO","COMPACT","LARGE","EV")
4const PREFS = Dict("MOTORCYCLE"=>["MOTO","COMPACT","LARGE"],"CAR"=>["COMPACT","LARGE"],"ECAR"=>["EV","COMPACT","LARGE"],"VAN"=>["LARGE"])
5Base.@kwdef mutable struct GarageSpot
6 id::String
7 level::Int
8 number::Int
9 kind::String
10 closed::Bool = false
11 plate::String = ""
12end
13struct Payment
14 time::Int
15 parking::Int
16 night::Int
17end
18Base.@kwdef mutable struct Vehicle
19 plate::String
20 kind::String
21 spots::Vector{GarageSpot}
22 entry::Int
23 kwh::Int = 0
24 cost::Int = 0
25 validation::Int = 0
26 payment::Union{Nothing,Payment} = nothing
27end
28struct WaitEntry
29 plate::String
30 kind::String
31 time::Int
32end
33struct ExitRecord
34 entry::Int
35 exit::Int
36 location::String
37 kind::String
38 total::Int
39 lost::Bool
40end
41const RATES = Dict("MOTORCYCLE"=>(100,800),"CAR"=>(200,1600),"ECAR"=>(200,1600),"VAN"=>(300,2400),"BUS"=>(600,4800))
42const FEES = Dict("NIGHT"=>300,"LOST"=>2000,"KWH"=>30)
43const LEVELS = Vector{GarageSpot}[]
44const SPOTS = Dict{String,GarageSpot}()
45const VEHICLES = Dict{String,Vehicle}()
46const QUEUE = WaitEntry[]
47const PASSES = Dict{String,Int}()
48const HISTORY = Dict{String,Vector{ExitRecord}}()
49const STATS = Dict(kind=>Dict(field=>0 for field in ("entries","exits","refused","stay","parking","night","lost","charge","total")) for kind in TYPES)
50const DAILY = Dict{Int,Dict{String,Dict{String,Int}}}()
51const NOW = Ref(0)
52const PEAK = Ref(0)
53const PEAK_TIME = Ref(0)
54fail(message) = throw(ArgumentError(message))
55date_of(time) = div(time,1440)+1
56time_text(time) = string(date_of(time),'-',@sprintf("%02d:%02d",div(mod(time,1440),60),mod(time,60)))
57duration(minutes) = @sprintf("%d:%02d",div(minutes,60),mod(minutes,60))
58one_decimal(q) = string(div(q,10),'.',mod(q,10))
59valid_pass(plate,time) = haskey(PASSES,plate) && date_of(time) <= PASSES[plate]
60function get_spot(id)
61 haskey(SPOTS,id) || fail("unknown spot $id")
62 return SPOTS[id]
63end
64function get_vehicle(plate)
65 haskey(VEHICLES,plate) || fail("$plate is not parked")
66 return VEHICLES[plate]
67end
68function location(v)
69 s = v.spots[1]
70 return v.kind == "BUS" ? "L$(s.level)-$(s.number)..$(s.number+2)" : s.id
71end
72location_key(v) = (v.spots[1].level,v.spots[1].number)
73free_spot(s) = !s.closed && isempty(s.plate)
74function assignment(kind)
75 if kind == "BUS"
76 for level in LEVELS
77 for i in 1:length(level)-2
78 triple = level[i:i+2]
79 all(s -> s.kind == "LARGE" && free_spot(s),triple) && return triple
80 end
81 end
82 else
83 for preferred in PREFS[kind]
84 for level in LEVELS
85 for s in level
86 s.kind == preferred && free_spot(s) && return [s]
87 end
88 end
89 end
90 end
91 return GarageSpot[]
92end
93function add_stat!(kind,field,amount,date)
94 STATS[kind][field] += amount
95 if field in ("entries","exits","refused","total")
96 if !haskey(DAILY,date)
97 DAILY[date] = Dict(k=>Dict(f=>0 for f in ("entries","exits","refused","revenue")) for k in TYPES)
98 end
99 dailyfield = field == "total" ? "revenue" : field
100 DAILY[date][kind][dailyfield] += amount
101 end
102end
103function park_vehicle!(plate,kind,chosen)
104 v = Vehicle(plate=plate,kind=kind,spots=chosen,entry=NOW[])
105 for s in chosen
106 s.plate = plate
107 end
108 VEHICLES[plate] = v
109 add_stat!(kind,"entries",1,date_of(NOW[]))
110 if length(VEHICLES) > PEAK[]
111 PEAK[],PEAK_TIME[] = length(VEHICLES),NOW[]
112 end
113 return v
114end
115function process_queue!(out)
116 remaining = WaitEntry[]
117 for waiting in QUEUE
118 chosen = assignment(waiting.kind)
119 if isempty(chosen)
120 push!(remaining,waiting)
121 else
122 v = park_vehicle!(waiting.plate,waiting.kind,chosen)
123 push!(out,"ADMIT $(v.plate) $(v.kind) $(location(v)) waited $(duration(NOW[]-waiting.time))")
124 end
125 end
126 empty!(QUEUE)
127 append!(QUEUE,remaining)
128end
129function giveups!()
130 out = String[]
131 remaining = WaitEntry[]
132 for waiting in QUEUE
133 if NOW[]-waiting.time >= 30
134 time = waiting.time+30
135 push!(out,"GIVEUP $(waiting.plate) $(waiting.kind) $(time_text(time))")
136 add_stat!(waiting.kind,"refused",1,date_of(time))
137 else
138 push!(remaining,waiting)
139 end
140 end
141 empty!(QUEUE)
142 append!(QUEUE,remaining)
143 return out
144end
145function interval_fee(v,a,b,free)
146 total = b-a
147 (valid_pass(v.plate,b) || total <= free) && return (0,0)
148 rate,cap = RATES[v.kind]
149 parking,i = 0,0
150 while total > 0
151 minutes = min(total,1440)
152 billable = minutes-(i == 0 ? free : 0)
153 parking += min(cld(billable,30)*rate,cap)
154 total -= minutes
155 i += 1
156 end
157 return parking,(date_of(b)-date_of(a))*FEES["NIGHT"]
158end
159function charges(v,lost=false)
160 if v.payment === nothing
161 parking,night = interval_fee(v,v.entry,NOW[],15+v.validation)
162 loss = lost ? FEES["LOST"] : 0
163 total = parking+night+loss+v.cost
164 return (parking,night,loss,v.cost,total,total)
165 end
166 extra,extranight = interval_fee(v,v.payment.time,NOW[],15)
167 parking,night = v.payment.parking+extra,v.payment.night+extranight
168 return (parking,night,0,v.cost,parking+night+v.cost,extra+extranight)
169end
170fee_line(label,v,f) = "$label $(v.plate) $(location(v)) stay $(duration(NOW[]-v.entry)) parking $(f[1]) night $(f[2]) lost $(f[3]) charge $(f[4]) total $(f[5]) due $(f[6])"
171function annotation(v)
172 text = valid_pass(v.plate,NOW[]) ? " PASS" : ""
173 v.validation != 0 && (text *= " VAL $(v.validation)")
174 v.kwh != 0 && (text *= " KWH $(v.kwh)")
175 v.payment !== nothing && (text *= " PAID")
176 return text
177end
178function move_vehicle!(v,spot)
179 old = location(v)
180 v.spots[1].plate = ""
181 spot.plate = v.plate
182 v.spots = [spot]
183 return "MOVE $(v.plate) $old -> $(spot.id)"
184end
185function occupancy_line(label,used,available,closed)
186 parts = [label]
187 for kind in KINDS
188 append!(parts,[kind,"$(used[kind])/$(available[kind])"])
189 end
190 return join(parts," ")*" closed $closed"
191end
192function execute(t)
193 cmd = t[1]
194 out = String[]
195 if cmd == "LEVEL"
196 levelno = length(LEVELS)+1
197 level = GarageSpot[]
198 for (index,kind) in enumerate(KINDS)
199 for i in 1:parse(Int,t[index+1])
200 number = length(level)+1
201 id = "L$levelno-$number"
202 s = GarageSpot(id=id,level=levelno,number=number,kind=kind)
203 SPOTS[id] = s
204 push!(level,s)
205 end
206 end
207 push!(LEVELS,level)
208 push!(out,"LEVEL $levelno spots $(length(level)) MOTO $(t[2]) COMPACT $(t[3]) LARGE $(t[4]) EV $(t[5])")
209 elseif cmd in ("CLOSE","OPEN")
210 s = get_spot(t[2])
211 if cmd == "CLOSE"
212 s.closed && fail("spot $(t[2]) is already closed")
213 !isempty(s.plate) && fail("spot $(t[2]) is occupied by $(s.plate)")
214 s.closed = true
215 else
216 s.closed || fail("spot $(t[2]) is not closed")
217 s.closed = false
218 end
219 push!(out,"$cmd $(s.id) $(s.kind)")
220 elseif cmd == "RATE"
221 haskey(RATES,t[2]) || fail("unknown type $(t[2])")
222 rate,cap = parse.(Int,t[3:4])
223 cap >= rate || fail("cap below rate for $(t[2])")
224 RATES[t[2]] = (rate,cap)
225 push!(out,"RATE $(t[2]) $(t[3]) $(t[4])")
226 elseif cmd == "FEE"
227 haskey(FEES,t[2]) || fail("unknown fee $(t[2])")
228 FEES[t[2]] = parse(Int,t[3])
229 push!(out,"FEE $(t[2]) $(t[3])")
230 elseif cmd == "ENTER"
231 plate,kind = t[2:3]
232 haskey(RATES,kind) || fail("unknown type $kind")
233 haskey(VEHICLES,plate) && fail("$plate is already parked at $(location(VEHICLES[plate]))")
234 any(w -> w.plate == plate,QUEUE) && fail("$plate is already waiting")
235 chosen = assignment(kind)
236 if !isempty(chosen)
237 v = park_vehicle!(plate,kind,chosen)
238 push!(out,"ENTER $plate $kind $(location(v))")
239 elseif length(QUEUE) < 3
240 push!(QUEUE,WaitEntry(plate,kind,NOW[]))
241 push!(out,"WAIT $plate $kind $(length(QUEUE))")
242 else
243 add_stat!(kind,"refused",1,date_of(NOW[]))
244 push!(out,"FULL $plate $kind")
245 end
246 elseif cmd == "LEAVE"
247 index = findfirst(w -> w.plate == t[2],QUEUE)
248 index === nothing && fail("$(t[2]) is not waiting")
249 w = QUEUE[index]
250 deleteat!(QUEUE,index)
251 push!(out,"LEAVE $(w.plate) $(w.kind)")
252 elseif cmd in ("EXIT","PAY","QUOTE","CHARGE","VALIDATE","MOVE")
253 v = get_vehicle(t[2])
254 if cmd == "EXIT"
255 lost = length(t) == 3
256 lost && v.payment !== nothing && fail("$(v.plate) has already paid")
257 f = charges(v,lost)
258 push!(out,fee_line("EXIT",v,f))
259 record = ExitRecord(v.entry,NOW[],location(v),v.kind,f[5],lost)
260 push!(get!(HISTORY,v.plate,ExitRecord[]),record)
261 for s in v.spots
262 s.plate = ""
263 end
264 delete!(VEHICLES,v.plate)
265 add_stat!(v.kind,"exits",1,date_of(NOW[]))
266 add_stat!(v.kind,"stay",NOW[]-v.entry,date_of(NOW[]))
267 for (index,field) in enumerate(("parking","night","lost","charge","total"))
268 add_stat!(v.kind,field,f[index],date_of(NOW[]))
269 end
270 elseif cmd == "PAY"
271 v.payment !== nothing && fail("$(v.plate) has already paid")
272 f = charges(v)
273 v.payment = Payment(NOW[],f[1],f[2])
274 push!(out,"PAY $(v.plate) parking $(f[1]) night $(f[2]) charge $(f[4]) total $(f[5])")
275 elseif cmd == "QUOTE"
276 push!(out,fee_line("QUOTE",v,charges(v)))
277 elseif cmd == "CHARGE"
278 v.kind == "ECAR" || fail("$(v.plate) is not an electric car")
279 v.spots[1].kind == "EV" || fail("$(v.plate) is not at a charger")
280 v.payment !== nothing && fail("$(v.plate) has already paid")
281 kwh = parse(Int,t[3])
282 v.kwh+kwh <= 100 || fail("charge limit exceeded for $(v.plate)")
283 cost = kwh*FEES["KWH"]
284 v.kwh += kwh
285 v.cost += cost
286 push!(out,"CHARGE $(v.plate) kwh $kwh cost $cost stay_kwh $(v.kwh) stay_cost $(v.cost)")
287 elseif cmd == "VALIDATE"
288 v.payment !== nothing && fail("$(v.plate) has already paid")
289 minutes = parse(Int,t[3])
290 v.validation+minutes <= 180 || fail("validation limit exceeded for $(v.plate)")
291 v.validation += minutes
292 push!(out,"VALIDATE $(v.plate) $minutes total $(v.validation)")
293 else
294 v.kind == "BUS" && fail("bus $(v.plate) cannot be moved")
295 s = get_spot(t[3])
296 v.spots[1].id == s.id && fail("$(v.plate) is already at $(s.id)")
297 s.closed && fail("spot $(s.id) is closed")
298 !isempty(s.plate) && fail("spot $(s.id) is occupied by $(s.plate)")
299 s.kind in PREFS[v.kind] || fail("$(v.kind) cannot use $(s.kind) spot $(s.id)")
300 push!(out,move_vehicle!(v,s))
301 end
302 elseif cmd == "PASS"
303 plate,days = t[2],parse(Int,t[3])
304 PASSES[plate] = valid_pass(plate,NOW[]) ? PASSES[plate]+days : date_of(NOW[])+days-1
305 push!(out,"PASS $plate until $(PASSES[plate])")
306 elseif cmd == "UNPASS"
307 valid_pass(t[2],NOW[]) || fail("$(t[2]) has no pass")
308 delete!(PASSES,t[2])
309 push!(out,"UNPASS $(t[2])")
310 elseif cmd == "REBALANCE"
311 targets = sort(filter(v -> v.kind != "BUS",collect(values(VEHICLES)));by=location_key)
312 moved = 0
313 for v in targets
314 current = findfirst(==(v.spots[1].kind),PREFS[v.kind])
315 current == 1 && continue
316 chosen = assignment(v.kind)
317 if !isempty(chosen) && findfirst(==(chosen[1].kind),PREFS[v.kind]) < current
318 push!(out,move_vehicle!(v,chosen[1]))
319 moved += 1
320 end
321 end
322 push!(out,"REBALANCE $moved")
323 elseif cmd == "WHERE"
324 plate = t[2]
325 if haskey(VEHICLES,plate)
326 v = VEHICLES[plate]
327 push!(out,"WHERE $plate $(location(v)) $(v.kind) $(time_text(v.entry)) stay $(duration(NOW[]-v.entry))$(annotation(v))")
328 else
329 index = findfirst(w -> w.plate == plate,QUEUE)
330 if index !== nothing
331 push!(out,"WHERE $plate waiting $index")
332 elseif haskey(HISTORY,plate)
333 push!(out,"WHERE $plate gone $(time_text(HISTORY[plate][end].exit))")
334 else
335 fail("unknown plate $plate")
336 end
337 end
338 elseif cmd in ("LIST","OVERSTAY")
339 if cmd == "LIST" && length(t) == 2
340 1 <= parse(Int,t[2]) <= length(LEVELS) || fail("unknown level $(t[2])")
341 end
342 targets = collect(values(VEHICLES))
343 if cmd == "LIST" && length(t) == 2
344 filter!(v -> v.spots[1].level == parse(Int,t[2]),targets)
345 elseif cmd == "OVERSTAY"
346 filter!(v -> NOW[]-v.entry >= 60parse(Int,t[2]),targets)
347 end
348 sort!(targets;by=v -> cmd == "OVERSTAY" ? (v.entry,location_key(v)...) : location_key(v))
349 header = cmd == "OVERSTAY" ? "$cmd $(t[2])" : cmd
350 push!(out,"$header $(length(targets))")
351 for v in targets
352 text = "$(location(v)) $(v.plate) $(v.kind) $(time_text(v.entry))"
353 text *= cmd == "LIST" ? annotation(v) : " stay $(duration(NOW[]-v.entry))"
354 push!(out,text)
355 end
356 elseif cmd == "MAP"
357 push!(out,"MAP $(length(LEVELS))")
358 for (index,level) in enumerate(LEVELS)
359 symbols = String[]
360 for s in level
361 if s.closed
362 push!(symbols,"#")
363 else
364 initial = string(first(s.kind))
365 push!(symbols,isempty(s.plate) ? lowercase(initial) : initial)
366 end
367 end
368 chunks = (join(chunk) for chunk in Iterators.partition(symbols,10))
369 push!(out,"L$index "*join(chunks," "))
370 end
371 elseif cmd == "OCCUPANCY"
372 push!(out,"OCCUPANCY vehicles $(length(VEHICLES)) waiting $(length(QUEUE))")
373 totalused,totalavailable = Dict(k=>0 for k in KINDS),Dict(k=>0 for k in KINDS)
374 totalclosed = 0
375 for (index,level) in enumerate(LEVELS)
376 used,available = Dict(k=>0 for k in KINDS),Dict(k=>0 for k in KINDS)
377 closedcount = 0
378 for s in level
379 if s.closed
380 closedcount += 1
381 else
382 available[s.kind] += 1
383 end
384 !isempty(s.plate) && (used[s.kind] += 1)
385 end
386 push!(out,occupancy_line("L$index",used,available,closedcount))
387 for kind in KINDS
388 totalused[kind] += used[kind]
389 totalavailable[kind] += available[kind]
390 end
391 totalclosed += closedcount
392 end
393 capacity = sum(values(totalavailable))
394 usage = capacity == 0 ? 0 : div(sum(values(totalused))*1000,capacity)
395 push!(out,occupancy_line("TOTAL",totalused,totalavailable,totalclosed)*" rate $(one_decimal(usage))%")
396 elseif cmd == "SIGN"
397 parts = ["SIGN"]
398 for kind in TYPES
399 total = 0
400 for level in LEVELS
401 run = 0
402 for s in level
403 if kind == "BUS"
404 if s.kind == "LARGE" && free_spot(s)
405 run += 1
406 else
407 total += div(run,3)
408 run = 0
409 end
410 elseif s.kind in PREFS[kind] && free_spot(s)
411 total += 1
412 end
413 end
414 kind == "BUS" && (total += div(run,3))
415 end
416 append!(parts,[kind,total == 0 ? "FULL" : string(total)])
417 end
418 push!(out,join(parts," "))
419 elseif cmd == "RATES"
420 push!(out,"RATES")
421 for kind in TYPES
422 push!(out,"$kind $(RATES[kind][1]) $(RATES[kind][2])")
423 end
424 push!(out,"NIGHT $(FEES["NIGHT"]) LOST $(FEES["LOST"]) KWH $(FEES["KWH"])")
425 elseif cmd == "PASSES"
426 plates = sort(filter(plate -> valid_pass(plate,NOW[]),collect(keys(PASSES))))
427 push!(out,"PASSES $(length(plates))")
428 for plate in plates
429 push!(out,"$plate until $(PASSES[plate])"*(haskey(VEHICLES,plate) ? " IN" : ""))
430 end
431 elseif cmd == "HISTORY"
432 haskey(HISTORY,t[2]) || fail("no history for $(t[2])")
433 records = HISTORY[t[2]]
434 paid = sum(r.total for r in records)
435 push!(out,"HISTORY $(t[2]) visits $(length(records)) paid $paid")
436 for r in records
437 push!(out,"$(time_text(r.entry)) $(time_text(r.exit)) $(r.location) $(r.kind) $(r.total)"*(r.lost ? " LOST" : ""))
438 end
439 elseif cmd == "REVENUE"
440 push!(out,"REVENUE")
441 total = Dict(key=>0 for key in keys(STATS[TYPES[1]]))
442 for kind in TYPES
443 s = STATS[kind]
444 push!(out,"$kind exits $(s["exits"]) parking $(s["parking"]) night $(s["night"]) lost $(s["lost"]) charge $(s["charge"]) total $(s["total"])")
445 for (key,value) in s
446 total[key] += value
447 end
448 end
449 push!(out,"TOTAL exits $(total["exits"]) parking $(total["parking"]) night $(total["night"]) lost $(total["lost"]) charge $(total["charge"]) total $(total["total"])")
450 elseif cmd == "DAILY"
451 date = parse(Int,t[2])
452 rows = get(DAILY,date,Dict(k=>Dict(f=>0 for f in ("entries","exits","refused","revenue")) for k in TYPES))
453 total = Dict(f=>0 for f in ("entries","exits","refused","revenue"))
454 lines = String[]
455 for kind in TYPES
456 s = rows[kind]
457 for (key,value) in s
458 total[key] += value
459 end
460 push!(lines,"$kind entries $(s["entries"]) exits $(s["exits"]) refused $(s["refused"]) revenue $(s["revenue"])")
461 end
462 push!(out,"DAILY $date entries $(total["entries"]) exits $(total["exits"]) refused $(total["refused"]) revenue $(total["revenue"])")
463 append!(out,lines)
464 elseif cmd == "STATS"
465 peaktext = PEAK[] == 0 ? "-" : time_text(PEAK_TIME[])
466 push!(out,"STATS parked $(length(VEHICLES)) peak $(PEAK[]) at $peaktext")
467 total = Dict(f=>0 for f in ("entries","exits","refused","stay"))
468 for kind in (TYPES...,"ALL")
469 s = kind == "ALL" ? total : STATS[kind]
470 if kind != "ALL"
471 for key in keys(total)
472 total[key] += s[key]
473 end
474 end
475 average = s["exits"] == 0 ? 0 : div(s["stay"]*10,s["exits"])
476 push!(out,"$kind entries $(s["entries"]) exits $(s["exits"]) refused $(s["refused"]) avg_stay $(one_decimal(average))")
477 end
478 end
479 cmd in ("LEVEL","OPEN","EXIT","MOVE","REBALANCE") && process_queue!(out)
480 return out
481end
482function main()
483 for i in 1:parse(Int,readline())
484 t = String.(split(readline(),' '))
485 hour,minute = parse.(Int,split(t[2],':'))
486 NOW[] = (parse(Int,t[1])-1)*1440+60hour+minute
487 foreach(println,giveups!())
488 try
489 foreach(println,execute(t[3:end]))
490 catch exception
491 exception isa ArgumentError || rethrow()
492 println("ERROR ",exception.msg)
493 end
494 end
495end
