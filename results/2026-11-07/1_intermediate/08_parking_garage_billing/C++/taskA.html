26string timeText(int t) {
27 int hour = t%1440/60, minute = t%60;
28 return std::to_string(t/1440+1) + "-" + (hour < 10 ? "0" : "") + std::to_string(hour) + ":" + (minute < 10 ? "0" : "") + std::to_string(minute);
29}
30string duration(int t) { return std::to_string(t/60) + ":" + (t%60 < 10 ? "0" : "") + std::to_string(t%60); }
31string tenth(int q) { return std::to_string(q/10) + "." + std::to_string(q%10); }
32int lookup(const vector<string> &names, const string &s) {
33 vector<string>::const_iterator it = std::find(names.begin(),names.end(),s);
34 return it == names.end() ? -1 : int(it-names.begin());

36struct Garage {
37 int now = 0, peak = 0, peakTime = 0;
38 std::array<int,5> rates = {100,200,200,300,600}, caps = {800,1600,1600,2400,4800};
39 std::array<int,3> fees = {300,2000,30};
40 vector<Spot> spots;
41 vector<vector<int>> levels;
42 vector<Vehicle> vehicles;
43 vector<Waiting> queue;
44 std::map<string,int> parked, passes;
45 std::map<string,vector<Record>> histories;
46 std::array<Counts,5> totals{};
47 std::map<int,std::array<Counts,5>> daily;
48 std::array<Revenue,5> revenue{};

51 void line(const string &s) { output.push_back(s); }
52 bool error(bool condition, const string &reason) {
53 if (condition) line("ERROR " + reason);
54 return condition;
55 }
56 string label(int s) const { return "L" + std::to_string(spots[s].level) + "-" + std::to_string(spots[s].number); }
57 string location(const Vehicle &v) const { return label(v.spot) + (v.type == 4 ? ".." + std::to_string(spots[v.spot].number+2) : ""); }
58 bool free(int s) const { return !spots[s].closed && spots[s].vehicle < 0; }
59 bool validPass(const string &p) const {
60 std::map<string,int>::const_iterator found = passes.find(p);
61 return found != passes.end() && now/1440+1 <= found->second;
62 }
63 int spotId(const string &s) const {
64 std::size_t dash = s.find('-');
65 int level = std::stoi(s.substr(1,dash-1)), number = std::stoi(s.substr(dash+1));
66 if (level > int(levels.size()) || number > int(levels[level-1].size())) return -1;
67 return levels[level-1][number-1];
68 }
69 int vehicle(const string &p) {
70 std::map<string,int>::const_iterator found = parked.find(p);
71 if (error(found == parked.end(),p + " is not parked")) return -1;
72 return found->second;
73 }
74 int waiting(const string &p) const {
75 for (int i = 0; i < int(queue.size()); ++i) if (queue[i].plate == p) return i;
76 return -1;
77 }
78 int preference(int type, int kind) const {
79 return int(std::find(preferences[type].begin(),preferences[type].end(),kind)-preferences[type].begin());
80 }
81 int assign(int type) const {
82 if (type == 4) {
83 for (const vector<int> &level : levels) {
84 for (int j = 0; j+2 < int(level.size()); ++j) {
85 int a = level[j], b = level[j+1], c = level[j+2];
86 if (spots[a].kind == 2 && spots[b].kind == 2 && spots[c].kind == 2 && free(a) && free(b) && free(c)) return a;
87 }
88 }
89 } else {
90 for (int kind : preferences[type]) {
91 for (int i = 0; i < int(spots.size()); ++i) if (spots[i].kind == kind && free(i)) return i;
92 }
93 }
94 return -1;
95 }
96 void occupy(int id, int s) {
97 Vehicle &v = vehicles[id];
98 v.spot = s;
99 for (int i = 0; i < (v.type == 4 ? 3 : 1); ++i) spots[s+i].vehicle = id;
100 }
101 void vacate(int id) {
102 const Vehicle &v = vehicles[id];
103 for (int i = 0; i < (v.type == 4 ? 3 : 1); ++i) spots[v.spot+i].vehicle = -1;
104 }
105 int park(const string &plate, int type, int spot) {
106 int id = int(vehicles.size());
107 vehicles.push_back({plate,type,spot,now});
108 parked[plate] = id;
109 occupy(id,spot);
110 ++totals[type].entries;
111 ++daily[now/1440+1][type].entries;
112 if (int(parked.size()) > peak) {
113 peak = int(parked.size());
114 peakTime = now;
115 }
116 return id;
117 }
118 void refused(int type, int time) {
119 ++totals[type].refused;
120 ++daily[time/1440+1][type].refused;
121 }
122 void giveups() {
123 vector<Waiting> remaining;
124 for (const Waiting &w : queue) {
125 if (now-w.joined >= 30) {
126 line("GIVEUP " + w.plate + " " + types[w.type] + " " + timeText(w.joined+30));
127 refused(w.type,w.joined+30);
128 } else remaining.push_back(w);
129 }
130 queue.swap(remaining);
131 }
132 void processQueue() {
133 vector<Waiting> remaining;
134 for (const Waiting &w : queue) {
135 int s = assign(w.type);
136 if (s < 0) remaining.push_back(w);
137 else {
138 int id = park(w.plate,w.type,s);
139 line("ADMIT " + w.plate + " " + types[w.type] + " " + location(vehicles[id]) + " waited " + duration(now-w.joined));
140 }
141 }
142 queue.swap(remaining);
143 }
144 std::array<int,2> interval(const Vehicle &v, int from, int freeTime) const {
145 int length = now-from;
146 if (validPass(v.plate) || length <= freeTime) return {0,0};
147 int parking = 0;
148 for (int i = 0; i*1440 < length; ++i) {
149 int minutes = std::min(length-i*1440,1440)-(i == 0 ? freeTime : 0);
150 parking += std::min((minutes+29)/30*rates[v.type],caps[v.type]);
151 }
152 return {parking,(now/1440-from/1440)*fees[0]};
153 }
154 Bill bill(const Vehicle &v, bool loss) const {
155 std::array<int,2> amount = interval(v,v.paid ? v.payment : v.entry,v.paid ? 15 : 15+v.validation);
156 int p = amount[0]+(v.paid ? v.parking : 0), n = amount[1]+(v.paid ? v.night : 0);
157 int l = loss ? fees[1] : 0, total = p+n+l+v.charge;
158 return {p,n,l,v.charge,total,v.paid ? amount[0]+amount[1] : total};
159 }
160 void feeLine(const string &heading, const Vehicle &v, Bill b) {
161 line(heading + " " + v.plate + " " + location(v) + " stay " + duration(now-v.entry) + " parking " + std::to_string(b.parking) + " night " + std::to_string(b.night) + " lost " + std::to_string(b.lost) + " charge " + std::to_string(b.charge) + " total " + std::to_string(b.total) + " due " + std::to_string(b.due));
162 }
163 string annotation(const Vehicle &v) const {
164 string s;
165 if (validPass(v.plate)) s += " PASS";
166 if (v.validation) s += " VAL " + std::to_string(v.validation);
167 if (v.energy) s += " KWH " + std::to_string(v.energy);
168 if (v.paid) s += " PAID";
169 return s;
170 }
171 vector<int> orderedVehicles() const {
172 vector<int> ids;
173 for (const auto &[plate, id] : parked) ids.push_back(id);
174 std::sort(ids.begin(),ids.end(),[this](int a,int b) { return vehicles[a].spot < vehicles[b].spot; });
175 return ids;
176 }
177 void move(int id, int s) {
178 string before = location(vehicles[id]);
179 vacate(id);
180 occupy(id,s);
181 line("MOVE " + vehicles[id].plate + " " + before + " -> " + label(s));
182 }
183 void execute(const vector<string> &a) {
184 string cmd = a[0];
185 if (cmd == "LEVEL") {
186 vector<int> level;
187 int counts[4];
188 for (int k = 0; k < 4; ++k) {
189 counts[k] = std::stoi(a[k+1]);
190 for (int i = 0; i < counts[k]; ++i) {
191 level.push_back(int(spots.size()));
192 spots.push_back({int(levels.size())+1,int(level.size()),k});
193 }
194 }
195 levels.push_back(level);
196 line("LEVEL " + std::to_string(levels.size()) + " spots " + std::to_string(level.size()) + " MOTO " + a[1] + " COMPACT " + a[2] + " LARGE " + a[3] + " EV " + a[4]);
197 processQueue();
198 } else if (cmd == "CLOSE" || cmd == "OPEN") {
199 int s = spotId(a[1]);
200 if (error(s < 0,"unknown spot " + a[1])) return;
201 Spot &spot = spots[s];
202 if (cmd == "CLOSE") {
203 if (error(spot.closed,"spot " + a[1] + " is already closed")) return;
204 if (error(spot.vehicle >= 0,"spot " + a[1] + " is occupied by " + (spot.vehicle >= 0 ? vehicles[spot.vehicle].plate : ""))) return;
205 spot.closed = true;
206 } else {
207 if (error(!spot.closed,"spot " + a[1] + " is not closed")) return;
208 spot.closed = false;
209 }
210 line(cmd + " " + a[1] + " " + kindNames[spot.kind]);
211 if (cmd == "OPEN") processQueue();
212 } else if (cmd == "RATE") {
213 int type = lookup(types,a[1]);
214 if (error(type < 0,"unknown type " + a[1])) return;
215 int rate = std::stoi(a[2]), cap = std::stoi(a[3]);
216 if (error(cap < rate,"cap below rate for " + a[1])) return;
217 rates[type] = rate;
218 caps[type] = cap;
219 line("RATE " + a[1] + " " + a[2] + " " + a[3]);
220 } else if (cmd == "FEE") {
221 int item = lookup({"NIGHT","LOST","KWH"},a[1]);
222 if (error(item < 0,"unknown fee " + a[1])) return;
223 fees[item] = std::stoi(a[2]);
224 line("FEE " + a[1] + " " + a[2]);
225 } else if (cmd == "ENTER") {
226 int type = lookup(types,a[2]);
227 if (error(type < 0,"unknown type " + a[2])) return;
228 std::map<string,int>::const_iterator found = parked.find(a[1]);
229 if (error(found != parked.end(),a[1] + " is already parked at " + (found != parked.end() ? location(vehicles[found->second]) : ""))) return;
230 if (error(waiting(a[1]) >= 0,a[1] + " is already waiting")) return;
231 int s = assign(type);
232 if (s >= 0) {
233 int id = park(a[1],type,s);
234 line("ENTER " + a[1] + " " + a[2] + " " + location(vehicles[id]));
235 } else if (queue.size() < 3) {
236 queue.push_back({a[1],type,now});
237 line("WAIT " + a[1] + " " + a[2] + " " + std::to_string(queue.size()));
238 } else {
239 refused(type,now);
240 line("FULL " + a[1] + " " + a[2]);
241 }
242 } else if (cmd == "LEAVE") {
243 int i = waiting(a[1]);
244 if (error(i < 0,a[1] + " is not waiting")) return;
245 line("LEAVE " + a[1] + " " + types[queue[i].type]);
246 queue.erase(queue.begin()+i);
247 } else if (cmd == "EXIT" || cmd == "PAY" || cmd == "QUOTE" || cmd == "CHARGE" || cmd == "VALIDATE" || cmd == "MOVE") {
248 int id = vehicle(a[1]);
249 if (id < 0) return;
250 Vehicle &v = vehicles[id];
251 if (cmd == "EXIT") {
252 bool loss = a.size() == 3;
253 if (error(loss && v.paid,a[1] + " has already paid")) return;
254 Bill b = bill(v,loss);
255 feeLine("EXIT",v,b);
256 histories[v.plate].push_back({v.entry,now,v.type,b.total,location(v),loss});
257 Counts &all = totals[v.type], &today = daily[now/1440+1][v.type];
258 ++all.exits;
259 ++today.exits;
260 all.stay += now-v.entry;
261 all.revenue += b.total;
262 today.revenue += b.total;
263 Revenue &r = revenue[v.type];
264 ++r.exits;
265 r.parking += b.parking;
266 r.night += b.night;
267 r.lost += b.lost;
268 r.charge += b.charge;
269 r.total += b.total;
270 vacate(id);
271 parked.erase(v.plate);
272 processQueue();
273 } else if (cmd == "QUOTE") feeLine("QUOTE",v,bill(v,false));
274 else if (cmd == "PAY") {
275 if (error(v.paid,a[1] + " has already paid")) return;
276 Bill b = bill(v,false);
277 v.paid = true;
278 v.payment = now;
279 v.parking = b.parking;
280 v.night = b.night;
281 line("PAY " + v.plate + " parking " + std::to_string(b.parking) + " night " + std::to_string(b.night) + " charge " + std::to_string(b.charge) + " total " + std::to_string(b.total));
282 } else if (cmd == "CHARGE") {
283 if (error(v.type != 2,a[1] + " is not an electric car") || error(spots[v.spot].kind != 3,a[1] + " is not at a charger") || error(v.paid,a[1] + " has already paid")) return;
284 int energy = std::stoi(a[2]);
285 if (error(v.energy+energy > 100,"charge limit exceeded for " + a[1])) return;
286 int cost = energy*fees[2];
287 v.energy += energy;
288 v.charge += cost;
289 line("CHARGE " + v.plate + " kwh " + a[2] + " cost " + std::to_string(cost) + " stay_kwh " + std::to_string(v.energy) + " stay_cost " + std::to_string(v.charge));
290 } else if (cmd == "VALIDATE") {
291 if (error(v.paid,a[1] + " has already paid")) return;
292 int minutes = std::stoi(a[2]);
293 if (error(v.validation+minutes > 180,"validation limit exceeded for " + a[1])) return;
294 v.validation += minutes;
295 line("VALIDATE " + v.plate + " " + a[2] + " total " + std::to_string(v.validation));
296 } else {
297 if (error(v.type == 4,"bus " + a[1] + " cannot be moved")) return;
298 int s = spotId(a[2]);
299 if (error(s < 0,"unknown spot " + a[2]) || error(s == v.spot,a[1] + " is already at " + a[2])) return;
300 if (error(spots[s].closed,"spot " + a[2] + " is closed")) return;
301 if (error(spots[s].vehicle >= 0,"spot " + a[2] + " is occupied by " + (spots[s].vehicle >= 0 ? vehicles[spots[s].vehicle].plate : ""))) return;
302 bool usable = std::find(preferences[v.type].begin(),preferences[v.type].end(),spots[s].kind) != preferences[v.type].end();
303 if (error(!usable,types[v.type] + " cannot use " + kindNames[spots[s].kind] + " spot " + a[2])) return;
304 move(id,s);
305 processQueue();
306 }
307 } else if (cmd == "PASS") {
308 int days = std::stoi(a[2]);
309 passes[a[1]] = validPass(a[1]) ? passes[a[1]]+days : now/1440+days;
310 line("PASS " + a[1] + " until " + std::to_string(passes[a[1]]));
311 } else if (cmd == "UNPASS") {
312 if (error(!validPass(a[1]),a[1] + " has no pass")) return;
313 passes.erase(a[1]);
314 line("UNPASS " + a[1]);
315 } else if (cmd == "REBALANCE") {
316 vector<int> ids = orderedVehicles();
317 int moved = 0;
318 for (int id : ids) {
319 const Vehicle &v = vehicles[id];
320 if (v.type == 4) continue;
321 int current = preference(v.type,spots[v.spot].kind);
322 if (!current) continue;
323 int s = assign(v.type);
324 if (s >= 0 && preference(v.type,spots[s].kind) < current) {
325 move(id,s);
326 ++moved;
327 }
328 }
329 line("REBALANCE " + std::to_string(moved));
330 processQueue();
331 } else if (cmd == "WHERE") {
332 std::map<string,int>::const_iterator found = parked.find(a[1]);
333 int q = waiting(a[1]);
334 if (found != parked.end()) {
335 const Vehicle &v = vehicles[found->second];
336 line("WHERE " + a[1] + " " + location(v) + " " + types[v.type] + " " + timeText(v.entry) + " stay " + duration(now-v.entry) + annotation(v));
337 } else if (q >= 0) line("WHERE " + a[1] + " waiting " + std::to_string(q+1));
338 else if (histories.count(a[1])) line("WHERE " + a[1] + " gone " + timeText(histories[a[1]].back().exit));
339 else error(true,"unknown plate " + a[1]);
340 } else if (cmd == "LIST" || cmd == "OVERSTAY") {
341 int level = cmd == "LIST" && a.size() == 2 ? std::stoi(a[1]) : 0;
342 if (error(level > int(levels.size()),"unknown level " + std::to_string(level))) return;
343 vector<int> ids;
344 for (int id : orderedVehicles()) {
345 if (level && spots[vehicles[id].spot].level != level) continue;
346 if (cmd == "OVERSTAY" && now-vehicles[id].entry < std::stoi(a[1])*60) continue;
347 ids.push_back(id);
348 }
349 if (cmd == "OVERSTAY") std::stable_sort(ids.begin(),ids.end(),[this](int a,int b) { return vehicles[a].entry < vehicles[b].entry; });
350 line(cmd + (cmd == "OVERSTAY" ? " " + a[1] : "") + " " + std::to_string(ids.size()));
351 for (int id : ids) {
352 const Vehicle &v = vehicles[id];
353 line(location(v) + " " + v.plate + " " + types[v.type] + " " + timeText(v.entry) + (cmd == "LIST" ? annotation(v) : " stay " + duration(now-v.entry)));
354 }
355 } else if (cmd == "MAP") {
356 line("MAP " + std::to_string(levels.size()));
357 const string initial = "mcle";
358 for (int i = 0; i < int(levels.size()); ++i) {
359 string symbols;
360 for (int j = 0; j < int(levels[i].size()); ++j) {
361 if (j && j%10 == 0) symbols += ' ';
362 const Spot &s = spots[levels[i][j]];
363 char c = initial[s.kind];
364 if (s.closed) c = '#';
365 else if (s.vehicle >= 0) c += 'A'-'a';
366 symbols += c;
367 }
368 line("L" + std::to_string(i+1) + " " + symbols);
369 }
370 } else if (cmd == "OCCUPANCY") {
371 line("OCCUPANCY vehicles " + std::to_string(parked.size()) + " waiting " + std::to_string(queue.size()));
372 std::array<int,4> allUsed{}, allAvailable{};
373 int allClosed = 0;
374 for (int i = 0; i < int(levels.size()); ++i) {
375 std::array<int,4> used{}, available{};
376 int closed = 0;
377 for (int id : levels[i]) {
378 const Spot &s = spots[id];
379 if (s.closed) ++closed;
380 else ++available[s.kind];
381 if (s.vehicle >= 0) ++used[s.kind];
382 }
383 string s = "L" + std::to_string(i+1);
384 for (int k = 0; k < 4; ++k) {
385 s += " " + kindNames[k] + " " + std::to_string(used[k]) + "/" + std::to_string(available[k]);
386 allUsed[k] += used[k];
387 allAvailable[k] += available[k];
388 }
389 allClosed += closed;
390 line(s + " closed " + std::to_string(closed));
391 }
392 string s = "TOTAL";
393 int used = 0, available = 0;
394 for (int k = 0; k < 4; ++k) {
395 s += " " + kindNames[k] + " " + std::to_string(allUsed[k]) + "/" + std::to_string(allAvailable[k]);
396 used += allUsed[k];
397 available += allAvailable[k];
398 }
399 line(s + " closed " + std::to_string(allClosed) + " rate " + tenth(available ? used*1000/available : 0) + "%");
400 } else if (cmd == "SIGN") {
401 std::array<int,4> freeCounts{};
402 int buses = 0;
403 for (const vector<int> &level : levels) {
404 int run = 0;
405 for (int id : level) {
406 if (free(id)) ++freeCounts[spots[id].kind];
407 if (free(id) && spots[id].kind == 2) ++run;
408 else {
409 buses += run/3;
410 run = 0;
411 }
412 }
413 buses += run/3;
414 }
415 string s = "SIGN";
416 for (int k = 0; k < 5; ++k) {
417 int count = buses;
418 if (k != 4) {
419 count = 0;
420 for (int kind : preferences[k]) count += freeCounts[kind];
421 }
422 s += " " + types[k] + " " + (count ? std::to_string(count) : "FULL");
423 }
424 line(s);
425 } else if (cmd == "RATES") {
426 line("RATES");
427 for (int k = 0; k < 5; ++k) line(types[k] + " " + std::to_string(rates[k]) + " " + std::to_string(caps[k]));
428 line("NIGHT " + std::to_string(fees[0]) + " LOST " + std::to_string(fees[1]) + " KWH " + std::to_string(fees[2]));
429 } else if (cmd == "PASSES") {
430 int count = 0;
431 for (const auto &[plate, last] : passes) count += now/1440+1 <= last;
432 line("PASSES " + std::to_string(count));
433 for (const auto &[plate, last] : passes) if (now/1440+1 <= last) line(plate + " until " + std::to_string(last) + (parked.count(plate) ? " IN" : ""));
434 } else if (cmd == "HISTORY") {
435 std::map<string,vector<Record>>::const_iterator found = histories.find(a[1]);
436 if (error(found == histories.end(),"no history for " + a[1])) return;
437 int total = 0;
438 for (const Record &r : found->second) total += r.total;
439 line("HISTORY " + a[1] + " visits " + std::to_string(found->second.size()) + " paid " + std::to_string(total));
440 for (const Record &r : found->second) line(timeText(r.entry) + " " + timeText(r.exit) + " " + r.location + " " + types[r.type] + " " + std::to_string(r.total) + (r.lost ? " LOST" : ""));
441 } else if (cmd == "REVENUE") {
442 line("REVENUE");
443 Revenue all;
444 for (int k = 0; k <= 5; ++k) {
445 Revenue r = k == 5 ? all : revenue[k];
446 line((k == 5 ? "TOTAL" : types[k]) + " exits " + std::to_string(r.exits) + " parking " + std::to_string(r.parking) + " night " + std::to_string(r.night) + " lost " + std::to_string(r.lost) + " charge " + std::to_string(r.charge) + " total " + std::to_string(r.total));
447 if (k < 5) {
448 all.exits += r.exits;
449 all.parking += r.parking;
450 all.night += r.night;
451 all.lost += r.lost;
452 all.charge += r.charge;
453 all.total += r.total;
454 }
455 }
456 } else if (cmd == "DAILY" || cmd == "STATS") {
457 std::array<Counts,5> counts = cmd == "DAILY" ? daily[std::stoi(a[1])] : totals;
458 Counts all;
459 for (Counts c : counts) {
460 all.entries += c.entries;
461 all.exits += c.exits;
462 all.refused += c.refused;
463 all.revenue += c.revenue;
464 all.stay += c.stay;
