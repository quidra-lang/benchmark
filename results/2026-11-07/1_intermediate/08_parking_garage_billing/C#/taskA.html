1using System;
2using System.Collections.Generic;
3using System.Linq;
4using System.Globalization;
5
6class Program {
7 static readonly string[] TYPES={"MOTORCYCLE","CAR","ECAR","VAN","BUS"};
8 static readonly string[] KINDS={"MOTO","COMPACT","LARGE","EV"};
9 static readonly int[][] PREFS={new[]{0,1,2},new[]{1,2},new[]{3,1,2},new[]{2},new[]{2}};
10 class Fault : Exception { public Fault(string message) : base(message) {} }
11 static void error(bool condition, string message) { if (condition) throw new Fault(message); }
12 static string time(int n) => $"{n/1440+1}-{n/60%24:D2}:{n%60:D2}";
13 static string duration(int n) => $"{n/60}:{n%60:D2}";
14 static string tenths(int n) { return n/10+"."+n%10; }
15 class Spot {
16 public int level, number, kind, order;
17 public bool closed;
18 public Vehicle occupant;
19 public Spot(int level, int number, int kind, int order) {
20 this.level=level;
21 this.number=number;
22 this.kind=kind;
23 this.order=order;
24 }
25 public string id() { return "L"+level+"-"+number; }
26 public bool free() { return !closed && occupant==null; }
27 }
28 class Vehicle {
29 public string plate;
30 public int kind, entry, energy, charge, validation, paidAt=-1, paidParking, paidNight;
31 public List<Spot> spots;
32 public Vehicle(string plate, int kind, int entry, List<Spot> spots) {
33 this.plate=plate;
34 this.kind=kind;
35 this.entry=entry;
36 this.spots=spots;
37 }
38 public string location() { return spots[0].id()+(kind==4 ? ".."+spots[2].number : ""); }
39 }
40 record Waiting(string plate, int kind, int joined);
41 record Visit(int entry, int exit, string location, int kind, int total, bool loss);
42 class Garage {
43 public int now, peak, peakAt;
44 public List<List<Spot>> levels=new();
45 public List<Spot> spots=new();
46 public Dictionary<string,Vehicle> parked=new();
47 public List<Waiting> queue=new();
48 public SortedDictionary<string,int> passes=new(StringComparer.Ordinal);
49 public Dictionary<string,List<Visit>> history=new();
50 public int[][] rates={new[]{100,800},new[]{200,1600},new[]{200,1600},new[]{300,2400},new[]{600,4800}};
51 public int night=300, lost=2000, kwh=30;
52 public int[][] stats=Enumerable.Range(0,5).Select(_ => new int[9]).ToArray();
53 public Dictionary<int,int[][]> days=new();
54 public List<string> output;
55 public int[][] daily(int date) {
56 if (!days.TryGetValue(date,out int[][] data)) {
57 data=Enumerable.Range(0,5).Select(_ => new int[4]).ToArray();
58 days[date]=data;
59 }
60 return data;
61 }
62 List<Visit> visitsFor(string plate) {
63 if (!history.TryGetValue(plate,out List<Visit> result)) {
64 result=new();
65 history[plate]=result;
66 }
67 return result;
68 }
69 public bool pass(string plate) { return passes.GetValueOrDefault(plate,0)>=now/1440+1; }
70 public Vehicle vehicle(string plate) {
71 Vehicle v=parked.GetValueOrDefault(plate);
72 error(v==null,plate+" is not parked");
73 return v;
74 }
75 public Spot spot(string id) {
76 string[] p=id.Substring(1).Split('-');
77 int l=int.Parse(p[0]), n=int.Parse(p[1]);
78 error(l>levels.Count || n>levels[l-1].Count,"unknown spot "+id);
79 return levels[l-1][n-1];
80 }
81 public List<Spot> assign(int kind) {
82 if (kind==4) {
83 foreach (List<Spot> level in levels) {
84 for (int i=0;i+2<level.Count;i++) {
85 List<Spot> triple=level.GetRange(i,3);
86 if (triple.All(s => s.kind==2 && s.free())) return triple;
87 }
88 }
89 } else {
90 foreach (int wanted in PREFS[kind]) {
91 Spot first=spots.FirstOrDefault(s => s.kind==wanted && s.free());
92 if (first!=null) return new List<Spot>{first};
93 }
94 }
95 return null;
96 }
97 public void occupy(Vehicle v) { foreach (Spot s in v.spots) s.occupant=v; }
98 public void free(Vehicle v) { foreach (Spot s in v.spots) s.occupant=null; }
99 public Vehicle park(string plate, int kind, List<Spot> assigned) {
100 Vehicle v=new Vehicle(plate,kind,now,assigned);
101 parked[plate]=v;
102 occupy(v);
103 stats[kind][0]++;
104 daily(now/1440+1)[kind][0]++;
105 if (parked.Count>peak) {
106 peak=parked.Count;
107 peakAt=now;
108 }
109 return v;
110 }
111 public void refuse(int kind, int at) {
112 stats[kind][2]++;
113 daily(at/1440+1)[kind][2]++;
114 }
115 public void giveup() {
116 foreach (Waiting w in queue.ToList()) {
117 if (now-w.joined>=30) {
118 queue.Remove(w);
119 refuse(w.kind,w.joined+30);
120 output.Add("GIVEUP "+w.plate+" "+TYPES[w.kind]+" "+time(w.joined+30));
121 }
122 }
123 }
124 public void admit() {
125 foreach (Waiting w in queue.ToList()) {
126 List<Spot> assigned=assign(w.kind);
127 if (assigned!=null) {
128 queue.Remove(w);
129 Vehicle v=park(w.plate,w.kind,assigned);
130 output.Add("ADMIT "+v.plate+" "+TYPES[v.kind]+" "+v.location()+" waited "+duration(now-w.joined));
131 }
132 }
133 }
134 public int[] interval(Vehicle v, int start, int free) {
135 int total=now-start;
136 if (pass(v.plate) || total<=free) return new int[]{0,0};
137 int parking=0;
138 for (int offset=0;offset<total;offset+=1440) {
139 int billable=Math.Min(total-offset,1440)-(offset==0 ? free : 0);
140 parking+=Math.Min((billable+29)/30*rates[v.kind][0],rates[v.kind][1]);
141 }
142 return new int[]{parking,(now/1440-start/1440)*night};
143 }
144 public int[] fees(Vehicle v, bool loss) {
145 int[] pair=interval(v,v.paidAt<0 ? v.entry : v.paidAt,v.paidAt<0 ? 15+v.validation : 15);
146 int parking=pair[0]+(v.paidAt<0 ? 0 : v.paidParking);
147 int surcharge=pair[1]+(v.paidAt<0 ? 0 : v.paidNight);
148 int ticket=loss ? lost : 0;
149 int total=parking+surcharge+ticket+v.charge;
150 return new int[]{parking,surcharge,ticket,v.charge,total,v.paidAt<0 ? total : pair[0]+pair[1]};
151 }
152 public string feeLine(string label, Vehicle v, int[] f) {
153 return label+" "+v.plate+" "+v.location()+" stay "+duration(now-v.entry)+" parking "+f[0]+
154 " night "+f[1]+" lost "+f[2]+" charge "+f[3]+" total "+f[4]+" due "+f[5];
155 }
156 public string annotation(Vehicle v) {
157 return (pass(v.plate) ? " PASS" : "")+(v.validation>0 ? " VAL "+v.validation : "")+
158 (v.energy>0 ? " KWH "+v.energy : "")+(v.paidAt>=0 ? " PAID" : "");
159 }
160 public List<Vehicle> ordered() {
161 return parked.Values.OrderBy(v => v.spots[0].order).ToList();
162 }
163 public void move(Vehicle v, List<Spot> target) {
164 string old=v.location();
165 free(v);
166 v.spots=target;
167 occupy(v);
168 output.Add("MOVE "+v.plate+" "+old+" -> "+v.location());
169 }
170 public int[] occupancy(List<Spot> target) {
171 int[] result=new int[9];
172 foreach (Spot s in target) {
173 if (s.closed) result[8]++;
174 else result[4+s.kind]++;
175 if (s.occupant!=null) result[s.kind]++;
176 }
177 return result;
178 }
179 public string occupancyLine(string label, int[] a) {
180 string result=label;
181 for (int k=0;k<4;k++) result+=" "+KINDS[k]+" "+a[k]+"/"+a[k+4];
182 return result+" closed "+a[8];
183 }
184 public string revenueLine(string label, int[] a) {
185 return label+" exits "+a[1]+" parking "+a[4]+" night "+a[5]+" lost "+a[6]+" charge "+a[7]+" total "+a[8];
186 }
187 public string statsLine(string label, int[] a) {
188 return label+" entries "+a[0]+" exits "+a[1]+" refused "+a[2]+" avg_stay "+tenths(a[1]==0 ? 0 : a[3]*10/a[1]);
189 }
190 public string dailyLine(string label, int[] a) { return label+" entries "+a[0]+" exits "+a[1]+" refused "+a[2]+" revenue "+a[3]; }
191 public int[] sum(int[][] a, int width) => Enumerable.Range(0,width).Select(i => a.Sum(row => row[i])).ToArray();
192 public void execute(string[] t) {
193 bool trigger=false;
194 string command=t[2];
195 switch (command) {
196 case "LEVEL": {
197 List<Spot> added=new();
198 int[] counts=new int[4];
199 for (int k=0;k<4;k++) {
200 counts[k]=int.Parse(t[k+3]);
201 for (int i=0;i<counts[k];i++) {
202 Spot s=new Spot(levels.Count+1,added.Count+1,k,spots.Count);
203 added.Add(s);
204 spots.Add(s);
205 }
206 }
207 levels.Add(added);
208 output.Add("LEVEL "+levels.Count+" spots "+added.Count+" MOTO "+counts[0]+" COMPACT "+counts[1]+" LARGE "+counts[2]+" EV "+counts[3]);
209 trigger=true;
210 break;
211 }
212 case "CLOSE":
213 case "OPEN": {
214 Spot s=spot(t[3]);
215 if (command.Equals("CLOSE")) {
216 error(s.closed,"spot "+t[3]+" is already closed");
217 error(s.occupant!=null,"spot "+t[3]+" is occupied by "+(s.occupant==null ? "" : s.occupant.plate));
218 s.closed=true;
219 } else {
220 error(!s.closed,"spot "+t[3]+" is not closed");
221 s.closed=false;
222 trigger=true;
223 }
224 output.Add(command+" "+s.id()+" "+KINDS[s.kind]);
225 break;
226 }
227 case "RATE": {
228 int k=Array.IndexOf(TYPES,t[3]);
229 error(k<0,"unknown type "+t[3]);
230 int rate=int.Parse(t[4]), cap=int.Parse(t[5]);
231 error(cap<rate,"cap below rate for "+t[3]);
232 rates[k]=new int[]{rate,cap};
233 output.Add("RATE "+t[3]+" "+rate+" "+cap);
234 break;
235 }
236 case "FEE": {
237 int amount=int.Parse(t[4]);
238 error(!new[]{"NIGHT","LOST","KWH"}.Contains(t[3]),"unknown fee "+t[3]);
239 switch (t[3]) {
240 case "NIGHT":
241 night=amount;
242 break;
243 case "LOST":
244 lost=amount;
245 break;
246 default:
247 kwh=amount;
248 break;
249 }
250 output.Add("FEE "+t[3]+" "+amount);
251 break;
252 }
253 case "ENTER": {
254 string plate=t[3];
255 int k=Array.IndexOf(TYPES,t[4]);
256 error(k<0,"unknown type "+t[4]);
257 Vehicle existing=parked.GetValueOrDefault(plate);
258 error(existing!=null,plate+" is already parked at "+(existing==null ? "" : existing.location()));
259 error(queue.Any(w => w.plate.Equals(plate)),plate+" is already waiting");
260 List<Spot> assigned=assign(k);
261 if (assigned!=null) output.Add("ENTER "+plate+" "+TYPES[k]+" "+park(plate,k,assigned).location());
262 else if (queue.Count<3) {
263 queue.Add(new Waiting(plate,k,now));
264 output.Add("WAIT "+plate+" "+TYPES[k]+" "+queue.Count);
265 } else {
266 refuse(k,now);
267 output.Add("FULL "+plate+" "+TYPES[k]);
268 }
269 break;
270 }
271 case "LEAVE": {
272 Waiting w=queue.FirstOrDefault(x => x.plate.Equals(t[3]));
273 error(w==null,t[3]+" is not waiting");
274 queue.Remove(w);
275 output.Add("LEAVE "+w.plate+" "+TYPES[w.kind]);
276 break;
277 }
278 case "EXIT":
279 case "PAY":
280 case "QUOTE": {
281 Vehicle v=vehicle(t[3]);
282 bool loss=t.Length==5;
283 error(v.paidAt>=0 && (command.Equals("PAY") || loss),v.plate+" has already paid");
284 int[] f=fees(v,loss);
285 if (command.Equals("PAY")) {
286 v.paidAt=now;
287 v.paidParking=f[0];
288 v.paidNight=f[1];
289 output.Add("PAY "+v.plate+" parking "+f[0]+" night "+f[1]+" charge "+f[3]+" total "+f[4]);
290 } else {
291 output.Add(feeLine(command,v,f));
292 if (command.Equals("EXIT")) {
293 visitsFor(v.plate).Add(new Visit(v.entry,now,v.location(),v.kind,f[4],loss));
294 stats[v.kind][1]++;
295 stats[v.kind][3]+=now-v.entry;
296 for (int i=0;i<5;i++) stats[v.kind][i+4]+=f[i];
297 int[] d=daily(now/1440+1)[v.kind];
298 d[1]++;
299 d[3]+=f[4];
300 free(v);
301 parked.Remove(v.plate);
302 trigger=true;
303 }
304 }
305 break;
306 }
307 case "CHARGE": {
308 Vehicle v=vehicle(t[3]);
309 error(v.kind!=2,v.plate+" is not an electric car");
310 error(v.spots[0].kind!=3,v.plate+" is not at a charger");
311 error(v.paidAt>=0,v.plate+" has already paid");
312 int energy=int.Parse(t[4]);
313 error(v.energy+energy>100,"charge limit exceeded for "+v.plate);
314 int cost=energy*kwh;
315 v.energy+=energy;
316 v.charge+=cost;
317 output.Add("CHARGE "+v.plate+" kwh "+energy+" cost "+cost+" stay_kwh "+v.energy+" stay_cost "+v.charge);
318 break;
319 }
320 case "VALIDATE": {
321 Vehicle v=vehicle(t[3]);
322 error(v.paidAt>=0,v.plate+" has already paid");
323 int minutes=int.Parse(t[4]);
324 error(v.validation+minutes>180,"validation limit exceeded for "+v.plate);
325 v.validation+=minutes;
326 output.Add("VALIDATE "+v.plate+" "+minutes+" total "+v.validation);
327 break;
328 }
329 case "PASS": {
330 int end=(pass(t[3]) ? passes[t[3]] : now/1440)+int.Parse(t[4]);
331 passes[t[3]]=end;
332 output.Add("PASS "+t[3]+" until "+end);
333 break;
334 }
335 case "UNPASS": {
336 error(!pass(t[3]),t[3]+" has no pass");
337 passes.Remove(t[3]);
338 output.Add("UNPASS "+t[3]);
339 break;
340 }
341 case "MOVE": {
342 Vehicle v=vehicle(t[3]);
343 error(v.kind==4,"bus "+v.plate+" cannot be moved");
344 Spot s=spot(t[4]);
345 error(v.spots[0]==s,v.plate+" is already at "+t[4]);
346 error(s.closed,"spot "+t[4]+" is closed");
347 error(s.occupant!=null,"spot "+t[4]+" is occupied by "+(s.occupant==null ? "" : s.occupant.plate));
348 error(!PREFS[v.kind].Contains(s.kind),TYPES[v.kind]+" cannot use "+KINDS[s.kind]+" spot "+t[4]);
349 move(v,new List<Spot>{s});
350 trigger=true;
351 break;
352 }
353 case "REBALANCE": {
354 int moved=0;
355 foreach (Vehicle v in ordered()) {
356 if (v.kind==4) continue;
357 List<Spot> target=assign(v.kind);
358 if (target!=null && Array.IndexOf(PREFS[v.kind],target[0].kind)<
359 Array.IndexOf(PREFS[v.kind],v.spots[0].kind)) {
360 move(v,target);
361 moved++;
362 }
363 }
364 output.Add("REBALANCE "+moved);
365 trigger=true;
366 break;
367 }
368 case "WHERE": {
369 Vehicle v=parked.GetValueOrDefault(t[3]);
370 int p=queue.FindIndex(w => w.plate == t[3]);
371 List<Visit> visits=history.GetValueOrDefault(t[3]);
372 error(v==null && p<0 && visits==null,"unknown plate "+t[3]);
373 if (v!=null) output.Add("WHERE "+v.plate+" "+v.location()+" "+TYPES[v.kind]+" "+time(v.entry)+" stay "+duration(now-v.entry)+annotation(v));
374 else if (p>=0) output.Add("WHERE "+t[3]+" waiting "+(p+1));
375 else output.Add("WHERE "+t[3]+" gone "+time(visits[visits.Count-1].exit));
376 break;
377 }
378 case "LIST": {
379 int l=t.Length==4 ? int.Parse(t[3]) : 0;
380 error(l>levels.Count,"unknown level "+l);
381 List<Vehicle> found=ordered().Where(v => l==0 || v.spots[0].level==l).ToList();
382 output.Add("LIST "+found.Count);
383 foreach (Vehicle v in found) output.Add(v.location()+" "+v.plate+" "+TYPES[v.kind]+" "+time(v.entry)+annotation(v));
384 break;
385 }
386 case "MAP": {
387 output.Add("MAP "+levels.Count);
388 for (int i=0;i<levels.Count;i++) {
389 var symbols=levels[i].Select(s => s.closed ? '#' : s.occupant!=null ? KINDS[s.kind][0] : char.ToLowerInvariant(KINDS[s.kind][0]));
390 var chunks=symbols.Chunk(10).Select(chars => new string(chars));
391 output.Add("L"+(i+1)+" "+string.Join(" ",chunks));
392 }
393 break;
394 }
395 case "OCCUPANCY": {
396 output.Add("OCCUPANCY vehicles "+parked.Count+" waiting "+queue.Count);
397 for (int i=0;i<levels.Count;i++) output.Add(occupancyLine("L"+(i+1),occupancy(levels[i])));
398 int[] a=occupancy(spots);
399 int used=a.Take(4).Sum(), available=a.Skip(4).Take(4).Sum();
400 output.Add(occupancyLine("TOTAL",a)+" rate "+tenths(available==0 ? 0 : used*1000/available)+"%");
401 break;
402 }
403 case "SIGN": {
404 string line="SIGN";
405 for (int k=0;k<5;k++) {
406 int count=0;
407 if (k==4) {
408 foreach (List<Spot> level in levels) {
409 int run=0;
410 foreach (Spot s in level) {
411 if (s.kind==2 && s.free()) {
412 run++;
413 if (run==3) {
414 count++;
415 run=0;
416 }
417 } else run=0;
418 }
419 }
420 } else {
421 count=spots.Count(s => PREFS[k].Contains(s.kind) && s.free());
422 }
423 line+=" "+TYPES[k]+" "+(count==0 ? "FULL" : count.ToString());
424 }
425 output.Add(line);
426 break;
427 }
428 case "RATES": {
429 output.Add("RATES");
430 for (int k=0;k<5;k++) output.Add(TYPES[k]+" "+rates[k][0]+" "+rates[k][1]);
431 output.Add("NIGHT "+night+" LOST "+lost+" KWH "+kwh);
432 break;
433 }
434 case "PASSES": {
435 List<string> found=passes.Keys.Where(pass).ToList();
436 output.Add("PASSES "+found.Count);
437 foreach (string plate in found) output.Add(plate+" until "+passes[plate]+(parked.ContainsKey(plate) ? " IN" : ""));
438 break;
439 }
440 case "OVERSTAY": {
441 int hours=int.Parse(t[3]);
442 List<Vehicle> found=ordered().Where(v => now-v.entry>=hours*60).OrderBy(v => v.entry).ToList();
443 output.Add("OVERSTAY "+hours+" "+found.Count);
444 foreach (Vehicle v in found) output.Add(v.location()+" "+v.plate+" "+TYPES[v.kind]+" "+time(v.entry)+" stay "+duration(now-v.entry));
445 break;
446 }
447 case "HISTORY": {
448 List<Visit> visits=history.GetValueOrDefault(t[3]);
449 error(visits==null,"no history for "+t[3]);
450 output.Add("HISTORY "+t[3]+" visits "+visits.Count+" paid "+visits.Sum(v => v.total));
451 foreach (Visit v in visits) output.Add(time(v.entry)+" "+time(v.exit)+" "+v.location+" "+TYPES[v.kind]+" "+v.total+(v.loss ? " LOST" : ""));
452 break;
453 }
454 case "REVENUE": {
455 output.Add("REVENUE");
456 for (int k=0;k<5;k++) output.Add(revenueLine(TYPES[k],stats[k]));
457 output.Add(revenueLine("TOTAL",sum(stats,9)));
458 break;
459 }
460 case "DAILY": {
461 int d=int.Parse(t[3]);
462 int[][] a=daily(d);
463 output.Add(dailyLine("DAILY "+d,sum(a,4)));
464 for (int k=0;k<5;k++) output.Add(dailyLine(TYPES[k],a[k]));
465 break;
466 }
467 case "STATS": {
468 output.Add("STATS parked "+parked.Count+" peak "+peak+" at "+(peak==0 ? "-" : time(peakAt)));
469 for (int k=0;k<5;k++) output.Add(statsLine(TYPES[k],stats[k]));
470 output.Add(statsLine("ALL",sum(stats,9)));
471 break;
472 }
473 }
474 if (trigger) admit();
475 }
476 public List<string> process(string line) {
477 string[] t=line.Split(' '), hm=t[1].Split(':');
478 now=(int.Parse(t[0])-1)*1440+int.Parse(hm[0])*60+int.Parse(hm[1]);
479 output=new();
480 giveup();
481 try { execute(t); }
482 catch (Fault f) { output.Add("ERROR "+f.Message); }
483 return output;
484 }
485 }
486 static void Main() {
487 CultureInfo.CurrentCulture=CultureInfo.InvariantCulture;
488 Garage garage=new();
489 int n=int.Parse(Console.ReadLine()!);
490 for (int i=0;i<n;i++) {
491 foreach (string line in garage.process(Console.ReadLine()!)) Console.WriteLine(line);
492 }
493 }
