3type Spot = {level: number; number: number; kind: string; code: string; closed: boolean; vehicle: Vehicle | null};
4type Vehicle = {plate: string; kind: string; spots: Spot[]; entered: number; energy: number; chargeCost: number; validation: number; payment: number[] | null};
5type Waiting = {plate: string; kind: string; joined: number};
6type ExitRecord = {entered: number; exited: number; location: string; kind: string; total: number; lost: boolean};
7type Totals = Record<string, number>;
8type Charges = {parking: number; night: number; lost: number; charge: number; total: number; due: number};
9const TYPES = ["MOTORCYCLE", "CAR", "ECAR", "VAN", "BUS"];
10const KINDS = ["MOTO", "COMPACT", "LARGE", "EV"];
11const PREF: Record<string, string[]> = {
12 MOTORCYCLE: ["MOTO", "COMPACT", "LARGE"],
13 CAR: ["COMPACT", "LARGE"],
14 ECAR: ["EV", "COMPACT", "LARGE"],
15 VAN: ["LARGE"]
16};
17const FIELDS = ["entries", "exits", "refused", "stay", "parking", "night", "lost", "charge", "total"];
18const dateOf = (t: number): number => Math.floor(t / 1440) + 1;
19const duration = (m: number): string => Math.floor(m / 60) + ":" + String(m % 60).padStart(2, "0");
20const stamp = (t: number): string => dateOf(t) + "-" + String(Math.floor(t % 1440 / 60)).padStart(2, "0") + ":" + String(t % 60).padStart(2, "0");
21const emptyTotals = (): Totals => Object.fromEntries(FIELDS.map(f => [f, 0]));
22const emptyTypes = (): Record<string, Totals> => Object.fromEntries(TYPES.map(k => [k, emptyTotals()]));
23const sumTotals = (rows: Totals[]): Totals => Object.fromEntries(FIELDS.map(f => [f, rows.reduce((a, r) => a + r[f], 0)]));
24const free = (s: Spot): boolean => !s.closed && s.vehicle === null;
25const location = (v: Vehicle): string => v.spots[0].code + (v.kind === "BUS" ? ".." + v.spots[2].number : "");
26const order = (a: Vehicle, b: Vehicle): number => a.spots[0].level - b.spots[0].level || a.spots[0].number - b.spots[0].number;
27class CommandError extends Error {}

29class Garage {
30 now = 0;
31 levels: Spot[][] = [];
32 spots: Spot[] = [];
33 spotCodes = new Map<string, Spot>();
34 vehicles = new Map<string, Vehicle>();
35 queue: Waiting[] = [];
36 passes = new Map<string, number>();
37 history = new Map<string, ExitRecord[]>();
38 rates: Record<string, number[]> = {MOTORCYCLE: [100, 800], CAR: [200, 1600], ECAR: [200, 1600], VAN: [300, 2400], BUS: [600, 4800]};
39 fees: Record<string, number> = {NIGHT: 300, LOST: 2000, KWH: 30};
40 totals = emptyTypes();
41 daily = new Map<number, Record<string, Totals>>();
42 peak = 0;
43 peakTime: number | null = null;
44 output: string[] = [];

47 this.output.push(parts.join(" "));

80 assignment(kind: string): Spot[] | null {
81 if (kind === "BUS") {
82 for (const level of this.levels) {
83 for (let i = 0; i + 2 < level.length; i++) {
84 const triple = level.slice(i, i + 3);
85 if (triple.every(s => s.kind === "LARGE" && free(s))) return triple;
86 }
87 }
88 } else {
89 for (const preferred of PREF[kind]) {
90 const s = this.spots.find(s => s.kind === preferred && free(s));
91 if (s) return [s];
92 }
93 }
94 return null;
95 }

97 park(plate: string, kind: string, spots: Spot[]): Vehicle {
98 const v: Vehicle = {plate, kind, spots, entered: this.now, energy: 0, chargeCost: 0, validation: 0, payment: null};
99 this.vehicles.set(plate, v);
100 for (const s of spots) s.vehicle = v;
101 this.count(kind, "entries", dateOf(this.now));
102 if (this.vehicles.size > this.peak) {
103 this.peak = this.vehicles.size;
104 this.peakTime = this.now;
105 }
106 return v;
107 }

109 giveUp(): void {
110 const remaining: Waiting[] = [];
111 for (const w of this.queue) {
112 const deadline = w.joined + 30;
113 if (deadline <= this.now) {
114 this.emit("GIVEUP", w.plate, w.kind, stamp(deadline));
115 this.count(w.kind, "refused", dateOf(deadline));
116 } else {
117 remaining.push(w);
118 }
119 }
120 this.queue = remaining;
121 }
122
123 processQueue(): void {
124 const remaining: Waiting[] = [];
125 for (const w of this.queue) {
126 const spots = this.assignment(w.kind);
127 if (spots === null) {
128 remaining.push(w);
129 } else {
130 const v = this.park(w.plate, w.kind, spots);
131 this.emit("ADMIT", w.plate, w.kind, location(v), "waited", duration(this.now - w.joined));
132 }
133 }
134 this.queue = remaining;
135 }

137 interval(v: Vehicle, start: number, end: number, grace: number): number[] {
138 const stay = end - start;
139 if (this.hasPass(v.plate, end) || stay <= grace) return [0, 0];
140 const [rate, cap] = this.rates[v.kind];
141 let parking = 0;
142 for (let offset = 0; offset < stay; offset += 1440) {
143 const billable = Math.min(stay - offset, 1440) - (offset === 0 ? grace : 0);
144 parking += Math.min(Math.floor((billable + 29) / 30) * rate, cap);
145 }
146 return [parking, (dateOf(end) - dateOf(start)) * this.fees.NIGHT];
147 }
148
149 charges(v: Vehicle, lost = false): Charges {
150 let parking: number;
151 let night: number;
152 let due: number;
153 const lostFee = lost ? this.fees.LOST : 0;
154 if (v.payment === null) {
155 [parking, night] = this.interval(v, v.entered, this.now, 15 + v.validation);
156 due = parking + night + lostFee + v.chargeCost;
157 } else {
158 const [paidTime, paidParking, paidNight] = v.payment;
159 const [additional, extraNight] = this.interval(v, paidTime, this.now, 15);
160 parking = paidParking + additional;
161 night = paidNight + extraNight;
162 due = additional + extraNight;
163 }
164 return {parking, night, lost: lostFee, charge: v.chargeCost, total: parking + night + lostFee + v.chargeCost, due};
165 }

168 this.emit(label, v.plate, location(v), "stay", duration(this.now - v.entered),
169 "parking", c.parking, "night", c.night, "lost", c.lost, "charge", c.charge, "total", c.total, "due", c.due);

172 annotation(v: Vehicle): string {
173 const parts: (string | number)[] = [];
174 if (this.hasPass(v.plate)) parts.push("PASS");
175 if (v.validation) parts.push("VAL", v.validation);
176 if (v.energy) parts.push("KWH", v.energy);
177 if (v.payment !== null) parts.push("PAID");
178 return parts.length ? " " + parts.join(" ") : "";

182 const old = location(v);
183 v.spots[0].vehicle = null;
184 v.spots = [s];
185 s.vehicle = v;
186 this.emit("MOVE", v.plate, old, "->", s.code);

190 const parts: (string | number)[] = [label, "exits", t.exits];
191 for (const f of ["parking", "night", "lost", "charge", "total"]) parts.push(f, t[f]);
192 this.emit(...parts);

199 statsLine(label: string, t: Totals): void {
200 const q = t.exits ? Math.floor(t.stay * 10 / t.exits) : 0;
201 this.emit(label, "entries", t.entries, "exits", t.exits, "refused", t.refused, "avg_stay", Math.floor(q / 10) + "." + q % 10);
202 }

204 command(cmd: string, a: string[]): void {
205 const plate = a[0];
206 switch (cmd) {
207 case "LEVEL": {
208 const counts = a.map(Number);
209 const level: Spot[] = [];
210 const index = this.levels.length + 1;
211 for (let k = 0; k < KINDS.length; k++) {
212 for (let j = 0; j < counts[k]; j++) {
213 const number = level.length + 1;
214 const s: Spot = {level: index, number, kind: KINDS[k], code: "L" + index + "-" + number, closed: false, vehicle: null};
215 level.push(s);
216 this.spots.push(s);
217 this.spotCodes.set(s.code, s);
218 }
219 }
220 this.levels.push(level);
221 this.emit("LEVEL", index, "spots", level.length, "MOTO", counts[0], "COMPACT", counts[1], "LARGE", counts[2], "EV", counts[3]);
222 this.processQueue();
223 break;
224 }
225 case "CLOSE": {
226 const s = this.spot(a[0]);
227 if (s.closed) throw new CommandError("spot " + s.code + " is already closed");
228 if (s.vehicle) throw new CommandError("spot " + s.code + " is occupied by " + s.vehicle.plate);
229 s.closed = true;
230 this.emit("CLOSE", s.code, s.kind);
231 break;
232 }
233 case "OPEN": {
234 const s = this.spot(a[0]);
235 if (!s.closed) throw new CommandError("spot " + s.code + " is not closed");
236 s.closed = false;
237 this.emit("OPEN", s.code, s.kind);
238 this.processQueue();
239 break;
240 }
241 case "RATE": {
242 if (!TYPES.includes(a[0])) throw new CommandError("unknown type " + a[0]);
243 const rate = Number(a[1]);
244 const cap = Number(a[2]);
245 if (cap < rate) throw new CommandError("cap below rate for " + a[0]);
246 this.rates[a[0]] = [rate, cap];
247 this.emit("RATE", a[0], rate, cap);
248 break;
249 }
250 case "FEE": {
251 if (!(a[0] in this.fees)) throw new CommandError("unknown fee " + a[0]);
252 this.fees[a[0]] = Number(a[1]);
253 this.emit("FEE", a[0], Number(a[1]));
254 break;
255 }
256 case "ENTER": {
257 const kind = a[1];
258 if (!TYPES.includes(kind)) throw new CommandError("unknown type " + kind);
259 if (this.vehicles.has(plate)) throw new CommandError(plate + " is already parked at " + location(this.vehicles.get(plate)!));
260 if (this.queue.some(w => w.plate === plate)) throw new CommandError(plate + " is already waiting");
261 const spots = this.assignment(kind);
262 if (spots !== null) {
263 const v = this.park(plate, kind, spots);
264 this.emit("ENTER", plate, kind, location(v));
265 } else if (this.queue.length < 3) {
266 this.queue.push({plate, kind, joined: this.now});
267 this.emit("WAIT", plate, kind, this.queue.length);
268 } else {
269 this.emit("FULL", plate, kind);
270 this.count(kind, "refused", dateOf(this.now));
271 }
272 break;
273 }
274 case "LEAVE": {
275 const i = this.queue.findIndex(w => w.plate === plate);
276 if (i < 0) throw new CommandError(plate + " is not waiting");
277 const [w] = this.queue.splice(i, 1);
278 this.emit("LEAVE", plate, w.kind);
279 break;
280 }
281 case "EXIT": {
282 const v = this.vehicle(plate);
283 const lost = a.length === 2;
284 if (lost) this.unpaid(v);
285 const c = this.charges(v, lost);
286 this.feeLine("EXIT", v, c);
287 if (!this.history.has(plate)) this.history.set(plate, []);
288 this.history.get(plate)!.push({entered: v.entered, exited: this.now, location: location(v), kind: v.kind, total: c.total, lost});
289 this.count(v.kind, "exits", dateOf(this.now));
290 const daily = this.dayTotals(dateOf(this.now))[v.kind];
291 for (const f of ["parking", "night", "lost", "charge", "total"]) {
292 this.totals[v.kind][f] += c[f as keyof Charges];
293 daily[f] += c[f as keyof Charges];
294 }
295 this.totals[v.kind].stay += this.now - v.entered;
296 daily.stay += this.now - v.entered;
297 for (const s of v.spots) s.vehicle = null;
298 this.vehicles.delete(plate);
299 this.processQueue();
300 break;
301 }
302 case "PAY": {
303 const v = this.vehicle(plate);
304 this.unpaid(v);
305 const c = this.charges(v);
306 v.payment = [this.now, c.parking, c.night];
307 this.emit("PAY", plate, "parking", c.parking, "night", c.night, "charge", c.charge, "total", c.total);
308 break;
309 }
310 case "QUOTE": {
311 const v = this.vehicle(plate);
312 this.feeLine("QUOTE", v, this.charges(v));
313 break;
314 }
315 case "CHARGE": {
316 const v = this.vehicle(plate);
317 const kwh = Number(a[1]);
318 if (v.kind !== "ECAR") throw new CommandError(plate + " is not an electric car");
319 if (v.spots[0].kind !== "EV") throw new CommandError(plate + " is not at a charger");
320 this.unpaid(v);
321 if (v.energy + kwh > 100) throw new CommandError("charge limit exceeded for " + plate);
322 const cost = kwh * this.fees.KWH;
323 v.energy += kwh;
324 v.chargeCost += cost;
325 this.emit("CHARGE", plate, "kwh", kwh, "cost", cost, "stay_kwh", v.energy, "stay_cost", v.chargeCost);
326 break;
327 }
328 case "VALIDATE": {
329 const v = this.vehicle(plate);
330 this.unpaid(v);
331 const minutes = Number(a[1]);
332 if (v.validation + minutes > 180) throw new CommandError("validation limit exceeded for " + plate);
333 v.validation += minutes;
334 this.emit("VALIDATE", plate, minutes, "total", v.validation);
335 break;
336 }
337 case "PASS": {
338 const last = this.hasPass(plate) ? this.passes.get(plate)! : dateOf(this.now) - 1;
339 this.passes.set(plate, last + Number(a[1]));
340 this.emit("PASS", plate, "until", this.passes.get(plate)!);
341 break;
342 }
343 case "UNPASS": {
344 if (!this.hasPass(plate)) throw new CommandError(plate + " has no pass");
345 this.passes.delete(plate);
346 this.emit("UNPASS", plate);
347 break;
348 }
349 case "MOVE": {
350 const v = this.vehicle(plate);
351 if (v.kind === "BUS") throw new CommandError("bus " + plate + " cannot be moved");
352 const s = this.spot(a[1]);
353 if (s === v.spots[0]) throw new CommandError(plate + " is already at " + s.code);
354 if (s.closed) throw new CommandError("spot " + s.code + " is closed");
355 if (s.vehicle) throw new CommandError("spot " + s.code + " is occupied by " + s.vehicle.plate);
356 if (!PREF[v.kind].includes(s.kind)) throw new CommandError(v.kind + " cannot use " + s.kind + " spot " + s.code);
357 this.moveVehicle(v, s);
358 this.processQueue();
359 break;
360 }
361 case "REBALANCE": {
362 const targets = [...this.vehicles.values()].filter(v => v.kind !== "BUS").sort(order);
363 let moved = 0;
364 for (const v of targets) {
365 const rank = PREF[v.kind].indexOf(v.spots[0].kind);
366 if (rank === 0) continue;
367 const spots = this.assignment(v.kind);
368 if (spots && PREF[v.kind].indexOf(spots[0].kind) < rank) {
369 this.moveVehicle(v, spots[0]);
370 moved++;
371 }
372 }
373 this.emit("REBALANCE", moved);
374 this.processQueue();
375 break;
376 }
377 case "WHERE": {
378 const v = this.vehicles.get(plate);
379 const position = this.queue.findIndex(w => w.plate === plate);
380 if (v) {
381 this.emit("WHERE", plate, location(v), v.kind, stamp(v.entered), "stay", duration(this.now - v.entered) + this.annotation(v));
382 } else if (position >= 0) {
383 this.emit("WHERE", plate, "waiting", position + 1);
384 } else {
385 const records = this.history.get(plate);
386 if (!records) throw new CommandError("unknown plate " + plate);
387 this.emit("WHERE", plate, "gone", stamp(records[records.length - 1].exited));
388 }
389 break;
390 }
391 case "LIST":
392 case "OVERSTAY": {
393 const level = a.length ? Number(a[0]) : null;
394 if (cmd === "LIST" && level !== null && level > this.levels.length) throw new CommandError("unknown level " + level);
395 const targets = [...this.vehicles.values()].filter(v => cmd === "LIST" ? level === null || v.spots[0].level === level : this.now - v.entered >= level! * 60);
396 targets.sort(cmd === "LIST" ? order : (a, b) => a.entered - b.entered || order(a, b));
397 if (cmd === "LIST") this.emit("LIST", targets.length);
398 else this.emit("OVERSTAY", level!, targets.length);
399 for (const v of targets) {
400 if (cmd === "LIST") this.emit(location(v), v.plate, v.kind, stamp(v.entered) + this.annotation(v));
401 else this.emit(location(v), v.plate, v.kind, stamp(v.entered), "stay", duration(this.now - v.entered));
402 }
403 break;
404 }
405 case "MAP": {
406 this.emit("MAP", this.levels.length);
407 for (let i = 0; i < this.levels.length; i++) {
408 const symbols = this.levels[i].map(s => s.closed ? "#" : s.vehicle ? s.kind[0] : s.kind[0].toLowerCase()).join("");
409 this.emit("L" + (i + 1), ...symbols.match(/.{1,10}/g)!);
410 }
411 break;
412 }
413 case "OCCUPANCY": {
414 this.emit("OCCUPANCY", "vehicles", this.vehicles.size, "waiting", this.queue.length);
415 const allUsed: Record<string, number> = Object.fromEntries(KINDS.map(k => [k, 0]));
416 const allAvailable: Record<string, number> = {...allUsed};
417 let allClosed = 0;
418 for (let i = 0; i < this.levels.length; i++) {
419 const used: Record<string, number> = Object.fromEntries(KINDS.map(k => [k, 0]));
420 const available: Record<string, number> = {...used};
421 let closed = 0;
422 for (const s of this.levels[i]) {
423 if (s.closed) closed++;
424 else available[s.kind]++;
425 if (s.vehicle) used[s.kind]++;
426 }
427 const parts: (string | number)[] = ["L" + (i + 1)];
428 for (const k of KINDS) {
429 parts.push(k, used[k] + "/" + available[k]);
430 allUsed[k] += used[k];
431 allAvailable[k] += available[k];
432 }
433 this.emit(...parts, "closed", closed);
434 allClosed += closed;
435 }
436 const parts: (string | number)[] = ["TOTAL"];
437 for (const k of KINDS) parts.push(k, allUsed[k] + "/" + allAvailable[k]);
438 const available = Object.values(allAvailable).reduce((a, b) => a + b, 0);
439 const used = Object.values(allUsed).reduce((a, b) => a + b, 0);
440 const q = available ? Math.floor(used * 1000 / available) : 0;
441 this.emit(...parts, "closed", allClosed, "rate", Math.floor(q / 10) + "." + q % 10 + "%");
442 break;
443 }
444 case "SIGN": {
445 const parts: (string | number)[] = ["SIGN"];
446 for (const kind of TYPES) {
447 let count = 0;
448 if (kind === "BUS") {
449 for (const level of this.levels) {
450 let run = 0;
451 for (const s of level) {
452 if (s.kind === "LARGE" && free(s)) run++;
453 else {
454 count += Math.floor(run / 3);
455 run = 0;
456 }
457 }
458 count += Math.floor(run / 3);
459 }
460 } else {
461 count = this.spots.filter(s => free(s) && PREF[kind].includes(s.kind)).length;
462 }
463 parts.push(kind, count || "FULL");
464 }
465 this.emit(...parts);
466 break;
467 }
468 case "RATES": {
469 this.emit("RATES");
470 for (const kind of TYPES) this.emit(kind, ...this.rates[kind]);
471 this.emit("NIGHT", this.fees.NIGHT, "LOST", this.fees.LOST, "KWH", this.fees.KWH);
472 break;
473 }
474 case "PASSES": {
475 const plates = [...this.passes.keys()].filter(p => this.hasPass(p)).sort();
476 this.emit("PASSES", plates.length);
477 for (const p of plates) this.emit(p, "until", this.passes.get(p)!, ...(this.vehicles.has(p) ? ["IN"] : []));
478 break;
479 }
480 case "HISTORY": {
481 const records = this.history.get(plate);
482 if (!records) throw new CommandError("no history for " + plate);
483 this.emit("HISTORY", plate, "visits", records.length, "paid", records.reduce((a, r) => a + r.total, 0));
484 for (const r of records) this.emit(stamp(r.entered), stamp(r.exited), r.location, r.kind, r.total, ...(r.lost ? ["LOST"] : []));
485 break;
486 }
487 case "REVENUE": {
488 this.emit("REVENUE");
489 for (const kind of TYPES) this.revenueLine(kind, this.totals[kind]);
490 this.revenueLine("TOTAL", sumTotals(Object.values(this.totals)));
491 break;
492 }
493 case "DAILY": {
494 const day = Number(a[0]);
495 const totals = this.daily.get(day) ?? emptyTypes();
496 this.dailyLine(["DAILY", day], sumTotals(Object.values(totals)));
497 for (const kind of TYPES) this.dailyLine([kind], totals[kind]);
498 break;
499 }
500 case "STATS": {
501 this.emit("STATS", "parked", this.vehicles.size, "peak", this.peak, "at", this.peakTime === null ? "-" : stamp(this.peakTime));
502 for (const kind of TYPES) this.statsLine(kind, this.totals[kind]);
503 this.statsLine("ALL", sumTotals(Object.values(this.totals)));
504 break;
505 }
506 }
507 }

509 execute(line: string): void {
510 const [day, clock, cmd, ...args] = line.split(" ");
511 const [hour, minute] = clock.split(":").map(Number);
512 this.now = (Number(day) - 1) * 1440 + hour * 60 + minute;
513 this.giveUp();
514 try {
515 this.command(cmd, args);
516 } catch (error) {
517 if (!(error instanceof CommandError)) throw error;
518 this.emit("ERROR", error.message);
519 }
520 }
