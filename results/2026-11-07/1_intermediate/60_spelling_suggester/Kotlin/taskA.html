4private val adjacentKeys = arrayOf(
5 "qswz", "ghnv", "dfvx", "cefrsx", "drsw", "cdgrtv", "bfhtvy", "bgjnuy",
6 "jkou", "hikmnu", "ijlmo", "kop", "jkn", "bhjm", "iklp", "lo", "aw",

9private val soundexDigits = CharArray(26).also { digits ->
10 arrayOf("bfpv", "cgjkqsxz", "dt", "l", "mn", "r", "aeiouy")
11 .forEachIndexed { index, letters ->
12 letters.forEach { letter -> digits[letter - 'a'] = if (index == 6) '0' else '1' + index }

37private fun numberArgument(text: String, minimum: Int, maximum: Int): Int {
38 if (!numberPattern.matches(text)) throw CommandError("BAD NUMBER $text")
39 val value = text.toInt()
40 if (value !in minimum..maximum) throw CommandError("BAD NUMBER $text")

44private fun decimalText(numerator: Long, denominator: Long, digits: Int): String {
45 val scale = if (digits == 1) 10L else 100L
46 val value = (2 * numerator * scale + denominator) / (2 * denominator)
47 return "${value / scale}.${(value % scale).toString().padStart(digits, '0')}"
48}

50private fun applyCase(original: String, word: String): String = when {
51 original.length >= 2 && original.all { it.isUpperCase() } -> word.uppercase()
52 original.first().isUpperCase() -> word.replaceFirstChar { it.uppercase() }
53 else -> word
54}

56private data class Token(
57 val number: Int, val original: String, val start: Int, val end: Int,
58 val gap: String, val previous: Token?
59) {
60 val word = original.lowercase()
61 val sentenceStart = previous == null || gap.any { it in ".!?" }
62 val repeated = previous != null && word == previous.word && gap.all { it == ' ' }
63 val mixedCase = original.drop(1).any { it.isUpperCase() } && !original.all { it.isUpperCase() }
64}

66private fun textTokens(text: String): List<Token> {
67 val tokens = mutableListOf<Token>()
68 var position = 0
69 for (match in tokenPattern.findAll(text)) {
70 val token = Token(tokens.size + 1, match.value, match.range.first,
71 match.range.last + 1, text.substring(position, match.range.first), tokens.lastOrNull())
72 tokens += token
73 position = token.end
74 }
75 return tokens
76}

78private fun substitutionCost(first: Char, second: Char, keyboard: Boolean): Int = when {
79 first == second -> 0
80 !keyboard || second in adjacentKeys[first - 'a'] -> 1
81 else -> 2
82}

84private fun isSwap(first: String, second: String, i: Int, j: Int) =
85 i >= 2 && j >= 2 && first[i - 1] == second[j - 2] && first[i - 2] == second[j - 1]
86
87private fun distanceTable(first: String, second: String, keyboard: Boolean): Array<IntArray> {
88 val m = first.length
89 val n = second.length
90 val gap = if (keyboard) 2 else 1
91 val table = Array(m + 1) { IntArray(n + 1) }
92 for (i in 0..m) table[i][0] = i * gap
93 for (j in 0..n) table[0][j] = j * gap
94 for (i in 1..m) {
95 for (j in 1..n) {
96 var best = minOf(table[i - 1][j] + gap, table[i][j - 1] + gap,
97 table[i - 1][j - 1] + substitutionCost(first[i - 1], second[j - 1], keyboard))
98 if (isSwap(first, second, i, j)) best = minOf(best, table[i - 2][j - 2] + 1)
99 table[i][j] = best
100 }
101 }
102 return table
103}
104
105private fun editDistance(first: String, second: String, keyboard: Boolean): Int =
106 distanceTable(first, second, keyboard)[first.length][second.length]
107
108private fun prefixDistance(word: String, other: String, keyboard: Boolean): Int =
109 distanceTable(word, other, keyboard)[word.length].min()
110
111private fun editOperations(first: String, second: String, keyboard: Boolean): Pair<Int, List<String>> {
112 val table = distanceTable(first, second, keyboard)
113 val gap = if (keyboard) 2 else 1
114 var i = first.length
115 var j = second.length
116 val operations = mutableListOf<String>()
117 while (i > 0 || j > 0) {
118 if (i >= 1 && j >= 1 && table[i][j] == table[i - 1][j - 1] +
119 substitutionCost(first[i - 1], second[j - 1], keyboard)) {
120 if (first[i - 1] != second[j - 1]) operations += "sub:${first[i - 1]}>${second[j - 1]}"
121 --i; --j
122 } else if (isSwap(first, second, i, j) && table[i][j] == table[i - 2][j - 2] + 1) {
123 operations += "swap:${first.substring(i - 2, i)}"
124 i -= 2; j -= 2
125 } else if (i >= 1 && table[i][j] == table[i - 1][j] + gap) {
126 operations += "del:${first[i - 1]}"
127 --i
128 } else {
129 operations += "ins:${second[j - 1]}"
130 --j
131 }
132 }
133 return table[first.length][second.length] to operations.asReversed()
134}

136private fun soundex(word: String): String {
137 val code = StringBuilder(word.first().uppercase())
138 var previous = if (word.first() in "hw") '0' else soundexDigits[word.first() - 'a']
139 for (letter in word.drop(1)) {
140 if (letter in "hw") continue
141 val digit = soundexDigits[letter - 'a']
142 if (digit != '0' && digit != previous) code.append(digit)
143 previous = digit
144 }
145 return (code.toString() + "000").take(4)
146}

148private data class Segmentation(val ways: Long, val words: List<String>?)
149private fun segment(word: String, dictionary: Map<String, Int>): Segmentation {
150 val n = word.length
151 val ways = LongArray(n + 1)
152 val best = arrayOfNulls<List<String>>(n + 1)
153 ways[n] = 1
154 best[n] = emptyList()
155 for (start in n - 1 downTo 0) {
156 for (end in start + 1..n) {
157 val piece = word.substring(start, end)
158 if (piece !in dictionary || ways[end] == 0L) continue
159 ways[start] += ways[end]
160 val candidate = listOf(piece) + best[end]!!
161 val current = best[start]
162 if (current == null || candidate.size < current.size ||
163 (candidate.size == current.size && lexicographic(candidate, current) < 0)) {
164 best[start] = candidate
165 }
166 }
167 }
168 return Segmentation(ways[0], best[0])
169}

171private fun lexicographic(first: List<String>, second: List<String>): Int {
172 for (index in 0 until minOf(first.size, second.size)) {
173 val comparison = first[index].compareTo(second[index])
174 if (comparison != 0) return comparison
175 }
176 return first.size.compareTo(second.size)
177}

185private class SpellChecker(initial: Map<String, Int>) {
186 private var dictionary = initial.toMutableMap()
187 private var ignored = mutableSetOf<String>()
188 private var rules = mutableMapOf<String, String>()
189 private var keyboard = false
190 private var baseLimit = 2
191 private var suggestionCount = 5
192 private val undoHistory = mutableListOf<UndoRecord>()
193 private val unknownLog = mutableMapOf<String, Int>()
194 private val correctionLog = mutableMapOf<String, Int>()
195 private val counters = counterNames.associateWith { 0 }.toMutableMap()
196 val output = mutableListOf<String>()
197
198 private fun emit(vararg parts: Any) { output += parts.joinToString(" ") }
199 private fun limit() = if (keyboard) 2 * baseLimit else baseLimit
200 private fun isKnown(word: String) = word in dictionary || word in ignored
201
202 private fun frequencyOrder(): List<String> = dictionary.entries.sortedWith(
203 compareByDescending<Map.Entry<String, Int>> { it.value }.thenBy { it.key }
204 ).map { it.key }
205
206 private fun requireEntry(word: String): Int = dictionary[word] ?: throw CommandError("NOT FOUND $word")
207
208 private fun candidates(word: String): List<Candidate> {
209 val gap = if (keyboard) 2 else 1
210 val maximum = limit()
211 return dictionary.mapNotNull { (other, frequency) ->
212 if (other == word || kotlin.math.abs(other.length - word.length) * gap > maximum) {
213 null
214 } else {
215 val distance = editDistance(word, other, keyboard)
216 if (distance <= maximum) Candidate(other, distance, frequency) else null
217 }
218 }.sortedWith(compareBy<Candidate> { it.distance }.thenByDescending { it.frequency }.thenBy { it.word })
219 }
220
221 private fun correction(word: String): String? = when {
222 isKnown(word) -> null
223 word in rules -> rules[word]
224 else -> candidates(word).firstOrNull()?.word
225 }
226
227 private fun recordUndo(command: String) {
228 undoHistory += UndoRecord(command, dictionary.toMutableMap(), ignored.toMutableSet(), rules.toMutableMap())
229 }
230
231 private fun logCorrection(word: String, replacement: String) {
232 val key = "$word>$replacement"
233 correctionLog[key] = correctionLog.getOrDefault(key, 0) + 1
234 counters["corrected"] = counters.getValue("corrected") + 1
235 }
236
237 private fun unknownFinding(token: Token): String {
238 val prefix = "UNKNOWN ${token.number} ${token.original}"
239 rules[token.word]?.let { return "$prefix RULE $it" }
240 val found = candidates(token.word)
241 return if (found.isEmpty()) "$prefix NONE" else
242 "$prefix SUGGEST ${found.take(suggestionCount).joinToString(" ") { it.word }}"
243 }
244
245 private fun checkText(text: String) {
246 val tokens = textTokens(text)
247 val findings = mutableListOf<String>()
248 val counts = findingNames.associateWith { 0 }.toMutableMap()
249 fun finding(kind: String, line: String) {
250 findings += line
251 counts[kind] = counts.getValue(kind) + 1
252 }
253 for (token in tokens) {
254 if (!isKnown(token.word)) {
255 finding("unknown", unknownFinding(token))
256 unknownLog[token.word] = unknownLog.getOrDefault(token.word, 0) + 1
257 }
258 if (token.mixedCase) finding("case", "CASE ${token.number} ${token.original}")
259 if (token.sentenceStart && token.original.first().isLowerCase())
260 finding("capital", "CAPITAL ${token.number} ${token.original}")
261 if (token.repeated) finding("repeats", "REPEAT ${token.number} ${token.original}")
262 }
263 counters["checked"] = counters.getValue("checked") + tokens.size
264 findingNames.forEach { counters[it] = counters.getValue(it) + counts.getValue(it) }
265 emit("CHECK words=${tokens.size}", findingNames.joinToString(" ") { "$it=${counts.getValue(it)}" })
266 output.addAll(findings)
267 }
268
269 private fun correctText(text: String) {
270 val result = StringBuilder()
271 var position = 0
272 var changed = 0
273 for (token in textTokens(text)) {
274 val replacement = correction(token.word)
275 result.append(token.gap)
276 if (replacement == null) {
277 result.append(token.original)
278 } else {
279 result.append(applyCase(token.original, replacement))
280 logCorrection(token.word, replacement)
281 changed++
282 }
283 position = token.end
284 }
285 result.append(text.substring(position))
286 output += "CORRECT changed=$changed: $result"
287 }
288
289 private fun fixText(text: String) {
290 val result = StringBuilder()
291 var position = 0
292 var removed = 0
293 var replaced = 0
294 var recased = 0
295 for (token in textTokens(text)) {
296 position = token.end
297 if (token.repeated) { removed++; continue }
298 val replacement = correction(token.word)
299 var revised = applyCase(token.original, replacement ?: token.word)
300 if (token.sentenceStart && revised[0].isLowerCase())
301 revised = revised.replaceFirstChar { it.uppercase() }
302 if (replacement != null) {
303 logCorrection(token.word, replacement)
304 replaced++
305 } else if (revised != token.original) {
306 recased++
307 }
308 result.append(token.gap).append(revised)
309 }
310 result.append(text.substring(position))
311 output += "FIX removed=$removed replaced=$replaced recased=$recased: $result"
312 }

314 private fun profile(text: String) {
315 val tokens = textTokens(text)
316 val size = tokens.size
317 val distinct = tokens.map { it.word }.toSet().size
318 val unknown = tokens.count { !isKnown(it.word) }
319 val sentences = tokens.count { it.sentenceStart }
320 val longest = tokens.maxByOrNull { it.original.length }?.word ?: "-"
321 val rate = if (size == 0) "0.0" else decimalText(100L * unknown, size.toLong(), 1)
322 val average = if (size == 0) "0.00" else
323 decimalText(tokens.sumOf { it.original.length }.toLong(), size.toLong(), 2)
324 emit("PROFILE words=$size distinct=$distinct unknown=$unknown rate=$rate avglen=$average longest=$longest sentences=$sentences")
325 }

327 private fun learn(text: String) {
328 recordUndo("LEARN")
329 var updated = 0
330 var added = 0
331 var skipped = 0
332 for (token in textTokens(text)) {
333 when {
334 token.word.length > 30 -> skipped++
335 token.word in dictionary -> { dictionary[token.word] = dictionary.getValue(token.word) + 1; updated++ }
336 else -> { dictionary[token.word] = 1; ignored.remove(token.word); added++ }
337 }
338 }
339 emit("LEARN updated=$updated added=$added skipped=$skipped")
340 }

342 private fun suggest(args: List<String>) {
343 val word = wordArgument(args[0])
344 when {
345 isKnown(word) -> emit("SUGGEST", word, "KNOWN")
346 word in rules -> emit("SUGGEST", word, "RULE", rules.getValue(word))
347 else -> {
348 val found = candidates(word)
349 val entries = found.take(suggestionCount).joinToString(" ") { "${it.word}:${it.distance}" }
350 output += "SUGGEST $word found=${found.size}${if (entries.isEmpty()) "" else " $entries"}"
351 }
352 }
353 }

356 val first = wordArgument(args[0]); val second = wordArgument(args[1])
357 val (distance, operations) = editOperations(first, second, keyboard)
358 output += (listOf("DIST", first, second, distance.toString()) + operations).joinToString(" ")

363 val code = soundex(word)

375 private fun prefix(args: List<String>) {
376 val word = wordArgument(args[0])
377 val count = numberArgument(args[1], 1, 20)
378 val found = frequencyOrder().filter { it.startsWith(word) }
379 output += (listOf("PREFIX", word, "count=${found.size}") +

383 private fun complete(args: List<String>) {
384 val word = wordArgument(args[0])
385 val count = numberArgument(args[1], 1, 20)
386 val threshold = limit() / 2
387 val found = dictionary.mapNotNull { (other, frequency) ->
388 val distance = prefixDistance(word, other, keyboard)
389 if (distance <= threshold) Candidate(other, distance, frequency) else null
390 }.sortedWith(compareBy<Candidate> { it.distance }.thenByDescending { it.frequency }.thenBy { it.word })
391 output += (listOf("COMPLETE", word, "count=${found.size}") +
392 found.take(count).map { "${it.word}:${it.distance}" }).joinToString(" ")
393 }

398 val found = dictionary.keys.filter { it >= word }.sorted()
399 output += (listOf("LIST", word, "count=${found.size}") +

445 private fun rule(args: List<String>) {
446 val source = wordArgument(args[0])
447 val target = wordArgument(args[1])
448 if (source in dictionary) throw CommandError("IN DICTIONARY $source")
449 if (target !in dictionary) throw CommandError("NOT FOUND $target")
450 recordUndo("RULE")
451 val previous = rules.put(source, target)
452 if (previous == null) emit("RULE", source, target)
453 else emit("RULE", source, target, "REPLACED", previous)
454 }

457 val word = wordArgument(args[0])
458 if (word !in rules) throw CommandError("NO RULE $word")
459 recordUndo("UNRULE")
460 val target = rules.remove(word)!!
461 emit("UNRULE", word, target)
462 }

507 private fun stats() {
508 val top = frequencyOrder().first()
509 val total = dictionary.values.sumOf { it.toLong() }
510 val characters = dictionary.keys.sumOf { it.length.toLong() }
511 emit("STATS words=${dictionary.size} total=$total avglen=${decimalText(characters, dictionary.size.toLong(), 2)}",
512 "top=$top:${dictionary.getValue(top)} ignored=${ignored.size} rules=${rules.size}",
513 "keyboard=${if (keyboard) "ON" else "OFF"} limit=${limit()}",
514 "suggestions=$suggestionCount undo=${undoHistory.size}")
515 output += (listOf("SESSION") + counterNames.map { "$it=${counters.getValue(it)}" }).joinToString(" ")
516 }

518 fun execute(line: String) {
519 val separator = line.indexOf(' ')
520 val command = if (separator == -1) line else line.substring(0, separator)
521 val rest = if (separator == -1) "" else line.substring(separator + 1)
522 try {
523 if (command in textCommands) {
524 if (separator == -1) throw CommandError("BAD ARGUMENTS $command")
525 when (command) {
526 "CHECK" -> checkText(rest)
527 "CORRECT" -> correctText(rest)
528 "FIX" -> fixText(rest)
529 "PROFILE" -> profile(rest)
530 "LEARN" -> learn(rest)
531 }
532 return
533 }
534 val expected = arities[command] ?: throw CommandError("UNKNOWN COMMAND $command")
535 val args = if (separator == -1) emptyList() else rest.split(' ')
536 if (args.size != expected) throw CommandError("BAD ARGUMENTS $command")
537 when (command) {
538 "SUGGEST" -> suggest(args)
539 "DIST" -> dist(args)
540 "SOUNDEX" -> { val word = wordArgument(args[0]); emit("SOUNDEX", word, soundex(word)) }
541 "SOUNDS" -> sounds(args)
542 "SPLIT" -> split(args)
543 "PREFIX" -> prefix(args)
544 "COMPLETE" -> complete(args)
545 "LIST" -> listWords(args)
546 "WORD" -> wordInfo(args)
547 "ADD" -> add(args)
548 "REMOVE" -> remove(args)
549 "IGNORE" -> ignore(args)
550 "UNIGNORE" -> unignore(args)
551 "RULE" -> rule(args)
552 "UNRULE" -> unrule(args)
553 "RULES" -> showRules()
554 "KEYBOARD" -> keyboardMode(args)
555 "SET" -> setOption(args)
556 "UNDO" -> undo()
557 "REPORT" -> report()
558 "HISTORY" -> history()
559 "STATS" -> stats()
560 }
561 } catch (error: CommandError) {
562 emit("ERROR", error.message!!)
563 }
564 }

567fun main() {
568 val input = BufferedReader(InputStreamReader(System.`in`))
569 val size = input.readLine().toInt()
570 val dictionary = mutableMapOf<String, Int>()
571 repeat(size) {
572 val entry = input.readLine().split(' ')
573 dictionary[entry[0]] = entry[1].toInt()
574 }
575 val checker = SpellChecker(dictionary)
576 val count = input.readLine().toInt()
577 repeat(count) { checker.execute(input.readLine()) }
578 print(checker.output.joinToString("\n", postfix = "\n"))
579}
