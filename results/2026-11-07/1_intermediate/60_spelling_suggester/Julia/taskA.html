30function word_argument(text)
31 occursin(r"^[A-Za-z]{1,30}$", text) || throw(CommandError("BAD WORD $text"))
32 return lowercase(text)
33end

35function number_argument(text, lower, upper)
36 occursin(r"^[0-9]{1,7}$", text) || throw(CommandError("BAD NUMBER $text"))
37 value = parse(Int, text)
38 lower <= value <= upper || throw(CommandError("BAD NUMBER $text"))
39 return value
40end

42function decimal_text(numerator, denominator, digits)
43 scale = 10^digits
44 value = div(2 * numerator * scale + denominator, 2 * denominator)
45 return "$(div(value, scale)).$(lpad(value % scale, digits, '0'))"
46end

52function apply_case(token_text, word)
53 if length(token_text) >= 2 && all(isuppercase, token_text)
54 return uppercase(word)
55 elseif isuppercase(token_text[1])
56 return uppercasefirst(word)
57 end
58 return word
59end

61struct Token
62 number::Int
63 text::String
64 word::String
65 stop::Int
66 gap::String
67 sentence_start::Bool
68 repeat::Bool
69end
70
71is_mixed_case(token::Token) =
72 any(isuppercase, token.text[2:end]) && !all(isuppercase, token.text)

74function tokenize(text)
75 tokens = Token[]
76 position = 1
77 for token_match in eachmatch(r"[A-Za-z]+", text)
78 token_text = String(token_match.match)
79 word = lowercase(token_text)
80 gap = text[position:token_match.offset-1]
81 is_sentence_start = isempty(tokens) || any(in(SENTENCE_MARKS), gap)
82 is_repeat = !isempty(tokens) && word == tokens[end].word && all(==(' '), gap)
83 stop = token_match.offset + length(token_text) - 1
84 token = Token(
85 length(tokens) + 1, token_text, word, stop, gap, is_sentence_start, is_repeat,
86 )
87 push!(tokens, token)
88 position = stop + 1
89 end
90 return tokens
91end

93gap_cost(keyboard) = keyboard ? 2 : 1
94
95function substitution_cost(x, y, keyboard)
96 x == y && return 0
97 keyboard || return 1
98 return y in ADJACENT_KEYS[x] ? 1 : 2
99end
100
101is_swap(a, b, i, j) = i >= 2 && j >= 2 && a[i] == b[j-1] && a[i-1] == b[j]
102
103function distance_table(a, b, keyboard)
104 gap = gap_cost(keyboard)
105 rows, columns = length(a), length(b)
106 table = zeros(Int, rows + 1, columns + 1)
107 table[:, 1] .= (0:rows) .* gap
108 table[1, :] .= (0:columns) .* gap
109 for i in 1:rows, j in 1:columns
110 best = min(
111 table[i, j+1] + gap,
112 table[i+1, j] + gap,
113 table[i, j] + substitution_cost(a[i], b[j], keyboard),
114 )
115 if is_swap(a, b, i, j)
116 best = min(best, table[i-1, j-1] + 1)
117 end
118 table[i+1, j+1] = best
119 end
120 return table
121end
122
123edit_distance(a, b, keyboard) = distance_table(a, b, keyboard)[end, end]
124
125prefix_distance(word, other, keyboard) = minimum(distance_table(word, other, keyboard)[end, :])
126
127function edit_operations(a, b, keyboard)
128 table = distance_table(a, b, keyboard)
129 gap = gap_cost(keyboard)
130 operations = String[]
131 i, j = length(a), length(b)
132 while i > 0 || j > 0
133 if i >= 1 && j >= 1 &&
134 table[i+1, j+1] == table[i, j] + substitution_cost(a[i], b[j], keyboard)
135 if a[i] != b[j]
136 push!(operations, "sub:$(a[i])>$(b[j])")
137 end
138 i, j = i - 1, j - 1
139 elseif is_swap(a, b, i, j) && table[i+1, j+1] == table[i-1, j-1] + 1
140 push!(operations, "swap:$(a[i-1])$(a[i])")
141 i, j = i - 2, j - 2
142 elseif i >= 1 && table[i+1, j+1] == table[i, j+1] + gap
143 push!(operations, "del:$(a[i])")
144 i -= 1
145 else
146 push!(operations, "ins:$(b[j])")
147 j -= 1
148 end
149 end
150 return table[end, end], reverse(operations)
151end

153function soundex(word)
154 code = string(uppercase(word[1]))
155 previous = word[1] in "hw" ? '0' : SOUNDEX_DIGITS[word[1]]
156 for letter in word[2:end]
157 letter in "hw" && continue
158 digit = SOUNDEX_DIGITS[letter]
159 if digit != '0' && digit != previous
160 code *= digit
161 end
162 previous = digit
163 end
164 return first(code * "000", 4)
165end

167function segment(word, dictionary)
168 len = length(word)
169 ways = zeros(Int, len + 1)
170 best = Vector{Union{Nothing,Vector{String}}}(nothing, len + 1)
171 ways[len+1] = 1
172 best[len+1] = String[]
173 for start in len:-1:1
174 for stop in start:len
175 piece = word[start:stop]
176 (haskey(dictionary, piece) && ways[stop+1] > 0) || continue
177 ways[start] += ways[stop+1]
178 candidate = [piece; best[stop+1]]
179 current = best[start]
180 if current === nothing || (length(candidate), candidate) < (length(current), current)
181 best[start] = candidate
182 end
183 end
184 end
185 return ways[1], best[1]
186end

214limit(checker) = checker.keyboard ? 2 * checker.base_limit : checker.base_limit

216is_known(checker, word) = haskey(checker.dictionary, word) || word in checker.ignored

220function require_entry(checker, word)
221 haskey(checker.dictionary, word) || throw(CommandError("NOT FOUND $word"))
222 return checker.dictionary[word]
223end
224
225function candidates(checker, word)
226 max_distance = limit(checker)
227 gap = gap_cost(checker.keyboard)
228 found = Tuple{Int,Int,String}[]
229 for (other, frequency) in checker.dictionary
230 if other == word || abs(length(other) - length(word)) * gap > max_distance
231 continue
232 end
233 distance = edit_distance(word, other, checker.keyboard)
234 if distance <= max_distance
235 push!(found, (distance, -frequency, other))
236 end
237 end
238 sort!(found)
239 return [(other, distance) for (distance, _, other) in found]
240end
241
242function correction(checker, word)
243 is_known(checker, word) && return nothing
244 haskey(checker.rules, word) && return checker.rules[word]
245 found = candidates(checker, word)
246 return isempty(found) ? nothing : found[1][1]
247end

249function record_undo!(checker, name)
250 record = UndoRecord(
251 name, copy(checker.dictionary), copy(checker.ignored), copy(checker.rules),
252 )
253 push!(checker.undo_history, record)
254end

256function log_correction!(checker, word, replacement)
257 key = (word, replacement)
258 checker.correction_log[key] = get(checker.correction_log, key, 0) + 1
259 checker.counters[:corrected] += 1
260end

262function unknown_finding(checker, token)
263 label = "UNKNOWN $(token.number) $(token.text)"
264 haskey(checker.rules, token.word) && return "$label RULE $(checker.rules[token.word])"
265 found = candidates(checker, token.word)
266 isempty(found) && return "$label NONE"
267 words = [other for (other, _) in first(found, checker.suggestion_count)]
268 return join([label, "SUGGEST", words...], " ")
269end

271function check!(checker, text)
272 tokens = tokenize(text)
273 findings = Pair{Symbol,String}[]
274 for token in tokens
275 if !is_known(checker, token.word)
276 push!(findings, :unknown => unknown_finding(checker, token))
277 checker.unknown_log[token.word] = get(checker.unknown_log, token.word, 0) + 1
278 end
279 if is_mixed_case(token)
280 push!(findings, :case => "CASE $(token.number) $(token.text)")
281 end
282 if token.sentence_start && islowercase(token.text[1])
283 push!(findings, :capital => "CAPITAL $(token.number) $(token.text)")
284 end
285 if token.repeat
286 push!(findings, :repeats => "REPEAT $(token.number) $(token.text)")
287 end
288 end
289 counts = Dict(
290 kind => count(finding -> finding.first == kind, findings) for kind in FINDING_KINDS
291 )
292 checker.counters[:checked] += length(tokens)
293 for kind in FINDING_KINDS
294 checker.counters[kind] += counts[kind]
295 end
296 emit(
297 "CHECK", "words=$(length(tokens))",
298 ("$kind=$(counts[kind])" for kind in FINDING_KINDS)...,
299 )
300 for finding in findings
301 emit(finding.second)
302 end
303end

305function correct!(checker, text)
306 pieces = String[]
307 position = 0
308 changed = 0
309 for token in tokenize(text)
310 replacement = correction(checker, token.word)
311 if replacement === nothing
312 push!(pieces, token.gap * token.text)
313 else
314 push!(pieces, token.gap * apply_case(token.text, replacement))
315 log_correction!(checker, token.word, replacement)
316 changed += 1
317 end
318 position = token.stop
319 end
320 push!(pieces, text[position+1:end])
321 emit("CORRECT changed=$changed:", join(pieces))
322end

324function fix!(checker, text)
325 pieces = String[]
326 position = 0
327 removed = 0
328 replaced = 0
329 recased = 0
330 for token in tokenize(text)
331 position = token.stop
332 if token.repeat
333 removed += 1
334 continue
335 end
336 replacement = correction(checker, token.word)
337 result = apply_case(token.text, something(replacement, token.word))
338 if token.sentence_start && islowercase(result[1])
339 result = uppercasefirst(result)
340 end
341 if replacement !== nothing
342 log_correction!(checker, token.word, replacement)
343 replaced += 1
344 elseif result != token.text
345 recased += 1
346 end
347 push!(pieces, token.gap * result)
348 end
349 push!(pieces, text[position+1:end])
350 emit("FIX", "removed=$removed", "replaced=$replaced", "recased=$recased:", join(pieces))
351end

353function profile(checker, text)
354 tokens = tokenize(text)
355 distinct = length(unique(token.word for token in tokens))
356 unknown = count(token -> !is_known(checker, token.word), tokens)
357 sentences = count(token -> token.sentence_start, tokens)
358 if isempty(tokens)
359 rate, average, longest = "0.0", "0.00", "-"
360 else
361 characters = sum(token -> length(token.text), tokens)
362 rate = decimal_text(100 * unknown, length(tokens), 1)
363 average = decimal_text(characters, length(tokens), 2)
364 longest = argmax(token -> length(token.text), tokens).word
365 end
366 emit(
367 "PROFILE", "words=$(length(tokens))", "distinct=$distinct", "unknown=$unknown",
368 "rate=$rate", "avglen=$average", "longest=$longest", "sentences=$sentences",
369 )
370end

372function learn!(checker, text)
373 record_undo!(checker, "LEARN")
374 updated = 0
375 added = 0
376 skipped = 0
377 for token in tokenize(text)
378 if length(token.word) > MAX_WORD_LENGTH
379 skipped += 1
380 elseif haskey(checker.dictionary, token.word)
381 checker.dictionary[token.word] += 1
382 updated += 1
383 else
384 checker.dictionary[token.word] = 1
385 delete!(checker.ignored, token.word)
386 added += 1
387 end
388 end
389 emit("LEARN", "updated=$updated", "added=$added", "skipped=$skipped")
390end

392function suggest(checker, arguments)
393 word = word_argument(arguments[1])
394 if is_known(checker, word)
395 emit("SUGGEST", word, "KNOWN")
396 elseif haskey(checker.rules, word)
397 emit("SUGGEST", word, "RULE", checker.rules[word])
398 else
399 found = candidates(checker, word)
400 shown = first(found, checker.suggestion_count)
401 emit(
402 "SUGGEST", word, "found=$(length(found))",
403 ("$other:$distance" for (other, distance) in shown)...,
404 )
405 end
406end

449function complete(checker, arguments)
450 word = word_argument(arguments[1])
451 shown = number_argument(arguments[2], 1, MAX_LISTED)
452 threshold = limit(checker) ÷ 2
453 matches = Tuple{Int,Int,String}[]
454 for (other, frequency) in checker.dictionary
455 distance = prefix_distance(word, other, checker.keyboard)
456 if distance <= threshold
457 push!(matches, (distance, -frequency, other))
458 end
459 end
460 sort!(matches)
461 emit(
462 "COMPLETE", word, "count=$(length(matches))",
463 ("$other:$distance" for (distance, _, other) in first(matches, shown))...,
464 )
465end

522function rule!(checker, arguments)
523 source = word_argument(arguments[1])
524 target = word_argument(arguments[2])
525 haskey(checker.dictionary, source) && throw(CommandError("IN DICTIONARY $source"))
526 haskey(checker.dictionary, target) || throw(CommandError("NOT FOUND $target"))
527 record_undo!(checker, "RULE")
528 previous = get(checker.rules, source, nothing)
529 checker.rules[source] = target
530 if previous === nothing
531 emit("RULE", source, target)
532 else
533 emit("RULE", source, target, "REPLACED", previous)
534 end
535end

597function stats(checker, _)

612const HANDLERS = Dict(
613 "CHECK" => (nothing, check!),
614 "CORRECT" => (nothing, correct!),
615 "FIX" => (nothing, fix!),
616 "PROFILE" => (nothing, profile),
617 "LEARN" => (nothing, learn!),
618 "SUGGEST" => (1, suggest),
619 "DIST" => (2, dist),
620 "SOUNDEX" => (1, soundex_code),
621 "SOUNDS" => (1, sounds),
622 "SPLIT" => (1, split_word),
623 "PREFIX" => (2, prefix),
624 "COMPLETE" => (2, complete),
625 "LIST" => (2, list_words),
626 "WORD" => (1, word_info),
627 "ADD" => (2, add!),
628 "REMOVE" => (1, remove!),
629 "IGNORE" => (1, ignore!),
630 "UNIGNORE" => (1, unignore!),
631 "RULE" => (2, rule!),
632 "UNRULE" => (1, unrule!),
633 "RULES" => (0, show_rules),
634 "KEYBOARD" => (1, keyboard_mode!),
635 "SET" => (2, set_option!),
636 "UNDO" => (0, undo!),
637 "REPORT" => (0, report),
638 "HISTORY" => (0, history),
639 "STATS" => (0, stats),
640)
641
642function execute!(checker, line)
643 parts = split(line, ' ', limit = 2)
644 name = parts[1]
645 try
646 haskey(HANDLERS, name) || throw(CommandError("UNKNOWN COMMAND $name"))
647 arity, handler = HANDLERS[name]
648 if arity === nothing
649 length(parts) == 2 || throw(CommandError("BAD ARGUMENTS $name"))
650 handler(checker, parts[2])
651 else
652 arguments = length(parts) == 2 ? split(parts[2], ' ') : SubString{String}[]
653 length(arguments) == arity || throw(CommandError("BAD ARGUMENTS $name"))
654 handler(checker, arguments)
655 end
656 catch err
657 err isa CommandError || rethrow()
658 emit("ERROR", err.message)
659 end
660end

664 word_count = parse(Int, readline())
665 for _ in 1:word_count
