string translate(string text, string source, string target)
    result := ""
    for: i < #text
        match source.find(text[i])
            nat index
                result += target[index]
            none
                result += text[i]
    return result

string lower(string text)
    return translate(text, "ABCDEFGHIJKLMNOPQRSTUVWXYZ", "abcdefghijklmnopqrstuvwxyz")
bool is_letter(string character)
    return "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz".contains(character)

bool is_digit(string character)
    return "0123456789".contains(character)

bool all_of(string text, func<bool(string)> predicate)
    for: i < #text
        if not predicate(text[i])
            return false
    return true

nat smaller(nat first, nat second)
    return if first < second then first else second

bool text_less(string first, string second)
    int[] a := first.codepoints()
    int[] b := second.codepoints()
    for: i < smaller(#a, #b)
        if a[i] != b[i]
            return a[i] < b[i]
    return #a < #b

bool contains<T>(T[] items, T item)
    for value := items[]
        if value == item
            return true
    return false

T[] head<T>(T[] items, nat count)
    return items[:<smaller(count, #items)]

T[] sorted_by<T>(T[] items, func<bool(T, T)> before)
    if #items <= 1
        return items
    middle := #items / 2
    left := sorted_by(items[:<middle], before)
    right := sorted_by(items[middle:], before)
    T[] merged := []
    nat i := 0
    nat j := 0
    while i < #left and j < #right
        if before(right[j], left[i])
            merged = merged.append(right[j])
            j += 1
        else
            merged = merged.append(left[i])
            i += 1
    return merged.concat(left[i:]).concat(right[j:])
void emit(string[] parts)
    println("{parts.join(" ")}")

string | error word_argument(string text)
    if not (1 <= #text <= 30 and all_of(text, is_letter))
        return error("BAD WORD {text}")
    return lower(text)

nat | error number_argument(string text, nat minimum, nat maximum)
    if not (1 <= #text <= 7 and all_of(text, is_digit))
        return error("BAD NUMBER {text}")
    value := try nat(text)
    if not (minimum <= value <= maximum)
        return error("BAD NUMBER {text}")
    return value
bool is_swap(int[] first, int[] second, nat i, nat j)
    return i >= 2 and j >= 2 and first[i - 1] == second[j - 2] and first[i - 2] == second[j - 1]

class EditMetric(bool keyboard)
    nat gap := if keyboard then 2 else 1
    bool[][] adjacent := array(26, fill = array(26, fill = false))
    local neighbors := [
        "qswz", "ghnv", "dfvx", "cefrsx", "drsw", "cdgrtv", "bfhtvy", "bgjnuy", "jkou",
        "hikmnu", "ijlmo", "kop", "jkn", "bhjm", "iklp", "lo", "aw", "deft",
        "adewxz", "fgry", "hijy", "bcfg", "aeqs", "cdsz", "ghtu", "asx",
    ]
    for: keys := neighbors[letter]
        for key := keys.codepoints()[]
            adjacent[letter][key - 97] = true

    nat substitution(int first, int second)
        if first == second
            return 0
        if not keyboard
            return 1
        return if adjacent[first - 97][second - 97] then 1 else 2

    nat[][] distance_table(string first, string second)
        int[] a := first.codepoints()
        int[] b := second.codepoints()
        nat[][] table := array(#a + 1, fill = array(#b + 1, fill = 0))
        for: i <= #a
            table[i][0] = i * gap
        for: j <= #b
            table[0][j] = j * gap
        for: 1 <= i <= #a
            for: 1 <= j <= #b
                best := smaller(table[i - 1][j], table[i][j - 1]) + gap
                best = smaller(best, table[i - 1][j - 1] + substitution(a[i - 1], b[j - 1]))
                if is_swap(a, b, i, j)
                    best = smaller(best, table[i - 2][j - 2] + 1)
                table[i][j] = best
        return table
    nat prefix_distance(string word, string other)
        nat[] row := distance_table(word, other)[#word]
        best := row[0]
        for value := row[]
            best = smaller(best, value)
        return best
class Candidate(string word, nat distance, nat frequency)
bool candidate_before(Candidate first, Candidate second)
    if first.distance != second.distance
        return first.distance < second.distance
    if first.frequency != second.frequency
        return first.frequency > second.frequency
    return text_less(first.word, second.word)
bool takes_text(string name)
    return contains(["CHECK", "CORRECT", "FIX", "PROFILE", "LEARN"], name)

nat | none argument_count(string name)
    if contains(["RULES", "UNDO", "REPORT", "HISTORY", "STATS"], name)
        return 0
    if contains(["DIST", "PREFIX", "COMPLETE", "LIST", "ADD", "RULE", "SET"], name)
        return 2
    single := [
        "SUGGEST", "SOUNDEX", "SOUNDS", "SPLIT", "WORD", "REMOVE", "IGNORE", "UNIGNORE", "UNRULE",
        "KEYBOARD",
    ]
    if contains(single, name)
        return 1
class SpellChecker(map<string, nat> dictionary)
    const nat max_listed := 20
    metric := EditMetric(false)
    nat base_limit := 2
    nat limit()
        return if metric.keyboard then 2 * base_limit else base_limit
    void | error complete(string[] arguments)
        word := try word_argument(arguments[0])
        count := try number_argument(arguments[1], 1, max_listed)
        nat threshold := limit() / 2
        Candidate[] matches := []
        for other := dictionary.keys()[], frequency := dictionary.values()[]
            nat distance := metric.prefix_distance(word, other)
            if distance <= threshold
                matches = matches.append(Candidate(other, distance, frequency))
        matches = sorted_by(matches, candidate_before)
        listed := [
            for match_entry := head(matches, count)[]
            then "{match_entry.word}:{match_entry.distance}"
        ]
        emit(["COMPLETE", word, "count={#matches}"].concat(listed))
        return
    void | error .keyboard_mode(string[] arguments)
        mode := arguments[0]
        if not contains(["ON", "OFF"], mode)
            return error("BAD MODE {mode}")
        metric = EditMetric(mode == "ON")
        emit(["KEYBOARD", mode])
        return

    void | error .set_option(string[] arguments)
        option := arguments[0]
        if not contains(["LIMIT", "SUGGESTIONS"], option)
            return error("BAD OPTION {option}")
        nat maximum := if option == "LIMIT" then 3 else 10
        value := try number_argument(arguments[1], 1, maximum)
        if option == "LIMIT"
            base_limit = value
        else
            suggestion_count = value
        emit(["SET", option, "{value}"])
        return
    void | error .run_command(string name, string[] arguments)
        elif name == "COMPLETE"
            try complete(arguments)
        elif name == "KEYBOARD"
            try .keyboard_mode(arguments)
        elif name == "SET"
            try .set_option(arguments)
    void | error .run(string line)
        name := line.split(" ")[0]
        separated := #name < #line
        rest := if separated then line[#name + 1:] else ""
        if takes_text(name)
            if not separated
                return error("BAD ARGUMENTS {name}")
            .run_text(name, rest)
            return
        match argument_count(name)
            nat expected
                string[] arguments := if separated then rest.split(" ") else []
                if #arguments != expected
                    return error("BAD ARGUMENTS {name}")
                try .run_command(name, arguments)
                return
    void .execute(string line)
        match .run(line)
            void
                void
            error problem
                emit(["ERROR", problem.message])
