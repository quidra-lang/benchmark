30function wordArgument(string $text): string
31{
32 if (!preg_match('/^[A-Za-z]{1,30}$/D', $text)) throw new CommandError("BAD WORD $text");
33 return strtolower($text);
34}

36function numberArgument(string $text, int $minimum, int $maximum): int
37{
38 if (!preg_match('/^[0-9]{1,7}$/D', $text)) throw new CommandError("BAD NUMBER $text");
39 $value = (int) $text;
40 if ($value < $minimum || $value > $maximum) throw new CommandError("BAD NUMBER $text");
41 return $value;
42}

44function decimalText(int $numerator, int $denominator, int $digits): string
45{
46 $scale = 10 ** $digits;
47 $value = intdiv(2 * $numerator * $scale + $denominator, 2 * $denominator);
48 return intdiv($value, $scale) . '.' . str_pad((string) ($value % $scale), $digits, '0', STR_PAD_LEFT);
49}

51function applyCase(string $original, string $word): string
52{
53 if (strlen($original) >= 2 && strtoupper($original) === $original) return strtoupper($word);
54 if ($original[0] >= 'A' && $original[0] <= 'Z') return ucfirst($word);
55 return $word;
56}

85function textTokens(string $text): array
86{
87 preg_match_all('/[A-Za-z]+/', $text, $matches, PREG_OFFSET_CAPTURE);
88 $tokens = [];
89 $previous = null;
90 $position = 0;
91 foreach ($matches[0] as [$spelling, $start]) {
92 $end = $start + strlen($spelling);
93 $token = new Token(count($tokens) + 1, $spelling, $start, $end,
94 substr($text, $position, $start - $position), $previous);
95 $tokens[] = $token;
96 $previous = $token;
97 $position = $end;
98 }
99 return $tokens;
100}

102function substitutionCost(string $first, string $second, bool $keyboard): int
103{
104 if ($first === $second) return 0;
105 if (!$keyboard) return 1;
106 return str_contains(ADJACENT_KEYS[$first], $second) ? 1 : 2;
107}

109function isSwap(string $first, string $second, int $i, int $j): bool
110{
111 return $i >= 2 && $j >= 2 &&
112 $first[$i - 1] === $second[$j - 2] && $first[$i - 2] === $second[$j - 1];
113}

115function distanceTable(string $first, string $second, bool $keyboard): array
116{
117 $gap = $keyboard ? 2 : 1;
118 $rows = strlen($first);
119 $columns = strlen($second);
120 $table = array_fill(0, $rows + 1, array_fill(0, $columns + 1, 0));
121 for ($i = 0; $i <= $rows; ++$i) $table[$i][0] = $i * $gap;
122 for ($j = 0; $j <= $columns; ++$j) $table[0][$j] = $j * $gap;
123 for ($i = 1; $i <= $rows; ++$i) {
124 for ($j = 1; $j <= $columns; ++$j) {
125 $best = min(
126 $table[$i - 1][$j] + $gap,
127 $table[$i][$j - 1] + $gap,
128 $table[$i - 1][$j - 1] + substitutionCost($first[$i - 1], $second[$j - 1], $keyboard)
129 );
130 if (isSwap($first, $second, $i, $j)) {
131 $best = min($best, $table[$i - 2][$j - 2] + 1);
132 }
133 $table[$i][$j] = $best;
134 }
135 }
136 return $table;
137}

139function editDistance(string $first, string $second, bool $keyboard): int
140{
141 return distanceTable($first, $second, $keyboard)[strlen($first)][strlen($second)];
142}
143
144function prefixDistance(string $word, string $other, bool $keyboard): int
145{
146 return min(distanceTable($word, $other, $keyboard)[strlen($word)]);
147}

149function editOperations(string $first, string $second, bool $keyboard): array
150{
151 $table = distanceTable($first, $second, $keyboard);
152 $gap = $keyboard ? 2 : 1;
153 $operations = [];
154 $i = strlen($first);
155 $j = strlen($second);
156 while ($i > 0 || $j > 0) {
157 if ($i >= 1 && $j >= 1 &&
158 $table[$i][$j] === $table[$i - 1][$j - 1] +
159 substitutionCost($first[$i - 1], $second[$j - 1], $keyboard)) {
160 if ($first[$i - 1] !== $second[$j - 1]) {
161 $operations[] = 'sub:' . $first[$i - 1] . '>' . $second[$j - 1];
162 }
163 --$i;
164 --$j;
165 } elseif (isSwap($first, $second, $i, $j) &&
166 $table[$i][$j] === $table[$i - 2][$j - 2] + 1) {
167 $operations[] = 'swap:' . $first[$i - 2] . $first[$i - 1];
168 $i -= 2;
169 $j -= 2;
170 } elseif ($i >= 1 && $table[$i][$j] === $table[$i - 1][$j] + $gap) {
171 $operations[] = 'del:' . $first[$i - 1];
172 --$i;
173 } else {
174 $operations[] = 'ins:' . $second[$j - 1];
175 --$j;
176 }
177 }
178 return [$table[strlen($first)][strlen($second)], array_reverse($operations)];
179}

181function spellSoundex(string $word): string
182{
183 static $digits = null;
184 if ($digits === null) {
185 $digits = [];
186 foreach (SOUNDEX_GROUPS as [$letters, $digit]) {
187 foreach (str_split($letters) as $letter) $digits[$letter] = $digit;
188 }
189 }
190 $code = strtoupper($word[0]);
191 $previous = str_contains('hw', $word[0]) ? '0' : $digits[$word[0]];
192 for ($i = 1; $i < strlen($word); ++$i) {
193 $letter = $word[$i];
194 if (str_contains('hw', $letter)) continue;
195 $digit = $digits[$letter];
196 if ($digit !== '0' && $digit !== $previous) $code .= $digit;
197 $previous = $digit;
198 }
199 return substr($code . '000', 0, 4);
200}

202function compareWordLists(array $first, array $second): int
203{
204 $length = min(count($first), count($second));
205 for ($i = 0; $i < $length; ++$i) {
206 $order = strcmp($first[$i], $second[$i]);
207 if ($order !== 0) return $order;
208 }
209 return count($first) <=> count($second);
210}

212function segment(string $word, array $dictionary): array
213{
214 $length = strlen($word);
215 $ways = array_fill(0, $length + 1, 0);
216 $best = array_fill(0, $length + 1, null);
217 $ways[$length] = 1;
218 $best[$length] = [];
219 for ($start = $length - 1; $start >= 0; --$start) {
220 for ($end = $start + 1; $end <= $length; ++$end) {
221 $piece = substr($word, $start, $end - $start);
222 if (!isset($dictionary[$piece]) || $ways[$end] === 0) continue;
223 $ways[$start] += $ways[$end];
224 $candidate = array_merge([$piece], $best[$end]);
225 $current = $best[$start];
226 if ($current === null || count($candidate) < count($current) ||
227 (count($candidate) === count($current) && compareWordLists($candidate, $current) < 0)) {
228 $best[$start] = $candidate;
229 }
230 }
231 }
232 return [$ways[0], $best[0]];
233}

235final class SpellChecker
236{
237 public array $output = [];
238 private array $dictionary;
239 private array $ignored = [];
240 private array $rules = [];
241 private bool $keyboard = false;
242 private int $baseLimit = 2;
243 private int $suggestionCount = 5;
244 private array $undoHistory = [];
245 private array $unknownLog = [];
246 private array $correctionLog = [];
247 private array $counters;
248
249 public function __construct(array $dictionary)
250 {
251 $this->dictionary = $dictionary;
252 $this->counters = array_fill_keys(COUNTERS, 0);
253 }
254
255 private function emit(string ...$parts): void
256 {
257 $this->output[] = implode(' ', $parts);
258 }
259
260 private function limit(): int
261 {
262 return $this->keyboard ? 2 * $this->baseLimit : $this->baseLimit;
263 }
264
265 private function isKnown(string $word): bool
266 {
267 return isset($this->dictionary[$word]) || isset($this->ignored[$word]);
268 }
269
270 private function frequencyOrder(): array
271 {
272 $words = array_keys($this->dictionary);
273 usort($words, function (string $a, string $b): int {
274 return ($this->dictionary[$b] <=> $this->dictionary[$a]) ?: strcmp($a, $b);
275 });
276 return $words;
277 }
278
279 private function requireEntry(string $word): int
280 {
281 if (!isset($this->dictionary[$word])) throw new CommandError("NOT FOUND $word");
282 return $this->dictionary[$word];
283 }
284
285 private function candidates(string $word): array
286 {
287 $limit = $this->limit();
288 $gap = $this->keyboard ? 2 : 1;
289 $found = [];
290 foreach ($this->dictionary as $other => $frequency) {
291 if ($other === $word || abs(strlen($other) - strlen($word)) * $gap > $limit) continue;
292 $distance = editDistance($word, $other, $this->keyboard);
293 if ($distance <= $limit) $found[] = [$other, $distance, $frequency];
294 }
295 usort($found, static function (array $a, array $b): int {
296 return ($a[1] <=> $b[1]) ?: (($b[2] <=> $a[2]) ?: strcmp($a[0], $b[0]));
297 });
298 return $found;
299 }
300
301 private function correction(string $word): ?string
302 {
303 if ($this->isKnown($word)) return null;
304 if (isset($this->rules[$word])) return $this->rules[$word];
305 $candidates = $this->candidates($word);
306 return $candidates ? $candidates[0][0] : null;
307 }
308
309 private function recordUndo(string $name): void
310 {
311 $this->undoHistory[] = [$name, $this->dictionary, $this->ignored, $this->rules];
312 }
313
314 private function logCorrection(string $word, string $replacement): void
315 {
316 $key = $word . '>' . $replacement;
317 $this->correctionLog[$key] = ($this->correctionLog[$key] ?? 0) + 1;
318 ++$this->counters['corrected'];
319 }
320
321 private function unknownFinding(Token $token): string
322 {
323 $prefix = 'UNKNOWN ' . $token->number . ' ' . $token->text;
324 if (isset($this->rules[$token->word])) {
325 return $prefix . ' RULE ' . $this->rules[$token->word];
326 }
327 $found = $this->candidates($token->word);
328 if (!$found) return $prefix . ' NONE';
329 $words = array_column(array_slice($found, 0, $this->suggestionCount), 0);
330 return $prefix . ' SUGGEST ' . implode(' ', $words);
331 }

333 private function cmdCheck(string $text): void
334 {
335 $tokens = textTokens($text);
336 $findings = [];
337 foreach ($tokens as $token) {
338 if (!$this->isKnown($token->word)) {
339 $findings[] = ['unknown', $this->unknownFinding($token)];
340 $this->unknownLog[$token->word] = ($this->unknownLog[$token->word] ?? 0) + 1;
341 }
342 if ($token->mixedCase()) {
343 $findings[] = ['case', 'CASE ' . $token->number . ' ' . $token->text];
344 }
345 if ($token->sentenceStart && ctype_lower($token->text[0])) {
346 $findings[] = ['capital', 'CAPITAL ' . $token->number . ' ' . $token->text];
347 }
348 if ($token->repeat) {
349 $findings[] = ['repeats', 'REPEAT ' . $token->number . ' ' . $token->text];
350 }
351 }
352 $counts = array_fill_keys(FINDINGS, 0);
353 foreach ($findings as [$kind, $_]) ++$counts[$kind];
354 $this->counters['checked'] += count($tokens);
355 foreach (FINDINGS as $kind) $this->counters[$kind] += $counts[$kind];
356 $items = ['CHECK', 'words=' . count($tokens)];
357 foreach (FINDINGS as $kind) $items[] = $kind . '=' . $counts[$kind];
358 $this->emit(...$items);
359 foreach ($findings as [$_, $line]) $this->emit($line);
360 }
361
362 private function cmdCorrect(string $text): void
363 {
364 $pieces = [];
365 $position = 0;
366 $changed = 0;
367 foreach (textTokens($text) as $token) {
368 $replacement = $this->correction($token->word);
369 if ($replacement === null) {
370 $pieces[] = $token->gap . $token->text;
371 } else {
372 $pieces[] = $token->gap . applyCase($token->text, $replacement);
373 $this->logCorrection($token->word, $replacement);
374 ++$changed;
375 }
376 $position = $token->end;
377 }
378 $pieces[] = substr($text, $position);
379 $this->emit('CORRECT', 'changed=' . $changed . ':', implode('', $pieces));
380 }
381
382 private function cmdFix(string $text): void
383 {
384 $pieces = [];
385 $position = 0;
386 $removed = 0;
387 $replaced = 0;
388 $recased = 0;
389 foreach (textTokens($text) as $token) {
390 $position = $token->end;
391 if ($token->repeat) { ++$removed; continue; }
392 $replacement = $this->correction($token->word);
393 $target = $replacement ?? $token->word;
394 $result = applyCase($token->text, $target);
395 if ($token->sentenceStart && ctype_lower($result[0])) $result = ucfirst($result);
396 if ($replacement !== null) {
397 $this->logCorrection($token->word, $replacement);
398 ++$replaced;
399 } elseif ($result !== $token->text) {
400 ++$recased;
401 }
402 $pieces[] = $token->gap . $result;
403 }
404 $pieces[] = substr($text, $position);
405 $this->emit('FIX', 'removed=' . $removed, 'replaced=' . $replaced,
406 'recased=' . $recased . ':', implode('', $pieces));
407 }

409 private function cmdProfile(string $text): void
410 {
411 $tokens = textTokens($text);
412 $seen = [];
413 $unknown = 0;
414 $sentences = 0;
415 $characters = 0;
416 $longest = '-';
417 $maxLength = 0;
418 foreach ($tokens as $token) {
419 $seen[$token->word] = true;
420 if (!$this->isKnown($token->word)) ++$unknown;
421 if ($token->sentenceStart) ++$sentences;
422 $characters += strlen($token->text);
423 if (strlen($token->text) > $maxLength) {
424 $maxLength = strlen($token->text);
425 $longest = $token->word;
426 }
427 }
428 $count = count($tokens);
429 $rate = $count ? decimalText(100 * $unknown, $count, 1) : '0.0';
430 $average = $count ? decimalText($characters, $count, 2) : '0.00';
431 $this->emit('PROFILE', 'words=' . $count, 'distinct=' . count($seen),
432 'unknown=' . $unknown, 'rate=' . $rate, 'avglen=' . $average,
433 'longest=' . $longest, 'sentences=' . $sentences);
434 }

436 private function cmdLearn(string $text): void
437 {
438 $this->recordUndo('LEARN');
439 $updated = 0; $added = 0; $skipped = 0;
440 foreach (textTokens($text) as $token) {
441 if (strlen($token->word) > 30) {
442 ++$skipped;
443 } elseif (isset($this->dictionary[$token->word])) {
444 ++$this->dictionary[$token->word];
445 ++$updated;
446 } else {
447 $this->dictionary[$token->word] = 1;
448 unset($this->ignored[$token->word]);
449 ++$added;
450 }
451 }
452 $this->emit('LEARN', 'updated=' . $updated, 'added=' . $added, 'skipped=' . $skipped);
453 }

628 private function cmdRules(): void
629 {

706 public function execute(string $line): void
707 {
708 $parts = explode(' ', $line, 2);
709 $name = $parts[0];
710 $hasArguments = count($parts) === 2;
711 $rest = $parts[1] ?? '';
712 try {
713 if (!in_array($name, TEXT_COMMANDS, true) && !array_key_exists($name, ARITIES)) {
714 throw new CommandError("UNKNOWN COMMAND $name");
715 }
716 if (in_array($name, TEXT_COMMANDS, true)) {
717 if (!$hasArguments) throw new CommandError("BAD ARGUMENTS $name");
718 match ($name) {
719 'CHECK' => $this->cmdCheck($rest),
720 'CORRECT' => $this->cmdCorrect($rest),
721 'FIX' => $this->cmdFix($rest),
722 'PROFILE' => $this->cmdProfile($rest),
723 'LEARN' => $this->cmdLearn($rest),
724 };
725 return;
726 }
727 $args = $hasArguments ? explode(' ', $rest) : [];
728 if (count($args) !== ARITIES[$name]) throw new CommandError("BAD ARGUMENTS $name");
729 match ($name) {
730 'SUGGEST' => $this->cmdSuggest($args),
731 'DIST' => $this->cmdDist($args),
732 'SOUNDEX' => $this->cmdSoundex($args),
733 'SOUNDS' => $this->cmdSounds($args),
734 'SPLIT' => $this->cmdSplit($args),
735 'PREFIX' => $this->cmdPrefix($args),
736 'COMPLETE' => $this->cmdComplete($args),
737 'LIST' => $this->cmdList($args),
738 'WORD' => $this->cmdWord($args),
739 'ADD' => $this->cmdAdd($args),
740 'REMOVE' => $this->cmdRemove($args),
741 'IGNORE' => $this->cmdIgnore($args),
742 'UNIGNORE' => $this->cmdUnignore($args),
743 'RULE' => $this->cmdRule($args),
744 'UNRULE' => $this->cmdUnrule($args),
745 'RULES' => $this->cmdRules(),
746 'KEYBOARD' => $this->cmdKeyboard($args),
747 'SET' => $this->cmdSet($args),
748 'UNDO' => $this->cmdUndo(),
749 'REPORT' => $this->cmdReport(),
750 'HISTORY' => $this->cmdHistory(),
751 'STATS' => $this->cmdStats(),
752 };
753 } catch (CommandError $error) {
754 $this->emit('ERROR', $error->getMessage());
755 }
756 }
