41decimal_text <- function(numerator, denominator, digits) {
42 scale <- 10^digits
43 value <- (2 * numerator * scale + denominator) %/% (2 * denominator)

76tokenize <- function(text) {
77 matches <- gregexpr("[[:alpha:]]+", text)[[1]]
78 tokens <- list()
79 if (matches[1] == -1L) {
80 return(tokens)
81 }
82 match_lengths <- attr(matches, "match.length")
83 position <- 1L
84 for (index in seq_along(matches)) {
85 start <- matches[[index]]
86 last <- start + match_lengths[[index]] - 1L
87 token_text <- substr(text, start, last)
88 word <- tolower(token_text)
89 gap <- substr(text, position, start - 1L)
90 tokens[[index]] <- list(
91 number = index,
92 text = token_text,
93 word = word,
94 last = last,
95 gap = gap,
96 sentence_start = index == 1L || grepl("[.!?]", gap),
97 is_repeat = index > 1L && word == tokens[[index - 1L]]$word && grepl("^ *$", gap)
98 )
99 position <- last + 1L
100 }
101 tokens
102}

107 if (x == y) {

118is_swap <- function(a, b, i, j) {
119 i >= 2L && j >= 2L && a[i] == b[j - 1L] && a[i - 1L] == b[j]
120}

122distance_table <- function(a, b, keyboard) {
123 gap <- gap_cost(keyboard)
124 rows <- length(a)
125 columns <- length(b)
126 table <- matrix(0L, rows + 1L, columns + 1L)
127 table[, 1] <- (0:rows) * gap
128 table[1, ] <- (0:columns) * gap
129 for (i in seq_len(rows)) {
130 for (j in seq_len(columns)) {
131 best <- min(
132 table[i, j + 1L] + gap,
133 table[i + 1L, j] + gap,
134 table[i, j] + substitution_cost(a[i], b[j], keyboard)
135 )
136 if (is_swap(a, b, i, j)) {
137 best <- min(best, table[i - 1L, j - 1L] + 1L)
138 }
139 table[i + 1L, j + 1L] <- best
140 }
141 }
142 table
143}

145edit_distance <- function(a, b, keyboard) {
146 table <- distance_table(a, b, keyboard)
147 table[nrow(table), ncol(table)]
148}
149
150prefix_distance <- function(a, b, keyboard) {
151 table <- distance_table(a, b, keyboard)
152 min(table[nrow(table), ])
153}
154
155edit_operations <- function(a, b, keyboard) {
156 table <- distance_table(a, b, keyboard)
157 gap <- gap_cost(keyboard)
158 operations <- character(0)
159 i <- length(a)
160 j <- length(b)
161 while (i > 0L || j > 0L) {
162 if (i >= 1L && j >= 1L &&
163 table[i + 1L, j + 1L] == table[i, j] + substitution_cost(a[i], b[j], keyboard)) {
164 if (a[i] != b[j]) {
165 operations <- c(operations, paste0("sub:", a[i], ">", b[j]))
166 }
167 i <- i - 1L
168 j <- j - 1L
169 } else if (is_swap(a, b, i, j) && table[i + 1L, j + 1L] == table[i - 1L, j - 1L] + 1L) {
170 operations <- c(operations, paste0("swap:", a[i - 1L], a[i]))
171 i <- i - 2L
172 j <- j - 2L
173 } else if (i >= 1L && table[i + 1L, j + 1L] == table[i, j + 1L] + gap) {
174 operations <- c(operations, paste0("del:", a[i]))
175 i <- i - 1L
176 } else {
177 operations <- c(operations, paste0("ins:", b[j]))
178 j <- j - 1L
179 }
180 }
181 list(distance = table[nrow(table), ncol(table)], operations = rev(operations))
182}

184soundex <- function(word) {
185 chars <- split_letters(word)
186 code <- toupper(chars[1])
187 previous <- if (chars[1] %in% c("h", "w")) "0" else SOUNDEX_DIGITS[[chars[1]]]
188 for (letter in chars[-1]) {
189 if (letter %in% c("h", "w")) next
190 digit <- SOUNDEX_DIGITS[[letter]]
191 if (digit != "0" && digit != previous) {
192 code <- paste0(code, digit)
193 }
194 previous <- digit
195 }
196 substr(paste0(code, "000"), 1L, 4L)
197}

199precedes <- function(candidate, current) {
200 if (length(candidate) != length(current)) {
201 return(length(candidate) < length(current))
202 }
203 differs <- candidate != current
204 any(differs) && candidate[differs][1] < current[differs][1]
205}

207segment <- function(word, words) {
208 size <- nchar(word)
209 ways <- integer(size + 1L)
210 best <- vector("list", size + 1L)
211 ways[size + 1L] <- 1L
212 best[[size + 1L]] <- character(0)
213 for (start in rev(seq_len(size))) {
214 for (last in start:size) {
215 piece <- substr(word, start, last)
216 if (!(piece %in% words) || ways[last + 1L] == 0L) next
217 ways[start] <- ways[start] + ways[last + 1L]
218 candidate <- c(piece, best[[last + 1L]])
219 if (is.null(best[[start]]) || precedes(candidate, best[[start]])) {
220 best[[start]] <- candidate
221 }
222 }
223 }
224 list(ways = ways[1], best = best[[1]])
225}

264candidates <- function(checker, word) {
265 max_distance <- limit(checker)
266 gap <- gap_cost(checker$keyboard)
267 others <- names(checker$dictionary)
268 others <- others[others != word & abs(nchar(others) - nchar(word)) * gap <= max_distance]
269 word_letters <- split_letters(word)
270 distances <- vapply(others, function(other) {
271 edit_distance(word_letters, split_letters(other), checker$keyboard)
272 }, integer(1), USE.NAMES = FALSE)
273 keep <- distances <= max_distance
274 rank_by_distance(checker, others[keep], distances[keep])
275}

312check_text <- function(checker, text) {
313 tokens <- tokenize(text)
314 counts <- setNames(integer(length(FINDING_KINDS)), FINDING_KINDS)
315 findings <- character(0)
316 add_finding <- function(kind, line) {
317 counts[[kind]] <<- counts[[kind]] + 1L
318 findings <<- c(findings, line)
319 }
320 for (token in tokens) {
321 if (!is_known(checker, token$word)) {
322 add_finding("unknown", unknown_finding(checker, token))
323 checker$unknown_log <- increment(checker$unknown_log, token$word)
324 }
325 if (is_mixed_case(token$text)) {
326 add_finding("case", paste("CASE", token$number, token$text))
327 }
328 if (token$sentence_start && grepl("^[[:lower:]]", token$text)) {
329 add_finding("capital", paste("CAPITAL", token$number, token$text))
330 }
331 if (token$is_repeat) {
332 add_finding("repeats", paste("REPEAT", token$number, token$text))
333 }
334 }
335 checker$counters[["checked"]] <- checker$counters[["checked"]] + length(tokens)
336 checker$counters[FINDING_KINDS] <- checker$counters[FINDING_KINDS] + counts
337 emit("CHECK", paste0("words=", length(tokens)), paste0(FINDING_KINDS, "=", counts))
338 writeLines(findings)
339}

341correct_text <- function(checker, text) {
342 pieces <- character(0)
343 position <- 0L
344 changed <- 0L
345 for (token in tokenize(text)) {
346 replacement <- correction(checker, token$word)
347 if (is.null(replacement)) {
348 pieces <- c(pieces, token$gap, token$text)
349 } else {
350 pieces <- c(pieces, token$gap, apply_case(token$text, replacement))
351 log_correction(checker, token$word, replacement)
352 changed <- changed + 1L
353 }
354 position <- token$last
355 }
356 pieces <- c(pieces, substring(text, position + 1L))
357 emit(paste0("CORRECT changed=", changed, ":"), paste(pieces, collapse = ""))
358}

360fix_text <- function(checker, text) {
361 pieces <- character(0)
362 position <- 0L
363 removed <- 0L
364 replaced <- 0L
365 recased <- 0L
366 for (token in tokenize(text)) {
367 position <- token$last
368 if (token$is_repeat) {
369 removed <- removed + 1L
370 next
371 }
372 replacement <- correction(checker, token$word)
373 result <- apply_case(token$text, replacement %||% token$word)
374 if (token$sentence_start && grepl("^[[:lower:]]", result)) {
375 result <- capitalize_first(result)
376 }
377 if (!is.null(replacement)) {
378 log_correction(checker, token$word, replacement)
379 replaced <- replaced + 1L
380 } else if (result != token$text) {
381 recased <- recased + 1L
382 }
383 pieces <- c(pieces, token$gap, result)
384 }
385 pieces <- c(pieces, substring(text, position + 1L))
386 emit(
387 "FIX", paste0("removed=", removed), paste0("replaced=", replaced),
388 paste0("recased=", recased, ":"), paste(pieces, collapse = "")
389 )
390}

392profile_text <- function(checker, text) {
393 tokens <- tokenize(text)
394 words <- vapply(tokens, function(token) token$word, character(1))
395 unknown <- sum(!vapply(words, function(word) is_known(checker, word), logical(1)))
396 sentences <- sum(vapply(tokens, function(token) token$sentence_start, logical(1)))
397 if (length(tokens) == 0L) {
398 rate <- "0.0"
399 average <- "0.00"
400 longest <- "-"
401 } else {
402 rate <- decimal_text(100 * unknown, length(tokens), 1L)
403 average <- decimal_text(sum(nchar(words)), length(tokens), 2L)
404 longest <- words[which.max(nchar(words))]
405 }
406 emit(
407 "PROFILE", paste0("words=", length(tokens)), paste0("distinct=", length(unique(words))),
408 paste0("unknown=", unknown), paste0("rate=", rate), paste0("avglen=", average),
409 paste0("longest=", longest), paste0("sentences=", sentences)
410 )
411}

683execute <- function(checker, line) {
684 name <- sub(" .*", "", line)
685 rest <- if (grepl(" ", line, fixed = TRUE)) sub("^[^ ]* ", "", line)
686 tryCatch(
687 {
688 if (!(name %in% names(COMMANDS))) command_error(paste("UNKNOWN COMMAND", name))
689 command <- COMMANDS[[name]]
690 if (is.na(command$arity)) {
691 if (is.null(rest)) command_error(paste("BAD ARGUMENTS", name))
692 command$handler(checker, rest)
693 } else {
694 arguments <- if (is.null(rest)) character(0) else strsplit(rest, " ", fixed = TRUE)[[1]]
695 if (length(arguments) != command$arity) command_error(paste("BAD ARGUMENTS", name))
696 command$handler(checker, arguments)
697 }
698 },
699 command_error = function(condition) emit("ERROR", conditionMessage(condition))
700 )
701}
