53func decimalText(_ numerator: Int, _ denominator: Int, _ digits: Int) -> String {
54 let scale = repeatElement(10, count: digits).reduce(1, *)
55 let value = (2 * numerator * scale + denominator) / (2 * denominator)
56 let fraction = String(value % scale)
57 return "\(value / scale).\(String(repeating: "0", count: digits - fraction.count))\(fraction)"
58}

74func applyCase(_ tokenText: String, _ word: String) -> String {
75 if tokenText.count >= 2 && tokenText.allSatisfy(\.isUppercase) {

88 let end: Int
89 let gap: String

98func tokenize(_ text: String) -> [Token] {
99 let characters = Array(text)
100 var tokens: [Token] = []
101 var position = 0
102 var index = 0
103 while index < characters.count {
104 guard characters[index].isLetter else {
105 index += 1
106 continue
107 }
108 let start = index
109 while index < characters.count && characters[index].isLetter {
110 index += 1
111 }
112 let gap = String(characters[position..<start])
113 let tokenText = String(characters[start..<index])
114 let word = tokenText.lowercased()
115 let previous = tokens.last
116 tokens.append(
117 Token(
118 number: tokens.count + 1, text: tokenText, word: word, end: index, gap: gap,
119 sentenceStart: previous == nil || gap.contains { sentenceMarks.contains($0) },
120 isRepeat: previous?.word == word && gap.allSatisfy { $0 == " " }))
121 position = index
122 }
123 return tokens
124}

126func gapCost(_ keyboard: Bool) -> Int {
127 keyboard ? 2 : 1
128}
129
130func substitutionCost(_ first: Character, _ second: Character, _ keyboard: Bool) -> Int {
131 if first == second {
132 return 0
133 }
134 if !keyboard {
135 return 1
136 }
137 return adjacentKeys[first, default: ""].contains(second) ? 1 : 2
138}

140func isSwap(_ first: [Character], _ second: [Character], _ i: Int, _ j: Int) -> Bool {
141 i >= 2 && j >= 2 && first[i - 1] == second[j - 2] && first[i - 2] == second[j - 1]
142}
143
144func distanceTable(_ first: [Character], _ second: [Character], _ keyboard: Bool) -> [[Int]] {
145 let gap = gapCost(keyboard)
146 let row = Array(repeating: 0, count: second.count + 1)
147 var table = Array(repeating: row, count: first.count + 1)
148 for i in 0...first.count {
149 table[i][0] = i * gap
150 }
151 for j in 0...second.count {
152 table[0][j] = j * gap
153 }
154 for i in 1...first.count {
155 for j in 1...second.count {
156 let substitution = substitutionCost(first[i - 1], second[j - 1], keyboard)
157 var best = min(
158 table[i - 1][j] + gap, table[i][j - 1] + gap, table[i - 1][j - 1] + substitution)
159 if isSwap(first, second, i, j) {
160 best = min(best, table[i - 2][j - 2] + 1)
161 }
162 table[i][j] = best
163 }
164 }
165 return table
166}
167
168func editDistance(_ first: String, _ second: String, _ keyboard: Bool) -> Int {
169 distanceTable(Array(first), Array(second), keyboard)[first.count][second.count]
170}
171
172func prefixDistance(_ word: String, _ other: String, _ keyboard: Bool) -> Int {
173 distanceTable(Array(word), Array(other), keyboard)[word.count].min()!
174}
175
176func editOperations(_ first: String, _ second: String, _ keyboard: Bool) -> (Int, [String]) {
177 let first = Array(first)
178 let second = Array(second)
179 let table = distanceTable(first, second, keyboard)
180 let gap = gapCost(keyboard)
181 var operations: [String] = []
182 var i = first.count
183 var j = second.count
184 while i > 0 || j > 0 {
185 if i >= 1 && j >= 1
186 && table[i][j]
187 == table[i - 1][j - 1] + substitutionCost(first[i - 1], second[j - 1], keyboard)
188 {
189 if first[i - 1] != second[j - 1] {
190 operations.append("sub:\(first[i - 1])>\(second[j - 1])")
191 }
192 i -= 1
193 j -= 1
194 } else if isSwap(first, second, i, j) && table[i][j] == table[i - 2][j - 2] + 1 {
195 operations.append("swap:\(first[i - 2])\(first[i - 1])")
196 i -= 2
197 j -= 2
198 } else if i >= 1 && table[i][j] == table[i - 1][j] + gap {
199 operations.append("del:\(first[i - 1])")
200 i -= 1
201 } else {
202 operations.append("ins:\(second[j - 1])")
203 j -= 1
204 }
205 }
206 return (table[first.count][second.count], operations.reversed())
207}

209func soundex(_ word: String) -> String {
210 var code = word.prefix(1).uppercased()
211 var previous = soundexDigits[word.first!, default: "0"]
212 for letter in word.dropFirst() where letter != "h" && letter != "w" {
213 let digit = soundexDigits[letter, default: "0"]
214 if digit != "0" && digit != previous {
215 code.append(digit)
216 }
217 previous = digit
218 }
219 return String((code + "000").prefix(4))
220}

222func segment(_ word: String, _ dictionary: [String: Int]) -> (ways: Int, best: [String]?) {
223 let characters = Array(word)
224 let length = characters.count
225 var ways = Array(repeating: 0, count: length + 1)
226 var best: [[String]?] = Array(repeating: nil, count: length + 1)
227 ways[length] = 1
228 best[length] = []
229 for start in stride(from: length - 1, through: 0, by: -1) {
230 for end in (start + 1)...length {
231 let piece = String(characters[start..<end])
232 guard dictionary[piece] != nil, let rest = best[end] else {
233 continue
234 }
235 ways[start] += ways[end]
236 let candidate = [piece] + rest
237 if let current = best[start], current.count <= candidate.count {
238 continue
239 }
240 best[start] = candidate
241 }
242 }
243 return (ways[0], best[0])
244}

303 private func candidates(_ word: String) -> [Match] {
304 let gap = gapCost(keyboard)
305 var found: [Match] = []
306 for (other, frequency) in dictionary
307 where other != word && abs(other.count - word.count) * gap <= limit {
308 let distance = editDistance(word, other, keyboard)
309 if distance <= limit {
310 found.append((distance, frequency, other))
311 }
312 }
313 return inDistanceOrder(found)
314 }

316 private func correction(_ word: String) -> String? {
317 guard !isKnown(word) else { return nil }
318 if let target = rules[word] {
319 return target
320 }
321 return candidates(word).first?.word

324 private func recordUndo(_ name: String) {
325 undoHistory.append(

329 private func logCorrection(_ word: String, _ replacement: String) {
330 correctionLog[CorrectionPair(word: word, correction: replacement), default: 0] += 1
331 counters["corrected", default: 0] += 1

334 private func unknownFinding(_ token: Token) -> String {
335 let prefix = "UNKNOWN \(token.number) \(token.text)"
336 if let target = rules[token.word] {
337 return "\(prefix) RULE \(target)"
338 }
339 let found = candidates(token.word)
340 if found.isEmpty {
341 return "\(prefix) NONE"
342 }
343 let words = found.prefix(suggestionCount).map(\.word)
344 return ([prefix, "SUGGEST"] + words).joined(separator: " ")

347 private func check(_ text: String) {
348 let tokens = tokenize(text)
349 var findings: [(kind: String, line: String)] = []
350 for token in tokens {
351 if !isKnown(token.word) {
352 findings.append(("unknown", unknownFinding(token)))
353 unknownLog[token.word, default: 0] += 1
354 }
355 if token.mixedCase {
356 findings.append(("case", "CASE \(token.number) \(token.text)"))
357 }
358 if token.sentenceStart && token.text.first?.isLowercase == true {
359 findings.append(("capital", "CAPITAL \(token.number) \(token.text)"))
360 }
361 if token.isRepeat {
362 findings.append(("repeats", "REPEAT \(token.number) \(token.text)"))
363 }
364 }
365 let counts = findingKinds.map { kind in findings.filter { $0.kind == kind }.count }
366 counters["checked", default: 0] += tokens.count
367 for (kind, count) in zip(findingKinds, counts) {
368 counters[kind, default: 0] += count
369 }
370 emit(["CHECK", "words=\(tokens.count)"] + zip(findingKinds, counts).map { "\($0)=\($1)" })
371 output.append(contentsOf: findings.map(\.line))
372 }

374 private func correct(_ text: String) {
375 var result = ""
376 var position = 0
377 var changed = 0
378 for token in tokenize(text) {
379 result += token.gap
380 if let replacement = correction(token.word) {
381 result += applyCase(token.text, replacement)
382 logCorrection(token.word, replacement)
383 changed += 1
384 } else {
385 result += token.text
386 }
387 position = token.end
388 }
389 result.append(contentsOf: text.dropFirst(position))
390 emit(["CORRECT changed=\(changed):", result])
391 }
392
393 private func fix(_ text: String) {
394 var result = ""
395 var position = 0
396 var removed = 0
397 var replaced = 0
398 var recased = 0
399 for token in tokenize(text) {
400 position = token.end
401 if token.isRepeat {
402 removed += 1
403 continue
404 }
405 let replacement = correction(token.word)
406 var fixed = applyCase(token.text, replacement ?? token.word)
407 if token.sentenceStart && fixed.first?.isLowercase == true {
408 fixed = capitalizeFirst(fixed)
409 }
410 if let replacement {
411 logCorrection(token.word, replacement)
412 replaced += 1
413 } else if fixed != token.text {
414 recased += 1
415 }
416 result += token.gap + fixed
417 }
418 result.append(contentsOf: text.dropFirst(position))
419 emit(["FIX", "removed=\(removed)", "replaced=\(replaced)", "recased=\(recased):", result])
420 }

422 private func profile(_ text: String) {
423 let tokens = tokenize(text)
424 let distinct = Set(tokens.map(\.word)).count
425 let unknown = tokens.filter { !isKnown($0.word) }.count
426 let sentences = tokens.filter(\.sentenceStart).count
427 var rate = "0.0"
428 var average = "0.00"
429 var longest = "-"
430 if let longestToken = tokens.max(by: { $0.text.count < $1.text.count }) {
431 let characters = tokens.map(\.text.count).reduce(0, +)
432 rate = decimalText(100 * unknown, tokens.count, 1)
433 average = decimalText(characters, tokens.count, 2)
434 longest = longestToken.word
435 }
436 emit([
437 "PROFILE", "words=\(tokens.count)", "distinct=\(distinct)", "unknown=\(unknown)",
438 "rate=\(rate)", "avglen=\(average)", "longest=\(longest)", "sentences=\(sentences)",
439 ])
440 }

442 private func learn(_ text: String) {
443 recordUndo("LEARN")
444 var updated = 0
445 var added = 0
446 var skipped = 0
447 for token in tokenize(text) {
448 if token.word.count > maxWordLength {
449 skipped += 1
450 } else if dictionary[token.word] != nil {
451 dictionary[token.word, default: 0] += 1
452 updated += 1
453 } else {
454 dictionary[token.word] = 1
455 ignored.remove(token.word)
456 added += 1
457 }
458 }
459 emit(["LEARN", "updated=\(updated)", "added=\(added)", "skipped=\(skipped)"])
460 }

683 private func handler(for name: String) -> Handler? {
684 switch name {
685 case "CHECK": .text(check)
686 case "CORRECT": .text(correct)
687 case "FIX": .text(fix)
688 case "PROFILE": .text(profile)
689 case "LEARN": .text(learn)
690 case "SUGGEST": .words(1, suggest)
691 case "DIST": .words(2, dist)
692 case "SOUNDEX": .words(1, soundexCode)
693 case "SOUNDS": .words(1, sounds)
694 case "SPLIT": .words(1, split)

716 func execute(_ line: String) {
717 let parts = line.split(separator: " ", maxSplits: 1, omittingEmptySubsequences: false)
718 let name = String(parts[0])
719 let rest = parts.count > 1 ? String(parts[1]) : nil
720 do {
721 switch handler(for: name) {
722 case nil:
723 throw CommandError("UNKNOWN COMMAND \(name)")
724 case .text(let run)?:
725 guard let rest else {
726 throw CommandError("BAD ARGUMENTS \(name)")
727 }
728 run(rest)
729 case .words(let arity, let run)?:
730 let arguments =
731 rest?.split(separator: " ", omittingEmptySubsequences: false).map(String.init)
732 ?? []
733 guard arguments.count == arity else {
734 throw CommandError("BAD ARGUMENTS \(name)")
735 }
736 try run(arguments)
737 }
738 } catch {
739 emit(["ERROR \(error)"])
740 }
741 }
742}
