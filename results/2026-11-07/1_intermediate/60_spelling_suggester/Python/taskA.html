42 if re.fullmatch(r"[0-9]{1,7}", text) is None:

50def decimal_text(numerator, denominator, digits):
51 scale = 10 ** digits
52 value = (2 * numerator * scale + denominator) // (2 * denominator)
53 return f"{value // scale}.{value % scale:0{digits}d}"

73class Token:
74 def __init__(self, number, text, start, end, gap, previous):
75 self.number = number
76 self.text = text
77 self.word = text.lower()
78 self.start = start
79 self.end = end
80 self.gap = gap
81 self.sentence_start = previous is None or any(
82 mark in gap for mark in SENTENCE_MARKS
83 )
84 self.repeat = (
85 previous is not None
86 and self.word == previous.word
87 and all(character == " " for character in gap)
88 )
89
90 def mixed_case(self):
91 has_inner_upper = any(character.isupper() for character in self.text[1:])
92 return has_inner_upper and not self.text.isupper()

95def tokenize(text):
96 tokens = []
97 previous = None
98 position = 0
99 for match in re.finditer(r"[A-Za-z]+", text):
100 token = Token(
101 len(tokens) + 1, match.group(), match.start(), match.end(),
102 text[position:match.start()], previous,
103 )
104 tokens.append(token)
105 previous = token
106 position = match.end()
107 return tokens

110def gap_cost(keyboard):
111 return 2 if keyboard else 1
112
113
114def substitution_cost(first, second, keyboard):
115 if first == second:
116 return 0
117 if not keyboard:
118 return 1
119 return 1 if second in ADJACENT_KEYS[first] else 2

122def is_swap(first, second, i, j):
123 return (
124 i >= 2 and j >= 2
125 and first[i - 1] == second[j - 2]
126 and first[i - 2] == second[j - 1]
127 )
128
129
130def distance_table(first, second, keyboard):
131 gap = gap_cost(keyboard)
132 rows, columns = len(first), len(second)
133 table = [[0] * (columns + 1) for _ in range(rows + 1)]
134 for i in range(rows + 1):
135 table[i][0] = i * gap
136 for j in range(columns + 1):
137 table[0][j] = j * gap
138 for i in range(1, rows + 1):
139 for j in range(1, columns + 1):
140 best = min(
141 table[i - 1][j] + gap,
142 table[i][j - 1] + gap,
143 table[i - 1][j - 1] + substitution_cost(first[i - 1], second[j - 1], keyboard),
144 )
145 if is_swap(first, second, i, j):
146 best = min(best, table[i - 2][j - 2] + 1)
147 table[i][j] = best
148 return table
149
150
151def edit_distance(first, second, keyboard):
152 return distance_table(first, second, keyboard)[len(first)][len(second)]
153
154
155def prefix_distance(word, other, keyboard):
156 return min(distance_table(word, other, keyboard)[len(word)])
157
158
159def edit_operations(first, second, keyboard):
160 table = distance_table(first, second, keyboard)
161 gap = gap_cost(keyboard)
162 operations = []
163 i, j = len(first), len(second)
164 while i > 0 or j > 0:
165 if i >= 1 and j >= 1 and table[i][j] == table[i - 1][j - 1] + substitution_cost(
166 first[i - 1], second[j - 1], keyboard
167 ):
168 if first[i - 1] != second[j - 1]:
169 operations.append(f"sub:{first[i - 1]}>{second[j - 1]}")
170 i, j = i - 1, j - 1
171 elif is_swap(first, second, i, j) and table[i][j] == table[i - 2][j - 2] + 1:
172 operations.append(f"swap:{first[i - 2]}{first[i - 1]}")
173 i, j = i - 2, j - 2
174 elif i >= 1 and table[i][j] == table[i - 1][j] + gap:
175 operations.append(f"del:{first[i - 1]}")
176 i -= 1
177 else:
178 operations.append(f"ins:{second[j - 1]}")
179 j -= 1
180 operations.reverse()
181 return table[len(first)][len(second)], operations

184def soundex(word):
185 code = word[0].upper()
186 previous = "0" if word[0] in "hw" else SOUNDEX_DIGITS[word[0]]
187 for letter in word[1:]:
188 if letter in "hw":
189 continue
190 digit = SOUNDEX_DIGITS[letter]
191 if digit != "0" and digit != previous:
192 code += digit
193 previous = digit
194 return (code + "000")[:4]

197def segment(word, dictionary):
198 length = len(word)
199 ways = [0] * (length + 1)
200 best = [None] * (length + 1)
201 ways[length] = 1
202 best[length] = []
203 for start in range(length - 1, -1, -1):
204 for end in range(start + 1, length + 1):
205 piece = word[start:end]
206 if piece not in dictionary or ways[end] == 0:
207 continue
208 ways[start] += ways[end]
209 candidate = [piece] + best[end]
210 current = best[start]
211 if current is None or (len(candidate), candidate) < (len(current), current):
212 best[start] = candidate
213 return ways[0], best[0]

270 def limit(self):
271 return 2 * self.base_limit if self.keyboard else self.base_limit

273 def is_known(self, word):
274 return word in self.dictionary or word in self.ignored

276 def frequency_order(self):
277 entries = sorted(self.dictionary.items(), key=frequency_key)
278 return [word for word, _ in entries]

280 def require_entry(self, word):
281 if word not in self.dictionary:
282 raise CommandError(f"NOT FOUND {word}")
283 return self.dictionary[word]

285 def candidates(self, word):
286 limit = self.limit()
287 gap = gap_cost(self.keyboard)
288 found = []
289 for other, frequency in self.dictionary.items():
290 if other == word or abs(len(other) - len(word)) * gap > limit:
291 continue
292 distance = edit_distance(word, other, self.keyboard)
293 if distance <= limit:
294 found.append((distance, -frequency, other))
295 found.sort()
296 return [(other, distance) for distance, _, other in found]

298 def correction(self, word):
299 if self.is_known(word):
300 return None
301 if word in self.rules:
302 return self.rules[word]
303 found = self.candidates(word)
304 return found[0][0] if found else None

306 def record_undo(self, name):
307 record = UndoRecord(name, self.dictionary, self.ignored, self.rules)
308 self.undo_history.append(record)

310 def log_correction(self, word, replacement):
311 key = (word, replacement)
312 self.correction_log[key] = self.correction_log.get(key, 0) + 1
313 self.counters["corrected"] += 1

315 def unknown_finding(self, token):
316 prefix = f"UNKNOWN {token.number} {token.text}"
317 if token.word in self.rules:
318 return f"{prefix} RULE {self.rules[token.word]}"
319 found = self.candidates(token.word)
320 if not found:
321 return f"{prefix} NONE"
322 words = [other for other, _ in found[:self.suggestion_count]]
323 return " ".join([prefix, "SUGGEST"] + words)

325 def check(self, text):
326 tokens = tokenize(text)
327 findings = []
328 for token in tokens:
329 if not self.is_known(token.word):
330 findings.append(("unknown", self.unknown_finding(token)))
331 self.unknown_log[token.word] = self.unknown_log.get(token.word, 0) + 1
332 if token.mixed_case():
333 findings.append(("case", f"CASE {token.number} {token.text}"))
334 if token.sentence_start and token.text[0].islower():
335 findings.append(("capital", f"CAPITAL {token.number} {token.text}"))
336 if token.repeat:
337 findings.append(("repeats", f"REPEAT {token.number} {token.text}"))
338 counts = {
339 kind: sum(found == kind for found, _ in findings) for kind in FINDING_KINDS
340 }
341 self.counters["checked"] += len(tokens)
342 for kind in FINDING_KINDS:
343 self.counters[kind] += counts[kind]
344 self.emit(
345 "CHECK", f"words={len(tokens)}",
346 *(f"{kind}={counts[kind]}" for kind in FINDING_KINDS),
347 )
348 for _, line in findings:
349 self.emit(line)

351 def correct(self, text):
352 pieces = []
353 position = 0
354 changed = 0
355 for token in tokenize(text):
356 replacement = self.correction(token.word)
357 if replacement is None:
358 pieces.append(token.gap + token.text)
359 else:
360 pieces.append(token.gap + apply_case(token.text, replacement))
361 self.log_correction(token.word, replacement)
362 changed += 1
363 position = token.end
364 pieces.append(text[position:])
365 self.emit(f"CORRECT changed={changed}:", "".join(pieces))
366
367 def fix(self, text):
368 pieces = []
369 position = 0
370 removed = 0
371 replaced = 0
372 recased = 0
373 for token in tokenize(text):
374 position = token.end
375 if token.repeat:
376 removed += 1
377 continue
378 replacement = self.correction(token.word)
379 target = token.word if replacement is None else replacement
380 result = apply_case(token.text, target)
381 if token.sentence_start and result[0].islower():
382 result = capitalize_first(result)
383 if replacement is not None:
384 self.log_correction(token.word, replacement)
385 replaced += 1
386 elif result != token.text:
387 recased += 1
388 pieces.append(token.gap + result)
389 pieces.append(text[position:])
390 self.emit(
391 "FIX", f"removed={removed}", f"replaced={replaced}",
392 f"recased={recased}:", "".join(pieces),
393 )

395 def profile(self, text):
396 tokens = tokenize(text)
397 distinct = len({token.word for token in tokens})
398 unknown = sum(not self.is_known(token.word) for token in tokens)
399 sentences = sum(token.sentence_start for token in tokens)
400 if tokens:
401 characters = sum(len(token.text) for token in tokens)
402 rate = decimal_text(100 * unknown, len(tokens), 1)
403 average = decimal_text(characters, len(tokens), 2)
404 longest = max(tokens, key=lambda token: len(token.text)).word
405 else:
406 rate, average, longest = "0.0", "0.00", "-"
407 self.emit(
408 "PROFILE", f"words={len(tokens)}", f"distinct={distinct}",
409 f"unknown={unknown}", f"rate={rate}", f"avglen={average}",
410 f"longest={longest}", f"sentences={sentences}",
411 )

413 def learn(self, text):
414 self.record_undo("LEARN")
415 updated = 0
416 added = 0
417 skipped = 0
418 for token in tokenize(text):
419 if len(token.word) > MAX_WORD_LENGTH:
420 skipped += 1
421 elif token.word in self.dictionary:
422 self.dictionary[token.word] += 1
423 updated += 1
424 else:
425 self.dictionary[token.word] = 1
426 self.ignored.discard(token.word)
427 added += 1
428 self.emit("LEARN", f"updated={updated}", f"added={added}", f"skipped={skipped}")

430 def suggest(self, arguments):
431 word = word_argument(arguments[0])
432 if self.is_known(word):
433 self.emit("SUGGEST", word, "KNOWN")
434 elif word in self.rules:
435 self.emit("SUGGEST", word, "RULE", self.rules[word])
436 else:
437 found = self.candidates(word)
438 self.emit(
439 "SUGGEST", word, f"found={len(found)}",
440 *(f"{other}:{distance}" for other, distance in found[:self.suggestion_count]),
441 )

444 first = word_argument(arguments[0])
445 second = word_argument(arguments[1])
446 distance, operations = edit_operations(first, second, self.keyboard)
447 self.emit("DIST", first, second, distance, *operations)

454 word = word_argument(arguments[0])
455 code = soundex(word)

462 def split(self, arguments):
463 word = word_argument(arguments[0])
464 ways, best = segment(word, self.dictionary)
465 if ways == 0:
466 self.emit("SPLIT", word, "ways=0", "->", "NONE")
467 else:
468 self.emit("SPLIT", word, f"ways={ways}", "->", *best)

470 def prefix(self, arguments):
471 word = word_argument(arguments[0])
472 count = number_argument(arguments[1], 1, MAX_LISTED)
473 matches = [other for other in self.frequency_order() if other.startswith(word)]
474 self.emit(
475 "PREFIX", word, f"count={len(matches)}",
476 *(f"{other}:{self.dictionary[other]}" for other in matches[:count]),

479 def complete(self, arguments):
480 word = word_argument(arguments[0])
481 count = number_argument(arguments[1], 1, MAX_LISTED)
482 threshold = self.limit() // 2
483 matches = []
484 for other, frequency in self.dictionary.items():
485 distance = prefix_distance(word, other, self.keyboard)
486 if distance <= threshold:
487 matches.append((distance, -frequency, other))
488 matches.sort()
489 self.emit(
490 "COMPLETE", word, f"count={len(matches)}",
491 *(f"{other}:{distance}" for distance, _, other in matches[:count]),
492 )

513 word = word_argument(arguments[0])
514 frequency = number_argument(arguments[1], 1, MAX_FREQUENCY)

547 def rule(self, arguments):
548 source = word_argument(arguments[0])
549 target = word_argument(arguments[1])
550 if source in self.dictionary:
551 raise CommandError(f"IN DICTIONARY {source}")
552 if target not in self.dictionary:
553 raise CommandError(f"NOT FOUND {target}")
554 self.record_undo("RULE")
555 previous = self.rules.get(source)
556 self.rules[source] = target
557 if previous is None:
558 self.emit("RULE", source, target)
559 else:
560 self.emit("RULE", source, target, "REPLACED", previous)

562 def unrule(self, arguments):
563 word = word_argument(arguments[0])

565 raise CommandError(f"NO RULE {word}")
566 self.record_undo("UNRULE")
567 target = self.rules.pop(word)
568 self.emit("UNRULE", word, target)

620 def stats(self, arguments):
621 top, top_frequency = min(self.dictionary.items(), key=frequency_key)
622 characters = sum(len(word) for word in self.dictionary)

624 "STATS", f"words={len(self.dictionary)}",

634 def execute(self, line):
635 name, separator, rest = line.partition(" ")
636 try:
637 if name not in self.handlers:
638 raise CommandError(f"UNKNOWN COMMAND {name}")
639 arity, handler = self.handlers[name]
640 if arity is None:
641 if not separator:
642 raise CommandError(f"BAD ARGUMENTS {name}")
643 handler(rest)
644 else:
645 arguments = rest.split(" ") if separator else []
646 if len(arguments) != arity:
647 raise CommandError(f"BAD ARGUMENTS {name}")
648 handler(arguments)
649 except CommandError as error:
650 self.emit("ERROR", error)
