98String decimalText(int numerator, int denominator, int digits) {
99 final scale = pow(10, digits).toInt();
100 final value = (2 * numerator * scale + denominator) ~/ (2 * denominator);
101 final fraction = (value % scale).toString().padLeft(digits, '0');
102 return '${value ~/ scale}.$fraction';
103}

105int byDecreasingValue(MapEntry<String, int> a, MapEntry<String, int> b) =>
106 a.value != b.value ? b.value - a.value : a.key.compareTo(b.key);

108int compareCandidates(Candidate a, Candidate b) {
109 if (a.distance != b.distance) {
110 return a.distance - b.distance;
111 }
112 if (a.frequency != b.frequency) {
113 return b.frequency - a.frequency;
114 }
115 return a.word.compareTo(b.word);
116}

125String applyCase(String tokenText, String word) {
126 if (tokenText.length >= 2 && tokenText == tokenText.toUpperCase()) {
127 return word.toUpperCase();
128 }
129 if (isUpperCase(tokenText[0])) {
130 return capitalizeFirst(word);
131 }
132 return word;

159List<Token> tokenize(String text) {
160 final tokens = <Token>[];
161 var position = 0;
162 for (final match in letterRun.allMatches(text)) {
163 final tokenText = match[0]!;
164 final word = tokenText.toLowerCase();
165 final gap = text.substring(position, match.start);
166 final previous = tokens.lastOrNull;
167 tokens.add(
168 Token(
169 tokens.length + 1,
170 tokenText,
171 word,
172 match.end,
173 gap,
174 sentenceStart: previous == null || gap.contains(sentenceMark),
175 repeat:
176 previous != null &&
177 word == previous.word &&
178 gap == ' ' * gap.length,
179 ),
180 );
181 position = match.end;
182 }
183 return tokens;
184}

186int gapCost(bool keyboard) => keyboard ? 2 : 1;

188int substitutionCost(String first, String second, bool keyboard) {
189 if (first == second) {
190 return 0;
191 }
192 if (!keyboard) {
193 return 1;
194 }
195 return adjacentKeys[first]!.contains(second) ? 1 : 2;
196}

198bool isSwap(String first, String second, int i, int j) =>
199 i >= 2 &&
200 j >= 2 &&
201 first[i - 1] == second[j - 2] &&
202 first[i - 2] == second[j - 1];
203
204List<List<int>> distanceTable(String first, String second, bool keyboard) {
205 final gap = gapCost(keyboard);
206 final rows = first.length;
207 final columns = second.length;
208 final table = List.generate(rows + 1, (_) => List.filled(columns + 1, 0));
209 for (var i = 0; i <= rows; i++) {
210 table[i][0] = i * gap;
211 }
212 for (var j = 0; j <= columns; j++) {
213 table[0][j] = j * gap;
214 }
215 for (var i = 1; i <= rows; i++) {
216 for (var j = 1; j <= columns; j++) {
217 var best = min(table[i - 1][j], table[i][j - 1]) + gap;
218 best = min(
219 best,
220 table[i - 1][j - 1] +
221 substitutionCost(first[i - 1], second[j - 1], keyboard),
222 );
223 if (isSwap(first, second, i, j)) {
224 best = min(best, table[i - 2][j - 2] + 1);
225 }
226 table[i][j] = best;
227 }
228 }
229 return table;
230}
231
232int editDistance(String first, String second, bool keyboard) =>
233 distanceTable(first, second, keyboard)[first.length][second.length];
234
235int prefixDistance(String word, String other, bool keyboard) =>
236 distanceTable(word, other, keyboard)[word.length].reduce(min);

238(int, List<String>) editOperations(String first, String second, bool keyboard) {
239 final table = distanceTable(first, second, keyboard);
240 final gap = gapCost(keyboard);
241 final operations = <String>[];
242 var i = first.length;
243 var j = second.length;
244 while (i > 0 || j > 0) {
245 if (i >= 1 &&
246 j >= 1 &&
247 table[i][j] ==
248 table[i - 1][j - 1] +
249 substitutionCost(first[i - 1], second[j - 1], keyboard)) {
250 if (first[i - 1] != second[j - 1]) {
251 operations.add('sub:${first[i - 1]}>${second[j - 1]}');
252 }
253 i--;
254 j--;
255 } else if (isSwap(first, second, i, j) &&
256 table[i][j] == table[i - 2][j - 2] + 1) {
257 operations.add('swap:${first[i - 2]}${first[i - 1]}');
258 i -= 2;
259 j -= 2;
260 } else if (i >= 1 && table[i][j] == table[i - 1][j] + gap) {
261 operations.add('del:${first[i - 1]}');
262 i--;
263 } else {
264 operations.add('ins:${second[j - 1]}');
265 j--;
266 }
267 }
268 return (table[first.length][second.length], operations.reversed.toList());
269}

271String soundex(String word) {
272 final first = word[0];
273 var code = first.toUpperCase();
274 var previous = 'hw'.contains(first) ? '0' : soundexDigits[first]!;
275 for (final letter in word.substring(1).split('')) {
276 if ('hw'.contains(letter)) {
277 continue;
278 }
279 final digit = soundexDigits[letter]!;
280 if (digit != '0' && digit != previous) {
281 code += digit;
282 }
283 previous = digit;
284 }
285 return '${code}000'.substring(0, 4);
286}
287
288int compareSegmentations(List<String> first, List<String> second) {
289 if (first.length != second.length) {
290 return first.length - second.length;
291 }
292 for (var i = 0; i < first.length; i++) {
293 final order = first[i].compareTo(second[i]);
294 if (order != 0) {
295 return order;
296 }
297 }
298 return 0;
299}

301(int, List<String>?) segment(String word, Map<String, int> dictionary) {
302 final length = word.length;
303 final ways = List.filled(length + 1, 0);
304 final best = List<List<String>?>.filled(length + 1, null);
305 ways[length] = 1;
306 best[length] = [];
307 for (var start = length - 1; start >= 0; start--) {
308 for (var end = start + 1; end <= length; end++) {
309 final piece = word.substring(start, end);
310 if (!dictionary.containsKey(piece) || ways[end] == 0) {
311 continue;
312 }
313 ways[start] += ways[end];
314 final candidate = [piece, ...best[end]!];
315 final current = best[start];
316 if (current == null || compareSegmentations(candidate, current) < 0) {
317 best[start] = candidate;
318 }
319 }
320 }
321 return (ways[0], best[0]);
322}

385 int requireEntry(String word) =>

388 List<Candidate> candidates(String word) {
389 final gap = gapCost(keyboard);
390 final found = <Candidate>[];
391 for (final MapEntry(key: other, value: frequency) in dictionary.entries) {
392 if (other == word || (other.length - word.length).abs() * gap > limit) {
393 continue;
394 }
395 final distance = editDistance(word, other, keyboard);
396 if (distance <= limit) {
397 found.add((word: other, frequency: frequency, distance: distance));
398 }
399 }
400 return found..sort(compareCandidates);
401 }

403 String? correction(String word) {
404 if (isKnown(word)) {
405 return null;
406 }
407 if (rules[word] case final target?) {
408 return target;
409 }
410 return candidates(word).firstOrNull?.word;

431 String unknownFinding(Token token) {
432 final prefix = 'UNKNOWN ${token.number} ${token.text}';
433 if (rules[token.word] case final target?) {
434 return '$prefix RULE $target';
435 }
436 final found = candidates(token.word);
437 if (found.isEmpty) {
438 return '$prefix NONE';
439 }
440 return [
441 prefix,
442 'SUGGEST',
443 for (final candidate in found.take(suggestionCount)) candidate.word,
444 ].join(' ');

447 void check(String text) {
448 final tokens = tokenize(text);
449 final findings = <(String, String)>[];
450 for (final token in tokens) {
451 if (!isKnown(token.word)) {
452 findings.add(('unknown', unknownFinding(token)));
453 unknownLog.update(token.word, (count) => count + 1, ifAbsent: () => 1);
454 }
455 if (token.mixedCase) {
456 findings.add(('case', 'CASE ${token.number} ${token.text}'));
457 }
458 if (token.sentenceStart && isLowerCase(token.text[0])) {
459 findings.add(('capital', 'CAPITAL ${token.number} ${token.text}'));
460 }
461 if (token.repeat) {
462 findings.add(('repeats', 'REPEAT ${token.number} ${token.text}'));
463 }
464 }
465 final counts = {
466 for (final kind in findingKinds)
467 kind: findings.where((finding) => finding.$1 == kind).length,
468 };
469 counters.update('checked', (count) => count + tokens.length);
470 for (final kind in findingKinds) {
471 counters.update(kind, (count) => count + counts[kind]!);
472 }
473 emit('CHECK words=${tokens.length}', [
474 for (final kind in findingKinds) '$kind=${counts[kind]}',
475 ]);
476 for (final (_, line) in findings) {
477 emit(line);
478 }
479 }

481 void correct(String text) {
482 final result = StringBuffer();
483 var position = 0;
484 var changed = 0;
485 for (final token in tokenize(text)) {
486 final replacement = correction(token.word);
487 if (replacement == null) {
488 result.write('${token.gap}${token.text}');
489 } else {
490 result.write('${token.gap}${applyCase(token.text, replacement)}');
491 logCorrection(token.word, replacement);
492 changed++;
493 }
494 position = token.end;
495 }
496 result.write(text.substring(position));
497 emit('CORRECT changed=$changed: $result');
498 }
499
500 void fix(String text) {
501 final result = StringBuffer();
502 var position = 0;
503 var removed = 0;
504 var replaced = 0;
505 var recased = 0;
506 for (final token in tokenize(text)) {
507 position = token.end;
508 if (token.repeat) {
509 removed++;
510 continue;
511 }
512 final replacement = correction(token.word);
513 var fixed = applyCase(token.text, replacement ?? token.word);
514 if (token.sentenceStart && isLowerCase(fixed[0])) {
515 fixed = capitalizeFirst(fixed);
516 }
517 if (replacement != null) {
518 logCorrection(token.word, replacement);
519 replaced++;
520 } else if (fixed != token.text) {
521 recased++;
522 }
523 result.write('${token.gap}$fixed');
524 }
525 result.write(text.substring(position));
526 emit('FIX removed=$removed replaced=$replaced recased=$recased: $result');
527 }

529 void profile(String text) {
530 final tokens = tokenize(text);
531 final distinct = tokens.map((token) => token.word).toSet().length;
532 final unknown = tokens.where((token) => !isKnown(token.word)).length;
533 final sentences = tokens.where((token) => token.sentenceStart).length;
534 final characters = tokens.fold(0, (sum, token) => sum + token.text.length);
535 final (rate, average, longest) = tokens.isEmpty
536 ? ('0.0', '0.00', '-')
537 : (
538 decimalText(100 * unknown, tokens.length, 1),
539 decimalText(characters, tokens.length, 2),
540 tokens.reduce((a, b) => b.text.length > a.text.length ? b : a).word,
541 );
542 emit('PROFILE', [
543 'words=${tokens.length}',
544 'distinct=$distinct',
545 'unknown=$unknown',
546 'rate=$rate',
547 'avglen=$average',
548 'longest=$longest',
549 'sentences=$sentences',
550 ]);
551 }

553 void learn(String text) {
554 recordUndo('LEARN');
555 var updated = 0;
556 var added = 0;
557 var skipped = 0;
558 for (final token in tokenize(text)) {
559 if (token.word.length > maxWordLength) {
560 skipped++;
561 } else if (dictionary.containsKey(token.word)) {
562 dictionary.update(token.word, (frequency) => frequency + 1);
563 updated++;
564 } else {
565 dictionary[token.word] = 1;
566 ignored.remove(token.word);
567 added++;
568 }
569 }
570 emit('LEARN updated=$updated added=$added skipped=$skipped');
571 }

635 void complete(List<String> arguments) {
636 final word = wordArgument(arguments[0]);
637 final count = numberArgument(arguments[1], 1, maxListed);
638 final threshold = limit ~/ 2;
639 final matches = <Candidate>[];
640 for (final MapEntry(key: other, value: frequency) in dictionary.entries) {
641 final distance = prefixDistance(word, other, keyboard);
642 if (distance <= threshold) {
643 matches.add((word: other, frequency: frequency, distance: distance));
644 }
645 }
646 matches.sort(compareCandidates);
647 emit('COMPLETE $word count=${matches.length}', [
648 for (final match in matches.take(count))
649 '${match.word}:${match.distance}',
650 ]);
651 }

783 final record = undoHistory.removeLast();
784 dictionary = record.dictionary;
785 ignored = record.ignored;
786 rules = record.rules;
787 emit('UNDO ${record.name}');
788 }

798 void history() {
799 final entries = correctionLog.entries.toList()
800 ..sort((a, b) {
801 if (a.value != b.value) {
802 return b.value - a.value;
803 }
804 final order = a.key.$1.compareTo(b.key.$1);
805 return order != 0 ? order : a.key.$2.compareTo(b.key.$2);
806 });
807 emit('HISTORY distinct=${entries.length}', [
808 for (final MapEntry(key: (word, replacement), value: count)
809 in entries.take(reportSize))
810 '$word>$replacement:$count',
811 ]);
812 }

841 void execute(String line) {
842 final space = line.indexOf(' ');
843 final name = space < 0 ? line : line.substring(0, space);
844 final rest = space < 0 ? null : line.substring(space + 1);
845 try {
846 if (textCommands[name] case final handler?) {
847 if (rest == null) {
848 throw CommandError('BAD ARGUMENTS $name');
849 }
850 handler(rest);
851 } else if (argumentCommands[name] case (final arity, final handler)?) {
852 final arguments = rest?.split(' ') ?? [];
853 if (arguments.length != arity) {
854 throw CommandError('BAD ARGUMENTS $name');
855 }
856 handler(arguments);
857 } else {
858 throw CommandError('UNKNOWN COMMAND $name');
859 }
860 } on CommandError catch (error) {
861 emit('ERROR ${error.message}');
862 }
863 }
864}
