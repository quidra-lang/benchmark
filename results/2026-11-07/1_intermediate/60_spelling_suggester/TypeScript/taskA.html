44function wordArgument(text: string): string {
45 if (!/^[A-Za-z]{1,30}$/.test(text)) throw new CommandError("BAD WORD " + text);
46 return text.toLowerCase();

49function numberArgument(text: string, minimum: number, maximum: number): number {
50 if (!/^[0-9]{1,7}$/.test(text)) throw new CommandError("BAD NUMBER " + text);
51 const value = Number(text);
52 if (value < minimum || value > maximum)
53 throw new CommandError("BAD NUMBER " + text);
54 return value;

57function decimalText(numerator: number, denominator: number, digits: number): string {
58 const scale = 10 ** digits;
59 const rounded = Math.floor((2 * numerator * scale + denominator) / (2 * denominator));
60 return Math.floor(rounded / scale) + "." + String(rounded % scale).padStart(digits, "0");
61}

63function applyCase(token: string, replacement: string): string {
64 if (token.length >= 2 && token === token.toUpperCase()) return replacement.toUpperCase();
65 if (/[A-Z]/.test(token[0])) return replacement[0].toUpperCase() + replacement.slice(1);
66 return replacement;
67}

69function tokens(text: string): WordToken[] {
70 const result: WordToken[] = [];
71 let previous: WordToken | null = null, lastEnd = 0;
72 for (const match of text.matchAll(/[A-Za-z]+/g)) {
73 const spelling = match[0];
74 const start = match.index ?? 0;
75 const gap = text.slice(lastEnd, start);
76 const word = spelling.toLowerCase();
77 const token = {
78 number: result.length + 1,
79 text: spelling,
80 word,
81 start,
82 end: start + spelling.length,
83 gap,
84 sentenceStart: previous === null || /[.!?]/.test(gap),
85 repeat: previous !== null && word === previous.word && /^ *$/.test(gap),
86 mixedCase: /[A-Z]/.test(spelling.slice(1)) && spelling !== spelling.toUpperCase()
87 };
88 result.push(token);
89 previous = token;
90 lastEnd = token.end;
91 }
92 return result;
93}

95const gapCost = (keyboard: boolean): number => keyboard ? 2 : 1;
96function subCost(a: string, b: string, keyboard: boolean): number {
97 return a === b ? 0 : !keyboard ? 1 : ADJACENT[a].includes(b) ? 1 : 2;
98}
99function swapped(a: string, b: string, i: number, j: number): boolean {
100 return i >= 2 && j >= 2 && a[i - 1] === b[j - 2] && a[i - 2] === b[j - 1];
101}
102
103function distanceTable(a: string, b: string, keyboard: boolean): number[][] {
104 const gap = gapCost(keyboard), p = a.length, q = b.length;
105 const d = Array.from({length: p + 1}, () => Array(q + 1).fill(0));
106 for (let i = 0; i <= p; ++i) d[i][0] = i * gap;
107 for (let j = 0; j <= q; ++j) d[0][j] = j * gap;
108 for (let i = 1; i <= p; ++i) {
109 for (let j = 1; j <= q; ++j) {
110 d[i][j] = Math.min(
111 d[i - 1][j] + gap,
112 d[i][j - 1] + gap,
113 d[i - 1][j - 1] + subCost(a[i - 1], b[j - 1], keyboard)
114 );
115 if (swapped(a, b, i, j)) d[i][j] = Math.min(d[i][j], d[i - 2][j - 2] + 1);
116 }
117 }
118 return d;
119}
120
121function editDistance(a: string, b: string, keyboard: boolean): number {
122 return distanceTable(a, b, keyboard)[a.length][b.length];
123}
124
125function prefixDistance(a: string, b: string, keyboard: boolean): number {
126 return Math.min(...distanceTable(a, b, keyboard)[a.length]);
127}
128
129function editOperations(a: string, b: string, keyboard: boolean): [number, string[]] {
130 const d = distanceTable(a, b, keyboard);
131 const gap = gapCost(keyboard), operations: string[] = [];
132 let i = a.length, j = b.length;
133 while (i > 0 || j > 0) {
134 if (i > 0 && j > 0 &&
135 d[i][j] === d[i - 1][j - 1] + subCost(a[i - 1], b[j - 1], keyboard)) {
136 if (a[i - 1] !== b[j - 1]) operations.push("sub:" + a[i - 1] + ">" + b[j - 1]);
137 --i; --j;
138 } else if (swapped(a, b, i, j) && d[i][j] === d[i - 2][j - 2] + 1) {
139 operations.push("swap:" + a.slice(i - 2, i));
140 i -= 2; j -= 2;
141 } else if (i > 0 && d[i][j] === d[i - 1][j] + gap) {
142 operations.push("del:" + a[i - 1]);
143 --i;
144 } else {
145 operations.push("ins:" + b[j - 1]);
146 --j;
147 }
148 }
149 operations.reverse();
150 return [d[a.length][b.length], operations];
151}

153function soundex(word: string): string {
154 let code = word[0].toUpperCase();
155 let previous = "hw".includes(word[0]) ? "0" : SOUND_DIGITS[word[0]];
156 for (const letter of word.slice(1)) {
157 if ("hw".includes(letter)) continue;
158 const digit = SOUND_DIGITS[letter];
159 if (digit !== "0" && digit !== previous) code += digit;
160 previous = digit;
161 }
162 return (code + "000").slice(0, 4);
163}

165function compareSequence(a: string[], b: string[]): number {
166 for (let i = 0; i < Math.min(a.length, b.length); ++i) {
167 const result = cmp(a[i], b[i]);
168 if (result !== 0) return result;
169 }
170 return a.length - b.length;
171}

173function segment(word: string, dictionary: Map<string, number>): [number, string[] | null] {
174 const n = word.length;
175 const ways: number[] = Array(n + 1).fill(0), best: (string[] | null)[] = Array(n + 1).fill(null);
176 ways[n] = 1; best[n] = [];
177 for (let start = n - 1; start >= 0; --start) {
178 for (let end = start + 1; end <= n; ++end) {
179 const piece = word.slice(start, end);
180 if (!dictionary.has(piece) || ways[end] === 0) continue;
181 ways[start] += ways[end];
182 const candidate = [piece, ...best[end]!];
183 if (best[start] === null || candidate.length < best[start].length ||
184 (candidate.length === best[start].length && compareSequence(candidate, best[start]) < 0))
185 best[start] = candidate;
186 }
187 }
188 return [ways[0], best[0]];
189}

191class SpellChecker {
192 dictionary: Map<string, number>;
193 ignored: Set<string>;
194 rules: Map<string, string>;
195 keyboard: boolean;
196 baseLimit: number;
197 suggestionCount: number;
198 undoHistory: UndoRecord[];
199 unknownLog: Map<string, number>;
200 correctionLog: Map<string, number>;
201 counters: Record<string, number>;
202 output: string[];

204 constructor(dictionary: Map<string, number>) {
205 this.dictionary = dictionary;
206 this.ignored = new Set();
207 this.rules = new Map();
208 this.keyboard = false;
209 this.baseLimit = 2;
210 this.suggestionCount = 5;
211 this.undoHistory = [];
212 this.unknownLog = new Map();
213 this.correctionLog = new Map();
214 this.counters = Object.fromEntries(COUNTERS.map(key => [key, 0]));
215 this.output = [];
216 }

218 emit(...parts: (string | number)[]): void {
219 this.output.push(parts.join(" "));

221
222 get limit(): number {
223 return this.keyboard ? this.baseLimit * 2 : this.baseLimit;
224 }
225
226 known(word: string): boolean {
227 return this.dictionary.has(word) || this.ignored.has(word);
228 }
229
230 frequencyOrder(): string[] {
231 return [...this.dictionary.keys()].sort((a, b) =>
232 this.dictionary.get(b)! - this.dictionary.get(a)! || cmp(a, b));
233 }
234
235 requireEntry(word: string): number {
236 if (!this.dictionary.has(word)) throw new CommandError("NOT FOUND " + word);
237 return this.dictionary.get(word)!;
238 }
239
240 candidates(word: string): Candidate[] {
241 const gap = gapCost(this.keyboard), limit = this.limit;
242 const found: Candidate[] = [];
243 for (const [other, frequency] of this.dictionary) {
244 if (other === word || Math.abs(other.length - word.length) * gap > limit) continue;
245 const distance = editDistance(word, other, this.keyboard);
246 if (distance <= limit) found.push({word: other, distance, frequency});
247 }
248 found.sort((a, b) =>
249 a.distance - b.distance || b.frequency - a.frequency || cmp(a.word, b.word));
250 return found;
251 }
252
253 correction(word: string): string | null {
254 if (this.known(word)) return null;
255 if (this.rules.has(word)) return this.rules.get(word)!;
256 const found = this.candidates(word);
257 return found.length > 0 ? found[0].word : null;
258 }
259
260 saveUndo(name: string): void {
261 this.undoHistory.push({
262 name, dictionary: new Map(this.dictionary),
263 ignored: new Set(this.ignored), rules: new Map(this.rules)
264 });
265 }

267 logCorrection(original: string, replacement: string): void {
268 const key = original + "\u0000" + replacement;
269 this.correctionLog.set(key, (this.correctionLog.get(key) ?? 0) + 1);
270 ++this.counters.corrected;
271 }
272
273 unknownFinding(token: WordToken): string {
274 const prefix = "UNKNOWN " + token.number + " " + token.text;
275 if (this.rules.has(token.word))
276 return prefix + " RULE " + this.rules.get(token.word)!;
277 const found = this.candidates(token.word);
278 if (found.length === 0) return prefix + " NONE";
279 return prefix + " SUGGEST " +
280 found.slice(0, this.suggestionCount).map(item => item.word).join(" ");
281 }

283 checkText(text: string): void {
284 const list = tokens(text), findings: string[] = [];
285 const totals = Object.fromEntries(FINDINGS.map(key => [key, 0]));
286 for (const token of list) {
287 if (!this.known(token.word)) {
288 findings.push(this.unknownFinding(token));
289 ++totals.unknown;
290 this.unknownLog.set(token.word, (this.unknownLog.get(token.word) ?? 0) + 1);
291 }
292 if (token.mixedCase) {
293 findings.push("CASE " + token.number + " " + token.text);
294 ++totals.case;
295 }
296 if (token.sentenceStart && /^[a-z]/.test(token.text)) {
297 findings.push("CAPITAL " + token.number + " " + token.text);
298 ++totals.capital;
299 }
300 if (token.repeat) {
301 findings.push("REPEAT " + token.number + " " + token.text);
302 ++totals.repeats;
303 }
304 }
305 this.counters.checked += list.length;
306 for (const key of FINDINGS) this.counters[key] += totals[key];
307 this.emit("CHECK", "words=" + list.length,
308 ...FINDINGS.map(key => key + "=" + totals[key]));
309 for (const finding of findings) this.emit(finding);
310 }
311
312 correctText(text: string): void {
313 let position = 0, changed = 0, result = "";
314 for (const token of tokens(text)) {
315 const replacement = this.correction(token.word);
316 if (replacement === null) {
317 result += token.gap + token.text;
318 } else {
319 result += token.gap + applyCase(token.text, replacement);
320 this.logCorrection(token.word, replacement);
321 ++changed;
322 }
323 position = token.end;
324 }
325 result += text.slice(position);
326 this.emit("CORRECT", "changed=" + changed + ":", result);
327 }
328
329 fixText(text: string): void {
330 let position = 0, removed = 0, replaced = 0, recased = 0, result = "";
331 for (const token of tokens(text)) {
332 position = token.end;
333 if (token.repeat) {
334 ++removed;
335 continue;
336 }
337 const replacement = this.correction(token.word);
338 const target = replacement === null ? token.word : replacement;
339 let converted = applyCase(token.text, target);
340 if (token.sentenceStart && /^[a-z]/.test(converted))
341 converted = converted[0].toUpperCase() + converted.slice(1);
342 if (replacement !== null) {
343 this.logCorrection(token.word, replacement);
344 ++replaced;
345 } else if (converted !== token.text) {
346 ++recased;
347 }
348 result += token.gap + converted;
349 }
350 result += text.slice(position);
351 this.emit("FIX", "removed=" + removed, "replaced=" + replaced,
352 "recased=" + recased + ":", result);
353 }

355 profileText(text: string): void {
356 const list = tokens(text);
357 const distinct = new Set(list.map(t => t.word)).size;
358 const unknown = list.filter(t => !this.known(t.word)).length;
359 const sentences = list.filter(t => t.sentenceStart).length;
360 let rate = "0.0", average = "0.00", longest = "-";
361 if (list.length > 0) {
362 rate = decimalText(100 * unknown, list.length, 1);
363 average = decimalText(list.reduce((total, t) => total + t.text.length, 0), list.length, 2);
364 longest = list.reduce((best, token) =>
365 token.text.length > best.text.length ? token : best).word;
366 }
367 this.emit("PROFILE", "words=" + list.length, "distinct=" + distinct,
368 "unknown=" + unknown, "rate=" + rate, "avglen=" + average,
369 "longest=" + longest, "sentences=" + sentences);
370 }

372 learnText(text: string): void {
373 this.saveUndo("LEARN");
374 let updated = 0, added = 0, skipped = 0;
375 for (const token of tokens(text)) {
376 if (token.word.length > 30) {
377 ++skipped;
378 } else if (this.dictionary.has(token.word)) {
379 this.dictionary.set(token.word, this.dictionary.get(token.word)! + 1);
380 ++updated;
381 } else {
382 this.dictionary.set(token.word, 1);
383 this.ignored.delete(token.word);
384 ++added;
385 }
386 }
387 this.emit("LEARN", "updated=" + updated, "added=" + added, "skipped=" + skipped);
388 }

400 }

434 complete([raw, countText]: string[]): void {
435 const word = wordArgument(raw), count = numberArgument(countText, 1, 20);
436 const threshold = Math.floor(this.limit / 2);
437 const found: Candidate[] = [];
438 for (const [other, frequency] of this.dictionary) {
439 const distance = prefixDistance(word, other, this.keyboard);
440 if (distance <= threshold) found.push({word: other, distance, frequency});
441 }
442 found.sort((a, b) => a.distance - b.distance ||
443 b.frequency - a.frequency || cmp(a.word, b.word));
444 this.emit("COMPLETE", word, "count=" + found.length,
445 ...found.slice(0, count).map(item => item.word + ":" + item.distance));
446 }

578 execute(line: string): void {
579 const separator = line.indexOf(" ");
580 const name = separator === -1 ? line : line.slice(0, separator);
581 const rest = separator === -1 ? "" : line.slice(separator + 1);
582 try {
583 if (!TEXT_COMMANDS.has(name) && !Object.hasOwn(ARITIES, name))
584 throw new CommandError("UNKNOWN COMMAND " + name);
585 if (TEXT_COMMANDS.has(name)) {
586 if (separator === -1) throw new CommandError("BAD ARGUMENTS " + name);
587 switch (name) {
588 case "CHECK": this.checkText(rest); break;
589 case "CORRECT": this.correctText(rest); break;
590 case "FIX": this.fixText(rest); break;
591 case "PROFILE": this.profileText(rest); break;
592 case "LEARN": this.learnText(rest); break;
593 }
594 return;
595 }
596 const argumentsList = separator === -1 ? [] : rest.split(" ");
597 if (argumentsList.length !== ARITIES[name])
598 throw new CommandError("BAD ARGUMENTS " + name);
599 switch (name) {
600 case "SUGGEST": this.suggest(argumentsList); break;
601 case "DIST": this.dist(argumentsList); break;
602 case "SOUNDEX": this.soundexCode(argumentsList); break;
603 case "SOUNDS": this.sounds(argumentsList); break;

609 case "ADD": this.add(argumentsList); break;

617 case "SET": this.setOption(argumentsList); break;
618 case "UNDO": this.undo(); break;
619 case "REPORT": this.report(); break;
620 case "HISTORY": this.history(); break;
621 case "STATS": this.stats(); break;
622 }
623 } catch (error) {
624 if (!(error instanceof CommandError)) throw error;
625 this.emit("ERROR", error.message);
626 }
