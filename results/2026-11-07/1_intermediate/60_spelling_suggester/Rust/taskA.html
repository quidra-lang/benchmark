22fn word_argument(text: &str) -> Result<String, String> {
23 if (1..=MAX_WORD_LENGTH).contains(&text.len()) && text.bytes().all(|c| c.is_ascii_alphabetic())
24 {
25 Ok(text.to_ascii_lowercase())
26 } else {
27 Err(format!("BAD WORD {text}"))
28 }
29}

31fn number_argument(text: &str, minimum: u32, maximum: u32) -> Result<u32, String> {
32 let valid = (1..=7).contains(&text.len()) && text.bytes().all(|c| c.is_ascii_digit());
33 match text.parse::<u32>() {
34 Ok(value) if valid && (minimum..=maximum).contains(&value) => Ok(value),
35 _ => Err(format!("BAD NUMBER {text}")),
36 }
37}

39fn decimal_text(numerator: usize, denominator: usize, digits: u32) -> String {
40 let scale = 10usize.pow(digits);
41 let value = (2 * numerator * scale + denominator) / (2 * denominator);
42 format!(
43 "{}.{:0width$}",
44 value / scale,
45 value % scale,
46 width = digits as usize
47 )
48}

58fn is_all_upper(text: &str) -> bool {
59 text.bytes().all(|c| c.is_ascii_uppercase())
60}

62fn capitalize_first(word: &str) -> String {
63 let mut result = word.to_string();
64 result[..1].make_ascii_uppercase();
65 result

68fn apply_case(token_text: &str, word: &str) -> String {
69 if token_text.len() >= 2 && is_all_upper(token_text) {
70 word.to_ascii_uppercase()
71 } else if token_text.starts_with(|c: char| c.is_ascii_uppercase()) {
72 capitalize_first(word)
73 } else {
74 word.to_string()
75 }

94fn tokenize(text: &str) -> Vec<Token> {
95 let bytes = text.as_bytes();
96 let mut tokens: Vec<Token> = Vec::new();
97 let mut position = 0;
98 let mut index = 0;
99 while index < bytes.len() {
100 if !bytes[index].is_ascii_alphabetic() {
101 index += 1;
102 continue;
103 }
104 let start = index;
105 while index < bytes.len() && bytes[index].is_ascii_alphabetic() {
106 index += 1;
107 }
108 let gap = &text[position..start];
109 let word = text[start..index].to_ascii_lowercase();
110 let previous = tokens.last();
111 let sentence_start = previous.is_none() || gap.contains(SENTENCE_MARKS);
112 let repeat = previous.is_some_and(|previous| previous.word == word)
113 && gap.bytes().all(|c| c == b' ');
114 tokens.push(Token {
115 number: tokens.len() + 1,
116 text: text[start..index].to_string(),
117 word,
118 end: index,
119 gap: gap.to_string(),
120 sentence_start,
121 repeat,
122 });
123 position = index;
124 }
125 tokens
126}

128fn gap_cost(keyboard: bool) -> usize {
129 if keyboard { 2 } else { 1 }
130}
131
132fn substitution_cost(first: u8, second: u8, keyboard: bool) -> usize {
133 if first == second {
134 0
135 } else if !keyboard {
136 1
137 } else if ADJACENT_KEYS[(first - b'a') as usize]
138 .as_bytes()
139 .contains(&second)
140 {
141 1
142 } else {
143 2
144 }
145}

147fn is_swap(first: &[u8], second: &[u8], i: usize, j: usize) -> bool {
148 i >= 2 && j >= 2 && first[i - 1] == second[j - 2] && first[i - 2] == second[j - 1]
149}
150
151fn distance_table(first: &[u8], second: &[u8], keyboard: bool) -> Vec<Vec<usize>> {
152 let gap = gap_cost(keyboard);
153 let mut table = vec![vec![0; second.len() + 1]; first.len() + 1];
154 for (i, row) in table.iter_mut().enumerate() {
155 row[0] = i * gap;
156 }
157 for (j, cell) in table[0].iter_mut().enumerate() {
158 *cell = j * gap;
159 }
160 for i in 1..=first.len() {
161 for j in 1..=second.len() {
162 let substitution = substitution_cost(first[i - 1], second[j - 1], keyboard);
163 let mut best = (table[i - 1][j] + gap)
164 .min(table[i][j - 1] + gap)
165 .min(table[i - 1][j - 1] + substitution);
166 if is_swap(first, second, i, j) {
167 best = best.min(table[i - 2][j - 2] + 1);
168 }
169 table[i][j] = best;
170 }
171 }
172 table
173}
174
175fn edit_distance(first: &str, second: &str, keyboard: bool) -> usize {
176 distance_table(first.as_bytes(), second.as_bytes(), keyboard)[first.len()][second.len()]
177}
178
179fn prefix_distance(word: &str, other: &str, keyboard: bool) -> usize {
180 let table = distance_table(word.as_bytes(), other.as_bytes(), keyboard);
181 *table[word.len()].iter().min().unwrap()
182}
183
184fn edit_operations(first: &str, second: &str, keyboard: bool) -> (usize, Vec<String>) {
185 let first = first.as_bytes();
186 let second = second.as_bytes();
187 let table = distance_table(first, second, keyboard);
188 let gap = gap_cost(keyboard);
189 let mut operations = Vec::new();
190 let (mut i, mut j) = (first.len(), second.len());
191 while i > 0 || j > 0 {
192 if i >= 1
193 && j >= 1
194 && table[i][j]
195 == table[i - 1][j - 1] + substitution_cost(first[i - 1], second[j - 1], keyboard)
196 {
197 if first[i - 1] != second[j - 1] {
198 operations.push(format!(
199 "sub:{}>{}",
200 first[i - 1] as char,
201 second[j - 1] as char
202 ));
203 }
204 i -= 1;
205 j -= 1;
206 } else if is_swap(first, second, i, j) && table[i][j] == table[i - 2][j - 2] + 1 {
207 operations.push(format!(
208 "swap:{}{}",
209 first[i - 2] as char,
210 first[i - 1] as char
211 ));
212 i -= 2;
213 j -= 2;
214 } else if i >= 1 && table[i][j] == table[i - 1][j] + gap {
215 operations.push(format!("del:{}", first[i - 1] as char));
216 i -= 1;
217 } else {
218 operations.push(format!("ins:{}", second[j - 1] as char));
219 j -= 1;
220 }
221 }
222 operations.reverse();
223 (table[first.len()][second.len()], operations)
224}

226fn soundex(word: &str) -> String {
227 let letters = word.as_bytes();
228 let mut code = String::from(letters[0].to_ascii_uppercase() as char);
229 let mut previous = SOUNDEX_DIGITS[(letters[0] - b'a') as usize];
230 for &letter in &letters[1..] {
231 if letter == b'h' || letter == b'w' {
232 continue;
233 }
234 let digit = SOUNDEX_DIGITS[(letter - b'a') as usize];
235 if digit != b'0' && digit != previous {
236 code.push(digit as char);
237 }
238 previous = digit;
239 }
240 code.push_str("000");
241 code.truncate(4);
242 code
243}

245fn segment<'a>(word: &'a str, dictionary: &HashMap<String, u32>) -> (u64, Option<Vec<&'a str>>) {
246 let length = word.len();
247 let mut ways = vec![0u64; length + 1];
248 let mut best: Vec<Option<Vec<&str>>> = vec![None; length + 1];
249 ways[length] = 1;
250 best[length] = Some(Vec::new());
251 for start in (0..length).rev() {
252 for end in start + 1..=length {
253 let piece = &word[start..end];
254 let Some(rest) = &best[end] else { continue };
255 if !dictionary.contains_key(piece) {
256 continue;
257 }
258 ways[start] += ways[end];
259 let candidate: Vec<&str> = std::iter::once(piece).chain(rest.iter().copied()).collect();
260 let better = match &best[start] {
261 None => true,
262 Some(current) => (candidate.len(), &candidate) < (current.len(), current),
263 };
264 if better {
265 best[start] = Some(candidate);
266 }
267 }
268 }
269 (ways[0], best.swap_remove(0))
270}

364 self.base_limit

368 fn is_known(&self, word: &str) -> bool {
369 self.dictionary.contains_key(word) || self.ignored.contains(word)
370 }

372 fn frequency_order(&self) -> Vec<&str> {
373 let mut entries: Vec<(&String, &u32)> = self.dictionary.iter().collect();
374 entries.sort_by_key(|&(word, &frequency)| (Reverse(frequency), word));
375 entries.into_iter().map(|(word, _)| word.as_str()).collect()
376 }

378 fn require_entry(&self, word: &str) -> Result<u32, String> {
379 self.dictionary
380 .get(word)
381 .copied()
382 .ok_or_else(|| format!("NOT FOUND {word}"))
383 }
384
385 fn candidates(&self, word: &str) -> Vec<(String, usize)> {
386 let limit = self.limit();
387 let gap = gap_cost(self.keyboard);
388 let mut found: Vec<(usize, Reverse<u32>, &String)> = self
389 .dictionary
390 .iter()
391 .filter(|(other, _)| *other != word && other.len().abs_diff(word.len()) * gap <= limit)
392 .map(|(other, &frequency)| {
393 (
394 edit_distance(word, other, self.keyboard),
395 Reverse(frequency),
396 other,
397 )
398 })
399 .filter(|(distance, _, _)| *distance <= limit)
400 .collect();
401 found.sort();
402 found
403 .into_iter()
404 .map(|(distance, _, other)| (other.clone(), distance))
405 .collect()
406 }

408 fn correction(&self, word: &str) -> Option<String> {
409 if self.is_known(word) {
410 return None;
411 }
412 if let Some(target) = self.rules.get(word) {
413 return Some(target.clone());
414 }
415 self.candidates(word)
416 .into_iter()
417 .next()
418 .map(|(other, _)| other)
419 }

430 fn log_correction(&mut self, word: &str, replacement: &str) {
431 let key = (word.to_string(), replacement.to_string());
432 *self.correction_log.entry(key).or_default() += 1;
433 self.counters.corrected += 1;
434 }

436 fn unknown_finding(&self, token: &Token) -> String {
437 let prefix = format!("UNKNOWN {} {}", token.number, token.text);
438 if let Some(target) = self.rules.get(&token.word) {
439 return format!("{prefix} RULE {target}");
440 }
441 let found = self.candidates(&token.word);
442 if found.is_empty() {
443 return format!("{prefix} NONE");
444 }
445 let words = found
446 .iter()
447 .take(self.suggestion_count)
448 .map(|(other, _)| other);
449 listing(format!("{prefix} SUGGEST"), words)
450 }

452 fn check(&mut self, text: &str) {
453 let tokens = tokenize(text);
454 let mut findings = Vec::new();
455 let mut counts = [0; 4];
456 for token in &tokens {
457 if !self.is_known(&token.word) {
458 findings.push(self.unknown_finding(token));
459 counts[0] += 1;
460 *self.unknown_log.entry(token.word.clone()).or_default() += 1;
461 }
462 if token.mixed_case() {
463 findings.push(format!("CASE {} {}", token.number, token.text));
464 counts[1] += 1;
465 }
466 if token.sentence_start && token.text.starts_with(|c: char| c.is_ascii_lowercase()) {
467 findings.push(format!("CAPITAL {} {}", token.number, token.text));
468 counts[2] += 1;
469 }
470 if token.repeat {
471 findings.push(format!("REPEAT {} {}", token.number, token.text));
472 counts[3] += 1;
473 }
474 }
475 self.counters.checked += tokens.len();
476 self.counters.unknown += counts[0];
477 self.counters.case += counts[1];
478 self.counters.capital += counts[2];
479 self.counters.repeats += counts[3];
480 let header = FINDING_KINDS
481 .iter()
482 .zip(counts)
483 .map(|(kind, count)| format!("{kind}={count}"));
484 self.emit(listing(format!("CHECK words={}", tokens.len()), header));
485 self.output.extend(findings);
486 }

488 fn correct(&mut self, text: &str) {
489 let mut result = String::new();
490 let mut position = 0;
491 let mut changed = 0;
492 for token in tokenize(text) {
493 result.push_str(&token.gap);
494 match self.correction(&token.word) {
495 Some(replacement) => {
496 result.push_str(&apply_case(&token.text, &replacement));
497 self.log_correction(&token.word, &replacement);
498 changed += 1;
499 }
500 None => result.push_str(&token.text),
501 }
502 position = token.end;
503 }
504 result.push_str(&text[position..]);
505 self.emit(format!("CORRECT changed={changed}: {result}"));
506 }

508 fn fix(&mut self, text: &str) {
509 let mut result = String::new();
510 let mut position = 0;
511 let mut removed = 0;
512 let mut replaced = 0;
513 let mut recased = 0;
514 for token in tokenize(text) {
515 position = token.end;
516 if token.repeat {
517 removed += 1;
518 continue;
519 }
520 let replacement = self.correction(&token.word);
521 let mut fixed = apply_case(&token.text, replacement.as_deref().unwrap_or(&token.word));
522 if token.sentence_start && fixed.starts_with(|c: char| c.is_ascii_lowercase()) {
523 fixed = capitalize_first(&fixed);
524 }
525 if let Some(replacement) = replacement {
526 self.log_correction(&token.word, &replacement);
527 replaced += 1;
528 } else if fixed != token.text {
529 recased += 1;
530 }
531 result.push_str(&token.gap);
532 result.push_str(&fixed);
533 }
534 result.push_str(&text[position..]);
535 self.emit(format!(
536 "FIX removed={removed} replaced={replaced} recased={recased}: {result}"
537 ));
538 }

540 fn profile(&mut self, text: &str) {
541 let tokens = tokenize(text);
542 let distinct = tokens
543 .iter()
544 .map(|token| &token.word)
545 .collect::<HashSet<_>>()
546 .len();
547 let unknown = tokens
548 .iter()
549 .filter(|token| !self.is_known(&token.word))
550 .count();
551 let sentences = tokens.iter().filter(|token| token.sentence_start).count();
552 let (rate, average, longest) =
553 match tokens.iter().min_by_key(|token| Reverse(token.text.len())) {
554 None => ("0.0".to_string(), "0.00".to_string(), "-"),
555 Some(longest) => {
556 let characters: usize = tokens.iter().map(|token| token.text.len()).sum();
557 (
558 decimal_text(100 * unknown, tokens.len(), 1),
559 decimal_text(characters, tokens.len(), 2),
560 longest.word.as_str(),
561 )
562 }
563 };
564 self.emit(format!(
565 "PROFILE words={} distinct={distinct} unknown={unknown} rate={rate} avglen={average} \
566 longest={longest} sentences={sentences}",
567 tokens.len()
568 ));
569 }

571 fn learn(&mut self, text: &str) {
572 self.record_undo("LEARN");
573 let mut updated = 0;
574 let mut added = 0;
575 let mut skipped = 0;
576 for token in tokenize(text) {
577 if token.word.len() > MAX_WORD_LENGTH {
578 skipped += 1;
579 } else if let Some(frequency) = self.dictionary.get_mut(&token.word) {
580 *frequency += 1;
581 updated += 1;
582 } else {
583 self.ignored.remove(&token.word);
584 self.dictionary.insert(token.word, 1);
585 added += 1;
586 }
587 }
588 self.emit(format!(
589 "LEARN updated={updated} added={added} skipped={skipped}"
590 ));
591 }

593 fn suggest(&mut self, arguments: &[&str]) -> CommandResult {
594 let word = word_argument(arguments[0])?;
595 let line = if self.is_known(&word) {
596 format!("SUGGEST {word} KNOWN")
597 } else if let Some(target) = self.rules.get(&word) {
598 format!("SUGGEST {word} RULE {target}")
599 } else {
600 let found = self.candidates(&word);
601 let shown = found
602 .iter()
603 .take(self.suggestion_count)
604 .map(|(other, distance)| format!("{other}:{distance}"));
605 listing(format!("SUGGEST {word} found={}", found.len()), shown)
606 };
607 self.emit(line);
608 Ok(())
609 }

652 fn prefix(&mut self, arguments: &[&str]) -> CommandResult {
653 let word = word_argument(arguments[0])?;
654 let count = number_argument(arguments[1], 1, MAX_LISTED)? as usize;
655 let matches: Vec<&str> = self
656 .frequency_order()
657 .into_iter()
658 .filter(|other| other.starts_with(&word))
659 .collect();
660 let shown = matches
661 .iter()
662 .take(count)
663 .map(|other| format!("{other}:{}", self.dictionary[*other]));
664 let line = listing(format!("PREFIX {word} count={}", matches.len()), shown);
665 self.emit(line);
666 Ok(())
667 }

669 fn complete(&mut self, arguments: &[&str]) -> CommandResult {
670 let word = word_argument(arguments[0])?;
671 let count = number_argument(arguments[1], 1, MAX_LISTED)? as usize;
672 let threshold = self.limit() / 2;
673 let mut matches: Vec<(usize, Reverse<u32>, &String)> = self
674 .dictionary
675 .iter()
676 .map(|(other, &frequency)| {
677 (
678 prefix_distance(&word, other, self.keyboard),
679 Reverse(frequency),
680 other,
681 )
682 })
683 .filter(|(distance, _, _)| *distance <= threshold)
684 .collect();
685 matches.sort();
686 let shown = matches
687 .iter()
688 .take(count)
689 .map(|(distance, _, other)| format!("{other}:{distance}"));

909 fn dispatch(&mut self, name: &str, rest: Option<&str>) -> CommandResult {
910 let Some(&(_, handler)) = COMMANDS.iter().find(|(command, _)| *command == name) else {
911 return Err(format!("UNKNOWN COMMAND {name}"));
912 };
913 let bad_arguments = || format!("BAD ARGUMENTS {name}");
914 match handler {
915 Handler::Text(run) => {
916 run(self, rest.ok_or_else(bad_arguments)?);
917 Ok(())
918 }
919 Handler::Words(arity, run) => {
920 let arguments: Vec<&str> =
921 rest.map_or_else(Vec::new, |rest| rest.split(' ').collect());
922 if arguments.len() != arity {
923 return Err(bad_arguments());
924 }
925 run(self, &arguments)
926 }
927 }
928 }
929
930 fn execute(&mut self, line: &str) {
931 let (name, rest) = match line.split_once(' ') {
932 Some((name, rest)) => (name, Some(rest)),
933 None => (line, None),
934 };
935 if let Err(message) = self.dispatch(name, rest) {
936 self.emit(format!("ERROR {message}"));
937 }
938 }

941fn main() {
942 let mut input = String::new();
943 io::stdin().read_to_string(&mut input).unwrap();
944 let mut lines = input.lines();
945 let word_count: usize = lines.next().unwrap().trim().parse().unwrap();
946 let mut dictionary = HashMap::new();
947 for _ in 0..word_count {
948 let (word, frequency) = lines.next().unwrap().split_once(' ').unwrap();
949 dictionary.insert(word.to_string(), frequency.trim().parse().unwrap());
950 }
951 let mut checker = SpellChecker::new(dictionary);
952 let command_count: usize = lines.next().unwrap().trim().parse().unwrap();
953 for _ in 0..command_count {
954 checker.execute(lines.next().unwrap_or(""));
955 }
956 let mut output = checker.output.join("\n");
957 output.push('\n');
958 io::stdout().write_all(output.as_bytes()).unwrap();
959}
