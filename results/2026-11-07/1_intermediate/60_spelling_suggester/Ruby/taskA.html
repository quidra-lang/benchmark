29def word_argument(text)
30 raise CommandError, "BAD WORD #{text}" unless /\A[A-Za-z]{1,30}\z/.match?(text)
31 text.downcase
32end

34def number_argument(text, minimum, maximum)
35 raise CommandError, "BAD NUMBER #{text}" unless /\A[0-9]{1,7}\z/.match?(text)
36 value = text.to_i
37 raise CommandError, "BAD NUMBER #{text}" unless value.between?(minimum, maximum)
38 value
39end

41def decimal_text(numerator, denominator, digits)
42 scale = 10**digits
43 rounded = (2 * numerator * scale + denominator) / (2 * denominator)
44 "#{rounded / scale}.#{(rounded % scale).to_s.rjust(digits, "0")}"
45end

47def apply_case(token, word)
48 return word.upcase if token.length >= 2 && token == token.upcase
49 return word[0].upcase + word[1..] if /\A[A-Z]/.match?(token)
50 word
51end

53Token = Struct.new(
54 :number, :text, :word, :start, :finish, :gap,
55 :sentence_start, :repeat, :mixed_case, keyword_init: true
56)

58def tokens(text)
59 result = []
60 previous = nil
61 previous_end = 0
62 text.to_enum(:scan, /[A-Za-z]+/).each do
63 match = Regexp.last_match
64 spelling = match[0]
65 gap = text[previous_end...match.begin(0)]
66 word = spelling.downcase
67 token = Token.new(
68 number: result.length + 1, text: spelling, word: word,
69 start: match.begin(0), finish: match.end(0), gap: gap,
70 sentence_start: previous.nil? || /[.!?]/.match?(gap),
71 repeat: !previous.nil? && previous.word == word && /\A *\z/.match?(gap),
72 mixed_case: /[A-Z]/.match?(spelling[1..]) && spelling != spelling.upcase
73 )
74 result << token
75 previous = token
76 previous_end = token.finish
77 end
78 result
79end

81def gap_cost(keyboard)
82 keyboard ? 2 : 1
83end
84
85def substitution_cost(first, second, keyboard)
86 return 0 if first == second
87 return 1 unless keyboard
88 ADJACENT_KEYS[first].include?(second) ? 1 : 2
89end

91def swapped?(first, second, i, j)
92 i >= 2 && j >= 2 && first[i - 1] == second[j - 2] && first[i - 2] == second[j - 1]
93end
94
95def distance_table(first, second, keyboard)
96 gap = gap_cost(keyboard)
97 rows, columns = first.length, second.length
98 table = Array.new(rows + 1) { Array.new(columns + 1, 0) }
99 (0..rows).each { |i| table[i][0] = i * gap }
100 (0..columns).each { |j| table[0][j] = j * gap }
101 (1..rows).each do |i|
102 (1..columns).each do |j|
103 table[i][j] = [
104 table[i - 1][j] + gap,
105 table[i][j - 1] + gap,
106 table[i - 1][j - 1] + substitution_cost(first[i - 1], second[j - 1], keyboard)
107 ].min
108 if swapped?(first, second, i, j)
109 table[i][j] = [table[i][j], table[i - 2][j - 2] + 1].min
110 end
111 end
112 end
113 table
114end
115
116def edit_distance(first, second, keyboard)
117 distance_table(first, second, keyboard)[first.length][second.length]
118end
119
120def prefix_distance(first, second, keyboard)
121 distance_table(first, second, keyboard).last.min
122end
123
124def edit_operations(first, second, keyboard)
125 table = distance_table(first, second, keyboard)
126 gap = gap_cost(keyboard)
127 operations = []
128 i, j = first.length, second.length
129 while i.positive? || j.positive?
130 if i >= 1 && j >= 1 &&
131 table[i][j] == table[i - 1][j - 1] + substitution_cost(first[i - 1], second[j - 1], keyboard)
132 operations << "sub:#{first[i - 1]}>#{second[j - 1]}" if first[i - 1] != second[j - 1]
133 i -= 1
134 j -= 1
135 elsif swapped?(first, second, i, j) && table[i][j] == table[i - 2][j - 2] + 1
136 operations << "swap:#{first[i - 2, 2]}"
137 i -= 2
138 j -= 2
139 elsif i >= 1 && table[i][j] == table[i - 1][j] + gap
140 operations << "del:#{first[i - 1]}"
141 i -= 1
142 else
143 operations << "ins:#{second[j - 1]}"
144 j -= 1
145 end
146 end
147 [table[first.length][second.length], operations.reverse]
148end

150def soundex(word)
151 code = word[0].upcase
152 previous = "hw".include?(word[0]) ? "0" : SOUNDEX_DIGITS[word[0]]
153 word[1..].each_char do |letter|
154 next if "hw".include?(letter)
155 digit = SOUNDEX_DIGITS[letter]
156 code += digit if digit != "0" && digit != previous
157 previous = digit
158 end
159 (code + "000")[0, 4]
160end

162def segment(word, dictionary)
163 n = word.length
164 ways = Array.new(n + 1, 0)
165 best = Array.new(n + 1)
166 ways[n] = 1
167 best[n] = []
168 (n - 1).downto(0) do |start|
169 (start + 1..n).each do |finish|
170 piece = word[start...finish]
171 next unless dictionary.key?(piece) && ways[finish].positive?
172 ways[start] += ways[finish]
173 candidate = [piece] + best[finish]
174 current = best[start]
175 if current.nil? || ([candidate.length, candidate] <=> [current.length, current]).negative?
176 best[start] = candidate
177 end
178 end
179 end
180 [ways[0], best[0]]
181end

183class SpellChecker
184 attr_reader :output
185
186 def initialize(dictionary)
187 @dictionary = dictionary
188 @ignored = Set.new
189 @rules = {}
190 @keyboard = false
191 @base_limit = 2
192 @suggestion_count = 5
193 @undo_history = []
194 @unknown_log = Hash.new(0)
195 @correction_log = Hash.new(0)
196 @counters = COUNTERS.to_h { |name| [name, 0] }
197 @output = []
198 end
199
200 def emit(*parts)
201 @output << parts.join(" ")
202 end
203
204 def limit
205 @keyboard ? 2 * @base_limit : @base_limit
206 end
207
208 def known?(word)
209 @dictionary.key?(word) || @ignored.include?(word)
210 end
211
212 def frequency_order
213 @dictionary.sort_by { |word, frequency| [-frequency, word] }.map(&:first)
214 end
215
216 def require_entry(word)
217 raise CommandError, "NOT FOUND #{word}" unless @dictionary.key?(word)
218 @dictionary[word]
219 end
220
221 def candidates(word)
222 gap = gap_cost(@keyboard)
223 found = []
224 @dictionary.each do |other, frequency|
225 next if other == word || (other.length - word.length).abs * gap > limit
226 distance = edit_distance(word, other, @keyboard)
227 found << [distance, -frequency, other] if distance <= limit
228 end
229 found.sort.map { |distance, _, other| [other, distance] }
230 end
231
232 def correction(word)
233 return nil if known?(word)
234 return @rules[word] if @rules.key?(word)
235 candidates(word).first&.first
236 end
237
238 def save_undo(name)
239 @undo_history << [name, @dictionary.dup, @ignored.dup, @rules.dup]
240 end
241
242 def log_correction(original, replacement)
243 @correction_log[[original, replacement]] += 1
244 @counters["corrected"] += 1
245 end
246
247 def unknown_finding(token)
248 prefix = "UNKNOWN #{token.number} #{token.text}"
249 return "#{prefix} RULE #{@rules[token.word]}" if @rules.key?(token.word)
250 found = candidates(token.word)
251 return "#{prefix} NONE" if found.empty?
252 "#{prefix} SUGGEST #{found.first(@suggestion_count).map(&:first).join(' ')}"
253 end
254
255 def check_text(text)
256 list = tokens(text)
257 findings = []
258 counts = FINDINGS.to_h { |kind| [kind, 0] }
259 list.each do |token|
260 unless known?(token.word)
261 findings << unknown_finding(token)
262 counts["unknown"] += 1
263 @unknown_log[token.word] += 1
264 end
265 if token.mixed_case
266 findings << "CASE #{token.number} #{token.text}"
267 counts["case"] += 1
268 end
269 if token.sentence_start && /\A[a-z]/.match?(token.text)
270 findings << "CAPITAL #{token.number} #{token.text}"
271 counts["capital"] += 1
272 end
273 if token.repeat
274 findings << "REPEAT #{token.number} #{token.text}"
275 counts["repeats"] += 1
276 end
277 end
278 @counters["checked"] += list.length
279 FINDINGS.each { |kind| @counters[kind] += counts[kind] }
280 emit("CHECK", "words=#{list.length}", *FINDINGS.map { |kind| "#{kind}=#{counts[kind]}" })
281 findings.each { |line| emit(line) }
282 end
283
284 def correct_text(text)
285 pieces = []
286 position = 0
287 changed = 0
288 tokens(text).each do |token|
289 replacement = correction(token.word)
290 if replacement.nil?
291 pieces << token.gap + token.text
292 else
293 pieces << token.gap + apply_case(token.text, replacement)
294 log_correction(token.word, replacement)
295 changed += 1
296 end
297 position = token.finish
298 end
299 pieces << text[position..]
300 emit("CORRECT", "changed=#{changed}:", pieces.join)
301 end
302
303 def fix_text(text)
304 pieces = []
305 position = 0
306 removed = 0
307 replaced = 0
308 recased = 0
309 tokens(text).each do |token|
310 position = token.finish
311 if token.repeat
312 removed += 1
313 next
314 end
315 replacement = correction(token.word)
316 target = replacement.nil? ? token.word : replacement
317 result = apply_case(token.text, target)
318 result = result[0].upcase + result[1..] if token.sentence_start && /\A[a-z]/.match?(result)
319 if replacement
320 log_correction(token.word, replacement)
321 replaced += 1
322 elsif result != token.text
323 recased += 1
324 end
325 pieces << token.gap + result
326 end
327 pieces << text[position..]
328 emit("FIX", "removed=#{removed}", "replaced=#{replaced}", "recased=#{recased}:", pieces.join)
329 end

331 def profile_text(text)
332 list = tokens(text)
333 distinct = list.map(&:word).uniq.length
334 unknown = list.count { |token| !known?(token.word) }
335 sentences = list.count(&:sentence_start)
336 if list.empty?
337 rate, avglen, longest = "0.0", "0.00", "-"
338 else
339 rate = decimal_text(100 * unknown, list.length, 1)
340 avglen = decimal_text(list.sum { |token| token.text.length }, list.length, 2)
341 longest = list.max_by { |token| token.text.length }.word
342 end
343 emit("PROFILE", "words=#{list.length}", "distinct=#{distinct}",
344 "unknown=#{unknown}", "rate=#{rate}", "avglen=#{avglen}",
345 "longest=#{longest}", "sentences=#{sentences}")
346 end

348 def learn_text(text)
349 save_undo("LEARN")
350 updated = added = skipped = 0
351 tokens(text).each do |token|
352 if token.word.length > 30
353 skipped += 1
354 elsif @dictionary.key?(token.word)
355 @dictionary[token.word] += 1
356 updated += 1
357 else
358 @dictionary[token.word] = 1
359 @ignored.delete(token.word)
360 added += 1
361 end
362 end
363 emit("LEARN", "updated=#{updated}", "added=#{added}", "skipped=#{skipped}")
364 end

366 def suggest(arguments)
367 word = word_argument(arguments[0])
368 if known?(word)
369 emit("SUGGEST", word, "KNOWN")
370 elsif @rules.key?(word)
371 emit("SUGGEST", word, "RULE", @rules[word])
372 else
373 found = candidates(word)
374 emit("SUGGEST", word, "found=#{found.length}",
375 *found.first(@suggestion_count).map { |other, distance| "#{other}:#{distance}" })
376 end
377 end

379 def dist(arguments)
380 first = word_argument(arguments[0])
381 second = word_argument(arguments[1])
382 distance, operations = edit_operations(first, second, @keyboard)
383 emit("DIST", first, second, distance, *operations)
384 end

386 def soundex_code(arguments)
387 word = word_argument(arguments[0])
388 emit("SOUNDEX", word, soundex(word))
389 end
390
391 def sounds(arguments)
392 word = word_argument(arguments[0])
393 code = soundex(word)
394 found = frequency_order.select { |other| other != word && soundex(other) == code }
395 emit("SOUNDS", word, code, "count=#{found.length}", *found.first(10))
396 end

398 def split_word(arguments)
399 word = word_argument(arguments[0])
400 ways, best = segment(word, @dictionary)
401 emit("SPLIT", word, "ways=#{ways}", "->", *(ways.zero? ? ["NONE"] : best))
402 end

404 def prefix(arguments)
405 word = word_argument(arguments[0])
406 count = number_argument(arguments[1], 1, 20)
407 found = frequency_order.select { |other| other.start_with?(word) }
408 emit("PREFIX", word, "count=#{found.length}",
409 *found.first(count).map { |other| "#{other}:#{@dictionary[other]}" })
410 end
411
412 def complete(arguments)
413 word = word_argument(arguments[0])
414 count = number_argument(arguments[1], 1, 20)
415 threshold = limit / 2
416 found = []
417 @dictionary.each do |other, frequency|
418 distance = prefix_distance(word, other, @keyboard)
419 found << [distance, -frequency, other] if distance <= threshold
420 end
421 found.sort!
422 emit("COMPLETE", word, "count=#{found.length}",
423 *found.first(count).map { |distance, _, other| "#{other}:#{distance}" })
424 end

426 def list_words(arguments)
427 word = word_argument(arguments[0])
428 count = number_argument(arguments[1], 1, 20)
429 found = @dictionary.keys.select { |other| other >= word }.sort
430 emit("LIST", word, "count=#{found.length}",
431 *found.first(count).map { |other| "#{other}:#{@dictionary[other]}" })
432 end

434 def word_info(arguments)
435 word = word_argument(arguments[0])
436 frequency = require_entry(word)
437 rank = frequency_order.index(word) + 1
438 emit("WORD", word, "freq=#{frequency}", "rank=#{rank}",
439 "soundex=#{soundex(word)}", "neighbors=#{candidates(word).length}")
440 end

442 def add(arguments)
443 word = word_argument(arguments[0])
444 frequency = number_argument(arguments[1], 1, 1_000_000)
445 raise CommandError, "DUPLICATE #{word}" if @dictionary.key?(word)
446 save_undo("ADD")
447 @dictionary[word] = frequency
448 @ignored.delete(word)
449 emit("ADD", word, frequency)
450 end

452 def remove(arguments)
453 word = word_argument(arguments[0])
454 frequency = require_entry(word)
455 save_undo("REMOVE")
456 @dictionary.delete(word)
457 emit("REMOVE", word, frequency)
458 end

460 def ignore(arguments)
461 word = word_argument(arguments[0])
462 raise CommandError, "IN DICTIONARY #{word}" if @dictionary.key?(word)
463 raise CommandError, "ALREADY IGNORED #{word}" if @ignored.include?(word)
464 save_undo("IGNORE")
465 @ignored.add(word)
466 emit("IGNORE", word)
467 end

469 def unignore(arguments)
470 word = word_argument(arguments[0])
471 raise CommandError, "NOT IGNORED #{word}" unless @ignored.include?(word)
472 save_undo("UNIGNORE")
473 @ignored.delete(word)
474 emit("UNIGNORE", word)
475 end

477 def rule(arguments)
478 source = word_argument(arguments[0])
479 target = word_argument(arguments[1])
480 raise CommandError, "IN DICTIONARY #{source}" if @dictionary.key?(source)
481 raise CommandError, "NOT FOUND #{target}" unless @dictionary.key?(target)
482 save_undo("RULE")
483 previous = @rules[source]
484 @rules[source] = target
485 if previous.nil?
486 emit("RULE", source, target)
487 else
488 emit("RULE", source, target, "REPLACED", previous)
489 end
490 end
491
492 def unrule(arguments)
493 word = word_argument(arguments[0])
494 raise CommandError, "NO RULE #{word}" unless @rules.key?(word)
495 save_undo("UNRULE")
496 target = @rules.delete(word)
497 emit("UNRULE", word, target)
498 end

500 def show_rules
501 rows = @rules.sort_by(&:first)
502 emit("RULES", "count=#{rows.length}", *rows.map { |source, target| "#{source}>#{target}" })
503 end

505 def keyboard_mode(arguments)
506 mode = arguments[0]
507 raise CommandError, "BAD MODE #{mode}" unless %w[ON OFF].include?(mode)
508 @keyboard = mode == "ON"

512 def set_option(arguments)
513 option = arguments[0]
514 maximum = { "LIMIT" => 3, "SUGGESTIONS" => 10 }[option]
515 raise CommandError, "BAD OPTION #{option}" if maximum.nil?
516 value = number_argument(arguments[1], 1, maximum)
517 if option == "LIMIT"
518 @base_limit = value
519 else
520 @suggestion_count = value
521 end
522 emit("SET", option, value)

525 def undo
526 raise CommandError, "NOTHING TO UNDO" if @undo_history.empty?
527 name, @dictionary, @ignored, @rules = @undo_history.pop
528 emit("UNDO", name)
529 end

531 def report
532 rows = @unknown_log.sort_by { |word, count| [-count, word] }
533 emit("REPORT", "distinct=#{rows.length}",
534 *rows.first(10).map { |word, count| "#{word}:#{count}" })
535 end

537 def history
538 rows = @correction_log.sort_by { |pair, count| [-count, pair[0], pair[1]] }
539 emit("HISTORY", "distinct=#{rows.length}",
540 *rows.first(10).map { |pair, count| "#{pair[0]}>#{pair[1]}:#{count}" })
541 end

543 def stats
544 top = frequency_order.first
545 emit("STATS", "words=#{@dictionary.length}",
546 "total=#{@dictionary.values.sum}",
547 "avglen=#{decimal_text(@dictionary.keys.sum(&:length), @dictionary.length, 2)}",
548 "top=#{top}:#{@dictionary[top]}", "ignored=#{@ignored.length}",
549 "rules=#{@rules.length}", "keyboard=#{@keyboard ? 'ON' : 'OFF'}",
550 "limit=#{limit}", "suggestions=#{@suggestion_count}", "undo=#{@undo_history.length}")
551 emit("SESSION", *COUNTERS.map { |kind| "#{kind}=#{@counters[kind]}" })
552 end

554 def execute(line)
555 name, separator, rest = line.partition(" ")
556 begin
557 unless TEXT_COMMANDS.include?(name) || ARITIES.key?(name)
558 raise CommandError, "UNKNOWN COMMAND #{name}"
559 end
560 if TEXT_COMMANDS.include?(name)
561 raise CommandError, "BAD ARGUMENTS #{name}" if separator.empty?
562 case name
563 when "CHECK" then check_text(rest)
564 when "CORRECT" then correct_text(rest)
565 when "FIX" then fix_text(rest)
566 when "PROFILE" then profile_text(rest)
567 when "LEARN" then learn_text(rest)
568 end
569 else
570 arguments = separator.empty? ? [] : rest.split(/ /, -1)
571 raise CommandError, "BAD ARGUMENTS #{name}" unless arguments.length == ARITIES[name]
572 case name
573 when "SUGGEST" then suggest(arguments)
574 when "DIST" then dist(arguments)
575 when "SOUNDEX" then soundex_code(arguments)
576 when "SOUNDS" then sounds(arguments)
577 when "SPLIT" then split_word(arguments)
578 when "PREFIX" then prefix(arguments)
579 when "COMPLETE" then complete(arguments)
580 when "LIST" then list_words(arguments)
581 when "WORD" then word_info(arguments)
582 when "ADD" then add(arguments)
583 when "REMOVE" then remove(arguments)
584 when "IGNORE" then ignore(arguments)
585 when "UNIGNORE" then unignore(arguments)
586 when "RULE" then rule(arguments)
587 when "UNRULE" then unrule(arguments)
588 when "RULES" then show_rules
589 when "KEYBOARD" then keyboard_mode(arguments)
590 when "SET" then set_option(arguments)
591 when "UNDO" then undo
592 when "REPORT" then report
593 when "HISTORY" then history
594 when "STATS" then stats
595 end
596 end
597 rescue CommandError => error
598 emit("ERROR", error.message)
599 end
600 end
601end
