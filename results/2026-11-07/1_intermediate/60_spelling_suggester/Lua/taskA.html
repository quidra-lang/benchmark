86local function number_argument(text, minimum, maximum)
87 local value = #text <= 7 and text:find("^%d+$") and tonumber(text)
88 if not value or value < minimum or value > maximum then
89 fail("BAD NUMBER " .. text)

94local function decimal_text(numerator, denominator, digits)
95 local scale = math.tointeger(10 ^ digits)
96 local value = (2 * numerator * scale + denominator) // (2 * denominator)
97 return string.format("%d.%0" .. digits .. "d", value // scale, value % scale)
98end

100local function capitalize_first(word)
101 return word:sub(1, 1):upper() .. word:sub(2)

104local function apply_case(token_text, word)
105 if #token_text >= 2 and token_text:find("^%u+$") then
106 return word:upper()
107 end
108 if token_text:find("^%u") then
109 return capitalize_first(word)
110 end
111 return word

114local function tokenize(text)
115 local tokens = {}
116 local previous
117 local position = 0
118 for start, token_text, after in text:gmatch("()([A-Za-z]+)()") do
119 local gap = text:sub(position + 1, start - 1)
120 local word = token_text:lower()
121 local token = {
122 number = #tokens + 1,
123 text = token_text,
124 word = word,
125 stop = after - 1,
126 gap = gap,
127 sentence_start = previous == nil or gap:find("[.!?]") ~= nil,
128 is_repeat = previous ~= nil and word == previous.word and gap:find("^ *$") ~= nil,
129 mixed_case = token_text:find("%u", 2) ~= nil and not token_text:find("^%u+$"),
130 }
131 tokens[#tokens + 1] = token
132 previous = token
133 position = token.stop
134 end
135 return tokens
136end
137
138local function gap_cost(keyboard)
139 return keyboard and 2 or 1
140end
141
142local function substitution_cost(first, second, keyboard)
143 if first == second then
144 return 0
145 end
146 if not keyboard then
147 return 1
148 end
149 return ADJACENT_KEYS[first]:find(second, 1, true) and 1 or 2
150end
151
152local function is_swap(first, second, i, j)
153 return i >= 2 and j >= 2
154 and first:sub(i, i) == second:sub(j - 1, j - 1)
155 and first:sub(i - 1, i - 1) == second:sub(j, j)
156end
157
158local function distance_table(first, second, keyboard)
159 local gap = gap_cost(keyboard)
160 local distances = {}
161 for i = 0, #first do
162 distances[i] = {[0] = i * gap}
163 end
164 for j = 1, #second do
165 distances[0][j] = j * gap
166 end
167 for i = 1, #first do
168 local row = distances[i]
169 local above = distances[i - 1]
170 for j = 1, #second do
171 local best = math.min(
172 above[j] + gap,
173 row[j - 1] + gap,
174 above[j - 1] + substitution_cost(first:sub(i, i), second:sub(j, j), keyboard)
175 )
176 if is_swap(first, second, i, j) then
177 best = math.min(best, distances[i - 2][j - 2] + 1)
178 end
179 row[j] = best
180 end
181 end
182 return distances
183end
184
185local function edit_distance(first, second, keyboard)
186 return distance_table(first, second, keyboard)[#first][#second]
187end
188
189local function prefix_distance(word, other, keyboard)
190 local row = distance_table(word, other, keyboard)[#word]
191 local best = row[0]
192 for j = 1, #other do
193 best = math.min(best, row[j])
194 end
195 return best
196end
197
198local function edit_operations(first, second, keyboard)
199 local distances = distance_table(first, second, keyboard)
200 local gap = gap_cost(keyboard)
201 local operations = {}
202 local i, j = #first, #second
203 while i > 0 or j > 0 do
204 local source, target = first:sub(i, i), second:sub(j, j)
205 if i >= 1 and j >= 1
206 and distances[i][j] == distances[i - 1][j - 1]
207 + substitution_cost(source, target, keyboard)
208 then
209 if source ~= target then
210 table.insert(operations, 1, "sub:" .. source .. ">" .. target)
211 end
212 i, j = i - 1, j - 1
213 elseif is_swap(first, second, i, j) and distances[i][j] == distances[i - 2][j - 2] + 1 then
214 table.insert(operations, 1, "swap:" .. first:sub(i - 1, i))
215 i, j = i - 2, j - 2
216 elseif i >= 1 and distances[i][j] == distances[i - 1][j] + gap then
217 table.insert(operations, 1, "del:" .. source)
218 i = i - 1
219 else
220 table.insert(operations, 1, "ins:" .. target)
221 j = j - 1
222 end
223 end
224 return distances[#first][#second], operations
225end

227local function soundex(word)
228 local first = word:sub(1, 1)
229 local code = first:upper()
230 local previous = SOUNDEX_DIGITS[first] or "0"
231 for letter in word:sub(2):gmatch(".") do
232 local digit = SOUNDEX_DIGITS[letter]
233 if digit then
234 if digit ~= "0" and digit ~= previous then
235 code = code .. digit
236 end
237 previous = digit
238 end
239 end
240 return (code .. "000"):sub(1, 4)
241end
242
243local function segmentation_less(first, second)
244 if #first ~= #second then
245 return #first < #second
246 end
247 for index = 1, #first do
248 if first[index] ~= second[index] then
249 return first[index] < second[index]
250 end
251 end
252 return false
253end

255local function segment(word, dictionary)
256 local length = #word
257 local ways = {[length + 1] = 1}
258 local best = {[length + 1] = {}}
259 for start = length, 1, -1 do
260 ways[start] = 0
261 for stop = start, length do
262 local piece = word:sub(start, stop)
263 if dictionary[piece] and ways[stop + 1] > 0 then
264 ways[start] = ways[start] + ways[stop + 1]
265 local candidate = {piece, table.unpack(best[stop + 1])}
266 if best[start] == nil or segmentation_less(candidate, best[start]) then
267 best[start] = candidate
268 end
269 end
270 end
271 end
272 return ways[1], best[1]
273end

275local function ranked_less(first, second)
276 if first.distance ~= second.distance then
277 return first.distance < second.distance
278 end
279 if first.frequency ~= second.frequency then
280 return first.frequency > second.frequency
281 end
282 return first.word < second.word
283end
284
285local function scored_item(entry)
286 return entry.word .. ":" .. entry.distance
287end

289local function emit_ranked(head, log)
290 local entries = {}
291 for key, count in pairs(log) do
292 entries[#entries + 1] = {key = key, count = count}
293 end
294 table.sort(entries, function(first, second)
295 if first.count ~= second.count then
296 return first.count > second.count
297 end
298 return first.key < second.key
299 end)
300 local items = take(entries, REPORT_SIZE, function(entry)
301 return entry.key .. ":" .. entry.count
302 end)
303 emit(head, "distinct=" .. #entries, table.unpack(items))
304end

317 rules = {},

321 undo_history = {},
322 unknown_log = {},
323 correction_log = {},

328function SpellChecker:limit()
329 return self.keyboard and 2 * self.base_limit or self.base_limit
330end

332function SpellChecker:is_known(word)
333 return self.dictionary[word] ~= nil or self.ignored[word] ~= nil

336function SpellChecker:frequency_order()
337 local dictionary = self.dictionary
338 return sorted_keys(dictionary, function(first, second)
339 if dictionary[first] ~= dictionary[second] then
340 return dictionary[first] > dictionary[second]
341 end
342 return first < second
343 end)

346function SpellChecker:with_frequencies(words, count)

352function SpellChecker:require_entry(word)

360function SpellChecker:candidates(word)
361 local limit = self:limit()
362 local gap = gap_cost(self.keyboard)
363 local found = {}
364 for other, frequency in pairs(self.dictionary) do
365 if other ~= word and math.abs(#other - #word) * gap <= limit then
366 local distance = edit_distance(word, other, self.keyboard)
367 if distance <= limit then
368 found[#found + 1] = {word = other, distance = distance, frequency = frequency}
369 end
370 end
371 end
372 table.sort(found, ranked_less)
373 return found
374end

376function SpellChecker:correction(word)
377 if self:is_known(word) then
378 return nil
379 end
380 if self.rules[word] then
381 return self.rules[word]
382 end
383 local found = self:candidates(word)
384 return found[1] and found[1].word
385end

402function SpellChecker:unknown_finding(token)
403 local prefix = "UNKNOWN " .. token.number .. " " .. token.text
404 local target = self.rules[token.word]
405 if target then
406 return prefix .. " RULE " .. target
407 end
408 local found = self:candidates(token.word)
409 if #found == 0 then
410 return prefix .. " NONE"
411 end
412 local words = take(found, self.suggestion_count, function(entry)
413 return entry.word
414 end)
415 return prefix .. " SUGGEST " .. table.concat(words, " ")

418function SpellChecker:check(text)
419 local tokens = tokenize(text)
420 local findings = {}
421 local counts = {}
422 for _, kind in ipairs(FINDING_KINDS) do
423 counts[kind] = 0
424 end
425 local function add_finding(kind, line)
426 counts[kind] = counts[kind] + 1
427 findings[#findings + 1] = line
428 end
429 for _, token in ipairs(tokens) do
430 if not self:is_known(token.word) then
431 add_finding("unknown", self:unknown_finding(token))
432 self.unknown_log[token.word] = (self.unknown_log[token.word] or 0) + 1
433 end
434 if token.mixed_case then
435 add_finding("case", "CASE " .. token.number .. " " .. token.text)
436 end
437 if token.sentence_start and token.text:find("^%l") then
438 add_finding("capital", "CAPITAL " .. token.number .. " " .. token.text)
439 end
440 if token.is_repeat then
441 add_finding("repeats", "REPEAT " .. token.number .. " " .. token.text)
442 end
443 end
444 self.counters.checked = self.counters.checked + #tokens
445 local summary = {}
446 for index, kind in ipairs(FINDING_KINDS) do
447 self.counters[kind] = self.counters[kind] + counts[kind]
448 summary[index] = kind .. "=" .. counts[kind]
449 end
450 emit("CHECK", "words=" .. #tokens, table.unpack(summary))
451 for _, line in ipairs(findings) do
452 print(line)
453 end
454end

456function SpellChecker:correct(text)
457 local pieces = {}
458 local position = 0
459 local changed = 0
460 for _, token in ipairs(tokenize(text)) do
461 local replacement = self:correction(token.word)
462 if replacement then
463 pieces[#pieces + 1] = token.gap .. apply_case(token.text, replacement)
464 self:log_correction(token.word, replacement)
465 changed = changed + 1
466 else
467 pieces[#pieces + 1] = token.gap .. token.text
468 end
469 position = token.stop
470 end
471 pieces[#pieces + 1] = text:sub(position + 1)
472 emit("CORRECT changed=" .. changed .. ":", table.concat(pieces))
473end
474
475function SpellChecker:fix(text)
476 local pieces = {}
477 local position = 0
478 local removed = 0
479 local replaced = 0
480 local recased = 0
481 for _, token in ipairs(tokenize(text)) do
482 position = token.stop
483 if token.is_repeat then
484 removed = removed + 1
485 else
486 local replacement = self:correction(token.word)
487 local result = apply_case(token.text, replacement or token.word)
488 if token.sentence_start and result:find("^%l") then
489 result = capitalize_first(result)
490 end
491 if replacement then
492 self:log_correction(token.word, replacement)
493 replaced = replaced + 1
494 elseif result ~= token.text then
495 recased = recased + 1
496 end
497 pieces[#pieces + 1] = token.gap .. result
498 end
499 end
500 pieces[#pieces + 1] = text:sub(position + 1)
501 emit(
502 "FIX", "removed=" .. removed, "replaced=" .. replaced,
503 "recased=" .. recased .. ":", table.concat(pieces)
504 )
505end

507function SpellChecker:profile(text)
508 local tokens = tokenize(text)
509 local forms = {}
510 local distinct = 0
511 local unknown = 0
512 local sentences = 0
513 local characters = 0
514 local longest
515 for _, token in ipairs(tokens) do
516 if not forms[token.word] then
517 forms[token.word] = true
518 distinct = distinct + 1
519 end
520 if not self:is_known(token.word) then
521 unknown = unknown + 1
522 end
523 if token.sentence_start then
524 sentences = sentences + 1
525 end
526 characters = characters + #token.text
527 if longest == nil or #token.text > #longest.text then
528 longest = token
529 end
530 end
531 local rate, average, longest_word = "0.0", "0.00", "-"
532 if #tokens > 0 then
533 rate = decimal_text(100 * unknown, #tokens, 1)
534 average = decimal_text(characters, #tokens, 2)
535 longest_word = longest.word
536 end
537 emit(
538 "PROFILE", "words=" .. #tokens, "distinct=" .. distinct,
539 "unknown=" .. unknown, "rate=" .. rate, "avglen=" .. average,
540 "longest=" .. longest_word, "sentences=" .. sentences
541 )
542end

544function SpellChecker:learn(text)
545 self:record_undo("LEARN")
546 local updated = 0
547 local added = 0
548 local skipped = 0
549 for _, token in ipairs(tokenize(text)) do
550 local word = token.word
551 if #word > MAX_WORD_LENGTH then
552 skipped = skipped + 1
553 elseif self.dictionary[word] then
554 self.dictionary[word] = self.dictionary[word] + 1
555 updated = updated + 1
556 else
557 self.dictionary[word] = 1
558 self.ignored[word] = nil
559 added = added + 1
560 end
561 end
562 emit("LEARN", "updated=" .. updated, "added=" .. added, "skipped=" .. skipped)
563end

625function SpellChecker:complete(arguments)
626 local word = word_argument(arguments[1])
627 local count = number_argument(arguments[2], 1, MAX_LISTED)
628 local threshold = self:limit() // 2
629 local matches = {}
630 for other, frequency in pairs(self.dictionary) do
631 local distance = prefix_distance(word, other, self.keyboard)
632 if distance <= threshold then
633 matches[#matches + 1] = {word = other, distance = distance, frequency = frequency}
634 end
635 end
636 table.sort(matches, ranked_less)
637 local items = take(matches, count, scored_item)
638 emit("COMPLETE", word, "count=" .. #matches, table.unpack(items))
639end

713function SpellChecker:rule(arguments)
714 local source = word_argument(arguments[1])
715 local target = word_argument(arguments[2])

717 fail("IN DICTIONARY " .. source)

719 if not self.dictionary[target] then
720 fail("NOT FOUND " .. target)

723 local previous = self.rules[source]
724 self.rules[source] = target
725 if previous then
726 emit("RULE", source, target, "REPLACED", previous)
727 else
728 emit("RULE", source, target)
729 end
730end

767 local value = number_argument(arguments[2], 1, maximum)
768 if option == "LIMIT" then
769 self.base_limit = value
770 else
771 self.suggestion_count = value
772 end
773 emit("SET", option, value)
774end

777 if #self.undo_history == 0 then
778 fail("NOTHING TO UNDO")
779 end
780 local record = table.remove(self.undo_history)
781 self.dictionary = record.dictionary
782 self.ignored = record.ignored
783 self.rules = record.rules
784 emit("UNDO", record.name)
785end

795function SpellChecker:stats()
796 local order = self:frequency_order()
797 local total = 0
798 local characters = 0
799 for word, frequency in pairs(self.dictionary) do
800 total = total + frequency
801 characters = characters + #word
802 end
803 local top = order[1]
804 emit(
805 "STATS", "words=" .. #order, "total=" .. total,
806 "avglen=" .. decimal_text(characters, #order, 2),
807 "top=" .. top .. ":" .. self.dictionary[top], "ignored=" .. count_keys(self.ignored),
808 "rules=" .. count_keys(self.rules), "keyboard=" .. (self.keyboard and "ON" or "OFF"),
809 "limit=" .. self:limit(), "suggestions=" .. self.suggestion_count,
810 "undo=" .. #self.undo_history
811 )
812 local session = {}
813 for index, name in ipairs(COUNTER_NAMES) do
814 session[index] = name .. "=" .. self.counters[name]
815 end
816 emit("SESSION", table.unpack(session))
817end

849function SpellChecker:dispatch(name, rest)
850 local command = COMMANDS[name]
851 if not command then
852 fail("UNKNOWN COMMAND " .. name)
853 end
854 if command.text then
855 if not rest then
856 fail("BAD ARGUMENTS " .. name)
857 end
858 command.run(self, rest)
859 else
860 local arguments = rest and split_spaces(rest) or {}
861 if #arguments ~= command.arity then
862 fail("BAD ARGUMENTS " .. name)
863 end
864 command.run(self, arguments)
865 end
866end
867
868function SpellChecker:execute(line)
869 local space = line:find(" ", 1, true)
870 local name = space and line:sub(1, space - 1) or line
871 local rest = space and line:sub(space + 1)
872 local ok, err = pcall(self.dispatch, self, name, rest)
873 if not ok then
874 if getmetatable(err) ~= CommandError then
875 error(err, 0)
876 end
877 print("ERROR " .. err.message)
878 end
879end

884 for _ = 1, word_count do
