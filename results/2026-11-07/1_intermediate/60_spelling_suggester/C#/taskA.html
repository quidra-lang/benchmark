27sealed record Token(int Number, string Text, int End, string Gap, bool SentenceStart, bool Repeat)
28{
29 const string SentenceMarks = ".!?";
30
31 public string Word { get; } = Text.ToLowerInvariant();
32
33 public bool MixedCase => Text.Skip(1).Any(char.IsUpper) && !Text.All(char.IsUpper);
34
35 public static List<Token> Tokenize(string text)
36 {
37 var tokens = new List<Token>();
38 int position = 0;
39 foreach (Match match in Regex.Matches(text, "[A-Za-z]+"))
40 {
41 string gap = text[position..match.Index];
42 string word = match.Value.ToLowerInvariant();
43 Token? previous = tokens.Count > 0 ? tokens[^1] : null;
44 bool sentenceStart = previous is null || gap.Any(mark => SentenceMarks.Contains(mark));
45 bool repeat = previous is not null && previous.Word == word && gap.All(c => c == ' ');
46 position = match.Index + match.Length;
47 tokens.Add(
48 new Token(tokens.Count + 1, match.Value, position, gap, sentenceStart, repeat));
49 }
50 return tokens;
51 }
52}

54static class EditDistance
55{
56 static readonly string[] AdjacentKeys =
57 [
58 "qswz", "ghnv", "dfvx", "cefrsx", "drsw", "cdgrtv", "bfhtvy", "bgjnuy", "jkou", "hikmnu",
59 "ijlmo", "kop", "jkn", "bhjm", "iklp", "lo", "aw", "deft", "adewxz", "fgry", "hijy",
60 "bcfg", "aeqs", "cdsz", "ghtu", "asx",
61 ];
62
63 public static int GapCost(bool keyboard) => keyboard ? 2 : 1;
64
65 static int SubstitutionCost(char first, char second, bool keyboard)
66 {
67 if (first == second)
68 return 0;
69 if (!keyboard)
70 return 1;
71 return AdjacentKeys[first - 'a'].Contains(second) ? 1 : 2;
72 }
73
74 static bool IsSwap(string first, string second, int i, int j) =>
75 i >= 2 && j >= 2 && first[i - 1] == second[j - 2] && first[i - 2] == second[j - 1];
76
77 static int[,] Table(string first, string second, bool keyboard)
78 {
79 int gap = GapCost(keyboard);
80 var table = new int[first.Length + 1, second.Length + 1];
81 for (int i = 0; i <= first.Length; i++)
82 table[i, 0] = i * gap;
83 for (int j = 0; j <= second.Length; j++)
84 table[0, j] = j * gap;
85 for (int i = 1; i <= first.Length; i++)
86 {
87 for (int j = 1; j <= second.Length; j++)
88 {
89 int best = Math.Min(table[i - 1, j] + gap, table[i, j - 1] + gap);
90 int substitution = SubstitutionCost(first[i - 1], second[j - 1], keyboard);
91 best = Math.Min(best, table[i - 1, j - 1] + substitution);
92 if (IsSwap(first, second, i, j))
93 best = Math.Min(best, table[i - 2, j - 2] + 1);
94 table[i, j] = best;
95 }
96 }
97 return table;
98 }
99
100 public static int Compute(string first, string second, bool keyboard) =>
101 Table(first, second, keyboard)[first.Length, second.Length];
102
103 public static int Prefix(string word, string other, bool keyboard)
104 {
105 int[,] table = Table(word, other, keyboard);
106 return Enumerable.Range(0, other.Length + 1).Min(j => table[word.Length, j]);
107 }
108
109 public static (int Distance, List<string> Operations) Operations(
110 string first, string second, bool keyboard)
111 {
112 int[,] table = Table(first, second, keyboard);
113 int gap = GapCost(keyboard);
114 var operations = new List<string>();
115 int i = first.Length;
116 int j = second.Length;
117 while (i > 0 || j > 0)
118 {
119 if (i >= 1 && j >= 1
120 && table[i, j] == table[i - 1, j - 1]
121 + SubstitutionCost(first[i - 1], second[j - 1], keyboard))
122 {
123 if (first[i - 1] != second[j - 1])
124 operations.Add($"sub:{first[i - 1]}>{second[j - 1]}");
125 i--;
126 j--;
127 }
128 else if (IsSwap(first, second, i, j) && table[i, j] == table[i - 2, j - 2] + 1)
129 {
130 operations.Add($"swap:{first[i - 2]}{first[i - 1]}");
131 i -= 2;
132 j -= 2;
133 }
134 else if (i >= 1 && table[i, j] == table[i - 1, j] + gap)
135 {
136 operations.Add($"del:{first[i - 1]}");
137 i--;
138 }
139 else
140 {
141 operations.Add($"ins:{second[j - 1]}");
142 j--;
143 }
144 }
145 operations.Reverse();
146 return (table[first.Length, second.Length], operations);
147 }
148}

150static class Soundex
151{
152 const string Digits = "01230120022455012623010202";
153
154 public static string Encode(string word)
155 {
156 var code = new StringBuilder().Append(char.ToUpperInvariant(word[0]));
157 char previous = Digits[word[0] - 'a'];
158 foreach (char letter in word[1..])
159 {
160 if (letter is 'h' or 'w')
161 continue;
162 char digit = Digits[letter - 'a'];
163 if (digit != '0' && digit != previous)
164 code.Append(digit);
165 previous = digit;
166 }
167 return code.Append("000").ToString()[..4];
168 }
169}

251 static string DecimalText(long numerator, long denominator, int digits)
252 {
253 long scale = (long)Math.Pow(10, digits);
254 long value = (2 * numerator * scale + denominator) / (2 * denominator);
255 return $"{value / scale}.{(value % scale).ToString($"D{digits}")}";

269 static (long Ways, List<string>? Best) Segment(string word, Dictionary<string, int> dictionary)
270 {
271 int length = word.Length;
272 var ways = new long[length + 1];
273 var best = new List<string>?[length + 1];
274 ways[length] = 1;
275 best[length] = [];
276 for (int start = length - 1; start >= 0; start--)
277 {
278 for (int end = start + 1; end <= length; end++)
279 {
280 string piece = word[start..end];
281 if (!dictionary.ContainsKey(piece) || ways[end] == 0)
282 continue;
283 ways[start] += ways[end];
284 List<string> candidate = [piece, .. best[end]!];
285 if (best[start] is not { } current || candidate.Count < current.Count)
286 best[start] = candidate;
287 }
288 }
289 return (ways[0], best[0]);
290 }

318 List<(string Word, int Distance)> Candidates(string word)
319 {
320 int limit = Limit;
321 int gap = EditDistance.GapCost(keyboard);
322 return InDistanceOrder(
323 dictionary.Keys
324 .Where(other =>
325 other != word && Math.Abs(other.Length - word.Length) * gap <= limit)
326 .Select(other =>
327 (Word: other, Distance: EditDistance.Compute(word, other, keyboard)))
328 .Where(item => item.Distance <= limit));
329 }

351 string UnknownFinding(Token token)
352 {
353 string prefix = $"UNKNOWN {token.Number} {token.Text}";
354 if (rules.TryGetValue(token.Word, out string? target))
355 return $"{prefix} RULE {target}";

363 void Check(string text)
364 {
365 var tokens = Token.Tokenize(text);
366 var findings = new List<(string Kind, string Line)>();
367 foreach (var token in tokens)
368 {
369 if (!IsKnown(token.Word))
370 {
371 findings.Add(("unknown", UnknownFinding(token)));
372 unknownLog[token.Word] = unknownLog.GetValueOrDefault(token.Word) + 1;
373 }
374 if (token.MixedCase)
375 findings.Add(("case", $"CASE {token.Number} {token.Text}"));
376 if (token.SentenceStart && char.IsLower(token.Text[0]))
377 findings.Add(("capital", $"CAPITAL {token.Number} {token.Text}"));
378 if (token.Repeat)
379 findings.Add(("repeats", $"REPEAT {token.Number} {token.Text}"));
380 }
381 var counts = FindingKinds.ToDictionary(
382 kind => kind, kind => findings.Count(finding => finding.Kind == kind));
383 counters["checked"] += tokens.Count;
384 foreach (string kind in FindingKinds)
385 counters[kind] += counts[kind];
386 var shown = FindingKinds.Select(kind => $"{kind}={counts[kind]}");
387 Emit(["CHECK", $"words={tokens.Count}", .. shown]);
388 foreach (var (_, line) in findings)
389 Emit(line);
390 }

392 void Correct(string text)
393 {
394 var result = new StringBuilder();
395 int position = 0;
396 int changed = 0;
397 foreach (var token in Token.Tokenize(text))
398 {
399 string? replacement = Correction(token.Word);
400 result.Append(token.Gap);
401 if (replacement is null)
402 {
403 result.Append(token.Text);
404 }
405 else
406 {
407 result.Append(ApplyCase(token.Text, replacement));
408 LogCorrection(token.Word, replacement);
409 changed++;
410 }
411 position = token.End;
412 }
413 result.Append(text[position..]);
414 Emit($"CORRECT changed={changed}:", result.ToString());
415 }

417 void Fix(string text)
418 {
419 var result = new StringBuilder();
420 int position = 0;
421 int removed = 0;
422 int replaced = 0;
423 int recased = 0;
424 foreach (var token in Token.Tokenize(text))
425 {
426 position = token.End;
427 if (token.Repeat)
428 {
429 removed++;
430 continue;
431 }
432 string? replacement = Correction(token.Word);
433 string fixedWord = ApplyCase(token.Text, replacement ?? token.Word);
434 if (token.SentenceStart && char.IsLower(fixedWord[0]))
435 fixedWord = CapitalizeFirst(fixedWord);
436 if (replacement is not null)
437 {
438 LogCorrection(token.Word, replacement);
439 replaced++;
440 }
441 else if (fixedWord != token.Text)
442 {
443 recased++;
444 }
445 result.Append(token.Gap).Append(fixedWord);
446 }
447 result.Append(text[position..]);
448 Emit(
449 "FIX", $"removed={removed}", $"replaced={replaced}", $"recased={recased}:",
450 result.ToString());
451 }

453 void Profile(string text)
454 {
455 var tokens = Token.Tokenize(text);
456 int distinct = tokens.Select(token => token.Word).Distinct().Count();
457 int unknown = tokens.Count(token => !IsKnown(token.Word));
458 int sentences = tokens.Count(token => token.SentenceStart);
459 string rate = "0.0";
460 string average = "0.00";
461 string longest = "-";
462 if (tokens.Count > 0)
463 {
464 int characters = tokens.Sum(token => token.Text.Length);
465 rate = DecimalText(100 * unknown, tokens.Count, 1);
466 average = DecimalText(characters, tokens.Count, 2);
467 longest = tokens.MaxBy(token => token.Text.Length)!.Word;
468 }
469 Emit(
470 "PROFILE", $"words={tokens.Count}", $"distinct={distinct}", $"unknown={unknown}",
471 $"rate={rate}", $"avglen={average}", $"longest={longest}", $"sentences={sentences}");
472 }

474 void Learn(string text)
475 {
476 RecordUndo("LEARN");
477 int updated = 0;
478 int added = 0;
479 int skipped = 0;
480 foreach (var token in Token.Tokenize(text))
481 {
482 if (token.Word.Length > MaxWordLength)
483 {
484 skipped++;
485 }
486 else if (dictionary.ContainsKey(token.Word))
487 {
488 dictionary[token.Word]++;
489 updated++;
490 }
491 else
492 {
493 dictionary[token.Word] = 1;
494 ignored.Remove(token.Word);
495 added++;
496 }
497 }
498 Emit("LEARN", $"updated={updated}", $"added={added}", $"skipped={skipped}");
499 }

750 public void Execute(string line)
751 {
752 int space = line.IndexOf(' ');
753 string name = space < 0 ? line : line[..space];
754 string? rest = space < 0 ? null : line[(space + 1)..];
755 try
756 {
757 if (textCommands.TryGetValue(name, out var runText))
758 {
759 if (rest is null)
760 throw new CommandException($"BAD ARGUMENTS {name}");
761 runText(rest);
762 }
763 else if (commands.TryGetValue(name, out var command))
764 {
765 string[] arguments = rest is null ? [] : rest.Split(' ');
766 if (arguments.Length != command.Arity)
767 throw new CommandException($"BAD ARGUMENTS {name}");
768 command.Run(arguments);
769 }
770 else
771 {
772 throw new CommandException($"UNKNOWN COMMAND {name}");
773 }
774 }
775 catch (CommandException error)
776 {
777 Emit($"ERROR {error.Message}");
778 }
779 }
