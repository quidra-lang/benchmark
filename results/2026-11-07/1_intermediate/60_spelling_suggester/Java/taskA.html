61 static String decimalText(long numerator, long denominator, int digits) {
62 long scale = digits == 1 ? 10 : 100;
63 long value = (2 * numerator * scale + denominator) / (2 * denominator);
64 String fraction = "" + (value % scale);
65 return value / scale + "." + "0".repeat(digits - fraction.length()) + fraction;
66 }

68 static String applyCase(String original, String word) {
69 if (original.length() >= 2 && original.equals(original.toUpperCase())) return word.toUpperCase();
70 if (Character.isUpperCase(original.charAt(0))) return Character.toUpperCase(word.charAt(0)) + word.substring(1);
71 return word;
72 }

74 static final class Token {
75 final int number, start, end;
76 final String text, word, gap;
77 final boolean sentenceStart, repeat;
78 Token(int number, String text, int start, int end, String gap, Token previous) {
79 this.number = number;
80 this.text = text;
81 this.word = text.toLowerCase();
82 this.start = start;
83 this.end = end;
84 this.gap = gap;
85 sentenceStart = previous == null || gap.indexOf('.') >= 0 ||
86 gap.indexOf('!') >= 0 || gap.indexOf('?') >= 0;
87 repeat = previous != null && word.equals(previous.word) &&
88 gap.chars().allMatch(ch -> ch == ' ');
89 }
90 boolean mixedCase() {
91 for (int i = 1; i < text.length(); ++i) {
92 if (Character.isUpperCase(text.charAt(i)) && !text.equals(text.toUpperCase())) return true;
93 }
94 return false;
95 }
96 }

98 static List<Token> textTokens(String text) {
99 List<Token> tokens = new ArrayList<>();
100 Matcher matcher = WORD_TOKEN.matcher(text);
101 Token previous = null;
102 int position = 0;
103 while (matcher.find()) {
104 Token token = new Token(tokens.size() + 1, matcher.group(), matcher.start(), matcher.end(),
105 text.substring(position, matcher.start()), previous);
106 tokens.add(token);
107 previous = token;
108 position = matcher.end();
109 }
110 return tokens;
111 }

113 static int substitutionCost(char first, char second, boolean keyboard) {
114 if (first == second) return 0;
115 if (!keyboard) return 1;
116 return ADJACENT_KEYS[first - 'a'].indexOf(second) >= 0 ? 1 : 2;
117 }

119 static boolean isSwap(String first, String second, int i, int j) {
120 return i >= 2 && j >= 2 && first.charAt(i - 1) == second.charAt(j - 2) &&
121 first.charAt(i - 2) == second.charAt(j - 1);
122 }

124 static int[][] distanceTable(String first, String second, boolean keyboard) {
125 int rows = first.length(), columns = second.length();
126 int gap = keyboard ? 2 : 1;
127 int[][] table = new int[rows + 1][columns + 1];
128 for (int i = 0; i <= rows; ++i) table[i][0] = i * gap;
129 for (int j = 0; j <= columns; ++j) table[0][j] = j * gap;
130 for (int i = 1; i <= rows; ++i) {
131 for (int j = 1; j <= columns; ++j) {
132 int best = Math.min(Math.min(table[i - 1][j] + gap, table[i][j - 1] + gap),
133 table[i - 1][j - 1] + substitutionCost(first.charAt(i - 1),
134 second.charAt(j - 1), keyboard));
135 if (isSwap(first, second, i, j)) best = Math.min(best, table[i - 2][j - 2] + 1);
136 table[i][j] = best;
137 }
138 }
139 return table;
140 }

142 static int editDistance(String first, String second, boolean keyboard) {
143 return distanceTable(first, second, keyboard)[first.length()][second.length()];
144 }
145
146 static int prefixDistance(String word, String other, boolean keyboard) {
147 return Arrays.stream(distanceTable(word, other, keyboard)[word.length()]).min().orElse(0);
148 }

150 record DistanceResult(int distance, List<String> operations) {}
151
152 static DistanceResult editOperations(String first, String second, boolean keyboard) {
153 int[][] table = distanceTable(first, second, keyboard);
154 int gap = keyboard ? 2 : 1;
155 List<String> operations = new ArrayList<>();
156 int i = first.length(), j = second.length();
157 while (i > 0 || j > 0) {
158 if (i >= 1 && j >= 1 && table[i][j] == table[i - 1][j - 1] +
159 substitutionCost(first.charAt(i - 1), second.charAt(j - 1), keyboard)) {
160 if (first.charAt(i - 1) != second.charAt(j - 1)) {
161 operations.add("sub:" + first.charAt(i - 1) + ">" + second.charAt(j - 1));
162 }
163 --i; --j;
164 } else if (isSwap(first, second, i, j) && table[i][j] == table[i - 2][j - 2] + 1) {
165 operations.add("swap:" + first.charAt(i - 2) + first.charAt(i - 1));
166 i -= 2; j -= 2;
167 } else if (i >= 1 && table[i][j] == table[i - 1][j] + gap) {
168 operations.add("del:" + first.charAt(i - 1));
169 --i;
170 } else {
171 operations.add("ins:" + second.charAt(j - 1));
172 --j;
173 }
174 }
175 java.util.Collections.reverse(operations);
176 return new DistanceResult(table[first.length()][second.length()], operations);
177 }

179 static String soundex(String word) {
180 StringBuilder code = new StringBuilder(word.substring(0, 1).toUpperCase());
181 char start = word.charAt(0);
182 String previous = start == 'h' || start == 'w' ? "0" : SOUNDEX_DIGITS[start - 'a'];
183 for (int i = 1; i < word.length(); ++i) {
184 char letter = word.charAt(i);
185 if (letter == 'h' || letter == 'w') continue;
186 String digit = SOUNDEX_DIGITS[letter - 'a'];
187 if (!digit.equals("0") && !digit.equals(previous)) code.append(digit);
188 previous = digit;
189 }
190 return (code + "000").substring(0, 4);
191 }

193 static int compareWordLists(List<String> first, List<String> second) {
194 for (int i = 0; i < Math.min(first.size(), second.size()); ++i) {
195 int cmp = first.get(i).compareTo(second.get(i));
196 if (cmp != 0) return cmp;
197 }
198 return Integer.compare(first.size(), second.size());
199 }

202
203 static Segmentation segment(String word, Map<String, Integer> dictionary) {
204 int length = word.length();
205 long[] ways = new long[length + 1];
206 @SuppressWarnings("unchecked")
207 List<String>[] best = new List[length + 1];
208 ways[length] = 1;
209 best[length] = new ArrayList<>();
210 for (int start = length - 1; start >= 0; --start) {
211 for (int end = start + 1; end <= length; ++end) {
212 String piece = word.substring(start, end);
213 if (!dictionary.containsKey(piece) || ways[end] == 0) continue;
214 ways[start] += ways[end];
215 List<String> candidate = new ArrayList<>();
216 candidate.add(piece);
217 candidate.addAll(best[end]);
218 if (best[start] == null || candidate.size() < best[start].size() ||
219 (candidate.size() == best[start].size() &&
220 compareWordLists(candidate, best[start]) < 0)) {
221 best[start] = candidate;
222 }
223 }
224 }
225 return new Segmentation(ways[0], best[0]);
226 }

269 List<Candidate> candidates(String word) {
270 int maxDistance = limit(), gap = keyboard ? 2 : 1;
271 List<Candidate> found = new ArrayList<>();
272 for (Map.Entry<String, Integer> entry : dictionary.entrySet()) {
273 String other = entry.getKey();
274 if (other.equals(word) || Math.abs(other.length() - word.length()) * gap > maxDistance) continue;
275 int distance = editDistance(word, other, keyboard);
276 if (distance <= maxDistance) found.add(new Candidate(other, distance, entry.getValue()));
277 }
278 found.sort((a, b) -> {
279 int byDistance = Integer.compare(a.distance(), b.distance());
280 if (byDistance != 0) return byDistance;
281 int byFrequency = Integer.compare(b.frequency(), a.frequency());
282 return byFrequency != 0 ? byFrequency : a.word().compareTo(b.word());
283 });
284 return found;
285 }

316 void checkText(String text) {
317 List<Token> tokens = textTokens(text);
318 List<String> findings = new ArrayList<>();
319 Map<String, Integer> counts = new HashMap<>();
320 for (String name : FINDINGS) counts.put(name, 0);
321 for (Token token : tokens) {
322 if (!isKnown(token.word)) {
323 findings.add(unknownFinding(token));
324 counts.merge("unknown", 1, Integer::sum);
325 unknownLog.merge(token.word, 1, Integer::sum);
326 }
327 if (token.mixedCase()) {
328 findings.add("CASE " + token.number + " " + token.text);
329 counts.merge("case", 1, Integer::sum);
330 }
331 if (token.sentenceStart && Character.isLowerCase(token.text.charAt(0))) {
332 findings.add("CAPITAL " + token.number + " " + token.text);
333 counts.merge("capital", 1, Integer::sum);
334 }
335 if (token.repeat) {
336 findings.add("REPEAT " + token.number + " " + token.text);
337 counts.merge("repeats", 1, Integer::sum);
338 }
339 }
340 counters.merge("checked", tokens.size(), Integer::sum);
341 StringBuilder header = new StringBuilder("CHECK words=").append(tokens.size());
342 for (String name : FINDINGS) {
343 int count = counts.get(name);
344 header.append(' ').append(name).append('=').append(count);
345 counters.merge(name, count, Integer::sum);
346 }
347 output.add(header.toString());
348 output.addAll(findings);
349 }

351 void correctText(String text) {
352 StringBuilder result = new StringBuilder();
353 int position = 0, changed = 0;
354 for (Token token : textTokens(text)) {
355 String replacement = correction(token.word);
356 result.append(token.gap);
357 if (replacement == null) {
358 result.append(token.text);
359 } else {
360 result.append(applyCase(token.text, replacement));
361 logCorrection(token.word, replacement);
362 ++changed;
363 }
364 position = token.end;
365 }
366 result.append(text.substring(position));
367 output.add("CORRECT changed=" + changed + ": " + result);
368 }

370 void fixText(String text) {
371 StringBuilder result = new StringBuilder();
372 int position = 0, removed = 0, replaced = 0, recased = 0;
373 for (Token token : textTokens(text)) {
374 position = token.end;
375 if (token.repeat) {
376 ++removed;
377 continue;
378 }
379 String replacement = correction(token.word);
380 String target = replacement == null ? token.word : replacement;
381 String revised = applyCase(token.text, target);
382 if (token.sentenceStart && Character.isLowerCase(revised.charAt(0))) {
383 revised = Character.toUpperCase(revised.charAt(0)) + revised.substring(1);
384 }
385 if (replacement != null) {
386 logCorrection(token.word, replacement);
387 ++replaced;
388 } else if (!revised.equals(token.text)) {
389 ++recased;
390 }
391 result.append(token.gap).append(revised);
392 }
393 result.append(text.substring(position));
394 output.add("FIX removed=" + removed + " replaced=" + replaced +
395 " recased=" + recased + ": " + result);
396 }

398 void profile(String text) {
399 List<Token> tokens = textTokens(text);
400 Set<String> distinct = new HashSet<>();
401 int unknown = 0, sentences = 0, characters = 0;
402 String longest = "-";
403 for (Token token : tokens) {
404 distinct.add(token.word);
405 if (!isKnown(token.word)) ++unknown;
406 if (token.sentenceStart) ++sentences;
407 characters += token.text.length();
408 if (longest.equals("-") || token.text.length() > longest.length()) longest = token.word;
409 }
410 int size = tokens.size();
411 String rate = size == 0 ? "0.0" : decimalText(100L * unknown, size, 1);
412 String average = size == 0 ? "0.00" : decimalText(characters, size, 2);
413 output.add("PROFILE words=" + size + " distinct=" + distinct.size() +
414 " unknown=" + unknown + " rate=" + rate + " avglen=" + average +
415 " longest=" + longest + " sentences=" + sentences);
416 }

677 void execute(String line) {
678 int space = line.indexOf(' ');
679 String name = space == -1 ? line : line.substring(0, space);
680 String rest = space == -1 ? "" : line.substring(space + 1);
681 try {
682 if (TEXT_COMMANDS.contains(name)) {
683 if (space < 0) throw new CommandError("BAD ARGUMENTS " + name);
684 switch (name) {
685 case "CHECK" -> checkText(rest);
686 case "CORRECT" -> correctText(rest);
687 case "FIX" -> fixText(rest);
688 case "PROFILE" -> profile(rest);
689 case "LEARN" -> learn(rest);
690 }
691 return;
692 }
693 Integer expected = ARITIES.get(name);
694 if (expected == null) throw new CommandError("UNKNOWN COMMAND " + name);
695 String[] args = space == -1 ? new String[0] : rest.split(" ", -1);
696 if (args.length != expected) throw new CommandError("BAD ARGUMENTS " + name);
697 switch (name) {
698 case "SUGGEST" -> suggest(args);
699 case "DIST" -> distance(args);
700 case "SOUNDEX" -> { String w = wordArgument(args[0]); emit("SOUNDEX", w, soundex(w)); }
701 case "SOUNDS" -> sounds(args);
702 case "SPLIT" -> split(args);
703 case "PREFIX" -> prefix(args);
704 case "COMPLETE" -> complete(args);
705 case "LIST" -> listWords(args);
706 case "WORD" -> wordInfo(args);
707 case "ADD" -> add(args);
708 case "REMOVE" -> remove(args);
709 case "IGNORE" -> ignore(args);
710 case "UNIGNORE" -> unignore(args);
711 case "RULE" -> rule(args);
712 case "UNRULE" -> unrule(args);
713 case "RULES" -> showRules();
714 case "KEYBOARD" -> keyboardMode(args);
715 case "SET" -> setOption(args);
716 case "UNDO" -> undo();
717 case "REPORT" -> report();
718 case "HISTORY" -> history();
719 case "STATS" -> stats();
720 default -> throw new IllegalStateException("Unhandled command " + name);
721 }
722 } catch (CommandError e) {
723 emit("ERROR", e.getMessage());
724 }
725 }
726 }

728 public static void main(String[] args) throws Exception {
729 BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
730 int count = Integer.parseInt(input.readLine());
731 Map<String, Integer> dictionary = new HashMap<>();
732 for (int i = 0; i < count; ++i) {
733 String[] entry = input.readLine().split(" ");
734 dictionary.put(entry[0], Integer.parseInt(entry[1]));
735 }
736 SpellChecker checker = new SpellChecker(dictionary);
737 int commands = Integer.parseInt(input.readLine());
738 for (int i = 0; i < commands; ++i) checker.execute(input.readLine());
739 System.out.print(String.join("\n", checker.output) + "\n");
740 }
