37bool isLetter(char ch) {
38 return ('a' <= ch && ch <= 'z') || ('A' <= ch && ch <= 'Z');
39}
40bool isUpper(char ch) { return 'A' <= ch && ch <= 'Z'; }
41bool isLower(char ch) { return 'a' <= ch && ch <= 'z'; }
42char toLower(char ch) { return isUpper(ch) ? char(ch - 'A' + 'a') : ch; }
43char toUpper(char ch) { return isLower(ch) ? char(ch - 'a' + 'A') : ch; }
44string lower(string word) { for (char& ch : word) ch = toLower(ch); return word; }
45string upper(string word) { for (char& ch : word) ch = toUpper(ch); return word; }
46string capitalized(string word) { word[0] = toUpper(word[0]); return word; }

48string wordArgument(const string& text) {
49 if (text.empty() || text.size() > 30 || !std::all_of(text.begin(), text.end(), isLetter))
50 throw CommandError("BAD WORD " + text);
51 return lower(text);
52}
53int numberArgument(const string& text, int minimum, int maximum) {
54 if (text.empty() || text.size() > 7 || !std::all_of(text.begin(), text.end(),
55 [](char ch) { return '0' <= ch && ch <= '9'; }))
56 throw CommandError("BAD NUMBER " + text);
57 int number = std::stoi(text);
58 if (number < minimum || number > maximum) throw CommandError("BAD NUMBER " + text);
59 return number;
60}
61string decimalText(long long numerator, long long denominator, int digits) {
62 const long long scale = digits == 1 ? 10 : 100;
63 const long long rounded = (2 * numerator * scale + denominator) / (2 * denominator);
64 string fraction = std::to_string(rounded % scale);
65 fraction.insert(fraction.begin(), static_cast<std::size_t>(digits) - fraction.size(), '0');
66 return std::to_string(rounded / scale) + "." + fraction;
67}
68string applyCase(const string& original, const string& word) {
69 if (original.size() >= 2 && std::all_of(original.begin(), original.end(), isUpper)) return upper(word);
70 if (isUpper(original[0])) return capitalized(word);
71 return word;
72}

74struct Token {
75 int number;
76 string text, word, gap;
77 std::size_t start, end;
78 bool sentenceStart, repeated, mixedCase;
79};
80vector<Token> textTokens(const string& text) {
81 vector<Token> result;
82 std::size_t position = 0;
83 while (position < text.size()) {
84 std::size_t start = position;
85 while (start < text.size() && !isLetter(text[start])) ++start;
86 if (start == text.size()) break;
87 std::size_t end = start;
88 while (end < text.size() && isLetter(text[end])) ++end;
89 string original = text.substr(start, end - start);
90 string word = lower(original);
91 string gap = text.substr(position, start - position);
92 bool sentenceStart = result.empty() || gap.find_first_of(".!?") != string::npos;
93 bool repeated = !result.empty() && result.back().word == word &&
94 std::all_of(gap.begin(), gap.end(), [](char ch) { return ch == ' '; });
95 bool mixedCase = false, allUpper = std::all_of(original.begin(), original.end(), isUpper);
96 for (std::size_t i = 1; i < original.size(); ++i) {
97 if (isUpper(original[i]) && !allUpper) mixedCase = true;
98 }
99 result.push_back({int(result.size() + 1), original, word, gap, start, end,
100 sentenceStart, repeated, mixedCase});
101 position = end;
102 }
103 return result;
104}

106int substitutionCost(char first, char second, bool keyboard) {
107 if (first == second) return 0;
108 if (!keyboard) return 1;
109 return std::string_view(ADJACENT_KEYS[first - 'a']).find(second) != std::string_view::npos ? 1 : 2;
110}
111bool isSwap(const string& a, const string& b, int i, int j) {
112 return i >= 2 && j >= 2 && a[i - 1] == b[j - 2] && a[i - 2] == b[j - 1];
113}
114vector<vector<int>> distanceTable(const string& first, const string& second, bool keyboard) {
115 int m = int(first.size()), n = int(second.size()), gap = keyboard ? 2 : 1;
116 vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
117 for (int i = 0; i <= m; ++i) dp[i][0] = i * gap;
118 for (int j = 0; j <= n; ++j) dp[0][j] = j * gap;
119 for (int i = 1; i <= m; ++i) {
120 for (int j = 1; j <= n; ++j) {
121 int best = std::min({dp[i - 1][j] + gap, dp[i][j - 1] + gap,
122 dp[i - 1][j - 1] + substitutionCost(first[i - 1], second[j - 1], keyboard)});
123 if (isSwap(first, second, i, j)) best = std::min(best, dp[i - 2][j - 2] + 1);
124 dp[i][j] = best;
125 }
126 }
127 return dp;
128}
129int editDistance(const string& a, const string& b, bool keyboard) {
130 return distanceTable(a, b, keyboard)[a.size()][b.size()];
131}
132int prefixDistance(const string& a, const string& b, bool keyboard) {
133 const auto row = distanceTable(a, b, keyboard)[a.size()];
134 return *std::min_element(row.begin(), row.end());
135}
136std::pair<int, vector<string>> editOperations(const string& first, const string& second, bool keyboard) {
137 const auto dp = distanceTable(first, second, keyboard);
138 const int gap = keyboard ? 2 : 1;
139 int i = int(first.size()), j = int(second.size());
140 vector<string> operations;
141 while (i > 0 || j > 0) {
142 if (i > 0 && j > 0 && dp[i][j] == dp[i - 1][j - 1] +
143 substitutionCost(first[i - 1], second[j - 1], keyboard)) {
144 if (first[i - 1] != second[j - 1])
145 operations.push_back("sub:" + string(1, first[i - 1]) + ">" + string(1, second[j - 1]));
146 --i; --j;
147 } else if (isSwap(first, second, i, j) && dp[i][j] == dp[i - 2][j - 2] + 1) {
148 operations.push_back("swap:" + first.substr(i - 2, 2));
149 i -= 2; j -= 2;
150 } else if (i > 0 && dp[i][j] == dp[i - 1][j] + gap) {
151 operations.push_back("del:" + string(1, first[i - 1]));
152 --i;
153 } else {
154 operations.push_back("ins:" + string(1, second[j - 1]));
155 --j;
156 }
157 }
158 std::reverse(operations.begin(), operations.end());
159 return {dp[first.size()][second.size()], operations};
160}

162string soundex(const string& word) {
163 static const array<char, 26> digits = [] {
164 array<char, 26> values{};
165 const array<string, 7> groups = {"bfpv", "cgjkqsxz", "dt", "l", "mn", "r", "aeiouy"};
166 for (int i = 0; i < 7; ++i) {
167 for (char ch : groups[i]) values[ch - 'a'] = i == 6 ? '0' : char('1' + i);
168 }
169 return values;
170 }();
171 string code(1, toUpper(word[0]));
172 char previous = word[0] == 'h' || word[0] == 'w' ? '0' : digits[word[0] - 'a'];
173 for (std::size_t i = 1; i < word.size(); ++i) {
174 char ch = word[i];
175 if (ch == 'h' || ch == 'w') continue;
176 char digit = digits[ch - 'a'];
177 if (digit != '0' && digit != previous) code.push_back(digit);
178 previous = digit;
179 }
180 code.append("000");
181 code.resize(4);
182 return code;
183}

185struct Segmentation { long long ways; vector<string> best; };
186Segmentation segment(const string& word, const map<string, int>& dictionary) {
187 const int length = int(word.size());
188 vector<long long> ways(length + 1, 0);
189 vector<optional<vector<string>>> best(length + 1);
190 ways[length] = 1;
191 best[length] = vector<string>{};
192 for (int start = length - 1; start >= 0; --start) {
193 for (int end = start + 1; end <= length; ++end) {
194 string piece = word.substr(start, end - start);
195 if (!dictionary.contains(piece) || ways[end] == 0) continue;
196 ways[start] += ways[end];
197 vector<string> candidate = {piece};
198 candidate.insert(candidate.end(), best[end]->begin(), best[end]->end());
199 if (!best[start] || candidate.size() < best[start]->size() ||
200 (candidate.size() == best[start]->size() && candidate < *best[start])) {
201 best[start] = std::move(candidate);
202 }
203 }
204 }
205 return {ways[0], best[0].value_or(vector<string>{})};
206}

228class SpellChecker {

231 map<string, string> rules;

239 void emit(const string& text) { output.push_back(text); }
240 void emit(const vector<string>& parts) { output.push_back(join(parts)); }
241 int limit() const { return keyboard ? 2 * baseLimit : baseLimit; }
242 bool isKnown(const string& word) const { return dictionary.contains(word) || ignored.contains(word); }
243
244 vector<string> frequencyOrder() const {
245 vector<string> words;
246 for (const auto& [word, frequency] : dictionary) words.push_back(word);
247 std::sort(words.begin(), words.end(), [this](const string& a, const string& b) {
248 int af = dictionary.at(a), bf = dictionary.at(b);
249 return af == bf ? a < b : af > bf;
250 });
251 return words;
252 }
253
254 int requireEntry(const string& word) const {
255 auto found = dictionary.find(word);
256 if (found == dictionary.end()) throw CommandError("NOT FOUND " + word);
257 return found->second;
258 }
259
260 vector<Candidate> candidates(const string& word) const {
261 vector<Candidate> found;
262 int maximum = limit(), gap = keyboard ? 2 : 1;
263 for (const auto& [other, frequency] : dictionary) {
264 if (other == word || std::abs(int(other.size()) - int(word.size())) * gap > maximum) continue;
265 int distance = editDistance(word, other, keyboard);
266 if (distance <= maximum) found.push_back({other, distance, frequency});
267 }
268 std::sort(found.begin(), found.end(), [](const Candidate& a, const Candidate& b) {
269 if (a.distance != b.distance) return a.distance < b.distance;
270 if (a.frequency != b.frequency) return a.frequency > b.frequency;
271 return a.word < b.word;
272 });
273 return found;
274 }

276 optional<string> correction(const string& word) const {
277 if (isKnown(word)) return std::nullopt;
278 if (auto iter = rules.find(word); iter != rules.end()) return iter->second;
279 auto found = candidates(word);
280 if (found.empty()) return std::nullopt;
281 return found.front().word;
282 }

288 ++correctionLog[word + ">" + replacement];

292 string unknownFinding(const Token& token) const {
293 string prefix = "UNKNOWN " + std::to_string(token.number) + " " + token.text;
294 if (auto iter = rules.find(token.word); iter != rules.end())
295 return prefix + " RULE " + iter->second;
296 auto found = candidates(token.word);
297 if (found.empty()) return prefix + " NONE";
298 prefix += " SUGGEST";
299 for (int i = 0; i < std::min(int(found.size()), suggestionCount); ++i)
300 prefix += " " + found[i].word;
301 return prefix;
302 }

304 void checkText(const string& text) {
305 vector<Token> tokens = textTokens(text);
306 vector<string> findings;
307 map<string, int> counts;
308 for (const Token& token : tokens) {
309 if (!isKnown(token.word)) {
310 findings.push_back(unknownFinding(token));
311 ++unknownLog[token.word];
312 ++counts["unknown"];
313 }
314 if (token.mixedCase) {
315 findings.push_back("CASE " + std::to_string(token.number) + " " + token.text);
316 ++counts["case"];
317 }
318 if (token.sentenceStart && isLower(token.text.front())) {
319 findings.push_back("CAPITAL " + std::to_string(token.number) + " " + token.text);
320 ++counts["capital"];
321 }
322 if (token.repeated) {
323 findings.push_back("REPEAT " + std::to_string(token.number) + " " + token.text);
324 ++counts["repeats"];
325 }
326 }
327 counters["checked"] += int(tokens.size());
328 string heading = "CHECK words=" + std::to_string(tokens.size());
329 for (const string kind : {"unknown", "case", "capital", "repeats"}) {
330 heading += " " + kind + "=" + std::to_string(counts[kind]);
331 counters[kind] += counts[kind];
332 }
333 emit(heading);
334 output.insert(output.end(), findings.begin(), findings.end());
335 }

337 void correctText(const string& text) {
338 string result;
339 std::size_t position = 0;
340 int changed = 0;
341 for (const Token& token : textTokens(text)) {
342 result += token.gap;
343 if (auto replacement = correction(token.word)) {
344 result += applyCase(token.text, *replacement);
345 logCorrection(token.word, *replacement);
346 ++changed;
347 } else {
348 result += token.text;
349 }
350 position = token.end;
351 }
352 result += text.substr(position);
353 emit("CORRECT changed=" + std::to_string(changed) + ": " + result);
354 }
355
356 void fixText(const string& text) {
357 string result;
358 std::size_t position = 0;
359 int removed = 0, replaced = 0, recased = 0;
360 for (const Token& token : textTokens(text)) {
361 position = token.end;
362 if (token.repeated) { ++removed; continue; }
363 auto replacement = correction(token.word);
364 string revised = applyCase(token.text, replacement.value_or(token.word));
365 if (token.sentenceStart && isLower(revised.front())) revised = capitalized(revised);
366 if (replacement) {
367 logCorrection(token.word, *replacement);
368 ++replaced;
369 } else if (revised != token.text) {
370 ++recased;
371 }
372 result += token.gap + revised;
373 }
374 result += text.substr(position);
375 emit("FIX removed=" + std::to_string(removed) + " replaced=" +
376 std::to_string(replaced) + " recased=" + std::to_string(recased) + ": " + result);
377 }

379 void profile(const string& text) {
380 auto tokens = textTokens(text);
381 set<string> distinct;
382 int unknown = 0, sentences = 0, characters = 0;
383 string longest = "-";
384 for (const Token& token : tokens) {
385 distinct.insert(token.word);
386 if (!isKnown(token.word)) ++unknown;
387 if (token.sentenceStart) ++sentences;
388 characters += int(token.text.size());
389 if (longest == "-" || token.text.size() > longest.size()) longest = token.word;
390 }
391 int n = int(tokens.size());
392 string rate = n == 0 ? "0.0" : decimalText(100LL * unknown, n, 1);
393 string average = n == 0 ? "0.00" : decimalText(characters, n, 2);
394 emit("PROFILE words=" + std::to_string(n) + " distinct=" + std::to_string(distinct.size()) +
395 " unknown=" + std::to_string(unknown) + " rate=" + rate + " avglen=" + average +
396 " longest=" + longest + " sentences=" + std::to_string(sentences));
397 }

400 recordUndo("LEARN");
401 int updated = 0, added = 0, skipped = 0;
402 for (const Token& token : textTokens(text)) {
403 if (token.word.size() > 30) {
404 ++skipped;
405 } else if (dictionary.contains(token.word)) {
406 ++dictionary[token.word];
407 ++updated;

411 ++added;
412 }
413 }
414 emit("LEARN updated=" + std::to_string(updated) + " added=" +
415 std::to_string(added) + " skipped=" + std::to_string(skipped));
416 }

587 string option = args[0];
588 if (option != "LIMIT" && option != "SUGGESTIONS") throw CommandError("BAD OPTION " + option);
589 int value = numberArgument(args[1], 1, option == "LIMIT" ? 3 : 10);
590 if (option == "LIMIT") baseLimit = value;
591 else suggestionCount = value;
592 emit("SET " + option + " " + std::to_string(value));
593 }
594
595 void undo() {
596 if (undoHistory.empty()) throw CommandError("NOTHING TO UNDO");
597 Snapshot record = std::move(undoHistory.back());
598 undoHistory.pop_back();
599 dictionary = std::move(record.dictionary);
600 ignored = std::move(record.ignored);
601 rules = std::move(record.rules);
602 emit("UNDO " + record.name);
603 }

605 void report() {
606 vector<std::pair<string, int>> entries(unknownLog.begin(), unknownLog.end());
607 std::sort(entries.begin(), entries.end(), [](const auto& a, const auto& b) {
608 return a.second != b.second ? a.second > b.second : a.first < b.first;
609 });
610 string result = "REPORT distinct=" + std::to_string(entries.size());
611 for (int i = 0; i < std::min(int(entries.size()), 10); ++i)
612 result += " " + entries[i].first + ":" + std::to_string(entries[i].second);
613 emit(result);
614 }

618 std::sort(entries.begin(), entries.end(), [](const auto& a, const auto& b) {
619 return a.second != b.second ? a.second > b.second : a.first < b.first;
620 });
621 string result = "HISTORY distinct=" + std::to_string(entries.size());
622 for (int i = 0; i < std::min(int(entries.size()), 10); ++i)
623 result += " " + entries[i].first + ":" + std::to_string(entries[i].second);

627 void stats() {
628 auto words = frequencyOrder();
629 const string& top = words.front();
630 long long total = 0, characters = 0;
631 for (const string& word : words) {
632 total += dictionary.at(word);
633 characters += word.size();
634 }
635 emit("STATS words=" + std::to_string(words.size()) + " total=" + std::to_string(total) +
636 " avglen=" + decimalText(characters, words.size(), 2) +
637 " top=" + top + ":" + std::to_string(dictionary.at(top)) +
638 " ignored=" + std::to_string(ignored.size()) + " rules=" + std::to_string(rules.size()) +
639 " keyboard=" + (keyboard ? "ON" : "OFF") + " limit=" + std::to_string(limit()) +
640 " suggestions=" + std::to_string(suggestionCount) +
641 " undo=" + std::to_string(undoHistory.size()));
642 string session = "SESSION";
643 for (const char* name : COUNTER_NAMES)
644 session += " " + string(name) + "=" + std::to_string(counters[name]);
645 emit(session);
646 }

653 void execute(const string& line) {
654 std::size_t space = line.find(' ');
655 string command = space == string::npos ? line : line.substr(0, space);
656 string rest = space == string::npos ? "" : line.substr(space + 1);
657 try {
658 if (command == "CHECK" || command == "CORRECT" || command == "FIX" ||
659 command == "PROFILE" || command == "LEARN") {
660 if (space == string::npos) throw CommandError("BAD ARGUMENTS " + command);
661 if (command == "CHECK") checkText(rest);
662 else if (command == "CORRECT") correctText(rest);
663 else if (command == "FIX") fixText(rest);
664 else if (command == "PROFILE") profile(rest);
665 else learn(rest);
666 return;
667 }
668 static const map<string, int> arities = {
669 {"SUGGEST", 1}, {"DIST", 2}, {"SOUNDEX", 1}, {"SOUNDS", 1},
670 {"SPLIT", 1}, {"PREFIX", 2}, {"COMPLETE", 2}, {"LIST", 2},
671 {"WORD", 1}, {"ADD", 2}, {"REMOVE", 1}, {"IGNORE", 1},
672 {"UNIGNORE", 1}, {"RULE", 2}, {"UNRULE", 1}, {"RULES", 0},
673 {"KEYBOARD", 1}, {"SET", 2}, {"UNDO", 0}, {"REPORT", 0},
674 {"HISTORY", 0}, {"STATS", 0}
675 };
676 auto found = arities.find(command);
677 if (found == arities.end()) throw CommandError("UNKNOWN COMMAND " + command);
678 vector<string> args;
679 if (space != string::npos) {
680 std::size_t start = 0;
681 while (true) {
682 std::size_t next = rest.find(' ', start);
683 args.push_back(rest.substr(start, next == string::npos ? next : next - start));
684 if (next == string::npos) break;
685 start = next + 1;
686 }
687 }
688 if (int(args.size()) != found->second) throw CommandError("BAD ARGUMENTS " + command);
689 if (command == "SUGGEST") suggest(args);
690 else if (command == "DIST") dist(args);
691 else if (command == "SOUNDEX") {
692 string word = wordArgument(args[0]);
693 emit("SOUNDEX " + word + " " + soundex(word));
694 }
695 else if (command == "SOUNDS") sounds(args);
696 else if (command == "SPLIT") split(args);
697 else if (command == "PREFIX") prefix(args);
698 else if (command == "COMPLETE") complete(args);
699 else if (command == "LIST") listWords(args);
700 else if (command == "WORD") wordInfo(args);
701 else if (command == "ADD") add(args);
702 else if (command == "REMOVE") remove(args);
703 else if (command == "IGNORE") ignore(args);
704 else if (command == "UNIGNORE") unignore(args);
705 else if (command == "RULE") rule(args);
706 else if (command == "UNRULE") unrule(args);
707 else if (command == "RULES") showRules();
708 else if (command == "KEYBOARD") keyboardMode(args);
709 else if (command == "SET") setOption(args);
710 else if (command == "UNDO") undo();
711 else if (command == "REPORT") report();
712 else if (command == "HISTORY") history();
713 else if (command == "STATS") stats();
714 } catch (const CommandError& error) {
715 emit("ERROR " + string(error.what()));
716 }
717 }

720 for (const string& line : output) std::cout << line << '\n';
721 }
