3type Tree = Map<string, string>;
4type Kind = 'merge' | 'revert' | 'cherry-pick';
5interface Commit {
6 id: string;
7 sequence: number;
8 parents: Commit[];
9 tree: Tree;
10 message: string;
11}
12interface Operation { kind: Kind; target: Commit }
13interface Stash { label: string; base: Commit; tree: Tree }
14interface MergeResult { tree: Tree; conflicts: string[]; contents: Map<string, string> }

16const output: string[] = [];
17const commits = new Map<string, Commit>();
18const branches = new Map<string, Commit>();
19const tags = new Map<string, Commit>();
20const stashes: Stash[] = [];
21const patterns: string[] = [];
22let sequence = 0;
23let working: Tree = new Map();
24let index: Tree = new Map();
25let conflicts = new Set<string>();
26let operation: Operation | null = null;
27let branch: string | null = 'main';
28let detached: Commit;

30function paths(...trees: Tree[]): string[] {
31 return [...new Set(trees.flatMap(tree => [...tree.keys()]))].sort();
32}
33function different(a: Tree, b: Tree): string[] {
34 return paths(a, b).filter(path => a.get(path) !== b.get(path));
35}
36function equal(a: Tree, b: Tree): boolean { return different(a, b).length === 0; }
37function match(target: Tree, source: Tree, path: string): void {
38 if (source.has(path)) target.set(path, source.get(path)!);
39 else target.delete(path);
40}
41function fail(message: string): never { throw new Error(`error: ${message}`); }
42function create(parents: Commit[], tree: Tree, message: string): Commit {
43 const s = sequence++;
44 let canonical = `commit ${s}\n`;
45 for (const parent of parents) canonical += `parent ${parent.id}\n`;
46 for (const path of [...tree.keys()].sort()) canonical += `file ${path} ${tree.get(path)!}\n`;
47 canonical += `message ${message}\n`;
48 let hash = 0;
49 for (let i = 0; i < canonical.length; i++) hash = (hash * 131 + canonical.charCodeAt(i)) % 2147483647;
50 const commit: Commit = {
51 id: hash.toString(16).padStart(8, '0'), sequence: s,
52 parents: [...parents], tree: new Map(tree), message
53 };
54 commits.set(commit.id, commit);
55 return commit;
56}
57const root = create([], new Map(), 'init');
58branches.set('main', root);
59detached = root;
60function head(): Commit { return branch === null ? detached : branches.get(branch)!; }
61function move(commit: Commit): void {
62 if (branch === null) detached = commit;
63 else branches.set(branch, commit);
64}
65function record(parents: Commit[], tree: Tree, message: string): Commit {
66 const commit = create(parents, tree, message);
67 const old = parents[0].tree;
68 let added = 0, modified = 0, deleted = 0;
69 for (const path of different(old, tree)) {
70 if (!old.has(path)) added++;
71 else if (!tree.has(path)) deleted++;
72 else modified++;
73 }
74 move(commit);
75 output.push(`[${branch ?? 'detached HEAD'} ${commit.id}] ${message} (${added} added, ${modified} modified, ${deleted} deleted)`);
76 return commit;
77}
78function resolve(rev: string): Commit {
79 const [anchor, count] = rev.split('~');
80 let commit = anchor === 'HEAD' ? head() : branches.get(anchor) ?? tags.get(anchor) ?? commits.get(anchor);
81 if (!commit) fail(`unknown revision '${rev}'`);
82 for (let i = 0; i < Number(count ?? 0); i++) {
83 if (commit.parents.length === 0) fail(`unknown revision '${rev}'`);
84 commit = commit.parents[0];
85 }
86 return commit;
87}
88function distances(start: Commit): Map<Commit, number> {
89 const result = new Map<Commit, number>([[start, 0]]);
90 const queue = [start];
91 for (let i = 0; i < queue.length; i++) {
92 const commit = queue[i];
93 for (const parent of commit.parents) if (!result.has(parent)) {
94 result.set(parent, result.get(commit)! + 1);
95 queue.push(parent);
96 }
97 }
98 return result;
99}
100function ancestor(a: Commit, b: Commit): boolean { return distances(b).has(a); }
101function mergeBase(a: Commit, b: Commit): Commit {
102 const da = distances(a), db = distances(b);
103 const common = [...da.keys()].filter(commit => db.has(commit));
104 const candidates = common.filter(commit => !common.some(other => other !== commit && ancestor(commit, other)));
105 candidates.sort((x, y) => Math.max(da.get(x)!, db.get(x)!) - Math.max(da.get(y)!, db.get(y)!) || y.sequence - x.sequence);
106 return candidates[0];
107}
108function summary(commit: Commit): string {
109 const refs: string[] = [];
110 if (head() === commit) refs.push(branch === null ? 'HEAD' : `HEAD -> ${branch}`);
111 for (const name of [...branches.keys()].sort()) if (name !== branch && branches.get(name) === commit) refs.push(name);
112 for (const name of [...tags.keys()].sort()) if (tags.get(name) === commit) refs.push(`tag: ${name}`);
113 return `${commit.id}${refs.length ? ` (${refs.join(', ')})` : ''} ${commit.message}`;
114}
115function changes(a: Tree, b: Tree): void {
116 const changed = different(a, b);
117 if (!changed.length) output.push('(no changes)');
118 for (const path of changed) {
119 if (!a.has(path)) output.push(`A ${path}: ${b.get(path)!}`);
120 else if (!b.has(path)) output.push(`D ${path}: ${a.get(path)!}`);
121 else output.push(`M ${path}: ${a.get(path)!} -> ${b.get(path)!}`);
122 }
123}
124function inProgress(): void { if (operation) fail(`${operation.kind} in progress`); }
125function dirty(): boolean {
126 return !equal(index, head().tree) || [...index.keys()].some(path => working.get(path) !== index.get(path));
127}
128function uncommitted(): void { if (dirty()) fail('uncommitted changes'); }
129function overwrite(check: Iterable<string>): void {
130 const h = head().tree;
131 for (const path of [...new Set(check)].sort()) {
132 if (index.get(path) !== h.get(path) || working.get(path) !== h.get(path)) {
133 fail(`local changes to '${path}' would be overwritten`);
134 }
135 }
136}
137function switchTree(tree: Tree): void {
138 const changed = different(head().tree, tree);
139 overwrite(changed);
140 for (const path of changed) { match(index, tree, path); match(working, tree, path); }
141}
142function merge(base: Tree, ours: Tree, theirs: Tree): MergeResult {
143 const tree: Tree = new Map(), found: string[] = [], contents = new Map<string, string>();
144 for (const path of paths(base, ours, theirs)) {
145 const b = base.get(path), o = ours.get(path), t = theirs.get(path);
146 if (t === b || o === t) match(tree, ours, path);
147 else if (o === b) match(tree, theirs, path);
148 else {
149 found.push(path);
150 match(tree, ours, path);
151 contents.set(path, `<<<<<<< ${o ?? '(deleted)'} ======= ${t ?? '(deleted)'} >>>>>>>`);
152 }
153 }
154 return { tree, conflicts: found, contents };
155}
156function apply(base: Tree, theirs: Tree, kind: Kind, target: Commit, message: string, parents: Commit[], mergeCommit?: Commit): void {
157 const result = merge(base, head().tree, theirs);
158 const changed = different(head().tree, result.tree);
159 overwrite([...changed, ...result.conflicts]);
160 if (kind === 'merge') output.push(`Merge base ${mergeCommit!.id}`);
161 if (kind !== 'merge' && !result.conflicts.length && !changed.length) {
162 output.push('nothing to commit');
163 return;
164 }
165 for (const path of changed) { match(index, result.tree, path); match(working, result.tree, path); }
166 if (!result.conflicts.length) { record(parents, result.tree, message); return; }
167 for (const path of result.conflicts) {
168 working.set(path, result.contents.get(path)!);
169 output.push(`CONFLICT ${path}`);
170 }
171 conflicts = new Set(result.conflicts);
172 operation = { kind, target };
173 output.push(kind === 'merge' ? 'Automatic merge failed; fix conflicts and then commit the result.' :
174 kind === 'revert' ? `error: could not revert ${target.id}` : `error: could not apply ${target.id}`);
175}
176function glob(pattern: string, path: string): boolean {
177 let row = Array<boolean>(path.length + 1).fill(false);
178 row[0] = true;
179 for (const ch of pattern) {
180 const next = Array<boolean>(path.length + 1).fill(false);
181 if (ch === '*') next[0] = row[0];
182 for (let i = 1; i <= path.length; i++) {
183 next[i] = ch === '*' ? row[i] || next[i - 1] : row[i - 1] && (ch === '?' || ch === path[i - 1]);
184 }
185 row = next;
186 }
187 return row[path.length];
188}
189function ignored(path: string): boolean {
190 return !index.has(path) && !conflicts.has(path) && patterns.some(pattern => glob(pattern, path));
191}
192function pathspec(path: string): never { return fail(`pathspec '${path}' did not match any files`); }
193function existingPath(tree: Tree, path: string, rev: string): string {
194 if (!tree.has(path)) fail(`path '${path}' does not exist in '${rev}'`);
195 return tree.get(path)!;
196}
197function reachable(roots: Commit[]): Set<Commit> {
198 const result = new Set<Commit>();
199 for (const commit of roots) for (const item of distances(commit).keys()) result.add(item);
200 return result;
201}
202function execute(line: string): void {
203 const words = line.split(' '), command = words[0], arg = words[1];
204 switch (command) {
205 case 'WRITE': working.set(arg, line.slice(line.indexOf(' ', 6) + 1)); break;
206 case 'REMOVE':
207 if (!working.has(arg)) fail(`no such file '${arg}'`);
208 working.delete(arg);
209 break;
210 case 'ADD':
211 if (arg === 'ALL') {
212 const selected = paths(working, index).filter(path => !ignored(path));
213 for (const path of selected) match(index, working, path);
214 conflicts.clear();
215 } else {
216 if (working.has(arg) && ignored(arg)) fail(`path '${arg}' is ignored`);
217 if (!working.has(arg) && !index.has(arg) && !conflicts.has(arg)) pathspec(arg);
218 match(index, working, arg);
219 conflicts.delete(arg);
220 }
221 break;
222 case 'RM':
223 if (!index.has(arg)) pathspec(arg);
224 index.delete(arg); working.delete(arg); conflicts.delete(arg);
225 break;
226 case 'UNSTAGE':
227 if (!index.has(arg) && !head().tree.has(arg)) pathspec(arg);
228 match(index, head().tree, arg);
229 break;
230 case 'RESTORE':
231 if (words.length === 2) {
232 if (!index.has(arg)) pathspec(arg);
233 working.set(arg, index.get(arg)!);
234 } else {
235 const tree = resolve(words[2]).tree;
236 const value = existingPath(tree, arg, words[2]);
237 index.set(arg, value); working.set(arg, value); conflicts.delete(arg);
238 }
239 break;
240 case 'IGNORE': patterns.push(arg); break;
241 case 'COMMIT': {
242 if (conflicts.size) fail('unresolved conflicts');
243 const parents = operation?.kind === 'merge' ? [head(), operation.target] : [head()];
244 if (parents.length === 1 && equal(index, head().tree)) { output.push('nothing to commit'); break; }
245 record(parents, index, line.slice(7));
246 operation = null;
247 break;
248 }
249 case 'AMEND':
250 inProgress();
251 if (!head().parents.length) fail('cannot amend the root commit');
252 record(head().parents, index, line.slice(6));
253 break;
254 case 'STATUS': {
255 output.push(branch === null ? `HEAD detached at ${head().id}` : `On branch ${branch}`);
256 if (operation) {
257 const verb = operation.kind === 'merge' ? 'Merging' : operation.kind === 'revert' ? 'Reverting' : 'Cherry-picking';
258 output.push(`${verb} ${operation.target.id}`);
259 }
260 const before = output.length;
261 const all = [...new Set([...paths(head().tree, index, working), ...conflicts])].sort();
262 for (const path of all) {
263 if (conflicts.has(path)) { output.push(`UU ${path}`); continue; }
264 if (index.has(path)) {
265 const x = !head().tree.has(path) ? 'A' : head().tree.get(path) !== index.get(path) ? 'M' : '.';
266 const y = !working.has(path) ? 'D' : working.get(path) !== index.get(path) ? 'M' : '.';
267 if (x !== '.' || y !== '.') output.push(`${x}${y} ${path}`);
268 } else {
269 if (head().tree.has(path)) output.push(`D. ${path}`);
270 if (working.has(path) && !ignored(path)) output.push(`?? ${path}`);
271 }
272 }
273 if (output.length === before) output.push('nothing to commit, working tree clean');
274 break;
275 }
276 case 'LS': {
277 const tree = arg ? resolve(arg).tree : working;
278 if (!tree.size) output.push('(empty)');
279 for (const path of [...tree.keys()].sort()) output.push(`${path} ${tree.get(path)!}`);
280 break;
281 }
282 case 'CAT':
283 if (words.length === 3) output.push(existingPath(resolve(words[2]).tree, arg, words[2]));
284 else {
285 if (!working.has(arg)) fail(`no such file '${arg}'`);
286 output.push(working.get(arg)!);
287 }
288 break;
289 case 'DIFF':
290 if (arg === 'STAGED') changes(head().tree, index);
291 else if (arg) { const a = resolve(arg); const b = resolve(words[2]); changes(a.tree, b.tree); }
292 else changes(index, new Map([...working].filter(([path]) => index.has(path))));
293 break;
294 case 'LOG': {
295 let current = head();
296 for (let i = 0; i < (arg ? Number(arg) : Infinity); i++) {
297 output.push(summary(current));
298 if (!current.parents.length) break;
299 current = current.parents[0];
300 }
301 break;
302 }
303 case 'SHOW': {
304 const commit = resolve(arg ?? 'HEAD');
305 output.push(`commit ${summary(commit)}`, `parents: ${commit.parents.length ? commit.parents.map(parent => parent.id).join(' ') : 'none'}`);
306 changes(commit.parents[0]?.tree ?? new Map(), commit.tree);
307 break;
308 }
309 case 'BLAME': {
310 const value = existingPath(head().tree, arg, 'HEAD');
311 let commit = head();
312 while (commit.parents.length && commit.parents[0].tree.get(arg) === value) commit = commit.parents[0];
313 output.push(`${commit.id} (${commit.message}) ${value}`);
314 break;
315 }
316 case 'BRANCH': case 'TAG': {
317 const refs = command === 'BRANCH' ? branches : tags;
318 const noun = command === 'BRANCH' ? 'branch' : 'tag';
319 if (arg === '-d') {
320 const name = words[2], commit = refs.get(name);
321 if (!commit) fail(`no such ${noun} '${name}'`);
322 if (command === 'BRANCH') {
323 if (branch === name) fail(`cannot delete current branch '${name}'`);
324 if (!ancestor(commit, head())) fail(`branch '${name}' is not fully merged`);
325 }
326 refs.delete(name);
327 output.push(`Deleted ${noun} ${name} (was ${commit.id})`);
328 } else {
329 if (refs.has(arg)) fail(`${noun} '${arg}' already exists`);
330 const commit = resolve(words[2] ?? 'HEAD');
331 refs.set(arg, commit);
332 output.push(`Created ${noun} ${arg} at ${commit.id}`);
333 }
334 break;
335 }
336 case 'BRANCHES':
337 if (branch === null) output.push(`* (HEAD detached at ${head().id})`);
338 for (const name of [...branches.keys()].sort()) {
339 const commit = branches.get(name)!;
340 output.push(`${branch === name ? '*' : ' '} ${name} ${commit.id} ${commit.message}`);
341 }
342 break;
343 case 'TAGS':
344 if (!tags.size) output.push('(no tags)');
345 for (const name of [...tags.keys()].sort()) {
346 const commit = tags.get(name)!;
347 output.push(`${name} ${commit.id} ${commit.message}`);
348 }
349 break;
350 case 'SWITCH': {
351 const commit = branches.get(arg);
352 if (!commit) fail(`no such branch '${arg}'`);
353 if (branch === arg) { output.push(`Already on '${arg}'`); break; }
354 inProgress(); switchTree(commit.tree); branch = arg;
355 output.push(`Switched to branch '${arg}'`);
356 break;
357 }
358 case 'DETACH': {
359 const commit = resolve(arg);
360 inProgress(); switchTree(commit.tree); branch = null; detached = commit;
361 output.push(`HEAD is now at ${commit.id} ${commit.message}`);
362 break;
363 }
364 case 'MERGE': {
365 const target = resolve(arg);
366 inProgress();
367 const old = head();
368 if (ancestor(target, old)) { output.push('Already up to date'); break; }
369 if (ancestor(old, target)) {
370 switchTree(target.tree); move(target);
371 output.push(`Fast-forward ${old.id}..${target.id}`);
372 break;
373 }
374 uncommitted();
375 const base = mergeBase(old, target);
376 apply(base.tree, target.tree, 'merge', target, `Merge ${arg}`, [old, target], base);
377 break;
378 }
379 case 'MERGEBASE': {
380 const a = resolve(arg), b = resolve(words[2]);
381 output.push(mergeBase(a, b).id);
382 break;
383 }
384 case 'REBASE': {
385 const onto = resolve(arg);
386 inProgress();
387 if (ancestor(onto, head())) { output.push('Already up to date'); break; }
388 uncommitted();
389 const list: Commit[] = [];
390 let current = head();
391 while (!ancestor(current, onto)) { list.push(current); current = current.parents[0]; }
392 list.reverse();
393 const picks = new Map<Commit, Tree>();
394 let tree = onto.tree;
395 for (const commit of list) {
396 const result = merge(commit.parents[0].tree, tree, commit.tree);
397 if (result.conflicts.length) fail(`could not rebase ${commit.id}: conflict in '${result.conflicts[0]}'`);
398 if (!equal(result.tree, tree)) { picks.set(commit, result.tree); tree = result.tree; }
399 }
400 switchTree(tree);
401 let target = onto;
402 for (const commit of list) {
403 if (picks.has(commit)) {
404 const next = create([target], picks.get(commit)!, commit.message);
405 output.push(`pick ${commit.id} -> ${next.id} ${commit.message}`);
406 target = next;
407 } else output.push(`skip ${commit.id} ${commit.message}`);
408 }
409 move(target);
410 output.push(`HEAD is now at ${target.id} ${target.message}`);
411 break;
412 }
413 case 'RESET': {
414 const target = resolve(words[2]), tree = target.tree;
415 if (arg === 'HARD') {
416 const all = [...new Set([...paths(index, tree), ...conflicts])];
417 for (const path of all) match(working, tree, path);
418 }
419 if (arg !== 'SOFT') index = new Map(tree);
420 move(target); conflicts.clear(); operation = null;
421 output.push(`HEAD is now at ${target.id} ${target.message}`);
422 break;
423 }
424 case 'REVERT': case 'CHERRYPICK': {
425 const target = resolve(arg);
426 inProgress(); uncommitted();
427 const parentTree = target.parents[0]?.tree ?? new Map<string, string>();
428 if (command === 'REVERT') apply(target.tree, parentTree, 'revert', target, `Revert "${target.message}"`, [head()]);
429 else apply(parentTree, target.tree, 'cherry-pick', target, target.message, [head()]);
430 break;
431 }
432 case 'STASH':
433 if (arg === 'LIST') {
434 if (!stashes.length) output.push('(no stash entries)');
435 stashes.forEach((stash, i) => output.push(`stash@{${i}}: ${stash.label}`));
436 } else if (arg === 'DROP') {
437 if (!stashes.length) fail('no stash entries');
438 output.push(`Dropped stash@{0}: ${stashes.shift()!.label}`);
439 } else if (arg === 'POP') {
440 if (!stashes.length) fail('no stash entries');
441 inProgress(); uncommitted();
442 const stash = stashes[0], result = merge(stash.base.tree, head().tree, stash.tree);
443 if (result.conflicts.length) fail(`stash conflict in '${result.conflicts[0]}'`);
444 const changed = different(head().tree, result.tree);
445 overwrite(changed);
446 for (const path of changed) match(working, result.tree, path);
447 stashes.shift();
448 output.push(`Restored stash@{0}: ${stash.label}`);
449 changes(head().tree, result.tree);
450 } else {
451 inProgress();
452 if (!dirty()) { output.push('No local changes to save'); break; }
453 const tree: Tree = new Map(), h = head();
454 for (const path of paths(index, h.tree)) {
455 if (working.has(path)) tree.set(path, working.get(path)!);
456 match(working, h.tree, path);
457 }
458 index = new Map(h.tree);
459 const label = `WIP on ${branch ?? '(no branch)'}: ${h.id} ${h.message}`;
460 stashes.unshift({ label, base: h, tree });
461 output.push(`Saved stash@{0}: ${label}`);
462 }
463 break;
464 case 'GRAPH': {
465 const all = [...reachable([head(), ...branches.values(), ...tags.values()])].sort((a, b) => b.sequence - a.sequence);
466 let lanes: Commit[] = [];
467 for (const commit of all) {
468 if (!lanes.includes(commit)) lanes.push(commit);
469 const k = lanes.indexOf(commit);
470 const symbols = lanes.map((item, i) => i === k ? '*' : item === commit ? '/' : '|');
471 output.push(`${symbols.join(' ')} ${summary(commit)}`);
472 lanes = lanes.filter((item, i) => item !== commit || i === k);
473 if (!commit.parents.length) lanes.splice(k, 1);
474 else {
475 lanes[k] = commit.parents[0];
476 if (commit.parents.length === 2 && !lanes.includes(commit.parents[1])) lanes.splice(k + 1, 0, commit.parents[1]);
477 }
478 }
479 break;
480 }
481 case 'DESCRIBE': {
482 const target = resolve(arg ?? 'HEAD'), distance = distances(target);
483 const names = [...tags.keys()].filter(name => distance.has(tags.get(name)!));
484 names.sort((a, b) => distance.get(tags.get(a)!)! - distance.get(tags.get(b)!)! || (a < b ? -1 : a > b ? 1 : 0));
485 if (!names.length) fail(`no tag can describe '${target.id}'`);
486 const name = names[0], d = distance.get(tags.get(name)!)!;
487 output.push(d === 0 ? name : `${name}-${d}-g${target.id}`);
488 break;
489 }
490 case 'GC': {
491 const roots = [head(), ...branches.values(), ...tags.values(), ...stashes.map(stash => stash.base)];
492 if (operation) roots.push(operation.target);
493 const keep = reachable(roots);
494 let removed = 0;
495 for (const [id, commit] of commits) if (!keep.has(commit)) { commits.delete(id); removed++; }
496 output.push(`Removed unreachable commits: ${removed}`);
497 break;
498 }
499 }
500}
501for (const line of fs.readFileSync(0, 'utf8').split(/\r?\n/)) {
502 if (line === 'EXIT') break;
503 if (!line) continue;
504 try { execute(line); }
505 catch (error) { output.push((error as Error).message); }
506}
507if (output.length) process.stdout.write(output.join('\n') + '\n');
