1local function bad(pattern, ...)
2 error(string.format(pattern, ...), 0)
3end
4local function split(text, limit)
5 local result, start = {}, 1
6 while not limit or #result + 1 < limit do
7 local at = text:find(" ", start, true)
8 if not at then
9 break
10 end
11 result[#result + 1] = text:sub(start, at - 1)
12 start = at + 1
13 end
14 result[#result + 1] = text:sub(start)
15 return result
16end
17local function clone(files)
18 local result = {}
19 for path, text in pairs(files) do
20 result[path] = text
21 end
22 return result
23end
24local function keys(...)
25 local all, result = {}, {}
26 for _, files in ipairs({...}) do
27 for path in pairs(files) do
28 all[path] = true
29 end
30 end
31 for path in pairs(all) do
32 result[#result + 1] = path
33 end
34 table.sort(result)
35 return result
36end
37local function differing(a, b)
38 local result = {}
39 for _, path in ipairs(keys(a, b)) do
40 if a[path] ~= b[path] then
41 result[#result + 1] = path
42 end
43 end
44 return result
45end
46local function same(a, b)
47 return #differing(a, b) == 0
48end
49local function match_path(target, source, path)
50 target[path] = source[path]
51end
52local function three_way(base, ours, theirs)
53 local merged, conflicts = {}, {}
54 for _, path in ipairs(keys(base, ours, theirs)) do
55 if theirs[path] == base[path] or ours[path] == theirs[path] then
56 merged[path] = ours[path]
57 elseif ours[path] == base[path] then
58 merged[path] = theirs[path]
59 else
60 conflicts[#conflicts + 1] = path
61 merged[path] = ours[path]
62 end
63 end
64 return merged, conflicts
65end
66local function distances(start)
67 local result, queue, at = {[start] = 0}, {start}, 1
68 while at <= #queue do
69 local current = queue[at]
70 at = at + 1
71 for _, parent in ipairs(current.parents) do
72 if result[parent] == nil then
73 result[parent] = result[current] + 1
74 queue[#queue + 1] = parent
75 end
76 end
77 end
78 return result
79end
80local function ancestor(a, b)
81 return distances(b)[a] ~= nil
82end
83local function merge_base(a, b)
84 local da, db, candidates, excluded = distances(a), distances(b), {}, {}
85 for c in pairs(da) do
86 if db[c] ~= nil then
87 candidates[c] = true
88 end
89 end
90 for c in pairs(candidates) do
91 for _, parent in ipairs(c.parents) do
92 excluded[parent] = true
93 end
94 end
95 local best, best_distance
96 for c in pairs(candidates) do
97 if not excluded[c] then
98 local distance = math.max(da[c], db[c])
99 if not best or distance < best_distance or distance == best_distance and c.sequence > best.sequence then
100 best, best_distance = c, distance
101 end
102 end
103 end
104 return best
105end
106local function glob(pattern, path)
107 local previous = {[0] = true}
108 for i = 1, #pattern do
109 local c, current = pattern:sub(i, i), {}
110 if c == "*" then
111 current[0] = previous[0]
112 end
113 for j = 1, #path do
114 if c == "*" then
115 current[j] = previous[j] or current[j - 1]
116 else
117 current[j] = previous[j - 1] and (c == "?" or c == path:sub(j, j))
118 end
119 end
120 previous = current
121 end
122 return previous[#path] or false
123end
124local function parent_tree(c)
125 return c.parents[1] and c.parents[1].files or {}
126end
127local function reachable(roots)
128 local result = {}
129 for _, root in ipairs(roots) do
130 for c in pairs(distances(root)) do
131 result[c] = true
132 end
133 end
134 return result
135end

140 r.branches.main = r:create({}, {}, "init")

145end
146function Repository:head()
147 return self.attached and self.branches[self.attached] or self.detached
148end
149function Repository:move_head(c)
150 if self.attached then
151 self.branches[self.attached] = c
152 else
153 self.detached = c
154 end
155end
156function Repository:create(parents, files, message)
157 local lines = {"commit " .. self.created}
158 for _, parent in ipairs(parents) do
159 lines[#lines + 1] = "parent " .. parent.id
160 end
161 for _, path in ipairs(keys(files)) do
162 lines[#lines + 1] = "file " .. path .. " " .. files[path]
163 end
164 lines[#lines + 1] = "message " .. message
165 local text, hash = table.concat(lines, "\n") .. "\n", 0
166 for i = 1, #text do
167 hash = (hash * 131 + text:byte(i)) % 2147483647
168 end
169 local c = {id = string.format("%08x", hash), sequence = self.created, parents = {table.unpack(parents)}, files = clone(files), message = message}
170 self.created = self.created + 1
171 self.commits[c.id] = c
172 return c
173end
174function Repository:record(parents, files, message)
175 local c = self:create(parents, files, message)
176 self:move_head(c)
177 local added, modified, deleted, old = 0, 0, 0, parents[1].files
178 for _, path in ipairs(differing(old, files)) do
179 if old[path] == nil then
180 added = added + 1
181 elseif files[path] == nil then
182 deleted = deleted + 1
183 else
184 modified = modified + 1
185 end
186 end
187 self:emit("[%s %s] %s (%d added, %d modified, %d deleted)", self.attached or "detached HEAD", c.id, message, added, modified, deleted)
188end
189function Repository:revision(rev)
190 local at = rev:find("~", 1, true)
191 local anchor = at and rev:sub(1, at - 1) or rev
192 local c
193 if anchor == "HEAD" then
194 c = self:head()
195 else
196 c = self.branches[anchor] or self.tags[anchor] or self.commits[anchor]
197 end
198 if at then
199 for _ = 1, tonumber(rev:sub(at + 1)) do
200 if c then
201 c = c.parents[1]
202 end
203 end
204 end
205 if not c then
206 bad("unknown revision '%s'", rev)
207 end
208 return c
209end
210function Repository:summary(c)
211 local refs = {}
212 if self:head() == c then
213 refs[#refs + 1] = self.attached and "HEAD -> " .. self.attached or "HEAD"
214 end
215 for _, name in ipairs(keys(self.branches)) do
216 if name ~= self.attached and self.branches[name] == c then
217 refs[#refs + 1] = name
218 end
219 end
220 for _, name in ipairs(keys(self.tags)) do
221 if self.tags[name] == c then
222 refs[#refs + 1] = "tag: " .. name
223 end
224 end
225 return c.id .. (#refs == 0 and "" or " (" .. table.concat(refs, ", ") .. ")") .. " " .. c.message
226end
227function Repository:changes(a, b)
228 local changed = differing(a, b)
229 if #changed == 0 then
230 self:emit("(no changes)")
231 return
232 end
233 for _, path in ipairs(changed) do
234 if a[path] == nil then
235 self:emit("A %s: %s", path, b[path])
236 elseif b[path] == nil then
237 self:emit("D %s: %s", path, a[path])
238 else
239 self:emit("M %s: %s -> %s", path, a[path], b[path])
240 end
241 end
242end
243function Repository:in_progress()
244 if self.pending then
245 bad("%s in progress", self.pending.kind)
246 end
247end
248function Repository:uncommitted()
249 if not same(self.index, self:head().files) then
250 return true
251 end
252 for path in pairs(self.index) do
253 if self.work[path] ~= self.index[path] then
254 return true
255 end
256 end
257 return false
258end
259function Repository:require_clean()
260 if self:uncommitted() then
261 bad("uncommitted changes")
262 end
263end
264function Repository:overwrite(selected)
265 table.sort(selected)
266 local head = self:head().files
267 for _, path in ipairs(selected) do
268 if self.index[path] ~= head[path] or self.work[path] ~= head[path] then
269 bad("local changes to '%s' would be overwritten", path)
270 end
271 end
272end
273function Repository:switch_tree(files)
274 local changed = differing(self:head().files, files)
275 self:overwrite(changed)
276 for _, path in ipairs(changed) do
277 match_path(self.index, files, path)
278 match_path(self.work, files, path)
279 end
280end
281function Repository:ignored(path)
282 if self.index[path] ~= nil or self.conflicts[path] then
283 return false
284 end
285 for _, pattern in ipairs(self.ignores) do
286 if glob(pattern, path) then
287 return true
288 end
289 end
290 return false
291end
292function Repository:apply(base, theirs, kind, target, message, parents, base_commit)
293 local merged, conflicts = three_way(base, self:head().files, theirs)
294 local changed = differing(self:head().files, merged)
295 local selected = {table.unpack(changed)}
296 for _, path in ipairs(conflicts) do
297 selected[#selected + 1] = path
298 end
299 self:overwrite(selected)
300 if kind == "merge" then
301 self:emit("Merge base %s", base_commit.id)
302 end
303 if kind ~= "merge" and #conflicts == 0 and same(merged, self:head().files) then
304 self:emit("nothing to commit")
305 return
306 end
307 for _, path in ipairs(changed) do
308 match_path(self.index, merged, path)
309 match_path(self.work, merged, path)
310 end
311 if #conflicts == 0 then
312 self:record(parents, merged, message)
313 return
314 end
315 self.conflicts = {}
316 for _, path in ipairs(conflicts) do
317 self.work[path] = "<<<<<<< " .. (self:head().files[path] or "(deleted)") .. " ======= " .. (theirs[path] or "(deleted)") .. " >>>>>>>"
318 self.conflicts[path] = true
319 self:emit("CONFLICT %s", path)
320 end
321 self.pending = {kind = kind, target = target}
322 if kind == "merge" then
323 self:emit("Automatic merge failed; fix conflicts and then commit the result.")
324 elseif kind == "revert" then
325 self:emit("error: could not revert %s", target.id)
326 else
327 self:emit("error: could not apply %s", target.id)
328 end
329end
330function Repository:roots(extra)
331 local result = {self:head()}
332 for _, c in pairs(self.branches) do
333 result[#result + 1] = c
334 end
335 for _, c in pairs(self.tags) do
336 result[#result + 1] = c
337 end
338 if extra then
339 for _, entry in ipairs(self.stash) do
340 result[#result + 1] = entry.base
341 end
342 if self.pending then
343 result[#result + 1] = self.pending.target
344 end
345 end
346 return result
347end
348function Repository:command(line)
349 local w = split(line)
350 local command, h = w[1], self:head()
351 if command == "WRITE" then
352 local args = split(line, 3)
353 self.work[args[2]] = args[3]
354 elseif command == "REMOVE" then
355 if self.work[w[2]] == nil then
356 bad("no such file '%s'", w[2])
357 end
358 self.work[w[2]] = nil
359 elseif command == "IGNORE" then
360 self.ignores[#self.ignores + 1] = w[2]
361 elseif command == "ADD" then
362 local path = w[2]
363 if path == "ALL" then
364 for _, path in ipairs(keys(self.work, self.index)) do
365 if not self:ignored(path) then
366 match_path(self.index, self.work, path)
367 end
368 end
369 self.conflicts = {}
370 else
371 if self.work[path] ~= nil and self:ignored(path) then
372 bad("path '%s' is ignored", path)
373 end
374 if self.work[path] == nil and self.index[path] == nil and not self.conflicts[path] then
375 bad("pathspec '%s' did not match any files", path)
376 end
377 match_path(self.index, self.work, path)
378 self.conflicts[path] = nil
379 end
380 elseif command == "RM" then
381 local path = w[2]
382 if self.index[path] == nil then
383 bad("pathspec '%s' did not match any files", path)
384 end
385 self.index[path], self.work[path], self.conflicts[path] = nil, nil, nil
386 elseif command == "UNSTAGE" then
387 local path = w[2]
388 if self.index[path] == nil and h.files[path] == nil then
389 bad("pathspec '%s' did not match any files", path)
390 end
391 match_path(self.index, h.files, path)
392 elseif command == "RESTORE" or command == "CAT" then
393 local path, files = w[2], command == "CAT" and self.work or self.index
394 if #w == 3 then
395 files = self:revision(w[3]).files
396 if files[path] == nil then
397 bad("path '%s' does not exist in '%s'", path, w[3])
398 end
399 elseif files[path] == nil then
400 if command == "CAT" then
401 bad("no such file '%s'", path)
402 else
403 bad("pathspec '%s' did not match any files", path)
404 end
405 end
406 if command == "CAT" then
407 self:emit("%s", files[path])
408 else
409 self.work[path] = files[path]
410 if #w == 3 then
411 self.index[path] = files[path]
412 self.conflicts[path] = nil
413 end
414 end
415 elseif command == "COMMIT" then
416 if next(self.conflicts) then
417 bad("unresolved conflicts")
418 end
419 local message, parents = split(line, 2)[2], {h}
420 if self.pending and self.pending.kind == "merge" then
421 parents[2] = self.pending.target
422 elseif same(self.index, h.files) then
423 self:emit("nothing to commit")
424 return
425 end
426 self:record(parents, self.index, message)
427 self.pending = nil
428 elseif command == "AMEND" then
429 self:in_progress()
430 if #h.parents == 0 then
431 bad("cannot amend the root commit")
432 end
433 self:record(h.parents, self.index, split(line, 2)[2])
434 elseif command == "STATUS" then
435 if self.attached then
436 self:emit("On branch %s", self.attached)
437 else
438 self:emit("HEAD detached at %s", h.id)
439 end
440 if self.pending then
441 self:emit("%s %s", ({merge = "Merging", revert = "Reverting", ["cherry-pick"] = "Cherry-picking"})[self.pending.kind], self.pending.target.id)
442 end
443 local printed = false
444 for _, path in ipairs(keys(h.files, self.index, self.work, self.conflicts)) do
445 if self.conflicts[path] then
446 self:emit("UU %s", path)
447 printed = true
448 elseif self.index[path] ~= nil then
449 local x = h.files[path] == nil and "A" or h.files[path] ~= self.index[path] and "M" or "."
450 local y = self.work[path] == nil and "D" or self.work[path] ~= self.index[path] and "M" or "."
451 if x ~= "." or y ~= "." then
452 self:emit("%s%s %s", x, y, path)
453 printed = true
454 end
455 else
456 if h.files[path] ~= nil then
457 self:emit("D. %s", path)
458 printed = true
459 end
460 if self.work[path] ~= nil and not self:ignored(path) then
461 self:emit("?? %s", path)
462 printed = true
463 end
464 end
465 end
466 if not printed then
467 self:emit("nothing to commit, working tree clean")
468 end
469 elseif command == "LS" then
470 local files = w[2] and self:revision(w[2]).files or self.work
471 local selected = keys(files)
472 if #selected == 0 then
473 self:emit("(empty)")
474 end
475 for _, path in ipairs(selected) do
476 self:emit("%s %s", path, files[path])
477 end
478 elseif command == "DIFF" then
479 if #w == 1 then
480 local work = {}
481 for path in pairs(self.index) do
482 match_path(work, self.work, path)
483 end
484 self:changes(self.index, work)
485 elseif #w == 2 then
486 self:changes(h.files, self.index)
487 else
488 local a = self:revision(w[2])
489 local b = self:revision(w[3])
490 self:changes(a.files, b.files)
491 end
492 elseif command == "LOG" then
493 local c, n, limit = h, 0, w[2] and tonumber(w[2]) or self.created
494 while c and n < limit do
495 self:emit("%s", self:summary(c))
496 n = n + 1
497 c = c.parents[1]
498 end
499 elseif command == "SHOW" then
500 local c = self:revision(w[2] or "HEAD")
501 self:emit("commit %s", self:summary(c))
502 local parents = {}
503 for _, parent in ipairs(c.parents) do
504 parents[#parents + 1] = parent.id
505 end
506 self:emit("parents: %s", #parents == 0 and "none" or table.concat(parents, " "))
507 self:changes(parent_tree(c), c.files)
508 elseif command == "BLAME" then
509 local path, c = w[2], h
510 if h.files[path] == nil then
511 bad("path '%s' does not exist in 'HEAD'", path)
512 end
513 local value = h.files[path]
514 while c.parents[1] and c.parents[1].files[path] == value do
515 c = c.parents[1]
516 end
517 self:emit("%s (%s) %s", c.id, c.message, value)
518 elseif command == "BRANCH" or command == "TAG" then
519 local refs, label = command == "TAG" and self.tags or self.branches, command == "TAG" and "tag" or "branch"
520 if w[2] == "-d" then
521 local name = w[3]
522 local c = refs[name]
523 if not c then
524 bad("no such %s '%s'", label, name)
525 end
526 if command == "BRANCH" then
527 if self.attached == name then
528 bad("cannot delete current branch '%s'", name)
529 end
530 if not ancestor(c, h) then
531 bad("branch '%s' is not fully merged", name)
532 end
533 end
534 refs[name] = nil
535 self:emit("Deleted %s %s (was %s)", label, name, c.id)
536 else
537 local name = w[2]
538 if refs[name] then
539 bad("%s '%s' already exists", label, name)
540 end
541 local c = self:revision(w[3] or "HEAD")
542 refs[name] = c
543 self:emit("Created %s %s at %s", label, name, c.id)
544 end
545 elseif command == "BRANCHES" then
546 if not self.attached then
547 self:emit("* (HEAD detached at %s)", h.id)
548 end
549 for _, name in ipairs(keys(self.branches)) do
550 local c = self.branches[name]
551 self:emit("%s %s %s %s", name == self.attached and "*" or " ", name, c.id, c.message)
552 end
553 elseif command == "TAGS" then
554 local names = keys(self.tags)
555 if #names == 0 then
556 self:emit("(no tags)")
557 end
558 for _, name in ipairs(names) do
559 local c = self.tags[name]
560 self:emit("%s %s %s", name, c.id, c.message)
561 end
562 elseif command == "SWITCH" then
563 local name, c = w[2], self.branches[w[2]]
564 if not c then
565 bad("no such branch '%s'", name)
566 end
567 if name == self.attached then
568 self:emit("Already on '%s'", name)
569 return
570 end
571 self:in_progress()
572 self:switch_tree(c.files)
573 self.attached, self.detached = name, nil
574 self:emit("Switched to branch '%s'", name)
575 elseif command == "DETACH" then
576 local c = self:revision(w[2])
577 self:in_progress()
578 self:switch_tree(c.files)
579 self.attached, self.detached = nil, c
580 self:emit("HEAD is now at %s %s", c.id, c.message)
581 elseif command == "MERGEBASE" then
582 local a = self:revision(w[2])
583 local b = self:revision(w[3])
584 self:emit("%s", merge_base(a, b).id)
585 elseif command == "MERGE" or command == "REVERT" or command == "CHERRYPICK" then
586 local c = self:revision(w[2])
587 self:in_progress()
588 if command == "MERGE" then
589 if ancestor(c, h) then
590 self:emit("Already up to date")
591 return
592 end
593 if ancestor(h, c) then
594 self:switch_tree(c.files)
595 self:move_head(c)
596 self:emit("Fast-forward %s..%s", h.id, c.id)
597 return
598 end
599 end
600 self:require_clean()
601 if command == "MERGE" then
602 local base = merge_base(h, c)
603 self:apply(base.files, c.files, "merge", c, "Merge " .. w[2], {h, c}, base)
604 elseif command == "REVERT" then
605 self:apply(c.files, parent_tree(c), "revert", c, 'Revert "' .. c.message .. '"', {h})
606 else
607 self:apply(parent_tree(c), c.files, "cherry-pick", c, c.message, {h})
608 end
609 elseif command == "REBASE" then
610 local upstream = self:revision(w[2])
611 self:in_progress()
612 if ancestor(upstream, h) then
613 self:emit("Already up to date")
614 return
615 end
616 self:require_clean()
617 local list, c = {}, h
618 while not ancestor(c, upstream) do
619 list[#list + 1] = c
620 c = c.parents[1]
621 end
622 for i = 1, #list // 2 do
623 list[i], list[#list + 1 - i] = list[#list + 1 - i], list[i]
624 end
625 local current, picks = upstream.files, {}
626 for i, old in ipairs(list) do
627 local merged, conflicts = three_way(parent_tree(old), current, old.files)
628 if #conflicts > 0 then
629 bad("could not rebase %s: conflict in '%s'", old.id, conflicts[1])
630 end
631 if not same(merged, current) then
632 picks[i], current = merged, merged
633 end
634 end
635 self:switch_tree(current)
636 local tip = upstream
637 for i, old in ipairs(list) do
638 if not picks[i] then
639 self:emit("skip %s %s", old.id, old.message)
640 else
641 local c = self:create({tip}, picks[i], old.message)
642 self:emit("pick %s -> %s %s", old.id, c.id, old.message)
643 tip = c
644 end
645 end
646 self:move_head(tip)
647 self:emit("HEAD is now at %s %s", tip.id, tip.message)
648 elseif command == "RESET" then
649 local c = self:revision(w[3])
650 if w[2] == "HARD" then
651 for _, path in ipairs(keys(self.index, c.files, self.conflicts)) do
652 match_path(self.work, c.files, path)
653 end
654 end
655 if w[2] ~= "SOFT" then
656 self.index = clone(c.files)
657 end
658 self:move_head(c)
659 self.conflicts, self.pending = {}, nil
660 self:emit("HEAD is now at %s %s", c.id, c.message)
661 elseif command == "STASH" then
662 local action = w[2]
663 if action == "LIST" then
664 if #self.stash == 0 then
665 self:emit("(no stash entries)")
666 end
667 for i, entry in ipairs(self.stash) do
668 self:emit("stash@{%d}: %s", i - 1, entry.label)
669 end
670 return
671 end
672 if action and #self.stash == 0 then
673 bad("no stash entries")
674 end
675 if action == "DROP" then
676 local entry = table.remove(self.stash, 1)
677 self:emit("Dropped stash@{0}: %s", entry.label)
678 return
679 end
680 self:in_progress()
681 if action == "POP" then
682 self:require_clean()
683 local entry = self.stash[1]
684 local merged, conflicts = three_way(entry.base.files, h.files, entry.files)
685 if #conflicts > 0 then
686 bad("stash conflict in '%s'", conflicts[1])
687 end
688 local changed = differing(h.files, merged)
689 self:overwrite(changed)
690 for _, path in ipairs(changed) do
691 match_path(self.work, merged, path)
692 end
693 table.remove(self.stash, 1)
694 self:emit("Restored stash@{0}: %s", entry.label)
695 self:changes(h.files, merged)
696 else
697 if not self:uncommitted() then
698 self:emit("No local changes to save")
699 return
700 end
701 local saved = {}
702 for _, path in ipairs(keys(self.index, h.files)) do
703 match_path(saved, self.work, path)
704 match_path(self.work, h.files, path)
705 end
706 self.index = clone(h.files)
707 local label = "WIP on " .. (self.attached or "(no branch)") .. ": " .. h.id .. " " .. h.message
708 table.insert(self.stash, 1, {label = label, base = h, files = saved})
709 self:emit("Saved stash@{0}: %s", label)
710 end
711 elseif command == "GRAPH" then
712 local ordered = {}
713 for c in pairs(reachable(self:roots(false))) do
714 ordered[#ordered + 1] = c
715 end
716 table.sort(ordered, function(a, b)
717 return a.sequence > b.sequence
718 end)
719 local lanes = {}
720 for _, c in ipairs(ordered) do
721 local at
722 for i, held in ipairs(lanes) do
723 if held == c then
724 at = i
725 break
726 end
727 end
728 if not at then
729 lanes[#lanes + 1] = c
730 at = #lanes
731 end
732 local symbols, kept = {}, {}
733 for i, held in ipairs(lanes) do
734 symbols[i] = held == c and (i == at and "*" or "/") or "|"
735 if held ~= c or i == at then
736 kept[#kept + 1] = held
737 end
738 end
739 self:emit("%s %s", table.concat(symbols, " "), self:summary(c))
740 lanes = kept
741 if #c.parents == 0 then
742 table.remove(lanes, at)
743 else
744 lanes[at] = c.parents[1]
745 if c.parents[2] then
746 local found = false
747 for _, held in ipairs(lanes) do
748 if held == c.parents[2] then
749 found = true
750 end
751 end
752 if not found then
753 table.insert(lanes, at + 1, c.parents[2])
754 end
755 end
756 end
757 end
758 elseif command == "DESCRIBE" then
759 local c = self:revision(w[2] or "HEAD")
760 local d, best, distance = distances(c), nil, nil
761 for _, name in ipairs(keys(self.tags)) do
762 local n = d[self.tags[name]]
763 if n ~= nil and (not best or n < distance) then
764 best, distance = name, n
765 end
766 end
767 if not best then
768 bad("no tag can describe '%s'", c.id)
769 end
770 if distance == 0 then
771 self:emit("%s", best)
772 else
773 self:emit("%s-%d-g%s", best, distance, c.id)
774 end
775 elseif command == "GC" then
776 local keep, removed = reachable(self:roots(true)), 0
777 for id, c in pairs(self.commits) do
778 if not keep[c] then
779 self.commits[id] = nil
780 removed = removed + 1
781 end
782 end
783 self:emit("Removed unreachable commits: %d", removed)
784 end
785end

795end

797 io.write(table.concat(repository.output, "\n"), "\n")
