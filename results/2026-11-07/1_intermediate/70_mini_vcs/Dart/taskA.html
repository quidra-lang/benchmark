12 : parents = List.of(parents), tree = Map.of(tree);

40List<String> paths(List<Tree> trees) =>
41 (trees.expand((tree) => tree.keys).toSet().toList()..sort());
42List<String> different(Tree a, Tree b) =>
43 paths([a, b]).where((path) => a[path] != b[path]).toList();
44bool equal(Tree a, Tree b) => different(a, b).isEmpty;
45void match(Tree target, Tree source, String path) {
46 if (source.containsKey(path)) {
47 target[path] = source[path]!;
48 } else {
49 target.remove(path);
50 }
51}
52Never fail(String message) => throw VcsError('error: $message');

54class Vcs {
55 final List<String> output = [];
56 final Map<String, Commit> commits = {};
57 final Map<String, Commit> branches = {};
58 final Map<String, Commit> tags = {};
59 final List<Stash> stashes = [];
60 final List<String> patterns = [];
61 int sequence = 0;
62 Tree working = {};
63 Tree index = {};
64 Set<String> conflicts = {};
65 Operation? operation;
66 String? branch = 'main';
67 late Commit detached;
68
69 Vcs() {
70 final root = create([], {}, 'init');
71 branches['main'] = root;
72 detached = root;
73 }
74
75 Commit get head => branch == null ? detached : branches[branch]!;
76
77 Commit create(List<Commit> parents, Tree tree, String message) {
78 final s = sequence++;
79 final canonical = StringBuffer('commit $s\n');
80 for (final parent in parents) {
81 canonical.writeln('parent ${parent.id}');
82 }
83 for (final path in tree.keys.toList()..sort()) {
84 canonical.writeln('file $path ${tree[path]}');
85 }
86 canonical.writeln('message $message');
87 var hash = 0;
88 for (final code in canonical.toString().codeUnits) {
89 hash = (hash * 131 + code) % 2147483647;
90 }
91 final commit = Commit(hash.toRadixString(16).padLeft(8, '0'), s,
92 parents, tree, message);
93 commits[commit.id] = commit;
94 return commit;
95 }
96
97 void move(Commit commit) {
98 if (branch == null) {
99 detached = commit;
100 } else {
101 branches[branch!] = commit;
102 }
103 }
104
105 Commit record(List<Commit> parents, Tree tree, String message) {
106 final commit = create(parents, tree, message);
107 final old = parents.first.tree;
108 var added = 0, modified = 0, deleted = 0;
109 for (final path in different(old, tree)) {
110 if (!old.containsKey(path)) {
111 added++;
112 } else if (!tree.containsKey(path)) {
113 deleted++;
114 } else {
115 modified++;
116 }
117 }
118 move(commit);
119 output.add('[${branch ?? 'detached HEAD'} ${commit.id}] $message '
120 '($added added, $modified modified, $deleted deleted)');
121 return commit;
122 }
123
124 Commit resolve(String rev) {
125 final parts = rev.split('~');
126 final anchor = parts.first;
127 Commit? commit = anchor == 'HEAD'
128 ? head : branches[anchor] ?? tags[anchor] ?? commits[anchor];
129 if (commit == null) fail("unknown revision '$rev'");
130 final count = parts.length == 1 ? 0 : int.parse(parts[1]);
131 for (var i = 0; i < count; i++) {
132 if (commit!.parents.isEmpty) fail("unknown revision '$rev'");
133 commit = commit.parents.first;
134 }
135 return commit!;
136 }
137
138 Map<Commit, int> distances(Commit start) {
139 final result = <Commit, int>{start: 0};
140 final queue = <Commit>[start];
141 for (var i = 0; i < queue.length; i++) {
142 final commit = queue[i];
143 for (final parent in commit.parents) {
144 if (!result.containsKey(parent)) {
145 result[parent] = result[commit]! + 1;
146 queue.add(parent);
147 }
148 }
149 }
150 return result;
151 }
152
153 bool ancestor(Commit a, Commit b) => distances(b).containsKey(a);
154
155 Commit mergeBase(Commit a, Commit b) {
156 final da = distances(a), db = distances(b);
157 final common = da.keys.where(db.containsKey).toList();
158 final candidates = common.where((commit) => !common.any((other) =>
159 other != commit && ancestor(commit, other))).toList();
160 int rank(Commit commit) => da[commit]! > db[commit]!
161 ? da[commit]! : db[commit]!;
162 candidates.sort((x, y) {
163 final order = rank(x).compareTo(rank(y));
164 return order != 0 ? order : y.sequence.compareTo(x.sequence);
165 });
166 return candidates.first;
167 }
168
169 String summary(Commit commit) {
170 final refs = <String>[];
171 if (head == commit) refs.add(branch == null ? 'HEAD' : 'HEAD -> $branch');
172 for (final name in branches.keys.toList()..sort()) {
173 if (name != branch && branches[name] == commit) refs.add(name);
174 }
175 for (final name in tags.keys.toList()..sort()) {
176 if (tags[name] == commit) refs.add('tag: $name');
177 }
178 return '${commit.id}${refs.isEmpty ? '' : ' (${refs.join(', ')})'} ${commit.message}';
179 }
180
181 void changes(Tree a, Tree b) {
182 final changed = different(a, b);
183 if (changed.isEmpty) output.add('(no changes)');
184 for (final path in changed) {
185 if (!a.containsKey(path)) {
186 output.add('A $path: ${b[path]}');
187 } else if (!b.containsKey(path)) {
188 output.add('D $path: ${a[path]}');
189 } else {
190 output.add('M $path: ${a[path]} -> ${b[path]}');
191 }
192 }
193 }
194
195 void inProgress() {
196 if (operation != null) fail('${operation!.kind} in progress');
197 }
198 bool get dirty => !equal(index, head.tree) ||
199 index.keys.any((path) => working[path] != index[path]);
200 void uncommitted() {
201 if (dirty) fail('uncommitted changes');
202 }
203 void overwrite(Iterable<String> check) {
204 final h = head.tree;
205 for (final path in check.toSet().toList()..sort()) {
206 if (index[path] != h[path] || working[path] != h[path]) {
207 fail("local changes to '$path' would be overwritten");
208 }
209 }
210 }
211 void switchTree(Tree tree) {
212 final changed = different(head.tree, tree);
213 overwrite(changed);
214 for (final path in changed) {
215 match(index, tree, path);
216 match(working, tree, path);
217 }
218 }
219
220 MergeResult merge(Tree base, Tree ours, Tree theirs) {
221 final tree = <String, String>{};
222 final found = <String>[];
223 final contents = <String, String>{};
224 for (final path in paths([base, ours, theirs])) {
225 final b = base[path], o = ours[path], t = theirs[path];
226 if (t == b || o == t) {
227 match(tree, ours, path);
228 } else if (o == b) {
229 match(tree, theirs, path);
230 } else {
231 found.add(path);
232 match(tree, ours, path);
233 contents[path] = '<<<<<<< ${o ?? '(deleted)'} ======= ${t ?? '(deleted)'} >>>>>>>';
234 }
235 }
236 return MergeResult(tree, found, contents);
237 }
238
239 void apply(Tree base, Tree theirs, String kind, Commit target,
240 String message, List<Commit> parents, [Commit? mergeCommit]) {
241 final result = merge(base, head.tree, theirs);
242 final changed = different(head.tree, result.tree);
243 overwrite([...changed, ...result.conflicts]);
244 if (kind == 'merge') output.add('Merge base ${mergeCommit!.id}');
245 if (kind != 'merge' && result.conflicts.isEmpty && changed.isEmpty) {
246 output.add('nothing to commit');
247 return;
248 }
249 for (final path in changed) {
250 match(index, result.tree, path);
251 match(working, result.tree, path);
252 }
253 if (result.conflicts.isEmpty) {
254 record(parents, result.tree, message);
255 return;
256 }
257 for (final path in result.conflicts) {
258 working[path] = result.contents[path]!;
259 output.add('CONFLICT $path');
260 }
261 conflicts = result.conflicts.toSet();
262 operation = Operation(kind, target);
263 output.add(kind == 'merge'
264 ? 'Automatic merge failed; fix conflicts and then commit the result.'
265 : kind == 'revert' ? 'error: could not revert ${target.id}'
266 : 'error: could not apply ${target.id}');
267 }

269 bool glob(String pattern, String path) {
270 var row = List<bool>.filled(path.length + 1, false);
271 row[0] = true;
272 for (final code in pattern.codeUnits) {
273 final next = List<bool>.filled(path.length + 1, false);
274 if (code == 42) next[0] = row[0];
275 for (var i = 1; i <= path.length; i++) {
276 next[i] = code == 42 ? row[i] || next[i - 1]
277 : row[i - 1] && (code == 63 || code == path.codeUnitAt(i - 1));
278 }
279 row = next;
280 }
281 return row[path.length];
282 }
283 bool ignored(String path) => !index.containsKey(path) &&
284 !conflicts.contains(path) && patterns.any((pattern) => glob(pattern, path));
285 Never pathspec(String path) => fail("pathspec '$path' did not match any files");
286 String existingPath(Tree tree, String path, String rev) {
287 if (!tree.containsKey(path)) fail("path '$path' does not exist in '$rev'");
288 return tree[path]!;
289 }
290 Set<Commit> reachable(Iterable<Commit> roots) {
291 final result = <Commit>{};
292 for (final commit in roots) {
293 result.addAll(distances(commit).keys);
294 }
295 return result;
296 }

298 void execute(String line) {
299 final words = line.split(' '), command = words.first;
300 final arg = words.length > 1 ? words[1] : '';
301 switch (command) {
302 case 'WRITE':
303 working[arg] = line.substring(line.indexOf(' ', 6) + 1);
304 case 'REMOVE':
305 if (!working.containsKey(arg)) fail("no such file '$arg'");
306 working.remove(arg);
307 case 'ADD':
308 if (arg == 'ALL') {
309 final selected = paths([working, index]).where((path) => !ignored(path)).toList();
310 for (final path in selected) {
311 match(index, working, path);
312 }
313 conflicts.clear();
314 } else {
315 if (working.containsKey(arg) && ignored(arg)) fail("path '$arg' is ignored");
316 if (!working.containsKey(arg) && !index.containsKey(arg) && !conflicts.contains(arg)) pathspec(arg);
317 match(index, working, arg);
318 conflicts.remove(arg);
319 }
320 case 'RM':
321 if (!index.containsKey(arg)) pathspec(arg);
322 index.remove(arg);
323 working.remove(arg);
324 conflicts.remove(arg);
325 case 'UNSTAGE':
326 if (!index.containsKey(arg) && !head.tree.containsKey(arg)) pathspec(arg);
327 match(index, head.tree, arg);
328 case 'RESTORE':
329 if (words.length == 2) {
330 if (!index.containsKey(arg)) pathspec(arg);
331 working[arg] = index[arg]!;
332 } else {
333 final tree = resolve(words[2]).tree;
334 final value = existingPath(tree, arg, words[2]);
335 index[arg] = value;
336 working[arg] = value;
337 conflicts.remove(arg);
338 }
339 case 'IGNORE':
340 patterns.add(arg);
341 case 'COMMIT':
342 if (conflicts.isNotEmpty) fail('unresolved conflicts');
343 final parents = operation?.kind == 'merge' ? [head, operation!.target] : [head];
344 if (parents.length == 1 && equal(index, head.tree)) {
345 output.add('nothing to commit');
346 } else {
347 record(parents, index, line.substring(7));
348 operation = null;
349 }
350 case 'AMEND':
351 inProgress();
352 if (head.parents.isEmpty) fail('cannot amend the root commit');
353 record(head.parents, index, line.substring(6));
354 case 'STATUS':
355 output.add(branch == null ? 'HEAD detached at ${head.id}' : 'On branch $branch');
356 if (operation != null) {
357 final verb = operation!.kind == 'merge' ? 'Merging'
358 : operation!.kind == 'revert' ? 'Reverting' : 'Cherry-picking';
359 output.add('$verb ${operation!.target.id}');
360 }
361 final before = output.length;
362 final all = {...paths([head.tree, index, working]), ...conflicts}.toList()..sort();
363 for (final path in all) {
364 if (conflicts.contains(path)) {
365 output.add('UU $path');
366 } else if (index.containsKey(path)) {
367 final x = !head.tree.containsKey(path) ? 'A'
368 : head.tree[path] != index[path] ? 'M' : '.';
369 final y = !working.containsKey(path) ? 'D'
370 : working[path] != index[path] ? 'M' : '.';
371 if (x != '.' || y != '.') output.add('$x$y $path');
372 } else {
373 if (head.tree.containsKey(path)) output.add('D. $path');
374 if (working.containsKey(path) && !ignored(path)) output.add('?? $path');
375 }
376 }
377 if (output.length == before) output.add('nothing to commit, working tree clean');
378 case 'LS':
379 final tree = arg.isEmpty ? working : resolve(arg).tree;
380 if (tree.isEmpty) output.add('(empty)');
381 for (final path in tree.keys.toList()..sort()) {
382 output.add('$path ${tree[path]}');
383 }
384 case 'CAT':
385 if (words.length == 3) {
386 output.add(existingPath(resolve(words[2]).tree, arg, words[2]));
387 } else {
388 if (!working.containsKey(arg)) fail("no such file '$arg'");
389 output.add(working[arg]!);
390 }
391 case 'DIFF':
392 if (arg == 'STAGED') {
393 changes(head.tree, index);
394 } else if (arg.isNotEmpty) {
395 final a = resolve(arg), b = resolve(words[2]);
396 changes(a.tree, b.tree);
397 } else {
398 changes(index, Map.fromEntries(working.entries.where((entry) => index.containsKey(entry.key))));
399 }
400 case 'LOG':
401 var current = head, i = 0;
402 while (true) {
403 output.add(summary(current));
404 i++;
405 if (current.parents.isEmpty || (arg.isNotEmpty && i == int.parse(arg))) break;
406 current = current.parents.first;
407 }
408 case 'SHOW':
409 final commit = resolve(arg.isEmpty ? 'HEAD' : arg);
410 output.add('commit ${summary(commit)}');
411 output.add('parents: ${commit.parents.isEmpty ? 'none' : commit.parents.map((parent) => parent.id).join(' ')}');
412 changes(commit.parents.isEmpty ? {} : commit.parents.first.tree, commit.tree);
413 case 'BLAME':
414 final value = existingPath(head.tree, arg, 'HEAD');
415 var commit = head;
416 while (commit.parents.isNotEmpty && commit.parents.first.tree[arg] == value) {
417 commit = commit.parents.first;
418 }
419 output.add('${commit.id} (${commit.message}) $value');
420 case 'BRANCH':
421 case 'TAG':
422 final refs = command == 'BRANCH' ? branches : tags;
423 final noun = command == 'BRANCH' ? 'branch' : 'tag';
424 if (arg == '-d') {
425 final name = words[2], commit = refs[words[2]];
426 if (commit == null) fail("no such $noun '$name'");
427 if (command == 'BRANCH') {
428 if (branch == name) fail("cannot delete current branch '$name'");
429 if (!ancestor(commit, head)) fail("branch '$name' is not fully merged");
430 }
431 refs.remove(name);
432 output.add('Deleted $noun $name (was ${commit.id})');
433 } else {
434 if (refs.containsKey(arg)) fail("$noun '$arg' already exists");
435 final commit = resolve(words.length > 2 ? words[2] : 'HEAD');
436 refs[arg] = commit;
437 output.add('Created $noun $arg at ${commit.id}');
438 }
439 case 'BRANCHES':
440 if (branch == null) output.add('* (HEAD detached at ${head.id})');
441 for (final name in branches.keys.toList()..sort()) {
442 final commit = branches[name]!;
443 output.add('${branch == name ? '*' : ' '} $name ${commit.id} ${commit.message}');
444 }
445 case 'TAGS':
446 if (tags.isEmpty) output.add('(no tags)');
447 for (final name in tags.keys.toList()..sort()) {
448 final commit = tags[name]!;
449 output.add('$name ${commit.id} ${commit.message}');
450 }
451 case 'SWITCH':
452 final commit = branches[arg];
453 if (commit == null) fail("no such branch '$arg'");
454 if (branch == arg) {
455 output.add("Already on '$arg'");
456 } else {
457 inProgress();
458 switchTree(commit.tree);
459 branch = arg;
460 output.add("Switched to branch '$arg'");
461 }
462 case 'DETACH':
463 final commit = resolve(arg);
464 inProgress();
465 switchTree(commit.tree);
466 branch = null;
467 detached = commit;
468 output.add('HEAD is now at ${commit.id} ${commit.message}');
469 case 'MERGE':
470 final target = resolve(arg);
471 inProgress();
472 final old = head;
473 if (ancestor(target, old)) {
474 output.add('Already up to date');
475 } else if (ancestor(old, target)) {
476 switchTree(target.tree);
477 move(target);
478 output.add('Fast-forward ${old.id}..${target.id}');
479 } else {
480 uncommitted();
481 final base = mergeBase(old, target);
482 apply(base.tree, target.tree, 'merge', target, 'Merge $arg', [old, target], base);
483 }
484 case 'MERGEBASE':
485 final a = resolve(arg), b = resolve(words[2]);
486 output.add(mergeBase(a, b).id);
487 case 'REBASE':
488 final onto = resolve(arg);
489 inProgress();
490 if (ancestor(onto, head)) {
491 output.add('Already up to date');
492 } else {
493 uncommitted();
494 final list = <Commit>[];
495 var current = head;
496 while (!ancestor(current, onto)) {
497 list.add(current);
498 current = current.parents.first;
499 }
500 final ordered = list.reversed.toList();
501 final picks = <Commit, Tree>{};
502 var tree = onto.tree;
503 for (final commit in ordered) {
504 final result = merge(commit.parents.first.tree, tree, commit.tree);
505 if (result.conflicts.isNotEmpty) {
506 fail("could not rebase ${commit.id}: conflict in '${result.conflicts.first}'");
507 }
508 if (!equal(result.tree, tree)) {
509 picks[commit] = result.tree;
510 tree = result.tree;
511 }
512 }
513 switchTree(tree);
514 var target = onto;
515 for (final commit in ordered) {
516 if (picks.containsKey(commit)) {
517 final next = create([target], picks[commit]!, commit.message);
518 output.add('pick ${commit.id} -> ${next.id} ${commit.message}');
519 target = next;
520 } else {
521 output.add('skip ${commit.id} ${commit.message}');
522 }
523 }
524 move(target);
525 output.add('HEAD is now at ${target.id} ${target.message}');
526 }
527 case 'RESET':
528 final target = resolve(words[2]), tree = target.tree;
529 if (arg == 'HARD') {
530 final all = {...paths([index, tree]), ...conflicts};
531 for (final path in all) {
532 match(working, tree, path);
533 }
534 }
535 if (arg != 'SOFT') index = Map.of(tree);
536 move(target);
537 conflicts.clear();
538 operation = null;
539 output.add('HEAD is now at ${target.id} ${target.message}');
540 case 'REVERT':
541 case 'CHERRYPICK':
542 final target = resolve(arg);
543 inProgress();
544 uncommitted();
545 final parentTree = target.parents.isEmpty ? <String, String>{} : target.parents.first.tree;
546 if (command == 'REVERT') {
547 apply(target.tree, parentTree, 'revert', target, 'Revert "${target.message}"', [head]);
548 } else {
549 apply(parentTree, target.tree, 'cherry-pick', target, target.message, [head]);
550 }
551 case 'STASH':
552 if (arg == 'LIST') {
553 if (stashes.isEmpty) output.add('(no stash entries)');
554 for (var i = 0; i < stashes.length; i++) {
555 output.add('stash@{$i}: ${stashes[i].label}');
556 }
557 } else if (arg == 'DROP') {
558 if (stashes.isEmpty) fail('no stash entries');
559 output.add('Dropped stash@{0}: ${stashes.removeAt(0).label}');
560 } else if (arg == 'POP') {
561 if (stashes.isEmpty) fail('no stash entries');
562 inProgress();
563 uncommitted();
564 final stash = stashes.first;
565 final result = merge(stash.base.tree, head.tree, stash.tree);
566 if (result.conflicts.isNotEmpty) fail("stash conflict in '${result.conflicts.first}'");
567 final changed = different(head.tree, result.tree);
568 overwrite(changed);
569 for (final path in changed) {
570 match(working, result.tree, path);
571 }
572 stashes.removeAt(0);
573 output.add('Restored stash@{0}: ${stash.label}');
574 changes(head.tree, result.tree);
575 } else {
576 inProgress();
577 if (!dirty) {
578 output.add('No local changes to save');
579 } else {
580 final tree = <String, String>{}, h = head;
581 for (final path in paths([index, h.tree])) {
582 if (working.containsKey(path)) tree[path] = working[path]!;
583 match(working, h.tree, path);
584 }
585 index = Map.of(h.tree);
586 final label = 'WIP on ${branch ?? '(no branch)'}: ${h.id} ${h.message}';
587 stashes.insert(0, Stash(label, h, tree));
588 output.add('Saved stash@{0}: $label');
589 }
590 }
591 case 'GRAPH':
592 final all = reachable([head, ...branches.values, ...tags.values]).toList()
593 ..sort((a, b) => b.sequence.compareTo(a.sequence));
594 var lanes = <Commit>[];
595 for (final commit in all) {
596 if (!lanes.contains(commit)) lanes.add(commit);
597 final k = lanes.indexOf(commit);
598 final symbols = <String>[];
599 for (var i = 0; i < lanes.length; i++) {
600 symbols.add(i == k ? '*' : lanes[i] == commit ? '/' : '|');
601 }
602 output.add('${symbols.join(' ')} ${summary(commit)}');
603 lanes = [for (var i = 0; i < lanes.length; i++) if (lanes[i] != commit || i == k) lanes[i]];
604 if (commit.parents.isEmpty) {
605 lanes.removeAt(k);
606 } else {
607 lanes[k] = commit.parents.first;
608 if (commit.parents.length == 2 && !lanes.contains(commit.parents[1])) {
609 lanes.insert(k + 1, commit.parents[1]);
610 }
611 }
612 }
613 case 'DESCRIBE':
614 final target = resolve(arg.isEmpty ? 'HEAD' : arg), distance = distances(target);
615 final names = tags.keys.where((name) => distance.containsKey(tags[name])).toList();
616 names.sort((a, b) {
617 final order = distance[tags[a]]!.compareTo(distance[tags[b]]!);
618 return order == 0 ? a.compareTo(b) : order;
619 });
620 if (names.isEmpty) fail("no tag can describe '${target.id}'");
621 final name = names.first, d = distance[tags[name]]!;
622 output.add(d == 0 ? name : '$name-$d-g${target.id}');
623 case 'GC':
624 final roots = [head, ...branches.values, ...tags.values, ...stashes.map((stash) => stash.base)];
625 if (operation != null) roots.add(operation!.target);
626 final keep = reachable(roots);
627 final before = commits.length;
628 commits.removeWhere((id, commit) => !keep.contains(commit));
629 output.add('Removed unreachable commits: ${before - commits.length}');
630 }
631 }
632}

634void main() {
635 final vcs = Vcs();
636 String? line;
637 while ((line = stdin.readLineSync()) != null && line != 'EXIT') {
638 try {
639 vcs.execute(line!);
640 } on VcsError catch (error) {
641 vcs.output.add(error.message);
642 }
643 }
644 if (vcs.output.isNotEmpty) stdout.write('${vcs.output.join('\n')}\n');
645}
