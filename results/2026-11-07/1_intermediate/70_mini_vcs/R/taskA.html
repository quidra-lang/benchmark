40create_commit <- function(parents, tree, message) {
41 lines <- c(paste("commit", state$sequence),
42 if (length(parents)) paste("parent", parents),
43 vapply(paths(tree), function(p) paste("file", p, tree[[p]]), character(1)),
44 paste("message", message))
45 text <- paste0(paste(lines, collapse = "\n"), "\n")
46 hash <- 0
47 for (code in as.integer(charToRaw(text))) hash <- (hash * 131 + code) %% 2147483647
48 id <- sprintf("%08x", as.integer(hash))
49 state$commits[[id]] <- list(id = id, seq = state$sequence, parents = parents,
50 tree = tree, message = message)
51 state$sequence <- state$sequence + 1
52 id
53}
54record_commit <- function(parents, tree, message) {
55 old <- commit_at(parents[[1]])$tree
56 changed <- differences(old, tree)
57 added <- sum(vapply(changed, function(p) is.null(old[[p]]), logical(1)))
58 deleted <- sum(vapply(changed, function(p) is.null(tree[[p]]), logical(1)))
59 modified <- length(changed) - added - deleted
60 id <- create_commit(parents, tree, message)
61 move_head(id)
62 location <- if (is.null(state$branch)) "detached HEAD" else state$branch
63 say("[", location, " ", id, "] ", message, " (", added, " added, ", modified,
64 " modified, ", deleted, " deleted)")

66resolve <- function(revision) {
67 parts <- strsplit(revision, "~", fixed = TRUE)[[1]]
68 anchor <- parts[[1]]
69 id <- if (anchor == "HEAD") head_id() else state$branches[[anchor]]
70 if (is.null(id)) id <- state$tags[[anchor]]
71 if (is.null(id) && !is.null(commit_at(anchor))) id <- anchor
72 steps <- if (length(parts) == 1L) 0L else as.integer(parts[[2]])
73 if (steps > 0L) {
74 for (i in seq_len(steps)) {
75 if (!is.null(id)) {
76 parents <- commit_at(id)$parents
77 id <- if (length(parents)) parents[[1]] else NULL
78 }
79 }
80 }
81 if (is.null(id)) fail(paste0("unknown revision '", revision, "'"))
82 id

84distances <- function(id) {
85 result <- list()
86 result[[id]] <- 0L
87 queue <- id
88 at <- 1L
89 while (at <= length(queue)) {
90 current <- queue[[at]]
91 at <- at + 1L
92 for (parent in commit_at(current)$parents) {
93 if (is.null(result[[parent]])) {
94 result[[parent]] <- result[[current]] + 1L
95 queue <- c(queue, parent)
96 }
97 }
98 }
99 result

101ancestor <- function(a, b) !is.null(distances(b)[[a]])
102merge_base <- function(a, b) {
103 da <- distances(a)
104 db <- distances(b)
105 common <- intersect(names(da), names(db))
106 candidates <- common[!vapply(common, function(x) {
107 any(vapply(setdiff(common, x), function(y) ancestor(x, y), logical(1)))
108 }, logical(1))]
109 scores <- vapply(candidates, function(x) max(da[[x]], db[[x]]), integer(1))
110 sequences <- vapply(candidates, function(x) commit_at(x)$seq, numeric(1))
111 candidates[[order(scores, -sequences)[[1]]]]
112}

164merge_trees <- function(base, ours, theirs) {
165 result <- list()
166 conflicts <- character()
167 markers <- list()
168 for (p in paths(base, ours, theirs)) {
169 b <- base[[p]]
170 o <- ours[[p]]
171 t <- theirs[[p]]
172 if (identical(t, b) || identical(o, t)) {
173 result <- match_path(result, ours, p)
174 } else if (identical(o, b)) {
175 result <- match_path(result, theirs, p)
176 } else {
177 conflicts <- c(conflicts, p)
178 result <- match_path(result, ours, p)
179 markers[[p]] <- paste0("<<<<<<< ", if (is.null(o)) "(deleted)" else o,
180 " ======= ", if (is.null(t)) "(deleted)" else t, " >>>>>>>")
181 }
182 }
183 list(tree = result, conflicts = conflicts, markers = markers)
184}
185apply_change <- function(base, theirs, kind, target, message, parents, merge_base_id = NULL) {
186 merged <- merge_trees(base, head_commit()$tree, theirs)
187 changed <- differences(head_commit()$tree, merged$tree)
188 check_overwrite(c(changed, merged$conflicts))
189 if (kind == "merge") say("Merge base ", merge_base_id)
190 if (kind != "merge" && !length(merged$conflicts) && equal_trees(merged$tree, head_commit()$tree)) {
191 say("nothing to commit")
192 return(invisible(NULL))
193 }
194 for (p in changed) {
195 state$index <- match_path(state$index, merged$tree, p)
196 state$work <- match_path(state$work, merged$tree, p)
197 }
198 if (!length(merged$conflicts)) {
199 record_commit(parents, merged$tree, message)
200 } else {
201 for (p in merged$conflicts) {
202 state$work[[p]] <- merged$markers[[p]]
203 say("CONFLICT ", p)
204 }
205 state$conflicts <- merged$conflicts
206 state$operation <- list(kind = kind, target = target)
207 say(switch(kind, merge = "Automatic merge failed; fix conflicts and then commit the result.",
208 revert = paste("error: could not revert", target),
209 paste("error: could not apply", target)))
210 }

222 if (!is.null(state$operation)) {

233 if (x != "." || y != ".") lines <- c(lines, paste0(x, y, " ", p))

242rebase <- function(target) {
243 check_progress()
244 if (ancestor(target, head_id())) {
245 say("Already up to date")
246 return(invisible(NULL))
247 }
248 check_uncommitted()
249 history <- character()
250 current <- head_id()
251 while (!ancestor(current, target)) {
252 history <- c(history, current)
253 current <- commit_at(current)$parents[[1]]
254 }
255 history <- rev(history)
256 tree <- commit_at(target)$tree
257 actions <- list()
258 for (id in history) {
259 old <- commit_at(id)
260 merged <- merge_trees(commit_at(old$parents[[1]])$tree, tree, old$tree)
261 if (length(merged$conflicts)) fail(paste0("could not rebase ", id, ": conflict in '", merged$conflicts[[1]], "'"))
262 if (equal_trees(merged$tree, tree)) {
263 actions[[length(actions) + 1L]] <- list(id = id, pick = FALSE)
264 } else {
265 tree <- merged$tree
266 actions[[length(actions) + 1L]] <- list(id = id, pick = TRUE, tree = tree)
267 }
268 }
269 switch_tree(tree)
270 for (action in actions) {
271 old <- commit_at(action$id)
272 if (action$pick) {
273 fresh <- create_commit(target, action$tree, old$message)
274 say("pick ", old$id, " -> ", fresh, " ", old$message)
275 target <- fresh
276 } else {
277 say("skip ", old$id, " ", old$message)
278 }
279 }
280 move_head(target)
281 head_report()
282}
283stash_command <- function(action) {
284 if (identical(action, "LIST")) {
285 if (!length(state$stash)) say("(no stash entries)")
286 for (i in seq_along(state$stash)) say("stash@{", i - 1L, "}: ", state$stash[[i]]$label)
287 } else if (identical(action, "DROP")) {
288 if (!length(state$stash)) fail("no stash entries")
289 entry <- state$stash[[1]]
290 state$stash <- state$stash[-1]
291 say("Dropped stash@{0}: ", entry$label)
292 } else if (identical(action, "POP")) {
293 if (!length(state$stash)) fail("no stash entries")
294 check_progress()
295 check_uncommitted()
296 entry <- state$stash[[1]]
297 merged <- merge_trees(commit_at(entry$base)$tree, head_commit()$tree, entry$tree)
298 if (length(merged$conflicts)) fail(paste0("stash conflict in '", merged$conflicts[[1]], "'"))
299 changed <- differences(head_commit()$tree, merged$tree)
300 check_overwrite(changed)
301 for (p in changed) state$work <- match_path(state$work, merged$tree, p)
302 state$stash <- state$stash[-1]
303 say("Restored stash@{0}: ", entry$label)
304 changes(head_commit()$tree, merged$tree)
305 } else {
306 check_progress()
307 if (!uncommitted()) {
308 say("No local changes to save")
309 return(invisible(NULL))
310 }
311 selected <- paths(state$index, head_commit()$tree)
312 saved <- list()
313 for (p in selected) {
314 saved <- match_path(saved, state$work, p)
315 state$work <- match_path(state$work, head_commit()$tree, p)
316 }
317 state$index <- head_commit()$tree
318 location <- if (is.null(state$branch)) "(no branch)" else state$branch
319 label <- paste0("WIP on ", location, ": ", head_id(), " ", head_commit()$message)
320 state$stash <- c(list(list(label = label, base = head_id(), tree = saved)), state$stash)
321 say("Saved stash@{0}: ", label)
322 }

329graph <- function() {
330 live <- reachable(c(head_id(), unlist(state$branches), unlist(state$tags)))
331 sequences <- vapply(live, function(id) commit_at(id)$seq, numeric(1))
332 lanes <- character()
333 for (id in live[order(-sequences)]) {
334 if (!(id %in% lanes)) lanes <- c(lanes, id)
335 first <- match(id, lanes)
336 symbols <- ifelse(lanes == id, "/", "|")
337 symbols[[first]] <- "*"
338 say(paste(symbols, collapse = " "), " ", summary_line(id))
339 lanes <- lanes[lanes != id | seq_along(lanes) == first]
340 parents <- commit_at(id)$parents
341 if (!length(parents)) {
342 lanes <- lanes[-first]
343 } else {
344 lanes[[first]] <- parents[[1]]
345 if (length(parents) == 2L && !(parents[[2]] %in% lanes)) lanes <- append(lanes, parents[[2]], after = first)
346 }
347 }

349execute <- function(line) {
350 words <- strsplit(line, " ", fixed = TRUE)[[1]]
351 command <- words[[1]]
352 args <- words[-1]
353 path <- if (length(args)) args[[1]] else NULL
354 rest <- if (length(args)) substring(line, nchar(command) + 2L) else ""
355 h <- head_commit()$tree
356 switch(command,
357 WRITE = {
358 state$work[[path]] <- substring(rest, nchar(path) + 2L)
359 },
360 REMOVE = {
361 if (is.null(state$work[[path]])) fail(paste0("no such file '", path, "'"))
362 state$work[[path]] <- NULL
363 },
364 ADD = {
365 if (path == "ALL") {
366 selected <- paths(state$work, state$index)
367 selected <- selected[!vapply(selected, ignored, logical(1))]
368 for (p in selected) state$index <- match_path(state$index, state$work, p)
369 state$conflicts <- character()
370 } else {
371 if (!is.null(state$work[[path]]) && ignored(path)) fail(paste0("path '", path, "' is ignored"))
372 if (is.null(state$work[[path]]) && is.null(state$index[[path]]) && !(path %in% state$conflicts)) pathspec(path)
373 state$index <- match_path(state$index, state$work, path)
374 state$conflicts <- setdiff(state$conflicts, path)
375 }
376 },
377 RM = {
378 if (is.null(state$index[[path]])) pathspec(path)
379 state$index[[path]] <- NULL
380 state$work[[path]] <- NULL
381 state$conflicts <- setdiff(state$conflicts, path)
382 },
383 UNSTAGE = {
384 if (is.null(state$index[[path]]) && is.null(h[[path]])) pathspec(path)
385 state$index <- match_path(state$index, h, path)
386 },
387 RESTORE = {
388 if (length(args) == 1L) {
389 if (is.null(state$index[[path]])) pathspec(path)
390 state$work[[path]] <- state$index[[path]]
391 } else {
392 tree <- commit_at(resolve(args[[2]]))$tree
393 if (is.null(tree[[path]])) missing_path(path, args[[2]])
394 state$index[[path]] <- tree[[path]]
395 state$work[[path]] <- tree[[path]]
396 state$conflicts <- setdiff(state$conflicts, path)
397 }
398 },
399 IGNORE = {
400 chars <- strsplit(path, "", fixed = TRUE)[[1]]
401 pieces <- vapply(chars, function(ch) {
402 if (ch == "*") ".*" else if (ch == "?") "." else if (ch == ".") "\\." else ch
403 }, character(1))
404 state$ignores <- c(state$ignores, paste0("\\A", paste0(pieces, collapse = ""), "\\z"))
405 },
406 COMMIT = {
407 if (length(state$conflicts)) fail("unresolved conflicts")
408 if (!is.null(state$operation) && state$operation$kind == "merge") {
409 record_commit(c(head_id(), state$operation$target), state$index, rest)
410 state$operation <- NULL
411 } else if (equal_trees(state$index, h)) {
412 say("nothing to commit")
413 } else {
414 record_commit(head_id(), state$index, rest)
415 state$operation <- NULL
416 }
417 },
418 AMEND = {
419 check_progress()
420 if (!length(head_commit()$parents)) fail("cannot amend the root commit")
421 record_commit(head_commit()$parents, state$index, rest)
422 },
423 STATUS = status(),
424 LS = {
425 tree <- if (length(args)) commit_at(resolve(path))$tree else state$work
426 if (!length(tree)) say("(empty)")
427 for (p in paths(tree)) say(p, " ", tree[[p]])
428 },
429 CAT = {
430 revision <- if (length(args) == 2L) args[[2]] else NULL
431 tree <- if (is.null(revision)) state$work else commit_at(resolve(revision))$tree
432 if (is.null(tree[[path]])) {
433 if (!is.null(revision)) missing_path(path, revision)
434 fail(paste0("no such file '", path, "'"))
435 }
436 say(tree[[path]])
437 },
438 DIFF = {
439 if (!length(args)) {
440 target <- state$work[intersect(names(state$work), names(state$index))]
441 changes(state$index, target)
442 } else if (path == "STAGED") {
443 changes(h, state$index)
444 } else {
445 a <- resolve(path)
446 b <- resolve(args[[2]])
447 changes(commit_at(a)$tree, commit_at(b)$tree)
448 }
449 },
450 LOG = {
451 current <- head_id()
452 count <- 0L
453 repeat {
454 say(summary_line(current))
455 count <- count + 1L
456 parents <- commit_at(current)$parents
457 if (!length(parents) || (!is.null(path) && count == as.integer(path))) break
458 current <- parents[[1]]
459 }
460 },
461 SHOW = {
462 id <- resolve(if (is.null(path)) "HEAD" else path)
463 commit <- commit_at(id)
464 say("commit ", summary_line(id))
465 say("parents: ", if (length(commit$parents)) paste(commit$parents, collapse = " ") else "none")
466 changes(if (length(commit$parents)) commit_at(commit$parents[[1]])$tree else list(), commit$tree)
467 },
468 BLAME = {
469 if (is.null(h[[path]])) missing_path(path, "HEAD")
470 value <- h[[path]]
471 id <- head_id()
472 repeat {
473 parents <- commit_at(id)$parents
474 if (!length(parents) || !identical(commit_at(parents[[1]])$tree[[path]], value)) break
475 id <- parents[[1]]
476 }
477 say(id, " (", commit_at(id)$message, ") ", value)
478 },
479 BRANCH = reference_command("branch", args),
480 TAG = reference_command("tag", args),
481 BRANCHES = {
482 if (is.null(state$branch)) say("* (HEAD detached at ", head_id(), ")")
483 for (name in paths(state$branches)) {
484 id <- state$branches[[name]]
485 mark <- if (!is.null(state$branch) && name == state$branch) "*" else " "
486 say(mark, " ", name, " ", id, " ", commit_at(id)$message)
487 }
488 },
489 TAGS = {
490 if (!length(state$tags)) say("(no tags)")
491 for (name in paths(state$tags)) {
492 id <- state$tags[[name]]
493 say(name, " ", id, " ", commit_at(id)$message)
494 }
495 },
496 SWITCH = {
497 if (is.null(state$branches[[path]])) fail(paste0("no such branch '", path, "'"))
498 if (identical(state$branch, path)) {
499 say("Already on '", path, "'")
500 } else {
501 check_progress()
502 switch_tree(commit_at(state$branches[[path]])$tree)
503 state$branch <- path
504 say("Switched to branch '", path, "'")
505 }
506 },
507 DETACH = {
508 id <- resolve(path)
509 check_progress()
510 switch_tree(commit_at(id)$tree)
511 state$branch <- NULL
512 state$detached <- id
513 head_report()
514 },
515 MERGE = {
516 target <- resolve(path)
517 check_progress()
518 old <- head_id()
519 if (ancestor(target, old)) {
520 say("Already up to date")
521 } else if (ancestor(old, target)) {
522 switch_tree(commit_at(target)$tree)
523 move_head(target)
524 say("Fast-forward ", old, "..", target)
525 } else {
526 check_uncommitted()
527 base <- merge_base(old, target)
528 apply_change(commit_at(base)$tree, commit_at(target)$tree, "merge", target,
529 paste("Merge", path), c(old, target), base)
530 }
531 },
532 MERGEBASE = {
533 a <- resolve(path)
534 b <- resolve(args[[2]])
535 say(merge_base(a, b))
536 },
537 REBASE = rebase(resolve(path)),
538 RESET = {
539 id <- resolve(args[[2]])
540 tree <- commit_at(id)$tree
541 if (path == "HARD") {
542 for (p in union(paths(state$index, tree), state$conflicts)) state$work <- match_path(state$work, tree, p)
543 }
544 if (path != "SOFT") state$index <- tree
545 move_head(id)
546 state$conflicts <- character()
547 state$operation <- NULL
548 head_report()
549 },
550 REVERT = revert_or_pick(path, TRUE),
551 CHERRYPICK = revert_or_pick(path, FALSE),
552 STASH = stash_command(path),
553 GRAPH = graph(),
554 DESCRIBE = {
555 id <- resolve(if (is.null(path)) "HEAD" else path)
556 distance <- distances(id)
557 candidates <- paths(state$tags)
558 candidates <- candidates[vapply(candidates, function(name) !is.null(distance[[state$tags[[name]]]]), logical(1))]
559 if (!length(candidates)) fail(paste0("no tag can describe '", id, "'"))
560 scores <- vapply(candidates, function(name) distance[[state$tags[[name]]]], integer(1))
561 name <- candidates[[which.min(scores)]]
562 d <- min(scores)
563 say(if (d == 0L) name else paste0(name, "-", d, "-g", id))
564 },
565 GC = {
566 roots <- c(head_id(), unlist(state$branches), unlist(state$tags),
567 vapply(state$stash, function(entry) entry$base, character(1)))
568 if (!is.null(state$operation)) roots <- c(roots, state$operation$target)
569 live <- reachable(roots)
570 old_size <- length(state$commits)
571 state$commits <- state$commits[intersect(names(state$commits), live)]
572 say("Removed unreachable commits: ", old_size - length(state$commits))
573 }
574 )

576reference_command <- function(kind, args) {
577 refs <- if (kind == "branch") state$branches else state$tags
578 if (args[[1]] == "-d") {
579 name <- args[[2]]
580 if (is.null(refs[[name]])) fail(paste0("no such ", kind, " '", name, "'"))
581 if (kind == "branch") {
582 if (identical(state$branch, name)) fail(paste0("cannot delete current branch '", name, "'"))
583 if (!ancestor(refs[[name]], head_id())) fail(paste0("branch '", name, "' is not fully merged"))
584 }
585 id <- refs[[name]]
586 refs[[name]] <- NULL
587 say("Deleted ", kind, " ", name, " (was ", id, ")")
588 } else {
589 name <- args[[1]]
590 if (!is.null(refs[[name]])) fail(paste0(kind, " '", name, "' already exists"))
591 id <- resolve(if (length(args) == 2L) args[[2]] else "HEAD")
592 refs[[name]] <- id
593 say("Created ", kind, " ", name, " at ", id)
594 }
595 if (kind == "branch") state$branches <- refs else state$tags <- refs
596}
597revert_or_pick <- function(revision, revert) {
598 id <- resolve(revision)
599 check_progress()
600 check_uncommitted()
601 commit <- commit_at(id)
602 parent_tree <- if (length(commit$parents)) commit_at(commit$parents[[1]])$tree else list()
603 if (revert) {
604 apply_change(commit$tree, parent_tree, "revert", id, paste0('Revert "', commit$message, '"'), head_id())
605 } else {
606 apply_change(parent_tree, commit$tree, "cherry-pick", id, commit$message, head_id())
607 }
608}
