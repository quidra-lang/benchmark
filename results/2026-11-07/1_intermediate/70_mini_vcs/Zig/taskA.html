22fn bad(gpa: Allocator, comptime pattern: Text, args: anytype) Error {
23 failure = std.fmt.allocPrint(gpa, pattern, args) catch return error.OutOfMemory;

27const Builder = struct {
28 gpa: Allocator,
29 bytes: std.ArrayList(u8) = .empty,
30 fn add(self: *Builder, text: Text) Allocator.Error!void {
31 try self.bytes.appendSlice(self.gpa, text);
32 }
33 fn print(self: *Builder, comptime pattern: Text, args: anytype) Allocator.Error!void {
34 try self.add(try std.fmt.allocPrint(self.gpa, pattern, args));
35 }
36 fn line(self: *Builder, comptime pattern: Text, args: anytype) Allocator.Error!void {
37 try self.print(pattern ++ "\n", args);
38 }
39 fn result(self: *const Builder) Text {
40 return self.bytes.items;
41 }

44fn split(gpa: Allocator, text: Text, separator: u8, limit: usize) Allocator.Error![]Text {
45 var result: std.ArrayList(Text) = .empty;
46 var start: usize = 0;
47 for (text, 0..) |c, i| {
48 if (c == separator and (limit == 0 or result.items.len + 1 < limit)) {
49 try result.append(gpa, text[start..i]);
50 start = i + 1;
51 }
52 }
53 try result.append(gpa, text[start..]);
54 return result.items;
55}

57fn join(gpa: Allocator, parts: []const Text, separator: Text) Allocator.Error!Text {
58 var out: Builder = .{ .gpa = gpa };
59 for (parts, 0..) |part, i| {
60 if (i > 0) try out.add(separator);
61 try out.add(part);
62 }
63 return out.result();
64}

66fn textLess(_: void, a: Text, b: Text) bool {
67 return std.mem.order(u8, a, b) == .lt;

70fn keys(gpa: Allocator, map: anytype) Allocator.Error![]Text {
71 var result: std.ArrayList(Text) = .empty;
72 var iterator = map.keyIterator();
73 while (iterator.next()) |key| try result.append(gpa, key.*);

78fn paths(gpa: Allocator, trees: []const Tree) Allocator.Error![]Text {
79 var all: std.StringHashMapUnmanaged(void) = .empty;
80 for (trees) |tree| {
81 var iterator = tree.keyIterator();
82 while (iterator.next()) |key| try all.put(gpa, key.*, {});
83 }
84 return keys(gpa, all);
85}

87fn clone(gpa: Allocator, tree: Tree) Allocator.Error!Tree {
88 var result: Tree = .empty;
89 var iterator = tree.iterator();
90 while (iterator.next()) |entry| try result.put(gpa, entry.key_ptr.*, entry.value_ptr.*);
91 return result;

94fn agree(a: Tree, b: Tree, path: Text) bool {
95 const left = a.get(path);
96 const right = b.get(path);
97 if (left == null or right == null) return left == null and right == null;
98 return equal(left.?, right.?);
99}

101fn differences(gpa: Allocator, a: Tree, b: Tree) Allocator.Error![]Text {
102 var result: std.ArrayList(Text) = .empty;
103 for (try paths(gpa, &.{ a, b })) |path| if (!agree(a, b, path)) {
104 try result.append(gpa, path);
105 };
106 return result.items;
107}

109fn same(gpa: Allocator, a: Tree, b: Tree) Allocator.Error!bool {
110 return (try differences(gpa, a, b)).len == 0;
111}

113fn matchPath(gpa: Allocator, target: *Tree, source: Tree, path: Text) Allocator.Error!void {
114 if (source.get(path)) |text| try target.put(gpa, path, text) else _ = target.remove(path);

119fn threeWay(gpa: Allocator, base: Tree, ours: Tree, theirs: Tree) Allocator.Error!Merged {
120 var result: Tree = .empty;
121 var conflicts: std.ArrayList(Text) = .empty;
122 for (try paths(gpa, &.{ base, ours, theirs })) |path| {
123 if (agree(theirs, base, path) or agree(ours, theirs, path)) try matchPath(gpa, &result, ours, path) else if (agree(ours, base, path)) try matchPath(gpa, &result, theirs, path) else {
124 try conflicts.append(gpa, path);
125 try matchPath(gpa, &result, ours, path);
126 }
127 }
128 return .{ .files = result, .conflicts = conflicts.items };
129}

137fn distances(gpa: Allocator, start: *Commit) Allocator.Error!Distances {
138 var result: Distances = .empty;
139 try result.put(gpa, start, 0);
140 var queue: std.ArrayList(*Commit) = .empty;
141 try queue.append(gpa, start);
142 var at: usize = 0;
143 while (at < queue.items.len) : (at += 1) {
144 const current = queue.items[at];
145 for (current.parents) |parent| {
146 if (!result.contains(parent)) {
147 try result.put(gpa, parent, result.get(current).? + 1);
148 try queue.append(gpa, parent);
149 }
150 }
151 }
152 return result;
153}

155fn ancestor(gpa: Allocator, a: *Commit, b: *Commit) Allocator.Error!bool {
156 return (try distances(gpa, b)).contains(a);

159fn mergeBase(gpa: Allocator, a: *Commit, b: *Commit) Allocator.Error!*Commit {
160 const da = try distances(gpa, a);
161 const db = try distances(gpa, b);
162 var common: CommitSet = .empty;
163 var excluded: CommitSet = .empty;
164 var distance_iterator = da.keyIterator();
165 while (distance_iterator.next()) |c| if (db.contains(c.*)) {
166 try common.put(gpa, c.*, {});
167 };
168 var common_iterator = common.keyIterator();
169 while (common_iterator.next()) |c| for (c.*.parents) |parent| {
170 try excluded.put(gpa, parent, {});
171 };
172 var best: ?*Commit = null;
173 var best_distance: usize = 0;
174 common_iterator = common.keyIterator();
175 while (common_iterator.next()) |key| {
176 const c = key.*;
177 if (excluded.contains(c)) continue;
178 const distance = @max(da.get(c).?, db.get(c).?);
179 if (best == null or distance < best_distance or (distance == best_distance and c.sequence > best.?.sequence)) {
180 best = c;
181 best_distance = distance;
182 }
183 }
184 return best.?;
185}

187fn reachable(gpa: Allocator, roots: []const *Commit) Allocator.Error!CommitSet {
188 var result: CommitSet = .empty;
189 for (roots) |root| {
190 const all = try distances(gpa, root);
191 var iterator = all.keyIterator();
192 while (iterator.next()) |c| try result.put(gpa, c.*, {});
193 }
194 return result;
195}

197fn glob(gpa: Allocator, pattern: Text, path: Text) Allocator.Error!bool {
198 var previous = try gpa.alloc(bool, path.len + 1);
199 @memset(previous, false);
200 previous[0] = true;
201 for (pattern) |c| {
202 const current = try gpa.alloc(bool, path.len + 1);
203 @memset(current, false);
204 if (c == '*') current[0] = previous[0];
205 for (1..path.len + 1) |j| current[j] = if (c == '*') previous[j] or current[j - 1] else previous[j - 1] and (c == '?' or c == path[j - 1]);
206 previous = current;
207 }
208 return previous[path.len];
209}

215const Repository = struct {
216 gpa: Allocator,
217 out: Builder,
218 commits: References = .empty,
219 branches: References = .empty,
220 tags: References = .empty,
221 attached: ?Text = "main",
222 detached: ?*Commit = null,
223 work: Tree = .empty,
224 index: Tree = .empty,
225 conflicts: Tree = .empty,
226 pending: ?Operation = null,
227 stash: std.ArrayList(StashEntry) = .empty,
228 ignores: std.ArrayList(Text) = .empty,
229 created: usize = 0,
230 fn init(self: *Repository) Error!void {
231 const root = try self.create(&.{}, .empty, "init");
232 try self.branches.put(self.gpa, "main", root);
233 }
234 fn head(self: *const Repository) *Commit {
235 return if (self.attached) |branch| self.branches.get(branch).? else self.detached.?;
236 }
237 fn moveHead(self: *Repository, commit: *Commit) Allocator.Error!void {
238 if (self.attached) |branch| try self.branches.put(self.gpa, branch, commit) else self.detached = commit;
239 }
240 fn create(self: *Repository, parents: []const *Commit, files: Tree, message: Text) Error!*Commit {
241 var canonical: Builder = .{ .gpa = self.gpa };
242 try canonical.line("commit {d}", .{self.created});
243 for (parents) |parent| try canonical.line("parent {s}", .{parent.id});
244 for (try keys(self.gpa, files)) |path| try canonical.line("file {s} {s}", .{ path, files.get(path).? });
245 try canonical.line("message {s}", .{message});
246 var hash: u64 = 0;
247 for (canonical.result()) |c| hash = (hash * 131 + c) % 2147483647;
248 const commit = try self.gpa.create(Commit);
249 commit.* = .{ .id = try std.fmt.allocPrint(self.gpa, "{x:0>8}", .{hash}), .sequence = self.created, .parents = try self.gpa.dupe(*Commit, parents), .files = try clone(self.gpa, files), .message = message };
250 self.created += 1;
251 try self.commits.put(self.gpa, commit.id, commit);
252 return commit;
253 }
254 fn record(self: *Repository, parents: []const *Commit, files: Tree, message: Text) Error!void {
255 const commit = try self.create(parents, files, message);
256 try self.moveHead(commit);
257 var added: usize = 0;
258 var modified: usize = 0;
259 var deleted: usize = 0;
260 const old = parents[0].files;
261 for (try differences(self.gpa, old, files)) |path| {
262 if (!old.contains(path)) added += 1 else if (!files.contains(path)) deleted += 1 else modified += 1;
263 }
264 try self.out.line("[{s} {s}] {s} ({d} added, {d} modified, {d} deleted)", .{ self.attached orelse "detached HEAD", commit.id, message, added, modified, deleted });
265 }
266 fn revision(self: *const Repository, rev: Text) Error!*Commit {
267 const at = std.mem.findScalar(u8, rev, '~');
268 const anchor = if (at) |i| rev[0..i] else rev;
269 var commit: ?*Commit = if (equal(anchor, "HEAD")) self.head() else self.branches.get(anchor) orelse self.tags.get(anchor) orelse self.commits.get(anchor);
270 if (at) |i| {
271 const n = std.fmt.parseInt(usize, rev[i + 1 ..], 10) catch unreachable;
272 for (0..n) |_| if (commit) |c| {
273 commit = if (c.parents.len == 0) null else c.parents[0];
274 };
275 }
276 return commit orelse return bad(self.gpa, "unknown revision '{s}'", .{rev});
277 }
278 fn summary(self: *const Repository, commit: *Commit) Error!Text {
279 const gpa = self.gpa;
280 var refs: std.ArrayList(Text) = .empty;
281 if (self.head() == commit) try refs.append(gpa, if (self.attached) |branch| try std.fmt.allocPrint(gpa, "HEAD -> {s}", .{branch}) else "HEAD");
282 for (try keys(gpa, self.branches)) |name| {
283 if ((self.attached == null or !equal(name, self.attached.?)) and self.branches.get(name).? == commit) try refs.append(gpa, name);
284 }
285 for (try keys(gpa, self.tags)) |name| if (self.tags.get(name).? == commit) {
286 try refs.append(gpa, try std.fmt.allocPrint(gpa, "tag: {s}", .{name}));
287 };
288 const references = if (refs.items.len == 0) "" else try std.fmt.allocPrint(gpa, " ({s})", .{try join(gpa, refs.items, ", ")});
289 return std.fmt.allocPrint(gpa, "{s}{s} {s}", .{ commit.id, references, commit.message });
290 }
291 fn changes(self: *Repository, a: Tree, b: Tree) Error!void {
292 const changed = try differences(self.gpa, a, b);
293 if (changed.len == 0) {
294 try self.out.add("(no changes)\n");
295 return;
296 }
297 for (changed) |path| {
298 if (!a.contains(path)) try self.out.line("A {s}: {s}", .{ path, b.get(path).? }) else if (!b.contains(path)) try self.out.line("D {s}: {s}", .{ path, a.get(path).? }) else try self.out.line("M {s}: {s} -> {s}", .{ path, a.get(path).?, b.get(path).? });
299 }
300 }
301 fn inProgress(self: *const Repository) Error!void {
302 if (self.pending) |pending| return bad(self.gpa, "{s} in progress", .{pending.kind});
303 }
304 fn uncommitted(self: *const Repository) Error!bool {
305 if (!try same(self.gpa, self.index, self.head().files)) return true;
306 var iterator = self.index.keyIterator();
307 while (iterator.next()) |path| if (!agree(self.index, self.work, path.*)) return true;
308 return false;
309 }
310 fn requireClean(self: *const Repository) Error!void {
311 if (try self.uncommitted()) return bad(self.gpa, "uncommitted changes", .{});
312 }
313 fn overwrite(self: *const Repository, selected: []Text) Error!void {
314 std.mem.sort(Text, selected, {}, textLess);
315 for (selected) |path| if (!agree(self.index, self.head().files, path) or !agree(self.work, self.head().files, path)) return bad(self.gpa, "local changes to '{s}' would be overwritten", .{path});
316 }
317 fn switchTree(self: *Repository, files: Tree) Error!void {
318 const changed = try differences(self.gpa, self.head().files, files);
319 try self.overwrite(changed);
320 for (changed) |path| {
321 try matchPath(self.gpa, &self.index, files, path);
322 try matchPath(self.gpa, &self.work, files, path);
323 }
324 }
325 fn ignored(self: *const Repository, path: Text) Error!bool {
326 if (self.index.contains(path) or self.conflicts.contains(path)) return false;
327 for (self.ignores.items) |pattern| if (try glob(self.gpa, pattern, path)) return true;
328 return false;
329 }
330 fn apply(self: *Repository, base: Tree, theirs: Tree, kind: Text, target: *Commit, message: Text, parents: []const *Commit, base_commit: ?*Commit) Error!void {
331 const gpa = self.gpa;
332 const merged = try threeWay(gpa, base, self.head().files, theirs);
333 const changed = try differences(gpa, self.head().files, merged.files);
334 var selected: std.ArrayList(Text) = .empty;
335 try selected.appendSlice(gpa, changed);
336 try selected.appendSlice(gpa, merged.conflicts);
337 try self.overwrite(selected.items);
338 if (equal(kind, "merge")) try self.out.line("Merge base {s}", .{base_commit.?.id});
339 if (!equal(kind, "merge") and merged.conflicts.len == 0 and try same(gpa, merged.files, self.head().files)) {
340 try self.out.add("nothing to commit\n");
341 return;
342 }
343 for (changed) |path| {
344 try matchPath(gpa, &self.index, merged.files, path);
345 try matchPath(gpa, &self.work, merged.files, path);
346 }
347 if (merged.conflicts.len == 0) {
348 try self.record(parents, merged.files, message);
349 return;
350 }
351 self.conflicts = .empty;
352 for (merged.conflicts) |path| {
353 const content = try std.fmt.allocPrint(gpa, "<<<<<<< {s} ======= {s} >>>>>>>", .{ self.head().files.get(path) orelse "(deleted)", theirs.get(path) orelse "(deleted)" });
354 try self.work.put(gpa, path, content);
355 try self.conflicts.put(gpa, path, "");
356 try self.out.line("CONFLICT {s}", .{path});
357 }
358 self.pending = .{ .kind = kind, .target = target };
359 if (equal(kind, "merge")) try self.out.add("Automatic merge failed; fix conflicts and then commit the result.\n") else if (equal(kind, "revert")) try self.out.line("error: could not revert {s}", .{target.id}) else try self.out.line("error: could not apply {s}", .{target.id});
360 }
361 fn roots(self: *const Repository, extra: bool) Allocator.Error![]*Commit {
362 var result: std.ArrayList(*Commit) = .empty;
363 try result.append(self.gpa, self.head());
364 var iterator = self.branches.valueIterator();
365 while (iterator.next()) |commit| try result.append(self.gpa, commit.*);
366 iterator = self.tags.valueIterator();
367 while (iterator.next()) |commit| try result.append(self.gpa, commit.*);
368 if (extra) {
369 for (self.stash.items) |entry| try result.append(self.gpa, entry.base);
370 if (self.pending) |pending| try result.append(self.gpa, pending.target);
371 }
372 return result.items;
373 }
374 fn command(self: *Repository, line: Text) Error!void {
375 const gpa = self.gpa;
376 const words = try split(gpa, line, ' ', 0);
377 const cmd = words[0];
378 const h = self.head();
379 if (equal(cmd, "WRITE")) {
380 const args = try split(gpa, line, ' ', 3);
381 try self.work.put(gpa, args[1], args[2]);
382 } else if (equal(cmd, "REMOVE")) {
383 if (!self.work.contains(words[1])) return bad(gpa, "no such file '{s}'", .{words[1]});
384 _ = self.work.remove(words[1]);
385 } else if (equal(cmd, "IGNORE")) try self.ignores.append(gpa, words[1]) else if (equal(cmd, "ADD")) {
386 const path = words[1];
387 if (equal(path, "ALL")) {
388 for (try paths(gpa, &.{ self.work, self.index })) |p| if (!try self.ignored(p)) {
389 try matchPath(gpa, &self.index, self.work, p);
390 };
391 self.conflicts = .empty;
392 } else {
393 if (self.work.contains(path) and try self.ignored(path)) return bad(gpa, "path '{s}' is ignored", .{path});
394 if (!self.work.contains(path) and !self.index.contains(path) and !self.conflicts.contains(path)) return bad(gpa, "pathspec '{s}' did not match any files", .{path});
395 try matchPath(gpa, &self.index, self.work, path);
396 _ = self.conflicts.remove(path);
397 }
398 } else if (equal(cmd, "RM")) {
399 const path = words[1];
400 if (!self.index.contains(path)) return bad(gpa, "pathspec '{s}' did not match any files", .{path});
401 _ = self.index.remove(path);
402 _ = self.work.remove(path);
403 _ = self.conflicts.remove(path);
404 } else if (equal(cmd, "UNSTAGE")) {
405 const path = words[1];
406 if (!self.index.contains(path) and !h.files.contains(path)) return bad(gpa, "pathspec '{s}' did not match any files", .{path});
407 try matchPath(gpa, &self.index, h.files, path);
408 } else if (equal(cmd, "RESTORE") or equal(cmd, "CAT")) {
409 const path = words[1];
410 var files = if (equal(cmd, "CAT")) self.work else self.index;
411 if (words.len == 3) {
412 files = (try self.revision(words[2])).files;
413 if (!files.contains(path)) return bad(gpa, "path '{s}' does not exist in '{s}'", .{ path, words[2] });
414 } else if (!files.contains(path)) {
415 if (equal(cmd, "CAT")) return bad(gpa, "no such file '{s}'", .{path});
416 return bad(gpa, "pathspec '{s}' did not match any files", .{path});
417 }
418 if (equal(cmd, "CAT")) try self.out.line("{s}", .{files.get(path).?}) else {
419 try self.work.put(gpa, path, files.get(path).?);
420 if (words.len == 3) {
421 try self.index.put(gpa, path, files.get(path).?);
422 _ = self.conflicts.remove(path);
423 }
424 }
425 } else if (equal(cmd, "COMMIT")) {
426 if (self.conflicts.count() > 0) return bad(gpa, "unresolved conflicts", .{});
427 const message = (try split(gpa, line, ' ', 2))[1];
428 var parents: std.ArrayList(*Commit) = .empty;
429 try parents.append(gpa, h);
430 if (self.pending != null and equal(self.pending.?.kind, "merge")) try parents.append(gpa, self.pending.?.target) else if (try same(gpa, self.index, h.files)) {
431 try self.out.add("nothing to commit\n");
432 return;
433 }
434 try self.record(parents.items, self.index, message);
435 self.pending = null;
436 } else if (equal(cmd, "AMEND")) {
437 try self.inProgress();
438 if (h.parents.len == 0) return bad(gpa, "cannot amend the root commit", .{});
439 try self.record(h.parents, self.index, (try split(gpa, line, ' ', 2))[1]);
440 } else if (equal(cmd, "STATUS")) {
441 if (self.attached) |branch| try self.out.line("On branch {s}", .{branch}) else try self.out.line("HEAD detached at {s}", .{h.id});
442 if (self.pending) |pending| try self.out.line("{s} {s}", .{ if (equal(pending.kind, "merge")) "Merging" else if (equal(pending.kind, "revert")) "Reverting" else "Cherry-picking", pending.target.id });
443 var printed = false;
444 for (try paths(gpa, &.{ h.files, self.index, self.work, self.conflicts })) |path| {
445 if (self.conflicts.contains(path)) {
446 try self.out.line("UU {s}", .{path});
447 printed = true;
448 } else if (self.index.contains(path)) {
449 const x: u8 = if (!h.files.contains(path)) 'A' else if (!agree(h.files, self.index, path)) 'M' else '.';
450 const y: u8 = if (!self.work.contains(path)) 'D' else if (!agree(self.work, self.index, path)) 'M' else '.';
451 if (x != '.' or y != '.') {
452 try self.out.line("{c}{c} {s}", .{ x, y, path });
453 printed = true;
454 }
455 } else {
456 if (h.files.contains(path)) {
457 try self.out.line("D. {s}", .{path});
458 printed = true;
459 }
460 if (self.work.contains(path) and !try self.ignored(path)) {
461 try self.out.line("?? {s}", .{path});
462 printed = true;
463 }
464 }
465 }
466 if (!printed) try self.out.add("nothing to commit, working tree clean\n");
467 } else if (equal(cmd, "LS")) {
468 const files = if (words.len == 2) (try self.revision(words[1])).files else self.work;
469 const selected = try keys(gpa, files);
470 if (selected.len == 0) try self.out.add("(empty)\n");
471 for (selected) |path| try self.out.line("{s} {s}", .{ path, files.get(path).? });
472 } else if (equal(cmd, "DIFF")) {
473 if (words.len == 1) {
474 var work: Tree = .empty;
475 for (try keys(gpa, self.index)) |path| try matchPath(gpa, &work, self.work, path);
476 try self.changes(self.index, work);
477 } else if (words.len == 2) try self.changes(h.files, self.index) else {
478 const a = try self.revision(words[1]);
479 const b = try self.revision(words[2]);
480 try self.changes(a.files, b.files);
481 }
482 } else if (equal(cmd, "LOG")) {
483 const limit = if (words.len == 2) std.fmt.parseInt(usize, words[1], 10) catch unreachable else self.created;
484 var c = h;
485 var n: usize = 0;
486 while (n < limit) : (n += 1) {
487 try self.out.line("{s}", .{try self.summary(c)});
488 if (c.parents.len == 0) break;
489 c = c.parents[0];
490 }
491 } else if (equal(cmd, "SHOW")) {
492 const c = try self.revision(if (words.len == 2) words[1] else "HEAD");
493 try self.out.line("commit {s}", .{try self.summary(c)});
494 var parent_ids: std.ArrayList(Text) = .empty;
495 for (c.parents) |parent| try parent_ids.append(gpa, parent.id);
496 try self.out.line("parents: {s}", .{if (parent_ids.items.len == 0) "none" else try join(gpa, parent_ids.items, " ")});
497 try self.changes(parentTree(c), c.files);
498 } else if (equal(cmd, "BLAME")) {
499 const path = words[1];
500 const content = h.files.get(path) orelse return bad(gpa, "path '{s}' does not exist in 'HEAD'", .{path});
501 var c = h;
502 while (c.parents.len > 0 and agree(c.parents[0].files, h.files, path)) c = c.parents[0];
503 try self.out.line("{s} ({s}) {s}", .{ c.id, c.message, content });
504 } else if (equal(cmd, "BRANCH") or equal(cmd, "TAG")) {
505 const refs = if (equal(cmd, "TAG")) &self.tags else &self.branches;
506 const label: Text = if (equal(cmd, "TAG")) "tag" else "branch";
507 if (equal(words[1], "-d")) {
508 const name = words[2];
509 const c = refs.get(name) orelse return bad(gpa, "no such {s} '{s}'", .{ label, name });
510 if (equal(cmd, "BRANCH")) {
511 if (self.attached != null and equal(self.attached.?, name)) return bad(gpa, "cannot delete current branch '{s}'", .{name});
512 if (!try ancestor(gpa, c, h)) return bad(gpa, "branch '{s}' is not fully merged", .{name});
513 }
514 _ = refs.remove(name);
515 try self.out.line("Deleted {s} {s} (was {s})", .{ label, name, c.id });
516 } else {
517 const name = words[1];
518 if (refs.contains(name)) return bad(gpa, "{s} '{s}' already exists", .{ label, name });
519 const c = try self.revision(if (words.len == 3) words[2] else "HEAD");
520 try refs.put(gpa, name, c);
521 try self.out.line("Created {s} {s} at {s}", .{ label, name, c.id });
522 }
523 } else if (equal(cmd, "BRANCHES")) {
524 if (self.attached == null) try self.out.line("* (HEAD detached at {s})", .{h.id});
525 for (try keys(gpa, self.branches)) |name| {
526 const c = self.branches.get(name).?;
527 try self.out.line("{s} {s} {s} {s}", .{ if (self.attached != null and equal(name, self.attached.?)) "*" else " ", name, c.id, c.message });
528 }
529 } else if (equal(cmd, "TAGS")) {
530 const names = try keys(gpa, self.tags);
531 if (names.len == 0) try self.out.add("(no tags)\n");
532 for (names) |name| {
533 const c = self.tags.get(name).?;
534 try self.out.line("{s} {s} {s}", .{ name, c.id, c.message });
535 }
536 } else if (equal(cmd, "SWITCH")) {
537 const name = words[1];
538 const c = self.branches.get(name) orelse return bad(gpa, "no such branch '{s}'", .{name});
539 if (self.attached != null and equal(name, self.attached.?)) {
540 try self.out.line("Already on '{s}'", .{name});
541 return;
542 }
543 try self.inProgress();
544 try self.switchTree(c.files);
545 self.attached = name;
546 self.detached = null;
547 try self.out.line("Switched to branch '{s}'", .{name});
548 } else if (equal(cmd, "DETACH")) {
549 const c = try self.revision(words[1]);
550 try self.inProgress();
551 try self.switchTree(c.files);
552 self.attached = null;
553 self.detached = c;
554 try self.out.line("HEAD is now at {s} {s}", .{ c.id, c.message });
555 } else if (equal(cmd, "MERGEBASE")) {
556 const a = try self.revision(words[1]);
557 const b = try self.revision(words[2]);
558 try self.out.line("{s}", .{(try mergeBase(gpa, a, b)).id});
559 } else if (equal(cmd, "MERGE") or equal(cmd, "REVERT") or equal(cmd, "CHERRYPICK")) {
560 const c = try self.revision(words[1]);
561 try self.inProgress();
562 if (equal(cmd, "MERGE")) {
563 if (try ancestor(gpa, c, h)) {
564 try self.out.add("Already up to date\n");
565 return;
566 }
567 if (try ancestor(gpa, h, c)) {
568 try self.switchTree(c.files);
569 try self.moveHead(c);
570 try self.out.line("Fast-forward {s}..{s}", .{ h.id, c.id });
571 return;
572 }
573 }
574 try self.requireClean();
575 if (equal(cmd, "MERGE")) {
576 const base = try mergeBase(gpa, h, c);
577 try self.apply(base.files, c.files, "merge", c, try std.fmt.allocPrint(gpa, "Merge {s}", .{words[1]}), &.{ h, c }, base);
578 } else if (equal(cmd, "REVERT")) try self.apply(c.files, parentTree(c), "revert", c, try std.fmt.allocPrint(gpa, "Revert \"{s}\"", .{c.message}), &.{h}, null) else try self.apply(parentTree(c), c.files, "cherry-pick", c, c.message, &.{h}, null);
579 } else if (equal(cmd, "REBASE")) {
580 const upstream = try self.revision(words[1]);
581 try self.inProgress();
582 if (try ancestor(gpa, upstream, h)) {
583 try self.out.add("Already up to date\n");
584 return;
585 }
586 try self.requireClean();
587 var list: std.ArrayList(*Commit) = .empty;
588 var c = h;
589 while (!try ancestor(gpa, c, upstream)) {
590 try list.append(gpa, c);
591 c = c.parents[0];
592 }
593 std.mem.reverse(*Commit, list.items);
594 var current = upstream.files;
595 var picks: std.ArrayList(?Tree) = .empty;
596 for (list.items) |old| {
597 const merged = try threeWay(gpa, parentTree(old), current, old.files);
598 if (merged.conflicts.len > 0) return bad(gpa, "could not rebase {s}: conflict in '{s}'", .{ old.id, merged.conflicts[0] });
599 if (try same(gpa, merged.files, current)) try picks.append(gpa, null) else {
600 try picks.append(gpa, merged.files);
601 current = merged.files;
602 }
603 }
604 try self.switchTree(current);
605 var tip = upstream;
606 for (list.items, picks.items) |old, picked| {
607 if (picked) |files| {
608 const next = try self.create(&.{tip}, files, old.message);
609 try self.out.line("pick {s} -> {s} {s}", .{ old.id, next.id, old.message });
610 tip = next;
611 } else try self.out.line("skip {s} {s}", .{ old.id, old.message });
612 }
613 try self.moveHead(tip);
614 try self.out.line("HEAD is now at {s} {s}", .{ tip.id, tip.message });
615 } else if (equal(cmd, "RESET")) {
616 const c = try self.revision(words[2]);
617 if (equal(words[1], "HARD")) {
618 for (try paths(gpa, &.{ self.index, c.files, self.conflicts })) |path| try matchPath(gpa, &self.work, c.files, path);
619 }
620 if (!equal(words[1], "SOFT")) self.index = try clone(gpa, c.files);
621 try self.moveHead(c);
622 self.conflicts = .empty;
623 self.pending = null;
624 try self.out.line("HEAD is now at {s} {s}", .{ c.id, c.message });
625 } else if (equal(cmd, "STASH")) {
626 const action: Text = if (words.len > 1) words[1] else "";
627 if (equal(action, "LIST")) {
628 if (self.stash.items.len == 0) try self.out.add("(no stash entries)\n");
629 for (self.stash.items, 0..) |entry, i| try self.out.line("stash@{{{d}}}: {s}", .{ i, entry.label });
630 return;
631 }
632 if (action.len > 0 and self.stash.items.len == 0) return bad(gpa, "no stash entries", .{});
633 if (equal(action, "DROP")) {
634 const entry = self.stash.orderedRemove(0);
635 try self.out.line("Dropped stash@{{0}}: {s}", .{entry.label});
636 return;
637 }
638 try self.inProgress();
639 if (equal(action, "POP")) {
640 try self.requireClean();
641 const entry = self.stash.items[0];
642 const merged = try threeWay(gpa, entry.base.files, h.files, entry.files);
643 if (merged.conflicts.len > 0) return bad(gpa, "stash conflict in '{s}'", .{merged.conflicts[0]});
644 const changed = try differences(gpa, h.files, merged.files);
645 try self.overwrite(changed);
646 for (changed) |path| try matchPath(gpa, &self.work, merged.files, path);
647 _ = self.stash.orderedRemove(0);
648 try self.out.line("Restored stash@{{0}}: {s}", .{entry.label});
649 try self.changes(h.files, merged.files);
650 } else {
651 if (!try self.uncommitted()) {
652 try self.out.add("No local changes to save\n");
653 return;
654 }
655 var saved: Tree = .empty;
656 for (try paths(gpa, &.{ self.index, h.files })) |path| {
657 try matchPath(gpa, &saved, self.work, path);
658 try matchPath(gpa, &self.work, h.files, path);
659 }
660 self.index = try clone(gpa, h.files);
661 const label = try std.fmt.allocPrint(gpa, "WIP on {s}: {s} {s}", .{ self.attached orelse "(no branch)", h.id, h.message });
662 try self.stash.insert(gpa, 0, .{ .label = label, .base = h, .files = saved });
663 try self.out.line("Saved stash@{{0}}: {s}", .{label});
664 }
665 } else if (equal(cmd, "GRAPH")) {
666 const all = try reachable(gpa, try self.roots(false));
667 var ordered: std.ArrayList(*Commit) = .empty;
668 var iterator = all.keyIterator();
669 while (iterator.next()) |c| try ordered.append(gpa, c.*);
670 std.mem.sort(*Commit, ordered.items, {}, commitLater);
671 var lanes: std.ArrayList(*Commit) = .empty;
672 for (ordered.items) |c| {
673 var at = std.mem.findScalar(*Commit, lanes.items, c);
674 if (at == null) {
675 at = lanes.items.len;
676 try lanes.append(gpa, c);
677 }
678 const position = at.?;
679 var symbols: std.ArrayList(Text) = .empty;
680 var kept: std.ArrayList(*Commit) = .empty;
681 for (lanes.items, 0..) |held, i| {
682 try symbols.append(gpa, if (held == c) (if (i == position) @as(Text, "*") else "/") else "|");
683 if (held != c or i == position) try kept.append(gpa, held);
684 }
685 try self.out.line("{s} {s}", .{ try join(gpa, symbols.items, " "), try self.summary(c) });
686 lanes = kept;
687 if (c.parents.len == 0) _ = lanes.orderedRemove(position) else {
688 lanes.items[position] = c.parents[0];
689 if (c.parents.len == 2 and std.mem.findScalar(*Commit, lanes.items, c.parents[1]) == null) try lanes.insert(gpa, position + 1, c.parents[1]);
690 }
691 }
692 } else if (equal(cmd, "DESCRIBE")) {
693 const c = try self.revision(if (words.len == 2) words[1] else "HEAD");
694 const d = try distances(gpa, c);
695 var best: ?Text = null;
696 var distance: usize = 0;
697 for (try keys(gpa, self.tags)) |name| {
698 if (d.get(self.tags.get(name).?)) |n| {
699 if (best == null or n < distance) {
700 best = name;
701 distance = n;
702 }
703 }
704 }
705 const name = best orelse return bad(gpa, "no tag can describe '{s}'", .{c.id});
706 if (distance == 0) try self.out.line("{s}", .{name}) else try self.out.line("{s}-{d}-g{s}", .{ name, distance, c.id });
707 } else if (equal(cmd, "GC")) {
708 const keep = try reachable(gpa, try self.roots(true));
709 var removed: usize = 0;
710 for (try keys(gpa, self.commits)) |id| if (!keep.contains(self.commits.get(id).?)) {
711 _ = self.commits.remove(id);
712 removed += 1;
713 };
714 try self.out.line("Removed unreachable commits: {d}", .{removed});
715 }
716 }
717};
718
719fn commitLater(_: void, a: *Commit, b: *Commit) bool {
720 return a.sequence > b.sequence;
721}
722
723pub fn main(init: std.process.Init) !void {
724 var storage = std.heap.ArenaAllocator.init(std.heap.page_allocator);
725 defer storage.deinit();
726 const gpa = storage.allocator();
727 var input_buffer: [4096]u8 = undefined;
728 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
729 var output_buffer: [4096]u8 = undefined;
730 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
731 const text = try input.interface.allocRemaining(gpa, .unlimited);
732 var repository: Repository = .{ .gpa = gpa, .out = .{ .gpa = gpa } };
733 try repository.init();
734 for (try split(gpa, text, '\n', 0)) |line| {
735 if (equal(line, "EXIT")) break;
736 repository.command(line) catch |err| {
737 if (err != error.Invalid) return err;
738 try repository.out.line("error: {s}", .{failure});
739 };
740 }
741 try output.interface.writeAll(repository.out.result());
742 try output.interface.flush();
743}
