49 private function paths(array ...$trees): array
50 {
51 $paths = [];
52 foreach ($trees as $tree) {
53 foreach ($tree as $path => $_) {
54 $paths[$path] = true;
55 }
56 }
57 $result = array_map('strval', array_keys($paths));
58 sort($result, SORT_STRING);
59 return $result;

62 private function differences(array $a, array $b): array
63 {

65 fn($p) => ($a[$p] ?? null) !== ($b[$p] ?? null)));

82 private function create(array $parents, array $tree, string $message): Commit
83 {
84 $text = "commit {$this->nextSequence}\n";
85 foreach ($parents as $parent) {
86 $text .= "parent {$parent->id}\n";
87 }
88 foreach ($this->paths($tree) as $path) {
89 $text .= "file $path {$tree[$path]}\n";
90 }
91 $text .= "message $message\n";
92 $hash = 0;
93 for ($i = 0; $i < strlen($text); $i++) {
94 $hash = ($hash * 131 + ord($text[$i])) % 2147483647;
95 }
96 $commit = new Commit(sprintf('%08x', $hash), $this->nextSequence++, $parents, $tree, $message);
97 $this->commits[$commit->id] = $commit;
98 return $commit;
99 }

101 private function record(array $parents, array $tree, string $message): void
102 {
103 $old = $parents[0]->tree;
104 $added = 0;
105 $modified = 0;
106 $deleted = 0;
107 foreach ($this->differences($old, $tree) as $path) {
108 if (!isset($old[$path])) {
109 $added++;
110 } elseif (!isset($tree[$path])) {
111 $deleted++;
112 } else {
113 $modified++;
114 }
115 }
116 $commit = $this->create($parents, $tree, $message);
117 $this->moveHead($commit);
118 $location = $this->branch ?? 'detached HEAD';
119 echo "[$location {$commit->id}] $message ($added added, $modified modified, $deleted deleted)\n";
120 }

122 private function resolve(string $revision): Commit
123 {
124 $parts = explode('~', $revision);
125 $anchor = $parts[0];
126 $commit = $anchor === 'HEAD' ? $this->head() :
127 ($this->branches[$anchor] ?? $this->tags[$anchor] ?? $this->commits[$anchor] ?? null);
128 for ($i = 0; $i < (int)($parts[1] ?? 0); $i++) {
129 $commit = $commit?->parents[0] ?? null;
130 }
131 if ($commit === null) {
132 throw new VcsError("unknown revision '$revision'");
133 }
134 return $commit;
135 }

137 private function distances(Commit $commit): array
138 {
139 $distances = [$commit->id => 0];
140 $queue = [$commit];
141 for ($i = 0; $i < count($queue); $i++) {
142 $current = $queue[$i];
143 foreach ($current->parents as $parent) {
144 if (!isset($distances[$parent->id])) {
145 $distances[$parent->id] = $distances[$current->id] + 1;
146 $queue[] = $parent;
147 }
148 }
149 }
150 return $distances;
151 }

153 private function ancestor(Commit $ancestor, Commit $descendant): bool
154 {
155 return isset($this->distances($descendant)[$ancestor->id]);

158 private function mergeBase(Commit $a, Commit $b): Commit
159 {
160 $da = $this->distances($a);
161 $db = $this->distances($b);
162 $common = array_intersect_key($da, $db);
163 $best = null;
164 $bestDistance = PHP_INT_MAX;
165 foreach ($common as $id => $_) {
166 $candidate = $this->commits[$id];
167 foreach ($common as $other => $_) {
168 if ($id !== $other && $this->ancestor($candidate, $this->commits[$other])) {
169 continue 2;
170 }
171 }
172 $distance = max($da[$id], $db[$id]);
173 if ($best === null || $distance < $bestDistance ||
174 ($distance === $bestDistance && $candidate->sequence > $best->sequence)) {
175 $best = $candidate;
176 $bestDistance = $distance;
177 }
178 }
179 return $best;
180 }

182 private function summary(Commit $commit): string
183 {
184 $refs = [];
185 if ($this->head() === $commit) {
186 $refs[] = $this->branch !== null ? "HEAD -> {$this->branch}" : 'HEAD';
187 }
188 foreach ($this->paths($this->branches) as $name) {
189 if ($this->branches[$name] === $commit && $name !== $this->branch) {
190 $refs[] = $name;
191 }
192 }
193 foreach ($this->paths($this->tags) as $name) {
194 if ($this->tags[$name] === $commit) {
195 $refs[] = "tag: $name";
196 }
197 }
198 return $commit->id . ($refs ? ' (' . implode(', ', $refs) . ')' : '') . " {$commit->message}";
199 }

204 if (!$changed) {
205 echo "(no changes)\n";

245 private function checkOverwrite(array $selected): void
246 {
247 $selected = array_unique($selected);
248 sort($selected, SORT_STRING);
249 $head = $this->head()->tree;
250 foreach ($selected as $path) {
251 if (($this->index[$path] ?? null) !== ($head[$path] ?? null) ||
252 ($this->work[$path] ?? null) !== ($head[$path] ?? null)) {
253 throw new VcsError("local changes to '$path' would be overwritten");
254 }

258 private function switchTree(array $tree): void
259 {
260 $changed = $this->differences($this->head()->tree, $tree);
261 $this->checkOverwrite($changed);
262 foreach ($changed as $path) {
263 $this->matchPath($this->index, $tree, $path);
264 $this->matchPath($this->work, $tree, $path);
265 }

268 private function mergeTrees(array $base, array $ours, array $theirs): array
269 {
270 $result = [];
271 $conflicts = [];
272 $markers = [];
273 foreach ($this->paths($base, $ours, $theirs) as $path) {
274 $b = $base[$path] ?? null;
275 $o = $ours[$path] ?? null;
276 $t = $theirs[$path] ?? null;
277 if ($t === $b || $o === $t) {
278 $this->matchPath($result, $ours, $path);
279 } elseif ($o === $b) {
280 $this->matchPath($result, $theirs, $path);
281 } else {
282 $conflicts[] = $path;
283 $this->matchPath($result, $ours, $path);
284 $markers[$path] = '<<<<<<< ' . ($o ?? '(deleted)') . ' ======= ' . ($t ?? '(deleted)') . ' >>>>>>>';
285 }
286 }
287 return [$result, $conflicts, $markers];
288 }

290 private function apply(array $base, array $theirs, string $kind, Commit $target,
291 string $message, array $parents, ?Commit $mergeBase = null): void
292 {
293 [$result, $conflicts, $markers] = $this->mergeTrees($base, $this->head()->tree, $theirs);
294 $changed = $this->differences($this->head()->tree, $result);
295 $this->checkOverwrite(array_merge($changed, $conflicts));
296 if ($kind === 'merge') {
297 echo "Merge base {$mergeBase->id}\n";
298 }
299 if ($kind !== 'merge' && !$conflicts && $this->equalTrees($result, $this->head()->tree)) {
300 echo "nothing to commit\n";
301 return;
302 }
303 foreach ($changed as $path) {
304 $this->matchPath($this->index, $result, $path);
305 $this->matchPath($this->work, $result, $path);
306 }
307 if (!$conflicts) {
308 $this->record($parents, $result, $message);
309 return;
310 }
311 foreach ($conflicts as $path) {
312 $this->work[$path] = $markers[$path];
313 echo "CONFLICT $path\n";
314 }
315 $this->conflicts = $conflicts;
316 $this->operation = [$kind, $target];
317 echo match ($kind) {
318 'merge' => 'Automatic merge failed; fix conflicts and then commit the result.',
319 'revert' => "error: could not revert {$target->id}",
320 default => "error: could not apply {$target->id}"
321 }, "\n";
322 }

353 private function status(): void
354 {
355 $head = $this->head();
356 echo $this->branch !== null ? "On branch {$this->branch}\n" : "HEAD detached at {$head->id}\n";
357 if ($this->operation !== null) {
358 $labels = ['merge' => 'Merging', 'revert' => 'Reverting', 'cherry-pick' => 'Cherry-picking'];
359 echo "{$labels[$this->operation[0]]} {$this->operation[1]->id}\n";
360 }
361 $lines = [];
362 $selected = array_unique(array_merge($this->paths($head->tree, $this->index, $this->work), $this->conflicts));
363 sort($selected, SORT_STRING);
364 foreach ($selected as $path) {
365 if (in_array($path, $this->conflicts, true)) {
366 $lines[] = "UU $path";
367 } elseif (isset($this->index[$path])) {
368 $x = !isset($head->tree[$path]) ? 'A' : ($head->tree[$path] !== $this->index[$path] ? 'M' : '.');
369 $y = !isset($this->work[$path]) ? 'D' : ($this->work[$path] !== $this->index[$path] ? 'M' : '.');
370 if ($x !== '.' || $y !== '.') {
371 $lines[] = "$x$y $path";
372 }
373 } else {
374 if (isset($head->tree[$path])) {
375 $lines[] = "D. $path";
376 }
377 if (isset($this->work[$path]) && !$this->ignored($path)) {
378 $lines[] = "?? $path";
379 }
380 }
381 }
382 echo implode("\n", $lines ?: ['nothing to commit, working tree clean']), "\n";
383 }

385 private function rebase(Commit $target): void
386 {
387 $this->checkProgress();
388 if ($this->ancestor($target, $this->head())) {
389 echo "Already up to date\n";
390 return;
391 }
392 $this->checkUncommitted();
393 $list = [];
394 $current = $this->head();
395 while (!$this->ancestor($current, $target)) {
396 $list[] = $current;
397 $current = $current->parents[0];
398 }
399 $list = array_reverse($list);
400 $tree = $target->tree;
401 $actions = [];
402 foreach ($list as $old) {
403 [$result, $conflicts] = $this->mergeTrees($old->parents[0]->tree, $tree, $old->tree);
404 if ($conflicts) {
405 throw new VcsError("could not rebase {$old->id}: conflict in '{$conflicts[0]}'");
406 }
407 if ($this->equalTrees($result, $tree)) {
408 $actions[] = [$old, null];
409 } else {
410 $tree = $result;
411 $actions[] = [$old, $result];
412 }
413 }
414 $this->switchTree($tree);
415 foreach ($actions as [$old, $result]) {
416 if ($result !== null) {
417 $fresh = $this->create([$target], $result, $old->message);
418 echo "pick {$old->id} -> {$fresh->id} {$old->message}\n";
419 $target = $fresh;
420 } else {
421 echo "skip {$old->id} {$old->message}\n";
422 }
423 }
424 $this->moveHead($target);
425 $this->headReport();
426 }

428 private function stashCommand(?string $action): void
429 {
430 if ($action === 'LIST') {
431 if (!$this->stash) {
432 echo "(no stash entries)\n";
433 }
434 foreach ($this->stash as $i => $entry) {
435 echo "stash@{{$i}}: {$entry[0]}\n";
436 }
437 } elseif ($action === 'DROP') {
438 if (!$this->stash) {
439 throw new VcsError('no stash entries');
440 }
441 $entry = array_shift($this->stash);
442 echo "Dropped stash@{0}: {$entry[0]}\n";
443 } elseif ($action === 'POP') {
444 if (!$this->stash) {
445 throw new VcsError('no stash entries');
446 }
447 $this->checkProgress();
448 $this->checkUncommitted();
449 [$label, $base, $saved] = $this->stash[0];
450 [$result, $conflicts] = $this->mergeTrees($base->tree, $this->head()->tree, $saved);
451 if ($conflicts) {
452 throw new VcsError("stash conflict in '{$conflicts[0]}'");
453 }
454 $changed = $this->differences($this->head()->tree, $result);
455 $this->checkOverwrite($changed);
456 foreach ($changed as $path) {
457 $this->matchPath($this->work, $result, $path);
458 }
459 array_shift($this->stash);
460 echo "Restored stash@{0}: $label\n";
461 $this->changes($this->head()->tree, $result);
462 } else {
463 $this->checkProgress();
464 if (!$this->uncommitted()) {
465 echo "No local changes to save\n";
466 return;
467 }
468 $head = $this->head();
469 $selected = $this->paths($this->index, $head->tree);
470 $saved = [];
471 foreach ($selected as $path) {
472 $this->matchPath($saved, $this->work, $path);
473 $this->matchPath($this->work, $head->tree, $path);
474 }
475 $this->index = $head->tree;
476 $location = $this->branch ?? '(no branch)';
477 $label = "WIP on $location: {$head->id} {$head->message}";
478 array_unshift($this->stash, [$label, $head, $saved]);
479 echo "Saved stash@{0}: $label\n";
480 }

492 private function graph(): void
493 {
494 $live = $this->reachable(array_merge([$this->head()], array_values($this->branches), array_values($this->tags)));
495 $ordered = array_values(array_intersect_key($this->commits, $live));
496 usort($ordered, fn($a, $b) => $b->sequence <=> $a->sequence);
497 $lanes = [];
498 foreach ($ordered as $commit) {
499 if (!in_array($commit, $lanes, true)) {
500 $lanes[] = $commit;
501 }
502 $first = array_search($commit, $lanes, true);
503 $symbols = [];
504 foreach ($lanes as $i => $lane) {
505 $symbols[] = $lane === $commit ? ($i === $first ? '*' : '/') : '|';
506 }
507 echo implode(' ', $symbols), ' ', $this->summary($commit), "\n";
508 $kept = [];
509 foreach ($lanes as $i => $lane) {
510 if ($lane !== $commit || $i === $first) {
511 $kept[] = $lane;
512 }
513 }
514 $lanes = $kept;
515 if (!$commit->parents) {
516 array_splice($lanes, $first, 1);
517 } else {
518 $lanes[$first] = $commit->parents[0];
519 $second = $commit->parents[1] ?? null;
520 if ($second !== null && !in_array($second, $lanes, true)) {
521 array_splice($lanes, $first + 1, 0, [$second]);
522 }
523 }
524 }

527 public function execute(string $line): void
528 {
529 [$command, $rest] = array_pad(explode(' ', $line, 2), 2, '');
530 $args = $rest === '' ? [] : explode(' ', $rest);
531 $path = $args[0] ?? null;
532 try {
533 switch ($command) {
534 case 'WRITE':
535 [$path, $text] = explode(' ', $rest, 2);
536 $this->work[$path] = $text;
537 break;
538 case 'REMOVE':
539 if (!isset($this->work[$path])) {
540 throw new VcsError("no such file '$path'");
541 }
542 unset($this->work[$path]);
543 break;
544 case 'ADD':
545 if ($path === 'ALL') {
546 $selected = array_filter($this->paths($this->work, $this->index), fn($p) => !$this->ignored($p));
547 foreach ($selected as $p) {
548 $this->matchPath($this->index, $this->work, $p);
549 }
550 $this->conflicts = [];
551 } else {
552 if (isset($this->work[$path]) && $this->ignored($path)) {
553 throw new VcsError("path '$path' is ignored");
554 }
555 if (!isset($this->work[$path]) && !isset($this->index[$path]) && !in_array($path, $this->conflicts, true)) {
556 $this->pathspec($path);
557 }
558 $this->matchPath($this->index, $this->work, $path);
559 $this->conflicts = array_values(array_diff($this->conflicts, [$path]));
560 }
561 break;
562 case 'RM':
563 if (!isset($this->index[$path])) {
564 $this->pathspec($path);
565 }
566 unset($this->index[$path], $this->work[$path]);
567 $this->conflicts = array_values(array_diff($this->conflicts, [$path]));
568 break;
569 case 'UNSTAGE':
570 if (!isset($this->index[$path]) && !isset($this->head()->tree[$path])) {
571 $this->pathspec($path);
572 }
573 $this->matchPath($this->index, $this->head()->tree, $path);
574 break;
575 case 'RESTORE':
576 if (count($args) === 1) {
577 if (!isset($this->index[$path])) {
578 $this->pathspec($path);
579 }
580 $this->work[$path] = $this->index[$path];
581 } else {
582 $tree = $this->resolve($args[1])->tree;
583 if (!isset($tree[$path])) {
584 $this->missingPath($path, $args[1]);
585 }
586 $this->index[$path] = $tree[$path];
587 $this->work[$path] = $tree[$path];
588 $this->conflicts = array_values(array_diff($this->conflicts, [$path]));
589 }
590 break;
591 case 'IGNORE':
592 $pattern = str_replace(['\\*', '\\?'], ['.*', '.'], preg_quote($path, '~'));
593 $this->ignores[] = '~\A' . $pattern . '\z~';
594 break;
595 case 'COMMIT':
596 if ($this->conflicts) {
597 throw new VcsError('unresolved conflicts');
598 }
599 if ($this->operation !== null && $this->operation[0] === 'merge') {
600 $this->record([$this->head(), $this->operation[1]], $this->index, $rest);
601 $this->operation = null;
602 } elseif ($this->equalTrees($this->index, $this->head()->tree)) {
603 echo "nothing to commit\n";
604 } else {
605 $this->record([$this->head()], $this->index, $rest);
606 $this->operation = null;
607 }
608 break;
609 case 'AMEND':
610 $this->checkProgress();
611 if (!$this->head()->parents) {
612 throw new VcsError('cannot amend the root commit');
613 }
614 $this->record($this->head()->parents, $this->index, $rest);
615 break;
616 case 'STATUS':
617 $this->status();
618 break;
619 case 'LS':
620 case 'CAT':
621 $revision = $command === 'LS' ? ($args[0] ?? null) : ($args[1] ?? null);
622 $tree = $revision !== null ? $this->resolve($revision)->tree : $this->work;
623 if ($command === 'LS') {
624 if (!$tree) {
625 echo "(empty)\n";
626 }
627 foreach ($this->paths($tree) as $p) {
628 echo "$p {$tree[$p]}\n";
629 }
630 } else {
631 if (!isset($tree[$path])) {
632 if ($revision !== null) {
633 $this->missingPath($path, $revision);
634 }
635 throw new VcsError("no such file '$path'");
636 }
637 echo $tree[$path], "\n";
638 }
639 break;
640 case 'DIFF':
641 if (!$args) {
642 $this->changes($this->index, array_intersect_key($this->work, $this->index));
643 } elseif ($path === 'STAGED') {
644 $this->changes($this->head()->tree, $this->index);
645 } else {
646 $a = $this->resolve($path);
647 $b = $this->resolve($args[1]);
648 $this->changes($a->tree, $b->tree);
649 }
650 break;
651 case 'LOG':
652 $current = $this->head();
653 $count = 0;
654 do {
655 echo $this->summary($current), "\n";
656 $count++;
657 if (!$current->parents || ($path !== null && $count === (int)$path)) {
658 break;
659 }
660 $current = $current->parents[0];
661 } while (true);
662 break;
663 case 'SHOW':
664 $commit = $this->resolve($path ?? 'HEAD');
665 echo 'commit ', $this->summary($commit), "\n";
666 echo 'parents: ', $commit->parents ? implode(' ', array_map(fn($p) => $p->id, $commit->parents)) : 'none', "\n";
667 $this->changes($commit->parents[0]->tree ?? [], $commit->tree);
668 break;
669 case 'BLAME':
670 $commit = $this->head();
671 if (!isset($commit->tree[$path])) {
672 $this->missingPath($path, 'HEAD');
673 }
674 $value = $commit->tree[$path];
675 while ($commit->parents && ($commit->parents[0]->tree[$path] ?? null) === $value) {
676 $commit = $commit->parents[0];
677 }
678 echo "{$commit->id} ({$commit->message}) $value\n";
679 break;
680 case 'BRANCH':
681 case 'TAG':
682 $refs =& $this->tags;
683 if ($command === 'BRANCH') {
684 $refs =& $this->branches;
685 }
686 $kind = strtolower($command);
687 if ($path === '-d') {
688 $name = $args[1];
689 if (!isset($refs[$name])) {
690 throw new VcsError("no such $kind '$name'");
691 }
692 if ($command === 'BRANCH') {
693 if ($this->branch === $name) {
694 throw new VcsError("cannot delete current branch '$name'");
695 }
696 if (!$this->ancestor($refs[$name], $this->head())) {
697 throw new VcsError("branch '$name' is not fully merged");
698 }
699 }
700 $old = $refs[$name];
701 unset($refs[$name]);
702 echo "Deleted $kind $name (was {$old->id})\n";
703 } else {
704 if (isset($refs[$path])) {
705 throw new VcsError("$kind '$path' already exists");
706 }
707 $commit = $this->resolve($args[1] ?? 'HEAD');
708 $refs[$path] = $commit;
709 echo "Created $kind $path at {$commit->id}\n";
710 }
711 unset($refs);
712 break;
713 case 'BRANCHES':
714 if ($this->branch === null) {
715 echo "* (HEAD detached at {$this->head()->id})\n";
716 }
717 foreach ($this->paths($this->branches) as $name) {
718 $commit = $this->branches[$name];
719 $mark = $name === $this->branch ? '*' : ' ';
720 echo "$mark $name {$commit->id} {$commit->message}\n";
721 }
722 break;
723 case 'TAGS':
724 if (!$this->tags) {
725 echo "(no tags)\n";
726 }
727 foreach ($this->paths($this->tags) as $name) {
728 $commit = $this->tags[$name];
729 echo "$name {$commit->id} {$commit->message}\n";
730 }
731 break;
732 case 'SWITCH':
733 if (!isset($this->branches[$path])) {
734 throw new VcsError("no such branch '$path'");
735 }
736 if ($this->branch === $path) {
737 echo "Already on '$path'\n";
738 } else {
739 $this->checkProgress();
740 $this->switchTree($this->branches[$path]->tree);
741 $this->branch = $path;
742 echo "Switched to branch '$path'\n";
743 }
744 break;
745 case 'DETACH':
746 $commit = $this->resolve($path);
747 $this->checkProgress();
748 $this->switchTree($commit->tree);
749 $this->branch = null;
750 $this->detached = $commit;
751 $this->headReport();
752 break;
753 case 'MERGE':
754 $target = $this->resolve($path);
755 $this->checkProgress();
756 $old = $this->head();
757 if ($this->ancestor($target, $old)) {
758 echo "Already up to date\n";
759 } elseif ($this->ancestor($old, $target)) {
760 $this->switchTree($target->tree);
761 $this->moveHead($target);
762 echo "Fast-forward {$old->id}..{$target->id}\n";
763 } else {
764 $this->checkUncommitted();
765 $base = $this->mergeBase($old, $target);
766 $this->apply($base->tree, $target->tree, 'merge', $target, "Merge $path", [$old, $target], $base);
767 }
768 break;
769 case 'MERGEBASE':
770 $a = $this->resolve($path);
771 $b = $this->resolve($args[1]);
772 echo $this->mergeBase($a, $b)->id, "\n";
773 break;
774 case 'REBASE':
775 $this->rebase($this->resolve($path));
776 break;
777 case 'RESET':
778 $target = $this->resolve($args[1]);
779 if ($path === 'HARD') {
780 $selected = array_unique(array_merge($this->paths($this->index, $target->tree), $this->conflicts));
781 foreach ($selected as $p) {
782 $this->matchPath($this->work, $target->tree, $p);
783 }
784 }
785 if ($path !== 'SOFT') {
786 $this->index = $target->tree;
787 }
788 $this->moveHead($target);
789 $this->conflicts = [];
790 $this->operation = null;
791 $this->headReport();
792 break;
793 case 'REVERT':
794 case 'CHERRYPICK':
795 $target = $this->resolve($path);
796 $this->checkProgress();
797 $this->checkUncommitted();
798 $parentTree = $target->parents[0]->tree ?? [];
799 if ($command === 'REVERT') {
800 $this->apply($target->tree, $parentTree, 'revert', $target, 'Revert "' . $target->message . '"', [$this->head()]);
801 } else {
802 $this->apply($parentTree, $target->tree, 'cherry-pick', $target, $target->message, [$this->head()]);
803 }
804 break;
805 case 'STASH':
806 $this->stashCommand($path);
807 break;
808 case 'GRAPH':
809 $this->graph();
810 break;
811 case 'DESCRIBE':
812 $commit = $this->resolve($path ?? 'HEAD');
813 $distance = $this->distances($commit);
814 $best = null;
815 $d = PHP_INT_MAX;
816 foreach ($this->paths($this->tags) as $name) {
817 $id = $this->tags[$name]->id;
818 if (isset($distance[$id]) && $distance[$id] < $d) {
819 $best = $name;
820 $d = $distance[$id];
821 }
822 }
823 if ($best === null) {
824 throw new VcsError("no tag can describe '{$commit->id}'");
825 }
826 echo $d === 0 ? $best : "$best-$d-g{$commit->id}", "\n";
827 break;
828 case 'GC':
829 $roots = array_merge([$this->head()], array_values($this->branches), array_values($this->tags));
830 foreach ($this->stash as $entry) {
831 $roots[] = $entry[1];
832 }
833 if ($this->operation !== null) {
834 $roots[] = $this->operation[1];
835 }
836 $live = $this->reachable($roots);
837 $oldSize = count($this->commits);
838 $this->commits = array_intersect_key($this->commits, $live);
839 echo 'Removed unreachable commits: ', $oldSize - count($this->commits), "\n";
840 break;
841 }
842 } catch (VcsError $error) {
843 echo 'error: ', $error->getMessage(), "\n";
844 }
845 }
846}
