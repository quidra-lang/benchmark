3const output = [];
4const commits = new Map();
5const branches = new Map();
6const tags = new Map();
7const stashes = [];
8const patterns = [];
9let sequence = 0;
10let working = new Map();
11let index = new Map();
12let conflicts = new Set();
13let operation = null;
14let branch = 'main';
15let detached;

17function paths(...trees) {
18 return [...new Set(trees.flatMap(tree => [...tree.keys()]))].sort();
19}
20function different(a, b) {
21 return paths(a, b).filter(path => a.get(path) !== b.get(path));
22}
23function equal(a, b) { return different(a, b).length === 0; }
24function match(target, source, path) {
25 if (source.has(path)) target.set(path, source.get(path));
26 else target.delete(path);
27}
28function fail(message) { throw new Error(`error: ${message}`); }
29function create(parents, tree, message) {
30 const s = sequence++;
31 let canonical = `commit ${s}\n`;
32 for (const parent of parents) canonical += `parent ${parent.id}\n`;
33 for (const path of [...tree.keys()].sort()) canonical += `file ${path} ${tree.get(path)}\n`;
34 canonical += `message ${message}\n`;
35 let hash = 0;
36 for (let i = 0; i < canonical.length; i++) hash = (hash * 131 + canonical.charCodeAt(i)) % 2147483647;
37 const commit = {
38 id: hash.toString(16).padStart(8, '0'), sequence: s,
39 parents: [...parents], tree: new Map(tree), message
40 };
41 commits.set(commit.id, commit);
42 return commit;
43}
44const root = create([], new Map(), 'init');
45branches.set('main', root);
46detached = root;
47function head() { return branch === null ? detached : branches.get(branch); }
48function move(commit) {
49 if (branch === null) detached = commit;
50 else branches.set(branch, commit);
51}
52function record(parents, tree, message) {
53 const commit = create(parents, tree, message);
54 const old = parents[0].tree;
55 let added = 0, modified = 0, deleted = 0;
56 for (const path of different(old, tree)) {
57 if (!old.has(path)) added++;
58 else if (!tree.has(path)) deleted++;
59 else modified++;
60 }
61 move(commit);
62 output.push(`[${branch ?? 'detached HEAD'} ${commit.id}] ${message} (${added} added, ${modified} modified, ${deleted} deleted)`);
63 return commit;
64}
65function resolve(rev) {
66 const [anchor, count] = rev.split('~');
67 let commit = anchor === 'HEAD' ? head() : branches.get(anchor) ?? tags.get(anchor) ?? commits.get(anchor);
68 if (!commit) fail(`unknown revision '${rev}'`);
69 for (let i = 0; i < Number(count ?? 0); i++) {
70 if (commit.parents.length === 0) fail(`unknown revision '${rev}'`);
71 commit = commit.parents[0];
72 }
73 return commit;
74}
75function distances(start) {
76 const result = new Map([[start, 0]]);
77 const queue = [start];
78 for (let i = 0; i < queue.length; i++) {
79 const commit = queue[i];
80 for (const parent of commit.parents) if (!result.has(parent)) {
81 result.set(parent, result.get(commit) + 1);
82 queue.push(parent);
83 }
84 }
85 return result;
86}
87function ancestor(a, b) { return distances(b).has(a); }
88function mergeBase(a, b) {
89 const da = distances(a), db = distances(b);
90 const common = [...da.keys()].filter(commit => db.has(commit));
91 const candidates = common.filter(commit => !common.some(other => other !== commit && ancestor(commit, other)));
92 candidates.sort((x, y) => Math.max(da.get(x), db.get(x)) - Math.max(da.get(y), db.get(y)) || y.sequence - x.sequence);
93 return candidates[0];
94}
95function summary(commit) {
96 const refs = [];
97 if (head() === commit) refs.push(branch === null ? 'HEAD' : `HEAD -> ${branch}`);
98 for (const name of [...branches.keys()].sort()) if (name !== branch && branches.get(name) === commit) refs.push(name);
99 for (const name of [...tags.keys()].sort()) if (tags.get(name) === commit) refs.push(`tag: ${name}`);
100 return `${commit.id}${refs.length ? ` (${refs.join(', ')})` : ''} ${commit.message}`;
101}
102function changes(a, b) {
103 const changed = different(a, b);
104 if (!changed.length) output.push('(no changes)');
105 for (const path of changed) {
106 if (!a.has(path)) output.push(`A ${path}: ${b.get(path)}`);
107 else if (!b.has(path)) output.push(`D ${path}: ${a.get(path)}`);
108 else output.push(`M ${path}: ${a.get(path)} -> ${b.get(path)}`);
109 }
110}
111function inProgress() { if (operation) fail(`${operation.kind} in progress`); }
112function dirty() {
113 return !equal(index, head().tree) || [...index.keys()].some(path => working.get(path) !== index.get(path));
114}
115function uncommitted() { if (dirty()) fail('uncommitted changes'); }
116function overwrite(check) {
117 const h = head().tree;
118 for (const path of [...new Set(check)].sort()) {
119 if (index.get(path) !== h.get(path) || working.get(path) !== h.get(path)) {
120 fail(`local changes to '${path}' would be overwritten`);
121 }
122 }
123}
124function switchTree(tree) {
125 const changed = different(head().tree, tree);
126 overwrite(changed);
127 for (const path of changed) { match(index, tree, path); match(working, tree, path); }
128}
129function merge(base, ours, theirs) {
130 const tree = new Map(), found = [], contents = new Map();
131 for (const path of paths(base, ours, theirs)) {
132 const b = base.get(path), o = ours.get(path), t = theirs.get(path);
133 if (t === b || o === t) match(tree, ours, path);
134 else if (o === b) match(tree, theirs, path);
135 else {
136 found.push(path);
137 match(tree, ours, path);
138 contents.set(path, `<<<<<<< ${o ?? '(deleted)'} ======= ${t ?? '(deleted)'} >>>>>>>`);
139 }
140 }
141 return { tree, conflicts: found, contents };
142}
143function apply(base, theirs, kind, target, message, parents, mergeCommit) {
144 const result = merge(base, head().tree, theirs);
145 const changed = different(head().tree, result.tree);
146 overwrite([...changed, ...result.conflicts]);
147 if (kind === 'merge') output.push(`Merge base ${mergeCommit.id}`);
148 if (kind !== 'merge' && !result.conflicts.length && !changed.length) {
149 output.push('nothing to commit');
150 return;
151 }
152 for (const path of changed) { match(index, result.tree, path); match(working, result.tree, path); }
153 if (!result.conflicts.length) { record(parents, result.tree, message); return; }
154 for (const path of result.conflicts) {
155 working.set(path, result.contents.get(path));
156 output.push(`CONFLICT ${path}`);
157 }
158 conflicts = new Set(result.conflicts);
159 operation = { kind, target };
160 output.push(kind === 'merge' ? 'Automatic merge failed; fix conflicts and then commit the result.' :
161 kind === 'revert' ? `error: could not revert ${target.id}` : `error: could not apply ${target.id}`);
162}
163function glob(pattern, path) {
164 let row = Array(path.length + 1).fill(false);
165 row[0] = true;
166 for (const ch of pattern) {
167 const next = Array(path.length + 1).fill(false);
168 if (ch === '*') next[0] = row[0];
169 for (let i = 1; i <= path.length; i++) {
170 next[i] = ch === '*' ? row[i] || next[i - 1] : row[i - 1] && (ch === '?' || ch === path[i - 1]);
171 }
172 row = next;
173 }
174 return row[path.length];
175}
176function ignored(path) {
177 return !index.has(path) && !conflicts.has(path) && patterns.some(pattern => glob(pattern, path));
178}
179function pathspec(path) { return fail(`pathspec '${path}' did not match any files`); }
180function existingPath(tree, path, rev) {
181 if (!tree.has(path)) fail(`path '${path}' does not exist in '${rev}'`);
182 return tree.get(path);
183}
184function reachable(roots) {
185 const result = new Set();
186 for (const commit of roots) for (const item of distances(commit).keys()) result.add(item);
187 return result;
188}
189function execute(line) {
190 const words = line.split(' '), command = words[0], arg = words[1];
191 switch (command) {
192 case 'WRITE': working.set(arg, line.slice(line.indexOf(' ', 6) + 1)); break;
193 case 'REMOVE':
194 if (!working.has(arg)) fail(`no such file '${arg}'`);
195 working.delete(arg);
196 break;
197 case 'ADD':
198 if (arg === 'ALL') {
199 const selected = paths(working, index).filter(path => !ignored(path));
200 for (const path of selected) match(index, working, path);
201 conflicts.clear();
202 } else {
203 if (working.has(arg) && ignored(arg)) fail(`path '${arg}' is ignored`);
204 if (!working.has(arg) && !index.has(arg) && !conflicts.has(arg)) pathspec(arg);
205 match(index, working, arg);
206 conflicts.delete(arg);
207 }
208 break;
209 case 'RM':
210 if (!index.has(arg)) pathspec(arg);
211 index.delete(arg); working.delete(arg); conflicts.delete(arg);
212 break;
213 case 'UNSTAGE':
214 if (!index.has(arg) && !head().tree.has(arg)) pathspec(arg);
215 match(index, head().tree, arg);
216 break;
217 case 'RESTORE':
218 if (words.length === 2) {
219 if (!index.has(arg)) pathspec(arg);
220 working.set(arg, index.get(arg));
221 } else {
222 const tree = resolve(words[2]).tree;
223 const value = existingPath(tree, arg, words[2]);
224 index.set(arg, value); working.set(arg, value); conflicts.delete(arg);
225 }
226 break;
227 case 'IGNORE': patterns.push(arg); break;
228 case 'COMMIT': {
229 if (conflicts.size) fail('unresolved conflicts');
230 const parents = operation?.kind === 'merge' ? [head(), operation.target] : [head()];
231 if (parents.length === 1 && equal(index, head().tree)) { output.push('nothing to commit'); break; }
232 record(parents, index, line.slice(7));
233 operation = null;
234 break;
235 }
236 case 'AMEND':
237 inProgress();
238 if (!head().parents.length) fail('cannot amend the root commit');
239 record(head().parents, index, line.slice(6));
240 break;
241 case 'STATUS': {
242 output.push(branch === null ? `HEAD detached at ${head().id}` : `On branch ${branch}`);
243 if (operation) {
244 const verb = operation.kind === 'merge' ? 'Merging' : operation.kind === 'revert' ? 'Reverting' : 'Cherry-picking';
245 output.push(`${verb} ${operation.target.id}`);
246 }
247 const before = output.length;
248 const all = [...new Set([...paths(head().tree, index, working), ...conflicts])].sort();
249 for (const path of all) {
250 if (conflicts.has(path)) { output.push(`UU ${path}`); continue; }
251 if (index.has(path)) {
252 const x = !head().tree.has(path) ? 'A' : head().tree.get(path) !== index.get(path) ? 'M' : '.';
253 const y = !working.has(path) ? 'D' : working.get(path) !== index.get(path) ? 'M' : '.';
254 if (x !== '.' || y !== '.') output.push(`${x}${y} ${path}`);
255 } else {
256 if (head().tree.has(path)) output.push(`D. ${path}`);
257 if (working.has(path) && !ignored(path)) output.push(`?? ${path}`);
258 }
259 }
260 if (output.length === before) output.push('nothing to commit, working tree clean');
261 break;
262 }
263 case 'LS': {
264 const tree = arg ? resolve(arg).tree : working;
265 if (!tree.size) output.push('(empty)');
266 for (const path of [...tree.keys()].sort()) output.push(`${path} ${tree.get(path)}`);
267 break;
268 }
269 case 'CAT':
270 if (words.length === 3) output.push(existingPath(resolve(words[2]).tree, arg, words[2]));
271 else {
272 if (!working.has(arg)) fail(`no such file '${arg}'`);
273 output.push(working.get(arg));
274 }
275 break;
276 case 'DIFF':
277 if (arg === 'STAGED') changes(head().tree, index);
278 else if (arg) { const a = resolve(arg); const b = resolve(words[2]); changes(a.tree, b.tree); }
279 else changes(index, new Map([...working].filter(([path]) => index.has(path))));
280 break;
281 case 'LOG': {
282 let current = head();
283 for (let i = 0; i < (arg ? Number(arg) : Infinity); i++) {
284 output.push(summary(current));
285 if (!current.parents.length) break;
286 current = current.parents[0];
287 }
288 break;
289 }
290 case 'SHOW': {
291 const commit = resolve(arg ?? 'HEAD');
292 output.push(`commit ${summary(commit)}`, `parents: ${commit.parents.length ? commit.parents.map(parent => parent.id).join(' ') : 'none'}`);
293 changes(commit.parents[0]?.tree ?? new Map(), commit.tree);
294 break;
295 }
296 case 'BLAME': {
297 const value = existingPath(head().tree, arg, 'HEAD');
298 let commit = head();
299 while (commit.parents.length && commit.parents[0].tree.get(arg) === value) commit = commit.parents[0];
300 output.push(`${commit.id} (${commit.message}) ${value}`);
301 break;
302 }
303 case 'BRANCH': case 'TAG': {
304 const refs = command === 'BRANCH' ? branches : tags;
305 const noun = command === 'BRANCH' ? 'branch' : 'tag';
306 if (arg === '-d') {
307 const name = words[2], commit = refs.get(name);
308 if (!commit) fail(`no such ${noun} '${name}'`);
309 if (command === 'BRANCH') {
310 if (branch === name) fail(`cannot delete current branch '${name}'`);
311 if (!ancestor(commit, head())) fail(`branch '${name}' is not fully merged`);
312 }
313 refs.delete(name);
314 output.push(`Deleted ${noun} ${name} (was ${commit.id})`);
315 } else {
316 if (refs.has(arg)) fail(`${noun} '${arg}' already exists`);
317 const commit = resolve(words[2] ?? 'HEAD');
318 refs.set(arg, commit);
319 output.push(`Created ${noun} ${arg} at ${commit.id}`);
320 }
321 break;
322 }
323 case 'BRANCHES':
324 if (branch === null) output.push(`* (HEAD detached at ${head().id})`);
325 for (const name of [...branches.keys()].sort()) {
326 const commit = branches.get(name);
327 output.push(`${branch === name ? '*' : ' '} ${name} ${commit.id} ${commit.message}`);
328 }
329 break;
330 case 'TAGS':
331 if (!tags.size) output.push('(no tags)');
332 for (const name of [...tags.keys()].sort()) {
333 const commit = tags.get(name);
334 output.push(`${name} ${commit.id} ${commit.message}`);
335 }
336 break;
337 case 'SWITCH': {
338 const commit = branches.get(arg);
339 if (!commit) fail(`no such branch '${arg}'`);
340 if (branch === arg) { output.push(`Already on '${arg}'`); break; }
341 inProgress(); switchTree(commit.tree); branch = arg;
342 output.push(`Switched to branch '${arg}'`);
343 break;
344 }
345 case 'DETACH': {
346 const commit = resolve(arg);
347 inProgress(); switchTree(commit.tree); branch = null; detached = commit;
348 output.push(`HEAD is now at ${commit.id} ${commit.message}`);
349 break;
350 }
351 case 'MERGE': {
352 const target = resolve(arg);
353 inProgress();
354 const old = head();
355 if (ancestor(target, old)) { output.push('Already up to date'); break; }
356 if (ancestor(old, target)) {
357 switchTree(target.tree); move(target);
358 output.push(`Fast-forward ${old.id}..${target.id}`);
359 break;
360 }
361 uncommitted();
362 const base = mergeBase(old, target);
363 apply(base.tree, target.tree, 'merge', target, `Merge ${arg}`, [old, target], base);
364 break;
365 }
366 case 'MERGEBASE': {
367 const a = resolve(arg), b = resolve(words[2]);
368 output.push(mergeBase(a, b).id);
369 break;
370 }
371 case 'REBASE': {
372 const onto = resolve(arg);
373 inProgress();
374 if (ancestor(onto, head())) { output.push('Already up to date'); break; }
375 uncommitted();
376 const list = [];
377 let current = head();
378 while (!ancestor(current, onto)) { list.push(current); current = current.parents[0]; }
379 list.reverse();
380 const picks = new Map();
381 let tree = onto.tree;
382 for (const commit of list) {
383 const result = merge(commit.parents[0].tree, tree, commit.tree);
384 if (result.conflicts.length) fail(`could not rebase ${commit.id}: conflict in '${result.conflicts[0]}'`);
385 if (!equal(result.tree, tree)) { picks.set(commit, result.tree); tree = result.tree; }
386 }
387 switchTree(tree);
388 let target = onto;
389 for (const commit of list) {
390 if (picks.has(commit)) {
391 const next = create([target], picks.get(commit), commit.message);
392 output.push(`pick ${commit.id} -> ${next.id} ${commit.message}`);
393 target = next;
394 } else output.push(`skip ${commit.id} ${commit.message}`);
395 }
396 move(target);
397 output.push(`HEAD is now at ${target.id} ${target.message}`);
398 break;
399 }
400 case 'RESET': {
401 const target = resolve(words[2]), tree = target.tree;
402 if (arg === 'HARD') {
403 const all = [...new Set([...paths(index, tree), ...conflicts])];
404 for (const path of all) match(working, tree, path);
405 }
406 if (arg !== 'SOFT') index = new Map(tree);
407 move(target); conflicts.clear(); operation = null;
408 output.push(`HEAD is now at ${target.id} ${target.message}`);
409 break;
410 }
411 case 'REVERT': case 'CHERRYPICK': {
412 const target = resolve(arg);
413 inProgress(); uncommitted();
414 const parentTree = target.parents[0]?.tree ?? new Map();
415 if (command === 'REVERT') apply(target.tree, parentTree, 'revert', target, `Revert "${target.message}"`, [head()]);
416 else apply(parentTree, target.tree, 'cherry-pick', target, target.message, [head()]);
417 break;
418 }
419 case 'STASH':
420 if (arg === 'LIST') {
421 if (!stashes.length) output.push('(no stash entries)');
422 stashes.forEach((stash, i) => output.push(`stash@{${i}}: ${stash.label}`));
423 } else if (arg === 'DROP') {
424 if (!stashes.length) fail('no stash entries');
425 output.push(`Dropped stash@{0}: ${stashes.shift().label}`);
426 } else if (arg === 'POP') {
427 if (!stashes.length) fail('no stash entries');
428 inProgress(); uncommitted();
429 const stash = stashes[0], result = merge(stash.base.tree, head().tree, stash.tree);
430 if (result.conflicts.length) fail(`stash conflict in '${result.conflicts[0]}'`);
431 const changed = different(head().tree, result.tree);
432 overwrite(changed);
433 for (const path of changed) match(working, result.tree, path);
434 stashes.shift();
435 output.push(`Restored stash@{0}: ${stash.label}`);
436 changes(head().tree, result.tree);
437 } else {
438 inProgress();
439 if (!dirty()) { output.push('No local changes to save'); break; }
440 const tree = new Map(), h = head();
441 for (const path of paths(index, h.tree)) {
442 if (working.has(path)) tree.set(path, working.get(path));
443 match(working, h.tree, path);
444 }
445 index = new Map(h.tree);
446 const label = `WIP on ${branch ?? '(no branch)'}: ${h.id} ${h.message}`;
447 stashes.unshift({ label, base: h, tree });
448 output.push(`Saved stash@{0}: ${label}`);
449 }
450 break;
451 case 'GRAPH': {
452 const all = [...reachable([head(), ...branches.values(), ...tags.values()])].sort((a, b) => b.sequence - a.sequence);
453 let lanes = [];
454 for (const commit of all) {
455 if (!lanes.includes(commit)) lanes.push(commit);
456 const k = lanes.indexOf(commit);
457 const symbols = lanes.map((item, i) => i === k ? '*' : item === commit ? '/' : '|');
458 output.push(`${symbols.join(' ')} ${summary(commit)}`);
459 lanes = lanes.filter((item, i) => item !== commit || i === k);
460 if (!commit.parents.length) lanes.splice(k, 1);
461 else {
462 lanes[k] = commit.parents[0];
463 if (commit.parents.length === 2 && !lanes.includes(commit.parents[1])) lanes.splice(k + 1, 0, commit.parents[1]);
464 }
465 }
466 break;
467 }
468 case 'DESCRIBE': {
469 const target = resolve(arg ?? 'HEAD'), distance = distances(target);
470 const names = [...tags.keys()].filter(name => distance.has(tags.get(name)));
471 names.sort((a, b) => distance.get(tags.get(a)) - distance.get(tags.get(b)) || (a < b ? -1 : a > b ? 1 : 0));
472 if (!names.length) fail(`no tag can describe '${target.id}'`);
473 const name = names[0], d = distance.get(tags.get(name));
474 output.push(d === 0 ? name : `${name}-${d}-g${target.id}`);
475 break;
476 }
477 case 'GC': {
478 const roots = [head(), ...branches.values(), ...tags.values(), ...stashes.map(stash => stash.base)];
479 if (operation) roots.push(operation.target);
480 const keep = reachable(roots);
481 let removed = 0;
482 for (const [id, commit] of commits) if (!keep.has(commit)) { commits.delete(id); removed++; }
483 output.push(`Removed unreachable commits: ${removed}`);
484 break;
485 }
486 }
487}
488for (const line of fs.readFileSync(0, 'utf8').split(/\r?\n/)) {
489 if (line === 'EXIT') break;
490 if (!line) continue;
491 try { execute(line); }
492 catch (error) { output.push(error.message); }
493}
494if (output.length) process.stdout.write(output.join('\n') + '\n');
