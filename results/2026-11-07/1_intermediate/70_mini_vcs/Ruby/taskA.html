1class VcsError < StandardError; end
2
3Commit = Struct.new(:id, :seq, :parents, :tree, :message)
4
5class MiniVcs
6 def initialize
7 @commits = {}
8 @next_sequence = 0
9 root = create([], {}, 'init')
10 @branches = { 'main' => root }
11 @tags = {}
12 @branch = 'main'
13 @detached = nil
14 @work = {}
15 @index = {}
16 @conflicts = []
17 @operation = nil
18 @stash = []
19 @ignores = []
20 end
21
22 def head
23 @branch ? @branches[@branch] : @detached
24 end
25
26 def move_head(commit)
27 if @branch
28 @branches[@branch] = commit
29 else
30 @detached = commit
31 end
32 end

34 def paths(*trees)
35 trees.flat_map(&:keys).uniq.sort
36 end
37
38 def differences(a, b)
39 paths(a, b).select { |p| a[p] != b[p] }
40 end
41
42 def match_path(destination, source, path)
43 if source.key?(path)
44 destination[path] = source[path]
45 else
46 destination.delete(path)
47 end
48 end

50 def create(parents, tree, message)
51 seq = @next_sequence
52 text = "commit #{seq}\n"
53 parents.each { |p| text << "parent #{p.id}\n" }
54 tree.keys.sort.each { |p| text << "file #{p} #{tree[p]}\n" }
55 text << "message #{message}\n"
56 hash = text.each_byte.reduce(0) { |h, c| (h * 131 + c) % 2_147_483_647 }
57 commit = Commit.new(format('%08x', hash), seq, parents.dup, tree.dup, message)
58 @commits[commit.id] = commit
59 @next_sequence += 1
60 commit
61 end
62
63 def record(parents, tree, message)
64 old = parents.first.tree
65 added = differences(old, tree).count { |p| !old.key?(p) }
66 deleted = differences(old, tree).count { |p| !tree.key?(p) }
67 modified = differences(old, tree).length - added - deleted
68 commit = create(parents, tree, message)
69 move_head(commit)
70 puts "[#{@branch || 'detached HEAD'} #{commit.id}] #{message} (#{added} added, #{modified} modified, #{deleted} deleted)"
71 end

73 def resolve(revision)
74 anchor, steps = revision.split('~')
75 commit = if anchor == 'HEAD'
76 head
77 else
78 @branches[anchor] || @tags[anchor] || @commits[anchor]
79 end
80 steps.to_i.times { commit = commit&.parents&.first }
81 raise VcsError, "unknown revision '#{revision}'" unless commit
82 commit
83 end

85 def distances(commit)
86 result = { commit.id => 0 }
87 queue = [commit]
88 queue.each do |current|
89 current.parents.each do |parent|
90 next if result.key?(parent.id)
91 result[parent.id] = result[current.id] + 1
92 queue << parent
93 end
94 end
95 result
96 end
97
98 def ancestor?(ancestor, descendant)
99 distances(descendant).key?(ancestor.id)
100 end
101
102 def merge_base(a, b)
103 da = distances(a)
104 db = distances(b)
105 common = (da.keys & db.keys).map { |id| @commits[id] }
106 candidates = common.reject do |x|
107 common.any? { |y| x != y && ancestor?(x, y) }
108 end
109 candidates.min_by { |x| [[da[x.id], db[x.id]].max, -x.seq] }
110 end

112 def summary(commit)
113 refs = []
114 if head == commit
115 refs << (@branch ? "HEAD -> #{@branch}" : 'HEAD')
116 end
117 @branches.keys.sort.each do |name|
118 refs << name if @branches[name] == commit && name != @branch
119 end
120 @tags.keys.sort.each { |name| refs << "tag: #{name}" if @tags[name] == commit }
121 "#{commit.id}#{refs.empty? ? '' : " (#{refs.join(', ')})"} #{commit.message}"
122 end
123
124 def changes(a, b)
125 changed = differences(a, b)
126 puts '(no changes)' if changed.empty?
127 changed.each do |p|
128 if !a.key?(p)
129 puts "A #{p}: #{b[p]}"
130 elsif !b.key?(p)
131 puts "D #{p}: #{a[p]}"
132 else
133 puts "M #{p}: #{a[p]} -> #{b[p]}"
134 end
135 end
136 end
137
138 def check_progress
139 raise VcsError, "#{@operation[0]} in progress" if @operation
140 end
141
142 def uncommitted?
143 @index != head.tree || @index.any? { |p, value| @work[p] != value }
144 end
145
146 def check_uncommitted
147 raise VcsError, 'uncommitted changes' if uncommitted?
148 end
149
150 def check_overwrite(selected)
151 selected.uniq.sort.each do |p|
152 if @index[p] != head.tree[p] || @work[p] != head.tree[p]
153 raise VcsError, "local changes to '#{p}' would be overwritten"
154 end
155 end
156 end
157
158 def switch_tree(tree)
159 changed = differences(head.tree, tree)
160 check_overwrite(changed)
161 changed.each do |p|
162 match_path(@index, tree, p)
163 match_path(@work, tree, p)
164 end
165 end
166
167 def merge_trees(base, ours, theirs)
168 result = {}
169 conflicts = []
170 markers = {}
171 paths(base, ours, theirs).each do |p|
172 if theirs[p] == base[p] || ours[p] == theirs[p]
173 match_path(result, ours, p)
174 elsif ours[p] == base[p]
175 match_path(result, theirs, p)
176 else
177 conflicts << p
178 match_path(result, ours, p)
179 markers[p] = "<<<<<<< #{ours[p] || '(deleted)'} ======= #{theirs[p] || '(deleted)'} >>>>>>>"
180 end
181 end
182 [result, conflicts, markers]
183 end
184
185 def apply(base, theirs, kind, target, message, parents, merge_base = nil)
186 result, conflicts, markers = merge_trees(base, head.tree, theirs)
187 changed = differences(head.tree, result)
188 check_overwrite(changed + conflicts)
189 puts "Merge base #{merge_base.id}" if kind == 'merge'
190 if kind != 'merge' && conflicts.empty? && result == head.tree
191 puts 'nothing to commit'
192 return
193 end
194 changed.each do |p|
195 match_path(@index, result, p)
196 match_path(@work, result, p)
197 end
198 if conflicts.empty?
199 record(parents, result, message)
200 else
201 conflicts.each do |p|
202 @work[p] = markers[p]
203 puts "CONFLICT #{p}"
204 end
205 @conflicts = conflicts
206 @operation = [kind, target]
207 puts case kind
208 when 'merge' then 'Automatic merge failed; fix conflicts and then commit the result.'
209 when 'revert' then "error: could not revert #{target.id}"
210 else "error: could not apply #{target.id}"
211 end
212 end
213 end
214
215 def ignored?(path)
216 !@index.key?(path) && !@conflicts.include?(path) && @ignores.any? { |pattern| pattern.match?(path) }
217 end
218
219 def pathspec(path)
220 raise VcsError, "pathspec '#{path}' did not match any files"
221 end
222
223 def missing_path(path, revision)
224 raise VcsError, "path '#{path}' does not exist in '#{revision}'"
225 end
226
227 def head_report
228 puts "HEAD is now at #{head.id} #{head.message}"
229 end
230
231 def status
232 puts(@branch ? "On branch #{@branch}" : "HEAD detached at #{head.id}")
233 if @operation
234 label = { 'merge' => 'Merging', 'revert' => 'Reverting', 'cherry-pick' => 'Cherry-picking' }
235 puts "#{label[@operation[0]]} #{@operation[1].id}"
236 end
237 lines = []
238 (paths(head.tree, @index, @work) | @conflicts).sort.each do |p|
239 if @conflicts.include?(p)
240 lines << "UU #{p}"
241 elsif @index.key?(p)
242 x = !head.tree.key?(p) ? 'A' : (head.tree[p] != @index[p] ? 'M' : '.')
243 y = !@work.key?(p) ? 'D' : (@work[p] != @index[p] ? 'M' : '.')
244 lines << "#{x}#{y} #{p}" unless x == '.' && y == '.'
245 else
246 lines << "D. #{p}" if head.tree.key?(p)
247 lines << "?? #{p}" if @work.key?(p) && !ignored?(p)
248 end
249 end
250 puts(lines.empty? ? ['nothing to commit, working tree clean'] : lines)
251 end

253 def rebase(target)
254 check_progress
255 if ancestor?(target, head)
256 puts 'Already up to date'
257 return
258 end
259 check_uncommitted
260 list = []
261 current = head
262 until ancestor?(current, target)
263 list << current
264 current = current.parents.first
265 end
266 list.reverse!
267 tree = target.tree
268 actions = list.map do |old|
269 result, conflicts, = merge_trees(old.parents.first.tree, tree, old.tree)
270 unless conflicts.empty?
271 raise VcsError, "could not rebase #{old.id}: conflict in '#{conflicts.first}'"
272 end
273 if result == tree
274 [old, nil]
275 else
276 tree = result
277 [old, result]
278 end
279 end
280 switch_tree(tree)
281 actions.each do |old, result|
282 if result
283 fresh = create([target], result, old.message)
284 puts "pick #{old.id} -> #{fresh.id} #{old.message}"
285 target = fresh
286 else
287 puts "skip #{old.id} #{old.message}"
288 end
289 end
290 move_head(target)
291 head_report
292 end

294 def stash_command(action)
295 if action == 'LIST'
296 puts '(no stash entries)' if @stash.empty?
297 @stash.each_with_index { |entry, i| puts "stash@{#{i}}: #{entry[0]}" }
298 elsif action == 'DROP'
299 raise VcsError, 'no stash entries' if @stash.empty?
300 puts "Dropped stash@{0}: #{@stash.shift[0]}"
301 elsif action == 'POP'
302 raise VcsError, 'no stash entries' if @stash.empty?
303 check_progress
304 check_uncommitted
305 label, base, saved = @stash.first
306 result, conflicts, = merge_trees(base.tree, head.tree, saved)
307 raise VcsError, "stash conflict in '#{conflicts.first}'" unless conflicts.empty?
308 changed = differences(head.tree, result)
309 check_overwrite(changed)
310 changed.each { |p| match_path(@work, result, p) }
311 @stash.shift
312 puts "Restored stash@{0}: #{label}"
313 changes(head.tree, result)
314 else
315 check_progress
316 unless uncommitted?
317 puts 'No local changes to save'
318 return
319 end
320 selected = paths(@index, head.tree)
321 saved = @work.select { |p, _| selected.include?(p) }
322 selected.each { |p| match_path(@work, head.tree, p) }
323 @index = head.tree.dup
324 label = "WIP on #{@branch || '(no branch)'}: #{head.id} #{head.message}"
325 @stash.unshift([label, head, saved])
326 puts "Saved stash@{0}: #{label}"
327 end
328 end

330 def reachable(roots)
331 roots.each_with_object({}) do |root, result|
332 distances(root).each_key { |id| result[id] = true }
333 end
334 end

336 def graph
337 live = reachable([head] + @branches.values + @tags.values)
338 lanes = []
339 live.keys.map { |id| @commits[id] }.sort_by { |c| -c.seq }.each do |commit|
340 lanes << commit unless lanes.include?(commit)
341 first = lanes.index(commit)
342 symbols = lanes.each_with_index.map { |c, i| c == commit ? (i == first ? '*' : '/') : '|' }
343 puts "#{symbols.join(' ')} #{summary(commit)}"
344 lanes = lanes.each_with_index.reject { |c, i| c == commit && i != first }.map(&:first)
345 if commit.parents.empty?
346 lanes.delete_at(first)
347 else
348 lanes[first] = commit.parents.first
349 second = commit.parents[1]
350 lanes.insert(first + 1, second) if second && !lanes.include?(second)
351 end
352 end
353 end

355 def execute(line)
356 command, rest = line.split(' ', 2)
357 args = rest ? rest.split(' ') : []
358 path = args[0]
359 case command
360 when 'WRITE'
361 path, text = rest.split(' ', 2)
362 @work[path] = text
363 when 'REMOVE'
364 raise VcsError, "no such file '#{path}'" unless @work.key?(path)
365 @work.delete(path)
366 when 'ADD'
367 if path == 'ALL'
368 selected = paths(@work, @index).reject { |p| ignored?(p) }
369 selected.each { |p| match_path(@index, @work, p) }
370 @conflicts.clear
371 else
372 raise VcsError, "path '#{path}' is ignored" if @work.key?(path) && ignored?(path)
373 pathspec(path) unless @work.key?(path) || @index.key?(path) || @conflicts.include?(path)
374 match_path(@index, @work, path)
375 @conflicts.delete(path)
376 end
377 when 'RM'
378 pathspec(path) unless @index.key?(path)
379 @index.delete(path)
380 @work.delete(path)
381 @conflicts.delete(path)
382 when 'UNSTAGE'
383 pathspec(path) unless @index.key?(path) || head.tree.key?(path)
384 match_path(@index, head.tree, path)
385 when 'RESTORE'
386 if args.length == 1
387 pathspec(path) unless @index.key?(path)
388 @work[path] = @index[path]
389 else
390 tree = resolve(args[1]).tree
391 missing_path(path, args[1]) unless tree.key?(path)
392 @index[path] = @work[path] = tree[path]
393 @conflicts.delete(path)
394 end
395 when 'IGNORE'
396 regex = path.each_char.map { |c| c == '*' ? '.*' : (c == '?' ? '.' : Regexp.escape(c)) }.join
397 @ignores << Regexp.new("\\A#{regex}\\z")
398 when 'COMMIT'
399 raise VcsError, 'unresolved conflicts' unless @conflicts.empty?
400 if @operation && @operation[0] == 'merge'
401 record([head, @operation[1]], @index, rest)
402 @operation = nil
403 elsif @index == head.tree
404 puts 'nothing to commit'
405 else
406 record([head], @index, rest)
407 @operation = nil
408 end
409 when 'AMEND'
410 check_progress
411 raise VcsError, 'cannot amend the root commit' if head.parents.empty?
412 record(head.parents, @index, rest)
413 when 'STATUS'
414 status
415 when 'LS', 'CAT'
416 revision = command == 'LS' ? args[0] : args[1]
417 tree = revision ? resolve(revision).tree : @work
418 if command == 'LS'
419 puts '(empty)' if tree.empty?
420 tree.keys.sort.each { |p| puts "#{p} #{tree[p]}" }
421 else
422 unless tree.key?(path)
423 missing_path(path, revision) if revision
424 raise VcsError, "no such file '#{path}'"
425 end
426 puts tree[path]
427 end
428 when 'DIFF'
429 if args.empty?
430 changes(@index, @work.select { |p, _| @index.key?(p) })
431 elsif path == 'STAGED'
432 changes(head.tree, @index)
433 else
434 a = resolve(path)
435 b = resolve(args[1])
436 changes(a.tree, b.tree)
437 end
438 when 'LOG'
439 current = head
440 count = 0
441 loop do
442 puts summary(current)
443 count += 1
444 break if current.parents.empty? || (path && count == path.to_i)
445 current = current.parents.first
446 end
447 when 'SHOW'
448 commit = resolve(path || 'HEAD')
449 puts "commit #{summary(commit)}"
450 puts "parents: #{commit.parents.empty? ? 'none' : commit.parents.map(&:id).join(' ')}"
451 changes(commit.parents.empty? ? {} : commit.parents.first.tree, commit.tree)
452 when 'BLAME'
453 missing_path(path, 'HEAD') unless head.tree.key?(path)
454 value = head.tree[path]
455 commit = head
456 commit = commit.parents.first while !commit.parents.empty? && commit.parents.first.tree[path] == value
457 puts "#{commit.id} (#{commit.message}) #{value}"
458 when 'BRANCH', 'TAG'
459 refs = command == 'BRANCH' ? @branches : @tags
460 kind = command.downcase
461 if path == '-d'
462 name = args[1]
463 raise VcsError, "no such #{kind} '#{name}'" unless refs.key?(name)
464 if command == 'BRANCH'
465 raise VcsError, "cannot delete current branch '#{name}'" if @branch == name
466 raise VcsError, "branch '#{name}' is not fully merged" unless ancestor?(refs[name], head)
467 end
468 old = refs.delete(name)
469 puts "Deleted #{kind} #{name} (was #{old.id})"
470 else
471 raise VcsError, "#{kind} '#{path}' already exists" if refs.key?(path)
472 commit = resolve(args[1] || 'HEAD')
473 refs[path] = commit
474 puts "Created #{kind} #{path} at #{commit.id}"
475 end
476 when 'BRANCHES'
477 puts "* (HEAD detached at #{head.id})" unless @branch
478 @branches.keys.sort.each do |name|
479 commit = @branches[name]
480 puts "#{name == @branch ? '*' : ' '} #{name} #{commit.id} #{commit.message}"
481 end
482 when 'TAGS'
483 puts '(no tags)' if @tags.empty?
484 @tags.keys.sort.each do |name|
485 commit = @tags[name]
486 puts "#{name} #{commit.id} #{commit.message}"
487 end
488 when 'SWITCH'
489 raise VcsError, "no such branch '#{path}'" unless @branches.key?(path)
490 if @branch == path
491 puts "Already on '#{path}'"
492 else
493 check_progress
494 switch_tree(@branches[path].tree)
495 @branch = path
496 puts "Switched to branch '#{path}'"
497 end
498 when 'DETACH'
499 commit = resolve(path)
500 check_progress
501 switch_tree(commit.tree)
502 @branch = nil
503 @detached = commit
504 head_report
505 when 'MERGE'
506 target = resolve(path)
507 check_progress
508 old = head
509 if ancestor?(target, old)
510 puts 'Already up to date'
511 elsif ancestor?(old, target)
512 switch_tree(target.tree)
513 move_head(target)
514 puts "Fast-forward #{old.id}..#{target.id}"
515 else
516 check_uncommitted
517 base = merge_base(old, target)
518 apply(base.tree, target.tree, 'merge', target, "Merge #{path}", [old, target], base)
519 end
520 when 'MERGEBASE'
521 a = resolve(path)
522 b = resolve(args[1])
523 puts merge_base(a, b).id
524 when 'REBASE'
525 rebase(resolve(path))
526 when 'RESET'
527 target = resolve(args[1])
528 if path == 'HARD'
529 (paths(@index, target.tree) | @conflicts).each { |p| match_path(@work, target.tree, p) }
530 end
531 @index = target.tree.dup unless path == 'SOFT'
532 move_head(target)
533 @conflicts.clear
534 @operation = nil
535 head_report
536 when 'REVERT', 'CHERRYPICK'
537 target = resolve(path)
538 check_progress
539 check_uncommitted
540 parent_tree = target.parents.empty? ? {} : target.parents.first.tree
541 if command == 'REVERT'
542 apply(target.tree, parent_tree, 'revert', target, "Revert \"#{target.message}\"", [head])
543 else
544 apply(parent_tree, target.tree, 'cherry-pick', target, target.message, [head])
545 end
546 when 'STASH'
547 stash_command(path)
548 when 'GRAPH'
549 graph
550 when 'DESCRIBE'
551 commit = resolve(path || 'HEAD')
552 distance = distances(commit)
553 candidates = @tags.keys.select { |name| distance.key?(@tags[name].id) }
554 raise VcsError, "no tag can describe '#{commit.id}'" if candidates.empty?
555 name = candidates.min_by { |tag| [distance[@tags[tag].id], tag] }
556 d = distance[@tags[name].id]
557 puts(d.zero? ? name : "#{name}-#{d}-g#{commit.id}")
558 when 'GC'
559 roots = [head] + @branches.values + @tags.values + @stash.map { |entry| entry[1] }
560 roots << @operation[1] if @operation
561 live = reachable(roots)
562 old_size = @commits.length
563 @commits.select! { |id, _| live.key?(id) }
564 puts "Removed unreachable commits: #{old_size - @commits.length}"
565 end
566 rescue VcsError => error
567 puts "error: #{error.message}"
568 end
569end

572STDIN.each_line do |line|
573 line = line.chomp
574 break if line == 'EXIT'
575 vcs.execute(line)
