1private typealias Tree = Map<String, String>
2private class Problem(val text: String) : RuntimeException(text)
3private fun fail(reason: String): Nothing = throw Problem("error: $reason")
4private data class Commit(val id: String, val parents: List<Int>, val tree: Tree, val message: String)
5private data class Stash(val label: String, val base: Int, val tree: Tree)
6private fun paths(vararg trees: Tree) = trees.flatMap {
7 it.keys
8}.toSortedSet()
9private fun differences(a: Tree, b: Tree) = paths(a,b).filter {
10 a[it] != b[it]
11}.toSortedSet()
12private fun matchPath(target: MutableMap<String,String>, source: Tree, path: String) {
13 val value = source[path]
14 if (value == null) target.remove(path) else target[path] = value
15}
16private fun changes(a: Tree, b: Tree): List<String> = differences(a,b).map {
17 p -> when {
18 p !in a -> "A $p: ${b[p]}"
19 p !in b -> "D $p: ${a[p]}"
20 else -> "M $p: ${a[p]} -> ${b[p]}"
21 }
22}.ifEmpty {
23 listOf("(no changes)")
24}
25private fun threeWay(base: Tree, ours: Tree, theirs: Tree): Pair<Tree,Map<String,String>> {
26 val merged = sortedMapOf<String,String>()
27 val conflicts = sortedMapOf<String,String>()
28 for (p in paths(base,ours,theirs)) {
29 val b = base[p]
30 val o = ours[p]
31 val t = theirs[p]
32 if (t == b || o == t) matchPath(merged,ours,p)
33 else if (o == b) matchPath(merged,theirs,p)
34 else {
35 matchPath(merged,ours,p)
36 conflicts[p] = "<<<<<<< ${o ?: "(deleted)"} ======= ${t ?: "(deleted)"} >>>>>>>"
37 }
38 }
39 return merged to conflicts
40}
41private fun glob(pattern: String, path: String): Boolean {
42 var previous = BooleanArray(path.length+1)
43 previous[0] = true
44 for (c in pattern) {
45 val next = BooleanArray(path.length+1)
46 if (c == '*') next[0] = previous[0]
47 for (i in path.indices) next[i+1] = if (c == '*') previous[i+1] || next[i] else previous[i] && (c == '?' || c == path[i])
48 previous = next
49 }
50 return previous[path.length]
51}
52private class Repository {
53 val commits = mutableListOf<Commit?>()
54 val branches = sortedMapOf<String,Int>()
55 val tags = sortedMapOf<String,Int>()
56 var head = 0
57 var attached: String? = "main"
58 val work = sortedMapOf<String,String>()
59 var index = mutableMapOf<String,String>()
60 val conflicts = sortedSetOf<String>()
61 var pending: Pair<String,Int>? = null
62 val stashes = mutableListOf<Stash>()
63 val ignores = mutableListOf<String>()
64 init {
65 create(emptyList(),emptyMap(),"init")
66 branches["main"] = 0
67 }
68 fun commit(i: Int) = commits[i]!!
69 fun h() = commit(head).tree
70 fun create(parents: List<Int>, tree: Tree, message: String): Int {
71 val sequence = commits.size
72 val text = buildString {
73 append("commit $sequence\n")
74 for (p in parents) append("parent ${commit(p).id}\n")
75 for ((p,v) in tree.toSortedMap()) append("file $p $v\n")
76 append("message $message\n")
77 }
78 var hash = 0L
79 for (c in text) hash = (hash*131+c.code)%2147483647
80 commits.add(Commit(hash.toString(16).padStart(8,'0'),parents.toList(),tree.toMap(),message))
81 return sequence
82 }
83 fun moveHead(target: Int) {
84 head = target
85 attached?.let {
86 branches[it] = target
87 }
88 }
89 fun record(parents: List<Int>, tree: Tree, message: String): String {
90 val old = commit(parents[0]).tree
91 val counts = IntArray(3)
92 for (p in differences(old,tree)) counts[if (p !in old) 0 else if (p !in tree) 2 else 1]++
93 val target = create(parents,tree,message)
94 moveHead(target)
95 return "[${attached ?: "detached HEAD"} ${commit(target).id}] $message (${counts[0]} added, ${counts[1]} modified, ${counts[2]} deleted)"
96 }
97 fun resolve(revision: String): Int {
98 val anchor = revision.substringBefore('~')
99 val count = if ('~' in revision) revision.substringAfter('~').toInt() else 0
100 var target: Int? = if (anchor == "HEAD") head else branches[anchor] ?: tags[anchor] ?: commits.indexOfFirst {
101 it?.id == anchor
102 }.takeIf {
103 it >= 0
104 }
105 repeat(count) {
106 target = target?.let {
107 commit(it).parents.firstOrNull()
108 }
109 }
110 return target ?: fail("unknown revision '$revision'")
111 }
112 fun distances(start: Int): Map<Int,Int> {
113 val result = mutableMapOf(start to 0)
114 val queue = ArrayDeque<Int>()
115 queue.addLast(start)
116 while (queue.isNotEmpty()) {
117 val i = queue.removeFirst()
118 for (p in commit(i).parents) if (p !in result) {
119 result[p] = result.getValue(i)+1
120 queue.addLast(p)
121 }
122 }
123 return result
124 }
125 fun ancestor(a: Int, b: Int) = a in distances(b)
126 fun mergeBase(x: Int, y: Int): Int {
127 val a = distances(x)
128 val b = distances(y)
129 val common = a.keys.intersect(b.keys)
130 return common.filter {
131 z -> common.none {
132 w -> z != w && ancestor(z,w)
133 }
134 }.minWith(compareBy<Int> {
135 maxOf(a.getValue(it),b.getValue(it))
136 }.thenByDescending {
137 it
138 })
139 }
140 fun inProgress() {
141 pending?.let {
142 fail("${it.first} in progress")
143 }
144 }
145 fun uncommitted() = index != h() || index.keys.any {
146 work[it] != index[it]
147 }
148 fun clean() {
149 if (uncommitted()) fail("uncommitted changes")
150 }
151 fun overwrite(selected: Set<String>) {
152 for (p in selected.sorted()) if (index[p] != h()[p] || work[p] != h()[p]) fail("local changes to '$p' would be overwritten")
153 }
154 fun switchTree(tree: Tree) {
155 val changed = differences(h(),tree)
156 overwrite(changed)
157 for (p in changed) {
158 matchPath(work,tree,p)
159 matchPath(index,tree,p)
160 }
161 }
162 fun ignored(p: String) = p !in index && p !in conflicts && ignores.any {
163 glob(it,p)
164 }
165 fun summary(i: Int): String {
166 val refs = mutableListOf<String>()
167 if (head == i) refs.add(attached?.let {
168 "HEAD -> $it"
169 } ?: "HEAD")
170 for ((b,target) in branches) if (target == i && attached != b) refs.add(b)
171 for ((tag,target) in tags) if (target == i) refs.add("tag: $tag")
172 return commit(i).id+(if (refs.isEmpty()) "" else " (${refs.joinToString(", ")})")+" "+commit(i).message
173 }
174 fun apply(base: Tree, theirs: Tree, kind: String, target: Int, message: String, parents: List<Int>, mergeBase: Int? = null): List<String> {
175 val (merged,clashes) = threeWay(base,h(),theirs)
176 overwrite(differences(h(),merged)+clashes.keys)
177 val out = mutableListOf<String>()
178 if (mergeBase != null) out.add("Merge base ${commit(mergeBase).id}")
179 if (kind != "merge" && clashes.isEmpty() && merged == h()) return out+"nothing to commit"
180 for (p in differences(h(),merged)) {
181 matchPath(work,merged,p)
182 matchPath(index,merged,p)
183 }
184 if (clashes.isEmpty()) out.add(record(parents,merged,message))
185 else {
186 for ((p,value) in clashes) {
187 work[p] = value
188 out.add("CONFLICT $p")
189 }
190 conflicts.clear()
191 conflicts.addAll(clashes.keys)
192 pending = kind to target
193 out.add(if (kind == "merge") "Automatic merge failed; fix conflicts and then commit the result." else "error: could not ${if (kind == "revert") "revert" else "apply"} ${commit(target).id}")
194 }
195 return out
196 }
197 fun reachable(includeSaved: Boolean): Set<Int> {
198 val starts = mutableListOf(head)
199 starts.addAll(branches.values)
200 starts.addAll(tags.values)
201 if (includeSaved) {
202 starts.addAll(stashes.map {
203 it.base
204 })
205 pending?.let {
206 starts.add(it.second)
207 }
208 }
209 return starts.flatMap {
210 distances(it).keys
211 }.toSet()
212 }
213 fun command(line: String): List<String> {
214 val words = line.split(' ')
215 val c = words[0]
216 val a = words.drop(1)
217 val out = mutableListOf<String>()
218 when(c) {
219 "WRITE" -> work[a[0]] = line.split(' ',limit=3)[2]
220 "REMOVE" -> if (work.remove(a[0]) == null) fail("no such file '${a[0]}'")
221 "ADD" -> if (a[0] == "ALL") {
222 val selected = paths(work,index).filter {
223 !ignored(it)
224 }
225 for (p in selected) matchPath(index,work,p)
226 conflicts.clear()
227 }
228 else {
229 val p = a[0]
230 if (p in work && ignored(p)) fail("path '$p' is ignored")
231 if (p !in work && p !in index && p !in conflicts) fail("pathspec '$p' did not match any files")
232 matchPath(index,work,p)
233 conflicts.remove(p)
234 }
235 "RM" -> {
236 if (index.remove(a[0]) == null) fail("pathspec '${a[0]}' did not match any files")
237 work.remove(a[0])
238 conflicts.remove(a[0])
239 }
240 "UNSTAGE" -> {
241 if (a[0] !in index && a[0] !in h()) fail("pathspec '${a[0]}' did not match any files")
242 matchPath(index,h(),a[0])
243 }
244 "RESTORE" -> if (a.size == 1) {
245 val value = index[a[0]] ?: fail("pathspec '${a[0]}' did not match any files")
246 work[a[0]] = value
247 }
248 else {
249 val target = resolve(a[1])
250 val value = commit(target).tree[a[0]] ?: fail("path '${a[0]}' does not exist in '${a[1]}'")
251 work[a[0]] = value
252 index[a[0]] = value
253 conflicts.remove(a[0])
254 }
255 "IGNORE" -> ignores.add(a[0])
256 "COMMIT","AMEND" -> {
257 val message = line.substringAfter(' ')
258 val parents = if (c == "AMEND") {
259 inProgress()
260 if (commit(head).parents.isEmpty()) fail("cannot amend the root commit")
261 commit(head).parents
262 }
263 else {
264 if (conflicts.isNotEmpty()) fail("unresolved conflicts")
265 if (pending?.first == "merge") listOf(head,pending!!.second) else listOf(head)
266 }
267 if (c == "COMMIT" && parents.size != 2 && index == h()) out.add("nothing to commit") else {
268 out.add(record(parents,index,message))
269 pending = null
270 }
271 }
272 "STATUS" -> {
273 out.add(attached?.let {
274 "On branch $it"
275 } ?: "HEAD detached at ${commit(head).id}")
276 pending?.let {
277 (kind,i) -> out.add("${when(kind) { "merge" -> "Merging"; "revert" -> "Reverting"; else -> "Cherry-picking" }} ${commit(i).id}")
278 }
279 val start = out.size
280 for (p in (paths(h(),index,work)+conflicts).sorted()) {
281 if (p in conflicts) out.add("UU $p")
282 else if (p in index) {
283 val x = if (p !in h()) 'A' else if (h()[p] != index[p]) 'M' else '.'
284 val y = if (p !in work) 'D' else if (work[p] != index[p]) 'M' else '.'
285 if (x != '.' || y != '.') out.add("$x$y $p")
286 }
287 else {
288 if (p in h()) out.add("D. $p")
289 if (p in work && !ignored(p)) out.add("?? $p")
290 }
291 }
292 if (out.size == start) out.add("nothing to commit, working tree clean")
293 }
294 "LS","CAT" -> {
295 val revision = if (c == "LS") a.firstOrNull() else a.getOrNull(1)
296 val tree = revision?.let {
297 commit(resolve(it)).tree
298 } ?: work
299 if (c == "CAT") out.add(tree[a[0]] ?: if (revision == null) fail("no such file '${a[0]}'") else fail("path '${a[0]}' does not exist in '$revision'"))
300 else out.addAll(tree.toSortedMap().map {
301 (p,v) -> "$p $v"
302 }.ifEmpty {
303 listOf("(empty)")
304 })
305 }
306 "DIFF" -> out.addAll(if (a.isEmpty()) changes(index,work.filterKeys {
307 it in index
308 }) else if (a[0] == "STAGED") changes(h(),index) else {
309 val x = resolve(a[0])
310 val y = resolve(a[1])
311 changes(commit(x).tree,commit(y).tree)
312 })
313 "LOG" -> {
314 val limit = a.firstOrNull()?.toInt() ?: Int.MAX_VALUE
315 var current = head
316 var count = 0
317 while (count < limit) {
318 out.add(summary(current))
319 count++
320 current = commit(current).parents.firstOrNull() ?: break
321 }
322 }
323 "SHOW" -> {
324 val target = resolve(a.firstOrNull() ?: "HEAD")
325 val original = commit(target)
326 out.add("commit ${summary(target)}")
327 out.add("parents: ${original.parents.map { commit(it).id }.joinToString(" ").ifEmpty { "none" }}")
328 out.addAll(changes(original.parents.firstOrNull()?.let {
329 commit(it).tree
330 } ?: emptyMap(),original.tree))
331 }
332 "BLAME" -> {
333 val value = h()[a[0]] ?: fail("path '${a[0]}' does not exist in 'HEAD'")
334 var i = head
335 while (commit(i).parents.isNotEmpty()) {
336 val p = commit(i).parents[0]
337 if (commit(p).tree[a[0]] != value) break
338 i = p
339 }
340 out.add("${commit(i).id} (${commit(i).message}) $value")
341 }
342 "BRANCH","TAG" -> {
343 val branch = c == "BRANCH"
344 val noun = if (branch) "branch" else "tag"
345 val refs = if (branch) branches else tags
346 if (a[0] == "-d") {
347 val name = a[1]
348 val target = refs[name] ?: fail("no such $noun '$name'")
349 if (branch && attached == name) fail("cannot delete current branch '$name'")
350 if (branch && !ancestor(target,head)) fail("branch '$name' is not fully merged")
351 refs.remove(name)
352 out.add("Deleted $noun $name (was ${commit(target).id})")
353 }
354 else {
355 val name = a[0]
356 if (name in refs) fail("$noun '$name' already exists")
357 val target = resolve(a.getOrNull(1) ?: "HEAD")
358 refs[name] = target
359 out.add("Created $noun $name at ${commit(target).id}")
360 }
361 }
362 "BRANCHES" -> {
363 if (attached == null) out.add("* (HEAD detached at ${commit(head).id})")
364 for ((b,i) in branches) out.add("${if (attached == b) "*" else " "} $b ${commit(i).id} ${commit(i).message}")
365 }
366 "TAGS" -> out.addAll(tags.map {
367 (tag,i) -> "$tag ${commit(i).id} ${commit(i).message}"
368 }.ifEmpty {
369 listOf("(no tags)")
370 })
371 "SWITCH" -> {
372 val target = branches[a[0]] ?: fail("no such branch '${a[0]}'")
373 if (attached == a[0]) out.add("Already on '${a[0]}'") else {
374 inProgress()
375 switchTree(commit(target).tree)
376 attached = a[0]
377 head = target
378 out.add("Switched to branch '${a[0]}'")
379 }
380 }
381 "DETACH" -> {
382 val target = resolve(a[0])
383 inProgress()
384 switchTree(commit(target).tree)
385 attached = null
386 head = target
387 out.add("HEAD is now at ${commit(target).id} ${commit(target).message}")
388 }
389 "MERGE" -> {
390 val target = resolve(a[0])
391 inProgress()
392 if (ancestor(target,head)) out.add("Already up to date") else if (ancestor(head,target)) {
393 val old = head
394 switchTree(commit(target).tree)
395 moveHead(target)
396 out.add("Fast-forward ${commit(old).id}..${commit(target).id}")
397 } else {
398 clean()
399 val base = mergeBase(head,target)
400 out.addAll(apply(commit(base).tree,commit(target).tree,"merge",target,"Merge ${a[0]}",listOf(head,target),base))
401 }
402 }
403 "MERGEBASE" -> {
404 val x = resolve(a[0])
405 val y = resolve(a[1])
406 out.add(commit(mergeBase(x,y)).id)
407 }
408 "REBASE" -> {
409 val target = resolve(a[0])
410 inProgress()
411 if (ancestor(target,head)) return listOf("Already up to date")
412 clean()
413 val list = mutableListOf<Int>()
414 var current = head
415 while (!ancestor(current,target)) {
416 list.add(current)
417 current = commit(current).parents[0]
418 }
419 list.reverse()
420 var tree = commit(target).tree
421 val planned = mutableListOf<Tree?>()
422 for (i in list) {
423 val original = commit(i)
424 val (merged,clashes) = threeWay(commit(original.parents[0]).tree,tree,original.tree)
425 if (clashes.isNotEmpty()) fail("could not rebase ${original.id}: conflict in '${clashes.keys.first()}'")
426 if (merged == tree) planned.add(null) else {
427 planned.add(merged)
428 tree = merged
429 }
430 }
431 switchTree(tree)
432 var parent = target
433 for ((position,i) in list.withIndex()) {
434 val original = commit(i)
435 val picked = planned[position]
436 if (picked == null) out.add("skip ${original.id} ${original.message}") else {
437 val next = create(listOf(parent),picked,original.message)
438 out.add("pick ${original.id} -> ${commit(next).id} ${original.message}")
439 parent = next
440 }
441 }
442 moveHead(parent)
443 out.add("HEAD is now at ${commit(parent).id} ${commit(parent).message}")
444 }
445 "RESET" -> {
446 val target = resolve(a[1])
447 val tree = commit(target).tree
448 if (a[0] == "HARD") for (p in paths(index,tree)+conflicts) matchPath(work,tree,p)
449 if (a[0] != "SOFT") index = tree.toMutableMap()
450 moveHead(target)
451 conflicts.clear()
452 pending = null
453 out.add("HEAD is now at ${commit(target).id} ${commit(target).message}")
454 }
455 "REVERT","CHERRYPICK" -> {
456 val target = resolve(a[0])
457 inProgress()
458 clean()
459 val original = commit(target)
460 val parent = original.parents.firstOrNull()?.let {
461 commit(it).tree
462 } ?: emptyMap()
463 if (c == "REVERT") out.addAll(apply(original.tree,parent,"revert",target,"Revert \"${original.message}\"",listOf(head))) else out.addAll(apply(parent,original.tree,"cherry-pick",target,original.message,listOf(head)))
464 }
465 "STASH" -> {
466 if (a.isEmpty()) {
467 inProgress()
468 if (!uncommitted()) out.add("No local changes to save") else {
469 val all = paths(index,h())
470 val saved = work.filterKeys {
471 it in all
472 }.toMap()
473 val tree = h()
474 for (p in all) matchPath(work,tree,p)
475 index = tree.toMutableMap()
476 val label = "WIP on ${attached ?: "(no branch)"}: ${commit(head).id} ${commit(head).message}"
477 stashes.add(0,Stash(label,head,saved))
478 out.add("Saved stash@{0}: $label")
479 }
480 }
481 else if (a[0] == "LIST") out.addAll(stashes.mapIndexed {
482 i,s -> "stash@{$i}: ${s.label}"
483 }.ifEmpty {
484 listOf("(no stash entries)")
485 })
486 else {
487 val stash = stashes.firstOrNull() ?: fail("no stash entries")
488 if (a[0] == "DROP") {
489 stashes.removeAt(0)
490 out.add("Dropped stash@{0}: ${stash.label}")
491 } else {
492 inProgress()
493 clean()
494 val (merged,clashes) = threeWay(commit(stash.base).tree,h(),stash.tree)
495 if (clashes.isNotEmpty()) fail("stash conflict in '${clashes.keys.first()}'")
496 val changed = differences(h(),merged)
497 overwrite(changed)
498 for (p in changed) matchPath(work,merged,p)
499 stashes.removeAt(0)
500 out.add("Restored stash@{0}: ${stash.label}")
501 out.addAll(changes(h(),merged))
502 }
503 }
504 }
505 "GRAPH" -> {
506 val lanes = mutableListOf<Int>()
507 for (i in reachable(false).sortedDescending()) {
508 if (i !in lanes) lanes.add(i)
509 val k = lanes.indexOf(i)
510 out.add(lanes.mapIndexed {
511 j,v -> if (j == k) "*" else if (v == i) "/" else "|"
512 }.joinToString(" ")+" "+summary(i))
513 for (j in lanes.lastIndex downTo k+1) if (lanes[j] == i) lanes.removeAt(j)
514 val parents = commit(i).parents
515 if (parents.isEmpty()) lanes.removeAt(k) else {
516 lanes[k] = parents[0]
517 if (parents.size == 2 && parents[1] !in lanes) lanes.add(k+1,parents[1])
518 }
519 }
520 }
521 "DESCRIBE" -> {
522 val target = resolve(a.firstOrNull() ?: "HEAD")
523 val distance = distances(target)
524 val choice = tags.mapNotNull {
525 (tag,i) -> distance[i]?.let {
526 it to tag
527 }
528 }.minWithOrNull(compareBy<Pair<Int,String>> {
529 it.first
530 }.thenBy {
531 it.second
532 }) ?: fail("no tag can describe '${commit(target).id}'")
533 out.add(if (choice.first == 0) choice.second else "${choice.second}-${choice.first}-g${commit(target).id}")
534 }
535 "GC" -> {
536 val keep = reachable(true)
537 var count = 0
538 for (i in commits.indices) if (commits[i] != null && i !in keep) {
539 commits[i] = null
540 count++
541 }
542 out.add("Removed unreachable commits: $count")
543 }
544 }
545 return out
546 }
547}
548fun main() {
549 val repository = Repository()
550 while (true) {
551 val line = readLine() ?: break
552 if (line == "EXIT") break
553 try {
554 repository.command(line).forEach(::println)
555 } catch (p: Problem) {
556 println(p.text)
557 }
558 }
559}
