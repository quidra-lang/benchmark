60 static SortedSet<string> Paths(params Tree[] trees)
61 {
62 SortedSet<string> result = new(StringComparer.Ordinal);
63 foreach (Tree tree in trees) result.UnionWith(tree.Keys);
64 return result;

67 static bool Agree(Tree left, Tree right, string path) => left.GetValueOrDefault(path) == right.GetValueOrDefault(path);
68
69 static SortedSet<string> Differences(Tree left, Tree right)
70 {
71 var result = Paths(left, right);
72 result.RemoveWhere(path => Agree(left, right, path));
73 return result;
74 }

76 static bool Equal(Tree left, Tree right) => left.Count == right.Count && left.All(entry => right.GetValueOrDefault(entry.Key) == entry.Value);

78 static void Match(Tree destination, Tree source, string path)
79 {
80 if (source.TryGetValue(path, out string? text)) destination[path] = text;
81 else destination.Remove(path);
82 }
83
84 static void Replace(Tree destination, Tree source)
85 {
86 destination.Clear();
87 foreach (var entry in source) destination[entry.Key] = entry.Value;

90 static Commit Create(List<Commit> parents, Tree tree, string message)
91 {
92 int sequence = nextSequence++;
93 StringBuilder text = new($"commit {sequence}\n");
94 foreach (Commit parent in parents) text.Append("parent ").Append(parent.Id).Append('\n');
95 foreach (var entry in tree) text.Append("file ").Append(entry.Key).Append(' ').Append(entry.Value).Append('\n');
96 text.Append("message ").Append(message).Append('\n');
97 long hash = 0;
98 foreach (char c in text.ToString()) hash = (hash * 131 + c) % 2147483647;
99 Commit commit = new(hash.ToString("x8", CultureInfo.InvariantCulture), sequence, parents, tree, message);
100 Commits.Add(commit.Id, commit);
101 return commit;
102 }
103
104 static void Record(List<Commit> parents, Tree tree, string message)
105 {
106 Commit commit = Create(parents, tree, message);
107 Tree before = parents[0].Files;
108 int added = 0, modified = 0, deleted = 0;
109 foreach (string path in Differences(before, tree))
110 {
111 if (!before.ContainsKey(path)) added++;
112 else if (!tree.ContainsKey(path)) deleted++;
113 else modified++;
114 }
115 MoveHead(commit);
116 Console.WriteLine($"[{branch ?? "detached HEAD"} {commit.Id}] {message} ({added} added, {modified} modified, {deleted} deleted)");
117 }

119 static Commit Resolve(string revision)
120 {
121 string[] parts = revision.Split('~');
122 string anchor = parts[0];
123 Commit? commit = anchor == "HEAD" ? Head : Branches.GetValueOrDefault(anchor) ?? Tags.GetValueOrDefault(anchor) ?? Commits.GetValueOrDefault(anchor);
124 int steps = parts.Length == 1 ? 0 : int.Parse(parts[1]);
125 while (commit is not null && steps-- > 0) commit = commit.Parents.Count == 0 ? null : commit.Parents[0];
126 Require(commit is not null, $"error: unknown revision '{revision}'");
127 return commit!;
128 }

130 static Dictionary<Commit, int> Distances(Commit start)
131 {
132 Dictionary<Commit, int> result = new() { [start] = 0 };
133 Queue<Commit> queue = new();
134 queue.Enqueue(start);
135 while (queue.Count > 0)
136 {
137 Commit commit = queue.Dequeue();
138 foreach (Commit parent in commit.Parents) if (result.TryAdd(parent, result[commit] + 1)) queue.Enqueue(parent);
139 }
140 return result;
141 }
142
143 static bool Ancestor(Commit ancestor, Commit descendant) => Distances(descendant).ContainsKey(ancestor);
144
145 static Commit MergeBase(Commit left, Commit right)
146 {
147 var a = Distances(left);
148 var b = Distances(right);
149 var common = a.Keys.Where(b.ContainsKey).ToList();
150 Commit? best = null;
151 int smallest = int.MaxValue;
152 foreach (Commit candidate in common)
153 {
154 if (common.Any(other => other != candidate && Ancestor(candidate, other))) continue;
155 int distance = Math.Max(a[candidate], b[candidate]);
156 if (best is null || distance < smallest || distance == smallest && candidate.Sequence > best.Sequence)
157 {
158 best = candidate;
159 smallest = distance;
160 }
161 }
162 return best!;
163 }

165 static string Summary(Commit commit)
166 {
167 List<string> refs = [];
168 if (Head == commit) refs.Add(branch is null ? "HEAD" : "HEAD -> " + branch);
169 foreach (var entry in Branches) if (entry.Value == commit && entry.Key != branch) refs.Add(entry.Key);
170 foreach (var entry in Tags) if (entry.Value == commit) refs.Add("tag: " + entry.Key);
171 return commit.Id + (refs.Count == 0 ? "" : " (" + string.Join(", ", refs) + ")") + " " + commit.Message;
172 }

174 static void Changes(Tree before, Tree after)
175 {
176 var paths = Differences(before, after);
177 if (paths.Count == 0) Console.WriteLine("(no changes)");
178 foreach (string path in paths) Console.WriteLine(!before.ContainsKey(path) ? $"A {path}: {after[path]}" : !after.ContainsKey(path) ? $"D {path}: {before[path]}" : $"M {path}: {before[path]} -> {after[path]}");
179 }

181 static void ProgressCheck() => Require(operation is null, $"error: {operation} in progress");
182
183 static bool Dirty() => !Equal(Index, Head.Files) || Index.Keys.Any(path => !Agree(Working, Index, path));
184
185 static void CleanCheck() => Require(!Dirty(), "error: uncommitted changes");
186
187 static void OverwriteCheck(IEnumerable<string> paths)
188 {
189 foreach (string path in paths.OrderBy(candidate => candidate, StringComparer.Ordinal)) Require(Agree(Index, Head.Files, path) && Agree(Working, Head.Files, path), $"error: local changes to '{path}' would be overwritten");
190 }
191
192 static void SwitchTree(Tree tree)
193 {
194 var paths = Differences(Head.Files, tree);
195 OverwriteCheck(paths);
196 foreach (string path in paths)
197 {
198 Match(Index, tree, path);
199 Match(Working, tree, path);
200 }
201 }

203 static MergeResult Merge(Tree baseTree, Tree ours, Tree theirs)
204 {
205 Tree result = new();
206 SortedSet<string> problems = new(StringComparer.Ordinal);
207 Dictionary<string, string> contents = new(StringComparer.Ordinal);
208 foreach (string path in Paths(baseTree, ours, theirs)) if (Agree(theirs, baseTree, path) || Agree(ours, theirs, path)) Match(result, ours, path);
209 else if (Agree(ours, baseTree, path)) Match(result, theirs, path);
210 else
211 {
212 problems.Add(path);
213 Match(result, ours, path);
214 contents[path] = "<<<<<<< " + ours.GetValueOrDefault(path, "(deleted)") + " ======= " + theirs.GetValueOrDefault(path, "(deleted)") + " >>>>>>>";
215 }
216 return new(result, problems, contents);
217 }

219 static void Apply(Tree baseTree, Tree theirs, string kind, Commit chosen, string message, List<Commit> parents, Commit? mergeBase)
220 {
221 MergeResult result = Merge(baseTree, Head.Files, theirs);
222 var changed = Differences(Head.Files, result.Files);
223 SortedSet<string> checkedPaths = new(changed, StringComparer.Ordinal);
224 checkedPaths.UnionWith(result.Conflicts);
225 OverwriteCheck(checkedPaths);
226 if (kind == "merge") Console.WriteLine("Merge base " + mergeBase!.Id);
227 else if (result.Conflicts.Count == 0 && changed.Count == 0)
228 {
229 Console.WriteLine("nothing to commit");
230 return;
231 }
232 foreach (string path in changed)
233 {
234 Match(Index, result.Files, path);
235 Match(Working, result.Files, path);
236 }
237 if (result.Conflicts.Count == 0) Record(parents, result.Files, message);
238 else
239 {
240 foreach (string path in result.Conflicts)
241 {
242 Working[path] = result.Contents[path];
243 Console.WriteLine("CONFLICT " + path);
244 }
245 Conflicts.Clear();
246 Conflicts.UnionWith(result.Conflicts);
247 operation = kind;
248 target = chosen;
249 Console.WriteLine(kind == "merge" ? "Automatic merge failed; fix conflicts and then commit the result." : kind == "revert" ? "error: could not revert " + chosen.Id : "error: could not apply " + chosen.Id);
250 }
251 }

253 static bool Glob(string pattern, string path)
254 {
255 bool[,] dp = new bool[pattern.Length + 1, path.Length + 1];
256 dp[0, 0] = true;
257 for (int i = 0; i < pattern.Length; i++) for (int j = 0; j <= path.Length; j++) if (pattern[i] == '*') dp[i + 1, j] = dp[i, j] || j > 0 && dp[i + 1, j - 1];
258 else if (j > 0) dp[i + 1, j] = dp[i, j - 1] && (pattern[i] == '?' || pattern[i] == path[j - 1]);
259 return dp[pattern.Length, path.Length];
260 }
261
262 static bool Ignored(string path) => !Index.ContainsKey(path) && !Conflicts.Contains(path) && Ignores.Any(pattern => Glob(pattern, path));
263
264 static void Pathspec(bool valid, string path) => Require(valid, $"error: pathspec '{path}' did not match any files");
265
266 static HashSet<Commit> Reachable(bool extra)
267 {
268 HashSet<Commit> result = [];
269 List<Commit> roots = [.. Branches.Values, .. Tags.Values, Head];
270 if (extra)
271 {
272 roots.AddRange(Stashes.Select(stash => stash.Base));
273 if (operation is not null) roots.Add(target!);
274 }
275 foreach (Commit root in roots) result.UnionWith(Distances(root).Keys);
276 return result;
277 }

279 static void Command(string line)
280 {
281 string[] words = line.Split(' ');
282 string command = words[0], path = words.Length > 1 ? words[1] : "";
283 switch (command)
284 {
285 case "WRITE":
286 Working[path] = line[(line.IndexOf(' ', line.IndexOf(' ') + 1) + 1)..];
287 break;
288 case "REMOVE":
289 Require(Working.ContainsKey(path), $"error: no such file '{path}'");
290 Working.Remove(path);
291 break;
292 case "IGNORE":
293 Ignores.Add(path);
294 break;
295 case "ADD":
296 if (path == "ALL")
297 {
298 var selected = Paths(Working, Index).Where(entry => !Ignored(entry)).ToList();
299 foreach (string entry in selected) Match(Index, Working, entry);
300 Conflicts.Clear();
301 }
302 else
303 {
304 Require(!Working.ContainsKey(path) || !Ignored(path), $"error: path '{path}' is ignored");
305 Pathspec(Working.ContainsKey(path) || Index.ContainsKey(path) || Conflicts.Contains(path), path);
306 Match(Index, Working, path);
307 Conflicts.Remove(path);
308 }
309 break;
310 case "RM":
311 Pathspec(Index.ContainsKey(path), path);
312 Index.Remove(path);
313 Working.Remove(path);
314 Conflicts.Remove(path);
315 break;
316 case "UNSTAGE":
317 Pathspec(Index.ContainsKey(path) || Head.Files.ContainsKey(path), path);
318 Match(Index, Head.Files, path);
319 break;
320 case "RESTORE":
321 if (words.Length == 2)
322 {
323 Pathspec(Index.ContainsKey(path), path);
324 Working[path] = Index[path];
325 }
326 else
327 {
328 Commit chosen = Resolve(words[2]);
329 Require(chosen.Files.ContainsKey(path), $"error: path '{path}' does not exist in '{words[2]}'");
330 Match(Index, chosen.Files, path);
331 Match(Working, chosen.Files, path);
332 Conflicts.Remove(path);
333 }
334 break;
335 case "COMMIT":
336 Require(Conflicts.Count == 0, "error: unresolved conflicts");
337 if (operation != "merge" && Equal(Index, Head.Files))
338 {
339 Console.WriteLine("nothing to commit");
340 break;
341 }
342 Record(operation == "merge" ? [Head, target!] : [Head], Index, line[7..]);
343 operation = null;
344 target = null;
345 break;
346 case "AMEND":
347 ProgressCheck();
348 Require(Head.Parents.Count > 0, "error: cannot amend the root commit");
349 Record(Head.Parents, Index, line[6..]);
350 break;
351 case "STATUS":
352 {
353 Console.WriteLine(branch is null ? "HEAD detached at " + Head.Id : "On branch " + branch);
354 if (operation is not null) Console.WriteLine((operation == "merge" ? "Merging " : operation == "revert" ? "Reverting " : "Cherry-picking ") + target!.Id);
355 var all = Paths(Head.Files, Index, Working);
356 all.UnionWith(Conflicts);
357 bool any = false;
358 foreach (string entry in all) if (Conflicts.Contains(entry))
359 {
360 Console.WriteLine("UU " + entry);
361 any = true;
362 }
363 else if (Index.ContainsKey(entry))
364 {
365 char x = !Head.Files.ContainsKey(entry) ? 'A' : Agree(Head.Files, Index, entry) ? '.' : 'M';
366 char y = !Working.ContainsKey(entry) ? 'D' : Agree(Working, Index, entry) ? '.' : 'M';
367 if (x != '.' || y != '.')
368 {
369 Console.WriteLine($"{x}{y} {entry}");
370 any = true;
371 }
372 }
373 else
374 {
375 if (Head.Files.ContainsKey(entry))
376 {
377 Console.WriteLine("D. " + entry);
378 any = true;
379 }
380 if (Working.ContainsKey(entry) && !Ignored(entry))
381 {
382 Console.WriteLine("?? " + entry);
383 any = true;
384 }
385 }
386 if (!any) Console.WriteLine("nothing to commit, working tree clean");
387 break;
388 }
389 case "LS":
390 {
391 Tree tree = words.Length == 1 ? Working : Resolve(path).Files;
392 if (tree.Count == 0) Console.WriteLine("(empty)");
393 foreach (var entry in tree) Console.WriteLine(entry.Key + " " + entry.Value);
394 break;
395 }
396 case "CAT":
397 {
398 Tree tree = words.Length == 2 ? Working : Resolve(words[2]).Files;
399 Require(tree.ContainsKey(path), words.Length == 2 ? $"error: no such file '{path}'" : $"error: path '{path}' does not exist in '{words[2]}'");
400 Console.WriteLine(tree[path]);
401 break;
402 }
403 case "DIFF":
404 if (words.Length == 1)
405 {
406 Tree tracked = new();
407 foreach (string entry in Index.Keys) if (Working.ContainsKey(entry)) tracked[entry] = Working[entry];
408 Changes(Index, tracked);
409 }
410 else if (words.Length == 2) Changes(Head.Files, Index);
411 else
412 {
413 Commit left = Resolve(path), right = Resolve(words[2]);
414 Changes(left.Files, right.Files);
415 }
416 break;
417 case "LOG":
418 {
419 int limit = words.Length == 1 ? int.MaxValue : int.Parse(path);
420 Commit chosen = Head;
421 while (limit-- > 0)
422 {
423 Console.WriteLine(Summary(chosen));
424 if (chosen.Parents.Count == 0) break;
425 chosen = chosen.Parents[0];
426 }
427 break;
428 }
429 case "SHOW":
430 {
431 Commit chosen = Resolve(words.Length == 1 ? "HEAD" : path);
432 Console.WriteLine("commit " + Summary(chosen));
433 Console.WriteLine("parents: " + (chosen.Parents.Count == 0 ? "none" : string.Join(" ", chosen.Parents.Select(parent => parent.Id))));
434 Changes(chosen.Parents.Count == 0 ? new() : chosen.Parents[0].Files, chosen.Files);
435 break;
436 }
437 case "BLAME":
438 {
439 Require(Head.Files.ContainsKey(path), $"error: path '{path}' does not exist in 'HEAD'");
440 Commit chosen = Head;
441 string text = chosen.Files[path];
442 while (chosen.Parents.Count > 0 && chosen.Parents[0].Files.GetValueOrDefault(path) == text) chosen = chosen.Parents[0];
443 Console.WriteLine($"{chosen.Id} ({chosen.Message}) {text}");
444 break;
445 }
446 case "BRANCH":
447 case "TAG":
448 {
449 bool tag = command == "TAG";
450 var refs = tag ? Tags : Branches;
451 string noun = tag ? "tag" : "branch";
452 if (path == "-d")
453 {
454 string name = words[2];
455 Require(refs.ContainsKey(name), $"error: no such {noun} '{name}'");
456 Commit chosen = refs[name];
457 if (!tag)
458 {
459 Require(name != branch, $"error: cannot delete current branch '{name}'");
460 Require(Ancestor(chosen, Head), $"error: branch '{name}' is not fully merged");
461 }
462 refs.Remove(name);
463 Console.WriteLine($"Deleted {noun} {name} (was {chosen.Id})");
464 }
465 else
466 {
467 Require(!refs.ContainsKey(path), $"error: {noun} '{path}' already exists");
468 Commit chosen = Resolve(words.Length == 2 ? "HEAD" : words[2]);
469 refs[path] = chosen;
470 Console.WriteLine($"Created {noun} {path} at {chosen.Id}");
471 }
472 break;
473 }
474 case "BRANCHES":
475 if (branch is null) Console.WriteLine($"* (HEAD detached at {Head.Id})");
476 foreach (var entry in Branches) Console.WriteLine((entry.Key == branch ? "* " : " ") + entry.Key + " " + entry.Value.Id + " " + entry.Value.Message);
477 break;
478 case "TAGS":
479 if (Tags.Count == 0) Console.WriteLine("(no tags)");
480 foreach (var entry in Tags) Console.WriteLine(entry.Key + " " + entry.Value.Id + " " + entry.Value.Message);
481 break;
482 case "SWITCH":
483 Require(Branches.ContainsKey(path), $"error: no such branch '{path}'");
484 if (path == branch)
485 {
486 Console.WriteLine($"Already on '{path}'");
487 break;
488 }
489 ProgressCheck();
490 SwitchTree(Branches[path].Files);
491 branch = path;
492 Console.WriteLine($"Switched to branch '{path}'");
493 break;
494 case "DETACH":
495 {
496 Commit chosen = Resolve(path);
497 ProgressCheck();
498 SwitchTree(chosen.Files);
499 branch = null;
500 detached = chosen;
501 Console.WriteLine($"HEAD is now at {chosen.Id} {chosen.Message}");
502 break;
503 }
504 case "MERGEBASE":
505 {
506 Commit left = Resolve(path), right = Resolve(words[2]);
507 Console.WriteLine(MergeBase(left, right).Id);
508 break;
509 }
510 case "MERGE":
511 {
512 Commit chosen = Resolve(path), before = Head;
513 ProgressCheck();
514 if (Ancestor(chosen, before))
515 {
516 Console.WriteLine("Already up to date");
517 break;
518 }
519 if (Ancestor(before, chosen))
520 {
521 SwitchTree(chosen.Files);
522 MoveHead(chosen);
523 Console.WriteLine($"Fast-forward {before.Id}..{chosen.Id}");
524 break;
525 }
526 CleanCheck();
527 Commit baseCommit = MergeBase(before, chosen);
528 Apply(baseCommit.Files, chosen.Files, "merge", chosen, "Merge " + path, [before, chosen], baseCommit);
529 break;
530 }
531 case "RESET":
532 {
533 Commit chosen = Resolve(words[2]);
534 if (path == "HARD")
535 {
536 var all = Paths(Index, chosen.Files);
537 all.UnionWith(Conflicts);
538 foreach (string entry in all) Match(Working, chosen.Files, entry);
539 }
540 if (path != "SOFT") Replace(Index, chosen.Files);
541 MoveHead(chosen);
542 Conflicts.Clear();
543 operation = null;
544 target = null;
545 Console.WriteLine($"HEAD is now at {chosen.Id} {chosen.Message}");
546 break;
547 }
548 case "REVERT":
549 case "CHERRYPICK":
550 {
551 Commit chosen = Resolve(path);
552 ProgressCheck();
553 CleanCheck();
554 Tree parent = chosen.Parents.Count == 0 ? new() : chosen.Parents[0].Files;
555 bool revert = command == "REVERT";
556 Apply(revert ? chosen.Files : parent, revert ? parent : chosen.Files, revert ? "revert" : "cherry-pick", chosen, revert ? "Revert \"" + chosen.Message + "\"" : chosen.Message, [Head], null);
557 break;
558 }
559 case "REBASE":
560 {
561 Commit upstream = Resolve(path);
562 ProgressCheck();
563 if (Ancestor(upstream, Head))
564 {
565 Console.WriteLine("Already up to date");
566 break;
567 }
568 CleanCheck();
569 List<Commit> old = [];
570 Commit cursor = Head;
571 while (!Ancestor(cursor, upstream))
572 {
573 old.Add(cursor);
574 cursor = cursor.Parents[0];
575 }
576 old.Reverse();
577 List<Replay> replay = [];
578 Tree tree = new(upstream.Files);
579 foreach (Commit chosen in old)
580 {
581 MergeResult result = Merge(chosen.Parents[0].Files, tree, chosen.Files);
582 Require(result.Conflicts.Count == 0, $"error: could not rebase {chosen.Id}: conflict in '{result.Conflicts.FirstOrDefault()}'");
583 replay.Add(new(chosen, result.Files, !Equal(tree, result.Files)));
584 tree = result.Files;
585 }
586 SwitchTree(tree);
587 Commit current = upstream;
588 foreach (Replay item in replay) if (item.Pick)
589 {
590 Commit created = Create([current], item.Files, item.Original.Message);
591 Console.WriteLine($"pick {item.Original.Id} -> {created.Id} {created.Message}");
592 current = created;
593 }
594 else Console.WriteLine($"skip {item.Original.Id} {item.Original.Message}");
595 MoveHead(current);
596 Console.WriteLine($"HEAD is now at {current.Id} {current.Message}");
597 break;
598 }
599 case "STASH":
600 {
601 if (words.Length > 1 && path == "LIST")
602 {
603 if (Stashes.Count == 0) Console.WriteLine("(no stash entries)");
604 for (int i = 0; i < Stashes.Count; i++) Console.WriteLine($"stash@{{{i}}}: {Stashes[i].Label}");
605 break;
606 }
607 if (words.Length > 1)
608 {
609 Require(Stashes.Count > 0, "error: no stash entries");
610 Stash stash = Stashes[0];
611 if (path == "DROP")
612 {
613 Stashes.RemoveAt(0);
614 Console.WriteLine("Dropped stash@{0}: " + stash.Label);
615 break;
616 }
617 ProgressCheck();
618 CleanCheck();
619 MergeResult result = Merge(stash.Base.Files, Head.Files, stash.Files);
620 Require(result.Conflicts.Count == 0, $"error: stash conflict in '{result.Conflicts.FirstOrDefault()}'");
621 var changed = Differences(Head.Files, result.Files);
622 OverwriteCheck(changed);
623 foreach (string entry in changed) Match(Working, result.Files, entry);
624 Stashes.RemoveAt(0);
625 Console.WriteLine("Restored stash@{0}: " + stash.Label);
626 Changes(Head.Files, result.Files);
627 }
628 else
629 {
630 ProgressCheck();
631 if (!Dirty())
632 {
633 Console.WriteLine("No local changes to save");
634 break;
635 }
636 Tree saved = new();
637 var tracked = Paths(Index, Head.Files);
638 foreach (string entry in tracked) if (Working.ContainsKey(entry)) saved[entry] = Working[entry];
639 foreach (string entry in tracked) Match(Working, Head.Files, entry);
640 Replace(Index, Head.Files);
641 string label = $"WIP on {branch ?? "(no branch)"}: {Head.Id} {Head.Message}";
642 Stashes.Insert(0, new(label, Head, saved));
643 Console.WriteLine("Saved stash@{0}: " + label);
644 }
645 break;
646 }
647 case "GRAPH":
648 {
649 List<Commit> lanes = [];
650 foreach (Commit chosen in Reachable(false).OrderByDescending(commit => commit.Sequence))
651 {
652 if (!lanes.Contains(chosen)) lanes.Add(chosen);
653 int first = lanes.IndexOf(chosen);
654 Console.WriteLine(string.Join(" ", lanes.Select((commit, i) => i == first ? "*" : commit == chosen ? "/" : "|")) + " " + Summary(chosen));
655 for (int i = lanes.Count - 1; i > first; i--) if (lanes[i] == chosen) lanes.RemoveAt(i);
656 if (chosen.Parents.Count == 0) lanes.RemoveAt(first);
657 else
658 {
659 lanes[first] = chosen.Parents[0];
660 if (chosen.Parents.Count == 2 && !lanes.Contains(chosen.Parents[1])) lanes.Insert(first + 1, chosen.Parents[1]);
661 }
662 }
663 break;
664 }
665 case "DESCRIBE":
666 {
667 Commit chosen = Resolve(words.Length == 1 ? "HEAD" : path);
668 var distances = Distances(chosen);
669 string? best = null;
670 int smallest = int.MaxValue;
671 foreach (var entry in Tags) if (distances.TryGetValue(entry.Value, out int distance) && distance < smallest)
672 {
673 best = entry.Key;
674 smallest = distance;
675 }
676 Require(best is not null, $"error: no tag can describe '{chosen.Id}'");
677 Console.WriteLine(best + (smallest == 0 ? "" : $"-{smallest}-g{chosen.Id}"));
678 break;
679 }
680 case "GC":
681 {
682 var keep = Reachable(true);
683 var removed = Commits.Where(entry => !keep.Contains(entry.Value)).Select(entry => entry.Key).ToList();
684 foreach (string id in removed) Commits.Remove(id);
685 Console.WriteLine($"Removed unreachable commits: {removed.Count}");
686 break;
687 }
688 }
689 }
