1use std::collections::{BTreeMap, BTreeSet, VecDeque};
2use std::io::{self, Read};
3type Tree = BTreeMap<String, String>;
4#[derive(Clone)]
5struct Commit {
6 id: String, parents: Vec<usize>, tree: Tree, message: String
7}
8#[derive(Clone)]
9struct Stash {
10 label: String, base: usize, tree: Tree
11}
12struct Repository {
13 commits: Vec<Option<Commit>>, branches: BTreeMap<String, usize>, tags: BTreeMap<String, usize>,
14 head: usize, attached: Option<String>, work: Tree, index: Tree, conflicts: BTreeSet<String>,
15 pending: Option<(String, usize)>, stashes: Vec<Stash>, ignores: Vec<String>,
16}
17fn paths(trees: &[&Tree]) -> BTreeSet<String> {
18 trees.iter().flat_map(|t| t.keys().cloned()).collect()
19}
20fn differences(a: &Tree, b: &Tree) -> BTreeSet<String> {
21 paths(&[a,b]).into_iter().filter(|p| a.get(p) != b.get(p)).collect()
22}
23fn match_path(target: &mut Tree, source: &Tree, p: &str) {
24 if let Some(value) = source.get(p) {
25 target.insert(p.into(),value.clone());
26 } else {
27 target.remove(p);
28 }
29}
30fn changes(a: &Tree, b: &Tree) -> Vec<String> {
31 let result: Vec<_> = differences(a,b).iter().map(|p| match (a.get(p),b.get(p)) {
32 (None,Some(v)) => format!("A {p}: {v}"), (Some(v),None) => format!("D {p}: {v}"),
33 (Some(x),Some(y)) => format!("M {p}: {x} -> {y}"), _ => unreachable!(),
34 }).collect();
35 if result.is_empty() {
36 vec!["(no changes)".into()]
37 } else {
38 result
39 }
40}
41fn three_way(base: &Tree, ours: &Tree, theirs: &Tree) -> (Tree,BTreeMap<String,String>) {
42 let mut merged = Tree::new();
43 let mut conflicts = BTreeMap::new();
44 for p in paths(&[base,ours,theirs]) {
45 let b = base.get(&p);
46 let o = ours.get(&p);
47 let t = theirs.get(&p);
48 if t == b || o == t {
49 match_path(&mut merged,ours,&p);
50 }
51 else if o == b {
52 match_path(&mut merged,theirs,&p);
53 }
54 else {
55 match_path(&mut merged,ours,&p);
56 conflicts.insert(p,format!("<<<<<<< {} ======= {} >>>>>>>",o.map_or("(deleted)",String::as_str),t.map_or("(deleted)",String::as_str)));
57 }
58 }
59 (merged,conflicts)
60}
61fn glob(pattern: &str, path: &str) -> bool {
62 let mut previous = vec![false; path.len()+1];
63 previous[0] = true;
64 for c in pattern.bytes() {
65 let mut next = vec![false; path.len()+1];
66 if c == b'*' {
67 next[0] = previous[0];
68 }
69 for (i,b) in path.bytes().enumerate() {
70 next[i+1] = if c == b'*' {
71 previous[i+1] || next[i]
72 } else {
73 previous[i] && (c == b'?' || c == b)
74 };
75 }
76 previous = next;
77 }
78 previous[path.len()]
79}
80impl Repository {
81 fn new() -> Self {
82 let mut repo = Self {
83 commits: Vec::new(), branches: BTreeMap::new(), tags: BTreeMap::new(), head: 0, attached: Some("main".into()), work: Tree::new(), index: Tree::new(), conflicts: BTreeSet::new(), pending: None, stashes: Vec::new(), ignores: Vec::new()
84 };
85 repo.create(Vec::new(),Tree::new(),"init".into());
86 repo.branches.insert("main".into(),0);
87 repo
88 }
89 fn commit(&self, i: usize) -> &Commit {
90 self.commits[i].as_ref().unwrap()
91 }
92 fn h(&self) -> &Tree {
93 &self.commit(self.head).tree
94 }
95 fn create(&mut self, parents: Vec<usize>, tree: Tree, message: String) -> usize {
96 let sequence = self.commits.len();
97 let mut text = format!("commit {sequence}\n");
98 for &p in &parents {
99 text += &format!("parent {}\n",self.commit(p).id);
100 }
101 for (p,v) in &tree {
102 text += &format!("file {p} {v}\n");
103 }
104 text += &format!("message {message}\n");
105 let hash = text.bytes().fold(0u64,|h,c| (h*131+c as u64)%2147483647);
106 self.commits.push(Some(Commit {
107 id: format!("{hash:08x}"), parents, tree, message
108 }));
109 sequence
110 }
111 fn move_head(&mut self, target: usize) {
112 self.head = target;
113 if let Some(branch) = &self.attached {
114 self.branches.insert(branch.clone(),target);
115 }
116 }
117 fn record(&mut self, parents: Vec<usize>, tree: Tree, message: String) -> String {
118 let old = &self.commit(parents[0]).tree;
119 let mut counts = [0;3];
120 for p in differences(old,&tree) {
121 counts[if !old.contains_key(&p) {
122 0
123 } else if !tree.contains_key(&p) {
124 2
125 } else {
126 1
127 }] += 1;
128 }
129 let target = self.create(parents,tree,message.clone());
130 self.move_head(target);
131 format!("[{} {}] {message} ({} added, {} modified, {} deleted)",self.attached.as_deref().unwrap_or("detached HEAD"),self.commit(target).id,counts[0],counts[1],counts[2])
132 }
133 fn resolve(&self, revision: &str) -> Result<usize,String> {
134 let (anchor,count) = revision.split_once('~').map_or((revision,0),|(a,n)|(a,n.parse::<usize>().unwrap()));
135 let mut result = if anchor == "HEAD" {
136 Some(self.head)
137 } else {
138 self.branches.get(anchor).or_else(||self.tags.get(anchor)).copied().or_else(||self.commits.iter().position(|c|c.as_ref().is_some_and(|c|c.id == anchor)))
139 };
140 for _ in 0..count {
141 result = result.and_then(|i|self.commit(i).parents.first().copied());
142 }
143 result.ok_or_else(||format!("error: unknown revision '{revision}'"))
144 }
145 fn distances(&self, start: usize) -> BTreeMap<usize,usize> {
146 let mut result = BTreeMap::from([(start,0)]);
147 let mut queue = VecDeque::from([start]);
148 while let Some(i) = queue.pop_front() {
149 let d = result[&i]+1;
150 for &p in &self.commit(i).parents {
151 if let std::collections::btree_map::Entry::Vacant(entry) = result.entry(p) {
152 entry.insert(d);
153 queue.push_back(p);
154 }
155 }
156 }
157 result
158 }
159 fn ancestor(&self, a: usize, b: usize) -> bool {
160 self.distances(b).contains_key(&a)
161 }
162 fn merge_base(&self, x: usize, y: usize) -> usize {
163 let a = self.distances(x);
164 let b = self.distances(y);
165 let common: Vec<_> = a.keys().filter(|c|b.contains_key(*c)).copied().collect();
166 common.iter().copied().filter(|&z|!common.iter().any(|&w|z != w && self.ancestor(z,w))).min_by_key(|&z|(a[&z].max(b[&z]),std::cmp::Reverse(z))).unwrap()
167 }
168 fn in_progress(&self) -> Result<(),String> {
169 if let Some((kind,_)) = &self.pending {
170 Err(format!("error: {kind} in progress"))
171 } else {
172 Ok(())
173 }
174 }
175 fn uncommitted(&self) -> bool {
176 self.index != *self.h() || self.index.keys().any(|p| self.work.get(p) != self.index.get(p))
177 }
178 fn clean(&self) -> Result<(),String> {
179 if self.uncommitted() {
180 Err("error: uncommitted changes".into())
181 } else {
182 Ok(())
183 }
184 }
185 fn overwrite(&self, paths: &BTreeSet<String>) -> Result<(),String> {
186 for p in paths {
187 if self.index.get(p) != self.h().get(p) || self.work.get(p) != self.h().get(p) {
188 return Err(format!("error: local changes to '{p}' would be overwritten"));
189 }
190 } Ok(())
191 }
192 fn switch_tree(&mut self, tree: &Tree) -> Result<(),String> {
193 let changed = differences(self.h(),tree);
194 self.overwrite(&changed)?;
195 for p in changed {
196 match_path(&mut self.work,tree,&p);
197 match_path(&mut self.index,tree,&p);
198 } Ok(())
199 }
200 fn ignored(&self, p: &str) -> bool {
201 !self.index.contains_key(p) && !self.conflicts.contains(p) && self.ignores.iter().any(|pattern|glob(pattern,p))
202 }
203 fn summary(&self, i: usize) -> String {
204 let commit = self.commit(i);
205 let mut references = Vec::new();
206 if self.head == i {
207 references.push(self.attached.as_ref().map_or("HEAD".into(),|b|format!("HEAD -> {b}")));
208 }
209 for (b,&target) in &self.branches {
210 if target == i && self.attached.as_ref() != Some(b) {
211 references.push(b.clone());
212 }
213 }
214 for (tag,&target) in &self.tags {
215 if target == i {
216 references.push(format!("tag: {tag}"));
217 }
218 }
219 format!("{}{} {}",commit.id,if references.is_empty() {
220 String::new()
221 } else {
222 format!(" ({})",references.join(", "))
223 },commit.message)
224 }
225 fn apply(&mut self, base: &Tree, theirs: &Tree, kind: &str, target: usize, message: String, parents: Vec<usize>, merge_base: Option<usize>) -> Result<Vec<String>,String> {
226 let (merged,conflicts) = three_way(base,self.h(),theirs);
227 let mut changed = differences(self.h(),&merged);
228 changed.extend(conflicts.keys().cloned());
229 self.overwrite(&changed)?;
230 let mut out = Vec::new();
231 if let Some(base) = merge_base {
232 out.push(format!("Merge base {}",self.commit(base).id));
233 }
234 if kind != "merge" && conflicts.is_empty() && merged == *self.h() {
235 out.push("nothing to commit".into());
236 return Ok(out);
237 }
238 for p in differences(self.h(),&merged) {
239 match_path(&mut self.work,&merged,&p);
240 match_path(&mut self.index,&merged,&p);
241 }
242 if conflicts.is_empty() {
243 out.push(self.record(parents,merged,message));
244 }
245 else {
246 for (p,value) in &conflicts {
247 self.work.insert(p.clone(),value.clone());
248 out.push(format!("CONFLICT {p}"));
249 }
250 self.conflicts = conflicts.into_keys().collect();
251 self.pending = Some((kind.into(),target));
252 out.push(if kind == "merge" {
253 "Automatic merge failed; fix conflicts and then commit the result.".into()
254 } else {
255 format!("error: could not {} {}",if kind == "revert" {
256 "revert"
257 } else {
258 "apply"
259 },self.commit(target).id)
260 });
261 }
262 Ok(out)
263 }
264 fn reachable(&self, include_saved: bool) -> BTreeSet<usize> {
265 let mut starts = vec![self.head];
266 starts.extend(self.branches.values().copied());
267 starts.extend(self.tags.values().copied());
268 if include_saved {
269 starts.extend(self.stashes.iter().map(|s|s.base));
270 if let Some((_,i)) = &self.pending {
271 starts.push(*i);
272 }
273 }
274 starts.into_iter().flat_map(|i|self.distances(i).into_keys()).collect()
275 }
276 fn command(&mut self, line: &str) -> Result<Vec<String>,String> {
277 let words: Vec<_> = line.split(' ').collect();
278 let c = words[0];
279 let a = &words[1..];
280 let mut out = Vec::new();
281 match c {
282 "WRITE" => {
283 self.work.insert(a[0].into(),line.splitn(3,' ').nth(2).unwrap().into());
284 }
285 "REMOVE" => {
286 if self.work.remove(a[0]).is_none() {
287 return Err(format!("error: no such file '{}'",a[0]));
288 }
289 }
290 "ADD" => {
291 if a[0] == "ALL" {
292 let selected: Vec<_> = paths(&[&self.work,&self.index]).into_iter().filter(|p|!self.ignored(p)).collect();
293 for p in selected {
294 match_path(&mut self.index,&self.work,&p);
295 }
296 self.conflicts.clear();
297 }
298 else {
299 let p = a[0];
300 if self.work.contains_key(p) && self.ignored(p) {
301 return Err(format!("error: path '{p}' is ignored"));
302 }
303 if !self.work.contains_key(p) && !self.index.contains_key(p) && !self.conflicts.contains(p) {
304 return Err(format!("error: pathspec '{p}' did not match any files"));
305 } match_path(&mut self.index,&self.work,p);
306 self.conflicts.remove(p);
307 }
308 }
309 "RM" => {
310 if self.index.remove(a[0]).is_none() {
311 return Err(format!("error: pathspec '{}' did not match any files",a[0]));
312 }
313 self.work.remove(a[0]);
314 self.conflicts.remove(a[0]);
315 }
316 "UNSTAGE" => {
317 if !self.index.contains_key(a[0]) && !self.h().contains_key(a[0]) {
318 return Err(format!("error: pathspec '{}' did not match any files",a[0]));
319 }
320 let tree = self.h().clone();
321 match_path(&mut self.index,&tree,a[0]);
322 }
323 "RESTORE" => {
324 if a.len() == 1 {
325 let value = self.index.get(a[0]).ok_or_else(||format!("error: pathspec '{}' did not match any files",a[0]))?.clone();
326 self.work.insert(a[0].into(),value);
327 }
328 else {
329 let i = self.resolve(a[1])?;
330 let value = self.commit(i).tree.get(a[0]).ok_or_else(||format!("error: path '{}' does not exist in '{}'",a[0],a[1]))?.clone();
331 self.work.insert(a[0].into(),value.clone());
332 self.index.insert(a[0].into(),value);
333 self.conflicts.remove(a[0]);
334 }
335 }
336 "IGNORE" => self.ignores.push(a[0].into()),
337 "COMMIT" | "AMEND" => {
338 let message = line.split_once(' ').unwrap().1.to_string();
339 let parents = if c == "AMEND" {
340 self.in_progress()?;
341 if self.commit(self.head).parents.is_empty() {
342 return Err("error: cannot amend the root commit".into());
343 }
344 self.commit(self.head).parents.clone()
345 }
346 else {
347 if !self.conflicts.is_empty() {
348 return Err("error: unresolved conflicts".into());
349 }
350 if let Some((kind,i)) = &self.pending {
351 if kind == "merge" {
352 vec![self.head,*i]
353 } else {
354 vec![self.head]
355 }
356 } else {
357 vec![self.head]
358 }
359 };
360 if c == "COMMIT" && parents.len() != 2 && self.index == *self.h() {
361 out.push("nothing to commit".into());
362 }
363 else {
364 out.push(self.record(parents,self.index.clone(),message));
365 self.pending = None;
366 }
367 }
368 "STATUS" => {
369 out.push(self.attached.as_ref().map_or(format!("HEAD detached at {}",self.commit(self.head).id),|b|format!("On branch {b}")));
370 if let Some((kind,i)) = &self.pending {
371 out.push(format!("{} {}",match kind.as_str() {
372 "merge" => "Merging", "revert" => "Reverting", _ => "Cherry-picking"
373 },self.commit(*i).id));
374 }
375 let start = out.len();
376 let mut all = paths(&[self.h(),&self.index,&self.work]);
377 all.extend(self.conflicts.iter().cloned());
378 for p in all {
379 if self.conflicts.contains(&p) {
380 out.push(format!("UU {p}"));
381 }
382 else if let Some(value) = self.index.get(&p) {
383 let x = if !self.h().contains_key(&p) {
384 'A'
385 } else if self.h().get(&p) != Some(value) {
386 'M'
387 } else {
388 '.'
389 };
390 let y = if !self.work.contains_key(&p) {
391 'D'
392 } else if self.work.get(&p) != Some(value) {
393 'M'
394 } else {
395 '.'
396 };
397 if x != '.' || y != '.' {
398 out.push(format!("{x}{y} {p}"));
399 }
400 }
401 else {
402 if self.h().contains_key(&p) {
403 out.push(format!("D. {p}"));
404 }
405 if self.work.contains_key(&p) && !self.ignored(&p) {
406 out.push(format!("?? {p}"));
407 }
408 }
409 }
410 if out.len() == start {
411 out.push("nothing to commit, working tree clean".into());
412 }
413 }
414 "LS" | "CAT" => {
415 let revision = if c == "LS" {
416 a.first().copied()
417 } else {
418 a.get(1).copied()
419 };
420 let tree = if let Some(r) = revision {
421 &self.commit(self.resolve(r)?).tree
422 } else {
423 &self.work
424 };
425 if c == "CAT" {
426 out.push(tree.get(a[0]).ok_or_else(||if let Some(r) = revision {
427 format!("error: path '{}' does not exist in '{r}'",a[0])
428 } else {
429 format!("error: no such file '{}'",a[0])
430 })?.clone());
431 }
432 else if tree.is_empty() {
433 out.push("(empty)".into());
434 } else {
435 out.extend(tree.iter().map(|(p,v)|format!("{p} {v}")));
436 }
437 }
438 "DIFF" => {
439 if a.is_empty() {
440 let work: Tree = self.work.iter().filter(|(p,_)|self.index.contains_key(*p)).map(|(p,v)|(p.clone(),v.clone())).collect();
441 out = changes(&self.index,&work);
442 } else if a[0] == "STAGED" {
443 out = changes(self.h(),&self.index);
444 } else {
445 let x = self.resolve(a[0])?;
446 let y = self.resolve(a[1])?;
447 out = changes(&self.commit(x).tree,&self.commit(y).tree);
448 }
449 }
450 "LOG" => {
451 let limit = a.first().map_or(usize::MAX,|s|s.parse().unwrap());
452 let mut current = self.head;
453 for _ in 0..limit {
454 out.push(self.summary(current));
455 if let Some(&p) = self.commit(current).parents.first() {
456 current = p;
457 } else {
458 break;
459 }
460 }
461 }
462 "SHOW" => {
463 let i = self.resolve(a.first().copied().unwrap_or("HEAD"))?;
464 let commit = self.commit(i);
465 out.push(format!("commit {}",self.summary(i)));
466 out.push(format!("parents: {}",if commit.parents.is_empty() {
467 "none".into()
468 } else {
469 commit.parents.iter().map(|&p|self.commit(p).id.clone()).collect::<Vec<_>>().join(" ")
470 }));
471 out.extend(changes(&commit.parents.first().map_or(Tree::new(),|&p|self.commit(p).tree.clone()),&commit.tree));
472 }
473 "BLAME" => {
474 let value = self.h().get(a[0]).ok_or_else(||format!("error: path '{}' does not exist in 'HEAD'",a[0]))?;
475 let mut i = self.head;
476 while let Some(&p) = self.commit(i).parents.first() {
477 if self.commit(p).tree.get(a[0]) != Some(value) {
478 break;
479 } i = p;
480 }
481 out.push(format!("{} ({}) {value}",self.commit(i).id,self.commit(i).message));
482 }
483 "BRANCH" | "TAG" => {
484 let branch = c == "BRANCH";
485 let noun = if branch {
486 "branch"
487 } else {
488 "tag"
489 };
490 if a[0] == "-d" {
491 let name = a[1];
492 let target = (if branch {
493 &self.branches
494 } else {
495 &self.tags
496 }).get(name).copied().ok_or_else(||format!("error: no such {noun} '{name}'"))?;
497 if branch && self.attached.as_deref() == Some(name) {
498 return Err(format!("error: cannot delete current branch '{name}'"));
499 }
500 if branch && !self.ancestor(target,self.head) {
501 return Err(format!("error: branch '{name}' is not fully merged"));
502 }
503 if branch {
504 self.branches.remove(name);
505 } else {
506 self.tags.remove(name);
507 }
508 out.push(format!("Deleted {noun} {name} (was {})",self.commit(target).id));
509 } else {
510 let name = a[0];
511 if (if branch {
512 &self.branches
513 } else {
514 &self.tags
515 }).contains_key(name) {
516 return Err(format!("error: {noun} '{name}' already exists"));
517 }
518 let target = self.resolve(a.get(1).copied().unwrap_or("HEAD"))?;
519 if branch {
520 self.branches.insert(name.into(),target);
521 } else {
522 self.tags.insert(name.into(),target);
523 }
524 out.push(format!("Created {noun} {name} at {}",self.commit(target).id));
525 }
526 }
527 "BRANCHES" => {
528 if self.attached.is_none() {
529 out.push(format!("* (HEAD detached at {})",self.commit(self.head).id));
530 }
531 for (b,&i) in &self.branches {
532 out.push(format!("{} {b} {} {}",if self.attached.as_ref() == Some(b) {
533 "*"
534 } else {
535 " "
536 },self.commit(i).id,self.commit(i).message));
537 }
538 }
539 "TAGS" => {
540 for (tag,&i) in &self.tags {
541 out.push(format!("{tag} {} {}",self.commit(i).id,self.commit(i).message));
542 }
543 if out.is_empty() {
544 out.push("(no tags)".into());
545 }
546 }
547 "SWITCH" => {
548 let target = self.branches.get(a[0]).copied().ok_or_else(||format!("error: no such branch '{}'",a[0]))?;
549 if self.attached.as_deref() == Some(a[0]) {
550 out.push(format!("Already on '{}'",a[0]));
551 } else {
552 self.in_progress()?;
553 self.switch_tree(&self.commit(target).tree.clone())?;
554 self.attached = Some(a[0].into());
555 self.head = target;
556 out.push(format!("Switched to branch '{}'",a[0]));
557 }
558 }
559 "DETACH" => {
560 let target = self.resolve(a[0])?;
561 self.in_progress()?;
562 self.switch_tree(&self.commit(target).tree.clone())?;
563 self.attached = None;
564 self.head = target;
565 out.push(format!("HEAD is now at {} {}",self.commit(target).id,self.commit(target).message));
566 }
567 "MERGE" => {
568 let target = self.resolve(a[0])?;
569 self.in_progress()?;
570 if self.ancestor(target,self.head) {
571 out.push("Already up to date".into());
572 } else if self.ancestor(self.head,target) {
573 let old = self.head;
574 self.switch_tree(&self.commit(target).tree.clone())?;
575 self.move_head(target);
576 out.push(format!("Fast-forward {}..{}",self.commit(old).id,self.commit(target).id));
577 } else {
578 self.clean()?;
579 let base = self.merge_base(self.head,target);
580 out = self.apply(&self.commit(base).tree.clone(),&self.commit(target).tree.clone(),"merge",target,format!("Merge {}",a[0]),vec![self.head,target],Some(base))?;
581 }
582 }
583 "MERGEBASE" => {
584 let x = self.resolve(a[0])?;
585 let y = self.resolve(a[1])?;
586 out.push(self.commit(self.merge_base(x,y)).id.clone());
587 }
588 "REBASE" => {
589 let target = self.resolve(a[0])?;
590 self.in_progress()?;
591 if self.ancestor(target,self.head) {
592 return Ok(vec!["Already up to date".into()]);
593 }
594 self.clean()?;
595 let mut list = Vec::new();
596 let mut current = self.head;
597 while !self.ancestor(current,target) {
598 list.push(current);
599 current = self.commit(current).parents[0];
600 } list.reverse();
601 let mut tree = self.commit(target).tree.clone();
602 let mut planned = Vec::new();
603 for &i in &list {
604 let commit = self.commit(i);
605 let (merged,conflicts) = three_way(&self.commit(commit.parents[0]).tree,&tree,&commit.tree);
606 if let Some(p) = conflicts.keys().next() {
607 return Err(format!("error: could not rebase {}: conflict in '{p}'",commit.id));
608 }
609 if merged == tree {
610 planned.push(None);
611 } else {
612 planned.push(Some(merged.clone()));
613 tree = merged;
614 }
615 }
616 self.switch_tree(&tree)?;
617 let mut parent = target;
618 for (i,planned) in list.into_iter().zip(planned) {
619 let commit = self.commit(i).clone();
620 if let Some(tree) = planned {
621 let next = self.create(vec![parent],tree,commit.message.clone());
622 out.push(format!("pick {} -> {} {}",commit.id,self.commit(next).id,commit.message));
623 parent = next;
624 } else {
625 out.push(format!("skip {} {}",commit.id,commit.message));
626 }
627 }
628 self.move_head(parent);
629 out.push(format!("HEAD is now at {} {}",self.commit(parent).id,self.commit(parent).message));
630 }
631 "RESET" => {
632 let target = self.resolve(a[1])?;
633 let tree = self.commit(target).tree.clone();
634 if a[0] == "HARD" {
635 let mut all = paths(&[&self.index,&tree]);
636 all.extend(self.conflicts.iter().cloned());
637 for p in all {
638 match_path(&mut self.work,&tree,&p);
639 }
640 }
641 if a[0] != "SOFT" {
642 self.index = tree;
643 }
644 self.move_head(target);
645 self.conflicts.clear();
646 self.pending = None;
647 out.push(format!("HEAD is now at {} {}",self.commit(target).id,self.commit(target).message));
648 }
649 "REVERT" | "CHERRYPICK" => {
650 let target = self.resolve(a[0])?;
651 self.in_progress()?;
652 self.clean()?;
653 let commit = self.commit(target).clone();
654 let parent = commit.parents.first().map_or(Tree::new(),|&p|self.commit(p).tree.clone());
655 let (base,theirs,kind,message) = if c == "REVERT" {
656 (commit.tree,parent,"revert",format!("Revert \"{}\"",commit.message))
657 } else {
658 (parent,commit.tree,"cherry-pick",commit.message)
659 };
660 out = self.apply(&base,&theirs,kind,target,message,vec![self.head],None)?;
661 }
662 "STASH" => {
663 if a.is_empty() {
664 self.in_progress()?;
665 if !self.uncommitted() {
666 out.push("No local changes to save".into());
667 } else {
668 let all = paths(&[&self.index,self.h()]);
669 let saved: Tree = all.iter().filter_map(|p|self.work.get(p).map(|v|(p.clone(),v.clone()))).collect();
670 let head = self.h().clone();
671 for p in all {
672 match_path(&mut self.work,&head,&p);
673 }
674 self.index = head;
675 let label = format!("WIP on {}: {} {}",self.attached.as_deref().unwrap_or("(no branch)"),self.commit(self.head).id,self.commit(self.head).message);
676 self.stashes.insert(0,Stash {
677 label: label.clone(),base:self.head,tree:saved
678 });
679 out.push(format!("Saved stash@{{0}}: {label}"));
680 }
681 }
682 else if a[0] == "LIST" {
683 out.extend(self.stashes.iter().enumerate().map(|(i,s)|format!("stash@{{{i}}}: {}",s.label)));
684 if out.is_empty() {
685 out.push("(no stash entries)".into());
686 }
687 }
688 else {
689 let stash = self.stashes.first().ok_or("error: no stash entries")?.clone();
690 if a[0] == "DROP" {
691 self.stashes.remove(0);
692 out.push(format!("Dropped stash@{{0}}: {}",stash.label));
693 } else {
694 self.in_progress()?;
695 self.clean()?;
696 let (merged,conflicts) = three_way(&self.commit(stash.base).tree,self.h(),&stash.tree);
697 if let Some(p) = conflicts.keys().next() {
698 return Err(format!("error: stash conflict in '{p}'"));
699 }
700 let changed = differences(self.h(),&merged);
701 self.overwrite(&changed)?;
702 for p in changed {
703 match_path(&mut self.work,&merged,&p);
704 }
705 self.stashes.remove(0);
706 out.push(format!("Restored stash@{{0}}: {}",stash.label));
707 out.extend(changes(self.h(),&merged));
708 }
709 }
710 }
711 "GRAPH" => {
712 let mut lanes = Vec::new();
713 for i in self.reachable(false).into_iter().rev() {
714 if !lanes.contains(&i) {
715 lanes.push(i);
716 }
717 let k = lanes.iter().position(|&v|v == i).unwrap();
718 let symbols = lanes.iter().enumerate().map(|(j,&v)|if j == k {
719 "*"
720 } else if v == i {
721 "/"
722 } else {
723 "|"
724 }).collect::<Vec<_>>().join(" ");
725 out.push(format!("{symbols} {}",self.summary(i)));
726 let mut j = lanes.len();
727 while j > k+1 {
728 j -= 1;
729 if lanes[j] == i {
730 lanes.remove(j);
731 }
732 }
733 let parents = &self.commit(i).parents;
734 if parents.is_empty() {
735 lanes.remove(k);
736 } else {
737 lanes[k] = parents[0];
738 if parents.len() == 2 && !lanes.contains(&parents[1]) {
739 lanes.insert(k+1,parents[1]);
740 }
741 }
742 }
743 }
744 "DESCRIBE" => {
745 let target = self.resolve(a.first().copied().unwrap_or("HEAD"))?;
746 let distance = self.distances(target);
747 let choice = self.tags.iter().filter_map(|(tag,i)|distance.get(i).map(|&d|(d,tag))).min();
748 let (d,tag) = choice.ok_or_else(||format!("error: no tag can describe '{}'",self.commit(target).id))?;
749 out.push(if d == 0 {
750 tag.clone()
751 } else {
752 format!("{tag}-{d}-g{}",self.commit(target).id)
753 });
754 }
755 "GC" => {
756 let keep = self.reachable(true);
757 let mut count = 0;
758 for (i,c) in self.commits.iter_mut().enumerate() {
759 if c.is_some() && !keep.contains(&i) {
760 *c = None;
761 count += 1;
762 }
763 }
764 out.push(format!("Removed unreachable commits: {count}"));
765 }
766 _ => unreachable!(),
767 }
768 Ok(out)
769 }
770}
771fn main() {
772 let mut input = String::new();
773 io::stdin().read_to_string(&mut input).unwrap();
774 let mut repository = Repository::new();
775 for line in input.lines().take_while(|&line|line != "EXIT") {
776 match repository.command(line) {
777 Ok(out) => for line in out {
778 println!("{line}");
779 }, Err(error) => println!("{error}")
780 }
781 }
