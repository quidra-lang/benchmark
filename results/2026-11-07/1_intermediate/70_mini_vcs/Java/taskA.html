78 static Commit create(List<Commit> parents, Map<String, String> tree, String message) {
79 int sequence = nextSequence++;
80 StringBuilder text = new StringBuilder("commit " + sequence + "\n");
81 for (Commit parent : parents) text.append("parent ").append(parent.id).append('\n');
82 for (String path : new TreeSet<>(tree.keySet())) text.append("file ").append(path).append(' ').append(tree.get(path)).append('\n');
83 text.append("message ").append(message).append('\n');
84 long hash = 0;
85 for (int i = 0; i < text.length(); i++) hash = (hash * 131 + text.charAt(i)) % 2147483647L;
86 String id = String.format(Locale.ROOT, "%08x", hash);
87 Commit commit = new Commit(id, sequence, parents, tree, message);
88 commits.put(id, commit);
89 return commit;
90 }

92 static void record(List<Commit> parents, Map<String, String> tree, String message) {
93 Commit commit = create(parents, tree, message);
94 int added = 0, modified = 0, deleted = 0;
95 Map<String, String> before = parents.getFirst().tree;
96 for (String path : differences(before, tree)) {
97 if (!before.containsKey(path)) added++;
98 else if (!tree.containsKey(path)) deleted++;
99 else modified++;
100 }
101 moveHead(commit);
102 System.out.println("[" + (branch == null ? "detached HEAD" : branch) + " " + commit.id + "] " + message + " (" + added + " added, " + modified + " modified, " + deleted + " deleted)");
103 }

105 static Commit resolve(String revision) {
106 String[] parts = revision.split("~");
107 String anchor = parts[0];
108 Commit commit = anchor.equals("HEAD") ? head() : branches.containsKey(anchor) ? branches.get(anchor) : tags.containsKey(anchor) ? tags.get(anchor) : commits.get(anchor);
109 int steps = parts.length == 1 ? 0 : Integer.parseInt(parts[1]);
110 while (commit != null && steps-- > 0) commit = commit.parents.isEmpty() ? null : commit.parents.getFirst();
111 require(commit != null, "error: unknown revision '" + revision + "'");
112 return commit;
113 }

115 static Map<Commit, Integer> distances(Commit start) {
116 Map<Commit, Integer> result = new HashMap<>();
117 ArrayDeque<Commit> queue = new ArrayDeque<>();
118 result.put(start, 0);
119 queue.add(start);
120 while (!queue.isEmpty()) {
121 Commit commit = queue.remove();
122 for (Commit parent : commit.parents) if (!result.containsKey(parent)) {
123 result.put(parent, result.get(commit) + 1);
124 queue.add(parent);
125 }
126 }
127 return result;
128 }

130 static boolean ancestor(Commit a, Commit b) {
131 return distances(b).containsKey(a);
132 }

134 static Commit mergeBase(Commit a, Commit b) {
135 Map<Commit, Integer> x = distances(a), y = distances(b);
136 List<Commit> common = x.keySet().stream().filter(y::containsKey).toList();
137 Commit best = null;
138 int bestDistance = Integer.MAX_VALUE;
139 for (Commit candidate : common) {
140 if (common.stream().anyMatch(other -> other != candidate && ancestor(candidate, other))) continue;
141 int distance = Math.max(x.get(candidate), y.get(candidate));
142 if (best == null || distance < bestDistance || distance == bestDistance && candidate.sequence > best.sequence) {
143 best = candidate;
144 bestDistance = distance;
145 }
146 }
147 return best;
148 }

150 static String summary(Commit commit) {
151 List<String> refs = new ArrayList<>();
152 if (head() == commit) refs.add(branch == null ? "HEAD" : "HEAD -> " + branch);
153 branches.forEach((name, value) -> {
154 if (value == commit && !name.equals(branch)) refs.add(name);
155 });
156 tags.forEach((name, value) -> {
157 if (value == commit) refs.add("tag: " + name);
158 });
159 return commit.id + (refs.isEmpty() ? "" : " (" + String.join(", ", refs) + ")") + " " + commit.message;
160 }

193 static Merge merge(Map<String, String> base, Map<String, String> ours, Map<String, String> theirs) {
194 TreeMap<String, String> result = new TreeMap<>();
195 TreeSet<String> problems = new TreeSet<>();
196 Map<String, String> contents = new HashMap<>();
197 for (String path : paths(base, ours, theirs)) {
198 if (agree(theirs, base, path) || agree(ours, theirs, path)) match(result, ours, path);
199 else if (agree(ours, base, path)) match(result, theirs, path);
200 else {
201 problems.add(path);
202 match(result, ours, path);
203 contents.put(path, "<<<<<<< " + ours.getOrDefault(path, "(deleted)") + " ======= " + theirs.getOrDefault(path, "(deleted)") + " >>>>>>>");
204 }
205 }
206 return new Merge(result, problems, contents);
207 }

209 static void apply(Map<String, String> base, Map<String, String> theirs, String kind, Commit chosen, String message, List<Commit> parents, Commit mergeBase) {
210 Merge result = merge(base, head().tree, theirs);
211 var changed = differences(head().tree, result.tree());
212 TreeSet<String> checked = new TreeSet<>(changed);
213 checked.addAll(result.conflicts());
214 overwriteCheck(checked);
215 if (kind.equals("merge")) System.out.println("Merge base " + mergeBase.id);
216 else if (result.conflicts().isEmpty() && changed.isEmpty()) {
217 System.out.println("nothing to commit");
218 return;
219 }
220 for (String path : changed) {
221 match(index, result.tree(), path);
222 match(working, result.tree(), path);
223 }
224 if (result.conflicts().isEmpty()) record(parents, result.tree(), message);
225 else {
226 for (String path : result.conflicts()) {
227 working.put(path, result.contents().get(path));
228 System.out.println("CONFLICT " + path);
229 }
230 conflicts.clear();
231 conflicts.addAll(result.conflicts());
232 operation = kind;
233 target = chosen;
234 System.out.println(kind.equals("merge") ? "Automatic merge failed; fix conflicts and then commit the result." : kind.equals("revert") ? "error: could not revert " + chosen.id : "error: could not apply " + chosen.id);
235 }
236 }

238 static boolean glob(String pattern, String path) {
239 boolean[][] dp = new boolean[pattern.length() + 1][path.length() + 1];
240 dp[0][0] = true;
241 for (int i = 0; i < pattern.length(); i++) for (int j = 0; j <= path.length(); j++) {
242 if (pattern.charAt(i) == '*') dp[i + 1][j] = dp[i][j] || j > 0 && dp[i + 1][j - 1];
243 else if (j > 0) dp[i + 1][j] = dp[i][j - 1] && (pattern.charAt(i) == '?' || pattern.charAt(i) == path.charAt(j - 1));
244 }
245 return dp[pattern.length()][path.length()];
246 }

248 static boolean ignored(String path) {
249 return !index.containsKey(path) && !conflicts.contains(path) && ignores.stream().anyMatch(pattern -> glob(pattern, path));
250 }

256 static Set<Commit> reachable(boolean extra) {
257 Set<Commit> result = new HashSet<>();
258 List<Commit> roots = new ArrayList<>(branches.values());
259 roots.addAll(tags.values());
260 roots.add(head());
261 if (extra) {
262 for (Stash stash : stashes) roots.add(stash.base());
263 if (operation != null) roots.add(target);
264 }
265 for (Commit root : roots) result.addAll(distances(root).keySet());
266 return result;
267 }

269 static void command(String line) {
270 String[] words = line.split(" ");
271 String command = words[0], path = words.length > 1 ? words[1] : null;
272 switch (command) {
273 case "WRITE":
274 working.put(path, line.substring(line.indexOf(' ', line.indexOf(' ') + 1) + 1));
275 break;
276 case "REMOVE":
277 require(working.containsKey(path), "error: no such file '" + path + "'");
278 working.remove(path);
279 break;
280 case "IGNORE":
281 ignores.add(path);
282 break;
283 case "ADD":
284 if (path.equals("ALL")) {
285 List<String> selected = paths(working, index).stream().filter(p -> !ignored(p)).toList();
286 for (String p : selected) match(index, working, p);
287 conflicts.clear();
288 } else {
289 require(!working.containsKey(path) || !ignored(path), "error: path '" + path + "' is ignored");
290 pathspec(working.containsKey(path) || index.containsKey(path) || conflicts.contains(path), path);
291 match(index, working, path);
292 conflicts.remove(path);
293 }
294 break;
295 case "RM":
296 pathspec(index.containsKey(path), path);
297 index.remove(path);
298 working.remove(path);
299 conflicts.remove(path);
300 break;
301 case "UNSTAGE":
302 pathspec(index.containsKey(path) || head().tree.containsKey(path), path);
303 match(index, head().tree, path);
304 break;
305 case "RESTORE":
306 if (words.length == 2) {
307 pathspec(index.containsKey(path), path);
308 working.put(path, index.get(path));
309 } else {
310 Commit chosen = resolve(words[2]);
311 require(chosen.tree.containsKey(path), "error: path '" + path + "' does not exist in '" + words[2] + "'");
312 match(index, chosen.tree, path);
313 match(working, chosen.tree, path);
314 conflicts.remove(path);
315 }
316 break;
317 case "COMMIT":
318 require(conflicts.isEmpty(), "error: unresolved conflicts");
319 if (!"merge".equals(operation) && index.equals(head().tree)) {
320 System.out.println("nothing to commit");
321 break;
322 }
323 record("merge".equals(operation) ? List.of(head(), target) : List.of(head()), index, line.substring(7));
324 operation = null;
325 target = null;
326 break;
327 case "AMEND":
328 progressCheck();
329 require(!head().parents.isEmpty(), "error: cannot amend the root commit");
330 record(head().parents, index, line.substring(6));
331 break;
332 case "STATUS":
333 {
334 System.out.println(branch == null ? "HEAD detached at " + head().id : "On branch " + branch);
335 if (operation != null) System.out.println((operation.equals("merge") ? "Merging " : operation.equals("revert") ? "Reverting " : "Cherry-picking ") + target.id);
336 var all = paths(head().tree, index, working);
337 all.addAll(conflicts);
338 boolean any = false;
339 for (String p : all) {
340 if (conflicts.contains(p)) {
341 System.out.println("UU " + p);
342 any = true;
343 } else if (index.containsKey(p)) {
344 char x = !head().tree.containsKey(p) ? 'A' : agree(head().tree, index, p) ? '.' : 'M';
345 char y = !working.containsKey(p) ? 'D' : agree(working, index, p) ? '.' : 'M';
346 if (x != '.' || y != '.') {
347 System.out.println("" + x + y + " " + p);
348 any = true;
349 }
350 } else {
351 if (head().tree.containsKey(p)) {
352 System.out.println("D. " + p);
353 any = true;
354 }
355 if (working.containsKey(p) && !ignored(p)) {
356 System.out.println("?? " + p);
357 any = true;
358 }
359 }
360 }
361 if (!any) System.out.println("nothing to commit, working tree clean");
362 break;
363 }
364 case "LS":
365 {
366 Map<String, String> tree = words.length == 1 ? working : resolve(path).tree;
367 if (tree.isEmpty()) System.out.println("(empty)");
368 else tree.forEach((p, text) -> System.out.println(p + " " + text));
369 break;
370 }
371 case "CAT":
372 {
373 Map<String, String> tree = words.length == 2 ? working : resolve(words[2]).tree;
374 require(tree.containsKey(path), words.length == 2 ? "error: no such file '" + path + "'" : "error: path '" + path + "' does not exist in '" + words[2] + "'");
375 System.out.println(tree.get(path));
376 break;
377 }
378 case "DIFF":
379 if (words.length == 1) {
380 TreeMap<String, String> tracked = new TreeMap<>(working);
381 tracked.keySet().retainAll(index.keySet());
382 changes(index, tracked);
383 } else if (words.length == 2) changes(head().tree, index);
384 else {
385 Commit a = resolve(path), b = resolve(words[2]);
386 changes(a.tree, b.tree);
387 }
388 break;
389 case "LOG":
390 {
391 int limit = words.length == 1 ? Integer.MAX_VALUE : Integer.parseInt(path);
392 Commit chosen = head();
393 while (limit-- > 0) {
394 System.out.println(summary(chosen));
395 if (chosen.parents.isEmpty()) break;
396 chosen = chosen.parents.getFirst();
397 }
398 break;
399 }
400 case "SHOW":
401 {
402 Commit chosen = resolve(words.length == 1 ? "HEAD" : path);
403 System.out.println("commit " + summary(chosen));
404 System.out.println("parents: " + (chosen.parents.isEmpty() ? "none" : String.join(" ", chosen.parents.stream().map(p -> p.id).toList())));
405 changes(chosen.parents.isEmpty() ? Map.of() : chosen.parents.getFirst().tree, chosen.tree);
406 break;
407 }
408 case "BLAME":
409 {
410 require(head().tree.containsKey(path), "error: path '" + path + "' does not exist in 'HEAD'");
411 Commit chosen = head();
412 String text = chosen.tree.get(path);
413 while (!chosen.parents.isEmpty() && Objects.equals(chosen.parents.getFirst().tree.get(path), text)) chosen = chosen.parents.getFirst();
414 System.out.println(chosen.id + " (" + chosen.message + ") " + text);
415 break;
416 }
417 case "BRANCH":
418 case "TAG":
419 {
420 boolean tag = command.equals("TAG");
421 var refs = tag ? tags : branches;
422 String noun = tag ? "tag" : "branch";
423 if (path.equals("-d")) {
424 String name = words[2];
425 require(refs.containsKey(name), "error: no such " + noun + " '" + name + "'");
426 Commit chosen = refs.get(name);
427 if (!tag) {
428 require(!name.equals(branch), "error: cannot delete current branch '" + name + "'");
429 require(ancestor(chosen, head()), "error: branch '" + name + "' is not fully merged");
430 }
431 refs.remove(name);
432 System.out.println("Deleted " + noun + " " + name + " (was " + chosen.id + ")");
433 } else {
434 require(!refs.containsKey(path), "error: " + noun + " '" + path + "' already exists");
435 Commit chosen = resolve(words.length == 2 ? "HEAD" : words[2]);
436 refs.put(path, chosen);
437 System.out.println("Created " + noun + " " + path + " at " + chosen.id);
438 }
439 break;
440 }
441 case "BRANCHES":
442 if (branch == null) System.out.println("* (HEAD detached at " + head().id + ")");
443 branches.forEach((name, chosen) -> System.out.println((name.equals(branch) ? "* " : " ") + name + " " + chosen.id + " " + chosen.message));
444 break;
445 case "TAGS":
446 if (tags.isEmpty()) System.out.println("(no tags)");
447 else tags.forEach((name, chosen) -> System.out.println(name + " " + chosen.id + " " + chosen.message));
448 break;
449 case "SWITCH":
450 {
451 require(branches.containsKey(path), "error: no such branch '" + path + "'");
452 if (path.equals(branch)) {
453 System.out.println("Already on '" + path + "'");
454 break;
455 }
456 progressCheck();
457 switchTree(branches.get(path).tree);
458 branch = path;
459 System.out.println("Switched to branch '" + path + "'");
460 break;
461 }
462 case "DETACH":
463 {
464 Commit chosen = resolve(path);
465 progressCheck();
466 switchTree(chosen.tree);
467 branch = null;
468 detached = chosen;
469 System.out.println("HEAD is now at " + chosen.id + " " + chosen.message);
470 break;
471 }
472 case "MERGEBASE":
473 {
474 Commit a = resolve(path), b = resolve(words[2]);
475 System.out.println(mergeBase(a, b).id);
476 break;
477 }
478 case "MERGE":
479 {
480 Commit chosen = resolve(path), before = head();
481 progressCheck();
482 if (ancestor(chosen, before)) {
483 System.out.println("Already up to date");
484 break;
485 }
486 if (ancestor(before, chosen)) {
487 switchTree(chosen.tree);
488 moveHead(chosen);
489 System.out.println("Fast-forward " + before.id + ".." + chosen.id);
490 break;
491 }
492 cleanCheck();
493 Commit base = mergeBase(before, chosen);
494 apply(base.tree, chosen.tree, "merge", chosen, "Merge " + path, List.of(before, chosen), base);
495 break;
496 }
497 case "RESET":
498 {
499 Commit chosen = resolve(words[2]);
500 if (path.equals("HARD")) {
501 var all = paths(index, chosen.tree);
502 all.addAll(conflicts);
503 for (String p : all) match(working, chosen.tree, p);
504 }
505 if (!path.equals("SOFT")) {
506 index.clear();
507 index.putAll(chosen.tree);
508 }
509 moveHead(chosen);
510 conflicts.clear();
511 operation = null;
512 target = null;
513 System.out.println("HEAD is now at " + chosen.id + " " + chosen.message);
514 break;
515 }
516 case "REVERT":
517 case "CHERRYPICK":
518 {
519 Commit chosen = resolve(path);
520 progressCheck();
521 cleanCheck();
522 Map<String, String> parent = chosen.parents.isEmpty() ? Map.of() : chosen.parents.getFirst().tree;
523 boolean revert = command.equals("REVERT");
524 apply(revert ? chosen.tree : parent, revert ? parent : chosen.tree, revert ? "revert" : "cherry-pick", chosen, revert ? "Revert \"" + chosen.message + "\"" : chosen.message, List.of(head()), null);
525 break;
526 }
527 case "REBASE":
528 {
529 Commit upstream = resolve(path);
530 progressCheck();
531 if (ancestor(upstream, head())) {
532 System.out.println("Already up to date");
533 break;
534 }
535 cleanCheck();
536 List<Commit> old = new ArrayList<>();
537 Commit cursor = head();
538 while (!ancestor(cursor, upstream)) {
539 old.add(cursor);
540 cursor = cursor.parents.getFirst();
541 }
542 Collections.reverse(old);
543 List<Replay> replay = new ArrayList<>();
544 TreeMap<String, String> tree = new TreeMap<>(upstream.tree);
545 for (Commit commit : old) {
546 Merge result = merge(commit.parents.getFirst().tree, tree, commit.tree);
547 require(result.conflicts().isEmpty(), "error: could not rebase " + commit.id + ": conflict in '" + (result.conflicts().isEmpty() ? "" : result.conflicts().first()) + "'");
548 boolean pick = !tree.equals(result.tree());
549 replay.add(new Replay(commit, result.tree(), pick));
550 tree = result.tree();
551 }
552 switchTree(tree);
553 Commit current = upstream;
554 for (Replay item : replay) if (item.pick()) {
555 Commit created = create(List.of(current), item.tree(), item.original().message);
556 System.out.println("pick " + item.original().id + " -> " + created.id + " " + created.message);
557 current = created;
558 } else System.out.println("skip " + item.original().id + " " + item.original().message);
559 moveHead(current);
560 System.out.println("HEAD is now at " + current.id + " " + current.message);
561 break;
562 }
563 case "STASH":
564 {
565 if (words.length > 1 && path.equals("LIST")) {
566 if (stashes.isEmpty()) System.out.println("(no stash entries)");
567 for (int i = 0; i < stashes.size(); i++) System.out.println("stash@{" + i + "}: " + stashes.get(i).label());
568 break;
569 }
570 if (words.length > 1) {
571 require(!stashes.isEmpty(), "error: no stash entries");
572 Stash stash = stashes.getFirst();
573 if (path.equals("DROP")) {
574 stashes.removeFirst();
575 System.out.println("Dropped stash@{0}: " + stash.label());
576 break;
577 }
578 progressCheck();
579 cleanCheck();
580 Merge result = merge(stash.base().tree, head().tree, stash.tree());
581 require(result.conflicts().isEmpty(), "error: stash conflict in '" + (result.conflicts().isEmpty() ? "" : result.conflicts().first()) + "'");
582 var changed = differences(head().tree, result.tree());
583 overwriteCheck(changed);
584 for (String p : changed) match(working, result.tree(), p);
585 stashes.removeFirst();
586 System.out.println("Restored stash@{0}: " + stash.label());
587 changes(head().tree, result.tree());
588 } else {
589 progressCheck();
590 if (!dirty()) {
591 System.out.println("No local changes to save");
592 break;
593 }
594 TreeMap<String, String> saved = new TreeMap<>();
595 var tracked = paths(index, head().tree);
596 for (String p : tracked) if (working.containsKey(p)) saved.put(p, working.get(p));
597 for (String p : tracked) match(working, head().tree, p);
598 index.clear();
599 index.putAll(head().tree);
600 String label = "WIP on " + (branch == null ? "(no branch)" : branch) + ": " + head().id + " " + head().message;
601 stashes.addFirst(new Stash(label, head(), saved));
602 System.out.println("Saved stash@{0}: " + label);
603 }
604 break;
605 }
606 case "GRAPH":
607 {
608 List<Commit> ordered = new ArrayList<>(reachable(false));
609 ordered.sort(Comparator.comparingInt((Commit c) -> c.sequence).reversed());
610 List<Commit> lanes = new ArrayList<>();
611 for (Commit commit : ordered) {
612 if (!lanes.contains(commit)) lanes.add(commit);
613 int first = lanes.indexOf(commit);
614 List<String> symbols = new ArrayList<>();
615 for (int i = 0; i < lanes.size(); i++) symbols.add(i == first ? "*" : lanes.get(i) == commit ? "/" : "|");
616 System.out.println(String.join(" ", symbols) + " " + summary(commit));
617 for (int i = lanes.size() - 1; i > first; i--) if (lanes.get(i) == commit) lanes.remove(i);
618 if (commit.parents.isEmpty()) lanes.remove(first);
619 else {
620 lanes.set(first, commit.parents.getFirst());
621 if (commit.parents.size() == 2 && !lanes.contains(commit.parents.get(1))) lanes.add(first + 1, commit.parents.get(1));
622 }
623 }
624 break;
625 }
626 case "DESCRIBE":
627 {
628 Commit chosen = resolve(words.length == 1 ? "HEAD" : path);
629 var distance = distances(chosen);
630 String best = null;
631 int minimum = Integer.MAX_VALUE;
632 for (var entry : tags.entrySet()) if (distance.containsKey(entry.getValue()) && distance.get(entry.getValue()) < minimum) {
633 best = entry.getKey();
634 minimum = distance.get(entry.getValue());
635 }
636 require(best != null, "error: no tag can describe '" + chosen.id + "'");
637 System.out.println(best + (minimum == 0 ? "" : "-" + minimum + "-g" + chosen.id));
638 break;
639 }
640 case "GC":
641 {
642 Set<Commit> keep = reachable(true);
643 int before = commits.size();
644 commits.values().removeIf(commit -> !keep.contains(commit));
645 System.out.println("Removed unreachable commits: " + (before - commits.size()));
646 break;
647 }
648 }
