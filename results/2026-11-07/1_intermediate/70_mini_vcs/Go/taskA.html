34func clone(files tree) tree {
35 result := tree{}
36 for path, text := range files {
37 result[path] = text
38 }
39 return result
40}

42func paths(trees ...tree) []string {
43 all := map[string]bool{}
44 for _, files := range trees {
45 for path := range files {
46 all[path] = true
47 }
48 }
49 result := []string{}
50 for path := range all {
51 result = append(result, path)
52 }
53 sort.Strings(result)
54 return result
55}

57func agree(a, b tree, path string) bool {
58 left, hasLeft := a[path]
59 right, hasRight := b[path]
60 return hasLeft == hasRight && left == right
61}
62
63func differences(a, b tree) []string {
64 result := []string{}
65 for _, path := range paths(a, b) {
66 if !agree(a, b, path) {
67 result = append(result, path)
68 }
69 }
70 return result
71}
72
73func same(a, b tree) bool {
74 return len(differences(a, b)) == 0
75}

77func matchPath(target, source tree, path string) {
78 if text, present := source[path]; present {
79 target[path] = text
80 } else {
81 delete(target, path)
82 }
83}

85func conflictText(ours, theirs tree, path string) string {
86 a, ok := ours[path]
87 if !ok {
88 a = "(deleted)"
89 }
90 b, ok := theirs[path]
91 if !ok {
92 b = "(deleted)"
93 }
94 return "<<<<<<< " + a + " ======= " + b + " >>>>>>>"
95}

97func threeWay(base, ours, theirs tree) (tree, []string) {
98 result, conflicts := tree{}, []string{}
99 for _, path := range paths(base, ours, theirs) {
100 if agree(theirs, base, path) || agree(ours, theirs, path) {
101 matchPath(result, ours, path)
102 } else if agree(ours, base, path) {
103 matchPath(result, theirs, path)
104 } else {
105 conflicts = append(conflicts, path)
106 matchPath(result, ours, path)
107 }
108 }
109 return result, conflicts
110}

112func distances(start *commit) map[*commit]int {
113 result, queue := map[*commit]int{start: 0}, []*commit{start}
114 for at := 0; at < len(queue); at++ {
115 current := queue[at]
116 for _, parent := range current.parents {
117 if _, found := result[parent]; !found {
118 result[parent] = result[current] + 1
119 queue = append(queue, parent)
120 }
121 }
122 }
123 return result
124}

126func ancestor(a, b *commit) bool {
127 _, exists := distances(b)[a]
128 return exists
129}

131func mergeBase(a, b *commit) *commit {
132 da, db := distances(a), distances(b)
133 candidates := map[*commit]bool{}
134 for c := range da {
135 if _, found := db[c]; found {
136 candidates[c] = true
137 }
138 }
139 excluded := map[*commit]bool{}
140 for c := range candidates {
141 for _, parent := range c.parents {
142 excluded[parent] = true
143 }
144 }
145 var best *commit
146 bestDistance := 0
147 for c := range candidates {
148 if excluded[c] {
149 continue
150 }
151 d := da[c]
152 if db[c] > d {
153 d = db[c]
154 }
155 if best == nil || d < bestDistance || d == bestDistance && c.sequence > best.sequence {
156 best, bestDistance = c, d
157 }
158 }
159 return best
160}

162func glob(pattern, path string) bool {
163 previous := make([]bool, len(path)+1)
164 previous[0] = true
165 for i := range pattern {
166 current := make([]bool, len(path)+1)
167 if pattern[i] == '*' {
168 current[0] = previous[0]
169 }
170 for j := 1; j <= len(path); j++ {
171 if pattern[i] == '*' {
172 current[j] = previous[j] || current[j-1]
173 } else {
174 current[j] = previous[j-1] && (pattern[i] == '?' || pattern[i] == path[j-1])
175 }
176 }
177 previous = current
178 }
179 return previous[len(path)]
180}

218func (r *repository) create(parents []*commit, files tree, message string) *commit {
219 var text strings.Builder
220 fmt.Fprintf(&text, "commit %d\n", r.created)
221 for _, parent := range parents {
222 fmt.Fprintf(&text, "parent %s\n", parent.id)
223 }
224 for _, path := range paths(files) {
225 fmt.Fprintf(&text, "file %s %s\n", path, files[path])
226 }
227 fmt.Fprintf(&text, "message %s\n", message)
228 hash := int64(0)
229 for _, c := range []byte(text.String()) {
230 hash = (hash*131 + int64(c)) % 2147483647
231 }
232 c := &commit{id: fmt.Sprintf("%08x", hash), sequence: r.created, parents: append([]*commit{}, parents...), files: clone(files), message: message}
233 r.created++
234 r.commits[c.id] = c
235 return c
236}

238func (r *repository) record(parents []*commit, files tree, message string) {
239 c := r.create(parents, files, message)
240 r.moveHead(c)
241 location := r.attached
242 if location == "" {
243 location = "detached HEAD"
244 }
245 added, modified, deleted := 0, 0, 0
246 old := parents[0].files
247 for _, path := range differences(old, files) {
248 if _, found := old[path]; !found {
249 added++
250 } else if _, found := files[path]; !found {
251 deleted++
252 } else {
253 modified++
254 }
255 }
256 fmt.Fprintf(r.out, "[%s %s] %s (%d added, %d modified, %d deleted)\n", location, c.id, message, added, modified, deleted)
257}

259func (r *repository) revision(rev string) (*commit, error) {
260 anchor, suffix, hasSuffix := strings.Cut(rev, "~")
261 var c *commit
262 if anchor == "HEAD" {
263 c = r.head()
264 } else if r.branches[anchor] != nil {
265 c = r.branches[anchor]
266 } else if r.tags[anchor] != nil {
267 c = r.tags[anchor]
268 } else {
269 c = r.commits[anchor]
270 }
271 if hasSuffix {
272 n, _ := strconv.Atoi(suffix)
273 for i := 0; i < n && c != nil; i++ {
274 if len(c.parents) == 0 {
275 c = nil
276 } else {
277 c = c.parents[0]
278 }
279 }
280 }
281 if c == nil {
282 return nil, fmt.Errorf("unknown revision '%s'", rev)
283 }
284 return c, nil
285}

296func (r *repository) summary(c *commit) string {
297 references := []string{}
298 if r.head() == c {
299 if r.attached == "" {
300 references = append(references, "HEAD")
301 } else {
302 references = append(references, "HEAD -> "+r.attached)
303 }
304 }
305 for _, name := range referenceNames(r.branches) {
306 if name != r.attached && r.branches[name] == c {
307 references = append(references, name)
308 }
309 }
310 for _, name := range referenceNames(r.tags) {
311 if r.tags[name] == c {
312 references = append(references, "tag: "+name)
313 }
314 }
315 text := c.id
316 if len(references) > 0 {
317 text += " (" + strings.Join(references, ", ") + ")"
318 }
319 return text + " " + c.message
320}

322func (r *repository) changes(a, b tree) {
323 changed := differences(a, b)
324 if len(changed) == 0 {
325 fmt.Fprintln(r.out, "(no changes)")
326 return
327 }
328 for _, path := range changed {
329 old, inOld := a[path]
330 text, inNew := b[path]
331 if !inOld {
332 fmt.Fprintf(r.out, "A %s: %s\n", path, text)
333 } else if !inNew {
334 fmt.Fprintf(r.out, "D %s: %s\n", path, old)
335 } else {
336 fmt.Fprintf(r.out, "M %s: %s -> %s\n", path, old, text)
337 }
338 }
339}

348func (r *repository) uncommitted() bool {
349 if !same(r.index, r.head().files) {
350 return true
351 }
352 for path := range r.index {
353 if !agree(r.index, r.work, path) {
354 return true
355 }
356 }
357 return false

367func (r *repository) overwrite(selected []string) error {
368 sort.Strings(selected)
369 for _, path := range selected {
370 if !agree(r.index, r.head().files, path) || !agree(r.work, r.head().files, path) {
371 return fmt.Errorf("local changes to '%s' would be overwritten", path)
372 }
373 }
374 return nil
375}

377func (r *repository) switchTree(files tree) error {
378 changed := differences(r.head().files, files)
379 if err := r.overwrite(changed); err != nil {
380 return err
381 }
382 for _, path := range changed {
383 matchPath(r.index, files, path)
384 matchPath(r.work, files, path)
385 }
386 return nil
387}

389func (r *repository) ignored(path string) bool {
390 if _, tracked := r.index[path]; tracked || r.conflicts[path] {
391 return false
392 }
393 for _, pattern := range r.ignores {
394 if glob(pattern, path) {
395 return true
396 }
397 }
398 return false
399}

401func (r *repository) apply(base, theirs tree, kind string, target *commit, message string, parents []*commit, baseCommit *commit) error {
402 merged, conflicts := threeWay(base, r.head().files, theirs)
403 changed := differences(r.head().files, merged)
404 if err := r.overwrite(append(append([]string{}, changed...), conflicts...)); err != nil {
405 return err
406 }
407 if kind == "merge" {
408 fmt.Fprintln(r.out, "Merge base "+baseCommit.id)
409 }
410 if kind != "merge" && len(conflicts) == 0 && same(merged, r.head().files) {
411 fmt.Fprintln(r.out, "nothing to commit")
412 return nil
413 }
414 for _, path := range changed {
415 matchPath(r.index, merged, path)
416 matchPath(r.work, merged, path)
417 }
418 if len(conflicts) == 0 {
419 r.record(parents, merged, message)
420 return nil
421 }
422 r.conflicts = map[string]bool{}
423 for _, path := range conflicts {
424 r.work[path] = conflictText(r.head().files, theirs, path)
425 r.conflicts[path] = true
426 fmt.Fprintln(r.out, "CONFLICT "+path)
427 }
428 r.pending = &operation{kind, target}
429 if kind == "merge" {
430 fmt.Fprintln(r.out, "Automatic merge failed; fix conflicts and then commit the result.")
431 } else if kind == "revert" {
432 fmt.Fprintln(r.out, "error: could not revert "+target.id)
433 } else {
434 fmt.Fprintln(r.out, "error: could not apply "+target.id)
435 }
436 return nil
437}

446func (r *repository) roots(extra bool) []*commit {
447 roots := []*commit{r.head()}
448 for _, c := range r.branches {

475func (r *repository) command(line string) error {
476 words := strings.Split(line, " ")
477 command := words[0]
478 h := r.head()
479 switch command {
480 case "WRITE":
481 args := strings.SplitN(line, " ", 3)
482 r.work[args[1]] = args[2]
483 case "REMOVE":
484 if _, exists := r.work[words[1]]; !exists {
485 return fmt.Errorf("no such file '%s'", words[1])
486 }
487 delete(r.work, words[1])
488 case "IGNORE":
489 r.ignores = append(r.ignores, words[1])
490 case "ADD":
491 path := words[1]
492 if path == "ALL" {
493 selected := paths(r.work, r.index)
494 for _, path := range selected {
495 if !r.ignored(path) {
496 matchPath(r.index, r.work, path)
497 }
498 }
499 r.conflicts = map[string]bool{}
500 } else {
501 _, inWork := r.work[path]
502 _, inIndex := r.index[path]
503 if inWork && r.ignored(path) {
504 return fmt.Errorf("path '%s' is ignored", path)
505 }
506 if !inWork && !inIndex && !r.conflicts[path] {
507 return fmt.Errorf("pathspec '%s' did not match any files", path)
508 }
509 matchPath(r.index, r.work, path)
510 delete(r.conflicts, path)
511 }
512 case "RM":
513 path := words[1]
514 if _, exists := r.index[path]; !exists {
515 return fmt.Errorf("pathspec '%s' did not match any files", path)
516 }
517 delete(r.index, path)
518 delete(r.work, path)
519 delete(r.conflicts, path)
520 case "UNSTAGE":
521 path := words[1]
522 _, inIndex := r.index[path]
523 _, inHead := h.files[path]
524 if !inIndex && !inHead {
525 return fmt.Errorf("pathspec '%s' did not match any files", path)
526 }
527 matchPath(r.index, h.files, path)
528 case "RESTORE", "CAT":
529 path, files := words[1], r.work
530 if command == "RESTORE" {
531 files = r.index
532 }
533 if len(words) == 3 {
534 c, err := r.revision(words[2])
535 if err != nil {
536 return err
537 }
538 files = c.files
539 if _, exists := files[path]; !exists {
540 return fmt.Errorf("path '%s' does not exist in '%s'", path, words[2])
541 }
542 } else if _, exists := files[path]; !exists {
543 if command == "CAT" {
544 return fmt.Errorf("no such file '%s'", path)
545 }
546 return fmt.Errorf("pathspec '%s' did not match any files", path)
547 }
548 if command == "CAT" {
549 fmt.Fprintln(r.out, files[path])
550 } else {
551 r.work[path] = files[path]
552 if len(words) == 3 {
553 r.index[path] = files[path]
554 delete(r.conflicts, path)
555 }
556 }
557 case "COMMIT":
558 if len(r.conflicts) > 0 {
559 return fmt.Errorf("unresolved conflicts")
560 }
561 message := strings.SplitN(line, " ", 2)[1]
562 parents := []*commit{h}
563 if r.pending != nil && r.pending.kind == "merge" {
564 parents = append(parents, r.pending.target)
565 } else if same(r.index, h.files) {
566 fmt.Fprintln(r.out, "nothing to commit")
567 return nil
568 }
569 r.record(parents, r.index, message)
570 r.pending = nil
571 case "AMEND":
572 if err := r.inProgress(); err != nil {
573 return err
574 }
575 if len(h.parents) == 0 {
576 return fmt.Errorf("cannot amend the root commit")
577 }
578 r.record(h.parents, r.index, strings.SplitN(line, " ", 2)[1])
579 case "STATUS":
580 if r.attached == "" {
581 fmt.Fprintln(r.out, "HEAD detached at "+h.id)
582 } else {
583 fmt.Fprintln(r.out, "On branch "+r.attached)
584 }
585 if r.pending != nil {
586 title := map[string]string{"merge": "Merging", "revert": "Reverting", "cherry-pick": "Cherry-picking"}[r.pending.kind]
587 fmt.Fprintln(r.out, title+" "+r.pending.target.id)
588 }
589 all := clone(r.work)
590 for path := range r.conflicts {
591 all[path] = ""
592 }
593 printed := false
594 for _, path := range paths(h.files, r.index, all) {
595 if r.conflicts[path] {
596 fmt.Fprintln(r.out, "UU "+path)
597 printed = true
598 continue
599 }
600 content, inIndex := r.index[path]
601 old, inHead := h.files[path]
602 work, inWork := r.work[path]
603 if inIndex {
604 x, y := byte('.'), byte('.')
605 if !inHead {
606 x = 'A'
607 } else if old != content {
608 x = 'M'
609 }
610 if !inWork {
611 y = 'D'
612 } else if work != content {
613 y = 'M'
614 }
615 if x != '.' || y != '.' {
616 fmt.Fprintf(r.out, "%c%c %s\n", x, y, path)
617 printed = true
618 }
619 } else {
620 if inHead {
621 fmt.Fprintln(r.out, "D. "+path)
622 printed = true
623 }
624 if inWork && !r.ignored(path) {
625 fmt.Fprintln(r.out, "?? "+path)
626 printed = true
627 }
628 }
629 }
630 if !printed {
631 fmt.Fprintln(r.out, "nothing to commit, working tree clean")
632 }
633 case "LS":
634 files := r.work
635 if len(words) == 2 {
636 c, err := r.revision(words[1])
637 if err != nil {
638 return err
639 }
640 files = c.files
641 }
642 if len(files) == 0 {
643 fmt.Fprintln(r.out, "(empty)")
644 }
645 for _, path := range paths(files) {
646 fmt.Fprintln(r.out, path+" "+files[path])
647 }
648 case "DIFF":
649 if len(words) == 1 {
650 work := tree{}
651 for path := range r.index {
652 matchPath(work, r.work, path)
653 }
654 r.changes(r.index, work)
655 } else if len(words) == 2 {
656 r.changes(h.files, r.index)
657 } else {
658 a, err := r.revision(words[1])
659 if err != nil {
660 return err
661 }
662 b, err := r.revision(words[2])
663 if err != nil {
664 return err
665 }
666 r.changes(a.files, b.files)
667 }
668 case "LOG":
669 limit := r.created
670 if len(words) == 2 {
671 limit, _ = strconv.Atoi(words[1])
672 }
673 for c, n := h, 0; c != nil && n < limit; n++ {
674 fmt.Fprintln(r.out, r.summary(c))
675 if len(c.parents) == 0 {
676 break
677 }
678 c = c.parents[0]
679 }
680 case "SHOW":
681 rev := "HEAD"
682 if len(words) == 2 {
683 rev = words[1]
684 }
685 c, err := r.revision(rev)
686 if err != nil {
687 return err
688 }
689 fmt.Fprintln(r.out, "commit "+r.summary(c))
690 parents := []string{}
691 for _, parent := range c.parents {
692 parents = append(parents, parent.id)
693 }
694 if len(parents) == 0 {
695 parents = append(parents, "none")
696 }
697 fmt.Fprintln(r.out, "parents: "+strings.Join(parents, " "))
698 r.changes(parentTree(c), c.files)
699 case "BLAME":
700 path := words[1]
701 content, exists := h.files[path]
702 if !exists {
703 return fmt.Errorf("path '%s' does not exist in 'HEAD'", path)
704 }
705 c := h
706 for len(c.parents) > 0 {
707 parent := c.parents[0]
708 if text, exists := parent.files[path]; !exists || text != content {
709 break
710 }
711 c = parent
712 }
713 fmt.Fprintf(r.out, "%s (%s) %s\n", c.id, c.message, content)
714 case "BRANCH", "TAG":
715 refs, label := r.branches, "branch"
716 if command == "TAG" {
717 refs, label = r.tags, "tag"
718 }
719 if words[1] == "-d" {
720 name := words[2]
721 c := refs[name]
722 if c == nil {
723 return fmt.Errorf("no such %s '%s'", label, name)
724 }
725 if command == "BRANCH" {
726 if r.attached == name {
727 return fmt.Errorf("cannot delete current branch '%s'", name)
728 }
729 if !ancestor(c, h) {
730 return fmt.Errorf("branch '%s' is not fully merged", name)
731 }
732 }
733 delete(refs, name)
734 fmt.Fprintf(r.out, "Deleted %s %s (was %s)\n", label, name, c.id)
735 } else {
736 name := words[1]
737 if refs[name] != nil {
738 return fmt.Errorf("%s '%s' already exists", label, name)
739 }
740 rev := "HEAD"
741 if len(words) == 3 {
742 rev = words[2]
743 }
744 c, err := r.revision(rev)
745 if err != nil {
746 return err
747 }
748 refs[name] = c
749 fmt.Fprintf(r.out, "Created %s %s at %s\n", label, name, c.id)
750 }
751 case "BRANCHES":
752 if r.attached == "" {
753 fmt.Fprintln(r.out, "* (HEAD detached at "+h.id+")")
754 }
755 for _, name := range referenceNames(r.branches) {
756 mark := " "
757 if r.attached == name {
758 mark = "*"
759 }
760 c := r.branches[name]
761 fmt.Fprintf(r.out, "%s %s %s %s\n", mark, name, c.id, c.message)
762 }
763 case "TAGS":
764 if len(r.tags) == 0 {
765 fmt.Fprintln(r.out, "(no tags)")
766 }
767 for _, name := range referenceNames(r.tags) {
768 c := r.tags[name]
769 fmt.Fprintf(r.out, "%s %s %s\n", name, c.id, c.message)
770 }
771 case "SWITCH":
772 name := words[1]
773 c := r.branches[name]
774 if c == nil {
775 return fmt.Errorf("no such branch '%s'", name)
776 }
777 if name == r.attached {
778 fmt.Fprintf(r.out, "Already on '%s'\n", name)
779 return nil
780 }
781 if err := r.inProgress(); err != nil {
782 return err
783 }
784 if err := r.switchTree(c.files); err != nil {
785 return err
786 }
787 r.attached, r.detached = name, nil
788 fmt.Fprintf(r.out, "Switched to branch '%s'\n", name)
789 case "DETACH":
790 c, err := r.revision(words[1])
791 if err != nil {
792 return err
793 }
794 if err := r.inProgress(); err != nil {
795 return err
796 }
797 if err := r.switchTree(c.files); err != nil {
798 return err
799 }
800 r.attached, r.detached = "", c
801 fmt.Fprintf(r.out, "HEAD is now at %s %s\n", c.id, c.message)
802 case "MERGEBASE":
803 a, err := r.revision(words[1])
804 if err != nil {
805 return err
806 }
807 b, err := r.revision(words[2])
808 if err != nil {
809 return err
810 }
811 fmt.Fprintln(r.out, mergeBase(a, b).id)
812 case "MERGE", "REVERT", "CHERRYPICK":
813 c, err := r.revision(words[1])
814 if err != nil {
815 return err
816 }
817 if err := r.inProgress(); err != nil {
818 return err
819 }
820 if command == "MERGE" {
821 if ancestor(c, h) {
822 fmt.Fprintln(r.out, "Already up to date")
823 return nil
824 }
825 if ancestor(h, c) {
826 if err := r.switchTree(c.files); err != nil {
827 return err
828 }
829 r.moveHead(c)
830 fmt.Fprintf(r.out, "Fast-forward %s..%s\n", h.id, c.id)
831 return nil
832 }
833 }
834 if err := r.requireClean(); err != nil {
835 return err
836 }
837 if command == "MERGE" {
838 base := mergeBase(h, c)
839 return r.apply(base.files, c.files, "merge", c, "Merge "+words[1], []*commit{h, c}, base)
840 }
841 if command == "REVERT" {
842 return r.apply(c.files, parentTree(c), "revert", c, "Revert \""+c.message+"\"", []*commit{h}, nil)
843 }
844 return r.apply(parentTree(c), c.files, "cherry-pick", c, c.message, []*commit{h}, nil)
845 case "REBASE":
846 upstream, err := r.revision(words[1])
847 if err != nil {
848 return err
849 }
850 if err := r.inProgress(); err != nil {
851 return err
852 }
853 if ancestor(upstream, h) {
854 fmt.Fprintln(r.out, "Already up to date")
855 return nil
856 }
857 if err := r.requireClean(); err != nil {
858 return err
859 }
860 list := []*commit{}
861 for c := h; !ancestor(c, upstream); c = c.parents[0] {
862 list = append(list, c)
863 }
864 for i, j := 0, len(list)-1; i < j; i, j = i+1, j-1 {
865 list[i], list[j] = list[j], list[i]
866 }
867 current := upstream.files
868 picks := make([]tree, len(list))
869 for i, c := range list {
870 merged, conflicts := threeWay(parentTree(c), current, c.files)
871 if len(conflicts) > 0 {
872 return fmt.Errorf("could not rebase %s: conflict in '%s'", c.id, conflicts[0])
873 }
874 if !same(merged, current) {
875 picks[i] = merged
876 current = merged
877 }
878 }
879 if err := r.switchTree(current); err != nil {
880 return err
881 }
882 tip := upstream
883 for i, c := range list {
884 if picks[i] == nil {
885 fmt.Fprintf(r.out, "skip %s %s\n", c.id, c.message)
886 } else {
887 next := r.create([]*commit{tip}, picks[i], c.message)
888 fmt.Fprintf(r.out, "pick %s -> %s %s\n", c.id, next.id, c.message)
889 tip = next
890 }
891 }
892 r.moveHead(tip)
893 fmt.Fprintf(r.out, "HEAD is now at %s %s\n", tip.id, tip.message)
894 case "RESET":
895 c, err := r.revision(words[2])
896 if err != nil {
897 return err
898 }
899 if words[1] == "HARD" {
900 all := tree{}
901 for path := range r.conflicts {
902 all[path] = ""
903 }
904 for _, path := range paths(r.index, c.files, all) {
905 matchPath(r.work, c.files, path)
906 }
907 }
908 if words[1] != "SOFT" {
909 r.index = clone(c.files)
910 }
911 r.moveHead(c)
912 r.conflicts = map[string]bool{}
913 r.pending = nil
914 fmt.Fprintf(r.out, "HEAD is now at %s %s\n", c.id, c.message)
915 case "STASH":
916 action := ""
917 if len(words) > 1 {
918 action = words[1]
919 }
920 if action == "LIST" {
921 if len(r.stash) == 0 {
922 fmt.Fprintln(r.out, "(no stash entries)")
923 }
924 for i, entry := range r.stash {
925 fmt.Fprintf(r.out, "stash@{%d}: %s\n", i, entry.label)
926 }
927 return nil
928 }
929 if action != "" && len(r.stash) == 0 {
930 return fmt.Errorf("no stash entries")
931 }
932 if action == "DROP" {
933 label := r.stash[0].label
934 r.stash = r.stash[1:]
935 fmt.Fprintln(r.out, "Dropped stash@{0}: "+label)
936 return nil
937 }
938 if err := r.inProgress(); err != nil {
939 return err
940 }
941 if action == "POP" {
942 if err := r.requireClean(); err != nil {
943 return err
944 }
945 entry := r.stash[0]
946 merged, conflicts := threeWay(entry.base.files, h.files, entry.files)
947 if len(conflicts) > 0 {
948 return fmt.Errorf("stash conflict in '%s'", conflicts[0])
949 }
950 changed := differences(h.files, merged)
951 if err := r.overwrite(changed); err != nil {
952 return err
953 }
954 for _, path := range changed {
955 matchPath(r.work, merged, path)
956 }
957 r.stash = r.stash[1:]
958 fmt.Fprintln(r.out, "Restored stash@{0}: "+entry.label)
959 r.changes(h.files, merged)
960 } else {
961 if !r.uncommitted() {
962 fmt.Fprintln(r.out, "No local changes to save")
963 return nil
964 }
965 saved := tree{}
966 for _, path := range paths(r.index, h.files) {
967 matchPath(saved, r.work, path)
968 matchPath(r.work, h.files, path)
969 }
970 r.index = clone(h.files)
971 location := r.attached
972 if location == "" {
973 location = "(no branch)"
974 }
975 label := "WIP on " + location + ": " + h.id + " " + h.message
976 r.stash = append([]stashEntry{{label, h, saved}}, r.stash...)
977 fmt.Fprintln(r.out, "Saved stash@{0}: "+label)
978 }
979 case "GRAPH":
980 all := reachable(r.roots(false))
981 ordered := []*commit{}
982 for c := range all {
983 ordered = append(ordered, c)
984 }
985 sort.Slice(ordered, func(i, j int) bool {
986 return ordered[i].sequence > ordered[j].sequence
987 })
988 lanes := []*commit{}
989 for _, c := range ordered {
990 at := -1
991 for i, held := range lanes {
992 if held == c {
993 at = i
994 break
995 }
996 }
997 if at < 0 {
998 at = len(lanes)
999 lanes = append(lanes, c)
1000 }
1001 symbols := []string{}
1002 kept := []*commit{}
1003 for i, held := range lanes {
1004 symbol := "|"
1005 if held == c {
1006 symbol = "/"
1007 if i == at {
1008 symbol = "*"
1009 }
1010 }
1011 symbols = append(symbols, symbol)
1012 if held != c || i == at {
1013 kept = append(kept, held)
1014 }
1015 }
1016 fmt.Fprintln(r.out, strings.Join(symbols, " ")+" "+r.summary(c))
1017 lanes = kept
1018 if len(c.parents) == 0 {
1019 lanes = append(lanes[:at], lanes[at+1:]...)
1020 } else {
1021 lanes[at] = c.parents[0]
1022 if len(c.parents) == 2 {
1023 exists := false
1024 for _, held := range lanes {
1025 if held == c.parents[1] {
1026 exists = true
1027 }
1028 }
1029 if !exists {
1030 lanes = append(lanes, nil)
1031 copy(lanes[at+2:], lanes[at+1:])
1032 lanes[at+1] = c.parents[1]
1033 }
1034 }
1035 }
1036 }
1037 case "DESCRIBE":
1038 rev := "HEAD"
1039 if len(words) == 2 {
1040 rev = words[1]
1041 }
1042 c, err := r.revision(rev)
1043 if err != nil {
1044 return err
1045 }
1046 d, best, distance := distances(c), "", 0
1047 for _, name := range referenceNames(r.tags) {
1048 if n, exists := d[r.tags[name]]; exists && (best == "" || n < distance) {
1049 best, distance = name, n
1050 }
1051 }
1052 if best == "" {
1053 return fmt.Errorf("no tag can describe '%s'", c.id)
1054 }
1055 if distance == 0 {
1056 fmt.Fprintln(r.out, best)
1057 } else {
1058 fmt.Fprintf(r.out, "%s-%d-g%s\n", best, distance, c.id)
1059 }
1060 case "GC":
1061 keep, removed := reachable(r.roots(true)), 0
1062 for id, c := range r.commits {
1063 if !keep[c] {
1064 delete(r.commits, id)
1065 removed++
1066 }
1067 }
1068 fmt.Fprintf(r.out, "Removed unreachable commits: %d\n", removed)
1069 }
1070 return nil
1071}
