87static char *format(const char *pattern, ...) {
88 va_list args, saved;
89 va_start(args, pattern);
90 va_copy(saved, args);
91 int n = vsnprintf(NULL, 0, pattern, args);
92 va_end(args);
93 char *out = allocate((size_t)n + 1);
94 vsnprintf(out, (size_t)n + 1, pattern, saved);
95 va_end(saved);

111static void append(Buffer *b, const char *s) {
112 size_t n = strlen(s);
113 if (b->length + n + 1 > b->capacity) {
114 b->capacity = (b->length + n + 1) * 2;
115 b->data = resize(b->data, b->capacity);
116 }
117 memcpy(b->data + b->length, s, n);
118 b->length += n;
119 b->data[b->length] = 0;
120}

244static Commit *create(Commit **parents, size_t count, Tree tree, char *message) {
245 Buffer text = {0};
246 append(&text, format("commit %zu\n", commit_count));
247 for (size_t i = 0; i < count; i++) append(&text, format("parent %s\n", parents[i]->id));
248 Strings sorted = keys(tree);
249 for (size_t i = 0; i < sorted.count; i++) append(&text, format("file %s %s\n", sorted.items[i], get(tree, sorted.items[i])));
250 append(&text, format("message %s\n", message));
251 uint64_t hash = 0;
252 for (size_t i = 0; i < text.length; i++) hash = (hash * 131 + (unsigned char)text.data[i]) % 2147483647;
253 Commit *commit = allocate(sizeof(*commit));
254 commit->id = format("%08llx", (unsigned long long)hash);
255 commit->sequence = commit_count;
256 commit->parent_count = count;
257 for (size_t i = 0; i < count; i++) commit->parents[i] = parents[i];
258 commit->tree = clone(tree);
259 commit->message = message;
260 commit->alive = true;
261 APPEND(commits, commit_count, commit);
262 return commit;
263}

265static void record(Commit **parents, size_t count, Tree tree, char *message) {
266 Commit *commit = create(parents, count, tree, message);
267 Tree before = parents[0]->tree;
268 int added = 0, modified = 0, deleted = 0;
269 Strings changed = differences(before, tree);
270 for (size_t i = 0; i < changed.count; i++) {
271 if (!get(before, changed.items[i])) added++;
272 else if (!get(tree, changed.items[i])) deleted++;
273 else modified++;
274 }
275 move_head(commit);
276 printf("[%s %s] %s (%d added, %d modified, %d deleted)\n", branch ? branch : "detached HEAD", commit->id, message, added, modified, deleted);
277}

279static Commit *resolve(const char *revision) {
280 const char *tilde = strchr(revision, '~');
281 char *anchor = tilde ? slice(revision, (size_t)(tilde - revision)) : copy(revision);
282 Commit *commit = !strcmp(anchor, "HEAD") ? head() : ref_get(branches, anchor);
283 if (!commit) commit = ref_get(tags, anchor);
284 if (!commit) for (size_t i = 0; i < commit_count; i++) if (commits[i]->alive && !strcmp(commits[i]->id, anchor)) commit = commits[i];
285 int steps = tilde ? atoi(tilde + 1) : 0;
286 while (commit && steps-- > 0) commit = commit->parent_count ? commit->parents[0] : NULL;
287 if (!commit) error("error: unknown revision '%s'", revision);
288 return commit;
289}

291static int *distances(Commit *start) {
292 int *result = allocate(commit_count * sizeof(*result));
293 for (size_t i = 0; i < commit_count; i++) result[i] = -1;
294 Commit **queue = NULL;
295 size_t count = 0, at = 0;
296 APPEND(queue, count, start);
297 result[start->sequence] = 0;
298 while (at < count) {
299 Commit *commit = queue[at++];
300 for (size_t i = 0; i < commit->parent_count; i++) {
301 Commit *parent = commit->parents[i];
302 if (result[parent->sequence] < 0) {
303 result[parent->sequence] = result[commit->sequence] + 1;
304 APPEND(queue, count, parent);
305 }
306 }
307 }
308 free(queue);
309 return result;
310}

312static bool ancestor(Commit *a, Commit *b) {
313 int *d = distances(b);

315 free(d);
316 return result;

319static Commit *merge_base(Commit *a, Commit *b) {
320 int *x = distances(a), *y = distances(b);
321 Commit *best = NULL;
322 int smallest = 0;
323 for (size_t i = 0; i < commit_count; i++) if (x[i] >= 0 && y[i] >= 0) {
324 bool candidate = true;
325 for (size_t j = 0; j < commit_count && candidate; j++) if (j != i && x[j] >= 0 && y[j] >= 0 && ancestor(commits[i], commits[j])) candidate = false;
326 if (!candidate) continue;
327 int distance = x[i] > y[i] ? x[i] : y[i];
328 if (!best || distance < smallest || (distance == smallest && i > best->sequence)) {
329 best = commits[i];
330 smallest = distance;
331 }
332 }
333 free(x);
334 free(y);
335 return best;
336}

338static char *summary(Commit *commit) {
339 Buffer refs = {0};
340 if (head() == commit) append(&refs, branch ? format("HEAD -> %s", branch) : "HEAD");
341 for (size_t i = 0; i < branches.count; i++) if (branches.items[i].commit == commit && (!branch || strcmp(branches.items[i].name, branch))) {
342 if (refs.length) append(&refs, ", ");
343 append(&refs, branches.items[i].name);
344 }
345 for (size_t i = 0; i < tags.count; i++) if (tags.items[i].commit == commit) {
346 if (refs.length) append(&refs, ", ");
347 append(&refs, format("tag: %s", tags.items[i].name));
348 }
349 return format("%s%s %s", commit->id, refs.length ? format(" (%s)", refs.data) : "", commit->message);
350}

391static Merge merge(Tree base, Tree ours, Tree theirs) {
392 Merge result = {0};
393 Strings all = paths(base, ours);
394 for (size_t i = 0; i < theirs.count; i++) set_add(&all, theirs.items[i].path);
395 all = ordered(all);
396 for (size_t i = 0; i < all.count; i++) {
397 char *path = all.items[i];
398 if (agree(theirs, base, path) || agree(ours, theirs, path)) match(&result.tree, ours, path);
399 else if (agree(ours, base, path)) match(&result.tree, theirs, path);
400 else {
401 set_add(&result.conflicts, path);
402 match(&result.tree, ours, path);
403 char *a = get(ours, path), *b = get(theirs, path);
404 put(&result.contents, path, format("<<<<<<< %s ======= %s >>>>>>>", a ? a : "(deleted)", b ? b : "(deleted)"));
405 }
406 }
407 return result;
408}

410static void apply(Tree base, Tree theirs, char *kind, Commit *chosen, char *message, Commit **parents, size_t parent_count, Commit *base_commit) {
411 Merge result = merge(base, head()->tree, theirs);
412 Strings changed = differences(head()->tree, result.tree), checked = {0};
413 for (size_t i = 0; i < changed.count; i++) set_add(&checked, changed.items[i]);
414 for (size_t i = 0; i < result.conflicts.count; i++) set_add(&checked, result.conflicts.items[i]);
415 overwrite_check(checked);
416 if (!strcmp(kind, "merge")) printf("Merge base %s\n", base_commit->id);
417 else if (!result.conflicts.count && !changed.count) {
418 puts("nothing to commit");
419 return;
420 }
421 for (size_t i = 0; i < changed.count; i++) {
422 match(&index_tree, result.tree, changed.items[i]);
423 match(&working, result.tree, changed.items[i]);
424 }
425 if (!result.conflicts.count) record(parents, parent_count, result.tree, message);
426 else {
427 for (size_t i = 0; i < result.conflicts.count; i++) {
428 char *path = result.conflicts.items[i];
429 put(&working, path, get(result.contents, path));
430 printf("CONFLICT %s\n", path);
431 }
432 conflicts = result.conflicts;
433 operation = kind;
434 target = chosen;
435 if (!strcmp(kind, "merge")) puts("Automatic merge failed; fix conflicts and then commit the result.");
436 else printf("error: could not %s %s\n", !strcmp(kind, "revert") ? "revert" : "apply", chosen->id);
437 }
438}

440static bool glob(const char *pattern, const char *path) {
441 size_t n = strlen(path), m = strlen(pattern);
442 bool *previous = allocate((n + 1) * sizeof(*previous));
443 previous[0] = true;
444 for (size_t i = 0; i < m; i++) {
445 bool *next = allocate((n + 1) * sizeof(*next));
446 for (size_t j = 0; j <= n; j++) next[j] = pattern[i] == '*' ? previous[j] || (j && next[j - 1]) : j && previous[j - 1] && (pattern[i] == '?' || pattern[i] == path[j - 1]);
447 free(previous);
448 previous = next;
449 }
450 bool result = previous[n];
451 free(previous);
452 return result;
453}

499static void command(const char *line) {
500 Strings words = split(line);
501 char **w = words.items, *cmd = w[0], *path = words.count > 1 ? w[1] : "";
502 if (!strcmp(cmd, "WRITE")) {
503 const char *space = strchr(strchr(line, ' ') + 1, ' ');
504 put(&working, path, copy(space + 1));
505 } else if (!strcmp(cmd, "REMOVE")) {
506 if (!get(working, path)) error("error: no such file '%s'", path);
507 remove_file(&working, path);
508 } else if (!strcmp(cmd, "IGNORE")) APPEND(ignores.items, ignores.count, path);
509 else if (!strcmp(cmd, "ADD")) {
510 if (!strcmp(path, "ALL")) {
511 Strings all = paths(working, index_tree), selected = {0};
512 for (size_t i = 0; i < all.count; i++) if (!ignored(all.items[i])) set_add(&selected, all.items[i]);
513 for (size_t i = 0; i < selected.count; i++) match(&index_tree, working, selected.items[i]);
514 conflicts.count = 0;
515 } else {
516 if (get(working, path) && ignored(path)) error("error: path '%s' is ignored", path);
517 pathspec(get(working, path) || get(index_tree, path) || contains(conflicts, path), path);
518 match(&index_tree, working, path);
519 set_remove(&conflicts, path);
520 }
521 } else if (!strcmp(cmd, "RM")) {
522 pathspec(get(index_tree, path) != NULL, path);
523 remove_file(&index_tree, path);
524 remove_file(&working, path);
525 set_remove(&conflicts, path);
526 } else if (!strcmp(cmd, "UNSTAGE")) {
527 pathspec(get(index_tree, path) || get(head()->tree, path), path);
528 match(&index_tree, head()->tree, path);
529 } else if (!strcmp(cmd, "RESTORE")) {
530 if (words.count == 2) {
531 pathspec(get(index_tree, path) != NULL, path);
532 put(&working, path, get(index_tree, path));
533 } else {
534 Commit *chosen = resolve(w[2]);
535 if (!get(chosen->tree, path)) error("error: path '%s' does not exist in '%s'", path, w[2]);
536 match(&index_tree, chosen->tree, path);
537 match(&working, chosen->tree, path);
538 set_remove(&conflicts, path);
539 }
540 } else if (!strcmp(cmd, "COMMIT")) {
541 if (conflicts.count) error("error: unresolved conflicts");
542 bool merging = operation && !strcmp(operation, "merge");
543 if (!merging && equal(index_tree, head()->tree)) {
544 puts("nothing to commit");
545 return;
546 }
547 Commit *parents[] = {head(), target};
548 record(parents, merging ? 2 : 1, index_tree, copy(strchr(line, ' ') + 1));
549 operation = NULL;
550 target = NULL;
551 } else if (!strcmp(cmd, "AMEND")) {
552 progress_check();
553 Commit *before = head();
554 if (!before->parent_count) error("error: cannot amend the root commit");
555 record(before->parents, before->parent_count, index_tree, copy(strchr(line, ' ') + 1));
556 } else if (!strcmp(cmd, "STATUS")) {
557 if (branch) printf("On branch %s\n", branch);
558 else printf("HEAD detached at %s\n", head()->id);
559 if (operation) printf("%s %s\n", !strcmp(operation, "merge") ? "Merging" : !strcmp(operation, "revert") ? "Reverting" : "Cherry-picking", target->id);
560 Strings all = paths(head()->tree, index_tree);
561 for (size_t i = 0; i < working.count; i++) set_add(&all, working.items[i].path);
562 for (size_t i = 0; i < conflicts.count; i++) set_add(&all, conflicts.items[i]);
563 all = ordered(all);
564 bool any = false;
565 for (size_t i = 0; i < all.count; i++) {
566 char *p = all.items[i];
567 if (contains(conflicts, p)) {
568 printf("UU %s\n", p);
569 any = true;
570 } else if (get(index_tree, p)) {
571 char x = !get(head()->tree, p) ? 'A' : agree(head()->tree, index_tree, p) ? '.' : 'M', y = !get(working, p) ? 'D' : agree(working, index_tree, p) ? '.' : 'M';
572 if (x != '.' || y != '.') {
573 printf("%c%c %s\n", x, y, p);
574 any = true;
575 }
576 } else {
577 if (get(head()->tree, p)) {
578 printf("D. %s\n", p);
579 any = true;
580 }
581 if (get(working, p) && !ignored(p)) {
582 printf("?? %s\n", p);
583 any = true;
584 }
585 }
586 }
587 if (!any) puts("nothing to commit, working tree clean");
588 } else if (!strcmp(cmd, "LS")) {
589 Tree tree = words.count == 1 ? working : resolve(path)->tree;
590 Strings all = keys(tree);
591 if (!all.count) puts("(empty)");
592 for (size_t i = 0; i < all.count; i++) printf("%s %s\n", all.items[i], get(tree, all.items[i]));
593 } else if (!strcmp(cmd, "CAT")) {
594 Tree tree = words.count == 2 ? working : resolve(w[2])->tree;
595 char *text = get(tree, path);
596 if (!text) {
597 if (words.count == 2) error("error: no such file '%s'", path);
598 else error("error: path '%s' does not exist in '%s'", path, w[2]);
599 }
600 puts(text);
601 } else if (!strcmp(cmd, "DIFF")) {
602 if (words.count == 1) {
603 Tree tracked = {0};
604 for (size_t i = 0; i < index_tree.count; i++) if (get(working, index_tree.items[i].path)) match(&tracked, working, index_tree.items[i].path);
605 changes(index_tree, tracked);
606 } else if (words.count == 2) changes(head()->tree, index_tree);
607 else {
608 Commit *a = resolve(path), *b = resolve(w[2]);
609 changes(a->tree, b->tree);
610 }
611 } else if (!strcmp(cmd, "LOG")) {
612 Commit *chosen = head();
613 int left = words.count == 1 ? (int)commit_count : atoi(path);
614 while (left-- > 0) {
615 puts(summary(chosen));
616 if (!chosen->parent_count) break;
617 chosen = chosen->parents[0];
618 }
619 } else if (!strcmp(cmd, "SHOW")) {
620 Commit *chosen = resolve(words.count == 1 ? "HEAD" : path);
621 printf("commit %s\nparents:", summary(chosen));
622 if (!chosen->parent_count) printf(" none");
623 for (size_t i = 0; i < chosen->parent_count; i++) printf(" %s", chosen->parents[i]->id);
624 putchar('\n');
625 changes(chosen->parent_count ? chosen->parents[0]->tree : (Tree){0}, chosen->tree);
626 } else if (!strcmp(cmd, "BLAME")) {
627 Commit *chosen = head();
628 char *text = get(chosen->tree, path);
629 if (!text) error("error: path '%s' does not exist in 'HEAD'", path);
630 while (chosen->parent_count && get(chosen->parents[0]->tree, path) && !strcmp(get(chosen->parents[0]->tree, path), text)) chosen = chosen->parents[0];
631 printf("%s (%s) %s\n", chosen->id, chosen->message, text);
632 } else if (!strcmp(cmd, "BRANCH") || !strcmp(cmd, "TAG")) {
633 bool tag = !strcmp(cmd, "TAG");
634 Refs *refs = tag ? &tags : &branches;
635 char *noun = tag ? "tag" : "branch";
636 if (!strcmp(path, "-d")) {
637 char *name = w[2];
638 Commit *chosen = ref_get(*refs, name);
639 if (!chosen) error("error: no such %s '%s'", noun, name);
640 if (!tag) {
641 if (branch && !strcmp(branch, name)) error("error: cannot delete current branch '%s'", name);
642 if (!ancestor(chosen, head())) error("error: branch '%s' is not fully merged", name);
643 }
644 ref_remove(refs, name);
645 printf("Deleted %s %s (was %s)\n", noun, name, chosen->id);
646 } else {
647 if (ref_get(*refs, path)) error("error: %s '%s' already exists", noun, path);
648 Commit *chosen = resolve(words.count == 2 ? "HEAD" : w[2]);
649 ref_put(refs, path, chosen);
650 printf("Created %s %s at %s\n", noun, path, chosen->id);
651 }
652 } else if (!strcmp(cmd, "BRANCHES")) {
653 if (!branch) printf("* (HEAD detached at %s)\n", head()->id);
654 for (size_t i = 0; i < branches.count; i++) printf("%s %s %s %s\n", branch && !strcmp(branch, branches.items[i].name) ? "*" : " ", branches.items[i].name, branches.items[i].commit->id, branches.items[i].commit->message);
655 } else if (!strcmp(cmd, "TAGS")) {
656 if (!tags.count) puts("(no tags)");
657 for (size_t i = 0; i < tags.count; i++) printf("%s %s %s\n", tags.items[i].name, tags.items[i].commit->id, tags.items[i].commit->message);
658 } else if (!strcmp(cmd, "SWITCH")) {
659 Commit *chosen = ref_get(branches, path);
660 if (!chosen) error("error: no such branch '%s'", path);
661 if (branch && !strcmp(branch, path)) {
662 printf("Already on '%s'\n", path);
663 return;
664 }
665 progress_check();
666 switch_tree(chosen->tree);
667 branch = path;
668 printf("Switched to branch '%s'\n", path);
669 } else if (!strcmp(cmd, "DETACH")) {
670 Commit *chosen = resolve(path);
671 progress_check();
672 switch_tree(chosen->tree);
673 branch = NULL;
674 detached = chosen;
675 printf("HEAD is now at %s %s\n", chosen->id, chosen->message);
676 } else if (!strcmp(cmd, "MERGEBASE")) {
677 Commit *a = resolve(path), *b = resolve(w[2]);
678 puts(merge_base(a, b)->id);
679 } else if (!strcmp(cmd, "MERGE")) {
680 Commit *chosen = resolve(path), *before = head();
681 progress_check();
682 if (ancestor(chosen, before)) {
683 puts("Already up to date");
684 return;
685 }
686 if (ancestor(before, chosen)) {
687 switch_tree(chosen->tree);
688 move_head(chosen);
689 printf("Fast-forward %s..%s\n", before->id, chosen->id);
690 return;
691 }
692 clean_check();
693 Commit *base = merge_base(before, chosen), *parents[] = {before, chosen};
694 apply(base->tree, chosen->tree, "merge", chosen, format("Merge %s", path), parents, 2, base);
695 } else if (!strcmp(cmd, "RESET")) {
696 Commit *chosen = resolve(w[2]);
697 if (!strcmp(path, "HARD")) {
698 Strings all = paths(index_tree, chosen->tree);
699 for (size_t i = 0; i < conflicts.count; i++) set_add(&all, conflicts.items[i]);
700 for (size_t i = 0; i < all.count; i++) match(&working, chosen->tree, all.items[i]);
701 }
702 if (strcmp(path, "SOFT")) index_tree = clone(chosen->tree);
703 move_head(chosen);
704 conflicts.count = 0;
705 operation = NULL;
706 target = NULL;
707 printf("HEAD is now at %s %s\n", chosen->id, chosen->message);
708 } else if (!strcmp(cmd, "REVERT") || !strcmp(cmd, "CHERRYPICK")) {
709 Commit *chosen = resolve(path);
710 progress_check();
711 clean_check();
712 Tree parent = chosen->parent_count ? chosen->parents[0]->tree : (Tree){0};
713 bool revert = !strcmp(cmd, "REVERT");
714 Commit *parents[] = {head()};
715 apply(revert ? chosen->tree : parent, revert ? parent : chosen->tree, revert ? "revert" : "cherry-pick", chosen, revert ? format("Revert \"%s\"", chosen->message) : chosen->message, parents, 1, NULL);
716 } else if (!strcmp(cmd, "REBASE")) {
717 Commit *upstream = resolve(path);
718 progress_check();
719 if (ancestor(upstream, head())) {
720 puts("Already up to date");
721 return;
722 }
723 clean_check();
724 Commit **old = NULL, *cursor = head();
725 size_t count = 0;
726 while (!ancestor(cursor, upstream)) {
727 APPEND(old, count, cursor);
728 cursor = cursor->parents[0];
729 }
730 Replay *replay = allocate(count * sizeof(*replay));
731 Tree tree = clone(upstream->tree);
732 for (size_t i = 0; i < count; i++) {
733 Commit *chosen = old[count - i - 1];
734 Merge result = merge(chosen->parents[0]->tree, tree, chosen->tree);
735 if (result.conflicts.count) error("error: could not rebase %s: conflict in '%s'", chosen->id, result.conflicts.items[0]);
736 replay[i] = (Replay){chosen, result.tree, !equal(tree, result.tree)};
737 tree = result.tree;
738 }
739 switch_tree(tree);
740 Commit *current = upstream;
741 for (size_t i = 0; i < count; i++) if (replay[i].pick) {
742 Commit *parents[] = {current};
743 Commit *created = create(parents, 1, replay[i].tree, replay[i].original->message);
744 printf("pick %s -> %s %s\n", replay[i].original->id, created->id, created->message);
745 current = created;
746 } else printf("skip %s %s\n", replay[i].original->id, replay[i].original->message);
747 move_head(current);
748 printf("HEAD is now at %s %s\n", current->id, current->message);
749 } else if (!strcmp(cmd, "STASH")) {
750 if (words.count > 1 && !strcmp(path, "LIST")) {
751 if (!stash_count) puts("(no stash entries)");
752 for (size_t i = 0; i < stash_count; i++) printf("stash@{%zu}: %s\n", i, stashes[i].label);
753 return;
754 }
755 if (words.count > 1) {
756 if (!stash_count) error("error: no stash entries");
757 Stash stash = stashes[0];
758 if (!strcmp(path, "DROP")) {
759 memmove(stashes, stashes + 1, (--stash_count) * sizeof(*stashes));
760 printf("Dropped stash@{0}: %s\n", stash.label);
761 return;
762 }
763 progress_check();
764 clean_check();
765 Merge result = merge(stash.base->tree, head()->tree, stash.tree);
766 if (result.conflicts.count) error("error: stash conflict in '%s'", result.conflicts.items[0]);
767 Strings changed = differences(head()->tree, result.tree);
768 overwrite_check(changed);
769 for (size_t i = 0; i < changed.count; i++) match(&working, result.tree, changed.items[i]);
770 memmove(stashes, stashes + 1, (--stash_count) * sizeof(*stashes));
771 printf("Restored stash@{0}: %s\n", stash.label);
772 changes(head()->tree, result.tree);
773 } else {
774 progress_check();
775 if (!dirty()) {
776 puts("No local changes to save");
777 return;
778 }
779 Tree saved = {0};
780 Strings tracked = paths(index_tree, head()->tree);
781 for (size_t i = 0; i < tracked.count; i++) if (get(working, tracked.items[i])) match(&saved, working, tracked.items[i]);
782 for (size_t i = 0; i < tracked.count; i++) match(&working, head()->tree, tracked.items[i]);
783 index_tree = clone(head()->tree);
784 char *label = format("WIP on %s: %s %s", branch ? branch : "(no branch)", head()->id, head()->message);
785 stashes = resize(stashes, (stash_count + 1) * sizeof(*stashes));
786 memmove(stashes + 1, stashes, stash_count * sizeof(*stashes));
787 stashes[0] = (Stash){label, head(), saved};
788 stash_count++;
789 printf("Saved stash@{0}: %s\n", label);
790 }
791 } else if (!strcmp(cmd, "GRAPH")) {
792 bool *keep = reachable(false);
793 Commit **lanes = NULL;
794 size_t count = 0;
795 for (size_t sequence = commit_count; sequence; sequence--) {
796 Commit *chosen = commits[sequence - 1];
797 if (!keep[sequence - 1]) continue;
798 if (!commit_in(lanes, count, chosen)) APPEND(lanes, count, chosen);
799 size_t first = 0;
800 while (lanes[first] != chosen) first++;
801 for (size_t i = 0; i < count; i++) printf("%s%s", i ? " " : "", i == first ? "*" : lanes[i] == chosen ? "/" : "|");
802 printf(" %s\n", summary(chosen));
803 size_t next = 0;
804 for (size_t i = 0; i < count; i++) if (i == first || lanes[i] != chosen) lanes[next++] = lanes[i];
805 count = next;
806 if (!chosen->parent_count) {
807 memmove(lanes + first, lanes + first + 1, (count - first - 1) * sizeof(*lanes));
808 count--;
809 } else {
810 lanes[first] = chosen->parents[0];
811 if (chosen->parent_count == 2 && !commit_in(lanes, count, chosen->parents[1])) {
812 lanes = resize(lanes, (count + 1) * sizeof(*lanes));
813 memmove(lanes + first + 2, lanes + first + 1, (count - first - 1) * sizeof(*lanes));
814 lanes[first + 1] = chosen->parents[1];
815 count++;
816 }
817 }
818 }
819 free(keep);
820 free(lanes);
821 } else if (!strcmp(cmd, "DESCRIBE")) {
822 Commit *chosen = resolve(words.count == 1 ? "HEAD" : path);
823 int *d = distances(chosen), best_distance = 0;
824 char *best = NULL;
825 for (size_t i = 0; i < tags.count; i++) {
826 int distance = d[tags.items[i].commit->sequence];
827 if (distance >= 0 && (!best || distance < best_distance)) {
828 best = tags.items[i].name;
829 best_distance = distance;
830 }
831 }
832 if (!best) error("error: no tag can describe '%s'", chosen->id);
833 if (!best_distance) puts(best);
834 else printf("%s-%d-g%s\n", best, best_distance, chosen->id);
835 free(d);
836 } else if (!strcmp(cmd, "GC")) {
837 bool *keep = reachable(true);
838 size_t removed = 0;
839 for (size_t i = 0; i < commit_count; i++) if (commits[i]->alive && !keep[i]) {
840 commits[i]->alive = false;
841 removed++;
842 }
843 printf("Removed unreachable commits: %zu\n", removed);
844 free(keep);
845 }
846}

848static char *read_line(void) {
849 Buffer b = {0};
850 int c;
851 while ((c = getchar()) != EOF && c != '\n') append(&b, (char[]){(char)c, 0});
852 if (c == EOF && !b.length) return NULL;
853 return finish(&b);
854}
855int main(void) {
856 ref_put(&branches, "main", create(NULL, 0, (Tree){0}, "init"));
857 char *line;
858 while ((line = read_line()) && strcmp(line, "EXIT")) {
859 if (setjmp(failure)) puts(error_reason);
860 else command(line);
861 }
862 return 0;
