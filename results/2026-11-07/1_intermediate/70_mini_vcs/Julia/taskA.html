1const Tree=Dict{String,String}
2struct CommandError <: Exception
3 message::String
4end
5fail(s) = throw(CommandError(s))
6check(b, s) = b ? nothing : fail(s)
7paths(trees...) = sort!(collect(union((Set(keys(t)) for t in trees)...)))
8agree(a, b, p) = get(a, p, nothing)==get(b, p, nothing)
9differing(a, b) = [p for p in paths(a, b) if !agree(a, b, p)]
10function matchpath!(target, source, p)
11 haskey(source, p) ? (target[p]=source[p]) : delete!(target, p)
12end
13function changes(a, b)
14 lines=String[]
15 for p in differing(a, b)
16 push!(
17 lines,
18 !haskey(a, p) ? "A $p: $(b[p])" :
19 !haskey(b, p) ? "D $p: $(a[p])" : "M $p: $(a[p]) -> $(b[p])",
20 )
21 end
22 isempty(lines) ? ["(no changes)"] : lines
23end
24function merge_trees(base, ours, theirs)
25 merged=Tree()
26 conflicts=String[]
27 for p in paths(base, ours, theirs)
28 if agree(theirs, base, p) || agree(ours, theirs, p)
29 matchpath!(merged, ours, p)
30 elseif agree(ours, base, p)
31 matchpath!(merged, theirs, p)
32 else
33 push!(conflicts, p)
34 matchpath!(merged, ours, p)
35 end
36 end
37 merged, conflicts
38end
39function globmatch(text, pattern)
40 previous=falses(length(text)+1)
41 previous[1]=true
42 for c in pattern
43 next=falses(length(text)+1)
44 c=='*' && (next[1]=previous[1])
45 for i in eachindex(text)
46 next[i+1]=c=='*' ? previous[i+1] || next[i] : previous[i] && (c=='?' || c==text[i])
47 end
48 previous=next
49 end
50 previous[end]
51end
52struct Commit
53 id::String
54 sequence::Int
55 parents::Vector{Int}
56 tree::Tree
57 message::String
58end

60 label::String

64mutable struct Repository
65 commits::Vector{Commit}
66 active::Dict{String,Int}
67 branches::Dict{String,Int}

70 detached::Int

78function createcommit!(r, parents, tree, message)
79 sequence=length(r.commits)
80 lines=["commit $sequence"]
81 append!(lines, ["parent $(r.commits[p].id)" for p in parents])
82 append!(lines, ["file $p $(tree[p])" for p in sort!(collect(keys(tree)))])
83 push!(lines, "message $message")
84 h=0
85 for c in join(lines, "\n")*"\n"
86 h=mod(h*131+Int(c), 2147483647)
87 end
88 id=lpad(string(h; base = 16), 8, '0')
89 push!(r.commits, Commit(id, sequence, copy(parents), copy(tree), message))
90 r.active[id]=sequence+1
91 sequence+1
92end
93function Repository()
94 r=Repository(
95 Commit[],
96 Dict{String,Int}(),
97 Dict{String,Int}(),
98 Dict{String,Int}(),
99 "main",
100 1,
101 Tree(),
102 Tree(),
103 Set{String}(),
104 nothing,
105 StashEntry[],
106 String[],
107 )
108 root=createcommit!(r, Int[], Tree(), "init")
109 r.branches["main"]=root
110 r
111end
112head(r) = r.branch===nothing ? r.detached : r.branches[r.branch]
113headtree(r) = r.commits[head(r)].tree
114parenttree(r, c) = isempty(c.parents) ? Tree() : r.commits[c.parents[1]].tree
115function movehead!(r, c)
116 r.branch===nothing ? (r.detached=c) : (r.branches[r.branch]=c)
117end
118function revision(r, rev)
119 parts=split(rev, '~')
120 anchor=parts[1]
121 c=anchor=="HEAD" ? head(r) :
122 haskey(r.branches, anchor) ? r.branches[anchor] :
123 haskey(r.tags, anchor) ? r.tags[anchor] : get(r.active, anchor, 0)
124 check(c!=0, "unknown revision '$rev'")
125 for _ in 1:(length(parts)==2 ? parse(Int, parts[2]) : 0)
126 check(!isempty(r.commits[c].parents), "unknown revision '$rev'")
127 c=r.commits[c].parents[1]
128 end
129 c
130end
131function distances(r, start)
132 result=Dict(start=>0)
133 queue=[start]
134 at=1
135 while at<=length(queue)
136 c=queue[at]
137 at+=1
138 for p in r.commits[c].parents
139 if !haskey(result, p)
140 result[p]=result[c]+1
141 push!(queue, p)
142 end
143 end
144 end
145 result
146end
147isancestor(r, a, b) = haskey(distances(r, b), a)
148function mergebase(r, a, b)
149 da=distances(r, a)
150 db=distances(r, b)
151 common=intersect(Set(keys(da)), Set(keys(db)))
152 parents=Set{Int}()
153 for c in common
154 union!(parents, r.commits[c].parents)
155 end
156 candidates=collect(setdiff(common, parents))
157 candidates[argmin([(max(da[c], db[c]), -r.commits[c].sequence) for c in candidates])]
158end
159function reachable(r, roots)
160 result=Set{Int}()
161 for c in roots
162 union!(result, keys(distances(r, c)))
163 end
164 result
165end
166function commit_summary(r, c)
167 value=r.commits[c]
168 references=String[]
169 if c==head(r)
170 push!(references, r.branch===nothing ? "HEAD" : "HEAD -> $(r.branch)")
171 end
172 for name in sort!(collect(keys(r.branches)))
173 r.branches[name]==c && name!=r.branch && push!(references, name)
174 end
175 for name in sort!(collect(keys(r.tags)))
176 r.tags[name]==c && push!(references, "tag: $name")
177 end
178 value.id*(isempty(references) ? "" : " ("*join(references, ", ")*")")*" "*value.message
179end
180ignored(r, p) =
181 !haskey(r.index, p) && !(p in r.conflicts) && any(pattern->globmatch(p, pattern), r.ignores)
182function checkprogress(r)
183 r.operation===nothing || fail("$(r.operation[1]) in progress")
184end
185uncommitted(r) = r.index!=headtree(r) || any(p->!agree(r.work, r.index, p), keys(r.index))
186checkclean(r) = check(!uncommitted(r), "uncommitted changes")
187function checkoverwrite(r, selected)
188 h=headtree(r)
189 for p in sort!(collect(selected))
190 check(
191 agree(r.index, h, p) && agree(r.work, h, p),
192 "local changes to '$p' would be overwritten",
193 )
194 end
195end
196function switchtree!(r, tree)
197 selected=differing(headtree(r), tree)
198 checkoverwrite(r, selected)
199 for p in selected
200 matchpath!(r.index, tree, p)
201 matchpath!(r.work, tree, p)
202 end
203end
204function recordcommit!(r, parents, tree, message)
205 c=createcommit!(r, parents, tree, message)
206 movehead!(r, c)
207 old=r.commits[parents[1]].tree
208 added=count(p->!haskey(old, p), keys(tree))
209 modified=count(p->haskey(old, p) && old[p]!=tree[p], keys(tree))
210 deleted=count(p->!haskey(tree, p), keys(old))
211 println(
212 "[$(r.branch===nothing ? "detached HEAD" : r.branch) $(r.commits[c].id)] $message ($added added, $modified modified, $deleted deleted)",
213 )
214end
215function applychange!(r, base, theirs, kind, target, message, parents, common = 0)
216 h=headtree(r)
217 merged, conflicts=merge_trees(base, h, theirs)
218 selected=differing(h, merged)
219 checkoverwrite(r, union(Set(selected), Set(conflicts)))
220 if kind=="merge"
221 println("Merge base $(r.commits[common].id)")
222 elseif isempty(conflicts) && merged==h
223 println("nothing to commit")
224 return
225 end
226 for p in selected
227 matchpath!(r.index, merged, p)
228 matchpath!(r.work, merged, p)
229 end
230 if isempty(conflicts)
231 recordcommit!(r, parents, merged, message)
232 return
233 end
234 for p in conflicts
235 r.work[p]="<<<<<<< $(get(h,p,"(deleted)")) ======= $(get(theirs,p,"(deleted)")) >>>>>>>"
236 println("CONFLICT $p")
237 end
238 r.conflicts=Set(conflicts)
239 r.operation=(kind, target)
240 println(
241 kind=="merge" ? "Automatic merge failed; fix conflicts and then commit the result." :
242 kind=="revert" ? "error: could not revert $(r.commits[target].id)" :
243 "error: could not apply $(r.commits[target].id)",
244 )
245end
246function status(r)
247 h=headtree(r)
248 lines=String[]
249 println(
250 r.branch===nothing ? "HEAD detached at $(r.commits[head(r)].id)" : "On branch $(r.branch)",
251 )
252 if r.operation!==nothing
253 kind, c=r.operation
254 title=kind=="merge" ? "Merging" : kind=="revert" ? "Reverting" : "Cherry-picking"
255 println("$title $(r.commits[c].id)")
256 end
257 for p in sort!(collect(union(Set(paths(h, r.index, r.work)), r.conflicts)))
258 if p in r.conflicts
259 push!(lines, "UU $p")
260 elseif haskey(r.index, p)
261 x=!haskey(h, p) ? "A" : h[p]!=r.index[p] ? "M" : "."
262 y=!haskey(r.work, p) ? "D" : r.work[p]!=r.index[p] ? "M" : "."
263 (x!="." || y!=".") && push!(lines, x*y*" "*p)
264 else
265 haskey(h, p) && push!(lines, "D. $p")
266 haskey(r.work, p) && !ignored(r, p) && push!(lines, "?? $p")
267 end
268 end
269 foreach(println, isempty(lines) ? ["nothing to commit, working tree clean"] : lines)
270end
271function execute!(r, line)
272 words=String.(split(line, ' '; limit = 2))
273 command=words[1]
274 argument=length(words)==1 ? "" : words[2]
275 args=isempty(argument) ? String[] :
276 command in ("COMMIT", "AMEND") ? [argument] :
277 String.(split(argument, ' '; limit = command=="WRITE" ? 2 : 0))
278 h=head(r)
279 hc=r.commits[h]
280 ht=hc.tree
281 if command=="WRITE"
282 r.work[args[1]]=args[2]
283 elseif command=="REMOVE"
284 p=args[1]
285 check(haskey(r.work, p), "no such file '$p'")
286 delete!(r.work, p)
287 elseif command=="IGNORE"
288 push!(r.ignores, args[1])
289 elseif command=="ADD"
290 p=args[1]
291 if p=="ALL"
292 selected=[p for p in paths(r.work, r.index) if !ignored(r, p)]
293 for p in selected
294 matchpath!(r.index, r.work, p)
295 end
296 empty!(r.conflicts)
297 else
298 check(!(haskey(r.work, p) && ignored(r, p)), "path '$p' is ignored")
299 check(
300 haskey(r.work, p) || haskey(r.index, p) || p in r.conflicts,
301 "pathspec '$p' did not match any files",
302 )
303 matchpath!(r.index, r.work, p)
304 delete!(r.conflicts, p)
305 end
306 elseif command=="RM"
307 p=args[1]
308 check(haskey(r.index, p), "pathspec '$p' did not match any files")
309 delete!(r.index, p)
310 delete!(r.work, p)
311 delete!(r.conflicts, p)
312 elseif command=="UNSTAGE"
313 p=args[1]
314 check(haskey(r.index, p) || haskey(ht, p), "pathspec '$p' did not match any files")
315 matchpath!(r.index, ht, p)
316 elseif command in ("RESTORE", "CAT")
317 p=args[1]
318 tree=length(args)==1 ? (command=="RESTORE" ? r.index : r.work) :
319 r.commits[revision(r, args[2])].tree
320 check(
321 haskey(tree, p),
322 length(args)==2 ? "path '$p' does not exist in '$(args[2])'" :
323 command=="CAT" ? "no such file '$p'" : "pathspec '$p' did not match any files",
324 )
325 if command=="CAT"
326 println(tree[p])
327 else
328 r.work[p]=tree[p]
329 if length(args)==2
330 r.index[p]=tree[p]
331 delete!(r.conflicts, p)
332 end
333 end
334 elseif command=="COMMIT"
335 check(isempty(r.conflicts), "unresolved conflicts")
336 if r.operation!==nothing && r.operation[1]=="merge"
337 recordcommit!(r, [h, r.operation[2]], r.index, args[1])
338 r.operation=nothing
339 elseif r.index==ht
340 println("nothing to commit")
341 else
342 recordcommit!(r, [h], r.index, args[1])
343 r.operation=nothing
344 end
345 elseif command=="AMEND"
346 checkprogress(r)
347 check(!isempty(hc.parents), "cannot amend the root commit")
348 recordcommit!(r, hc.parents, r.index, args[1])
349 elseif command=="STATUS"
350 status(r)
351 elseif command=="LS"
352 tree=isempty(args) ? r.work : r.commits[revision(r, args[1])].tree
353 isempty(tree) && println("(empty)")
354 for p in sort!(collect(keys(tree)))
355 println(p, " ", tree[p])
356 end
357 elseif command=="DIFF"
358 if isempty(args)
359 old=r.index
360 new=Tree(p=>r.work[p] for p in keys(r.index) if haskey(r.work, p))
361 elseif args[1]=="STAGED"
362 old=ht
363 new=r.index
364 else
365 a=revision(r, args[1])
366 b=revision(r, args[2])
367 old=r.commits[a].tree
368 new=r.commits[b].tree
369 end
370 foreach(println, changes(old, new))
371 elseif command=="LOG"
372 limit=isempty(args) ? typemax(Int) : parse(Int, args[1])
373 c=h
374 printed=0
375 while true
376 println(commit_summary(r, c))
377 printed+=1
378 (isempty(r.commits[c].parents) || printed==limit) && break
379 c=r.commits[c].parents[1]
380 end
381 elseif command=="SHOW"
382 c=revision(r, isempty(args) ? "HEAD" : args[1])
383 value=r.commits[c]
384 println("commit ", commit_summary(r, c))
385 println(
386 "parents: ",
387 isempty(value.parents) ? "none" : join([r.commits[p].id for p in value.parents], ' '),
388 )
389 foreach(println, changes(parenttree(r, value), value.tree))
390 elseif command=="BLAME"
391 p=args[1]
392 check(haskey(ht, p), "path '$p' does not exist in 'HEAD'")
393 c=h
394 text=ht[p]
395 while !isempty(r.commits[c].parents)
396 parent=r.commits[c].parents[1]
397 get(r.commits[parent].tree, p, nothing)==text || break
398 c=parent
399 end
400 println("$(r.commits[c].id) ($(r.commits[c].message)) $text")
401 elseif command in ("BRANCH", "TAG")
402 refs=command=="BRANCH" ? r.branches : r.tags
403 noun=lowercase(command)
404 remove=args[1]=="-d"
405 name=args[remove ? 2 : 1]
406 if remove
407 check(haskey(refs, name), "no such $noun '$name'")
408 c=refs[name]
409 if command=="BRANCH"
410 check(name!=r.branch, "cannot delete current branch '$name'")
411 check(isancestor(r, c, h), "branch '$name' is not fully merged")
412 end
413 delete!(refs, name)
414 println("Deleted $noun $name (was $(r.commits[c].id))")
415 else
416 check(!haskey(refs, name), "$noun '$name' already exists")
417 c=revision(r, length(args)==2 ? args[2] : "HEAD")
418 refs[name]=c
419 println("Created $noun $name at $(r.commits[c].id)")
420 end
421 elseif command in ("BRANCHES", "TAGS")
422 if command=="BRANCHES"
423 r.branch===nothing && println("* (HEAD detached at $(hc.id))")
424 for name in sort!(collect(keys(r.branches)))
425 value=r.commits[r.branches[name]]
426 println(name==r.branch ? "*" : " ", " $name $(value.id) $(value.message)")
427 end
428 else
429 isempty(r.tags) && println("(no tags)")
430 for name in sort!(collect(keys(r.tags)))
431 value=r.commits[r.tags[name]]
432 println("$name $(value.id) $(value.message)")
433 end
434 end
435 elseif command=="SWITCH"
436 name=args[1]
437 check(haskey(r.branches, name), "no such branch '$name'")
438 if name==r.branch
439 println("Already on '$name'")
440 else
441 checkprogress(r)
442 switchtree!(r, r.commits[r.branches[name]].tree)
443 r.branch=name
444 println("Switched to branch '$name'")
445 end
446 elseif command=="DETACH"
447 c=revision(r, args[1])
448 checkprogress(r)
449 switchtree!(r, r.commits[c].tree)
450 r.branch=nothing
451 r.detached=c
452 println("HEAD is now at $(r.commits[c].id) $(r.commits[c].message)")
453 elseif command=="MERGEBASE"
454 a=revision(r, args[1])
455 b=revision(r, args[2])
456 println(r.commits[mergebase(r, a, b)].id)
457 elseif command=="MERGE"
458 c=revision(r, args[1])
459 checkprogress(r)
460 if isancestor(r, c, h)
461 println("Already up to date")
462 elseif isancestor(r, h, c)
463 switchtree!(r, r.commits[c].tree)
464 movehead!(r, c)
465 println("Fast-forward $(hc.id)..$(r.commits[c].id)")
466 else
467 checkclean(r)
468 base=mergebase(r, h, c)
469 applychange!(
470 r,
471 r.commits[base].tree,
472 r.commits[c].tree,
473 "merge",
474 c,
475 "Merge $(args[1])",
476 [h, c],
477 base,
478 )
479 end
480 elseif command in ("REVERT", "CHERRYPICK")
481 c=revision(r, args[1])
482 checkprogress(r)
483 checkclean(r)
484 value=r.commits[c]
485 parent=parenttree(r, value)
486 if command=="REVERT"
487 applychange!(r, value.tree, parent, "revert", c, "Revert \"$(value.message)\"", [h])
488 else
489 applychange!(r, parent, value.tree, "cherry-pick", c, value.message, [h])
490 end
491 elseif command=="RESET"
492 c=revision(r, args[2])
493 tree=r.commits[c].tree
494 if args[1]=="HARD"
495 for p in union(Set(paths(r.index, tree)), r.conflicts)
496 matchpath!(r.work, tree, p)
497 end
498 end
499 args[1]!="SOFT" && (r.index=copy(tree))
500 movehead!(r, c)
501 empty!(r.conflicts)
502 r.operation=nothing
503 println("HEAD is now at $(r.commits[c].id) $(r.commits[c].message)")
504 elseif command=="REBASE"
505 upstream=revision(r, args[1])
506 checkprogress(r)
507 if isancestor(r, upstream, h)
508 println("Already up to date")
509 return
510 end
511 checkclean(r)
512 upstream_ancestors=distances(r, upstream)
513 replay=Int[]
514 c=h
515 while !haskey(upstream_ancestors, c)
516 push!(replay, c)
517 c=r.commits[c].parents[1]
518 end
519 reverse!(replay)
520 tree=r.commits[upstream].tree
521 plans=Union{Nothing,Tree}[]
522 for c in replay
523 value=r.commits[c]
524 merged, conflicts=merge_trees(parenttree(r, value), tree, value.tree)
525 check(
526 isempty(conflicts),
527 "could not rebase $(value.id): conflict in '$(isempty(conflicts) ? "" : conflicts[1])'",
528 )
529 if merged==tree
530 push!(plans, nothing)
531 else
532 push!(plans, merged)
533 tree=merged
534 end
535 end
536 switchtree!(r, tree)
537 tip=upstream
538 for (c, picked) in zip(replay, plans)
539 value=r.commits[c]
540 if picked===nothing
541 println("skip $(value.id) $(value.message)")
542 else
543 next=createcommit!(r, [tip], picked, value.message)
544 println("pick $(value.id) -> $(r.commits[next].id) $(value.message)")
545 tip=next
546 end
547 end
548 movehead!(r, tip)
549 println("HEAD is now at $(r.commits[tip].id) $(r.commits[tip].message)")
550 elseif command=="STASH"
551 option=isempty(args) ? "" : args[1]
552 if option=="LIST"
553 isempty(r.stash) && println("(no stash entries)")
554 for (i, e) in enumerate(r.stash)
555 println("stash@{$(i-1)}: $(e.label)")
556 end
557 elseif option=="DROP"
558 check(!isempty(r.stash), "no stash entries")
559 e=popfirst!(r.stash)
560 println("Dropped stash@{0}: $(e.label)")
561 elseif option=="POP"
562 check(!isempty(r.stash), "no stash entries")
563 checkprogress(r)
564 checkclean(r)
565 e=r.stash[1]
566 merged, conflicts=merge_trees(r.commits[e.base].tree, ht, e.tree)
567 check(
568 isempty(conflicts),
569 "stash conflict in '$(isempty(conflicts) ? "" : conflicts[1])'",
570 )
571 selected=differing(ht, merged)
572 checkoverwrite(r, selected)
573 for p in selected
574 matchpath!(r.work, merged, p)
575 end
576 popfirst!(r.stash)
577 println("Restored stash@{0}: $(e.label)")
578 foreach(println, changes(ht, merged))
579 else
580 checkprogress(r)
581 if !uncommitted(r)
582 println("No local changes to save")
583 return
584 end
585 tracked=paths(r.index, ht)
586 saved=Tree(p=>r.work[p] for p in tracked if haskey(r.work, p))
587 for p in tracked
588 matchpath!(r.work, ht, p)
589 end
590 r.index=copy(ht)
591 label="WIP on $(r.branch===nothing ? "(no branch)" : r.branch): $(hc.id) $(hc.message)"
592 pushfirst!(r.stash, StashEntry(label, h, saved))
593 println("Saved stash@{0}: $label")
594 end
595 elseif command=="DESCRIBE"
596 c=revision(r, isempty(args) ? "HEAD" : args[1])
597 ds=distances(r, c)
598 best=nothing
599 distance=0
600 for name in sort!(collect(keys(r.tags)))
601 target=r.tags[name]
602 if haskey(ds, target) && (best===nothing || ds[target]<distance)
603 best=name
604 distance=ds[target]
605 end
606 end
607 check(best!==nothing, "no tag can describe '$(r.commits[c].id)'")
608 println(distance==0 ? best : "$best-$distance-g$(r.commits[c].id)")
609 elseif command in ("GRAPH", "GC")
610 roots=[h; collect(values(r.branches)); collect(values(r.tags))]
611 if command=="GC"
612 append!(roots, [e.base for e in r.stash])
613 r.operation!==nothing && push!(roots, r.operation[2])
614 kept=reachable(r, roots)
615 removed=[id for (id, c) in r.active if !(c in kept)]
616 for id in removed
617 delete!(r.active, id)
618 end
619 println("Removed unreachable commits: $(length(removed))")
620 else
621 selected=sort!(collect(reachable(r, roots)); by = c->r.commits[c].sequence, rev = true)
622 lanes=Int[]
623 for c in selected
624 !(c in lanes) && push!(lanes, c)
625 at=findfirst(==(c), lanes)
626 symbols=[i==at ? "*" : lane==c ? "/" : "|" for (i, lane) in enumerate(lanes)]
627 println(join(symbols, ' '), " ", commit_summary(r, c))
628 lanes=[lane for (i, lane) in enumerate(lanes) if i==at || lane!=c]
629 parents=r.commits[c].parents
630 if isempty(parents)
631 deleteat!(lanes, at)
632 else
633 lanes[at]=parents[1]
634 length(parents)==2 && !(parents[2] in lanes) && insert!(lanes, at+1, parents[2])
635 end
636 end
637 end
638 end
639end
640function main()
641 repository=Repository()
642 for line in eachline()
643 line=="EXIT" && break
