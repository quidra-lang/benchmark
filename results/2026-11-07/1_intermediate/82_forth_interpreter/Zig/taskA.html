92 fn initialize(self: *Machine) Error!void {
93 try self.memory.append(self.gpa, 10);
94 var iterator = std.mem.tokenizeScalar(u8, builtinNames, ' ');
95 while (iterator.next()) |name| {
96 const word = try self.gpa.create(Word);
97 word.* = .{ .name = name, .kind = .builtin };
98 try self.builtins.append(self.gpa, word);
99 }
100 }
101 fn read(self: *Machine) ?Text {
102 while (self.position < self.source.len and self.source[self.position] == ' ') self.position += 1;
103 const start = self.position;
104 while (self.position < self.source.len and self.source[self.position] != ' ') self.position += 1;
105 return if (start == self.position) null else self.source[start..self.position];
106 }
107 fn readName(self: *Machine, raw: Text) Error!Text {
108 return self.read() orelse return issue(raw, "missing name", -16);
109 }
110 fn lookup(self: *Machine, name: Text) ?*Word {
111 var i = self.dictionary.items.len;
112 while (i > 0) {
113 i -= 1;
114 if (sameName(self.dictionary.items[i].name, name)) return self.dictionary.items[i];
115 }
116 for (self.builtins.items) |word| if (sameName(word.name, name)) return word;
117 return null;
118 }
119 fn numeric(raw: Text, rawBase: Int) Error!?Int {
120 if (raw.len == 3 and raw[0] == '\'' and raw[2] == '\'') return @as(Int, raw[1]);
121 var text = raw;
122 var base = rawBase;
123 if (raw.len > 0) switch (raw[0]) {
124 '$' => {
125 base = 16;
126 text = raw[1..];
127 },
128 '#' => {
129 base = 10;
130 text = raw[1..];
131 },
132 '%' => {
133 base = 2;
134 text = raw[1..];
135 },
136 else => if (base < 2 or base > 36) return issue(raw, "invalid base", -40),
137 };
138 const negative = std.mem.startsWith(u8, text, "-");
139 if (negative) text = text[1..];
140 if (text.len == 0) return null;
141 for (text) |character| {
142 const c = std.ascii.toUpper(character);
143 const digit: Int = if (std.ascii.isDigit(c)) c - '0' else if (c >= 'A' and c <= 'Z') c - 'A' + 10 else -1;
144 if (digit < 0 or digit >= base) return null;
145 }
146 var value: Int = 0;
147 for (text) |character| {
148 const c = std.ascii.toUpper(character);
149 const digit: Int = if (std.ascii.isDigit(c)) c - '0' else c - 'A' + 10;
150 value = value * base + digit;
151 }
152 return if (negative) -value else value;
153 }
154 fn checkName(self: *Machine, raw: Text) Error!void {
155 _ = self;
156 for (compileOnly) |name| if (sameName(raw, name)) return issue(raw, "invalid name", -19);
157 for (interpretOnly) |name| if (sameName(raw, name)) return issue(raw, "invalid name", -19);
158 for (bothModes) |name| if (sameName(raw, name)) return issue(raw, "invalid name", -19);
159 if (try numeric(raw, 10) != null) return issue(raw, "invalid name", -19);
160 }
161 fn take(self: *Machine, count: usize, raw: Text) Error![4]Int {
162 if (self.data.items.len < count) return issue(raw, "stack underflow", -4);
163 var values: [4]Int = .{ 0, 0, 0, 0 };
164 const start = self.data.items.len - count;
165 @memcpy(values[0..count], self.data.items[start..]);
166 self.data.shrinkRetainingCapacity(start);
167 return values;
168 }
169 fn push(self: *Machine, value: Int, raw: Text) Error!void {
170 try self.data.append(self.gpa, value);
171 if (self.data.items.len > 1000) return issue(raw, "stack overflow", -3);
172 }
173 fn pushAll(self: *Machine, values: []const Int, raw: Text) Error!void {
174 for (values) |value| try self.push(value, raw);
175 }
176 fn returnNeed(self: *Machine, count: usize, raw: Text) Error!void {

179 fn returnPush(self: *Machine, value: Int, raw: Text) Error!void {

183 fn address(self: *Machine, value: Int, raw: Text) Error!usize {
184 if (value < 0 or @rem(value, 8) != 0 or value >= si(self.memory.items.len) * 8) return issue(raw, "invalid address", -9);
185 return u(@divTrunc(value, 8));
186 }
187 fn formatNumber(self: *Machine, value: Int, raw: Text, buffer: *[64]u8) Error!Text {
188 const base = self.memory.items[0];
189 if (base < 2 or base > 36) return issue(raw, "invalid base", -40);
190 const digits = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ";
191 var absolute = if (value < 0) -value else value;
192 var start: usize = buffer.len;
193 while (true) {
194 start -= 1;
195 buffer[start] = digits[u(@mod(absolute, base))];
196 absolute = @divTrunc(absolute, base);
197 if (absolute == 0) break;
198 }
199 if (value < 0) {
200 start -= 1;
201 buffer[start] = '-';
202 }
203 return buffer[start..];
204 }

212 fn executionToken(self: *Machine, word: *Word) Error!Int {
213 if (self.tokenNumbers.get(word)) |token| return token;
214 try self.tokens.append(self.gpa, word);
215 const token = si(self.tokens.items.len);
216 try self.tokenNumbers.put(self.gpa, word, token);
217 return token;
218 }
219 fn validToken(self: *Machine, value: Int, raw: Text) Error!*Word {
220 if (value < 1 or value > si(self.tokens.items.len)) return issue(raw, "invalid argument", -24);
221 return self.tokens.items[u(value - 1)];
222 }
223 fn executeToken(self: *Machine, value: Int, raw: Text) Error!void {
224 const word = try self.validToken(value, raw);
225 try self.execute(word, word.name);
226 }

232 fn topControl(self: *Machine, raw: Text, kinds: []const Text) Error!*Control {
233 if (self.controls.items.len == 0 or !member(self.controls.items[self.controls.items.len - 1].kind, kinds)) return issue(raw, "control structure mismatch", null);
234 return &self.controls.items[self.controls.items.len - 1];
235 }
236 fn compileControl(self: *Machine, raw: Text, name: Text) Error!bool {
237 if (equal(name, "IF")) {
238 try self.controls.append(self.gpa, .{ .kind = "IF", .index = try self.addAction(.{ .kind = "zero", .raw = raw }) });
239 } else if (equal(name, "ELSE")) {
240 const top = try self.topControl(raw, &.{"IF"});
241 const index = try self.addAction(.{ .kind = "jump", .raw = raw });
242 self.compiling.?.actions.items[top.index].target = self.compiling.?.actions.items.len;
243 top.kind = "ELSE";
244 top.index = index;
245 } else if (equal(name, "THEN")) {
246 const top = try self.topControl(raw, &.{ "IF", "ELSE" });
247 self.compiling.?.actions.items[top.index].target = self.compiling.?.actions.items.len;
248 _ = self.controls.pop();
249 } else if (equal(name, "BEGIN")) {
250 try self.controls.append(self.gpa, .{ .kind = "BEGIN", .start = self.compiling.?.actions.items.len });
251 } else if (member(name, &.{ "UNTIL", "AGAIN" })) {
252 const top = try self.topControl(raw, &.{"BEGIN"});
253 _ = try self.addAction(.{ .kind = if (equal(name, "UNTIL")) "zero" else "jump", .raw = raw, .target = top.start });
254 _ = self.controls.pop();
255 } else if (equal(name, "WHILE")) {
256 const top = try self.topControl(raw, &.{"BEGIN"});
257 top.kind = "WHILE";
258 top.index = try self.addAction(.{ .kind = "zero", .raw = raw });
259 } else if (equal(name, "REPEAT")) {
260 const top = try self.topControl(raw, &.{"WHILE"});
261 _ = try self.addAction(.{ .kind = "jump", .raw = raw, .target = top.start });
262 self.compiling.?.actions.items[top.index].target = self.compiling.?.actions.items.len;
263 _ = self.controls.pop();
264 } else if (member(name, &.{ "DO", "?DO" })) {
265 const index = try self.addAction(.{ .kind = name, .raw = raw });
266 try self.controls.append(self.gpa, .{ .kind = name, .index = index, .start = index + 1 });
267 } else if (member(name, &.{ "LOOP", "+LOOP" })) {
268 const top = try self.topControl(raw, &.{ "DO", "?DO" });
269 _ = try self.addAction(.{ .kind = name, .raw = raw, .target = top.start });
270 const finish = self.compiling.?.actions.items.len;
271 self.compiling.?.actions.items[top.index].target = finish;
272 for (top.leaves.items) |index| self.compiling.?.actions.items[index].target = finish;
273 _ = self.controls.pop();
274 } else if (equal(name, "LEAVE")) {
275 var found: ?usize = null;
276 var i = self.controls.items.len;
277 while (i > 0) {
278 i -= 1;
279 if (member(self.controls.items[i].kind, &.{ "DO", "?DO" })) {
280 found = i;
281 break;
282 }
283 }
284 const index = found orelse return issue(raw, "control structure mismatch", null);
285 const actionIndex = try self.addAction(.{ .kind = "LEAVE", .raw = raw });
286 try self.controls.items[index].leaves.append(self.gpa, actionIndex);
287 } else if (equal(name, "CASE")) {
288 try self.controls.append(self.gpa, .{ .kind = "CASE" });
289 } else if (equal(name, "OF")) {
290 _ = try self.topControl(raw, &.{"CASE"});
291 try self.controls.append(self.gpa, .{ .kind = "OF", .index = try self.addAction(.{ .kind = "OF", .raw = raw }) });
292 } else if (equal(name, "ENDOF")) {
293 const top = try self.topControl(raw, &.{"OF"});
294 const index = try self.addAction(.{ .kind = "jump", .raw = raw });
295 self.compiling.?.actions.items[top.index].target = self.compiling.?.actions.items.len;
296 _ = self.controls.pop();
297 const parent = try self.topControl(raw, &.{"CASE"});
298 try parent.ends.append(self.gpa, index);
299 } else if (equal(name, "ENDCASE")) {
300 const top = try self.topControl(raw, &.{"CASE"});
301 _ = try self.addAction(.{ .kind = "ENDCASE", .raw = raw });
302 for (top.ends.items) |index| self.compiling.?.actions.items[index].target = self.compiling.?.actions.items.len;
303 _ = self.controls.pop();
304 } else return false;
305 return true;
306 }
307 fn readString(self: *Machine, raw: Text) Error!Text {
308 if (self.position == self.source.len) return issue(raw, "unterminated string", null);
309 const start = self.position + 1;
310 const end = std.mem.findScalar(u8, self.source[start..], '"') orelse return issue(raw, "unterminated string", null);
311 self.position = start + end + 1;
312 return self.source[start .. start + end];
313 }
314 fn declareLocals(self: *Machine, raw: Text) Error!void {
315 var names: std.ArrayList(Text) = .empty;
316 var ignored = false;
317 var closed = false;
318 while (self.read()) |name| {
319 if (sameName(name, ":}")) {
320 closed = true;
321 break;
322 }
323 if (equal(name, "--")) ignored = true;
324 if (!ignored) try names.append(self.gpa, name);
325 }
326 if (!closed) return issue(raw, "unterminated locals", null);
327 var slots: std.ArrayList(usize) = .empty;
328 for (names.items) |name| {
329 try self.checkName(name);
330 const slot = self.compiling.?.localCount;
331 self.compiling.?.localCount += 1;
332 try self.compiling.?.localNames.put(self.gpa, try uppercase(self.gpa, name), slot);
333 try slots.append(self.gpa, slot);
334 }
335 _ = try self.addAction(.{ .kind = "locals", .raw = raw, .slots = slots.items });
336 }
337 fn localSlot(self: *Machine, name: Text) ?usize {
338 var iterator = self.compiling.?.localNames.iterator();
339 while (iterator.next()) |entry| if (sameName(name, entry.key_ptr.*)) return entry.value_ptr.*;
340 return null;
341 }
342 fn assign(self: *Machine, raw: Text, name: Text) Error!void {
343 const targetName = try self.readName(raw);
344 if (equal(name, "TO") and self.compiling != null) {
345 if (self.localSlot(targetName)) |slot| {
346 _ = try self.addAction(.{ .kind = "localSet", .raw = raw, .target = slot });
347 return;
348 }
349 }
350 const target = self.lookup(targetName) orelse return issue(targetName, "undefined word", -13);
351 const required: WordKind = if (equal(name, "TO")) .value else .deferred;
352 if (target.kind != required) return issue(targetName, if (equal(name, "TO")) "not a value" else "not deferred", null);
353 const action: Action = .{ .kind = name, .raw = raw, .word = target };
354 if (self.compiling != null) {
355 _ = try self.addAction(action);
356 } else try self.runtimeAction(action, &.{});
357 }
358 fn process(self: *Machine, raw: Text) Error!void {
359 const name = try uppercase(self.gpa, raw);
360 if (equal(name, "\\")) {
361 self.position = self.source.len;
362 return;
363 }
364 if (equal(name, "(")) {
365 const end = std.mem.findScalar(u8, self.source[self.position..], ')') orelse return issue(raw, "unterminated comment", null);
366 self.position += end + 1;
367 return;
368 }
369 if (self.compiling == null and member(name, &compileOnly)) return issue(raw, "compile only", null);
370 if (self.compiling != null and member(name, &interpretOnly)) return issue(raw, "interpret only", null);
371 if (member(name, &.{ ".\"", "ABORT\"" })) {
372 const text = try self.readString(raw);
373 if (self.compiling == null) {
374 try self.write(text);
375 } else {
376 _ = try self.addAction(.{ .kind = name, .raw = raw, .text = text });
377 }
378 return;
379 }
380 if (member(name, &.{ "TO", "IS" })) return self.assign(raw, name);
381 if (member(name, &.{ ":", ":NONAME" })) {
382 const rawName = if (equal(name, ":")) try self.readName(raw) else raw;
383 if (equal(name, ":")) try self.checkName(rawName);
384 const word = try self.gpa.create(Word);
385 word.* = .{ .name = if (equal(name, ":NONAME")) ":NONAME" else try uppercase(self.gpa, rawName), .rawName = rawName, .kind = .colon, .here = self.memory.items.len };
386 self.compiling = word;
387 self.controls.clearRetainingCapacity();
388 return;
389 }
390 if (equal(name, "FORGET")) {
391 const targetName = try self.readName(raw);
392 var found: ?usize = null;
393 var i = self.dictionary.items.len;
394 while (i > 0) {
395 i -= 1;
396 if (sameName(self.dictionary.items[i].name, targetName)) {
397 found = i;
398 break;
399 }
400 }
401 const index = found orelse return issue(targetName, "undefined word", -13);
402 self.memory.shrinkRetainingCapacity(self.dictionary.items[index].here);
403 self.dictionary.shrinkRetainingCapacity(index);
404 return;
405 }
406 if (self.compiling) |word| {
407 if (try self.compileControl(raw, name)) return;
408 if (equal(name, ";")) {
409 if (self.controls.items.len > 0) {
410 const top = self.controls.items[self.controls.items.len - 1].kind;
411 const kind = if (equal(top, "ELSE")) "IF" else if (equal(top, "WHILE")) "BEGIN" else top;
412 return issue(raw, std.fmt.bufPrint(&reasonBuffer, "unclosed {s}", .{kind}) catch unreachable, null);
413 }
414 self.compiling = null;
415 if (equal(word.name, ":NONAME")) {
416 try self.push(try self.executionToken(word), raw);
417 } else try self.dictionary.append(self.gpa, word);
418 return;
419 }
420 if (equal(name, "DOES>")) {
421 if (self.controls.items.len != 0) return issue(raw, "control structure mismatch", null);
422 _ = try self.addAction(.{ .kind = "DOES>", .raw = raw, .word = word });
423 return;
424 }
425 if (equal(name, "EXIT")) {
426 _ = try self.addAction(.{ .kind = "EXIT", .raw = raw });
427 return;
428 }
429 if (equal(name, "RECURSE")) {
430 _ = try self.addAction(.{ .kind = "call", .raw = raw, .word = word });
431 return;
432 }
433 if (equal(name, "[']")) {
434 const targetName = try self.readName(raw);
435 const target = self.lookup(targetName) orelse return issue(targetName, "undefined word", -13);
436 _ = try self.addAction(.{ .kind = "push", .raw = raw, .value = try self.executionToken(target) });
437 return;
438 }
439 if (equal(name, "{:")) return self.declareLocals(raw);
440 if (member(name, &.{ ">R", "R>", "R@", "I", "J", "UNLOOP" })) {
441 _ = try self.addAction(.{ .kind = name, .raw = raw });
442 return;
443 }
444 if (self.localSlot(name)) |slot| {
445 _ = try self.addAction(.{ .kind = "localGet", .raw = raw, .target = slot });
446 return;
447 }
448 if (self.lookup(raw)) |target| {
449 _ = try self.addAction(.{ .kind = "call", .raw = raw, .word = target });
450 return;
451 }
452 if (try numeric(raw, self.memory.items[0])) |value| {
453 _ = try self.addAction(.{ .kind = "push", .raw = raw, .value = value });
454 return;
455 }
456 return issue(raw, "undefined word", -13);
457 }
458 if (self.lookup(raw)) |target| return self.execute(target, raw);
459 if (try numeric(raw, self.memory.items[0])) |value| return self.push(value, raw);
460 return issue(raw, "undefined word", -13);
461 }
462 fn runtimeAction(self: *Machine, action: Action, locals: []Int) Error!void {
463 const raw = action.raw;
464 if (equal(action.kind, "push")) {
465 try self.push(action.value, raw);
466 } else if (equal(action.kind, "call")) {
467 try self.execute(action.word.?, raw);
468 } else if (equal(action.kind, ".\"")) {
469 try self.write(action.text);
470 } else if (equal(action.kind, "ABORT\"")) {
471 if ((try self.take(1, raw))[0] != 0) return issue(raw, action.text, -2);
472 } else if (equal(action.kind, "localGet")) {
473 try self.push(locals[action.target], raw);
474 } else if (equal(action.kind, "localSet")) {
475 locals[action.target] = (try self.take(1, raw))[0];
476 } else if (equal(action.kind, "locals")) {
477 if (self.data.items.len < action.slots.len) return issue(raw, "stack underflow", -4);
478 const start = self.data.items.len - action.slots.len;
479 for (action.slots, self.data.items[start..]) |slot, value| locals[slot] = value;
480 self.data.shrinkRetainingCapacity(start);
481 } else if (equal(action.kind, "TO")) {
482 self.memory.items[action.word.?.address] = (try self.take(1, raw))[0];
483 } else if (equal(action.kind, "IS")) {
484 const value = (try self.take(1, raw))[0];
485 _ = try self.validToken(value, raw);
486 action.word.?.token = value;
487 } else if (equal(action.kind, ">R")) {
488 try self.returnPush((try self.take(1, raw))[0], raw);
489 } else if (equal(action.kind, "R>")) {
490 try self.returnNeed(1, raw);
491 const value = self.returns.pop().?;
492 try self.push(value, raw);
493 } else if (member(action.kind, &.{ "R@", "I", "J" })) {
494 const count: usize = if (equal(action.kind, "J")) 3 else 1;
495 try self.returnNeed(count, raw);
496 try self.push(self.returns.items[self.returns.items.len - count], raw);
497 } else if (equal(action.kind, "UNLOOP")) {
498 try self.returnNeed(2, raw);
499 self.returns.shrinkRetainingCapacity(self.returns.items.len - 2);
500 }
501 }
502 fn definition(self: *Machine, word: *Word, start: usize, raw: Text) Error!void {
503 if (self.depth == 100) return issue(raw, "nesting too deep", -7);
504 self.depth += 1;
505 const saved = self.returns.items.len;
506 const locals = try std.heap.page_allocator.alloc(Int, word.localCount);
507 defer std.heap.page_allocator.free(locals);
508 @memset(locals, 0);
509 var pc = start;
510 while (pc < word.actions.items.len) {
511 const action = word.actions.items[pc];
512 pc += 1;
513 if (equal(action.kind, "EXIT")) break;
514 if (equal(action.kind, "DOES>")) {
515 if (self.dictionary.items.len == 0 or self.dictionary.items[self.dictionary.items.len - 1].kind != .create) return issue(action.raw, "no CREATE word", -31);
516 const newest = self.dictionary.items[self.dictionary.items.len - 1];
517 newest.does = action.word;
518 newest.doesStart = pc;
519 break;
520 }
521 if (equal(action.kind, "jump")) {
522 pc = action.target;
523 } else if (equal(action.kind, "zero")) {
524 if ((try self.take(1, action.raw))[0] == 0) pc = action.target;
525 } else if (member(action.kind, &.{ "DO", "?DO" })) {
526 const values = try self.take(2, action.raw);
527 if (equal(action.kind, "?DO") and values[0] == values[1]) {
528 pc = action.target;
529 } else {
530 try self.returnPush(values[0], action.raw);
531 try self.returnPush(values[1], action.raw);
532 }
533 } else if (member(action.kind, &.{ "LOOP", "+LOOP" })) {
534 const step: Int = if (equal(action.kind, "+LOOP")) (try self.take(1, action.raw))[0] else 1;
535 try self.returnNeed(2, action.raw);
536 const index = self.returns.items[self.returns.items.len - 1];
537 const limit = self.returns.items[self.returns.items.len - 2];
538 const next = index + step;
539 if ((index - limit < 0) != (next - limit < 0)) {
540 self.returns.shrinkRetainingCapacity(self.returns.items.len - 2);
541 } else {
542 self.returns.items[self.returns.items.len - 1] = next;
543 pc = action.target;
544 }
545 } else if (equal(action.kind, "LEAVE")) {
546 try self.returnNeed(2, action.raw);
547 self.returns.shrinkRetainingCapacity(self.returns.items.len - 2);
548 pc = action.target;
549 } else if (equal(action.kind, "OF")) {
550 const values = try self.take(2, action.raw);
551 if (values[0] != values[1]) {
552 try self.push(values[0], action.raw);
553 pc = action.target;
554 }
555 } else if (equal(action.kind, "ENDCASE")) {
556 _ = try self.take(1, action.raw);
557 } else try self.runtimeAction(action, locals);
558 }
559 if (self.returns.items.len > saved) self.returns.shrinkRetainingCapacity(saved);
560 self.depth -= 1;
561 }
562 fn execute(self: *Machine, initialWord: *Word, initialRaw: Text) Error!void {
563 var word = initialWord;
564 var raw = initialRaw;
565 while (word.kind == .deferred) {
566 if (word.token == 0) return issue(raw, "deferred word not set", -21);
567 word = try self.validToken(word.token, raw);
568 raw = word.name;
569 }
570 switch (word.kind) {
571 .colon => try self.definition(word, 0, raw),
572 .create => {
573 try self.push(si(word.address) * 8, raw);
574 if (word.does) |part| try self.definition(part, word.doesStart, raw);
575 },
576 .variable => try self.push(si(word.address) * 8, raw),
577 .constant => try self.push(word.value, raw),
578 .value => try self.push(self.memory.items[word.address], raw),
579 .deferred => unreachable,
580 .builtin => try self.builtin(word.name, raw),
581 }
582 }
583 fn builtin(self: *Machine, name: Text, raw: Text) Error!void {
584 if (member(name, &.{ "CREATE", "VARIABLE", "CONSTANT", "VALUE", "DEFER" })) {
585 const rawName = try self.readName(raw);
586 try self.checkName(rawName);
587 const value: Int = if (member(name, &.{ "CONSTANT", "VALUE" })) (try self.take(1, raw))[0] else 0;
588 const word = try self.gpa.create(Word);
589 word.* = .{ .name = try uppercase(self.gpa, rawName), .rawName = rawName, .kind = if (equal(name, "CREATE")) .create else if (equal(name, "VARIABLE")) .variable else if (equal(name, "CONSTANT")) .constant else if (equal(name, "VALUE")) .value else .deferred, .here = self.memory.items.len, .address = self.memory.items.len, .value = value };
590 try self.dictionary.append(self.gpa, word);
591 if (member(name, &.{ "VARIABLE", "VALUE" })) try self.memory.append(self.gpa, value);
592 return;
593 }
594 if (equal(name, "'")) {
595 const targetName = try self.readName(raw);
596 const target = self.lookup(targetName) orelse return issue(targetName, "undefined word", -13);
597 try self.push(try self.executionToken(target), raw);
598 return;
599 }
600 if (equal(name, "EXECUTE")) return self.executeToken((try self.take(1, raw))[0], raw);
601 if (equal(name, "CATCH")) {
602 const token = (try self.take(1, raw))[0];
603 const target = try self.validToken(token, raw);
604 const saved = try std.heap.page_allocator.dupe(Int, self.data.items);
605 defer std.heap.page_allocator.free(saved);
606 const returnSize = self.returns.items.len;
607 const depth = self.depth;
608 var code: Int = 0;
609 self.execute(target, target.name) catch |err| {
610 if (err != error.Invalid or failure.code == null) return err;
611 self.data.clearRetainingCapacity();
612 try self.data.appendSlice(self.gpa, saved);
613 if (self.returns.items.len > returnSize) self.returns.shrinkRetainingCapacity(returnSize);
614 self.depth = depth;
615 code = failure.code.?;
616 };
617 try self.push(code, raw);
618 return;
619 }
620 if (equal(name, "THROW")) {
621 const code = (try self.take(1, raw))[0];
622 if (code != 0) return issue(raw, std.fmt.bufPrint(&reasonBuffer, "exception {d}", .{code}) catch unreachable, code);
623 return;
624 }
625 if (equal(name, "ABORT")) return issue(raw, "aborted", -1);
626 if (equal(name, "WORDS")) {
627 var i = self.dictionary.items.len;
628 while (i > 0) {
629 i -= 1;
630 try self.write(self.dictionary.items[i].name);
631 try self.write(" ");
632 }
633 return;
634 }
635 if (equal(name, "CR")) return self.flush();
636 if (equal(name, "SPACE")) return self.write(" ");
637 if (equal(name, ".S")) {
638 var buffer: [64]u8 = undefined;
639 const count = try self.formatNumber(si(self.data.items.len), raw, &buffer);
640 try self.write("<");
641 try self.write(count);
642 try self.write("> ");
643 for (self.data.items) |value| {
644 try self.write(try self.formatNumber(value, raw, &buffer));
645 try self.write(" ");
646 }
647 return;
648 }
649 if (equal(name, "TRUE")) return self.push(-1, raw);
650 if (equal(name, "FALSE")) return self.push(0, raw);
651 if (equal(name, "BL")) return self.push(32, raw);
652 if (equal(name, "BASE")) return self.push(0, raw);
653 if (equal(name, "HERE")) return self.push(si(self.memory.items.len) * 8, raw);
654 if (equal(name, "DEPTH")) return self.push(si(self.data.items.len), raw);
655 if (equal(name, "HEX")) {
656 self.memory.items[0] = 16;
657 return;
658 }
659 if (equal(name, "DECIMAL")) {
660 self.memory.items[0] = 10;
661 return;
662 }
663 var count: usize = 1;
664 if (member(name, &.{ "+", "-", "*", "/", "MOD", "/MOD", "MIN", "MAX", "=", "<>", "<", ">", "AND", "OR", "XOR", "SWAP", "OVER", "NIP", "TUCK", "2DUP", "2DROP", ".R", "!", "+!" })) count = 2;
665 if (member(name, &.{ "*/", "*/MOD", "WITHIN", "ROT", "-ROT" })) count = 3;
666 if (member(name, &.{ "2SWAP", "2OVER" })) count = 4;
667 const values = try self.take(count, raw);
668 var a = values[0];
669 const b = values[1];
670 if (equal(name, "+")) {
671 try self.push(a + b, raw);
672 } else if (equal(name, "-")) {
673 try self.push(a - b, raw);
674 } else if (equal(name, "*")) {
675 try self.push(a * b, raw);
676 } else if (member(name, &.{ "/", "MOD", "/MOD", "*/", "*/MOD" })) {
677 const dividend = if (count == 3) a * b else a;
678 const divisor = if (count == 3) values[2] else b;
679 if (divisor == 0) return issue(raw, "division by zero", -10);
680 const q = @divFloor(dividend, divisor);
681 const r = dividend - divisor * q;
682 if (equal(name, "MOD")) {
683 try self.push(r, raw);
684 } else if (member(name, &.{ "/MOD", "*/MOD" })) {
685 try self.pushAll(&.{ r, q }, raw);
686 } else try self.push(q, raw);
687 } else if (equal(name, "NEGATE")) {
688 try self.push(-a, raw);
689 } else if (equal(name, "ABS")) {
690 try self.push(if (a < 0) -a else a, raw);
691 } else if (equal(name, "MIN")) {
692 try self.push(@min(a, b), raw);
693 } else if (equal(name, "MAX")) {
694 try self.push(@max(a, b), raw);
695 } else if (equal(name, "1+")) {
696 try self.push(a + 1, raw);
697 } else if (equal(name, "1-")) {
698 try self.push(a - 1, raw);
699 } else if (equal(name, "2*")) {
700 try self.push(2 * a, raw);
701 } else if (equal(name, "2/")) {
702 try self.push(@divFloor(a, 2), raw);
703 } else if (equal(name, "=")) {
704 try self.push(flag(a == b), raw);
705 } else if (equal(name, "<>")) {
706 try self.push(flag(a != b), raw);
707 } else if (equal(name, "<")) {
708 try self.push(flag(a < b), raw);
709 } else if (equal(name, ">")) {
710 try self.push(flag(a > b), raw);
711 } else if (equal(name, "0=")) {
712 try self.push(flag(a == 0), raw);
713 } else if (equal(name, "0<>")) {
714 try self.push(flag(a != 0), raw);
715 } else if (equal(name, "0<")) {
716 try self.push(flag(a < 0), raw);
717 } else if (equal(name, "0>")) {
718 try self.push(flag(a > 0), raw);
719 } else if (equal(name, "WITHIN")) {
720 try self.push(flag(b <= a and a < values[2]), raw);
721 } else if (equal(name, "AND")) {
722 try self.push(a & b, raw);
723 } else if (equal(name, "OR")) {
724 try self.push(a | b, raw);
725 } else if (equal(name, "XOR")) {
726 try self.push(a ^ b, raw);
727 } else if (equal(name, "INVERT")) {
728 try self.push(~a, raw);
729 } else if (equal(name, "DUP")) {
730 try self.pushAll(&.{ a, a }, raw);
731 } else if (equal(name, "?DUP")) {
732 try self.push(a, raw);
733 if (a != 0) try self.push(a, raw);
734 } else if (member(name, &.{ "DROP", "2DROP" })) {} else if (equal(name, "SWAP")) {
735 try self.pushAll(&.{ b, a }, raw);
736 } else if (equal(name, "OVER")) {
737 try self.pushAll(&.{ a, b, a }, raw);
738 } else if (equal(name, "ROT")) {
739 try self.pushAll(&.{ b, values[2], a }, raw);
740 } else if (equal(name, "-ROT")) {
741 try self.pushAll(&.{ values[2], a, b }, raw);
742 } else if (equal(name, "NIP")) {
743 try self.push(b, raw);
744 } else if (equal(name, "TUCK")) {
745 try self.pushAll(&.{ b, a, b }, raw);
746 } else if (equal(name, "2DUP")) {
747 try self.pushAll(&.{ a, b, a, b }, raw);
748 } else if (equal(name, "2SWAP")) {
749 try self.pushAll(&.{ values[2], values[3], a, b }, raw);
750 } else if (equal(name, "2OVER")) {
751 try self.pushAll(&.{ a, b, values[2], values[3], a, b }, raw);
752 } else if (member(name, &.{ "PICK", "ROLL" })) {
753 if (a < 0) return issue(raw, "invalid argument", -24);
754 if (a >= si(self.data.items.len)) return issue(raw, "stack underflow", -4);
755 const index = self.data.items.len - 1 - u(a);
756 const value = self.data.items[index];
757 if (equal(name, "ROLL")) _ = self.data.orderedRemove(index);
758 try self.push(value, raw);
759 } else if (equal(name, ".")) {
760 var buffer: [64]u8 = undefined;
761 try self.write(try self.formatNumber(a, raw, &buffer));
762 try self.write(" ");
763 } else if (equal(name, ".R")) {
764 var buffer: [64]u8 = undefined;
765 const text = try self.formatNumber(a, raw, &buffer);
766 var padding = b - si(text.len);
767 while (padding > 0) : (padding -= 1) try self.write(" ");
768 try self.write(text);
769 } else if (equal(name, "?")) {
770 const cell = try self.address(a, raw);
771 var buffer: [64]u8 = undefined;
772 try self.write(try self.formatNumber(self.memory.items[cell], raw, &buffer));
773 try self.write(" ");
774 } else if (equal(name, "EMIT")) {
775 if (a < 32 or a > 126) return issue(raw, "invalid argument", -24);
776 try self.pending.append(self.gpa, @intCast(a));
777 } else if (equal(name, "SPACES")) {
778 while (a > 0) : (a -= 1) try self.write(" ");
779 } else if (equal(name, "!")) {
780 const cell = try self.address(b, raw);
781 self.memory.items[cell] = a;
782 } else if (equal(name, "@")) {
783 try self.push(self.memory.items[try self.address(a, raw)], raw);
784 } else if (equal(name, "+!")) {
785 const cell = try self.address(b, raw);
786 self.memory.items[cell] += a;
787 } else if (equal(name, ",")) {
788 try self.memory.append(self.gpa, a);
789 } else if (equal(name, "ALLOT")) {
790 if (a < 0 or @rem(a, 8) != 0) return issue(raw, "invalid argument", -24);
791 const start = self.memory.items.len;
792 try self.memory.resize(self.gpa, start + u(@divTrunc(a, 8)));
793 @memset(self.memory.items[start..], 0);
794 } else if (equal(name, "CELLS")) {
795 try self.push(8 * a, raw);
796 } else if (equal(name, "CELL+")) {
797 try self.push(a + 8, raw);
798 }
799 }
800 fn processLine(self: *Machine, line: Text) Error!void {
801 self.source = line;
802 self.position = 0;
803 while (self.read()) |raw| {
804 self.process(raw) catch |err| {
805 if (err != error.Invalid) return err;
806 if (self.pending.items.len > 0) try self.flush();
807 try self.out.line("ERROR {s}: {s}", .{ failure.word, failure.reason });
808 self.data.clearRetainingCapacity();
809 self.returns.clearRetainingCapacity();
810 self.compiling = null;
811 self.controls.clearRetainingCapacity();
812 self.depth = 0;
813 return;
814 };
815 }
816 if (self.pending.items.len > 0) try self.flush();
817 }
818};

820pub fn main(init: std.process.Init) !void {
821 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
822 defer arena.deinit();
823 const gpa = arena.allocator();
824 var inputBuffer: [8192]u8 = undefined;
825 var input = std.Io.File.stdin().readerStreaming(init.io, &inputBuffer);
826 const text = try input.interface.allocRemaining(gpa, .unlimited);
827 var lines = std.mem.splitScalar(u8, text, '\n');
828 var out: Builder = .{ .gpa = gpa };
829 var machine: Machine = .{ .gpa = gpa, .out = &out };
830 try machine.initialize();
831 while (lines.next()) |raw| {
832 const line = std.mem.trimEnd(u8, raw, "\r");
833 if (equal(line, "BYE")) {
834 if (machine.compiling) |word| try out.line("ERROR {s}: unterminated definition", .{word.rawName});
835 break;
836 }
837 try machine.processLine(line);
838 }
839 var outputBuffer: [8192]u8 = undefined;
840 var output = std.Io.File.stdout().writerStreaming(init.io, &outputBuffer);
841 try output.interface.writeAll(out.bytes.items);
842 try output.interface.flush();
843}
