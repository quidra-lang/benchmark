137 static long? ParseNumber(string raw, long radix)
138 {
139 if (raw.Length == 3 && raw[0] == '\'' && raw[2] == '\'') return raw[1];
140 string text = Upper(raw);
141 if (text.Length > 0 && "$#%".Contains(text[0]))
142 {
143 radix = text[0] == '$' ? 16 : text[0] == '#' ? 10 : 2;
144 text = text[1..];
145 }
146 else if (radix < 2 || radix > 36) return null;
147 bool negative = text.StartsWith('-');
148 if (negative) text = text[1..];
149 if (text.Length == 0) return null;
150 foreach (char c in text)
151 {
152 int digit = c is >= '0' and <= '9' ? c - '0' : c is >= 'A' and <= 'Z' ? c - 'A' + 10 : 36;
153 if (digit >= radix) return null;
154 }
155 long value = 0;
156 foreach (char c in text) value = value * radix + (c <= '9' ? c - '0' : c - 'A' + 10);
157 return negative ? -value : value;
158 }
159 long? Number(string raw)
160 {
161 bool direct = raw.Length == 3 && raw[0] == '\'' && raw[2] == '\'' || raw.Length > 0 && "$#%".Contains(raw[0]);
162 long radix = memory[0];
163 if (!direct && (radix < 2 || radix > 36)) Error(raw, "invalid base", -40);
164 return ParseNumber(raw, radix);
165 }

220 string OutputNumber(long value, string word)
221 {
222 long radix = memory[0];
223 if (radix < 2 || radix > 36) Error(word, "invalid base", -40);
224 long magnitude = Math.Abs(value);
225 StringBuilder text = new();
226 do
227 {
228 int digit = (int)(magnitude % radix);
229 text.Append((char)(digit < 10 ? '0' + digit : 'A' + digit - 10));
230 magnitude /= radix;
231 }
232 while (magnitude != 0);
233 if (value < 0) text.Append('-');
234 return new string(text.ToString().Reverse().ToArray());
235 }

252 void Compile(string raw)
253 {
254 string word = Upper(raw);
255 int at = compiling!.Instructions.Count;
256 switch (word)
257 {
258 case "IF":
259 Emit("zerojump", raw);
260 controls.Add(new("IF", at, 0));
261 break;
262 case "ELSE":
263 {
264 Control c = Top(raw, "IF");
265 Emit("jump", raw);
266 compiling.Instructions[c.Index].Target = at + 1;
267 c.Kind = "ELSE";
268 c.Index = at;
269 break;
270 }
271 case "THEN":
272 {
273 Control c = Top(raw, "IF", "ELSE");
274 compiling.Instructions[c.Index].Target = at;
275 RemoveLast(controls);
276 break;
277 }
278 case "BEGIN":
279 controls.Add(new("BEGIN", at, at));
280 break;
281 case "UNTIL":
282 case "AGAIN":
283 {
284 Control c = Top(raw, "BEGIN");
285 Emit(word == "UNTIL" ? "zerojump" : "jump", raw).Target = c.Begin;
286 RemoveLast(controls);
287 break;
288 }
289 case "WHILE":
290 {
291 Control c = Top(raw, "BEGIN");
292 Emit("zerojump", raw);
293 c.Kind = "WHILE";
294 c.Index = at;
295 break;
296 }
297 case "REPEAT":
298 {
299 Control c = Top(raw, "WHILE");
300 Emit("jump", raw).Target = c.Begin;
301 compiling.Instructions[c.Index].Target = at + 1;
302 RemoveLast(controls);
303 break;
304 }
305 case "DO":
306 case "?DO":
307 Emit(word, raw);
308 controls.Add(new(word, at, at + 1));
309 break;
310 case "LOOP":
311 case "+LOOP":
312 {
313 Control c = Top(raw, "DO", "?DO");
314 Emit(word, raw).Target = c.Begin;
315 compiling.Instructions[c.Index].Target = at + 1;
316 foreach (int leave in c.Exits) compiling.Instructions[leave].Target = at + 1;
317 RemoveLast(controls);
318 break;
319 }
320 case "LEAVE":
321 {
322 Control? loop = controls.LastOrDefault(c => c.Kind is "DO" or "?DO");
323 if (loop == null) Syntax(raw, "control structure mismatch");
324 Emit("LEAVE", raw);
325 loop!.Exits.Add(at);
326 break;
327 }
328 case "CASE":
329 controls.Add(new("CASE", at, 0));
330 break;
331 case "OF":
332 Top(raw, "CASE");
333 Emit("OF", raw);
334 controls.Add(new("OF", at, 0));
335 break;
336 case "ENDOF":
337 {
338 Control of = Top(raw, "OF");
339 RemoveLast(controls);
340 Control c = Top(raw, "CASE");
341 Emit("jump", raw);
342 compiling.Instructions[of.Index].Target = at + 1;
343 c.Exits.Add(at);
344 break;
345 }
346 case "ENDCASE":
347 {
348 Control c = Top(raw, "CASE");
349 Emit("DROP", raw);
350 foreach (int jump in c.Exits) compiling.Instructions[jump].Target = at + 1;
351 RemoveLast(controls);
352 break;
353 }
354 case "EXIT":
355 Emit("EXIT", raw);
356 break;
357 case "RECURSE":
358 Emit("call", raw).Word = compiling;
359 break;
360 case "DOES>":
361 if (controls.Count > 0) Syntax(raw, "control structure mismatch");
362 Emit("DOES>", raw);
363 break;
364 case ">R":
365 case "R>":
366 case "R@":
367 case "I":
368 case "J":
369 case "UNLOOP":
370 Emit(word, raw);
371 break;
372 case "[']":
373 {
374 string name = ReadName(raw);
375 Word? target = Find(name);
376 if (target == null) Error(name, "undefined word", -13);
377 Emit("push", raw).Value = Xt(target!);
378 break;
379 }
380 case "{:":
381 {
382 List<string> names = new();
383 string? name;
384 bool closed = false;
385 while ((name = ReadWord()) != null)
386 {
387 if (Upper(name) == ":}")
388 {
389 closed = true;
390 break;
391 }
392 names.Add(name);
393 }
394 if (!closed) Syntax(raw, "unterminated locals");
395 List<int> locals = new();
396 foreach (string local in names)
397 {
398 if (local == "--") break;
399 CheckName(local);
400 int index = compiling.LocalCount++;
401 compiling.Locals[Upper(local)] = index;
402 locals.Add(index);
403 }
404 Emit("locals", raw).Locals = locals.ToArray();
405 break;
406 }
407 case ";":
408 {
409 if (controls.Count > 0)
410 {
411 string kind = controls[^1].Kind;
412 if (kind == "ELSE") kind = "IF";
413 if (kind == "WHILE") kind = "BEGIN";
414 Syntax(raw, "unclosed " + kind);
415 }
416 Word complete = compiling;
417 compiling = null;
418 if (complete.Name == ":NONAME") Push(Xt(complete), raw);
419 else AddUser(complete);
420 break;
421 }
422 default:
423 if (compiling.Locals.TryGetValue(word, out int localIndex)) Emit("local", raw).Local = localIndex;
424 else
425 {
426 Word? target = Find(word);
427 if (target != null) Emit("call", raw).Word = target;
428 else
429 {
430 long? value = Number(raw);
431 if (!value.HasValue) Error(raw, "undefined word", -13);
432 Emit("push", raw).Value = value!.Value;
433 }
434 }
435 break;
436 }
437 }

439 void Assign(string raw, bool deferred)
440 {
441 string name = ReadName(raw);
442 if (!deferred && compiling != null && compiling.Locals.TryGetValue(Upper(name), out int local))
443 {
444 Emit("setlocal", raw).Local = local;
445 return;
446 }
447 Word? target = Find(name);
448 if (target == null) Error(name, "undefined word", -13);
449 if (target!.Kind != (deferred ? "defer" : "value")) Syntax(name, deferred ? "not deferred" : "not a value");
450 if (compiling != null) Emit(deferred ? "is" : "to", raw).Word = target;
451 else
452 {
453 long value = Pop(1, raw)[0];
454 if (deferred)
455 {
456 Token(value, raw);
457 target.Deferred = value;
458 }
459 else memory[(int)(target.Address / 8)] = value;
460 }

463 void Process(string raw)
464 {
465 string word = Upper(raw);
466 if (word == "\\")
467 {
468 position = line.Length;
469 return;
470 }
471 if (word == "(")
472 {
473 int end = line.IndexOf(')', position);
474 if (end < 0) Syntax(raw, "unterminated comment");
475 position = end + 1;
476 return;
477 }
478 if (compiling == null && Listed(word, CompileOnly)) Syntax(raw, "compile only");
479 if (compiling != null && Listed(word, InterpretOnly)) Syntax(raw, "interpret only");
480 if (word is ".\"" or "ABORT\"")
481 {
482 int end = position < line.Length ? line.IndexOf('"', position + 1) : -1;
483 if (end < 0) Syntax(raw, "unterminated string");
484 string text = line[(position + 1)..end];
485 position = end + 1;
486 if (compiling != null) Emit(word == ".\"" ? "write" : "abortstring", raw).Text = text;
487 else pending.Append(text);
488 return;
489 }
490 if (word is "TO" or "IS")
491 {
492 Assign(raw, word == "IS");
493 return;
494 }
495 if (word is ":" or ":NONAME")
496 {
497 string name = word == ":" ? ReadName(raw) : raw;
498 if (word == ":") CheckName(name);
499 compiling = new Word(word == ":" ? Upper(name) : ":NONAME", name, "colon", memory.Count * 8L);
500 controls.Clear();
501 return;
502 }
503 if (word == "FORGET")
504 {
505 string name = ReadName(raw);
506 Word? target = newest.GetValueOrDefault(Upper(name));
507 if (target == null) Error(name, "undefined word", -13);
508 int index = target!.DictionaryIndex;
509 long here = target.Here;
510 for (int i = dictionary.Count - 1; i >= index; i--)
511 {
512 Word removed = dictionary[i];
513 if (removed.Previous == null) newest.Remove(removed.Name);
514 else newest[removed.Name] = removed.Previous;
515 }
516 dictionary.RemoveRange(index, dictionary.Count - index);
517 memory.RemoveRange((int)(here / 8), memory.Count - (int)(here / 8));
518 return;
519 }
520 if (compiling != null)
521 {
522 Compile(raw);
523 return;
524 }
525 Word? target = Find(word);
526 if (target != null) Execute(target, raw);
527 else
528 {
529 long? value = Number(raw);
530 if (!value.HasValue) Error(raw, "undefined word", -13);
531 Push(value!.Value, raw);
532 }
533 }

535 void Definition(Word word, int start, string spelling)
536 {
537 if (depth == 100) Error(spelling, "nesting too deep", -7);
538 depth++;
539 int returnDepth = returns.Count;
540 long[] locals = new long[word.LocalCount];
541 int pc = start;
542 while (pc < word.Instructions.Count)
543 {
544 Instruction instruction = word.Instructions[pc++];
545 string raw = instruction.Spelling;
546 switch (instruction.Kind)
547 {
548 case "push":
549 Push(instruction.Value, raw);
550 break;
551 case "local":
552 Push(locals[instruction.Local], raw);
553 break;
554 case "setlocal":
555 locals[instruction.Local] = Pop(1, raw)[0];
556 break;
557 case "locals":
558 {
559 long[] values = Pop(instruction.Locals.Length, raw);
560 for (int i = 0; i < values.Length; i++) locals[instruction.Locals[i]] = values[i];
561 break;
562 }
563 case "call":
564 Execute(instruction.Word!, raw);
565 break;
566 case "write":
567 pending.Append(instruction.Text);
568 break;
569 case "abortstring":
570 if (Pop(1, raw)[0] != 0) Error(raw, instruction.Text, -2);
571 break;
572 case "jump":
573 pc = instruction.Target;
574 break;
575 case "zerojump":
576 if (Pop(1, raw)[0] == 0) pc = instruction.Target;
577 break;
578 case "DO":
579 case "?DO":
580 {
581 long[] values = Pop(2, raw);
582 if (instruction.Kind == "?DO" && values[0] == values[1]) pc = instruction.Target;
583 else
584 {
585 ReturnPush(values[0], raw);
586 ReturnPush(values[1], raw);
587 }
588 break;
589 }
590 case "LOOP":
591 case "+LOOP":
592 {
593 long step = instruction.Kind == "LOOP" ? 1 : Pop(1, raw)[0];
594 NeedReturn(2, raw);
595 long index = returns[^1], limit = returns[^2], before = index - limit, after = index + step - limit;
596 if ((before < 0) != (after < 0)) returns.RemoveRange(returns.Count - 2, 2);
597 else
598 {
599 returns[^1] = index + step;
600 pc = instruction.Target;
601 }
602 break;
603 }
604 case "LEAVE":
605 NeedReturn(2, raw);
606 returns.RemoveRange(returns.Count - 2, 2);
607 pc = instruction.Target;
608 break;
609 case "OF":
610 {
611 long[] values = Pop(2, raw);
612 if (values[0] != values[1])
613 {
614 Push(values[0], raw);
615 pc = instruction.Target;
616 }
617 break;
618 }
619 case "DROP":
620 Pop(1, raw);
621 break;
622 case ">R":
623 ReturnPush(Pop(1, raw)[0], raw);
624 break;
625 case "R>":
626 NeedReturn(1, raw);
627 Push(RemoveLast(returns), raw);
628 break;
629 case "R@":
630 case "I":
631 NeedReturn(1, raw);
632 Push(returns[^1], raw);
633 break;
634 case "J":
635 NeedReturn(3, raw);
636 Push(returns[^3], raw);
637 break;
638 case "UNLOOP":
639 NeedReturn(2, raw);
640 returns.RemoveRange(returns.Count - 2, 2);
641 break;
642 case "to":
643 memory[(int)(instruction.Word!.Address / 8)] = Pop(1, raw)[0];
644 break;
645 case "is":
646 {
647 long value = Pop(1, raw)[0];
648 Token(value, raw);
649 instruction.Word!.Deferred = value;
650 break;
651 }
652 case "EXIT":
653 pc = word.Instructions.Count;
654 break;
655 case "DOES>":
656 if (dictionary.Count == 0 || dictionary[^1].Kind != "create") Error(raw, "no CREATE word", -31);
657 Word created = dictionary[^1];
658 created.DoesOwner = word;
659 created.DoesStart = pc;
660 pc = word.Instructions.Count;
661 break;
662 }
663 }
664 if (returns.Count > returnDepth) returns.RemoveRange(returnDepth, returns.Count - returnDepth);
665 depth--;
666 }

668 void Execute(Word word, string spelling)
669 {
670 while (word.Kind == "defer")
671 {
672 if (word.Deferred == 0) Error(spelling, "deferred word not set", -21);
673 word = Token(word.Deferred, spelling);
674 spelling = word.Name;
675 }
676 switch (word.Kind)
677 {
678 case "builtin":
679 Builtin(word.Name, spelling);
680 break;
681 case "colon":
682 Definition(word, 0, spelling);
683 break;
684 case "constant":
685 Push(word.Value, spelling);
686 break;
687 case "value":
688 Push(memory[(int)(word.Address / 8)], spelling);
689 break;
690 case "create":
691 case "variable":
692 Push(word.Address, spelling);
693 if (word.DoesOwner != null) Definition(word.DoesOwner, word.DoesStart, spelling);
694 break;
695 }

698 static long FloorDivide(long a, long b)
699 {
700 long quotient = a / b, remainder = a % b;
701 return remainder != 0 && (remainder < 0) != (b < 0) ? quotient - 1 : quotient;
702 }

704 void Builtin(string word, string raw)
705 {
706 switch (word)
707 {
708 case "+":
709 case "-":
710 case "*":
711 case "/":
712 case "MOD":
713 case "/MOD":
714 case "MIN":
715 case "MAX":
716 case "=":
717 case "<>":
718 case "<":
719 case ">":
720 case "AND":
721 case "OR":
722 case "XOR":
723 {
724 long[] v = Pop(2, raw);
725 long a = v[0], b = v[1];
726 if (word is "/" or "MOD" or "/MOD")
727 {
728 if (b == 0) Error(raw, "division by zero", -10);
729 long q = FloorDivide(a, b), r = a - b * q;
730 if (word != "/") Push(r, raw);
731 if (word != "MOD") Push(q, raw);
732 }
733 else Push(word switch
734 {
735 "+" => a + b,
736 "-" => a - b,
737 "*" => a * b,
738 "MIN" => Math.Min(a, b),
739 "MAX" => Math.Max(a, b),
740 "=" => a == b ? -1 : 0,
741 "<>" => a != b ? -1 : 0,
742 "<" => a < b ? -1 : 0,
743 ">" => a > b ? -1 : 0,
744 "AND" => a & b,
745 "OR" => a | b,
746 _ => a ^ b
747 }, raw);
748 break;
749 }
750 case "*/":
751 case "*/MOD":
752 {
753 long[] v = Pop(3, raw);
754 if (v[2] == 0) Error(raw, "division by zero", -10);
755 long product = v[0] * v[1], quotient = FloorDivide(product, v[2]);
756 if (word == "*/MOD") Push(product - v[2] * quotient, raw);
757 Push(quotient, raw);
758 break;
759 }
760 case "NEGATE":
761 case "ABS":
762 case "1+":
763 case "1-":
764 case "2*":
765 case "2/":
766 case "0=":
767 case "0<>":
768 case "0<":
769 case "0>":
770 case "INVERT":
771 case "CELLS":
772 case "CELL+":
773 {
774 long a = Pop(1, raw)[0];
775 Push(word switch
776 {
777 "NEGATE" => -a,
778 "ABS" => Math.Abs(a),
779 "1+" => a + 1,
780 "1-" => a - 1,
781 "2*" => 2 * a,
782 "2/" => FloorDivide(a, 2),
783 "0=" => a == 0 ? -1 : 0,
784 "0<>" => a != 0 ? -1 : 0,
785 "0<" => a < 0 ? -1 : 0,
786 "0>" => a > 0 ? -1 : 0,
787 "INVERT" => ~a,
788 "CELLS" => a * 8,
789 _ => a + 8
790 }, raw);
791 break;
792 }
793 case "WITHIN":
794 {
795 long[] v = Pop(3, raw);
796 Push(v[1] <= v[0] && v[0] < v[2] ? -1 : 0, raw);
797 break;
798 }
799 case "TRUE":
800 case "FALSE":
801 case "BL":
802 case "BASE":
803 case "HERE":
804 case "DEPTH":
805 Push(word == "TRUE" ? -1 : word == "BL" ? 32 : word == "HERE" ? memory.Count * 8L : word == "DEPTH" ? data.Count : 0, raw);
806 break;
807 case "DUP":
808 case "?DUP":
809 case "DROP":
810 {
811 long a = Pop(1, raw)[0];
812 if (word != "DROP")
813 {
814 Push(a, raw);
815 if (word == "DUP" || a != 0) Push(a, raw);
816 }
817 break;
818 }
819 case "SWAP":
820 case "OVER":
821 case "NIP":
822 case "TUCK":
823 case "2DUP":
824 case "2DROP":
825 {
826 long[] v = Pop(2, raw);
827 long a = v[0], b = v[1];
828 switch (word)
829 {
830 case "SWAP":
831 Push(b, raw);
832 Push(a, raw);
833 break;
834 case "OVER":
835 Push(a, raw);
836 Push(b, raw);
837 Push(a, raw);
838 break;
839 case "NIP":
840 Push(b, raw);
841 break;
842 case "TUCK":
843 Push(b, raw);
844 Push(a, raw);
845 Push(b, raw);
846 break;
847 case "2DUP":
848 Push(a, raw);
849 Push(b, raw);
850 Push(a, raw);
851 Push(b, raw);
852 break;
853 }
854 break;
855 }
856 case "ROT":
857 case "-ROT":
858 {
859 long[] v = Pop(3, raw);
860 foreach (int i in word == "ROT" ? new[] { 1, 2, 0 } : new[] { 2, 0, 1 }) Push(v[i], raw);
861 break;
862 }
863 case "2SWAP":
864 case "2OVER":
865 {
866 long[] v = Pop(4, raw);
867 foreach (int i in word == "2SWAP" ? new[] { 2, 3, 0, 1 } : new[] { 0, 1, 2, 3, 0, 1 }) Push(v[i], raw);
868 break;
869 }
870 case "PICK":
871 case "ROLL":
872 {
873 long index = Pop(1, raw)[0];
874 if (index < 0) Error(raw, "invalid argument", -24);
875 if (index >= data.Count) Error(raw, "stack underflow", -4);
876 int at = data.Count - 1 - (int)index;
877 long value = data[at];
878 if (word == "ROLL") data.RemoveAt(at);
879 Push(value, raw);
880 break;
881 }
882 case ".":
883 pending.Append(OutputNumber(Pop(1, raw)[0], raw)).Append(' ');
884 break;
885 case ".R":
886 {
887 long[] v = Pop(2, raw);
888 string text = OutputNumber(v[0], raw);
889 if (v[1] > text.Length) pending.Append(' ', (int)v[1] - text.Length);
890 pending.Append(text);
891 break;
892 }
893 case ".S":
894 {
895 string count = OutputNumber(data.Count, raw);
896 pending.Append('<').Append(count).Append("> ");
897 foreach (long value in data) pending.Append(OutputNumber(value, raw)).Append(' ');
898 break;
899 }
900 case "?":
901 {
902 int at = Address(Pop(1, raw)[0], raw);
903 pending.Append(OutputNumber(memory[at], raw)).Append(' ');
904 break;
905 }
906 case "CR":
907 Flush();
908 break;
909 case "SPACE":
910 pending.Append(' ');
911 break;
912 case "SPACES":
913 {
914 long n = Pop(1, raw)[0];
915 if (n > 0) pending.Append(' ', (int)n);
916 break;
917 }
918 case "EMIT":
919 {
920 long value = Pop(1, raw)[0];
921 if (value < 32 || value > 126) Error(raw, "invalid argument", -24);
922 pending.Append((char)value);
923 break;
924 }
925 case "!":
926 case "+!":
927 {
928 long[] v = Pop(2, raw);
929 int at = Address(v[1], raw);
930 memory[at] = word == "!" ? v[0] : memory[at] + v[0];
931 break;
932 }
933 case "@":
934 {
935 int at = Address(Pop(1, raw)[0], raw);
936 Push(memory[at], raw);
937 break;
938 }
939 case ",":
940 memory.Add(Pop(1, raw)[0]);
941 break;
942 case "ALLOT":
943 {
944 long n = Pop(1, raw)[0];
945 if (n < 0 || n % 8 != 0) Error(raw, "invalid argument", -24);
946 for (long i = 0; i < n / 8; i++) memory.Add(0);
947 break;
948 }
949 case "HEX":
950 case "DECIMAL":
951 memory[0] = word == "HEX" ? 16 : 10;
952 break;
953 case "CREATE":
954 case "VARIABLE":
955 case "CONSTANT":
956 case "VALUE":
957 case "DEFER":
958 {
959 string name = ReadName(raw);
960 CheckName(name);
961 long value = word is "CONSTANT" or "VALUE" ? Pop(1, raw)[0] : 0;
962 Word created = new(Upper(name), name, word.ToLowerInvariant(), memory.Count * 8L);
963 created.Address = created.Here;
964 created.Value = value;
965 AddUser(created);
966 if (word is "VARIABLE" or "VALUE") memory.Add(value);
967 break;
968 }
969 case "'":
970 {
971 string name = ReadName(raw);
972 Word? target = Find(name);
973 if (target == null) Error(name, "undefined word", -13);
974 Push(Xt(target!), raw);
975 break;
976 }
977 case "EXECUTE":
978 {
979 Word target = Token(Pop(1, raw)[0], raw);
980 Execute(target, target.Name);
981 break;
982 }
983 case "CATCH":
984 {
985 Word target = Token(Pop(1, raw)[0], raw);
986 List<long> saved = new(data);
987 int returnDepth = returns.Count, savedDepth = depth;
988 long code = 0;
989 try
990 {
991 Execute(target, target.Name);
992 }
993 catch (ForthError failure)
994 {
995 if (!failure.Code.HasValue) throw;
996 data.Clear();
997 data.AddRange(saved);
998 if (returns.Count > returnDepth) returns.RemoveRange(returnDepth, returns.Count - returnDepth);
999 depth = savedDepth;
1000 code = failure.Code.Value;
1001 }
1002 Push(code, raw);
1003 break;
1004 }
1005 case "THROW":
1006 {
1007 long code = Pop(1, raw)[0];
1008 if (code != 0) Error(raw, "exception " + code, code);
1009 break;
1010 }
1011 case "ABORT":
1012 Error(raw, "aborted", -1);
1013 break;
1014 case "WORDS":
1015 for (int i = dictionary.Count - 1; i >= 0; i--) pending.Append(dictionary[i].Name).Append(' ');
1016 break;
1017 }
1018 }

1020 void Input(string text)
1021 {
1022 line = text;
1023 position = 0;
1024 try
1025 {
1026 string? raw;
1027 while ((raw = ReadWord()) != null) Process(raw);
1028 }
1029 catch (ForthError failure)
1030 {
1031 if (pending.Length > 0) Flush();
1032 Console.WriteLine($"ERROR {failure.Word}: {failure.Reason}");
1033 data.Clear();
1034 returns.Clear();
1035 depth = 0;
1036 compiling = null;
1037 controls.Clear();
1038 }
1039 if (pending.Length > 0) Flush();
1040 }
