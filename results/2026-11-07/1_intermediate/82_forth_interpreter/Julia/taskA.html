112function parsedigits(text, base)
113 negative=startswith(text, '-')
114 digits=negative ? text[2:end] : text
115 isempty(digits) && return nothing
116 value=0
117 for c in uppercase(digits)
118 index=findfirst(==(c), DIGITS)
119 (index===nothing || index>base) && return nothing
120 value=value*base+index-1
121 end
122 negative ? -value : value
123end
124function parsenumber(r, text, base = nothing)
125 length(text)==3 && text[1]=='\'' && text[3]=='\'' && return Int(text[2])
126 text[1] in ('\$', '#', '%') &&
127 return parsedigits(text[2:end], text[1]=='\$' ? 16 : text[1]=='#' ? 10 : 2)
128 parsedigits(text, base===nothing ? radix(r) : base)
129end
130function literal(r, text)

135function formatnumber(r, number)
136 base=radix(r)
137 value=abs(number)
138 text=string(DIGITS[mod(value, base)+1])
139 while value>=base
140 value=div(value, base)
141 text=string(DIGITS[mod(value, base)+1])*text
142 end
143 number<0 ? "-"*text : text
144end

212function executeword!(r, word, token)
213 try
214 while word.kind=="defer"
215 word.target===nothing && failure("deferred word not set", -21)
216 word=word.target
217 token=word.name
218 end
219 if word.kind=="builtin"
220 builtin!(r, word.name)
221 elseif word.kind=="colon"
222 rundefinition!(r, word.definition, 1)
223 elseif word.kind in ("create", "variable")
224 pushdata!(r, word.address)
225 word.does!==nothing && rundefinition!(r, word.does...)
226 elseif word.kind=="constant"
227 pushdata!(r, word.value)
228 elseif word.kind=="value"
229 pushdata!(r, r.memory[div(word.address, 8)+1])
230 end
231 catch error
232 error isa ForthFailure || rethrow()
233 error.token===nothing && (error.token=token)
234 rethrow()
235 end
236end
237function builtin!(r, name)
238 if name in (
239 "+",
240 "-",
241 "*",
242 "/",
243 "MOD",
244 "/MOD",
245 "*/",
246 "*/MOD",
247 "MIN",
248 "MAX",
249 "=",
250 "<>",
251 "<",
252 ">",
253 "AND",
254 "OR",
255 "XOR",
256 "WITHIN",
257 )
258 values=takevalues!(r, name in ("*/", "*/MOD", "WITHIN") ? 3 : 2)
259 a, b=values[1:2]
260 if name in ("/", "MOD", "/MOD", "*/", "*/MOD")
261 if length(values)==3
262 a*=b
263 b=values[3]
264 end
265 b==0 && failure("division by zero", -10)
266 q=fld(a, b)
267 rem=a-b*q
268 pushvalues!(r, name in ("/MOD", "*/MOD") ? [rem, q] : [name=="MOD" ? rem : q])
269 else
270 value=name=="+" ? a+b :
271 name=="-" ? a-b :
272 name=="*" ? a*b :
273 name=="MIN" ? min(a, b) :
274 name=="MAX" ? max(a, b) :
275 name=="AND" ? a&b :
276 name=="OR" ? a|b :
277 name=="XOR" ? xor(a, b) :
278 -Int(
279 name=="=" ? a==b :
280 name=="<>" ? a!=b : name=="<" ? a<b : name==">" ? a>b : b<=a<values[3],
281 )
282 pushdata!(r, value)
283 end
284 elseif name in (
285 "NEGATE",
286 "ABS",
287 "1+",
288 "1-",
289 "2*",
290 "2/",
291 "0=",
292 "0<>",
293 "0<",
294 "0>",
295 "INVERT",
296 "CELLS",
297 "CELL+",
298 )
299 a=popdata!(r)
300 value=name=="NEGATE" ? -a :
301 name=="ABS" ? abs(a) :
302 name=="1+" ? a+1 :
303 name=="1-" ? a-1 :
304 name=="2*" ? 2a :
305 name=="2/" ? fld(a, 2) :
306 name=="INVERT" ? -a-1 :
307 name=="CELLS" ? 8a :
308 name=="CELL+" ? a+8 :
309 -Int(name=="0=" ? a==0 : name=="0<>" ? a!=0 : name=="0<" ? a<0 : a>0)
310 pushdata!(r, value)
311 elseif name in (
312 "DUP",
313 "?DUP",
314 "DROP",
315 "SWAP",
316 "OVER",
317 "ROT",
318 "-ROT",
319 "NIP",
320 "TUCK",
321 "2DUP",
322 "2DROP",
323 "2SWAP",
324 "2OVER",
325 )
326 n=name in ("DUP", "?DUP", "DROP") ? 1 :
327 name in ("ROT", "-ROT") ? 3 : name in ("2SWAP", "2OVER") ? 4 : 2
328 v=takevalues!(r, n)
329 out=name=="DUP" ? v[[1, 1]] :
330 name=="?DUP" ? (v[1]==0 ? v : v[[1, 1]]) :
331 name in ("DROP", "2DROP") ? Int[] :
332 name=="SWAP" ? v[[2, 1]] :
333 name=="OVER" ? v[[1, 2, 1]] :
334 name=="ROT" ? v[[2, 3, 1]] :
335 name=="-ROT" ? v[[3, 1, 2]] :
336 name=="NIP" ? v[[2]] :
337 name=="TUCK" ? v[[2, 1, 2]] :
338 name=="2DUP" ? v[[1, 2, 1, 2]] : name=="2SWAP" ? v[[3, 4, 1, 2]] : v[[1, 2, 3, 4, 1, 2]]
339 pushvalues!(r, out)
340 elseif name in ("TRUE", "FALSE", "DEPTH", "BL", "HERE", "BASE")
341 pushdata!(
342 r,
343 name=="TRUE" ? -1 :
344 name=="FALSE" || name=="BASE" ? 0 :
345 name=="DEPTH" ? length(r.stack) : name=="BL" ? 32 : here(r),
346 )
347 elseif name in ("PICK", "ROLL")
348 index=popdata!(r)
349 index<0 && failure("invalid argument", -24)
350 length(r.stack)<=index && failure("stack underflow", -4)
351 at=length(r.stack)-index
352 value=r.stack[at]
353 if name=="ROLL"
354 deleteat!(r.stack, at)
355 end
356 pushdata!(r, value)
357 elseif name=="."
358 value=popdata!(r)
359 writepending!(r, formatnumber(r, value)*" ")
360 elseif name==".R"
361 value, width=takevalues!(r, 2)
362 text=formatnumber(r, value)
363 writepending!(r, repeat(" ", max(0, width-length(text)))*text)
364 elseif name==".S"
365 text="<$(formatnumber(r,length(r.stack)))> "
366 for value in r.stack
367 text*=formatnumber(r, value)*" "
368 end
369 writepending!(r, text)
370 elseif name=="?"
371 address=popdata!(r)
372 value=r.memory[cell(r, address)]
373 writepending!(r, formatnumber(r, value)*" ")
374 elseif name=="CR"
375 flushline!(r)
376 elseif name=="EMIT"
377 code=popdata!(r)
378 32<=code<=126 || failure("invalid argument", -24)
379 writepending!(r, string(Char(code)))
380 elseif name=="SPACE"
381 writepending!(r, " ")
382 elseif name=="SPACES"
383 count=popdata!(r)
384 writepending!(r, repeat(" ", max(0, count)))
385 elseif name in ("!", "+!")
386 value, address=takevalues!(r, 2)
387 at=cell(r, address)
388 r.memory[at]=name=="!" ? value : r.memory[at]+value
389 elseif name=="@"
390 address=popdata!(r)
391 pushdata!(r, r.memory[cell(r, address)])
392 elseif name==","
393 push!(r.memory, popdata!(r))
394 elseif name=="ALLOT"
395 count=popdata!(r)
396 (count<0 || count%8!=0) && failure("invalid argument", -24)
397 append!(r.memory, zeros(Int, div(count, 8)))
398 elseif name in ("HEX", "DECIMAL")
399 r.memory[1]=name=="HEX" ? 16 : 10
400 elseif name in ("CREATE", "VARIABLE", "CONSTANT", "VALUE", "DEFER")
401 word=ForthWord(newname!(r), lowercase(name))
402 word.address=here(r)
403 name in ("CONSTANT", "VALUE") && (word.value=popdata!(r))
404 addword!(r, word)
405 name in ("VARIABLE", "VALUE") && push!(r.memory, name=="VALUE" ? word.value : 0)
406 elseif name=="'"
407 word=existing(r, readname!(r))
408 pushdata!(r, executiontoken!(r, word))
409 elseif name=="EXECUTE"
410 word=tokenword(r, popdata!(r))
411 executeword!(r, word, word.name)
412 elseif name=="CATCH"
413 word=tokenword(r, popdata!(r))
414 saved=copy(r.stack)
415 returnsize=length(r.returns)
416 depth=r.depth
417 try
418 executeword!(r, word, word.name)
419 catch error
420 error isa ForthFailure || rethrow()
421 error.code===nothing && rethrow()
422 r.stack=saved
423 resize!(r.returns, min(returnsize, length(r.returns)))
424 r.depth=depth
425 pushdata!(r, error.code)
426 return
427 end
428 pushdata!(r, 0)
429 elseif name=="THROW"
430 code=popdata!(r)
431 code!=0 && failure("exception $code", code)
432 elseif name=="ABORT"
433 failure("aborted", -1)
434 elseif name=="WORDS"
435 for word in Iterators.reverse(r.dictionary)
436 writepending!(r, word.name*" ")
437 end
438 end
439end
440function rundefinition!(r, definition, start)
441 r.depth==100 && failure("nesting too deep", -7)
442 r.depth+=1
443 returnsize=length(r.returns)
444 frame=zeros(Int, definition.localcount)
445 ip=start
446 while ip<=length(definition.body)
447 action=definition.body[ip]
448 next=ip+1
449 op=action.op
450 arg=action.argument
451 try
452 if op=="call"
453 executeword!(r, arg, action.token)
454 elseif op=="push"
455 pushdata!(r, arg)
456 elseif op=="local"
457 pushdata!(r, frame[arg])
458 elseif op==".\""
459 writepending!(r, arg)
460 elseif op=="branch"
461 next=arg
462 elseif op=="zbranch"
463 popdata!(r)==0 && (next=arg)
464 elseif op in ("DO", "?DO")
465 limit, index=takevalues!(r, 2)
466 if op=="?DO" && limit==index
467 next=arg
468 else
469 pushreturn!(r, limit)
470 pushreturn!(r, index)
471 end
472 elseif op in ("LOOP", "+LOOP")
473 step=op=="LOOP" ? 1 : popdata!(r)
474 requirereturn(r, 2)
475 limit, index=r.returns[(end-1):end]
476 if (index-limit<0)!=(index+step-limit<0)
477 resize!(r.returns, length(r.returns)-2)
478 else
479 r.returns[end]=index+step
480 next=arg
481 end
482 elseif op in ("LEAVE", "UNLOOP")
483 requirereturn(r, 2)
484 resize!(r.returns, length(r.returns)-2)
485 op=="LEAVE" && (next=arg)
486 elseif op==">R"
487 pushreturn!(r, popdata!(r))
488 elseif op=="R>"
489 requirereturn(r, 1)
490 pushdata!(r, pop!(r.returns))
491 elseif op in ("R@", "I", "J")
492 count=op=="J" ? 3 : 1
493 requirereturn(r, count)
494 pushdata!(r, r.returns[end-count+1])
495 elseif op=="OF"
496 selector, candidate=takevalues!(r, 2)
497 if selector!=candidate
498 pushdata!(r, selector)
499 next=arg
500 end
501 elseif op=="ENDCASE"
502 popdata!(r)
503 elseif op=="ABORT\""
504 popdata!(r)!=0 && failure(arg, -2)
505 elseif op=="{:"
506 values=takevalues!(r, length(arg))
507 for (slot, value) in zip(arg, values)
508 frame[slot]=value
509 end
510 elseif op=="to_local"
511 frame[arg]=popdata!(r)
512 elseif op=="to_value"
513 r.memory[div(arg.address, 8)+1]=popdata!(r)
514 elseif op=="IS"
515 setdeferred!(r, arg)
516 elseif op=="RECURSE"
517 rundefinition!(r, arg, 1)
518 elseif op=="DOES>"
519 (isempty(r.dictionary) || r.dictionary[end].kind!="create") &&
520 failure("no CREATE word", -31)
521 r.dictionary[end].does=(definition, arg)
522 next=length(definition.body)+1
523 elseif op=="EXIT"
524 next=length(definition.body)+1
525 end
526 catch error
527 error isa ForthFailure || rethrow()
528 error.token===nothing && (error.token=action.token)
529 rethrow()
530 end
531 ip=next
532 end
533 resize!(r.returns, min(returnsize, length(r.returns)))
534 r.depth-=1
535end
536function addinstruction!(r, op, token, argument = nothing)
537 push!(r.compiling.body, Instruction(op, token, argument))
538 length(r.compiling.body)
539end
540function patch!(definition, indices)
541 for index in indices
542 definition.body[index].argument=length(definition.body)+1
543 end
544end
545function requiretop(r, kinds)
546 controls=r.compiling.controls
547 (isempty(controls) || !(controls[end].kind in kinds)) && failure("control structure mismatch")
548end
549function compiletoken!(r, token, name)
550 d=r.compiling
551 controls=d.controls
552 if name=="IF"
553 push!(controls, ControlRecord("IF", addinstruction!(r, "zbranch", token), Int[]))
554 elseif name=="ELSE"
555 requiretop(r, ("IF",))
556 index=addinstruction!(r, "branch", token)
557 patch!(d, [pop!(controls).origin])
558 push!(controls, ControlRecord("ELSE", index, Int[]))
559 elseif name=="THEN"
560 requiretop(r, ("IF", "ELSE"))
561 patch!(d, [pop!(controls).origin])
562 elseif name=="BEGIN"
563 push!(controls, ControlRecord("BEGIN", length(d.body)+1, Int[]))
564 elseif name in ("UNTIL", "AGAIN")
565 requiretop(r, ("BEGIN",))
566 addinstruction!(r, name=="UNTIL" ? "zbranch" : "branch", token, pop!(controls).origin)
567 elseif name=="WHILE"
568 requiretop(r, ("BEGIN",))
569 index=addinstruction!(r, "zbranch", token)
570 controls[end]=ControlRecord("WHILE", controls[end].origin, [index])
571 elseif name=="REPEAT"
572 requiretop(r, ("WHILE",))
573 loop=pop!(controls)
574 addinstruction!(r, "branch", token, loop.origin)
575 patch!(d, loop.references)
576 elseif name in ("DO", "?DO")
577 index=addinstruction!(r, name, token)
578 push!(controls, ControlRecord(name, index+1, name=="?DO" ? [index] : Int[]))
579 elseif name in ("LOOP", "+LOOP")
580 requiretop(r, ("DO", "?DO"))
581 loop=pop!(controls)
582 addinstruction!(r, name, token, loop.origin)
583 patch!(d, loop.references)
584 elseif name=="LEAVE"
585 index=findlast(c->c.kind in ("DO", "?DO"), controls)
586 index===nothing && failure("control structure mismatch")
587 push!(controls[index].references, addinstruction!(r, "LEAVE", token))
588 elseif name=="CASE"
589 push!(controls, ControlRecord("CASE", 0, Int[]))
590 elseif name=="OF"
591 requiretop(r, ("CASE",))
592 push!(controls, ControlRecord("OF", addinstruction!(r, "OF", token), Int[]))
593 elseif name=="ENDOF"
594 requiretop(r, ("OF",))
595 index=addinstruction!(r, "branch", token)
596 patch!(d, [pop!(controls).origin])
597 push!(controls[end].references, index)
598 elseif name=="ENDCASE"
599 requiretop(r, ("CASE",))
600 addinstruction!(r, "ENDCASE", token)
601 patch!(d, pop!(controls).references)
602 elseif name=="RECURSE"
603 addinstruction!(r, "RECURSE", token, d)
604 elseif name=="DOES>"
605 !isempty(controls) && failure("control structure mismatch")
606 addinstruction!(r, "DOES>", token, length(d.body)+2)
607 elseif name==";"
608 if !isempty(controls)
609 kind=controls[end].kind
610 kind=kind=="ELSE" ? "IF" : kind=="WHILE" ? "BEGIN" : kind
611 failure("unclosed $kind")
612 end
613 r.compiling=nothing
614 word=ForthWord(isempty(d.name) ? ":NONAME" : d.name, "colon")
615 word.definition=d
616 word.here=d.here
617 if isempty(d.name)
618 pushdata!(r, executiontoken!(r, word))
619 else
620 push!(r.dictionary, word)
621 end
622 elseif name=="[']"
623 word=existing(r, readname!(r))
624 addinstruction!(r, "push", token, executiontoken!(r, word))
625 elseif name=="{:"
626 names=String[]
627 while true
628 word=readword!(r)
629 word===nothing && failure("unterminated locals")
630 word==":}" && break
631 push!(names, word)
632 end
633 at=findfirst(==("--"), names)
634 at!==nothing && resize!(names, at-1)
635 for word in names
636 checkname(r, word)
637 end
638 slots=Int[]
639 for word in names
640 d.localcount+=1
641 d.locals[uppercase(word)]=d.localcount
642 push!(slots, d.localcount)
643 end
644 addinstruction!(r, "{:", token, slots)
645 elseif name in ("EXIT", ">R", "R>", "R@", "I", "J", "UNLOOP")
646 addinstruction!(r, name, token)
647 elseif haskey(d.locals, name)
648 addinstruction!(r, "local", token, d.locals[name])
649 else
650 word=findword(r, name)
651 word===nothing ? addinstruction!(r, "push", token, literal(r, token)) :
652 addinstruction!(r, "call", token, word)
653 end
654end
655function processtoken!(r, token)
656 name=uppercase(token)
657 compiling=r.compiling!==nothing
658 if name=="\\"
659 r.position=length(r.line)+1
660 elseif name=="("
661 at=r.position<=length(r.line) ? findnext(')', r.line, r.position) : nothing
662 at===nothing && failure("unterminated comment")
663 r.position=at+1
664 elseif !compiling && name in COMPILE_ONLY
665 failure("compile only")
666 elseif compiling && name in INTERPRET_ONLY
667 failure("interpret only")
668 elseif name in (".\"", "ABORT\"")
669 text=readstring!(r)
670 compiling ? addinstruction!(r, name, token, text) : writepending!(r, text)
671 elseif name in ("TO", "IS")
672 raw=readname!(r)
673 key=uppercase(raw)
674 if name=="TO" && compiling && haskey(r.compiling.locals, key)
675 addinstruction!(r, "to_local", token, r.compiling.locals[key])
676 return
677 end
678 word=existing(r, raw)
679 word.kind==(name=="TO" ? "value" : "defer") ||
680 failure(name=="TO" ? "not a value" : "not deferred", nothing, raw)
681 if compiling
682 addinstruction!(r, name=="TO" ? "to_value" : "IS", token, word)
683 elseif name=="TO"
684 r.memory[div(word.address, 8)+1]=popdata!(r)
685 else
686 setdeferred!(r, word)
687 end
688 elseif name in (":", ":NONAME")
689 raw=name==":" ? readname!(r) : token
690 name==":" && checkname(r, raw)
691 r.compiling=Definition(name==":" ? uppercase(raw) : "", raw, here(r))
692 elseif name=="FORGET"
693 raw=readname!(r)
694 index=findlast(w->w.name==uppercase(raw), r.dictionary)
695 index===nothing && failure("undefined word", -13, raw)
696 resize!(r.memory, div(r.dictionary[index].here, 8))
697 resize!(r.dictionary, index-1)
698 elseif compiling
699 compiletoken!(r, token, name)
700 else
701 word=findword(r, name)
702 word===nothing ? pushdata!(r, literal(r, token)) : executeword!(r, word, token)
703 end
704end
705function main()
706 r=Interpreter()
707 for line in eachline()
708 if line=="BYE"
709 r.compiling!==nothing &&
710 println("ERROR $(r.compiling.display): unterminated definition")
711 break
712 end
713 r.line=line
714 r.position=1
715 token=""
716 try
717 while true
718 next=readword!(r)
719 next===nothing && break
720 token=next
721 processtoken!(r, token)
722 end
723 catch error
724 error isa ForthFailure || rethrow()
725 !isempty(r.pending) && flushline!(r)
726 println("ERROR $(error.token===nothing ? token : error.token): $(error.reason)")
727 empty!(r.stack)
728 empty!(r.returns)
729 r.depth=0
730 r.compiling=nothing
731 end
732 !isempty(r.pending) && flushline!(r)
733 end
734end
735main()
