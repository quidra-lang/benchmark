50static bool listed(const char *word, const char *list) {
51 size_t length = strlen(word);
52 for (const char *at = list; *at;) {
53 const char *end = strchr(at, ' ');
54 size_t n = end ? (size_t)(end - at) : strlen(at);
55 if (n == length && !memcmp(at, word, n)) return true;
56 if (!end) break;
57 at = end + 1;
58 }
59 return false;

250static bool parse_number(const char *raw, long long radix, long long *value) {
251 size_t length = strlen(raw);
252 if (length == 3 && raw[0] == '\'' && raw[2] == '\'') {
253 *value = (unsigned char)raw[1];
254 return true;
255 }
256 const char *text = raw;
257 if (*text && strchr("$#%", *text)) {
258 radix = *text == '$' ? 16 : *text == '#' ? 10 : 2;
259 text++;
260 } else if (radix < 2 || radix > 36) return false;
261 bool negative = *text == '-';
262 if (negative) text++;
263 if (!*text) return false;
264 for (const char *p = text; *p; p++) if (digit_value(*p) >= radix) return false;
265 long long result = 0;
266 for (; *text; text++) result = result * radix + digit_value(*text);
267 *value = negative ? -result : result;
268 return true;
269}

271static bool number(const char *raw, long long *value) {
272 size_t length = strlen(raw);
273 bool direct = (length == 3 && raw[0] == '\'' && raw[2] == '\'') || (*raw && strchr("$#%", *raw));
274 if (!direct && (memory[0] < 2 || memory[0] > 36)) error(raw, "invalid base", -40);
275 return parse_number(raw, memory[0], value);
276}

332static void write_char(char c) {
333 if (c == ' ') {
334 pending.trailing++;
335 return;
336 }
337 size_t needed = pending.length + pending.trailing + 2;
338 if (needed > pending.capacity) {
339 size_t capacity = pending.capacity ? pending.capacity : 64;
340 while (capacity < needed) capacity *= 2;
341 char *next = realloc(pending.text, capacity);
342 if (!next) exit(EXIT_FAILURE);
343 pending.text = next;
344 pending.capacity = capacity;
345 }
346 memset(pending.text + pending.length, ' ', pending.trailing);
347 pending.length += pending.trailing;
348 pending.trailing = 0;
349 pending.text[pending.length++] = c;
350 pending.text[pending.length] = '\0';
351}

368static void output_number(long long value, const char *word, char text[64]) {
369 long long radix = memory[0];
370 if (radix < 2 || radix > 36) error(word, "invalid base", -40);
371 unsigned long long magnitude = (unsigned long long)(value < 0 ? -value : value);
372 int count = 0;
373 do {
374 int digit = (int)(magnitude % (unsigned long long)radix);
375 text[count++] = (char)(digit < 10 ? '0' + digit : 'A' + digit - 10);
376 magnitude /= (unsigned long long)radix;
377 }
378 while (magnitude);
379 if (value < 0) text[count++] = '-';
380 text[count] = '\0';
381 for (int i = 0; i < count / 2; i++) {
382 char c = text[i];
383 text[i] = text[count - i - 1];
384 text[count - i - 1] = c;
385 }
386}

421static void compile(const char *raw, const char *word) {
422 int at = compiling->code.count;
423 if (equal(word, "IF")) {
424 emit("zerojump", raw);
425 add_control("IF", at, 0);
426 } else if (equal(word, "ELSE")) {
427 Control *c = top(raw, "IF");
428 emit("jump", raw);
429 compiling->code.items[c->index].target = at + 1;
430 c->kind = "ELSE";
431 c->index = at;
432 } else if (equal(word, "THEN")) {
433 Control *c = top(raw, "IF ELSE");
434 compiling->code.items[c->index].target = at;
435 control_count--;
436 } else if (equal(word, "BEGIN")) add_control("BEGIN", at, at);
437 else if (equal(word, "UNTIL") || equal(word, "AGAIN")) {
438 Control *c = top(raw, "BEGIN");
439 emit(equal(word, "UNTIL") ? "zerojump" : "jump", raw)->target = c->begin;
440 control_count--;
441 } else if (equal(word, "WHILE")) {
442 Control *c = top(raw, "BEGIN");
443 emit("zerojump", raw);
444 c->kind = "WHILE";
445 c->index = at;
446 } else if (equal(word, "REPEAT")) {
447 Control *c = top(raw, "WHILE");
448 emit("jump", raw)->target = c->begin;
449 compiling->code.items[c->index].target = at + 1;
450 control_count--;
451 } else if (equal(word, "DO") || equal(word, "?DO")) {
452 emit(word, raw);
453 add_control(word, at, at + 1);
454 } else if (equal(word, "LOOP") || equal(word, "+LOOP")) {
455 Control *c = top(raw, "DO ?DO");
456 emit(word, raw)->target = c->begin;
457 compiling->code.items[c->index].target = at + 1;
458 for (int i = 0; i < c->exits.count; i++) compiling->code.items[c->exits.items[i]].target = at + 1;
459 control_count--;
460 } else if (equal(word, "LEAVE")) {
461 Control *loop = NULL;
462 for (int i = control_count - 1; i >= 0; i--) if (equal(controls[i].kind, "DO") || equal(controls[i].kind, "?DO")) {
463 loop = &controls[i];
464 break;
465 }
466 if (!loop) syntax_error(raw, "control structure mismatch");
467 emit("LEAVE", raw);
468 add_integer(&loop->exits, at);
469 } else if (equal(word, "CASE")) add_control("CASE", at, 0);
470 else if (equal(word, "OF")) {
471 top(raw, "CASE");
472 emit("OF", raw);
473 add_control("OF", at, 0);
474 } else if (equal(word, "ENDOF")) {
475 Control *of = top(raw, "OF");
476 int index = of->index;
477 control_count--;
478 Control *c = top(raw, "CASE");
479 emit("jump", raw);
480 compiling->code.items[index].target = at + 1;
481 add_integer(&c->exits, at);
482 } else if (equal(word, "ENDCASE")) {
483 Control *c = top(raw, "CASE");
484 emit("DROP", raw);
485 for (int i = 0; i < c->exits.count; i++) compiling->code.items[c->exits.items[i]].target = at + 1;
486 control_count--;
487 } else if (equal(word, "EXIT")) emit("EXIT", raw);
488 else if (equal(word, "RECURSE")) emit("call", raw)->word = compiling;
489 else if (equal(word, "DOES>")) {
490 if (control_count) syntax_error(raw, "control structure mismatch");
491 emit("DOES>", raw);
492 } else if (listed(word, ">R R> R@ I J UNLOOP")) emit(word, raw);
493 else if (equal(word, "[']")) {
494 const char *name = read_name(raw);
495 Word *target = find(name);
496 if (!target) error(name, "undefined word", -13);
497 emit("push", raw)->value = xt(target);
498 } else if (equal(word, "{:")) {
499 const char **names = allocate((strlen(input_line) + 1) * sizeof(char *));
500 int count = 0;
501 bool closed = false;
502 char *name;
503 while ((name = read_word())) {
504 if (same_name(name, ":}")) {
505 closed = true;
506 break;
507 }
508 names[count++] = name;
509 }
510 if (!closed) syntax_error(raw, "unterminated locals");
511 int *locals = allocate((size_t)count * sizeof(int));
512 int local_count = 0;
513 for (int i = 0; i < count; i++) {
514 if (equal(names[i], "--")) break;
515 check_name(names[i]);
516 locals[local_count++] = add_local(compiling, names[i]);
517 }
518 Instruction *instruction = emit("locals", raw);
519 instruction->locals = locals;
520 instruction->local_count = local_count;
521 } else if (equal(word, ";")) {
522 if (control_count) {
523 const char *kind = controls[control_count - 1].kind;
524 if (equal(kind, "ELSE")) kind = "IF";
525 if (equal(kind, "WHILE")) kind = "BEGIN";
526 raise_error(raw, 0, false, "unclosed %s", kind);
527 }
528 Word *complete = compiling;
529 compiling = NULL;
530 if (equal(complete->name, ":NONAME")) push(xt(complete), raw);
531 else define_word(complete);
532 } else {
533 int local = local_index(compiling, word);
534 if (local >= 0) emit("local", raw)->local = local;
535 else {
536 Word *target = find(word);
537 if (target) emit("call", raw)->word = target;
538 else {
539 long long value;
540 if (!number(raw, &value)) error(raw, "undefined word", -13);
541 emit("push", raw)->value = value;
542 }
543 }
544 }
545}

548static void assign(const char *raw, bool deferred) {
549 const char *name = read_name(raw);
550 int local = !deferred && compiling ? local_index(compiling, name) : -1;
551 if (local >= 0) {
552 emit("setlocal", raw)->local = local;
553 return;
554 }
555 Word *target = find(name);
556 if (!target) error(name, "undefined word", -13);
557 if (!equal(target->kind, deferred ? "defer" : "value")) syntax_error(name, deferred ? "not deferred" : "not a value");
558 if (compiling) emit(deferred ? "is" : "to", raw)->word = target;
559 else {
560 long long value = pop(raw);
561 if (deferred) {
562 token(value, raw);
563 target->deferred = value;
564 } else memory[target->address / 8] = value;
565 }

568static void process(const char *raw) {
569 const char *word = upper(raw);
570 if (equal(word, "\\")) {
571 position = strlen(input_line);
572 return;
573 }
574 if (equal(word, "(")) {
575 const char *end = strchr(input_line + position, ')');
576 if (!end) syntax_error(raw, "unterminated comment");
577 position = (size_t)(end - input_line) + 1;
578 return;
579 }
580 if (!compiling && listed(word, COMPILE_ONLY)) syntax_error(raw, "compile only");
581 if (compiling && listed(word, INTERPRET_ONLY)) syntax_error(raw, "interpret only");
582 if (equal(word, ".\"") || equal(word, "ABORT\"")) {
583 const char *end = input_line[position] ? strchr(input_line + position + 1, '"') : NULL;
584 if (!end) syntax_error(raw, "unterminated string");
585 const char *text = part(input_line + position + 1, (size_t)(end - input_line - position - 1));
586 position = (size_t)(end - input_line) + 1;
587 if (compiling) emit(equal(word, ".\"") ? "write" : "abortstring", raw)->text = text;
588 else write_text(text);
589 return;
590 }
591 if (equal(word, "TO") || equal(word, "IS")) {
592 assign(raw, equal(word, "IS"));
593 return;
594 }
595 if (equal(word, ":") || equal(word, ":NONAME")) {
596 const char *name = equal(word, ":") ? read_name(raw) : raw;
597 if (equal(word, ":")) check_name(name);
598 compiling = new_word(equal(word, ":") ? upper(name) : ":NONAME", name, "colon", memory_count * 8LL);
599 control_count = 0;
600 return;
601 }
602 if (equal(word, "FORGET")) {
603 const char *name = read_name(raw);
604 Word *found = find(name);
605 if (!found || found->dictionary_index < 0) error(name, "undefined word", -13);
606 int index = found->dictionary_index;
607 memory_count = (int)(found->here / 8);
608 for (int i = dictionary.count - 1; i >= index; i--) {
609 Word *removed = dictionary.items[i];
610 entry(removed->name, false)->latest = removed->previous;
611 }
612 dictionary.count = index;
613 return;
614 }
615 if (compiling) {
616 compile(raw, word);
617 return;
618 }
619 Word *target = find(word);
620 if (target) execute(target, raw);
621 else {
622 long long value;
623 if (!number(raw, &value)) error(raw, "undefined word", -13);
624 push(value, raw);
625 }
626}

628static void definition(Word *word, int start, const char *spelling) {
629 if (definition_depth == 100) error(spelling, "nesting too deep", -7);
630 definition_depth++;
631 int return_depth = return_count;
632 RuntimeAllocation *allocation = runtime_allocate((size_t)word->local_count * sizeof(long long));
633 long long *locals = allocation->pointer;
634 int pc = start;
635 while (pc < word->code.count) {
636 Instruction *instruction = &word->code.items[pc++];
637 const char *raw = instruction->spelling, *kind = instruction->kind;
638 if (equal(kind, "push")) push(instruction->value, raw);
639 else if (equal(kind, "local")) push(locals[instruction->local], raw);
640 else if (equal(kind, "setlocal")) locals[instruction->local] = pop(raw);
641 else if (equal(kind, "locals")) {
642 int count = instruction->local_count;
643 if (data_count < count) error(raw, "stack underflow", -4);
644 for (int i = 0; i < count; i++) locals[instruction->locals[i]] = data[data_count - count + i];
645 data_count -= count;
646 } else if (equal(kind, "call")) execute(instruction->word, raw);
647 else if (equal(kind, "write")) write_text(instruction->text);
648 else if (equal(kind, "abortstring")) {
649 if (pop(raw)) error(raw, instruction->text, -2);
650 } else if (equal(kind, "jump")) pc = instruction->target;
651 else if (equal(kind, "zerojump")) {
652 if (!pop(raw)) pc = instruction->target;
653 } else if (equal(kind, "DO") || equal(kind, "?DO")) {
654 long long values[2];
655 pop_values(2, raw, values);
656 if (equal(kind, "?DO") && values[0] == values[1]) pc = instruction->target;
657 else {
658 return_push(values[0], raw);
659 return_push(values[1], raw);
660 }
661 } else if (equal(kind, "LOOP") || equal(kind, "+LOOP")) {
662 long long step = equal(kind, "LOOP") ? 1 : pop(raw);
663 need_return(2, raw);
664 long long index = returns[return_count - 1], limit = returns[return_count - 2], before = index - limit, after = index + step - limit;
665 if ((before < 0) != (after < 0)) return_count -= 2;
666 else {
667 returns[return_count - 1] = index + step;
668 pc = instruction->target;
669 }
670 } else if (equal(kind, "LEAVE")) {
671 need_return(2, raw);
672 return_count -= 2;
673 pc = instruction->target;
674 } else if (equal(kind, "OF")) {
675 long long values[2];
676 pop_values(2, raw, values);
677 if (values[0] != values[1]) {
678 push(values[0], raw);
679 pc = instruction->target;
680 }
681 } else if (equal(kind, "DROP")) pop(raw);
682 else if (equal(kind, ">R")) return_push(pop(raw), raw);
683 else if (equal(kind, "R>")) {
684 need_return(1, raw);
685 push(returns[--return_count], raw);
686 } else if (equal(kind, "R@") || equal(kind, "I")) {
687 need_return(1, raw);
688 push(returns[return_count - 1], raw);
689 } else if (equal(kind, "J")) {
690 need_return(3, raw);
691 push(returns[return_count - 3], raw);
692 } else if (equal(kind, "UNLOOP")) {
693 need_return(2, raw);
694 return_count -= 2;
695 } else if (equal(kind, "to")) memory[instruction->word->address / 8] = pop(raw);
696 else if (equal(kind, "is")) {
697 long long value = pop(raw);
698 token(value, raw);
699 instruction->word->deferred = value;
700 } else if (equal(kind, "EXIT")) pc = word->code.count;
701 else if (equal(kind, "DOES>")) {
702 if (!dictionary.count || !equal(dictionary.items[dictionary.count - 1]->kind, "create")) error(raw, "no CREATE word", -31);
703 Word *created = dictionary.items[dictionary.count - 1];
704 created->does_owner = word;
705 created->does_start = pc;
706 pc = word->code.count;
707 }
708 }
709 if (return_count > return_depth) return_count = return_depth;
710 definition_depth--;
711 runtime_release(allocation);
712}

714static long long floor_divide(long long a, long long b) {
715 long long quotient = a / b, remainder = a % b;
716 return remainder && ((remainder < 0) != (b < 0)) ? quotient - 1 : quotient;
717}

719static void builtin(const char *word, const char *raw) {
720 if (listed(word, "+ - * / MOD /MOD MIN MAX = <> < > AND OR XOR")) {
721 long long values[2];
722 pop_values(2, raw, values);
723 long long a = values[0], b = values[1];
724 if (listed(word, "/ MOD /MOD")) {
725 if (!b) error(raw, "division by zero", -10);
726 long long quotient = floor_divide(a, b), remainder = a - b * quotient;
727 if (!equal(word, "/")) push(remainder, raw);
728 if (!equal(word, "MOD")) push(quotient, raw);
729 } else {
730 long long result;
731 if (equal(word, "+")) result = a + b;
732 else if (equal(word, "-")) result = a - b;
733 else if (equal(word, "*")) result = a * b;
734 else if (equal(word, "MIN")) result = a < b ? a : b;
735 else if (equal(word, "MAX")) result = a > b ? a : b;
736 else if (equal(word, "=")) result = a == b ? -1 : 0;
737 else if (equal(word, "<>")) result = a != b ? -1 : 0;
738 else if (equal(word, "<")) result = a < b ? -1 : 0;
739 else if (equal(word, ">")) result = a > b ? -1 : 0;
740 else if (equal(word, "AND")) result = a & b;
741 else if (equal(word, "OR")) result = a | b;
742 else result = a ^ b;
743 push(result, raw);
744 }
745 } else if (equal(word, "*/") || equal(word, "*/MOD")) {
746 long long values[3];
747 pop_values(3, raw, values);
748 if (!values[2]) error(raw, "division by zero", -10);
749 long long product = values[0] * values[1], quotient = floor_divide(product, values[2]);
750 if (equal(word, "*/MOD")) push(product - values[2] * quotient, raw);
751 push(quotient, raw);
752 } else if (listed(word, "NEGATE ABS 1+ 1- 2* 2/ 0= 0<> 0< 0> INVERT CELLS CELL+")) {
753 long long a = pop(raw), result;
754 if (equal(word, "NEGATE")) result = -a;
755 else if (equal(word, "ABS")) result = a < 0 ? -a : a;
756 else if (equal(word, "1+")) result = a + 1;
757 else if (equal(word, "1-")) result = a - 1;
758 else if (equal(word, "2*")) result = a * 2;
759 else if (equal(word, "2/")) result = floor_divide(a, 2);
760 else if (equal(word, "0=")) result = a == 0 ? -1 : 0;
761 else if (equal(word, "0<>")) result = a != 0 ? -1 : 0;
762 else if (equal(word, "0<")) result = a < 0 ? -1 : 0;
763 else if (equal(word, "0>")) result = a > 0 ? -1 : 0;
764 else if (equal(word, "INVERT")) result = ~a;
765 else if (equal(word, "CELLS")) result = a * 8;
766 else result = a + 8;
767 push(result, raw);
768 } else if (equal(word, "WITHIN")) {
769 long long values[3];
770 pop_values(3, raw, values);
771 push(values[1] <= values[0] && values[0] < values[2] ? -1 : 0, raw);
772 } else if (listed(word, "TRUE FALSE BL BASE HERE DEPTH")) push(equal(word, "TRUE") ? -1 : equal(word, "BL") ? 32 : equal(word, "HERE") ? memory_count * 8LL : equal(word, "DEPTH") ? data_count : 0, raw);
773 else if (listed(word, "DUP ?DUP DROP")) {
774 long long a = pop(raw);
775 if (!equal(word, "DROP")) {
776 push(a, raw);
777 if (equal(word, "DUP") || a) push(a, raw);
778 }
779 } else if (listed(word, "SWAP OVER NIP TUCK 2DUP 2DROP")) {
780 long long values[2];
781 pop_values(2, raw, values);
782 long long a = values[0], b = values[1];
783 if (equal(word, "SWAP")) {
784 push(b, raw);
785 push(a, raw);
786 } else if (equal(word, "OVER")) {
787 push(a, raw);
788 push(b, raw);
789 push(a, raw);
790 } else if (equal(word, "NIP")) push(b, raw);
791 else if (equal(word, "TUCK")) {
792 push(b, raw);
793 push(a, raw);
794 push(b, raw);
795 } else if (equal(word, "2DUP")) {
796 push(a, raw);
797 push(b, raw);
798 push(a, raw);
799 push(b, raw);
800 }
801 } else if (equal(word, "ROT") || equal(word, "-ROT")) {
802 long long values[3];
803 pop_values(3, raw, values);
804 const int forward[3] = {1, 2, 0}, reverse[3] = {2, 0, 1};
805 const int *order = equal(word, "ROT") ? forward : reverse;
806 for (int i = 0; i < 3; i++) push(values[order[i]], raw);
807 } else if (equal(word, "2SWAP") || equal(word, "2OVER")) {
808 long long values[4];
809 pop_values(4, raw, values);
810 const int swap[4] = {2, 3, 0, 1}, over[6] = {0, 1, 2, 3, 0, 1};
811 const int *order = equal(word, "2SWAP") ? swap : over;
812 int count = equal(word, "2SWAP") ? 4 : 6;
813 for (int i = 0; i < count; i++) push(values[order[i]], raw);
814 } else if (equal(word, "PICK") || equal(word, "ROLL")) {
815 long long index = pop(raw);
816 if (index < 0) error(raw, "invalid argument", -24);
817 if (index >= data_count) error(raw, "stack underflow", -4);
818 int at = data_count - 1 - (int)index;
819 long long value = data[at];
820 if (equal(word, "ROLL")) {
821 memmove(data + at, data + at + 1, (size_t)(data_count - at - 1) * sizeof(long long));
822 data_count--;
823 }
824 push(value, raw);
825 } else if (equal(word, ".")) {
826 long long value = pop(raw);
827 char text[64];
828 output_number(value, raw, text);
829 write_text(text);
830 write_char(' ');
831 } else if (equal(word, ".R")) {
832 long long values[2];
833 pop_values(2, raw, values);
834 char text[64];
835 output_number(values[0], raw, text);
836 long long width = values[1] - (long long)strlen(text);
837 write_spaces(width);
838 write_text(text);
839 } else if (equal(word, ".S")) {
840 char text[64];
841 output_number(data_count, raw, text);
842 write_char('<');
843 write_text(text);
844 write_text("> ");
845 for (int i = 0; i < data_count; i++) {
846 output_number(data[i], raw, text);
847 write_text(text);
848 write_char(' ');
849 }
850 } else if (equal(word, "?")) {
851 int at = address(pop(raw), raw);
852 char text[64];
853 output_number(memory[at], raw, text);
854 write_text(text);
855 write_char(' ');
856 } else if (equal(word, "CR")) flush();
857 else if (equal(word, "SPACE")) write_char(' ');
858 else if (equal(word, "SPACES")) write_spaces(pop(raw));
859 else if (equal(word, "EMIT")) {
860 long long value = pop(raw);
861 if (value < 32 || value > 126) error(raw, "invalid argument", -24);
862 write_char((char)value);
863 } else if (equal(word, "!") || equal(word, "+!")) {
864 long long values[2];
865 pop_values(2, raw, values);
866 int at = address(values[1], raw);
867 memory[at] = equal(word, "!") ? values[0] : memory[at] + values[0];
868 } else if (equal(word, "@")) {
869 int at = address(pop(raw), raw);
870 push(memory[at], raw);
871 } else if (equal(word, ",")) {
872 long long value = pop(raw);
873 memory[memory_count++] = value;
874 } else if (equal(word, "ALLOT")) {
875 long long count = pop(raw);
876 if (count < 0 || count % 8) error(raw, "invalid argument", -24);
877 for (long long i = 0; i < count / 8; i++) memory[memory_count++] = 0;
878 } else if (equal(word, "HEX") || equal(word, "DECIMAL")) memory[0] = equal(word, "HEX") ? 16 : 10;
879 else if (listed(word, "CREATE VARIABLE CONSTANT VALUE DEFER")) {
880 const char *name = read_name(raw);
881 check_name(name);
882 long long value = equal(word, "CONSTANT") || equal(word, "VALUE") ? pop(raw) : 0;
883 const char *kind = equal(word, "CREATE") ? "create" : equal(word, "VARIABLE") ? "variable" : equal(word, "CONSTANT") ? "constant" : equal(word, "VALUE") ? "value" : "defer";
884 Word *created = new_word(upper(name), name, kind, memory_count * 8LL);
885 created->address = created->here;
886 created->value = value;
887 define_word(created);
888 if (equal(word, "VARIABLE") || equal(word, "VALUE")) memory[memory_count++] = value;
889 } else if (equal(word, "'")) {
890 const char *name = read_name(raw);
891 Word *target = find(name);
892 if (!target) error(name, "undefined word", -13);
893 push(xt(target), raw);
894 } else if (equal(word, "EXECUTE")) {
895 Word *target = token(pop(raw), raw);
896 execute(target, target->name);
897 } else if (equal(word, "CATCH")) {
898 Word *target = token(pop(raw), raw);
899 int saved_count = data_count, saved_return = return_count, saved_depth = definition_depth;
900 RuntimeAllocation *saved = runtime_allocate((size_t)saved_count * sizeof(long long));
901 if (saved_count) memcpy(saved->pointer, data, (size_t)saved_count * sizeof(long long));
902 ErrorContext context;
903 context.previous = error_context;
904 context.line = false;
905 error_context = &context;
906 long long code = 0;
907 if (setjmp(context.jump) == 0) execute(target, target->name);
908 else {
909 runtime_restore(saved);
910 data_count = saved_count;
911 if (saved_count) memcpy(data, saved->pointer, (size_t)saved_count * sizeof(long long));
912 if (return_count > saved_return) return_count = saved_return;
913 definition_depth = saved_depth;
914 code = failure_code;
915 }
916 error_context = context.previous;
917 runtime_release(saved);
918 push(code, raw);
919 } else if (equal(word, "THROW")) {
920 long long code = pop(raw);
921 if (code) raise_error(raw, code, true, "exception %lld", code);
922 } else if (equal(word, "ABORT")) error(raw, "aborted", -1);
923 else if (equal(word, "WORDS")) {
924 for (int i = dictionary.count - 1; i >= 0; i--) {
925 write_text(dictionary.items[i]->name);
926 write_char(' ');
927 }
928 }
929}

931static void execute(Word *word, const char *spelling) {
932 while (equal(word->kind, "defer")) {
933 if (!word->deferred) error(spelling, "deferred word not set", -21);
934 word = token(word->deferred, spelling);
935 spelling = word->name;
936 }
937 if (equal(word->kind, "builtin")) builtin(word->name, spelling);
938 else if (equal(word->kind, "colon")) definition(word, 0, spelling);
939 else if (equal(word->kind, "constant")) push(word->value, spelling);
940 else if (equal(word->kind, "value")) push(memory[word->address / 8], spelling);
941 else if (equal(word->kind, "create") || equal(word->kind, "variable")) {
942 push(word->address, spelling);
943 if (word->does_owner) definition(word->does_owner, word->does_start, spelling);
944 }
945}

947static char *read_line(void) {
948 size_t length = 0, capacity = 128;
949 char *line = malloc(capacity);
950 if (!line) exit(EXIT_FAILURE);
951 int c;
952 while ((c = getchar()) != EOF && c != '\n') {
953 if (length + 1 == capacity) {
954 capacity *= 2;
955 char *next = realloc(line, capacity);
956 if (!next) exit(EXIT_FAILURE);
957 line = next;
958 }
959 line[length++] = (char)c;
960 }
961 if (c == EOF && !length) {
962 free(line);
963 return NULL;
964 }
965 line[length] = '\0';
966 return line;
967}

969static void input(const char *text) {
970 input_line = text;
971 position = 0;
972 ErrorContext context;
973 context.previous = error_context;
974 context.line = true;
975 error_context = &context;
976 if (setjmp(context.jump) == 0) {
977 char *raw;
978 while ((raw = read_word())) process(raw);
979 } else {
980 if (pending.length || pending.trailing) flush();
981 printf("ERROR %s: %s\n", failure_word, failure_reason);
982 data_count = return_count = definition_depth = 0;
983 compiling = NULL;
984 control_count = 0;
985 runtime_restore(NULL);
986 }
987 error_context = context.previous;
988 if (pending.length || pending.trailing) flush();
989}
