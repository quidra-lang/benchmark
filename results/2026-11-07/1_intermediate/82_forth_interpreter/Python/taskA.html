76def parse_digits(text, base):
77 negative = text.startswith('-')
78 digits = text[1:] if negative else text
79 if not digits:
80 return None
81 value = 0
82 for char in digits.upper():
83 digit = DIGITS.find(char)
84 if not 0 <= digit < base:
85 return None
86 value = value * base + digit
87 return -value if negative else value

198 def push(self, value):
199 if len(self.stack) >= STACK_LIMIT:
200 raise ForthError('stack overflow', -3)
201 self.stack.append(value)
202
203 def pop(self):
204 if not self.stack:
205 raise ForthError('stack underflow', -4)
206 return self.stack.pop()
207
208 def take(self, count):
209 split = len(self.stack) - count
210 if split < 0:
211 raise ForthError('stack underflow', -4)
212 values = self.stack[split:]
213 del self.stack[split:]
214 return values

216 def push_return(self, value):
217 if len(self.return_stack) >= STACK_LIMIT:
218 raise ForthError('return stack overflow', -5)
219 self.return_stack.append(value)
220
221 def require_return(self, count):
222 if len(self.return_stack) < count:
223 raise ForthError('return stack underflow', -6)
224
225 def check_stack_index(self, index):
226 if index < 0:
227 raise ForthError('invalid argument', -24)
228 if len(self.stack) <= index:
229 raise ForthError('stack underflow', -4)

231 def pick(self):
232 index = self.pop()
233 self.check_stack_index(index)
234 self.push(self.stack[-1 - index])

236 def roll(self):
237 index = self.pop()
238 self.check_stack_index(index)
239 self.stack.append(self.stack.pop(-1 - index))

272 def parse_number(self, text, base=None):
273 if len(text) == 3 and text[0] == text[2] == "'":
274 return ord(text[1])
275 if text[0] in PREFIXES:
276 return parse_digits(text[1:], PREFIXES[text[0]])
277 return parse_digits(text, self.radix() if base is None else base)

285 def format_number(self, number):
286 base = self.radix()
287 value = abs(number)
288 digits = DIGITS[value % base]
289 while value >= base:
290 value //= base
291 digits = DIGITS[value % base] + digits
292 return '-' + digits if number < 0 else digits

323 line = self.line
324 start = self.pos

326 start += 1

329 end += 1

410 def catch(self):
411 word = self.xt_word(self.pop())
412 saved_stack = self.stack.copy()
413 saved_return_depth = len(self.return_stack)
414 saved_depth = self.depth
415 try:
416 self.execute(word, word.name)
417 except ForthError as error:
418 if error.code is None:
419 raise
420 self.stack[:] = saved_stack
421 del self.return_stack[saved_return_depth:]
422 self.depth = saved_depth
423 self.push(error.code)
424 else:
425 self.push(0)

435 def execute(self, word, token):
436 try:
437 match word.kind:
438 case 'builtin':
439 word.action()
440 case 'colon':
441 self.run(word.action, 0)
442 case 'create' | 'variable':
443 self.push(word.address)
444 if word.does is not None:
445 self.run(*word.does)
446 case 'constant':
447 self.push(word.value)
448 case 'value':
449 self.push(self.memory[word.address // CELL])
450 case 'defer':
451 if word.target is None:
452 raise ForthError('deferred word not set', -21)
453 self.execute(word.target, word.target.name)
454 except ForthError as error:
455 if error.word is None:
456 error.word = token
457 raise

459 def run(self, definition, start):
460 if self.depth == DEPTH_LIMIT:
461 raise ForthError('nesting too deep', -7)
462 self.depth += 1
463 return_depth = len(self.return_stack)
464 frame = [0] * definition.local_count
465 body = definition.body
466 ip = start
467 try:
468 while ip < len(body):
469 action = body[ip]
470 ip = self.step(definition, action, ip + 1, frame)
471 except ForthError as error:
472 if error.word is None:
473 error.word = action.token
474 raise
475 del self.return_stack[return_depth:]
476 self.depth -= 1

478 def step(self, definition, action, ip, frame):
479 match action.op:
480 case 'call':
481 self.execute(action.arg, action.token)
482 case 'push':
483 self.push(action.arg)
484 case 'local':
485 self.push(frame[action.arg])
486 case '."':
487 self.write(action.arg)
488 case 'branch':
489 return action.arg
490 case 'zbranch':
491 if self.pop() == 0:
492 return action.arg
493 case 'DO' | '?DO':
494 limit, index = self.take(2)
495 if action.op == '?DO' and limit == index:
496 return action.arg
497 self.push_return(limit)
498 self.push_return(index)
499 case 'LOOP' | '+LOOP':
500 increment = 1 if action.op == 'LOOP' else self.pop()
501 self.require_return(2)
502 limit, index = self.return_stack[-2:]
503 if (index - limit < 0) != (index + increment - limit < 0):
504 del self.return_stack[-2:]
505 else:
506 self.return_stack[-1] = index + increment
507 return action.arg
508 case 'LEAVE':
509 self.require_return(2)
510 del self.return_stack[-2:]
511 return action.arg
512 case 'UNLOOP':
513 self.require_return(2)
514 del self.return_stack[-2:]
515 case '>R':
516 self.push_return(self.pop())
517 case 'R>':
518 self.require_return(1)
519 self.push(self.return_stack.pop())
520 case 'R@' | 'I':
521 self.require_return(1)
522 self.push(self.return_stack[-1])
523 case 'J':
524 self.require_return(3)
525 self.push(self.return_stack[-3])
526 case 'OF':
527 selector, candidate = self.take(2)
528 if selector != candidate:
529 self.push(selector)
530 return action.arg
531 case 'ENDCASE':
532 self.pop()
533 case 'ABORT"':
534 if self.pop():
535 raise ForthError(action.arg, -2)
536 case '{:':
537 for slot, value in zip(action.arg, self.take(len(action.arg))):
538 frame[slot] = value
539 case 'to_local':
540 frame[action.arg] = self.pop()
541 case 'to_value':
542 self.memory[action.arg.address // CELL] = self.pop()
543 case 'IS':
544 self.set_deferred(action.arg)
545 case 'RECURSE':
546 self.run(action.arg, 0)
547 case 'DOES>':
548 if not self.dictionary or self.dictionary[-1].kind != 'create':
549 raise ForthError('no CREATE word', -31)
550 self.dictionary[-1].does = (definition, action.arg)
551 return len(definition.body)
552 case 'EXIT':
553 return len(definition.body)
554 return ip

556 def process_line(self, line):
557 self.line = line
558 self.pos = 0
559 try:
560 while (token := self.read_word()) is not None:
561 self.interpret_token(token)
562 except ForthError as error:
563 self.report(error, token)
564 if self.pending:
565 self.flush()

567 def report(self, error, token):
568 if self.pending:
569 self.flush()
570 print(f'ERROR {token if error.word is None else error.word}: {error.reason}')
571 self.stack.clear()
572 self.return_stack.clear()
573 self.depth = 0
574 self.definition = None

576 def interpret_token(self, token):
577 name = token.upper()
578 compiling = self.definition is not None
579 match name:
580 case '\\':
581 self.pos = len(self.line)
582 case '(':
583 end = self.line.find(')', self.pos)
584 if end < 0:
585 raise ForthError('unterminated comment')
586 self.pos = end + 1
587 case _ if not compiling and name in COMPILE_ONLY:
588 raise ForthError('compile only')
589 case _ if compiling and name in INTERPRET_ONLY:
590 raise ForthError('interpret only')
591 case '."' | 'ABORT"':
592 text = self.read_string()
593 if compiling:
594 self.emit(name, token, text)
595 else:
596 self.write(text)
597 case 'TO':
598 self.assign_value(token)
599 case 'IS':
600 self.assign_deferred(token)
601 case ':' | ':NONAME':
602 self.start_definition(token, name)
603 case 'FORGET':
604 self.forget()
605 case _ if compiling:
606 self.compile_token(token, name)
607 case _:
608 word = self.find(name)
609 if word is None:
610 self.push(self.parse_literal(token))
611 else:
612 self.execute(word, token)

614 def assign_value(self, token):
615 name = self.read_name()
616 definition = self.definition
617 if definition is not None and name.upper() in definition.local_names:
618 self.emit('to_local', token, definition.local_names[name.upper()])
619 return
620 word = self.find_existing(name)
621 if word.kind != 'value':
622 raise ForthError('not a value', word=name)
623 if definition is None:
624 self.memory[word.address // CELL] = self.pop()
625 else:
626 self.emit('to_value', token, word)

628 def assign_deferred(self, token):

638 def start_definition(self, token, name):
639 if name == ':NONAME':

646 def end_definition(self):
647 definition = self.definition
648 if definition.control:
649 kind = definition.control[-1].kind
650 raise ForthError(f'unclosed {UNCLOSED_ALIASES.get(kind, kind)}')
651 self.definition = None
652 if definition.name is None:
653 self.push(self.xt_of(Word(':NONAME', 'colon', definition)))
654 else:
655 word = Word(definition.name, 'colon', definition, here=definition.here)

658 def forget(self):
659 name = self.read_name()
660 key = name.upper()
661 for index in range(len(self.dictionary) - 1, -1, -1):
662 word = self.dictionary[index]
663 if word.name == key:
664 del self.dictionary[index:]
665 del self.memory[word.here // CELL:]

684 def declare_locals(self, token):
685 names = []
686 while (word := self.read_word()) != ':}':
687 if word is None:
688 raise ForthError('unterminated locals')
689 names.append(word)
690 if '--' in names:
691 names = names[:names.index('--')]
692 for name in names:
693 self.check_name(name)
694 definition = self.definition
695 slots = []
696 for name in names:
697 definition.local_names[name.upper()] = definition.local_count
698 slots.append(definition.local_count)
699 definition.local_count += 1
700 self.emit('{:', token, slots)

702 def compile_token(self, token, name):
703 definition = self.definition
704 control = definition.control
705 match name:
706 case 'IF':
707 control.append(Control('IF', self.emit('zbranch', token)))
708 case 'ELSE':
709 self.check_top('IF')
710 branch = self.emit('branch', token)
711 self.patch(control.pop().origin)
712 control.append(Control('ELSE', branch))
713 case 'THEN':
714 self.check_top('IF', 'ELSE')
715 self.patch(control.pop().origin)
716 case 'BEGIN':
717 control.append(Control('BEGIN', len(definition.body)))
718 case 'UNTIL' | 'AGAIN':
719 self.check_top('BEGIN')
720 op = 'zbranch' if name == 'UNTIL' else 'branch'
721 self.emit(op, token, control.pop().origin)
722 case 'WHILE':
723 self.check_top('BEGIN')
724 exit_branch = self.emit('zbranch', token)
725 control[-1] = Control('WHILE', control[-1].origin, [exit_branch])
726 case 'REPEAT':
727 self.check_top('WHILE')
728 loop = control.pop()
729 self.emit('branch', token, loop.origin)
730 self.patch(*loop.refs)
731 case 'DO' | '?DO':
732 start = self.emit(name, token)
733 control.append(Control(name, start + 1, [start] if name == '?DO' else []))
734 case 'LOOP' | '+LOOP':
735 self.check_top('DO', '?DO')
736 loop = control.pop()
737 self.emit(name, token, loop.origin)
738 self.patch(*loop.refs)
739 case 'LEAVE':
740 loops = [record for record in control if record.kind in ('DO', '?DO')]
741 if not loops:
742 raise ForthError('control structure mismatch')
743 loops[-1].refs.append(self.emit('LEAVE', token))
744 case 'CASE':
745 control.append(Control('CASE'))
746 case 'OF':
747 self.check_top('CASE')
748 control.append(Control('OF', self.emit('OF', token)))
749 case 'ENDOF':
750 self.check_top('OF')
751 branch = self.emit('branch', token)
752 self.patch(control.pop().origin)
753 control[-1].refs.append(branch)
754 case 'ENDCASE':
755 self.check_top('CASE')
756 self.emit('ENDCASE', token)
757 self.patch(*control.pop().refs)
758 case 'RECURSE':
759 self.emit('RECURSE', token, definition)
760 case 'DOES>':
761 if control:
762 raise ForthError('control structure mismatch')
763 self.emit('DOES>', token, len(definition.body) + 1)
764 case ';':
765 self.end_definition()
766 case "[']":
767 word = self.find_existing(self.read_name())
768 self.emit('push', token, self.xt_of(word))
769 case '{:':
770 self.declare_locals(token)
771 case 'EXIT' | '>R' | 'R>' | 'R@' | 'I' | 'J' | 'UNLOOP':
772 self.emit(name, token)
773 case _ if name in definition.local_names:
774 self.emit('local', token, definition.local_names[name])
775 case _:
776 word = self.find(name)
777 if word is None:
778 self.emit('push', token, self.parse_literal(token))
779 else:
780 self.emit('call', token, word)
781
782 def finish(self):
783 if self.definition is not None:
784 print(f'ERROR {self.definition.display}: unterminated definition')
