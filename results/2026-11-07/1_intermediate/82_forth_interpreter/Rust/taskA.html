120fn compile_only(n: &str) -> bool {
121 [
122 "IF", "ELSE", "THEN", "DO", "?DO", "LOOP", "+LOOP", "LEAVE", "UNLOOP", "I", "J", "BEGIN",
123 "UNTIL", "WHILE", "REPEAT", "AGAIN", "CASE", "OF", "ENDOF", "ENDCASE", "EXIT", "RECURSE",
124 "DOES>", ";", ">R", "R>", "R@", "ABORT\"", "[']", "{:",
125 ]
126 .contains(&n)
127}
128
129fn interpret_only(n: &str) -> bool {
130 [":", ":NONAME", "FORGET"].contains(&n)
131}
132
133fn both(n: &str) -> bool {
134 [".\"", "(", "\\", "TO", "IS"].contains(&n)
135}

241 fn number(&self, raw: &str, owner: &str, base: i64) -> Running<Option<i64>> {
242 let bytes = raw.as_bytes();
243 if bytes.len() == 3 && bytes[0] == b'\'' && bytes[2] == b'\'' {
244 return Ok(Some(bytes[1] as i64));
245 }
246 let upper = raw.to_ascii_uppercase();
247 let mut s = upper.as_str();
248 let mut radix = base;
249 if let Some(c) = s.as_bytes().first() {
250 if b"$#%".contains(c) {
251 radix = match c {
252 b'$' => 16,
253 b'#' => 10,
254 _ => 2,
255 };
256 s = &s[1..];
257 } else if !(2..=36).contains(&radix) {
258 return Err(fault(owner, "invalid base", Some(-40)));
259 }
260 } else if !(2..=36).contains(&radix) {
261 return Err(fault(owner, "invalid base", Some(-40)));
262 }
263 let sign = if let Some(rest) = s.strip_prefix('-') {
264 s = rest;
265 -1
266 } else {
267 1
268 };
269 if s.is_empty() {
270 return Ok(None);
271 }
272 let mut digits = Vec::new();
273 for c in s.bytes() {
274 let d = if c.is_ascii_digit() {
275 (c - b'0') as i64
276 } else if c.is_ascii_uppercase() {
277 (c - b'A' + 10) as i64
278 } else {
279 return Ok(None);
280 };
281 if d >= radix {
282 return Ok(None);
283 }
284 digits.push(d);
285 }
286 let mut value = 0;
287 for d in digits {
288 value = value * radix + d;
289 }
290 Ok(Some(sign * value))
291 }

294 let n = raw.to_ascii_uppercase();
295 if compile_only(&n)
296 || interpret_only(&n)
297 || both(&n)
298 || self.number(raw, raw, 10)?.is_some()
299 {
300 Err(fault(raw, "invalid name", Some(-19)))
301 } else {
302 Ok(())
303 }
304 }

331 fn format(n: i64, base: i64) -> String {
332 let mut v = n.abs();
333 let mut out = Vec::new();
334 loop {
335 out.push(b"0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"[(v % base) as usize] as char);
336 v /= base;
337 if v == 0 {
338 break;
339 }
340 }
341 if n < 0 {
342 out.push('-');
343 }
344 out.into_iter().rev().collect()
345 }

357 q

361 fn execute_definition(&mut self, definition: usize, start: usize, owner: &str) -> Running<()> {
362 if self.depth == 100 {
363 return Err(fault(owner, "nesting too deep", Some(-7)));
364 }
365 self.depth += 1;
366 let old_return = self.returns.len();
367 let mut locals = vec![0; self.words[definition].locals];
368 let mut pc = start;
369 while pc < self.words[definition].code.len() {
370 let i = self.words[definition].code[pc].clone();
371 pc += 1;
372 let s = i.spelling.as_str();
373 match i.op.as_str() {
374 "literal" => self.push(i.value, s)?,
375 "call" => self.execute(i.word, s)?,
376 "local" => self.push(locals[i.value as usize], s)?,
377 "local-set" => locals[i.value as usize] = self.pop(1, s)?[0],
378 "locals" => {
379 let a = self.pop(i.slots.len(), s)?;
380 for (&slot, v) in i.slots.iter().zip(a) {
381 locals[slot] = v;
382 }
383 }
384 "value-set" => {
385 let v = self.pop(1, s)?[0];
386 self.memory[self.words[i.word].address / 8] = v;
387 }
388 "defer-set" => {
389 let v = self.pop(1, s)?[0];
390 self.words[i.word].deferred = Some(self.token_word(v, s)?);
391 }
392 "jump" => pc = i.target,
393 "zero" => {
394 if self.pop(1, s)?[0] == 0 {
395 pc = i.target;
396 }
397 }
398 "exit" => break,
399 "does" => {
400 let latest = self
401 .users
402 .last()
403 .copied()
404 .ok_or_else(|| fault(s, "no CREATE word", Some(-31)))?;
405 if self.words[latest].kind != "create" {
406 return Err(fault(s, "no CREATE word", Some(-31)));
407 }
408 self.words[latest].does = Some((definition, pc));
409 break;
410 }
411 "abort-string" => {
412 if self.pop(1, s)?[0] != 0 {
413 return Err(fault(s, i.text.clone(), Some(-2)));
414 }
415 }
416 "write" => self.pending.push_str(&i.text),
417 "do" | "?do" => {
418 let a = self.pop(2, s)?;
419 if i.op == "?do" && a[0] == a[1] {
420 pc = i.target;
421 } else {
422 self.return_push(a[0], s)?;
423 self.return_push(a[1], s)?;
424 }
425 }
426 "loop" | "+loop" => {
427 let step = if i.op == "loop" {
428 1
429 } else {
430 self.pop(1, s)?[0]
431 };
432 self.return_need(2, s)?;
433 let p = self.returns.len() - 1;
434 let index = self.returns[p];
435 let limit = self.returns[p - 1];
436 let next = index + step;
437 if (index - limit < 0) != (next - limit < 0) {
438 self.returns.truncate(p - 1);
439 } else {
440 self.returns[p] = next;
441 pc = i.target;
442 }
443 }
444 "leave" => {
445 self.return_need(2, s)?;
446 self.returns.truncate(self.returns.len() - 2);
447 pc = i.target;
448 }
449 "of" => {
450 let a = self.pop(2, s)?;
451 if a[0] != a[1] {
452 self.push(a[0], s)?;
453 pc = i.target;
454 }
455 }
456 "endcase" => {
457 self.pop(1, s)?;
458 }
459 ">R" => {
460 let v = self.pop(1, s)?[0];
461 self.return_push(v, s)?;
462 }
463 "R>" => {
464 self.return_need(1, s)?;
465 let v = self.returns.pop().unwrap();
466 self.push(v, s)?;
467 }
468 "R@" | "I" => {
469 self.return_need(1, s)?;
470 self.push(*self.returns.last().unwrap(), s)?;
471 }
472 "J" => {
473 self.return_need(3, s)?;
474 self.push(self.returns[self.returns.len() - 3], s)?;
475 }
476 "UNLOOP" => {
477 self.return_need(2, s)?;
478 self.returns.truncate(self.returns.len() - 2);
479 }
480 _ => unreachable!(),
481 }
482 }
483 if self.returns.len() > old_return {
484 self.returns.truncate(old_return);
485 }
486 self.depth -= 1;
487 Ok(())
488 }

490 fn execute(&mut self, mut i: usize, spelling: &str) -> Running<()> {
491 let mut owner = spelling.to_string();
492 while self.words[i].kind == "defer" {
493 i = self.words[i]
494 .deferred
495 .ok_or_else(|| fault(&owner, "deferred word not set", Some(-21)))?;
496 owner = self.words[i].name.clone();
497 }
498 let spelling = owner.as_str();
499 match self.words[i].kind.as_str() {
500 "colon" => self.execute_definition(i, 0, spelling),
501 "create" | "variable" => {
502 self.push(self.words[i].address as i64, spelling)?;
503 if let Some((d, start)) = self.words[i].does {
504 self.execute_definition(d, start, spelling)?;
505 }
506 Ok(())
507 }
508 "constant" => self.push(self.words[i].value, spelling),
509 "value" => self.push(self.memory[self.words[i].address / 8], spelling),
510 _ => {
511 let name = self.words[i].name.clone();
512 self.builtin(&name, spelling)
513 }
514 }
515 }

517 fn builtin(&mut self, name: &str, owner: &str) -> Running<()> {
518 match name {
519 "+" | "-" | "*" | "/" | "MOD" | "/MOD" | "MIN" | "MAX" | "=" | "<>" | "<" | ">"
520 | "AND" | "OR" | "XOR" => {
521 let a = self.pop(2, owner)?;
522 let (x, y) = (a[0], a[1]);
523 if ["/", "MOD", "/MOD"].contains(&name) && y == 0 {
524 return Err(fault(owner, "division by zero", Some(-10)));
525 }
526 let v = match name {
527 "+" => x + y,
528 "-" => x - y,
529 "*" => x * y,
530 "/" | "/MOD" => Self::quotient(x, y),
531 "MOD" => x - y * Self::quotient(x, y),
532 "MIN" => x.min(y),
533 "MAX" => x.max(y),
534 "=" => -i64::from(x == y),
535 "<>" => -i64::from(x != y),
536 "<" => -i64::from(x < y),
537 ">" => -i64::from(x > y),
538 "AND" => x & y,
539 "OR" => x | y,
540 _ => x ^ y,
541 };
542 if name == "/MOD" {
543 self.push(x - y * v, owner)?;
544 }
545 self.push(v, owner)?;
546 }
547 "*/" | "*/MOD" => {
548 let a = self.pop(3, owner)?;
549 if a[2] == 0 {
550 return Err(fault(owner, "division by zero", Some(-10)));
551 }
552 let p = a[0] * a[1];
553 let q = Self::quotient(p, a[2]);
554 if name == "*/MOD" {
555 self.push(p - a[2] * q, owner)?;
556 }
557 self.push(q, owner)?;
558 }
559 "NEGATE" | "ABS" | "1+" | "1-" | "2*" | "2/" | "0=" | "0<>" | "0<" | "0>"
560 | "INVERT" | "CELLS" | "CELL+" => {
561 let a = self.pop(1, owner)?[0];
562 let v = match name {
563 "NEGATE" => -a,
564 "ABS" => a.abs(),
565 "1+" => a + 1,
566 "1-" => a - 1,
567 "2*" => 2 * a,
568 "2/" => Self::quotient(a, 2),
569 "0=" => -i64::from(a == 0),
570 "0<>" => -i64::from(a != 0),
571 "0<" => -i64::from(a < 0),
572 "0>" => -i64::from(a > 0),
573 "INVERT" => !a,
574 "CELLS" => 8 * a,
575 _ => a + 8,
576 };
577 self.push(v, owner)?;
578 }
579 "WITHIN" => {
580 let a = self.pop(3, owner)?;
581 self.push(-i64::from(a[1] <= a[0] && a[0] < a[2]), owner)?;
582 }
583 "TRUE" => self.push(-1, owner)?,
584 "FALSE" | "BASE" => self.push(0, owner)?,
585 "BL" => self.push(32, owner)?,
586 "DEPTH" => self.push(self.data.len() as i64, owner)?,
587 "HERE" => self.push(self.memory.len() as i64 * 8, owner)?,
588 "DUP" | "?DUP" | "DROP" | "SWAP" | "OVER" | "ROT" | "-ROT" | "NIP" | "TUCK"
589 | "2DUP" | "2DROP" | "2SWAP" | "2OVER" => {
590 let count = match name {
591 "DUP" | "?DUP" | "DROP" => 1,
592 "ROT" | "-ROT" => 3,
593 "2SWAP" | "2OVER" => 4,
594 _ => 2,
595 };
596 let a = self.pop(count, owner)?;
597 let values = match name {
598 "DUP" => vec![a[0], a[0]],
599 "?DUP" => {
600 if a[0] == 0 {
601 a.clone()
602 } else {
603 vec![a[0], a[0]]
604 }
605 }
606 "DROP" | "2DROP" => Vec::new(),
607 "SWAP" => vec![a[1], a[0]],
608 "OVER" => vec![a[0], a[1], a[0]],
609 "ROT" => vec![a[1], a[2], a[0]],
610 "-ROT" => vec![a[2], a[0], a[1]],
611 "NIP" => vec![a[1]],
612 "TUCK" => vec![a[1], a[0], a[1]],
613 "2DUP" => vec![a[0], a[1], a[0], a[1]],
614 "2SWAP" => vec![a[2], a[3], a[0], a[1]],
615 _ => vec![a[0], a[1], a[2], a[3], a[0], a[1]],
616 };
617 self.push_all(&values, owner)?;
618 }
619 "PICK" | "ROLL" => {
620 let u = self.pop(1, owner)?[0];
621 if u < 0 {
622 return Err(fault(owner, "invalid argument", Some(-24)));
623 }
624 if u >= self.data.len() as i64 {
625 return Err(fault(owner, "stack underflow", Some(-4)));
626 }
627 let i = self.data.len() - 1 - u as usize;
628 let v = if name == "ROLL" {
629 self.data.remove(i)
630 } else {
631 self.data[i]
632 };
633 self.push(v, owner)?;
634 }
635 "." => {
636 let n = self.pop(1, owner)?[0];
637 let s = Self::format(n, self.radix(owner)?);
638 self.pending.push_str(&s);
639 self.pending.push(' ');
640 }
641 ".R" => {
642 let a = self.pop(2, owner)?;
643 let s = Self::format(a[0], self.radix(owner)?);
644 if a[1] > s.len() as i64 {
645 self.pending.push_str(&" ".repeat(a[1] as usize - s.len()));
646 }
647 self.pending.push_str(&s);
648 }
649 ".S" => {
650 let b = self.radix(owner)?;
651 self.pending
652 .push_str(&format!("<{}> ", Self::format(self.data.len() as i64, b)));
653 for &n in &self.data {
654 self.pending.push_str(&Self::format(n, b));
655 self.pending.push(' ');
656 }
657 }
658 "?" => {
659 let n = self.pop(1, owner)?[0];
660 let i = self.address(n, owner)?;
661 let s = Self::format(self.memory[i], self.radix(owner)?);
662 self.pending.push_str(&s);
663 self.pending.push(' ');
664 }
665 "CR" => self.flush(),
666 "EMIT" => {
667 let c = self.pop(1, owner)?[0];
668 if !(32..=126).contains(&c) {
669 return Err(fault(owner, "invalid argument", Some(-24)));
670 }
671 self.pending.push(c as u8 as char);
672 }
673 "SPACE" => self.pending.push(' '),
674 "SPACES" => {
675 let n = self.pop(1, owner)?[0];
676 if n > 0 {
677 self.pending.push_str(&" ".repeat(n as usize));
678 }
679 }
680 "!" | "+!" => {
681 let a = self.pop(2, owner)?;
682 let i = self.address(a[1], owner)?;
683 self.memory[i] = if name == "!" {
684 a[0]
685 } else {
686 self.memory[i] + a[0]
687 };
688 }
689 "@" => {
690 let n = self.pop(1, owner)?[0];
691 let i = self.address(n, owner)?;
692 self.push(self.memory[i], owner)?;
693 }
694 "," => {
695 let v = self.pop(1, owner)?[0];
696 self.memory.push(v);
697 }
698 "ALLOT" => {
699 let n = self.pop(1, owner)?[0];
700 if n < 0 || n % 8 != 0 {
701 return Err(fault(owner, "invalid argument", Some(-24)));
702 }
703 self.memory.resize(self.memory.len() + (n / 8) as usize, 0);
704 }
705 "HEX" => self.memory[0] = 16,
706 "DECIMAL" => self.memory[0] = 10,
707 "CREATE" | "VARIABLE" | "CONSTANT" | "VALUE" | "DEFER" => {
708 let raw = self.read_name(owner)?;
709 self.check_name(&raw)?;
710 let value = if name == "CONSTANT" || name == "VALUE" {
711 self.pop(1, owner)?[0]
712 } else {
713 0
714 };
715 let mut w = Word::new(
716 &raw.to_ascii_uppercase(),
717 &raw,
718 &name.to_ascii_lowercase(),
719 self.memory.len() * 8,
720 );
721 w.address = self.memory.len() * 8;
722 w.value = value;
723 let i = self.words.len();
724 self.words.push(w);
725 self.users.push(i);
726 if name == "VARIABLE" || name == "VALUE" {
727 self.memory.push(value);
728 }
729 }
730 "'" => {
731 let raw = self.read_name(owner)?;
732 let w = self
733 .find(&raw)
734 .ok_or_else(|| fault(&raw, "undefined word", Some(-13)))?;
735 let xt = self.token(w);
736 self.push(xt, owner)?;
737 }
738 "EXECUTE" => {
739 let xt = self.pop(1, owner)?[0];
740 let w = self.token_word(xt, owner)?;
741 let n = self.words[w].name.clone();
742 self.execute(w, &n)?;
743 }
744 "CATCH" => {
745 let xt = self.pop(1, owner)?[0];
746 let w = self.token_word(xt, owner)?;
747 let saved = self.data.clone();
748 let size = self.returns.len();
749 let level = self.depth;
750 let n = self.words[w].name.clone();
751 let code = match self.execute(w, &n) {
752 Ok(()) => 0,
753 Err(e) => {
754 let Some(code) = e.code else {
755 return Err(e);
756 };
757 self.data = saved;
758 if self.returns.len() > size {
759 self.returns.truncate(size);
760 }
761 self.depth = level;
762 code
763 }
764 };
765 self.push(code, owner)?;
766 }
767 "THROW" => {
768 let n = self.pop(1, owner)?[0];
769 if n != 0 {
770 return Err(fault(owner, format!("exception {}", n), Some(n)));
771 }
772 }
773 "ABORT" => return Err(fault(owner, "aborted", Some(-1))),
774 "WORDS" => {
775 for &w in self.users.iter().rev() {
776 self.pending.push_str(&self.words[w].name);
777 self.pending.push(' ');
778 }
779 }
780 _ => unreachable!(),
781 }
782 Ok(())
783 }
784
785 fn string(&mut self, owner: &str) -> Running<String> {
786 if self.at == self.line.len() {
787 return Err(fault(owner, "unterminated string", None));
788 }
789 let start = self.at + 1;
790 let offset = self.line[start..]
791 .find('"')
792 .ok_or_else(|| fault(owner, "unterminated string", None))?;
793 self.at = start + offset + 1;
794 Ok(self.line[start..start + offset].into())
795 }
796
797 fn target(&self, raw: &str, kind: &str) -> Running<usize> {
798 let w = self
799 .find(raw)
800 .ok_or_else(|| fault(raw, "undefined word", Some(-13)))?;
801 if self.words[w].kind != kind {
802 Err(fault(
803 raw,
804 if kind == "value" {
805 "not a value"
806 } else {
807 "not deferred"
808 },
809 None,
810 ))
811 } else {
812 Ok(w)
813 }
814 }
815
816 fn process(&mut self, raw: &str) -> Running<()> {
817 let name = raw.to_ascii_uppercase();
818 if name == "\\" {
819 self.at = self.line.len();
820 return Ok(());
821 }
822 if name == "(" {
823 let end = self.line[self.at..]
824 .find(')')
825 .ok_or_else(|| fault(raw, "unterminated comment", None))?;
826 self.at += end + 1;
827 return Ok(());
828 }
829 if self.compiler.is_none() && compile_only(&name) {
830 return Err(fault(raw, "compile only", None));
831 }
832 if self.compiler.is_some() && interpret_only(&name) {
833 return Err(fault(raw, "interpret only", None));
834 }
835 if name == ".\"" || name == "ABORT\"" {
836 let text = self.string(raw)?;
837 if let Some(c) = &self.compiler {
838 let mut i = Instruction::new(
839 if name == ".\"" {
840 "write"
841 } else {
842 "abort-string"
843 },
844 raw,
845 );
846 i.text = text;
847 self.words[c.word].code.push(i);
848 } else {
849 self.pending.push_str(&text);
850 }
851 return Ok(());
852 }
853 if name == "TO" || name == "IS" {
854 let n = self.read_name(raw)?;
855 let upper = n.to_ascii_uppercase();
856 if name == "TO" {
857 if let Some(c) = &self.compiler {
858 if let Some(&slot) = c.locals.get(&upper) {
859 let mut i = Instruction::new("local-set", raw);
860 i.value = slot as i64;
861 self.words[c.word].code.push(i);
862 return Ok(());
863 }
864 }
865 }
866 let w = self.target(&n, if name == "TO" { "value" } else { "defer" })?;
867 if let Some(c) = &self.compiler {
868 let mut i = Instruction::new(
869 if name == "TO" {
870 "value-set"
871 } else {
872 "defer-set"
873 },
874 raw,
875 );
876 i.word = w;
877 self.words[c.word].code.push(i);
878 } else {
879 let v = self.pop(1, raw)?[0];
880 if name == "TO" {
881 self.memory[self.words[w].address / 8] = v;
882 } else {
883 self.words[w].deferred = Some(self.token_word(v, raw)?);
884 }
885 }
886 return Ok(());
887 }
888 if name == ":" || name == ":NONAME" {
889 let n = if name == ":" {
890 let n = self.read_name(raw)?;
891 self.check_name(&n)?;
892 n
893 } else {
894 raw.into()
895 };
896 let i = self.words.len();
897 let upper = if name == ":" {
898 n.to_ascii_uppercase()
899 } else {
900 ":NONAME".into()
901 };
902 self.words
903 .push(Word::new(&upper, &n, "colon", self.memory.len() * 8));
904 self.compiler = Some(Compiler {
905 word: i,
906 anonymous: name == ":NONAME",
907 locals: BTreeMap::new(),
908 control: Vec::new(),
909 });
910 return Ok(());
911 }
912 if name == "FORGET" {
913 let n = self.read_name(raw)?;
914 let upper = n.to_ascii_uppercase();
915 let i = self
916 .users
917 .iter()
918 .rposition(|&w| self.words[w].name == upper)
919 .ok_or_else(|| fault(&n, "undefined word", Some(-13)))?;
920 let here = self.words[self.users[i]].here / 8;
921 self.users.truncate(i);
922 self.memory.truncate(here);
923 return Ok(());
924 }
925 if let Some(mut c) = self.compiler.take() {
926 let keep = self.compile(raw, &name, &mut c)?;
927 if keep {
928 self.compiler = Some(c);
929 }
930 return Ok(());
931 }
932 if let Some(w) = self.find(raw) {
933 self.execute(w, raw)
934 } else {
935 let n = self
936 .number(raw, raw, self.memory[0])?
937 .ok_or_else(|| fault(raw, "undefined word", Some(-13)))?;
938 self.push(n, raw)
939 }
940 }

942 fn instruction(
943 &mut self,
944 c: &Compiler,
945 op: &str,
946 raw: &str,
947 value: i64,
948 target: usize,
949 ) -> usize {
950 let code = &mut self.words[c.word].code;
951 let mut i = Instruction::new(op, raw);
952 i.value = value;
953 i.target = target;
954 code.push(i);
955 code.len() - 1
956 }

958 fn top(c: &Compiler, kinds: &[&str], raw: &str) -> Running<()> {
959 if c.control
960 .last()
961 .is_some_and(|t| kinds.contains(&t.kind.as_str()))
962 {
963 Ok(())
964 } else {
965 Err(fault(raw, "control structure mismatch", None))
966 }
967 }
968
969 fn compile(&mut self, raw: &str, name: &str, c: &mut Compiler) -> Running<bool> {
970 match name {
971 ";" => {
972 if let Some(t) = c.control.last() {
973 let kind = match t.kind.as_str() {
974 "ELSE" => "IF",
975 "WHILE" => "BEGIN",
976 k => k,
977 };
978 return Err(fault(raw, format!("unclosed {}", kind), None));
979 }
980 if c.anonymous {
981 let xt = self.token(c.word);
982 self.push(xt, raw)?;
983 } else {
984 self.users.push(c.word);
985 }
986 return Ok(false);
987 }
988 "IF" => {
989 let i = self.instruction(c, "zero", raw, 0, 0);
990 c.control.push(Control::new("IF", i, 0));
991 }
992 "ELSE" => {
993 Self::top(c, &["IF"], raw)?;
994 let t = c.control.pop().unwrap();
995 let i = self.instruction(c, "jump", raw, 0, 0);
996 self.words[c.word].code[t.index].target = i + 1;
997 c.control.push(Control::new("ELSE", i, 0));
998 }
999 "THEN" => {
1000 Self::top(c, &["IF", "ELSE"], raw)?;
1001 let t = c.control.pop().unwrap();
1002 let end = self.words[c.word].code.len();
1003 self.words[c.word].code[t.index].target = end;
1004 }
1005 "BEGIN" => c
1006 .control
1007 .push(Control::new("BEGIN", self.words[c.word].code.len(), 0)),
1008 "UNTIL" | "AGAIN" => {
1009 Self::top(c, &["BEGIN"], raw)?;
1010 let t = c.control.pop().unwrap();
1011 self.instruction(
1012 c,
1013 if name == "UNTIL" { "zero" } else { "jump" },
1014 raw,
1015 0,
1016 t.index,
1017 );
1018 }
1019 "WHILE" => {
1020 Self::top(c, &["BEGIN"], raw)?;
1021 let t = c.control.pop().unwrap();
1022 let i = self.instruction(c, "zero", raw, 0, 0);
1023 c.control.push(Control::new("WHILE", i, t.index));
1024 }
1025 "REPEAT" => {
1026 Self::top(c, &["WHILE"], raw)?;
1027 let t = c.control.pop().unwrap();
1028 self.instruction(c, "jump", raw, 0, t.begin);
1029 let end = self.words[c.word].code.len();
1030 self.words[c.word].code[t.index].target = end;
1031 }
1032 "DO" | "?DO" => {
1033 let i = self.instruction(c, if name == "DO" { "do" } else { "?do" }, raw, 0, 0);
1034 c.control.push(Control::new(name, i, 0));
1035 }
1036 "LOOP" | "+LOOP" => {
1037 Self::top(c, &["DO", "?DO"], raw)?;
1038 let t = c.control.pop().unwrap();
1039 self.instruction(
1040 c,
1041 if name == "LOOP" { "loop" } else { "+loop" },
1042 raw,
1043 0,
1044 t.index + 1,
1045 );
1046 let end = self.words[c.word].code.len();
1047 self.words[c.word].code[t.index].target = end;
1048 for i in t.leaves {
1049 self.words[c.word].code[i].target = end;
1050 }
1051 }
1052 "LEAVE" => {
1053 let p = c
1054 .control
1055 .iter()
1056 .rposition(|t| t.kind == "DO" || t.kind == "?DO")
1057 .ok_or_else(|| fault(raw, "control structure mismatch", None))?;
1058 let i = self.instruction(c, "leave", raw, 0, 0);
1059 c.control[p].leaves.push(i);
1060 }
1061 "CASE" => c
1062 .control
1063 .push(Control::new("CASE", self.words[c.word].code.len(), 0)),
1064 "OF" => {
1065 Self::top(c, &["CASE"], raw)?;
1066 let i = self.instruction(c, "of", raw, 0, 0);
1067 c.control.push(Control::new("OF", i, 0));
1068 }
1069 "ENDOF" => {
1070 Self::top(c, &["OF"], raw)?;
1071 let t = c.control.pop().unwrap();
1072 let i = self.instruction(c, "jump", raw, 0, 0);
1073 self.words[c.word].code[t.index].target = i + 1;
1074 c.control.last_mut().unwrap().ends.push(i);
1075 }
1076 "ENDCASE" => {
1077 Self::top(c, &["CASE"], raw)?;
1078 let t = c.control.pop().unwrap();
1079 self.instruction(c, "endcase", raw, 0, 0);
1080 let end = self.words[c.word].code.len();
1081 for i in t.ends {
1082 self.words[c.word].code[i].target = end;
1083 }
1084 }
1085 "EXIT" => {
1086 self.instruction(c, "exit", raw, 0, 0);
1087 }
1088 "DOES>" => {
1089 if !c.control.is_empty() {
1090 return Err(fault(raw, "control structure mismatch", None));
1091 }
1092 self.instruction(c, "does", raw, 0, 0);
1093 }
1094 "RECURSE" => {
1095 let mut i = Instruction::new("call", raw);
1096 i.word = c.word;
1097 self.words[c.word].code.push(i);
1098 }
1099 "[']" => {
1100 let n = self.read_name(raw)?;
1101 let word = self
1102 .find(&n)
1103 .ok_or_else(|| fault(&n, "undefined word", Some(-13)))?;
1104 let xt = self.token(word);
1105 self.instruction(c, "literal", raw, xt, 0);
1106 }
1107 "{:" => {
1108 let mut names = Vec::new();
1109 let mut closed = false;
1110 while let Some(n) = self.read_word() {
1111 if n.to_ascii_uppercase() == ":}" {
1112 closed = true;
1113 break;
1114 }
1115 names.push(n);
1116 }
1117 if !closed {
1118 return Err(fault(raw, "unterminated locals", None));
1119 }
1120 let mut i = Instruction::new("locals", raw);
1121 for n in names.iter().take_while(|n| n.as_str() != "--") {
1122 self.check_name(n)?;
1123 let slot = self.words[c.word].locals;
1124 self.words[c.word].locals += 1;
1125 c.locals.insert(n.to_ascii_uppercase(), slot);
1126 i.slots.push(slot);
1127 }
1128 self.words[c.word].code.push(i);
1129 }
1130 ">R" | "R>" | "R@" | "I" | "J" | "UNLOOP" => {
1131 self.instruction(c, name, raw, 0, 0);
1132 }
1133 _ => {
1134 if let Some(&slot) = c.locals.get(name) {
1135 self.instruction(c, "local", raw, slot as i64, 0);
1136 } else if let Some(word) = self.find(raw) {
1137 let mut i = Instruction::new("call", raw);
1138 i.word = word;
1139 self.words[c.word].code.push(i);
1140 } else {
1141 let n = self
1142 .number(raw, raw, self.memory[0])?
1143 .ok_or_else(|| fault(raw, "undefined word", Some(-13)))?;
1144 self.instruction(c, "literal", raw, n, 0);
1145 }
1146 }
1147 }
1148 Ok(true)
1149 }

1151 fn input(&mut self, text: &str) {
1152 self.line = text.into();
1153 self.at = 0;
1154 while let Some(word) = self.read_word() {
1155 if let Err(e) = self.process(&word) {
1156 if !self.pending.is_empty() {
1157 self.flush();
1158 }
1159 self.output.push(format!("ERROR {}: {}", e.word, e.reason));
1160 self.data.clear();
1161 self.returns.clear();
1162 self.compiler = None;
1163 self.depth = 0;
1164 break;
1165 }
1166 }
1167 if !self.pending.is_empty() {
1168 self.flush();
1169 }
1170 }
