148 static Long parseNumber(String raw, long base) {
149 if (raw.length() == 3 && raw.charAt(0) == '\'' && raw.charAt(2) == '\'') return (long)raw.charAt(1);
150 String text = upper(raw);
151 if (!text.isEmpty() && "$#%".indexOf(text.charAt(0)) >= 0) {
152 base = text.charAt(0) == '$' ? 16 : text.charAt(0) == '#' ? 10 : 2;
153 text = text.substring(1);
154 } else if (base < 2 || base > 36) return null;
155 boolean negative = text.startsWith("-");
156 if (negative) text = text.substring(1);
157 if (text.isEmpty()) return null;
158 long result = 0;
159 for (char c : text.toCharArray()) {
160 int digit = c >= '0' && c <= '9' ? c - '0' : c >= 'A' && c <= 'Z' ? c - 'A' + 10 : 36;
161 if (digit >= base) return null;
162 result = result * base + digit;
163 }
164 return negative ? -result : result;
165 }
166 Long number(String raw) {
167 boolean direct = raw.length() == 3 && raw.charAt(0) == '\'' && raw.charAt(2) == '\'' || !raw.isEmpty() && "$#%".indexOf(raw.charAt(0)) >= 0;
168 long base = memory.get(0);
169 if (!direct && (base < 2 || base > 36)) error(raw, "invalid base", -40);
170 return parseNumber(raw, base);
171 }

178 void push(long value, String word) {
179 if (data.size() == 1000) error(word, "stack overflow", -3);
180 data.add(value);
181 }
182 long[] pop(int count, String word) {
183 if (data.size() < count) error(word, "stack underflow", -4);
184 long[] result = new long[count];
185 int first = data.size() - count;
186 for (int i = 0; i < count; i++) result[i] = data.get(first + i);
187 data.subList(first, data.size()).clear();
188 return result;
189 }

243 void compile(String raw) {
244 String word = upper(raw);
245 int at = compiling.instructions.size();
246 switch (word) {
247 case "IF" -> {
248 emit("zerojump", raw);
249 controls.add(new Control("IF", at, 0));
250 }
251 case "ELSE" -> {
252 Control c = top(raw, "IF");
253 emit("jump", raw);
254 compiling.instructions.get(c.index).target = at + 1;
255 c.kind = "ELSE";
256 c.index = at;
257 }
258 case "THEN" -> {
259 Control c = top(raw, "IF", "ELSE");
260 compiling.instructions.get(c.index).target = at;
261 removeControl();
262 }
263 case "BEGIN" -> controls.add(new Control("BEGIN", at, at));
264 case "UNTIL", "AGAIN" -> {
265 Control c = top(raw, "BEGIN");
266 emit(word.equals("UNTIL") ? "zerojump" : "jump", raw).target = c.begin;
267 removeControl();
268 }
269 case "WHILE" -> {
270 Control c = top(raw, "BEGIN");
271 emit("zerojump", raw);
272 c.kind = "WHILE";
273 c.index = at;
274 }
275 case "REPEAT" -> {
276 Control c = top(raw, "WHILE");
277 emit("jump", raw).target = c.begin;
278 compiling.instructions.get(c.index).target = at + 1;
279 removeControl();
280 }
281 case "DO", "?DO" -> {
282 emit(word, raw);
283 controls.add(new Control(word, at, at + 1));
284 }
285 case "LOOP", "+LOOP" -> {
286 Control c = top(raw, "DO", "?DO");
287 emit(word, raw).target = c.begin;
288 compiling.instructions.get(c.index).target = at + 1;
289 for (int leave : c.exits) compiling.instructions.get(leave).target = at + 1;
290 removeControl();
291 }
292 case "LEAVE" -> {
293 Control loop = null;
294 for (int i = controls.size() - 1; i >= 0; i--) if (controls.get(i).kind.equals("DO") || controls.get(i).kind.equals("?DO")) {
295 loop = controls.get(i);
296 break;
297 }
298 if (loop == null) syntax(raw, "control structure mismatch");
299 emit("LEAVE", raw);
300 loop.exits.add(at);
301 }
302 case "CASE" -> controls.add(new Control("CASE", at, 0));
303 case "OF" -> {
304 top(raw, "CASE");
305 emit("OF", raw);
306 controls.add(new Control("OF", at, 0));
307 }
308 case "ENDOF" -> {
309 Control of = top(raw, "OF");
310 removeControl();
311 Control c = top(raw, "CASE");
312 emit("jump", raw);
313 compiling.instructions.get(of.index).target = at + 1;
314 c.exits.add(at);
315 }
316 case "ENDCASE" -> {
317 Control c = top(raw, "CASE");
318 emit("DROP", raw);
319 for (int jump : c.exits) compiling.instructions.get(jump).target = at + 1;
320 removeControl();
321 }
322 case "EXIT" -> emit("EXIT", raw);
323 case "RECURSE" -> emit("call", raw).word = compiling;
324 case "DOES>" -> {
325 if (!controls.isEmpty()) syntax(raw, "control structure mismatch");
326 emit("DOES>", raw);
327 }
328 case ">R", "R>", "R@", "I", "J", "UNLOOP" -> emit(word, raw);
329 case "[']" -> {
330 String name = readName(raw);
331 Word target = find(name);
332 if (target == null) error(name, "undefined word", -13);
333 emit("push", raw).value = xt(target);
334 }
335 case "{:" -> {
336 List<String> names = new ArrayList<>();
337 String name;
338 boolean closed = false;
339 while ((name = readWord()) != null) {
340 if (upper(name).equals(":}")) {
341 closed = true;
342 break;
343 }
344 names.add(name);
345 }
346 if (!closed) syntax(raw, "unterminated locals");
347 List<Integer> locals = new ArrayList<>();
348 for (String local : names) {
349 if (local.equals("--")) break;
350 checkName(local);
351 int index = compiling.localCount++;
352 compiling.locals.put(upper(local), index);
353 locals.add(index);
354 }
355 emit("locals", raw).locals = locals.stream().mapToInt(Integer::intValue).toArray();
356 }
357 case ";" -> {
358 if (!controls.isEmpty()) {
359 String kind = controls.getLast().kind;
360 if (kind.equals("ELSE")) kind = "IF";
361 if (kind.equals("WHILE")) kind = "BEGIN";
362 syntax(raw, "unclosed " + kind);
363 }
364 Word complete = compiling;
365 compiling = null;
366 if (complete.name.equals(":NONAME")) push(xt(complete), raw);
367 else addUser(complete);
368 }
369 default -> {
370 Integer local = compiling.locals.get(word);
371 if (local != null) emit("local", raw).local = local;
372 else {
373 Word target = find(word);
374 if (target != null) emit("call", raw).word = target;
375 else {
376 Long value = number(raw);
377 if (value == null) error(raw, "undefined word", -13);
378 emit("push", raw).value = value;
379 }
380 }
381 }
382 }
383 }

385 void assign(String raw, boolean deferred) {
386 String name = readName(raw);
387 Integer local = !deferred && compiling != null ? compiling.locals.get(upper(name)) : null;
388 if (local != null) {
389 emit("setlocal", raw).local = local;
390 return;
391 }
392 Word target = find(name);
393 if (target == null) error(name, "undefined word", -13);
394 if (!target.kind.equals(deferred ? "defer" : "value")) syntax(name, deferred ? "not deferred" : "not a value");
395 if (compiling != null) emit(deferred ? "is" : "to", raw).word = target;
396 else {
397 long value = pop(1, raw)[0];
398 if (deferred) {
399 token(value, raw);
400 target.deferred = value;
401 } else memory.set((int)(target.address / 8), value);
402 }

405 void process(String raw) {
406 String word = upper(raw);
407 if (word.equals("\\")) {
408 position = line.length();
409 return;
410 }
411 if (word.equals("(")) {
412 int end = line.indexOf(')', position);
413 if (end < 0) syntax(raw, "unterminated comment");
414 position = end + 1;
415 return;
416 }
417 if (compiling == null && listed(word, COMPILE_ONLY)) syntax(raw, "compile only");
418 if (compiling != null && listed(word, INTERPRET_ONLY)) syntax(raw, "interpret only");
419 if (word.equals(".\"") || word.equals("ABORT\"")) {
420 int end = position < line.length() ? line.indexOf('"', position + 1) : -1;
421 if (end < 0) syntax(raw, "unterminated string");
422 String text = line.substring(position + 1, end);
423 position = end + 1;
424 if (compiling != null) emit(word.equals(".\"") ? "write" : "abortstring", raw).text = text;
425 else pending.append(text);
426 return;
427 }
428 if (word.equals("TO") || word.equals("IS")) {
429 assign(raw, word.equals("IS"));
430 return;
431 }
432 if (word.equals(":") || word.equals(":NONAME")) {
433 String name = word.equals(":") ? readName(raw) : raw;
434 if (word.equals(":")) checkName(name);
435 compiling = new Word(word.equals(":") ? upper(name) : ":NONAME", name, "colon", memory.size() * 8L);
436 controls.clear();
437 return;
438 }
439 if (word.equals("FORGET")) {
440 String name = readName(raw);
441 Word target = newest.get(upper(name));
442 if (target == null) error(name, "undefined word", -13);
443 int index = target.dictionaryIndex;
444 long here = target.here;
445 for (int i = dictionary.size() - 1; i >= index; i--) {
446 Word removed = dictionary.get(i);
447 if (removed.previous == null) newest.remove(removed.name);
448 else newest.put(removed.name, removed.previous);
449 }
450 dictionary.subList(index, dictionary.size()).clear();
451 memory.subList((int)(here / 8), memory.size()).clear();
452 return;
453 }
454 if (compiling != null) {
455 compile(raw);
456 return;
457 }
458 Word target = find(word);
459 if (target != null) execute(target, raw);
460 else {
461 Long value = number(raw);
462 if (value == null) error(raw, "undefined word", -13);
463 push(value, raw);
464 }
465 }

467 void definition(Word word, int start, String spelling) {
468 if (depth == 100) error(spelling, "nesting too deep", -7);
469 depth++;
470 int returnDepth = returns.size();
471 long[] locals = new long[word.localCount];
472 int pc = start;
473 while (pc < word.instructions.size()) {
474 Instruction instruction = word.instructions.get(pc++);
475 String raw = instruction.spelling;
476 switch (instruction.kind) {
477 case "push" -> push(instruction.value, raw);
478 case "local" -> push(locals[instruction.local], raw);
479 case "setlocal" -> locals[instruction.local] = pop(1, raw)[0];
480 case "locals" -> {
481 long[] values = pop(instruction.locals.length, raw);
482 for (int i = 0; i < values.length; i++) locals[instruction.locals[i]] = values[i];
483 }
484 case "call" -> execute(instruction.word, raw);
485 case "write" -> pending.append(instruction.text);
486 case "abortstring" -> {
487 if (pop(1, raw)[0] != 0) error(raw, instruction.text, -2);
488 }
489 case "jump" -> pc = instruction.target;
490 case "zerojump" -> {
491 if (pop(1, raw)[0] == 0) pc = instruction.target;
492 }
493 case "DO", "?DO" -> {
494 long[] values = pop(2, raw);
495 if (instruction.kind.equals("?DO") && values[0] == values[1]) pc = instruction.target;
496 else {
497 returnPush(values[0], raw);
498 returnPush(values[1], raw);
499 }
500 }
501 case "LOOP", "+LOOP" -> {
502 long step = instruction.kind.equals("LOOP") ? 1 : pop(1, raw)[0];
503 needReturn(2, raw);
504 long index = returns.getLast(), limit = returns.get(returns.size() - 2), before = index - limit, after = index + step - limit;
505 if ((before < 0) != (after < 0)) returns.subList(returns.size() - 2, returns.size()).clear();
506 else {
507 returns.set(returns.size() - 1, index + step);
508 pc = instruction.target;
509 }
510 }
511 case "LEAVE" -> {
512 needReturn(2, raw);
513 returns.subList(returns.size() - 2, returns.size()).clear();
514 pc = instruction.target;
515 }
516 case "OF" -> {
517 long[] values = pop(2, raw);
518 if (values[0] != values[1]) {
519 push(values[0], raw);
520 pc = instruction.target;
521 }
522 }
523 case "DROP" -> pop(1, raw);
524 case ">R" -> returnPush(pop(1, raw)[0], raw);
525 case "R>" -> {
526 needReturn(1, raw);
527 push(returns.removeLast(), raw);
528 }
529 case "R@", "I" -> {
530 needReturn(1, raw);
531 push(returns.getLast(), raw);
532 }
533 case "J" -> {
534 needReturn(3, raw);
535 push(returns.get(returns.size() - 3), raw);
536 }
537 case "UNLOOP" -> {
538 needReturn(2, raw);
539 returns.subList(returns.size() - 2, returns.size()).clear();
540 }
541 case "to" -> memory.set((int)(instruction.word.address / 8), pop(1, raw)[0]);
542 case "is" -> {
543 long value = pop(1, raw)[0];
544 token(value, raw);
545 instruction.word.deferred = value;
546 }
547 case "EXIT" -> pc = word.instructions.size();
548 case "DOES>" -> {
549 if (dictionary.isEmpty() || !dictionary.getLast().kind.equals("create")) error(raw, "no CREATE word", -31);
550 Word created = dictionary.getLast();
551 created.doesOwner = word;
552 created.doesStart = pc;
553 pc = word.instructions.size();
554 }
555 default -> throw new IllegalStateException();
556 }
557 }
558 if (returns.size() > returnDepth) returns.subList(returnDepth, returns.size()).clear();
559 depth--;
560 }

562 void execute(Word word, String spelling) {
563 while (word.kind.equals("defer")) {
564 if (word.deferred == 0) error(spelling, "deferred word not set", -21);
565 word = token(word.deferred, spelling);
566 spelling = word.name;
567 }
568 switch (word.kind) {
569 case "builtin" -> builtin(word.name, spelling);
570 case "colon" -> definition(word, 0, spelling);
571 case "constant" -> push(word.value, spelling);
572 case "value" -> push(memory.get((int)(word.address / 8)), spelling);
573 case "create", "variable" -> {
574 push(word.address, spelling);
575 if (word.doesOwner != null) definition(word.doesOwner, word.doesStart, spelling);
576 }
577 default -> throw new IllegalStateException();
578 }
579 }

581 void builtin(String word, String raw) {
582 switch (word) {
583 case "+", "-", "*", "/", "MOD", "/MOD", "MIN", "MAX", "=", "<>", "<", ">", "AND", "OR", "XOR" -> {
584 long[] v = pop(2, raw);
585 long a = v[0], b = v[1];
586 if (word.equals("/") || word.equals("MOD") || word.equals("/MOD")) {
587 if (b == 0) error(raw, "division by zero", -10);
588 long q = Math.floorDiv(a, b), r = a - b * q;
589 if (!word.equals("/")) push(r, raw);
590 if (!word.equals("MOD")) push(q, raw);
591 } else push(switch (word) {
592 case "+" -> a + b;
593 case "-" -> a - b;
594 case "*" -> a * b;
595 case "MIN" -> Math.min(a, b);
596 case "MAX" -> Math.max(a, b);
597 case "=" -> a == b ? -1 : 0;
598 case "<>" -> a != b ? -1 : 0;
599 case "<" -> a < b ? -1 : 0;
600 case ">" -> a > b ? -1 : 0;
601 case "AND" -> a & b;
602 case "OR" -> a | b;
603 default -> a ^ b;
604 }, raw);
605 }
606 case "*/", "*/MOD" -> {
607 long[] v = pop(3, raw);
608 if (v[2] == 0) error(raw, "division by zero", -10);
609 long product = v[0] * v[1], quotient = Math.floorDiv(product, v[2]);
610 if (word.equals("*/MOD")) push(product - v[2] * quotient, raw);
611 push(quotient, raw);
612 }
613 case "NEGATE", "ABS", "1+", "1-", "2*", "2/", "0=", "0<>", "0<", "0>", "INVERT", "CELLS", "CELL+" -> {
614 long a = pop(1, raw)[0];
615 push(switch (word) {
616 case "NEGATE" -> -a;
617 case "ABS" -> Math.abs(a);
618 case "1+" -> a + 1;
619 case "1-" -> a - 1;
620 case "2*" -> 2 * a;
621 case "2/" -> Math.floorDiv(a, 2);
622 case "0=" -> a == 0 ? -1 : 0;
623 case "0<>" -> a != 0 ? -1 : 0;
624 case "0<" -> a < 0 ? -1 : 0;
625 case "0>" -> a > 0 ? -1 : 0;
626 case "INVERT" -> ~a;
627 case "CELLS" -> a * 8;
628 default -> a + 8;
629 }, raw);
630 }
631 case "WITHIN" -> {
632 long[] v = pop(3, raw);
633 push(v[1] <= v[0] && v[0] < v[2] ? -1 : 0, raw);
634 }
635 case "TRUE", "FALSE", "BL", "BASE", "HERE", "DEPTH" -> push(word.equals("TRUE") ? -1 : word.equals("BL") ? 32 : word.equals("HERE") ? memory.size() * 8L : word.equals("DEPTH") ? data.size() : 0, raw);
636 case "DUP", "?DUP", "DROP" -> {
637 long a = pop(1, raw)[0];
638 if (!word.equals("DROP")) {
639 push(a, raw);
640 if (word.equals("DUP") || a != 0) push(a, raw);
641 }
642 }
643 case "SWAP", "OVER", "NIP", "TUCK", "2DUP", "2DROP" -> {
644 long[] v = pop(2, raw);
645 long a = v[0], b = v[1];
646 switch (word) {
647 case "SWAP" -> {
648 push(b, raw);
649 push(a, raw);
650 }
651 case "OVER" -> {
652 push(a, raw);
653 push(b, raw);
654 push(a, raw);
655 }
656 case "NIP" -> push(b, raw);
657 case "TUCK" -> {
658 push(b, raw);
659 push(a, raw);
660 push(b, raw);
661 }
662 case "2DUP" -> {
663 push(a, raw);
664 push(b, raw);
665 push(a, raw);
666 push(b, raw);
667 }
668 }
669 }
670 case "ROT", "-ROT" -> {
671 long[] v = pop(3, raw);
672 int[] order = word.equals("ROT") ? new int[]{1, 2, 0} : new int[]{2, 0, 1};
673 for (int i : order) push(v[i], raw);
674 }
675 case "2SWAP", "2OVER" -> {
676 long[] v = pop(4, raw);
677 int[] order = word.equals("2SWAP") ? new int[]{2, 3, 0, 1} : new int[]{0, 1, 2, 3, 0, 1};
678 for (int i : order) push(v[i], raw);
679 }
680 case "PICK", "ROLL" -> {
681 long index = pop(1, raw)[0];
682 if (index < 0) error(raw, "invalid argument", -24);
683 if (index >= data.size()) error(raw, "stack underflow", -4);
684 int at = data.size() - 1 - (int)index;
685 long value = word.equals("ROLL") ? data.remove(at) : data.get(at);
686 push(value, raw);
687 }
688 case "." -> pending.append(outputNumber(pop(1, raw)[0], raw)).append(' ');
689 case ".R" -> {
690 long[] v = pop(2, raw);
691 String text = outputNumber(v[0], raw);
692 if (v[1] > text.length()) pending.spaces(v[1] - text.length());
693 pending.append(text);
694 }
695 case ".S" -> {
696 String count = outputNumber(data.size(), raw);
697 pending.append('<').append(count).append("> ");
698 for (long value : data) pending.append(outputNumber(value, raw)).append(' ');
699 }
700 case "?" -> {
701 int address = address(pop(1, raw)[0], raw);
702 pending.append(outputNumber(memory.get(address), raw)).append(' ');
703 }
704 case "CR" -> flush();
705 case "SPACE" -> pending.append(' ');
706 case "SPACES" -> {
707 long n = pop(1, raw)[0];
708 if (n > 0) pending.spaces(n);
709 }
710 case "EMIT" -> {
711 long value = pop(1, raw)[0];
712 if (value < 32 || value > 126) error(raw, "invalid argument", -24);
713 pending.append((char)value);
714 }
715 case "!", "+!" -> {
716 long[] v = pop(2, raw);
717 int at = address(v[1], raw);
718 memory.set(at, word.equals("!") ? v[0] : memory.get(at) + v[0]);
719 }
720 case "@" -> {
721 int at = address(pop(1, raw)[0], raw);
722 push(memory.get(at), raw);
723 }
724 case "," -> memory.add(pop(1, raw)[0]);
725 case "ALLOT" -> {
726 long n = pop(1, raw)[0];
727 if (n < 0 || n % 8 != 0) error(raw, "invalid argument", -24);
728 for (long i = 0; i < n / 8; i++) memory.add(0L);
729 }
730 case "HEX", "DECIMAL" -> memory.set(0, word.equals("HEX") ? 16L : 10L);
731 case "CREATE", "VARIABLE", "CONSTANT", "VALUE", "DEFER" -> {
732 String name = readName(raw);
733 checkName(name);
734 long value = word.equals("CONSTANT") || word.equals("VALUE") ? pop(1, raw)[0] : 0;
735 Word created = new Word(upper(name), name, word.toLowerCase(java.util.Locale.ROOT), memory.size() * 8L);
736 created.address = created.here;
737 created.value = value;
738 addUser(created);
739 if (word.equals("VARIABLE") || word.equals("VALUE")) memory.add(value);
740 }
741 case "'" -> {
742 String name = readName(raw);
743 Word target = find(name);
744 if (target == null) error(name, "undefined word", -13);
745 push(xt(target), raw);
746 }
747 case "EXECUTE" -> {
748 Word target = token(pop(1, raw)[0], raw);
749 execute(target, target.name);
750 }
751 case "CATCH" -> {
752 Word target = token(pop(1, raw)[0], raw);
753 List<Long> saved = new ArrayList<>(data);
754 int returnDepth = returns.size(), savedDepth = depth;
755 long code = 0;
756 try {
757 execute(target, target.name);
758 } catch (ForthError failure) {
759 if (failure.code == null) throw failure;
760 data.clear();
761 data.addAll(saved);
762 if (returns.size() > returnDepth) returns.subList(returnDepth, returns.size()).clear();
763 depth = savedDepth;
764 code = failure.code;
765 }
766 push(code, raw);
767 }
768 case "THROW" -> {
769 long code = pop(1, raw)[0];
770 if (code != 0) error(raw, "exception " + code, code);
771 }
772 case "ABORT" -> error(raw, "aborted", -1);
773 case "WORDS" -> {
774 for (int i = dictionary.size() - 1; i >= 0; i--) pending.append(dictionary.get(i).name).append(' ');
775 }
776 default -> throw new IllegalStateException();
777 }
778 }

780 void input(String text) {
781 line = text;
782 position = 0;
783 try {
784 String raw;
785 while ((raw = readWord()) != null) process(raw);
786 } catch (ForthError failure) {
787 if (!pending.isEmpty()) flush();
788 System.out.println("ERROR " + failure.word + ": " + failure.reason);
789 data.clear();
790 returns.clear();
791 depth = 0;
792 compiling = null;
793 controls.clear();
794 }
795 if (!pending.isEmpty()) flush();
796 }
