99 fun parseNumber(raw: String, owner: String, base: Long = memory[0]): Long? {
100 if (raw.length == 3 && raw.first() == '\'' && raw.last() == '\'') return raw[1].code.toLong()
101 var text = raw.uppercase()
102 var radix = base
103 if (text.firstOrNull() in listOf('$', '#', '%')) {
104 radix = when (text[0]) {
105 '$' -> 16
106 '#' -> 10
107 else -> 2
108 }
109 text = text.drop(1)
110 } else if (radix !in 2L..36L) fault(owner, "invalid base", -40)
111 var sign = 1L
112 if (text.startsWith('-')) {
113 sign = -1
114 text = text.drop(1)
115 }
116 if (text.isEmpty()) return null
117 val digits = mutableListOf<Int>()
118 for (c in text) {
119 val digit = if (c in '0'..'9') c - '0' else if (c in 'A'..'Z') c - 'A' + 10 else return null
120 if (digit >= radix) return null
121 digits.add(digit)
122 }
123 return sign * digits.fold(0L) {
124 value, digit -> value * radix + digit
125 }
126 }

154 val out = StringBuilder()

174 fun executeDefinition(definition: Word, start: Int, owner: String) {
175 if (depth == 100) fault(owner, "nesting too deep", -7)
176 depth++
177 val oldReturn = returns.size
178 val locals = LongArray(definition.locals)
179 var pc = start
180 try {
181 while (pc < definition.code.size) {
182 val i = definition.code[pc]
183 pc++
184 when (i.op) {
185 "literal" -> push(i.value, i.spelling)
186 "call" -> execute(i.word!!, i.spelling)
187 "local" -> push(locals[i.value.toInt()], i.spelling)
188 "local-set" -> locals[i.value.toInt()] = pop(1, i.spelling)[0]
189 "locals" -> {
190 val a = pop(i.slots.size, i.spelling)
191 for (j in i.slots.indices) locals[i.slots[j]] = a[j]
192 }
193 "value-set" -> memory[i.word!!.address / 8] = pop(1, i.spelling)[0]
194 "defer-set" -> i.word!!.deferred = tokenWord(pop(1, i.spelling)[0], i.spelling)
195 "jump" -> pc = i.target
196 "zero" -> if (pop(1, i.spelling)[0] == 0L) pc = i.target
197 "exit" -> throw Exit()
198 "does" -> {
199 val latest = users.lastOrNull()
200 if (latest == null || latest.kind != "create") fault(i.spelling, "no CREATE word", -31)
201 latest.does = definition
202 latest.doesStart = pc
203 throw Exit()
204 }
205 "abort-string" -> if (pop(1, i.spelling)[0] != 0L) fault(i.spelling, i.text, -2)
206 "write" -> pending += i.text
207 "do", "?do" -> {
208 val a = pop(2, i.spelling)
209 if (i.op == "?do" && a[0] == a[1]) pc = i.target else {
210 returnPush(a[0], i.spelling)
211 returnPush(a[1], i.spelling)
212 }
213 }
214 "loop", "+loop" -> {
215 val step = if (i.op == "loop") 1L else pop(1, i.spelling)[0]
216 returnNeed(2, i.spelling)
217 val index = returns.last()
218 val limit = returns[returns.lastIndex - 1]
219 val next = index + step
220 if ((index - limit < 0) != (next - limit < 0)) {
221 returns.removeAt(returns.lastIndex)
222 returns.removeAt(returns.lastIndex)
223 } else {
224 returns[returns.lastIndex] = next
225 pc = i.target
226 }
227 }
228 "leave" -> {
229 returnNeed(2, i.spelling)
230 repeat(2) {
231 returns.removeAt(returns.lastIndex)
232 }
233 pc = i.target
234 }
235 "of" -> {
236 val a = pop(2, i.spelling)
237 if (a[0] != a[1]) {
238 push(a[0], i.spelling)
239 pc = i.target
240 }
241 }
242 "endcase" -> pop(1, i.spelling)
243 ">R" -> returnPush(pop(1, i.spelling)[0], i.spelling)
244 "R>" -> {
245 returnNeed(1, i.spelling)
246 push(returns.removeAt(returns.lastIndex), i.spelling)
247 }
248 "R@", "I" -> {
249 returnNeed(1, i.spelling)
250 push(returns.last(), i.spelling)
251 }
252 "J" -> {
253 returnNeed(3, i.spelling)
254 push(returns[returns.size - 3], i.spelling)
255 }
256 "UNLOOP" -> {
257 returnNeed(2, i.spelling)
258 repeat(2) {
259 returns.removeAt(returns.lastIndex)
260 }
261 }
262 }
263 }
264 } catch (_: Exit) {
265 }
266 while (returns.size > oldReturn) returns.removeAt(returns.lastIndex)
267 depth--
268 }

270 fun execute(initial: Word, originalSpelling: String) {
271 var word = initial
272 var spelling = originalSpelling
273 while (word.kind == "defer") {
274 val target = word.deferred ?: fault(spelling, "deferred word not set", -21)
275 word = target
276 spelling = target.name
277 }
278 when (word.kind) {
279 "colon" -> executeDefinition(word, 0, spelling)
280 "create", "variable" -> {
281 push(word.address.toLong(), spelling)
282 word.does?.let {
283 executeDefinition(it, word.doesStart, spelling)
284 }
285 }
286 "constant" -> push(word.value, spelling)
287 "value" -> push(memory[word.address / 8], spelling)
288 else -> builtin(word.name, spelling)
289 }
290 }

292 fun builtin(name: String, owner: String) {
293 when (name) {
294 "+", "-", "*", "/", "MOD", "/MOD", "MIN", "MAX", "=", "<>", "<", ">", "AND", "OR", "XOR" -> {
295 val a = pop(2, owner)
296 val x = a[0]
297 val y = a[1]
298 if (name in setOf("/", "MOD", "/MOD") && y == 0L) fault(owner, "division by zero", -10)
299 val v = when (name) {
300 "+" -> x + y
301 "-" -> x - y
302 "*" -> x * y
303 "/", "/MOD" -> quotient(x, y)
304 "MOD" -> x - y * quotient(x, y)
305 "MIN" -> minOf(x, y)
306 "MAX" -> maxOf(x, y)
307 "=" -> if (x == y) -1L else 0L
308 "<>" -> if (x != y) -1L else 0L
309 "<" -> if (x < y) -1L else 0L
310 ">" -> if (x > y) -1L else 0L
311 "AND" -> x and y
312 "OR" -> x or y
313 else -> x xor y
314 }
315 if (name == "/MOD") push(x - y * v, owner)
316 push(v, owner)
317 }
318 "*/", "*/MOD" -> {
319 val a = pop(3, owner)
320 if (a[2] == 0L) fault(owner, "division by zero", -10)
321 val p = a[0] * a[1]
322 val q = quotient(p, a[2])
323 if (name == "*/MOD") push(p - a[2] * q, owner)
324 push(q, owner)
325 }
326 "NEGATE", "ABS", "1+", "1-", "2*", "2/", "0=", "0<>", "0<", "0>", "INVERT", "CELLS", "CELL+" -> {
327 val a = pop(1, owner)[0]
328 push(when (name) {
329 "NEGATE" -> -a
330 "ABS" -> kotlin.math.abs(a)
331 "1+" -> a + 1
332 "1-" -> a - 1
333 "2*" -> 2 * a
334 "2/" -> quotient(a, 2)
335 "0=" -> if (a == 0L) -1 else 0
336 "0<>" -> if (a != 0L) -1 else 0
337 "0<" -> if (a < 0) -1 else 0
338 "0>" -> if (a > 0) -1 else 0
339 "INVERT" -> a.inv()
340 "CELLS" -> 8 * a
341 else -> a + 8
342 }, owner)
343 }
344 "WITHIN" -> {
345 val a = pop(3, owner)
346 push(if (a[1] <= a[0] && a[0] < a[2]) -1 else 0, owner)
347 }
348 "TRUE" -> push(-1, owner)
349 "FALSE", "BASE" -> push(0, owner)
350 "BL" -> push(32, owner)
351 "DEPTH" -> push(data.size.toLong(), owner)
352 "HERE" -> push(memory.size * 8L, owner)
353 "DUP", "?DUP", "DROP", "SWAP", "OVER", "ROT", "-ROT", "NIP", "TUCK", "2DUP", "2DROP", "2SWAP", "2OVER" -> {
354 val count = when (name) {
355 "DUP", "?DUP", "DROP" -> 1
356 "ROT", "-ROT" -> 3
357 "2SWAP", "2OVER" -> 4
358 else -> 2
359 }
360 val a = pop(count, owner)
361 val values = when (name) {
362 "DUP" -> listOf(a[0], a[0])
363 "?DUP" -> if (a[0] == 0L) a else listOf(a[0], a[0])
364 "DROP", "2DROP" -> emptyList()
365 "SWAP" -> listOf(a[1], a[0])
366 "OVER" -> a + a[0]
367 "ROT" -> listOf(a[1], a[2], a[0])
368 "-ROT" -> listOf(a[2], a[0], a[1])
369 "NIP" -> listOf(a[1])
370 "TUCK" -> listOf(a[1], a[0], a[1])
371 "2DUP" -> a + a
372 "2SWAP" -> listOf(a[2], a[3], a[0], a[1])
373 else -> a + a.take(2)
374 }
375 pushAll(values, owner)
376 }
377 "PICK", "ROLL" -> {
378 val u = pop(1, owner)[0]
379 if (u < 0) fault(owner, "invalid argument", -24)
380 if (u >= data.size) fault(owner, "stack underflow", -4)
381 val i = data.size - 1 - u.toInt()
382 val v = if (name == "ROLL") data.removeAt(i) else data[i]
383 push(v, owner)
384 }
385 "." -> {
386 val n = pop(1, owner)[0]
387 pending += formatted(n, radix(owner)) + " "
388 }
389 ".R" -> {
390 val a = pop(2, owner)
391 val s = formatted(a[0], radix(owner))
392 if (a[1] > s.length) pending += " ".repeat((a[1] - s.length).toInt())
393 pending += s
394 }
395 ".S" -> {
396 val b = radix(owner)
397 pending += "<${formatted(data.size.toLong(), b)}> " + data.joinToString("") {
398 formatted(it, b) + " "
399 }
400 }
401 "?" -> {
402 val i = address(pop(1, owner)[0], owner)
403 pending += formatted(memory[i], radix(owner)) + " "
404 }
405 "CR" -> flush()
406 "EMIT" -> {
407 val c = pop(1, owner)[0]
408 if (c !in 32L..126L) fault(owner, "invalid argument", -24)
409 pending += c.toInt().toChar()
410 }
411 "SPACE" -> pending += " "
412 "SPACES" -> {
413 val n = pop(1, owner)[0]
414 if (n > 0) pending += " ".repeat(n.toInt())
415 }
416 "!", "+!" -> {
417 val a = pop(2, owner)
418 val i = address(a[1], owner)
419 memory[i] = if (name == "!") a[0] else memory[i] + a[0]
420 }
421 "@" -> {
422 val i = address(pop(1, owner)[0], owner)
423 push(memory[i], owner)
424 }
425 "," -> memory.add(pop(1, owner)[0])
426 "ALLOT" -> {
427 val n = pop(1, owner)[0]
428 if (n < 0 || n % 8 != 0L) fault(owner, "invalid argument", -24)
429 repeat((n / 8).toInt()) {
430 memory.add(0)
431 }
432 }
433 "HEX" -> memory[0] = 16
434 "DECIMAL" -> memory[0] = 10
435 "CREATE", "VARIABLE", "CONSTANT", "VALUE", "DEFER" -> {
436 val raw = readName(owner)
437 checkName(raw)
438 val value = if (name in setOf("CONSTANT", "VALUE")) pop(1, owner)[0] else 0L
439 val word = Word(raw.uppercase(), raw, name.lowercase(), memory.size * 8)
440 word.address = memory.size * 8
441 word.value = value
442 users.add(word)
443 if (name == "VARIABLE" || name == "VALUE") memory.add(value)
444 }
445 "'" -> {
446 val raw = readName(owner)
447 val word = find(raw) ?: fault(raw, "undefined word", -13)
448 push(token(word), owner)
449 }
450 "EXECUTE" -> {
451 val word = tokenWord(pop(1, owner)[0], owner)
452 execute(word, word.name)
453 }
454 "CATCH" -> {
455 val word = tokenWord(pop(1, owner)[0], owner)
456 val saved = data.toList()
457 val size = returns.size
458 val level = depth
459 var code = 0L
460 try {
461 execute(word, word.name)
462 } catch (e: Fault) {
463 if (e.code == null) throw e
464 data.clear()
465 data.addAll(saved)
466 while (returns.size > size) returns.removeAt(returns.lastIndex)
467 depth = level
468 code = e.code
469 }
470 push(code, owner)
471 }
472 "THROW" -> {
473 val n = pop(1, owner)[0]
474 if (n != 0L) fault(owner, "exception $n", n)
475 }
476 "ABORT" -> fault(owner, "aborted", -1)
477 "WORDS" -> for (w in users.asReversed()) pending += w.name + " "
478 }
479 }

496 fun process(raw: String) {
497 val name = raw.uppercase()
498 val compiling = compiler
499 if (name == "\\") {
500 at = line.length
501 return
502 }
503 if (name == "(") {
504 val end = line.indexOf(')', at)
505 if (end < 0) fault(raw, "unterminated comment")
506 at = end + 1
507 return
508 }
509 if (compiling == null && name in compileOnly) fault(raw, "compile only")
510 if (compiling != null && name in interpretOnly) fault(raw, "interpret only")
511 if (name == ".\"" || name == "ABORT\"") {
512 val text = string(raw)
513 if (compiling == null) pending += text else compiling.word.code.add(Instruction(if (name == ".\"") "write" else "abort-string", raw, text = text))
514 return
515 }
516 if (name == "TO" || name == "IS") {
517 val n = readName(raw)
518 val upper = n.uppercase()
519 if (name == "TO" && compiling != null && upper in compiling.locals) {
520 compiling.word.code.add(Instruction("local-set", raw, compiling.locals.getValue(upper).toLong()))
521 return
522 }
523 val w = target(n, if (name == "TO") "value" else "defer")
524 if (compiling != null) compiling.word.code.add(Instruction(if (name == "TO") "value-set" else "defer-set", raw, word = w))
525 else if (name == "TO") memory[w.address / 8] = pop(1, raw)[0] else w.deferred = tokenWord(pop(1, raw)[0], raw)
526 return
527 }
528 if (name == ":" || name == ":NONAME") {
529 val n = if (name == ":") readName(raw).also(::checkName) else raw
530 compiler = Compiler(Word(if (name == ":") n.uppercase() else ":NONAME", n, "colon", memory.size * 8), name == ":NONAME")
531 return
532 }
533 if (name == "FORGET") {
534 val n = readName(raw)
535 val i = users.indexOfLast {
536 it.name == n.uppercase()
537 }
538 if (i < 0) fault(n, "undefined word", -13)
539 val here = users[i].here / 8
540 while (users.size > i) users.removeAt(users.lastIndex)
541 while (memory.size > here) memory.removeAt(memory.lastIndex)
542 return
543 }
544 if (compiling != null) {
545 compile(raw, name, compiling)
546 return
547 }
548 val w = find(raw)
549 if (w != null) execute(w, raw) else {
550 val n = parseNumber(raw, raw) ?: fault(raw, "undefined word", -13)
551 push(n, raw)
552 }
553 }

558 fun mismatch(): Nothing = fault(raw, "control structure mismatch")

560 fun top(vararg kinds: String): Control {
561 val t = c.control.lastOrNull() ?: mismatch()
562 if (t.kind !in kinds) mismatch()
563 return t

566 fun append(op: String, value: Long = 0, target: Int = 0): Int {
567 code.add(Instruction(op, raw, value, target = target))
568 return code.lastIndex
569 }
570 when (name) {
571 ";" -> {
572 if (c.control.isNotEmpty()) {
573 val k = c.control.last().kind
574 fault(raw, "unclosed " + when (k) {
575 "ELSE" -> "IF"
576 "WHILE" -> "BEGIN"
577 else -> k
578 })
579 }
580 compiler = null
581 if (c.anonymous) push(token(c.word), raw) else users.add(c.word)
582 }
583 "IF" -> c.control.add(Control("IF", append("zero")))
584 "ELSE" -> {
585 val t = top("IF")
586 c.control.removeAt(c.control.lastIndex)
587 val i = append("jump")
588 code[t.index].target = code.size
589 c.control.add(Control("ELSE", i))
590 }
591 "THEN" -> {
592 val t = top("IF", "ELSE")
593 c.control.removeAt(c.control.lastIndex)
594 code[t.index].target = code.size
595 }
596 "BEGIN" -> c.control.add(Control("BEGIN", code.size))
597 "UNTIL", "AGAIN" -> {
598 val t = top("BEGIN")
599 c.control.removeAt(c.control.lastIndex)
600 append(if (name == "UNTIL") "zero" else "jump", target = t.index)
601 }
602 "WHILE" -> {
603 val t = top("BEGIN")
604 c.control.removeAt(c.control.lastIndex)
605 c.control.add(Control("WHILE", append("zero"), t.index))
606 }
607 "REPEAT" -> {
608 val t = top("WHILE")
609 c.control.removeAt(c.control.lastIndex)
610 append("jump", target = t.begin)
611 code[t.index].target = code.size
612 }
613 "DO", "?DO" -> c.control.add(Control(name, append(if (name == "DO") "do" else "?do")))
614 "LOOP", "+LOOP" -> {
615 val t = top("DO", "?DO")
616 c.control.removeAt(c.control.lastIndex)
617 append(if (name == "LOOP") "loop" else "+loop", target = t.index + 1)
618 code[t.index].target = code.size
619 for (i in t.leaves) code[i].target = code.size
620 }
621 "LEAVE" -> {
622 val t = c.control.lastOrNull {
623 it.kind == "DO" || it.kind == "?DO"
624 } ?: mismatch()
625 t.leaves.add(append("leave"))
626 }
627 "CASE" -> c.control.add(Control("CASE", code.size))
628 "OF" -> {
629 top("CASE")
630 c.control.add(Control("OF", append("of")))
631 }
632 "ENDOF" -> {
633 val t = top("OF")
634 c.control.removeAt(c.control.lastIndex)
635 val i = append("jump")
636 code[t.index].target = code.size
637 top("CASE").ends.add(i)
638 }
639 "ENDCASE" -> {
640 val t = top("CASE")
641 c.control.removeAt(c.control.lastIndex)
642 append("endcase")
643 for (i in t.ends) code[i].target = code.size
644 }
645 "EXIT" -> append("exit")
646 "DOES>" -> {
647 if (c.control.isNotEmpty()) mismatch()
648 append("does")
649 }
650 "RECURSE" -> code.add(Instruction("call", raw, word = c.word))
651 "[']" -> {
652 val n = readName(raw)
653 val word = find(n) ?: fault(n, "undefined word", -13)
654 append("literal", token(word))
655 }
656 "{:" -> {
657 val names = mutableListOf<String>()
658 var closed = false
659 while (true) {
660 val n = readWord() ?: break
661 if (n.uppercase() == ":}") {
662 closed = true
663 break
664 }
665 names.add(n)
666 }
667 if (!closed) fault(raw, "unterminated locals")
668 val slots = mutableListOf<Int>()
669 for (n in names.takeWhile {
670 it != "--"
671 }) {
672 checkName(n)
673 val slot = c.word.locals++
674 slots.add(slot)
675 c.locals[n.uppercase()] = slot
676 }
677 code.add(Instruction("locals", raw, slots = slots))
678 }
679 ">R", "R>", "R@", "I", "J", "UNLOOP" -> append(name)
680 else -> if (name in c.locals) append("local", c.locals.getValue(name).toLong()) else {
681 val word = find(raw)
682 if (word != null) code.add(Instruction("call", raw, word = word)) else append("literal", parseNumber(raw, raw) ?: fault(raw, "undefined word", -13))
683 }
684 }
685 }

687 fun input(text: String) {
688 line = text
689 at = 0
690 try {
691 while (true) process(readWord() ?: break)
692 } catch (e: Fault) {
693 if (pending.isNotEmpty()) flush()
694 output.add("ERROR ${e.word}: ${e.reason}")
695 data.clear()
696 returns.clear()
697 compiler = null
698 depth = 0
699 }
700 if (pending.isNotEmpty()) flush()
701 }
702}
