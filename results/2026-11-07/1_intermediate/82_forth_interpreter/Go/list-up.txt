func (m *machine) take(count int, raw string) []int64 {
	if len(m.data) < count {
		problem(raw, "stack underflow", -4)
	}
	values := append([]int64(nil), m.data[len(m.data)-count:]...)
	m.data = m.data[:len(m.data)-count]
	return values
