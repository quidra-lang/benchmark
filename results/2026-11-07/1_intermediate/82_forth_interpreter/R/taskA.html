52 number <- function(text, base = state$memory[[1]]) {
53 if (nchar(text) == 3L && substr(text, 1L, 1L) == "'" && substr(text, 3L, 3L) == "'") return(utf8ToInt(substr(text, 2L, 2L)))
54 first <- substr(text, 1L, 1L)
55 if (first %in% c("$", "#", "%")) {
56 base <- c("$" = 16, "#" = 10, "%" = 2)[[first]]
57 text <- substring(text, 2L)
58 } else if (base < 2 || base > 36) {
59 fail(text, "invalid base")
60 }
61 negative <- startsWith(text, "-")
62 digits <- upper(if (negative) substring(text, 2L) else text)
63 if (digits == "") return(NULL)
64 chars <- utf8ToInt(digits)
65 values <- ifelse(chars >= 48L & chars <= 57L, chars - 48L, ifelse(chars >= 65L & chars <= 90L, chars - 55L, -1L))
66 if (any(values < 0L | values >= base)) return(NULL)
67 result <- 0
68 for (d in values) result <- result * base + d
69 if (negative) -result else result
70 }

117 divmod <- function(a, b) {
118 x <- abs(a)
119 y <- abs(b)
120 q <- floor(x / y)
121 r <- x - q * y
122 if (r < 0) {
123 q <- q - 1
124 r <- r + y
125 } else if (r >= y) {
126 q <- q + 1
127 r <- r - y
128 }
129 if ((a < 0) != (b < 0)) {
130 q <- -q
131 if (r != 0) {
132 q <- q - 1
133 r <- y - r
134 }
135 }
136 c(q, if (b < 0) -r else r)
137 }

177 compile <- function(source) {
178 key <- upper(source)
179 if (key == "IF") {
180 record("IF", at = emit("zero_jump", source, NULL))
181 } else if (key == "ELSE") {
182 mismatch(source, "IF")
183 old <- pop_control()
184 index <- emit("jump", source, NULL)
185 patch(old$at, length(state$compiling$code) + 1L)
186 record("ELSE", at = index)
187 } else if (key == "THEN") {
188 mismatch(source, c("IF", "ELSE"))
189 patch(pop_control()$at, length(state$compiling$code) + 1L)
190 } else if (key == "BEGIN") {
191 record("BEGIN", begin = length(state$compiling$code) + 1L)
192 } else if (key %in% c("UNTIL", "AGAIN")) {
193 mismatch(source, "BEGIN")
194 item <- pop_control()
195 emit(if (key == "UNTIL") "zero_jump" else "jump", source, item$begin)
196 } else if (key == "WHILE") {
197 mismatch(source, "BEGIN")
198 item <- pop_control()
199 record("WHILE", begin = item$begin, at = emit("zero_jump", source, NULL))
200 } else if (key == "REPEAT") {
201 mismatch(source, "WHILE")
202 item <- pop_control()
203 emit("jump", source, item$begin)
204 patch(item$at, length(state$compiling$code) + 1L)
205 } else if (key %in% c("DO", "?DO")) {
206 record(key, at = emit("do", source, NULL, key == "?DO"), leaves = integer())
207 } else if (key %in% c("LOOP", "+LOOP")) {
208 mismatch(source, c("DO", "?DO"))
209 item <- pop_control()
210 emit("loop", source, item$at + 1L, key == "+LOOP")
211 finish <- length(state$compiling$code) + 1L
212 patch(item$at, finish)
213 for (index in item$leaves) patch(index, finish)
214 } else if (key == "LEAVE") {
215 found <- NULL
216 for (item in rev(state$control)) {
217 if (item$kind %in% c("DO", "?DO")) {
218 found <- item
219 break
220 }
221 }
222 if (is.null(found)) fail(source, "control structure mismatch")
223 found$leaves <- c(found$leaves, emit("leave", source, NULL))
224 } else if (key == "CASE") {
225 record("CASE", ends = integer())
226 } else if (key == "OF") {
227 mismatch(source, "CASE")
228 record("OF", at = emit("of", source, NULL))
229 } else if (key == "ENDOF") {
230 mismatch(source, "OF")
231 item <- pop_control()
232 case <- top()
233 case$ends <- c(case$ends, emit("jump", source, NULL))
234 patch(item$at, length(state$compiling$code) + 1L)
235 } else if (key == "ENDCASE") {
236 mismatch(source, "CASE")
237 item <- pop_control()
238 emit("drop", source)
239 for (index in item$ends) patch(index, length(state$compiling$code) + 1L)
240 } else if (key == "EXIT") {
241 emit("exit", source)
242 } else if (key == "RECURSE") {
243 emit("call", source, state$compiling)
244 } else if (key == "DOES>") {
245 if (length(state$control) != 0L) fail(source, "control structure mismatch")
246 emit("does", source, state$compiling, length(state$compiling$code) + 2L)
247 } else if (key == ";") {
248 if (length(state$control) != 0L) {
249 kind <- top()$kind
250 if (kind == "ELSE") kind <- "IF"
251 if (kind == "WHILE") kind <- "BEGIN"
252 fail(source, paste0("unclosed ", kind))
253 }
254 w <- state$compiling
255 state$compiling <- NULL
256 if (state$anonymous) push(token(w), source) else state$dictionary[[length(state$dictionary) + 1L]] <- w
257 } else if (key == "[']") {
258 name <- read_name(source)
259 w <- lookup(name)
260 if (is.null(w)) fail(name, "undefined word")
261 emit("literal", source, token(w))
262 } else if (key == "{:") {
263 names <- character()
264 repeat {
265 name <- read_word()
266 if (is.null(name)) fail(source, "unterminated locals")
267 if (upper(name) == ":}") break
268 names <- c(names, name)
269 }
270 slots <- integer()
271 for (name in names) {
272 if (name == "--") break
273 key_name <- check_name(name)
274 state$compiling$local_count <- state$compiling$local_count + 1L
275 slot <- state$compiling$local_count
276 assign(key_name, slot, envir = state$locals)
277 slots <- c(slots, slot)
278 }
279 emit("locals", source, slots)
280 } else if (key %in% c(">R", "R>", "R@", "I", "J", "UNLOOP")) {
281 emit("return_stack", source, key)
282 } else if (exists(key, envir = state$locals, inherits = FALSE)) {
283 emit("local", source, get(key, envir = state$locals, inherits = FALSE))
284 } else {
285 w <- lookup(source)
286 if (!is.null(w)) {
287 emit("call", source, w)
288 } else {
289 value <- number(source)
290 if (is.null(value)) fail(source, "undefined word")
291 emit("literal", source, value)
292 }
293 }

304 process <- function(source) {
305 key <- upper(source)
306 if (key == "\\") {
307 state$position <- nchar(state$line) + 1L
308 return(invisible(NULL))
309 }
310 if (key == "(") {
311 remainder <- substring(state$line, state$position)
312 finish <- regexpr(")", remainder, fixed = TRUE)[[1]]
313 if (finish < 0L) fail(source, "unterminated comment")
314 state$position <- state$position + finish
315 return(invisible(NULL))
316 }
317 if (is.null(state$compiling) && key %in% compile_only) fail(source, "compile only")
318 if (!is.null(state$compiling) && key %in% interpret_only) fail(source, "interpret only")
319 if (key %in% c('."', 'ABORT"')) {
320 start <- state$position + 1L
321 finish <- regexpr('"', substring(state$line, start), fixed = TRUE)[[1]]
322 if (state$position > nchar(state$line) || finish < 0L) fail(source, "unterminated string")
323 finish <- start + finish - 1L
324 text <- substr(state$line, start, finish - 1L)
325 state$position <- finish + 1L
326 if (is.null(state$compiling)) write(text) else emit(if (key == '."') "write" else "abort_string", source, text)
327 } else if (key %in% c("TO", "IS")) {
328 name <- read_name(source)
329 if (key == "TO" && !is.null(state$compiling) && exists(upper(name), envir = state$locals, inherits = FALSE)) {
330 emit("set_local", source, get(upper(name), envir = state$locals, inherits = FALSE))
331 return(invisible(NULL))
332 }
333 w <- lookup(name)
334 if (is.null(w)) fail(name, "undefined word")
335 expected <- if (key == "TO") "value" else "defer"
336 if (w$kind != expected) fail(name, if (key == "TO") "not a value" else "not deferred")
337 if (is.null(state$compiling)) assign_word(w, source, key == "IS") else emit(if (key == "TO") "to" else "is", source, w)
338 } else if (key %in% c(":", ":NONAME")) {
339 name <- if (key == ":") read_name(source) else source
340 if (key == ":") check_name(name)
341 state$compiling <- word(upper(name), "colon", here = length(state$memory) * 8)
342 state$definition_name <- name
343 state$anonymous <- key == ":NONAME"
344 state$control <- list()
345 state$locals <- new.env(parent = emptyenv())
346 } else if (key == "FORGET") {
347 name <- read_name(source)
348 index <- NULL
349 for (i in rev(seq_along(state$dictionary))) {
350 if (state$dictionary[[i]]$name == upper(name)) {
351 index <- i
352 break
353 }
354 }
355 if (is.null(index)) fail(name, "undefined word")
356 state$memory <- head(state$memory, state$dictionary[[index]]$here / 8)
357 state$dictionary <- head(state$dictionary, index - 1L)
358 } else if (!is.null(state$compiling)) {
359 compile(source)
360 } else {
361 w <- lookup(source)
362 if (!is.null(w)) {
363 execute(w, source)
364 } else {
365 value <- number(source)
366 if (is.null(value)) fail(source, "undefined word")
367 push(value, source)
368 }
369 }
370 }
371 return_operation <- function(key, source) {
372 if (key == ">R") {
373 return_push(take(1L, source)[[1]], source)
374 } else if (key == "R>") {

376 value <- tail(state$returns, 1L)
377 state$returns <- head(state$returns, -1L)
378 push(value, source)
379 } else if (key %in% c("R@", "I")) {
380 return_need(1L, source)
381 push(tail(state$returns, 1L), source)
382 } else if (key == "J") {
383 return_need(3L, source)

388 }
389 }
390 execute_definition <- function(w, start, source) {
391 if (state$depth == 100L) fail(source, "nesting too deep")
392 state$depth <- state$depth + 1L
393 return_size <- length(state$returns)
394 locals <- numeric(w$local_count)
395 pc <- start
396 while (pc <= length(w$code)) {
397 instruction <- w$code[[pc]]
398 pc <- pc + 1L
399 op <- instruction$op
400 spelling <- instruction$source
401 args <- instruction$args
402 a <- if (length(args) >= 1L) args[[1]] else NULL
403 b <- if (length(args) >= 2L) args[[2]] else NULL
404 if (op == "literal") {
405 push(a, spelling)
406 } else if (op == "call") {
407 execute(a, spelling)
408 } else if (op == "write") {
409 write(a)
410 } else if (op == "local") {
411 push(locals[[a]], spelling)
412 } else if (op == "set_local") {
413 locals[[a]] <- take(1L, spelling)[[1]]
414 } else if (op == "locals") {
415 values <- take(length(a), spelling)
416 locals[a] <- values
417 } else if (op %in% c("to", "is")) {
418 assign_word(a, spelling, op == "is")
419 } else if (op == "exit") {
420 break
421 } else if (op == "does") {
422 created <- if (length(state$dictionary) == 0L) NULL else tail(state$dictionary, 1L)[[1]]
423 if (is.null(created) || created$kind != "create") fail(spelling, "no CREATE word")
424 created$does_word <- a
425 created$does_start <- b
426 break
427 } else if (op == "jump") {
428 pc <- a
429 } else if (op == "zero_jump") {
430 if (take(1L, spelling)[[1]] == 0) pc <- a
431 } else if (op == "do") {
432 values <- take(2L, spelling)
433 if (b && values[[1]] == values[[2]]) {
434 pc <- a
435 } else {
436 return_push(values[[1]], spelling)
437 return_push(values[[2]], spelling)
438 }
439 } else if (op == "loop") {
440 step <- if (b) take(1L, spelling)[[1]] else 1
441 return_need(2L, spelling)
442 last <- length(state$returns)
443 index <- state$returns[[last]]
444 limit <- state$returns[[last - 1L]]
445 next_index <- index + step
446 if ((index - limit < 0) != (next_index - limit < 0)) {
447 state$returns <- head(state$returns, -2L)
448 } else {
449 state$returns[[last]] <- next_index
450 pc <- a
451 }
452 } else if (op == "leave") {
453 return_need(2L, spelling)
454 state$returns <- head(state$returns, -2L)
455 pc <- a
456 } else if (op == "of") {
457 values <- take(2L, spelling)
458 if (values[[1]] != values[[2]]) {
459 push(values[[1]], spelling)
460 pc <- a
461 }
462 } else if (op == "drop") {
463 take(1L, spelling)
464 } else if (op == "return_stack") {
465 return_operation(a, spelling)
466 } else if (op == "abort_string") {
467 if (take(1L, spelling)[[1]] != 0) fail(spelling, a, -2)
468 }
469 }
470 if (length(state$returns) > return_size) state$returns <- head(state$returns, return_size)
471 state$depth <- state$depth - 1L
472 }

478 execute <- function(w, source) {
479 while (w$kind == "defer") {
480 if (is.null(w$value)) fail(source, "deferred word not set")
481 if (!valid_token(w$value)) fail(source, "invalid argument")
482 w <- state$tokens[[w$value]]

485 if (w$kind == "colon") {

487 } else if (w$kind %in% c("create", "variable")) {
488 push(w$value, source)
489 if (!is.null(w$does_word)) execute_definition(w$does_word, w$does_start, source)
490 } else if (w$kind == "constant") {
491 push(w$value, source)
492 } else if (w$kind == "value") {
493 push(state$memory[[w$value / 8 + 1L]], source)
494 } else {
495 builtin(w$name, source)
496 }

498 builtin <- function(key, source) {
499 binary <- c("+", "-", "*", "/", "MOD", "/MOD", "MIN", "MAX", "=", "<>", "<", ">", "AND", "OR", "XOR")
500 unary <- c("NEGATE", "ABS", "1+", "1-", "2*", "2/", "0=", "0<>", "0<", "0>", "INVERT", "CELLS", "CELL+")
501 stack_counts <- c(DUP = 1L, "?DUP" = 1L, DROP = 1L, SWAP = 2L, OVER = 2L, ROT = 3L, "-ROT" = 3L, NIP = 2L, TUCK = 2L, "2DUP" = 2L, "2DROP" = 2L, "2SWAP" = 4L, "2OVER" = 4L)
502 if (key %in% binary) {
503 values <- take(2L, source)
504 a <- values[[1]]
505 b <- values[[2]]
506 if (key %in% c("/", "MOD", "/MOD")) {
507 if (b == 0) fail(source, "division by zero")
508 qr <- divmod(a, b)
509 push_all(if (key == "/") qr[[1]] else if (key == "MOD") qr[[2]] else rev(qr), source)
510 } else {
511 result <- switch(key, "+" = a + b, "-" = a - b, "*" = a * b, MIN = min(a, b), MAX = max(a, b), "=" = if (a == b) -1 else 0, "<>" = if (a != b) -1 else 0, "<" = if (a < b) -1 else 0, ">" = if (a > b) -1 else 0, AND = bitwAnd(as.integer(a), as.integer(b)), OR = bitwOr(as.integer(a), as.integer(b)), XOR = bitwXor(as.integer(a), as.integer(b)))
512 push(result, source)
513 }
514 } else if (key %in% unary) {
515 a <- take(1L, source)[[1]]
516 result <- switch(key, NEGATE = -a, ABS = abs(a), "1+" = a + 1, "1-" = a - 1, "2*" = 2 * a, "2/" = divmod(a, 2)[[1]], "0=" = if (a == 0) -1 else 0, "0<>" = if (a != 0) -1 else 0, "0<" = if (a < 0) -1 else 0, "0>" = if (a > 0) -1 else 0, INVERT = -a - 1, CELLS = a * 8, "CELL+" = a + 8)
517 push(result, source)
518 } else if (key %in% names(stack_counts)) {
519 values <- take(stack_counts[[key]], source)
520 indices <- switch(key, DUP = c(1L, 1L), "?DUP" = if (values[[1]] == 0) 1L else c(1L, 1L), DROP = integer(), SWAP = c(2L, 1L), OVER = c(1L, 2L, 1L), ROT = c(2L, 3L, 1L), "-ROT" = c(3L, 1L, 2L), NIP = 2L, TUCK = c(2L, 1L, 2L), "2DUP" = c(1L, 2L, 1L, 2L), "2DROP" = integer(), "2SWAP" = c(3L, 4L, 1L, 2L), "2OVER" = c(1L, 2L, 3L, 4L, 1L, 2L))
521 push_all(values[indices], source)
522 } else if (key %in% c("*/", "*/MOD")) {
523 values <- take(3L, source)
524 if (values[[3]] == 0) fail(source, "division by zero")
525 qr <- divmod(values[[1]] * values[[2]], values[[3]])
526 push_all(if (key == "*/") qr[[1]] else rev(qr), source)
527 } else if (key == "WITHIN") {
528 values <- take(3L, source)
529 push(if (values[[2]] <= values[[1]] && values[[1]] < values[[3]]) -1 else 0, source)
530 } else if (key %in% c("TRUE", "FALSE", "BL", "BASE", "HERE")) {
531 value <- switch(key, "TRUE" = -1, "FALSE" = 0, BL = 32, BASE = 0, HERE = length(state$memory) * 8)
532 push(value, source)
533 } else if (key == "DEPTH") {
534 push(length(state$stack), source)
535 } else if (key %in% c("PICK", "ROLL")) {
536 u <- take(1L, source)[[1]]
537 if (u < 0) fail(source, "invalid argument")
538 if (length(state$stack) <= u) fail(source, "stack underflow")
539 index <- length(state$stack) - u
540 value <- state$stack[[index]]
541 if (key == "ROLL") state$stack <- state$stack[-index]
542 push(value, source)
543 } else if (key %in% c(".", ".R", "?")) {
544 values <- take(if (key == ".R") 2L else 1L, source)
545 n <- if (key == "?") state$memory[[address(values[[1]], source)]] else values[[1]]
546 text <- output_number(n, source)
547 write(if (key == ".R") paste0(strrep(" ", max(0, values[[2]] - nchar(text))), text) else paste0(text, " "))
548 } else if (key == ".S") {
549 output_number(0, source)
550 write(paste0("<", output_number(length(state$stack), source), "> "))
551 for (value in state$stack) write(paste0(output_number(value, source), " "))
552 } else if (key == "CR") {
553 flush()
554 } else if (key == "EMIT") {
555 c <- take(1L, source)[[1]]
556 if (c < 32 || c > 126) fail(source, "invalid argument")
557 write(intToUtf8(c))
558 } else if (key == "SPACE") {
559 write(" ")
560 } else if (key == "SPACES") {
561 write(strrep(" ", max(0, take(1L, source)[[1]])))
562 } else if (key %in% c("!", "+!")) {
563 values <- take(2L, source)
564 index <- address(values[[2]], source)
565 state$memory[[index]] <- if (key == "!") values[[1]] else state$memory[[index]] + values[[1]]
566 } else if (key == "@") {
567 push(state$memory[[address(take(1L, source)[[1]], source)]], source)
568 } else if (key == ",") {
569 state$memory <- c(state$memory, take(1L, source)[[1]])
570 } else if (key == "ALLOT") {
571 n <- take(1L, source)[[1]]
572 if (n < 0 || n %% 8 != 0) fail(source, "invalid argument")
573 state$memory <- c(state$memory, numeric(n / 8))
574 } else if (key %in% c("HEX", "DECIMAL")) {
575 state$memory[[1]] <- if (key == "HEX") 16 else 10
576 } else if (key %in% c("CREATE", "VARIABLE", "CONSTANT", "VALUE", "DEFER")) {
577 name <- check_name(read_name(source))
578 value <- if (key %in% c("CONSTANT", "VALUE")) take(1L, source)[[1]] else NULL
579 here <- length(state$memory) * 8
580 stored <- if (key %in% c("CREATE", "VARIABLE", "VALUE")) here else value
581 state$dictionary[[length(state$dictionary) + 1L]] <- word(name, tolower(key), stored, here)
582 if (key %in% c("VARIABLE", "VALUE")) state$memory <- c(state$memory, if (key == "VARIABLE") 0 else value)
583 } else if (key == "'") {
584 name <- read_name(source)
585 w <- lookup(name)
586 if (is.null(w)) fail(name, "undefined word")
587 push(token(w), source)
588 } else if (key == "EXECUTE") {
589 execute_token(take(1L, source)[[1]], source)
590 } else if (key == "CATCH") {
591 xt <- take(1L, source)[[1]]
592 if (!valid_token(xt)) fail(source, "invalid argument")
593 saved <- state$stack
594 saved_returns <- length(state$returns)
595 saved_depth <- state$depth
596 code <- tryCatch({
597 execute_token(xt, source)
598 0
599 }, forth_error = function(exception) {
600 if (is.null(exception$code)) stop(exception)
601 state$stack <- saved
602 if (length(state$returns) > saved_returns) state$returns <- head(state$returns, saved_returns)
603 state$depth <- saved_depth
604 exception$code
605 })
606 push(code, source)
607 } else if (key == "THROW") {
608 n <- take(1L, source)[[1]]
609 if (n != 0) fail(source, paste0("exception ", sprintf("%.0f", n)), n)
610 } else if (key == "ABORT") {
611 fail(source, "aborted")
612 } else if (key == "WORDS") {
613 for (w in rev(state$dictionary)) write(paste0(w$name, " "))
614 }
615 }
616 input <- file("stdin", open = "r")
617 on.exit(close(input))
618 repeat {
619 lines <- readLines(input, n = 1L, warn = FALSE)
620 if (length(lines) == 0L) break
621 state$line <- lines[[1]]
622 if (state$line == "BYE") {
623 if (!is.null(state$compiling)) cat("ERROR ", state$definition_name, ": unterminated definition\n", sep = "")
624 break
625 }
626 state$position <- 1L
627 tryCatch({
628 repeat {
629 source <- read_word()
630 if (is.null(source)) break
631 process(source)
632 }
633 }, forth_error = function(exception) {
634 if (nchar(state$pending) > 0L) flush()
635 cat("ERROR ", exception$word, ": ", exception$reason, "\n", sep = "")
636 state$stack <- numeric()
637 state$returns <- numeric()
638 state$depth <- 0L
639 state$compiling <- NULL
640 })
641 if (nchar(state$pending) > 0L) flush()
642 }
643}

645forth()
