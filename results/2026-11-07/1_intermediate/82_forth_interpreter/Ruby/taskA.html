29class Forth
30 COMPILE_ONLY = %w[IF ELSE THEN DO ?DO LOOP +LOOP LEAVE UNLOOP I J BEGIN
31 UNTIL WHILE REPEAT AGAIN CASE OF ENDOF ENDCASE EXIT RECURSE
32 DOES> ; >R R> R@ ABORT" ['] {:].freeze
33 INTERPRET_ONLY = %w[: :NONAME FORGET].freeze
34 SPECIAL = (COMPILE_ONLY + INTERPRET_ONLY + ['."', '(', '\\', 'TO', 'IS']).freeze
35 BUILTINS = %w[+ - * / MOD /MOD */ */MOD NEGATE ABS MIN MAX 1+ 1- 2* 2/
36 = <> < > 0= 0<> 0< 0> WITHIN TRUE FALSE AND OR XOR INVERT
37 DUP ?DUP DROP SWAP OVER ROT -ROT NIP TUCK 2DUP 2DROP 2SWAP
38 2OVER DEPTH PICK ROLL . .R .S ? CR EMIT SPACE SPACES BL ! @
39 +! , ALLOT HERE CELLS CELL+ BASE HEX DECIMAL CREATE VARIABLE
40 CONSTANT VALUE DEFER ' EXECUTE CATCH THROW ABORT WORDS].freeze
41 ERROR_CODES = {
42 'aborted' => -1, 'stack overflow' => -3, 'stack underflow' => -4,
43 'return stack overflow' => -5, 'return stack underflow' => -6,
44 'nesting too deep' => -7, 'invalid address' => -9, 'division by zero' => -10,
45 'undefined word' => -13, 'missing name' => -16, 'invalid name' => -19,
46 'deferred word not set' => -21, 'invalid argument' => -24,
47 'no CREATE word' => -31, 'invalid base' => -40
48 }.freeze

67 @position += 1 while @position < @line.length && @line[@position] == ' '

72 @line[start...@position]

79 def number(text, base = @memory[0])
80 return text[1].ord if text.length == 3 && text[0] == "'" && text[2] == "'"
81
82 if %w[$ # %].include?(text[0])
83 base = { '$' => 16, '#' => 10, '%' => 2 }[text[0]]
84 text = text[1..]
85 elsif !(2..36).cover?(base)
86 error(text, 'invalid base')
87 end
88 sign = text.start_with?('-') ? -1 : 1
89 digits = sign == -1 ? text[1..] : text
90 return nil if digits.empty?
91
92 result = 0
93 digits.upcase.each_byte do |c|
94 d = if (48..57).cover?(c)
95 c - 48
96 elsif (65..90).cover?(c)
97 c - 55
98 else
99 return nil
100 end
101 return nil if d >= base
102
103 result = result * base + d
104 end
105 sign * result
106 end

178 def compile(source)
179 key = source.upcase
180 case key
181 when 'IF'
182 @control << { kind: 'IF', at: emit(:zero_jump, source, nil) }
183 when 'ELSE'
184 mismatch(source, ['IF'])
185 old = @control.pop
186 index = emit(:jump, source, nil)
187 @compiling.code[old[:at]][2] = @compiling.code.length
188 @control << { kind: 'ELSE', at: index }
189 when 'THEN'
190 mismatch(source, %w[IF ELSE])
191 @compiling.code[@control.pop[:at]][2] = @compiling.code.length
192 when 'BEGIN'
193 @control << { kind: 'BEGIN', begin: @compiling.code.length }
194 when 'UNTIL', 'AGAIN'
195 mismatch(source, ['BEGIN'])
196 record = @control.pop
197 emit(key == 'UNTIL' ? :zero_jump : :jump, source, record[:begin])
198 when 'WHILE'
199 mismatch(source, ['BEGIN'])
200 record = @control.pop
201 @control << { kind: 'WHILE', begin: record[:begin], at: emit(:zero_jump, source, nil) }
202 when 'REPEAT'
203 mismatch(source, ['WHILE'])
204 record = @control.pop
205 emit(:jump, source, record[:begin])
206 @compiling.code[record[:at]][2] = @compiling.code.length
207 when 'DO', '?DO'
208 @control << { kind: key, at: emit(:do, source, nil, key == '?DO'), leaves: [] }
209 when 'LOOP', '+LOOP'
210 mismatch(source, %w[DO ?DO])
211 record = @control.pop
212 emit(:loop, source, record[:at] + 1, key == '+LOOP')
213 finish = @compiling.code.length
214 @compiling.code[record[:at]][2] = finish
215 record[:leaves].each { |at| @compiling.code[at][2] = finish }
216 when 'LEAVE'
217 record = @control.reverse.find { |item| %w[DO ?DO].include?(item[:kind]) }
218 error(source, 'control structure mismatch') unless record
219 record[:leaves] << emit(:leave, source, nil)
220 when 'CASE'
221 @control << { kind: 'CASE', ends: [] }
222 when 'OF'
223 mismatch(source, ['CASE'])
224 @control << { kind: 'OF', at: emit(:of, source, nil) }
225 when 'ENDOF'
226 mismatch(source, ['OF'])
227 record = @control.pop
228 @control[-1][:ends] << emit(:jump, source, nil)
229 @compiling.code[record[:at]][2] = @compiling.code.length
230 when 'ENDCASE'
231 mismatch(source, ['CASE'])
232 record = @control.pop
233 emit(:drop, source)
234 record[:ends].each { |at| @compiling.code[at][2] = @compiling.code.length }
235 when 'EXIT'
236 emit(:exit, source)
237 when 'RECURSE'
238 emit(:call, source, @compiling)
239 when 'DOES>'
240 error(source, 'control structure mismatch') unless @control.empty?
241 emit(:does, source, @compiling, @compiling.code.length + 1)
242 when ';'
243 unless @control.empty?
244 kind = @control[-1][:kind]
245 kind = 'IF' if kind == 'ELSE'
246 kind = 'BEGIN' if kind == 'WHILE'
247 error(source, "unclosed #{kind}")
248 end
249 word = @compiling
250 @compiling = nil
251 if @anonymous
252 push(token(word), source)
253 else
254 @dictionary << word
255 end
256 when "[']"
257 name = read_name(source)
258 word = lookup(name) || error(name, 'undefined word')
259 emit(:literal, source, token(word))
260 when '{:'
261 names = []
262 loop do
263 name = read_word
264 error(source, 'unterminated locals') unless name
265 break if name.upcase == ':}'
266 names << name
267 end
268 names = names.take_while { |name| name != '--' }
269 slots = names.map do |name|
270 key_name = check_name(name)
271 index = @compiling.local_count
272 @compiling.local_count += 1
273 @locals[key_name] = index
274 index
275 end
276 emit(:locals, source, slots)
277 when '>R', 'R>', 'R@', 'I', 'J', 'UNLOOP'
278 emit(:return_stack, source, key)
279 else
280 if @locals.key?(key)
281 emit(:local, source, @locals[key])
282 elsif (word = lookup(source))
283 emit(:call, source, word)
284 elsif !(value = number(source)).nil?
285 emit(:literal, source, value)
286 else
287 error(source, 'undefined word')
288 end
289 end
290 end

292 def process(source)
293 key = source.upcase
294 case key
295 when '\\'
296 @position = @line.length
297 return
298 when '('
299 finish = @line.index(')', @position)
300 error(source, 'unterminated comment') unless finish
301 @position = finish + 1
302 return
303 end
304 error(source, 'compile only') if !@compiling && COMPILE_ONLY.include?(key)
305 error(source, 'interpret only') if @compiling && INTERPRET_ONLY.include?(key)
306 case key
307 when '."', 'ABORT"'
308 start = @position + 1
309 finish = @line.index('"', start)
310 error(source, 'unterminated string') unless @position < @line.length && finish
311 string = @line[start...finish]
312 @position = finish + 1
313 if @compiling
314 emit(key == '."' ? :write : :abort_string, source, string)
315 else
316 @pending << string
317 end
318 when 'TO', 'IS'
319 name = read_name(source)
320 if key == 'TO' && @compiling && @locals.key?(name.upcase)
321 emit(:set_local, source, @locals[name.upcase])
322 return
323 end
324 word = lookup(name) || error(name, 'undefined word')
325 error(name, key == 'TO' ? 'not a value' : 'not deferred') unless word.kind == (key == 'TO' ? :value : :defer)
326 if @compiling
327 emit(key == 'TO' ? :to : :is, source, word)
328 else
329 assign(word, source, key == 'IS')
330 end
331 when ':', ':NONAME'
332 name = key == ':' ? read_name(source) : source
333 check_name(name) if key == ':'
334 @compiling = ForthWord.new(name.upcase, :colon, nil, @memory.length * 8)
335 @definition_name = name
336 @anonymous = key == ':NONAME'
337 @control = []
338 @locals = {}
339 when 'FORGET'
340 name = read_name(source)
341 index = @dictionary.rindex { |word| word.name == name.upcase }
342 error(name, 'undefined word') unless index
343 @memory = @memory.take(@dictionary[index].here / 8)
344 @dictionary = @dictionary.take(index)
345 else
346 if @compiling
347 compile(source)
348 elsif (word = lookup(source))
349 execute(word, source)
350 elsif !(value = number(source)).nil?
351 push(value, source)
352 else
353 error(source, 'undefined word')
354 end
355 end
356 end

358 def assign(word, source, deferred)
359 value = take(1, source)[0]
360 if deferred
361 error(source, 'invalid argument') unless valid_token?(value)
362 word.value = value
363 else
364 @memory[word.value / 8] = value
365 end
366 end

368 def return_operation(key, source)
369 case key
370 when '>R'
371 return_push(take(1, source)[0], source)
372 when 'R>'
373 return_need(1, source)
374 push(@returns.pop, source)
375 when 'R@', 'I'
376 return_need(1, source)
377 push(@returns[-1], source)
378 when 'J'
379 return_need(3, source)
380 push(@returns[-3], source)
381 when 'UNLOOP'
382 return_need(2, source)
383 @returns.pop(2)
384 end
385 end

387 def execute_definition(word, start, source)
388 error(source, 'nesting too deep') if @depth == 100
389 @depth += 1
390 return_size = @returns.length
391 locals = Array.new(word.local_count, 0)
392 pc = start
393 while pc < word.code.length
394 op, spelling, a, b = word.code[pc]
395 pc += 1
396 case op
397 when :literal
398 push(a, spelling)
399 when :call
400 execute(a, spelling)
401 when :write
402 @pending << a
403 when :local
404 push(locals[a], spelling)
405 when :set_local
406 locals[a] = take(1, spelling)[0]
407 when :locals
408 values = take(a.length, spelling)
409 a.each_with_index { |slot, i| locals[slot] = values[i] }
410 when :to, :is
411 assign(a, spelling, op == :is)
412 when :exit
413 break
414 when :does
415 created = @dictionary[-1]
416 error(spelling, 'no CREATE word') unless created && created.kind == :create
417 created.does_word = a
418 created.does_start = b
419 break
420 when :jump
421 pc = a
422 when :zero_jump
423 pc = a if take(1, spelling)[0] == 0
424 when :do
425 limit, index = take(2, spelling)
426 if b && limit == index
427 pc = a
428 else
429 return_push(limit, spelling)
430 return_push(index, spelling)
431 end
432 when :loop
433 step = b ? take(1, spelling)[0] : 1
434 return_need(2, spelling)
435 index = @returns[-1]
436 limit = @returns[-2]
437 next_index = index + step
438 if (index - limit).negative? != (next_index - limit).negative?
439 @returns.pop(2)
440 else
441 @returns[-1] = next_index
442 pc = a
443 end
444 when :leave
445 return_need(2, spelling)
446 @returns.pop(2)
447 pc = a
448 when :of
449 x, y = take(2, spelling)
450 if x != y
451 push(x, spelling)
452 pc = a
453 end
454 when :drop
455 take(1, spelling)
456 when :return_stack
457 return_operation(a, spelling)
458 when :abort_string
459 error(spelling, a, -2) if take(1, spelling)[0] != 0
460 end
461 end
462 @returns.slice!(return_size..-1) if @returns.length > return_size
463 @depth -= 1
464 end

466 def execute_token(value, source)
467 error(source, 'invalid argument') unless valid_token?(value)
468 word = @tokens[value - 1]
469 execute(word, word.name)
470 end

472 def execute(word, source)
473 while word.kind == :defer
474 error(source, 'deferred word not set') unless word.value
475 error(source, 'invalid argument') unless valid_token?(word.value)
476 word = @tokens[word.value - 1]
477 source = word.name
478 end
479 case word.kind
480 when :colon
481 execute_definition(word, 0, source)
482 when :create, :variable
483 push(word.value, source)
484 execute_definition(word.does_word, word.does_start, source) if word.does_word
485 when :constant
486 push(word.value, source)
487 when :value
488 push(@memory[word.value / 8], source)
489 when :builtin
490 builtin(word.name, source)
491 end
492 end

494 def builtin(key, source)
495 case key
496 when '+', '-', '*', '/', 'MOD', '/MOD', 'MIN', 'MAX', '=', '<>', '<', '>', 'AND', 'OR', 'XOR'
497 a, b = take(2, source)
498 case key
499 when '+' then push(a + b, source)
500 when '-' then push(a - b, source)
501 when '*' then push(a * b, source)
502 when '/', 'MOD', '/MOD'
503 error(source, 'division by zero') if b == 0
504 q, r = a.divmod(b)
505 push_all(key == '/' ? [q] : key == 'MOD' ? [r] : [r, q], source)
506 when 'MIN' then push([a, b].min, source)
507 when 'MAX' then push([a, b].max, source)
508 when '=' then push(a == b ? -1 : 0, source)
509 when '<>' then push(a != b ? -1 : 0, source)
510 when '<' then push(a < b ? -1 : 0, source)
511 when '>' then push(a > b ? -1 : 0, source)
512 when 'AND' then push(a & b, source)
513 when 'OR' then push(a | b, source)
514 when 'XOR' then push(a ^ b, source)
515 end
516 when '*/', '*/MOD'
517 a, b, c = take(3, source)
518 error(source, 'division by zero') if c == 0
519 q, r = (a * b).divmod(c)
520 push_all(key == '*/' ? [q] : [r, q], source)
521 when 'NEGATE', 'ABS', '1+', '1-', '2*', '2/', '0=', '0<>', '0<', '0>', 'INVERT', 'CELLS', 'CELL+'
522 a = take(1, source)[0]
523 value = case key
524 when 'NEGATE' then -a
525 when 'ABS' then a.abs
526 when '1+' then a + 1
527 when '1-' then a - 1
528 when '2*' then a * 2
529 when '2/' then a.div(2)
530 when '0=' then a == 0 ? -1 : 0
531 when '0<>' then a != 0 ? -1 : 0
532 when '0<' then a < 0 ? -1 : 0
533 when '0>' then a > 0 ? -1 : 0
534 when 'INVERT' then -a - 1
535 when 'CELLS' then a * 8
536 when 'CELL+' then a + 8
537 end
538 push(value, source)
539 when 'WITHIN'
540 n, lo, hi = take(3, source)
541 push(lo <= n && n < hi ? -1 : 0, source)
542 when 'TRUE', 'FALSE', 'BL', 'BASE', 'HERE'
543 push({ 'TRUE' => -1, 'FALSE' => 0, 'BL' => 32, 'BASE' => 0, 'HERE' => @memory.length * 8 }[key], source)
544 when 'DUP', '?DUP', 'DROP', 'SWAP', 'OVER', 'ROT', '-ROT', 'NIP', 'TUCK', '2DUP', '2DROP', '2SWAP', '2OVER'
545 count = { 'DUP' => 1, '?DUP' => 1, 'DROP' => 1, 'SWAP' => 2, 'OVER' => 2,
546 'ROT' => 3, '-ROT' => 3, 'NIP' => 2, 'TUCK' => 2, '2DUP' => 2,
547 '2DROP' => 2, '2SWAP' => 4, '2OVER' => 4 }[key]
548 values = take(count, source)
549 a, b, c, d = values
550 result = case key
551 when 'DUP' then [a, a]
552 when '?DUP' then a == 0 ? [a] : [a, a]
553 when 'DROP', '2DROP' then []
554 when 'SWAP' then [b, a]
555 when 'OVER' then [a, b, a]
556 when 'ROT' then [b, c, a]
557 when '-ROT' then [c, a, b]
558 when 'NIP' then [b]
559 when 'TUCK' then [b, a, b]
560 when '2DUP' then [a, b, a, b]
561 when '2SWAP' then [c, d, a, b]
562 when '2OVER' then [a, b, c, d, a, b]
563 end
564 push_all(result, source)
565 when 'DEPTH'
566 push(@stack.length, source)
567 when 'PICK', 'ROLL'
568 u = take(1, source)[0]
569 error(source, 'invalid argument') if u.negative?
570 error(source, 'stack underflow') if @stack.length <= u
571 value = key == 'PICK' ? @stack[-1 - u] : @stack.delete_at(@stack.length - 1 - u)
572 push(value, source)
573 when '.', '.R', '?'
574 values = take(key == '.R' ? 2 : 1, source)
575 n = key == '?' ? @memory[address(values[0], source)] : values[0]
576 text = output_number(n, source)
577 @pending << (key == '.R' ? text.rjust([values[1], 0].max) : text + ' ')
578 when '.S'
579 output_number(0, source)
580 @pending << '<' + output_number(@stack.length, source) + '> '
581 @stack.each { |value| @pending << output_number(value, source) + ' ' }
582 when 'CR'
583 flush
584 when 'EMIT'
585 c = take(1, source)[0]
586 error(source, 'invalid argument') unless (32..126).cover?(c)
587 @pending << c.chr
588 when 'SPACE'
589 @pending << ' '
590 when 'SPACES'
591 @pending << ' ' * [take(1, source)[0], 0].max
592 when '!', '+!'
593 n, a = take(2, source)
594 index = address(a, source)
595 @memory[index] = key == '!' ? n : @memory[index] + n
596 when '@'
597 push(@memory[address(take(1, source)[0], source)], source)
598 when ','
599 @memory << take(1, source)[0]
600 when 'ALLOT'
601 n = take(1, source)[0]
602 error(source, 'invalid argument') if n.negative? || n % 8 != 0
603 @memory.concat(Array.new(n / 8, 0))
604 when 'HEX', 'DECIMAL'
605 @memory[0] = key == 'HEX' ? 16 : 10
606 when 'CREATE', 'VARIABLE', 'CONSTANT', 'VALUE', 'DEFER'
607 name = check_name(read_name(source))
608 value = %w[CONSTANT VALUE].include?(key) ? take(1, source)[0] : nil
609 here = @memory.length * 8
610 stored = %w[CREATE VARIABLE VALUE].include?(key) ? here : value
611 @dictionary << ForthWord.new(name, key.downcase.to_sym, stored, here)
612 @memory << (key == 'VARIABLE' ? 0 : value) if %w[VARIABLE VALUE].include?(key)
613 when "'"
614 name = read_name(source)
615 word = lookup(name) || error(name, 'undefined word')
616 push(token(word), source)
617 when 'EXECUTE'
618 execute_token(take(1, source)[0], source)
619 when 'CATCH'
620 xt = take(1, source)[0]
621 error(source, 'invalid argument') unless valid_token?(xt)
622 saved = @stack.dup
623 saved_returns = @returns.length
624 saved_depth = @depth
625 begin
626 execute_token(xt, source)
627 rescue ForthError => exception
628 raise if exception.code.nil?
629 @stack = saved
630 @returns.slice!(saved_returns..-1) if @returns.length > saved_returns
631 @depth = saved_depth
632 push(exception.code, source)
633 else
634 push(0, source)
635 end
636 when 'THROW'
637 n = take(1, source)[0]
638 error(source, "exception #{n}", n) unless n == 0
639 when 'ABORT'
640 error(source, 'aborted')
641 when 'WORDS'
642 @dictionary.reverse_each { |word| @pending << word.name + ' ' }
643 end
644 end

646 def run
647 STDIN.each_line do |input|
648 @line = input.delete_suffix("\n")
649 if @line == 'BYE'
650 puts "ERROR #{@definition_name}: unterminated definition" if @compiling
651 break
652 end
653 @position = 0
654 begin
655 while (source = read_word)
656 process(source)
657 end
658 rescue ForthError => exception
659 flush unless @pending.empty?
660 puts "ERROR #{exception.word}: #{exception.reason}"
661 @stack.clear
662 @returns.clear
663 @depth = 0
664 @compiling = nil
665 end
666 flush unless @pending.empty?
667 end
668 end
669end
