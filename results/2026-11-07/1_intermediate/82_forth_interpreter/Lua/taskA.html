82function Machine:numeric(raw, base)
83 if #raw == 3 and raw:sub(1, 1) == "'" and raw:sub(3, 3) == "'" then
84 return raw:byte(2)
85 end
86 local text, first = raw:upper(), raw:sub(1, 1)
87 if first == "$" then
88 base, text = 16, text:sub(2)
89 elseif first == "#" then
90 base, text = 10, text:sub(2)
91 elseif first == "%" then
92 base, text = 2, text:sub(2)
93 elseif base < 2 or base > 36 then
94 problem(raw, "invalid base", -40)
95 end
96 local negative = text:sub(1, 1) == "-"
97 if negative then
98 text = text:sub(2)
99 end
100 if text == "" then
101 return nil
102 end
103 local digits = {}
104 for i = 1, #text do
105 local c, digit = text:byte(i), -1
106 if c >= 48 and c <= 57 then
107 digit = c - 48
108 elseif c >= 65 and c <= 90 then
109 digit = c - 65 + 10
110 end
111 if digit < 0 or digit >= base then
112 return nil
113 end
114 digits[i] = digit
115 end
116 local value = 0
117 for _, digit in ipairs(digits) do
118 value = value * base + digit
119 end
120 return negative and -value or value
121end

129 end
130end

137 result[i] = self.data[start + i]
138 end
139 for i = #self.data, start + 1, -1 do
140 self.data[i] = nil
141 end
142 return result
143end

167function Machine:format(value, raw)
168 local base = self.memory[1]
169 if base < 2 or base > 36 then
170 problem(raw, "invalid base", -40)
171 end
172 local negative, digits = value < 0, "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
173 if negative then
174 value = -value
175 end
176 local result = ""
177 repeat
178 local digit = value % base
179 result = digits:sub(digit + 1, digit + 1) .. result
180 value = value // base
181 until value == 0
182 return (negative and "-" or "") .. result
183end

221function Machine:compile_control(raw, name)
222 if name == "IF" then
223 self.controls[#self.controls + 1] = {
224 kind = "IF",
225 index = self:add_action({kind = "zero", raw = raw})
226 }
227 elseif name == "ELSE" then
228 local top = self:top_control(raw, "IF")
229 local index = self:add_action({kind = "jump", raw = raw})
230 self.compiling.actions[top.index].target = #self.compiling.actions + 1
231 top.kind, top.index = "ELSE", index
232 elseif name == "THEN" then
233 local top = self:top_control(raw, "IF", "ELSE")
234 self.compiling.actions[top.index].target = #self.compiling.actions + 1
235 table.remove(self.controls)
236 elseif name == "BEGIN" then
237 self.controls[#self.controls + 1] = {kind = "BEGIN", start = #self.compiling.actions + 1}
238 elseif name == "UNTIL" or name == "AGAIN" then
239 local top = self:top_control(raw, "BEGIN")
240 self:add_action(
241 {kind = name == "UNTIL" and "zero" or "jump", raw = raw, target = top.start}
242 )
243 table.remove(self.controls)
244 elseif name == "WHILE" then
245 local top = self:top_control(raw, "BEGIN")
246 top.kind, top.index = "WHILE", self:add_action({kind = "zero", raw = raw})
247 elseif name == "REPEAT" then
248 local top = self:top_control(raw, "WHILE")
249 self:add_action({kind = "jump", raw = raw, target = top.start})
250 self.compiling.actions[top.index].target = #self.compiling.actions + 1
251 table.remove(self.controls)
252 elseif name == "DO" or name == "?DO" then
253 local index = self:add_action({kind = name, raw = raw})
254 self.controls[#self.controls + 1] = {
255 kind = name,
256 index = index,
257 start = index + 1,
258 leaves = {}
259 }
260 elseif name == "LOOP" or name == "+LOOP" then
261 local top = self:top_control(raw, "DO", "?DO")
262 self:add_action({kind = name, raw = raw, target = top.start})
263 local finish = #self.compiling.actions + 1
264 self.compiling.actions[top.index].target = finish
265 for _, index in ipairs(top.leaves) do
266 self.compiling.actions[index].target = finish
267 end
268 table.remove(self.controls)
269 elseif name == "LEAVE" then
270 local top
271 for i = #self.controls, 1, -1 do
272 if self.controls[i].kind == "DO" or self.controls[i].kind == "?DO" then
273 top = self.controls[i]
274 break
275 end
276 end
277 if not top then
278 syntax(raw, "control structure mismatch")
279 end
280 top.leaves[#top.leaves + 1] = self:add_action({kind = "LEAVE", raw = raw})
281 elseif name == "CASE" then
282 self.controls[#self.controls + 1] = {kind = "CASE", ends = {}}
283 elseif name == "OF" then
284 self:top_control(raw, "CASE")
285 self.controls[#self.controls + 1] = {
286 kind = "OF",
287 index = self:add_action({kind = "OF", raw = raw})
288 }
289 elseif name == "ENDOF" then
290 local top = self:top_control(raw, "OF")
291 local index = self:add_action({kind = "jump", raw = raw})
292 self.compiling.actions[top.index].target = #self.compiling.actions + 1
293 table.remove(self.controls)
294 local parent = self:top_control(raw, "CASE")
295 parent.ends[#parent.ends + 1] = index
296 elseif name == "ENDCASE" then
297 local top = self:top_control(raw, "CASE")
298 self:add_action({kind = "ENDCASE", raw = raw})
299 for _, index in ipairs(top.ends) do
300 self.compiling.actions[index].target = #self.compiling.actions + 1
301 end
302 table.remove(self.controls)
303 else
304 return false
305 end
306 return true
307end

322 while true do
323 local name = self:read()
324 if not name then
325 break
326 end
327 if name:upper() == ":}" then

333 end

351function Machine:assign(raw, name)
352 local target_name = self:read_name(raw)
353 if name == "TO" and self.compiling then
354 local slot = self.compiling.local_names[target_name:upper()]
355 if slot then
356 self:add_action({kind = "localSet", raw = raw, target = slot})
357 return
358 end
359 end
360 local target = self:lookup(target_name)
361 if not target then
362 problem(target_name, "undefined word", -13)
363 end
364 local required, reason =
365 name == "TO" and "value" or "defer",
366 name == "TO" and "not a value" or "not deferred"
367 if target.kind ~= required then
368 syntax(target_name, reason)
369 end
370 local action = {kind = name, raw = raw, word = target}
371 if self.compiling then
372 self:add_action(action)
373 else
374 self:runtime_action(action, {})
375 end
376end
377function Machine:process(raw)
378 local name = raw:upper()
379 if name == "\\" then
380 self.position = #self.line + 1
381 return
382 end
383 if name == "(" then
384 local closing = self.line:find(")", self.position, true)
385 if not closing then
386 syntax(raw, "unterminated comment")
387 end
388 self.position = closing + 1
389 return
390 end
391 if not self.compiling and compile_only[name] then
392 syntax(raw, "compile only")
393 end
394 if self.compiling and interpret_only[name] then
395 syntax(raw, "interpret only")
396 end
397 if name == '."' or name == 'ABORT"' then
398 local text = self:read_string(raw)
399 if self.compiling then
400 self:add_action({kind = name, raw = raw, text = text})
401 else
402 self:write(text)
403 end
404 return
405 end
406 if name == "TO" or name == "IS" then
407 self:assign(raw, name)
408 return
409 end
410 if name == ":" or name == ":NONAME" then
411 local raw_name = raw
412 if name == ":" then
413 raw_name = self:read_name(raw)
414 self:check_name(raw_name)
415 end
416 self.compiling = {
417 name = name == ":NONAME" and ":NONAME" or raw_name:upper(),
418 raw_name = raw_name,
419 kind = "colon",
420 here = #self.memory * 8,
421 actions = {},
422 local_names = {},
423 local_count = 0
424 }
425 self.controls = {}
426 return
427 end
428 if name == "FORGET" then
429 local target_name, index = self:read_name(raw), nil
430 for i = #self.dictionary, 1, -1 do
431 if self.dictionary[i].name == target_name:upper() then
432 index = i
433 break
434 end
435 end
436 if not index then
437 problem(target_name, "undefined word", -13)
438 end
439 for i = #self.memory, self.dictionary[index].here // 8 + 1, -1 do
440 self.memory[i] = nil
441 end
442 for i = #self.dictionary, index, -1 do
443 self.dictionary[i] = nil
444 end
445 return
446 end
447 if self.compiling then
448 if self:compile_control(raw, name) then
449 return
450 end
451 if name == ";" then
452 if #self.controls > 0 then
453 local kind = self.controls[#self.controls].kind
454 if kind == "ELSE" then
455 kind = "IF"
456 elseif kind == "WHILE" then
457 kind = "BEGIN"
458 end
459 syntax(raw, "unclosed " .. kind)
460 end
461 local finished = self.compiling
462 self.compiling = nil
463 if finished.name == ":NONAME" then
464 self:push(self:execution_token(finished), raw)
465 else
466 self.dictionary[#self.dictionary + 1] = finished
467 end
468 return
469 elseif name == "DOES>" then
470 if #self.controls > 0 then
471 syntax(raw, "control structure mismatch")
472 end
473 self:add_action({kind = "DOES>", raw = raw, word = self.compiling})
474 return
475 elseif name == "EXIT" then
476 self:add_action({kind = "EXIT", raw = raw})
477 return
478 elseif name == "RECURSE" then
479 self:add_action({kind = "call", raw = raw, word = self.compiling})
480 return
481 elseif name == "[']" then
482 local target_name = self:read_name(raw)
483 local target = self:lookup(target_name)
484 if not target then
485 problem(target_name, "undefined word", -13)
486 end
487 self:add_action({kind = "push", raw = raw, value = self:execution_token(target)})
488 return
489 elseif name == "{:" then
490 self:locals(raw)
491 return
492 end
493 if
494 name == ">R" or name == "R>" or name == "R@" or name == "I" or name == "J" or
495 name == "UNLOOP"
496 then
497 self:add_action({kind = name, raw = raw})
498 return
499 end
500 local slot = self.compiling.local_names[name]
501 if slot then
502 self:add_action({kind = "localGet", raw = raw, target = slot})
503 return
504 end
505 local target = self:lookup(raw)
506 if target then
507 self:add_action({kind = "call", raw = raw, word = target})
508 return
509 end
510 local value = self:numeric(raw, self.memory[1])
511 if value ~= nil then
512 self:add_action({kind = "push", raw = raw, value = value})
513 return
514 end
515 problem(raw, "undefined word", -13)
516 end
517 local target = self:lookup(raw)
518 if target then
519 self:execute(target, raw)
520 return
521 end
522 local value = self:numeric(raw, self.memory[1])
523 if value ~= nil then
524 self:push(value, raw)
525 return
526 end
527 problem(raw, "undefined word", -13)
528end
529function Machine:runtime_action(action, locals)
530 local kind, raw = action.kind, action.raw
531 if kind == "push" then
532 self:push(action.value, raw)
533 elseif kind == "call" then
534 self:execute(action.word, raw)
535 elseif kind == '."' then
536 self:write(action.text)
537 elseif kind == 'ABORT"' then
538 if self:take(1, raw)[1] ~= 0 then
539 problem(raw, action.text, -2)
540 end
541 elseif kind == "localGet" then
542 self:push(locals[action.target], raw)
543 elseif kind == "localSet" then
544 locals[action.target] = self:take(1, raw)[1]
545 elseif kind == "locals" then
546 local values = self:take(#action.slots, raw)
547 for i, slot in ipairs(action.slots) do
548 locals[slot] = values[i]
549 end
550 elseif kind == "TO" then
551 self.memory[action.word.address // 8 + 1] = self:take(1, raw)[1]
552 elseif kind == "IS" then
553 local value = self:take(1, raw)[1]
554 self:valid_token(value, raw)
555 action.word.token = value
556 elseif kind == ">R" then
557 self:return_push(self:take(1, raw)[1], raw)
558 elseif kind == "R>" then
559 self:return_need(1, raw)
560 self:push(table.remove(self.returns), raw)
561 elseif kind == "R@" or kind == "I" or kind == "J" then
562 local count = kind == "J" and 3 or 1
563 self:return_need(count, raw)
564 self:push(self.returns[#self.returns - count + 1], raw)
565 elseif kind == "UNLOOP" then
566 self:return_need(2, raw)
567 table.remove(self.returns)
568 table.remove(self.returns)
569 end
570end
571function Machine:definition(word, start, raw)
572 if self.depth == 100 then
573 problem(raw, "nesting too deep", -7)
574 end
575 self.depth = self.depth + 1
576 local saved, locals, pc = #self.returns, {}, start
577 for i = 1, word.local_count do
578 locals[i] = 0
579 end
580 while pc <= #word.actions do
581 local action = word.actions[pc]
582 pc = pc + 1
583 local kind = action.kind
584 if kind == "EXIT" then
585 break
586 elseif kind == "DOES>" then
587 local newest = self.dictionary[#self.dictionary]
588 if not newest or newest.kind ~= "create" then
589 problem(action.raw, "no CREATE word", -31)
590 end
591 newest.does, newest.does_start = action.word, pc
592 break
593 elseif kind == "jump" then
594 pc = action.target
595 elseif kind == "zero" then
596 if self:take(1, action.raw)[1] == 0 then
597 pc = action.target
598 end
599 elseif kind == "DO" or kind == "?DO" then
600 local values = self:take(2, action.raw)
601 if kind == "?DO" and values[1] == values[2] then
602 pc = action.target
603 else
604 self:return_push(values[1], action.raw)
605 self:return_push(values[2], action.raw)
606 end
607 elseif kind == "LOOP" or kind == "+LOOP" then
608 local step = kind == "+LOOP" and self:take(1, action.raw)[1] or 1
609 self:return_need(2, action.raw)
610 local index, limit = self.returns[#self.returns], self.returns[#self.returns - 1]
611 local next_index = index + step
612 if (index - limit < 0) ~= (next_index - limit < 0) then
613 table.remove(self.returns)
614 table.remove(self.returns)
615 else
616 self.returns[#self.returns] = next_index
617 pc = action.target
618 end
619 elseif kind == "LEAVE" then
620 self:return_need(2, action.raw)
621 table.remove(self.returns)
622 table.remove(self.returns)
623 pc = action.target
624 elseif kind == "OF" then
625 local values = self:take(2, action.raw)
626 if values[1] ~= values[2] then
627 self:push(values[1], action.raw)
628 pc = action.target
629 end
630 elseif kind == "ENDCASE" then
631 self:take(1, action.raw)
632 else
633 self:runtime_action(action, locals)
634 end
635 end
636 for i = #self.returns, saved + 1, -1 do
637 self.returns[i] = nil
638 end
639 self.depth = self.depth - 1
640end
641function Machine:execute(word, raw)
642 while word.kind == "defer" do
643 if not word.token then
644 problem(raw, "deferred word not set", -21)
645 end
646 word = self:valid_token(word.token, raw)
647 raw = word.name
648 end
649 if word.kind == "colon" then
650 self:definition(word, 1, raw)
651 elseif word.kind == "create" then
652 self:push(word.address, raw)
653 if word.does then
654 self:definition(word.does, word.does_start, raw)
655 end
656 elseif word.kind == "variable" then
657 self:push(word.address, raw)
658 elseif word.kind == "constant" then
659 self:push(word.value, raw)
660 elseif word.kind == "value" then
661 self:push(self.memory[word.address // 8 + 1], raw)
662 else
663 self:builtin(word.name, raw)
664 end
665end

667for _, entry in ipairs(

671 {4, "2SWAP 2OVER"}

677end
678function Machine:builtin(name, raw)
679 if
680 name == "CREATE" or name == "VARIABLE" or name == "CONSTANT" or name == "VALUE" or
681 name == "DEFER"
682 then
683 local raw_name = self:read_name(raw)
684 self:check_name(raw_name)
685 local value = 0
686 if name == "CONSTANT" or name == "VALUE" then
687 value = self:take(1, raw)[1]
688 end
689 self.dictionary[#self.dictionary + 1] = {
690 name = raw_name:upper(),
691 raw_name = raw_name,
692 kind = name:lower(),
693 here = #self.memory * 8,
694 address = #self.memory * 8,
695 value = value
696 }
697 if name == "VARIABLE" or name == "VALUE" then
698 self.memory[#self.memory + 1] = value
699 end
700 return
701 elseif name == "'" then
702 local target_name = self:read_name(raw)
703 local target = self:lookup(target_name)
704 if not target then
705 problem(target_name, "undefined word", -13)
706 end
707 self:push(self:execution_token(target), raw)
708 return
709 elseif name == "EXECUTE" then
710 self:execute_token(self:take(1, raw)[1], raw)
711 return
712 elseif name == "CATCH" then
713 local token = self:take(1, raw)[1]
714 local target = self:valid_token(token, raw)
715 local saved, return_size, depth = copy(self.data), #self.returns, self.depth
716 local ok, caught =
717 pcall(
718 function()
719 self:execute(target, target.name)
720 end
721 )
722 local code = 0
723 if not ok then
724 if type(caught) ~= "table" or not caught.forth or caught.code == nil then
725 error(caught, 0)
726 end
727 self.data = saved
728 for i = #self.returns, return_size + 1, -1 do
729 self.returns[i] = nil
730 end
731 self.depth, code = depth, caught.code
732 end
733 self:push(code, raw)
734 return
735 elseif name == "THROW" then
736 local code = self:take(1, raw)[1]
737 if code ~= 0 then
738 problem(raw, "exception " .. code, code)
739 end
740 return
741 elseif name == "ABORT" then
742 problem(raw, "aborted", -1)
743 elseif name == "WORDS" then
744 for i = #self.dictionary, 1, -1 do
745 self:write(self.dictionary[i].name .. " ")
746 end
747 return
748 elseif name == "CR" then
749 self:flush()
750 return
751 elseif name == "SPACE" then
752 self:write(" ")
753 return
754 elseif name == ".S" then
755 local count, values = self:format(#self.data, raw), {}
756 for i, value in ipairs(self.data) do
757 values[i] = self:format(value, raw)
758 end
759 self:write("<" .. count .. "> ")
760 for _, value in ipairs(values) do
761 self:write(value .. " ")
762 end
763 return
764 elseif name == "TRUE" then
765 self:push(-1, raw)
766 return
767 elseif name == "FALSE" then
768 self:push(0, raw)
769 return
770 elseif name == "BL" then
771 self:push(32, raw)
772 return
773 elseif name == "BASE" then
774 self:push(0, raw)
775 return
776 elseif name == "HERE" then
777 self:push(#self.memory * 8, raw)
778 return
779 elseif name == "DEPTH" then
780 self:push(#self.data, raw)
781 return
782 elseif name == "HEX" then
783 self.memory[1] = 16
784 return
785 elseif name == "DECIMAL" then
786 self.memory[1] = 10
787 return
788 end
789 local values = self:take(effects[name] or 1, raw)
790 local a, b = values[1], values[2]
791 local function push(...)
792 for _, value in ipairs({...}) do
793 self:push(value, raw)
794 end
795 end
796 if name == "+" then
797 push(a + b)
798 elseif name == "-" then
799 push(a - b)
800 elseif name == "*" then
801 push(a * b)
802 elseif name == "/" or name == "MOD" or name == "/MOD" or name == "*/" or name == "*/MOD" then
803 local dividend, divisor = a, b
804 if #values == 3 then
805 dividend, divisor = a * b, values[3]
806 end
807 if divisor == 0 then
808 problem(raw, "division by zero", -10)
809 end
810 local q = dividend // divisor
811 local r = dividend - divisor * q
812 if name == "MOD" then
813 push(r)
814 elseif name == "/MOD" or name == "*/MOD" then
815 push(r, q)
816 else
817 push(q)
818 end
819 elseif name == "NEGATE" then
820 push(-a)
821 elseif name == "ABS" then
822 push(math.abs(a))
823 elseif name == "MIN" then
824 push(math.min(a, b))
825 elseif name == "MAX" then
826 push(math.max(a, b))
827 elseif name == "1+" then
828 push(a + 1)
829 elseif name == "1-" then
830 push(a - 1)
831 elseif name == "2*" then
832 push(2 * a)
833 elseif name == "2/" then
834 push(a // 2)
835 elseif name == "=" then
836 push(flag(a == b))
837 elseif name == "<>" then
838 push(flag(a ~= b))
839 elseif name == "<" then
840 push(flag(a < b))
841 elseif name == ">" then
842 push(flag(a > b))
843 elseif name == "0=" then
844 push(flag(a == 0))
845 elseif name == "0<>" then
846 push(flag(a ~= 0))
847 elseif name == "0<" then
848 push(flag(a < 0))
849 elseif name == "0>" then
850 push(flag(a > 0))
851 elseif name == "WITHIN" then
852 push(flag(b <= a and a < values[3]))
853 elseif name == "AND" then
854 push(a & b)
855 elseif name == "OR" then
856 push(a | b)
857 elseif name == "XOR" then
858 push(a ~ b)
859 elseif name == "INVERT" then
860 push(~a)
861 elseif name == "DUP" then
862 push(a, a)
863 elseif name == "?DUP" then
864 push(a)
865 if a ~= 0 then
866 push(a)
867 end
868 elseif name == "DROP" or name == "2DROP" then
869 elseif name == "SWAP" then
870 push(b, a)
871 elseif name == "OVER" then
872 push(a, b, a)
873 elseif name == "ROT" then
874 push(b, values[3], a)
875 elseif name == "-ROT" then
876 push(values[3], a, b)
877 elseif name == "NIP" then
878 push(b)
879 elseif name == "TUCK" then
880 push(b, a, b)
881 elseif name == "2DUP" then
882 push(a, b, a, b)
883 elseif name == "2SWAP" then
884 push(values[3], values[4], a, b)
885 elseif name == "2OVER" then
886 push(a, b, values[3], values[4], a, b)
887 elseif name == "PICK" or name == "ROLL" then
888 if a < 0 then
889 problem(raw, "invalid argument", -24)
890 end
891 if a >= #self.data then
892 problem(raw, "stack underflow", -4)
893 end
894 local index = #self.data - a
895 local value = self.data[index]
896 if name == "ROLL" then
897 table.remove(self.data, index)
898 end
899 push(value)
900 elseif name == "." then
901 self:write(self:format(a, raw) .. " ")
902 elseif name == ".R" then
903 local text = self:format(a, raw)
904 self:write(string.rep(" ", math.max(0, b - #text)) .. text)
905 elseif name == "?" then
906 local address = self:address(a, raw)
907 self:write(self:format(self.memory[address], raw) .. " ")
908 elseif name == "EMIT" then
909 if a < 32 or a > 126 then
910 problem(raw, "invalid argument", -24)
911 end
912 self:write(string.char(a))
913 elseif name == "SPACES" then
914 if a > 0 then
915 self:write(string.rep(" ", a))
916 end
917 elseif name == "!" then
918 self.memory[self:address(b, raw)] = a
919 elseif name == "@" then
920 push(self.memory[self:address(a, raw)])
921 elseif name == "+!" then
922 local address = self:address(b, raw)
923 self.memory[address] = self.memory[address] + a
924 elseif name == "," then
925 self.memory[#self.memory + 1] = a
926 elseif name == "ALLOT" then
927 if a < 0 or a % 8 ~= 0 then
928 problem(raw, "invalid argument", -24)
929 end
930 for _ = 1, a // 8 do
931 self.memory[#self.memory + 1] = 0
932 end
933 elseif name == "CELLS" then
934 push(8 * a)
935 elseif name == "CELL+" then
936 push(a + 8)
937 end
938end
939function Machine:process_line(line)
940 self.line, self.position = line, 1
941 local ok, caught =
942 pcall(
943 function()
944 while true do
945 local raw = self:read()
946 if not raw then
947 break
948 end
949 self:process(raw)
950 end
951 end
952 )
953 if not ok then
954 if type(caught) ~= "table" or not caught.forth then
955 error(caught, 0)
956 end
957 if #self.pending > 0 then
958 self:flush()
959 end
960 output[#output + 1] = "ERROR " .. caught.word .. ": " .. caught.reason
961 self.data, self.returns, self.compiling, self.controls, self.depth = {}, {}, nil, {}, 0
962 elseif #self.pending > 0 then
963 self:flush()
964 end
965end
966
967local machine = Machine.new()
968for line in io.lines() do
969 if line == "BYE" then
970 if machine.compiling then
971 output[#output + 1] =
972 "ERROR " .. machine.compiling.raw_name .. ": unterminated definition"
973 end
974 break
975 end
976 machine:process_line(line)
977end
978if #output > 0 then
979 io.write(table.concat(output, "\n"), "\n")
980end
