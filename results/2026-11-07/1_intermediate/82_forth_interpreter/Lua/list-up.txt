function Machine:take(count, raw)
    if #self.data < count then
        problem(raw, "stack underflow", -4)
    end
    local result, start = {}, #self.data - count
    for i = 1, count do
        result[i] = self.data[start + i]
    end
    for i = #self.data, start + 1, -1 do
        self.data[i] = nil
    end
    return result
