611processToken =: 3 : 0
612 token=.y
613 name=.toupper token
614 compiling=.CurrentDef>:0
615 if. name-:,'\' do. Position=:#InputLine
616 elseif. name-:,'(' do.
617 tail=.Position}.InputLine
618 at=.tail i.')'
619 if. at=#tail do. uncaught 'unterminated comment' end.
620 Position=:Position+at+1
621 elseif. (-.compiling)*.(<name)e.CompileOnly do. uncaught 'compile only'
622 elseif. compiling*.(<name)e.InterpretOnly do. uncaught 'interpret only'
623 elseif. (<name)e.('.';'ABORT'),&.><DQ do.
624 text=.readString''
625 if. compiling do. name emitText (<token),<text else. Pending=:Pending,text end.
626 elseif. (<name)e.words 'TO IS' do.
627 raw=.readName''
628 key=.toupper raw
629 if. compiling*.name-:'TO' do.
630 slot=.localIndex key
631 if. slot>:0 do. 'to_local'emitArgument (<token),<slot return. end.
632 end.
633 id=.existingWord raw
634 if. name-:'TO' do. expected=.'value'
635 else. expected=.'defer' end.
636 if. -.expected-:1{::id{::WordPool do.
637 if. name-:'TO' do. reason=.'not a value' else. reason=.'not deferred' end.
638 fail (<reason),(<0),(<0),<raw
639 end.
640 if. compiling do.
641 if. name-:'TO' do. op=.'to_value' else. op=.'IS' end.
642 op emitArgument (<token),<id
643 elseif. name-:'TO' do. value=.popData''
644 index=.<.(3{::id{::WordPool)%8
645 Memory=:value index}Memory
646 else. setDeferred id end.
647 elseif. (<name)e.words ': :NONAME' do.
648 if. name-:,':' do. raw=.readName''
649 checkName raw
650 key=.toupper raw
651 else. raw=.token
652 key=.'' end.
653 CurrentDef=:#Definitions
654 definition=.(<key),(<raw),(<hereAddress''),(<0$a:),(<0$a:),<0$a:
655 Definitions=:Definitions,<definition
656 elseif. name-:'FORGET' do.
657 raw=.readName''
658 key=.toupper raw
659 for_i. |.i.#Dictionary do.
660 id=.i{Dictionary
661 if. key-:0{::id{::WordPool do.
662 Memory=:(<.(5{::id{::WordPool)%8){.Memory
663 Dictionary=:i{.Dictionary
664 return.
665 end.
666 end.
667 fail (<'undefined word'),(<_13),(<1),<raw
668 elseif. compiling do. compileToken (<token),<name
669 else.
670 id=.findWord name
671 if. id<0 do. pushData literal token else. id executeWord token end.
672 end.
673)
674processLine =: 3 : 0
675 InputLine=:y
676 Position=:0
677 token=.''
678 try.
679 while. 1 do. token=.readWord''
680 if. 0=#token do. break. end.
681 processToken token
682 end.
683 catcht.
684 if. #Pending do. flushLine'' end.
685 if. 0=#ErrorWord do. ErrorWord=:token end.
686 say 'ERROR ',ErrorWord,': ',ErrorReason
687 Stack=:Returns=:i.0
688 Frames=:0$a:
689 NestingDepth=:0
690 CurrentDef=:_1
691 end.
692 if. #Pending do. flushLine'' end.
693)
694main =: 3 : 0
695 Stack=:Returns=:Dictionary=:Xts=:i.0
696 Memory=:,10
697 WordPool=:Definitions=:Frames=:0$a:
698 for_name. BuiltinNames do. WordPool=:WordPool,< (>name)newWord 'builtin' end.
699 CurrentDef=:_1
700 NestingDepth=:0
701 Pending=:''
702 lines=.<;._2 (1!:1 <'/dev/stdin'),LF
703 for_line. lines do.
704 line=.>line
705 if. line-:'BYE' do.
706 if. CurrentDef>:0 do. say 'ERROR ',(1{::CurrentDef{::Definitions),': unterminated definition' end.
707 break.
708 end.
709 processLine line
710 end.
