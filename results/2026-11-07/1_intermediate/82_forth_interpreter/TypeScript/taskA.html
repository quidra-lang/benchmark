17type Action =

130 number(raw: string, radix = this.memory[0]): bigint | null {
131 if (raw.length === 3 && raw[0] === "'" && raw[2] === "'") return BigInt(raw.charCodeAt(1));
132 let text = upper(raw);
133 let base: number;
134 if ("$#%".includes(text[0])) {
135 base = text[0] === "$" ? 16 : text[0] === "#" ? 10 : 2;
136 text = text.slice(1);
137 } else {
138 if (radix < 2n || radix > 36n) fail(raw, "invalid base", -40n);
139 base = Number(radix);
140 }
141 const negative = text.startsWith("-");
142 if (negative) text = text.slice(1);
143 if (text === "") return null;
144 let value = 0n;
145 for (const character of text) {
146 const code = character.charCodeAt(0),
147 digit =
148 code >= 48 && code <= 57
149 ? code - 48
150 : code >= 65 && code <= 90
151 ? code - 55
152 : -1;
153 if (digit < 0 || digit >= base) return null;
154 value = value * BigInt(base) + BigInt(digit);
155 }
156 return negative ? -value : value;

212 define(kind: string, raw: string): void {
213 const name = this.name(raw);
214 this.checkName(name);
215 const value = kind === "CONSTANT" || kind === "VALUE" ? this.pop(1, raw)[0] : 0n;
216 const here = this.memory.length * 8,

222 execute(word: Word, raw: string): void {
223 while (word.kind === "defer") {
224 const token = word.deferred;
225 if (token === null) fail(raw, "deferred word not set", -21n);
226 if (!this.validToken(token)) fail(raw, "invalid argument", -24n);
227 word = this.tokens[Number(token - 1n)];
228 raw = word.name;
229 }
230 switch (word.kind) {
231 case "builtin":
232 this.builtin(word.name, raw);
233 return;
234 case "colon":
235 this.definition(word, 0, raw);
236 return;
237 case "constant":
238 this.push(word.value, raw);
239 return;
240 case "value":
241 this.push(this.memory[word.address / 8], raw);
242 return;
243 case "create":
244 case "variable":
245 this.push(BigInt(word.address), raw);
246 if (word.does !== null) this.definition(word.does.definition, word.does.start, raw);
247 return;
248 }

250 builtin(name: string, raw: string): void {
251 if (["CREATE", "VARIABLE", "CONSTANT", "VALUE", "DEFER"].includes(name)) {
252 this.define(name, raw);
253 return;
254 }
255 if (name === "'") {

265 }

280 return;

285 if (name === "THROW") {

288 return;
289 }

546 definition(definition: Word, start: number, raw: string): void {
547 if (this.depth === 100) fail(raw, "nesting too deep", -7n);
548 this.depth++;
549 const returns = this.returns.length;
550 const locals: bigint[] = Array(definition.localCount).fill(0n);
551 for (let pc = start; pc < definition.actions.length; pc++) {
552 const action = definition.actions[pc],
553 word = action.raw;
554 switch (action.kind) {
555 case "literal":
556 this.push(action.value, word);
557 break;
558 case "call":
559 this.execute(action.word, word);
560 break;
561 case "write":
562 this.pending += action.text;
563 break;
564 case "abort":
565 if (this.pop(1, word)[0] !== 0n) fail(word, action.text, -2n);
566 break;
567 case "exit":
568 pc = definition.actions.length;
569 break;
570 case "recurse":
571 this.definition(definition, 0, word);
572 break;
573 case "does": {
574 const newest = this.dictionary.at(-1);
575 if (newest === undefined || newest.kind !== "create")
576 fail(word, "no CREATE word", -31n);
577 newest.does = { definition, start: pc + 1 };
578 pc = definition.actions.length;
579 break;
580 }
581 case "local-get":
582 this.push(locals[action.slot], word);
583 break;
584 case "local-set":
585 locals[action.slot] = this.pop(1, word)[0];
586 break;
587 case "locals": {
588 const values = this.pop(action.slots.length, word);
589 action.slots.forEach((slot, i) => {
590 locals[slot] = values[i];
591 });
592 break;
593 }
594 case "value-set":
595 this.memory[action.word.address / 8] = this.pop(1, word)[0];
596 break;
597 case "defer-set": {
598 const value = this.pop(1, word)[0];
599 if (!this.validToken(value)) fail(word, "invalid argument", -24n);
600 action.word.deferred = value;
601 break;
602 }
603 case "branch":
604 if (this.pop(1, word)[0] === 0n) pc = action.target - 1;
605 break;
606 case "jump":
607 pc = action.target - 1;
608 break;
609 case "loop-enter": {
610 const [limit, index] = this.pop(2, word);
611 if (action.conditional && limit === index) pc = action.target - 1;
612 else {
613 this.pushReturn(limit, word);
614 this.pushReturn(index, word);
615 }
616 break;
617 }
618 case "loop-next": {
619 const step = action.increment ? 1n : this.pop(1, word)[0];
620 this.needReturns(2, word);
621 const index = this.returns.at(-1)!,
622 limit = this.returns.at(-2)!,
623 next = index + step;
624 if (index - limit < 0n !== next - limit < 0n) this.returns.length -= 2;
625 else {
626 this.returns[this.returns.length - 1] = next;
627 pc = action.target - 1;
628 }
629 break;
630 }
631 case "leave":
632 this.needReturns(2, word);
633 this.returns.length -= 2;
634 pc = action.target - 1;
635 break;
636 case "of": {
637 const [a, b] = this.pop(2, word);
638 if (a !== b) {
639 this.push(a, word);
640 pc = action.target - 1;
641 }
642 break;
643 }
644 case "endcase":
645 this.pop(1, word);
646 break;
647 case "return":
648 if (action.operation === ">R") this.pushReturn(this.pop(1, word)[0], word);
649 else {
650 const count =
651 action.operation === "J" ? 3 : action.operation === "UNLOOP" ? 2 : 1;
652 this.needReturns(count, word);
653 if (action.operation === "UNLOOP") this.returns.length -= 2;
654 else if (action.operation === "R>") this.push(this.returns.pop()!, word);
655 else this.push(this.returns[this.returns.length - count], word);
656 }
657 break;
658 }
659 }
660 if (this.returns.length > returns) this.returns.length = returns;
661 this.depth--;
662 }

664 const index = this.compiling!.actions.length;
665 this.compiling!.actions.push(action);

671 compile(raw: string): void {
672 const name = upper(raw),
673 definition = this.compiling!,
674 top = this.control.at(-1),
675 index = definition.actions.length;
676 if (name === ";") {
677 if (top !== undefined) {
678 const kind = top.kind === "ELSE" ? "IF" : top.kind === "WHILE" ? "BEGIN" : top.kind;
679 fail(raw, "unclosed " + kind);
680 }
681 this.compiling = null;
682 if (definition.anonymous) this.push(this.xt(definition), raw);
683 else this.dictionary.push(definition);
684 return;
685 }
686 if (name === "{:") {
687 const words: string[] = [];
688 let closed = false;
689 let word: string | null;
690 while ((word = this.read()) !== null) {
691 if (upper(word) === ":}") {
692 closed = true;
693 break;
694 }
695 words.push(word);
696 }
697 if (!closed) fail(raw, "unterminated locals");
698 const marker = words.indexOf("--"),
699 names = marker < 0 ? words : words.slice(0, marker);
700 const slots: number[] = [];
701 for (const name of names) this.checkName(name);
702 for (const name of names) {
703 const slot = definition.localCount++;
704 definition.locals.set(upper(name), slot);
705 slots.push(slot);
706 }
707 this.emit({ kind: "locals", raw, slots });
708 return;
709 }
710 if (name === "[']") {
711 const name = this.name(raw),
712 word = this.lookup(name);
713 if (word === null) fail(name, "undefined word", -13n);
714 this.emit({ kind: "literal", raw, value: this.xt(word) });
715 return;
716 }
717 if (name === "DOES>") {
718 if (this.control.length) this.mismatch(raw);
719 this.emit({ kind: "does", raw });
720 return;
721 }
722 if (name === "EXIT" || name === "RECURSE") {
723 this.emit({ kind: name === "EXIT" ? "exit" : "recurse", raw });
724 return;
725 }
726 if ([">R", "R>", "R@", "I", "J", "UNLOOP"].includes(name)) {
727 this.emit({ kind: "return", raw, operation: name });
728 return;
729 }
730 if (name === "IF") {
731 this.control.push({ kind: "IF", index: this.emit({ kind: "branch", raw, target: 0 }) });
732 return;
733 }
734 if (name === "ELSE") {
735 if (top === undefined || top.kind !== "IF") this.mismatch(raw);
736 this.emit({ kind: "jump", raw, target: 0 });
737 (definition.actions[top.index] as TargetedAction).target = index + 1;
738 this.control[this.control.length - 1] = { kind: "ELSE", index };
739 return;
740 }
741 if (name === "THEN") {
742 if (top === undefined || (top.kind !== "IF" && top.kind !== "ELSE")) this.mismatch(raw);
743 this.emit({ kind: "noop", raw });
744 (definition.actions[top.index] as TargetedAction).target = index + 1;
745 this.control.pop();
746 return;
747 }
748 if (name === "BEGIN") {
749 this.emit({ kind: "noop", raw });
750 this.control.push({ kind: "BEGIN", index });
751 return;
752 }
753 if (name === "UNTIL" || name === "AGAIN") {
754 if (top === undefined || top.kind !== "BEGIN") this.mismatch(raw);
755 this.emit({ kind: name === "UNTIL" ? "branch" : "jump", raw, target: top.index + 1 });
756 this.control.pop();
757 return;
758 }
759 if (name === "WHILE") {
760 if (top === undefined || top.kind !== "BEGIN") this.mismatch(raw);
761 this.emit({ kind: "branch", raw, target: 0 });
762 this.control[this.control.length - 1] = { kind: "WHILE", index, begin: top.index };
763 return;
764 }
765 if (name === "REPEAT") {
766 if (top === undefined || top.kind !== "WHILE") this.mismatch(raw);
767 this.emit({ kind: "jump", raw, target: top.begin + 1 });
768 (definition.actions[top.index] as TargetedAction).target = index + 1;
769 this.control.pop();
770 return;
771 }
772 if (name === "DO" || name === "?DO") {
773 this.emit({ kind: "loop-enter", raw, conditional: name === "?DO", target: 0 });
774 this.control.push({ kind: name, index, leaves: [] });
775 return;
776 }
777 if (name === "LOOP" || name === "+LOOP") {
778 if (top === undefined || (top.kind !== "DO" && top.kind !== "?DO")) this.mismatch(raw);
779 this.emit({
780 kind: "loop-next",
781 raw,
782 increment: name === "LOOP",
783 target: top.index + 1,
784 });
785 (definition.actions[top.index] as TargetedAction).target = index + 1;
786 for (const leave of top.leaves)
787 (definition.actions[leave] as TargetedAction).target = index + 1;
788 this.control.pop();
789 return;
790 }
791 if (name === "LEAVE") {
792 const loop = [...this.control]
793 .reverse()
794 .find(
795 (record): record is Extract<Control, { leaves: number[] }> =>
796 record.kind === "DO" || record.kind === "?DO",
797 );
798 if (loop === undefined) this.mismatch(raw);
799 loop.leaves.push(this.emit({ kind: "leave", raw, target: 0 }));
800 return;
801 }
802 if (name === "CASE") {
803 this.emit({ kind: "noop", raw });
804 this.control.push({ kind: "CASE", index, endofs: [] });
805 return;
806 }
807 if (name === "OF") {
808 if (top === undefined || top.kind !== "CASE") this.mismatch(raw);
809 this.emit({ kind: "of", raw, target: 0 });
810 this.control.push({ kind: "OF", index });
811 return;
812 }
813 if (name === "ENDOF") {
814 if (top === undefined || top.kind !== "OF") this.mismatch(raw);
815 this.emit({ kind: "jump", raw, target: 0 });
816 (definition.actions[top.index] as TargetedAction).target = index + 1;
817 this.control.pop();
818 (this.control.at(-1)! as Extract<Control, { kind: "CASE" }>).endofs.push(index);
819 return;
820 }
821 if (name === "ENDCASE") {
822 if (top === undefined || top.kind !== "CASE") this.mismatch(raw);
823 this.emit({ kind: "endcase", raw });
824 for (const endof of top.endofs)
825 (definition.actions[endof] as TargetedAction).target = index + 1;
826 this.control.pop();
827 return;
828 }
829 if (definition.locals.has(name)) {
830 this.emit({ kind: "local-get", raw, slot: definition.locals.get(name)! });
831 return;
832 }
833 const word = this.lookup(name);
834 if (word !== null) {
835 this.emit({ kind: "call", raw, word });
836 return;
837 }
838 const value = this.number(raw);
839 if (value === null) fail(raw, "undefined word", -13n);
840 this.emit({ kind: "literal", raw, value });
841 }
842 assign(raw: string, kind: string): void {
843 const name = this.name(raw),
844 normalized = upper(name);
845 if (kind === "TO" && this.compiling !== null && this.compiling.locals.has(normalized)) {
846 this.emit({ kind: "local-set", raw, slot: this.compiling.locals.get(normalized)! });
847 return;
848 }
849 const word = this.lookup(name);
850 if (word === null) fail(name, "undefined word", -13n);
851 if (word.kind !== (kind === "TO" ? "value" : "defer"))
852 fail(name, kind === "TO" ? "not a value" : "not deferred");
853 if (this.compiling !== null)
854 this.emit({ kind: kind === "TO" ? "value-set" : "defer-set", raw, word });
855 else {
856 const value = this.pop(1, raw)[0];
857 if (kind === "TO") this.memory[word.address / 8] = value;
858 else {
859 if (!this.validToken(value)) fail(raw, "invalid argument", -24n);
860 word.deferred = value;
861 }
862 }
863 }
864 process(raw: string): void {
865 const name = upper(raw);
866 if (name === "\\") {
867 this.position = this.line.length;
868 return;
869 }
870 if (name === "(") {
871 const end = this.line.indexOf(")", this.position);
872 if (end < 0) fail(raw, "unterminated comment");
873 this.position = end + 1;
874 return;
875 }
876 if (this.compiling === null && compileOnly.has(name)) fail(raw, "compile only");
877 if (this.compiling !== null && interpretOnly.has(name)) fail(raw, "interpret only");
878 if (name === '."' || name === 'ABORT"') {
879 const start = this.position + 1,
880 end = this.line.indexOf('"', start);
881 if (this.position === this.line.length || end < 0) fail(raw, "unterminated string");
882 const text = this.line.slice(start, end);
883 this.position = end + 1;
884 if (this.compiling === null) this.pending += text;
885 else this.emit({ kind: name === '."' ? "write" : "abort", raw, text });
886 return;
887 }
888 if (name === "TO" || name === "IS") {
889 this.assign(raw, name);
890 return;
891 }
892 if (name === ":" || name === ":NONAME") {
893 const wordName = name === ":" ? this.name(raw) : raw;
894 if (name === ":") this.checkName(wordName);
895 this.compiling = newWord(
896 name === ":" ? upper(wordName) : ":NONAME",
897 "colon",
898 this.memory.length * 8,
899 );
900 this.compiling.rawName = wordName;
901 this.compiling.anonymous = name === ":NONAME";
902 this.control = [];
903 return;
904 }
905 if (name === "FORGET") {
906 const name = this.name(raw),
907 normalized = upper(name);
908 let index = this.dictionary.length - 1;
909 while (index >= 0 && this.dictionary[index].name !== normalized) index--;
910 if (index < 0) fail(name, "undefined word", -13n);
911 this.memory.length = this.dictionary[index].here / 8;
912 this.dictionary.length = index;
913 return;
914 }
915 if (this.compiling !== null) {
916 this.compile(raw);
917 return;
918 }
919 const word = this.lookup(name);
920 if (word !== null) {
921 this.execute(word, raw);
922 return;
923 }
924 const value = this.number(raw);
925 if (value === null) fail(raw, "undefined word", -13n);
926 this.push(value, raw);
927 }
928 run(line: string): void {
929 this.line = line;
930 this.position = 0;
931 try {
932 let word: string | null;
933 while ((word = this.read()) !== null) this.process(word);
934 } catch (error) {
935 if (!(error instanceof ForthError)) throw error;
936 if (this.pending.length) this.flush();
937 this.output.push(`ERROR ${error.word}: ${error.message}`);
938 this.stack = [];
939 this.returns = [];
940 this.depth = 0;
941 this.compiling = null;
942 this.control = [];
943 }
944 if (this.pending.length) this.flush();
945 }
946}
