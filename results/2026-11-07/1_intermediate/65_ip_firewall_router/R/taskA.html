24address <- function(word) {
25 if (!grepl("^(0|[1-9][0-9]{0,2})(\\.(0|[1-9][0-9]{0,2})){3}$", word)) fail(paste0("bad address ", word))
26 parts <- as.numeric(strsplit(word, ".", fixed = TRUE)[[1]])
27 if (any(parts > 255)) fail(paste0("bad address ", word))
28 sum(parts * 256^(3:0))
29}
30ip <- function(number) paste(floor(number / 256^(3:0)) %% 256, collapse = ".")
31network <- function(number, length) floor(number / 2^(32 - length)) * 2^(32 - length)
32parse_prefix <- function(word, hosts = FALSE) {
33 parts <- strsplit(paste0(word, "!"), "/", fixed = TRUE)[[1]]
34 parts[length(parts)] <- sub("!$", "", parts[length(parts)])
35 parsed <- tryCatch({
36 if (length(parts) != 2 || !grepl("^(0|[1-9][0-9]?)$", parts[2]) || as.numeric(parts[2]) > 32) fail("invalid")
37 c(address(parts[1]), as.numeric(parts[2]))
38 }, router_error = function(e) NULL)
39 if (is.null(parsed)) fail(paste0("bad prefix ", word))
40 if (!hosts && parsed[1] != network(parsed[1], parsed[2])) fail(paste0("host bits set in ", word))
41 parsed
42}

64port_range <- function(word) {
65 if (word == "ANY") return(list(low = 0, high = 65535, any = TRUE))
66 parts <- as.numeric(strsplit(word, "-", fixed = TRUE)[[1]])
67 low <- parts[1]
68 high <- parts[length(parts)]
69 if (low > 65535 || high > 65535 || low > high) fail(paste0("bad port range ", word))
70 list(low = low, high = high, any = FALSE)
71}
72range_text <- function(range) {
73 if (range$any) "any" else if (range$low == range$high) fmt(range$low) else paste0(range$low, "-", range$high)

77evaluate_acl <- function(name, direction, protocol, source, sport, destination, dport, established) {
78 pair <- interfaces[[name]]$acl[[direction]]
79 if (!length(pair$rules)) return(NULL)
80 rules <- pair$rules[order(vapply(pair$rules, function(rule) rule$seq, numeric(1)))]
81 for (rule in rules) {
82 if (!(rule$proto %in% c("ANY", protocol)) || !contains(rule$src, source) || !contains(rule$dst, destination) ||
83 (rule$proto %in% c("TCP", "UDP") && (sport < rule$sports$low || sport > rule$sports$high || dport < rule$dports$low || dport > rule$dports$high)) ||
84 (rule$established && !established)) next
85 rule$hits <- rule$hits + 1
86 if (rule$action == "PERMIT") return(NULL)
87 if (rule$action == "DENY") return(paste0("DROP acl ", name, " ", direction, " ", rule$seq))
88 return(paste0("REJECT acl ", name, " ", direction, " ", rule$seq, " ", if (protocol == "TCP") "tcp-reset" else "icmp-prohibited"))
89 }
90 pair$default <- pair$default + 1
91 paste0("DROP acl ", name, " ", direction, " default")
92}
93packet <- function(name, protocol, flags, source, sport, destination, dport, ttl) {
94 input <- interfaces[[name]]
95 if (!input$up) return(c("DROP down", "down"))
96 for (number in c(source, destination)) {
97 if (contains(c(0, 8), number) || contains(c(127 * 2^24, 8), number) || contains(c(224 * 2^24, 3), number)) return(c("DROP martian", "martian"))
98 }
99 if (!is.null(local_name(source))) return(c("DROP martian", "martian"))
100 if (input$rpf != "off") {
101 reverse <- select_route(source)
102 if (is.null(reverse) || (input$rpf == "strict" && reverse$dev != name)) return(c("DROP rpf", "rpf"))
103 }
104 tuple <- list(protocol, source, sport, destination, dport)
105 pure_syn <- protocol == "TCP" && flags == "S"
106 connection <- find_item(connections, function(item) tuple_equal(item$orig, tuple) || tuple_equal(item$reply, tuple))
107 direction <- if (is.null(connection)) NULL else if (tuple_equal(connection$orig, tuple)) "orig" else "reply"
108 if (!is.null(connection) && connection$state == "CLOSED" && pure_syn) {
109 connections <<- Filter(function(item) !identical(item, connection), connections)
110 connection <- NULL
111 }
112 if (protocol == "TCP" && is.null(connection) && !pure_syn) return(c("DROP invalid", "invalid"))
113 verdict <- evaluate_acl(name, "in", protocol, source, sport, destination, dport, !is.null(connection))
114 if (!is.null(verdict)) return(c(verdict, "acl_in"))
115 translated_destination <- destination
116 translated_dport <- dport
117 mapping <- NULL
118 if (input$role == "outside") {
119 static <- find_item(static_nat, function(item) item$outside == destination)
120 forward <- if (destination == input$address) find_item(forwards, function(item) item$dev == name && item$proto == protocol && item$port == dport) else NULL
121 if (!is.null(static)) {
122 translated_destination <- static$inside
123 } else if (!is.null(forward)) {
124 translated_destination <- forward$inside
125 translated_dport <- forward$inport
126 } else if (destination == input$address) {
127 mapping <- find_item(pat, function(item) item$dev == name && item$proto == protocol && item$port == dport)
128 if (!is.null(mapping)) {
129 translated_destination <- mapping$inside
130 translated_dport <- mapping$inport
131 }
132 }
133 }
134 local <- local_name(translated_destination)
135 if (!is.null(local)) return(c(paste0("LOCAL ", local), "local"))
136 if (ttl == 1) return(c("DROP ttl", "ttl"))
137 selected <- select_route(translated_destination)
138 if (is.null(selected)) return(c("DROP noroute", "noroute"))
139 output_name <- selected$dev
140 output <- interfaces[[output_name]]
141 verdict <- evaluate_acl(output_name, "out", protocol, source, sport, translated_destination, translated_dport, !is.null(connection))
142 if (!is.null(verdict)) return(c(verdict, "acl_out"))
143 translated_source <- source
144 translated_sport <- sport
145 if (input$role == "inside" && output$role == "outside") {
146 static <- find_item(static_nat, function(item) item$inside == source)
147 forward <- find_item(forwards, function(item) item$dev == output_name && item$proto == protocol && item$inside == source && item$inport == sport)
148 if (!is.null(static)) {
149 translated_source <- static$outside
150 } else if (!is.null(forward)) {
151 translated_source <- output$address
152 translated_sport <- forward$port
153 } else if (!is.null(output$pool)) {
154 mapping <- find_item(pat, function(item) item$dev == output_name && item$proto == protocol && item$inside == source && item$inport == sport)
155 if (is.null(mapping)) {
156 used <- vapply(Filter(function(item) item$dev == output_name, c(pat, forwards)), function(item) item$port, numeric(1))
157 available <- setdiff(seq(20000, 19999 + output$pool), used)
158 if (!length(available)) return(c("DROP exhausted", "exhausted"))
159 mapping <- object(dev = output_name, proto = protocol, inside = source, inport = sport, port = available[1], last = clock)
160 pat[[length(pat) + 1]] <<- mapping
161 }
162 translated_source <- output$address
163 translated_sport <- mapping$port
164 }
165 }
166 if (!is.null(mapping)) mapping$last <- clock
167 if (is.null(connection)) {
168 connection_number <<- connection_number + 1
169 connection <- object(number = connection_number, proto = protocol, orig = tuple,
170 reply = list(protocol, translated_destination, translated_dport, translated_source, translated_sport),
171 input = name, output = output_name, state = if (protocol == "TCP") "SYN_SENT" else "UNREPLIED",
172 orig_packets = 0, reply_packets = 0, orig_fin = FALSE, reply_fin = FALSE, last = clock)
173 connections[[length(connections) + 1]] <<- connection
174 direction <- "new"
175 }
176 side <- if (direction == "reply") "reply" else "orig"
177 packet_field <- paste0(side, "_packets")
178 connection[[packet_field]] <- connection[[packet_field]] + 1
179 connection$last <- clock
180 if (protocol != "TCP") {
181 if (direction == "reply") connection$state <- "REPLIED"
182 } else if (grepl("R", flags, fixed = TRUE)) {
183 connection$state <- "CLOSED"
184 } else {
185 if (connection$state == "SYN_SENT" && direction == "reply" && grepl("S", flags, fixed = TRUE) && grepl("A", flags, fixed = TRUE)) connection$state <- "ESTABLISHED"
186 if (grepl("F", flags, fixed = TRUE) && connection$state %in% c("ESTABLISHED", "FIN_WAIT")) {
187 connection[[paste0(side, "_fin")]] <- TRUE
188 connection$state <- if (connection$orig_fin && connection$reply_fin) "CLOSED" else "FIN_WAIT"
189 }
190 }
191 output$tx <- output$tx + 1
192 selected$hits <- selected$hits + 1
193 next_hop <- if (is.null(selected$hop)) "direct" else paste0("via ", ip(selected$hop))
194 text <- paste0("FORWARD ", output_name, " ", next_hop, " ttl=", ttl - 1)
195 if (translated_destination != destination || translated_dport != dport) text <- paste0(text, " dnat=", ip(translated_destination), ":", translated_dport)
196 if (translated_source != source || translated_sport != sport) text <- paste0(text, " snat=", ip(translated_source), ":", translated_sport)
197 c(paste0(text, " conn=", connection$number, " ", direction), "forwarded")
198}

212show_state <- function(what) {
213 sorted_names <- sort(names(interfaces), method = "radix")
214 if (what == "IFACES") {
215 emit("show ifaces count=", length(interfaces))
216 for (name in sorted_names) {
217 item <- interfaces[[name]]
218 text <- paste0(name, " ", ip(item$address), "/", item$length, " ", if (item$up) "up" else "down", " ", item$role, " rpf=", item$rpf)
219 if (!is.null(item$pool)) text <- paste0(text, " pat=", sum(vapply(pat, function(mapping) mapping$dev == name, logical(1))), "/", item$pool)
220 emit(text)
221 }
222 } else if (what == "ROUTES") {
223 emit("show routes count=", length(routes))
224 ordering <- order(vapply(routes, function(item) item$prefix[1], numeric(1)), vapply(routes, function(item) item$prefix[2], numeric(1)), vapply(routes, function(item) item$metric, numeric(1)), seq_along(routes), method = "radix")
225 for (item in routes[ordering]) emit(route_text(item), " hits=", item$hits, if (!interfaces[[item$dev]]$up) " inactive" else "")
226 } else if (what == "ACL") {
227 total <- sum(vapply(interfaces, function(item) sum(vapply(item$acl, function(pair) length(pair$rules), integer(1))), numeric(1)))
228 emit("show acl count=", total)
229 for (name in sorted_names) {
230 for (direction in c("in", "out")) {
231 pair <- interfaces[[name]]$acl[[direction]]
232 if (!length(pair$rules)) next
233 ordering <- order(vapply(pair$rules, function(rule) rule$seq, numeric(1)))
234 for (rule in pair$rules[ordering]) emit("acl ", name, " ", direction, " ", rule$seq, " ", rule$action, " ", rule$proto, " ", prefix_text(rule$src), " ", range_text(rule$sports), " ", prefix_text(rule$dst), " ", range_text(rule$dports), if (rule$established) " established" else "", " hits=", rule$hits)
235 emit("acl ", name, " ", direction, " default DENY hits=", pair$default)
236 }
237 }
238 } else if (what == "NAT") {
239 emit("show nat static=", length(static_nat), " forward=", length(forwards), " pat=", length(pat))
240 for (item in static_nat[order(vapply(static_nat, function(item) item$inside, numeric(1)))]) emit("static ", ip(item$inside), " ", ip(item$outside))
241 ordering <- order(vapply(forwards, function(item) item$dev, character(1)), vapply(forwards, function(item) item$port, numeric(1)), vapply(forwards, function(item) item$proto, character(1)), method = "radix")
242 for (item in forwards[ordering]) emit("forward ", item$dev, " ", item$proto, " ", item$port, " ", ip(item$inside), ":", item$inport)
243 ordering <- order(vapply(pat, function(item) item$dev, character(1)), vapply(pat, function(item) item$port, numeric(1)), method = "radix")
244 for (item in pat[ordering]) emit("pat ", item$dev, " ", item$port, " ", item$proto, " ", ip(item$inside), ":", item$inport, " idle=", fmt(clock - item$last))
245 } else if (what == "CONNS") {
246 emit("show conns count=", length(connections))
247 for (item in connections) {
248 original <- item$orig
249 reply <- item$reply
250 emit("conn ", item$number, " ", item$proto, " ", ip(original[[2]]), ":", original[[3]], " > ", ip(original[[4]]), ":", original[[5]], " reply ", ip(reply[[2]]), ":", reply[[3]], " > ", ip(reply[[4]]), ":", reply[[5]], " in=", item$input, " out=", item$output, " state=", item$state, " pkts=", item$orig_packets, "/", item$reply_packets, " idle=", fmt(clock - item$last))
251 }
252 } else if (what == "COUNTERS") {
253 dropped <- sum(counts[drop_reasons])
254 emit("show counters packets=", counts["forwarded"] + counts["local"] + dropped, " forwarded=", counts["forwarded"], " local=", counts["local"], " dropped=", dropped, " errors=", counts["errors"])
255 emit("drops ", paste(paste0(drop_reasons, "=", counts[drop_reasons]), collapse = " "))
256 for (name in sorted_names) {
257 item <- interfaces[[name]]
258 emit("iface ", name, " rx=", item$rx, " tx=", item$tx, " drop=", item$drop)
259 }
260 }

262command <- function(words) {
263 kind <- words[1]
264 a <- words[-1]
265 name <- a[1]
266 if (kind == "IFACE") {
267 if (name %in% names(interfaces)) fail(paste0("duplicate interface ", name))
268 prefix <- parse_prefix(a[2], TRUE)
269 number <- prefix[1]
270 length <- prefix[2]
271 if (length < 8 || length > 30) fail(paste0("bad prefix ", a[2]))
272 base <- network(number, length)
273 if (number == base || number == base + 2^(32 - length) - 1) fail(paste0("bad host address ", a[2]))
274 for (other in names(interfaces)) {
275 item <- interfaces[[other]]
276 common <- min(length, item$length)
277 if (network(number, common) == network(item$address, common)) fail(paste0("subnet overlaps ", other))
278 }
279 interfaces[[name]] <<- object(address = number, length = length, up = TRUE, role = "inside", pool = NULL, rpf = "off", rx = 0, tx = 0, drop = 0,
280 acl = list("in" = object(rules = list(), default = 0), "out" = object(rules = list(), default = 0)))
281 routes[[length(routes) + 1]] <<- object(type = "C", prefix = c(base, length), dev = name, hop = NULL, metric = 0, hits = 0)
282 emit("ok iface ", name, " ", a[2], " network ", ip(base), "/", length)
283 } else if (kind %in% c("UP", "DOWN")) {
284 item <- iface(name)
285 state <- kind == "UP"
286 if (item$up == state) fail(paste0("interface ", name, " already ", if (state) "up" else "down"))
287 item$up <- state
288 emit("ok ", tolower(kind), " ", name)
289 } else if (kind == "ROUTE") {
290 prefix <- parse_prefix(name)
291 via <- a[2] == "VIA"
292 hop <- if (via) address(a[3]) else NULL
293 if (!via) iface(a[3])
294 metric <- as.numeric(a[4])
295 if (metric < 1 || metric > 255) fail(paste0("bad metric ", a[4]))
296 output <- a[3]
297 if (via) {
298 if (!is.null(local_name(hop))) fail(paste0("nexthop ", a[3], " is local"))
299 output <- NULL
300 for (key in names(interfaces)) {
301 item <- interfaces[[key]]
302 if (contains(c(item$address, item$length), hop)) output <- key
303 }
304 if (is.null(output)) fail(paste0("nexthop ", a[3], " unreachable"))
305 }
306 if (!is.null(find_item(routes, function(item) item$type == "S" && identical(item$prefix, prefix) && item$dev == output && identical(item$hop, hop)))) fail(paste0("duplicate route ", name))
307 item <- object(type = "S", prefix = prefix, dev = output, hop = hop, metric = metric, hits = 0)
308 routes[[length(routes) + 1]] <<- item
309 emit("ok route ", route_text(item))
310 } else if (kind == "NOROUTE") {
311 prefix <- parse_prefix(name)
312 remaining <- Filter(function(item) !(item$type == "S" && identical(item$prefix, prefix)), routes)
313 removed <- length(routes) - length(remaining)
314 if (removed == 0) fail(paste0("no static route ", name))
315 routes <<- remaining
316 emit("ok noroute ", name, " removed=", removed)
317 } else if (kind == "LOOKUP") {
318 selected <- select_route(address(name))
319 emit("lookup ", name, " ", if (is.null(selected)) "none" else route_text(selected))
320 } else if (kind == "ACL") {
321 item <- iface(name)
322 direction <- tolower(a[2])
323 sequence <- as.numeric(a[3])
324 if (sequence < 1 || sequence > 9999) fail(paste0("bad sequence ", a[3]))
325 source <- acl_prefix(a[6])
326 sports <- port_range(a[7])
327 destination <- acl_prefix(a[8])
328 dports <- port_range(a[9])
329 if (a[5] %in% c("ICMP", "ANY") && (a[7] != "ANY" || a[9] != "ANY")) fail(paste0("ports not allowed for ", a[5]))
330 pair <- item$acl[[direction]]
331 if (!is.null(find_item(pair$rules, function(rule) rule$seq == sequence))) fail(paste0("duplicate sequence ", a[3]))
332 pair$rules[[length(pair$rules) + 1]] <- object(seq = sequence, action = a[4], proto = a[5], src = source, sports = sports, dst = destination, dports = dports, established = length(a) == 10, hits = 0)
333 emit("ok acl ", name, " ", direction, " ", sequence)
334 } else if (kind == "NOACL") {
335 pair <- iface(name)$acl[[tolower(a[2])]]
336 rule <- find_item(pair$rules, function(item) item$seq == as.numeric(a[3]))
337 direction <- tolower(a[2])
338 if (is.null(rule)) fail(paste0("no rule ", a[3], " on ", name, " ", direction))
339 pair$rules <- Filter(function(item) !identical(item, rule), pair$rules)
340 emit("ok noacl ", name, " ", direction, " ", a[3], " hits=", rule$hits)
341 } else if (kind == "NAT") {
342 if (name == "STATIC") {
343 inside <- address(a[2])
344 outside <- address(a[3])
345 if (!is.null(local_name(inside))) fail(paste0("address ", a[2], " is local"))
346 if (!is.null(local_name(outside))) fail(paste0("address ", a[3], " is local"))
347 if (!is.null(find_item(static_nat, function(item) item$inside == inside))) fail(paste0("duplicate static ", a[2]))
348 if (!is.null(find_item(static_nat, function(item) item$outside == outside))) fail(paste0("duplicate static ", a[3]))
349 static_nat[[length(static_nat) + 1]] <<- object(inside = inside, outside = outside)
350 emit("ok nat static ", a[2], " ", a[3])
351 } else if (name == "OUTSIDE") {
352 item <- iface(a[2])
353 if (item$role == "outside") fail(paste0("interface ", a[2], " already outside"))
354 item$role <- "outside"
355 emit("ok nat outside ", a[2])
356 } else if (name == "PAT") {
357 item <- iface(a[2])
358 count <- as.numeric(a[3])
359 if (count < 1 || count > 1000) fail(paste0("bad pool size ", a[3]))
360 if (!is.null(item$pool)) fail(paste0("pat already on ", a[2]))
361 item$pool <- count
362 item$role <- "outside"
363 emit("ok nat pat ", a[2], " ", ip(item$address), " 20000-", 19999 + count)
364 } else {
365 item <- iface(a[2])
366 port <- as.numeric(a[4])
367 if (port < 1 || port > 65535) fail(paste0("bad port ", a[4]))
368 inside <- address(a[5])
369 inport <- as.numeric(a[6])
370 if (inport < 1 || inport > 65535) fail(paste0("bad port ", a[6]))
371 if (item$role != "outside") fail(paste0("interface ", a[2], " not outside"))
372 if (!is.null(local_name(inside))) fail(paste0("address ", a[5], " is local"))
373 if (!is.null(find_item(forwards, function(rule) rule$dev == a[2] && rule$proto == a[3] && rule$port == port))) fail(paste0("duplicate forward ", a[2], " ", a[3], " ", a[4]))
374 forwards[[length(forwards) + 1]] <<- object(dev = a[2], proto = a[3], port = port, inside = inside, inport = inport)
375 emit("ok nat forward ", a[2], " ", a[3], " ", port, " ", a[5], ":", inport)
376 }
377 } else if (kind == "RPF") {
378 item <- iface(name)
379 item$rpf <- tolower(a[2])
380 emit("ok rpf ", name, " ", item$rpf)
381 } else if (kind == "TIMEOUT") {
382 seconds <- as.numeric(a[2])
383 if (seconds < 1 || seconds > 86400) fail(paste0("bad timeout ", a[2]))
384 timeouts[name] <<- seconds
385 emit("ok timeout ", name, " ", seconds)
386 } else if (kind == "TICK") {
387 seconds <- as.numeric(name)
388 if (seconds < 1 || seconds > 86400) fail(paste0("bad tick ", name))
389 clock <<- clock + seconds
390 conns <- length(connections)
391 mappings <- length(pat)
392 connections <<- Filter(function(item) {
393 timeout <- if (item$proto == "TCP") switch(item$state, SYN_SENT = 60, FIN_WAIT = 60, CLOSED = 10, timeouts["TCP"]) else timeouts[item$proto]
394 clock - item$last < timeout
395 }, connections)
396 pat <<- Filter(function(item) clock - item$last < timeouts[item$proto], pat)
397 emit("ok tick time=", fmt(clock), " expired conns=", conns - length(connections), " nat=", mappings - length(pat))
398 } else if (kind == "PACKET") {
399 item <- iface(name)
400 protocol <- a[2]
401 flags <- a[3]
402 if (protocol != "TCP" && flags != "-") fail(paste0("flags not allowed for ", protocol))
403 if (protocol == "TCP" && flags != "-" && (!grepl("^[SAFR]+$", flags) || length(unique(strsplit(flags, "", fixed = TRUE)[[1]])) != nchar(flags))) fail(paste0("bad flags ", flags))
404 source <- address(a[4])
405 sport <- as.numeric(a[5])
406 if (sport > 65535) fail(paste0("bad port ", a[5]))
407 destination <- address(a[6])
408 dport <- as.numeric(a[7])
409 if (dport > 65535) fail(paste0("bad port ", a[7]))
410 ttl <- as.numeric(a[8])
411 if (ttl < 1 || ttl > 255) fail(paste0("bad ttl ", a[8]))
412 packet_number <<- packet_number + 1
413 item$rx <- item$rx + 1
414 result <- packet(name, protocol, flags, source, sport, destination, dport, ttl)
