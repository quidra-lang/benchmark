147static bool parse_address(const char *s, uint32_t *result) {
148 uint32_t address = 0;
149 for (int i = 0; i < 4; i++) {
150 int value = 0, digits = 0;
151 const char *start = s;
152 while (*s >= '0' && *s <= '9') {
153 value = value * 10 + *s++ - '0';
154 if (++digits > 3) return false;
155 }
156 if (!digits || (digits > 1 && *start == '0') || value > 255) return false;
157 address = (address << 8) | (uint32_t)value;
158 if (i < 3) {
159 if (*s++ != '.') return false;
160 } else if (*s) return false;
161 }
162 *result = address;
163 return true;
164}

166static uint32_t address(const char *text) {
167 uint32_t value;
168 if (!parse_address(text, &value)) error("bad address %s", text);
169 return value;
170}

172static bool parse_prefix(const char *s, Prefix *result) {
173 const char *slash = strchr(s, '/');
174 if (!slash || strchr(slash + 1, '/')) return false;
175 uint32_t value;
176 if (!parse_address(slice(s, (size_t)(slash - s)), &value)) return false;
177 const char *length = slash + 1;
178 size_t n = strlen(length);
179 if (!n || n > 2 || (n > 1 && *length == '0')) return false;
180 for (size_t i = 0; i < n; i++) if (length[i] < '0' || length[i] > '9') return false;
181 int bits = atoi(length);
182 if (bits > 32) return false;
183 *result = (Prefix){value, bits};
184 return true;
185}

187static Prefix prefix(const char *s, bool any) {
188 if (any && !strcmp(s, "ANY")) return (Prefix){0, 0};
189 Prefix p;
190 if (!parse_prefix(s, &p)) error("bad prefix %s", s);
191 if (p.address != network(p)) error("host bits set in %s", s);
192 return p;
193}

232static Route *select_route(uint32_t value) {
233 Route *best = NULL;
234 for (size_t i = 0; i < route_count; i++) {
235 Route *r = routes[i];
236 if (r->output->up && contains(r->prefix, value) && (!best || r->prefix.length > best->prefix.length || (r->prefix.length == best->prefix.length && r->metric < best->metric))) best = r;

257static char *acl(Interface *iface, int direction, Tuple packet, uint32_t source, int sport, uint32_t destination, int dport, bool established) {
258 if (!iface->acl_count[direction]) return NULL;
259 const char *dir = direction ? "out" : "in";
260 for (size_t i = 0; i < iface->acl_count[direction]; i++) {
261 Rule *r = iface->acl[direction][i];
262 if ((strcmp(r->protocol, "ANY") && strcmp(r->protocol, packet.protocol)) || !contains(r->source, source) || !contains(r->destination, destination) || (r->established && !established)) continue;
263 if ((!strcmp(r->protocol, "TCP") || !strcmp(r->protocol, "UDP")) && (sport < r->sports.first || sport > r->sports.last || dport < r->dports.first || dport > r->dports.last)) continue;
264 r->hits++;
265 if (!strcmp(r->action, "PERMIT")) return NULL;
266 bool reject = !strcmp(r->action, "REJECT");
267 return format("%s acl %s %s %d%s", reject ? "REJECT" : "DROP", iface->name, dir, r->sequence, reject ? (!strcmp(packet.protocol, "TCP") ? " tcp-reset" : " icmp-prohibited") : "");
268 }
269 iface->defaults[direction]++;
270 return format("DROP acl %s %s default", iface->name, dir);
271}

292static char *packet(Interface *input, Tuple original, const char *flags, int ttl) {
293 input->rx++;
294 if (!input->up) return drop(input, 0, "DROP down");
295 if (martian(original.source) || martian(original.destination) || local_address(original.source)) return drop(input, 1, "DROP martian");
296 if (strcmp(input->rpf, "off")) {
297 Route *r = select_route(original.source);
298 if (!r || (!strcmp(input->rpf, "strict") && r->output != input)) return drop(input, 2, "DROP rpf");
299 }
300 bool pure_syn = !strcmp(original.protocol, "TCP") && !strcmp(flags, "S");
301 Connection *connection = NULL;
302 const char *direction = "new";
303 for (size_t i = 0; i < connection_count; i++) {
304 Connection *c = connections[i];
305 bool orig = tuple_equal(c->orig, original), reply = !orig && tuple_equal(c->reply, original);
306 if (!orig && !reply) continue;
307 if (!strcmp(c->state, "CLOSED") && pure_syn) remove_connection(i);
308 else {
309 connection = c;
310 direction = orig ? "orig" : "reply";
311 }
312 break;
313 }
314 if (!strcmp(original.protocol, "TCP") && !connection && !pure_syn) return drop(input, 3, "DROP invalid");
315 char *verdict = acl(input, 0, original, original.source, original.sport, original.destination, original.dport, connection != NULL);
316 if (verdict) return drop(input, 4, verdict);
317 uint32_t source = original.source, destination = original.destination;
318 int sport = original.sport, dport = original.dport;
319 Pat *used = NULL;
320 if (input->outside) {
321 StaticNat *mapping = NULL;
322 for (size_t i = 0; i < static_count; i++) if (statics[i]->outside == destination) {
323 mapping = statics[i];
324 break;
325 }
326 if (mapping) destination = mapping->inside;
327 else if (destination == input->prefix.address) {
328 Forward *f = NULL;
329 for (size_t i = 0; i < forward_count; i++) if (forwards[i]->interface == input && !strcmp(forwards[i]->protocol, original.protocol) && forwards[i]->port == dport) {
330 f = forwards[i];
331 break;
332 }
333 if (f) {
334 destination = f->inside;
335 dport = f->inport;
336 } else for (size_t i = 0; i < pat_count; i++) if (pats[i]->interface == input && !strcmp(pats[i]->protocol, original.protocol) && pats[i]->port == dport) {
337 used = pats[i];
338 destination = used->inside;
339 dport = used->inport;
340 break;
341 }
342 }
343 }
344 Interface *target = local_address(destination);
345 if (target) {
346 local++;
347 return format("LOCAL %s", target->name);
348 }
349 if (ttl == 1) return drop(input, 5, "DROP ttl");
350 Route *route = select_route(destination);
351 if (!route) return drop(input, 6, "DROP noroute");
352 Interface *output = route->output;
353 verdict = acl(output, 1, original, source, sport, destination, dport, connection != NULL);
354 if (verdict) return drop(input, 7, verdict);
355 if (!input->outside && output->outside) {
356 StaticNat *mapping = NULL;
357 for (size_t i = 0; i < static_count; i++) if (statics[i]->inside == source) {
358 mapping = statics[i];
359 break;
360 }
361 if (mapping) source = mapping->outside;
362 else {
363 Forward *f = NULL;
364 for (size_t i = 0; i < forward_count; i++) if (forwards[i]->interface == output && !strcmp(forwards[i]->protocol, original.protocol) && forwards[i]->inside == source && forwards[i]->inport == sport) {
365 f = forwards[i];
366 break;
367 }
368 if (f) {
369 source = output->prefix.address;
370 sport = f->port;
371 } else if (output->pool) {
372 for (size_t i = 0; i < pat_count; i++) if (pats[i]->interface == output && !strcmp(pats[i]->protocol, original.protocol) && pats[i]->inside == source && pats[i]->inport == sport) {
373 used = pats[i];
374 break;
375 }
376 if (!used) {
377 int port;
378 for (port = 20000; port < 20000 + output->pool; port++) {
379 bool occupied = false;
380 for (size_t i = 0; i < pat_count; i++) if (pats[i]->interface == output && pats[i]->port == port) occupied = true;
381 for (size_t i = 0; i < forward_count; i++) if (forwards[i]->interface == output && forwards[i]->port == port) occupied = true;
382 if (!occupied) break;
383 }
384 if (port == 20000 + output->pool) return drop(input, 8, "DROP exhausted");
385 used = allocate(sizeof(*used));
386 *used = (Pat){output, original.protocol, source, sport, port, clock_time};
387 APPEND(pats, pat_count, used);
388 }
389 source = output->prefix.address;
390 sport = used->port;
391 }
392 }
393 }
394 if (used) used->last = clock_time;
395 if (!connection) {
396 connection = allocate(sizeof(*connection));
397 connection->number = ++connection_number;
398 connection->orig = original;
399 connection->reply = (Tuple){original.protocol, destination, source, dport, sport};
400 connection->input = input;
401 connection->output = output;
402 connection->state = !strcmp(original.protocol, "TCP") ? "SYN_SENT" : "UNREPLIED";
403 APPEND(connections, connection_count, connection);
404 }
405 if (!strcmp(direction, "reply")) connection->reply_packets++;
406 else connection->orig_packets++;
407 connection->last = clock_time;
408 if (strcmp(original.protocol, "TCP")) {
409 if (!strcmp(direction, "reply")) connection->state = "REPLIED";
410 } else if (strchr(flags, 'R')) connection->state = "CLOSED";
411 else {
412 if (!strcmp(connection->state, "SYN_SENT") && !strcmp(direction, "reply") && strchr(flags, 'S') && strchr(flags, 'A')) connection->state = "ESTABLISHED";
413 if (strchr(flags, 'F') && (!strcmp(connection->state, "ESTABLISHED") || !strcmp(connection->state, "FIN_WAIT"))) {
414 if (!strcmp(direction, "reply")) connection->reply_fin = true;
415 else connection->orig_fin = true;
416 connection->state = connection->orig_fin && connection->reply_fin ? "CLOSED" : "FIN_WAIT";
417 }
418 }
419 output->tx++;
420 route->hits++;
421 forwarded++;
422 return format("FORWARD %s %s ttl=%d%s%s conn=%d %s", output->name, route->via ? format("via %s", ip(route->hop)) : "direct", ttl - 1, destination != original.destination || dport != original.dport ? format(" dnat=%s:%d", ip(destination), dport) : "", source != original.source || sport != original.sport ? format(" snat=%s:%d", ip(source), sport) : "", connection->number, direction);
423}

429static int route_compare(const void *a, const void *b) {
430 const Route *x = *(Route *const *)a, *y = *(Route *const *)b;
431 if (x->prefix.address != y->prefix.address) return x->prefix.address > y->prefix.address ? 1 : -1;
432 if (x->prefix.length != y->prefix.length) return x->prefix.length - y->prefix.length;
433 if (x->metric != y->metric) return x->metric - y->metric;
434 return (x->order > y->order) - (x->order < y->order);
435}

526static void execute(char **a, size_t argc) {
527 if (!strcmp(a[0], "IFACE")) {
528 for (size_t i = 0; i < interface_count; i++) if (!strcmp(interfaces[i]->name, a[1])) error("duplicate interface %s", a[1]);
529 Prefix p;
530 if (!parse_prefix(a[2], &p) || p.length < 8 || p.length > 30) error("bad prefix %s", a[2]);
531 uint32_t host = p.address & ~mask(p.length);
532 if (!host || host == ~mask(p.length)) error("bad host address %s", a[2]);
533 for (size_t i = 0; i < interface_count; i++) {
534 Prefix other = interfaces[i]->prefix;
535 int k = p.length < other.length ? p.length : other.length;
536 if ((p.address & mask(k)) == (other.address & mask(k))) error("subnet overlaps %s", interfaces[i]->name);
537 }
538 Interface *f = allocate(sizeof(*f));
539 f->name = a[1];
540 f->prefix = p;
541 f->up = true;
542 f->rpf = "off";
543 APPEND(interfaces, interface_count, f);
544 add_route('C', (Prefix){network(p), p.length}, f, false, 0, 0);
545 printf("ok iface %s %s network %s/%d\n", a[1], a[2], ip(network(p)), p.length);
546 } else if (!strcmp(a[0], "UP") || !strcmp(a[0], "DOWN")) {

568 if (r->type == 'S' && equal_prefix(r->prefix, p) && r->output == output && r->via == via && (!via || r->hop == hop)) error("duplicate route %s", a[1]);
569 }
570 printf("ok route %s\n", route_text(add_route('S', p, output, via, hop, metric)));
571 } else if (!strcmp(a[0], "NOROUTE")) {
572 Prefix p = prefix(a[1], false);

574 for (size_t i = 0; i < route_count; i++) if (routes[i]->type == 'S' && equal_prefix(routes[i]->prefix, p)) count++;
575 if (!count) error("no static route %s", a[1]);

577 for (size_t i = 0; i < route_count; i++) if (!(routes[i]->type == 'S' && equal_prefix(routes[i]->prefix, p))) routes[next++] = routes[i];
578 route_count = next;

580 } else if (!strcmp(a[0], "LOOKUP")) {

583 } else if (!strcmp(a[0], "ACL")) {

593 r->dports = ports(a[9]);

595 for (size_t i = 0; i < f->acl_count[d]; i++) if (f->acl[d][i]->sequence == r->sequence) error("duplicate sequence %s", a[3]);
596 r->established = argc == 11;
597 APPEND(f->acl[d], f->acl_count[d], r);
598 qsort(f->acl[d], f->acl_count[d], sizeof(*f->acl[d]), rule_compare);
599 printf("ok acl %s %s %s\n", a[1], d ? "out" : "in", a[3]);
600 } else if (!strcmp(a[0], "NOACL")) {
601 Interface *f = find_interface(a[1]);

603 size_t i;

605 if (i == f->acl_count[d]) error("no rule %s on %s %s", a[3], a[1], d ? "out" : "in");
606 int hits = f->acl[d][i]->hits;
607 memmove(f->acl[d] + i, f->acl[d] + i + 1, (f->acl_count[d] - i - 1) * sizeof(*f->acl[d]));

609 printf("ok noacl %s %s %s hits=%d\n", a[1], d ? "out" : "in", a[3], hits);
610 } else if (!strcmp(a[0], "NAT")) {
611 if (!strcmp(a[1], "STATIC")) {
612 uint32_t inside = address(a[2]);

641 Forward *n = allocate(sizeof(*n));

649 f->rpf = !strcmp(a[2], "STRICT") ? "strict" : !strcmp(a[2], "LOOSE") ? "loose" : "off";

672 } else if (!strcmp(a[0], "PACKET")) {

695 }
696 for (size_t i = 0; i < route_count; i++) routes[i]->hits = 0;

701static char *read_line(void) {
702 char *s = NULL;
703 size_t n = 0;
704 int c;
705 while ((c = getchar()) != EOF && c != '\n') {
706 s = grow(s, n + 2);
707 s[n++] = (char)c;
708 }
709 if (c == EOF && !n) return NULL;
710 s = grow(s, n + 1);
711 s[n] = 0;
712 return s;
713}
714
715int main(void) {
716 char *first = read_line();
717 int count = atoi(first);
718 for (int i = 0; i < count; i++) {
719 char *line = read_line(), **words = NULL;
720 size_t argc = 0, start = 0;
721 for (size_t j = 0;; j++) if (!line[j] || line[j] == ' ') {
722 APPEND(words, argc, slice(line + start, j - start));
723 start = j + 1;
724 if (!line[j]) break;
725 }
726 if (!setjmp(failure)) execute(words, argc);
727 }
728 return 0;
729}
