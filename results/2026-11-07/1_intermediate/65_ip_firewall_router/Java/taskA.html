108 static long mask(int length) {
109 return length == 0 ? 0 : (0xffffffffL << (32 - length)) & 0xffffffffL;
110 }

112 static String ip(long value) {
113 return (value >>> 24) + "." + ((value >>> 16) & 255) + "." + ((value >>> 8) & 255) + "." + (value & 255);
114 }

116 static Long parseAddress(String text) {
117 String[] fields = text.split("\\.", -1);
118 if (fields.length != 4) return null;
119 long value = 0;
120 for (String field : fields) {
121 if (!field.matches("0|[1-9][0-9]{0,2}")) return null;
122 int part = Integer.parseInt(field);
123 if (part > 255) return null;
124 value = (value << 8) | part;
125 }
126 return value;
127 }

129 static long address(String text) {

135 static Prefix parsePrefix(String text) {
136 String[] parts = text.split("/", -1);
137 if (parts.length != 2 || !parts[1].matches("0|[1-9][0-9]?")) return null;
138 Long value = parseAddress(parts[0]);
139 int length = Integer.parseInt(parts[1]);
140 return value == null || length > 32 ? null : new Prefix(value, length);
141 }

147 if (prefix.address() != prefix.network()) error("host bits set in " + text);
148 return prefix;
149 }

182 static String acl(Iface iface, int direction, Tuple tuple, long source, int sport, long destination, int dport, boolean established) {
183 var rules = iface.acl.get(direction);
184 if (rules.isEmpty()) return null;
185 String dir = direction == 0 ? "in" : "out";
186 for (Rule rule : rules.values()) {
187 if (!(rule.proto.equals("ANY") || rule.proto.equals(tuple.proto())) || !rule.source.contains(source) || !rule.destination.contains(destination) || rule.established && !established) continue;
188 if ((rule.proto.equals("TCP") || rule.proto.equals("UDP")) && (!rule.sports.contains(sport) || !rule.dports.contains(dport))) continue;
189 rule.hits++;
190 if (rule.action.equals("PERMIT")) return null;
191 return (rule.action.equals("DENY") ? "DROP" : "REJECT") + " acl " + iface.name + " " + dir + " " + rule.sequence + (rule.action.equals("REJECT") ? tuple.proto().equals("TCP") ? " tcp-reset" : " icmp-prohibited" : "");
192 }
193 iface.defaults[direction]++;
194 return "DROP acl " + iface.name + " " + dir + " default";
195 }

207 static String packet(Iface input, Tuple original, String flags, int ttl) {
208 input.rx++;
209 if (!input.up) return drop(input, 0, "DROP down");
210 if (martian(original.source()) || martian(original.destination()) || localAddress(original.source()) != null) return drop(input, 1, "DROP martian");
211 if (!input.rpf.equals("off")) {
212 Route reverse = select(original.source());
213 if (reverse == null || input.rpf.equals("strict") && reverse.output != input) return drop(input, 2, "DROP rpf");
214 }
215 boolean pureSyn = original.proto().equals("TCP") && flags.equals("S");
216 Connection connection = null;
217 String direction = "new";
218 for (Connection candidate : connections) {
219 if (candidate.orig.equals(original)) {
220 connection = candidate;
221 direction = "orig";
222 break;
223 }
224 if (candidate.reply.equals(original)) {
225 connection = candidate;
226 direction = "reply";
227 break;
228 }
229 }
230 if (connection != null && connection.state.equals("CLOSED") && pureSyn) {
231 connections.remove(connection);
232 connection = null;
233 direction = "new";
234 }
235 if (original.proto().equals("TCP") && connection == null && !pureSyn) return drop(input, 3, "DROP invalid");
236 String verdict = acl(input, 0, original, original.source(), original.sport(), original.destination(), original.dport(), connection != null);
237 if (verdict != null) return drop(input, 4, verdict);
238 long destination = original.destination(), source = original.source();
239 int dport = original.dport(), sport = original.sport();
240 Pat used = null;
241 if (input.outside) {
242 StaticNat mapping = null;
243 for (StaticNat candidate : statics) if (candidate.outside() == destination) {
244 mapping = candidate;
245 break;
246 }
247 if (mapping != null) destination = mapping.inside();
248 else if (destination == input.prefix.address()) {
249 Forward forwarding = null;
250 for (Forward candidate : forwards) if (candidate.iface() == input && candidate.proto().equals(original.proto()) && candidate.port() == dport) {
251 forwarding = candidate;
252 break;
253 }
254 if (forwarding != null) {
255 destination = forwarding.inside();
256 dport = forwarding.inport();
257 } else for (Pat candidate : pats) if (candidate.iface == input && candidate.proto.equals(original.proto()) && candidate.port == dport) {
258 used = candidate;
259 destination = candidate.inside;
260 dport = candidate.inport;
261 break;
262 }
263 }
264 }
265 Iface target = localAddress(destination);
266 if (target != null) {
267 local++;
268 return "LOCAL " + target.name;
269 }
270 if (ttl == 1) return drop(input, 5, "DROP ttl");
271 Route route = select(destination);
272 if (route == null) return drop(input, 6, "DROP noroute");
273 Iface output = route.output;
274 verdict = acl(output, 1, original, source, sport, destination, dport, connection != null);
275 if (verdict != null) return drop(input, 7, verdict);
276 if (!input.outside && output.outside) {
277 StaticNat mapping = null;
278 for (StaticNat candidate : statics) if (candidate.inside() == source) {
279 mapping = candidate;
280 break;
281 }
282 if (mapping != null) source = mapping.outside();
283 else {
284 Forward forwarding = null;
285 for (Forward candidate : forwards) if (candidate.iface() == output && candidate.proto().equals(original.proto()) && candidate.inside() == source && candidate.inport() == sport) {
286 forwarding = candidate;
287 break;
288 }
289 if (forwarding != null) {
290 source = output.prefix.address();
291 sport = forwarding.port();
292 } else if (output.pool > 0) {
293 for (Pat candidate : pats) if (candidate.iface == output && candidate.proto.equals(original.proto()) && candidate.inside == source && candidate.inport == sport) {
294 used = candidate;
295 break;
296 }
297 if (used == null) {
298 int port;
299 for (port = 20000; port < 20000 + output.pool; port++) {
300 boolean occupied = false;
301 for (Pat candidate : pats) if (candidate.iface == output && candidate.port == port) occupied = true;
302 for (Forward candidate : forwards) if (candidate.iface() == output && candidate.port() == port) occupied = true;
303 if (!occupied) break;
304 }
305 if (port == 20000 + output.pool) return drop(input, 8, "DROP exhausted");
306 used = new Pat(output, original.proto(), source, sport, port);
307 pats.add(used);
308 }
309 source = output.prefix.address();
310 sport = used.port;
311 }
312 }
313 }
314 if (used != null) used.last = clock;
315 if (connection == null) {
316 connection = new Connection();
317 connection.number = ++connectionNumber;
318 connection.orig = original;
319 connection.reply = new Tuple(original.proto(), destination, dport, source, sport);
320 connection.input = input;
321 connection.output = output;
322 connection.state = original.proto().equals("TCP") ? "SYN_SENT" : "UNREPLIED";
323 connections.add(connection);
324 }
325 if (direction.equals("reply")) connection.replyPackets++;
326 else connection.origPackets++;
327 connection.last = clock;
328 if (!original.proto().equals("TCP")) {
329 if (direction.equals("reply")) connection.state = "REPLIED";
330 } else if (flags.contains("R")) connection.state = "CLOSED";
331 else {
332 if (connection.state.equals("SYN_SENT") && direction.equals("reply") && flags.contains("S") && flags.contains("A")) connection.state = "ESTABLISHED";
333 if (flags.contains("F") && (connection.state.equals("ESTABLISHED") || connection.state.equals("FIN_WAIT"))) {
334 if (direction.equals("reply")) connection.replyFin = true;
335 else connection.origFin = true;
336 connection.state = connection.origFin && connection.replyFin ? "CLOSED" : "FIN_WAIT";
337 }
338 }
339 output.tx++;
340 route.hits++;
341 forwarded++;
342 return "FORWARD " + output.name + (route.hop == null ? " direct" : " via " + ip(route.hop)) + " ttl=" + (ttl - 1) + (destination != original.destination() || dport != original.dport() ? " dnat=" + ip(destination) + ":" + dport : "") + (source != original.source() || sport != original.sport() ? " snat=" + ip(source) + ":" + sport : "") + " conn=" + connection.number + " " + direction;
343 }

389 static void execute(String[] a) {
390 switch (a[0]) {
391 case "IFACE": {
392 for (Iface iface : interfaces) if (iface.name.equals(a[1])) error("duplicate interface " + a[1]);
393 Prefix p = parsePrefix(a[2]);
394 if (p == null || p.length() < 8 || p.length() > 30) error("bad prefix " + a[2]);
395 long host = p.address() & (0xffffffffL ^ mask(p.length()));
396 if (host == 0 || host == (0xffffffffL ^ mask(p.length()))) error("bad host address " + a[2]);
397 for (Iface iface : interfaces) if ((p.address() & mask(Math.min(p.length(), iface.prefix.length()))) == (iface.prefix.address() & mask(Math.min(p.length(), iface.prefix.length())))) error("subnet overlaps " + iface.name);
398 Iface iface = new Iface(a[1], p);
399 interfaces.add(iface);
400 routes.add(new Route('C', new Prefix(p.network(), p.length()), iface, null, 0));
401 System.out.println("ok iface " + a[1] + " " + a[2] + " network " + ip(p.network()) + "/" + p.length());
402 break;
403 }
404 case "UP": case "DOWN": {
405 Iface iface = iface(a[1]);
406 boolean up = a[0].equals("UP");
407 if (iface.up == up) error("interface " + a[1] + " already " + (up ? "up" : "down"));
408 iface.up = up;
409 System.out.println("ok " + a[0].toLowerCase(Locale.ROOT) + " " + a[1]);
410 break;
411 }
412 case "ROUTE": {
413 Prefix p = prefix(a[1], false);
414 boolean via = a[2].equals("VIA");
415 Long hop = via ? address(a[3]) : null;
416 Iface output = via ? null : iface(a[3]);
417 int metric = range(a[4], 1, 255, "bad metric");
418 if (via) {
419 if (localAddress(hop) != null) error("nexthop " + a[3] + " is local");
420 for (Iface iface : interfaces) if (iface.prefix.contains(hop)) {
421 output = iface;
422 break;
423 }
424 if (output == null) error("nexthop " + a[3] + " unreachable");
425 }
426 for (Route route : routes) if (route.type == 'S' && route.prefix.equals(p) && route.output == output && Objects.equals(route.hop, hop)) error("duplicate route " + a[1]);
427 Route route = new Route('S', p, output, hop, metric);
428 routes.add(route);
429 System.out.println("ok route " + route);
430 break;
431 }
432 case "NOROUTE": {
433 Prefix p = prefix(a[1], false);
434 int old = routes.size();
435 boolean exists = routes.stream().anyMatch(r -> r.type == 'S' && r.prefix.equals(p));
436 if (!exists) error("no static route " + a[1]);
437 routes.removeIf(r -> r.type == 'S' && r.prefix.equals(p));
438 System.out.println("ok noroute " + a[1] + " removed=" + (old - routes.size()));
439 break;
440 }
441 case "LOOKUP": {
442 Route route = select(address(a[1]));
443 System.out.println("lookup " + a[1] + " " + (route == null ? "none" : route));
444 break;
445 }
446 case "ACL": {
447 Iface iface = iface(a[1]);
448 int d = a[2].equals("IN") ? 0 : 1;
449 Rule rule = new Rule();
450 rule.sequence = range(a[3], 1, 9999, "bad sequence");
451 rule.action = a[4];
452 rule.proto = a[5];
453 rule.source = prefix(a[6], true);
454 rule.sports = ports(a[7]);
455 rule.destination = prefix(a[8], true);
456 rule.dports = ports(a[9]);
457 if ((a[5].equals("ICMP") || a[5].equals("ANY")) && (!a[7].equals("ANY") || !a[9].equals("ANY"))) error("ports not allowed for " + a[5]);
458 if (iface.acl.get(d).containsKey(rule.sequence)) error("duplicate sequence " + a[3]);
459 rule.established = a.length == 11;
460 iface.acl.get(d).put(rule.sequence, rule);
461 System.out.println("ok acl " + a[1] + " " + a[2].toLowerCase(Locale.ROOT) + " " + a[3]);
462 break;
463 }
464 case "NOACL": {
465 Iface iface = iface(a[1]);
466 int d = a[2].equals("IN") ? 0 : 1;
467 Rule rule = iface.acl.get(d).get(Integer.parseInt(a[3]));
468 if (rule == null) error("no rule " + a[3] + " on " + a[1] + " " + a[2].toLowerCase(Locale.ROOT));
469 iface.acl.get(d).remove(rule.sequence);
470 System.out.println("ok noacl " + a[1] + " " + a[2].toLowerCase(Locale.ROOT) + " " + a[3] + " hits=" + rule.hits);
471 break;
472 }
473 case "NAT": {
474 if (a[1].equals("STATIC")) {
475 long inside = address(a[2]), outside = address(a[3]);
476 if (localAddress(inside) != null) error("address " + a[2] + " is local");
477 if (localAddress(outside) != null) error("address " + a[3] + " is local");
478 for (StaticNat n : statics) if (n.inside() == inside) error("duplicate static " + a[2]);
479 for (StaticNat n : statics) if (n.outside() == outside) error("duplicate static " + a[3]);
480 statics.add(new StaticNat(inside, outside));
481 System.out.println("ok nat static " + a[2] + " " + a[3]);
482 } else {
483 Iface iface = iface(a[2]);
484 if (a[1].equals("OUTSIDE")) {
485 if (iface.outside) error("interface " + a[2] + " already outside");
486 iface.outside = true;
487 System.out.println("ok nat outside " + a[2]);
488 } else if (a[1].equals("PAT")) {
489 int pool = range(a[3], 1, 1000, "bad pool size");
490 if (iface.pool > 0) error("pat already on " + a[2]);
491 iface.pool = pool;
492 iface.outside = true;
493 System.out.println("ok nat pat " + a[2] + " " + ip(iface.prefix.address()) + " 20000-" + (19999 + pool));
494 } else {
495 int port = range(a[4], 1, 65535, "bad port");
496 long inside = address(a[5]);
497 int inport = range(a[6], 1, 65535, "bad port");
498 if (!iface.outside) error("interface " + a[2] + " not outside");
499 if (localAddress(inside) != null) error("address " + a[5] + " is local");
500 for (Forward f : forwards) if (f.iface() == iface && f.proto().equals(a[3]) && f.port() == port) error("duplicate forward " + a[2] + " " + a[3] + " " + a[4]);
501 forwards.add(new Forward(iface, a[3], port, inside, inport));
502 System.out.println("ok nat forward " + a[2] + " " + a[3] + " " + a[4] + " " + a[5] + ":" + a[6]);
503 }
504 }
505 break;
506 }
507 case "RPF": {
508 Iface iface = iface(a[1]);
509 iface.rpf = a[2].toLowerCase(Locale.ROOT);
510 System.out.println("ok rpf " + a[1] + " " + iface.rpf);
511 break;
512 }
513 case "TIMEOUT": {
514 int seconds = range(a[2], 1, 86400, "bad timeout");
515 timeouts.put(a[1], seconds);
516 System.out.println("ok timeout " + a[1] + " " + a[2]);
517 break;
518 }
519 case "TICK": {
520 clock += range(a[1], 1, 86400, "bad tick");
521 int c = connections.size(), n = pats.size();
522 connections.removeIf(x -> clock - x.last >= (x.orig.proto().equals("TCP") ? switch (x.state) {
523 case "SYN_SENT", "FIN_WAIT" -> 60; case "CLOSED" -> 10; default -> timeouts.get("TCP");
524 }
525 : timeouts.get(x.orig.proto())));
526 pats.removeIf(x -> clock - x.last >= timeouts.get(x.proto));
527 System.out.println("ok tick time=" + clock + " expired conns=" + (c - connections.size()) + " nat=" + (n - pats.size()));
528 break;
529 }
530 case "PACKET": {
531 Iface iface = iface(a[1]);
532 if (!a[2].equals("TCP") && !a[3].equals("-")) error("flags not allowed for " + a[2]);
533 if (a[2].equals("TCP") && !a[3].equals("-")) {
534 Set<Character> seen = new HashSet<>();
535 for (char c : a[3].toCharArray()) if ("SAFR".indexOf(c) < 0 || !seen.add(c)) error("bad flags " + a[3]);
536 }
537 long source = address(a[4]);
538 int sport = range(a[5], 0, 65535, "bad port");
539 long destination = address(a[6]);
540 int dport = range(a[7], 0, 65535, "bad port"), ttl = range(a[8], 1, 255, "bad ttl");
541 packetNumber++;
542 String verdict = packet(iface, new Tuple(a[2], source, sport, destination, dport), a[3], ttl);
543 System.out.println("#" + packetNumber + " " + a[2] + " " + a[3] + " " + a[4] + ":" + a[5] + " > " + a[6] + ":" + a[7] + " " + verdict);
544 break;
545 }
546 case "SHOW": show(a[1]);
547 break;
548 case "CLEAR": if (a[1].equals("CONNS")) {
549 System.out.println("ok clear conns=" + connections.size() + " nat=" + pats.size());
550 connections.clear();
551 pats.clear();
552 } else {
553 forwarded = local = errors = 0;
554 Arrays.fill(drops, 0);
555 for (Iface iface : interfaces) {
556 iface.rx = iface.tx = iface.drop = 0;
557 Arrays.fill(iface.defaults, 0);
558 for (var rules : iface.acl) for (Rule rule : rules.values()) rule.hits = 0;
559 }
560 for (Route route : routes) route.hits = 0;
561 System.out.println("ok clear counters");
562 }
563 }
564 }
