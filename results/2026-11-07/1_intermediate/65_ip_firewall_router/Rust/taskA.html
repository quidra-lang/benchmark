3fn ip(word: &str) -> Result<u32, String> {
4 let fields: Vec<_> = word.split('.').collect();
5 if fields.len() != 4 { return Err(format!("bad address {word}")); }
6 let mut address = 0;
7 for field in fields {
8 if field.is_empty() || field.len() > 3 || (field.len() > 1 && field.starts_with('0')) || !field.bytes().all(|c| c.is_ascii_digit()) {
9 return Err(format!("bad address {word}"));
10 }
11 let n: u32 = field.parse().unwrap();
12 if n > 255 { return Err(format!("bad address {word}")); }
13 address = address * 256 + n;
14 }
15 Ok(address)
16}
17fn address(a: u32) -> String { format!("{}.{}.{}.{}", a>>24, (a>>16)&255, (a>>8)&255, a&255) }
18fn mask(length: u32) -> u32 { if length == 0 { 0 } else { u32::MAX << (32-length) } }
19#[derive(Clone, Copy, PartialEq, Eq)]
20struct Prefix { network: u32, length: u32 }
21impl Prefix {
22 fn contains(self, a: u32) -> bool { a & mask(self.length) == self.network & mask(self.length) }
23 fn notation(self) -> String { format!("{}/{}", address(self.network), self.length) }
24 fn acl_notation(self) -> String { if self.length == 0 { "any".into() } else { self.notation() } }
25}
26fn prefix(word: &str, host_bits: bool) -> Result<Prefix, String> {
27 let fields: Vec<_> = word.split('/').collect();
28 let bad = || format!("bad prefix {word}");
29 if fields.len() != 2 { return Err(bad()); }
30 let network = ip(fields[0]).map_err(|_| bad())?;
31 let s = fields[1];
32 if s.is_empty() || s.len() > 2 || (s.len() == 2 && s.starts_with('0')) || !s.bytes().all(|c| c.is_ascii_digit()) { return Err(bad()); }
33 let length: u32 = s.parse().unwrap();
34 if length > 32 { return Err(bad()); }
35 if host_bits && network & !mask(length) != 0 { return Err(format!("host bits set in {word}")); }
36 Ok(Prefix { network, length })
37}
38#[derive(Clone)]
39struct Ports { low: i64, high: i64, any: bool }
40impl Ports {
41 fn notation(&self) -> String { if self.any { "any".into() } else if self.low == self.high { self.low.to_string() } else { format!("{}-{}", self.low, self.high) } }
42 fn contains(&self, port: i64) -> bool { self.low <= port && port <= self.high }
43}
44fn ports(word: &str) -> Result<Ports, String> {
45 if word == "ANY" { return Ok(Ports { low: 0, high: 65535, any: true }); }
46 let values: Vec<i64> = word.split('-').map(|s| s.parse().unwrap()).collect();
47 let low = values[0];
48 let high = *values.last().unwrap();
49 if high > 65535 || low > 65535 || low > high { return Err(format!("bad port range {word}")); }
50 Ok(Ports { low, high, any: false })
51}

70const REASONS: [&str;9] = ["down", "martian", "rpf", "invalid", "acl_in", "ttl", "noroute", "acl_out", "exhausted"];
71struct Router {

77fn protocol(p: &str) -> usize { match p { "TCP" => 0, "UDP" => 1, _ => 2 } }
78impl Router {
79 fn new() -> Self {
80 Self { ifaces: Vec::new(), routes: Vec::new(), rules: Vec::new(), defaults: Vec::new(), statics: Vec::new(), forwards: Vec::new(), pats: Vec::new(), conns: Vec::new(), timeouts: [300,60,30], clock: 0, packet_number: 0, connection_number: 0, forwarded: 0, local: 0, drops: [0;9], errors: 0 }

82 fn iface(&self, name: &str) -> Result<usize,String> { self.ifaces.iter().position(|i| i.name == name).ok_or_else(|| format!("no interface {name}")) }
83 fn local_address(&self, a: u32) -> Option<usize> { self.ifaces.iter().position(|i| i.address == a) }
84 fn select(&self, a: u32) -> Option<usize> {
85 let mut selected = None;
86 for (i,r) in self.routes.iter().enumerate() {
87 if !self.ifaces[r.iface].up || !r.dest.contains(a) { continue; }
88 if selected.is_none_or(|j: usize| r.dest.length > self.routes[j].dest.length || (r.dest.length == self.routes[j].dest.length && r.metric < self.routes[j].metric)) { selected = Some(i); }
89 }
90 selected
91 }
92 fn route_text(&self, r: &Route) -> String {
93 format!("{} {}{} dev {} metric {}", r.kind, r.dest.notation(), r.hop.map_or(String::new(), |h| format!(" via {}",address(h))), self.ifaces[r.iface].name, r.metric)
94 }
95 fn acl(&mut self, iface: usize, dir: usize, tuple: &Tuple, existing: bool) -> Option<String> {
96 let mut indices: Vec<_> = self.rules.iter().enumerate().filter(|(_,r)| r.iface==iface && r.dir==dir).map(|(i,_)| i).collect();
97 if indices.is_empty() { return None; }
98 indices.sort_by_key(|&i| self.rules[i].seq);
99 let name = &self.ifaces[iface].name;
100 let d = if dir == 0 { "in" } else { "out" };
101 for i in indices {
102 let r = &mut self.rules[i];
103 if (r.proto != "ANY" && r.proto != tuple.proto) || !r.src.contains(tuple.src) || !r.dst.contains(tuple.dst) || (r.established && !existing) { continue; }
104 if (r.proto == "TCP" || r.proto == "UDP") && (!r.sports.contains(tuple.sport) || !r.dports.contains(tuple.dport)) { continue; }
105 r.hits += 1;
106 return match r.action.as_str() {
107 "PERMIT" => None,
108 "DENY" => Some(format!("DROP acl {name} {d} {}", r.seq)),
109 _ => Some(format!("REJECT acl {name} {d} {} {}", r.seq, if tuple.proto == "TCP" { "tcp-reset" } else { "icmp-prohibited" })),
110 };
111 }
112 self.defaults[iface][dir] += 1;
113 Some(format!("DROP acl {name} {d} default"))
114 }
115 fn packet(&mut self, input: usize, flags: &str, original: Tuple, ttl: i64) -> (String, i32) {
116 if !self.ifaces[input].up { return ("DROP down".into(),0); }
117 let martian = |a: u32| a>>24 == 0 || a>>24 == 127 || a >= 0xe0000000;
118 if martian(original.src) || martian(original.dst) || self.local_address(original.src).is_some() { return ("DROP martian".into(),1); }
119 if self.ifaces[input].rpf != "off" {
120 let route = self.select(original.src);
121 if route.is_none() || (self.ifaces[input].rpf == "strict" && self.routes[route.unwrap()].iface != input) { return ("DROP rpf".into(),2); }
122 }
123 let pure_syn = original.proto == "TCP" && flags == "S";
124 let mut found = self.conns.iter().enumerate().find_map(|(i,c)| if c.orig == original { Some((i,0)) } else if c.reply == original { Some((i,1)) } else { None });
125 if let Some((i,_)) = found {
126 if self.conns[i].state == "CLOSED" && pure_syn { self.conns.remove(i); found = None; }
127 }
128 if original.proto == "TCP" && found.is_none() && !pure_syn { return ("DROP invalid".into(),3); }
129 if let Some(verdict) = self.acl(input,0,&original,found.is_some()) { return (verdict,4); }
130 let (mut dst, mut dport) = (original.dst,original.dport);
131 let mut used = None;
132 if self.ifaces[input].outside {
133 if let Some(&(inside,_)) = self.statics.iter().find(|(_,outside)| *outside == dst) { dst = inside; }
134 else if dst == self.ifaces[input].address {
135 if let Some(f) = self.forwards.iter().find(|f| f.iface==input && f.proto==original.proto && f.port==dport) { dst=f.inside; dport=f.inport; }
136 else if let Some((i,p)) = self.pats.iter().enumerate().find(|(_,p)| p.iface==input && p.proto==original.proto && p.port==dport) { dst=p.inside; dport=p.inport; used=Some(i); }
137 }
138 }
139 if let Some(i) = self.local_address(dst) { return (format!("LOCAL {}",self.ifaces[i].name),-2); }
140 if ttl == 1 { return ("DROP ttl".into(),5); }
141 let Some(route) = self.select(dst) else { return ("DROP noroute".into(),6); };
142 let output = self.routes[route].iface;
143 let outgoing = Tuple { dst, dport, ..original.clone() };
144 if let Some(verdict) = self.acl(output,1,&outgoing,found.is_some()) { return (verdict,7); }
145 let (mut src, mut sport) = (original.src,original.sport);
146 if !self.ifaces[input].outside && self.ifaces[output].outside {
147 if let Some(&(_,outside)) = self.statics.iter().find(|(inside,_)| *inside == src) { src=outside; }
148 else if let Some(f) = self.forwards.iter().find(|f| f.iface==output && f.proto==original.proto && f.inside==src && f.inport==sport) { src=self.ifaces[output].address; sport=f.port; }
149 else if let Some(count) = self.ifaces[output].pool {
150 let mapping = self.pats.iter().position(|p| p.iface==output && p.proto==original.proto && p.inside==src && p.inport==sport);
151 let index = if let Some(i) = mapping { i } else {
152 let port = (20000..20000+count).find(|&p| !self.pats.iter().any(|m| m.iface==output && m.port==p) && !self.forwards.iter().any(|f| f.iface==output && f.port==p));
153 let Some(port) = port else { return ("DROP exhausted".into(),8); };
154 self.pats.push(Pat { iface: output, proto: original.proto.clone(), inside: src, inport: sport, port, used: self.clock });
155 self.pats.len()-1
156 };
157 used=Some(index); src=self.ifaces[output].address; sport=self.pats[index].port;
158 }
159 }
160 if let Some(i) = used { self.pats[i].used=self.clock; }
161 let (connection,side,direction) = if let Some((i,side)) = found { (i,side,if side==0 { "orig" } else { "reply" }) } else {
162 self.connection_number += 1;
163 self.conns.push(Connection { number: self.connection_number, orig: original.clone(), reply: Tuple { proto: original.proto.clone(), src: dst, sport: dport, dst: src, dport: sport }, input, output, state: if original.proto=="TCP" { "SYN_SENT".into() } else { "UNREPLIED".into() }, packets: [0;2], used: self.clock, fin: [false;2] });
164 (self.conns.len()-1,0,"new")
165 };
166 let c = &mut self.conns[connection];
167 c.packets[side] += 1; c.used=self.clock;
168 if original.proto != "TCP" { if side==1 { c.state="REPLIED".into(); } }
169 else if flags.contains('R') { c.state="CLOSED".into(); }
170 else {
171 if c.state=="SYN_SENT" && side==1 && flags.contains('S') && flags.contains('A') { c.state="ESTABLISHED".into(); }
172 if flags.contains('F') && (c.state=="ESTABLISHED" || c.state=="FIN_WAIT") { c.fin[side]=true; c.state=if c.fin[0] && c.fin[1] { "CLOSED".into() } else { "FIN_WAIT".into() }; }
173 }
174 let number = c.number;
175 self.ifaces[output].tx += 1; self.routes[route].hits += 1;
176 let hop = self.routes[route].hop.map_or("direct".into(),|a| format!("via {}",address(a)));
177 let mut verdict = format!("FORWARD {} {hop} ttl={}",self.ifaces[output].name,ttl-1);
178 if (dst,dport)!=(original.dst,original.dport) { verdict+=&format!(" dnat={}:{}",address(dst),dport); }
179 if (src,sport)!=(original.src,original.sport) { verdict+=&format!(" snat={}:{}",address(src),sport); }
180 verdict+=&format!(" conn={number} {direction}");
181 (verdict,-1)
182 }
183 fn command(&mut self, words: &[&str]) -> Result<Vec<String>,String> {
184 let w = words;
185 let n = |i: usize| w[i].parse::<i64>().unwrap();
186 let dir = |s: &str| if s=="IN" { 0 } else { 1 };
187 let single = match w[0] {
188 "IFACE" => {
189 if self.ifaces.iter().any(|i| i.name==w[1]) { return Err(format!("duplicate interface {}",w[1])); }
190 let p = prefix(w[2],false)?;
191 if !(8..=30).contains(&p.length) { return Err(format!("bad prefix {}",w[2])); }
192 let host = p.network & !mask(p.length);
193 if host==0 || host==!mask(p.length) { return Err(format!("bad host address {}",w[2])); }
194 if let Some(i) = self.ifaces.iter().find(|i| (i.address & mask(i.length.min(p.length))) == (p.network & mask(i.length.min(p.length)))) { return Err(format!("subnet overlaps {}",i.name)); }
195 let index=self.ifaces.len();
196 self.ifaces.push(Interface { name:w[1].into(),address:p.network,length:p.length,up:true,outside:false,pool:None,rpf:"off".into(),rx:0,tx:0,drop:0 });
197 self.defaults.push([0;2]);
198 let dest=Prefix { network:p.network & mask(p.length),length:p.length };
199 self.routes.push(Route { kind:'C',dest,iface:index,hop:None,metric:0,hits:0 });
200 format!("ok iface {} {} network {}",w[1],w[2],dest.notation())
201 }
202 "UP" | "DOWN" => {
203 let i=self.iface(w[1])?; let up=w[0]=="UP";
204 if self.ifaces[i].up==up { return Err(format!("interface {} already {}",w[1],w[0].to_ascii_lowercase())); }
205 self.ifaces[i].up=up; format!("ok {} {}",w[0].to_ascii_lowercase(),w[1])
206 }
207 "ROUTE" => {
208 let dest=prefix(w[1],true)?;
209 let hop=if w[2]=="VIA" { Some(ip(w[3])?) } else { self.iface(w[3])?; None };
210 if !(1..=255).contains(&n(4)) { return Err(format!("bad metric {}",w[4])); }
211 let iface=if let Some(hop)=hop {
212 if self.local_address(hop).is_some() { return Err(format!("nexthop {} is local",w[3])); }
213 self.ifaces.iter().position(|i| Prefix { network:i.address,length:i.length }.contains(hop)).ok_or_else(|| format!("nexthop {} unreachable",w[3]))?
214 } else { self.iface(w[3])? };
215 if self.routes.iter().any(|r| r.kind=='S' && r.dest==dest && r.iface==iface && r.hop==hop) { return Err(format!("duplicate route {}",w[1])); }
216 let route=Route { kind:'S',dest,iface,hop,metric:n(4),hits:0 };
217 let text=self.route_text(&route); self.routes.push(route); format!("ok route {text}")
218 }
219 "NOROUTE" => {
220 let dest=prefix(w[1],true)?; let old=self.routes.len();
221 if !self.routes.iter().any(|r| r.kind=='S' && r.dest==dest) { return Err(format!("no static route {}",w[1])); }
222 self.routes.retain(|r| r.kind!='S' || r.dest!=dest); format!("ok noroute {} removed={}",w[1],old-self.routes.len())
223 }
224 "LOOKUP" => { let a=ip(w[1])?; format!("lookup {} {}",w[1],self.select(a).map_or("none".into(),|i| self.route_text(&self.routes[i]))) }
225 "ACL" => {
226 let iface=self.iface(w[1])?; let d=dir(w[2]);
227 if !(1..=9999).contains(&n(3)) { return Err(format!("bad sequence {}",w[3])); }
228 let src=if w[6]=="ANY" { Prefix { network:0,length:0 } } else { prefix(w[6],true)? };
229 let sports=ports(w[7])?;
230 let dst=if w[8]=="ANY" { Prefix { network:0,length:0 } } else { prefix(w[8],true)? };
231 let dports=ports(w[9])?;
232 if (w[5]=="ICMP" || w[5]=="ANY") && (!sports.any || !dports.any) { return Err(format!("ports not allowed for {}",w[5])); }
233 if self.rules.iter().any(|r| r.iface==iface && r.dir==d && r.seq==n(3)) { return Err(format!("duplicate sequence {}",w[3])); }
234 self.rules.push(Rule { iface,dir:d,seq:n(3),action:w[4].into(),proto:w[5].into(),src,sports,dst,dports,established:w.len()==11,hits:0 });
235 format!("ok acl {} {} {}",w[1],w[2].to_ascii_lowercase(),w[3])
236 }
237 "NOACL" => {
238 let iface=self.iface(w[1])?; let d=dir(w[2]);
239 let i=self.rules.iter().position(|r| r.iface==iface && r.dir==d && r.seq==n(3)).ok_or_else(|| format!("no rule {} on {} {}",w[3],w[1],w[2].to_ascii_lowercase()))?;
240 let hits=self.rules.remove(i).hits; format!("ok noacl {} {} {} hits={hits}",w[1],w[2].to_ascii_lowercase(),w[3])
241 }
242 "NAT" => match w[1] {
243 "STATIC" => {
244 let inside=ip(w[2])?; let outside=ip(w[3])?;
245 for (a,word) in [(inside,w[2]),(outside,w[3])] { if self.local_address(a).is_some() { return Err(format!("address {word} is local")); } }
246 if self.statics.iter().any(|(a,_)| *a==inside) { return Err(format!("duplicate static {}",w[2])); }
247 if self.statics.iter().any(|(_,a)| *a==outside) { return Err(format!("duplicate static {}",w[3])); }
248 self.statics.push((inside,outside)); format!("ok nat static {} {}",w[2],w[3])
249 }
250 "OUTSIDE" => {
251 let i=self.iface(w[2])?; if self.ifaces[i].outside { return Err(format!("interface {} already outside",w[2])); }
252 self.ifaces[i].outside=true; format!("ok nat outside {}",w[2])
253 }
254 "PAT" => {
255 let i=self.iface(w[2])?;
256 if !(1..=1000).contains(&n(3)) { return Err(format!("bad pool size {}",w[3])); }
257 if self.ifaces[i].pool.is_some() { return Err(format!("pat already on {}",w[2])); }
258 self.ifaces[i].pool=Some(n(3)); self.ifaces[i].outside=true;
259 format!("ok nat pat {} {} 20000-{}",w[2],address(self.ifaces[i].address),20000+n(3)-1)
260 }
261 _ => {
262 let i=self.iface(w[2])?;
263 if !(1..=65535).contains(&n(4)) { return Err(format!("bad port {}",w[4])); }
264 let inside=ip(w[5])?;
265 if !(1..=65535).contains(&n(6)) { return Err(format!("bad port {}",w[6])); }
266 if !self.ifaces[i].outside { return Err(format!("interface {} not outside",w[2])); }
267 if self.local_address(inside).is_some() { return Err(format!("address {} is local",w[5])); }
268 if self.forwards.iter().any(|f| f.iface==i && f.proto==w[3] && f.port==n(4)) { return Err(format!("duplicate forward {} {} {}",w[2],w[3],w[4])); }
269 self.forwards.push(Forward { iface:i,proto:w[3].into(),port:n(4),inside,inport:n(6) });
270 format!("ok nat forward {} {} {} {}:{}",w[2],w[3],w[4],w[5],w[6])
271 }
272 },
273 "RPF" => { let i=self.iface(w[1])?; self.ifaces[i].rpf=w[2].to_ascii_lowercase(); format!("ok rpf {} {}",w[1],self.ifaces[i].rpf) }
274 "TIMEOUT" => {
275 if !(1..=86400).contains(&n(2)) { return Err(format!("bad timeout {}",w[2])); }
276 self.timeouts[protocol(w[1])]=n(2); format!("ok timeout {} {}",w[1],w[2])
277 }
278 "TICK" => {
279 if !(1..=86400).contains(&n(1)) { return Err(format!("bad tick {}",w[1])); }
280 self.clock+=n(1); let old=self.conns.len(); let old_nat=self.pats.len();
281 let clock = self.clock;
282 let timeouts = self.timeouts;
283 self.conns.retain(|c| { let timeout=if c.orig.proto=="TCP" { match c.state.as_str() { "SYN_SENT"|"FIN_WAIT"=>60,"CLOSED"=>10,_=>timeouts[0] } } else { timeouts[protocol(&c.orig.proto)] }; clock-c.used<timeout });
284 self.pats.retain(|p| clock-p.used<timeouts[protocol(&p.proto)]);
285 format!("ok tick time={} expired conns={} nat={}",self.clock,old-self.conns.len(),old_nat-self.pats.len())
286 }
287 "PACKET" => {
288 let i=self.iface(w[1])?;
289 if w[2]!="TCP" && w[3]!="-" { return Err(format!("flags not allowed for {}",w[2])); }
290 if w[2]=="TCP" && w[3]!="-" && (w[3].chars().any(|c| !"SAFR".contains(c)) || w[3].chars().any(|c| w[3].matches(c).count()>1)) { return Err(format!("bad flags {}",w[3])); }
291 let src=ip(w[4])?;
292 if !(0..=65535).contains(&n(5)) { return Err(format!("bad port {}",w[5])); }
293 let dst=ip(w[6])?;
294 if !(0..=65535).contains(&n(7)) { return Err(format!("bad port {}",w[7])); }
295 if !(1..=255).contains(&n(8)) { return Err(format!("bad ttl {}",w[8])); }
296 self.packet_number+=1; self.ifaces[i].rx+=1;
297 let (verdict,reason)=self.packet(i,w[3],Tuple { proto:w[2].into(),src,sport:n(5),dst,dport:n(7) },n(8));
298 if reason==-1 { self.forwarded+=1; } else if reason==-2 { self.local+=1; } else { self.drops[reason as usize]+=1; self.ifaces[i].drop+=1; }
299 format!("#{} {} {} {}:{} > {}:{} {verdict}",self.packet_number,w[2],w[3],w[4],w[5],w[6],w[7])
300 }
301 "SHOW" => return Ok(self.show(w[1])),
302 "CLEAR" => if w[1]=="CONNS" {
303 let text=format!("ok clear conns={} nat={}",self.conns.len(),self.pats.len()); self.conns.clear(); self.pats.clear(); text
304 } else {
305 self.forwarded=0; self.local=0; self.drops=[0;9]; self.errors=0;
306 for i in &mut self.ifaces { i.rx=0; i.tx=0; i.drop=0; }
307 for r in &mut self.routes { r.hits=0; } for r in &mut self.rules { r.hits=0; } self.defaults.fill([0;2]);
308 "ok clear counters".into()
309 },
310 _ => unreachable!(),
311 };
312 Ok(vec![single])
313 }
314 fn show(&self, what: &str) -> Vec<String> {
315 let mut lines=Vec::new();
316 let mut names: Vec<_>=(0..self.ifaces.len()).collect(); names.sort_by_key(|&i| &self.ifaces[i].name);
317 match what {
318 "IFACES" => {
319 lines.push(format!("show ifaces count={}",self.ifaces.len()));
320 for i in names { let f=&self.ifaces[i]; let mut text=format!("{} {}/{} {} {} rpf={}",f.name,address(f.address),f.length,if f.up { "up" } else { "down" },if f.outside { "outside" } else { "inside" },f.rpf); if let Some(pool)=f.pool { text+=&format!(" pat={}/{pool}",self.pats.iter().filter(|p| p.iface==i).count()); } lines.push(text); }
321 }
322 "ROUTES" => {
323 lines.push(format!("show routes count={}",self.routes.len()));
324 let mut routes: Vec<_>=self.routes.iter().enumerate().collect(); routes.sort_by_key(|(i,r)| (r.dest.network,r.dest.length,r.metric,*i));
325 for (_,r) in routes { lines.push(format!("{} hits={}{}",self.route_text(r),r.hits,if self.ifaces[r.iface].up { "" } else { " inactive" })); }
326 }
327 "ACL" => {
328 lines.push(format!("show acl count={}",self.rules.len()));
329 for i in names { for d in 0..2 {
330 let mut rules: Vec<_>=self.rules.iter().filter(|r| r.iface==i && r.dir==d).collect(); rules.sort_by_key(|r| r.seq); if rules.is_empty() { continue; }
331 let name=&self.ifaces[i].name; let dir=if d==0 { "in" } else { "out" };
332 for r in rules { lines.push(format!("acl {name} {dir} {} {} {} {} {} {} {}{} hits={}",r.seq,r.action,r.proto,r.src.acl_notation(),r.sports.notation(),r.dst.acl_notation(),r.dports.notation(),if r.established { " established" } else { "" },r.hits)); }
333 lines.push(format!("acl {name} {dir} default DENY hits={}",self.defaults[i][d]));
334 } }
335 }
336 "NAT" => {
337 lines.push(format!("show nat static={} forward={} pat={}",self.statics.len(),self.forwards.len(),self.pats.len()));
338 let mut statics=self.statics.clone(); statics.sort_by_key(|p| p.0); for (a,b) in statics { lines.push(format!("static {} {}",address(a),address(b))); }
339 let mut forwards: Vec<_>=self.forwards.iter().collect();
340 forwards.sort_by(|a,b| self.ifaces[a.iface].name.cmp(&self.ifaces[b.iface].name).then(a.port.cmp(&b.port)).then(a.proto.cmp(&b.proto)));
341 for f in forwards { lines.push(format!("forward {} {} {} {}:{}",self.ifaces[f.iface].name,f.proto,f.port,address(f.inside),f.inport)); }
342 let mut pats: Vec<_>=self.pats.iter().collect();
343 pats.sort_by(|a,b| self.ifaces[a.iface].name.cmp(&self.ifaces[b.iface].name).then(a.port.cmp(&b.port)));
344 for p in pats { lines.push(format!("pat {} {} {} {}:{} idle={}",self.ifaces[p.iface].name,p.port,p.proto,address(p.inside),p.inport,self.clock-p.used)); }
345 }
346 "CONNS" => {
347 lines.push(format!("show conns count={}",self.conns.len()));
348 for c in &self.conns { lines.push(format!("conn {} {} {} reply {} in={} out={} state={} pkts={}/{} idle={}",c.number,c.orig.proto,c.orig.notation(),c.reply.notation(),self.ifaces[c.input].name,self.ifaces[c.output].name,c.state,c.packets[0],c.packets[1],self.clock-c.used)); }
349 }
350 _ => {
351 let dropped:i64=self.drops.iter().sum();
352 lines.push(format!("show counters packets={} forwarded={} local={} dropped={dropped} errors={}",self.forwarded+self.local+dropped,self.forwarded,self.local,self.errors));
353 lines.push(format!("drops {}",REASONS.iter().zip(self.drops).map(|(r,n)| format!("{r}={n}")).collect::<Vec<_>>().join(" ")));
354 for i in names { let f=&self.ifaces[i]; lines.push(format!("iface {} rx={} tx={} drop={}",f.name,f.rx,f.tx,f.drop)); }
355 }
356 }
357 lines
358 }
359}
360fn main() {
361 let mut input=String::new(); io::stdin().read_to_string(&mut input).unwrap();
362 let mut lines=input.lines(); let count:usize=lines.next().unwrap().parse().unwrap();
363 let mut router=Router::new();
364 for line in lines.take(count) {
365 match router.command(&line.split(' ').collect::<Vec<_>>()) {
366 Ok(output) => for line in output { println!("{line}"); },
367 Err(reason) => { router.errors+=1; println!("error: {reason}"); }
368 }
369 }
