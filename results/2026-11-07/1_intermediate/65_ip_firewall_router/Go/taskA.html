17func decimal(s string) bool {
18 if s == "" || len(s) > 1 && s[0] == '0' {
19 return false
20 }
21 for _, c := range s {
22 if c < '0' || c > '9' {
23 return false
24 }
25 }
26 return true
27}

29func number(s string) int {
30 n, _ := strconv.Atoi(s)
31 return n
32}
33
34func parseAddress(s string) (uint32, bool) {
35 parts := strings.Split(s, ".")
36 if len(parts) != 4 {
37 return 0, false
38 }
39 var a uint32
40 for _, p := range parts {
41 if len(p) > 3 || !decimal(p) || number(p) > 255 {
42 return 0, false
43 }
44 a = a*256 + uint32(number(p))
45 }
46 return a, true
47}
48
49func address(a uint32) string {
50 return fmt.Sprintf("%d.%d.%d.%d", a>>24, (a>>16)&255, (a>>8)&255, a&255)
51}
52
53func parsePrefix(s string) (prefix, bool) {
54 parts := strings.Split(s, "/")
55 if len(parts) != 2 || len(parts[1]) > 2 || !decimal(parts[1]) || number(parts[1]) > 32 {
56 return prefix{}, false
57 }
58 a, ok := parseAddress(parts[0])
59 return prefix{a, number(parts[1])}, ok
60}
61
62func network(a uint32, length int) uint32 {
63 if length == 0 {
64 return 0
65 }
66 return a & (^uint32(0) << uint(32-length))
67}
68
69func (p prefix) contains(a uint32) bool {
70 return network(a, p.length) == network(p.address, p.length)
71}
72
73func (p prefix) String() string {
74 return fmt.Sprintf("%s/%d", address(p.address), p.length)
75}
76
77func checkedAddress(s string) (uint32, error) {
78 a, ok := parseAddress(s)
79 if !ok {
80 return 0, fmt.Errorf("bad address %s", s)
81 }
82 return a, nil
83}
84
85func checkedPrefix(s string) (prefix, error) {
86 p, ok := parsePrefix(s)
87 if !ok {
88 return p, fmt.Errorf("bad prefix %s", s)
89 }
90 if network(p.address, p.length) != p.address {
91 return p, fmt.Errorf("host bits set in %s", s)
92 }
93 return p, nil
94}

96type portRange struct {
97 lo, hi int
98 any bool
99}
100
101func parsePorts(s string) (portRange, error) {
102 if s == "ANY" {
103 return portRange{0, 65535, true}, nil
104 }
105 parts := strings.Split(s, "-")
106 lo, hi := number(parts[0]), number(parts[len(parts)-1])
107 if lo > 65535 || hi > 65535 || lo > hi {
108 return portRange{}, fmt.Errorf("bad port range %s", s)
109 }
110 return portRange{lo, hi, false}, nil
111}
112
113func (p portRange) String() string {
114 if p.any {
115 return "any"
116 }
117 if p.lo == p.hi {
118 return strconv.Itoa(p.lo)
119 }
120 return fmt.Sprintf("%d-%d", p.lo, p.hi)
121}

230func (r *router) selectRoute(a uint32) *route {
231 var best *route
232 for _, v := range r.routes {
233 if v.out.up && v.p.contains(a) && (best == nil || v.p.length > best.p.length || v.p.length == best.p.length && v.metric < best.metric) {
234 best = v
235 }
236 }
237 return best
238}

255func (r *router) acl(i *iface, d int, t tuple, existing bool) string {
256 if len(i.rules[d]) == 0 {
257 return ""
258 }
259 for _, a := range i.rules[d] {
260 if a.proto != "ANY" && a.proto != t.proto || !a.src.contains(t.src) || !a.dst.contains(t.dst) || a.established && !existing {
261 continue
262 }
263 if (a.proto == "TCP" || a.proto == "UDP") && (t.sport < a.sports.lo || t.sport > a.sports.hi || t.dport < a.dports.lo || t.dport > a.dports.hi) {
264 continue
265 }
266 a.hits++
267 if a.action == "PERMIT" {
268 return ""
269 }
270 dir := []string{"in", "out"}[d]
271 if a.action == "DENY" {
272 return fmt.Sprintf("DROP acl %s %s %d", i.name, dir, a.seq)
273 }
274 reason := "icmp-prohibited"
275 if t.proto == "TCP" {
276 reason = "tcp-reset"
277 }
278 return fmt.Sprintf("REJECT acl %s %s %d %s", i.name, dir, a.seq, reason)
279 }
280 i.defaults[d]++
281 return fmt.Sprintf("DROP acl %s %s default", i.name, []string{"in", "out"}[d])
282}

284func martian(a uint32) bool {
285 return a>>24 == 0 || a>>24 == 127 || a>>29 == 7
286}

288func (r *router) packetVerdict(i *iface, t tuple, flags string, ttl int) (string, string) {
289 drop := func(reason string) (string, string) {
290 return "DROP " + reason, reason
291 }
292 if !i.up {
293 return drop("down")
294 }
295 if martian(t.src) || martian(t.dst) || r.localAddress(t.src) != nil {
296 return drop("martian")
297 }
298 if i.rpf != "off" {
299 back := r.selectRoute(t.src)
300 if back == nil || i.rpf == "strict" && back.out != i {
301 return drop("rpf")
302 }
303 }
304 pure := t.proto == "TCP" && flags == "S"
305 var conn *connection
306 way := "new"
307 for n, c := range r.conns {
308 if c.orig == t {
309 conn = c
310 way = "orig"
311 } else if c.reply == t {
312 conn = c
313 way = "reply"
314 }
315 if conn != nil {
316 if conn.state == "CLOSED" && pure {
317 r.conns = append(r.conns[:n], r.conns[n+1:]...)
318 conn = nil
319 way = "new"
320 }
321 break
322 }
323 }
324 if t.proto == "TCP" && conn == nil && !pure {
325 return drop("invalid")
326 }
327 if verdict := r.acl(i, 0, t, conn != nil); verdict != "" {
328 return verdict, "acl_in"
329 }
330 translated := t
331 var used *pat
332 if i.outside {
333 matched := false
334 for _, m := range r.statics {
335 if m.outside == t.dst {
336 translated.dst = m.inside
337 matched = true
338 break
339 }
340 }
341 if !matched && t.dst == i.p.address {
342 for _, f := range r.forwards {
343 if f.out == i && f.proto == t.proto && f.port == t.dport {
344 translated.dst = f.inside
345 translated.dport = f.inport
346 matched = true
347 break
348 }
349 }
350 if !matched {
351 for _, m := range i.mappings {
352 if m.proto == t.proto && m.outport == t.dport {
353 translated.dst = m.inside
354 translated.dport = m.inport
355 used = m
356 break
357 }
358 }
359 }
360 }
361 }
362 if local := r.localAddress(translated.dst); local != nil {
363 return "LOCAL " + local.name, "local"
364 }
365 if ttl == 1 {
366 return drop("ttl")
367 }
368 route := r.selectRoute(translated.dst)
369 if route == nil {
370 return drop("noroute")
371 }
372 o := route.out
373 if verdict := r.acl(o, 1, translated, conn != nil); verdict != "" {
374 return verdict, "acl_out"
375 }
376 if !i.outside && o.outside {
377 matched := false
378 for _, m := range r.statics {
379 if m.inside == t.src {
380 translated.src = m.outside
381 matched = true
382 break
383 }
384 }
385 if !matched {
386 for _, f := range r.forwards {
387 if f.out == o && f.proto == t.proto && f.inside == t.src && f.inport == t.sport {
388 translated.src = o.p.address
389 translated.sport = f.port
390 matched = true
391 break
392 }
393 }
394 }
395 if !matched && o.pool > 0 {
396 for _, m := range o.mappings {
397 if m.proto == t.proto && m.inside == t.src && m.inport == t.sport {
398 used = m
399 break
400 }
401 }
402 if used == nil {
403 port := 20000
404 for ; port < 20000+o.pool; port++ {
405 busy := false
406 for _, m := range o.mappings {
407 if m.outport == port {
408 busy = true
409 }
410 }
411 for _, f := range r.forwards {
412 if f.out == o && f.port == port {
413 busy = true
414 }
415 }
416 if !busy {
417 break
418 }
419 }
420 if port == 20000+o.pool {
421 return drop("exhausted")
422 }
423 used = &pat{t.proto, t.src, t.sport, port, r.clock}
424 o.mappings = append(o.mappings, used)
425 }
426 translated.src = o.p.address
427 translated.sport = used.outport
428 }
429 }
430 if used != nil {
431 used.last = r.clock
432 }
433 if conn == nil {
434 r.connNumber++
435 state := "UNREPLIED"
436 if t.proto == "TCP" {
437 state = "SYN_SENT"
438 }
439 conn = &connection{number: r.connNumber, orig: t, reply: tuple{t.proto, translated.dst, translated.dport, translated.src, translated.sport}, in: i, out: o, state: state}
440 r.conns = append(r.conns, conn)
441 way = "new"
442 }
443 if way == "reply" {
444 conn.replyCount++
445 } else {
446 conn.origCount++
447 }
448 conn.last = r.clock
449 if t.proto != "TCP" {
450 if way == "reply" {
451 conn.state = "REPLIED"
452 }
453 } else if strings.Contains(flags, "R") {
454 conn.state = "CLOSED"
455 } else {
456 if conn.state == "SYN_SENT" && way == "reply" && strings.Contains(flags, "S") && strings.Contains(flags, "A") {
457 conn.state = "ESTABLISHED"
458 }
459 if strings.Contains(flags, "F") && (conn.state == "ESTABLISHED" || conn.state == "FIN_WAIT") {
460 side := 0
461 if way == "reply" {
462 side = 1
463 }
464 conn.fin[side] = true
465 if conn.fin[0] && conn.fin[1] {
466 conn.state = "CLOSED"
467 } else {
468 conn.state = "FIN_WAIT"
469 }
470 }
471 }
472 o.tx++
473 route.hits++
474 via := "direct"
475 if route.hop != nil {
476 via = "via " + address(*route.hop)
477 }
478 v := fmt.Sprintf("FORWARD %s %s ttl=%d", o.name, via, ttl-1)
479 if translated.dst != t.dst || translated.dport != t.dport {
480 v += fmt.Sprintf(" dnat=%s:%d", address(translated.dst), translated.dport)
481 }
482 if translated.src != t.src || translated.sport != t.sport {
483 v += fmt.Sprintf(" snat=%s:%d", address(translated.src), translated.sport)
484 }
485 v += fmt.Sprintf(" conn=%d %s", conn.number, way)
486 return v, "forwarded"
487}

489func (r *router) expire() (int, int) {
490 old := len(r.conns)
491 kept := r.conns[:0]
492 for _, c := range r.conns {
493 timeout := r.timeouts[c.orig.proto]
494 if c.orig.proto == "TCP" {
495 if c.state == "SYN_SENT" || c.state == "FIN_WAIT" {
496 timeout = 60
497 } else if c.state == "CLOSED" {
498 timeout = 10
499 }
500 }
501 if r.clock-c.last < timeout {
502 kept = append(kept, c)
503 }
504 }
505 r.conns = kept
506 removed := 0
507 for _, i := range r.ifaces {
508 ms := i.mappings[:0]
509 for _, m := range i.mappings {
510 if r.clock-m.last >= r.timeouts[m.proto] {
511 removed++
512 } else {
513 ms = append(ms, m)
514 }
515 }
516 i.mappings = ms
517 }
518 return old - len(kept), removed
519}

521func (r *router) show(what string) string {
522 out := []string{}
523 add := func(format string, args ...any) {
524 out = append(out, fmt.Sprintf(format, args...))
525 }
526 is := r.sortedIfaces()
527 switch what {
528 case "IFACES":
529 add("show ifaces count=%d", len(is))
530 for _, i := range is {
531 status, role := "down", "inside"
532 if i.up {
533 status = "up"
534 }
535 if i.outside {
536 role = "outside"
537 }
538 s := fmt.Sprintf("%s %s %s %s rpf=%s", i.name, i.p, status, role, i.rpf)
539 if i.pool > 0 {
540 s += fmt.Sprintf(" pat=%d/%d", len(i.mappings), i.pool)
541 }
542 out = append(out, s)
543 }
544 case "ROUTES":
545 rs := append([]*route{}, r.routes...)
546 sort.SliceStable(rs, func(i, j int) bool {
547 a, b := rs[i], rs[j]
548 if a.p.address != b.p.address {
549 return a.p.address < b.p.address
550 }
551 if a.p.length != b.p.length {
552 return a.p.length < b.p.length
553 }
554 return a.metric < b.metric
555 })
556 add("show routes count=%d", len(rs))
557 for _, v := range rs {
558 s := fmt.Sprintf("%s hits=%d", v, v.hits)
559 if !v.out.up {
560 s += " inactive"
561 }
562 out = append(out, s)
563 }
564 case "ACL":
565 count := 0
566 for _, i := range is {
567 count += len(i.rules[0]) + len(i.rules[1])
568 }
569 add("show acl count=%d", count)
570 for _, i := range is {
571 for d, rules := range i.rules {
572 dir := []string{"in", "out"}[d]
573 for _, a := range rules {
574 src, dst := "any", "any"
575 if a.src.length > 0 {
576 src = a.src.String()
577 }
578 if a.dst.length > 0 {
579 dst = a.dst.String()
580 }
581 est := ""
582 if a.established {
583 est = " established"
584 }
585 add("acl %s %s %d %s %s %s %s %s %s%s hits=%d", i.name, dir, a.seq, a.action, a.proto, src, a.sports, dst, a.dports, est, a.hits)
586 }
587 if len(rules) > 0 {
588 add("acl %s %s default DENY hits=%d", i.name, dir, i.defaults[d])
589 }
590 }
591 }
592 case "NAT":
593 count := 0
594 for _, i := range is {
595 count += len(i.mappings)
596 }
597 add("show nat static=%d forward=%d pat=%d", len(r.statics), len(r.forwards), count)
598 ss := append([]staticNAT{}, r.statics...)
599 sort.Slice(ss, func(i, j int) bool {
600 return ss[i].inside < ss[j].inside
601 })
602 for _, s := range ss {
603 add("static %s %s", address(s.inside), address(s.outside))
604 }
605 fs := append([]*forward{}, r.forwards...)
606 sort.Slice(fs, func(i, j int) bool {
607 a, b := fs[i], fs[j]
608 if a.out.name != b.out.name {
609 return a.out.name < b.out.name
610 }
611 if a.port != b.port {
612 return a.port < b.port
613 }
614 return a.proto < b.proto
615 })
616 for _, f := range fs {
617 add("forward %s %s %d %s:%d", f.out.name, f.proto, f.port, address(f.inside), f.inport)
618 }
619 for _, i := range is {
620 ms := append([]*pat{}, i.mappings...)
621 sort.Slice(ms, func(i, j int) bool {
622 return ms[i].outport < ms[j].outport
623 })
624 for _, m := range ms {
625 add("pat %s %d %s %s:%d idle=%d", i.name, m.outport, m.proto, address(m.inside), m.inport, r.clock-m.last)
626 }
627 }
628 case "CONNS":
629 add("show conns count=%d", len(r.conns))
630 for _, c := range r.conns {
631 a, b := c.orig, c.reply
632 add("conn %d %s %s:%d > %s:%d reply %s:%d > %s:%d in=%s out=%s state=%s pkts=%d/%d idle=%d", c.number, a.proto, address(a.src), a.sport, address(a.dst), a.dport, address(b.src), b.sport, address(b.dst), b.dport, c.in.name, c.out.name, c.state, c.origCount, c.replyCount, r.clock-c.last)
633 }
634 case "COUNTERS":
635 dropped := 0
636 for _, n := range r.drops {
637 dropped += n
638 }
639 add("show counters packets=%d forwarded=%d local=%d dropped=%d errors=%d", r.forwarded+r.local+dropped, r.forwarded, r.local, dropped, r.errors)
640 s := "drops"
641 for _, reason := range reasons {
642 s += fmt.Sprintf(" %s=%d", reason, r.drops[reason])
643 }
644 out = append(out, s)
645 for _, i := range is {
646 add("iface %s rx=%d tx=%d drop=%d", i.name, i.rx, i.tx, i.drop)
647 }
648 }
649 return strings.Join(out, "\n")
650}

652func (r *router) command(w []string) (string, error) {
653 bad := func(format string, args ...any) (string, error) {
654 return "", fmt.Errorf(format, args...)
655 }
656 cmd := w[0]
657 switch cmd {
658 case "IFACE":
659 name := w[1]
660 if r.byName[name] != nil {
661 return bad("duplicate interface %s", name)
662 }
663 p, ok := parsePrefix(w[2])
664 if !ok || p.length < 8 || p.length > 30 {
665 return bad("bad prefix %s", w[2])
666 }
667 net := network(p.address, p.length)
668 if p.address == net || p.address == net|(^uint32(0)>>uint(p.length)) {
669 return bad("bad host address %s", w[2])
670 }
671 for _, i := range r.ifaces {
672 k := p.length
673 if i.p.length < k {
674 k = i.p.length
675 }
676 if network(p.address, k) == network(i.p.address, k) {
677 return bad("subnet overlaps %s", i.name)
678 }
679 }
680 i := &iface{name: name, p: p, up: true, rpf: "off"}
681 r.ifaces = append(r.ifaces, i)
682 r.byName[name] = i
683 r.routes = append(r.routes, &route{kind: "C", p: prefix{net, p.length}, out: i})
684 return fmt.Sprintf("ok iface %s %s network %s/%d", name, w[2], address(net), p.length), nil
685 case "UP", "DOWN":
686 i, err := r.interfaceNamed(w[1])
687 if err != nil {
688 return "", err
689 }
690 status := strings.ToLower(cmd)
691 if i.up == (cmd == "UP") {
692 return bad("interface %s already %s", i.name, status)
693 }
694 i.up = cmd == "UP"
695 return "ok " + status + " " + i.name, nil
696 case "ROUTE":
697 p, err := checkedPrefix(w[1])
698 if err != nil {
699 return "", err
700 }
701 var out *iface
702 var hop *uint32
703 if w[2] == "VIA" {
704 a, e := checkedAddress(w[3])
705 if e != nil {
706 return "", e
707 }
708 hop = &a
709 } else {
710 out, err = r.interfaceNamed(w[3])
711 if err != nil {
712 return "", err
713 }
714 }
715 metric := number(w[4])
716 if metric < 1 || metric > 255 {
717 return bad("bad metric %s", w[4])
718 }
719 if hop != nil {
720 if r.localAddress(*hop) != nil {
721 return bad("nexthop %s is local", w[3])
722 }
723 for _, i := range r.ifaces {
724 if i.p.contains(*hop) {
725 out = i
726 break
727 }
728 }
729 if out == nil {
730 return bad("nexthop %s unreachable", w[3])
731 }
732 }
733 for _, v := range r.routes {
734 sameHop := v.hop == nil && hop == nil || v.hop != nil && hop != nil && *v.hop == *hop
735 if v.kind == "S" && v.p == p && v.out == out && sameHop {
736 return bad("duplicate route %s", w[1])
737 }
738 }
739 v := &route{kind: "S", p: p, out: out, hop: hop, metric: metric}
740 r.routes = append(r.routes, v)
741 return "ok route " + v.String(), nil
742 case "NOROUTE":
743 p, err := checkedPrefix(w[1])
744 if err != nil {
745 return "", err
746 }
747 count := 0
748 for _, v := range r.routes {
749 if v.kind == "S" && v.p == p {
750 count++
751 }
752 }
753 if count == 0 {
754 return bad("no static route %s", w[1])
755 }
756 kept := r.routes[:0]
757 for _, v := range r.routes {
758 if v.kind != "S" || v.p != p {
759 kept = append(kept, v)
760 }
761 }
762 r.routes = kept
763 return fmt.Sprintf("ok noroute %s removed=%d", w[1], count), nil
764 case "LOOKUP":
765 a, err := checkedAddress(w[1])
766 if err != nil {
767 return "", err
768 }
769 v := r.selectRoute(a)
770 s := "none"
771 if v != nil {
772 s = v.String()
773 }
774 return "lookup " + w[1] + " " + s, nil
775 case "ACL":
776 i, err := r.interfaceNamed(w[1])
777 if err != nil {
778 return "", err
779 }
780 seq := number(w[3])
781 if seq < 1 || seq > 9999 {
782 return bad("bad sequence %s", w[3])
783 }
784 read := func(s string) (prefix, error) {
785 if s == "ANY" {
786 return prefix{}, nil
787 }
788 return checkedPrefix(s)
789 }
790 src, err := read(w[6])
791 if err != nil {
792 return "", err
793 }
794 sports, err := parsePorts(w[7])
795 if err != nil {
796 return "", err
797 }
798 dst, err := read(w[8])
799 if err != nil {
800 return "", err
801 }
802 dports, err := parsePorts(w[9])
803 if err != nil {
804 return "", err
805 }
806 if (w[5] == "ICMP" || w[5] == "ANY") && (!sports.any || !dports.any) {
807 return bad("ports not allowed for %s", w[5])
808 }
809 d := direction(w[2])
810 for _, a := range i.rules[d] {
811 if a.seq == seq {
812 return bad("duplicate sequence %s", w[3])
813 }
814 }
815 i.rules[d] = append(i.rules[d], &rule{seq: seq, action: w[4], proto: w[5], src: src, dst: dst, sports: sports, dports: dports, established: len(w) == 11})
816 sort.Slice(i.rules[d], func(a, b int) bool {
817 return i.rules[d][a].seq < i.rules[d][b].seq
818 })
819 return fmt.Sprintf("ok acl %s %s %s", i.name, strings.ToLower(w[2]), w[3]), nil
820 case "NOACL":
821 i, err := r.interfaceNamed(w[1])
822 if err != nil {
823 return "", err
824 }
825 d := direction(w[2])
826 for n, a := range i.rules[d] {
827 if a.seq == number(w[3]) {
828 i.rules[d] = append(i.rules[d][:n], i.rules[d][n+1:]...)
829 return fmt.Sprintf("ok noacl %s %s %s hits=%d", i.name, strings.ToLower(w[2]), w[3], a.hits), nil
830 }
831 }
832 return bad("no rule %s on %s %s", w[3], i.name, strings.ToLower(w[2]))
833 case "NAT":
834 switch w[1] {
835 case "STATIC":
836 a, err := checkedAddress(w[2])
837 if err != nil {
838 return "", err
839 }
840 b, err := checkedAddress(w[3])
841 if err != nil {
842 return "", err
843 }
844 if r.localAddress(a) != nil {
845 return bad("address %s is local", w[2])
846 }
847 if r.localAddress(b) != nil {
848 return bad("address %s is local", w[3])
849 }
850 for _, m := range r.statics {
851 if m.inside == a {
852 return bad("duplicate static %s", w[2])
853 }
854 }
855 for _, m := range r.statics {
856 if m.outside == b {
857 return bad("duplicate static %s", w[3])
858 }
859 }
860 r.statics = append(r.statics, staticNAT{a, b})
861 return "ok nat static " + w[2] + " " + w[3], nil
862 case "OUTSIDE", "PAT", "FORWARD":
863 i, err := r.interfaceNamed(w[2])
864 if err != nil {
865 return "", err
866 }
867 switch w[1] {
868 case "OUTSIDE":
869 if i.outside {
870 return bad("interface %s already outside", i.name)
871 }
872 i.outside = true
873 return "ok nat outside " + i.name, nil
874 case "PAT":
875 n := number(w[3])
876 if n < 1 || n > 1000 {
877 return bad("bad pool size %s", w[3])
878 }
879 if i.pool > 0 {
880 return bad("pat already on %s", i.name)
881 }
882 i.pool = n
883 i.outside = true
884 return fmt.Sprintf("ok nat pat %s %s 20000-%d", i.name, address(i.p.address), 19999+n), nil
885 case "FORWARD":
886 port := number(w[4])
887 if port < 1 || port > 65535 {
888 return bad("bad port %s", w[4])
889 }
890 a, err := checkedAddress(w[5])
891 if err != nil {
892 return "", err
893 }
894 inport := number(w[6])
895 if inport < 1 || inport > 65535 {
896 return bad("bad port %s", w[6])
897 }
898 if !i.outside {
899 return bad("interface %s not outside", i.name)
900 }
901 if r.localAddress(a) != nil {
902 return bad("address %s is local", w[5])
903 }
904 for _, f := range r.forwards {
905 if f.out == i && f.proto == w[3] && f.port == port {
906 return bad("duplicate forward %s %s %s", i.name, w[3], w[4])
907 }
908 }
909 r.forwards = append(r.forwards, &forward{i, w[3], port, a, inport})
910 return fmt.Sprintf("ok nat forward %s %s %s %s:%s", i.name, w[3], w[4], w[5], w[6]), nil
911 }
912 }
913 case "RPF":
914 i, err := r.interfaceNamed(w[1])
915 if err != nil {
916 return "", err
917 }
918 i.rpf = strings.ToLower(w[2])
919 return "ok rpf " + i.name + " " + i.rpf, nil
920 case "TIMEOUT":
921 n := number(w[2])
922 if n < 1 || n > 86400 {
923 return bad("bad timeout %s", w[2])
924 }
925 r.timeouts[w[1]] = n
926 return "ok timeout " + w[1] + " " + w[2], nil
927 case "TICK":
928 n := number(w[1])
929 if n < 1 || n > 86400 {
930 return bad("bad tick %s", w[1])
931 }
932 r.clock += n
933 c, m := r.expire()
934 return fmt.Sprintf("ok tick time=%d expired conns=%d nat=%d", r.clock, c, m), nil
935 case "PACKET":
936 i, err := r.interfaceNamed(w[1])
937 if err != nil {
938 return "", err
939 }
940 proto, flags := w[2], w[3]
941 if proto != "TCP" && flags != "-" {
942 return bad("flags not allowed for %s", proto)
943 }
944 if proto == "TCP" && flags != "-" {
945 seen := map[rune]bool{}
946 for _, c := range flags {
947 if !strings.ContainsRune("SAFR", c) || seen[c] {
948 return bad("bad flags %s", flags)
949 }
950 seen[c] = true
951 }
952 }
953 src, err := checkedAddress(w[4])
954 if err != nil {
955 return "", err
956 }
957 sp := number(w[5])
958 if sp > 65535 {
959 return bad("bad port %s", w[5])
960 }
961 dst, err := checkedAddress(w[6])
962 if err != nil {
963 return "", err
964 }
965 dp := number(w[7])
966 if dp > 65535 {
967 return bad("bad port %s", w[7])
968 }
969 ttl := number(w[8])
970 if ttl < 1 || ttl > 255 {
971 return bad("bad ttl %s", w[8])
972 }
973 r.packet++
974 i.rx++
975 v, reason := r.packetVerdict(i, tuple{proto, src, sp, dst, dp}, flags, ttl)
976 switch reason {
977 case "forwarded":
978 r.forwarded++
979 case "local":
980 r.local++
981 default:
982 r.drops[reason]++
983 i.drop++
984 }
985 return fmt.Sprintf("#%d %s %s %s:%s > %s:%s %s", r.packet, proto, flags, w[4], w[5], w[6], w[7], v), nil
986 case "SHOW":
987 return r.show(w[1]), nil
988 case "CLEAR":
989 if w[1] == "CONNS" {
990 c, m := len(r.conns), 0
991 r.conns = nil
992 for _, i := range r.ifaces {
993 m += len(i.mappings)
994 i.mappings = nil
995 }
996 return fmt.Sprintf("ok clear conns=%d nat=%d", c, m), nil
997 }
998 r.forwarded = 0
999 r.local = 0
1000 r.errors = 0
1001 r.drops = map[string]int{}
1002 for _, i := range r.ifaces {
1003 i.rx = 0
1004 i.tx = 0
1005 i.drop = 0
1006 i.defaults = [2]int{}
1007 for _, rs := range i.rules {
1008 for _, a := range rs {
1009 a.hits = 0
1010 }
1011 }
1012 }
1013 for _, v := range r.routes {
1014 v.hits = 0
1015 }
1016 return "ok clear counters", nil
1017 }
1018 return "", nil
1019}
