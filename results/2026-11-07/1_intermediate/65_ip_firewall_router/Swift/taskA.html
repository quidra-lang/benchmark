3struct Problem: Error { let reason: String }
4func ip(_ word: String) throws -> Int {
5 let fields = word.components(separatedBy: ".")
6 guard fields.count == 4 else {
7 throw Problem(reason: "bad address \(word)")
8 }
9 var a = 0
10 for field in fields {
11 guard (1...3).contains(field.count), !(field.count > 1 && field.hasPrefix("0")),
12 field.utf8.allSatisfy({ (48...57).contains($0) }), let n = Int(field),
13 n <= 255
14 else {
15 throw Problem(reason: "bad address \(word)")
16 }
17 a = a * 256 + n
18 }
19 return a
20}
21func address(_ a: Int) -> String {
22 "\(a >> 24).\((a >> 16) & 255).\((a >> 8) & 255).\(a & 255)"
23}
24func mask(_ length: Int) -> Int {
25 length == 0 ? 0 : (0xffff_ffff << (32 - length)) & 0xffff_ffff
26}
27struct Prefix: Equatable {
28 let network: Int, length: Int
29 func contains(_ a: Int) -> Bool {
30 a & mask(length) == network & mask(length)
31 }
32 var notation: String {
33 "\(address(network))/\(length)"
34 }
35 var aclNotation: String {
36 length == 0 ? "any" : notation
37 }
38}
39func prefix(_ word: String, _ hostBits: Bool = true) throws -> Prefix {
40 let fields = word.components(separatedBy: "/")
41 guard fields.count == 2, let a = try? ip(fields[0]), (1...2).contains(fields[1].count),
42 !(fields[1].count == 2 && fields[1].hasPrefix("0")), fields[1].utf8.allSatisfy({ (48...57).contains($0) }),
43 let length = Int(fields[1]), length <= 32
44 else {
45 throw Problem(reason: "bad prefix \(word)")
46 }
47 if hostBits && a & (0xffff_ffff ^ mask(length)) != 0 {
48 throw Problem(reason: "host bits set in \(word)")
49 }
50 return Prefix(network: a, length: length)
51}
52struct Ports {
53 let low: Int, high: Int, any: Bool
54 var notation: String {
55 any ? "any" : (low == high ? String(low) : "\(low)-\(high)")
56 }
57 func contains(_ n: Int) -> Bool {
58 low <= n && n <= high
59 }
60}
61func ports(_ word: String) throws -> Ports {
62 if word == "ANY" {
63 return Ports(low: 0, high: 65535, any: true)
64 }
65 let values = word.components(separatedBy: "-").map { Int($0)! }
66 let a = values[0], b = values.last!
67 guard a <= 65535 && b <= 65535 && a <= b else {
68 throw Problem(reason: "bad port range \(word)")
69 }
70 return Ports(low: a, high: b, any: false)
71}

176 var selected: Int? = nil

178 if let j = selected {
179 if r.dest.length > routes[j].dest
180 .length || (r.dest.length == routes[j].dest.length && r.metric < routes[j].metric)

183 }
184 } else {
185 selected = i
186 }

195 func acl(_ i: Int, _ d: Int, _ tuple: PacketTuple, _ existing: Bool) -> String? {
196 let list = rules.filter { $0.iface == i && $0.dir == d }.sorted { $0.seq < $1.seq }
197 if list.isEmpty {
198 return nil
199 }
200 let name = ifaces[i].name, dir = d == 0 ? "in" : "out"
201 for r in list {
202 if (r.proto != "ANY" && r.proto != tuple.proto) || !r.src.contains(tuple.src) || !r.dst
203 .contains(tuple.dst) || (r.established && !existing)
204 {
205 continue
206 }
207 if (r.proto == "TCP" || r.proto == "UDP") &&
208 (!r.sports.contains(tuple.sport) || !r.dports.contains(tuple.dport))
209 {
210 continue
211 }
212 r.hits += 1
213 if r.action == "PERMIT" {
214 return nil
215 }
216 if r.action == "DENY" {
217 return "DROP acl \(name) \(dir) \(r.seq)"
218 }
219 return "REJECT acl \(name) \(dir) \(r.seq) \(tuple.proto == "TCP" ? "tcp-reset" : "icmp-prohibited")"
220 }
221 ifaces[i].defaults[d] += 1
222 return "DROP acl \(name) \(dir) default"
223 }
224 func packet(_ input: Int, _ flags: String, _ orig: PacketTuple, _ ttl: Int) -> (String, Int) {
225 if !ifaces[input].up {
226 return ("DROP down", 0)
227 }
228 func martian(_ a: Int) -> Bool {
229 a >> 24 == 0 || a >> 24 == 127 || a >= 0xe000_0000
230 }
231 if martian(orig.src) || martian(orig.dst) || localAddress(orig.src) != nil {
232 return ("DROP martian", 1)
233 }
234 if ifaces[input].rpf != "off" {
235 guard let r = select(orig.src) else {
236 return ("DROP rpf", 2)
237 }
238 if ifaces[input].rpf == "strict" && routes[r].iface != input {
239 return ("DROP rpf", 2)
240 }
241 }
242 let pureSyn = orig.proto == "TCP" && flags == "S"
243 var found: (Int, Int)? = nil
244 for (i, c) in conns.enumerated() {
245 if c.orig == orig {
246 found = (i, 0)
247 break
248 }
249 if c.reply == orig {
250 found = (i, 1)
251 break
252 }
253 }
254 if let (i, _) = found, conns[i].state == "CLOSED" && pureSyn {
255 conns.remove(at: i)
256 found = nil
257 }
258 if orig.proto == "TCP" && found == nil && !pureSyn {
259 return ("DROP invalid", 3)
260 }
261 if let verdict = acl(input, 0, orig, found != nil) {
262 return (verdict, 4)
263 }
264 var dst = orig.dst, dport = orig.dport, used: Pat? = nil
265 if ifaces[input].outside {
266 if let pair = statics.first(where: { $0.1 == dst }) {
267 dst = pair.0
268 } else if dst == ifaces[input].address {
269 if let f = forwards
270 .first(where: { $0.iface == input && $0.proto == orig.proto && $0.port == dport })
271 {
272 dst = f.inside
273 dport = f.inport
274 } else if let p = pats
275 .first(where: { $0.iface == input && $0.proto == orig.proto && $0.port == dport })
276 {
277 dst = p.inside
278 dport = p.inport
279 used = p
280 }
281 }
282 }
283 if let i = localAddress(dst) {
284 return ("LOCAL \(ifaces[i].name)", -2)
285 }
286 if ttl == 1 {
287 return ("DROP ttl", 5)
288 }
289 guard let r = select(dst) else {
290 return ("DROP noroute", 6)
291 }
292 let output = routes[r].iface
293 let outgoing = PacketTuple(proto: orig.proto, src: orig.src, sport: orig.sport, dst: dst, dport: dport)
294 if let verdict = acl(output, 1, outgoing, found != nil) {
295 return (verdict, 7)
296 }
297 var src = orig.src, sport = orig.sport
298 if !ifaces[input].outside && ifaces[output].outside {
299 if let pair = statics.first(where: { $0.0 == src }) {
300 src = pair.1
301 } else if let f = forwards
302 .first(where: { $0.iface == output && $0.proto == orig.proto && $0.inside == src && $0.inport == sport
303 })
304 {
305 src = ifaces[output].address
306 sport = f.port
307 } else if let count = ifaces[output].pool {
308 var mapping = pats
309 .first { $0.iface == output && $0.proto == orig.proto && $0.inside == src && $0.inport == sport }
310 if mapping == nil {
311 guard let port = (20000..<(20000 + count))
312 .first(where: { p in
313 !pats.contains { $0.iface == output && $0.port == p } && !forwards
314 .contains { $0.iface == output && $0.port == p }
315 })
316 else {
317 return (
318 "DROP exhausted",
319 8
320 )
321 }
322 mapping = Pat(output, orig.proto, src, sport, port, clock)
323 pats.append(mapping!)
324 }
325 used = mapping
326 src = ifaces[output].address
327 sport = mapping!.port
328 }
329 }
330 used?.used = clock
331 let connection: Connection, side: Int, direction: String
332 if let (i, d) = found {
333 connection = conns[i]
334 side = d
335 direction = d == 0 ? "orig" : "reply"
336 } else {
337 connectionNumber += 1
338 let reply = PacketTuple(proto: orig.proto, src: dst, sport: dport, dst: src, dport: sport)
339 connection = Connection(connectionNumber, orig, reply, input, output, clock)
340 conns.append(connection)
341 side = 0
342 direction = "new"
343 }
344 connection.packets[side] += 1
345 connection.used = clock
346 if orig.proto != "TCP" {
347 if side == 1 {
348 connection.state = "REPLIED"
349 }
350 } else if flags.contains("R") {
351 connection.state = "CLOSED"
352 } else {
353 if connection.state == "SYN_SENT" && side == 1 && flags.contains("S") && flags
354 .contains("A")
355 {
356 connection.state = "ESTABLISHED"
357 }
358 if flags
359 .contains("F") &&
360 (connection.state == "ESTABLISHED" || connection.state == "FIN_WAIT")
361 {
362 connection.fin[side] = true
363 connection.state = connection.fin[0] && connection.fin[1] ? "CLOSED" : "FIN_WAIT"
364 }
365 }
366 ifaces[output].tx += 1
367 routes[r].hits += 1
368 let hop = routes[r].hop.map { "via " + address($0) } ?? "direct"
369 var verdict = "FORWARD \(ifaces[output].name) \(hop) ttl=\(ttl - 1)"
370 if dst != orig.dst || dport != orig.dport {
371 verdict += " dnat=\(address(dst)):\(dport)"
372 }
373 if src != orig.src || sport != orig.sport {
374 verdict += " snat=\(address(src)):\(sport)"
375 }
376 verdict += " conn=\(connection.number) \(direction)"
377 return (verdict, -1)
378 }
379 func command(_ w: [String]) throws -> [String] {
380 func n(_ i: Int) -> Int {
381 Int(w[i])!
382 }
383 func d(_ s: String) -> Int {
384 s == "IN" ? 0 : 1
385 }
386 func require(_ condition: Bool, _ reason: String) throws {
387 if !condition {
388 throw Problem(reason: reason)
389 }
390 }
391 let text: String
392 switch w[0] {
393 case "IFACE":
394 try require(!ifaces.contains { $0.name == w[1] }, "duplicate interface \(w[1])")
395 let p = try prefix(w[2], false)
396 try require((8...30).contains(p.length), "bad prefix \(w[2])")
397 let host = p.network & (0xffff_ffff ^ mask(p.length))
398 try require(host != 0 && host != (0xffff_ffff ^ mask(p.length)), "bad host address \(w[2])")
399 if let old = ifaces.first(where: { $0.address & mask(min($0.length, p.length)) == p.network & mask(min(
400 $0.length,
401 p.length
402 )) }) {
403 throw Problem(reason: "subnet overlaps \(old.name)")
404 }
405 let i = ifaces.count, dest = Prefix(network: p.network & mask(p.length), length: p.length)
406 ifaces.append(Interface(w[1], p.network, p.length))
407 routes.append(Route("C", dest, i, nil, 0))
408 text = "ok iface \(w[1]) \(w[2]) network \(dest.notation)"
409 case "UP", "DOWN":
410 let i = try iface(w[1]), up = w[0] == "UP"
411 try require(ifaces[i].up != up, "interface \(w[1]) already \(w[0].lowercased())")
412 ifaces[i].up = up
413 text = "ok \(w[0].lowercased()) \(w[1])"
414 case "ROUTE":
415 let dest = try prefix(w[1])
416 var hop: Int? = nil
417 var output = 0
418 if w[2] == "VIA" {
419 hop = try ip(w[3])
420 } else {
421 output = try iface(w[3])
422 }
423 try require((1...255).contains(n(4)), "bad metric \(w[4])")
424 if let hop {
425 try require(localAddress(hop) == nil, "nexthop \(w[3]) is local")
426 guard let i = ifaces
427 .firstIndex(where: { Prefix(network: $0.address, length: $0.length).contains(hop) })
428 else {
429 throw Problem(reason: "nexthop \(w[3]) unreachable")
430 }
431 output = i
432 }
433 try require(
434 !routes.contains { $0.kind == "S" && $0.dest == dest && $0.iface == output && $0.hop == hop },
435 "duplicate route \(w[1])"
436 )
437 let route = Route("S", dest, output, hop, n(4))
438 text = "ok route " + routeText(route)
439 routes.append(route)
440 case "NOROUTE":
441 let dest = try prefix(w[1]), old = routes.count
442 try require(routes.contains { $0.kind == "S" && $0.dest == dest }, "no static route \(w[1])")
443 routes.removeAll { $0.kind == "S" && $0.dest == dest }
444 text = "ok noroute \(w[1]) removed=\(old - routes.count)"
445 case "LOOKUP":
446 let a = try ip(w[1])
447 text = "lookup \(w[1]) \(select(a).map { routeText(routes[$0]) } ?? "none")"
448 case "ACL":
449 let i = try iface(w[1]), dir = d(w[2])
450 try require((1...9999).contains(n(3)), "bad sequence \(w[3])")
451 let src: Prefix
452 if w[6] == "ANY" {
453 src = Prefix(network: 0, length: 0)
454 } else {
455 src = try prefix(w[6])
456 }
457 let sports = try ports(w[7])
458 let dst: Prefix
459 if w[8] == "ANY" {
460 dst = Prefix(network: 0, length: 0)
461 } else {
462 dst = try prefix(w[8])
463 }
464 let dports = try ports(w[9])
465 try require(!["ICMP", "ANY"].contains(w[5]) || (sports.any && dports.any), "ports not allowed for \(w[5])")
466 try require(
467 !rules.contains { $0.iface == i && $0.dir == dir && $0.seq == n(3) },
468 "duplicate sequence \(w[3])"
469 )
470 rules.append(Rule(i, dir, n(3), w[4], w[5], src, sports, dst, dports, w.count == 11))
471 text = "ok acl \(w[1]) \(w[2].lowercased()) \(w[3])"
472 case "NOACL":
473 let i = try iface(w[1]), dir = d(w[2])
474 guard let at = rules.firstIndex(where: { $0.iface == i && $0.dir == dir && $0.seq == n(3) })
475 else {
476 throw Problem(reason: "no rule \(w[3]) on \(w[1]) \(w[2].lowercased())")
477 }
478 let rule = rules.remove(at: at)
479 text = "ok noacl \(w[1]) \(w[2].lowercased()) \(w[3]) hits=\(rule.hits)"
480 case "NAT":
481 switch w[1] {
482 case "STATIC":
483 let inside = try ip(w[2]), outside = try ip(w[3])
484 try require(localAddress(inside) == nil, "address \(w[2]) is local")
485 try require(localAddress(outside) == nil, "address \(w[3]) is local")
486 try require(!statics.contains { $0.0 == inside }, "duplicate static \(w[2])")
487 try require(!statics.contains { $0.1 == outside }, "duplicate static \(w[3])")
488 statics.append((inside, outside))
489 text = "ok nat static \(w[2]) \(w[3])"
490 case "OUTSIDE":
491 let i = try iface(w[2])
492 try require(!ifaces[i].outside, "interface \(w[2]) already outside")
493 ifaces[i].outside = true
494 text = "ok nat outside \(w[2])"
495 case "PAT":
496 let i = try iface(w[2])
497 try require((1...1000).contains(n(3)), "bad pool size \(w[3])")
498 try require(ifaces[i].pool == nil, "pat already on \(w[2])")
499 ifaces[i].pool = n(3)
500 ifaces[i].outside = true
501 text = "ok nat pat \(w[2]) \(address(ifaces[i].address)) 20000-\(20000 + n(3) - 1)"
502 default:
503 let i = try iface(w[2])
504 try require((1...65535).contains(n(4)), "bad port \(w[4])")
505 let inside = try ip(w[5])
506 try require((1...65535).contains(n(6)), "bad port \(w[6])")
507 try require(ifaces[i].outside, "interface \(w[2]) not outside")
508 try require(localAddress(inside) == nil, "address \(w[5]) is local")
509 try require(
510 !forwards.contains { $0.iface == i && $0.proto == w[3] && $0.port == n(4) },
511 "duplicate forward \(w[2]) \(w[3]) \(w[4])"
512 )
513 forwards.append(Forward(iface: i, proto: w[3], port: n(4), inside: inside, inport: n(6)))
514 text = "ok nat forward \(w[2]) \(w[3]) \(w[4]) \(w[5]):\(w[6])"
515 }
516 case "RPF":
517 let i = try iface(w[1])
518 ifaces[i].rpf = w[2].lowercased()
519 text = "ok rpf \(w[1]) \(w[2].lowercased())"
520 case "TIMEOUT":
521 try require((1...86400).contains(n(2)), "bad timeout \(w[2])")
522 timeouts[protocolIndex(w[1])] = n(2)
523 text = "ok timeout \(w[1]) \(w[2])"
524 case "TICK":
525 try require((1...86400).contains(n(1)), "bad tick \(w[1])")
526 clock += n(1)
527 let old = conns.count, oldNat = pats.count
528 conns.removeAll { c in
529 let timeout: Int
530 if c.orig.proto == "TCP" {
531 switch c.state { case "SYN_SENT", "FIN_WAIT":
532 timeout = 60
533 case "CLOSED":
534 timeout = 10
535 default:
536 timeout = timeouts[0] }
537 } else {
538 timeout = timeouts[protocolIndex(c.orig.proto)]
539 }
540 return clock - c.used >= timeout
541 }
542 pats.removeAll { clock - $0.used >= timeouts[protocolIndex($0.proto)] }
543 text = "ok tick time=\(clock) expired conns=\(old - conns.count) nat=\(oldNat - pats.count)"
544 case "PACKET":
545 let i = try iface(w[1])
546 try require(w[2] == "TCP" || w[3] == "-", "flags not allowed for \(w[2])")
547 if w[2] == "TCP" && w[3] != "-" {
548 try require(
549 w[3].allSatisfy { "SAFR".contains($0) } && Set(w[3]).count == w[3].count,
550 "bad flags \(w[3])"
551 )
552 }
553 let src = try ip(w[4])
554 try require((0...65535).contains(n(5)), "bad port \(w[5])")
555 let dst = try ip(w[6])
556 try require((0...65535).contains(n(7)), "bad port \(w[7])")
557 try require((1...255).contains(n(8)), "bad ttl \(w[8])")
558 packetNumber += 1
559 ifaces[i].rx += 1
560 let (verdict, reason) = packet(
561 i,
562 w[3],
563 PacketTuple(proto: w[2], src: src, sport: n(5), dst: dst, dport: n(7)),
564 n(8)
565 )
566 if reason == -1 {
567 forwarded += 1
568 } else if reason == -2 {
569 local += 1
570 } else {
571 drops[reason] += 1
572 ifaces[i].drop += 1
573 }
574 text = "#\(packetNumber) \(w[2]) \(w[3]) \(w[4]):\(w[5]) > \(w[6]):\(w[7]) \(verdict)"
575 case "SHOW":
576 return show(w[1])
577 default:
578 if w[1] == "CONNS" {
579 text = "ok clear conns=\(conns.count) nat=\(pats.count)"
580 conns.removeAll()
581 pats.removeAll()
582 } else {
583 forwarded = 0
584 local = 0
585 drops = Array(repeating: 0, count: 9)
586 errors = 0
587 for i in ifaces {
588 i.rx = 0
589 i.tx = 0
590 i.drop = 0
591 i.defaults = [0, 0]
592 }
593 for r in routes {
594 r.hits = 0
595 }
596 for r in rules {
597 r.hits = 0
598 }
599 text = "ok clear counters"
600 }
601 }
602 return [text]
603 }
604 func show(_ what: String) -> [String] {
605 let names = ifaces.indices.sorted { ifaces[$0].name < ifaces[$1].name }
606 var lines: [String] = []
607 switch what {
608 case "IFACES":
609 lines.append("show ifaces count=\(ifaces.count)")
610 for i in names {

614 s += " pat=\(pats.filter { $0.iface == i }.count)/\(pool)"

617 }
618 case "ROUTES":
619 lines.append("show routes count=\(routes.count)")
620 let indices = routes.indices.sorted { a, b in let x = routes[a], y = routes[b]
621 if x.dest.network != y.dest.network {
622 return x.dest.network < y.dest.network
623 }
624 if x.dest.length != y.dest.length {
625 return x.dest.length < y.dest.length
626 }
627 if x.metric != y.metric {
628 return x.metric < y.metric
629 }
630 return a < b
631 }
632 for i in indices {
633 let r = routes[i]
634 lines.append(routeText(r) + " hits=\(r.hits)" + (ifaces[r.iface].up ? "" : " inactive"))
635 }
636 case "ACL":
637 lines.append("show acl count=\(rules.count)")
638 for i in names {
639 for d in 0...1 {
640 let list = rules.filter { $0.iface == i && $0.dir == d }.sorted { $0.seq < $1.seq }
641 if list.isEmpty {
642 continue
643 }
644 let name = ifaces[i].name, dir = d == 0 ? "in" : "out"
645 for r in list {
646 lines
647 .append(
648 "acl \(name) \(dir) \(r.seq) \(r.action) \(r.proto) \(r.src.aclNotation) \(r.sports.notation) \(r.dst.aclNotation) \(r.dports.notation)\(r.established ? " established" : "") hits=\(r.hits)"
649 )
650 }
651 lines.append("acl \(name) \(dir) default DENY hits=\(ifaces[i].defaults[d])")
652 }
653 }
654 case "NAT":
655 lines.append("show nat static=\(statics.count) forward=\(forwards.count) pat=\(pats.count)")
656 for (a, b) in statics.sorted(by: { $0.0 < $1.0 }) {
657 lines.append("static \(address(a)) \(address(b))")
658 }
659 for f in forwards
660 .sorted(by: { a, b in
661 if ifaces[a.iface].name != ifaces[b.iface].name {
662 return ifaces[a.iface].name < ifaces[b.iface].name
663 }
664 if a.port != b.port {
665 return a.port < b.port
666 }
667 return a.proto < b.proto
668 })
669 {
670 lines.append("forward \(ifaces[f.iface].name) \(f.proto) \(f.port) \(address(f.inside)):\(f.inport)")
671 }
672 for p in pats
673 .sorted(by: { a, b in
674 if ifaces[a.iface].name != ifaces[b.iface].name {
675 return ifaces[a.iface].name < ifaces[b.iface].name
676 }
677 return a.port < b.port
678 })
679 {
680 lines
681 .append(
682 "pat \(ifaces[p.iface].name) \(p.port) \(p.proto) \(address(p.inside)):\(p.inport) idle=\(clock - p.used)"
683 )
684 }
685 case "CONNS":
686 lines.append("show conns count=\(conns.count)")
687 for c in conns {
688 lines
689 .append(
690 "conn \(c.number) \(c.orig.proto) \(c.orig.notation) reply \(c.reply.notation) in=\(ifaces[c.input].name) out=\(ifaces[c.output].name) state=\(c.state) pkts=\(c.packets[0])/\(c.packets[1]) idle=\(clock - c.used)"
691 )
692 }
693 default:
694 let dropped = drops.reduce(0,+)
695 lines
696 .append(
697 "show counters packets=\(forwarded + local + dropped) forwarded=\(forwarded) local=\(local) dropped=\(dropped) errors=\(errors)"
698 )
699 lines
700 .append("drops " + reasons.indices.map { reasons[$0] + "=" + String(drops[$0]) }.joined(separator: " "))
701 for i in names {
702 let f = ifaces[i]
703 lines.append("iface \(f.name) rx=\(f.rx) tx=\(f.tx) drop=\(f.drop)")
704 }
705 }
706 return lines
707 }
