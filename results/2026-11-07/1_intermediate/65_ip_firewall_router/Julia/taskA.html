6function decimal(s, maxdigits)
7 1<=length(s)<=maxdigits && all(c->'0'<=c<='9', s) && (length(s)==1 || first(s)!='0') ?
8 parse(Int, s) : nothing
9end
10function address(s)
11 fields=split(s, '.'; keepempty = true)
12 length(fields)==4 || return nothing
13 a=0
14 for field in fields
15 x=decimal(field, 3)
16 (x===nothing || x>255) && return nothing
17 a=256a+x
18 end
19 a
20end
21function prefix(s)
22 parts=split(s, '/'; keepempty = true)
23 length(parts)==2 || return nothing
24 a=address(parts[1])
25 n=decimal(parts[2], 2)
26 a===nothing || n===nothing || n>32 ? nothing : (a, n)
27end
28network(a, n) = fld(a, 2^(32-n))*2^(32-n)
29inprefix(a, n, b) = network(a, n)==network(b, n)
30ip(a) = join((mod(fld(a, 2^k), 256) for k in (24, 16, 8, 0)), ".")
31cidr(a, n) = "$(ip(a))/$n"
32endpoint(a, p) = "$(ip(a)):$p"
33flowtext(t) = "$(endpoint(t[2],t[3])) > $(endpoint(t[4],t[5]))"
34function requireaddress(s)
35 a=address(s)
36 a===nothing && fail("bad address $s")
37 a
38end
39function requireprefix(s; acl = false)
40 acl && s=="ANY" && return (0, 0)
41 p=prefix(s)
42 p===nothing && fail("bad prefix $s")
43 network(p...)==p[1] || fail("host bits set in $s")
44 p
45end
46function requireint(s, lo, hi, message)
47 n=parse(Int, s)
48 lo<=n<=hi || fail(message)
49 n
50end
51function portrange(s)
52 s=="ANY" && return (0, 65535, "any")
53 b=parse.(Int, split(s, '-'))
54 lo=first(b)
55 hi=last(b)
56 lo<=hi<=65535 || fail("bad port range $s")
57 lo, hi, lo==hi ? string(lo) : "$lo-$hi"
58end

165function iface(r, name)
166 n=findfirst(i->i.name==name, r.interfaces)
167 n===nothing && fail("no interface $name")
168 r.interfaces[n]
169end
170localiface(r, a) = findfirst(i->i.addr==a, r.interfaces)
171function selectroute(r, a)
172 best=nothing
173 for route in r.routes
174 route.iface.up && inprefix(route.addr, route.length, a) || continue
175 if best===nothing ||
176 route.length>best.length ||
177 route.length==best.length && route.metric<best.metric
178 best=route
179 end
180 end
181 best
182end
183function evaluate!(i, d, t, established)
184 isempty(i.rules[d]) && return nothing
185 p, s, sp, dst, dp=t
186 for rule in i.rules[d]
187 rule.proto in ("ANY", p) && inprefix(rule.src..., s) && inprefix(rule.dst..., dst) ||
188 continue
189 rule.proto in ("TCP", "UDP") &&
190 !(rule.sports[1]<=sp<=rule.sports[2] && rule.dports[1]<=dp<=rule.dports[2]) &&
191 continue
192 rule.established && !established && continue
193 rule.hits+=1
194 label="acl $(i.name) $d $(rule.seq)"
195 return rule.action=="PERMIT" ? nothing :
196 rule.action=="DENY" ? "DROP $label" :
197 "REJECT $label "*(p=="TCP" ? "tcp-reset" : "icmp-prohibited")
198 end
199 i.defaults[d]+=1
200 "DROP acl $(i.name) $d default"
201end
202function record!(c, d, flags, clock)
203 d=="reply" ? (c.replycount+=1) : (c.origcount+=1)
204 c.used=clock
205 if c.orig[1]!="TCP"
206 d=="reply" && (c.state="REPLIED")
207 elseif occursin('R', flags)
208 c.state="CLOSED"
209 else
210 c.state=="SYN_SENT" &&
211 d=="reply" &&
212 occursin('S', flags) &&
213 occursin('A', flags) &&
214 (c.state="ESTABLISHED")
215 if occursin('F', flags) && c.state in ("ESTABLISHED", "FIN_WAIT")
216 d=="reply" ? (c.replyfin=true) : (c.origfin=true)
217 c.state=c.origfin && c.replyfin ? "CLOSED" : "FIN_WAIT"
218 end
219 end
220end
221function process!(r, i, p, flags, s, sp, d, dp, ttl)
222 !i.up && return "down", "DROP down"
223 if any(
224 x->inprefix(x[1], x[2], s) || inprefix(x[1], x[2], d),
225 ((0, 8), (2130706432, 8), (3758096384, 3)),
226 ) || localiface(r, s)!==nothing
227 return "martian", "DROP martian"
228 end
229 if i.rpf!="off"
230 route=selectroute(r, s)
231 (route===nothing || i.rpf=="strict" && route.iface!==i) && return "rpf", "DROP rpf"
232 end
233 t=(p, s, sp, d, dp)
234 pure=p=="TCP" && flags=="S"
235 c=nothing
236 direction=""
237 for (j, conn) in enumerate(r.conns)
238 dir=conn.orig==t ? "orig" : conn.reply==t ? "reply" : ""
239 isempty(dir) && continue
240 if conn.state=="CLOSED" && pure
241 deleteat!(r.conns, j)
242 else
243 c=conn
244 direction=dir
245 end
246 break
247 end
248 p=="TCP" && c===nothing && !pure && return "invalid", "DROP invalid"
249 verdict=evaluate!(i, "in", t, c!==nothing)
250 verdict!==nothing && return "acl_in", verdict
251 nd=d
252 np=dp
253 used=nothing
254 if i.outside
255 st=findfirst(x->x[2]==d, r.statics)
256 if st!==nothing
257 nd=r.statics[st][1]
258 elseif d==i.addr
259 f=findfirst(x->x.iface===i && x.proto==p && x.port==dp, r.forwards)
260 m=findfirst(x->x.proto==p && x.port==dp, i.pat)
261 if f!==nothing
262 nd=r.forwards[f].inside
263 np=r.forwards[f].inport
264 elseif m!==nothing
265 used=i.pat[m]
266 nd=used.inside
267 np=used.inport
268 end
269 end
270 end
271 loc=localiface(r, nd)
272 loc!==nothing && return "local", "LOCAL $(r.interfaces[loc].name)"
273 ttl==1 && return "ttl", "DROP ttl"
274 route=selectroute(r, nd)
275 route===nothing && return "noroute", "DROP noroute"
276 o=route.iface
277 verdict=evaluate!(o, "out", (p, s, sp, nd, np), c!==nothing)
278 verdict!==nothing && return "acl_out", verdict
279 ns=s
280 nsp=sp
281 if !i.outside && o.outside
282 st=findfirst(x->x[1]==s, r.statics)
283 f=findfirst(x->x.iface===o && x.proto==p && x.inside==s && x.inport==sp, r.forwards)
284 if st!==nothing
285 ns=r.statics[st][2]
286 elseif f!==nothing
287 ns=o.addr
288 nsp=r.forwards[f].port
289 elseif o.pool>0
290 m=findfirst(x->x.proto==p && x.inside==s && x.inport==sp, o.pat)
291 if m===nothing
292 port=findfirst(
293 x->!any(y->y.port==x, o.pat) && !any(y->y.iface===o && y.port==x, r.forwards),
294 20000:(19999+o.pool),
295 )
296 port===nothing && return "exhausted", "DROP exhausted"
297 used=Pat(p, s, sp, 19999+port, r.clock)
298 push!(o.pat, used)
299 else
300 used=o.pat[m]
301 end
302 ns=o.addr
303 nsp=used.port
304 end
305 end
306 used!==nothing && (used.used=r.clock)
307 if c===nothing
308 r.serial+=1
309 c=Connection(
310 r.serial,
311 t,
312 (p, nd, np, ns, nsp),
313 i,
314 o,
315 p=="TCP" ? "SYN_SENT" : "UNREPLIED",
316 0,
317 0,
318 r.clock,
319 false,
320 false,
321 )
322 push!(r.conns, c)
323 direction="new"
324 end
325 record!(c, direction, flags, r.clock)
326 o.tx+=1
327 route.hits+=1
328 verdict="FORWARD $(o.name) "*(route.hop===nothing ? "direct" : "via $(ip(route.hop))")*" ttl=$(ttl-1)"
329 (nd, np)!=(d, dp) && (verdict*=" dnat=$(endpoint(nd,np))")
330 (ns, nsp)!=(s, sp) && (verdict*=" snat=$(endpoint(ns,nsp))")
331 "forwarded", verdict*" conn=$(c.number) $direction"
332end
333aclprefix(p) = p[2]==0 ? "any" : cidr(p...)
334function showstate(r, what)
335 interfaces=sort(r.interfaces; by = i->i.name)
336 if what=="IFACES"
337 println("show ifaces count=$(length(interfaces))")
338 for i in interfaces
339 println(
340 "$(i.name) $(cidr(i.addr,i.length)) $(i.up ? "up" : "down") $(i.outside ? "outside" : "inside") rpf=$(i.rpf)",
341 i.pool>0 ? " pat=$(length(i.pat))/$(i.pool)" : "",
342 )
343 end
344 elseif what=="ROUTES"
345 println("show routes count=$(length(r.routes))")
346 for route in sort(r.routes; by = x->(x.addr, x.length, x.metric), alg = Base.Sort.MergeSort)
347 println(notation(route), " hits=$(route.hits)", route.iface.up ? "" : " inactive")
348 end
349 elseif what=="ACL"
350 println(
351 "show acl count=$(sum(length(i.rules[d]) for i in interfaces for d in ("in","out");init=0))",
352 )
353 for i in interfaces, d in ("in", "out")
354 isempty(i.rules[d]) && continue
355 for rule in i.rules[d]
356 println(
357 "acl $(i.name) $d $(rule.seq) $(rule.action) $(rule.proto) $(aclprefix(rule.src)) $(rule.sports[3]) $(aclprefix(rule.dst)) $(rule.dports[3])",
358 rule.established ? " established" : "",
359 " hits=$(rule.hits)",
360 )
361 end
362 println("acl $(i.name) $d default DENY hits=$(i.defaults[d])")
363 end
364 elseif what=="NAT"
365 println(
366 "show nat static=$(length(r.statics)) forward=$(length(r.forwards)) pat=$(sum(length(i.pat) for i in interfaces;init=0))",
367 )
368 for (a, b) in sort(r.statics; by = first)
369 println("static $(ip(a)) $(ip(b))")
370 end
371 for f in sort(r.forwards; by = x->(x.iface.name, x.port, x.proto))
372 println("forward $(f.iface.name) $(f.proto) $(f.port) $(endpoint(f.inside,f.inport))")
373 end
374 for i in interfaces, m in sort(i.pat; by = x->x.port)
375 println(
376 "pat $(i.name) $(m.port) $(m.proto) $(endpoint(m.inside,m.inport)) idle=$(r.clock-m.used)",
377 )
378 end
379 elseif what=="CONNS"
380 println("show conns count=$(length(r.conns))")
381 for c in r.conns
382 println(
383 "conn $(c.number) $(c.orig[1]) $(flowtext(c.orig)) reply $(flowtext(c.reply)) in=$(c.inbound.name) out=$(c.outbound.name) state=$(c.state) pkts=$(c.origcount)/$(c.replycount) idle=$(r.clock-c.used)",
384 )
385 end
386 else
387 dropped=sum(r.counts[k] for k in DROPS)
388 forwarded=r.counts["forwarded"]
389 localcount=r.counts["local"]
390 println(
391 "show counters packets=$(forwarded+localcount+dropped) forwarded=$forwarded local=$localcount dropped=$dropped errors=$(r.errors)",
392 )
393 println("drops ", join(("$k=$(r.counts[k])" for k in DROPS), " "))
394 for i in interfaces
395 println("iface $(i.name) rx=$(i.rx) tx=$(i.tx) drop=$(i.drop)")
396 end
397 end
398end
399function execute!(r, line)
400 words=String.(split(line))
401 cmd=words[1]
402 a=words[2:end]
403 if cmd=="IFACE"
404 name, word=a
405 any(i->i.name==name, r.interfaces) && fail("duplicate interface $name")
406 p=prefix(word)
407 (p===nothing || !(8<=p[2]<=30)) && fail("bad prefix $word")
408 addr, len=p
409 host=addr-network(addr, len)
410 host in (0, 2^(32-len)-1) && fail("bad host address $word")
411 for i in r.interfaces
412 inprefix(i.addr, min(len, i.length), addr) && fail("subnet overlaps $(i.name)")
413 end
414 i=Interface(name, addr, len)
415 push!(r.interfaces, i)
416 push!(r.routes, Route("C", network(addr, len), len, i, nothing, 0, 0))
417 println("ok iface $name $word network $(cidr(network(addr,len),len))")
418 elseif cmd in ("UP", "DOWN")
419 i=iface(r, a[1])
420 state=cmd=="UP"
421 word=lowercase(cmd)
422 i.up==state && fail("interface $(i.name) already $word")
423 i.up=state
424 println("ok $word $(i.name)")
425 elseif cmd=="ROUTE"
426 word, mode, target, m=a
427 addr, len=requireprefix(word)
428 hop=nothing
429 i=nothing
430 if mode=="VIA"
431 hop=requireaddress(target)
432 else
433 i=iface(r, target)
434 end
435 metric=requireint(m, 1, 255, "bad metric $m")
436 if mode=="VIA"
437 localiface(r, hop)!==nothing && fail("nexthop $target is local")
438 idx=findfirst(x->inprefix(x.addr, x.length, hop), r.interfaces)
439 idx===nothing && fail("nexthop $target unreachable")
440 i=r.interfaces[idx]
441 end
442 any(
443 x->x.kind=="S" && x.addr==addr && x.length==len && x.iface===i && x.hop==hop,
444 r.routes,
445 ) && fail("duplicate route $word")
446 route=Route("S", addr, len, i, hop, metric, 0)
447 push!(r.routes, route)
448 println("ok route ", notation(route))
449 elseif cmd=="NOROUTE"
450 addr, len=requireprefix(a[1])
451 removed=count(x->x.kind=="S" && x.addr==addr && x.length==len, r.routes)
452 removed==0 && fail("no static route $(a[1])")
453 filter!(x->!(x.kind=="S" && x.addr==addr && x.length==len), r.routes)
454 println("ok noroute $(a[1]) removed=$removed")
455 elseif cmd=="LOOKUP"
456 route=selectroute(r, requireaddress(a[1]))
457 println("lookup $(a[1]) ", route===nothing ? "none" : notation(route))
458 elseif cmd=="ACL"
459 name, d, seq, action, p, src, sports, dst, dports=a[1:9]
460 i=iface(r, name)
461 n=requireint(seq, 1, 9999, "bad sequence $seq")
462 s=requireprefix(src; acl = true)
463 sp=portrange(sports)
464 dest=requireprefix(dst; acl = true)
465 dp=portrange(dports)
466 p in ("ICMP", "ANY") && (sports!="ANY" || dports!="ANY") && fail("ports not allowed for $p")
467 dir=lowercase(d)
468 any(x->x.seq==n, i.rules[dir]) && fail("duplicate sequence $seq")
469 push!(i.rules[dir], Rule(n, action, p, s, sp, dest, dp, last(a)=="ESTABLISHED", 0))
470 sort!(i.rules[dir]; by = x->x.seq)
471 println("ok acl $name $dir $seq")
472 elseif cmd=="NOACL"
473 name, d, seq=a
474 i=iface(r, name)
475 dir=lowercase(d)
476 n=parse(Int, seq)
477 idx=findfirst(x->x.seq==n, i.rules[dir])
478 idx===nothing && fail("no rule $seq on $name $dir")
479 hits=i.rules[dir][idx].hits
480 deleteat!(i.rules[dir], idx)
481 println("ok noacl $name $dir $seq hits=$hits")
482 elseif cmd=="NAT"
483 kind=a[1]
484 a=a[2:end]
485 if kind=="STATIC"
486 x, y=a
487 inside=requireaddress(x)
488 outside=requireaddress(y)
489 for (addr, word) in ((inside, x), (outside, y))
490 localiface(r, addr)!==nothing && fail("address $word is local")
491 end
492 any(s->s[1]==inside, r.statics) && fail("duplicate static $x")
493 any(s->s[2]==outside, r.statics) && fail("duplicate static $y")
494 push!(r.statics, (inside, outside))
495 println("ok nat static $x $y")
496 elseif kind=="OUTSIDE"
497 i=iface(r, a[1])
498 i.outside && fail("interface $(i.name) already outside")
499 i.outside=true
500 println("ok nat outside $(i.name)")
501 elseif kind=="PAT"
502 name, word=a
503 i=iface(r, name)
504 n=requireint(word, 1, 1000, "bad pool size $word")
505 i.pool>0 && fail("pat already on $name")
506 i.pool=n
507 i.outside=true
508 println("ok nat pat $name $(ip(i.addr)) 20000-$(19999+n)")
509 else
510 name, p, word, insideword, inword=a
511 i=iface(r, name)
512 port=requireint(word, 1, 65535, "bad port $word")
513 inside=requireaddress(insideword)
514 inport=requireint(inword, 1, 65535, "bad port $inword")
515 i.outside || fail("interface $name not outside")
516 localiface(r, inside)!==nothing && fail("address $insideword is local")
517 any(f->f.iface===i && f.proto==p && f.port==port, r.forwards) &&
518 fail("duplicate forward $name $p $word")
519 push!(r.forwards, Forward(i, p, port, inside, inport))
520 println("ok nat forward $name $p $word $insideword:$inword")
521 end
522 elseif cmd=="RPF"
523 i=iface(r, a[1])
524 i.rpf=lowercase(a[2])
525 println("ok rpf $(i.name) $(i.rpf)")
526 elseif cmd=="TIMEOUT"
527 p, word=a
528 r.timeouts[p]=requireint(word, 1, 86400, "bad timeout $word")
529 println("ok timeout $p $word")
530 elseif cmd=="TICK"
531 word=a[1]
532 r.clock+=requireint(word, 1, 86400, "bad tick $word")
533 nc=length(r.conns)
534 nm=sum(length(i.pat) for i in r.interfaces; init = 0)
535 filter!(
536 c->r.clock-c.used < (
537 c.orig[1]!="TCP" || c.state=="ESTABLISHED" ? r.timeouts[c.orig[1]] :
538 c.state=="CLOSED" ? 10 : 60
539 ),
540 r.conns,
541 )
542 for i in r.interfaces
543 filter!(m->r.clock-m.used<r.timeouts[m.proto], i.pat)
544 end
545 println(
546 "ok tick time=$(r.clock) expired conns=$(nc-length(r.conns)) nat=$(nm-sum(length(i.pat) for i in r.interfaces;init=0))",
547 )
548 elseif cmd=="PACKET"
549 name, p, flags, sw, spw, dw, dpw, tw=a
550 i=iface(r, name)
551 p!="TCP" && flags!="-" && fail("flags not allowed for $p")
552 p=="TCP" &&
553 flags!="-" &&
554 (!all(c->c in "SAFR", flags) || length(unique(flags))!=length(flags)) &&
555 fail("bad flags $flags")
556 s=requireaddress(sw)
557 sp=requireint(spw, 0, 65535, "bad port $spw")
558 d=requireaddress(dw)
559 dp=requireint(dpw, 0, 65535, "bad port $dpw")
560 ttl=requireint(tw, 1, 255, "bad ttl $tw")
561 r.packets+=1
562 i.rx+=1
563 reason, verdict=process!(r, i, p, flags=="-" ? "" : flags, s, sp, d, dp, ttl)
564 r.counts[reason]+=1
565 reason in DROPS && (i.drop+=1)
566 println("#$(r.packets) $p $flags $sw:$spw > $dw:$dpw $verdict")
567 elseif cmd=="SHOW"
568 showstate(r, a[1])
569 elseif a[1]=="COUNTERS"
570 for k in keys(r.counts)
571 r.counts[k]=0
572 end
573 r.errors=0
574 for i in r.interfaces
575 i.rx=i.tx=i.drop=0
576 for d in ("in", "out")
577 i.defaults[d]=0
578 for rule in i.rules[d]
579 rule.hits=0
580 end
581 end
582 end
583 for route in r.routes
584 route.hits=0
585 end
586 println("ok clear counters")
587 else
588 nc=length(r.conns)
589 nm=sum(length(i.pat) for i in r.interfaces; init = 0)
590 empty!(r.conns)
591 for i in r.interfaces
592 empty!(i.pat)
593 end
594 println("ok clear conns=$nc nat=$nm")
595 end
596end
597function main()
598 r=Router()
599 n=parse(Int, readline())
600 for _ in 1:n
601 line=readline()
602 try
603 execute!(r, line)
604 catch e
605 e isa CommandError || rethrow()
606 r.errors+=1
607 println("error: $(e.message)")
608 end
609 end
610end
