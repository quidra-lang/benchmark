30std::optional<int> decimal(const std::string& text, std::size_t max_digits) {
31 if (text.empty() || text.size() > max_digits || (text.size() > 1 && text[0] == '0'))
32 return std::nullopt;
33 int value = 0;
34 for (char c : text) {
35 if (c < '0' || c > '9') return std::nullopt;
36 value = value * 10 + c - '0';
37 }
38 return value;
39}

41std::optional<Address> parse_address(const std::string& text) {
42 Address value = 0;
43 std::size_t position = 0;
44 for (int i = 0; i < 4; ++i) {
45 std::size_t end = text.find('.', position);
46 if ((i < 3) == (end == std::string::npos)) return std::nullopt;
47 std::optional<int> field = decimal(text.substr(position, end == std::string::npos
48 ? std::string::npos : end - position), 3);
49 if (!field || *field > 255) return std::nullopt;
50 value = value * 256 + static_cast<Address>(*field);
51 if (i < 3) position = end + 1;
52 }
53 return value;
54}

56std::optional<Prefix> parse_prefix(const std::string& text) {
57 std::size_t slash = text.find('/');
58 if (slash == std::string::npos || text.find('/', slash + 1) != std::string::npos)
59 return std::nullopt;
60 std::optional<Address> address = parse_address(text.substr(0, slash));
61 std::optional<int> length = decimal(text.substr(slash + 1), 2);
62 if (!address || !length || *length > 32) return std::nullopt;
63 return Prefix{*address, *length};
64}

66Address mask(int length) {
67 return length == 0 ? Address{0} : ~Address{0} << (32 - length);
68}
69
70Address network(Address address, int length) {
71 return address & mask(length);
72}
73
74bool contains(Prefix prefix, Address address) {
75 return network(prefix.address, prefix.length) == network(address, prefix.length);
76}

78std::string address_text(Address address) {
79 return std::to_string(address >> 24) + "." + std::to_string((address >> 16) & 255) + "." +
80 std::to_string((address >> 8) & 255) + "." + std::to_string(address & 255);
81}

83std::string prefix_text(Prefix prefix) {
84 return address_text(prefix.address) + "/" + std::to_string(prefix.length);
85}

87std::string endpoint(Address address, int port) {
88 return address_text(address) + ":" + std::to_string(port);

97Prefix require_prefix(const std::string& word, bool acl = false) {
98 if (acl && word == "ANY") return {0, 0};
99 std::optional<Prefix> prefix = parse_prefix(word);
100 if (!prefix) throw std::runtime_error("bad prefix " + word);
101 if (network(prefix->address, prefix->length) != prefix->address)
102 throw std::runtime_error("host bits set in " + word);
103 return *prefix;
104}

117struct Ports {
118 int low = 0, high = 65535;
119 bool any = true;
120 bool contains(int port) const { return low <= port && port <= high; }
121 std::string text() const {
122 if (any) return "any";
123 return std::to_string(low) + (low == high ? "" : "-" + std::to_string(high));
124 }
125};
126
127Ports require_ports(const std::string& word) {
128 if (word == "ANY") return Ports{};
129 std::size_t dash = word.find('-');
130 int low = std::stoi(word.substr(0, dash));
131 int high = dash == std::string::npos ? low : std::stoi(word.substr(dash + 1));
132 if (low > 65535 || high > 65535 || low > high)
133 throw std::runtime_error("bad port range " + word);
134 return {low, high, false};
135}

137struct Flow {
138 std::string protocol;
139 Address source;
140 int sport;
141 Address destination;

143
144 bool operator==(const Flow& other) const {
145 return std::tie(protocol, source, sport, destination, dport) ==
146 std::tie(other.protocol, other.source, other.sport, other.destination, other.dport);

151struct Rule {
152 int sequence;
153 std::string action, protocol;
154 Prefix source;
155 Ports sports;
156 Prefix destination;
157 Ports dports;
158 bool established;
159 int hits = 0;
160
161 bool matches(const Flow& flow, bool existing) const {
162 if (protocol != "ANY" && protocol != flow.protocol) return false;
163 if (!contains(source, flow.source) || !contains(destination, flow.destination)) return false;
164 if ((protocol == "TCP" || protocol == "UDP") &&
165 (!sports.contains(flow.sport) || !dports.contains(flow.dport))) return false;
166 return !established || existing;
167 }

216struct Connection {
217 int number;
218 Flow orig, reply;
219 std::string input, output, state;
220 int orig_packets = 0, reply_packets = 0, last_used = 0;
221 bool orig_fin = false, reply_fin = false;
222
223 void record(const std::string& direction, const std::string& flags, int clock) {
224 bool reply_side = direction == "reply";
225 if (reply_side) ++reply_packets;
226 else ++orig_packets;
227 last_used = clock;
228 if (orig.protocol != "TCP") {
229 if (reply_side) state = "REPLIED";
230 return;
231 }
232 if (flags.find('R') != std::string::npos) {
233 state = "CLOSED";
234 return;
235 }
236 if (state == "SYN_SENT" && reply_side && flags.find('S') != std::string::npos &&
237 flags.find('A') != std::string::npos) state = "ESTABLISHED";
238 if (flags.find('F') != std::string::npos && (state == "ESTABLISHED" || state == "FIN_WAIT")) {
239 if (reply_side) reply_fin = true;
240 else orig_fin = true;
241 state = orig_fin && reply_fin ? "CLOSED" : "FIN_WAIT";
242 }
243 }
244};

267 for (auto& item : interfaces) if (item.second.address == address) return &item.second;

271 Route* select(Address address) {
272 Route* best = nullptr;
273 for (Route& route : routes) {
274 if (!interfaces.at(route.interface).up || !contains(route.prefix, address)) continue;
275 if (!best || route.prefix.length > best->prefix.length ||
276 (route.prefix.length == best->prefix.length && route.metric < best->metric)) best = &route;
277 }
278 return best;
279 }

281 std::string evaluate(Interface& interface, int direction, const Flow& flow, bool existing) {
282 AccessList& acl = interface.acls[direction];
283 if (acl.rules.empty()) return "";
284 std::string label = "acl " + interface.name + (direction == 0 ? " in " : " out ");
285 for (auto& entry : acl.rules) {
286 Rule& rule = entry.second;
287 if (!rule.matches(flow, existing)) continue;
288 ++rule.hits;
289 if (rule.action == "PERMIT") return "";
290 if (rule.action == "DENY") return "DROP " + label + std::to_string(rule.sequence);
291 return "REJECT " + label + std::to_string(rule.sequence) +
292 (flow.protocol == "TCP" ? " tcp-reset" : " icmp-prohibited");
293 }
294 ++acl.default_hits;
295 return "DROP " + label + "default";
296 }

298 std::pair<Connection*, std::string> find_connection(const Flow& flow, bool pure_syn) {
299 for (auto it = connections.begin(); it != connections.end(); ++it) {
300 std::string direction;
301 if (it->orig == flow) direction = "orig";
302 else if (it->reply == flow) direction = "reply";
303 else continue;
304 if (it->state == "CLOSED" && pure_syn) {
305 connections.erase(it);
306 return {nullptr, ""};
307 }
308 return {&*it, direction};
309 }
310 return {nullptr, ""};
311 }

313 std::pair<std::string, std::string> process(Interface& input, const Flow& original,
314 const std::string& flags, int ttl) {
315 auto drop = [](const std::string& reason) {
316 return std::make_pair(reason, "DROP " + reason);
317 };
318 if (!input.up) return drop("down");
319 for (Prefix prefix : std::array<Prefix, 3>{{{0, 8}, {0x7f000000U, 8}, {0xe0000000U, 3}}})
320 if (contains(prefix, original.source) || contains(prefix, original.destination)) return drop("martian");
321 if (local(original.source)) return drop("martian");
322 if (input.rpf != "off") {
323 Route* route = select(original.source);
324 if (!route || (input.rpf == "strict" && route->interface != input.name)) return drop("rpf");
325 }
326 bool pure_syn = original.protocol == "TCP" && flags == "S";
327 std::pair<Connection*, std::string> found = find_connection(original, pure_syn);
328 Connection* connection = found.first;
329 std::string direction = found.second;
330 if (original.protocol == "TCP" && !connection && !pure_syn) return drop("invalid");
331 std::string verdict = evaluate(input, 0, original, connection != nullptr);
332 if (!verdict.empty()) return {"acl_in", verdict};
333 Flow translated = original;
334 Pat* used = nullptr;
335 if (input.outside) {
336 bool done = false;
337 for (const StaticNat& mapping : statics) {
338 if (mapping.outside != original.destination) continue;
339 translated.destination = mapping.inside;
340 done = true;
341 break;
342 }
343 if (!done && original.destination == input.address) {
344 for (const Forward& forward : forwards) {
345 if (forward.interface == input.name && forward.protocol == original.protocol &&
346 forward.outside_port == original.dport) {
347 translated.destination = forward.inside;
348 translated.dport = forward.inside_port;
349 done = true;
350 break;
351 }
352 }
353 if (!done) {
354 for (Pat& mapping : input.mappings) {
355 if (mapping.protocol == original.protocol && mapping.outside_port == original.dport) {
356 translated.destination = mapping.inside;
357 translated.dport = mapping.inside_port;
358 used = &mapping;
359 break;
360 }
361 }
362 }
363 }
364 }
365 if (Interface* destination = local(translated.destination))
366 return {"local", "LOCAL " + destination->name};
367 if (ttl == 1) return drop("ttl");
368 Route* route = select(translated.destination);
369 if (!route) return drop("noroute");
370 Interface& output = interfaces.at(route->interface);
371 verdict = evaluate(output, 1, translated, connection != nullptr);
372 if (!verdict.empty()) return {"acl_out", verdict};
373 if (!input.outside && output.outside) {
374 bool done = false;
375 for (const StaticNat& mapping : statics) {
376 if (mapping.inside != original.source) continue;
377 translated.source = mapping.outside;
378 done = true;
379 break;
380 }
381 if (!done) {
382 for (const Forward& forward : forwards) {
383 if (forward.interface == output.name && forward.protocol == original.protocol &&
384 forward.inside == original.source && forward.inside_port == original.sport) {
385 translated.source = output.address;
386 translated.sport = forward.outside_port;
387 done = true;
388 break;
389 }
390 }
391 }
392 if (!done && output.pool_size) {
393 for (Pat& mapping : output.mappings) {
394 if (mapping.protocol == original.protocol && mapping.inside == original.source &&
395 mapping.inside_port == original.sport) {
396 used = &mapping;
397 break;
398 }
399 }
400 if (!used) {
401 std::set<int> occupied;
402 for (const Pat& mapping : output.mappings) occupied.insert(mapping.outside_port);
403 for (const Forward& forward : forwards)
404 if (forward.interface == output.name) occupied.insert(forward.outside_port);
405 int port = 20000;
406 while (port < 20000 + output.pool_size && occupied.count(port)) ++port;
407 if (port == 20000 + output.pool_size) return drop("exhausted");
408 output.mappings.push_back({original.protocol, original.source, original.sport, port, clock});
409 used = &output.mappings.back();
410 }
411 translated.source = output.address;
412 translated.sport = used->outside_port;
413 }
414 }
415 if (used) used->last_used = clock;
416 if (!connection) {
417 Flow reply{original.protocol, translated.destination, translated.dport,
418 translated.source, translated.sport};
419 connections.push_back({++connection_number, original, reply, input.name, output.name,
420 original.protocol == "TCP" ? "SYN_SENT" : "UNREPLIED"});
421 connection = &connections.back();
422 direction = "new";
423 }
424 connection->record(direction, flags, clock);
425 ++output.tx;
426 ++route->hits;
427 verdict = "FORWARD " + output.name + (route->hop ? " via " + address_text(*route->hop) : " direct") +
428 " ttl=" + std::to_string(ttl - 1);
429 if (translated.destination != original.destination || translated.dport != original.dport)
430 verdict += " dnat=" + endpoint(translated.destination, translated.dport);
431 if (translated.source != original.source || translated.sport != original.sport)
432 verdict += " snat=" + endpoint(translated.source, translated.sport);
433 verdict += " conn=" + std::to_string(connection->number) + " " + direction;
434 return {"forwarded", verdict};
435 }

522 void configure(const std::vector<std::string>& w) {
523 const std::string& command = w[0];
524 if (command == "IFACE") {
525 if (interfaces.count(w[1])) throw std::runtime_error("duplicate interface " + w[1]);
526 std::optional<Prefix> prefix = parse_prefix(w[2]);
527 if (!prefix || prefix->length < 8 || prefix->length > 30) throw std::runtime_error("bad prefix " + w[2]);
528 Address host = prefix->address & ~mask(prefix->length);
529 if (host == 0 || host == ~mask(prefix->length)) throw std::runtime_error("bad host address " + w[2]);
530 for (const std::string& name : definition_order) {
531 const Interface& existing = interfaces.at(name);
532 int length = std::min(prefix->length, existing.length);
533 if (network(prefix->address, length) == network(existing.address, length))
534 throw std::runtime_error("subnet overlaps " + name);
535 }
536 Interface interface;
537 interface.name = w[1];
538 interface.address = prefix->address;
539 interface.length = prefix->length;
540 interfaces.emplace(w[1], std::move(interface));
541 definition_order.push_back(w[1]);
542 Prefix connected{network(prefix->address, prefix->length), prefix->length};
543 routes.push_back({'C', connected, w[1], std::nullopt, 0});
544 std::cout << "ok iface " << w[1] << ' ' << w[2] << " network " << prefix_text(connected) << '\n';
545 } else if (command == "UP" || command == "DOWN") {
546 Interface& interface = require_interface(w[1]);
547 bool up = command == "UP";
548 if (interface.up == up) throw std::runtime_error("interface " + w[1] + " already " + lowercase(command));
549 interface.up = up;
550 std::cout << "ok " << lowercase(command) << ' ' << w[1] << '\n';
551 } else if (command == "ROUTE") {
552 Prefix prefix = require_prefix(w[1]);
553 std::optional<Address> hop;
554 std::string output;
555 if (w[2] == "VIA") hop = require_address(w[3]);
556 else output = require_interface(w[3]).name;
557 int metric = require_number(w[4], 1, 255, "bad metric");
558 if (hop) {
559 if (local(*hop)) throw std::runtime_error("nexthop " + w[3] + " is local");
560 for (const auto& entry : interfaces)
561 if (contains({entry.second.address, entry.second.length}, *hop)) output = entry.first;
562 if (output.empty()) throw std::runtime_error("nexthop " + w[3] + " unreachable");
563 }
564 for (const Route& route : routes)
565 if (route.type == 'S' && route.prefix.address == prefix.address && route.prefix.length == prefix.length &&
566 route.interface == output && route.hop == hop) throw std::runtime_error("duplicate route " + w[1]);
567 routes.push_back({'S', prefix, output, hop, metric});
568 std::cout << "ok route " << routes.back().text() << '\n';
569 } else if (command == "NOROUTE") {
570 Prefix prefix = require_prefix(w[1]);
571 auto matches = [&](const Route& route) {
572 return route.type == 'S' && route.prefix.address == prefix.address && route.prefix.length == prefix.length;
573 };
574 std::size_t removed = static_cast<std::size_t>(std::count_if(routes.begin(), routes.end(), matches));
575 if (!removed) throw std::runtime_error("no static route " + w[1]);
576 routes.erase(std::remove_if(routes.begin(), routes.end(), matches), routes.end());
577 std::cout << "ok noroute " << w[1] << " removed=" << removed << '\n';
578 } else if (command == "LOOKUP") {
579 Address address = require_address(w[1]);
580 Route* route = select(address);
581 std::cout << "lookup " << w[1] << ' ' << (route ? route->text() : "none") << '\n';
582 } else if (command == "ACL") {
583 Interface& interface = require_interface(w[1]);
584 int sequence = require_number(w[3], 1, 9999, "bad sequence");
585 Prefix source = require_prefix(w[6], true);
586 Ports sports = require_ports(w[7]);
587 Prefix destination = require_prefix(w[8], true);
588 Ports dports = require_ports(w[9]);
589 if ((w[5] == "ICMP" || w[5] == "ANY") && (w[7] != "ANY" || w[9] != "ANY"))
590 throw std::runtime_error("ports not allowed for " + w[5]);
591 AccessList& acl = interface.acls[w[2] == "IN" ? 0 : 1];
592 if (acl.rules.count(sequence)) throw std::runtime_error("duplicate sequence " + w[3]);
593 acl.rules.emplace(sequence, Rule{sequence, w[4], w[5], source, sports, destination, dports, w.size() == 11});
594 std::cout << "ok acl " << w[1] << ' ' << lowercase(w[2]) << ' ' << w[3] << '\n';
595 } else if (command == "NOACL") {
596 Interface& interface = require_interface(w[1]);
597 AccessList& acl = interface.acls[w[2] == "IN" ? 0 : 1];
598 auto rule = acl.rules.find(std::stoi(w[3]));
599 if (rule == acl.rules.end()) throw std::runtime_error("no rule " + w[3] + " on " + w[1] + " " + lowercase(w[2]));
600 int hits = rule->second.hits;
601 acl.rules.erase(rule);
602 std::cout << "ok noacl " << w[1] << ' ' << lowercase(w[2]) << ' ' << w[3] << " hits=" << hits << '\n';
603 } else if (command == "NAT") {
604 if (w[1] == "STATIC") {
605 Address inside = require_address(w[2]), outside = require_address(w[3]);
606 if (local(inside)) throw std::runtime_error("address " + w[2] + " is local");
607 if (local(outside)) throw std::runtime_error("address " + w[3] + " is local");
608 for (StaticNat mapping : statics)
609 if (mapping.inside == inside) throw std::runtime_error("duplicate static " + w[2]);
610 for (StaticNat mapping : statics)
611 if (mapping.outside == outside) throw std::runtime_error("duplicate static " + w[3]);
612 statics.push_back({inside, outside});
613 std::cout << "ok nat static " << w[2] << ' ' << w[3] << '\n';
614 } else {
615 Interface& interface = require_interface(w[2]);
616 if (w[1] == "OUTSIDE") {
617 if (interface.outside) throw std::runtime_error("interface " + w[2] + " already outside");
618 interface.outside = true;
619 std::cout << "ok nat outside " << w[2] << '\n';
620 } else if (w[1] == "PAT") {
621 int count = require_number(w[3], 1, 1000, "bad pool size");
622 if (interface.pool_size) throw std::runtime_error("pat already on " + w[2]);
623 interface.pool_size = count;
624 interface.outside = true;
625 std::cout << "ok nat pat " << w[2] << ' ' << address_text(interface.address) << " 20000-" << 20000 + count - 1 << '\n';
626 } else {
627 int port = require_number(w[4], 1, 65535, "bad port");
628 Address inside = require_address(w[5]);
629 int inport = require_number(w[6], 1, 65535, "bad port");
630 if (!interface.outside) throw std::runtime_error("interface " + w[2] + " not outside");
631 if (local(inside)) throw std::runtime_error("address " + w[5] + " is local");
632 for (const Forward& forward : forwards)
633 if (forward.interface == w[2] && forward.protocol == w[3] && forward.outside_port == port)
634 throw std::runtime_error("duplicate forward " + w[2] + " " + w[3] + " " + w[4]);
635 forwards.push_back({w[2], w[3], port, inside, inport});
636 std::cout << "ok nat forward " << w[2] << ' ' << w[3] << ' ' << w[4] << ' ' << w[5] << ':' << w[6] << '\n';
637 }
638 }
639 } else if (command == "RPF") {
640 Interface& interface = require_interface(w[1]);
641 interface.rpf = lowercase(w[2]);
642 std::cout << "ok rpf " << w[1] << ' ' << interface.rpf << '\n';
643 } else if (command == "TIMEOUT") {
644 int seconds = require_number(w[2], 1, 86400, "bad timeout");
645 timeouts[w[1]] = seconds;
646 std::cout << "ok timeout " << w[1] << ' ' << w[2] << '\n';
647 } else if (command == "TICK") {
648 int seconds = require_number(w[1], 1, 86400, "bad tick");
649 clock += seconds;
650 std::size_t before = connections.size();
651 connections.erase(std::remove_if(connections.begin(), connections.end(), [&](const Connection& connection) {
652 int timeout = timeouts.at(connection.orig.protocol);
653 if (connection.orig.protocol == "TCP") {
654 if (connection.state == "SYN_SENT" || connection.state == "FIN_WAIT") timeout = 60;
655 if (connection.state == "CLOSED") timeout = 10;
656 }
657 return clock - connection.last_used >= timeout;
658 }), connections.end());
659 std::size_t removed = 0;
660 for (auto& entry : interfaces) {
661 std::vector<Pat>& mappings = entry.second.mappings;
662 std::size_t old_size = mappings.size();
663 mappings.erase(std::remove_if(mappings.begin(), mappings.end(), [&](const Pat& mapping) {
664 return clock - mapping.last_used >= timeouts.at(mapping.protocol);
665 }), mappings.end());
666 removed += old_size - mappings.size();
667 }
668 std::cout << "ok tick time=" << clock << " expired conns=" << before - connections.size() << " nat=" << removed << '\n';
669 } else if (command == "PACKET") {
670 Interface& interface = require_interface(w[1]);
671 if (w[2] != "TCP" && w[3] != "-") throw std::runtime_error("flags not allowed for " + w[2]);
672 if (w[2] == "TCP" && w[3] != "-") {
673 std::set<char> seen;
674 for (char c : w[3])
675 if (std::string("SAFR").find(c) == std::string::npos || !seen.insert(c).second)
676 throw std::runtime_error("bad flags " + w[3]);
677 }
678 Address source = require_address(w[4]);
679 int sport = require_number(w[5], 0, 65535, "bad port");
680 Address destination = require_address(w[6]);
681 int dport = require_number(w[7], 0, 65535, "bad port");
682 int ttl = require_number(w[8], 1, 255, "bad ttl");
683 ++packet_number;
684 ++interface.rx;
685 std::pair<std::string, std::string> result = process(interface, {w[2], source, sport, destination, dport},
686 w[3] == "-" ? "" : w[3], ttl);
687 ++counts[result.first];
688 if (result.first != "forwarded" && result.first != "local") ++interface.drop;
689 std::cout << '#' << packet_number << ' ' << w[2] << ' ' << w[3] << ' ' << w[4] << ':' << w[5]
690 << " > " << w[6] << ':' << w[7] << ' ' << result.second << '\n';
691 } else if (command == "SHOW") {
692 show(w[1]);
693 } else if (command == "CLEAR" && w[1] == "COUNTERS") {
694 for (auto& count : counts) count.second = 0;
695 errors = 0;
696 for (auto& entry : interfaces) {
697 Interface& interface = entry.second;
698 interface.rx = interface.tx = interface.drop = 0;
699 for (AccessList& acl : interface.acls) {
700 acl.default_hits = 0;
701 for (auto& rule : acl.rules) rule.second.hits = 0;
702 }
703 }
704 for (Route& route : routes) route.hits = 0;
705 std::cout << "ok clear counters\n";
706 } else if (command == "CLEAR") {
707 std::size_t removed = 0;
708 for (auto& entry : interfaces) {
709 removed += entry.second.mappings.size();
710 entry.second.mappings.clear();
711 }
712 std::cout << "ok clear conns=" << connections.size() << " nat=" << removed << '\n';
713 connections.clear();
714 }
715 }

717public:
718 void execute(const std::string& line) {
719 try {
720 configure(words(line));
721 } catch (const std::runtime_error& error) {
722 ++errors;
723 std::cout << "error: " << error.what() << '\n';
724 }
725 }
