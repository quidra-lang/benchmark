3private fun ip(word: String): Long {
4 val fields = word.split('.')
5 if (fields.size != 4) throw Problem("bad address $word")
6 var address = 0L
7 for (field in fields) {
8 if (
9 field.length !in 1..3 ||
10 (field.length > 1 && field[0] == '0') ||
11 field.any { it !in '0'..'9' } ||
12 field.toInt() > 255
13 )
14 throw Problem("bad address $word")
15 address = address * 256 + field.toInt()
16 }
17 return address
18}

20private fun address(a: Long) = "${a shr 24}.${(a shr 16) and 255}.${(a shr 8) and 255}.${a and 255}"

22private fun mask(length: Int) =
23 if (length == 0) 0L else (0xffffffffL shl (32 - length)) and 0xffffffffL

25private data class Prefix(val network: Long, val length: Int) {
26 fun contains(a: Long) = (a and mask(length)) == (network and mask(length))
27
28 fun notation() = "${address(network)}/$length"
29
30 fun aclNotation() = if (length == 0) "any" else notation()
31}
32
33private fun prefix(word: String, hostBits: Boolean = true): Prefix {
34 val fields = word.split('/')
35 if (fields.size != 2) throw Problem("bad prefix $word")
36 val a =
37 try {
38 ip(fields[0])
39 } catch (_: Problem) {
40 throw Problem("bad prefix $word")
41 }
42 val s = fields[1]
43 if (
44 s.length !in 1..2 ||
45 (s.length == 2 && s[0] == '0') ||
46 s.any { it !in '0'..'9' } ||
47 s.toInt() > 32
48 )
49 throw Problem("bad prefix $word")
50 val length = s.toInt()
51 if (hostBits && (a and (0xffffffffL xor mask(length))) != 0L)
52 throw Problem("host bits set in $word")
53 return Prefix(a, length)
54}

56private data class Ports(val low: Int, val high: Int, val any: Boolean) {
57 fun contains(n: Int) = n in low..high
58
59 fun notation() = if (any) "any" else if (low == high) "$low" else "$low-$high"
60}
61
62private fun ports(word: String): Ports {
63 if (word == "ANY") return Ports(0, 65535, true)
64 val values = word.split('-').map(String::toInt)
65 val a = values.first()
66 val b = values.last()
67 if (a > 65535 || b > 65535 || a > b) throw Problem("bad port range $word")
68 return Ports(a, b, false)
69}

174 fun iface(name: String): Int =

178 fun localAddress(a: Long): Int? = ifaces.indexOfFirst { it.address == a }.takeIf { it >= 0 }

180 fun select(a: Long): Int? {
181 var selected: Int? = null
182 for ((i, r) in routes.withIndex()) {
183 if (!ifaces[r.iface].up || !r.dest.contains(a)) continue
184 val old = selected?.let { routes[it] }
185 if (
186 old == null ||
187 r.dest.length > old.dest.length ||
188 (r.dest.length == old.dest.length && r.metric < old.metric)
189 )
190 selected = i
191 }
192 return selected
193 }

198 fun acl(i: Int, d: Int, tuple: PacketTuple, existing: Boolean): String? {
199 val list = rules.filter { it.iface == i && it.dir == d }.sortedBy { it.seq }
200 if (list.isEmpty()) return null
201 val name = ifaces[i].name
202 val dir = if (d == 0) "in" else "out"
203 for (r in list) {
204 if (
205 (r.proto != "ANY" && r.proto != tuple.proto) ||
206 !r.src.contains(tuple.src) ||
207 !r.dst.contains(tuple.dst) ||
208 (r.established && !existing)
209 )
210 continue
211 if (
212 r.proto in listOf("TCP", "UDP") &&
213 (!r.sports.contains(tuple.sport) || !r.dports.contains(tuple.dport))
214 )
215 continue
216 r.hits++
217 return when (r.action) {
218 "PERMIT" -> null
219 "DENY" -> "DROP acl $name $dir ${r.seq}"
220 else ->
221 "REJECT acl $name $dir ${r.seq} ${if (tuple.proto == "TCP") "tcp-reset" else "icmp-prohibited"}"
222 }
223 }
224 ifaces[i].defaults[d]++
225 return "DROP acl $name $dir default"
226 }

228 fun packet(input: Int, flags: String, orig: PacketTuple, ttl: Int): Pair<String, Int> {
229 if (!ifaces[input].up) return "DROP down" to 0
230 fun martian(a: Long) = a shr 24 == 0L || a shr 24 == 127L || a >= 0xe0000000L
231 if (martian(orig.src) || martian(orig.dst) || localAddress(orig.src) != null)
232 return "DROP martian" to 1
233 if (ifaces[input].rpf != "off") {
234 val r = select(orig.src) ?: return "DROP rpf" to 2
235 if (ifaces[input].rpf == "strict" && routes[r].iface != input) return "DROP rpf" to 2
236 }
237 val pureSyn = orig.proto == "TCP" && flags == "S"
238 var found: Connection? = null
239 var side = 0
240 for (c in conns) {
241 if (c.orig == orig) {
242 found = c
243 side = 0
244 break
245 }
246 if (c.reply == orig) {
247 found = c
248 side = 1
249 break
250 }
251 }
252 if (found?.state == "CLOSED" && pureSyn) {
253 conns.remove(found)
254 found = null
255 }
256 if (orig.proto == "TCP" && found == null && !pureSyn) return "DROP invalid" to 3
257 acl(input, 0, orig, found != null)?.let {
258 return it to 4
259 }
260 var dst = orig.dst
261 var dport = orig.dport
262 var used: Pat? = null
263 if (ifaces[input].outside) {
264 val static = statics.firstOrNull { it.second == dst }
265 if (static != null) dst = static.first
266 else if (dst == ifaces[input].address) {
267 val f = forwards.firstOrNull {
268 it.iface == input && it.proto == orig.proto && it.port == dport
269 }
270 if (f != null) {
271 dst = f.inside
272 dport = f.inport
273 } else {
274 val p = pats.firstOrNull {
275 it.iface == input && it.proto == orig.proto && it.port == dport
276 }
277 if (p != null) {
278 dst = p.inside
279 dport = p.inport
280 used = p
281 }
282 }
283 }
284 }
285 localAddress(dst)?.let {
286 return "LOCAL ${ifaces[it].name}" to -2
287 }
288 if (ttl == 1) return "DROP ttl" to 5
289 val r = select(dst) ?: return "DROP noroute" to 6
290 val output = routes[r].iface
291 acl(output, 1, orig.copy(dst = dst, dport = dport), found != null)?.let {
292 return it to 7
293 }
294 var src = orig.src
295 var sport = orig.sport
296 if (!ifaces[input].outside && ifaces[output].outside) {
297 val static = statics.firstOrNull { it.first == src }
298 val forward = forwards.firstOrNull {
299 it.iface == output &&
300 it.proto == orig.proto &&
301 it.inside == src &&
302 it.inport == sport
303 }
304 if (static != null) src = static.second
305 else if (forward != null) {
306 src = ifaces[output].address
307 sport = forward.port
308 } else {
309 val count = ifaces[output].pool
310 if (count != null) {
311 var mapping = pats.firstOrNull {
312 it.iface == output &&
313 it.proto == orig.proto &&
314 it.inside == src &&
315 it.inport == sport
316 }
317 if (mapping == null) {
318 val port =
319 (20000 until 20000 + count).firstOrNull { p ->
320 pats.none { it.iface == output && it.port == p } &&
321 forwards.none { it.iface == output && it.port == p }
322 } ?: return "DROP exhausted" to 8
323 mapping = Pat(output, orig.proto, src, sport, port, clock)
324 pats.add(mapping)
325 }
326 used = mapping
327 src = ifaces[output].address
328 sport = mapping.port
329 }
330 }
331 }
332 used?.used = clock
333 val direction: String
334 val c: Connection
335 if (found != null) {
336 c = found
337 direction = if (side == 0) "orig" else "reply"
338 } else {
339 connectionNumber++
340 c =
341 Connection(
342 connectionNumber,
343 orig,
344 PacketTuple(orig.proto, dst, dport, src, sport),
345 input,
346 output,
347 clock,
348 )
349 conns.add(c)
350 side = 0
351 direction = "new"
352 }
353 c.packets[side]++
354 c.used = clock
355 if (orig.proto != "TCP") {
356 if (side == 1) c.state = "REPLIED"
357 } else if ('R' in flags) c.state = "CLOSED"
358 else {
359 if (c.state == "SYN_SENT" && side == 1 && 'S' in flags && 'A' in flags)
360 c.state = "ESTABLISHED"
361 if ('F' in flags && c.state in listOf("ESTABLISHED", "FIN_WAIT")) {
362 c.fin[side] = true
363 c.state = if (c.fin.all { it }) "CLOSED" else "FIN_WAIT"
364 }
365 }
366 ifaces[output].tx++
367 routes[r].hits++
368 val hop = routes[r].hop?.let { "via ${address(it)}" } ?: "direct"
369 var verdict = "FORWARD ${ifaces[output].name} $hop ttl=${ttl-1}"
370 if (dst != orig.dst || dport != orig.dport) verdict += " dnat=${address(dst)}:$dport"
371 if (src != orig.src || sport != orig.sport) verdict += " snat=${address(src)}:$sport"
372 return "$verdict conn=${c.number} $direction" to -1
373 }

375 fun command(w: List<String>): List<String> {
376 fun n(i: Int) = w[i].toInt()
377 fun d(s: String) = if (s == "IN") 0 else 1
378 fun require(condition: Boolean, reason: String) {
379 if (!condition) throw Problem(reason)
380 }
381 val text =
382 when (w[0]) {
383 "IFACE" -> {
384 require(ifaces.none { it.name == w[1] }, "duplicate interface ${w[1]}")
385 val p = prefix(w[2], false)
386 require(p.length in 8..30, "bad prefix ${w[2]}")
387 val host = p.network and (0xffffffffL xor mask(p.length))
388 require(
389 host != 0L && host != (0xffffffffL xor mask(p.length)),
390 "bad host address ${w[2]}",
391 )
392 ifaces
393 .firstOrNull {
394 (it.address and mask(minOf(it.length, p.length))) ==
395 (p.network and mask(minOf(it.length, p.length)))
396 }
397 ?.let { throw Problem("subnet overlaps ${it.name}") }
398 val i = ifaces.size
399 val dest = Prefix(p.network and mask(p.length), p.length)
400 ifaces.add(Interface(w[1], p.network, p.length))
401 routes.add(Route("C", dest, i, null, 0))
402 "ok iface ${w[1]} ${w[2]} network ${dest.notation()}"
403 }
404 "UP",
405 "DOWN" -> {
406 val i = iface(w[1])
407 val up = w[0] == "UP"
408 require(ifaces[i].up != up, "interface ${w[1]} already ${w[0].lowercase()}")
409 ifaces[i].up = up
410 "ok ${w[0].lowercase()} ${w[1]}"
411 }
412 "ROUTE" -> {
413 val dest = prefix(w[1])
414 var hop: Long? = null
415 var output = 0
416 if (w[2] == "VIA") hop = ip(w[3]) else output = iface(w[3])
417 require(n(4) in 1..255, "bad metric ${w[4]}")
418 val nextHop = hop
419 if (nextHop != null) {
420 require(localAddress(nextHop) == null, "nexthop ${w[3]} is local")
421 output = ifaces.indexOfFirst {
422 Prefix(it.address, it.length).contains(nextHop)
423 }
424 require(output >= 0, "nexthop ${w[3]} unreachable")
425 }
426 require(
427 routes.none {
428 it.kind == "S" && it.dest == dest && it.iface == output && it.hop == hop
429 },
430 "duplicate route ${w[1]}",
431 )
432 val r = Route("S", dest, output, hop, n(4))
433 routes.add(r)
434 "ok route ${routeText(r)}"
435 }
436 "NOROUTE" -> {
437 val dest = prefix(w[1])
438 val old = routes.size
439 require(
440 routes.any { it.kind == "S" && it.dest == dest },
441 "no static route ${w[1]}",
442 )
443 routes.removeAll { it.kind == "S" && it.dest == dest }
444 "ok noroute ${w[1]} removed=${old-routes.size}"
445 }
446 "LOOKUP" -> {
447 val a = ip(w[1])
448 "lookup ${w[1]} ${select(a)?.let { routeText(routes[it]) } ?: "none"}"
449 }
450 "ACL" -> {
451 val i = iface(w[1])
452 val dir = d(w[2])
453 require(n(3) in 1..9999, "bad sequence ${w[3]}")
454 val src = if (w[6] == "ANY") Prefix(0, 0) else prefix(w[6])
455 val sports = ports(w[7])
456 val dst = if (w[8] == "ANY") Prefix(0, 0) else prefix(w[8])
457 val dports = ports(w[9])
458 require(
459 w[5] !in listOf("ICMP", "ANY") || (sports.any && dports.any),
460 "ports not allowed for ${w[5]}",
461 )
462 require(
463 rules.none { it.iface == i && it.dir == dir && it.seq == n(3) },
464 "duplicate sequence ${w[3]}",
465 )
466 rules.add(
467 Rule(i, dir, n(3), w[4], w[5], src, sports, dst, dports, w.size == 11),
468 )
469 "ok acl ${w[1]} ${w[2].lowercase()} ${w[3]}"
470 }
471 "NOACL" -> {
472 val i = iface(w[1])
473 val dir = d(w[2])
474 val at = rules.indexOfFirst { it.iface == i && it.dir == dir && it.seq == n(3) }
475 require(at >= 0, "no rule ${w[3]} on ${w[1]} ${w[2].lowercase()}")
476 val rule = rules.removeAt(at)
477 "ok noacl ${w[1]} ${w[2].lowercase()} ${w[3]} hits=${rule.hits}"
478 }
479 "NAT" ->
480 when (w[1]) {
481 "STATIC" -> {
482 val inside = ip(w[2])
483 val outside = ip(w[3])
484 require(localAddress(inside) == null, "address ${w[2]} is local")
485 require(localAddress(outside) == null, "address ${w[3]} is local")
486 require(statics.none { it.first == inside }, "duplicate static ${w[2]}")
487 require(
488 statics.none { it.second == outside },
489 "duplicate static ${w[3]}",
490 )
491 statics.add(inside to outside)
492 "ok nat static ${w[2]} ${w[3]}"
493 }
494 "OUTSIDE" -> {
495 val i = iface(w[2])
496 require(!ifaces[i].outside, "interface ${w[2]} already outside")
497 ifaces[i].outside = true
498 "ok nat outside ${w[2]}"
499 }
500 "PAT" -> {
501 val i = iface(w[2])
502 require(n(3) in 1..1000, "bad pool size ${w[3]}")
503 require(ifaces[i].pool == null, "pat already on ${w[2]}")
504 ifaces[i].pool = n(3)
505 ifaces[i].outside = true
506 "ok nat pat ${w[2]} ${address(ifaces[i].address)} 20000-${20000+n(3)-1}"
507 }
508 else -> {
509 val i = iface(w[2])
510 require(n(4) in 1..65535, "bad port ${w[4]}")
511 val inside = ip(w[5])
512 require(n(6) in 1..65535, "bad port ${w[6]}")
513 require(ifaces[i].outside, "interface ${w[2]} not outside")
514 require(localAddress(inside) == null, "address ${w[5]} is local")
515 require(
516 forwards.none {
517 it.iface == i && it.proto == w[3] && it.port == n(4)
518 },
519 "duplicate forward ${w[2]} ${w[3]} ${w[4]}",
520 )
521 forwards.add(Forward(i, w[3], n(4), inside, n(6)))
522 "ok nat forward ${w[2]} ${w[3]} ${w[4]} ${w[5]}:${w[6]}"
523 }
524 }
525 "RPF" -> {
526 val i = iface(w[1])
527 ifaces[i].rpf = w[2].lowercase()
528 "ok rpf ${w[1]} ${w[2].lowercase()}"
529 }
530 "TIMEOUT" -> {
531 require(n(2) in 1..86400, "bad timeout ${w[2]}")
532 timeouts[protocolIndex(w[1])] = n(2)
533 "ok timeout ${w[1]} ${w[2]}"
534 }
535 "TICK" -> {
536 require(n(1) in 1..86400, "bad tick ${w[1]}")
537 clock += n(1)
538 val old = conns.size
539 val oldNat = pats.size
540 conns.removeAll { c ->
541 val timeout =
542 if (c.orig.proto == "TCP")
543 when (c.state) {
544 "SYN_SENT",
545 "FIN_WAIT" -> 60
546 "CLOSED" -> 10
547 else -> timeouts[0]
548 }
549 else timeouts[protocolIndex(c.orig.proto)]
550 clock - c.used >= timeout
551 }
552 pats.removeAll { clock - it.used >= timeouts[protocolIndex(it.proto)] }
553 "ok tick time=$clock expired conns=${old-conns.size} nat=${oldNat-pats.size}"
554 }
555 "PACKET" -> {
556 val i = iface(w[1])
557 require(w[2] == "TCP" || w[3] == "-", "flags not allowed for ${w[2]}")
558 if (w[2] == "TCP" && w[3] != "-")
559 require(
560 w[3].all { it in "SAFR" } && w[3].toSet().size == w[3].length,
561 "bad flags ${w[3]}",
562 )
563 val src = ip(w[4])
564 require(n(5) in 0..65535, "bad port ${w[5]}")
565 val dst = ip(w[6])
566 require(n(7) in 0..65535, "bad port ${w[7]}")
567 require(n(8) in 1..255, "bad ttl ${w[8]}")
568 packetNumber++
569 ifaces[i].rx++
570 val (verdict, reason) =
571 packet(i, w[3], PacketTuple(w[2], src, n(5), dst, n(7)), n(8))
572 when (reason) {
573 -1 -> forwarded++
574 -2 -> local++
575 else -> {
576 drops[reason]++
577 ifaces[i].drop++
578 }
579 }
580 "#$packetNumber ${w[2]} ${w[3]} ${w[4]}:${w[5]} > ${w[6]}:${w[7]} $verdict"
581 }
582 "SHOW" -> return show(w[1])
583 else ->
584 if (w[1] == "CONNS") {
585 val result = "ok clear conns=${conns.size} nat=${pats.size}"
586 conns.clear()
587 pats.clear()
588 result
589 } else {
590 forwarded = 0
591 local = 0
592 drops.fill(0)
593 errors = 0
594 ifaces.forEach {
595 it.rx = 0
596 it.tx = 0
597 it.drop = 0
598 it.defaults.fill(0)
599 }
600 routes.forEach { it.hits = 0 }
601 rules.forEach { it.hits = 0 }
602 "ok clear counters"
603 }
604 }
605 return listOf(text)
606 }

609 val names = ifaces.indices.sortedBy { ifaces[it].name }
610 val lines = mutableListOf<String>()

618 (f.pool?.let { " pat=${pats.count { p -> p.iface == i }}/$it" } ?: ""),
619 )
620 }
621 }
622 "ROUTES" -> {
623 lines.add("show routes count=${routes.size}")
624 for (i in
625 routes.indices.sortedWith(
626 compareBy(

631 ),
632 )) {

636 )
637 }
638 }

642 val list = rules.filter { it.iface == i && it.dir == d }.sortedBy { it.seq }
643 if (list.isEmpty()) continue
644 val name = ifaces[i].name
645 val dir = if (d == 0) "in" else "out"
646 for (r in list) lines.add(

648 )

650 }
651 }

653 lines.add(
654 "show nat static=${statics.size} forward=${forwards.size} pat=${pats.size}",
655 )
656 for ((a, b) in statics.sortedBy { it.first }) lines.add(
657 "static ${address(a)} ${address(b)}",

660 forwards.sortedWith(
661 compareBy({ ifaces[it.iface].name }, { it.port }, { it.proto }),

665 for (p in pats.sortedWith(compareBy({ ifaces[it.iface].name }, { it.port }))) lines
666 .add(
667 "pat ${ifaces[p.iface].name} ${p.port} ${p.proto} ${address(p.inside)}:${p.inport} idle=${clock-p.used}",
668 )
669 }
670 "CONNS" -> {
671 lines.add("show conns count=${conns.size}")
672 for (c in conns) lines.add(
673 "conn ${c.number} ${c.orig.proto} ${c.orig.notation()} reply ${c.reply.notation()} in=${ifaces[c.input].name} out=${ifaces[c.output].name} state=${c.state} pkts=${c.packets[0]}/${c.packets[1]} idle=${clock-c.used}",
674 )
675 }
676 else -> {
677 val dropped = drops.sum()
678 lines.add(
679 "show counters packets=${forwarded+local+dropped} forwarded=$forwarded local=$local dropped=$dropped errors=$errors",
680 )
681 lines.add(
682 "drops " + reasons.indices.joinToString(" ") { "${reasons[it]}=${drops[it]}" },
683 )
684 for (i in names) {
685 val f = ifaces[i]
686 lines.add("iface ${f.name} rx=${f.rx} tx=${f.tx} drop=${f.drop}")
687 }
688 }
689 }
690 return lines
691 }
692}
