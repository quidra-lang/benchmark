3class Program
4{
5 record Prefix(uint Address, int Length)
6 {
7 public uint Network => Address & Mask(Length);
8 public bool Contains(uint value) => (value & Mask(Length)) == Network;
9 public override string ToString() => $"{Ip(Address)}/{Length}";
10 }
11 record Ports(int First, int Last, bool Any)
12 {
13 public bool Contains(int port) => First <= port && port <= Last;
14 public override string ToString() => Any ? "any" : First == Last ? $"{First}" : $"{First}-{Last}";
15 }
16 class Interface(string name, Prefix prefix)
17 {
18 public string Name = name, Rpf = "off";
19 public Prefix Prefix = prefix;
20 public bool Up = true, Outside;
21 public int Pool, Rx, Tx, Drop;
22 public SortedDictionary<int, Rule>[] Acl = [new(), new()];
23 public int[] Defaults = new int[2];
24 }
25 class Route(char type, Prefix prefix, Interface output, uint? hop, int metric)
26 {
27 public char Type = type;
28 public Prefix Prefix = prefix;
29 public Interface Output = output;
30 public uint? Hop = hop;
31 public int Metric = metric, Hits;
32 public override string ToString() => $"{Type} {Prefix}" + (Hop is null ? "" : $" via {Ip(Hop.Value)}") + $" dev {Output.Name} metric {Metric}";
33 }
34 class Rule
35 {
36 public int Sequence, Hits;
37 public required string Action, Protocol;
38 public required Prefix Source, Destination;
39 public required Ports SourcePorts, DestinationPorts;
40 public bool Established;
41 }
42 record StaticNat(uint Inside, uint Outside);
43 record Forward(Interface Interface, string Protocol, int Port, uint Inside, int InsidePort);
44 class Pat(Interface iface, string protocol, uint inside, int insidePort, int port)
45 {
46 public Interface Interface = iface;
47 public string Protocol = protocol;
48 public uint Inside = inside;
49 public int InsidePort = insidePort, Port = port, Last = clock;
50 }
51 record PacketTuple(string Protocol, uint Source, int SourcePort, uint Destination, int DestinationPort);
52 class Connection
53 {
54 public int Number, OriginalPackets, ReplyPackets, Last;
55 public required PacketTuple Original, Reply;
56 public required Interface Input, Output;
57 public required string State;
58 public bool OriginalFin, ReplyFin;

60 class CommandError(string message) : Exception(message);
61 static readonly List<Interface> Interfaces = [];
62 static readonly List<Route> Routes = [];
63 static readonly List<StaticNat> Statics = [];
64 static readonly List<Forward> Forwards = [];
65 static readonly List<Pat> Pats = [];
66 static readonly List<Connection> Connections = [];
67 static readonly Dictionary<string, int> Timeouts = new() { ["TCP"] = 300, ["UDP"] = 60, ["ICMP"] = 30 };
68 static readonly string[] Reasons = ["down", "martian", "rpf", "invalid", "acl_in", "ttl", "noroute", "acl_out", "exhausted"];
69 static readonly int[] Drops = new int[9];
70 static int clock, packetNumber, connectionNumber, forwarded, local, errors;
71 static uint Mask(int length) => length == 0 ? 0 : uint.MaxValue << (32 - length);
72 static string Ip(uint value) => $"{value >> 24}.{(value >> 16) & 255}.{(value >> 8) & 255}.{value & 255}";
73 static uint? ParseAddress(string text)
74 {
75 string[] fields = text.Split('.');
76 if (fields.Length != 4) return null;
77 uint address = 0;
78 foreach (string field in fields)
79 {
80 if (field.Length is < 1 or > 3 || field.Length > 1 && field[0] == '0' || field.Any(c => c is < '0' or > '9')) return null;
81 uint part = uint.Parse(field, CultureInfo.InvariantCulture);
82 if (part > 255) return null;
83 address = (address << 8) | part;
84 }
85 return address;
86 }
87 static uint Address(string text) => ParseAddress(text) ?? throw new CommandError("bad address " + text);
88 static Prefix? ParsePrefix(string text)
89 {
90 string[] parts = text.Split('/');
91 if (parts.Length != 2 || parts[1].Length is < 1 or > 2 || parts[1].Length > 1 && parts[1][0] == '0' || parts[1].Any(c => c is < '0' or > '9')) return null;
92 uint? address = ParseAddress(parts[0]);
93 int length = int.Parse(parts[1]);
94 return address is null || length > 32 ? null : new(address.Value, length);
95 }
96 static Prefix CheckedPrefix(string text, bool any = false)
97 {
98 if (any && text == "ANY") return new(0, 0);
99 Prefix prefix = ParsePrefix(text) ?? throw new CommandError("bad prefix " + text);
100 if (prefix.Address != prefix.Network) throw new CommandError("host bits set in " + text);
101 return prefix;
102 }
103 static Interface FindInterface(string name) => Interfaces.Find(iface => iface.Name == name) ?? throw new CommandError("no interface " + name);
104 static Interface? LocalAddress(uint address) => Interfaces.Find(iface => iface.Prefix.Address == address);
105 static int Range(string text, int minimum, int maximum, string message)
106 {
107 int value = int.Parse(text);
108 if (value < minimum || value > maximum) throw new CommandError(message + " " + text);
109 return value;
110 }
111 static Ports ParsePorts(string text)
112 {
113 if (text == "ANY") return new(0, 65535, true);
114 string[] parts = text.Split('-');
115 int first = int.Parse(parts[0]), last = int.Parse(parts[^1]);
116 if (first > 65535 || last > 65535 || first > last) throw new CommandError("bad port range " + text);
117 return new(first, last, false);
118 }
119 static Route? SelectRoute(uint address)
120 {
121 Route? best = null;
122 foreach (Route route in Routes)
123 if (route.Output.Up && route.Prefix.Contains(address) && (best is null || route.Prefix.Length > best.Prefix.Length || route.Prefix.Length == best.Prefix.Length && route.Metric < best.Metric)) best = route;
124 return best;
125 }
126 static string? EvaluateAcl(Interface iface, int direction, PacketTuple packet, uint source, int sourcePort, uint destination, int destinationPort, bool established)
127 {
128 var rules = iface.Acl[direction];
129 if (rules.Count == 0) return null;
130 string dir = direction == 0 ? "in" : "out";
131 foreach (Rule rule in rules.Values)
132 {
133 if (rule.Protocol != "ANY" && rule.Protocol != packet.Protocol || !rule.Source.Contains(source) || !rule.Destination.Contains(destination) || rule.Established && !established) continue;
134 if (rule.Protocol is "TCP" or "UDP" && (!rule.SourcePorts.Contains(sourcePort) || !rule.DestinationPorts.Contains(destinationPort))) continue;
135 rule.Hits++;
136 if (rule.Action == "PERMIT") return null;
137 return (rule.Action == "DENY" ? "DROP" : "REJECT") + $" acl {iface.Name} {dir} {rule.Sequence}" + (rule.Action == "REJECT" ? packet.Protocol == "TCP" ? " tcp-reset" : " icmp-prohibited" : "");
138 }
139 iface.Defaults[direction]++;
140 return $"DROP acl {iface.Name} {dir} default";
141 }
142 static bool Martian(uint value) => value >> 24 is 0 or 127 || value >> 29 == 7;
143 static string Drop(Interface input, int reason, string verdict) { Drops[reason]++; input.Drop++; return verdict; }
144 static string ProcessPacket(Interface input, PacketTuple packet, string flags, int ttl)
145 {
146 input.Rx++;
147 if (!input.Up) return Drop(input, 0, "DROP down");
148 if (Martian(packet.Source) || Martian(packet.Destination) || LocalAddress(packet.Source) is not null) return Drop(input, 1, "DROP martian");
149 if (input.Rpf != "off")
150 {
151 Route? reverse = SelectRoute(packet.Source);
152 if (reverse is null || input.Rpf == "strict" && reverse.Output != input) return Drop(input, 2, "DROP rpf");
153 }
154 bool pureSyn = packet.Protocol == "TCP" && flags == "S";
155 Connection? connection = null;
156 string direction = "new";
157 foreach (Connection candidate in Connections)
158 {
159 if (candidate.Original == packet) { connection = candidate; direction = "orig"; break; }
160 if (candidate.Reply == packet) { connection = candidate; direction = "reply"; break; }
161 }
162 if (connection is not null && connection.State == "CLOSED" && pureSyn) { Connections.Remove(connection); connection = null; direction = "new"; }
163 if (packet.Protocol == "TCP" && connection is null && !pureSyn) return Drop(input, 3, "DROP invalid");
164 string? verdict = EvaluateAcl(input, 0, packet, packet.Source, packet.SourcePort, packet.Destination, packet.DestinationPort, connection is not null);
165 if (verdict is not null) return Drop(input, 4, verdict);
166 uint source = packet.Source, destination = packet.Destination;
167 int sourcePort = packet.SourcePort, destinationPort = packet.DestinationPort;
168 Pat? used = null;
169 if (input.Outside)
170 {
171 StaticNat? mapping = Statics.Find(entry => entry.Outside == destination);
172 if (mapping is not null) destination = mapping.Inside;
173 else if (destination == input.Prefix.Address)
174 {
175 Forward? forward = Forwards.Find(rule => rule.Interface == input && rule.Protocol == packet.Protocol && rule.Port == destinationPort);
176 if (forward is not null) { destination = forward.Inside; destinationPort = forward.InsidePort; }
177 else
178 {
179 used = Pats.Find(entry => entry.Interface == input && entry.Protocol == packet.Protocol && entry.Port == destinationPort);
180 if (used is not null) { destination = used.Inside; destinationPort = used.InsidePort; }
181 }
182 }
183 }
184 Interface? target = LocalAddress(destination);
185 if (target is not null) { local++; return "LOCAL " + target.Name; }
186 if (ttl == 1) return Drop(input, 5, "DROP ttl");
187 Route? route = SelectRoute(destination);
188 if (route is null) return Drop(input, 6, "DROP noroute");
189 Interface output = route.Output;
190 verdict = EvaluateAcl(output, 1, packet, source, sourcePort, destination, destinationPort, connection is not null);
191 if (verdict is not null) return Drop(input, 7, verdict);
192 if (!input.Outside && output.Outside)
193 {
194 StaticNat? mapping = Statics.Find(entry => entry.Inside == source);
195 if (mapping is not null) source = mapping.Outside;
196 else
197 {
198 Forward? forward = Forwards.Find(rule => rule.Interface == output && rule.Protocol == packet.Protocol && rule.Inside == source && rule.InsidePort == sourcePort);
199 if (forward is not null) { source = output.Prefix.Address; sourcePort = forward.Port; }
200 else if (output.Pool > 0)
201 {
202 used = Pats.Find(entry => entry.Interface == output && entry.Protocol == packet.Protocol && entry.Inside == source && entry.InsidePort == sourcePort);
203 if (used is null)
204 {
205 int port;
206 for (port = 20000; port < 20000 + output.Pool; port++)
207 if (!Pats.Any(entry => entry.Interface == output && entry.Port == port) && !Forwards.Any(rule => rule.Interface == output && rule.Port == port)) break;
208 if (port == 20000 + output.Pool) return Drop(input, 8, "DROP exhausted");
209 used = new(output, packet.Protocol, source, sourcePort, port); Pats.Add(used);
210 }
211 source = output.Prefix.Address; sourcePort = used.Port;
212 }
213 }
214 }
215 if (used is not null) used.Last = clock;
216 if (connection is null)
217 {
218 connection = new() { Number = ++connectionNumber, Original = packet, Reply = new(packet.Protocol, destination, destinationPort, source, sourcePort), Input = input, Output = output, State = packet.Protocol == "TCP" ? "SYN_SENT" : "UNREPLIED" };
219 Connections.Add(connection);
220 }
221 if (direction == "reply") connection.ReplyPackets++; else connection.OriginalPackets++;
222 connection.Last = clock;
223 if (packet.Protocol != "TCP") { if (direction == "reply") connection.State = "REPLIED"; }
224 else if (flags.Contains('R')) connection.State = "CLOSED";
225 else
226 {
227 if (connection.State == "SYN_SENT" && direction == "reply" && flags.Contains('S') && flags.Contains('A')) connection.State = "ESTABLISHED";
228 if (flags.Contains('F') && connection.State is "ESTABLISHED" or "FIN_WAIT")
229 {
230 if (direction == "reply") connection.ReplyFin = true; else connection.OriginalFin = true;
231 connection.State = connection.OriginalFin && connection.ReplyFin ? "CLOSED" : "FIN_WAIT";
232 }
233 }
234 output.Tx++; route.Hits++; forwarded++;
235 return $"FORWARD {output.Name}" + (route.Hop is null ? " direct" : " via " + Ip(route.Hop.Value)) + $" ttl={ttl - 1}"
236 + (destination != packet.Destination || destinationPort != packet.DestinationPort ? $" dnat={Ip(destination)}:{destinationPort}" : "")
237 + (source != packet.Source || sourcePort != packet.SourcePort ? $" snat={Ip(source)}:{sourcePort}" : "")
238 + $" conn={connection.Number} {direction}";
239 }
240 static IEnumerable<Interface> SortedInterfaces() => Interfaces.OrderBy(iface => iface.Name, StringComparer.Ordinal);
241 static string AclPrefix(Prefix prefix) => prefix.Length == 0 ? "any" : prefix.ToString();
242 static void Show(string what)
243 {
244 switch (what)
245 {
246 case "IFACES":
247 Console.WriteLine($"show ifaces count={Interfaces.Count}");
248 foreach (Interface iface in SortedInterfaces()) Console.WriteLine($"{iface.Name} {iface.Prefix} {(iface.Up ? "up" : "down")} {(iface.Outside ? "outside" : "inside")} rpf={iface.Rpf}" + (iface.Pool == 0 ? "" : $" pat={Pats.Count(mapping => mapping.Interface == iface)}/{iface.Pool}"));
249 break;
250 case "ROUTES":
251 Console.WriteLine($"show routes count={Routes.Count}");
252 foreach (Route route in Routes.OrderBy(entry => entry.Prefix.Address).ThenBy(entry => entry.Prefix.Length).ThenBy(entry => entry.Metric)) Console.WriteLine($"{route} hits={route.Hits}" + (route.Output.Up ? "" : " inactive"));
253 break;
254 case "ACL":
255 Console.WriteLine($"show acl count={Interfaces.Sum(iface => iface.Acl.Sum(rules => rules.Count))}");
256 foreach (Interface iface in SortedInterfaces()) for (int d = 0; d < 2; d++)
257 {
258 if (iface.Acl[d].Count == 0) continue;
259 string dir = d == 0 ? "in" : "out";
260 foreach (Rule rule in iface.Acl[d].Values) Console.WriteLine($"acl {iface.Name} {dir} {rule.Sequence} {rule.Action} {rule.Protocol} {AclPrefix(rule.Source)} {rule.SourcePorts} {AclPrefix(rule.Destination)} {rule.DestinationPorts}" + (rule.Established ? " established" : "") + $" hits={rule.Hits}");
261 Console.WriteLine($"acl {iface.Name} {dir} default DENY hits={iface.Defaults[d]}");
262 }
263 break;
264 case "NAT":
265 Console.WriteLine($"show nat static={Statics.Count} forward={Forwards.Count} pat={Pats.Count}");
266 foreach (StaticNat mapping in Statics.OrderBy(entry => entry.Inside)) Console.WriteLine($"static {Ip(mapping.Inside)} {Ip(mapping.Outside)}");
267 foreach (Forward rule in Forwards.OrderBy(entry => entry.Interface.Name, StringComparer.Ordinal).ThenBy(entry => entry.Port).ThenBy(entry => entry.Protocol, StringComparer.Ordinal)) Console.WriteLine($"forward {rule.Interface.Name} {rule.Protocol} {rule.Port} {Ip(rule.Inside)}:{rule.InsidePort}");
268 foreach (Pat mapping in Pats.OrderBy(entry => entry.Interface.Name, StringComparer.Ordinal).ThenBy(entry => entry.Port)) Console.WriteLine($"pat {mapping.Interface.Name} {mapping.Port} {mapping.Protocol} {Ip(mapping.Inside)}:{mapping.InsidePort} idle={clock - mapping.Last}");
269 break;
270 case "CONNS":
271 Console.WriteLine($"show conns count={Connections.Count}");
272 foreach (Connection c in Connections) Console.WriteLine($"conn {c.Number} {c.Original.Protocol} {Ip(c.Original.Source)}:{c.Original.SourcePort} > {Ip(c.Original.Destination)}:{c.Original.DestinationPort} reply {Ip(c.Reply.Source)}:{c.Reply.SourcePort} > {Ip(c.Reply.Destination)}:{c.Reply.DestinationPort} in={c.Input.Name} out={c.Output.Name} state={c.State} pkts={c.OriginalPackets}/{c.ReplyPackets} idle={clock - c.Last}");
273 break;
274 case "COUNTERS":
275 int dropped = Drops.Sum();
276 Console.WriteLine($"show counters packets={forwarded + local + dropped} forwarded={forwarded} local={local} dropped={dropped} errors={errors}");
277 Console.WriteLine("drops " + string.Join(" ", Reasons.Select((reason, index) => $"{reason}={Drops[index]}")));
278 foreach (Interface iface in SortedInterfaces()) Console.WriteLine($"iface {iface.Name} rx={iface.Rx} tx={iface.Tx} drop={iface.Drop}");
279 break;
280 }
281 }
282 static void Execute(string[] args)
283 {
284 switch (args[0])
285 {
286 case "IFACE":
287 {
288 if (Interfaces.Any(existing => existing.Name == args[1])) throw new CommandError("duplicate interface " + args[1]);
289 Prefix? prefix = ParsePrefix(args[2]);
290 if (prefix is null || prefix.Length is < 8 or > 30) throw new CommandError("bad prefix " + args[2]);
291 uint host = prefix.Address & ~Mask(prefix.Length);
292 if (host == 0 || host == ~Mask(prefix.Length)) throw new CommandError("bad host address " + args[2]);
293 foreach (Interface existing in Interfaces)
294 if ((prefix.Address & Mask(Math.Min(prefix.Length, existing.Prefix.Length))) == (existing.Prefix.Address & Mask(Math.Min(prefix.Length, existing.Prefix.Length)))) throw new CommandError("subnet overlaps " + existing.Name);
295 Interface iface = new(args[1], prefix); Interfaces.Add(iface); Routes.Add(new('C', new(prefix.Network, prefix.Length), iface, null, 0));
296 Console.WriteLine($"ok iface {args[1]} {args[2]} network {Ip(prefix.Network)}/{prefix.Length}"); break;
297 }
298 case "UP": case "DOWN":
299 {
300 Interface iface = FindInterface(args[1]); bool up = args[0] == "UP";
301 if (iface.Up == up) throw new CommandError($"interface {args[1]} already {(up ? "up" : "down")}");
302 iface.Up = up; Console.WriteLine($"ok {args[0].ToLowerInvariant()} {args[1]}"); break;
303 }
304 case "ROUTE":
305 {
306 Prefix prefix = CheckedPrefix(args[1]); bool via = args[2] == "VIA";
307 uint? hop = via ? Address(args[3]) : null;
308 Interface? output = via ? null : FindInterface(args[3]);
309 int metric = Range(args[4], 1, 255, "bad metric");
310 if (via)
311 {
312 if (LocalAddress(hop!.Value) is not null) throw new CommandError($"nexthop {args[3]} is local");
313 output = Interfaces.Find(iface => iface.Prefix.Contains(hop.Value));
314 if (output is null) throw new CommandError($"nexthop {args[3]} unreachable");
315 }
316 if (Routes.Any(route => route.Type == 'S' && route.Prefix == prefix && route.Output == output && route.Hop == hop)) throw new CommandError("duplicate route " + args[1]);
317 Route added = new('S', prefix, output!, hop, metric); Routes.Add(added); Console.WriteLine("ok route " + added); break;
318 }
319 case "NOROUTE":
320 {
321 Prefix prefix = CheckedPrefix(args[1]);
322 if (!Routes.Any(route => route.Type == 'S' && route.Prefix == prefix)) throw new CommandError("no static route " + args[1]);
323 int removed = Routes.RemoveAll(route => route.Type == 'S' && route.Prefix == prefix);
324 Console.WriteLine($"ok noroute {args[1]} removed={removed}"); break;
325 }
326 case "LOOKUP": Console.WriteLine($"lookup {args[1]} {SelectRoute(Address(args[1]))?.ToString() ?? "none"}"); break;
327 case "ACL":
328 {
329 Interface iface = FindInterface(args[1]); int direction = args[2] == "IN" ? 0 : 1;
330 int sequence = Range(args[3], 1, 9999, "bad sequence");
331 Prefix source = CheckedPrefix(args[6], true); Ports sourcePorts = ParsePorts(args[7]);
332 Prefix destination = CheckedPrefix(args[8], true); Ports destinationPorts = ParsePorts(args[9]);
333 if (args[5] is "ICMP" or "ANY" && (args[7] != "ANY" || args[9] != "ANY")) throw new CommandError("ports not allowed for " + args[5]);
334 if (iface.Acl[direction].ContainsKey(sequence)) throw new CommandError("duplicate sequence " + args[3]);
335 iface.Acl[direction][sequence] = new() { Sequence = sequence, Action = args[4], Protocol = args[5], Source = source, SourcePorts = sourcePorts, Destination = destination, DestinationPorts = destinationPorts, Established = args.Length == 11 };
336 Console.WriteLine($"ok acl {args[1]} {args[2].ToLowerInvariant()} {args[3]}"); break;
337 }
338 case "NOACL":
339 {
340 Interface iface = FindInterface(args[1]); int direction = args[2] == "IN" ? 0 : 1;
341 if (!iface.Acl[direction].TryGetValue(int.Parse(args[3]), out Rule? rule)) throw new CommandError($"no rule {args[3]} on {args[1]} {args[2].ToLowerInvariant()}");
342 iface.Acl[direction].Remove(rule.Sequence); Console.WriteLine($"ok noacl {args[1]} {args[2].ToLowerInvariant()} {args[3]} hits={rule.Hits}"); break;
343 }
344 case "NAT":
345 if (args[1] == "STATIC")
346 {
347 uint inside = Address(args[2]), outside = Address(args[3]);
348 if (LocalAddress(inside) is not null) throw new CommandError($"address {args[2]} is local");
349 if (LocalAddress(outside) is not null) throw new CommandError($"address {args[3]} is local");
350 if (Statics.Any(mapping => mapping.Inside == inside)) throw new CommandError("duplicate static " + args[2]);
351 if (Statics.Any(mapping => mapping.Outside == outside)) throw new CommandError("duplicate static " + args[3]);
352 Statics.Add(new(inside, outside)); Console.WriteLine($"ok nat static {args[2]} {args[3]}");
353 }
354 else
355 {
356 Interface iface = FindInterface(args[2]);
357 if (args[1] == "OUTSIDE")
358 {
359 if (iface.Outside) throw new CommandError($"interface {args[2]} already outside");
360 iface.Outside = true; Console.WriteLine("ok nat outside " + args[2]);
361 }
362 else if (args[1] == "PAT")
363 {
364 int count = Range(args[3], 1, 1000, "bad pool size");
365 if (iface.Pool > 0) throw new CommandError("pat already on " + args[2]);
366 iface.Pool = count; iface.Outside = true; Console.WriteLine($"ok nat pat {args[2]} {Ip(iface.Prefix.Address)} 20000-{19999 + count}");
367 }
368 else
369 {
370 int port = Range(args[4], 1, 65535, "bad port"); uint inside = Address(args[5]); int insidePort = Range(args[6], 1, 65535, "bad port");
371 if (!iface.Outside) throw new CommandError($"interface {args[2]} not outside");
372 if (LocalAddress(inside) is not null) throw new CommandError($"address {args[5]} is local");
373 if (Forwards.Any(rule => rule.Interface == iface && rule.Protocol == args[3] && rule.Port == port)) throw new CommandError($"duplicate forward {args[2]} {args[3]} {args[4]}");
374 Forwards.Add(new(iface, args[3], port, inside, insidePort)); Console.WriteLine($"ok nat forward {args[2]} {args[3]} {args[4]} {args[5]}:{args[6]}");
375 }
376 }
377 break;
378 case "RPF":
379 {
380 Interface iface = FindInterface(args[1]); iface.Rpf = args[2].ToLowerInvariant(); Console.WriteLine($"ok rpf {args[1]} {iface.Rpf}"); break;
381 }
382 case "TIMEOUT": Timeouts[args[1]] = Range(args[2], 1, 86400, "bad timeout"); Console.WriteLine($"ok timeout {args[1]} {args[2]}"); break;
383 case "TICK":
384 {
385 clock += Range(args[1], 1, 86400, "bad tick");
386 int removed = Connections.RemoveAll(connection => clock - connection.Last >= (connection.Original.Protocol == "TCP" ? connection.State switch { "SYN_SENT" or "FIN_WAIT" => 60, "CLOSED" => 10, _ => Timeouts["TCP"] } : Timeouts[connection.Original.Protocol]));
387 int mappings = Pats.RemoveAll(mapping => clock - mapping.Last >= Timeouts[mapping.Protocol]);
388 Console.WriteLine($"ok tick time={clock} expired conns={removed} nat={mappings}"); break;
389 }
390 case "PACKET":
391 {
392 Interface iface = FindInterface(args[1]);
393 if (args[2] != "TCP" && args[3] != "-") throw new CommandError("flags not allowed for " + args[2]);
394 if (args[2] == "TCP" && args[3] != "-" && (args[3].Any(c => !"SAFR".Contains(c)) || args[3].Distinct().Count() != args[3].Length)) throw new CommandError("bad flags " + args[3]);
395 uint source = Address(args[4]); int sourcePort = Range(args[5], 0, 65535, "bad port");
396 uint destination = Address(args[6]); int destinationPort = Range(args[7], 0, 65535, "bad port"), ttl = Range(args[8], 1, 255, "bad ttl");
397 packetNumber++;
398 string verdict = ProcessPacket(iface, new(args[2], source, sourcePort, destination, destinationPort), args[3], ttl);
399 Console.WriteLine($"#{packetNumber} {args[2]} {args[3]} {args[4]}:{args[5]} > {args[6]}:{args[7]} {verdict}"); break;
400 }
401 case "SHOW": Show(args[1]); break;
402 case "CLEAR":
403 if (args[1] == "CONNS") { Console.WriteLine($"ok clear conns={Connections.Count} nat={Pats.Count}"); Connections.Clear(); Pats.Clear(); }
404 else
405 {
406 forwarded = local = errors = 0; Array.Clear(Drops);
407 foreach (Interface iface in Interfaces) { iface.Rx = iface.Tx = iface.Drop = 0; Array.Clear(iface.Defaults); foreach (var rules in iface.Acl) foreach (Rule rule in rules.Values) rule.Hits = 0; }
408 foreach (Route route in Routes) route.Hits = 0;
409 Console.WriteLine("ok clear counters");
410 }
411 break;
412 }
413 }
414 static void Main()
415 {
416 int count = int.Parse(Console.ReadLine()!);
417 for (int i = 0; i < count; i++)
418 try { Execute(Console.ReadLine()!.Split(' ')); }
419 catch (CommandError error) { errors++; Console.WriteLine("error: " + error.Message); }
420 }
