1local function split(s, separator)
2 local out, at = {}, 1
3 while true do
4 local p = s:find(separator, at, true)
5 if not p then
6 out[#out + 1] = s:sub(at)
7 return out
8 end
9 out[#out + 1] = s:sub(at, p - 1)
10 at = p + #separator
11 end
12end

14local function bad(format, ...)
15 error(string.format(format, ...), 0)
16end

18local function decimal(s)
19 return s == "0" or s:match("^[1-9][0-9]*$") ~= nil
20end
21
22local function address(a)
23 return string.format("%d.%d.%d.%d",(a >> 24) & 255,(a >> 16) & 255,(a >> 8) & 255, a & 255)
24end
25
26local function parse_address(s)
27 local fields = split(s, ".")
28 if #fields ~= 4 then
29 return nil
30 end
31 local a = 0
32 for _, p in ipairs(fields) do
33 if #p > 3 or not decimal(p) or tonumber(p) > 255 then
34 return nil
35 end
36 a = a * 256 + tonumber(p)
37 end
38 return a
39end
40
41local function network(a, length)
42 return a &((0xffffffff <<(32 - length)) & 0xffffffff)
43end
44
45local function parse_prefix(s)
46 local parts = split(s, "/")
47 if #parts ~= 2 or #parts[2] > 2 or not decimal(parts[2]) or tonumber(parts[2]) > 32 then
48 return nil
49 end
50 local a = parse_address(parts[1])
51 if not a then
52 return nil
53 end
54 return { address = a, length = tonumber(parts[2]) }
55end
56
57local function contains(p, a)
58 return network(a, p.length) == network(p.address, p.length)
59end
60
61local function prefix_text(p)
62 return address(p.address) .. "/" .. p.length
63end
64
65local function checked_address(s)
66 local a = parse_address(s)
67 if not a then
68 bad("bad address %s", s)
69 end
70 return a
71end
72
73local function checked_prefix(s)
74 local p = parse_prefix(s)
75 if not p then
76 bad("bad prefix %s", s)
77 end
78 if p.address ~= network(p.address, p.length) then
79 bad("host bits set in %s", s)
80 end
81 return p
82end

84local function same_prefix(a, b)
85 return a.address == b.address and a.length == b.length
86end

88local function ports(s)
89 if s == "ANY" then
90 return { lo = 0, hi = 65535, any = true }
91 end
92 local p = split(s, "-")
93 local lo, hi = tonumber(p[1]), tonumber(p[#p])
94 if lo > 65535 or hi > 65535 or lo > hi then
95 bad("bad port range %s", s)
96 end
97 return { lo = lo, hi = hi, any = false }
98end
99
100local function ports_text(p)
101 if p.any then
102 return "any"
103 end
104 if p.lo == p.hi then
105 return tostring(p.lo)
106 end
107 return p.lo .. "-" .. p.hi
108end

118local ifaces, by_name, routes, statics, forwards, conns = {}, {}, {}, {}, {}, {}
119local timeouts = { TCP = 300, UDP = 60, ICMP = 30 }
120local clock, packet_number, connection_number = 0, 0, 0
121local forwarded, local_count, error_count, drops = 0, 0, 0, {}
122local REASONS = { "down", "martian", "rpf", "invalid", "acl_in", "ttl", "noroute", "acl_out", "exhausted" }

139local function select_route(a)
140 local best
141 for _, r in ipairs(routes) do
142 if r.out.up and contains(r.prefix, a) and (not best or r.prefix.length > best.prefix.length or r.prefix.length == best.prefix.length and r.metric < best.metric) then
143 best = r
144 end
145 end
146 return best
147end

149local function route_text(r)
150 return r.kind .. " " .. prefix_text(r.prefix) ..(r.hop and " via " .. address(r.hop) or "") .. " dev " .. r.out.name .. " metric " .. r.metric
151end

153local function same_tuple(a, b)
154 return a.proto == b.proto and a.src == b.src and a.sport == b.sport and a.dst == b.dst and a.dport == b.dport
155end

157local function evaluate_acl(i, direction, t, existing)
158 local rules = i.rules[direction]
159 if #rules == 0 then
160 return nil
161 end
162 for _, r in ipairs(rules) do
163 local match =(r.proto == "ANY" or r.proto == t.proto) and contains(r.src, t.src) and contains(r.dst, t.dst) and (not r.established or existing)
164 if r.proto == "TCP" or r.proto == "UDP" then
165 match = match and t.sport >= r.sports.lo and t.sport <= r.sports.hi and t.dport >= r.dports.lo and t.dport <= r.dports.hi
166 end
167 if match then
168 r.hits = r.hits + 1
169 if r.action == "PERMIT" then
170 return nil
171 end
172 if r.action == "DENY" then
173 return string.format("DROP acl %s %s %d", i.name, direction, r.seq)
174 end
175 return string.format("REJECT acl %s %s %d %s", i.name, direction, r.seq, t.proto == "TCP" and "tcp-reset" or "icmp-prohibited")
176 end
177 end
178 i.defaults[direction] = i.defaults[direction] + 1
179 return "DROP acl " .. i.name .. " " .. direction .. " default"
180end

182local function martian(a)
183 return a >> 24 == 0 or a >> 24 == 127 or a >> 29 == 7
184end

186local function packet(i, t, flags, ttl)
187 local function drop(reason)
188 return "DROP " .. reason, reason
189 end
190 if not i.up then
191 return drop("down")
192 end
193 if martian(t.src) or martian(t.dst) or local_address(t.src) then
194 return drop("martian")
195 end
196 if i.rpf ~= "off" then
197 local back = select_route(t.src)
198 if not back or i.rpf == "strict" and back.out ~= i then
199 return drop("rpf")
200 end
201 end
202 local pure = t.proto == "TCP" and flags == "S"
203 local conn, way = nil, "new"
204 for n, c in ipairs(conns) do
205 if same_tuple(t, c.orig) then
206 conn, way = c, "orig"
207 elseif same_tuple(t, c.reply) then
208 conn, way = c, "reply"
209 end
210 if conn then
211 if c.state == "CLOSED" and pure then
212 table.remove(conns, n)
213 conn, way = nil, "new"
214 end
215 break
216 end
217 end
218 if t.proto == "TCP" and not conn and not pure then
219 return drop("invalid")
220 end
221 local verdict = evaluate_acl(i, "in", t, conn ~= nil)
222 if verdict then
223 return verdict, "acl_in"
224 end
225 local translated = { proto = t.proto, src = t.src, sport = t.sport, dst = t.dst, dport = t.dport }
226 local used
227 if i.outside then
228 local matched = false
229 for _, m in ipairs(statics) do
230 if m.outside == t.dst then
231 translated.dst = m.inside
232 matched = true
233 break
234 end
235 end
236 if not matched and t.dst == i.prefix.address then
237 for _, f in ipairs(forwards) do
238 if f.out == i and f.proto == t.proto and f.port == t.dport then
239 translated.dst, translated.dport = f.inside, f.inport
240 matched = true
241 break
242 end
243 end
244 if not matched then
245 for _, m in ipairs(i.mappings) do
246 if m.proto == t.proto and m.outport == t.dport then
247 translated.dst, translated.dport, used = m.inside, m.inport, m
248 break
249 end
250 end
251 end
252 end
253 end
254 local local_iface = local_address(translated.dst)
255 if local_iface then
256 return "LOCAL " .. local_iface.name, "local"
257 end
258 if ttl == 1 then
259 return drop("ttl")
260 end
261 local route = select_route(translated.dst)
262 if not route then
263 return drop("noroute")
264 end
265 local o = route.out
266 verdict = evaluate_acl(o, "out", translated, conn ~= nil)
267 if verdict then
268 return verdict, "acl_out"
269 end
270 if not i.outside and o.outside then
271 local matched = false
272 for _, m in ipairs(statics) do
273 if m.inside == t.src then
274 translated.src = m.outside
275 matched = true
276 break
277 end
278 end
279 if not matched then
280 for _, f in ipairs(forwards) do
281 if f.out == o and f.proto == t.proto and f.inside == t.src and f.inport == t.sport then
282 translated.src, translated.sport = o.prefix.address, f.port
283 matched = true
284 break
285 end
286 end
287 end
288 if not matched and o.pool then
289 for _, m in ipairs(o.mappings) do
290 if m.proto == t.proto and m.inside == t.src and m.inport == t.sport then
291 used = m
292 break
293 end
294 end
295 if not used then
296 local chosen
297 for port = 20000, 19999 + o.pool do
298 local busy = false
299 for _, m in ipairs(o.mappings) do
300 if m.outport == port then
301 busy = true
302 end
303 end
304 for _, f in ipairs(forwards) do
305 if f.out == o and f.port == port then
306 busy = true
307 end
308 end
309 if not busy then
310 chosen = port
311 break
312 end
313 end
314 if not chosen then
315 return drop("exhausted")
316 end
317 used = { proto = t.proto, inside = t.src, inport = t.sport, outport = chosen, last = clock }
318 o.mappings[#o.mappings + 1] = used
319 end
320 translated.src, translated.sport = o.prefix.address, used.outport
321 end
322 end
323 if used then
324 used.last = clock
325 end
326 if not conn then
327 connection_number = connection_number + 1
328 conn = { number = connection_number, orig = t, reply = { proto = t.proto, src = translated.dst, sport = translated.dport, dst = translated.src, dport = translated.sport }, input = i, out = o, state = t.proto == "TCP" and "SYN_SENT" or "UNREPLIED", orig_count = 0, reply_count = 0, fin = {}, last = clock }
329 conns[#conns + 1] = conn
330 way = "new"
331 end
332 if way == "reply" then
333 conn.reply_count = conn.reply_count + 1
334 else
335 conn.orig_count = conn.orig_count + 1
336 end
337 conn.last = clock
338 if t.proto ~= "TCP" then
339 if way == "reply" then
340 conn.state = "REPLIED"
341 end
342 elseif flags:find("R", 1, true) then
343 conn.state = "CLOSED"
344 else
345 if conn.state == "SYN_SENT" and way == "reply" and flags:find("S", 1, true) and flags:find("A", 1, true) then
346 conn.state = "ESTABLISHED"
347 end
348 if flags:find("F", 1, true) and (conn.state == "ESTABLISHED" or conn.state == "FIN_WAIT") then
349 conn.fin[way == "reply" and "reply" or "orig"] = true
350 conn.state = conn.fin.orig and conn.fin.reply and "CLOSED" or "FIN_WAIT"
351 end
352 end
353 o.tx, route.hits = o.tx + 1, route.hits + 1
354 local v = string.format("FORWARD %s %s ttl=%d", o.name, route.hop and "via " .. address(route.hop) or "direct", ttl - 1)
355 if translated.dst ~= t.dst or translated.dport ~= t.dport then
356 v = v .. string.format(" dnat=%s:%d", address(translated.dst), translated.dport)
357 end
358 if translated.src ~= t.src or translated.sport ~= t.sport then
359 v = v .. string.format(" snat=%s:%d", address(translated.src), translated.sport)
360 end
361 return v .. string.format(" conn=%d %s", conn.number, way), "forwarded"
362end

364local function expire()
365 local kept, removed_nat = {}, 0
366 for _, c in ipairs(conns) do
367 local timeout = timeouts[c.orig.proto]
368 if c.orig.proto == "TCP" then
369 if c.state == "SYN_SENT" or c.state == "FIN_WAIT" then
370 timeout = 60
371 elseif c.state == "CLOSED" then
372 timeout = 10
373 end
374 end
375 if clock - c.last < timeout then
376 kept[#kept + 1] = c
377 end
378 end
379 local removed = #conns - #kept
380 conns = kept
381 for _, i in ipairs(ifaces) do
382 kept = {}
383 for _, m in ipairs(i.mappings) do
384 if clock - m.last >= timeouts[m.proto] then
385 removed_nat = removed_nat + 1
386 else
387 kept[#kept + 1] = m
388 end
389 end
390 i.mappings = kept
391 end
392 return removed, removed_nat
393end

395local function show(what)
396 local out = {}
397 local function add(format, ...)
398 out[#out + 1] = string.format(format, ...)
399 end
400 local ordered = sorted(ifaces, function (a, b)
401 return a.name < b.name
402 end)
403 if what == "IFACES" then
404 add("show ifaces count=%d", #ifaces)
405 for _, i in ipairs(ordered) do
406 local s = string.format("%s %s %s %s rpf=%s", i.name, prefix_text(i.prefix), i.up and "up" or "down", i.outside and "outside" or "inside", i.rpf)
407 if i.pool then
408 s = s .. string.format(" pat=%d/%d", #i.mappings, i.pool)
409 end
410 out[#out + 1] = s
411 end
412 elseif what == "ROUTES" then
413 local positions = {}
414 for n, r in ipairs(routes) do
415 positions[r] = n
416 end
417 local rs = sorted(routes, function (a, b)
418 if a.prefix.address ~= b.prefix.address then
419 return a.prefix.address < b.prefix.address
420 end
421 if a.prefix.length ~= b.prefix.length then
422 return a.prefix.length < b.prefix.length
423 end
424 if a.metric ~= b.metric then
425 return a.metric < b.metric
426 end
427 return positions[a] < positions[b]
428 end)
429 add("show routes count=%d", #rs)
430 for _, r in ipairs(rs) do
431 add("%s hits=%d%s", route_text(r), r.hits, r.out.up and "" or " inactive")
432 end
433 elseif what == "ACL" then
434 local count = 0
435 for _, i in ipairs(ifaces) do
436 count = count + #i.rules["in"] + #i.rules.out
437 end
438 add("show acl count=%d", count)
439 for _, i in ipairs(ordered) do
440 for _, d in ipairs({ "in", "out" }) do
441 for _, r in ipairs(i.rules[d]) do
442 add("acl %s %s %d %s %s %s %s %s %s%s hits=%d", i.name, d, r.seq, r.action, r.proto, r.src.length == 0 and "any" or prefix_text(r.src), ports_text(r.sports), r.dst.length == 0 and "any" or prefix_text(r.dst), ports_text(r.dports), r.established and " established" or "", r.hits)
443 end
444 if #i.rules[d] > 0 then
445 add("acl %s %s default DENY hits=%d", i.name, d, i.defaults[d])
446 end
447 end
448 end
449 elseif what == "NAT" then
450 local count = 0
451 for _, i in ipairs(ifaces) do
452 count = count + #i.mappings
453 end
454 add("show nat static=%d forward=%d pat=%d", #statics, #forwards, count)
455 for _, m in ipairs(sorted(statics, function (a, b)
456 return a.inside < b.inside
457 end)) do
458 add("static %s %s", address(m.inside), address(m.outside))
459 end
460 local fs = sorted(forwards, function (a, b)
461 if a.out.name ~= b.out.name then
462 return a.out.name < b.out.name
463 end
464 if a.port ~= b.port then
465 return a.port < b.port
466 end
467 return a.proto < b.proto
468 end)
469 for _, f in ipairs(fs) do
470 add("forward %s %s %d %s:%d", f.out.name, f.proto, f.port, address(f.inside), f.inport)
471 end
472 for _, i in ipairs(ordered) do
473 for _, m in ipairs(sorted(i.mappings, function (a, b)
474 return a.outport < b.outport
475 end)) do
476 add("pat %s %d %s %s:%d idle=%d", i.name, m.outport, m.proto, address(m.inside), m.inport, clock - m.last)
477 end
478 end
479 elseif what == "CONNS" then
480 add("show conns count=%d", #conns)
481 for _, c in ipairs(conns) do
482 local a, b = c.orig, c.reply
483 add("conn %d %s %s:%d > %s:%d reply %s:%d > %s:%d in=%s out=%s state=%s pkts=%d/%d idle=%d", c.number, a.proto, address(a.src), a.sport, address(a.dst), a.dport, address(b.src), b.sport, address(b.dst), b.dport, c.input.name, c.out.name, c.state, c.orig_count, c.reply_count, clock - c.last)
484 end
485 elseif what == "COUNTERS" then
486 local total = 0
487 for _, count in pairs(drops) do
488 total = total + count
489 end
490 add("show counters packets=%d forwarded=%d local=%d dropped=%d errors=%d", forwarded + local_count + total, forwarded, local_count, total, error_count)
491 local parts = { "drops" }
492 for _, reason in ipairs(REASONS) do
493 parts[#parts + 1] = reason .. "=" ..(drops[reason] or 0)
494 end
495 out[#out + 1] = table.concat(parts, " ")
496 for _, i in ipairs(ordered) do
497 add("iface %s rx=%d tx=%d drop=%d", i.name, i.rx, i.tx, i.drop)
498 end
499 end
500 return table.concat(out, "\n")
501end

503local function command(w)
504 local cmd = w[1]
505 if cmd == "IFACE" then
506 if by_name[w[2]] then
507 bad("duplicate interface %s", w[2])
508 end
509 local p = parse_prefix(w[3])
510 if not p or p.length < 8 or p.length > 30 then
511 bad("bad prefix %s", w[3])
512 end
513 local net = network(p.address, p.length)
514 if p.address == net or p.address ==(net |(0xffffffff >> p.length)) then
515 bad("bad host address %s", w[3])
516 end
517 for _, i in ipairs(ifaces) do
518 local length = math.min(p.length, i.prefix.length)
519 if network(p.address, length) == network(i.prefix.address, length) then
520 bad("subnet overlaps %s", i.name)
521 end
522 end
523 local i = { name = w[2], prefix = p, up = true, outside = false, rpf = "off", rx = 0, tx = 0, drop = 0, rules = { ["in"] = {}, out = {} }, defaults = { ["in"] = 0, out = 0 }, mappings = {} }
524 ifaces[#ifaces + 1], by_name[i.name] = i, i
525 routes[#routes + 1] = { kind = "C", prefix = { address = net, length = p.length }, out = i, metric = 0, hits = 0 }
526 return string.format("ok iface %s %s network %s/%d", i.name, w[3], address(net), p.length)
527 elseif cmd == "UP" or cmd == "DOWN" then
528 local i, status = interface(w[2]), cmd:lower()
529 if i.up ==(cmd == "UP") then
530 bad("interface %s already %s", i.name, status)
531 end
532 i.up = cmd == "UP"
533 return "ok " .. status .. " " .. i.name
534 elseif cmd == "ROUTE" then
535 local p, hop, out = checked_prefix(w[2]), nil, nil
536 if w[3] == "VIA" then
537 hop = checked_address(w[4])
538 else
539 out = interface(w[4])
540 end
541 local metric = tonumber(w[5])
542 if metric < 1 or metric > 255 then
543 bad("bad metric %s", w[5])
544 end
545 if hop then
546 if local_address(hop) then
547 bad("nexthop %s is local", w[4])
548 end
549 for _, i in ipairs(ifaces) do
550 if contains(i.prefix, hop) then
551 out = i
552 break
553 end
554 end
555 if not out then
556 bad("nexthop %s unreachable", w[4])
557 end
558 end
559 for _, r in ipairs(routes) do
560 if r.kind == "S" and same_prefix(r.prefix, p) and r.out == out and r.hop == hop then
561 bad("duplicate route %s", w[2])
562 end
563 end
564 local r = { kind = "S", prefix = p, out = out, hop = hop, metric = metric, hits = 0 }
565 routes[#routes + 1] = r
566 return "ok route " .. route_text(r)
567 elseif cmd == "NOROUTE" then
568 local p, kept, count = checked_prefix(w[2]), {}, 0
569 for _, r in ipairs(routes) do
570 if r.kind == "S" and same_prefix(r.prefix, p) then
571 count = count + 1
572 else
573 kept[#kept + 1] = r
574 end
575 end
576 if count == 0 then
577 bad("no static route %s", w[2])
578 end
579 routes = kept
580 return string.format("ok noroute %s removed=%d", w[2], count)
581 elseif cmd == "LOOKUP" then
582 local r = select_route(checked_address(w[2]))
583 return "lookup " .. w[2] .. " " ..(r and route_text(r) or "none")
584 elseif cmd == "ACL" then
585 local i, seq, d = interface(w[2]), tonumber(w[4]), w[3]:lower()
586 if seq < 1 or seq > 9999 then
587 bad("bad sequence %s", w[4])
588 end
589 local function acl_prefix(s)
590 if s == "ANY" then
591 return { address = 0, length = 0 }
592 end
593 return checked_prefix(s)
594 end
595 local src = acl_prefix(w[7])
596 local sports = ports(w[8])
597 local dst = acl_prefix(w[9])
598 local dports = ports(w[10])
599 if (w[6] == "ICMP" or w[6] == "ANY") and (not sports.any or not dports.any) then
600 bad("ports not allowed for %s", w[6])
601 end
602 for _, r in ipairs(i.rules[d]) do
603 if r.seq == seq then
604 bad("duplicate sequence %s", w[4])
605 end
606 end
607 i.rules[d][#i.rules[d] + 1] = { seq = seq, action = w[5], proto = w[6], src = src, sports = sports, dst = dst, dports = dports, established = #w == 11, hits = 0 }
608 table.sort(i.rules[d], function (a, b)
609 return a.seq < b.seq
610 end)
611 return "ok acl " .. i.name .. " " .. d .. " " .. w[4]
612 elseif cmd == "NOACL" then
613 local i, d = interface(w[2]), w[3]:lower()
614 for n, r in ipairs(i.rules[d]) do
615 if r.seq == tonumber(w[4]) then
616 table.remove(i.rules[d], n)
617 return string.format("ok noacl %s %s %s hits=%d", i.name, d, w[4], r.hits)
618 end
619 end
620 bad("no rule %s on %s %s", w[4], i.name, d)
621 elseif cmd == "NAT" then
622 if w[2] == "STATIC" then
623 local a = checked_address(w[3])
624 local b = checked_address(w[4])
625 if local_address(a) then
626 bad("address %s is local", w[3])
627 end
628 if local_address(b) then
629 bad("address %s is local", w[4])
630 end
631 for _, m in ipairs(statics) do
632 if m.inside == a then
633 bad("duplicate static %s", w[3])
634 end
635 end
636 for _, m in ipairs(statics) do
637 if m.outside == b then
638 bad("duplicate static %s", w[4])
639 end
640 end
641 statics[#statics + 1] = { inside = a, outside = b }
642 return "ok nat static " .. w[3] .. " " .. w[4]
643 end
644 local i = interface(w[3])
645 if w[2] == "OUTSIDE" then
646 if i.outside then
647 bad("interface %s already outside", i.name)
648 end
649 i.outside = true
650 return "ok nat outside " .. i.name
651 elseif w[2] == "PAT" then
652 local n = tonumber(w[4])
653 if n < 1 or n > 1000 then
654 bad("bad pool size %s", w[4])
655 end
656 if i.pool then
657 bad("pat already on %s", i.name)
658 end
659 i.pool, i.outside = n, true
660 return string.format("ok nat pat %s %s 20000-%d", i.name, address(i.prefix.address), 19999 + n)
661 else
662 local port = tonumber(w[5])
663 if port < 1 or port > 65535 then
664 bad("bad port %s", w[5])
665 end
666 local a = checked_address(w[6])
667 local inport = tonumber(w[7])
668 if inport < 1 or inport > 65535 then
669 bad("bad port %s", w[7])
670 end
671 if not i.outside then
672 bad("interface %s not outside", i.name)
673 end
674 if local_address(a) then
675 bad("address %s is local", w[6])
676 end
677 for _, f in ipairs(forwards) do
678 if f.out == i and f.proto == w[4] and f.port == port then
679 bad("duplicate forward %s %s %s", i.name, w[4], w[5])
680 end
681 end
682 forwards[#forwards + 1] = { out = i, proto = w[4], port = port, inside = a, inport = inport }
683 return string.format("ok nat forward %s %s %s %s:%s", i.name, w[4], w[5], w[6], w[7])
684 end
685 elseif cmd == "RPF" then
686 local i = interface(w[2])
687 i.rpf = w[3]:lower()
688 return "ok rpf " .. i.name .. " " .. i.rpf
689 elseif cmd == "TIMEOUT" then
690 local n = tonumber(w[3])
691 if n < 1 or n > 86400 then
692 bad("bad timeout %s", w[3])
693 end
694 timeouts[w[2]] = n
695 return "ok timeout " .. w[2] .. " " .. w[3]
696 elseif cmd == "TICK" then
697 local n = tonumber(w[2])
698 if n < 1 or n > 86400 then
699 bad("bad tick %s", w[2])
700 end
701 clock = clock + n
702 local c, m = expire()
703 return string.format("ok tick time=%d expired conns=%d nat=%d", clock, c, m)
704 elseif cmd == "PACKET" then
705 local i, proto, flags = interface(w[2]), w[3], w[4]
706 if proto ~= "TCP" and flags ~= "-" then
707 bad("flags not allowed for %s", proto)
708 end
709 if proto == "TCP" and flags ~= "-" then
710 local seen = {}
711 for c in flags:gmatch(".") do
712 if not ("SAFR"):find(c, 1, true) or seen[c] then
713 bad("bad flags %s", flags)
714 end
715 seen[c] = true
716 end
717 end
718 local src = checked_address(w[5])
719 local sp = tonumber(w[6])
720 if sp > 65535 then
721 bad("bad port %s", w[6])
722 end
723 local dst = checked_address(w[7])
724 local dp = tonumber(w[8])
725 if dp > 65535 then
726 bad("bad port %s", w[8])
727 end
728 local ttl = tonumber(w[9])
729 if ttl < 1 or ttl > 255 then
730 bad("bad ttl %s", w[9])
731 end
732 packet_number, i.rx = packet_number + 1, i.rx + 1
733 local verdict, reason = packet(i, { proto = proto, src = src, sport = sp, dst = dst, dport = dp }, flags, ttl)
734 if reason == "forwarded" then
735 forwarded = forwarded + 1
736 elseif reason == "local" then
737 local_count = local_count + 1
738 else
739 drops[reason] =(drops[reason] or 0) + 1
740 i.drop = i.drop + 1
741 end
742 return string.format("#%d %s %s %s:%s > %s:%s %s", packet_number, proto, flags, w[5], w[6], w[7], w[8], verdict)
743 elseif cmd == "SHOW" then
744 return show(w[2])
745 elseif cmd == "CLEAR" then
746 if w[2] == "CONNS" then
747 local count, mappings = #conns, 0
748 conns = {}
749 for _, i in ipairs(ifaces) do
750 mappings = mappings + #i.mappings
751 i.mappings = {}
752 end
753 return string.format("ok clear conns=%d nat=%d", count, mappings)
754 end
755 forwarded, local_count, error_count, drops = 0, 0, 0, {}
756 for _, i in ipairs(ifaces) do
757 i.rx, i.tx, i.drop = 0, 0, 0
758 i.defaults = { ["in"] = 0, out = 0 }
759 for _, rs in pairs(i.rules) do
760 for _, r in ipairs(rs) do
761 r.hits = 0
762 end
763 end
764 end
765 for _, r in ipairs(routes) do
766 r.hits = 0
767 end
768 return "ok clear counters"
769 end
770end
