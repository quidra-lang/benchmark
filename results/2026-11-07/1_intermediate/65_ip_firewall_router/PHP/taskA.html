31 foreach ($items as $item) {

39 private function address(string $word): int
40 {
41 if (!preg_match('/\A(?:0|[1-9][0-9]{0,2})(?:\.(?:0|[1-9][0-9]{0,2})){3}\z/', $word)) {
42 $this->fail("bad address $word");
43 }
44 $value = 0;
45 foreach (explode('.', $word) as $part) {
46 if ((int)$part > 255) {
47 $this->fail("bad address $word");
48 }
49 $value = $value * 256 + (int)$part;
50 }
51 return $value;
52 }

54 private function ip(int $value): string
55 {
56 return implode('.', array_map(fn($shift) => ($value >> $shift) & 255, [24, 16, 8, 0]));
57 }

59 private function network(int $address, int $length): int
60 {
61 $size = 2 ** (32 - $length);
62 return intdiv($address, $size) * $size;
63 }

65 private function prefix(string $word, bool $hosts = false): array
66 {
67 $parts = explode('/', $word);
68 try {
69 if (count($parts) !== 2 || !preg_match('/\A(?:0|[1-9][0-9]?)\z/', $parts[1]) || (int)$parts[1] > 32) {
70 $this->fail('invalid');
71 }
72 $address = $this->address($parts[0]);
73 } catch (RouterFault $error) {
74 $this->fail("bad prefix $word");
75 }
76 $length = (int)$parts[1];
77 if (!$hosts && $address !== $this->network($address, $length)) {
78 $this->fail("host bits set in $word");
79 }
80 return [$address, $length];
81 }

106 private function route(int $address): ?object
107 {
108 $best = null;
109 foreach ($this->routes as $item) {
110 if ($this->interfaces[$item->dev]->up && $this->contains($item->prefix, $address) &&
111 ($best === null || $item->prefix[1] > $best->prefix[1] || ($item->prefix[1] === $best->prefix[1] && $item->metric < $best->metric))) {
112 $best = $item;
113 }
114 }
115 return $best;

124 private function portRange(string $word): array
125 {
126 if ($word === 'ANY') {
127 return [0, 65535, true];
128 }
129 $parts = explode('-', $word);
130 $low = (int)$parts[0];
131 $high = (int)($parts[1] ?? $parts[0]);
132 if ($low > 65535 || $high > 65535 || $low > $high) {
133 $this->fail("bad port range $word");
134 }
135 return [$low, $high, false];
136 }

153 private function acl(string $name, string $direction, string $protocol, int $source, int $sport, int $destination, int $dport, bool $established): ?string
154 {
155 $pair = $this->interfaces[$name]->acl[$direction];
156 if ($pair->rules === []) {
157 return null;
158 }
159 $rules = $pair->rules;
160 usort($rules, fn($a, $b) => $a->seq <=> $b->seq);
161 foreach ($rules as $rule) {
162 if (($rule->proto !== 'ANY' && $rule->proto !== $protocol) || !$this->contains($rule->src, $source) || !$this->contains($rule->dst, $destination) ||
163 (in_array($rule->proto, ['TCP', 'UDP'], true) && ($sport < $rule->sports[0] || $sport > $rule->sports[1] || $dport < $rule->dports[0] || $dport > $rule->dports[1])) ||
164 ($rule->established && !$established)) {
165 continue;
166 }
167 $rule->hits++;
168 return match ($rule->action) {
169 'PERMIT' => null,
170 'DENY' => "DROP acl $name $direction $rule->seq",
171 default => "REJECT acl $name $direction $rule->seq " . ($protocol === 'TCP' ? 'tcp-reset' : 'icmp-prohibited')
172 };
173 }
174 $pair->default++;
175 return "DROP acl $name $direction default";
176 }

178 private function packet(string $name, string $protocol, string $flags, int $source, int $sport, int $destination, int $dport, int $ttl): array
179 {
180 $input = $this->interfaces[$name];
181 if (!$input->up) return ['DROP down', 'down'];
182 foreach ([$source, $destination] as $address) {
183 if ($this->contains([0, 8], $address) || $this->contains([127 * 2 ** 24, 8], $address) || $this->contains([224 * 2 ** 24, 3], $address)) {
184 return ['DROP martian', 'martian'];
185 }
186 }
187 if ($this->local($source) !== null) return ['DROP martian', 'martian'];
188 if ($input->rpf !== 'off') {
189 $reverse = $this->route($source);
190 if ($reverse === null || ($input->rpf === 'strict' && $reverse->dev !== $name)) return ['DROP rpf', 'rpf'];
191 }
192 $tuple = [$protocol, $source, $sport, $destination, $dport];
193 $pureSyn = $protocol === 'TCP' && $flags === 'S';
194 $connection = $this->find($this->connections, fn($item) => $item->orig === $tuple || $item->reply === $tuple);
195 $direction = $connection === null ? null : ($connection->orig === $tuple ? 'orig' : 'reply');
196 if ($connection !== null && $connection->state === 'CLOSED' && $pureSyn) {
197 $this->connections = array_values(array_filter($this->connections, fn($item) => $item !== $connection));
198 $connection = null;
199 }
200 if ($protocol === 'TCP' && $connection === null && !$pureSyn) return ['DROP invalid', 'invalid'];
201 $verdict = $this->acl($name, 'in', $protocol, $source, $sport, $destination, $dport, $connection !== null);
202 if ($verdict !== null) return [$verdict, 'acl_in'];
203 $translatedDestination = $destination;
204 $translatedDport = $dport;
205 $mapping = null;
206 if ($input->role === 'outside') {
207 $static = $this->find($this->static, fn($item) => $item->outside === $destination);
208 $forward = $destination === $input->address ? $this->find($this->forwards, fn($item) => $item->dev === $name && $item->proto === $protocol && $item->port === $dport) : null;
209 if ($static !== null) {
210 $translatedDestination = $static->inside;
211 } elseif ($forward !== null) {
212 $translatedDestination = $forward->inside;
213 $translatedDport = $forward->inport;
214 } elseif ($destination === $input->address) {
215 $mapping = $this->find($this->pat, fn($item) => $item->dev === $name && $item->proto === $protocol && $item->port === $dport);
216 if ($mapping !== null) {
217 $translatedDestination = $mapping->inside;
218 $translatedDport = $mapping->inport;
219 }
220 }
221 }
222 $local = $this->local($translatedDestination);
223 if ($local !== null) return ["LOCAL $local", 'local'];
224 if ($ttl === 1) return ['DROP ttl', 'ttl'];
225 $selected = $this->route($translatedDestination);
226 if ($selected === null) return ['DROP noroute', 'noroute'];
227 $outputName = $selected->dev;
228 $output = $this->interfaces[$outputName];
229 $verdict = $this->acl($outputName, 'out', $protocol, $source, $sport, $translatedDestination, $translatedDport, $connection !== null);
230 if ($verdict !== null) return [$verdict, 'acl_out'];
231 $translatedSource = $source;
232 $translatedSport = $sport;
233 if ($input->role === 'inside' && $output->role === 'outside') {
234 $static = $this->find($this->static, fn($item) => $item->inside === $source);
235 $forward = $this->find($this->forwards, fn($item) => $item->dev === $outputName && $item->proto === $protocol && $item->inside === $source && $item->inport === $sport);
236 if ($static !== null) {
237 $translatedSource = $static->outside;
238 } elseif ($forward !== null) {
239 $translatedSource = $output->address;
240 $translatedSport = $forward->port;
241 } elseif ($output->pool !== null) {
242 $mapping = $this->find($this->pat, fn($item) => $item->dev === $outputName && $item->proto === $protocol && $item->inside === $source && $item->inport === $sport);
243 if ($mapping === null) {
244 $used = [];
245 foreach ([...$this->pat, ...$this->forwards] as $item) {
246 if ($item->dev === $outputName) $used[$item->port] = true;
247 }
248 $port = 20000;
249 while ($port < 20000 + $output->pool && isset($used[$port])) $port++;
250 if ($port === 20000 + $output->pool) return ['DROP exhausted', 'exhausted'];
251 $mapping = (object)['dev' => $outputName, 'proto' => $protocol, 'inside' => $source, 'inport' => $sport, 'port' => $port, 'last' => $this->clock];
252 $this->pat[] = $mapping;
253 }
254 $translatedSource = $output->address;
255 $translatedSport = $mapping->port;
256 }
257 }
258 if ($mapping !== null) $mapping->last = $this->clock;
259 if ($connection === null) {
260 $connection = (object)['number' => ++$this->connectionNumber, 'proto' => $protocol, 'orig' => $tuple,
261 'reply' => [$protocol, $translatedDestination, $translatedDport, $translatedSource, $translatedSport],
262 'input' => $name, 'output' => $outputName, 'state' => $protocol === 'TCP' ? 'SYN_SENT' : 'UNREPLIED',
263 'origPackets' => 0, 'replyPackets' => 0, 'origFin' => false, 'replyFin' => false, 'last' => $this->clock];
264 $this->connections[] = $connection;
265 $direction = 'new';
266 }
267 $side = $direction === 'reply' ? 'reply' : 'orig';
268 $packetField = $side . 'Packets';
269 $connection->{$packetField}++;
270 $connection->last = $this->clock;
271 if ($protocol !== 'TCP') {
272 if ($direction === 'reply') $connection->state = 'REPLIED';
273 } elseif (str_contains($flags, 'R')) {
274 $connection->state = 'CLOSED';
275 } else {
276 if ($connection->state === 'SYN_SENT' && $direction === 'reply' && str_contains($flags, 'S') && str_contains($flags, 'A')) $connection->state = 'ESTABLISHED';
277 if (str_contains($flags, 'F') && in_array($connection->state, ['ESTABLISHED', 'FIN_WAIT'], true)) {
278 $finField = $side . 'Fin';
279 $connection->{$finField} = true;
280 $connection->state = $connection->origFin && $connection->replyFin ? 'CLOSED' : 'FIN_WAIT';
281 }
282 }
283 $output->tx++;
284 $selected->hits++;
285 $nextHop = $selected->hop === null ? 'direct' : 'via ' . $this->ip($selected->hop);
286 $text = "FORWARD $outputName $nextHop ttl=" . ($ttl - 1);
287 if ([$translatedDestination, $translatedDport] !== [$destination, $dport]) $text .= ' dnat=' . $this->ip($translatedDestination) . ":$translatedDport";
288 if ([$translatedSource, $translatedSport] !== [$source, $sport]) $text .= ' snat=' . $this->ip($translatedSource) . ":$translatedSport";
289 return [$text . " conn=$connection->number $direction", 'forwarded'];
290 }

292 private function resetCounters(): void
293 {
294 $this->counts = array_fill_keys([...self::DROPS, 'forwarded', 'local', 'errors'], 0);
295 foreach ($this->interfaces as $item) {
296 $item->rx = $item->tx = $item->drop = 0;
297 foreach ($item->acl as $pair) {
298 $pair->default = 0;
299 foreach ($pair->rules as $rule) $rule->hits = 0;
300 }
301 }
302 foreach ($this->routes as $item) $item->hits = 0;
303 }

305 private function show(string $what): void
306 {
307 $names = array_keys($this->interfaces);
308 sort($names, SORT_STRING);
309 switch ($what) {
310 case 'IFACES':
311 echo 'show ifaces count=' . count($names) . "\n";
312 foreach ($names as $name) {
313 $item = $this->interfaces[$name];
314 $text = $name . ' ' . $this->ip($item->address) . "/$item->length " . ($item->up ? 'up' : 'down') . " $item->role rpf=$item->rpf";
315 if ($item->pool !== null) $text .= ' pat=' . count(array_filter($this->pat, fn($mapping) => $mapping->dev === $name)) . "/$item->pool";
316 echo "$text\n";
317 }
318 break;
319 case 'ROUTES':
320 echo 'show routes count=' . count($this->routes) . "\n";
321 $routes = $this->routes;
322 usort($routes, fn($a, $b) => [$a->prefix[0], $a->prefix[1], $a->metric] <=> [$b->prefix[0], $b->prefix[1], $b->metric]);
323 foreach ($routes as $item) echo $this->routeText($item) . " hits=$item->hits" . ($this->interfaces[$item->dev]->up ? '' : ' inactive') . "\n";
324 break;
325 case 'ACL':
326 $count = 0;
327 foreach ($this->interfaces as $item) foreach ($item->acl as $pair) $count += count($pair->rules);
328 echo "show acl count=$count\n";
329 foreach ($names as $name) {
330 foreach (['in', 'out'] as $direction) {
331 $pair = $this->interfaces[$name]->acl[$direction];
332 if ($pair->rules === []) continue;
333 $rules = $pair->rules;
334 usort($rules, fn($a, $b) => $a->seq <=> $b->seq);
335 foreach ($rules as $rule) {
336 echo "acl $name $direction $rule->seq $rule->action $rule->proto " . $this->prefixText($rule->src) . ' ' . $this->rangeText($rule->sports) . ' ' .
337 $this->prefixText($rule->dst) . ' ' . $this->rangeText($rule->dports) . ($rule->established ? ' established' : '') . " hits=$rule->hits\n";
338 }
339 echo "acl $name $direction default DENY hits=$pair->default\n";
340 }
341 }
342 break;
343 case 'NAT':
344 echo 'show nat static=' . count($this->static) . ' forward=' . count($this->forwards) . ' pat=' . count($this->pat) . "\n";
345 $static = $this->static;
346 usort($static, fn($a, $b) => $a->inside <=> $b->inside);
347 foreach ($static as $item) echo 'static ' . $this->ip($item->inside) . ' ' . $this->ip($item->outside) . "\n";
348 $forwards = $this->forwards;
349 usort($forwards, fn($a, $b) => strcmp($a->dev, $b->dev) ?: ($a->port <=> $b->port) ?: strcmp($a->proto, $b->proto));
350 foreach ($forwards as $item) echo "forward $item->dev $item->proto $item->port " . $this->ip($item->inside) . ":$item->inport\n";
351 $pat = $this->pat;
352 usort($pat, fn($a, $b) => strcmp($a->dev, $b->dev) ?: ($a->port <=> $b->port));
353 foreach ($pat as $item) echo "pat $item->dev $item->port $item->proto " . $this->ip($item->inside) . ":$item->inport idle=" . ($this->clock - $item->last) . "\n";
354 break;
355 case 'CONNS':
356 echo 'show conns count=' . count($this->connections) . "\n";
357 foreach ($this->connections as $item) {
358 [, $source, $sport, $destination, $dport] = $item->orig;
359 [, $replySource, $replySport, $replyDestination, $replyDport] = $item->reply;
360 echo "conn $item->number $item->proto " . $this->ip($source) . ":$sport > " . $this->ip($destination) . ":$dport reply " . $this->ip($replySource) . ":$replySport > " .
361 $this->ip($replyDestination) . ":$replyDport in=$item->input out=$item->output state=$item->state pkts=$item->origPackets/$item->replyPackets idle=" . ($this->clock - $item->last) . "\n";
362 }
363 break;
364 case 'COUNTERS':
365 $dropped = array_sum(array_map(fn($reason) => $this->counts[$reason], self::DROPS));
366 echo 'show counters packets=' . ($this->counts['forwarded'] + $this->counts['local'] + $dropped) . ' forwarded=' . $this->counts['forwarded'] . ' local=' . $this->counts['local'] . " dropped=$dropped errors=" . $this->counts['errors'] . "\n";
367 echo 'drops ' . implode(' ', array_map(fn($reason) => "$reason=" . $this->counts[$reason], self::DROPS)) . "\n";
368 foreach ($names as $name) {
369 $item = $this->interfaces[$name];
370 echo "iface $name rx=$item->rx tx=$item->tx drop=$item->drop\n";
371 }
372 break;
373 }

376 private function command(array $words): void
377 {
378 $kind = array_shift($words);
379 $a = $words;
380 $name = $a[0];
381 switch ($kind) {
382 case 'IFACE':
383 if (isset($this->interfaces[$name])) $this->fail("duplicate interface $name");
384 [$address, $length] = $this->prefix($a[1], true);
385 if ($length < 8 || $length > 30) $this->fail("bad prefix $a[1]");
386 $base = $this->network($address, $length);
387 if ($address === $base || $address === $base + 2 ** (32 - $length) - 1) $this->fail("bad host address $a[1]");
388 foreach ($this->interfaces as $otherName => $item) {
389 $common = min($length, $item->length);
390 if ($this->network($address, $common) === $this->network($item->address, $common)) $this->fail("subnet overlaps $otherName");
391 }
392 $this->interfaces[$name] = (object)['address' => $address, 'length' => $length, 'up' => true, 'role' => 'inside', 'pool' => null, 'rpf' => 'off', 'rx' => 0, 'tx' => 0, 'drop' => 0,
393 'acl' => ['in' => (object)['rules' => [], 'default' => 0], 'out' => (object)['rules' => [], 'default' => 0]]];
394 $this->routes[] = (object)['type' => 'C', 'prefix' => [$base, $length], 'dev' => $name, 'hop' => null, 'metric' => 0, 'hits' => 0];
395 echo "ok iface $name $a[1] network " . $this->ip($base) . "/$length\n";
396 break;
397 case 'UP':
398 case 'DOWN':
399 $item = $this->iface($name);
400 $state = $kind === 'UP';
401 if ($item->up === $state) $this->fail("interface $name already " . ($state ? 'up' : 'down'));
402 $item->up = $state;
403 echo 'ok ' . strtolower($kind) . " $name\n";
404 break;
405 case 'ROUTE':
406 $prefix = $this->prefix($name);
407 $via = $a[1] === 'VIA';
408 $hop = $via ? $this->address($a[2]) : null;
409 if (!$via) $this->iface($a[2]);
410 $metric = (int)$a[3];
411 if ($metric < 1 || $metric > 255) $this->fail("bad metric $a[3]");
412 $output = $a[2];
413 if ($via) {
414 if ($this->local($hop) !== null) $this->fail("nexthop $a[2] is local");
415 $output = null;
416 foreach ($this->interfaces as $key => $item) if ($this->contains([$item->address, $item->length], $hop)) $output = $key;
417 if ($output === null) $this->fail("nexthop $a[2] unreachable");
418 }
419 if ($this->find($this->routes, fn($item) => $item->type === 'S' && $item->prefix === $prefix && $item->dev === $output && $item->hop === $hop)) $this->fail("duplicate route $name");
420 $item = (object)['type' => 'S', 'prefix' => $prefix, 'dev' => $output, 'hop' => $hop, 'metric' => $metric, 'hits' => 0];
421 $this->routes[] = $item;
422 echo 'ok route ' . $this->routeText($item) . "\n";
423 break;
424 case 'NOROUTE':
425 $prefix = $this->prefix($name);
426 $remaining = array_values(array_filter($this->routes, fn($item) => $item->type !== 'S' || $item->prefix !== $prefix));
427 $removed = count($this->routes) - count($remaining);
428 if ($removed === 0) $this->fail("no static route $name");
429 $this->routes = $remaining;
430 echo "ok noroute $name removed=$removed\n";
431 break;
432 case 'LOOKUP':
433 $selected = $this->route($this->address($name));
434 echo "lookup $name " . ($selected === null ? 'none' : $this->routeText($selected)) . "\n";
435 break;
436 case 'ACL':
437 $item = $this->iface($name);
438 $direction = strtolower($a[1]);
439 $seq = (int)$a[2];
440 if ($seq < 1 || $seq > 9999) $this->fail("bad sequence $a[2]");
441 $source = $this->aclPrefix($a[5]);
442 $sports = $this->portRange($a[6]);
443 $destination = $this->aclPrefix($a[7]);
444 $dports = $this->portRange($a[8]);
445 if (in_array($a[4], ['ICMP', 'ANY'], true) && ($a[6] !== 'ANY' || $a[8] !== 'ANY')) $this->fail("ports not allowed for $a[4]");
446 $pair = $item->acl[$direction];
447 if ($this->find($pair->rules, fn($rule) => $rule->seq === $seq)) $this->fail("duplicate sequence $a[2]");
448 $pair->rules[] = (object)['seq' => $seq, 'action' => $a[3], 'proto' => $a[4], 'src' => $source, 'sports' => $sports, 'dst' => $destination, 'dports' => $dports, 'established' => count($a) === 10, 'hits' => 0];
449 echo "ok acl $name $direction $seq\n";
450 break;
451 case 'NOACL':
452 $pair = $this->iface($name)->acl[strtolower($a[1])];
453 $seq = (int)$a[2];
454 $rule = $this->find($pair->rules, fn($item) => $item->seq === $seq);
455 $direction = strtolower($a[1]);
456 if ($rule === null) $this->fail("no rule $a[2] on $name $direction");
457 $pair->rules = array_values(array_filter($pair->rules, fn($item) => $item !== $rule));
458 echo "ok noacl $name $direction $seq hits=$rule->hits\n";
459 break;
460 case 'NAT':
461 if ($name === 'STATIC') {
462 $inside = $this->address($a[1]);
463 $outside = $this->address($a[2]);
464 if ($this->local($inside) !== null) $this->fail("address $a[1] is local");
465 if ($this->local($outside) !== null) $this->fail("address $a[2] is local");
466 if ($this->find($this->static, fn($item) => $item->inside === $inside)) $this->fail("duplicate static $a[1]");
467 if ($this->find($this->static, fn($item) => $item->outside === $outside)) $this->fail("duplicate static $a[2]");
468 $this->static[] = (object)['inside' => $inside, 'outside' => $outside];
469 echo "ok nat static $a[1] $a[2]\n";
470 } elseif ($name === 'OUTSIDE') {
471 $item = $this->iface($a[1]);
472 if ($item->role === 'outside') $this->fail("interface $a[1] already outside");
473 $item->role = 'outside';
474 echo "ok nat outside $a[1]\n";
475 } elseif ($name === 'PAT') {
476 $item = $this->iface($a[1]);
477 $count = (int)$a[2];
478 if ($count < 1 || $count > 1000) $this->fail("bad pool size $a[2]");
479 if ($item->pool !== null) $this->fail("pat already on $a[1]");
480 $item->pool = $count;
481 $item->role = 'outside';
482 echo "ok nat pat $a[1] " . $this->ip($item->address) . ' 20000-' . (19999 + $count) . "\n";
483 } else {
484 $item = $this->iface($a[1]);
485 $port = (int)$a[3];
486 if ($port < 1 || $port > 65535) $this->fail("bad port $a[3]");
487 $inside = $this->address($a[4]);
488 $inport = (int)$a[5];
489 if ($inport < 1 || $inport > 65535) $this->fail("bad port $a[5]");
490 if ($item->role !== 'outside') $this->fail("interface $a[1] not outside");
491 if ($this->local($inside) !== null) $this->fail("address $a[4] is local");
492 if ($this->find($this->forwards, fn($rule) => $rule->dev === $a[1] && $rule->proto === $a[2] && $rule->port === $port)) $this->fail("duplicate forward $a[1] $a[2] $a[3]");
493 $this->forwards[] = (object)['dev' => $a[1], 'proto' => $a[2], 'port' => $port, 'inside' => $inside, 'inport' => $inport];
494 echo "ok nat forward $a[1] $a[2] $port $a[4]:$inport\n";
495 }
496 break;
497 case 'RPF':
498 $item = $this->iface($name);
499 $item->rpf = strtolower($a[1]);
500 echo "ok rpf $name $item->rpf\n";
501 break;
502 case 'TIMEOUT':
503 $seconds = (int)$a[1];
504 if ($seconds < 1 || $seconds > 86400) $this->fail("bad timeout $a[1]");
505 $this->timeouts[$name] = $seconds;
506 echo "ok timeout $name $seconds\n";
507 break;
508 case 'TICK':
509 $seconds = (int)$name;
510 if ($seconds < 1 || $seconds > 86400) $this->fail("bad tick $name");
511 $this->clock += $seconds;
512 $conns = count($this->connections);
513 $mappings = count($this->pat);
514 $this->connections = array_values(array_filter($this->connections, function ($item) {
515 $timeout = $item->proto === 'TCP' ? (['SYN_SENT' => 60, 'FIN_WAIT' => 60, 'CLOSED' => 10][$item->state] ?? $this->timeouts['TCP']) : $this->timeouts[$item->proto];
516 return $this->clock - $item->last < $timeout;
517 }));
518 $this->pat = array_values(array_filter($this->pat, fn($item) => $this->clock - $item->last < $this->timeouts[$item->proto]));
519 echo "ok tick time=$this->clock expired conns=" . ($conns - count($this->connections)) . ' nat=' . ($mappings - count($this->pat)) . "\n";
520 break;
521 case 'PACKET':
522 $item = $this->iface($name);
523 $protocol = $a[1];
524 $flags = $a[2];
525 if ($protocol !== 'TCP' && $flags !== '-') $this->fail("flags not allowed for $protocol");
526 if ($protocol === 'TCP' && $flags !== '-' && (!preg_match('/\A[SAFR]+\z/', $flags) || count(array_unique(str_split($flags))) !== strlen($flags))) $this->fail("bad flags $flags");
527 $source = $this->address($a[3]);
528 $sport = (int)$a[4];
529 if ($sport > 65535) $this->fail("bad port $a[4]");
530 $destination = $this->address($a[5]);
531 $dport = (int)$a[6];
532 if ($dport > 65535) $this->fail("bad port $a[6]");
533 $ttl = (int)$a[7];
534 if ($ttl < 1 || $ttl > 255) $this->fail("bad ttl $a[7]");
535 $this->packetNumber++;
536 $item->rx++;
537 [$verdict, $reason] = $this->packet($name, $protocol, $flags, $source, $sport, $destination, $dport, $ttl);
538 $this->counts[$reason]++;
539 if (in_array($reason, self::DROPS, true)) $item->drop++;
540 echo "#$this->packetNumber $protocol $flags $a[3]:$a[4] > $a[5]:$a[6] $verdict\n";
541 break;
542 case 'SHOW':
543 $this->show($name);
544 break;
545 case 'CLEAR':
546 if ($name === 'COUNTERS') {
547 $this->resetCounters();
548 echo "ok clear counters\n";
549 } else {
550 echo 'ok clear conns=' . count($this->connections) . ' nat=' . count($this->pat) . "\n";
551 $this->connections = [];
552 $this->pat = [];
553 }
554 break;
555 }
556 }
557
558 public function run(): void
559 {
560 $count = (int)fgets(STDIN);
561 for ($index = 0; $index < $count; $index++) {
562 try {
563 $this->command(explode(' ', trim(fgets(STDIN))));
564 } catch (RouterFault $error) {
565 $this->counts['errors']++;
566 echo 'error: ' . $error->getMessage() . "\n";
567 }
568 }
569 }
570}
