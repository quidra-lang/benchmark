121String addressText(int address) => [24, 16, 8, 0].map((shift) => (address >> shift) & 255).join('.');
122int network(int address, int length) => (address >> (32 - length)) << (32 - length);
123bool contains(Prefix prefix, int address) => network(prefix.address, prefix.length) == network(address, prefix.length);

125int parseAddress(String text) {
126 final fields = text.split('.');
127 if (fields.length != 4 || fields.any((field) => !RegExp(r'^(?:0|[1-9][0-9]{0,2})$').hasMatch(field) || int.parse(field) > 255)) {
128 throw RouterFailure('bad address $text');
129 }
130 return fields.fold<int>(0, (value, field) => value * 256 + int.parse(field));
131}
132
133Prefix parsePrefix(String text, {bool checkHost = true}) {
134 final fields = text.split('/');
135 final int address;
136 try {
137 if (fields.length != 2 || !RegExp(r'^(?:0|[1-9][0-9]?)$').hasMatch(fields[1]) || int.parse(fields[1]) > 32) throw RouterFailure('');
138 address = parseAddress(fields[0]);
139 } on RouterFailure {
140 throw RouterFailure('bad prefix $text');
141 }
142 final length = int.parse(fields[1]);
143 if (checkHost && network(address, length) != address) throw RouterFailure('host bits set in $text');
144 return Prefix(address, length);
145}

147PortRange parsePorts(String text) {
148 if (text == 'ANY') return PortRange(0, 65535, true);
149 final values = text.split('-').map(int.parse).toList();
150 final low = values.first;
151 final high = values.last;
152 if (low > 65535 || high > 65535 || low > high) throw RouterFailure('bad port range $text');
153 return PortRange(low, high, false);
154}

156String prefixText(Prefix prefix) => prefix.length == 0 ? 'any' : '${addressText(prefix.address)}/${prefix.length}';
157String portsText(PortRange ports) => ports.any ? 'any' : ports.low == ports.high ? '${ports.low}' : '${ports.low}-${ports.high}';
158String routeText(Route route) => '${route.type} ${addressText(route.address)}/${route.length}${route.hop == null ? '' : ' via ${addressText(route.hop!)}'} dev ${route.output.name} metric ${route.metric}';
159String tupleText(PacketTuple tuple) => '${addressText(tuple.source)}:${tuple.sport} > ${addressText(tuple.destination)}:${tuple.dport}';
160bool sameTuple(PacketTuple a, PacketTuple b) => a.protocol == b.protocol && a.source == b.source && a.sport == b.sport && a.destination == b.destination && a.dport == b.dport;

187 Route? select(int address) {
188 Route? best;
189 for (final route in routes) {
190 if (!route.output.up || !contains(route, address)) continue;
191 if (best == null || route.length > best.length || route.length == best.length && route.metric < best.metric) best = route;
192 }
193 return best;
194 }

196 String? evaluateAcl(Interface iface, String direction, PacketTuple tuple, bool established) {
197 final acl = iface.acl[direction]!;
198 if (acl.rules.isEmpty) return null;
199 final rules = acl.rules.toList()..sort((a, b) => a.sequence.compareTo(b.sequence));
200 for (final rule in rules) {
201 if (rule.protocol != 'ANY' && rule.protocol != tuple.protocol) continue;
202 if (!contains(rule.source, tuple.source) || !contains(rule.destination, tuple.destination)) continue;
203 if ((rule.protocol == 'TCP' || rule.protocol == 'UDP') && (tuple.sport < rule.sports.low || tuple.sport > rule.sports.high || tuple.dport < rule.dports.low || tuple.dport > rule.dports.high)) continue;
204 if (rule.established && !established) continue;
205 rule.hits++;
206 if (rule.action == 'PERMIT') return null;
207 final prefix = '${iface.name} $direction ${rule.sequence}';
208 return rule.action == 'DENY' ? 'DROP acl $prefix' : 'REJECT acl $prefix ${tuple.protocol == 'TCP' ? 'tcp-reset' : 'icmp-prohibited'}';
209 }
210 acl.denied++;
211 return 'DROP acl ${iface.name} $direction default';
212 }

214 String processPacket(Interface input, PacketTuple tuple, String flags, int ttl) {
215 String finish(String verdict, String reason) {
216 if (reason == 'forwarded') {
217 forwarded++;
218 } else if (reason == 'local') {
219 local++;
220 } else {
221 drops[reason] = drops[reason]! + 1;
222 input.drop++;
223 }
224 return verdict;
225 }
226 if (!input.up) return finish('DROP down', 'down');
227 bool martian(int address) => address < 16777216 || address >> 24 == 127 || address >= 3758096384;
228 if (martian(tuple.source) || martian(tuple.destination) || localInterface(tuple.source) != null) return finish('DROP martian', 'martian');
229 if (input.rpf != 'off') {
230 final reverse = select(tuple.source);
231 if (reverse == null || input.rpf == 'strict' && reverse.output != input) return finish('DROP rpf', 'rpf');
232 }
233 final pureSyn = tuple.protocol == 'TCP' && flags == 'S';
234 Connection? connection;
235 var direction = 'new';
236 for (final candidate in connections) {
237 if (sameTuple(tuple, candidate.original)) {
238 connection = candidate;
239 direction = 'orig';
240 break;
241 }
242 if (sameTuple(tuple, candidate.reply)) {
243 connection = candidate;
244 direction = 'reply';
245 break;
246 }
247 }
248 if (connection != null && connection.state == 'CLOSED' && pureSyn) {
249 connections.remove(connection);
250 connection = null;
251 direction = 'new';
252 }
253 if (tuple.protocol == 'TCP' && connection == null && !pureSyn) return finish('DROP invalid', 'invalid');
254 final inbound = evaluateAcl(input, 'in', tuple, connection != null);
255 if (inbound != null) return finish(inbound, 'acl_in');
256 var destination = tuple.destination;
257 var dport = tuple.dport;
258 Pat? used;
259 if (input.outside) {
260 final mapping = find(staticNat, (mapping) => mapping.outside == destination);
261 if (mapping != null) {
262 destination = mapping.inside;
263 } else if (destination == input.address) {
264 final forward = find(forwards, (rule) => rule.output == input && rule.protocol == tuple.protocol && rule.outport == dport);
265 if (forward != null) {
266 destination = forward.inside;
267 dport = forward.inport;
268 } else {
269 used = find(input.pat, (mapping) => mapping.protocol == tuple.protocol && mapping.outport == dport);
270 if (used != null) {
271 destination = used.inside;
272 dport = used.inport;
273 }
274 }
275 }
276 }
277 final localTarget = localInterface(destination);
278 if (localTarget != null) return finish('LOCAL ${localTarget.name}', 'local');
279 if (ttl == 1) return finish('DROP ttl', 'ttl');
280 final route = select(destination);
281 if (route == null) return finish('DROP noroute', 'noroute');
282 final output = route.output;
283 final outbound = evaluateAcl(output, 'out', PacketTuple(tuple.protocol, tuple.source, tuple.sport, destination, dport), connection != null);
284 if (outbound != null) return finish(outbound, 'acl_out');
285 var source = tuple.source;
286 var sport = tuple.sport;
287 if (!input.outside && output.outside) {
288 final mapping = find(staticNat, (mapping) => mapping.inside == source);
289 if (mapping != null) {
290 source = mapping.outside;
291 } else {
292 final forward = find(forwards, (rule) => rule.output == output && rule.protocol == tuple.protocol && rule.inside == source && rule.inport == sport);
293 if (forward != null) {
294 source = output.address;
295 sport = forward.outport;
296 } else if (output.pool != null) {
297 used = find(output.pat, (mapping) => mapping.protocol == tuple.protocol && mapping.inside == source && mapping.inport == sport);
298 if (used == null) {
299 var port = 20000;
300 while (port < 20000 + output.pool! && (output.pat.any((mapping) => mapping.outport == port) || forwards.any((rule) => rule.output == output && rule.outport == port))) port++;
301 if (port == 20000 + output.pool!) return finish('DROP exhausted', 'exhausted');
302 used = Pat(tuple.protocol, source, sport, port, clock);
303 output.pat.add(used);
304 }
305 source = output.address;
306 sport = used.outport;
307 }
308 }
309 }
310 if (used != null) used.used = clock;
311 if (connection == null) {
312 connection = Connection(++connectionNumber, tuple, PacketTuple(tuple.protocol, destination, dport, source, sport), input, output, clock);
313 connections.add(connection);
314 direction = 'new';
315 }
316 if (direction == 'reply') {
317 connection.replyPackets++;
318 } else {
319 connection.origPackets++;
320 }
321 connection.used = clock;
322 if (tuple.protocol != 'TCP') {
323 if (direction == 'reply') connection.state = 'REPLIED';
324 } else if (flags.contains('R')) {
325 connection.state = 'CLOSED';
326 } else {
327 if (connection.state == 'SYN_SENT' && direction == 'reply' && flags.contains('S') && flags.contains('A')) connection.state = 'ESTABLISHED';
328 if (flags.contains('F') && (connection.state == 'ESTABLISHED' || connection.state == 'FIN_WAIT')) {
329 if (direction == 'reply') {
330 connection.replyFin = true;
331 } else {
332 connection.origFin = true;
333 }
334 connection.state = connection.origFin && connection.replyFin ? 'CLOSED' : 'FIN_WAIT';
335 }
336 }
337 output.tx++;
338 route.hits++;
339 var verdict = 'FORWARD ${output.name} ${route.hop == null ? 'direct' : 'via ${addressText(route.hop!)}'} ttl=${ttl - 1}';
340 if (destination != tuple.destination || dport != tuple.dport) verdict += ' dnat=${addressText(destination)}:$dport';
341 if (source != tuple.source || sport != tuple.sport) verdict += ' snat=${addressText(source)}:$sport';
342 return finish('$verdict conn=${connection.number} $direction', 'forwarded');
343 }

345 void command(List<String> words) {
346 final command = words[0];
347 if (command == 'IFACE') {
348 final name = words[1];
349 final text = words[2];
350 if (interfaces.containsKey(name)) throw RouterFailure('duplicate interface $name');
351 final prefix = parsePrefix(text, checkHost: false);
352 if (prefix.length < 8 || prefix.length > 30) throw RouterFailure('bad prefix $text');
353 final net = network(prefix.address, prefix.length);
354 if (prefix.address == net || prefix.address == net + (1 << (32 - prefix.length)) - 1) throw RouterFailure('bad host address $text');
355 for (final iface in interfaces.values) {
356 final length = prefix.length < iface.length ? prefix.length : iface.length;
357 if (network(prefix.address, length) == network(iface.address, length)) throw RouterFailure('subnet overlaps ${iface.name}');
358 }
359 final iface = Interface(name, prefix.address, prefix.length);
360 interfaces[name] = iface;
361 routes.add(Route('C', net, prefix.length, iface, null, 0, routeOrder++));
362 output.add('ok iface $name $text network ${addressText(net)}/${prefix.length}');
363 } else if (command == 'UP' || command == 'DOWN') {
364 final interface = iface(words[1]);
365 final up = command == 'UP';
366 if (interface.up == up) throw RouterFailure('interface ${interface.name} already ${up ? 'up' : 'down'}');
367 interface.up = up;
368 output.add('ok ${command.toLowerCase()} ${interface.name}');
369 } else if (command == 'ROUTE') {
370 final prefix = parsePrefix(words[1]);
371 Interface? outputInterface;
372 int? hop;
373 if (words[2] == 'VIA') {
374 hop = parseAddress(words[3]);
375 } else {
376 outputInterface = iface(words[3]);
377 }
378 final metric = int.parse(words[4]);
379 if (metric < 1 || metric > 255) throw RouterFailure('bad metric ${words[4]}');
380 if (hop != null) {
381 if (localInterface(hop) != null) throw RouterFailure('nexthop ${words[3]} is local');
382 outputInterface = find(interfaces.values, (interface) => contains(interface, hop!));
383 if (outputInterface == null) throw RouterFailure('nexthop ${words[3]} unreachable');
384 }
385 if (routes.any((route) => route.type == 'S' && route.address == prefix.address && route.length == prefix.length && route.output == outputInterface && route.hop == hop)) throw RouterFailure('duplicate route ${words[1]}');
386 final route = Route('S', prefix.address, prefix.length, outputInterface!, hop, metric, routeOrder++);
387 routes.add(route);
388 output.add('ok route ${routeText(route)}');
389 } else if (command == 'NOROUTE') {
390 final prefix = parsePrefix(words[1]);
391 bool matches(Route route) => route.type == 'S' && route.address == prefix.address && route.length == prefix.length;
392 final removed = routes.where(matches).length;
393 if (removed == 0) throw RouterFailure('no static route ${words[1]}');
394 routes.removeWhere(matches);
395 output.add('ok noroute ${words[1]} removed=$removed');
396 } else if (command == 'LOOKUP') {
397 final route = select(parseAddress(words[1]));
398 output.add('lookup ${words[1]} ${route == null ? 'none' : routeText(route)}');
399 } else if (command == 'ACL') {
400 final interface = iface(words[1]);
401 final direction = words[2].toLowerCase();
402 final sequence = int.parse(words[3]);
403 if (sequence < 1 || sequence > 9999) throw RouterFailure('bad sequence ${words[3]}');
404 final source = words[6] == 'ANY' ? Prefix(0, 0) : parsePrefix(words[6]);
405 final sports = parsePorts(words[7]);
406 final destination = words[8] == 'ANY' ? Prefix(0, 0) : parsePrefix(words[8]);
407 final dports = parsePorts(words[9]);
408 if ((words[5] == 'ICMP' || words[5] == 'ANY') && (!sports.any || !dports.any)) throw RouterFailure('ports not allowed for ${words[5]}');
409 final acl = interface.acl[direction]!;
410 if (acl.rules.any((rule) => rule.sequence == sequence)) throw RouterFailure('duplicate sequence ${words[3]}');
411 acl.rules.add(Rule(sequence, words[4], words[5], source, sports, destination, dports, words.length == 11));
412 output.add('ok acl ${interface.name} $direction ${words[3]}');
413 } else if (command == 'NOACL') {
414 final interface = iface(words[1]);
415 final direction = words[2].toLowerCase();
416 final acl = interface.acl[direction]!;
417 final rule = find(acl.rules, (rule) => rule.sequence == int.parse(words[3]));
418 if (rule == null) throw RouterFailure('no rule ${words[3]} on ${interface.name} $direction');
419 acl.rules.remove(rule);
420 output.add('ok noacl ${interface.name} $direction ${words[3]} hits=${rule.hits}');
421 } else if (command == 'NAT') {
422 natCommand(words);
423 } else if (command == 'RPF') {
424 final interface = iface(words[1]);
425 interface.rpf = words[2].toLowerCase();
426 output.add('ok rpf ${interface.name} ${interface.rpf}');
427 } else if (command == 'TIMEOUT') {
428 final seconds = int.parse(words[2]);
429 if (seconds < 1 || seconds > 86400) throw RouterFailure('bad timeout ${words[2]}');
430 timeouts[words[1]] = seconds;
431 output.add('ok timeout ${words[1]} ${words[2]}');
432 } else if (command == 'TICK') {
433 final seconds = int.parse(words[1]);
434 if (seconds < 1 || seconds > 86400) throw RouterFailure('bad tick ${words[1]}');
435 clock += seconds;
436 final before = connections.length;
437 connections.removeWhere((connection) {
438 final protocol = connection.original.protocol;
439 final timeout = protocol != 'TCP' || connection.state == 'ESTABLISHED' ? timeouts[protocol]! : connection.state == 'CLOSED' ? 10 : 60;
440 return clock - connection.used >= timeout;
441 });
442 var removed = 0;
443 for (final interface in interfaces.values) {
444 final count = interface.pat.length;
445 interface.pat.removeWhere((mapping) => clock - mapping.used >= timeouts[mapping.protocol]!);
446 removed += count - interface.pat.length;
447 }
448 output.add('ok tick time=$clock expired conns=${before - connections.length} nat=$removed');
449 } else if (command == 'PACKET') {
450 final interface = iface(words[1]);
451 final protocol = words[2];
452 final flags = words[3];
453 if (protocol != 'TCP' && flags != '-') throw RouterFailure('flags not allowed for $protocol');
454 if (protocol == 'TCP' && flags != '-' && (!RegExp(r'^[SAFR]+$').hasMatch(flags) || flags.split('').toSet().length != flags.length)) throw RouterFailure('bad flags $flags');
455 final source = parseAddress(words[4]);
456 final sport = int.parse(words[5]);
457 if (sport > 65535) throw RouterFailure('bad port ${words[5]}');
458 final destination = parseAddress(words[6]);
459 final dport = int.parse(words[7]);
460 if (dport > 65535) throw RouterFailure('bad port ${words[7]}');
461 final ttl = int.parse(words[8]);
462 if (ttl < 1 || ttl > 255) throw RouterFailure('bad ttl ${words[8]}');
463 packetNumber++;
464 interface.rx++;
465 final verdict = processPacket(interface, PacketTuple(protocol, source, sport, destination, dport), flags, ttl);
466 output.add('#$packetNumber $protocol $flags ${words[4]}:${words[5]} > ${words[6]}:${words[7]} $verdict');
467 } else if (command == 'SHOW') {
468 show(words[1]);
469 } else if (command == 'CLEAR') {
470 if (words[1] == 'CONNS') {
471 final mappings = interfaces.values.fold<int>(0, (total, interface) => total + interface.pat.length);
472 output.add('ok clear conns=${connections.length} nat=$mappings');
473 connections.clear();
474 for (final interface in interfaces.values) interface.pat.clear();
475 } else {
476 forwarded = local = errors = 0;
477 for (final reason in dropReasons) drops[reason] = 0;
478 for (final route in routes) route.hits = 0;
479 for (final interface in interfaces.values) {
480 interface.rx = interface.tx = interface.drop = 0;
481 for (final acl in interface.acl.values) {
482 acl.denied = 0;
483 for (final rule in acl.rules) rule.hits = 0;
484 }
485 }
486 output.add('ok clear counters');
487 }
488 }
489 }

491 void natCommand(List<String> words) {
492 if (words[1] == 'STATIC') {
493 final inside = parseAddress(words[2]);
494 final outside = parseAddress(words[3]);
495 if (localInterface(inside) != null) throw RouterFailure('address ${words[2]} is local');
496 if (localInterface(outside) != null) throw RouterFailure('address ${words[3]} is local');
497 if (staticNat.any((mapping) => mapping.inside == inside)) throw RouterFailure('duplicate static ${words[2]}');
498 if (staticNat.any((mapping) => mapping.outside == outside)) throw RouterFailure('duplicate static ${words[3]}');
499 staticNat.add(StaticNat(inside, outside));
500 output.add('ok nat static ${words[2]} ${words[3]}');
501 return;
502 }
503 final interface = iface(words[2]);
504 if (words[1] == 'OUTSIDE') {
505 if (interface.outside) throw RouterFailure('interface ${interface.name} already outside');
506 interface.outside = true;
507 output.add('ok nat outside ${interface.name}');
508 } else if (words[1] == 'PAT') {
509 final count = int.parse(words[3]);
510 if (count < 1 || count > 1000) throw RouterFailure('bad pool size ${words[3]}');
511 if (interface.pool != null) throw RouterFailure('pat already on ${interface.name}');
512 interface.pool = count;
513 interface.outside = true;
514 output.add('ok nat pat ${interface.name} ${addressText(interface.address)} 20000-${20000 + count - 1}');
515 } else {
516 final protocol = words[3];
517 final outport = int.parse(words[4]);
518 if (outport < 1 || outport > 65535) throw RouterFailure('bad port ${words[4]}');
519 final inside = parseAddress(words[5]);
520 final inport = int.parse(words[6]);
521 if (inport < 1 || inport > 65535) throw RouterFailure('bad port ${words[6]}');
522 if (!interface.outside) throw RouterFailure('interface ${interface.name} not outside');
523 if (localInterface(inside) != null) throw RouterFailure('address ${words[5]} is local');
524 if (forwards.any((rule) => rule.output == interface && rule.protocol == protocol && rule.outport == outport)) throw RouterFailure('duplicate forward ${interface.name} $protocol ${words[4]}');
525 forwards.add(Forward(interface, protocol, outport, inside, inport));
526 output.add('ok nat forward ${interface.name} $protocol ${words[4]} ${words[5]}:${words[6]}');
527 }
528 }

530 void show(String what) {
531 final sortedInterfaces = orderedInterfaces();
532 if (what == 'IFACES') {
533 output.add('show ifaces count=${sortedInterfaces.length}');
534 for (final interface in sortedInterfaces) output.add('${interface.name} ${addressText(interface.address)}/${interface.length} ${interface.up ? 'up' : 'down'} ${interface.outside ? 'outside' : 'inside'} rpf=${interface.rpf}${interface.pool == null ? '' : ' pat=${interface.pat.length}/${interface.pool}'}');
535 } else if (what == 'ROUTES') {
536 output.add('show routes count=${routes.length}');
537 final sorted = routes.toList()..sort((a, b) {
538 for (final comparison in [a.address.compareTo(b.address), a.length.compareTo(b.length), a.metric.compareTo(b.metric), a.order.compareTo(b.order)]) {
539 if (comparison != 0) return comparison;
540 }
541 return 0;
542 });
543 for (final route in sorted) output.add('${routeText(route)} hits=${route.hits}${route.output.up ? '' : ' inactive'}');
544 } else if (what == 'ACL') {
545 final count = sortedInterfaces.fold<int>(0, (total, interface) => total + interface.acl['in']!.rules.length + interface.acl['out']!.rules.length);
546 output.add('show acl count=$count');
547 for (final interface in sortedInterfaces) {
548 for (final direction in ['in', 'out']) {
549 final acl = interface.acl[direction]!;
550 if (acl.rules.isEmpty) continue;
551 final rules = acl.rules.toList()..sort((a, b) => a.sequence.compareTo(b.sequence));
552 for (final rule in rules) output.add('acl ${interface.name} $direction ${rule.sequence} ${rule.action} ${rule.protocol} ${prefixText(rule.source)} ${portsText(rule.sports)} ${prefixText(rule.destination)} ${portsText(rule.dports)}${rule.established ? ' established' : ''} hits=${rule.hits}');
553 output.add('acl ${interface.name} $direction default DENY hits=${acl.denied}');
554 }
555 }
556 } else if (what == 'NAT') {
557 final count = sortedInterfaces.fold<int>(0, (total, interface) => total + interface.pat.length);
558 output.add('show nat static=${staticNat.length} forward=${forwards.length} pat=$count');
559 final mappings = staticNat.toList()..sort((a, b) => a.inside.compareTo(b.inside));
560 for (final mapping in mappings) output.add('static ${addressText(mapping.inside)} ${addressText(mapping.outside)}');
561 final rules = forwards.toList()..sort((a, b) {
562 final name = a.output.name.compareTo(b.output.name);
563 if (name != 0) return name;
564 final port = a.outport.compareTo(b.outport);
565 return port == 0 ? a.protocol.compareTo(b.protocol) : port;
566 });
567 for (final rule in rules) output.add('forward ${rule.output.name} ${rule.protocol} ${rule.outport} ${addressText(rule.inside)}:${rule.inport}');
568 for (final interface in sortedInterfaces) {
569 final pat = interface.pat.toList()..sort((a, b) => a.outport.compareTo(b.outport));
570 for (final mapping in pat) output.add('pat ${interface.name} ${mapping.outport} ${mapping.protocol} ${addressText(mapping.inside)}:${mapping.inport} idle=${clock - mapping.used}');
571 }
572 } else if (what == 'CONNS') {
573 output.add('show conns count=${connections.length}');
574 for (final connection in connections) output.add('conn ${connection.number} ${connection.original.protocol} ${tupleText(connection.original)} reply ${tupleText(connection.reply)} in=${connection.input.name} out=${connection.output.name} state=${connection.state} pkts=${connection.origPackets}/${connection.replyPackets} idle=${clock - connection.used}');
575 } else {
576 final dropped = dropReasons.fold<int>(0, (total, reason) => total + drops[reason]!);
577 output.add('show counters packets=${forwarded + local + dropped} forwarded=$forwarded local=$local dropped=$dropped errors=$errors');
578 output.add('drops ${dropReasons.map((reason) => '$reason=${drops[reason]}').join(' ')}');
579 for (final interface in sortedInterfaces) output.add('iface ${interface.name} rx=${interface.rx} tx=${interface.tx} drop=${interface.drop}');
580 }
581 }
582
583 void run(List<String> lines) {
584 final count = int.parse(lines[0]);
585 for (final line in lines.skip(1).take(count)) {
586 try {
587 command(line.split(' '));
588 } on RouterFailure catch (error) {
589 errors++;
590 output.add('error: ${error.message}');
591 }
592 }
593 }
594}

597 final router = Router();
598 router.run((await stdin.transform(utf8.decoder).join()).split('\n'));
599 stdout.write('${router.output.join('\n')}\n');
600}
