22def parse_decimal(text, max_digits):
23 if not 1 <= len(text) <= max_digits:
24 return None
25 if any(character not in DIGITS for character in text):
26 return None
27 if len(text) > 1 and text[0] == "0":
28 return None
29 return int(text)

32def parse_address(text):
33 fields = text.split(".")
34 if len(fields) != 4:
35 return None
36 value = 0
37 for field in fields:
38 number = parse_decimal(field, 3)
39 if number is None or number > 255:
40 return None
41 value = value * 256 + number
42 return value

45def parse_prefix(text):
46 if text.count("/") != 1:
47 return None
48 address_part, length_part = text.split("/")
49 address = parse_address(address_part)
50 length = parse_decimal(length_part, 2)
51 if address is None or length is None or length > 32:
52 return None
53 return address, length

56def network_mask(length):
57 return (ALL_BITS << (32 - length)) & ALL_BITS

60def network_of(address, length):

119class PortRange:
120 def __init__(self, low, high, label):
121 self.low = low
122 self.high = high
123 self.label = label
124
125 def contains(self, port):
126 return self.low <= port <= self.high

129def require_port_range(text):
130 if text == "ANY":
131 return PortRange(0, MAX_PORT, "any")
132 bounds = [int(part) for part in text.split("-")]
133 low, high = bounds[0], bounds[-1]
134 if any(bound > MAX_PORT for bound in bounds) or low > high:
135 raise CommandError(f"bad port range {text}")
136 label = str(low) if low == high else f"{low}-{high}"
137 return PortRange(low, high, label)

140class AccessRule:
141 def __init__(self, sequence, action, protocol, source, source_ports,
142 destination, destination_ports, established):
143 self.sequence = sequence
144 self.action = action
145 self.protocol = protocol
146 self.source = source
147 self.source_ports = source_ports
148 self.destination = destination
149 self.destination_ports = destination_ports
150 self.established = established
151 self.hits = 0
152
153 def matches(self, flow, established):
154 protocol, source, source_port, destination, destination_port = flow
155 if self.protocol != "ANY" and self.protocol != protocol:
156 return False
157 if not contains(*self.source, source) or not contains(*self.destination, destination):
158 return False
159 if self.protocol in ("TCP", "UDP"):
160 if not self.source_ports.contains(source_port):
161 return False
162 if not self.destination_ports.contains(destination_port):
163 return False
164 return established or not self.established
165
166 def description(self, interface_name, direction):
167 parts = [
168 "acl", interface_name, direction, self.sequence, self.action, self.protocol,
169 acl_prefix_text(self.source), self.source_ports.label,
170 acl_prefix_text(self.destination), self.destination_ports.label,
171 ]
172 if self.established:
173 parts.append("established")
174 parts.append(f"hits={self.hits}")
175 return " ".join(str(part) for part in parts)

183class AccessList:
184 def __init__(self, interface_name, direction):
185 self.interface_name = interface_name
186 self.direction = direction
187 self.rules = []
188 self.default_hits = 0
189
190 def find(self, sequence):
191 for rule in self.rules:
192 if rule.sequence == sequence:
193 return rule
194 return None
195
196 def add(self, rule):
197 self.rules.append(rule)
198 self.rules.sort(key=lambda item: item.sequence)
199
200 def evaluate(self, flow, established):
201 if not self.rules:
202 return None
203 protocol = flow[0]
204 label = f"acl {self.interface_name} {self.direction}"
205 for rule in self.rules:
206 if not rule.matches(flow, established):
207 continue
208 rule.hits += 1
209 if rule.action == "PERMIT":
210 return None
211 if rule.action == "DENY":
212 return f"DROP {label} {rule.sequence}"
213 reason = "tcp-reset" if protocol == "TCP" else "icmp-prohibited"
214 return f"REJECT {label} {rule.sequence} {reason}"
215 self.default_hits += 1
216 return f"DROP {label} default"
217
218 def clear_counters(self):
219 self.default_hits = 0
220 for rule in self.rules:
221 rule.hits = 0

302class Connection:
303 def __init__(self, number, orig, reply, inbound, outbound):
304 self.number = number
305 self.protocol = orig[0]
306 self.orig = orig
307 self.reply = reply
308 self.inbound = inbound
309 self.outbound = outbound
310 self.state = "SYN_SENT" if self.protocol == "TCP" else "UNREPLIED"
311 self.orig_packets = 0
312 self.reply_packets = 0
313 self.last_used = 0
314 self.orig_fin = False
315 self.reply_fin = False
316
317 def record(self, direction, flags, now):
318 if direction == "reply":
319 self.reply_packets += 1
320 else:
321 self.orig_packets += 1
322 self.last_used = now
323 self.update_state(direction, flags)
324
325 def update_state(self, direction, flags):
326 if self.protocol != "TCP":
327 if direction == "reply":
328 self.state = "REPLIED"
329 return
330 if "R" in flags:
331 self.state = "CLOSED"
332 return
333 if self.state == "SYN_SENT" and direction == "reply" and "S" in flags and "A" in flags:
334 self.state = "ESTABLISHED"
335 if "F" in flags and self.state in ("ESTABLISHED", "FIN_WAIT"):
336 if direction == "reply":
337 self.reply_fin = True
338 else:
339 self.orig_fin = True
340 self.state = "CLOSED" if self.orig_fin and self.reply_fin else "FIN_WAIT"
341
342 def timeout(self, timeouts):
343 if self.protocol != "TCP":
344 return timeouts[self.protocol]
345 if self.state == "ESTABLISHED":
346 return timeouts["TCP"]
347 if self.state == "CLOSED":
348 return 10
349 return 60
350
351 def description(self, now):
352 return " ".join((
353 "conn", str(self.number), self.protocol, flow_text(self.orig),
354 "reply", flow_text(self.reply),
355 f"in={self.inbound.name}", f"out={self.outbound.name}", f"state={self.state}",
356 f"pkts={self.orig_packets}/{self.reply_packets}", f"idle={now - self.last_used}",
357 ))

423 def select_route(self, address):
424 best = None
425 for route in self.routes:
426 if not route.active() or not contains(route.network, route.length, address):
427 continue
428 if best is None or route.length > best.length:
429 best = route
430 elif route.length == best.length and route.metric < best.metric:
431 best = route

434 def add_interface(self, arguments):
435 name, cidr = arguments
436 if name in self.interfaces:
437 raise CommandError(f"duplicate interface {name}")
438 prefix = parse_prefix(cidr)
439 if prefix is None or not 8 <= prefix[1] <= 30:
440 raise CommandError(f"bad prefix {cidr}")
441 address, length = prefix
442 host_mask = ALL_BITS ^ network_mask(length)
443 host = address & host_mask
444 if host == 0 or host == host_mask:
445 raise CommandError(f"bad host address {cidr}")
446 for other in self.interfaces.values():
447 if contains(other.address, min(length, other.length), address):
448 raise CommandError(f"subnet overlaps {other.name}")
449 interface = Interface(name, address, length)
450 self.interfaces[name] = interface
451 network = network_of(address, length)
452 self.routes.append(Route("C", network, length, interface, None, 0))
453 self.emit("ok", "iface", name, cidr, "network", prefix_text(network, length))

470 def add_route(self, arguments):
471 prefix_word, mode, target, metric_word = arguments
472 network, length = require_prefix(prefix_word)
473 next_hop = None
474 interface = None
475 if mode == "VIA":
476 next_hop = require_address(target)
477 else:
478 interface = self.interface(target)
479 metric = require_number(metric_word, 1, 255, f"bad metric {metric_word}")
480 if mode == "VIA":
481 if self.interface_with_address(next_hop) is not None:
482 raise CommandError(f"nexthop {target} is local")
483 interface = self.interface_containing(next_hop)
484 if interface is None:
485 raise CommandError(f"nexthop {target} unreachable")
486 for route in self.routes:
487 if (route.is_static_to(network, length) and route.interface is interface
488 and route.next_hop == next_hop):
489 raise CommandError(f"duplicate route {prefix_word}")
490 route = Route("S", network, length, interface, next_hop, metric)
491 self.routes.append(route)
492 self.emit("ok", "route", route.notation())

497 kept = [route for route in self.routes if not route.is_static_to(network, length)]
498 removed = len(self.routes) - len(kept)
499 if removed == 0:
500 raise CommandError(f"no static route {prefix_word}")
501 self.routes = kept
502 self.emit("ok", "noroute", prefix_word, f"removed={removed}")

509 def add_rule(self, arguments):
510 (name, direction_word, sequence_word, action, protocol, source_word,
511 source_ports_word, destination_word, destination_ports_word) = arguments[:9]
512 interface = self.interface(name)
513 sequence = require_number(sequence_word, 1, 9999, f"bad sequence {sequence_word}")
514 source = require_acl_prefix(source_word)
515 source_ports = require_port_range(source_ports_word)
516 destination = require_acl_prefix(destination_word)
517 destination_ports = require_port_range(destination_ports_word)
518 if protocol in ("ICMP", "ANY") and (
519 source_ports_word != "ANY" or destination_ports_word != "ANY"
520 ):
521 raise CommandError(f"ports not allowed for {protocol}")
522 direction = direction_word.lower()
523 access_list = interface.acls[direction]
524 if access_list.find(sequence) is not None:
525 raise CommandError(f"duplicate sequence {sequence_word}")
526 established = arguments[-1] == "ESTABLISHED"
527 access_list.add(AccessRule(
528 sequence, action, protocol, source, source_ports,
529 destination, destination_ports, established,
530 ))
531 self.emit("ok", "acl", name, direction, sequence_word)

533 def remove_rule(self, arguments):
534 name, direction_word, sequence_word = arguments
535 interface = self.interface(name)
536 direction = direction_word.lower()
537 access_list = interface.acls[direction]

539 if rule is None:
540 raise CommandError(f"no rule {sequence_word} on {name} {direction}")
541 access_list.rules.remove(rule)
542 self.emit("ok", "noacl", name, direction, sequence_word, f"hits={rule.hits}")

553 def add_static(self, arguments):
554 inside_word, outside_word = arguments
555 inside = require_address(inside_word)
556 outside = require_address(outside_word)
557 for address, word in ((inside, inside_word), (outside, outside_word)):
558 if self.interface_with_address(address) is not None:
559 raise CommandError(f"address {word} is local")

563 raise CommandError(f"duplicate static {outside_word}")

565 self.emit("ok", "nat", "static", inside_word, outside_word)

623 def tick(self, arguments):
624 seconds_word = arguments[0]
625 seconds = require_number(seconds_word, 1, MAX_SECONDS, f"bad tick {seconds_word}")
626 self.clock += seconds
627 kept = [
628 connection for connection in self.connections
629 if self.clock - connection.last_used < connection.timeout(self.timeouts)
630 ]
631 expired_connections = len(self.connections) - len(kept)
632 self.connections = kept
633 expired_mappings = 0
634 for interface in self.interfaces.values():
635 kept_mappings = [
636 mapping for mapping in interface.pat_mappings
637 if self.clock - mapping.last_used < self.timeouts[mapping.protocol]
638 ]
639 expired_mappings += len(interface.pat_mappings) - len(kept_mappings)
640 interface.pat_mappings = kept_mappings
641 self.emit(
642 "ok", "tick", f"time={self.clock}", "expired",
643 f"conns={expired_connections}", f"nat={expired_mappings}",
644 )

646 def packet(self, arguments):
647 (name, protocol, flags, source_word, source_port_word,
648 destination_word, destination_port_word, ttl_word) = arguments
649 interface = self.interface(name)
650 if protocol != "TCP" and flags != "-":
651 raise CommandError(f"flags not allowed for {protocol}")
652 if protocol == "TCP" and flags != "-" and not valid_tcp_flags(flags):
653 raise CommandError(f"bad flags {flags}")
654 source = require_address(source_word)
655 source_port = require_number(
656 source_port_word, 0, MAX_PORT, f"bad port {source_port_word}"
657 )
658 destination = require_address(destination_word)
659 destination_port = require_number(
660 destination_port_word, 0, MAX_PORT, f"bad port {destination_port_word}"
661 )
662 ttl = require_number(ttl_word, 1, 255, f"bad ttl {ttl_word}")
663 self.packet_count += 1
664 interface.rx += 1
665 packet = Packet(
666 interface, protocol, "" if flags == "-" else flags,
667 source, source_port, destination, destination_port, ttl,
668 )
669 reason, verdict = self.process(packet)
670 self.counts[reason] += 1
671 if reason in DROP_REASONS:
672 interface.drop += 1
673 self.emit(
674 f"#{self.packet_count}", protocol, flags,
675 f"{source_word}:{source_port_word}", ">",
676 f"{destination_word}:{destination_port_word}", verdict,
677 )

685 def find_connection(self, packet):
686 flow = packet.flow()
687 for connection in self.connections:
688 if connection.orig == flow:
689 direction = "orig"
690 elif connection.reply == flow:
691 direction = "reply"
692 else:
693 continue
694 if connection.state == "CLOSED" and packet.pure_syn():
695 self.connections.remove(connection)
696 return None, None
697 return connection, direction
698 return None, None

700 def translate_destination(self, inbound, packet):
701 for mapping in self.statics:
702 if mapping.outside == packet.destination:
703 return mapping.inside, packet.destination_port, None
704 if packet.destination == inbound.address:
705 for forward in self.forwards:
706 if (forward.interface is inbound and forward.protocol == packet.protocol
707 and forward.outside_port == packet.destination_port):
708 return forward.inside_address, forward.inside_port, None
709 for mapping in inbound.pat_mappings:
710 if (mapping.protocol == packet.protocol
711 and mapping.outside_port == packet.destination_port):
712 return mapping.inside_address, mapping.inside_port, mapping
713 return packet.destination, packet.destination_port, None

715 def free_pat_port(self, interface):
716 used = {mapping.outside_port for mapping in interface.pat_mappings}
717 for forward in self.forwards:
718 if forward.interface is interface:
719 used.add(forward.outside_port)
720 for port in range(PAT_FIRST_PORT, PAT_FIRST_PORT + interface.pat_size):
721 if port not in used:
722 return port
723 return None

725 def translate_source(self, outbound, packet):
726 for mapping in self.statics:
727 if mapping.inside == packet.source:
728 return mapping.outside, packet.source_port, None
729 for forward in self.forwards:
730 if (forward.interface is outbound and forward.protocol == packet.protocol
731 and forward.inside_address == packet.source
732 and forward.inside_port == packet.source_port):
733 return outbound.address, forward.outside_port, None
734 if outbound.pat_size is None:
735 return packet.source, packet.source_port, None
736 for mapping in outbound.pat_mappings:
737 if (mapping.protocol == packet.protocol
738 and mapping.inside_address == packet.source
739 and mapping.inside_port == packet.source_port):
740 return outbound.address, mapping.outside_port, mapping
741 port = self.free_pat_port(outbound)
742 if port is None:
743 return None
744 mapping = PatMapping(packet.protocol, packet.source, packet.source_port, port, self.clock)
745 outbound.pat_mappings.append(mapping)
746 return outbound.address, port, mapping

748 def process(self, packet):
749 inbound = packet.interface
750 if not inbound.up:
751 return "down", "DROP down"
752 if self.martian(packet.source, packet.destination):
753 return "martian", "DROP martian"
754 if inbound.rpf != "off":
755 route = self.select_route(packet.source)
756 if route is None or (inbound.rpf == "strict" and route.interface is not inbound):
757 return "rpf", "DROP rpf"
758 connection, direction = self.find_connection(packet)
759 established = connection is not None
760 if packet.protocol == "TCP" and not established and not packet.pure_syn():
761 return "invalid", "DROP invalid"
762 verdict = inbound.acls["in"].evaluate(packet.flow(), established)
763 if verdict is not None:
764 return "acl_in", verdict
765 destination, destination_port, mapping = packet.destination, packet.destination_port, None
766 if inbound.outside:
767 destination, destination_port, mapping = self.translate_destination(inbound, packet)
768 local = self.interface_with_address(destination)
769 if local is not None:
770 return "local", "LOCAL " + local.name
771 if packet.ttl == 1:
772 return "ttl", "DROP ttl"
773 route = self.select_route(destination)
774 if route is None:
775 return "noroute", "DROP noroute"
776 outbound = route.interface
777 translated_flow = (
778 packet.protocol, packet.source, packet.source_port, destination, destination_port,
779 )
780 verdict = outbound.acls["out"].evaluate(translated_flow, established)
781 if verdict is not None:
782 return "acl_out", verdict
783 source, source_port = packet.source, packet.source_port
784 if not inbound.outside and outbound.outside:
785 translation = self.translate_source(outbound, packet)
786 if translation is None:
787 return "exhausted", "DROP exhausted"
788 source, source_port, mapping = translation
789 if mapping is not None:
790 mapping.last_used = self.clock
791 if connection is None:
792 self.connection_count += 1
793 reply = (packet.protocol, destination, destination_port, source, source_port)
794 connection = Connection(
795 self.connection_count, packet.flow(), reply, inbound, outbound
796 )
797 self.connections.append(connection)
798 direction = "new"
799 connection.record(direction, packet.flags, self.clock)
800 outbound.tx += 1
801 route.hits += 1
802 hop = "direct" if route.next_hop is None else "via " + address_text(route.next_hop)
803 parts = ["FORWARD", outbound.name, hop, f"ttl={packet.ttl - 1}"]
804 if (destination, destination_port) != (packet.destination, packet.destination_port):
805 parts.append("dnat=" + endpoint(destination, destination_port))
806 if (source, source_port) != (packet.source, packet.source_port):
807 parts.append("snat=" + endpoint(source, source_port))
808 parts.extend((f"conn={connection.number}", direction))
809 return "forwarded", " ".join(parts)

838 def show_acl(self):

905 def clear(self, arguments):
906 if arguments[0] == "COUNTERS":
907 self.clear_counters()

911 def clear_counters(self):
912 self.counts = self.empty_counts()
913 self.errors = 0
914 for interface in self.interfaces.values():
915 interface.rx = 0
916 interface.tx = 0
917 interface.drop = 0
918 for access_list in interface.acls.values():
919 access_list.clear_counters()
920 for route in self.routes:
921 route.hits = 0
922 self.emit("ok", "clear", "counters")

924 def clear_connections(self):
925 removed_connections = len(self.connections)
926 removed_mappings = 0
927 for interface in self.interfaces.values():
928 removed_mappings += len(interface.pat_mappings)
929 interface.pat_mappings = []
930 self.connections = []
931 self.emit("ok", "clear", f"conns={removed_connections}", f"nat={removed_mappings}")

933 def execute(self, line):
934 command, *arguments = line.split()
935 handlers = {
936 "IFACE": self.add_interface,
937 "UP": self.bring_up,
938 "DOWN": self.bring_down,
939 "ROUTE": self.add_route,
940 "NOROUTE": self.remove_route,
941 "ACL": self.add_rule,
942 "NOACL": self.remove_rule,
943 "NAT": self.nat,
944 "RPF": self.set_rpf,
945 "TIMEOUT": self.set_timeout,
946 "TICK": self.tick,
947 "LOOKUP": self.lookup,
948 "PACKET": self.packet,
949 "SHOW": self.show,
950 "CLEAR": self.clear,
951 }
952 try:
953 handlers[command](arguments)
954 except CommandError as error:
955 self.errors += 1
956 self.emit(f"error: {error}")
