3type Prefix = { address: number; length: number };
4type PortRange = { low: number; high: number; any: boolean };
5type Rule = { sequence: number; action: string; protocol: string; source: Prefix; sports: PortRange; destination: Prefix; dports: PortRange; established: boolean; hits: number };
6type Acl = { rules: Rule[]; denied: number };
7type Pat = { protocol: string; inside: number; inport: number; outport: number; used: number };
8type Interface = { name: string; address: number; length: number; up: boolean; outside: boolean; pool: number | null; rpf: string; rx: number; tx: number; drop: number; acl: Record<string, Acl>; pat: Pat[] };
9type Route = { type: string; address: number; length: number; output: Interface; hop: number | null; metric: number; hits: number; order: number };
10type StaticNat = { inside: number; outside: number };
11type Forward = { output: Interface; protocol: string; outport: number; inside: number; inport: number };
12type Tuple = { protocol: string; source: number; sport: number; destination: number; dport: number };
13type Connection = { number: number; original: Tuple; reply: Tuple; input: Interface; output: Interface; state: string; origPackets: number; replyPackets: number; used: number; origFin: boolean; replyFin: boolean };

15const dropReasons = ['down', 'martian', 'rpf', 'invalid', 'acl_in', 'ttl', 'noroute', 'acl_out', 'exhausted'];
16function addressText(address: number): string {
17 return [24, 16, 8, 0].map(shift => Math.floor(address / 2 ** shift) % 256).join('.');
18}
19function network(address: number, length: number): number {
20 const size = 2 ** (32 - length);
21 return Math.floor(address / size) * size;
22}
23function contains(prefix: Prefix, address: number): boolean {
24 return network(prefix.address, prefix.length) === network(address, prefix.length);
25}
26function parseAddress(text: string): number {
27 const fields = text.split('.');
28 if (fields.length !== 4 || fields.some(field => !/^(?:0|[1-9][0-9]{0,2})$/.test(field) || Number(field) > 255)) throw new Error(`bad address ${text}`);
29 return fields.reduce((value, field) => value * 256 + Number(field), 0);
30}
31function parsePrefix(text: string, checkHost = true): Prefix {
32 const fields = text.split('/');
33 let address: number;
34 try {
35 if (fields.length !== 2 || !/^(?:0|[1-9][0-9]?)$/.test(fields[1]) || Number(fields[1]) > 32) throw new Error();
36 address = parseAddress(fields[0]);
37 } catch {
38 throw new Error(`bad prefix ${text}`);
39 }
40 const length = Number(fields[1]);
41 if (checkHost && network(address, length) !== address) throw new Error(`host bits set in ${text}`);
42 return { address, length };
43}
44function parsePorts(text: string): PortRange {
45 if (text === 'ANY') return { low: 0, high: 65535, any: true };
46 const values = text.split('-').map(Number);
47 const low = values[0];
48 const high = values[values.length - 1];
49 if (high > 65535 || low > 65535 || low > high) throw new Error(`bad port range ${text}`);
50 return { low, high, any: false };
51}
52function prefixText(prefix: Prefix): string {
53 return prefix.length === 0 ? 'any' : `${addressText(prefix.address)}/${prefix.length}`;
54}
55function portsText(ports: PortRange): string {
56 return ports.any ? 'any' : ports.low === ports.high ? String(ports.low) : `${ports.low}-${ports.high}`;
57}
58function routeText(route: Route): string {
59 return `${route.type} ${addressText(route.address)}/${route.length}${route.hop === null ? '' : ` via ${addressText(route.hop)}`} dev ${route.output.name} metric ${route.metric}`;
60}
61function tupleText(tuple: Tuple): string {
62 return `${addressText(tuple.source)}:${tuple.sport} > ${addressText(tuple.destination)}:${tuple.dport}`;
63}
64function sameTuple(left: Tuple, right: Tuple): boolean {
65 return left.protocol === right.protocol && left.source === right.source && left.sport === right.sport && left.destination === right.destination && left.dport === right.dport;
66}
67function compareText(left: string, right: string): number {
68 return left === right ? 0 : left < right ? -1 : 1;
69}

71class Router {
72 interfaces = new Map<string, Interface>();
73 routes: Route[] = [];
74 staticNat: StaticNat[] = [];
75 forwards: Forward[] = [];
76 connections: Connection[] = [];
77 timeouts: Record<string, number> = { TCP: 300, UDP: 60, ICMP: 30 };
78 clock = 0;
79 packetNumber = 0;
80 connectionNumber = 0;
81 routeOrder = 0;
82 forwarded = 0;
83 local = 0;
84 errors = 0;
85 drops: Record<string, number> = Object.fromEntries(dropReasons.map(reason => [reason, 0]));
86 output: string[] = [];

88 iface(name: string): Interface {
89 const iface = this.interfaces.get(name);
90 if (iface === undefined) throw new Error(`no interface ${name}`);
91 return iface;
92 }
93 localInterface(address: number): Interface | undefined {
94 return [...this.interfaces.values()].find(iface => iface.address === address);
95 }
96 orderedInterfaces(): Interface[] {
97 return [...this.interfaces.values()].sort((a, b) => compareText(a.name, b.name));
98 }
99 select(address: number): Route | null {
100 let best: Route | null = null;
101 for (const route of this.routes) {
102 if (!route.output.up || !contains(route, address)) continue;
103 if (best === null || route.length > best.length || route.length === best.length && route.metric < best.metric) best = route;
104 }
105 return best;
106 }
107 evaluateAcl(iface: Interface, direction: string, tuple: Tuple, established: boolean): string | null {
108 const acl = iface.acl[direction];
109 if (acl.rules.length === 0) return null;
110 for (const rule of [...acl.rules].sort((a, b) => a.sequence - b.sequence)) {
111 if (rule.protocol !== 'ANY' && rule.protocol !== tuple.protocol) continue;
112 if (!contains(rule.source, tuple.source) || !contains(rule.destination, tuple.destination)) continue;
113 if ((rule.protocol === 'TCP' || rule.protocol === 'UDP') && (tuple.sport < rule.sports.low || tuple.sport > rule.sports.high || tuple.dport < rule.dports.low || tuple.dport > rule.dports.high)) continue;
114 if (rule.established && !established) continue;
115 rule.hits++;
116 if (rule.action === 'PERMIT') return null;
117 const prefix = `${iface.name} ${direction} ${rule.sequence}`;
118 return rule.action === 'DENY' ? `DROP acl ${prefix}` : `REJECT acl ${prefix} ${tuple.protocol === 'TCP' ? 'tcp-reset' : 'icmp-prohibited'}`;
119 }
120 acl.denied++;
121 return `DROP acl ${iface.name} ${direction} default`;
122 }
123 processPacket(input: Interface, tuple: Tuple, flags: string, ttl: number): string {
124 const finish = (verdict: string, reason: string): string => {
125 if (reason === 'forwarded') this.forwarded++;
126 else if (reason === 'local') this.local++;
127 else {
128 this.drops[reason]++;
129 input.drop++;
130 }
131 return verdict;
132 };
133 if (!input.up) return finish('DROP down', 'down');
134 const martian = (address: number): boolean => address < 2 ** 24 || Math.floor(address / 2 ** 24) === 127 || address >= 224 * 2 ** 24;
135 if (martian(tuple.source) || martian(tuple.destination) || this.localInterface(tuple.source) !== undefined) return finish('DROP martian', 'martian');
136 if (input.rpf !== 'off') {
137 const reverse = this.select(tuple.source);
138 if (reverse === null || input.rpf === 'strict' && reverse.output !== input) return finish('DROP rpf', 'rpf');
139 }
140 const pureSyn = tuple.protocol === 'TCP' && flags === 'S';
141 let connection: Connection | null = null;
142 let direction = 'new';
143 for (const candidate of this.connections) {
144 if (sameTuple(tuple, candidate.original)) {
145 connection = candidate;
146 direction = 'orig';
147 break;
148 }
149 if (sameTuple(tuple, candidate.reply)) {
150 connection = candidate;
151 direction = 'reply';
152 break;
153 }
154 }
155 if (connection !== null && connection.state === 'CLOSED' && pureSyn) {
156 this.connections = this.connections.filter(candidate => candidate !== connection);
157 connection = null;
158 direction = 'new';
159 }
160 if (tuple.protocol === 'TCP' && connection === null && !pureSyn) return finish('DROP invalid', 'invalid');
161 const inbound = this.evaluateAcl(input, 'in', tuple, connection !== null);
162 if (inbound !== null) return finish(inbound, 'acl_in');
163 let destination = tuple.destination;
164 let dport = tuple.dport;
165 let used: Pat | null = null;
166 if (input.outside) {
167 const mapping = this.staticNat.find(mapping => mapping.outside === destination);
168 if (mapping !== undefined) destination = mapping.inside;
169 else if (destination === input.address) {
170 const forward = this.forwards.find(rule => rule.output === input && rule.protocol === tuple.protocol && rule.outport === dport);
171 if (forward !== undefined) {
172 destination = forward.inside;
173 dport = forward.inport;
174 } else {
175 used = input.pat.find(mapping => mapping.protocol === tuple.protocol && mapping.outport === dport) ?? null;
176 if (used !== null) {
177 destination = used.inside;
178 dport = used.inport;
179 }
180 }
181 }
182 }
183 const local = this.localInterface(destination);
184 if (local !== undefined) return finish(`LOCAL ${local.name}`, 'local');
185 if (ttl === 1) return finish('DROP ttl', 'ttl');
186 const route = this.select(destination);
187 if (route === null) return finish('DROP noroute', 'noroute');
188 const output = route.output;
189 const outbound = this.evaluateAcl(output, 'out', { ...tuple, destination, dport }, connection !== null);
190 if (outbound !== null) return finish(outbound, 'acl_out');
191 let source = tuple.source;
192 let sport = tuple.sport;
193 if (!input.outside && output.outside) {
194 const mapping = this.staticNat.find(mapping => mapping.inside === source);
195 if (mapping !== undefined) source = mapping.outside;
196 else {
197 const forward = this.forwards.find(rule => rule.output === output && rule.protocol === tuple.protocol && rule.inside === source && rule.inport === sport);
198 if (forward !== undefined) {
199 source = output.address;
200 sport = forward.outport;
201 } else if (output.pool !== null) {
202 used = output.pat.find(mapping => mapping.protocol === tuple.protocol && mapping.inside === source && mapping.inport === sport) ?? null;
203 if (used === null) {
204 let port = 20000;
205 while (port < 20000 + output.pool && (output.pat.some(mapping => mapping.outport === port) || this.forwards.some(rule => rule.output === output && rule.outport === port))) port++;
206 if (port === 20000 + output.pool) return finish('DROP exhausted', 'exhausted');
207 used = { protocol: tuple.protocol, inside: source, inport: sport, outport: port, used: this.clock };
208 output.pat.push(used);
209 }
210 source = output.address;
211 sport = used.outport;
212 }
213 }
214 }
215 if (used !== null) used.used = this.clock;
216 if (connection === null) {
217 connection = {
218 number: ++this.connectionNumber, original: tuple,
219 reply: { protocol: tuple.protocol, source: destination, sport: dport, destination: source, dport: sport },
220 input, output, state: tuple.protocol === 'TCP' ? 'SYN_SENT' : 'UNREPLIED',
221 origPackets: 0, replyPackets: 0, used: this.clock, origFin: false, replyFin: false,
222 };
223 this.connections.push(connection);
224 direction = 'new';
225 }
226 if (direction === 'reply') connection.replyPackets++;
227 else connection.origPackets++;
228 connection.used = this.clock;
229 if (tuple.protocol !== 'TCP') {
230 if (direction === 'reply') connection.state = 'REPLIED';
231 } else if (flags.includes('R')) connection.state = 'CLOSED';
232 else {
233 if (connection.state === 'SYN_SENT' && direction === 'reply' && flags.includes('S') && flags.includes('A')) connection.state = 'ESTABLISHED';
234 if (flags.includes('F') && (connection.state === 'ESTABLISHED' || connection.state === 'FIN_WAIT')) {
235 if (direction === 'reply') connection.replyFin = true;
236 else connection.origFin = true;
237 connection.state = connection.origFin && connection.replyFin ? 'CLOSED' : 'FIN_WAIT';
238 }
239 }
240 output.tx++;
241 route.hits++;
242 let verdict = `FORWARD ${output.name} ${route.hop === null ? 'direct' : `via ${addressText(route.hop)}`} ttl=${ttl - 1}`;
243 if (destination !== tuple.destination || dport !== tuple.dport) verdict += ` dnat=${addressText(destination)}:${dport}`;
244 if (source !== tuple.source || sport !== tuple.sport) verdict += ` snat=${addressText(source)}:${sport}`;
245 return finish(`${verdict} conn=${connection.number} ${direction}`, 'forwarded');
246 }

248 command(words: string[]): void {
249 const command = words[0];
250 if (command === 'IFACE') {
251 const [name, text] = words.slice(1);
252 if (this.interfaces.has(name)) throw new Error(`duplicate interface ${name}`);
253 const prefix = parsePrefix(text, false);
254 if (prefix.length < 8 || prefix.length > 30) throw new Error(`bad prefix ${text}`);
255 const net = network(prefix.address, prefix.length);
256 if (prefix.address === net || prefix.address === net + 2 ** (32 - prefix.length) - 1) throw new Error(`bad host address ${text}`);
257 for (const iface of this.interfaces.values()) {
258 const length = Math.min(prefix.length, iface.length);
259 if (network(prefix.address, length) === network(iface.address, length)) throw new Error(`subnet overlaps ${iface.name}`);
260 }
261 const iface: Interface = { name, ...prefix, up: true, outside: false, pool: null, rpf: 'off', rx: 0, tx: 0, drop: 0, acl: { in: { rules: [], denied: 0 }, out: { rules: [], denied: 0 } }, pat: [] };
262 this.interfaces.set(name, iface);
263 this.routes.push({ type: 'C', address: net, length: prefix.length, output: iface, hop: null, metric: 0, hits: 0, order: this.routeOrder++ });
264 this.output.push(`ok iface ${name} ${text} network ${addressText(net)}/${prefix.length}`);
265 } else if (command === 'UP' || command === 'DOWN') {
266 const iface = this.iface(words[1]);
267 const up = command === 'UP';
268 if (iface.up === up) throw new Error(`interface ${iface.name} already ${up ? 'up' : 'down'}`);
269 iface.up = up;
270 this.output.push(`ok ${command.toLowerCase()} ${iface.name}`);
271 } else if (command === 'ROUTE') {
272 const prefix = parsePrefix(words[1]);
273 let output: Interface | undefined;
274 let hop: number | null = null;
275 if (words[2] === 'VIA') hop = parseAddress(words[3]);
276 else output = this.iface(words[3]);
277 const metric = Number(words[4]);
278 if (metric < 1 || metric > 255) throw new Error(`bad metric ${words[4]}`);
279 if (hop !== null) {
280 if (this.localInterface(hop) !== undefined) throw new Error(`nexthop ${words[3]} is local`);
281 output = [...this.interfaces.values()].find(iface => contains(iface, hop!));
282 if (output === undefined) throw new Error(`nexthop ${words[3]} unreachable`);
283 }
284 if (this.routes.some(route => route.type === 'S' && route.address === prefix.address && route.length === prefix.length && route.output === output && route.hop === hop)) throw new Error(`duplicate route ${words[1]}`);
285 const route: Route = { type: 'S', ...prefix, output: output!, hop, metric, hits: 0, order: this.routeOrder++ };
286 this.routes.push(route);
287 this.output.push(`ok route ${routeText(route)}`);
288 } else if (command === 'NOROUTE') {
289 const prefix = parsePrefix(words[1]);
290 const matches = (route: Route): boolean => route.type === 'S' && route.address === prefix.address && route.length === prefix.length;
291 const removed = this.routes.filter(matches).length;
292 if (removed === 0) throw new Error(`no static route ${words[1]}`);
293 this.routes = this.routes.filter(route => !matches(route));
294 this.output.push(`ok noroute ${words[1]} removed=${removed}`);
295 } else if (command === 'LOOKUP') {
296 const route = this.select(parseAddress(words[1]));
297 this.output.push(`lookup ${words[1]} ${route === null ? 'none' : routeText(route)}`);
298 } else if (command === 'ACL') {
299 const iface = this.iface(words[1]);
300 const direction = words[2].toLowerCase();
301 const sequence = Number(words[3]);
302 if (sequence < 1 || sequence > 9999) throw new Error(`bad sequence ${words[3]}`);
303 const source = words[6] === 'ANY' ? { address: 0, length: 0 } : parsePrefix(words[6]);
304 const sports = parsePorts(words[7]);
305 const destination = words[8] === 'ANY' ? { address: 0, length: 0 } : parsePrefix(words[8]);
306 const dports = parsePorts(words[9]);
307 if ((words[5] === 'ICMP' || words[5] === 'ANY') && (!sports.any || !dports.any)) throw new Error(`ports not allowed for ${words[5]}`);
308 const acl = iface.acl[direction];
309 if (acl.rules.some(rule => rule.sequence === sequence)) throw new Error(`duplicate sequence ${words[3]}`);
310 acl.rules.push({ sequence, action: words[4], protocol: words[5], source, sports, destination, dports, established: words.length === 11, hits: 0 });
311 this.output.push(`ok acl ${iface.name} ${direction} ${words[3]}`);
312 } else if (command === 'NOACL') {
313 const iface = this.iface(words[1]);
314 const direction = words[2].toLowerCase();
315 const acl = iface.acl[direction];
316 const rule = acl.rules.find(rule => rule.sequence === Number(words[3]));
317 if (rule === undefined) throw new Error(`no rule ${words[3]} on ${iface.name} ${direction}`);
318 acl.rules = acl.rules.filter(candidate => candidate !== rule);
319 this.output.push(`ok noacl ${iface.name} ${direction} ${words[3]} hits=${rule.hits}`);
320 } else if (command === 'NAT') this.natCommand(words);
321 else if (command === 'RPF') {
322 const iface = this.iface(words[1]);
323 iface.rpf = words[2].toLowerCase();
324 this.output.push(`ok rpf ${iface.name} ${iface.rpf}`);
325 } else if (command === 'TIMEOUT') {
326 const seconds = Number(words[2]);
327 if (seconds < 1 || seconds > 86400) throw new Error(`bad timeout ${words[2]}`);
328 this.timeouts[words[1]] = seconds;
329 this.output.push(`ok timeout ${words[1]} ${words[2]}`);
330 } else if (command === 'TICK') {
331 const seconds = Number(words[1]);
332 if (seconds < 1 || seconds > 86400) throw new Error(`bad tick ${words[1]}`);
333 this.clock += seconds;
334 const before = this.connections.length;
335 this.connections = this.connections.filter(connection => {
336 const protocol = connection.original.protocol;
337 const timeout = protocol !== 'TCP' || connection.state === 'ESTABLISHED' ? this.timeouts[protocol] : connection.state === 'CLOSED' ? 10 : 60;
338 return this.clock - connection.used < timeout;
339 });
340 let removed = 0;
341 for (const iface of this.interfaces.values()) {
342 const count = iface.pat.length;
343 iface.pat = iface.pat.filter(mapping => this.clock - mapping.used < this.timeouts[mapping.protocol]);
344 removed += count - iface.pat.length;
345 }
346 this.output.push(`ok tick time=${this.clock} expired conns=${before - this.connections.length} nat=${removed}`);
347 } else if (command === 'PACKET') {
348 const iface = this.iface(words[1]);
349 const protocol = words[2];
350 const flags = words[3];
351 if (protocol !== 'TCP' && flags !== '-') throw new Error(`flags not allowed for ${protocol}`);
352 if (protocol === 'TCP' && flags !== '-' && (!/^[SAFR]+$/.test(flags) || new Set(flags).size !== flags.length)) throw new Error(`bad flags ${flags}`);
353 const source = parseAddress(words[4]);
354 const sport = Number(words[5]);
355 if (sport > 65535) throw new Error(`bad port ${words[5]}`);
356 const destination = parseAddress(words[6]);
357 const dport = Number(words[7]);
358 if (dport > 65535) throw new Error(`bad port ${words[7]}`);
359 const ttl = Number(words[8]);
360 if (ttl < 1 || ttl > 255) throw new Error(`bad ttl ${words[8]}`);
361 this.packetNumber++;
362 iface.rx++;
363 const verdict = this.processPacket(iface, { protocol, source, sport, destination, dport }, flags, ttl);
364 this.output.push(`#${this.packetNumber} ${protocol} ${flags} ${words[4]}:${words[5]} > ${words[6]}:${words[7]} ${verdict}`);
365 } else if (command === 'SHOW') this.show(words[1]);
366 else if (command === 'CLEAR') {
367 if (words[1] === 'CONNS') {
368 const mappings = [...this.interfaces.values()].reduce((total, iface) => total + iface.pat.length, 0);
369 this.output.push(`ok clear conns=${this.connections.length} nat=${mappings}`);
370 this.connections = [];
371 for (const iface of this.interfaces.values()) iface.pat = [];
372 } else {
373 this.forwarded = this.local = this.errors = 0;
374 for (const reason of dropReasons) this.drops[reason] = 0;
375 for (const route of this.routes) route.hits = 0;
376 for (const iface of this.interfaces.values()) {
377 iface.rx = iface.tx = iface.drop = 0;
378 for (const acl of Object.values(iface.acl)) {
379 acl.denied = 0;
380 for (const rule of acl.rules) rule.hits = 0;
381 }
382 }
383 this.output.push('ok clear counters');
384 }
385 }
386 }

388 natCommand(words: string[]): void {
389 if (words[1] === 'STATIC') {
390 const inside = parseAddress(words[2]);
391 const outside = parseAddress(words[3]);
392 if (this.localInterface(inside) !== undefined) throw new Error(`address ${words[2]} is local`);
393 if (this.localInterface(outside) !== undefined) throw new Error(`address ${words[3]} is local`);
394 if (this.staticNat.some(mapping => mapping.inside === inside)) throw new Error(`duplicate static ${words[2]}`);
395 if (this.staticNat.some(mapping => mapping.outside === outside)) throw new Error(`duplicate static ${words[3]}`);
396 this.staticNat.push({ inside, outside });
397 this.output.push(`ok nat static ${words[2]} ${words[3]}`);
398 return;
399 }
400 const iface = this.iface(words[2]);
401 if (words[1] === 'OUTSIDE') {
402 if (iface.outside) throw new Error(`interface ${iface.name} already outside`);
403 iface.outside = true;
404 this.output.push(`ok nat outside ${iface.name}`);
405 } else if (words[1] === 'PAT') {
406 const count = Number(words[3]);
407 if (count < 1 || count > 1000) throw new Error(`bad pool size ${words[3]}`);
408 if (iface.pool !== null) throw new Error(`pat already on ${iface.name}`);
409 iface.pool = count;
410 iface.outside = true;
411 this.output.push(`ok nat pat ${iface.name} ${addressText(iface.address)} 20000-${20000 + count - 1}`);
412 } else {
413 const protocol = words[3];
414 const outport = Number(words[4]);
415 if (outport < 1 || outport > 65535) throw new Error(`bad port ${words[4]}`);
416 const inside = parseAddress(words[5]);
417 const inport = Number(words[6]);
418 if (inport < 1 || inport > 65535) throw new Error(`bad port ${words[6]}`);
419 if (!iface.outside) throw new Error(`interface ${iface.name} not outside`);
420 if (this.localInterface(inside) !== undefined) throw new Error(`address ${words[5]} is local`);
421 if (this.forwards.some(rule => rule.output === iface && rule.protocol === protocol && rule.outport === outport)) throw new Error(`duplicate forward ${iface.name} ${protocol} ${words[4]}`);
422 this.forwards.push({ output: iface, protocol, outport, inside, inport });
423 this.output.push(`ok nat forward ${iface.name} ${protocol} ${words[4]} ${words[5]}:${words[6]}`);
424 }
425 }

427 show(what: string): void {
428 const interfaces = this.orderedInterfaces();
429 if (what === 'IFACES') {
430 this.output.push(`show ifaces count=${interfaces.length}`);
431 for (const iface of interfaces) this.output.push(`${iface.name} ${addressText(iface.address)}/${iface.length} ${iface.up ? 'up' : 'down'} ${iface.outside ? 'outside' : 'inside'} rpf=${iface.rpf}${iface.pool === null ? '' : ` pat=${iface.pat.length}/${iface.pool}`}`);
432 } else if (what === 'ROUTES') {
433 this.output.push(`show routes count=${this.routes.length}`);
434 for (const route of [...this.routes].sort((a, b) => a.address - b.address || a.length - b.length || a.metric - b.metric || a.order - b.order)) this.output.push(`${routeText(route)} hits=${route.hits}${route.output.up ? '' : ' inactive'}`);
435 } else if (what === 'ACL') {
436 this.output.push(`show acl count=${interfaces.reduce((total, iface) => total + iface.acl.in.rules.length + iface.acl.out.rules.length, 0)}`);
437 for (const iface of interfaces) {
438 for (const direction of ['in', 'out']) {
439 const acl = iface.acl[direction];
440 if (acl.rules.length === 0) continue;
441 for (const rule of [...acl.rules].sort((a, b) => a.sequence - b.sequence)) this.output.push(`acl ${iface.name} ${direction} ${rule.sequence} ${rule.action} ${rule.protocol} ${prefixText(rule.source)} ${portsText(rule.sports)} ${prefixText(rule.destination)} ${portsText(rule.dports)}${rule.established ? ' established' : ''} hits=${rule.hits}`);
442 this.output.push(`acl ${iface.name} ${direction} default DENY hits=${acl.denied}`);
443 }
444 }
445 } else if (what === 'NAT') {
446 const count = interfaces.reduce((total, iface) => total + iface.pat.length, 0);
447 this.output.push(`show nat static=${this.staticNat.length} forward=${this.forwards.length} pat=${count}`);
448 for (const mapping of [...this.staticNat].sort((a, b) => a.inside - b.inside)) this.output.push(`static ${addressText(mapping.inside)} ${addressText(mapping.outside)}`);
449 for (const rule of [...this.forwards].sort((a, b) => compareText(a.output.name, b.output.name) || a.outport - b.outport || compareText(a.protocol, b.protocol))) this.output.push(`forward ${rule.output.name} ${rule.protocol} ${rule.outport} ${addressText(rule.inside)}:${rule.inport}`);
450 for (const iface of interfaces) for (const mapping of [...iface.pat].sort((a, b) => a.outport - b.outport)) this.output.push(`pat ${iface.name} ${mapping.outport} ${mapping.protocol} ${addressText(mapping.inside)}:${mapping.inport} idle=${this.clock - mapping.used}`);
451 } else if (what === 'CONNS') {
452 this.output.push(`show conns count=${this.connections.length}`);
453 for (const connection of this.connections) this.output.push(`conn ${connection.number} ${connection.original.protocol} ${tupleText(connection.original)} reply ${tupleText(connection.reply)} in=${connection.input.name} out=${connection.output.name} state=${connection.state} pkts=${connection.origPackets}/${connection.replyPackets} idle=${this.clock - connection.used}`);
454 } else {
455 const dropped = dropReasons.reduce((total, reason) => total + this.drops[reason], 0);
456 this.output.push(`show counters packets=${this.forwarded + this.local + dropped} forwarded=${this.forwarded} local=${this.local} dropped=${dropped} errors=${this.errors}`);
457 this.output.push('drops ' + dropReasons.map(reason => `${reason}=${this.drops[reason]}`).join(' '));
458 for (const iface of interfaces) this.output.push(`iface ${iface.name} rx=${iface.rx} tx=${iface.tx} drop=${iface.drop}`);
459 }
460 }
461
462 run(lines: string[]): void {
463 const count = Number(lines[0]);
464 for (const line of lines.slice(1, count + 1)) {
465 try {
466 this.command(line.split(' '));
467 } catch (error) {
468 this.errors++;
469 this.output.push(`error: ${(error as Error).message}`);
470 }
471 }
472 }
473}
