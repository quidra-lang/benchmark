1class RouterError < StandardError
2end
3
4class Router
5 DROP_REASONS = %w[down martian rpf invalid acl_in ttl noroute acl_out exhausted].freeze
6
7 def initialize
8 @interfaces = {}
9 @routes = []
10 @static = []
11 @forwards = []
12 @pat = []
13 @connections = []
14 @timeouts = { 'TCP' => 300, 'UDP' => 60, 'ICMP' => 30 }
15 @clock = @packet_number = @connection_number = 0
16 reset_counters
17 end
18
19 def error(message)
20 raise RouterError, message
21 end
22
23 def address(word)
24 error("bad address #{word}") unless word.match?(/\A(?:0|[1-9][0-9]{0,2})(?:\.(?:0|[1-9][0-9]{0,2})){3}\z/) && word.split('.').all? { |part| part.to_i <= 255 }
25 word.split('.').reduce(0) { |value, part| value * 256 + part.to_i }
26 end
27
28 def ip(value)
29 [24, 16, 8, 0].map { |shift| (value >> shift) & 255 }.join('.')
30 end
31
32 def network(value, length)
33 size = 2**(32 - length)
34 value / size * size
35 end
36
37 def prefix(word, hosts = false)
38 pieces = word.split('/', -1)
39 begin
40 valid = pieces.length == 2 && pieces[1].match?(/\A(?:0|[1-9][0-9]?)\z/) && pieces[1].to_i <= 32
41 error('invalid') unless valid
42 value = address(pieces[0])
43 rescue RouterError
44 error("bad prefix #{word}")
45 end
46 length = pieces[1].to_i
47 error("host bits set in #{word}") if !hosts && value != network(value, length)
48 [value, length]
49 end
50
51 def contains?(prefix, value)
52 network(prefix[0], prefix[1]) == network(value, prefix[1])
53 end
54
55 def iface(name)
56 error("no interface #{name}") unless @interfaces.key?(name)
57 @interfaces[name]
58 end
59
60 def local(value)
61 @interfaces.find { |_name, item| item[:address] == value }&.first
62 end
63
64 def route(value)
65 @routes.each_with_index.select { |item, _| @interfaces[item[:dev]][:up] && contains?(item[:prefix], value) }
66 .min_by { |item, index| [-item[:prefix][1], item[:metric], index] }&.first
67 end
68
69 def route_text(item)
70 text = "#{item[:type]} #{ip(item[:prefix][0])}/#{item[:prefix][1]}"
71 text += " via #{ip(item[:hop])}" if item[:hop]
72 text + " dev #{item[:dev]} metric #{item[:metric]}"
73 end
74
75 def port_range(word)
76 return [0, 65535, true] if word == 'ANY'
77 numbers = word.split('-').map(&:to_i)
78 low = numbers[0]
79 high = numbers.fetch(1, low)
80 error("bad port range #{word}") if high > 65535 || low > 65535 || low > high
81 [low, high, false]
82 end
83
84 def range_text(range)
85 return 'any' if range[2]
86 range[0] == range[1] ? range[0].to_s : "#{range[0]}-#{range[1]}"
87 end
88
89 def acl_prefix(word)
90 word == 'ANY' ? [0, 0] : prefix(word)
91 end
92
93 def prefix_text(value)
94 value[1] == 0 ? 'any' : "#{ip(value[0])}/#{value[1]}"
95 end

97 def acl(name, direction, protocol, source, sport, destination, dport, established)
98 pair = @interfaces[name][:acl][direction]
99 return nil if pair[:rules].empty?
100 rule = pair[:rules].sort_by { |item| item[:seq] }.find do |item|
101 (item[:proto] == 'ANY' || item[:proto] == protocol) && contains?(item[:src], source) && contains?(item[:dst], destination) &&
102 (!%w[TCP UDP].include?(item[:proto]) || (sport.between?(item[:sports][0], item[:sports][1]) && dport.between?(item[:dports][0], item[:dports][1]))) &&
103 (!item[:established] || established)
104 end
105 unless rule
106 pair[:default] += 1
107 return "DROP acl #{name} #{direction} default"
108 end
109 rule[:hits] += 1
110 case rule[:action]
111 when 'PERMIT' then nil
112 when 'DENY' then "DROP acl #{name} #{direction} #{rule[:seq]}"
113 else "REJECT acl #{name} #{direction} #{rule[:seq]} #{protocol == 'TCP' ? 'tcp-reset' : 'icmp-prohibited'}"
114 end
115 end

117 def packet(name, protocol, flags, source, sport, destination, dport, ttl)
118 input = @interfaces[name]
119 return ['DROP down', 'down'] unless input[:up]
120 martian = [source, destination].any? { |value| contains?([0, 8], value) || contains?([127 * 2**24, 8], value) || contains?([224 * 2**24, 3], value) }
121 return ['DROP martian', 'martian'] if martian || local(source)
122 if input[:rpf] != 'off'
123 reverse = route(source)
124 return ['DROP rpf', 'rpf'] unless reverse && (input[:rpf] != 'strict' || reverse[:dev] == name)
125 end
126 tuple = [protocol, source, sport, destination, dport]
127 pure_syn = protocol == 'TCP' && flags == 'S'
128 connection = nil
129 direction = nil
130 @connections.each do |item|
131 if item[:orig] == tuple || item[:reply] == tuple
132 connection = item
133 direction = item[:orig] == tuple ? 'orig' : 'reply'
134 break
135 end
136 end
137 if connection && connection[:state] == 'CLOSED' && pure_syn
138 @connections.delete(connection)
139 connection = nil
140 end
141 return ['DROP invalid', 'invalid'] if protocol == 'TCP' && !connection && !pure_syn
142 verdict = acl(name, 'in', protocol, source, sport, destination, dport, !connection.nil?)
143 return [verdict, 'acl_in'] if verdict
144 translated_destination = destination
145 translated_dport = dport
146 mapping = nil
147 if input[:role] == 'outside'
148 static = @static.find { |item| item[:outside] == destination }
149 forward = @forwards.find { |item| item[:dev] == name && item[:proto] == protocol && item[:port] == dport } if destination == input[:address]
150 if static
151 translated_destination = static[:inside]
152 elsif forward
153 translated_destination = forward[:inside]
154 translated_dport = forward[:inport]
155 elsif destination == input[:address]
156 mapping = @pat.find { |item| item[:dev] == name && item[:proto] == protocol && item[:port] == dport }
157 if mapping
158 translated_destination = mapping[:inside]
159 translated_dport = mapping[:inport]
160 end
161 end
162 end
163 local_name = local(translated_destination)
164 return ["LOCAL #{local_name}", 'local'] if local_name
165 return ['DROP ttl', 'ttl'] if ttl == 1
166 selected = route(translated_destination)
167 return ['DROP noroute', 'noroute'] unless selected
168 output_name = selected[:dev]
169 output = @interfaces[output_name]
170 verdict = acl(output_name, 'out', protocol, source, sport, translated_destination, translated_dport, !connection.nil?)
171 return [verdict, 'acl_out'] if verdict
172 translated_source = source
173 translated_sport = sport
174 if input[:role] == 'inside' && output[:role] == 'outside'
175 static = @static.find { |item| item[:inside] == source }
176 forward = @forwards.find { |item| item[:dev] == output_name && item[:proto] == protocol && item[:inside] == source && item[:inport] == sport }
177 if static
178 translated_source = static[:outside]
179 elsif forward
180 translated_source = output[:address]
181 translated_sport = forward[:port]
182 elsif output[:pool]
183 mapping = @pat.find { |item| item[:dev] == output_name && item[:proto] == protocol && item[:inside] == source && item[:inport] == sport }
184 unless mapping
185 used = @pat.select { |item| item[:dev] == output_name }.map { |item| item[:port] } + @forwards.select { |item| item[:dev] == output_name }.map { |item| item[:port] }
186 port = (20000...(20000 + output[:pool])).find { |candidate| !used.include?(candidate) }
187 return ['DROP exhausted', 'exhausted'] unless port
188 mapping = { dev: output_name, proto: protocol, inside: source, inport: sport, port: port, last: @clock }
189 @pat << mapping
190 end
191 translated_source = output[:address]
192 translated_sport = mapping[:port]
193 end
194 end
195 mapping[:last] = @clock if mapping
196 unless connection
197 @connection_number += 1
198 connection = { number: @connection_number, proto: protocol, orig: tuple,
199 reply: [protocol, translated_destination, translated_dport, translated_source, translated_sport],
200 input: name, output: output_name, state: protocol == 'TCP' ? 'SYN_SENT' : 'UNREPLIED',
201 orig_packets: 0, reply_packets: 0, orig_fin: false, reply_fin: false, last: @clock }
202 @connections << connection
203 direction = 'new'
204 end
205 side = direction == 'reply' ? :reply : :orig
206 connection[side == :reply ? :reply_packets : :orig_packets] += 1
207 connection[:last] = @clock
208 if protocol != 'TCP'
209 connection[:state] = 'REPLIED' if direction == 'reply'
210 elsif flags.include?('R')
211 connection[:state] = 'CLOSED'
212 else
213 if connection[:state] == 'SYN_SENT' && direction == 'reply' && flags.include?('S') && flags.include?('A')
214 connection[:state] = 'ESTABLISHED'
215 end
216 if flags.include?('F') && %w[ESTABLISHED FIN_WAIT].include?(connection[:state])
217 connection[side == :reply ? :reply_fin : :orig_fin] = true
218 connection[:state] = connection[:orig_fin] && connection[:reply_fin] ? 'CLOSED' : 'FIN_WAIT'
219 end
220 end
221 output[:tx] += 1
222 selected[:hits] += 1
223 next_hop = selected[:hop] ? "via #{ip(selected[:hop])}" : 'direct'
224 text = "FORWARD #{output_name} #{next_hop} ttl=#{ttl - 1}"
225 text += " dnat=#{ip(translated_destination)}:#{translated_dport}" if [translated_destination, translated_dport] != [destination, dport]
226 text += " snat=#{ip(translated_source)}:#{translated_sport}" if [translated_source, translated_sport] != [source, sport]
227 [text + " conn=#{connection[:number]} #{direction}", 'forwarded']
228 end

230 def reset_counters
231 @counts = { 'forwarded' => 0, 'local' => 0, 'errors' => 0 }
232 DROP_REASONS.each { |reason| @counts[reason] = 0 }
233 @interfaces.each_value do |item|
234 item[:rx] = item[:tx] = item[:drop] = 0
235 item[:acl].each_value do |pair|
236 pair[:default] = 0
237 pair[:rules].each { |rule| rule[:hits] = 0 }
238 end
239 end
240 @routes.each { |item| item[:hits] = 0 }
241 end

243 def show(what)
244 names = @interfaces.keys.sort
245 case what
246 when 'IFACES'
247 puts "show ifaces count=#{names.length}"
248 names.each do |name|
249 item = @interfaces[name]
250 text = "#{name} #{ip(item[:address])}/#{item[:length]} #{item[:up] ? 'up' : 'down'} #{item[:role]} rpf=#{item[:rpf]}"
251 text += " pat=#{@pat.count { |mapping| mapping[:dev] == name }}/#{item[:pool]}" if item[:pool]
252 puts text
253 end
254 when 'ROUTES'
255 puts "show routes count=#{@routes.length}"
256 @routes.each_with_index.sort_by { |item, index| [item[:prefix][0], item[:prefix][1], item[:metric], index] }.each do |item, _|
257 puts route_text(item) + " hits=#{item[:hits]}" + (@interfaces[item[:dev]][:up] ? '' : ' inactive')
258 end
259 when 'ACL'
260 puts "show acl count=#{@interfaces.values.sum { |item| item[:acl].values.sum { |pair| pair[:rules].length } }}"
261 names.each do |name|
262 %w[in out].each do |direction|
263 pair = @interfaces[name][:acl][direction]
264 next if pair[:rules].empty?
265 pair[:rules].sort_by { |rule| rule[:seq] }.each do |rule|
266 puts "acl #{name} #{direction} #{rule[:seq]} #{rule[:action]} #{rule[:proto]} #{prefix_text(rule[:src])} #{range_text(rule[:sports])} #{prefix_text(rule[:dst])} #{range_text(rule[:dports])}" + (rule[:established] ? ' established' : '') + " hits=#{rule[:hits]}"
267 end
268 puts "acl #{name} #{direction} default DENY hits=#{pair[:default]}"
269 end
270 end
271 when 'NAT'
272 puts "show nat static=#{@static.length} forward=#{@forwards.length} pat=#{@pat.length}"
273 @static.sort_by { |item| item[:inside] }.each { |item| puts "static #{ip(item[:inside])} #{ip(item[:outside])}" }
274 @forwards.sort_by { |item| [item[:dev], item[:port], item[:proto]] }.each { |item| puts "forward #{item[:dev]} #{item[:proto]} #{item[:port]} #{ip(item[:inside])}:#{item[:inport]}" }
275 @pat.sort_by { |item| [item[:dev], item[:port]] }.each { |item| puts "pat #{item[:dev]} #{item[:port]} #{item[:proto]} #{ip(item[:inside])}:#{item[:inport]} idle=#{@clock - item[:last]}" }
276 when 'CONNS'
277 puts "show conns count=#{@connections.length}"
278 @connections.each do |item|
279 _, source, sport, destination, dport = item[:orig]
280 _, reply_source, reply_sport, reply_destination, reply_dport = item[:reply]
281 puts "conn #{item[:number]} #{item[:proto]} #{ip(source)}:#{sport} > #{ip(destination)}:#{dport} reply #{ip(reply_source)}:#{reply_sport} > #{ip(reply_destination)}:#{reply_dport} in=#{item[:input]} out=#{item[:output]} state=#{item[:state]} pkts=#{item[:orig_packets]}/#{item[:reply_packets]} idle=#{@clock - item[:last]}"
282 end
283 when 'COUNTERS'
284 dropped = DROP_REASONS.sum { |reason| @counts[reason] }
285 puts "show counters packets=#{@counts['forwarded'] + @counts['local'] + dropped} forwarded=#{@counts['forwarded']} local=#{@counts['local']} dropped=#{dropped} errors=#{@counts['errors']}"
286 puts 'drops ' + DROP_REASONS.map { |reason| "#{reason}=#{@counts[reason]}" }.join(' ')
287 names.each do |name|
288 item = @interfaces[name]
289 puts "iface #{name} rx=#{item[:rx]} tx=#{item[:tx]} drop=#{item[:drop]}"
290 end
291 end
292 end

294 def command(words)
295 kind, *args = words
296 name = args[0]
297 case kind
298 when 'IFACE'
299 error("duplicate interface #{name}") if @interfaces.key?(name)
300 value, length = prefix(args[1], true)
301 error("bad prefix #{args[1]}") unless length.between?(8, 30)
302 base = network(value, length)
303 error("bad host address #{args[1]}") if value == base || value == base + 2**(32 - length) - 1
304 overlap = @interfaces.find { |_key, item| network(value, [length, item[:length]].min) == network(item[:address], [length, item[:length]].min) }
305 error("subnet overlaps #{overlap[0]}") if overlap
306 @interfaces[name] = { address: value, length: length, up: true, role: 'inside', pool: nil, rpf: 'off', rx: 0, tx: 0, drop: 0,
307 acl: { 'in' => { rules: [], default: 0 }, 'out' => { rules: [], default: 0 } } }
308 @routes << { type: 'C', prefix: [base, length], dev: name, hop: nil, metric: 0, hits: 0 }
309 puts "ok iface #{name} #{args[1]} network #{ip(base)}/#{length}"
310 when 'UP', 'DOWN'
311 item = iface(name)
312 state = kind == 'UP'
313 error("interface #{name} already #{state ? 'up' : 'down'}") if item[:up] == state
314 item[:up] = state
315 puts "ok #{kind.downcase} #{name}"
316 when 'ROUTE'
317 destination = prefix(name)
318 via = args[1] == 'VIA'
319 hop = via ? address(args[2]) : nil
320 iface(args[2]) unless via
321 metric = args[3].to_i
322 error("bad metric #{args[3]}") unless metric.between?(1, 255)
323 if via
324 error("nexthop #{args[2]} is local") if local(hop)
325 output = @interfaces.find { |_key, item| contains?([item[:address], item[:length]], hop) }&.first
326 error("nexthop #{args[2]} unreachable") unless output
327 else
328 output = args[2]
329 end
330 error("duplicate route #{name}") if @routes.any? { |item| item[:type] == 'S' && item[:prefix] == destination && item[:dev] == output && item[:hop] == hop }
331 item = { type: 'S', prefix: destination, dev: output, hop: hop, metric: metric, hits: 0 }
332 @routes << item
333 puts "ok route #{route_text(item)}"
334 when 'NOROUTE'
335 destination = prefix(name)
336 removed = @routes.count { |item| item[:type] == 'S' && item[:prefix] == destination }
337 error("no static route #{name}") if removed == 0
338 @routes.reject! { |item| item[:type] == 'S' && item[:prefix] == destination }
339 puts "ok noroute #{name} removed=#{removed}"
340 when 'LOOKUP'
341 selected = route(address(name))
342 puts "lookup #{name} #{selected ? route_text(selected) : 'none'}"
343 when 'ACL'
344 item = iface(name)
345 direction = args[1].downcase
346 sequence = args[2].to_i
347 error("bad sequence #{args[2]}") unless sequence.between?(1, 9999)
348 source = acl_prefix(args[5])
349 sports = port_range(args[6])
350 destination = acl_prefix(args[7])
351 dports = port_range(args[8])
352 error("ports not allowed for #{args[4]}") if %w[ICMP ANY].include?(args[4]) && (args[6] != 'ANY' || args[8] != 'ANY')
353 pair = item[:acl][direction]
354 error("duplicate sequence #{args[2]}") if pair[:rules].any? { |rule| rule[:seq] == sequence }
355 pair[:rules] << { seq: sequence, action: args[3], proto: args[4], src: source, sports: sports, dst: destination, dports: dports, established: args.length == 10, hits: 0 }
356 puts "ok acl #{name} #{direction} #{sequence}"
357 when 'NOACL'
358 pair = iface(name)[:acl][args[1].downcase]
359 rule = pair[:rules].find { |item| item[:seq] == args[2].to_i }
360 error("no rule #{args[2]} on #{name} #{args[1].downcase}") unless rule
361 pair[:rules].delete(rule)
362 puts "ok noacl #{name} #{args[1].downcase} #{args[2]} hits=#{rule[:hits]}"
363 when 'NAT'
364 case name
365 when 'STATIC'
366 inside = address(args[1])
367 outside = address(args[2])
368 [1, 2].each { |index| error("address #{args[index]} is local") if local(index == 1 ? inside : outside) }
369 error("duplicate static #{args[1]}") if @static.any? { |item| item[:inside] == inside }
370 error("duplicate static #{args[2]}") if @static.any? { |item| item[:outside] == outside }
371 @static << { inside: inside, outside: outside }
372 puts "ok nat static #{args[1]} #{args[2]}"
373 when 'OUTSIDE'
374 item = iface(args[1])
375 error("interface #{args[1]} already outside") if item[:role] == 'outside'
376 item[:role] = 'outside'
377 puts "ok nat outside #{args[1]}"
378 when 'PAT'
379 item = iface(args[1])
380 count = args[2].to_i
381 error("bad pool size #{args[2]}") unless count.between?(1, 1000)
382 error("pat already on #{args[1]}") if item[:pool]
383 item[:pool] = count
384 item[:role] = 'outside'
385 puts "ok nat pat #{args[1]} #{ip(item[:address])} 20000-#{19999 + count}"
386 when 'FORWARD'
387 item = iface(args[1])
388 port = args[3].to_i
389 error("bad port #{args[3]}") unless port.between?(1, 65535)
390 inside = address(args[4])
391 inport = args[5].to_i
392 error("bad port #{args[5]}") unless inport.between?(1, 65535)
393 error("interface #{args[1]} not outside") unless item[:role] == 'outside'
394 error("address #{args[4]} is local") if local(inside)
395 error("duplicate forward #{args[1]} #{args[2]} #{args[3]}") if @forwards.any? { |rule| rule[:dev] == args[1] && rule[:proto] == args[2] && rule[:port] == port }
396 @forwards << { dev: args[1], proto: args[2], port: port, inside: inside, inport: inport }
397 puts "ok nat forward #{args[1]} #{args[2]} #{port} #{args[4]}:#{inport}"
398 end
399 when 'RPF'
400 iface(name)[:rpf] = args[1].downcase
401 puts "ok rpf #{name} #{args[1].downcase}"
402 when 'TIMEOUT'
403 error("bad timeout #{args[1]}") unless args[1].to_i.between?(1, 86400)
404 @timeouts[name] = args[1].to_i
405 puts "ok timeout #{name} #{args[1]}"
406 when 'TICK'
407 error("bad tick #{name}") unless name.to_i.between?(1, 86400)
408 @clock += name.to_i
409 conns = @connections.length
410 mappings = @pat.length
411 @connections.reject! do |item|
412 timeout = item[:proto] == 'TCP' ? { 'SYN_SENT' => 60, 'FIN_WAIT' => 60, 'CLOSED' => 10 }.fetch(item[:state], @timeouts['TCP']) : @timeouts[item[:proto]]
413 @clock - item[:last] >= timeout
414 end
415 @pat.reject! { |item| @clock - item[:last] >= @timeouts[item[:proto]] }
416 puts "ok tick time=#{@clock} expired conns=#{conns - @connections.length} nat=#{mappings - @pat.length}"
417 when 'PACKET'
418 item = iface(name)
419 protocol, flags = args[1], args[2]
420 error("flags not allowed for #{protocol}") if protocol != 'TCP' && flags != '-'
421 error("bad flags #{flags}") if protocol == 'TCP' && flags != '-' && (!flags.match?(/\A[SAFR]+\z/) || flags.chars.uniq.length != flags.length)
422 source = address(args[3])
423 sport = args[4].to_i
424 error("bad port #{args[4]}") unless sport.between?(0, 65535)
425 destination = address(args[5])
426 dport = args[6].to_i
427 error("bad port #{args[6]}") unless dport.between?(0, 65535)
428 ttl = args[7].to_i
429 error("bad ttl #{args[7]}") unless ttl.between?(1, 255)
430 @packet_number += 1
431 item[:rx] += 1
432 verdict, reason = packet(name, protocol, flags, source, sport, destination, dport, ttl)
433 @counts[reason] += 1
434 item[:drop] += 1 if DROP_REASONS.include?(reason)
435 puts "##{@packet_number} #{protocol} #{flags} #{args[3]}:#{args[4]} > #{args[5]}:#{args[6]} #{verdict}"
436 when 'SHOW'
437 show(name)
438 when 'CLEAR'
439 if name == 'COUNTERS'
440 reset_counters
441 puts 'ok clear counters'
442 else
443 puts "ok clear conns=#{@connections.length} nat=#{@pat.length}"
444 @connections.clear
445 @pat.clear
446 end
447 end
448 end

450 def run
451 count = STDIN.readline.to_i
452 count.times do
453 begin
454 command(STDIN.readline.split)
455 rescue RouterError => exception
456 @counts['errors'] += 1
457 puts "error: #{exception.message}"
458 end
459 end
460 end
461end
