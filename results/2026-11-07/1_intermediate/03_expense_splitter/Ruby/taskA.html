20 def amount(value)
21 whole, cents = value.split(".")
22 whole.to_i * 100 + cents.to_i
23 end

25 def money(value)
26 format("%d.%02d", value.div(100), value % 100)
27 end

29 def signed(value)
30 return "+" + money(value) if value.positive?
31 return "-" + money(-value) if value.negative?
32 "0.00"
33 end

35 def percent(part, whole)
36 money((part * 10_000).div(whole)) + "%"
37 end

39 def rate_text(value)
40 format("%d.%04d", value.div(10_000), value % 10_000)
41 end

43 def distribute(total, weights)
44 sum = weights.sum
45 portions = weights.map { |weight| (total * weight).div(sum) }
46 order = weights.each_index.sort_by { |i| [-(total * weights[i] % sum), i] }
47 (total - portions.sum).times { |i| portions[order[i]] += 1 }
48 portions
49 end

51 def equal(total, count)
52 q, m = total.divmod(count)
53 Array.new(count) { |i| q + (i < m ? 1 : 0) }
54 end

56 def note(item)
57 item[:currency].empty? ? "" : " [#{money(item[:amount])} #{item[:currency]}]"
58 end

60 def names(active = false)
61 @members.keys.select { |name| !active || @members[name][:active] }.sort

64 def member(name, active = true)
65 value = @members[name]
66 fail_with("unknown member #{name}") unless value
67 fail_with("inactive member #{name}") if active && !value[:active]

70 def involved?(expense, name)
71 (expense[:payers] + expense[:participants]).any? { |person| person[:name] == name }
72 end

74 def net(expense, name)
75 paid = expense[:payers].each_with_index.sum do |person, i|
76 person[:name] == name ? expense[:contributions][i] : 0
77 end
78 share = expense[:participants].each_with_index.sum do |person, i|
79 person[:name] == name ? expense[:shares][i] : 0
80 end
81 paid - share
82 end

84 def get_expense(id, frozen = false)
85 expense = @expenses.find { |item| item[:id] == id }
86 fail_with("unknown expense #{id}") unless expense
87 if frozen
88 (expense[:payers] + expense[:participants]).each do |person|
89 fail_with("inactive member #{person[:name]}") unless @members[person[:name]][:active]
90 end
91 end
92 expense
93 end

95 def convert(value, currency)
96 return value if currency.empty?
97 fail_with("unknown currency #{currency}") unless @currencies.key?(currency)
98 (value * @currencies[currency] + 5000).div(10_000)
99 end

101 def specification(value)
102 text, currency = value.split("@")
103 currency ||= ""
104 original = amount(text)
105 total = convert(original, currency)
106 fail_with("invalid amount") if original.zero? || total.zero?
107 { amount: original, currency: currency, total: total }
108 end

110 def original_shares(expense, total)
111 weights = expense[:participants].map { |person| person[:value] }
112 portions = case expense[:mode]
113 when "EQUAL"
114 equal(expense[:amount], weights.length)
115 when "EXACT"
116 weights
117 when "ADJUST"
118 equal(expense[:amount] - weights.sum, weights.length).each_with_index.map do |value, i|
119 value + weights[i]
120 end
121 else
122 distribute(expense[:amount], weights)
123 end
124 distribute(total, portions)
125 end

127 def prepare(id, payer, spec, category, mode, parts, previous = nil)
128 member(payer)
129 converted = specification(spec)
130 participants = []
131 if mode == "EQUAL" && parts[0] == "ALL"
132 participants = names(true).map { |name| { name: name, value: 0 } }
133 else
134 seen = {}
135 parts.each do |part|
136 name, value = part.split(":")
137 member(name)
138 fail_with("duplicate member #{name}") if seen[name]
139 seen[name] = true
140 specified = if value.nil?
141 0
142 elsif %w[EXACT ADJUST].include?(mode)
143 amount(value)
144 else
145 value.to_i
146 end
147 participants << { name: name, value: specified }
148 end
149 end
150 sum = participants.sum { |person| person[:value] }
151 a = converted[:amount]
152 if (mode == "EXACT" && sum != a) || (mode == "PERCENT" && sum != 10_000) ||
153 (mode == "SHARES" && sum.zero?) || (mode == "ADJUST" && sum > a)
154 fail_with("splits do not add up")
155 end
156 expense = {
157 id: id, day: previous ? previous[:day] : @day,
158 record: previous ? previous[:record] : 0, **converted,
159 category: category, mode: mode, payers: [{ name: payer, value: a }],
160 participants: participants, contributions: [converted[:total]],
161 interval: previous ? previous[:interval] : 0,
162 next: previous ? previous[:next] : 0,
163 copies: previous ? previous[:copies] : 0
164 }
165 expense[:shares] = original_shares(expense, expense[:total])
166 expense
167 end

169 def balances
170 values = names.to_h { |name| [name, 0] }
171 @expenses.each do |expense|
172 expense[:payers].each_with_index do |person, i|
173 values[person[:name]] += expense[:contributions][i]
174 end
175 expense[:participants].each_with_index do |person, i|
176 values[person[:name]] -= expense[:shares][i]
177 end
178 end
179 @settlements.each do |payment|
180 values[payment[:from]] += payment[:total]
181 values[payment[:to]] -= payment[:total]
182 end
183 values
184 end

186 def spent(category)
187 @expenses.select { |expense| expense[:category] == category }.sum { |expense| expense[:total] }
188 end
189
190 def warning(expense)
191 budget = @budgets[expense[:category]]
192 total = spent(expense[:category])
193 budget && total > budget ? [" WARNING #{expense[:category]} over budget by #{money(total - budget)}"] : []
194 end

196 def debts
197 result = Hash.new(0)
198 @expenses.each do |expense|
199 people = names.select { |name| involved?(expense, name) }
200 nets = people.map { |name| { name: name, value: net(expense, name) } }
201 borrowers = nets.select { |person| person[:value].negative? }
202 lenders = nets.select { |person| person[:value].positive? }
203 i = 0
204 j = 0
205 while i < borrowers.length && j < lenders.length
206 value = [-borrowers[i][:value], lenders[j][:value]].min
207 result[[borrowers[i][:name], lenders[j][:name]]] += value
208 borrowers[i][:value] += value
209 lenders[j][:value] -= value
210 i += 1 if borrowers[i][:value].zero?
211 j += 1 if lenders[j][:value].zero?
212 end
213 end
214 @settlements.each do |payment|
215 result[[payment[:from], payment[:to]]] -= payment[:total]
216 end
217 result
218 end
219
220 def greedy(balances)
221 values = balances.dup
222 result = []
223 loop do
224 debtors = values.select { |_, value| value.negative? }
225 break if debtors.empty?
226 from, debt = debtors.min_by { |name, value| [value, name] }
227 to, credit = values.select { |_, value| value.positive? }.min_by { |name, value| [-value, name] }
228 value = [-debt, credit].min
229 result << { from: from, to: to, amount: value }
230 values[from] += value
231 values[to] -= value
232 end
233 result
234 end

236 def transfer_text(transfer, indent)
237 "#{indent}#{transfer[:from]} -> #{transfer[:to]} #{money(transfer[:amount])}"
238 end

240 def optimal
241 values = balances
242 people = names.reject { |name| values[name].zero? }
243 fail_with("too many members") if people.length > 12
244 full = (1 << people.length) - 1
245 sums = Array.new(full + 1, 0)
246 (1..full).each do |mask|
247 bit = mask & -mask
248 index = bit.bit_length - 1
249 sums[mask] = sums[mask ^ bit] + values[people[index]]
250 end
251 memo = { 0 => 0 }
252 best = lambda do |mask|
253 return memo[mask] if memo.key?(mask)
254 first = mask & -mask
255 maximum = 0
256 sub = mask
257 while sub.positive?
258 if (sub & first).positive? && sums[sub].zero?
259 maximum = [maximum, 1 + best.call(mask ^ sub)].max
260 end
261 sub = (sub - 1) & mask
262 end
263 memo[mask] = maximum
264 end
265 list = lambda do |mask|
266 people.each_with_index.filter_map { |name, i| name if (mask & (1 << i)).positive? }
267 end
268 groups = []
269 mask = full
270 while mask.positive?
271 first = mask & -mask
272 candidates = []
273 sub = mask
274 while sub.positive?
275 if (sub & first).positive? && sums[sub].zero? && 1 + best.call(mask ^ sub) == best.call(mask)
276 candidates << sub
277 end
278 sub = (sub - 1) & mask
279 end
280 chosen = candidates.min_by { |candidate| list.call(candidate) }
281 groups << list.call(chosen)
282 mask ^= chosen
283 end
284 output = ["SIMPLIFY OPTIMAL transfers #{people.length - groups.length} groups #{groups.length}"]
285 groups.each do |group|
286 output << " group " + group.join(" ")
287 subset = group.to_h { |name| [name, values[name]] }
288 output.concat(greedy(subset).map { |transfer| transfer_text(transfer, " ") })
289 end
290 output
291 end

293 def command(line)
294 name, *args = line.split(" ")
295 output = []
296 case name
297 when "DAY"
298 day = args[0].to_i
299 fail_with("invalid day #{args[0]}") if day <= @day
300 output << "DAY #{day}"
301 loop do
302 pending = @expenses.select { |expense| expense[:interval].positive? && expense[:next] <= day }
303 .min_by { |expense| [expense[:next], expense[:record]] }
304 break unless pending
305 template = pending
306 active = (template[:payers] + template[:participants]).all? { |person| @members[person[:name]][:active] }
307 total = convert(template[:amount], template[:currency])
308 if !active || total.zero?
309 template[:interval] = 0
310 output << " CANCEL #{template[:id]}"
311 next
312 end
313 template[:copies] += 1
314 id = "#{template[:id]}##{template[:copies] + 1}"
315 @serial += 1
316 copy = template.merge(
317 id: id, day: template[:next], record: @serial, total: total,
318 payers: template[:payers].map(&:dup), participants: template[:participants].map(&:dup),
319 shares: original_shares(template, total),
320 contributions: distribute(total, template[:payers].map { |person| person[:value] }),
321 interval: 0, next: 0, copies: 0
322 )
323 @expenses << copy
324 @used[id] = true
325 template[:next] += template[:interval]
326 output << " RECUR #{id} #{money(total)}"
327 output.concat(warning(copy))
328 end
329 @day = day
330 when "MEMBER"
331 old = @members[args[0]]
332 fail_with("duplicate member #{args[0]}") if old && old[:active]
333 @members[args[0]] = { active: true, day: @day }
334 output << "MEMBER #{args[0]} #{old ? 'rejoined' : 'joined'}"
335 when "LEAVE"
336 member(args[0])
337 fail_with("member has a balance #{args[0]}") unless balances[args[0]].zero?
338 @members[args[0]] = { active: false, day: @day }
339 output << "LEAVE #{args[0]}"
340 when "CURRENCY"
341 old = @currencies[args[0]]
342 value = args[1].to_i
343 @currencies[args[0]] = value
344 output << "CURRENCY #{args[0]} #{rate_text(value)}" + (old ? " was #{rate_text(old)}" : "")
345 when "EXPENSE", "EDIT"
346 previous = nil
347 if name == "EXPENSE"
348 fail_with("duplicate id #{args[0]}") if @used[args[0]]
349 else
350 previous = get_expense(args[0], true)
351 end
352 expense = prepare(args[0], args[1], args[2], args[3], args[4], args.drop(5), previous)
353 if previous
354 index = @expenses.index { |item| item.equal?(previous) }
355 @expenses[index] = expense
356 else
357 @serial += 1
358 expense[:record] = @serial
359 @expenses << expense
360 @used[expense[:id]] = true
361 end
362 shares = expense[:participants].each_with_index.map do |person, i|
363 "#{person[:name]} #{money(expense[:shares][i])}"
364 end
365 output << "#{name} #{expense[:id]} #{money(expense[:total])}#{note(expense)} = " + shares.join(" + ")
366 output.concat(warning(expense))
367 when "PAYERS"
368 expense = get_expense(args[0], true)
369 payers = []
370 seen = {}
371 args.drop(1).each do |part|
372 payer, value = part.split(":")
373 member(payer)
374 fail_with("duplicate member #{payer}") if seen[payer]
375 seen[payer] = true
376 payers << { name: payer, value: amount(value) }
377 end
378 fail_with("payers do not add up") unless payers.sum { |person| person[:value] } == expense[:amount]
379 expense[:payers] = payers
380 expense[:contributions] = distribute(expense[:total], payers.map { |person| person[:value] })
381 contributions = payers.each_with_index.map { |person, i| "#{person[:name]} #{money(expense[:contributions][i])}" }
382 output << "PAYERS #{expense[:id]} #{money(expense[:total])}#{note(expense)} = " + contributions.join(" + ")
383 when "DELETE"
384 expense = get_expense(args[0], true)
385 @expenses.delete_if { |item| item.equal?(expense) }
386 output << "DELETE #{args[0]}"
387 when "RECURRING"
388 expense = get_expense(args[0])
389 interval = args[1].to_i
390 expense[:interval] = interval
391 expense[:next] = @day + interval
392 suffix = interval.positive? ? " interval #{interval} next #{expense[:next]}" : " stopped"
393 output << "RECURRING #{args[0]}" + suffix
394 when "SETTLE"
395 member(args[0])
396 member(args[1])
397 fail_with("same member #{args[0]}") if args[0] == args[1]
398 converted = specification(args[2])
399 @serial += 1
400 payment = { from: args[0], to: args[1], **converted, day: @day, record: @serial }
401 @settlements << payment
402 output << "SETTLE #{payment[:from]} -> #{payment[:to]} #{money(payment[:total])}#{note(payment)}"
403 when "BUDGET"
404 value = amount(args[1])
405 if value.positive?
406 @budgets[args[0]] = value
407 else
408 @budgets.delete(args[0])
409 end
410 output << "BUDGET #{args[0]} " + (value.positive? ? money(value) : "removed")
411 when "BALANCES"
412 values = balances
413 output << "BALANCES"
414 output.concat(names(true).map { |person| " #{person} #{signed(values[person])}" })
415 when "OWES", "MATRIX"
416 member(args[0], false) if name == "OWES"
417 values = debts
418 debt = ->(a, b) { values[[a, b]] - values[[b, a]] }
419 output << (name == "OWES" ? "OWES #{args[0]}" : "MATRIX")
420 if name == "OWES"
421 names.reject { |person| person == args[0] }.each do |person|
422 value = debt.call(args[0], person)
423 if value.positive?
424 output << " #{args[0]} owes #{person} #{money(value)}"
425 elsif value.negative?
426 output << " #{person} owes #{args[0]} #{money(-value)}"
427 end
428 end
429 else
430 people = names(true)
431 if people.empty?
432 output << " (none)"
433 else
434 cells = people.map { |a| people.map { |b| a == b ? "-" : money([0, debt.call(a, b)].max) } }
435 l = people.map(&:length).max
436 w = (people + cells.flatten).map(&:length).max
437 output << " " * l + people.map { |person| " " + person.rjust(w) }.join
438 people.each_with_index do |person, i|
439 output << person.ljust(l) + cells[i].map { |cell| " " + cell.rjust(w) }.join
440 end
441 end
442 end
443 when "HISTORY"
444 member(args[0], false)
445 person = args[0]
446 items = @expenses.select { |expense| involved?(expense, person) }.map do |expense|
447 { day: expense[:day], record: expense[:record], label: expense[:id], change: net(expense, person) }
448 end
449 @settlements.select { |payment| payment[:from] == person || payment[:to] == person }.each do |payment|
450 items << {
451 day: payment[:day], record: payment[:record], label: "#{payment[:from]}->#{payment[:to]}",
452 change: payment[:from] == person ? payment[:total] : -payment[:total]
453 }
454 end
455 items.sort_by! { |item| [item[:day], item[:record]] }
456 running = 0
457 output << "HISTORY #{person}"
458 items.each do |item|
459 running += item[:change]
460 output << " day #{item[:day]} #{item[:label]} #{signed(item[:change])} balance #{signed(running)}"
461 end
462 when "CATEGORY"
463 output << "CATEGORY"
464 groups = @expenses.group_by { |expense| expense[:category] }
465 category_totals = groups.transform_values { |group| group.sum { |expense| expense[:total] } }
466 total = category_totals.values.sum
467 category_totals.keys.sort_by { |category| [-category_totals[category], category] }.each do |category|
468 value = category_totals[category]
469 output << " #{category} #{money(value)} #{percent(value, total)}"
470 totals = Hash.new(0)
471 groups[category].each do |expense|
472 expense[:participants].each_with_index { |person, i| totals[person[:name]] += expense[:shares][i] }
473 end
474 totals.select { |_, share| share.positive? }.sort_by { |person, share| [-share, person] }.each do |person, share|
475 output << " #{person} #{money(share)} #{percent(share, value)}"
476 end
477 end
478 when "BUDGETS"
479 output << "BUDGETS"
480 @budgets.keys.sort.each do |category|
481 budget = @budgets[category]
482 value = spent(category)
483 suffix = value <= budget ? " left #{money(budget - value)}" : " over #{money(value - budget)}"
484 output << " #{category} spent #{money(value)} budget #{money(budget)} used #{percent(value, budget)}" + suffix
485 end
486 when "TOTALS"
487 output << "TOTALS"
488 names.each do |person|
489 paid = 0
490 share = 0
491 @expenses.each do |expense|
492 expense[:payers].each_with_index { |payer, i| paid += expense[:contributions][i] if payer[:name] == person }
493 expense[:participants].each_with_index { |part, i| share += expense[:shares][i] if part[:name] == person }
494 end
495 sent = @settlements.select { |payment| payment[:from] == person }.sum { |payment| payment[:total] }
496 received = @settlements.select { |payment| payment[:to] == person }.sum { |payment| payment[:total] }
497 output << " #{person} paid #{money(paid)} share #{money(share)} sent #{money(sent)} received #{money(received)}"
498 end
499 currencies = @expenses.map { |expense| expense[:currency] }.reject(&:empty?).uniq.sort
500 currencies.each do |currency|
501 selected = @expenses.select { |expense| expense[:currency] == currency }
502 original = selected.sum { |expense| expense[:amount] }
503 total = selected.sum { |expense| expense[:total] }
504 output << " currency #{currency} expenses #{selected.length} amount #{money(original)} base #{money(total)}"
505 end
506 total = @expenses.sum { |expense| expense[:total] }
507 settled = @settlements.sum { |payment| payment[:total] }
508 output << " expenses #{@expenses.length} total #{money(total)} settlements #{@settlements.length} total #{money(settled)}"
509 when "MEMBERS"
510 output << "MEMBERS"
511 names.each do |person|
512 info = @members[person]
513 count = @expenses.count { |expense| involved?(expense, person) }
514 output << " #{person} #{info[:active] ? 'active' : 'left'} day #{info[:day]} expenses #{count}"
515 end
516 when "EXPENSES"
517 member(args[1], false) if args[0] == "MEMBER"
518 selected = @expenses.select do |expense|
519 args[0] == "ALL" ||
520 (args[0] == "CATEGORY" && expense[:category] == args[1]) ||
521 (args[0] == "MEMBER" && involved?(expense, args[1])) ||
522 (args[0] == "DAYS" && expense[:day].between?(args[1].to_i, args[2].to_i))
523 end.sort_by { |expense| [expense[:day], expense[:record]] }
524 output << line
525 selected.each do |expense|
526 payers = expense[:payers].map { |person| person[:name] }.join(",")
527 output << " day #{expense[:day]} #{expense[:id]} #{expense[:category]} #{money(expense[:total])} by #{payers}"
528 end
529 output << " count #{selected.length} total #{money(selected.sum { |expense| expense[:total] })}"
530 when "PERIODS"
531 k = args[0].to_i
532 groups = @expenses.group_by { |expense| (expense[:day] - 1).div(k) }
533 output << "PERIODS #{k}"
534 groups.keys.sort.each do |index|
535 group = groups[index]
536 total = group.sum { |expense| expense[:total] }
537 output << " days #{index * k + 1}-#{index * k + k} expenses #{group.length} total #{money(total)}"
538 end
539 when "SHOW"
540 expense = get_expense(args[0])
541 output << "SHOW #{expense[:id]}"
542 output << " day #{expense[:day]} category #{expense[:category]} mode #{expense[:mode]} total #{money(expense[:total])}#{note(expense)}"
543 expense[:payers].each_with_index do |person, i|
544 output << " paid #{person[:name]} #{money(expense[:contributions][i])}"
545 end
546 units = expense[:participants].sum { |person| person[:value] }
547 expense[:participants].each_with_index do |person, i|
548 suffix = case expense[:mode]
549 when "EXACT"
550 " " + money(person[:value])
551 when "PERCENT"
552 " " + money(person[:value]) + "%"
553 when "SHARES"
554 " #{person[:value]}/#{units}"
555 when "ADJUST"
556 " +" + money(person[:value])
557 else
558 ""
559 end
560 output << " share #{person[:name]} #{money(expense[:shares][i])}" + suffix
561 end
562 if expense[:interval].positive?
563 output << " recurring interval #{expense[:interval]} next #{expense[:next]}"
564 end
565 when "SIMPLIFY"
566 return optimal if args[0] == "OPTIMAL"
567 transfers = greedy(balances)
568 output << "SIMPLIFY GREEDY transfers #{transfers.length}"
569 output.concat(transfers.map { |transfer| transfer_text(transfer, " ") })
570 end
571 if %w[BALANCES OWES HISTORY CATEGORY BUDGETS MEMBERS PERIODS].include?(name) && output.length == 1
572 output << " (none)"
573 end
574 output
575 end
576end
