64 static long[] distribute(long total, List<Long> weights) {
65 long sum = weights.stream().mapToLong(Long::longValue).sum();
66 long[] result = new long[weights.size()], remainders = new long[weights.size()];
67 List<Integer> order = new ArrayList<>();
68 long allocated = 0;
69 for (int i = 0; i < weights.size(); i++) {
70 result[i] = total*weights.get(i)/sum;
71 remainders[i] = total*weights.get(i)%sum;
72 allocated += result[i];
73 order.add(i);
74 }
75 order.sort(Comparator.<Integer>comparingLong(i -> remainders[i]).reversed().thenComparingInt(i -> i));
76 for (int i = 0; i < total-allocated; i++) result[order.get(i)]++;
77 return result;
78 }
79 static List<Long> weights(List<Part> parts) { return parts.stream().map(Part::value).toList(); }
80 void compute(Expense e) {
81 long a = e.amount.value();
82 e.total = convert(e.amount);
83 long sum = e.participants.stream().mapToLong(Part::value).sum();
84 long[] portions = new long[e.participants.size()];
85 if (e.mode.equals("EXACT")) {
86 if (sum != a) fail("splits do not add up");
87 portions = e.participants.stream().mapToLong(Part::value).toArray();
88 } else if (e.mode.equals("PERCENT") || e.mode.equals("SHARES")) {
89 if (e.mode.equals("PERCENT") && sum != 10000 || e.mode.equals("SHARES") && sum == 0) fail("splits do not add up");
90 portions = distribute(a,weights(e.participants));
91 } else {
92 long extras = e.mode.equals("ADJUST") ? sum : 0;
93 if (extras > a) fail("splits do not add up");
94 int k = portions.length;
95 for (int i = 0; i < k; i++) {
96 portions[i] = (a-extras)/k+(i < (a-extras)%k ? 1 : 0)+(e.mode.equals("ADJUST") ? e.participants.get(i).value() : 0);
97 }
98 }
99 e.shares = distribute(e.total,Arrays.stream(portions).boxed().toList());
100 e.contributions = distribute(e.total,weights(e.payers));
101 }
102 Expense build(String[] tokens) {
103 active(tokens[2]);
104 Expense e = new Expense();
105 e.amount = amount(tokens[3]);
106 long total = convert(e.amount);
107 if (e.amount.value() == 0 || total == 0) fail("invalid amount");
108 e.category = tokens[4];
109 e.mode = tokens[5];
110 e.payers = List.of(new Part(tokens[2],e.amount.value()));
111 e.participants = new ArrayList<>();
112 Set<String> seen = new HashSet<>();
113 if (e.mode.equals("EQUAL") && tokens[6].equals("ALL")) {
114 for (var member : members.entrySet()) {
115 if (member.getValue().active()) e.participants.add(new Part(member.getKey(),0));
116 }
117 } else {
118 for (int i = 6; i < tokens.length; i++) {
119 String[] part = tokens[i].split(":");
120 active(part[0]);
121 if (!seen.add(part[0])) fail("duplicate member "+part[0]);
122 long value = part.length == 1 ? 0 : e.mode.equals("EXACT") || e.mode.equals("ADJUST") ? money(part[1]) : Long.parseLong(part[1]);
123 e.participants.add(new Part(part[0],value));
124 }
125 }
126 compute(e);
127 return e;
128 }

135 TreeMap<String,Long> net(Expense e) {
136 TreeMap<String,Long> result = new TreeMap<>();
137 for (int i = 0; i < e.payers.size(); i++) result.merge(e.payers.get(i).name(),e.contributions[i],Long::sum);
138 for (int i = 0; i < e.participants.size(); i++) result.merge(e.participants.get(i).name(),-e.shares[i],Long::sum);
139 return result;
140 }
141 TreeMap<String,Long> balances() {
142 TreeMap<String,Long> result = new TreeMap<>();
143 for (String name : members.keySet()) result.put(name,0L);
144 for (Expense e : expenses.values()) net(e).forEach((name,value) -> result.merge(name,value,Long::sum));
145 for (Settlement s : settlements) {
146 result.merge(s.from(),s.total(),Long::sum);
147 result.merge(s.to(),-s.total(),Long::sum);
148 }
149 return result;
150 }

165 static List<Transfer> greedy(Map<String,Long> balances) {
166 TreeMap<String,Long> work = new TreeMap<>(balances);
167 List<Transfer> result = new ArrayList<>();
168 while (!work.isEmpty()) {
169 var sender = Collections.min(work.entrySet(),Map.Entry.comparingByValue());
170 var receiver = Collections.max(work.entrySet(),Map.Entry.comparingByValue());
171 long minimum = sender.getValue(), maximum = receiver.getValue();
172 if (minimum >= 0) break;
173 String from = sender.getKey(), to = receiver.getKey();
174 long value = Math.min(-minimum,maximum);
175 result.add(new Transfer(from,to,value));
176 work.put(from,minimum+value);
177 work.put(to,maximum-value);
178 }
179 return result;
180 }
181 long[][] debts(List<String> names) {
182 int n = names.size();
183 Map<String,Integer> index = new HashMap<>();
184 for (int i = 0; i < n; i++) index.put(names.get(i),i);
185 long[][] result = new long[n][n];
186 for (Expense e : expenses.values()) {
187 TreeMap<String,Long> amounts = net(e);
188 List<String> borrowers = amounts.entrySet().stream().filter(p -> p.getValue() < 0).map(Map.Entry::getKey).toList();
189 List<String> lenders = amounts.entrySet().stream().filter(p -> p.getValue() > 0).map(Map.Entry::getKey).toList();
190 int a = 0, b = 0;
191 while (a < borrowers.size() && b < lenders.size()) {
192 String x = borrowers.get(a), y = lenders.get(b);
193 long value = Math.min(-amounts.get(x),amounts.get(y));
194 int i = index.get(x), j = index.get(y);
195 result[i][j] += value;
196 result[j][i] -= value;
197 amounts.put(x,amounts.get(x)+value);
198 amounts.put(y,amounts.get(y)-value);
199 if (amounts.get(x) == 0) a++;
200 if (amounts.get(y) == 0) b++;
201 }
202 }
203 for (Settlement s : settlements) {
204 int i = index.get(s.from()), j = index.get(s.to());
205 result[i][j] -= s.total();
206 result[j][i] += s.total();
207 }
208 return result;
209 }
210 static int[] indexes(int mask, int n) {
211 return IntStream.range(0,n).filter(i -> (mask&(1<<i)) != 0).toArray();
212 }
213 List<String> execute(String line) {
214 String[] t = line.split(" ");
215 String command = t[0];
216 List<String> out = new ArrayList<>();
217 switch (command) {
218 case "DAY": {
219 long d = Long.parseLong(t[1]);
220 if (d <= day) fail("invalid day "+t[1]);
221 out.add("DAY "+d);
222 while (true) {
223 Expense template = expenses.values().stream().filter(e -> e.interval > 0 && e.next <= d)
224 .min(Comparator.comparingLong((Expense e) -> e.next).thenComparingLong(e -> e.record)).orElse(null);
225 if (template == null) break;
226 boolean active = template.payers.stream().allMatch(p -> members.get(p.name()).active()) &&
227 template.participants.stream().allMatch(p -> members.get(p.name()).active());
228 if (!active || convert(template.amount) == 0) {
229 template.interval = 0;
230 out.add(" CANCEL "+template.id);
231 } else {
232 Expense copy = new Expense();
233 copy.id = template.id+"#"+(template.copies+2);
234 copy.day = template.next;
235 copy.record = ++serial;
236 copy.amount = template.amount;
237 copy.category = template.category;
238 copy.mode = template.mode;
239 copy.payers = template.payers;
240 copy.participants = template.participants;
241 compute(copy);
242 expenses.put(copy.id,copy);
243 used.add(copy.id);
244 template.copies++;
245 template.next += template.interval;
246 out.add(" RECUR "+copy.id+" "+fmt(copy.total));
247 warning(copy,out);
248 }
249 }
250 day = d;
251 break;
252 }
253 case "MEMBER": {
254 Member old = members.get(t[1]);
255 if (old != null && old.active()) fail("duplicate member "+t[1]);
256 members.put(t[1],new Member(true,day));
257 out.add("MEMBER "+t[1]+(old == null ? " joined" : " rejoined"));
258 break;
259 }
260 case "LEAVE": {
261 active(t[1]);
262 if (balances().get(t[1]) != 0) fail("member has a balance "+t[1]);
263 members.put(t[1],new Member(false,day));
264 out.add("LEAVE "+t[1]);
265 break;
266 }
267 case "CURRENCY": {
268 long value = Long.parseLong(t[2]);
269 Long old = currencies.put(t[1],value);
270 out.add("CURRENCY "+t[1]+" "+rate(value)+(old == null ? "" : " was "+rate(old)));
271 break;
272 }
273 case "EXPENSE", "EDIT": {
274 Expense old = null;
275 if (command.equals("EXPENSE")) {
276 if (used.contains(t[1])) fail("duplicate id "+t[1]);
277 } else {
278 old = expense(t[1]);
279 thawed(old);
280 }
281 Expense e = build(t);
282 e.id = t[1];
283 e.day = old == null ? day : old.day;
284 e.record = old == null ? ++serial : old.record;
285 if (old != null) {
286 e.interval = old.interval;
287 e.next = old.next;
288 e.copies = old.copies;
289 }
290 expenses.put(e.id,e);
291 used.add(e.id);
292 out.add(command+" "+e.id+" "+fmt(e.total)+note(e.amount)+" = "+pairs(e.participants,e.shares));
293 warning(e,out);
294 break;
295 }
296 case "PAYERS": {
297 Expense e = expense(t[1]);
298 thawed(e);
299 List<Part> payers = new ArrayList<>();
300 Set<String> seen = new HashSet<>();
301 long total = 0;
302 for (int i = 2; i < t.length; i++) {
303 String[] p = t[i].split(":");
304 active(p[0]);
305 if (!seen.add(p[0])) fail("duplicate member "+p[0]);
306 long value = money(p[1]);
307 payers.add(new Part(p[0],value));

309 }
310 if (total != e.amount.value()) fail("payers do not add up");

359 out.add(command.equals("OWES") ? "OWES "+t[1] : "MATRIX");

526 for (Transfer x : transfers) out.add(" "+x.from()+" -> "+x.to()+" "+fmt(x.amount()));
527 } else {
528 List<String> names = new ArrayList<>(all.keySet());
529 int n = names.size();
530 if (n > 12) fail("too many members");
531 int size = 1<<n;
532 long[] sums = new long[size];
533 int[] dp = new int[size];
534 Arrays.fill(dp,-1000);
535 dp[0] = 0;
536 for (int mask = 1; mask < size; mask++) {
537 int bit = Integer.numberOfTrailingZeros(mask);
538 sums[mask] = sums[mask^(1<<bit)]+all.get(names.get(bit));
539 int first = mask&-mask;
540 for (int sub = mask; sub > 0; sub = (sub-1)&mask) {
541 if ((sub&first) != 0 && sums[sub] == 0) dp[mask] = Math.max(dp[mask],1+dp[mask^sub]);
542 }
543 }
544 int remaining = size-1;
545 List<Integer> groups = new ArrayList<>();
546 while (remaining != 0) {
547 int first = remaining&-remaining, best = 0;
548 for (int sub = remaining; sub > 0; sub = (sub-1)&remaining) {
549 if ((sub&first) != 0 && sums[sub] == 0 && 1+dp[remaining^sub] == dp[remaining] &&
550 (best == 0 || Arrays.compare(indexes(sub,n),indexes(best,n)) < 0)) best = sub;
551 }
552 groups.add(best);
553 remaining ^= best;
554 }
555 List<List<Transfer>> transfers = new ArrayList<>();
556 for (int group : groups) {
557 TreeMap<String,Long> amounts = new TreeMap<>();
558 for (int i : indexes(group,n)) amounts.put(names.get(i),all.get(names.get(i)));
559 transfers.add(greedy(amounts));
560 }
561 out.add("SIMPLIFY OPTIMAL transfers "+transfers.stream().mapToInt(List::size).sum()+" groups "+groups.size());
562 for (int i = 0; i < groups.size(); i++) {
563 List<String> groupNames = Arrays.stream(indexes(groups.get(i),n)).mapToObj(names::get).toList();
564 out.add(" group "+String.join(" ",groupNames));
565 for (Transfer x : transfers.get(i)) out.add(" "+x.from()+" -> "+x.to()+" "+fmt(x.amount()));
566 }
567 }
568 break;
569 }
570 }
571 if (Set.of("BALANCES","OWES","HISTORY","CATEGORY","BUDGETS","MEMBERS","PERIODS").contains(command) && out.size() == 1) {
572 out.add(" (none)");
573 }
574 return out;
575 }
