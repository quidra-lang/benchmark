57fn integer(s: &str) -> Integer {
58 s.parse().unwrap()
59}
60fn amount(s: &str) -> Integer {
61 let (a, b) = s.split_once('.').unwrap();
62 integer(a) * 100 + integer(b)
63}
64fn money(n: Integer) -> String {
65 format!("{}.{:02}", n / 100, n % 100)
66}
67fn signed_money(n: Integer) -> String {
68 if n > 0 {
69 format!("+{}", money(n))
70 } else if n < 0 {
71 format!("-{}", money(-n))
72 } else {
73 "0.00".to_owned()
74 }
75}
76fn rate_text(n: Integer) -> String {
77 format!("{}.{:04}", n / 10000, n % 10000)
78}
79fn percentage(p: Integer, w: Integer) -> String {
80 format!("{}%", money(p * 10000 / w))
81}
82fn amount_spec(s: &str) -> (Integer, String) {
83 let (value, currency) = s.split_once('@').unwrap_or((s, ""));
84 (amount(value), currency.to_owned())
85}
86fn note(value: Integer, currency: &str) -> String {
87 if currency.is_empty() {
88 String::new()
89 } else {
90 format!(" [{} {currency}]", money(value))
91 }
92}
93fn distribute(total: Integer, weights: &[Integer]) -> Vec<Integer> {
94 let sum: Integer = weights.iter().sum();
95 let mut result: Vec<Integer> = weights.iter().map(|w| total * w / sum).collect();
96 let remainders: Vec<Integer> = weights.iter().map(|w| total * w % sum).collect();
97 let left = total - result.iter().sum::<Integer>();
98 let mut order: Vec<usize> = (0..weights.len()).collect();
99 order.sort_by(|&i, &j| remainders[j].cmp(&remainders[i]));
100 for &i in &order[..left as usize] {
101 result[i] += 1;
102 }
103 result
104}
105fn equal_distribution(total: Integer, count: usize) -> Vec<Integer> {
106 (0..count)
107 .map(|i| {
108 total / count as Integer + Integer::from((i as Integer) < total % count as Integer)
109 })
110 .collect()
111}
112fn contributions(e: &mut Expense) {
113 let weights: Vec<Integer> = e.payers.iter().map(|p| p.value).collect();
114 for (p, value) in e.payers.iter_mut().zip(distribute(e.total, &weights)) {
115 p.result = value;
116 }
117}
118fn net(e: &Expense, count: usize) -> Vec<Integer> {
119 let mut result = vec![0; count];
120 for p in &e.payers {
121 result[p.member] += p.result;
122 }
123 for p in &e.participants {
124 result[p.member] -= p.result;
125 }
126 result
127}
128fn involved(e: &Expense, id: usize) -> bool {
129 e.payers
130 .iter()
131 .chain(&e.participants)
132 .any(|p| p.member == id)
133}
134fn greedy(group: &[usize], original: &[Integer]) -> Vec<Transfer> {
135 let mut values = original.to_vec();
136 let mut result = Vec::new();
137 loop {
138 let (mut from, mut to): (Option<usize>, Option<usize>) = (None, None);
139 for &i in group {
140 if values[i] < 0 && from.is_none_or(|j| values[i] < values[j]) {
141 from = Some(i);
142 }
143 if values[i] > 0 && to.is_none_or(|j| values[i] > values[j]) {
144 to = Some(i);
145 }
146 }
147 let Some(from) = from else {
148 break;
149 };
150 let to = to.unwrap();
151 let amount = (-values[from]).min(values[to]);
152 result.push(Transfer { from, to, amount });
153 values[from] += amount;
154 values[to] -= amount;
155 }
156 result
157}
158fn best_groups(mask: usize, sums: &[Integer], memo: &mut [Option<usize>]) -> usize {
159 if let Some(value) = memo[mask] {
160 return value;
161 }
162 let first = 1usize << mask.trailing_zeros();
163 let mut sub = mask;
164 let mut result = 0;
165 while sub > 0 {
166 if sub & first != 0 && sums[sub] == 0 {
167 result = result.max(1 + best_groups(mask ^ sub, sums, memo));
168 }
169 sub = (sub - 1) & mask;
170 }
171 memo[mask] = Some(result);
172 result
173}
174impl Engine {
175 fn convert(&self, value: Integer, currency: &str) -> Result<Integer, String> {
176 if currency.is_empty() {
177 return Ok(value);
178 }
179 let rate = self
180 .currencies
181 .get(currency)
182 .ok_or_else(|| format!("unknown currency {currency}"))?;
183 Ok((value * rate + 5000) / 10000)
184 }
185 fn member_id(&self, name: &str, active: bool) -> Result<usize, String> {
186 let id = self
187 .members
188 .iter()
189 .position(|m| m.name == name)
190 .ok_or_else(|| format!("unknown member {name}"))?;
191 if active && !self.members[id].active {
192 return Err(format!("inactive member {name}"));
193 }
194 Ok(id)
195 }
196 fn ordered_members(&self, active: bool) -> Vec<usize> {
197 let mut order: Vec<usize> = (0..self.members.len())
198 .filter(|&i| !active || self.members[i].active)
199 .collect();
200 order.sort_by(|&i, &j| self.members[i].name.cmp(&self.members[j].name));
201 order
202 }
203 fn expense_id(&self, id: &str) -> Result<usize, String> {
204 self.expenses
205 .iter()
206 .position(|e| e.id == id)
207 .ok_or_else(|| format!("unknown expense {id}"))
208 }
209 fn check_frozen(&self, e: &Expense) -> Result<(), String> {
210 for p in e.payers.iter().chain(&e.participants) {
211 if !self.members[p.member].active {
212 return Err(format!("inactive member {}", self.members[p.member].name));
213 }
214 }
215 Ok(())
216 }
217 fn compute(&self, e: &mut Expense) -> Result<(), String> {
218 e.total = self.convert(e.amount, &e.currency)?;
219 let specified: Vec<Integer> = e.participants.iter().map(|p| p.value).collect();
220 let portions = match e.mode.as_str() {
221 "EQUAL" => equal_distribution(e.amount, specified.len()),
222 "EXACT" => specified,
223 "ADJUST" => {
224 let extras: Integer = specified.iter().sum();
225 equal_distribution(e.amount - extras, specified.len())
226 .iter()
227 .zip(&specified)
228 .map(|(a, b)| a + b)
229 .collect()
230 }
231 _ => distribute(e.amount, &specified),
232 };
233 for (p, share) in e
234 .participants
235 .iter_mut()
236 .zip(distribute(e.total, &portions))
237 {
238 p.result = share;
239 }
240 contributions(e);
241 Ok(())
242 }
243 fn draft_expense(&self, a: &[&str], previous: Option<&Expense>) -> Result<Expense, String> {
244 let mut e = previous.cloned().unwrap_or_default();
245 let payer = self.member_id(a[2], true)?;
246 let (value, currency) = amount_spec(a[3]);
247 let total = self.convert(value, &currency)?;
248 if value == 0 || total == 0 {
249 return Err("invalid amount".to_owned());
250 }
251 e.amount = value;
252 e.currency = currency;
253 e.category = a[4].to_owned();
254 e.mode = a[5].to_owned();
255 e.payers = vec![Part {
256 member: payer,
257 value,
258 result: 0,
259 }];
260 e.participants.clear();
261 if e.mode == "EQUAL" && a[6] == "ALL" {
262 for i in self.ordered_members(true) {
263 e.participants.push(Part {
264 member: i,
265 value: 0,
266 result: 0,
267 });
268 }
269 } else {
270 let mut seen = HashSet::new();
271 for item in &a[6..] {
272 let (name, specified) = item.split_once(':').unwrap_or((*item, ""));
273 let member = self.member_id(name, true)?;
274 if !seen.insert(member) {
275 return Err(format!("duplicate member {name}"));
276 }
277 let value = if specified.is_empty() {
278 0
279 } else if e.mode == "PERCENT" || e.mode == "SHARES" {
280 integer(specified)
281 } else {
282 amount(specified)
283 };
284 e.participants.push(Part {
285 member,
286 value,
287 result: 0,
288 });
289 }
290 }
291 let sum: Integer = e.participants.iter().map(|p| p.value).sum();
292 if e.mode == "EXACT" && sum != e.amount
293 || e.mode == "PERCENT" && sum != 10000
294 || e.mode == "SHARES" && sum == 0
295 || e.mode == "ADJUST" && sum > e.amount
296 {
297 return Err("splits do not add up".to_owned());
298 }
299 self.compute(&mut e)?;
300 Ok(e)
301 }
302 fn balances(&self) -> Vec<Integer> {
303 let mut result = vec![0; self.members.len()];
304 for e in &self.expenses {
305 for (i, value) in net(e, self.members.len()).into_iter().enumerate() {
306 result[i] += value;
307 }
308 }
309 for s in &self.settlements {
310 result[s.from] += s.total;
311 result[s.to] -= s.total;
312 }
313 result
314 }
315 fn debts(&self) -> Vec<Vec<Integer>> {
316 let count = self.members.len();
317 let mut result = vec![vec![0; count]; count];
318 let order = self.ordered_members(false);
319 for e in &self.expenses {
320 let mut change = net(e, count);
321 let borrowers: Vec<usize> = order.iter().copied().filter(|&i| change[i] < 0).collect();
322 let lenders: Vec<usize> = order.iter().copied().filter(|&i| change[i] > 0).collect();
323 let (mut i, mut j) = (0, 0);
324 while i < borrowers.len() && j < lenders.len() {
325 let (from, to) = (borrowers[i], lenders[j]);
326 let value = (-change[from]).min(change[to]);
327 result[from][to] += value;
328 change[from] += value;
329 change[to] -= value;
330 if change[from] == 0 {
331 i += 1;
332 }
333 if change[to] == 0 {
334 j += 1;
335 }
336 }
337 }
338 for s in &self.settlements {
339 result[s.from][s.to] -= s.total;
340 }
341 for i in 0..count {
342 for j in i + 1..count {
343 let value = result[i][j] - result[j][i];
344 result[i][j] = value;
345 result[j][i] = -value;
346 }
347 }
348 result
349 }
350 fn spent(&self, category: &str) -> Integer {
351 self.expenses
352 .iter()
353 .filter(|e| e.category == category)
354 .map(|e| e.total)
355 .sum()
356 }
357 fn warning(&self, e: &Expense, out: &mut Vec<String>) {
358 if let Some(&budget) = self.budgets.get(&e.category) {
359 let total = self.spent(&e.category);
360 if total > budget {
361 out.push(format!(
362 " WARNING {} over budget by {}",
363 e.category,
364 money(total - budget)
365 ));
366 }
367 }
368 }
369 fn parts_text(&self, parts: &[Part]) -> String {
370 parts
371 .iter()
372 .map(|p| format!("{} {}", self.members[p.member].name, money(p.result)))
373 .collect::<Vec<_>>()
374 .join(" + ")
375 }
376 fn chronological(&self) -> Vec<usize> {
377 let mut order: Vec<usize> = (0..self.expenses.len()).collect();
378 order.sort_by_key(|&i| (self.expenses[i].day, self.expenses[i].serial));
379 order
380 }

382 transfers
383 .iter()
384 .map(|t| {
385 format!(
386 "{indent}{} -> {} {}",

390 )

392 .collect()
393 }
394 fn execute(&mut self, line: &str) -> Result<Vec<String>, String> {
395 let a: Vec<&str> = line.split(' ').collect();
396 let command = a[0];
397 let mut out = Vec::new();
398 match command {
399 "DAY" => {
400 let day = integer(a[1]);
401 if day <= self.day {
402 return Err(format!("invalid day {}", a[1]));
403 }
404 out.push(format!("DAY {day}"));
405 loop {
406 let selected = self
407 .expenses
408 .iter()
409 .enumerate()
410 .filter(|(_, e)| e.interval > 0 && e.next <= day)
411 .min_by_key(|(_, e)| (e.next, e.serial))
412 .map(|(i, _)| i);
413 let Some(i) = selected else {
414 break;
415 };
416 if self.check_frozen(&self.expenses[i]).is_err()
417 || self.convert(self.expenses[i].amount, &self.expenses[i].currency)? == 0
418 {
419 out.push(format!(" CANCEL {}", self.expenses[i].id));
420 self.expenses[i].interval = 0;
421 continue;
422 }
423 let mut copy = self.expenses[i].clone();
424 copy.id = format!("{}#{}", copy.id, copy.copies + 2);
425 copy.day = copy.next;
426 self.serial += 1;
427 copy.serial = self.serial;
428 copy.interval = 0;
429 copy.next = 0;
430 copy.copies = 0;
431 self.expenses[i].copies += 1;
432 let interval = self.expenses[i].interval;
433 self.expenses[i].next += interval;
434 self.compute(&mut copy)?;
435 self.used.insert(copy.id.clone());
436 out.push(format!(" RECUR {} {}", copy.id, money(copy.total)));
437 self.expenses.push(copy);
438 self.warning(self.expenses.last().unwrap(), &mut out);
439 }
440 self.day = day;
441 }
442 "MEMBER" => {
443 let previous = self.members.iter().position(|m| m.name == a[1]);
444 if let Some(i) = previous {
445 if self.members[i].active {
446 return Err(format!("duplicate member {}", a[1]));
447 }
448 self.members[i].active = true;
449 self.members[i].day = self.day;
450 } else {
451 self.members.push(Member {
452 name: a[1].to_owned(),
453 active: true,
454 day: self.day,
455 });
456 }
457 out.push(format!(
458 "MEMBER {} {}",
459 a[1],
460 if previous.is_some() {
461 "rejoined"
462 } else {
463 "joined"
464 }
465 ));
466 }
467 "LEAVE" => {
468 let i = self.member_id(a[1], true)?;
469 if self.balances()[i] != 0 {
470 return Err(format!("member has a balance {}", a[1]));
471 }
472 self.members[i].active = false;
473 self.members[i].day = self.day;
474 out.push(format!("LEAVE {}", a[1]));
475 }
476 "CURRENCY" => {
477 let value = integer(a[2]);
478 let mut result = format!("CURRENCY {} {}", a[1], rate_text(value));
479 if let Some(old) = self.currencies.insert(a[1].to_owned(), value) {
480 result.push_str(&format!(" was {}", rate_text(old)));
481 }
482 out.push(result);
483 }
484 "EXPENSE" | "EDIT" => {
485 let previous = if command == "EXPENSE" {
486 if self.used.contains(a[1]) {
487 return Err(format!("duplicate id {}", a[1]));
488 }
489 None
490 } else {
491 let i = self.expense_id(a[1])?;
492 self.check_frozen(&self.expenses[i])?;
493 Some(i)
494 };
495 let mut e = self.draft_expense(&a, previous.map(|i| &self.expenses[i]))?;
496 let position = previous.unwrap_or(self.expenses.len());
497 if let Some(i) = previous {
498 self.expenses[i] = e;
499 } else {
500 e.id = a[1].to_owned();
501 e.day = self.day;
502 self.serial += 1;
503 e.serial = self.serial;
504 self.used.insert(e.id.clone());
505 self.expenses.push(e);
506 }
507 let e = &self.expenses[position];
508 out.push(format!(
509 "{command} {} {}{} = {}",
510 e.id,
511 money(e.total),
512 note(e.amount, &e.currency),
513 self.parts_text(&e.participants)
514 ));
515 self.warning(e, &mut out);
516 }
517 "PAYERS" => {
518 let i = self.expense_id(a[1])?;
519 self.check_frozen(&self.expenses[i])?;
520 let mut payers = Vec::new();
521 let mut seen = HashSet::new();
522 let mut sum = 0;
523 for item in &a[2..] {
524 let (name, payment) = item.split_once(':').unwrap();
525 let member = self.member_id(name, true)?;
526 if !seen.insert(member) {
527 return Err(format!("duplicate member {name}"));
528 }
529 let value = amount(payment);
530 sum += value;
531 payers.push(Part {
532 member,
533 value,
534 result: 0,
535 });
536 }
537 let e = &mut self.expenses[i];
538 if sum != e.amount {
539 return Err("payers do not add up".to_owned());
540 }
541 e.payers = payers;
542 contributions(e);
543 let e = &self.expenses[i];
544 out.push(format!(
545 "PAYERS {} {}{} = {}",
546 e.id,
547 money(e.total),
548 note(e.amount, &e.currency),
549 self.parts_text(&e.payers)
550 ));
551 }
552 "DELETE" => {
553 let i = self.expense_id(a[1])?;
554 self.check_frozen(&self.expenses[i])?;
555 self.expenses.remove(i);
556 out.push(format!("DELETE {}", a[1]));
557 }
558 "RECURRING" => {
559 let i = self.expense_id(a[1])?;
560 let e = &mut self.expenses[i];
561 e.interval = integer(a[2]);
562 if e.interval > 0 {
563 e.next = self.day + e.interval;
564 out.push(format!(
565 "RECURRING {} interval {} next {}",
566 e.id, e.interval, e.next
567 ));
568 } else {
569 out.push(format!("RECURRING {} stopped", e.id));
570 }
571 }
572 "SETTLE" => {
573 let from = self.member_id(a[1], true)?;
574 let to = self.member_id(a[2], true)?;
575 if from == to {
576 return Err(format!("same member {}", a[1]));
577 }
578 let (amount, currency) = amount_spec(a[3]);
579 let total = self.convert(amount, &currency)?;
580 if amount == 0 || total == 0 {
581 return Err("invalid amount".to_owned());
582 }
583 self.serial += 1;
584 out.push(format!(
585 "SETTLE {} -> {} {}{}",
586 a[1],
587 a[2],
588 money(total),
589 note(amount, &currency)
590 ));
591 self.settlements.push(Settlement {
592 from,
593 to,
594 amount,
595 total,
596 day: self.day,
597 serial: self.serial,
598 currency,
599 });
600 }
601 "BUDGET" => {
602 let value = amount(a[2]);
603 if value > 0 {
604 self.budgets.insert(a[1].to_owned(), value);
605 out.push(format!("BUDGET {} {}", a[1], money(value)));
606 } else {
607 self.budgets.remove(a[1]);
608 out.push(format!("BUDGET {} removed", a[1]));
609 }
610 }
611 "BALANCES" => {
612 out.push("BALANCES".to_owned());
613 let values = self.balances();
614 for i in self.ordered_members(true) {
615 out.push(format!(
616 " {} {}",
617 self.members[i].name,
618 signed_money(values[i])
619 ));
620 }
621 }
622 "OWES" => {
623 let id = self.member_id(a[1], false)?;
624 out.push(format!("OWES {}", a[1]));
625 let values = self.debts();
626 for i in self.ordered_members(false) {
627 if i == id || values[id][i] == 0 {
628 continue;
629 }
630 let (from, to) = if values[id][i] > 0 { (id, i) } else { (i, id) };
631 out.push(format!(
632 " {} owes {} {}",
633 self.members[from].name,
634 self.members[to].name,
635 money(values[from][to])
636 ));
637 }
638 }
639 "HISTORY" => {
640 let id = self.member_id(a[1], false)?;
641 let mut items: Vec<(Integer, Integer, Integer, String)> = Vec::new();
642 for e in &self.expenses {
643 if involved(e, id) {
644 items.push((
645 e.day,
646 e.serial,
647 net(e, self.members.len())[id],
648 e.id.clone(),
649 ));
650 }
651 }
652 for s in &self.settlements {
653 if s.from == id || s.to == id {
654 items.push((
655 s.day,
656 s.serial,
657 if s.from == id { s.total } else { -s.total },
658 format!("{}->{}", self.members[s.from].name, self.members[s.to].name),
659 ));
660 }
661 }
662 items.sort_by_key(|item| (item.0, item.1));
663 out.push(format!("HISTORY {}", a[1]));
664 let mut running = 0;
665 for (day, _, change, label) in items {
666 running += change;
667 out.push(format!(
668 " day {day} {label} {} balance {}",
669 signed_money(change),
670 signed_money(running)
671 ));
672 }
673 }
674 "MATRIX" => {
675 out.push("MATRIX".to_owned());
676 let order = self.ordered_members(true);
677 if order.is_empty() {
678 out.push(" (none)".to_owned());
679 } else {
680 let values = self.debts();
681 let l = order
682 .iter()
683 .map(|&i| self.members[i].name.len())
684 .max()
685 .unwrap();
686 let mut w = l;
687 for &i in &order {
688 for &j in &order {
689 if i != j {
690 w = w.max(money(values[i][j].max(0)).len());
691 }
692 }
693 }
694 let mut header = " ".repeat(l);
695 for &i in &order {
696 header.push_str(&format!(" {:>w$}", self.members[i].name));
697 }
698 out.push(header);
699 for &i in &order {
700 let mut row = format!("{:<l$}", self.members[i].name);
701 for &j in &order {
702 let cell = if i == j {
703 "-".to_owned()
704 } else {
705 money(values[i][j].max(0))
706 };
707 row.push_str(&format!(" {cell:>w$}"));
708 }
709 out.push(row);
710 }
711 }
712 }
713 "CATEGORY" => {
714 out.push("CATEGORY".to_owned());
715 let mut totals: BTreeMap<&str, Integer> = BTreeMap::new();
716 let mut total = 0;
717 for e in &self.expenses {
718 *totals.entry(&e.category).or_default() += e.total;
719 total += e.total;
720 }
721 let mut categories: Vec<&str> = totals.keys().copied().collect();
722 categories.sort_by(|&x, &y| totals[y].cmp(&totals[x]).then(x.cmp(y)));
723 for category in categories {
724 out.push(format!(
725 " {category} {} {}",
726 money(totals[category]),
727 percentage(totals[category], total)
728 ));
729 let mut shares = vec![0; self.members.len()];
730 for e in &self.expenses {
731 if e.category == category {
732 for p in &e.participants {
733 shares[p.member] += p.result;
734 }
735 }
736 }
737 let mut order = self.ordered_members(false);
738 order.sort_by(|&i, &j| shares[j].cmp(&shares[i]));
739 for i in order {
740 if shares[i] != 0 {
741 out.push(format!(
742 " {} {} {}",
743 self.members[i].name,
744 money(shares[i]),
745 percentage(shares[i], totals[category])
746 ));
747 }
748 }
749 }
750 }
751 "BUDGETS" => {
752 out.push("BUDGETS".to_owned());
753 for (category, &budget) in &self.budgets {
754 let spent = self.spent(category);
755 let suffix = if spent <= budget {
756 format!("left {}", money(budget - spent))
757 } else {
758 format!("over {}", money(spent - budget))
759 };
760 out.push(format!(
761 " {category} spent {} budget {} used {} {suffix}",
762 money(spent),
763 money(budget),
764 percentage(spent, budget)
765 ));
766 }
767 }
768 "TOTALS" => {
769 out.push("TOTALS".to_owned());
770 let count = self.members.len();
771 let (mut paid, mut shares, mut sent, mut received) = (
772 vec![0; count],
773 vec![0; count],
774 vec![0; count],
775 vec![0; count],
776 );
777 let (mut expense_total, mut settlement_total) = (0, 0);
778 for e in &self.expenses {
779 expense_total += e.total;
780 for p in &e.payers {
781 paid[p.member] += p.result;
782 }
783 for p in &e.participants {
784 shares[p.member] += p.result;
785 }
786 }
787 for s in &self.settlements {
788 settlement_total += s.total;
789 sent[s.from] += s.total;
790 received[s.to] += s.total;
791 }
792 for i in self.ordered_members(false) {
793 out.push(format!(
794 " {} paid {} share {} sent {} received {}",
795 self.members[i].name,
796 money(paid[i]),
797 money(shares[i]),
798 money(sent[i]),
799 money(received[i])
800 ));
801 }
802 for currency in self.currencies.keys() {
803 let (mut count, mut original, mut base) = (0, 0, 0);
804 for e in &self.expenses {
805 if &e.currency == currency {
806 count += 1;
807 original += e.amount;
808 base += e.total;
809 }
810 }
811 if count > 0 {
812 out.push(format!(
813 " currency {currency} expenses {count} amount {} base {}",
814 money(original),
815 money(base)
816 ));
817 }
818 }
819 out.push(format!(
820 " expenses {} total {} settlements {} total {}",
821 self.expenses.len(),
822 money(expense_total),
823 self.settlements.len(),
824 money(settlement_total)
825 ));
826 }
827 "MEMBERS" => {
828 out.push("MEMBERS".to_owned());
829 for i in self.ordered_members(false) {
830 let count = self.expenses.iter().filter(|e| involved(e, i)).count();
831 let m = &self.members[i];
832 out.push(format!(
833 " {} {} day {} expenses {count}",
834 m.name,
835 if m.active { "active" } else { "left" },
836 m.day
837 ));
838 }
839 }
840 "EXPENSES" => {
841 let member = if a[1] == "MEMBER" {
842 Some(self.member_id(a[2], false)?)
843 } else {
844 None
845 };
846 out.push(line.to_owned());
847 let (mut count, mut total) = (0, 0);
848 for i in self.chronological() {
849 let e = &self.expenses[i];
850 let matches = match a[1] {
851 "ALL" => true,
852 "CATEGORY" => e.category == a[2],
853 "MEMBER" => involved(e, member.unwrap()),
854 _ => e.day >= integer(a[2]) && e.day <= integer(a[3]),
855 };
856 if !matches {
857 continue;
858 }
859 let names = e
860 .payers
861 .iter()
862 .map(|p| self.members[p.member].name.as_str())
863 .collect::<Vec<_>>()
864 .join(",");
865 out.push(format!(
866 " day {} {} {} {} by {names}",
867 e.day,
868 e.id,
869 e.category,
870 money(e.total)
871 ));
872 count += 1;
873 total += e.total;
874 }
875 out.push(format!(" count {count} total {}", money(total)));
876 }
877 "PERIODS" => {
878 let k = integer(a[1]);
879 let mut groups: BTreeMap<Integer, (usize, Integer)> = BTreeMap::new();
880 for e in &self.expenses {
881 let group = groups.entry((e.day - 1) / k).or_default();
882 group.0 += 1;
883 group.1 += e.total;
884 }
885 out.push(format!("PERIODS {k}"));
886 for (i, (count, total)) in groups {
887 out.push(format!(
888 " days {}-{} expenses {count} total {}",
889 i * k + 1,
890 i * k + k,
891 money(total)
892 ));
893 }
894 }
895 "SHOW" => {
896 let e = &self.expenses[self.expense_id(a[1])?];
897 out.push(format!("SHOW {}", a[1]));
898 out.push(format!(
899 " day {} category {} mode {} total {}{}",
900 e.day,
901 e.category,
902 e.mode,
903 money(e.total),
904 note(e.amount, &e.currency)
905 ));
906 for p in &e.payers {
907 out.push(format!(
908 " paid {} {}",
909 self.members[p.member].name,
910 money(p.result)
911 ));
912 }
913 let units: Integer = e.participants.iter().map(|p| p.value).sum();
914 for p in &e.participants {
915 let suffix = match e.mode.as_str() {
916 "EXACT" => format!(" {}", money(p.value)),
917 "PERCENT" => format!(" {}%", money(p.value)),
918 "SHARES" => format!(" {}/{units}", p.value),
919 "ADJUST" => format!(" +{}", money(p.value)),
920 _ => String::new(),
921 };
922 out.push(format!(
923 " share {} {}{suffix}",
924 self.members[p.member].name,
925 money(p.result)
926 ));
927 }
928 if e.interval > 0 {
929 out.push(format!(
930 " recurring interval {} next {}",
931 e.interval, e.next
932 ));
933 }
934 }
935 "SIMPLIFY" => {
936 let values = self.balances();
937 let order: Vec<usize> = self
938 .ordered_members(false)
939 .into_iter()
940 .filter(|&i| values[i] != 0)
941 .collect();
942 if a[1] == "GREEDY" {
943 let transfers = greedy(&order, &values);
944 out.push(format!("SIMPLIFY GREEDY transfers {}", transfers.len()));
945 out.extend(self.transfer_lines(&transfers, " "));
946 } else {
947 let n = order.len();
948 if n > 12 {
949 return Err("too many members".to_owned());
950 }
951 let mut sums = vec![0; 1 << n];
952 for mask in 1usize..1 << n {
953 let bit = mask.trailing_zeros() as usize;
954 sums[mask] = sums[mask ^ (1 << bit)] + values[order[bit]];
955 }
956 let mut memo = vec![None; 1 << n];
957 memo[0] = Some(0);
958 let mut remaining = (1 << n) - 1;
959 let group_count = best_groups(remaining, &sums, &mut memo);
960 out.push(format!(
961 "SIMPLIFY OPTIMAL transfers {} groups {group_count}",
962 n - group_count
963 ));
964 while remaining > 0 {
965 let first = 1usize << remaining.trailing_zeros();
966 let (mut chosen, mut chosen_list) = (0, Vec::new());
967 let mut sub = remaining;
968 while sub > 0 {
969 if sub & first != 0
970 && sums[sub] == 0
971 && 1 + best_groups(remaining ^ sub, &sums, &mut memo)
972 == best_groups(remaining, &sums, &mut memo)
973 {
974 let list: Vec<usize> =
975 (0..n).filter(|&i| sub & (1 << i) != 0).collect();
976 if chosen == 0 || list < chosen_list {
977 chosen = sub;
978 chosen_list = list;
979 }
980 }
981 sub = (sub - 1) & remaining;
982 }
983 let group: Vec<usize> = chosen_list.iter().map(|&i| order[i]).collect();
984 let names = group
985 .iter()
986 .map(|&i| self.members[i].name.as_str())
987 .collect::<Vec<_>>()
988 .join(" ");
989 out.push(format!(" group {names}"));
990 out.extend(self.transfer_lines(&greedy(&group, &values), " "));
991 remaining ^= chosen;
992 }
993 }
994 }
995 _ => unreachable!(),
996 }
997 if [
998 "BALANCES", "OWES", "HISTORY", "CATEGORY", "BUDGETS", "MEMBERS", "PERIODS",
999 ]
1000 .contains(&command)
1001 && out.len() == 1
1002 {
1003 out.push(" (none)".to_owned());
1004 }
1005 Ok(out)
1006 }
1007}
