46func fail(message string) { panic(message) }
47func integer(s string) int64 {
48 n, _ := strconv.ParseInt(s, 10, 64)
49 return n
50}
51func amount(s string) int64 {
52 whole, fraction, _ := strings.Cut(s, ".")
53 return integer(whole)*100+integer(fraction)
54}
55func money(n int64) string { return fmt.Sprintf("%d.%02d", n/100, n%100) }
56func signedMoney(n int64) string {
57 if n > 0 { return "+"+money(n) }
58 if n < 0 { return "-"+money(-n) }
59 return "0.00"
60}
61func rateText(n int64) string { return fmt.Sprintf("%d.%04d", n/10000, n%10000) }
62func percentage(p, w int64) string { return money(p*10000/w)+"%" }
63func amountSpec(s string) (int64, string) {
64 value, currency, _ := strings.Cut(s, "@")
65 return amount(value), currency
66}
67func convert(value int64, currency string) int64 {
68 if currency == "" { return value }
69 rate, exists := currencies[currency]
70 if !exists { fail("unknown currency "+currency) }
71 return (value*rate+5000)/10000
72}
73func note(value int64, currency string) string {
74 if currency == "" { return "" }
75 return " ["+money(value)+" "+currency+"]"
76}
77func memberID(name string, active bool) int {
78 for i, m := range members {
79 if m.name == name {
80 if active && !m.active { fail("inactive member "+name) }
81 return i
82 }
83 }
84 fail("unknown member "+name)
85 return -1
86}
87func orderedMembers(active bool) []int {
88 var order []int
89 for i, m := range members {
90 if !active || m.active { order = append(order, i) }
91 }
92 sort.Slice(order, func(i, j int) bool { return members[order[i]].name < members[order[j]].name })
93 return order
94}
95func expenseID(id string) int {
96 for i, e := range expenses {
97 if e.id == id { return i }
98 }
99 fail("unknown expense "+id)
100 return -1
101}
102func checkFrozen(e expense) {
103 for _, list := range [][]part{e.payers, e.participants} {
104 for _, p := range list {
105 if !members[p.member].active { fail("inactive member "+members[p.member].name) }
106 }
107 }
108}
109func allActive(e expense) bool {
110 for _, list := range [][]part{e.payers, e.participants} {
111 for _, p := range list {
112 if !members[p.member].active { return false }
113 }
114 }
115 return true
116}
117func distribute(total int64, weights []int64) []int64 {
118 var sum int64
119 for _, w := range weights { sum += w }
120 result, remainders := make([]int64, len(weights)), make([]int64, len(weights))
121 order := make([]int, len(weights))
122 left := total
123 for i, w := range weights {
124 product := total*w
125 result[i], remainders[i], order[i] = product/sum, product%sum, i
126 left -= result[i]
127 }
128 sort.SliceStable(order, func(i, j int) bool { return remainders[order[i]] > remainders[order[j]] })
129 for i := int64(0); i < left; i++ { result[order[i]]++ }
130 return result
131}
132func equalDistribution(total int64, count int) []int64 {
133 result := make([]int64, count)
134 for i := range result {
135 result[i] = total/int64(count)
136 if int64(i) < total%int64(count) { result[i]++ }
137 }
138 return result
139}
140func contributions(e *expense) {
141 var weights []int64
142 for _, p := range e.payers { weights = append(weights, p.value) }
143 values := distribute(e.total, weights)
144 for i := range e.payers { e.payers[i].result = values[i] }
145}
146func compute(e *expense) {
147 e.total = convert(e.amount, e.currency)
148 var specified []int64
149 var sum int64
150 for _, p := range e.participants {
151 specified = append(specified, p.value)
152 sum += p.value
153 }
154 var portions []int64
155 switch e.mode {
156 case "EQUAL": portions = equalDistribution(e.amount, len(specified))
157 case "EXACT": portions = specified
158 case "ADJUST":
159 portions = equalDistribution(e.amount-sum, len(specified))
160 for i := range portions { portions[i] += specified[i] }
161 default: portions = distribute(e.amount, specified)
162 }
163 shares := distribute(e.total, portions)
164 for i := range e.participants { e.participants[i].result = shares[i] }
165 contributions(e)
166}
167func draftExpense(a []string, previous *expense) expense {
168 e := expense{}
169 if previous != nil { e = *previous }
170 payer := memberID(a[2], true)
171 value, currency := amountSpec(a[3])
172 total := convert(value, currency)
173 if value == 0 || total == 0 { fail("invalid amount") }
174 e.amount, e.currency, e.category, e.mode = value, currency, a[4], a[5]
175 e.payers = []part{{payer, value, 0}}
176 e.participants = nil
177 if e.mode == "EQUAL" && a[6] == "ALL" {
178 for _, i := range orderedMembers(true) { e.participants = append(e.participants, part{member: i}) }
179 } else {
180 seen := map[int]bool{}
181 for _, item := range a[6:] {
182 name, specified, found := strings.Cut(item, ":")
183 id := memberID(name, true)
184 if seen[id] { fail("duplicate member "+name) }
185 seen[id] = true
186 var value int64
187 if found {
188 if e.mode == "PERCENT" || e.mode == "SHARES" { value = integer(specified) } else { value = amount(specified) }
189 }
190 e.participants = append(e.participants, part{id, value, 0})
191 }
192 }
193 var sum int64
194 for _, p := range e.participants { sum += p.value }
195 if e.mode == "EXACT" && sum != e.amount || e.mode == "PERCENT" && sum != 10000 || e.mode == "SHARES" && sum == 0 || e.mode == "ADJUST" && sum > e.amount { fail("splits do not add up") }
196 compute(&e)
197 return e
198}
199func net(e expense) []int64 {
200 result := make([]int64, len(members))
201 for _, p := range e.payers { result[p.member] += p.result }
202 for _, p := range e.participants { result[p.member] -= p.result }
203 return result
204}
205func involved(e expense, id int) bool {
206 for _, list := range [][]part{e.payers, e.participants} {
207 for _, p := range list {
208 if p.member == id { return true }
209 }
210 }
211 return false
212}
213func balances() []int64 {
214 result := make([]int64, len(members))
215 for _, e := range expenses {
216 for i, v := range net(e) { result[i] += v }
217 }
218 for _, s := range settlements {
219 result[s.from] += s.total
220 result[s.to] -= s.total
221 }
222 return result
223}
224func debts() [][]int64 {
225 result := make([][]int64, len(members))
226 for i := range result { result[i] = make([]int64, len(members)) }
227 order := orderedMembers(false)
228 for _, e := range expenses {
229 change := net(e)
230 var borrowers, lenders []int
231 for _, i := range order {
232 if change[i] < 0 { borrowers = append(borrowers, i) }
233 if change[i] > 0 { lenders = append(lenders, i) }
234 }
235 for i, j := 0, 0; i < len(borrowers) && j < len(lenders); {
236 from, to := borrowers[i], lenders[j]
237 value := min(-change[from], change[to])
238 result[from][to] += value
239 change[from] += value
240 change[to] -= value
241 if change[from] == 0 { i++ }
242 if change[to] == 0 { j++ }
243 }
244 }
245 for _, s := range settlements { result[s.from][s.to] -= s.total }
246 for i := range result {
247 for j := i+1; j < len(result); j++ {
248 value := result[i][j]-result[j][i]
249 result[i][j], result[j][i] = value, -value
250 }
251 }
252 return result
253}
254func spent(category string) int64 {
255 var total int64
256 for _, e := range expenses {
257 if e.category == category { total += e.total }
258 }
259 return total
260}
261func warning(e expense) []string {
262 budget, exists := budgets[e.category]
263 total := spent(e.category)
264 if exists && total > budget { return []string{" WARNING "+e.category+" over budget by "+money(total-budget)} }
265 return nil
266}
267func partsText(parts []part) string {
268 var result []string
269 for _, p := range parts { result = append(result, members[p.member].name+" "+money(p.result)) }
270 return strings.Join(result, " + ")
271}
272func chronological() []int {
273 order := make([]int, len(expenses))
274 for i := range order { order[i] = i }
275 sort.Slice(order, func(i, j int) bool {
276 x, y := expenses[order[i]], expenses[order[j]]
277 if x.day != y.day { return x.day < y.day }
278 return x.serial < y.serial
279 })
280 return order
281}
282func greedy(group []int, original []int64) []transfer {
283 values := slices.Clone(original)
284 var result []transfer
285 for {
286 from, to := -1, -1
287 for _, i := range group {
288 if values[i] < 0 && (from < 0 || values[i] < values[from]) { from = i }
289 if values[i] > 0 && (to < 0 || values[i] > values[to]) { to = i }
290 }
291 if from < 0 { break }
292 value := min(-values[from], values[to])
293 result = append(result, transfer{from, to, value})
294 values[from] += value
295 values[to] -= value
296 }
297 return result
298}
299func transferLines(transfers []transfer, indent string) []string {
300 var result []string
301 for _, t := range transfers { result = append(result, indent+members[t.from].name+" -> "+members[t.to].name+" "+money(t.amount)) }
302 return result
303}
304func execute(line string) []string {
305 a := strings.Split(line, " ")
306 command := a[0]
307 var out []string
308 switch command {
309 case "DAY":
310 day := integer(a[1])
311 if day <= currentDay { fail("invalid day "+a[1]) }
312 out = append(out, "DAY "+a[1])
313 for {
314 selected := -1
315 for i, e := range expenses {
316 if e.interval > 0 && e.next <= day && (selected < 0 || e.next < expenses[selected].next || e.next == expenses[selected].next && e.serial < expenses[selected].serial) { selected = i }
317 }
318 if selected < 0 { break }
319 original := &expenses[selected]
320 if !allActive(*original) || convert(original.amount, original.currency) == 0 {
321 out = append(out, " CANCEL "+original.id)
322 original.interval = 0
323 continue
324 }
325 copy := *original
326 copy.payers = slices.Clone(original.payers)
327 copy.participants = slices.Clone(original.participants)
328 copy.id = original.id+"#"+strconv.FormatInt(original.copies+2, 10)
329 copy.day = original.next
330 serial++
331 copy.serial = serial
332 copy.interval, copy.next, copy.copies = 0, 0, 0
333 original.copies++
334 original.next += original.interval
335 compute(&copy)
336 used[copy.id] = true
337 expenses = append(expenses, copy)
338 out = append(out, " RECUR "+copy.id+" "+money(copy.total))
339 out = append(out, warning(copy)...)
340 }
341 currentDay = day
342 case "MEMBER":
343 id := -1
344 for i, m := range members {
345 if m.name == a[1] { id = i }
346 }
347 if id >= 0 && members[id].active { fail("duplicate member "+a[1]) }
348 action := "joined"
349 if id < 0 { members = append(members, member{a[1], true, currentDay}) } else {
350 members[id].active, members[id].day = true, currentDay
351 action = "rejoined"
352 }
353 out = append(out, "MEMBER "+a[1]+" "+action)
354 case "LEAVE":
355 id := memberID(a[1], true)
356 if balances()[id] != 0 { fail("member has a balance "+a[1]) }
357 members[id].active, members[id].day = false, currentDay
358 out = append(out, "LEAVE "+a[1])
359 case "CURRENCY":
360 value := integer(a[2])
361 result := "CURRENCY "+a[1]+" "+rateText(value)
362 if old, exists := currencies[a[1]]; exists { result += " was "+rateText(old) }
363 currencies[a[1]] = value
364 out = append(out, result)
365 case "EXPENSE", "EDIT":
366 id := -1
367 if command == "EXPENSE" {
368 if used[a[1]] { fail("duplicate id "+a[1]) }
369 } else {
370 id = expenseID(a[1])
371 checkFrozen(expenses[id])
372 }
373 var previous *expense
374 if id >= 0 { previous = &expenses[id] }
375 e := draftExpense(a, previous)
376 if id < 0 {
377 e.id, e.day = a[1], currentDay
378 serial++
379 e.serial = serial
380 used[e.id] = true
381 expenses = append(expenses, e)
382 } else { expenses[id] = e }
383 out = append(out, command+" "+e.id+" "+money(e.total)+note(e.amount, e.currency)+" = "+partsText(e.participants))
384 out = append(out, warning(e)...)
385 case "PAYERS":
386 id := expenseID(a[1])
387 checkFrozen(expenses[id])
388 var payers []part
389 seen := map[int]bool{}
390 var sum int64
391 for _, item := range a[2:] {
392 name, payment, _ := strings.Cut(item, ":")
393 member := memberID(name, true)
394 if seen[member] { fail("duplicate member "+name) }
395 seen[member] = true
396 value := amount(payment)
397 sum += value
398 payers = append(payers, part{member, value, 0})
399 }
400 e := &expenses[id]
401 if sum != e.amount { fail("payers do not add up") }
402 e.payers = payers
403 contributions(e)
404 out = append(out, "PAYERS "+e.id+" "+money(e.total)+note(e.amount, e.currency)+" = "+partsText(e.payers))
405 case "DELETE":
406 id := expenseID(a[1])
407 checkFrozen(expenses[id])
408 expenses = slices.Delete(expenses, id, id+1)
409 out = append(out, "DELETE "+a[1])
410 case "RECURRING":
411 e := &expenses[expenseID(a[1])]
412 e.interval = integer(a[2])
413 if e.interval > 0 {
414 e.next = currentDay+e.interval
415 out = append(out, fmt.Sprintf("RECURRING %s interval %d next %d", e.id, e.interval, e.next))
416 } else { out = append(out, "RECURRING "+e.id+" stopped") }
417 case "SETTLE":
418 from, to := memberID(a[1], true), memberID(a[2], true)
419 if from == to { fail("same member "+a[1]) }
420 value, currency := amountSpec(a[3])
421 total := convert(value, currency)
422 if value == 0 || total == 0 { fail("invalid amount") }
423 serial++
424 settlements = append(settlements, settlement{from, to, value, total, currentDay, serial, currency})
425 out = append(out, "SETTLE "+a[1]+" -> "+a[2]+" "+money(total)+note(value, currency))
426 case "BUDGET":
427 value := amount(a[2])
428 if value > 0 {
429 budgets[a[1]] = value
430 out = append(out, "BUDGET "+a[1]+" "+money(value))
431 } else {
432 delete(budgets, a[1])
433 out = append(out, "BUDGET "+a[1]+" removed")
434 }
435 case "BALANCES":
436 out = append(out, "BALANCES")
437 values := balances()
438 for _, i := range orderedMembers(true) { out = append(out, " "+members[i].name+" "+signedMoney(values[i])) }
439 case "OWES":
440 id := memberID(a[1], false)
441 out = append(out, "OWES "+a[1])
442 values := debts()
443 for _, i := range orderedMembers(false) {
444 if i == id || values[id][i] == 0 { continue }
445 from, to := id, i
446 if values[id][i] < 0 { from, to = i, id }
447 out = append(out, " "+members[from].name+" owes "+members[to].name+" "+money(values[from][to]))
448 }
449 case "HISTORY":
450 id := memberID(a[1], false)
451 type item struct {
452 day, serial, change int64
453 label string
454 }
455 var items []item
456 for _, e := range expenses {
457 if involved(e, id) { items = append(items, item{e.day, e.serial, net(e)[id], e.id}) }
458 }
459 for _, s := range settlements {
460 if s.from == id || s.to == id {
461 change := s.total
462 if s.to == id { change = -change }
463 items = append(items, item{s.day, s.serial, change, members[s.from].name+"->"+members[s.to].name})
464 }
465 }
466 sort.Slice(items, func(i, j int) bool {
467 if items[i].day != items[j].day { return items[i].day < items[j].day }
468 return items[i].serial < items[j].serial
469 })
470 out = append(out, "HISTORY "+a[1])
471 var running int64
472 for _, item := range items {
473 running += item.change
474 out = append(out, fmt.Sprintf(" day %d %s %s balance %s", item.day, item.label, signedMoney(item.change), signedMoney(running)))
475 }
476 case "MATRIX":
477 out = append(out, "MATRIX")
478 order := orderedMembers(true)
479 if len(order) == 0 {
480 out = append(out, " (none)")
481 break
482 }
483 values := debts()
484 l := 0
485 for _, i := range order { l = max(l, len(members[i].name)) }
486 w := l
487 for _, i := range order {
488 for _, j := range order {
489 if i != j {
490 w = max(w, len(money(max(int64(0), values[i][j]))))
491 }
492 }
493 }
494 header := strings.Repeat(" ", l)
495 for _, i := range order { header += fmt.Sprintf(" %*s", w, members[i].name) }
496 out = append(out, header)
497 for _, i := range order {
498 row := fmt.Sprintf("%-*s", l, members[i].name)
499 for _, j := range order {
500 cell := "-"
501 if i != j { cell = money(max(int64(0), values[i][j])) }
502 row += fmt.Sprintf(" %*s", w, cell)
503 }
504 out = append(out, row)
505 }
506 case "CATEGORY":
507 out = append(out, "CATEGORY")
508 totals := map[string]int64{}
509 var total int64
510 for _, e := range expenses {
511 totals[e.category] += e.total
512 total += e.total
513 }
514 categories := slices.Sorted(maps.Keys(totals))
515 sort.SliceStable(categories, func(i, j int) bool { return totals[categories[i]] > totals[categories[j]] })
516 for _, category := range categories {
517 out = append(out, " "+category+" "+money(totals[category])+" "+percentage(totals[category], total))
518 shares := make([]int64, len(members))
519 for _, e := range expenses {
520 if e.category == category {
521 for _, p := range e.participants {
522 shares[p.member] += p.result
523 }
524 }
525 }
526 order := orderedMembers(false)
527 sort.SliceStable(order, func(i, j int) bool { return shares[order[i]] > shares[order[j]] })
528 for _, i := range order {
529 if shares[i] != 0 { out = append(out, " "+members[i].name+" "+money(shares[i])+" "+percentage(shares[i], totals[category])) }
530 }
531 }
532 case "BUDGETS":
533 out = append(out, "BUDGETS")
534 for _, category := range slices.Sorted(maps.Keys(budgets)) {
535 s, b := spent(category), budgets[category]
536 suffix := " left "+money(b-s)
537 if s > b { suffix = " over "+money(s-b) }
538 out = append(out, " "+category+" spent "+money(s)+" budget "+money(b)+" used "+percentage(s, b)+suffix)
539 }
540 case "TOTALS":
541 out = append(out, "TOTALS")
542 paid, shares := make([]int64, len(members)), make([]int64, len(members))
543 sent, received := make([]int64, len(members)), make([]int64, len(members))
544 var expenseTotal, settlementTotal int64
545 for _, e := range expenses {
546 expenseTotal += e.total
547 for _, p := range e.payers { paid[p.member] += p.result }
548 for _, p := range e.participants { shares[p.member] += p.result }
549 }
550 for _, s := range settlements {
551 settlementTotal += s.total
552 sent[s.from] += s.total
553 received[s.to] += s.total
554 }
555 for _, i := range orderedMembers(false) { out = append(out, " "+members[i].name+" paid "+money(paid[i])+" share "+money(shares[i])+" sent "+money(sent[i])+" received "+money(received[i])) }
556 for _, currency := range slices.Sorted(maps.Keys(currencies)) {
557 count := 0
558 var original, base int64
559 for _, e := range expenses {
560 if e.currency == currency {
561 count++
562 original += e.amount
563 base += e.total
564 }
565 }
566 if count > 0 { out = append(out, fmt.Sprintf(" currency %s expenses %d amount %s base %s", currency, count, money(original), money(base))) }
567 }
568 out = append(out, fmt.Sprintf(" expenses %d total %s settlements %d total %s", len(expenses), money(expenseTotal), len(settlements), money(settlementTotal)))
569 case "MEMBERS":
570 out = append(out, "MEMBERS")
571 for _, i := range orderedMembers(false) {
572 count := 0
573 for _, e := range expenses {
574 if involved(e, i) { count++ }
575 }
576 status := "left"
577 if members[i].active { status = "active" }
578 out = append(out, fmt.Sprintf(" %s %s day %d expenses %d", members[i].name, status, members[i].day, count))
579 }
580 case "EXPENSES":
581 id := -1
582 if a[1] == "MEMBER" { id = memberID(a[2], false) }
583 out = append(out, line)
584 count := 0
585 var total int64
586 for _, i := range chronological() {
587 e := expenses[i]
588 matches := a[1] == "ALL" || a[1] == "CATEGORY" && e.category == a[2] || a[1] == "MEMBER" && involved(e, id) || a[1] == "DAYS" && e.day >= integer(a[2]) && e.day <= integer(a[3])
589 if !matches { continue }
590 var names []string
591 for _, p := range e.payers { names = append(names, members[p.member].name) }
592 out = append(out, fmt.Sprintf(" day %d %s %s %s by %s", e.day, e.id, e.category, money(e.total), strings.Join(names, ",")))
593 count++
594 total += e.total
595 }
596 out = append(out, fmt.Sprintf(" count %d total %s", count, money(total)))
597 case "PERIODS":
598 k := integer(a[1])
599 type group struct {
600 count int
601 total int64
602 }
603 groups := map[int64]group{}
604 for _, e := range expenses {
605 i := (e.day-1)/k
606 g := groups[i]
607 g.count++
608 g.total += e.total
609 groups[i] = g
610 }
611 order := slices.Sorted(maps.Keys(groups))
612 out = append(out, "PERIODS "+a[1])
613 for _, i := range order { out = append(out, fmt.Sprintf(" days %d-%d expenses %d total %s", i*k+1, i*k+k, groups[i].count, money(groups[i].total))) }
614 case "SHOW":
615 e := expenses[expenseID(a[1])]
616 out = append(out, "SHOW "+a[1], fmt.Sprintf(" day %d category %s mode %s total %s%s", e.day, e.category, e.mode, money(e.total), note(e.amount, e.currency)))
617 for _, p := range e.payers { out = append(out, " paid "+members[p.member].name+" "+money(p.result)) }
618 var units int64
619 for _, p := range e.participants { units += p.value }
620 for _, p := range e.participants {
621 suffix := ""
622 switch e.mode {
623 case "EXACT": suffix = " "+money(p.value)
624 case "PERCENT": suffix = " "+money(p.value)+"%"
625 case "SHARES": suffix = fmt.Sprintf(" %d/%d", p.value, units)
626 case "ADJUST": suffix = " +"+money(p.value)
627 }
628 out = append(out, " share "+members[p.member].name+" "+money(p.result)+suffix)
629 }
630 if e.interval > 0 { out = append(out, fmt.Sprintf(" recurring interval %d next %d", e.interval, e.next)) }
631 case "SIMPLIFY":
632 values := balances()
633 var order []int
634 for _, i := range orderedMembers(false) {
635 if values[i] != 0 { order = append(order, i) }
636 }
637 if a[1] == "GREEDY" {
638 transfers := greedy(order, values)
639 out = append(out, fmt.Sprintf("SIMPLIFY GREEDY transfers %d", len(transfers)))
640 out = append(out, transferLines(transfers, " ")...)
641 } else {
642 n := len(order)
643 if n > 12 { fail("too many members") }
644 sums := make([]int64, 1<<n)
645 for mask := 1; mask < 1<<n; mask++ {
646 bit := 0
647 for mask&(1<<bit) == 0 { bit++ }
648 sums[mask] = sums[mask^(1<<bit)]+values[order[bit]]
649 }
650 memo := make([]int, 1<<n)
651 for i := 1; i < len(memo); i++ { memo[i] = -1 }
652 var best func(int) int
653 best = func(mask int) int {
654 if memo[mask] >= 0 { return memo[mask] }
655 first, result := mask & -mask, 0
656 for sub := mask; sub > 0; sub = (sub-1)&mask {
657 if sub&first != 0 && sums[sub] == 0 { result = max(result, 1+best(mask^sub)) }
658 }
659 memo[mask] = result
660 return result
661 }
662 remaining := (1 << n) - 1
663 groupCount := best(remaining)
664 out = append(out, fmt.Sprintf("SIMPLIFY OPTIMAL transfers %d groups %d", n-groupCount, groupCount))
665 for remaining > 0 {
666 first, chosen := remaining & -remaining, 0
667 var chosenList []int
668 for sub := remaining; sub > 0; sub = (sub-1)&remaining {
669 if sub&first == 0 || sums[sub] != 0 || 1+best(remaining^sub) != best(remaining) { continue }
670 var list []int
671 for i := 0; i < n; i++ {
672 if sub&(1<<i) != 0 { list = append(list, i) }
673 }
674 if chosen == 0 || slices.Compare(list, chosenList) < 0 { chosen, chosenList = sub, list }
675 }
676 var group []int
677 var names []string
678 for _, i := range chosenList {
679 group = append(group, order[i])
680 names = append(names, members[order[i]].name)
681 }
682 out = append(out, " group "+strings.Join(names, " "))
683 out = append(out, transferLines(greedy(group, values), " ")...)
684 remaining ^= chosen
685 }
686 }
687 }
688 if strings.Contains(" BALANCES OWES HISTORY CATEGORY BUDGETS MEMBERS PERIODS ", " "+command+" ") && len(out) == 1 { out = append(out, " (none)") }
689 return out
690}
