30 static void Fail(string message) { throw new Problem(message); }
31 static long Money(string s) => long.Parse(s.Replace(".",""),CultureInfo.InvariantCulture);
32 static Amount ParseAmount(string s) {
33 string[] parts = s.Split('@');
34 return new Amount(Money(parts[0]),parts.Length == 2 ? parts[1] : "");
35 }
36 static string Fmt(long n) => $"{n/100}.{n%100:D2}";
37 static string Signed(long n) => n > 0 ? "+"+Fmt(n) : n < 0 ? "-"+Fmt(-n) : "0.00";
38 static string Rate(long n) => $"{n/10000}.{n%10000:D4}";
39 static string Percent(long p, long w) { return Fmt(p*10000/w)+"%"; }
40 static string Note(Amount a) { return a.Currency.Length == 0 ? "" : " ["+Fmt(a.Value)+" "+a.Currency+"]"; }
41 void Registered(string name) {
42 if (!members.ContainsKey(name)) Fail("unknown member "+name);
43 }
44 void Active(string name) {
45 Registered(name);
46 if (!members[name].Active) Fail("inactive member "+name);
47 }
48 long ConvertAmount(Amount a) {
49 if (a.Currency.Length == 0) return a.Value;
50 if (!currencies.ContainsKey(a.Currency)) Fail("unknown currency "+a.Currency);
51 return (a.Value*currencies[a.Currency]+5000)/10000;
52 }
53 Expense GetExpense(string id) {
54 Expense e = expenses.GetValueOrDefault(id);
55 if (e == null) Fail("unknown expense "+id);
56 return e;
57 }
58 void Thawed(Expense e) {
59 foreach (Part p in e.payers) {
60 if (!members[p.Name].Active) Fail("inactive member "+p.Name);
61 }
62 foreach (Part p in e.participants) {
63 if (!members[p.Name].Active) Fail("inactive member "+p.Name);
64 }
65 }
66 static long[] Distribute(long total, List<long> weights) {
67 long sum = weights.Sum();
68 long[] result = new long[weights.Count], remainders = new long[weights.Count];
69 List<int> order = new();
70 long allocated = 0;
71 for (int i = 0; i < weights.Count; i++) {
72 result[i] = total*weights[i]/sum;
73 remainders[i] = total*weights[i]%sum;
74 allocated += result[i];
75 order.Add(i);
76 }
77 order = order.OrderByDescending(i => remainders[i]).ThenBy(i => i).ToList();
78 for (int i = 0; i < total-allocated; i++) result[order[i]]++;
79 return result;
80 }
81 static List<long> Weights(List<Part> parts) { return parts.Select(p => p.Value).ToList(); }
82 void Compute(Expense e) {
83 long a = e.amount.Value;
84 e.total = ConvertAmount(e.amount);
85 long sum = e.participants.Sum(p => p.Value);
86 long[] portions = new long[e.participants.Count];
87 if (e.mode.Equals("EXACT")) {
88 if (sum != a) Fail("splits do not add up");
89 portions = e.participants.Select(p => p.Value).ToArray();
90 } else if (e.mode.Equals("PERCENT") || e.mode.Equals("SHARES")) {
91 if (e.mode.Equals("PERCENT") && sum != 10000 || e.mode.Equals("SHARES") && sum == 0) Fail("splits do not add up");
92 portions = Distribute(a,Weights(e.participants));
93 } else {
94 long extras = e.mode.Equals("ADJUST") ? sum : 0;
95 if (extras > a) Fail("splits do not add up");
96 int k = portions.Length;
97 for (int i = 0; i < k; i++) {
98 portions[i] = (a-extras)/k+(i < (a-extras)%k ? 1 : 0)+(e.mode.Equals("ADJUST") ? e.participants[i].Value : 0);
99 }
100 }
101 e.shares = Distribute(e.total,portions.ToList());
102 e.contributions = Distribute(e.total,Weights(e.payers));
103 }
104 Expense Build(string[] tokens) {
105 Active(tokens[2]);
106 Expense e = new Expense();
107 e.amount = ParseAmount(tokens[3]);
108 long total = ConvertAmount(e.amount);
109 if (e.amount.Value == 0 || total == 0) Fail("invalid amount");
110 e.category = tokens[4];
111 e.mode = tokens[5];
112 e.payers = new List<Part> { new Part(tokens[2],e.amount.Value) };
113 e.participants = new();
114 HashSet<string> seen = new(StringComparer.Ordinal);
115 if (e.mode.Equals("EQUAL") && tokens[6].Equals("ALL")) {
116 foreach (var member in members) {
117 if (member.Value.Active) e.participants.Add(new Part(member.Key,0));
118 }
119 } else {
120 for (int i = 6; i < tokens.Length; i++) {
121 string[] part = tokens[i].Split(':');
122 Active(part[0]);
123 if (!seen.Add(part[0])) Fail("duplicate member "+part[0]);
124 long value = part.Length == 1 ? 0 : e.mode.Equals("EXACT") || e.mode.Equals("ADJUST") ? Money(part[1]) : long.Parse(part[1]);
125 e.participants.Add(new Part(part[0],value));
126 }
127 }
128 Compute(e);
129 return e;
130 }
131 List<Expense> Chronological() {
132 return expenses.Values.OrderBy(e => e.day).ThenBy(e => e.record).ToList();
133 }
134 static bool Includes(Expense e, string name) {
135 return e.payers.Any(p => p.Name == name) || e.participants.Any(p => p.Name == name);
136 }
137 SortedDictionary<string,long> Net(Expense e) {
138 SortedDictionary<string,long> result = new(StringComparer.Ordinal);
139 for (int i = 0; i < e.payers.Count; i++) AddBalance(result,e.payers[i].Name,e.contributions[i]);
140 for (int i = 0; i < e.participants.Count; i++) AddBalance(result,e.participants[i].Name,-e.shares[i]);
141 return result;
142 }
143 static void AddBalance(IDictionary<string,long> values, string name, long change) {
144 values[name] = (values.TryGetValue(name,out long old) ? old : 0)+change;
145 }
146 SortedDictionary<string,long> Balances() {
147 SortedDictionary<string,long> result = new(StringComparer.Ordinal);
148 foreach (string name in members.Keys) result[name] = 0L;
149 foreach (Expense e in expenses.Values) {
150 foreach (var item in Net(e)) AddBalance(result,item.Key,item.Value);
151 }
152 foreach (Settlement s in settlements) {
153 AddBalance(result,s.From,s.Total);
154 AddBalance(result,s.To,-s.Total);
155 }
156 return result;
157 }
158 List<string> ActiveNames() { return members.Where(p => p.Value.Active).Select(p => p.Key).ToList(); }
159 SortedDictionary<string,long> CategoryTotals() {
160 SortedDictionary<string,long> result = new(StringComparer.Ordinal);
161 foreach (Expense e in expenses.Values) AddBalance(result,e.category,e.total);
162 return result;
163 }
164 void Warning(Expense e, List<string> output) {
165 long? budget = budgets.TryGetValue(e.category,out long limit) ? limit : null;
166 long spent = CategoryTotals().GetValueOrDefault(e.category);
167 if (budget != null && spent > budget.Value) output.Add(" WARNING "+e.category+" over budget by "+Fmt(spent-budget.Value));
168 }
169 static string Pairs(List<Part> people, long[] values) =>
170 string.Join(" + ",people.Zip(values,(p,value) => p.Name+" "+Fmt(value)));
171 static List<Transfer> Greedy(IDictionary<string,long> balances) {
172 SortedDictionary<string,long> work = new SortedDictionary<string,long>(balances,StringComparer.Ordinal);
173 List<Transfer> result = new();
174 while (work.Count > 0) {
175 var sender = work.MinBy(p => p.Value);
176 var receiver = work.MaxBy(p => p.Value);
177 long minimum = sender.Value, maximum = receiver.Value;
178 if (minimum >= 0) break;
179 string from = sender.Key, to = receiver.Key;
180 long value = Math.Min(-minimum,maximum);
181 result.Add(new Transfer(from,to,value));
182 work[from] = minimum+value;
183 work[to] = maximum-value;
184 }
185 return result;
186 }
187 long[][] Debts(List<string> names) {
188 int n = names.Count;
189 Dictionary<string,int> index = new(StringComparer.Ordinal);
190 for (int i = 0; i < n; i++) index[names[i]] = i;
191 long[][] result = Enumerable.Range(0,n).Select(_ => new long[n]).ToArray();
192 foreach (Expense e in expenses.Values) {
193 SortedDictionary<string,long> amounts = Net(e);
194 List<string> borrowers = amounts.Where(p => p.Value < 0).Select(p => p.Key).ToList();
195 List<string> lenders = amounts.Where(p => p.Value > 0).Select(p => p.Key).ToList();
196 int a = 0, b = 0;
197 while (a < borrowers.Count && b < lenders.Count) {
198 string x = borrowers[a], y = lenders[b];
199 long value = Math.Min(-amounts[x],amounts[y]);
200 int i = index[x], j = index[y];
201 result[i][j] += value;
202 result[j][i] -= value;
203 amounts[x] = amounts[x]+value;
204 amounts[y] = amounts[y]-value;
205 if (amounts[x] == 0) a++;
206 if (amounts[y] == 0) b++;
207 }
208 }
209 foreach (Settlement s in settlements) {
210 int i = index[s.From], j = index[s.To];
211 result[i][j] -= s.Total;
212 result[j][i] += s.Total;
213 }
214 return result;
215 }
216 static int[] Indexes(int mask, int n) => Enumerable.Range(0,n).Where(i => (mask&(1<<i)) != 0).ToArray();
217 List<string> Execute(string line) {
218 string[] t = line.Split(' ');
219 string command = t[0];
220 List<string> output = new();
221 switch (command) {
222 case "DAY": {
223 long d = long.Parse(t[1]);
224 if (d <= day) Fail("invalid day "+t[1]);
225 output.Add("DAY "+d);
226 while (true) {
227 Expense template = expenses.Values.Where(e => e.interval > 0 && e.next <= d)
228 .OrderBy(e => e.next).ThenBy(e => e.record).FirstOrDefault();
229 if (template == null) break;
230 bool active = template.payers.Concat(template.participants).All(p => members[p.Name].Active);
231 if (!active || ConvertAmount(template.amount) == 0) {
232 template.interval = 0;
233 output.Add(" CANCEL "+template.id);
234 } else {
235 Expense copy = new Expense();
236 copy.id = template.id+"#"+(template.copies+2);
237 copy.day = template.next;
238 copy.record = ++serial;
239 copy.amount = template.amount;
240 copy.category = template.category;
241 copy.mode = template.mode;
242 copy.payers = template.payers;
243 copy.participants = template.participants;
244 Compute(copy);
245 expenses[copy.id] = copy;
246 used.Add(copy.id);
247 template.copies++;
248 template.next += template.interval;
249 output.Add(" RECUR "+copy.id+" "+Fmt(copy.total));
250 Warning(copy,output);
251 }
252 }
253 day = d;
254 break;
255 }
256 case "MEMBER": {
257 Member old = members.GetValueOrDefault(t[1]);
258 if (old != null && old.Active) Fail("duplicate member "+t[1]);
259 members[t[1]] = new Member(true,day);
260 output.Add("MEMBER "+t[1]+(old == null ? " joined" : " rejoined"));
261 break;
262 }
263 case "LEAVE": {
264 Active(t[1]);
265 if (Balances()[t[1]] != 0) Fail("member has a balance "+t[1]);
266 members[t[1]] = new Member(false,day);
267 output.Add("LEAVE "+t[1]);
268 break;
269 }
270 case "CURRENCY": {
271 long value = long.Parse(t[2]);
272 long? old = currencies.TryGetValue(t[1],out long previous) ? previous : null;
273 currencies[t[1]] = value;
274 output.Add("CURRENCY "+t[1]+" "+Rate(value)+(old == null ? "" : " was "+Rate(old.Value)));
275 break;
276 }
277 case "EXPENSE":
278 case "EDIT": {
279 Expense old = null;
280 if (command.Equals("EXPENSE")) {
281 if (used.Contains(t[1])) Fail("duplicate id "+t[1]);
282 } else {
283 old = GetExpense(t[1]);
284 Thawed(old);
285 }
286 Expense e = Build(t);
287 e.id = t[1];
288 e.day = old == null ? day : old.day;
289 e.record = old == null ? ++serial : old.record;
290 if (old != null) {
291 e.interval = old.interval;
292 e.next = old.next;
293 e.copies = old.copies;
294 }
295 expenses[e.id] = e;
296 used.Add(e.id);
297 output.Add(command+" "+e.id+" "+Fmt(e.total)+Note(e.amount)+" = "+Pairs(e.participants,e.shares));
298 Warning(e,output);
299 break;
300 }
301 case "PAYERS": {
302 Expense e = GetExpense(t[1]);
303 Thawed(e);
304 List<Part> payers = new();
305 HashSet<string> seen = new(StringComparer.Ordinal);
306 long total = 0;
307 for (int i = 2; i < t.Length; i++) {
308 string[] p = t[i].Split(':');
309 Active(p[0]);
310 if (!seen.Add(p[0])) Fail("duplicate member "+p[0]);
311 long value = Money(p[1]);
312 payers.Add(new Part(p[0],value));
313 total += value;
314 }
315 if (total != e.amount.Value) Fail("payers do not add up");
316 long[] contributions = Distribute(e.total,Weights(payers));
317 e.payers = payers;
318 e.contributions = contributions;
319 output.Add("PAYERS "+e.id+" "+Fmt(e.total)+Note(e.amount)+" = "+Pairs(payers,contributions));
320 break;
321 }
322 case "DELETE": {
323 Expense e = GetExpense(t[1]);
324 Thawed(e);
325 expenses.Remove(e.id);
326 output.Add("DELETE "+e.id);
327 break;
328 }
329 case "RECURRING": {
330 Expense e = GetExpense(t[1]);
331 e.interval = long.Parse(t[2]);
332 e.next = day+e.interval;
333 output.Add("RECURRING "+e.id+(e.interval == 0 ? " stopped" : " interval "+e.interval+" next "+e.next));
334 break;
335 }
336 case "SETTLE": {
337 Active(t[1]);
338 Active(t[2]);
339 if (t[1].Equals(t[2])) Fail("same member "+t[1]);
340 Amount a = ParseAmount(t[3]);
341 long total = ConvertAmount(a);
342 if (a.Value == 0 || total == 0) Fail("invalid amount");
343 settlements.Add(new Settlement(day,++serial,t[1],t[2],a,total));
344 output.Add("SETTLE "+t[1]+" -> "+t[2]+" "+Fmt(total)+Note(a));
345 break;
346 }
347 case "BUDGET": {
348 long value = Money(t[2]);
349 if (value == 0) budgets.Remove(t[1]);
350 else budgets[t[1]] = value;
351 output.Add("BUDGET "+t[1]+(value == 0 ? " removed" : " "+Fmt(value)));
352 break;
353 }
354 case "BALANCES": {
355 output.Add("BALANCES");
356 IDictionary<string,long> balances = Balances();
357 foreach (string name in ActiveNames()) output.Add(" "+name+" "+Signed(balances[name]));
358 break;
359 }
360 case "OWES":
361 case "MATRIX": {
362 if (command.Equals("OWES")) Registered(t[1]);
363 List<string> names = members.Keys.ToList();
364 long[][] debt = Debts(names);
365 output.Add(command.Equals("OWES") ? "OWES "+t[1] : "MATRIX");
366 if (command.Equals("OWES")) {
367 int i = names.IndexOf(t[1]);
368 for (int j = 0; j < names.Count; j++) {
369 if (debt[i][j] > 0) output.Add(" "+t[1]+" owes "+names[j]+" "+Fmt(debt[i][j]));
370 if (debt[i][j] < 0) output.Add(" "+names[j]+" owes "+t[1]+" "+Fmt(-debt[i][j]));
371 }
372 } else {
373 List<string> active = ActiveNames();
374 if (active.Count == 0) output.Add(" (none)");
375 else {
376 int l = active.Max(name => name.Length), width = l;
377 foreach (string a in active) {
378 foreach (string b in active) width = Math.Max(width,a.Equals(b) ? 1 : Fmt(Math.Max(0,debt[names.IndexOf(a)][names.IndexOf(b)])).Length);
379 }
380 StringBuilder header = new StringBuilder(new string(' ',l));
381 foreach (string name in active) header.Append(" ").Append(name.PadLeft(width));
382 output.Add(header.ToString());
383 foreach (string a in active) {
384 StringBuilder row = new StringBuilder(a.PadRight(l));
385 foreach (string b in active) {
386 string cell = a.Equals(b) ? "-" : Fmt(Math.Max(0,debt[names.IndexOf(a)][names.IndexOf(b)]));
387 row.Append(" ").Append(cell.PadLeft(width));
388 }
389 output.Add(row.ToString());
390 }
391 }
392 }
393 break;
394 }
395 case "HISTORY": {
396 Registered(t[1]);
397 List<Entry> entries = new();
398 foreach (Expense e in expenses.Values) {
399 if (Includes(e,t[1])) entries.Add(new Entry(e.day,e.record,e.id,Net(e)[t[1]]));
400 }
401 foreach (Settlement s in settlements) {
402 if (s.From.Equals(t[1]) || s.To.Equals(t[1])) entries.Add(new Entry(s.Day,s.Record,s.From+"->"+s.To,s.From.Equals(t[1]) ? s.Total : -s.Total));
403 }
404 entries = entries.OrderBy(e => e.Day).ThenBy(e => e.Record).ToList();
405 output.Add("HISTORY "+t[1]);
406 long balance = 0;
407 foreach (Entry e in entries) {
408 balance += e.Change;
409 output.Add(" day "+e.Day+" "+e.Label+" "+Signed(e.Change)+" balance "+Signed(balance));
410 }
411 break;
412 }
413 case "CATEGORY": {
414 output.Add("CATEGORY");
415 SortedDictionary<string,long> totals = CategoryTotals();
416 long all = totals.Values.Sum();
417 List<string> categories = totals.Keys.OrderByDescending(name => totals[name]).ThenBy(name => name,StringComparer.Ordinal).ToList();
418 foreach (string category in categories) {
419 long total = totals[category];
420 output.Add(" "+category+" "+Fmt(total)+" "+Percent(total,all));
421 SortedDictionary<string,long> shares = new(StringComparer.Ordinal);
422 foreach (Expense e in expenses.Values) {
423 if (e.category.Equals(category)) {
424 for (int i = 0; i < e.participants.Count; i++) AddBalance(shares,e.participants[i].Name,e.shares[i]);
425 }
426 }
427 foreach (string name in shares.Keys.Where(name => shares[name] != 0).OrderByDescending(name => shares[name]).ThenBy(name => name,StringComparer.Ordinal).ToList()) {
428 output.Add(" "+name+" "+Fmt(shares[name])+" "+Percent(shares[name],total));
429 }
430 }
431 break;
432 }
433 case "BUDGETS": {
434 output.Add("BUDGETS");
435 SortedDictionary<string,long> totals = CategoryTotals();
436 foreach (var budget in budgets) {
437 long spent = totals.GetValueOrDefault(budget.Key), b = budget.Value;
438 output.Add(" "+budget.Key+" spent "+Fmt(spent)+" budget "+Fmt(b)+" used "+Percent(spent,b)+
439 (spent <= b ? " left "+Fmt(b-spent) : " over "+Fmt(spent-b)));
440 }
441 break;
442 }
443 case "TOTALS": {
444 output.Add("TOTALS");
445 foreach (string name in members.Keys) {
446 long paid = 0, share = 0, sent = 0, received = 0;
447 foreach (Expense e in expenses.Values) {
448 for (int i = 0; i < e.payers.Count; i++) {
449 if (e.payers[i].Name.Equals(name)) paid += e.contributions[i];
450 }
451 for (int i = 0; i < e.participants.Count; i++) {
452 if (e.participants[i].Name.Equals(name)) share += e.shares[i];
453 }
454 }
455 sent = settlements.Where(s => s.From == name).Sum(s => s.Total);
456 received = settlements.Where(s => s.To == name).Sum(s => s.Total);
457 output.Add(" "+name+" paid "+Fmt(paid)+" share "+Fmt(share)+" sent "+Fmt(sent)+" received "+Fmt(received));
458 }
459 foreach (string currency in currencies.Keys) {
460 List<Expense> group = expenses.Values.Where(e => e.amount.Currency == currency).ToList();
461 if (group.Count != 0) output.Add(" currency "+currency+" expenses "+group.Count+" amount "+Fmt(group.Sum(e => e.amount.Value))+" base "+Fmt(group.Sum(e => e.total)));
462 }
463 output.Add(" expenses "+expenses.Count+" total "+Fmt(expenses.Values.Sum(e => e.total))+
464 " settlements "+settlements.Count+" total "+Fmt(settlements.Sum(s => s.Total)));
465 break;
466 }
467 case "MEMBERS": {
468 output.Add("MEMBERS");
469 foreach (var m in members) {
470 long count = expenses.Values.Count(e => Includes(e,m.Key));
471 output.Add(" "+m.Key+(m.Value.Active ? " active" : " left")+" day "+m.Value.Day+" expenses "+count);
472 }
473 break;
474 }
475 case "EXPENSES": {
476 if (t[1].Equals("MEMBER")) Registered(t[2]);
477 output.Add(line);
478 long count = 0, total = 0;
479 foreach (Expense e in Chronological()) {
480 bool matches = t[1] switch {
481 "CATEGORY" => e.category.Equals(t[2]),
482 "MEMBER" => Includes(e,t[2]),
483 "DAYS" => e.day >= long.Parse(t[2]) && e.day <= long.Parse(t[3]),
484 _ => true
485 };
486 if (matches) {
487 output.Add(" day "+e.day+" "+e.id+" "+e.category+" "+Fmt(e.total)+" by "+string.Join(",",e.payers.Select(p => p.Name).ToList()));
488 count++;
489 total += e.total;
490 }
491 }
492 output.Add(" count "+count+" total "+Fmt(total));
493 break;
494 }
495 case "PERIODS": {
496 long k = long.Parse(t[1]);
497 output.Add("PERIODS "+k);
498 var groups = expenses.Values.GroupBy(e => (e.day-1)/k).OrderBy(g => g.Key);
499 foreach (var group in groups) {
500 long i = group.Key;
501 output.Add(" days "+(i*k+1)+"-"+(i*k+k)+" expenses "+group.Count()+" total "+Fmt(group.Sum(e => e.total)));
502 }
503 break;
504 }
505 case "SHOW": {
506 Expense e = GetExpense(t[1]);
507 output.Add("SHOW "+e.id);
508 output.Add(" day "+e.day+" category "+e.category+" mode "+e.mode+" total "+Fmt(e.total)+Note(e.amount));
509 for (int i = 0; i < e.payers.Count; i++) output.Add(" paid "+e.payers[i].Name+" "+Fmt(e.contributions[i]));
510 long sum = e.participants.Sum(p => p.Value);
511 for (int i = 0; i < e.participants.Count; i++) {
512 Part p = e.participants[i];
513 string suffix = e.mode switch {
514 "EXACT" => " "+Fmt(p.Value),
515 "PERCENT" => " "+Fmt(p.Value)+"%",
516 "SHARES" => " "+p.Value+"/"+sum,
517 "ADJUST" => " +"+Fmt(p.Value),
518 _ => ""
519 };
520 output.Add(" share "+p.Name+" "+Fmt(e.shares[i])+suffix);
521 }
522 if (e.interval > 0) output.Add(" recurring interval "+e.interval+" next "+e.next);
523 break;
524 }
525 case "SIMPLIFY": {
526 SortedDictionary<string,long> all = Balances();
527 foreach (string key in all.Keys.Where(key => all[key] == 0).ToList()) all.Remove(key);
528 if (t[1].Equals("GREEDY")) {
529 List<Transfer> transfers = Greedy(all);
530 output.Add("SIMPLIFY GREEDY transfers "+transfers.Count);
531 foreach (Transfer x in transfers) output.Add(" "+x.From+" -> "+x.To+" "+Fmt(x.Amount));
532 } else {
533 List<string> names = all.Keys.ToList();
534 int n = names.Count;
535 if (n > 12) Fail("too many members");
536 int size = 1<<n;
537 long[] sums = new long[size];
538 int[] dp = new int[size];
539 Array.Fill(dp,-1000);
540 dp[0] = 0;
541 for (int mask = 1; mask < size; mask++) {
542 int bit = System.Numerics.BitOperations.TrailingZeroCount((uint)mask);
543 sums[mask] = sums[mask^(1<<bit)]+all[names[bit]];
544 int first = mask&-mask;
545 for (int sub = mask; sub > 0; sub = (sub-1)&mask) {
546 if ((sub&first) != 0 && sums[sub] == 0) dp[mask] = Math.Max(dp[mask],1+dp[mask^sub]);
547 }
548 }
549 int remaining = size-1;
550 List<int> groups = new();
551 while (remaining != 0) {
552 int first = remaining&-remaining, best = 0;
553 for (int sub = remaining; sub > 0; sub = (sub-1)&remaining) {
554 if ((sub&first) != 0 && sums[sub] == 0 && 1+dp[remaining^sub] == dp[remaining] &&
555 (best == 0 || Indexes(sub,n).AsSpan().SequenceCompareTo(Indexes(best,n)) < 0)) best = sub;
556 }
557 groups.Add(best);
558 remaining ^= best;
559 }
560 List<List<Transfer>> transfers = new();
561 foreach (int group in groups) {
562 SortedDictionary<string,long> amounts = new(StringComparer.Ordinal);
563 foreach (int i in Indexes(group,n)) amounts[names[i]] = all[names[i]];
564 transfers.Add(Greedy(amounts));
565 }
566 output.Add("SIMPLIFY OPTIMAL transfers "+transfers.Sum(group => group.Count)+" groups "+groups.Count);
567 for (int i = 0; i < groups.Count; i++) {
568 List<string> groupNames = Indexes(groups[i],n).Select(index => names[index]).ToList();
569 output.Add(" group "+string.Join(" ",groupNames));
570 foreach (Transfer x in transfers[i]) output.Add(" "+x.From+" -> "+x.To+" "+Fmt(x.Amount));
571 }
572 }
573 break;
574 }
575 }
576 if (new HashSet<string> { "BALANCES","OWES","HISTORY","CATEGORY","BUDGETS","MEMBERS","PERIODS" }.Contains(command) && output.Count == 1) {
577 output.Add(" (none)");
578 }
579 return output;
580 }
