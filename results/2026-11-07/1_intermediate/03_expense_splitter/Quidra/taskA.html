30int[] largest_remainder(int total, int[] weights)
31 whole := weights.sum()
32 int[] result := []
33 int[] order := []
34 for: 0 <= k < int(#weights)
35 result = result.append(total * weights[k] / whole)
36 order = order.append(k)
37 p := k
38 while p > 0 and total * weights[k] % whole > total * weights[order[p - 1]] % whole
39 order[p] = order[p - 1]
40 p -= 1
41 order[p] = k
42 remainder := total - result.sum()
43 for: 0 <= k < remainder
44 result[order[k]] += 1
45 return result

101Expense computed(const Expense original, int rate)
102 result := original
103 result.total = converted_total(result.amount, rate)
104 portions := result.specified
105 if result.mode == "EQUAL"
106 portions = equal_distribution(result.amount, int(#result.participants))
107 elif result.mode == "PERCENT" or result.mode == "SHARES"
108 portions = largest_remainder(result.amount, result.specified)
109 elif result.mode == "ADJUST"
110 base := equal_distribution(result.amount - result.specified.sum(), int(#result.participants))
111 portions = base + result.specified
112 result.shares = largest_remainder(result.total, portions)
113 result.contributions = largest_remainder(result.total, result.payments)
114 return result

175int[][] debts_of(const Ledger state)
176 int[][] raw := array(#state.members, fill = array(#state.members, fill = 0))
177 names := member_names(state)
178 for expense := state.expenses[]
179 net := expense_net(state, expense)
180 int[] borrowers := []
181 int[] lenders := []
182 for name := names[]
183 k := member_id(state, name)
184 if net[k] < 0
185 borrowers = borrowers.append(k)
186 elif net[k] > 0
187 lenders = lenders.append(k)
188 b := 0
189 l := 0
190 while b < int(#borrowers) and l < int(#lenders)
191 sender := borrowers[b]
192 receiver := lenders[l]
193 amount := math.min(-net[sender], net[receiver])
194 raw[sender][receiver] += amount
195 net[sender] += amount
196 net[receiver] -= amount
197 if net[sender] == 0
198 b += 1
199 if net[receiver] == 0
200 l += 1
201 for settlement := state.settlements[]
202 raw[member_id(state, settlement.sender)][member_id(state, settlement.receiver)] -= settlement.total
203 int[][] result := array(#state.members, fill = array(#state.members, fill = 0))
204 for: 0 <= r < int(#state.members)
205 for: 0 <= c < int(#state.members)
206 result[r][c] = raw[r][c] - raw[c][r]
207 return result

235string[] greedy_transfers(string[] names, int[] amounts, string indent)
236 remaining := amounts
237 string[] result := []
238 while true
239 sender := -1
240 receiver := -1
241 for: 0 <= k < int(#names)
242 if remaining[k] < 0 and (sender < 0 or remaining[k] < remaining[sender])
243 sender = k
244 if remaining[k] > 0 and (receiver < 0 or remaining[k] > remaining[receiver])
245 receiver = k
246 if sender < 0
247 break
248 amount := math.min(-remaining[sender], remaining[receiver])
249 result = result.append(indent + names[sender] + " -> " + names[receiver] + " " + money(amount))
250 remaining[sender] += amount
251 remaining[receiver] -= amount
252 return result

269string[] simplify(const Ledger state, bool optimal)
270 balances := balances_of(state)
271 string[] names := []
272 int[] amounts := []
273 for name := member_names(state)[]
274 amount := balances[member_id(state, name)]
275 if amount != 0
276 names = names.append(name)
277 amounts = amounts.append(amount)
278 if not optimal
279 transfers := greedy_transfers(names, amounts, " ")
280 return ["SIMPLIFY GREEDY transfers " + string(#transfers)].concat(transfers)
281 count := int(#names)
282 if count > 12
283 return ["ERROR too many members"]
284 int[] powers := []
285 size := 1
286 for: 0 <= k < count
287 powers = powers.append(size)
288 size *= 2
289 sums := array(nat(size), fill = 0)
290 dp := array(nat(size), fill = 0)
291 for: 1 <= mask < size
292 best := 0
293 sum_found := false
294 for: 0 <= k < count
295 if mask / powers[k] % 2 == 1
296 prior := mask - powers[k]
297 best = math.max(best, dp[prior])
298 if not sum_found
299 sums[mask] = sums[prior] + amounts[k]
300 sum_found = true
301 dp[mask] = best + (if sums[mask] == 0 then 1 else 0)
302 mask := size - 1
303 string[] output := []
304 transfer_count := 0
305 groups := 0
306 while mask > 0
307 first := subset_indices(mask, powers)[0]
308 selected := -1
309 int[] best_members := []
310 for: 1 <= candidate < size
311 if candidate / powers[first] % 2 == 0 or sums[candidate] != 0
312 continue
313 if not subset_of(candidate, mask, powers)
314 continue
315 if 1 + dp[mask - candidate] != dp[mask]
316 continue
317 members := subset_indices(candidate, powers)
318 if selected < 0 or lex_before(members, best_members)
319 selected = candidate
320 best_members = members
321 string[] group_names := names[best_members]
322 int[] group_amounts := amounts[best_members]
323 output = output.append(" group " + group_names.join(" "))
324 transfers := greedy_transfers(group_names, group_amounts, " ")
325 output = output.concat(transfers)
326 transfer_count += int(#transfers)
327 groups += 1
328 mask -= selected
329 return ["SIMPLIFY OPTIMAL transfers " + string(transfer_count) + " groups " + string(groups)].concat(output)
