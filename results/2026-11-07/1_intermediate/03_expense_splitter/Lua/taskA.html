105local function parse_amount(text)
106 local whole, fraction = text:match("^(%d+)%.(%d%d)$")
107 return tonumber(whole) * 100 + tonumber(fraction)
108end

154local function equal_distribution(value, k)
155 local result = {}
156 for i = 1, k do
157 result[i] = value // k + (i <= value % k and 1 or 0)
158 end
159 return result
160end

162local function largest_remainder(value, weights)
163 local whole = total(weights)
164 local result, order = {}, {}
165 for i, weight in ipairs(weights) do
166 result[i] = (value * weight) // whole
167 order[i] = i
168 end
169 table.sort(order, function(a, b)
170 local ra, rb = (value * weights[a]) % whole, (value * weights[b]) % whole
171 if ra ~= rb then
172 return ra > rb
173 end
174 return a < b
175 end)
176 for i = 1, value - total(result) do
177 result[order[i]] = result[order[i]] + 1
178 end
179 return result
180end

182local function portions(expense)
183 if expense.mode == "EQUAL" then
184 return equal_distribution(expense.amount, #expense.values)
185 elseif expense.mode == "EXACT" then
186 return expense.values
187 elseif expense.mode == "PERCENT" or expense.mode == "SHARES" then
188 return largest_remainder(expense.amount, expense.values)
189 end
190 local result = equal_distribution(expense.amount - total(expense.values), #expense.values)
191 for i, extra in ipairs(expense.values) do
192 result[i] = result[i] + extra
193 end
194 return result
195end

197local function recompute(state, expense)
198 expense.total = converted(state, expense.amount, expense.currency)
199 expense.shares = largest_remainder(expense.total, portions(expense))
200 expense.contributions = largest_remainder(expense.total, expense.payments)
201 return expense
202end

204local function build_expense(state, id, payer, spec, category, mode, parts)
205 check_member(state, payer)
206 local amount, currency = amount_spec(spec)
207 local converted_total = converted(state, amount, currency)
208 if amount == 0 or converted_total == 0 then
209 fail("invalid amount")
210 end
211 local names, values = {}, {}
212 if mode == "EQUAL" and #parts == 1 and parts[1] == "ALL" then
213 names = active_names(state)
214 for i = 1, #names do
215 values[i] = 0
216 end
217 else
218 for _, part in ipairs(parts) do
219 local name, text = part:match("^([^:]+):(.+)$")
220 name = name or part
221 check_member(state, name)
222 if contains(names, name) then
223 fail("duplicate member " .. name)
224 end
225 local value = 0
226 if text then
227 if mode == "EXACT" or mode == "ADJUST" then
228 value = parse_amount(text)
229 else
230 value = tonumber(text)
231 end
232 end
233 names[#names + 1] = name
234 values[#values + 1] = value
235 end
236 end
237 local sum = total(values)
238 local valid = true
239 if mode == "EXACT" then
240 valid = sum == amount
241 elseif mode == "PERCENT" then
242 valid = sum == 10000
243 elseif mode == "SHARES" then
244 valid = sum >= 1
245 elseif mode == "ADJUST" then
246 valid = sum <= amount
247 end
248 if not valid then
249 fail("splits do not add up")
250 end
251 return recompute(state, {id = id, day = state.day, record = 0, payers = {payer}, payments = {amount},
252 amount = amount, currency = currency, category = category, mode = mode, participants = names,
253 values = values, total = converted_total, shares = {}, contributions = {}, interval = 0, next_day = 0, copies = 0})
254end

256local function expense_net(expense)
257 local result = {}
258 for _, name in ipairs(expense_names(expense)) do
259 result[name] = 0
260 end
261 for i, name in ipairs(expense.payers) do
262 result[name] = result[name] + expense.contributions[i]
263 end
264 for i, name in ipairs(expense.participants) do
265 result[name] = result[name] - expense.shares[i]
266 end
267 return result
268end

270local function balances(state)
271 local result = {}
272 for name in pairs(state.members) do
273 result[name] = 0
274 end
275 for _, expense in pairs(state.expenses) do
276 for name, net in pairs(expense_net(expense)) do
277 result[name] = result[name] + net
278 end
279 end
280 for _, settlement in ipairs(state.settlements) do
281 result[settlement.sender] = result[settlement.sender] + settlement.total
282 result[settlement.receiver] = result[settlement.receiver] - settlement.total
283 end
284 return result
285end

287local function greedy_transfers(names, original)
288 local remaining = {}
289 for _, name in ipairs(names) do
290 remaining[name] = original[name]
291 end
292 local output = {}
293 while #names > 0 do
294 local sender, receiver = names[1], names[1]
295 for _, name in ipairs(names) do
296 if remaining[name] < remaining[sender] then
297 sender = name
298 end
299 if remaining[name] > remaining[receiver] then
300 receiver = name
301 end
302 end
303 if remaining[sender] >= 0 then
304 break
305 end
306 local amount = math.min(-remaining[sender], remaining[receiver])
307 output[#output + 1] = {sender, receiver, amount}
308 remaining[sender] = remaining[sender] + amount
309 remaining[receiver] = remaining[receiver] - amount
310 end
311 return output
312end

314local function debts(state)
315 local result = {}
316 local function add(a, b, value)
317 result[a] = result[a] or {}
318 result[a][b] = (result[a][b] or 0) + value
319 end
320 for _, expense in pairs(state.expenses) do
321 local net = expense_net(expense)
322 local borrowers, lenders = {}, {}
323 for _, name in ipairs(sorted_keys(net)) do
324 if net[name] < 0 then
325 borrowers[#borrowers + 1] = name
326 elseif net[name] > 0 then
327 lenders[#lenders + 1] = name
328 end
329 end
330 local i, j = 1, 1
331 while i <= #borrowers and j <= #lenders do
332 local sender, receiver = borrowers[i], lenders[j]
333 local amount = math.min(-net[sender], net[receiver])
334 add(sender, receiver, amount)
335 net[sender] = net[sender] + amount
336 net[receiver] = net[receiver] - amount
337 if net[sender] == 0 then
338 i = i + 1
339 end
340 if net[receiver] == 0 then
341 j = j + 1
342 end
343 end
344 end
345 for _, s in ipairs(state.settlements) do
346 add(s.sender, s.receiver, -s.total)
347 end
348 return result
349end

394local function advance_day(state, day)
395 if day <= state.day then
396 fail("invalid day " .. day)
397 end
398 local output = {"DAY " .. day}
399 while true do
400 local template
401 for _, expense in pairs(state.expenses) do
402 if expense.interval > 0 and expense.next_day <= day then
403 if not template or expense.next_day < template.next_day or (expense.next_day == template.next_day and expense.record < template.record) then
404 template = expense
405 end
406 end
407 end
408 if not template then
409 break
410 end
411 local active = true
412 for _, name in ipairs(expense_names(template)) do
413 if not state.members[name].active then
414 active = false
415 end
416 end
417 if not active or converted(state, template.amount, template.currency) == 0 then
418 template.interval, template.next_day = 0, 0
419 output[#output + 1] = " CANCEL " .. template.id
420 else
421 local copy_id = template.id .. "#" .. (template.copies + 2)
422 local copy = copy_expense(template)
423 copy.id, copy.day = copy_id, template.next_day
424 state.serial = state.serial + 1
425 copy.record = state.serial
426 copy.interval, copy.next_day, copy.copies = 0, 0, 0
427 recompute(state, copy)
428 state.expenses[copy_id] = copy
429 state.used[copy_id] = true
430 template.copies = template.copies + 1
431 template.next_day = template.next_day + template.interval
432 output[#output + 1] = " RECUR " .. copy_id .. " " .. amount_text(copy.total)
433 local warnings = budget_warning(state, copy.category)
434 table.move(warnings,1,#warnings,#output+1,output)
435 end
436 end
437 state.day = day
438 return output
439end

441local function vector_less(a, b)
442 for i = 1, math.min(#a, #b) do
443 if a[i] ~= b[i] then
444 return a[i] < b[i]
445 end
446 end
447 return #a < #b
448end

450local function optimal_groups(names, bal)
451 local n = #names
452 local limit = 1 << n
453 local sums, bit_index = {[0] = 0}, {}
454 for i = 1, n do
455 bit_index[1 << (i - 1)] = i
456 end
457 for mask = 1, limit - 1 do
458 local bit = mask & -mask
459 sums[mask] = sums[mask ~ bit] + bal[names[bit_index[bit]]]
460 end
461 local memo = {[0] = 0}
462 local function best(mask)
463 if memo[mask] then
464 return memo[mask]
465 end
466 local first, value, subset = mask & -mask, 0, mask
467 while subset ~= 0 do
468 if (subset & first) ~= 0 and sums[subset] == 0 then
469 value = math.max(value, 1 + best(mask ~ subset))
470 end
471 subset = (subset - 1) & mask
472 end
473 memo[mask] = value
474 return value
475 end
476 local groups, mask = {}, limit - 1
477 while mask ~= 0 do
478 local first, selected, selected_indices = mask & -mask, 0, {}
479 local target, subset = best(mask), mask
480 while subset ~= 0 do
481 if (subset & first) ~= 0 and sums[subset] == 0 and 1 + best(mask ~ subset) == target then
482 local indices = {}
483 for i = 1, n do
484 if (subset & (1 << (i - 1))) ~= 0 then
485 indices[#indices + 1] = i
486 end
487 end
488 if selected == 0 or vector_less(indices, selected_indices) then
489 selected, selected_indices = subset, indices
490 end
491 end
492 subset = (subset - 1) & mask
493 end
494 local group = {}
495 for i, index in ipairs(selected_indices) do
496 group[i] = names[index]
497 end
498 groups[#groups + 1] = group
499 mask = mask ~ selected
500 end
501 return groups
502end

504local function ledger_summary(command, args, line, state)
505 local output = {command}
506 local names, expenses = sorted_keys(state.members), chronological(state)
507 if command == "BALANCES" then
508 local bal = balances(state)
509 for _, name in ipairs(active_names(state)) do
510 output[#output + 1] = " " .. name .. " " .. signed_text(bal[name])
511 end
512 elseif command == "OWES" then
513 local name = args[1]
514 check_member(state, name, false)
515 output[1] = "OWES " .. name
516 local map = debts(state)
517 for _, other in ipairs(names) do
518 if name ~= other then
519 local amount = net_debt(map, name, other)
520 if amount > 0 then
521 output[#output + 1] = " " .. name .. " owes " .. other .. " " .. amount_text(amount)
522 elseif amount < 0 then
523 output[#output + 1] = " " .. other .. " owes " .. name .. " " .. amount_text(-amount)
524 end
525 end
526 end
527 elseif command == "HISTORY" then
528 local name = args[1]
529 check_member(state, name, false)
530 output[1] = "HISTORY " .. name
531 local records = {}
532 for _, e in ipairs(expenses) do
533 if expense_has(e, name) then
534 records[#records + 1] = {e.day, e.record, e.id, expense_net(e)[name]}
535 end
536 end
537 for _, s in ipairs(state.settlements) do
538 if name == s.sender or name == s.receiver then
539 records[#records + 1] = {s.day, s.record, s.sender .. "->" .. s.receiver, name == s.sender and s.total or -s.total}
540 end
541 end
542 table.sort(records, function(a, b)
543 if a[1] ~= b[1] then
544 return a[1] < b[1]
545 end
546 return a[2] < b[2]
547 end)
548 local running = 0
549 for _, record in ipairs(records) do
550 local day, serial, label, change = table.unpack(record)
551 running = running + change
552 output[#output + 1] = " day " .. day .. " " .. label .. " " .. signed_text(change) .. " balance " .. signed_text(running)
553 end
554 elseif command == "MATRIX" then
555 local active = active_names(state)
556 if #active == 0 then
557 return {"MATRIX", " (none)"}
558 end
559 local map, cells, left, width = debts(state), {}, 0, 0
560 for _, name in ipairs(active) do
561 left = math.max(left, #name)
562 width = math.max(width, #name)
563 end
564 for i, a in ipairs(active) do
565 cells[i] = {}
566 for j, b in ipairs(active) do
567 local cell = a == b and "-" or amount_text(math.max(0, net_debt(map, a, b)))
568 cells[i][j] = cell
569 width = math.max(width, #cell)
570 end
571 end
572 local header = string.rep(" ", left)
573 for _, name in ipairs(active) do
574 header = header .. " " .. string.format("%"..width.."s",name)
575 end
576 output[#output + 1] = header
577 for i, name in ipairs(active) do
578 local row = string.format("%-"..left.."s",name)
579 for _, cell in ipairs(cells[i]) do
580 row = row .. " " .. string.format("%"..width.."s",cell)
581 end
582 output[#output + 1] = row
583 end
584 elseif command == "CATEGORY" then
585 local totals, whole = category_totals(state), 0
586 for _, value in pairs(totals) do
587 whole = whole + value
588 end
589 local categories = sorted_keys(totals)
590 table.sort(categories, function(a, b)
591 if totals[a] ~= totals[b] then
592 return totals[a] > totals[b]
593 end
594 return a < b
595 end)
596 for _, category in ipairs(categories) do
597 local value = totals[category]
598 output[#output + 1] = " " .. category .. " " .. amount_text(value) .. " " .. percentage(value, whole)
599 local shares = {}
600 for _, name in ipairs(names) do
601 shares[name] = 0
602 end
603 for _, e in ipairs(expenses) do
604 if e.category == category then
605 for i, name in ipairs(e.participants) do
606 shares[name] = shares[name] + e.shares[i]
607 end
608 end
609 end
610 local people = {}
611 for _, name in ipairs(names) do
612 if shares[name] > 0 then
613 people[#people + 1] = name
614 end
615 end
616 table.sort(people, function(a, b)
617 if shares[a] ~= shares[b] then
618 return shares[a] > shares[b]
619 end
620 return a < b
621 end)
622 for _, name in ipairs(people) do
623 output[#output + 1] = " " .. name .. " " .. amount_text(shares[name]) .. " " .. percentage(shares[name], value)
624 end
625 end
626 elseif command == "BUDGETS" then
627 local totals = category_totals(state)
628 for _, category in ipairs(sorted_keys(state.budgets)) do
629 local spent, budget = totals[category] or 0, state.budgets[category]
630 local label = spent <= budget and "left" or "over"
631 output[#output + 1] = " " .. category .. " spent " .. amount_text(spent) .. " budget " .. amount_text(budget) .. " used " .. percentage(spent, budget) .. " " .. label .. " " .. amount_text(math.abs(budget - spent))
632 end
633 elseif command == "TOTALS" then
634 for _, name in ipairs(names) do
635 local paid, share, sent, received = 0, 0, 0, 0
636 for _, e in ipairs(expenses) do
637 for i, payer in ipairs(e.payers) do
638 if payer == name then
639 paid = paid + e.contributions[i]
640 end
641 end
642 for i, person in ipairs(e.participants) do
643 if person == name then
644 share = share + e.shares[i]
645 end
646 end
647 end
648 for _, s in ipairs(state.settlements) do
649 if s.sender == name then
650 sent = sent + s.total
651 end
652 if s.receiver == name then
653 received = received + s.total
654 end
655 end
656 output[#output + 1] = " " .. name .. " paid " .. amount_text(paid) .. " share " .. amount_text(share) .. " sent " .. amount_text(sent) .. " received " .. amount_text(received)
657 end
658 local currencies, expense_total, settlement_total = {}, 0, 0
659 for _, e in ipairs(expenses) do
660 expense_total = expense_total + e.total
661 if e.currency ~= "" then
662 local value = currencies[e.currency] or {count = 0, amount = 0, total = 0}
663 value.count = value.count + 1
664 value.amount = value.amount + e.amount
665 value.total = value.total + e.total
666 currencies[e.currency] = value
667 end
668 end
669 for _, currency in ipairs(sorted_keys(currencies)) do
670 local value = currencies[currency]
671 output[#output + 1] = " currency " .. currency .. " expenses " .. value.count .. " amount " .. amount_text(value.amount) .. " base " .. amount_text(value.total)
672 end
673 for _, s in ipairs(state.settlements) do
674 settlement_total = settlement_total + s.total
675 end
676 output[#output + 1] = " expenses " .. #expenses .. " total " .. amount_text(expense_total) .. " settlements " .. #state.settlements .. " total " .. amount_text(settlement_total)
677 elseif command == "MEMBERS" then
678 for _, name in ipairs(names) do
679 local member, count = state.members[name], 0
680 for _, e in ipairs(expenses) do
681 if expense_has(e, name) then
682 count = count + 1
683 end
684 end
685 output[#output + 1] = " " .. name .. " " .. (member.active and "active" or "left") .. " day " .. member.day .. " expenses " .. count
686 end
687 elseif command == "EXPENSES" then
688 output[1] = line
689 local mode, count, sum = args[1], 0, 0
690 if mode == "MEMBER" then
691 check_member(state, args[2], false)
692 end
693 for _, e in ipairs(expenses) do
694 local matches = mode == "ALL" or (mode == "CATEGORY" and e.category == args[2]) or
695 (mode == "MEMBER" and expense_has(e, args[2])) or
696 (mode == "DAYS" and e.day >= tonumber(args[2]) and e.day <= tonumber(args[3]))
697 if matches then
698 count, sum = count + 1, sum + e.total
699 output[#output + 1] = " day " .. e.day .. " " .. e.id .. " " .. e.category .. " " .. amount_text(e.total) .. " by " .. table.concat(e.payers, ",")
700 end
701 end
702 output[#output + 1] = " count " .. count .. " total " .. amount_text(sum)
703 elseif command == "PERIODS" then
704 local k = tonumber(args[1])
705 output[1] = "PERIODS " .. k
706 local groups = {}
707 for _, e in ipairs(expenses) do
708 local i = (e.day - 1) // k
709 groups[i] = groups[i] or {count = 0, total = 0}
710 groups[i].count = groups[i].count + 1
711 groups[i].total = groups[i].total + e.total
712 end
713 for _, i in ipairs(sorted_keys(groups)) do
714 local group = groups[i]
715 output[#output + 1] = " days " .. (i * k + 1) .. "-" .. (i * k + k) .. " expenses " .. group.count .. " total " .. amount_text(group.total)
716 end
717 elseif command == "SHOW" then
718 local e = expense_ref(state, args[1])
719 output[1] = "SHOW " .. e.id
720 output[#output + 1] = " day " .. e.day .. " category " .. e.category .. " mode " .. e.mode .. " total " .. amount_text(e.total) .. currency_note(e)
721 for i, name in ipairs(e.payers) do
722 output[#output + 1] = " paid " .. name .. " " .. amount_text(e.contributions[i])
723 end
724 for i, name in ipairs(e.participants) do
725 local value, suffix = e.values[i], ""
726 if e.mode == "EXACT" then
727 suffix = " " .. amount_text(value)
728 elseif e.mode == "PERCENT" then
729 suffix = " " .. amount_text(value) .. "%"
730 elseif e.mode == "SHARES" then
731 suffix = " " .. value .. "/" .. total(e.values)
732 elseif e.mode == "ADJUST" then
733 suffix = " +" .. amount_text(value)
734 end
735 output[#output + 1] = " share " .. name .. " " .. amount_text(e.shares[i]) .. suffix
736 end
737 if e.interval > 0 then
738 output[#output + 1] = " recurring interval " .. e.interval .. " next " .. e.next_day
739 end
740 elseif command == "SIMPLIFY" then
741 local bal, selected = balances(state), {}
742 for _, name in ipairs(names) do
743 if bal[name] ~= 0 then
744 selected[#selected + 1] = name
745 end
746 end
747 if args[1] == "GREEDY" then
748 local transfers = greedy_transfers(selected, bal)
749 output[1] = "SIMPLIFY GREEDY transfers " .. #transfers
750 for _, transfer in ipairs(transfers) do
751 output[#output + 1] = " " .. transfer[1] .. " -> " .. transfer[2] .. " " .. amount_text(transfer[3])
752 end
753 else
754 if #selected > 12 then
755 fail("too many members")
756 end
757 local groups = optimal_groups(selected, bal)
758 local transfers, count = {}, 0
759 for i, group in ipairs(groups) do
760 transfers[i] = greedy_transfers(group, bal)
761 count = count + #transfers[i]
762 end
763 output[1] = "SIMPLIFY OPTIMAL transfers " .. count .. " groups " .. #groups
764 for i, group in ipairs(groups) do
765 output[#output + 1] = " group " .. table.concat(group, " ")
766 for _, transfer in ipairs(transfers[i]) do
767 output[#output + 1] = " " .. transfer[1] .. " -> " .. transfer[2] .. " " .. amount_text(transfer[3])
768 end
769 end
770 end
771 end
772 if contains({"BALANCES", "OWES", "HISTORY", "CATEGORY", "BUDGETS", "MEMBERS", "PERIODS"}, command) and #output == 1 then
773 output[2] = " (none)"
774 end
775 return output
776end

778local function execute(line, state)
779 local tokens = words(line)
780 local command = table.remove(tokens, 1)
781 local args = tokens
782 if command == "DAY" then
783 return advance_day(state, tonumber(args[1]))
784 elseif command == "MEMBER" then
785 local name = args[1]
786 if state.members[name] and state.members[name].active then
787 fail("duplicate member " .. name)
788 end
789 local label = state.members[name] and "rejoined" or "joined"
790 state.members[name] = {active = true, day = state.day}
791 return {"MEMBER " .. name .. " " .. label}
792 elseif command == "LEAVE" then
793 local name = args[1]
794 check_member(state, name)
795 if balances(state)[name] ~= 0 then
796 fail("member has a balance " .. name)
797 end
798 state.members[name] = {active = false, day = state.day}
799 return {"LEAVE " .. name}
800 elseif command == "CURRENCY" then
801 local currency, rate = args[1], tonumber(args[2])
802 local suffix = state.currencies[currency] and " was " .. rate_text(state.currencies[currency]) or ""
803 state.currencies[currency] = rate
804 return {"CURRENCY " .. currency .. " " .. rate_text(rate) .. suffix}
805 elseif command == "EXPENSE" or command == "EDIT" then
806 local id, payer, spec, category, mode = table.unpack(args, 1, 5)
807 local original
808 if command == "EXPENSE" then
809 if state.used[id] then
810 fail("duplicate id " .. id)
811 end
812 else
813 original = expense_ref(state, id, true)
814 end
815 local parts = {table.unpack(args,6)}
816 local expense = build_expense(state, id, payer, spec, category, mode, parts)
817 if original then
818 expense.day, expense.record = original.day, original.record
819 expense.interval, expense.next_day, expense.copies = original.interval, original.next_day, original.copies
820 else
821 state.serial = state.serial + 1
822 expense.record = state.serial
823 state.used[id] = true
824 end
825 state.expenses[id] = expense
826 local output = {expense_line(command, expense)}
827 local warnings = budget_warning(state, category)
828 table.move(warnings,1,#warnings,#output+1,output)
829 return output
830 elseif command == "PAYERS" then
831 local expense = expense_ref(state, args[1], true)
832 local names, payments = {}, {}
833 for i = 2, #args do
834 local name, amount = args[i]:match("^([^:]+):(.+)$")
835 check_member(state, name)
836 if contains(names, name) then
837 fail("duplicate member " .. name)
838 end
839 names[#names + 1] = name
840 payments[#payments + 1] = parse_amount(amount)
841 end
842 if total(payments) ~= expense.amount then
843 fail("payers do not add up")
844 end
845 local contributions = largest_remainder(expense.total, payments)
846 expense.payers, expense.payments, expense.contributions = names, payments, contributions
847 return {expense_line(command, expense, true)}
848 elseif command == "DELETE" then
849 expense_ref(state, args[1], true)
850 state.expenses[args[1]] = nil
851 return {"DELETE " .. args[1]}
852 elseif command == "RECURRING" then
853 local expense = expense_ref(state, args[1])
854 local interval = tonumber(args[2])
855 expense.interval = interval
856 expense.next_day = interval == 0 and 0 or state.day + interval
857 if interval == 0 then
858 return {"RECURRING " .. expense.id .. " stopped"}
859 end
860 return {"RECURRING " .. expense.id .. " interval " .. interval .. " next " .. expense.next_day}
861 elseif command == "SETTLE" then
862 local sender, receiver, spec = table.unpack(args)
863 check_member(state, sender)
864 check_member(state, receiver)
865 if sender == receiver then
866 fail("same member " .. sender)
867 end
868 local amount, currency = amount_spec(spec)
869 local value = converted(state, amount, currency)
870 if amount == 0 or value == 0 then
871 fail("invalid amount")
872 end
873 state.serial = state.serial + 1
874 local settlement = {day = state.day, record = state.serial, sender = sender, receiver = receiver,
875 amount = amount, currency = currency, total = value}
876 state.settlements[#state.settlements + 1] = settlement
877 return {"SETTLE " .. sender .. " -> " .. receiver .. " " .. amount_text(value) .. currency_note(settlement)}
878 elseif command == "BUDGET" then
879 local category, amount = args[1], parse_amount(args[2])
880 if amount == 0 then
881 state.budgets[category] = nil
882 return {"BUDGET " .. category .. " removed"}
883 end
884 state.budgets[category] = amount
885 return {"BUDGET " .. category .. " " .. amount_text(amount)}
886 end
887 return ledger_summary(command, args, line, state)
888end

890local state = {day = 1, serial = 0, members = {}, currencies = {}, expenses = {}, used = {}, settlements = {}, budgets = {}}
891local count = tonumber(io.read("l"))
892for i = 1, count do
893 local valid, output = pcall(execute, io.read("l"), state)
894 if valid then
895 for _, line in ipairs(output) do
896 io.write(line, "\n")
897 end
898 else
899 io.write("ERROR ", output, "\n")
900 end
901end
