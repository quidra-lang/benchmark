40function distribute(total, weights) {
41 const sum = weights.reduce((a, b) => a + b, 0);
42 const portions = weights.map(weight => Number(BigInt(total) * BigInt(weight) / BigInt(sum)));
43 const order = weights.map((weight, i) => ({ i, remainder: BigInt(total) * BigInt(weight) % BigInt(sum) }));
44 order.sort((a, b) => a.remainder === b.remainder ? a.i - b.i : a.remainder > b.remainder ? -1 : 1);
45 const remaining = total - portions.reduce((a, b) => a + b, 0);
46 for (let i = 0; i < remaining; i++) {
47 portions[order[i].i]++;
48 }
49 return portions;
50}

52function equal(total, count) {
53 const q = ratio(total, count);
54 return Array.from({ length: count }, (_, i) => q + (i < total % count ? 1 : 0));

65function net(expense, name) {
66 const paid = expense.payers.reduce((sum, person, i) => sum + (person.name === name ? expense.contributions[i] : 0), 0);
67 const share = expense.participants.reduce((sum, person, i) => sum + (person.name === name ? expense.shares[i] : 0), 0);
68 return paid - share;
69}

75function greedy(balances) {
76 const values = new Map(balances);
77 const result = [];
78 while (true) {
79 const debtors = [...values].filter(([, value]) => value < 0).sort((a, b) => a[1] - b[1] || lex(a[0], b[0]));
80 const creditors = [...values].filter(([, value]) => value > 0).sort((a, b) => b[1] - a[1] || lex(a[0], b[0]));
81 if (!debtors.length) {
82 return result;
83 }
84 const [from, debt] = debtors[0];
85 const [to, credit] = creditors[0];
86 const value = Math.min(-debt, credit);
87 result.push({ from, to, amount: value });
88 values.set(from, debt + value);
89 values.set(to, credit - value);
90 }
91}

157 prepare(id, payer, spec, category, mode,
158 parts, previous) {
159 this.member(payer);
160 const converted = this.specification(spec);
161 const participants = [];
162 if (mode === "EQUAL" && parts[0] === "ALL") {
163 participants.push(...this.names(true).map(name => ({ name, value: 0 })));
164 } else {
165 const seen = new Set();
166 for (const part of parts) {
167 const [name, value] = part.split(":");
168 this.member(name);
169 if (seen.has(name)) {
170 fail("duplicate member " + name);
171 }
172 seen.add(name);
173 const specified = value === undefined ? 0 :
174 mode === "EXACT" || mode === "ADJUST" ? amount(value) : Number(value);
175 participants.push({ name, value: specified });
176 }
177 }
178 const a = converted.amount;
179 const sum = participants.reduce((sum, person) => sum + person.value, 0);
180 if ((mode === "EXACT" && sum !== a) || (mode === "PERCENT" && sum !== 10000) ||
181 (mode === "SHARES" && sum === 0) || (mode === "ADJUST" && sum > a)) {
182 fail("splits do not add up");
183 }
184 const expense = {
185 id, day: previous?.day ?? this.day, record: previous?.record ?? 0,
186 ...converted, category, mode, payers: [{ name: payer, value: a }], participants,
187 shares: [], contributions: [converted.total],
188 interval: previous?.interval ?? 0, next: previous?.next ?? 0, copies: previous?.copies ?? 0
189 };
190 expense.shares = this.originalShares(expense, expense.total);
191 return expense;
192 }

194 balances() {
195 const values = new Map(this.names().map((name) => [name, 0]));
196 for (const expense of this.expenses) {
197 expense.payers.forEach((person, i) => values.set(person.name, values.get(person.name) + expense.contributions[i]));
198 expense.participants.forEach((person, i) => values.set(person.name, values.get(person.name) - expense.shares[i]));
199 }
200 for (const payment of this.settlements) {
201 values.set(payment.from, values.get(payment.from) + payment.total);
202 values.set(payment.to, values.get(payment.to) - payment.total);
203 }
204 return values;
205 }

207 spent(category) {
208 return this.expenses.filter(expense => expense.category === category).reduce((sum, expense) => sum + expense.total, 0);

218 debts() {
219 const result = new Map();
220 const add = (from, to, value) => {
221 const key = from + ":" + to;
222 result.set(key, (result.get(key) ?? 0) + value);
223 };
224 for (const expense of this.expenses) {
225 const people = this.names().filter(name => involved(expense, name)).map(name => ({ name, value: net(expense, name) }));
226 const borrowers = people.filter(person => person.value < 0);
227 const lenders = people.filter(person => person.value > 0);
228 let i = 0;
229 let j = 0;
230 while (i < borrowers.length && j < lenders.length) {
231 const value = Math.min(-borrowers[i].value, lenders[j].value);
232 add(borrowers[i].name, lenders[j].name, value);
233 borrowers[i].value += value;
234 lenders[j].value -= value;
235 if (!borrowers[i].value) {
236 i++;
237 }
238 if (!lenders[j].value) {
239 j++;
240 }
241 }
242 }
243 for (const payment of this.settlements) {
244 add(payment.from, payment.to, -payment.total);
245 }
246 return result;
247 }

253 optimal() {
254 const balances = this.balances();
255 const names = this.names().filter(name => balances.get(name) !== 0);
256 if (names.length > 12) {
257 fail("too many members");
258 }
259 const full = (1 << names.length) - 1;
260 const sums = new Array(full + 1).fill(0);
261 for (let mask = 1; mask <= full; mask++) {
262 const bit = mask & -mask;
263 const index = 31 - Math.clz32(bit);
264 sums[mask] = sums[mask ^ bit] + balances.get(names[index]);
265 }
266 const memo = new Map([[0, 0]]);
267 const best = (mask) => {
268 const known = memo.get(mask);
269 if (known !== undefined) {
270 return known;
271 }
272 const first = mask & -mask;
273 let maximum = 0;
274 for (let sub = mask; sub; sub = (sub - 1) & mask) {
275 if ((sub & first) && sums[sub] === 0) {
276 maximum = Math.max(maximum, 1 + best(mask ^ sub));
277 }
278 }
279 memo.set(mask, maximum);
280 return maximum;
281 };
282 const list = (mask) => names.filter((_, i) => mask & (1 << i));
283 const compare = (a, b) => {
284 const left = list(a);
285 const right = list(b);
286 for (let i = 0; i < Math.min(left.length, right.length); i++) {
287 const c = lex(left[i], right[i]);
288 if (c) {
289 return c;
290 }
291 }
292 return left.length - right.length;
293 };
294 const groups = [];
295 let mask = full;
296 while (mask) {
297 const candidates = [];
298 const first = mask & -mask;
299 for (let sub = mask; sub; sub = (sub - 1) & mask) {
300 if ((sub & first) && sums[sub] === 0 && 1 + best(mask ^ sub) === best(mask)) {
301 candidates.push(sub);
302 }
303 }
304 candidates.sort(compare);
305 groups.push(list(candidates[0]));
306 mask ^= candidates[0];
307 }
308 const output = ["SIMPLIFY OPTIMAL transfers " + (names.length - groups.length) + " groups " + groups.length];
309 for (const group of groups) {
310 output.push(" group " + group.join(" "));
311 const subset = new Map(group.map((name) => [name, balances.get(name)]));
312 output.push(...greedy(subset).map(transfer => transferText(transfer, " ")));
313 }
314 return output;
315 }
