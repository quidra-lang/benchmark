floor_div <- function(n,d) {
    quotient <- trunc(n/d)
    remainder <- n-quotient*d
    if (remainder < 0) {
        quotient <- quotient-1
    } else if (remainder >= d) {
        quotient <- quotient+1
    }
    quotient
