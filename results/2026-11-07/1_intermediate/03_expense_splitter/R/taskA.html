3floor_div <- function(n,d) {
4 quotient <- trunc(n/d)
5 remainder <- n-quotient*d
6 if (remainder < 0) {
7 quotient <- quotient-1
8 } else if (remainder >= d) {
9 quotient <- quotient+1
10 }
11 quotient
12}

82equal_distribution <- function(total,k) {
83 quotient <- floor_div(total,k)
84 quotient+as.numeric(seq_len(k) <= total-quotient*k)
85}

87largest_remainder <- function(total,weights) {
88 whole <- sum(weights)
89 result <- vapply(weights,function(weight) floor_div(total*weight,whole),numeric(1))
90 remainders <- total*weights-result*whole
91 ordering <- order(-remainders,seq_along(weights))
92 for (i in seq_len(total-sum(result))) {
93 result[ordering[i]] <- result[ordering[i]]+1
94 }
95 unname(result)
96}

98portions <- function(expense) {
99 values <- expense$values
100 if (expense$mode == "EQUAL") return(equal_distribution(expense$amount,length(values)))
101 if (expense$mode == "EXACT") return(values)
102 if (expense$mode %in% c("PERCENT","SHARES")) return(largest_remainder(expense$amount,values))
103 equal_distribution(expense$amount-sum(values),length(values))+values
104}

106recompute <- function(state,expense) {
107 expense$total <- converted(state,expense$amount,expense$currency)
108 expense$shares <- largest_remainder(expense$total,portions(expense))
109 expense$contributions <- largest_remainder(expense$total,expense$payments)
110 expense

113build_expense <- function(state,id,payer,spec,category,mode,parts) {
114 check_member(state,payer)
115 parsed <- amount_spec(spec)
116 amount <- parsed$amount
117 currency <- parsed$currency
118 total <- converted(state,amount,currency)
119 if (amount == 0 || total == 0) fail("invalid amount")
120 names <- character(0)
121 values <- numeric(0)
122 if (mode == "EQUAL" && identical(parts,"ALL")) {
123 names <- active_names(state)
124 values <- rep(0,length(names))
125 } else {
126 for (part in parts) {
127 pieces <- strsplit(part,":",fixed=TRUE)[[1]]
128 name <- pieces[1]
129 check_member(state,name)
130 if (name %in% names) fail(paste("duplicate member",name))
131 value <- 0
132 if (length(pieces) == 2) value <- if (mode %in% c("EXACT","ADJUST")) parse_amount(pieces[2]) else as.numeric(pieces[2])
133 names <- c(names,name)
134 values <- c(values,value)
135 }
136 }
137 valid <- switch(mode,EXACT=sum(values) == amount,PERCENT=sum(values) == 10000,
138 SHARES=sum(values) >= 1,ADJUST=sum(values) <= amount,EQUAL=TRUE)
139 if (!valid) fail("splits do not add up")
140 recompute(state,list(id=id,day=state$day,record=0,payers=payer,payments=amount,
141 amount=amount,currency=currency,category=category,mode=mode,participants=names,
142 values=values,total=total,shares=numeric(0),contributions=numeric(0),interval=0,next_day=0,copies=0))
143}

167greedy_transfers <- function(names,original) {
168 remaining <- original[names]
169 output <- list()
170 while (length(remaining) > 0 && min(remaining) < 0) {
171 sender <- names[which.min(remaining)]
172 receiver <- names[which.max(remaining)]
173 amount <- min(-remaining[[sender]],remaining[[receiver]])
174 output[[length(output)+1]] <- list(sender=sender,receiver=receiver,amount=amount)
175 remaining[sender] <- remaining[sender]+amount
176 remaining[receiver] <- remaining[receiver]-amount
177 }
178 output
179}

181debts <- function(state) {
182 names <- registered_names(state)
183 result <- matrix(0,length(names),length(names),dimnames=list(names,names))
184 for (expense in state$expenses) {
185 net <- expense_net(expense)
186 borrowers <- sort(names(net)[net < 0],method="radix")
187 lenders <- sort(names(net)[net > 0],method="radix")
188 i <- 1
189 j <- 1
190 while (i <= length(borrowers) && j <= length(lenders)) {
191 sender <- borrowers[i]
192 receiver <- lenders[j]
193 amount <- min(-net[[sender]],net[[receiver]])
194 result[sender,receiver] <- result[sender,receiver]+amount
195 net[sender] <- net[sender]+amount
196 net[receiver] <- net[receiver]-amount
197 if (net[[sender]] == 0) i <- i+1
198 if (net[[receiver]] == 0) j <- j+1
199 }
200 }
201 for (s in state$settlements) {
202 result[s$sender,s$receiver] <- result[s$sender,s$receiver]-s$total
203 }
204 result
205}

230advance_day <- function(state,day) {
231 if (day <= state$day) fail(paste("invalid day",integer_text(day)))
232 output <- paste("DAY",integer_text(day))
233 repeat {
234 due <- Filter(function(e) e$interval > 0 && e$next_day <= day,state$expenses)
235 if (length(due) == 0) break
236 ordering <- order(vapply(due,function(e) e$next_day,numeric(1)),vapply(due,function(e) e$record,numeric(1)))
237 template <- due[[ordering[1]]]
238 id <- template$id
239 active <- all(vapply(expense_names(template),function(name) state$members[[name]]$active,logical(1)))
240 if (!active || converted(state,template$amount,template$currency) == 0) {
241 state$expenses[[id]]$interval <- 0
242 state$expenses[[id]]$next_day <- 0
243 output <- c(output,paste(" CANCEL",id))
244 next
245 }
246 copy_id <- paste0(id,"#",integer_text(template$copies+2))
247 copy <- template
248 copy$id <- copy_id
249 copy$day <- template$next_day
250 state$serial <- state$serial+1
251 copy$record <- state$serial
252 copy$interval <- 0
253 copy$next_day <- 0
254 copy$copies <- 0
255 copy <- recompute(state,copy)
256 state$expenses[[copy_id]] <- copy
257 state$used <- c(state$used,copy_id)
258 state$expenses[[id]]$copies <- template$copies+1
259 state$expenses[[id]]$next_day <- template$next_day+template$interval
260 output <- c(output,paste(" RECUR",copy_id,amount_text(copy$total)),budget_warning(state,copy$category))
261 }
262 state$day <- day
263 output
264}

266vector_less <- function(a,b) {
267 for (i in seq_len(min(length(a),length(b)))) {
268 if (a[i] != b[i]) return(a[i] < b[i])
269 }
270 length(a) < length(b)

273optimal_groups <- function(names,bal) {
274 n <- length(names)
275 limit <- bitwShiftL(1,n)
276 sums <- rep(0,limit)
277 bits <- bitwShiftL(1,seq_len(n)-1)
278 for (mask in seq_len(limit-1)) {
279 bit <- bitwAnd(mask,-mask)
280 index <- match(bit,bits)
281 sums[mask+1] <- sums[bitwXor(mask,bit)+1]+bal[[names[index]]]
282 }
283 memo <- rep(-1,limit)
284 memo[1] <- 0
285 best <- function(mask) {
286 if (memo[mask+1] >= 0) return(memo[mask+1])
287 first <- bitwAnd(mask,-mask)
288 value <- 0
289 subset <- mask
290 while (subset != 0) {
291 if (bitwAnd(subset,first) != 0 && sums[subset+1] == 0) value <- max(value,1+best(bitwXor(mask,subset)))
292 subset <- bitwAnd(subset-1,mask)
293 }
294 memo[mask+1] <<- value
295 value
296 }
297 groups <- list()
298 mask <- limit-1
299 while (mask != 0) {
300 first <- bitwAnd(mask,-mask)
301 selected <- 0
302 selected_indices <- integer(0)
303 target <- best(mask)
304 subset <- mask
305 while (subset != 0) {
306 if (bitwAnd(subset,first) != 0 && sums[subset+1] == 0 && 1+best(bitwXor(mask,subset)) == target) {
307 indices <- which(bitwAnd(subset,bits) != 0)
308 if (selected == 0 || vector_less(indices,selected_indices)) {
309 selected <- subset
310 selected_indices <- indices
311 }
312 }
313 subset <- bitwAnd(subset-1,mask)
314 }
315 groups[[length(groups)+1]] <- names[selected_indices]
316 mask <- bitwXor(mask,selected)
317 }
318 groups
319}

321ledger_summary <- function(command,args,line,state) {
322 output <- command
323 names <- registered_names(state)
324 expenses <- chronological(state)
325 if (command == "BALANCES") {
326 bal <- balances(state)
327 for (name in active_names(state)) {
328 output <- c(output,paste(" ",name,signed_text(bal[[name]])))
329 }
330 } else if (command == "OWES") {
331 name <- args[1]
332 check_member(state,name,FALSE)
333 output <- paste("OWES",name)
334 table <- debts(state)
335 for (other in names) {
336 if (other == name) next
337 amount <- net_debt(table,name,other)
338 if (amount > 0) output <- c(output,paste(" ",name,"owes",other,amount_text(amount)))
339 if (amount < 0) output <- c(output,paste(" ",other,"owes",name,amount_text(-amount)))
340 }
341 } else if (command == "HISTORY") {
342 name <- args[1]
343 check_member(state,name,FALSE)
344 output <- paste("HISTORY",name)
345 records <- list()
346 for (expense in expenses) {
347 if (expense_has(expense,name)) records[[length(records)+1]] <- list(day=expense$day,record=expense$record,label=expense$id,change=expense_net(expense)[[name]])
348 }
349 for (s in state$settlements) {
350 if (name %in% c(s$sender,s$receiver)) {
351 change <- if (name == s$sender) s$total else -s$total
352 records[[length(records)+1]] <- list(day=s$day,record=s$record,label=paste0(s$sender,"->",s$receiver),change=change)
353 }
354 }
355 ordering <- order(vapply(records,function(r) r$day,numeric(1)),vapply(records,function(r) r$record,numeric(1)))
356 running <- 0
357 for (record in records[ordering]) {
358 running <- running+record$change
359 output <- c(output,paste(" day",integer_text(record$day),record$label,signed_text(record$change),"balance",signed_text(running)))
360 }
361 } else if (command == "MATRIX") {
362 active <- active_names(state)
363 if (length(active) == 0) return(c("MATRIX"," (none)"))
364 table <- debts(state)
365 cells <- lapply(active,function(a) vapply(active,function(b) if (a == b) "-" else amount_text(max(0,net_debt(table,a,b))),character(1)))
366 left <- max(nchar(active))
367 width <- max(left,max(nchar(unlist(cells,use.names=FALSE))))
368 output <- c(output,paste0(strrep(" ",left),paste0(" ",sprintf("%*s",width,active),collapse="")))
369 for (i in seq_along(active)) {
370 row <- sprintf("%-*s",left,active[i])
371 output <- c(output,paste0(row,paste0(" ",sprintf("%*s",width,cells[[i]]),collapse="")))
372 }
373 } else if (command == "CATEGORY") {
374 totals <- category_totals(state)
375 whole <- sum(totals)
376 categories <- names(totals)[order(-totals,names(totals),method="radix")]
377 for (category in categories) {
378 total <- totals[[category]]
379 output <- c(output,paste(" ",category,amount_text(total),percentage(total,whole)))
380 shares <- setNames(rep(0,length(names)),names)
381 for (expense in expenses) {
382 if (expense$category == category) {
383 for (i in seq_along(expense$participants)) {
384 name <- expense$participants[i]
385 shares[name] <- shares[name]+expense$shares[i]
386 }
387 }
388 }
389 people <- names[shares > 0]
390 people <- people[order(-shares[people],people,method="radix")]
391 for (name in people) {
392 output <- c(output,paste(" ",name,amount_text(shares[[name]]),percentage(shares[[name]],total)))
393 }
394 }
395 } else if (command == "BUDGETS") {
396 totals <- category_totals(state)
397 for (category in sort(names(state$budgets),method="radix")) {
398 spent <- if (category %in% names(totals)) totals[[category]] else 0
399 budget <- state$budgets[[category]]
400 label <- if (spent <= budget) "left" else "over"
401 output <- c(output,paste(" ",category,"spent",amount_text(spent),"budget",amount_text(budget),"used",percentage(spent,budget),label,amount_text(abs(budget-spent))))
402 }
403 } else if (command == "TOTALS") {
404 for (name in names) {
405 paid <- 0
406 share <- 0
407 sent <- 0
408 received <- 0
409 for (expense in expenses) {
410 paid <- paid+sum(expense$contributions[expense$payers == name])
411 share <- share+sum(expense$shares[expense$participants == name])
412 }
413 for (s in state$settlements) {
414 if (s$sender == name) sent <- sent+s$total
415 if (s$receiver == name) received <- received+s$total
416 }
417 output <- c(output,paste(" ",name,"paid",amount_text(paid),"share",amount_text(share),"sent",amount_text(sent),"received",amount_text(received)))
418 }
419 currencies <- sort(unique(vapply(expenses,function(e) e$currency,character(1))),method="radix")
420 for (currency in currencies[currencies != ""]) {
421 selected <- Filter(function(e) e$currency == currency,expenses)
422 output <- c(output,paste(" currency",currency,"expenses",length(selected),"amount",amount_text(sum(vapply(selected,function(e) e$amount,numeric(1)))),"base",amount_text(sum(vapply(selected,function(e) e$total,numeric(1))))))
423 }
424 output <- c(output,paste(" expenses",length(expenses),"total",amount_text(sum(vapply(expenses,function(e) e$total,numeric(1)))),"settlements",length(state$settlements),"total",amount_text(sum(vapply(state$settlements,function(s) s$total,numeric(1))))))
425 } else if (command == "MEMBERS") {
426 for (name in names) {
427 member <- state$members[[name]]
428 status <- if (member$active) "active" else "left"
429 count <- sum(vapply(expenses,function(e) expense_has(e,name),logical(1)))
430 output <- c(output,paste(" ",name,status,"day",integer_text(member$day),"expenses",count))
431 }
432 } else if (command == "EXPENSES") {
433 output <- line
434 mode <- args[1]
435 if (mode == "MEMBER") check_member(state,args[2],FALSE)
436 selected <- Filter(function(e) {
437 mode == "ALL" || (mode == "CATEGORY" && e$category == args[2]) ||
438 (mode == "MEMBER" && expense_has(e,args[2])) ||
439 (mode == "DAYS" && e$day >= as.numeric(args[2]) && e$day <= as.numeric(args[3]))
440 },expenses)
441 for (e in selected) {
442 output <- c(output,paste(" day",integer_text(e$day),e$id,e$category,amount_text(e$total),"by",paste(e$payers,collapse=",")))
443 }
444 output <- c(output,paste(" count",length(selected),"total",amount_text(sum(vapply(selected,function(e) e$total,numeric(1))))))
445 } else if (command == "PERIODS") {
446 k <- as.numeric(args[1])
447 output <- paste("PERIODS",integer_text(k))
448 periods <- vapply(expenses,function(e) floor_div(e$day-1,k),numeric(1))
449 for (i in sort(unique(periods))) {
450 group <- expenses[periods == i]
451 output <- c(output,paste(" days",paste0(integer_text(i*k+1),"-",integer_text(i*k+k)),"expenses",length(group),"total",amount_text(sum(vapply(group,function(e) e$total,numeric(1))))))
452 }
453 } else if (command == "SHOW") {
454 e <- expense_ref(state,args[1])
455 output <- paste("SHOW",e$id)
456 output <- c(output,paste0(" day ",integer_text(e$day)," category ",e$category," mode ",e$mode," total ",amount_text(e$total),currency_note(e)))
457 for (i in seq_along(e$payers)) {
458 output <- c(output,paste(" paid",e$payers[i],amount_text(e$contributions[i])))
459 }
460 for (i in seq_along(e$participants)) {
461 value <- e$values[i]
462 suffix <- ""
463 if (e$mode == "EXACT") suffix <- paste0(" ",amount_text(value))
464 if (e$mode == "PERCENT") suffix <- paste0(" ",amount_text(value),"%")
465 if (e$mode == "SHARES") suffix <- paste0(" ",integer_text(value),"/",integer_text(sum(e$values)))
466 if (e$mode == "ADJUST") suffix <- paste0(" +",amount_text(value))
467 output <- c(output,paste0(" share ",e$participants[i]," ",amount_text(e$shares[i]),suffix))
468 }
469 if (e$interval > 0) output <- c(output,paste(" recurring interval",integer_text(e$interval),"next",integer_text(e$next_day)))
470 } else if (command == "SIMPLIFY") {
471 bal <- balances(state)
472 selected <- names[bal != 0]
473 if (args[1] == "GREEDY") {
474 transfers <- greedy_transfers(selected,bal)
475 output <- paste("SIMPLIFY GREEDY transfers",length(transfers))
476 for (t in transfers) {
477 output <- c(output,paste(" ",t$sender,"->",t$receiver,amount_text(t$amount)))
478 }
479 } else {
480 if (length(selected) > 12) fail("too many members")
481 groups <- optimal_groups(selected,bal)
482 transfers <- lapply(groups,greedy_transfers,bal)
483 output <- paste("SIMPLIFY OPTIMAL transfers",sum(lengths(transfers)),"groups",length(groups))
484 for (i in seq_along(groups)) {
485 output <- c(output,paste(" group",paste(groups[[i]],collapse=" ")))
486 for (t in transfers[[i]]) {
487 output <- c(output,paste(" ",t$sender,"->",t$receiver,amount_text(t$amount)))
488 }
489 }
490 }
491 }
492 if (command %in% c("BALANCES","OWES","HISTORY","CATEGORY","BUDGETS","MEMBERS","PERIODS") && length(output) == 1) output <- c(output," (none)")
493 output
494}

496execute <- function(line,state) {
497 tokens <- strsplit(line," ",fixed=TRUE)[[1]]
498 command <- tokens[1]
499 args <- tokens[-1]
500 if (command == "DAY") return(advance_day(state,as.numeric(args[1])))
501 if (command == "MEMBER") {
502 name <- args[1]
503 if (name %in% names(state$members) && state$members[[name]]$active) fail(paste("duplicate member",name))
504 label <- if (name %in% names(state$members)) "rejoined" else "joined"
505 state$members[[name]] <- list(active=TRUE,day=state$day)
506 return(paste("MEMBER",name,label))
507 }
508 if (command == "LEAVE") {
509 name <- args[1]
510 check_member(state,name)
511 if (balances(state)[[name]] != 0) fail(paste("member has a balance",name))
512 state$members[[name]] <- list(active=FALSE,day=state$day)
513 return(paste("LEAVE",name))
514 }
515 if (command == "CURRENCY") {
516 currency <- args[1]
517 rate <- as.numeric(args[2])
518 suffix <- if (currency %in% names(state$currencies)) paste0(" was ",rate_text(state$currencies[[currency]])) else ""
519 state$currencies[[currency]] <- rate
520 return(paste0("CURRENCY ",currency," ",rate_text(rate),suffix))
521 }
522 if (command %in% c("EXPENSE","EDIT")) {
523 id <- args[1]
524 if (command == "EXPENSE") {
525 if (id %in% state$used) fail(paste("duplicate id",id))
526 } else {
527 original <- expense_ref(state,id,TRUE)
528 }
529 expense <- build_expense(state,id,args[2],args[3],args[4],args[5],args[-(1:5)])
530 if (command == "EDIT") {
531 for (key in c("day","record","interval","next_day","copies")) {
532 expense[[key]] <- original[[key]]
533 }
534 } else {
535 state$serial <- state$serial+1
536 expense$record <- state$serial
537 state$used <- c(state$used,id)
538 }
539 state$expenses[[id]] <- expense
540 return(c(expense_line(command,expense),budget_warning(state,expense$category)))
541 }
542 if (command == "PAYERS") {
543 expense <- expense_ref(state,args[1],TRUE)
544 names <- character(0)
545 payments <- numeric(0)
546 for (part in args[-1]) {
547 pieces <- strsplit(part,":",fixed=TRUE)[[1]]
548 name <- pieces[1]
549 check_member(state,name)
550 if (name %in% names) fail(paste("duplicate member",name))
551 names <- c(names,name)
552 payments <- c(payments,parse_amount(pieces[2]))
553 }
554 if (sum(payments) != expense$amount) fail("payers do not add up")
555 expense$payers <- names
556 expense$payments <- payments
557 expense$contributions <- largest_remainder(expense$total,payments)
558 state$expenses[[expense$id]] <- expense
559 return(expense_line(command,expense,TRUE))
560 }
561 if (command == "DELETE") {
562 expense_ref(state,args[1],TRUE)
563 state$expenses[[args[1]]] <- NULL
564 return(paste("DELETE",args[1]))
565 }
566 if (command == "RECURRING") {
567 expense <- expense_ref(state,args[1])
568 interval <- as.numeric(args[2])
569 expense$interval <- interval
570 expense$next_day <- if (interval == 0) 0 else state$day+interval
571 state$expenses[[expense$id]] <- expense
572 if (interval == 0) return(paste("RECURRING",expense$id,"stopped"))
573 return(paste("RECURRING",expense$id,"interval",integer_text(interval),"next",integer_text(expense$next_day)))
574 }
575 if (command == "SETTLE") {
576 sender <- args[1]
577 receiver <- args[2]
578 check_member(state,sender)
579 check_member(state,receiver)
580 if (sender == receiver) fail(paste("same member",sender))
581 parsed <- amount_spec(args[3])
582 total <- converted(state,parsed$amount,parsed$currency)
583 if (parsed$amount == 0 || total == 0) fail("invalid amount")
584 state$serial <- state$serial+1
585 settlement <- list(day=state$day,record=state$serial,sender=sender,receiver=receiver,
586 amount=parsed$amount,currency=parsed$currency,total=total)
587 state$settlements[[length(state$settlements)+1]] <- settlement
588 return(paste0("SETTLE ",sender," -> ",receiver," ",amount_text(total),currency_note(settlement)))
589 }
590 if (command == "BUDGET") {
591 category <- args[1]
592 amount <- parse_amount(args[2])
593 if (amount == 0) {
594 state$budgets[[category]] <- NULL
595 return(paste("BUDGET",category,"removed"))
596 }
597 state$budgets[[category]] <- amount
598 return(paste("BUDGET",category,amount_text(amount)))
599 }
600 ledger_summary(command,args,line,state)
601}
