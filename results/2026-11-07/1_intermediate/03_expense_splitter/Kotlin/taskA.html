1import java.util.TreeMap
2
3data class Amount(val value: Long, val currency: String = "")
4data class Member(val active: Boolean, val day: Long)
5data class Part(val name: String, val value: Long)
6data class Settlement(val day: Long, val record: Long, val from: String, val to: String, val amount: Amount, val total: Long)
7data class Entry(val day: Long, val record: Long, val label: String, val change: Long)
8data class Transfer(val from: String, val to: String, val amount: Long)
9data class Expense(var id: String = "", var category: String = "", var mode: String = "",
10 var amount: Amount = Amount(0), var payers: List<Part> = emptyList(), var participants: List<Part> = emptyList(),
11 var contributions: List<Long> = emptyList(), var shares: List<Long> = emptyList(), var day: Long = 0,
12 var record: Long = 0, var total: Long = 0, var interval: Long = 0, var next: Long = 0, var copies: Long = 0)
13class Problem(message: String) : RuntimeException(message)
14
15class Splitter {
16 private val members = TreeMap<String,Member>()
17 private val currencies = TreeMap<String,Long>()
18 private val budgets = TreeMap<String,Long>()
19 private val expenses = linkedMapOf<String,Expense>()
20 private val used = mutableSetOf<String>()
21 private val settlements = mutableListOf<Settlement>()
22 private var day = 1L
23 private var serial = 0L
24 private fun fail(message: String): Nothing = throw Problem(message)
25 private fun money(s: String) = s.replace(".","").toLong()
26 private fun amount(s: String): Amount {
27 val parts = s.split('@')
28 return Amount(money(parts[0]),parts.getOrElse(1) { "" })
29 }
30 private fun fmt(n: Long) = "${n/100}.${(n%100).toString().padStart(2,'0')}"
31 private fun signed(n: Long) = if (n > 0) "+${fmt(n)}" else if (n < 0) "-${fmt(-n)}" else "0.00"
32 private fun rate(n: Long) = "${n/10000}.${(n%10000).toString().padStart(4,'0')}"
33 private fun percent(p: Long, w: Long) = "${fmt(p*10000/w)}%"
34 private fun note(a: Amount) = if (a.currency.isEmpty()) "" else " [${fmt(a.value)} ${a.currency}]"
35 private fun registered(name: String) {
36 if (name !in members) fail("unknown member $name")
37 }
38 private fun active(name: String) {
39 registered(name)
40 if (!members.getValue(name).active) fail("inactive member $name")
41 }
42 private fun convert(a: Amount): Long {
43 if (a.currency.isEmpty()) return a.value
44 val rate = currencies[a.currency] ?: fail("unknown currency ${a.currency}")
45 return (a.value*rate+5000)/10000
46 }
47 private fun expense(id: String) = expenses[id] ?: fail("unknown expense $id")
48 private fun thawed(e: Expense) {
49 for (p in e.payers+e.participants) {
50 if (!members.getValue(p.name).active) fail("inactive member ${p.name}")
51 }
52 }
53 private fun distribute(total: Long, weights: List<Long>): List<Long> {
54 val sum = weights.sum()
55 val result = weights.map { total*it/sum }.toMutableList()
56 val remainders = weights.map { total*it%sum }
57 val order = weights.indices.sortedWith(compareByDescending<Int> { remainders[it] }.thenBy { it })
58 val left = (total-result.sum()).toInt()
59 for (i in 0 until left) result[order[i]]++
60 return result
61 }
62 private fun compute(e: Expense) {
63 val a = e.amount.value
64 e.total = convert(e.amount)
65 val sum = e.participants.sumOf { it.value }
66 val portions = when (e.mode) {
67 "EXACT" -> {
68 if (sum != a) fail("splits do not add up")
69 e.participants.map { it.value }
70 }
71 "PERCENT", "SHARES" -> {
72 if (e.mode == "PERCENT" && sum != 10000L || e.mode == "SHARES" && sum == 0L) fail("splits do not add up")
73 distribute(a,e.participants.map { it.value })
74 }
75 else -> {
76 val extras = if (e.mode == "ADJUST") sum else 0L
77 if (extras > a) fail("splits do not add up")
78 val k = e.participants.size
79 e.participants.mapIndexed { i,p ->
80 (a-extras)/k+(if (i < (a-extras)%k) 1 else 0)+(if (e.mode == "ADJUST") p.value else 0L)
81 }
82 }
83 }
84 e.shares = distribute(e.total,portions)
85 e.contributions = distribute(e.total,e.payers.map { it.value })
86 }
87 private fun build(t: List<String>): Expense {
88 active(t[2])
89 val a = amount(t[3])
90 if (convert(a) == 0L || a.value == 0L) fail("invalid amount")
91 val e = Expense(category = t[4],mode = t[5],amount = a,payers = listOf(Part(t[2],a.value)))
92 e.participants = if (e.mode == "EQUAL" && t[6] == "ALL") {
93 members.filterValues { it.active }.keys.map { Part(it,0) }
94 } else {
95 val seen = mutableSetOf<String>()
96 t.drop(6).map { token ->
97 val p = token.split(':')
98 active(p[0])
99 if (!seen.add(p[0])) fail("duplicate member ${p[0]}")
100 val value = if (p.size == 1) 0L else if (e.mode in setOf("EXACT","ADJUST")) money(p[1]) else p[1].toLong()
101 Part(p[0],value)
102 }
103 }
104 compute(e)
105 return e
106 }
107 private fun chronological() = expenses.values.sortedWith(compareBy<Expense> { it.day }.thenBy { it.record })
108 private fun includes(e: Expense, name: String) = (e.payers+e.participants).any { it.name == name }
109 private fun net(e: Expense): TreeMap<String,Long> {
110 val result = TreeMap<String,Long>()
111 for ((i,p) in e.payers.withIndex()) result[p.name] = result.getOrDefault(p.name,0)+e.contributions[i]
112 for ((i,p) in e.participants.withIndex()) result[p.name] = result.getOrDefault(p.name,0)-e.shares[i]
113 return result
114 }
115 private fun balances(): TreeMap<String,Long> {
116 val result = TreeMap<String,Long>()
117 for (name in members.keys) result[name] = 0
118 for (e in expenses.values) {
119 for ((name,value) in net(e)) result[name] = result.getValue(name)+value
120 }
121 for (s in settlements) {
122 result[s.from] = result.getValue(s.from)+s.total
123 result[s.to] = result.getValue(s.to)-s.total
124 }
125 return result
126 }
127 private fun activeNames() = members.filterValues { it.active }.keys.toList()
128 private fun categoryTotals(): TreeMap<String,Long> {
129 val result = TreeMap<String,Long>()
130 for (e in expenses.values) result[e.category] = result.getOrDefault(e.category,0)+e.total
131 return result
132 }
133 private fun warning(e: Expense, output: MutableList<String>) {
134 val budget = budgets[e.category] ?: return
135 val spent = categoryTotals().getValue(e.category)
136 if (spent > budget) output.add(" WARNING ${e.category} over budget by ${fmt(spent-budget)}")
137 }
138 private fun pairs(people: List<Part>, values: List<Long>) = people.indices.joinToString(" + ") { "${people[it].name} ${fmt(values[it])}" }
139 private fun greedy(balances: Map<String,Long>): List<Transfer> {
140 val work = TreeMap(balances)
141 val result = mutableListOf<Transfer>()
142 while (work.isNotEmpty()) {
143 val sender = work.entries.minBy { it.value }
144 val receiver = work.entries.maxBy { it.value }
145 val minimum = sender.value
146 val maximum = receiver.value
147 if (minimum >= 0) break
148 val from = sender.key
149 val to = receiver.key
150 val value = minOf(-minimum,maximum)
151 result.add(Transfer(from,to,value))
152 work[from] = minimum+value
153 work[to] = maximum-value
154 }
155 return result
156 }
157 private fun debts(names: List<String>): Array<LongArray> {
158 val index = names.withIndex().associate { it.value to it.index }
159 val result = Array(names.size) { LongArray(names.size) }
160 for (e in expenses.values) {
161 val amounts = net(e)
162 val borrowers = amounts.filterValues { it < 0 }.keys.toList()
163 val lenders = amounts.filterValues { it > 0 }.keys.toList()
164 var a = 0
165 var b = 0
166 while (a < borrowers.size && b < lenders.size) {
167 val x = borrowers[a]
168 val y = lenders[b]
169 val value = minOf(-amounts.getValue(x),amounts.getValue(y))
170 val i = index.getValue(x)
171 val j = index.getValue(y)
172 result[i][j] += value
173 result[j][i] -= value
174 amounts[x] = amounts.getValue(x)+value
175 amounts[y] = amounts.getValue(y)-value
176 if (amounts[x] == 0L) a++
177 if (amounts[y] == 0L) b++
178 }
179 }
180 for (s in settlements) {
181 val i = index.getValue(s.from)
182 val j = index.getValue(s.to)
183 result[i][j] -= s.total
184 result[j][i] += s.total
185 }
186 return result
187 }
188 private fun indexes(mask: Int, n: Int) = (0 until n).filter { mask and (1 shl it) != 0 }
189 private fun lexLess(a: List<Int>, b: List<Int>): Boolean {
190 for (i in 0 until minOf(a.size,b.size)) {
191 if (a[i] != b[i]) return a[i] < b[i]
192 }
193 return a.size < b.size
194 }
195 fun execute(line: String): List<String> {
196 val t = line.split(' ')
197 val command = t[0]
198 val output = mutableListOf<String>()
199 when (command) {
200 "DAY" -> {
201 val d = t[1].toLong()
202 if (d <= day) fail("invalid day ${t[1]}")
203 output.add("DAY $d")
204 while (true) {
205 val template = expenses.values.filter { it.interval > 0 && it.next <= d }
206 .minWithOrNull(compareBy<Expense> { it.next }.thenBy { it.record }) ?: break
207 if ((template.payers+template.participants).any { !members.getValue(it.name).active } || convert(template.amount) == 0L) {
208 template.interval = 0
209 output.add(" CANCEL ${template.id}")
210 } else {
211 val copy = template.copy(id = "${template.id}#${template.copies+2}",day = template.next,
212 record = ++serial,interval = 0,next = 0,copies = 0)
213 compute(copy)
214 expenses[copy.id] = copy
215 used.add(copy.id)
216 template.copies++
217 template.next += template.interval
218 output.add(" RECUR ${copy.id} ${fmt(copy.total)}")
219 warning(copy,output)
220 }
221 }
222 day = d
223 }
224 "MEMBER" -> {
225 val old = members[t[1]]
226 if (old?.active == true) fail("duplicate member ${t[1]}")
227 members[t[1]] = Member(true,day)
228 output.add("MEMBER ${t[1]} "+if (old == null) "joined" else "rejoined")
229 }
230 "LEAVE" -> {
231 active(t[1])
232 if (balances().getValue(t[1]) != 0L) fail("member has a balance ${t[1]}")
233 members[t[1]] = Member(false,day)
234 output.add("LEAVE ${t[1]}")
235 }
236 "CURRENCY" -> {
237 val value = t[2].toLong()
238 val old = currencies.put(t[1],value)
239 output.add("CURRENCY ${t[1]} ${rate(value)}"+if (old == null) "" else " was ${rate(old)}")
240 }
241 "EXPENSE", "EDIT" -> {
242 val old = if (command == "EDIT") expense(t[1]).also(::thawed) else null
243 if (old == null && t[1] in used) fail("duplicate id ${t[1]}")
244 val e = build(t)
245 e.id = t[1]
246 e.day = old?.day ?: day
247 e.record = old?.record ?: ++serial
248 e.interval = old?.interval ?: 0
249 e.next = old?.next ?: 0
250 e.copies = old?.copies ?: 0
251 expenses[e.id] = e
252 used.add(e.id)
253 output.add("$command ${e.id} ${fmt(e.total)}${note(e.amount)} = ${pairs(e.participants,e.shares)}")
254 warning(e,output)
255 }
256 "PAYERS" -> {
257 val e = expense(t[1])
258 thawed(e)
259 val seen = mutableSetOf<String>()
260 val payers = t.drop(2).map { token ->
261 val (name,value) = token.split(':')
262 active(name)
263 if (!seen.add(name)) fail("duplicate member $name")
264 Part(name,money(value))
265 }
266 if (payers.sumOf { it.value } != e.amount.value) fail("payers do not add up")
267 val contributions = distribute(e.total,payers.map { it.value })
268 e.payers = payers
269 e.contributions = contributions
270 output.add("PAYERS ${e.id} ${fmt(e.total)}${note(e.amount)} = ${pairs(payers,contributions)}")
271 }
272 "DELETE" -> {
273 val e = expense(t[1])
274 thawed(e)
275 expenses.remove(e.id)
276 output.add("DELETE ${e.id}")
277 }
278 "RECURRING" -> {
279 val e = expense(t[1])
280 e.interval = t[2].toLong()
281 e.next = day+e.interval
282 output.add("RECURRING ${e.id}"+if (e.interval == 0L) " stopped" else " interval ${e.interval} next ${e.next}")
283 }
284 "SETTLE" -> {
285 active(t[1])
286 active(t[2])
287 if (t[1] == t[2]) fail("same member ${t[1]}")
288 val a = amount(t[3])
289 val total = convert(a)
290 if (a.value == 0L || total == 0L) fail("invalid amount")
291 settlements.add(Settlement(day,++serial,t[1],t[2],a,total))
292 output.add("SETTLE ${t[1]} -> ${t[2]} ${fmt(total)}${note(a)}")
293 }
294 "BUDGET" -> {
295 val value = money(t[2])
296 if (value == 0L) budgets.remove(t[1]) else budgets[t[1]] = value
297 output.add("BUDGET ${t[1]}"+if (value == 0L) " removed" else " ${fmt(value)}")
298 }
299 "BALANCES" -> {
300 output.add("BALANCES")
301 val balances = balances()
302 for (name in activeNames()) output.add(" $name ${signed(balances.getValue(name))}")
303 }
304 "OWES", "MATRIX" -> {
305 if (command == "OWES") registered(t[1])
306 val names = members.keys.toList()
307 val debt = debts(names)
308 output.add(if (command == "OWES") "OWES ${t[1]}" else "MATRIX")
309 if (command == "OWES") {
310 val i = names.indexOf(t[1])
311 for (j in names.indices) {
312 if (debt[i][j] > 0) output.add(" ${t[1]} owes ${names[j]} ${fmt(debt[i][j])}")
313 if (debt[i][j] < 0) output.add(" ${names[j]} owes ${t[1]} ${fmt(-debt[i][j])}")
314 }
315 } else {
316 val active = activeNames()
317 if (active.isEmpty()) output.add(" (none)")
318 else {
319 val l = active.maxOf { it.length }
320 fun cell(a: String,b: String) = if (a == b) "-" else fmt(maxOf(0,debt[names.indexOf(a)][names.indexOf(b)]))
321 val width = maxOf(l,active.maxOf { a -> active.maxOf { b -> cell(a,b).length } })
322 output.add(" ".repeat(l)+active.joinToString("") { " "+it.padStart(width) })
323 for (a in active) output.add(a.padEnd(l)+active.joinToString("") { " "+cell(a,it).padStart(width) })
324 }
325 }
326 }
327 "HISTORY" -> {
328 registered(t[1])
329 val entries = mutableListOf<Entry>()
330 for (e in expenses.values) {
331 if (includes(e,t[1])) entries.add(Entry(e.day,e.record,e.id,net(e).getValue(t[1])))
332 }
333 for (s in settlements) {
334 if (t[1] == s.from || t[1] == s.to) entries.add(Entry(s.day,s.record,"${s.from}->${s.to}",if (t[1] == s.from) s.total else -s.total))
335 }
336 output.add("HISTORY ${t[1]}")
337 var balance = 0L
338 for (e in entries.sortedWith(compareBy<Entry> { it.day }.thenBy { it.record })) {
339 balance += e.change
340 output.add(" day ${e.day} ${e.label} ${signed(e.change)} balance ${signed(balance)}")
341 }
342 }
343 "CATEGORY" -> {
344 output.add("CATEGORY")
345 val totals = categoryTotals()
346 val all = totals.values.sum()
347 for (category in totals.keys.sortedWith(compareByDescending<String> { totals.getValue(it) }.thenBy { it })) {
348 val total = totals.getValue(category)
349 output.add(" $category ${fmt(total)} ${percent(total,all)}")
350 val shares = mutableMapOf<String,Long>()
351 for (e in expenses.values.filter { it.category == category }) {
352 for ((i,p) in e.participants.withIndex()) shares[p.name] = shares.getOrDefault(p.name,0)+e.shares[i]
353 }
354 for (name in shares.keys.filter { shares.getValue(it) > 0 }.sortedWith(compareByDescending<String> { shares.getValue(it) }.thenBy { it })) {
355 output.add(" $name ${fmt(shares.getValue(name))} ${percent(shares.getValue(name),total)}")
356 }
357 }
358 }
359 "BUDGETS" -> {
360 output.add("BUDGETS")
361 val totals = categoryTotals()
362 for ((category,budget) in budgets) {
363 val spent = totals.getOrDefault(category,0)
364 output.add(" $category spent ${fmt(spent)} budget ${fmt(budget)} used ${percent(spent,budget)}"+
365 if (spent <= budget) " left ${fmt(budget-spent)}" else " over ${fmt(spent-budget)}")
366 }
367 }
368 "TOTALS" -> {
369 output.add("TOTALS")
370 for (name in members.keys) {
371 var paid = 0L
372 var share = 0L
373 for (e in expenses.values) {
374 for ((i,p) in e.payers.withIndex()) {
375 if (p.name == name) paid += e.contributions[i]
376 }
377 for ((i,p) in e.participants.withIndex()) {
378 if (p.name == name) share += e.shares[i]
379 }
380 }
381 val sent = settlements.filter { it.from == name }.sumOf { it.total }
382 val received = settlements.filter { it.to == name }.sumOf { it.total }
383 output.add(" $name paid ${fmt(paid)} share ${fmt(share)} sent ${fmt(sent)} received ${fmt(received)}")
384 }
385 for (currency in currencies.keys) {
386 val group = expenses.values.filter { it.amount.currency == currency }
387 if (group.isNotEmpty()) output.add(" currency $currency expenses ${group.size} amount ${fmt(group.sumOf { it.amount.value })} base ${fmt(group.sumOf { it.total })}")
388 }
389 output.add(" expenses ${expenses.size} total ${fmt(expenses.values.sumOf { it.total })} settlements ${settlements.size} total ${fmt(settlements.sumOf { it.total })}")
390 }
391 "MEMBERS" -> {
392 output.add("MEMBERS")
393 for ((name,m) in members) {
394 val count = expenses.values.count { includes(it,name) }
395 output.add(" $name ${if (m.active) "active" else "left"} day ${m.day} expenses $count")
396 }
397 }
398 "EXPENSES" -> {
399 if (t[1] == "MEMBER") registered(t[2])
400 output.add(line)
401 var count = 0
402 var total = 0L
403 for (e in chronological()) {
404 val matches = when (t[1]) {
405 "CATEGORY" -> e.category == t[2]
406 "MEMBER" -> includes(e,t[2])
407 "DAYS" -> e.day in t[2].toLong()..t[3].toLong()
408 else -> true
409 }
410 if (matches) {
411 output.add(" day ${e.day} ${e.id} ${e.category} ${fmt(e.total)} by ${e.payers.joinToString(",") { it.name }}")
412 count++
413 total += e.total
414 }
415 }
416 output.add(" count $count total ${fmt(total)}")
417 }
418 "PERIODS" -> {
419 val k = t[1].toLong()
420 output.add("PERIODS $k")
421 val groups = expenses.values.groupBy { (it.day-1)/k }.toSortedMap()
422 for ((i,group) in groups) output.add(" days ${i*k+1}-${i*k+k} expenses ${group.size} total ${fmt(group.sumOf { it.total })}")
423 }
424 "SHOW" -> {
425 val e = expense(t[1])
426 output.add("SHOW ${e.id}")
427 output.add(" day ${e.day} category ${e.category} mode ${e.mode} total ${fmt(e.total)}${note(e.amount)}")
428 for ((i,p) in e.payers.withIndex()) output.add(" paid ${p.name} ${fmt(e.contributions[i])}")
429 val sum = e.participants.sumOf { it.value }
430 for ((i,p) in e.participants.withIndex()) {
431 val suffix = when (e.mode) {
432 "EXACT" -> " ${fmt(p.value)}"
433 "PERCENT" -> " ${fmt(p.value)}%"
434 "SHARES" -> " ${p.value}/$sum"
435 "ADJUST" -> " +${fmt(p.value)}"
436 else -> ""
437 }
438 output.add(" share ${p.name} ${fmt(e.shares[i])}$suffix")
439 }
440 if (e.interval > 0) output.add(" recurring interval ${e.interval} next ${e.next}")
441 }
442 "SIMPLIFY" -> {
443 val all = balances().filterValues { it != 0L }
444 if (t[1] == "GREEDY") {
445 val transfers = greedy(all)
446 output.add("SIMPLIFY GREEDY transfers ${transfers.size}")
447 for (x in transfers) output.add(" ${x.from} -> ${x.to} ${fmt(x.amount)}")
448 } else {
449 val names = all.keys.toList()
450 val n = names.size
451 if (n > 12) fail("too many members")
452 val size = 1 shl n
453 val sums = LongArray(size)
454 val dp = IntArray(size) { -1000 }
455 dp[0] = 0
456 for (mask in 1 until size) {
457 val bit = Integer.numberOfTrailingZeros(mask)
458 sums[mask] = sums[mask xor (1 shl bit)]+all.getValue(names[bit])
459 val first = mask and -mask
460 var sub = mask
461 while (sub > 0) {
462 if (sub and first != 0 && sums[sub] == 0L) dp[mask] = maxOf(dp[mask],1+dp[mask xor sub])
463 sub = (sub-1) and mask
464 }
465 }
466 val groups = mutableListOf<Int>()
467 var remaining = size-1
468 while (remaining != 0) {
469 val first = remaining and -remaining
470 var best = 0
471 var sub = remaining
472 while (sub > 0) {
473 if (sub and first != 0 && sums[sub] == 0L && 1+dp[remaining xor sub] == dp[remaining] &&
474 (best == 0 || lexLess(indexes(sub,n),indexes(best,n)))) best = sub
475 sub = (sub-1) and remaining
476 }
477 groups.add(best)
478 remaining = remaining xor best
479 }
480 val transfers = groups.map { mask -> greedy(indexes(mask,n).associate { names[it] to all.getValue(names[it]) }) }
481 output.add("SIMPLIFY OPTIMAL transfers ${transfers.sumOf { it.size }} groups ${groups.size}")
482 for (i in groups.indices) {
483 output.add(" group ${indexes(groups[i],n).joinToString(" ") { names[it] }}")
484 for (x in transfers[i]) output.add(" ${x.from} -> ${x.to} ${fmt(x.amount)}")
485 }
486 }
487 }
488 }
489 if (command in setOf("BALANCES","OWES","HISTORY","CATEGORY","BUDGETS","MEMBERS","PERIODS") && output.size == 1) {
490 output.add(" (none)")
491 }
492 return output
493 }
494}

496fun main() {
497 val input = System.`in`.bufferedReader()
498 val output = System.out.bufferedWriter()
499 val app = Splitter()
500 repeat(input.readLine().toInt()) {
501 val lines = try {
502 app.execute(input.readLine())
503 } catch (problem: Problem) {
504 listOf("ERROR ${problem.message}")
505 }
506 for (line in lines) {
507 output.write(line)
508 output.newLine()
509 }
510 }
511 output.flush()
512}
