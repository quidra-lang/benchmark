74static char *copy_string(const char *s) { return slice(s, strlen(s)); }
75static char *format(const char *pattern, ...) {
76 va_list args, copy;
77 va_start(args, pattern);
78 va_copy(copy, args);
79 int length = vsnprintf(NULL, 0, pattern, copy);
80 va_end(copy);
81 char *s = allocate((size_t)length + 1);
82 vsnprintf(s, (size_t)length + 1, pattern, args);
83 va_end(args);
84 return s;

86static void push_owned(Strings *s, char *item) {
87 if (s->count == s->capacity) {
88 s->capacity = s->capacity ? s->capacity * 2 : 8;
89 s->items = realloc(s->items, (size_t)s->capacity * sizeof *s->items);
90 if (!s->items) exit(EXIT_FAILURE);
91 }
92 s->items[s->count++] = item;
93}
94static void push(Strings *s, const char *item) { push_owned(s, copy_string(item)); }
95static void free_strings(Strings *s) {
96 for (int i = 0; i < s->count; ++i) free(s->items[i]);
97 free(s->items);
98 *s = (Strings){0};
99}
100static Strings split(const char *s, char delimiter, int limit) {
101 Strings result = {0};
102 const char *start = s;
103 for (const char *p = s;; ++p) {
104 if (!*p || (*p == delimiter && (!limit || result.count + 1 < limit))) {
105 push_owned(&result, slice(start, (size_t)(p - start)));
106 start = p + 1;
107 if (!*p) break;
108 }
109 }
110 return result;
111}
112static char *join(const Strings *s, const char *separator) {
113 size_t size = 1, sep = strlen(separator);
114 for (int i = 0; i < s->count; ++i) size += strlen(s->items[i]) + (i ? sep : 0);
115 char *result = allocate(size), *p = result;
116 for (int i = 0; i < s->count; ++i) {
117 if (i) {
118 memcpy(p, separator, sep);
119 p += sep;
120 }
121 size_t n = strlen(s->items[i]);
122 memcpy(p, s->items[i], n);
123 p += n;
124 }
125 *p = '\0';
126 return result;
127}

154static Integer convert(Integer value, const char *currency) {
155 if (!*currency) return value;
156 for (int i = 0; i < currency_count; ++i) {
157 if (!strcmp(currencies[i].code, currency)) return (value * currencies[i].rate + 5000) / 10000;
158 }
159 named_error("unknown currency", currency);
160 return 0;
161}

168static int member_id(const char *name, bool active) {
169 for (int i = 0; i < member_count; ++i) {
170 if (!strcmp(members[i].name, name)) {
171 if (active && !members[i].active) named_error("inactive member", name);
172 return i;
173 }
174 }
175 named_error("unknown member", name);
176 return -1;

178static int ordered_members(int *order, bool active) {
179 int count = 0;
180 for (int i = 0; i < member_count; ++i) {
181 if (active && !members[i].active) continue;
182 int j = count;
183 while (j > 0 && strcmp(members[order[j - 1]].name, members[i].name) > 0) {
184 order[j] = order[j - 1];
185 --j;
186 }
187 order[j] = i;
188 ++count;
189 }
190 return count;
191}

197static int frozen_member(const Expense *e) {
198 for (int i = 0; i < e->payer_count; ++i) if (!members[e->payers[i].member].active) return e->payers[i].member;
199 for (int i = 0; i < e->participant_count; ++i) if (!members[e->participants[i].member].active) return e->participants[i].member;
200 return -1;
201}
202static void check_frozen(const Expense *e) {
203 int id = frozen_member(e);
204 if (id >= 0) named_error("inactive member", members[id].name);
205}
206static void distribute(Integer total, const Integer *weights, int count, Integer *result) {
207 Integer sum = 0, remainders[30], left = total;
208 int order[30];
209 for (int i = 0; i < count; ++i) sum += weights[i];
210 for (int i = 0; i < count; ++i) {
211 Integer product = total * weights[i];
212 result[i] = product / sum;
213 remainders[i] = product % sum;
214 left -= result[i];
215 order[i] = i;
216 }
217 for (int i = 1; i < count; ++i) {
218 int id = order[i], j = i;
219 while (j > 0 && remainders[order[j - 1]] < remainders[id]) {
220 order[j] = order[j - 1];
221 --j;
222 }
223 order[j] = id;
224 }
225 for (int i = 0; i < left; ++i) ++result[order[i]];
226}
227static void equal_distribution(Integer total, int count, Integer *result) {
228 for (int i = 0; i < count; ++i) result[i] = total / count + (i < total % count ? 1 : 0);
229}
230static void contributions(Expense *e) {
231 Integer weights[30], values[30];
232 for (int i = 0; i < e->payer_count; ++i) weights[i] = e->payers[i].value;
233 distribute(e->total, weights, e->payer_count, values);
234 for (int i = 0; i < e->payer_count; ++i) e->payers[i].result = values[i];
235}
236static void compute(Expense *e) {
237 e->total = convert(e->amount, e->currency);
238 Integer specified[30], portions[30], shares[30], sum = 0;
239 for (int i = 0; i < e->participant_count; ++i) {
240 specified[i] = e->participants[i].value;
241 sum += specified[i];
242 }
243 if (!strcmp(e->mode, "EQUAL")) equal_distribution(e->amount, e->participant_count, portions);
244 else if (!strcmp(e->mode, "EXACT")) memcpy(portions, specified, (size_t)e->participant_count * sizeof *portions);
245 else if (!strcmp(e->mode, "ADJUST")) {
246 equal_distribution(e->amount - sum, e->participant_count, portions);
247 for (int i = 0; i < e->participant_count; ++i) portions[i] += specified[i];
248 } else distribute(e->amount, specified, e->participant_count, portions);
249 distribute(e->total, portions, e->participant_count, shares);
250 for (int i = 0; i < e->participant_count; ++i) e->participants[i].result = shares[i];
251 contributions(e);
252}
253static Expense draft_expense(const Strings *a, const Expense *previous) {
254 Expense e = previous ? *previous : (Expense){0};
255 int payer = member_id(a->items[2], true);
256 amount_spec(a->items[3], &e.amount, e.currency);
257 Integer total = convert(e.amount, e.currency);
258 if (!e.amount || !total) fail("invalid amount");
259 strcpy(e.category, a->items[4]);
260 strcpy(e.mode, a->items[5]);
261 e.payer_count = 1;
262 e.payers[0] = (Part){payer, e.amount, 0};
263 e.participant_count = 0;
264 if (!strcmp(e.mode, "EQUAL") && !strcmp(a->items[6], "ALL")) {
265 int order[30], count = ordered_members(order, true);
266 for (int i = 0; i < count; ++i) e.participants[e.participant_count++] = (Part){order[i], 0, 0};
267 } else {
268 bool seen[30] = {false};
269 for (int i = 6; i < a->count; ++i) {
270 const char *colon = strchr(a->items[i], ':');
271 char name[11];
272 size_t length = colon ? (size_t)(colon - a->items[i]) : strlen(a->items[i]);
273 memcpy(name, a->items[i], length);
274 name[length] = '\0';
275 int id = member_id(name, true);
276 if (seen[id]) named_error("duplicate member", name);
277 seen[id] = true;
278 Integer value = 0;
279 if (colon) value = !strcmp(e.mode, "PERCENT") || !strcmp(e.mode, "SHARES") ? integer(colon + 1) : amount(colon + 1);
280 e.participants[e.participant_count++] = (Part){id, value, 0};
281 }
282 }
283 Integer sum = 0;
284 for (int i = 0; i < e.participant_count; ++i) sum += e.participants[i].value;
285 if ((!strcmp(e.mode, "EXACT") && sum != e.amount) || (!strcmp(e.mode, "PERCENT") && sum != 10000) || (!strcmp(e.mode, "SHARES") && sum == 0) || (!strcmp(e.mode, "ADJUST") && sum > e.amount)) fail("splits do not add up");
286 compute(&e);
287 return e;
288}
289static void net(const Expense *e, Integer *result) {
290 memset(result, 0, 30 * sizeof *result);
291 for (int i = 0; i < e->payer_count; ++i) result[e->payers[i].member] += e->payers[i].result;
292 for (int i = 0; i < e->participant_count; ++i) result[e->participants[i].member] -= e->participants[i].result;
293}
294static bool involved(const Expense *e, int id) {
295 for (int i = 0; i < e->payer_count; ++i) if (e->payers[i].member == id) return true;
296 for (int i = 0; i < e->participant_count; ++i) if (e->participants[i].member == id) return true;
297 return false;
298}
299static void balances(Integer *result) {
300 memset(result, 0, 30 * sizeof *result);
301 for (int i = 0; i < expense_count; ++i) {
302 Integer change[30];
303 net(&expenses[i], change);
304 for (int j = 0; j < member_count; ++j) result[j] += change[j];
305 }
306 for (int i = 0; i < settlement_count; ++i) {
307 result[settlements[i].from] += settlements[i].total;
308 result[settlements[i].to] -= settlements[i].total;
309 }
310}
311static void debts(Integer result[30][30]) {
312 memset(result, 0, 30 * 30 * sizeof(Integer));
313 int order[30], count = ordered_members(order, false);
314 for (int i = 0; i < expense_count; ++i) {
315 Integer change[30];
316 net(&expenses[i], change);
317 int borrowers[30], lenders[30], nb = 0, nl = 0;
318 for (int j = 0; j < count; ++j) {
319 int id = order[j];
320 if (change[id] < 0) borrowers[nb++] = id;
321 if (change[id] > 0) lenders[nl++] = id;
322 }
323 int b = 0, l = 0;
324 while (b < nb && l < nl) {
325 int from = borrowers[b], to = lenders[l];
326 Integer value = -change[from] < change[to] ? -change[from] : change[to];
327 result[from][to] += value;
328 change[from] += value;
329 change[to] -= value;
330 if (!change[from]) ++b;
331 if (!change[to]) ++l;
332 }
333 }
334 for (int i = 0; i < settlement_count; ++i) result[settlements[i].from][settlements[i].to] -= settlements[i].total;
335 for (int i = 0; i < member_count; ++i) {
336 for (int j = i + 1; j < member_count; ++j) {
337 Integer value = result[i][j] - result[j][i];
338 result[i][j] = value;
339 result[j][i] = -value;
340 }
341 }
342}
343static Integer spent(const char *category) {
344 Integer total = 0;

349 for (int i = 0; i < budget_count; ++i) {

371static int chronological(int *order) {
372 for (int i = 0; i < expense_count; ++i) {
373 int j = i;
374 while (j > 0 && (expenses[order[j - 1]].day > expenses[i].day || (expenses[order[j - 1]].day == expenses[i].day && expenses[order[j - 1]].serial > expenses[i].serial))) {
375 order[j] = order[j - 1];
376 --j;
377 }

382static int greedy(const int *group, int count, const Integer *original, Transfer *result) {
383 Integer values[30];
384 memcpy(values, original, sizeof values);
385 int length = 0;
386 for (;;) {
387 int from = -1, to = -1;
388 for (int i = 0; i < count; ++i) {
389 int id = group[i];
390 if (values[id] < 0 && (from < 0 || values[id] < values[from])) from = id;
391 if (values[id] > 0 && (to < 0 || values[id] > values[to])) to = id;
392 }
393 if (from < 0) break;
394 Integer value = -values[from] < values[to] ? -values[from] : values[to];
395 result[length++] = (Transfer){from, to, value};
396 values[from] += value;
397 values[to] -= value;
398 }
399 return length;
400}

408static int best_groups(int mask, const Integer *sums, int *memo) {
409 if (memo[mask] >= 0) return memo[mask];
410 int first = mask & -mask, result = 0;
411 for (int sub = mask; sub; sub = (sub - 1) & mask) {
412 if ((sub & first) && sums[sub] == 0) {
413 int candidate = 1 + best_groups(mask ^ sub, sums, memo);
414 if (candidate > result) result = candidate;
415 }
416 }
417 return memo[mask] = result;
418}
419static bool mask_less(int a, int b, int count) {
420 int i = 0, j = 0;
421 for (;;) {
422 while (i < count && !(a & (1 << i))) ++i;
423 while (j < count && !(b & (1 << j))) ++j;
424 if (i == count || j == count) return i == count && j != count;
425 if (i != j) return i < j;
426 ++i;
427 ++j;
428 }
429}

445static void execute(const Strings *a, const char *line, Strings *out) {
446 const char *command = a->items[0];
447 if (!strcmp(command, "DAY")) {
448 Integer day = integer(a->items[1]);
449 if (day <= current_day) named_error("invalid day", a->items[1]);
450 push_owned(out, format("DAY %" PRId64, day));
451 for (;;) {
452 int selected = -1;
453 for (int i = 0; i < expense_count; ++i) {
454 Expense *e = &expenses[i];
455 if (e->interval > 0 && e->next <= day && (selected < 0 || e->next < expenses[selected].next || (e->next == expenses[selected].next && e->serial < expenses[selected].serial))) selected = i;
456 }
457 if (selected < 0) break;
458 Expense *original = &expenses[selected];
459 if (frozen_member(original) >= 0 || convert(original->amount, original->currency) == 0) {
460 push_owned(out, format(" CANCEL %s", original->id));
461 original->interval = 0;
462 continue;
463 }
464 Expense copy = *original;
465 copy.id = format("%s#%" PRId64, original->id, original->copies + 2);
466 copy.day = original->next;
467 copy.serial = ++serial;
468 copy.interval = copy.next = copy.copies = 0;
469 ++original->copies;
470 original->next += original->interval;
471 compute(&copy);
472 used[used_count++] = copy.id;
473 expenses[expense_count++] = copy;
474 push_owned(out, format(" RECUR %s %s", copy.id, m(copy.total)));
475 warning(&copy, out);
476 }
477 current_day = day;
478 } else if (!strcmp(command, "MEMBER")) {
479 int id = -1;
480 for (int i = 0; i < member_count; ++i) if (!strcmp(members[i].name, a->items[1])) id = i;
481 if (id >= 0 && members[id].active) named_error("duplicate member", a->items[1]);
482 bool rejoined = id >= 0;
483 if (id < 0) {
484 id = member_count++;
485 strcpy(members[id].name, a->items[1]);
486 }
487 members[id].active = true;
488 members[id].day = current_day;
489 push_owned(out, format("MEMBER %s %s", a->items[1], rejoined ? "rejoined" : "joined"));
490 } else if (!strcmp(command, "LEAVE")) {
491 int id = member_id(a->items[1], true);
492 Integer values[30];
493 balances(values);
494 if (values[id]) named_error("member has a balance", a->items[1]);
495 members[id].active = false;
496 members[id].day = current_day;
497 push_owned(out, format("LEAVE %s", a->items[1]));
498 } else if (!strcmp(command, "CURRENCY")) {
499 Integer value = integer(a->items[2]);
500 int id = -1;
501 for (int i = 0; i < currency_count; ++i) if (!strcmp(currencies[i].code, a->items[1])) id = i;
502 const char *new_rate = keep(rate_text(value));
503 if (id < 0) {
504 push_owned(out, format("CURRENCY %s %s", a->items[1], new_rate));
505 id = currency_count++;
506 strcpy(currencies[id].code, a->items[1]);
507 } else push_owned(out, format("CURRENCY %s %s was %s", a->items[1], new_rate, keep(rate_text(currencies[id].rate))));
508 currencies[id].rate = value;
509 } else if (!strcmp(command, "EXPENSE") || !strcmp(command, "EDIT")) {
510 int id = -1;
511 if (!strcmp(command, "EXPENSE")) {
512 for (int i = 0; i < used_count; ++i) if (!strcmp(used[i], a->items[1])) named_error("duplicate id", a->items[1]);
513 } else {
514 id = expense_id(a->items[1]);
515 check_frozen(&expenses[id]);
516 }
517 Expense e = draft_expense(a, id < 0 ? NULL : &expenses[id]);
518 if (id < 0) {
519 e.id = copy_string(a->items[1]);
520 e.day = current_day;
521 e.serial = ++serial;
522 used[used_count++] = e.id;
523 expenses[expense_count++] = e;
524 } else expenses[id] = e;
525 const char *parts = keep(parts_text(e.participants, e.participant_count));
526 push_owned(out, format("%s %s %s%s = %s", command, e.id, m(e.total), currency_note(e.amount, e.currency), parts));
527 warning(&e, out);
528 } else if (!strcmp(command, "PAYERS")) {
529 int id = expense_id(a->items[1]);
530 check_frozen(&expenses[id]);
531 Part payers[30];
532 int count = 0;
533 Integer sum = 0;
534 bool seen[30] = {false};
535 for (int i = 2; i < a->count; ++i) {
536 const char *colon = strchr(a->items[i], ':');
537 char name[11];
538 memcpy(name, a->items[i], (size_t)(colon - a->items[i]));
539 name[colon - a->items[i]] = '\0';
540 int member = member_id(name, true);
541 if (seen[member]) named_error("duplicate member", name);
542 seen[member] = true;
543 Integer value = amount(colon + 1);
544 sum += value;
545 payers[count++] = (Part){member, value, 0};
546 }
547 Expense *e = &expenses[id];
548 if (sum != e->amount) fail("payers do not add up");
549 memcpy(e->payers, payers, (size_t)count * sizeof *payers);
550 e->payer_count = count;
551 contributions(e);
552 push_owned(out, format("PAYERS %s %s%s = %s", e->id, m(e->total), currency_note(e->amount, e->currency), keep(parts_text(e->payers, e->payer_count))));
553 } else if (!strcmp(command, "DELETE")) {
554 int id = expense_id(a->items[1]);
555 check_frozen(&expenses[id]);

888static char *read_line(void) {
889 size_t capacity = 256, length = 0;
890 char *line = allocate(capacity);
891 int c;
892 while ((c = getchar()) != EOF && c != '\n') {
893 if (length + 1 == capacity) {
894 capacity *= 2;
895 line = realloc(line, capacity);
896 if (!line) exit(EXIT_FAILURE);
897 }
898 line[length++] = (char)c;
899 }
900 if (!length && c == EOF) {
901 free(line);
902 return NULL;
903 }
904 line[length] = '\0';
905 return line;
906}
907int main(void) {
908 char *line = read_line();
909 if (!line) return 0;
910 int count = atoi(line);
911 free(line);
912 for (int i = 0; i < count; ++i) {
913 line = read_line();
914 if (!line) break;
915 Strings a = split(line, ' ', 0);
916 if (setjmp(recovery) == 0) {
917 Strings out = {0};
918 execute(&a, line, &out);
919 for (int j = 0; j < out.count; ++j) puts(out.items[j]);
920 free_strings(&out);
921 } else {
922 printf("ERROR %s\n", error_message);
923 free(error_message);
924 error_message = NULL;
925 }
926 free_strings(&temporary);
927 free_strings(&a);
928 free(line);
929 }
930 for (int i = 0; i < used_count; ++i) free(used[i]);
931 return 0;
