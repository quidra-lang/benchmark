110function equal_distribution(int $total, int $k): array
111{
112 $result = [];
113 for ($i = 0; $i < $k; ++$i) {
114 $result[] = intdiv($total, $k) + ($i < $total % $k ? 1 : 0);
115 }
116 return $result;

119function largest_remainder(int $total, array $weights): array
120{
121 $whole = array_sum($weights);
122 $result = array_map(fn($weight) => intdiv($total * $weight, $whole), $weights);
123 $order = array_keys($weights);
124 usort($order, function($a, $b) use ($total, $weights, $whole) {
125 $ra = $total * $weights[$a] % $whole;
126 $rb = $total * $weights[$b] % $whole;
127 return $ra === $rb ? $a <=> $b : $rb <=> $ra;
128 });
129 $remaining = $total - array_sum($result);
130 for ($i = 0; $i < $remaining; ++$i) {
131 ++$result[$order[$i]];
132 }
133 return $result;
134}

136function portions(array $expense): array
137{
138 $values = $expense['values'];
139 if ($expense['mode'] === 'EQUAL') {
140 return equal_distribution($expense['amount'], count($values));
141 }
142 if ($expense['mode'] === 'EXACT') {
143 return $values;
144 }
145 if (in_array($expense['mode'], ['PERCENT', 'SHARES'], true)) {
146 return largest_remainder($expense['amount'], $values);
147 }
148 $base = equal_distribution($expense['amount'] - array_sum($values), count($values));
149 return array_map(fn($a, $b) => $a + $b, $base, $values);
150}

152function recompute(array $state, array $expense): array
153{
154 $expense['total'] = converted($state, $expense['amount'], $expense['currency']);
155 $expense['shares'] = largest_remainder($expense['total'], portions($expense));
156 $expense['contributions'] = largest_remainder($expense['total'], $expense['payments']);
157 return $expense;
158}

160function build_expense(array $state, string $id, string $payer, string $spec, string $category, string $mode, array $parts): array
161{
162 check_member($state, $payer);
163 [$amount, $currency] = amount_spec($spec);
164 $total = converted($state, $amount, $currency);
165 if ($amount === 0 || $total === 0) {
166 fail('invalid amount');
167 }
168 $names = [];
169 $values = [];
170 if ($mode === 'EQUAL' && $parts === ['ALL']) {
171 $names = active_names($state);
172 $values = array_fill(0, count($names), 0);
173 } else {
174 foreach ($parts as $part) {
175 $pieces = explode(':', $part);
176 $name = $pieces[0];
177 check_member($state, $name);
178 if (in_array($name, $names, true)) {
179 fail("duplicate member $name");
180 }
181 $value = 0;
182 if (isset($pieces[1])) {
183 $value = in_array($mode, ['EXACT', 'ADJUST'], true) ? parse_amount($pieces[1]) : (int) $pieces[1];
184 }
185 $names[] = $name;
186 $values[] = $value;
187 }
188 }
189 $sum = array_sum($values);
190 $valid = match ($mode) {
191 'EXACT' => $sum === $amount,
192 'PERCENT' => $sum === 10000,
193 'SHARES' => $sum >= 1,
194 'ADJUST' => $sum <= $amount,
195 default => true
196 };
197 if (!$valid) {
198 fail('splits do not add up');
199 }
200 return recompute($state, ['id' => $id, 'day' => $state['day'], 'record' => 0, 'payers' => [$payer],
201 'payments' => [$amount], 'amount' => $amount, 'currency' => $currency, 'category' => $category,
202 'mode' => $mode, 'participants' => $names, 'values' => $values, 'total' => $total, 'shares' => [],
203 'contributions' => [], 'interval' => 0, 'next' => 0, 'copies' => 0]);
204}

219 $result[$settlement['receiver']] -= $settlement['total'];

236function greedy_transfers(array $names, array $original): array
237{
238 $remaining = array_combine($names,array_map(fn($name) => $original[$name],$names));
239 $output = [];
240 while ($remaining !== [] && min($remaining) < 0) {
241 $sender = array_search(min($remaining),$remaining,true);
242 $receiver = array_search(max($remaining),$remaining,true);
243 $amount = min(-$remaining[$sender], $remaining[$receiver]);
244 $output[] = [$sender, $receiver, $amount];
245 $remaining[$sender] += $amount;
246 $remaining[$receiver] -= $amount;
247 }
248 return $output;
249}

251function debts(array $state): array
252{
253 $result = [];
254 foreach ($state['expenses'] as $expense) {
255 $net = expense_net($expense);
256 $borrowers = array_keys(array_filter($net, fn($value) => $value < 0));
257 $lenders = array_keys(array_filter($net, fn($value) => $value > 0));
258 sort($borrowers, SORT_STRING);
259 sort($lenders, SORT_STRING);
260 $i = 0;
261 $j = 0;
262 while ($i < count($borrowers) && $j < count($lenders)) {
263 $sender = $borrowers[$i];
264 $receiver = $lenders[$j];
265 $amount = min(-$net[$sender], $net[$receiver]);
266 $result[$sender][$receiver] = ($result[$sender][$receiver] ?? 0) + $amount;
267 $net[$sender] += $amount;
268 $net[$receiver] -= $amount;
269 if ($net[$sender] === 0) {
270 ++$i;
271 }
272 if ($net[$receiver] === 0) {
273 ++$j;
274 }
275 }
276 }
277 foreach ($state['settlements'] as $settlement) {
278 $a = $settlement['sender'];
279 $b = $settlement['receiver'];
280 $result[$a][$b] = ($result[$a][$b] ?? 0) - $settlement['total'];
281 }
282 return $result;
283}

318function advance_day(array &$state, int $day): array
319{
320 if ($day <= $state['day']) {
321 fail("invalid day $day");
322 }
323 $output = ["DAY $day"];
324 while (true) {
325 $due = array_values(array_filter($state['expenses'], fn($e) => $e['interval'] > 0 && $e['next'] <= $day));
326 if ($due === []) {
327 break;
328 }
329 usort($due, fn($a, $b) => [$a['next'], $a['record']] <=> [$b['next'], $b['record']]);
330 $template = $due[0];
331 $id = $template['id'];
332 $active = array_all(expense_names($template),fn($name) => $state['members'][$name]['active']);
333 if (!$active || converted($state, $template['amount'], $template['currency']) === 0) {
334 $state['expenses'][$id]['interval'] = 0;
335 $state['expenses'][$id]['next'] = 0;
336 $output[] = " CANCEL $id";
337 continue;
338 }
339 $copyId = $id . '#' . ($template['copies'] + 2);
340 $copy = $template;
341 $copy['id'] = $copyId;
342 $copy['day'] = $template['next'];
343 $copy['record'] = ++$state['serial'];
344 $copy['interval'] = 0;
345 $copy['next'] = 0;
346 $copy['copies'] = 0;
347 $copy = recompute($state, $copy);
348 $state['expenses'][$copyId] = $copy;
349 $state['used'][$copyId] = true;
350 ++$state['expenses'][$id]['copies'];
351 $state['expenses'][$id]['next'] += $template['interval'];
352 $output[] = " RECUR $copyId " . amount_text($copy['total']);
353 $output = [...$output, ...budget_warning($state, $copy['category'])];
354 }
355 $state['day'] = $day;
356 return $output;
357}

359function vector_less(array $a, array $b): bool
360{
361 for ($i = 0; $i < min(count($a), count($b)); ++$i) {
362 if ($a[$i] !== $b[$i]) {
363 return $a[$i] < $b[$i];
364 }
365 }
366 return count($a) < count($b);
367}

369function optimal_groups(array $names, array $bal): array
370{
371 $n = count($names);
372 $limit = 1 << $n;
373 $sums = array_fill(0, $limit, 0);
374 $bitIndex = [];
375 for ($i = 0; $i < $n; ++$i) {
376 $bitIndex[1 << $i] = $i;
377 }
378 for ($mask = 1; $mask < $limit; ++$mask) {
379 $bit = $mask & -$mask;
380 $sums[$mask] = $sums[$mask ^ $bit] + $bal[$names[$bitIndex[$bit]]];
381 }
382 $memo = array_fill(0, $limit, -1);
383 $memo[0] = 0;
384 $best = function($mask) use (&$best, &$memo, $sums) {
385 if ($memo[$mask] >= 0) {
386 return $memo[$mask];
387 }
388 $first = $mask & -$mask;
389 $value = 0;
390 for ($subset = $mask; $subset !== 0; $subset = ($subset - 1) & $mask) {
391 if (($subset & $first) !== 0 && $sums[$subset] === 0) {
392 $value = max($value, 1 + $best($mask ^ $subset));
393 }
394 }
395 return $memo[$mask] = $value;
396 };
397 $groups = [];
398 $mask = $limit - 1;
399 while ($mask !== 0) {
400 $first = $mask & -$mask;
401 $selected = 0;
402 $selectedIndices = [];
403 $target = $best($mask);
404 for ($subset = $mask; $subset !== 0; $subset = ($subset - 1) & $mask) {
405 if (($subset & $first) !== 0 && $sums[$subset] === 0 && 1 + $best($mask ^ $subset) === $target) {
406 $indices = [];
407 for ($i = 0; $i < $n; ++$i) {
408 if (($subset & (1 << $i)) !== 0) {
409 $indices[] = $i;
410 }
411 }
412 if ($selected === 0 || vector_less($indices, $selectedIndices)) {
413 $selected = $subset;
414 $selectedIndices = $indices;
415 }
416 }
417 }
418 $groups[] = array_map(fn($i) => $names[$i], $selectedIndices);
419 $mask ^= $selected;
420 }
421 return $groups;
422}

424function ledger_summary(string $command, array $args, string $line, array $state): array
425{
426 $output = [$command];
427 $names = registered_names($state);
428 $expenses = chronological($state);
429 if ($command === 'BALANCES') {
430 $bal = balances($state);
431 foreach (active_names($state) as $name) {
432 $output[] = ' ' . $name . ' ' . signed_text($bal[$name]);
433 }
434 } elseif ($command === 'OWES') {
435 $name = $args[0];
436 check_member($state, $name, false);
437 $output[0] = "OWES $name";
438 $table = debts($state);
439 foreach ($names as $other) {
440 if ($other === $name) {
441 continue;
442 }
443 $amount = net_debt($table, $name, $other);
444 if ($amount > 0) {
445 $output[] = " $name owes $other " . amount_text($amount);
446 } elseif ($amount < 0) {
447 $output[] = " $other owes $name " . amount_text(-$amount);
448 }
449 }
450 } elseif ($command === 'HISTORY') {
451 $name = $args[0];
452 check_member($state, $name, false);
453 $output[0] = "HISTORY $name";
454 $records = [];
455 foreach ($expenses as $expense) {
456 if (expense_has($expense, $name)) {
457 $records[] = [$expense['day'], $expense['record'], $expense['id'], expense_net($expense)[$name]];
458 }
459 }
460 foreach ($state['settlements'] as $s) {
461 if ($name === $s['sender'] || $name === $s['receiver']) {
462 $records[] = [$s['day'], $s['record'], $s['sender'] . '->' . $s['receiver'], $name === $s['sender'] ? $s['total'] : -$s['total']];
463 }
464 }
465 usort($records, fn($a, $b) => [$a[0], $a[1]] <=> [$b[0], $b[1]]);
466 $running = 0;
467 foreach ($records as [$day, $record, $label, $change]) {
468 $running += $change;
469 $output[] = " day $day $label " . signed_text($change) . ' balance ' . signed_text($running);
470 }
471 } elseif ($command === 'MATRIX') {
472 $active = active_names($state);
473 if ($active === []) {
474 return ['MATRIX', ' (none)'];
475 }
476 $table = debts($state);
477 $cells = [];
478 $left = max(array_map('strlen', $active));
479 $width = $left;
480 foreach ($active as $a) {
481 $row = [];
482 foreach ($active as $b) {
483 $cell = $a === $b ? '-' : amount_text(max(0, net_debt($table, $a, $b)));
484 $width = max($width, strlen($cell));
485 $row[] = $cell;
486 }
487 $cells[] = $row;
488 }
489 $output[] = str_repeat(' ', $left) . implode('', array_map(fn($name) => ' ' . str_pad($name, $width, ' ', STR_PAD_LEFT), $active));
490 foreach ($active as $i => $name) {
491 $output[] = str_pad($name, $left) . implode('', array_map(fn($cell) => ' ' . str_pad($cell, $width, ' ', STR_PAD_LEFT), $cells[$i]));
492 }
493 } elseif ($command === 'CATEGORY') {
494 $totals = category_totals($state);
495 $whole = array_sum($totals);
496 $categories = array_keys($totals);
497 usort($categories, fn($a, $b) => $totals[$a] === $totals[$b] ? strcmp($a, $b) : $totals[$b] <=> $totals[$a]);
498 foreach ($categories as $category) {
499 $total = $totals[$category];
500 $output[] = " $category " . amount_text($total) . ' ' . percentage($total, $whole);
501 $shares = array_fill_keys($names, 0);
502 foreach ($expenses as $expense) {
503 if ($expense['category'] === $category) {
504 foreach ($expense['participants'] as $i => $name) {
505 $shares[$name] += $expense['shares'][$i];
506 }
507 }
508 }
509 $people = array_keys(array_filter($shares, fn($value) => $value > 0));
510 usort($people, fn($a, $b) => $shares[$a] === $shares[$b] ? strcmp($a, $b) : $shares[$b] <=> $shares[$a]);
511 foreach ($people as $name) {
512 $output[] = " $name " . amount_text($shares[$name]) . ' ' . percentage($shares[$name], $total);
513 }
514 }
515 } elseif ($command === 'BUDGETS') {
516 $totals = category_totals($state);
517 $categories = array_keys($state['budgets']);
518 sort($categories, SORT_STRING);
519 foreach ($categories as $category) {
520 $spent = $totals[$category] ?? 0;
521 $budget = $state['budgets'][$category];
522 $label = $spent <= $budget ? 'left' : 'over';
523 $output[] = " $category spent " . amount_text($spent) . ' budget ' . amount_text($budget) . ' used ' . percentage($spent, $budget) . " $label " . amount_text(abs($budget - $spent));
524 }
525 } elseif ($command === 'TOTALS') {
526 foreach ($names as $name) {
527 $paid = 0;
528 $share = 0;
529 $sent = 0;
530 $received = 0;
531 foreach ($expenses as $expense) {
532 $payer = array_search($name,$expense['payers'],true);
533 $participant = array_search($name,$expense['participants'],true);
534 if ($payer !== false) $paid += $expense['contributions'][$payer];
535 if ($participant !== false) $share += $expense['shares'][$participant];
536 }
537 foreach ($state['settlements'] as $s) {
538 if ($s['sender'] === $name) {
539 $sent += $s['total'];
540 }
541 if ($s['receiver'] === $name) {
542 $received += $s['total'];
543 }
544 }
545 $output[] = " $name paid " . amount_text($paid) . ' share ' . amount_text($share) . ' sent ' . amount_text($sent) . ' received ' . amount_text($received);
546 }
547 $currencies = array_values(array_unique(array_filter(array_column($expenses, 'currency'), fn($code) => $code !== '')));
548 sort($currencies, SORT_STRING);
549 foreach ($currencies as $currency) {
550 $selected = array_filter($expenses, fn($e) => $e['currency'] === $currency);
551 $output[] = " currency $currency expenses " . count($selected) . ' amount ' . amount_text(array_sum(array_column($selected, 'amount'))) . ' base ' . amount_text(array_sum(array_column($selected, 'total')));
552 }
553 $output[] = ' expenses ' . count($expenses) . ' total ' . amount_text(array_sum(array_column($expenses, 'total'))) . ' settlements ' . count($state['settlements']) . ' total ' . amount_text(array_sum(array_column($state['settlements'], 'total')));
554 } elseif ($command === 'MEMBERS') {
555 foreach ($names as $name) {
556 $member = $state['members'][$name];
557 $status = $member['active'] ? 'active' : 'left';
558 $count = count(array_filter($expenses, fn($e) => expense_has($e, $name)));
559 $output[] = " $name $status day $member[day] expenses $count";
560 }
561 } elseif ($command === 'EXPENSES') {
562 $output[0] = $line;
563 $mode = $args[0];
564 if ($mode === 'MEMBER') {
565 check_member($state, $args[1], false);
566 }
567 $selected = array_values(array_filter($expenses, function($e) use ($mode, $args) {
568 return match ($mode) {
569 'ALL' => true,
570 'CATEGORY' => $e['category'] === $args[1],
571 'MEMBER' => expense_has($e, $args[1]),
572 'DAYS' => $e['day'] >= (int) $args[1] && $e['day'] <= (int) $args[2]
573 };
574 }));
575 foreach ($selected as $e) {
576 $output[] = " day $e[day] $e[id] $e[category] " . amount_text($e['total']) . ' by ' . implode(',', $e['payers']);
577 }
578 $output[] = ' count ' . count($selected) . ' total ' . amount_text(array_sum(array_column($selected, 'total')));
579 } elseif ($command === 'PERIODS') {
580 $k = (int) $args[0];
581 $output[0] = "PERIODS $k";
582 $groups = [];
583 foreach ($expenses as $e) {
584 $groups[intdiv($e['day'] - 1, $k)][] = $e;
585 }
586 ksort($groups, SORT_NUMERIC);
587 foreach ($groups as $i => $group) {
588 $output[] = ' days ' . ($i * $k + 1) . '-' . ($i * $k + $k) . ' expenses ' . count($group) . ' total ' . amount_text(array_sum(array_column($group, 'total')));
589 }
590 } elseif ($command === 'SHOW') {
591 $e = expense_ref($state, $args[0]);
592 $output[0] = "SHOW $e[id]";
593 $output[] = " day $e[day] category $e[category] mode $e[mode] total " . amount_text($e['total']) . currency_note($e);
594 foreach ($e['payers'] as $i => $name) {
595 $output[] = " paid $name " . amount_text($e['contributions'][$i]);
596 }
597 foreach ($e['participants'] as $i => $name) {
598 $value = $e['values'][$i];
599 $suffix = match ($e['mode']) {
600 'EXACT' => ' ' . amount_text($value),
601 'PERCENT' => ' ' . amount_text($value) . '%',
602 'SHARES' => " $value/" . array_sum($e['values']),
603 'ADJUST' => ' +' . amount_text($value),
604 default => ''
605 };
606 $output[] = " share $name " . amount_text($e['shares'][$i]) . $suffix;
607 }
608 if ($e['interval'] > 0) {
609 $output[] = " recurring interval $e[interval] next $e[next]";
610 }
611 } elseif ($command === 'SIMPLIFY') {
612 $bal = balances($state);
613 $selected = array_values(array_filter($names, fn($name) => $bal[$name] !== 0));
614 if ($args[0] === 'GREEDY') {
615 $transfers = greedy_transfers($selected, $bal);
616 $output[0] = 'SIMPLIFY GREEDY transfers ' . count($transfers);
617 foreach ($transfers as [$sender, $receiver, $value]) {
618 $output[] = " $sender -> $receiver " . amount_text($value);
619 }
620 } else {
621 if (count($selected) > 12) {
622 fail('too many members');
623 }
624 $groups = optimal_groups($selected, $bal);
625 $transfers = array_map(fn($group) => greedy_transfers($group, $bal), $groups);
626 $count = array_sum(array_map('count', $transfers));
627 $output[0] = "SIMPLIFY OPTIMAL transfers $count groups " . count($groups);
628 foreach ($groups as $i => $group) {
629 $output[] = ' group ' . implode(' ', $group);
630 foreach ($transfers[$i] as [$sender, $receiver, $value]) {
631 $output[] = " $sender -> $receiver " . amount_text($value);
632 }
633 }
634 }
635 }
636 if (in_array($command, ['BALANCES', 'OWES', 'HISTORY', 'CATEGORY', 'BUDGETS', 'MEMBERS', 'PERIODS'], true) && count($output) === 1) {
637 $output[] = ' (none)';
638 }
639 return $output;
640}

642function execute(string $line, array &$state): array
643{
644 $tokens = explode(' ', $line);
645 $command = array_shift($tokens);
646 $args = $tokens;
647 if ($command === 'DAY') {
648 return advance_day($state, (int) $args[0]);
649 }
650 if ($command === 'MEMBER') {
651 $name = $args[0];
652 if (isset($state['members'][$name]) && $state['members'][$name]['active']) {
653 fail("duplicate member $name");
654 }
655 $label = isset($state['members'][$name]) ? 'rejoined' : 'joined';
656 $state['members'][$name] = ['active' => true, 'day' => $state['day']];
657 return ["MEMBER $name $label"];
658 }
659 if ($command === 'LEAVE') {
660 $name = $args[0];
661 check_member($state, $name);
662 if (balances($state)[$name] !== 0) {
663 fail("member has a balance $name");
664 }
665 $state['members'][$name] = ['active' => false, 'day' => $state['day']];
666 return ["LEAVE $name"];
667 }
668 if ($command === 'CURRENCY') {
669 [$currency, $rate] = [$args[0], (int) $args[1]];
670 $suffix = isset($state['currencies'][$currency]) ? ' was ' . rate_text($state['currencies'][$currency]) : '';
671 $state['currencies'][$currency] = $rate;
672 return ["CURRENCY $currency " . rate_text($rate) . $suffix];
673 }
674 if ($command === 'EXPENSE' || $command === 'EDIT') {
675 [$id, $payer, $spec, $category, $mode] = array_slice($args, 0, 5);
676 if ($command === 'EXPENSE') {
677 if (isset($state['used'][$id])) {
678 fail("duplicate id $id");
679 }
680 } else {
681 $original = expense_ref($state, $id, true);
682 }
683 $expense = build_expense($state, $id, $payer, $spec, $category, $mode, array_slice($args, 5));
684 if ($command === 'EDIT') {
685 foreach (['day', 'record', 'interval', 'next', 'copies'] as $key) {
686 $expense[$key] = $original[$key];
687 }
688 } else {
689 $expense['record'] = ++$state['serial'];
690 $state['used'][$id] = true;
691 }
692 $state['expenses'][$id] = $expense;
693 return [expense_line($command, $expense), ...budget_warning($state, $category)];
694 }
695 if ($command === 'PAYERS') {
696 $expense = expense_ref($state, $args[0], true);
697 $names = [];
698 $payments = [];
699 foreach (array_slice($args, 1) as $part) {
700 [$name, $amount] = explode(':', $part);
701 check_member($state, $name);
702 if (in_array($name, $names, true)) {
703 fail("duplicate member $name");
704 }
705 $names[] = $name;
706 $payments[] = parse_amount($amount);
707 }
708 if (array_sum($payments) !== $expense['amount']) {
709 fail('payers do not add up');
710 }
711 $expense['payers'] = $names;
712 $expense['payments'] = $payments;
713 $expense['contributions'] = largest_remainder($expense['total'], $payments);
714 $state['expenses'][$expense['id']] = $expense;
715 return [expense_line($command, $expense, true)];
716 }
717 if ($command === 'DELETE') {
718 expense_ref($state, $args[0], true);
719 unset($state['expenses'][$args[0]]);
720 return ["DELETE $args[0]"];
721 }
722 if ($command === 'RECURRING') {
723 $expense = expense_ref($state, $args[0]);
724 $interval = (int) $args[1];
725 $expense['interval'] = $interval;
726 $expense['next'] = $interval === 0 ? 0 : $state['day'] + $interval;
727 $state['expenses'][$expense['id']] = $expense;
728 return [$interval === 0 ? "RECURRING $expense[id] stopped" : "RECURRING $expense[id] interval $interval next $expense[next]"];
729 }
730 if ($command === 'SETTLE') {
731 [$sender, $receiver, $spec] = $args;
732 check_member($state, $sender);
733 check_member($state, $receiver);
734 if ($sender === $receiver) {
735 fail("same member $sender");
736 }
737 [$amount, $currency] = amount_spec($spec);
738 $total = converted($state, $amount, $currency);
739 if ($amount === 0 || $total === 0) {
740 fail('invalid amount');
741 }
742 $settlement = ['day' => $state['day'], 'record' => ++$state['serial'], 'sender' => $sender,
743 'receiver' => $receiver, 'amount' => $amount, 'currency' => $currency, 'total' => $total];
744 $state['settlements'][] = $settlement;
745 return ["SETTLE $sender -> $receiver " . amount_text($total) . currency_note($settlement)];
746 }
747 if ($command === 'BUDGET') {
748 $category = $args[0];
749 $amount = parse_amount($args[1]);
750 if ($amount === 0) {
751 unset($state['budgets'][$category]);
752 return ["BUDGET $category removed"];
753 }
754 $state['budgets'][$category] = $amount;
755 return ["BUDGET $category " . amount_text($amount)];
756 }
757 return ledger_summary($command, $args, $line, $state);
758}
759
760$state = ['day' => 1, 'serial' => 0, 'members' => [], 'currencies' => [], 'expenses' => [],
761 'used' => [], 'settlements' => [], 'budgets' => []];
762$count = (int) fgets(STDIN);
763for ($i = 0; $i < $count; ++$i) {
764 $line = rtrim(fgets(STDIN), "\r\n");
765 try {
766 foreach (execute($line, $state) as $output) {
767 echo $output, "\n";
768 }
769 } catch (InvalidArgumentException $exception) {
770 echo 'ERROR ', $exception->getMessage(), "\n";
771 }
772}
