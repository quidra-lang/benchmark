35def equal_distribution(total, count):
36 quotient, remainder = divmod(total, count)
37 return [quotient + (i < remainder) for i in range(count)]

40def distribute(total, weights):
41 whole = sum(weights)
42 portions = [total * weight // whole for weight in weights]
43 order = sorted(
44 range(len(weights)),
45 key=lambda i: (-(total * weights[i] % whole), i),
46 )
47 remainder = total - sum(portions)
48 for i in order[:remainder]:
49 portions[i] += 1
50 return portions

86def greedy(balances):
87 remaining = balances.copy()
88 transfers = []
89 while True:
90 debtors = [name for name in remaining if remaining[name] < 0]
91 if not debtors:
92 return transfers
93 creditors = [name for name in remaining if remaining[name] > 0]
94 sender = min(debtors, key=lambda name: (remaining[name], name))
95 receiver = min(creditors, key=lambda name: (-remaining[name], name))
96 value = min(-remaining[sender], remaining[receiver])
97 transfers.append((sender, receiver, value))
98 remaining[sender] += value
99 remaining[receiver] -= value

144 def convert(self, value, currency):
145 if not currency:
146 return value

151 def specification(self, token):
152 parts = token.split("@")
153 original = amount(parts[0])
154 currency = parts[1] if len(parts) == 2 else ""
155 total = self.convert(original, currency)
156 if original == 0 or total == 0:
157 raise Invalid("invalid amount")
158 return {"amount": original, "currency": currency, "total": total}

160 def portions(self, expense):
161 weights = [value for _, value in expense["participants"]]
162 original = expense["amount"]
163 mode = expense["mode"]
164 if mode == "EQUAL":
165 return equal_distribution(original, len(weights))
166 if mode == "EXACT":
167 return weights
168 if mode == "ADJUST":
169 base = equal_distribution(original - sum(weights), len(weights))
170 return [value + extra for value, extra in zip(base, weights)]
171 return distribute(original, weights)

173 def prepare(self, args, previous=None):
174 reference, payer, specification, category, mode = args[:5]
175 self.member(payer)
176 converted = self.specification(specification)
177 participants = []
178 if mode == "EQUAL" and args[5] == "ALL":
179 participants = [(name, 0) for name in self.names(active=True)]
180 else:
181 seen = set()
182 for part in args[5:]:
183 pieces = part.split(":")
184 name = pieces[0]
185 self.member(name)
186 if name in seen:
187 raise Invalid("duplicate member " + name)
188 seen.add(name)
189 value = 0
190 if len(pieces) == 2:
191 if mode in ("EXACT", "ADJUST"):
192 value = amount(pieces[1])
193 else:
194 value = int(pieces[1])
195 participants.append((name, value))
196 total_values = sum(value for _, value in participants)
197 original = converted["amount"]
198 invalid = (
199 mode == "EXACT" and total_values != original
200 or mode == "PERCENT" and total_values != 10000
201 or mode == "SHARES" and total_values == 0
202 or mode == "ADJUST" and total_values > original
203 )
204 if invalid:
205 raise Invalid("splits do not add up")
206 expense = {
207 "id": reference,
208 "day": previous["day"] if previous else self.day,
209 "record": previous["record"] if previous else 0,
210 **converted,
211 "category": category,
212 "mode": mode,
213 "payers": [(payer, original)],
214 "participants": participants,
215 "contributions": [converted["total"]],
216 "interval": previous["interval"] if previous else 0,
217 "next": previous["next"] if previous else 0,
218 "copies": previous["copies"] if previous else 0,
219 }
220 expense["shares"] = distribute(expense["total"], self.portions(expense))
221 return expense

223 def balances(self):
224 values = dict.fromkeys(self.members, 0)
225 for expense in self.expenses.values():
226 for (name, _), value in zip(

231 expense["participants"], expense["shares"]
232 ):
233 values[name] -= value
234 for payment in self.settlements:
235 values[payment["from"]] += payment["total"]
236 values[payment["to"]] -= payment["total"]
237 return values

253 def debts(self):
254 totals = defaultdict(int)
255 for expense in self.expenses.values():
256 people = [
257 (name, net(expense, name))
258 for name in self.names() if involved(expense, name)
259 ]
260 borrowers = [[name, -value] for name, value in people if value < 0]
261 lenders = [[name, value] for name, value in people if value > 0]
262 i = 0
263 j = 0
264 while i < len(borrowers) and j < len(lenders):
265 value = min(borrowers[i][1], lenders[j][1])
266 totals[borrowers[i][0], lenders[j][0]] += value
267 borrowers[i][1] -= value
268 lenders[j][1] -= value
269 if borrowers[i][1] == 0:
270 i += 1
271 if lenders[j][1] == 0:
272 j += 1
273 for payment in self.settlements:
274 totals[payment["from"], payment["to"]] -= payment["total"]
275 return totals

277 def advance_day(self, token):
278 day = int(token)
279 if day <= self.day:
280 raise Invalid("invalid day " + token)
281 output = [f"DAY {day}"]
282 while True:
283 pending = [
284 expense for expense in self.expenses.values()
285 if expense["interval"] and expense["next"] <= day
286 ]
287 if not pending:
288 break
289 template = min(pending, key=lambda item: (item["next"], item["record"]))
290 active = all(
291 self.members[name]["active"]
292 for name, _ in template["payers"] + template["participants"]
293 )
294 total = self.convert(template["amount"], template["currency"])
295 if not active or total == 0:
296 template["interval"] = 0
297 output.append(" CANCEL " + template["id"])
298 continue
299 template["copies"] += 1
300 reference = f"{template['id']}#{template['copies'] + 1}"
301 copy = {
302 **template,
303 "id": reference,
304 "day": template["next"],
305 "record": self.next_record(),
306 "payers": template["payers"].copy(),
307 "participants": template["participants"].copy(),
308 "total": total,
309 "shares": distribute(total, self.portions(template)),
310 "contributions": distribute(
311 total, [value for _, value in template["payers"]]
312 ),
313 "interval": 0,
314 "next": 0,
315 "copies": 0,
316 }
317 self.expenses[reference] = copy
318 self.used.add(reference)
319 template["next"] += template["interval"]
320 output.append(f" RECUR {reference} {money(total)}")
321 output.extend(self.warning(copy))
322 self.day = day
323 return output

404 def matrix(self):
405 names = self.names(active=True)
406 if not names:
407 return ["MATRIX", " (none)"]
408 debts = self.debts()
409 cells = [
410 [
411 "-" if row == column else money(max(
412 0, debts[row, column] - debts[column, row]
413 ))
414 for column in names
415 ]
416 for row in names
417 ]
418 name_width = max(map(len, names))
419 width = max(map(len, names + [cell for row in cells for cell in row]))
420 output = [
421 "MATRIX",
422 " " * name_width + "".join(" " + name.rjust(width) for name in names),
423 ]
424 for name, row in zip(names, cells):
425 output.append(
426 name.ljust(name_width)
427 + "".join(" " + cell.rjust(width) for cell in row)
428 )
429 return output

431 def categories(self):
432 output = ["CATEGORY"]
433 groups = defaultdict(list)
434 for expense in self.expenses.values():
435 groups[expense["category"]].append(expense)

437 total = sum(totals.values())

439 spent = totals[category]

444 shares[name] += value
445 ordered = sorted(shares, key=lambda name: (-shares[name], name))
446 for name in ordered:
447 value = shares[name]
448 if value:
449 output.append(
450 f" {name} {money(value)} {percentage(value, spent)}"
451 )

562 def optimal(self):
563 balances = self.balances()
564 names = [name for name in self.names() if balances[name]]
565 if len(names) > 12:
566 raise Invalid("too many members")
567 full = (1 << len(names)) - 1
568 sums = [0] * (full + 1)
569 for mask in range(1, full + 1):
570 bit = mask & -mask
571 index = bit.bit_length() - 1
572 sums[mask] = sums[mask ^ bit] + balances[names[index]]
573
574 @cache
575 def best(mask):
576 if not mask:
577 return 0
578 first = mask & -mask
579 maximum = 0
580 sub = mask
581 while sub:
582 if sub & first and sums[sub] == 0:
583 maximum = max(maximum, 1 + best(mask ^ sub))
584 sub = (sub - 1) & mask
585 return maximum
586
587 def group_names(mask):
588 return [name for i, name in enumerate(names) if mask & (1 << i)]
589
590 groups = []
591 mask = full
592 while mask:
593 first = mask & -mask
594 candidates = []
595 sub = mask
596 while sub:
597 if (
598 sub & first and sums[sub] == 0
599 and 1 + best(mask ^ sub) == best(mask)
600 ):
601 candidates.append(sub)
602 sub = (sub - 1) & mask
603 chosen = min(candidates, key=group_names)
604 groups.append(group_names(chosen))
605 mask ^= chosen
606 output = [
607 f"SIMPLIFY OPTIMAL transfers {len(names) - len(groups)}"
608 f" groups {len(groups)}"
609 ]
610 for group in groups:
611 output.append(" group " + " ".join(group))
612 transfers = greedy({name: balances[name] for name in group})
613 output.extend(transfer_line(transfer, " ") for transfer in transfers)
614 return output

616 def command(self, line):
617 name, *args = line.split()
618 output = []
619 if name == "DAY":
620 return self.advance_day(args[0])
621 if name in ("EXPENSE", "EDIT"):
622 return self.record_expense(name, args)
623 if name == "PAYERS":
624 return self.change_payers(args)
625 if name == "MEMBER":
626 old = self.members.get(args[0])
627 if old and old["active"]:
628 raise Invalid("duplicate member " + args[0])
629 self.members[args[0]] = {"active": True, "day": self.day}
630 output.append(f"MEMBER {args[0]} {'rejoined' if old else 'joined'}")
631 elif name == "LEAVE":
632 self.member(args[0])
633 if self.balances()[args[0]] != 0:
634 raise Invalid("member has a balance " + args[0])
635 self.members[args[0]] = {"active": False, "day": self.day}
636 output.append("LEAVE " + args[0])
637 elif name == "CURRENCY":
638 old = self.currencies.get(args[0])
639 value = int(args[1])
640 self.currencies[args[0]] = value
641 suffix = "" if old is None else " was " + rate_text(old)
642 output.append(f"CURRENCY {args[0]} {rate_text(value)}{suffix}")
643 elif name == "DELETE":
644 self.get_expense(args[0], frozen=True)
645 del self.expenses[args[0]]
646 output.append("DELETE " + args[0])
647 elif name == "RECURRING":
648 expense = self.get_expense(args[0])
649 interval = int(args[1])
650 expense["interval"] = interval
651 expense["next"] = self.day + interval
652 suffix = (
653 f" interval {interval} next {expense['next']}"
654 if interval else " stopped"
655 )
656 output.append("RECURRING " + args[0] + suffix)
657 elif name == "SETTLE":
658 self.member(args[0])
659 self.member(args[1])
660 if args[0] == args[1]:
661 raise Invalid("same member " + args[0])
662 converted = self.specification(args[2])
663 payment = {
664 "from": args[0], "to": args[1], **converted,
665 "day": self.day, "record": self.next_record(),
666 }
667 self.settlements.append(payment)
668 output.append(
669 f"SETTLE {args[0]} -> {args[1]} {money(payment['total'])}"
670 + currency_note(payment)
671 )
672 elif name == "BUDGET":
673 value = amount(args[1])
674 if value:
675 self.budgets[args[0]] = value
676 else:
677 self.budgets.pop(args[0], None)
678 output.append(f"BUDGET {args[0]} " + (money(value) if value else "removed"))
679 elif name == "BALANCES":
680 balances = self.balances()
681 output = ["BALANCES"] + [
682 f" {person} {signed(balances[person])}"
683 for person in self.names(active=True)
684 ]
685 elif name == "OWES":
686 self.member(args[0], active=False)
687 debts = self.debts()
688 output = ["OWES " + args[0]]
689 for person in self.names():
690 if person == args[0]:
691 continue
692 value = debts[args[0], person] - debts[person, args[0]]
693 if value > 0:
694 output.append(f" {args[0]} owes {person} {money(value)}")
695 elif value < 0:
696 output.append(f" {person} owes {args[0]} {money(-value)}")
697 elif name == "HISTORY":
698 output = self.history(args[0])
699 elif name == "MATRIX":
700 output = self.matrix()
701 elif name == "CATEGORY":
702 output = self.categories()
703 elif name == "BUDGETS":
704 output = ["BUDGETS"]
705 for category, budget in sorted(self.budgets.items()):
706 spent = self.spent(category)
707 suffix = (
708 " left " + money(budget - spent) if spent <= budget
709 else " over " + money(spent - budget)
710 )
711 output.append(
712 f" {category} spent {money(spent)} budget {money(budget)}"
713 f" used {percentage(spent, budget)}{suffix}"
714 )
715 elif name == "TOTALS":
716 output = self.totals()
717 elif name == "MEMBERS":
718 output = ["MEMBERS"]
719 for person in self.names():
720 member = self.members[person]
721 count = sum(involved(expense, person) for expense in self.expenses.values())
722 status = "active" if member["active"] else "left"
723 output.append(
724 f" {person} {status} day {member['day']} expenses {count}"
725 )
726 elif name == "EXPENSES":
727 output = self.expense_list(line, args)
728 elif name == "PERIODS":
729 width = int(args[0])
730 groups = defaultdict(list)
731 for expense in self.expenses.values():
732 groups[(expense["day"] - 1) // width].append(expense)
733 output = [f"PERIODS {width}"]
734 for index, group in sorted(groups.items()):
735 total = sum(expense["total"] for expense in group)
736 output.append(
737 f" days {index * width + 1}-{index * width + width}"
738 f" expenses {len(group)} total {money(total)}"
739 )
740 elif name == "SHOW":
741 output = self.show(args[0])
