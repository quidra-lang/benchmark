63final class Splitter {
64 var members: [String:Member] = [:]
65 var currencies: [String:Int] = [:]
66 var budgets: [String:Int] = [:]
67 var expenses: [String:Expense] = [:]
68 var used: Set<String> = []
69 var settlements: [Settlement] = []
70 var day = 1
71 var serial = 0
72 func fail(_ message: String) -> Problem { Problem(message: message) }
73 func money(_ s: String) -> Int { Int(s.replacingOccurrences(of: ".",with: ""))! }
74 func amount(_ s: String) -> Amount {
75 let p = s.split(separator: "@").map(String.init)
76 return Amount(value: money(p[0]),currency: p.count == 2 ? p[1] : "")
77 }
78 func fmt(_ n: Int) -> String {
79 String(format: "%ld.%02ld",locale: Locale(identifier: "en_US_POSIX"),n/100,n%100)
80 }
81 func signed(_ n: Int) -> String { n > 0 ? "+"+fmt(n) : (n < 0 ? "-"+fmt(-n) : "0.00") }
82 func rate(_ n: Int) -> String {
83 String(format: "%ld.%04ld",locale: Locale(identifier: "en_US_POSIX"),n/10000,n%10000)
84 }
85 func percent(_ p: Int, _ w: Int) -> String { fmt(p*10000/w)+"%" }
86 func note(_ a: Amount) -> String { a.currency.isEmpty ? "" : " [\(fmt(a.value)) \(a.currency)]" }
87 func registered(_ name: String) throws {
88 guard members[name] != nil else { throw fail("unknown member "+name) }
89 }
90 func active(_ name: String) throws {
91 try registered(name)
92 guard members[name]!.active else { throw fail("inactive member "+name) }
93 }
94 func convert(_ a: Amount) throws -> Int {
95 if a.currency.isEmpty { return a.value }
96 guard let r = currencies[a.currency] else { throw fail("unknown currency "+a.currency) }
97 return (a.value*r+5000)/10000
98 }
99 func expense(_ id: String) throws -> Expense {
100 guard let e = expenses[id] else { throw fail("unknown expense "+id) }
101 return e
102 }
103 func thawed(_ e: Expense) throws {
104 for p in e.payers+e.participants {
105 guard members[p.name]!.active else { throw fail("inactive member "+p.name) }
106 }
107 }
108 func distribute(_ total: Int, _ weights: [Int]) -> [Int] {
109 let sum = weights.reduce(0,+)
110 var result = weights.map { total*$0/sum }
111 let remainders = weights.map { total*$0%sum }
112 let order = weights.indices.sorted {
113 remainders[$0] == remainders[$1] ? $0 < $1 : remainders[$0] > remainders[$1]
114 }
115 let left = total-result.reduce(0,+)
116 for i in 0..<left { result[order[i]] += 1 }
117 return result
118 }
119 func compute(_ e: Expense) throws {
120 let a = e.amount.value
121 e.total = try convert(e.amount)
122 let sum = e.participants.reduce(0) { $0+$1.value }
123 let portions: [Int]
124 switch e.mode {
125 case "EXACT":
126 guard sum == a else { throw fail("splits do not add up") }
127 portions = e.participants.map { $0.value }
128 case "PERCENT", "SHARES":
129 if (e.mode == "PERCENT" && sum != 10000) || (e.mode == "SHARES" && sum == 0) {
130 throw fail("splits do not add up")
131 }
132 portions = distribute(a,e.participants.map { $0.value })
133 default:
134 let extras = e.mode == "ADJUST" ? sum : 0
135 guard extras <= a else { throw fail("splits do not add up") }
136 let k = e.participants.count
137 portions = e.participants.enumerated().map { i,p in
138 (a-extras)/k+(i < (a-extras)%k ? 1 : 0)+(e.mode == "ADJUST" ? p.value : 0)
139 }
140 }
141 e.shares = distribute(e.total,portions)
142 e.contributions = distribute(e.total,e.payers.map { $0.value })
143 }
144 func build(_ t: [String]) throws -> Expense {
145 try active(t[2])
146 let e = Expense()
147 e.amount = amount(t[3])
148 let total = try convert(e.amount)
149 guard e.amount.value > 0 && total > 0 else { throw fail("invalid amount") }
150 e.category = t[4]
151 e.mode = t[5]
152 e.payers = [Part(name: t[2],value: e.amount.value)]
153 if e.mode == "EQUAL" && t[6] == "ALL" {
154 e.participants = activeNames().map { Part(name: $0,value: 0) }
155 } else {
156 var seen: Set<String> = []
157 for token in t.dropFirst(6) {
158 let p = token.split(separator: ":").map(String.init)
159 try active(p[0])
160 guard seen.insert(p[0]).inserted else { throw fail("duplicate member "+p[0]) }
161 let value = p.count == 1 ? 0 : ((e.mode == "EXACT" || e.mode == "ADJUST") ? money(p[1]) : Int(p[1])!)
162 e.participants.append(Part(name: p[0],value: value))
163 }
164 }
165 try compute(e)
166 return e
167 }
168 func chronological() -> [Expense] {
169 expenses.values.sorted { $0.day == $1.day ? $0.record < $1.record : $0.day < $1.day }
170 }
171 func includes(_ e: Expense, _ name: String) -> Bool { (e.payers+e.participants).contains { $0.name == name } }
172 func net(_ e: Expense) -> [String:Int] {
173 var result: [String:Int] = [:]
174 for (i,p) in e.payers.enumerated() { result[p.name,default: 0] += e.contributions[i] }
175 for (i,p) in e.participants.enumerated() { result[p.name,default: 0] -= e.shares[i] }
176 return result
177 }
178 func balances() -> [String:Int] {
179 var result = Dictionary(uniqueKeysWithValues: members.keys.map { ($0,0) })
180 for e in expenses.values {
181 for (name,value) in net(e) { result[name,default: 0] += value }
182 }
183 for s in settlements {
184 result[s.from,default: 0] += s.total
185 result[s.to,default: 0] -= s.total
186 }
187 return result
188 }
189 func activeNames() -> [String] { members.keys.filter { members[$0]!.active }.sorted() }
190 func categoryTotals() -> [String:Int] {
191 var result: [String:Int] = [:]
192 for e in expenses.values { result[e.category,default: 0] += e.total }
193 return result
194 }
195 func warning(_ e: Expense, _ output: inout [String]) {
196 if let budget = budgets[e.category] {
197 let spent = categoryTotals()[e.category]!
198 if spent > budget { output.append(" WARNING \(e.category) over budget by \(fmt(spent-budget))") }
199 }
200 }
201 func pairs(_ people: [Part], _ values: [Int]) -> String {
202 people.indices.map { people[$0].name+" "+fmt(values[$0]) }.joined(separator: " + ")
203 }
204 func greedy(_ balances: [String:Int]) -> [Transfer] {
205 var work = balances
206 let names = work.keys.sorted()
207 var result: [Transfer] = []
208 while !work.isEmpty {
209 let from = names.min { work[$0]! < work[$1]! }!
210 let to = names.max { work[$0]! < work[$1]! }!
211 let minimum = work[from]!
212 let maximum = work[to]!
213 if minimum >= 0 { break }
214 let value = min(-minimum,maximum)
215 result.append(Transfer(from: from,to: to,amount: value))
216 work[from] = minimum+value
217 work[to] = maximum-value
218 }
219 return result
220 }
221 func debts(_ names: [String]) -> [[Int]] {
222 let index = Dictionary(uniqueKeysWithValues: names.enumerated().map { ($0.element,$0.offset) })
223 var result = Array(repeating: Array(repeating: 0,count: names.count),count: names.count)
224 for e in expenses.values {
225 var amounts = net(e)
226 let borrowers = amounts.keys.filter { amounts[$0]! < 0 }.sorted()
227 let lenders = amounts.keys.filter { amounts[$0]! > 0 }.sorted()
228 var a = 0
229 var b = 0
230 while a < borrowers.count && b < lenders.count {
231 let x = borrowers[a]
232 let y = lenders[b]
233 let value = min(-amounts[x]!,amounts[y]!)
234 let i = index[x]!
235 let j = index[y]!
236 result[i][j] += value
237 result[j][i] -= value
238 amounts[x]! += value
239 amounts[y]! -= value
240 if amounts[x] == 0 { a += 1 }
241 if amounts[y] == 0 { b += 1 }
242 }
243 }
244 for s in settlements {
245 let i = index[s.from]!
246 let j = index[s.to]!
247 result[i][j] -= s.total
248 result[j][i] += s.total
249 }
250 return result
251 }
252 func indexes(_ mask: Int, _ n: Int) -> [Int] { (0..<n).filter { mask & (1 << $0) != 0 } }
253 func execute(_ line: String) throws -> [String] {
254 let t = line.split(separator: " ").map(String.init)
255 let command = t[0]
256 var output: [String] = []
257 switch command {
258 case "DAY":
259 let d = Int(t[1])!
260 guard d > day else { throw fail("invalid day "+t[1]) }
261 output.append("DAY \(d)")
262 while let template = expenses.values.filter({ $0.interval > 0 && $0.next <= d }).min(by: {
263 $0.next == $1.next ? $0.record < $1.record : $0.next < $1.next
264 }) {
265 let allActive = (template.payers+template.participants).allSatisfy { members[$0.name]!.active }
266 let total = try convert(template.amount)
267 if !allActive || total == 0 {
268 template.interval = 0
269 output.append(" CANCEL "+template.id)
270 } else {
271 let copy = template.copyContents()
272 copy.id = template.id+"#\(template.copies+2)"
273 copy.day = template.next
274 serial += 1
275 copy.record = serial
276 try compute(copy)
277 expenses[copy.id] = copy
278 used.insert(copy.id)
279 template.copies += 1
280 template.next += template.interval
281 output.append(" RECUR \(copy.id) \(fmt(copy.total))")
282 warning(copy,&output)
283 }
284 }
285 day = d
286 case "MEMBER":
287 let old = members[t[1]]
288 guard old?.active != true else { throw fail("duplicate member "+t[1]) }
289 members[t[1]] = Member(active: true,day: day)
290 output.append("MEMBER "+t[1]+(old == nil ? " joined" : " rejoined"))
291 case "LEAVE":
292 try active(t[1])
293 guard balances()[t[1]]! == 0 else { throw fail("member has a balance "+t[1]) }
294 members[t[1]] = Member(active: false,day: day)
295 output.append("LEAVE "+t[1])
296 case "CURRENCY":
297 let value = Int(t[2])!
298 let old = currencies.updateValue(value,forKey: t[1])
299 output.append("CURRENCY \(t[1]) \(rate(value))"+(old.map { " was "+rate($0) } ?? ""))
300 case "EXPENSE", "EDIT":
301 let old: Expense?
302 if command == "EDIT" {
303 old = try expense(t[1])
304 try thawed(old!)
305 } else {
306 guard !used.contains(t[1]) else { throw fail("duplicate id "+t[1]) }
307 old = nil
308 }
309 let e = try build(t)
310 e.id = t[1]
311 e.day = old?.day ?? day
312 if let old {
313 e.record = old.record
314 e.interval = old.interval
315 e.next = old.next
316 e.copies = old.copies
317 } else {
318 serial += 1
319 e.record = serial
320 }
321 expenses[e.id] = e
322 used.insert(e.id)
323 output.append("\(command) \(e.id) \(fmt(e.total))\(note(e.amount)) = \(pairs(e.participants,e.shares))")
324 warning(e,&output)
325 case "PAYERS":
326 let e = try expense(t[1])
327 try thawed(e)
328 var payers: [Part] = []
329 var seen: Set<String> = []
330 for token in t.dropFirst(2) {
331 let p = token.split(separator: ":").map(String.init)
332 try active(p[0])
333 guard seen.insert(p[0]).inserted else { throw fail("duplicate member "+p[0]) }
334 payers.append(Part(name: p[0],value: money(p[1])))
335 }
336 guard payers.reduce(0,{ $0+$1.value }) == e.amount.value else { throw fail("payers do not add up") }
337 let contributions = distribute(e.total,payers.map { $0.value })
338 e.payers = payers
339 e.contributions = contributions
340 output.append("PAYERS \(e.id) \(fmt(e.total))\(note(e.amount)) = \(pairs(payers,contributions))")
341 case "DELETE":
342 let e = try expense(t[1])
343 try thawed(e)
344 expenses.removeValue(forKey: e.id)
345 output.append("DELETE "+e.id)
346 case "RECURRING":
347 let e = try expense(t[1])
348 e.interval = Int(t[2])!
349 e.next = day+e.interval
350 output.append("RECURRING "+e.id+(e.interval == 0 ? " stopped" : " interval \(e.interval) next \(e.next)"))
351 case "SETTLE":
352 try active(t[1])
353 try active(t[2])
354 guard t[1] != t[2] else { throw fail("same member "+t[1]) }
355 let a = amount(t[3])
356 let total = try convert(a)
357 guard a.value > 0 && total > 0 else { throw fail("invalid amount") }
358 serial += 1
359 settlements.append(Settlement(day: day,record: serial,from: t[1],to: t[2],amount: a,total: total))
360 output.append("SETTLE \(t[1]) -> \(t[2]) \(fmt(total))\(note(a))")
361 case "BUDGET":
362 let value = money(t[2])
363 if value == 0 { budgets.removeValue(forKey: t[1]) }
364 else { budgets[t[1]] = value }
365 output.append("BUDGET "+t[1]+(value == 0 ? " removed" : " "+fmt(value)))
366 case "BALANCES":
367 output.append("BALANCES")
368 let values = balances()
369 for name in activeNames() { output.append(" "+name+" "+signed(values[name]!)) }
370 case "OWES", "MATRIX":
371 if command == "OWES" { try registered(t[1]) }
372 let names = members.keys.sorted()
373 let index = Dictionary(uniqueKeysWithValues: names.enumerated().map { ($0.element,$0.offset) })
374 let debt = debts(names)
375 output.append(command == "OWES" ? "OWES "+t[1] : "MATRIX")
376 if command == "OWES" {
377 let i = index[t[1]]!
378 for j in names.indices {
379 if debt[i][j] > 0 { output.append(" \(t[1]) owes \(names[j]) \(fmt(debt[i][j]))") }
380 if debt[i][j] < 0 { output.append(" \(names[j]) owes \(t[1]) \(fmt(-debt[i][j]))") }
381 }
382 } else {
383 let active = activeNames()
384 if active.isEmpty { output.append(" (none)") }
385 else {
386 let l = active.map { $0.count }.max()!
387 func cell(_ a: String, _ b: String) -> String { a == b ? "-" : fmt(max(0,debt[index[a]!][index[b]!])) }
388 let width = max(l,active.flatMap { a in active.map { cell(a,$0).count } }.max()!)
389 func right(_ s: String) -> String { String(format: "%*@",width,s) }
390 output.append(String(repeating: " ",count: l)+active.map { " "+right($0) }.joined())
391 for a in active {
392 output.append(a+String(repeating: " ",count: l-a.count)+active.map { " "+right(cell(a,$0)) }.joined())
393 }
394 }
395 }
396 case "HISTORY":
397 try registered(t[1])
398 var entries: [Entry] = []
399 for e in expenses.values {
400 if includes(e,t[1]) { entries.append(Entry(day: e.day,record: e.record,label: e.id,change: net(e)[t[1]]!)) }
401 }
402 for s in settlements {
403 if s.from == t[1] || s.to == t[1] {
404 entries.append(Entry(day: s.day,record: s.record,label: s.from+"->"+s.to,change: s.from == t[1] ? s.total : -s.total))
405 }
406 }
407 entries.sort { $0.day == $1.day ? $0.record < $1.record : $0.day < $1.day }
408 output.append("HISTORY "+t[1])
409 var balance = 0
410 for e in entries {
411 balance += e.change
412 output.append(" day \(e.day) \(e.label) \(signed(e.change)) balance \(signed(balance))")
413 }
414 case "CATEGORY":
415 output.append("CATEGORY")
416 let totals = categoryTotals()
417 let all = totals.values.reduce(0,+)
418 for category in totals.keys.sorted(by: { totals[$0] == totals[$1] ? $0 < $1 : totals[$0]! > totals[$1]! }) {
419 let total = totals[category]!
420 output.append(" \(category) \(fmt(total)) \(percent(total,all))")
421 var shares: [String:Int] = [:]
422 for e in expenses.values where e.category == category {
423 for (i,p) in e.participants.enumerated() { shares[p.name,default: 0] += e.shares[i] }
424 }
425 let names = shares.keys.filter { shares[$0]! > 0 }.sorted { shares[$0] == shares[$1] ? $0 < $1 : shares[$0]! > shares[$1]! }
426 for name in names { output.append(" \(name) \(fmt(shares[name]!)) \(percent(shares[name]!,total))") }
427 }
428 case "BUDGETS":
429 output.append("BUDGETS")
430 let totals = categoryTotals()
431 for category in budgets.keys.sorted() {
432 let b = budgets[category]!
433 let s = totals[category] ?? 0
434 output.append(" \(category) spent \(fmt(s)) budget \(fmt(b)) used \(percent(s,b))"+
435 (s <= b ? " left "+fmt(b-s) : " over "+fmt(s-b)))
436 }
437 case "TOTALS":
438 output.append("TOTALS")
439 for name in members.keys.sorted() {
440 var paid = 0
441 var share = 0
442 for e in expenses.values {
443 for (i,p) in e.payers.enumerated() where p.name == name { paid += e.contributions[i] }
444 for (i,p) in e.participants.enumerated() where p.name == name { share += e.shares[i] }
445 }
446 let sent = settlements.filter { $0.from == name }.reduce(0) { $0+$1.total }
447 let received = settlements.filter { $0.to == name }.reduce(0) { $0+$1.total }
448 output.append(" \(name) paid \(fmt(paid)) share \(fmt(share)) sent \(fmt(sent)) received \(fmt(received))")
449 }
450 for currency in currencies.keys.sorted() {
451 let group = expenses.values.filter { $0.amount.currency == currency }
452 if !group.isEmpty {
453 output.append(" currency \(currency) expenses \(group.count) amount \(fmt(group.reduce(0) { $0+$1.amount.value })) base \(fmt(group.reduce(0) { $0+$1.total }))")
454 }
455 }
456 output.append(" expenses \(expenses.count) total \(fmt(expenses.values.reduce(0) { $0+$1.total })) settlements \(settlements.count) total \(fmt(settlements.reduce(0) { $0+$1.total }))")
457 case "MEMBERS":
458 output.append("MEMBERS")
459 for name in members.keys.sorted() {
460 let m = members[name]!
461 let count = expenses.values.filter { includes($0,name) }.count
462 output.append(" \(name) \(m.active ? "active" : "left") day \(m.day) expenses \(count)")
463 }
464 case "EXPENSES":
465 if t[1] == "MEMBER" { try registered(t[2]) }
466 output.append(line)
467 var count = 0
468 var total = 0
469 for e in chronological() {
470 let matches: Bool
471 switch t[1] {
472 case "CATEGORY": matches = e.category == t[2]
473 case "MEMBER": matches = includes(e,t[2])
474 case "DAYS": matches = e.day >= Int(t[2])! && e.day <= Int(t[3])!
475 default: matches = true
476 }
477 if matches {
478 output.append(" day \(e.day) \(e.id) \(e.category) \(fmt(e.total)) by "+e.payers.map { $0.name }.joined(separator: ","))
479 count += 1
480 total += e.total
481 }
482 }
483 output.append(" count \(count) total \(fmt(total))")
484 case "PERIODS":
485 let k = Int(t[1])!
486 output.append("PERIODS \(k)")
487 let groups = Dictionary(grouping: expenses.values,by: { ($0.day-1)/k })
488 for i in groups.keys.sorted() {
489 let group = groups[i]!
490 output.append(" days \(i*k+1)-\(i*k+k) expenses \(group.count) total \(fmt(group.reduce(0) { $0+$1.total }))")
491 }
492 case "SHOW":
493 let e = try expense(t[1])
494 output.append("SHOW "+e.id)
495 output.append(" day \(e.day) category \(e.category) mode \(e.mode) total \(fmt(e.total))\(note(e.amount))")
496 for (i,p) in e.payers.enumerated() { output.append(" paid \(p.name) \(fmt(e.contributions[i]))") }
497 let sum = e.participants.reduce(0) { $0+$1.value }
498 for (i,p) in e.participants.enumerated() {
499 let suffix: String
500 switch e.mode {
501 case "EXACT": suffix = " "+fmt(p.value)
502 case "PERCENT": suffix = " "+fmt(p.value)+"%"
503 case "SHARES": suffix = " \(p.value)/\(sum)"
504 case "ADJUST": suffix = " +"+fmt(p.value)
505 default: suffix = ""
506 }
507 output.append(" share \(p.name) \(fmt(e.shares[i]))"+suffix)
508 }
509 if e.interval > 0 { output.append(" recurring interval \(e.interval) next \(e.next)") }
510 case "SIMPLIFY":
511 let all = balances().filter { $0.value != 0 }
512 if t[1] == "GREEDY" {
513 let transfers = greedy(all)
514 output.append("SIMPLIFY GREEDY transfers \(transfers.count)")
515 for x in transfers { output.append(" \(x.from) -> \(x.to) \(fmt(x.amount))") }
516 } else {
517 let names = all.keys.sorted()
518 let n = names.count
519 guard n <= 12 else { throw fail("too many members") }
520 let size = 1 << n
521 var sums = Array(repeating: 0,count: size)
522 var dp = Array(repeating: -1000,count: size)
523 dp[0] = 0
524 if size > 1 {
525 for mask in 1..<size {
526 let bit = mask.trailingZeroBitCount
527 sums[mask] = sums[mask ^ (1 << bit)]+all[names[bit]]!
528 let first = mask & -mask
529 var sub = mask
530 while sub > 0 {
531 if sub & first != 0 && sums[sub] == 0 { dp[mask] = max(dp[mask],1+dp[mask ^ sub]) }
532 sub = (sub-1) & mask
533 }
534 }
535 }
536 var remaining = size-1
537 var groups: [Int] = []
538 while remaining != 0 {
539 let first = remaining & -remaining
540 var best = 0
541 var sub = remaining
542 while sub > 0 {
543 if sub & first != 0 && sums[sub] == 0 && 1+dp[remaining ^ sub] == dp[remaining] &&
544 (best == 0 || indexes(sub,n).lexicographicallyPrecedes(indexes(best,n))) { best = sub }
545 sub = (sub-1) & remaining
546 }
547 groups.append(best)
548 remaining ^= best
549 }
550 let transfers = groups.map { mask in
551 greedy(Dictionary(uniqueKeysWithValues: indexes(mask,n).map { (names[$0],all[names[$0]]!) }))
552 }
553 output.append("SIMPLIFY OPTIMAL transfers \(transfers.reduce(0) { $0+$1.count }) groups \(groups.count)")
554 for i in groups.indices {
555 output.append(" group "+indexes(groups[i],n).map { names[$0] }.joined(separator: " "))
556 for x in transfers[i] { output.append(" \(x.from) -> \(x.to) \(fmt(x.amount))") }
557 }
558 }
559 default:
560 preconditionFailure("Unreachable command")
561 }
562 if ["BALANCES","OWES","HISTORY","CATEGORY","BUDGETS","MEMBERS","PERIODS"].contains(command) && output.count == 1 {
