1local RULES = {STUDENT={3,14,25,1000,2,50},STAFF={10,28,10,2000,5,100}}
2local books,members,closed = {},{},{}
3local book_order,member_order = {},{}
4local day = 1
5local counts = {borrowed=0,returned=0,renewed=0,lost=0,FINE=0,FEE=0,PRICE=0,paid=0,waived=0}
6local function fail(message) error(message,0) end
7local function size(map)
8 local total = 0
9 for _ in pairs(map) do total = total+1 end
10 return total
11end
12local function keys(map)
13 local values = {}
14 for key in pairs(map) do values[#values+1] = key end
15 table.sort(values)
16 return values
17end
18local function append(target,items)
19 table.move(items,1,#items,#target+1,target)
20end
21local function find(values,needle)
22 for index,value in ipairs(values) do
23 if value == needle then return index end
24 end
25 return nil
26end
27local function cents(text)
28 local whole,fraction = text:match('^(%d+)%.(%d%d)$')
29 return tonumber(whole)*100+tonumber(fraction)
30end
31local function money(value) return string.format('%d.%02d',value//100,value%100) end
32local function isbn(text)
33 local s = text:gsub('-','')
34 if #s == 10 and s:match('^[0-9]+[0-9X]$') and s:sub(1,9):match('^[0-9]+$') then
35 local total = 0
36 for index = 1,10 do
37 local char = s:sub(index,index)
38 total = total+(11-index)*(char == 'X' and 10 or tonumber(char))
39 end
40 if total%11 ~= 0 then fail('invalid isbn') end
41 s = '978' .. s:sub(1,9)
42 total = 0
43 for index = 1,12 do total = total+tonumber(s:sub(index,index))*(index%2 == 1 and 1 or 3) end
44 return s .. ((10-total%10)%10)
45 end
46 if #s == 13 and s:match('^[0-9]+$') and (s:sub(1,3) == '978' or s:sub(1,3) == '979') then
47 local total = 0
48 for index = 1,13 do total = total+tonumber(s:sub(index,index))*(index%2 == 1 and 1 or 3) end
49 if total%10 == 0 then return s end
50 end
51 fail('invalid isbn')
52end
53local function get_member(id)
54 if not members[id] then fail('unknown member') end
55 return members[id]
56end
57local function get_book(text)
58 local id = isbn(text)
59 if not books[id] then fail('unknown book') end
60 return books[id]
61end
62local function balance(m)
63 local total = 0
64 for _,item in ipairs(m.items) do total = total+item.unpaid end
65 return total
66end
67local function holdings(b)
68 local total = 0
69 for _,c in ipairs(b.copies) do
70 if c.state == 'AVAILABLE' or c.state == 'LOANED' or c.state == 'HOLD' then total = total+1 end
71 end
72 return total
73end
74local function reservations(m)
75 local total = size(m.holds)
76 for _,id in ipairs(book_order) do
77 if find(books[id].queue,m.id) then total = total+1 end
78 end
79 return total
80end
81local function open_day(date)
82 while closed[date] do date = date+1 end
83 return date
84end
85local function late_fine(c)
86 local m = members[c.member]
87 local late = math.max(0,day-c.due)
88 return late,math.min(late*RULES[m.type][3],RULES[m.type][4])
89end
90local function charge(m,kind,id,amount)
91 if amount == 0 then return end
92 m.items[#m.items+1] = {day=day,kind=kind,isbn=id,amount=amount,unpaid=amount}
93 counts[kind] = counts[kind]+amount
94end
95local function release_copy(b,c,out)
96 if #b.queue == 0 then
97 c.state = 'AVAILABLE'
98 return
99 end
100 local id = table.remove(b.queue,1)
101 c.state,c.member,c.expiry = 'HOLD',id,open_day(day+3)
102 members[id].holds[b.id] = c
103 out[#out+1] = 'HOLD ' .. b.id .. ' copy ' .. c.number .. ' ' .. id .. ' until ' .. c.expiry
104end
105local function end_loan(b,c,result)
106 local m = members[c.member]
107 local late,fine = late_fine(c)
108 m.loans[b.id] = nil
109 charge(m,'FINE',b.id,fine)
110 m.history[#m.history+1] = result .. ' ' .. b.id .. ' copy ' .. c.number .. ' from ' .. c.start .. ' to ' .. day .. ' late ' .. late .. ' fine ' .. money(fine)
111 return m,late,fine
112end
113local function lose_copy(b,c,heading,out)
114 local m,late,fine = end_loan(b,c,'LOST')
115 m.history[#m.history] = m.history[#m.history] .. ' price ' .. money(b.price)
116 charge(m,'PRICE',b.id,b.price)
117 c.state = 'LOST'
118 counts.lost = counts.lost+1
119 out[#out+1] = heading .. ' ' .. m.id .. ' ' .. b.id .. ' copy ' .. c.number .. ' late ' .. late .. ' fine ' .. money(fine) .. ' price ' .. money(b.price) .. ' balance ' .. money(balance(m))
120 if holdings(b) == 0 then
121 for _,id in ipairs(b.queue) do out[#out+1] = 'DROPPED ' .. b.id .. ' ' .. id end
122 b.queue = {}
123 end
124end
125local function renewal_reason(b,c)
126 if c.due < day then return 'loan overdue' end
127 if c.renewals == 2 then return 'renewal limit' end
128 return #b.queue > 0 and 'title reserved' or ''
129end
130local function renew_loan(m,b,c)
131 c.due = open_day(day+RULES[m.type][2])
132 c.renewals = c.renewals+1
133 counts.renewed = counts.renewed+1
134 return 'RENEW ' .. m.id .. ' ' .. b.id .. ' copy ' .. c.number .. ' due ' .. c.due .. ' renewals ' .. c.renewals
135end
136local function sorted_copies(map,date)
137 local copies = {}
138 for _,c in pairs(map) do copies[#copies+1] = c end
139 table.sort(copies,function(a,b)
140 if a[date] ~= b[date] then return a[date] < b[date] end
141 return a.isbn < b.isbn
142 end)
143 return copies
144end
145local function execute(line)
146 local t,out = {},{}
147 for token in line:gmatch('%S+') do t[#t+1] = token end
148 local cmd = t[1]
149 local function emit(text) out[#out+1] = text end
150 if cmd == 'DAY' then
151 local target = tonumber(t[2])
152 if target <= day then fail('invalid day') end
153 emit('DAY ' .. target)
154 while day < target do
155 local next_day = target+1
156 for _,m in pairs(members) do
157 for _,c in pairs(m.loans) do next_day = math.min(next_day,c.due+365) end
158 for _,c in pairs(m.holds) do next_day = math.min(next_day,c.expiry+1) end
159 end
160 if next_day > target then
161 day = target
162 break
163 end
164 day = next_day
165 local losses = {}
166 for _,id in ipairs(member_order) do
167 local m = members[id]
168 for _,key in ipairs(keys(m.loans)) do
169 local c = m.loans[key]
170 if c.due+365 <= day then losses[#losses+1] = c end
171 end
172 end
173 for _,c in ipairs(losses) do lose_copy(books[c.isbn],c,'AUTOLOST',out) end
174 local expired = {}
175 for _,id in ipairs(book_order) do
176 for _,c in ipairs(books[id].copies) do
177 if c.state == 'HOLD' and c.expiry < day then expired[#expired+1] = c end
178 end
179 end
180 for _,c in ipairs(expired) do
181 local b,m = books[c.isbn],members[c.member]
182 local fee = RULES[m.type][6]
183 charge(m,'FEE',b.id,fee)
184 m.holds[b.id] = nil
185 emit('EXPIRED ' .. b.id .. ' copy ' .. c.number .. ' ' .. m.id .. ' fee ' .. money(fee))
186 release_copy(b,c,out)
187 end
188 end
189 elseif cmd == 'CLOSED' then
190 local date = tonumber(t[2])
191 if date <= day then fail('invalid day') end
192 if closed[date] then fail('already closed') end
193 closed[date] = true
194 emit('CLOSED ' .. date)
195 for _,id in ipairs(book_order) do
196 for _,c in ipairs(books[id].copies) do
197 if c.state == 'LOANED' and c.due == date then
198 c.due = open_day(date)
199 emit('MOVED ' .. id .. ' copy ' .. c.number .. ' ' .. c.member .. ' due ' .. c.due)
200 elseif c.state == 'HOLD' and c.expiry == date then
201 c.expiry = open_day(date)
202 emit('MOVED ' .. id .. ' copy ' .. c.number .. ' ' .. c.member .. ' until ' .. c.expiry)
203 end
204 end
205 end
206 elseif cmd == 'ADDBOOK' then
207 local id,name = isbn(t[2]),table.concat(t,' ',5)
208 if books[id] and books[id].name ~= name then fail('title mismatch') end
209 if not books[id] then
210 books[id] = {id=id,name=name,price=0,copies={},queue={},borrowed=0}
211 book_order[#book_order+1] = id
212 table.sort(book_order)
213 end
214 local b,added = books[id],{}
215 b.price = cents(t[4])
216 for _ = 1,tonumber(t[3]) do
217 local c = {isbn=id,number=#b.copies+1,state='AVAILABLE',member='',start=0,due=0,renewals=0,expiry=0}
218 b.copies[#b.copies+1] = c
219 added[#added+1] = c
220 end
221 emit('ADDBOOK ' .. id .. ' added ' .. t[3] .. ' total ' .. holdings(b))
222 for _,c in ipairs(added) do release_copy(b,c,out) end
223 elseif cmd == 'ADDMEMBER' then
224 if members[t[2]] then fail('duplicate member') end
225 if not RULES[t[3]] then fail('unknown type') end
226 local m = {id=t[2],type=t[3],name=table.concat(t,' ',4),loans={},holds={},items={},history={}}
227 members[m.id] = m
228 member_order[#member_order+1] = m.id
229 table.sort(member_order)
230 local rule = RULES[m.type]
231 emit('ADDMEMBER ' .. m.id .. ' ' .. m.type .. ' loans ' .. rule[1] .. ' days ' .. rule[2])
232 elseif cmd == 'BORROW' or cmd == 'RETURN' or cmd == 'RENEW' or cmd == 'RESERVE' or cmd == 'CANCEL' or cmd == 'LOST' then
233 local m = get_member(t[2])
234 local b = get_book(t[3])
235 local id,rule = b.id,RULES[m.type]
236 local loan,hold = m.loans[id],m.holds[id]
237 if cmd == 'BORROW' then
238 if closed[day] then fail('library closed') end
239 if loan then fail('already borrowed') end
240 if balance(m) > 500 then fail('member blocked') end
241 for _,c in pairs(m.loans) do
242 if c.due < day then fail('overdue loans') end
243 end
244 if size(m.loans) >= rule[1] then fail('loan limit') end
245 local chosen = hold
246 if not chosen then
247 for _,c in ipairs(b.copies) do
248 if c.state == 'AVAILABLE' then
249 chosen = c
250 break
251 end
252 end
253 end
254 if not chosen then
255 for _,c in ipairs(b.copies) do
256 if c.state == 'HOLD' then fail('all copies on hold') end
257 end
258 fail('no copy available')
259 end
260 m.holds[id] = nil
261 chosen.state,chosen.member,chosen.start = 'LOANED',m.id,day
262 chosen.due,chosen.renewals = open_day(day+rule[2]),0
263 m.loans[id] = chosen
264 b.borrowed,counts.borrowed = b.borrowed+1,counts.borrowed+1
265 emit('BORROW ' .. m.id .. ' ' .. id .. ' copy ' .. chosen.number .. ' due ' .. chosen.due)
266 elseif cmd == 'RETURN' or cmd == 'RENEW' or cmd == 'LOST' then
267 if not loan then fail('not borrowed') end
268 if cmd == 'RETURN' then
269 local _,late,fine = end_loan(b,loan,'RETURNED')
270 counts.returned = counts.returned+1
271 emit('RETURN ' .. m.id .. ' ' .. id .. ' copy ' .. loan.number .. ' late ' .. late .. ' fine ' .. money(fine) .. ' balance ' .. money(balance(m)))
272 release_copy(b,loan,out)
273 elseif cmd == 'LOST' then
274 lose_copy(b,loan,'LOST',out)
275 else
276 if balance(m) > 500 then fail('member blocked') end
277 local reason = renewal_reason(b,loan)
278 if reason ~= '' then fail(reason) end
279 emit(renew_loan(m,b,loan))
280 end
281 elseif cmd == 'RESERVE' then
282 if loan then fail('already borrowed') end
283 if hold or find(b.queue,m.id) then fail('already reserved') end
284 if balance(m) > 500 then fail('member blocked') end
285 if reservations(m) >= rule[5] then fail('reservation limit') end
286 if holdings(b) == 0 then fail('no copies') end
287 for _,c in ipairs(b.copies) do
288 if c.state == 'AVAILABLE' then fail('copy available') end
289 end
290 b.queue[#b.queue+1] = m.id
291 local position,dues = #b.queue,{}
292 for _,c in ipairs(b.copies) do
293 if c.state == 'LOANED' then dues[#dues+1] = c.due end
294 end
295 table.sort(dues)
296 emit('RESERVE ' .. m.id .. ' ' .. id .. ' position ' .. position .. ' estimate ' .. (dues[position] or 'none'))
297 else
298 local position = find(b.queue,m.id)
299 if not position and not hold then fail('not reserved') end
300 if position then
301 table.remove(b.queue,position)
302 emit('CANCEL ' .. m.id .. ' ' .. id .. ' queue')
303 else
304 m.holds[id] = nil
305 emit('CANCEL ' .. m.id .. ' ' .. id .. ' hold copy ' .. hold.number)
306 release_copy(b,hold,out)
307 end
308 end
309 elseif cmd == 'RENEWALL' or cmd == 'PAYFINE' or cmd == 'WAIVE' or cmd == 'FINES' or cmd == 'MEMBER' or cmd == 'HISTORY' then
310 local m = get_member(t[2])
311 if cmd == 'RENEWALL' then
312 if size(m.loans) == 0 then fail('no loans') end
313 if balance(m) > 500 then fail('member blocked') end
314 local renewed,skipped = 0,0
315 for _,id in ipairs(keys(m.loans)) do
316 local reason = renewal_reason(books[id],m.loans[id])
317 if reason == '' then
318 emit(renew_loan(m,books[id],m.loans[id]))
319 renewed = renewed+1
320 else
321 emit('SKIP ' .. id .. ' ' .. reason)
322 skipped = skipped+1
323 end
324 end
325 emit('RENEWALL ' .. m.id .. ' renewed ' .. renewed .. ' skipped ' .. skipped)
326 elseif cmd == 'PAYFINE' then
327 if balance(m) == 0 then fail('no fines') end
328 local remaining,paid,lines = cents(t[3]),0,{}
329 for index,item in ipairs(m.items) do
330 local applied = math.min(remaining,item.unpaid)
331 if applied > 0 then
332 item.unpaid,remaining,paid = item.unpaid-applied,remaining-applied,paid+applied
333 lines[#lines+1] = 'ITEM ' .. index .. ' ' .. item.kind .. ' paid ' .. money(applied) .. ' remaining ' .. money(item.unpaid)
334 end
335 end
336 counts.paid = counts.paid+paid
337 emit('PAYFINE ' .. m.id .. ' paid ' .. money(paid) .. ' change ' .. money(remaining) .. ' balance ' .. money(balance(m)))
338 append(out,lines)
339 elseif cmd == 'WAIVE' then
340 local index = tonumber(t[3])
341 local item = m.items[index]
342 if not item then fail('unknown item') end
343 if item.unpaid == 0 then fail('item settled') end
344 local amount = item.unpaid
345 item.unpaid = 0
346 counts.waived = counts.waived+amount
347 emit('WAIVE ' .. m.id .. ' item ' .. index .. ' ' .. item.kind .. ' amount ' .. money(amount) .. ' balance ' .. money(balance(m)))
348 elseif cmd == 'FINES' then
349 local count = 0
350 for _,item in ipairs(m.items) do
351 if item.unpaid > 0 then count = count+1 end
352 end
353 emit('FINES ' .. m.id .. ' items ' .. count .. ' balance ' .. money(balance(m)))
354 for index,item in ipairs(m.items) do
355 if item.unpaid > 0 then emit(index .. ' day ' .. item.day .. ' ' .. item.kind .. ' ' .. item.isbn .. ' amount ' .. money(item.amount) .. ' remaining ' .. money(item.unpaid)) end
356 end
357 elseif cmd == 'HISTORY' then
358 emit('HISTORY ' .. m.id .. ' ' .. #m.history)
359 append(out,m.history)
360 else
361 local rule = RULES[m.type]
362 local state = balance(m) > 500 and 'BLOCKED' or 'OK'
363 emit('MEMBER ' .. m.id .. ' ' .. m.type .. ' ' .. m.name)
364 emit('STATUS loans ' .. size(m.loans) .. '/' .. rule[1] .. ' reservations ' .. reservations(m) .. '/' .. rule[5] .. ' balance ' .. money(balance(m)) .. ' ' .. state)
365 for _,c in ipairs(sorted_copies(m.loans,'due')) do
366 local text = 'LOAN ' .. c.isbn .. ' copy ' .. c.number .. ' due ' .. c.due .. ' renewals ' .. c.renewals
367 if c.due < day then
368 local late,fine = late_fine(c)
369 text = text .. ' late ' .. late .. ' fine ' .. money(fine)
370 end
371 emit(text)
372 end
373 for _,c in ipairs(sorted_copies(m.holds,'expiry')) do emit('HOLD ' .. c.isbn .. ' copy ' .. c.number .. ' until ' .. c.expiry) end
374 for _,id in ipairs(book_order) do
375 local position = find(books[id].queue,m.id)
376 if position then emit('QUEUE ' .. id .. ' position ' .. position) end
377 end
378 end
379 elseif cmd == 'WITHDRAW' or cmd == 'BOOK' then
380 local b = get_book(t[2])
381 if cmd == 'WITHDRAW' then
382 local c = b.copies[tonumber(t[3])]
383 if not c then fail('unknown copy') end
384 if c.state ~= 'AVAILABLE' then fail('copy not available') end
385 c.state = 'WITHDRAWN'
386 emit('WITHDRAW ' .. b.id .. ' copy ' .. c.number .. ' total ' .. holdings(b))
387 else
388 emit('BOOK ' .. b.id .. ' ' .. b.name)
389 emit('PRICE ' .. money(b.price) .. ' copies ' .. holdings(b) .. ' borrowed ' .. b.borrowed)
390 for _,c in ipairs(b.copies) do
391 local text = 'COPY ' .. c.number .. ' ' .. c.state
392 if c.state == 'LOANED' then text = text .. ' ' .. c.member .. ' due ' .. c.due
393 elseif c.state == 'HOLD' then text = text .. ' ' .. c.member .. ' until ' .. c.expiry end
394 emit(text)
395 end
396 local parts = {'QUEUE',#b.queue}
397 append(parts,b.queue)
398 emit(table.concat(parts,' '))
399 end
400 elseif cmd == 'SEARCH' then
401 local keyword = table.concat(t,' ',2):lower()
402 local found = {}
403 for _,b in pairs(books) do
404 if b.name:lower():find(keyword,1,true) then found[#found+1] = b end
405 end
406 table.sort(found,function(a,b)
407 if a.name:lower() ~= b.name:lower() then return a.name:lower() < b.name:lower() end
408 return a.id < b.id
409 end)
410 emit('SEARCH ' .. #found)
411 for _,b in ipairs(found) do
412 local available = 0
413 for _,c in ipairs(b.copies) do
414 if c.state == 'AVAILABLE' then available = available+1 end
415 end
416 emit(b.id .. ' ' .. available .. '/' .. holdings(b) .. ' ' .. b.name)
417 end
418 elseif cmd == 'OVERDUE' or cmd == 'NOTICES' then
419 local lines = {}
420 if cmd == 'OVERDUE' then
421 local loans = {}
422 for _,m in pairs(members) do
423 for _,c in pairs(m.loans) do
424 if c.due < day then loans[#loans+1] = c end
425 end
426 end
427 table.sort(loans,function(a,b)
428 if a.due ~= b.due then return a.due < b.due end
429 if a.member ~= b.member then return a.member < b.member end
430 return a.isbn < b.isbn
431 end)
432 for _,c in ipairs(loans) do
433 local late,fine = late_fine(c)
434 lines[#lines+1] = c.member .. ' ' .. c.isbn .. ' copy ' .. c.number .. ' due ' .. c.due .. ' late ' .. late .. ' fine ' .. money(fine)
435 end
436 else
437 for _,id in ipairs(member_order) do
438 local m = members[id]
439 local loans = sorted_copies(m.loans,'due')
440 for _,c in ipairs(loans) do
441 if c.due < day then lines[#lines+1] = 'OVERDUE ' .. id .. ' ' .. c.isbn .. ' due ' .. c.due end
442 end
443 for _,c in ipairs(loans) do
444 if c.due >= day and c.due <= day+2 then lines[#lines+1] = 'DUESOON ' .. id .. ' ' .. c.isbn .. ' due ' .. c.due end
445 end
446 for _,c in ipairs(sorted_copies(m.holds,'expiry')) do lines[#lines+1] = 'PICKUP ' .. id .. ' ' .. c.isbn .. ' until ' .. c.expiry end
447 end
448 end
449 emit(cmd .. ' ' .. #lines)
450 append(out,lines)
451 elseif cmd == 'MEMBERS' then
452 emit('MEMBERS ' .. #member_order)
453 for _,id in ipairs(member_order) do
454 local m = members[id]
455 emit(id .. ' ' .. m.type .. ' loans ' .. size(m.loans) .. ' reservations ' .. reservations(m) .. ' balance ' .. money(balance(m)) .. ' ' .. (balance(m) > 500 and 'BLOCKED' or 'OK'))
456 end
457 elseif cmd == 'INVENTORY' then
458 local totals = {AVAILABLE=0,LOANED=0,HOLD=0,LOST=0,WITHDRAWN=0}
459 local lines = {}
460 for _,id in ipairs(book_order) do
461 local b = books[id]
462 local n = {AVAILABLE=0,LOANED=0,HOLD=0,LOST=0,WITHDRAWN=0}
463 for _,c in ipairs(b.copies) do n[c.state] = n[c.state]+1 end
464 for state,count in pairs(n) do totals[state] = totals[state]+count end
465 lines[#lines+1] = id .. ' copies ' .. holdings(b) .. ' available ' .. n.AVAILABLE .. ' loaned ' .. n.LOANED .. ' hold ' .. n.HOLD .. ' queue ' .. #b.queue
466 end
467 local holding = totals.AVAILABLE+totals.LOANED+totals.HOLD
468 emit('INVENTORY titles ' .. #book_order .. ' copies ' .. holding .. ' available ' .. totals.AVAILABLE .. ' loaned ' .. totals.LOANED .. ' hold ' .. totals.HOLD .. ' lost ' .. totals.LOST .. ' withdrawn ' .. totals.WITHDRAWN)
469 append(out,lines)
470 elseif cmd == 'STATS' then
471 local active,outstanding = 0,0
472 for _,m in pairs(members) do
473 active,outstanding = active+size(m.loans),outstanding+balance(m)
474 end
475 emit('STATS day ' .. day .. ' titles ' .. #book_order .. ' members ' .. #member_order .. ' closed ' .. size(closed))
476 emit('LOANS borrowed ' .. counts.borrowed .. ' returned ' .. counts.returned .. ' renewed ' .. counts.renewed .. ' lost ' .. counts.lost .. ' active ' .. active)
477 emit('MONEY fines ' .. money(counts.FINE) .. ' fees ' .. money(counts.FEE) .. ' charges ' .. money(counts.PRICE) .. ' paid ' .. money(counts.paid) .. ' waived ' .. money(counts.waived) .. ' outstanding ' .. money(outstanding))
478 local top = {}
479 for _,b in pairs(books) do
480 if b.borrowed > 0 then top[#top+1] = b end
481 end
482 table.sort(top,function(a,b)
483 if a.borrowed ~= b.borrowed then return a.borrowed > b.borrowed end
484 return a.id < b.id
485 end)
486 for rank = 1,math.min(3,#top) do
487 local b = top[rank]
488 emit('TOP ' .. rank .. ' ' .. b.id .. ' ' .. b.borrowed .. ' ' .. b.name)
489 end
490 end
491 return out
492end
493local n = tonumber(io.read('*l'))
494for _ = 1,n do
495 local ok,out = pcall(execute,io.read('*l'))
496 if ok then
497 for _,line in ipairs(out) do io.write(line,'\n') end
498 else
499 io.write('ERROR ',out,'\n')
500 end
501end
502
