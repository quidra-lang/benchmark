6 static class Fault extends RuntimeException {

9 static void check(boolean condition, String reason) { if (condition) throw new Fault(reason); }
10 static class Item {
11 int day, amount, unpaid;
12 String kind, isbn;
13 Item(int day, String kind, String isbn, int amount) {
14 this.day=day;
15 this.kind=kind;
16 this.isbn=isbn;
17 this.amount=amount;
18 unpaid=amount;
19 }

21 static class Member {
22 String id, type, name;
23 int limit, period, rate, cap, reservationLimit, fee, queued;
24 TreeMap<String,Copy> loans=new TreeMap<>(), holds=new TreeMap<>();
25 List<Item> items=new ArrayList<>();
26 List<String> history=new ArrayList<>();
27 Member(String id, String type, String name) {
28 this.id=id;
29 this.type=type;
30 this.name=name;
31 boolean student=type.equals("STUDENT");
32 limit=student ? 3 : 10;
33 period=student ? 14 : 28;
34 rate=student ? 25 : 10;
35 cap=student ? 1000 : 2000;
36 reservationLimit=student ? 2 : 5;
37 fee=student ? 50 : 100;
38 }
39 int balance() { return items.stream().mapToInt(i -> i.unpaid).sum(); }
40 int reservations() { return queued+holds.size(); }
41 boolean blocked() { return balance()>500; }
42 String status() { return blocked() ? "BLOCKED" : "OK"; }
43 }
44 static class Title {
45 String isbn, name;
46 int price, borrowed;
47 List<Copy> copies=new ArrayList<>();
48 List<Member> queue=new ArrayList<>();
49 Title(String isbn, String name, int price) {
50 this.isbn=isbn;
51 this.name=name;
52 this.price=price;
53 }
54 int count(String state) { return (int)copies.stream().filter(c -> c.state.equals(state)).count(); }
55 int holdings() { return count("AVAILABLE")+count("LOANED")+count("HOLD"); }
56 }
57 static class Copy {
58 Title title;
59 int number, loanDay, due, renewals, expiry;
60 String state="AVAILABLE";
61 Member owner;
62 Copy(Title title, int number) {
63 this.title=title;
64 this.number=number;
65 }

67 static class Library {
68 int day=1, borrowed, returned, renewed, lost, fines, fees, charges, paid, waived;
69 TreeMap<String,Title> titles=new TreeMap<>();
70 TreeMap<String,Member> members=new TreeMap<>();
71 Set<Integer> closed=new HashSet<>();
72 List<Copy> copies=new ArrayList<>();
73 List<String> output;
74 Member member(String id) {
75 Member m=members.get(id);
76 check(m==null,"unknown member");
77 return m;
78 }
79 Title book(String raw) {
80 String isbn=normalize(raw);
81 Title b=titles.get(isbn);
82 check(b==null,"unknown book");
83 return b;
84 }
85 Copy loan(Member m, Title b) {
86 Copy c=m.loans.get(b.isbn);
87 check(c==null,"not borrowed");
88 return c;
89 }
90 int open(int d) {
91 while (closed.contains(d)) d++;
92 return d;
93 }
94 int late(Copy c) { return Math.max(0,day-c.due); }
95 int fine(Copy c) { return Math.min(late(c)*c.owner.rate,c.owner.cap); }
96 void charge(Member m, String kind, Title b, int amount) {
97 if (amount==0) return;
98 m.items.add(new Item(day,kind,b.isbn,amount));
99 switch (kind) {
100 case "FINE":
101 fines+=amount;
102 break;
103 case "FEE":
104 fees+=amount;
105 break;
106 default: charges+=amount;
107 }
108 }
109 void release(Copy c) {
110 Title b=c.title;
111 c.owner=null;
112 if (b.queue.isEmpty()) c.state="AVAILABLE";
113 else {
114 Member m=b.queue.remove(0);
115 m.queued--;
116 c.state="HOLD";
117 c.owner=m;
118 c.expiry=open(day+3);
119 m.holds.put(b.isbn,c);
120 output.add("HOLD "+b.isbn+" copy "+c.number+" "+m.id+" until "+c.expiry);
121 }
122 }
123 int[] endLoan(Copy c, boolean loss) {
124 Member m=c.owner;
125 int l=late(c), f=fine(c);
126 m.loans.remove(c.title.isbn);
127 charge(m,"FINE",c.title,f);
128 m.history.add((loss ? "LOST " : "RETURNED ")+c.title.isbn+" copy "+c.number+
129 " from "+c.loanDay+" to "+day+" late "+l+" fine "+money(f)+
130 (loss ? " price "+money(c.title.price) : ""));
131 return new int[]{l,f};
132 }
133 void loss(Copy c, String heading) {
134 Member m=c.owner;
135 Title b=c.title;
136 int[] values=endLoan(c,true);
137 charge(m,"PRICE",b,b.price);
138 c.state="LOST";
139 c.owner=null;
140 lost++;
141 output.add(heading+" "+m.id+" "+b.isbn+" copy "+c.number+" late "+values[0]+" fine "+money(values[1])+
142 " price "+money(b.price)+" balance "+money(m.balance()));
143 if (b.holdings()==0) {
144 for (Member queued:b.queue) {
145 queued.queued--;
146 output.add("DROPPED "+b.isbn+" "+queued.id);
147 }
148 b.queue.clear();
149 }
150 }
151 String renewalReason(Copy c) {
152 if (c.due<day) return "loan overdue";
153 if (c.renewals==2) return "renewal limit";
154 if (!c.title.queue.isEmpty()) return "title reserved";
155 return null;
156 }
157 void renew(Copy c) {
158 c.due=open(day+c.owner.period);
159 c.renewals++;
160 renewed++;
161 output.add("RENEW "+c.owner.id+" "+c.title.isbn+" copy "+c.number+" due "+c.due+" renewals "+c.renewals);
162 }
163 List<Copy> orderedLoans(Member m) {
164 return m.loans.values().stream().sorted(Comparator.comparingInt((Copy c) -> c.due).thenComparing(c -> c.title.isbn)).toList();
165 }
166 List<Copy> orderedHolds(Member m) {
167 return m.holds.values().stream().sorted(Comparator.comparingInt((Copy c) -> c.expiry).thenComparing(c -> c.title.isbn)).toList();
168 }
169 void showMember(Member m) {
170 output.add("MEMBER "+m.id+" "+m.type+" "+m.name);
171 output.add("STATUS loans "+m.loans.size()+"/"+m.limit+" reservations "+m.reservations()+"/"+m.reservationLimit+
172 " balance "+money(m.balance())+" "+m.status());
173 for (Copy c:orderedLoans(m)) output.add("LOAN "+c.title.isbn+" copy "+c.number+" due "+c.due+" renewals "+c.renewals+
174 (c.due<day ? " late "+late(c)+" fine "+money(fine(c)) : ""));
175 for (Copy c:orderedHolds(m)) output.add("HOLD "+c.title.isbn+" copy "+c.number+" until "+c.expiry);
176 for (Title b:titles.values()) {
177 int p=b.queue.indexOf(m);
178 if (p>=0) output.add("QUEUE "+b.isbn+" position "+(p+1));
179 }
180 }
181 void execute(String line) {
182 String[] t=line.split(" ");
183 switch (t[0]) {
184 case "DAY": {
185 int d=Integer.parseInt(t[1]);
186 check(d<=day,"invalid day");
187 output.add("DAY "+d);
188 while (day<d) {
189 day++;
190 List<Copy> losses=copies.stream().filter(c -> c.state.equals("LOANED") && c.due+365<=day)
191 .sorted(Comparator.comparing((Copy c) -> c.owner.id).thenComparing(c -> c.title.isbn)).toList();
192 for (Copy c:losses) loss(c,"AUTOLOST");
193 List<Copy> expired=copies.stream().filter(c -> c.state.equals("HOLD") && c.expiry<day)
194 .sorted(Comparator.comparing((Copy c) -> c.title.isbn).thenComparingInt(c -> c.number)).toList();
195 for (Copy c:expired) {
196 Member m=c.owner;
197 charge(m,"FEE",c.title,m.fee);
198 m.holds.remove(c.title.isbn);
199 output.add("EXPIRED "+c.title.isbn+" copy "+c.number+" "+m.id+" fee "+money(m.fee));
200 release(c);
201 }
202 }
203 break;
204 }
205 case "CLOSED": {
206 int d=Integer.parseInt(t[1]);
207 check(d<=day,"invalid day");
208 check(closed.contains(d),"already closed");
209 closed.add(d);
210 output.add("CLOSED "+d);
211 for (Title b:titles.values()) {
212 for (Copy c:b.copies) {
213 if (c.state.equals("LOANED") && c.due==d) {
214 c.due=open(d);
215 output.add("MOVED "+b.isbn+" copy "+c.number+" "+c.owner.id+" due "+c.due);
216 } else if (c.state.equals("HOLD") && c.expiry==d) {
217 c.expiry=open(d);
218 output.add("MOVED "+b.isbn+" copy "+c.number+" "+c.owner.id+" until "+c.expiry);
219 }
220 }
221 }
222 break;
223 }
224 case "ADDBOOK": {
225 String isbn=normalize(t[1]), name=String.join(" ",Arrays.copyOfRange(t,4,t.length));
226 Title b=titles.get(isbn);
227 check(b!=null && !b.name.equals(name),"title mismatch");
228 int price=parseMoney(t[3]), count=Integer.parseInt(t[2]);
229 if (b==null) {
230 b=new Title(isbn,name,price);
231 titles.put(isbn,b);
232 }
233 b.price=price;
234 List<Copy> added=new ArrayList<>();
235 for (int i=0;i<count;i++) {
236 Copy c=new Copy(b,b.copies.size()+1);
237 b.copies.add(c);
238 copies.add(c);
239 added.add(c);
240 }
241 output.add("ADDBOOK "+isbn+" added "+count+" total "+b.holdings());
242 for (Copy c:added) release(c);
243 break;
244 }
245 case "ADDMEMBER": {
246 check(members.containsKey(t[1]),"duplicate member");
247 check(!t[2].equals("STUDENT") && !t[2].equals("STAFF"),"unknown type");
248 Member m=new Member(t[1],t[2],String.join(" ",Arrays.copyOfRange(t,3,t.length)));
249 members.put(m.id,m);
250 output.add("ADDMEMBER "+m.id+" "+m.type+" loans "+m.limit+" days "+m.period);
251 break;
252 }
253 case "WITHDRAW": {
254 Title b=book(t[1]);
255 int n=Integer.parseInt(t[2]);
256 check(n<1 || n>b.copies.size(),"unknown copy");
257 Copy c=b.copies.get(n-1);
258 check(!c.state.equals("AVAILABLE"),"copy not available");
259 c.state="WITHDRAWN";
260 output.add("WITHDRAW "+b.isbn+" copy "+n+" total "+b.holdings());
261 break;
262 }
263 case "BORROW": {
264 Member m=member(t[1]);
265 Title b=book(t[2]);
266 check(closed.contains(day),"library closed");
267 check(m.loans.containsKey(b.isbn),"already borrowed");
268 check(m.blocked(),"member blocked");
269 check(m.loans.values().stream().anyMatch(c -> c.due<day),"overdue loans");
270 check(m.loans.size()>=m.limit,"loan limit");
271 Copy c=m.holds.get(b.isbn);
272 if (c==null) c=b.copies.stream().filter(x -> x.state.equals("AVAILABLE")).findFirst().orElse(null);
273 check(c==null,b.count("HOLD")>0 ? "all copies on hold" : "no copy available");
274 m.holds.remove(b.isbn);
275 c.state="LOANED";
276 c.owner=m;
277 c.loanDay=day;
278 c.due=open(day+m.period);
279 c.renewals=0;
280 m.loans.put(b.isbn,c);
281 b.borrowed++;
282 borrowed++;
283 output.add("BORROW "+m.id+" "+b.isbn+" copy "+c.number+" due "+c.due);
284 break;
285 }
286 case "RETURN": {
287 Member m=member(t[1]);
288 Title b=book(t[2]);
289 Copy c=loan(m,b);
290 int[] values=endLoan(c,false);
291 returned++;
292 output.add("RETURN "+m.id+" "+b.isbn+" copy "+c.number+" late "+values[0]+" fine "+money(values[1])+" balance "+money(m.balance()));
293 release(c);
294 break;
295 }
296 case "RENEW": {
297 Member m=member(t[1]);
298 Copy c=loan(m,book(t[2]));
299 check(m.blocked(),"member blocked");
300 String reason=renewalReason(c);
301 check(reason!=null,reason);
302 renew(c);
303 break;
304 }
305 case "RENEWALL": {
306 Member m=member(t[1]);
307 check(m.loans.isEmpty(),"no loans");
308 check(m.blocked(),"member blocked");
309 int done=0, skipped=0;
310 for (Copy c:m.loans.values()) {
311 String reason=renewalReason(c);
312 if (reason!=null) {
313 output.add("SKIP "+c.title.isbn+" "+reason);
314 skipped++;
315 } else {
316 renew(c);
317 done++;
318 }
319 }
320 output.add("RENEWALL "+m.id+" renewed "+done+" skipped "+skipped);
321 break;
322 }
323 case "RESERVE": {
324 Member m=member(t[1]);
325 Title b=book(t[2]);
326 check(m.loans.containsKey(b.isbn),"already borrowed");
327 check(b.queue.contains(m) || m.holds.containsKey(b.isbn),"already reserved");
328 check(m.blocked(),"member blocked");
329 check(m.reservations()>=m.reservationLimit,"reservation limit");
330 check(b.holdings()==0,"no copies");
331 check(b.count("AVAILABLE")>0,"copy available");
332 b.queue.add(m);
333 m.queued++;
334 int p=b.queue.size();
335 List<Integer> dues=b.copies.stream().filter(c -> c.state.equals("LOANED")).map(c -> c.due).sorted().toList();
336 output.add("RESERVE "+m.id+" "+b.isbn+" position "+p+" estimate "+(p<=dues.size() ? dues.get(p-1).toString() : "none"));
337 break;
338 }
339 case "CANCEL": {
340 Member m=member(t[1]);
341 Title b=book(t[2]);
342 int p=b.queue.indexOf(m);
343 Copy c=m.holds.get(b.isbn);
344 check(p<0 && c==null,"not reserved");
345 if (p>=0) {
346 b.queue.remove(p);
347 m.queued--;
348 output.add("CANCEL "+m.id+" "+b.isbn+" queue");
349 } else {
350 m.holds.remove(b.isbn);
351 output.add("CANCEL "+m.id+" "+b.isbn+" hold copy "+c.number);
352 release(c);
353 }
354 break;
355 }
356 case "LOST": {
357 Member m=member(t[1]);
358 loss(loan(m,book(t[2])),"LOST");
359 break;
360 }
361 case "PAYFINE": {
362 Member m=member(t[1]);
363 check(m.balance()==0,"no fines");
364 int amount=parseMoney(t[2]), rest=amount;
365 List<String> details=new ArrayList<>();
366 for (int i=0;i<m.items.size() && rest>0;i++) {
367 Item item=m.items.get(i);
368 int applied=Math.min(rest,item.unpaid);
369 if (applied==0) continue;
370 rest-=applied;
371 item.unpaid-=applied;
372 details.add("ITEM "+(i+1)+" "+item.kind+" paid "+money(applied)+" remaining "+money(item.unpaid));
373 }
374 paid+=amount-rest;
375 output.add("PAYFINE "+m.id+" paid "+money(amount-rest)+" change "+money(rest)+" balance "+money(m.balance()));
376 output.addAll(details);
377 break;
378 }
379 case "WAIVE": {
380 Member m=member(t[1]);
381 int n=Integer.parseInt(t[2]);
382 check(n<1 || n>m.items.size(),"unknown item");
383 Item item=m.items.get(n-1);
384 check(item.unpaid==0,"item settled");
385 int amount=item.unpaid;
386 item.unpaid=0;
387 waived+=amount;
388 output.add("WAIVE "+m.id+" item "+n+" "+item.kind+" amount "+money(amount)+" balance "+money(m.balance()));
389 break;
390 }
391 case "FINES": {
392 Member m=member(t[1]);
393 output.add("FINES "+m.id+" items "+m.items.stream().filter(i -> i.unpaid>0).count()+" balance "+money(m.balance()));
394 for (int i=0;i<m.items.size();i++) {
395 Item item=m.items.get(i);
396 if (item.unpaid>0) output.add((i+1)+" day "+item.day+" "+item.kind+" "+item.isbn+" amount "+money(item.amount)+" remaining "+money(item.unpaid));
397 }
398 break;
399 }
400 case "SEARCH": {
401 String keyword=String.join(" ",Arrays.copyOfRange(t,1,t.length)).toLowerCase(Locale.ROOT);
402 List<Title> found=titles.values().stream().filter(b -> b.name.toLowerCase(Locale.ROOT).contains(keyword))
403 .sorted(Comparator.comparing((Title b) -> b.name.toLowerCase(Locale.ROOT)).thenComparing(b -> b.isbn)).toList();
404 output.add("SEARCH "+found.size());
405 for (Title b:found) output.add(b.isbn+" "+b.count("AVAILABLE")+"/"+b.holdings()+" "+b.name);
406 break;
407 }
408 case "OVERDUE": {
409 List<Copy> found=copies.stream().filter(c -> c.state.equals("LOANED") && c.due<day)
410 .sorted(Comparator.comparingInt((Copy c) -> c.due).thenComparing(c -> c.owner.id).thenComparing(c -> c.title.isbn)).toList();
411 output.add("OVERDUE "+found.size());
412 for (Copy c:found) output.add(c.owner.id+" "+c.title.isbn+" copy "+c.number+" due "+c.due+" late "+late(c)+" fine "+money(fine(c)));
413 break;
414 }
415 case "NOTICES": {
416 List<String> notices=new ArrayList<>();
417 for (Member m:members.values()) {
418 List<Copy> loans=orderedLoans(m);
419 for (Copy c:loans) if (c.due<day) notices.add("OVERDUE "+m.id+" "+c.title.isbn+" due "+c.due);
420 for (Copy c:loans) if (c.due>=day && c.due<=day+2) notices.add("DUESOON "+m.id+" "+c.title.isbn+" due "+c.due);
421 for (Copy c:orderedHolds(m)) notices.add("PICKUP "+m.id+" "+c.title.isbn+" until "+c.expiry);
422 }
423 output.add("NOTICES "+notices.size());
424 output.addAll(notices);
425 break;
426 }
427 case "MEMBER":
428 showMember(member(t[1]));
429 break;
430 case "MEMBERS": {
431 output.add("MEMBERS "+members.size());
432 for (Member m:members.values()) output.add(m.id+" "+m.type+" loans "+m.loans.size()+" reservations "+m.reservations()+
433 " balance "+money(m.balance())+" "+m.status());
434 break;
435 }
436 case "BOOK": {
437 Title b=book(t[1]);
438 output.add("BOOK "+b.isbn+" "+b.name);
439 output.add("PRICE "+money(b.price)+" copies "+b.holdings()+" borrowed "+b.borrowed);
440 for (Copy c:b.copies) output.add("COPY "+c.number+" "+c.state+
441 (c.state.equals("LOANED") ? " "+c.owner.id+" due "+c.due : c.state.equals("HOLD") ? " "+c.owner.id+" until "+c.expiry : ""));
442 output.add("QUEUE "+b.queue.size()+(b.queue.isEmpty() ? "" : " "+String.join(" ",b.queue.stream().map(m -> m.id).toList())));
443 break;
444 }
445 case "INVENTORY": {
446 long a=copies.stream().filter(c -> c.state.equals("AVAILABLE")).count();
447 long l=copies.stream().filter(c -> c.state.equals("LOANED")).count();
448 long h=copies.stream().filter(c -> c.state.equals("HOLD")).count();
449 long x=copies.stream().filter(c -> c.state.equals("LOST")).count();
450 long w=copies.stream().filter(c -> c.state.equals("WITHDRAWN")).count();
451 output.add("INVENTORY titles "+titles.size()+" copies "+(a+l+h)+" available "+a+" loaned "+l+" hold "+h+" lost "+x+" withdrawn "+w);
452 for (Title b:titles.values()) output.add(b.isbn+" copies "+b.holdings()+" available "+b.count("AVAILABLE")+" loaned "+b.count("LOANED")+
453 " hold "+b.count("HOLD")+" queue "+b.queue.size());
454 break;
455 }
456 case "HISTORY": {
457 Member m=member(t[1]);
458 output.add("HISTORY "+m.id+" "+m.history.size());
459 output.addAll(m.history);
460 break;
461 }
462 case "STATS": {
463 int active=members.values().stream().mapToInt(m -> m.loans.size()).sum();
464 int outstanding=members.values().stream().mapToInt(Member::balance).sum();
465 output.add("STATS day "+day+" titles "+titles.size()+" members "+members.size()+" closed "+closed.size());
466 output.add("LOANS borrowed "+borrowed+" returned "+returned+" renewed "+renewed+" lost "+lost+" active "+active);
467 output.add("MONEY fines "+money(fines)+" fees "+money(fees)+" charges "+money(charges)+" paid "+money(paid)+" waived "+money(waived)+" outstanding "+money(outstanding));
468 List<Title> top=titles.values().stream().filter(b -> b.borrowed>0)
469 .sorted(Comparator.comparingInt((Title b) -> b.borrowed).reversed().thenComparing(b -> b.isbn)).limit(3).toList();
470 for (int i=0;i<top.size();i++) {
471 Title b=top.get(i);
472 output.add("TOP "+(i+1)+" "+b.isbn+" "+b.borrowed+" "+b.name);
473 }
474 break;
475 }
476 }
477 }
478 List<String> process(String line) {
479 output=new ArrayList<>();
480 try { execute(line); }
481 catch (Fault fault) { output.add("ERROR "+fault.getMessage()); }
482 return output;
483 }

485 static int checksum(String s) {
486 return IntStream.range(0,s.length()).map(i -> (i%2==0 ? 1 : 3)*(s.charAt(i)-'0')).sum();
487 }
488 static String normalize(String raw) {
489 String s=raw.replace("-","");
490 if (s.matches("[0-9]{9}[0-9X]")) {
491 int sum=IntStream.range(0,10).map(i -> (10-i)*(s.charAt(i)=='X' ? 10 : s.charAt(i)-'0')).sum();
492 check(sum%11!=0,"invalid isbn");
493 String prefix="978"+s.substring(0,9);
494 sum=checksum(prefix);
495 return prefix+(10-sum%10)%10;
496 }
497 check(!s.matches("97[89][0-9]{10}"),"invalid isbn");
498 int sum=checksum(s);
499 check(sum%10!=0,"invalid isbn");
500 return s;
501 }
502 static int parseMoney(String s) { return Integer.parseInt(s.replace(".","")); }
503 static String money(int n) { return String.format(Locale.ROOT,"%d.%02d",n/100,n%100); }
504 public static void main(String[] args) throws IOException {
505 BufferedReader input=new BufferedReader(new InputStreamReader(System.in));
506 PrintWriter output=new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));
507 Library library=new Library();
508 int n=Integer.parseInt(input.readLine());
509 for (int i=0;i<n;i++) {
510 for (String line:library.process(input.readLine())) output.println(line);
511 }
512 output.flush();
513 }
