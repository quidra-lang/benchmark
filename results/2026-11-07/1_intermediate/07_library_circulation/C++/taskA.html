37string join(const vector<string> &a, int start) {
38 string result;
39 for (int i = start; i < int(a.size()); ++i) {
40 if (i != start) result += ' ';
41 result += a[i];
42 }
43 return result;

45string money(int value) {
46 return std::to_string(value/100) + "." + (value%100 < 10 ? "0" : "") + std::to_string(value%100);
47}
48int cents(const string &s) {
49 std::size_t dot = s.find('.');
50 return std::stoi(s.substr(0,dot))*100 + std::stoi(s.substr(dot+1));
51}
52string lower(string s) {
53 for (char &c : s) if (c >= 'A' && c <= 'Z') c += 'a'-'A';
54 return s;
55}
56string normalize(const string &input) {
57 string s = input;
58 s.erase(std::remove(s.begin(),s.end(),'-'),s.end());
59 if (s.size() == 10) {
60 int sum = 0;
61 for (int i = 0; i < 10; ++i) {
62 if (!(s[i] >= '0' && s[i] <= '9') && !(i == 9 && s[i] == 'X')) return "";
63 sum += (10-i)*(s[i] == 'X' ? 10 : s[i]-'0');
64 }
65 if (sum%11) return "";
66 s = "978" + s.substr(0,9);
67 sum = 0;
68 for (int i = 0; i < 12; ++i) sum += (i%2 ? 3 : 1)*(s[i]-'0');
69 s += char('0'+(10-sum%10)%10);
70 return s;
71 }
72 if (s.size() != 13 || (s.substr(0,3) != "978" && s.substr(0,3) != "979")) return "";
73 int sum = 0;
74 for (int i = 0; i < 13; ++i) {
75 if (s[i] < '0' || s[i] > '9') return "";
76 sum += (i%2 ? 3 : 1)*(s[i]-'0');
77 }
78 return sum%10 ? "" : s;
79}
80struct Library {
81 int day = 1, borrowed = 0, returned = 0, renewed = 0, lost = 0, paid = 0, waived = 0;

89 void line(const string &s) { output.push_back(s); }
90 bool error(bool condition, const string &reason) {
91 if (condition) line("ERROR " + reason);
92 return condition;
93 }
94 int open(int d) const {
95 while (closed.count(d)) ++d;
96 return d;
97 }
98 int holdings(int t) const { return titles[t].counts[AVAILABLE]+titles[t].counts[LOANED]+titles[t].counts[HOLD]; }
99 Copy &copy(Ref r) { return titles[r.title].copies[r.copy]; }
100 const Copy &copy(Ref r) const { return titles[r.title].copies[r.copy]; }
101 void state(Ref r, State next) {
102 Title &t = titles[r.title];
103 --t.counts[t.copies[r.copy].state];
104 t.copies[r.copy].state = next;
105 ++t.counts[next];
106 }
107 static void remove(vector<Ref> &refs, Ref r) {
108 refs.erase(std::find_if(refs.begin(),refs.end(),[r](Ref q) { return q.title == r.title && q.copy == r.copy; }));
109 }
110 int find(const vector<Ref> &refs, int title) const {
111 for (Ref r : refs) if (r.title == title) return r.copy;
112 return -1;
113 }
114 int member(const string &id) {
115 std::map<string,int>::const_iterator found = memberIds.find(id);
116 if (error(found == memberIds.end(),"unknown member")) return -1;
117 return found->second;
118 }
119 int title(const string &input) {
120 string isbn = normalize(input);
121 if (error(isbn.empty(),"invalid isbn")) return -1;
122 std::map<string,int>::const_iterator found = titleIds.find(isbn);
123 if (error(found == titleIds.end(),"unknown book")) return -1;
124 return found->second;
125 }
126 bool pair(const vector<string> &a, int &m, int &t) {
127 m = member(a[1]);
128 if (m < 0) return false;
129 t = title(a[2]);
130 return t >= 0;
131 }
132 int late(Ref r) const { return std::max(0,day-copy(r).due); }
133 int fine(Ref r) const {
134 const Member &m = members[copy(r).member];
135 return std::min(late(r)*m.rate,m.cap);
136 }
137 void charge(int m, int kind, int t, int amount) {
138 if (!amount) return;
139 members[m].charges.push_back({day,kind,t,amount,amount});
140 members[m].balance += amount;
141 charged[kind] += amount;
142 }
143 void release(Ref r) {
144 Title &t = titles[r.title];
145 if (t.queue.empty()) {
146 state(r,AVAILABLE);
147 return;
148 }
149 int m = t.queue.front();
150 t.queue.erase(t.queue.begin());
151 state(r,HOLD);
152 Copy &c = copy(r);
153 c.member = m;
154 c.expiry = open(day+3);
155 members[m].holds.push_back(r);
156 line("HOLD " + t.isbn + " copy " + std::to_string(r.copy+1) + " " + members[m].id + " until " + std::to_string(c.expiry));
157 }
158 void dropQueue(int t) {
159 if (holdings(t)) return;
160 for (int m : titles[t].queue) {
161 --members[m].reservations;
162 line("DROPPED " + titles[t].isbn + " " + members[m].id);
163 }
164 titles[t].queue.clear();
165 }
166 void endLoan(Ref r, bool loss, const string &heading) {
167 Copy c = copy(r);
168 int m = c.member, l = late(r), f = fine(r);
169 Member &person = members[m];
170 Title &book = titles[r.title];
171 remove(person.loans,r);
172 charge(m,0,r.title,f);
173 string history = string(loss ? "LOST" : "RETURNED") + " " + book.isbn + " copy " + std::to_string(r.copy+1) + " from " + std::to_string(c.from) + " to " + std::to_string(day) + " late " + std::to_string(l) + " fine " + money(f);
174 if (loss) {
175 history += " price " + money(book.price);
176 charge(m,2,r.title,book.price);
177 state(r,LOST);
178 ++lost;
179 } else ++returned;
180 person.history.push_back(history);
181 string result = heading + " " + person.id + " " + book.isbn + " copy " + std::to_string(r.copy+1) + " late " + std::to_string(l) + " fine " + money(f);
182 if (loss) result += " price " + money(book.price);
183 line(result + " balance " + money(person.balance));
184 if (loss) dropQueue(r.title);
185 else release(r);
186 }
187 string renewalReason(Ref r) const {
188 if (late(r)) return "loan overdue";
189 if (copy(r).renewals == 2) return "renewal limit";
190 if (!titles[r.title].queue.empty()) return "title reserved";
191 return "";
192 }
193 void renew(Ref r) {
194 Copy &c = copy(r);
195 Member &m = members[c.member];
196 c.due = open(day+m.period);
197 ++c.renewals;
198 ++renewed;
199 line("RENEW " + m.id + " " + titles[r.title].isbn + " copy " + std::to_string(r.copy+1) + " due " + std::to_string(c.due) + " renewals " + std::to_string(c.renewals));
200 }
201 vector<Ref> ordered(const vector<Ref> &refs, bool holds = false) const {
202 vector<Ref> result = refs;
203 std::sort(result.begin(),result.end(),[this,holds](Ref a, Ref b) {
204 int x = holds ? copy(a).expiry : copy(a).due, y = holds ? copy(b).expiry : copy(b).due;
205 return std::tie(x,titles[a.title].isbn) < std::tie(y,titles[b.title].isbn);
206 });
207 return result;
208 }
209 void execute(const vector<string> &a) {
210 const string &cmd = a[0];
211 if (cmd == "DAY") {
212 int target = std::stoi(a[1]);
213 if (error(target <= day,"invalid day")) return;
214 line("DAY " + a[1]);
215 while (day < target) {
216 ++day;
217 vector<Ref> losses;
218 for (const auto &[memberId, id] : memberIds) {
219 for (Ref r : members[id].loans) if (copy(r).due+365 <= day) losses.push_back(r);
220 }
221 for (Ref r : losses) endLoan(r,true,"AUTOLOST");

223 for (const auto &[isbn, id] : titleIds) {
224 int t = id;
225 for (int i = 0; i < int(titles[t].copies.size()); ++i) {
226 const Copy &c = titles[t].copies[i];
227 if (c.state == HOLD && c.expiry < day) expiries.push_back({t,i});
228 }

230 for (Ref r : expiries) {
231 int m = copy(r).member;

234 --members[m].reservations;

236 release(r);
237 }
238 }
239 } else if (cmd == "CLOSED") {
240 int d = std::stoi(a[1]);
241 if (error(d <= day,"invalid day") || error(closed.count(d),"already closed")) return;
242 closed.insert(d);
243 line("CLOSED " + a[1]);
244 for (const auto &[isbn, id] : titleIds) {
245 Title &t = titles[id];
246 for (int i = 0; i < int(t.copies.size()); ++i) {
247 Copy &c = t.copies[i];
248 if (c.state == LOANED && c.due == d) {
249 c.due = open(d);
250 line("MOVED " + t.isbn + " copy " + std::to_string(i+1) + " " + members[c.member].id + " due " + std::to_string(c.due));
251 } else if (c.state == HOLD && c.expiry == d) {
252 c.expiry = open(d);
253 line("MOVED " + t.isbn + " copy " + std::to_string(i+1) + " " + members[c.member].id + " until " + std::to_string(c.expiry));
254 }
255 }
256 }
257 } else if (cmd == "ADDBOOK") {
258 string isbn = normalize(a[1]), name = join(a,4);
259 if (error(isbn.empty(),"invalid isbn")) return;
260 std::map<string,int>::const_iterator found = titleIds.find(isbn);
261 int t = found == titleIds.end() ? -1 : found->second;
262 if (error(t >= 0 && titles[t].name != name,"title mismatch")) return;
263 if (t < 0) {
264 t = int(titles.size());

266 book.isbn = isbn;
267 book.name = name;
268 book.folded = lower(name);
269 titles.push_back(book);
270 titleIds[isbn] = t;
271 }
272 Title &book = titles[t];
273 book.price = cents(a[3]);
274 int start = int(book.copies.size()), count = std::stoi(a[2]);
275 book.copies.resize(start+count);
276 book.counts[AVAILABLE] += count;
277 line("ADDBOOK " + isbn + " added " + a[2] + " total " + std::to_string(holdings(t)));
278 for (int i = start; i < start+count; ++i) release({t,i});
279 } else if (cmd == "ADDMEMBER") {
280 if (error(memberIds.count(a[1]),"duplicate member") || error(a[2] != "STUDENT" && a[2] != "STAFF","unknown type")) return;
281 Member m;
282 m.id = a[1];
283 m.type = a[2];
284 m.name = join(a,3);
285 bool staff = m.type == "STAFF";
286 m.limit = staff ? 10 : 3;
287 m.period = staff ? 28 : 14;
288 m.rate = staff ? 10 : 25;
289 m.cap = staff ? 2000 : 1000;
290 m.reserveLimit = staff ? 5 : 2;
291 m.fee = staff ? 100 : 50;
292 memberIds[m.id] = int(members.size());
293 members.push_back(m);
294 line("ADDMEMBER " + m.id + " " + m.type + " loans " + std::to_string(m.limit) + " days " + std::to_string(m.period));
295 } else if (cmd == "WITHDRAW") {
296 int t = title(a[1]);
297 if (t < 0) return;
298 int i = std::stoi(a[2])-1;
299 if (error(i < 0 || i >= int(titles[t].copies.size()),"unknown copy")) return;
300 if (error(titles[t].copies[i].state != AVAILABLE,"copy not available")) return;
301 state({t,i},WITHDRAWN);
302 line("WITHDRAW " + titles[t].isbn + " copy " + a[2] + " total " + std::to_string(holdings(t)));
303 } else if (cmd == "BORROW" || cmd == "RETURN" || cmd == "RENEW" || cmd == "RESERVE" || cmd == "CANCEL" || cmd == "LOST") {
304 int m, t;
305 if (!pair(a,m,t)) return;
306 Member &person = members[m];
307 Title &book = titles[t];
308 int loan = find(person.loans,t), hold = find(person.holds,t);
309 vector<int>::iterator queued = std::find(book.queue.begin(),book.queue.end(),m);
310 if (cmd == "BORROW") {
311 if (error(closed.count(day),"library closed") || error(loan >= 0,"already borrowed") || error(person.balance > 500,"member blocked")) return;
312 bool overdue = std::any_of(person.loans.begin(),person.loans.end(),[this](Ref r) { return late(r) > 0; });
313 if (error(overdue,"overdue loans") || error(int(person.loans.size()) >= person.limit,"loan limit")) return;
314 const auto available = std::find_if(book.copies.begin(),book.copies.end(),[](const Copy &c) { return c.state == AVAILABLE; });
315 int i = hold >= 0 ? hold : available == book.copies.end() ? -1 : int(available-book.copies.begin());
316 if (error(i < 0,book.counts[HOLD] ? "all copies on hold" : "no copy available")) return;
317 Ref r{t,i};
318 if (hold >= 0) {
319 remove(person.holds,r);
320 --person.reservations;
321 }
322 state(r,LOANED);
323 Copy &c = copy(r);
324 c.member = m;
325 c.from = day;
326 c.due = open(day+person.period);
327 c.renewals = 0;
328 person.loans.push_back(r);
329 std::sort(person.loans.begin(),person.loans.end(),[this](Ref x,Ref y) { return titles[x.title].isbn < titles[y.title].isbn; });
330 ++borrowed;
331 ++book.borrowed;
332 line("BORROW " + person.id + " " + book.isbn + " copy " + std::to_string(i+1) + " due " + std::to_string(c.due));
333 } else if (cmd == "RETURN" || cmd == "LOST" || cmd == "RENEW") {
334 if (error(loan < 0,"not borrowed")) return;
335 Ref r{t,loan};
336 if (cmd == "RENEW") {
337 if (error(person.balance > 500,"member blocked")) return;
338 string reason = renewalReason(r);
339 if (error(!reason.empty(),reason)) return;
340 renew(r);
341 } else endLoan(r,cmd == "LOST",cmd);
342 } else if (cmd == "RESERVE") {
343 if (error(loan >= 0,"already borrowed") || error(queued != book.queue.end() || hold >= 0,"already reserved") || error(person.balance > 500,"member blocked") || error(person.reservations >= person.reserveLimit,"reservation limit") || error(!holdings(t),"no copies") || error(book.counts[AVAILABLE] > 0,"copy available")) return;
344 book.queue.push_back(m);
345 ++person.reservations;
346 vector<int> dues;
347 for (const Copy &c : book.copies) if (c.state == LOANED) dues.push_back(c.due);
348 std::sort(dues.begin(),dues.end());
349 int p = int(book.queue.size());
350 line("RESERVE " + person.id + " " + book.isbn + " position " + std::to_string(p) + " estimate " + (p <= int(dues.size()) ? std::to_string(dues[p-1]) : "none"));
351 } else {
352 if (error(queued == book.queue.end() && hold < 0,"not reserved")) return;
353 --person.reservations;
354 if (queued != book.queue.end()) {
355 book.queue.erase(queued);
356 line("CANCEL " + person.id + " " + book.isbn + " queue");
357 } else {
358 Ref r{t,hold};
359 remove(person.holds,r);
360 line("CANCEL " + person.id + " " + book.isbn + " hold copy " + std::to_string(hold+1));
361 release(r);
362 }
363 }
364 } else if (cmd == "RENEWALL") {
365 int m = member(a[1]);
366 if (m < 0) return;
367 if (error(members[m].loans.empty(),"no loans") || error(members[m].balance > 500,"member blocked")) return;
368 int good = 0, skipped = 0;
369 for (Ref r : members[m].loans) {
370 string reason = renewalReason(r);
371 if (reason.empty()) {
372 renew(r);
373 ++good;
374 } else {
375 line("SKIP " + titles[r.title].isbn + " " + reason);
376 ++skipped;
377 }
378 }
379 line("RENEWALL " + a[1] + " renewed " + std::to_string(good) + " skipped " + std::to_string(skipped));
380 } else if (cmd == "PAYFINE" || cmd == "WAIVE" || cmd == "FINES") {
381 int m = member(a[1]);
382 if (m < 0) return;
383 Member &person = members[m];
384 if (cmd == "PAYFINE") {
385 if (error(!person.balance,"no fines")) return;
386 int amount = cents(a[2]), remaining = amount;
387 vector<string> details;
388 for (int i = 0; i < int(person.charges.size()) && remaining; ++i) {
389 Charge &c = person.charges[i];
390 int applied = std::min(remaining,c.unpaid);
391 if (!applied) continue;
392 c.unpaid -= applied;
393 remaining -= applied;
394 details.push_back("ITEM " + std::to_string(i+1) + " " + kinds[c.kind] + " paid " + money(applied) + " remaining " + money(c.unpaid));
395 }
396 person.balance -= amount-remaining;
397 paid += amount-remaining;
398 line("PAYFINE " + person.id + " paid " + money(amount-remaining) + " change " + money(remaining) + " balance " + money(person.balance));
399 for (const string &s : details) line(s);
400 } else if (cmd == "WAIVE") {
401 int i = std::stoi(a[2])-1;
402 if (error(i < 0 || i >= int(person.charges.size()),"unknown item")) return;
403 Charge &c = person.charges[i];
404 if (error(!c.unpaid,"item settled")) return;
405 int amount = c.unpaid;
406 c.unpaid = 0;
407 person.balance -= amount;
408 waived += amount;
409 line("WAIVE " + person.id + " item " + a[2] + " " + kinds[c.kind] + " amount " + money(amount) + " balance " + money(person.balance));
410 } else {
411 int count = std::count_if(person.charges.begin(),person.charges.end(),[](const Charge &c) { return c.unpaid > 0; });
412 line("FINES " + person.id + " items " + std::to_string(count) + " balance " + money(person.balance));
413 for (int i = 0; i < int(person.charges.size()); ++i) {
414 const Charge &c = person.charges[i];
415 if (c.unpaid) line(std::to_string(i+1) + " day " + std::to_string(c.day) + " " + kinds[c.kind] + " " + titles[c.title].isbn + " amount " + money(c.amount) + " remaining " + money(c.unpaid));
416 }
417 }
418 } else if (cmd == "SEARCH") {
419 string keyword = lower(join(a,1));
420 vector<int> found;
421 for (int t = 0; t < int(titles.size()); ++t) if (titles[t].folded.find(keyword) != string::npos) found.push_back(t);
422 std::sort(found.begin(),found.end(),[this](int a,int b) { return std::tie(titles[a].folded,titles[a].isbn) < std::tie(titles[b].folded,titles[b].isbn); });
423 line("SEARCH " + std::to_string(found.size()));
424 for (int t : found) line(titles[t].isbn + " " + std::to_string(titles[t].counts[AVAILABLE]) + "/" + std::to_string(holdings(t)) + " " + titles[t].name);
425 } else if (cmd == "OVERDUE") {
426 vector<Ref> refs;
427 for (const Member &m : members) for (Ref r : m.loans) if (late(r)) refs.push_back(r);
428 std::sort(refs.begin(),refs.end(),[this](Ref a,Ref b) { return std::tie(copy(a).due,members[copy(a).member].id,titles[a.title].isbn) < std::tie(copy(b).due,members[copy(b).member].id,titles[b.title].isbn); });
429 line("OVERDUE " + std::to_string(refs.size()));
430 for (Ref r : refs) line(members[copy(r).member].id + " " + titles[r.title].isbn + " copy " + std::to_string(r.copy+1) + " due " + std::to_string(copy(r).due) + " late " + std::to_string(late(r)) + " fine " + money(fine(r)));
431 } else if (cmd == "NOTICES") {
432 vector<string> notices;
433 for (const auto &[memberId, id] : memberIds) {
434 const Member &m = members[id];
435 for (Ref r : ordered(m.loans)) if (copy(r).due <= day+2) notices.push_back(string(late(r) ? "OVERDUE " : "DUESOON ") + m.id + " " + titles[r.title].isbn + " due " + std::to_string(copy(r).due));
436 for (Ref r : ordered(m.holds,true)) notices.push_back("PICKUP " + m.id + " " + titles[r.title].isbn + " until " + std::to_string(copy(r).expiry));
437 }
438 line("NOTICES " + std::to_string(notices.size()));
439 for (const string &s : notices) line(s);
440 } else if (cmd == "MEMBER" || cmd == "MEMBERS" || cmd == "HISTORY") {
441 int target = -1;
442 if (cmd != "MEMBERS") {
443 target = member(a[1]);
444 if (target < 0) return;
445 }
446 if (cmd == "HISTORY") {
447 line("HISTORY " + a[1] + " " + std::to_string(members[target].history.size()));
448 for (const string &s : members[target].history) line(s);
449 return;
450 }
451 if (cmd == "MEMBERS") line("MEMBERS " + std::to_string(members.size()));
452 for (const auto &[memberId, id] : memberIds) {
453 if (target >= 0 && id != target) continue;
454 const Member &m = members[id];
455 string status = m.balance > 500 ? "BLOCKED" : "OK";
456 if (cmd == "MEMBERS") {
457 line(m.id + " " + m.type + " loans " + std::to_string(m.loans.size()) + " reservations " + std::to_string(m.reservations) + " balance " + money(m.balance) + " " + status);
458 continue;
459 }
460 line("MEMBER " + m.id + " " + m.type + " " + m.name);
461 line("STATUS loans " + std::to_string(m.loans.size()) + "/" + std::to_string(m.limit) + " reservations " + std::to_string(m.reservations) + "/" + std::to_string(m.reserveLimit) + " balance " + money(m.balance) + " " + status);
462 for (Ref r : ordered(m.loans)) {
463 string s = "LOAN " + titles[r.title].isbn + " copy " + std::to_string(r.copy+1) + " due " + std::to_string(copy(r).due) + " renewals " + std::to_string(copy(r).renewals);
464 if (late(r)) s += " late " + std::to_string(late(r)) + " fine " + money(fine(r));
465 line(s);
466 }
