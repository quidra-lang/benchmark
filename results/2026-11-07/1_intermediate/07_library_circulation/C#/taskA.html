7class Program {
8 class Fault : Exception {
9 public Fault(string reason) : base(reason) {}
10 }
11 static void check(bool condition, string reason) { if (condition) throw new Fault(reason); }
12 class Item {
13 public int day, amount, unpaid;
14 public string kind, isbn;
15 public Item(int day, string kind, string isbn, int amount) {
16 this.day=day;
17 this.kind=kind;
18 this.isbn=isbn;
19 this.amount=amount;
20 unpaid=amount;
21 }
22 }
23 class Member {
24 public string id, type, name;
25 public int limit, period, rate, cap, reservationLimit, fee, queued;
26 public SortedDictionary<string,Copy> loans=new(StringComparer.Ordinal), holds=new(StringComparer.Ordinal);
27 public List<Item> items=new();
28 public List<string> history=new();
29 public Member(string id, string type, string name) {
30 this.id=id;
31 this.type=type;
32 this.name=name;
33 bool student=type.Equals("STUDENT");
34 limit=student ? 3 : 10;
35 period=student ? 14 : 28;
36 rate=student ? 25 : 10;
37 cap=student ? 1000 : 2000;
38 reservationLimit=student ? 2 : 5;
39 fee=student ? 50 : 100;
40 }
41 public int balance() { return items.Sum(i => i.unpaid); }
42 public int reservations() { return queued+holds.Count; }
43 public bool blocked() { return balance()>500; }
44 public string status() { return blocked() ? "BLOCKED" : "OK"; }
45 }
46 class Title {
47 public string isbn, name;
48 public int price, borrowed;
49 public List<Copy> copies=new();
50 public List<Member> queue=new();
51 public Title(string isbn, string name, int price) {
52 this.isbn=isbn;
53 this.name=name;
54 this.price=price;
55 }
56 public int count(string state) { return (int)copies.Count(c => c.state.Equals(state)); }
57 public int holdings() { return count("AVAILABLE")+count("LOANED")+count("HOLD"); }
58 }
59 class Copy {
60 public Title title;
61 public int number, loanDay, due, renewals, expiry;
62 public string state="AVAILABLE";
63 public Member owner;
64 public Copy(Title title, int number) {
65 this.title=title;
66 this.number=number;
67 }
68 }
69 class Library {
70 public int day=1, borrowed, returned, renewed, lost, fines, fees, charges, paid, waived;
71 public SortedDictionary<string,Title> titles=new(StringComparer.Ordinal);
72 public SortedDictionary<string,Member> members=new(StringComparer.Ordinal);
73 public HashSet<int> closed=new();
74 public List<Copy> copies=new();
75 public List<string> output;
76 public Member member(string id) {
77 Member m=members.GetValueOrDefault(id);
78 check(m==null,"unknown member");
79 return m;
80 }
81 public Title book(string raw) {
82 string isbn=normalize(raw);
83 Title b=titles.GetValueOrDefault(isbn);
84 check(b==null,"unknown book");
85 return b;
86 }
87 public Copy loan(Member m, Title b) {
88 Copy c=m.loans.GetValueOrDefault(b.isbn);
89 check(c==null,"not borrowed");
90 return c;
91 }
92 public int open(int d) {
93 while (closed.Contains(d)) d++;
94 return d;
95 }
96 public int late(Copy c) { return Math.Max(0,day-c.due); }
97 public int fine(Copy c) { return Math.Min(late(c)*c.owner.rate,c.owner.cap); }
98 public void charge(Member m, string kind, Title b, int amount) {
99 if (amount==0) return;
100 m.items.Add(new Item(day,kind,b.isbn,amount));
101 switch (kind) {
102 case "FINE":
103 fines+=amount;
104 break;
105 case "FEE":
106 fees+=amount;
107 break;
108 default:
109 charges+=amount;
110 break;
111 }
112 }
113 public void release(Copy c) {
114 Title b=c.title;
115 c.owner=null;
116 if (b.queue.Count==0) c.state="AVAILABLE";
117 else {
118 Member m=b.queue[0];
119 b.queue.RemoveAt(0);
120 m.queued--;
121 c.state="HOLD";
122 c.owner=m;
123 c.expiry=open(day+3);
124 m.holds[b.isbn]=c;
125 output.Add("HOLD "+b.isbn+" copy "+c.number+" "+m.id+" until "+c.expiry);
126 }
127 }
128 public int[] endLoan(Copy c, bool loss) {
129 Member m=c.owner;
130 int l=late(c), f=fine(c);
131 m.loans.Remove(c.title.isbn);
132 charge(m,"FINE",c.title,f);
133 m.history.Add((loss ? "LOST " : "RETURNED ")+c.title.isbn+" copy "+c.number+
134 " from "+c.loanDay+" to "+day+" late "+l+" fine "+money(f)+
135 (loss ? " price "+money(c.title.price) : ""));
136 return new int[]{l,f};
137 }
138 public void loss(Copy c, string heading) {
139 Member m=c.owner;
140 Title b=c.title;
141 int[] values=endLoan(c,true);
142 charge(m,"PRICE",b,b.price);
143 c.state="LOST";
144 c.owner=null;
145 lost++;
146 output.Add(heading+" "+m.id+" "+b.isbn+" copy "+c.number+" late "+values[0]+" fine "+money(values[1])+
147 " price "+money(b.price)+" balance "+money(m.balance()));
148 if (b.holdings()==0) {
149 foreach (Member queued in b.queue) {
150 queued.queued--;
151 output.Add("DROPPED "+b.isbn+" "+queued.id);
152 }
153 b.queue.Clear();
154 }
155 }
156 public string renewalReason(Copy c) {
157 if (c.due<day) return "loan overdue";
158 if (c.renewals==2) return "renewal limit";
159 if (c.title.queue.Count!=0) return "title reserved";
160 return null;
161 }
162 public void renew(Copy c) {
163 c.due=open(day+c.owner.period);
164 c.renewals++;
165 renewed++;
166 output.Add("RENEW "+c.owner.id+" "+c.title.isbn+" copy "+c.number+" due "+c.due+" renewals "+c.renewals);
167 }
168 public List<Copy> orderedLoans(Member m) {
169 return m.loans.Values.OrderBy(c => c.due).ThenBy(c => c.title.isbn,StringComparer.Ordinal).ToList();
170 }
171 public List<Copy> orderedHolds(Member m) {
172 return m.holds.Values.OrderBy(c => c.expiry).ThenBy(c => c.title.isbn,StringComparer.Ordinal).ToList();
173 }
174 public void showMember(Member m) {
175 output.Add("MEMBER "+m.id+" "+m.type+" "+m.name);
176 output.Add("STATUS loans "+m.loans.Count+"/"+m.limit+" reservations "+m.reservations()+"/"+m.reservationLimit+
177 " balance "+money(m.balance())+" "+m.status());
178 foreach (Copy c in orderedLoans(m)) output.Add("LOAN "+c.title.isbn+" copy "+c.number+" due "+c.due+" renewals "+c.renewals+
179 (c.due<day ? " late "+late(c)+" fine "+money(fine(c)) : ""));
180 foreach (Copy c in orderedHolds(m)) output.Add("HOLD "+c.title.isbn+" copy "+c.number+" until "+c.expiry);
181 foreach (Title b in titles.Values) {
182 int p=b.queue.IndexOf(m);
183 if (p>=0) output.Add("QUEUE "+b.isbn+" position "+(p+1));
184 }
185 }
186 public void execute(string line) {
187 string[] t=line.Split(' ');
188 switch (t[0]) {
189 case "DAY": {
190 int d=int.Parse(t[1]);
191 check(d<=day,"invalid day");
192 output.Add("DAY "+d);
193 while (day<d) {
194 day++;
195 List<Copy> losses=copies.Where(c => c.state.Equals("LOANED") && c.due+365<=day)
196 .OrderBy(c => c.owner.id,StringComparer.Ordinal).ThenBy(c => c.title.isbn,StringComparer.Ordinal).ToList();
197 foreach (Copy c in losses) loss(c,"AUTOLOST");
198 List<Copy> expired=copies.Where(c => c.state.Equals("HOLD") && c.expiry<day)
199 .OrderBy(c => c.title.isbn,StringComparer.Ordinal).ThenBy(c => c.number).ToList();
200 foreach (Copy c in expired) {
201 Member m=c.owner;
202 charge(m,"FEE",c.title,m.fee);
203 m.holds.Remove(c.title.isbn);
204 output.Add("EXPIRED "+c.title.isbn+" copy "+c.number+" "+m.id+" fee "+money(m.fee));
205 release(c);
206 }
207 }
208 break;
209 }
210 case "CLOSED": {
211 int d=int.Parse(t[1]);
212 check(d<=day,"invalid day");
213 check(closed.Contains(d),"already closed");
214 closed.Add(d);
215 output.Add("CLOSED "+d);
216 foreach (Title b in titles.Values) {
217 foreach (Copy c in b.copies) {
218 if (c.state.Equals("LOANED") && c.due==d) {
219 c.due=open(d);
220 output.Add("MOVED "+b.isbn+" copy "+c.number+" "+c.owner.id+" due "+c.due);
221 } else if (c.state.Equals("HOLD") && c.expiry==d) {
222 c.expiry=open(d);
223 output.Add("MOVED "+b.isbn+" copy "+c.number+" "+c.owner.id+" until "+c.expiry);
224 }
225 }
226 }
227 break;
228 }
229 case "ADDBOOK": {
230 string isbn=normalize(t[1]), name=string.Join(" ",t.Skip(4));
231 Title b=titles.GetValueOrDefault(isbn);
232 check(b!=null && !b.name.Equals(name),"title mismatch");
233 int price=parseMoney(t[3]), count=int.Parse(t[2]);
234 if (b==null) {
235 b=new Title(isbn,name,price);
236 titles[isbn]=b;
237 }
238 b.price=price;
239 List<Copy> added=new();
240 for (int i=0;i<count;i++) {
241 Copy c=new Copy(b,b.copies.Count+1);
242 b.copies.Add(c);
243 copies.Add(c);
244 added.Add(c);
245 }
246 output.Add("ADDBOOK "+isbn+" added "+count+" total "+b.holdings());
247 foreach (Copy c in added) release(c);
248 break;
249 }
250 case "ADDMEMBER": {
251 check(members.ContainsKey(t[1]),"duplicate member");
252 check(!t[2].Equals("STUDENT") && !t[2].Equals("STAFF"),"unknown type");
253 Member m=new Member(t[1],t[2],string.Join(" ",t.Skip(3)));
254 members[m.id]=m;
255 output.Add("ADDMEMBER "+m.id+" "+m.type+" loans "+m.limit+" days "+m.period);
256 break;
257 }
258 case "WITHDRAW": {
259 Title b=book(t[1]);
260 int n=int.Parse(t[2]);
261 check(n<1 || n>b.copies.Count,"unknown copy");
262 Copy c=b.copies[n-1];
263 check(!c.state.Equals("AVAILABLE"),"copy not available");
264 c.state="WITHDRAWN";
265 output.Add("WITHDRAW "+b.isbn+" copy "+n+" total "+b.holdings());
266 break;
267 }
268 case "BORROW": {
269 Member m=member(t[1]);
270 Title b=book(t[2]);
271 check(closed.Contains(day),"library closed");
272 check(m.loans.ContainsKey(b.isbn),"already borrowed");
273 check(m.blocked(),"member blocked");
274 check(m.loans.Values.Any(c => c.due<day),"overdue loans");
275 check(m.loans.Count>=m.limit,"loan limit");
276 Copy c=m.holds.GetValueOrDefault(b.isbn);
277 if (c==null) c=b.copies.FirstOrDefault(x => x.state.Equals("AVAILABLE"));
278 check(c==null,b.count("HOLD")>0 ? "all copies on hold" : "no copy available");
279 m.holds.Remove(b.isbn);
280 c.state="LOANED";
281 c.owner=m;
282 c.loanDay=day;
283 c.due=open(day+m.period);
284 c.renewals=0;
285 m.loans[b.isbn]=c;
286 b.borrowed++;
287 borrowed++;
288 output.Add("BORROW "+m.id+" "+b.isbn+" copy "+c.number+" due "+c.due);
289 break;
290 }
291 case "RETURN": {
292 Member m=member(t[1]);
293 Title b=book(t[2]);
294 Copy c=loan(m,b);
295 int[] values=endLoan(c,false);
296 returned++;
297 output.Add("RETURN "+m.id+" "+b.isbn+" copy "+c.number+" late "+values[0]+" fine "+money(values[1])+" balance "+money(m.balance()));
298 release(c);
299 break;
300 }
301 case "RENEW": {
302 Member m=member(t[1]);
303 Copy c=loan(m,book(t[2]));
304 check(m.blocked(),"member blocked");
305 string reason=renewalReason(c);
306 check(reason!=null,reason);
307 renew(c);
308 break;
309 }
310 case "RENEWALL": {
311 Member m=member(t[1]);
312 check(m.loans.Count==0,"no loans");
313 check(m.blocked(),"member blocked");
314 int done=0, skipped=0;
315 foreach (Copy c in m.loans.Values) {
316 string reason=renewalReason(c);
317 if (reason!=null) {
318 output.Add("SKIP "+c.title.isbn+" "+reason);
319 skipped++;
320 } else {
321 renew(c);
322 done++;
323 }
324 }
325 output.Add("RENEWALL "+m.id+" renewed "+done+" skipped "+skipped);
326 break;
327 }
328 case "RESERVE": {
329 Member m=member(t[1]);
330 Title b=book(t[2]);
331 check(m.loans.ContainsKey(b.isbn),"already borrowed");
332 check(b.queue.Contains(m) || m.holds.ContainsKey(b.isbn),"already reserved");
333 check(m.blocked(),"member blocked");
334 check(m.reservations()>=m.reservationLimit,"reservation limit");
335 check(b.holdings()==0,"no copies");
336 check(b.count("AVAILABLE")>0,"copy available");
337 b.queue.Add(m);
338 m.queued++;
339 int p=b.queue.Count;
340 List<int> dues=b.copies.Where(c => c.state.Equals("LOANED")).Select(c => c.due).OrderBy(d => d).ToList();
341 output.Add("RESERVE "+m.id+" "+b.isbn+" position "+p+" estimate "+(p<=dues.Count ? dues[p-1].ToString() : "none"));
342 break;
343 }
344 case "CANCEL": {
345 Member m=member(t[1]);
346 Title b=book(t[2]);
347 int p=b.queue.IndexOf(m);
348 Copy c=m.holds.GetValueOrDefault(b.isbn);
349 check(p<0 && c==null,"not reserved");
350 if (p>=0) {
351 b.queue.RemoveAt(p);
352 m.queued--;
353 output.Add("CANCEL "+m.id+" "+b.isbn+" queue");
354 } else {
355 m.holds.Remove(b.isbn);
356 output.Add("CANCEL "+m.id+" "+b.isbn+" hold copy "+c.number);
357 release(c);
358 }
359 break;
360 }
361 case "LOST": {
362 Member m=member(t[1]);
363 loss(loan(m,book(t[2])),"LOST");
364 break;
365 }
366 case "PAYFINE": {
367 Member m=member(t[1]);
368 check(m.balance()==0,"no fines");
369 int amount=parseMoney(t[2]), rest=amount;
370 List<string> details=new();
371 for (int i=0;i<m.items.Count && rest>0;i++) {
372 Item item=m.items[i];
373 int applied=Math.Min(rest,item.unpaid);
374 if (applied==0) continue;
375 rest-=applied;
376 item.unpaid-=applied;
377 details.Add("ITEM "+(i+1)+" "+item.kind+" paid "+money(applied)+" remaining "+money(item.unpaid));
378 }
379 paid+=amount-rest;
380 output.Add("PAYFINE "+m.id+" paid "+money(amount-rest)+" change "+money(rest)+" balance "+money(m.balance()));
381 output.AddRange(details);
382 break;
383 }
384 case "WAIVE": {
385 Member m=member(t[1]);
386 int n=int.Parse(t[2]);
387 check(n<1 || n>m.items.Count,"unknown item");
388 Item item=m.items[n-1];
389 check(item.unpaid==0,"item settled");
390 int amount=item.unpaid;
391 item.unpaid=0;
392 waived+=amount;
393 output.Add("WAIVE "+m.id+" item "+n+" "+item.kind+" amount "+money(amount)+" balance "+money(m.balance()));
394 break;
395 }
396 case "FINES": {
397 Member m=member(t[1]);
398 output.Add("FINES "+m.id+" items "+m.items.Count(i => i.unpaid>0)+" balance "+money(m.balance()));
399 for (int i=0;i<m.items.Count;i++) {
400 Item item=m.items[i];
401 if (item.unpaid>0) output.Add((i+1)+" day "+item.day+" "+item.kind+" "+item.isbn+" amount "+money(item.amount)+" remaining "+money(item.unpaid));
402 }
403 break;
404 }
405 case "SEARCH": {
406 string keyword=string.Join(" ",t.Skip(1)).ToLowerInvariant();
407 List<Title> found=titles.Values.Where(b => b.name.ToLowerInvariant().Contains(keyword))
408 .OrderBy(b => b.name.ToLowerInvariant(),StringComparer.Ordinal).ThenBy(b => b.isbn,StringComparer.Ordinal).ToList();
409 output.Add("SEARCH "+found.Count);
410 foreach (Title b in found) output.Add(b.isbn+" "+b.count("AVAILABLE")+"/"+b.holdings()+" "+b.name);
411 break;
412 }
413 case "OVERDUE": {
414 List<Copy> found=copies.Where(c => c.state.Equals("LOANED") && c.due<day)
415 .OrderBy(c => c.due).ThenBy(c => c.owner.id,StringComparer.Ordinal).ThenBy(c => c.title.isbn,StringComparer.Ordinal).ToList();
416 output.Add("OVERDUE "+found.Count);
417 foreach (Copy c in found) output.Add(c.owner.id+" "+c.title.isbn+" copy "+c.number+" due "+c.due+" late "+late(c)+" fine "+money(fine(c)));
418 break;
419 }
420 case "NOTICES": {
421 List<string> notices=new();
422 foreach (Member m in members.Values) {
423 List<Copy> loans=orderedLoans(m);
424 foreach (Copy c in loans) if (c.due<day) notices.Add("OVERDUE "+m.id+" "+c.title.isbn+" due "+c.due);
425 foreach (Copy c in loans) if (c.due>=day && c.due<=day+2) notices.Add("DUESOON "+m.id+" "+c.title.isbn+" due "+c.due);
426 foreach (Copy c in orderedHolds(m)) notices.Add("PICKUP "+m.id+" "+c.title.isbn+" until "+c.expiry);
427 }
428 output.Add("NOTICES "+notices.Count);
429 output.AddRange(notices);
430 break;
431 }
432 case "MEMBER":
433 showMember(member(t[1]));
434 break;
435 case "MEMBERS": {
436 output.Add("MEMBERS "+members.Count);
437 foreach (Member m in members.Values) output.Add(m.id+" "+m.type+" loans "+m.loans.Count+" reservations "+m.reservations()+
438 " balance "+money(m.balance())+" "+m.status());
439 break;
440 }
441 case "BOOK": {
442 Title b=book(t[1]);
443 output.Add("BOOK "+b.isbn+" "+b.name);
444 output.Add("PRICE "+money(b.price)+" copies "+b.holdings()+" borrowed "+b.borrowed);
445 foreach (Copy c in b.copies) output.Add("COPY "+c.number+" "+c.state+
446 (c.state.Equals("LOANED") ? " "+c.owner.id+" due "+c.due : c.state.Equals("HOLD") ? " "+c.owner.id+" until "+c.expiry : ""));
447 output.Add("QUEUE "+b.queue.Count+(b.queue.Count==0 ? "" : " "+string.Join(" ",b.queue.Select(m => m.id))));
448 break;
449 }
450 case "INVENTORY": {
451 int a=copies.Count(c => c.state == "AVAILABLE"), l=copies.Count(c => c.state == "LOANED");
452 int h=copies.Count(c => c.state == "HOLD"), x=copies.Count(c => c.state == "LOST");
453 int w=copies.Count(c => c.state == "WITHDRAWN");
454 output.Add("INVENTORY titles "+titles.Count+" copies "+(a+l+h)+" available "+a+" loaned "+l+" hold "+h+" lost "+x+" withdrawn "+w);
455 foreach (Title b in titles.Values) output.Add(b.isbn+" copies "+b.holdings()+" available "+b.count("AVAILABLE")+" loaned "+b.count("LOANED")+
456 " hold "+b.count("HOLD")+" queue "+b.queue.Count);
457 break;
458 }
459 case "HISTORY": {
460 Member m=member(t[1]);
461 output.Add("HISTORY "+m.id+" "+m.history.Count);
462 output.AddRange(m.history);
463 break;
464 }
465 case "STATS": {
466 int active=members.Values.Sum(m => m.loans.Count);
467 int outstanding=members.Values.Sum(m => m.balance());
468 output.Add("STATS day "+day+" titles "+titles.Count+" members "+members.Count+" closed "+closed.Count);
469 output.Add("LOANS borrowed "+borrowed+" returned "+returned+" renewed "+renewed+" lost "+lost+" active "+active);
470 output.Add("MONEY fines "+money(fines)+" fees "+money(fees)+" charges "+money(charges)+" paid "+money(paid)+" waived "+money(waived)+" outstanding "+money(outstanding));
471 List<Title> top=titles.Values.Where(b => b.borrowed>0)
472 .OrderByDescending(b => b.borrowed).ThenBy(b => b.isbn,StringComparer.Ordinal).Take(3).ToList();
473 for (int i=0;i<top.Count;i++) {
474 Title b=top[i];
475 output.Add("TOP "+(i+1)+" "+b.isbn+" "+b.borrowed+" "+b.name);
476 }
477 break;
478 }
479 }
480 }
481 public List<string> process(string line) {
482 output=new();
483 try { execute(line); }
484 catch (Fault fault) { output.Add("ERROR "+fault.Message); }
485 return output;
486 }

488 static int checksum(string s) => s.Select((c,i) => (i%2==0 ? 1 : 3)*(c-'0')).Sum();
489 static string normalize(string raw) {
490 string s=raw.Replace("-","");
491 if (matches(s,"[0-9]{9}[0-9X]")) {
492 int sum=s.Select((c,i) => (10-i)*(c=='X' ? 10 : c-'0')).Sum();
493 check(sum%11!=0,"invalid isbn");
494 string prefix="978"+s.Substring(0,9);
495 sum=checksum(prefix);
496 return prefix+(10-sum%10)%10;
497 }
498 check(!matches(s,"97[89][0-9]{10}"),"invalid isbn");
499 int sum13=checksum(s);
500 check(sum13%10!=0,"invalid isbn");
501 return s;
502 }
503 static int parseMoney(string s) => int.Parse(s.Replace(".",""),CultureInfo.InvariantCulture);
504 static string money(int n) => $"{n/100}.{n%100:D2}";
505 static bool matches(string s, string pattern) => Regex.IsMatch(s,"\\A(?:"+pattern+")\\z",RegexOptions.CultureInvariant);
506 static void Main() {
507 CultureInfo.CurrentCulture=CultureInfo.InvariantCulture;
508 Library library=new();
509 int n=int.Parse(Console.ReadLine()!);
510 for (int i=0;i<n;i++) {
511 foreach (string line in library.process(Console.ReadLine()!)) Console.WriteLine(line);
512 }
