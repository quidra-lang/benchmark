int member_id(const State state, string text)
    for: 0 <= k < int(#state.patrons)
        if state.patrons[k].id == text
            return k
    return -1
int title_id(const State state, string text)
    for: 0 <= k < int(#state.titles)
        if state.titles[k].isbn == text
            return k
    return -1
int rule(const State state, int person, int field)
    student := [3, 14, 25, 1000, 2, 50]
    staff := [10, 28, 10, 2000, 5, 100]
    return if state.patrons[person].category == "STUDENT" then student[field] else staff[field]
bool is_closed(const State state, int date)
    for entry := state.closed[]
        if entry == date
            return true
    return false
int open_day(const State state, int date)
    while is_closed(state, date)
        date += 1
    return date
int balance(const State state, int person)
    unpaid := [for item := state.patrons[person].items[] then item.unpaid]
    return unpaid.sum()
int copy_of(const State state, int person, int book, int status)
    for index := state.titles[book].copies[]
        if state.copies[index].status == status and state.copies[index].member == person
            return index
    return -1
int late_days(const State state, int index)
    return math.max(0, state.day - state.copies[index].due)
int fine_amount(const State state, int index)
    return math.min(late_days(state, index) * rule(state, state.copies[index].member, 2), rule(state, state.copies[index].member, 3))
void charge(State &state, int person, int book, string kind, int amount)
    if amount > 0
        state.patrons[person].items = state.patrons[person].items.append(Charge(state.day, kind, book, amount, amount))
        state.totals[if kind == "FINE" then 4 elif kind == "FEE" then 5 else 6] += amount
void release_copy(State &state, int index)
    book := state.copies[index].title
    state.copies[index].member = -1
    if #state.titles[book].queue == 0
        state.copies[index].status = 0
        return
    person := state.titles[book].queue[0]
    state.titles[book].queue = state.titles[book].queue[1:]
    state.copies[index].status = 2
    state.copies[index].member = person
    state.copies[index].expiry = open_day(state, state.day + 3)
    emit("HOLD " + state.titles[book].isbn + " copy " + string(state.copies[index].number) + " " + state.patrons[person].id + " until " + string(state.copies[index].expiry))
void end_loan(State &state, int index, bool lost)
    item := state.copies[index]
    late := late_days(state, index)
    amount := fine_amount(state, index)
    charge(&state, item.member, item.title, "FINE", amount)
    entry := (if lost then "LOST" else "RETURNED") + " " + state.titles[item.title].isbn + " copy " + string(item.number) + " from " + string(item.loan_day) + " to " + string(state.day) + " late " + string(late) + " fine " + money(amount)
    state.patrons[item.member].history = state.patrons[item.member].history.append(entry)
void advance_day(State &state, int target)
    emit("DAY " + string(target))
    while true
        next_day := target + 1
        for: 0 <= copy_index < state.copy_used
            item := state.copies[copy_index]
            if item.status == 1
                next_day = math.min(next_day, math.max(state.day + 1, item.due + 365))
            elif item.status == 2
                next_day = math.min(next_day, math.max(state.day + 1, item.expiry + 1))
        if next_day > target
            state.day = target
            return
        state.day = next_day
        Ordered[] expiries := []
        for: 0 <= k < state.copy_used
            item := state.copies[k]
            if item.status == 2 and item.expiry < state.day
                expiries = expiries.append(Ordered(k, state.titles[item.title].isbn + padded(item.number)))
        for index := ordered(expiries)[]
            item := state.copies[index]
            amount := rule(state, item.member, 5)
            charge(&state, item.member, item.title, "FEE", amount)
            emit("EXPIRED " + state.titles[item.title].isbn + " copy " + string(item.number) + " " + state.patrons[item.member].id + " fee " + money(amount))
            release_copy(&state, index)
void command(State &state, string line)
    words := line.split(" ")
    action := words[0]
    if action == "DAY" or action == "CLOSED"
        date := parse_integer(words[1])
        if date <= state.day
            error_line("invalid day")
        elif action == "DAY"
            advance_day(&state, date)
    personal := ["BORROW", "RETURN", "RENEW", "RENEWALL", "RESERVE", "CANCEL", "LOST", "PAYFINE", "WAIVE", "FINES", "MEMBER", "HISTORY"]
    needs_member := false
    for name := personal[]
        if action == name
            needs_member = true
    if needs_member
        person := member_id(state, words[1])
        if person < 0
            error_line("unknown member")
            return
        id := state.patrons[person].id
        normalized := isbn_value(words[2])
        book := title_id(state, normalized)
        if normalized == ""
            error_line("invalid isbn")
            return
        if book < 0
            error_line("unknown book")
            return
        loan := copy_of(state, person, book, 1)
        elif action == "RETURN" or action == "LOST" or action == "RENEW"
            if loan < 0
                error_line("not borrowed")
            elif action == "LOST"
            else
                late := late_days(state, loan)
                amount := fine_amount(state, loan)
                end_loan(&state, loan, false)
                state.totals[1] += 1
                emit("RETURN " + id + " " + normalized + " copy " + string(state.copies[loan].number) + " late " + string(late) + " fine " + money(amount) + " balance " + money(balance(state, person)))
                release_copy(&state, loan)
