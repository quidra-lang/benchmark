37string isbn_value(string text)
38 raw := [for k := (0:<int(#text))[] if text[k] != "-" then text[k]].join("")
39 if #raw != 10 and #raw != 13
40 return ""
41 total := 0
42 for: 0 <= k < int(#raw)
43 int digit
44 match "0123456789".find(raw[k])
45 nat position
46 digit = int(position)
47 none
48 digit = -1
49 if #raw == 10 and k == 9 and raw[k] == "X"
50 digit = 10
51 if digit < 0
52 return ""
53 total += digit * (if #raw == 10 then 10 - k elif k % 2 == 0 then 1 else 3)
54 if #raw == 10
55 if total % 11 != 0
56 return ""
57 raw = "978" + raw[:<9]
58 total = 0
59 for: 0 <= k < 12
60 total += parse_integer(raw[k]) * (if k % 2 == 0 then 1 else 3)
61 return raw + string((10 - total % 10) % 10)
62 if total % 10 != 0 or (not raw.starts_with("978") and not raw.starts_with("979"))
63 return ""
64 return raw
