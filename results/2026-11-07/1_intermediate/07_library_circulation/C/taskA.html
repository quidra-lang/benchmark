52static char *format(const char *pattern, ...) {
53 va_list args, measure;
54 va_start(args,pattern);
55 va_copy(measure,args);
56 int length = vsnprintf(NULL,0,pattern,measure);
57 va_end(measure);
58 if (length < 0) exit(EXIT_FAILURE);
59 Temp *t = resize(NULL,sizeof(*t)+(size_t)length+1);
60 t->next = temps;
61 temps = t;
62 vsnprintf(t->data,(size_t)length+1,pattern,args);
63 va_end(args);
64 return t->data;
65}
66static void clear_temps(void) {
67 while (temps) {
68 Temp *next = temps->next;
69 free(temps);
70 temps = next;
71 }
72}

90static char *join(char **a, int count, int start) {
91 char result[256] = "";
92 for (int i = start; i < count; ++i) {
93 if (i != start) strcat(result," ");
94 strcat(result,a[i]);
95 }
96 return format("%s",result);
97}
98static void lower(char *destination, const char *s) {
99 while (*s) {
100 *destination++ = (char)tolower((unsigned char)*s);
101 ++s;
102 }
103 *destination = '\0';
104}
105static char *normalize(const char *input) {
106 char s[18];
107 int length = 0;
108 for (; *input; ++input) if (*input != '-') s[length++] = *input;
109 s[length] = '\0';
110 if (length == 10) {
111 int sum = 0;
112 for (int i = 0; i < 10; ++i) {
113 if (!(s[i] >= '0' && s[i] <= '9') && !(i == 9 && s[i] == 'X')) return NULL;
114 sum += (10-i)*(s[i] == 'X' ? 10 : s[i]-'0');
115 }
116 if (sum%11) return NULL;
117 char *result = format("978%.9s0",s);
118 sum = 0;
119 for (int i = 0; i < 12; ++i) sum += (i%2 ? 3 : 1)*(result[i]-'0');
120 result[12] = (char)('0'+(10-sum%10)%10);
121 return result;
122 }
123 if (length != 13 || (strncmp(s,"978",3) && strncmp(s,"979",3))) return NULL;
124 int sum = 0;
125 for (int i = 0; i < 13; ++i) {
126 if (s[i] < '0' || s[i] > '9') return NULL;
127 sum += (i%2 ? 3 : 1)*(s[i]-'0');
128 }
129 return sum%10 ? NULL : format("%s",s);
130}

151static int open_day(int d) {
152 while (d <= 3650 && closed[d]) ++d;
153 return d;
154}
155static Copy *copy(Ref r) { return &titles[r.title]->copies[r.copy]; }
156static int holdings(int t) { return titles[t]->counts[0]+titles[t]->counts[1]+titles[t]->counts[2]; }
157static void state(Ref r, int next) {
158 --titles[r.title]->counts[copy(r)->state];
159 copy(r)->state = next;
160 ++titles[r.title]->counts[next];
161}
162static int find_ref(const Ref *refs, int count, int t) {
163 for (int i = 0; i < count; ++i) if (refs[i].title == t) return refs[i].copy;
164 return -1;
165}
166static void remove_ref(Ref *refs, int *count, Ref r) {
167 for (int i = 0; i < *count; ++i) {
168 if (refs[i].title == r.title && refs[i].copy == r.copy) {
169 memmove(refs+i,refs+i+1,(size_t)(*count-i-1)*sizeof(*refs));
170 --*count;
171 return;
172 }
173 }
174}
175static void remove_queue(Title *t, int i) {
176 memmove(t->queue+i,t->queue+i+1,(size_t)(t->queue_count-i-1)*sizeof(*t->queue));
177 --t->queue_count;
178}
179static int queue_position(const Title *t, int m) {
180 for (int i = 0; i < t->queue_count; ++i) if (t->queue[i] == m) return i;
181 return -1;
182}
183static int late(Ref r) { return day > copy(r)->due ? day-copy(r)->due : 0; }
184static int fine(Ref r) {
185 Member *m = members[copy(r)->member];
186 int amount = late(r)*m->rate;
187 return amount < m->cap ? amount : m->cap;
188}
189static void charge(int m, int kind, int t, int amount) {
190 if (!amount) return;
191 Member *p = members[m];
192 if (p->charge_count == p->charge_capacity) {
193 p->charge_capacity = p->charge_capacity ? 2*p->charge_capacity : 8;
194 p->charges = resize(p->charges,(size_t)p->charge_capacity*sizeof(*p->charges));
195 }
196 p->charges[p->charge_count++] = (Charge){day,kind,t,amount,amount};
197 p->balance += amount;
198 charged[kind] += amount;
199}
200static void history(Member *m, const char *s) {
201 if (m->history_count == m->history_capacity) {
202 m->history_capacity = m->history_capacity ? 2*m->history_capacity : 8;
203 m->history = resize(m->history,(size_t)m->history_capacity*sizeof(*m->history));
204 }
205 m->history[m->history_count++] = save(s);
206}
207static void release(Ref r) {
208 Title *t = titles[r.title];
209 if (!t->queue_count) {
210 state(r,AVAILABLE);
211 return;
212 }
213 int m = t->queue[0];
214 remove_queue(t,0);
215 state(r,HOLD);
216 copy(r)->member = m;
217 copy(r)->expiry = open_day(day+3);
218 members[m]->holds[members[m]->hold_count++] = r;
219 line("HOLD %s copy %d %s until %d",t->isbn,r.copy+1,members[m]->id,copy(r)->expiry);
220}
221static void drop_queue(int t) {
222 if (holdings(t)) return;
223 Title *book = titles[t];
224 for (int i = 0; i < book->queue_count; ++i) {
225 int m = book->queue[i];
226 --members[m]->reservations;
227 line("DROPPED %s %s",book->isbn,members[m]->id);
228 }
229 book->queue_count = 0;
230}
231static void end_loan(Ref r, bool loss, const char *heading) {
232 Copy c = *copy(r);
233 int m = c.member, l = late(r), f = fine(r);
234 Member *p = members[m];
235 Title *t = titles[r.title];
236 remove_ref(p->loans,&p->loan_count,r);
237 charge(m,0,r.title,f);
238 char *entry = format("%s %s copy %d from %d to %d late %d fine %s",loss ? "LOST" : "RETURNED",t->isbn,r.copy+1,c.from,day,l,money(f));
239 if (loss) {
240 entry = format("%s price %s",entry,money(t->price));
241 charge(m,2,r.title,t->price);
242 state(r,LOST);
243 ++lost;
244 } else ++returned;
245 history(p,entry);
246 if (loss) line("%s %s %s copy %d late %d fine %s price %s balance %s",heading,p->id,t->isbn,r.copy+1,l,money(f),money(t->price),money(p->balance));
247 else line("RETURN %s %s copy %d late %d fine %s balance %s",p->id,t->isbn,r.copy+1,l,money(f),money(p->balance));
248 if (loss) drop_queue(r.title);
249 else release(r);
250}
251static const char *renewal_reason(Ref r) {
252 if (late(r)) return "loan overdue";
253 if (copy(r)->renewals == 2) return "renewal limit";
254 if (titles[r.title]->queue_count) return "title reserved";
255 return NULL;
256}
257static void renew(Ref r) {
258 Copy *c = copy(r);
259 Member *m = members[c->member];
260 c->due = open_day(day+m->period);
261 ++c->renewals;
262 ++renewed;
263 line("RENEW %s %s copy %d due %d renewals %d",m->id,titles[r.title]->isbn,r.copy+1,c->due,c->renewals);
264}
265static int isbn_order(const void *left, const void *right) {
266 const Ref *a = left, *b = right;
267 return strcmp(titles[a->title]->isbn,titles[b->title]->isbn);
268}
269static int due_order(const void *left, const void *right) {
270 const Ref *a = left, *b = right;
271 if (copy(*a)->due != copy(*b)->due) return copy(*a)->due < copy(*b)->due ? -1 : 1;
272 int order = strcmp(members[copy(*a)->member]->id,members[copy(*b)->member]->id);
273 return order ? order : isbn_order(left,right);
274}
275static int expiry_order(const void *left, const void *right) {
276 const Ref *a = left, *b = right;
277 if (copy(*a)->expiry != copy(*b)->expiry) return copy(*a)->expiry < copy(*b)->expiry ? -1 : 1;
278 return isbn_order(left,right);
279}
280static int integer_order(const void *left, const void *right) {
281 int a = *(const int *)left, b = *(const int *)right;
282 return a < b ? -1 : a > b ? 1 : 0;
283}
284static int search_order(const void *left, const void *right) {
285 int a = *(const int *)left, b = *(const int *)right;
286 int order = strcmp(titles[a]->folded,titles[b]->folded);
287 return order ? order : strcmp(titles[a]->isbn,titles[b]->isbn);
288}
289static int top_order(const void *left, const void *right) {
290 int a = *(const int *)left, b = *(const int *)right;
291 if (titles[a]->borrowed != titles[b]->borrowed) return titles[a]->borrowed > titles[b]->borrowed ? -1 : 1;
292 return strcmp(titles[a]->isbn,titles[b]->isbn);
293}
294static void execute(char **a, int count) {
295 const char *cmd = a[0];
296 if (!strcmp(cmd,"DAY")) {
297 int target = atoi(a[1]);
298 if (error(target <= day,"invalid day")) return;
299 line("DAY %d",target);
300 while (day < target) {
301 ++day;
302 Ref losses[10000];
303 int loss_count = 0;
304 for (int j = 0; j < member_count; ++j) {
305 Member *m = members[member_order[j]];
306 for (int i = 0; i < m->loan_count; ++i) if (copy(m->loans[i])->due+365 <= day) losses[loss_count++] = m->loans[i];
307 }
308 for (int i = 0; i < loss_count; ++i) end_loan(losses[i],true,"AUTOLOST");
309 Ref expiries[20000];
310 int expiry_count = 0;
311 for (int j = 0; j < title_count; ++j) {
312 int t = title_order[j];
313 for (int i = 0; i < titles[t]->copy_count; ++i) {
314 Ref r = {t,i};
315 if (copy(r)->state == HOLD && copy(r)->expiry < day) expiries[expiry_count++] = r;
316 }
317 }
318 for (int i = 0; i < expiry_count; ++i) {
319 Ref r = expiries[i];
320 int m = copy(r)->member;
321 charge(m,1,r.title,members[m]->fee);
322 remove_ref(members[m]->holds,&members[m]->hold_count,r);
323 --members[m]->reservations;
324 line("EXPIRED %s copy %d %s fee %s",titles[r.title]->isbn,r.copy+1,members[m]->id,money(members[m]->fee));
325 release(r);
326 }
327 }
328 } else if (!strcmp(cmd,"CLOSED")) {
329 int d = atoi(a[1]);
330 if (error(d <= day,"invalid day") || error(closed[d],"already closed")) return;
331 closed[d] = true;
332 ++closed_count;
333 line("CLOSED %d",d);
334 for (int j = 0; j < title_count; ++j) {
335 Title *t = titles[title_order[j]];
336 for (int i = 0; i < t->copy_count; ++i) {
337 Copy *c = &t->copies[i];
338 if (c->state == LOANED && c->due == d) {
339 c->due = open_day(d);
340 line("MOVED %s copy %d %s due %d",t->isbn,i+1,members[c->member]->id,c->due);
341 } else if (c->state == HOLD && c->expiry == d) {
342 c->expiry = open_day(d);
343 line("MOVED %s copy %d %s until %d",t->isbn,i+1,members[c->member]->id,c->expiry);
344 }
345 }
346 }
347 } else if (!strcmp(cmd,"ADDBOOK")) {
348 char *isbn = normalize(a[1]), *name = join(a,count,4);
349 if (error(!isbn,"invalid isbn")) return;
350 int t = find_title(isbn);
351 if (error(t >= 0 && strcmp(titles[t]->name,name),"title mismatch")) return;
352 if (t < 0) {
353 t = title_count;
354 titles[t] = calloc(1,sizeof(*titles[t]));
355 if (!titles[t]) exit(EXIT_FAILURE);
356 strcpy(titles[t]->isbn,isbn);
357 strcpy(titles[t]->name,name);
358 lower(titles[t]->folded,name);
359 int position = title_count;
360 while (position && strcmp(titles[title_order[position-1]]->isbn,isbn) > 0) {
361 title_order[position] = title_order[position-1];
362 --position;
363 }
364 title_order[position] = t;
365 ++title_count;
366 }
367 Title *book = titles[t];
368 book->price = cents(a[3]);
369 int start = book->copy_count, added = atoi(a[2]);
370 book->copy_count += added;
371 book->counts[AVAILABLE] += added;
372 line("ADDBOOK %s added %d total %d",isbn,added,holdings(t));
373 for (int i = start; i < book->copy_count; ++i) release((Ref){t,i});
374 } else if (!strcmp(cmd,"ADDMEMBER")) {
375 if (error(find_member(a[1]) >= 0,"duplicate member") || error(strcmp(a[2],"STUDENT") && strcmp(a[2],"STAFF"),"unknown type")) return;
376 int id = member_count;
377 Member *m = calloc(1,sizeof(*m));
378 if (!m) exit(EXIT_FAILURE);
379 members[id] = m;
380 strcpy(m->id,a[1]);
381 strcpy(m->type,a[2]);
382 strcpy(m->name,join(a,count,3));
383 bool staff = !strcmp(m->type,"STAFF");
384 m->limit = staff ? 10 : 3;
385 m->period = staff ? 28 : 14;
386 m->rate = staff ? 10 : 25;
387 m->cap = staff ? 2000 : 1000;
388 m->reserve_limit = staff ? 5 : 2;
389 m->fee = staff ? 100 : 50;
390 int position = member_count;
391 while (position && strcmp(members[member_order[position-1]]->id,m->id) > 0) {
392 member_order[position] = member_order[position-1];
393 --position;
394 }
395 member_order[position] = id;
396 ++member_count;
397 line("ADDMEMBER %s %s loans %d days %d",m->id,m->type,m->limit,m->period);
398 } else if (!strcmp(cmd,"WITHDRAW")) {
399 int t = title(a[1]);
400 if (t < 0) return;
401 int i = atoi(a[2])-1;
402 if (error(i < 0 || i >= titles[t]->copy_count,"unknown copy") || error(titles[t]->copies[i].state != AVAILABLE,"copy not available")) return;
403 state((Ref){t,i},WITHDRAWN);
404 line("WITHDRAW %s copy %d total %d",titles[t]->isbn,i+1,holdings(t));
405 } else if (!strcmp(cmd,"BORROW") || !strcmp(cmd,"RETURN") || !strcmp(cmd,"RENEW") || !strcmp(cmd,"RESERVE") || !strcmp(cmd,"CANCEL") || !strcmp(cmd,"LOST")) {
406 int m = member(a[1]);
407 if (m < 0) return;
408 int t = title(a[2]);
409 if (t < 0) return;
410 Member *p = members[m];
411 Title *book = titles[t];
412 int loan = find_ref(p->loans,p->loan_count,t), hold = find_ref(p->holds,p->hold_count,t), q = queue_position(book,m);
413 if (!strcmp(cmd,"BORROW")) {
414 if (error(closed[day],"library closed") || error(loan >= 0,"already borrowed") || error(p->balance > 500,"member blocked")) return;
415 bool overdue = false;
416 for (int i = 0; i < p->loan_count; ++i) overdue |= late(p->loans[i]) > 0;
417 if (error(overdue,"overdue loans") || error(p->loan_count >= p->limit,"loan limit")) return;
418 int available = -1;
419 for (int i = 0; i < book->copy_count; ++i) {
420 if (book->copies[i].state == AVAILABLE) {
421 available = i;
422 break;
423 }
424 }
425 int i = hold >= 0 ? hold : available;
426 if (error(i < 0,book->counts[HOLD] ? "all copies on hold" : "no copy available")) return;
427 Ref r = {t,i};
428 if (hold >= 0) {
429 remove_ref(p->holds,&p->hold_count,r);
430 --p->reservations;
431 }
432 state(r,LOANED);
433 *copy(r) = (Copy){LOANED,m,day,open_day(day+p->period),0,0};
434 p->loans[p->loan_count++] = r;
435 qsort(p->loans,(size_t)p->loan_count,sizeof(*p->loans),isbn_order);
436 ++borrowed;
437 ++book->borrowed;
438 line("BORROW %s %s copy %d due %d",p->id,book->isbn,i+1,copy(r)->due);
439 } else if (!strcmp(cmd,"RETURN") || !strcmp(cmd,"LOST") || !strcmp(cmd,"RENEW")) {
440 if (error(loan < 0,"not borrowed")) return;
441 Ref r = {t,loan};
442 if (!strcmp(cmd,"RENEW")) {
443 if (error(p->balance > 500,"member blocked")) return;
444 const char *reason = renewal_reason(r);
445 if (error(reason != NULL,reason)) return;
446 renew(r);
447 } else end_loan(r,!strcmp(cmd,"LOST"),cmd);
448 } else if (!strcmp(cmd,"RESERVE")) {
449 if (error(loan >= 0,"already borrowed") || error(q >= 0 || hold >= 0,"already reserved") || error(p->balance > 500,"member blocked") || error(p->reservations >= p->reserve_limit,"reservation limit") || error(!holdings(t),"no copies") || error(book->counts[AVAILABLE] > 0,"copy available")) return;
450 book->queue[book->queue_count++] = m;
451 ++p->reservations;
452 int dues[99], length = 0;
453 for (int i = 0; i < book->copy_count; ++i) if (book->copies[i].state == LOANED) dues[length++] = book->copies[i].due;
454 qsort(dues,(size_t)length,sizeof(*dues),integer_order);
455 line("RESERVE %s %s position %d estimate %s",p->id,book->isbn,book->queue_count,book->queue_count <= length ? format("%d",dues[book->queue_count-1]) : "none");
456 } else {
457 if (error(q < 0 && hold < 0,"not reserved")) return;
458 --p->reservations;
459 if (q >= 0) {
460 remove_queue(book,q);
461 line("CANCEL %s %s queue",p->id,book->isbn);
462 } else {
463 Ref r = {t,hold};
464 remove_ref(p->holds,&p->hold_count,r);
465 line("CANCEL %s %s hold copy %d",p->id,book->isbn,hold+1);
466 release(r);
467 }
468 }
469 } else if (!strcmp(cmd,"RENEWALL")) {
470 int m = member(a[1]);
471 if (m < 0) return;
472 Member *p = members[m];
473 if (error(!p->loan_count,"no loans") || error(p->balance > 500,"member blocked")) return;
474 int good = 0, skipped = 0;
475 for (int i = 0; i < p->loan_count; ++i) {
476 Ref r = p->loans[i];
477 const char *reason = renewal_reason(r);
478 if (reason) {
479 line("SKIP %s %s",titles[r.title]->isbn,reason);
480 ++skipped;
481 } else {
482 renew(r);
483 ++good;
484 }
485 }
486 line("RENEWALL %s renewed %d skipped %d",p->id,good,skipped);
487 } else if (!strcmp(cmd,"PAYFINE") || !strcmp(cmd,"WAIVE") || !strcmp(cmd,"FINES")) {
488 int m = member(a[1]);
489 if (m < 0) return;
490 Member *p = members[m];
491 if (!strcmp(cmd,"PAYFINE")) {
492 if (error(!p->balance,"no fines")) return;
493 int amount = cents(a[2]), remaining = amount;
494 int *applied = calloc((size_t)p->charge_count,sizeof(*applied));
495 if (!applied) exit(EXIT_FAILURE);
496 for (int i = 0; i < p->charge_count && remaining; ++i) {
497 Charge *c = &p->charges[i];
498 applied[i] = remaining < c->unpaid ? remaining : c->unpaid;
499 c->unpaid -= applied[i];
500 remaining -= applied[i];
501 }
502 p->balance -= amount-remaining;
503 paid += amount-remaining;
504 line("PAYFINE %s paid %s change %s balance %s",p->id,money(amount-remaining),money(remaining),money(p->balance));
505 for (int i = 0; i < p->charge_count; ++i) if (applied[i]) line("ITEM %d %s paid %s remaining %s",i+1,kinds[p->charges[i].kind],money(applied[i]),money(p->charges[i].unpaid));
506 free(applied);
507 } else if (!strcmp(cmd,"WAIVE")) {
508 int i = atoi(a[2])-1;
509 if (error(i < 0 || i >= p->charge_count,"unknown item")) return;
510 Charge *c = &p->charges[i];
511 if (error(!c->unpaid,"item settled")) return;
512 int amount = c->unpaid;
513 c->unpaid = 0;
514 p->balance -= amount;
515 waived += amount;
516 line("WAIVE %s item %d %s amount %s balance %s",p->id,i+1,kinds[c->kind],money(amount),money(p->balance));
517 } else {
518 int count_unpaid = 0;
519 for (int i = 0; i < p->charge_count; ++i) count_unpaid += p->charges[i].unpaid > 0;
520 line("FINES %s items %d balance %s",p->id,count_unpaid,money(p->balance));
