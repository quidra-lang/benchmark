1import Foundation
2
3enum Fault: Error { case reason(String) }
4func failIf(_ condition: Bool, _ reason: String) throws {
5 if condition { throw Fault.reason(reason) }
6}
7func matches(_ s: String, _ pattern: String) -> Bool {
8 s.range(of:"\\A(?:"+pattern+")\\z",options:.regularExpression) != nil
9}
10func checksum(_ s: String) -> Int {
11 s.utf8.enumerated().reduce(0) { $0+($1.offset%2 == 0 ? 1 : 3)*(Int($1.element)-48) }
12}
13func normalize(_ raw: String) throws -> String {
14 let s = raw.replacingOccurrences(of:"-",with:"")
15 if matches(s,"[0-9]{9}[0-9X]") {
16 let sum = s.utf8.enumerated().reduce(0) { $0+(10-$1.offset)*($1.element == 88 ? 10 : Int($1.element)-48) }
17 try failIf(sum%11 != 0,"invalid isbn")
18 let prefix = "978"+String(s.prefix(9))
19 return prefix+String((10-checksum(prefix)%10)%10)
20 }
21 try failIf(!matches(s,"97[89][0-9]{10}"),"invalid isbn")
22 try failIf(checksum(s)%10 != 0,"invalid isbn")
23 return s
24}
25func parseMoney(_ s: String) -> Int { Int(s.replacingOccurrences(of: ".",with: ""))! }
26func money(_ n: Int) -> String {
27 String(format: "%ld.%02ld",locale: Locale(identifier: "en_US_POSIX"),n/100,n%100)
28}
29final class Item {
30 let day: Int
31 let kind: String
32 let isbn: String
33 let amount: Int
34 var unpaid: Int
35 init(_ day: Int, _ kind: String, _ isbn: String, _ amount: Int) {
36 self.day = day
37 self.kind = kind
38 self.isbn = isbn
39 self.amount = amount
40 unpaid = amount
41 }
42}
43final class Member {
44 let id: String
45 let type: String
46 let name: String
47 let limit: Int
48 let period: Int
49 let rate: Int
50 let cap: Int
51 let reservationLimit: Int
52 let fee: Int
53 var queued = 0
54 var loans: [String:Copy] = [:]
55 var holds: [String:Copy] = [:]
56 var items: [Item] = []
57 var history: [String] = []
58 init(_ id: String, _ type: String, _ name: String) {
59 self.id = id
60 self.type = type
61 self.name = name
62 let student = type == "STUDENT"
63 limit = student ? 3 : 10
64 period = student ? 14 : 28
65 rate = student ? 25 : 10
66 cap = student ? 1000 : 2000
67 reservationLimit = student ? 2 : 5
68 fee = student ? 50 : 100
69 }
70 func balance() -> Int { items.reduce(0) { $0+$1.unpaid } }
71 func reservations() -> Int { queued+holds.count }
72 func blocked() -> Bool { balance() > 500 }
73 func status() -> String { blocked() ? "BLOCKED" : "OK" }
74}
75final class Title {
76 let isbn: String
77 let name: String
78 var price: Int
79 var borrowed = 0
80 var copies: [Copy] = []
81 var queue: [Member] = []
82 init(_ isbn: String, _ name: String, _ price: Int) {
83 self.isbn = isbn
84 self.name = name
85 self.price = price
86 }
87 func count(_ state: String) -> Int { copies.filter { $0.state == state }.count }
88 func holdings() -> Int { count("AVAILABLE")+count("LOANED")+count("HOLD") }
89}
90final class Copy {
91 unowned let title: Title
92 let number: Int
93 var state = "AVAILABLE"
94 weak var owner: Member?
95 var loanDay = 0
96 var due = 0
97 var renewals = 0
98 var expiry = 0
99 init(_ title: Title, _ number: Int) {
100 self.title = title
101 self.number = number
102 }
103}
104final class Library {
105 var day = 1
106 var borrowed = 0
107 var returned = 0
108 var renewed = 0
109 var lost = 0
110 var fines = 0
111 var fees = 0
112 var charges = 0
113 var paid = 0
114 var waived = 0
115 var titles: [String:Title] = [:]
116 var members: [String:Member] = [:]
117 var closed = Set<Int>()
118 var copies: [Copy] = []
119 var output: [String] = []
120 func orderedTitles() -> [Title] { titles.keys.sorted().map { titles[$0]! } }
121 func orderedMembers() -> [Member] { members.keys.sorted().map { members[$0]! } }
122 func member(_ id: String) throws -> Member {
123 guard let m = members[id] else { throw Fault.reason("unknown member") }
124 return m
125 }
126 func book(_ raw: String) throws -> Title {
127 let isbn = try normalize(raw)
128 guard let b = titles[isbn] else { throw Fault.reason("unknown book") }
129 return b
130 }
131 func loan(_ m: Member, _ b: Title) throws -> Copy {
132 guard let c = m.loans[b.isbn] else { throw Fault.reason("not borrowed") }
133 return c
134 }
135 func open(_ start: Int) -> Int {
136 var d = start
137 while closed.contains(d) { d += 1 }
138 return d
139 }
140 func late(_ c: Copy) -> Int { max(0,day-c.due) }
141 func fine(_ c: Copy) -> Int { min(late(c)*c.owner!.rate,c.owner!.cap) }
142 func charge(_ m: Member, _ kind: String, _ b: Title, _ amount: Int) {
143 if amount == 0 { return }
144 m.items.append(Item(day,kind,b.isbn,amount))
145 switch kind {
146 case "FINE": fines += amount
147 case "FEE": fees += amount
148 default: charges += amount
149 }
150 }
151 func release(_ c: Copy) {
152 let b = c.title
153 c.owner = nil
154 if b.queue.isEmpty { c.state = "AVAILABLE" }
155 else {
156 let m = b.queue.removeFirst()
157 m.queued -= 1
158 c.state = "HOLD"
159 c.owner = m
160 c.expiry = open(day+3)
161 m.holds[b.isbn] = c
162 output.append("HOLD \(b.isbn) copy \(c.number) \(m.id) until \(c.expiry)")
163 }
164 }
165 func endLoan(_ c: Copy, _ loss: Bool) -> (Int,Int) {
166 let m = c.owner!
167 let l = late(c)
168 let f = fine(c)
169 m.loans.removeValue(forKey:c.title.isbn)
170 charge(m,"FINE",c.title,f)
171 m.history.append((loss ? "LOST " : "RETURNED ")+"\(c.title.isbn) copy \(c.number)"+
172 " from \(c.loanDay) to \(day) late \(l) fine \(money(f))"+(loss ? " price \(money(c.title.price))" : ""))
173 return (l,f)
174 }
175 func loss(_ c: Copy, _ heading: String) {
176 let m = c.owner!
177 let b = c.title
178 let (l,f) = endLoan(c,true)
179 charge(m,"PRICE",b,b.price)
180 c.state = "LOST"
181 c.owner = nil
182 lost += 1
183 output.append("\(heading) \(m.id) \(b.isbn) copy \(c.number) late \(l) fine \(money(f)) price \(money(b.price)) balance \(money(m.balance()))")
184 if b.holdings() == 0 {
185 for queued in b.queue {
186 queued.queued -= 1
187 output.append("DROPPED \(b.isbn) \(queued.id)")
188 }
189 b.queue.removeAll()
190 }
191 }
192 func renewalReason(_ c: Copy) -> String? {
193 if c.due < day { return "loan overdue" }
194 if c.renewals == 2 { return "renewal limit" }
195 if !c.title.queue.isEmpty { return "title reserved" }
196 return nil
197 }
198 func renew(_ c: Copy) {
199 c.due = open(day+c.owner!.period)
200 c.renewals += 1
201 renewed += 1
202 output.append("RENEW \(c.owner!.id) \(c.title.isbn) copy \(c.number) due \(c.due) renewals \(c.renewals)")
203 }
204 func orderedLoans(_ m: Member) -> [Copy] {
205 m.loans.values.sorted {
206 if $0.due != $1.due { return $0.due < $1.due }
207 return $0.title.isbn < $1.title.isbn
208 }
209 }
210 func orderedHolds(_ m: Member) -> [Copy] {
211 m.holds.values.sorted {
212 if $0.expiry != $1.expiry { return $0.expiry < $1.expiry }
213 return $0.title.isbn < $1.title.isbn
214 }
215 }
216 func showMember(_ m: Member) {
217 output.append("MEMBER \(m.id) \(m.type) \(m.name)")
218 output.append("STATUS loans \(m.loans.count)/\(m.limit) reservations \(m.reservations())/\(m.reservationLimit) balance \(money(m.balance())) \(m.status())")
219 for c in orderedLoans(m) {
220 output.append("LOAN \(c.title.isbn) copy \(c.number) due \(c.due) renewals \(c.renewals)"+
221 (c.due < day ? " late \(late(c)) fine \(money(fine(c)))" : ""))
222 }
223 for c in orderedHolds(m) { output.append("HOLD \(c.title.isbn) copy \(c.number) until \(c.expiry)") }
224 for b in orderedTitles() {
225 if let p = b.queue.firstIndex(where:{ $0 === m }) { output.append("QUEUE \(b.isbn) position \(p+1)") }
226 }
227 }
228 func execute(_ line: String) throws {
229 let t = line.split(separator:" ").map(String.init)
230 switch t[0] {
231 case "DAY":
232 let d = Int(t[1])!
233 try failIf(d <= day,"invalid day")
234 output.append("DAY \(d)")
235 while day < d {
236 day += 1
237 let losses = copies.filter { $0.state == "LOANED" && $0.due+365 <= day }.sorted {
238 if $0.owner!.id != $1.owner!.id { return $0.owner!.id < $1.owner!.id }
239 return $0.title.isbn < $1.title.isbn
240 }
241 for c in losses { loss(c,"AUTOLOST") }
242 let expired = copies.filter { $0.state == "HOLD" && $0.expiry < day }.sorted {
243 if $0.title.isbn != $1.title.isbn { return $0.title.isbn < $1.title.isbn }
244 return $0.number < $1.number
245 }
246 for c in expired {
247 let m = c.owner!
248 charge(m,"FEE",c.title,m.fee)
249 m.holds.removeValue(forKey:c.title.isbn)
250 output.append("EXPIRED \(c.title.isbn) copy \(c.number) \(m.id) fee \(money(m.fee))")
251 release(c)
252 }
253 }
254 case "CLOSED":
255 let d = Int(t[1])!
256 try failIf(d <= day,"invalid day")
257 try failIf(closed.contains(d),"already closed")
258 closed.insert(d)
259 output.append("CLOSED \(d)")
260 for b in orderedTitles() {
261 for c in b.copies {
262 if c.state == "LOANED" && c.due == d {
263 c.due = open(d)
264 output.append("MOVED \(b.isbn) copy \(c.number) \(c.owner!.id) due \(c.due)")
265 } else if c.state == "HOLD" && c.expiry == d {
266 c.expiry = open(d)
267 output.append("MOVED \(b.isbn) copy \(c.number) \(c.owner!.id) until \(c.expiry)")
268 }
269 }
270 }
271 case "ADDBOOK":
272 let isbn = try normalize(t[1])
273 let name = t.dropFirst(4).joined(separator:" ")
274 let existing = titles[isbn]
275 try failIf(existing != nil && existing!.name != name,"title mismatch")
276 let count = Int(t[2])!
277 let price = parseMoney(t[3])
278 let b = existing ?? Title(isbn,name,price)
279 titles[isbn] = b
280 b.price = price
281 var added: [Copy] = []
282 for _ in 0..<count {
283 let c = Copy(b,b.copies.count+1)
284 b.copies.append(c)
285 copies.append(c)
286 added.append(c)
287 }
288 output.append("ADDBOOK \(isbn) added \(count) total \(b.holdings())")
289 for c in added { release(c) }
290 case "ADDMEMBER":
291 try failIf(members[t[1]] != nil,"duplicate member")
292 try failIf(t[2] != "STUDENT" && t[2] != "STAFF","unknown type")
293 let m = Member(t[1],t[2],t.dropFirst(3).joined(separator:" "))
294 members[m.id] = m
295 output.append("ADDMEMBER \(m.id) \(m.type) loans \(m.limit) days \(m.period)")
296 case "WITHDRAW":
297 let b = try book(t[1])
298 let n = Int(t[2])!
299 try failIf(n < 1 || n > b.copies.count,"unknown copy")
300 let c = b.copies[n-1]
301 try failIf(c.state != "AVAILABLE","copy not available")
302 c.state = "WITHDRAWN"
303 output.append("WITHDRAW \(b.isbn) copy \(n) total \(b.holdings())")
304 case "BORROW":
305 let m = try member(t[1])
306 let b = try book(t[2])
307 try failIf(closed.contains(day),"library closed")
308 try failIf(m.loans[b.isbn] != nil,"already borrowed")
309 try failIf(m.blocked(),"member blocked")
310 try failIf(m.loans.values.contains { $0.due < day },"overdue loans")
311 try failIf(m.loans.count >= m.limit,"loan limit")
312 guard let c = m.holds[b.isbn] ?? b.copies.first(where:{ $0.state == "AVAILABLE" }) else {
313 throw Fault.reason(b.count("HOLD") > 0 ? "all copies on hold" : "no copy available")
314 }
315 m.holds.removeValue(forKey:b.isbn)
316 c.state = "LOANED"
317 c.owner = m
318 c.loanDay = day
319 c.due = open(day+m.period)
320 c.renewals = 0
321 m.loans[b.isbn] = c
322 b.borrowed += 1
323 borrowed += 1
324 output.append("BORROW \(m.id) \(b.isbn) copy \(c.number) due \(c.due)")
325 case "RETURN":
326 let m = try member(t[1])
327 let b = try book(t[2])
328 let c = try loan(m,b)
329 let (l,f) = endLoan(c,false)
330 returned += 1
331 output.append("RETURN \(m.id) \(b.isbn) copy \(c.number) late \(l) fine \(money(f)) balance \(money(m.balance()))")
332 release(c)
333 case "RENEW":
334 let m = try member(t[1])
335 let c = try loan(m,book(t[2]))
336 try failIf(m.blocked(),"member blocked")
337 if let reason = renewalReason(c) { throw Fault.reason(reason) }
338 renew(c)
339 case "RENEWALL":
340 let m = try member(t[1])
341 try failIf(m.loans.isEmpty,"no loans")
342 try failIf(m.blocked(),"member blocked")
343 var done = 0
344 var skipped = 0
345 for isbn in m.loans.keys.sorted() {
346 let c = m.loans[isbn]!
347 if let reason = renewalReason(c) {
348 output.append("SKIP \(c.title.isbn) \(reason)")
349 skipped += 1
350 } else {
351 renew(c)
352 done += 1
353 }
354 }
355 output.append("RENEWALL \(m.id) renewed \(done) skipped \(skipped)")
356 case "RESERVE":
357 let m = try member(t[1])
358 let b = try book(t[2])
359 try failIf(m.loans[b.isbn] != nil,"already borrowed")
360 try failIf(b.queue.contains { $0 === m } || m.holds[b.isbn] != nil,"already reserved")
361 try failIf(m.blocked(),"member blocked")
362 try failIf(m.reservations() >= m.reservationLimit,"reservation limit")
363 try failIf(b.holdings() == 0,"no copies")
364 try failIf(b.count("AVAILABLE") > 0,"copy available")
365 b.queue.append(m)
366 m.queued += 1
367 let p = b.queue.count
368 let dues = b.copies.filter { $0.state == "LOANED" }.map { $0.due }.sorted()
369 let estimate = p <= dues.count ? String(dues[p-1]) : "none"
370 output.append("RESERVE \(m.id) \(b.isbn) position \(p) estimate \(estimate)")
371 case "CANCEL":
372 let m = try member(t[1])
373 let b = try book(t[2])
374 let p = b.queue.firstIndex { $0 === m }
375 let c = m.holds[b.isbn]
376 try failIf(p == nil && c == nil,"not reserved")
377 if let p {
378 b.queue.remove(at:p)
379 m.queued -= 1
380 output.append("CANCEL \(m.id) \(b.isbn) queue")
381 } else {
382 m.holds.removeValue(forKey:b.isbn)
383 output.append("CANCEL \(m.id) \(b.isbn) hold copy \(c!.number)")
384 release(c!)
385 }
386 case "LOST":
387 let m = try member(t[1])
388 try loss(loan(m,book(t[2])),"LOST")
389 case "PAYFINE":
390 let m = try member(t[1])
391 try failIf(m.balance() == 0,"no fines")
392 let amount = parseMoney(t[2])
393 var rest = amount
394 var details: [String] = []
395 for (i,item) in m.items.enumerated() {
396 if rest == 0 { break }
397 let applied = min(rest,item.unpaid)
398 if applied == 0 { continue }
399 rest -= applied
400 item.unpaid -= applied
401 details.append("ITEM \(i+1) \(item.kind) paid \(money(applied)) remaining \(money(item.unpaid))")
402 }
403 paid += amount-rest
404 output.append("PAYFINE \(m.id) paid \(money(amount-rest)) change \(money(rest)) balance \(money(m.balance()))")
405 output.append(contentsOf:details)
406 case "WAIVE":
407 let m = try member(t[1])
408 let n = Int(t[2])!
409 try failIf(n < 1 || n > m.items.count,"unknown item")
410 let item = m.items[n-1]
411 try failIf(item.unpaid == 0,"item settled")
412 let amount = item.unpaid
413 item.unpaid = 0
414 waived += amount
415 output.append("WAIVE \(m.id) item \(n) \(item.kind) amount \(money(amount)) balance \(money(m.balance()))")
416 case "FINES":
417 let m = try member(t[1])
418 output.append("FINES \(m.id) items \(m.items.filter { $0.unpaid > 0 }.count) balance \(money(m.balance()))")
419 for (i,item) in m.items.enumerated() where item.unpaid > 0 {
420 output.append("\(i+1) day \(item.day) \(item.kind) \(item.isbn) amount \(money(item.amount)) remaining \(money(item.unpaid))")
421 }
422 case "SEARCH":
423 let keyword = t.dropFirst().joined(separator:" ").lowercased()
424 let found = titles.values.filter { $0.name.lowercased().contains(keyword) }.sorted {
425 let a = $0.name.lowercased()
426 let b = $1.name.lowercased()
427 return a == b ? $0.isbn < $1.isbn : a < b
428 }
429 output.append("SEARCH \(found.count)")
430 for b in found { output.append("\(b.isbn) \(b.count("AVAILABLE"))/\(b.holdings()) \(b.name)") }
431 case "OVERDUE":
432 let found = copies.filter { $0.state == "LOANED" && $0.due < day }.sorted {
433 if $0.due != $1.due { return $0.due < $1.due }
434 if $0.owner!.id != $1.owner!.id { return $0.owner!.id < $1.owner!.id }
435 return $0.title.isbn < $1.title.isbn
436 }
437 output.append("OVERDUE \(found.count)")
438 for c in found { output.append("\(c.owner!.id) \(c.title.isbn) copy \(c.number) due \(c.due) late \(late(c)) fine \(money(fine(c)))") }
439 case "NOTICES":
440 var notices: [String] = []
441 for m in orderedMembers() {
442 let loans = orderedLoans(m)
443 for c in loans where c.due < day { notices.append("OVERDUE \(m.id) \(c.title.isbn) due \(c.due)") }
444 for c in loans where c.due >= day && c.due <= day+2 { notices.append("DUESOON \(m.id) \(c.title.isbn) due \(c.due)") }
445 for c in orderedHolds(m) { notices.append("PICKUP \(m.id) \(c.title.isbn) until \(c.expiry)") }
446 }
447 output.append("NOTICES \(notices.count)")
448 output.append(contentsOf:notices)
449 case "MEMBER": try showMember(member(t[1]))
450 case "MEMBERS":
451 output.append("MEMBERS \(members.count)")
452 for m in orderedMembers() {
453 output.append("\(m.id) \(m.type) loans \(m.loans.count) reservations \(m.reservations()) balance \(money(m.balance())) \(m.status())")
454 }
455 case "BOOK":
456 let b = try book(t[1])
457 output.append("BOOK \(b.isbn) \(b.name)")
458 output.append("PRICE \(money(b.price)) copies \(b.holdings()) borrowed \(b.borrowed)")
459 for c in b.copies {
460 let detail: String
461 if c.state == "LOANED" { detail = " \(c.owner!.id) due \(c.due)" }
462 else if c.state == "HOLD" { detail = " \(c.owner!.id) until \(c.expiry)" }
463 else { detail = "" }
464 output.append("COPY \(c.number) \(c.state)\(detail)")
465 }
466 output.append("QUEUE \(b.queue.count)"+(b.queue.isEmpty ? "" : " "+b.queue.map { $0.id }.joined(separator:" ")))
467 case "INVENTORY":
468 let a = copies.filter { $0.state == "AVAILABLE" }.count
469 let l = copies.filter { $0.state == "LOANED" }.count
470 let h = copies.filter { $0.state == "HOLD" }.count
471 let x = copies.filter { $0.state == "LOST" }.count
472 let w = copies.filter { $0.state == "WITHDRAWN" }.count
473 output.append("INVENTORY titles \(titles.count) copies \(a+l+h) available \(a) loaned \(l) hold \(h) lost \(x) withdrawn \(w)")
474 for b in orderedTitles() {
475 output.append("\(b.isbn) copies \(b.holdings()) available \(b.count("AVAILABLE")) loaned \(b.count("LOANED")) hold \(b.count("HOLD")) queue \(b.queue.count)")
476 }
477 case "HISTORY":
478 let m = try member(t[1])
479 output.append("HISTORY \(m.id) \(m.history.count)")
480 output.append(contentsOf:m.history)
481 case "STATS":
482 let active = members.values.reduce(0) { $0+$1.loans.count }
483 let outstanding = members.values.reduce(0) { $0+$1.balance() }
484 output.append("STATS day \(day) titles \(titles.count) members \(members.count) closed \(closed.count)")
485 output.append("LOANS borrowed \(borrowed) returned \(returned) renewed \(renewed) lost \(lost) active \(active)")
486 output.append("MONEY fines \(money(fines)) fees \(money(fees)) charges \(money(charges)) paid \(money(paid)) waived \(money(waived)) outstanding \(money(outstanding))")
487 let top = titles.values.filter { $0.borrowed > 0 }.sorted {
488 $0.borrowed == $1.borrowed ? $0.isbn < $1.isbn : $0.borrowed > $1.borrowed
489 }.prefix(3)
490 for (i,b) in top.enumerated() { output.append("TOP \(i+1) \(b.isbn) \(b.borrowed) \(b.name)") }
491 default: break
492 }
493 }
494 func process(_ line: String) -> [String] {
495 output = []
496 do { try execute(line) }
497 catch Fault.reason(let reason) { output.append("ERROR "+reason) }
498 catch { fatalError("Unexpected error") }
499 return output
500 }
