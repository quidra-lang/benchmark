3class Invalid < StandardError
4end

6class Library
7 RULES = {
8 "STUDENT" => { limit: 3, period: 14, fine: 25, cap: 1000, reservations: 2, fee: 50 },
9 "STAFF" => { limit: 10, period: 28, fine: 10, cap: 2000, reservations: 5, fee: 100 }
10 }.freeze
11
12 def initialize
13 @day = 1
14 @closed = Set.new
15 @titles = {}
16 @members = {}
17 @counts = { borrowed: 0, returned: 0, renewed: 0, lost: 0, FINE: 0, FEE: 0, PRICE: 0, paid: 0, waived: 0 }
18 end
19
20 def fail_with(reason)
21 raise Invalid, reason
22 end
23
24 def amount(text)
25 whole, fraction = text.split(".")
26 whole.to_i * 100 + fraction.to_i
27 end
28
29 def money(value)
30 format("%d.%02d", value.div(100), value % 100)
31 end
32
33 def isbn(text)
34 raw = text.delete("-")
35 if raw.match?(/\A[0-9]{9}[0-9X]\z/)
36 sum = raw.chars.each_with_index.sum { |c, i| (10 - i) * (c == "X" ? 10 : c.to_i) }
37 fail_with("invalid isbn") unless (sum % 11).zero?
38 prefix = "978" + raw[0, 9]
39 checksum = prefix.chars.each_with_index.sum { |c, i| c.to_i * (i.even? ? 1 : 3) }
40 return prefix + ((10 - checksum % 10) % 10).to_s
41 end
42 if raw.match?(/\A(?:978|979)[0-9]{10}\z/)
43 sum = raw.chars.each_with_index.sum { |c, i| c.to_i * (i.even? ? 1 : 3) }
44 return raw if (sum % 10).zero?
45 end
46 fail_with("invalid isbn")
47 end

49 def member(id)
50 @members[id] || fail_with("unknown member")
51 end
52
53 def title(token)
54 @titles[isbn(token)] || fail_with("unknown book")
55 end
56
57 def common(id, token)
58 person = member(id)
59 [person, title(token)]
60 end
61
62 def all_copies
63 @titles.values.flat_map { |book| book[:copies] }
64 end
65
66 def holdings(book)
67 book[:copies].count { |copy| %w[AVAILABLE LOANED HOLD].include?(copy[:state]) }
68 end
69
70 def balance(person)
71 person[:items].sum { |item| item[:unpaid] }
72 end
73
74 def reservations(person)
75 person[:holds].length + @titles.values.count { |book| book[:queue].include?(person[:id]) }
76 end
77
78 def open_day(day)
79 day += 1 while @closed.include?(day)
80 day
81 end
82
83 def late(copy)
84 days = [0, @day - copy[:due]].max
85 rule = RULES[@members[copy[:member]][:type]]
86 [days, [days * rule[:fine], rule[:cap]].min]
87 end
88
89 def charge(person, kind, book, value)
90 return if value.zero?
91 person[:items] << {
92 no: person[:items].length + 1, day: @day, kind: kind, isbn: book[:isbn], amount: value, unpaid: value
93 }
94 @counts[kind.to_sym] += value
95 end
96
97 def release(book, copy)
98 if book[:queue].empty?
99 copy[:state] = "AVAILABLE"
100 copy[:member] = nil
101 return []
102 end
103 id = book[:queue].shift
104 copy[:state] = "HOLD"
105 copy[:member] = id
106 copy[:expiry] = open_day(@day + 3)
107 @members[id][:holds][book[:isbn]] = copy
108 ["HOLD #{book[:isbn]} copy #{copy[:no]} #{id} until #{copy[:expiry]}"]
109 end
110
111 def end_loan(book, copy, result)
112 person = @members[copy[:member]]
113 days, fine = late(copy)
114 person[:loans].delete(book[:isbn])
115 charge(person, "FINE", book, fine)
116 person[:history] << "#{result} #{book[:isbn]} copy #{copy[:no]} from #{copy[:loan_day]} to #{@day} late #{days} fine #{money(fine)}"
117 [days, fine]
118 end
119
120 def loss(book, copy, heading)
121 person = @members[copy[:member]]
122 days, fine = end_loan(book, copy, "LOST")
123 person[:history][-1] += " price #{money(book[:price])}"
124 charge(person, "PRICE", book, book[:price])
125 copy[:state] = "LOST"
126 copy[:member] = nil
127 @counts[:lost] += 1
128 output = ["#{heading} #{person[:id]} #{book[:isbn]} copy #{copy[:no]} late #{days} fine #{money(fine)} " +
129 "price #{money(book[:price])} balance #{money(balance(person))}"]
130 if holdings(book).zero? && !book[:queue].empty?
131 output.concat(book[:queue].map { |id| "DROPPED #{book[:isbn]} #{id}" })
132 book[:queue] = []
133 end
134 output
135 end
136
137 def renewal_reason(book, copy)
138 return "loan overdue" if copy[:due] < @day
139 return "renewal limit" if copy[:renewals] == 2
140 book[:queue].empty? ? nil : "title reserved"
141 end
142
143 def renew(person, book, copy)
144 copy[:due] = open_day(@day + RULES[person[:type]][:period])
145 copy[:renewals] += 1
146 @counts[:renewed] += 1
147 "RENEW #{person[:id]} #{book[:isbn]} copy #{copy[:no]} due #{copy[:due]} renewals #{copy[:renewals]}"
148 end
149
150 def advance(day)
151 fail_with("invalid day") if day <= @day
152 output = ["DAY #{day}"]
153 while @day < day
154 @day += 1
155 losses = all_copies.select { |copy| copy[:state] == "LOANED" && copy[:due] + 365 <= @day }
156 .sort_by { |copy| [copy[:member], copy[:isbn]] }
157 losses.each { |copy| output.concat(loss(@titles[copy[:isbn]], copy, "AUTOLOST")) }
158 expired = all_copies.select { |copy| copy[:state] == "HOLD" && copy[:expiry] < @day }
159 .sort_by { |copy| [copy[:isbn], copy[:no]] }
160 expired.each do |copy|
161 book = @titles[copy[:isbn]]
162 person = @members[copy[:member]]
163 fee = RULES[person[:type]][:fee]
164 charge(person, "FEE", book, fee)
165 person[:holds].delete(copy[:isbn])
166 output << "EXPIRED #{copy[:isbn]} copy #{copy[:no]} #{person[:id]} fee #{money(fee)}"
167 output.concat(release(book, copy))
168 end
169 end
170 output
171 end

173 def command(line)
174 name, *args = line.split(" ")
175 return advance(args[0].to_i) if name == "DAY"
176 if name == "CLOSED"
177 day = args[0].to_i
178 fail_with("invalid day") if day <= @day
179 fail_with("already closed") if @closed.include?(day)
180 @closed << day
181 output = ["CLOSED #{day}"]
182 all_copies.sort_by { |copy| [copy[:isbn], copy[:no]] }.each do |copy|
183 if copy[:state] == "LOANED" && copy[:due] == day
184 copy[:due] = open_day(day)
185 output << "MOVED #{copy[:isbn]} copy #{copy[:no]} #{copy[:member]} due #{copy[:due]}"
186 elsif copy[:state] == "HOLD" && copy[:expiry] == day
187 copy[:expiry] = open_day(day)
188 output << "MOVED #{copy[:isbn]} copy #{copy[:no]} #{copy[:member]} until #{copy[:expiry]}"
189 end
190 end
191 return output
192 end
193 if name == "ADDBOOK"
194 id = isbn(args[0])
195 book_name = args.drop(3).join(" ")
196 book = @titles[id]
197 fail_with("title mismatch") if book && book[:name] != book_name
198 unless book
199 book = { isbn: id, name: book_name, price: amount(args[2]), copies: [], queue: [], borrowed: 0 }
200 @titles[id] = book
201 end
202 book[:price] = amount(args[2])
203 added = Array.new(args[1].to_i) do
204 copy = {
205 isbn: id, no: book[:copies].length + 1, state: "AVAILABLE",
206 member: nil, loan_day: 0, due: 0, renewals: 0, expiry: 0
207 }
208 book[:copies] << copy
209 copy
210 end
211 output = ["ADDBOOK #{id} added #{added.length} total #{holdings(book)}"]
212 added.each { |copy| output.concat(release(book, copy)) }
213 return output
214 end
215 if name == "ADDMEMBER"
216 fail_with("duplicate member") if @members.key?(args[0])
217 fail_with("unknown type") unless RULES.key?(args[1])
218 @members[args[0]] = {
219 id: args[0], type: args[1], name: args.drop(2).join(" "),
220 loans: {}, holds: {}, items: [], history: []
221 }
222 return ["ADDMEMBER #{args[0]} #{args[1]} loans #{RULES[args[1]][:limit]} days #{RULES[args[1]][:period]}"]
223 end
224 if name == "WITHDRAW"
225 book = title(args[0])
226 number = args[1].to_i
227 fail_with("unknown copy") unless number.between?(1, book[:copies].length)
228 copy = book[:copies][number - 1]
229 fail_with("copy not available") unless copy[:state] == "AVAILABLE"
230 copy[:state] = "WITHDRAWN"
231 return ["WITHDRAW #{book[:isbn]} copy #{copy[:no]} total #{holdings(book)}"]
232 end
233 return circulate(name, args) if %w[BORROW RETURN RENEW RESERVE CANCEL LOST].include?(name)
234 if name == "RENEWALL"
235 person = member(args[0])
236 fail_with("no loans") if person[:loans].empty?
237 fail_with("member blocked") if balance(person) > 500
238 output = []
239 renewed = 0
240 skipped = 0
241 person[:loans].values.sort_by { |copy| copy[:isbn] }.each do |copy|
242 book = @titles[copy[:isbn]]
243 reason = renewal_reason(book, copy)
244 if reason
245 output << "SKIP #{copy[:isbn]} #{reason}"
246 skipped += 1
247 else
248 output << renew(person, book, copy)
249 renewed += 1
250 end
251 end
252 output << "RENEWALL #{person[:id]} renewed #{renewed} skipped #{skipped}"
253 return output
254 end
255 return charges(name, args) if %w[PAYFINE WAIVE FINES].include?(name)
256 inspect_library(name, args)
257 end

259 def circulate(name, args)
260 person, book = common(args[0], args[1])
261 loan = person[:loans][book[:isbn]]
262 hold = person[:holds][book[:isbn]]
263 rule = RULES[person[:type]]
264 if name == "BORROW"
265 fail_with("library closed") if @closed.include?(@day)
266 fail_with("already borrowed") if loan
267 fail_with("member blocked") if balance(person) > 500
268 fail_with("overdue loans") if person[:loans].values.any? { |copy| copy[:due] < @day }
269 fail_with("loan limit") if person[:loans].length >= rule[:limit]
270 copy = hold || book[:copies].find { |item| item[:state] == "AVAILABLE" }
271 unless copy
272 fail_with(book[:copies].any? { |item| item[:state] == "HOLD" } ? "all copies on hold" : "no copy available")
273 end
274 person[:holds].delete(book[:isbn])
275 copy.merge!(state: "LOANED", member: person[:id], loan_day: @day,
276 due: open_day(@day + rule[:period]), renewals: 0)
277 person[:loans][book[:isbn]] = copy
278 book[:borrowed] += 1
279 @counts[:borrowed] += 1
280 return ["BORROW #{person[:id]} #{book[:isbn]} copy #{copy[:no]} due #{copy[:due]}"]
281 end
282 if %w[RETURN LOST RENEW].include?(name)
283 fail_with("not borrowed") unless loan
284 return loss(book, loan, "LOST") if name == "LOST"
285 if name == "RETURN"
286 days, fine = end_loan(book, loan, "RETURNED")
287 @counts[:returned] += 1
288 output = ["RETURN #{person[:id]} #{book[:isbn]} copy #{loan[:no]} late #{days} fine #{money(fine)} balance #{money(balance(person))}"]
289 return output + release(book, loan)
290 end
291 fail_with("member blocked") if balance(person) > 500
292 reason = renewal_reason(book, loan)
293 fail_with(reason) if reason
294 return [renew(person, book, loan)]
295 end
296 queued = book[:queue].include?(person[:id])
297 if name == "CANCEL"
298 fail_with("not reserved") unless queued || hold
299 if queued
300 book[:queue].delete(person[:id])
301 return ["CANCEL #{person[:id]} #{book[:isbn]} queue"]
302 end
303 person[:holds].delete(book[:isbn])
304 return ["CANCEL #{person[:id]} #{book[:isbn]} hold copy #{hold[:no]}"] + release(book, hold)
305 end
306 fail_with("already borrowed") if loan
307 fail_with("already reserved") if queued || hold
308 fail_with("member blocked") if balance(person) > 500
309 fail_with("reservation limit") if reservations(person) >= rule[:reservations]
310 fail_with("no copies") if holdings(book).zero?
311 fail_with("copy available") if book[:copies].any? { |copy| copy[:state] == "AVAILABLE" }
312 book[:queue] << person[:id]
313 days = book[:copies].select { |copy| copy[:state] == "LOANED" }.map { |copy| copy[:due] }.sort
314 position = book[:queue].length
315 ["RESERVE #{person[:id]} #{book[:isbn]} position #{position} estimate #{days[position - 1] || 'none'}"]
316 end

318 def charges(name, args)
319 person = member(args[0])
320 if name == "PAYFINE"
321 fail_with("no fines") if balance(person).zero?
322 remaining = amount(args[1])
323 items = []
324 person[:items].each do |item|
325 applied = [remaining, item[:unpaid]].min
326 if applied.positive?
327 item[:unpaid] -= applied
328 remaining -= applied
329 items << "ITEM #{item[:no]} #{item[:kind]} paid #{money(applied)} remaining #{money(item[:unpaid])}"
330 end
331 break if remaining.zero?
332 end
333 paid = amount(args[1]) - remaining
334 @counts[:paid] += paid
335 return ["PAYFINE #{person[:id]} paid #{money(paid)} change #{money(remaining)} balance #{money(balance(person))}", *items]
336 end
337 if name == "WAIVE"
338 no = args[1].to_i
339 fail_with("unknown item") unless no.between?(1, person[:items].length)
340 item = person[:items][no - 1]
341 fail_with("item settled") if item[:unpaid].zero?
342 value = item[:unpaid]
343 item[:unpaid] = 0
344 @counts[:waived] += value
345 return ["WAIVE #{person[:id]} item #{no} #{item[:kind]} amount #{money(value)} balance #{money(balance(person))}"]
346 end
347 items = person[:items].select { |item| item[:unpaid].positive? }
348 output = ["FINES #{person[:id]} items #{items.length} balance #{money(balance(person))}"]
349 lines = items.map do |item|
350 "#{item[:no]} day #{item[:day]} #{item[:kind]} #{item[:isbn]} amount #{money(item[:amount])} remaining #{money(item[:unpaid])}"
351 end
352 output + lines
353 end

355 def inspect_library(name, args)
356 if name == "SEARCH"
357 keyword = args.join(" ").downcase
358 titles = @titles.values.select { |book| book[:name].downcase.include?(keyword) }
359 .sort_by { |book| [book[:name].downcase, book[:isbn]] }
360 lines = titles.map do |book|
361 available = book[:copies].count { |copy| copy[:state] == "AVAILABLE" }
362 "#{book[:isbn]} #{available}/#{holdings(book)} #{book[:name]}"
363 end
364 return ["SEARCH #{titles.length}"] + lines
365 end
366 if name == "OVERDUE"
367 copies = all_copies.select { |copy| copy[:state] == "LOANED" && copy[:due] < @day }
368 .sort_by { |copy| [copy[:due], copy[:member], copy[:isbn]] }
369 lines = copies.map do |copy|
370 days, fine = late(copy)
371 "#{copy[:member]} #{copy[:isbn]} copy #{copy[:no]} due #{copy[:due]} late #{days} fine #{money(fine)}"
372 end
373 return ["OVERDUE #{copies.length}"] + lines
374 end
375 if name == "NOTICES"
376 output = []
377 @members.keys.sort.each do |id|
378 person = @members[id]
379 loans = person[:loans].values.sort_by { |copy| [copy[:due], copy[:isbn]] }
380 loans.select { |copy| copy[:due] < @day }.each do |copy|
381 output << "OVERDUE #{id} #{copy[:isbn]} due #{copy[:due]}"
382 end
383 loans.select { |copy| copy[:due].between?(@day, @day + 2) }.each do |copy|
384 output << "DUESOON #{id} #{copy[:isbn]} due #{copy[:due]}"
385 end
386 person[:holds].values.sort_by { |copy| [copy[:expiry], copy[:isbn]] }.each do |copy|
387 output << "PICKUP #{id} #{copy[:isbn]} until #{copy[:expiry]}"
388 end
389 end
390 return ["NOTICES #{output.length}", *output]
391 end
392 if name == "HISTORY"
393 person = member(args[0])
394 return ["HISTORY #{person[:id]} #{person[:history].length}", *person[:history]]
395 end
396 if name == "MEMBER" || name == "MEMBERS"
397 members = name == "MEMBER" ? [member(args[0])] : @members.values.sort_by { |person| person[:id] }
398 output = name == "MEMBERS" ? ["MEMBERS #{members.length}"] : []
399 members.each do |person|
400 value = balance(person)
401 state = value > 500 ? "BLOCKED" : "OK"
402 if name == "MEMBERS"
403 output << "#{person[:id]} #{person[:type]} loans #{person[:loans].length} reservations #{reservations(person)} balance #{money(value)} #{state}"
404 next
405 end
406 rule = RULES[person[:type]]
407 output << "MEMBER #{person[:id]} #{person[:type]} #{person[:name]}"
408 output << "STATUS loans #{person[:loans].length}/#{rule[:limit]} reservations #{reservations(person)}/#{rule[:reservations]} balance #{money(value)} #{state}"
409 person[:loans].values.sort_by { |copy| [copy[:due], copy[:isbn]] }.each do |copy|
410 days, fine = late(copy)
411 suffix = days.positive? ? " late #{days} fine #{money(fine)}" : ""
412 output << "LOAN #{copy[:isbn]} copy #{copy[:no]} due #{copy[:due]} renewals #{copy[:renewals]}" + suffix
413 end
414 person[:holds].values.sort_by { |copy| [copy[:expiry], copy[:isbn]] }.each do |copy|
415 output << "HOLD #{copy[:isbn]} copy #{copy[:no]} until #{copy[:expiry]}"
416 end
417 @titles.values.sort_by { |book| book[:isbn] }.each do |book|
418 index = book[:queue].index(person[:id])
419 output << "QUEUE #{book[:isbn]} position #{index + 1}" if index
420 end
421 end
422 return output
423 end
424 if name == "BOOK"
425 book = title(args[0])
426 output = ["BOOK #{book[:isbn]} #{book[:name]}", "PRICE #{money(book[:price])} copies #{holdings(book)} borrowed #{book[:borrowed]}"]
427 book[:copies].each do |copy|
428 line = "COPY #{copy[:no]} #{copy[:state]}"
429 line += " #{copy[:member]} due #{copy[:due]}" if copy[:state] == "LOANED"
430 line += " #{copy[:member]} until #{copy[:expiry]}" if copy[:state] == "HOLD"
431 output << line
432 end
433 output << "QUEUE #{book[:queue].length}" + (book[:queue].empty? ? "" : " " + book[:queue].join(" "))
434 return output
435 end
436 if name == "INVENTORY"
437 states = all_copies.map { |copy| copy[:state] }.tally(Hash.new(0))
438 total = states["AVAILABLE"] + states["LOANED"] + states["HOLD"]
439 output = ["INVENTORY titles #{@titles.length} copies #{total} available #{states['AVAILABLE']} " +
440 "loaned #{states['LOANED']} hold #{states['HOLD']} lost #{states['LOST']} withdrawn #{states['WITHDRAWN']}"]
441 @titles.values.sort_by { |book| book[:isbn] }.each do |book|
442 counts = book[:copies].map { |copy| copy[:state] }.tally
443 output << "#{book[:isbn]} copies #{holdings(book)} available #{counts.fetch('AVAILABLE', 0)} " +
444 "loaned #{counts.fetch('LOANED', 0)} hold #{counts.fetch('HOLD', 0)} queue #{book[:queue].length}"
445 end
446 return output
447 end
448 c = @counts
449 outstanding = @members.values.sum { |person| balance(person) }
450 active = all_copies.count { |copy| copy[:state] == "LOANED" }
451 output = [
452 "STATS day #{@day} titles #{@titles.length} members #{@members.length} closed #{@closed.length}",
453 "LOANS borrowed #{c[:borrowed]} returned #{c[:returned]} renewed #{c[:renewed]} lost #{c[:lost]} active #{active}",
454 "MONEY fines #{money(c[:FINE])} fees #{money(c[:FEE])} charges #{money(c[:PRICE])} " +
455 "paid #{money(c[:paid])} waived #{money(c[:waived])} outstanding #{money(outstanding)}"
456 ]
457 top = @titles.values.select { |book| book[:borrowed].positive? }.sort_by { |book| [-book[:borrowed], book[:isbn]] }.take(3)
458 top.each_with_index { |book, i| output << "TOP #{i + 1} #{book[:isbn]} #{book[:borrowed]} #{book[:name]}" }
459 output
460 end
461end
462
463lines = STDIN.read.split("\n")
464library = Library.new
465output = []
466lines[0].to_i.times do |i|
467 begin
468 output.concat(library.command(lines[i + 1]))
469 rescue Invalid => error
470 output << "ERROR " + error.message
471 end
472end
473STDOUT.write(output.join("\n") + "\n")
