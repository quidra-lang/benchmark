1using Printf
2const RULES = Dict("STUDENT"=>(3,14,25,1000,2,50),"STAFF"=>(10,28,10,2000,5,100))
3Base.@kwdef mutable struct BookCopy
4 isbn::String
5 number::Int
6 state::String = "AVAILABLE"
7 member::String = ""
8 start::Int = 0
9 due::Int = 0
10 renewals::Int = 0
11 expiry::Int = 0
12end
13Base.@kwdef mutable struct Title
14 id::String
15 name::String
16 price::Int
17 copies::Vector{BookCopy} = BookCopy[]
18 queue::Vector{String} = String[]
19 borrowed::Int = 0
20end
21mutable struct ChargeItem
22 day::Int
23 kind::String
24 isbn::String
25 amount::Int
26 unpaid::Int
27end
28Base.@kwdef mutable struct LibMember
29 id::String
30 kind::String
31 name::String
32 loans::Dict{String,BookCopy} = Dict{String,BookCopy}()
33 holds::Dict{String,BookCopy} = Dict{String,BookCopy}()
34 items::Vector{ChargeItem} = ChargeItem[]
35 history::Vector{String} = String[]
36end
37const CURRENT_DAY = Ref(1)
38const CLOSED_DAYS = Set{Int}()
39const BOOKS = Dict{String,Title}()
40const BOOK_ORDER = String[]
41const MEMBERS = Dict{String,LibMember}()
42const MEMBER_ORDER = String[]
43const COUNTS = Dict(key=>0 for key in ("borrowed","returned","renewed","lost","FINE","FEE","PRICE","paid","waived"))
44fail(message) = throw(ArgumentError(message))
45function cents(text)
46 whole,fraction = split(text,'.')
47 return 100parse(Int,whole)+parse(Int,fraction)
48end
49money(value) = @sprintf("%d.%02d",div(value,100),mod(value,100))
50function isbn(text)
51 s = replace(text,"-"=>"")
52 if occursin(r"^[0-9]{9}[0-9X]$",s)
53 total = sum((11-i)*(s[i] == 'X' ? 10 : Int(s[i]-'0')) for i in 1:10)
54 mod(total,11) == 0 || fail("invalid isbn")
55 s = "978"*s[1:9]
56 total = sum(Int(s[i]-'0')*(isodd(i) ? 1 : 3) for i in 1:12)
57 return s*string(mod(10-mod(total,10),10))
58 elseif occursin(r"^97[89][0-9]{10}$",s)
59 total = sum(Int(s[i]-'0')*(isodd(i) ? 1 : 3) for i in 1:13)
60 mod(total,10) == 0 && return s
61 end
62 fail("invalid isbn")
63end
64function get_member(id)
65 haskey(MEMBERS,id) || fail("unknown member")
66 return MEMBERS[id]
67end
68function get_book(text)
69 id = isbn(text)
70 haskey(BOOKS,id) || fail("unknown book")
71 return BOOKS[id]
72end
73balance(m) = sum((item.unpaid for item in m.items);init=0)
74holdings(b) = count(c -> c.state in ("AVAILABLE","LOANED","HOLD"),b.copies)
75reservations(m) = length(m.holds)+count(b -> m.id in b.queue,values(BOOKS))
76function open_day(date)
77 while date in CLOSED_DAYS
78 date += 1
79 end
80 return date
81end
82function late_fine(c)
83 m = MEMBERS[c.member]
84 late = max(0,CURRENT_DAY[]-c.due)
85 return late,min(late*RULES[m.kind][3],RULES[m.kind][4])
86end
87function charge!(m,kind,id,amount)
88 amount == 0 && return
89 push!(m.items,ChargeItem(CURRENT_DAY[],kind,id,amount,amount))
90 COUNTS[kind] += amount
91end
92function release_copy!(b,c,out)
93 if isempty(b.queue)
94 c.state = "AVAILABLE"
95 return
96 end
97 id = popfirst!(b.queue)
98 c.state = "HOLD"
99 c.member = id
100 c.expiry = open_day(CURRENT_DAY[]+3)
101 MEMBERS[id].holds[b.id] = c
102 push!(out,"HOLD $(b.id) copy $(c.number) $id until $(c.expiry)")
103end
104function end_loan!(b,c,result)
105 m = MEMBERS[c.member]
106 late,fine = late_fine(c)
107 delete!(m.loans,b.id)
108 charge!(m,"FINE",b.id,fine)
109 push!(m.history,"$result $(b.id) copy $(c.number) from $(c.start) to $(CURRENT_DAY[]) late $late fine $(money(fine))")
110 return m,late,fine
111end
112function lose_copy!(b,c,heading,out)
113 m,late,fine = end_loan!(b,c,"LOST")
114 m.history[end] *= " price $(money(b.price))"
115 charge!(m,"PRICE",b.id,b.price)
116 c.state = "LOST"
117 COUNTS["lost"] += 1
118 push!(out,"$heading $(m.id) $(b.id) copy $(c.number) late $late fine $(money(fine)) price $(money(b.price)) balance $(money(balance(m)))")
119 if holdings(b) == 0
120 for id in b.queue
121 push!(out,"DROPPED $(b.id) $id")
122 end
123 empty!(b.queue)
124 end
125end
126function renewal_reason(b,c)
127 c.due < CURRENT_DAY[] && return "loan overdue"
128 c.renewals == 2 && return "renewal limit"
129 return isempty(b.queue) ? "" : "title reserved"
130end
131function renew_loan!(m,b,c)
132 c.due = open_day(CURRENT_DAY[]+RULES[m.kind][2])
133 c.renewals += 1
134 COUNTS["renewed"] += 1
135 return "RENEW $(m.id) $(b.id) copy $(c.number) due $(c.due) renewals $(c.renewals)"
136end
137sorted_copies(copies,date) = sort(collect(values(copies));by=c -> (getproperty(c,date),c.isbn))
138function execute(line)
139 t = String.(split(line,' '))
140 cmd = t[1]
141 out = String[]
142 if cmd == "DAY"
143 target = parse(Int,t[2])
144 target > CURRENT_DAY[] || fail("invalid day")
145 push!(out,"DAY $target")
146 while CURRENT_DAY[] < target
147 nextday = target+1
148 for m in values(MEMBERS)
149 for c in values(m.loans)
150 nextday = min(nextday,c.due+365)
151 end
152 for c in values(m.holds)
153 nextday = min(nextday,c.expiry+1)
154 end
155 end
156 if nextday > target
157 CURRENT_DAY[] = target
158 break
159 end
160 CURRENT_DAY[] = nextday
161 losses = BookCopy[]
162 for id in MEMBER_ORDER
163 m = MEMBERS[id]
164 for key in sort(collect(keys(m.loans)))
165 c = m.loans[key]
166 c.due+365 <= CURRENT_DAY[] && push!(losses,c)
167 end
168 end
169 for c in losses
170 lose_copy!(BOOKS[c.isbn],c,"AUTOLOST",out)
171 end
172 expired = BookCopy[]
173 for id in BOOK_ORDER
174 for c in BOOKS[id].copies
175 c.state == "HOLD" && c.expiry < CURRENT_DAY[] && push!(expired,c)
176 end
177 end
178 for c in expired
179 b,m = BOOKS[c.isbn],MEMBERS[c.member]
180 fee = RULES[m.kind][6]
181 charge!(m,"FEE",b.id,fee)
182 delete!(m.holds,b.id)
183 push!(out,"EXPIRED $(b.id) copy $(c.number) $(m.id) fee $(money(fee))")
184 release_copy!(b,c,out)
185 end
186 end
187 elseif cmd == "CLOSED"
188 date = parse(Int,t[2])
189 date > CURRENT_DAY[] || fail("invalid day")
190 date in CLOSED_DAYS && fail("already closed")
191 push!(CLOSED_DAYS,date)
192 push!(out,"CLOSED $date")
193 for id in BOOK_ORDER
194 for c in BOOKS[id].copies
195 if c.state == "LOANED" && c.due == date
196 c.due = open_day(date)
197 push!(out,"MOVED $id copy $(c.number) $(c.member) due $(c.due)")
198 elseif c.state == "HOLD" && c.expiry == date
199 c.expiry = open_day(date)
200 push!(out,"MOVED $id copy $(c.number) $(c.member) until $(c.expiry)")
201 end
202 end
203 end
204 elseif cmd == "ADDBOOK"
205 id = isbn(t[2])
206 name = join(t[5:end]," ")
207 haskey(BOOKS,id) && BOOKS[id].name != name && fail("title mismatch")
208 if !haskey(BOOKS,id)
209 BOOKS[id] = Title(id=id,name=name,price=0)
210 push!(BOOK_ORDER,id)
211 sort!(BOOK_ORDER)
212 end
213 b = BOOKS[id]
214 b.price = cents(t[4])
215 added = BookCopy[]
216 for i in 1:parse(Int,t[3])
217 c = BookCopy(isbn=id,number=length(b.copies)+1)
218 push!(b.copies,c)
219 push!(added,c)
220 end
221 push!(out,"ADDBOOK $id added $(t[3]) total $(holdings(b))")
222 for c in added
223 release_copy!(b,c,out)
224 end
225 elseif cmd == "ADDMEMBER"
226 haskey(MEMBERS,t[2]) && fail("duplicate member")
227 haskey(RULES,t[3]) || fail("unknown type")
228 m = LibMember(id=t[2],kind=t[3],name=join(t[4:end]," "))
229 MEMBERS[m.id] = m
230 push!(MEMBER_ORDER,m.id)
231 sort!(MEMBER_ORDER)
232 rule = RULES[m.kind]
233 push!(out,"ADDMEMBER $(m.id) $(m.kind) loans $(rule[1]) days $(rule[2])")
234 elseif cmd in ("BORROW","RETURN","RENEW","RESERVE","CANCEL","LOST")
235 m = get_member(t[2])
236 b = get_book(t[3])
237 id = b.id
238 loan,hold = get(m.loans,id,nothing),get(m.holds,id,nothing)
239 rule = RULES[m.kind]
240 if cmd == "BORROW"
241 CURRENT_DAY[] in CLOSED_DAYS && fail("library closed")
242 loan !== nothing && fail("already borrowed")
243 balance(m) > 500 && fail("member blocked")
244 any(c -> c.due < CURRENT_DAY[],values(m.loans)) && fail("overdue loans")
245 length(m.loans) < rule[1] || fail("loan limit")
246 chosen = hold
247 if chosen === nothing
248 index = findfirst(c -> c.state == "AVAILABLE",b.copies)
249 index !== nothing && (chosen = b.copies[index])
250 end
251 if chosen === nothing
252 any(c -> c.state == "HOLD",b.copies) && fail("all copies on hold")
253 fail("no copy available")
254 end
255 delete!(m.holds,id)
256 chosen.state,chosen.member = "LOANED",m.id
257 chosen.start = CURRENT_DAY[]
258 chosen.due = open_day(CURRENT_DAY[]+rule[2])
259 chosen.renewals = 0
260 m.loans[id] = chosen
261 b.borrowed += 1
262 COUNTS["borrowed"] += 1
263 push!(out,"BORROW $(m.id) $id copy $(chosen.number) due $(chosen.due)")
264 elseif cmd in ("RETURN","RENEW","LOST")
265 loan !== nothing || fail("not borrowed")
266 if cmd == "RETURN"
267 member,late,fine = end_loan!(b,loan,"RETURNED")
268 COUNTS["returned"] += 1
269 push!(out,"RETURN $(m.id) $id copy $(loan.number) late $late fine $(money(fine)) balance $(money(balance(m)))")
270 release_copy!(b,loan,out)
271 elseif cmd == "LOST"
272 lose_copy!(b,loan,"LOST",out)
273 else
274 balance(m) > 500 && fail("member blocked")
275 reason = renewal_reason(b,loan)
276 isempty(reason) || fail(reason)
277 push!(out,renew_loan!(m,b,loan))
278 end
279 elseif cmd == "RESERVE"
280 loan !== nothing && fail("already borrowed")
281 (hold !== nothing || m.id in b.queue) && fail("already reserved")
282 balance(m) > 500 && fail("member blocked")
283 reservations(m) < rule[5] || fail("reservation limit")
284 holdings(b) > 0 || fail("no copies")
285 any(c -> c.state == "AVAILABLE",b.copies) && fail("copy available")
286 push!(b.queue,m.id)
287 position = length(b.queue)
288 dues = sort([c.due for c in b.copies if c.state == "LOANED"])
289 estimate = position <= length(dues) ? string(dues[position]) : "none"
290 push!(out,"RESERVE $(m.id) $id position $position estimate $estimate")
291 else
292 position = findfirst(==(m.id),b.queue)
293 position !== nothing || hold !== nothing || fail("not reserved")
294 if position !== nothing
295 deleteat!(b.queue,position)
296 push!(out,"CANCEL $(m.id) $id queue")
297 else
298 delete!(m.holds,id)
299 push!(out,"CANCEL $(m.id) $id hold copy $(hold.number)")
300 release_copy!(b,hold,out)
301 end
302 end
303 elseif cmd in ("RENEWALL","PAYFINE","WAIVE","FINES","MEMBER","HISTORY")
304 m = get_member(t[2])
305 if cmd == "RENEWALL"
306 isempty(m.loans) && fail("no loans")
307 balance(m) > 500 && fail("member blocked")
308 renewed,skipped = 0,0
309 for id in sort(collect(keys(m.loans)))
310 reason = renewal_reason(BOOKS[id],m.loans[id])
311 if isempty(reason)
312 push!(out,renew_loan!(m,BOOKS[id],m.loans[id]))
313 renewed += 1
314 else
315 push!(out,"SKIP $id $reason")
316 skipped += 1
317 end
318 end
319 push!(out,"RENEWALL $(m.id) renewed $renewed skipped $skipped")
320 elseif cmd == "PAYFINE"
321 balance(m) > 0 || fail("no fines")
322 remaining,paid = cents(t[3]),0
323 lines = String[]
324 for (index,item) in enumerate(m.items)
325 applied = min(remaining,item.unpaid)
326 if applied > 0
327 item.unpaid -= applied
328 remaining -= applied
329 paid += applied
330 push!(lines,"ITEM $index $(item.kind) paid $(money(applied)) remaining $(money(item.unpaid))")
331 end
332 end
333 COUNTS["paid"] += paid
334 push!(out,"PAYFINE $(m.id) paid $(money(paid)) change $(money(remaining)) balance $(money(balance(m)))")
335 append!(out,lines)
336 elseif cmd == "WAIVE"
337 index = parse(Int,t[3])
338 1 <= index <= length(m.items) || fail("unknown item")
339 item = m.items[index]
340 item.unpaid > 0 || fail("item settled")
341 amount = item.unpaid
342 item.unpaid = 0
343 COUNTS["waived"] += amount
344 push!(out,"WAIVE $(m.id) item $index $(item.kind) amount $(money(amount)) balance $(money(balance(m)))")
345 elseif cmd == "FINES"
346 n = count(item -> item.unpaid > 0,m.items)
347 push!(out,"FINES $(m.id) items $n balance $(money(balance(m)))")
348 for (index,item) in enumerate(m.items)
349 item.unpaid > 0 && push!(out,"$index day $(item.day) $(item.kind) $(item.isbn) amount $(money(item.amount)) remaining $(money(item.unpaid))")
350 end
351 elseif cmd == "HISTORY"
352 push!(out,"HISTORY $(m.id) $(length(m.history))")
353 append!(out,m.history)
354 else
355 rule = RULES[m.kind]
356 state = balance(m) > 500 ? "BLOCKED" : "OK"
357 push!(out,"MEMBER $(m.id) $(m.kind) $(m.name)")
358 push!(out,"STATUS loans $(length(m.loans))/$(rule[1]) reservations $(reservations(m))/$(rule[5]) balance $(money(balance(m))) $state")
359 for c in sorted_copies(m.loans,:due)
360 text = "LOAN $(c.isbn) copy $(c.number) due $(c.due) renewals $(c.renewals)"
361 if c.due < CURRENT_DAY[]
362 late,fine = late_fine(c)
363 text *= " late $late fine $(money(fine))"
364 end
365 push!(out,text)
366 end
367 for c in sorted_copies(m.holds,:expiry)
368 push!(out,"HOLD $(c.isbn) copy $(c.number) until $(c.expiry)")
369 end
370 for id in BOOK_ORDER
371 position = findfirst(==(m.id),BOOKS[id].queue)
372 position !== nothing && push!(out,"QUEUE $id position $position")
373 end
374 end
375 elseif cmd in ("WITHDRAW","BOOK")
376 b = get_book(t[2])
377 if cmd == "WITHDRAW"
378 index = parse(Int,t[3])
379 1 <= index <= length(b.copies) || fail("unknown copy")
380 c = b.copies[index]
381 c.state == "AVAILABLE" || fail("copy not available")
382 c.state = "WITHDRAWN"
383 push!(out,"WITHDRAW $(b.id) copy $(c.number) total $(holdings(b))")
384 else
385 push!(out,"BOOK $(b.id) $(b.name)")
386 push!(out,"PRICE $(money(b.price)) copies $(holdings(b)) borrowed $(b.borrowed)")
387 for c in b.copies
388 text = "COPY $(c.number) $(c.state)"
389 if c.state == "LOANED"
390 text *= " $(c.member) due $(c.due)"
391 elseif c.state == "HOLD"
392 text *= " $(c.member) until $(c.expiry)"
393 end
394 push!(out,text)
395 end
396 push!(out,join(vcat(["QUEUE",string(length(b.queue))],b.queue)," "))
397 end
398 elseif cmd == "SEARCH"
399 keyword = lowercase(join(t[2:end]," "))
400 found = filter(b -> occursin(keyword,lowercase(b.name)),collect(values(BOOKS)))
401 sort!(found;by=b -> (lowercase(b.name),b.id))
402 push!(out,"SEARCH $(length(found))")
403 for b in found
404 available = count(c -> c.state == "AVAILABLE",b.copies)
405 push!(out,"$(b.id) $available/$(holdings(b)) $(b.name)")
406 end
407 elseif cmd in ("OVERDUE","NOTICES")
408 lines = String[]
409 if cmd == "OVERDUE"
410 loans = [c for m in values(MEMBERS) for c in values(m.loans) if c.due < CURRENT_DAY[]]
411 sort!(loans;by=c -> (c.due,c.member,c.isbn))
412 for c in loans
413 late,fine = late_fine(c)
414 push!(lines,"$(c.member) $(c.isbn) copy $(c.number) due $(c.due) late $late fine $(money(fine))")
415 end
416 else
417 for id in MEMBER_ORDER
418 m = MEMBERS[id]
419 loans = sorted_copies(m.loans,:due)
420 for c in loans
421 c.due < CURRENT_DAY[] && push!(lines,"OVERDUE $id $(c.isbn) due $(c.due)")
422 end
423 for c in loans
424 CURRENT_DAY[] <= c.due <= CURRENT_DAY[]+2 && push!(lines,"DUESOON $id $(c.isbn) due $(c.due)")
425 end
426 for c in sorted_copies(m.holds,:expiry)
427 push!(lines,"PICKUP $id $(c.isbn) until $(c.expiry)")
428 end
429 end
430 end
431 push!(out,"$cmd $(length(lines))")
432 append!(out,lines)
433 elseif cmd == "MEMBERS"
434 push!(out,"MEMBERS $(length(MEMBERS))")
435 for id in MEMBER_ORDER
436 m = MEMBERS[id]
437 state = balance(m) > 500 ? "BLOCKED" : "OK"
438 push!(out,"$id $(m.kind) loans $(length(m.loans)) reservations $(reservations(m)) balance $(money(balance(m))) $state")
439 end
440 elseif cmd == "INVENTORY"
441 totals = Dict(state=>0 for state in ("AVAILABLE","LOANED","HOLD","LOST","WITHDRAWN"))
442 lines = String[]
443 for id in BOOK_ORDER
444 b = BOOKS[id]
445 n = Dict(state=>0 for state in keys(totals))
446 for c in b.copies
447 n[c.state] += 1
448 end
449 for (state,total) in n
450 totals[state] += total
451 end
452 push!(lines,"$id copies $(holdings(b)) available $(n["AVAILABLE"]) loaned $(n["LOANED"]) hold $(n["HOLD"]) queue $(length(b.queue))")
453 end
454 holding = totals["AVAILABLE"]+totals["LOANED"]+totals["HOLD"]
455 push!(out,"INVENTORY titles $(length(BOOKS)) copies $holding available $(totals["AVAILABLE"]) loaned $(totals["LOANED"]) hold $(totals["HOLD"]) lost $(totals["LOST"]) withdrawn $(totals["WITHDRAWN"])")
456 append!(out,lines)
457 elseif cmd == "STATS"
458 active = sum((length(m.loans) for m in values(MEMBERS));init=0)
459 outstanding = sum((balance(m) for m in values(MEMBERS));init=0)
460 push!(out,"STATS day $(CURRENT_DAY[]) titles $(length(BOOKS)) members $(length(MEMBERS)) closed $(length(CLOSED_DAYS))")
461 push!(out,"LOANS borrowed $(COUNTS["borrowed"]) returned $(COUNTS["returned"]) renewed $(COUNTS["renewed"]) lost $(COUNTS["lost"]) active $active")
462 push!(out,"MONEY fines $(money(COUNTS["FINE"])) fees $(money(COUNTS["FEE"])) charges $(money(COUNTS["PRICE"])) paid $(money(COUNTS["paid"])) waived $(money(COUNTS["waived"])) outstanding $(money(outstanding))")
463 top = sort(filter(b -> b.borrowed > 0,collect(values(BOOKS)));by=b -> (-b.borrowed,b.id))
464 for (rank,b) in enumerate(top[1:min(3,length(top))])
465 push!(out,"TOP $rank $(b.id) $(b.borrowed) $(b.name)")
466 end
467 end
468 return out
469end
470function main()
471 for i in 1:parse(Int,readline())
472 line = readline()
473 try
474 foreach(println,execute(line))
475 catch exception
476 exception isa ArgumentError || rethrow()
477 println("ERROR ",exception.msg)
478 end
479 end
480end
481main()
