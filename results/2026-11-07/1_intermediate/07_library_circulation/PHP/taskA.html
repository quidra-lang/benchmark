10function fail(string $message): never { throw new InvalidArgumentException($message); }
11function cents(string $text): int
12{
13 [$whole,$fraction] = explode('.', $text);

16function money(int $v): string { return intdiv($v,100) . '.' . sprintf('%02d',$v%100); }
17function isbn(string $text): string
18{
19 $s = str_replace('-','',$text);
20 if (preg_match('/^[0-9]{9}[0-9X]$/D',$s)) {
21 $sum = 0;
22 for ($i=0; $i<10; ++$i) $sum += (10-$i) * ($s[$i]==='X' ? 10 : (int)$s[$i]);
23 if ($sum%11 !== 0) fail('invalid isbn');
24 $s = '978' . substr($s,0,9);
25 $sum = 0;
26 for ($i=0; $i<12; ++$i) $sum += (int)$s[$i] * ($i%2===0 ? 1 : 3);
27 return $s . ((10-$sum%10)%10);
28 }
29 if (preg_match('/^97[89][0-9]{10}$/D',$s)) {
30 $sum = 0;
31 for ($i=0; $i<13; ++$i) $sum += (int)$s[$i] * ($i%2===0 ? 1 : 3);
32 if ($sum%10===0) return $s;
33 }
34 fail('invalid isbn');
35}

61function open_day(int $date): int

63 global $closed;
64 while (isset($closed[$date])) ++$date;

67function late_fine(object $c): array
68{
69 global $day,$members;
70 $m = $members[$c->member];
71 $late = max(0,$day-$c->due);
72 return [$late,min($late*RULES[$m->type][2],RULES[$m->type][3])];

74function charge(object $m,string $kind,string $id,int $amount): void
75{
76 global $day,$counts;
77 if ($amount===0) return;
78 $m->items[] = (object)['day'=>$day,'kind'=>$kind,'isbn'=>$id,'amount'=>$amount,'unpaid'=>$amount];
79 $counts[$kind] += $amount;

81function release_copy(object $b,object $c,array &$out): void
82{
83 global $day,$members;
84 if ($b->queue===[]) {
85 $c->state = 'AVAILABLE';
86 return;
87 }
88 $id = array_shift($b->queue);
89 $c->state = 'HOLD';
90 $c->member = $id;
91 $c->expiry = open_day($day+3);
92 $members[$id]->holds[$b->id] = $c;
93 $out[] = "HOLD $b->id copy $c->number $id until $c->expiry";
94}
95function end_loan(object $b,object $c,string $result): array
96{
97 global $day,$members;
98 $m = $members[$c->member];
99 [$late,$fine] = late_fine($c);
100 unset($m->loans[$b->id]);
101 charge($m,'FINE',$b->id,$fine);
102 $m->history[] = "$result $b->id copy $c->number from $c->start to $day late $late fine " . money($fine);
103 return [$m,$late,$fine];
104}
105function lose_copy(object $b,object $c,string $heading,array &$out): void
106{
107 global $counts;
108 [$m,$late,$fine] = end_loan($b,$c,'LOST');
109 $last = count($m->history)-1;
110 $m->history[$last] .= ' price ' . money($b->price);
111 charge($m,'PRICE',$b->id,$b->price);
112 $c->state = 'LOST';
113 ++$counts['lost'];
114 $out[] = "$heading $m->id $b->id copy $c->number late $late fine " . money($fine) . ' price ' . money($b->price) . ' balance ' . money(balance($m));
115 if (holdings($b)===0) {
116 foreach ($b->queue as $id) $out[] = "DROPPED $b->id $id";
117 $b->queue = [];
118 }
119}
120function renewal_reason(object $b,object $c): string
121{
122 global $day;
123 if ($c->due < $day) return 'loan overdue';
124 if ($c->renewals===2) return 'renewal limit';
125 return $b->queue!==[] ? 'title reserved' : '';

127function renew_loan(object $m,object $b,object $c): string
128{
129 global $day,$counts;
130 $c->due = open_day($day+RULES[$m->type][1]);
131 ++$c->renewals;
132 ++$counts['renewed'];
133 return "RENEW $m->id $b->id copy $c->number due $c->due renewals $c->renewals";
134}
135function sorted_copies(array $copies,string $date): array
136{
137 $values = array_values($copies);
138 usort($values,fn($a,$b)=>[$a->$date,$a->isbn] <=> [$b->$date,$b->isbn]);
139 return $values;
140}
141function execute(string $line): array
142{
143 global $day,$closed,$books,$bookOrder,$members,$memberOrder,$counts;
144 $t = explode(' ',$line);
145 $cmd = $t[0];
146 $out = [];
147 if ($cmd==='DAY') {
148 $target = (int)$t[1];
149 if ($target <= $day) fail('invalid day');
150 $out[] = "DAY $target";
151 while ($day < $target) {
152 $next = $target+1;
153 foreach ($members as $m) {
154 foreach ($m->loans as $c) $next = min($next,$c->due+365);
155 foreach ($m->holds as $c) $next = min($next,$c->expiry+1);
156 }
157 if ($next > $target) {
158 $day = $target;
159 break;
160 }
161 $day = $next;
162 $losses = [];
163 foreach ($memberOrder as $id) {
164 $m = $members[$id];
165 $ids = array_keys($m->loans);
166 sort($ids,SORT_STRING);
167 foreach ($ids as $key) {
168 $c = $m->loans[$key];
169 if ($c->due+365 <= $day) $losses[] = $c;
170 }
171 }
172 foreach ($losses as $c) lose_copy($books[$c->isbn],$c,'AUTOLOST',$out);
173 $expired = [];
174 foreach ($bookOrder as $id) {
175 foreach ($books[$id]->copies as $c) {
176 if ($c->state==='HOLD' && $c->expiry < $day) $expired[] = $c;
177 }
178 }
179 foreach ($expired as $c) {
180 $b = $books[$c->isbn];
181 $m = $members[$c->member];
182 $fee = RULES[$m->type][5];
183 charge($m,'FEE',$b->id,$fee);
184 unset($m->holds[$b->id]);
185 $out[] = "EXPIRED $b->id copy $c->number $m->id fee " . money($fee);
186 release_copy($b,$c,$out);
187 }
188 }
189 } elseif ($cmd==='CLOSED') {
190 $date = (int)$t[1];
191 if ($date <= $day) fail('invalid day');
192 if (isset($closed[$date])) fail('already closed');
193 $closed[$date] = true;
194 $out[] = "CLOSED $date";
195 foreach ($bookOrder as $id) {
196 foreach ($books[$id]->copies as $c) {
197 if ($c->state==='LOANED' && $c->due===$date) {
198 $c->due = open_day($date);
199 $out[] = "MOVED $id copy $c->number $c->member due $c->due";
200 } elseif ($c->state==='HOLD' && $c->expiry===$date) {
201 $c->expiry = open_day($date);
202 $out[] = "MOVED $id copy $c->number $c->member until $c->expiry";
203 }
204 }
205 }
206 } elseif ($cmd==='ADDBOOK') {
207 $id = isbn($t[1]);
208 $name = implode(' ',array_slice($t,4));
209 if (isset($books[$id]) && $books[$id]->name!==$name) fail('title mismatch');
210 if (!isset($books[$id])) {
211 $books[$id] = (object)['id'=>$id,'name'=>$name,'price'=>0,'copies'=>[],'queue'=>[],'borrowed'=>0];
212 $bookOrder[] = $id;
213 sort($bookOrder,SORT_STRING);
214 }
215 $b = $books[$id];
216 $b->price = cents($t[3]);
217 $added = [];
218 for ($i=0; $i<(int)$t[2]; ++$i) {
219 $c = (object)['isbn'=>$id,'number'=>count($b->copies)+1,'state'=>'AVAILABLE','member'=>'','start'=>0,'due'=>0,'renewals'=>0,'expiry'=>0];
220 $b->copies[] = $c;
221 $added[] = $c;
222 }
223 $out[] = "ADDBOOK $id added $t[2] total " . holdings($b);
224 foreach ($added as $c) release_copy($b,$c,$out);
225 } elseif ($cmd==='ADDMEMBER') {
226 if (isset($members[$t[1]])) fail('duplicate member');
227 if (!isset(RULES[$t[2]])) fail('unknown type');
228 $m = (object)['id'=>$t[1],'type'=>$t[2],'name'=>implode(' ',array_slice($t,3)),'loans'=>[],'holds'=>[],'items'=>[],'history'=>[]];
229 $members[$m->id] = $m;
230 $memberOrder[] = $m->id;
231 sort($memberOrder,SORT_STRING);
232 $rule = RULES[$m->type];
233 $out[] = "ADDMEMBER $m->id $m->type loans $rule[0] days $rule[1]";
234 } elseif (in_array($cmd,['BORROW','RETURN','RENEW','RESERVE','CANCEL','LOST'],true)) {
235 $m = get_member($t[1]);
236 $b = get_book($t[2]);
237 $id = $b->id;
238 $loan = $m->loans[$id] ?? null;
239 $hold = $m->holds[$id] ?? null;
240 $rule = RULES[$m->type];
241 if ($cmd==='BORROW') {
242 if (isset($closed[$day])) fail('library closed');
243 if ($loan!==null) fail('already borrowed');
244 if (balance($m)>500) fail('member blocked');
245 if (array_any($m->loans,fn($c)=>$c->due<$day)) fail('overdue loans');
246 if (count($m->loans)>=$rule[0]) fail('loan limit');
247 $chosen = $hold ?? array_find($b->copies,fn($c)=>$c->state==='AVAILABLE');
248 if ($chosen===null) {
249 if (array_any($b->copies,fn($c)=>$c->state==='HOLD')) fail('all copies on hold');
250 fail('no copy available');
251 }
252 unset($m->holds[$id]);
253 $chosen->state = 'LOANED';
254 $chosen->member = $m->id;
255 $chosen->start = $day;
256 $chosen->due = open_day($day+$rule[1]);
257 $chosen->renewals = 0;
258 $m->loans[$id] = $chosen;
259 ++$b->borrowed;
260 ++$counts['borrowed'];
261 $out[] = "BORROW $m->id $id copy $chosen->number due $chosen->due";
262 } elseif (in_array($cmd,['RETURN','RENEW','LOST'],true)) {
263 if ($loan===null) fail('not borrowed');
264 if ($cmd==='RETURN') {
265 [$member,$late,$fine] = end_loan($b,$loan,'RETURNED');
266 ++$counts['returned'];
267 $out[] = "RETURN $m->id $id copy $loan->number late $late fine " . money($fine) . ' balance ' . money(balance($m));
268 release_copy($b,$loan,$out);
269 } elseif ($cmd==='LOST') {
270 lose_copy($b,$loan,'LOST',$out);
271 } else {
272 if (balance($m)>500) fail('member blocked');
273 $reason = renewal_reason($b,$loan);
274 if ($reason!=='') fail($reason);
275 $out[] = renew_loan($m,$b,$loan);
276 }
277 } elseif ($cmd==='RESERVE') {
278 if ($loan!==null) fail('already borrowed');
279 if ($hold!==null || in_array($m->id,$b->queue,true)) fail('already reserved');
280 if (balance($m)>500) fail('member blocked');
281 if (reservations($m)>=$rule[4]) fail('reservation limit');
282 if (holdings($b)===0) fail('no copies');
283 if (array_any($b->copies,fn($c)=>$c->state==='AVAILABLE')) fail('copy available');
284 $b->queue[] = $m->id;
285 $position = count($b->queue);
286 $dues = array_column(array_filter($b->copies,fn($c)=>$c->state==='LOANED'),'due');
287 sort($dues,SORT_NUMERIC);
288 $estimate = $dues[$position-1] ?? 'none';
289 $out[] = "RESERVE $m->id $id position $position estimate $estimate";
290 } else {
291 $position = array_search($m->id,$b->queue,true);
292 if ($position===false && $hold===null) fail('not reserved');
293 if ($position!==false) {
294 array_splice($b->queue,$position,1);
295 $out[] = "CANCEL $m->id $id queue";
296 } else {
297 unset($m->holds[$id]);
298 $out[] = "CANCEL $m->id $id hold copy $hold->number";
299 release_copy($b,$hold,$out);
300 }
301 }
302 } elseif (in_array($cmd,['RENEWALL','PAYFINE','WAIVE','FINES','MEMBER','HISTORY'],true)) {
303 $m = get_member($t[1]);
304 if ($cmd==='RENEWALL') {
305 if ($m->loans===[]) fail('no loans');
306 if (balance($m)>500) fail('member blocked');
307 $ids = array_keys($m->loans);
308 sort($ids,SORT_STRING);
309 $renewed = 0;
310 $skipped = 0;
311 foreach ($ids as $id) {
312 $reason = renewal_reason($books[$id],$m->loans[$id]);
313 if ($reason==='') {
314 $out[] = renew_loan($m,$books[$id],$m->loans[$id]);
315 ++$renewed;
316 } else {
317 $out[] = "SKIP $id $reason";
318 ++$skipped;
319 }
320 }
321 $out[] = "RENEWALL $m->id renewed $renewed skipped $skipped";
322 } elseif ($cmd==='PAYFINE') {
323 if (balance($m)===0) fail('no fines');
324 $remaining = cents($t[2]);
325 $paid = 0;
326 $lines = [];
327 foreach ($m->items as $index=>$item) {
328 $applied = min($remaining,$item->unpaid);
329 if ($applied>0) {
330 $item->unpaid -= $applied;
331 $remaining -= $applied;
332 $paid += $applied;
333 $lines[] = 'ITEM ' . ($index+1) . " $item->kind paid " . money($applied) . ' remaining ' . money($item->unpaid);
334 }
335 }
336 $counts['paid'] += $paid;
337 $out[] = "PAYFINE $m->id paid " . money($paid) . ' change ' . money($remaining) . ' balance ' . money(balance($m));
338 $out = [...$out,...$lines];
339 } elseif ($cmd==='WAIVE') {
340 $index = (int)$t[2]-1;
341 if (!isset($m->items[$index])) fail('unknown item');
342 $item = $m->items[$index];
343 if ($item->unpaid===0) fail('item settled');
344 $amount = $item->unpaid;
345 $item->unpaid = 0;
346 $counts['waived'] += $amount;
347 $out[] = "WAIVE $m->id item $t[2] $item->kind amount " . money($amount) . ' balance ' . money(balance($m));
348 } elseif ($cmd==='FINES') {
349 $n = count(array_filter($m->items,fn($item)=>$item->unpaid>0));
350 $out[] = "FINES $m->id items $n balance " . money(balance($m));
351 foreach ($m->items as $index=>$item) {
352 if ($item->unpaid>0) $out[] = ($index+1) . " day $item->day $item->kind $item->isbn amount " . money($item->amount) . ' remaining ' . money($item->unpaid);
353 }
354 } elseif ($cmd==='HISTORY') {
355 $out[] = "HISTORY $m->id " . count($m->history);
356 $out = [...$out,...$m->history];
357 } else {
358 $rule = RULES[$m->type];
359 $state = balance($m)>500 ? 'BLOCKED' : 'OK';
360 $out[] = "MEMBER $m->id $m->type $m->name";
361 $out[] = 'STATUS loans ' . count($m->loans) . "/$rule[0] reservations " . reservations($m) . "/$rule[4] balance " . money(balance($m)) . " $state";
362 foreach (sorted_copies($m->loans,'due') as $c) {
363 $text = "LOAN $c->isbn copy $c->number due $c->due renewals $c->renewals";
364 if ($c->due<$day) {
365 [$late,$fine] = late_fine($c);
366 $text .= " late $late fine " . money($fine);
367 }
368 $out[] = $text;
369 }
370 foreach (sorted_copies($m->holds,'expiry') as $c) $out[] = "HOLD $c->isbn copy $c->number until $c->expiry";
371 foreach ($bookOrder as $id) {
372 $position = array_search($m->id,$books[$id]->queue,true);
373 if ($position!==false) $out[] = "QUEUE $id position " . ($position+1);
374 }
375 }
376 } elseif ($cmd==='WITHDRAW' || $cmd==='BOOK') {
377 $b = get_book($t[1]);
378 if ($cmd==='WITHDRAW') {
379 $index = (int)$t[2]-1;
380 if (!isset($b->copies[$index])) fail('unknown copy');
381 $c = $b->copies[$index];
382 if ($c->state!=='AVAILABLE') fail('copy not available');
383 $c->state = 'WITHDRAWN';
384 $out[] = "WITHDRAW $b->id copy $c->number total " . holdings($b);
385 } else {
386 $out[] = "BOOK $b->id $b->name";
387 $out[] = 'PRICE ' . money($b->price) . ' copies ' . holdings($b) . " borrowed $b->borrowed";
388 foreach ($b->copies as $c) {
389 $text = "COPY $c->number $c->state";
390 if ($c->state==='LOANED') $text .= " $c->member due $c->due";
391 elseif ($c->state==='HOLD') $text .= " $c->member until $c->expiry";
392 $out[] = $text;
393 }
394 $out[] = implode(' ',['QUEUE',count($b->queue),...$b->queue]);
395 }
396 } elseif ($cmd==='SEARCH') {
397 $keyword = strtolower(implode(' ',array_slice($t,1)));
398 $found = array_values(array_filter($books,fn($b)=>str_contains(strtolower($b->name),$keyword)));
399 usort($found,fn($a,$b)=>strcmp(strtolower($a->name),strtolower($b->name)) ?: strcmp($a->id,$b->id));
400 $out[] = 'SEARCH ' . count($found);
401 foreach ($found as $b) {
402 $available = count(array_filter($b->copies,fn($c)=>$c->state==='AVAILABLE'));
403 $out[] = "$b->id $available/" . holdings($b) . " $b->name";
404 }
405 } elseif ($cmd==='OVERDUE' || $cmd==='NOTICES') {
406 $lines = [];
407 if ($cmd==='OVERDUE') {
408 $loans = [];
409 foreach ($members as $m) foreach ($m->loans as $c) if ($c->due<$day) $loans[] = $c;
410 usort($loans,fn($a,$b)=>($a->due <=> $b->due) ?: strcmp($a->member,$b->member) ?: strcmp($a->isbn,$b->isbn));
411 foreach ($loans as $c) {
412 [$late,$fine] = late_fine($c);
413 $lines[] = "$c->member $c->isbn copy $c->number due $c->due late $late fine " . money($fine);
414 }
415 } else {
416 foreach ($memberOrder as $id) {
417 $m = $members[$id];
418 $loans = sorted_copies($m->loans,'due');
419 foreach ($loans as $c) if ($c->due<$day) $lines[] = "OVERDUE $id $c->isbn due $c->due";
420 foreach ($loans as $c) if ($c->due>=$day && $c->due<=$day+2) $lines[] = "DUESOON $id $c->isbn due $c->due";
421 foreach (sorted_copies($m->holds,'expiry') as $c) $lines[] = "PICKUP $id $c->isbn until $c->expiry";
422 }
423 }
424 $out[] = "$cmd " . count($lines);
425 $out = [...$out,...$lines];
426 } elseif ($cmd==='MEMBERS') {
427 $out[] = 'MEMBERS ' . count($members);
428 foreach ($memberOrder as $id) {
429 $m = $members[$id];
430 $state = balance($m)>500 ? 'BLOCKED' : 'OK';
431 $out[] = "$id $m->type loans " . count($m->loans) . ' reservations ' . reservations($m) . ' balance ' . money(balance($m)) . " $state";
432 }
433 } elseif ($cmd==='INVENTORY') {
434 $totals = array_fill_keys(['AVAILABLE','LOANED','HOLD','LOST','WITHDRAWN'],0);
435 $lines = [];
436 foreach ($bookOrder as $id) {
437 $b = $books[$id];
438 $n = array_count_values(array_column($b->copies,'state')) + array_fill_keys(array_keys($totals),0);
439 foreach ($n as $state=>$count) $totals[$state] += $count;
440 $lines[] = "$id copies " . holdings($b) . " available $n[AVAILABLE] loaned $n[LOANED] hold $n[HOLD] queue " . count($b->queue);
441 }
442 $holding = $totals['AVAILABLE']+$totals['LOANED']+$totals['HOLD'];
443 $out[] = 'INVENTORY titles ' . count($books) . " copies $holding available $totals[AVAILABLE] loaned $totals[LOANED] hold $totals[HOLD] lost $totals[LOST] withdrawn $totals[WITHDRAWN]";
444 $out = [...$out,...$lines];
445 } elseif ($cmd==='STATS') {
446 $active = array_sum(array_map(fn($m)=>count($m->loans),$members));
447 $outstanding = array_sum(array_map('balance',$members));
448 $out[] = "STATS day $day titles " . count($books) . ' members ' . count($members) . ' closed ' . count($closed);
449 $out[] = "LOANS borrowed $counts[borrowed] returned $counts[returned] renewed $counts[renewed] lost $counts[lost] active $active";
450 $out[] = 'MONEY fines ' . money($counts['FINE']) . ' fees ' . money($counts['FEE']) . ' charges ' . money($counts['PRICE']) . ' paid ' . money($counts['paid']) . ' waived ' . money($counts['waived']) . ' outstanding ' . money($outstanding);
451 $top = array_values(array_filter($books,fn($b)=>$b->borrowed>0));
452 usort($top,fn($a,$b)=>[-$a->borrowed,$a->id] <=> [-$b->borrowed,$b->id]);
453 foreach (array_slice($top,0,3) as $rank=>$b) $out[] = 'TOP ' . ($rank+1) . " $b->id $b->borrowed $b->name";
454 }
455 return $out;
456}
