11cents <- function(text) {

15money <- function(v) sprintf("%d.%02d",as.integer(v%/%100L),as.integer(v%%100L))
16isbn <- function(text) {
17 s <- gsub("-","",text,fixed=TRUE)
18 digits <- strsplit(s,"",fixed=TRUE)[[1L]]
19 if (grepl("^[0-9]{9}[0-9X]$",s,perl=TRUE)) {
20 values <- vapply(digits,function(char) if (char=="X") 10L else as.integer(char),integer(1L))
21 if (sum((10:1)*values)%%11L!=0L) fail("invalid isbn")
22 s <- paste0("978",substr(s,1L,9L))
23 values <- as.integer(strsplit(s,"",fixed=TRUE)[[1L]])
24 total <- sum(values*rep(c(1L,3L),6L))
25 return(paste0(s,(10L-total%%10L)%%10L))
26 }
27 if (grepl("^97[89][0-9]{10}$",s,perl=TRUE)) {
28 total <- sum(as.integer(digits)*rep(c(1L,3L),length.out=13L))
29 if (total%%10L==0L) return(s)
30 }
31 fail("invalid isbn")
32}

55late_fine <- function(c) {
56 m <- members[[c$member]]

60charge <- function(m,kind,id,amount) {
61 if (amount==0L) return(invisible(NULL))
62 m$items[[length(m$items)+1L]] <- object(day=day,kind=kind,isbn=id,amount=amount,unpaid=amount)
63 counts[[kind]] <<- counts[[kind]]+amount

65release_copy <- function(b,c) {
66 if (!length(b$queue)) {
67 c$state <- "AVAILABLE"
68 return(character())
69 }
70 id <- b$queue[1L]
71 b$queue <- b$queue[-1L]
72 c$state <- "HOLD"
73 c$member <- id
74 c$expiry <- open_day(day+3L)
75 m <- members[[id]]
76 m$holds[[b$id]] <- c
77 paste("HOLD",b$id,"copy",c$number,id,"until",c$expiry)
78}
79end_loan <- function(b,c,result) {
80 m <- members[[c$member]]
81 fine <- late_fine(c)
82 rm(list=b$id,envir=m$loans)
83 charge(m,"FINE",b$id,fine[2L])
84 m$history <- c(m$history,paste(result,b$id,"copy",c$number,"from",c$start,"to",day,"late",fine[1L],"fine",money(fine[2L])))
85 list(member=m,late=fine[1L],fine=fine[2L])
86}
87lose_copy <- function(b,c,heading) {
88 ended <- end_loan(b,c,"LOST")
89 m <- ended$member
90 last <- length(m$history)
91 m$history[last] <- paste(m$history[last],"price",money(b$price))
92 charge(m,"PRICE",b$id,b$price)
93 c$state <- "LOST"
94 counts$lost <<- counts$lost+1L
95 out <- paste(heading,m$id,b$id,"copy",c$number,"late",ended$late,"fine",money(ended$fine),"price",money(b$price),"balance",money(balance(m)))
96 if (holdings(b)==0L) {
97 for (id in b$queue) out <- c(out,paste("DROPPED",b$id,id))
98 b$queue <- character()
99 }
100 out
101}
102renewal_reason <- function(b,c) {

104 if (c$renewals==2L) return("renewal limit")
105 if (length(b$queue)) "title reserved" else ""
106}
107renew_loan <- function(m,b,c) {
108 c$due <- open_day(day+rules[[m$type]][2L])
109 c$renewals <- c$renewals+1L
110 counts$renewed <<- counts$renewed+1L
111 paste("RENEW",m$id,b$id,"copy",c$number,"due",c$due,"renewals",c$renewals)
112}
113sorted_copies <- function(map,date) {
114 copies <- map_values(map)
115 dates <- vapply(copies,function(c) c[[date]],integer(1L))
116 ids <- vapply(copies,function(c) c$isbn,character(1L))
117 copies[order(dates,ids,method="radix")]
118}
119execute <- function(line) {
120 t <- strsplit(line," ",fixed=TRUE)[[1L]]
121 cmd <- t[1L]
122 out <- character()
123 emit <- function(lines) out <<- c(out,lines)
124 if (cmd=="DAY") {
125 target <- as.integer(t[2L])
126 if (target<=day) fail("invalid day")
127 emit(paste("DAY",target))
128 while (day<target) {
129 next_day <- target+1L
130 for (id in member_order) {
131 m <- members[[id]]
132 for (c in map_values(m$loans)) next_day <- min(next_day,c$due+365L)
133 for (c in map_values(m$holds)) next_day <- min(next_day,c$expiry+1L)
134 }
135 if (next_day>target) {
136 day <<- target
137 break
138 }
139 day <<- next_day
140 losses <- list()
141 for (id in member_order) {
142 m <- members[[id]]
143 for (key in sort(ls(m$loans,all.names=TRUE),method="radix")) {
144 c <- m$loans[[key]]
145 if (c$due+365L<=day) losses[[length(losses)+1L]] <- c
146 }
147 }
148 for (c in losses) emit(lose_copy(books[[c$isbn]],c,"AUTOLOST"))
149 expired <- list()
150 for (id in book_order) {
151 for (c in books[[id]]$copies) {
152 if (c$state=="HOLD" && c$expiry<day) expired[[length(expired)+1L]] <- c
153 }
154 }
155 for (c in expired) {
156 b <- books[[c$isbn]]
157 m <- members[[c$member]]
158 fee <- rules[[m$type]][6L]
159 charge(m,"FEE",b$id,fee)
160 rm(list=b$id,envir=m$holds)
161 emit(paste("EXPIRED",b$id,"copy",c$number,m$id,"fee",money(fee)))
162 emit(release_copy(b,c))
163 }
164 }
165 } else if (cmd=="CLOSED") {
166 date <- as.integer(t[2L])
167 if (date<=day) fail("invalid day")
168 if (date %in% closed) fail("already closed")
169 closed <<- c(closed,date)
170 emit(paste("CLOSED",date))
171 for (id in book_order) {
172 for (c in books[[id]]$copies) {
173 if (c$state=="LOANED" && c$due==date) {
174 c$due <- open_day(date)
175 emit(paste("MOVED",id,"copy",c$number,c$member,"due",c$due))
176 } else if (c$state=="HOLD" && c$expiry==date) {
177 c$expiry <- open_day(date)
178 emit(paste("MOVED",id,"copy",c$number,c$member,"until",c$expiry))
179 }
180 }
181 }
182 } else if (cmd=="ADDBOOK") {
183 id <- isbn(t[2L])
184 name <- paste(t[5:length(t)],collapse=" ")
185 b <- books[[id]]
186 if (!is.null(b) && b$name!=name) fail("title mismatch")
187 if (is.null(b)) {
188 b <- object(id=id,name=name,price=0L,copies=list(),queue=character(),borrowed=0L)
189 books[[id]] <- b
190 book_order <<- sort(c(book_order,id),method="radix")
191 }
192 b$price <- cents(t[4L])
193 added <- list()
194 for (i in seq_len(as.integer(t[3L]))) {
195 c <- object(isbn=id,number=length(b$copies)+1L,state="AVAILABLE",member="",start=0L,due=0L,renewals=0L,expiry=0L)
196 b$copies[[length(b$copies)+1L]] <- c
197 added[[length(added)+1L]] <- c
198 }
199 emit(paste("ADDBOOK",id,"added",t[3L],"total",holdings(b)))
200 for (c in added) emit(release_copy(b,c))
201 } else if (cmd=="ADDMEMBER") {
202 if (!is.null(members[[t[2L]]])) fail("duplicate member")
203 if (is.null(rules[[t[3L]]])) fail("unknown type")
204 m <- object(id=t[2L],type=t[3L],name=paste(t[4:length(t)],collapse=" "),loans=registry(),holds=registry(),items=list(),history=character())
205 members[[m$id]] <- m
206 member_order <<- sort(c(member_order,m$id),method="radix")
207 rule <- rules[[m$type]]
208 emit(paste("ADDMEMBER",m$id,m$type,"loans",rule[1L],"days",rule[2L]))
209 } else if (cmd %in% c("BORROW","RETURN","RENEW","RESERVE","CANCEL","LOST")) {
210 m <- get_member(t[2L])
211 b <- get_book(t[3L])
212 id <- b$id
213 loan <- m$loans[[id]]
214 hold <- m$holds[[id]]
215 rule <- rules[[m$type]]
216 if (cmd=="BORROW") {
217 if (day %in% closed) fail("library closed")
218 if (!is.null(loan)) fail("already borrowed")
219 if (balance(m)>500L) fail("member blocked")
220 for (c in map_values(m$loans)) if (c$due<day) fail("overdue loans")
221 if (length(ls(m$loans,all.names=TRUE))>=rule[1L]) fail("loan limit")
222 chosen <- hold
223 if (is.null(chosen)) {
224 for (c in b$copies) {
225 if (c$state=="AVAILABLE") {
226 chosen <- c
227 break
228 }
229 }
230 }
231 if (is.null(chosen)) {
232 for (c in b$copies) if (c$state=="HOLD") fail("all copies on hold")
233 fail("no copy available")
234 }
235 if (!is.null(hold)) rm(list=id,envir=m$holds)
236 chosen$state <- "LOANED"
237 chosen$member <- m$id
238 chosen$start <- day
239 chosen$due <- open_day(day+rule[2L])
240 chosen$renewals <- 0L
241 m$loans[[id]] <- chosen
242 b$borrowed <- b$borrowed+1L
243 counts$borrowed <<- counts$borrowed+1L
244 emit(paste("BORROW",m$id,id,"copy",chosen$number,"due",chosen$due))
245 } else if (cmd %in% c("RETURN","RENEW","LOST")) {
246 if (is.null(loan)) fail("not borrowed")
247 if (cmd=="RETURN") {
248 ended <- end_loan(b,loan,"RETURNED")
249 counts$returned <<- counts$returned+1L
250 emit(paste("RETURN",m$id,id,"copy",loan$number,"late",ended$late,"fine",money(ended$fine),"balance",money(balance(m))))
251 emit(release_copy(b,loan))
252 } else if (cmd=="LOST") {
253 emit(lose_copy(b,loan,"LOST"))
254 } else {
255 if (balance(m)>500L) fail("member blocked")
256 reason <- renewal_reason(b,loan)
257 if (reason!="") fail(reason)
258 emit(renew_loan(m,b,loan))
259 }
260 } else if (cmd=="RESERVE") {
261 if (!is.null(loan)) fail("already borrowed")
262 if (!is.null(hold) || m$id %in% b$queue) fail("already reserved")
263 if (balance(m)>500L) fail("member blocked")
264 if (reservations(m)>=rule[5L]) fail("reservation limit")
265 if (holdings(b)==0L) fail("no copies")
266 for (c in b$copies) if (c$state=="AVAILABLE") fail("copy available")
267 b$queue <- c(b$queue,m$id)
268 position <- length(b$queue)
269 dues <- sort(vapply(Filter(function(c) c$state=="LOANED",b$copies),function(c) c$due,integer(1L)))
270 estimate <- if (position<=length(dues)) as.character(dues[position]) else "none"
271 emit(paste("RESERVE",m$id,id,"position",position,"estimate",estimate))
272 } else {
273 position <- match(m$id,b$queue)
274 if (is.na(position) && is.null(hold)) fail("not reserved")
275 if (!is.na(position)) {
276 b$queue <- b$queue[-position]
277 emit(paste("CANCEL",m$id,id,"queue"))
278 } else {
279 rm(list=id,envir=m$holds)
280 emit(paste("CANCEL",m$id,id,"hold copy",hold$number))
281 emit(release_copy(b,hold))
282 }
283 }
284 } else if (cmd %in% c("RENEWALL","PAYFINE","WAIVE","FINES","MEMBER","HISTORY")) {
285 m <- get_member(t[2L])
286 if (cmd=="RENEWALL") {
287 ids <- sort(ls(m$loans,all.names=TRUE),method="radix")
288 if (!length(ids)) fail("no loans")
289 if (balance(m)>500L) fail("member blocked")
290 renewed <- skipped <- 0L
291 for (id in ids) {
292 reason <- renewal_reason(books[[id]],m$loans[[id]])
293 if (reason=="") {
294 emit(renew_loan(m,books[[id]],m$loans[[id]]))
295 renewed <- renewed+1L
296 } else {
297 emit(paste("SKIP",id,reason))
298 skipped <- skipped+1L
299 }
300 }
301 emit(paste("RENEWALL",m$id,"renewed",renewed,"skipped",skipped))
302 } else if (cmd=="PAYFINE") {
303 if (balance(m)==0L) fail("no fines")
304 remaining <- cents(t[3L])
305 paid <- 0L
306 lines <- character()
307 for (index in seq_along(m$items)) {
308 item <- m$items[[index]]
309 applied <- min(remaining,item$unpaid)
310 if (applied>0L) {
311 item$unpaid <- item$unpaid-applied
312 remaining <- remaining-applied
313 paid <- paid+applied
314 lines <- c(lines,paste("ITEM",index,item$kind,"paid",money(applied),"remaining",money(item$unpaid)))
315 }
316 }
317 counts$paid <<- counts$paid+paid
318 emit(paste("PAYFINE",m$id,"paid",money(paid),"change",money(remaining),"balance",money(balance(m))))
319 emit(lines)
320 } else if (cmd=="WAIVE") {
321 index <- as.integer(t[3L])
322 if (index<1L || index>length(m$items)) fail("unknown item")
323 item <- m$items[[index]]
324 if (item$unpaid==0L) fail("item settled")
325 amount <- item$unpaid
326 item$unpaid <- 0L
327 counts$waived <<- counts$waived+amount
328 emit(paste("WAIVE",m$id,"item",index,item$kind,"amount",money(amount),"balance",money(balance(m))))
329 } else if (cmd=="FINES") {
330 n <- sum(vapply(m$items,function(item) item$unpaid>0L,logical(1L)))
331 emit(paste("FINES",m$id,"items",n,"balance",money(balance(m))))
332 for (index in seq_along(m$items)) {
333 item <- m$items[[index]]
334 if (item$unpaid>0L) emit(paste(index,"day",item$day,item$kind,item$isbn,"amount",money(item$amount),"remaining",money(item$unpaid)))
335 }
336 } else if (cmd=="HISTORY") {
337 emit(paste("HISTORY",m$id,length(m$history)))
338 emit(m$history)
339 } else {
340 rule <- rules[[m$type]]
341 state <- if (balance(m)>500L) "BLOCKED" else "OK"
342 emit(paste("MEMBER",m$id,m$type,m$name))
343 emit(paste("STATUS loans",paste0(length(ls(m$loans,all.names=TRUE)),"/",rule[1L]),"reservations",paste0(reservations(m),"/",rule[5L]),"balance",money(balance(m)),state))
344 for (c in sorted_copies(m$loans,"due")) {
345 text <- paste("LOAN",c$isbn,"copy",c$number,"due",c$due,"renewals",c$renewals)
346 if (c$due<day) {
347 fine <- late_fine(c)
348 text <- paste(text,"late",fine[1L],"fine",money(fine[2L]))
349 }
350 emit(text)
351 }
352 for (c in sorted_copies(m$holds,"expiry")) emit(paste("HOLD",c$isbn,"copy",c$number,"until",c$expiry))
353 for (id in book_order) {
354 position <- match(m$id,books[[id]]$queue)
355 if (!is.na(position)) emit(paste("QUEUE",id,"position",position))
356 }
357 }
358 } else if (cmd %in% c("WITHDRAW","BOOK")) {
359 b <- get_book(t[2L])
360 if (cmd=="WITHDRAW") {
361 index <- as.integer(t[3L])
362 if (index<1L || index>length(b$copies)) fail("unknown copy")
363 c <- b$copies[[index]]
364 if (c$state!="AVAILABLE") fail("copy not available")
365 c$state <- "WITHDRAWN"
366 emit(paste("WITHDRAW",b$id,"copy",c$number,"total",holdings(b)))
367 } else {
368 emit(paste("BOOK",b$id,b$name))
369 emit(paste("PRICE",money(b$price),"copies",holdings(b),"borrowed",b$borrowed))
370 for (c in b$copies) {
371 text <- paste("COPY",c$number,c$state)
372 if (c$state=="LOANED") text <- paste(text,c$member,"due",c$due) else if (c$state=="HOLD") text <- paste(text,c$member,"until",c$expiry)
373 emit(text)
374 }
375 emit(paste(c("QUEUE",length(b$queue),b$queue),collapse=" "))
376 }
377 } else if (cmd=="SEARCH") {
378 keyword <- chartr("ABCDEFGHIJKLMNOPQRSTUVWXYZ","abcdefghijklmnopqrstuvwxyz",paste(t[-1L],collapse=" "))
379 lower <- function(text) chartr("ABCDEFGHIJKLMNOPQRSTUVWXYZ","abcdefghijklmnopqrstuvwxyz",text)
380 found <- Filter(function(b) grepl(keyword,lower(b$name),fixed=TRUE),map_values(books))
381 names <- vapply(found,function(b) lower(b$name),character(1L))
382 ids <- vapply(found,function(b) b$id,character(1L))
383 found <- found[order(names,ids,method="radix")]
384 emit(paste("SEARCH",length(found)))
385 for (b in found) {
386 available <- sum(vapply(b$copies,function(c) c$state=="AVAILABLE",logical(1L)))
387 emit(paste(b$id,paste0(available,"/",holdings(b)),b$name))
388 }
389 } else if (cmd %in% c("OVERDUE","NOTICES")) {
390 lines <- character()
391 if (cmd=="OVERDUE") {
392 loans <- list()
393 for (id in member_order) {
394 for (c in map_values(members[[id]]$loans)) if (c$due<day) loans[[length(loans)+1L]] <- c
395 }
396 dues <- vapply(loans,function(c) c$due,integer(1L))
397 mids <- vapply(loans,function(c) c$member,character(1L))
398 ids <- vapply(loans,function(c) c$isbn,character(1L))
399 loans <- loans[order(dues,mids,ids,method="radix")]
400 for (c in loans) {
401 fine <- late_fine(c)
402 lines <- c(lines,paste(c$member,c$isbn,"copy",c$number,"due",c$due,"late",fine[1L],"fine",money(fine[2L])))
403 }
404 } else {
405 for (id in member_order) {
406 m <- members[[id]]
407 loans <- sorted_copies(m$loans,"due")
408 for (c in loans) if (c$due<day) lines <- c(lines,paste("OVERDUE",id,c$isbn,"due",c$due))
409 for (c in loans) if (c$due>=day && c$due<=day+2L) lines <- c(lines,paste("DUESOON",id,c$isbn,"due",c$due))
410 for (c in sorted_copies(m$holds,"expiry")) lines <- c(lines,paste("PICKUP",id,c$isbn,"until",c$expiry))
411 }
412 }
413 emit(paste(cmd,length(lines)))
414 emit(lines)
415 } else if (cmd=="MEMBERS") {
416 emit(paste("MEMBERS",length(member_order)))
417 for (id in member_order) {
418 m <- members[[id]]
419 state <- if (balance(m)>500L) "BLOCKED" else "OK"
420 emit(paste(id,m$type,"loans",length(ls(m$loans,all.names=TRUE)),"reservations",reservations(m),"balance",money(balance(m)),state))
421 }
422 } else if (cmd=="INVENTORY") {
423 totals <- setNames(rep(0L,5L),c("AVAILABLE","LOANED","HOLD","LOST","WITHDRAWN"))
424 lines <- character()
425 for (id in book_order) {
426 b <- books[[id]]
427 states <- vapply(b$copies, function(c) c$state, character(1L))
428 n <- setNames(as.integer(table(factor(states, levels = names(totals)))), names(totals))
429 totals <- totals+n
430 lines <- c(lines,paste(id,"copies",holdings(b),"available",n[["AVAILABLE"]],"loaned",n[["LOANED"]],"hold",n[["HOLD"]],"queue",length(b$queue)))
431 }
432 holding <- totals[["AVAILABLE"]]+totals[["LOANED"]]+totals[["HOLD"]]
433 emit(paste("INVENTORY titles",length(book_order),"copies",holding,"available",totals[["AVAILABLE"]],"loaned",totals[["LOANED"]],"hold",totals[["HOLD"]],"lost",totals[["LOST"]],"withdrawn",totals[["WITHDRAWN"]]))
434 emit(lines)
435 } else if (cmd=="STATS") {
436 active <- outstanding <- 0L
437 for (id in member_order) {
438 active <- active+length(ls(members[[id]]$loans,all.names=TRUE))
439 outstanding <- outstanding+balance(members[[id]])
440 }
441 emit(paste("STATS day",day,"titles",length(book_order),"members",length(member_order),"closed",length(closed)))
442 emit(paste("LOANS borrowed",counts$borrowed,"returned",counts$returned,"renewed",counts$renewed,"lost",counts$lost,"active",active))
443 emit(paste("MONEY fines",money(counts$FINE),"fees",money(counts$FEE),"charges",money(counts$PRICE),"paid",money(counts$paid),"waived",money(counts$waived),"outstanding",money(outstanding)))
444 top <- Filter(function(b) b$borrowed>0L,map_values(books))
445 totals <- vapply(top,function(b) b$borrowed,integer(1L))
446 ids <- vapply(top,function(b) b$id,character(1L))
447 top <- top[order(-totals,ids,method="radix")]
448 for (rank in seq_len(min(3L,length(top)))) {
449 b <- top[[rank]]
450 emit(paste("TOP",rank,b$id,b$borrowed,b$name))
451 }
452 }
453 out
454}
