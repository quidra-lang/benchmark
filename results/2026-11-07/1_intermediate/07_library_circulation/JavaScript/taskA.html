4class Invalid extends Error {}

6function fail(reason) {
7 throw new Invalid(reason);

10function amount(text) {
11 const [whole, fraction] = text.split(".");
12 return Number(whole) * 100 + Number(fraction);
13}
14
15function money(value) {
16 return Math.floor(value / 100) + "." + String(value % 100).padStart(2, "0");
17}
18
19function lex(a, b) {
20 return a < b ? -1 : a > b ? 1 : 0;
21}

23function isbn(text) {
24 const raw = text.replace(/-/g, "");
25 if (/^[0-9]{9}[0-9X]$/.test(raw)) {
26 const sum = raw.split("").reduce((total, c, i) => total + (10 - i) * (c === "X" ? 10 : Number(c)), 0);
27 if (sum % 11 !== 0) {
28 fail("invalid isbn");
29 }
30 const prefix = "978" + raw.slice(0, 9);
31 const checksum = prefix.split("").reduce((total, c, i) => total + Number(c) * (i % 2 === 0 ? 1 : 3), 0);
32 return prefix + (10 - checksum % 10) % 10;
33 }
34 if (/^(978|979)[0-9]{10}$/.test(raw)) {
35 const sum = raw.split("").reduce((total, c, i) => total + Number(c) * (i % 2 === 0 ? 1 : 3), 0);
36 if (sum % 10 === 0) {
37 return raw;
38 }
39 }
40 return fail("invalid isbn");
41}

43const rules = {
44 STUDENT: { limit: 3, period: 14, fine: 25, cap: 1000, reservations: 2, fee: 50 },
45 STAFF: { limit: 10, period: 28, fine: 10, cap: 2000, reservations: 5, fee: 100 }
46};

48function holdings(title) {
49 return title.copies.filter(copy => ["AVAILABLE", "LOANED", "HOLD"].includes(copy.state)).length;
50}
51
52function copyOrder(a, b) {
53 return lex(a.isbn, b.isbn) || a.no - b.no;
54}
55
56function loanOrder(a, b) {
57 return a.due - b.due || lex(a.isbn, b.isbn);
58}
59
60function holdOrder(a, b) {
61 return a.expiry - b.expiry || lex(a.isbn, b.isbn);
62}

64class Library {
65 day = 1;
66 closed = new Set();
67 titles = new Map();
68 members = new Map();
69 counts = { borrowed: 0, returned: 0, renewed: 0, lost: 0, FINE: 0, FEE: 0, PRICE: 0, paid: 0, waived: 0 };
70
71 member(id) {
72 const member = this.members.get(id);
73 if (!member) {
74 fail("unknown member");
75 }
76 return member;
77 }
78
79 title(token) {
80 const title = this.titles.get(isbn(token));
81 if (!title) {
82 fail("unknown book");
83 }
84 return title;
85 }
86
87 common(id, token) {
88 const member = this.member(id);
89 const title = this.title(token);
90 return [member, title];
91 }
92
93 allCopies() {
94 return [...this.titles.values()].flatMap(title => title.copies);
95 }
96
97 balance(member) {
98 return member.items.reduce((sum, item) => sum + item.unpaid, 0);
99 }
100
101 reservations(member) {
102 return member.holds.size + [...this.titles.values()].filter(title => title.queue.includes(member.id)).length;
103 }
104
105 openDay(day) {
106 while (this.closed.has(day)) {
107 day++;
108 }
109 return day;
110 }
111
112 late(copy) {
113 const days = Math.max(0, this.day - copy.due);
114 const rule = rules[this.members.get(copy.member).type];
115 return [days, Math.min(days * rule.fine, rule.cap)];
116 }
117
118 charge(member, kind, title, value) {
119 if (!value) {
120 return;
121 }
122 member.items.push({ no: member.items.length + 1, day: this.day, kind, isbn: title.isbn, amount: value, unpaid: value });
123 this.counts[kind] += value;
124 }
125
126 release(title, copy) {
127 if (!title.queue.length) {
128 copy.state = "AVAILABLE";
129 copy.member = "";
130 return [];
131 }
132 const id = title.queue.shift();
133 copy.state = "HOLD";
134 copy.member = id;
135 copy.expiry = this.openDay(this.day + 3);
136 this.members.get(id).holds.set(title.isbn, copy);
137 return ["HOLD " + title.isbn + " copy " + copy.no + " " + id + " until " + copy.expiry];
138 }
139
140 endLoan(title, copy, result) {
141 const member = this.members.get(copy.member);
142 const [late, fine] = this.late(copy);
143 member.loans.delete(title.isbn);
144 this.charge(member, "FINE", title, fine);
145 member.history.push(result + " " + title.isbn + " copy " + copy.no + " from " + copy.loanDay +
146 " to " + this.day + " late " + late + " fine " + money(fine));
147 return [late, fine];
148 }
149
150 loss(title, copy, heading) {
151 const member = this.members.get(copy.member);
152 const [late, fine] = this.endLoan(title, copy, "LOST");
153 member.history[member.history.length - 1] += " price " + money(title.price);
154 this.charge(member, "PRICE", title, title.price);
155 copy.state = "LOST";
156 copy.member = "";
157 this.counts.lost++;
158 const output = [heading + " " + member.id + " " + title.isbn + " copy " + copy.no + " late " + late +
159 " fine " + money(fine) + " price " + money(title.price) + " balance " + money(this.balance(member))];
160 if (!holdings(title) && title.queue.length) {
161 output.push(...title.queue.map(id => "DROPPED " + title.isbn + " " + id));
162 title.queue = [];
163 }
164 return output;
165 }
166
167 renewalReason(title, copy) {
168 if (copy.due < this.day) {
169 return "loan overdue";
170 }
171 if (copy.renewals === 2) {
172 return "renewal limit";
173 }
174 return title.queue.length ? "title reserved" : "";
175 }
176
177 renew(member, title, copy) {
178 copy.due = this.openDay(this.day + rules[member.type].period);
179 copy.renewals++;
180 this.counts.renewed++;
181 return "RENEW " + member.id + " " + title.isbn + " copy " + copy.no + " due " + copy.due + " renewals " + copy.renewals;
182 }
183
184 advance(day) {
185 if (day <= this.day) {
186 fail("invalid day");
187 }
188 const output = ["DAY " + day];
189 while (this.day < day) {
190 this.day++;
191 const losses = this.allCopies().filter(copy => copy.state === "LOANED" && copy.due + 365 <= this.day)
192 .sort((a, b) => lex(a.member, b.member) || lex(a.isbn, b.isbn));
193 for (const copy of losses) {
194 output.push(...this.loss(this.titles.get(copy.isbn), copy, "AUTOLOST"));
195 }
196 const expired = this.allCopies().filter(copy => copy.state === "HOLD" && copy.expiry < this.day).sort(copyOrder);
197 for (const copy of expired) {
198 const title = this.titles.get(copy.isbn);
199 const member = this.members.get(copy.member);
200 const fee = rules[member.type].fee;
201 this.charge(member, "FEE", title, fee);
202 member.holds.delete(copy.isbn);
203 output.push("EXPIRED " + copy.isbn + " copy " + copy.no + " " + member.id + " fee " + money(fee));
204 output.push(...this.release(title, copy));
205 }
206 }
207 return output;
208 }

210 command(line) {
211 const [name, ...args] = line.split(" ");
212 if (name === "DAY") {
213 return this.advance(Number(args[0]));
214 }
215 if (name === "CLOSED") {
216 const day = Number(args[0]);
217 if (day <= this.day) {
218 fail("invalid day");
219 }
220 if (this.closed.has(day)) {
221 fail("already closed");
222 }
223 this.closed.add(day);
224 const output = ["CLOSED " + day];
225 for (const copy of this.allCopies().sort(copyOrder)) {
226 if (copy.state === "LOANED" && copy.due === day) {
227 copy.due = this.openDay(day);
228 output.push("MOVED " + copy.isbn + " copy " + copy.no + " " + copy.member + " due " + copy.due);
229 } else if (copy.state === "HOLD" && copy.expiry === day) {
230 copy.expiry = this.openDay(day);
231 output.push("MOVED " + copy.isbn + " copy " + copy.no + " " + copy.member + " until " + copy.expiry);
232 }
233 }
234 return output;
235 }
236 if (name === "ADDBOOK") {
237 const id = isbn(args[0]);
238 const bookName = args.slice(3).join(" ");
239 let title = this.titles.get(id);
240 if (title && title.name !== bookName) {
241 fail("title mismatch");
242 }
243 if (!title) {
244 title = { isbn: id, name: bookName, price: amount(args[2]), copies: [], queue: [], borrowed: 0 };
245 this.titles.set(id, title);
246 }
247 title.price = amount(args[2]);
248 const added = [];
249 for (let i = 0; i < Number(args[1]); i++) {
250 const copy = { isbn: id, no: title.copies.length + 1, state: "AVAILABLE", member: "", loanDay: 0, due: 0, renewals: 0, expiry: 0 };
251 title.copies.push(copy);
252 added.push(copy);
253 }
254 const output = ["ADDBOOK " + id + " added " + added.length + " total " + holdings(title)];
255 for (const copy of added) {
256 output.push(...this.release(title, copy));
257 }
258 return output;
259 }
260 if (name === "ADDMEMBER") {
261 if (this.members.has(args[0])) {
262 fail("duplicate member");
263 }
264 if (!Object.hasOwn(rules, args[1])) {
265 fail("unknown type");
266 }
267 this.members.set(args[0], {
268 id: args[0], type: args[1], name: args.slice(2).join(" "),
269 loans: new Map(), holds: new Map(), items: [], history: []
270 });
271 return ["ADDMEMBER " + args[0] + " " + args[1] + " loans " + rules[args[1]].limit + " days " + rules[args[1]].period];
272 }
273 if (name === "WITHDRAW") {
274 const title = this.title(args[0]);
275 const number = Number(args[1]);
276 if (number < 1 || number > title.copies.length) {
277 fail("unknown copy");
278 }
279 const copy = title.copies[number - 1];
280 if (copy.state !== "AVAILABLE") {
281 fail("copy not available");
282 }
283 copy.state = "WITHDRAWN";
284 return ["WITHDRAW " + title.isbn + " copy " + copy.no + " total " + holdings(title)];
285 }
286 if (["BORROW", "RETURN", "RENEW", "RESERVE", "CANCEL", "LOST"].includes(name)) {
287 return this.circulate(name, args);
288 }
289 if (name === "RENEWALL") {
290 const member = this.member(args[0]);
291 if (!member.loans.size) {
292 fail("no loans");
293 }
294 if (this.balance(member) > 500) {
295 fail("member blocked");
296 }
297 const copies = [...member.loans.values()].sort((a, b) => lex(a.isbn, b.isbn));
298 const output = [];
299 let renewed = 0;
300 let skipped = 0;
301 for (const copy of copies) {
302 const title = this.titles.get(copy.isbn);
303 const reason = this.renewalReason(title, copy);
304 if (reason) {
305 output.push("SKIP " + copy.isbn + " " + reason);
306 skipped++;
307 } else {
308 output.push(this.renew(member, title, copy));
309 renewed++;
310 }
311 }
312 output.push("RENEWALL " + member.id + " renewed " + renewed + " skipped " + skipped);
313 return output;
314 }
315 if (["PAYFINE", "WAIVE", "FINES"].includes(name)) {
316 return this.charges(name, args);
317 }
318 return this.inspect(name, args);
319 }

321 circulate(name, args) {
322 const [member, title] = this.common(args[0], args[1]);
323 const loan = member.loans.get(title.isbn);
324 const hold = member.holds.get(title.isbn);
325 const rule = rules[member.type];
326 if (name === "BORROW") {
327 if (this.closed.has(this.day)) {
328 fail("library closed");
329 }
330 if (loan) {
331 fail("already borrowed");
332 }
333 if (this.balance(member) > 500) {
334 fail("member blocked");
335 }
336 if ([...member.loans.values()].some(copy => copy.due < this.day)) {
337 fail("overdue loans");
338 }
339 if (member.loans.size >= rule.limit) {
340 fail("loan limit");
341 }
342 const copy = hold ?? title.copies.find(copy => copy.state === "AVAILABLE");
343 if (!copy) {
344 fail(title.copies.some(copy => copy.state === "HOLD") ? "all copies on hold" : "no copy available");
345 }
346 member.holds.delete(title.isbn);
347 copy.state = "LOANED";
348 copy.member = member.id;
349 copy.loanDay = this.day;
350 copy.due = this.openDay(this.day + rule.period);
351 copy.renewals = 0;
352 member.loans.set(title.isbn, copy);
353 title.borrowed++;
354 this.counts.borrowed++;
355 return ["BORROW " + member.id + " " + title.isbn + " copy " + copy.no + " due " + copy.due];
356 }
357 if (name === "RETURN" || name === "LOST" || name === "RENEW") {
358 if (!loan) {
359 fail("not borrowed");
360 }
361 if (name === "LOST") {
362 return this.loss(title, loan, "LOST");
363 }
364 if (name === "RETURN") {
365 const [late, fine] = this.endLoan(title, loan, "RETURNED");
366 this.counts.returned++;
367 return ["RETURN " + member.id + " " + title.isbn + " copy " + loan.no + " late " + late +
368 " fine " + money(fine) + " balance " + money(this.balance(member)), ...this.release(title, loan)];
369 }
370 if (this.balance(member) > 500) {
371 fail("member blocked");
372 }
373 const reason = this.renewalReason(title, loan);
374 if (reason) {
375 fail(reason);
376 }
377 return [this.renew(member, title, loan)];
378 }
379 const queued = title.queue.includes(member.id);
380 if (name === "CANCEL") {
381 if (!queued && !hold) {
382 fail("not reserved");
383 }
384 if (queued) {
385 title.queue.splice(title.queue.indexOf(member.id), 1);
386 return ["CANCEL " + member.id + " " + title.isbn + " queue"];
387 }
388 member.holds.delete(title.isbn);
389 return ["CANCEL " + member.id + " " + title.isbn + " hold copy " + hold.no, ...this.release(title, hold)];
390 }
391 if (loan) {
392 fail("already borrowed");
393 }
394 if (queued || hold) {
395 fail("already reserved");
396 }
397 if (this.balance(member) > 500) {
398 fail("member blocked");
399 }
400 if (this.reservations(member) >= rule.reservations) {
401 fail("reservation limit");
402 }
403 if (!holdings(title)) {
404 fail("no copies");
405 }
406 if (title.copies.some(copy => copy.state === "AVAILABLE")) {
407 fail("copy available");
408 }
409 title.queue.push(member.id);
410 const days = title.copies.filter(copy => copy.state === "LOANED").map(copy => copy.due).sort((a, b) => a - b);
411 const position = title.queue.length;
412 return ["RESERVE " + member.id + " " + title.isbn + " position " + position + " estimate " + (days[position - 1] ?? "none")];
413 }

415 charges(name, args) {
416 const member = this.member(args[0]);
417 if (name === "PAYFINE") {
418 if (!this.balance(member)) {
419 fail("no fines");
420 }
421 let remaining = amount(args[1]);
422 const items = [];
423 for (const item of member.items) {
424 const applied = Math.min(remaining, item.unpaid);
425 if (applied) {
426 item.unpaid -= applied;
427 remaining -= applied;
428 items.push("ITEM " + item.no + " " + item.kind + " paid " + money(applied) + " remaining " + money(item.unpaid));
429 }
430 if (!remaining) {
431 break;
432 }
433 }
434 const paid = amount(args[1]) - remaining;
435 this.counts.paid += paid;
436 return ["PAYFINE " + member.id + " paid " + money(paid) + " change " + money(remaining) + " balance " + money(this.balance(member)), ...items];
437 }
438 if (name === "WAIVE") {
439 const no = Number(args[1]);
440 if (no < 1 || no > member.items.length) {
441 fail("unknown item");
442 }
443 const item = member.items[no - 1];
444 if (!item.unpaid) {
445 fail("item settled");
446 }
447 const value = item.unpaid;
448 item.unpaid = 0;
449 this.counts.waived += value;
450 return ["WAIVE " + member.id + " item " + no + " " + item.kind + " amount " + money(value) + " balance " + money(this.balance(member))];
451 }
452 const items = member.items.filter(item => item.unpaid);
453 return ["FINES " + member.id + " items " + items.length + " balance " + money(this.balance(member)),
454 ...items.map(item => item.no + " day " + item.day + " " + item.kind + " " + item.isbn +
455 " amount " + money(item.amount) + " remaining " + money(item.unpaid))];
456 }

458 inspect(name, args) {
459 if (name === "SEARCH") {
460 const keyword = args.join(" ").toLowerCase();
461 const titles = [...this.titles.values()].filter(title => title.name.toLowerCase().includes(keyword))
462 .sort((a, b) => lex(a.name.toLowerCase(), b.name.toLowerCase()) || lex(a.isbn, b.isbn));
463 return ["SEARCH " + titles.length, ...titles.map(title => title.isbn + " " +
464 title.copies.filter(copy => copy.state === "AVAILABLE").length + "/" + holdings(title) + " " + title.name)];
465 }
466 if (name === "OVERDUE") {
467 const copies = this.allCopies().filter(copy => copy.state === "LOANED" && copy.due < this.day)
468 .sort((a, b) => a.due - b.due || lex(a.member, b.member) || lex(a.isbn, b.isbn));
469 return ["OVERDUE " + copies.length, ...copies.map(copy => {
470 const [late, fine] = this.late(copy);
471 return copy.member + " " + copy.isbn + " copy " + copy.no + " due " + copy.due + " late " + late + " fine " + money(fine);
472 })];
473 }
474 if (name === "NOTICES") {
475 const output = [];
476 for (const id of [...this.members.keys()].sort(lex)) {
477 const member = this.members.get(id);
478 const loans = [...member.loans.values()].sort(loanOrder);
479 for (const copy of loans.filter(copy => copy.due < this.day)) {
480 output.push("OVERDUE " + id + " " + copy.isbn + " due " + copy.due);
481 }
482 for (const copy of loans.filter(copy => copy.due >= this.day && copy.due <= this.day + 2)) {
483 output.push("DUESOON " + id + " " + copy.isbn + " due " + copy.due);
484 }
485 for (const copy of [...member.holds.values()].sort(holdOrder)) {
486 output.push("PICKUP " + id + " " + copy.isbn + " until " + copy.expiry);
487 }
488 }
489 return ["NOTICES " + output.length, ...output];
490 }
491 if (name === "HISTORY") {
492 const member = this.member(args[0]);
493 return ["HISTORY " + member.id + " " + member.history.length, ...member.history];
494 }
495 if (name === "MEMBER" || name === "MEMBERS") {
496 const members = name === "MEMBER" ? [this.member(args[0])] :
497 [...this.members.values()].sort((a, b) => lex(a.id, b.id));
498 const output = name === "MEMBERS" ? ["MEMBERS " + members.length] : [];
499 for (const member of members) {
500 const balance = this.balance(member);
501 const state = balance > 500 ? "BLOCKED" : "OK";
502 if (name === "MEMBERS") {
503 output.push(member.id + " " + member.type + " loans " + member.loans.size + " reservations " +
504 this.reservations(member) + " balance " + money(balance) + " " + state);
505 continue;
506 }
507 const rule = rules[member.type];
508 output.push("MEMBER " + member.id + " " + member.type + " " + member.name,
509 "STATUS loans " + member.loans.size + "/" + rule.limit + " reservations " +
510 this.reservations(member) + "/" + rule.reservations + " balance " + money(balance) + " " + state);
511 for (const copy of [...member.loans.values()].sort(loanOrder)) {
512 const [late, fine] = this.late(copy);
513 output.push("LOAN " + copy.isbn + " copy " + copy.no + " due " + copy.due + " renewals " + copy.renewals +
514 (late ? " late " + late + " fine " + money(fine) : ""));
515 }
516 for (const copy of [...member.holds.values()].sort(holdOrder)) {
517 output.push("HOLD " + copy.isbn + " copy " + copy.no + " until " + copy.expiry);
518 }
519 for (const title of [...this.titles.values()].sort((a, b) => lex(a.isbn, b.isbn))) {
520 const index = title.queue.indexOf(member.id);
521 if (index >= 0) {
522 output.push("QUEUE " + title.isbn + " position " + (index + 1));
523 }
524 }
525 }
526 return output;
527 }
528 if (name === "BOOK") {
529 const title = this.title(args[0]);
530 const output = ["BOOK " + title.isbn + " " + title.name, "PRICE " + money(title.price) +
531 " copies " + holdings(title) + " borrowed " + title.borrowed];
532 for (const copy of title.copies) {
533 let line = "COPY " + copy.no + " " + copy.state;
534 if (copy.state === "LOANED") {
535 line += " " + copy.member + " due " + copy.due;
536 } else if (copy.state === "HOLD") {
537 line += " " + copy.member + " until " + copy.expiry;
538 }
539 output.push(line);
540 }
541 output.push("QUEUE " + title.queue.length + (title.queue.length ? " " + title.queue.join(" ") : ""));
542 return output;
543 }
544 if (name === "INVENTORY") {
545 const states = { AVAILABLE: 0, LOANED: 0, HOLD: 0, LOST: 0, WITHDRAWN: 0 };
546 for (const copy of this.allCopies()) {
547 states[copy.state]++;
548 }
549 const total = states.AVAILABLE + states.LOANED + states.HOLD;
550 const output = ["INVENTORY titles " + this.titles.size + " copies " + total + " available " + states.AVAILABLE +
551 " loaned " + states.LOANED + " hold " + states.HOLD + " lost " + states.LOST + " withdrawn " + states.WITHDRAWN];
552 for (const title of [...this.titles.values()].sort((a, b) => lex(a.isbn, b.isbn))) {
553 const count = (state) => title.copies.filter(copy => copy.state === state).length;
554 output.push(title.isbn + " copies " + holdings(title) + " available " + count("AVAILABLE") +
555 " loaned " + count("LOANED") + " hold " + count("HOLD") + " queue " + title.queue.length);
556 }
557 return output;
558 }
559 const c = this.counts;
560 const outstanding = [...this.members.values()].reduce((sum, member) => sum + this.balance(member), 0);
561 const output = ["STATS day " + this.day + " titles " + this.titles.size + " members " + this.members.size + " closed " + this.closed.size,
562 "LOANS borrowed " + c.borrowed + " returned " + c.returned + " renewed " + c.renewed + " lost " + c.lost +
563 " active " + this.allCopies().filter(copy => copy.state === "LOANED").length,
564 "MONEY fines " + money(c.FINE) + " fees " + money(c.FEE) + " charges " + money(c.PRICE) +
565 " paid " + money(c.paid) + " waived " + money(c.waived) + " outstanding " + money(outstanding)];
566 const top = [...this.titles.values()].filter(title => title.borrowed > 0)
567 .sort((a, b) => b.borrowed - a.borrowed || lex(a.isbn, b.isbn)).slice(0, 3);
568 output.push(...top.map((title, i) => "TOP " + (i + 1) + " " + title.isbn + " " + title.borrowed + " " + title.name));
569 return output;
570 }
571}

573const lines = readFileSync(0, "utf8").split("\n");
574const library = new Library();
575const output = [];
576for (let i = 1; i <= Number(lines[0]); i++) {
577 try {
578 output.push(...library.command(lines[i]));
579 } catch (error) {
580 if (!(error instanceof Invalid)) {
581 throw error;
582 }
583 output.push("ERROR " + error.message);
584 }
585}
586process.stdout.write(output.join("\n") + "\n");
