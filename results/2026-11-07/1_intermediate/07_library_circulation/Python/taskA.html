21def amount(text):
22 whole, fraction = text.split(".")
23 return int(whole) * 100 + int(fraction)

27 return f"{value // 100}.{value % 100:02d}"

30def normalize_isbn(text):
31 raw = text.replace("-", "")
32 if (
33 len(raw) == 10
34 and raw[:9].isdigit()
35 and raw[-1] in "0123456789X"
36 ):
37 total = sum(
38 (10 - i) * (10 if c == "X" else int(c))
39 for i, c in enumerate(raw)
40 )
41 if total % 11:
42 raise Invalid("invalid isbn")
43 prefix = "978" + raw[:9]
44 checksum = sum(
45 int(c) * (1 if i % 2 == 0 else 3)
46 for i, c in enumerate(prefix)
47 )
48 return prefix + str((10 - checksum % 10) % 10)
49 if (
50 len(raw) == 13 and raw[:3] in ("978", "979")
51 and raw.isdigit()
52 ):
53 total = sum(
54 int(c) * (1 if i % 2 == 0 else 3)
55 for i, c in enumerate(raw)
56 )
57 if total % 10 == 0:
58 return raw
59 raise Invalid("invalid isbn")

81class Library:
82 def __init__(self):
83 self.day = 1
84 self.closed = set()
85 self.titles = {}
86 self.members = {}
87 self.counts = {
88 "borrowed": 0, "returned": 0, "renewed": 0, "lost": 0,
89 "FINE": 0, "FEE": 0, "PRICE": 0, "paid": 0, "waived": 0,
90 }

92 def member(self, identifier):
93 if identifier not in self.members:
94 raise Invalid("unknown member")
95 return self.members[identifier]
96
97 def title(self, token):
98 identifier = normalize_isbn(token)
99 if identifier not in self.titles:
100 raise Invalid("unknown book")
101 return self.titles[identifier]
102
103 def common(self, identifier, token):
104 member = self.member(identifier)
105 title = self.title(token)
106 return member, title

108 def all_copies(self):
109 return [
110 copy for title in self.titles.values()
111 for copy in title["copies"]
112 ]
113
114 def balance(self, member):
115 return sum(item["unpaid"] for item in member["items"])
116
117 def reservations(self, member):
118 queued = sum(
119 member["id"] in title["queue"] for title in self.titles.values()
120 )
121 return len(member["holds"]) + queued
122
123 def open_day(self, day):
124 while day in self.closed:
125 day += 1
126 return day
127
128 def late(self, copy):
129 days = max(0, self.day - copy["due"])
130 rule = RULES[self.members[copy["member"]]["type"]]
131 return days, min(days * rule["fine"], rule["cap"])
132
133 def charge(self, member, kind, title, value):
134 if not value:
135 return
136 member["items"].append({
137 "no": len(member["items"]) + 1,
138 "day": self.day,
139 "kind": kind,
140 "isbn": title["isbn"],
141 "amount": value,
142 "unpaid": value,
143 })
144 self.counts[kind] += value
145
146 def release(self, title, copy):
147 if not title["queue"]:
148 copy["state"] = "AVAILABLE"
149 copy["member"] = None
150 return []
151 identifier = title["queue"].pop(0)
152 copy["state"] = "HOLD"
153 copy["member"] = identifier
154 copy["expiry"] = self.open_day(self.day + 3)
155 self.members[identifier]["holds"][title["isbn"]] = copy
156 return [
157 f"HOLD {title['isbn']} copy {copy['no']}"
158 f" {identifier} until {copy['expiry']}"
159 ]
160
161 def end_loan(self, title, copy, result):
162 member = self.members[copy["member"]]
163 days, fine = self.late(copy)
164 del member["loans"][title["isbn"]]
165 self.charge(member, "FINE", title, fine)
166 member["history"].append(
167 f"{result} {title['isbn']} copy {copy['no']}"
168 f" from {copy['loan_day']} to {self.day}"
169 f" late {days} fine {money(fine)}"
170 )
171 return days, fine
172
173 def loss(self, title, copy, heading):
174 member = self.members[copy["member"]]
175 days, fine = self.end_loan(title, copy, "LOST")
176 member["history"][-1] += " price " + money(title["price"])
177 self.charge(member, "PRICE", title, title["price"])
178 copy["state"] = "LOST"
179 copy["member"] = None
180 self.counts["lost"] += 1
181 output = [
182 f"{heading} {member['id']} {title['isbn']} copy {copy['no']}"
183 f" late {days} fine {money(fine)} price {money(title['price'])}"
184 f" balance {money(self.balance(member))}"
185 ]
186 if not holdings(title) and title["queue"]:
187 output.extend(
188 f"DROPPED {title['isbn']} {identifier}"
189 for identifier in title["queue"]
190 )
191 title["queue"].clear()
192 return output

194 def renewal_reason(self, title, copy):
195 if copy["due"] < self.day:
196 return "loan overdue"
197 if copy["renewals"] == 2:
198 return "renewal limit"
199 if title["queue"]:
200 return "title reserved"
201 return None

203 def renew(self, member, title, copy):
204 copy["due"] = self.open_day(self.day + RULES[member["type"]]["period"])
205 copy["renewals"] += 1
206 self.counts["renewed"] += 1
207 return (
208 f"RENEW {member['id']} {title['isbn']} copy {copy['no']}"
209 f" due {copy['due']} renewals {copy['renewals']}"
210 )

212 def advance(self, day):
213 if day <= self.day:
214 raise Invalid("invalid day")
215 output = [f"DAY {day}"]
216 while self.day < day:
217 self.day += 1
218 losses = [
219 copy for copy in self.all_copies()
220 if copy["state"] == "LOANED" and copy["due"] + 365 <= self.day
221 ]
222 losses.sort(key=lambda copy: (copy["member"], copy["isbn"]))
223 for copy in losses:
224 output.extend(self.loss(self.titles[copy["isbn"]], copy, "AUTOLOST"))
225 expired = [
226 copy for copy in self.all_copies()
227 if copy["state"] == "HOLD" and copy["expiry"] < self.day
228 ]
229 for copy in sorted(expired, key=copy_order):
230 title = self.titles[copy["isbn"]]
231 member = self.members[copy["member"]]
232 fee = RULES[member["type"]]["fee"]
233 self.charge(member, "FEE", title, fee)
234 del member["holds"][copy["isbn"]]
235 output.append(
236 f"EXPIRED {copy['isbn']} copy {copy['no']}"
237 f" {member['id']} fee {money(fee)}"
238 )
239 output.extend(self.release(title, copy))
240 return output

242 def close_day(self, day):
243 if day <= self.day:
244 raise Invalid("invalid day")
245 if day in self.closed:
246 raise Invalid("already closed")
247 self.closed.add(day)
248 output = [f"CLOSED {day}"]
249 for copy in sorted(self.all_copies(), key=copy_order):
250 if copy["state"] == "LOANED" and copy["due"] == day:
251 copy["due"] = self.open_day(day)
252 output.append(
253 f"MOVED {copy['isbn']} copy {copy['no']}"
254 f" {copy['member']} due {copy['due']}"
255 )
256 elif copy["state"] == "HOLD" and copy["expiry"] == day:
257 copy["expiry"] = self.open_day(day)
258 output.append(
259 f"MOVED {copy['isbn']} copy {copy['no']}"
260 f" {copy['member']} until {copy['expiry']}"
261 )
262 return output

264 def add_book(self, args):
265 identifier = normalize_isbn(args[0])
266 name = " ".join(args[3:])
267 title = self.titles.get(identifier)
268 if title and title["name"] != name:
269 raise Invalid("title mismatch")
270 if title is None:
271 title = {
272 "isbn": identifier, "name": name, "price": amount(args[2]),
273 "copies": [], "queue": [], "borrowed": 0,
274 }
275 self.titles[identifier] = title
276 title["price"] = amount(args[2])
277 added = []
278 for _ in range(int(args[1])):
279 copy = {
280 "isbn": identifier, "no": len(title["copies"]) + 1,
281 "state": "AVAILABLE", "member": None, "loan_day": 0,
282 "due": 0, "renewals": 0, "expiry": 0,
283 }
284 title["copies"].append(copy)
285 added.append(copy)
286 output = [
287 f"ADDBOOK {identifier} added {len(added)} total {holdings(title)}"
288 ]
289 for copy in added:
290 output.extend(self.release(title, copy))
291 return output

293 def add_member(self, args):
294 identifier, kind = args[:2]
295 if identifier in self.members:
296 raise Invalid("duplicate member")
297 if kind not in RULES:
298 raise Invalid("unknown type")
299 self.members[identifier] = {
300 "id": identifier, "type": kind, "name": " ".join(args[2:]),
301 "loans": {}, "holds": {}, "items": [], "history": [],
302 }
303 return [
304 f"ADDMEMBER {identifier} {kind}"
305 f" loans {RULES[kind]['limit']} days {RULES[kind]['period']}"
306 ]

308 def circulate(self, name, args):
309 member, title = self.common(args[0], args[1])
310 identifier = title["isbn"]
311 loan = member["loans"].get(identifier)
312 hold = member["holds"].get(identifier)
313 rule = RULES[member["type"]]
314 if name == "BORROW":
315 if self.day in self.closed:
316 raise Invalid("library closed")
317 if loan is not None:
318 raise Invalid("already borrowed")
319 if self.balance(member) > 500:
320 raise Invalid("member blocked")
321 if any(copy["due"] < self.day for copy in member["loans"].values()):
322 raise Invalid("overdue loans")
323 if len(member["loans"]) >= rule["limit"]:
324 raise Invalid("loan limit")
325 copy = hold
326 if copy is None:
327 copy = next(
328 (copy for copy in title["copies"] if copy["state"] == "AVAILABLE"),
329 None,
330 )
331 if copy is None:
332 if any(copy["state"] == "HOLD" for copy in title["copies"]):
333 raise Invalid("all copies on hold")
334 raise Invalid("no copy available")
335 member["holds"].pop(identifier, None)
336 copy.update(
337 state="LOANED", member=member["id"], loan_day=self.day,
338 due=self.open_day(self.day + rule["period"]), renewals=0,
339 )
340 member["loans"][identifier] = copy
341 title["borrowed"] += 1
342 self.counts["borrowed"] += 1
343 return [
344 f"BORROW {member['id']} {identifier}"
345 f" copy {copy['no']} due {copy['due']}"
346 ]
347 if name in ("RETURN", "LOST", "RENEW"):
348 if loan is None:
349 raise Invalid("not borrowed")
350 if name == "LOST":
351 return self.loss(title, loan, "LOST")
352 if name == "RETURN":
353 days, fine = self.end_loan(title, loan, "RETURNED")
354 self.counts["returned"] += 1
355 output = [
356 f"RETURN {member['id']} {identifier} copy {loan['no']}"
357 f" late {days} fine {money(fine)}"
358 f" balance {money(self.balance(member))}"
359 ]
360 return output + self.release(title, loan)
361 if self.balance(member) > 500:
362 raise Invalid("member blocked")
363 reason = self.renewal_reason(title, loan)
364 if reason:
365 raise Invalid(reason)
366 return [self.renew(member, title, loan)]
367 queued = member["id"] in title["queue"]
368 if name == "CANCEL":
369 if not queued and hold is None:
370 raise Invalid("not reserved")
371 if queued:
372 title["queue"].remove(member["id"])
373 return [f"CANCEL {member['id']} {identifier} queue"]
374 del member["holds"][identifier]
375 output = [f"CANCEL {member['id']} {identifier} hold copy {hold['no']}"]
376 return output + self.release(title, hold)
377 if loan is not None:
378 raise Invalid("already borrowed")
379 if queued or hold is not None:
380 raise Invalid("already reserved")
381 if self.balance(member) > 500:
382 raise Invalid("member blocked")
383 if self.reservations(member) >= rule["reservations"]:
384 raise Invalid("reservation limit")
385 if not holdings(title):
386 raise Invalid("no copies")
387 if any(copy["state"] == "AVAILABLE" for copy in title["copies"]):
388 raise Invalid("copy available")
389 title["queue"].append(member["id"])
390 position = len(title["queue"])
391 days = sorted(
392 copy["due"] for copy in title["copies"] if copy["state"] == "LOANED"
393 )
394 estimate = days[position - 1] if position <= len(days) else "none"
395 return [
396 f"RESERVE {member['id']} {identifier}"
397 f" position {position} estimate {estimate}"
398 ]

400 def renew_all(self, identifier):
401 member = self.member(identifier)
402 if not member["loans"]:
403 raise Invalid("no loans")
404 if self.balance(member) > 500:
405 raise Invalid("member blocked")
406 output = []
407 renewed = 0
408 skipped = 0
409 for copy in sorted(member["loans"].values(), key=lambda copy: copy["isbn"]):
410 title = self.titles[copy["isbn"]]
411 reason = self.renewal_reason(title, copy)
412 if reason:
413 output.append(f"SKIP {copy['isbn']} {reason}")
414 skipped += 1
415 else:
416 output.append(self.renew(member, title, copy))
417 renewed += 1
418 output.append(
419 f"RENEWALL {identifier} renewed {renewed} skipped {skipped}"
420 )
421 return output

423 def charges(self, name, args):
424 member = self.member(args[0])
425 if name == "PAYFINE":
426 if not self.balance(member):
427 raise Invalid("no fines")
428 remaining = amount(args[1])
429 items = []
430 for item in member["items"]:
431 applied = min(remaining, item["unpaid"])
432 if applied:
433 item["unpaid"] -= applied
434 remaining -= applied
435 items.append(
436 f"ITEM {item['no']} {item['kind']} paid {money(applied)}"
437 f" remaining {money(item['unpaid'])}"
438 )
439 if not remaining:
440 break
441 paid = amount(args[1]) - remaining
442 self.counts["paid"] += paid
443 return [
444 f"PAYFINE {member['id']} paid {money(paid)} change {money(remaining)}"
445 f" balance {money(self.balance(member))}",
446 *items,
447 ]
448 if name == "WAIVE":
449 number = int(args[1])
450 if not 1 <= number <= len(member["items"]):
451 raise Invalid("unknown item")
452 item = member["items"][number - 1]
453 if not item["unpaid"]:
454 raise Invalid("item settled")
455 value = item["unpaid"]
456 item["unpaid"] = 0
457 self.counts["waived"] += value
458 return [
459 f"WAIVE {member['id']} item {number} {item['kind']}"
460 f" amount {money(value)} balance {money(self.balance(member))}"
461 ]
462 items = [item for item in member["items"] if item["unpaid"]]
463 output = [
464 f"FINES {member['id']} items {len(items)}"
465 f" balance {money(self.balance(member))}"
466 ]
467 output.extend(
468 f"{item['no']} day {item['day']} {item['kind']} {item['isbn']}"
469 f" amount {money(item['amount'])} remaining {money(item['unpaid'])}"
470 for item in items
471 )
472 return output

474 def member_lines(self, member):
475 value = self.balance(member)
476 state = "BLOCKED" if value > 500 else "OK"
477 rule = RULES[member["type"]]
478 output = [
479 f"MEMBER {member['id']} {member['type']} {member['name']}",
480 f"STATUS loans {len(member['loans'])}/{rule['limit']}"
481 f" reservations {self.reservations(member)}/{rule['reservations']}"
482 f" balance {money(value)} {state}",
483 ]
484 for copy in sorted(member["loans"].values(), key=loan_order):
485 days, fine = self.late(copy)
486 suffix = f" late {days} fine {money(fine)}" if days else ""
487 output.append(
488 f"LOAN {copy['isbn']} copy {copy['no']} due {copy['due']}"
489 f" renewals {copy['renewals']}{suffix}"
490 )
491 for copy in sorted(member["holds"].values(), key=hold_order):
492 output.append(
493 f"HOLD {copy['isbn']} copy {copy['no']} until {copy['expiry']}"
494 )
495 for title in sorted(self.titles.values(), key=lambda title: title["isbn"]):
496 if member["id"] in title["queue"]:
497 position = title["queue"].index(member["id"]) + 1
498 output.append(f"QUEUE {title['isbn']} position {position}")
499 return output

501 def notices(self):
502 output = []
503 for identifier, member in sorted(self.members.items()):
504 loans = sorted(member["loans"].values(), key=loan_order)
505 for copy in loans:
506 if copy["due"] < self.day:
507 output.append(
508 f"OVERDUE {identifier} {copy['isbn']} due {copy['due']}"
509 )
510 for copy in loans:
511 if self.day <= copy["due"] <= self.day + 2:
512 output.append(
513 f"DUESOON {identifier} {copy['isbn']} due {copy['due']}"
514 )
515 for copy in sorted(member["holds"].values(), key=hold_order):
516 output.append(
517 f"PICKUP {identifier} {copy['isbn']} until {copy['expiry']}"
518 )
519 return [f"NOTICES {len(output)}", *output]

521 def book_lines(self, title):
522 output = [
523 f"BOOK {title['isbn']} {title['name']}",
524 f"PRICE {money(title['price'])} copies {holdings(title)}"
525 f" borrowed {title['borrowed']}",

528 line = f"COPY {copy['no']} {copy['state']}"
529 if copy["state"] == "LOANED":

535 output.append(f"QUEUE {len(title['queue'])}" + (" " + queue if queue else ""))

538 def inventory(self):
539 states = Counter(copy["state"] for copy in self.all_copies())
540 total = states["AVAILABLE"] + states["LOANED"] + states["HOLD"]
541 output = [
542 f"INVENTORY titles {len(self.titles)} copies {total}"
543 f" available {states['AVAILABLE']} loaned {states['LOANED']}"
544 f" hold {states['HOLD']} lost {states['LOST']}"
545 f" withdrawn {states['WITHDRAWN']}"
546 ]
547 for title in sorted(self.titles.values(), key=lambda title: title["isbn"]):
548 count = Counter(copy["state"] for copy in title["copies"])
549 output.append(
550 f"{title['isbn']} copies {holdings(title)}"
551 f" available {count['AVAILABLE']} loaned {count['LOANED']}"
552 f" hold {count['HOLD']} queue {len(title['queue'])}"
553 )
554 return output

556 def statistics(self):
557 counts = self.counts
558 active = sum(copy["state"] == "LOANED" for copy in self.all_copies())
559 outstanding = sum(self.balance(member) for member in self.members.values())
560 output = [
561 f"STATS day {self.day} titles {len(self.titles)}"
562 f" members {len(self.members)} closed {len(self.closed)}",
563 f"LOANS borrowed {counts['borrowed']} returned {counts['returned']}"
564 f" renewed {counts['renewed']} lost {counts['lost']} active {active}",
565 f"MONEY fines {money(counts['FINE'])} fees {money(counts['FEE'])}"
566 f" charges {money(counts['PRICE'])} paid {money(counts['paid'])}"
567 f" waived {money(counts['waived'])} outstanding {money(outstanding)}",
568 ]
569 top = sorted(
570 (title for title in self.titles.values() if title["borrowed"]),
571 key=lambda title: (-title["borrowed"], title["isbn"]),
572 )[:3]
573 output.extend(
574 f"TOP {i} {title['isbn']} {title['borrowed']} {title['name']}"
575 for i, title in enumerate(top, 1)
576 )
577 return output

579 def inspect(self, name, args):
580 if name == "SEARCH":
581 keyword = " ".join(args).lower()
582 titles = sorted(
583 (
584 title for title in self.titles.values()
585 if keyword in title["name"].lower()
586 ),
587 key=lambda title: (title["name"].lower(), title["isbn"]),
588 )
589 output = [f"SEARCH {len(titles)}"]
590 for title in titles:
591 available = sum(
592 copy["state"] == "AVAILABLE" for copy in title["copies"]
593 )
594 output.append(
595 f"{title['isbn']} {available}/{holdings(title)} {title['name']}"
596 )
597 return output
598 if name == "OVERDUE":
599 copies = sorted(
600 (
601 copy for copy in self.all_copies()
602 if copy["state"] == "LOANED" and copy["due"] < self.day
603 ),
604 key=lambda copy: (copy["due"], copy["member"], copy["isbn"]),
605 )
606 output = [f"OVERDUE {len(copies)}"]
607 for copy in copies:
608 days, fine = self.late(copy)
609 output.append(
610 f"{copy['member']} {copy['isbn']} copy {copy['no']}"
611 f" due {copy['due']} late {days} fine {money(fine)}"
612 )
613 return output
614 if name == "NOTICES":
615 return self.notices()
616 if name == "MEMBER":
617 return self.member_lines(self.member(args[0]))
618 if name == "MEMBERS":
619 output = [f"MEMBERS {len(self.members)}"]
620 for identifier, member in sorted(self.members.items()):
621 value = self.balance(member)
622 state = "BLOCKED" if value > 500 else "OK"
623 output.append(
624 f"{identifier} {member['type']} loans {len(member['loans'])}"
625 f" reservations {self.reservations(member)}"
626 f" balance {money(value)} {state}"
627 )
628 return output
629 if name == "BOOK":
630 return self.book_lines(self.title(args[0]))
631 if name == "INVENTORY":
632 return self.inventory()
633 if name == "HISTORY":
634 member = self.member(args[0])
635 return [
636 f"HISTORY {member['id']} {len(member['history'])}",
637 *member["history"],
638 ]
639 return self.statistics()

641 def command(self, line):
642 name, *args = line.split()
643 if name == "DAY":
644 return self.advance(int(args[0]))
645 if name == "CLOSED":
646 return self.close_day(int(args[0]))
647 if name == "ADDBOOK":
648 return self.add_book(args)
649 if name == "ADDMEMBER":
650 return self.add_member(args)
651 if name == "WITHDRAW":
652 title = self.title(args[0])
653 number = int(args[1])
654 if not 1 <= number <= len(title["copies"]):
655 raise Invalid("unknown copy")
656 copy = title["copies"][number - 1]
657 if copy["state"] != "AVAILABLE":
658 raise Invalid("copy not available")
659 copy["state"] = "WITHDRAWN"
660 return [
661 f"WITHDRAW {title['isbn']} copy {copy['no']}"
662 f" total {holdings(title)}"
663 ]
664 if name in ("BORROW", "RETURN", "RENEW", "RESERVE", "CANCEL", "LOST"):
665 return self.circulate(name, args)
666 if name == "RENEWALL":
667 return self.renew_all(args[0])
668 if name in ("PAYFINE", "WAIVE", "FINES"):
669 return self.charges(name, args)
670 return self.inspect(name, args)
