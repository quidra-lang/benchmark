3const Allocator = std.mem.Allocator;
4const Error = error{BadCommand} || Allocator.Error || std.Io.Writer.Error;
5fn integer(text: []const u8) i64 {
6 return std.fmt.parseInt(i64, text, 10) catch unreachable;
7}
8
9fn amount(text: []const u8) i64 {
10 const dot = std.mem.findScalar(u8, text, '.').?;
11 return integer(text[0..dot]) * 100 + integer(text[dot + 1 ..]);
12}
13
14fn money(gpa: Allocator, n: i64) Allocator.Error![]const u8 {
15 const value: u64 = @intCast(n);
16 return std.fmt.allocPrint(gpa, "{d}.{d:0>2}", .{ value / 100, value % 100 });
17}
18
19fn split(gpa: Allocator, text: []const u8) Allocator.Error![][]const u8 {
20 var result: std.ArrayList([]const u8) = .empty;
21 var it = std.mem.splitScalar(u8, text, ' ');
22 while (it.next()) |part|
23 try result.append(gpa, part);
24 return result.items;
25}

27fn normalizeISBN(gpa: Allocator, token: []const u8) (error{InvalidISBN} || Allocator.Error)![]const u8 {
28 const s = try std.mem.replaceOwned(u8, gpa, token, "-", "");
29 if (s.len == 10) {
30 for (s[0..9]) |ch| {
31 if (!std.ascii.isDigit(ch))
32 return error.InvalidISBN;
33 }
34 if (!std.ascii.isDigit(s[9]) and s[9] != 'X')
35 return error.InvalidISBN;
36 var total: i64 = 0;
37 for (s[0..10], 0..) |ch, i|
38 total += @as(i64, @intCast(10 - i)) * @as(i64, if (ch == 'X') 10 else ch - '0');
39 if (@mod(total, 11) != 0)
40 return error.InvalidISBN;
41 const result = try gpa.alloc(u8, 13);
42 @memcpy(result[0..3], "978");
43 @memcpy(result[3..12], s[0..9]);
44 var sum: usize = 0;
45 for (result[0..12], 0..) |ch, i|
46 sum += @as(usize, if (i % 2 == 0) 1 else 3) * (ch - '0');
47 result[12] = @as(u8, @intCast((10 - sum % 10) % 10)) + '0';
48 return result;
49 }
50 if (s.len == 13 and (std.mem.eql(u8, s[0..3], "978") or std.mem.eql(u8, s[0..3], "979"))) {
51 var sum: usize = 0;
52 for (s[0..13], 0..) |ch, i| {
53 if (!std.ascii.isDigit(ch))
54 return error.InvalidISBN;
55 sum += @as(usize, if (i % 2 == 0) 1 else 3) * (ch - '0');
56 }
57 if (sum % 10 == 0)
58 return gpa.dupe(u8, s[0..13]);
59 }
60 return error.InvalidISBN;
61}

69const Member = struct {
70 id: []const u8,
71 kind: MemberType,
72 name: []const u8,
73 items: std.ArrayList(Charge) = .empty,
74 history: std.ArrayList([]const u8) = .empty,
75
76 fn loanLimit(self: *const Member) usize {
77 return if (self.kind == .STUDENT) 3 else 10;
78 }
79
80 fn period(self: *const Member) i64 {
81 return if (self.kind == .STUDENT) 14 else 28;
82 }
83
84 fn rate(self: *const Member) i64 {
85 return if (self.kind == .STUDENT) 25 else 10;
86 }
87
88 fn cap(self: *const Member) i64 {
89 return if (self.kind == .STUDENT) 1000 else 2000;
90 }
91
92 fn reservationLimit(self: *const Member) usize {
93 return if (self.kind == .STUDENT) 2 else 5;
94 }
95
96 fn fee(self: *const Member) i64 {
97 return if (self.kind == .STUDENT) 50 else 100;
98 }
99
100 fn balance(self: *const Member) i64 {
101 var total: i64 = 0;
102 for (self.items.items) |item|
103 total += item.unpaid;
104 return total;
105 }
106
107 fn blocked(self: *const Member) bool {
108 return self.balance() > 500;
109 }

112const CopyState = enum { AVAILABLE, LOANED, HOLD, LOST, WITHDRAWN };
113
114const Title = struct {
115 isbn: []const u8,
116 name: []const u8,
117 folded: []const u8,
118 price: i64,
119 borrowed: usize = 0,
120 copies: std.ArrayList(*Copy) = .empty,
121 queue: std.ArrayList(*Member) = .empty,
122
123 fn countState(self: *const Title, state: CopyState) usize {
124 var count: usize = 0;
125 for (self.copies.items) |c| {
126 if (c.state == state)
127 count += 1;
128 }
129 return count;
130 }
131
132 fn holdings(self: *const Title) usize {
133 return self.countState(.AVAILABLE) + self.countState(.LOANED) + self.countState(.HOLD);
134 }
135};

137const Copy = struct {
138 title: *Title,
139 number: usize,
140 state: CopyState = .AVAILABLE,
141 member: ?*Member = null,
142 loan_day: i64 = 0,
143 due: i64 = 0,
144 renewals: usize = 0,
145 expiry: i64 = 0,

148const LoanEnd = struct { late: i64, fine: i64 };
149
150fn titleCopy(m: *Member, t: *Title, state: CopyState) ?*Copy {
151 for (t.copies.items) |c| {
152 if (c.state == state and c.member == m)
153 return c;
154 }
155 return null;

158const Library = struct {
159 gpa: Allocator,
160 day: i64 = 1,
161 titles: std.StringArrayHashMapUnmanaged(*Title) = .empty,
162 members: std.StringArrayHashMapUnmanaged(*Member) = .empty,
163 closed: std.AutoHashMapUnmanaged(i64, void) = .empty,
164 borrowed: usize = 0,
165 returned: usize = 0,
166 renewed: usize = 0,
167 lost: usize = 0,
168 charged: [3]i64 = .{ 0, 0, 0 },
169 paid: i64 = 0,
170 waived: i64 = 0,
171 error_message: []const u8 = "",
172
173 fn fail(self: *Library, message: []const u8) error{BadCommand} {
174 self.error_message = message;
175 return error.BadCommand;
176 }
177
178 fn normalized(self: *Library, token: []const u8) Error![]const u8 {
179 return normalizeISBN(self.gpa, token) catch |err| {
180 if (err == error.InvalidISBN)
181 return self.fail("invalid isbn");
182 return err;
183 };
184 }

186 fn findMember(self: *Library, id: []const u8) Error!*Member {
187 return self.members.get(id) orelse self.fail("unknown member");
188 }
189
190 fn findTitle(self: *Library, token: []const u8) Error!*Title {
191 return self.titles.get(try self.normalized(token)) orelse self.fail("unknown book");
192 }

194 fn titlesSorted(self: *const Library) Allocator.Error![]*Title {
195 const result = try self.gpa.dupe(*Title, self.titles.values());
196 std.mem.sort(*Title, result, {}, titleLess);
197 return result;
198 }
199
200 fn membersSorted(self: *const Library) Allocator.Error![]*Member {
201 const result = try self.gpa.dupe(*Member, self.members.values());
202 std.mem.sort(*Member, result, {}, memberLess);
203 return result;
204 }

206 fn copiesInState(self: *const Library, state: CopyState, owner: ?*Member) Allocator.Error![]*Copy {
207 var result: std.ArrayList(*Copy) = .empty;
208 for (self.titles.values()) |t| for (t.copies.items) |c| {
209 if (c.state == state and (owner == null or c.member == owner))
210 try result.append(self.gpa, c);
211 };
212 return result.items;
213 }
214
215 fn reservations(self: *const Library, m: *Member) usize {
216 var count: usize = 0;
217 for (self.titles.values()) |t| {
218 if (std.mem.findScalar(*Member, t.queue.items, m) != null)
219 count += 1;
220 for (t.copies.items) |c| {
221 if (c.state == .HOLD and c.member == m)
222 count += 1;
223 }
224 }
225 return count;
226 }
227
228 fn openDay(self: *const Library, d: i64) i64 {
229 var current = d;
230 while (self.closed.contains(current))
231 current += 1;
232 return current;
233 }
234
235 fn daysLate(self: *const Library, c: *const Copy) i64 {
236 return @max(0, self.day - c.due);
237 }
238
239 fn lateFine(self: *const Library, c: *const Copy) i64 {
240 return @min(self.daysLate(c) * c.member.?.rate(), c.member.?.cap());
241 }
242
243 fn addCharge(self: *Library, m: *Member, t: *Title, kind: ChargeKind, value: i64) Allocator.Error!void {
244 if (value == 0)
245 return;
246 try m.items.append(self.gpa, .{ .day = self.day, .kind = kind, .isbn = t.isbn, .value = value, .unpaid = value });
247 self.charged[@backingInt(kind)] += value;
248 }
249
250 fn release(self: *Library, c: *Copy, out: *std.Io.Writer) Error!void {
251 if (c.title.queue.items.len == 0) {
252 c.state = .AVAILABLE;
253 c.member = null;
254 return;
255 }
256 const m = c.title.queue.orderedRemove(0);
257 c.state = .HOLD;
258 c.member = m;
259 c.expiry = self.openDay(self.day + 3);
260 try out.print("HOLD {s} copy {d} {s} until {d}\n", .{ c.title.isbn, c.number, m.id, c.expiry });
261 }
262
263 fn endLoan(self: *Library, c: *Copy, result: []const u8) Error!LoanEnd {
264 const m = c.member.?;
265 const late = self.daysLate(c);
266 const value = self.lateFine(c);
267 try self.addCharge(m, c.title, .FINE, value);
268 try m.history.append(self.gpa, try std.fmt.allocPrint(self.gpa, "{s} {s} copy {d} from {d} to {d} late {d} fine {s}", .{ result, c.title.isbn, c.number, c.loan_day, self.day, late, try money(self.gpa, value) }));
269 c.state = .AVAILABLE;
270 c.member = null;
271 return .{ .late = late, .fine = value };
272 }
273
274 fn processLoss(self: *Library, c: *Copy, heading: []const u8, out: *std.Io.Writer) Error!void {
275 const m = c.member.?;
276 const t = c.title;
277 const result = try self.endLoan(c, "LOST");
278 const last = m.history.items.len - 1;
279 m.history.items[last] = try std.fmt.allocPrint(self.gpa, "{s} price {s}", .{ m.history.items[last], try money(self.gpa, t.price) });
280 try self.addCharge(m, t, .PRICE, t.price);
281 c.state = .LOST;
282 self.lost += 1;
283 try out.print("{s} {s} {s} copy {d} late {d} fine {s} price {s} balance {s}\n", .{ heading, m.id, t.isbn, c.number, result.late, try money(self.gpa, result.fine), try money(self.gpa, t.price), try money(self.gpa, m.balance()) });
284 if (t.holdings() == 0) {
285 for (t.queue.items) |member|
286 try out.print("DROPPED {s} {s}\n", .{ t.isbn, member.id });
287 t.queue.clearRetainingCapacity();
288 }
289 }

291 fn renewalProblem(self: *const Library, c: *const Copy) ?[]const u8 {
292 if (c.due < self.day)
293 return "loan overdue";
294 if (c.renewals == 2)
295 return "renewal limit";
296 if (c.title.queue.items.len > 0)
297 return "title reserved";
298 return null;
299 }
300
301 fn performRenew(self: *Library, c: *Copy, out: *std.Io.Writer) Error!void {
302 c.due = self.openDay(self.day + c.member.?.period());
303 c.renewals += 1;
304 self.renewed += 1;
305 try out.print("RENEW {s} {s} copy {d} due {d} renewals {d}\n", .{ c.member.?.id, c.title.isbn, c.number, c.due, c.renewals });
306 }

308 fn execute(self: *Library, line: []const u8, out: *std.Io.Writer) Error!void {
309 const a = try split(self.gpa, line);
310 const op = a[0];
311 if (std.mem.eql(u8, op, "DAY")) {
312 const target = integer(a[1]);
313 if (target <= self.day)
314 return self.fail("invalid day");
315 const ordered_titles = try self.titlesSorted();
316 const loan_order = try self.copiesInState(.LOANED, null);
317 std.mem.sort(*Copy, loan_order, {}, automaticLess);
318 try out.print("DAY {d}\n", .{target});
319 while (self.day < target) {
320 self.day += 1;
321 var automatic: std.ArrayList(*Copy) = .empty;
322 for (loan_order) |c| {
323 if (c.state == .LOANED and c.due + 365 <= self.day)
324 try automatic.append(self.gpa, c);
325 }
326 for (automatic.items) |c|
327 try self.processLoss(c, "AUTOLOST", out);
328 var expired: std.ArrayList(*Copy) = .empty;
329 for (ordered_titles) |t| for (t.copies.items) |c| {
330 if (c.state == .HOLD and c.expiry < self.day)
331 try expired.append(self.gpa, c);
332 };
333 for (expired.items) |c| {
334 const m = c.member.?;
335 try self.addCharge(m, c.title, .FEE, m.fee());
336 try out.print("EXPIRED {s} copy {d} {s} fee {s}\n", .{ c.title.isbn, c.number, m.id, try money(self.gpa, m.fee()) });
337 try self.release(c, out);
338 }
339 }
340 } else if (std.mem.eql(u8, op, "CLOSED")) {
341 const target = integer(a[1]);
342 if (target <= self.day)
343 return self.fail("invalid day");
344 if (self.closed.contains(target))
345 return self.fail("already closed");
346 try self.closed.put(self.gpa, target, {});
347 try out.print("CLOSED {d}\n", .{target});
348 for (try self.titlesSorted()) |t| for (t.copies.items) |c| {
349 if (c.state == .LOANED and c.due == target) {
350 c.due = self.openDay(target);
351 try out.print("MOVED {s} copy {d} {s} due {d}\n", .{ t.isbn, c.number, c.member.?.id, c.due });
352 } else if (c.state == .HOLD and c.expiry == target) {
353 c.expiry = self.openDay(target);
354 try out.print("MOVED {s} copy {d} {s} until {d}\n", .{ t.isbn, c.number, c.member.?.id, c.expiry });
355 }
356 };
357 } else if (std.mem.eql(u8, op, "ADDBOOK")) {
358 const isbn = try self.normalized(a[1]);
359 const name = try std.mem.join(self.gpa, " ", a[4..]);
360 const existing = self.titles.get(isbn);
361 if (existing) |t| {
362 if (!std.mem.eql(u8, t.name, name))
363 return self.fail("title mismatch");
364 }
365 const t = existing orelse blk: {
366 const title = try self.gpa.create(Title);
367 title.* = .{ .isbn = isbn, .name = name, .folded = try std.ascii.allocLowerString(self.gpa, name), .price = amount(a[3]) };
368 try self.titles.put(self.gpa, isbn, title);
369 break :blk title;
370 };
371 t.price = amount(a[3]);
372 const count: usize = @intCast(integer(a[2]));
373 const first = t.copies.items.len;
374 for (0..count) |_| {
375 const c = try self.gpa.create(Copy);
376 c.* = .{ .title = t, .number = t.copies.items.len + 1 };
377 try t.copies.append(self.gpa, c);
378 }
379 try out.print("ADDBOOK {s} added {d} total {d}\n", .{ isbn, count, t.holdings() });
380 for (t.copies.items[first..]) |c|
381 try self.release(c, out);
382 } else if (std.mem.eql(u8, op, "ADDMEMBER")) {
383 if (self.members.contains(a[1]))
384 return self.fail("duplicate member");
385 const kind = std.meta.stringToEnum(MemberType, a[2]) orelse return self.fail("unknown type");
386 const m = try self.gpa.create(Member);
387 m.* = .{ .id = a[1], .kind = kind, .name = try std.mem.join(self.gpa, " ", a[3..]) };
388 try self.members.put(self.gpa, m.id, m);
389 try out.print("ADDMEMBER {s} {s} loans {d} days {d}\n", .{ m.id, @tagName(m.kind), m.loanLimit(), m.period() });
390 } else if (std.mem.eql(u8, op, "WITHDRAW")) {
391 const t = try self.findTitle(a[1]);
392 const n = integer(a[2]);
393 if (n < 1 or n > @as(i64, @intCast(t.copies.items.len)))
394 return self.fail("unknown copy");
395 const c = t.copies.items[@intCast(n - 1)];
396 if (c.state != .AVAILABLE)
397 return self.fail("copy not available");
398 c.state = .WITHDRAWN;
399 try out.print("WITHDRAW {s} copy {d} total {d}\n", .{ t.isbn, n, t.holdings() });
400 } else if (std.mem.eql(u8, op, "BORROW") or std.mem.eql(u8, op, "RETURN") or std.mem.eql(u8, op, "RENEW") or std.mem.eql(u8, op, "RESERVE") or std.mem.eql(u8, op, "CANCEL") or std.mem.eql(u8, op, "LOST")) {
401 const m = try self.findMember(a[1]);
402 const t = try self.findTitle(a[2]);
403 const current = titleCopy(m, t, .LOANED);
404 const held = titleCopy(m, t, .HOLD);
405 if (std.mem.eql(u8, op, "BORROW")) {
406 if (self.closed.contains(self.day))
407 return self.fail("library closed");
408 if (current != null)
409 return self.fail("already borrowed");
410 if (m.blocked())
411 return self.fail("member blocked");
412 const loans = try self.copiesInState(.LOANED, m);
413 for (loans) |c| {
414 if (c.due < self.day)
415 return self.fail("overdue loans");
416 }
417 if (loans.len >= m.loanLimit())
418 return self.fail("loan limit");
419 var selected = held;
420 if (selected == null) for (t.copies.items) |c| {
421 if (c.state == .AVAILABLE) {
422 selected = c;
423 break;
424 }
425 };
426 const c = selected orelse return self.fail(if (t.countState(.HOLD) > 0) "all copies on hold" else "no copy available");
427 c.state = .LOANED;
428 c.member = m;
429 c.loan_day = self.day;
430 c.due = self.openDay(self.day + m.period());
431 c.renewals = 0;
432 t.borrowed += 1;
433 self.borrowed += 1;
434 try out.print("BORROW {s} {s} copy {d} due {d}\n", .{ m.id, t.isbn, c.number, c.due });
435 } else if (std.mem.eql(u8, op, "RETURN") or std.mem.eql(u8, op, "LOST")) {
436 const c = current orelse return self.fail("not borrowed");
437 if (std.mem.eql(u8, op, "LOST"))
438 try self.processLoss(c, "LOST", out)
439 else {
440 const result = try self.endLoan(c, "RETURNED");
441 self.returned += 1;
442 try out.print("RETURN {s} {s} copy {d} late {d} fine {s} balance {s}\n", .{ m.id, t.isbn, c.number, result.late, try money(self.gpa, result.fine), try money(self.gpa, m.balance()) });
443 try self.release(c, out);
444 }
445 } else if (std.mem.eql(u8, op, "RENEW")) {
446 const c = current orelse return self.fail("not borrowed");
447 if (m.blocked())
448 return self.fail("member blocked");
449 if (self.renewalProblem(c)) |reason|
450 return self.fail(reason);
451 try self.performRenew(c, out);
452 } else if (std.mem.eql(u8, op, "RESERVE")) {
453 if (current != null)
454 return self.fail("already borrowed");
455 if (std.mem.findScalar(*Member, t.queue.items, m) != null or held != null)
456 return self.fail("already reserved");
457 if (m.blocked())
458 return self.fail("member blocked");
459 if (self.reservations(m) >= m.reservationLimit())
460 return self.fail("reservation limit");
461 if (t.holdings() == 0)
462 return self.fail("no copies");
463 if (t.countState(.AVAILABLE) > 0)
464 return self.fail("copy available");
465 try t.queue.append(self.gpa, m);
466 const p = t.queue.items.len;
467 var due: std.ArrayList(i64) = .empty;
468 for (t.copies.items) |c| {
469 if (c.state == .LOANED)
470 try due.append(self.gpa, c.due);
471 }
472 std.mem.sort(i64, due.items, {}, std.sort.asc(i64));
473 try out.print("RESERVE {s} {s} position {d} estimate ", .{ m.id, t.isbn, p });
474 if (p <= due.items.len)
475 try out.print("{d}\n", .{due.items[p - 1]})
476 else
477 try out.writeAll("none\n");
478 } else {
479 const queued = std.mem.findScalar(*Member, t.queue.items, m);
480 if (queued == null and held == null)
481 return self.fail("not reserved");
482 if (queued) |i| {
483 _ = t.queue.orderedRemove(i);
484 try out.print("CANCEL {s} {s} queue\n", .{ m.id, t.isbn });
485 } else {
486 const c = held.?;
487 try out.print("CANCEL {s} {s} hold copy {d}\n", .{ m.id, t.isbn, c.number });
488 try self.release(c, out);
489 }
490 }
491 } else if (std.mem.eql(u8, op, "RENEWALL")) {
492 const m = try self.findMember(a[1]);
493 const loans = try self.copiesInState(.LOANED, m);
494 if (loans.len == 0)
495 return self.fail("no loans");
496 if (m.blocked())
497 return self.fail("member blocked");
498 std.mem.sort(*Copy, loans, {}, copyISBNLess);
499 var success: usize = 0;
500 var skipped: usize = 0;
501 for (loans) |c| {
502 if (self.renewalProblem(c)) |reason| {
503 try out.print("SKIP {s} {s}\n", .{ c.title.isbn, reason });
504 skipped += 1;
505 } else {
506 try self.performRenew(c, out);
507 success += 1;
508 }
509 }
510 try out.print("RENEWALL {s} renewed {d} skipped {d}\n", .{ m.id, success, skipped });
511 } else if (std.mem.eql(u8, op, "PAYFINE")) {
512 const m = try self.findMember(a[1]);
513 if (m.balance() == 0)
514 return self.fail("no fines");
515 const Applied = struct { item: usize, value: i64 };
516 var applied: std.ArrayList(Applied) = .empty;
517 const total = amount(a[2]);
518 var remaining = total;
519 for (m.items.items, 0..) |*item, i| {
520 if (remaining == 0)
521 break;
522 const value = @min(item.unpaid, remaining);
523 if (value == 0)
524 continue;
525 item.unpaid -= value;
526 remaining -= value;
527 try applied.append(self.gpa, .{ .item = i, .value = value });
528 }
529 self.paid += total - remaining;
530 try out.print("PAYFINE {s} paid {s} change {s} balance {s}\n", .{ m.id, try money(self.gpa, total - remaining), try money(self.gpa, remaining), try money(self.gpa, m.balance()) });
531 for (applied.items) |entry| {
532 const item = m.items.items[entry.item];
533 try out.print("ITEM {d} {s} paid {s} remaining {s}\n", .{ entry.item + 1, @tagName(item.kind), try money(self.gpa, entry.value), try money(self.gpa, item.unpaid) });
534 }
535 } else if (std.mem.eql(u8, op, "WAIVE")) {
536 const m = try self.findMember(a[1]);
537 const n = integer(a[2]);
538 if (n < 1 or n > @as(i64, @intCast(m.items.items.len)))
539 return self.fail("unknown item");
540 const item = &m.items.items[@intCast(n - 1)];
541 if (item.unpaid == 0)
542 return self.fail("item settled");
543 const value = item.unpaid;
544 item.unpaid = 0;
545 self.waived += value;
546 try out.print("WAIVE {s} item {d} {s} amount {s} balance {s}\n", .{ m.id, n, @tagName(item.kind), try money(self.gpa, value), try money(self.gpa, m.balance()) });
547 } else if (std.mem.eql(u8, op, "FINES")) {
548 const m = try self.findMember(a[1]);
549 var count: usize = 0;
550 for (m.items.items) |item| {
551 if (item.unpaid > 0)
552 count += 1;
553 }
554 try out.print("FINES {s} items {d} balance {s}\n", .{ m.id, count, try money(self.gpa, m.balance()) });
555 for (m.items.items, 0..) |item, i| {
556 if (item.unpaid > 0)
557 try out.print("{d} day {d} {s} {s} amount {s} remaining {s}\n", .{ i + 1, item.day, @tagName(item.kind), item.isbn, try money(self.gpa, item.value), try money(self.gpa, item.unpaid) });
558 }
559 } else if (std.mem.eql(u8, op, "SEARCH")) {
560 const keyword = try std.ascii.allocLowerString(self.gpa, try std.mem.join(self.gpa, " ", a[1..]));
561 var selected: std.ArrayList(*Title) = .empty;
562 for (try self.titlesSorted()) |t| {
563 if (std.mem.find(u8, t.folded, keyword) != null)
564 try selected.append(self.gpa, t);
565 }
566 std.mem.sort(*Title, selected.items, {}, searchLess);
567 try out.print("SEARCH {d}\n", .{selected.items.len});
568 for (selected.items) |t|
569 try out.print("{s} {d}/{d} {s}\n", .{ t.isbn, t.countState(.AVAILABLE), t.holdings(), t.name });
570 } else if (std.mem.eql(u8, op, "OVERDUE")) {
571 var copies: std.ArrayList(*Copy) = .empty;
572 for (try self.copiesInState(.LOANED, null)) |c| {
573 if (c.due < self.day)
574 try copies.append(self.gpa, c);
575 }
576 std.mem.sort(*Copy, copies.items, {}, overdueLess);
577 try out.print("OVERDUE {d}\n", .{copies.items.len});
578 for (copies.items) |c|
579 try out.print("{s} {s} copy {d} due {d} late {d} fine {s}\n", .{ c.member.?.id, c.title.isbn, c.number, c.due, self.daysLate(c), try money(self.gpa, self.lateFine(c)) });
580 } else if (std.mem.eql(u8, op, "NOTICES")) {
581 var lines: std.ArrayList([]const u8) = .empty;
582 for (try self.membersSorted()) |m| {
583 const loans = try self.copiesInState(.LOANED, m);
584 std.mem.sort(*Copy, loans, false, dueLess);
585 for (loans) |c| {
586 if (c.due < self.day)
587 try lines.append(self.gpa, try std.fmt.allocPrint(self.gpa, "OVERDUE {s} {s} due {d}", .{ m.id, c.title.isbn, c.due }));
588 }
589 for (loans) |c| {
590 if (c.due >= self.day and c.due <= self.day + 2)
591 try lines.append(self.gpa, try std.fmt.allocPrint(self.gpa, "DUESOON {s} {s} due {d}", .{ m.id, c.title.isbn, c.due }));
592 }
593 const holds = try self.copiesInState(.HOLD, m);
594 std.mem.sort(*Copy, holds, true, dueLess);
595 for (holds) |c|
596 try lines.append(self.gpa, try std.fmt.allocPrint(self.gpa, "PICKUP {s} {s} until {d}", .{ m.id, c.title.isbn, c.expiry }));
597 }
598 try out.print("NOTICES {d}\n", .{lines.items.len});
599 for (lines.items) |text|
600 try out.print("{s}\n", .{text});
601 } else if (std.mem.eql(u8, op, "MEMBER") or std.mem.eql(u8, op, "MEMBERS")) {
602 const single = std.mem.eql(u8, op, "MEMBER");
603 const ms = if (single) try self.gpa.dupe(*Member, &.{try self.findMember(a[1])}) else try self.membersSorted();
604 if (!single)
605 try out.print("MEMBERS {d}\n", .{ms.len});
606 for (ms) |m| {
607 const loans = try self.copiesInState(.LOANED, m);
608 std.mem.sort(*Copy, loans, false, dueLess);
609 const status: []const u8 = if (m.blocked()) "BLOCKED" else "OK";
610 if (!single) {
611 try out.print("{s} {s} loans {d} reservations {d} balance {s} {s}\n", .{ m.id, @tagName(m.kind), loans.len, self.reservations(m), try money(self.gpa, m.balance()), status });
612 continue;
613 }
614 try out.print("MEMBER {s} {s} {s}\nSTATUS loans {d}/{d} reservations {d}/{d} balance {s} {s}\n", .{ m.id, @tagName(m.kind), m.name, loans.len, m.loanLimit(), self.reservations(m), m.reservationLimit(), try money(self.gpa, m.balance()), status });
615 for (loans) |c| {
616 try out.print("LOAN {s} copy {d} due {d} renewals {d}", .{ c.title.isbn, c.number, c.due, c.renewals });
617 if (c.due < self.day)
618 try out.print(" late {d} fine {s}", .{ self.daysLate(c), try money(self.gpa, self.lateFine(c)) });
619 try out.writeByte('\n');
620 }
621 const holds = try self.copiesInState(.HOLD, m);
622 std.mem.sort(*Copy, holds, true, dueLess);
623 for (holds) |c|
624 try out.print("HOLD {s} copy {d} until {d}\n", .{ c.title.isbn, c.number, c.expiry });
625 for (try self.titlesSorted()) |t| {
626 if (std.mem.findScalar(*Member, t.queue.items, m)) |i|
627 try out.print("QUEUE {s} position {d}\n", .{ t.isbn, i + 1 });
628 }
629 }
630 } else if (std.mem.eql(u8, op, "BOOK")) {
631 const t = try self.findTitle(a[1]);
632 try out.print("BOOK {s} {s}\nPRICE {s} copies {d} borrowed {d}\n", .{ t.isbn, t.name, try money(self.gpa, t.price), t.holdings(), t.borrowed });
633 for (t.copies.items) |c| {
634 try out.print("COPY {d} {s}", .{ c.number, @tagName(c.state) });
635 if (c.state == .LOANED)
636 try out.print(" {s} due {d}", .{ c.member.?.id, c.due })
637 else if (c.state == .HOLD) try out.print(" {s} until {d}", .{ c.member.?.id, c.expiry });
638 try out.writeByte('\n');
639 }
640 try out.print("QUEUE {d}", .{t.queue.items.len});
641 for (t.queue.items) |m|
642 try out.print(" {s}", .{m.id});
643 try out.writeByte('\n');
644 } else if (std.mem.eql(u8, op, "INVENTORY")) {
645 var counts = [_]usize{ 0, 0, 0, 0, 0 };
646 for (self.titles.values()) |t| for (t.copies.items) |c| {
647 counts[@backingInt(c.state)] += 1;
648 };
649 try out.print("INVENTORY titles {d} copies {d} available {d} loaned {d} hold {d} lost {d} withdrawn {d}\n", .{ self.titles.values().len, counts[0] + counts[1] + counts[2], counts[0], counts[1], counts[2], counts[3], counts[4] });
650 for (try self.titlesSorted()) |t|
651 try out.print("{s} copies {d} available {d} loaned {d} hold {d} queue {d}\n", .{ t.isbn, t.holdings(), t.countState(.AVAILABLE), t.countState(.LOANED), t.countState(.HOLD), t.queue.items.len });
652 } else if (std.mem.eql(u8, op, "HISTORY")) {
653 const m = try self.findMember(a[1]);
654 try out.print("HISTORY {s} {d}\n", .{ m.id, m.history.items.len });
655 for (m.history.items) |text|
656 try out.print("{s}\n", .{text});
657 } else if (std.mem.eql(u8, op, "STATS")) {
658 try out.print("STATS day {d} titles {d} members {d} closed {d}\n", .{ self.day, self.titles.values().len, self.members.values().len, self.closed.count() });
659 try out.print("LOANS borrowed {d} returned {d} renewed {d} lost {d} active {d}\n", .{ self.borrowed, self.returned, self.renewed, self.lost, (try self.copiesInState(.LOANED, null)).len });
660 var outstanding: i64 = 0;
661 for (self.members.values()) |m|
662 outstanding += m.balance();
663 try out.print("MONEY fines {s} fees {s} charges {s} paid {s} waived {s} outstanding {s}\n", .{ try money(self.gpa, self.charged[0]), try money(self.gpa, self.charged[1]), try money(self.gpa, self.charged[2]), try money(self.gpa, self.paid), try money(self.gpa, self.waived), try money(self.gpa, outstanding) });
664 var top: std.ArrayList(*Title) = .empty;
665 for (try self.titlesSorted()) |t| {
666 if (t.borrowed > 0)
667 try top.append(self.gpa, t);
668 }
669 std.mem.sort(*Title, top.items, {}, topLess);
670 for (top.items[0..@min(3, top.items.len)], 0..) |t, i|
671 try out.print("TOP {d} {s} {d} {s}\n", .{ i + 1, t.isbn, t.borrowed, t.name });
672 }
673 }

676fn titleLess(_: void, a: *Title, b: *Title) bool {
677 return std.mem.order(u8, a.isbn, b.isbn) == .lt;
678}
679fn memberLess(_: void, a: *Member, b: *Member) bool {
680 return std.mem.order(u8, a.id, b.id) == .lt;
681}
682fn copyISBNLess(_: void, a: *Copy, b: *Copy) bool {
683 return std.mem.order(u8, a.title.isbn, b.title.isbn) == .lt;
684}
685fn dueLess(expiry: bool, a: *Copy, b: *Copy) bool {
686 const x = if (expiry) a.expiry else a.due;
687 const y = if (expiry) b.expiry else b.due;
688 return x < y or (x == y and copyISBNLess({}, a, b));
689}
690fn automaticLess(_: void, a: *Copy, b: *Copy) bool {
691 const order = std.mem.order(u8, a.member.?.id, b.member.?.id);
692 return order == .lt or (order == .eq and copyISBNLess({}, a, b));
693}
694fn overdueLess(_: void, a: *Copy, b: *Copy) bool {
695 return a.due < b.due or (a.due == b.due and automaticLess({}, a, b));
696}
697fn searchLess(_: void, a: *Title, b: *Title) bool {
698 const order = std.mem.order(u8, a.folded, b.folded);
699 return order == .lt or (order == .eq and titleLess({}, a, b));
700}
701fn topLess(_: void, a: *Title, b: *Title) bool {
702 return a.borrowed > b.borrowed or (a.borrowed == b.borrowed and titleLess({}, a, b));
703}

705pub fn main(init: std.process.Init) !void {
706 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
707 defer arena.deinit();
708 const gpa = arena.allocator();
709 var input_buffer: [4096]u8 = undefined;
710 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
711 var output_buffer: [8192]u8 = undefined;
712 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
713 var library = Library{ .gpa = gpa };
714 const count: usize = @intCast(integer((try input.interface.takeDelimiter('\n')).?));
715 for (0..count) |_| {
716 const line = try gpa.dupe(u8, (try input.interface.takeDelimiter('\n')).?);
717 var draft: std.Io.Writer.Allocating = .init(gpa);
718 library.execute(line, &draft.writer) catch |err| {
719 if (err == error.BadCommand) {
720 try output.interface.print("ERROR {s}\n", .{library.error_message});
721 continue;
722 }
723 return err;
724 };
725 try output.interface.writeAll(draft.writer.buffered());
726 }
727 try output.interface.flush();
728}
