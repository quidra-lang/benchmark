1use std::collections::{BTreeMap, BTreeSet};
2use std::io::{self, Read};
3
4type Result<T> = std::result::Result<T, &'static str>;
5macro_rules! reject {
6 ($condition:expr, $reason:expr) => {
7 if $condition {
8 return Err($reason);
9 }
10 };
11}
12const AVAILABLE: usize = 0;
13const LOANED: usize = 1;
14const HOLD: usize = 2;
15const LOST: usize = 3;
16const WITHDRAWN: usize = 4;
17const STATES: [&str; 5] = ["AVAILABLE", "LOANED", "HOLD", "LOST", "WITHDRAWN"];
18const KINDS: [&str; 3] = ["FINE", "FEE", "PRICE"];
19#[derive(Clone, Copy, PartialEq, Eq)]
20struct Ref {
21 title: usize,
22 copy: usize,
23}
24#[derive(Clone, Copy, Default)]
25struct Copy {
26 state: usize,
27 member: usize,
28 from: i32,
29 due: i32,
30 renewals: i32,
31 expiry: i32,
32}
33#[derive(Default)]
34struct Title {
35 isbn: String,
36 name: String,
37 folded: String,
38 price: i32,
39 borrowed: i32,
40 counts: [i32; 5],
41 copies: Vec<Copy>,
42 queue: Vec<usize>,
43}
44struct Charge {
45 day: i32,
46 kind: usize,
47 title: usize,
48 amount: i32,
49 unpaid: i32,
50}
51#[derive(Default)]
52struct Member {
53 id: String,
54 kind: String,
55 name: String,
56 limit: usize,
57 period: i32,
58 rate: i32,

69#[derive(Default)]
70struct Library {
71 day: i32,
72 borrowed: i32,
73 returned: i32,
74 renewed: i32,
75 lost: i32,
76 paid: i32,
77 waived: i32,
78 charged: [i32; 3],
79 closed: BTreeSet<i32>,
80 title_ids: BTreeMap<String, usize>,
81 member_ids: BTreeMap<String, usize>,
82 titles: Vec<Title>,
83 members: Vec<Member>,
84 output: Vec<String>,
85}
86fn number(s: &str) -> i32 {
87 s.parse().unwrap()
88}
89fn money(n: i32) -> String {
90 format!("{}.{:02}", n / 100, n % 100)
91}
92fn cents(s: &str) -> i32 {
93 let (whole, fraction) = s.split_once('.').unwrap();
94 number(whole) * 100 + number(fraction)
95}
96fn normalize(input: &str) -> Option<String> {
97 let mut s = input.replace('-', "");
98 if s.len() == 10 {
99 let mut sum = 0;
100 for (i, c) in s.bytes().enumerate() {
101 if !c.is_ascii_digit() && !(i == 9 && c == b'X') {
102 return None;
103 }
104 let value = if c == b'X' { 10 } else { i32::from(c - b'0') };
105 sum += (10 - i as i32) * value;
106 }
107 if sum % 11 != 0 {
108 return None;
109 }
110 s = "978".to_owned() + &s[..9];
111 sum = s
112 .bytes()
113 .enumerate()
114 .map(|(i, c)| (if i % 2 == 0 { 1 } else { 3 }) * i32::from(c - b'0'))
115 .sum();
116 s.push((b'0' + ((10 - sum % 10) % 10) as u8) as char);
117 return Some(s);
118 }
119 if s.len() != 13
120 || !(s.starts_with("978") || s.starts_with("979"))
121 || !s.bytes().all(|c| c.is_ascii_digit())
122 {
123 return None;
124 }
125 let sum: i32 = s
126 .bytes()
127 .enumerate()
128 .map(|(i, c)| (if i % 2 == 0 { 1 } else { 3 }) * i32::from(c - b'0'))
129 .sum();
130 if sum % 10 == 0 { Some(s) } else { None }
131}
132fn remove_ref(refs: &mut Vec<Ref>, r: Ref) {
133 let i = refs.iter().position(|&other| other == r).unwrap();
134 refs.remove(i);
135}
136fn find_ref(refs: &[Ref], title: usize) -> Option<usize> {
137 refs.iter().find(|r| r.title == title).map(|r| r.copy)
138}
139impl Library {
140 fn new() -> Self {
141 Self {
142 day: 1,
143 ..Self::default()
144 }
145 }
146 fn open(&self, mut d: i32) -> i32 {
147 while self.closed.contains(&d) {
148 d += 1;
149 }
150 d
151 }
152 fn cp(&self, r: Ref) -> &Copy {
153 &self.titles[r.title].copies[r.copy]
154 }
155 fn holdings(&self, t: usize) -> i32 {
156 self.titles[t].counts[..3].iter().sum()
157 }
158 fn state(&mut self, r: Ref, next: usize) {
159 let t = &mut self.titles[r.title];
160 t.counts[t.copies[r.copy].state] -= 1;
161 t.copies[r.copy].state = next;
162 t.counts[next] += 1;
163 }
164 fn member(&self, id: &str) -> Result<usize> {
165 self.member_ids.get(id).copied().ok_or("unknown member")
166 }
167 fn title(&self, input: &str) -> Result<usize> {
168 let isbn = normalize(input).ok_or("invalid isbn")?;
169 self.title_ids.get(&isbn).copied().ok_or("unknown book")
170 }
171 fn late(&self, r: Ref) -> i32 {
172 (self.day - self.cp(r).due).max(0)
173 }
174 fn fine(&self, r: Ref) -> i32 {
175 let m = &self.members[self.cp(r).member];
176 (self.late(r) * m.rate).min(m.cap)
177 }
178 fn charge(&mut self, m: usize, kind: usize, title: usize, amount: i32) {
179 if amount == 0 {
180 return;
181 }
182 self.members[m].charges.push(Charge {
183 day: self.day,
184 kind,
185 title,
186 amount,
187 unpaid: amount,
188 });
189 self.members[m].balance += amount;
190 self.charged[kind] += amount;
191 }
192 fn release(&mut self, r: Ref) {
193 if self.titles[r.title].queue.is_empty() {
194 self.state(r, AVAILABLE);
195 return;
196 }
197 let m = self.titles[r.title].queue.remove(0);
198 let expiry = self.open(self.day + 3);
199 self.state(r, HOLD);
200 self.titles[r.title].copies[r.copy].member = m;
201 self.titles[r.title].copies[r.copy].expiry = expiry;
202 self.members[m].holds.push(r);
203 self.output.push(format!(
204 "HOLD {} copy {} {} until {}",
205 self.titles[r.title].isbn,
206 r.copy + 1,
207 self.members[m].id,
208 expiry
209 ));
210 }
211 fn drop_queue(&mut self, t: usize) {
212 if self.holdings(t) != 0 {
213 return;
214 }
215 let queue = std::mem::take(&mut self.titles[t].queue);
216 for m in queue {
217 self.members[m].reservations -= 1;
218 self.output.push(format!(
219 "DROPPED {} {}",
220 self.titles[t].isbn, self.members[m].id
221 ));
222 }
223 }
224 fn end_loan(&mut self, r: Ref, loss: bool, heading: &str) {
225 let c = *self.cp(r);
226 let (m, late, fine) = (c.member, self.late(r), self.fine(r));
227 remove_ref(&mut self.members[m].loans, r);
228 self.charge(m, 0, r.title, fine);
229 let mut history = format!(
230 "{} {} copy {} from {} to {} late {} fine {}",
231 if loss { "LOST" } else { "RETURNED" },
232 self.titles[r.title].isbn,
233 r.copy + 1,
234 c.from,
235 self.day,
236 late,
237 money(fine)
238 );
239 let mut result = format!(
240 "{} {} {} copy {} late {} fine {}",
241 heading,
242 self.members[m].id,
243 self.titles[r.title].isbn,
244 r.copy + 1,
245 late,
246 money(fine)
247 );
248 if loss {
249 let price = self.titles[r.title].price;
250 history.push_str(&format!(" price {}", money(price)));
251 result.push_str(&format!(" price {}", money(price)));
252 self.charge(m, 2, r.title, price);
253 self.state(r, LOST);
254 self.lost += 1;
255 } else {
256 self.returned += 1;
257 }
258 self.members[m].history.push(history);
259 self.output.push(format!(
260 "{} balance {}",
261 result,
262 money(self.members[m].balance)
263 ));
264 if loss {
265 self.drop_queue(r.title);
266 } else {
267 self.release(r);
268 }
269 }
270 fn renewal_reason(&self, r: Ref) -> Option<&'static str> {
271 if self.late(r) != 0 {
272 return Some("loan overdue");
273 }
274 if self.cp(r).renewals == 2 {
275 return Some("renewal limit");
276 }
277 if !self.titles[r.title].queue.is_empty() {
278 return Some("title reserved");
279 }
280 None
281 }
282 fn renew(&mut self, r: Ref) {
283 let m = self.cp(r).member;
284 let due = self.open(self.day + self.members[m].period);
285 let c = &mut self.titles[r.title].copies[r.copy];
286 c.due = due;
287 c.renewals += 1;
288 self.renewed += 1;
289 self.output.push(format!(
290 "RENEW {} {} copy {} due {} renewals {}",
291 self.members[m].id,
292 self.titles[r.title].isbn,
293 r.copy + 1,
294 due,
295 self.cp(r).renewals
296 ));
297 }
298 fn ordered(&self, refs: &[Ref], holds: bool) -> Vec<Ref> {
299 let mut result = refs.to_vec();
300 result.sort_by(|&a, &b| {
301 let x = if holds {
302 self.cp(a).expiry
303 } else {
304 self.cp(a).due
305 };
306 let y = if holds {
307 self.cp(b).expiry
308 } else {
309 self.cp(b).due
310 };
311 (x, &self.titles[a.title].isbn).cmp(&(y, &self.titles[b.title].isbn))
312 });
313 result
314 }
315 fn execute(&mut self, a: &[&str]) -> Result<()> {
316 let cmd = a[0];
317 match cmd {
318 "DAY" => {
319 let target = number(a[1]);
320 reject!(target <= self.day, "invalid day");
321 self.output.push(format!("DAY {}", target));
322 while self.day < target {
323 self.day += 1;
324 let mut losses = Vec::new();
325 for &m in self.member_ids.values() {
326 for &r in &self.members[m].loans {
327 if self.cp(r).due + 365 <= self.day {
328 losses.push(r);
329 }
330 }
331 }
332 for r in losses {
333 self.end_loan(r, true, "AUTOLOST");
334 }
335 let mut expiries = Vec::new();
336 for &t in self.title_ids.values() {
337 for (i, c) in self.titles[t].copies.iter().enumerate() {
338 if c.state == HOLD && c.expiry < self.day {
339 expiries.push(Ref { title: t, copy: i });
340 }
341 }
342 }
343 for r in expiries {
344 let m = self.cp(r).member;
345 self.charge(m, 1, r.title, self.members[m].fee);
346 remove_ref(&mut self.members[m].holds, r);
347 self.members[m].reservations -= 1;
348 self.output.push(format!(
349 "EXPIRED {} copy {} {} fee {}",
350 self.titles[r.title].isbn,
351 r.copy + 1,
352 self.members[m].id,
353 money(self.members[m].fee)
354 ));
355 self.release(r);
356 }
357 }
358 }
359 "CLOSED" => {
360 let d = number(a[1]);
361 reject!(d <= self.day, "invalid day");
362 reject!(self.closed.contains(&d), "already closed");
363 self.closed.insert(d);
364 self.output.push(format!("CLOSED {}", d));
365 let next = self.open(d);
366 for &t in self.title_ids.values() {
367 let book = &mut self.titles[t];
368 let isbn = &book.isbn;
369 for (i, c) in book.copies.iter_mut().enumerate() {
370 if c.state == LOANED && c.due == d {
371 c.due = next;
372 self.output.push(format!(
373 "MOVED {} copy {} {} due {}",
374 isbn,
375 i + 1,
376 self.members[c.member].id,
377 next
378 ));
379 } else if c.state == HOLD && c.expiry == d {
380 c.expiry = next;
381 self.output.push(format!(
382 "MOVED {} copy {} {} until {}",
383 isbn,
384 i + 1,
385 self.members[c.member].id,
386 next
387 ));
388 }
389 }
390 }
391 }
392 "ADDBOOK" => {
393 let isbn = normalize(a[1]).ok_or("invalid isbn")?;
394 let name = a[4..].join(" ");
395 let t = if let Some(&id) = self.title_ids.get(&isbn) {
396 reject!(self.titles[id].name != name, "title mismatch");
397 id
398 } else {
399 let id = self.titles.len();
400 self.titles.push(Title {
401 isbn: isbn.clone(),
402 folded: name.to_ascii_lowercase(),
403 name,
404 ..Title::default()
405 });
406 self.title_ids.insert(isbn.clone(), id);
407 id
408 };
409 self.titles[t].price = cents(a[3]);
410 let (start, added) = (self.titles[t].copies.len(), number(a[2]) as usize);
411 self.titles[t].copies.resize(start + added, Copy::default());
412 self.titles[t].counts[AVAILABLE] += added as i32;
413 self.output.push(format!(
414 "ADDBOOK {} added {} total {}",
415 isbn,
416 added,
417 self.holdings(t)
418 ));
419 for copy in start..start + added {
420 self.release(Ref { title: t, copy });
421 }
422 }
423 "ADDMEMBER" => {
424 reject!(self.member_ids.contains_key(a[1]), "duplicate member");
425 reject!(a[2] != "STUDENT" && a[2] != "STAFF", "unknown type");
426 let staff = a[2] == "STAFF";
427 let m = Member {
428 id: a[1].to_owned(),
429 kind: a[2].to_owned(),
430 name: a[3..].join(" "),
431 limit: if staff { 10 } else { 3 },
432 period: if staff { 28 } else { 14 },
433 rate: if staff { 10 } else { 25 },
434 cap: if staff { 2000 } else { 1000 },
435 reserve_limit: if staff { 5 } else { 2 },
436 fee: if staff { 100 } else { 50 },
437 ..Member::default()
438 };
439 self.output.push(format!(
440 "ADDMEMBER {} {} loans {} days {}",
441 m.id, m.kind, m.limit, m.period
442 ));
443 self.member_ids.insert(m.id.clone(), self.members.len());
444 self.members.push(m);
445 }
446 "WITHDRAW" => {
447 let t = self.title(a[1])?;
448 let i = number(a[2]) - 1;
449 reject!(
450 i < 0 || i as usize >= self.titles[t].copies.len(),
451 "unknown copy"
452 );
453 let r = Ref {
454 title: t,
455 copy: i as usize,
456 };
457 reject!(self.cp(r).state != AVAILABLE, "copy not available");
458 self.state(r, WITHDRAWN);
459 self.output.push(format!(
460 "WITHDRAW {} copy {} total {}",
461 self.titles[t].isbn,
462 i + 1,
463 self.holdings(t)
464 ));
465 }
466 "BORROW" | "RETURN" | "RENEW" | "RESERVE" | "CANCEL" | "LOST" => {
467 let m = self.member(a[1])?;
468 let t = self.title(a[2])?;
469 let loan = find_ref(&self.members[m].loans, t);
470 let held = find_ref(&self.members[m].holds, t);
471 let queued = self.titles[t].queue.iter().position(|&id| id == m);
472 match cmd {
473 "BORROW" => {
474 reject!(self.closed.contains(&self.day), "library closed");
475 reject!(loan.is_some(), "already borrowed");
476 reject!(self.members[m].balance > 500, "member blocked");
477 reject!(
478 self.members[m].loans.iter().any(|&r| self.late(r) != 0),
479 "overdue loans"
480 );
481 reject!(
482 self.members[m].loans.len() >= self.members[m].limit,
483 "loan limit"
484 );
485 let available = self.titles[t]
486 .copies
487 .iter()
488 .position(|c| c.state == AVAILABLE);
489 let i = held
490 .or(available)
491 .ok_or(if self.titles[t].counts[HOLD] != 0 {
492 "all copies on hold"
493 } else {
494 "no copy available"
495 })?;
496 let r = Ref { title: t, copy: i };
497 if held.is_some() {
498 remove_ref(&mut self.members[m].holds, r);
499 self.members[m].reservations -= 1;
500 }
501 self.state(r, LOANED);
502 self.titles[t].copies[i] = Copy {
503 state: LOANED,
504 member: m,
505 from: self.day,
506 due: self.open(self.day + self.members[m].period),
507 renewals: 0,
508 expiry: 0,
509 };
510 self.members[m].loans.push(r);
511 let titles = &self.titles;
512 self.members[m]
513 .loans
514 .sort_by(|a, b| titles[a.title].isbn.cmp(&titles[b.title].isbn));
515 self.borrowed += 1;
516 self.titles[t].borrowed += 1;
517 self.output.push(format!(
518 "BORROW {} {} copy {} due {}",
519 self.members[m].id,
520 self.titles[t].isbn,
521 i + 1,
522 self.cp(r).due
523 ));
524 }
525 "RETURN" | "LOST" | "RENEW" => {
526 let r = Ref {
527 title: t,
528 copy: loan.ok_or("not borrowed")?,
529 };
530 if cmd == "RENEW" {
531 reject!(self.members[m].balance > 500, "member blocked");
532 if let Some(reason) = self.renewal_reason(r) {
533 return Err(reason);
534 }
535 self.renew(r);
536 } else {
537 self.end_loan(r, cmd == "LOST", cmd);
538 }
539 }
540 "RESERVE" => {
541 reject!(loan.is_some(), "already borrowed");
542 reject!(queued.is_some() || held.is_some(), "already reserved");
543 reject!(self.members[m].balance > 500, "member blocked");
544 reject!(
545 self.members[m].reservations >= self.members[m].reserve_limit,
546 "reservation limit"
547 );
548 reject!(self.holdings(t) == 0, "no copies");
549 reject!(self.titles[t].counts[AVAILABLE] > 0, "copy available");
550 self.titles[t].queue.push(m);
551 self.members[m].reservations += 1;
552 let mut dues: Vec<i32> = self.titles[t]
553 .copies
554 .iter()
555 .filter(|c| c.state == LOANED)
556 .map(|c| c.due)
557 .collect();
558 dues.sort();
559 let p = self.titles[t].queue.len();
560 let estimate = dues
561 .get(p - 1)
562 .map(|d| d.to_string())
563 .unwrap_or_else(|| "none".to_owned());
564 self.output.push(format!(
565 "RESERVE {} {} position {} estimate {}",
566 self.members[m].id, self.titles[t].isbn, p, estimate
567 ));
568 }
569 _ => {
570 reject!(queued.is_none() && held.is_none(), "not reserved");
571 self.members[m].reservations -= 1;
572 if let Some(q) = queued {
573 self.titles[t].queue.remove(q);
574 self.output.push(format!(
575 "CANCEL {} {} queue",
576 self.members[m].id, self.titles[t].isbn
577 ));
578 } else {
579 let r = Ref {
580 title: t,
581 copy: held.unwrap(),
582 };
583 remove_ref(&mut self.members[m].holds, r);
584 self.output.push(format!(
585 "CANCEL {} {} hold copy {}",
586 self.members[m].id,
587 self.titles[t].isbn,
588 r.copy + 1
589 ));
590 self.release(r);
591 }
592 }
593 }
594 }
595 "RENEWALL" => {
596 let m = self.member(a[1])?;
597 reject!(self.members[m].loans.is_empty(), "no loans");
598 reject!(self.members[m].balance > 500, "member blocked");
599 let (mut good, mut skipped) = (0, 0);
600 for r in self.members[m].loans.clone() {
601 if let Some(reason) = self.renewal_reason(r) {
602 self.output
603 .push(format!("SKIP {} {}", self.titles[r.title].isbn, reason));
604 skipped += 1;
605 } else {
606 self.renew(r);
607 good += 1;
608 }
609 }
610 self.output.push(format!(
611 "RENEWALL {} renewed {} skipped {}",
612 self.members[m].id, good, skipped
613 ));
614 }
615 "PAYFINE" | "WAIVE" | "FINES" => {
616 let m = self.member(a[1])?;
617 let p = &mut self.members[m];
618 match cmd {
619 "PAYFINE" => {
620 reject!(p.balance == 0, "no fines");
621 let amount = cents(a[2]);
622 let mut remaining = amount;
623 let mut details = Vec::new();
624 for (i, c) in p.charges.iter_mut().enumerate() {
625 let applied = remaining.min(c.unpaid);
626 if applied == 0 {
627 continue;
628 }
629 c.unpaid -= applied;
630 remaining -= applied;
631 details.push(format!(
632 "ITEM {} {} paid {} remaining {}",
633 i + 1,
634 KINDS[c.kind],
635 money(applied),
636 money(c.unpaid)
637 ));
638 if remaining == 0 {
639 break;
640 }
641 }
642 p.balance -= amount - remaining;
643 self.paid += amount - remaining;
644 self.output.push(format!(
645 "PAYFINE {} paid {} change {} balance {}",
646 p.id,
647 money(amount - remaining),
648 money(remaining),
649 money(p.balance)
650 ));
651 self.output.extend(details);
652 }
653 "WAIVE" => {
654 let i = number(a[2]) - 1;
655 reject!(i < 0 || i as usize >= p.charges.len(), "unknown item");
656 let c = &mut p.charges[i as usize];
657 reject!(c.unpaid == 0, "item settled");
658 let amount = c.unpaid;
659 c.unpaid = 0;
660 p.balance -= amount;
661 self.waived += amount;
662 self.output.push(format!(
663 "WAIVE {} item {} {} amount {} balance {}",
664 p.id,
665 i + 1,
666 KINDS[c.kind],
667 money(amount),
668 money(p.balance)
669 ));
670 }
671 _ => {
672 self.output.push(format!(
673 "FINES {} items {} balance {}",
674 p.id,
675 p.charges.iter().filter(|c| c.unpaid > 0).count(),
676 money(p.balance)
677 ));
678 for (i, c) in p.charges.iter().enumerate() {
679 if c.unpaid > 0 {
680 self.output.push(format!(
681 "{} day {} {} {} amount {} remaining {}",
682 i + 1,
683 c.day,
684 KINDS[c.kind],
685 self.titles[c.title].isbn,
686 money(c.amount),
687 money(c.unpaid)
688 ));
689 }
690 }
691 }
692 }
693 }
694 "SEARCH" => {
695 let keyword = a[1..].join(" ").to_ascii_lowercase();
696 let mut found: Vec<usize> = (0..self.titles.len())
697 .filter(|&i| self.titles[i].folded.contains(&keyword))
698 .collect();
699 found.sort_by(|&a, &b| {
700 (&self.titles[a].folded, &self.titles[a].isbn)
701 .cmp(&(&self.titles[b].folded, &self.titles[b].isbn))
702 });
703 self.output.push(format!("SEARCH {}", found.len()));
704 for t in found {
705 self.output.push(format!(
706 "{} {}/{} {}",
707 self.titles[t].isbn,
708 self.titles[t].counts[AVAILABLE],
709 self.holdings(t),
710 self.titles[t].name
711 ));
712 }
713 }
714 "OVERDUE" => {
715 let mut refs: Vec<Ref> = self
716 .members
717 .iter()
718 .flat_map(|m| m.loans.iter().copied())
719 .filter(|&r| self.late(r) > 0)
720 .collect();
721 refs.sort_by(|&a, &b| {
722 (
723 self.cp(a).due,
724 &self.members[self.cp(a).member].id,
725 &self.titles[a.title].isbn,
726 )
727 .cmp(&(
728 self.cp(b).due,
729 &self.members[self.cp(b).member].id,
730 &self.titles[b.title].isbn,
731 ))
732 });
733 self.output.push(format!("OVERDUE {}", refs.len()));
734 for r in refs {
735 self.output.push(format!(
736 "{} {} copy {} due {} late {} fine {}",
737 self.members[self.cp(r).member].id,
738 self.titles[r.title].isbn,
739 r.copy + 1,
740 self.cp(r).due,
741 self.late(r),
742 money(self.fine(r))
743 ));
744 }
745 }
746 "NOTICES" => {
747 let mut notices = Vec::new();
748 for &id in self.member_ids.values() {
749 let m = &self.members[id];
750 for r in self.ordered(&m.loans, false) {
751 if self.cp(r).due <= self.day + 2 {
752 notices.push(format!(
753 "{} {} {} due {}",
754 if self.late(r) > 0 {
755 "OVERDUE"
756 } else {
757 "DUESOON"
758 },
759 m.id,
760 self.titles[r.title].isbn,
761 self.cp(r).due
762 ));
763 }
764 }
765 for r in self.ordered(&m.holds, true) {
766 notices.push(format!(
767 "PICKUP {} {} until {}",
768 m.id,
769 self.titles[r.title].isbn,
770 self.cp(r).expiry
771 ));
772 }
773 }
774 self.output.push(format!("NOTICES {}", notices.len()));
775 self.output.extend(notices);
776 }
777 "MEMBER" | "MEMBERS" | "HISTORY" => {
778 let target = if cmd == "MEMBERS" {
779 None
780 } else {
781 Some(self.member(a[1])?)
782 };
783 if cmd == "HISTORY" {
784 let m = &self.members[target.unwrap()];
785 self.output
786 .push(format!("HISTORY {} {}", m.id, m.history.len()));
787 self.output.extend(m.history.iter().cloned());
788 return Ok(());
789 }
790 if cmd == "MEMBERS" {
791 self.output.push(format!("MEMBERS {}", self.members.len()));
792 }
793 for &id in self.member_ids.values() {
794 if target.is_some() && target != Some(id) {
795 continue;
796 }
797 let m = &self.members[id];
798 let status = if m.balance > 500 { "BLOCKED" } else { "OK" };
799 if cmd == "MEMBERS" {
800 self.output.push(format!(
801 "{} {} loans {} reservations {} balance {} {}",
802 m.id,
803 m.kind,
804 m.loans.len(),
805 m.reservations,
806 money(m.balance),
807 status
808 ));
809 continue;
810 }
811 self.output
812 .push(format!("MEMBER {} {} {}", m.id, m.kind, m.name));
813 self.output.push(format!(
814 "STATUS loans {}/{} reservations {}/{} balance {} {}",
815 m.loans.len(),
816 m.limit,
817 m.reservations,
818 m.reserve_limit,
819 money(m.balance),
820 status
821 ));
822 for r in self.ordered(&m.loans, false) {
823 let mut s = format!(
824 "LOAN {} copy {} due {} renewals {}",
825 self.titles[r.title].isbn,
826 r.copy + 1,
827 self.cp(r).due,
828 self.cp(r).renewals
829 );
830 if self.late(r) > 0 {
831 s.push_str(&format!(
832 " late {} fine {}",
833 self.late(r),
834 money(self.fine(r))
835 ));
836 }
837 self.output.push(s);
838 }
839 for r in self.ordered(&m.holds, true) {
840 self.output.push(format!(
841 "HOLD {} copy {} until {}",
842 self.titles[r.title].isbn,
843 r.copy + 1,
844 self.cp(r).expiry
845 ));
846 }
847 for (isbn, &t) in &self.title_ids {
848 if let Some(q) = self.titles[t].queue.iter().position(|&m| m == id) {
849 self.output
850 .push(format!("QUEUE {} position {}", isbn, q + 1));
851 }
852 }
853 }
854 }
855 "BOOK" => {
856 let t = self.title(a[1])?;
857 let book = &self.titles[t];
858 self.output
859 .push(format!("BOOK {} {}", book.isbn, book.name));
860 self.output.push(format!(
861 "PRICE {} copies {} borrowed {}",
862 money(book.price),
863 self.holdings(t),
864 book.borrowed
865 ));
866 for (i, c) in book.copies.iter().enumerate() {
867 let mut s = format!("COPY {} {}", i + 1, STATES[c.state]);
868 if c.state == LOANED || c.state == HOLD {
869 s.push_str(&format!(
870 " {} {} {}",
871 self.members[c.member].id,
872 if c.state == LOANED { "due" } else { "until" },
873 if c.state == LOANED { c.due } else { c.expiry }
874 ));
875 }
876 self.output.push(s);
877 }
878 let mut queue = format!("QUEUE {}", book.queue.len());
879 for &m in &book.queue {
880 queue.push_str(&format!(" {}", self.members[m].id));
881 }
882 self.output.push(queue);
883 }
884 "INVENTORY" => {
885 let mut totals = [0; 5];
886 for t in &self.titles {
887 for (i, n) in t.counts.iter().enumerate() {
888 totals[i] += n;
889 }
890 }
891 self.output.push(format!("INVENTORY titles {} copies {} available {} loaned {} hold {} lost {} withdrawn {}", self.titles.len(), totals[0] + totals[1] + totals[2], totals[0], totals[1], totals[2], totals[3], totals[4]));
892 for &id in self.title_ids.values() {
893 let t = &self.titles[id];
894 self.output.push(format!(
895 "{} copies {} available {} loaned {} hold {} queue {}",
896 t.isbn,
897 self.holdings(id),
898 t.counts[0],
899 t.counts[1],
900 t.counts[2],
901 t.queue.len()
902 ));
903 }
904 }
905 "STATS" => {
906 let active: usize = self.members.iter().map(|m| m.loans.len()).sum();
907 let outstanding: i32 = self.members.iter().map(|m| m.balance).sum();
908 self.output.push(format!(
909 "STATS day {} titles {} members {} closed {}",
910 self.day,
911 self.titles.len(),
912 self.members.len(),
913 self.closed.len()
914 ));
915 self.output.push(format!(
916 "LOANS borrowed {} returned {} renewed {} lost {} active {}",
917 self.borrowed, self.returned, self.renewed, self.lost, active
918 ));
919 self.output.push(format!(
920 "MONEY fines {} fees {} charges {} paid {} waived {} outstanding {}",
921 money(self.charged[0]),
922 money(self.charged[1]),
923 money(self.charged[2]),
924 money(self.paid),
925 money(self.waived),
926 money(outstanding)
927 ));
928 let mut top: Vec<usize> = (0..self.titles.len())
929 .filter(|&t| self.titles[t].borrowed > 0)
930 .collect();
931 top.sort_by(|&a, &b| {
932 (-self.titles[a].borrowed, &self.titles[a].isbn)
933 .cmp(&(-self.titles[b].borrowed, &self.titles[b].isbn))
934 });
935 for (i, &t) in top.iter().take(3).enumerate() {
936 self.output.push(format!(
937 "TOP {} {} {} {}",
938 i + 1,
939 self.titles[t].isbn,
940 self.titles[t].borrowed,
941 self.titles[t].name
942 ));
943 }
944 }
945 _ => unreachable!(),
946 }
947 Ok(())
948 }
949}
950fn main() {
951 let mut input = String::new();
952 io::stdin().read_to_string(&mut input).unwrap();
953 let mut lines = input.lines();
954 let count: usize = lines.next().unwrap().parse().unwrap();
955 let mut library = Library::new();
956 let mut output = String::new();
957 for _ in 0..count {
958 let arguments: Vec<&str> = lines.next().unwrap().split(' ').collect();
959 library.output.clear();
960 if let Err(reason) = library.execute(&arguments) {
961 library.output.clear();
962 library.output.push(format!("ERROR {}", reason));
963 }
964 for line in &library.output {
965 output.push_str(line);
966 output.push('\n');
967 }
968 }
969 print!("{}", output);
970}
