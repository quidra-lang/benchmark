4class LibraryError implements Exception {
5 final String message;
6 LibraryError(this.message);
7}
8Never fail(String message) => throw LibraryError(message);
9int amount(String text) {
10 final p = text.split('.');
11 return int.parse(p[0]) * 100 + int.parse(p[1]);
12}
13
14String money(int n) => '${n ~/ 100}.${(n % 100).toString().padLeft(2, '0')}';
15String normalizeISBN(String text) {
16 final s = text.replaceAll('-', '');
17 bool digit(int c) => c >= 48 && c <= 57;
18 int checksum(String digits) => digits.codeUnits.indexed.fold(0,
19 (sum, entry) => sum + (entry.$1.isEven ? 1 : 3) * (entry.$2 - 48));
20 if (s.length == 10 && s.substring(0, 9).codeUnits.every(digit) && (digit(s.codeUnitAt(9)) || s[9] == 'X')) {
21 final total = s.codeUnits.indexed.fold(0,
22 (sum, entry) => sum + (10 - entry.$1) * (entry.$2 == 88 ? 10 : entry.$2 - 48));
23 if (total % 11 != 0)
24 fail('invalid isbn');
25 final prefix = '978${s.substring(0, 9)}';
26 return '$prefix${(10 - checksum(prefix) % 10) % 10}';
27 }
28 if (s.length == 13 && s.codeUnits.every(digit) && (s.startsWith('978') || s.startsWith('979'))) {
29 if (checksum(s) % 10 == 0)
30 return s;
31 }
32 fail('invalid isbn');
33}

35class Charge {
36 final int day, amount;
37 final String kind, isbn;
38 int unpaid;
39 Charge(this.day, this.kind, this.isbn, this.amount): unpaid = amount;
40}

42class Member {
43 final String id, type, name;
44 final items = <Charge>[], history = <String>[];
45 Member(this.id, this.type, this.name);
46 bool get student => type == 'STUDENT';
47 int get loanLimit => student ? 3 : 10;
48 int get period => student ? 14 : 28;
49 int get rate => student ? 25 : 10;
50 int get cap => student ? 1000 : 2000;
51 int get reservationLimit => student ? 2 : 5;
52 int get fee => student ? 50 : 100;
53 int get balance => items.fold(0, (s, i) => s + i.unpaid);
54 bool get blocked => balance > 500;
55}

57class Title {
58 final String isbn, name, folded;
59 int price, borrowed = 0;
60 final copies = <Copy>[], queue = <Member>[];
61 Title(this.isbn, this.name, this.price): folded = name.toLowerCase();
62 int get holdings => copies.where((c) => ['AVAILABLE', 'LOANED', 'HOLD'].contains(c.state)).length;
63 int count(String state) => copies.where((c) => c.state == state).length;
64}

66class Copy {
67 final Title title;
68 final int number;
69 String state = 'AVAILABLE';
70 Member? member;
71 int loanDay = 0, due = 0, renewals = 0, expiry = 0;
72 Copy(this.title, this.number);
73}

75class Library {
76 int day = 1, borrowed = 0, returned = 0, renewed = 0, lost = 0, paid = 0, waived = 0;
77 final charged = <String, int>{ 'FINE': 0, 'FEE': 0, 'PRICE': 0 };
78 final titles = <String, Title>{}, members = <String, Member>{}, closed = <int>{};
79 List<Title> books() => titles.values.toList()..sort((a, b) => a.isbn.compareTo(b.isbn));
80 List<Member> people() => members.values.toList()..sort((a, b) => a.id.compareTo(b.id));
81 List<Copy> copies() => titles.values.expand((t) => t.copies).toList();
82 List<Copy> loans([Member? m]) => titles.values.expand((t) => t.copies).where((c) => c.state == 'LOANED'
83 && (m == null || c.member == m)).toList();
84 List<Copy> holds([Member? m]) => titles.values.expand((t) => t.copies).where((c) => c.state == 'HOLD'
85 && (m == null || c.member == m)).toList();
86 int reservations(Member m) => holds(m).length + titles.values.where((t) => t.queue.contains(m)).length;
87 Member member(String id) {
88 final m = members[id];
89 if (m == null)
90 fail('unknown member');
91 return m;
92 }
93 Title book(String token) {
94 final isbn = normalizeISBN(token), t = titles[isbn];
95 if (t == null)
96 fail('unknown book');
97 return t;
98 }
99 Copy? loan(Member m, Title t) => t.copies.where((c) => c.state == 'LOANED' && c.member == m).firstOrNull;
100 Copy? hold(Member m, Title t) => t.copies.where((c) => c.state == 'HOLD' && c.member == m).firstOrNull;
101
102 int openDay(int t) {
103 while (closed.contains(t))
104 t++;
105 return t;
106 }
107
108 int daysLate(Copy c) => max(0, day - c.due);
109 int fine(Copy c) => min(daysLate(c) * c.member!.rate, c.member!.cap);
110
111 void charge(Member m, Title t, String kind, int value) {
112 if (value == 0)
113 return;
114 m.items.add(Charge(day, kind, t.isbn, value));
115 charged[kind] = charged[kind]! + value;
116 }
117
118 void release(Copy c, StringBuffer out) {
119 if (c.title.queue.isEmpty) {
120 c.state = 'AVAILABLE';
121 c.member = null;
122 return;
123 }
124 final m = c.title.queue.removeAt(0);
125 c.state = 'HOLD';
126 c.member = m;
127 c.expiry = openDay(day + 3);
128 out.writeln('HOLD ${c.title.isbn} copy ${c.number} ${m.id} until ${c.expiry}');
129 }
130
131 (int, int) endLoan(Copy c, String result) {
132 final m = c.member!, late = daysLate(c), value = fine(c);
133 charge(m, c.title, 'FINE', value);
134 m.history.add('$result ${c.title.isbn} copy ${c.number} from ${c.loanDay} to $day late $late fine ${money(value)}');
135 c.state = 'AVAILABLE';
136 c.member = null;
137 return (late, value);
138 }
139
140 void processLoss(Copy c, String heading, StringBuffer out) {
141 final m = c.member!, t = c.title, result = endLoan(c, 'LOST');
142 m.history[m.history.length - 1] += ' price ${money(t.price)}';
143 charge(m, t, 'PRICE', t.price);
144 c.state = 'LOST';
145 lost++;
146 out.writeln('$heading ${m.id} ${t.isbn} copy ${c.number} late ${result.$1} fine ${money(result.$2)} price ${money(t.price)} balance ${money(m.balance)}');
147 if (t.holdings == 0) {
148 for (final member in t.queue)
149 out.writeln('DROPPED ${t.isbn} ${member.id}');
150 t.queue.clear();
151 }
152 }
153 String? renewalProblem(Copy c) => c.due < day ? 'loan overdue' : c.renewals == 2 ? 'renewal limit' : c.title.queue.isNotEmpty ? 'title reserved' : null;
154 void renew(Copy c, StringBuffer out) {
155 c.due = openDay(day + c.member!.period);
156 c.renewals++;
157 renewed++;
158 out.writeln('RENEW ${c.member!.id} ${c.title.isbn} copy ${c.number} due ${c.due} renewals ${c.renewals}');
159 }
160 List<Copy> dueOrder(List<Copy> cs, [bool expiry = false]) => cs..sort((a, b) {
161 final x = expiry ? a.expiry : a.due, y = expiry ? b.expiry : b.due;
162 return x != y ? x.compareTo(y) : a.title.isbn.compareTo(b.title.isbn);
163 });
164 String execute(String line) {
165 final a = line.split(' '), op = a[0], out = StringBuffer();
166 if (op == 'DAY') {
167 final target = int.parse(a[1]);
168 if (target <= day)
169 fail('invalid day');
170 final loanOrder = loans()..sort((a, b) => a.member!.id != b.member!.id ? a.member!.id.compareTo(b.member!.id) : a.title.isbn.compareTo(b.title.isbn));
171 final copyOrder = books().expand((t) => t.copies).toList();
172 out.writeln('DAY $target');
173 while (day < target) {
174 day++;
175 final automatic = loanOrder.where((c) => c.state == 'LOANED' && c.due + 365 <= day).toList();
176 for (final c in automatic)
177 processLoss(c, 'AUTOLOST', out);
178 final expired = copyOrder.where((c) => c.state == 'HOLD' && c.expiry < day).toList();
179 for (final c in expired) {
180 final m = c.member!;
181 charge(m, c.title, 'FEE', m.fee);
182 out.writeln('EXPIRED ${c.title.isbn} copy ${c.number} ${m.id} fee ${money(m.fee)}');
183 release(c, out);
184 }
185 }
186 } else if (op == 'CLOSED') {
187 final target = int.parse(a[1]);
188 if (target <= day)
189 fail('invalid day');
190 if (closed.contains(target))
191 fail('already closed');
192 closed.add(target);
193 out.writeln('CLOSED $target');
194 for (final t in books()) {
195 for (final c in t.copies) {
196 if (c.state == 'LOANED' && c.due == target) {
197 c.due = openDay(target);
198 out.writeln('MOVED ${t.isbn} copy ${c.number} ${c.member!.id} due ${c.due}');
199 } else if (c.state == 'HOLD' && c.expiry == target) {
200 c.expiry = openDay(target);
201 out.writeln('MOVED ${t.isbn} copy ${c.number} ${c.member!.id} until ${c.expiry}');
202 }
203 }
204 }
205 } else if (op == 'ADDBOOK') {
206 final isbn = normalizeISBN(a[1]), name = a.skip(4).join(' '), old = titles[isbn];
207 if (old != null && old.name != name)
208 fail('title mismatch');
209 final t = old ?? Title(isbn, name, amount(a[3])), count = int.parse(a[2]);
210 t.price = amount(a[3]);
211 titles[isbn] = t;
212 final added = [for (var i = 0; i < count; i++) Copy(t, t.copies.length + i + 1)];
213 t.copies.addAll(added);
214 out.writeln('ADDBOOK $isbn added $count total ${t.holdings}');
215 for (final c in added)
216 release(c, out);
217 } else if (op == 'ADDMEMBER') {
218 if (members.containsKey(a[1]))
219 fail('duplicate member');
220 if (!['STUDENT', 'STAFF'].contains(a[2]))
221 fail('unknown type');
222 final m = Member(a[1], a[2], a.skip(3).join(' '));
223 members[m.id] = m;
224 out.writeln('ADDMEMBER ${m.id} ${m.type} loans ${m.loanLimit} days ${m.period}');
225 } else if (op == 'WITHDRAW') {
226 final t = book(a[1]), n = int.parse(a[2]);
227 if (n < 1 || n > t.copies.length)
228 fail('unknown copy');
229 final c = t.copies[n - 1];
230 if (c.state != 'AVAILABLE')
231 fail('copy not available');
232 c.state = 'WITHDRAWN';
233 out.writeln('WITHDRAW ${t.isbn} copy $n total ${t.holdings}');
234 } else if (['BORROW', 'RETURN', 'RENEW', 'RESERVE', 'CANCEL', 'LOST'].contains(op)) {
235 final m = member(a[1]), t = book(a[2]), current = loan(m, t), held = hold(m, t);
236 if (op == 'BORROW') {
237 if (closed.contains(day))
238 fail('library closed');
239 if (current != null)
240 fail('already borrowed');
241 if (m.blocked)
242 fail('member blocked');
243 final activeLoans = loans(m);
244 if (activeLoans.any((c) => c.due < day))
245 fail('overdue loans');
246 if (activeLoans.length >= m.loanLimit)
247 fail('loan limit');
248 final c = held ?? t.copies.where((c) => c.state == 'AVAILABLE').firstOrNull;
249 if (c == null)
250 fail(t.count('HOLD') > 0 ? 'all copies on hold' : 'no copy available');
251 c.state = 'LOANED';
252 c.member = m;
253 c.loanDay = day;
254 c.due = openDay(day + m.period);
255 c.renewals = 0;
256 t.borrowed++;
257 borrowed++;
258 out.writeln('BORROW ${m.id} ${t.isbn} copy ${c.number} due ${c.due}');
259 } else if (op == 'RETURN' || op == 'LOST') {
260 if (current == null)
261 fail('not borrowed');
262 if (op == 'LOST')
263 processLoss(current, 'LOST', out);
264 else {
265 final result = endLoan(current, 'RETURNED');
266 returned++;
267 out.writeln('RETURN ${m.id} ${t.isbn} copy ${current.number} late ${result.$1} fine ${money(result.$2)} balance ${money(m.balance)}');
268 release(current, out);
269 }
270 } else if (op == 'RENEW') {
271 if (current == null)
272 fail('not borrowed');
273 if (m.blocked)
274 fail('member blocked');
275 final reason = renewalProblem(current);
276 if (reason != null)
277 fail(reason);
278 renew(current, out);
279 } else if (op == 'RESERVE') {
280 if (current != null)
281 fail('already borrowed');
282 if (t.queue.contains(m) || held != null)
283 fail('already reserved');
284 if (m.blocked)
285 fail('member blocked');
286 if (reservations(m) >= m.reservationLimit)
287 fail('reservation limit');
288 if (t.holdings == 0)
289 fail('no copies');
290 if (t.count('AVAILABLE') > 0)
291 fail('copy available');
292 t.queue.add(m);
293 final p = t.queue.length, due = t.copies.where((c) => c.state == 'LOANED').map((c) => c.due).toList()..sort();
294 out.writeln('RESERVE ${m.id} ${t.isbn} position $p estimate ${p <= due.length ? due[p - 1] : 'none'}');
295 } else {
296 if (!t.queue.contains(m) && held == null)
297 fail('not reserved');
298 if (t.queue.remove(m))
299 out.writeln('CANCEL ${m.id} ${t.isbn} queue');
300 else {
301 out.writeln('CANCEL ${m.id} ${t.isbn} hold copy ${held!.number}');
302 release(held!, out);
303 }
304 }
305 } else if (op == 'RENEWALL') {
306 final m = member(a[1]), cs = loans(m)..sort((a, b) => a.title.isbn.compareTo(b.title.isbn));
307 if (cs.isEmpty)
308 fail('no loans');
309 if (m.blocked)
310 fail('member blocked');
311 var success = 0, skipped = 0;
312 for (final c in cs) {
313 final reason = renewalProblem(c);
314 if (reason == null) {
315 renew(c, out);
316 success++;
317 } else {
318 out.writeln('SKIP ${c.title.isbn} $reason');
319 skipped++;
320 }
321 }
322 out.writeln('RENEWALL ${m.id} renewed $success skipped $skipped');
323 } else if (op == 'PAYFINE') {
324 final m = member(a[1]);
325 if (m.balance == 0)
326 fail('no fines');
327 var remaining = amount(a[2]);
328 final applied = <(int, int)>[];
329 for (var i = 0; i < m.items.length && remaining > 0; i++) {
330 final item = m.items[i], value = min(item.unpaid, remaining);
331 if (value == 0)
332 continue;
333 item.unpaid -= value;
334 remaining -= value;
335 applied.add((i, value));
336 }
337 final value = amount(a[2]) - remaining;
338 paid += value;
339 out.writeln('PAYFINE ${m.id} paid ${money(value)} change ${money(remaining)} balance ${money(m.balance)}');
340 for (final entry in applied)
341 out.writeln('ITEM ${entry.$1 + 1} ${m.items[entry.$1].kind} paid ${money(entry.$2)} remaining ${money(m.items[entry.$1].unpaid)}');
342 } else if (op == 'WAIVE') {
343 final m = member(a[1]), n = int.parse(a[2]);
344 if (n < 1 || n > m.items.length)
345 fail('unknown item');
346 final item = m.items[n - 1];
347 if (item.unpaid == 0)
348 fail('item settled');
349 final value = item.unpaid;
350 item.unpaid = 0;
351 waived += value;
352 out.writeln('WAIVE ${m.id} item $n ${item.kind} amount ${money(value)} balance ${money(m.balance)}');
353 } else if (op == 'FINES') {
354 final m = member(a[1]);
355 out.writeln('FINES ${m.id} items ${m.items.where((i) => i.unpaid != 0).length} balance ${money(m.balance)}');
356 for (var i = 0; i < m.items.length; i++) {
357 final item = m.items[i];
358 if (item.unpaid > 0)
359 out.writeln('${i + 1} day ${item.day} ${item.kind} ${item.isbn} amount ${money(item.amount)} remaining ${money(item.unpaid)}');
360 }
361 } else if (op == 'SEARCH') {
362 final keyword = a.skip(1).join(' ').toLowerCase(), selected = titles.values.where((t) => t.folded.contains(keyword)).toList();
363 selected.sort((a, b) => a.folded != b.folded ? a.folded.compareTo(b.folded) : a.isbn.compareTo(b.isbn));
364 out.writeln('SEARCH ${selected.length}');
365 for (final t in selected)
366 out.writeln('${t.isbn} ${t.count('AVAILABLE')}/${t.holdings} ${t.name}');
367 } else if (op == 'OVERDUE') {
368 final cs = loans().where((c) => c.due < day).toList()..sort((a, b) => a.due != b.due ? a.due.compareTo(b.due) : a.member!.id != b.member!.id ? a.member!.id.compareTo(b.member!.id) : a.title.isbn.compareTo(b.title.isbn));
369 out.writeln('OVERDUE ${cs.length}');
370 for (final c in cs)
371 out.writeln('${c.member!.id} ${c.title.isbn} copy ${c.number} due ${c.due} late ${daysLate(c)} fine ${money(fine(c))}');
372 } else if (op == 'NOTICES') {
373 final lines = <String>[];
374 for (final m in people()) {
375 final cs = dueOrder(loans(m));
376 for (final c in cs.where((c) => c.due < day))
377 lines.add('OVERDUE ${m.id} ${c.title.isbn} due ${c.due}');
378 for (final c in cs.where((c) => c.due >= day && c.due <= day + 2))
379 lines.add('DUESOON ${m.id} ${c.title.isbn} due ${c.due}');
380 for (final c in dueOrder(holds(m), true))
381 lines.add('PICKUP ${m.id} ${c.title.isbn} until ${c.expiry}');
382 }
383 out.writeln('NOTICES ${lines.length}');
384 for (final text in lines)
385 out.writeln(text);
386 } else if (op == 'MEMBER' || op == 'MEMBERS') {
387 final ms = op == 'MEMBER' ? [member(a[1])] : people();
388 if (op == 'MEMBERS')
389 out.writeln('MEMBERS ${ms.length}');
390 for (final m in ms) {
391 final cs = dueOrder(loans(m));
392 if (op == 'MEMBERS') {
393 out.writeln('${m.id} ${m.type} loans ${cs.length} reservations ${reservations(m)} balance ${money(m.balance)} ${m.blocked ? 'BLOCKED' : 'OK'}');
394 continue;
395 }
396 out.writeln('MEMBER ${m.id} ${m.type} ${m.name}');
397 out.writeln('STATUS loans ${cs.length}/${m.loanLimit} reservations ${reservations(m)}/${m.reservationLimit} balance ${money(m.balance)} ${m.blocked ? 'BLOCKED' : 'OK'}');
398 for (final c in cs)
399 out.writeln('LOAN ${c.title.isbn} copy ${c.number} due ${c.due} renewals ${c.renewals}${c.due < day ? ' late ${daysLate(c)} fine ${money(fine(c))}' : ''}');
400 for (final c in dueOrder(holds(m), true))
401 out.writeln('HOLD ${c.title.isbn} copy ${c.number} until ${c.expiry}');
402 for (final t in books())
403 if (t.queue.contains(m)) out.writeln('QUEUE ${t.isbn} position ${t.queue.indexOf(m) + 1}');
404 }
405 } else if (op == 'BOOK') {
406 final t = book(a[1]);
407 out.writeln('BOOK ${t.isbn} ${t.name}');
408 out.writeln('PRICE ${money(t.price)} copies ${t.holdings} borrowed ${t.borrowed}');
409 for (final c in t.copies)
410 out.writeln('COPY ${c.number} ${c.state}${c.state == 'LOANED' ? ' ${c.member!.id} due ${c.due}' : c.state == 'HOLD' ? ' ${c.member!.id} until ${c.expiry}' : ''}');
411 out.writeln('QUEUE ${t.queue.length}${t.queue.isEmpty ? '' : ' ${t.queue.map((m) => m.id).join(' ')}'}');
412 } else if (op == 'INVENTORY') {
413 final cs = copies();
414 int count(String state) => cs.where((c) => c.state == state).length;
415 out.writeln('INVENTORY titles ${titles.length} copies ${titles.values.fold(0, (s, t) => s + t.holdings)} available ${count('AVAILABLE')} loaned ${count('LOANED')} hold ${count('HOLD')} lost ${count('LOST')} withdrawn ${count('WITHDRAWN')}');
416 for (final t in books())
417 out.writeln('${t.isbn} copies ${t.holdings} available ${t.count('AVAILABLE')} loaned ${t.count('LOANED')} hold ${t.count('HOLD')} queue ${t.queue.length}');
418 } else if (op == 'HISTORY') {
419 final m = member(a[1]);
420 out.writeln('HISTORY ${m.id} ${m.history.length}');
421 for (final text in m.history)
422 out.writeln(text);
423 } else if (op == 'STATS') {
424 out.writeln('STATS day $day titles ${titles.length} members ${members.length} closed ${closed.length}');
425 out.writeln('LOANS borrowed $borrowed returned $returned renewed $renewed lost $lost active ${loans().length}');
426 out.writeln('MONEY fines ${money(charged['FINE']!)} fees ${money(charged['FEE']!)} charges ${money(charged['PRICE']!)} paid ${money(paid)} waived ${money(waived)} outstanding ${money(members.values.fold(0, (s, m) => s + m.balance))}');
427 final top = titles.values.where((t) => t.borrowed > 0).toList()..sort((a, b) => a.borrowed != b.borrowed ? b.borrowed.compareTo(a.borrowed) : a.isbn.compareTo(b.isbn));
428 for (var i = 0; i < top.length && i < 3; i++)
429 out.writeln('TOP ${i + 1} ${top[i].isbn} ${top[i].borrowed} ${top[i].name}');
430 }
431 return out.toString();
432 }
433}

435void main() {
436 final library = Library(), count = int.parse(stdin.readLineSync()!);
437 for (var i = 0; i < count; i++) {
438 final line = stdin.readLineSync()!;
439 try {
440 stdout.write(library.execute(line));
441 } on LibraryError catch (error) {
442 stdout.writeln('ERROR ${error.message}');
443 }
444 }
