13const (
14 available = iota
15 loaned
16 hold
17 lostState
18 withdrawn
19)
20var states = []string{"AVAILABLE", "LOANED", "HOLD", "LOST", "WITHDRAWN"}
21var kinds = []string{"FINE", "FEE", "PRICE"}
22type ref struct { title, copy int }
23type bookCopy struct { state, member, from, due, renewals, expiry int }
24type title struct {
25 isbn, name, folded string
26 price, borrowed int
27 counts [5]int
28 copies []bookCopy
29 queue []int
30}
31type charge struct { day, kind, title, amount, unpaid int }
32type member struct {
33 id, kind, name string
34 limit, period, rate, cap, reserveLimit, fee int
35 balance, reservations int
36 loans, holds []ref
37 charges []charge
38 history []string
39}
40type library struct {
41 day, borrowed, returned, renewed, lost, paid, waived int
42 charged [3]int
43 closed map[int]bool
44 titleIDs, memberIDs map[string]int
45 titleOrder, memberOrder []int
46 titles []*title
47 members []*member
48 output []string
49}
50func number(s string) int {
51 n,_ := strconv.Atoi(s)
52 return n
53}
54func money(n int) string { return fmt.Sprintf("%d.%02d", n/100, n%100) }
55func cents(s string) int {
56 whole,fraction,_ := strings.Cut(s,".")
57 return number(whole)*100+number(fraction)
58}
59func normalize(input string) string {
60 s := strings.ReplaceAll(input,"-","")
61 if len(s) == 10 {
62 sum := 0
63 for i:=0; i<10; i++ {
64 if !(s[i] >= '0' && s[i] <= '9') && !(i == 9 && s[i] == 'X') { return "" }
65 value := int(s[i]-'0')
66 if s[i] == 'X' { value = 10 }
67 sum += (10-i)*value
68 }
69 if sum%11 != 0 { return "" }
70 s = "978"+s[:9]
71 sum = 0
72 for i:=0; i<12; i++ {
73 weight := 1
74 if i%2 == 1 { weight = 3 }
75 sum += weight*int(s[i]-'0')
76 }
77 return s+string(byte('0'+(10-sum%10)%10))
78 }
79 if len(s) != 13 || !(strings.HasPrefix(s,"978") || strings.HasPrefix(s,"979")) { return "" }
80 sum := 0
81 for i:=0; i<13; i++ {
82 if s[i] < '0' || s[i] > '9' { return "" }
83 weight := 1
84 if i%2 == 1 { weight = 3 }
85 sum += weight*int(s[i]-'0')
86 }
87 if sum%10 != 0 { return "" }
88 return s
89}
90func (l *library) line(pattern string, args ...any) { l.output = append(l.output,fmt.Sprintf(pattern,args...)) }
91func (l *library) error(condition bool, reason string) bool {
92 if condition { l.line("ERROR %s",reason) }
93 return condition
94}
95func (l *library) open(d int) int {
96 for l.closed[d] { d++ }
97 return d
98}
99func (l *library) copy(r ref) *bookCopy { return &l.titles[r.title].copies[r.copy] }
100func (l *library) holdings(t int) int {
101 c := l.titles[t].counts
102 return c[0]+c[1]+c[2]
103}
104func (l *library) state(r ref, next int) {
105 t := l.titles[r.title]
106 t.counts[t.copies[r.copy].state]--
107 t.copies[r.copy].state = next
108 t.counts[next]++
109}
110func removeRef(refs []ref, r ref) []ref {
111 if i := slices.Index(refs,r); i >= 0 { return slices.Delete(refs,i,i+1) }
112 return refs
113}
114func findRef(refs []ref, t int) int {
115 for _,r := range refs {
116 if r.title == t { return r.copy }
117 }
118 return -1
119}
120func (l *library) getMember(id string) int {
121 m,ok := l.memberIDs[id]
122 if l.error(!ok,"unknown member") { return -1 }
123 return m
124}
125func (l *library) getTitle(input string) int {
126 isbn := normalize(input)
127 if l.error(isbn == "","invalid isbn") { return -1 }
128 t,ok := l.titleIDs[isbn]
129 if l.error(!ok,"unknown book") { return -1 }
130 return t
131}
132func (l *library) late(r ref) int {
133 return max(0,l.day-l.copy(r).due)
134}
135func (l *library) fine(r ref) int {
136 p := l.members[l.copy(r).member]
137 return min(l.late(r)*p.rate,p.cap)
138}
139func (l *library) charge(m, kind, t, amount int) {
140 if amount == 0 { return }
141 p := l.members[m]
142 p.charges = append(p.charges,charge{l.day,kind,t,amount,amount})
143 p.balance += amount
144 l.charged[kind] += amount
145}
146func (l *library) release(r ref) {
147 t := l.titles[r.title]
148 if len(t.queue) == 0 {
149 l.state(r,available)
150 return
151 }
152 m := t.queue[0]
153 t.queue = t.queue[1:]
154 l.state(r,hold)
155 c := l.copy(r)
156 c.member = m
157 c.expiry = l.open(l.day+3)
158 l.members[m].holds = append(l.members[m].holds,r)
159 l.line("HOLD %s copy %d %s until %d",t.isbn,r.copy+1,l.members[m].id,c.expiry)
160}
161func (l *library) dropQueue(t int) {
162 if l.holdings(t) != 0 { return }
163 for _,m := range l.titles[t].queue {
164 l.members[m].reservations--
165 l.line("DROPPED %s %s",l.titles[t].isbn,l.members[m].id)
166 }
167 l.titles[t].queue = nil
168}
169func (l *library) endLoan(r ref, loss bool, heading string) {
170 c := *l.copy(r)
171 m,late,fine := c.member,l.late(r),l.fine(r)
172 p,t := l.members[m],l.titles[r.title]
173 p.loans = removeRef(p.loans,r)
174 l.charge(m,0,r.title,fine)
175 result := "RETURNED"
176 if loss { result = "LOST" }
177 history := fmt.Sprintf("%s %s copy %d from %d to %d late %d fine %s",result,t.isbn,r.copy+1,c.from,l.day,late,money(fine))
178 if loss {
179 history += " price "+money(t.price)
180 l.charge(m,2,r.title,t.price)
181 l.state(r,lostState)
182 l.lost++
183 } else { l.returned++ }
184 p.history = append(p.history,history)
185 s := fmt.Sprintf("%s %s %s copy %d late %d fine %s",heading,p.id,t.isbn,r.copy+1,late,money(fine))
186 if loss { s += " price "+money(t.price) }
187 l.line("%s balance %s",s,money(p.balance))
188 if loss { l.dropQueue(r.title) } else { l.release(r) }
189}
190func (l *library) renewalReason(r ref) string {
191 if l.late(r) != 0 { return "loan overdue" }
192 if l.copy(r).renewals == 2 { return "renewal limit" }
193 if len(l.titles[r.title].queue) != 0 { return "title reserved" }
194 return ""
195}
196func (l *library) renew(r ref) {
197 c := l.copy(r)
198 p := l.members[c.member]
199 c.due = l.open(l.day+p.period)
200 c.renewals++
201 l.renewed++
202 l.line("RENEW %s %s copy %d due %d renewals %d",p.id,l.titles[r.title].isbn,r.copy+1,c.due,c.renewals)
203}
204func (l *library) ordered(refs []ref, holds bool) []ref {
205 result := slices.Clone(refs)
206 sort.Slice(result,func(i,j int) bool {
207 a,b := result[i],result[j]
208 x,y := l.copy(a).due,l.copy(b).due
209 if holds { x,y = l.copy(a).expiry,l.copy(b).expiry }
210 if x != y { return x < y }
211 return l.titles[a.title].isbn < l.titles[b.title].isbn
212 })
213 return result
214}
215func (l *library) execute(a []string) {
216 cmd := a[0]
217 switch cmd {
218 case "DAY":
219 target := number(a[1])
220 if l.error(target <= l.day,"invalid day") { return }
221 l.line("DAY %d",target)
222 for l.day < target {
223 l.day++
224 losses := []ref{}
225 for _,id := range l.memberOrder {
226 for _,r := range l.members[id].loans {
227 if l.copy(r).due+365 <= l.day { losses = append(losses,r) }
228 }
229 }
230 for _,r := range losses { l.endLoan(r,true,"AUTOLOST") }
231 expiries := []ref{}
232 for _,t := range l.titleOrder {
233 for i,c := range l.titles[t].copies {
234 if c.state == hold && c.expiry < l.day { expiries = append(expiries,ref{t,i}) }
235 }
236 }
237 for _,r := range expiries {
238 m := l.copy(r).member
239 l.charge(m,1,r.title,l.members[m].fee)
240 l.members[m].holds = removeRef(l.members[m].holds,r)
241 l.members[m].reservations--
242 l.line("EXPIRED %s copy %d %s fee %s",l.titles[r.title].isbn,r.copy+1,l.members[m].id,money(l.members[m].fee))
243 l.release(r)
244 }
245 }
246 case "CLOSED":
247 d := number(a[1])
248 if l.error(d <= l.day,"invalid day") || l.error(l.closed[d],"already closed") { return }
249 l.closed[d] = true
250 l.line("CLOSED %d",d)
251 for _,id := range l.titleOrder {
252 t := l.titles[id]
253 for i := range t.copies {
254 c := &t.copies[i]
255 if c.state == loaned && c.due == d {
256 c.due = l.open(d)
257 l.line("MOVED %s copy %d %s due %d",t.isbn,i+1,l.members[c.member].id,c.due)
258 } else if c.state == hold && c.expiry == d {
259 c.expiry = l.open(d)
260 l.line("MOVED %s copy %d %s until %d",t.isbn,i+1,l.members[c.member].id,c.expiry)
261 }
262 }
263 }
264 case "ADDBOOK":
265 isbn,name := normalize(a[1]),strings.Join(a[4:]," ")
266 if l.error(isbn == "","invalid isbn") { return }
267 t,exists := l.titleIDs[isbn]
268 if l.error(exists && l.titles[t].name != name,"title mismatch") { return }
269 if !exists {
270 t = len(l.titles)
271 l.titles = append(l.titles,&title{isbn:isbn,name:name,folded:strings.ToLower(name)})
272 l.titleIDs[isbn] = t
273 position := sort.Search(len(l.titleOrder),func(i int) bool { return l.titles[l.titleOrder[i]].isbn > isbn })
274 l.titleOrder = slices.Insert(l.titleOrder,position,t)
275 }
276 book := l.titles[t]
277 book.price = cents(a[3])
278 start,added := len(book.copies),number(a[2])
279 book.copies = append(book.copies,make([]bookCopy,added)...)
280 book.counts[available] += added
281 l.line("ADDBOOK %s added %d total %d",isbn,added,l.holdings(t))
282 for i:=start; i<len(book.copies); i++ { l.release(ref{t,i}) }
283 case "ADDMEMBER":
284 _,exists := l.memberIDs[a[1]]
285 if l.error(exists,"duplicate member") || l.error(a[2] != "STUDENT" && a[2] != "STAFF","unknown type") { return }
286 m := &member{id:a[1],kind:a[2],name:strings.Join(a[3:]," "),limit:3,period:14,rate:25,cap:1000,reserveLimit:2,fee:50}
287 if m.kind == "STAFF" { m.limit,m.period,m.rate,m.cap,m.reserveLimit,m.fee = 10,28,10,2000,5,100 }
288 id := len(l.members)
289 l.memberIDs[m.id] = id
290 l.members = append(l.members,m)
291 position := sort.Search(len(l.memberOrder),func(i int) bool { return l.members[l.memberOrder[i]].id > m.id })
292 l.memberOrder = slices.Insert(l.memberOrder,position,id)
293 l.line("ADDMEMBER %s %s loans %d days %d",m.id,m.kind,m.limit,m.period)
294 case "WITHDRAW":
295 t := l.getTitle(a[1])
296 if t < 0 { return }
297 i := number(a[2])-1
298 if l.error(i < 0 || i >= len(l.titles[t].copies),"unknown copy") { return }
299 if l.error(l.titles[t].copies[i].state != available,"copy not available") { return }
300 l.state(ref{t,i},withdrawn)
301 l.line("WITHDRAW %s copy %d total %d",l.titles[t].isbn,i+1,l.holdings(t))
302 case "BORROW","RETURN","RENEW","RESERVE","CANCEL","LOST":
303 m := l.getMember(a[1])
304 if m < 0 { return }
305 t := l.getTitle(a[2])
306 if t < 0 { return }
307 p,book := l.members[m],l.titles[t]
308 loan,held,q := findRef(p.loans,t),findRef(p.holds,t),slices.Index(book.queue,m)
309 switch cmd {
310 case "BORROW":
311 if l.error(l.closed[l.day],"library closed") || l.error(loan >= 0,"already borrowed") || l.error(p.balance > 500,"member blocked") { return }
312 overdue := false
313 for _,r := range p.loans { overdue = overdue || l.late(r) != 0 }
314 if l.error(overdue,"overdue loans") || l.error(len(p.loans) >= p.limit,"loan limit") { return }
315 first := -1
316 for i,c := range book.copies {
317 if c.state == available {
318 first = i
319 break
320 }
321 }
322 i := first
323 if held >= 0 { i = held }
324 reason := "no copy available"
325 if book.counts[hold] > 0 { reason = "all copies on hold" }
326 if l.error(i < 0,reason) { return }
327 r := ref{t,i}
328 if held >= 0 {
329 p.holds = removeRef(p.holds,r)
330 p.reservations--
331 }
332 l.state(r,loaned)
333 *l.copy(r) = bookCopy{state:loaned,member:m,from:l.day,due:l.open(l.day+p.period)}
334 p.loans = append(p.loans,r)
335 sort.Slice(p.loans,func(i,j int) bool { return l.titles[p.loans[i].title].isbn < l.titles[p.loans[j].title].isbn })
336 l.borrowed++
337 book.borrowed++
338 l.line("BORROW %s %s copy %d due %d",p.id,book.isbn,i+1,l.copy(r).due)
339 case "RETURN","LOST","RENEW":
340 if l.error(loan < 0,"not borrowed") { return }
341 r := ref{t,loan}
342 if cmd == "RENEW" {
343 if l.error(p.balance > 500,"member blocked") { return }
344 reason := l.renewalReason(r)
345 if l.error(reason != "",reason) { return }
346 l.renew(r)
347 } else { l.endLoan(r,cmd == "LOST",cmd) }
348 case "RESERVE":
349 if l.error(loan >= 0,"already borrowed") || l.error(q >= 0 || held >= 0,"already reserved") || l.error(p.balance > 500,"member blocked") || l.error(p.reservations >= p.reserveLimit,"reservation limit") || l.error(l.holdings(t) == 0,"no copies") || l.error(book.counts[available] > 0,"copy available") { return }
350 book.queue = append(book.queue,m)
351 p.reservations++
352 dues := []int{}
353 for _,c := range book.copies {
354 if c.state == loaned { dues = append(dues,c.due) }
355 }
356 sort.Ints(dues)
357 position := len(book.queue)
358 estimate := "none"
359 if position <= len(dues) { estimate = strconv.Itoa(dues[position-1]) }
360 l.line("RESERVE %s %s position %d estimate %s",p.id,book.isbn,position,estimate)
361 case "CANCEL":
362 if l.error(q < 0 && held < 0,"not reserved") { return }
363 p.reservations--
364 if q >= 0 {
365 book.queue = slices.Delete(book.queue,q,q+1)
366 l.line("CANCEL %s %s queue",p.id,book.isbn)
367 } else {
368 r := ref{t,held}
369 p.holds = removeRef(p.holds,r)
370 l.line("CANCEL %s %s hold copy %d",p.id,book.isbn,held+1)
371 l.release(r)
372 }
373 }
374 case "RENEWALL":
375 m := l.getMember(a[1])
376 if m < 0 { return }
377 p := l.members[m]
378 if l.error(len(p.loans) == 0,"no loans") || l.error(p.balance > 500,"member blocked") { return }
379 good,skipped := 0,0
380 for _,r := range p.loans {
381 reason := l.renewalReason(r)
382 if reason == "" {
383 l.renew(r)
384 good++
385 } else {
386 l.line("SKIP %s %s",l.titles[r.title].isbn,reason)
387 skipped++
388 }
389 }
390 l.line("RENEWALL %s renewed %d skipped %d",p.id,good,skipped)
391 case "PAYFINE","WAIVE","FINES":
392 m := l.getMember(a[1])
393 if m < 0 { return }
394 p := l.members[m]
395 switch cmd {
396 case "PAYFINE":
397 if l.error(p.balance == 0,"no fines") { return }
398 amount := cents(a[2])
399 remaining := amount
400 details := []string{}
401 for i := range p.charges {
402 c := &p.charges[i]
403 applied := min(remaining,c.unpaid)
404 if applied == 0 { continue }
405 c.unpaid -= applied
406 remaining -= applied
407 details = append(details,fmt.Sprintf("ITEM %d %s paid %s remaining %s",i+1,kinds[c.kind],money(applied),money(c.unpaid)))
408 if remaining == 0 { break }
409 }
410 p.balance -= amount-remaining
411 l.paid += amount-remaining
412 l.line("PAYFINE %s paid %s change %s balance %s",p.id,money(amount-remaining),money(remaining),money(p.balance))
413 l.output = append(l.output,details...)
414 case "WAIVE":
415 i := number(a[2])-1
416 if l.error(i < 0 || i >= len(p.charges),"unknown item") { return }
417 c := &p.charges[i]
418 if l.error(c.unpaid == 0,"item settled") { return }
419 amount := c.unpaid
420 c.unpaid = 0
421 p.balance -= amount
422 l.waived += amount
423 l.line("WAIVE %s item %d %s amount %s balance %s",p.id,i+1,kinds[c.kind],money(amount),money(p.balance))
424 case "FINES":
425 count := 0
426 for _,c := range p.charges {
427 if c.unpaid > 0 { count++ }
428 }
429 l.line("FINES %s items %d balance %s",p.id,count,money(p.balance))
430 for i,c := range p.charges {
431 if c.unpaid > 0 { l.line("%d day %d %s %s amount %s remaining %s",i+1,c.day,kinds[c.kind],l.titles[c.title].isbn,money(c.amount),money(c.unpaid)) }
432 }
433 }
434 case "SEARCH":
435 keyword := strings.ToLower(strings.Join(a[1:]," "))
436 found := []int{}
437 for i,t := range l.titles {
438 if strings.Contains(t.folded,keyword) { found = append(found,i) }
439 }
440 sort.Slice(found,func(i,j int) bool {
441 a,b := l.titles[found[i]],l.titles[found[j]]
442 if a.folded != b.folded { return a.folded < b.folded }
443 return a.isbn < b.isbn
444 })
445 l.line("SEARCH %d",len(found))
446 for _,id := range found {
447 t := l.titles[id]
448 l.line("%s %d/%d %s",t.isbn,t.counts[available],l.holdings(id),t.name)
449 }
450 case "OVERDUE":
451 refs := []ref{}
452 for _,m := range l.members {
453 for _,r := range m.loans {
454 if l.late(r) > 0 { refs = append(refs,r) }
455 }
456 }
457 sort.Slice(refs,func(i,j int) bool {
458 a,b := refs[i],refs[j]
459 ca,cb := l.copy(a),l.copy(b)
460 if ca.due != cb.due { return ca.due < cb.due }
461 ma,mb := l.members[ca.member].id,l.members[cb.member].id
462 if ma != mb { return ma < mb }
463 return l.titles[a.title].isbn < l.titles[b.title].isbn
464 })
465 l.line("OVERDUE %d",len(refs))
466 for _,r := range refs { l.line("%s %s copy %d due %d late %d fine %s",l.members[l.copy(r).member].id,l.titles[r.title].isbn,r.copy+1,l.copy(r).due,l.late(r),money(l.fine(r))) }
467 case "NOTICES":
468 notices := []string{}
469 for _,id := range l.memberOrder {
470 m := l.members[id]
471 for _,r := range l.ordered(m.loans,false) {
472 if l.copy(r).due > l.day+2 { continue }
473 heading := "DUESOON"
474 if l.late(r) > 0 { heading = "OVERDUE" }
475 notices = append(notices,fmt.Sprintf("%s %s %s due %d",heading,m.id,l.titles[r.title].isbn,l.copy(r).due))
476 }
477 for _,r := range l.ordered(m.holds,true) { notices = append(notices,fmt.Sprintf("PICKUP %s %s until %d",m.id,l.titles[r.title].isbn,l.copy(r).expiry)) }
478 }
479 l.line("NOTICES %d",len(notices))
480 l.output = append(l.output,notices...)
481 case "MEMBER","MEMBERS","HISTORY":
482 target := -1
483 if cmd != "MEMBERS" {
484 target = l.getMember(a[1])
485 if target < 0 { return }
486 }
487 if cmd == "HISTORY" {
488 m := l.members[target]
489 l.line("HISTORY %s %d",m.id,len(m.history))
490 l.output = append(l.output,m.history...)
491 return
492 }
493 if cmd == "MEMBERS" { l.line("MEMBERS %d",len(l.members)) }
494 for _,id := range l.memberOrder {
495 if target >= 0 && id != target { continue }
496 m := l.members[id]
497 status := "OK"
498 if m.balance > 500 { status = "BLOCKED" }
499 if cmd == "MEMBERS" {
500 l.line("%s %s loans %d reservations %d balance %s %s",m.id,m.kind,len(m.loans),m.reservations,money(m.balance),status)
501 continue
502 }
503 l.line("MEMBER %s %s %s",m.id,m.kind,m.name)
504 l.line("STATUS loans %d/%d reservations %d/%d balance %s %s",len(m.loans),m.limit,m.reservations,m.reserveLimit,money(m.balance),status)
505 for _,r := range l.ordered(m.loans,false) {
506 s := fmt.Sprintf("LOAN %s copy %d due %d renewals %d",l.titles[r.title].isbn,r.copy+1,l.copy(r).due,l.copy(r).renewals)
507 if l.late(r) > 0 { s += fmt.Sprintf(" late %d fine %s",l.late(r),money(l.fine(r))) }
508 l.line("%s",s)
509 }
510 for _,r := range l.ordered(m.holds,true) { l.line("HOLD %s copy %d until %d",l.titles[r.title].isbn,r.copy+1,l.copy(r).expiry) }
511 for _,t := range l.titleOrder {
512 q := slices.Index(l.titles[t].queue,id)
513 if q >= 0 { l.line("QUEUE %s position %d",l.titles[t].isbn,q+1) }
514 }
515 }
516 case "BOOK":
517 t := l.getTitle(a[1])
518 if t < 0 { return }
519 book := l.titles[t]
520 l.line("BOOK %s %s",book.isbn,book.name)
521 l.line("PRICE %s copies %d borrowed %d",money(book.price),l.holdings(t),book.borrowed)
522 for i,c := range book.copies {
523 s := fmt.Sprintf("COPY %d %s",i+1,states[c.state])
524 if c.state == loaned { s += fmt.Sprintf(" %s due %d",l.members[c.member].id,c.due) } else if c.state == hold { s += fmt.Sprintf(" %s until %d",l.members[c.member].id,c.expiry) }
525 l.line("%s",s)
526 }

544 outstanding += m.balance
545 }
546 l.line("STATS day %d titles %d members %d closed %d",l.day,len(l.titles),len(l.members),len(l.closed))
547 l.line("LOANS borrowed %d returned %d renewed %d lost %d active %d",l.borrowed,l.returned,l.renewed,l.lost,active)

549 top := []int{}
