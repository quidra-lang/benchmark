48std::string trim(const std::string& text) {
49 std::size_t begin = text.find_first_not_of(' ');
50 if (begin == std::string::npos) return "";
51 return text.substr(begin, text.find_last_not_of(' ') - begin + 1);
52}

54bool identifier(const std::string& text) {
55 static const std::regex pattern("[A-Za-z_][A-Za-z0-9_]*");
56 return std::regex_match(text, pattern);
57}
58
59bool subscript(const std::string& text) {
60 static const std::regex pattern("0|[1-9][0-9]*");
61 return std::regex_match(text, pattern);
62}
63
64bool numeric_text(const std::string& text) {
65 static const std::regex pattern("-?[0-9]{1,9}");
66 return std::regex_match(text, pattern);
67}

69std::vector<std::string> split(const std::string& text, char delimiter,
70 bool respect_quotes = false) {
71 std::vector<std::string> parts;
72 bool quoted = false;
73 std::size_t start = 0;
74 for (std::size_t i = 0; i < text.size(); ++i) {
75 if (respect_quotes && text[i] == '\'') quoted = !quoted;
76 if (text[i] == delimiter && !quoted) {
77 parts.push_back(text.substr(start, i - start));
78 start = i + 1;
79 }
80 }
81 parts.push_back(text.substr(start));
82 return parts;
83}

85bool valid_name(const std::string& name) {
86 if (name == "." || name == "@index" || name == "@number" ||
87 name == "@first" || name == "@last") return true;
88 for (const std::string& part : split(name, '.')) {
89 if (!identifier(part) && !subscript(part)) return false;
90 }
91 return true;
92}

101std::size_t index_number(const std::string& text) {
102 std::size_t value = 0;
103 for (char c : text) {
104 std::size_t digit = static_cast<std::size_t>(c - '0');
105 if (value > (std::numeric_limits<std::size_t>::max() - digit) / 10)
106 return std::numeric_limits<std::size_t>::max();
107 value = value * 10 + digit;
108 }
109 return value;
110}

112std::vector<Part> parse_key(const std::string& key) {
113 static const std::regex pattern(
114 "[A-Za-z_][A-Za-z0-9_]*(\\.[A-Za-z_][A-Za-z0-9_]*|\\[(0|[1-9][0-9]*)\\])*");
115 if (!std::regex_match(key, pattern))
116 throw std::runtime_error("bad key '" + key + "'");
117 std::vector<Part> parts;
118 std::size_t position = 0;
119 while (position < key.size()) {
120 if (key[position] == '.') ++position;
121 bool index = key[position] == '[';
122 std::size_t start = position + (index ? 1 : 0);
123 position = start;
124 if (index) {
125 while (key[position] != ']') ++position;
126 } else {
127 while (position < key.size() && key[position] != '.' && key[position] != '[')
128 ++position;
129 }
130 std::string token = key.substr(start, position - start);
131 if (index) ++position;
132 parts.push_back({index, token, index ? index_number(token) : 0,
133 key.substr(0, position)});
134 }
135 return parts;
136}

138const Value* get(const Value& value, const Part& part) {
139 if (part.index) {
140 if (value.kind == Kind::List && part.number < value.elements.size())
141 return &value.elements[part.number];
142 } else if (value.kind == Kind::Map) {
143 auto found = value.fields.find(part.key);
144 if (found != value.fields.end()) return &found->second;
145 }
146 return nullptr;
147}
148
149Value& put(Value& container, const Part& part, Value value) {
150 if (!part.index) {
151 container.fields[part.key] = std::move(value);
152 return container.fields[part.key];
153 }
154 if (part.number == container.elements.size())
155 container.elements.push_back(std::move(value));
156 else
157 container.elements[part.number] = std::move(value);
158 return container.elements[part.number];
159}

161Value read_scalar(const std::string& text, std::size_t& position) {
162 if (position < text.size() && text[position] == '"') {
163 std::string result;
164 ++position;
165 while (position < text.size()) {
166 char c = text[position++];
167 if (c == '"') return Value(result);
168 if (c == '\\') {
169 if (position == text.size()) throw std::runtime_error("invalid value");
170 c = text[position++];
171 if (c == 'n') c = '\n';
172 else if (c != '"' && c != '\\') throw std::runtime_error("invalid value");
173 }
174 result += c;
175 }
176 throw std::runtime_error("invalid value");
177 }
178 std::size_t end = text.find_first_of(",]", position);
179 if (end == std::string::npos) end = text.size();
180 std::string word = text.substr(position, end - position);
181 position = end;
182 if (word == "true" || word == "false") return Value(word == "true");
183 static const std::regex integer("0|-?[1-9][0-9]*");
184 if (!std::regex_match(word, integer)) throw std::runtime_error("invalid value");
185 return Value(std::stoll(word));
186}

188Value parse_value(const std::string& notation) {
189 try {
190 if (notation == "[]") return Value::container(Kind::List);
191 if (notation == "{}") return Value::container(Kind::Map);
192 std::size_t position = 0;
193 if (!notation.empty() && notation[0] == '[') {
194 Value result = Value::container(Kind::List);
195 position = 1;
196 while (true) {
197 result.elements.push_back(read_scalar(notation, position));
198 if (notation.substr(position) == "]") return result;
199 if (notation.compare(position, 2, ", ") != 0)
200 throw std::runtime_error("invalid value");
201 position += 2;
202 }
203 }
204 Value value = read_scalar(notation, position);
205 if (position != notation.size()) throw std::runtime_error("invalid value");
206 return value;
207 } catch (const std::exception&) {
208 throw std::runtime_error("bad value '" + notation + "'");
209 }
210}

212void assign(Value& root, const std::string& line) {
213 std::size_t separator = line.find(" = ");
214 if (separator == std::string::npos) throw std::runtime_error("bad line");
215 std::vector<Part> parts = parse_key(line.substr(0, separator));
216 Value value = parse_value(line.substr(separator + 3));
217 Value updated = root;
218 Value* container = &updated;
219 for (std::size_t i = 0; i < parts.size(); ++i) {
220 const Part& part = parts[i];
221 if (part.index && part.number > container->elements.size())
222 throw std::runtime_error("index gap at '" + part.prefix + "'");
223 if (i + 1 == parts.size()) {
224 put(*container, part, std::move(value));
225 break;
226 }
227 Kind required = parts[i + 1].index ? Kind::List : Kind::Map;
228 const Value* child = get(*container, part);
229 if (child && child->kind != required)
230 throw std::runtime_error("type conflict at '" + part.prefix + "'");
231 if (!child) put(*container, part, Value::container(required));
232 container = part.index ? &container->elements[part.number]
233 : &container->fields.at(part.key);
234 }
235 root = std::move(updated);
236}
237
238void unset(Value& root, const std::string& key) {
239 std::vector<Part> parts = parse_key(key);
240 Value* container = &root;
241 for (std::size_t i = 0; i < parts.size(); ++i) {
242 const Part& part = parts[i];
243 if (!get(*container, part)) throw std::runtime_error("no value at '" + key + "'");
244 if (i + 1 == parts.size()) {
245 if (part.index)
246 container->elements.erase(container->elements.begin() + part.number);
247 else
248 container->fields.erase(part.key);
249 return;
250 }
251 container = part.index ? &container->elements[part.number]
252 : &container->fields.at(part.key);
253 }
254}

256bool truthy(const Value& value) {
257 switch (value.kind) {
258 case Kind::Absent: return false;
259 case Kind::Integer: return value.integer != 0;
260 case Kind::Boolean: return value.boolean;
261 case Kind::String: return !value.text.empty();
262 case Kind::List: return !value.elements.empty();
263 case Kind::Map: return !value.fields.empty();
264 }
265 return false;
266}
267
268std::string stringify(const Value& value) {
269 switch (value.kind) {
270 case Kind::Absent: return "";
271 case Kind::Integer: return std::to_string(value.integer);
272 case Kind::String: return value.text;
273 case Kind::Boolean: return value.boolean ? "true" : "false";
274 case Kind::List: {
275 std::string result;
276 for (std::size_t i = 0; i < value.elements.size(); ++i) {
277 if (i) result += ", ";
278 result += stringify(value.elements[i]);
279 }
280 return result;
281 }
282 case Kind::Map: throw std::runtime_error("cannot render map");
283 }
284 return "";
285}

287Integer number(const Value& value, const std::string& filter) {
288 if (value.kind == Kind::Absent) return 0;
289 if (value.kind == Kind::Integer) return value.integer;
290 if (value.kind == Kind::String && numeric_text(value.text)) return std::stoll(value.text);
291 throw std::runtime_error("not a number for '" + filter + "'");
292}

303bool valid_arguments(const Filter& filter) {
304 const std::string& name = filter.name;
305 const std::vector<Value>& a = filter.arguments;
306 auto is = [&](std::size_t i, Kind kind) { return i < a.size() && a[i].kind == kind; };
307 if (name == "default" || name == "eq") return a.size() == 1;
308 if (name == "pad" || name == "lpad")
309 return (a.size() == 1 || a.size() == 2) && is(0, Kind::Integer) &&
310 a[0].integer >= 0 &&
311 (a.size() == 1 || (is(1, Kind::String) && a[1].text.size() == 1));
312 if (name == "truncate") return a.size() == 1 && is(0, Kind::Integer) && a[0].integer >= 0;
313 if (name == "replace")
314 return a.size() == 2 && is(0, Kind::String) && is(1, Kind::String) && !a[0].text.empty();
315 if (name == "join" || name == "pluck") return a.size() == 1 && is(0, Kind::String);
316 if (name == "sort") return a.empty() || (a.size() == 1 && is(0, Kind::String));
317 if (name == "div" || name == "mod")
318 return a.size() == 1 && is(0, Kind::Integer) && a[0].integer != 0;
319 if (name == "add" || name == "sub" || name == "mul" || name == "gt" || name == "lt")
320 return a.size() == 1 && is(0, Kind::Integer);
321 return a.empty();
322}

324Expression parse_expression(const std::string& text, int line) {
325 static const std::set<std::string> names = {
326 "upper", "lower", "title", "trim", "length", "first", "last", "reverse", "items",
327 "comma", "sum", "min", "max", "default", "eq", "pad", "lpad", "truncate", "replace",
328 "join", "sort", "pluck", "add", "sub", "mul", "div", "mod", "gt", "lt"
329 };
330 std::vector<std::string> pieces = split(text, '|', true);
331 Expression expression{trim(pieces[0]), {}};
332 if (!valid_name(expression.name))
333 throw TemplateError(line, "invalid name '" + expression.name + "'");
334 for (std::size_t i = 1; i < pieces.size(); ++i) {
335 std::vector<std::string> items = split(trim(pieces[i]), ':', true);
336 Filter filter{items[0], {}};
337 if (!names.count(filter.name))
338 throw TemplateError(line, "unknown filter '" + filter.name + "'");
339 bool valid = true;
340 for (std::size_t j = 1; j < items.size(); ++j) {
341 const std::string& argument = items[j];
342 if (argument.size() >= 2 && argument.front() == '\'' && argument.back() == '\'' &&
343 argument.find('\'', 1) == argument.size() - 1)
344 filter.arguments.emplace_back(argument.substr(1, argument.size() - 2));
345 else if (numeric_text(argument))
346 filter.arguments.emplace_back(std::stoll(argument));
347 else
348 valid = false;
349 }
350 if (!valid || !valid_arguments(filter))
351 throw TemplateError(line, "bad arguments for '" + filter.name + "'");
352 expression.filters.push_back(std::move(filter));
353 }
354 return expression;
355}

357struct Node {

360 std::string text;
361 std::string indentation;
362 Expression expression;
363 std::size_t closing = 0;
364};

366std::vector<Node> analyze(const Source& source) {
367 std::vector<Node> nodes;
368 std::vector<std::pair<std::string, int>> sections;
369 std::vector<std::size_t> openings;
370 std::string opening = "{{", closing = "}}";
371 for (const SourceLine& line : source) {
372 std::vector<Node> tokens;
373 std::size_t position = 0, tag_count = 0;
374 bool only_spaces = true;
375 while (true) {
376 std::size_t start = line.text.find(opening, position);
377 std::string text = line.text.substr(position, start == std::string::npos
378 ? std::string::npos : start - position);
379 if (!trim(text).empty()) only_spaces = false;
380 Node literal;
381 literal.text = std::move(text);
382 tokens.push_back(std::move(literal));
383 if (start == std::string::npos) break;
384 std::size_t after = start + opening.size();
385 bool triple = after < line.text.size() && line.text[after] == '{';
386 std::string terminator = (triple ? "}" : "") + closing;
387 std::size_t end = line.text.find(terminator, after + (triple ? 1 : 0));
388 if (end == std::string::npos) throw TemplateError(line.number, "unclosed tag");
389 std::string content = trim(line.text.substr(after + (triple ? 1 : 0),
390 end - after - (triple ? 1 : 0)));
391 position = end + terminator.size();
392 Node tag;
393 tag.line = line.number;
394 tag.kind = triple ? '&' : 'v';
395 tag.text = content;
396 if (!triple && !content.empty() && std::string("#^/!>&=").find(content[0]) != std::string::npos) {
397 tag.kind = content[0];
398 if (tag.kind != '=') tag.text = trim(content.substr(1));
399 }
400 if (tag.kind == '=') {
401 std::vector<std::string> delimiters;
402 if (content.size() >= 2 && content.back() == '=') {
403 std::istringstream input(content.substr(1, content.size() - 2));
404 std::string item;
405 while (input >> item) delimiters.push_back(item);
406 }
407 if (delimiters.size() != 2 || delimiters[0].find('=') != std::string::npos ||
408 delimiters[1].find('=') != std::string::npos)
409 throw TemplateError(line.number, "invalid delimiters");
410 opening = delimiters[0];
411 closing = delimiters[1];
412 } else if (tag.kind == 'v' || tag.kind == '&' || tag.kind == '#' || tag.kind == '^') {
413 tag.expression = parse_expression(tag.text, line.number);
414 if (tag.kind == '#' || tag.kind == '^')
415 sections.emplace_back(tag.expression.name, line.number);
416 } else if (tag.kind == '/') {
417 if (!valid_name(tag.text)) throw TemplateError(line.number, "invalid name '" + tag.text + "'");
418 if (sections.empty()) throw TemplateError(line.number, "unexpected closing '" + tag.text + "'");
419 if (sections.back().first != tag.text)
420 throw TemplateError(line.number, "mismatched closing '" + tag.text +
421 "', expected '" + sections.back().first + "'");
422 sections.pop_back();
423 } else if (tag.kind == '>' && !identifier(tag.text)) {
424 throw TemplateError(line.number, "invalid name '" + tag.text + "'");
425 }
426 tag.indentation = line.text.substr(0, start);
427 tokens.push_back(std::move(tag));
428 ++tag_count;
429 }
430 bool standalone = tag_count == 1 && only_spaces &&
431 std::string("#^/!>=").find(tokens[1].kind) != std::string::npos;
432 for (Node& node : tokens) {
433 if (node.kind == 't' && (standalone || node.text.empty())) continue;
434 if (node.kind == '>' && !standalone) node.indentation.clear();
435 if (node.kind == '#' || node.kind == '^') openings.push_back(nodes.size());
436 if (node.kind == '/') {
437 nodes[openings.back()].closing = nodes.size();
438 openings.pop_back();
439 }
440 nodes.push_back(std::move(node));
441 }
442 if (!standalone) {
443 Node newline;
444 newline.text = "\n";
445 nodes.push_back(std::move(newline));
446 }
447 }
448 if (!sections.empty())
449 throw TemplateError(sections.back().second, "unclosed section '" + sections.back().first + "'");
450 return nodes;
451}

461int sort_rank(const Value& value) {
462 switch (value.kind) {
463 case Kind::Absent: return 0;
464 case Kind::Integer: return 1;
465 case Kind::String: return 2;
466 case Kind::Boolean: return 3;
467 default: return 4;
468 }
469}
470
471bool less(const Value& left, const Value& right) {
472 int a = sort_rank(left), b = sort_rank(right);
473 if (a != b) return a < b;
474 if (a == 1) return left.integer < right.integer;
475 if (a == 2) return left.text < right.text;
476 if (a == 3) return left.boolean < right.boolean;
477 return false;
478}

480Value apply_filter(const Filter& filter, Value value) {
481 const std::string& name = filter.name;
482 const std::vector<Value>& a = filter.arguments;
483 if (name == "default") return truthy(value) ? value : a[0];
484 if (name == "eq") return Value(stringify(value) == stringify(a[0]));
485 if (name == "length") {
486 std::size_t length = value.kind == Kind::List ? value.elements.size()
487 : value.kind == Kind::Map ? value.fields.size() : stringify(value).size();
488 return Value(static_cast<Integer>(length));
489 }
490 if (name == "first" || name == "last") {
491 if (value.kind != Kind::List) return value;
492 if (value.elements.empty()) return Value();
493 return name == "first" ? value.elements.front() : value.elements.back();
494 }
495 if (name == "reverse") {
496 if (value.kind == Kind::List) std::reverse(value.elements.begin(), value.elements.end());
497 if (value.kind == Kind::String) std::reverse(value.text.begin(), value.text.end());
498 return value;
499 }
500 if (name == "items") {
501 if (value.kind != Kind::Map) return value;
502 Value result = Value::container(Kind::List);
503 for (const auto& entry : value.fields) {
504 Value item = Value::container(Kind::Map);
505 item.fields["key"] = Value(entry.first);
506 item.fields["value"] = entry.second;
507 result.elements.push_back(std::move(item));
508 }
509 return result;
510 }
511 if (name == "sort") {
512 if (value.kind != Kind::List) return value;
513 std::stable_sort(value.elements.begin(), value.elements.end(), [&](const Value& l, const Value& r) {
514 return a.empty() ? less(l, r) : less(field(l, a[0].text), field(r, a[0].text));
515 });
516 return value;
517 }
518 if (name == "pluck") {
519 if (value.kind != Kind::List) return value;
520 for (Value& element : value.elements) element = field(element, a[0].text);
521 return value;
522 }
523 if (name == "sum" || name == "min" || name == "max") {
524 if (value.kind != Kind::List) return Value(number(value, name));
525 if (value.elements.empty()) return name == "sum" ? Value(Integer{0}) : Value();
526 Integer result = 0;
527 for (std::size_t i = 0; i < value.elements.size(); ++i) {
528 Integer current = number(value.elements[i], name);
529 if (name == "sum") result += current;
530 else if (!i) result = current;
531 else if (name == "min") result = std::min(result, current);
532 else result = std::max(result, current);
533 }
534 return Value(result);
535 }
536 if (name == "comma") {
537 Integer n = number(value, name);
538 std::string digits = std::to_string(n < 0 ? -n : n), result;
539 for (std::size_t i = 0; i < digits.size(); ++i) {
540 if (i && (digits.size() - i) % 3 == 0) result += ',';
541 result += digits[i];
542 }
543 return Value((n < 0 ? "-" : "") + result);
544 }
545 if (name == "add" || name == "sub" || name == "mul" || name == "div" ||
546 name == "mod" || name == "gt" || name == "lt") {
547 Integer n = number(value, name), operand = a[0].integer;
548 if (name == "add") return Value(n + operand);
549 if (name == "sub") return Value(n - operand);
550 if (name == "mul") return Value(n * operand);
551 if (name == "gt") return Value(n > operand);
552 if (name == "lt") return Value(n < operand);
553 Integer quotient = n / operand;
554 if (n % operand != 0 && (n < 0) != (operand < 0)) --quotient;
555 return Value(name == "div" ? quotient : n - operand * quotient);
556 }
557 if (name == "join" && value.kind == Kind::List) {
558 std::string result;
559 for (std::size_t i = 0; i < value.elements.size(); ++i) {
560 if (i) result += a[0].text;
561 result += stringify(value.elements[i]);
562 }
563 return Value(result);
564 }
565 std::string text = stringify(value);
566 if (name == "upper" || name == "lower" || name == "title") {
567 for (std::size_t i = 0; i < text.size(); ++i) {
568 bool upper = name == "upper" || (name == "title" && (i == 0 || text[i - 1] == ' '));
569 if (upper && text[i] >= 'a' && text[i] <= 'z') text[i] -= 'a' - 'A';
570 if (!upper && text[i] >= 'A' && text[i] <= 'Z') text[i] += 'a' - 'A';
571 }
572 } else if (name == "trim") {
573 text = trim(text);
574 } else if (name == "pad" || name == "lpad") {
575 if (static_cast<Integer>(text.size()) < a[0].integer) {
576 std::string padding(static_cast<std::size_t>(a[0].integer) - text.size(),
577 a.size() == 2 ? a[1].text[0] : ' ');
578 text = name == "pad" ? text + padding : padding + text;
579 }
580 } else if (name == "truncate") {
581 if (static_cast<Integer>(text.size()) > a[0].integer)
582 text = text.substr(0, static_cast<std::size_t>(a[0].integer)) + "...";
583 } else if (name == "replace") {
584 std::string result;
585 std::size_t position = 0;
586 while (true) {
587 std::size_t found = text.find(a[0].text, position);
588 if (found == std::string::npos) {
589 result += text.substr(position);
590 break;
591 }
592 result += text.substr(position, found - position) + a[1].text;
593 position = found + a[0].text.size();
594 }
595 text = std::move(result);
596 }
597 return Value(text);
598}

600std::string escape_html(const std::string& text) {
601 std::string result;
602 for (char c : text) {
603 switch (c) {
604 case '&':
605 result += "&amp;";
606 break;
607 case '<':
608 result += "&lt;";
609 break;
610 case '>':
611 result += "&gt;";
612 break;
613 case '"':
614 result += "&quot;";
615 break;
616 case '\'':
617 result += "&#39;";
618 break;
619 default: result += c;
620 }
621 }
622 return result;
623}

625class Renderer {
626 const std::map<std::string, Source>& partials;
627 bool strict;
628 std::vector<Value> contexts;
629 std::vector<std::pair<std::size_t, std::size_t>> loops;
630 std::string output;
631
632 std::pair<bool, Value> resolve(const std::string& name) const {
633 if (name == ".") return {true, contexts.back()};
634 if (name[0] == '@') {
635 if (loops.empty()) return {false, Value()};
636 std::size_t i = loops.back().first, n = loops.back().second;
637 if (name == "@index") return {true, Value(static_cast<Integer>(i))};
638 if (name == "@number") return {true, Value(static_cast<Integer>(i + 1))};
639 return {true, Value(name == "@first" ? i == 0 : i + 1 == n)};
640 }
641 std::vector<Part> parts;
642 for (const std::string& token : split(name, '.')) {
643 bool index = subscript(token);
644 parts.push_back({index, token, index ? index_number(token) : 0, ""});
645 }
646 for (auto context = contexts.rbegin(); context != contexts.rend(); ++context) {
647 const Value* current = get(*context, parts[0]);
648 if (!current) continue;
649 for (std::size_t i = 1; i < parts.size() && current; ++i) current = get(*current, parts[i]);
650 return current ? std::make_pair(true, *current) : std::make_pair(false, Value());
651 }
652 return {false, Value()};
653 }
654
655 Value evaluate(const Expression& expression, bool require) const {
656 std::pair<bool, Value> found = resolve(expression.name);
657 if (!found.first && require) throw std::runtime_error("missing '" + expression.name + "'");
658 for (const Filter& filter : expression.filters)
659 found.second = apply_filter(filter, std::move(found.second));
660 return found.second;
661 }
662
663 void render(const std::vector<Node>& nodes, std::size_t begin, std::size_t end, int depth) {
664 for (std::size_t i = begin; i < end; ++i) {
665 const Node& node = nodes[i];
666 try {
667 if (node.kind == 't') {
668 output += node.text;
669 } else if (node.kind == 'v' || node.kind == '&') {
670 std::string text = stringify(evaluate(node.expression, strict));
671 output += node.kind == '&' ? text : escape_html(text);
672 } else if (node.kind == '#' || node.kind == '^') {
673 Value value = evaluate(node.expression, false);
674 if (node.kind == '^') {
675 if (!truthy(value)) render(nodes, i + 1, node.closing, depth);
676 } else if (value.kind == Kind::List) {
677 for (std::size_t j = 0; j < value.elements.size(); ++j) {
678 contexts.push_back(value.elements[j]);
679 loops.emplace_back(j, value.elements.size());
680 render(nodes, i + 1, node.closing, depth);
681 loops.pop_back();
682 contexts.pop_back();
683 }
684 } else if (truthy(value)) {
685 contexts.push_back(std::move(value));
686 render(nodes, i + 1, node.closing, depth);
687 contexts.pop_back();
688 }
689 i = node.closing;
690 } else if (node.kind == '>') {
691 auto found = partials.find(node.text);
692 if (found == partials.end()) throw std::runtime_error("unknown partial '" + node.text + "'");
693 if (depth == 8) throw std::runtime_error("partial depth exceeded at '" + node.text + "'");
694 Source source = found->second;
695 for (SourceLine& line : source)
696 if (!line.text.empty()) line.text = node.indentation + line.text;
697 std::vector<Node> partial_nodes = analyze(source);
698 render(partial_nodes, 0, partial_nodes.size(), depth + 1);
699 }
700 } catch (const TemplateError&) {
701 throw;
702 } catch (const std::runtime_error& error) {
703 throw TemplateError(node.line, error.what());
704 }
705 }
706 }
707
708public:
709 Renderer(const Value& root, const std::map<std::string, Source>& partials, bool strict)
710 : partials(partials), strict(strict), contexts{root} {}
711
712 std::string run(const std::vector<Node>& nodes) {
713 render(nodes, 0, nodes.size(), 0);
714 return output;
715 }
716};
717
718std::string notation(const Value& value) {
719 if (value.kind == Kind::String) {
720 std::string text = "\"";
721 for (char c : value.text) {
722 if (c == '\\' || c == '"') text += '\\';
723 text += c == '\n' ? "\\n" : std::string(1, c);
724 }
725 return text + '"';
726 }
727 if (value.kind == Kind::List) return "[]";
728 if (value.kind == Kind::Map) return "{}";
729 return stringify(value);
730}
731
732void leaves(const Value& value, const std::string& path, std::vector<std::string>& output) {
733 if (value.kind == Kind::Map && !value.fields.empty()) {
734 for (const auto& entry : value.fields)
735 leaves(entry.second, path.empty() ? entry.first : path + "." + entry.first, output);
736 } else if (value.kind == Kind::List && !value.elements.empty()) {
737 for (std::size_t i = 0; i < value.elements.size(); ++i)
738 leaves(value.elements[i], path + "[" + std::to_string(i) + "]", output);
739 } else if (!path.empty()) {
740 output.push_back(path + " = " + notation(value));
741 }
742}
743
744void render_template(const std::string& name, const Source& source, const Value& root,
745 const std::map<std::string, Source>& partials, bool strict) {
746 try {
747 std::vector<Node> nodes = analyze(source);
748 std::string result = Renderer(root, partials, strict).run(nodes);
749 std::size_t lines = static_cast<std::size_t>(std::count(result.begin(), result.end(), '\n'));
750 if (!result.empty() && result.back() != '\n') ++lines;
751 std::cout << "TEMPLATE " << name << " OK lines=" << lines << " chars=" << result.size() << '\n';
752 std::cout << result;
753 if (!result.empty() && result.back() != '\n') std::cout << '\n';
754 } catch (const TemplateError& error) {
755 std::cout << "TEMPLATE " << name << " ERROR line " << error.line << ": " << error.what() << '\n';
756 }
757}

759int main() {
760 Value root = Value::container(Kind::Map);
761 std::map<std::string, Source> partials, templates;
762 bool strict = false;
763 int number = 0;
764 std::string command;
765 while (std::getline(std::cin, command)) {
766 ++number;
767 if (command == "END") break;
768 if (command == "DATA" || command.compare(0, 8, "PARTIAL ") == 0 ||
769 command.compare(0, 9, "TEMPLATE ") == 0) {
770 std::string marker = command == "DATA" ? "ENDDATA"
771 : command[0] == 'P' ? "ENDPARTIAL" : "ENDTEMPLATE";
772 Source body;
773 std::string line;
774 while (std::getline(std::cin, line)) {
775 ++number;
776 if (line == marker) break;
777 body.push_back({number, line});
778 }
779 if (command == "DATA") {
780 int set_count = 0, unset_count = 0;
781 std::vector<std::string> errors;
782 for (const SourceLine& data : body) {
783 if (data.text.empty() || data.text[0] == '#') continue;
784 try {
785 if (data.text.compare(0, 6, "UNSET ") == 0) {
786 unset(root, data.text.substr(6));
787 ++unset_count;
788 } else {
789 assign(root, data.text);
790 ++set_count;
791 }
792 } catch (const std::runtime_error& error) {
793 errors.push_back("ERROR line " + std::to_string(data.number) + ": " + error.what());
794 }
795 }
796 std::cout << "DATA set=" << set_count << " unset=" << unset_count << " errors=" << errors.size() << '\n';
797 for (const std::string& error : errors) std::cout << error << '\n';
798 } else if (command[0] == 'P') {
799 std::string name = command.substr(8);
800 try {
801 analyze(body);
802 bool replaced = partials.count(name) != 0;
803 partials[name] = body;
804 std::cout << "PARTIAL " << name << (replaced ? " REPLACED" : " DEFINED")
805 << " lines=" << body.size() << '\n';
806 } catch (const TemplateError& error) {
807 std::cout << "PARTIAL " << name << " ERROR line " << error.line << ": " << error.what() << '\n';
808 }
809 } else {
810 std::string name = command.substr(9);
811 templates[name] = body;
812 render_template(name, body, root, partials, strict);
813 }
814 } else if (command.compare(0, 7, "RENDER ") == 0) {
815 std::string name = command.substr(7);
816 auto found = templates.find(name);
817 if (found == templates.end())
818 std::cout << "TEMPLATE " << name << " ERROR line " << number << ": unknown template '" << name << "'\n";
819 else
820 render_template(name, found->second, root, partials, strict);
821 } else if (command == "DUMP") {
822 std::vector<std::string> output;
823 leaves(root, "", output);
824 std::cout << "DUMP leaves=" << output.size() << '\n';
825 for (const std::string& line : output) std::cout << line << '\n';
826 } else if (command == "RESET") {
827 root = Value::container(Kind::Map);
828 partials.clear();
829 templates.clear();
830 std::cout << "RESET\n";
831 } else if (command.compare(0, 7, "STRICT ") == 0) {
832 strict = command == "STRICT ON";
833 std::cout << command << '\n';
834 }
835 }
