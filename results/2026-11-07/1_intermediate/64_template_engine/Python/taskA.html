6IDENTIFIER = r"[A-Za-z_][A-Za-z0-9_]*"
7SUBSCRIPT = r"0|[1-9][0-9]*"

10KEY_PATTERN = re.compile(rf"{IDENTIFIER}(?:\.{IDENTIFIER}|\[(?:{SUBSCRIPT})\])*")
11KEY_PART_PATTERN = re.compile(rf"({IDENTIFIER})|\[({SUBSCRIPT})\]")
12NAME_PATTERN = re.compile(rf"\.|@(?:index|number|first|last)|(?:{SEGMENT})(?:\.(?:{SEGMENT}))*")
13INTEGER_PATTERN = re.compile(r"0|-?[1-9][0-9]*")
14NUMBER_TEXT_PATTERN = re.compile(r"-?[0-9]{1,9}")
15STRING_ARGUMENT_PATTERN = re.compile(r"'[^']*'")

79def has_part(value, part):
80 if isinstance(part, str):
81 return isinstance(value, dict) and part in value
82 return isinstance(value, list) and part < len(value)

85def put(container, part, value):
86 if isinstance(container, list) and part == len(container):
87 container.append(value)

92def parse_key(key):
93 if not KEY_PATTERN.fullmatch(key):
94 raise DataError(f"bad key '{key}'")
95 parts = []
96 for match in KEY_PART_PATTERN.finditer(key):
97 identifier, subscript = match.groups()
98 parts.append((identifier if identifier else int(subscript), key[: match.end()]))
99 return parts

102def read_string(text, start):
103 characters = []
104 position = start
105 while position < len(text):
106 char = text[position]
107 if char == '"':
108 return "".join(characters), position + 1
109 if char == "\\":
110 escaped = STRING_ESCAPES.get(text[position + 1 : position + 2])
111 if escaped is None:
112 raise ValueError
113 characters.append(escaped)
114 position += 2
115 else:
116 characters.append(char)
117 position += 1
118 raise ValueError

121def read_scalar(text, start):
122 if text.startswith('"', start):
123 return read_string(text, start + 1)
124 end = start
125 while end < len(text) and text[end] not in ",]":
126 end += 1
127 word = text[start:end]
128 if word in ("true", "false"):
129 return word == "true", end
130 if INTEGER_PATTERN.fullmatch(word):
131 return int(word), end
132 raise ValueError

135def read_list(text):
136 elements = []
137 position = 1
138 while True:
139 element, position = read_scalar(text, position)
140 elements.append(element)
141 if text.startswith(", ", position):
142 position += 2
143 elif text[position:] == "]":
144 return elements
145 else:
146 raise ValueError

149def parse_value(notation):
150 try:
151 if notation == "[]":
152 return []
153 if notation == "{}":
154 return {}
155 if notation.startswith("["):
156 return read_list(notation)
157 value, end = read_scalar(notation, 0)
158 if end != len(notation):
159 raise ValueError
160 return value
161 except ValueError:
162 raise DataError(f"bad value '{notation}'") from None

165def notation_of(value):
166 if isinstance(value, str):
167 escaped = value.replace("\\", "\\\\").replace('"', '\\"').replace("\n", "\\n")
168 return f'"{escaped}"'
169 if isinstance(value, bool):
170 return "true" if value else "false"
171 if isinstance(value, int):
172 return str(value)
173 return "[]" if isinstance(value, list) else "{}"

176def collect_leaves(value, path, leaves):
177 if isinstance(value, dict) and value:
178 for key in sorted(value):
179 collect_leaves(value[key], f"{path}.{key}", leaves)
180 elif isinstance(value, list) and value:
181 for index, element in enumerate(value):
182 collect_leaves(element, f"{path}[{index}]", leaves)
183 else:
184 leaves.append(f"{path} = {notation_of(value)}")

187def split_unquoted(text, separator):
188 pieces = []
189 start = 0
190 quoted = False
191 for position, char in enumerate(text):
192 if char == "'":
193 quoted = not quoted
194 elif char == separator and not quoted:
195 pieces.append(text[start:position])
196 start = position + 1
197 pieces.append(text[start:])
198 return pieces

201def parse_argument(text):
202 if STRING_ARGUMENT_PATTERN.fullmatch(text):
203 return text[1:-1]
204 if NUMBER_TEXT_PATTERN.fullmatch(text):
205 return int(text)
206 return None

209def valid_arguments(name, arguments):
210 kinds = tuple(type(argument) for argument in arguments)
211 match name:
212 case "default" | "eq":
213 return len(arguments) == 1
214 case "pad" | "lpad":
215 if kinds not in ((int,), (int, str)) or arguments[0] < 0:
216 return False
217 return len(arguments) == 1 or len(arguments[1]) == 1
218 case "truncate":
219 return kinds == (int,) and arguments[0] >= 0
220 case "replace":
221 return kinds == (str, str) and arguments[0] != ""
222 case "join" | "pluck":
223 return kinds == (str,)
224 case "sort":
225 return kinds in ((), (str,))
226 case "div" | "mod":
227 return kinds == (int,) and arguments[0] != 0
228 case "add" | "sub" | "mul" | "gt" | "lt":
229 return kinds == (int,)
230 case _:
231 return not arguments

234def parse_expression(text, number):
235 pieces = split_unquoted(text, "|")
236 name = pieces[0].strip()
237 if not NAME_PATTERN.fullmatch(name):
238 raise TemplateError(number, f"invalid name '{name}'")
239 filters = []
240 for piece in pieces[1:]:
241 filter_name, *texts = split_unquoted(piece.strip(), ":")
242 if filter_name not in FILTERS:
243 raise TemplateError(number, f"unknown filter '{filter_name}'")
244 arguments = [parse_argument(argument) for argument in texts]
245 if None in arguments or not valid_arguments(filter_name, arguments):
246 raise TemplateError(number, f"bad arguments for '{filter_name}'")
247 filters.append((filter_name, arguments))
248 return Expression(name, filters)

251def read_tag(line, start, opening, closing, number):
252 after = start + len(opening)
253 if line.startswith("{", after):
254 end = line.find("}" + closing, after + 1)
255 if end < 0:
256 raise TemplateError(number, "unclosed tag")
257 return Tag("raw", number, line[after + 1 : end].strip()), end + 1 + len(closing)
258 end = line.find(closing, after)
259 if end < 0:
260 raise TemplateError(number, "unclosed tag")
261 content = line[after:end].strip()
262 kind = SIGILS.get(content[:1], "variable")
263 if kind in ("variable", "delimiter"):
264 return Tag(kind, number, content), end + len(closing)
265 return Tag(kind, number, content[1:].strip()), end + len(closing)

268def parse_delimiters(content, number):
269 items = content[1:-1].split()
270 if (
271 len(content) < 2
272 or not content.endswith("=")
273 or len(items) != 2
274 or any("=" in item for item in items)
275 ):
276 raise TemplateError(number, "invalid delimiters")
277 return items

280def check_tag(tag, open_sections):
281 match tag.kind:
282 case "variable" | "raw":
283 tag.expression = parse_expression(tag.argument, tag.number)
284 case "section" | "inverted":
285 tag.expression = parse_expression(tag.argument, tag.number)
286 open_sections.append(tag)
287 case "close":
288 name = tag.argument
289 if not NAME_PATTERN.fullmatch(name):
290 raise TemplateError(tag.number, f"invalid name '{name}'")
291 if not open_sections:
292 raise TemplateError(tag.number, f"unexpected closing '{name}'")
293 expected = open_sections[-1].expression.name
294 if name != expected:
295 raise TemplateError(
296 tag.number, f"mismatched closing '{name}', expected '{expected}'"
297 )
298 open_sections.pop()
299 case "partial":
300 if not IDENTIFIER_PATTERN.fullmatch(tag.argument):
301 raise TemplateError(tag.number, f"invalid name '{tag.argument}'")

304def place_tokens(tokens, targets):
305 texts = tokens[0::2]
306 tags = tokens[1::2]
307 if len(tags) == 1 and tags[0].kind in STANDALONE_KINDS and not "".join(texts).strip():
308 if tags[0].kind == "partial":
309 tags[0].indentation = texts[0]
310 tokens = tags
311 else:
312 tokens.append("\n")
313 for token in tokens:
314 if isinstance(token, str):
315 if token:
316 targets[-1].append(token)
317 elif token.kind in ("section", "inverted"):
318 targets[-1].append(token)
319 targets.append(token.children)
320 elif token.kind == "close":
321 targets.pop()
322 elif token.kind not in ("comment", "delimiter"):
323 targets[-1].append(token)

326def parse_source(lines):
327 opening, closing = "{{", "}}"
328 open_sections = []
329 nodes = []
330 targets = [nodes]
331 for number, line in lines:
332 tokens = []
333 position = 0
334 while True:
335 start = line.find(opening, position)
336 if start < 0:
337 tokens.append(line[position:])
338 break
339 tokens.append(line[position:start])
340 tag, position = read_tag(line, start, opening, closing, number)
341 if tag.kind == "delimiter":
342 opening, closing = parse_delimiters(tag.argument, number)
343 else:
344 check_tag(tag, open_sections)
345 tokens.append(tag)
346 place_tokens(tokens, targets)
347 if open_sections:
348 innermost = open_sections[-1]
349 raise TemplateError(innermost.number, f"unclosed section '{innermost.expression.name}'")
350 return nodes

353def to_string(value):
354 match value:
355 case None:
356 return ""
357 case bool():
358 return "true" if value else "false"
359 case int():
360 return str(value)
361 case str():
362 return value
363 case list():
364 return ", ".join(to_string(element) for element in value)
365 case _:
366 raise RenderError("cannot render map")

369def to_number(value, filter_name):
370 if value is None:
371 return 0
372 if type(value) is int:
373 return value
374 if isinstance(value, str) and NUMBER_TEXT_PATTERN.fullmatch(value):
375 return int(value)
376 raise RenderError(f"not a number for '{filter_name}'")

379def sort_rank(value):
380 match value:
381 case None:
382 return (0, 0)
383 case bool():
384 return (3, value)
385 case int():
386 return (1, value)
387 case str():
388 return (2, value)
389 case _:
390 return (4, 0)

397def title_case(text):
398 return "".join(
399 char.upper() if index == 0 or text[index - 1] == " " else char.lower()
400 for index, char in enumerate(text)
401 )

404def aggregate(name, value):
405 if not isinstance(value, list):
406 return to_number(value, name)
407 numbers = [to_number(element, name) for element in value]
408 if name == "sum":
409 return sum(numbers)
410 if not numbers:
411 return None
412 return min(numbers) if name == "min" else max(numbers)

415def apply_filter(name, arguments, value):
416 match name:
417 case "upper":
418 return to_string(value).upper()
419 case "lower":
420 return to_string(value).lower()
421 case "title":
422 return title_case(to_string(value))
423 case "trim":
424 return to_string(value).strip(" ")
425 case "length":
426 if isinstance(value, (list, dict)):
427 return len(value)
428 return len(to_string(value))
429 case "first" | "last":
430 if not isinstance(value, list):
431 return value
432 if not value:
433 return None
434 return value[0] if name == "first" else value[-1]
435 case "reverse":
436 return value[::-1] if isinstance(value, (list, str)) else value
437 case "items":
438 if not isinstance(value, dict):
439 return value
440 return [{"key": key, "value": value[key]} for key in sorted(value)]
441 case "comma":
442 return f"{to_number(value, name):,}"
443 case "sum" | "min" | "max":
444 return aggregate(name, value)
445 case "default":
446 return value or arguments[0]
447 case "eq":
448 return to_string(value) == to_string(arguments[0])
449 case "pad" | "lpad":
450 text = to_string(value)
451 fill = arguments[1] if len(arguments) == 2 else " "
452 if name == "pad":
453 return text.ljust(arguments[0], fill)
454 return text.rjust(arguments[0], fill)
455 case "truncate":
456 text = to_string(value)
457 limit = arguments[0]
458 return text[:limit] + "..." if len(text) > limit else text
459 case "replace":
460 return to_string(value).replace(arguments[0], arguments[1])
461 case "join":
462 if not isinstance(value, list):
463 return to_string(value)
464 return arguments[0].join(to_string(element) for element in value)
465 case "sort":
466 if not isinstance(value, list):
467 return value
468 if not arguments:
469 return sorted(value, key=sort_rank)
470 return sorted(value, key=lambda element: sort_rank(field_of(element, arguments[0])))
471 case "pluck":
472 if not isinstance(value, list):
473 return value
474 return [field_of(element, arguments[0]) for element in value]
475 case _:
476 return ARITHMETIC[name](to_number(value, name), arguments[0])

479class Renderer:
480 def __init__(self, root, partials, strict):
481 self.partials = partials
482 self.strict = strict
483 self.contexts = [root]
484 self.loops = []
485 self.output = []
486
487 def render(self, nodes, depth):
488 for node in nodes:
489 if isinstance(node, str):
490 self.output.append(node)
491 continue
492 try:
493 self.render_tag(node, depth)
494 except RenderError as error:
495 raise TemplateError(node.number, str(error)) from None
496
497 def render_tag(self, tag, depth):
498 match tag.kind:
499 case "variable" | "raw":
500 text = to_string(self.evaluate(tag.expression, self.strict))
501 self.output.append(text if tag.kind == "raw" else text.translate(HTML_ESCAPES))
502 case "section":
503 self.render_section(tag, depth)
504 case "inverted":
505 if not self.evaluate(tag.expression, False):
506 self.render(tag.children, depth)
507 case "partial":
508 self.render_partial(tag, depth)
509
510 def render_section(self, tag, depth):
511 value = self.evaluate(tag.expression, False)
512 if isinstance(value, list):
513 for index, element in enumerate(value):
514 self.contexts.append(element)
515 self.loops.append((index, len(value)))
516 self.render(tag.children, depth)
517 self.contexts.pop()
518 self.loops.pop()
519 elif value:
520 self.contexts.append(value)
521 self.render(tag.children, depth)
522 self.contexts.pop()
523
524 def render_partial(self, tag, depth):
525 name = tag.argument
526 if name not in self.partials:
527 raise RenderError(f"unknown partial '{name}'")
528 if depth == MAX_PARTIAL_DEPTH:
529 raise RenderError(f"partial depth exceeded at '{name}'")
530 source = [
531 (number, tag.indentation + text if text else text)
532 for number, text in self.partials[name]
533 ]
534 self.render(parse_source(source), depth + 1)
535
536 def evaluate(self, expression, strict):
537 found, value = self.resolve(expression.name)
538 if not found and strict:
539 raise RenderError(f"missing '{expression.name}'")
540 for name, arguments in expression.filters:
541 value = apply_filter(name, arguments, value)
542 return value
543
544 def resolve(self, name):
545 if name == ".":
546 return True, self.contexts[-1]
547 if name.startswith("@"):
548 if not self.loops:
549 return False, None
550 index, count = self.loops[-1]
551 variables = {
552 "@index": index,
553 "@number": index + 1,
554 "@first": index == 0,
555 "@last": index == count - 1,
556 }
557 return True, variables[name]
558 parts = [int(part) if part.isdigit() else part for part in name.split(".")]
559 for context in reversed(self.contexts):
560 if has_part(context, parts[0]):
561 value = context
562 for part in parts:
563 if not has_part(value, part):
564 return False, None
565 value = value[part]
566 return True, value
567 return False, None

571 end = lines.index(marker, start)
572 return list(enumerate(lines[start:end], start + 1)), end + 1

575class TemplateEngine:
576 def __init__(self):
577 self.strict = False
578 self.reset()
579
580 def reset(self):
581 self.root = {}
582 self.partials = {}
583 self.templates = {}
584
585 def run(self, lines):
586 position = 0
587 while lines[position] != "END":
588 line = lines[position]
589 number = position + 1
590 command, _, name = line.partition(" ")
591 position = number
592 if line == "DATA":
593 body, position = read_block(lines, position, "ENDDATA")
594 self.load_data(body)
595 elif command == "PARTIAL":
596 body, position = read_block(lines, position, "ENDPARTIAL")
597 self.define_partial(name, body)
598 elif command == "TEMPLATE":
599 body, position = read_block(lines, position, "ENDTEMPLATE")
600 self.templates[name] = body
601 self.render(name)
602 elif command == "RENDER":
603 if name in self.templates:
604 self.render(name)
605 else:
606 print(f"TEMPLATE {name} ERROR line {number}: unknown template '{name}'")
607 elif line == "DUMP":
608 self.dump()
609 elif command == "STRICT":
610 self.strict = name == "ON"
611 print(line)
612 elif line == "RESET":
613 self.reset()
614 print("RESET")
615
616 def load_data(self, body):
617 assigned = 0
618 unset = 0
619 errors = []
620 for number, line in body:
621 if not line or line.startswith("#"):
622 continue
623 try:
624 if line.startswith("UNSET "):
625 self.unset(line[6:])
626 unset += 1
627 else:
628 self.assign(line)
629 assigned += 1
630 except DataError as error:
631 errors.append(f"ERROR line {number}: {error}")
632 print(f"DATA set={assigned} unset={unset} errors={len(errors)}")
633 for error in errors:
634 print(error)
635
636 def assign(self, line):
637 key, separator, notation = line.partition(" = ")
638 if not separator:
639 raise DataError("bad line")
640 parts = parse_key(key)
641 value = parse_value(notation)
642 container = self.root
643 detached = None
644 for position, (part, prefix) in enumerate(parts):
645 if isinstance(part, int) and part > len(container):
646 raise DataError(f"index gap at '{prefix}'")
647 if position == len(parts) - 1:
648 put(container, part, value)
649 break
650 required = dict if isinstance(parts[position + 1][0], str) else list
651 if has_part(container, part):
652 child = container[part]
653 if not isinstance(child, required):
654 raise DataError(f"type conflict at '{prefix}'")
655 else:
656 child = required()
657 if detached is None:
658 detached = (container, part, child)
659 else:
660 put(container, part, child)
661 container = child
662 if detached is not None:
663 put(*detached)
664
665 def unset(self, key):
666 parts = parse_key(key)
667 parent = target = self.root
668 for part, _ in parts:
669 if not has_part(target, part):
670 raise DataError(f"no value at '{key}'")
671 parent, target = target, target[part]
672 del parent[part]
673
674 def define_partial(self, name, body):
675 try:
676 parse_source(body)
677 except TemplateError as error:
678 print(f"PARTIAL {name} ERROR line {error.number}: {error.reason}")
679 return
680 status = "REPLACED" if name in self.partials else "DEFINED"
681 self.partials[name] = body
682 print(f"PARTIAL {name} {status} lines={len(body)}")
683
684 def render(self, name):
685 renderer = Renderer(self.root, self.partials, self.strict)
686 try:
687 renderer.render(parse_source(self.templates[name]), 0)
688 except TemplateError as error:
689 print(f"TEMPLATE {name} ERROR line {error.number}: {error.reason}")
690 return
691 result = "".join(renderer.output)
692 lines = result.split("\n")
693 if lines[-1] == "":
694 lines.pop()
695 print(f"TEMPLATE {name} OK lines={len(lines)} chars={len(result)}")
696 for line in lines:
697 print(line)
698
699 def dump(self):
700 leaves = []
701 for key in sorted(self.root):
702 collect_leaves(self.root[key], key, leaves)
703 print(f"DUMP leaves={len(leaves)}")
704 for leaf in leaves:
705 print(leaf)
