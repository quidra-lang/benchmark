11const identifier = /^[A-Za-z_][A-Za-z0-9_]*$/;
12const keyPattern = /^[A-Za-z_][A-Za-z0-9_]*(?:\.[A-Za-z_][A-Za-z0-9_]*|\[(?:0|[1-9][0-9]*)\])*$/;
13const namePattern = /^(?:\.|@(?:index|number|first|last)|(?:[A-Za-z_][A-Za-z0-9_]*|0|[1-9][0-9]*)(?:\.(?:[A-Za-z_][A-Za-z0-9_]*|0|[1-9][0-9]*))*)$/;
14const integerPattern = /^(?:0|-?[1-9][0-9]*)$/;
15const numericPattern = /^-?[0-9]{1,9}$/;
16const noArguments = new Set(['upper', 'lower', 'title', 'trim', 'length', 'first', 'last', 'reverse', 'items', 'comma', 'sum', 'min', 'max']);
17const numericFilters = new Set(['add', 'sub', 'mul', 'div', 'mod', 'gt', 'lt']);
18const filters = new Set([...noArguments, ...numericFilters, 'default', 'eq', 'pad', 'lpad', 'truncate', 'replace', 'join', 'sort', 'pluck']);
19const standaloneKinds = new Set(['section', 'inverted', 'close', 'comment', 'partial', 'delimiter']);
20const sigils: Record<string, string> = { '#': 'section', '^': 'inverted', '/': 'close', '!': 'comment', '>': 'partial', '&': 'raw', '=': 'delimiter' };
21function strip(text: string): string {
22 let start = 0;
23 let end = text.length;
24 while (start < end && text[start] === ' ') start++;
25 while (end > start && text[end - 1] === ' ') end--;
26 return text.slice(start, end);
27}

43function hasPart(value: Value, part: Part): boolean {
44 return typeof part === 'string' ? value instanceof Map && value.has(part) : Array.isArray(value) && part < value.length;
45}
46
47function getPart(value: Value, part: Part): Value {
48 return typeof part === 'string' ? (value as Map<string, Value>).get(part)! : (value as Value[])[part];
49}
50
51function putPart(container: Value, part: Part, value: Value): void {
52 if (typeof part === 'string') (container as Map<string, Value>).set(part, value);
53 else (container as Value[])[part] = value;
54}

56function parseKey(key: string): { part: Part; prefix: string }[] {
57 if (!keyPattern.test(key)) throw new Error(`bad key '${key}'`);
58 return [...key.matchAll(/([A-Za-z_][A-Za-z0-9_]*)|\[(0|[1-9][0-9]*)\]/g)].map(match => ({
59 part: match[1] === undefined ? Number(match[2]) : match[1],
60 prefix: key.slice(0, match.index! + match[0].length),
61 }));
62}

64function readScalar(text: string, start: number): [Value, number] {
65 if (text[start] === '"') {
66 let result = '';
67 for (let position = start + 1; position < text.length; position++) {
68 const char = text[position];
69 if (char === '"') return [result, position + 1];
70 if (char === '\\') {
71 const next = text[++position];
72 if (next !== '"' && next !== '\\' && next !== 'n') throw new Error();
73 result += next === 'n' ? '\n' : next;
74 } else result += char;
75 }
76 throw new Error();
77 }
78 let end = start;
79 while (end < text.length && text[end] !== ',' && text[end] !== ']') end++;
80 const word = text.slice(start, end);
81 if (word === 'true' || word === 'false') return [word === 'true', end];
82 if (integerPattern.test(word)) return [Number(word), end];
83 throw new Error();
84}

86function parseValue(text: string): Value {
87 try {
88 if (text === '[]') return [];
89 if (text === '{}') return new Map<string, Value>();
90 if (text.startsWith('[')) {
91 const values: Value[] = [];
92 let position = 1;
93 while (true) {
94 const [value, end] = readScalar(text, position);
95 values.push(value);
96 if (text.slice(end) === ']') return values;
97 if (text.slice(end, end + 2) !== ', ') throw new Error();
98 position = end + 2;
99 }
100 }
101 const [value, end] = readScalar(text, 0);
102 if (end !== text.length) throw new Error();
103 return value;
104 } catch {
105 throw new Error(`bad value '${text}'`);
106 }
107}

109function splitUnquoted(text: string, separator: string): string[] {
110 const parts: string[] = [];
111 let quoted = false;
112 let start = 0;
113 for (let index = 0; index < text.length; index++) {
114 if (text[index] === "'") quoted = !quoted;
115 else if (text[index] === separator && !quoted) {
116 parts.push(text.slice(start, index));
117 start = index + 1;
118 }
119 }
120 parts.push(text.slice(start));
121 return parts;
122}

124function validArguments(name: string, args: Argument[]): boolean {
125 const number = args.length > 0 && typeof args[0] === 'number';
126 const string = args.length > 0 && typeof args[0] === 'string';
127 if (noArguments.has(name)) return args.length === 0;
128 if (name === 'default' || name === 'eq') return args.length === 1;
129 if (name === 'sort') return args.length === 0 || args.length === 1 && string;
130 if (name === 'join' || name === 'pluck') return args.length === 1 && string;
131 if (name === 'replace') return args.length === 2 && string && args[0] !== '' && typeof args[1] === 'string';
132 if (name === 'pad' || name === 'lpad') {
133 return number && (args[0] as number) >= 0 && (args.length === 1 || args.length === 2 && typeof args[1] === 'string' && args[1].length === 1);
134 }
135 if (name === 'truncate') return args.length === 1 && number && (args[0] as number) >= 0;
136 return args.length === 1 && number && (!(name === 'div' || name === 'mod') || args[0] !== 0);
137}

139function parseExpression(text: string, number: number): Expression {
140 const parts = splitUnquoted(text, '|');
141 const name = strip(parts[0]);
142 if (!namePattern.test(name)) throw new TemplateError(number, `invalid name '${name}'`);
143 const chain: Filter[] = [];
144 for (const part of parts.slice(1)) {
145 const [filter, ...notations] = splitUnquoted(strip(part), ':');
146 if (!filters.has(filter)) throw new TemplateError(number, `unknown filter '${filter}'`);
147 const args: Argument[] = [];
148 let valid = true;
149 for (const notation of notations) {
150 if (/^'[^']*'$/.test(notation)) args.push(notation.slice(1, -1));
151 else if (numericPattern.test(notation)) args.push(Number(notation));
152 else valid = false;
153 }
154 if (!valid || !validArguments(filter, args)) throw new TemplateError(number, `bad arguments for '${filter}'`);
155 chain.push({ name: filter, args });
156 }
157 return { name, filters: chain };
158}

160function parseSource(source: SourceLine[]): Node[] {
161 let opening = '{{';
162 let closing = '}}';
163 const nodes: Node[] = [];
164 const targets: Node[][] = [nodes];
165 const sections: Tag[] = [];
166 for (const { number, text } of source) {
167 let position = 0;
168 const tokens: Node[] = [];
169 const lineTags: Tag[] = [];
170 while (true) {
171 const start = text.indexOf(opening, position);
172 if (start < 0) {
173 tokens.push(text.slice(position));
174 break;
175 }
176 tokens.push(text.slice(position, start));
177 const after = start + opening.length;
178 const triple = text[after] === '{';
179 const terminator = triple ? '}' + closing : closing;
180 const end = text.indexOf(terminator, after + (triple ? 1 : 0));
181 if (end < 0) throw new TemplateError(number, 'unclosed tag');
182 const content = strip(text.slice(after + (triple ? 1 : 0), end));
183 const kind = triple ? 'raw' : sigils[content[0]] ?? 'variable';
184 const argument = triple || kind === 'variable' || kind === 'delimiter' ? content : strip(content.slice(1));
185 const tag = new Tag(kind, number, argument);
186 position = end + terminator.length;
187 if (kind === 'delimiter') {
188 const words = strip(content.slice(1, -1)).split(/ +/);
189 if (content.length < 2 || !content.endsWith('=') || words.length !== 2 || words.some(word => word.includes('='))) {
190 throw new TemplateError(number, 'invalid delimiters');
191 }
192 [opening, closing] = words;
193 } else if (kind === 'variable' || kind === 'raw' || kind === 'section' || kind === 'inverted') {
194 tag.expression = parseExpression(argument, number);
195 if (kind === 'section' || kind === 'inverted') sections.push(tag);
196 } else if (kind === 'close') {
197 if (!namePattern.test(argument)) throw new TemplateError(number, `invalid name '${argument}'`);
198 if (sections.length === 0) throw new TemplateError(number, `unexpected closing '${argument}'`);
199 const expected = sections[sections.length - 1].expression!.name;
200 if (argument !== expected) throw new TemplateError(number, `mismatched closing '${argument}', expected '${expected}'`);
201 sections.pop();
202 } else if (kind === 'partial' && !identifier.test(argument)) {
203 throw new TemplateError(number, `invalid name '${argument}'`);
204 }
205 lineTags.push(tag);
206 tokens.push(tag);
207 }
208 const standalone = lineTags.length === 1 && standaloneKinds.has(lineTags[0].kind) && tokens.every(token => typeof token !== 'string' || /^ *$/.test(token));
209 let placed: Node[] = tokens;
210 if (standalone) {
211 if (lineTags[0].kind === 'partial') lineTags[0].indentation = tokens[0] as string;
212 placed = lineTags;
213 } else placed.push('\n');
214 for (const token of placed) {
215 const target = targets[targets.length - 1];
216 if (typeof token === 'string') {
217 if (token !== '') target.push(token);
218 } else if (token.kind === 'section' || token.kind === 'inverted') {
219 target.push(token);
220 targets.push(token.children);
221 } else if (token.kind === 'close') targets.pop();
222 else if (token.kind !== 'comment' && token.kind !== 'delimiter') target.push(token);
223 }
224 }
225 if (sections.length > 0) {
226 const tag = sections[sections.length - 1];
227 throw new TemplateError(tag.number, `unclosed section '${tag.expression!.name}'`);
228 }
229 return nodes;
230}

232function toText(value: Value): string {
233 if (value === null) return '';
234 if (value instanceof Map) throw new Error('cannot render map');
235 if (Array.isArray(value)) return value.map(toText).join(', ');
236 return String(value);
237}

239function truthy(value: Value): boolean {
240 if (value instanceof Map) return value.size > 0;
241 if (Array.isArray(value)) return value.length > 0;
242 return value !== null && value !== false && value !== 0 && value !== '';
243}

245function toNumber(value: Value, filter: string): number {
246 if (value === null) return 0;
247 if (typeof value === 'number') return value;
248 if (typeof value === 'string' && !value.includes('\n') && numericPattern.test(value)) return Number(value);
249 throw new Error(`not a number for '${filter}'`);
250}

252function field(value: Value, key: string): Value {
253 return value instanceof Map ? value.get(key) ?? null : null;
254}

256function sortRank(value: Value): number {
257 if (value === null) return 0;
258 if (typeof value === 'number') return 1;
259 if (typeof value === 'string') return 2;
260 if (typeof value === 'boolean') return 3;
261 return 4;
262}
263
264function compareValues(left: Value, right: Value): number {
265 const rank = sortRank(left) - sortRank(right);
266 if (rank !== 0) return rank;
267 if (sortRank(left) === 0 || sortRank(left) === 4) return 0;
268 if (left === right) return 0;
269 if (typeof left === 'number' && typeof right === 'number') return left - right;
270 if (typeof left === 'string' && typeof right === 'string') return left < right ? -1 : 1;
271 return Number(left) - Number(right);
272}

274function applyFilter(filter: Filter, value: Value): Value {
275 const { name, args } = filter;
276 const n = args[0] as number;
277 const s = args[0] as string;
278 switch (name) {
279 case 'upper': return toText(value).toUpperCase();
280 case 'lower': return toText(value).toLowerCase();
281 case 'title': return [...toText(value)].map((char, index, chars) => index === 0 || chars[index - 1] === ' ' ? char.toUpperCase() : char.toLowerCase()).join('');
282 case 'trim': return strip(toText(value));
283 case 'length': return value instanceof Map ? value.size : Array.isArray(value) ? value.length : toText(value).length;
284 case 'first': return Array.isArray(value) ? value[0] ?? null : value;
285 case 'last': return Array.isArray(value) ? value[value.length - 1] ?? null : value;
286 case 'reverse': return Array.isArray(value) ? [...value].reverse() : typeof value === 'string' ? [...value].reverse().join('') : value;
287 case 'items': return value instanceof Map ? [...value.keys()].sort().map(key => new Map<string, Value>([['key', key], ['value', value.get(key)!]])) : value;
288 case 'comma': {
289 const number = toNumber(value, name);
290 return (number < 0 ? '-' : '') + String(Math.abs(number)).replace(/\B(?=(\d{3})+(?!\d))/g, ',');
291 }
292 case 'sum': case 'min': case 'max': {
293 if (!Array.isArray(value)) return toNumber(value, name);
294 const numbers = value.map(element => toNumber(element, name));
295 if (name === 'sum') return numbers.reduce((total, number) => total + number, 0);
296 if (numbers.length === 0) return null;
297 return numbers.reduce((best, number) => name === 'min' ? Math.min(best, number) : Math.max(best, number));
298 }
299 case 'default': return truthy(value) ? value : args[0];
300 case 'eq': return toText(value) === toText(args[0]);
301 case 'pad': return toText(value).padEnd(n, args.length === 2 ? args[1] as string : ' ');
302 case 'lpad': return toText(value).padStart(n, args.length === 2 ? args[1] as string : ' ');
303 case 'truncate': {
304 const text = toText(value);
305 return text.length > n ? text.slice(0, n) + '...' : text;
306 }
307 case 'replace': return toText(value).split(s).join(args[1] as string);
308 case 'join': return Array.isArray(value) ? value.map(toText).join(s) : toText(value);
309 case 'pluck': return Array.isArray(value) ? value.map(element => field(element, s)) : value;
310 case 'sort': return Array.isArray(value) ? value.map((element, index) => ({ element, index, key: args.length === 0 ? element : field(element, s) })).sort((a, b) => compareValues(a.key, b.key) || a.index - b.index).map(item => item.element) : value;
311 default: {
312 const number = toNumber(value, name);
313 switch (name) {
314 case 'add': return number + n;
315 case 'sub': return number - n;
316 case 'mul': return number * n;
317 case 'div': return Math.floor(number / n);
318 case 'mod': return number - n * Math.floor(number / n);
319 case 'gt': return number > n;
320 default: return number < n;
321 }
322 }
323 }

326class Renderer {
327 contexts: Value[];
328 loops: { index: number; count: number }[] = [];
329 output: string[] = [];
330
331 constructor(root: Map<string, Value>, private partials: Map<string, SourceLine[]>, private strict: boolean) {
332 this.contexts = [root];
333 }
334
335 resolve(name: string): [boolean, Value] {
336 if (name === '.') return [true, this.contexts[this.contexts.length - 1]];
337 if (name.startsWith('@')) {
338 if (this.loops.length === 0) return [false, null];
339 const { index, count } = this.loops[this.loops.length - 1];
340 if (name === '@index') return [true, index];
341 if (name === '@number') return [true, index + 1];
342 return [true, name === '@first' ? index === 0 : index === count - 1];
343 }
344 const parts: Part[] = name.split('.').map(part => /^[0-9]+$/.test(part) ? Number(part) : part);
345 for (let index = this.contexts.length - 1; index >= 0; index--) {
346 let value = this.contexts[index];
347 if (!hasPart(value, parts[0])) continue;
348 for (const part of parts) {
349 if (!hasPart(value, part)) return [false, null];
350 value = getPart(value, part);
351 }
352 return [true, value];
353 }
354 return [false, null];
355 }
356
357 evaluate(expression: Expression, strict: boolean): Value {
358 let [found, value] = this.resolve(expression.name);
359 if (!found && strict) throw new Error(`missing '${expression.name}'`);
360 for (const filter of expression.filters) value = applyFilter(filter, value);
361 return value;
362 }
363
364 render(nodes: Node[], depth: number): void {
365 for (const node of nodes) {
366 if (typeof node === 'string') {
367 this.output.push(node);
368 continue;
369 }
370 try {
371 if (node.kind === 'variable' || node.kind === 'raw') {
372 const text = toText(this.evaluate(node.expression!, this.strict));
373 const escaped: Record<string, string> = { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' };
374 this.output.push(node.kind === 'raw' ? text : text.replace(/[&<>"']/g, char => escaped[char]));
375 } else if (node.kind === 'section') {
376 const value = this.evaluate(node.expression!, false);
377 if (Array.isArray(value)) {
378 for (let index = 0; index < value.length; index++) {
379 this.contexts.push(value[index]);
380 this.loops.push({ index, count: value.length });
381 this.render(node.children, depth);
382 this.loops.pop();
383 this.contexts.pop();
384 }
385 } else if (truthy(value)) {
386 this.contexts.push(value);
387 this.render(node.children, depth);
388 this.contexts.pop();
389 }
390 } else if (node.kind === 'inverted') {
391 if (!truthy(this.evaluate(node.expression!, false))) this.render(node.children, depth);
392 } else if (node.kind === 'partial') {
393 const source = this.partials.get(node.argument);
394 if (source === undefined) throw new Error(`unknown partial '${node.argument}'`);
395 if (depth === 8) throw new Error(`partial depth exceeded at '${node.argument}'`);
396 this.render(parseSource(source.map(line => ({ number: line.number, text: line.text === '' ? '' : node.indentation + line.text }))), depth + 1);
397 }
398 } catch (error) {
399 if (error instanceof TemplateError) throw error;
400 throw new TemplateError(node.number, (error as Error).message);
401 }
402 }
403 }
404}

406class TemplateEngine {
407 root = new Map<string, Value>();
408 partials = new Map<string, SourceLine[]>();
409 templates = new Map<string, SourceLine[]>();
410 strict = false;
411 output: string[] = [];
412
413 assign(line: string): void {
414 const separator = line.indexOf(' = ');
415 if (separator < 0) throw new Error('bad line');
416 const parts = parseKey(line.slice(0, separator));
417 const value = parseValue(line.slice(separator + 3));
418 let container: Value = this.root;
419 let detached: { container: Value; part: Part; child: Value } | null = null;
420 for (let index = 0; index < parts.length; index++) {
421 const { part, prefix } = parts[index];
422 if (typeof part === 'number' && part > (container as Value[]).length) throw new Error(`index gap at '${prefix}'`);
423 if (index === parts.length - 1) {
424 putPart(container, part, value);
425 break;
426 }
427 const mapRequired = typeof parts[index + 1].part === 'string';
428 let child: Value;
429 if (hasPart(container, part)) {
430 child = getPart(container, part);
431 if (!(mapRequired ? child instanceof Map : Array.isArray(child))) throw new Error(`type conflict at '${prefix}'`);
432 } else {
433 child = mapRequired ? new Map<string, Value>() : [];
434 if (detached === null) detached = { container, part, child };
435 else putPart(container, part, child);
436 }
437 container = child;
438 }
439 if (detached !== null) putPart(detached.container, detached.part, detached.child);
440 }
441
442 unset(key: string): void {
443 const parts = parseKey(key);
444 let container: Value = this.root;
445 for (let index = 0; index < parts.length; index++) {
446 const part = parts[index].part;
447 if (!hasPart(container, part)) throw new Error(`no value at '${key}'`);
448 if (index === parts.length - 1) {
449 if (typeof part === 'string') (container as Map<string, Value>).delete(part);
450 else (container as Value[]).splice(part, 1);
451 } else container = getPart(container, part);
452 }
453 }
454
455 loadData(body: SourceLine[]): void {
456 let assigned = 0;
457 let unset = 0;
458 const errors: string[] = [];
459 for (const { number, text } of body) {
460 if (text === '' || text.startsWith('#')) continue;
461 try {
462 if (text.startsWith('UNSET ')) {
463 this.unset(text.slice(6));
464 unset++;
465 } else {
466 this.assign(text);
467 assigned++;
468 }
469 } catch (error) {
470 errors.push(`ERROR line ${number}: ${(error as Error).message}`);
471 }
472 }
473 this.output.push(`DATA set=${assigned} unset=${unset} errors=${errors.length}`, ...errors);
474 }
475
476 render(name: string): void {
477 try {
478 const renderer = new Renderer(this.root, this.partials, this.strict);
479 renderer.render(parseSource(this.templates.get(name)!), 0);
480 const result = renderer.output.join('');
481 const lines = result === '' ? [] : result.split('\n');
482 if (result.endsWith('\n')) lines.pop();
483 this.output.push(`TEMPLATE ${name} OK lines=${lines.length} chars=${result.length}`, ...lines);
484 } catch (error) {
485 const failure = error as TemplateError;
486 this.output.push(`TEMPLATE ${name} ERROR line ${failure.number}: ${failure.message}`);
487 }
488 }
489
490 dump(): void {
491 const leaves: string[] = [];
492 const visit = (value: Value, path: string): void => {
493 if (value instanceof Map && value.size > 0) {
494 for (const key of [...value.keys()].sort()) visit(value.get(key)!, `${path}.${key}`);
495 } else if (Array.isArray(value) && value.length > 0) {
496 value.forEach((element, index) => visit(element, `${path}[${index}]`));
497 } else {
498 const notation = typeof value === 'string' ? '"' + value.replace(/\\/g, '\\\\').replace(/"/g, '\\"').replace(/\n/g, '\\n') + '"' : value instanceof Map ? '{}' : Array.isArray(value) ? '[]' : String(value);
499 leaves.push(`${path} = ${notation}`);
500 }
501 };
502 for (const key of [...this.root.keys()].sort()) visit(this.root.get(key)!, key);
503 this.output.push(`DUMP leaves=${leaves.length}`, ...leaves);
504 }
505
506 run(lines: string[]): void {
507 let position = 0;
508 while (lines[position] !== 'END') {
509 const line = lines[position++];
510 const number = position;
511 const [command, name] = line.split(' ');
512 if (command === 'DATA' || command === 'PARTIAL' || command === 'TEMPLATE') {
513 const body: SourceLine[] = [];
514 while (lines[position] !== `END${command}`) body.push({ number: position + 1, text: lines[position++] });
515 position++;
516 if (command === 'DATA') this.loadData(body);
517 else if (command === 'TEMPLATE') {
518 this.templates.set(name, body);
519 this.render(name);
520 } else {
521 try {
522 parseSource(body);
523 const status = this.partials.has(name) ? 'REPLACED' : 'DEFINED';
524 this.partials.set(name, body);
525 this.output.push(`PARTIAL ${name} ${status} lines=${body.length}`);
526 } catch (error) {
527 const failure = error as TemplateError;
528 this.output.push(`PARTIAL ${name} ERROR line ${failure.number}: ${failure.message}`);
529 }
530 }
531 } else if (command === 'RENDER') {
532 if (this.templates.has(name)) this.render(name);
533 else this.output.push(`TEMPLATE ${name} ERROR line ${number}: unknown template '${name}'`);
534 } else if (command === 'DUMP') this.dump();
535 else if (command === 'STRICT') {
536 this.strict = name === 'ON';
537 this.output.push(line);
538 } else if (command === 'RESET') {
539 this.root.clear();
540 this.partials.clear();
541 this.templates.clear();
542 this.output.push('RESET');
543 }
544 }
545 }
546}

549engine.run(fs.readFileSync(0, 'utf8').split('\n'));
550process.stdout.write(engine.output.length === 0 ? '' : engine.output.join('\n') + '\n');
