10class TemplateEngine
11 IDENTIFIER = /\A[A-Za-z_][A-Za-z0-9_]*\z/
12 NAME = /\A(?:\.|@(?:index|number|first|last)|(?:[A-Za-z_][A-Za-z0-9_]*|0|[1-9][0-9]*)(?:\.(?:[A-Za-z_][A-Za-z0-9_]*|0|[1-9][0-9]*))*)\z/
13 NO_ARGUMENTS = %w[upper lower title trim length first last reverse items comma sum min max].freeze
14 FILTERS = (NO_ARGUMENTS + %w[default eq pad lpad truncate replace join sort pluck add sub mul div mod gt lt]).freeze

18 @partials = {}
19 @templates = {}

24 text.gsub(/\A +| +\z/, '')

31 def parse_key(key)
32 return nil unless key.match?(/\A[A-Za-z_][A-Za-z0-9_]*(?:(?:\.[A-Za-z_][A-Za-z0-9_]*)|(?:\[(?:0|[1-9][0-9]*)\]))*\z/)
33
34 parts = []
35 key.scan(/[A-Za-z_][A-Za-z0-9_]*|\[(0|[1-9][0-9]*)\]/) do
36 match = Regexp.last_match
37 parts << [match[1] ? match[1].to_i : match[0], key[0...match.end(0)]]
38 end
39 parts
40 end

42 def scalar(text, offset)
43 if text[offset] == '"'
44 result = +''
45 pos = offset + 1
46 while pos < text.length
47 char = text[pos]
48 return [result, pos + 1] if char == '"'
49 if char == '\\'
50 pos += 1
51 char = { '"' => '"', '\\' => '\\', 'n' => "\n" }[text[pos]]
52 raise ArgumentError if char.nil?
53 end
54 result << char
55 pos += 1
56 end
57 raise ArgumentError
58 end
59 match = /\A(?:0|-?[1-9][0-9]*|true|false)/.match(text[offset..])
60 raise ArgumentError unless match
61
62 word = match[0]
63 value = case word
64 when 'true' then true
65 when 'false' then false
66 else word.to_i
67 end
68 [value, offset + word.length]
69 end

71 def parse_value(text)
72 return [] if text == '[]'
73 return {} if text == '{}'
74 if text.start_with?('[')
75 result = []
76 pos = 1
77 loop do
78 value, pos = scalar(text, pos)
79 result << value
80 return result if text[pos] == ']' && pos == text.length - 1
81 raise ArgumentError unless text[pos, 2] == ', '
82
83 pos += 2
84 end
85 else
86 value, pos = scalar(text, 0)
87 raise ArgumentError unless pos == text.length
88
89 value
90 end
91 end

93 def has_part?(value, part)
94 (part.is_a?(String) && value.is_a?(Hash) && value.key?(part)) ||
95 (part.is_a?(Integer) && value.is_a?(Array) && part < value.length)
96 end
97
98 def set_value(container, parts, value, index = 0)
99 part, prefix = parts[index]
100 fail_with("index gap at '#{prefix}'") if part.is_a?(Integer) && part > container.length
101 if index == parts.length - 1
102 container[part] = value
103 return
104 end
105 required = parts[index + 1][0].is_a?(Integer) ? Array : Hash
106 if has_part?(container, part)
107 fail_with("type conflict at '#{prefix}'") unless container[part].is_a?(required)
108 else
109 container[part] = required.new
110 end
111 set_value(container[part], parts, value, index + 1)
112 end
113
114 def unset_value(container, parts, key, index = 0)
115 part = parts[index][0]
116 fail_with("no value at '#{key}'") unless has_part?(container, part)
117 if index == parts.length - 1
118 container.is_a?(Array) ? container.delete_at(part) : container.delete(part)
119 else
120 unset_value(container[part], parts, key, index + 1)
121 end
122 end

124 def data(body)
125 counts = { set: 0, unset: 0 }
126 errors = []
127 body.each do |text, line|
128 next if text.empty? || text.start_with?('#')
129 begin
130 deleting = text.start_with?('UNSET ')
131 if deleting
132 key = text[6..]
133 else
134 fail_with('bad line') unless text.include?(' = ')
135 key, notation = text.split(' = ', 2)
136 end
137 parts = parse_key(key)
138 fail_with("bad key '#{key}'") unless parts
139 unless deleting
140 begin
141 value = parse_value(notation)
142 rescue ArgumentError
143 fail_with("bad value '#{notation}'")
144 end
145 end
146 changed = Marshal.load(Marshal.dump(@root))
147 if deleting
148 unset_value(changed, parts, key)
149 else
150 set_value(changed, parts, value)
151 end
152 @root = changed
153 counts[deleting ? :unset : :set] += 1
154 rescue TemplateError => error
155 errors << "ERROR line #{line}: #{error.message}"
156 end
157 end
158 puts "DATA set=#{counts[:set]} unset=#{counts[:unset]} errors=#{errors.length}"
159 puts errors
160 end

162 def split_quoted(text, separator)
163 result = []
164 start = 0
165 quoted = false
166 text.each_char.with_index do |char, index|
167 quoted = !quoted if char == "'"
168 if char == separator && !quoted
169 result << text[start...index]
170 start = index + 1
171 end
172 end
173 result << text[start..]
174 result
175 end

177 def expression(text, line)
178 pieces = split_quoted(text, '|').map { |part| strip_spaces(part) }
179 name = pieces.shift
180 fail_with("invalid name '#{name}'", line) unless NAME.match?(name)
181 filters = pieces.map do |piece|
182 filter, *arguments = split_quoted(piece, ':')
183 fail_with("unknown filter '#{filter}'", line) unless FILTERS.include?(filter)
184 valid = true
185 args = arguments.map do |argument|
186 if argument.match?(/\A'[^']*'\z/)
187 argument[1...-1]
188 elsif argument.match?(/\A-?[0-9]{1,9}\z/)
189 argument.to_i
190 else
191 valid = false
192 nil
193 end
194 end
195 integers = args.all? { |arg| arg.is_a?(Integer) }
196 strings = args.all? { |arg| arg.is_a?(String) }
197 valid &&= if NO_ARGUMENTS.include?(filter)
198 args.empty?
199 elsif %w[default eq].include?(filter)
200 args.length == 1
201 elsif %w[pad lpad].include?(filter)
202 [1, 2].include?(args.length) && args[0].is_a?(Integer) && args[0] >= 0 &&
203 (args.length == 1 || (args[1].is_a?(String) && args[1].length == 1))
204 elsif filter == 'truncate'
205 args.length == 1 && integers && args[0] >= 0
206 elsif filter == 'replace'
207 args.length == 2 && strings && !args[0].empty?
208 elsif %w[join pluck].include?(filter)
209 args.length == 1 && strings
210 elsif filter == 'sort'
211 args.length <= 1 && strings
212 else
213 args.length == 1 && integers && (!%w[div mod].include?(filter) || args[0] != 0)
214 end
215 fail_with("bad arguments for '#{filter}'", line) unless valid
216 [filter, args]
217 end
218 [name, filters]
219 end

221 def analyze(body)
222 opening = '{{'
223 closing = '}}'
224 tokens = []
225 sections = []
226 body.each do |text, line|
227 local = []
228 pos = 0
229 loop do
230 start = text.index(opening, pos)
231 unless start
232 local << { kind: :text, text: text[pos..], line: line }
233 break
234 end
235 local << { kind: :text, text: text[pos...start], line: line }
236 content_start = start + opening.length
237 triple = text[content_start] == '{'
238 content_start += 1 if triple
239 finish_delimiter = triple ? '}' + closing : closing
240 finish = text.index(finish_delimiter, content_start)
241 fail_with('unclosed tag', line) unless finish
242 content = strip_spaces(text[content_start...finish])
243 pos = finish + finish_delimiter.length
244 marker = triple ? '&' : content[0]
245 kinds = { '#' => :section, '^' => :inverted, '/' => :close, '!' => :comment,
246 '>' => :partial, '&' => :raw, '=' => :delimiter }
247 kind = triple ? :raw : kinds.fetch(marker, :variable)
248 argument = triple || kind == :variable ? content : strip_spaces(content[1..])
249 token = { kind: kind, line: line, text: argument, indent: '' }
250 case kind
251 when :variable, :raw, :section, :inverted
252 token[:expression] = expression(argument, line)
253 if [:section, :inverted].include?(kind)
254 token[:name] = token[:expression][0]
255 sections << token
256 end
257 when :close
258 fail_with("invalid name '#{argument}'", line) unless NAME.match?(argument)
259 fail_with("unexpected closing '#{argument}'", line) if sections.empty?
260 expected = sections.last[:name]
261 fail_with("mismatched closing '#{argument}', expected '#{expected}'", line) unless expected == argument
262 sections.pop
263 when :partial
264 fail_with("invalid name '#{argument}'", line) unless IDENTIFIER.match?(argument)
265 when :delimiter
266 delimiters = content.length >= 2 && content.end_with?('=') ? content[1...-1].split(' ') : []
267 fail_with('invalid delimiters', line) unless delimiters.length == 2 && delimiters.none? { |item| item.include?('=') }
268 opening, closing = delimiters
269 end
270 local << token
271 end
272 tags = local.reject { |token| token[:kind] == :text }
273 standalone = tags.length == 1 && [:section, :inverted, :close, :comment, :partial, :delimiter].include?(tags[0][:kind]) &&
274 local.select { |token| token[:kind] == :text }.all? { |token| token[:text].match?(/\A *\z/) }
275 if standalone
276 tags[0][:indent] = local[0][:text] if tags[0][:kind] == :partial
277 tokens.concat(tags)
278 else
279 tokens.concat(local)
280 tokens << { kind: :text, text: "\n", line: line }
281 end
282 end
283 unless sections.empty?
284 token = sections.last
285 fail_with("unclosed section '#{token[:name]}'", token[:line])
286 end
287 root = []
288 stack = [root]
289 tokens.each do |token|
290 if token[:kind] == :close
291 stack.pop
292 else
293 stack.last << token
294 if [:section, :inverted].include?(token[:kind])
295 token[:children] = []
296 stack << token[:children]
297 end
298 end
299 end
300 root
301 end

303 def stringify(value)
304 case value
305 when nil then ''
306 when String then value
307 when Integer then value.to_s
308 when true then 'true'
309 when false then 'false'
310 when Array then value.map { |item| stringify(item) }.join(', ')
311 else fail_with('cannot render map')
312 end
313 end

315 def truthy?(value)
316 !(value.nil? || value == false || value == 0 || (value.respond_to?(:empty?) && value.empty?))
317 end

319 def number(value, filter)
320 return 0 if value.nil?
321 return value if value.is_a?(Integer)
322 return value.to_i if value.is_a?(String) && value.match?(/\A-?[0-9]{1,9}\z/)
323
324 fail_with("not a number for '#{filter}'")
325 end

327 def sort_key(value)
328 case value
329 when nil then [0, 0]
330 when Integer then [1, value]
331 when String then [2, value]
332 when true, false then [3, value ? 1 : 0]
333 else [4, 0]
334 end
335 end

337 def filter_value(value, filter, args)
338 case filter
339 when 'upper' then stringify(value).upcase
340 when 'lower' then stringify(value).downcase
341 when 'title'
342 stringify(value).downcase.gsub(/\A.|(?<= )./) { |char| char.upcase }
343 when 'trim' then strip_spaces(stringify(value))
344 when 'length' then value.is_a?(Array) || value.is_a?(Hash) ? value.length : stringify(value).length
345 when 'first' then value.is_a?(Array) ? value.first : value
346 when 'last' then value.is_a?(Array) ? value.last : value
347 when 'reverse' then value.is_a?(Array) || value.is_a?(String) ? value.reverse : value
348 when 'items' then value.is_a?(Hash) ? value.keys.sort.map { |key| { 'key' => key, 'value' => value[key] } } : value
349 when 'comma'
350 n = number(value, filter)
351 (n < 0 ? '-' : '') + n.abs.to_s.reverse.scan(/.{1,3}/).join(',').reverse
352 when 'sum', 'min', 'max'
353 if value.is_a?(Array)
354 values = value.map { |item| number(item, filter) }
355 filter == 'sum' ? values.sum : values.public_send(filter)
356 else
357 number(value, filter)
358 end
359 when 'default' then truthy?(value) ? value : args[0]
360 when 'eq' then stringify(value) == stringify(args[0])
361 when 'pad' then stringify(value).ljust(args[0], args.fetch(1, ' '))
362 when 'lpad' then stringify(value).rjust(args[0], args.fetch(1, ' '))
363 when 'truncate'
364 text = stringify(value)
365 text.length > args[0] ? text[0, args[0]] + '...' : text
366 when 'replace' then stringify(value).gsub(args[0]) { args[1] }
367 when 'join' then value.is_a?(Array) ? value.map { |item| stringify(item) }.join(args[0]) : stringify(value)
368 when 'sort'
369 return value unless value.is_a?(Array)
370 value.each_with_index.sort_by do |item, index|
371 key = args.empty? ? item : (item.is_a?(Hash) ? item[args[0]] : nil)
372 sort_key(key) + [index]
373 end.map(&:first)
374 when 'pluck' then value.is_a?(Array) ? value.map { |item| item.is_a?(Hash) ? item[args[0]] : nil } : value
375 when 'add' then number(value, filter) + args[0]
376 when 'sub' then number(value, filter) - args[0]
377 when 'mul' then number(value, filter) * args[0]
378 when 'div' then number(value, filter).div(args[0])
379 when 'mod' then number(value, filter) % args[0]
380 when 'gt' then number(value, filter) > args[0]
381 when 'lt' then number(value, filter) < args[0]
382 end
383 end

385 def resolve(name, contexts, loops)
386 return [true, contexts.last] if name == '.'
387 if name.start_with?('@')
388 return [false, nil] if loops.empty?
389 index, count = loops.last
390 return [true, { '@index' => index, '@number' => index + 1, '@first' => index == 0, '@last' => index == count - 1 }[name]]
391 end
392 parts = name.split('.').map { |part| part.match?(/\A[0-9]+\z/) ? part.to_i : part }
393 context = contexts.reverse.find { |item| has_part?(item, parts[0]) }
394 return [false, nil] if context.nil?
395 parts.each do |part|
396 return [false, nil] unless has_part?(context, part)
397 context = context[part]
398 end
399 [true, context]
400 end

402 def evaluate(token, contexts, loops)
403 name, filters = token[:expression]
404 found, value = resolve(name, contexts, loops)
405 if @strict && [:variable, :raw].include?(token[:kind]) && !found
406 fail_with("missing '#{name}'")
407 end
408 filters.each { |filter, args| value = filter_value(value, filter, args) }
409 value
410 end

412 def html(text)
413 text.gsub(/[&<>"']/) { |char| { '&' => '&amp;', '<' => '&lt;', '>' => '&gt;', '"' => '&quot;', "'" => '&#39;' }[char] }
414 end

416 def render(nodes, contexts, loops, depth)
417 output = +''
418 nodes.each do |token|
419 begin
420 case token[:kind]
421 when :text then output << token[:text]
422 when :variable, :raw
423 text = stringify(evaluate(token, contexts, loops))
424 output << (token[:kind] == :variable ? html(text) : text)
425 when :section
426 value = evaluate(token, contexts, loops)
427 if value.is_a?(Array)
428 value.each_with_index do |item, index|
429 output << render(token[:children], contexts + [item], loops + [[index, value.length]], depth)
430 end
431 elsif truthy?(value)
432 output << render(token[:children], contexts + [value], loops, depth)
433 end
434 when :inverted
435 output << render(token[:children], contexts, loops, depth) unless truthy?(evaluate(token, contexts, loops))
436 when :partial
437 name = token[:text]
438 fail_with("unknown partial '#{name}'") unless @partials.key?(name)
439 fail_with("partial depth exceeded at '#{name}'") if depth == 8
440 body = @partials[name].map { |text, line| [text.empty? ? text : token[:indent] + text, line] }
441 output << render(analyze(body), contexts, loops, depth + 1)
442 end
443 rescue TemplateError => error
444 error.line ||= token[:line]
445 raise
446 end
447 end
448 output
449 end

451 def render_template(name, line)
452 unless @templates.key?(name)
453 puts "TEMPLATE #{name} ERROR line #{line}: unknown template '#{name}'"
454 return
455 end
456 begin
457 result = render(analyze(@templates[name]), [@root], [], 0)
458 count = result.empty? ? 0 : result.count("\n") + (result.end_with?("\n") ? 0 : 1)
459 puts "TEMPLATE #{name} OK lines=#{count} chars=#{result.length}"
460 print result
461 print "\n" unless result.empty? || result.end_with?("\n")
462 rescue TemplateError => error
463 puts "TEMPLATE #{name} ERROR line #{error.line}: #{error.message}"
464 end
465 end

467 def dump_leaves(value, path, output)
468 if value.is_a?(Hash) && !value.empty?
469 value.keys.sort.each { |key| dump_leaves(value[key], path.empty? ? key : "#{path}.#{key}", output) }
470 elsif value.is_a?(Array) && !value.empty?
471 value.each_with_index { |item, index| dump_leaves(item, "#{path}[#{index}]", output) }
472 elsif !path.empty?
473 notation = case value
474 when String then '"' + value.gsub(/[\\"\n]/) { |char| { '\\' => '\\\\', '"' => '\\"', "\n" => '\\n' }[char] } + '"'
475 when Hash then '{}'
476 when Array then '[]'
477 else stringify(value)
478 end
479 output << "#{path} = #{notation}"
480 end
481 end

483 def run
484 input = STDIN.each_line.map { |line| line.chomp("\n").delete_suffix("\r") }
485 index = 0
486 while index < input.length
487 command = input[index]
488 line = index + 1
489 index += 1
490 break if command == 'END'
491 kind, name = command.split(' ', 2)
492 if %w[DATA PARTIAL TEMPLATE].include?(kind)
493 body = []
494 ending = { 'DATA' => 'ENDDATA', 'PARTIAL' => 'ENDPARTIAL', 'TEMPLATE' => 'ENDTEMPLATE' }[kind]
495 while input[index] != ending
496 body << [input[index], index + 1]
497 index += 1
498 end
499 index += 1
500 if kind == 'DATA'
501 data(body)
502 elsif kind == 'PARTIAL'
503 begin
504 analyze(body)
505 action = @partials.key?(name) ? 'REPLACED' : 'DEFINED'
506 @partials[name] = body
507 puts "PARTIAL #{name} #{action} lines=#{body.length}"
508 rescue TemplateError => error
509 puts "PARTIAL #{name} ERROR line #{error.line}: #{error.message}"
510 end
511 else
512 @templates[name] = body
513 render_template(name, line)
514 end
515 elsif kind == 'RENDER'
516 render_template(name, line)
517 elsif kind == 'DUMP'
518 leaves = []
519 dump_leaves(@root, '', leaves)
520 puts "DUMP leaves=#{leaves.length}"
521 puts leaves
522 elsif kind == 'STRICT'
523 @strict = name == 'ON'
524 puts command
525 elsif kind == 'RESET'
526 @root = {}
527 @partials = {}
528 @templates = {}
529 puts 'RESET'
530 end
531 end
532 end
