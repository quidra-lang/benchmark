48 private function scalar(string $text, int &$position): mixed
49 {
50 if (($text[$position] ?? '') === '"') {
51 $position++;
52 $result = '';
53 while ($position < strlen($text)) {
54 $char = $text[$position++];
55 if ($char === '"') {
56 return $result;
57 }
58 if ($char === '\\') {
59 $escaped = $text[$position++] ?? '';
60 $char = match ($escaped) {
61 '"' => '"', '\\' => '\\', 'n' => "\n", default => null
62 };
63 if ($char === null) {
64 throw new InvalidArgumentException();
65 }
66 }
67 $result .= $char;
68 }
69 throw new InvalidArgumentException();
70 }
71 if (!preg_match('/\A(?:0|-?[1-9][0-9]*|true|false)/', substr($text, $position), $match)) {
72 throw new InvalidArgumentException();
73 }
74 $position += strlen($match[0]);
75 return match ($match[0]) {
76 'true' => true, 'false' => false, default => (int)$match[0]
77 };

80 private function value(string $text): mixed
81 {
82 if ($text === '[]') {
83 return [];
84 }
85 if ($text === '{}') {
86 return new stdClass();
87 }
88 if (str_starts_with($text, '[')) {
89 $result = [];
90 $position = 1;
91 while (true) {
92 $result[] = $this->scalar($text, $position);
93 if (($text[$position] ?? '') === ']' && $position === strlen($text) - 1) {
94 return $result;
95 }
96 if (substr($text, $position, 2) !== ', ') {
97 throw new InvalidArgumentException();
98 }
99 $position += 2;
100 }
101 }
102 $position = 0;
103 $result = $this->scalar($text, $position);
104 if ($position !== strlen($text)) {
105 throw new InvalidArgumentException();
106 }
107 return $result;

121 private function assign(mixed &$container, array $parts, mixed $value, int $index = 0): void
122 {
123 [$part, $prefix] = $parts[$index];
124 if (is_int($part) && $part > count($container)) {
125 $this->fail("index gap at '$prefix'");
126 }
127 if ($index === count($parts) - 1) {
128 if (is_object($container)) {
129 $container->{$part} = $value;
130 } else {
131 $container[$part] = $value;
132 }
133 return;
134 }
135 $list = is_int($parts[$index + 1][0]);
136 if ($this->has($container, $part)) {
137 $child = $this->get($container, $part);
138 if (($list && !is_array($child)) || (!$list && !is_object($child))) {
139 $this->fail("type conflict at '$prefix'");
140 }
141 } else {
142 $child = $list ? [] : new stdClass();
143 }
144 $this->assign($child, $parts, $value, $index + 1);
145 if (is_object($container)) {
146 $container->{$part} = $child;
147 } else {
148 $container[$part] = $child;
149 }
150 }

152 private function remove(mixed &$container, array $parts, string $key, int $index = 0): void
153 {
154 $part = $parts[$index][0];
155 if (!$this->has($container, $part)) {
156 $this->fail("no value at '$key'");
157 }
158 if ($index === count($parts) - 1) {
159 if (is_object($container)) {
160 unset($container->{$part});
161 } else {
162 array_splice($container, $part, 1);
163 }
164 } else {
165 $child = $this->get($container, $part);
166 $this->remove($child, $parts, $key, $index + 1);
167 if (is_object($container)) {
168 $container->{$part} = $child;
169 } else {
170 $container[$part] = $child;
171 }

175 private function data(array $body): void
176 {
177 $set = 0;
178 $unset = 0;
179 $errors = [];
180 foreach ($body as [$text, $line]) {
181 if ($text === '' || str_starts_with($text, '#')) {
182 continue;
183 }
184 try {
185 $deleting = str_starts_with($text, 'UNSET ');
186 if ($deleting) {
187 $key = substr($text, 6);
188 } else {
189 if (!str_contains($text, ' = ')) {
190 $this->fail('bad line');
191 }
192 [$key, $notation] = explode(' = ', $text, 2);
193 }
194 $parts = $this->keyParts($key);
195 if ($parts === null) {
196 $this->fail("bad key '$key'");
197 }
198 if (!$deleting) {
199 try {
200 $value = $this->value($notation);
201 } catch (InvalidArgumentException $error) {
202 $this->fail("bad value '$notation'");
203 }
204 }
205 $changed = unserialize(serialize($this->root));
206 if ($deleting) {
207 $this->remove($changed, $parts, $key);
208 $unset++;
209 } else {
210 $this->assign($changed, $parts, $value);
211 $set++;
212 }
213 $this->root = $changed;
214 } catch (TemplateFault $error) {
215 $errors[] = "ERROR line $line: {$error->getMessage()}";
216 }
217 }
218 echo "DATA set=$set unset=$unset errors=" . count($errors) . "\n";
219 foreach ($errors as $error) {
220 echo "$error\n";
221 }
222 }

224 private function splitQuoted(string $text, string $separator): array
225 {
226 $parts = [];
227 $quoted = false;
228 $start = 0;
229 for ($index = 0; $index < strlen($text); $index++) {
230 if ($text[$index] === "'") {
231 $quoted = !$quoted;
232 }
233 if ($text[$index] === $separator && !$quoted) {
234 $parts[] = substr($text, $start, $index - $start);
235 $start = $index + 1;
236 }
237 }
238 $parts[] = substr($text, $start);
239 return $parts;

242 private function expression(string $text, int $line): array
243 {
244 $pieces = array_map(fn($part) => trim($part, ' '), $this->splitQuoted($text, '|'));
245 $name = array_shift($pieces);
246 if (!preg_match(self::NAME, $name)) {
247 $this->fail("invalid name '$name'", $line);
248 }
249 $filters = [];
250 foreach ($pieces as $piece) {
251 $arguments = $this->splitQuoted($piece, ':');
252 $filter = array_shift($arguments);
253 if (!in_array($filter, self::NO_ARGS, true) && !in_array($filter, self::WITH_ARGS, true)) {
254 $this->fail("unknown filter '$filter'", $line);
255 }
256 $args = [];
257 $valid = true;
258 foreach ($arguments as $argument) {
259 if (preg_match("/\A'[^']*'\z/", $argument)) {
260 $args[] = substr($argument, 1, -1);
261 } elseif (preg_match('/\A-?[0-9]{1,9}\z/', $argument)) {
262 $args[] = (int)$argument;
263 } else {
264 $valid = false;
265 }
266 }
267 $count = count($args);
268 $integers = count(array_filter($args, 'is_int')) === $count;
269 $strings = count(array_filter($args, 'is_string')) === $count;
270 if (in_array($filter, self::NO_ARGS, true)) {
271 $valid = $valid && $count === 0;
272 } elseif (in_array($filter, ['default', 'eq'], true)) {
273 $valid = $valid && $count === 1;
274 } elseif (in_array($filter, ['pad', 'lpad'], true)) {
275 $valid = $valid && ($count === 1 || $count === 2) && is_int($args[0]) && $args[0] >= 0 &&
276 ($count === 1 || (is_string($args[1]) && strlen($args[1]) === 1));
277 } elseif ($filter === 'truncate') {
278 $valid = $valid && $count === 1 && $integers && $args[0] >= 0;
279 } elseif ($filter === 'replace') {
280 $valid = $valid && $count === 2 && $strings && $args[0] !== '';
281 } elseif (in_array($filter, ['join', 'pluck'], true)) {
282 $valid = $valid && $count === 1 && $strings;
283 } elseif ($filter === 'sort') {
284 $valid = $valid && $count <= 1 && $strings;
285 } else {
286 $valid = $valid && $count === 1 && $integers && (!in_array($filter, ['div', 'mod'], true) || $args[0] !== 0);
287 }
288 if (!$valid) {
289 $this->fail("bad arguments for '$filter'", $line);
290 }
291 $filters[] = [$filter, $args];
292 }
293 return [$name, $filters];
294 }

312 private function analyze(array $body): array
313 {
314 $opening = '{{';
315 $closing = '}}';
316 $tokens = [];
317 $sections = [];
318 $kinds = ['#' => 'section', '^' => 'inverted', '/' => 'close', '!' => 'comment', '>' => 'partial', '&' => 'raw', '=' => 'delimiter'];
319 foreach ($body as [$text, $line]) {
320 $local = [];
321 $position = 0;
322 while (true) {
323 $start = strpos($text, $opening, $position);
324 if ($start === false) {
325 $local[] = ['kind' => 'text', 'text' => substr($text, $position), 'line' => $line];
326 break;
327 }
328 $local[] = ['kind' => 'text', 'text' => substr($text, $position, $start - $position), 'line' => $line];
329 $contentStart = $start + strlen($opening);
330 $triple = ($text[$contentStart] ?? '') === '{';
331 if ($triple) {
332 $contentStart++;
333 }
334 $endDelimiter = ($triple ? '}' : '') . $closing;
335 $finish = strpos($text, $endDelimiter, $contentStart);
336 if ($finish === false) {
337 $this->fail('unclosed tag', $line);
338 }
339 $content = trim(substr($text, $contentStart, $finish - $contentStart), ' ');
340 $position = $finish + strlen($endDelimiter);
341 $kind = $triple ? 'raw' : ($kinds[$content[0] ?? ''] ?? 'variable');
342 $argument = $triple || $kind === 'variable' ? $content : trim(substr($content, 1), ' ');
343 $token = ['kind' => $kind, 'text' => $argument, 'line' => $line, 'indent' => ''];
344 if (in_array($kind, ['variable', 'raw', 'section', 'inverted'], true)) {
345 $token['expression'] = $this->expression($argument, $line);
346 if (in_array($kind, ['section', 'inverted'], true)) {
347 $sections[] = [$token['expression'][0], $line];
348 }
349 } elseif ($kind === 'close') {
350 if (!preg_match(self::NAME, $argument)) {
351 $this->fail("invalid name '$argument'", $line);
352 }
353 if ($sections === []) {
354 $this->fail("unexpected closing '$argument'", $line);
355 }
356 $expected = $sections[array_key_last($sections)][0];
357 if ($expected !== $argument) {
358 $this->fail("mismatched closing '$argument', expected '$expected'", $line);
359 }
360 array_pop($sections);
361 } elseif ($kind === 'partial') {
362 if (!preg_match(self::IDENTIFIER, $argument)) {
363 $this->fail("invalid name '$argument'", $line);
364 }
365 } elseif ($kind === 'delimiter') {
366 $delimiters = strlen($content) >= 2 && str_ends_with($content, '=')
367 ? preg_split('/ +/', trim(substr($content, 1, -1), ' '), -1, PREG_SPLIT_NO_EMPTY) : [];
368 if (count($delimiters) !== 2 || str_contains(implode('', $delimiters), '=')) {
369 $this->fail('invalid delimiters', $line);
370 }
371 [$opening, $closing] = $delimiters;
372 }
373 $local[] = $token;
374 }
375 $tags = array_values(array_filter($local, fn($token) => $token['kind'] !== 'text'));
376 $onlySpaces = true;
377 foreach ($local as $token) {
378 if ($token['kind'] === 'text' && trim($token['text'], ' ') !== '') {
379 $onlySpaces = false;
380 }
381 }
382 if (count($tags) === 1 && in_array($tags[0]['kind'], ['section', 'inverted', 'close', 'comment', 'partial', 'delimiter'], true) && $onlySpaces) {
383 if ($tags[0]['kind'] === 'partial') {
384 $tags[0]['indent'] = $local[0]['text'];
385 }
386 $tokens[] = $tags[0];
387 } else {
388 array_push($tokens, ...$local);
389 $tokens[] = ['kind' => 'text', 'text' => "\n", 'line' => $line];
390 }
391 }
392 if ($sections !== []) {
393 [$name, $line] = $sections[array_key_last($sections)];
394 $this->fail("unclosed section '$name'", $line);
395 }
396 $index = 0;
397 return $this->tree($tokens, $index);
398 }

441 private function filter(mixed $value, string $filter, array $args): mixed
442 {
443 switch ($filter) {
444 case 'upper': return strtoupper($this->stringify($value));
445 case 'lower': return strtolower($this->stringify($value));
446 case 'title': return preg_replace_callback('/\A.|(?<= )./', fn($match) => strtoupper($match[0]), strtolower($this->stringify($value)));
447 case 'trim': return trim($this->stringify($value), ' ');
448 case 'length': return is_array($value) || is_object($value) ? count((array)$value) : strlen($this->stringify($value));
449 case 'first': return is_array($value) ? ($value[0] ?? null) : $value;
450 case 'last': return is_array($value) ? ($value === [] ? null : $value[array_key_last($value)]) : $value;
451 case 'reverse': return is_array($value) ? array_reverse($value) : (is_string($value) ? strrev($value) : $value);
452 case 'items':
453 if (!is_object($value)) {
454 return $value;
455 }
456 $map = get_object_vars($value);
457 ksort($map, SORT_STRING);
458 $items = [];
459 foreach ($map as $key => $item) {
460 $items[] = (object)['key' => (string)$key, 'value' => $item];
461 }
462 return $items;
463 case 'comma': return number_format($this->number($value, $filter), 0, '.', ',');
464 case 'sum':
465 case 'min':
466 case 'max':
467 if (!is_array($value)) {
468 return $this->number($value, $filter);
469 }
470 $numbers = array_map(fn($item) => $this->number($item, $filter), $value);
471 return $filter === 'sum' ? array_sum($numbers) : ($numbers === [] ? null : ($filter === 'min' ? min($numbers) : max($numbers)));
472 case 'default': return $this->truthy($value) ? $value : $args[0];
473 case 'eq': return $this->stringify($value) === $this->stringify($args[0]);
474 case 'pad': return str_pad($this->stringify($value), $args[0], $args[1] ?? ' ', STR_PAD_RIGHT);
475 case 'lpad': return str_pad($this->stringify($value), $args[0], $args[1] ?? ' ', STR_PAD_LEFT);
476 case 'truncate':
477 $text = $this->stringify($value);
478 return strlen($text) > $args[0] ? substr($text, 0, $args[0]) . '...' : $text;
479 case 'replace': return str_replace($args[0], $args[1], $this->stringify($value));
480 case 'join': return is_array($value) ? implode($args[0], array_map(fn($item) => $this->stringify($item), $value)) : $this->stringify($value);
481 case 'sort':
482 if (!is_array($value)) {
483 return $value;
484 }
485 $items = [];
486 foreach ($value as $index => $item) {
487 $key = $args === [] ? $item : (is_object($item) && property_exists($item, $args[0]) ? $item->{$args[0]} : null);
488 $items[] = [$this->sortKey($key), $index, $item];
489 }
490 usort($items, function ($a, $b) {
491 $comparison = $a[0][0] <=> $b[0][0];
492 if ($comparison === 0) {
493 $comparison = $a[0][0] === 2 ? strcmp($a[0][1], $b[0][1]) : ($a[0][1] <=> $b[0][1]);
494 }
495 return $comparison !== 0 ? $comparison : ($a[1] <=> $b[1]);
496 });
497 return array_map(fn($item) => $item[2], $items);
498 case 'pluck': return is_array($value) ? array_map(fn($item) => is_object($item) && property_exists($item, $args[0]) ? $item->{$args[0]} : null, $value) : $value;
499 case 'add': return $this->number($value, $filter) + $args[0];
500 case 'sub': return $this->number($value, $filter) - $args[0];
501 case 'mul': return $this->number($value, $filter) * $args[0];
502 case 'div': return (int)floor($this->number($value, $filter) / $args[0]);
503 case 'mod':
504 $number = $this->number($value, $filter);
505 return $number - $args[0] * (int)floor($number / $args[0]);
506 case 'gt': return $this->number($value, $filter) > $args[0];
507 case 'lt': return $this->number($value, $filter) < $args[0];
508 }
509 return $value;
510 }

512 private function resolve(string $name, array $contexts, array $loops): array
513 {
514 if ($name === '.') {
515 return [true, $contexts[array_key_last($contexts)]];
516 }
517 if (str_starts_with($name, '@')) {
518 if ($loops === []) {
519 return [false, null];
520 }
521 [$index, $count] = $loops[array_key_last($loops)];
522 return [true, match ($name) {
523 '@index' => $index, '@number' => $index + 1, '@first' => $index === 0, '@last' => $index === $count - 1
524 }];
525 }
526 $parts = array_map(fn($part) => ctype_digit($part) ? (int)$part : $part, explode('.', $name));
527 $found = false;
528 foreach (array_reverse($contexts) as $context) {
529 if ($this->has($context, $parts[0])) {
530 $found = true;
531 break;
532 }
533 }
534 if (!$found) {
535 return [false, null];
536 }
537 foreach ($parts as $part) {
538 if (!$this->has($context, $part)) {
539 return [false, null];
540 }
541 $context = $this->get($context, $part);
542 }
543 return [true, $context];
544 }

546 private function evaluate(array $token, array $contexts, array $loops): mixed
547 {
548 [$name, $filters] = $token['expression'];
549 [$found, $value] = $this->resolve($name, $contexts, $loops);
550 if ($this->strict && in_array($token['kind'], ['variable', 'raw'], true) && !$found) {
551 $this->fail("missing '$name'");
552 }
553 foreach ($filters as [$filter, $args]) {
554 $value = $this->filter($value, $filter, $args);
555 }
556 return $value;
557 }

564 private function render(array $nodes, array $contexts, array $loops, int $depth): string
565 {
566 $output = '';
567 foreach ($nodes as $token) {
568 try {
569 switch ($token['kind']) {
570 case 'text':
571 $output .= $token['text'];
572 break;
573 case 'variable':
574 case 'raw':
575 $text = $this->stringify($this->evaluate($token, $contexts, $loops));
576 $output .= $token['kind'] === 'variable' ? $this->html($text) : $text;
577 break;
578 case 'section':
579 $value = $this->evaluate($token, $contexts, $loops);
580 if (is_array($value)) {
581 foreach ($value as $index => $item) {
582 $output .= $this->render($token['children'], [...$contexts, $item], [...$loops, [$index, count($value)]], $depth);
583 }
584 } elseif ($this->truthy($value)) {
585 $output .= $this->render($token['children'], [...$contexts, $value], $loops, $depth);
586 }
587 break;
588 case 'inverted':
589 if (!$this->truthy($this->evaluate($token, $contexts, $loops))) {
590 $output .= $this->render($token['children'], $contexts, $loops, $depth);
591 }
592 break;
593 case 'partial':
594 $name = $token['text'];
595 if (!array_key_exists($name, $this->partials)) {
596 $this->fail("unknown partial '$name'");
597 }
598 if ($depth === 8) {
599 $this->fail("partial depth exceeded at '$name'");
600 }
601 $body = array_map(fn($entry) => [$entry[0] === '' ? '' : $token['indent'] . $entry[0], $entry[1]], $this->partials[$name]);
602 $output .= $this->render($this->analyze($body), $contexts, $loops, $depth + 1);
603 break;
604 }
605 } catch (TemplateFault $error) {
606 $error->sourceLine ??= $token['line'];
607 throw $error;
608 }
609 }
610 return $output;
611 }

613 private function renderTemplate(string $name, int $line): void
614 {
615 if (!array_key_exists($name, $this->templates)) {
616 echo "TEMPLATE $name ERROR line $line: unknown template '$name'\n";
617 return;
618 }
619 try {
620 $result = $this->render($this->analyze($this->templates[$name]), [$this->root], [], 0);
621 $count = $result === '' ? 0 : substr_count($result, "\n") + (str_ends_with($result, "\n") ? 0 : 1);
622 echo "TEMPLATE $name OK lines=$count chars=" . strlen($result) . "\n";
623 echo $result;
624 if ($result !== '' && !str_ends_with($result, "\n")) {
625 echo "\n";
626 }
627 } catch (TemplateFault $error) {
628 echo "TEMPLATE $name ERROR line {$error->sourceLine}: {$error->getMessage()}\n";
629 }
630 }
631
632 private function leaves(mixed $value, string $path, array &$output): void
633 {
634 if (is_object($value) && get_object_vars($value) !== []) {
635 $map = get_object_vars($value);
636 ksort($map, SORT_STRING);
637 foreach ($map as $key => $item) {
638 $this->leaves($item, $path === '' ? $key : "$path.$key", $output);
639 }
640 } elseif (is_array($value) && $value !== []) {
641 foreach ($value as $index => $item) {
642 $this->leaves($item, $path . "[$index]", $output);
643 }
644 } elseif ($path !== '') {
645 $notation = match (true) {
646 is_string($value) => '"' . strtr($value, ['\\' => '\\\\', '"' => '\\"', "\n" => '\\n']) . '"',
647 is_object($value) => '{}', is_array($value) => '[]', default => $this->stringify($value)
648 };
649 $output[] = "$path = $notation";
650 }
651 }
652
653 public function run(): void
654 {
655 $input = [];
656 while (($line = fgets(STDIN)) !== false) {
657 $input[] = rtrim($line, "\r\n");
658 }
659 $index = 0;
660 while ($index < count($input)) {
661 $command = $input[$index];
662 $line = ++$index;
663 if ($command === 'END') {
664 break;
665 }
666 $words = explode(' ', $command, 2);
667 $kind = $words[0];
668 $name = $words[1] ?? '';
669 if (in_array($kind, ['DATA', 'PARTIAL', 'TEMPLATE'], true)) {
670 $body = [];
671 $ending = ['DATA' => 'ENDDATA', 'PARTIAL' => 'ENDPARTIAL', 'TEMPLATE' => 'ENDTEMPLATE'][$kind];
672 while ($input[$index] !== $ending) {
673 $body[] = [$input[$index], $index + 1];
674 $index++;
675 }
676 $index++;
677 if ($kind === 'DATA') {
678 $this->data($body);
679 } elseif ($kind === 'PARTIAL') {
680 try {
681 $this->analyze($body);
682 $action = array_key_exists($name, $this->partials) ? 'REPLACED' : 'DEFINED';
683 $this->partials[$name] = $body;
684 echo "PARTIAL $name $action lines=" . count($body) . "\n";
685 } catch (TemplateFault $error) {
686 echo "PARTIAL $name ERROR line {$error->sourceLine}: {$error->getMessage()}\n";
687 }
688 } else {
689 $this->templates[$name] = $body;
690 $this->renderTemplate($name, $line);
691 }
692 } elseif ($kind === 'RENDER') {
693 $this->renderTemplate($name, $line);
694 } elseif ($kind === 'DUMP') {
695 $leaves = [];
696 $this->leaves($this->root, '', $leaves);
697 echo 'DUMP leaves=' . count($leaves) . "\n";
698 foreach ($leaves as $leaf) {
699 echo "$leaf\n";
700 }
701 } elseif ($kind === 'STRICT') {
702 $this->strict = $name === 'ON';
703 echo "$command\n";
704 } elseif ($kind === 'RESET') {
705 $this->root = new stdClass();
706 $this->partials = [];
707 $this->templates = [];
708 echo "RESET\n";
709 }
710 }
711 }
712}
