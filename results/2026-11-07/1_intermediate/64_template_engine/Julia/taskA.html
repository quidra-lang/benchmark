18function put!(v, p, x)
19 if v isa Vector && p == length(v)
20 push!(v, x)
21 else

25function keyparts(key, n)
26 occursin(KEY, key) || fail(n, "bad key '$key'")
27 [
28 (
29 m.captures[1] === nothing ? parse(Int, m.captures[2]) : String(m.captures[1]),
30 part(key, 1, m.offset+length(m.match)-1),
31 ) for m in eachmatch(r"([A-Za-z_][A-Za-z0-9_]*)|\[([0-9]+)\]", key)
32 ]
33end
34function scalar(s, at)
35 if at <= length(s) && s[at] == '"'
36 out = IOBuffer()
37 at += 1
38 while at <= length(s)
39 c = s[at]
40 at += 1
41 c == '"' && return String(take!(out)), at
42 if c == '\\'
43 at <= length(s) || error("value")
44 c = s[at]
45 at += 1
46 c in ('"', '\\', 'n') || error("value")
47 c == 'n' && (c = '\n')
48 end
49 print(out, c)
50 end
51 error("value")
52 end
53 finish = at
54 while finish <= length(s) && !(s[finish] in ",]")
55 finish += 1
56 end
57 word = part(s, at, finish-1)
58 word == "true" && return true, finish
59 word == "false" && return false, finish
60 occursin(r"^(?:0|-?[1-9][0-9]*)$", word) || error("value")
61 parse(Int, word), finish
62end
63function value(s, n)
64 try
65 s == "[]" && return Any[]
66 s == "{}" && return Dict{String,Any}()
67 if startswith(s, "[")
68 list = Any[]
69 at = 2
70 while true
71 x, at = scalar(s, at)
72 push!(list, x)
73 part(s, at) == "]" && return list
74 startswith(part(s, at), ", ") || error("value")
75 at += 2
76 end
77 end
78 x, at = scalar(s, 1)
79 at == length(s)+1 || error("value")
80 x
81 catch
82 fail(n, "bad value '$s'")
83 end
84end
85function assign!(root, line, n)
86 separator = findfirst(" = ", line)
87 separator === nothing && fail(n, "bad line")
88 key = part(line, 1, first(separator)-1)
89 parts = keyparts(key, n)
90 x = value(part(line, last(separator)+1), n)
91 current = root
92 for (i, (p, prefix)) in enumerate(parts)
93 p isa Int && p > length(current) && fail(n, "index gap at '$prefix'")
94 if i == length(parts)
95 put!(current, p, x)
96 break
97 end
98 kind = parts[i+1][1] isa String ? Dict : Vector
99 if haspart(current, p)
100 getpart(current, p) isa kind || fail(n, "type conflict at '$prefix'")
101 else
102 put!(current, p, kind === Dict ? Dict{String,Any}() : Any[])
103 end
104 current = getpart(current, p)
105 end
106end
107function unset!(root, key, n)
108 parts = keyparts(key, n)
109 current = root
110 for (i, (p, _)) in enumerate(parts)
111 haspart(current, p) || fail(n, "no value at '$key'")
112 if i == length(parts)
113 current isa Vector ? deleteat!(current, p+1) : delete!(current, p)
114 else
115 current = getpart(current, p)
116 end
117 end
118end

120function text(v, n)
121 v === nothing && return ""
122 v isa String && return v
123 v isa Bool && return v ? "true" : "false"
124 v isa Int && return string(v)
125 v isa Vector && return join((text(x, n) for x in v), ", ")
126 fail(n, "cannot render map")
127end
128function number(v, f, n)
129 v === nothing && return 0
130 v isa Int && !(v isa Bool) && return v
131 v isa String && occursin(NUMBER, v) && return parse(Int, v)
132 fail(n, "not a number for '$f'")
133end
134function unquoted(s, separator)
135 result = String[]
136 at = 1
137 quoted = false
138 for i in eachindex(s)
139 if s[i] == '\''
140 quoted = !quoted
141 elseif s[i] == separator && !quoted
142 push!(result, part(s, at, i-1))
143 at=i+1
144 end
145 end
146 push!(result, part(s, at))
147 result
148end
149function argumentsvalid(f, a)
150 ints = all(x->x isa Int, a)
151 strings = all(x->x isa String, a)
152 f in ("default", "eq") && return length(a)==1
153 f in ("pad", "lpad") && return length(a) in (1, 2) &&
154 a[1] isa Int &&
155 a[1]>=0 &&
156 (length(a)==1 || a[2] isa String && length(a[2])==1)
157 f=="truncate" && return length(a)==1 && ints && a[1]>=0
158 f=="replace" && return length(a)==2 && strings && !isempty(a[1])
159 f in ("join", "pluck") && return length(a)==1 && strings
160 f=="sort" && return length(a)<=1 && strings
161 f in ("div", "mod") && return length(a)==1 && ints && a[1]!=0
162 f in ("add", "sub", "mul", "gt", "lt") && return length(a)==1 && ints
163 isempty(a)
164end
165function expression(s, n)
166 pieces = unquoted(s, '|')
167 name = trim(pieces[1])
168 occursin(NAME, name) || fail(n, "invalid name '$name'")
169 filters = Tuple{String,Vector{Any}}[]
170 for piece in pieces[2:end]
171 fields = unquoted(trim(piece), ':')
172 f = fields[1]
173 f in FILTERS || fail(n, "unknown filter '$f'")
174 a = Any[]
175 for raw in fields[2:end]
176 if occursin(r"^'[^']*'$", raw)
177 push!(a, part(raw, 2, length(raw)-1))
178 elseif occursin(NUMBER, raw)
179 push!(a, parse(Int, raw))
180 else
181 fail(n, "bad arguments for '$f'")
182 end
183 end
184 argumentsvalid(f, a) || fail(n, "bad arguments for '$f'")
185 push!(filters, (f, a))
186 end
187 name, filters
188end

198function source(lines)
199 opening="{{"
200 closing="}}"
201 roots=Node[]
202 targets=[roots]
203 stack=Node[]
204 for (n, line) in lines
205 tokens=Node[]
206 at=1
207 while true
208 found=findnext(opening, line, at)
209 if found===nothing
210 push!(tokens, Node('t', n, part(line, at)))
211 break
212 end
213 start=first(found)
214 push!(tokens, Node('t', n, part(line, at, start-1)))
215 after=start+length(opening)
216 triple=startswith(part(line, after), "{")
217 stop=triple ? "}"*closing : closing
218 finish=findnext(stop, line, after+triple)
219 finish===nothing && fail(n, "unclosed tag")
220 content=trim(part(line, after+triple, first(finish)-1))
221 at=last(finish)+1
222 k=triple ? '&' : !isempty(content) && first(content) in "#^/!>&=" ? first(content) : 'v'
223 arg=k in ('v', '=') || triple ? content : trim(part(content, 2))
224 tag=Node(k, n, arg)
225 if k=='='
226 fields=length(content)>=2 && endswith(content, "=") ?
227 split(part(content, 2, length(content)-1), ' '; keepempty = false) : String[]
228 length(fields)==2 && all(x->!occursin('=', x), fields) ||
229 fail(n, "invalid delimiters")
230 opening, closing=String.(fields)
231 elseif k in ('v', '&', '#', '^')
232 tag.expression=expression(arg, n)
233 k in ('#', '^') && push!(stack, tag)
234 elseif k=='/'
235 occursin(NAME, arg) || fail(n, "invalid name '$arg'")
236 isempty(stack) && fail(n, "unexpected closing '$arg'")
237 expected=last(stack).expression[1]
238 arg==expected || fail(n, "mismatched closing '$arg', expected '$expected'")
239 pop!(stack)
240 elseif k=='>'
241 occursin(IDENT, arg) || fail(n, "invalid name '$arg'")
242 end
243 push!(tokens, tag)
244 end
245 tags=filter(x->x.kind!='t', tokens)
246 standalone=length(tags)==1 &&
247 tags[1].kind in "#^/!>=" &&
248 all(x->x.kind!='t' || isempty(trim(x.argument)), tokens)
249 if standalone
250 tags[1].kind=='>' && (tags[1].indent=tokens[1].argument)
251 tokens=tags
252 else
253 push!(tokens, Node('t', n, "\n"))
254 end
255 for node in tokens
256 if node.kind=='/'
257 pop!(targets)
258 elseif node.kind in ('#', '^')
259 push!(last(targets), node)
260 push!(targets, node.children)
261 elseif !(node.kind in ('!', '='))
262 push!(last(targets), node)
263 end
264 end
265 end
266 !isempty(stack) && fail(last(stack).number, "unclosed section '$(last(stack).expression[1])'")
267 roots
268end
269field(v, k) = v isa Dict ? get(v, k, nothing) : nothing
270rank(v) =
271 v===nothing ? (0, 0) :
272 v isa Bool ? (3, Int(v)) : v isa Int ? (1, v) : v isa String ? (2, v) : (4, 0)
273function applyfilter(f, a, v, n)
274 f=="upper" && return uppercase(text(v, n))
275 f=="lower" && return lowercase(text(v, n))
276 if f=="title"
277 s=text(v, n)
278 return join(
279 i==1 || s[i-1]==' ' ? uppercase(string(s[i])) : lowercase(string(s[i])) for
280 i in eachindex(s)
281 )
282 end
283 f=="trim" && return trim(text(v, n))
284 f=="length" && return v isa Union{Vector,Dict} ? length(v) : length(text(v, n))
285 f in ("first", "last") &&
286 return !(v isa Vector) ? v : isempty(v) ? nothing : f=="first" ? first(v) : last(v)
287 f=="reverse" && return v isa Union{Vector,String} ? reverse(v) : v
288 f=="items" && return v isa Dict ?
289 Any[Dict{String,Any}("key"=>k, "value"=>v[k]) for k in sort!(collect(keys(v)))] : v
290 if f=="comma"
291 x=number(v, f, n)
292 s=string(abs(x))
293 groups=String[]
294 while length(s)>3
295 pushfirst!(groups, part(s, length(s)-2))
296 s=part(s, 1, length(s)-3)
297 end
298 return (x<0 ? "-" : "")*join(pushfirst!(groups, s), ",")
299 end
300 if f in ("sum", "min", "max")
301 v isa Vector || return number(v, f, n)
302 nums=[number(x, f, n) for x in v]
303 return f=="sum" ? sum(nums; init = 0) :
304 isempty(nums) ? nothing : f=="min" ? minimum(nums) : maximum(nums)
305 end
306 f=="default" && return truth(v) ? v : a[1]
307 f=="eq" && return text(v, n)==text(a[1], n)
308 if f in ("pad", "lpad")
309 s=text(v, n)
310 fill=repeat(length(a)==1 ? " " : a[2], max(0, a[1]-length(s)))
311 return f=="pad" ? s*fill : fill*s
312 end
313 if f=="truncate"
314 s=text(v, n)
315 return length(s)>a[1] ? part(s, 1, a[1])*"..." : s
316 end
317 f=="replace" && return replace(text(v, n), a[1]=>a[2])
318 f=="join" && return v isa Vector ? join((text(x, n) for x in v), a[1]) : text(v, n)
319 f=="sort" && return v isa Vector ?
320 sort(v; by = x->rank(isempty(a) ? x : field(x, a[1])), alg = Base.Sort.MergeSort) : v
321 f=="pluck" && return v isa Vector ? Any[field(x, a[1]) for x in v] : v
322 x=number(v, f, n)
323 y=a[1]
324 f=="add" ? x+y :
325 f=="sub" ? x-y :
326 f=="mul" ? x*y : f=="div" ? fld(x, y) : f=="mod" ? mod(x, y) : f=="gt" ? x>y : x<y
327end
328function resolve(name, contexts, loops)
329 name=="." && return true, last(contexts)
330 if startswith(name, "@")
331 isempty(loops) && return false, nothing
332 i, count=last(loops)
333 return true, name=="@index" ? i : name=="@number" ? i+1 : name=="@first" ? i==0 : i==count-1
334 end
335 parts=Any[occursin(r"^[0-9]+$", p) ? parse(BigInt, p) : String(p) for p in split(name, '.')]
336 for context in reverse(contexts)
337 if haspart(context, parts[1])
338 v=context
339 for p in parts
340 haspart(v, p) || return false, nothing
341 v=getpart(v, p)
342 end
343 return true, v
344 end
345 end
346 false, nothing
347end
348escapehtml(s) = replace(s, '&'=>"&amp;", '<'=>"&lt;", '>'=>"&gt;", '"'=>"&quot;", '\''=>"&#39;")
349function render(nodes, contexts, loops, partials, strict, depth = 0)
350 out=IOBuffer()
351 for node in nodes
352 k=node.kind
353 n=node.number
354 if k=='t'
355 print(out, node.argument)
356 continue
357 end
358 if k=='>'
359 name=node.argument
360 haskey(partials, name) || fail(n, "unknown partial '$name'")
361 depth==8 && fail(n, "partial depth exceeded at '$name'")
362 lines=[(i, isempty(s) ? s : node.indent*s) for (i, s) in partials[name]]
363 print(out, render(source(lines), contexts, loops, partials, strict, depth+1))
364 continue
365 end
366 name, filters=node.expression
367 found, v=resolve(name, contexts, loops)
368 !found && strict && k in ('v', '&') && fail(n, "missing '$name'")
369 for (f, a) in filters
370 v=applyfilter(f, a, v, n)
371 end
372 if k in ('v', '&')
373 s=text(v, n)
374 print(out, k=='&' ? s : escapehtml(s))
375 elseif k=='^'
376 !truth(v) && print(out, render(node.children, contexts, loops, partials, strict, depth))
377 elseif v isa Vector
378 for (i, x) in enumerate(v)
379 print(
380 out,
381 render(
382 node.children,
383 [contexts; Any[x]],
384 [loops; [(i-1, length(v))]],
385 partials,
386 strict,
387 depth,
388 ),
389 )
390 end
391 elseif truth(v)
392 print(out, render(node.children, [contexts; Any[v]], loops, partials, strict, depth))
393 end
394 end
395 String(take!(out))
396end
397notation(v) =
398 v isa String ? "\""*replace(v, '\\'=>"\\\\", '"'=>"\\\"", '\n'=>"\\n")*"\"" :
399 v isa Bool ? (v ? "true" : "false") : v isa Int ? string(v) : v isa Vector ? "[]" : "{}"
400function leaves!(result, v, path)
401 if v isa Dict && !isempty(v)
402 for k in sort!(collect(keys(v)))
403 leaves!(result, v[k], isempty(path) ? k : path*"."*k)
404 end
405 elseif v isa Vector && !isempty(v)
406 for (i, x) in enumerate(v)
407 leaves!(result, x, path*"[$(i-1)]")
408 end
409 elseif !isempty(path)
410 push!(result, path*" = "*notation(v))
411 end
412end
413function showtemplate(name, templates, root, partials, strict)
414 try
415 result=render(source(templates[name]), Any[root], Tuple{Int,Int}[], partials, strict)
416 lines=isempty(result) ? String[] : String.(split(result, '\n'; keepempty = true))
417 !isempty(lines) && isempty(last(lines)) && pop!(lines)
418 println("TEMPLATE $name OK lines=$(length(lines)) chars=$(length(result))")
419 foreach(println, lines)
420 catch e
421 e isa TemplateError || rethrow()
422 println("TEMPLATE $name ERROR line $(e.number): $(e.reason)")
423 end
424end
425function main()
426 lines=readlines()
427 root=Dict{String,Any}()
428 partials=Dict{String,Vector{Tuple{Int,String}}}()
429 templates=copy(partials)
430 strict=false
431 at=1
432 while lines[at]!="END"
433 line=lines[at]
434 n=at
435 fields=split(line, ' '; limit = 2)
436 cmd=fields[1]
437 name=length(fields)==2 ? String(fields[2]) : ""
438 at+=1
439 if cmd in ("DATA", "PARTIAL", "TEMPLATE")
440 marker=cmd=="DATA" ? "ENDDATA" : cmd=="PARTIAL" ? "ENDPARTIAL" : "ENDTEMPLATE"
441 body=Tuple{Int,String}[]
442 while lines[at]!=marker
443 push!(body, (at, lines[at]))
444 at+=1
445 end
446 at+=1
447 if cmd=="DATA"
448 sets=0
449 unsets=0
450 errors=String[]
451 for (i, s) in body
452 (isempty(s) || startswith(s, "#")) && continue
453 candidate=deepcopy(root)
454 try
455 if startswith(s, "UNSET ")
456 unset!(candidate, part(s, 7), i)
457 unsets+=1
458 else
459 assign!(candidate, s, i)
460 sets+=1
461 end
462 root=candidate
463 catch e
464 e isa TemplateError || rethrow()
465 push!(errors, "ERROR line $(e.number): $(e.reason)")
466 end
467 end
468 println("DATA set=$sets unset=$unsets errors=$(length(errors))")
469 foreach(println, errors)
470 elseif cmd=="PARTIAL"
471 try
472 source(body)
473 status=haskey(partials, name) ? "REPLACED" : "DEFINED"
474 partials[name]=body
475 println("PARTIAL $name $status lines=$(length(body))")
476 catch e
477 e isa TemplateError || rethrow()
478 println("PARTIAL $name ERROR line $(e.number): $(e.reason)")
479 end
480 else
481 templates[name]=body
482 showtemplate(name, templates, root, partials, strict)
483 end
484 elseif cmd=="RENDER"
485 if haskey(templates, name)
486 showtemplate(name, templates, root, partials, strict)
487 else
488 println("TEMPLATE $name ERROR line $n: unknown template '$name'")
489 end
490 elseif cmd=="DUMP"
491 result=String[]
492 leaves!(result, root, "")
493 println("DUMP leaves=$(length(result))")
494 foreach(println, result)
495 elseif cmd=="STRICT"
496 strict=name=="ON"
497 println(line)
498 elseif cmd=="RESET"
499 empty!(root)
500 empty!(partials)
501 empty!(templates)
502 println("RESET")
503 end
504 end
505end
