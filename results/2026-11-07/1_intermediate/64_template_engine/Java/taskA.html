84 static List<Part> keyParts(String key) {
85 if (!key.matches(IDENT + "(?:\\." + IDENT + "|\\[" + INDEX + "\\])*")) throw new Failure("bad key '" + key + "'");
86 List<Part> result = new ArrayList<>();
87 Matcher matcher = Pattern.compile("(" + IDENT + ")|\\[(" + INDEX + ")\\]").matcher(key);
88 while (matcher.find()) result.add(new Part(matcher.group(1) != null ? matcher.group(1) : Integer.parseInt(matcher.group(2)), key.substring(0, matcher.end())));
89 return result;

98 Object scalar() {
99 if (pos < text.length() && text.charAt(pos) == '"') {
100 pos++;
101 StringBuilder out = new StringBuilder();
102 while (pos < text.length()) {
103 char c = text.charAt(pos++);
104 if (c == '"') return out.toString();
105 if (c == '\\') {
106 if (pos == text.length()) throw new Failure("");
107 c = text.charAt(pos++);
108 if (c == 'n') c = '\n';
109 else if (c != '"' && c != '\\') throw new Failure("");
110 }
111 out.append(c);
112 }
113 throw new Failure("");
114 }
115 int start = pos;
116 while (pos < text.length() && text.charAt(pos) != ',' && text.charAt(pos) != ']') pos++;
117 String word = text.substring(start, pos);
118 if (word.equals("true") || word.equals("false")) return Boolean.valueOf(word);
119 if (!word.matches("0|-?[1-9][0-9]*")) throw new Failure("");
120 return Integer.parseInt(word);
121 }
122 Object read() {
123 if (text.equals("{}")) {
124 pos = 2;
125 return new TreeMap<String, Object>();
126 }
127 if (text.equals("[]")) {
128 pos = 2;
129 return new ArrayList<Object>();
130 }
131 if (!text.startsWith("[")) return scalar();
132 pos = 1;
133 List<Object> result = new ArrayList<>();
134 while (true) {
135 result.add(scalar());
136 if (text.startsWith(", ", pos)) pos += 2;
137 else if (text.substring(pos).equals("]")) {
138 pos++;
139 return result;
140 } else throw new Failure("");
141 }
142 }

156 static void data(List<Line> lines) {
157 int sets = 0, unsets = 0;
158 List<String> errors = new ArrayList<>();
159 for (Line line : lines) {
160 String text = line.text();
161 if (text.isEmpty() || text.startsWith("#")) continue;
162 try {
163 boolean remove = text.startsWith("UNSET ");
164 int equal = text.indexOf(" = ");
165 if (!remove && equal < 0) throw new Failure("bad line");
166 String key = remove ? text.substring(6) : text.substring(0, equal);
167 List<Part> parts = keyParts(key);
168 Object item = remove ? null : value(text.substring(equal + 3));
169 Map<String, Object> candidate = map(copy(root));
170 Object container = candidate;
171 for (int i = 0; i < parts.size(); i++) {
172 Part part = parts.get(i);
173 boolean found = has(container, part.key());
174 if (remove) {
175 if (!found) throw new Failure("no value at '" + key + "'");
176 if (i == parts.size() - 1) {
177 if (part.key() instanceof String) map(container).remove(part.key());
178 else list(container).remove((int) part.key());
179 } else container = get(container, part.key());
180 } else {
181 if (part.key() instanceof Integer n && n > list(container).size()) throw new Failure("index gap at '" + part.prefix() + "'");
182 if (i == parts.size() - 1) put(container, part.key(), item);
183 else {
184 boolean wantMap = parts.get(i + 1).key() instanceof String;
185 if (found) {
186 Object next = get(container, part.key());
187 if (wantMap ? !(next instanceof Map) : !(next instanceof List)) throw new Failure("type conflict at '" + part.prefix() + "'");
188 container = next;
189 } else {
190 Object next = wantMap ? new TreeMap<String, Object>() : new ArrayList<Object>();
191 put(container, part.key(), next);
192 container = next;
193 }
194 }
195 }
196 }
197 root = candidate;
198 if (remove) unsets++;
199 else sets++;
200 } catch (Failure ex) {
201 errors.add("ERROR line " + line.number() + ": " + ex.getMessage());
202 }
203 }
204 System.out.println("DATA set=" + sets + " unset=" + unsets + " errors=" + errors.size());
205 errors.forEach(System.out::println);
206 }

208 static List<String> split(String text, char separator) {
209 List<String> result = new ArrayList<>();
210 int start = 0;
211 boolean quote = false;
212 for (int i = 0; i < text.length(); i++) {
213 if (text.charAt(i) == '\'') quote = !quote;
214 else if (text.charAt(i) == separator && !quote) {
215 result.add(text.substring(start, i));
216 start = i + 1;
217 }
218 }
219 result.add(text.substring(start));
220 return result;

223 static boolean arguments(String name, List<Object> args) {
224 int n = args.size();
225 boolean integer = n > 0 && args.get(0) instanceof Integer;
226 boolean string = n > 0 && args.get(0) instanceof String;
227 return switch (name) {
228 case "default", "eq" -> n == 1;
229 case "pad", "lpad" -> (n == 1 || n == 2) && integer && (int) args.get(0) >= 0 && (n == 1 || args.get(1) instanceof String s && s.length() == 1);
230 case "truncate" -> n == 1 && integer && (int) args.get(0) >= 0;
231 case "replace" -> n == 2 && string && !((String) args.get(0)).isEmpty() && args.get(1) instanceof String;
232 case "join", "pluck" -> n == 1 && string;
233 case "sort" -> n == 0 || n == 1 && string;
234 case "div", "mod" -> n == 1 && integer && (int) args.get(0) != 0;
235 case "add", "sub", "mul", "gt", "lt" -> n == 1 && integer;
236 default -> n == 0;
237 };
238 }

240 static Expression expression(String text, int line) {
241 List<String> pieces = split(text, '|');
242 String name = strip(pieces.get(0));
243 if (!name.matches(NAME)) throw new Failure(line, "invalid name '" + name + "'");
244 List<Filter> filters = new ArrayList<>();
245 for (String piece : pieces.subList(1, pieces.size())) {
246 List<String> fields = split(strip(piece), ':');
247 String filter = fields.get(0);
248 if (!FILTERS.contains(filter)) throw new Failure(line, "unknown filter '" + filter + "'");
249 List<Object> args = new ArrayList<>();
250 boolean valid = true;
251 for (String arg : fields.subList(1, fields.size())) {
252 if (arg.matches("'[^']*'")) args.add(arg.substring(1, arg.length() - 1));
253 else if (arg.matches("-?[0-9]{1,9}")) args.add(Integer.parseInt(arg));
254 else valid = false;
255 }
256 if (!valid || !arguments(filter, args)) throw new Failure(line, "bad arguments for '" + filter + "'");
257 filters.add(new Filter(filter, args));
258 }
259 return new Expression(name, filters);
260 }

262 static List<Node> analyze(List<Line> source, String indentation) {
263 List<Node> result = new ArrayList<>();
264 Deque<Node> open = new ArrayDeque<>();
265 Deque<List<Node>> levels = new ArrayDeque<>();
266 levels.push(result);
267 String opening = "{{", closing = "}}";
268 for (Line line : source) {
269 String text = line.text().isEmpty() ? "" : indentation + line.text();
270 List<Node> nodes = new ArrayList<>();
271 int pos = 0, tags = 0;
272 while (true) {
273 int start = text.indexOf(opening, pos);
274 if (start < 0) {
275 nodes.add(new Node('T', line.number(), text.substring(pos)));
276 break;
277 }
278 nodes.add(new Node('T', line.number(), text.substring(pos, start)));
279 int begin = start + opening.length();
280 boolean triple = begin < text.length() && text.charAt(begin) == '{';
281 if (triple) begin++;
282 String endText = (triple ? "}" : "") + closing;
283 int end = text.indexOf(endText, begin);
284 if (end < 0) throw new Failure(line.number(), "unclosed tag");
285 String content = strip(text.substring(begin, end));
286 char kind = triple ? 'R' : !content.isEmpty() && "#^/!>&=".indexOf(content.charAt(0)) >= 0 ? content.charAt(0) : 'V';
287 String arg = kind == 'V' || triple ? content : strip(content.substring(1));
288 if (kind == '&') kind = 'R';
289 Node node = new Node(kind, line.number(), arg);
290 tags++;
291 if (kind == '=') {
292 if (content.length() < 2 || !content.endsWith("=")) throw new Failure(line.number(), "invalid delimiters");
293 String middle = strip(content.substring(1, content.length() - 1));
294 String[] pair = middle.split(" +", -1);
295 if (pair.length != 2 || pair[0].isEmpty() || pair[1].isEmpty() || pair[0].contains("=") || pair[1].contains("=")) throw new Failure(line.number(), "invalid delimiters");
296 opening = pair[0];
297 closing = pair[1];
298 } else if (kind == 'V' || kind == 'R' || kind == '#' || kind == '^') {
299 node.expression = expression(arg, line.number());
300 if (kind == '#' || kind == '^') open.push(node);
301 } else if (kind == '/') {
302 if (!arg.matches(NAME)) throw new Failure(line.number(), "invalid name '" + arg + "'");
303 if (open.isEmpty()) throw new Failure(line.number(), "unexpected closing '" + arg + "'");
304 String expected = open.peek().expression.name();
305 if (!arg.equals(expected)) throw new Failure(line.number(), "mismatched closing '" + arg + "', expected '" + expected + "'");
306 open.pop();
307 } else if (kind == '>' && !arg.matches(IDENT)) throw new Failure(line.number(), "invalid name '" + arg + "'");
308 nodes.add(node);
309 pos = end + endText.length();
310 }
311 Node only = null;
312 boolean spaces = true;
313 for (Node node : nodes) {
314 if (node.kind == 'T') spaces &= node.text.matches(" *");
315 else only = node;
316 }
317 boolean standalone = tags == 1 && spaces && "#^/!>=".indexOf(only.kind) >= 0;
318 if (standalone && only.kind == '>') only.indent = nodes.get(0).text;
319 if (!standalone) nodes.add(new Node('T', line.number(), "\n"));
320 for (Node node : nodes) {
321 if (standalone && node.kind == 'T') continue;
322 if (node.kind == '/') levels.pop();
323 else {
324 levels.peek().add(node);
325 if (node.kind == '#' || node.kind == '^') levels.push(node.children);
326 }
327 }
328 }
329 if (!open.isEmpty()) throw new Failure(open.peek().line, "unclosed section '" + open.peek().expression.name() + "'");
330 return result;
331 }

369 int r = Integer.compare(rank(left), rank(right));
370 if (r != 0) return r;
371 return switch (rank(left)) {
372 case 1 -> Integer.compare((int) left, (int) right);
373 case 2 -> ((String) left).compareTo((String) right);
374 case 3 -> Boolean.compare((boolean) left, (boolean) right);
375 default -> 0;

379 static Object apply(Object value, Filter filter) {
380 String f = filter.name();
381 List<Object> a = filter.args();
382 switch (f) {
383 case "default": return truth(value) ? value : a.get(0);
384 case "eq": return string(value).equals(string(a.get(0)));
385 case "length": return value instanceof Map ? map(value).size() : value instanceof List ? list(value).size() : string(value).length();
386 case "first", "last": return value instanceof List ? list(value).isEmpty() ? null : list(value).get(f.equals("first") ? 0 : list(value).size() - 1) : value;
387 case "reverse": if (value instanceof List) {
388 List<Object> result = new ArrayList<>(list(value));
389 Collections.reverse(result);
390 return result;
391 }
392 return value instanceof String s ? new StringBuilder(s).reverse().toString() : value;
393 case "items": if (value instanceof Map) {
394 List<Object> result = new ArrayList<>();
395 for (String key : new TreeSet<>(map(value).keySet())) {
396 Map<String, Object> item = new TreeMap<>();
397 item.put("key", key);
398 item.put("value", map(value).get(key));
399 result.add(item);
400 }
401 return result;
402 }
403 return value;
404 case "sort", "pluck": if (!(value instanceof List)) return value;
405 List<Object> result = new ArrayList<>();
406 if (f.equals("pluck")) for (Object item : list(value)) result.add(sortKey(item, a));
407 else {
408 result.addAll(list(value));
409 result.sort((x, y) -> compare(sortKey(x, a), sortKey(y, a)));
410 }
411 return result;
412 case "sum", "min", "max": if (!(value instanceof List)) return number(value, f);
413 Integer total = f.equals("sum") ? 0 : null;
414 for (Object item : list(value)) {
415 int n = number(item, f);
416 total = total == null ? n : f.equals("sum") ? total + n : f.equals("min") ? Math.min(total, n) : Math.max(total, n);
417 }
418 return total;
419 case "add", "sub", "mul", "div", "mod", "gt", "lt": int n = number(value, f), argument = (int) a.get(0);
420 return switch (f) {
421 case "add" -> n + argument;
422 case "sub" -> n - argument;
423 case "mul" -> n * argument;
424 case "div" -> Math.floorDiv(n, argument);
425 case "mod" -> Math.floorMod(n, argument);
426 case "gt" -> n > argument;
427 default -> n < argument;
428 };
429 case "comma": String digits = Integer.toString(number(value, f));
430 String sign = digits.startsWith("-") ? "-" : "";
431 if (!sign.isEmpty()) digits = digits.substring(1);
432 StringBuilder comma = new StringBuilder(sign);
433 for (int i = 0; i < digits.length(); i++) {
434 if (i > 0 && (digits.length() - i) % 3 == 0) comma.append(',');
435 comma.append(digits.charAt(i));
436 }
437 return comma.toString();
438 case "join": if (value instanceof List) {
439 List<String> parts = new ArrayList<>();
440 for (Object item : list(value)) parts.add(string(item));
441 return String.join((String) a.get(0), parts);
442 }
443 return string(value);
444 }
445 String text = string(value);
446 return switch (f) {
447 case "upper" -> text.toUpperCase(Locale.ROOT);
448 case "lower" -> text.toLowerCase(Locale.ROOT);
449 case "title" -> {
450 StringBuilder out = new StringBuilder();
451 for (int i = 0; i < text.length(); i++) out.append(i == 0 || text.charAt(i - 1) == ' ' ? Character.toUpperCase(text.charAt(i)) : Character.toLowerCase(text.charAt(i)));
452 yield out.toString();
453 }
454 case "trim" -> strip(text);
455 case "pad", "lpad" -> {
456 int count = Math.max(0, (int) a.get(0) - text.length());
457 String padding = (a.size() == 1 ? " " : (String) a.get(1)).repeat(count);
458 yield f.equals("pad") ? text + padding : padding + text;
459 }
460 case "truncate" -> text.length() > (int) a.get(0) ? text.substring(0, (int) a.get(0)) + "..." : text;
461 case "replace" -> text.replace((String) a.get(0), (String) a.get(1));
462 default -> throw new IllegalStateException();
463 };
464 }

474 record Resolved(boolean found, Object value) {} static Resolved resolve(String name, List<Object> contexts, List<int[]> loops) {
475 if (name.equals(".")) return new Resolved(true, contexts.getLast());
476 if (name.startsWith("@")) {
477 if (loops.isEmpty()) return new Resolved(false, null);
478 int[] loop = loops.getLast();
479 Object value = switch (name) {
480 case "@index" -> loop[0];
481 case "@number" -> loop[0] + 1;
482 case "@first" -> loop[0] == 0;
483 default -> loop[0] == loop[1] - 1;
484 };
485 return new Resolved(true, value);
486 }
487 String[] parts = name.split("\\.");
488 Object value = null;
489 boolean found = false;
490 for (int i = contexts.size() - 1; i >= 0; i--) if (has(contexts.get(i), indexPart(parts[0]))) {
491 value = contexts.get(i);
492 found = true;
493 break;
494 }
495 if (!found) return new Resolved(false, null);
496 for (String part : parts) {
497 Object key = indexPart(part);
498 if (!has(value, key)) return new Resolved(false, null);
499 value = get(value, key);
500 }
501 return new Resolved(true, value);
502 }

508 static void render(List<Node> nodes, List<Object> contexts, List<int[]> loops, int depth, StringBuilder out) {
509 for (Node node : nodes) {
510 try {
511 if (node.kind == 'T') {
512 out.append(node.text);
513 continue;
514 }
515 if (node.kind == '>') {
516 if (!partials.containsKey(node.text)) throw new Failure("unknown partial '" + node.text + "'");
517 if (depth == 8) throw new Failure("partial depth exceeded at '" + node.text + "'");
518 render(analyze(partials.get(node.text), node.indent), contexts, loops, depth + 1, out);
519 continue;
520 }
521 if (node.expression == null) continue;
522 Resolved resolved = resolve(node.expression.name(), contexts, loops);
523 if (strict && (node.kind == 'V' || node.kind == 'R') && !resolved.found()) throw new Failure("missing '" + node.expression.name() + "'");
524 Object value = resolved.value();
525 for (Filter filter : node.expression.filters()) value = apply(value, filter);
526 if (node.kind == 'V' || node.kind == 'R') {
527 String text = string(value);
528 out.append(node.kind == 'V' ? html(text) : text);
529 } else if (node.kind == '^') {
530 if (!truth(value)) render(node.children, contexts, loops, depth, out);
531 } else if (value instanceof List) {
532 List<Object> items = list(value);
533 for (int i = 0; i < items.size(); i++) {
534 contexts.add(items.get(i));
535 loops.add(new int[]{i, items.size()});
536 render(node.children, contexts, loops, depth, out);
537 contexts.removeLast();
538 loops.removeLast();
539 }
540 } else if (truth(value)) {
541 contexts.add(value);
542 render(node.children, contexts, loops, depth, out);
543 contexts.removeLast();
544 }
545 } catch (Failure ex) {
546 if (ex.line != 0) throw ex;
547 throw new Failure(node.line, ex.getMessage());
548 }
549 }
550 }

569 static void leaves(Object value, String path, List<String> out) {
570 if (value instanceof Map && !map(value).isEmpty()) for (String key : new TreeSet<>(map(value).keySet())) leaves(map(value).get(key), path.isEmpty() ? key : path + "." + key, out);
571 else if (value instanceof List && !list(value).isEmpty()) for (int i = 0; i < list(value).size(); i++) leaves(list(value).get(i), path + "[" + i + "]", out);
572 else {
573 String notation = value instanceof String s ? "\"" + s.replace("\\", "\\\\").replace("\"", "\\\"").replace("\n", "\\n") + "\"" : value instanceof Map ? "{}" : value instanceof List ? "[]" : string(value);
574 out.add(path + " = " + notation);

578 public static void main(String[] args) throws IOException {
579 BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
580 int number = 0;
581 String command;
582 while ((command = input.readLine()) != null) {
583 number++;
584 if (command.equals("END")) break;

587 String name = type.equals("DATA") ? "" : command.substring(type.length() + 1);
588 List<Line> body = new ArrayList<>();
589 int start = number;
590 String text;
591 while ((text = input.readLine()) != null) {
592 number++;
593 if (text.equals("END" + type)) break;
594 body.add(new Line(number, text));
595 }
596 if (type.equals("DATA")) data(body);
597 else if (type.equals("TEMPLATE")) {
598 templates.put(name, body);
599 renderTemplate(name, start);
600 } else {
601 try {
602 analyze(body, "");
603 boolean existed = partials.containsKey(name);
604 partials.put(name, body);
605 System.out.println("PARTIAL " + name + (existed ? " REPLACED" : " DEFINED") + " lines=" + body.size());
606 } catch (Failure ex) {
607 System.out.println("PARTIAL " + name + " ERROR line " + ex.line + ": " + ex.getMessage());
608 }
609 }
610 } else if (command.startsWith("RENDER ")) renderTemplate(command.substring(7), number);
611 else if (command.equals("DUMP")) {
612 List<String> out = new ArrayList<>();
613 for (String key : root.keySet()) leaves(root.get(key), key, out);
614 System.out.println("DUMP leaves=" + out.size());
615 out.forEach(System.out::println);
616 } else if (command.equals("RESET")) {
617 root = new TreeMap<>();
618 partials.clear();
619 templates.clear();
620 System.out.println(command);
621 } else {
622 strict = command.equals("STRICT ON");
623 System.out.println(command);
624 }
625 }
