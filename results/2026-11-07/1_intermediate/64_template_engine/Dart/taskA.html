10final identifier = RegExp(r'^[A-Za-z_][A-Za-z0-9_]*$');
11final keyPattern = RegExp(r'^[A-Za-z_][A-Za-z0-9_]*(?:\.[A-Za-z_][A-Za-z0-9_]*|\[(?:0|[1-9][0-9]*)\])*$');
12final namePattern = RegExp(r'^(?:\.|@(?:index|number|first|last)|(?:[A-Za-z_][A-Za-z0-9_]*|0|[1-9][0-9]*)(?:\.(?:[A-Za-z_][A-Za-z0-9_]*|0|[1-9][0-9]*))*)$');
13final integerPattern = RegExp(r'^(?:0|-?[1-9][0-9]*)$');
14final numericPattern = RegExp(r'^-?[0-9]{1,9}$');
15final noArguments = {'upper', 'lower', 'title', 'trim', 'length', 'first', 'last', 'reverse', 'items', 'comma', 'sum', 'min', 'max'};
16final numericFilters = {'add', 'sub', 'mul', 'div', 'mod', 'gt', 'lt'};
17final filterNames = {...noArguments, ...numericFilters, 'default', 'eq', 'pad', 'lpad', 'truncate', 'replace', 'join', 'sort', 'pluck'};
18final standaloneKinds = {'section', 'inverted', 'close', 'comment', 'partial', 'delimiter'};
19const sigils = {'#': 'section', '^': 'inverted', '/': 'close', '!': 'comment', '>': 'partial', '&': 'raw', '=': 'delimiter'};
20String strip(String text) {
21 var start = 0;
22 var end = text.length;
23 while (start < end && text[start] == ' ') start++;
24 while (end > start && text[end - 1] == ' ') end--;
25 return text.substring(start, end);
26}

61bool hasPart(Value value, Object part) {
62 if (part is String) return value is Map<String, Value> && value.containsKey(part);
63 return part is int && value is List<Value> && part < value.length;
64}

66Value getPart(Value value, Object part) => part is String
67 ? (value as Map<String, Value>)[part]
68 : (value as List<Value>)[part as int];

70void putPart(Value container, Object part, Value value) {
71 if (part is String) {
72 (container as Map<String, Value>)[part] = value;
73 } else {
74 final list = container as List<Value>;
75 final index = part as int;
76 if (index == list.length) {
77 list.add(value);
78 } else {
79 list[index] = value;
80 }
81 }

84List<KeyPart> parseKey(String key) {
85 if (!keyPattern.hasMatch(key)) throw EngineFailure("bad key '$key'");
86 return RegExp(r'([A-Za-z_][A-Za-z0-9_]*)|\[(0|[1-9][0-9]*)\]').allMatches(key).map((match) => (
87 part: match[1] ?? int.parse(match[2]!),
88 prefix: key.substring(0, match.end),
89 )).toList();
90}

92(Value, int) readScalar(String text, int start) {
93 if (start < text.length && text[start] == '"') {
94 final result = StringBuffer();
95 for (var position = start + 1; position < text.length; position++) {
96 final char = text[position];
97 if (char == '"') return (result.toString(), position + 1);
98 if (char == r'\') {
99 position++;
100 if (position == text.length) throw const FormatException();
101 final next = text[position];
102 if (next != '"' && next != r'\' && next != 'n') throw const FormatException();
103 result.write(next == 'n' ? '\n' : next);
104 } else {
105 result.write(char);
106 }
107 }
108 throw const FormatException();
109 }
110 var end = start;
111 while (end < text.length && text[end] != ',' && text[end] != ']') end++;
112 final word = text.substring(start, end);
113 if (word == 'true' || word == 'false') return (word == 'true', end);
114 if (integerPattern.hasMatch(word)) return (int.parse(word), end);
115 throw const FormatException();
116}

118Value parseValue(String text) {
119 try {
120 if (text == '[]') return <Value>[];
121 if (text == '{}') return <String, Value>{};
122 if (text.startsWith('[')) {
123 final values = <Value>[];
124 var position = 1;
125 while (true) {
126 final (value, end) = readScalar(text, position);
127 values.add(value);
128 if (text.substring(end) == ']') return values;
129 if (!text.startsWith(', ', end)) throw const FormatException();
130 position = end + 2;
131 }
132 }
133 final (value, end) = readScalar(text, 0);
134 if (end != text.length) throw const FormatException();
135 return value;
136 } on FormatException {
137 throw EngineFailure("bad value '$text'");
138 }
139}

141List<String> splitUnquoted(String text, String separator) {
142 final parts = <String>[];
143 var quoted = false;
144 var start = 0;
145 for (var index = 0; index < text.length; index++) {
146 if (text[index] == "'") {
147 quoted = !quoted;
148 } else if (text[index] == separator && !quoted) {
149 parts.add(text.substring(start, index));
150 start = index + 1;
151 }
152 }
153 parts.add(text.substring(start));
154 return parts;
155}

157bool validArguments(String name, List<FilterArgument> args) {
158 final number = args.isNotEmpty && args[0] is int;
159 final string = args.isNotEmpty && args[0] is String;
160 if (noArguments.contains(name)) return args.isEmpty;
161 if (name == 'default' || name == 'eq') return args.length == 1;
162 if (name == 'sort') return args.isEmpty || args.length == 1 && string;
163 if (name == 'join' || name == 'pluck') return args.length == 1 && string;
164 if (name == 'replace') return args.length == 2 && string && args[0] != '' && args[1] is String;
165 if (name == 'pad' || name == 'lpad') {
166 return number && (args[0] as int) >= 0 && (args.length == 1 || args.length == 2 && args[1] is String && (args[1] as String).length == 1);
167 }
168 if (name == 'truncate') return args.length == 1 && number && (args[0] as int) >= 0;
169 return args.length == 1 && number && (!(name == 'div' || name == 'mod') || args[0] != 0);
170}

172Expression parseExpression(String text, int number) {
173 final parts = splitUnquoted(text, '|');
174 final name = strip(parts[0]);
175 if (!namePattern.hasMatch(name)) throw TemplateFailure(number, "invalid name '$name'");
176 final chain = <TemplateFilter>[];
177 for (final part in parts.skip(1)) {
178 final pieces = splitUnquoted(strip(part), ':');
179 final filter = pieces[0];
180 if (!filterNames.contains(filter)) throw TemplateFailure(number, "unknown filter '$filter'");
181 final args = <FilterArgument>[];
182 var valid = true;
183 for (final notation in pieces.skip(1)) {
184 if (RegExp(r"^'[^']*'$").hasMatch(notation)) {
185 args.add(notation.substring(1, notation.length - 1));
186 } else if (numericPattern.hasMatch(notation)) {
187 args.add(int.parse(notation));
188 } else {
189 valid = false;
190 }
191 }
192 if (!valid || !validArguments(filter, args)) throw TemplateFailure(number, "bad arguments for '$filter'");
193 chain.add(TemplateFilter(filter, args));
194 }
195 return Expression(name, chain);
196}

198List<TemplateNode> parseSource(List<SourceLine> source) {
199 var opening = '{{';
200 var closing = '}}';
201 final nodes = <TemplateNode>[];
202 final targets = <List<TemplateNode>>[nodes];
203 final sections = <Tag>[];
204 for (final line in source) {
205 final text = line.text;
206 final number = line.number;
207 var position = 0;
208 final tokens = <TemplateNode>[];
209 final tags = <Tag>[];
210 while (true) {
211 final start = text.indexOf(opening, position);
212 if (start < 0) {
213 tokens.add(text.substring(position));
214 break;
215 }
216 tokens.add(text.substring(position, start));
217 final after = start + opening.length;
218 final triple = after < text.length && text[after] == '{';
219 final terminator = triple ? '}$closing' : closing;
220 final end = text.indexOf(terminator, after + (triple ? 1 : 0));
221 if (end < 0) throw TemplateFailure(number, 'unclosed tag');
222 final content = strip(text.substring(after + (triple ? 1 : 0), end));
223 final kind = triple ? 'raw' : sigils[content.isEmpty ? '' : content[0]] ?? 'variable';
224 final argument = triple || kind == 'variable' || kind == 'delimiter' ? content : strip(content.substring(1));
225 final tag = Tag(kind, number, argument);
226 position = end + terminator.length;
227 if (kind == 'delimiter') {
228 if (content.length < 2 || !content.endsWith('=')) throw TemplateFailure(number, 'invalid delimiters');
229 final words = strip(content.substring(1, content.length - 1)).split(RegExp(r' +'));
230 if (words.length != 2 || words.any((word) => word.contains('='))) throw TemplateFailure(number, 'invalid delimiters');
231 opening = words[0];
232 closing = words[1];
233 } else if (kind == 'variable' || kind == 'raw' || kind == 'section' || kind == 'inverted') {
234 tag.expression = parseExpression(argument, number);
235 if (kind == 'section' || kind == 'inverted') sections.add(tag);
236 } else if (kind == 'close') {
237 if (!namePattern.hasMatch(argument)) throw TemplateFailure(number, "invalid name '$argument'");
238 if (sections.isEmpty) throw TemplateFailure(number, "unexpected closing '$argument'");
239 final expected = sections.last.expression!.name;
240 if (argument != expected) throw TemplateFailure(number, "mismatched closing '$argument', expected '$expected'");
241 sections.removeLast();
242 } else if (kind == 'partial' && !identifier.hasMatch(argument)) {
243 throw TemplateFailure(number, "invalid name '$argument'");
244 }
245 tags.add(tag);
246 tokens.add(tag);
247 }
248 final standalone = tags.length == 1 && standaloneKinds.contains(tags[0].kind) && tokens.every((token) => token is! String || RegExp(r'^ *$').hasMatch(token));
249 final List<TemplateNode> placed;
250 if (standalone) {
251 if (tags[0].kind == 'partial') tags[0].indentation = tokens[0] as String;
252 placed = tags;
253 } else {
254 tokens.add('\n');
255 placed = tokens;
256 }
257 for (final token in placed) {
258 if (token is String) {
259 if (token.isNotEmpty) targets.last.add(token);
260 } else {
261 final tag = token as Tag;
262 if (tag.kind == 'section' || tag.kind == 'inverted') {
263 targets.last.add(tag);
264 targets.add(tag.children);
265 } else if (tag.kind == 'close') {
266 targets.removeLast();
267 } else if (tag.kind != 'comment' && tag.kind != 'delimiter') {
268 targets.last.add(tag);
269 }
270 }
271 }
272 }
273 if (sections.isNotEmpty) {
274 final tag = sections.last;
275 throw TemplateFailure(tag.number, "unclosed section '${tag.expression!.name}'");
276 }
277 return nodes;
278}

280String toText(Value value) {
281 if (value == null) return '';
282 if (value is Map<String, Value>) throw EngineFailure('cannot render map');
283 if (value is List<Value>) return value.map(toText).join(', ');
284 return value.toString();
285}

287bool truthy(Value value) {
288 if (value is Map<String, Value>) return value.isNotEmpty;
289 if (value is List<Value>) return value.isNotEmpty;
290 return value != null && value != false && value != 0 && value != '';
291}

293int toNumber(Value value, String filter) {
294 if (value == null) return 0;
295 if (value is int) return value;
296 if (value is String && !value.contains('\n') && numericPattern.hasMatch(value)) return int.parse(value);
297 throw EngineFailure("not a number for '$filter'");
298}

300Value field(Value value, String key) => value is Map<String, Value> ? value[key] : null;
301int sortRank(Value value) => value == null ? 0 : value is int ? 1 : value is String ? 2 : value is bool ? 3 : 4;
302
303int compareValues(Value left, Value right) {
304 final rank = sortRank(left) - sortRank(right);
305 if (rank != 0) return rank;
306 if (left is int && right is int) return left.compareTo(right);
307 if (left is String && right is String) return left.compareTo(right);
308 if (left is bool && right is bool) return left == right ? 0 : left ? 1 : -1;
309 return 0;
310}

312Value applyFilter(TemplateFilter filter, Value value) {
313 final name = filter.name;
314 final args = filter.arguments;
315 switch (name) {
316 case 'upper': return toText(value).toUpperCase();
317 case 'lower': return toText(value).toLowerCase();
318 case 'title':
319 final text = toText(value);
320 return List.generate(text.length, (i) => i == 0 || text[i - 1] == ' ' ? text[i].toUpperCase() : text[i].toLowerCase()).join();
321 case 'trim': return strip(toText(value));
322 case 'length': return value is Map<String, Value> ? value.length : value is List<Value> ? value.length : toText(value).length;
323 case 'first': return value is List<Value> ? value.isEmpty ? null : value.first : value;
324 case 'last': return value is List<Value> ? value.isEmpty ? null : value.last : value;
325 case 'reverse': return value is List<Value> ? value.reversed.toList() : value is String ? value.split('').reversed.join() : value;
326 case 'items':
327 if (value is! Map<String, Value>) return value;
328 final keys = value.keys.toList()..sort();
329 return keys.map<Value>((key) => <String, Value>{'key': key, 'value': value[key]}).toList();
330 case 'comma':
331 final number = toNumber(value, name);
332 final digits = number.abs().toString();
333 final groups = <String>[];
334 for (var end = digits.length; end > 0; end -= 3) {
335 groups.add(digits.substring(end > 3 ? end - 3 : 0, end));
336 }
337 return '${number < 0 ? '-' : ''}${groups.reversed.join(',')}';
338 case 'sum': case 'min': case 'max':
339 if (value is! List<Value>) return toNumber(value, name);
340 final numbers = value.map((element) => toNumber(element, name)).toList();
341 if (name == 'sum') return numbers.fold<int>(0, (total, number) => total + number);
342 if (numbers.isEmpty) return null;
343 return numbers.reduce((a, b) => name == 'min' ? a < b ? a : b : a > b ? a : b);
344 case 'default': return truthy(value) ? value : args[0];
345 case 'eq': return toText(value) == toText(args[0]);
346 case 'pad': return toText(value).padRight(args[0] as int, args.length == 2 ? args[1] as String : ' ');
347 case 'lpad': return toText(value).padLeft(args[0] as int, args.length == 2 ? args[1] as String : ' ');
348 case 'truncate':
349 final text = toText(value);
350 final limit = args[0] as int;
351 return text.length > limit ? '${text.substring(0, limit)}...' : text;
352 case 'replace': return toText(value).replaceAll(args[0] as String, args[1] as String);
353 case 'join': return value is List<Value> ? value.map(toText).join(args[0] as String) : toText(value);
354 case 'pluck': return value is List<Value> ? value.map((element) => field(element, args[0] as String)).toList() : value;
355 case 'sort':
356 if (value is! List<Value>) return value;
357 final decorated = List.generate(value.length, (i) => (value: value[i], index: i, key: args.isEmpty ? value[i] : field(value[i], args[0] as String)));
358 decorated.sort((a, b) {
359 final comparison = compareValues(a.key, b.key);
360 return comparison == 0 ? a.index.compareTo(b.index) : comparison;
361 });
362 return decorated.map((item) => item.value).toList();
363 default:
364 final number = toNumber(value, name);
365 final n = args[0] as int;
366 switch (name) {
367 case 'add': return number + n;
368 case 'sub': return number - n;
369 case 'mul': return number * n;
370 case 'div': return (number / n).floor();
371 case 'mod': return number - n * (number / n).floor();
372 case 'gt': return number > n;
373 default: return number < n;
374 }
375 }
376}

378class Renderer {
379 final Map<String, List<SourceLine>> partials;
380 final bool strict;
381 final List<Value> contexts;
382 final List<({int index, int count})> loops = [];
383 final StringBuffer output = StringBuffer();
384 Renderer(Map<String, Value> root, this.partials, this.strict) : contexts = [root];
385
386 (bool, Value) resolve(String name) {
387 if (name == '.') return (true, contexts.last);
388 if (name.startsWith('@')) {
389 if (loops.isEmpty) return (false, null);
390 final loop = loops.last;
391 if (name == '@index') return (true, loop.index);
392 if (name == '@number') return (true, loop.index + 1);
393 return (true, name == '@first' ? loop.index == 0 : loop.index == loop.count - 1);
394 }
395 final parts = name.split('.').map<Object>((part) => RegExp(r'^[0-9]+$').hasMatch(part) ? int.tryParse(part) ?? BigInt.parse(part) : part).toList();
396 for (final context in contexts.reversed) {
397 if (!hasPart(context, parts[0])) continue;
398 Value value = context;
399 for (final part in parts) {
400 if (!hasPart(value, part)) return (false, null);
401 value = getPart(value, part);
402 }
403 return (true, value);
404 }
405 return (false, null);
406 }
407
408 Value evaluate(Expression expression, bool strict) {
409 final (found, resolved) = resolve(expression.name);
410 if (!found && strict) throw EngineFailure("missing '${expression.name}'");
411 Value value = resolved;
412 for (final filter in expression.filters) value = applyFilter(filter, value);
413 return value;
414 }

416 void render(List<TemplateNode> nodes, int depth) {
417 for (final node in nodes) {
418 if (node is String) {
419 output.write(node);
420 continue;
421 }
422 final tag = node as Tag;
423 try {
424 if (tag.kind == 'variable' || tag.kind == 'raw') {
425 final text = toText(evaluate(tag.expression!, strict));
426 const escapes = {'&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'};
427 output.write(tag.kind == 'raw' ? text : text.replaceAllMapped(RegExp('[&<>"\']'), (match) => escapes[match[0]]!));
428 } else if (tag.kind == 'section') {
429 final value = evaluate(tag.expression!, false);
430 if (value is List<Value>) {
431 for (var index = 0; index < value.length; index++) {
432 contexts.add(value[index]);
433 loops.add((index: index, count: value.length));
434 render(tag.children, depth);
435 loops.removeLast();
436 contexts.removeLast();
437 }
438 } else if (truthy(value)) {
439 contexts.add(value);
440 render(tag.children, depth);
441 contexts.removeLast();
442 }
443 } else if (tag.kind == 'inverted') {
444 if (!truthy(evaluate(tag.expression!, false))) render(tag.children, depth);
445 } else if (tag.kind == 'partial') {
446 final source = partials[tag.argument];
447 if (source == null) throw EngineFailure("unknown partial '${tag.argument}'");
448 if (depth == 8) throw EngineFailure("partial depth exceeded at '${tag.argument}'");
449 render(parseSource(source.map((line) => (number: line.number, text: line.text.isEmpty ? '' : tag.indentation + line.text)).toList()), depth + 1);
450 }
451 } on EngineFailure catch (error) {
452 throw TemplateFailure(tag.number, error.reason);
453 }
454 }
455 }
456}

458class TemplateEngine {
459 final Map<String, Value> root = {};
460 final Map<String, List<SourceLine>> partials = {};
461 final Map<String, List<SourceLine>> templates = {};
462 bool strict = false;
463 final List<String> output = [];
464
465 void assign(String line) {
466 final separator = line.indexOf(' = ');
467 if (separator < 0) throw EngineFailure('bad line');
468 final parts = parseKey(line.substring(0, separator));
469 final value = parseValue(line.substring(separator + 3));
470 Value container = root;
471 ({Value container, Object part, Value child})? detached;
472 for (var index = 0; index < parts.length; index++) {
473 final part = parts[index].part;
474 final prefix = parts[index].prefix;
475 if (part is int && part > (container as List<Value>).length) throw EngineFailure("index gap at '$prefix'");
476 if (index == parts.length - 1) {
477 putPart(container, part, value);
478 break;
479 }
480 final mapRequired = parts[index + 1].part is String;
481 Value child;
482 if (hasPart(container, part)) {
483 child = getPart(container, part);
484 if (!(mapRequired ? child is Map<String, Value> : child is List<Value>)) throw EngineFailure("type conflict at '$prefix'");
485 } else {
486 child = mapRequired ? <String, Value>{} : <Value>[];
487 if (detached == null) {
488 detached = (container: container, part: part, child: child);
489 } else {
490 putPart(container, part, child);
491 }
492 }
493 container = child;
494 }
495 if (detached != null) putPart(detached.container, detached.part, detached.child);
496 }
497
498 void unset(String key) {
499 final parts = parseKey(key);
500 Value container = root;
501 for (var index = 0; index < parts.length; index++) {
502 final part = parts[index].part;
503 if (!hasPart(container, part)) throw EngineFailure("no value at '$key'");
504 if (index == parts.length - 1) {
505 if (part is String) {
506 (container as Map<String, Value>).remove(part);
507 } else {
508 (container as List<Value>).removeAt(part as int);
509 }
510 } else {
511 container = getPart(container, part);
512 }
513 }
514 }
515
516 void loadData(List<SourceLine> body) {
517 var assigned = 0;
518 var unsetCount = 0;
519 final errors = <String>[];
520 for (final line in body) {
521 if (line.text.isEmpty || line.text.startsWith('#')) continue;
522 try {
523 if (line.text.startsWith('UNSET ')) {
524 unset(line.text.substring(6));
525 unsetCount++;
526 } else {
527 assign(line.text);
528 assigned++;
529 }
530 } on EngineFailure catch (error) {
531 errors.add('ERROR line ${line.number}: ${error.reason}');
532 }
533 }
534 output.add('DATA set=$assigned unset=$unsetCount errors=${errors.length}');
535 output.addAll(errors);
536 }
537
538 void render(String name) {
539 try {
540 final renderer = Renderer(root, partials, strict);
541 renderer.render(parseSource(templates[name]!), 0);
542 final result = renderer.output.toString();
543 final lines = result.isEmpty ? <String>[] : result.split('\n');
544 if (result.endsWith('\n')) lines.removeLast();
545 output.add('TEMPLATE $name OK lines=${lines.length} chars=${result.length}');
546 output.addAll(lines);
547 } on TemplateFailure catch (error) {
548 output.add('TEMPLATE $name ERROR line ${error.number}: ${error.reason}');
549 }
550 }

552 void dump() {
553 final leaves = <String>[];
554 void visit(Value value, String path) {
555 if (value is Map<String, Value> && value.isNotEmpty) {
556 final keys = value.keys.toList()..sort();
557 for (final key in keys) visit(value[key], '$path.$key');
558 } else if (value is List<Value> && value.isNotEmpty) {
559 for (var index = 0; index < value.length; index++) visit(value[index], '$path[$index]');
560 } else {
561 final notation = value is String ? '"${value.replaceAll(r'\', r'\\').replaceAll('"', r'\"').replaceAll('\n', r'\n')}"' : value is Map<String, Value> ? '{}' : value is List<Value> ? '[]' : value.toString();
562 leaves.add('$path = $notation');
563 }
564 }
565 final keys = root.keys.toList()..sort();
566 for (final key in keys) visit(root[key], key);
567 output.add('DUMP leaves=${leaves.length}');
568 output.addAll(leaves);
569 }

571 void run(List<String> lines) {
572 var position = 0;
573 while (lines[position] != 'END') {
574 final line = lines[position++];
575 final number = position;
576 final parts = line.split(' ');
577 final command = parts[0];
578 final name = parts.length > 1 ? parts[1] : '';
579 if (command == 'DATA' || command == 'PARTIAL' || command == 'TEMPLATE') {
580 final body = <SourceLine>[];
581 while (lines[position] != 'END$command') {
582 body.add((number: position + 1, text: lines[position]));
583 position++;
584 }
585 position++;
586 if (command == 'DATA') {
587 loadData(body);
588 } else if (command == 'TEMPLATE') {
589 templates[name] = body;
590 render(name);
591 } else {
592 try {
593 parseSource(body);
594 final status = partials.containsKey(name) ? 'REPLACED' : 'DEFINED';
595 partials[name] = body;
596 output.add('PARTIAL $name $status lines=${body.length}');
597 } on TemplateFailure catch (error) {
598 output.add('PARTIAL $name ERROR line ${error.number}: ${error.reason}');
599 }
600 }
601 } else if (command == 'RENDER') {
602 if (templates.containsKey(name)) {
603 render(name);
604 } else {
605 output.add("TEMPLATE $name ERROR line $number: unknown template '$name'");
606 }
607 } else if (command == 'DUMP') {
608 dump();
609 } else if (command == 'STRICT') {
610 strict = name == 'ON';
611 output.add(line);
612 } else if (command == 'RESET') {
613 root.clear();
614 partials.clear();
615 templates.clear();
616 output.add('RESET');
617 }
618 }
619 }
620}
621
622Future<void> main() async {
623 final engine = TemplateEngine();
624 engine.run((await stdin.transform(utf8.decoder).join()).split('\n'));
625 if (engine.output.isNotEmpty) stdout.write('${engine.output.join('\n')}\n');
626}
