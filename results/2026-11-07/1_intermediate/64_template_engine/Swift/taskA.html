12func cut(_ s: String, _ a: Int, _ b: Int? = nil) -> String {
13 String(decoding: Array(s.utf8)[a..<(b ?? s.utf8.count)], as: UTF8.self)
14}
15func find(_ s: String, _ needle: String, _ at: Int = 0) -> Int? {
16 let tail = cut(s, at)
17 guard let range = tail.range(of: needle) else {
18 return nil
19 }
20 return at + tail.distance(from: tail.startIndex, to: range.lowerBound)
21}
22func trim(_ s: String) -> String {
23 s.trimmingCharacters(in: CharacterSet(charactersIn: " "))
24}
25func letter(_ c: UInt8) -> Bool {
26 (65...90).contains(c) || (97...122).contains(c)
27}
28func digit(_ c: UInt8) -> Bool {
29 (48...57).contains(c)
30}
31func identifier(_ s: String) -> Bool {
32 let b = Array(s.utf8)
33 return !b.isEmpty && (letter(b[0]) || b[0] == 95) && b.allSatisfy { letter($0) || digit($0) || $0 == 95 }
34}
35func subscriptName(_ s: String) -> Bool {
36 let b = Array(s.utf8)
37 return s == "0" || (!b.isEmpty && b[0] != 48 && b.allSatisfy(digit))
38}
39func integerArgument(_ s: String) -> Int? {
40 let b = Array((s.hasPrefix("-") ? cut(s, 1) : s).utf8)
41 return (1...9).contains(b.count) && b.allSatisfy(digit) ? Int(s) : nil
42}

44enum Part { case key(String), index(Int) }
45struct KeyPart { let part: Part
46 let prefix: String
47}
48func keyParts(_ s: String) -> [KeyPart]? {
49 let b = Array(s.utf8)
50 var at = 0, result: [KeyPart] = []
51 while true {
52 let start = at
53 while at < b.count && (letter(b[at]) || digit(b[at]) || b[at] == 95) {
54 at += 1
55 }
56 guard identifier(cut(s, start, at)) else {
57 return nil
58 }
59 result.append(KeyPart(part: .key(cut(s, start, at)), prefix: cut(s, 0, at)))
60 while true {
61 if at == b.count {
62 return result
63 }
64 if b[at] == 46 {
65 at += 1
66 break
67 }
68 guard b[at] == 91, let end = find(s, "]", at + 1), subscriptName(cut(s, at + 1, end)), let index = Int(cut(
69 s,
70 at + 1,
71 end
72 )) else {
73 return nil
74 }
75 at = end + 1
76 result.append(KeyPart(part: .index(index), prefix: cut(s, 0, at)))
77 }
78 }

80func scalar(_ s: String, _ at: inout Int) -> Value? {
81 let b = Array(s.utf8)
82 if at < b.count && b[at] == 34 {
83 at += 1
84 var text: [UInt8] = []
85 while at < b.count {
86 let c = b[at]
87 at += 1
88 if c == 34 {
89 return .string(String(decoding: text, as: UTF8.self))
90 }
91 if c == 92 {
92 guard at < b.count else {
93 return nil
94 }
95 switch b[at] {
96 case 34:
97 text.append(34)
98 case 92:
99 text.append(92)
100 case 110:
101 text.append(10)
102 default:
103 return nil
104 }
105 at += 1
106 } else {
107 text.append(c)
108 }
109 }
110 return nil
111 }
112 let start = at
113 while at < b.count && b[at] != 44 && b[at] != 93 {
114 at += 1
115 }
116 let word = cut(s, start, at)
117 if word == "true" {
118 return .boolean(true)
119 }
120 if word == "false" {
121 return .boolean(false)
122 }
123 let digits = word.hasPrefix("-") ? cut(word, 1) : word
124 guard word != "-0", subscriptName(digits), let n = Int(word) else {
125 return nil
126 }
127 return .integer(n)
128}
129func parseValue(_ s: String) -> Value? {
130 if s == "{}" {
131 return .map([:])
132 }
133 if s == "[]" {
134 return .list([])
135 }
136 var at = 0
137 if s.hasPrefix("[") {
138 at = 1
139 var list: [Value] = []
140 while true {
141 guard let value = scalar(s, &at) else {
142 return nil
143 }
144 list.append(value)
145 if cut(s, at).hasPrefix("]") {
146 return at + 1 == s.utf8.count ? .list(list) : nil
147 }
148 guard cut(s, at).hasPrefix(", ") else {
149 return nil
150 }
151 at += 2
152 }
153 }
154 guard let value = scalar(s, &at), at == s.utf8.count else {
155 return nil
156 }
157 return value
158}
159func child(_ value: Value, _ part: Part) -> Value? {
160 switch (value, part) {
161 case (.map(let map), .key(let key)):
162 return map[key]
163 case (.list(let list), .index(let index)):
164 return index < list.count ? list[index] : nil
165 default:
166 return nil
167 }
168}
169func assigning(_ container: Value, _ parts: ArraySlice<KeyPart>, _ value: Value) throws -> Value {
170 let first = parts.first!
171 if case .list(let list) = container, case .index(let i) = first.part, i > list.count {
172 throw Problem(reason: "index gap at '\(first.prefix)'")
173 }
174 var next = value
175 if parts.count > 1 {
176 let mapNeeded: Bool
177 if case .key = parts.dropFirst().first!.part {
178 mapNeeded = true
179 } else {
180 mapNeeded = false
181 }
182 let old = child(container, first.part) ?? (mapNeeded ? .map([:]) : .list([]))
183 switch (old, mapNeeded) {
184 case (.map, true), (.list, false):
185 break
186 default:
187 throw Problem(reason: "type conflict at '\(first.prefix)'")
188 }
189 next = try assigning(old, parts.dropFirst(), value)
190 }
191 switch (container, first.part) {
192 case (.map(var map), .key(let key)):
193 map[key] = next
194 return .map(map)
195 case (.list(var list), .index(let i)):
196 if i == list.count {
197 list.append(next)
198 } else {
199 list[i] = next
200 }
201 return .list(list)
202 default:
203 fatalError("invalid container")
204 }
205}
206func removing(_ container: Value, _ parts: ArraySlice<KeyPart>) -> Value? {
207 let part = parts.first!.part
208 guard let old = child(container, part) else {
209 return nil
210 }
211 if parts.count == 1 {
212 switch (container, part) {
213 case (.map(var map), .key(let key)):
214 map.removeValue(forKey: key)
215 return .map(map)
216 case (.list(var list), .index(let i)):
217 list.remove(at: i)
218 return .list(list)
219 default:
220 return nil
221 }
222 }
223 guard let next = removing(old, parts.dropFirst()) else {
224 return nil
225 }
226 switch (container, part) {
227 case (.map(var map), .key(let key)):
228 map[key] = next
229 return .map(map)
230 case (.list(var list), .index(let i)):
231 list[i] = next
232 return .list(list)
233 default:
234 return nil
235 }
236}
237func stringify(_ v: Value) throws -> String {
238 switch v {
239 case .absent:
240 return ""
241 case .string(let s):
242 return s
243 case .integer(let n):
244 return String(n)
245 case .boolean(let b):
246 return b ? "true" : "false"
247 case .list(let list):
248 return try list.map(stringify).joined(separator: ", ")
249 case .map:
250 throw Problem(reason: "cannot render map")
251 }

254 switch v {
255 case .absent:
256 return false
257 case .string(let s):
258 return !s.isEmpty
259 case .integer(let n):
260 return n != 0
261 case .boolean(let b):
262 return b
263 case .list(let list):
264 return !list.isEmpty
265 case .map(let map):

269func number(_ v: Value, _ filter: String) throws -> Int {
270 switch v {
271 case .absent:
272 return 0
273 case .integer(let n):
274 return n
275 case .string(let s):
276 if let n = integerArgument(s) {
277 return n
278 }
279 default:
280 break
281 }
282 throw Problem(reason: "not a number for '\(filter)'")
283}
284func quotedSplit(_ s: String, _ separator: UInt8) -> [String] {
285 var quoted = false, start = 0, result: [String] = []
286 for (i, c) in s.utf8.enumerated() {
287 if c == 39 {
288 quoted.toggle()
289 }
290 if c == separator && !quoted {
291 result.append(cut(s, start, i))
292 start = i + 1
293 }
294 }
295 result.append(cut(s, start))
296 return result
297}
298func validName(_ s: String) -> Bool {
299 s == "." || ["@index", "@number", "@first", "@last"].contains(s) || s.components(separatedBy: ".")
300 .allSatisfy { identifier($0) || subscriptName($0) }

302struct Filter { let name: String
303 let args: [Value]

305struct Expression { let name: String
306 let filters: [Filter]

308func expression(_ s: String) throws -> Expression {
309 let pieces = quotedSplit(s, 124), name = trim(pieces[0])
310 guard validName(name) else {
311 throw Problem(reason: "invalid name '\(name)'")
312 }
313 let noArgs = [
314 "upper",
315 "lower",
316 "title",
317 "trim",
318 "length",
319 "first",
320 "last",
321 "reverse",
322 "items",
323 "comma",
324 "sum",
325 "min",
326 "max"
327 ]
328 let others = [
329 "default",
330 "eq",
331 "pad",
332 "lpad",
333 "truncate",
334 "replace",
335 "join",
336 "sort",
337 "pluck",
338 "add",
339 "sub",
340 "mul",
341 "div",
342 "mod",
343 "gt",
344 "lt"
345 ]
346 var filters: [Filter] = []
347 for piece in pieces.dropFirst() {
348 let items = quotedSplit(trim(piece), 58), f = items[0]
349 guard noArgs.contains(f) || others.contains(f) else {
350 throw Problem(reason: "unknown filter '\(f)'")
351 }
352 var args: [Value] = []
353 for arg in items.dropFirst() {
354 if arg.utf8.count >= 2 && arg.hasPrefix("'") && arg.hasSuffix("'") && !cut(arg, 1, arg.utf8.count - 1)
355 .contains("'")
356 {
357 args.append(.string(cut(arg, 1, arg.utf8.count - 1)))
358 } else if let n = integerArgument(arg) {
359 args.append(.integer(n))
360 } else {
361 throw Problem(reason: "bad arguments for '\(f)'")
362 }
363 }
364 func stringArg(_ i: Int) -> String? {
365 if i < args.count, case .string(let s) = args[i] {
366 return s
367 }
368 return nil
369 }
370 func intArg(_ i: Int) -> Int? {
371 if i < args.count, case .integer(let n) = args[i] {
372 return n
373 }
374 return nil
375 }
376 var good = false
377 if noArgs.contains(f) {
378 good = args.isEmpty
379 } else {
380 switch f {
381 case "default", "eq":
382 good = args.count == 1
383 case "pad",
384 "lpad":
385 good = (args.count == 1 || args.count == 2) && (intArg(0) ?? -1) >= 0 &&
386 (args.count == 1 || stringArg(1)?.utf8.count == 1)
387 case "truncate":
388 good = args.count == 1 && (intArg(0) ?? -1) >= 0
389 case "replace":
390 good = args.count == 2 && !(stringArg(0) ?? "").isEmpty && stringArg(1) != nil
391 case "join", "pluck":
392 good = args.count == 1 && stringArg(0) != nil
393 case "sort":
394 good = args.isEmpty || (args.count == 1 && stringArg(0) != nil)
395 case "div", "mod":
396 good = args.count == 1 && intArg(0) != nil && intArg(0) != 0
397 default:
398 good = args.count == 1 && intArg(0) != nil
399 }
400 }
401 guard good else {
402 throw Problem(reason: "bad arguments for '\(f)'")
403 }
404 filters.append(Filter(name: f, args: args))
405 }
406 return Expression(name: name, filters: filters)
407}
408func sortCompare(_ a: Value, _ b: Value) -> Int {
409 func rank(_ v: Value) -> Int {
410 switch v { case .absent:
411 return 0
412 case .integer:
413 return 1
414 case .string:
415 return 2
416 case .boolean:
417 return 3
418 default:
419 return 4 }
420 }
421 if rank(a) != rank(b) {
422 return rank(a) < rank(b) ? -1 : 1
423 }
424 switch (a, b) {
425 case (.integer(let a), .integer(let b)):
426 return a == b ? 0 : (a < b ? -1 : 1)
427 case (.string(let a), .string(let b)):
428 return a == b ? 0 : (a < b ? -1 : 1)
429 case (.boolean(let a), .boolean(let b)):
430 return a == b ? 0 : (!a ? -1 : 1)
431 default:
432 return 0
433 }
434}
435func applyFilter(_ v: Value, _ f: Filter) throws -> Value {
436 let name = f.name
437 func n() -> Int {
438 if case .integer(let x) = f.args[0] {
439 return x
440 }
441 fatalError("integer argument")
442 }
443 func s(_ i: Int) -> String {
444 if case .string(let x) = f.args[i] {
445 return x
446 }
447 fatalError("string argument")
448 }
449 switch name {
450 case "default":
451 return truthy(v) ? v : f.args[0]
452 case "eq":
453 return .boolean(try stringify(v) == stringify(f.args[0]))
454 case "length":
455 switch v { case .list(let l):
456 return .integer(l.count)
457 case .map(let m):
458 return .integer(m.count)
459 default:
460 return .integer(try stringify(v).utf8.count) }
461 case "first", "last":
462 if case .list(let l) = v {
463 return (name == "first" ? l.first : l.last) ?? .absent
464 }
465 return v
466 case "reverse":
467 switch v { case .list(let l):
468 return .list(Array(l.reversed()))
469 case .string(let t):
470 return .string(String(t.reversed()))
471 default:
472 return v }
473 case "items":
474 if case .map(let m) = v {
475 return .list(m.keys.sorted().map { .map(["key": .string($0), "value": m[$0]!]) })
476 }
477 return v
478 case "sort":
479 guard case .list(let l) = v else {
480 return v
481 }
482 func key(_ item: Value) -> Value {
483 if f.args.isEmpty {
484 return item
485 }
486 if case .map(let m) = item {
487 return m[s(0)] ?? .absent
488 }
489 return .absent
490 }
491 return .list(l.enumerated().sorted { let order = sortCompare(key($0.element), key($1.element))
492 return order == 0 ? $0.offset < $1.offset : order < 0
493 }.map { $0.element })
494 case "pluck":
495 if case .list(let l) = v {
496 return .list(l.map {
497 if case .map(let m) = $0 {
498 return m[s(0)] ?? .absent
499 }
500 return .absent
501 })
502 }
503 return v
504 case "sum", "min", "max":
505 if case .list(let l) = v {
506 let nums = try l.map { try number($0, name) }
507 if name == "sum" {
508 return .integer(nums.reduce(0, +))
509 }
510 return (name == "min" ? nums.min() : nums.max()).map(Value.integer) ?? .absent
511 }
512 return .integer(try number(v, name))
513 case "add", "sub", "mul", "div", "mod", "gt", "lt":
514 let a = try number(v, name), b = n()
515 func floorQuotient() -> Int {
516 let q = a / b
517 return a % b != 0 && (a < 0) != (b < 0) ? q - 1 : q
518 }
519 switch name { case "add":
520 return .integer(a + b)
521 case "sub":
522 return .integer(a - b)
523 case "mul":
524 return .integer(a * b)
525 case "div":
526 return .integer(floorQuotient())
527 case "mod":
528 return .integer(a - b * floorQuotient())
529 case "gt":
530 return .boolean(a > b)
531 default:
532 return .boolean(a < b) }
533 case "comma":
534 let a = try number(v, name), digits = Array(String(abs(a)))
535 var text = a < 0 ? "-" : ""
536 for (i, c) in digits.enumerated() {
537 if i > 0 && (digits.count - i) % 3 == 0 {
538 text += ","
539 }
540 text.append(c)
541 }
542 return .string(text)
543 case "join":
544 if case .list(let l) = v {
545 return .string(try l.map(stringify).joined(separator: s(0)))
546 }
547 return .string(try stringify(v))
548 default:
549 let text = try stringify(v)
550 switch name {
551 case "upper":
552 return .string(text.uppercased())
553 case "lower":
554 return .string(text.lowercased())
555 case "trim":
556 return .string(trim(text))
557 case "title":
558 var first = true, result = ""
559 for c in text {
560 result += first ? String(c).uppercased() : String(c).lowercased()
561 first = c == " "
562 }
563 return .string(result)
564 case "pad", "lpad":
565 let padding = String(repeating: f.args.count == 2 ? s(1) : " ", count: max(0, n() - text.utf8.count))
566 return .string(name == "pad" ? text + padding : padding + text)
567 case "truncate":
568 return .string(text.utf8.count > n() ? cut(text, 0, n()) + "..." : text)
569 default:
570 return .string(text.replacingOccurrences(of: s(0), with: s(1)))
571 }
572 }

575struct SourceLine { let line: Int
576 let text: String
577}
578struct Tag {
579 var kind: Character
580 var text: String
581 var expr: Expression? = nil
582 var line: Int
583 var indent = ""
584 var end = 0
585}
586func analyze(_ source: [SourceLine]) throws -> [Tag] {
587 var open = "{{", close = "}}", stack: [(String, Int)] = [], tokens: [Tag] = []
588 for row in source {
589 let text = row.text, line = row.line
590 var at = 0, tags: [Tag] = []
591 while let start = find(text, open, at) {
592 tags.append(Tag(kind: "t", text: cut(text, at, start), line: line))
593 var begin = start + open.utf8.count
594 let triple = cut(text, begin).hasPrefix("{")
595 if triple {
596 begin += 1
597 }
598 let ending = triple ? "}" + close : close
599 guard let finish = find(text, ending, begin) else {
600 throw Problem(line: line, reason: "unclosed tag")
601 }
602 let content = trim(cut(text, begin, finish))
603 at = finish + ending.utf8.count
604 let kind: Character, arg: String
605 if triple {
606 kind = "&"
607 arg = content
608 } else if let first = content.first, "#^/!>&=".contains(first) {
609 kind = first
610 arg = trim(cut(content, 1))
611 } else {
612 kind = "v"
613 arg = content
614 }
615 var tag = Tag(kind: kind, text: arg, line: line)
616 do {
617 switch kind {
618 case "v", "&", "#", "^":
619 let expr = try expression(arg)
620 tag.expr = expr
621 if kind == "#" || kind == "^" {
622 stack.append((expr.name, line))
623 }
624 case "/":
625 guard validName(arg) else {
626 throw Problem(reason: "invalid name '\(arg)'")
627 }
628 guard let top = stack.last else {
629 throw Problem(reason: "unexpected closing '\(arg)'")
630 }
631 guard top.0 == arg
632 else {
633 throw Problem(reason: "mismatched closing '\(arg)', expected '\(top.0)'")
634 }
635 stack.removeLast()
636 case ">":
637 if !identifier(arg) {
638 throw Problem(reason: "invalid name '\(arg)'")
639 }
640 case "=":
641 guard content.utf8.count >= 2 && content.hasSuffix("=")
642 else {
643 throw Problem(reason: "invalid delimiters")
644 }
645 let pair = cut(content, 1, content.utf8.count - 1).split(separator: " ").map(String.init)
646 guard pair.count == 2 && pair.allSatisfy({ !$0.contains("=") })
647 else {
648 throw Problem(reason: "invalid delimiters")
649 }
650 open = pair[0]
651 close = pair[1]
652 default:
653 break
654 }
655 } catch let error as Problem { throw Problem(line: line, reason: error.reason) }
656 tags.append(tag)
657 }
658 tags.append(Tag(kind: "t", text: cut(text, at), line: line))
659 let positions = tags.indices.filter { tags[$0].kind != "t" }
660 if positions.count == 1 && "#^/!>=".contains(tags[positions[0]].kind) && tags.filter({ $0.kind == "t" })
661 .allSatisfy({ $0.text.allSatisfy { $0 == " " } })
662 {
663 let i = positions[0]
664 var tag = tags[i]
665 if tag.kind == ">" {
666 tag.indent = tags[..<i].map(\.text).joined()
667 }
668 tokens.append(tag)
669 } else {
670 tokens += tags
671 tokens.append(Tag(kind: "t", text: "\n", line: line))
672 }
673 }
674 if let top = stack.last {
675 throw Problem(line: top.1, reason: "unclosed section '\(top.0)'")
676 }
677 var openings: [Int] = []
678 for i in tokens.indices {
679 if tokens[i].kind == "#" || tokens[i].kind == "^" {
680 openings.append(i)
681 }
682 if tokens[i].kind == "/" {
683 let start = openings.removeLast()
684 tokens[start].end = i
685 }
686 }
687 return tokens
688}
689func resolve(_ name: String, _ contexts: [Value], _ loops: [(Int, Int)]) -> Value? {
690 if name == "." {
691 return contexts.last
692 }
693 if name.hasPrefix("@") {
694 guard let (i, n) = loops.last else {
695 return nil
696 }
697 switch name { case "@index":
698 return .integer(i)
699 case "@number":
700 return .integer(i + 1)
701 case "@first":
702 return .boolean(i == 0)
703 default:
704 return .boolean(i == n - 1) }
705 }
706 let parts: [Part] = name.components(separatedBy: ".")
707 .map { subscriptName($0) ? .index(Int($0) ?? Int.max) : .key($0) }
708 for context in contexts.reversed() {
709 if var value = child(context, parts[0]) {
710 for part in parts.dropFirst() {
711 guard let next = child(value, part)
712 else {
713 return nil
714 }
715 value = next
716 }
717 return value
718 }
719 }
720 return nil
721}
722func escape(_ s: String) -> String {
723 s.map { c in switch c { case "&":
724 return "&amp;"
725 case "<":
726 return "&lt;"
727 case ">":
728 return "&gt;"
729 case "\"":
730 return "&quot;"
731 case "'":
732 return "&#39;"
733 default:
734 return String(c) } }.joined()
735}
736func render(
737 _ tags: [Tag],
738 _ start: Int,
739 _ end: Int,
740 _ contexts: [Value],
741 _ loops: [(Int, Int)],
742 _ partials: [String: [SourceLine]],
743 _ strict: Bool,
744 _ depth: Int
745) throws -> String {
746 var result = "", at = start
747 while at < end {
748 let tag = tags[at]
749 do {
750 switch tag.kind {
751 case "t":
752 result += tag.text
753 case "v", "&", "#", "^":
754 let expr = tag.expr!, found = resolve(expr.name, contexts, loops)
755 if strict && (tag.kind == "v" || tag.kind == "&") && found ==
756 nil
757 {
758 throw Problem(reason: "missing '\(expr.name)'")
759 }
760 var value = found ?? .absent
761 for filter in expr.filters {
762 value = try applyFilter(value, filter)
763 }
764 if tag.kind == "v" || tag.kind == "&" {
765 let text = try stringify(value)
766 result += tag.kind == "v" ? escape(text) : text
767 } else {
768 if tag.kind == "^" {
769 if !truthy(value) {
770 result += try render(
771 tags,
772 at + 1,
773 tag.end,
774 contexts,
775 loops,
776 partials,
777 strict,
778 depth
779 )
780 }
781 } else if case .list(let list) = value {
782 for (i, item) in list.enumerated() {
783 result += try render(
784 tags,
785 at + 1,
786 tag.end,
787 contexts + [item],
788 loops + [(i, list.count)],
789 partials,
790 strict,
791 depth
792 )
793 }
794 } else if truthy(value) {
795 result += try render(
796 tags,
797 at + 1,
798 tag.end,
799 contexts + [value],
800 loops,
801 partials,
802 strict,
803 depth
804 )
805 }
806 at = tag.end
807 }
808 case ">":
809 guard let source = partials[tag.text] else {
810 throw Problem(reason: "unknown partial '\(tag.text)'")
811 }
812 guard depth < 8 else {
813 throw Problem(reason: "partial depth exceeded at '\(tag.text)'")
814 }
815 let lines = source.map { SourceLine(line: $0.line, text: $0.text.isEmpty ? "" : tag.indent + $0.text) }
816 let parsed = try analyze(lines)
817 result += try render(parsed, 0, parsed.count, contexts, loops, partials, strict, depth + 1)
818 default:
819 break
820 }
821 } catch let error as Problem {
822 throw Problem(line: error.line == 0 ? tag.line : error.line, reason: error.reason)
823 }
824 at += 1
825 }
826 return result
827}
828func dump(_ v: Value, _ path: String, _ leaves: inout [String]) {
829 switch v {
830 case .map(let map) where !map.isEmpty:
831 for key in map.keys.sorted() {
832 dump(map[key]!, path.isEmpty ? key : path + "." + key, &leaves)
833 }
834 case .list(let list) where !list.isEmpty:
835 for (i, value) in list.enumerated() {
836 dump(value, "\(path)[\(i)]", &leaves)
837 }
838 default:
839 guard !path.isEmpty else {
840 return
841 }
842 let notation: String
843 switch v {
844 case .string(let s):
845 notation = "\"" + s.replacingOccurrences(of: "\\", with: "\\\\").replacingOccurrences(
846 of: "\"",
847 with: "\\\""
848 ).replacingOccurrences(of: "\n", with: "\\n") + "\""
849 case .list:
850 notation = "[]"
851 case .map:
852 notation = "{}"
853 default:
854 notation = try! stringify(v)
855 }
856 leaves.append(path + " = " + notation)
857 }
858}
859func outputTemplate(
860 _ name: String,
861 _ source: [SourceLine],
862 _ root: Value,
863 _ partials: [String: [SourceLine]],
864 _ strict: Bool
865) {
866 do {
867 let tags = try analyze(source), text = try render(tags, 0, tags.count, [root], [], partials, strict, 0)
868 let count = text.isEmpty ? 0 : text.filter { $0 == "\n" }.count + (text.hasSuffix("\n") ? 0 : 1)
869 print("TEMPLATE \(name) OK lines=\(count) chars=\(text.utf8.count)")
870 if !text.isEmpty {
871 print(text, terminator: text.hasSuffix("\n") ? "" : "\n")
872 }
873 } catch let error as Problem { print("TEMPLATE \(name) ERROR line \(error.line): \(error.reason)") }
874 catch { fatalError("unexpected error") }
875}
876
877var lines: [String] = []
878while let line = readLine() {
879 lines.append(line)
880}
881var at = 0, root = Value.map([:]), partials: [String: [SourceLine]] = [:], templates: [String: [SourceLine]] = [:],
882 strict = false
883while at < lines.count {
884 let command = lines[at], number = at + 1
885 at += 1
886 if command == "END" {
887 break
888 }
889 if command == "DATA" {
890 var set = 0, unset = 0, errors: [String] = []
891 while lines[at] != "ENDDATA" {
892 let line = lines[at], lineNumber = at + 1
893 at += 1
894 if line.isEmpty || line.hasPrefix("#") {
895 continue
896 }
897 do {
898 let deletion = line.hasPrefix("UNSET "), key: String, notation: String?
899 if deletion {
900 key = cut(line, 6)
901 notation = nil
902 } else {
903 guard let pos = find(line, " = ") else {
904 throw Problem(reason: "bad line")
905 }
906 key = cut(line, 0, pos)
907 notation = cut(line, pos + 3)
908 }
909 guard let parts = keyParts(key) else {
910 throw Problem(reason: "bad key '\(key)'")
911 }
912 if let notation {
913 guard let value = parseValue(notation) else {
914 throw Problem(reason: "bad value '\(notation)'")
915 }
916 root = try assigning(root, parts[...], value)
917 set += 1
918 } else {
919 guard let value = removing(root, parts[...]) else {
920 throw Problem(reason: "no value at '\(key)'")
921 }
922 root = value
923 unset += 1
924 }
925 } catch let error as Problem { errors.append("ERROR line \(lineNumber): \(error.reason)") }
926 catch { fatalError("unexpected error") }
927 }
928 at += 1
929 print("DATA set=\(set) unset=\(unset) errors=\(errors.count)")
930 for error in errors {
931 print(error)
932 }
933 } else if command.hasPrefix("PARTIAL ") || command.hasPrefix("TEMPLATE ") {
934 let partial = command.hasPrefix("PARTIAL "), name = command.components(separatedBy: " ")[1]
935 let end = partial ? "ENDPARTIAL" : "ENDTEMPLATE"
936 var source: [SourceLine] = []
937 while lines[at] != end {
938 source.append(SourceLine(line: at + 1, text: lines[at]))
939 at += 1
940 }
941 at += 1
942 if partial {
943 do {
944 _ = try analyze(source)
945 print("PARTIAL \(name) \(partials[name] == nil ? "DEFINED" : "REPLACED") lines=\(source.count)")
946 partials[name] = source
947 } catch let error as Problem { print("PARTIAL \(name) ERROR line \(error.line): \(error.reason)") }
948 catch { fatalError("unexpected error") }
949 } else {
950 templates[name] = source
951 outputTemplate(name, source, root, partials, strict)
952 }
953 } else if command.hasPrefix("RENDER ") {
954 let name = cut(command, 7)
955 if let source = templates[name] {
956 outputTemplate(name, source, root, partials, strict)
957 } else {
958 print("TEMPLATE \(name) ERROR line \(number): unknown template '\(name)'")
959 }
960 } else if command == "DUMP" {
961 var leaves: [String] = []
962 dump(root, "", &leaves)
963 print("DUMP leaves=\(leaves.count)")
964 for leaf in leaves {
965 print(leaf)
966 }
967 } else if command == "RESET" {
968 root = .map([:])
969 partials.removeAll()
970 templates.removeAll()
971 print("RESET")
972 } else {
973 strict = command == "STRICT ON"
974 print(command)
975 }
976}
