81local function key_parts(key)
82 local parts, pos = {}, 1
83 while pos <= #key do
84 local p
85 if pos > 1 and key:sub(pos, pos) == "[" then
86 local close = key:find("]", pos + 1, true)
87 if not close or not subscript(key:sub(pos + 1, close - 1)) then
88 bad("bad key '" .. key .. "'")
89 end
90 p = tonumber(key:sub(pos + 1, close - 1))
91 pos = close + 1
92 else
93 if pos > 1 then
94 if key:sub(pos, pos) ~= "." then
95 bad("bad key '" .. key .. "'")
96 end
97 pos = pos + 1
98 end
99 local name = key:match("^([A-Za-z_][A-Za-z0-9_]*)", pos)
100 if not name then
101 bad("bad key '" .. key .. "'")
102 end
103 p, pos = name, pos + #name
104 end
105 parts[#parts + 1] = { part = p, prefix = key:sub(1, pos - 1) }
106 end
107 if #parts == 0 then
108 bad("bad key '" .. key .. "'")
109 end
110 return parts
111end

113local function scalar(s, pos)
114 if s:sub(pos, pos) == '"' then
115 local chars = {}
116 pos = pos + 1
117 while pos <= #s do
118 local c = s:sub(pos, pos)
119 pos = pos + 1
120 if c == '"' then
121 return table.concat(chars), pos
122 end
123 if c == "\\" then
124 local escapes = { n = "\n", ['"'] = '"', ["\\"] = "\\" }
125 c = escapes[s:sub(pos, pos)]
126 if not c then
127 bad("invalid")
128 end
129 pos = pos + 1
130 end
131 chars[#chars + 1] = c
132 end
133 bad("invalid")
134 end
135 local stop = pos
136 while stop <= #s and s:sub(stop, stop) ~= "," and s:sub(stop, stop) ~= "]" do
137 stop = stop + 1
138 end
139 local word = s:sub(pos, stop - 1)
140 if word == "true" then
141 return true, stop
142 end
143 if word == "false" then
144 return false, stop
145 end
146 if word == "0" or word:match("^%-?[1-9][0-9]*$") then
147 return tonumber(word), stop
148 end
149 bad("invalid")
150end

152local function parse_value(s)
153 local ok, value = pcall(function ()
154 if s == "[]" then
155 return list()
156 end
157 if s == "{}" then
158 return map()
159 end
160 if starts(s, "[") then
161 local items, pos = {}, 2
162 while true do
163 local v
164 v, pos = scalar(s, pos)
165 items[#items + 1] = v
166 if s:sub(pos) == "]" then
167 return list(items)
168 end
169 if s:sub(pos, pos + 1) ~= ", " then
170 bad("invalid")
171 end
172 pos = pos + 2
173 end
174 end
175 local v, stop = scalar(s, 1)
176 if stop ~= #s + 1 then
177 bad("invalid")
178 end
179 return v
180 end)
181 if not ok then
182 bad("bad value '" .. s .. "'")
183 end
184 return value
185end

187local function get(v, p)
188 if type(p) == "number" and kind(v) == "list" and p < #v.items then
189 return v.items[p + 1], true
190 end
191 if type(p) == "string" and kind(v) == "map" and v.items[p] ~= nil then
192 return v.items[p], true
193 end
194 return ABSENT, false
195end

197local function change(v, parts, at, new_value, remove, key)
198 local p, prefix = parts[at].part, parts[at].prefix
199 local child, exists = get(v, p)
200 if remove and not exists then
201 bad("no value at '" .. key .. "'")
202 end
203 if not remove and type(p) == "number" and p > #v.items then
204 bad("index gap at '" .. prefix .. "'")
205 end
206 if at < #parts then
207 if not remove then
208 local required = type(parts[at + 1].part) == "number" and "list" or "map"
209 if not exists then
210 child = required == "list" and list() or map()
211 elseif kind(child) ~= required then
212 bad("type conflict at '" .. prefix .. "'")
213 end
214 end
215 new_value = change(child, parts, at + 1, new_value, remove, key)
216 end
217 local out = {}
218 for k, x in pairs(v.items) do
219 out[k] = x
220 end
221 if kind(v) == "list" then
222 if at == #parts and remove then
223 table.remove(out, p + 1)
224 else
225 out[p + 1] = new_value
226 end
227 return list(out)
228 end
229 if at == #parts and remove then
230 out[p] = nil
231 else
232 out[p] = new_value
233 end
234 return map(out)
235end

237local function split_quoted(s, separator)
238 local out, start, quoted = {}, 1, false
239 for i = 1, #s do
240 local c = s:sub(i, i)
241 if c == "'" then
242 quoted = not quoted
243 elseif c == separator and not quoted then
244 out[#out + 1] = s:sub(start, i - 1)
245 start = i + 1
246 end
247 end
248 out[#out + 1] = s:sub(start)
249 return out
250end
251local FILTERS = {}
252for name in("upper lower title trim length first last reverse items comma sum min max default eq pad lpad truncate replace join sort pluck add sub mul div mod gt lt"):gmatch("%S+") do
253 FILTERS[name] = true
254end

256local function parse_expression(s, number)
257 local pieces = split_quoted(s, "|")
258 local expression = { name = trim(pieces[1]), filters = {} }
259 if not valid_name(expression.name) then
260 fail(number, "invalid name '" .. expression.name .. "'")
261 end
262 for i = 2, #pieces do
263 local fields = split_quoted(trim(pieces[i]), ":")
264 local name, args, valid = fields[1], {}, true
265 if not FILTERS[name] then
266 fail(number, "unknown filter '" .. name .. "'")
267 end
268 for j = 2, #fields do
269 local a = fields[j]
270 if #a >= 2 and a:sub(1, 1) == "'" and a:sub(- 1) == "'" and not a:sub(2, - 2):find("'", 1, true) then
271 args[#args + 1] = a:sub(2, - 2)
272 elseif numeric_text(a) then
273 args[#args + 1] = tonumber(a)
274 else
275 valid = false
276 end
277 end
278 local n = #args
279 if name == "default" or name == "eq" then
280 valid = valid and n == 1
281 elseif name == "pad" or name == "lpad" then
282 valid = valid and (n == 1 or n == 2) and type(args[1]) == "number" and args[1] >= 0 and (n == 1 or type(args[2]) == "string" and #args[2] == 1)
283 elseif name == "truncate" then
284 valid = valid and n == 1 and type(args[1]) == "number" and args[1] >= 0
285 elseif name == "replace" then
286 valid = valid and n == 2 and type(args[1]) == "string" and args[1] ~= "" and type(args[2]) == "string"
287 elseif name == "join" or name == "pluck" then
288 valid = valid and n == 1 and type(args[1]) == "string"
289 elseif name == "sort" then
290 valid = valid and (n == 0 or n == 1 and type(args[1]) == "string")
291 elseif name == "div" or name == "mod" then
292 valid = valid and n == 1 and type(args[1]) == "number" and args[1] ~= 0
293 elseif name == "add" or name == "sub" or name == "mul" or name == "gt" or name == "lt" then
294 valid = valid and n == 1 and type(args[1]) == "number"
295 else
296 valid = valid and n == 0
297 end
298 if not valid then
299 fail(number, "bad arguments for '" .. name .. "'")
300 end
301 expression.filters[#expression.filters + 1] = { name = name, args = args }
302 end
303 return expression
304end
305local STANDALONE = { ["#"] = true, ["^"] = true, ["/"] = true, ["!"] = true, [">"] = true, ["="] = true }

307local function parse_source(lines)
308 local root, opens = {}, {}
309 local targets = { root }
310 local opening, closing = "{{", "}}"
311 for _, line in ipairs(lines) do
312 local tokens, pos, count, spaces, only_tag = {}, 1, 0, true, nil
313 while true do
314 local start = line.text:find(opening, pos, true)
315 local text = line.text:sub(pos, start and start - 1 or #line.text)
316 tokens[#tokens + 1] = text
317 spaces = spaces and trim(text) == ""
318 if not start then
319 break
320 end
321 local after, close = start + #opening, closing
322 local triple = line.text:sub(after, after) == "{"
323 if triple then
324 after = after + 1
325 close = "}" .. closing
326 end
327 local stop = line.text:find(close, after, true)
328 if not stop then
329 fail(line.number, "unclosed tag")
330 end
331 pos = stop + #close
332 local content = trim(line.text:sub(after, stop - 1))
333 local tag = { kind = "v", text = content, number = line.number, children = {}, indent = "" }
334 if triple then
335 tag.kind = "&"
336 elseif #content > 0 and ("#^/!>&="):find(content:sub(1, 1), 1, true) then
337 tag.kind = content:sub(1, 1)
338 if tag.kind ~= "=" then
339 tag.text = trim(content:sub(2))
340 end
341 end
342 if tag.kind == "=" then
343 local ds = {}
344 if #content >= 2 and content:sub(- 1) == "=" then
345 for x in content:sub(2, - 2):gmatch("[^ ]+") do
346 ds[#ds + 1] = x
347 end
348 end
349 if #ds ~= 2 or ds[1]:find("=", 1, true) or ds[2]:find("=", 1, true) then
350 fail(line.number, "invalid delimiters")
351 end
352 opening, closing = ds[1], ds[2]
353 elseif tag.kind == "v" or tag.kind == "&" or tag.kind == "#" or tag.kind == "^" then
354 tag.expression = parse_expression(tag.text, line.number)
355 if tag.kind == "#" or tag.kind == "^" then
356 opens[#opens + 1] = tag
357 end
358 elseif tag.kind == "/" then
359 if not valid_name(tag.text) then
360 fail(line.number, "invalid name '" .. tag.text .. "'")
361 end
362 if #opens == 0 then
363 fail(line.number, "unexpected closing '" .. tag.text .. "'")
364 end
365 local expected = opens[#opens].expression.name
366 if tag.text ~= expected then
367 fail(line.number, "mismatched closing '" .. tag.text .. "', expected '" .. expected .. "'")
368 end
369 opens[#opens] = nil
370 elseif tag.kind == ">" and not identifier(tag.text) then
371 fail(line.number, "invalid name '" .. tag.text .. "'")
372 end
373 tokens[#tokens + 1] = tag
374 count, only_tag = count + 1, tag
375 end
376 if count == 1 and spaces and STANDALONE[only_tag.kind] then
377 if only_tag.kind == ">" then
378 only_tag.indent = tokens[1]
379 end
380 tokens = { only_tag }
381 else
382 tokens[#tokens + 1] = "\n"
383 end
384 for _, token in ipairs(tokens) do
385 local target = targets[#targets]
386 if type(token) == "string" then
387 target[#target + 1] = token
388 elseif token.kind == "#" or token.kind == "^" then
389 target[#target + 1] = token
390 targets[#targets + 1] = token.children
391 elseif token.kind == "/" then
392 targets[#targets] = nil
393 elseif token.kind ~= "!" and token.kind ~= "=" then
394 target[#target + 1] = token
395 end
396 end
397 end
398 if #opens > 0 then
399 local tag = opens[#opens]
400 fail(tag.number, "unclosed section '" .. tag.expression.name .. "'")
401 end
402 return root
403end

405local function truth(v)
406 local k = kind(v)
407 if k == "absent" then
408 return false
409 end
410 if k == "boolean" then
411 return v
412 end
413 if k == "number" then
414 return v ~= 0
415 end
416 if k == "string" then
417 return v ~= ""
418 end
419 return next(v.items) ~= nil
420end
421local stringify
422stringify = function (v)
423 local k = kind(v)
424 if k == "absent" then
425 return ""
426 end
427 if k == "string" then
428 return v
429 end
430 if k == "number" then
431 return string.format("%.0f", v)
432 end
433 if k == "boolean" then
434 return v and "true" or "false"
435 end
436 if k == "map" then
437 bad("cannot render map")
438 end
439 local out = {}
440 for i, x in ipairs(v.items) do
441 out[i] = stringify(x)
442 end
443 return table.concat(out, ", ")
444end

459local function field(v, key)
460 if kind(v) == "map" and v.items[key] ~= nil then
461 return v.items[key]

463 return ABSENT

465local RANK = { absent = 0, number = 1, string = 2, boolean = 3, list = 4, map = 4 }

467local function less(a, b)
468 local ka, kb = kind(a), kind(b)
469 if RANK[ka] ~= RANK[kb] then
470 return RANK[ka] < RANK[kb]
471 end
472 if ka == "number" or ka == "string" then
473 return a < b
474 end
475 if ka == "boolean" then
476 return not a and b
477 end
478 return false
479end

481local function apply(f, v)
482 local name, args, k = f.name, f.args, kind(v)
483 if name == "default" then
484 if not truth(v) then
485 return args[1]
486 end
487 return v
488 elseif name == "length" then
489 if k == "list" then
490 return #v.items
491 elseif k == "map" then
492 return #keys(v.items)
493 end
494 elseif name == "first" or name == "last" then
495 if k ~= "list" then
496 return v
497 end
498 if #v.items == 0 then
499 return ABSENT
500 end
501 return v.items[name == "first" and 1 or #v.items]
502 elseif name == "reverse" then
503 if k == "string" then
504 return v:reverse()
505 end
506 if k ~= "list" then
507 return v
508 end
509 local out = {}
510 for i = #v.items, 1, - 1 do
511 out[#out + 1] = v.items[i]
512 end
513 return list(out)
514 elseif name == "items" then
515 if k ~= "map" then
516 return v
517 end
518 local out = {}
519 for _, key in ipairs(keys(v.items)) do
520 out[#out + 1] = map({ key = key, value = v.items[key] })
521 end
522 return list(out)
523 elseif name == "sort" or name == "pluck" then
524 if k ~= "list" then
525 return v
526 end
527 local out = {}
528 if name == "pluck" then
529 for i, x in ipairs(v.items) do
530 out[i] = field(x, args[1])
531 end
532 else
533 local entries = {}
534 for i, x in ipairs(v.items) do
535 local sort_key = x
536 if #args > 0 then
537 sort_key = field(x, args[1])
538 end
539 entries[i] = { value = x, key = sort_key, index = i }
540 end
541 table.sort(entries, function (a, b)
542 if less(a.key, b.key) then
543 return true
544 end
545 if less(b.key, a.key) then
546 return false
547 end
548 return a.index < b.index
549 end)
550 for i, entry in ipairs(entries) do
551 out[i] = entry.value
552 end
553 end
554 return list(out)
555 elseif name == "sum" or name == "min" or name == "max" then
556 if k ~= "list" then
557 return number(v, name)
558 end
559 if #v.items == 0 and name ~= "sum" then
560 return ABSENT
561 end
562 local result = 0
563 for i, x in ipairs(v.items) do
564 local n = number(x, name)
565 if name == "sum" then
566 result = result + n
567 elseif i == 1 or name == "min" and n < result or name == "max" and n > result then
568 result = n
569 end
570 end
571 return result
572 elseif name == "comma" or name == "add" or name == "sub" or name == "mul" or name == "div" or name == "mod" or name == "gt" or name == "lt" then
573 local n = number(v, name)
574 if name == "add" then
575 return n + args[1]
576 elseif name == "sub" then
577 return n - args[1]
578 elseif name == "mul" then
579 return n * args[1]
580 elseif name == "div" then
581 return math.floor(n / args[1])
582 elseif name == "mod" then
583 return n - args[1] * math.floor(n / args[1])
584 elseif name == "gt" then
585 return n > args[1]
586 elseif name == "lt" then
587 return n < args[1]
588 end
589 local text, sign = string.format("%.0f", math.abs(n)), n < 0 and "-" or ""
590 text = text:reverse():gsub("(%d%d%d)", "%1,"):reverse():gsub("^,", "")
591 return sign .. text
592 elseif name == "join" and k == "list" then
593 local out = {}
594 for i, x in ipairs(v.items) do
595 out[i] = stringify(x)
596 end
597 return table.concat(out, args[1])
598 end
599 local s = stringify(v)
600 if name == "upper" then
601 return s:upper()
602 elseif name == "lower" then
603 return s:lower()
604 elseif name == "title" then
605 local out = {}
606 for i = 1, #s do
607 local c = s:sub(i, i)
608 out[i] =(i == 1 or s:sub(i - 1, i - 1) == " ") and c:upper() or c:lower()
609 end
610 return table.concat(out)
611 elseif name == "trim" then
612 return trim(s)
613 elseif name == "length" then
614 return #s
615 elseif name == "eq" then
616 return s == stringify(args[1])
617 elseif name == "pad" or name == "lpad" then
618 local fill =(args[2] or " "):rep(math.max(0, args[1] - #s))
619 return name == "pad" and s .. fill or fill .. s
620 elseif name == "truncate" then
621 return #s > args[1] and s:sub(1, args[1]) .. "..." or s
622 elseif name == "replace" then
623 local out, at = {}, 1
624 while true do
625 local p = s:find(args[1], at, true)
626 if not p then
627 out[#out + 1] = s:sub(at)
628 break
629 end
630 out[#out + 1] = s:sub(at, p - 1)
631 out[#out + 1] = args[2]
632 at = p + #args[1]
633 end
634 return table.concat(out)
635 elseif name == "join" then
636 return s
637 end
638 return v
639end
640local ESCAPES = { ["&"] = "&amp;", ["<"] = "&lt;", [">"] = "&gt;", ['"'] = "&quot;", ["'"] = "&#39;" }

642local function render(nodes, root, partials, strict)
643 local contexts, loops, out = { root }, {}, {}
644 local function resolve(name)
645 if name == "." then
646 return contexts[#contexts], true
647 end
648 if starts(name, "@") then
649 local loop = loops[#loops]
650 if not loop then
651 return ABSENT, false
652 end
653 if name == "@index" then
654 return loop.index, true
655 elseif name == "@number" then
656 return loop.index + 1, true
657 elseif name == "@first" then
658 return loop.index == 0, true
659 else
660 return loop.index == loop.count - 1, true
661 end
662 end
663 local parts = split(name, ".")
664 for i, p in ipairs(parts) do
665 if subscript(p) then
666 parts[i] = tonumber(p)
667 end
668 end
669 for i = #contexts, 1, - 1 do
670 local _, exists = get(contexts[i], parts[1])
671 if exists then
672 local v = contexts[i]
673 for _, p in ipairs(parts) do
674 v, exists = get(v, p)
675 if not exists then
676 return ABSENT, false
677 end
678 end
679 return v, true
680 end
681 end
682 return ABSENT, false
683 end
684 local function evaluate(expression, check)
685 local v, found = resolve(expression.name)
686 if check and not found then
687 bad("missing '" .. expression.name .. "'")
688 end
689 for _, f in ipairs(expression.filters) do
690 v = apply(f, v)
691 end
692 return v
693 end
694 local visit
695 visit = function (content, depth)
696 for _, tag in ipairs(content) do
697 if type(tag) == "string" then
698 out[#out + 1] = tag
699 else
700 local ok, err = pcall(function ()
701 if tag.kind == "v" or tag.kind == "&" then
702 local s = stringify(evaluate(tag.expression, strict))
703 out[#out + 1] = tag.kind == "v" and s:gsub("[&<>\"']", ESCAPES) or s
704 elseif tag.kind == "#" or tag.kind == "^" then
705 local v = evaluate(tag.expression, false)
706 if tag.kind == "^" then
707 if not truth(v) then
708 visit(tag.children, depth)
709 end
710 elseif kind(v) == "list" then
711 for i, x in ipairs(v.items) do
712 contexts[#contexts + 1] = x
713 loops[#loops + 1] = { index = i - 1, count = #v.items }
714 visit(tag.children, depth)
715 contexts[#contexts], loops[#loops] = nil, nil
716 end
717 elseif truth(v) then
718 contexts[#contexts + 1] = v
719 visit(tag.children, depth)
720 contexts[#contexts] = nil
721 end
722 elseif tag.kind == ">" then
723 local body = partials[tag.text]
724 if not body then
725 bad("unknown partial '" .. tag.text .. "'")
726 end
727 if depth == 8 then
728 bad("partial depth exceeded at '" .. tag.text .. "'")
729 end
730 local source = {}
731 for i, line in ipairs(body) do
732 source[i] = { number = line.number, text = line.text ~= "" and tag.indent .. line.text or "" }
733 end
734 visit(parse_source(source), depth + 1)
735 end
736 end)
737 if not ok then
738 if type(err) == "table" then
739 error(err, 0)
740 end
741 fail(tag.number, err)
742 end
743 end
744 end
745 end
746 visit(nodes, 0)
747 return table.concat(out)
748end

757 return "{}"

762local function collect(v, path, out)
763 if kind(v) == "map" and next(v.items) then
764 for _, k in ipairs(keys(v.items)) do
765 collect(v.items[k], path == "" and k or path .. "." .. k, out)
766 end

768 for i, x in ipairs(v.items) do
769 collect(x, path .. "[" ..(i - 1) .. "]", out)

771 elseif path ~= "" then
772 out[#out + 1] = path .. " = " .. notation(v)

783 return render(parse_source(templates[name]), root, partials, strict)
784 end)
785 if not ok then

787 return

790 if result ~= "" and result:sub(- 1) ~= "\n" then
791 count = count + 1
792 end

810 pos = pos + 1

812 pos = pos + 1
