30 static bool Matches(string text, string pattern) => Regex.IsMatch(text, @"\A(?:" + pattern + @")\z");
31 static string Strip(string text) => text.Trim(' ');
32 static bool Has(object? value, object key) => key is string name
33 ? value is Dictionary<string, object?> map && map.ContainsKey(name)
34 : value is List<object?> list && (int)key >= 0 && (int)key < list.Count;
35 static object? Get(object? value, object key) => key is string name
36 ? ((Dictionary<string, object?>)value!)[name] : ((List<object?>)value!)[(int)key];
37 static void Put(object container, object key, object? value)
38 {
39 if (key is string name) ((Dictionary<string, object?>)container)[name] = value;
40 else
41 {
42 var list = (List<object?>)container;
43 if ((int)key == list.Count) list.Add(value); else list[(int)key] = value;
44 }
45 }
46 static object? Copy(object? value) => value switch
47 {
48 Dictionary<string, object?> map => map.ToDictionary(pair => pair.Key, pair => Copy(pair.Value), StringComparer.Ordinal),
49 List<object?> list => list.Select(Copy).ToList(),
50 _ => value
51 };
52 static List<KeyPart> ParseKey(string key)
53 {
54 if (!Matches(key, Identifier + @"(?:\." + Identifier + @"|\[" + Subscript + @"\])*"))
55 throw new Failure($"bad key '{key}'");
56 List<KeyPart> parts = [];
57 foreach (Match match in Regex.Matches(key, "(" + Identifier + @")|\[(" + Subscript + @")\]"))
58 parts.Add(new(match.Groups[1].Success ? match.Groups[1].Value : int.Parse(match.Groups[2].Value), key[..(match.Index + match.Length)]));
59 return parts;
60 }
61 class ValueReader(string text)
62 {
63 public int Position;
64 object Scalar()
65 {
66 if (Position < text.Length && text[Position] == '"')
67 {
68 Position++;
69 StringBuilder result = new();
70 while (Position < text.Length)
71 {
72 char c = text[Position++];
73 if (c == '"') return result.ToString();
74 if (c == '\\')
75 {
76 if (Position == text.Length) throw new Failure("");
77 c = text[Position++];
78 if (c == 'n') c = '\n';
79 else if (c is not ('"' or '\\')) throw new Failure("");
80 }
81 result.Append(c);
82 }
83 throw new Failure("");
84 }
85 int start = Position;
86 while (Position < text.Length && text[Position] is not (',' or ']')) Position++;
87 string word = text[start..Position];
88 if (word is "true" or "false") return word == "true";
89 if (!Matches(word, "0|-?[1-9][0-9]*")) throw new Failure("");
90 return int.Parse(word);
91 }
92 public object Read()
93 {
94 if (text == "{}") { Position = 2; return new Dictionary<string, object?>(StringComparer.Ordinal); }
95 if (text == "[]") { Position = 2; return new List<object?>(); }
96 if (!text.StartsWith('[')) return Scalar();
97 Position = 1;
98 List<object?> result = [];
99 while (true)
100 {
101 result.Add(Scalar());
102 if (text[Position..].StartsWith(", ", StringComparison.Ordinal)) Position += 2;
103 else if (text[Position..] == "]") { Position++; return result; }
104 else throw new Failure("");
105 }
106 }
107 }
108 static object ParseValue(string notation)
109 {
110 try
111 {
112 ValueReader reader = new(notation);
113 object result = reader.Read();
114 if (reader.Position != notation.Length) throw new Failure("");
115 return result;
116 }
117 catch (Exception error) when (error is Failure or OverflowException or FormatException)
118 { throw new Failure($"bad value '{notation}'"); }
119 }
120 static void Data(List<SourceLine> lines)
121 {
122 int sets = 0, unsets = 0;
123 List<string> errors = [];
124 foreach (var line in lines)
125 {
126 string text = line.Text;
127 if (text.Length == 0 || text.StartsWith('#')) continue;
128 try
129 {
130 bool remove = text.StartsWith("UNSET ", StringComparison.Ordinal);
131 int equal = text.IndexOf(" = ", StringComparison.Ordinal);
132 if (!remove && equal < 0) throw new Failure("bad line");
133 string key = remove ? text[6..] : text[..equal];
134 var parts = ParseKey(key);
135 object? item = remove ? null : ParseValue(text[(equal + 3)..]);
136 var candidate = (Dictionary<string, object?>)Copy(root)!;
137 object? container = candidate;
138 for (int i = 0; i < parts.Count; i++)
139 {
140 var part = parts[i];
141 bool found = Has(container, part.Key);
142 if (remove)
143 {
144 if (!found) throw new Failure($"no value at '{key}'");
145 if (i == parts.Count - 1)
146 {
147 if (part.Key is string name) ((Dictionary<string, object?>)container!).Remove(name);
148 else ((List<object?>)container!).RemoveAt((int)part.Key);
149 }
150 else container = Get(container, part.Key);
151 }
152 else
153 {
154 if (part.Key is int index && index > ((List<object?>)container!).Count)
155 throw new Failure($"index gap at '{part.Prefix}'");
156 if (i == parts.Count - 1) Put(container!, part.Key, item);
157 else
158 {
159 bool wantMap = parts[i + 1].Key is string;
160 if (found)
161 {
162 object? next = Get(container, part.Key);
163 if (wantMap ? next is not Dictionary<string, object?> : next is not List<object?>)
164 throw new Failure($"type conflict at '{part.Prefix}'");
165 container = next;
166 }
167 else
168 {
169 object next = wantMap ? new Dictionary<string, object?>(StringComparer.Ordinal) : new List<object?>();
170 Put(container!, part.Key, next);
171 container = next;
172 }
173 }
174 }
175 }
176 root = candidate;
177 if (remove) unsets++; else sets++;
178 }
179 catch (Failure error) { errors.Add($"ERROR line {line.Number}: {error.Message}"); }
180 }
181 Console.WriteLine($"DATA set={sets} unset={unsets} errors={errors.Count}");
182 errors.ForEach(Console.WriteLine);
183 }
184 static List<string> SplitUnquoted(string text, char separator)
185 {
186 List<string> result = [];
187 int start = 0;
188 bool quoted = false;
189 for (int i = 0; i < text.Length; i++)
190 if (text[i] == '\'') quoted = !quoted;
191 else if (text[i] == separator && !quoted) { result.Add(text[start..i]); start = i + 1; }
192 result.Add(text[start..]);
193 return result;
194 }
195 static bool ValidArguments(string name, List<object> args)
196 {
197 int count = args.Count;
198 bool integer = count > 0 && args[0] is int;
199 bool text = count > 0 && args[0] is string;
200 return name switch
201 {
202 "default" or "eq" => count == 1,
203 "pad" or "lpad" => count is 1 or 2 && integer && (int)args[0] >= 0 && (count == 1 || args[1] is string s && s.Length == 1),
204 "truncate" => count == 1 && integer && (int)args[0] >= 0,
205 "replace" => count == 2 && text && ((string)args[0]).Length > 0 && args[1] is string,
206 "join" or "pluck" => count == 1 && text,
207 "sort" => count == 0 || count == 1 && text,
208 "div" or "mod" => count == 1 && integer && (int)args[0] != 0,
209 "add" or "sub" or "mul" or "gt" or "lt" => count == 1 && integer,
210 _ => count == 0
211 };
212 }
213 static Expression ParseExpression(string text, int line)
214 {
215 var pieces = SplitUnquoted(text, '|');
216 string name = Strip(pieces[0]);
217 if (!Matches(name, NamePattern)) throw new Failure($"invalid name '{name}'", line);
218 List<Filter> filters = [];
219 foreach (string piece in pieces.Skip(1))
220 {
221 var fields = SplitUnquoted(Strip(piece), ':');
222 string filter = fields[0];
223 if (!Filters.Contains(filter)) throw new Failure($"unknown filter '{filter}'", line);
224 List<object> arguments = [];
225 bool valid = true;
226 foreach (string argument in fields.Skip(1))
227 if (Matches(argument, "'[^']*'")) arguments.Add(argument[1..^1]);
228 else if (Matches(argument, "-?[0-9]{1,9}")) arguments.Add(int.Parse(argument));
229 else valid = false;
230 if (!valid || !ValidArguments(filter, arguments)) throw new Failure($"bad arguments for '{filter}'", line);
231 filters.Add(new(filter, arguments));
232 }
233 return new(name, filters);
234 }
235 static List<Node> Analyze(List<SourceLine> source, string indentation)
236 {
237 List<Node> result = [];
238 Stack<Node> open = new();
239 Stack<List<Node>> levels = new();
240 levels.Push(result);
241 string opening = "{{", closing = "}}";
242 foreach (var line in source)
243 {
244 string text = line.Text.Length == 0 ? "" : indentation + line.Text;
245 List<Node> nodes = [];
246 int position = 0, tags = 0;
247 while (true)
248 {
249 int start = text.IndexOf(opening, position, StringComparison.Ordinal);
250 if (start < 0) { nodes.Add(new('T', line.Number, text[position..])); break; }
251 nodes.Add(new('T', line.Number, text[position..start]));
252 int begin = start + opening.Length;
253 bool triple = begin < text.Length && text[begin] == '{';
254 if (triple) begin++;
255 string endText = (triple ? "}" : "") + closing;
256 int end = text.IndexOf(endText, begin, StringComparison.Ordinal);
257 if (end < 0) throw new Failure("unclosed tag", line.Number);
258 string content = Strip(text[begin..end]);
259 char kind = triple ? 'R' : content.Length > 0 && "#^/!>&=".Contains(content[0]) ? content[0] : 'V';
260 string argument = kind == 'V' || triple ? content : Strip(content[1..]);
261 if (kind == '&') kind = 'R';
262 Node node = new(kind, line.Number, argument);
263 tags++;
264 if (kind == '=')
265 {
266 if (content.Length < 2 || !content.EndsWith('=')) throw new Failure("invalid delimiters", line.Number);
267 string[] pair = Strip(content[1..^1]).Split(' ', StringSplitOptions.RemoveEmptyEntries);
268 if (pair.Length != 2 || pair.Any(s => s.Contains('='))) throw new Failure("invalid delimiters", line.Number);
269 opening = pair[0]; closing = pair[1];
270 }
271 else if (kind is 'V' or 'R' or '#' or '^')
272 {
273 node.Expression = ParseExpression(argument, line.Number);
274 if (kind is '#' or '^') open.Push(node);
275 }
276 else if (kind == '/')
277 {
278 if (!Matches(argument, NamePattern)) throw new Failure($"invalid name '{argument}'", line.Number);
279 if (open.Count == 0) throw new Failure($"unexpected closing '{argument}'", line.Number);
280 string expected = open.Peek().Expression!.Name;
281 if (argument != expected) throw new Failure($"mismatched closing '{argument}', expected '{expected}'", line.Number);
282 open.Pop();
283 }
284 else if (kind == '>' && !Matches(argument, Identifier)) throw new Failure($"invalid name '{argument}'", line.Number);
285 nodes.Add(node);
286 position = end + endText.Length;
287 }
288 Node? only = nodes.FirstOrDefault(node => node.Kind != 'T');
289 bool standalone = tags == 1 && nodes.Where(node => node.Kind == 'T').All(node => node.Text.All(c => c == ' ')) && "#^/!>=".Contains(only!.Kind);
290 if (standalone && only!.Kind == '>') only.Indentation = nodes[0].Text;
291 if (!standalone) nodes.Add(new('T', line.Number, "\n"));
292 foreach (Node node in nodes)
293 {
294 if (standalone && node.Kind == 'T') continue;
295 if (node.Kind == '/') levels.Pop();
296 else
297 {
298 levels.Peek().Add(node);
299 if (node.Kind is '#' or '^') levels.Push(node.Children);
300 }
301 }
302 }
303 if (open.Count > 0) throw new Failure($"unclosed section '{open.Peek().Expression!.Name}'", open.Peek().Line);
304 return result;
305 }
306 static bool Truth(object? value) => value switch
307 {
308 null => false, bool b => b, int n => n != 0, string s => s.Length > 0,
309 List<object?> list => list.Count > 0, Dictionary<string, object?> map => map.Count > 0,
310 _ => throw new InvalidOperationException()
311 };
312 static string StringValue(object? value) => value switch
313 {
314 null => "", string s => s, bool b => b ? "true" : "false", int n => n.ToString(System.Globalization.CultureInfo.InvariantCulture),
315 List<object?> list => string.Join(", ", list.Select(StringValue)),
316 _ => throw new Failure("cannot render map")
317 };
318 static int Number(object? value, string filter) => value switch
319 {
320 null => 0, int n => n,
321 string s when Matches(s, "-?[0-9]{1,9}") => int.Parse(s),
322 _ => throw new Failure($"not a number for '{filter}'")
323 };
324 static int Rank(object? value) => value switch { null => 0, int => 1, string => 2, bool => 3, _ => 4 };
325 static object? SortKey(object? value, List<object> args) => args.Count == 0 ? value
326 : value is Dictionary<string, object?> map ? map.GetValueOrDefault((string)args[0]) : null;
327 static int Compare(object? left, object? right)
328 {
329 int order = Rank(left).CompareTo(Rank(right));
330 if (order != 0) return order;
331 return Rank(left) switch { 1 => ((int)left!).CompareTo((int)right!), 2 => StringComparer.Ordinal.Compare((string)left!, (string)right!), 3 => ((bool)left!).CompareTo((bool)right!), _ => 0 };
332 }
333 static object? Apply(object? value, Filter filter)
334 {
335 string name = filter.Name;
336 var args = filter.Arguments;
337 switch (name)
338 {
339 case "default": return Truth(value) ? value : args[0];
340 case "eq": return StringValue(value) == StringValue(args[0]);
341 case "length": return value is Dictionary<string, object?> map ? map.Count : value is List<object?> list ? list.Count : StringValue(value).Length;
342 case "first": case "last": return value is List<object?> items ? items.Count == 0 ? null : items[name == "first" ? 0 : items.Count - 1] : value;
343 case "reverse": return value is List<object?> reversed ? reversed.AsEnumerable().Reverse().ToList() : value is string s ? new string(s.Reverse().ToArray()) : value;
344 case "items":
345 if (value is not Dictionary<string, object?> entries) return value;
346 return entries.OrderBy(pair => pair.Key, StringComparer.Ordinal).Select(pair => (object?)new Dictionary<string, object?> { ["key"] = pair.Key, ["value"] = pair.Value }).ToList();
347 case "sort": case "pluck":
348 if (value is not List<object?> values) return value;
349 if (name == "pluck") return values.Select(item => SortKey(item, args)).ToList();
350 return values.OrderBy(item => SortKey(item, args), Comparer<object?>.Create(Compare)).ToList();
351 case "sum": case "min": case "max":
352 if (value is not List<object?> numbers) return Number(value, name);
353 int? total = name == "sum" ? 0 : null;
354 foreach (object? item in numbers)
355 {
356 int n = Number(item, name);
357 total = total is null ? n : name == "sum" ? total + n : name == "min" ? Math.Min(total.Value, n) : Math.Max(total.Value, n);
358 }
359 return total;
360 case "add": case "sub": case "mul": case "div": case "mod": case "gt": case "lt":
361 int number = Number(value, name), argument = (int)args[0];
362 int quotient = name is "div" or "mod" ? number / argument : 0;
363 if (name is "div" or "mod" && number % argument != 0 && (number < 0) != (argument < 0)) quotient--;
364 return name switch { "add" => number + argument, "sub" => number - argument, "mul" => number * argument, "div" => quotient, "mod" => number - argument * quotient, "gt" => number > argument, _ => number < argument };
365 case "comma": return Number(value, name).ToString("N0", System.Globalization.CultureInfo.InvariantCulture);
366 case "join": return value is List<object?> joined ? string.Join((string)args[0], joined.Select(StringValue)) : StringValue(value);
367 }
368 string text = StringValue(value);
369 switch (name)
370 {
371 case "upper": return text.ToUpperInvariant();
372 case "lower": return text.ToLowerInvariant();
373 case "trim": return Strip(text);
374 case "title":
375 StringBuilder title = new();
376 for (int i = 0; i < text.Length; i++) title.Append(i == 0 || text[i - 1] == ' ' ? char.ToUpperInvariant(text[i]) : char.ToLowerInvariant(text[i]));
377 return title.ToString();
378 case "pad": case "lpad":
379 string padding = new(args.Count == 1 ? ' ' : ((string)args[1])[0], Math.Max(0, (int)args[0] - text.Length));
380 return name == "pad" ? text + padding : padding + text;
381 case "truncate": return text.Length > (int)args[0] ? text[..(int)args[0]] + "..." : text;
382 case "replace": return text.Replace((string)args[0], (string)args[1], StringComparison.Ordinal);
383 default: throw new InvalidOperationException();
384 }
385 }
386 static object IndexPart(string part) => Matches(part, Subscript) ? int.TryParse(part, out int number) ? number : -1 : part;
387 static (bool Found, object? Value) Resolve(string name, List<object?> contexts, List<(int Index, int Count)> loops)
388 {
389 if (name == ".") return (true, contexts[^1]);
390 if (name.StartsWith('@'))
391 {
392 if (loops.Count == 0) return (false, null);
393 var loop = loops[^1];
394 object value = name switch { "@index" => loop.Index, "@number" => loop.Index + 1, "@first" => loop.Index == 0, _ => loop.Index == loop.Count - 1 };
395 return (true, value);
396 }
397 string[] parts = name.Split('.');
398 object? current = null;
399 bool found = false;
400 for (int i = contexts.Count - 1; i >= 0; i--)
401 if (Has(contexts[i], IndexPart(parts[0]))) { current = contexts[i]; found = true; break; }
402 if (!found) return (false, null);
403 foreach (string part in parts)
404 {
405 object key = IndexPart(part);
406 if (!Has(current, key)) return (false, null);
407 current = Get(current, key);
408 }
409 return (true, current);
410 }
411 static string Html(string text) => text.Replace("&", "&amp;").Replace("<", "&lt;").Replace(">", "&gt;").Replace("\"", "&quot;").Replace("'", "&#39;");
412 static void Render(List<Node> nodes, List<object?> contexts, List<(int Index, int Count)> loops, int depth, StringBuilder output)
413 {
414 foreach (Node node in nodes)
415 {
416 try
417 {
418 if (node.Kind == 'T') { output.Append(node.Text); continue; }
419 if (node.Kind == '>')
420 {
421 if (!Partials.TryGetValue(node.Text, out var partial)) throw new Failure($"unknown partial '{node.Text}'");
422 if (depth == 8) throw new Failure($"partial depth exceeded at '{node.Text}'");
423 Render(Analyze(partial, node.Indentation), contexts, loops, depth + 1, output);
424 continue;
425 }
426 if (node.Expression is null) continue;
427 var resolved = Resolve(node.Expression.Name, contexts, loops);
428 if (strict && node.Kind is 'V' or 'R' && !resolved.Found) throw new Failure($"missing '{node.Expression.Name}'");
429 object? value = resolved.Value;
430 foreach (Filter filter in node.Expression.Filters) value = Apply(value, filter);
431 if (node.Kind is 'V' or 'R') { string text = StringValue(value); output.Append(node.Kind == 'V' ? Html(text) : text); }
432 else if (node.Kind == '^') { if (!Truth(value)) Render(node.Children, contexts, loops, depth, output); }
433 else if (value is List<object?> items)
434 {
435 for (int i = 0; i < items.Count; i++)
436 {
437 contexts.Add(items[i]); loops.Add((i, items.Count));
438 Render(node.Children, contexts, loops, depth, output);
439 contexts.RemoveAt(contexts.Count - 1); loops.RemoveAt(loops.Count - 1);
440 }
441 }
442 else if (Truth(value)) { contexts.Add(value); Render(node.Children, contexts, loops, depth, output); contexts.RemoveAt(contexts.Count - 1); }
443 }
444 catch (Failure error) when (error.Line == 0) { throw new Failure(error.Message, node.Line); }
445 }
446 }
447 static void RenderTemplate(string name, int line)
448 {
449 try
450 {
451 if (!Templates.TryGetValue(name, out var body)) throw new Failure($"unknown template '{name}'", line);
452 StringBuilder output = new();
453 Render(Analyze(body, ""), [root], [], 0, output);
454 string result = output.ToString();
455 int count = result.Length == 0 ? 0 : result.Count(c => c == '\n') + (result.EndsWith('\n') ? 0 : 1);
456 Console.WriteLine($"TEMPLATE {name} OK lines={count} chars={result.Length}");
457 Console.Write(result);
458 if (result.Length > 0 && !result.EndsWith('\n')) Console.WriteLine();
459 }
460 catch (Failure error) { Console.WriteLine($"TEMPLATE {name} ERROR line {error.Line}: {error.Message}"); }
461 }
462 static void Leaves(object? value, string path, List<string> output)
463 {
464 if (value is Dictionary<string, object?> map && map.Count > 0)
465 foreach (var pair in map.OrderBy(pair => pair.Key, StringComparer.Ordinal)) Leaves(pair.Value, path.Length == 0 ? pair.Key : path + "." + pair.Key, output);
466 else if (value is List<object?> list && list.Count > 0)
467 for (int i = 0; i < list.Count; i++) Leaves(list[i], $"{path}[{i}]", output);
468 else
469 {
470 string notation = value switch { string s => "\"" + s.Replace("\\", "\\\\").Replace("\"", "\\\"").Replace("\n", "\\n") + "\"", Dictionary<string, object?> => "{}", List<object?> => "[]", _ => StringValue(value) };
471 output.Add(path + " = " + notation);
472 }
473 }
474 static void Main()
475 {
476 int number = 0;
477 string? command;
478 while ((command = Console.ReadLine()) is not null)
479 {
480 number++;
481 if (command == "END") break;
482 if (command == "DATA" || command.StartsWith("PARTIAL ", StringComparison.Ordinal) || command.StartsWith("TEMPLATE ", StringComparison.Ordinal))
483 {
484 string type = command.Split(' ')[0], name = type == "DATA" ? "" : command[(type.Length + 1)..];
485 List<SourceLine> body = [];
486 int start = number;
487 string? text;
488 while ((text = Console.ReadLine()) is not null) { number++; if (text == "END" + type) break; body.Add(new(number, text)); }
489 if (type == "DATA") Data(body);
490 else if (type == "TEMPLATE") { Templates[name] = body; RenderTemplate(name, start); }
491 else
492 {
493 try
494 {
495 Analyze(body, "");
496 bool existed = Partials.ContainsKey(name); Partials[name] = body;
497 Console.WriteLine($"PARTIAL {name} {(existed ? "REPLACED" : "DEFINED")} lines={body.Count}");
498 }
499 catch (Failure error) { Console.WriteLine($"PARTIAL {name} ERROR line {error.Line}: {error.Message}"); }
500 }
501 }
502 else if (command.StartsWith("RENDER ", StringComparison.Ordinal)) RenderTemplate(command[7..], number);
503 else if (command == "DUMP")
504 {
505 List<string> output = [];
506 foreach (var pair in root.OrderBy(pair => pair.Key, StringComparer.Ordinal)) Leaves(pair.Value, pair.Key, output);
507 Console.WriteLine($"DUMP leaves={output.Count}"); output.ForEach(Console.WriteLine);
508 }
509 else if (command == "RESET") { root = new(StringComparer.Ordinal); Partials.Clear(); Templates.Clear(); Console.WriteLine(command); }
510 else { strict = command == "STRICT ON"; Console.WriteLine(command); }
511 }
512 }
