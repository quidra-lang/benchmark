13 last <- nchar(text)
14 while (first <= last && substr(text, first, first) == " ") first <- first + 1

16 if (first > last) "" else substr(text, first, last)

25key_parts <- function(key) {
26 pattern <- "^[A-Za-z_][A-Za-z0-9_]*((\\.[A-Za-z_][A-Za-z0-9_]*)|(\\[(0|[1-9][0-9]*)\\]))*$"
27 if (!grepl(pattern, key)) return(NULL)
28 matches <- gregexpr("[A-Za-z_][A-Za-z0-9_]*|\\[(0|[1-9][0-9]*)\\]", key, perl = TRUE)[[1]]
29 lengths <- attr(matches, "match.length")

31 part <- substr(key, matches[i], matches[i] + lengths[i] - 1)
32 index <- startsWith(part, "[")
33 list(part = if (index) as.numeric(substr(part, 2, nchar(part) - 1)) else part,
34 prefix = substr(key, 1, matches[i] + lengths[i] - 1))
35 })

37parse_scalar <- function(text, position) {
38 if (text_slice(text, position, position) == '"') {
39 position <- position + 1
40 output <- ""
41 while (position <= nchar(text)) {
42 char <- substr(text, position, position)
43 position <- position + 1
44 if (char == '"') return(list(value = value("string", output), position = position))
45 if (char == "\\") {
46 char <- text_slice(text, position, position)
47 position <- position + 1
48 if (!(char %in% c('"', "\\", "n"))) stop("bad value")
49 if (char == "n") char <- "\n"
50 }
51 output <- paste0(output, char)
52 }
53 stop("bad value")
54 }
55 rest <- text_slice(text, position)
56 match <- regexpr("^(0|-?[1-9][0-9]*|true|false)", rest, perl = TRUE)
57 if (match[1] < 0) stop("bad value")
58 word <- regmatches(rest, match)
59 parsed <- if (word %in% c("true", "false")) value("boolean", word == "true") else value("integer", as.numeric(word))
60 list(value = parsed, position = position + nchar(word))
61}
62parse_value <- function(text) {
63 if (text == "[]") return(value("list", list()))
64 if (text == "{}") return(value("map", list()))
65 if (startsWith(text, "[")) {
66 output <- list()
67 position <- 2
68 repeat {
69 parsed <- parse_scalar(text, position)
70 output[[length(output) + 1]] <- parsed$value
71 position <- parsed$position
72 if (text_slice(text, position, position) == "]" && position == nchar(text)) return(value("list", output))
73 if (text_slice(text, position, position + 1) != ", ") stop("bad value")
74 position <- position + 2
75 }
76 }
77 parsed <- parse_scalar(text, 1)
78 if (parsed$position != nchar(text) + 1) stop("bad value")
79 parsed$value

92assign_value <- function(container, parts, new_value, index = 1) {
93 part <- parts[[index]]$part
94 prefix <- parts[[index]]$prefix
95 if (is.numeric(part) && part > length(container$data)) fail(paste0("index gap at '", prefix, "'"))
96 if (index == length(parts)) return(put_part(container, part, new_value))
97 kind <- if (is.numeric(parts[[index + 1]]$part)) "list" else "map"
98 if (has_part(container, part)) {
99 child <- get_part(container, part)
100 if (child$kind != kind) fail(paste0("type conflict at '", prefix, "'"))
101 } else {
102 child <- value(kind, list())
103 }
104 put_part(container, part, assign_value(child, parts, new_value, index + 1))
105}

115data_block <- function(body) {
116 sets <- 0
117 unsets <- 0
118 errors <- character()
119 for (entry in body) {
120 text <- entry$text
121 if (text == "" || startsWith(text, "#")) next
122 tryCatch({
123 deleting <- startsWith(text, "UNSET ")
124 if (deleting) {
125 key <- substring(text, 7)
126 } else {
127 at <- regexpr(" = ", text, fixed = TRUE)[1]
128 if (at < 0) fail("bad line")
129 key <- text_slice(text, 1, at - 1)
130 notation <- text_slice(text, at + 3)
131 }
132 parts <- key_parts(key)
133 if (is.null(parts)) fail(paste0("bad key '", key, "'"))
134 if (!deleting) {
135 parsed <- tryCatch(parse_value(notation), error = function(e) NULL)
136 if (is.null(parsed)) fail(paste0("bad value '", notation, "'"))
137 }
138 if (deleting) {
139 root <<- unset_value(root, parts, key)
140 unsets <- unsets + 1
141 } else {
142 root <<- assign_value(root, parts, parsed)
143 sets <- sets + 1
144 }
145 }, template_error = function(e) {
146 errors <<- c(errors, paste0("ERROR line ", entry$line, ": ", e$message))
147 })
148 }
149 cat(sprintf("DATA set=%d unset=%d errors=%d\n", sets, unsets, length(errors)))
150 if (length(errors)) cat(paste0(errors, "\n"), sep = "")
151}
152split_quoted <- function(text, separator) {
153 output <- character()
154 quoted <- FALSE
155 start <- 1
156 letters <- chars(text)
157 for (i in seq_along(letters)) {
158 if (letters[i] == "'") quoted <- !quoted
159 if (letters[i] == separator && !quoted) {
160 output <- c(output, text_slice(text, start, i - 1))
161 start <- i + 1
162 }
163 }
164 c(output, text_slice(text, start))
165}
166parse_expression <- function(text, line) {
167 pieces <- vapply(split_quoted(text, "|"), strip_spaces, character(1))
168 name <- pieces[1]
169 if (!grepl(name_pattern, name)) fail(paste0("invalid name '", name, "'"), line)
170 filters <- list()
171 for (piece in pieces[-1]) {
172 parts <- split_quoted(piece, ":")
173 filter <- parts[1]
174 if (!(filter %in% all_filters)) fail(paste0("unknown filter '", filter, "'"), line)
175 valid <- TRUE
176 args <- lapply(parts[-1], function(argument) {
177 if (grepl("^'[^']*'$", argument)) return(value("string", text_slice(argument, 2, nchar(argument) - 1)))
178 if (grepl("^-?[0-9]{1,9}$", argument)) return(value("integer", as.numeric(argument)))
179 valid <<- FALSE
180 absent
181 })
182 count <- length(args)
183 integers <- all(vapply(args, function(a) a$kind == "integer", logical(1)))
184 strings <- all(vapply(args, function(a) a$kind == "string", logical(1)))
185 correct <- if (filter %in% no_arguments) {
186 count == 0
187 } else if (filter %in% c("default", "eq")) {
188 count == 1
189 } else if (filter %in% c("pad", "lpad")) {
190 count %in% c(1, 2) && args[[1]]$kind == "integer" && args[[1]]$data >= 0 &&
191 (count == 1 || (args[[2]]$kind == "string" && nchar(args[[2]]$data) == 1))
192 } else if (filter == "truncate") {
193 count == 1 && integers && args[[1]]$data >= 0
194 } else if (filter == "replace") {
195 count == 2 && strings && args[[1]]$data != ""
196 } else if (filter %in% c("join", "pluck")) {
197 count == 1 && strings
198 } else if (filter == "sort") {
199 count <= 1 && strings
200 } else {
201 count == 1 && integers && (!(filter %in% c("div", "mod")) || args[[1]]$data != 0)
202 }
203 if (!valid || !correct) fail(paste0("bad arguments for '", filter, "'"), line)
204 filters[[length(filters) + 1]] <- list(name = filter, args = args)
205 }
206 list(name = name, filters = filters)
207}

212analyze <- function(body) {
213 opening <- "{{"
214 closing <- "}}"
215 tokens <- list()
216 sections <- list()
217 kinds <- c("#" = "section", "^" = "inverted", "/" = "close", "!" = "comment", ">" = "partial", "&" = "raw", "=" = "delimiter")
218 for (entry in body) {
219 text <- entry$text
220 line <- entry$line
221 local <- list()
222 position <- 1
223 repeat {
224 start <- find_from(text, opening, position)
225 if (start < 0) {
226 local[[length(local) + 1]] <- list(kind = "text", text = text_slice(text, position), line = line)
227 break
228 }
229 local[[length(local) + 1]] <- list(kind = "text", text = text_slice(text, position, start - 1), line = line)
230 content_start <- start + nchar(opening)
231 triple <- text_slice(text, content_start, content_start) == "{"
232 if (triple) content_start <- content_start + 1
233 end_delimiter <- paste0(if (triple) "}" else "", closing)
234 finish <- find_from(text, end_delimiter, content_start)
235 if (finish < 0) fail("unclosed tag", line)
236 content <- strip_spaces(text_slice(text, content_start, finish - 1))
237 position <- finish + nchar(end_delimiter)
238 marker <- text_slice(content, 1, 1)
239 kind <- if (triple) "raw" else if (marker %in% names(kinds)) unname(kinds[marker]) else "variable"
240 argument <- if (triple || kind == "variable") content else strip_spaces(text_slice(content, 2))
241 token <- list(kind = kind, text = argument, line = line, indent = "")
242 if (kind %in% c("variable", "raw", "section", "inverted")) {
243 token$expression <- parse_expression(argument, line)
244 if (kind %in% c("section", "inverted")) sections[[length(sections) + 1]] <- list(name = token$expression$name, line = line)
245 } else if (kind == "close") {
246 if (!grepl(name_pattern, argument)) fail(paste0("invalid name '", argument, "'"), line)
247 if (!length(sections)) fail(paste0("unexpected closing '", argument, "'"), line)
248 expected <- sections[[length(sections)]]$name
249 if (expected != argument) fail(paste0("mismatched closing '", argument, "', expected '", expected, "'"), line)
250 sections <- sections[-length(sections)]
251 } else if (kind == "partial") {
252 if (!grepl(identifier, argument)) fail(paste0("invalid name '", argument, "'"), line)
253 } else if (kind == "delimiter") {
254 delimiters <- character()
255 if (nchar(content) >= 2 && endsWith(content, "=")) {
256 inner <- strip_spaces(text_slice(content, 2, nchar(content) - 1))
257 if (inner != "") delimiters <- strsplit(inner, " +")[[1]]
258 }
259 if (length(delimiters) != 2 || any(grepl("=", delimiters, fixed = TRUE))) fail("invalid delimiters", line)
260 opening <- delimiters[1]
261 closing <- delimiters[2]
262 }
263 local[[length(local) + 1]] <- token
264 }
265 tags <- Filter(function(t) t$kind != "text", local)
266 standalone <- length(tags) == 1 && tags[[1]]$kind %in% c("section", "inverted", "close", "comment", "partial", "delimiter") &&
267 all(vapply(local, function(t) t$kind != "text" || grepl("^ *$", t$text), logical(1)))
268 if (standalone) {
269 if (tags[[1]]$kind == "partial") tags[[1]]$indent <- local[[1]]$text
270 tokens <- c(tokens, tags)
271 } else {
272 tokens <- c(tokens, local, list(list(kind = "text", text = "\n", line = line)))
273 }
274 }
275 if (length(sections)) {
276 section <- sections[[length(sections)]]
277 fail(paste0("unclosed section '", section$name, "'"), section$line)
278 }
279 position <- 1
280 tree <- function() {
281 nodes <- list()
282 while (position <= length(tokens)) {
283 token <- tokens[[position]]
284 position <<- position + 1
285 if (token$kind == "close") break
286 if (token$kind %in% c("section", "inverted")) token$children <- tree()
287 nodes[[length(nodes) + 1]] <- token
288 }
289 nodes
290 }
291 tree()
292}

303number <- function(v, filter) {
304 if (v$kind == "absent") return(0)
305 if (v$kind == "integer") return(v$data)
306 if (v$kind == "string" && grepl("^-?[0-9]{1,9}$", v$data)) return(as.numeric(v$data))

309apply_filter <- function(v, filter, args) {
310 argument <- if (length(args)) args[[1]]$data else NULL
311 if (filter %in% c("upper", "lower", "title", "trim")) {
312 text <- stringify(v)
313 text <- switch(filter, upper = toupper(text), lower = tolower(text), trim = strip_spaces(text), title = {
314 letters <- chars(tolower(text))
315 for (i in seq_along(letters)) if (i == 1 || letters[i - 1] == " ") letters[i] <- toupper(letters[i])
316 paste(letters, collapse = "")
317 })
318 return(value("string", text))
319 }
320 if (filter == "length") return(value("integer", if (v$kind %in% c("list", "map")) length(v$data) else nchar(stringify(v))))
321 if (filter %in% c("first", "last")) {
322 if (v$kind != "list") return(v)
323 if (!length(v$data)) return(absent)
324 return(v$data[[if (filter == "first") 1 else length(v$data)]])
325 }
326 if (filter == "reverse") {
327 if (v$kind == "list") return(value("list", rev(v$data)))
328 if (v$kind == "string") return(value("string", paste(rev(chars(v$data)), collapse = "")))
329 return(v)
330 }
331 if (filter == "items") {
332 if (v$kind != "map") return(v)
333 return(value("list", lapply(sort(names(v$data), method = "radix"), function(key) value("map", list(key = value("string", key), value = v$data[[key]])))))
334 }
335 if (filter == "comma") {
336 n <- number(v, filter)
337 digits <- chars(sprintf("%.0f", abs(n)))
338 groups <- split(digits, ceiling(seq_along(digits) / 3 + ((3 - length(digits) %% 3) %% 3) / 3))
339 return(value("string", paste0(if (n < 0) "-" else "", paste(vapply(groups, paste, character(1), collapse = ""), collapse = ","))))
340 }
341 if (filter %in% c("sum", "min", "max")) {
342 if (v$kind != "list") return(value("integer", number(v, filter)))
343 numbers <- vapply(v$data, function(item) number(item, filter), numeric(1))
344 if (!length(numbers) && filter != "sum") return(absent)
345 return(value("integer", switch(filter, sum = sum(numbers), min = min(numbers), max = max(numbers))))
346 }
347 if (filter == "default") return(if (truthy(v)) v else args[[1]])
348 if (filter == "eq") return(value("boolean", stringify(v) == stringify(args[[1]])))
349 if (filter %in% c("pad", "lpad", "truncate", "replace", "join")) {
350 if (filter == "join" && v$kind == "list") return(value("string", paste(vapply(v$data, stringify, character(1)), collapse = argument)))
351 text <- stringify(v)
352 if (filter %in% c("pad", "lpad")) {
353 padding <- strrep(if (length(args) == 2) args[[2]]$data else " ", max(0, argument - nchar(text)))
354 text <- if (filter == "pad") paste0(text, padding) else paste0(padding, text)
355 } else if (filter == "truncate" && nchar(text) > argument) {
356 text <- paste0(text_slice(text, 1, argument), "...")
357 } else if (filter == "replace") {
358 pieces <- character()
359 position <- 1
360 repeat {
361 at <- find_from(text, argument, position)
362 if (at < 0) break
363 pieces <- c(pieces, text_slice(text, position, at - 1), args[[2]]$data)
364 position <- at + nchar(argument)
365 }
366 text <- paste0(c(pieces, text_slice(text, position)), collapse = "")
367 }
368 return(value("string", text))
369 }
370 if (filter %in% c("sort", "pluck")) {
371 if (v$kind != "list") return(v)
372 keys <- lapply(v$data, function(item) {
373 if (filter == "sort" && !length(args)) item else if (item$kind == "map" && argument %in% names(item$data)) item$data[[argument]] else absent
374 })
375 if (filter == "pluck") return(value("list", keys))
376 ranks <- vapply(keys, function(k) switch(k$kind, absent = 0, integer = 1, string = 2, boolean = 3, 4), numeric(1))
377 numeric_keys <- vapply(keys, function(k) if (k$kind %in% c("integer", "boolean")) as.numeric(k$data) else 0, numeric(1))
378 string_keys <- vapply(keys, function(k) if (k$kind == "string") k$data else "", character(1))
379 return(value("list", v$data[order(ranks, numeric_keys, string_keys, seq_along(keys), method = "radix")]))
380 }
381 n <- number(v, filter)
382 result <- switch(filter, add = n + argument, sub = n - argument, mul = n * argument,
383 div = floor(n / argument), mod = n - argument * floor(n / argument), gt = n > argument, lt = n < argument)
384 value(if (filter %in% c("gt", "lt")) "boolean" else "integer", result)
385}
386resolve <- function(name, contexts, loops) {
387 if (name == ".") return(list(found = TRUE, value = contexts[[length(contexts)]]))
388 if (startsWith(name, "@")) {
389 if (!length(loops)) return(list(found = FALSE, value = absent))
390 loop <- loops[[length(loops)]]
391 output <- switch(name, "@index" = value("integer", loop$i), "@number" = value("integer", loop$i + 1), "@first" = value("boolean", loop$i == 0), "@last" = value("boolean", loop$i == loop$n - 1))
392 return(list(found = TRUE, value = output))
393 }
394 parts <- lapply(strsplit(name, ".", fixed = TRUE)[[1]], function(part) if (grepl("^[0-9]+$", part)) as.numeric(part) else part)
395 found <- FALSE
396 for (context in rev(contexts)) {
397 if (has_part(context, parts[[1]])) {
398 found <- TRUE
399 break
400 }
401 }
402 if (!found) return(list(found = FALSE, value = absent))
403 for (part in parts) {
404 if (!has_part(context, part)) return(list(found = FALSE, value = absent))
405 context <- get_part(context, part)
406 }
407 list(found = TRUE, value = context)
408}

421render <- function(nodes, contexts, loops, depth) {
422 output <- character()
423 for (token in nodes) {
424 text <- tryCatch({
425 if (token$kind == "text") {
426 token$text
427 } else if (token$kind %in% c("variable", "raw")) {
428 result <- stringify(evaluate(token, contexts, loops))
429 if (token$kind == "variable") escape_html(result) else result
430 } else if (token$kind == "section") {
431 v <- evaluate(token, contexts, loops)
432 pieces <- character()
433 if (v$kind == "list") {
434 for (i in seq_along(v$data)) pieces <- c(pieces, render(token$children, c(contexts, list(v$data[[i]])), c(loops, list(list(i = i - 1, n = length(v$data)))), depth))
435 } else if (truthy(v)) {
436 pieces <- render(token$children, c(contexts, list(v)), loops, depth)
437 }
438 paste(pieces, collapse = "")
439 } else if (token$kind == "inverted") {
440 if (!truthy(evaluate(token, contexts, loops))) render(token$children, contexts, loops, depth) else ""
441 } else if (token$kind == "partial") {
442 name <- token$text
443 if (!(name %in% names(partials))) fail(paste0("unknown partial '", name, "'"))
444 if (depth == 8) fail(paste0("partial depth exceeded at '", name, "'"))
445 body <- lapply(partials[[name]], function(entry) {
446 if (entry$text != "") entry$text <- paste0(token$indent, entry$text)
447 entry
448 })
449 render(analyze(body), contexts, loops, depth + 1)
450 } else ""
451 }, template_error = function(e) {
452 if (is.null(e$line)) e$line <- token$line
453 stop(e)
454 })
455 output <- c(output, text)
456 }
457 paste(output, collapse = "")

476 } else if (v$kind == "list" && length(v$data)) {

485 output

492 position <- position + 1

494 at <- regexpr(" ", command, fixed = TRUE)[1]

496 name <- if (at < 0) "" else substring(command, at + 1)
497 if (kind %in% c("DATA", "PARTIAL", "TEMPLATE")) {
498 ending <- switch(kind, DATA = "ENDDATA", PARTIAL = "ENDPARTIAL", TEMPLATE = "ENDTEMPLATE")
499 body <- list()
500 while (input[position] != ending) {
501 body[[length(body) + 1]] <- list(text = input[position], line = position)
502 position <- position + 1
503 }
504 position <- position + 1
505 if (kind == "DATA") {
506 data_block(body)
507 } else if (kind == "PARTIAL") {
508 tryCatch({
509 analyze(body)
510 action <- if (name %in% names(partials)) "REPLACED" else "DEFINED"
511 partials[name] <- list(body)
512 cat(sprintf("PARTIAL %s %s lines=%d\n", name, action, length(body)))
513 }, template_error = function(e) cat(sprintf("PARTIAL %s ERROR line %d: %s\n", name, e$line, e$message)))
514 } else {
515 templates[name] <- list(body)
516 render_template(name, line)
517 }
518 } else if (kind == "RENDER") {
519 render_template(name, line)
520 } else if (kind == "DUMP") {
521 leaves <- dump_leaves(root)
522 cat(sprintf("DUMP leaves=%d\n", length(leaves)))
523 if (length(leaves)) cat(paste0(leaves, "\n"), sep = "")
524 } else if (kind == "STRICT") {
525 strict <- name == "ON"
526 cat(command, "\n", sep = "")
527 } else if (kind == "RESET") {
528 root <- value("map", list())
529 partials <- list()
530 templates <- list()
531 cat("RESET\n")
532 }
