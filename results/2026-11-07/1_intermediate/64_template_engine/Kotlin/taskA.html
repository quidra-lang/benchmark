32private fun keyParts(s: String): List<KeyPart>? {
33 var at = 0
34 val parts = mutableListOf<KeyPart>()
35 while (true) {
36 val start = at
37 while (
38 at < s.length &&
39 (s[at] in 'A'..'Z' || s[at] in 'a'..'z' || s[at] in '0'..'9' || s[at] == '_')
40 ) at++
41 if (!identifier(s.substring(start, at))) return null
42 parts.add(KeyPart(Key(s.substring(start, at)), s.substring(0, at)))
43 while (true) {
44 if (at == s.length) return parts
45 if (s[at] == '.') {
46 at++
47 break
48 }
49 if (s[at] != '[') return null
50 val end = s.indexOf(']', at + 1)
51 if (end < 0 || !subscript(s.substring(at + 1, end))) return null
52 val index = s.substring(at + 1, end).toIntOrNull() ?: return null
53 at = end + 1
54 parts.add(KeyPart(Index(index), s.substring(0, at)))
55 }
56 }
57}

59private class ValueParser(private val text: String) {
60 private var at = 0
61
62 private fun scalar(): Value? {
63 if (text.getOrNull(at) == '"') {
64 at++
65 val s = StringBuilder()
66 while (at < text.length) {
67 when (val c = text[at++]) {
68 '"' -> return Text(s.toString())
69 '\\' -> {
70 if (at == text.length) return null
71 s.append(
72 when (text[at++]) {
73 '"' -> '"'
74 '\\' -> '\\'
75 'n' -> '\n'
76 else -> return null
77 },
78 )
79 }
80 else -> s.append(c)
81 }
82 }
83 return null
84 }
85 val start = at
86 while (at < text.length && text[at] != ',' && text[at] != ']') at++
87 val word = text.substring(start, at)
88 return when (word) {
89 "true" -> BooleanValue(true)
90 "false" -> BooleanValue(false)
91 else ->
92 if (word != "-0" && subscript(word.removePrefix("-")))
93 word.toLongOrNull()?.let(::NumberValue)
94 else null
95 }
96 }

98 fun parse(): Value? {
99 if (text == "{}") return MapValue(emptyMap())
100 if (text == "[]") return ListValue(emptyList())
101 if (text.startsWith('[')) {
102 at = 1
103 val list = mutableListOf<Value>()
104 while (true) {
105 list.add(scalar() ?: return null)
106 if (text.getOrNull(at) == ']')
107 return if (at + 1 == text.length) ListValue(list) else null
108 if (!text.startsWith(", ", at)) return null
109 at += 2
110 }
111 }
112 val value = scalar()
113 return if (at == text.length) value else null
114 }
115}

117private fun child(value: Value, part: Part): Value? =
118 when {
119 value is MapValue && part is Key -> value.values[part.name]
120 value is ListValue && part is Index -> value.values.getOrNull(part.index)
121 else -> null
122 }
123
124private fun assign(container: Value, parts: List<KeyPart>, at: Int, value: Value): Value {
125 val (part, prefix) = parts[at]
126 if (container is ListValue && part is Index && part.index > container.values.size)
127 throw Problem("index gap at '$prefix'")
128 val next =
129 if (at == parts.lastIndex) value
130 else {
131 val mapNeeded = parts[at + 1].part is Key
132 val old =
133 child(container, part)
134 ?: if (mapNeeded) MapValue(emptyMap()) else ListValue(emptyList())
135 if ((mapNeeded && old !is MapValue) || (!mapNeeded && old !is ListValue))
136 throw Problem("type conflict at '$prefix'")
137 assign(old, parts, at + 1, value)
138 }
139 return when {
140 container is MapValue && part is Key -> MapValue(container.values + (part.name to next))
141 container is ListValue && part is Index -> {
142 val list = container.values.toMutableList()
143 if (part.index == list.size) list.add(next) else list[part.index] = next
144 ListValue(list)
145 }
146 else -> error("invalid container")
147 }
148}
149
150private fun remove(container: Value, parts: List<KeyPart>, at: Int): Value? {
151 val part = parts[at].part
152 val old = child(container, part) ?: return null
153 if (at == parts.lastIndex)
154 return when {
155 container is MapValue && part is Key -> MapValue(container.values - part.name)
156 container is ListValue && part is Index ->
157 ListValue(container.values.toMutableList().also { it.removeAt(part.index) })
158 else -> null
159 }
160 val next = remove(old, parts, at + 1) ?: return null
161 return when {
162 container is MapValue && part is Key -> MapValue(container.values + (part.name to next))
163 container is ListValue && part is Index ->
164 ListValue(container.values.toMutableList().also { it[part.index] = next })
165 else -> null
166 }
167}

197private fun quotedSplit(s: String, separator: Char): List<String> {
198 var quoted = false
199 var start = 0
200 val pieces = mutableListOf<String>()
201 for ((i, c) in s.withIndex()) {
202 if (c == '\'') quoted = !quoted
203 if (c == separator && !quoted) {
204 pieces.add(s.substring(start, i))
205 start = i + 1
206 }
207 }
208 pieces.add(s.substring(start))
209 return pieces
210}

217private data class Filter(val name: String, val args: List<Value>)

219private data class Expression(val name: String, val filters: List<Filter>)

221private fun expression(s: String): Expression {
222 val pieces = quotedSplit(s, '|')
223 val name = pieces[0].trim(' ')
224 if (!validName(name)) throw Problem("invalid name '$name'")
225 val noArgs =
226 setOf(
227 "upper",
228 "lower",
229 "title",
230 "trim",
231 "length",
232 "first",
233 "last",
234 "reverse",
235 "items",
236 "comma",
237 "sum",
238 "min",
239 "max",
240 )
241 val others =
242 setOf(
243 "default",
244 "eq",
245 "pad",
246 "lpad",
247 "truncate",
248 "replace",
249 "join",
250 "sort",
251 "pluck",
252 "add",
253 "sub",
254 "mul",
255 "div",
256 "mod",
257 "gt",
258 "lt",
259 )
260 val filters =
261 pieces.drop(1).map { piece ->
262 val items = quotedSplit(piece.trim(' '), ':')
263 val f = items[0]
264 if (f !in noArgs && f !in others) throw Problem("unknown filter '$f'")
265 val args =
266 items.drop(1).map { arg ->
267 if (
268 arg.length >= 2 &&
269 arg.first() == '\'' &&
270 arg.last() == '\'' &&
271 '\'' !in arg.substring(1, arg.lastIndex)
272 )
273 Text(arg.substring(1, arg.lastIndex))
274 else
275 integerArgument(arg)?.let(::NumberValue)
276 ?: throw Problem("bad arguments for '$f'")
277 }
278 fun n(i: Int) = (args.getOrNull(i) as? NumberValue)?.value
279 fun t(i: Int) = (args.getOrNull(i) as? Text)?.value
280 val valid =
281 if (f in noArgs) args.isEmpty()
282 else
283 when (f) {
284 "default",
285 "eq" -> args.size == 1
286 "pad",
287 "lpad" ->
288 args.size in 1..2 &&
289 (n(0) ?: -1L) >= 0 &&
290 (args.size == 1 || t(1)?.length == 1)
291 "truncate" -> args.size == 1 && (n(0) ?: -1L) >= 0
292 "replace" -> args.size == 2 && !t(0).isNullOrEmpty() && t(1) != null
293 "join",
294 "pluck" -> args.size == 1 && t(0) != null
295 "sort" -> args.isEmpty() || (args.size == 1 && t(0) != null)
296 "div",
297 "mod" -> args.size == 1 && n(0) != null && n(0) != 0L
298 else -> args.size == 1 && n(0) != null
299 }
300 if (!valid) throw Problem("bad arguments for '$f'")
301 Filter(f, args)
302 }
303 return Expression(name, filters)
304}

306private fun sortCompare(a: Value, b: Value): Int {
307 fun rank(v: Value) =
308 when (v) {
309 Absent -> 0
310 is NumberValue -> 1
311 is Text -> 2
312 is BooleanValue -> 3
313 else -> 4
314 }
315 val order = rank(a).compareTo(rank(b))
316 if (order != 0) return order
317 return when {
318 a is NumberValue && b is NumberValue -> a.value.compareTo(b.value)
319 a is Text && b is Text -> a.value.compareTo(b.value)
320 a is BooleanValue && b is BooleanValue -> a.value.compareTo(b.value)
321 else -> 0
322 }
323}

325private fun applyFilter(v: Value, f: Filter): Value {
326 val name = f.name
327 fun n() = (f.args[0] as NumberValue).value
328 fun s(i: Int) = (f.args[i] as Text).value
329 return when (name) {
330 "default" -> if (truthy(v)) v else f.args[0]
331 "eq" -> BooleanValue(stringify(v) == stringify(f.args[0]))
332 "length" ->
333 NumberValue(
334 (when (v) {
335 is ListValue -> v.values.size
336 is MapValue -> v.values.size
337 else -> stringify(v).length
338 })
339 .toLong(),
340 )
341 "first",
342 "last" ->
343 if (v is ListValue)
344 (if (name == "first") v.values.firstOrNull() else v.values.lastOrNull()) ?: Absent
345 else v
346 "reverse" ->
347 when (v) {
348 is ListValue -> ListValue(v.values.reversed())
349 is Text -> Text(v.value.reversed())
350 else -> v
351 }
352 "items" ->
353 if (v is MapValue)
354 ListValue(
355 v.values.toSortedMap().map { (k, value) ->
356 MapValue(mapOf("key" to Text(k), "value" to value))
357 },
358 )
359 else v
360 "sort" ->
361 if (v is ListValue) {
362 fun key(value: Value) =
363 if (f.args.isEmpty()) value
364 else (value as? MapValue)?.values?.get(s(0)) ?: Absent
365 ListValue(v.values.sortedWith { a, b -> sortCompare(key(a), key(b)) })
366 } else v
367 "pluck" ->
368 if (v is ListValue)
369 ListValue(v.values.map { (it as? MapValue)?.values?.get(s(0)) ?: Absent })
370 else v
371 "sum",
372 "min",
373 "max" ->
374 if (v is ListValue) {
375 val numbers = v.values.map { number(it, name) }
376 when (name) {
377 "sum" -> NumberValue(numbers.sum())
378 "min" -> numbers.minOrNull()?.let(::NumberValue) ?: Absent
379 else -> numbers.maxOrNull()?.let(::NumberValue) ?: Absent
380 }
381 } else NumberValue(number(v, name))
382 "add",
383 "sub",
384 "mul",
385 "div",
386 "mod",
387 "gt",
388 "lt" -> {
389 val a = number(v, name)
390 val b = n()
391 fun floor(): Long {
392 val q = a / b
393 return if (a % b != 0L && (a < 0) != (b < 0)) q - 1 else q
394 }
395 when (name) {
396 "add" -> NumberValue(a + b)
397 "sub" -> NumberValue(a - b)
398 "mul" -> NumberValue(a * b)
399 "div" -> NumberValue(floor())
400 "mod" -> NumberValue(a - b * floor())
401 "gt" -> BooleanValue(a > b)
402 else -> BooleanValue(a < b)
403 }
404 }
405 "comma" -> {
406 val a = number(v, name)
407 val digits = kotlin.math.abs(a).toString()
408 Text(
409 (if (a < 0) "-" else "") +
410 digits
411 .mapIndexed { i, c ->
412 (if (i > 0 && (digits.length - i) % 3 == 0) "," else "") + c
413 }
414 .joinToString(""),
415 )
416 }
417 "join" ->
418 Text(
419 if (v is ListValue) v.values.joinToString(s(0), transform = ::stringify)
420 else stringify(v),
421 )
422 else -> {
423 val text = stringify(v)
424 Text(
425 when (name) {
426 "upper" -> text.uppercase()
427 "lower" -> text.lowercase()
428 "trim" -> text.trim(' ')
429 "title" ->
430 text
431 .mapIndexed { i, c ->
432 if (i == 0 || text[i - 1] == ' ') c.uppercaseChar()
433 else c.lowercaseChar()
434 }
435 .joinToString("")
436 "pad" -> text.padEnd(n().toInt(), if (f.args.size == 2) s(1)[0] else ' ')
437 "lpad" -> text.padStart(n().toInt(), if (f.args.size == 2) s(1)[0] else ' ')
438 "truncate" -> if (text.length > n()) text.take(n().toInt()) + "..." else text
439 else -> text.replace(s(0), s(1))
440 },
441 )
442 }
443 }
444}

457private fun analyze(source: List<SourceLine>): List<Tag> {
458 var open = "{{"
459 var close = "}}"
460 val stack = mutableListOf<Pair<String, Int>>()
461 val tokens = mutableListOf<Tag>()
462 for ((line, text) in source) {
463 var at = 0
464 val row = mutableListOf<Tag>()
465 while (true) {
466 val start = text.indexOf(open, at)
467 if (start < 0) break
468 row.add(Tag('t', text.substring(at, start), line))
469 var begin = start + open.length
470 val triple = text.startsWith("{", begin)
471 if (triple) begin++
472 val ending = if (triple) "}$close" else close
473 val finish = text.indexOf(ending, begin)
474 if (finish < 0) throw Problem("unclosed tag", line)
475 val content = text.substring(begin, finish).trim(' ')
476 at = finish + ending.length
477 val kind = if (triple) '&' else content.firstOrNull()?.takeIf { it in "#^/!>&=" } ?: 'v'
478 val arg = if (!triple && kind != 'v') content.substring(1).trim(' ') else content
479 var expr: Expression? = null
480 try {
481 when (kind) {
482 'v',
483 '&',
484 '#',
485 '^' -> {
486 expr = expression(arg)
487 if (kind == '#' || kind == '^') stack.add(expr.name to line)
488 }
489 '/' -> {
490 if (!validName(arg)) throw Problem("invalid name '$arg'")
491 val top = stack.lastOrNull() ?: throw Problem("unexpected closing '$arg'")
492 if (top.first != arg)
493 throw Problem("mismatched closing '$arg', expected '${top.first}'")
494 stack.removeAt(stack.lastIndex)
495 }
496 '>' -> if (!identifier(arg)) throw Problem("invalid name '$arg'")
497 '=' -> {
498 if (content.length < 2 || !content.endsWith('='))
499 throw Problem("invalid delimiters")
500 val pair =
501 content.substring(1, content.lastIndex).split(' ').filter {
502 it.isNotEmpty()
503 }
504 if (pair.size != 2 || pair.any { '=' in it })
505 throw Problem("invalid delimiters")
506 open = pair[0]
507 close = pair[1]
508 }
509 }
510 } catch (p: Problem) {
511 throw Problem(p.reason, line)
512 }
513 row.add(Tag(kind, arg, line, expr))
514 }
515 row.add(Tag('t', text.substring(at), line))
516 val positions = row.indices.filter { row[it].kind != 't' }
517 if (
518 positions.size == 1 &&
519 row[positions[0]].kind in "#^/!>=" &&
520 row.filter { it.kind == 't' }.all { it.text.all { c -> c == ' ' } }
521 ) {
522 val i = positions[0]
523 val tag = row[i]
524 if (tag.kind == '>') tag.indent = row.take(i).joinToString("") { it.text }
525 tokens.add(tag)
526 } else {
527 tokens.addAll(row)
528 tokens.add(Tag('t', "\n", line))
529 }
530 }
531 stack.lastOrNull()?.let { throw Problem("unclosed section '${it.first}'", it.second) }
532 val openings = mutableListOf<Int>()
533 for (i in tokens.indices) {
534 if (tokens[i].kind == '#' || tokens[i].kind == '^') openings.add(i)
535 if (tokens[i].kind == '/') tokens[openings.removeAt(openings.lastIndex)].end = i
536 }
537 return tokens
538}

540private fun resolve(name: String, contexts: List<Value>, loops: List<Pair<Int, Int>>): Value? {
541 if (name == ".") return contexts.last()
542 if (name.startsWith('@')) {
543 val (i, n) = loops.lastOrNull() ?: return null
544 return when (name) {
545 "@index" -> NumberValue(i.toLong())
546 "@number" -> NumberValue(i + 1L)
547 "@first" -> BooleanValue(i == 0)
548 else -> BooleanValue(i == n - 1)
549 }
550 }
551 val parts =
552 name.split('.').map {
553 if (subscript(it)) Index(it.toIntOrNull() ?: Int.MAX_VALUE) else Key(it)
554 }
555 for (context in contexts.asReversed()) {
556 var value = child(context, parts[0]) ?: continue
557 for (part in parts.drop(1)) value = child(value, part) ?: return null
558 return value
559 }
560 return null
561}

563private fun escape(s: String) =
564 s.map {
565 when (it) {
566 '&' -> "&amp;"
567 '<' -> "&lt;"
568 '>' -> "&gt;"
569 '"' -> "&quot;"
570 '\'' -> "&#39;"
571 else -> it.toString()
572 }
573 }
574 .joinToString("")

576private fun render(
577 tags: List<Tag>,
578 start: Int,
579 end: Int,
580 contexts: List<Value>,
581 loops: List<Pair<Int, Int>>,
582 partials: Map<String, List<SourceLine>>,
583 strict: Boolean,
584 depth: Int,
585): String = buildString {
586 var at = start
587 while (at < end) {
588 val tag = tags[at]
589 try {
590 when (tag.kind) {
591 't' -> append(tag.text)
592 'v',
593 '&',
594 '#',
595 '^' -> {
596 val expr = tag.expr!!
597 val found = resolve(expr.name, contexts, loops)
598 if (strict && tag.kind in "v&" && found == null)
599 throw Problem("missing '${expr.name}'")
600 var value: Value = found ?: Absent
601 for (filter in expr.filters) value = applyFilter(value, filter)
602 if (tag.kind in "v&") {
603 val text = stringify(value)
604 append(if (tag.kind == 'v') escape(text) else text)
605 } else {
606 if (tag.kind == '^') {
607 if (!truthy(value))
608 append(
609 render(
610 tags,
611 at + 1,
612 tag.end,
613 contexts,
614 loops,
615 partials,
616 strict,
617 depth,
618 ),
619 )
620 } else if (value is ListValue) {
621 for ((i, item) in value.values.withIndex()) append(
622 render(
623 tags,
624 at + 1,
625 tag.end,
626 contexts + item,
627 loops + (i to value.values.size),
628 partials,
629 strict,
630 depth,
631 ),
632 )
633 } else if (truthy(value))
634 append(
635 render(
636 tags,
637 at + 1,
638 tag.end,
639 contexts + value,
640 loops,
641 partials,
642 strict,
643 depth,
644 ),
645 )
646 at = tag.end
647 }
648 }
649 '>' -> {
650 val source =
651 partials[tag.text] ?: throw Problem("unknown partial '${tag.text}'")
652 if (depth == 8) throw Problem("partial depth exceeded at '${tag.text}'")
653 val parsed =
654 analyze(
655 source.map {
656 SourceLine(
657 it.line,
658 if (it.text.isEmpty()) "" else tag.indent + it.text,
659 )
660 },
661 )
662 append(
663 render(
664 parsed,
665 0,
666 parsed.size,
667 contexts,
668 loops,
669 partials,
670 strict,
671 depth + 1,
672 ),
673 )
674 }
675 }
676 } catch (p: Problem) {
677 throw Problem(p.reason, if (p.line == 0) tag.line else p.line)
678 }
679 at++
680 }
681}

683private fun dump(v: Value, path: String, leaves: MutableList<String>) {
684 when {
685 v is MapValue && v.values.isNotEmpty() ->
686 for ((key, value) in v.values.toSortedMap()) dump(
687 value,
688 if (path.isEmpty()) key else "$path.$key",
689 leaves,
690 )
691 v is ListValue && v.values.isNotEmpty() ->
692 for ((i, value) in v.values.withIndex()) dump(value, "$path[$i]", leaves)
693 path.isNotEmpty() -> {
694 val notation =
695 when (v) {
696 is Text ->
697 "\"" +
698 v.value
699 .replace("\\", "\\\\")
700 .replace("\"", "\\\"")
701 .replace("\n", "\\n") +
702 "\""
703 is MapValue -> "{}"
704 is ListValue -> "[]"
705 else -> stringify(v)
706 }
707 leaves.add("$path = $notation")
708 }
709 }
710}

734fun main() {
735 val lines = generateSequence(::readLine).toList()
736 var at = 0
737 var root: Value = MapValue(emptyMap())
738 val partials = mutableMapOf<String, List<SourceLine>>()
739 val templates = mutableMapOf<String, List<SourceLine>>()
740 var strict = false
741 while (at < lines.size) {
742 val command = lines[at]
743 val number = ++at
744 if (command == "END") break
745 if (command == "DATA") {
746 var set = 0
747 var unset = 0
748 val errors = mutableListOf<String>()
749 while (lines[at] != "ENDDATA") {
750 val line = lines[at]
751 val lineNumber = ++at
752 if (line.isEmpty() || line.startsWith('#')) continue
753 try {
754 val deletion = line.startsWith("UNSET ")
755 val pos = if (deletion) -1 else line.indexOf(" = ")
756 if (!deletion && pos < 0) throw Problem("bad line")
757 val key = if (deletion) line.substring(6) else line.substring(0, pos)
758 val parts = keyParts(key) ?: throw Problem("bad key '$key'")
759 if (deletion) {
760 root = remove(root, parts, 0) ?: throw Problem("no value at '$key'")
761 unset++
762 } else {
763 val notation = line.substring(pos + 3)
764 val value =
765 ValueParser(notation).parse() ?: throw Problem("bad value '$notation'")
766 root = assign(root, parts, 0, value)
767 set++
768 }
769 } catch (p: Problem) {
770 errors.add("ERROR line $lineNumber: ${p.reason}")
771 }
772 }
773 at++
774 println("DATA set=$set unset=$unset errors=${errors.size}")
775 errors.forEach(::println)
776 } else if (command.startsWith("PARTIAL ") || command.startsWith("TEMPLATE ")) {
777 val partial = command.startsWith("PARTIAL ")
778 val name = command.substringAfter(' ')
779 val end = if (partial) "ENDPARTIAL" else "ENDTEMPLATE"
780 val source = mutableListOf<SourceLine>()
781 while (lines[at] != end) {
782 source.add(SourceLine(at + 1, lines[at]))
783 at++
784 }
785 at++
786 if (partial) {
787 try {
788 analyze(source)
789 println(
790 "PARTIAL $name ${if (name in partials) "REPLACED" else "DEFINED"} lines=${source.size}",
791 )
792 partials[name] = source
793 } catch (p: Problem) {
794 println("PARTIAL $name ERROR line ${p.line}: ${p.reason}")
795 }
796 } else {
797 templates[name] = source
798 outputTemplate(name, source, root, partials, strict)
799 }
800 } else if (command.startsWith("RENDER ")) {
801 val name = command.substring(7)
802 val source = templates[name]
803 if (source != null) outputTemplate(name, source, root, partials, strict)
804 else println("TEMPLATE $name ERROR line $number: unknown template '$name'")
805 } else if (command == "DUMP") {
806 val leaves = mutableListOf<String>()
807 dump(root, "", leaves)
808 println("DUMP leaves=${leaves.size}")
809 leaves.forEach(::println)
810 } else if (command == "RESET") {
811 root = MapValue(emptyMap())
812 partials.clear()
813 templates.clear()
814 println("RESET")
815 } else {
816 strict = command == "STRICT ON"
817 println(command)
818 }
819 }
