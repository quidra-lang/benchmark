3const identifier = /^[A-Za-z_][A-Za-z0-9_]*$/;
4const keyPattern = /^[A-Za-z_][A-Za-z0-9_]*(?:\.[A-Za-z_][A-Za-z0-9_]*|\[(?:0|[1-9][0-9]*)\])*$/;
5const namePattern = /^(?:\.|@(?:index|number|first|last)|(?:[A-Za-z_][A-Za-z0-9_]*|0|[1-9][0-9]*)(?:\.(?:[A-Za-z_][A-Za-z0-9_]*|0|[1-9][0-9]*))*)$/;
6const integerPattern = /^(?:0|-?[1-9][0-9]*)$/;
7const numericPattern = /^-?[0-9]{1,9}$/;
8const noArguments = new Set(['upper', 'lower', 'title', 'trim', 'length', 'first', 'last', 'reverse', 'items', 'comma', 'sum', 'min', 'max']);
9const numericFilters = new Set(['add', 'sub', 'mul', 'div', 'mod', 'gt', 'lt']);
10const filters = new Set([...noArguments, ...numericFilters, 'default', 'eq', 'pad', 'lpad', 'truncate', 'replace', 'join', 'sort', 'pluck']);
11const standaloneKinds = new Set(['section', 'inverted', 'close', 'comment', 'partial', 'delimiter']);
12const sigils = { '#': 'section', '^': 'inverted', '/': 'close', '!': 'comment', '>': 'partial', '&': 'raw', '=': 'delimiter' };
13function strip(text) {
14 let start = 0;
15 let end = text.length;
16 while (start < end && text[start] === ' ') start++;
17 while (end > start && text[end - 1] === ' ') end--;
18 return text.slice(start, end);
19}

40function hasPart(value, part) {
41 return typeof part === 'string' ? value instanceof Map && value.has(part) : Array.isArray(value) && part < value.length;
42}

44function getPart(value, part) {
45 return typeof part === 'string' ? value.get(part) : value[part];
46}

48function putPart(container, part, value) {
49 if (typeof part === 'string') container.set(part, value);
50 else container[part] = value;
51}

53function parseKey(key) {
54 if (!keyPattern.test(key)) throw new Error(`bad key '${key}'`);
55 return [...key.matchAll(/([A-Za-z_][A-Za-z0-9_]*)|\[(0|[1-9][0-9]*)\]/g)].map(match => ({
56 part: match[1] === undefined ? Number(match[2]) : match[1],
57 prefix: key.slice(0, match.index + match[0].length),
58 }));
59}

61function readScalar(text, start) {
62 if (text[start] === '"') {
63 let result = '';
64 for (let position = start + 1; position < text.length; position++) {
65 const char = text[position];
66 if (char === '"') return [result, position + 1];
67 if (char === '\\') {
68 const next = text[++position];
69 if (next !== '"' && next !== '\\' && next !== 'n') throw new Error();
70 result += next === 'n' ? '\n' : next;
71 } else result += char;
72 }
73 throw new Error();
74 }
75 let end = start;
76 while (end < text.length && text[end] !== ',' && text[end] !== ']') end++;
77 const word = text.slice(start, end);
78 if (word === 'true' || word === 'false') return [word === 'true', end];
79 if (integerPattern.test(word)) return [Number(word), end];
80 throw new Error();
81}

83function parseValue(text) {
84 try {
85 if (text === '[]') return [];
86 if (text === '{}') return new Map();
87 if (text.startsWith('[')) {
88 const values = [];
89 let position = 1;
90 while (true) {
91 const [value, end] = readScalar(text, position);
92 values.push(value);
93 if (text.slice(end) === ']') return values;
94 if (text.slice(end, end + 2) !== ', ') throw new Error();
95 position = end + 2;
96 }
97 }
98 const [value, end] = readScalar(text, 0);
99 if (end !== text.length) throw new Error();
100 return value;
101 } catch {
102 throw new Error(`bad value '${text}'`);
103 }
104}

106function splitUnquoted(text, separator) {
107 const parts = [];
108 let quoted = false;
109 let start = 0;
110 for (let index = 0; index < text.length; index++) {
111 if (text[index] === "'") quoted = !quoted;
112 else if (text[index] === separator && !quoted) {
113 parts.push(text.slice(start, index));
114 start = index + 1;
115 }
116 }
117 parts.push(text.slice(start));
118 return parts;
119}

121function validArguments(name, args) {
122 const number = args.length > 0 && typeof args[0] === 'number';
123 const string = args.length > 0 && typeof args[0] === 'string';
124 if (noArguments.has(name)) return args.length === 0;
125 if (name === 'default' || name === 'eq') return args.length === 1;
126 if (name === 'sort') return args.length === 0 || args.length === 1 && string;
127 if (name === 'join' || name === 'pluck') return args.length === 1 && string;
128 if (name === 'replace') return args.length === 2 && string && args[0] !== '' && typeof args[1] === 'string';
129 if (name === 'pad' || name === 'lpad') {
130 return number && args[0] >= 0 && (args.length === 1 || args.length === 2 && typeof args[1] === 'string' && args[1].length === 1);
131 }
132 if (name === 'truncate') return args.length === 1 && number && args[0] >= 0;
133 return args.length === 1 && number && (!(name === 'div' || name === 'mod') || args[0] !== 0);
134}

136function parseExpression(text, number) {
137 const parts = splitUnquoted(text, '|');
138 const name = strip(parts[0]);
139 if (!namePattern.test(name)) throw new TemplateError(number, `invalid name '${name}'`);
140 const chain = [];
141 for (const part of parts.slice(1)) {
142 const [filter, ...notations] = splitUnquoted(strip(part), ':');
143 if (!filters.has(filter)) throw new TemplateError(number, `unknown filter '${filter}'`);
144 const args = [];
145 let valid = true;
146 for (const notation of notations) {
147 if (/^'[^']*'$/.test(notation)) args.push(notation.slice(1, -1));
148 else if (numericPattern.test(notation)) args.push(Number(notation));
149 else valid = false;
150 }
151 if (!valid || !validArguments(filter, args)) throw new TemplateError(number, `bad arguments for '${filter}'`);
152 chain.push({ name: filter, args });
153 }
154 return { name, filters: chain };
155}

157function parseSource(source) {
158 let opening = '{{';
159 let closing = '}}';
160 const nodes = [];
161 const targets = [nodes];
162 const sections = [];
163 for (const { number, text } of source) {
164 let position = 0;
165 const tokens = [];
166 const lineTags = [];
167 while (true) {
168 const start = text.indexOf(opening, position);
169 if (start < 0) {
170 tokens.push(text.slice(position));
171 break;
172 }
173 tokens.push(text.slice(position, start));
174 const after = start + opening.length;
175 const triple = text[after] === '{';
176 const terminator = triple ? '}' + closing : closing;
177 const end = text.indexOf(terminator, after + (triple ? 1 : 0));
178 if (end < 0) throw new TemplateError(number, 'unclosed tag');
179 const content = strip(text.slice(after + (triple ? 1 : 0), end));
180 const kind = triple ? 'raw' : sigils[content[0]] ?? 'variable';
181 const argument = triple || kind === 'variable' || kind === 'delimiter' ? content : strip(content.slice(1));
182 const tag = new Tag(kind, number, argument);
183 position = end + terminator.length;
184 if (kind === 'delimiter') {
185 const words = strip(content.slice(1, -1)).split(/ +/);
186 if (content.length < 2 || !content.endsWith('=') || words.length !== 2 || words.some(word => word.includes('='))) {
187 throw new TemplateError(number, 'invalid delimiters');
188 }
189 [opening, closing] = words;
190 } else if (kind === 'variable' || kind === 'raw' || kind === 'section' || kind === 'inverted') {
191 tag.expression = parseExpression(argument, number);
192 if (kind === 'section' || kind === 'inverted') sections.push(tag);
193 } else if (kind === 'close') {
194 if (!namePattern.test(argument)) throw new TemplateError(number, `invalid name '${argument}'`);
195 if (sections.length === 0) throw new TemplateError(number, `unexpected closing '${argument}'`);
196 const expected = sections[sections.length - 1].expression.name;
197 if (argument !== expected) throw new TemplateError(number, `mismatched closing '${argument}', expected '${expected}'`);
198 sections.pop();
199 } else if (kind === 'partial' && !identifier.test(argument)) {
200 throw new TemplateError(number, `invalid name '${argument}'`);
201 }
202 lineTags.push(tag);
203 tokens.push(tag);
204 }
205 const standalone = lineTags.length === 1 && standaloneKinds.has(lineTags[0].kind) && tokens.every(token => typeof token !== 'string' || /^ *$/.test(token));
206 let placed = tokens;
207 if (standalone) {
208 if (lineTags[0].kind === 'partial') lineTags[0].indentation = tokens[0];
209 placed = lineTags;
210 } else placed.push('\n');
211 for (const token of placed) {
212 const target = targets[targets.length - 1];
213 if (typeof token === 'string') {
214 if (token !== '') target.push(token);
215 } else if (token.kind === 'section' || token.kind === 'inverted') {
216 target.push(token);
217 targets.push(token.children);
218 } else if (token.kind === 'close') targets.pop();
219 else if (token.kind !== 'comment' && token.kind !== 'delimiter') target.push(token);
220 }
221 }
222 if (sections.length > 0) {
223 const tag = sections[sections.length - 1];
224 throw new TemplateError(tag.number, `unclosed section '${tag.expression.name}'`);
225 }
226 return nodes;
227}

229function toText(value) {
230 if (value === null) return '';
231 if (value instanceof Map) throw new Error('cannot render map');
232 if (Array.isArray(value)) return value.map(toText).join(', ');
233 return String(value);
234}

236function truthy(value) {
237 if (value instanceof Map) return value.size > 0;
238 if (Array.isArray(value)) return value.length > 0;
239 return value !== null && value !== false && value !== 0 && value !== '';
240}

242function toNumber(value, filter) {
243 if (value === null) return 0;
244 if (typeof value === 'number') return value;
245 if (typeof value === 'string' && !value.includes('\n') && numericPattern.test(value)) return Number(value);
246 throw new Error(`not a number for '${filter}'`);
247}

249function field(value, key) {
250 return value instanceof Map ? value.get(key) ?? null : null;
251}

253function sortRank(value) {
254 if (value === null) return 0;
255 if (typeof value === 'number') return 1;
256 if (typeof value === 'string') return 2;
257 if (typeof value === 'boolean') return 3;
258 return 4;
259}
260
261function compareValues(left, right) {
262 const rank = sortRank(left) - sortRank(right);
263 if (rank !== 0) return rank;
264 if (sortRank(left) === 0 || sortRank(left) === 4) return 0;
265 if (left === right) return 0;
266 if (typeof left === 'number' && typeof right === 'number') return left - right;
267 if (typeof left === 'string' && typeof right === 'string') return left < right ? -1 : 1;
268 return Number(left) - Number(right);
269}

271function applyFilter(filter, value) {
272 const { name, args } = filter;
273 const n = args[0];
274 const s = args[0];
275 switch (name) {
276 case 'upper': return toText(value).toUpperCase();
277 case 'lower': return toText(value).toLowerCase();
278 case 'title': return [...toText(value)].map((char, index, chars) => index === 0 || chars[index - 1] === ' ' ? char.toUpperCase() : char.toLowerCase()).join('');
279 case 'trim': return strip(toText(value));
280 case 'length': return value instanceof Map ? value.size : Array.isArray(value) ? value.length : toText(value).length;
281 case 'first': return Array.isArray(value) ? value[0] ?? null : value;
282 case 'last': return Array.isArray(value) ? value[value.length - 1] ?? null : value;
283 case 'reverse': return Array.isArray(value) ? [...value].reverse() : typeof value === 'string' ? [...value].reverse().join('') : value;
284 case 'items': return value instanceof Map ? [...value.keys()].sort().map(key => new Map([['key', key], ['value', value.get(key)]])) : value;
285 case 'comma': {
286 const number = toNumber(value, name);
287 return (number < 0 ? '-' : '') + String(Math.abs(number)).replace(/\B(?=(\d{3})+(?!\d))/g, ',');
288 }
289 case 'sum': case 'min': case 'max': {
290 if (!Array.isArray(value)) return toNumber(value, name);
291 const numbers = value.map(element => toNumber(element, name));
292 if (name === 'sum') return numbers.reduce((total, number) => total + number, 0);
293 if (numbers.length === 0) return null;
294 return numbers.reduce((best, number) => name === 'min' ? Math.min(best, number) : Math.max(best, number));
295 }
296 case 'default': return truthy(value) ? value : args[0];
297 case 'eq': return toText(value) === toText(args[0]);
298 case 'pad': return toText(value).padEnd(n, args.length === 2 ? args[1] : ' ');
299 case 'lpad': return toText(value).padStart(n, args.length === 2 ? args[1] : ' ');
300 case 'truncate': {
301 const text = toText(value);
302 return text.length > n ? text.slice(0, n) + '...' : text;
303 }
304 case 'replace': return toText(value).split(s).join(args[1]);
305 case 'join': return Array.isArray(value) ? value.map(toText).join(s) : toText(value);
306 case 'pluck': return Array.isArray(value) ? value.map(element => field(element, s)) : value;
307 case 'sort': return Array.isArray(value) ? value.map((element, index) => ({ element, index, key: args.length === 0 ? element : field(element, s) })).sort((a, b) => compareValues(a.key, b.key) || a.index - b.index).map(item => item.element) : value;
308 default: {
309 const number = toNumber(value, name);
310 switch (name) {
311 case 'add': return number + n;
312 case 'sub': return number - n;
313 case 'mul': return number * n;
314 case 'div': return Math.floor(number / n);
315 case 'mod': return number - n * Math.floor(number / n);
316 case 'gt': return number > n;
317 default: return number < n;
318 }
319 }
320 }

323class Renderer {
324 contexts;
325 loops = [];
326 output = [];
327
328 constructor(root, partials, strict) {
329 this.partials = partials;
330 this.strict = strict;
331 this.contexts = [root];
332 }
333
334 resolve(name) {
335 if (name === '.') return [true, this.contexts[this.contexts.length - 1]];
336 if (name.startsWith('@')) {
337 if (this.loops.length === 0) return [false, null];
338 const { index, count } = this.loops[this.loops.length - 1];
339 if (name === '@index') return [true, index];
340 if (name === '@number') return [true, index + 1];
341 return [true, name === '@first' ? index === 0 : index === count - 1];
342 }
343 const parts = name.split('.').map(part => /^[0-9]+$/.test(part) ? Number(part) : part);
344 for (let index = this.contexts.length - 1; index >= 0; index--) {
345 let value = this.contexts[index];
346 if (!hasPart(value, parts[0])) continue;
347 for (const part of parts) {
348 if (!hasPart(value, part)) return [false, null];
349 value = getPart(value, part);
350 }
351 return [true, value];
352 }
353 return [false, null];
354 }
355
356 evaluate(expression, strict) {
357 let [found, value] = this.resolve(expression.name);
358 if (!found && strict) throw new Error(`missing '${expression.name}'`);
359 for (const filter of expression.filters) value = applyFilter(filter, value);
360 return value;
361 }
362
363 render(nodes, depth) {
364 for (const node of nodes) {
365 if (typeof node === 'string') {
366 this.output.push(node);
367 continue;
368 }
369 try {
370 if (node.kind === 'variable' || node.kind === 'raw') {
371 const text = toText(this.evaluate(node.expression, this.strict));
372 const escaped = { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' };
373 this.output.push(node.kind === 'raw' ? text : text.replace(/[&<>"']/g, char => escaped[char]));
374 } else if (node.kind === 'section') {
375 const value = this.evaluate(node.expression, false);
376 if (Array.isArray(value)) {
377 for (let index = 0; index < value.length; index++) {
378 this.contexts.push(value[index]);
379 this.loops.push({ index, count: value.length });
380 this.render(node.children, depth);
381 this.loops.pop();
382 this.contexts.pop();
383 }
384 } else if (truthy(value)) {
385 this.contexts.push(value);
386 this.render(node.children, depth);
387 this.contexts.pop();
388 }
389 } else if (node.kind === 'inverted') {
390 if (!truthy(this.evaluate(node.expression, false))) this.render(node.children, depth);
391 } else if (node.kind === 'partial') {
392 const source = this.partials.get(node.argument);
393 if (source === undefined) throw new Error(`unknown partial '${node.argument}'`);
394 if (depth === 8) throw new Error(`partial depth exceeded at '${node.argument}'`);
395 this.render(parseSource(source.map(line => ({ number: line.number, text: line.text === '' ? '' : node.indentation + line.text }))), depth + 1);
396 }
397 } catch (error) {
398 if (error instanceof TemplateError) throw error;
399 throw new TemplateError(node.number, error.message);
400 }
401 }
402 }
403}

405class TemplateEngine {
406 root = new Map();
407 partials = new Map();
408 templates = new Map();
409 strict = false;
410 output = [];
411
412 assign(line) {
413 const separator = line.indexOf(' = ');
414 if (separator < 0) throw new Error('bad line');
415 const parts = parseKey(line.slice(0, separator));
416 const value = parseValue(line.slice(separator + 3));
417 let container = this.root;
418 let detached = null;
419 for (let index = 0; index < parts.length; index++) {
420 const { part, prefix } = parts[index];
421 if (typeof part === 'number' && part > container.length) throw new Error(`index gap at '${prefix}'`);
422 if (index === parts.length - 1) {
423 putPart(container, part, value);
424 break;
425 }
426 const mapRequired = typeof parts[index + 1].part === 'string';
427 let child;
428 if (hasPart(container, part)) {
429 child = getPart(container, part);
430 if (!(mapRequired ? child instanceof Map : Array.isArray(child))) throw new Error(`type conflict at '${prefix}'`);
431 } else {
432 child = mapRequired ? new Map() : [];
433 if (detached === null) detached = { container, part, child };
434 else putPart(container, part, child);
435 }
436 container = child;
437 }
438 if (detached !== null) putPart(detached.container, detached.part, detached.child);
439 }
440
441 unset(key) {
442 const parts = parseKey(key);
443 let container = this.root;
444 for (let index = 0; index < parts.length; index++) {
445 const part = parts[index].part;
446 if (!hasPart(container, part)) throw new Error(`no value at '${key}'`);
447 if (index === parts.length - 1) {
448 if (typeof part === 'string') container.delete(part);
449 else container.splice(part, 1);
450 } else container = getPart(container, part);
451 }
452 }
453
454 loadData(body) {
455 let assigned = 0;
456 let unset = 0;
457 const errors = [];
458 for (const { number, text } of body) {
459 if (text === '' || text.startsWith('#')) continue;
460 try {
461 if (text.startsWith('UNSET ')) {
462 this.unset(text.slice(6));
463 unset++;
464 } else {
465 this.assign(text);
466 assigned++;
467 }
468 } catch (error) {
469 errors.push(`ERROR line ${number}: ${error.message}`);
470 }
471 }
472 this.output.push(`DATA set=${assigned} unset=${unset} errors=${errors.length}`, ...errors);
473 }
474
475 render(name) {
476 try {
477 const renderer = new Renderer(this.root, this.partials, this.strict);
478 renderer.render(parseSource(this.templates.get(name)), 0);
479 const result = renderer.output.join('');
480 const lines = result === '' ? [] : result.split('\n');
481 if (result.endsWith('\n')) lines.pop();
482 this.output.push(`TEMPLATE ${name} OK lines=${lines.length} chars=${result.length}`, ...lines);
483 } catch (error) {
484 const failure = error;
485 this.output.push(`TEMPLATE ${name} ERROR line ${failure.number}: ${failure.message}`);
486 }
487 }
488
489 dump() {
490 const leaves = [];
491 const visit = (value, path) => {
492 if (value instanceof Map && value.size > 0) {
493 for (const key of [...value.keys()].sort()) visit(value.get(key), `${path}.${key}`);
494 } else if (Array.isArray(value) && value.length > 0) {
495 value.forEach((element, index) => visit(element, `${path}[${index}]`));
496 } else {
497 const notation = typeof value === 'string' ? '"' + value.replace(/\\/g, '\\\\').replace(/"/g, '\\"').replace(/\n/g, '\\n') + '"' : value instanceof Map ? '{}' : Array.isArray(value) ? '[]' : String(value);
498 leaves.push(`${path} = ${notation}`);
499 }
500 };
501 for (const key of [...this.root.keys()].sort()) visit(this.root.get(key), key);
502 this.output.push(`DUMP leaves=${leaves.length}`, ...leaves);
503 }
504
505 run(lines) {
506 let position = 0;
507 while (lines[position] !== 'END') {
508 const line = lines[position++];
509 const number = position;
510 const [command, name] = line.split(' ');
511 if (command === 'DATA' || command === 'PARTIAL' || command === 'TEMPLATE') {
512 const body = [];
513 while (lines[position] !== `END${command}`) body.push({ number: position + 1, text: lines[position++] });
514 position++;
515 if (command === 'DATA') this.loadData(body);
516 else if (command === 'TEMPLATE') {
517 this.templates.set(name, body);
518 this.render(name);
519 } else {
520 try {
521 parseSource(body);
522 const status = this.partials.has(name) ? 'REPLACED' : 'DEFINED';
523 this.partials.set(name, body);
524 this.output.push(`PARTIAL ${name} ${status} lines=${body.length}`);
525 } catch (error) {
526 const failure = error;
527 this.output.push(`PARTIAL ${name} ERROR line ${failure.number}: ${failure.message}`);
528 }
529 }
530 } else if (command === 'RENDER') {
531 if (this.templates.has(name)) this.render(name);
532 else this.output.push(`TEMPLATE ${name} ERROR line ${number}: unknown template '${name}'`);
533 } else if (command === 'DUMP') this.dump();
534 else if (command === 'STRICT') {
535 this.strict = name === 'ON';
536 this.output.push(line);
537 } else if (command === 'RESET') {
538 this.root.clear();
539 this.partials.clear();
540 this.templates.clear();
541 this.output.push('RESET');
542 }
543 }
544 }
545}

548engine.run(fs.readFileSync(0, 'utf8').split('\n'));
549process.stdout.write(engine.output.length === 0 ? '' : engine.output.join('\n') + '\n');
