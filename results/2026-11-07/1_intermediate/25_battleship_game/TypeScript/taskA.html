61const advance = (state: number): number => 48271 * state % 2147483647;
62const otherPlayer = (player: PlayerIndex): PlayerIndex => player === 0 ? 1 : 0;
63const coordinate = (cell: number): string => String.fromCharCode(65 + Math.floor(cell / 10)) + (cell % 10 + 1);
64const parseCell = (word: string): number => (word.charCodeAt(0) - 65) * 10 + Number(word.slice(1)) - 1;
65const shipRange = (cells: readonly number[]): string => `${coordinate(cells[0])}-${coordinate(cells[cells.length - 1])}`;
66const touching = (a: number, b: number): boolean => Math.abs(Math.floor(a / 10) - Math.floor(b / 10)) <= 1 && Math.abs(a % 10 - b % 10) <= 1;
67const emptyFleet = (): Fleet => [[], [], [], [], []];
68const isHit = (shot: Shot): boolean => shot.kind === 'HIT' || shot.kind === 'SUNK';
69const shotText = (shot: Shot): string => shot.kind === 'SUNK' ? `SUNK ${ships[shot.ship]}` : shot.kind;

74function parseCommand(words: readonly string[]): Command {
75 const kind = words[0] as Command['kind'];
76 const player = (): PlayerIndex => (Number(words[1]) - 1) as PlayerIndex;
77 const ship = (): ShipIndex => ships.indexOf(words[2] as ShipName) as ShipIndex;
78 const pair = (): readonly [Strategy, Strategy] => [words[3] as Strategy, words[4] as Strategy];
79 switch (kind) {
80 case 'PLACE': return { kind, player: player(), ship: ship(), cell: parseCell(words[3]), direction: words[4] as Direction };
81 case 'REMOVE': return { kind, player: player(), ship: ship() };
82 case 'AUTOPLACE': case 'SEED': return { kind, player: player(), seed: Number(words[2]) };
83 case 'LAYOUT': return { kind, player: player(), layout: words[2] };
84 case 'REEF': return { kind, cell: parseCell(words[1]) };
85 case 'SALVO': case 'CHAIN': case 'TOUCH': return { kind, enabled: words[1] === 'ON' };
86 case 'FIRE': return { kind, player: player(), cells: words.slice(2).map(parseCell) };
87 case 'AI': case 'HINT': return { kind, player: player(), strategy: words[2] as Strategy };
88 case 'TORPEDO': return { kind, player: player(), row: words[2].charCodeAt(0) - 65, side: words[3] as Side };
89 case 'SONAR': return { kind, player: player(), cell: parseCell(words[2]) };
90 case 'RESIGN': case 'DENSITY': case 'SHOW': case 'RADAR': case 'EXPORT': case 'FLEET': return { kind, player: player() };
91 case 'CONTROL': return { kind, player: player(), controller: words[2] as Controller };
92 case 'MATCH': return { kind, seeds: [Number(words[1]), Number(words[2])], strategies: pair() };
93 case 'SERIES': return { kind, count: Number(words[1]), seed: Number(words[2]), strategies: pair() };
94 case 'TOURNEY': return { kind, count: Number(words[1]), seed: Number(words[2]) };
95 case 'NEW': case 'STATUS': case 'STATS': return { kind };
96 }
97}

99function step(cell: number, dr: number, dc: number): number {
100 const row = Math.floor(cell / 10) + dr;
101 const column = cell % 10 + dc;
102 return row >= 0 && row < 10 && column >= 0 && column < 10 ? row * 10 + column : -1;
103}

105function footprint(start: number, direction: Direction, length: number): readonly number[] | null {
106 const position = direction === 'H' ? start % 10 : Math.floor(start / 10);
107 if (position + length > 10) return null;
108 return Array.from({ length }, (_, index) => start + index * (direction === 'H' ? 1 : 10));
109}

113 readonly attacked: boolean[] = Array<boolean>(100).fill(false);
114 readonly candidates: number[] = [];

127 shipAt(cell: number): ShipIndex | null { return shipIndices.find(ship => this.fleet[ship].includes(cell)) ?? null; }
128 sunk(ship: ShipIndex): boolean { return this.fleet[ship].length !== 0 && this.fleet[ship].every(cell => this.attacked[cell]); }
129 get placed(): number { return this.fleet.filter(cells => cells.length !== 0).length; }
130 get afloat(): number { return shipIndices.filter(ship => this.fleet[ship].length !== 0 && !this.sunk(ship)).length; }
131 get defeated(): boolean { return shipIndices.every(ship => this.sunk(ship)); }

150 placementProblem(player: Player, cells: readonly number[] | null): string | null {
151 if (cells === null) return 'out of bounds';
152 if (cells.some(cell => this.reefs.has(cell))) return 'on reef';
153 for (const ship of shipIndices) {
154 if (cells.some(cell => player.fleet[ship].includes(cell))) return `overlaps ${ships[ship]}`;
155 }
156 if (!this.touch) {
157 for (const ship of shipIndices) {
158 if (cells.some(cell => player.fleet[ship].some(other => touching(cell, other)))) return `touches ${ships[ship]}`;
159 }
160 }
161 return null;
162 }

165 return shipIndices.map(ship => `PLACED ${p + 1} ${ships[ship]} ${shipRange(this.players[p].fleet[ship])}`);

168 autoPlace(p: PlayerIndex, seed: number): number {
169 const player = this.players[p];
170 player.random = seed;
171 player.fleet = emptyFleet();
172 let attempts = 0;
173 for (const ship of shipIndices) {
174 while (player.fleet[ship].length === 0) {
175 attempts++;
176 player.random = advance(player.random);
177 const direction: Direction = player.random % 2 === 0 ? 'H' : 'V';
178 player.random = advance(player.random);
179 const span = 11 - lengths[ship];
180 const index = player.random % (10 * span);
181 const start = direction === 'H' ? Math.floor(index / span) * 10 + index % span : index;
182 const cells = footprint(start, direction, lengths[ship]);
183 if (this.placementProblem(player, cells) === null) player.fleet[ship] = cells!;
184 }
185 }
186 return attempts;
187 }

189 view(p: PlayerIndex): CellView[] {
190 const opponent = this.players[otherPlayer(p)];
191 return Array.from({ length: 100 }, (_, cell): CellView => {
192 if (this.reefs.has(cell)) return '~';
193 if (!opponent.attacked[cell]) return '.';
194 const ship = opponent.shipAt(cell);
195 return ship === null ? 'o' : opponent.sunk(ship) ? '#' : 'x';
196 });
197 }
198
199 excluded(view: readonly CellView[]): boolean[] {
200 const result = Array<boolean>(100).fill(false);
201 if (!this.touch) {
202 for (let cell = 0; cell < 100; cell++) {
203 if (view[cell] !== '#') continue;
204 for (let dr = -1; dr <= 1; dr++) {
205 for (let dc = -1; dc <= 1; dc++) {
206 if (dr === 0 && dc === 0) continue;
207 const next = step(cell, dr, dc);
208 if (next !== -1) result[next] = true;
209 }
210 }
211 }
212 }
213 return result;
214 }

216 density(p: PlayerIndex, view: readonly CellView[], excluded: readonly boolean[]): number[] {
217 const scores = Array<number>(100).fill(0);
218 for (const ship of shipIndices) {
219 if (this.players[otherPlayer(p)].sunk(ship)) continue;
220 for (let start = 0; start < 100; start++) {
221 for (const direction of ['H', 'V'] as const) {
222 const cells = footprint(start, direction, lengths[ship]);
223 if (cells === null || cells.some(cell => excluded[cell] || (view[cell] !== '.' && view[cell] !== 'x'))) continue;
224 const weight = 1 + 10 * cells.filter(cell => view[cell] === 'x').length;
225 for (const cell of cells) scores[cell] += weight;
226 }
227 }
228 }
229 return scores;
230 }

232 choose(p: PlayerIndex, strategy: Strategy): Choice {
233 const player = this.players[p];
234 const view = this.view(p);
235 const excluded = this.excluded(view);
236 const candidate = (cell: number): boolean => cell >= 0 && cell < 100 && view[cell] === '.' && !excluded[cell];
237 if (strategy === 'PROB') {
238 const scores = this.density(p, view, excluded);
239 let chosen = -1;
240 for (let cell = 0; cell < 100; cell++) {
241 if (view[cell] === '.' && (chosen === -1 || scores[cell] > scores[chosen])) chosen = cell;
242 }
243 return { cell: chosen, info: scores[chosen], nextRandom: player.random };
244 }
245 if (strategy === 'RANDOM') {
246 const cells = Array.from({ length: 100 }, (_, cell) => cell).filter(candidate);
247 const random = advance(player.random);
248 return { cell: cells[random % cells.length], info: cells.length, nextRandom: random };
249 }
250 for (let cell = 0; cell < 100; cell++) {
251 if (view[cell] !== 'x') continue;
252 for (const [dr, dc] of [[0, 1], [1, 0]]) {
253 let front = step(cell, dr, dc);
254 if (front === -1 || view[front] !== 'x') continue;
255 while (front !== -1 && view[front] === 'x') front = step(front, dr, dc);
256 const back = step(cell, -dr, -dc);
257 if (candidate(front)) return { cell: front, info: 'LINE', nextRandom: player.random };
258 if (candidate(back)) return { cell: back, info: 'LINE', nextRandom: player.random };
259 }
260 }
261 for (let i = player.candidates.length - 1; i >= 0; i--) {
262 const cell = player.candidates[i];
263 if (candidate(cell)) return { cell, info: 'STACK', nextRandom: player.random };
264 }
265 const shortest = Math.min(...shipIndices.filter(ship => !this.players[otherPlayer(p)].sunk(ship)).map(ship => lengths[ship]));
266 for (let cell = 0; cell < 100; cell++) {
267 if (candidate(cell) && (Math.floor(cell / 10) + cell % 10) % shortest === 0) {
268 return { cell, info: 'PARITY', nextRandom: player.random };
269 }
270 }
271 throw new Error('No strategy candidate');
272 }

274 shoot(p: PlayerIndex, cell: number): Shot {
275 const player = this.players[p];
276 const opponent = this.players[otherPlayer(p)];
277 let shot: Shot = { kind: 'ALREADY' };
278 if (!opponent.attacked[cell]) {
279 opponent.attacked[cell] = true;
280 const ship = opponent.shipAt(cell);
281 shot = ship === null ? { kind: 'MISS' } : opponent.sunk(ship) ? { kind: 'SUNK', ship } : { kind: 'HIT' };
282 }
283 if (shot.kind === 'HIT') {
284 for (const [dr, dc] of [[0, -1], [1, 0], [0, 1], [-1, 0]]) {
285 const next = step(cell, dr, dc);
286 if (next !== -1) player.candidates.push(next);
287 }
288 }
289 player.shots++;
290 if (isHit(shot)) {
291 player.hits++;
292 player.streak++;
293 player.bestStreak = Math.max(player.bestStreak, player.streak);
294 } else player.streak = 0;
295 if (shot.kind === 'SUNK') player.sinks++;
296 return shot;
297 }

299 allowance(p: PlayerIndex): number { return this.salvo ? this.players[p].afloat : 1; }

301 finishTurn(p: PlayerIndex, hit: boolean, output: string[], mode: OutputMode): void {
302 this.players[p].turns++;
303 if (this.players[otherPlayer(p)].defeated) {
304 this.phase = 'OVER';
305 this.winner = p;
306 this.players[p].wins++;
307 if (mode === 'normal') output.push(`WINNER ${p + 1}`);
308 } else this.current = this.chain && hit ? p : otherPlayer(p);
309 }
310
311 shootingTurn(p: PlayerIndex, strategy: Strategy | null, cells: readonly number[], output: string[], mode: OutputMode = 'normal'): void {
312 this.phase = 'BATTLE';
313 const count = this.allowance(p);
314 let hit = false;
315 for (let i = 0; i < count; i++) {
316 let cell: number;
317 if (strategy === null) cell = cells[i];
318 else {
319 const choice = this.choose(p, strategy);
320 cell = choice.cell;
321 this.players[p].random = choice.nextRandom;
322 }
323 const shot = this.shoot(p, cell);
324 hit ||= isHit(shot);
325 if (mode === 'normal') output.push(`SHOT ${p + 1} ${coordinate(cell)} ${shotText(shot)}`);
326 if (mode === 'match' && shot.kind === 'SUNK') output.push(`SINK ${p + 1} ${ships[shot.ship]} ${this.players[p].shots}`);
327 if (this.players[otherPlayer(p)].defeated) break;
328 }
329 this.finishTurn(p, hit, output, mode);
330 }

332 checkTurn(p: PlayerIndex): void {
333 if (this.phase === 'OVER') fail('game over');
334 if (this.phase === 'SETUP' && this.players.some(player => player.placed !== 5)) fail('fleets not ready');
335 if (p !== this.current) fail('not your turn');

338 specialTurn(command: SpecialCommand, output: string[]): void {
339 this.phase = 'BATTLE';
340 const p = command.player;
341 const player = this.players[p];
342 const opponent = this.players[otherPlayer(p)];
343 let hit = false;
344 if (command.kind === 'SONAR') {
345 player.sonar = true;
346 let count = 0;
347 for (let dr = -1; dr <= 1; dr++) {
348 for (let dc = -1; dc <= 1; dc++) {
349 const cell = step(command.cell, dr, dc);
350 if (cell !== -1 && !opponent.attacked[cell] && opponent.shipAt(cell) !== null) count++;
351 }
352 }
353 output.push(`SONAR ${p + 1} ${coordinate(command.cell)} ${count}`);
354 } else {
355 player.torpedo = true;
356 let result = 'MISS';
357 let shot: Shot | null = null;
358 for (let index = 0; index < 10; index++) {
359 const cell = command.row * 10 + (command.side === 'W' ? index : 9 - index);
360 if (this.reefs.has(cell)) {
361 result = `REEF ${coordinate(cell)}`;
362 break;
363 }
364 if (opponent.shipAt(cell) !== null) {
365 shot = this.shoot(p, cell);
366 result = `${shotText(shot)} ${coordinate(cell)}`;
367 hit = isHit(shot);
368 break;
369 }
370 opponent.attacked[cell] = true;
371 }
372 if (shot === null) {
373 player.shots++;
374 player.streak = 0;
375 }
376 output.push(`TORPEDO ${p + 1} ${String.fromCharCode(65 + command.row)} ${command.side} ${result}`);
377 }
378 this.finishTurn(p, hit, output, 'normal');
379 }

385 if (controller === 'HUMAN') break;

390 match(seed1: number, seed2: number, methods: readonly [Strategy, Strategy], output: string[] | null = null): MatchResult {
391 const game = new Game();
392 game.salvo = this.salvo;
393 game.chain = this.chain;
394 game.touch = this.touch;
395 game.reefs = new Set(this.reefs);
396 game.autoPlace(0, seed1);
397 game.autoPlace(1, seed2);
398 game.phase = 'BATTLE';
399 const lines = output ?? [];
400 while (!game.finished) {
401 game.shootingTurn(game.current, methods[game.current], [], lines, output === null ? 'quiet' : 'match');
402 }
403 return { winner: game.winner!, shots: [game.players[0].shots, game.players[1].shots], hits: [game.players[0].hits, game.players[1].hits] };
404 }

406 layout(p: PlayerIndex, text: string): string[] {
407 const grid = text.replaceAll('/', '');
408 const fleet = emptyFleet();
409 for (const ship of shipIndices) {
410 const cells = Array.from({ length: 100 }, (_, cell) => cell).filter(cell => grid[cell] === symbols[ship]);
411 if (cells.length !== lengths[ship]) fail(`${ships[ship]} has ${cells.length} cells`);
412 const horizontal = cells.every((cell, index) => cell === cells[0] + index && Math.floor(cell / 10) === Math.floor(cells[0] / 10));
413 const vertical = cells.every((cell, index) => cell === cells[0] + 10 * index);
414 if (!horizontal && !vertical) fail(`${ships[ship]} not straight`);
415 if (cells.some(cell => this.reefs.has(cell))) fail(`${ships[ship]} on reef`);
416 fleet[ship] = cells;
417 }
418 if (!this.touch) {
419 for (const first of shipIndices) {
420 for (const second of shipIndices) {
421 if (second <= first) continue;
422 if (fleet[first].some(cell => fleet[second].some(other => touching(cell, other)))) fail(`${ships[first]} touches ${ships[second]}`);
423 }
424 }
425 }
426 this.players[p].fleet = fleet;
427 return [`LAYOUT ${p + 1}`, ...this.placementOutput(p)];
428 }

430 boardOutput(p: PlayerIndex, radar: boolean): string[] {
431 const player = this.players[p];

433 if (this.reefs.has(cell)) return '~';

443 densityOutput(p: PlayerIndex): string[] {
444 const view = this.view(p);
445 const scores = this.density(p, view, this.excluded(view));
446 const output = [`DENSITY ${p + 1}`, ' ' + Array.from({ length: 10 }, (_, i) => String(i + 1).padStart(5)).join('')];
447 let best = -1;
448 for (let row = 0; row < 10; row++) {
449 let text = String.fromCharCode(65 + row);
450 for (let column = 0; column < 10; column++) {
451 const cell = row * 10 + column;
452 const value = view[cell] === '.' ? String(scores[cell]) : view[cell] === '~' ? '~' : '-';
453 text += value.padStart(5);
454 if (view[cell] === '.' && (best === -1 || scores[cell] > scores[best])) best = cell;
455 }
456 output.push(text);
457 }
458 output.push(`BEST ${coordinate(best)} ${scores[best]}`);
459 return output;
460 }

479 status(): string {
480 const prefix = `STATUS ${this.phase} salvo ${this.salvo ? 'ON' : 'OFF'} chain ${this.chain ? 'ON' : 'OFF'} touch ${this.touch ? 'ON' : 'OFF'} reefs ${this.reefs.size}`;

494const setupCommands = new Set<Command['kind']>(['PLACE', 'REMOVE', 'AUTOPLACE', 'LAYOUT', 'REEF', 'SALVO', 'CHAIN', 'TOUCH']);
495
496class Program {
497 game = new Game();
498
499 execute(command: Command): string[] {
500 if (command.kind === 'NEW') {
501 this.game = new Game([this.game.players[0].wins, this.game.players[1].wins]);
502 return ['NEW'];
503 }
504 const game = this.game;
505 if (setupCommands.has(command.kind) && game.phase !== 'SETUP') fail('setup closed');
506 switch (command.kind) {
507 case 'PLACE': {
508 const { player: p, ship } = command;
509 const player = game.players[p];
510 if (player.fleet[ship].length !== 0) fail(`${ships[ship]} already placed`);
511 const cells = footprint(command.cell, command.direction, lengths[ship]);
512 const problem = game.placementProblem(player, cells);
513 if (problem !== null) fail(problem);
514 player.fleet[ship] = cells!;
515 return [`PLACED ${p + 1} ${ships[ship]} ${shipRange(cells!)}`];
516 }
517 case 'REMOVE': {
518 const { player: p, ship } = command;
519 if (game.players[p].fleet[ship].length === 0) fail(`${ships[ship]} not placed`);
520 game.players[p].fleet[ship] = [];
521 return [`REMOVED ${p + 1} ${ships[ship]}`];
522 }
523 case 'AUTOPLACE': {
524 const attempts = game.autoPlace(command.player, command.seed);
525 return [`AUTOPLACE ${command.player + 1} attempts ${attempts}`, ...game.placementOutput(command.player)];
526 }
527 case 'LAYOUT': return game.layout(command.player, command.layout);
528 case 'REEF': {
529 const cell = command.cell;
530 if (game.reefs.has(cell)) fail(`reef exists at ${coordinate(cell)}`);
531 for (const owner of playerIndices) {
532 const ship = game.players[owner].shipAt(cell);
533 if (ship !== null) fail(`${coordinate(cell)} occupied by ${owner + 1} ${ships[ship]}`);
534 }
535 game.reefs.add(cell);
536 return [`REEF ${coordinate(cell)}`];
537 }
538 case 'SALVO': case 'CHAIN': case 'TOUCH':
539 if (command.kind === 'SALVO') game.salvo = command.enabled;
540 else if (command.kind === 'CHAIN') game.chain = command.enabled;
541 else game.touch = command.enabled;
542 return [`${command.kind} ${command.enabled ? 'ON' : 'OFF'}`];
543 case 'FIRE': case 'AI': case 'TORPEDO': case 'SONAR': {
544 const p = command.player;
545 game.checkTurn(p);
546 const output: string[] = [];
547 if (command.kind === 'FIRE') {
548 const allowance = game.allowance(p);
549 if (command.cells.length !== allowance) fail(`expected ${allowance} shots`);
550 for (const cell of command.cells) if (game.reefs.has(cell)) fail(`reef at ${coordinate(cell)}`);
551 game.shootingTurn(p, null, command.cells, output);
552 } else if (command.kind === 'AI') game.shootingTurn(p, command.strategy, [], output);
553 else {
554 if (command.kind === 'TORPEDO' && game.players[p].torpedo) fail('torpedo used');
555 if (command.kind === 'SONAR' && game.players[p].sonar) fail('sonar used');
556 game.specialTurn(command, output);
557 }
558 game.automatic(output);
559 return output;
560 }
561 case 'RESIGN': {
562 if (game.phase !== 'BATTLE') fail('not in battle');
563 const winner = otherPlayer(command.player);
564 game.phase = 'OVER';
565 game.winner = winner;
566 game.players[winner].wins++;
567 return [`RESIGN ${command.player + 1}`, `WINNER ${winner + 1}`];
568 }
569 case 'CONTROL':
570 game.players[command.player].controller = command.controller;
571 return [`CONTROL ${command.player + 1} ${command.controller}`];
572 case 'SEED':
573 game.players[command.player].random = command.seed;
574 return [`SEED ${command.player + 1} ${command.seed}`];
575 case 'HINT': {
576 if (game.phase === 'OVER') fail('game over');
577 const choice = game.choose(command.player, command.strategy);
578 return [`HINT ${command.player + 1} ${command.strategy} ${coordinate(choice.cell)} ${choice.info}`];
579 }
580 case 'DENSITY':
581 if (game.phase === 'OVER') fail('game over');
582 return game.densityOutput(command.player);
583 case 'SHOW': return game.boardOutput(command.player, false);
584 case 'RADAR': return game.boardOutput(command.player, true);
585 case 'EXPORT': return [`EXPORT ${command.player + 1} ${game.export(command.player)}`];
586 case 'FLEET': return game.fleetOutput(command.player);
587 case 'STATUS': return [game.status()];
588 case 'STATS': return game.statistics();
589 case 'MATCH': {
590 const output = [`MATCH 1 ${command.strategies[0]} 2 ${command.strategies[1]}`];
591 const match = game.match(command.seeds[0], command.seeds[1], command.strategies, output);
592 output.push(`RESULT winner ${match.winner + 1} shots ${match.shots[0]} ${match.shots[1]} hits ${match.hits[0]} ${match.hits[1]}`);
593 return output;
594 }
595 case 'SERIES': {
596 let random = command.seed;
597 const wins = [0, 0];
598 const shots = [0, 0];
599 const fastest: (number | null)[] = [null, null];
600 for (let i = 0; i < command.count; i++) {
601 random = advance(random);
602 const seed1 = random;
603 random = advance(random);
604 const match = game.match(seed1, random, command.strategies);
605 wins[match.winner]++;
606 for (const seat of playerIndices) shots[seat] += match.shots[seat];
607 const wonShots = match.shots[match.winner];
608 fastest[match.winner] = Math.min(fastest[match.winner] ?? wonShots, wonShots);
609 }
610 return [`SERIES ${command.count} wins ${wins[0]} ${wins[1]} shots ${shots[0]} ${shots[1]} fastest ${fastest[0] ?? '-'} ${fastest[1] ?? '-'}`];
611 }
612 case 'TOURNEY': {
613 let random = command.seed;
614 const totals = strategies.map((strategy, index) => ({ strategy, index, wins: 0, shots: 0 }));
615 for (let first = 0; first < 3; first++) {
616 for (let second = 0; second < 3; second++) {
617 if (first === second) continue;
618 for (let i = 0; i < command.count; i++) {
619 random = advance(random);
620 const seed1 = random;
621 random = advance(random);
622 const match = game.match(seed1, random, [strategies[first], strategies[second]]);
623 totals[match.winner === 0 ? first : second].wins++;
624 totals[first].shots += match.shots[0];
625 totals[second].shots += match.shots[1];
626 }
627 }
628 }
629 totals.sort((a, b) => b.wins - a.wins || a.shots - b.shots || a.index - b.index);
630 return [`TOURNEY ${command.count}`, ...totals.map((total, rank) => `${rank + 1} ${total.strategy} wins ${total.wins} shots ${total.shots}`)];
631 }
632 }
633 }
634}

636const lines = readFileSync(0, 'utf8').split(/\r?\n/);
637const count = Number(lines[0]);
638const program = new Program();
639for (let i = 1; i <= count; i++) {
640 try {
641 process.stdout.write(program.execute(parseCommand(lines[i].split(' '))).join('\n') + '\n');
642 } catch (error) {
643 if (!(error instanceof CommandError)) throw error;
644 process.stdout.write(`ERROR ${error.message}\n`);
645 }
