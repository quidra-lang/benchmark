8local function emit(...)
9 local words = { ... }
10 for i = 1, #words do words[i] = tostring(words[i]) end
11 io.write(table.concat(words, " "), "\n")
12end
13local function fail(message) error({ command_error = true, message = message }, 0) end
14local function fill(size, value)
15 local result = {}
16 for i = 1, size do result[i] = value end
17 return result
18end
19local function copy_list(values)
20 local result = {}
21 for i, value in ipairs(values) do result[i] = value end
22 return result
23end
24local function advance(random) return 48271 * random % 2147483647 end
25local function coordinate(cell) return string.char(65 + (cell - 1) // 10) .. ((cell - 1) % 10 + 1) end
26local function parse_cell(text) return (text:byte(1) - 65) * 10 + tonumber(text:sub(2)) end
27local function switch_name(enabled) return enabled and "ON" or "OFF" end
28local function step(cell, dr, dc)
29 local row, column = (cell - 1) // 10 + dr, (cell - 1) % 10 + dc
30 if row >= 0 and row < 10 and column >= 0 and column < 10 then return row * 10 + column + 1 end
31end
32local function adjacent(a, b)
33 return math.abs((a - 1) // 10 - (b - 1) // 10) <= 1 and math.abs((a - 1) % 10 - (b - 1) % 10) <= 1
34end
35local surroundings = {}
36for cell = 1, 100 do
37 local nearby = {}
38 for dr = -1, 1 do
39 for dc = -1, 1 do
40 if dr ~= 0 or dc ~= 0 then
41 local other = step(cell, dr, dc)
42 if other then nearby[#nearby + 1] = other end
43 end
44 end
45 end
46 surroundings[cell] = nearby
47end
48local function position(start, stride, length)
49 local offset = stride == 1 and (start - 1) % 10 or (start - 1) // 10
50 if offset + length > 10 then return nil end
51 local cells = {}
52 for i = 0, length - 1 do cells[#cells + 1] = start + i * stride end
53 return cells
54end
55local function touching(first, second)
56 for _, a in ipairs(first) do
57 for _, b in ipairs(second) do if adjacent(a, b) then return true end end
58 end
59 return false
60end
61local function ship_range(cells) return coordinate(cells[1]) .. "-" .. coordinate(cells[#cells]) end
62
63local placements = {}
64for number, length in ipairs(lengths) do
65 placements[number] = {}
66 for cell = 1, 100 do
67 for _, stride in ipairs({ 1, 10 }) do
68 local cells = position(cell, stride, length)
69 if cells then placements[number][#placements[number] + 1] = cells end
70 end
71 end
72end

76function Player.new(seed)
77 return setmetatable({ fleet = fill(5, false), board = fill(100, 0), attacked = fill(100, false),
78 candidates = {}, random = seed, controller = "HUMAN", torpedo_used = false, sonar_used = false,
79 shots = 0, hits = 0, sinks = 0, turns = 0, streak = 0, longest = 0, wins = 0 }, Player)

81function Player:clear_fleet() self.fleet, self.board = fill(5, false), fill(100, 0) end
82function Player:put_ship(number, cells)
83 self.fleet[number] = cells
84 for _, cell in ipairs(cells) do self.board[cell] = number end
85end
86function Player:attacked_count(number)
87 local count = 0
88 if self.fleet[number] then
89 for _, cell in ipairs(self.fleet[number]) do if self.attacked[cell] then count = count + 1 end end
90 end
91 return count
92end
93function Player:sunk(number) return self.fleet[number] and self:attacked_count(number) == lengths[number] end
94function Player:placed()
95 local count = 0
96 for _, cells in ipairs(self.fleet) do if cells then count = count + 1 end end
97 return count
98end
99function Player:afloat()
100 local count = 0
101 for number, cells in ipairs(self.fleet) do if cells and not self:sunk(number) then count = count + 1 end end
102 return count
103end
104function Player:defeated()
105 for number = 1, 5 do if not self:sunk(number) then return false end end
106 return true
107end
108function Player:record(result)
109 self.shots = self.shots + 1
110 if result == "HIT" or result == "SUNK" then
111 self.hits, self.streak = self.hits + 1, self.streak + 1
112 self.longest = math.max(self.longest, self.streak)
113 else
114 self.streak = 0
115 end
116 if result == "SUNK" then self.sinks = self.sinks + 1 end
117end

125function Game:allowance(player) return self.salvo and self.players[player]:afloat() or 1 end
126function Game:placement_problem(player, cells)
127 if not cells then return "out of bounds" end
128 for _, cell in ipairs(cells) do if self.reefs[cell] then return "on reef" end end
129 for number = 1, 5 do
130 for _, cell in ipairs(cells) do
131 if player.board[cell] == number then return "overlaps " .. ship_names[number] end
132 end
133 end
134 if not self.touch then
135 for number, other in ipairs(player.fleet) do
136 if other and touching(cells, other) then return "touches " .. ship_names[number] end
137 end
138 end
139end

145function Game:auto_place(player, seed)
146 local owner, attempts = self.players[player], 0
147 owner.random = seed
148 owner:clear_fleet()
149 for number, length in ipairs(lengths) do
150 repeat
151 attempts = attempts + 1
152 owner.random = advance(owner.random)
153 local stride = owner.random % 2 == 0 and 1 or 10
154 owner.random = advance(owner.random)
155 local k = owner.random % (10 * (11 - length))
156 local start = stride == 1 and k // (11 - length) * 10 + k % (11 - length) + 1 or k + 1
157 local cells = position(start, stride, length)
158 if not self:placement_problem(owner, cells) then owner:put_ship(number, cells) end
159 until owner.fleet[number]
160 end
161 return attempts
162end
163function Game:layout_fleet(layout)
164 local flat, fleet = layout:gsub("/", ""), {}
165 for number, symbol in ipairs(symbols) do
166 local cells = {}
167 for cell = 1, 100 do if flat:sub(cell, cell) == symbol then cells[#cells + 1] = cell end end
168 if #cells ~= lengths[number] then fail(ship_names[number] .. " has " .. #cells .. " cells") end
169 local stride = cells[2] - cells[1]
170 local expected = (stride == 1 or stride == 10) and position(cells[1], stride, #cells)
171 if not expected then fail(ship_names[number] .. " not straight") end
172 for i, cell in ipairs(cells) do if cell ~= expected[i] then fail(ship_names[number] .. " not straight") end end
173 for _, cell in ipairs(cells) do if self.reefs[cell] then fail(ship_names[number] .. " on reef") end end
174 fleet[number] = cells
175 end
176 if not self.touch then
177 for first = 1, 4 do
178 for second = first + 1, 5 do
179 if touching(fleet[first], fleet[second]) then fail(ship_names[first] .. " touches " .. ship_names[second]) end
180 end
181 end
182 end
183 return fleet
184end
185function Game:view(player)
186 local opponent = self.players[3 - player]
187 local view = { cells = {}, remaining = {}, excluded = fill(100, false) }
188 for number = 1, 5 do view.remaining[number] = not opponent:sunk(number) end
189 for cell = 1, 100 do
190 local number = opponent.board[cell]
191 if self.reefs[cell] then view.cells[cell] = "~"
192 elseif not opponent.attacked[cell] then view.cells[cell] = "."
193 elseif number == 0 then view.cells[cell] = "o"
194 else view.cells[cell] = view.remaining[number] and "x" or "#" end
195 end
196 if not self.touch then
197 for cell, value in ipairs(view.cells) do
198 if value == "#" then
199 for _, other in ipairs(surroundings[cell]) do view.excluded[other] = true end
200 end
201 end
202 end
203 return view
204end
205local function candidate(view, cell) return cell and view.cells[cell] == "." and not view.excluded[cell] end
206local function density(view)
207 local scores = fill(100, 0)
208 for number = 1, 5 do
209 if view.remaining[number] then
210 for _, cells in ipairs(placements[number]) do
211 local possible, hits = true, 0
212 for _, cell in ipairs(cells) do
213 if view.excluded[cell] or view.cells[cell] ~= "." and view.cells[cell] ~= "x" then possible = false; break end
214 if view.cells[cell] == "x" then hits = hits + 1 end
215 end
216 if possible then
217 for _, cell in ipairs(cells) do scores[cell] = scores[cell] + 1 + 10 * hits end
218 end
219 end
220 end
221 end
222 return scores
223end
224local function best_cell(view, scores)
225 local best
226 for cell = 1, 100 do
227 if view.cells[cell] == "." and (not best or scores[cell] > scores[best]) then best = cell end
228 end
229 return best
230end
231function Game:choose(player, strategy)
232 local owner, view = self.players[player], self:view(player)
233 if strategy == "PROB" then
234 local scores = density(view)
235 local cell = best_cell(view, scores)
236 return cell, scores[cell], owner.random
237 elseif strategy == "RANDOM" then
238 local cells = {}
239 for cell = 1, 100 do if candidate(view, cell) then cells[#cells + 1] = cell end end
240 local random = advance(owner.random)
241 return cells[random % #cells + 1], #cells, random
242 end
243 for cell = 1, 100 do
244 if view.cells[cell] == "x" then
245 for _, direction in ipairs({ { 0, 1 }, { 1, 0 } }) do
246 local dr, dc = direction[1], direction[2]
247 local front = step(cell, dr, dc)
248 if front and view.cells[front] == "x" then
249 repeat front = step(front, dr, dc) until not front or view.cells[front] ~= "x"
250 if candidate(view, front) then return front, "LINE", owner.random end
251 local back = step(cell, -dr, -dc)
252 if candidate(view, back) then return back, "LINE", owner.random end
253 end
254 end
255 end
256 end
257 for i = #owner.candidates, 1, -1 do
258 local cell = owner.candidates[i]
259 if candidate(view, cell) then return cell, "STACK", owner.random end
260 end
261 local shortest = 5
262 for number, remaining in ipairs(view.remaining) do if remaining then shortest = math.min(shortest, lengths[number]) end end
263 for cell = 1, 100 do
264 if candidate(view, cell) and ((cell - 1) // 10 + (cell - 1) % 10) % shortest == 0 then return cell, "PARITY", owner.random end
265 end
266end
267local function is_hit(result) return result == "HIT" or result == "SUNK" end
268local function shot_text(result, number) return result == "SUNK" and "SUNK " .. ship_names[number] or result end
269function Game:shoot(player, cell)
270 local owner, opponent = self.players[player], self.players[3 - player]
271 local number, result = opponent.board[cell], "ALREADY"
272 if not opponent.attacked[cell] then
273 opponent.attacked[cell] = true
274 result = number == 0 and "MISS" or opponent:sunk(number) and "SUNK" or "HIT"
275 end
276 if result == "HIT" then
277 for _, direction in ipairs({ { 0, -1 }, { 1, 0 }, { 0, 1 }, { -1, 0 } }) do
278 local other = step(cell, direction[1], direction[2])
279 if other then owner.candidates[#owner.candidates + 1] = other end
280 end
281 end
282 owner:record(result)
283 return result, number
284end
285function Game:finish_turn(player, hit, mode)
286 local owner = self.players[player]
287 owner.turns = owner.turns + 1
288 if self.players[3 - player]:defeated() then
289 self.phase, self.winner = "OVER", player
290 owner.wins = owner.wins + 1
291 if mode == "normal" then emit("WINNER", player) end
292 else
293 self.current = self.chain and hit and player or 3 - player
294 end
295end
296function Game:shooting_turn(player, strategy, cells, mode)
297 local hit, allowance = false, self:allowance(player)
298 for i = 1, allowance do
299 local cell
300 if strategy then
301 local chosen, _, random = self:choose(player, strategy)
302 cell = chosen
303 self.players[player].random = random
304 else
305 cell = cells[i]
306 end
307 local result, number = self:shoot(player, cell)
308 hit = hit or is_hit(result)
309 if mode == "normal" then emit("SHOT", player, coordinate(cell), shot_text(result, number))
310 elseif mode == "sinks" and result == "SUNK" then emit("SINK", player, ship_names[number], self.players[player].shots) end
311 if self.players[3 - player]:defeated() then break end
312 end
313 self:finish_turn(player, hit, mode)
314end
315function Game:automatic()
316 while self.phase == "BATTLE" do
317 local player = self.current
318 local controller = self.players[player].controller
319 if controller == "HUMAN" then break end
320 self:shooting_turn(player, controller, nil, "normal")
321 end
322end
323function Game:torpedo_turn(player, row, side)
324 local owner, opponent = self.players[player], self.players[3 - player]
325 owner.torpedo_used = true
326 local result, hit, stopped_ship = "MISS", false, false
327 local start, last, stride = 1, 10, 1
328 if side == "E" then start, last, stride = 10, 1, -1 end
329 for column = start, last, stride do
330 local cell = (row - 1) * 10 + column
331 if self.reefs[cell] then
332 result = "REEF " .. coordinate(cell)
333 break
334 elseif opponent.board[cell] ~= 0 then
335 local shot, number = self:shoot(player, cell)
336 result, hit, stopped_ship = shot_text(shot, number) .. " " .. coordinate(cell), is_hit(shot), true
337 break
338 end
339 opponent.attacked[cell] = true
340 end
341 if not stopped_ship then owner:record("MISS") end
342 emit("TORPEDO", player, string.char(64 + row), side, result)
343 self:finish_turn(player, hit, "normal")
344end
345function Game:sonar_turn(player, center)
346 self.players[player].sonar_used = true
347 local opponent, count = self.players[3 - player], 0
348 for cell = 1, 100 do
349 if adjacent(cell, center) and opponent.board[cell] ~= 0 and not opponent.attacked[cell] then count = count + 1 end
350 end
351 emit("SONAR", player, coordinate(center), count)
352 self:finish_turn(player, false, "normal")
353end

429local function play_match(current, first_seed, second_seed, first_strategy, second_strategy, mode)
430 local game = Game.new()
431 game.salvo, game.chain, game.touch = current.salvo, current.chain, current.touch
432 game.reefs = copy_list(current.reefs)
433 game:auto_place(1, first_seed)
434 game:auto_place(2, second_seed)
435 game.phase = "BATTLE"
436 while game.phase == "BATTLE" do
437 local player = game.current
438 game:shooting_turn(player, player == 1 and first_strategy or second_strategy, nil, mode)
439 end
440 return game
441end
442function Game:series_output(count, seed, first_strategy, second_strategy)
443 local wins, shots, fastest = { 0, 0 }, { 0, 0 }, {}
444 for _ = 1, count do
445 local first_seed = advance(seed)
446 seed = advance(first_seed)
447 local game = play_match(self, first_seed, seed, first_strategy, second_strategy, "quiet")
448 local winner = game.winner
449 wins[winner] = wins[winner] + 1
450 for player = 1, 2 do shots[player] = shots[player] + game.players[player].shots end
451 local won_shots = game.players[winner].shots
452 fastest[winner] = fastest[winner] and math.min(fastest[winner], won_shots) or won_shots
453 end
454 emit("SERIES", count, "wins", wins[1], wins[2], "shots", shots[1], shots[2], "fastest", fastest[1] or "-", fastest[2] or "-")
455end
456function Game:tourney_output(count, seed)
457 local wins, shots = { 0, 0, 0 }, { 0, 0, 0 }
458 for first = 1, 3 do
459 for second = 1, 3 do
460 if first ~= second then
461 for _ = 1, count do
462 local first_seed = advance(seed)
463 seed = advance(first_seed)
464 local game = play_match(self, first_seed, seed, strategies[first], strategies[second], "quiet")
465 local winner = game.winner == 1 and first or second
466 wins[winner] = wins[winner] + 1
467 shots[first], shots[second] = shots[first] + game.players[1].shots, shots[second] + game.players[2].shots
468 end
469 end
470 end
471 end
472 local ranking = { 1, 2, 3 }
473 table.sort(ranking, function(a, b)
474 if wins[a] ~= wins[b] then return wins[a] > wins[b] end
475 if shots[a] ~= shots[b] then return shots[a] < shots[b] end
476 return a < b
477 end)
478 emit("TOURNEY", count)
479 for rank, number in ipairs(ranking) do emit(rank, strategies[number], "wins", wins[number], "shots", shots[number]) end
480end

482local setup_commands = { PLACE = true, REMOVE = true, AUTOPLACE = true, LAYOUT = true, REEF = true, SALVO = true, CHAIN = true, TOUCH = true }
483local turn_commands = { FIRE = true, AI = true, TORPEDO = true, SONAR = true }
484local player_commands = { PLACE = true, REMOVE = true, AUTOPLACE = true, LAYOUT = true, FIRE = true, AI = true, TORPEDO = true,
485 SONAR = true, RESIGN = true, CONTROL = true, SEED = true, HINT = true, DENSITY = true, SHOW = true, RADAR = true, EXPORT = true, FLEET = true }

487local function execute(game, words)
488 local command = words[1]
489 if command == "NEW" then
490 local replacement = Game.new()
491 for player = 1, 2 do replacement.players[player].wins = game.players[player].wins end
492 emit("NEW")
493 return replacement
494 end
495 local player = player_commands[command] and tonumber(words[2])
496 if setup_commands[command] and game.phase ~= "SETUP" then fail("setup closed") end
497 if turn_commands[command] then
498 if game.phase == "OVER" then fail("game over") end
499 if game.phase == "SETUP" and (game.players[1]:placed() < 5 or game.players[2]:placed() < 5) then fail("fleets not ready") end
500 if player ~= game.current then fail("not your turn") end
501 local cells = {}
502 if command == "FIRE" then
503 local expected = game:allowance(player)
504 if #words - 2 ~= expected then fail("expected " .. expected .. " shots") end
505 for i = 3, #words do
506 local cell = parse_cell(words[i])
507 if game.reefs[cell] then fail("reef at " .. coordinate(cell)) end
508 cells[#cells + 1] = cell
509 end
510 elseif command == "TORPEDO" and game.players[player].torpedo_used then fail("torpedo used")
511 elseif command == "SONAR" and game.players[player].sonar_used then fail("sonar used") end
512 game.phase = "BATTLE"
513 if command == "FIRE" then game:shooting_turn(player, nil, cells, "normal")
514 elseif command == "AI" then game:shooting_turn(player, words[3], nil, "normal")
515 elseif command == "TORPEDO" then game:torpedo_turn(player, words[3]:byte() - 64, words[4])
516 else game:sonar_turn(player, parse_cell(words[3])) end
517 game:automatic()
518 return game
519 end
520 if command == "PLACE" or command == "REMOVE" then
521 local owner, number = game.players[player], ship_numbers[words[3]]
522 if command == "PLACE" then
523 if owner.fleet[number] then fail(ship_names[number] .. " already placed") end
524 local cells = position(parse_cell(words[4]), words[5] == "H" and 1 or 10, lengths[number])
525 local problem = game:placement_problem(owner, cells)
526 if problem then fail(problem) end
527 owner:put_ship(number, cells)
528 emit("PLACED", player, ship_names[number], ship_range(cells))
529 else
530 if not owner.fleet[number] then fail(ship_names[number] .. " not placed") end
531 for _, cell in ipairs(owner.fleet[number]) do owner.board[cell] = 0 end
532 owner.fleet[number] = false
533 emit("REMOVED", player, ship_names[number])
534 end
535 elseif command == "AUTOPLACE" then
536 local attempts = game:auto_place(player, tonumber(words[3]))
537 emit("AUTOPLACE", player, "attempts", attempts)
538 game:placement_output(player)
539 elseif command == "LAYOUT" then
540 local fleet = game:layout_fleet(words[3])
541 game.players[player]:clear_fleet()
542 for number, cells in ipairs(fleet) do game.players[player]:put_ship(number, cells) end
543 emit("LAYOUT", player)
544 game:placement_output(player)
545 elseif command == "REEF" then
546 local cell = parse_cell(words[2])
547 if game.reefs[cell] then fail("reef exists at " .. coordinate(cell)) end
548 for owner = 1, 2 do
549 local number = game.players[owner].board[cell]
550 if number ~= 0 then fail(coordinate(cell) .. " occupied by " .. owner .. " " .. ship_names[number]) end
551 end
552 game.reefs[cell] = true
553 emit("REEF", coordinate(cell))
554 elseif command == "SALVO" or command == "CHAIN" or command == "TOUCH" then
555 game[command:lower()] = words[2] == "ON"
556 emit(command, words[2])
557 elseif command == "CONTROL" then
558 game.players[player].controller = words[3]
559 emit("CONTROL", player, words[3])
560 elseif command == "SEED" then
561 game.players[player].random = tonumber(words[3])
562 emit("SEED", player, game.players[player].random)
563 elseif command == "RESIGN" then
564 if game.phase ~= "BATTLE" then fail("not in battle") end
565 game.phase, game.winner = "OVER", 3 - player
566 game.players[game.winner].wins = game.players[game.winner].wins + 1
567 emit("RESIGN", player)
568 emit("WINNER", game.winner)
569 elseif command == "HINT" then
570 if game.phase == "OVER" then fail("game over") end
571 local cell, info = game:choose(player, words[3])
572 emit("HINT", player, words[3], coordinate(cell), info)
573 elseif command == "DENSITY" then
574 if game.phase == "OVER" then fail("game over") end
575 game:density_output(player)
576 elseif command == "SHOW" or command == "RADAR" then game:board_output(player, command == "RADAR")
577 elseif command == "EXPORT" then game:export_fleet(player)
578 elseif command == "FLEET" then game:fleet_output(player)
579 elseif command == "STATUS" then game:status_output()
580 elseif command == "STATS" then game:statistics_output()
581 elseif command == "MATCH" then
582 emit("MATCH", 1, words[4], 2, words[5])
583 local trial = play_match(game, tonumber(words[2]), tonumber(words[3]), words[4], words[5], "sinks")
584 local first, second = trial.players[1], trial.players[2]
585 emit("RESULT winner", trial.winner, "shots", first.shots, second.shots, "hits", first.hits, second.hits)
586 elseif command == "SERIES" then game:series_output(tonumber(words[2]), tonumber(words[3]), words[4], words[5])
587 elseif command == "TOURNEY" then game:tourney_output(tonumber(words[2]), tonumber(words[3])) end
588 return game
589end

591local function input_line() return (io.read("l"):gsub("\r$", "")) end
592local game = Game.new()
593local count = tonumber(input_line())
594for _ = 1, count do
595 local words = {}
596 for word in input_line():gmatch("%S+") do words[#words + 1] = word end
597 local ok, result = pcall(execute, game, words)
598 if ok then game = result
599 elseif type(result) == "table" and result.command_error then emit("ERROR", result.message)
600 else error(result, 0) end
601end
