func fail(_ message: String) throws -> Never {
    throw CommandError(message: message)
}
struct Game {
    mutating func shoot(_ p: Int, cell: Int) -> Shot {
        players[p].shots += 1
        return result
    }
    mutating func finishTurn(_ p: Int, hit: Bool, output: inout [String], mode: OutputMode) {
        players[p].turns += 1
        if players[1 - p].defeated {
            phase = .over
            winner = p
            players[p].wins += 1
            if mode == .normal {
                output.append("WINNER \(p + 1)")
            }
        } else {
            current = chain && hit ? p : 1 - p
        }
    }
    mutating func shootingTurn(
        _ p: Int,
        strategy: Strategy?,
        cells: [Int],
        output: inout [String],
        mode: OutputMode = .normal
    ) {
        phase = .battle
        let count = allowance(p)
        var hit = false
        for index in 0..<count {
            let cell: Int
            if let strategy {
                cell = choose(p, strategy: strategy).cell
            } else {
                cell = cells[index]
            }
            let result = shoot(p, cell: cell)
            hit = hit || result.hit
            if mode == .normal {
                output.append("SHOT \(p + 1) \(coordinate(cell)) \(result.text)")
            }
            if players[1 - p].defeated {
                break
            }
        }
        finishTurn(p, hit: hit, output: &output, mode: mode)
    }
    func checkTurn(_ p: Int) throws {
        if phase == .over {
            try fail("game over")
        }
        if phase == .setup && players.contains(where: { $0.placed != 5 }) {
            try fail("fleets not ready")
        }
        if p != current {
            try fail("not your turn")
        }
    }
let turnCommands: Set<String> = ["FIRE", "AI", "TORPEDO", "SONAR"]
struct Program {
    var game = Game()
    mutating func execute(_ words: [String]) throws -> [String] {
        let command = words[0]
        if turnCommands.contains(command) {
            let p = Int(words[1])! - 1
            try game.checkTurn(p)
            var output: [String] = []
            if command == "FIRE" {
                let allowance = game.allowance(p)
                if words.count - 2 != allowance {
                    try fail("expected \(allowance) shots")
                }
                let cells = words.dropFirst(2).map(parseCell)
                for cell in cells where game.reefs.contains(cell) {
                    try fail("reef at \(coordinate(cell))")
                }
                game.shootingTurn(p, strategy: nil, cells: cells, output: &output)
            } else if command == "AI" {
                game.shootingTurn(
                    p,
                    strategy: Strategy(rawValue: words[2])!,
                    cells: [],
                    output: &output
                )
            } else {
                if command == "TORPEDO" && game.players[p].torpedo {
                    try fail("torpedo used")
                }
                if command == "SONAR" && game.players[p].sonar {
                    try fail("sonar used")
                }
                game.specialTurn(p, command: command, args: Array(words.dropFirst(2)), output: &output)
            }
            game.automatic(output: &output)
            return output
        }
let commandCount = Int(readLine()!)!
var program = Program()
for _ in 0..<commandCount {
    let words = readLine()!.split(separator: " ").map(String.init)
    do {
        print(try program.execute(words).joined(separator: "\n"))
    } catch let error as CommandError {
        print("ERROR \(error.message)")
    } catch {
        fatalError(String(describing: error))
    }
}
