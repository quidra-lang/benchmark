14def coordinate(cell) = "#{(65 + cell / 10).chr}#{cell % 10 + 1}"
15def parse_cell(text) = (text.ord - 65) * 10 + text[1..].to_i - 1
16def advance(state) = 48_271 * state % 2_147_483_647
17def switch_name(enabled) = enabled ? 'ON' : 'OFF'

19def step(cell, dr, dc)
20 row, column = cell / 10 + dr, cell % 10 + dc
21 row * 10 + column if row.between?(0, 9) && column.between?(0, 9)
22end

24NEIGHBORS = Array.new(100) do |cell|
25 (-1..1).flat_map do |dr|
26 (-1..1).filter_map { |dc| step(cell, dr, dc) unless dr.zero? && dc.zero? }
27 end.freeze
28end.freeze

65class Player

67 attr_accessor :random, :controller, :torpedo_used, :sonar_used, :wins

69 def initialize(seed)
70 @fleet = Array.new(5)
71 @board = Array.new(100)
72 @attacked = Array.new(100, false)
73 @candidates = []
74 @stats = Stats.new

81 def sunk?(index) = fleet[index] && fleet[index].cells.all? { |cell| attacked[cell] }
82 def placed = fleet.count { |ship| ship }
83 def afloat = fleet.each_index.count { |index| fleet[index] && !sunk?(index) }
84
85 def set_ship(index, ship)
86 fleet[index]&.cells&.each { |cell| board[cell] = nil }
87 fleet[index] = ship
88 ship&.cells&.each { |cell| board[cell] = index }
89 end

91 def clear_fleet
92 fleet.each_index { |index| set_ship(index, nil) }
93 end

96View = Data.define(:cells, :excluded, :remaining) do
97 def candidate?(cell)
98 cell && cell.between?(0, 99) && cells[cell] == :unattacked && !excluded[cell]
99 end
100
101 def scores
102 result = Array.new(100, 0)
103 remaining.each do |ship|
104 100.times do |start|
105 [true, false].each do |horizontal|
106 placement = position(start, horizontal, SHIP_LENGTHS[ship])
107 next unless placement
108
109 possible = placement.cells.all? do |cell|
110 !excluded[cell] && (cells[cell] == :unattacked || cells[cell] == :hit)
111 end
112 next unless possible
113
114 weight = 1 + 10 * placement.cells.count { |cell| cells[cell] == :hit }
115 placement.cells.each { |cell| result[cell] += weight }
116 end
117 end
118 end
119 result
120 end
121
122 def best(scores)
123 cells.each_index.select { |cell| cells[cell] == :unattacked }
124 .min_by { |cell| [-scores[cell], cell] }
125 end
126end

153 def placement_problem(player, ship)
154 return 'out of bounds' unless ship
155 return 'on reef' if ship.cells.any? { |cell| reefs[cell] }
156
157 board = players[player].board
158 overlap = ship.cells.filter_map { |cell| board[cell] }.min
159 return "overlaps #{SHIP_NAMES[overlap]}" if overlap
160
161 unless touch
162 adjacent = ship.cells.flat_map { |cell| NEIGHBORS[cell] }.filter_map { |cell| board[cell] }.min
163 return "touches #{SHIP_NAMES[adjacent]}" if adjacent
164 end
165 nil
166 end

168 def auto_place(player, seed)
169 owner = players[player]
170 owner.random = seed
171 owner.clear_fleet
172 attempts = 0
173 SHIP_LENGTHS.each_with_index do |length, index|
174 loop do
175 attempts += 1
176 owner.random = advance(owner.random)
177 horizontal = owner.random.even?
178 owner.random = advance(owner.random)
179 k = owner.random % (10 * (11 - length))
180 start = horizontal ? k / (11 - length) * 10 + k % (11 - length) : k
181 ship = position(start, horizontal, length)
182 next if placement_problem(player, ship)
183
184 owner.set_ship(index, ship)
185 break
186 end
187 end
188 attempts
189 end

191 def layout(player, text)
192 grid = text.delete('/')
193 fleet = SHIP_NAMES.each_index.map do |index|
194 cells = (0...100).select { |cell| grid[cell] == SHIP_SYMBOLS[index] }
195 name, length = SHIP_NAMES[index], SHIP_LENGTHS[index]
196 raise CommandError, "#{name} has #{cells.length} cells" unless cells.length == length
197
198 horizontal = cells.each_with_index.all? { |cell, i| cell == cells[0] + i && cell / 10 == cells[0] / 10 }
199 vertical = cells.each_with_index.all? { |cell, i| cell == cells[0] + 10 * i }
200 raise CommandError, "#{name} not straight" unless horizontal || vertical
201 raise CommandError, "#{name} on reef" if cells.any? { |cell| reefs[cell] }
202
203 Ship.new(cells.freeze)
204 end
205 unless touch
206 5.times do |first|
207 (first + 1...5).each do |second|
208 adjacent = fleet[first].cells.any? { |cell| NEIGHBORS[cell].intersect?(fleet[second].cells) }
209 raise CommandError, "#{SHIP_NAMES[first]} touches #{SHIP_NAMES[second]}" if adjacent
210 end
211 end
212 end
213 players[player].clear_fleet
214 fleet.each_with_index { |ship, index| players[player].set_ship(index, ship) }
215 end

233 def view(player)
234 opponent = players[1 - player]
235 sunk = SHIP_NAMES.each_index.map { |index| opponent.sunk?(index) }
236 cells = Array.new(100) do |cell|
237 ship = opponent.board[cell]
238 if reefs[cell]
239 :reef
240 elsif !opponent.attacked[cell]
241 :unattacked
242 elsif ship.nil?
243 :miss
244 else
245 sunk[ship] ? :sunk : :hit
246 end
247 end
248 excluded = Array.new(100, false)
249 unless touch
250 100.times do |cell|
251 NEIGHBORS[cell].each { |neighbor| excluded[neighbor] = true } if cells[cell] == :sunk
252 end
253 end
254 remaining = SHIP_NAMES.each_index.reject { |index| sunk[index] }
255 View.new(cells.freeze, excluded.freeze, remaining.freeze)
256 end

258 def choose(player, strategy)
259 sight = view(player)
260 random = players[player].random
261 case strategy
262 when 'PROB'
263 scores = sight.scores
264 cell = sight.best(scores)
265 return Choice.new(cell, scores[cell], random)
266 when 'RANDOM'
267 candidates = (0...100).select { |cell| sight.candidate?(cell) }
268 random = advance(random)
269 return Choice.new(candidates[random % candidates.length], candidates.length, random)
270 end
271
272 100.times do |cell|
273 next unless sight.cells[cell] == :hit
274
275 LINE_DIRECTIONS.each do |dr, dc|
276 front = step(cell, dr, dc)
277 next unless front && sight.cells[front] == :hit
278
279 front = step(front, dr, dc) while front && sight.cells[front] == :hit
280 back = step(cell, -dr, -dc)
281 return Choice.new(front, 'LINE', random) if sight.candidate?(front)
282 return Choice.new(back, 'LINE', random) if sight.candidate?(back)
283 end
284 end
285 players[player].candidates.reverse_each do |cell|
286 return Choice.new(cell, 'STACK', random) if sight.candidate?(cell)
287 end
288 shortest = sight.remaining.map { |index| SHIP_LENGTHS[index] }.min
289 cell = (0...100).find { |at| sight.candidate?(at) && ((at / 10 + at % 10) % shortest).zero? }
290 Choice.new(cell, 'PARITY', random)
291 end

293 def shoot(player, cell)
294 owner, opponent = players[player], players[1 - player]
295 if opponent.attacked[cell]
296 result = Shot.new(:already, nil)
297 else
298 opponent.attacked[cell] = true
299 index = opponent.board[cell]
300 result = if index.nil?
301 Shot.new(:miss, nil)
302 elsif opponent.sunk?(index)
303 Shot.new(:sunk, index)
304 else
305 Shot.new(:hit, nil)
306 end
307 if result.kind == :hit
308 STACK_DIRECTIONS.each do |dr, dc|
309 neighbor = step(cell, dr, dc)
310 owner.candidates << neighbor if neighbor
311 end
312 end
313 end
314 owner.stats.record(result)
315 result
316 end

318 def finish_turn(player, hit, mode, output)
319 players[player].stats.turns += 1
320 if players[1 - player].afloat.zero?
321 @phase, @winner = :over, player
322 players[player].wins += 1
323 output << "WINNER #{player + 1}" if mode == :normal
324 else
325 @current = chain && hit ? player : 1 - player
326 end
327 end

329 def shooting_turn(player, strategy: nil, targets: nil, output: nil, mode: :normal)
330 count = allowance(player)
331 @phase = :battle
332 hit = false
333 count.times do |i|
334 if strategy
335 choice = choose(player, strategy)
336 players[player].random = choice.random
337 cell = choice.cell
338 else
339 cell = targets[i]
340 end
341 result = shoot(player, cell)
342 hit ||= result.hit?
343 output << "SHOT #{player + 1} #{coordinate(cell)} #{result.text}" if mode == :normal
344 if mode == :match && result.kind == :sunk
345 output << "SINK #{player + 1} #{SHIP_NAMES[result.ship]} #{players[player].stats.shots}"
346 end
347 break if players[1 - player].afloat.zero?
348 end
349 finish_turn(player, hit, mode, output)
350 end

352 def torpedo(player, row, west, output)
353 @phase = :battle
354 players[player].torpedo_used = true
355 opponent = players[1 - player]
356 text, shot = 'MISS', nil
357 10.times do |i|
358 cell = row * 10 + (west ? i : 9 - i)
359 if reefs[cell]
360 text = "REEF #{coordinate(cell)}"
361 break
362 end
363 if opponent.board[cell]
364 shot = shoot(player, cell)
365 text = "#{shot.text} #{coordinate(cell)}"
366 break
367 end
368 opponent.attacked[cell] = true
369 end
370 players[player].stats.record(Shot.new(:miss, nil)) unless shot
371 output << "TORPEDO #{player + 1} #{(65 + row).chr} #{west ? 'W' : 'E'} #{text}"
372 finish_turn(player, shot&.hit?, :normal, output)
373 end

375 def sonar(player, cell, output)
376 @phase = :battle
377 players[player].sonar_used = true
378 opponent = players[1 - player]
379 count = [cell, *NEIGHBORS[cell]].count { |at| opponent.board[at] && !opponent.attacked[at] }
380 output << "SONAR #{player + 1} #{coordinate(cell)} #{count}"
381 finish_turn(player, false, :normal, output)
382 end

462 def status

480 def simulate(first_seed, second_seed, first_strategy, second_strategy, sinks: nil)
481 game = Game.new(salvo: salvo, chain: chain, touch: touch, reefs: reefs)
482 game.auto_place(0, first_seed)
483 game.auto_place(1, second_seed)
484 strategies = [first_strategy, second_strategy]
485 until game.phase == :over
486 game.shooting_turn(game.current, strategy: strategies[game.current], output: sinks, mode: sinks ? :match : :silent)
487 end
488 MatchResult.new(game.winner, game.players.map { |player| player.stats.shots }.freeze,
489 game.players.map { |player| player.stats.hits }.freeze)
490 end

493class Program
494 def initialize = @game = Game.new
495
496 def execute(words)
497 command = words[0]
498 game = @game
499 game.check_setup if SETUP_COMMANDS.include?(command)
500 if TURN_COMMANDS.include?(command)
501 player = words[1].to_i - 1
502 game.check_turn(player)
503 output = []
504 case command
505 when 'FIRE'
506 targets = words.drop(2).map { |word| parse_cell(word) }
507 allowance = game.allowance(player)
508 raise CommandError, "expected #{allowance} shots" unless targets.length == allowance
509
510 targets.each do |cell|
511 raise CommandError, "reef at #{coordinate(cell)}" if game.reefs[cell]
512 end
513 game.shooting_turn(player, targets: targets, output: output)
514 when 'AI'
515 game.shooting_turn(player, strategy: words[2], output: output)
516 when 'TORPEDO'
517 raise CommandError, 'torpedo used' if game.players[player].torpedo_used
518
519 game.torpedo(player, words[2].ord - 65, words[3] == 'W', output)
520 when 'SONAR'
521 raise CommandError, 'sonar used' if game.players[player].sonar_used
522
523 game.sonar(player, parse_cell(words[2]), output)
524 end
525 game.automatic(output)
526 return output
527 end
528
529 case command
530 when 'NEW'
531 wins = game.players.map(&:wins)
532 @game = Game.new
533 wins.each_with_index { |count, player| @game.players[player].wins = count }
534 ['NEW']
535 when 'PLACE'
536 player, index = words[1].to_i - 1, SHIP_NAMES.index(words[2])
537 raise CommandError, "#{words[2]} already placed" if game.players[player].fleet[index]
538
539 ship = position(parse_cell(words[3]), words[4] == 'H', SHIP_LENGTHS[index])
540 problem = game.placement_problem(player, ship)
541 raise CommandError, problem if problem
542
543 game.players[player].set_ship(index, ship)
544 ["PLACED #{player + 1} #{words[2]} #{ship.range}"]
545 when 'REMOVE'
546 player, index = words[1].to_i - 1, SHIP_NAMES.index(words[2])
547 raise CommandError, "#{words[2]} not placed" unless game.players[player].fleet[index]
548
549 game.players[player].set_ship(index, nil)
550 ["REMOVED #{player + 1} #{words[2]}"]
551 when 'AUTOPLACE'
552 player = words[1].to_i - 1
553 attempts = game.auto_place(player, words[2].to_i)
554 ["AUTOPLACE #{player + 1} attempts #{attempts}", *game.placed_lines(player)]
555 when 'LAYOUT'
556 player = words[1].to_i - 1
557 game.layout(player, words[2])
558 ["LAYOUT #{player + 1}", *game.placed_lines(player)]
559 when 'REEF'
560 cell = parse_cell(words[1])
561 game.add_reef(cell)
562 ["REEF #{coordinate(cell)}"]
563 when 'SALVO', 'CHAIN', 'TOUCH'
564 game.public_send("#{command.downcase}=", words[1] == 'ON')
565 ["#{command} #{words[1]}"]
566 when 'CONTROL'
567 player = words[1].to_i - 1
568 game.players[player].controller = words[2]
569 ["CONTROL #{player + 1} #{words[2]}"]
570 when 'SEED'
571 player, seed = words[1].to_i - 1, words[2].to_i
572 game.players[player].random = seed
573 ["SEED #{player + 1} #{seed}"]
574 when 'RESIGN'
575 game.resign(words[1].to_i - 1)
576 when 'HINT'
577 raise CommandError, 'game over' if game.phase == :over
578
579 player = words[1].to_i - 1
580 choice = game.choose(player, words[2])
581 ["HINT #{player + 1} #{words[2]} #{coordinate(choice.cell)} #{choice.info}"]
582 when 'DENSITY'
583 raise CommandError, 'game over' if game.phase == :over
584
585 game.density(words[1].to_i - 1)
586 when 'SHOW', 'RADAR' then game.show(words[1].to_i - 1, command == 'RADAR')
587 when 'EXPORT'
588 player = words[1].to_i - 1
589 ["EXPORT #{player + 1} #{game.export_fleet(player)}"]
590 when 'FLEET' then game.fleet_lines(words[1].to_i - 1)
591 when 'STATUS' then [game.status]
592 when 'STATS' then game.stats_lines
593 when 'MATCH', 'SERIES', 'TOURNEY' then simulation(words)
594 end
595 end

597 def simulation(words)
598 if words[0] == 'MATCH'
599 output = ["MATCH 1 #{words[3]} 2 #{words[4]}"]
600 match = @game.simulate(words[1].to_i, words[2].to_i, words[3], words[4], sinks: output)
601 output << "RESULT winner #{match.winner + 1} shots #{match.shots.join(' ')} hits #{match.hits.join(' ')}"
602 return output
603 end
604
605 count, random = words[1].to_i, words[2].to_i
606 if words[0] == 'SERIES'
607 wins, shots, fastest = [0, 0], [0, 0], [nil, nil]
608 count.times do
609 random = advance(random)
610 first_seed = random
611 random = advance(random)
612 match = @game.simulate(first_seed, random, words[3], words[4])
613 wins[match.winner] += 1
614 2.times { |player| shots[player] += match.shots[player] }
615 winning_shots = match.shots[match.winner]
616 fastest[match.winner] = [fastest[match.winner] || winning_shots, winning_shots].min
617 end
618 return ["SERIES #{count} wins #{wins.join(' ')} shots #{shots.join(' ')} fastest #{fastest.map { |value| value || '-' }.join(' ')}"]
619 end
620
621 wins, shots = [0, 0, 0], [0, 0, 0]
622 STRATEGIES.each_index do |first|
623 STRATEGIES.each_index do |second|
624 next if first == second
625
626 count.times do
627 random = advance(random)
628 first_seed = random
629 random = advance(random)
630 match = @game.simulate(first_seed, random, STRATEGIES[first], STRATEGIES[second])
631 wins[match.winner.zero? ? first : second] += 1
632 shots[first] += match.shots[0]
633 shots[second] += match.shots[1]
634 end
635 end
636 end
637 ranking = STRATEGIES.each_index.sort_by { |index| [-wins[index], shots[index], STRATEGIES[index]] }
