void finish_turn(Game &game, int player, bool hit, int mode)
    game.players[player].turns += 1
    if defeated(game.players[1 - player])
        game.phase = "OVER"
        game.winner = player
        game.players[player].wins += 1
        if mode == 0
            emit("WINNER {player + 1}")
    else
        game.current = if game.chain and hit then player else 1 - player
void shooting_turn(Game &game, int player, string strategy, const int[] cells, mode := 0)
    game.phase = "BATTLE"
    count := allowance(game, player)
    hit := false
    for: 0 <= index < count
        int cell
        if strategy == ""
            cell = cells[index]
        else
            choice := choose(game, player, strategy)
            cell = choice.cell
            game.players[player].state = choice.state
        result := shoot(&game, player, cell)
        hit = hit or result.hit
        if mode == 0
            emit("SHOT {player + 1} {coordinate(cell)} {result.text}")
        if defeated(game.players[1 - player])
            break
    finish_turn(&game, player, hit, mode)
void | error command(Program &program, const string[] tokens)
    op := tokens[0]
    Game game := program.game
    player := 0
    player_commands := ["PLACE", "REMOVE", "AUTOPLACE", "LAYOUT", "FIRE", "AI", "TORPEDO", "SONAR", "RESIGN", "CONTROL", "SEED", "HINT", "DENSITY", "SHOW", "RADAR", "EXPORT", "FLEET"]
    if index_of(player_commands, op) >= 0
        player = int(tokens[1]) - 1
    if index_of(["FIRE", "AI", "TORPEDO", "SONAR"], op) >= 0
        if game.phase == "OVER"
            return error("game over")
        if game.phase == "SETUP" and (placed_count(game.players[0]) < 5 or placed_count(game.players[1]) < 5)
            return error("fleets not ready")
        if player != game.current
            return error("not your turn")
        if op == "FIRE"
            count := allowance(game, player)
            if int(#tokens) - 2 != count
                return error("expected {count} shots")
            cells := [for token := tokens[2:][] then parse_cell(token)]
            for cell := cells[]
                if game.reefs[cell]
                    return error("reef at {coordinate(cell)}")
            shooting_turn(&game, player, "", cells)
        elif op == "AI"
            shooting_turn(&game, player, tokens[2], [])
        elif op == "TORPEDO"
            if game.players[player].torpedo
                return error("torpedo used")
            torpedo_turn(&game, player, tokens[2].codepoints()[0] - 65, tokens[3])
        else
            if game.players[player].sonar
                return error("sonar used")
            sonar_turn(&game, player, parse_cell(tokens[2]))
        automatic(&game)
        program.game = game
        return void
