247Choice choose(const Game game, int player, string strategy)
248 view := view_of(game, player)
249 excluded := excluded_cells(game, view)
250 candidates := [for cell := view.indices()[] then view[cell] == "." and not excluded[cell]]
251 generator := game.players[player].state
252 if strategy == "PROB"
253 scores := density(game, player, view, excluded)
254 unknown := [for cell := view.indices()[] if view[cell] == "." then int(cell)]
255 best := [for cell := unknown[] then scores[cell]].max()
256 tied := [for cell := unknown[] if scores[cell] == best then cell]
257 return Choice(tied[0], "{best}", generator)
258 if strategy == "RANDOM"
259 choices := [for cell := candidates.indices()[] if candidates[cell] then int(cell)]
260 generator = advance(generator)
261 return Choice(choices[generator % int(#choices)], "{#choices}", generator)
262 for: 0 <= cell < 100
263 if view[cell] != "x"
264 continue
265 for vertical := [false, true][]
266 dr := if vertical then 1 else 0
267 dc := if vertical then 0 else 1
268 front := step(cell, dr, dc)
269 if front < 0 or view[front] != "x"
270 continue
271 while front >= 0 and view[front] == "x"
272 front = step(front, dr, dc)
273 back := step(cell, -dr, -dc)
274 if can_choose(candidates, front)
275 return Choice(front, "LINE", generator)
276 if can_choose(candidates, back)
277 return Choice(back, "LINE", generator)
278 for cell := game.players[player].candidates[::-1][]
279 if candidates[cell]
280 return Choice(cell, "STACK", generator)
281 int[] remaining := []
282 for: 0 <= ship < 5
283 if not is_sunk(game.players[1 - player], ship)
284 remaining = remaining.append(ship_length(ship))
285 shortest := remaining.min()
286 parity := [for cell := candidates.indices()[] if candidates[cell] and (int(cell) / 10 + int(cell) % 10) % shortest == 0 then int(cell)]
287 return Choice(parity[0], "PARITY", generator)
