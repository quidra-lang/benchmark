26step <- function(cell, dr, dc) {
27 row <- (cell - 1L) %/% 10L + dr
28 column <- (cell - 1L) %% 10L + dc
29 if (row < 0L || row >= 10L || column < 0L || column >= 10L) return(0L)
30 row * 10L + column + 1L
31}

33footprint <- function(start, direction, length) {
34 position <- if (direction == "H") (start - 1L) %% 10L else (start - 1L) %/% 10L
35 if (position + length > 10L) return(integer())
36 start + (seq_len(length) - 1L) * if (direction == "H") 1L else 10L
37}

39AROUND <- lapply(seq_len(100L), function(cell) {
40 cells <- integer()
41 for (dr in -1L:1L) for (dc in -1L:1L) {
42 if (dr == 0L && dc == 0L) next
43 neighbor <- step(cell, dr, dc)
44 if (neighbor != 0L) cells <- c(cells, neighbor)
45 }
46 cells
47})
48ORTHOGONAL <- lapply(seq_len(100L), function(cell) {
49 cells <- c(step(cell, 0L, -1L), step(cell, 1L, 0L), step(cell, 0L, 1L), step(cell, -1L, 0L))
50 cells[cells != 0L]
51})

89placement_problem <- function(game, player, cells) {
90 if (!length(cells)) return("out of bounds")
91 if (any(game$reefs[cells])) return("on reef")
92 fleet <- game$players[[player]]$fleet
93 for (ship in seq_len(5L)) if (any(cells %in% fleet[[ship]])) return(paste("overlaps", SHIP_NAMES[ship]))
94 if (!game$touch) {
95 for (ship in seq_len(5L)) if (touching(cells, fleet[[ship]])) return(paste("touches", SHIP_NAMES[ship]))
96 }
97 NULL

104auto_place <- function(game, player, seed) {
105 owner <- game$players[[player]]
106 owner$state <- seed
107 owner$fleet <- replicate(5L, integer(), simplify = FALSE)
108 attempts <- 0
109 for (ship in seq_len(5L)) {
110 while (!length(owner$fleet[[ship]])) {
111 attempts <- attempts + 1
112 owner$state <- advance(owner$state)
113 direction <- if (owner$state %% 2L == 0L) "H" else "V"
114 owner$state <- advance(owner$state)
115 span <- 11L - SHIP_LENGTHS[ship]
116 index <- owner$state %% (10L * span)
117 start <- if (direction == "H") index %/% span * 10L + index %% span + 1L else index + 1L
118 cells <- footprint(start, direction, SHIP_LENGTHS[ship])
119 if (is.null(placement_problem(game, player, cells))) owner$fleet[[ship]] <- cells
120 }
121 }
122 attempts
123}

125layout_fleet <- function(game, text) {
126 chars <- strsplit(gsub("/", "", text, fixed = TRUE), "", fixed = TRUE)[[1L]]
127 fleet <- vector("list", 5L)
128 for (ship in seq_len(5L)) {
129 cells <- which(chars == SHIP_SYMBOLS[ship])
130 if (length(cells) != SHIP_LENGTHS[ship]) fail(paste(SHIP_NAMES[ship], "has", length(cells), "cells"))
131 if (!identical(cells, footprint(cells[1L], "H", SHIP_LENGTHS[ship])) &&
132 !identical(cells, footprint(cells[1L], "V", SHIP_LENGTHS[ship]))) fail(paste(SHIP_NAMES[ship], "not straight"))
133 if (any(game$reefs[cells])) fail(paste(SHIP_NAMES[ship], "on reef"))
134 fleet[[ship]] <- cells
135 }
136 if (!game$touch) {
137 for (first in seq_len(4L)) for (second in seq.int(first + 1L, 5L)) {
138 if (touching(fleet[[first]], fleet[[second]])) fail(paste(SHIP_NAMES[first], "touches", SHIP_NAMES[second]))
139 }
140 }
141 fleet
142}

174choose <- function(game, player, strategy) {
175 view <- view_of(game, player)
176 excluded <- excluded_cells(game, view)
177 candidates <- view == "." & !excluded
178 state <- game$players[[player]]$state
179 result <- function(cell, info, generator = state) list(cell = cell, info = info, state = generator)
180 if (strategy == "PROB") {
181 scores <- density(game, player, view, excluded)
182 unknown <- which(view == ".")
183 cell <- unknown[which.max(scores[unknown])]
184 return(result(cell, scores[cell]))
185 }
186 if (strategy == "RANDOM") {
187 cells <- which(candidates)
188 generator <- advance(state)
189 return(result(cells[generator %% length(cells) + 1L], length(cells), generator))
190 }
191 valid <- function(cell) cell > 0L && candidates[cell]
192 for (cell in which(view == "x")) {
193 for (direction in list(c(0L, 1L), c(1L, 0L))) {
194 dr <- direction[1L]
195 dc <- direction[2L]
196 front <- step(cell, dr, dc)
197 if (front == 0L || view[front] != "x") next
198 while (front != 0L && view[front] == "x") front <- step(front, dr, dc)
199 back <- step(cell, -dr, -dc)
200 if (valid(front)) return(result(front, "LINE"))
201 if (valid(back)) return(result(back, "LINE"))
202 }
203 }
204 for (cell in rev(game$players[[player]]$candidates)) if (candidates[cell]) return(result(cell, "STACK"))
205 remaining <- !vapply(seq_len(5L), function(ship) is_sunk(game$players[[3L - player]], ship), logical(1L))
206 shortest <- min(SHIP_LENGTHS[remaining])
207 cells <- which(candidates)
208 parity <- ((cells - 1L) %/% 10L + (cells - 1L) %% 10L) %% shortest == 0L
209 result(cells[which(parity)[1L]], "PARITY")
210}

212shoot <- function(game, player, cell) {
213 attacker <- game$players[[player]]
214 opponent <- game$players[[3L - player]]
215 text <- "ALREADY"
216 hit <- FALSE
217 sunk <- 0L
218 if (!opponent$attacked[cell]) {
219 opponent$attacked[cell] <- TRUE
220 ship <- board_of(opponent)[cell]
221 if (ship == 0L) text <- "MISS"
222 else {
223 hit <- TRUE
224 if (is_sunk(opponent, ship)) {
225 text <- paste("SUNK", SHIP_NAMES[ship])
226 sunk <- ship
227 } else {
228 text <- "HIT"
229 attacker$candidates <- c(attacker$candidates, ORTHOGONAL[[cell]])
230 }
231 }
232 }
233 attacker$shots <- attacker$shots + 1L
234 if (hit) {
235 attacker$hits <- attacker$hits + 1L
236 attacker$streak <- attacker$streak + 1L
237 attacker$best_streak <- max(attacker$best_streak, attacker$streak)
238 } else attacker$streak <- 0L
239 if (sunk > 0L) attacker$sinks <- attacker$sinks + 1L
240 list(text = text, hit = hit, sunk = sunk)
241}

254shooting_turn <- function(game, player, strategy, cells = integer(), mode = "normal") {
255 game$phase <- "BATTLE"
256 count <- allowance(game, player)
257 hit <- FALSE
258 for (index in seq_len(count)) {
259 if (is.null(strategy)) cell <- cells[index]
260 else {
261 choice <- choose(game, player, strategy)
262 cell <- choice$cell
263 game$players[[player]]$state <- choice$state
264 }
265 shot <- shoot(game, player, cell)
266 hit <- hit || shot$hit
267 if (mode == "normal") emit("SHOT", player, coordinate(cell), shot$text)
268 if (mode == "match" && shot$sunk > 0L) emit("SINK", player, SHIP_NAMES[shot$sunk], game$players[[player]]$shots)
269 if (defeated(game$players[[3L - player]])) break
270 }
271 finish_turn(game, player, hit, mode)
272}

283torpedo_turn <- function(game, player, row, side) {
284 game$phase <- "BATTLE"
285 owner <- game$players[[player]]
286 opponent <- game$players[[3L - player]]
287 owner$torpedo <- TRUE
288 occupied <- board_of(opponent)
289 columns <- if (side == "W") seq_len(10L) else 10L:1L
290 result <- "MISS"
291 hit <- FALSE
292 stopped_ship <- FALSE
293 for (column in columns) {
294 cell <- (row - 1L) * 10L + column
295 if (game$reefs[cell]) {
296 result <- paste("REEF", coordinate(cell))
297 break
298 }
299 if (occupied[cell] > 0L) {
300 shot <- shoot(game, player, cell)
301 result <- paste(shot$text, coordinate(cell))
302 hit <- shot$hit
303 stopped_ship <- TRUE
304 break
305 }
306 opponent$attacked[cell] <- TRUE
307 }
308 if (!stopped_ship) {
309 owner$shots <- owner$shots + 1L
310 owner$streak <- 0L
311 }
312 emit("TORPEDO", player, LETTERS[row], side, result)
313 finish_turn(game, player, hit, "normal")
314}

396play_match <- function(current, first_seed, second_seed, first_strategy, second_strategy, mode = "quiet") {
397 game <- new_game()
398 game$salvo <- current$salvo
399 game$chain <- current$chain
400 game$touch <- current$touch
401 game$reefs <- current$reefs
402 auto_place(game, 1L, first_seed)
403 auto_place(game, 2L, second_seed)
404 game$phase <- "BATTLE"
405 strategies <- c(first_strategy, second_strategy)
406 while (game$phase == "BATTLE") {
407 player <- game$current
408 shooting_turn(game, player, strategies[player], mode = mode)
409 }
410 game
411}

413series <- function(current, count, seed, first_strategy, second_strategy) {
414 wins <- shots <- c(0L, 0L)
415 fastest <- rep(NA_integer_, 2L)
416 generator <- seed
417 for (index in seq_len(count)) {
418 first_seed <- advance(generator)
419 generator <- advance(first_seed)
420 game <- play_match(current, first_seed, generator, first_strategy, second_strategy)
421 winner <- game$winner
422 wins[winner] <- wins[winner] + 1L
423 shots <- shots + vapply(game$players, function(player) player$shots, integer(1L))
424 won_shots <- game$players[[winner]]$shots
425 fastest[winner] <- if (is.na(fastest[winner])) won_shots else min(fastest[winner], won_shots)
426 }
427 minimums <- ifelse(is.na(fastest), "-", as.character(fastest))
428 emit("SERIES", count, "wins", wins, "shots", shots, "fastest", minimums)
429}

431tourney <- function(current, count, seed) {
432 wins <- shots <- integer(3L)
433 generator <- seed
434 for (first in seq_len(3L)) for (second in seq_len(3L)) {
435 if (first == second) next
436 for (index in seq_len(count)) {
437 first_seed <- advance(generator)
438 generator <- advance(first_seed)
439 game <- play_match(current, first_seed, generator, STRATEGIES[first], STRATEGIES[second])
440 winner <- if (game$winner == 1L) first else second
441 wins[winner] <- wins[winner] + 1L
442 shots[c(first, second)] <- shots[c(first, second)] + vapply(game$players, function(player) player$shots, integer(1L))
443 }
444 }
445 ranking <- order(-wins, shots, STRATEGIES, method = "radix")
446 emit("TOURNEY", count)
447 for (rank in seq_along(ranking)) {
448 strategy <- ranking[rank]
449 emit(rank, STRATEGIES[strategy], "wins", wins[strategy], "shots", shots[strategy])
450 }
451}

453command <- function(program, tokens) {
454 op <- tokens[1L]
455 if (op == "NEW") {
456 wins <- vapply(program$game$players, function(player) player$wins, integer(1L))
457 program$game <- new_game(wins)
458 emit("NEW")
459 return(invisible(NULL))
460 }
461 game <- program$game
462 player <- if (op %in% PLAYER_COMMANDS) as.integer(tokens[2L]) else 0L
463 if (op %in% SETUP_COMMANDS && game$phase != "SETUP") fail("setup closed")
464 if (op %in% TURN_COMMANDS) {
465 if (game$phase == "OVER") fail("game over")
466 if (game$phase == "SETUP" && any(vapply(game$players, placed_count, integer(1L)) < 5L)) fail("fleets not ready")
467 if (player != game$current) fail("not your turn")
468 if (op == "FIRE") {
469 count <- allowance(game, player)
470 if (length(tokens) - 2L != count) fail(paste("expected", count, "shots"))
471 cells <- vapply(tokens[-c(1L, 2L)], parse_cell, integer(1L), USE.NAMES = FALSE)
472 for (cell in cells) if (game$reefs[cell]) fail(paste("reef at", coordinate(cell)))
473 shooting_turn(game, player, NULL, cells)
474 } else if (op == "AI") shooting_turn(game, player, tokens[3L])
475 else if (op == "TORPEDO") {
476 if (game$players[[player]]$torpedo) fail("torpedo used")
477 torpedo_turn(game, player, match(tokens[3L], LETTERS), tokens[4L])
478 } else {
479 if (game$players[[player]]$sonar) fail("sonar used")
480 sonar_turn(game, player, parse_cell(tokens[3L]))
481 }
482 automatic(game)
483 return(invisible(NULL))
484 }
485 switch(op,
486 PLACE =, REMOVE = {
487 ship <- match(tokens[3L], SHIP_NAMES)
488 owner <- game$players[[player]]
489 if (op == "PLACE") {
490 if (length(owner$fleet[[ship]])) fail(paste(SHIP_NAMES[ship], "already placed"))
491 cells <- footprint(parse_cell(tokens[4L]), tokens[5L], SHIP_LENGTHS[ship])
492 problem <- placement_problem(game, player, cells)
493 if (!is.null(problem)) fail(problem)
494 owner$fleet[[ship]] <- cells
495 emit("PLACED", player, SHIP_NAMES[ship], ship_range(cells))
496 } else {
497 if (!length(owner$fleet[[ship]])) fail(paste(SHIP_NAMES[ship], "not placed"))
498 owner$fleet[[ship]] <- integer()
499 emit("REMOVED", player, SHIP_NAMES[ship])
500 }
501 },
502 AUTOPLACE = {
503 attempts <- auto_place(game, player, as.integer(tokens[3L]))
504 emit("AUTOPLACE", player, "attempts", attempts)
505 placement_output(game, player)
506 },
507 LAYOUT = {
508 fleet <- layout_fleet(game, tokens[3L])
509 game$players[[player]]$fleet <- fleet
510 emit("LAYOUT", player)
511 placement_output(game, player)
512 },
513 REEF = {
514 cell <- parse_cell(tokens[2L])
515 if (game$reefs[cell]) fail(paste("reef exists at", coordinate(cell)))
516 for (owner in seq_len(2L)) {
517 ship <- board_of(game$players[[owner]])[cell]
518 if (ship > 0L) fail(paste(coordinate(cell), "occupied by", owner, SHIP_NAMES[ship]))
519 }
520 game$reefs[cell] <- TRUE
521 emit("REEF", coordinate(cell))
522 },
523 SALVO =, CHAIN =, TOUCH = {
524 game[[tolower(op)]] <- tokens[2L] == "ON"
525 emit(op, tokens[2L])
526 },
527 CONTROL = {
528 game$players[[player]]$controller <- tokens[3L]
529 emit("CONTROL", player, tokens[3L])
530 },
531 SEED = {
532 game$players[[player]]$state <- as.integer(tokens[3L])
533 emit("SEED", player, game$players[[player]]$state)
534 },
535 RESIGN = {
536 if (game$phase != "BATTLE") fail("not in battle")
537 game$phase <- "OVER"
538 game$winner <- 3L - player
539 game$players[[game$winner]]$wins <- game$players[[game$winner]]$wins + 1L
540 emit("RESIGN", player)
