76function advance(int $state): int

78 return (48271 * $state) % 2147483647;
79}

84function step(int $cell, int $dr, int $dc): int
85{
86 $row = intdiv($cell, 10) + $dr;
87 $column = ($cell % 10) + $dc;
88 return $row >= 0 && $row < 10 && $column >= 0 && $column < 10 ? $row * 10 + $column : -1;
89}
90function surroundings(int $cell): array
91{
92 static $cache = null;
93 if ($cache === null) {
94 $cache = array_fill(0, 100, []);
95 for ($at = 0; $at < 100; $at++) {
96 for ($dr = -1; $dr <= 1; $dr++) {
97 for ($dc = -1; $dc <= 1; $dc++) {
98 if ($dr === 0 && $dc === 0) {
99 continue;
100 }
101 $neighbor = step($at, $dr, $dc);
102 if ($neighbor >= 0) {
103 $cache[$at][] = $neighbor;
104 }
105 }
106 }
107 }
108 }
109 return $cache[$cell];
110}
111function positions(int $start, bool $horizontal, int $length): ?array
112{
113 if (($horizontal ? $start % 10 : intdiv($start, 10)) + $length > 10) {
114 return null;
115 }
116 $cells = [];
117 for ($i = 0; $i < $length; $i++) {
118 $cells[] = $start + $i * ($horizontal ? 1 : 10);
119 }
120 return $cells;
121}
122function touching(array $first, array $second): bool
123{
124 foreach ($first as $cell) {
125 foreach (surroundings($cell) as $neighbor) {
126 if (in_array($neighbor, $second, true)) {
127 return true;
128 }
129 }
130 }
131 return false;
132}

134final class Player
135{
136 public array $ships = [[], [], [], [], []];
137 public array $owner;
138 public array $attacked;
139 public array $candidates = [];
140 public ?Strategy $controller = null;
141 public bool $torpedoUsed = false,
142 $sonarUsed = false;
143 public int $shots = 0,
144 $hits = 0,
145 $sinks = 0,
146 $turns = 0,
147 $streak = 0,
148 $bestStreak = 0,
149 $wins = 0;
150 public function __construct(public int $random)
151 {
152 $this->owner = array_fill(0, 100, -1);
153 $this->attacked = array_fill(0, 100, false);
154 }
155 public function sunk(int $ship): bool
156 {
157 if ($this->ships[$ship] === []) {
158 return false;
159 }
160 foreach ($this->ships[$ship] as $cell) {
161 if (!$this->attacked[$cell]) {
162 return false;
163 }
164 }
165 return true;
166 }
167 public function placed(): int
168 {
169 return count(array_filter($this->ships, static fn(array $cells): bool => $cells !== []));
170 }
171 public function afloat(): int
172 {
173 $count = 0;
174 for ($ship = 0; $ship < 5; $ship++) {
175 if ($this->ships[$ship] !== [] && !$this->sunk($ship)) {
176 $count++;
177 }
178 }
179 return $count;
180 }
181 public function setShip(int $ship, array $cells): void
182 {
183 foreach ($this->ships[$ship] as $cell) {
184 $this->owner[$cell] = -1;
185 }
186 $this->ships[$ship] = $cells;
187 foreach ($cells as $cell) {
188 $this->owner[$cell] = $ship;
189 }
190 }
191 public function clearFleet(): void
192 {
193 $this->ships = [[], [], [], [], []];
194 $this->owner = array_fill(0, 100, -1);
195 }
196 public function record(Shot $result): void
197 {
198 $this->shots++;
199 if ($result->isHit()) {
200 $this->hits++;
201 $this->streak++;
202 $this->bestStreak = max($this->bestStreak, $this->streak);
203 } else {
204 $this->streak = 0;
205 }
206 if ($result->kind === ShotKind::Sunk) {
207 $this->sinks++;
208 }
209 }
210}

268 public function placementProblem(int $player, ?array $cells): ?string
269 {
270 if ($cells === null) {
271 return "out of bounds";
272 }
273 foreach ($cells as $cell) {
274 if ($this->reefs[$cell]) {
275 return "on reef";
276 }
277 }
278 $first = 5;
279 foreach ($cells as $cell) {
280 $ship = $this->players[$player]->owner[$cell];
281 if ($ship >= 0) {
282 $first = min($first, $ship);
283 }
284 }
285 if ($first < 5) {
286 return "overlaps " . SHIP_NAMES[$first];
287 }
288 if (!$this->touch) {
289 foreach ($cells as $cell) {
290 foreach (surroundings($cell) as $neighbor) {
291 $ship = $this->players[$player]->owner[$neighbor];
292 if ($ship >= 0) {
293 $first = min($first, $ship);
294 }
295 }
296 }
297 if ($first < 5) {
298 return "touches " . SHIP_NAMES[$first];
299 }
300 }
301 return null;

303 public function autoPlace(int $player, int $seed): int
304 {
305 $owner = $this->players[$player];
306 $owner->random = $seed;
307 $owner->clearFleet();
308 $attempts = 0;
309 for ($ship = 0; $ship < 5; $ship++) {
310 while (true) {
311 $attempts++;
312 $length = SHIP_LENGTHS[$ship];
313 $owner->random = advance($owner->random);
314 $horizontal = $owner->random % 2 === 0;
315 $owner->random = advance($owner->random);
316 $k = $owner->random % (10 * (11 - $length));
317 $start = $horizontal ? intdiv($k, 11 - $length) * 10 + ($k % (11 - $length)) : $k;
318 $cells = positions($start, $horizontal, $length);
319 if ($this->placementProblem($player, $cells) !== null) {
320 continue;
321 }
322 $owner->setShip($ship, $cells);
323 break;
324 }
325 }
326 return $attempts;

328 public function layout(int $player, string $text): void
329 {
330 $grid = str_replace("/", "", $text);
331 $fleet = [];
332 for ($ship = 0; $ship < 5; $ship++) {
333 $cells = [];
334 for ($cell = 0; $cell < 100; $cell++) {
335 if ($grid[$cell] === SHIP_SYMBOLS[$ship]) {
336 $cells[] = $cell;
337 }
338 }
339 $name = SHIP_NAMES[$ship];
340 $length = SHIP_LENGTHS[$ship];
341 if (count($cells) !== $length) {
342 throw new CommandError("$name has " . count($cells) . " cells");
343 }
344 $horizontal = true;
345 $vertical = true;
346 for ($i = 0; $i < $length; $i++) {
347 $horizontal =
348 $horizontal &&
349 $cells[$i] === $cells[0] + $i &&
350 intdiv($cells[$i], 10) === intdiv($cells[0], 10);
351 $vertical = $vertical && $cells[$i] === $cells[0] + 10 * $i;
352 }
353 if (!$horizontal && !$vertical) {
354 throw new CommandError("$name not straight");
355 }
356 foreach ($cells as $cell) {
357 if ($this->reefs[$cell]) {
358 throw new CommandError("$name on reef");
359 }
360 }
361 $fleet[] = $cells;
362 }
363 if (!$this->touch) {
364 for ($first = 0; $first < 5; $first++) {
365 for ($second = $first + 1; $second < 5; $second++) {
366 if (touching($fleet[$first], $fleet[$second])) {
367 throw new CommandError(
368 SHIP_NAMES[$first] . " touches " . SHIP_NAMES[$second],
369 );
370 }
371 }
372 }
373 }
374 $this->players[$player]->clearFleet();
375 foreach ($fleet as $ship => $cells) {
376 $this->players[$player]->setShip($ship, $cells);
377 }
378 }

386 public function view(int $player): View
387 {
388 $opponent = $this->players[1 - $player];
389 $sunk = [];
390 $remaining = [];
391 for ($ship = 0; $ship < 5; $ship++) {
392 $sunk[$ship] = $opponent->sunk($ship);
393 if (!$sunk[$ship]) {
394 $remaining[] = $ship;
395 }
396 }
397 $cells = [];
398 for ($cell = 0; $cell < 100; $cell++) {
399 $ship = $opponent->owner[$cell];
400 $cells[] = $this->reefs[$cell]
401 ? ViewKind::Reef
402 : (!$opponent->attacked[$cell]
403 ? ViewKind::Unattacked
404 : ($ship < 0
405 ? ViewKind::Miss
406 : ($sunk[$ship]
407 ? ViewKind::Sunk
408 : ViewKind::Hit)));
409 }
410 $excluded = array_fill(0, 100, false);
411 if (!$this->touch) {
412 foreach ($cells as $cell => $kind) {
413 if ($kind === ViewKind::Sunk) {
414 foreach (surroundings($cell) as $neighbor) {
415 $excluded[$neighbor] = true;
416 };
417 }
418 }
419 }
420 return new View($cells, $excluded, $remaining);
421 }
422 public static function scores(View $view): array
423 {
424 $scores = array_fill(0, 100, 0);
425 foreach ($view->remaining as $ship) {
426 for ($start = 0; $start < 100; $start++) {
427 foreach ([true, false] as $horizontal) {
428 $cells = positions($start, $horizontal, SHIP_LENGTHS[$ship]);
429 if ($cells === null) {
430 continue;
431 }
432 $possible = true;
433 $hits = 0;
434 foreach ($cells as $cell) {
435 if (
436 $view->excluded[$cell] ||
437 ($view->cells[$cell] !== ViewKind::Unattacked &&
438 $view->cells[$cell] !== ViewKind::Hit)
439 ) {
440 $possible = false;
441 }
442 if ($view->cells[$cell] === ViewKind::Hit) {
443 $hits++;
444 }
445 }
446 if ($possible) {
447 foreach ($cells as $cell) {
448 $scores[$cell] += 1 + 10 * $hits;
449 };
450 }
451 }
452 }
453 }
454 return $scores;
455 }
456 public static function best(View $view, array $scores): int
457 {
458 $chosen = -1;
459 foreach ($view->cells as $cell => $kind) {
460 if (
461 $kind === ViewKind::Unattacked &&
462 ($chosen < 0 || $scores[$cell] > $scores[$chosen])
463 ) {
464 $chosen = $cell;
465 }
466 }
467 return $chosen;
468 }
469 public function choose(int $player, Strategy $strategy, bool $advanceRandom = true): Choice
470 {
471 $view = $this->view($player);
472 if ($strategy === Strategy::Prob) {
473 $scores = self::scores($view);
474 $cell = self::best($view, $scores);
475 return new Choice($cell, (string) $scores[$cell]);
476 }
477 if ($strategy === Strategy::Random) {
478 $candidates = [];
479 for ($cell = 0; $cell < 100; $cell++) {
480 if ($view->candidate($cell)) {
481 $candidates[] = $cell;
482 }
483 }
484 $random = advance($this->players[$player]->random);
485 if ($advanceRandom) {
486 $this->players[$player]->random = $random;
487 }
488 return new Choice(
489 $candidates[$random % count($candidates)],
490 (string) count($candidates),
491 );
492 }
493 foreach ($view->cells as $cell => $kind) {
494 if ($kind !== ViewKind::Hit) {
495 continue;
496 }
497 foreach (LINE_DIRECTIONS as [$dr, $dc]) {
498 $front = step($cell, $dr, $dc);
499 if ($front < 0 || $view->cells[$front] !== ViewKind::Hit) {
500 continue;
501 }
502 while ($front >= 0 && $view->cells[$front] === ViewKind::Hit) {
503 $front = step($front, $dr, $dc);
504 }
505 $back = step($cell, -$dr, -$dc);
506 if ($view->candidate($front)) {
507 return new Choice($front, "LINE");
508 }
509 if ($view->candidate($back)) {
510 return new Choice($back, "LINE");
511 }
512 }
513 }
514 $stack = $this->players[$player]->candidates;
515 for ($i = count($stack) - 1; $i >= 0; $i--) {
516 if ($view->candidate($stack[$i])) {
517 return new Choice($stack[$i], "STACK");
518 }
519 }
520 $shortest = min(
521 array_map(static fn(int $ship): int => SHIP_LENGTHS[$ship], $view->remaining),
522 );
523 for ($cell = 0; $cell < 100; $cell++) {
524 if ($view->candidate($cell) && (intdiv($cell, 10) + ($cell % 10)) % $shortest === 0) {
525 return new Choice($cell, "PARITY");
526 }
527 }
528 throw new LogicException("No strategy candidate");
529 }
530 public function shoot(int $player, int $cell): Shot
531 {
532 $owner = $this->players[$player];
533 $opponent = $this->players[1 - $player];
534 if ($opponent->attacked[$cell]) {
535 $result = new Shot(ShotKind::Already);
536 } else {
537 $opponent->attacked[$cell] = true;
538 $ship = $opponent->owner[$cell];
539 $result =
540 $ship < 0
541 ? new Shot(ShotKind::Miss)
542 : ($opponent->sunk($ship)
543 ? new Shot(ShotKind::Sunk, $ship)
544 : new Shot(ShotKind::Hit));
545 if ($result->kind === ShotKind::Hit) {
546 foreach (STACK_DIRECTIONS as [$dr, $dc]) {
547 $neighbor = step($cell, $dr, $dc);
548 if ($neighbor >= 0) {
549 $owner->candidates[] = $neighbor;
550 }
551 }
552 }
553 }
554 $owner->record($result);
555 return $result;
556 }
557 private function finishTurn(int $player, bool $hit, LogMode $mode, array &$output): void
558 {
559 $this->players[$player]->turns++;
560 if ($this->players[1 - $player]->afloat() === 0) {
561 $this->phase = Phase::Over;
562 $this->winner = $player;
563 $this->players[$player]->wins++;
564 if ($mode === LogMode::Normal) {
565 $output[] = "WINNER " . ($player + 1);
566 }
567 } else {
568 $this->current = $this->chain && $hit ? $player : 1 - $player;
569 }
570 }
571 public function shootingTurn(
572 int $player,
573 ?Strategy $strategy,
574 array $targets,
575 LogMode $mode,
576 array &$output,
577 ): void {
578 $count = $this->allowance($player);
579 $this->phase = Phase::Battle;
580 $hit = false;
581 for ($i = 0; $i < $count; $i++) {
582 $cell = $strategy === null ? $targets[$i] : $this->choose($player, $strategy)->cell;
583 $result = $this->shoot($player, $cell);
584 $hit = $hit || $result->isHit();
585 if ($mode === LogMode::Normal) {
586 $output[] =
587 "SHOT " . ($player + 1) . " " . coordinate($cell) . " " . $result->text();
588 }
589 if ($mode === LogMode::Sinks && $result->kind === ShotKind::Sunk) {
590 $output[] =
591 "SINK " .
592 ($player + 1) .
593 " " .
594 SHIP_NAMES[$result->ship] .
595 " " .
596 $this->players[$player]->shots;
597 }
598 if ($this->players[1 - $player]->afloat() === 0) {
599 break;
600 }
601 }
602 $this->finishTurn($player, $hit, $mode, $output);
603 }
604 public function torpedo(int $player, int $row, bool $west, array &$output): void
605 {
606 $this->phase = Phase::Battle;
607 $this->players[$player]->torpedoUsed = true;
608 $opponent = $this->players[1 - $player];
609 $result = "MISS";
610 $stoppedAtShip = false;
611 $hit = false;
612 for ($i = 0; $i < 10; $i++) {
613 $cell = $row * 10 + ($west ? $i : 9 - $i);
614 if ($this->reefs[$cell]) {
615 $result = "REEF " . coordinate($cell);
616 break;
617 }
618 if ($opponent->owner[$cell] >= 0) {
619 $shot = $this->shoot($player, $cell);
620 $result = $shot->text() . " " . coordinate($cell);
621 $hit = $shot->isHit();
622 $stoppedAtShip = true;
623 break;
624 }
625 $opponent->attacked[$cell] = true;
626 }
627 if (!$stoppedAtShip) {
628 $this->players[$player]->record(new Shot(ShotKind::Miss));
629 }
630 $output[] =
631 "TORPEDO " .
632 ($player + 1) .
633 " " .
634 chr(65 + $row) .
635 " " .
636 ($west ? "W" : "E") .
637 " " .
638 $result;
639 $this->finishTurn($player, $hit, LogMode::Normal, $output);
640 }
641 public function sonar(int $player, int $cell, array &$output): void
642 {
643 $this->phase = Phase::Battle;
644 $this->players[$player]->sonarUsed = true;
645 $opponent = $this->players[1 - $player];
646 $count = 0;
647 foreach ([$cell, ...surroundings($cell)] as $at) {

649 $count++;
650 }
651 }
652 $output[] = "SONAR " . ($player + 1) . " " . coordinate($cell) . " " . $count;
653 $this->finishTurn($player, false, LogMode::Normal, $output);
654 }
