17 reason::String
18end

21coordinate(cell::Int) = string('A' + div(cell - 1, 10), (cell - 1) % 10 + 1)

27function neighbor(cell::Int, dr::Int, dc::Int)::Int
28 row, column = div(cell - 1, 10) + dr, (cell - 1) % 10 + dc
29 0 <= row < 10 && 0 <= column < 10 ? row * 10 + column + 1 : 0
30end

32adjacent(first::Int, second::Int) =
33 abs(div(first - 1, 10) - div(second - 1, 10)) <= 1 && abs((first - 1) % 10 - (second - 1) % 10) <= 1

35struct Ship
36 start::Int
37 stride::Int
38end
39ship_cells(ship::Ship, size::Int) = ship.start:ship.stride:ship.start + (size - 1) * ship.stride
40in_bounds(ship::Ship, size::Int) = (ship.stride == 1 ? (ship.start - 1) % 10 : div(ship.start - 1, 10)) + size <= 10
41ship_range(ship::Ship, size::Int) = string(coordinate(ship.start), "-", coordinate(ship.start + (size - 1) * ship.stride))
42touching(first::Ship, first_size::Int, second::Ship, second_size::Int) =
43 any(a -> any(b -> adjacent(a, b), ship_cells(second, second_size)), ship_cells(first, first_size))
44
45const PLACEMENTS = [[Ship(start, stride) for start in 1:100 for stride in (1, 10)
46 if in_bounds(Ship(start, stride), size)] for size in SHIP_LENGTHS]

95function attacked_count(player::Player, number::Int)::Int
96 ship = player.fleet[number]
97 ship === nothing ? 0 : count(cell -> player.attacked[cell], ship_cells(ship, SHIP_LENGTHS[number]))
98end
99is_sunk(player::Player, number::Int) = player.fleet[number] !== nothing && attacked_count(player, number) == SHIP_LENGTHS[number]
100placed_count(player::Player) = count(ship -> ship !== nothing, player.fleet)
101afloat_count(player::Player) = count(number -> player.fleet[number] !== nothing && !is_sunk(player, number), 1:5)
102defeated(player::Player) = all(number -> is_sunk(player, number), 1:5)
103allowance(game::Game, player::Int) = game.salvo ? afloat_count(game.players[player]) : 1

105function placement_problem(game::Game, player::Player, number::Int, ship::Ship)::Union{Nothing,String}
106 size = SHIP_LENGTHS[number]
107 in_bounds(ship, size) || return "out of bounds"
108 cells = ship_cells(ship, size)
109 any(cell -> game.reefs[cell], cells) && return "on reef"
110 for other in 1:5
111 any(cell -> player.board[cell] == other, cells) && return "overlaps $(SHIP_NAMES[other])"
112 end
113 if !game.touch
114 for other in 1:5
115 placed = player.fleet[other]
116 if placed !== nothing && touching(ship, size, placed, SHIP_LENGTHS[other])
117 return "touches $(SHIP_NAMES[other])"
118 end
119 end
120 end
121 nothing
122end

131function auto_place!(game::Game, player::Int, seed::Int64)::Int64
132 owner = game.players[player]
133 owner.random = seed
134 clear_fleet!(owner)
135 attempts = Int64(0)
136 for number in 1:5
137 size = SHIP_LENGTHS[number]
138 while owner.fleet[number] === nothing
139 attempts += 1
140 owner.random = advance(owner.random)
141 stride = iseven(owner.random) ? 1 : 10
142 owner.random = advance(owner.random)
143 index = Int(owner.random % (10 * (11 - size)))
144 start = stride == 1 ? div(index, 11 - size) * 10 + index % (11 - size) + 1 : index + 1
145 ship = Ship(start, stride)
146 placement_problem(game, owner, number, ship) === nothing && put_ship!(owner, number, ship)
147 end
148 end
149 attempts
150end

152function layout_fleet(game::Game, layout::String)::Vector{Ship}
153 chars = collect(replace(layout, "/" => ""))
154 fleet = Ship[]
155 for number in 1:5
156 cells = findall(==(SHIP_SYMBOLS[number]), chars)
157 length(cells) == SHIP_LENGTHS[number] || fail("$(SHIP_NAMES[number]) has $(length(cells)) cells")
158 ship = Ship(cells[1], cells[2] - cells[1])
159 if !(ship.stride in (1, 10)) || !in_bounds(ship, length(cells)) || cells != ship_cells(ship, length(cells))
160 fail("$(SHIP_NAMES[number]) not straight")
161 end
162 any(cell -> game.reefs[cell], cells) && fail("$(SHIP_NAMES[number]) on reef")
163 push!(fleet, ship)
164 end
165 if !game.touch
166 for first in 1:4, second in first+1:5
167 touching(fleet[first], SHIP_LENGTHS[first], fleet[second], SHIP_LENGTHS[second]) &&
168 fail("$(SHIP_NAMES[first]) touches $(SHIP_NAMES[second])")
169 end
170 end
171 fleet
172end

174struct View
175 cells::Vector{CellView}
176 excluded::BitVector
177 remaining::BitVector
178end

180function view_of(game::Game, player::Int)::View
181 opponent = game.players[3 - player]
182 remaining = BitVector([!is_sunk(opponent, number) for number in 1:5])
183 cells = fill(UNATTACKED, 100)
184 for cell in 1:100
185 number = opponent.board[cell]
186 cells[cell] = game.reefs[cell] ? REEF : !opponent.attacked[cell] ? UNATTACKED :
187 number == 0 ? MISS_VIEW : remaining[number] ? HIT_VIEW : SUNK_VIEW
188 end
189 excluded = falses(100)
190 if !game.touch
191 for cell in findall(==(SUNK_VIEW), cells), dr in -1:1, dc in -1:1
192 dr == dc == 0 && continue
193 other = neighbor(cell, dr, dc)
194 other != 0 && (excluded[other] = true)
195 end
196 end
197 View(cells, excluded, remaining)
198end
199candidate(view::View, cell::Int) = 1 <= cell <= 100 && view.cells[cell] == UNATTACKED && !view.excluded[cell]
200
201function density(view::View)::Vector{Int}
202 scores = zeros(Int, 100)
203 for number in 1:5
204 view.remaining[number] || continue
205 for ship in PLACEMENTS[number]
206 cells = ship_cells(ship, SHIP_LENGTHS[number])
207 all(cell -> !view.excluded[cell] && view.cells[cell] in (UNATTACKED, HIT_VIEW), cells) || continue
208 weight = 1 + 10 * count(cell -> view.cells[cell] == HIT_VIEW, cells)
209 scores[cells] .+= weight
210 end
211 end
212 scores
213end
214
215function best_cell(view::View, scores::Vector{Int})::Int
216 cells = findall(==(UNATTACKED), view.cells)
217 cells[argmax(scores[cells])]
218end

220struct Choice
221 cell::Int
222 info::Union{String,Int}
223 next_random::Int64
224end
225
226function choose(game::Game, player::Int, strategy::Strategy)::Choice
227 view = view_of(game, player)
228 owner = game.players[player]
229 if strategy == PROB
230 scores = density(view)
231 cell = best_cell(view, scores)
232 return Choice(cell, scores[cell], owner.random)
233 elseif strategy == RANDOM
234 cells = [cell for cell in 1:100 if candidate(view, cell)]
235 next = advance(owner.random)
236 return Choice(cells[Int(next % length(cells)) + 1], length(cells), next)
237 end
238 for cell in findall(==(HIT_VIEW), view.cells), (dr, dc) in ((0, 1), (1, 0))
239 front = neighbor(cell, dr, dc)
240 (front == 0 || view.cells[front] != HIT_VIEW) && continue
241 while front != 0 && view.cells[front] == HIT_VIEW
242 front = neighbor(front, dr, dc)
243 end
244 for target in (front, neighbor(cell, -dr, -dc))
245 candidate(view, target) && return Choice(target, "LINE", owner.random)
246 end
247 end
248 for cell in Iterators.reverse(owner.candidates)
249 candidate(view, cell) && return Choice(cell, "STACK", owner.random)
250 end
251 shortest = minimum(SHIP_LENGTHS[number] for number in 1:5 if view.remaining[number])
252 cell = something(findfirst(cell -> candidate(view, cell) && (div(cell - 1, 10) + (cell - 1) % 10) % shortest == 0, 1:100))
253 Choice(cell, "PARITY", owner.random)
254end

256struct Shot
257 result::ShotResult
258 ship::Union{Nothing,Int}
259end
260is_hit(shot::Shot) = shot.result == HIT || shot.result == SUNK
261shot_text(shot::Shot) = shot.result == SUNK ? "SUNK $(SHIP_NAMES[something(shot.ship)])" : string(shot.result)

263function record_shot!(statistics::Statistics, shot::Shot)
264 statistics.shots += 1
265 if is_hit(shot)
266 statistics.hits += 1
267 statistics.streak += 1
268 statistics.longest = max(statistics.longest, statistics.streak)
269 else
270 statistics.streak = 0
271 end
272 shot.result == SUNK && (statistics.sinks += 1)
273end

275function shoot!(game::Game, player::Int, cell::Int)::Shot
276 owner, opponent = game.players[player], game.players[3 - player]
277 number = opponent.board[cell]
278 result = ALREADY
279 if !opponent.attacked[cell]
280 opponent.attacked[cell] = true
281 result = number == 0 ? MISS : is_sunk(opponent, number) ? SUNK : HIT
282 end
283 if result == HIT
284 for (dr, dc) in ((0, -1), (1, 0), (0, 1), (-1, 0))
285 other = neighbor(cell, dr, dc)
286 other != 0 && push!(owner.candidates, other)
287 end
288 end
289 shot = Shot(result, number == 0 ? nothing : number)
290 record_shot!(owner.statistics, shot)
291 shot
292end

294function finish_turn!(game::Game, player::Int, hit::Bool, mode::OutputMode)
295 statistics = game.players[player].statistics
296 statistics.turns += 1
297 if defeated(game.players[3 - player])
298 game.phase = OVER
299 game.winner = player
300 statistics.wins += 1
301 mode == NORMAL && emit("WINNER", player)
302 else
303 game.current = game.chain && hit ? player : 3 - player
304 end
305end

307function shooting_turn!(game::Game, player::Int, strategy::Union{Nothing,Strategy}, cells::Vector{Int}=Int[], mode::OutputMode=NORMAL)
308 hit = false
309 for index in 1:allowance(game, player)
310 if strategy === nothing
311 cell = cells[index]
312 else
313 choice = choose(game, player, strategy)
314 cell = choice.cell
315 game.players[player].random = choice.next_random
316 end
317 shot = shoot!(game, player, cell)
318 hit |= is_hit(shot)
319 mode == NORMAL && emit("SHOT", player, coordinate(cell), shot_text(shot))
320 if mode == SINKS_ONLY && shot.result == SUNK
321 emit("SINK", player, SHIP_NAMES[something(shot.ship)], game.players[player].statistics.shots)
322 end
323 defeated(game.players[3 - player]) && break
324 end
325 finish_turn!(game, player, hit, mode)
326end

328function automatic!(game::Game)
329 while game.phase == BATTLE
330 player = game.current
331 strategy = game.players[player].controller
332 strategy === nothing && break
333 shooting_turn!(game, player, strategy)
334 end

337function torpedo_turn!(game::Game, player::Int, row::Int, side::String)
338 owner, opponent = game.players[player], game.players[3 - player]
339 owner.torpedo_used = true
340 columns = side == "W" ? (1:10) : (10:-1:1)
341 result = "MISS"
342 hit = false
343 stopped_ship = false
344 for column in columns
345 cell = (row - 1) * 10 + column
346 if game.reefs[cell]
347 result = "REEF $(coordinate(cell))"
348 break
349 elseif opponent.board[cell] != 0
350 shot = shoot!(game, player, cell)
351 result = "$(shot_text(shot)) $(coordinate(cell))"
352 hit = is_hit(shot)
353 stopped_ship = true
354 break
355 end
356 opponent.attacked[cell] = true
357 end
358 stopped_ship || record_shot!(owner.statistics, Shot(MISS, nothing))
359 emit("TORPEDO", player, 'A' + row - 1, side, result)
360 finish_turn!(game, player, hit, NORMAL)
361end

449function play_match(current::Game, first_seed::Int64, second_seed::Int64, strategies::NTuple{2,Strategy}, mode::OutputMode=QUIET)::Game
450 game = Game()
451 game.salvo, game.chain, game.touch = current.salvo, current.chain, current.touch
452 game.reefs = copy(current.reefs)
453 auto_place!(game, 1, first_seed)
454 auto_place!(game, 2, second_seed)
455 game.phase = BATTLE
456 while game.phase == BATTLE
457 player = game.current
458 shooting_turn!(game, player, strategies[player], Int[], mode)
459 end

463function series_output(game::Game, count::Int, seed::Int64, strategies::NTuple{2,Strategy})
464 wins, shots = zeros(Int64, 2), zeros(Int64, 2)
465 fastest = Union{Nothing,Int64}[nothing, nothing]
466 generator = seed
467 for _ in 1:count
468 first_seed = advance(generator)
469 generator = advance(first_seed)
470 trial = play_match(game, first_seed, generator, strategies)
471 winner = something(trial.winner)
472 wins[winner] += 1
473 shots .+= [player.statistics.shots for player in trial.players]
474 winning_shots = trial.players[winner].statistics.shots
475 best = fastest[winner]
476 fastest[winner] = best === nothing ? winning_shots : min(best, winning_shots)
477 end
478 emit("SERIES", count, "wins", wins..., "shots", shots..., "fastest", something(fastest[1], "-"), something(fastest[2], "-"))
479end

481function tourney_output(game::Game, count::Int, seed::Int64)
482 wins, shots = zeros(Int64, 3), zeros(Int64, 3)
483 generator = seed
484 for first in 1:3, second in 1:3
485 first == second && continue
486 for _ in 1:count
487 first_seed = advance(generator)
488 generator = advance(first_seed)
489 trial = play_match(game, first_seed, generator, (STRATEGIES[first], STRATEGIES[second]))
490 wins[something(trial.winner) == 1 ? first : second] += 1
491 shots[first] += trial.players[1].statistics.shots
492 shots[second] += trial.players[2].statistics.shots
493 end
494 end
495 ranking = sort(collect(1:3); by=number -> (-wins[number], shots[number], string(STRATEGIES[number])))
496 emit("TOURNEY", count)
497 for (rank, number) in enumerate(ranking)
498 emit(rank, STRATEGIES[number], "wins", wins[number], "shots", shots[number])
499 end

502function execute!(game::Game, tokens::Vector{String})::Game
503 command = tokens[1]
504 if command == "NEW"
505 replacement = Game()
506 for player in 1:2
507 replacement.players[player].statistics.wins = game.players[player].statistics.wins
508 end
509 emit("NEW")
510 return replacement
511 end
512 player = command in PLAYER_COMMANDS ? parse(Int, tokens[2]) : 0
513 command in SETUP_COMMANDS && game.phase != SETUP && fail("setup closed")
514 if command in TURN_COMMANDS
515 game.phase == OVER && fail("game over")
516 game.phase == SETUP && any(owner -> placed_count(owner) < 5, game.players) && fail("fleets not ready")
517 player == game.current || fail("not your turn")
518 cells = Int[]
519 if command == "FIRE"
520 expected = allowance(game, player)
521 length(tokens) - 2 == expected || fail("expected $expected shots")
522 cells = parse_cell.(tokens[3:end])
523 for cell in cells
524 game.reefs[cell] && fail("reef at $(coordinate(cell))")
525 end
526 elseif command == "TORPEDO"
527 game.players[player].torpedo_used && fail("torpedo used")
528 elseif command == "SONAR"
529 game.players[player].sonar_used && fail("sonar used")
530 end
531 game.phase = BATTLE
532 if command == "FIRE"
533 shooting_turn!(game, player, nothing, cells)
534 elseif command == "AI"
535 shooting_turn!(game, player, STRATEGY_BY_NAME[tokens[3]])
536 elseif command == "TORPEDO"
537 torpedo_turn!(game, player, Int(tokens[3][1] - 'A') + 1, tokens[4])
538 else
539 sonar_turn!(game, player, parse_cell(tokens[3]))
540 end
541 automatic!(game)
542 return game
543 end
544 if command == "PLACE" || command == "REMOVE"
545 number = ship_number(tokens[3])
546 owner = game.players[player]
547 if command == "PLACE"
548 owner.fleet[number] === nothing || fail("$(SHIP_NAMES[number]) already placed")
549 ship = Ship(parse_cell(tokens[4]), tokens[5] == "H" ? 1 : 10)
550 problem = placement_problem(game, owner, number, ship)
551 problem === nothing || fail(problem)
552 put_ship!(owner, number, ship)
553 emit("PLACED", player, SHIP_NAMES[number], ship_range(ship, SHIP_LENGTHS[number]))
554 else
555 ship = owner.fleet[number]
556 ship === nothing && fail("$(SHIP_NAMES[number]) not placed")
557 owner.board[ship_cells(ship, SHIP_LENGTHS[number])] .= 0
558 owner.fleet[number] = nothing
559 emit("REMOVED", player, SHIP_NAMES[number])
560 end
561 elseif command == "AUTOPLACE"
562 attempts = auto_place!(game, player, parse(Int64, tokens[3]))
563 emit("AUTOPLACE", player, "attempts", attempts)
564 placement_output(game, player)
565 elseif command == "LAYOUT"
566 fleet = layout_fleet(game, tokens[3])
567 clear_fleet!(game.players[player])
568 for number in 1:5
569 put_ship!(game.players[player], number, fleet[number])
570 end
571 emit("LAYOUT", player)
572 placement_output(game, player)
573 elseif command == "REEF"
574 cell = parse_cell(tokens[2])
575 game.reefs[cell] && fail("reef exists at $(coordinate(cell))")
576 for owner in 1:2
577 number = game.players[owner].board[cell]
578 number != 0 && fail("$(coordinate(cell)) occupied by $owner $(SHIP_NAMES[number])")
579 end
580 game.reefs[cell] = true
581 emit("REEF", coordinate(cell))
582 elseif command in ("SALVO", "CHAIN", "TOUCH")
583 enabled = tokens[2] == "ON"
584 if command == "SALVO"
585 game.salvo = enabled
586 elseif command == "CHAIN"
587 game.chain = enabled
588 else
589 game.touch = enabled
590 end
591 emit(command, tokens[2])
592 elseif command == "CONTROL"
593 game.players[player].controller = tokens[3] == "HUMAN" ? nothing : STRATEGY_BY_NAME[tokens[3]]
594 emit("CONTROL", player, tokens[3])
595 elseif command == "SEED"
596 game.players[player].random = parse(Int64, tokens[3])
597 emit("SEED", player, game.players[player].random)
598 elseif command == "RESIGN"
599 game.phase == BATTLE || fail("not in battle")
600 game.phase = OVER
601 game.winner = 3 - player
602 game.players[3 - player].statistics.wins += 1
603 emit("RESIGN", player)
604 emit("WINNER", 3 - player)
605 elseif command == "HINT"
606 game.phase == OVER && fail("game over")
607 choice = choose(game, player, STRATEGY_BY_NAME[tokens[3]])
608 emit("HINT", player, tokens[3], coordinate(choice.cell), choice.info)
609 elseif command == "DENSITY"
610 game.phase == OVER && fail("game over")
611 density_output(game, player)
612 elseif command == "SHOW" || command == "RADAR"
613 board_output(game, player, command == "RADAR")
614 elseif command == "EXPORT"
615 emit("EXPORT", player, export_fleet(game.players[player]))
616 elseif command == "FLEET"
617 fleet_output(game, player)
618 elseif command == "STATUS"
619 status_output(game)
620 elseif command == "STATS"
621 statistics_output(game)
622 elseif command == "MATCH"
623 strategies = (STRATEGY_BY_NAME[tokens[4]], STRATEGY_BY_NAME[tokens[5]])
624 emit("MATCH", 1, strategies[1], 2, strategies[2])
625 trial = play_match(game, parse(Int64, tokens[2]), parse(Int64, tokens[3]), strategies, SINKS_ONLY)
626 first, second = trial.players
627 emit("RESULT winner", something(trial.winner), "shots", first.statistics.shots, second.statistics.shots,
628 "hits", first.statistics.hits, second.statistics.hits)
629 elseif command == "SERIES"
630 series_output(game, parse(Int, tokens[2]), parse(Int64, tokens[3]), (STRATEGY_BY_NAME[tokens[4]], STRATEGY_BY_NAME[tokens[5]]))
631 elseif command == "TOURNEY"
632 tourney_output(game, parse(Int, tokens[2]), parse(Int64, tokens[3]))
633 end
634 game
635end
