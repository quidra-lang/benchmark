4SIZE = 10
5ROW_LETTERS = "ABCDEFGHIJ"
6SHIP_NAMES = ("CARRIER", "BATTLESHIP", "CRUISER", "SUBMARINE", "DESTROYER")
7SHIP_LENGTHS = (5, 4, 3, 3, 2)
8SHIP_SYMBOLS = "CBRSD"
9RULES = ("SALVO", "CHAIN", "TOUCH")
10STRATEGIES = ("HUNT", "PROB", "RANDOM")
11TOURNEY_PAIRS = (
12 ("HUNT", "PROB"), ("HUNT", "RANDOM"), ("PROB", "HUNT"),
13 ("PROB", "RANDOM"), ("RANDOM", "HUNT"), ("RANDOM", "PROB"),

15DIRECTIONS = {"H": (0, 1), "V": (1, 0)}

41def inside(cell):
42 row, col = cell
43 return 0 <= row < SIZE and 0 <= col < SIZE

46def surrounding(cell):
47 row, col = cell
48 cells = []
49 for dr in (-1, 0, 1):
50 for dc in (-1, 0, 1):
51 neighbour = (row + dr, col + dc)
52 if (dr, dc) != (0, 0) and inside(neighbour):
53 cells.append(neighbour)
54 return cells

57def orthogonal(cell):
58 row, col = cell
59 neighbours = [(row + dr, col + dc) for dr, dc in NEIGHBOUR_STEPS]
60 return [neighbour for neighbour in neighbours if inside(neighbour)]

81def is_straight(cells):
82 row, col = cells[0]
83 horizontal = all(cell == (row, col + step) for step, cell in enumerate(cells))
84 vertical = all(cell == (row + step, col) for step, cell in enumerate(cells))
85 return horizontal or vertical

96class Player:
97 def __init__(self, number):
98 self.number = number
99 self.ships = [None] * len(SHIP_NAMES)
100 self.attacked = set()
101 self.candidates = []
102 self.random = number
103 self.controller = "HUMAN"
104 self.torpedo_used = False
105 self.sonar_used = False
106 self.shots = 0
107 self.hits = 0
108 self.sinks = 0
109 self.turns = 0
110 self.streak = 0
111 self.longest_streak = 0
112 self.wins = 0
113
114 def advance_random(self):
115 self.random = next_random(self.random)
116 return self.random
117
118 def ship_at(self, cell):
119 for index, cells in enumerate(self.ships):
120 if cells is not None and cell in cells:
121 return index
122 return None
123
124 def attacked_count(self, index):
125 return sum(cell in self.attacked for cell in self.ships[index])
126
127 def is_sunk(self, index):
128 cells = self.ships[index]
129 return cells is not None and all(cell in self.attacked for cell in cells)
130
131 def placed_count(self):
132 return sum(cells is not None for cells in self.ships)
133
134 def fully_placed(self):
135 return self.placed_count() == len(SHIP_NAMES)
136
137 def afloat_count(self):
138 return sum(
139 cells is not None and not self.is_sunk(index)
140 for index, cells in enumerate(self.ships)
141 )
142
143 def defeated(self):
144 return all(self.is_sunk(index) for index in range(len(SHIP_NAMES)))
145
146 def record_shot(self, result):
147 self.shots += 1
148 if is_hit(result):
149 self.hits += 1
150 self.streak += 1
151 self.longest_streak = max(self.longest_streak, self.streak)
152 else:
153 self.streak = 0
154 if result.startswith("SUNK"):
155 self.sinks += 1
156
157 def accuracy(self):
158 if self.shots == 0:
159 return "-"
160 return f"{self.hits * 100 // self.shots}%"

163class View:
164 def __init__(self, game, number):
165 target = game.opponent(number)
166 self.states = {}
167 for cell in CELLS:
168 if cell in game.reefs:
169 self.states[cell] = "~"
170 elif cell in target.attacked:
171 self.states[cell] = "o"
172 else:
173 self.states[cell] = "."
174 for index, cells in enumerate(target.ships):
175 if cells is None:
176 continue
177 mark = "#" if target.is_sunk(index) else "x"
178 for cell in cells:
179 if cell in target.attacked:
180 self.states[cell] = mark
181 self.excluded = set()
182 if not game.rules["TOUCH"]:
183 for cell in CELLS:
184 if self.states[cell] == "#":
185 self.excluded.update(surrounding(cell))
186 self.remaining = [
187 index for index in range(len(SHIP_NAMES)) if not target.is_sunk(index)
188 ]
189
190 def state(self, cell):
191 return self.states.get(cell)
192
193 def is_candidate(self, cell):
194 return self.state(cell) == "." and cell not in self.excluded
195
196 def candidate_cells(self):
197 return [cell for cell in CELLS if self.is_candidate(cell)]
198
199 def hunt(self, stack):
200 for row, col in CELLS:
201 if self.states[(row, col)] != "x":
202 continue
203 for dr, dc in DIRECTIONS.values():
204 front = (row + dr, col + dc)
205 if self.state(front) != "x":
206 continue
207 while self.state(front) == "x":
208 front = (front[0] + dr, front[1] + dc)
209 back = (row - dr, col - dc)
210 if self.is_candidate(front):
211 return front, "LINE"
212 if self.is_candidate(back):
213 return back, "LINE"
214 for cell in reversed(stack):
215 if self.is_candidate(cell):
216 return cell, "STACK"
217 shortest = min(SHIP_LENGTHS[index] for index in self.remaining)
218 for row, col in CELLS:
219 if (row + col) % shortest == 0 and self.is_candidate((row, col)):
220 return (row, col), "PARITY"
221 return None
222
223 def possible(self, cells):
224 return all(
225 self.state(cell) in (".", "x") and cell not in self.excluded
226 for cell in cells
227 )
228
229 def scores(self):
230 scores = dict.fromkeys(CELLS, 0)
231 for index in self.remaining:
232 for start in CELLS:
233 for direction in DIRECTIONS:
234 cells = ship_cells(start, direction, SHIP_LENGTHS[index])
235 if not self.possible(cells):
236 continue
237 hits = sum(self.states[cell] == "x" for cell in cells)
238 for cell in cells:
239 scores[cell] += 1 + 10 * hits
240 return scores
241
242 def best(self, scores):
243 best_cell = None
244 for cell in CELLS:
245 if self.states[cell] != ".":
246 continue
247 if best_cell is None or scores[cell] > scores[best_cell]:
248 best_cell = cell
249 return best_cell, scores[best_cell]

252class Game:
253 def __init__(self):
254 self.phase = "SETUP"
255 self.current = 1
256 self.winner = None
257 self.rules = dict.fromkeys(RULES, False)
258 self.reefs = set()
259 self.players = (Player(1), Player(2))
260 self.sink_log = []
261
262 def player(self, number):
263 return self.players[number - 1]
264
265 def opponent(self, number):
266 return self.players[2 - number]
267
268 def fleets_ready(self):
269 return all(player.fully_placed() for player in self.players)
270
271 def placement_problem(self, player, cells):
272 if not all(inside(cell) for cell in cells):
273 return "out of bounds"
274 if any(cell in self.reefs for cell in cells):
275 return "on reef"
276 for index, placed in enumerate(player.ships):
277 if placed is not None and any(cell in placed for cell in cells):
278 return "overlaps " + SHIP_NAMES[index]
279 if not self.rules["TOUCH"]:
280 for index, placed in enumerate(player.ships):
281 if placed is not None and ships_touch(placed, cells):
282 return "touches " + SHIP_NAMES[index]
283 return None

285 def auto_place(self, number, seed):
286 player = self.player(number)
287 player.random = seed
288 player.ships = [None] * len(SHIP_NAMES)
289 attempts = 0
290 for index, length in enumerate(SHIP_LENGTHS):
291 starts = SIZE + 1 - length
292 while player.ships[index] is None:
293 attempts += 1
294 direction = "H" if player.advance_random() % 2 == 0 else "V"
295 k = player.advance_random() % (SIZE * starts)
296 if direction == "H":
297 start = divmod(k, starts)
298 else:
299 start = divmod(k, SIZE)
300 cells = ship_cells(start, direction, length)
301 if self.placement_problem(player, cells) is None:
302 player.ships[index] = cells
303 return attempts

305 def shot_allowance(self, number):
306 if self.rules["SALVO"]:
307 return self.player(number).afloat_count()
308 return 1
309
310 def begin_battle(self):
311 if self.phase == "SETUP":
312 self.phase = "BATTLE"
313
314 def shoot(self, number, cell):
315 shooter = self.player(number)
316 target = self.opponent(number)
317 sunk_ship = None
318 if cell in target.attacked:
319 result = "ALREADY"
320 else:
321 target.attacked.add(cell)
322 index = target.ship_at(cell)
323 if index is None:
324 result = "MISS"
325 elif target.is_sunk(index):
326 sunk_ship = SHIP_NAMES[index]
327 result = "SUNK " + sunk_ship
328 else:
329 result = "HIT"
330 shooter.candidates.extend(orthogonal(cell))
331 shooter.record_shot(result)
332 if sunk_ship is not None:
333 self.sink_log.append((number, sunk_ship, shooter.shots))
334 return result

336 def choose_target(self, number, strategy, commit=True):
337 view = View(self, number)
338 player = self.player(number)
339 if strategy == "HUNT":
340 return view.hunt(player.candidates)
341 if strategy == "PROB":
342 return view.best(view.scores())
343 cells = view.candidate_cells()
344 value = next_random(player.random)
345 if commit:
346 player.random = value
347 return cells[value % len(cells)], len(cells)

349 def ai_turn(self, number, strategy):
350 shots = []
351 for _ in range(self.shot_allowance(number)):
352 cell, _ = self.choose_target(number, strategy)
353 shots.append((cell, self.shoot(number, cell)))
354 if self.opponent(number).defeated():
355 break
356 return shots

358 def torpedo(self, number, row, side):
359 shooter = self.player(number)
360 target = self.opponent(number)
361 shooter.torpedo_used = True
362 columns = range(SIZE) if side == "W" else range(SIZE - 1, -1, -1)
363 result = "MISS"
364 for col in columns:
365 cell = (row, col)
366 if cell in self.reefs:
367 result = "REEF " + cell_name(cell)
368 break
369 if target.ship_at(cell) is not None:
370 return self.shoot(number, cell) + " " + cell_name(cell)
371 target.attacked.add(cell)
372 shooter.record_shot(result)
373 return result

375 def sonar(self, number, cell):
376 self.player(number).sonar_used = True
377 target = self.opponent(number)
378 area = [cell] + surrounding(cell)
379 return sum(
380 target.ship_at(spot) is not None and spot not in target.attacked
381 for spot in area
382 )

384 def end_turn(self, number, results):
385 player = self.player(number)
386 player.turns += 1
387 if self.opponent(number).defeated():
388 self.phase = "OVER"
389 self.winner = number
390 player.wins += 1
391 return True
392 if self.rules["CHAIN"] and any(is_hit(result) for result in results):
393 self.current = number
394 else:
395 self.current = 3 - number
396 return False
397
398 def resign(self, number):
399 self.phase = "OVER"
400 self.winner = 3 - number
401 self.player(self.winner).wins += 1
402
403 def board_symbol(self, player, cell):
404 if cell in self.reefs:
405 return "~"
406 index = player.ship_at(cell)
407 attacked = cell in player.attacked
408 if index is None:
409 return "o" if attacked else "."
410 symbol = SHIP_SYMBOLS[index]
411 return symbol.lower() if attacked else symbol

414def play_match(base, seeds, strategies):
415 game = Game()
416 game.rules = dict(base.rules)
417 game.reefs = set(base.reefs)
418 for number, seed in zip((1, 2), seeds):
419 game.auto_place(number, seed)
420 game.phase = "BATTLE"
421 game.current = 1
422 while game.phase == "BATTLE":
423 number = game.current
424 shots = game.ai_turn(number, strategies[number - 1])
425 game.end_turn(number, [result for _, result in shots])
426 return game

523 def layout(self, arguments):
524 number, text = int(arguments[0]), arguments[1]
525 game = self.setup_game()
526 rows = text.split("/")
527 fleet = []
528 for name, length, symbol in zip(SHIP_NAMES, SHIP_LENGTHS, SHIP_SYMBOLS):
529 cells = [(row, col) for row, col in CELLS if rows[row][col] == symbol]
530 if len(cells) != length:

533 raise CommandError(f"{name} not straight")
534 if any(cell in game.reefs for cell in cells):
535 raise CommandError(f"{name} on reef")
536 fleet.append(cells)
537 if not game.rules["TOUCH"]:
538 for first in range(len(fleet)):
539 for second in range(first + 1, len(fleet)):
540 if ships_touch(fleet[first], fleet[second]):
541 raise CommandError(
542 f"{SHIP_NAMES[first]} touches {SHIP_NAMES[second]}"
543 )
544 player = game.player(number)
545 player.ships = fleet
546 self.emit("LAYOUT", number)
547 self.emit_fleet(number, player)

713 def status(self, arguments):
714 game = self.game
715 first, second = game.players
716 parts = ["STATUS", game.phase]
717 for rule in RULES:
718 parts.extend((rule.lower(), switch_text(game.rules[rule])))
719 parts.extend(("reefs", len(game.reefs)))
720 if game.phase == "SETUP":
721 parts.extend(("placed", first.placed_count(), second.placed_count()))
722 else:
723 if game.phase == "BATTLE":
724 parts.extend((
725 "turn", game.current, "shots", game.shot_allowance(game.current),
726 ))
727 else:
728 parts.extend(("winner", game.winner))
729 parts.extend(("afloat", first.afloat_count(), second.afloat_count()))
730 self.emit(*parts)

741 def match(self, arguments):
742 seeds = int(arguments[0]), int(arguments[1])
743 strategies = arguments[2], arguments[3]
744 game = play_match(self.game, seeds, strategies)
745 self.emit("MATCH", 1, strategies[0], 2, strategies[1])
746 for number, ship, shots in game.sink_log:
747 self.emit("SINK", number, ship, shots)
748 first, second = game.players
749 self.emit(
750 "RESULT", "winner", game.winner, "shots", first.shots, second.shots,
751 "hits", first.hits, second.hits,
752 )

754 def series(self, arguments):
755 count, state = int(arguments[0]), int(arguments[1])
756 strategies = arguments[2], arguments[3]
757 wins = [0, 0]
758 shots = [0, 0]
759 fastest = [None, None]
760 for _ in range(count):
761 seeds = match_seeds(state)
762 state = seeds[1]
763 game = play_match(self.game, seeds, strategies)
764 for side, player in enumerate(game.players):
765 shots[side] += player.shots
766 winner = game.winner - 1
767 wins[winner] += 1
768 winner_shots = game.players[winner].shots
769 if fastest[winner] is None or winner_shots < fastest[winner]:
770 fastest[winner] = winner_shots
771 fastest_text = ["-" if value is None else value for value in fastest]
772 self.emit(
773 "SERIES", count, "wins", *wins, "shots", *shots, "fastest", *fastest_text
774 )

776 def tourney(self, arguments):
777 count, state = int(arguments[0]), int(arguments[1])
778 wins = dict.fromkeys(STRATEGIES, 0)
779 shots = dict.fromkeys(STRATEGIES, 0)
780 for pair in TOURNEY_PAIRS:
781 for _ in range(count):
782 seeds = match_seeds(state)
783 state = seeds[1]
784 game = play_match(self.game, seeds, pair)
785 wins[pair[game.winner - 1]] += 1
786 for strategy, player in zip(pair, game.players):
787 shots[strategy] += player.shots
788 ranking = sorted(STRATEGIES, key=lambda name: (-wins[name], shots[name], name))
789 self.emit("TOURNEY", count)
790 for rank, strategy in enumerate(ranking, 1):
791 self.emit(rank, strategy, "wins", wins[strategy], "shots", shots[strategy])

793 def execute(self, line):
794 command, *arguments = line.split()
795 handlers = {
796 "PLACE": self.place,
797 "REMOVE": self.remove,
798 "AUTOPLACE": self.autoplace,
799 "LAYOUT": self.layout,
800 "REEF": self.reef,
801 "FIRE": self.fire,
802 "AI": self.ai,
803 "TORPEDO": self.torpedo,
804 "SONAR": self.sonar,
805 "RESIGN": self.resign,
806 "NEW": self.new_game,
807 "CONTROL": self.control,
808 "SEED": self.seed,

811 "SHOW": self.show,
812 "RADAR": self.radar,
813 "EXPORT": self.export,

815 "STATUS": self.status,
816 "STATS": self.stats,
817 "MATCH": self.match,
818 "SERIES": self.series,
819 "TOURNEY": self.tourney,
820 }

824 else:
825 handlers[command](arguments)
826 except CommandError as error:
