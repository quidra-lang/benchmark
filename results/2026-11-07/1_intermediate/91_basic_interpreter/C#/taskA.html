84 public static List<Token> Tokenize(string text) {
85 List<Token> tokens = [];
86 for (int p = 0; p < text.Length;) {
87 char ch = text[p];
88 if (ch == ' ') {
89 p++;
90 continue;
91 }
92 if (ch == '"') {
93 int end = text.IndexOf('"', p + 1);
94 if (end < 0)
95 Syntax();
96 tokens.Add(new("S", text[p..(end + 1)], text[(p + 1)..end]));
97 p = end + 1;
98 } else if (ch is >= '0' and <= '9') {
99 int start = p;
100 while (p < text.Length && text[p] is >= '0' and <= '9')
101 p++;
102 string word = text[start..p];
103 if (!int.TryParse(word, out int value) || value > 32767)
104 Syntax();
105 tokens.Add(new("I", word, value));
106 } else if (ch is >= 'A' and <= 'Z') {
107 if (text.AsSpan(p).StartsWith("REM", StringComparison.Ordinal)) {
108 tokens.Add(new("W", "REM", "REM"));
109 break;
110 }
111 int start = p;
112 while (p < text.Length && (text[p] is >= 'A' and <= 'Z' or >= '0' and <= '9'))
113 p++;
114 if (p < text.Length && text[p] == '$')
115 p++;
116 string word = text[start..p];
117 tokens.Add(new("W", word, word));
118 } else {
119 if (!"+-*/^=<>() ,;:".Contains(ch))
120 Syntax();
121 string word = p + 1 < text.Length ? text.Substring(p, 2) : ch.ToString();
122 if (word is not("<>" or "<=" or ">="))
123 word = ch.ToString();
124 tokens.Add(new("P", word, word));
125 p += word.Length;
126 }
127 }
128 tokens.Add(new("E", "", ""));
129 return tokens;
130 }

133sealed class Parser {
134 readonly List<Token> tokens;
135 int position;
136 public Parser(List<Token> tokens) {
137 this.tokens = tokens;
138 }
139 Token Current => tokens[position];
140 bool At(string word) => (Current.Kind is "P" or "W") && Current.Text == word;
141 bool Take(string word) {
142 if (!At(word))
143 return false;
144 position++;
145 return true;
146 }
147 void Expect(string word) {
148 if (!Take(word))
149 Basic.Syntax();
150 }
151 int Number() {
152 if (Current.Kind != "I")
153 Basic.Syntax();
154 return (int)tokens[position++].Value;
155 }
156 Expr Integer(int level = 0) {
157 Expr result = Expression(level);
158 if (result.Type != "I")
159 Basic.Syntax();
160 return result;
161 }
162 Expr Target() {
163 Token token = Current;
164 if (token.Kind != "W" || !Basic.Variable(token.Text))
165 Basic.Syntax();
166 position++;
167 Expr result = new("variable", Basic.NameType(token.Text), token.Text);
168 if (Take("(")) {
169 result.Kind = "array";
170 result.Args.Add(Integer());
171 if (Take(","))
172 result.Args.Add(Integer());
173 Expect(")");
174 }
175 return result;
176 }
177 Expr Primary() {
178 Token token = Current;
179 if (token.Kind is "I" or "S") {
180 position++;
181 return new("constant", token.Kind, token.Value);
182 }
183 if (Take("(")) {
184 Expr value = Expression();
185 Expect(")");
186 return value;
187 }
188 if (token.Kind == "W" && Basic.Variable(token.Text))
189 return Target();
190 if (Take("ERR"))
191 return new("builtin", "I", "ERR");
192 if (Take("ERL"))
193 return new("builtin", "I", "ERL");
194 if (token.Kind == "W" && Basic.Signatures.ContainsKey(token.Text)) {
195 position++;
196 Expect("(");
197 List<Expr> args = [Expression()];
198 while (Take(","))
199 args.Add(Expression());
200 Expect(")");
201 if (!Basic.Signatures[token.Text].Contains(string.Concat(args.Select(e => e.Type))))
202 Basic.Syntax();
203 return new("call", Basic.NameType(token.Text), token.Text, args);
204 }
205 Basic.Syntax();
206 throw new InvalidOperationException();
207 }
208 Expr Expression(int level = 0) {
209 if (level == 2 && Take("NOT"))
210 return new("unary", "I", "NOT", [Integer(2)]);
211 if (level == 6 && Take("-"))
212 return new("unary", "I", "-", [Integer(6)]);
213 if (level == 8)
214 return Primary();
215 Expr left = Expression(level + 1);
216 while (Basic.Ops[level].Contains(Current.Text)) {
217 string op = Current.Text;
218 position++;
219 Expr right = Expression(level == 7 ? 6 : level + 1);
220 if (left.Type != right.Type || (left.Type == "S" && level != 3 && op != "+"))
221 Basic.Syntax();
222 left = new("binary", level == 3 ? "I" : left.Type, op, [left, right]);
223 if (level == 7)
224 break;
225 }
226 return left;
227 }
228 Clause ParseClause() => Current.Kind == "I" ? new(Number(), []) : new(null, Listing(true));
229 List<Statement> Listing(bool inside = false) {
230 List<Statement> body = [ParseStatement(inside)];
231 while (Take(":"))
232 body.Add(ParseStatement(inside));
233 return body;
234 }
235 Statement ParseStatement(bool inside) {
236 string word = Current.Text;
237 if (inside && new[] { "IF", "FOR", "NEXT", "WHILE", "WEND", "DATA" }.Contains(word))
238 Basic.Syntax();
239 Statement s = new(word);
240 foreach (string kind in new[] { "REM", "RETURN", "WEND", "END", "STOP" })
241 if (Take(kind))
242 return s;
243 if (Take("PRINT")) {
244 bool lastValue = false;
245 while (Current.Kind != "E" && !At(":") && !At("ELSE")) {
246 if (Take(";")) {
247 s.Items.Add(new(";", null));
248 lastValue = false;
249 } else if (Take(",")) {
250 s.Items.Add(new(",", null));
251 lastValue = false;
252 } else {
253 string kind = At("TAB") ? "TAB" : At("SPC") ? "SPC" : "value";
254 if (kind == "value" && lastValue)
255 Basic.Syntax();
256 Expr expr;
257 if (kind == "value")
258 expr = Expression();
259 else {
260 position++;
261 Expect("(");
262 expr = Integer();
263 Expect(")");
264 }
265 s.Items.Add(new(kind, expr));
266 lastValue = kind == "value";
267 }
268 }
269 return s;
270 }
271 if (Take("INPUT") || Take("READ") || Take("DIM")) {
272 do {
273 Expr target = Target();
274 if (word == "DIM" && target.Kind != "array")
275 Basic.Syntax();
276 s.Targets.Add(target);
277 } while (Take(","));
278 return s;
279 }
280 if (Take("DATA")) {
281 do {
282 bool negative = Take("-");
283 Token token = Current;
284 if (token.Kind != "I" && (token.Kind != "S" || negative))
285 Basic.Syntax();
286 position++;
287 s.Values.Add(negative ? -(int)token.Value : token.Value);
288 } while (Take(","));
289 return s;
290 }
291 if (Take("IF")) {
292 s.Expr = Integer();
293 Expect("THEN");
294 s.Clauses.Add(ParseClause());
295 if (Take("ELSE"))
296 s.Clauses.Add(ParseClause());
297 if (Current.Kind != "E")
298 Basic.Syntax();
299 return s;
300 }
301 if (Take("GOTO") || Take("GOSUB")) {
302 s.Numbers.Add(Number());
303 return s;
304 }
305 if (Take("ON")) {
306 if (Take("ERROR")) {
307 s.Kind = "handler";
308 Expect("GOTO");
309 s.Numbers.Add(Number());
310 } else {
311 s.Expr = Integer();
312 if (Take("GOTO"))
313 s.Mode = "GOTO";
314 else {
315 Expect("GOSUB");
316 s.Mode = "GOSUB";
317 }
318 s.Numbers.Add(Number());
319 while (Take(","))
320 s.Numbers.Add(Number());
321 }
322 return s;
323 }
324 if (Take("RESUME")) {
325 if (Take("NEXT"))
326 s.Mode = "NEXT";
327 else if (Current.Kind == "I") {
328 s.Mode = "line";
329 s.Numbers.Add(Number());
330 }
331 return s;
332 }
333 if (Take("FOR")) {
334 Token token = Current;
335 if (token.Kind != "W" || !Basic.Variable(token.Text) ||
336 Basic.NameType(token.Text) != "I")
337 Basic.Syntax();
338 s.Name = token.Text;
339 position++;
340 Expect("=");
341 s.Expressions.Add(Integer());
342 Expect("TO");
343 s.Expressions.Add(Integer());
344 s.Expressions.Add(Take("STEP") ? Integer() : new Expr("constant", "I", 1));
345 return s;
346 }
347 if (Take("NEXT")) {
348 Token token = Current;
349 if (token.Kind == "W" && Basic.Variable(token.Text) &&
350 Basic.NameType(token.Text) == "I") {
351 s.Name = token.Text;
352 position++;
353 }
354 return s;
355 }
356 if (Take("WHILE") || Take("RANDOMIZE")) {
357 s.Expr = Integer();
358 return s;
359 }
360 if (Take("RESTORE")) {
361 if (Current.Kind == "I")
362 s.Numbers.Add(Number());
363 return s;
364 }
365 Take("LET");
366 s.Kind = "assign";
367 s.Target = Target();
368 Expect("=");
369 s.Expr = Expression();
370 if (s.Expr.Type != s.Target.Type)
371 Basic.Syntax();
372 return s;
373 }
374 public List<Statement> Parse() {
375 List<Statement> result = Listing();
376 if (Current.Kind != "E")
377 Basic.Syntax();
378 return result;
379 }
380}

406 void EndLine() {

410 void Message(string line) {
411 if (buffer.Length > 0)
412 EndLine();
413 output.Add(line);

415 static int Checked(long value) {
416 if (value is < -32768 or > 32767)
417 Basic.Fault("OVERFLOW");
418 return (int)value;
419 }
420 object ValueOf(string name) => variables.TryGetValue(name, out object? value) ? value
421 : Basic.NameType(name) == "S" ? ""
422 : 0;
423 void Declare(string name, int[] bounds) {
424 int count = 1;
425 foreach (int b in bounds)
426 count *= b + 1;
427 object initial = Basic.NameType(name) == "S" ? "" : 0;
428 arrays[name] = new(bounds, Enumerable.Repeat(initial, count).ToArray());
429 }
430 Address GetAddress(Expr target) {
431 string name = (string)target.Value;
432 if (target.Kind == "variable")
433 return new(name, null);
434 int[] indices = target.Args.Select(e => (int)Evaluate(e)).ToArray();
435 if (!arrays.ContainsKey(name))
436 Declare(name, Enumerable.Repeat(10, indices.Length).ToArray());
437 ArrayDef array = arrays[name];
438 if (indices.Length != array.Bounds.Length)
439 Basic.Fault("SUBSCRIPT OUT OF RANGE");
440 int index = 0;
441 for (int i = 0; i < indices.Length; i++) {
442 int value = indices[i];
443 if (value < 0 || value > array.Bounds[i])
444 Basic.Fault("SUBSCRIPT OUT OF RANGE");
445 index = index * (array.Bounds[i] + 1) + value;
446 }
447 return new(name, index);
448 }
449 object Load(Address addr) => addr.Index == null ? ValueOf(addr.Name)
450 : arrays[addr.Name].Values[addr.Index.Value];
451 void Store(Address addr, object value) {
452 if (addr.Index == null)
453 variables[addr.Name] = value;
454 else
455 arrays[addr.Name].Values[addr.Index.Value] = value;
456 }
457 object Evaluate(Expr e) {
458 object op = e.Value;
459 if (e.Kind == "constant")
460 return op;
461 if (e.Kind is "variable" or "array")
462 return Load(GetAddress(e));
463 if (e.Kind == "builtin")
464 return (string)op == "ERR" ? err : erl;
465 if (e.Kind == "unary") {
466 int a = (int)Evaluate(e.Args[0]);
467 return (string)op == "NOT" ? ~a : Checked(-(long)a);
468 }
469 object[] values = e.Args.Select(Evaluate).ToArray();
470 if (e.Kind == "call")
471 return Call((string)op, values);
472 object a0 = values[0], b0 = values[1];
473 if (Basic.Ops[3].Contains((string)op)) {
474 int cmp = a0 is string x ? string.CompareOrdinal(x, (string)b0)
475 : ((int)a0).CompareTo((int)b0);
476 bool success = (string)op switch { "=" => cmp == 0, "<>" => cmp != 0, "<" => cmp < 0,
477 ">" => cmp > 0, "<=" => cmp <= 0, ">=" => cmp >= 0,
478 _ => false };
479 return success ? -1 : 0;
480 }
481 if (e.Type == "S") {
482 string value = (string)a0 + (string)b0;
483 if (value.Length > 255)
484 Basic.Fault("STRING TOO LONG");
485 return value;
486 }
487 int a = (int)a0, b = (int)b0;
488 switch ((string)op) {
489 case "+":
490 return Checked((long)a + b);
491 case "-":
492 return Checked((long)a - b);
493 case "*":
494 return Checked((long)a * b);
495 case "/":
496 if (b == 0)
497 Basic.Fault("DIVISION BY ZERO");
498 return Checked(a / b);
499 case "MOD":
500 if (b == 0)
501 Basic.Fault("DIVISION BY ZERO");
502 return a - b * (a / b);
503 case "^":
504 if (b < 0)
505 Basic.Fault("ILLEGAL FUNCTION CALL");
506 if (Math.Abs(a) > 1 && b > 15)
507 Basic.Fault("OVERFLOW");
508 long result = 1;
509 for (int i = 0; i < b; i++)
510 result = Checked(result * a);
511 return (int)result;
512 case "AND":
513 return a & b;
514 case "OR":
515 return a | b;
516 }
517 Basic.Fault("ILLEGAL FUNCTION CALL");
518 return 0;
519 }
520 object Call(string name, object[] args) {
521 object first = args[0];
522 object? second = args.Length > 1 ? args[1] : null;
523 object? third = args.Length > 2 ? args[2] : null;
524 switch (name) {
525 case "ABS":
526 return Checked(Math.Abs((long)(int)first));
527 case "SGN":
528 return Math.Sign((int)first);
529 case "INT":
530 return first;
531 case "SQR":
532 if ((int)first < 0)
533 Basic.Fault("ILLEGAL FUNCTION CALL");
534 return (int)Math.Sqrt((int)first);
535 case "RND":
536 int n = (int)first;
537 if (n < 1)
538 Basic.Fault("ILLEGAL FUNCTION CALL");
539 random = (int)(48271L * random % 2147483647);
540 return random % n + 1;
541 case "LEN":
542 return ((string)first).Length;
543 case "ASC":
544 string source = (string)first;
545 if (source.Length == 0)
546 Basic.Fault("ILLEGAL FUNCTION CALL");
547 return (int)source[0];
548 case "VAL":
549 Match match = Regex.Match((string)first, @"\A *([+-]?[0-9]+)");
550 if (!match.Success)
551 return 0;
552 if (!long.TryParse(match.Groups[1].Value, out long number))
553 Basic.Fault("OVERFLOW");
554 return Checked(number);
555 case "CHR$":
556 int code = (int)first;
557 if (code < 32 || code > 126)
558 Basic.Fault("ILLEGAL FUNCTION CALL");
559 return ((char)code).ToString();
560 case "STR$":
561 return ((int)first).ToString();
562 case "LEFT$":
563 int count = (int)second!;
564 if (count < 0)
565 Basic.Fault("ILLEGAL FUNCTION CALL");
566 return ((string)first)[..Math.Min(count, ((string)first).Length)];
567 case "RIGHT$":
568 int k = (int)second!;
569 if (k < 0)
570 Basic.Fault("ILLEGAL FUNCTION CALL");
571 string text = (string)first;
572 return text[Math.Max(0, text.Length - k)..];
573 case "MID$":
574 int start = (int)second!;
575 int? length = third == null ? null : (int)third;
576 if (start < 1 || (length != null && length < 0))
577 Basic.Fault("ILLEGAL FUNCTION CALL");
578 string word = (string)first;
579 int begin = Math.Min(start - 1, word.Length);
580 return word.Substring(begin, length == null
581 ? word.Length - begin
582 : Math.Min(length.Value, word.Length - begin));
583 case "INSTR":
584 return ((string)first).IndexOf((string)second!, StringComparison.Ordinal) + 1;
585 }
586 Basic.Fault("ILLEGAL FUNCTION CALL");
587 return 0;
588 }
589 Statement Line(int number) {
590 if(!entries.TryGetValue(number,out Statement? s))
591 Basic.Fault("UNDEFINED LINE NUMBER");
592 return s!;
593 }
594 void Push(Frame f) {
595 if (stack.Count == 100)
596 Basic.Fault("OUT OF MEMORY");
597 stack.Add(f);

599 int Search(string kind, string name = "") {
600 for (int i = stack.Count - 1; i >= 0; i--) {
601 Frame f = stack[i];
602 if (f.Kind == "GOSUB")
603 break;
604 if (f.Kind == kind && (name == "" || f.Name == name))
605 return i;
606 }
607 return -1;
608 }
609 Statement? Matching(Statement s, string opening, string closing) {
610 int start = sequence.IndexOf(s), depth = 0;
611 for (int i = start + 1; i < sequence.Count; i++) {
612 Statement t = sequence[i];
613 if (t.Kind == opening)
614 depth++;
615 else if (t.Kind == closing) {
616 if (depth == 0)
617 return t.Next;
618 depth--;
619 }
620 }
621 Basic.Fault(opening + " WITHOUT " + closing);
622 return null;
623 }
624 Statement Subroutine(int number, Statement? after) {
625 Statement target = Line(number);
626 Push(new("GOSUB", after));
627 return target;
628 }
629 Statement? Execute(Statement s) {
630 Statement? following = s.Next;
631 switch (s.Kind) {
632 case "assign":
633 Address addr = GetAddress(s.Target!);
634 Store(addr, Evaluate(s.Expr!));
635 break;
636 case "PRINT":
637 foreach (PrintItem item in s.Items) {
638 if (item.Kind == ",") {
639 buffer += new string(' ', 10 - buffer.Length % 10);
640 } else if (item.Kind != ";") {
641 object value = Evaluate(item.Expr!);
642 if (item.Kind == "value")
643 buffer += value.ToString();
644 else {
645 int space = (int)value;
646 if (space < 0)
647 Basic.Fault("ILLEGAL FUNCTION CALL");
648 buffer += new string(
649 ' ', item.Kind == "SPC" ? space : Math.Max(0, space - buffer.Length));
650 }
651 }
652 }
653 if (s.Items.Count == 0 || s.Items[^1].Kind is not(";" or ","))
654 EndLine();
655 break;
656 case "INPUT":
657 case "READ":
658 foreach (Expr target in s.Targets) {
659 Address place = GetAddress(target);
660 object value;
661 if (s.Kind == "READ") {
662 if (dataPosition == data.Count)
663 Basic.Fault("OUT OF DATA");
664 value = data[dataPosition++].Value;
665 if (Basic.ValueType(value) != target.Type)
666 Basic.Fault("TYPE MISMATCH");
667 } else {
668 while (true) {
669 if (inputPosition == inputs.Count)
670 Basic.Fault("INPUT PAST END");
671 string word = inputs[inputPosition++];
672 if (target.Type == "S") {
673 value = word;
674 break;
675 }
676 if (Regex.IsMatch(word, @"\A-?[0-9]+\z") && int.TryParse(word, out int v) &&
677 v is >= -32768 and <= 32767) {
678 value = v;
679 break;
680 }
681 Message("?REDO FROM START");
682 }
683 }
684 Store(place, value);
685 }
686 break;
687 case "RESTORE":
688 if (s.Numbers.Count == 0)
689 dataPosition = 0;
690 else {
691 int number = s.Numbers[0];
692 Line(number);
693 dataPosition = data.FindIndex(d => d.Line >= number);
694 if (dataPosition < 0)
695 dataPosition = data.Count;
696 }
697 break;
698 case "IF":
699 int ci = (int)Evaluate(s.Expr!) == 0 ? 1 : 0;
700 if (ci < s.Clauses.Count) {
701 Clause chosen = s.Clauses[ci];
702 following = chosen.Line == null ? chosen.Body[0] : Line(chosen.Line.Value);
703 }
704 break;
705 case "GOTO":
706 following = Line(s.Numbers[0]);
707 break;
708 case "GOSUB":
709 following = Subroutine(s.Numbers[0], following);
710 break;
711 case "ON":
712 int index = (int)Evaluate(s.Expr!);
713 if (index < 0)
714 Basic.Fault("ILLEGAL FUNCTION CALL");
715 if (index >= 1 && index <= s.Numbers.Count) {
716 following = s.Mode == "GOTO" ? Line(s.Numbers[index - 1])
717 : Subroutine(s.Numbers[index - 1], following);
718 }
719 break;
720 case "RETURN":
721 int frame = stack.FindLastIndex(f => f.Kind == "GOSUB");
722 if (frame < 0)
723 Basic.Fault("RETURN WITHOUT GOSUB");
724 following = stack[frame].Position;
725 stack.RemoveRange(frame, stack.Count - frame);
726 break;
727 case "FOR":
728 int[] numbers = s.Expressions.Select(e => (int)Evaluate(e)).ToArray();
729 int from = numbers[0], end = numbers[1], step = numbers[2];
730 if (step == 0)
731 Basic.Fault("ILLEGAL FUNCTION CALL");
732 int ix = Search("FOR", s.Name);
733 if (ix >= 0)
734 stack.RemoveRange(ix, stack.Count - ix);
735 variables[s.Name] = from;
736 if ((step > 0 && from > end) || (step < 0 && from < end))
737 following = Matching(s, "FOR", "NEXT");
738 else
739 Push(new("FOR", following, s.Name, end, step));
740 break;
741 case "NEXT":
742 int fidx = Search("FOR", s.Name);
743 if (fidx < 0)
744 Basic.Fault("NEXT WITHOUT FOR");
745 stack.RemoveRange(fidx + 1, stack.Count - fidx - 1);
746 Frame current = stack[fidx];
747 int count = Checked((long)(int)ValueOf(current.Name) + current.Step);
748 variables[current.Name] = count;
749 if ((current.Step > 0 && count > current.Limit) ||
750 (current.Step < 0 && count < current.Limit))
751 stack.RemoveAt(fidx);
752 else
753 following = current.Position;
754 break;
755 case "WHILE":
756 if ((int)Evaluate(s.Expr!) == 0)
757 following = Matching(s, "WHILE", "WEND");
758 else
759 Push(new("WHILE", s));
760 break;
761 case "WEND":
762 int widx = Search("WHILE");
763 if (widx < 0)
764 Basic.Fault("WEND WITHOUT WHILE");
765 following = stack[widx].Position;
766 stack.RemoveRange(widx, stack.Count - widx);
767 break;
768 case "DIM":
769 foreach (Expr target in s.Targets) {
770 string name = (string)target.Value;
771 int[] bounds = target.Args.Select(e => (int)Evaluate(e)).ToArray();
772 if (arrays.ContainsKey(name))
773 Basic.Fault("DUPLICATE DEFINITION");
774 if (bounds.Any(v => v < 0))
775 Basic.Fault("ILLEGAL FUNCTION CALL");
776 long size = 1;
777 foreach (int b in bounds)
778 size *= b + 1;
779 if (size > 10000)
780 Basic.Fault("OUT OF MEMORY");
781 Declare(name, bounds);
782 }
783 break;
784 case "RANDOMIZE":
785 int seed = (int)Evaluate(s.Expr!);
786 if (seed < 1)
787 Basic.Fault("ILLEGAL FUNCTION CALL");
788 random = seed;
789 break;
790 case "handler":
791 if (s.Numbers[0] != 0)
792 Line(s.Numbers[0]);
793 handler = s.Numbers[0];
794 break;
795 case "RESUME":
796 if (!handling)
797 Basic.Fault("RESUME WITHOUT ERROR");
798 following = s.Mode == "line" ? Line(s.Numbers[0])
799 : s.Mode == "NEXT" ? remembered!.Next
800 : remembered;
801 handling = false;
802 break;
803 case "STOP":
804 Message("BREAK IN LINE " + s.Line);
805 return null;
806 case "END":
807 return null;
808 }
809 return following;
810 }
811 void Connect(List<Statement> body, Statement? after, int number) {
812 Statement? following = after;
813 for (int i = body.Count - 1; i >= 0; i--) {
814 Statement s = body[i];
815 s.Line = number;
816 s.Next = following;
817 foreach (Clause c in s.Clauses)
818 Connect(c.Body, following, number);
819 following = s;
820 }
821 }
822 List<string> Finish() {
823 if (buffer.Length > 0)
824 EndLine();
825 output.Add("READY");
826 return output;
827 }
828 public List<string> Run(Dictionary<int, string> source) {
829 List<ProgramLine> parsed = [];
830 foreach (int number in source.Keys.Order()) {
831 try {
832 parsed.Add(new(number, new Parser(Basic.Tokenize(source[number])).Parse()));
833 } catch (SyntaxFault) {
834 Message("ERROR IN LINE " + number + ": SYNTAX ERROR");
835 return Finish();
836 }
837 }
838 Statement? after = null;
839 for (int i = parsed.Count - 1; i >= 0; i--) {
840 ProgramLine e = parsed[i];
841 Connect(e.Body, after, e.Number);
842 entries[e.Number] = e.Body[0];
843 after = e.Body[0];
844 }
845 foreach (ProgramLine e in parsed)
846 foreach (Statement s in e.Body) {
847 sequence.Add(s);
848 if (s.Kind == "DATA")
849 foreach (object value in s.Values)
850 data.Add(new(s.Line, value));
851 }
852 Statement? position = after;
853 while (position != null) {
854 Statement current = position;
855 try {
856 position = Execute(current);
857 } catch (BasicFault error) {
858 if (handler != 0 && !handling) {
859 err = Basic.Errors.GetValueOrDefault(error.Message, 0);
860 erl = current.Line;
861 remembered = current;
862 handling = true;
863 try {
864 position = Line(handler);
865 } catch (BasicFault secondary) {
866 Message("ERROR IN LINE " + current.Line + ": " + secondary.Message);
867 position = null;
868 }
869 } else {
870 Message("ERROR IN LINE " + current.Line + ": " + error.Message);
871 position = null;
872 }
873 }
874 }
875 return Finish();
876 }
877}
