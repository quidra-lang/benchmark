function tokenize(text) {
    const tokens = [];
    for (let p = 0; p < text.length;) {
        const ch = text[p];
        if (ch === " ") {
            p++;
            continue;
        }
        if (ch === '"') {
            const end = text.indexOf('"', p + 1);
            if (end < 0) syntax();
            tokens.push({ kind: "S", text: text.slice(p, end + 1), value: text.slice(p + 1, end) });
            p = end + 1;
        } else if (/[0-9]/.test(ch)) {
            const start = p;
            while (p < text.length && /[0-9]/.test(text[p])) p++;
            const word = text.slice(start, p),
                value = Number(word);
            if (value > 32767) syntax();
            tokens.push({ kind: "I", text: word, value });
        } else if (/[A-Z]/.test(ch)) {
            const start = p;
            while (p < text.length && /[A-Z0-9]/.test(text[p])) p++;
            if (text[p] === "$") p++;
            const word = text.slice(start, p);
            tokens.push({ kind: "W", text: word, value: word });
        } else {
            let word = text.slice(p, p + 2);
            if (!["<>", "<=", ">="].includes(word)) word = ch;
            if (!"+-*/^=<>() ,;:".includes(ch)) syntax();
            tokens.push({ kind: "P", text: word, value: word });
            p += word.length;
        }
    }
    tokens.push({ kind: "E", text: "", value: "" });
    return tokens;
}

class Parser {
    position = 0;
    constructor(tokens) {
        this.tokens = tokens;
    }
    current() {
        return this.tokens[this.position];
    }
    at(word) {
        const t = this.current();
        return (t.kind === "P" || t.kind === "W") && t.text === word;
    }
    take(word) {
        if (!this.at(word)) return false;
        this.position++;
        return true;
    }
    expect(word) {
        if (!this.take(word)) syntax();
    }
    integer() {
        const expr = this.expression();
        if (expr.type !== "I") syntax();
        return expr;
    }
    expression(level = 0) {
        if (level === 2 && this.take("NOT"))
            return new Expr("unary", "I", "NOT", [this.integerAt(2)]);
        if (level === 6 && this.take("-")) return new Expr("unary", "I", "-", [this.integerAt(6)]);
        if (level === 8) return this.primary();
        let left = this.expression(level + 1);
        const ops = [
            ["OR"],
            ["AND"],
            [],
            ["=", "<>", "<", ">", "<=", ">="],
            ["+", "-"],
            ["*", "/", "MOD"],
            [],
            ["^"],
        ][level];
        while (ops.some((op) => this.at(op))) {
            const op = this.current().text;
            this.position++;
            const right = this.expression(level === 7 ? 6 : level + 1);
            if (left.type !== right.type || (left.type === "S" && level !== 3 && op !== "+"))
                syntax();
            left = new Expr("binary", level === 3 ? "I" : left.type, op, [left, right]);
            if (level === 7) break;
        }
        return left;
    }
    integerAt(level) {
        const e = this.expression(level);
        if (e.type !== "I") syntax();
        return e;
    }
    primary() {
        const t = this.current();
        if (t.kind === "I" || t.kind === "S") {
            this.position++;
            return new Expr("constant", t.kind, t.value);
        }
        if (this.take("(")) {
            const e = this.expression();
            this.expect(")");
            return e;
        }
        if (t.kind === "W" && signatures.has(t.text)) {
            this.position++;
            this.expect("(");
            const args = [this.expression()];
            while (this.take(",")) args.push(this.expression());
            this.expect(")");
            if (!signatures.get(t.text).includes(args.map((e) => e.type).join(""))) syntax();
            return new Expr("call", nameType(t.text), t.text, args);
        }
        return syntax();
    }
    list(clause = false) {
        const body = [this.statement(clause)];
        while (this.take(":")) body.push(this.statement(clause));
        return body;
    }
    statement(inClause) {
        const word = this.current().text;
        if (inClause && ["IF", "FOR", "NEXT", "WHILE", "WEND", "DATA"].includes(word)) syntax();
        const s = new Statement(word);
        if (this.take("PRINT")) {
            let expressionLast = false;
            while (this.current().kind !== "E" && !this.at(":") && !this.at("ELSE")) {
                if (this.take(";")) {
                    s.items.push({ kind: ";", expr: null });
                    expressionLast = false;
                } else if (this.take(",")) {
                    s.items.push({ kind: ",", expr: null });
                    expressionLast = false;
                } else {
                    const kind = this.at("TAB") ? "TAB" : this.at("SPC") ? "SPC" : "value";
                    if (kind === "value" && expressionLast) syntax();
                    let expr;
                    } else expr = this.expression();
                    s.items.push({ kind, expr });
                    expressionLast = kind === "value";
                }
            }
            return s;
        }
    parse() {
        const body = this.list();
        if (this.current().kind !== "E") syntax();
        return body;
    }
    random = 1;
    handler = 0;
    handling = false;
    buffer = "";
    output = [];
    endLine() {
        this.output.push(this.buffer.replace(/ +$/, ""));
        this.buffer = "";
    }
    message(text) {
        if (this.buffer !== "") this.endLine();
        this.output.push(text);
    }
    checked(n) {
        if (!Number.isFinite(n) || n < -32768 || n > 32767) fault("OVERFLOW");
        return n;
    }
    evaluate(e) {
        const op = e.value;
        if (e.kind === "constant") return op;
        if (e.kind === "unary") {
            const a = this.evaluate(e.args[0]);
            return op === "NOT" ? ~a : this.checked(-a);
        }
        const args = e.args.map((a) => this.evaluate(a));
        if (e.kind === "call") return this.call(op, args);

            y = b;
        switch (op) {
            case "/":
                if (y === 0) fault("DIVISION BY ZERO");
                return this.checked(Math.trunc(x / y));
            case "MOD":
                if (y === 0) fault("DIVISION BY ZERO");
                return x % y;
            case "AND":
                return x & y;
            case "OR":
                return x | y;
        }
        throw new Error("invalid expression");
    }
    call(name, args) {
        const n = args[0],
        switch (name) {

                        const value = this.evaluate(item.expr);

    }

            s.line = line;
            s.next = next;
            next = s;
        }
    }

        const parsed = [];
        for (const line of [...source.keys()].sort((a, b) => a - b)) {
            try {
                parsed.push([line, new Parser(tokenize(source.get(line))).parse()]);
            } catch (e) {

                    position = null;
                }
            }
        }
        return this.finish();
    }
