72function tokenize(text) {
73 const tokens = [];
74 for (let p = 0; p < text.length;) {
75 const ch = text[p];
76 if (ch === " ") {
77 p++;
78 continue;
79 }
80 if (ch === '"') {
81 const end = text.indexOf('"', p + 1);
82 if (end < 0) syntax();
83 tokens.push({ kind: "S", text: text.slice(p, end + 1), value: text.slice(p + 1, end) });
84 p = end + 1;
85 } else if (/[0-9]/.test(ch)) {
86 const start = p;
87 while (p < text.length && /[0-9]/.test(text[p])) p++;
88 const word = text.slice(start, p),
89 value = Number(word);
90 if (value > 32767) syntax();
91 tokens.push({ kind: "I", text: word, value });
92 } else if (/[A-Z]/.test(ch)) {
93 if (text.startsWith("REM", p)) {
94 tokens.push({ kind: "W", text: "REM", value: "REM" });
95 break;
96 }
97 const start = p;
98 while (p < text.length && /[A-Z0-9]/.test(text[p])) p++;
99 if (text[p] === "$") p++;
100 const word = text.slice(start, p);
101 tokens.push({ kind: "W", text: word, value: word });
102 } else {
103 let word = text.slice(p, p + 2);
104 if (!["<>", "<=", ">="].includes(word)) word = ch;
105 if (!"+-*/^=<>() ,;:".includes(ch)) syntax();
106 tokens.push({ kind: "P", text: word, value: word });
107 p += word.length;
108 }
109 }
110 tokens.push({ kind: "E", text: "", value: "" });
111 return tokens;
112}

114class Parser {
115 position = 0;
116 constructor(tokens) {
117 this.tokens = tokens;
118 }
119 current() {
120 return this.tokens[this.position];
121 }
122 at(word) {
123 const t = this.current();
124 return (t.kind === "P" || t.kind === "W") && t.text === word;
125 }
126 take(word) {
127 if (!this.at(word)) return false;
128 this.position++;
129 return true;
130 }
131 expect(word) {
132 if (!this.take(word)) syntax();
133 }
134 integer() {
135 const expr = this.expression();
136 if (expr.type !== "I") syntax();
137 return expr;
138 }
139 number() {
140 const t = this.current();
141 if (t.kind !== "I") syntax();
142 this.position++;
143 return t.value;
144 }
145 target() {
146 const token = this.current();
147 if (token.kind !== "W" || !variable(token.text)) syntax();
148 this.position++;
149 const result = new Expr("variable", nameType(token.text), token.text);
150 if (this.take("(")) {
151 result.kind = "array";
152 result.args.push(this.integer());
153 if (this.take(",")) result.args.push(this.integer());
154 this.expect(")");
155 }
156 return result;
157 }
158 expression(level = 0) {
159 if (level === 2 && this.take("NOT"))
160 return new Expr("unary", "I", "NOT", [this.integerAt(2)]);
161 if (level === 6 && this.take("-")) return new Expr("unary", "I", "-", [this.integerAt(6)]);
162 if (level === 8) return this.primary();
163 let left = this.expression(level + 1);
164 const ops = [
165 ["OR"],
166 ["AND"],
167 [],
168 ["=", "<>", "<", ">", "<=", ">="],
169 ["+", "-"],
170 ["*", "/", "MOD"],
171 [],
172 ["^"],
173 ][level];
174 while (ops.some((op) => this.at(op))) {
175 const op = this.current().text;
176 this.position++;
177 const right = this.expression(level === 7 ? 6 : level + 1);
178 if (left.type !== right.type || (left.type === "S" && level !== 3 && op !== "+"))
179 syntax();
180 left = new Expr("binary", level === 3 ? "I" : left.type, op, [left, right]);
181 if (level === 7) break;
182 }
183 return left;
184 }
185 integerAt(level) {
186 const e = this.expression(level);
187 if (e.type !== "I") syntax();
188 return e;
189 }
190 primary() {
191 const t = this.current();
192 if (t.kind === "I" || t.kind === "S") {
193 this.position++;
194 return new Expr("constant", t.kind, t.value);
195 }
196 if (this.take("(")) {
197 const e = this.expression();
198 this.expect(")");
199 return e;
200 }
201 if (t.kind === "W" && variable(t.text)) return this.target();
202 if (this.take("ERR")) return new Expr("builtin", "I", "ERR");
203 if (this.take("ERL")) return new Expr("builtin", "I", "ERL");
204 if (t.kind === "W" && signatures.has(t.text)) {
205 this.position++;
206 this.expect("(");
207 const args = [this.expression()];
208 while (this.take(",")) args.push(this.expression());
209 this.expect(")");
210 if (!signatures.get(t.text).includes(args.map((e) => e.type).join(""))) syntax();
211 return new Expr("call", nameType(t.text), t.text, args);
212 }
213 return syntax();
214 }
215 clause() {
216 if (this.current().kind === "I") return { line: this.number(), body: [] };
217 return { line: null, body: this.list(true) };
218 }
219 list(clause = false) {
220 const body = [this.statement(clause)];
221 while (this.take(":")) body.push(this.statement(clause));
222 return body;
223 }
224 statement(inClause) {
225 const word = this.current().text;
226 if (inClause && ["IF", "FOR", "NEXT", "WHILE", "WEND", "DATA"].includes(word)) syntax();
227 const s = new Statement(word);
228 if (
229 this.take("REM") ||
230 this.take("RETURN") ||
231 this.take("WEND") ||
232 this.take("END") ||
233 this.take("STOP")
234 )
235 return s;
236 if (this.take("PRINT")) {
237 let expressionLast = false;
238 while (this.current().kind !== "E" && !this.at(":") && !this.at("ELSE")) {
239 if (this.take(";")) {
240 s.items.push({ kind: ";", expr: null });
241 expressionLast = false;
242 } else if (this.take(",")) {
243 s.items.push({ kind: ",", expr: null });
244 expressionLast = false;
245 } else {
246 const kind = this.at("TAB") ? "TAB" : this.at("SPC") ? "SPC" : "value";
247 if (kind === "value" && expressionLast) syntax();
248 let expr;
249 if (kind !== "value") {
250 this.position++;
251 this.expect("(");
252 expr = this.integer();
253 this.expect(")");
254 } else expr = this.expression();
255 s.items.push({ kind, expr });
256 expressionLast = kind === "value";
257 }
258 }
259 return s;
260 }
261 if (this.take("INPUT") || this.take("READ") || this.take("DIM")) {
262 do {
263 const target = this.target();
264 if (word === "DIM" && target.kind !== "array") syntax();
265 s.targets.push(target);
266 } while (this.take(","));
267 return s;
268 }
269 if (this.take("DATA")) {
270 do {
271 const negative = this.take("-"),
272 t = this.current();
273 if (t.kind !== "I" && (t.kind !== "S" || negative)) syntax();
274 this.position++;
275 s.values.push(negative ? -t.value : t.value);
276 } while (this.take(","));
277 return s;
278 }
279 if (this.take("IF")) {
280 s.expr = this.integer();
281 this.expect("THEN");
282 s.clauses.push(this.clause());
283 if (this.take("ELSE")) s.clauses.push(this.clause());
284 if (this.current().kind !== "E") syntax();
285 return s;
286 }
287 if (this.take("GOTO") || this.take("GOSUB")) {
288 s.numbers.push(this.number());
289 return s;
290 }
291 if (this.take("ON")) {
292 if (this.take("ERROR")) {
293 s.kind = "handler";
294 this.expect("GOTO");
295 s.numbers.push(this.number());
296 } else {
297 s.expr = this.integer();
298 if (this.take("GOTO")) s.mode = "GOTO";
299 else {
300 this.expect("GOSUB");
301 s.mode = "GOSUB";
302 }
303 do {
304 s.numbers.push(this.number());
305 } while (this.take(","));
306 }
307 return s;
308 }
309 if (this.take("RESUME")) {
310 if (this.take("NEXT")) s.mode = "NEXT";
311 else if (this.current().kind === "I") {
312 s.mode = "line";
313 s.numbers.push(this.number());
314 }
315 return s;
316 }
317 if (this.take("FOR")) {
318 const t = this.current();
319 if (t.kind !== "W" || !variable(t.text) || nameType(t.text) !== "I") syntax();
320 s.name = t.text;
321 this.position++;
322 this.expect("=");
323 s.expressions.push(this.integer());
324 this.expect("TO");
325 s.expressions.push(this.integer());
326 s.expressions.push(this.take("STEP") ? this.integer() : new Expr("constant", "I", 1));
327 return s;
328 }
329 if (this.take("NEXT")) {
330 if (
331 this.current().kind === "W" &&
332 variable(this.current().text) &&
333 nameType(this.current().text) === "I"
334 )
335 s.name = this.tokens[this.position++].text;
336 return s;
337 }
338 if (this.take("WHILE") || this.take("RANDOMIZE")) {
339 s.expr = this.integer();
340 return s;
341 }
342 if (this.take("RESTORE")) {
343 if (this.current().kind === "I") s.numbers.push(this.number());
344 return s;
345 }
346 this.take("LET");
347 s.kind = "assign";
348 s.target = this.target();
349 this.expect("=");
350 s.expr = this.expression();
351 if (s.expr.type !== s.target.type) syntax();
352 return s;
353 }
354 parse() {
355 const body = this.list();
356 if (this.current().kind !== "E") syntax();
357 return body;
358 }
359}

361class Machine {
362 variables = new Map();
363 arrays = new Map();
364 stack = [];
365 entries = new Map();
366 sequence = [];
367 data = [];
368 dataPosition = 0;
369 inputPosition = 0;
370 random = 1;
371 handler = 0;
372 handling = false;
373 remembered = null;
374 err = 0;
375 erl = 0;
376 buffer = "";
377 output = [];
378 constructor(inputs) {
379 this.inputs = inputs;
380 }
381 endLine() {
382 this.output.push(this.buffer.replace(/ +$/, ""));
383 this.buffer = "";
384 }
385 message(text) {
386 if (this.buffer !== "") this.endLine();
387 this.output.push(text);
388 }
389 checked(n) {
390 if (!Number.isFinite(n) || n < -32768 || n > 32767) fault("OVERFLOW");
391 return n;
392 }
393 get(name) {
394 return this.variables.get(name) ?? (nameType(name) === "S" ? "" : 0);
395 }
396 declare(name, bounds) {
397 const count = bounds.reduce((a, b) => a * (b + 1), 1);
398 this.arrays.set(name, {
399 bounds,
400 values: Array(count).fill(nameType(name) === "S" ? "" : 0),
401 });
402 }
403 address(target) {
404 const name = target.value;
405 if (target.kind === "variable") return { name, index: null };
406 const indices = target.args.map((e) => this.evaluate(e));
407 if (!this.arrays.has(name))
408 this.declare(
409 name,
410 indices.map(() => 10),
411 );
412 const array = this.arrays.get(name);
413 if (
414 indices.length !== array.bounds.length ||
415 indices.some((n, i) => n < 0 || n > array.bounds[i])
416 )
417 fault("SUBSCRIPT OUT OF RANGE");
418 let index = 0;
419 indices.forEach((n, i) => {
420 index = index * (array.bounds[i] + 1) + n;
421 });
422 return { name, index };
423 }
424 load(address) {
425 return address.index === null
426 ? this.get(address.name)
427 : this.arrays.get(address.name).values[address.index];
428 }
429 store(address, value) {
430 if (address.index === null) this.variables.set(address.name, value);
431 else this.arrays.get(address.name).values[address.index] = value;
432 }
433 evaluate(e) {
434 const op = e.value;
435 if (e.kind === "constant") return op;
436 if (e.kind === "variable" || e.kind === "array") return this.load(this.address(e));
437 if (e.kind === "builtin") return op === "ERR" ? this.err : this.erl;
438 if (e.kind === "unary") {
439 const a = this.evaluate(e.args[0]);
440 return op === "NOT" ? ~a : this.checked(-a);
441 }
442 const args = e.args.map((a) => this.evaluate(a));
443 if (e.kind === "call") return this.call(op, args);
444 const [a, b] = args;
445 if (["=", "<>", "<", ">", "<=", ">="].includes(op)) {
446 const order = a === b ? 0 : a < b ? -1 : 1;
447 return {
448 "=": order === 0,
449 "<>": order !== 0,
450 "<": order < 0,
451 ">": order > 0,
452 "<=": order <= 0,
453 ">=": order >= 0,
454 }[op]
455 ? -1
456 : 0;
457 }
458 if (e.type === "S") {
459 const text = a + b;
460 if (text.length > 255) fault("STRING TOO LONG");
461 return text;
462 }
463 const x = a,
464 y = b;
465 switch (op) {
466 case "+":
467 return this.checked(x + y);
468 case "-":
469 return this.checked(x - y);
470 case "*":
471 return this.checked(x * y);
472 case "/":
473 if (y === 0) fault("DIVISION BY ZERO");
474 return this.checked(Math.trunc(x / y));
475 case "MOD":
476 if (y === 0) fault("DIVISION BY ZERO");
477 return x % y;
478 case "^":
479 if (y < 0) fault("ILLEGAL FUNCTION CALL");
480 return this.checked(x ** y);
481 case "AND":
482 return x & y;
483 case "OR":
484 return x | y;
485 }
486 throw new Error("invalid expression");
487 }
488 call(name, args) {
489 const n = args[0],
490 s = args[0],
491 k = args[1];
492 switch (name) {
493 case "ABS":
494 return this.checked(Math.abs(n));
495 case "SGN":
496 return Math.sign(n);
497 case "INT":
498 return n;
499 case "SQR":
500 if (n < 0) fault("ILLEGAL FUNCTION CALL");
501 return Math.floor(Math.sqrt(n));
502 case "RND":
503 if (n < 1) fault("ILLEGAL FUNCTION CALL");
504 this.random = (48271 * this.random) % 2147483647;
505 return (this.random % n) + 1;
506 case "LEN":
507 return s.length;
508 case "ASC":
509 if (s === "") fault("ILLEGAL FUNCTION CALL");
510 return s.charCodeAt(0);
511 case "VAL": {
512 const m = /^ *([+-]?[0-9]+)/.exec(s);
513 return this.checked(m ? Number(m[1]) : 0);
514 }
515 case "CHR$":
516 if (n < 32 || n > 126) fault("ILLEGAL FUNCTION CALL");
517 return String.fromCharCode(n);
518 case "STR$":
519 return String(n);
520 case "LEFT$":
521 if (k < 0) fault("ILLEGAL FUNCTION CALL");
522 return s.slice(0, k);
523 case "RIGHT$":
524 if (k < 0) fault("ILLEGAL FUNCTION CALL");
525 return k === 0 ? "" : s.slice(-k);
526 case "MID$": {
527 if (k < 1 || (args.length === 3 && args[2] < 0)) fault("ILLEGAL FUNCTION CALL");
528 return s.slice(k - 1, args.length === 3 ? k - 1 + args[2] : undefined);
529 }
530 case "INSTR":
531 return s.indexOf(args[1]) + 1;
532 }
533 throw new Error("invalid function");
534 }
535 line(number) {
536 const s = this.entries.get(number);
537 if (!s) fault("UNDEFINED LINE NUMBER");
538 return s;
539 }
540 push(frame) {
541 if (this.stack.length === 100) fault("OUT OF MEMORY");
542 this.stack.push(frame);
543 }
544 search(kind, name = "") {
545 for (let i = this.stack.length - 1; i >= 0; i--) {
546 const f = this.stack[i];
547 if (f.kind === "GOSUB") break;
548 if (f.kind === kind && (!name || f.name === name)) return i;
549 }
550 return -1;
551 }
552 matching(s, open, close) {
553 let depth = 0;
554 for (let i = this.sequence.indexOf(s) + 1; i < this.sequence.length; i++) {
555 const t = this.sequence[i];
556 if (t.kind === open) depth++;
557 else if (t.kind === close) {
558 if (depth === 0) return t.next;
559 depth--;
560 }
561 }
562 return fault(open + " WITHOUT " + close);
563 }
564 subroutine(number, after) {
565 const target = this.line(number);
566 this.push({ kind: "GOSUB", position: after });
567 return target;
568 }
569 execute(s) {
570 let next = s.next;
571 switch (s.kind) {
572 case "REM":
573 case "DATA":
574 break;
575 case "assign": {
576 const address = this.address(s.target);
577 this.store(address, this.evaluate(s.expr));
578 break;
579 }
580 case "PRINT":
581 for (const item of s.items) {
582 if (item.kind === ",")
583 this.buffer += " ".repeat(10 - (this.buffer.length % 10));
584 else if (item.kind !== ";") {
585 const value = this.evaluate(item.expr);
586 if (item.kind === "value") this.buffer += String(value);
587 else {
588 const n = value;
589 if (n < 0) fault("ILLEGAL FUNCTION CALL");
590 this.buffer += " ".repeat(
591 item.kind === "SPC" ? n : Math.max(0, n - this.buffer.length),
592 );
593 }
594 }
595 }
596 if (s.items.length === 0 || ![";", ","].includes(s.items[s.items.length - 1].kind))
597 this.endLine();
598 break;
599 case "INPUT":
600 case "READ":
601 for (const target of s.targets) {
602 const address = this.address(target);
603 let value;
604 if (s.kind === "READ") {
605 if (this.dataPosition === this.data.length) fault("OUT OF DATA");
606 value = this.data[this.dataPosition++].value;
607 if (valueType(value) !== target.type) fault("TYPE MISMATCH");
608 } else {
609 while (true) {
610 if (this.inputPosition === this.inputs.length) fault("INPUT PAST END");
611 const text = this.inputs[this.inputPosition++];
612 if (target.type === "S") {
613 value = text;
614 break;
615 }
616 const n = Number(text);
617 if (/^-?[0-9]+$/.test(text) && n >= -32768 && n <= 32767) {
618 value = n;
619 break;
620 }
621 this.message("?REDO FROM START");
622 }
623 }
624 this.store(address, value);
625 }
626 break;
627 case "RESTORE":
628 if (s.numbers.length) {
629 this.line(s.numbers[0]);
630 const p = this.data.findIndex((d) => d.line >= s.numbers[0]);
631 this.dataPosition = p < 0 ? this.data.length : p;
632 } else this.dataPosition = 0;
633 break;
634 case "IF": {
635 const clause = s.clauses[this.evaluate(s.expr) !== 0 ? 0 : 1];
636 if (clause) next = clause.line !== null ? this.line(clause.line) : clause.body[0];
637 break;
638 }
639 case "GOTO":
640 next = this.line(s.numbers[0]);
641 break;
642 case "GOSUB":
643 next = this.subroutine(s.numbers[0], next);
644 break;
645 case "ON": {
646 const n = this.evaluate(s.expr);
647 if (n < 0) fault("ILLEGAL FUNCTION CALL");
648 if (n >= 1 && n <= s.numbers.length)
649 next =
650 s.mode === "GOTO"
651 ? this.line(s.numbers[n - 1])
652 : this.subroutine(s.numbers[n - 1], next);
653 break;
654 }
655 case "RETURN": {
656 let i = this.stack.length - 1;
657 while (i >= 0 && this.stack[i].kind !== "GOSUB") i--;
658 if (i < 0) fault("RETURN WITHOUT GOSUB");
659 next = this.stack[i].position;
660 this.stack.length = i;
661 break;
662 }
663 case "FOR": {
664 const [start, final, step] = s.expressions.map((e) => this.evaluate(e));
665 if (step === 0) fault("ILLEGAL FUNCTION CALL");
666 const i = this.search("FOR", s.name);
667 if (i >= 0) this.stack.length = i;
668 this.variables.set(s.name, start);
669 if (step > 0 ? start > final : start < final)
670 next = this.matching(s, "FOR", "NEXT");
671 else this.push({ kind: "FOR", name: s.name, final, step, position: next });
672 break;
673 }
674 case "NEXT": {
675 const i = this.search("FOR", s.name);
676 if (i < 0) fault("NEXT WITHOUT FOR");
677 this.stack.length = i + 1;
678 const f = this.stack[i],
679 step = f.step,
680 final = f.final;
681 const value = this.checked(this.get(f.name) + step);
682 this.variables.set(f.name, value);
683 if (step > 0 ? value > final : value < final) this.stack.pop();
684 else next = f.position;
685 break;
686 }
687 case "WHILE":
688 if (this.evaluate(s.expr) !== 0) this.push({ kind: "WHILE", position: s });
689 else next = this.matching(s, "WHILE", "WEND");
690 break;
691 case "WEND": {
692 const i = this.search("WHILE");
693 if (i < 0) fault("WEND WITHOUT WHILE");
694 next = this.stack[i].position;
695 this.stack.length = i;
696 break;
697 }
698 case "DIM":
699 for (const target of s.targets) {
700 const name = target.value,
701 bounds = target.args.map((e) => this.evaluate(e));
702 if (this.arrays.has(name)) fault("DUPLICATE DEFINITION");
703 if (bounds.some((n) => n < 0)) fault("ILLEGAL FUNCTION CALL");
704 if (bounds.reduce((a, b) => a * (b + 1), 1) > 10000) fault("OUT OF MEMORY");
705 this.declare(name, bounds);
706 }
707 break;
708 case "RANDOMIZE": {
709 const n = this.evaluate(s.expr);
710 if (n < 1) fault("ILLEGAL FUNCTION CALL");
711 this.random = n;
712 break;
713 }
714 case "handler":
715 if (s.numbers[0] !== 0) this.line(s.numbers[0]);
716 this.handler = s.numbers[0];
717 break;
718 case "RESUME":
719 if (!this.handling) fault("RESUME WITHOUT ERROR");
720 next =
721 s.mode === "line"
722 ? this.line(s.numbers[0])
723 : s.mode === "NEXT"
724 ? this.remembered.next
725 : this.remembered;
726 this.handling = false;
727 break;
728 case "STOP":
729 this.message("BREAK IN LINE " + s.line);
730 return null;
731 case "END":
732 return null;
733 }
734 return next;
735 }
736 connect(body, after, line) {
737 let next = after;
738 for (let i = body.length - 1; i >= 0; i--) {
739 const s = body[i];
740 s.line = line;
741 s.next = next;
742 for (const c of s.clauses) this.connect(c.body, next, line);
743 next = s;
744 }
745 }
746 run(source) {
747 const parsed = [];
748 for (const line of [...source.keys()].sort((a, b) => a - b)) {
749 try {
750 parsed.push([line, new Parser(tokenize(source.get(line))).parse()]);
751 } catch (e) {
752 if (!(e instanceof SyntaxFault)) throw e;
753 this.message(`ERROR IN LINE ${line}: SYNTAX ERROR`);
754 return this.finish();
755 }
756 }
757 let after = null;
758 for (let i = parsed.length - 1; i >= 0; i--) {
759 const [line, body] = parsed[i];
760 this.connect(body, after, line);
761 this.entries.set(line, body[0]);
762 after = body[0];
763 }
764 this.sequence = parsed.flatMap(([, body]) => body);
765 for (const s of this.sequence)
766 if (s.kind === "DATA")
767 for (const value of s.values) this.data.push({ line: s.line, value });
768 let position = after;
769 while (position !== null) {
770 const statement = position;
771 try {
772 position = this.execute(statement);
773 } catch (e) {
774 if (!(e instanceof RuntimeFault)) throw e;
775 if (this.handler !== 0 && !this.handling) {
776 this.err = errorCodes.get(e.message);
777 this.erl = statement.line;
778 this.remembered = statement;
779 this.handling = true;
780 position = this.line(this.handler);
781 } else {
782 this.message(`ERROR IN LINE ${statement.line}: ${e.message}`);
783 position = null;
784 }
785 }
786 }
787 return this.finish();
788 }
789 finish() {
790 if (this.buffer !== "") this.endLine();
791 this.output.push("READY");
792 return this.output;
793 }
