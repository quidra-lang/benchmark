110List<Token> tokenize(String text) {
111 final tokens = <Token>[];
112 var p = 0;
113 while (p < text.length) {
114 final ch = text[p];
115 if (ch == ' ') {
116 p++;
117 continue;
118 }
119 if (ch == '"') {
120 final end = text.indexOf('"', p + 1);
121 if (end < 0) syntax();
122 tokens.add(
123 Token('S', text.substring(p, end + 1), text.substring(p + 1, end)),
124 );
125 p = end + 1;
126 } else if (digit(ch)) {
127 final start = p;
128 while (p < text.length && digit(text[p])) p++;
129 final word = text.substring(start, p), value = int.tryParse(word);
130 if (value == null || value > 32767) syntax();
131 tokens.add(Token('I', word, value));
132 } else if (upper(ch)) {
133 if (text.startsWith('REM', p)) {
134 tokens.add(const Token('W', 'REM', 'REM'));
135 break;
136 }
137 final start = p;
138 while (p < text.length && (upper(text[p]) || digit(text[p]))) p++;
139 if (p < text.length && text[p] == '\$') p++;
140 final word = text.substring(start, p);
141 tokens.add(Token('W', word, word));
142 } else {
143 if (!'+-*/^=<>() ,;:'.contains(ch)) syntax();
144 final pair = text.substring(p, math.min(p + 2, text.length));
145 final word = ['<>', '<=', '>='].contains(pair) ? pair : ch;
146 tokens.add(Token('P', word, word));
147 p += word.length;
148 }
149 }
150 tokens.add(const Token('E', '', ''));
151 return tokens;
152}

154class Parser {
155 final List<Token> tokens;
156 int pos = 0;
157 Parser(this.tokens);
158 Token get current => tokens[pos];
159 bool at(String s) =>
160 (current.kind == 'P' || current.kind == 'W') && current.text == s;
161 bool take(String s) {
162 if (!at(s)) return false;
163 pos++;
164 return true;
165 }
166
167 void expect(String s) {
168 if (!take(s)) syntax();
169 }
170
171 int number() {
172 if (current.kind != 'I') syntax();
173 return tokens[pos++].value as int;
174 }
175
176 Expr integer([int level = 0]) {
177 final e = expression(level);
178 if (e.type != 'I') syntax();
179 return e;
180 }
181
182 Expr target() {
183 final token = current;
184 if (token.kind != 'W' || !variableName(token.text)) syntax();
185 pos++;
186 final result = Expr('variable', nameType(token.text), token.text);
187 if (take('(')) {
188 result.kind = 'array';
189 result.args.add(integer());
190 if (take(',')) result.args.add(integer());
191 expect(')');
192 }
193 return result;
194 }
195
196 Expr primary() {
197 final token = current;
198 if (token.kind == 'I' || token.kind == 'S') {
199 pos++;
200 return Expr('constant', token.kind, token.value);
201 }
202 if (take('(')) {
203 final e = expression();
204 expect(')');
205 return e;
206 }
207 if (token.kind == 'W' && variableName(token.text)) return target();
208 if (take('ERR')) return Expr('builtin', 'I', 'ERR');
209 if (take('ERL')) return Expr('builtin', 'I', 'ERL');
210 if (token.kind == 'W' && signatures.containsKey(token.text)) {
211 pos++;
212 expect('(');
213 final args = <Expr>[expression()];
214 while (take(',')) args.add(expression());
215 expect(')');
216 if (!signatures[token.text]!.contains(args.map((e) => e.type).join()))
217 syntax();
218 return Expr('call', nameType(token.text), token.text, args);
219 }
220 syntax();
221 }
222
223 Expr expression([int level = 0]) {
224 if (level == 2 && take('NOT'))
225 return Expr('unary', 'I', 'NOT', [integer(2)]);
226 if (level == 6 && take('-')) return Expr('unary', 'I', '-', [integer(6)]);
227 if (level == 8) return primary();
228 var left = expression(level + 1);
229 while (operators[level].contains(current.text)) {
230 final op = current.text;
231 pos++;
232 final right = expression(level == 7 ? 6 : level + 1);
233 if (left.type != right.type ||
234 (left.type == 'S' && level != 3 && op != '+'))
235 syntax();
236 left = Expr('binary', level == 3 ? 'I' : left.type, op, [left, right]);
237 if (level == 7) break;
238 }
239 return left;
240 }
241
242 Clause clause() =>
243 current.kind == 'I' ? Clause(number(), []) : Clause(null, listing(true));
244 List<Statement> listing([bool inside = false]) {
245 final body = <Statement>[statement(inside)];
246 while (take(':')) body.add(statement(inside));
247 return body;
248 }
249
250 Statement statement(bool inside) {
251 final word = current.text;
252 if (inside && ['IF', 'FOR', 'NEXT', 'WHILE', 'WEND', 'DATA'].contains(word))
253 syntax();
254 final s = Statement(word);
255 for (final kind in ['REM', 'RETURN', 'WEND', 'END', 'STOP']) {
256 if (take(kind)) return s;
257 }
258 if (take('PRINT')) {
259 var expressionLast = false;
260 while (current.kind != 'E' && !at(':') && !at('ELSE')) {
261 if (take(';')) {
262 s.items.add(const PrintItem(';', null));
263 expressionLast = false;
264 } else if (take(',')) {
265 s.items.add(const PrintItem(',', null));
266 expressionLast = false;
267 } else {
268 final kind = at('TAB')
269 ? 'TAB'
270 : at('SPC')
271 ? 'SPC'
272 : 'value';
273 if (kind == 'value' && expressionLast) syntax();
274 late Expr e;
275 if (kind == 'value') {
276 e = expression();
277 } else {
278 pos++;
279 expect('(');
280 e = integer();
281 expect(')');
282 }
283 s.items.add(PrintItem(kind, e));
284 expressionLast = kind == 'value';
285 }
286 }
287 return s;
288 }
289 if (take('INPUT') || take('READ') || take('DIM')) {
290 do {
291 final t = target();
292 if (word == 'DIM' && t.kind != 'array') syntax();
293 s.targets.add(t);
294 } while (take(','));
295 return s;
296 }
297 if (take('DATA')) {
298 do {
299 final negative = take('-');
300 final t = current;
301 if (t.kind != 'I' && (t.kind != 'S' || negative)) syntax();
302 pos++;
303 s.values.add(negative ? -(t.value as int) : t.value);
304 } while (take(','));
305 return s;
306 }
307 if (take('IF')) {
308 s.expr = integer();
309 expect('THEN');
310 s.clauses.add(clause());
311 if (take('ELSE')) s.clauses.add(clause());
312 if (current.kind != 'E') syntax();
313 return s;
314 }
315 if (take('GOTO') || take('GOSUB')) {
316 s.numbers.add(number());
317 return s;
318 }
319 if (take('ON')) {
320 if (take('ERROR')) {
321 s.kind = 'handler';
322 expect('GOTO');
323 s.numbers.add(number());
324 } else {
325 s.expr = integer();
326 if (take('GOTO'))
327 s.mode = 'GOTO';
328 else {
329 expect('GOSUB');
330 s.mode = 'GOSUB';
331 }
332 s.numbers.add(number());
333 while (take(',')) s.numbers.add(number());
334 }
335 return s;
336 }
337 if (take('RESUME')) {
338 if (take('NEXT'))
339 s.mode = 'NEXT';
340 else if (current.kind == 'I') {
341 s.mode = 'line';
342 s.numbers.add(number());
343 }
344 return s;
345 }
346 if (take('FOR')) {
347 final t = current;
348 if (t.kind != 'W' || !variableName(t.text) || nameType(t.text) != 'I')
349 syntax();
350 s.name = t.text;
351 pos++;
352 expect('=');
353 s.expressions.add(integer());
354 expect('TO');
355 s.expressions.add(integer());
356 s.expressions.add(take('STEP') ? integer() : Expr('constant', 'I', 1));
357 return s;
358 }
359 if (take('NEXT')) {
360 final t = current;
361 if (t.kind == 'W' && variableName(t.text) && nameType(t.text) == 'I') {
362 s.name = t.text;
363 pos++;
364 }
365 return s;
366 }
367 if (take('WHILE') || take('RANDOMIZE')) {
368 s.expr = integer();
369 return s;
370 }
371 if (take('RESTORE')) {
372 if (current.kind == 'I') s.numbers.add(number());
373 return s;
374 }
375 take('LET');
376 s.kind = 'assign';
377 s.target = target();
378 expect('=');
379 s.expr = expression();
380 if (s.expr!.type != s.target!.type) syntax();
381 return s;
382 }
383
384 List<Statement> parse() {
385 final result = listing();
386 if (current.kind != 'E') syntax();
387 return result;
388 }
389}

455 if (n < -32768 || n > 32767) fault('OVERFLOW');

461 var length = 1;

469 Address address(Expr target) {
470 final name = target.value as String;
471 if (target.kind == 'variable') return Address(name, null);
472 final indices = target.args.map((e) => evaluate(e) as int).toList();
473 if (!arrays.containsKey(name))
474 declare(name, List<int>.filled(indices.length, 10));
475 final arr = arrays[name]!;
476 if (indices.length != arr.bounds.length) fault('SUBSCRIPT OUT OF RANGE');
477 var index = 0;
478 for (var i = 0; i < indices.length; i++) {
479 final n = indices[i];
480 if (n < 0 || n > arr.bounds[i]) fault('SUBSCRIPT OUT OF RANGE');
481 index = index * (arr.bounds[i] + 1) + n;
482 }
483 return Address(name, index);
484 }

486 Object load(Address address) => address.index == null
487 ? valueOf(address.name)
488 : arrays[address.name]!.values[address.index!];

490 if (a.index == null)
491 variables[a.name] = value;
492 else
493 arrays[a.name]!.values[a.index!] = value;

496 Object evaluate(Expr e) {
497 final op = e.value;
498 if (e.kind == 'constant') return op;
499 if (e.kind == 'variable' || e.kind == 'array') return load(address(e));
500 if (e.kind == 'builtin') return op == 'ERR' ? err : erl;
501 if (e.kind == 'unary') {
502 final a = evaluate(e.args[0]) as int;
503 return op == 'NOT' ? ~a : checked(-a);
504 }
505 final args = e.args.map(evaluate).toList();
506 if (e.kind == 'call') return call(op as String, args);
507 final a = args[0];
508 final b = args[1];
509 if (operators[3].contains(op)) {
510 final cmp = a is String
511 ? a.compareTo(b as String)
512 : (a as int).compareTo(b as int);
513 final success = switch (op) {
514 '=' => cmp == 0,
515 '<>' => cmp != 0,
516 '<' => cmp < 0,
517 '>' => cmp > 0,
518 '<=' => cmp <= 0,
519 '>=' => cmp >= 0,
520 _ => false,
521 };
522 return success ? -1 : 0;
523 }
524 if (e.type == 'S') {
525 final result = (a as String) + (b as String);
526 if (result.length > 255) fault('STRING TOO LONG');
527 return result;
528 }
529 final x = a as int, y = b as int;
530 switch (op) {
531 case '+':
532 return checked(x + y);
533 case '-':
534 return checked(x - y);
535 case '*':
536 return checked(x * y);
537 case '/':
538 if (y == 0) fault('DIVISION BY ZERO');
539 return checked(x ~/ y);
540 case 'MOD':
541 if (y == 0) fault('DIVISION BY ZERO');
542 return x - y * (x ~/ y);
543 case '^':
544 if (y < 0) fault('ILLEGAL FUNCTION CALL');
545 if (x.abs() > 1 && y > 15) fault('OVERFLOW');
546 var result = 1;
547 for (var i = 0; i < y; i++) result = checked(result * x);
548 return result;
549 case 'AND':
550 return x & y;
551 case 'OR':
552 return x | y;
553 }
554 fault('ILLEGAL FUNCTION CALL');
555 }
556
557 Object call(String name, List<Object> args) {
558 final a = args[0];
559 final b = args.length > 1 ? args[1] : null;
560 final c = args.length > 2 ? args[2] : null;
561 switch (name) {
562 case 'ABS':
563 return checked((a as int).abs());
564 case 'SGN':
565 final n = a as int;
566 return n < 0
567 ? -1
568 : n > 0
569 ? 1
570 : 0;
571 case 'INT':
572 return a;
573 case 'SQR':
574 final n = a as int;
575 if (n < 0) fault('ILLEGAL FUNCTION CALL');
576 return math.sqrt(n).floor();
577 case 'RND':
578 final n = a as int;
579 if (n < 1) fault('ILLEGAL FUNCTION CALL');
580 random = (48271 * random) % 2147483647;
581 return random % n + 1;
582 case 'LEN':
583 return (a as String).length;
584 case 'ASC':
585 final s = a as String;
586 if (s.isEmpty) fault('ILLEGAL FUNCTION CALL');
587 return s.codeUnitAt(0);
588 case 'VAL':
589 final match = RegExp(r'^ *([+-]?[0-9]+)').firstMatch(a as String);
590 return checked(match == null ? 0 : (int.tryParse(match[1]!) ?? 32768));
591 case r'CHR$':
592 final n = a as int;
593 if (n < 32 || n > 126) fault('ILLEGAL FUNCTION CALL');
594 return String.fromCharCode(n);
595 case r'STR$':
596 return a.toString();
597 case r'LEFT$':
598 final s = a as String, n = b as int;
599 if (n < 0) fault('ILLEGAL FUNCTION CALL');
600 return s.substring(0, math.min(n, s.length));
601 case r'RIGHT$':
602 final s = a as String, n = b as int;
603 if (n < 0) fault('ILLEGAL FUNCTION CALL');
604 return s.substring(s.length - math.min(n, s.length));
605 case r'MID$':
606 final s = a as String, start = b as int, len = c as int?;
607 if (start < 1 || (len != null && len < 0))
608 fault('ILLEGAL FUNCTION CALL');
609 final from = math.min(start - 1, s.length);
610 return s.substring(
611 from,
612 len == null ? s.length : math.min(from + len, s.length),
613 );
614 case 'INSTR':
615 return (a as String).indexOf(b as String) + 1;
616 }
617 fault('ILLEGAL FUNCTION CALL');
618 }

631 int search(String kind, [String name = '']) {
632 for (var i = stack.length - 1; i >= 0; i--) {
633 final f = stack[i];
634 if (f.kind == 'GOSUB') break;
635 if (f.kind == kind && (name.isEmpty || f.name == name)) return i;

640 Statement? matching(Statement s, String opening, String closing) {
641 final start = sequence.indexOf(s);
642 var depth = 0;
643 for (var i = start + 1; i < sequence.length; i++) {
644 final t = sequence[i];
645 if (t.kind == opening)
646 depth++;
647 else if (t.kind == closing) {
648 if (depth == 0) return t.next;
649 depth--;
650 }
651 }
652 fault(opening + ' WITHOUT ' + closing);
653 }

661 Statement? execute(Statement s) {
662 Statement? following = s.next;
663 switch (s.kind) {
664 case 'assign':
665 final place = address(s.target!);
666 store(place, evaluate(s.expr!));
667 break;
668 case 'PRINT':
669 for (final item in s.items) {
670 if (item.kind == ',') {
671 buffer += ''.padRight(10 - buffer.length % 10);
672 } else if (item.kind != ';') {
673 final value = evaluate(item.expr!);
674 if (item.kind == 'value') {
675 buffer += value.toString();
676 } else {
677 final n = value as int;
678 if (n < 0) fault('ILLEGAL FUNCTION CALL');
679 buffer += ''.padRight(
680 item.kind == 'SPC' ? n : math.max(0, n - buffer.length),
681 );
682 }
683 }
684 }
685 if (s.items.isEmpty || ![';', ','].contains(s.items.last.kind))
686 endLine();
687 break;
688 case 'INPUT':
689 case 'READ':
690 for (final target in s.targets) {
691 final place = address(target);
692 late Object value;
693 if (s.kind == 'READ') {
694 if (dataPos == data.length) fault('OUT OF DATA');
695 value = data[dataPos++].value;
696 if (valueType(value) != target.type) fault('TYPE MISMATCH');
697 } else {
698 while (true) {
699 if (inputPos == inputs.length) fault('INPUT PAST END');
700 final text = inputs[inputPos++];
701 if (target.type == 'S') {
702 value = text;
703 break;
704 }
705 final n = int.tryParse(text);
706 if (RegExp(r'^-?[0-9]+$').hasMatch(text) &&
707 n != null &&
708 n >= -32768 &&
709 n <= 32767) {
710 value = n;
711 break;
712 }
713 message('?REDO FROM START');
714 }
715 }
716 store(place, value);
717 }
718 break;
719 case 'RESTORE':
720 if (s.numbers.isEmpty)
721 dataPos = 0;
722 else {
723 final number = s.numbers[0];
724 line(number);
725 dataPos = data.indexWhere((entry) => entry.line >= number);
726 if (dataPos < 0) dataPos = data.length;
727 }
728 break;
729 case 'IF':
730 final ix = (evaluate(s.expr!) as int) == 0 ? 1 : 0;
731 if (ix < s.clauses.length) {
732 final selected = s.clauses[ix];
733 following = selected.line == null
734 ? selected.body[0]
735 : line(selected.line!);
736 }
737 break;
738 case 'GOTO':
739 following = line(s.numbers[0]);
740 break;
741 case 'GOSUB':
742 following = subroutine(s.numbers[0], following);
743 break;
744 case 'ON':
745 final n = evaluate(s.expr!) as int;
746 if (n < 0) fault('ILLEGAL FUNCTION CALL');
747 if (n >= 1 && n <= s.numbers.length) {
748 following = s.mode == 'GOTO'
749 ? line(s.numbers[n - 1])
750 : subroutine(s.numbers[n - 1], following);
751 }
752 break;
753 case 'RETURN':
754 final ix = stack.lastIndexWhere((f) => f.kind == 'GOSUB');
755 if (ix < 0) fault('RETURN WITHOUT GOSUB');
756 following = stack[ix].position;
757 stack.removeRange(ix, stack.length);
758 break;
759 case 'FOR':
760 final values = s.expressions.map((e) => evaluate(e) as int).toList();
761 final start = values[0], limit = values[1], step = values[2];
762 if (step == 0) fault('ILLEGAL FUNCTION CALL');
763 final ix = search('FOR', s.name);
764 if (ix >= 0) stack.removeRange(ix, stack.length);
765 variables[s.name] = start;
766 if ((step > 0 && start > limit) || (step < 0 && start < limit)) {
767 following = matching(s, 'FOR', 'NEXT');
768 } else
769 push(Frame('FOR', following, s.name, limit, step));
770 break;
771 case 'NEXT':
772 final ix = search('FOR', s.name);
773 if (ix < 0) fault('NEXT WITHOUT FOR');
774 stack.removeRange(ix + 1, stack.length);
775 final frame = stack[ix];
776 final value = checked((valueOf(frame.name) as int) + frame.step);
777 variables[frame.name] = value;
778 if ((frame.step > 0 && value > frame.limit) ||
779 (frame.step < 0 && value < frame.limit))
780 stack.removeLast();
781 else
782 following = frame.position;
783 break;
784 case 'WHILE':
785 if ((evaluate(s.expr!) as int) == 0)
786 following = matching(s, 'WHILE', 'WEND');
787 else
788 push(Frame('WHILE', s));
789 break;
790 case 'WEND':
791 final ix = search('WHILE');
792 if (ix < 0) fault('WEND WITHOUT WHILE');
793 following = stack[ix].position;
794 stack.removeRange(ix, stack.length);
795 break;
796 case 'DIM':
797 for (final t in s.targets) {
798 final name = t.value as String;
799 final bounds = t.args.map((e) => evaluate(e) as int).toList();
800 if (arrays.containsKey(name)) fault('DUPLICATE DEFINITION');
801 if (bounds.any((n) => n < 0)) fault('ILLEGAL FUNCTION CALL');
802 var size = 1;
803 for (final n in bounds) size *= n + 1;
804 if (size > 10000) fault('OUT OF MEMORY');
805 declare(name, bounds);
806 }
807 break;
808 case 'RANDOMIZE':
809 final n = evaluate(s.expr!) as int;
810 if (n < 1) fault('ILLEGAL FUNCTION CALL');
811 random = n;
812 break;
813 case 'handler':
814 if (s.numbers[0] != 0) line(s.numbers[0]);
815 handler = s.numbers[0];
816 break;
817 case 'RESUME':
818 if (!handling) fault('RESUME WITHOUT ERROR');
819 following = s.mode == 'line'
820 ? line(s.numbers[0])
821 : s.mode == 'NEXT'
822 ? remembered!.next
823 : remembered;
824 handling = false;
825 break;
826 case 'STOP':
827 message('BREAK IN LINE ' + s.line.toString());
828 return null;
829 case 'END':
830 return null;
831 }
832 return following;
833 }

835 void connect(List<Statement> body, Statement? after, int lineNumber) {
836 Statement? following = after;
837 for (final s in body.reversed) {
838 s.line = lineNumber;
839 s.next = following;
840 for (final c in s.clauses) connect(c.body, following, lineNumber);
841 following = s;
842 }
843 }
844
845 List<String> finish() {
846 if (buffer.isNotEmpty) endLine();
847 output.add('READY');
848 return output;
849 }

851 List<String> run(Map<int, String> source) {
852 final parsed = <ProgramLine>[];
853 final numbers = source.keys.toList()..sort();
854 for (final n in numbers) {
855 try {
856 parsed.add(ProgramLine(n, Parser(tokenize(source[n]!)).parse()));
857 } on SyntaxFault {
858 message('ERROR IN LINE ' + n.toString() + ': SYNTAX ERROR');
859 return finish();
860 }
861 }
862 Statement? after;
863 for (final e in parsed.reversed) {
864 connect(e.body, after, e.number);
865 entries[e.number] = e.body[0];
866 after = e.body[0];
867 }
868 for (final e in parsed) {
869 for (final s in e.body) {
870 sequence.add(s);
871 if (s.kind == 'DATA')
872 for (final v in s.values) data.add(Entry(s.line, v));
873 }
874 }
875 var pos = after;
876 while (pos != null) {
877 final current = pos;
878 try {
879 pos = execute(current);
880 } on BasicFault catch (failure) {
881 if (handler != 0 && !handling) {
882 err = errorCodes[failure.reason] ?? 0;
883 erl = current.line;
884 remembered = current;
885 handling = true;
886 try {
887 pos = line(handler);
888 } on BasicFault catch (secondary) {
889 message(
890 'ERROR IN LINE ' +
891 current.line.toString() +
892 ': ' +
893 secondary.reason,
894 );
895 pos = null;
896 }
897 } else {
898 message(
899 'ERROR IN LINE ' + current.line.toString() + ': ' + failure.reason,
900 );
901 pos = null;
902 }
903 }
904 }
905 return finish();
906 }
