286class Array:
287 def __init__(self, name, bounds):
288 self.bounds = bounds
289 self.values = [default_value(name)] * math.prod(bound + 1 for bound in bounds)

291 def index(self, subscripts):
292 if len(subscripts) != len(self.bounds):
293 raise BasicError('SUBSCRIPT OUT OF RANGE')
294 index = 0
295 for subscript, bound in zip(subscripts, self.bounds):
296 if not 0 <= subscript <= bound:
297 raise BasicError('SUBSCRIPT OUT OF RANGE')
298 index = index * (bound + 1) + subscript
299 return index

302def tokenize(content):
303 tokens = []
304 position = 0
305 while position < len(content):
306 found = TOKEN_PATTERN.match(content, position)
307 if found is None:
308 raise BasicSyntaxError()
309 kind = found.lastgroup
310 value = found.group(kind)
311 if kind == 'number':
312 value = int(value)
313 if value > MAX_INTEGER:
314 raise BasicSyntaxError()
315 elif kind == 'word' and VARIABLE_PATTERN.fullmatch(value):
316 kind = 'variable'
317 elif kind == 'word' and value not in RESERVED_WORDS and value not in FUNCTIONS:
318 raise BasicSyntaxError()
319 elif kind != 'string':
320 kind = 'keyword'
321 tokens.append(Token(kind, value))
322 position = found.end()
323 return tokens

326def is_string(expression):
327 match expression:
328 case Constant(value):
329 return isinstance(value, str)
330 case Variable(name) | ArrayElement(name) | FunctionCall(name):
331 return name.endswith('$')
332 case BinaryOp('+', left):
333 return is_string(left)
334 return False

337def require_integer(expression):
338 if is_string(expression):
339 raise BasicSyntaxError()
340 return expression

343def make_binary(symbol, left, right):
344 if symbol == '+' or symbol in COMPARISONS:
345 if is_string(left) != is_string(right):
346 raise BasicSyntaxError()
347 else:
348 require_integer(left)
349 require_integer(right)
350 return BinaryOp(symbol, left, right)

353class Parser:
354 def __init__(self, tokens):
355 self.tokens = [*tokens, Token('end', None)]
356 self.position = 0
357
358 def peek(self):
359 return self.tokens[self.position]
360
361 def advance(self):
362 token = self.tokens[self.position]
363 self.position += 1
364 return token
365
366 def at(self, *keywords):
367 token = self.peek()
368 return token.kind == 'keyword' and token.value in keywords
369
370 def accept(self, keyword):
371 if self.at(keyword):
372 self.position += 1
373 return True
374 return False
375
376 def expect(self, keyword):
377 if not self.accept(keyword):
378 raise BasicSyntaxError()
379
380 def expect_end(self):
381 if self.peek().kind != 'end':
382 raise BasicSyntaxError()
383
384 def at_statement_end(self):
385 return self.peek().kind == 'end' or self.at(':', 'ELSE')
386
387 def parse_list(self, parse_item, separator=','):
388 items = [parse_item()]
389 while self.accept(separator):
390 items.append(parse_item())
391 return items
392
393 def parse_line(self):
394 statements = self.parse_statements(in_clause=False)
395 self.expect_end()
396 return statements
397
398 def parse_statements(self, in_clause):
399 return self.parse_list(lambda: self.parse_statement(in_clause), ':')
400
401 def parse_statement(self, in_clause):
402 token = self.peek()
403 if token.kind == 'variable':
404 return self.parse_assignment()
405 if token.kind != 'keyword' or in_clause and token.value in NON_CLAUSE_STATEMENTS:
406 raise BasicSyntaxError()
407 self.advance()
408 match token.value:
409 case 'REM':
410 return Rem()
411 case 'LET':
412 return self.parse_assignment()
413 case 'PRINT':
414 return Print(self.parse_print_items())
415 case 'INPUT':
416 return Input(self.parse_list(self.parse_target))
417 case 'READ':
418 return Read(self.parse_list(self.parse_target))
419 case 'DATA':
420 return Data(self.parse_list(self.parse_data_item))
421 case 'IF':
422 return self.parse_if()
423 case 'GOTO':
424 return Goto(self.parse_line_number())
425 case 'GOSUB':
426 return Gosub(self.parse_line_number())
427 case 'RETURN':
428 return Return()
429 case 'ON':
430 return self.parse_on()
431 case 'RESUME':
432 if self.accept('NEXT'):
433 return Resume('NEXT')
434 return Resume(self.parse_optional_line_number())
435 case 'FOR':
436 return self.parse_for()
437 case 'NEXT':
438 has_variable = self.peek().kind == 'variable'
439 return Next(self.parse_integer_variable() if has_variable else None)
440 case 'WHILE':
441 return While(self.parse_integer_expression())
442 case 'WEND':
443 return Wend()
444 case 'DIM':
445 return Dim(self.parse_list(self.parse_array_declaration))
446 case 'RESTORE':
447 return Restore(self.parse_optional_line_number())
448 case 'RANDOMIZE':
449 return Randomize(self.parse_integer_expression())
450 case 'END':
451 return End()
452 case 'STOP':
453 return Stop()
454 raise BasicSyntaxError()
455
456 def parse_assignment(self):
457 target = self.parse_target()
458 self.expect('=')
459 expression = self.parse_expression()
460 if is_string(target) != is_string(expression):
461 raise BasicSyntaxError()
462 return Let(target, expression)
463
464 def parse_print_items(self):
465 items = []
466 while not self.at_statement_end():
467 if self.at(';', ','):
468 items.append(self.advance().value)
469 elif self.at('TAB', 'SPC'):
470 name = self.advance().value
471 self.expect('(')
472 argument = self.parse_integer_expression()
473 self.expect(')')
474 items.append(PrintFunction(name, argument))
475 elif items and isinstance(items[-1], Expression):
476 raise BasicSyntaxError()
477 else:
478 items.append(self.parse_expression())
479 return items

481 def parse_data_item(self):
482 negative = self.accept('-')
483 token = self.advance()
484 if token.kind == 'number':
485 return -token.value if negative else token.value
486 if token.kind == 'string' and not negative:
487 return token.value
488 raise BasicSyntaxError()

490 def parse_if(self):
491 condition = self.parse_integer_expression()
492 self.expect('THEN')
493 then_clause = self.parse_clause()
494 else_clause = self.parse_clause() if self.accept('ELSE') else None
495 self.expect_end()
496 return If(condition, then_clause, else_clause)
497
498 def parse_clause(self):
499 if self.peek().kind == 'number':
500 return self.advance().value
501 return self.parse_statements(in_clause=True)
502
503 def parse_on(self):
504 if self.accept('ERROR'):
505 self.expect('GOTO')
506 return OnError(self.parse_line_number())
507 selector = self.parse_integer_expression()
508 if not self.at('GOTO', 'GOSUB'):
509 raise BasicSyntaxError()
510 is_gosub = self.advance().value == 'GOSUB'
511 return OnJump(selector, self.parse_list(self.parse_line_number), is_gosub)

513 def parse_for(self):
514 variable = self.parse_integer_variable()
515 self.expect('=')
516 start = self.parse_integer_expression()
517 self.expect('TO')
518 limit = self.parse_integer_expression()
519 step = self.parse_integer_expression() if self.accept('STEP') else Constant(1)
520 return For(variable, start, limit, step)

522 def parse_line_number(self):
523 token = self.advance()
524 if token.kind != 'number':
525 raise BasicSyntaxError()
526 return token.value
527
528 def parse_optional_line_number(self):
529 return self.advance().value if self.peek().kind == 'number' else None
530
531 def parse_integer_variable(self):
532 token = self.advance()
533 if token.kind != 'variable' or token.value.endswith('$'):
534 raise BasicSyntaxError()
535 return token.value
536
537 def parse_target(self):
538 token = self.advance()
539 if token.kind != 'variable':
540 raise BasicSyntaxError()
541 return self.parse_reference(token.value)
542
543 def parse_reference(self, name):
544 if not self.accept('('):
545 return Variable(name)
546 subscripts = self.parse_list(self.parse_integer_expression)
547 self.expect(')')
548 if len(subscripts) > 2:
549 raise BasicSyntaxError()
550 return ArrayElement(name, subscripts)
551
552 def parse_array_declaration(self):
553 declaration = self.parse_target()
554 if not isinstance(declaration, ArrayElement):
555 raise BasicSyntaxError()
556 return declaration
557
558 def parse_integer_expression(self):
559 return require_integer(self.parse_expression())
560
561 def parse_binary(self, operators, parse_operand):
562 left = parse_operand()
563 while self.at(*operators):
564 symbol = self.advance().value
565 left = make_binary(symbol, left, parse_operand())
566 return left
567
568 def parse_expression(self):
569 return self.parse_binary(('OR',), self.parse_conjunction)
570
571 def parse_conjunction(self):
572 return self.parse_binary(('AND',), self.parse_negation)
573
574 def parse_negation(self):
575 if self.accept('NOT'):
576 return UnaryOp('NOT', require_integer(self.parse_negation()))
577 return self.parse_comparison()
578
579 def parse_comparison(self):
580 return self.parse_binary(COMPARISONS, self.parse_sum)
581
582 def parse_sum(self):
583 return self.parse_binary(('+', '-'), self.parse_product)
584
585 def parse_product(self):
586 return self.parse_binary(('*', '/', 'MOD'), self.parse_unary)
587
588 def parse_unary(self):
589 if self.accept('-'):
590 return UnaryOp('-', require_integer(self.parse_unary()))
591 return self.parse_power()
592
593 def parse_power(self):
594 base = self.parse_primary()
595 if self.accept('^'):
596 return make_binary('^', base, self.parse_unary())
597 return base
598
599 def parse_primary(self):
600 token = self.advance()
601 match token:
602 case Token('number' | 'string', value):
603 return Constant(value)
604 case Token('variable', name):
605 return self.parse_reference(name)
606 case Token('keyword', 'ERR' | 'ERL'):
607 return ErrorValue(token.value)
608 case Token('keyword', '('):
609 expression = self.parse_expression()
610 self.expect(')')
611 return expression
612 case Token('keyword', name) if name in FUNCTIONS:
613 return self.parse_function_call(name)
614 raise BasicSyntaxError()
615
616 def parse_function_call(self, name):
617 self.expect('(')
618 arguments = self.parse_list(self.parse_expression)
619 self.expect(')')
620 signature = ''.join('S' if is_string(argument) else 'I' for argument in arguments)
621 if signature not in FUNCTIONS[name]:
622 raise BasicSyntaxError()
623 return FunctionCall(name, arguments)

626def check_overflow(value):
627 if not MIN_INTEGER <= value <= MAX_INTEGER:
628 raise BasicError('OVERFLOW')
629 return value

632def truncated_quotient(dividend, divisor):
633 quotient = abs(dividend) // abs(divisor)
634 return quotient if (dividend < 0) == (divisor < 0) else -quotient

637def apply_operator(symbol, left, right):
638 match symbol:
639 case '+' if isinstance(left, str):
640 if len(left) + len(right) > MAX_STRING_LENGTH:
641 raise BasicError('STRING TOO LONG')
642 return left + right
643 case '+':
644 return check_overflow(left + right)
645 case '-':
646 return check_overflow(left - right)
647 case '*':
648 return check_overflow(left * right)
649 case '/' | 'MOD' if right == 0:
650 raise BasicError('DIVISION BY ZERO')
651 case '/':
652 return check_overflow(truncated_quotient(left, right))
653 case 'MOD':
654 return left - right * truncated_quotient(left, right)
655 case '^':
656 if right < 0:
657 raise BasicError('ILLEGAL FUNCTION CALL')
658 return check_overflow(left ** right)
659 case 'AND':
660 return left & right
661 case 'OR':
662 return left | right
663 return -1 if COMPARISONS[symbol](left, right) else 0

674def link(statements, following):
675 for statement, successor in zip(statements, [*statements[1:], following]):
676 statement.next = successor
677 if isinstance(statement, If):
678 for clause in (statement.then_clause, statement.else_clause):
679 if isinstance(clause, list):
680 for inner in clause:
681 inner.line = statement.line
682 link(clause, successor)

685def find_matching(statement, opening, closing):
686 depth = 0
687 current = statement.next
688 while current is not None:
689 if isinstance(current, opening):
690 depth += 1
691 elif isinstance(current, closing):
692 if depth == 0:
693 return current
694 depth -= 1
695 current = current.next
696 return None

703class Interpreter:
704 def __init__(self, program, input_lines):
705 self.line_starts = {number: statements[0] for number, statements in program.items()}
706 self.data = []
707 sequence = []
708 for number, statements in program.items():
709 for statement in statements:
710 statement.line = number
711 if isinstance(statement, Data):
712 self.data.extend((number, value) for value in statement.values)
713 sequence.extend(statements)
714 link(sequence, None)
715 self.first_statement = sequence[0] if sequence else None
716 self.variables = {name: default_value(name) for name in VARIABLE_NAMES}
717 self.arrays = {}
718 self.stack = []
719 self.data_position = 0
720 self.input_lines = iter(input_lines)
721 self.random_state = 1
722 self.handler_line = 0
723 self.handling_error = False
724 self.error_code = 0
725 self.error_line = 0
726 self.error_statement = None
727 self.output = ''

729 def run(self):
730 statement = self.first_statement
731 while statement is not None:
732 try:
733 statement = self.execute(statement)
734 except BasicError as error:
735 if self.handler_line == 0 or self.handling_error:
736 self.message(f'ERROR IN LINE {statement.line}: {error}')
737 break
738 self.error_code = ERROR_CODES[str(error)]
739 self.error_line = statement.line
740 self.error_statement = statement
741 self.handling_error = True
742 statement = self.line_starts[self.handler_line]
743 if self.output:
744 self.end_line()

746 def end_line(self):
747 print(self.output.rstrip(' '))
748 self.output = ''

750 def message(self, text):
751 if self.output:
752 self.end_line()

768 def push(self, frame):
769 if len(self.stack) >= STACK_LIMIT:
770 raise BasicError('OUT OF MEMORY')
771 self.stack.append(frame)

773 def find_frame(self, kind, variable=None):
774 for index in reversed(range(len(self.stack))):
775 frame = self.stack[index]
776 if isinstance(frame, kind) and (variable is None or frame.variable == variable):
777 return index
778 if isinstance(frame, GosubFrame):
779 return None
780 return None

782 def pop_frame(self, kind, error):
783 index = self.find_frame(kind)
784 if index is None:
785 raise BasicError(error)
786 frame = self.stack[index]
787 del self.stack[index:]
788 return frame

790 def locate(self, target):
791 if isinstance(target, Variable):
792 return self.variables, target.name
793 subscripts = [self.evaluate(subscript) for subscript in target.subscripts]
794 if target.name not in self.arrays:
795 bounds = [DEFAULT_UPPER_BOUND] * len(subscripts)
796 self.arrays[target.name] = Array(target.name, bounds)
797 array = self.arrays[target.name]
798 return array.values, array.index(subscripts)

800 def evaluate(self, expression):
801 match expression:
802 case Constant(value):
803 return value
804 case Variable() | ArrayElement():
805 container, key = self.locate(expression)
806 return container[key]
807 case ErrorValue(name):
808 return self.error_code if name == 'ERR' else self.error_line
809 case FunctionCall(name, arguments):
810 return self.call(name, [self.evaluate(argument) for argument in arguments])
811 case UnaryOp('-', operand):
812 return check_overflow(-self.evaluate(operand))
813 case UnaryOp('NOT', operand):
814 return ~self.evaluate(operand)
815 case BinaryOp(symbol, left, right):
816 return apply_operator(symbol, self.evaluate(left), self.evaluate(right))

818 def call(self, name, arguments):
819 match name, arguments:
820 case 'ABS', [number]:
821 return check_overflow(abs(number))
822 case 'SGN', [number]:
823 return (number > 0) - (number < 0)
824 case 'INT', [number]:
825 return number
826 case 'SQR', [number]:
827 if number < 0:
828 raise BasicError('ILLEGAL FUNCTION CALL')
829 return math.isqrt(number)
830 case 'RND', [limit]:
831 if limit < 1:
832 raise BasicError('ILLEGAL FUNCTION CALL')
833 self.random_state = 48271 * self.random_state % 2147483647
834 return self.random_state % limit + 1
835 case 'LEN', [text]:
836 return len(text)
837 case 'ASC', [text]:
838 if not text:
839 raise BasicError('ILLEGAL FUNCTION CALL')
840 return ord(text[0])
841 case 'VAL', [text]:
842 return val(text)
843 case 'CHR$', [code]:
844 if not 32 <= code <= 126:
845 raise BasicError('ILLEGAL FUNCTION CALL')
846 return chr(code)
847 case 'STR$', [number]:
848 return str(number)
849 case 'LEFT$', [text, count]:
850 if count < 0:
851 raise BasicError('ILLEGAL FUNCTION CALL')
852 return text[:count]
853 case 'RIGHT$', [text, count]:
854 if count < 0:
855 raise BasicError('ILLEGAL FUNCTION CALL')
856 return text[max(len(text) - count, 0):]
857 case 'MID$', [text, start, *length]:
858 if start < 1 or length and length[0] < 0:
859 raise BasicError('ILLEGAL FUNCTION CALL')
860 part = text[start - 1:]
861 return part[:length[0]] if length else part
862 case 'INSTR', [text, pattern]:
863 return text.find(pattern) + 1

865 def execute(self, statement):
866 match statement:
867 case Let(target, expression):
868 container, key = self.locate(target)
869 container[key] = self.evaluate(expression)
870 case Print(items):
871 self.print_items(items)
872 case Input(targets):
873 for target in targets:
874 self.read_input(target)
875 case Read(targets):
876 for target in targets:
877 self.read_data(target)
878 case If(condition, then_clause, else_clause):
879 clause = then_clause if self.evaluate(condition) != 0 else else_clause
880 if isinstance(clause, int):
881 return self.go_to(clause)
882 if clause is not None:
883 return clause[0]
884 case Goto(target_line):
885 return self.go_to(target_line)
886 case Gosub(target_line):
887 return self.gosub(target_line, statement.next)
888 case Return():
889 return self.pop_frame(GosubFrame, 'RETURN WITHOUT GOSUB').return_position
890 case OnJump(selector, target_lines, is_gosub):
891 choice = self.evaluate(selector)
892 if choice < 0:
893 raise BasicError('ILLEGAL FUNCTION CALL')
894 if 1 <= choice <= len(target_lines):
895 target_line = target_lines[choice - 1]
896 if is_gosub:
897 return self.gosub(target_line, statement.next)
898 return self.go_to(target_line)
899 case OnError(target_line):
900 if target_line != 0:
901 self.check_line(target_line)
902 self.handler_line = target_line
903 case Resume(target):
904 return self.resume(target)
905 case For():
906 return self.start_loop(statement)
907 case Next():
908 return self.next_iteration(statement)
909 case While(condition):
910 if self.evaluate(condition) != 0:
911 self.push(WhileFrame(statement))
912 else:
913 wend = find_matching(statement, While, Wend)
914 if wend is None:
915 raise BasicError('WHILE WITHOUT WEND')
916 return wend.next
917 case Wend():
918 return self.pop_frame(WhileFrame, 'WEND WITHOUT WHILE').statement
919 case Dim(arrays):
920 for declaration in arrays:
921 self.declare(declaration)
922 case Restore(None):
923 self.data_position = 0
924 case Restore(target_line):
925 self.check_line(target_line)
926 self.data_position = next(
927 (index for index, (line, _) in enumerate(self.data) if line >= target_line),
928 len(self.data),
929 )
930 case Randomize(seed):
931 value = self.evaluate(seed)
932 if value < 1:
933 raise BasicError('ILLEGAL FUNCTION CALL')
934 self.random_state = value
935 case End():
936 return None
937 case Stop():
938 self.message(f'BREAK IN LINE {statement.line}')
939 return None
940 return statement.next

942 def print_items(self, items):
943 for item in items:
944 match item:
945 case ';':
946 pass
947 case ',':
948 self.output += ' ' * (PRINT_ZONE_WIDTH - len(self.output) % PRINT_ZONE_WIDTH)
949 case PrintFunction(name, argument):
950 width = self.evaluate(argument)
951 if width < 0:
952 raise BasicError('ILLEGAL FUNCTION CALL')
953 if name == 'TAB':
954 self.output = self.output.ljust(width)
955 else:
956 self.output += ' ' * width
957 case _:
958 self.output += str(self.evaluate(item))
959 if not items or items[-1] not in (';', ','):
960 self.end_line()

962 def read_input(self, target):
963 container, key = self.locate(target)
964 while True:
965 line = next(self.input_lines, None)
966 if line is None:
967 raise BasicError('INPUT PAST END')
968 if is_string(target):
969 container[key] = line
970 return
971 if INTEGER_INPUT_PATTERN.fullmatch(line) and MIN_INTEGER <= int(line) <= MAX_INTEGER:
972 container[key] = int(line)
973 return
974 self.message('?REDO FROM START')

976 def read_data(self, target):
977 container, key = self.locate(target)
978 if self.data_position == len(self.data):
979 raise BasicError('OUT OF DATA')
980 _, value = self.data[self.data_position]
981 self.data_position += 1
982 if isinstance(value, str) != is_string(target):
983 raise BasicError('TYPE MISMATCH')
984 container[key] = value

986 def resume(self, target):
987 if not self.handling_error:
988 raise BasicError('RESUME WITHOUT ERROR')
989 match target:
990 case None:
991 position = self.error_statement
992 case 'NEXT':
993 position = self.error_statement.next
994 case line:
995 position = self.go_to(line)
996 self.handling_error = False
997 return position

999 def start_loop(self, statement):
1000 start = self.evaluate(statement.start)
1001 limit = self.evaluate(statement.limit)
1002 step = self.evaluate(statement.step)
1003 if step == 0:
1004 raise BasicError('ILLEGAL FUNCTION CALL')
1005 index = self.find_frame(ForFrame, statement.variable)
1006 if index is not None:
1007 del self.stack[index:]
1008 self.variables[statement.variable] = start
1009 if loop_finished(step, start, limit):
1010 matching = find_matching(statement, For, Next)
1011 if matching is None:
1012 raise BasicError('FOR WITHOUT NEXT')
1013 return matching.next
1014 self.push(ForFrame(statement.variable, limit, step, statement.next))
1015 return statement.next
1016
1017 def next_iteration(self, statement):
1018 index = self.find_frame(ForFrame, statement.variable)
1019 if index is None:
1020 raise BasicError('NEXT WITHOUT FOR')
1021 del self.stack[index + 1:]
1022 frame = self.stack[index]
1023 value = check_overflow(self.variables[frame.variable] + frame.step)
1024 self.variables[frame.variable] = value
1025 if loop_finished(frame.step, value, frame.limit):
1026 self.stack.pop()
1027 return statement.next
1028 return frame.loop_start

1030 def declare(self, declaration):
1031 bounds = [self.evaluate(bound) for bound in declaration.subscripts]
1032 if declaration.name in self.arrays:
1033 raise BasicError('DUPLICATE DEFINITION')
1034 if any(bound < 0 for bound in bounds):
1035 raise BasicError('ILLEGAL FUNCTION CALL')
1036 if math.prod(bound + 1 for bound in bounds) > MAX_ARRAY_ELEMENTS:
1037 raise BasicError('OUT OF MEMORY')
1038 self.arrays[declaration.name] = Array(declaration.name, bounds)

1046 except BasicSyntaxError:

1048 break
1049 else:
