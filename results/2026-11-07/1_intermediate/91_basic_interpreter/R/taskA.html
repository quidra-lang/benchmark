71tokenize <- function(text) {
72 tokens <- list()
73 p <- 1L
74 n <- nchar(text,type="bytes")
75 take_word <- function(start,stop) substr(text,start,stop)
76 add <- function(kind,word,value) {
77 tokens[[length(tokens)+1L]] <<- list(kind=kind,text=word,value=value)
78 }
79 while (p<=n) {
80 ch <- substr(text,p,p)
81 if (ch==" ") {
82 p <- p+1L
83 } else if (ch=="\"") {
84 ending <- regexpr("\"",substring(text,p+1L),fixed=TRUE)[[1]]
85 if (ending<0L) syntax()
86 ending <- p+ending
87 add("S",take_word(p,ending),take_word(p+1L,ending-1L))
88 p <- ending+1L
89 } else if (grepl("^[0-9]$",ch)) {
90 start <- p
91 while (p<=n && grepl("^[0-9]$",substr(text,p,p))) p <- p+1L
92 word <- take_word(start,p-1L)
93 value <- suppressWarnings(as.numeric(word))
94 if (!is.finite(value) || value>32767) syntax()
95 add("I",word,value)
96 } else if (grepl("^[A-Z]$",ch)) {
97 if (substr(text,p,p+2L)=="REM") {
98 add("W","REM","REM")
99 break
100 }
101 start <- p
102 while (p<=n && grepl("^[A-Z0-9]$",substr(text,p,p))) p <- p+1L
103 if (p<=n && substr(text,p,p)=="$") p <- p+1L
104 word <- take_word(start,p-1L)
105 add("W",word,word)
106 } else {
107 if (!grepl(ch,"+-*/^=<>() ,;:",fixed=TRUE)) syntax()
108 word <- substr(text,p,p+1L)
109 if (!word %in% c("<>","<=",">=")) word <- ch
110 add("P",word,word)
111 p <- p+nchar(word,type="bytes")
112 }
113 }
114 add("E","","")
115 tokens
116}

118Parser <- function(tokens) {
119 p <- new.env(parent=emptyenv())
120 p$tokens <- tokens
121 p$position <- 1L
122 p$current <- function() p$tokens[[p$position]]
123 p$at <- function(word) {
124 t <- p$current()
125 t$kind %in% c("W","P") && t$text==word
126 }
127 p$take <- function(word) {
128 if (!p$at(word)) return(FALSE)
129 p$position <- p$position+1L
130 TRUE
131 }
132 p$expect <- function(word) if (!p$take(word)) syntax()
133 p$number <- function() {
134 t <- p$current()
135 if (t$kind!="I") syntax()
136 p$position <- p$position+1L
137 t$value
138 }
139 p$integer <- function(level=0L) {
140 e <- p$expression(level)
141 if (e$type!="I") syntax()
142 e
143 }
144 p$target <- function() {
145 t <- p$current()
146 if (t$kind!="W" || !variable_name(t$text)) syntax()
147 p$position <- p$position+1L
148 args <- list()
149 if (p$take("(")) {
150 args[[1]] <- p$integer()
151 if (p$take(",")) args[[2]] <- p$integer()
152 p$expect(")")
153 }
154 expr(if (length(args)==0) "variable" else "array",
155 name_type(t$text),t$text,args)
156 }
157 p$primary <- function() {
158 t <- p$current()
159 if (t$kind %in% c("I","S")) {
160 p$position <- p$position+1L
161 return(expr("constant",t$kind,t$value))
162 }
163 if (p$take("(")) {
164 e <- p$expression()
165 p$expect(")")
166 return(e)
167 }
168 if (t$kind=="W" && variable_name(t$text)) return(p$target())
169 if (p$take("ERR")) return(expr("builtin","I","ERR"))
170 if (p$take("ERL")) return(expr("builtin","I","ERL"))
171 if (t$kind=="W" && !is.null(signatures[[t$text]])) {
172 p$position <- p$position+1L
173 p$expect("(")
174 args <- list(p$expression())
175 while (p$take(",")) args[[length(args)+1L]] <- p$expression()
176 p$expect(")")
177 signature <- paste(vapply(args,function(e) e$type,""),collapse="")
178 if (!signature %in% signatures[[t$text]]) syntax()
179 return(expr("call",name_type(t$text),t$text,args))
180 }
181 syntax()
182 }
183 p$expression <- function(level=0L) {
184 if (level==2L && p$take("NOT"))
185 return(expr("unary","I","NOT",list(p$integer(2L))))
186 if (level==6L && p$take("-"))
187 return(expr("unary","I","-",list(p$integer(6L))))
188 if (level==8L) return(p$primary())
189 left <- p$expression(level+1L)
190 while (p$current()$text %in% operators[[level+1L]]) {
191 op <- p$current()$text
192 p$position <- p$position+1L
193 right <- p$expression(if (level==7L) 6L else level+1L)
194 if (left$type!=right$type ||
195 (left$type=="S" && level!=3L && op!="+")) syntax()
196 left <- expr("binary",if (level==3L) "I" else left$type,
197 op,list(left,right))
198 if (level==7L) break
199 }
200 left
201 }
202 p$clause <- function() {
203 if (p$current()$kind=="I") return(list(line=p$number(),body=list()))
204 list(line=NULL,body=p$listing(TRUE))

206 p$listing <- function(in_clause=FALSE) {
207 statements <- list(p$statement(in_clause))
208 while (p$take(":"))
209 statements[[length(statements)+1L]] <- p$statement(in_clause)
210 statements
211 }
212 p$statement <- function(in_clause) {
213 word <- p$current()$text
214 if (in_clause && word %in% c("IF","FOR","NEXT","WHILE","WEND","DATA"))
215 syntax()
216 s <- new_stmt(word)
217 for (kind in c("REM","RETURN","WEND","END","STOP"))
218 if (p$take(kind)) return(s)
219 if (p$take("PRINT")) {
220 expression_last <- FALSE
221 while (p$current()$kind!="E" && !p$at(":") && !p$at("ELSE")) {
222 if (p$take(";")) {
223 s$items[[length(s$items)+1L]] <- list(kind=";",node=NULL)
224 expression_last <- FALSE
225 } else if (p$take(",")) {
226 s$items[[length(s$items)+1L]] <- list(kind=",",node=NULL)
227 expression_last <- FALSE
228 } else {
229 kind <- if (p$at("TAB")) "TAB" else if (p$at("SPC")) "SPC" else "value"
230 if (kind=="value" && expression_last) syntax()
231 if (kind=="value") {
232 e <- p$expression()
233 } else {
234 p$position <- p$position+1L
235 p$expect("(")
236 e <- p$integer()
237 p$expect(")")
238 }
239 s$items[[length(s$items)+1L]] <- list(kind=kind,node=e)
240 expression_last <- kind=="value"
241 }
242 }
243 return(s)
244 }
245 if (p$take("INPUT") || p$take("READ") || p$take("DIM")) {
246 repeat {
247 target <- p$target()
248 if (word=="DIM" && target$kind!="array") syntax()
249 s$targets[[length(s$targets)+1L]] <- target
250 if (!p$take(",")) break
251 }
252 return(s)
253 }
254 if (p$take("DATA")) {
255 repeat {
256 negative <- p$take("-")
257 t <- p$current()
258 if (t$kind!="I" && (t$kind!="S" || negative)) syntax()
259 p$position <- p$position+1L
260 s$values[[length(s$values)+1L]] <- if (negative) -t$value else t$value
261 if (!p$take(",")) break
262 }
263 return(s)
264 }
265 if (p$take("IF")) {
266 s$expr <- p$integer()
267 p$expect("THEN")
268 s$clauses[[1]] <- p$clause()
269 if (p$take("ELSE")) s$clauses[[2]] <- p$clause()
270 if (p$current()$kind!="E") syntax()
271 return(s)
272 }
273 if (p$take("GOTO") || p$take("GOSUB")) {
274 s$numbers <- c(s$numbers,p$number())
275 return(s)
276 }
277 if (p$take("ON")) {
278 if (p$take("ERROR")) {
279 s$kind <- "handler"
280 p$expect("GOTO")
281 s$numbers <- c(s$numbers,p$number())
282 } else {
283 s$expr <- p$integer()
284 if (p$take("GOTO")) {
285 s$mode <- "GOTO"
286 } else {
287 p$expect("GOSUB")
288 s$mode <- "GOSUB"
289 }
290 s$numbers <- c(s$numbers,p$number())
291 while (p$take(",")) s$numbers <- c(s$numbers,p$number())
292 }
293 return(s)
294 }
295 if (p$take("RESUME")) {
296 if (p$take("NEXT")) {
297 s$mode <- "NEXT"
298 } else if (p$current()$kind=="I") {
299 s$mode <- "line"
300 s$numbers <- c(s$numbers,p$number())
301 }
302 return(s)
303 }
304 if (p$take("FOR")) {
305 t <- p$current()
306 if (t$kind!="W" || !variable_name(t$text) || name_type(t$text)!="I")
307 syntax()
308 s$name <- t$text
309 p$position <- p$position+1L
310 p$expect("=")
311 s$expressions[[1]] <- p$integer()
312 p$expect("TO")
313 s$expressions[[2]] <- p$integer()
314 s$expressions[[3]] <- if (p$take("STEP")) p$integer() else
315 expr("constant","I",1)
316 return(s)
317 }
318 if (p$take("NEXT")) {
319 t <- p$current()
320 if (t$kind=="W" && variable_name(t$text) && name_type(t$text)=="I") {
321 s$name <- t$text
322 p$position <- p$position+1L
323 }
324 return(s)
325 }
326 if (p$take("WHILE") || p$take("RANDOMIZE")) {
327 s$expr <- p$integer()
328 return(s)
329 }
330 if (p$take("RESTORE")) {
331 if (p$current()$kind=="I") s$numbers <- c(s$numbers,p$number())
332 return(s)
333 }
334 p$take("LET")
335 s$kind <- "assign"
336 s$target <- p$target()
337 p$expect("=")
338 s$expr <- p$expression()
339 if (s$expr$type!=s$target$type) syntax()
340 s
341 }
342 p$parse <- function() {
343 result <- p$listing()
344 if (p$current()$kind!="E") syntax()
345 result
346 }
347 p
348}

424 m$evaluate <- function(e) {
425 op <- e$value
426 if (e$kind=="constant") return(op)
427 if (e$kind %in% c("variable","array")) return(m$load(m$address(e)))
428 if (e$kind=="builtin") return(if (op=="ERR") m$err else m$erl)
429 if (e$kind=="unary") {
430 a <- m$evaluate(e$args[[1]])
431 if (op=="NOT") return(bitwNot(as.integer(a)))
432 return(m$checked(-a))
433 }
434 values <- lapply(e$args,m$evaluate)
435 if (e$kind=="call") return(m$call(op,values))
436 a <- values[[1]]
437 b <- values[[2]]
438 if (op %in% operators[[4]]) {
439 result <- switch(op,"="=a==b,"<>"=a!=b,
440 "<"=a<b,">"=a>b,"<="=a<=b,">="=a>=b)
441 return(if (result) -1 else 0)
442 }
443 if (e$type=="S") {
444 value <- paste0(a,b)
445 if (nchar(value,type="bytes")>255) fault("STRING TOO LONG")
446 return(value)
447 }
448 if (op=="+") return(m$checked(a+b))
449 if (op=="-") return(m$checked(a-b))
450 if (op=="*") return(m$checked(a*b))
451 if (op=="/" || op=="MOD") {
452 if (b==0) fault("DIVISION BY ZERO")
453 q <- if ((a<0)!=(b<0)) -floor(abs(a)/abs(b)) else floor(abs(a)/abs(b))
454 return(if (op=="/") m$checked(q) else a-b*q)
455 }
456 if (op=="^") {
457 if (b<0) fault("ILLEGAL FUNCTION CALL")
458 if (abs(a)>1 && b>15) fault("OVERFLOW")
459 return(m$checked(a^b))
460 }
461 if (op=="AND") return(bitwAnd(as.integer(a),as.integer(b)))
462 if (op=="OR") return(bitwOr(as.integer(a),as.integer(b)))
463 fault("ILLEGAL FUNCTION CALL")
464 }
465 m$call <- function(name,args) {
466 a <- args[[1]]
467 b <- if (length(args)>1) args[[2]] else NULL
468 c <- if (length(args)>2) args[[3]] else NULL
469 if (name=="ABS") return(m$checked(abs(a)))
470 if (name=="SGN") return(sign(a))
471 if (name=="INT") return(a)
472 if (name=="SQR") {
473 if (a<0) fault("ILLEGAL FUNCTION CALL")
474 return(floor(sqrt(a)))
475 }
476 if (name=="RND") {
477 if (a<1) fault("ILLEGAL FUNCTION CALL")
478 m$random <- (48271*m$random)%%2147483647
479 return(m$random%%a+1)
480 }
481 if (name=="LEN") return(nchar(a,type="bytes"))
482 if (name=="ASC") {
483 if (a=="") fault("ILLEGAL FUNCTION CALL")
484 return(as.integer(charToRaw(substr(a,1,1))))
485 }
486 if (name=="VAL") {
487 word <- regmatches(a,regexpr("^ *[+-]?[0-9]+",a))
488 return(m$checked(if (length(word) && nzchar(word)) as.numeric(word) else 0))
489 }
490 if (name=="CHR$") {
491 if (a<32 || a>126) fault("ILLEGAL FUNCTION CALL")
492 return(intToUtf8(a))
493 }
494 if (name=="STR$") return(as.character(a))
495 if (name=="LEFT$" || name=="RIGHT$") {
496 if (b<0) fault("ILLEGAL FUNCTION CALL")
497 if (name=="LEFT$") return(substr(a,1,b))
498 if (b==0) return("")
499 return(substr(a,max(1,nchar(a)-b+1),nchar(a)))
500 }
501 if (name=="MID$") {
502 if (b<1 || (!is.null(c) && c<0)) fault("ILLEGAL FUNCTION CALL")
503 if (!is.null(c)) return(substr(a,b,b+c-1))
504 return(substr(a,b,nchar(a)))
505 }
506 if (name=="INSTR") {
507 if (b=="") return(1)
508 at <- regexpr(b,a,fixed=TRUE)[[1]]
509 return(if (at<0) 0 else at)
510 }
511 fault("ILLEGAL FUNCTION CALL")
512 }

552 m$execute <- function(s) {
553 following <- s$following
554 if (s$kind=="assign") {
555 place <- m$address(s$target)
556 m$store(place,m$evaluate(s$expr))
557 } else if (s$kind=="PRINT") {
558 for (item in s$items) {
559 if (item$kind==",") {
560 m$buffer <- paste0(m$buffer,
561 strrep(" ",10-nchar(m$buffer)%%10))
562 } else if (item$kind!=";") {
563 value <- m$evaluate(item$node)
564 if (item$kind=="value") {
565 m$buffer <- paste0(m$buffer,as.character(value))
566 } else {
567 if (value<0) fault("ILLEGAL FUNCTION CALL")
568 amount <- if (item$kind=="SPC") value else
569 max(0,value-nchar(m$buffer,type="bytes"))
570 m$buffer <- paste0(m$buffer,strrep(" ",amount))
571 }
572 }
573 }
574 if (!length(s$items) ||
575 !(tail(s$items,1L)[[1]]$kind %in% c(";",","))) m$end_line()
576 } else if (s$kind %in% c("INPUT","READ")) {
577 for (target in s$targets) {
578 place <- m$address(target)
579 if (s$kind=="READ") {
580 if (m$data_position>length(m$data)) fault("OUT OF DATA")
581 value <- m$data[[m$data_position]]$value
582 m$data_position <- m$data_position+1L
583 if (value_type(value)!=target$type) fault("TYPE MISMATCH")
584 } else {
585 repeat {
586 if (m$input_position>length(m$inputs)) fault("INPUT PAST END")
587 text <- m$inputs[[m$input_position]]
588 m$input_position <- m$input_position+1L
589 if (target$type=="S") {value <- text;break}
590 valid <- grepl("^-?[0-9]+$",text)
591 n <- suppressWarnings(as.numeric(text))
592 if (valid && is.finite(n) && n>=-32768 && n<=32767) {
593 value <- n
594 break
595 }
596 m$message("?REDO FROM START")
597 }
598 }
599 m$store(place,value)
600 }
601 } else if (s$kind=="RESTORE") {
602 if (!length(s$numbers)) {
603 m$data_position <- 1L
604 } else {
605 number <- s$numbers[[1]]
606 m$line(number)
607 m$data_position <- length(m$data)+1L
608 for (i in seq_along(m$data)) {
609 if (m$data[[i]]$line>=number) {
610 m$data_position <- i
611 break
612 }
613 }
614 }
615 } else if (s$kind=="IF") {
616 index <- if (m$evaluate(s$expr)==0) 2L else 1L
617 if (index<=length(s$clauses)) {
618 chosen <- s$clauses[[index]]
619 following <- if (is.null(chosen$line)) chosen$body[[1]] else
620 m$line(chosen$line)
621 }
622 } else if (s$kind=="GOTO") {
623 following <- m$line(s$numbers[[1]])
624 } else if (s$kind=="GOSUB") {
625 following <- m$subroutine(s$numbers[[1]],following)
626 } else if (s$kind=="ON") {
627 n <- m$evaluate(s$expr)
628 if (n<0) fault("ILLEGAL FUNCTION CALL")
629 if (n>=1 && n<=length(s$numbers)) {
630 following <- if (s$mode=="GOTO") m$line(s$numbers[[n]]) else
631 m$subroutine(s$numbers[[n]],following)
632 }
633 } else if (s$kind=="RETURN") {
634 indices <- which(vapply(m$stack,function(f) f$kind=="GOSUB",FALSE))
635 if (!length(indices)) fault("RETURN WITHOUT GOSUB")
636 index <- max(indices)
637 following <- m$stack[[index]]$position
638 m$stack <- if (index>1) m$stack[seq_len(index-1L)] else list()
639 } else if (s$kind=="FOR") {
640 values <- vapply(s$expressions,m$evaluate,0.0)
641 start <- values[[1]]
642 final <- values[[2]]
643 step <- values[[3]]
644 if (step==0) fault("ILLEGAL FUNCTION CALL")
645 index <- m$search("FOR",s$name)
646 if (!is.null(index))
647 m$stack <- if (index>1) m$stack[seq_len(index-1L)] else list()
648 assign(s$name,start,envir=m$variables)
649 if ((step>0 && start>final) || (step<0 && start<final)) {
650 following <- m$matching(s,"FOR","NEXT")
651 } else {
652 m$push(list(kind="FOR",position=following,name=s$name,
653 final=final,step=step))
654 }
655 } else if (s$kind=="NEXT") {
656 index <- m$search("FOR",s$name)
657 if (is.null(index)) fault("NEXT WITHOUT FOR")
658 m$stack <- m$stack[seq_len(index)]
659 frame <- m$stack[[index]]
660 value <- m$checked(m$get(frame$name)+frame$step)
661 assign(frame$name,value,envir=m$variables)
662 if ((frame$step>0 && value>frame$final) ||
663 (frame$step<0 && value<frame$final)) {
664 m$stack <- if (index>1) m$stack[seq_len(index-1L)] else list()
665 } else {
666 following <- frame$position
667 }
668 } else if (s$kind=="WHILE") {
669 if (m$evaluate(s$expr)==0) {
670 following <- m$matching(s,"WHILE","WEND")
671 } else {
672 m$push(list(kind="WHILE",position=s,name=""))
673 }
674 } else if (s$kind=="WEND") {
675 index <- m$search("WHILE")
676 if (is.null(index)) fault("WEND WITHOUT WHILE")
677 following <- m$stack[[index]]$position
678 m$stack <- if (index>1) m$stack[seq_len(index-1L)] else list()
679 } else if (s$kind=="DIM") {
680 for (target in s$targets) {
681 name <- target$value
682 bounds <- vapply(target$args,m$evaluate,0.0)
683 if (exists(name,envir=m$arrays,inherits=FALSE))
684 fault("DUPLICATE DEFINITION")
685 if (any(bounds<0)) fault("ILLEGAL FUNCTION CALL")
686 if (prod(bounds+1)>10000) fault("OUT OF MEMORY")
687 m$declare(name,bounds)
688 }
689 } else if (s$kind=="RANDOMIZE") {
690 n <- m$evaluate(s$expr)
691 if (n<1) fault("ILLEGAL FUNCTION CALL")
692 m$random <- n
693 } else if (s$kind=="handler") {
694 if (s$numbers[[1]]!=0) m$line(s$numbers[[1]])
695 m$handler <- s$numbers[[1]]
696 } else if (s$kind=="RESUME") {
697 if (!m$handling) fault("RESUME WITHOUT ERROR")
698 following <- if (s$mode=="line") {
699 m$line(s$numbers[[1]])
700 } else if (s$mode=="NEXT") {
701 m$remembered$following
702 } else {
703 m$remembered
704 }
705 m$handling <- FALSE
706 } else if (s$kind=="STOP") {
707 m$message(paste0("BREAK IN LINE ",s$line))
708 return(NULL)
709 } else if (s$kind=="END") {
710 return(NULL)
711 }
712 following
713 }
714 m$connect <- function(body,after,number) {
715 following <- after
716 for (i in rev(seq_along(body))) {
717 s <- body[[i]]
718 s$line <- number
719 s$following <- following
720 for (clause in s$clauses) m$connect(clause$body,following,number)
721 following <- s
722 }
723 }
724 m$finish <- function() {
725 if (nchar(m$buffer,type="bytes")>0) m$end_line()
726 c(m$output,"READY")
727 }
728 m$run <- function(source) {
729 parsed <- list()
730 for (key in sort(as.numeric(names(source)))) {
731 number <- as.numeric(key)
732 code <- source[[as.character(key)]]
733 body <- tryCatch(Parser(tokenize(code))$parse(),error=function(e) e)
734 if (inherits(body,"error")) {
735 m$message(paste0("ERROR IN LINE ",number,": SYNTAX ERROR"))
736 return(m$finish())
737 }
738 parsed[[length(parsed)+1L]] <- list(number=number,body=body)
739 }
740 after <- NULL
741 for (entry in rev(parsed)) {
742 m$connect(entry$body,after,entry$number)
743 assign(as.character(entry$number),entry$body[[1]],envir=m$entries)
744 after <- entry$body[[1]]
745 }
746 for (entry in parsed) {
747 for (s in entry$body) {
748 m$sequence[[length(m$sequence)+1L]] <- s
749 if (s$kind=="DATA") {
750 for (value in s$values)
751 m$data[[length(m$data)+1L]] <- list(line=s$line,value=value)
752 }
753 }
754 }
755 position <- after
756 while (!is.null(position)) {
757 current <- position
758 result <- tryCatch(m$execute(current),error=function(e) e)
759 if (inherits(result,"error")) {
760 failure <- conditionMessage(result)
761 if (m$handler!=0 && !m$handling) {
762 m$err <- if (failure %in% names(error_codes))
763 unname(error_codes[[failure]]) else 0
764 m$erl <- current$line
765 m$remembered <- current
766 m$handling <- TRUE
767 position <- tryCatch(m$line(m$handler),error=function(e) {
768 m$message(paste0("ERROR IN LINE ",current$line,": ",
769 conditionMessage(e)))
770 NULL
771 })
772 } else {
773 m$message(paste0("ERROR IN LINE ",current$line,": ",failure))
774 position <- NULL
775 }
776 } else {
777 position <- result
778 }
779 }
780 m$finish()
781 }
782 m
783}
