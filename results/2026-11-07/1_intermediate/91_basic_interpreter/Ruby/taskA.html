90def tokenize(text)
91 tokens = []
92 p = 0
93 while p < text.length
94 ch = text[p]
95 if ch == " "
96 p += 1
97 elsif ch == "\""
98 ending = text.index("\"", p + 1)
99 syntax! unless ending
100 tokens << Token.new("S", text[p..ending], text[p + 1...ending])
101 p = ending + 1
102 elsif /[0-9]/.match?(ch)
103 start = p
104 p += 1 while p < text.length && /[0-9]/.match?(text[p])
105 word = text[start...p]
106 value = word.to_i
107 syntax! if value > 32767
108 tokens << Token.new("I", word, value)
109 elsif /[A-Z]/.match?(ch)
110 if text[p, 3] == "REM"
111 tokens << Token.new("W", "REM", "REM")
112 break
113 end
114
115 start = p
116 p += 1 while p < text.length && /[A-Z0-9]/.match?(text[p])
117 p += 1 if text[p] == "$"
118 word = text[start...p]
119 tokens << Token.new("W", word, word)
120 else
121 word = text[p, 2]
122 word = ch unless %w[<> <= >=].include?(word)
123 syntax! unless "+-*/^=<>() ,;:".include?(ch)
124 tokens << Token.new("P", word, word)
125 p += word.length
126 end
127 end
128
129 tokens << Token.new("E", "", "")
130end

132class Parser
133 def initialize(tokens)
134 @tokens = tokens
135 @position = 0
136 end
137
138 def current = @tokens[@position]
139 def at?(word) = %w[P W].include?(current.kind) && current.text == word
140 def take(word)
141 return false unless at?(word)
142 @position += 1
143 true
144 end
145
146 def expect(word)
147 syntax! unless take(word)
148 end
149
150 def integer
151 e = expression
152 syntax! unless e.type == "I"
153 e
154 end
155
156 def integer_at(level)
157 e = expression(level)
158 syntax! unless e.type == "I"
159 e
160 end
161
162 def number
163 token = current
164 syntax! unless token.kind == "I"
165 @position += 1
166 token.value
167 end
168
169 def target
170 token = current
171 syntax! unless token.kind == "W" && variable?(token.text)
172 @position += 1
173 args = []
174 if take("(")
175 args << integer
176 args << integer if take(",")
177 expect(")")
178 end
179
180 expr(args.empty? ? "variable" : "array", name_type(token.text), token.text, args)
181 end
182
183 def expression(level = 0)
184 return expr("unary", "I", "NOT", [integer_at(2)]) if level == 2 && take("NOT")
185 return expr("unary", "I", "-", [integer_at(6)]) if level == 6 && take("-")
186 return primary if level == 8
187 left = expression(level + 1)
188 while OPERATORS[level].any? { |op| at?(op) }
189 op = current.text
190 @position += 1
191 right = expression(level == 7 ? 6 : level + 1)
192 syntax! if left.type != right.type || (left.type == "S" && level != 3 && op != "+")
193 left = expr("binary", level == 3 ? "I" : left.type, op, [left, right])
194 break if level == 7
195 end
196
197 left
198 end
199
200 def primary
201 token = current
202 if %w[I S].include?(token.kind)
203 @position += 1
204 return expr("constant", token.kind, token.value)
205 end
206
207 if take("(")
208 e = expression
209 expect(")")
210 return e
211 end
212
213 return target if token.kind == "W" && variable?(token.text)
214 return expr("builtin", "I", "ERR") if take("ERR")
215 return expr("builtin", "I", "ERL") if take("ERL")
216 if token.kind == "W" && SIGNATURES.key?(token.text)
217 @position += 1
218 expect("(")
219 args = [expression]
220 args << expression while take(",")
221 expect(")")
222 syntax! unless SIGNATURES[token.text].include?(args.map(&:type).join)
223 return expr("call", name_type(token.text), token.text, args)
224 end
225
226 syntax!
227 end
228
229 def clause
230 current.kind == "I" ? Clause.new(number, []) : Clause.new(nil, list(true))
231 end
232
233 def list(in_clause = false)
234 statements = [statement(in_clause)]
235 statements << statement(in_clause) while take(":")
236 statements
237 end
238
239 def statement(in_clause)
240 word = current.text
241 syntax! if in_clause && %w[IF FOR NEXT WHILE WEND DATA].include?(word)
242 s = Statement.new(word)
243 if %w[REM RETURN WEND END STOP].any? { |name| take(name) }
244 return s
245 end
246
247 if take("PRINT")
248 expression_last = false
249 while current.kind != "E" && !at?(":") && !at?("ELSE")
250 if take(";")
251 s.items << PrintItem.new(";", nil)
252 expression_last = false
253 elsif take(",")
254 s.items << PrintItem.new(",", nil)
255 expression_last = false
256 else
257 kind = at?("TAB") ? "TAB" : at?("SPC") ? "SPC" : "value"
258 syntax! if kind == "value" && expression_last
259 if kind == "value"
260 e = expression
261 else
262 @position += 1
263 expect("(")
264 e = integer
265 expect(")")
266 end
267
268 s.items << PrintItem.new(kind, e)
269 expression_last = kind == "value"
270 end
271 end
272
273 return s
274 end
275
276 if take("INPUT") || take("READ") || take("DIM")
277 loop do
278 t = target
279 syntax! if word == "DIM" && t.kind != "array"
280 s.targets << t
281 break unless take(",")
282 end
283
284 return s
285 end
286
287 if take("DATA")
288 loop do
289 negative = take("-")
290 token = current
291 syntax! unless token.kind == "I" || (token.kind == "S" && !negative)
292 @position += 1
293 s.values << (negative ? -token.value : token.value)
294 break unless take(",")
295 end
296
297 return s
298 end
299
300 if take("IF")
301 s.expr = integer
302 expect("THEN")
303 s.clauses << clause
304 s.clauses << clause if take("ELSE")
305 syntax! unless current.kind == "E"
306 return s
307 end
308
309 if take("GOTO") || take("GOSUB")
310 s.numbers << number
311 return s
312 end
313
314 if take("ON")
315 if take("ERROR")
316 s.kind = "handler"
317 expect("GOTO")
318 s.numbers << number
319 else
320 s.expr = integer
321 if take("GOTO")
322 s.mode = "GOTO"
323 else
324 expect("GOSUB")
325 s.mode = "GOSUB"
326 end
327
328 s.numbers << number
329 s.numbers << number while take(",")
330 end
331
332 return s
333 end
334
335 if take("RESUME")
336 if take("NEXT")
337 s.mode = "NEXT"
338 elsif current.kind == "I"
339 s.mode = "line"
340 s.numbers << number
341 end
342
343 return s
344 end
345
346 if take("FOR")
347 token = current
348 syntax! unless token.kind == "W" && variable?(token.text) && name_type(token.text) == "I"
349 s.name = token.text
350 @position += 1
351 expect("=")
352 s.expressions << integer
353 expect("TO")
354 s.expressions << integer
355 s.expressions << (take("STEP") ? integer : expr("constant", "I", 1))
356 return s
357 end
358
359 if take("NEXT")
360 if current.kind == "W" && variable?(current.text) && name_type(current.text) == "I"
361 s.name = current.text
362 @position += 1
363 end
364
365 return s
366 end
367
368 if take("WHILE") || take("RANDOMIZE")
369 s.expr = integer
370 return s
371 end
372
373 if take("RESTORE")
374 s.numbers << number if current.kind == "I"
375 return s
376 end
377
378 take("LET")
379 s.kind = "assign"
380 s.target = target
381 expect("=")
382 s.expr = expression
383 syntax! unless s.expr.type == s.target.type
384 s
385 end
386
387 def parse
388 result = list
389 syntax! unless current.kind == "E"
390 result
391 end
392end

394class Machine
395 def initialize(inputs)
396 @inputs = inputs
397 @variables = {}
398 @arrays = {}
399 @stack = []
400 @entries = {}
401 @sequence = []
402 @data = []
403 @data_position = 0
404 @input_position = 0
405 @random = 1
406 @handler = 0
407 @handling = false
408 @remembered = nil
409 @err = 0
410 @erl = 0
411 @buffer = ""
412 @output = []
413 end

415 def end_line
416 @output << @buffer.sub(/ +\z/, "")
417 @buffer = ""

421 end_line unless @buffer.empty?
422 @output << text

425 def checked(n)
426 fault!("OVERFLOW") unless n.between?(-32768, 32767)
427 n
428 end

430 def get(name) = @variables.fetch(name) { name_type(name) == "S" ? "" : 0 }
431 def declare(name, bounds)
432 @arrays[name] = BasicArray.new(
433 bounds,
434 Array.new(bounds.map { |b| b + 1 }.inject(1, :*), name_type(name) == "S" ? "" : 0)
435 )
436 end

438 def address(target)
439 name = target.value
440 return Address.new(name, nil) if target.kind == "variable"
441 indices = target.args.map { |e| evaluate(e) }
442 declare(name, Array.new(indices.length, 10)) unless @arrays.key?(name)
443 array = @arrays[name]
444 if indices.length != array.bounds.length || indices.each_with_index.any? { |n, i| n < 0 || n > array.bounds[i] }
445 fault!("SUBSCRIPT OUT OF RANGE")
446 end
447
448 index = 0
449 indices.each_with_index { |n, i| index = index * (array.bounds[i] + 1) + n }
450 Address.new(name, index)
451 end
452
453 def load(address) = address.index.nil? ? get(address.name) : @arrays[address.name].values[address.index]
454 def store(address, value)
455 if address.index.nil?
456 @variables[address.name] = value
457 else
458 @arrays[address.name].values[address.index] = value
459 end
460 end
461
462 def quotient(a, b) = (a < 0) != (b < 0) ? -(a.abs / b.abs) : a.abs / b.abs
463 def evaluate(e)
464 op = e.value
465 return op if e.kind == "constant"
466 return load(address(e)) if %w[variable array].include?(e.kind)
467 return op == "ERR" ? @err : @erl if e.kind == "builtin"
468 if e.kind == "unary"
469 a = evaluate(e.args[0])
470 return op == "NOT" ? ~a : checked(-a)
471 end
472
473 args = e.args.map { |a| evaluate(a) }
474 return call(op, args) if e.kind == "call"
475 a, b = args
476 if %w[= <> < > <= >=].include?(op)
477 order = a <=> b
478 return {
479 "=" => order.zero?,
480 "<>" => !order.zero?,
481 "<" => order < 0,
482 ">" => order > 0,
483 "<=" => order <= 0,
484 ">=" => order >= 0
485 }[op] ? -1 : 0
486 end
487
488 if e.type == "S"
489 text = a + b
490 fault!("STRING TOO LONG") if text.length > 255
491 return text
492 end
493
494 case op
495 when "+"
496 checked(a + b)
497 when "-"
498 checked(a - b)
499 when "*"
500 checked(a * b)
501 when "/"
502 fault!("DIVISION BY ZERO") if b.zero?
503 checked(quotient(a, b))
504 when "MOD"
505 fault!("DIVISION BY ZERO") if b.zero?
506 a - b * quotient(a, b)
507 when "^"
508 fault!("ILLEGAL FUNCTION CALL") if b < 0
509 fault!("OVERFLOW") if a.abs > 1 && b > 15
510 checked(a ** b)
511 when "AND"
512 a & b
513 when "OR"
514 a | b
515 end
516 end
517
518 def call(name, args)
519 a, b, c = args
520 case name
521 when "ABS"
522 checked(a.abs)
523 when "SGN"
524 a <=> 0
525 when "INT"
526 a
527 when "SQR"
528 fault!("ILLEGAL FUNCTION CALL") if a < 0
529 Integer.sqrt(a)
530 when "RND"
531 fault!("ILLEGAL FUNCTION CALL") if a < 1
532 @random = 48271 * @random % 2147483647
533 @random % a + 1
534 when "LEN"
535 a.length
536 when "ASC"
537 fault!("ILLEGAL FUNCTION CALL") if a.empty?
538 a.ord
539 when "VAL"
540 match = /\A *([+-]?[0-9]+)/.match(a)
541 checked(match ? match[1].to_i : 0)
542 when "CHR$"
543 fault!("ILLEGAL FUNCTION CALL") unless a.between?(32, 126)
544 a.chr
545 when "STR$"
546 a.to_s
547 when "LEFT$"
548 fault!("ILLEGAL FUNCTION CALL") if b < 0
549 a[0, b]
550 when "RIGHT$"
551 fault!("ILLEGAL FUNCTION CALL") if b < 0
552 count = [b, a.length].min
553 a[-count, count]
554 when "MID$"
555 fault!("ILLEGAL FUNCTION CALL") if b < 1 || (args.length == 3 && c < 0)
556 text = a[b - 1..] || ""
557 args.length == 3 ? text[0, c] : text
558 when "INSTR"
559 (a.index(b) || -1) + 1
560 end
561 end
562
563 def line(number) = @entries[number] || fault!("UNDEFINED LINE NUMBER")
564 def push(frame)
565 fault!("OUT OF MEMORY") if @stack.length == 100
566 @stack << frame
567 end
568
569 def search(kind, name = "")
570 (@stack.length - 1).downto(0) do |i|
571 f = @stack[i]
572 break if f.kind == "GOSUB"
573 return i if f.kind == kind && (name.empty? || f.name == name)
574 end
575
576 -1
577 end
578
579 def matching(s, opening, closing)
580 depth = 0
581 @sequence.drop(@sequence.index(s) + 1).each do |t|
582 if t.kind == opening
583 depth += 1
584 elsif t.kind == closing
585 return t.next if depth.zero?
586 depth -= 1
587 end
588 end
589
590 fault!(opening + " WITHOUT " + closing)
591 end
592
593 def subroutine(number, after)
594 target = line(number)
595 push(Frame.new("GOSUB", after, nil, nil, nil))
596 target
597 end
598
599 def execute(s)
600 following = s.next
601 case s.kind
602 when "assign"
603 place = address(s.target)
604 store(place, evaluate(s.expr))
605 when "PRINT"
606 s.items.each do |item|
607 if item.kind == ","
608 @buffer += " " * (10 - @buffer.length % 10)
609 elsif item.kind != ";"
610 value = evaluate(item.expr)
611 if item.kind == "value"
612 @buffer += value.to_s
613 else
614 fault!("ILLEGAL FUNCTION CALL") if value < 0
615 @buffer += " " * (item.kind == "SPC" ? value : [0, value - @buffer.length].max)
616 end
617 end
618 end
619
620 end_line if s.items.empty? || ![";", ","].include?(s.items.last.kind)
621 when "INPUT", "READ"
622 s.targets.each do |target|
623 place = address(target)
624 if s.kind == "READ"
625 fault!("OUT OF DATA") if @data_position == @data.length
626 value = @data[@data_position][1]
627 @data_position += 1
628 fault!("TYPE MISMATCH") unless value_type(value) == target.type
629 else
630 loop do
631 fault!("INPUT PAST END") if @input_position == @inputs.length
632 text = @inputs[@input_position]
633 @input_position += 1
634 if target.type == "S"
635 value = text
636 break
637 end
638
639 n = text.to_i
640 if /\A-?[0-9]+\z/.match?(text) && n.between?(-32768, 32767)
641 value = n
642 break
643 end
644
645 message("?REDO FROM START")
646 end
647 end
648
649 store(place, value)
650 end
651
652 when "RESTORE"
653 if s.numbers.empty?
654 @data_position = 0
655 else
656 line(s.numbers[0])
657 @data_position = @data.index { |number, _| number >= s.numbers[0] } || @data.length
658 end
659
660 when "IF"
661 chosen = s.clauses[evaluate(s.expr).zero? ? 1 : 0]
662 following = chosen.line ? line(chosen.line) : chosen.body[0] if chosen
663 when "GOTO"
664 following = line(s.numbers[0])
665 when "GOSUB"
666 following = subroutine(s.numbers[0], following)
667 when "ON"
668 n = evaluate(s.expr)
669 fault!("ILLEGAL FUNCTION CALL") if n < 0
670 if n.between?(1, s.numbers.length)
671 following = s.mode == "GOTO" ? line(s.numbers[n - 1]) : subroutine(s.numbers[n - 1], following)
672 end
673
674 when "RETURN"
675 index = @stack.rindex { |f| f.kind == "GOSUB" }
676 fault!("RETURN WITHOUT GOSUB") unless index
677 following = @stack[index].position
678 @stack = @stack.first(index)
679 when "FOR"
680 start, final, step = s.expressions.map { |e| evaluate(e) }
681 fault!("ILLEGAL FUNCTION CALL") if step.zero?
682 index = search("FOR", s.name)
683 @stack = @stack.first(index) if index >= 0
684 @variables[s.name] = start
685 if step > 0 ? start > final : start < final
686 following = matching(s, "FOR", "NEXT")
687 else
688 push(Frame.new("FOR", following, s.name, final, step))
689 end
690
691 when "NEXT"
692 index = search("FOR", s.name)
693 fault!("NEXT WITHOUT FOR") if index < 0
694 @stack = @stack.first(index + 1)
695 f = @stack[index]
696 value = checked(get(f.name) + f.step)
697 @variables[f.name] = value
698 if f.step > 0 ? value > f.final : value < f.final
699 @stack.pop
700 else
701 following = f.position
702 end
703
704 when "WHILE"
705 if evaluate(s.expr).zero?
706 following = matching(s, "WHILE", "WEND")
707 else
708 push(Frame.new("WHILE", s, nil, nil, nil))
709 end
710
711 when "WEND"
712 index = search("WHILE")
713 fault!("WEND WITHOUT WHILE") if index < 0
714 following = @stack[index].position
715 @stack = @stack.first(index)
716 when "DIM"
717 s.targets.each do |target|
718 name = target.value
719 bounds = target.args.map { |e| evaluate(e) }
720 fault!("DUPLICATE DEFINITION") if @arrays.key?(name)
721 fault!("ILLEGAL FUNCTION CALL") if bounds.any?(&:negative?)
722 fault!("OUT OF MEMORY") if bounds.map { |n| n + 1 }.inject(1, :*) > 10000
723 declare(name, bounds)
724 end
725
726 when "RANDOMIZE"
727 n = evaluate(s.expr)
728 fault!("ILLEGAL FUNCTION CALL") if n < 1
729 @random = n
730 when "handler"
731 line(s.numbers[0]) unless s.numbers[0].zero?
732 @handler = s.numbers[0]
733 when "RESUME"
734 fault!("RESUME WITHOUT ERROR") unless @handling
735 following = s.mode == "line" ? line(s.numbers[0]) : s.mode == "NEXT" ? @remembered.next : @remembered
736 @handling = false
737 when "STOP"
738 message("BREAK IN LINE #{s.line}")
739 return nil
740 when "END"
741 return nil
742 end
743
744 following
745 end

747 def connect(body, after, number)
748 following = after
749 body.reverse_each do |s|
750 s.line = number
751 s.next = following
752 s.clauses.each { |c| connect(c.body, following, number) }
753 following = s
754 end
755 end

757 def run(source)
758 parsed = []
759 source.keys.sort.each do |number|
760 begin
761 parsed << [number, Parser.new(tokenize(source[number])).parse]
762 rescue SyntaxFault
763 message("ERROR IN LINE #{number}: SYNTAX ERROR")
764 return finish
765 end
766 end
767
768 after = nil
769 parsed.reverse_each do |number, body|
770 connect(body, after, number)
771 @entries[number] = body[0]
772 after = body[0]
773 end
774
775 @sequence = parsed.flat_map(&:last)
776 @sequence.each { |s| s.values.each { |v| @data << [s.line, v] } if s.kind == "DATA" }
777 position = after
778 while position
779 statement = position
780 begin
781 position = execute(statement)
782 rescue RuntimeFault => failure
783 if @handler != 0 && !@handling
784 @err = ERROR_CODES[failure.message]
785 @erl = statement.line
786 @remembered = statement
787 @handling = true
788 position = line(@handler)
789 else
790 message("ERROR IN LINE #{statement.line}: #{failure.message}")
791 position = nil
792 end
793 end
794 end
795
796 finish
797 end

799 def finish
800 end_line unless @buffer.empty?
801 @output << "READY"
802 @output
803 end
804end
