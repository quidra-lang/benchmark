88function tokenize(text: string): Token[] {
89 const tokens: Token[] = [];
90 for (let p = 0; p < text.length;) {
91 const ch = text[p];
92 if (ch === " ") {
93 p++;
94 continue;
95 }
96 if (ch === '"') {
97 const end = text.indexOf('"', p + 1);
98 if (end < 0) syntax();
99 tokens.push({ kind: "S", text: text.slice(p, end + 1), value: text.slice(p + 1, end) });
100 p = end + 1;
101 } else if (/[0-9]/.test(ch)) {
102 const start = p;
103 while (p < text.length && /[0-9]/.test(text[p])) p++;
104 const word = text.slice(start, p),
105 value = Number(word);
106 if (value > 32767) syntax();
107 tokens.push({ kind: "I", text: word, value });
108 } else if (/[A-Z]/.test(ch)) {
109 if (text.startsWith("REM", p)) {
110 tokens.push({ kind: "W", text: "REM", value: "REM" });
111 break;
112 }
113 const start = p;
114 while (p < text.length && /[A-Z0-9]/.test(text[p])) p++;
115 if (text[p] === "$") p++;
116 const word = text.slice(start, p);
117 tokens.push({ kind: "W", text: word, value: word });
118 } else {
119 let word = text.slice(p, p + 2);
120 if (!["<>", "<=", ">="].includes(word)) word = ch;
121 if (!"+-*/^=<>() ,;:".includes(ch)) syntax();
122 tokens.push({ kind: "P", text: word, value: word });
123 p += word.length;
124 }
125 }
126 tokens.push({ kind: "E", text: "", value: "" });
127 return tokens;
128}

130class Parser {
131 position = 0;
132 constructor(public tokens: Token[]) {}
133 current(): Token {
134 return this.tokens[this.position];
135 }
136 at(word: string): boolean {
137 const t = this.current();
138 return (t.kind === "P" || t.kind === "W") && t.text === word;
139 }
140 take(word: string): boolean {
141 if (!this.at(word)) return false;
142 this.position++;
143 return true;
144 }
145 expect(word: string): void {
146 if (!this.take(word)) syntax();
147 }
148 integer(): Expr {
149 const expr = this.expression();
150 if (expr.type !== "I") syntax();
151 return expr;
152 }
153 number(): number {
154 const t = this.current();
155 if (t.kind !== "I") syntax();
156 this.position++;
157 return t.value as number;
158 }
159 target(): Expr {
160 const token = this.current();
161 if (token.kind !== "W" || !variable(token.text)) syntax();
162 this.position++;
163 const result = new Expr("variable", nameType(token.text), token.text);
164 if (this.take("(")) {
165 result.kind = "array";
166 result.args.push(this.integer());
167 if (this.take(",")) result.args.push(this.integer());
168 this.expect(")");
169 }
170 return result;
171 }
172 expression(level = 0): Expr {
173 if (level === 2 && this.take("NOT"))
174 return new Expr("unary", "I", "NOT", [this.integerAt(2)]);
175 if (level === 6 && this.take("-")) return new Expr("unary", "I", "-", [this.integerAt(6)]);
176 if (level === 8) return this.primary();
177 let left = this.expression(level + 1);
178 const ops = [
179 ["OR"],
180 ["AND"],
181 [],
182 ["=", "<>", "<", ">", "<=", ">="],
183 ["+", "-"],
184 ["*", "/", "MOD"],
185 [],
186 ["^"],
187 ][level];
188 while (ops.some((op) => this.at(op))) {
189 const op = this.current().text;
190 this.position++;
191 const right = this.expression(level === 7 ? 6 : level + 1);
192 if (left.type !== right.type || (left.type === "S" && level !== 3 && op !== "+"))
193 syntax();
194 left = new Expr("binary", level === 3 ? "I" : left.type, op, [left, right]);
195 if (level === 7) break;
196 }
197 return left;
198 }
199 integerAt(level: number): Expr {
200 const e = this.expression(level);
201 if (e.type !== "I") syntax();
202 return e;
203 }
204 primary(): Expr {
205 const t = this.current();
206 if (t.kind === "I" || t.kind === "S") {
207 this.position++;
208 return new Expr("constant", t.kind as BasicType, t.value);
209 }
210 if (this.take("(")) {
211 const e = this.expression();
212 this.expect(")");
213 return e;
214 }
215 if (t.kind === "W" && variable(t.text)) return this.target();
216 if (this.take("ERR")) return new Expr("builtin", "I", "ERR");
217 if (this.take("ERL")) return new Expr("builtin", "I", "ERL");
218 if (t.kind === "W" && signatures.has(t.text)) {
219 this.position++;
220 this.expect("(");
221 const args = [this.expression()];
222 while (this.take(",")) args.push(this.expression());
223 this.expect(")");
224 if (!signatures.get(t.text)!.includes(args.map((e) => e.type).join(""))) syntax();
225 return new Expr("call", nameType(t.text), t.text, args);
226 }
227 return syntax();
228 }
229 clause(): Clause {
230 if (this.current().kind === "I") return { line: this.number(), body: [] };
231 return { line: null, body: this.list(true) };
232 }
233 list(clause = false): Statement[] {
234 const body = [this.statement(clause)];
235 while (this.take(":")) body.push(this.statement(clause));
236 return body;
237 }
238 statement(inClause: boolean): Statement {
239 const word = this.current().text;
240 if (inClause && ["IF", "FOR", "NEXT", "WHILE", "WEND", "DATA"].includes(word)) syntax();
241 const s = new Statement(word);
242 if (
243 this.take("REM") ||
244 this.take("RETURN") ||
245 this.take("WEND") ||
246 this.take("END") ||
247 this.take("STOP")
248 )
249 return s;
250 if (this.take("PRINT")) {
251 let expressionLast = false;
252 while (this.current().kind !== "E" && !this.at(":") && !this.at("ELSE")) {
253 if (this.take(";")) {
254 s.items.push({ kind: ";", expr: null });
255 expressionLast = false;
256 } else if (this.take(",")) {
257 s.items.push({ kind: ",", expr: null });
258 expressionLast = false;
259 } else {
260 const kind = this.at("TAB") ? "TAB" : this.at("SPC") ? "SPC" : "value";
261 if (kind === "value" && expressionLast) syntax();
262 let expr: Expr;
263 if (kind !== "value") {
264 this.position++;
265 this.expect("(");
266 expr = this.integer();
267 this.expect(")");
268 } else expr = this.expression();
269 s.items.push({ kind, expr });
270 expressionLast = kind === "value";
271 }
272 }
273 return s;
274 }
275 if (this.take("INPUT") || this.take("READ") || this.take("DIM")) {
276 do {
277 const target = this.target();
278 if (word === "DIM" && target.kind !== "array") syntax();
279 s.targets.push(target);
280 } while (this.take(","));
281 return s;
282 }
283 if (this.take("DATA")) {
284 do {
285 const negative = this.take("-"),
286 t = this.current();
287 if (t.kind !== "I" && (t.kind !== "S" || negative)) syntax();
288 this.position++;
289 s.values.push(negative ? -(t.value as number) : t.value);
290 } while (this.take(","));
291 return s;
292 }
293 if (this.take("IF")) {
294 s.expr = this.integer();
295 this.expect("THEN");
296 s.clauses.push(this.clause());
297 if (this.take("ELSE")) s.clauses.push(this.clause());
298 if (this.current().kind !== "E") syntax();
299 return s;
300 }
301 if (this.take("GOTO") || this.take("GOSUB")) {
302 s.numbers.push(this.number());
303 return s;
304 }
305 if (this.take("ON")) {
306 if (this.take("ERROR")) {
307 s.kind = "handler";
308 this.expect("GOTO");
309 s.numbers.push(this.number());
310 } else {
311 s.expr = this.integer();
312 if (this.take("GOTO")) s.mode = "GOTO";
313 else {
314 this.expect("GOSUB");
315 s.mode = "GOSUB";
316 }
317 do {
318 s.numbers.push(this.number());
319 } while (this.take(","));
320 }
321 return s;
322 }
323 if (this.take("RESUME")) {
324 if (this.take("NEXT")) s.mode = "NEXT";
325 else if (this.current().kind === "I") {
326 s.mode = "line";
327 s.numbers.push(this.number());
328 }
329 return s;
330 }
331 if (this.take("FOR")) {
332 const t = this.current();
333 if (t.kind !== "W" || !variable(t.text) || nameType(t.text) !== "I") syntax();
334 s.name = t.text;
335 this.position++;
336 this.expect("=");
337 s.expressions.push(this.integer());
338 this.expect("TO");
339 s.expressions.push(this.integer());
340 s.expressions.push(this.take("STEP") ? this.integer() : new Expr("constant", "I", 1));
341 return s;
342 }
343 if (this.take("NEXT")) {
344 if (
345 this.current().kind === "W" &&
346 variable(this.current().text) &&
347 nameType(this.current().text) === "I"
348 )
349 s.name = this.tokens[this.position++].text;
350 return s;
351 }
352 if (this.take("WHILE") || this.take("RANDOMIZE")) {
353 s.expr = this.integer();
354 return s;
355 }
356 if (this.take("RESTORE")) {
357 if (this.current().kind === "I") s.numbers.push(this.number());
358 return s;
359 }
360 this.take("LET");
361 s.kind = "assign";
362 s.target = this.target();
363 this.expect("=");
364 s.expr = this.expression();
365 if (s.expr.type !== s.target.type) syntax();
366 return s;
367 }
368 parse(): Statement[] {
369 const body = this.list();
370 if (this.current().kind !== "E") syntax();
371 return body;
372 }
373}

390class Machine {
391 variables = new Map<string, Value>();
392 arrays = new Map<string, BasicArray>();
393 stack: Frame[] = [];
394 entries = new Map<number, Statement>();
395 sequence: Statement[] = [];
396 data: { line: number; value: Value }[] = [];
397 dataPosition = 0;
398 inputPosition = 0;
399 random = 1;
400 handler = 0;
401 handling = false;
402 remembered: Statement | null = null;
403 err = 0;
404 erl = 0;
405 buffer = "";
406 output: string[] = [];
407 constructor(public inputs: string[]) {}
408 endLine(): void {
409 this.output.push(this.buffer.replace(/ +$/, ""));
410 this.buffer = "";
411 }
412 message(text: string): void {
413 if (this.buffer !== "") this.endLine();
414 this.output.push(text);
415 }
416 checked(n: number): number {
417 if (!Number.isFinite(n) || n < -32768 || n > 32767) fault("OVERFLOW");
418 return n;
419 }
420 get(name: string): Value {
421 return this.variables.get(name) ?? (nameType(name) === "S" ? "" : 0);
422 }
423 declare(name: string, bounds: number[]): void {
424 const count = bounds.reduce((a, b) => a * (b + 1), 1);
425 this.arrays.set(name, {
426 bounds,
427 values: Array(count).fill(nameType(name) === "S" ? "" : 0),
428 });
429 }
430 address(target: Expr): Address {
431 const name = target.value as string;
432 if (target.kind === "variable") return { name, index: null };
433 const indices = target.args.map((e) => this.evaluate(e) as number);
434 if (!this.arrays.has(name))
435 this.declare(
436 name,
437 indices.map(() => 10),
438 );
439 const array = this.arrays.get(name)!;
440 if (
441 indices.length !== array.bounds.length ||
442 indices.some((n, i) => n < 0 || n > array.bounds[i])
443 )
444 fault("SUBSCRIPT OUT OF RANGE");
445 let index = 0;
446 indices.forEach((n, i) => {
447 index = index * (array.bounds[i] + 1) + n;
448 });
449 return { name, index };
450 }
451 load(address: Address): Value {
452 return address.index === null
453 ? this.get(address.name)
454 : this.arrays.get(address.name)!.values[address.index];
455 }
456 store(address: Address, value: Value): void {
457 if (address.index === null) this.variables.set(address.name, value);
458 else this.arrays.get(address.name)!.values[address.index] = value;
459 }
460 evaluate(e: Expr): Value {
461 const op = e.value;
462 if (e.kind === "constant") return op;
463 if (e.kind === "variable" || e.kind === "array") return this.load(this.address(e));
464 if (e.kind === "builtin") return op === "ERR" ? this.err : this.erl;
465 if (e.kind === "unary") {
466 const a = this.evaluate(e.args[0]) as number;
467 return op === "NOT" ? ~a : this.checked(-a);
468 }
469 const args = e.args.map((a) => this.evaluate(a));
470 if (e.kind === "call") return this.call(op as string, args);
471 const [a, b] = args;
472 if (["=", "<>", "<", ">", "<=", ">="].includes(op as string)) {
473 const order = a === b ? 0 : a < b ? -1 : 1;
474 return (
475 {
476 "=": order === 0,
477 "<>": order !== 0,
478 "<": order < 0,
479 ">": order > 0,
480 "<=": order <= 0,
481 ">=": order >= 0,
482 } as Record<string, boolean>
483 )[op]
484 ? -1
485 : 0;
486 }
487 if (e.type === "S") {
488 const text = (a as string) + (b as string);
489 if (text.length > 255) fault("STRING TOO LONG");
490 return text;
491 }
492 const x = a as number,
493 y = b as number;
494 switch (op) {
495 case "+":
496 return this.checked(x + y);
497 case "-":
498 return this.checked(x - y);
499 case "*":
500 return this.checked(x * y);
501 case "/":
502 if (y === 0) fault("DIVISION BY ZERO");
503 return this.checked(Math.trunc(x / y));
504 case "MOD":
505 if (y === 0) fault("DIVISION BY ZERO");
506 return x % y;
507 case "^":
508 if (y < 0) fault("ILLEGAL FUNCTION CALL");
509 return this.checked(x ** y);
510 case "AND":
511 return x & y;
512 case "OR":
513 return x | y;
514 }
515 throw new Error("invalid expression");
516 }
517 call(name: string, args: Value[]): Value {
518 const n = args[0] as number,
519 s = args[0] as string,
520 k = args[1] as number;
521 switch (name) {
522 case "ABS":
523 return this.checked(Math.abs(n));
524 case "SGN":
525 return Math.sign(n);
526 case "INT":
527 return n;
528 case "SQR":
529 if (n < 0) fault("ILLEGAL FUNCTION CALL");
530 return Math.floor(Math.sqrt(n));
531 case "RND":
532 if (n < 1) fault("ILLEGAL FUNCTION CALL");
533 this.random = (48271 * this.random) % 2147483647;
534 return (this.random % n) + 1;
535 case "LEN":
536 return s.length;
537 case "ASC":
538 if (s === "") fault("ILLEGAL FUNCTION CALL");
539 return s.charCodeAt(0);
540 case "VAL": {
541 const m = /^ *([+-]?[0-9]+)/.exec(s);
542 return this.checked(m ? Number(m[1]) : 0);
543 }
544 case "CHR$":
545 if (n < 32 || n > 126) fault("ILLEGAL FUNCTION CALL");
546 return String.fromCharCode(n);
547 case "STR$":
548 return String(n);
549 case "LEFT$":
550 if (k < 0) fault("ILLEGAL FUNCTION CALL");
551 return s.slice(0, k);
552 case "RIGHT$":
553 if (k < 0) fault("ILLEGAL FUNCTION CALL");
554 return k === 0 ? "" : s.slice(-k);
555 case "MID$": {
556 if (k < 1 || (args.length === 3 && (args[2] as number) < 0))
557 fault("ILLEGAL FUNCTION CALL");
558 return s.slice(k - 1, args.length === 3 ? k - 1 + (args[2] as number) : undefined);
559 }
560 case "INSTR":
561 return s.indexOf(args[1] as string) + 1;
562 }
563 throw new Error("invalid function");
564 }
565 line(number: number): Statement {
566 const s = this.entries.get(number);
567 if (!s) fault("UNDEFINED LINE NUMBER");
568 return s;
569 }
570 push(frame: Frame): void {
571 if (this.stack.length === 100) fault("OUT OF MEMORY");
572 this.stack.push(frame);
573 }
574 search(kind: string, name = ""): number {
575 for (let i = this.stack.length - 1; i >= 0; i--) {
576 const f = this.stack[i];
577 if (f.kind === "GOSUB") break;
578 if (f.kind === kind && (!name || f.name === name)) return i;
579 }
580 return -1;
581 }
582 matching(s: Statement, open: string, close: string): Statement | null {
583 let depth = 0;
584 for (let i = this.sequence.indexOf(s) + 1; i < this.sequence.length; i++) {
585 const t = this.sequence[i];
586 if (t.kind === open) depth++;
587 else if (t.kind === close) {
588 if (depth === 0) return t.next;
589 depth--;
590 }
591 }
592 return fault(open + " WITHOUT " + close);
593 }
594 subroutine(number: number, after: Statement | null): Statement {
595 const target = this.line(number);
596 this.push({ kind: "GOSUB", position: after });
597 return target;
598 }
599 execute(s: Statement): Statement | null {
600 let next = s.next;
601 switch (s.kind) {
602 case "REM":
603 case "DATA":
604 break;
605 case "assign": {
606 const address = this.address(s.target!);
607 this.store(address, this.evaluate(s.expr!));
608 break;
609 }
610 case "PRINT":
611 for (const item of s.items) {
612 if (item.kind === ",")
613 this.buffer += " ".repeat(10 - (this.buffer.length % 10));
614 else if (item.kind !== ";") {
615 const value = this.evaluate(item.expr!);
616 if (item.kind === "value") this.buffer += String(value);
617 else {
618 const n = value as number;
619 if (n < 0) fault("ILLEGAL FUNCTION CALL");
620 this.buffer += " ".repeat(
621 item.kind === "SPC" ? n : Math.max(0, n - this.buffer.length),
622 );
623 }
624 }
625 }
626 if (s.items.length === 0 || ![";", ","].includes(s.items[s.items.length - 1].kind))
627 this.endLine();
628 break;
629 case "INPUT":
630 case "READ":
631 for (const target of s.targets) {
632 const address = this.address(target);
633 let value: Value;
634 if (s.kind === "READ") {
635 if (this.dataPosition === this.data.length) fault("OUT OF DATA");
636 value = this.data[this.dataPosition++].value;
637 if (valueType(value) !== target.type) fault("TYPE MISMATCH");
638 } else {
639 while (true) {
640 if (this.inputPosition === this.inputs.length) fault("INPUT PAST END");
641 const text = this.inputs[this.inputPosition++];
642 if (target.type === "S") {
643 value = text;
644 break;
645 }
646 const n = Number(text);
647 if (/^-?[0-9]+$/.test(text) && n >= -32768 && n <= 32767) {
648 value = n;
649 break;
650 }
651 this.message("?REDO FROM START");
652 }
653 }
654 this.store(address, value);
655 }
656 break;
657 case "RESTORE":
658 if (s.numbers.length) {
659 this.line(s.numbers[0]);
660 const p = this.data.findIndex((d) => d.line >= s.numbers[0]);
661 this.dataPosition = p < 0 ? this.data.length : p;
662 } else this.dataPosition = 0;
663 break;
664 case "IF": {
665 const clause = s.clauses[(this.evaluate(s.expr!) as number) !== 0 ? 0 : 1];
666 if (clause) next = clause.line !== null ? this.line(clause.line) : clause.body[0];
667 break;
668 }
669 case "GOTO":
670 next = this.line(s.numbers[0]);
671 break;
672 case "GOSUB":
673 next = this.subroutine(s.numbers[0], next);
674 break;
675 case "ON": {
676 const n = this.evaluate(s.expr!) as number;
677 if (n < 0) fault("ILLEGAL FUNCTION CALL");
678 if (n >= 1 && n <= s.numbers.length)
679 next =
680 s.mode === "GOTO"
681 ? this.line(s.numbers[n - 1])
682 : this.subroutine(s.numbers[n - 1], next);
683 break;
684 }
685 case "RETURN": {
686 let i = this.stack.length - 1;
687 while (i >= 0 && this.stack[i].kind !== "GOSUB") i--;
688 if (i < 0) fault("RETURN WITHOUT GOSUB");
689 next = this.stack[i].position;
690 this.stack.length = i;
691 break;
692 }
693 case "FOR": {
694 const [start, final, step] = s.expressions.map((e) => this.evaluate(e) as number);
695 if (step === 0) fault("ILLEGAL FUNCTION CALL");
696 const i = this.search("FOR", s.name);
697 if (i >= 0) this.stack.length = i;
698 this.variables.set(s.name, start);
699 if (step > 0 ? start > final : start < final)
700 next = this.matching(s, "FOR", "NEXT");
701 else this.push({ kind: "FOR", name: s.name, final, step, position: next });
702 break;
703 }
704 case "NEXT": {
705 const i = this.search("FOR", s.name);
706 if (i < 0) fault("NEXT WITHOUT FOR");
707 this.stack.length = i + 1;
708 const f = this.stack[i],
709 step = f.step!,
710 final = f.final!;
711 const value = this.checked((this.get(f.name!) as number) + step);
712 this.variables.set(f.name!, value);
713 if (step > 0 ? value > final : value < final) this.stack.pop();
714 else next = f.position;
715 break;
716 }
717 case "WHILE":
718 if ((this.evaluate(s.expr!) as number) !== 0)
719 this.push({ kind: "WHILE", position: s });
720 else next = this.matching(s, "WHILE", "WEND");
721 break;
722 case "WEND": {
723 const i = this.search("WHILE");
724 if (i < 0) fault("WEND WITHOUT WHILE");
725 next = this.stack[i].position;
726 this.stack.length = i;
727 break;
728 }
729 case "DIM":
730 for (const target of s.targets) {
731 const name = target.value as string,
732 bounds = target.args.map((e) => this.evaluate(e) as number);
733 if (this.arrays.has(name)) fault("DUPLICATE DEFINITION");
734 if (bounds.some((n) => n < 0)) fault("ILLEGAL FUNCTION CALL");
735 if (bounds.reduce((a, b) => a * (b + 1), 1) > 10000) fault("OUT OF MEMORY");
736 this.declare(name, bounds);
737 }
738 break;
739 case "RANDOMIZE": {
740 const n = this.evaluate(s.expr!) as number;
741 if (n < 1) fault("ILLEGAL FUNCTION CALL");
742 this.random = n;
743 break;
744 }
745 case "handler":
746 if (s.numbers[0] !== 0) this.line(s.numbers[0]);
747 this.handler = s.numbers[0];
748 break;
749 case "RESUME":
750 if (!this.handling) fault("RESUME WITHOUT ERROR");
751 next =
752 s.mode === "line"
753 ? this.line(s.numbers[0])
754 : s.mode === "NEXT"
755 ? this.remembered!.next
756 : this.remembered;
757 this.handling = false;
758 break;
759 case "STOP":
760 this.message("BREAK IN LINE " + s.line);
761 return null;
762 case "END":
763 return null;
764 }
765 return next;
766 }
767 connect(body: Statement[], after: Statement | null, line: number): void {
768 let next = after;
769 for (let i = body.length - 1; i >= 0; i--) {
770 const s = body[i];
771 s.line = line;
772 s.next = next;
773 for (const c of s.clauses) this.connect(c.body, next, line);
774 next = s;
775 }
776 }
777 run(source: Map<number, string>): string[] {
778 const parsed: [number, Statement[]][] = [];
779 for (const line of [...source.keys()].sort((a, b) => a - b)) {
780 try {
781 parsed.push([line, new Parser(tokenize(source.get(line)!)).parse()]);
782 } catch (e) {
783 if (!(e instanceof SyntaxFault)) throw e;
784 this.message(`ERROR IN LINE ${line}: SYNTAX ERROR`);
785 return this.finish();
786 }
787 }
788 let after: Statement | null = null;
789 for (let i = parsed.length - 1; i >= 0; i--) {
790 const [line, body] = parsed[i];
791 this.connect(body, after, line);
792 this.entries.set(line, body[0]);
793 after = body[0];
794 }
795 this.sequence = parsed.flatMap(([, body]) => body);
796 for (const s of this.sequence)
797 if (s.kind === "DATA")
798 for (const value of s.values) this.data.push({ line: s.line, value });
799 let position = after;
800 while (position !== null) {
801 const statement = position;
802 try {
803 position = this.execute(statement);
804 } catch (e) {
805 if (!(e instanceof RuntimeFault)) throw e;
806 if (this.handler !== 0 && !this.handling) {
807 this.err = errorCodes.get(e.message)!;
808 this.erl = statement.line;
809 this.remembered = statement;
810 this.handling = true;
811 position = this.line(this.handler);
812 } else {
813 this.message(`ERROR IN LINE ${statement.line}: ${e.message}`);
814 position = null;
815 }
816 }
817 }
818 return this.finish();
819 }
820 finish(): string[] {
821 if (this.buffer !== "") this.endLine();
822 this.output.push("READY");
823 return this.output;
824 }
825}
