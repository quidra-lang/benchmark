134void *allocate(Machine *m, size_t size) {
135 void *memory = calloc(1, size ? size : 1);
136 if (!memory) {
137 fputs("out of memory\n", stderr);
138 exit(1);
139 }
140 if (m->allocation_count == m->allocation_capacity) {
141 m->allocation_capacity = m->allocation_capacity ? m->allocation_capacity * 2 : 64;
142 m->allocations = realloc(m->allocations,
143 (size_t)m->allocation_capacity * sizeof(void *));
144 if (!m->allocations) exit(1);
145 }
146 m->allocations[m->allocation_count++] = memory;
147 return memory;
148}

192int is_variable(const char *name) {
193 size_t size = strlen(name);
194 if (size < 1 || size > 3 || name[0] < 'A' || name[0] > 'Z') return 0;
195 if (size == 1) return 1;
196 if (size == 2) return (name[1] >= '0' && name[1] <= '9') || name[1] == '$';
197 return name[1] >= '0' && name[1] <= '9' && name[2] == '$';
198}
199
200int variable_index(const char *name) {
201 int column = name[0] - 'A';
202 int digit = name[1] >= '0' && name[1] <= '9' ? name[1] - '0' + 1 : 0;
203 int string = name[strlen(name) - 1] == '$';
204 return (column * 11 + digit) * 2 + string;
205}

229int find_function(const char *name) {
230 const char *names[] = {
231 "ABS", "SGN", "INT", "SQR", "RND", "LEN", "ASC", "VAL",
232 "CHR$", "STR$", "LEFT$", "RIGHT$", "MID$", "INSTR"
233 };
234 for (int i = 0; i < 14; ++i)
235 if (strcmp(name, names[i]) == 0) return i;
236 return -1;
237}

239int signature(int function, const char *types) {
240 if (function <= 4 || function == 8 || function == 9)
241 return strcmp(types, "I") == 0;
242 if (function == 5 || function == 6 || function == 7)
243 return strcmp(types, "S") == 0;
244 if (function == 10 || function == 11)
245 return strcmp(types, "SI") == 0;
246 if (function == 12)
247 return strcmp(types, "SI") == 0 || strcmp(types, "SII") == 0;
248 return function == 13 && strcmp(types, "SS") == 0;
249}

259void tokenize(Machine *m, const char *source) {
260 int pos = 0;
261 m->token_count = 0;
262 m->position = 0;
263 while (source[pos]) {
264 char c = source[pos];
265 if (c == ' ') {
266 ++pos;
267 continue;
268 }
269 Token token = {0};
270 int start = pos;
271 if (c == '"') {
272 ++pos;
273 while (source[pos] && source[pos] != '"') ++pos;
274 if (!source[pos]) syntax(m);
275 token.kind = 'S';
276 int width = pos - start + 1;
277 memcpy(token.text, source + start, (size_t)width);
278 char temp[256] = {0};
279 memcpy(temp, source + start + 1, (size_t)(width - 2));
280 token.value = text_value(copy(m, temp));
281 ++pos;
282 } else if (token_digit(c)) {
283 while (token_digit(source[pos])) ++pos;
284 token.kind = 'I';
285 memcpy(token.text, source + start, (size_t)(pos - start));
286 int64_t n = 0;
287 for (int i = start; i < pos; ++i) {
288 n = n * 10 + source[i] - '0';
289 if (n > 32767) syntax(m);
290 }
291 token.value = integer_value(n);
292 } else if (token_upper(c)) {
293 if (strncmp(source + pos, "REM", 3) == 0) {
294 strcpy(token.text, "REM");
295 token.kind = 'W';
296 token.value = text_value(copy(m, "REM"));
297 m->tokens[m->token_count++] = token;
298 break;
299 }
300 while (token_upper(source[pos]) || token_digit(source[pos])) ++pos;
301 if (source[pos] == '$') ++pos;
302 token.kind = 'W';
303 memcpy(token.text, source + start, (size_t)(pos - start));
304 token.value = text_value(copy(m, token.text));
305 } else {
306 if (!strchr("+-*/^=<>() ,;:", c)) syntax(m);
307 token.kind = 'P';
308 token.text[0] = c;
309 ++pos;
310 if ((c == '<' || c == '>') && source[pos] &&
311 ((c == '<' && source[pos] == '>') || source[pos] == '=')) {
312 token.text[1] = source[pos++];
313 }
314 token.value = text_value(copy(m, token.text));
315 }
316 m->tokens[m->token_count++] = token;
317 if (m->token_count >= 255) syntax(m);
318 }
319 Token end = {0};
320 end.kind = 'E';
321 m->tokens[m->token_count++] = end;
322}

373int expression_node(Machine *m, const char *kind, char type, const char *name,
374 IntVec args) {
375 Expression result = {0};
376 result.kind = strcmp(kind, "constant") == 0 ? 0 :
377 strcmp(kind, "variable") == 0 ? 1 :
378 strcmp(kind, "array") == 0 ? 2 :
379 strcmp(kind, "builtin") == 0 ? 3 :
380 strcmp(kind, "unary") == 0 ? 4 :
381 strcmp(kind, "call") == 0 ? 5 : 6;
382 result.type = type;
383 strncpy(result.name, name, sizeof(result.name) - 1);
384 result.args = args;
385 return add_expression(m, result);
386}

417int target(Machine *m) {
418 Token token = *peek(m);
419 if (token.kind != 'W' || !is_variable(token.text)) syntax(m);
420 ++m->position;
421 IntVec args = {0};
422 int array = take(m, "(");
423 if (array) {
424 int_push(&args, integer(m, 0));
425 if (take(m, ",")) int_push(&args, integer(m, 0));
426 expect(m, ")");
427 }
428 return expression_node(m, array ? "array" : "variable",
429 name_type(token.text), token.text, args);
430}

432int primary(Machine *m) {
433 Token token = *peek(m);
434 if (token.kind == 'I' || token.kind == 'S') {
435 ++m->position;
436 return constant(m, token.value);
437 }
438 if (take(m, "(")) {
439 int result = parse_expression(m, 0);
440 expect(m, ")");
441 return result;
442 }
443 if (token.kind == 'W' && is_variable(token.text)) return target(m);
444 if (take(m, "ERR")) return expression_node(m, "builtin", 'I', "ERR", (IntVec){0});
445 if (take(m, "ERL")) return expression_node(m, "builtin", 'I', "ERL", (IntVec){0});
446 int function = token.kind == 'W' ? find_function(token.text) : -1;
447 if (function >= 0) {
448 ++m->position;
449 expect(m, "(");
450 IntVec args = {0};
451 int_push(&args, parse_expression(m, 0));
452 while (take(m, ",")) int_push(&args, parse_expression(m, 0));
453 expect(m, ")");
454 char types[32] = {0};
455 if (args.count >= 31) syntax(m);
456 for (int i = 0; i < args.count; ++i)
457 types[i] = m->expressions[args.items[i]].type;
458 if (!signature(function, types)) syntax(m);
459 return expression_node(m, "call", name_type(token.text), token.text, args);
460 }
461 syntax(m);
462 return -1;
463}

465int is_operator(int level, const char *op) {
466 if (level == 0) return strcmp(op, "OR") == 0;
467 if (level == 1) return strcmp(op, "AND") == 0;
468 if (level == 3) return strcmp(op, "=") == 0 || strcmp(op, "<>") == 0 ||
469 strcmp(op, "<") == 0 || strcmp(op, ">") == 0 ||
470 strcmp(op, "<=") == 0 || strcmp(op, ">=") == 0;
471 if (level == 4) return strcmp(op, "+") == 0 || strcmp(op, "-") == 0;
472 if (level == 5) return strcmp(op, "*") == 0 || strcmp(op, "/") == 0 ||
473 strcmp(op, "MOD") == 0;
474 return level == 7 && strcmp(op, "^") == 0;
475}
476
477int parse_expression(Machine *m, int level) {
478 if (level == 2 && take(m, "NOT")) return unary(m, "NOT", integer(m, 2));
479 if (level == 6 && take(m, "-")) return unary(m, "-", integer(m, 6));
480 if (level == 8) return primary(m);
481 int left = parse_expression(m, level + 1);
482 while (is_operator(level, peek(m)->text)) {
483 char op[16];
484 strcpy(op, peek(m)->text);
485 ++m->position;
486 int right = parse_expression(m, level == 7 ? 6 : level + 1);
487 char type = m->expressions[left].type;
488 if (type != m->expressions[right].type ||
489 (type == 'S' && level != 3 && strcmp(op, "+") != 0)) syntax(m);
490 left = binary(m, level == 3 ? 'I' : type, op, left, right);
491 if (level == 7) break;
492 }
493 return left;
494}

517int parse_statement(Machine *m, int in_clause) {
518 char word[256];
519 strcpy(word, peek(m)->text);
520 if (in_clause && (strcmp(word, "IF") == 0 || strcmp(word, "FOR") == 0 ||
521 strcmp(word, "NEXT") == 0 || strcmp(word, "WHILE") == 0 ||
522 strcmp(word, "WEND") == 0 || strcmp(word, "DATA") == 0)) syntax(m);
523 Statement s = new_statement(word);
524 if (take(m, "REM") || take(m, "RETURN") || take(m, "WEND") ||
525 take(m, "END") || take(m, "STOP")) return add_statement(m, s);
526 if (take(m, "PRINT")) {
527 int last = 0;
528 while (peek(m)->kind != 'E' && !at(m, ":") && !at(m, "ELSE")) {
529 if (take(m, ";")) {
530 print_push(&s.items, (PrintItem){.kind = 1, .expr = -1});
531 last = 0;
532 } else if (take(m, ",")) {
533 print_push(&s.items, (PrintItem){.kind = 2, .expr = -1});
534 last = 0;
535 } else {
536 int mode = at(m, "TAB") ? 3 : at(m, "SPC") ? 4 : 0;
537 if (!mode && last) syntax(m);
538 int expr;
539 if (!mode) expr = parse_expression(m, 0);
540 else {
541 ++m->position;
542 expect(m, "(");
543 expr = integer(m, 0);
544 expect(m, ")");
545 }
546 print_push(&s.items, (PrintItem){.kind = mode, .expr = expr});
547 last = !mode;
548 }
549 }
550 return add_statement(m, s);
551 }
552 if (take(m, "INPUT") || take(m, "READ") || take(m, "DIM")) {
553 do {
554 int t = target(m);
555 if (strcmp(word, "DIM") == 0 && m->expressions[t].kind != 2) syntax(m);
556 int_push(&s.targets, t);
557 } while (take(m, ","));
558 return add_statement(m, s);
559 }
560 if (take(m, "DATA")) {
561 do {
562 int negative = take(m, "-");
563 Token token = *peek(m);
564 if (token.kind != 'I' && (token.kind != 'S' || negative)) syntax(m);
565 ++m->position;
566 value_push(&s.values, negative ?
567 integer_value(-token.value.number) : token.value);
568 } while (take(m, ","));
569 return add_statement(m, s);
570 }
571 if (take(m, "IF")) {
572 s.expr = integer(m, 0);
573 expect(m, "THEN");
574 s.clauses[s.clause_count++] = parse_clause(m);
575 if (take(m, "ELSE")) s.clauses[s.clause_count++] = parse_clause(m);
576 if (peek(m)->kind != 'E') syntax(m);
577 return add_statement(m, s);
578 }
579 if (take(m, "GOTO") || take(m, "GOSUB")) {
580 int_push(&s.numbers, number_token(m));
581 return add_statement(m, s);
582 }
583 if (take(m, "ON")) {
584 if (take(m, "ERROR")) {
585 strcpy(s.kind, "handler");
586 expect(m, "GOTO");
587 int_push(&s.numbers, number_token(m));
588 } else {
589 s.expr = integer(m, 0);
590 if (take(m, "GOTO")) strcpy(s.mode, "GOTO");
591 else {
592 expect(m, "GOSUB");
593 strcpy(s.mode, "GOSUB");
594 }
595 do {
596 int_push(&s.numbers, number_token(m));
597 } while (take(m, ","));
598 }
599 return add_statement(m, s);
600 }
601 if (take(m, "RESUME")) {
602 if (take(m, "NEXT")) strcpy(s.mode, "NEXT");
603 else if (peek(m)->kind == 'I') {
604 strcpy(s.mode, "line");
605 int_push(&s.numbers, number_token(m));
606 }
607 return add_statement(m, s);
608 }
609 if (take(m, "FOR")) {
610 Token token = *peek(m);
611 if (token.kind != 'W' || !is_variable(token.text) ||
612 name_type(token.text) != 'I') syntax(m);
613 strcpy(s.name, token.text);
614 ++m->position;
615 expect(m, "=");
616 int_push(&s.expressions, integer(m, 0));
617 expect(m, "TO");
618 int_push(&s.expressions, integer(m, 0));
619 int_push(&s.expressions, take(m, "STEP") ?
620 integer(m, 0) : constant(m, integer_value(1)));
621 return add_statement(m, s);
622 }
623 if (take(m, "NEXT")) {
624 if (peek(m)->kind == 'W' && is_variable(peek(m)->text) &&
625 name_type(peek(m)->text) == 'I') {
626 strcpy(s.name, peek(m)->text);
627 ++m->position;
628 }
629 return add_statement(m, s);
630 }
631 if (take(m, "WHILE") || take(m, "RANDOMIZE")) {
632 s.expr = integer(m, 0);
633 return add_statement(m, s);
634 }
635 if (take(m, "RESTORE")) {
636 if (peek(m)->kind == 'I') int_push(&s.numbers, number_token(m));
637 return add_statement(m, s);
638 }
639 take(m, "LET");
640 strcpy(s.kind, "assign");
641 s.target = target(m);
642 expect(m, "=");
643 s.expr = parse_expression(m, 0);
644 if (m->expressions[s.target].type != m->expressions[s.expr].type) syntax(m);
645 return add_statement(m, s);
646}

648IntVec parse_list(Machine *m, int in_clause) {
649 IntVec body = {0};
650 int_push(&body, parse_statement(m, in_clause));
651 while (take(m, ":")) int_push(&body, parse_statement(m, in_clause));
652 return body;
653}

655IntVec parse(Machine *m, const char *text) {
656 tokenize(m, text);
657 IntVec body = parse_list(m, 0);
658 if (peek(m)->kind != 'E') syntax(m);
659 return body;
660}

662void connect(Machine *m, IntVec body, int after, int line) {
663 int next = after;
664 for (int i = body.count - 1; i >= 0; --i) {
665 int id = body.items[i];
666 Statement *s = &m->statements[id];
667 s->line = line;
668 s->next = next;
669 for (int j = 0; j < s->clause_count; ++j)
670 connect(m, s->clauses[j].body, next, line);
671 next = id;
672 }
673}

676 while (m->buffer_length > 0 && m->buffer[m->buffer_length - 1] == ' ')
677 --m->buffer_length;
678 m->buffer[m->buffer_length] = 0;
679 puts(m->buffer);
680 m->buffer[0] = 0;
681 m->buffer_length = 0;
682}

689void append(Machine *m, const char *text) {
690 size_t length = strlen(text);
691 if (m->buffer_length + length >= sizeof(m->buffer)) exit(1);
692 memcpy(m->buffer + m->buffer_length, text, length);
693 m->buffer_length += (int)length;
694 m->buffer[m->buffer_length] = 0;
695}

706void declare(Machine *m, const char *name, const int *bounds, int dimensions) {
707 int index = variable_index(name);
708 BasicArray *array = allocate(m, sizeof(BasicArray));
709 array->dims = dimensions;
710 array->count = 1;
711 for (int i = 0; i < dimensions; ++i) {
712 array->bounds[i] = bounds[i];
713 array->count *= bounds[i] + 1;
714 }
715 array->values = allocate(m, (size_t)array->count * sizeof(Value));
716 Value initial = zero_value(m, name);
717 for (int i = 0; i < array->count; ++i) array->values[i] = initial;
718 m->arrays[index] = array;
719}

723Address address(Machine *m, int expr_id) {
724 Expression expr = m->expressions[expr_id];
725 Address loc = {0};
726 strcpy(loc.name, expr.name);
727 loc.index = -1;
728 if (expr.kind == 1) return loc;
729 int bounds[2] = {0};
730 for (int i = 0; i < expr.args.count; ++i)
731 bounds[i] = (int)evaluate(m, expr.args.items[i]).number;
732 int slot = variable_index(loc.name);
733 if (!m->arrays[slot]) {
734 int defaults[2] = {10, 10};
735 declare(m, loc.name, defaults, expr.args.count);
736 }
737 BasicArray *array = m->arrays[slot];
738 if (expr.args.count != array->dims) fail(m, "SUBSCRIPT OUT OF RANGE");
739 int offset = 0;
740 for (int i = 0; i < expr.args.count; ++i) {
741 if (bounds[i] < 0 || bounds[i] > array->bounds[i])
742 fail(m, "SUBSCRIPT OUT OF RANGE");
743 offset = offset * (array->bounds[i] + 1) + bounds[i];
744 }
745 loc.index = offset;
746 return loc;
747}

764int64_t integer_power(Machine *m, int64_t base, int64_t exponent) {
765 if (exponent < 0) fail(m, "ILLEGAL FUNCTION CALL");
766 int64_t result = 1;
767 while (exponent > 0) {
768 if (exponent % 2) result = checked(m, result * base);
769 exponent /= 2;
770 if (exponent) base = checked(m, base * base);
771 }
772 return result;
773}

775int compare(Value a, Value b) {
776 if (a.type == 'S') {
777 int order = strcmp(a.text, b.text);
778 return (order > 0) - (order < 0);
779 }
780 return (a.number > b.number) - (a.number < b.number);
781}

783Value call(Machine *m, const char *name, const Value *args, int count) {
784 int64_t n = args[0].number;
785 const char *s = args[0].type == 'S' ? args[0].text : "";
786 int64_t k = count > 1 ? args[1].number : 0;
787 if (strcmp(name, "ABS") == 0)
788 return integer_value(checked(m, n < 0 ? -n : n));
789 if (strcmp(name, "SGN") == 0)
790 return integer_value((n > 0) - (n < 0));
791 if (strcmp(name, "INT") == 0) return integer_value(n);
792 if (strcmp(name, "SQR") == 0) {
793 if (n < 0) fail(m, "ILLEGAL FUNCTION CALL");
794 int64_t root = 0;
795 while ((root + 1) * (root + 1) <= n) ++root;
796 return integer_value(root);
797 }
798 if (strcmp(name, "RND") == 0) {
799 if (n < 1) fail(m, "ILLEGAL FUNCTION CALL");
800 m->random_state = m->random_state * 48271 % 2147483647;
801 return integer_value(m->random_state % n + 1);
802 }
803 if (strcmp(name, "LEN") == 0) return integer_value((int64_t)strlen(s));
804 if (strcmp(name, "ASC") == 0) {
805 if (!*s) fail(m, "ILLEGAL FUNCTION CALL");
806 return integer_value((unsigned char)s[0]);
807 }
808 if (strcmp(name, "VAL") == 0) {
809 while (*s == ' ') ++s;
810 int sign = 1;
811 if (*s == '-' || *s == '+') sign = *s++ == '-' ? -1 : 1;
812 if (!token_digit(*s)) return integer_value(0);
813 int64_t result = 0;
814 while (token_digit(*s)) {
815 result = result * 10 + *s++ - '0';
816 if (result > 32768) fail(m, "OVERFLOW");
817 }
818 return integer_value(checked(m, sign * result));
819 }
820 if (strcmp(name, "CHR$") == 0) {
821 if (n < 32 || n > 126) fail(m, "ILLEGAL FUNCTION CALL");
822 char text[2] = {(char)n, 0};
823 return text_value(copy(m, text));
824 }
825 if (strcmp(name, "STR$") == 0) {
826 char text[32];
827 snprintf(text, sizeof(text), "%" PRId64, n);
828 return text_value(copy(m, text));
829 }
830 if (strcmp(name, "LEFT$") == 0 || strcmp(name, "RIGHT$") == 0) {
831 if (k < 0) fail(m, "ILLEGAL FUNCTION CALL");
832 size_t length = strlen(s);
833 size_t amount = (size_t)k < length ? (size_t)k : length;
834 char *result = allocate(m, amount + 1);
835 const char *source = strcmp(name, "LEFT$") == 0 ? s : s + length - amount;
836 memcpy(result, source, amount);
837 return text_value(result);
838 }
839 if (strcmp(name, "MID$") == 0) {
840 int64_t length = count == 3 ? args[2].number : (int64_t)strlen(s);
841 if (k < 1 || length < 0) fail(m, "ILLEGAL FUNCTION CALL");
842 size_t size = strlen(s);
843 size_t offset = (size_t)(k - 1) < size ? (size_t)(k - 1) : size;
844 size_t amount = (size_t)length < size - offset ? (size_t)length : size - offset;
845 char *result = allocate(m, amount + 1);
846 memcpy(result, s + offset, amount);
847 return text_value(result);
848 }
849 if (strcmp(name, "INSTR") == 0) {
850 const char *found = strstr(s, args[1].text);
851 return integer_value(found ? found - s + 1 : 0);
852 }
853 fail(m, "ILLEGAL FUNCTION CALL");
854 return integer_value(0);
855}

857Value evaluate(Machine *m, int id) {
858 Expression expr = m->expressions[id];
859 if (expr.kind == 0) return expr.value;
860 if (expr.kind == 1 || expr.kind == 2) return load(m, address(m, id));
861 if (expr.kind == 3)
862 return integer_value(strcmp(expr.name, "ERR") == 0 ? m->err : m->erl);
863 if (expr.kind == 4) {
864 int64_t a = evaluate(m, expr.args.items[0]).number;
865 return integer_value(strcmp(expr.name, "NOT") == 0 ? ~a : checked(m, -a));
866 }
867 Value args[8] = {0};
868 for (int i = 0; i < expr.args.count; ++i)
869 args[i] = evaluate(m, expr.args.items[i]);
870 if (expr.kind == 5) return call(m, expr.name, args, expr.args.count);
871 Value a = args[0], b = args[1];
872 const char *op = expr.name;
873 if (strcmp(op, "=") == 0 || strcmp(op, "<>") == 0 ||
874 strcmp(op, "<") == 0 || strcmp(op, ">") == 0 ||
875 strcmp(op, "<=") == 0 || strcmp(op, ">=") == 0) {
876 int cmp = compare(a, b);
877 int yes = strcmp(op, "=") == 0 ? cmp == 0 :
878 strcmp(op, "<>") == 0 ? cmp != 0 :
879 strcmp(op, "<") == 0 ? cmp < 0 :
880 strcmp(op, ">") == 0 ? cmp > 0 :
881 strcmp(op, "<=") == 0 ? cmp <= 0 : cmp >= 0;
882 return integer_value(yes ? -1 : 0);
883 }
884 if (expr.type == 'S') {
885 size_t len = strlen(a.text) + strlen(b.text);
886 if (len > 255) fail(m, "STRING TOO LONG");
887 char *text = allocate(m, len + 1);
888 strcpy(text, a.text);
889 strcat(text, b.text);
890 return text_value(text);
891 }
892 int64_t x = a.number, y = b.number;
893 if (strcmp(op, "+") == 0) return integer_value(checked(m, x + y));
894 if (strcmp(op, "-") == 0) return integer_value(checked(m, x - y));
895 if (strcmp(op, "*") == 0) return integer_value(checked(m, x * y));
896 if (strcmp(op, "/") == 0 || strcmp(op, "MOD") == 0) {
897 if (y == 0) fail(m, "DIVISION BY ZERO");
898 return integer_value(strcmp(op, "MOD") == 0 ? x % y : checked(m, x / y));
899 }
900 if (strcmp(op, "^") == 0) return integer_value(integer_power(m, x, y));
901 if (strcmp(op, "AND") == 0) return integer_value(x & y);
902 if (strcmp(op, "OR") == 0) return integer_value(x | y);
903 fail(m, "ILLEGAL FUNCTION CALL");
904 return integer_value(0);
905}

927int matching(Machine *m, int id, const char *open, const char *close) {
928 int start = 0, depth = 0;
929 for (int i = 0; i < m->sequence.count; ++i)
930 if (m->sequence.items[i] == id) {
931 start = i + 1;
932 break;
933 }
934 for (int i = start; i < m->sequence.count; ++i) {
935 Statement *s = &m->statements[m->sequence.items[i]];
936 if (strcmp(s->kind, open) == 0) ++depth;
937 else if (strcmp(s->kind, close) == 0) {
938 if (!depth) return s->next;
939 --depth;
940 }
941 }
942 char error[80];
943 snprintf(error, sizeof(error), "%s WITHOUT %s", open, close);
944 fail(m, error);
945 return -1;
946}

954int execute(Machine *m, int id) {
955 Statement *s = &m->statements[id];
956 int next = s->next;
957 if (strcmp(s->kind, "REM") == 0 || strcmp(s->kind, "DATA") == 0)
958 return next;
959 if (strcmp(s->kind, "assign") == 0) {
960 Address place = address(m, s->target);
961 store(m, place, evaluate(m, s->expr));
962 } else if (strcmp(s->kind, "PRINT") == 0) {
963 for (int i = 0; i < s->items.count; ++i) {
964 PrintItem *item = &s->items.items[i];
965 if (item->kind == 2) {
966 append_spaces(m, 10 - m->buffer_length % 10);
967 } else if (item->kind != 1) {
968 Value value = evaluate(m, item->expr);
969 if (item->kind == 0) {
970 if (value.type == 'S') append(m, value.text);
971 else {
972 char text[32];
973 snprintf(text, sizeof(text), "%" PRId64, value.number);
974 append(m, text);
975 }
976 } else {
977 int n = (int)value.number;
978 if (n < 0) fail(m, "ILLEGAL FUNCTION CALL");
979 append_spaces(m, item->kind == 4 ? n :
980 n > m->buffer_length ? n - m->buffer_length : 0);
981 }
982 }
983 }
984 if (!s->items.count ||
985 (s->items.items[s->items.count - 1].kind != 1 &&
986 s->items.items[s->items.count - 1].kind != 2))
987 end_line(m);
988 } else if (strcmp(s->kind, "INPUT") == 0 || strcmp(s->kind, "READ") == 0) {
989 for (int i = 0; i < s->targets.count; ++i) {
990 int id = s->targets.items[i];
991 Address place = address(m, id);
992 Value value;
993 if (strcmp(s->kind, "READ") == 0) {
994 if (m->data_position == m->data.count) fail(m, "OUT OF DATA");
995 value = m->data.items[m->data_position++];
996 if (value.type != m->expressions[id].type)
997 fail(m, "TYPE MISMATCH");
998 } else {
999 for (;;) {
1000 if (m->input_position == m->input_count) fail(m, "INPUT PAST END");
1001 char *text = m->inputs[m->input_position++];
1002 if (m->expressions[id].type == 'S') {
1003 value = text_value(text);
1004 break;
1005 }
1006 const char *p = text;
1007 if (*p == '-') ++p;
1008 int valid = *p != 0;
1009 for (; *p; ++p) if (!token_digit(*p)) valid = 0;
1010 if (valid) {
1011 int64_t number = 0;
1012 int negative = text[0] == '-';
1013 p = text + negative;
1014 for (; *p; ++p) {
1015 number = number * 10 + *p - '0';
1016 if (number > 32768) { valid = 0; break; }
1017 }
1018 number = negative ? -number : number;
1019 if (valid && number >= -32768 && number <= 32767) {
1020 value = integer_value(number);
1021 break;
1022 }
1023 }
1024 message(m, "?REDO FROM START");
1025 }
1026 }
1027 store(m, place, value);
1028 }
1029 } else if (strcmp(s->kind, "RESTORE") == 0) {
1030 if (!s->numbers.count) m->data_position = 0;
1031 else {
1032 int line = s->numbers.items[0];
1033 lookup(m, line);
1034 m->data_position = m->data.count;
1035 for (int i = 0; i < m->data.count; ++i)
1036 if (m->data_lines.items[i] >= line) {
1037 m->data_position = i;
1038 break;
1039 }
1040 }
1041 } else if (strcmp(s->kind, "IF") == 0) {
1042 int branch = evaluate(m, s->expr).number != 0 ? 0 : 1;
1043 if (branch < s->clause_count) {
1044 Clause *clause = &s->clauses[branch];
1045 next = clause->number >= 0 ? lookup(m, clause->number) :
1046 clause->body.items[0];
1047 }
1048 } else if (strcmp(s->kind, "GOTO") == 0) {
1049 next = lookup(m, s->numbers.items[0]);
1050 } else if (strcmp(s->kind, "GOSUB") == 0) {
1051 next = subroutine(m, s->numbers.items[0], next);
1052 } else if (strcmp(s->kind, "ON") == 0) {
1053 int n = (int)evaluate(m, s->expr).number;
1054 if (n < 0) fail(m, "ILLEGAL FUNCTION CALL");
1055 if (n >= 1 && n <= s->numbers.count) {
1056 int number = s->numbers.items[n - 1];
1057 next = strcmp(s->mode, "GOTO") == 0 ?
1058 lookup(m, number) : subroutine(m, number, next);
1059 }
1060 } else if (strcmp(s->kind, "RETURN") == 0) {
1061 int found = m->stack_count - 1;
1062 while (found >= 0 && m->stack[found].kind != 3) --found;
1063 if (found < 0) fail(m, "RETURN WITHOUT GOSUB");
1064 next = m->stack[found].position;
1065 m->stack_count = found;
1066 } else if (strcmp(s->kind, "FOR") == 0) {
1067 int values[3];
1068 for (int i = 0; i < 3; ++i)
1069 values[i] = (int)evaluate(m, s->expressions.items[i]).number;
1070 int start = values[0], end = values[1], step = values[2];
1071 if (!step) fail(m, "ILLEGAL FUNCTION CALL");
1072 int found = search(m, 1, s->name);
1073 if (found >= 0) m->stack_count = found;
1074 int slot = variable_index(s->name);
1075 m->vars[slot] = integer_value(start);
1076 m->present[slot] = 1;
1077 if (step > 0 ? start > end : start < end) {
1078 next = matching(m, id, "FOR", "NEXT");
1079 } else {
1080 Frame frame = {.kind = 1, .position = next, .limit = end, .step = step};
1081 strcpy(frame.name, s->name);
1082 push_frame(m, frame);
1083 }
1084 } else if (strcmp(s->kind, "NEXT") == 0) {
1085 int found = search(m, 1, s->name);
1086 if (found < 0) fail(m, "NEXT WITHOUT FOR");
1087 m->stack_count = found + 1;
1088 Frame frame = m->stack[found];
1089 int value = checked(m, get(m, frame.name).number + frame.step);
1090 int slot = variable_index(frame.name);
1091 m->vars[slot] = integer_value(value);
1092 m->present[slot] = 1;
1093 if (frame.step > 0 ? value > frame.limit : value < frame.limit)
1094 --m->stack_count;
1095 else next = frame.position;
1096 } else if (strcmp(s->kind, "WHILE") == 0) {
1097 if (evaluate(m, s->expr).number)
1098 push_frame(m, (Frame){.kind = 2, .position = id});
1099 else next = matching(m, id, "WHILE", "WEND");
1100 } else if (strcmp(s->kind, "WEND") == 0) {
1101 int found = search(m, 2, "");
1102 if (found < 0) fail(m, "WEND WITHOUT WHILE");
1103 next = m->stack[found].position;
1104 m->stack_count = found;
1105 } else if (strcmp(s->kind, "DIM") == 0) {
1106 for (int i = 0; i < s->targets.count; ++i) {
1107 Expression *target = &m->expressions[s->targets.items[i]];
1108 int slot = variable_index(target->name);
1109 if (m->arrays[slot]) fail(m, "DUPLICATE DEFINITION");
1110 int bounds[2] = {0};
1111 int count = 1;
1112 for (int j = 0; j < target->args.count; ++j) {
1113 bounds[j] = (int)evaluate(m, target->args.items[j]).number;
1114 if (bounds[j] < 0) fail(m, "ILLEGAL FUNCTION CALL");
1115 count *= bounds[j] + 1;
1116 }
1117 if (count > 10000) fail(m, "OUT OF MEMORY");
1118 declare(m, target->name, bounds, target->args.count);
1119 }
1120 } else if (strcmp(s->kind, "RANDOMIZE") == 0) {
1121 int n = (int)evaluate(m, s->expr).number;
1122 if (n < 1) fail(m, "ILLEGAL FUNCTION CALL");
1123 m->random_state = n;
1124 } else if (strcmp(s->kind, "handler") == 0) {
1125 if (s->numbers.items[0]) lookup(m, s->numbers.items[0]);
1126 m->handler = s->numbers.items[0];
1127 } else if (strcmp(s->kind, "RESUME") == 0) {
1128 if (!m->handling) fail(m, "RESUME WITHOUT ERROR");
1129 next = strcmp(s->mode, "line") == 0 ? lookup(m, s->numbers.items[0]) :
1130 strcmp(s->mode, "NEXT") == 0 ?
1131 m->statements[m->remembered].next : m->remembered;
1132 m->handling = 0;
1133 } else if (strcmp(s->kind, "STOP") == 0) {
1134 char text[80];
1135 snprintf(text, sizeof(text), "BREAK IN LINE %d", s->line);
1136 message(m, text);
1137 return -1;
1138 } else if (strcmp(s->kind, "END") == 0) {
1139 return -1;
1140 }
1141 return next;
1142}

1144int error_code(const char *reason) {
1145 const char *errors[] = {
1146 "NEXT WITHOUT FOR", "RETURN WITHOUT GOSUB", "OUT OF DATA",
1147 "ILLEGAL FUNCTION CALL", "OVERFLOW", "OUT OF MEMORY",
1148 "UNDEFINED LINE NUMBER", "SUBSCRIPT OUT OF RANGE", "DUPLICATE DEFINITION",
1149 "DIVISION BY ZERO", "TYPE MISMATCH", "STRING TOO LONG",
1150 "RESUME WITHOUT ERROR", "FOR WITHOUT NEXT", "WHILE WITHOUT WEND",
1151 "WEND WITHOUT WHILE", "INPUT PAST END"
1152 };
1153 const int codes[] = {1, 3, 4, 5, 6, 7, 8, 9, 10, 11, 13, 15, 20, 26, 29, 30, 62};
1154 for (int i = 0; i < 17; ++i)
1155 if (strcmp(reason, errors[i]) == 0) return codes[i];
1156 return 0;
1157}

1159void run(Machine *m, const char *contents[], const int line_numbers[], int line_count) {
1160 IntVec parsed[200] = {0};
1161 for (int i = 0; i < 10000; ++i) m->entries[i] = -1;
1162 for (int i = 0; i < line_count; ++i) {
1163 if (setjmp(m->jump) != 0) {
1164 printf("ERROR IN LINE %d: SYNTAX ERROR\nREADY\n", line_numbers[i]);
1165 return;
1166 }
1167 parsed[i] = parse(m, contents[i]);
1168 }
1169 int following = -1;
1170 for (int i = line_count - 1; i >= 0; --i) {
1171 connect(m, parsed[i], following, line_numbers[i]);
1172 m->entries[line_numbers[i]] = parsed[i].items[0];
1173 following = parsed[i].items[0];
1174 }
1175 for (int i = 0; i < line_count; ++i) {
1176 for (int j = 0; j < parsed[i].count; ++j) {
1177 int id = parsed[i].items[j];
1178 int_push(&m->sequence, id);
1179 Statement *s = &m->statements[id];
1180 if (strcmp(s->kind, "DATA") == 0) {
1181 for (int k = 0; k < s->values.count; ++k) {
1182 value_push(&m->data, s->values.items[k]);
1183 int_push(&m->data_lines, s->line);
1184 }
1185 }
1186 }
1187 }
1188 int position = following;
1189 while (position >= 0) {
1190 int current = position;
1191 if (setjmp(m->jump) == 0) {
1192 position = execute(m, current);
1193 } else if (m->handler && !m->handling) {
1194 m->err = error_code(m->failure);
1195 m->erl = m->statements[current].line;
1196 m->remembered = current;
1197 m->handling = 1;
1198 position = lookup(m, m->handler);
1199 } else {
1200 char text[256];
1201 snprintf(text, sizeof(text), "ERROR IN LINE %d: %s",
1202 m->statements[current].line, m->failure);
1203 message(m, text);
1204 position = -1;
1205 }
1206 }
1207 if (m->buffer_length) end_line(m);
1208 puts("READY");
1209}
