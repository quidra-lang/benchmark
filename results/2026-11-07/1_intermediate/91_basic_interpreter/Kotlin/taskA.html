62private fun tokenize(source: String): List<Token> {
63 val tokens = mutableListOf<Token>()
64 var p = 0
65 while (p < source.length) {
66 val c = source[p]
67 if (c == ' ') {
68 p++
69 } else if (c == '"') {
70 val end = source.indexOf('"', p + 1)
71 if (end < 0) syntax()
72 tokens.add(Token("S", source.substring(p, end + 1), source.substring(p + 1, end)))
73 p = end + 1
74 } else if (c.isDigit()) {
75 val start = p
76 while (p < source.length && source[p].isDigit()) p++
77 val text = source.substring(start, p)
78 val number = text.toLongOrNull() ?: syntax()
79 if (number > 32767) syntax()
80 tokens.add(Token("I", text, number.toInt()))
81 } else if (c in 'A'..'Z') {
82 if (source.startsWith("REM", p)) {
83 tokens.add(Token("W", "REM", "REM"))
84 break
85 }
86 val start = p
87 while (p < source.length && (source[p] in 'A'..'Z' || source[p].isDigit())) p++
88 if (p < source.length && source[p] == '$') p++
89 val text = source.substring(start, p)
90 tokens.add(Token("W", text, text))
91 } else {
92 var text = source.substring(p, (p + 2).coerceAtMost(source.length))
93 if (text !in listOf("<>", "<=", ">=")) text = c.toString()
94 if (c !in "+-*/^=<>() ,;:") syntax()
95 tokens.add(Token("P", text, text))
96 p += text.length
97 }
98 }
99 tokens.add(Token("E", "", ""))
100 return tokens
101}

103private class Parser(private val tokens: List<Token>) {
104 private var index = 0
105 private fun now() = tokens[index]
106 private fun at(text: String) = now().type in listOf("W", "P") && now().text == text
107 private fun take(text: String): Boolean {
108 if (!at(text)) return false
109 index++
110 return true
111 }
112 private fun expect(text: String) {
113 if (!take(text)) syntax()
114 }
115 private fun number(): Int {
116 if (now().type != "I") syntax()
117 return tokens[index++].value as Int
118 }
119 private fun integer(level: Int = 0): Expr {
120 val e = expression(level)
121 if (e.type != "I") syntax()
122 return e
123 }
124 private fun target(): Expr {
125 val token = now()
126 if (token.type != "W" || !variable(token.text)) syntax()
127 index++
128 val e = Expr("variable", nameType(token.text), token.text)
129 if (take("(")) {
130 e.kind = "array"
131 e.args.add(integer())
132 if (take(",")) e.args.add(integer())
133 expect(")")
134 }
135 return e
136 }
137 private fun primary(): Expr {
138 val token = now()
139 if (token.type == "I" || token.type == "S") {
140 index++
141 return Expr("constant", token.type, token.value)
142 }
143 if (take("(")) {
144 val e = expression()
145 expect(")")
146 return e
147 }
148 if (token.type == "W" && variable(token.text)) return target()
149 if (take("ERR")) return Expr("builtin", "I", "ERR")
150 if (take("ERL")) return Expr("builtin", "I", "ERL")
151 if (token.type == "W" && signatures.containsKey(token.text)) {
152 index++
153 expect("(")
154 val args = mutableListOf(expression())
155 while (take(",")) args.add(expression())
156 expect(")")
157 if (args.joinToString("") { it.type } !in signatures.getValue(token.text)) syntax()
158 return Expr("call", nameType(token.text), token.text, args)
159 }
160 syntax()
161 }
162 private fun expression(level: Int = 0): Expr {
163 if (level == 2 && take("NOT")) return Expr("unary", "I", "NOT", mutableListOf(integer(2)))
164 if (level == 6 && take("-")) return Expr("unary", "I", "-", mutableListOf(integer(6)))
165 if (level == 8) return primary()
166 var left = expression(level + 1)
167 val operators = listOf(
168 listOf("OR"), listOf("AND"), emptyList(),
169 listOf("=", "<>", "<", ">", "<=", ">="),
170 listOf("+", "-"), listOf("*", "/", "MOD"),
171 emptyList(), listOf("^")
172 )[level]
173 while (operators.any { at(it) }) {
174 val op = tokens[index++].text
175 val right = expression(if (level == 7) 6 else level + 1)
176 if (left.type != right.type ||
177 (left.type == "S" && level != 3 && op != "+")) syntax()
178 left = Expr("binary", if (level == 3) "I" else left.type, op, mutableListOf(left, right))
179 if (level == 7) break
180 }
181 return left
182 }
183 private fun clause(): Clause =
184 if (now().type == "I") Clause(number(), emptyList()) else Clause(null, list(true))
185 private fun list(inClause: Boolean = false): List<Statement> {
186 val result = mutableListOf(statement(inClause))
187 while (take(":")) result.add(statement(inClause))
188 return result
189 }
190 private fun statement(inClause: Boolean): Statement {
191 val word = now().text
192 if (inClause && word in listOf("IF", "FOR", "NEXT", "WHILE", "WEND", "DATA")) syntax()
193 val s = Statement(word)
194 if (take("REM") || take("RETURN") || take("WEND") || take("END") || take("STOP"))
195 return s
196 if (take("PRINT")) {
197 var lastValue = false
198 while (now().type != "E" && !at(":") && !at("ELSE")) {
199 if (take(";")) {
200 s.items.add(PrintItem(";", null))
201 lastValue = false
202 } else if (take(",")) {
203 s.items.add(PrintItem(",", null))
204 lastValue = false
205 } else {
206 val mode = if (at("TAB")) "TAB" else if (at("SPC")) "SPC" else "value"
207 if (mode == "value" && lastValue) syntax()
208 val e = if (mode == "value") expression() else {
209 index++
210 expect("(")
211 val value = integer()
212 expect(")")
213 value
214 }
215 s.items.add(PrintItem(mode, e))
216 lastValue = mode == "value"
217 }
218 }
219 return s
220 }
221 if (take("INPUT") || take("READ") || take("DIM")) {
222 do {
223 val target = target()
224 if (word == "DIM" && target.kind != "array") syntax()
225 s.targets.add(target)
226 } while (take(","))
227 return s
228 }
229 if (take("DATA")) {
230 do {
231 val negative = take("-")
232 val token = now()
233 if (token.type != "I" && (token.type != "S" || negative)) syntax()
234 index++
235 s.values.add(if (negative) -(token.value as Int) else token.value)
236 } while (take(","))
237 return s
238 }
239 if (take("IF")) {
240 s.expr = integer()
241 expect("THEN")
242 s.clauses.add(clause())
243 if (take("ELSE")) s.clauses.add(clause())
244 if (now().type != "E") syntax()
245 return s
246 }
247 if (take("GOTO") || take("GOSUB")) {
248 s.numbers.add(number())
249 return s
250 }
251 if (take("ON")) {
252 if (take("ERROR")) {
253 s.kind = "handler"
254 expect("GOTO")
255 s.numbers.add(number())
256 } else {
257 s.expr = integer()
258 s.mode = if (take("GOTO")) "GOTO" else {
259 expect("GOSUB")
260 "GOSUB"
261 }
262 do { s.numbers.add(number()) } while (take(","))
263 }
264 return s
265 }
266 if (take("RESUME")) {
267 if (take("NEXT")) s.mode = "NEXT"
268 else if (now().type == "I") {
269 s.mode = "line"
270 s.numbers.add(number())
271 }
272 return s
273 }
274 if (take("FOR")) {
275 val token = now()
276 if (token.type != "W" || !variable(token.text) || nameType(token.text) != "I") syntax()
277 s.name = token.text
278 index++
279 expect("=")
280 s.expressions.add(integer())
281 expect("TO")
282 s.expressions.add(integer())
283 s.expressions.add(if (take("STEP")) integer() else Expr("constant", "I", 1))
284 return s
285 }
286 if (take("NEXT")) {
287 if (now().type == "W" && variable(now().text) && nameType(now().text) == "I")
288 s.name = tokens[index++].text
289 return s
290 }
291 if (take("WHILE") || take("RANDOMIZE")) {
292 s.expr = integer()
293 return s
294 }
295 if (take("RESTORE")) {
296 if (now().type == "I") s.numbers.add(number())
297 return s
298 }
299 take("LET")
300 s.kind = "assign"
301 s.target = target()
302 expect("=")
303 s.expr = expression()
304 if (s.expr!!.type != s.target!!.type) syntax()
305 return s
306 }
307 fun parse(): List<Statement> {
308 val result = list()
309 if (now().type != "E") syntax()
310 return result
311 }
312}

314private class Machine(private val inputs: List<String>) {
315 private val vars = mutableMapOf<String, Any>()
316 private val arrays = mutableMapOf<String, BasicArray>()
317 private val stack = mutableListOf<Frame>()
318 private val lines = mutableMapOf<Int, Statement>()
319 private val sequence = mutableListOf<Statement>()
320 private val data = mutableListOf<Pair<Int, Any>>()
321 private var dataIndex = 0
322 private var inputIndex = 0
323 private var random = 1L
324 private var handler = 0
325 private var handling = false
326 private var remembered: Statement? = null
327 private var err = 0
328 private var erl = 0
329 private var buffer = ""
330 private val output = mutableListOf<String>()
331
332 private fun endLine() {
333 output.add(buffer.trimEnd(' '))
334 buffer = ""
335 }
336 private fun message(s: String) {
337 if (buffer.isNotEmpty()) endLine()
338 output.add(s)
339 }
340 private fun checked(n: Long): Int {
341 if (n < -32768 || n > 32767) fault("OVERFLOW")
342 return n.toInt()
343 }
344 private fun get(name: String): Any = vars[name] ?: if (nameType(name) == "S") "" else 0
345 private fun declare(name: String, bounds: List<Int>) {
346 val count = bounds.fold(1) { a, b -> a * (b + 1) }
347 val zero: Any = if (nameType(name) == "S") "" else 0
348 arrays[name] = BasicArray(bounds, MutableList(count) { zero })
349 }
350 private fun address(expr: Expr): Address {
351 val name = expr.value as String
352 if (expr.kind == "variable") return Address(name, null)
353 val indices = expr.args.map { evaluate(it) as Int }
354 if (!arrays.containsKey(name)) declare(name, indices.map { 10 })
355 val array = arrays.getValue(name)
356 if (indices.size != array.bounds.size ||
357 indices.indices.any { indices[it] < 0 || indices[it] > array.bounds[it] })
358 fault("SUBSCRIPT OUT OF RANGE")
359 var offset = 0
360 indices.forEachIndexed { i, value -> offset = offset * (array.bounds[i] + 1) + value }
361 return Address(name, offset)
362 }
363 private fun load(a: Address): Any =
364 if (a.index == null) get(a.name) else arrays.getValue(a.name).values[a.index]
365 private fun store(a: Address, value: Any) {
366 if (a.index == null) vars[a.name] = value else arrays.getValue(a.name).values[a.index] = value
367 }
368 private fun power(a: Int, b: Int): Int {
369 if (b < 0) fault("ILLEGAL FUNCTION CALL")
370 if (b == 0) return 1
371 var result = 1L
372 var base = a.toLong()
373 var exponent = b
374 while (exponent > 0) {
375 if (exponent % 2 == 1) result = checked(result * base).toLong()
376 exponent /= 2
377 if (exponent > 0) base = checked(base * base).toLong()
378 }
379 return result.toInt()
380 }
381 private fun evaluate(e: Expr): Any {
382 val op = e.value
383 if (e.kind == "constant") return op
384 if (e.kind == "variable" || e.kind == "array") return load(address(e))
385 if (e.kind == "builtin") return if (op == "ERR") err else erl
386 if (e.kind == "unary") {
387 val value = evaluate(e.args[0]) as Int
388 return if (op == "NOT") value.inv() else checked(-value.toLong())
389 }
390 val args = e.args.map { evaluate(it) }
391 if (e.kind == "call") return call(op as String, args)
392 val a = args[0]
393 val b = args[1]
394 if (op in listOf("=", "<>", "<", ">", "<=", ">=")) {
395 val cmp = if (a == b) 0 else if (a is String) a.compareTo(b as String)
396 else (a as Int).compareTo(b as Int)
397 return if (when (op) {
398 "=" -> cmp == 0; "<>" -> cmp != 0; "<" -> cmp < 0
399 ">" -> cmp > 0; "<=" -> cmp <= 0; else -> cmp >= 0
400 }) -1 else 0
401 }
402 if (e.type == "S") {
403 val text = (a as String) + (b as String)
404 if (text.length > 255) fault("STRING TOO LONG")
405 return text
406 }
407 val x = a as Int
408 val y = b as Int
409 return when (op) {
410 "+" -> checked(x.toLong() + y)
411 "-" -> checked(x.toLong() - y)
412 "*" -> checked(x.toLong() * y)
413 "/" -> if (y == 0) fault("DIVISION BY ZERO") else checked(x.toLong() / y)
414 "MOD" -> if (y == 0) fault("DIVISION BY ZERO") else x % y
415 "^" -> power(x, y)
416 "AND" -> x and y
417 "OR" -> x or y
418 else -> syntax()
419 }
420 }
421 private fun call(name: String, args: List<Any>): Any {
422 val n = args[0] as? Int ?: 0
423 val s = args[0] as? String ?: ""
424 val k = if (args.size > 1) args[1] as? Int ?: 0 else 0
425 return when (name) {
426 "ABS" -> checked(abs(n.toLong()))
427 "SGN" -> n.compareTo(0)
428 "INT" -> n
429 "SQR" -> if (n < 0) fault("ILLEGAL FUNCTION CALL") else sqrt(n.toDouble()).toInt()
430 "RND" -> {
431 if (n < 1) fault("ILLEGAL FUNCTION CALL")
432 random = 48271 * random % 2147483647
433 (random % n + 1).toInt()
434 }
435 "LEN" -> s.length
436 "ASC" -> if (s.isEmpty()) fault("ILLEGAL FUNCTION CALL") else s[0].code
437 "VAL" -> checked(Regex("^ *([+-]?[0-9]+)").find(s)?.groupValues?.get(1)?.toLongOrNull() ?: 0L)
438 "CHR$" -> if (n !in 32..126) fault("ILLEGAL FUNCTION CALL") else n.toChar().toString()
439 "STR$" -> n.toString()
440 "LEFT$" -> if (k < 0) fault("ILLEGAL FUNCTION CALL") else s.take(k)
441 "RIGHT$" -> if (k < 0) fault("ILLEGAL FUNCTION CALL") else s.takeLast(k)
442 "MID$" -> {
443 val length = if (args.size == 3) args[2] as Int else s.length
444 if (k < 1 || length < 0) fault("ILLEGAL FUNCTION CALL")
445 if (k > s.length) "" else s.substring(k - 1, (k - 1 + length).coerceAtMost(s.length))
446 }
447 "INSTR" -> s.indexOf(args[1] as String) + 1
448 else -> syntax()
449 }
450 }
451 private fun line(n: Int): Statement = lines[n] ?: fault("UNDEFINED LINE NUMBER")
452 private fun push(frame: Frame) {
453 if (stack.size == 100) fault("OUT OF MEMORY")
454 stack.add(frame)
455 }
456 private fun search(kind: String, name: String = ""): Int {
457 for (i in stack.indices.reversed()) {
458 val f = stack[i]
459 if (f.kind == "GOSUB") break
460 if (f.kind == kind && (name.isEmpty() || f.name == name)) return i
461 }
462 return -1
463 }
464 private fun matching(s: Statement, open: String, close: String): Statement? {
465 var depth = 0
466 for (i in (sequence.indexOf(s) + 1) until sequence.size) {
467 val t = sequence[i]
468 if (t.kind == open) depth++
469 else if (t.kind == close) {
470 if (depth == 0) return t.next
471 depth--
472 }
473 }
474 fault("$open WITHOUT $close")
475 }
476 private fun subroutine(number: Int, after: Statement?): Statement {
477 val destination = line(number)
478 push(Frame("GOSUB", after))
479 return destination
480 }
481 private fun execute(s: Statement): Statement? {
482 var next = s.next
483 when (s.kind) {
484 "REM", "DATA" -> Unit
485 "assign" -> {
486 val place = address(s.target!!)
487 store(place, evaluate(s.expr!!))
488 }
489 "PRINT" -> {
490 for (item in s.items) {
491 if (item.kind == ",") buffer += " ".repeat(10 - buffer.length % 10)
492 else if (item.kind != ";") {
493 val value = evaluate(item.expr!!)
494 if (item.kind == "value") buffer += value.toString()
495 else {
496 val count = value as Int
497 if (count < 0) fault("ILLEGAL FUNCTION CALL")
498 buffer += " ".repeat(if (item.kind == "SPC") count else (count - buffer.length).coerceAtLeast(0))
499 }
500 }
501 }
502 if (s.items.isEmpty() || s.items.last().kind !in listOf(";", ",")) endLine()
503 }
504 "INPUT", "READ" -> for (target in s.targets) {
505 val place = address(target)
506 val value: Any
507 if (s.kind == "READ") {
508 if (dataIndex == data.size) fault("OUT OF DATA")
509 value = data[dataIndex++].second
510 if (valueType(value) != target.type) fault("TYPE MISMATCH")
511 } else {
512 var received: Any
513 while (true) {
514 if (inputIndex == inputs.size) fault("INPUT PAST END")
515 val text = inputs[inputIndex++]
516 if (target.type == "S") {
517 received = text
518 break
519 }
520 val number = if (Regex("-?[0-9]+").matches(text)) text.toIntOrNull() else null
521 if (number != null && number in -32768..32767) {
522 received = number
523 break
524 }
525 message("?REDO FROM START")
526 }
527 value = received
528 }
529 store(place, value)
530 }
531 "RESTORE" -> {
532 if (s.numbers.isNotEmpty()) {
533 val number = s.numbers[0]
534 line(number)
535 val found = data.indexOfFirst { it.first >= number }
536 dataIndex = if (found < 0) data.size else found
537 } else dataIndex = 0
538 }
539 "IF" -> {
540 val clause = s.clauses.getOrNull(if ((evaluate(s.expr!!) as Int) != 0) 0 else 1)
541 if (clause != null) next = if (clause.number == null) clause.body[0] else line(clause.number)
542 }
543 "GOTO" -> next = line(s.numbers[0])
544 "GOSUB" -> next = subroutine(s.numbers[0], next)
545 "ON" -> {
546 val index = evaluate(s.expr!!) as Int
547 if (index < 0) fault("ILLEGAL FUNCTION CALL")
548 if (index in 1..s.numbers.size) {
549 val target = s.numbers[index - 1]
550 next = if (s.mode == "GOTO") line(target) else subroutine(target, next)
551 }
552 }
553 "RETURN" -> {
554 val found = stack.indexOfLast { it.kind == "GOSUB" }
555 if (found < 0) fault("RETURN WITHOUT GOSUB")
556 next = stack[found].position
557 stack.subList(found, stack.size).clear()
558 }
559 "FOR" -> {
560 val values = s.expressions.map { evaluate(it) as Int }
561 val start = values[0]
562 val limit = values[1]
563 val step = values[2]
564 if (step == 0) fault("ILLEGAL FUNCTION CALL")
565 val found = search("FOR", s.name)
566 if (found >= 0) stack.subList(found, stack.size).clear()
567 vars[s.name] = start
568 if (if (step > 0) start > limit else start < limit) next = matching(s, "FOR", "NEXT")
569 else push(Frame("FOR", next, s.name, limit, step))
570 }
571 "NEXT" -> {
572 val found = search("FOR", s.name)
573 if (found < 0) fault("NEXT WITHOUT FOR")
574 stack.subList(found + 1, stack.size).clear()
575 val frame = stack[found]
576 val value = checked((get(frame.name) as Int).toLong() + frame.step)
577 vars[frame.name] = value
578 if (if (frame.step > 0) value > frame.limit else value < frame.limit)
579 stack.removeAt(stack.lastIndex)
580 else next = frame.position
581 }
582 "WHILE" -> {
583 if ((evaluate(s.expr!!) as Int) != 0) push(Frame("WHILE", s))
584 else next = matching(s, "WHILE", "WEND")
585 }
586 "WEND" -> {
587 val found = search("WHILE")
588 if (found < 0) fault("WEND WITHOUT WHILE")
589 next = stack[found].position
590 stack.subList(found, stack.size).clear()
591 }
592 "DIM" -> for (target in s.targets) {
593 val name = target.value as String
594 val bounds = target.args.map { evaluate(it) as Int }
595 if (arrays.containsKey(name)) fault("DUPLICATE DEFINITION")
596 if (bounds.any { it < 0 }) fault("ILLEGAL FUNCTION CALL")
597 if (bounds.fold(1L) { a, b -> a * (b + 1) } > 10000) fault("OUT OF MEMORY")
598 declare(name, bounds)
599 }
600 "RANDOMIZE" -> {
601 val value = evaluate(s.expr!!) as Int
602 if (value < 1) fault("ILLEGAL FUNCTION CALL")
603 random = value.toLong()
604 }
605 "handler" -> {
606 if (s.numbers[0] != 0) line(s.numbers[0])
607 handler = s.numbers[0]
608 }
609 "RESUME" -> {
610 if (!handling) fault("RESUME WITHOUT ERROR")
611 next = when (s.mode) {
612 "line" -> line(s.numbers[0])
613 "NEXT" -> remembered!!.next
614 else -> remembered
615 }
616 handling = false
617 }
618 "STOP" -> {
619 message("BREAK IN LINE " + s.line)
620 return null
621 }
622 "END" -> return null
623 }
624 return next
625 }
626 private fun connect(body: List<Statement>, after: Statement?, line: Int) {
627 var following = after
628 for (s in body.asReversed()) {
629 s.line = line
630 s.next = following
631 s.clauses.forEach { connect(it.body, following, line) }
632 following = s
633 }
634 }
635 fun run(source: Map<Int, String>): List<String> {
636 val parsed = mutableListOf<Pair<Int, List<Statement>>>()
637 for (line in source.keys.sorted()) {
638 try {
639 parsed.add(line to Parser(tokenize(source.getValue(line))).parse())
640 } catch (_: SyntaxFault) {
641 message("ERROR IN LINE $line: SYNTAX ERROR")
642 return finish()
643 }
644 }
645 var following: Statement? = null
646 for ((number, statements) in parsed.asReversed()) {
647 connect(statements, following, number)
648 lines[number] = statements.first()
649 following = statements.first()
650 }
651 sequence.addAll(parsed.flatMap { it.second })
652 for (s in sequence) if (s.kind == "DATA")
653 for (value in s.values) data.add(s.line to value)
654 var position = following
655 while (position != null) {
656 val statement = position
657 try {
658 position = execute(statement)
659 } catch (problem: RuntimeFault) {
660 if (handler != 0 && !handling) {
661 err = errorCodes[problem.message] ?: 0
662 erl = statement.line
663 remembered = statement
664 handling = true
665 position = line(handler)
666 } else {
667 message("ERROR IN LINE " + statement.line + ": " + problem.message)
668 position = null
669 }
670 }
671 }
672 return finish()
673 }
674 private fun finish(): List<String> {
675 if (buffer.isNotEmpty()) endLine()
676 output.add("READY")
677 return output
678 }
679}

681fun main() {
682 val reader = BufferedReader(InputStreamReader(System.`in`))
683 val sessions = reader.readLine()?.toIntOrNull() ?: return
684 val output = StringBuilder()
685 repeat(sessions) {
686 val source = mutableMapOf<Int, String>()
687 while (true) {
688 val line = reader.readLine() ?: break
689 if (line == "RUN") break
690 val result = Regex("([0-9]+)(?: +(.*))?").matchEntire(line) ?: continue
691 val number = result.groupValues[1].toInt()
692 if (result.groups[2] == null) source.remove(number)
693 else source[number] = result.groupValues[2]
694 }
695 val data = mutableListOf<String>()
696 while (true) {
697 val line = reader.readLine() ?: break
698 if (line == "END") break
699 data.add(line)
700 }
701 for (line in Machine(data).run(source)) output.append(line).append('\n')
702 }
703 print(output.toString())
704}
