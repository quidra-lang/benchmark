81std::vector<Token> tokenize(const std::string& source) {
82 std::vector<Token> result;
83 std::size_t pos = 0;
84 while (pos < source.size()) {
85 char c = source[pos];
86 if (c == ' ') {
87 ++pos;
88 } else if (c == '"') {
89 auto end = source.find('"', pos + 1);
90 if (end == std::string::npos) syntax();
91 result.push_back({'S', source.substr(pos, end - pos + 1),
92 source.substr(pos + 1, end - pos - 1)});
93 pos = end + 1;
94 } else if (digit(c)) {
95 auto start = pos;
96 while (pos < source.size() && digit(source[pos])) ++pos;
97 std::string word = source.substr(start, pos - start);
98 long long value = 0;
99 for (char letter : word) {
100 value = value * 10 + letter - '0';
101 if (value > 32767) syntax();
102 }
103 result.push_back({'I', word, static_cast<int>(value)});
104 } else if (upper(c)) {
105 if (source.compare(pos, 3, "REM") == 0) {
106 result.push_back({'W', "REM", std::string("REM")});
107 break;
108 }
109 auto start = pos;
110 while (pos < source.size() && (upper(source[pos]) || digit(source[pos]))) ++pos;
111 if (pos < source.size() && source[pos] == '$') ++pos;
112 std::string word = source.substr(start, pos - start);
113 result.push_back({'W', word, word});
114 } else {
115 std::string word(1, c);
116 if (pos + 1 < source.size()) {
117 std::string pair = source.substr(pos, 2);
118 if (pair == "<>" || pair == "<=" || pair == ">=") word = pair;
119 }
120 if (std::string("+-*/^=<>() ,;:").find(c) == std::string::npos) syntax();
121 result.push_back({'P', word, word});
122 pos += word.size();
123 }
124 }
125 result.push_back({'E', "", std::string("")});
126 return result;
127}

174class Parser {
175 std::vector<Token> tokens;
176 std::size_t pos = 0;
177
178 const Token& current() const {
179 return tokens[pos];
180 }
181
182 bool at(const std::string& word) const {
183 return (current().kind == 'P' || current().kind == 'W')
184 && current().text == word;
185 }
186
187 bool take(const std::string& word) {
188 if (!at(word)) return false;
189 ++pos;
190 return true;
191 }
192
193 void expect(const std::string& word) {
194 if (!take(word)) syntax();
195 }
196
197 int number() {
198 if (current().kind != 'I') syntax();
199 return std::get<int>(tokens[pos++].value);
200 }
201
202 Expr integer(int level = 0) {
203 Expr value = expression(level);
204 if (value->type != 'I') syntax();
205 return value;
206 }
207
208 Expr target() {
209 const Token& token = current();
210 if (token.kind != 'W' || !variable(token.text)) syntax();
211 std::string name = token.text;
212 ++pos;
213 Expr result = expression_node("variable", name_type(name), name);
214 if (take("(")) {
215 result->kind = "array";
216 result->args.push_back(integer());
217 if (take(",")) result->args.push_back(integer());
218 expect(")");
219 }
220 return result;
221 }
222
223 Expr primary() {
224 const Token& token = current();
225 if (token.kind == 'I' || token.kind == 'S') {
226 ++pos;
227 return expression_node("constant", token.kind, token.value);
228 }
229 if (take("(")) {
230 Expr value = expression();
231 expect(")");
232 return value;
233 }
234 if (token.kind == 'W' && variable(token.text)) return target();
235 if (take("ERR")) return expression_node("builtin", 'I', std::string("ERR"));
236 if (take("ERL")) return expression_node("builtin", 'I', std::string("ERL"));
237 if (token.kind == 'W' && signatures.contains(token.text)) {
238 std::string name = token.text;
239 ++pos;
240 expect("(");
241 std::vector<Expr> args{expression()};
242 while (take(",")) args.push_back(expression());
243 expect(")");
244 std::string types;
245 for (auto& e : args) types += e->type;
246 auto& expected = signatures.at(name);
247 if (std::find(expected.begin(), expected.end(), types) == expected.end()) syntax();
248 return expression_node("call", name_type(name), name, std::move(args));
249 }
250 syntax();
251 }
252
253 Expr expression(int level = 0) {
254 if (level == 2 && take("NOT"))
255 return expression_node("unary", 'I', std::string("NOT"), {integer(2)});
256 if (level == 6 && take("-"))
257 return expression_node("unary", 'I', std::string("-"), {integer(6)});
258 if (level == 8) return primary();
259 Expr left = expression(level + 1);
260 const std::vector<std::vector<std::string>> ops = {
261 {"OR"}, {"AND"}, {}, {"=", "<>", "<", ">", "<=", ">="},
262 {"+", "-"}, {"*", "/", "MOD"}, {}, {"^"}
263 };
264 while (std::find(ops[level].begin(), ops[level].end(), current().text) != ops[level].end()) {
265 std::string op = tokens[pos++].text;
266 Expr right = expression(level == 7 ? 6 : level + 1);
267 if (left->type != right->type ||
268 (left->type == 'S' && level != 3 && op != "+")) syntax();
269 left = expression_node("binary", level == 3 ? 'I' : left->type,
270 op, {left, right});
271 if (level == 7) break;
272 }
273 return left;
274 }
275
276 Clause clause() {
277 if (current().kind == 'I') return {number(), {}};
278 return {std::nullopt, list(true)};
279 }
280
281 std::vector<Stmt> list(bool in_clause = false) {
282 std::vector<Stmt> result{statement(in_clause)};
283 while (take(":")) result.push_back(statement(in_clause));
284 return result;
285 }
286
287 Stmt statement(bool in_clause) {
288 std::string word = current().text;
289 if (in_clause && std::set<std::string>{"IF", "FOR", "NEXT", "WHILE", "WEND", "DATA"}.contains(word))
290 syntax();
291 Stmt s = std::make_shared<Statement>();
292 s->kind = word;
293 if (take("REM") || take("RETURN") || take("WEND") || take("END") || take("STOP"))
294 return s;
295 if (take("PRINT")) {
296 bool value_last = false;
297 while (current().kind != 'E' && !at(":") && !at("ELSE")) {
298 if (take(";")) {
299 s->items.push_back({";", nullptr});
300 value_last = false;
301 } else if (take(",")) {
302 s->items.push_back({",", nullptr});
303 value_last = false;
304 } else {
305 std::string mode = at("TAB") ? "TAB" : at("SPC") ? "SPC" : "value";
306 if (mode == "value" && value_last) syntax();
307 Expr value;
308 if (mode == "value") {
309 value = expression();
310 } else {
311 ++pos;
312 expect("(");
313 value = integer();
314 expect(")");
315 }
316 s->items.push_back({mode, value});
317 value_last = mode == "value";
318 }
319 }
320 return s;
321 }
322 if (take("INPUT") || take("READ") || take("DIM")) {
323 do {
324 Expr ref = target();
325 if (word == "DIM" && ref->kind != "array") syntax();
326 s->targets.push_back(ref);
327 } while (take(","));
328 return s;
329 }
330 if (take("DATA")) {
331 do {
332 bool negative = take("-");
333 Token token = current();
334 if (token.kind != 'I' && (token.kind != 'S' || negative)) syntax();
335 ++pos;
336 if (negative) s->values.push_back(-std::get<int>(token.value));
337 else s->values.push_back(token.value);
338 } while (take(","));
339 return s;
340 }
341 if (take("IF")) {
342 s->expr = integer();
343 expect("THEN");
344 s->clauses.push_back(clause());
345 if (take("ELSE")) s->clauses.push_back(clause());
346 if (current().kind != 'E') syntax();
347 return s;
348 }
349 if (take("GOTO") || take("GOSUB")) {
350 s->numbers.push_back(number());
351 return s;
352 }
353 if (take("ON")) {
354 if (take("ERROR")) {
355 s->kind = "handler";
356 expect("GOTO");
357 s->numbers.push_back(number());
358 } else {
359 s->expr = integer();
360 if (take("GOTO")) s->mode = "GOTO";
361 else {
362 expect("GOSUB");
363 s->mode = "GOSUB";
364 }
365 do {
366 s->numbers.push_back(number());
367 } while (take(","));
368 }
369 return s;
370 }
371 if (take("RESUME")) {
372 if (take("NEXT")) s->mode = "NEXT";
373 else if (current().kind == 'I') {
374 s->mode = "line";
375 s->numbers.push_back(number());
376 }
377 return s;
378 }
379 if (take("FOR")) {
380 Token token = current();
381 if (token.kind != 'W' || !variable(token.text) || name_type(token.text) != 'I')
382 syntax();
383 s->name = token.text;
384 ++pos;
385 expect("=");
386 s->expressions.push_back(integer());
387 expect("TO");
388 s->expressions.push_back(integer());
389 s->expressions.push_back(take("STEP") ? integer() :
390 expression_node("constant", 'I', 1));
391 return s;
392 }
393 if (take("NEXT")) {
394 if (current().kind == 'W' && variable(current().text) &&
395 name_type(current().text) == 'I')
396 s->name = tokens[pos++].text;
397 return s;
398 }
399 if (take("WHILE") || take("RANDOMIZE")) {
400 s->expr = integer();
401 return s;
402 }
403 if (take("RESTORE")) {
404 if (current().kind == 'I') s->numbers.push_back(number());
405 return s;
406 }
407 take("LET");
408 s->kind = "assign";
409 s->target = target();
410 expect("=");
411 s->expr = expression();
412 if (s->expr->type != s->target->type) syntax();
413 return s;
414 }

474 int checked(std::int64_t value) {
475 if (value < -32768 || value > 32767) fault("OVERFLOW");
476 return static_cast<int>(value);

479 Value get(const std::string& name) {
480 auto found = variables.find(name);
481 if (found != variables.end()) return found->second;
482 return name_type(name) == 'S' ? Value(std::string("")) : Value(0);
483 }

485 void declare(const std::string& name, const std::vector<int>& bounds) {
486 std::size_t count = 1;
487 for (int bound : bounds) count *= static_cast<std::size_t>(bound + 1);
488 Value zero = name_type(name) == 'S' ? Value(std::string("")) : Value(0);
489 arrays[name] = {bounds, std::vector<Value>(count, zero)};
490 }
491
492 Address address(const Expr& target) {
493 std::string name = std::get<std::string>(target->value);
494 if (target->kind == "variable") return {name, std::nullopt};
495 std::vector<int> indices;
496 for (auto& arg : target->args) indices.push_back(std::get<int>(evaluate(arg)));
497 if (!arrays.contains(name)) declare(name, std::vector<int>(indices.size(), 10));
498 BasicArray& array = arrays.at(name);
499 if (indices.size() != array.bounds.size()) fault("SUBSCRIPT OUT OF RANGE");
500 std::size_t index = 0;
501 for (std::size_t i = 0; i < indices.size(); ++i) {
502 int value = indices[i];
503 if (value < 0 || value > array.bounds[i]) fault("SUBSCRIPT OUT OF RANGE");
504 index = index * (array.bounds[i] + 1) + value;
505 }
506 return {name, index};
507 }

518 int integer_power(int base, int exponent) {
519 if (exponent < 0) fault("ILLEGAL FUNCTION CALL");
520 std::int64_t result = 1;
521 std::int64_t current = base;
522 while (exponent > 0) {
523 if (exponent % 2) result = checked(result * current);
524 exponent /= 2;
525 if (exponent) current = checked(current * current);
526 }
527 return static_cast<int>(result);
528 }
529
530 Value evaluate(const Expr& expression) {
531 const Value& op = expression->value;
532 if (expression->kind == "constant") return op;
533 if (expression->kind == "variable" || expression->kind == "array")
534 return load(address(expression));
535 if (expression->kind == "builtin")
536 return std::get<std::string>(op) == "ERR" ? err : erl;
537 if (expression->kind == "unary") {
538 int number = std::get<int>(evaluate(expression->args[0]));
539 return std::get<std::string>(op) == "NOT" ? ~number : checked(-static_cast<std::int64_t>(number));
540 }
541 std::vector<Value> args;
542 for (auto& part : expression->args) args.push_back(evaluate(part));
543 if (expression->kind == "call") return call(std::get<std::string>(op), args);
544 const std::string& operation = std::get<std::string>(op);
545 const Value& a = args[0];
546 const Value& b = args[1];
547 if (std::set<std::string>{"=", "<>", "<", ">", "<=", ">="}.contains(operation)) {
548 int order = 0;
549 if (std::holds_alternative<std::string>(a))
550 order = std::get<std::string>(a).compare(std::get<std::string>(b));
551 else {
552 int left = std::get<int>(a), right = std::get<int>(b);
553 order = (left > right) - (left < right);
554 }
555 bool yes = operation == "=" ? order == 0 :
556 operation == "<>" ? order != 0 :
557 operation == "<" ? order < 0 :
558 operation == ">" ? order > 0 :
559 operation == "<=" ? order <= 0 : order >= 0;
560 return yes ? -1 : 0;
561 }
562 if (expression->type == 'S') {
563 std::string value = std::get<std::string>(a) + std::get<std::string>(b);
564 if (value.size() > 255) fault("STRING TOO LONG");
565 return value;
566 }
567 int x = std::get<int>(a), y = std::get<int>(b);
568 if (operation == "+") return checked(static_cast<std::int64_t>(x) + y);
569 if (operation == "-") return checked(static_cast<std::int64_t>(x) - y);
570 if (operation == "*") return checked(static_cast<std::int64_t>(x) * y);
571 if (operation == "/") {
572 if (y == 0) fault("DIVISION BY ZERO");
573 return checked(static_cast<std::int64_t>(x) / y);
574 }
575 if (operation == "MOD") {
576 if (y == 0) fault("DIVISION BY ZERO");
577 return x % y;
578 }
579 if (operation == "^") return integer_power(x, y);
580 if (operation == "AND") return x & y;
581 if (operation == "OR") return x | y;
582 syntax();
583 }

585 Value call(const std::string& name, const std::vector<Value>& args) {
586 int n = std::holds_alternative<int>(args[0]) ? std::get<int>(args[0]) : 0;
587 std::string s = std::holds_alternative<std::string>(args[0]) ?
588 std::get<std::string>(args[0]) : "";
589 int k = args.size() > 1 && std::holds_alternative<int>(args[1]) ?
590 std::get<int>(args[1]) : 0;
591 if (name == "ABS") return checked(std::llabs(static_cast<std::int64_t>(n)));
592 if (name == "SGN") return (n > 0) - (n < 0);
593 if (name == "INT") return n;
594 if (name == "SQR") {
595 if (n < 0) fault("ILLEGAL FUNCTION CALL");
596 return static_cast<int>(std::sqrt(n));
597 }
598 if (name == "RND") {
599 if (n < 1) fault("ILLEGAL FUNCTION CALL");
600 random_state = (48271 * random_state) % 2147483647;
601 return static_cast<int>(random_state % n) + 1;
602 }
603 if (name == "LEN") return static_cast<int>(s.size());
604 if (name == "ASC") {
605 if (s.empty()) fault("ILLEGAL FUNCTION CALL");
606 return static_cast<int>(static_cast<unsigned char>(s[0]));
607 }
608 if (name == "VAL") {
609 std::smatch match;
610 static const std::regex pattern(R"(^ *([+-]?[0-9]+))");
611 if (!std::regex_search(s, match, pattern)) return 0;
612 std::int64_t result = 0;
613 try {
614 result = std::stoll(match[1].str());
615 } catch (const std::out_of_range&) {
616 fault("OVERFLOW");
617 }
618 return checked(result);
619 }
620 if (name == "CHR$") {
621 if (n < 32 || n > 126) fault("ILLEGAL FUNCTION CALL");
622 return std::string(1, static_cast<char>(n));
623 }
624 if (name == "STR$") return std::to_string(n);
625 if (name == "LEFT$") {
626 if (k < 0) fault("ILLEGAL FUNCTION CALL");
627 return s.substr(0, static_cast<std::size_t>(k));
628 }
629 if (name == "RIGHT$") {
630 if (k < 0) fault("ILLEGAL FUNCTION CALL");
631 return s.substr(s.size() - std::min(s.size(), static_cast<std::size_t>(k)));
632 }
633 if (name == "MID$") {
634 if (k < 1) fault("ILLEGAL FUNCTION CALL");
635 int length = args.size() == 3 ? std::get<int>(args[2]) :
636 static_cast<int>(s.size());
637 if (length < 0) fault("ILLEGAL FUNCTION CALL");
638 if (static_cast<std::size_t>(k) > s.size()) return std::string("");
639 return s.substr(k - 1, length);
640 }
641 if (name == "INSTR") {
642 auto pos = s.find(std::get<std::string>(args[1]));
643 return pos == std::string::npos ? 0 : static_cast<int>(pos + 1);
644 }
645 syntax();
646 }

660 for (int i = static_cast<int>(stack.size()) - 1; i >= 0; --i) {

668 Statement* matching(Statement* s, const std::string& open, const std::string& close) {
669 auto found = std::find(sequence.begin(), sequence.end(), s);
670 int depth = 0;
671 for (auto it = found + 1; it != sequence.end(); ++it) {
672 Statement* current = *it;
673 if (current->kind == open) ++depth;
674 else if (current->kind == close) {
675 if (depth == 0) return current->next;
676 --depth;
677 }
678 }
679 fault(open + " WITHOUT " + close);
680 }

688 Statement* execute(Statement* s) {
689 Statement* next = s->next;
690 if (s->kind == "REM" || s->kind == "DATA") return next;
691 if (s->kind == "assign") {
692 Address place = address(s->target);
693 store(place, evaluate(s->expr));
694 } else if (s->kind == "PRINT") {
695 for (auto& item : s->items) {
696 if (item.kind == ",") {
697 buffer.append(10 - buffer.size() % 10, ' ');
698 } else if (item.kind != ";") {
699 Value v = evaluate(item.expr);
700 if (item.kind == "value") {
701 buffer += std::holds_alternative<std::string>(v) ?
702 std::get<std::string>(v) :
703 std::to_string(std::get<int>(v));
704 } else {
705 int n = std::get<int>(v);
706 if (n < 0) fault("ILLEGAL FUNCTION CALL");
707 std::size_t count = item.kind == "SPC" ? n :
708 std::max<std::int64_t>(0, static_cast<std::int64_t>(n) - static_cast<std::int64_t>(buffer.size()));
709 buffer.append(count, ' ');
710 }
711 }
712 }
713 if (s->items.empty() ||
714 (s->items.back().kind != ";" && s->items.back().kind != ","))
715 end_line();
716 } else if (s->kind == "INPUT" || s->kind == "READ") {
717 for (auto& target : s->targets) {
718 Address place = address(target);
719 Value value;
720 if (s->kind == "READ") {
721 if (data_position == data.size()) fault("OUT OF DATA");
722 value = data[data_position++].second;
723 if (value_type(value) != target->type) fault("TYPE MISMATCH");
724 } else {
725 for (;;) {
726 if (input_position == inputs.size()) fault("INPUT PAST END");
727 std::string text = inputs[input_position++];
728 if (target->type == 'S') {
729 value = text;
730 break;
731 }
732 static const std::regex pattern(R"(-?[0-9]+)");
733 if (std::regex_match(text, pattern)) {
734 try {
735 auto n = std::stoll(text);
736 if (n >= -32768 && n <= 32767) {
737 value = static_cast<int>(n);
738 break;
739 }
740 } catch (const std::out_of_range&) {}
741 }
742 message("?REDO FROM START");
743 }
744 }
745 store(place, value);
746 }
747 } else if (s->kind == "RESTORE") {
748 if (s->numbers.empty()) data_position = 0;
749 else {
750 int target = s->numbers[0];
751 line(target);
752 auto it = std::find_if(data.begin(), data.end(),
753 [target](const auto& item) { return item.first >= target; });
754 data_position = static_cast<std::size_t>(it - data.begin());
755 }
756 } else if (s->kind == "IF") {
757 auto index = std::get<int>(evaluate(s->expr)) != 0 ? 0u : 1u;
758 if (index < s->clauses.size()) {
759 Clause& clause = s->clauses[index];
760 next = clause.line ? line(*clause.line) : clause.body.front().get();
761 }
762 } else if (s->kind == "GOTO") {
763 next = line(s->numbers[0]);
764 } else if (s->kind == "GOSUB") {
765 next = subroutine(s->numbers[0], next);
766 } else if (s->kind == "ON") {
767 int index = std::get<int>(evaluate(s->expr));
768 if (index < 0) fault("ILLEGAL FUNCTION CALL");
769 if (index > 0 && static_cast<std::size_t>(index) <= s->numbers.size()) {
770 int destination = s->numbers[index - 1];
771 next = s->mode == "GOTO" ? line(destination) : subroutine(destination, next);
772 }
773 } else if (s->kind == "RETURN") {
774 int found = static_cast<int>(stack.size()) - 1;
775 while (found >= 0 && stack[found].kind != "GOSUB") --found;
776 if (found < 0) fault("RETURN WITHOUT GOSUB");
777 next = stack[found].position;
778 stack.resize(found);
779 } else if (s->kind == "FOR") {
780 int start = std::get<int>(evaluate(s->expressions[0]));
781 int last = std::get<int>(evaluate(s->expressions[1]));
782 int step = std::get<int>(evaluate(s->expressions[2]));
783 if (step == 0) fault("ILLEGAL FUNCTION CALL");
784 int found = search("FOR", s->name);
785 if (found >= 0) stack.resize(found);
786 variables[s->name] = start;
787 if (step > 0 ? start > last : start < last)
788 next = matching(s, "FOR", "NEXT");
789 else push({"FOR", next, s->name, last, step});
790 } else if (s->kind == "NEXT") {
791 int found = search("FOR", s->name);
792 if (found < 0) fault("NEXT WITHOUT FOR");
793 stack.resize(found + 1);
794 Frame frame = stack[found];
795 int value = checked(static_cast<std::int64_t>(std::get<int>(get(frame.name))) + frame.step);
796 variables[frame.name] = value;
797 if (frame.step > 0 ? value > frame.end : value < frame.end)
798 stack.pop_back();
799 else next = frame.position;
800 } else if (s->kind == "WHILE") {
801 if (std::get<int>(evaluate(s->expr)) != 0) push({"WHILE", s});
802 else next = matching(s, "WHILE", "WEND");
803 } else if (s->kind == "WEND") {
804 int found = search("WHILE");
805 if (found < 0) fault("WEND WITHOUT WHILE");
806 next = stack[found].position;
807 stack.resize(found);
808 } else if (s->kind == "DIM") {
809 for (auto& target : s->targets) {
810 std::string name = std::get<std::string>(target->value);
811 std::vector<int> bounds;
812 for (auto& arg : target->args) bounds.push_back(std::get<int>(evaluate(arg)));
813 if (arrays.contains(name)) fault("DUPLICATE DEFINITION");
814 std::int64_t elements = 1;
815 for (int bound : bounds) {
816 if (bound < 0) fault("ILLEGAL FUNCTION CALL");
817 elements *= bound + 1;
818 }
819 if (elements > 10000) fault("OUT OF MEMORY");
820 declare(name, bounds);
821 }
822 } else if (s->kind == "RANDOMIZE") {
823 int n = std::get<int>(evaluate(s->expr));
824 if (n < 1) fault("ILLEGAL FUNCTION CALL");
825 random_state = n;
826 } else if (s->kind == "handler") {
827 if (s->numbers[0] != 0) line(s->numbers[0]);
828 handler = s->numbers[0];
829 } else if (s->kind == "RESUME") {
830 if (!handling) fault("RESUME WITHOUT ERROR");
831 next = s->mode == "line" ? line(s->numbers[0]) :
832 s->mode == "NEXT" ? remembered->next : remembered;
833 handling = false;
834 } else if (s->kind == "STOP") {
835 message("BREAK IN LINE " + std::to_string(s->line));
836 return nullptr;
837 } else if (s->kind == "END") {
838 return nullptr;
839 }
840 return next;
841 }
842
843 void connect(const std::vector<Stmt>& body, Statement* after, int number) {
844 Statement* next = after;
845 for (auto it = body.rbegin(); it != body.rend(); ++it) {
846 Statement* s = it->get();
847 s->line = number;
848 s->next = next;
849 for (auto& clause : s->clauses) connect(clause.body, next, number);
850 next = s;
851 }
852 }
853
854 std::vector<std::string> finish() {
855 if (!buffer.empty()) end_line();
856 output.push_back("READY");
857 return output;
858 }
859
860public:
861 explicit Machine(std::vector<std::string> input) : inputs(std::move(input)) {}
862
863 std::vector<std::string> run(const std::map<int, std::string>& source) {
864 std::vector<std::pair<int, std::vector<Stmt>>> parsed;
865 for (auto& [number, text] : source) {
866 try {
867 parsed.push_back({number, Parser(tokenize(text)).parse()});
868 } catch (const SyntaxError&) {
869 message("ERROR IN LINE " + std::to_string(number) + ": SYNTAX ERROR");
870 return finish();
871 }
872 }
873 Statement* after = nullptr;
874 for (auto it = parsed.rbegin(); it != parsed.rend(); ++it) {
875 connect(it->second, after, it->first);
876 entries[it->first] = it->second.front().get();
877 after = it->second.front().get();
878 }
879 for (auto& [number, body] : parsed) {
880 for (auto& statement : body) {
881 sequence.push_back(statement.get());
882 if (statement->kind == "DATA")
883 for (auto& value : statement->values) data.emplace_back(number, value);
884 }
885 }
886 Statement* position = after;
887 while (position != nullptr) {
888 Statement* statement = position;
889 try {
890 position = execute(statement);
891 } catch (const BasicError& error) {
892 if (handler != 0 && !handling) {
893 err = error_codes.at(error.what());
894 erl = statement->line;
895 remembered = statement;
896 handling = true;
897 position = line(handler);
898 } else {
899 message("ERROR IN LINE " + std::to_string(statement->line) +
900 ": " + error.what());
901 position = nullptr;
902 }
903 }
904 }
905 return finish();
906 }
