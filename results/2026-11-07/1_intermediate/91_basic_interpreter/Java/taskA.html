75 static List<Token> tokenize(String text) {
76 List<Token> tokens = new ArrayList<>();
77 for (int p = 0; p < text.length();) {
78 char c = text.charAt(p);
79 if (c == ' ') {
80 p++;
81 continue;
82 }
83 if (c == '"') {
84 int end = text.indexOf('"', p + 1);
85 if (end < 0)
86 syntax();
87 tokens.add(new Token("S", text.substring(p, end + 1), text.substring(p + 1, end)));
88 p = end + 1;
89 } else if (digit(c)) {
90 int start = p;
91 while (p < text.length() && digit(text.charAt(p)))
92 p++;
93 String word = text.substring(start, p);
94 int n;
95 try {
96 n = Integer.parseInt(word);
97 } catch (NumberFormatException error) {
98 syntax();
99 throw error;
100 }
101 if (n > 32767)
102 syntax();
103 tokens.add(new Token("I", word, n));
104 } else if (upper(c)) {
105 if (text.startsWith("REM", p)) {
106 tokens.add(new Token("W", "REM", "REM"));
107 break;
108 }
109 int start = p;
110 while (p < text.length() && (upper(text.charAt(p)) || digit(text.charAt(p))))
111 p++;
112 if (p < text.length() && text.charAt(p) == '$')
113 p++;
114 String word = text.substring(start, p);
115 tokens.add(new Token("W", word, word));
116 } else {
117 if ("+-*/^=<>() ,;:".indexOf(c) < 0)
118 syntax();
119 String pair = p + 1 < text.length() ? text.substring(p, p + 2) : String.valueOf(c);
120 String word = List.of("<>", "<=", ">=").contains(pair) ? pair : String.valueOf(c);
121 tokens.add(new Token("P", word, word));
122 p += word.length();
123 }
124 }
125 tokens.add(new Token("E", "", ""));
126 return tokens;
127 }

175 boolean take(String word) {
176 if (!at(word))
177 return false;
178 pos++;
179 return true;
180 }
181 void expect(String word) {
182 if (!take(word))
183 syntax();
184 }
185 int number() {
186 if (!current().kind().equals("I"))
187 syntax();
188 return (int)tokens.get(pos++).value();

190 Expr integer() {
191 return integer(0);
192 }
193 Expr integer(int level) {
194 Expr e = expression(level);
195 if (!e.type.equals("I"))
196 syntax();
197 return e;
198 }
199 Expr target() {
200 Token token = current();
201 if (!token.kind().equals("W") || !variable(token.text()))
202 syntax();
203 pos++;
204 Expr result = new Expr("variable", nameType(token.text()), token.text());
205 if (take("(")) {
206 result.kind = "array";
207 result.args.add(integer());
208 if (take(","))
209 result.args.add(integer());
210 expect(")");
211 }
212 return result;
213 }
214 Expr primary() {
215 Token token = current();
216 if (token.kind().equals("I") || token.kind().equals("S")) {
217 pos++;
218 return new Expr("constant", token.kind(), token.value());
219 }
220 if (take("(")) {
221 Expr e = expression();
222 expect(")");
223 return e;
224 }
225 if (token.kind().equals("W") && variable(token.text()))
226 return target();
227 if (take("ERR"))
228 return new Expr("builtin", "I", "ERR");
229 if (take("ERL"))
230 return new Expr("builtin", "I", "ERL");
231 if (token.kind().equals("W") && signatures.containsKey(token.text())) {
232 pos++;
233 expect("(");
234 List<Expr> args = new ArrayList<>();
235 args.add(expression());
236 while (take(","))
237 args.add(expression());
238 expect(")");
239 StringBuilder types = new StringBuilder();
240 for (Expr e : args)
241 types.append(e.type);
242 if (!signatures.get(token.text()).contains(types.toString()))
243 syntax();
244 return new Expr("call", nameType(token.text()), token.text(), args);
245 }
246 syntax();
247 return null;
248 }
249 Expr expression() {
250 return expression(0);
251 }
252 Expr expression(int level) {
253 if (level == 2 && take("NOT"))
254 return new Expr("unary", "I", "NOT", List.of(integer(2)));
255 if (level == 6 && take("-"))
256 return new Expr("unary", "I", "-", List.of(integer(6)));
257 if (level == 8)
258 return primary();
259 Expr left = expression(level + 1);
260 while (operators.get(level).contains(current().text())) {
261 String op = current().text();
262 pos++;
263 Expr right = expression(level == 7 ? 6 : level + 1);
264 if (!left.type.equals(right.type) ||
265 left.type.equals("S") && level != 3 && !op.equals("+"))
266 syntax();
267 left = new Expr("binary", level == 3 ? "I" : left.type, op, List.of(left, right));
268 if (level == 7)
269 break;
270 }
271 return left;
272 }
273 Clause clause() {
274 if (current().kind().equals("I"))
275 return new Clause(number(), List.of());
276 return new Clause(null, listing(true));
277 }
278 List<Statement> listing() {
279 return listing(false);
280 }
281 List<Statement> listing(boolean inside) {
282 List<Statement> body = new ArrayList<>();
283 body.add(statement(inside));
284 while (take(":"))
285 body.add(statement(inside));
286 return body;
287 }
288 Statement statement(boolean inside) {
289 String word = current().text();
290 if (inside && List.of("IF", "FOR", "NEXT", "WHILE", "WEND", "DATA").contains(word))
291 syntax();
292 Statement s = new Statement(word);
293 for (String kind : List.of("REM", "RETURN", "WEND", "END", "STOP")) {
294 if (take(kind))
295 return s;
296 }
297 if (take("PRINT")) {
298 boolean lastValue = false;
299 while (!current().kind().equals("E") && !at(":") && !at("ELSE")) {
300 if (take(";")) {
301 s.items.add(new PrintItem(";", null));
302 lastValue = false;
303 } else if (take(",")) {
304 s.items.add(new PrintItem(",", null));
305 lastValue = false;
306 } else {
307 String kind = at("TAB") ? "TAB" : at("SPC") ? "SPC" : "value";
308 if (kind.equals("value") && lastValue)
309 syntax();
310 Expr e;
311 if (kind.equals("value"))
312 e = expression();
313 else {
314 pos++;
315 expect("(");
316 e = integer();
317 expect(")");
318 }
319 s.items.add(new PrintItem(kind, e));
320 lastValue = kind.equals("value");
321 }
322 }
323 return s;
324 }
325 if (take("INPUT") || take("READ") || take("DIM")) {
326 do {
327 Expr target = target();
328 if (word.equals("DIM") && !target.kind.equals("array"))
329 syntax();
330 s.targets.add(target);
331 } while (take(","));
332 return s;
333 }
334 if (take("DATA")) {
335 do {
336 boolean negative = take("-");
337 Token t = current();
338 if (!t.kind().equals("I") && (!t.kind().equals("S") || negative))
339 syntax();
340 pos++;
341 s.values.add(negative ? -(int)t.value() : t.value());
342 } while (take(","));
343 return s;
344 }
345 if (take("IF")) {
346 s.expr = integer();
347 expect("THEN");
348 s.clauses.add(clause());
349 if (take("ELSE"))
350 s.clauses.add(clause());
351 if (!current().kind().equals("E"))
352 syntax();
353 return s;
354 }
355 if (take("GOTO") || take("GOSUB")) {
356 s.numbers.add(number());
357 return s;
358 }
359 if (take("ON")) {
360 if (take("ERROR")) {
361 s.kind = "handler";
362 expect("GOTO");
363 s.numbers.add(number());
364 } else {
365 s.expr = integer();
366 if (take("GOTO"))
367 s.mode = "GOTO";
368 else {
369 expect("GOSUB");
370 s.mode = "GOSUB";
371 }
372 s.numbers.add(number());
373 while (take(","))
374 s.numbers.add(number());
375 }
376 return s;
377 }
378 if (take("RESUME")) {
379 if (take("NEXT"))
380 s.mode = "NEXT";
381 else if (current().kind().equals("I")) {
382 s.mode = "line";
383 s.numbers.add(number());
384 }
385 return s;
386 }
387 if (take("FOR")) {
388 Token t = current();
389 if (!t.kind().equals("W") || !variable(t.text()) || !nameType(t.text()).equals("I"))
390 syntax();
391 s.name = t.text();
392 pos++;
393 expect("=");
394 s.expressions.add(integer());
395 expect("TO");
396 s.expressions.add(integer());
397 s.expressions.add(take("STEP") ? integer() : new Expr("constant", "I", 1));
398 return s;
399 }
400 if (take("NEXT")) {
401 Token t = current();
402 if (t.kind().equals("W") && variable(t.text()) && nameType(t.text()).equals("I")) {
403 s.name = t.text();
404 pos++;
405 }
406 return s;
407 }
408 if (take("WHILE") || take("RANDOMIZE")) {
409 s.expr = integer();
410 return s;
411 }
412 if (take("RESTORE")) {
413 if (current().kind().equals("I"))
414 s.numbers.add(number());
415 return s;
416 }
417 take("LET");
418 s.kind = "assign";
419 s.target = target();
420 expect("=");
421 s.expr = expression();
422 if (!s.expr.type.equals(s.target.type))
423 syntax();
424 return s;
425 }
426 List<Statement> parse() {
427 List<Statement> result = listing();
428 if (!current().kind().equals("E"))
429 syntax();
430 return result;
431 }
432 }

471 static int checked(long n) {
472 if (n < -32768 || n > 32767)
473 fault("OVERFLOW");
474 return (int)n;
475 }
476 Object valueOf(String name) {
477 return variables.getOrDefault(name, nameType(name).equals("S") ? "" : 0);
478 }
479 void declare(String name, int[] bounds) {
480 int count = 1;
481 for (int b : bounds)
482 count *= b + 1;
483 Object[] values = new Object[count];
484 Arrays.fill(values, nameType(name).equals("S") ? "" : 0);
485 arrays.put(name, new ArrayDef(bounds, values));
486 }
487 Address address(Expr target) {
488 String name = (String)target.value;
489 if (target.kind.equals("variable"))
490 return new Address(name, null);
491 int[] indices = new int[target.args.size()];
492 for (int i = 0; i < indices.length; i++)
493 indices[i] = (int)evaluate(target.args.get(i));
494 if (!arrays.containsKey(name)) {
495 int[] bounds = new int[indices.length];
496 Arrays.fill(bounds, 10);
497 declare(name, bounds);
498 }
499 ArrayDef array = arrays.get(name);
500 if (indices.length != array.bounds().length)
501 fault("SUBSCRIPT OUT OF RANGE");
502 int index = 0;
503 for (int i = 0; i < indices.length; i++) {
504 int value = indices[i];
505 if (value < 0 || value > array.bounds()[i])
506 fault("SUBSCRIPT OUT OF RANGE");
507 index = index * (array.bounds()[i] + 1) + value;
508 }
509 return new Address(name, index);
510 }
511 Object load(Address a) {
512 if (a.index() == null)
513 return valueOf(a.name());
514 return arrays.get(a.name()).values()[a.index()];
515 }
516 void store(Address a, Object value) {
517 if (a.index() == null)
518 variables.put(a.name(), value);
519 else
520 arrays.get(a.name()).values()[a.index()] = value;
521 }
522 Object evaluate(Expr e) {
523 Object op = e.value;
524 if (e.kind.equals("constant"))
525 return op;
526 if (e.kind.equals("variable") || e.kind.equals("array"))
527 return load(address(e));
528 if (e.kind.equals("builtin"))
529 return op.equals("ERR") ? err : erl;
530 if (e.kind.equals("unary")) {
531 int n = (int)evaluate(e.args.get(0));
532 return op.equals("NOT") ? ~n : checked(-(long)n);
533 }
534 List<Object> args = new ArrayList<>();
535 for (Expr child : e.args)
536 args.add(evaluate(child));
537 if (e.kind.equals("call"))
538 return call((String)op, args);
539 Object left = args.get(0), right = args.get(1);
540 if (operators.get(3).contains(op)) {
541 int cmp = left instanceof String ? ((String)left).compareTo((String)right)
542 : Integer.compare((int)left, (int)right);
543 boolean success = switch ((String)op) {
544 case "=" -> cmp == 0;
545 case "<>" -> cmp != 0;
546 case "<" -> cmp < 0;
547 case ">" -> cmp > 0;
548 case "<=" -> cmp <= 0;
549 case ">=" -> cmp >= 0;
550 default -> false;
551 };
552 return success ? -1 : 0;
553 }
554 if (e.type.equals("S")) {
555 String s = (String)left + (String)right;
556 if (s.length() > 255)
557 fault("STRING TOO LONG");
558 return s;
559 }
560 int a = (int)left, b = (int)right;
561 switch ((String)op) {
562 case "+":
563 return checked((long)a + b);
564 case "-":
565 return checked((long)a - b);
566 case "*":
567 return checked((long)a * b);
568 case "/":
569 if (b == 0)
570 fault("DIVISION BY ZERO");
571 return checked(a / b);
572 case "MOD":
573 if (b == 0)
574 fault("DIVISION BY ZERO");
575 return a - b * (a / b);
576 case "^":
577 if (b < 0)
578 fault("ILLEGAL FUNCTION CALL");
579 if (Math.abs(a) > 1 && b > 15)
580 fault("OVERFLOW");
581 long total = 1;
582 for (int i = 0; i < b; i++)
583 total = checked(total * a);
584 return (int)total;
585 case "AND":
586 return a & b;
587 case "OR":
588 return a | b;
589 }
590 fault("ILLEGAL FUNCTION CALL");
591 return 0;
592 }
593 Object call(String name, List<Object> args) {
594 Object a = args.get(0), b = args.size() > 1 ? args.get(1) : null,
595 c = args.size() > 2 ? args.get(2) : null;
596 switch (name) {
597 case "ABS":
598 return checked(Math.abs((long)(int)a));
599 case "SGN":
600 return Integer.compare((int)a, 0);
601 case "INT":
602 return a;
603 case "SQR":
604 if ((int)a < 0)
605 fault("ILLEGAL FUNCTION CALL");
606 return (int)Math.sqrt((int)a);
607 case "RND":
608 int n = (int)a;
609 if (n < 1)
610 fault("ILLEGAL FUNCTION CALL");
611 random = (int)(48271L * random % 2147483647L);
612 return random % n + 1;
613 case "LEN":
614 return ((String)a).length();
615 case "ASC":
616 if (((String)a).isEmpty())
617 fault("ILLEGAL FUNCTION CALL");
618 return (int)((String)a).charAt(0);
619 case "VAL":
620 Matcher m = Pattern.compile("^ *([+-]?[0-9]+)").matcher((String)a);
621 if (!m.find())
622 return 0;
623 long number;
624 try {
625 number = Long.parseLong(m.group(1));
626 } catch (NumberFormatException e) {
627 fault("OVERFLOW");
628 return 0;
629 }
630 return checked(number);
631 case "CHR$":
632 int code = (int)a;
633 if (code < 32 || code > 126)
634 fault("ILLEGAL FUNCTION CALL");
635 return Character.toString(code);
636 case "STR$":
637 return Integer.toString((int)a);
638 case "LEFT$":
639 int count = (int)b;
640 if (count < 0)
641 fault("ILLEGAL FUNCTION CALL");
642 return ((String)a).substring(0, Math.min(count, ((String)a).length()));
643 case "RIGHT$":
644 int k = (int)b;
645 if (k < 0)
646 fault("ILLEGAL FUNCTION CALL");
647 String word = (String)a;
648 return word.substring(Math.max(0, word.length() - k));
649 case "MID$":
650 int start = (int)b;
651 Integer len = c == null ? null : (int)c;
652 if (start < 1 || len != null && len < 0)
653 fault("ILLEGAL FUNCTION CALL");
654 String text = (String)a;
655 int begin = Math.min(start - 1, text.length());
656 int end = len == null ? text.length() : Math.min(text.length(), begin + len);
657 return text.substring(begin, end);
658 case "INSTR":
659 return ((String)a).indexOf((String)b) + 1;
660 }
661 fault("ILLEGAL FUNCTION CALL");
662 return 0;
663 }

685 Statement matching(Statement s, String opening, String closing) {
686 int start = sequence.indexOf(s), depth = 0;
687 for (int i = start + 1; i < sequence.size(); i++) {
688 Statement t = sequence.get(i);

691 else if (t.kind.equals(closing)) {

694 depth--;
695 }

705 Statement execute(Statement s) {
706 Statement following = s.next;
707 switch (s.kind) {
708 case "assign": {
709 Address a = address(s.target);
710 store(a, evaluate(s.expr));
711 break;
712 }
713 case "PRINT": {
714 for (PrintItem item : s.items) {
715 if (item.kind().equals(",")) {
716 buffer.append(" ".repeat(10 - buffer.length() % 10));
717 } else if (!item.kind().equals(";")) {
718 Object value = evaluate(item.expr());
719 if (item.kind().equals("value"))
720 buffer.append(value);
721 else {
722 int n = (int)value;
723 if (n < 0)
724 fault("ILLEGAL FUNCTION CALL");
725 buffer.append(" ".repeat(
726 item.kind().equals("SPC") ? n : Math.max(0, n - buffer.length())));
727 }
728 }
729 }
730 if (s.items.isEmpty() || !List.of(";", ",").contains(s.items.getLast().kind()))
731 endLine();
732 break;
733 }
734 case "INPUT", "READ": {
735 for (Expr target : s.targets) {
736 Address a = address(target);
737 Object value;
738 if (s.kind.equals("READ")) {
739 if (dataPos >= data.size())
740 fault("OUT OF DATA");
741 value = data.get(dataPos++).value();
742 if (!valueType(value).equals(target.type))
743 fault("TYPE MISMATCH");
744 } else {
745 while (true) {
746 if (inputPos >= inputs.size())
747 fault("INPUT PAST END");
748 String text = inputs.get(inputPos++);
749 if (target.type.equals("S")) {
750 value = text;
751 break;
752 }
753 Integer n = null;
754 try {
755 n = Integer.parseInt(text);
756 } catch (NumberFormatException ignored) {
757 }
758 if (n != null && n >= -32768 && n <= 32767 &&
759 text.matches("-?[0-9]+")) {
760 value = n;
761 break;
762 }
763 message("?REDO FROM START");
764 }
765 }
766 store(a, value);
767 }
768 break;
769 }
770 case "RESTORE": {
771 if (s.numbers.isEmpty())
772 dataPos = 0;
773 else {
774 int number = s.numbers.get(0);
775 line(number);
776 dataPos = data.size();
777 for (int i = 0; i < data.size(); i++) {
778 if (data.get(i).line() >= number) {
779 dataPos = i;
780 break;
781 }
782 }
783 }
784 break;
785 }
786 case "IF": {
787 int index = (int)evaluate(s.expr) == 0 ? 1 : 0;
788 if (index < s.clauses.size()) {
789 Clause chosen = s.clauses.get(index);
790 following = chosen.line() == null ? chosen.body().get(0) : line(chosen.line());
791 }
792 break;
793 }
794 case "GOTO":
795 following = line(s.numbers.get(0));
796 break;
797 case "GOSUB":
798 following = subroutine(s.numbers.get(0), following);
799 break;
800 case "ON": {
801 int ix = (int)evaluate(s.expr);
802 if (ix < 0)
803 fault("ILLEGAL FUNCTION CALL");
804 if (ix >= 1 && ix <= s.numbers.size()) {
805 following = s.mode.equals("GOTO")
806 ? line(s.numbers.get(ix - 1))
807 : subroutine(s.numbers.get(ix - 1), following);
808 }
809 break;
810 }
811 case "RETURN": {
812 int ix = -1;
813 for (int i = stack.size() - 1; i >= 0; i--) {
814 if (stack.get(i).kind().equals("GOSUB")) {
815 ix = i;
816 break;
817 }
818 }
819 if (ix < 0)
820 fault("RETURN WITHOUT GOSUB");
821 following = stack.get(ix).position();
822 stack.subList(ix, stack.size()).clear();
823 break;
824 }
825 case "FOR": {
826 int start = (int)evaluate(s.expressions.get(0));
827 int limit = (int)evaluate(s.expressions.get(1));
828 int step = (int)evaluate(s.expressions.get(2));
829 if (step == 0)
830 fault("ILLEGAL FUNCTION CALL");
831 int ix = search("FOR", s.name);
832 if (ix >= 0)
833 stack.subList(ix, stack.size()).clear();
834 variables.put(s.name, start);
835 if (step > 0 && start > limit || step < 0 && start < limit)
836 following = matching(s, "FOR", "NEXT");
837 else
838 push(new Frame("FOR", following, s.name, limit, step));
839 break;
840 }
841 case "NEXT": {
842 int ix = search("FOR", s.name);
843 if (ix < 0)
844 fault("NEXT WITHOUT FOR");
845 stack.subList(ix + 1, stack.size()).clear();
846 Frame f = stack.get(ix);
847 int value = checked((long)(int)valueOf(f.name()) + f.step());
848 variables.put(f.name(), value);
849 if (f.step() > 0 && value > f.limit() || f.step() < 0 && value < f.limit())
850 stack.remove(ix);
851 else
852 following = f.position();
853 break;
854 }
855 case "WHILE": {
856 if ((int)evaluate(s.expr) == 0)
857 following = matching(s, "WHILE", "WEND");
858 else
859 push(new Frame("WHILE", s));
860 break;
861 }
862 case "WEND": {
863 int ix = search("WHILE", "");
864 if (ix < 0)
865 fault("WEND WITHOUT WHILE");
866 following = stack.get(ix).position();
867 stack.subList(ix, stack.size()).clear();
868 break;
869 }
870 case "DIM": {
871 for (Expr target : s.targets) {
872 String name = (String)target.value;
873 int[] bounds = new int[target.args.size()];
874 for (int i = 0; i < bounds.length; i++)
875 bounds[i] = (int)evaluate(target.args.get(i));
876 if (arrays.containsKey(name))
877 fault("DUPLICATE DEFINITION");
878 int count = 1;
879 for (int bound : bounds) {
880 if (bound < 0)
881 fault("ILLEGAL FUNCTION CALL");
882 count *= bound + 1;
883 }
884 if (count > 10000)
885 fault("OUT OF MEMORY");
886 declare(name, bounds);
887 }
888 break;
889 }
890 case "RANDOMIZE": {
891 int n = (int)evaluate(s.expr);
892 if (n < 1)
893 fault("ILLEGAL FUNCTION CALL");
894 random = n;
895 break;
896 }
897 case "handler": {
898 if (s.numbers.get(0) != 0)
899 line(s.numbers.get(0));
900 handler = s.numbers.get(0);
901 break;
902 }
903 case "RESUME": {
904 if (!handling)
905 fault("RESUME WITHOUT ERROR");
906 following = s.mode.equals("line") ? line(s.numbers.get(0))
907 : s.mode.equals("NEXT") ? remembered.next
908 : remembered;
909 handling = false;
910 break;
911 }
912 case "STOP":
913 message("BREAK IN LINE " + s.line);
914 return null;
915 case "END":
916 return null;
917 }
918 return following;
919 }
920 void connect(List<Statement> body, Statement after, int number) {
921 Statement following = after;
922 for (int i = body.size() - 1; i >= 0; i--) {
923 Statement s = body.get(i);
924 s.line = number;
925 s.next = following;
926 for (Clause c : s.clauses)
927 connect(c.body(), following, number);
928 following = s;
929 }
930 }
931 List<String> finish() {
932 if (!buffer.isEmpty())
933 endLine();
934 output.add("READY");
935 return output;
936 }
937 List<String> run(Map<Integer, String> source) {
938 List<ProgramLine> parsed = new ArrayList<>();
939 List<Integer> numbers = new ArrayList<>(source.keySet());
940 numbers.sort(Comparator.naturalOrder());
941 for (int number : numbers) {
942 try {
943 parsed.add(
944 new ProgramLine(number, new Parser(tokenize(source.get(number))).parse()));
945 } catch (SyntaxFault failure) {
946 message("ERROR IN LINE " + number + ": SYNTAX ERROR");
947 return finish();
948 }
949 }
950 Statement after = null;
951 for (int i = parsed.size() - 1; i >= 0; i--) {
952 ProgramLine e = parsed.get(i);
953 connect(e.body(), after, e.number());
954 entries.put(e.number(), e.body().get(0));
955 after = e.body().get(0);
956 }
957 for (ProgramLine e : parsed) {
958 for (Statement s : e.body()) {
959 sequence.add(s);
960 if (s.kind.equals("DATA")) {
961 for (Object value : s.values)
962 data.add(new DataValue(s.line, value));
963 }
964 }
965 }
966 Statement position = after;
967 while (position != null) {
968 Statement s = position;
969 try {
970 position = execute(s);
971 } catch (BasicFault failure) {
972 if (handler != 0 && !handling) {
973 err = errorCodes.getOrDefault(failure.getMessage(), 0);
974 erl = s.line;
975 remembered = s;
976 handling = true;
977 try {
978 position = line(handler);
979 } catch (BasicFault secondary) {
980 message("ERROR IN LINE " + s.line + ": " + secondary.getMessage());
981 position = null;
982 }
983 } else {
984 message("ERROR IN LINE " + s.line + ": " + failure.getMessage());
985 position = null;
986 }
987 }
988 }
989 return finish();
990 }

995 int count = Integer.parseInt(reader.readLine());

1009 }

1019 }
