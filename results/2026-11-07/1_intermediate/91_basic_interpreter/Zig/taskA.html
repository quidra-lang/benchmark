61fn lex(s: S) ![]Token {
62 var ts: std.ArrayList(Token) = .empty;
63 var i: usize = 0;
64 while (i < s.len) {
65 if (s[i] == ' ') {
66 i += 1;
67 continue;
68 }
69 const start = i;
70 var k: TokenKind = .sym;
71 var value: S = "";
72 if (s[i] == '"') {
73 i += 1;
74 const begin = i;
75 while (i < s.len and s[i] != '"') i += 1;
76 if (i == s.len) return syntax();
77 value = s[begin..i];
78 i += 1;
79 k = .str;
80 } else if (std.ascii.isDigit(s[i])) {
81 while (i < s.len and std.ascii.isDigit(s[i])) i += 1;
82 const n = std.fmt.parseInt(i64, s[start..i], 10) catch return syntax();
83 if (n > 32767) return syntax();
84 k = .num;
85 } else if (std.ascii.isUpper(s[i])) {
86 if (std.mem.startsWith(u8, s[i..], "REM")) {
87 try ts.append(A, .{ .k = .word, .s = "REM" });
88 break;
89 }
90 while (i < s.len and (std.ascii.isUpper(s[i]) or std.ascii.isDigit(s[i]))) i += 1;
91 if (i < s.len and s[i] == '$') i += 1;
92 k = .word;
93 } else if (i + 1 < s.len and (eq(s[i .. i + 2], "<>") or eq(s[i .. i + 2], "<=") or eq(s[i .. i + 2], ">="))) i += 2 else if (std.mem.indexOfScalar(u8, "+-*/^=<>() ,;:", s[i]) != null) i += 1 else return syntax();
94 if (k != .str) value = s[start..i];
95 try ts.append(A, .{ .k = k, .s = value });
96 }
97 return ts.toOwnedSlice(A);
98}

100fn varId(s: S) ?usize {
101 if (s.len == 0) return null;
102 const str = s[s.len - 1] == '$';
103 const v = if (str) s[0 .. s.len - 1] else s;
104 if (v.len < 1 or v.len > 2 or !std.ascii.isUpper(v[0]) or (v.len == 2 and !std.ascii.isDigit(v[1]))) return null;
105 return @as(usize, v[0] - 'A') * 11 + (if (v.len == 1) @as(usize, 0) else v[1] - '0' + 1) + (if (str) @as(usize, 286) else 0);
106}

133 fn peek(p: *Parser) S {

136 fn accept(p: *Parser, s: S) bool {
137 if (!eq(p.peek(), s)) return false;
138 p.i += 1;

141 fn expect(p: *Parser, s: S) !void {
142 if (!p.accept(s)) return syntax();
143 }
144 fn num(p: *Parser) !usize {
145 if (p.i == p.ts.len or p.ts[p.i].k != .num) return syntax();
146 const n = try std.fmt.parseInt(usize, p.peek(), 10);
147 p.i += 1;
148 return n;

150 fn integerExpr(p: *Parser) !*Expr {
151 const e = try p.parse(1);
152 if (e.string) return syntax();
153 return e;
154 }
155 fn target(p: *Parser, scalar: bool, integer_only: bool) !*Expr {
156 const s = p.peek();
157 const id = varId(s) orelse return syntax();
158 if (integer_only and id >= 286) return syntax();
159 p.i += 1;
160 var e: Expr = .{ .k = .variable, .s = s, .id = id, .string = id >= 286 };
161 if (!scalar and p.accept("(")) {
162 var args: std.ArrayList(*Expr) = .empty;
163 try args.append(A, try p.integerExpr());
164 if (p.accept(",")) try args.append(A, try p.integerExpr());
165 try p.expect(")");
166 e.k = .array;
167 e.args = try args.toOwnedSlice(A);
168 }
169 return expression(e);
170 }
171 fn primary(p: *Parser) anyerror!*Expr {
172 if (p.accept("-")) {
173 const x = try p.parse(7);
174 if (x.string) return syntax();
175 return expression(.{ .k = .unary, .s = "-", .args = try A.dupe(*Expr, &.{x}) });
176 }
177 if (p.accept("(")) {
178 const x = try p.parse(1);
179 try p.expect(")");
180 return x;
181 }
182 if (p.i == p.ts.len) return syntax();
183 const t = p.ts[p.i];
184 if (t.k == .num) {
185 p.i += 1;
186 return expression(.{ .k = .value, .v = integer(try std.fmt.parseInt(i64, t.s, 10)) });
187 }
188 if (t.k == .str) {
189 p.i += 1;
190 return expression(.{ .k = .value, .string = true, .v = .{ .string = true, .s = t.s } });
191 }
192 if (eq(t.s, "ERR") or eq(t.s, "ERL")) {
193 p.i += 1;
194 return expression(.{ .k = .special, .s = t.s });
195 }
196 if (varId(t.s) != null) return p.target(false, false);
197 if (functionName(t.s)) {
198 p.i += 1;
199 try p.expect("(");
200 var args: std.ArrayList(*Expr) = .empty;
201 while (true) {
202 try args.append(A, try p.parse(1));
203 if (!p.accept(",")) break;
204 }
205 try p.expect(")");
206 const s = t.s;
207 const count: usize = if (eq(s, "LEFT$") or eq(s, "RIGHT$") or eq(s, "INSTR")) 2 else 1;
208 if (eq(s, "MID$")) {
209 if (args.items.len != 2 and args.items.len != 3) return syntax();
210 } else if (args.items.len != count) return syntax();
211 for (args.items, 0..) |a, i| {
212 const want = if (eq(s, "LEN") or eq(s, "ASC") or eq(s, "VAL") or eq(s, "INSTR")) true else if (eq(s, "LEFT$") or eq(s, "RIGHT$") or eq(s, "MID$")) i == 0 else false;
213 if (a.string != want) return syntax();
214 }
215 return expression(.{ .k = .call, .s = s, .string = eq(s, "CHR$") or eq(s, "STR$") or eq(s, "LEFT$") or eq(s, "RIGHT$") or eq(s, "MID$"), .args = try args.toOwnedSlice(A) });
216 }
217 return syntax();
218 }
219 fn parse(p: *Parser, min: u8) anyerror!*Expr {
220 var n: *Expr = undefined;
221 if (min <= 3 and p.accept("NOT")) {
222 const x = try p.parse(3);
223 if (x.string) return syntax();
224 n = try expression(.{ .k = .unary, .s = "NOT", .args = try A.dupe(*Expr, &.{x}) });
225 } else n = try p.primary();
226 while (true) {
227 const op = p.peek();
228 const pr: u8 = if (eq(op, "OR")) 1 else if (eq(op, "AND")) 2 else if (eq(op, "=") or eq(op, "<>") or eq(op, "<") or eq(op, ">") or eq(op, "<=") or eq(op, ">=")) 4 else if (eq(op, "+") or eq(op, "-")) 5 else if (eq(op, "*") or eq(op, "/") or eq(op, "MOD")) 6 else if (eq(op, "^")) 8 else 0;
229 if (pr == 0 or pr < min) break;
230 p.i += 1;
231 const r = try p.parse(if (pr == 8) 7 else pr + 1);
232 if ((pr == 4 or eq(op, "+")) and n.string != r.string) return syntax();
233 if (pr != 4 and !eq(op, "+") and (n.string or r.string)) return syntax();
234 n = try expression(.{ .k = .binary, .s = op, .string = eq(op, "+") and n.string, .args = try A.dupe(*Expr, &.{ n, r }) });
235 }
236 return n;
237 }
238 fn ended(p: *Parser) bool {
239 return p.i == p.ts.len or eq(p.peek(), ":") or eq(p.peek(), "ELSE");
240 }
241 fn list(p: *Parser, clause: bool) anyerror![]*Stmt {
242 var a: std.ArrayList(*Stmt) = .empty;
243 while (true) {
244 try a.append(A, try p.stmt(clause));
245 if (!p.accept(":")) break;
246 }
247 return a.toOwnedSlice(A);

249 fn stmt(p: *Parser, clause: bool) anyerror!*Stmt {
250 const kind = p.peek();
251 if (kind.len == 0) return syntax();
252 p.i += 1;
253 var s: Stmt = .{ .kind = kind, .line = p.line };
254 var args: std.ArrayList(*Expr) = .empty;
255 var targets: std.ArrayList(*Expr) = .empty;
256 var nums: std.ArrayList(usize) = .empty;
257 if (eq(kind, "REM")) {} else if (eq(kind, "LET") or varId(kind) != null) {
258 s.kind = "LET";
259 if (!eq(kind, "LET")) p.i -= 1;
260 const target = try p.target(false, false);
261 try p.expect("=");
262 const value = try p.parse(1);
263 if (target.string != value.string) return syntax();
264 try targets.append(A, target);
265 try args.append(A, value);
266 } else if (eq(kind, "PRINT")) {
267 var need_separator = false;
268 while (!p.ended()) {
269 if (p.accept(";") or p.accept(",")) {
270 try args.append(A, try expression(.{ .k = .special, .s = p.ts[p.i - 1].s }));
271 need_separator = false;
272 } else {
273 if (eq(p.peek(), "TAB") or eq(p.peek(), "SPC")) {
274 const op = p.peek();
275 p.i += 1;
276 try p.expect("(");
277 const e = try p.integerExpr();
278 try p.expect(")");
279 try args.append(A, try expression(.{ .k = .special, .s = op, .args = try A.dupe(*Expr, &.{e}) }));
280 need_separator = false;
281 } else {
282 if (need_separator) return syntax();
283 try args.append(A, try p.parse(1));
284 need_separator = true;
285 }
286 }
287 }
288 } else if (eq(kind, "INPUT") or eq(kind, "READ") or eq(kind, "DIM")) {
289 while (true) {
290 const e = try p.target(false, false);
291 if (eq(kind, "DIM") and e.k != .array) return syntax();
292 try targets.append(A, e);
293 if (!p.accept(",")) break;
294 }
295 } else if (eq(kind, "DATA")) {
296 if (clause) return syntax();
297 while (true) {
298 const neg = p.accept("-");
299 if (p.i == p.ts.len) return syntax();
300 const t = p.ts[p.i];
301 if (t.k != .num and t.k != .str) return syntax();
302 if (neg and t.k != .num) return syntax();
303 p.i += 1;
304 try args.append(A, try expression(.{ .k = .value, .string = t.k == .str, .v = if (t.k == .str) .{ .string = true, .s = t.s } else integer((if (neg) @as(i64, -1) else 1) * try std.fmt.parseInt(i64, t.s, 10)) }));
305 if (!p.accept(",")) break;
306 }
307 } else if (eq(kind, "IF")) {
308 if (clause) return syntax();
309 try args.append(A, try p.integerExpr());
310 try p.expect("THEN");
311 if (p.i < p.ts.len and p.ts[p.i].k == .num) s.yesline = try p.num() else s.yes = try p.list(true);
312 if (p.accept("ELSE")) {
313 if (p.i < p.ts.len and p.ts[p.i].k == .num) s.noline = try p.num() else s.no = try p.list(true);
314 }
315 if (p.i != p.ts.len) return syntax();
316 } else if (eq(kind, "GOTO") or eq(kind, "GOSUB")) try nums.append(A, try p.num()) else if (eq(kind, "ON")) {
317 if (p.accept("ERROR")) {
318 s.kind = "ERROR";
319 try p.expect("GOTO");
320 try nums.append(A, try p.num());
321 } else {
322 try args.append(A, try p.integerExpr());
323 if (p.accept("GOTO")) s.option = "GOTO" else if (p.accept("GOSUB")) s.option = "GOSUB" else return syntax();
324 while (true) {
325 try nums.append(A, try p.num());
326 if (!p.accept(",")) break;
327 }
328 }
329 } else if (eq(kind, "RESUME")) {
330 if (p.accept("NEXT")) s.option = "NEXT" else if (p.i < p.ts.len and p.ts[p.i].k == .num) try nums.append(A, try p.num());
331 } else if (eq(kind, "FOR")) {
332 if (clause) return syntax();
333 try targets.append(A, try p.target(true, true));
334 try p.expect("=");
335 try args.append(A, try p.integerExpr());
336 try p.expect("TO");
337 try args.append(A, try p.integerExpr());
338 if (p.accept("STEP")) try args.append(A, try p.integerExpr()) else try args.append(A, try expression(.{ .k = .value, .v = integer(1) }));
339 } else if (eq(kind, "NEXT")) {
340 if (clause) return syntax();
341 if (!p.ended()) try targets.append(A, try p.target(true, true));
342 } else if (eq(kind, "WHILE")) {
343 if (clause) return syntax();
344 try args.append(A, try p.integerExpr());
345 } else if (eq(kind, "WEND")) {
346 if (clause) return syntax();
347 } else if (eq(kind, "RESTORE")) {
348 if (!p.ended()) try nums.append(A, try p.num());
349 } else if (eq(kind, "RANDOMIZE")) try args.append(A, try p.integerExpr()) else if (!eq(kind, "RETURN") and !eq(kind, "END") and !eq(kind, "STOP")) return syntax();
350 if (!p.ended()) return syntax();
351 s.args = try args.toOwnedSlice(A);
352 s.targets = try targets.toOwnedSlice(A);
353 s.nums = try nums.toOwnedSlice(A);
354 return statement(s);
355 }

382 fn create(m: *Machine, id: usize, bounds: [2]usize, dim: usize) !void {
383 const count = (bounds[0] + 1) * (if (dim == 2) bounds[1] + 1 else 1);
384 if (count > 10000) return runtime(7);
385 const values = try A.alloc(Value, count);
386 @memset(values, .{ .string = id >= 286 });
387 m.arrays[id] = .{ .bounds = bounds, .dim = dim, .values = values };
388 }
389 fn reference(m: *Machine, e: *Expr) anyerror!*Value {
390 if (e.k == .variable) return &m.vars[e.id];
391 var indices: [2]i64 = .{ 0, 0 };
392 for (e.args, 0..) |a, i| indices[i] = (try m.eval(a)).n;
393 if (m.arrays[e.id] == null) try m.create(e.id, .{ 10, 10 }, e.args.len);
394 const array = &m.arrays[e.id].?;
395 if (array.dim != e.args.len) return runtime(9);
396 for (indices[0..e.args.len], 0..) |v, i| if (v < 0 or v > @as(i64, @intCast(array.bounds[i]))) return runtime(9);
397 const index: usize = @intCast(if (e.args.len == 1) indices[0] else indices[0] * @as(i64, @intCast(array.bounds[1] + 1)) + indices[1]);
398 return &array.values[index];
399 }
400 fn eval(m: *Machine, e: *Expr) anyerror!Value {
401 switch (e.k) {
402 .value => return e.v,
403 .variable, .array => return (try m.reference(e)).*,
404 .special => return integer(if (eq(e.s, "ERR")) m.err else m.erl),
405 .unary => {
406 const v = (try m.eval(e.args[0])).n;
407 return integer(if (eq(e.s, "NOT")) -1 - v else try checked(-v));
408 },
409 .call => {
410 const vals = try A.alloc(Value, e.args.len);
411 for (e.args, 0..) |a, i| vals[i] = try m.eval(a);
412 return m.function(e.s, vals);
413 },
414 .binary => {
415 const a = try m.eval(e.args[0]);
416 const b = try m.eval(e.args[1]);
417 const op = e.s;
418 if (eq(op, "+") and a.string) {
419 const s = try fmt("{s}{s}", .{ a.s, b.s });
420 if (s.len > 255) return runtime(15);
421 return .{ .string = true, .s = s };
422 }
423 if (eq(op, "AND")) return integer(a.n & b.n);
424 if (eq(op, "OR")) return integer(a.n | b.n);
425 if (eq(op, "+")) return integer(try checked(a.n + b.n));
426 if (eq(op, "-")) return integer(try checked(a.n - b.n));
427 if (eq(op, "*")) return integer(try checked(a.n * b.n));
428 if (eq(op, "/") or eq(op, "MOD")) {
429 if (b.n == 0) return runtime(11);
430 return integer(if (eq(op, "MOD")) @rem(a.n, b.n) else try checked(@divTrunc(a.n, b.n)));
431 }
432 if (eq(op, "^")) {
433 if (b.n < 0) return runtime(5);
434 var n: i64 = 1;
435 var base = a.n;
436 var exp = b.n;
437 while (exp > 0) {
438 if (@mod(exp, 2) == 1) n = try checked(n * base);
439 exp = @divTrunc(exp, 2);
440 if (exp > 0) base = try checked(base * base);
441 }
442 return integer(n);
443 }
444 const order = if (a.string) std.mem.order(u8, a.s, b.s) else if (a.n < b.n) std.math.Order.lt else if (a.n > b.n) std.math.Order.gt else std.math.Order.eq;
445 const yes = if (eq(op, "=")) order == .eq else if (eq(op, "<>")) order != .eq else if (eq(op, "<")) order == .lt else if (eq(op, "<=")) order != .gt else if (eq(op, ">")) order == .gt else order != .lt;
446 return integer(if (yes) -1 else 0);
447 },
448 }
449 }
450 fn function(m: *Machine, s: S, a: []Value) !Value {
451 const x = a[0];
452 if (eq(s, "ABS")) return integer(try checked(@intCast(@abs(x.n))));
453 if (eq(s, "SGN")) return integer(if (x.n < 0) -1 else if (x.n > 0) 1 else 0);
454 if (eq(s, "INT")) return x;
455 if (eq(s, "SQR")) {
456 if (x.n < 0) return runtime(5);
457 var n: i64 = 0;
458 while ((n + 1) * (n + 1) <= x.n) n += 1;
459 return integer(n);
460 }
461 if (eq(s, "RND")) {
462 if (x.n < 1) return runtime(5);
463 m.random = m.random * 48271 % 2147483647;
464 return integer(@as(i64, @intCast(m.random % @as(u64, @intCast(x.n)))) + 1);
465 }
466 if (eq(s, "LEN")) return integer(@intCast(x.s.len));
467 if (eq(s, "ASC")) {
468 if (x.s.len == 0) return runtime(5);
469 return integer(x.s[0]);
470 }
471 if (eq(s, "VAL")) {
472 var u = std.mem.trimStart(u8, x.s, " ");
473 var sign: i64 = 1;
474 if (u.len > 0 and (u[0] == '-' or u[0] == '+')) {
475 if (u[0] == '-') sign = -1;
476 u = u[1..];
477 }
478 var n: i64 = 0;
479 for (u) |c| {
480 if (!std.ascii.isDigit(c)) break;
481 n = n * 10 + c - '0';
482 }
483 return integer(try checked(sign * n));
484 }
485 if (eq(s, "CHR$")) {
486 if (x.n < 32 or x.n > 126) return runtime(5);
487 return .{ .string = true, .s = try fmt("{c}", .{@as(u8, @intCast(x.n))}) };
488 }
489 if (eq(s, "STR$")) return .{ .string = true, .s = try fmt("{d}", .{x.n}) };
490 if (eq(s, "INSTR")) return integer(if (std.mem.indexOf(u8, x.s, a[1].s)) |i| @intCast(i + 1) else 0);
491 if (eq(s, "LEFT$") or eq(s, "RIGHT$")) {
492 if (a[1].n < 0) return runtime(5);
493 const n = @min(x.s.len, @as(usize, @intCast(a[1].n)));
494 return .{ .string = true, .s = if (eq(s, "LEFT$")) x.s[0..n] else x.s[x.s.len - n ..] };
495 }
496 if (a[1].n < 1 or (a.len == 3 and a[2].n < 0)) return runtime(5);
497 const begin = @min(x.s.len, @as(usize, @intCast(a[1].n - 1)));
498 const end = if (a.len == 3) @min(x.s.len, begin + @as(usize, @intCast(a[2].n))) else x.s.len;
499 return .{ .string = true, .s = x.s[begin..end] };
500 }
501 fn goto(m: *Machine, n: usize) !*Stmt {
502 return m.lines.get(n) orelse runtime(8);
503 }
504 fn push(m: *Machine, f: Frame) !void {
505 if (m.stack.items.len == 100) return runtime(7);
506 try m.stack.append(A, f);
507 }
508 fn search(m: *Machine, kind: FrameKind, id: ?usize) ?usize {
509 var i = m.stack.items.len;
510 while (i > 0) {
511 i -= 1;
512 const f = m.stack.items[i];
513 if (f.k == .gosub) break;
514 if (f.k == kind and (id == null or f.id == id.?)) return i;
515 }
516 return null;
517 }
518 fn matching(m: *Machine, s: *Stmt, open: S, close: S) !?*Stmt {
519 var depth: usize = 0;
520 for (m.program[s.index + 1 ..]) |q| {
521 if (eq(q.kind, open)) depth += 1;
522 if (eq(q.kind, close)) {
523 if (depth == 0) return q.next;
524 depth -= 1;
525 }
526 }
527 return runtime(if (eq(open, "FOR")) 26 else 29);
528 }
529 fn execute(m: *Machine, s: *Stmt) anyerror!?*Stmt {
530 const c = s.kind;
531 var next = s.next;
532 if (eq(c, "LET")) {
533 const target = try m.reference(s.targets[0]);
534 const value = try m.eval(s.args[0]);
535 target.* = value;
536 } else if (eq(c, "PRINT")) {
537 for (s.args) |e| {
538 if (e.k == .special and !eq(e.s, "ERR") and !eq(e.s, "ERL")) {
539 var count: usize = 0;
540 if (eq(e.s, ",")) count = 10 - pending.items.len % 10 else if (eq(e.s, "TAB") or eq(e.s, "SPC")) {
541 const n = (try m.eval(e.args[0])).n;
542 if (n < 0) return runtime(5);
543 count = if (eq(e.s, "SPC")) @intCast(n) else @as(usize, @intCast(n)) -| pending.items.len;
544 }
545 try pending.appendNTimes(A, ' ', count);
546 } else try pending.appendSlice(A, try display(try m.eval(e)));
547 }
548 if (s.args.len == 0 or !(eq(s.args[s.args.len - 1].s, ";") or eq(s.args[s.args.len - 1].s, ","))) try endLine();
549 } else if (eq(c, "INPUT")) {
550 for (s.targets) |e| {
551 const target = try m.reference(e);
552 while (true) {
553 if (m.input == m.inputs.len) return runtime(62);
554 const data = m.inputs[m.input];
555 m.input += 1;
556 if (e.string) {
557 target.* = .{ .string = true, .s = data };
558 break;
559 }
560 var valid = data.len > 0;
561 const off: usize = if (data.len > 0 and data[0] == '-') 1 else 0;
562 if (off == data.len) valid = false;
563 for (data[off..]) |ch| if (!std.ascii.isDigit(ch)) {
564 valid = false;
565 };
566 const n = std.fmt.parseInt(i64, data, 10) catch 32768;
567 if (valid and n >= -32768 and n <= 32767) {
568 target.* = integer(n);
569 break;
570 }
571 try message("?REDO FROM START");
572 }
573 }
574 } else if (eq(c, "READ")) {
575 for (s.targets) |e| {
576 const target = try m.reference(e);
577 if (m.read == m.data.items.len) return runtime(4);
578 const v = m.data.items[m.read].v;
579 m.read += 1;
580 if (v.string != e.string) return runtime(13);
581 target.* = v;
582 }
583 } else if (eq(c, "DIM")) {
584 for (s.targets) |e| {
585 var bounds: [2]usize = .{ 0, 0 };
586 var negative = false;
587 for (e.args, 0..) |a, i| {
588 const n = (try m.eval(a)).n;
589 if (n < 0) negative = true else bounds[i] = @intCast(n);
590 }
591 if (m.arrays[e.id] != null) return runtime(10);
592 if (negative) return runtime(5);
593 try m.create(e.id, bounds, e.args.len);
594 }
595 } else if (eq(c, "IF")) {
596 const yes = (try m.eval(s.args[0])).n != 0;
597 if (yes) {
598 if (s.yesline) |n| next = try m.goto(n) else if (s.yes.len > 0) next = s.yes[0];
599 } else {
600 if (s.noline) |n| next = try m.goto(n) else if (s.no.len > 0) next = s.no[0];
601 }
602 } else if (eq(c, "GOTO") or eq(c, "GOSUB")) {
603 const target = try m.goto(s.nums[0]);
604 if (eq(c, "GOSUB")) try m.push(.{ .k = .gosub, .position = s.next });
605 next = target;
606 } else if (eq(c, "ON")) {
607 const n = (try m.eval(s.args[0])).n;
608 if (n < 0) return runtime(5);
609 if (n > 0 and n <= @as(i64, @intCast(s.nums.len))) {
610 const target = try m.goto(s.nums[@intCast(n - 1)]);
611 if (eq(s.option, "GOSUB")) try m.push(.{ .k = .gosub, .position = s.next });
612 next = target;
613 }
614 } else if (eq(c, "RETURN")) {
615 var i = m.stack.items.len;
616 var found = false;
617 while (i > 0) {
618 i -= 1;
619 if (m.stack.items[i].k == .gosub) {
620 next = m.stack.items[i].position;
621 m.stack.shrinkRetainingCapacity(i);
622 found = true;
623 break;
624 }
625 }
626 if (!found) return runtime(3);
627 } else if (eq(c, "FOR")) {
628 const a = (try m.eval(s.args[0])).n;
629 const b = (try m.eval(s.args[1])).n;
630 const step = (try m.eval(s.args[2])).n;
631 if (step == 0) return runtime(5);
632 const id = s.targets[0].id;
633 if (m.search(.forloop, id)) |i| m.stack.shrinkRetainingCapacity(i);
634 m.vars[id] = integer(a);
635 if ((step > 0 and a > b) or (step < 0 and a < b)) next = try m.matching(s, "FOR", "NEXT") else try m.push(.{ .k = .forloop, .position = s.next, .id = id, .final = b, .step = step });
636 } else if (eq(c, "NEXT")) {
637 const i = m.search(.forloop, if (s.targets.len > 0) s.targets[0].id else null) orelse return runtime(1);
638 m.stack.shrinkRetainingCapacity(i + 1);
639 const f = m.stack.items[i];
640 const n = try checked(m.vars[f.id].n + f.step);
641 m.vars[f.id] = integer(n);
642 if ((f.step > 0 and n > f.final) or (f.step < 0 and n < f.final)) m.stack.shrinkRetainingCapacity(i) else next = f.position;
643 } else if (eq(c, "WHILE")) {
644 if ((try m.eval(s.args[0])).n != 0) try m.push(.{ .k = .whileloop, .position = s }) else next = try m.matching(s, "WHILE", "WEND");
645 } else if (eq(c, "WEND")) {
646 const i = m.search(.whileloop, null) orelse return runtime(30);
647 next = m.stack.items[i].position;
648 m.stack.shrinkRetainingCapacity(i);
649 } else if (eq(c, "RESTORE")) {
650 if (s.nums.len > 0) {
651 _ = try m.goto(s.nums[0]);
652 m.read = 0;
653 while (m.read < m.data.items.len and m.data.items[m.read].line < s.nums[0]) m.read += 1;
654 } else m.read = 0;
655 } else if (eq(c, "RANDOMIZE")) {
656 const n = (try m.eval(s.args[0])).n;
657 if (n < 1) return runtime(5);
658 m.random = @intCast(n);
659 } else if (eq(c, "ERROR")) {
660 const n = s.nums[0];
661 if (n != 0) {
662 _ = try m.goto(n);
663 }
664 m.handler = n;
665 } else if (eq(c, "RESUME")) {
666 if (!m.handling) return runtime(20);
667 if (s.nums.len > 0) next = try m.goto(s.nums[0]) else if (eq(s.option, "NEXT")) next = m.error_stmt.?.next else next = m.error_stmt;
668 m.handling = false;
669 } else if (eq(c, "END")) next = null else if (eq(c, "STOP")) {
670 try message(try fmt("BREAK IN LINE {d}", .{s.line}));
671 next = null;
672 }
673 return next;
674 }
675 fn run(m: *Machine) !void {
676 for (286..572) |i| m.vars[i].string = true;
677 for (m.program) |s| {
678 if (!m.lines.contains(s.line)) try m.lines.put(A, s.line, s);
679 if (eq(s.kind, "DATA")) for (s.args) |e| {
680 try m.data.append(A, .{ .line = s.line, .v = e.v });
681 };
682 }
683 var pc: ?*Stmt = if (m.program.len > 0) m.program[0] else null;
684 while (pc) |s| {
685 pc = m.execute(s) catch |e| {
686 if (e != error.Runtime) return e;
687 if (m.handler != 0 and !m.handling) {
688 m.err = @intCast(error_code);
689 m.erl = @intCast(s.line);
690 m.error_stmt = s;
691 m.handling = true;
692 pc = try m.goto(m.handler);
693 continue;
694 }
695 var msg: S = "";
696 for (codes, messages) |code, label| if (code == error_code) {
697 msg = label;
698 break;
699 };
700 try message(try fmt("ERROR IN LINE {d}: {s}", .{ s.line, msg }));
701 break;
702 };
703 }
704 }
705};

707fn wire(list: []*Stmt, next: ?*Stmt) void {
708 for (list, 0..) |s, i| {
709 s.next = if (i + 1 < list.len) list[i + 1] else next;
710 wire(s.yes, s.next);
711 wire(s.no, s.next);
712 }

715pub fn main(init: std.process.Init) !void {
716 var ib: [8192]u8 = undefined;
717 var input = std.Io.File.stdin().readerStreaming(init.io, &ib);
718 const all = try input.interface.allocRemaining(A, .unlimited);
719 var it = std.mem.splitScalar(u8, all, '\n');
720 const count = try std.fmt.parseInt(usize, it.next() orelse "0", 10);
721 for (0..count) |_| {
722 var stored: [10000]?S = @as([10000]?S, @splat(null));
723 while (it.next()) |raw| {
724 const s = std.mem.trimEnd(u8, raw, "\r");
725 if (eq(s, "RUN")) break;
726 const space = std.mem.indexOfScalar(u8, s, ' ');
727 const n = try std.fmt.parseInt(usize, if (space) |i| s[0..i] else s, 10);
728 stored[n] = if (space) |i| std.mem.trimStart(u8, s[i + 1 ..], " ") else null;
729 }
730 var inputs: std.ArrayList(S) = .empty;
731 while (it.next()) |raw| {
732 const s = std.mem.trimEnd(u8, raw, "\r");
733 if (eq(s, "END")) break;
734 try inputs.append(A, s);
735 }
736 var program: std.ArrayList(*Stmt) = .empty;
737 var bad: ?usize = null;
738 for (stored, 0..) |source, n| if (source) |s| {
739 const ts = lex(s) catch |e| {
740 if (e != error.Syntax) return e;
741 bad = n;
742 break;
743 };
744 var p = Parser{ .ts = ts, .line = n };
745 const statements = p.list(false) catch |e| {
746 if (e != error.Syntax) return e;
747 bad = n;
748 break;
749 };
750 if (p.i != p.ts.len) {
751 bad = n;
752 break;
753 }
754 try program.appendSlice(A, statements);
755 };
756 if (bad) |n| try message(try fmt("ERROR IN LINE {d}: SYNTAX ERROR", .{n})) else {
757 wire(program.items, null);
758 for (program.items, 0..) |s, i| s.index = i;
759 var m = Machine{ .program = program.items, .inputs = inputs.items };
760 try m.run();
761 }
762 if (pending.items.len > 0) try endLine();
763 try message("READY");
764 }
765 var ob: [8192]u8 = undefined;
766 var sink = std.Io.File.stdout().writerStreaming(init.io, &ob);
767 try sink.interface.writeAll(output.items);
768 try sink.interface.flush();
769}
