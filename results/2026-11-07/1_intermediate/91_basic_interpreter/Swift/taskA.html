121}

127func variable(_ name: String) -> Bool {
128 let bytes = Array(name.utf8)
129 if bytes.isEmpty || !upper(bytes[0]) { return false }
130 if bytes.count == 1 { return true }
131 if bytes.count == 2 {
132 return digit(bytes[1]) || bytes[1] == 36
133 }
134 return bytes.count == 3 && digit(bytes[1]) && bytes[2] == 36
135}

141func tokenize(_ source: String) throws -> [Token] {
142 let bytes = Array(source.utf8)
143 var position = 0
144 var tokens: [Token] = []
145 while position < bytes.count {
146 let c = bytes[position]
147 if c == 32 {
148 position += 1
149 } else if c == 34 {
150 let start = position
151 position += 1
152 while position < bytes.count && bytes[position] != 34 {
153 position += 1
154 }
155 if position == bytes.count { try syntax() }
156 tokens.append(Token(
157 kind: "S",
158 text: String(decoding: bytes[start...position], as: UTF8.self),
159 value: .text(String(decoding: bytes[(start + 1)..<position], as: UTF8.self))
160 ))
161 position += 1
162 } else if digit(c) {
163 let start = position
164 while position < bytes.count && digit(bytes[position]) {
165 position += 1
166 }
167 let word = String(decoding: bytes[start..<position], as: UTF8.self)
168 guard let n = Int(word), n <= 32767 else { try syntax() }
169 tokens.append(Token(kind: "I", text: word, value: .integer(n)))
170 } else if upper(c) {
171 if position + 2 < bytes.count &&
172 bytes[position] == 82 && bytes[position + 1] == 69 &&
173 bytes[position + 2] == 77 {
174 tokens.append(Token(kind: "W", text: "REM", value: .text("REM")))
175 break
176 }
177 let start = position
178 while position < bytes.count && (upper(bytes[position]) || digit(bytes[position])) {
179 position += 1
180 }
181 if position < bytes.count && bytes[position] == 36 {
182 position += 1
183 }
184 let word = String(decoding: bytes[start..<position], as: UTF8.self)
185 tokens.append(Token(kind: "W", text: word, value: .text(word)))
186 } else {
187 let candidate = String(decoding: bytes[position..<min(position + 2, bytes.count)], as: UTF8.self)
188 let word = ["<>", "<=", ">="].contains(candidate)
189 ? candidate : String(decoding: bytes[position...position], as: UTF8.self)
190 if !"+-*/^=<>() ,;:".utf8.contains(c) {
191 try syntax()
192 }
193 tokens.append(Token(kind: "P", text: word, value: .text(word)))
194 position += word.utf8.count
195 }
196 }
197 tokens.append(Token(kind: "E", text: "", value: .text("")))
198 return tokens
199}

211 private func at(_ word: String) -> Bool {
212 (current.kind == "P" || current.kind == "W") && current.text == word
213 }

215 @discardableResult
216 private func take(_ word: String) -> Bool {
217 if !at(word) { return false }
218 index += 1
219 return true
220 }
221
222 private func expect(_ word: String) throws {
223 if !take(word) { try syntax() }
224 }
225
226 private func number() throws -> Int {
227 if current.kind != "I" { try syntax() }
228 let result = current.value.number
229 index += 1
230 return result
231 }
232
233 private func integer(_ level: Int = 0) throws -> Expression {
234 let result = try expression(level)
235 if result.type != "I" { try syntax() }
236 return result
237 }
238
239 private func target() throws -> Expression {
240 let token = current
241 if token.kind != "W" || !variable(token.text) { try syntax() }
242 index += 1
243 let result = Expression("variable", nameType(token.text), .text(token.text))
244 if take("(") {
245 result.kind = "array"
246 result.args.append(try integer())
247 if take(",") { result.args.append(try integer()) }
248 try expect(")")
249 }
250 return result
251 }
252
253 private func primary() throws -> Expression {
254 let token = current
255 if token.kind == "I" || token.kind == "S" {
256 index += 1
257 return Expression("constant", token.kind, token.value)
258 }
259 if take("(") {
260 let value = try expression()
261 try expect(")")
262 return value
263 }
264 if token.kind == "W" && variable(token.text) { return try target() }
265 if take("ERR") { return Expression("builtin", "I", .text("ERR")) }
266 if take("ERL") { return Expression("builtin", "I", .text("ERL")) }
267 if token.kind == "W", let expected = signatures[token.text] {
268 index += 1
269 try expect("(")
270 var arguments = [try expression()]
271 while take(",") { arguments.append(try expression()) }
272 try expect(")")
273 if !expected.contains(arguments.map(\.type).joined()) { try syntax() }
274 return Expression("call", nameType(token.text), .text(token.text), arguments)
275 }
276 try syntax()
277 }
278
279 private func expression(_ level: Int = 0) throws -> Expression {
280 if level == 2 && take("NOT") {
281 return Expression("unary", "I", .text("NOT"), [try integer(2)])
282 }
283 if level == 6 && take("-") {
284 return Expression("unary", "I", .text("-"), [try integer(6)])
285 }
286 if level == 8 { return try primary() }
287 var left = try expression(level + 1)
288 let operators = [
289 ["OR"], ["AND"], [],
290 ["=", "<>", "<", ">", "<=", ">="],
291 ["+", "-"], ["*", "/", "MOD"], [], ["^"]
292 ][level]
293 while operators.contains(where: { at($0) }) {
294 let op = current.text
295 index += 1
296 let right = try expression(level == 7 ? 6 : level + 1)
297 if left.type != right.type ||
298 (left.type == "S" && level != 3 && op != "+") { try syntax() }
299 left = Expression("binary", level == 3 ? "I" : left.type,
300 .text(op), [left, right])
301 if level == 7 { break }
302 }
303 return left
304 }

306 private func clause() throws -> Clause {
307 if current.kind == "I" { return Clause(number: try number(), body: []) }
308 return Clause(number: nil, body: try list(true))
309 }

311 private func list(_ insideClause: Bool = false) throws -> [Statement] {
312 var result = [try statement(insideClause)]
313 while take(":") { result.append(try statement(insideClause)) }
314 return result

317 private func statement(_ insideClause: Bool) throws -> Statement {
318 let word = current.text
319 if insideClause && ["IF", "FOR", "NEXT", "WHILE", "WEND", "DATA"].contains(word) {
320 try syntax()
321 }
322 let s = Statement(word)
323 if take("REM") || take("RETURN") || take("WEND") ||
324 take("END") || take("STOP") {
325 return s
326 }
327 if take("PRINT") {
328 var lastWasValue = false
329 while current.kind != "E" && !at(":") && !at("ELSE") {
330 if take(";") {
331 s.items.append(PrintItem(kind: ";", expression: nil))
332 lastWasValue = false
333 } else if take(",") {
334 s.items.append(PrintItem(kind: ",", expression: nil))
335 lastWasValue = false
336 } else {
337 let mode = at("TAB") ? "TAB" : at("SPC") ? "SPC" : "value"
338 if mode == "value" && lastWasValue { try syntax() }
339 let value: Expression
340 if mode == "value" {
341 value = try expression()
342 } else {
343 index += 1
344 try expect("(")
345 value = try integer()
346 try expect(")")
347 }
348 s.items.append(PrintItem(kind: mode, expression: value))
349 lastWasValue = mode == "value"
350 }
351 }
352 return s
353 }
354 if take("INPUT") || take("READ") || take("DIM") {
355 repeat {
356 let target = try self.target()
357 if word == "DIM" && target.kind != "array" { try syntax() }
358 s.targets.append(target)
359 } while take(",")
360 return s
361 }
362 if take("DATA") {
363 repeat {
364 let negative = take("-")
365 let token = current
366 if token.kind != "I" && (token.kind != "S" || negative) {
367 try syntax()
368 }
369 index += 1
370 s.values.append(negative ? .integer(-token.value.number) : token.value)
371 } while take(",")
372 return s
373 }
374 if take("IF") {
375 s.expression = try integer()
376 try expect("THEN")
377 s.clauses.append(try clause())
378 if take("ELSE") { s.clauses.append(try clause()) }
379 if current.kind != "E" { try syntax() }
380 return s
381 }
382 if take("GOTO") || take("GOSUB") {
383 s.numbers.append(try number())
384 return s
385 }
386 if take("ON") {
387 if take("ERROR") {
388 s.kind = "handler"
389 try expect("GOTO")
390 s.numbers.append(try number())
391 } else {
392 s.expression = try integer()
393 if take("GOTO") {
394 s.mode = "GOTO"
395 } else {
396 try expect("GOSUB")
397 s.mode = "GOSUB"
398 }
399 repeat { s.numbers.append(try number()) } while take(",")
400 }
401 return s
402 }
403 if take("RESUME") {
404 if take("NEXT") {
405 s.mode = "NEXT"
406 } else if current.kind == "I" {
407 s.mode = "line"
408 s.numbers.append(try number())
409 }
410 return s
411 }
412 if take("FOR") {
413 let token = current
414 if token.kind != "W" || !variable(token.text) ||
415 nameType(token.text) != "I" { try syntax() }
416 s.name = token.text
417 index += 1
418 try expect("=")
419 s.expressions.append(try integer())
420 try expect("TO")
421 s.expressions.append(try integer())
422 s.expressions.append(take("STEP") ? try integer() :
423 Expression("constant", "I", .integer(1)))
424 return s
425 }
426 if take("NEXT") {
427 if current.kind == "W" && variable(current.text) &&
428 nameType(current.text) == "I" {
429 s.name = current.text
430 index += 1
431 }
432 return s
433 }
434 if take("WHILE") || take("RANDOMIZE") {
435 s.expression = try integer()
436 return s
437 }
438 if take("RESTORE") {
439 if current.kind == "I" { s.numbers.append(try number()) }
440 return s
441 }
442 _ = take("LET")
443 s.kind = "assign"
444 s.target = try target()
445 try expect("=")
446 s.expression = try expression()
447 if s.expression!.type != s.target!.type { try syntax() }
448 return s
449 }
450
451 func parse() throws -> [Statement] {
452 let result = try list()
453 if current.kind != "E" { try syntax() }
454 return result
455 }
456}

534 private func address(_ expression: Expression) throws -> Address {
535 let name = expression.value.string
536 if expression.kind == "variable" { return Address(name: name, index: nil) }
537 let indices = try expression.args.map { try evaluate($0).number }
538 if arrays[name] == nil { declare(name, Array(repeating: 10, count: indices.count)) }
539 let array = arrays[name]!
540 if indices.count != array.bounds.count { try fault("SUBSCRIPT OUT OF RANGE") }
541 var index = 0
542 for (i, value) in indices.enumerated() {
543 if value < 0 || value > array.bounds[i] { try fault("SUBSCRIPT OUT OF RANGE") }
544 index = index * (array.bounds[i] + 1) + value
545 }
546 return Address(name: name, index: index)
547 }

562 private func power(_ base: Int, _ exponent: Int) throws -> Int {
563 if exponent < 0 { try fault("ILLEGAL FUNCTION CALL") }
564 var n = exponent
565 var x = base
566 var result = 1
567 while n > 0 {
568 if n % 2 != 0 { result = try checked(result * x) }
569 n /= 2
570 if n > 0 { x = try checked(x * x) }
571 }
572 return result
573 }

575 private func evaluate(_ expression: Expression) throws -> Value {
576 let op = expression.value.stringIfText
577 if expression.kind == "constant" { return expression.value }
578 if expression.kind == "variable" || expression.kind == "array" {
579 return load(try address(expression))
580 }
581 if expression.kind == "builtin" {
582 return .integer(op == "ERR" ? err : erl)
583 }
584 if expression.kind == "unary" {
585 let a = try evaluate(expression.args[0]).number
586 return .integer(op == "NOT" ? ~a : try checked(-a))
587 }
588 let args = try expression.args.map { try evaluate($0) }
589 if expression.kind == "call" { return try call(op, args) }
590 let a = args[0], b = args[1]
591 if ["=", "<>", "<", ">", "<=", ">="].contains(op) {
592 let order: Int
593 if a == b { order = 0 }
594 else if a.type == "S" {
595 order = a.string < b.string ? -1 : 1
596 } else {
597 order = a.number < b.number ? -1 : 1
598 }
599 let answer: Bool
600 switch op {
601 case "=": answer = order == 0
602 case "<>": answer = order != 0
603 case "<": answer = order < 0
604 case ">": answer = order > 0
605 case "<=": answer = order <= 0
606 default: answer = order >= 0
607 }
608 return .integer(answer ? -1 : 0)
609 }
610 if expression.type == "S" {
611 let result = a.string + b.string
612 if result.utf8.count > 255 { try fault("STRING TOO LONG") }
613 return .text(result)
614 }
615 let x = a.number, y = b.number
616 switch op {
617 case "+": return .integer(try checked(x + y))
618 case "-": return .integer(try checked(x - y))
619 case "*": return .integer(try checked(x * y))
620 case "/":
621 if y == 0 { try fault("DIVISION BY ZERO") }
622 return .integer(try checked(x / y))
623 case "MOD":
624 if y == 0 { try fault("DIVISION BY ZERO") }
625 return .integer(x % y)
626 case "^": return .integer(try power(x, y))
627 case "AND": return .integer(x & y)
628 case "OR": return .integer(x | y)
629 default: try syntax()
630 }
631 }

633 private func call(_ name: String, _ args: [Value]) throws -> Value {
634 let n = args[0].type == "I" ? args[0].number : 0
635 let s = args[0].type == "S" ? args[0].string : ""
636 let k = args.count > 1 && args[1].type == "I" ? args[1].number : 0
637 switch name {
638 case "ABS": return .integer(try checked(abs(n)))
639 case "SGN": return .integer(n == 0 ? 0 : n > 0 ? 1 : -1)
640 case "INT": return .integer(n)
641 case "SQR":
642 if n < 0 { try fault("ILLEGAL FUNCTION CALL") }
643 return .integer(Int(sqrt(Double(n))))
644 case "RND":
645 if n < 1 { try fault("ILLEGAL FUNCTION CALL") }
646 randomState = randomState * 48271 % 2147483647
647 return .integer(randomState % n + 1)
648 case "LEN": return .integer(s.utf8.count)
649 case "ASC":
650 if s.isEmpty { try fault("ILLEGAL FUNCTION CALL") }
651 return .integer(Int(s.utf8.first!))
652 case "VAL":
653 let bytes = Array(s.utf8)
654 var i = 0
655 while i < bytes.count && bytes[i] == 32 { i += 1 }
656 let start = i
657 if i < bytes.count && (bytes[i] == 43 || bytes[i] == 45) { i += 1 }
658 let digits = i
659 while i < bytes.count && digit(bytes[i]) { i += 1 }
660 if digits == i { return .integer(0) }
661 let value = Int(String(decoding: bytes[start..<i], as: UTF8.self))
662 guard let value else { try fault("OVERFLOW") }
663 return .integer(try checked(value))
664 case "CHR$":
665 if n < 32 || n > 126 { try fault("ILLEGAL FUNCTION CALL") }
666 return .text(String(UnicodeScalar(n)!))
667 case "STR$": return .text(String(n))
668 case "LEFT$":
669 if k < 0 { try fault("ILLEGAL FUNCTION CALL") }
670 return .text(String(s.prefix(k)))
671 case "RIGHT$":
672 if k < 0 { try fault("ILLEGAL FUNCTION CALL") }
673 return .text(String(s.suffix(k)))
674 case "MID$":
675 let length = args.count == 3 ? args[2].number : s.count
676 if k < 1 || length < 0 { try fault("ILLEGAL FUNCTION CALL") }
677 return .text(String(s.dropFirst(k - 1).prefix(length)))
678 case "INSTR":
679 guard let range = s.range(of: args[1].string) else { return .integer(0) }
680 return .integer(s.distance(from: s.startIndex, to: range.lowerBound) + 1)
681 default: try syntax()
682 }
683 }

704 private func matching(_ s: Statement, _ open: String, _ close: String) throws -> Statement? {
705 let index = sequence.firstIndex { $0 === s }!
706 var depth = 0
707 if index + 1 < sequence.count {
708 for i in (index + 1)..<sequence.count {
709 let candidate = sequence[i]
710 if candidate.kind == open { depth += 1 }
711 else if candidate.kind == close {
712 if depth == 0 { return candidate.next }
713 depth -= 1
714 }
715 }
716 }
717 try fault(open + " WITHOUT " + close)
718 }

726 private func execute(_ s: Statement) throws -> Statement? {
727 var next = s.next
728 switch s.kind {
729 case "REM", "DATA": break
730 case "assign":
731 let place = try address(s.target!)
732 store(place, try evaluate(s.expression!))
733 case "PRINT":
734 for item in s.items {
735 if item.kind == "," {
736 buffer += String(repeating: " ", count: 10 - buffer.utf8.count % 10)
737 } else if item.kind != ";" {
738 let value = try evaluate(item.expression!)
739 if item.kind == "value" {
740 buffer += value.rendered
741 } else {
742 let amount = value.number
743 if amount < 0 { try fault("ILLEGAL FUNCTION CALL") }
744 let spaces = item.kind == "SPC"
745 ? amount : max(0, amount - buffer.utf8.count)
746 buffer += String(repeating: " ", count: spaces)
747 }
748 }
749 }
750 if s.items.isEmpty || ![";", ","].contains(s.items.last!.kind) { endLine() }
751 case "INPUT", "READ":
752 for target in s.targets {
753 let place = try address(target)
754 let value: Value
755 if s.kind == "READ" {
756 if dataPosition == data.count { try fault("OUT OF DATA") }
757 value = data[dataPosition].value
758 dataPosition += 1
759 if value.type != target.type { try fault("TYPE MISMATCH") }
760 } else {
761 var input: Value
762 while true {
763 if inputPosition == inputs.count { try fault("INPUT PAST END") }
764 let text = inputs[inputPosition]
765 inputPosition += 1
766 if target.type == "S" {
767 input = .text(text)
768 break
769 }
770 let bytes = Array(text.utf8)
771 let start = bytes.first == 45 ? 1 : 0
772 if start < bytes.count &&
773 bytes[start...].allSatisfy({ digit($0) }),
774 let number = Int(text), number >= -32768 && number <= 32767 {
775 input = .integer(number)
776 break
777 }
778 message("?REDO FROM START")
779 }
780 value = input
781 }
782 store(place, value)
783 }
784 case "RESTORE":
785 if let number = s.numbers.first {
786 _ = try line(number)
787 dataPosition = data.firstIndex { $0.line >= number } ?? data.count
788 } else {
789 dataPosition = 0
790 }
791 case "IF":
792 let index = try evaluate(s.expression!).number != 0 ? 0 : 1
793 if index < s.clauses.count {
794 let clause = s.clauses[index]
795 next = try clause.number.map { try line($0) } ?? clause.body.first
796 }
797 case "GOTO": next = try line(s.numbers[0])
798 case "GOSUB": next = try subroutine(s.numbers[0], next)
799 case "ON":
800 let number = try evaluate(s.expression!).number
801 if number < 0 { try fault("ILLEGAL FUNCTION CALL") }
802 if number >= 1 && number <= s.numbers.count {
803 let destination = s.numbers[number - 1]
804 next = try s.mode == "GOTO" ? line(destination) : subroutine(destination, next)
805 }
806 case "RETURN":
807 guard let i = stack.lastIndex(where: { $0.kind == "GOSUB" })
808 else { try fault("RETURN WITHOUT GOSUB") }
809 next = stack[i].position
810 stack.removeSubrange(i..<stack.count)
811 case "FOR":
812 let values = try s.expressions.map { try evaluate($0).number }
813 let start = values[0], last = values[1], step = values[2]
814 if step == 0 { try fault("ILLEGAL FUNCTION CALL") }
815 let found = search("FOR", s.name)
816 if found >= 0 { stack.removeSubrange(found..<stack.count) }
817 variables[s.name] = .integer(start)
818 if step > 0 ? start > last : start < last {
819 next = try matching(s, "FOR", "NEXT")
820 } else {
821 try push(Frame("FOR", next, name: s.name, limit: last, step: step))
822 }
823 case "NEXT":
824 let found = search("FOR", s.name)
825 if found < 0 { try fault("NEXT WITHOUT FOR") }
826 stack.removeSubrange((found + 1)..<stack.count)
827 let frame = stack[found]
828 let value = try checked(get(frame.name).number + frame.step)
829 variables[frame.name] = .integer(value)
830 if frame.step > 0 ? value > frame.limit : value < frame.limit {
831 stack.removeLast()
832 } else {
833 next = frame.position
834 }
835 case "WHILE":
836 if try evaluate(s.expression!).number != 0 {
837 try push(Frame("WHILE", s))
838 } else {
839 next = try matching(s, "WHILE", "WEND")
840 }
841 case "WEND":
842 let found = search("WHILE")
843 if found < 0 { try fault("WEND WITHOUT WHILE") }
844 next = stack[found].position
845 stack.removeSubrange(found..<stack.count)
846 case "DIM":
847 for target in s.targets {
848 let name = target.value.string
849 let bounds = try target.args.map { try evaluate($0).number }
850 if arrays[name] != nil { try fault("DUPLICATE DEFINITION") }
851 if bounds.contains(where: { $0 < 0 }) { try fault("ILLEGAL FUNCTION CALL") }
852 if bounds.reduce(1, { $0 * ($1 + 1) }) > 10000 { try fault("OUT OF MEMORY") }
853 declare(name, bounds)
854 }
855 case "RANDOMIZE":
856 let n = try evaluate(s.expression!).number
857 if n < 1 { try fault("ILLEGAL FUNCTION CALL") }
858 randomState = n
859 case "handler":
860 if s.numbers[0] != 0 { _ = try line(s.numbers[0]) }
861 handler = s.numbers[0]
862 case "RESUME":
863 if !handling { try fault("RESUME WITHOUT ERROR") }
864 if s.mode == "line" {
865 next = try line(s.numbers[0])
866 } else if s.mode == "NEXT" {
867 next = remembered!.next
868 } else {
869 next = remembered
870 }
871 handling = false
872 case "STOP":
873 message("BREAK IN LINE " + String(s.line))
874 return nil
875 case "END": return nil
876 default: break
877 }
878 return next
879 }

881 private func connect(_ body: [Statement], _ after: Statement?, _ number: Int) {
882 var next = after
883 for statement in body.reversed() {
884 statement.line = number
885 statement.next = next
886 for clause in statement.clauses {
887 connect(clause.body, next, number)
888 }
889 next = statement
890 }
891 }

893 private func finish() -> [String] {
894 if !buffer.isEmpty { endLine() }
895 output.append("READY")
896 return output
897 }

899 func run(_ source: [Int: String]) -> [String] {
900 var parsed: [(line: Int, statements: [Statement])] = []
901 for number in source.keys.sorted() {
902 do {
903 let statements = try Parser(try tokenize(source[number]!)).parse()
904 parsed.append((number, statements))
905 } catch is SyntaxFault {
906 message("ERROR IN LINE " + String(number) + ": SYNTAX ERROR")
907 return finish()
908 } catch {
909 message("ERROR IN LINE " + String(number) + ": SYNTAX ERROR")
910 return finish()
911 }
912 }
913 var after: Statement?
914 for (number, statements) in parsed.reversed() {
915 connect(statements, after, number)
916 entries[number] = statements.first!
917 after = statements.first
918 }
919 for (number, statements) in parsed {
920 for statement in statements {
921 sequence.append(statement)
922 if statement.kind == "DATA" {
923 for value in statement.values { data.append((number, value)) }
924 }
925 }
926 }
927 var position = after
928 while let statement = position {
929 do {
930 position = try execute(statement)
931 } catch let error as BasicFault {
932 if handler != 0 && !handling {
933 err = errorCodes[error.message] ?? 0
934 erl = statement.line
935 remembered = statement
936 handling = true
937 position = try? line(handler)
938 } else {
939 message("ERROR IN LINE " + String(statement.line) + ": " + error.message)
940 position = nil
941 }
942 } catch {
943 position = nil
944 }
945 }
946 return finish()
947 }
948}
