48def revcomp(s)
49 s.tr('ACGTRYWSKMBVDHN', 'TGCAYRWSMKVBHDN').reverse

52def gc(s)

56def fixed(value, places)
57 scale = 10**places
58 format("%d.%0#{places}d", value / scale, value % scale)
59end

61def pct(a, b)
62 fixed(10000 * a / b, 2)
63end

65def transition?(a, b)
66 %w[AG GA CT TC].include?(a + b)
67end

69def frequency_order(table)
70 table.keys.sort_by { |item| [-table[item], item] }
71end

73def codons(s, frame)
74 strand = frame.positive? ? s : revcomp(s)
75 strand[(frame.abs - 1)..].to_s.scan(/.../)
76end

78def occurrences(s, motif)
79 (0..(s.length - motif.length)).select do |i|
80 motif.each_char.with_index.all? { |symbol, j| IUPAC[symbol].include?(s[i + j]) }
81 end.map { |i| i + 1 }
82end

84def repeat_count(s, start, width)
85 unit = s[start, width]
86 count = 0
87 count += 1 while s[start + count * width, width] == unit
88 count
89end

91def primitive?(unit)
92 (1...unit.length).none? { |d| (unit.length % d).zero? && unit[0, d] * (unit.length / d) == unit }
93end

95def longest_palindrome(s)
96 best = 0
97 (1...s.length).each do |center|
98 left = center - 1
99 right = center
100 while left >= 0 && right < s.length && COMPLEMENT[s[left]] == s[right]
101 best = [best, right - left + 1].max
102 left -= 1
103 right += 1
104 end
105 end
106 best
107end

109def temperature(s)
110 n = s.length
111 amount = gc(s)
112 n < 14 ? 10 * (2 * (n - amount) + 4 * amount) : (649 * n + 410 * amount - 6724) / n
113end

115def strand_mass(s)
116 s.each_char.sum { |b| BASE_MASS[b] } - (s.length - 1) * 18015
117end

135 def kinds(cmd, count)
136 case cmd
137 when 'SEQ' then [:new, :sequence]
138 when 'SLICE' then [:name, [1, :n], [1, :n], :new]
139 when 'JOIN' then [:new] + Array.new(count - 1, :name)
140 when 'MUTATE' then [:name, [1, :n], :base]
141 when 'TRANSLATE', 'CODONS', 'PROTEIN' then [:name, [-3, 3, true]]
142 when 'ORFS' then [:name, [1, 200]]
143 when 'GCWINDOW' then [:name, [1, 500], [1, 500]]
144 when 'KMERS' then [:name, [1, 12], [1, 100]]
145 when 'FIND' then [:name, :motif]
146 when 'RESTRICT' then [:name] + Array.new(count - 1, :enzyme)
147 when 'PALINDROMES' then [:name, [2, 20], [2, 20]]
148 when 'REPEATS' then [:name, [1, 6], [1, 6], [2, 50]]
149 when 'ALIGN' then [:name, :name, :mode, [1, 20], [-20, 0], [-20, -1]]
150 when 'FOLD' then [:name, [0, 10]]
151 when 'ASSEMBLE' then [:new, [1, 100]] + Array.new(count - 2, :name)
152 when 'UNPACK' then [:new, [1, 500], :hex]
153 else Array.new(count, :name)
154 end
155 end

157 def validate(cmd, args)
158 raise CommandError, "unknown command #{cmd}" unless ARITY.key?(cmd)
159 minimum, maximum = ARITY[cmd]
160 raise CommandError, 'wrong argument count' unless (minimum..maximum).cover?(args.length)
161 values = []
162 seen = []
163 kinds(cmd, args.length).zip(args).each do |kind, text|
164 value = text
165 if kind.is_a?(Array)
166 upper = kind[1] == :n ? @registry[args[0]].length : kind[1]
167 unless /\A-?[0-9]{1,6}\z/.match?(text)
168 raise CommandError, "invalid number #{text}"
169 end
170 value = text.to_i
171 if value < kind[0] || value > upper || (kind[2] && value.zero?)
172 raise CommandError, "invalid number #{text}"
173 end
174 elsif kind == :name
175 raise CommandError, "unknown sequence #{text}" unless @registry.key?(text)
176 elsif kind == :new
177 raise CommandError, "invalid name #{text}" unless /\A[a-z][a-z0-9]{0,9}\z/.match?(text)
178 elsif kind == :sequence
179 text.each_char.with_index do |base, i|
180 raise CommandError, "invalid base #{base} at #{i + 1}" unless BASES.include?(base)
181 end
182 elsif kind == :base
183 raise CommandError, "invalid base #{text}" unless text.length == 1 && BASES.include?(text)
184 elsif kind == :motif
185 unless (1..30).cover?(text.length) && text.each_char.all? { |c| IUPAC.key?(c) }
186 raise CommandError, "invalid motif #{text}"
187 end
188 elsif kind == :enzyme
189 raise CommandError, "unknown enzyme #{text}" unless ENZYMES.key?(text)
190 raise CommandError, "duplicate enzyme #{text}" if seen.include?(text)
191 seen << text
192 elsif kind == :mode
193 raise CommandError, "invalid mode #{text}" unless %w[GLOBAL LOCAL].include?(text)
194 elsif kind == :hex
195 unless /\A[0-9A-F]+\z/.match?(text) && text.length == 2 * ((values[1] + 3) / 4)
196 raise CommandError, 'invalid hex'
197 end
198 end
199 values << value
200 end
201 values
202 end

207 @registry[name] = s

211 def translation(cmd, s, frame)
212 triplets = codons(s, frame)
213 protein = triplets.map { |c| CODE[c] }.join
214 if cmd == 'TRANSLATE'
215 emit(protein.empty? ? 'none' : protein)
216 elsif cmd == 'CODONS'
217 table = triplets.tally
218 frequency_order(table).each { |c| emit(c, CODE[c], table[c]) }
219 emit('none') if table.empty?
220 else
221 protein = protein.partition('*').first
222 raise CommandError, 'empty protein' if protein.empty?
223 table = protein.chars.tally
224 mass = protein.each_char.sum { |a| AMINO_MASS[a] } + 18015
225 charge = protein.count('KR') - protein.count('DE')
226 emit(protein)
227 emit('length', protein.length, 'mass', fixed(mass, 3), 'charge', charge.positive? ? "+#{charge}" : charge)
228 emit(*frequency_order(table).map { |a| "#{a}=#{table[a]}" })
229 end
230 end

232 def orfs(s, minimum)
233 results = []
234 [1, 2, 3, -1, -2, -3].each do |frame|
235 triplets = codons(s, frame)
236 start = nil
237 triplets.each_with_index do |codon, p|
238 if start.nil?
239 start = p if codon == 'ATG'
240 elsif CODE[codon] == '*'
241 if p - start >= minimum
242 left = frame.abs - 1 + 3 * start
243 right = frame.abs - 1 + 3 * (p + 1)
244 left, right = s.length - right, s.length - left if frame.negative?
245 protein = triplets[start...p].map { |c| CODE[c] }.join
246 results << [left + 1, right, frame, protein]
247 end
248 start = nil
249 end
250 end
251 end
252 results.sort_by(&:first).each do |left, right, frame, protein|
253 emit(format('%+d', frame), left, right, protein.length, protein)
254 end
255 emit('none') if results.empty?
256 end

258 def restrict(s, enzymes)
259 combined = []
260 enzymes.each do |enzyme|
261 motif, offset = ENZYMES[enzyme]
262 cuts = occurrences(s, motif).map { |p| p + offset - 1 }.select { |c| c >= 1 && c < s.length }
263 emit_list(enzyme, cuts)
264 combined.concat(cuts)
265 end
266 endpoints = [0, *combined.uniq.sort, s.length]
267 fragments = endpoints.each_cons(2).map { |a, b| [a + 1, b] }
268 emit_list('fragments', fragments.map { |a, b| "#{a}-#{b}" })
269 emit_list('sizes', fragments.map { |a, b| b - a + 1 }.sort.reverse)
270 end

272 def repeats(s, minimum, maximum, limit)
273 raise CommandError, 'invalid range' if minimum > maximum
274 results = []
275 (minimum..maximum).each do |width|
276 position = 0
277 while position + width <= s.length
278 unit = s[position, width]
279 count = repeat_count(s, position, width)
280 if primitive?(unit) && count >= limit
281 results << [position, width, count, unit]
282 position += count * width
283 else
284 position += 1
285 end
286 end
287 end
288 results.sort_by { |p, w, _, _| [p, w] }.each do |position, width, count, unit|
289 emit("#{position + 1}-#{position + width * count}", "#{unit}x#{count}")
290 end
291 emit('none') if results.empty?
292 end

294 def align(a, b, mode, match, mismatch, gap)
295 m = a.length
296 n = b.length
297 local = mode == 'LOCAL'
298 table = Array.new(m + 1) { Array.new(n + 1, 0) }
299 unless local
300 (0..m).each { |i| table[i][0] = i * gap }
301 (0..n).each { |j| table[0][j] = j * gap }
302 end
303 best = 0
304 end_i = 0
305 end_j = 0
306 (1..m).each do |i|
307 (1..n).each do |j|
308 diagonal = table[i - 1][j - 1] + (a[i - 1] == b[j - 1] ? match : mismatch)
309 value = [diagonal, table[i - 1][j] + gap, table[i][j - 1] + gap].max
310 table[i][j] = local ? [0, value].max : value
311 if local && table[i][j] > best
312 best = table[i][j]
313 end_i = i
314 end_j = j
315 end
316 end
317 end
318 unless local
319 end_i = m
320 end_j = n
321 best = table[m][n]
322 end
323 if local && best.zero?
324 emit('score', 0)
325 return
326 end
327 i = end_i
328 j = end_j
329 row_a = []
330 row_b = []
331 while (local && table[i][j] != 0) || (!local && (i != 0 || j != 0))
332 if i.positive? && j.positive? && table[i][j] == table[i - 1][j - 1] + (a[i - 1] == b[j - 1] ? match : mismatch)
333 row_a << a[i - 1]
334 row_b << b[j - 1]
335 i -= 1
336 j -= 1
337 elsif i.positive? && table[i][j] == table[i - 1][j] + gap
338 row_a << a[i - 1]
339 row_b << '-'
340 i -= 1
341 else
342 row_a << '-'
343 row_b << b[j - 1]
344 j -= 1
345 end
346 end
347 row_a.reverse!
348 row_b.reverse!
349 middle = row_a.zip(row_b).map { |x, y| x == '-' || y == '-' ? '-' : x == y ? '|' : '*' }.join
350 emit('score', best, 'a', "#{i + 1}-#{end_i}", 'b', "#{j + 1}-#{end_j}",
351 'identity', "#{middle.count('|')}/#{middle.length}", 'gaps', middle.count('-'))
352 emit(row_a.join)
353 emit(middle)
354 emit(row_b.join)
355 end

357 def consensus(sequences)
358 n = sequences[0].length
359 raise CommandError, 'length mismatch' unless sequences.all? { |s| s.length == n }
360 profile = BASES.each_char.to_h { |b| [b, []] }
361 consensus = ''
362 ambiguous = ''
363 n.times do |i|
364 table = sequences.map { |s| s[i] }.tally
365 largest = table.values.max
366 tied = BASES.each_char.select { |b| table[b] == largest }.join
367 BASES.each_char { |b| profile[b] << table.fetch(b, 0) }
368 consensus += tied[0]
369 ambiguous += SYMBOL[tied]
370 end
371 BASES.each_char { |b| emit(b, *profile[b]) }
372 emit('consensus', consensus)
373 emit('iupac', ambiguous)
374 end

376 def primer(s)
377 n = s.length
378 amount = gc(s)
379 tm = temperature(s)
380 clamp = gc(s[-5, 5] || s)
381 longest_run = s.each_char.chunk(&:itself).map { |_, run| run.length }.max
382 repeated = 0
383 (0...(n - 1)).each do |i|
384 repeated = [repeated, repeat_count(s, i, 2)].max if s[i] != s[i + 1]
385 end
386 selfcomp = longest_palindrome(s)
387 items = [
388 ['length', n, (18..25).cover?(n)],
389 ['gc', pct(amount, n), 40 * n <= 100 * amount && 100 * amount <= 60 * n],
390 ['tm', fixed(tm, 1), (520..620).cover?(tm)],
391 ['clamp', clamp, (1..3).cover?(clamp)],
392 ['run', longest_run, longest_run <= 4],
393 ['repeat', repeated, repeated <= 4],
394 ['selfcomp', selfcomp, selfcomp <= 6]
395 ]
396 failed = []
397 items.each do |label, value, passed|
398 emit(label, value, passed ? 'OK' : 'FAIL')
399 failed << label unless passed
400 end
401 emit(failed.empty? ? 'PASS' : 'FAIL', *failed)
402 end

404 def fold(s, loop)
405 n = s.length
406 raise CommandError, 'too long' if n > 100
407 table = Array.new(n) { Array.new(n, 0) }
408 value = ->(i, j) { i > j ? 0 : table[i][j] }
409 pair = ->(i, j) { %w[AT TA CG GC GT TG].include?(s[i] + s[j]) }
410 (2..n).each do |width|
411 (0..(n - width)).each do |i|
412 j = i + width - 1
413 best = value.call(i, j - 1)
414 (i...(j - loop)).each do |k|
415 best = [best, value.call(i, k - 1) + 1 + value.call(k + 1, j - 1)].max if pair.call(k, j)
416 end
417 table[i][j] = best
418 end
419 end
420 structure = Array.new(n, '.')
421 pending = [[0, n - 1]]
422 until pending.empty?
423 i, j = pending.pop
424 next if value.call(i, j).zero?
425 if value.call(i, j) == value.call(i, j - 1)
426 pending << [i, j - 1]
427 next
428 end
429 (i...(j - loop)).each do |k|
430 next unless pair.call(k, j) && value.call(i, k - 1) + 1 + value.call(k + 1, j - 1) == value.call(i, j)
431 structure[k] = '('
432 structure[j] = ')'
433 pending.push([i, k - 1], [k + 1, j - 1])
434 break
435 end
436 end
437 emit(structure.join)
438 emit('pairs', table[0][n - 1])
439 end

441 def assemble(name, minimum, sequences)
442 unique = sequences.uniq
443 fragments = unique.reject { |s| unique.any? { |t| t != s && t.include?(s) } }
444 while fragments.length >= 2
445 best = -1
446 chosen_i = 0
447 chosen_j = 1
448 fragments.each_with_index do |x, i|
449 fragments.each_with_index do |y, j|
450 next if i == j
451 overlap = 0
452 ([x.length, y.length].min - 1).downto(1) do |width|
453 if x[-width, width] == y[0, width]
454 overlap = width
455 break
456 end
457 end
458 if overlap > best
459 best = overlap
460 chosen_i = i
461 chosen_j = j
462 end
463 end
464 end
465 break if best < minimum
466 merged = fragments[chosen_i] + fragments[chosen_j][best..]
467 fragments = fragments.each_with_index.filter_map do |fragment, i|
468 if i == chosen_i
469 merged
470 elsif i != chosen_j && !merged.include?(fragment)
471 fragment
472 end
473 end
474 end
475 longest = fragments.max_by(&:length)
476 registration = store(name, longest)
477 emit('contigs', fragments.length)
478 fragments.each { |s| emit(s.length, s) }
479 emit(registration)
480 end

482 def command(cmd, a)
483 case cmd
484 when 'SEQ'
485 emit(store(a[0], a[1]))
486 return
487 when 'UNPACK'
488 codes = []
489 a[2].scan(/../).each do |hex|
490 value = hex.to_i(16)
491 codes.concat([value / 64, value / 16 % 4, value / 4 % 4, value % 4])
492 end
493 raise CommandError, 'invalid padding' if codes.drop(a[1]).any? { |c| c != 0 }
494 emit(store(a[0], codes.first(a[1]).map { |c| BASES[c] }.join))
495 return
496 when 'JOIN'
497 emit(store(a[0], a.drop(1).map { |name| @registry[name] }.join))
498 return
499 when 'ASSEMBLE'
500 assemble(a[0], a[1], a.drop(2).map { |name| @registry[name] })
501 return
502 when 'LIST'
503 @registry.keys.sort.each { |name| emit(name, @registry[name].length, 'nt') }
504 emit('no sequences') if @registry.empty?
505 return
506 end
507 name = a[0]
508 s = @registry[name]
509 n = s.length
510 case cmd
511 when 'SLICE'
512 start, finish = a[1, 2]
513 part = start <= finish ? s[(start - 1)...finish] : revcomp(s[(finish - 1)...start])
514 emit(store(a[3], part))
515 when 'MUTATE'
516 position, base = a[1, 2]
517 old = s[position - 1]
518 raise CommandError, 'no change' if old == base
519 @registry[name] = s[0, position - 1] + base + s[position..]
520 emit(name, "#{old}#{position}#{base}", transition?(old, base) ? 'transition' : 'transversion')
521 when 'DEL'
522 @registry.delete(name)
523 emit('deleted', name)
524 when 'REVCOMP'
525 emit(revcomp(s))
526 when 'COMPLEMENT'
527 emit(s.tr('ACGT', 'TGCA'))
528 when 'REVERSE'
529 emit(s.reverse)
530 when 'TRANSCRIBE'
531 emit(s.tr('T', 'U'))
532 when 'TRANSLATE', 'CODONS', 'PROTEIN'
533 translation(cmd, s, a[1])
534 when 'ORFS'
535 orfs(s, a[1])
536 when 'GC'
537 table = s.chars.tally
538 emit(*BASES.each_char.map { |b| "#{b}=#{table.fetch(b, 0)}" }, "GC=#{pct(gc(s), n)}%")
539 when 'GCWINDOW'
540 width, step = a[1, 2]
541 raise CommandError, 'window too large' if width > n
542 0.step(n - width, step) { |i| emit("#{i + 1}-#{i + width}", pct(gc(s[i, width]), width) + '%') }
543 when 'SKEW'
544 values = [0]
545 s.each_char { |b| values << values.last + (b == 'G' ? 1 : 0) - (b == 'C' ? 1 : 0) }
546 [['min', values.min], ['max', values.max]].each do |label, extreme|
547 emit(label, extreme, 'at', *values.each_index.select { |i| values[i] == extreme })
548 end
549 when 'KMERS'
550 width, limit = a[1, 2]
551 raise CommandError, 'k too large' if width > n
552 table = (0..(n - width)).map { |i| s[i, width] }.tally
553 emit(n - width + 1, 'windows', table.length, 'distinct')
554 frequency_order(table).first(limit).each { |word| emit(word, table[word]) }
555 when 'FIND'
556 emit_list('forward', occurrences(s, a[1]))
557 emit_list('reverse', occurrences(s, revcomp(a[1])))
558 when 'RESTRICT'
559 restrict(s, a.drop(1))
560 when 'PALINDROMES'
561 minimum, maximum = a[1, 2]
562 raise CommandError, 'invalid range' if minimum > maximum
563 found = false
564 n.times do |i|
565 (minimum..[maximum, n - i].min).each do |width|
566 part = s[i, width]
567 next unless part == revcomp(part)
568 emit(i + 1, width, part)
569 found = true
570 end
571 end
572 emit('none') unless found
573 when 'REPEATS'
574 repeats(s, *a.drop(1))
575 when 'HAMMING'
576 other = @registry[a[1]]
577 raise CommandError, 'length mismatch' unless other.length == n
578 distance = s.chars.zip(other.chars).count { |x, y| x != y }
579 transitions = s.chars.zip(other.chars).count { |x, y| transition?(x, y) }
580 emit('distance', distance, 'transitions', transitions, 'transversions', distance - transitions)
581 when 'ALIGN'
582 align(s, @registry[a[1]], *a.drop(2))
583 when 'CONSENSUS'
584 consensus(a.map { |id| @registry[id] })
585 when 'TM'
586 emit('Tm', fixed(temperature(s), 1), n < 14 ? 'wallace' : 'gc')
587 when 'MASS'
588 single = strand_mass(s)
589 emit('ss', fixed(single, 3), 'ds', fixed(single + strand_mass(revcomp(s)), 3))
590 when 'PRIMER'
591 primer(s)
592 when 'FOLD'
593 fold(s, a[1])
594 when 'PACK'
595 codes = s.each_char.map { |b| BASES.index(b) }
596 codes << 0 until (codes.length % 4).zero?
597 encoded = codes.each_slice(4).map { |c| format('%02X', 64 * c[0] + 16 * c[1] + 4 * c[2] + c[3]) }.join
598 emit(n, encoded)
599 end
600 end
