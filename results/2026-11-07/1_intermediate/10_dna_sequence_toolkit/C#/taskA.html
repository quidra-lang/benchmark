97 static char complementBase(char c) { return BASES[3-BASES.IndexOf(c)]; }
98 static int selfcomp(string s) {
99 int maximum=0;
100 for (int center=1;center<s.Length;center++) {
101 int left=center-1,right=center;
102 while (left>=0 && right<s.Length && complementBase(s[left])==s[right]) {
103 maximum=Math.Max(maximum,right-left+1);
104 left--;
105 right++;
106 }
107 }
108 return maximum;
109 }
110 static int tm(string s) {
111 int n=s.Length,g=gc(s);

117 static int properOverlap(string a, string b) {
118 string joined=b+"#"+a;
119 int[] prefix=new int[joined.Length];
120 for (int i=1;i<joined.Length;i++) {
121 int j=prefix[i-1];
122 while (j>0 && joined[i]!=joined[j]) j=prefix[j-1];
123 if (joined[i]==joined[j]) j++;
124 prefix[i]=j;
125 }
126 int result=prefix[^1], limit=Math.Min(a.Length,b.Length);
127 while (result>=limit) result=prefix[result-1];
128 return result;
129 }

169 public void validate(string command, string[] a, int[] nums, string[] sequences) {
170 error(!schemas.ContainsKey(command),"unknown command "+command);
171 int[] count=arities[command];
172 error(a.Length<count[0] || a.Length>count[1],"wrong argument count");
173 string[] schema=schemas[command];
174 HashSet<string> enzymes=new();
175 for (int i=0;i<a.Length;i++) {
176 string kind=schema[Math.Min(i,schema.Length-1)], value=a[i];
177 switch (kind) {
178 case "NAME":
179 error(!registry.ContainsKey(value),"unknown sequence "+value);
180 sequences[i]=registry[value];
181 break;
182 case "NEW":
183 error(!matches(value,"[a-z][a-z0-9]{0,9}"),"invalid name "+value);
184 break;
185 case "SEQ":
186 for (int j=0;j<value.Length;j++) error(BASES.IndexOf(value[j])<0,"invalid base "+value[j]+" at "+(j+1));
187 break;
188 case "BASE":
189 error(value.Length!=1 || BASES.IndexOf(value[0])<0,"invalid base "+value);
190 break;
191 case "MOTIF":
192 error(!matches(value,"[ACGTRYSWKMBDHVN]{1,30}"),"invalid motif "+value);
193 break;
194 case "ENZYME":
195 error(!ENZYMES.Contains(value),"unknown enzyme "+value);
196 error(!enzymes.Add(value),"duplicate enzyme "+value);
197 break;
198 case "MODE":
199 error(!value.Equals("GLOBAL") && !value.Equals("LOCAL"),"invalid mode "+value);
200 break;
201 case "HEX":
202 error(value.Length!=2*((nums[1]+3)/4) || !matches(value,"[0-9A-F]+"),"invalid hex");
203 break;
204 case "FRAME":
205 nums[i]=number(value,-3,3);
206 error(nums[i]==0,"invalid number "+value);
207 break;
208 case "POS":
209 nums[i]=number(value,1,sequences[0].Length);
210 break;
211 default:
212 string[] bounds=kind.Split(':');
213 nums[i]=number(value,int.Parse(bounds[0]),int.Parse(bounds[1]));
214 break;
215 }
216 }

218 public void align(string a, string b, bool local, int match, int mismatch, int gap) {
219 int m=a.Length,n=b.Length;
220 int[][] h=Matrix(m+1,n+1);
221 if (!local) {
222 for (int i=0;i<=m;i++) h[i][0]=i*gap;
223 for (int j=0;j<=n;j++) h[0][j]=j*gap;
224 }
225 int best=0,bi=0,bj=0;
226 for (int i=1;i<=m;i++) {
227 for (int j=1;j<=n;j++) {
228 int value=Math.Max(h[i-1][j-1]+(a[i-1]==b[j-1] ? match : mismatch),Math.Max(h[i-1][j]+gap,h[i][j-1]+gap));
229 h[i][j]=local ? Math.Max(0,value) : value;
230 if (local && h[i][j]>best) {
231 best=h[i][j];
232 bi=i;
233 bj=j;
234 }
235 }
236 }
237 if (!local) {
238 bi=m;
239 bj=n;
240 best=h[m][n];
241 }
242 if (local && best==0) {
243 output.Add("score 0");
244 return;
245 }
246 int traceI=bi,traceJ=bj;
247 StringBuilder x=new StringBuilder(),y=new StringBuilder();
248 while (local ? h[traceI][traceJ]!=0 : traceI!=0 || traceJ!=0) {
249 if (traceI>0 && traceJ>0 && h[traceI][traceJ]==h[traceI-1][traceJ-1]+(a[traceI-1]==b[traceJ-1] ? match : mismatch)) {
250 x.Append(a[--traceI]);
251 y.Append(b[--traceJ]);
252 } else if (traceI>0 && h[traceI][traceJ]==h[traceI-1][traceJ]+gap) {
253 x.Append(a[--traceI]);
254 y.Append('-');
255 } else {
256 x.Append('-');
257 y.Append(b[--traceJ]);
258 }
259 }
260 string left=reverse(x.ToString()),right=reverse(y.ToString());
261 StringBuilder middle=new StringBuilder();
262 int identity=0,gaps=0;
263 for (int k=0;k<left.Length;k++) {
264 char c=left[k]=='-' || right[k]=='-' ? '-' : left[k]==right[k] ? '|' : '*';
265 middle.Append(c);
266 if (c=='|') identity++;
267 if (c=='-') gaps++;
268 }
269 output.Add("score "+best+" a "+(traceI+1)+"-"+bi+" b "+(traceJ+1)+"-"+bj+" identity "+identity+"/"+left.Length+" gaps "+gaps);
270 output.Add(left);
271 output.Add(middle.ToString());
272 output.Add(right);
273 }

276 public void fold(string s, int loop) {
277 int n=s.Length;
278 error(n>100,"too long");
279 int[][] f=Matrix(n,n);
280 for (int length=1;length<=n;length++) {
281 for (int i=0;i+length<=n;i++) {
282 int j=i+length-1;
283 f[i][j]=value(f,i,j-1);
284 for (int k=i;k<=j-loop-1;k++) if (pair(s[k],s[j])) f[i][j]=Math.Max(f[i][j],value(f,i,k-1)+1+value(f,k+1,j-1));
285 }
286 }
287 char[] structure=new char[n];
288 Array.Fill(structure,'.');
289 Stack<int[]> intervals=new();
290 intervals.Push(new int[]{0,n-1});
291 while (intervals.Count!=0) {
292 int[] interval=intervals.Pop();
293 int i=interval[0],j=interval[1];
294 if (value(f,i,j)==0) continue;
295 if (f[i][j]==value(f,i,j-1)) intervals.Push(new int[]{i,j-1});
296 else {
297 for (int k=i;k<=j-loop-1;k++) {
298 if (pair(s[k],s[j]) && value(f,i,k-1)+1+value(f,k+1,j-1)==f[i][j]) {
299 structure[k]='(';
300 structure[j]=')';
301 intervals.Push(new int[]{i,k-1});
302 intervals.Push(new int[]{k+1,j-1});
303 break;
304 }
305 }
306 }
307 }
308 output.Add(new string(structure));
309 output.Add("pairs "+f[0][n-1]);
310 }
311 public void primer(string s) {

313 for (int i=1;i<n;i++) {

329 public void assemble(string name, int minimum, string[] sequences) {
330 List<string> unique=sequences.Skip(2).Distinct(StringComparer.Ordinal).ToList();
331 List<string> fragments=unique.Where(s => !unique.Any(t => t!=s && t.Contains(s))).ToList();
332 while (fragments.Count>=2) {
333 int best=-1,bx=0,by=0;
334 for (int x=0;x<fragments.Count;x++) {
335 for (int y=0;y<fragments.Count;y++) {
336 if (x==y) continue;
337 string a=fragments[x],b=fragments[y];
338 int overlap=properOverlap(a,b);
339 if (overlap>best) {
340 best=overlap;
341 bx=x;
342 by=y;
343 }
344 }
345 }
346 if (best<minimum) break;
347 string merged=fragments[bx]+fragments[by].Substring(best);
348 fragments[bx]=merged;
349 fragments.RemoveAt(by);
350 int keep=bx-(by<bx ? 1 : 0);
351 for (int i=fragments.Count-1;i>=0;i--) {
352 if (i!=keep && merged.Contains(fragments[i])) {
353 fragments.RemoveAt(i);
354 if (i<keep) keep--;
355 }
356 }
357 }
358 string longest=fragments.MaxBy(s => s.Length)!;
359 error(longest.Length>500,"too long");
360 output.Add("contigs "+fragments.Count);
361 foreach (string s in fragments) output.Add(s.Length+" "+s);
362 store(name,longest);
363 }
364 public void execute(string line) {
365 string[] t=line.Split(' '),a=t.Skip(1).ToArray();
366 int[] nums=new int[a.Length];
367 string[] sequences=new string[a.Length];
368 string command=t[0];
369 validate(command,a,nums,sequences);
370 string s=a.Length==0 ? "" : sequences[0];
371 int n=s==null ? 0 : s.Length;
372 switch (command) {
373 case "SEQ":
374 store(a[0],a[1]);
375 break;
376 case "SLICE": {
377 int left=Math.Min(nums[1],nums[2]),right=Math.Max(nums[1],nums[2]);
378 string part=s.Substring(left-1,right-left+1);
379 store(a[3],nums[1]<=nums[2] ? part : revcomp(part));
380 break;
381 }
382 case "JOIN":
383 store(a[0],string.Join("",sequences.Skip(1)));
384 break;
385 case "MUTATE": {
386 char old=s[nums[1]-1],next=a[2][0];
387 error(old==next,"no change");
388 registry[a[0]]=s.Substring(0,nums[1]-1)+next+s.Substring(nums[1]);
389 output.Add(a[0]+" "+old+nums[1]+next+" "+(transition(old,next) ? "transition" : "transversion"));
390 break;
391 }
392 case "DEL":
393 registry.Remove(a[0]);
394 output.Add("deleted "+a[0]);
395 break;
396 case "LIST":
397 if (registry.Count==0) output.Add("no sequences");
398 else foreach (string name in registry.Keys) output.Add(name+" "+registry[name].Length+" nt");
399 break;
400 case "REVCOMP":
401 output.Add(revcomp(s));
402 break;
403 case "COMPLEMENT":
404 output.Add(complement(s));
405 break;
406 case "REVERSE":
407 output.Add(reverse(s));
408 break;
409 case "TRANSCRIBE":
410 output.Add(s.Replace('T','U'));
411 break;
412 case "TRANSLATE": {
413 string protein=translation(makeCodons(s,nums[1]));
414 output.Add(protein.Length==0 ? "none" : protein);
415 break;
416 }
417 case "CODONS": {
418 Dictionary<string,int> counts=wordCounts(makeCodons(s,nums[1]));
419 if (counts.Count==0) output.Add("none");
420 foreach (string c in frequency(counts)) output.Add(c+" "+GENETIC[c]+" "+counts[c]);
421 break;
422 }
423 case "PROTEIN": {
424 string protein=translation(makeCodons(s,nums[1])).Split('*')[0];
425 error(protein.Length==0,"empty protein");
426 int mass=18015+protein.Sum(c => RESIDUES[AMINO.IndexOf(c)]);
427 int charge=protein.Count(c => "KR".Contains(c))-protein.Count(c => "DE".Contains(c));
428 Dictionary<string,int> counts=wordCounts(protein.Select(c => c.ToString()));
429 output.Add(protein);
430 output.Add("length "+protein.Length+" mass "+milli(mass)+" charge "+(charge>0 ? "+" : "")+charge);
431 output.Add(string.Join(" ",frequency(counts).Select(c => c+"="+counts[c])));
432 break;
433 }
434 case "ORFS": {
435 List<Orf> found=new();
436 foreach (int frame in new int[]{1,2,3,-1,-2,-3}) {
437 List<string> codons=makeCodons(s,frame);
438 int start=-1;
439 for (int i=0;i<codons.Count;i++) {
440 if (start<0 && codons[i].Equals("ATG")) start=i;
441 else if (start>=0 && GENETIC[codons[i]]=='*') {
442 if (i-start>=nums[1]) {
443 int left=Math.Abs(frame)-1+3*start,right=Math.Abs(frame)-1+3*(i+1)-1;
444 found.Add(new Orf(frame,frame>0 ? left+1 : n-right,frame>0 ? right+1 : n-left,translation(codons.GetRange(start,i-start))));
445 }
446 start=-1;
447 }
448 }
449 }
450 found=found.OrderBy(o => o.left).ToList();
451 if (found.Count==0) output.Add("none");
452 foreach (Orf o in found) output.Add((o.frame>0 ? "+" : "")+o.frame+" "+o.left+" "+o.right+" "+o.protein.Length+" "+o.protein);
453 break;
454 }
455 case "GC": {
456 string lineOut=string.Join(" ",BASES.Select(c => c+"="+s.Count(x => x==c)));
457 output.Add(lineOut+" GC="+pct(gc(s),n)+"%");
458 break;
459 }
460 case "GCWINDOW":
461 error(nums[1]>n,"window too large");
462 for (int i=0;i+nums[1]<=n;i+=nums[2]) output.Add((i+1)+"-"+(i+nums[1])+" "+pct(gc(s.Substring(i,nums[1])),nums[1])+"%");
463 break;
464 case "SKEW": {
465 int[] skew=new int[n+1];
466 for (int i=0;i<n;i++) skew[i+1]=skew[i]+(s[i]=='G' ? 1 : s[i]=='C' ? -1 : 0);
467 foreach (bool minimum in new bool[]{true,false}) {
468 int v=minimum ? skew.Min() : skew.Max();
469 List<int> at=Enumerable.Range(0,n+1).Where(i => skew[i]==v).ToList();
470 output.Add((minimum ? "min " : "max ")+v+" at "+list(at));
471 }
472 break;
473 }
474 case "KMERS": {
475 error(nums[1]>n,"k too large");
476 List<string> words=Enumerable.Range(0,n-nums[1]+1).Select(i => s.Substring(i,nums[1])).ToList();
477 Dictionary<string,int> counts=wordCounts(words);
478 output.Add(words.Count+" windows "+counts.Count+" distinct");
479 foreach (string word in frequency(counts).Take(nums[2])) output.Add(word+" "+counts[word]);
480 break;
481 }
482 case "FIND":
483 output.Add("forward "+list(occurrences(s,a[1])));
484 output.Add("reverse "+list(occurrences(s,revcomp(a[1]))));
485 break;
486 case "RESTRICT": {
487 SortedSet<int> cuts=new();
488 for (int i=1;i<a.Length;i++) {
489 int e=Array.IndexOf(ENZYMES,a[i]);
490 List<int> found=new();
491 foreach (int p in occurrences(s,SITES[e])) {
492 int c=p+CUTS[e]-1;
493 if (c>=1 && c<n) {
494 found.Add(c);
495 cuts.Add(c);
496 }
497 }
498 output.Add(a[i]+" "+list(found));
499 }
500 List<string> fragments=new();
501 List<int> sizes=new(),ends=new(cuts);
502 ends.Add(n);
503 int previous=0;
504 foreach (int end in ends) {
505 fragments.Add((previous+1)+"-"+end);
506 sizes.Add(end-previous);
507 previous=end;
508 }
509 sizes.Sort((leftSize,rightSize) => rightSize.CompareTo(leftSize));
510 output.Add("fragments "+list(fragments));
511 output.Add("sizes "+list(sizes));
512 break;
513 }
514 case "PALINDROMES": {
515 error(nums[1]>nums[2],"invalid range");
516 int count=0;
517 for (int i=0;i<n;i++) {
518 for (int length=nums[1];length<=nums[2] && i+length<=n;length++) {
519 string part=s.Substring(i,length);
520 if (part.Equals(revcomp(part))) {
521 output.Add((i+1)+" "+length+" "+part);
522 count++;
523 }
524 }
525 }
526 if (count==0) output.Add("none");
527 break;
528 }
529 case "REPEATS": {
530 error(nums[1]>nums[2],"invalid range");
531 List<Repeat> found=new();
532 for (int u=nums[1];u<=nums[2];u++) {
533 int i=0;
534 while (i+u<=n) {
535 string unit=s.Substring(i,u);
536 int count=repeatCount(s,i,u);
537 if (primitive(unit) && count>=nums[3]) {
538 found.Add(new Repeat(i+1,i+count*u,u,unit,count));
539 i+=count*u;
540 } else i++;
541 }
542 }
543 found=found.OrderBy(r => r.start).ThenBy(r => r.unitLength).ToList();
544 if (found.Count==0) output.Add("none");
545 foreach (Repeat r in found) output.Add(r.start+"-"+r.end+" "+r.unit+"x"+r.count);
546 break;
547 }
548 case "HAMMING": {
549 string b=sequences[1];
550 error(n!=b.Length,"length mismatch");
551 int distance=0,transitions=0;
552 for (int i=0;i<n;i++) {
553 if (s[i]!=b[i]) {
554 distance++;
555 if (transition(s[i],b[i])) transitions++;
556 }
557 }
558 output.Add("distance "+distance+" transitions "+transitions+" transversions "+(distance-transitions));
559 break;
560 }
561 case "ALIGN":
562 align(s,sequences[1],a[2].Equals("LOCAL"),nums[3],nums[4],nums[5]);
563 break;
564 case "CONSENSUS": {
565 error(sequences.Any(sequence => sequence.Length!=n),"length mismatch");
566 int[][] profile=Matrix(4,n);
567 StringBuilder consensus=new StringBuilder(),iupac=new StringBuilder();
568 for (int i=0;i<n;i++) {
569 foreach (string sequence in sequences) profile[BASES.IndexOf(sequence[i])][i]++;
570 int maximum=profile.Max(row => row[i]),mask=0;
571 bool chosen=false;
572 for (int b=0;b<4;b++) {
573 if (profile[b][i]==maximum) {
574 mask|=1<<b;
575 if (!chosen) {
576 consensus.Append(BASES[b]);
577 chosen=true;
578 }
579 }
580 }
581 iupac.Append(SYMBOLS[Array.IndexOf(MASKS,mask)]);
582 }
583 for (int b=0;b<4;b++) output.Add(BASES[b]+" "+list(profile[b]));
584 output.Add("consensus "+consensus);
585 output.Add("iupac "+iupac);
586 break;
587 }
588 case "TM":
589 output.Add("Tm "+formatFixed(tm(s),10,1)+" "+(n<14 ? "wallace" : "gc"));
590 break;
591 case "MASS":
592 output.Add("ss "+milli(mass(s))+" ds "+milli(mass(s)+mass(revcomp(s))));
593 break;
594 case "PRIMER":
595 primer(s);
596 break;
597 case "FOLD":
598 fold(s,nums[1]);
599 break;
600 case "ASSEMBLE":
601 assemble(a[0],nums[1],sequences);
602 break;
603 case "PACK": {
604 StringBuilder hex=new StringBuilder();
605 for (int i=0;i<n;i+=4) {
606 int value=0;
607 for (int j=0;j<4;j++) value=4*value+(i+j<n ? BASES.IndexOf(s[i+j]) : 0);
608 hex.Append(value.ToString("X2",CultureInfo.InvariantCulture));
609 }
610 output.Add(n+" "+hex);
611 break;
612 }
613 case "UNPACK": {
614 StringBuilder decoded=new StringBuilder();
615 foreach (byte value in Convert.FromHexString(a[2])) {
616 foreach (int shift in new int[]{6,4,2,0}) decoded.Append(BASES[value>>shift&3]);
617 }
618 for (int i=nums[1];i<decoded.Length;i++) error(decoded[i]!='A',"invalid padding");
619 store(a[0],decoded.ToString(0,nums[1]));
620 break;
621 }
622 }
623 }
624 public List<string> process(string line) {
625 output=new();
626 try { execute(line); }
627 catch (Fault fault) {
628 output.Clear();
629 output.Add("ERROR "+fault.Message);
630 }
631 return output;
632 }
