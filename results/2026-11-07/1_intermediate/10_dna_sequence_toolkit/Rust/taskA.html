6fn complement(base: u8) -> u8 {
7 match base {
8 b'A' => b'T',
9 b'C' => b'G',
10 b'G' => b'C',
11 b'T' => b'A',
12 b'R' => b'Y',
13 b'Y' => b'R',
14 b'W' => b'W',
15 b'S' => b'S',
16 b'K' => b'M',
17 b'M' => b'K',
18 b'B' => b'V',
19 b'D' => b'H',
20 b'H' => b'D',
21 b'V' => b'B',
22 b'N' => b'N',
23 _ => unreachable!("validated DNA or IUPAC symbol"),
24 }
25}

117fn arity(command: &str) -> Option<(usize, usize)> {
118 match command {
119 "SEQ" | "TRANSLATE" | "CODONS" | "PROTEIN" | "ORFS" | "FIND" |
120 "HAMMING" | "FOLD" => Some((2, 2)),
121 "SLICE" | "REPEATS" => Some((4, 4)),
122 "JOIN" => Some((3, 9)),
123 "MUTATE" | "GCWINDOW" | "KMERS" | "PALINDROMES" | "UNPACK" => Some((3, 3)),
124 "DEL" | "REVCOMP" | "COMPLEMENT" | "REVERSE" | "TRANSCRIBE" |
125 "GC" | "SKEW" | "TM" | "MASS" | "PRIMER" | "PACK" => Some((1, 1)),
126 "LIST" => Some((0, 0)),
127 "RESTRICT" => Some((2, 7)),
128 "ALIGN" => Some((6, 6)),
129 "CONSENSUS" => Some((2, 10)),
130 "ASSEMBLE" => Some((4, 10)),
131 _ => None,
132 }

135fn kinds(command: &str, count: usize) -> Vec<&'static str> {
136 match command {
137 "SEQ" => vec!["new", "sequence"],
138 "SLICE" => vec!["name", "position", "position", "new"],
139 "MUTATE" => vec!["name", "position", "base"],
140 "TRANSLATE" | "CODONS" | "PROTEIN" => vec!["name", "frame"],
141 "ORFS" => vec!["name", "orf"],
142 "GCWINDOW" => vec!["name", "window", "window"],
143 "KMERS" => vec!["name", "kmer", "limit"],
144 "FIND" => vec!["name", "motif"],
145 "PALINDROMES" => vec!["name", "palindrome", "palindrome"],
146 "REPEATS" => vec!["name", "repeat", "repeat", "repeat_count"],
147 "ALIGN" => vec!["name", "name", "mode", "match", "mismatch", "gap"],
148 "FOLD" => vec!["name", "loop"],
149 "UNPACK" => vec!["new", "length", "hex"],
150 _ => {
151 let mut result = vec!["name"; count];
152 if command == "JOIN" || command == "ASSEMBLE" {
153 result[0] = "new";
154 }
155 if command == "ASSEMBLE" {
156 result[1] = "overlap";
157 }
158 if command == "RESTRICT" {
159 for item in &mut result[1..] {
160 *item = "enzyme";
161 }
162 }
163 result
164 }
165 }

168fn number(text: &str, minimum: i32, maximum: i32, nonzero: bool) -> Result<i32, String> {
169 let digits = text.strip_prefix('-').unwrap_or(text);
170 if digits.is_empty() || digits.len() > 6 || !digits.bytes().all(|c| c.is_ascii_digit()) {
171 return Err(format!("invalid number {text}"));
172 }
173 let value = text.parse::<i32>().map_err(|_| format!("invalid number {text}"))?;
174 if value < minimum || value > maximum || (nonzero && value == 0) {
175 return Err(format!("invalid number {text}"));
176 }
177 Ok(value)
178}

192fn fixed(value: i32, places: u32) -> String {
193 let scale = 10_i32.pow(places);
194 format!("{}.{:0width$}", value / scale, value % scale, width = places as usize)
195}

201fn transition(a: u8, b: u8) -> bool {
202 a != b && matches!(
203 (a, b), (b'A', b'G') | (b'G', b'A') | (b'C', b'T') | (b'T', b'C')
204 )

207fn frequency_order(counts: &BTreeMap<String, usize>) -> Vec<String> {
208 let mut names: Vec<String> = counts.keys().cloned().collect();
209 names.sort_by(|a, b| counts[b].cmp(&counts[a]).then_with(|| a.cmp(b)));
210 names
211}

213fn codons(sequence: &str, frame: i32) -> Vec<String> {
214 let strand = if frame > 0 { sequence.to_owned() } else { revcomp(sequence) };
215 let offset = frame.unsigned_abs() as usize - 1;
216 (offset..strand.len().saturating_sub(2))
217 .step_by(3)
218 .map(|i| strand[i..i + 3].to_owned())
219 .collect()
220}
221
222fn occurrences(sequence: &str, motif: &str) -> Vec<usize> {
223 let mut positions = Vec::new();
224 if motif.len() > sequence.len() {
225 return positions;
226 }
227 for i in 0..=(sequence.len() - motif.len()) {
228 if motif.bytes().enumerate().all(|(j, symbol)| {
229 iupac(symbol).expect("validated motif").contains(&sequence.as_bytes()[i + j])
230 }) {
231 positions.push(i + 1);
232 }
233 }
234 positions
235}

237fn repeat_count(sequence: &str, start: usize, width: usize) -> usize {
238 let unit = &sequence[start..start + width];
239 let mut count = 0;
240 while start + (count + 1) * width <= sequence.len() &&
241 &sequence[start + count * width..start + (count + 1) * width] == unit {
242 count += 1;
243 }
244 count
245}

247fn primitive(unit: &str) -> bool {
248 (1..unit.len()).all(|width| {
249 unit.len() % width != 0 ||
250 (width..unit.len()).any(|i| unit.as_bytes()[i] != unit.as_bytes()[i % width])
251 })
252}
253
254fn longest_palindrome(sequence: &str) -> usize {
255 let mut longest = 0;
256 for center in 1..sequence.len() {
257 let mut left = center as isize - 1;
258 let mut right = center;
259 while left >= 0 && right < sequence.len() &&
260 complement(sequence.as_bytes()[left as usize]) == sequence.as_bytes()[right] {
261 longest = longest.max(right - left as usize + 1);
262 left -= 1;
263 right += 1;
264 }
265 }
266 longest
267}

269fn temperature(sequence: &str) -> i32 {
270 let n = sequence.len() as i32;
271 let amount = gc(sequence) as i32;
272 if n < 14 {
273 10 * (2 * (n - amount) + 4 * amount)
274 } else {
275 (649 * n + 410 * amount - 6724) / n
276 }
277}

279fn strand_mass(sequence: &str) -> i32 {
280 sequence.bytes().map(base_mass).sum::<i32>() - (sequence.len() as i32 - 1) * 18015
281}

305 fn validate(&self, command: &str, args: &[&str]) -> Result<(), String> {
306 let (minimum, maximum) =
307 arity(command).ok_or_else(|| format!("unknown command {command}"))?;
308 if args.len() < minimum || args.len() > maximum {
309 return Err("wrong argument count".to_owned());
310 }
311 let mut seen = HashSet::new();
312 for (i, (kind, text)) in kinds(command, args.len()).iter().zip(args.iter()).enumerate() {
313 let text = *text;
314 match *kind {
315 "position" | "frame" | "orf" | "window" | "kmer" | "limit" |
316 "palindrome" | "repeat" | "repeat_count" | "match" |
317 "mismatch" | "gap" | "loop" | "overlap" | "length" => {
318 let (min, max) = match *kind {
319 "position" => (1, self.sequences[args[0]].len() as i32),
320 "frame" => (-3, 3),
321 "orf" => (1, 200),
322 "kmer" => (1, 12),
323 "limit" | "overlap" => (1, 100),
324 "palindrome" => (2, 20),
325 "repeat" => (1, 6),
326 "repeat_count" => (2, 50),
327 "match" => (1, 20),
328 "mismatch" => (-20, 0),
329 "gap" => (-20, -1),
330 "loop" => (0, 10),
331 _ => (1, 500),
332 };
333 number(text, min, max, *kind == "frame")?;
334 }
335 "name" => {
336 if !self.sequences.contains_key(text) {
337 return Err(format!("unknown sequence {text}"));
338 }
339 }
340 "new" => {
341 let bytes = text.as_bytes();
342 if bytes.is_empty() || bytes.len() > 10 ||
343 !(b'a'..=b'z').contains(&bytes[0]) ||
344 !bytes.iter().all(|&c| c.is_ascii_lowercase() || c.is_ascii_digit()) {
345 return Err(format!("invalid name {text}"));
346 }
347 }
348 "sequence" => {
349 for (position, base) in text.bytes().enumerate() {
350 if !BASES.contains(&base) {
351 return Err(format!("invalid base {} at {}", base as char, position + 1));
352 }
353 }
354 }
355 "base" => {
356 if text.len() != 1 || !BASES.contains(&text.as_bytes()[0]) {
357 return Err(format!("invalid base {text}"));
358 }
359 }
360 "motif" => {
361 if text.is_empty() || text.len() > 30 ||
362 !text.bytes().all(|c| iupac(c).is_some()) {
363 return Err(format!("invalid motif {text}"));
364 }
365 }
366 "enzyme" => {
367 if enzyme(text).is_none() {
368 return Err(format!("unknown enzyme {text}"));
369 }
370 if !seen.insert(text) {
371 return Err(format!("duplicate enzyme {text}"));
372 }
373 }
374 "mode" => {
375 if text != "GLOBAL" && text != "LOCAL" {
376 return Err(format!("invalid mode {text}"));
377 }
378 }
379 "hex" => {
380 let length = args[1].parse::<usize>().expect("validated preceding length");
381 if text.len() != 2 * length.div_ceil(4) ||
382 !text.bytes().all(|c| c.is_ascii_digit() || (b'A'..=b'F').contains(&c)) {
383 return Err("invalid hex".to_owned());
384 }
385 }
386 _ => unreachable!("all argument kinds accounted for"),
387 }
388 let _ = i;
389 }
390 Ok(())
391 }

393 fn store(&mut self, name: &str, sequence: String) -> Result<String, String> {
394 if sequence.len() > 500 {
395 return Err("too long".to_owned());
396 }
397 let action = if self.sequences.contains_key(name) { "replaced" } else { "stored" };
398 let result = format!("{action} {name} {} nt", sequence.len());
399 self.sequences.insert(name.to_owned(), sequence);
400 Ok(result)
401 }

403 fn translation(&mut self, command: &str, name: &str, frame: i32) -> Result<(), String> {
404 let triplets = codons(&self.sequences[name], frame);
405 let mut protein: String = triplets.iter().map(|c| amino(c)).collect();
406 match command {
407 "TRANSLATE" => self.emit(if protein.is_empty() { "none" } else { &protein }),
408 "CODONS" => {
409 let mut counts = BTreeMap::new();
410 for codon in triplets {
411 *counts.entry(codon).or_insert(0_usize) += 1;
412 }
413 if counts.is_empty() {
414 self.emit("none");
415 }
416 for codon in frequency_order(&counts) {
417 self.emit(format!("{codon} {} {}", amino(&codon), counts[&codon]));
418 }
419 }
420 _ => {
421 if let Some(stop) = protein.find('*') {
422 protein.truncate(stop);
423 }
424 if protein.is_empty() {
425 return Err("empty protein".to_owned());
426 }
427 let mut counts: BTreeMap<String, usize> = BTreeMap::new();
428 let mut mass = 18015;
429 let mut charge = 0;
430 for letter in protein.bytes() {
431 *counts.entry((letter as char).to_string()).or_insert(0) += 1;
432 mass += amino_mass(letter);
433 charge += i32::from(letter == b'K' || letter == b'R');
434 charge -= i32::from(letter == b'D' || letter == b'E');
435 }
436 self.emit(protein.clone());
437 let charge_text = if charge > 0 { format!("+{charge}") } else { charge.to_string() };
438 self.emit(format!(
439 "length {} mass {} charge {}", protein.len(), fixed(mass, 3), charge_text
440 ));
441 let composition = frequency_order(&counts).iter()
442 .map(|amino| format!("{amino}={}", counts[amino]))
443 .collect::<Vec<_>>().join(" ");
444 self.emit(composition);
445 }
446 }
447 Ok(())
448 }

450 fn orfs(&mut self, sequence: &str, minimum: usize) {
451 let mut rows: Vec<(usize, usize, i32, String)> = Vec::new();
452 for frame in [1_i32, 2, 3, -1, -2, -3] {
453 let triplets = codons(sequence, frame);
454 let mut start: Option<usize> = None;
455 for (position, codon) in triplets.iter().enumerate() {
456 if start.is_none() {
457 if codon == "ATG" {
458 start = Some(position);
459 }
460 } else if amino(codon) == '*' {
461 let first = start.expect("inside an ORF");
462 if position - first >= minimum {
463 let mut left = frame.unsigned_abs() as usize - 1 + 3 * first;
464 let mut right = frame.unsigned_abs() as usize - 1 + 3 * (position + 1);
465 if frame < 0 {
466 (left, right) = (sequence.len() - right, sequence.len() - left);
467 }
468 let protein: String = triplets[first..position]
469 .iter().map(|codon| amino(codon)).collect();
470 rows.push((left + 1, right, frame, protein));
471 }
472 start = None;
473 }
474 }
475 }
476 rows.sort();
477 if rows.is_empty() {
478 self.emit("none");
479 }
480 for (left, right, frame, protein) in rows {
481 self.emit(format!(
482 "{frame:+} {left} {right} {} {protein}", protein.len()
483 ));
484 }
485 }

487 fn restriction(&mut self, sequence: &str, selected: &[&str]) {
488 let mut all_cuts = BTreeSet::new();
489 for &name in selected {
490 let (motif, offset) = enzyme(name).expect("validated enzyme");
491 let mut cuts = Vec::new();
492 for position in occurrences(sequence, motif) {
493 let cut = position as i32 + offset - 1;
494 if cut >= 1 && cut < sequence.len() as i32 {
495 cuts.push(cut);
496 all_cuts.insert(cut);
497 }
498 }
499 self.emit(format!("{name} {}", list_text(&cuts)));
500 }
501 let mut endpoints = vec![0];
502 endpoints.extend(all_cuts);
503 endpoints.push(sequence.len() as i32);
504 let mut fragments = Vec::new();
505 let mut sizes = Vec::new();
506 for pair in endpoints.windows(2) {
507 fragments.push(format!("{}-{}", pair[0] + 1, pair[1]));
508 sizes.push(pair[1] - pair[0]);
509 }
510 sizes.sort_by(|a, b| b.cmp(a));
511 self.emit(format!("fragments {}", list_text(&fragments)));
512 self.emit(format!("sizes {}", list_text(&sizes)));
513 }

515 fn repeats(&mut self, sequence: &str, minimum: usize, maximum: usize, count: usize)
516 -> Result<(), String>
517 {
518 if minimum > maximum {
519 return Err("invalid range".to_owned());
520 }
521 let mut rows: Vec<(usize, usize, usize, String)> = Vec::new();
522 for width in minimum..=maximum {
523 let mut position = 0;
524 while position + width <= sequence.len() {
525 let unit = &sequence[position..position + width];
526 let repeated = repeat_count(sequence, position, width);
527 if primitive(unit) && repeated >= count {
528 rows.push((position, width, repeated, unit.to_owned()));
529 position += repeated * width;
530 } else {
531 position += 1;
532 }
533 }
534 }
535 rows.sort();
536 if rows.is_empty() {
537 self.emit("none");
538 }
539 for (position, width, repeated, unit) in rows {
540 self.emit(format!(
541 "{}-{} {unit}x{repeated}", position + 1, position + width * repeated
542 ));
543 }
544 Ok(())
545 }

547 fn alignment(&mut self, a: &str, b: &str, mode: &str,
548 match_score: i32, mismatch: i32, gap: i32) {
549 let m = a.len();
550 let n = b.len();
551 let local = mode == "LOCAL";
552 let mut table = vec![vec![0; n + 1]; m + 1];
553 if !local {
554 for (i, row) in table.iter_mut().enumerate() {
555 row[0] = i as i32 * gap;
556 }
557 for j in 0..=n {
558 table[0][j] = j as i32 * gap;
559 }
560 }
561 let (mut best, mut end_i, mut end_j) = (0, 0, 0);
562 for i in 1..=m {
563 for j in 1..=n {
564 let score = if a.as_bytes()[i-1] == b.as_bytes()[j-1] {
565 match_score
566 } else {
567 mismatch
568 };
569 let candidate = (table[i-1][j-1] + score)
570 .max(table[i-1][j] + gap)
571 .max(table[i][j-1] + gap);
572 table[i][j] = if local { candidate.max(0) } else { candidate };
573 if local && table[i][j] > best {
574 (best, end_i, end_j) = (table[i][j], i, j);
575 }
576 }
577 }
578 if !local {
579 (best, end_i, end_j) = (table[m][n], m, n);
580 }
581 if local && best == 0 {
582 self.emit("score 0");
583 return;
584 }
585 let (mut i, mut j) = (end_i, end_j);
586 let (mut row_a, mut row_b) = (Vec::new(), Vec::new());
587 while (local && table[i][j] != 0) || (!local && (i != 0 || j != 0)) {
588 let score = if i > 0 && j > 0 && a.as_bytes()[i-1] == b.as_bytes()[j-1] {
589 match_score
590 } else {
591 mismatch
592 };
593 if i > 0 && j > 0 && table[i][j] == table[i-1][j-1] + score {
594 i -= 1;
595 j -= 1;
596 row_a.push(a.as_bytes()[i]);
597 row_b.push(b.as_bytes()[j]);
598 } else if i > 0 && table[i][j] == table[i-1][j] + gap {
599 i -= 1;
600 row_a.push(a.as_bytes()[i]);
601 row_b.push(b'-');
602 } else {
603 j -= 1;
604 row_a.push(b'-');
605 row_b.push(b.as_bytes()[j]);
606 }
607 }
608 row_a.reverse();
609 row_b.reverse();
610 let mut marks = Vec::new();
611 for (&left, &right) in row_a.iter().zip(&row_b) {
612 marks.push(if left == b'-' || right == b'-' { b'-' }
613 else if left == right { b'|' } else { b'*' });
614 }
615 let identical = marks.iter().filter(|&&mark| mark == b'|').count();
616 let gaps = marks.iter().filter(|&&mark| mark == b'-').count();
617 self.emit(format!(
618 "score {best} a {}-{end_i} b {}-{end_j} identity {identical}/{} gaps {gaps}",
619 i + 1, j + 1, marks.len()
620 ));
621 self.emit(String::from_utf8(row_a).expect("ASCII"));
622 self.emit(String::from_utf8(marks).expect("ASCII"));
623 self.emit(String::from_utf8(row_b).expect("ASCII"));
624 }

626 fn consensus(&mut self, names: &[&str]) -> Result<(), String> {
627 let n = self.sequences[names[0]].len();
628 if names.iter().any(|name| self.sequences[*name].len() != n) {
629 return Err("length mismatch".to_owned());
630 }
631 let mut profile: [Vec<usize>; 4] = std::array::from_fn(|_| Vec::new());
632 let (mut consensus, mut ambiguous) = (String::new(), String::new());
633 for position in 0..n {
634 let mut counts = [0_usize; 4];
635 for name in names {
636 let base = self.sequences[*name].as_bytes()[position];
637 let index = BASES.iter().position(|&c| c == base).expect("validated sequence");
638 counts[index] += 1;
639 }
640 let largest = *counts.iter().max().expect("four base counts");
641 let mut tied = String::new();
642 for (i, &base) in BASES.iter().enumerate() {
643 profile[i].push(counts[i]);
644 if counts[i] == largest {
645 tied.push(base as char);
646 }
647 }
648 consensus.push(tied.chars().next().expect("nonempty base set"));
649 ambiguous.push(symbol(&tied));
650 }
651 for (i, counts) in profile.iter().enumerate() {
652 self.emit(format!("{} {}", BASES[i] as char, list_text(counts)));
653 }
654 self.emit(format!("consensus {consensus}"));
655 self.emit(format!("iupac {ambiguous}"));
656 Ok(())
657 }

659 fn primer(&mut self, sequence: &str) {
660 let n = sequence.len();
661 let amount = gc(sequence);
662 let tm = temperature(sequence);
663 let clamp = gc(&sequence[n.saturating_sub(5)..]);
664 let mut longest_run = 0;
665 let mut run = 0;
666 let mut previous = 0;
667 for base in sequence.bytes() {
668 run = if base == previous { run + 1 } else { 1 };
669 longest_run = longest_run.max(run);
670 previous = base;
671 }
672 let mut repeated = 0;
673 for i in 0..n.saturating_sub(1) {
674 if sequence.as_bytes()[i] != sequence.as_bytes()[i+1] {
675 repeated = repeated.max(repeat_count(sequence, i, 2));
676 }
677 }
678 let selfcomp = longest_palindrome(sequence);
679 let checks = [
680 ("length", n.to_string(), (18..=25).contains(&n)),
681 ("gc", percentage(amount, n), 40*n <= 100*amount && 100*amount <= 60*n),
682 ("tm", fixed(tm, 1), (520..=620).contains(&tm)),
683 ("clamp", clamp.to_string(), (1..=3).contains(&clamp)),
684 ("run", longest_run.to_string(), longest_run <= 4),
685 ("repeat", repeated.to_string(), repeated <= 4),
686 ("selfcomp", selfcomp.to_string(), selfcomp <= 6),
687 ];
688 let mut failed = Vec::new();
689 for (label, value, passed) in checks {
690 self.emit(format!("{label} {value} {}", if passed { "OK" } else { "FAIL" }));
691 if !passed {
692 failed.push(label);
693 }
694 }
695 self.emit(if failed.is_empty() {
696 "PASS".to_owned()
697 } else {
698 format!("FAIL {}", failed.join(" "))
699 });
700 }

702 fn paired(a: u8, b: u8) -> bool {
703 matches!((a, b),
704 (b'A', b'T') | (b'T', b'A') | (b'C', b'G') | (b'G', b'C') |
705 (b'G', b'T') | (b'T', b'G'))
706 }

708 fn cell(table: &[Vec<i32>], i: isize, j: isize) -> i32 {
709 if i > j { 0 } else { table[i as usize][j as usize] }
710 }

712 fn folding(&mut self, sequence: &str, loop_size: usize) -> Result<(), String> {
713 let n = sequence.len();
714 if n > 100 {
715 return Err("too long".to_owned());
716 }
717 let mut table = vec![vec![0; n]; n];
718 let bases = sequence.as_bytes();
719 for width in 2..=n {
720 for i in 0..=(n - width) {
721 let j = i + width - 1;
722 let mut best = Self::cell(&table, i as isize, (j - 1) as isize);
723 let limit = j.saturating_sub(loop_size);
724 for k in i..limit {
725 if Self::paired(bases[k], bases[j]) {
726 let candidate = Self::cell(&table, i as isize, k as isize - 1) + 1 +
727 Self::cell(&table, k as isize + 1, j as isize - 1);
728 best = best.max(candidate);
729 }
730 }
731 table[i][j] = best;
732 }
733 }
734 let mut structure = vec![b'.'; n];
735 let mut pending = vec![(0_isize, n as isize - 1)];
736 while let Some((i, j)) = pending.pop() {
737 if i > j || Self::cell(&table, i, j) == 0 {
738 continue;
739 }
740 if Self::cell(&table, i, j) == Self::cell(&table, i, j - 1) {
741 pending.push((i, j - 1));
742 continue;
743 }
744 let limit = (j - loop_size as isize).max(i);
745 for k in i..limit {
746 let k_usize = k as usize;
747 if Self::paired(bases[k_usize], bases[j as usize]) &&
748 Self::cell(&table, i, k-1) + 1 + Self::cell(&table, k+1, j-1) ==
749 Self::cell(&table, i, j) {
750 structure[k_usize] = b'(';
751 structure[j as usize] = b')';
752 pending.push((i, k-1));
753 pending.push((k+1, j-1));
754 break;
755 }
756 }
757 }
758 self.emit(String::from_utf8(structure).expect("ASCII"));
759 self.emit(format!("pairs {}", table[0][n-1]));
760 Ok(())
761 }

763 fn assembly(&mut self, name: &str, minimum: usize, sequences: &[String])
764 -> Result<(), String>
765 {
766 let mut unique = Vec::new();
767 let mut seen = HashSet::new();
768 for sequence in sequences {
769 if seen.insert(sequence.clone()) {
770 unique.push(sequence.clone());
771 }
772 }
773 let mut fragments: Vec<String> = unique.iter()
774 .filter(|sequence| !unique.iter().any(|other|
775 *sequence != other && other.contains(sequence.as_str())))
776 .cloned().collect();
777 while fragments.len() >= 2 {
778 let (mut best, mut chosen) = (0_usize, None);
779 for i in 0..fragments.len() {
780 for j in 0..fragments.len() {
781 if i == j {
782 continue;
783 }
784 let a = &fragments[i];
785 let b = &fragments[j];
786 let mut overlap = 0;
787 for width in (1..a.len().min(b.len())).rev() {
788 if a.ends_with(&b[..width]) {
789 overlap = width;
790 break;
791 }
792 }
793 if chosen.is_none() || overlap > best {
794 best = overlap;
795 chosen = Some((i, j));
796 }
797 }
798 }
799 if best < minimum {
800 break;
801 }
802 let (first, second) = chosen.expect("at least two fragments");
803 let merged = fragments[first].clone() + &fragments[second][best..];
804 fragments = fragments.iter().enumerate().filter_map(|(i, sequence)| {
805 if i == first {
806 Some(merged.clone())
807 } else if i != second && !merged.contains(sequence) {
808 Some(sequence.clone())
809 } else {
810 None
811 }
812 }).collect();
813 }
814 let longest = fragments.iter()
815 .max_by_key(|sequence| (sequence.len(), std::cmp::Reverse(
816 fragments.iter().position(|item| item == *sequence).expect("present")
817 )))
818 .expect("at least one fragment").clone();
819 let registration = self.store(name, longest)?;
820 self.emit(format!("contigs {}", fragments.len()));
821 for fragment in fragments {
822 self.emit(format!("{} {fragment}", fragment.len()));
823 }
824 self.emit(registration);
825 Ok(())
826 }

828 fn command(&mut self, command: &str, args: &[&str]) -> Result<(), String> {
829 match command {
830 "SEQ" => {
831 let result = self.store(args[0], args[1].to_owned())?;
832 self.emit(result);
833 return Ok(());
834 }
835 "UNPACK" => {
836 let length = args[1].parse::<usize>().expect("validated number");
837 let mut codes = Vec::new();
838 for bytes in args[2].as_bytes().chunks_exact(2) {
839 let number = u8::from_str_radix(
840 std::str::from_utf8(bytes).expect("ASCII"), 16
841 ).expect("validated hex");
842 codes.extend([
843 number / 64, number / 16 % 4, number / 4 % 4, number % 4
844 ]);
845 }
846 if codes[length..].iter().any(|&code| code != 0) {
847 return Err("invalid padding".to_owned());
848 }
849 let sequence: String = codes[..length].iter()
850 .map(|&code| BASES[code as usize] as char).collect();
851 let result = self.store(args[0], sequence)?;
852 self.emit(result);
853 return Ok(());
854 }
855 "JOIN" => {
856 let sequence: String = args[1..].iter()
857 .map(|name| self.sequences[*name].as_str()).collect();
858 let result = self.store(args[0], sequence)?;
859 self.emit(result);
860 return Ok(());
861 }
862 "ASSEMBLE" => {
863 let minimum = args[1].parse::<usize>().expect("validated overlap");
864 let fragments: Vec<String> = args[2..].iter()
865 .map(|name| self.sequences[*name].clone()).collect();
866 return self.assembly(args[0], minimum, &fragments);
867 }
868 "LIST" => {
869 if self.sequences.is_empty() {
870 self.emit("no sequences");
871 }
872 let entries: Vec<(String, usize)> = self.sequences.iter()
873 .map(|(name, sequence)| (name.clone(), sequence.len())).collect();
874 for (name, length) in entries {
875 self.emit(format!("{name} {length} nt"));
876 }
877 return Ok(());
878 }
879 _ => {}
880 }
881
882 let name = args[0].to_owned();
883 let sequence = self.sequences[&name].clone();
884 let n = sequence.len();
885 match command {
886 "SLICE" => {
887 let start = args[1].parse::<usize>().expect("validated position");
888 let end = args[2].parse::<usize>().expect("validated position");
889 let segment = if start <= end {
890 sequence[start-1..end].to_owned()
891 } else {
892 revcomp(&sequence[end-1..start])
893 };
894 let result = self.store(args[3], segment)?;
895 self.emit(result);
896 }
897 "MUTATE" => {
898 let position = args[1].parse::<usize>().expect("validated position");
899 let before = sequence.as_bytes()[position-1];
900 let after = args[2].as_bytes()[0];
901 if before == after {
902 return Err("no change".to_owned());
903 }
904 let mut replacement = sequence.into_bytes();
905 replacement[position-1] = after;
906 self.sequences.insert(
907 name.clone(), String::from_utf8(replacement).expect("ASCII")
908 );
909 self.emit(format!(
910 "{name} {}{position}{} {}", before as char, after as char,
911 if transition(before, after) { "transition" } else { "transversion" }
912 ));
913 }
914 "DEL" => {
915 self.sequences.remove(&name);
916 self.emit(format!("deleted {name}"));
917 }
918 "REVCOMP" => self.emit(revcomp(&sequence)),
919 "COMPLEMENT" => self.emit(complementary(&sequence)),
920 "REVERSE" => self.emit(sequence.chars().rev().collect::<String>()),
921 "TRANSCRIBE" => self.emit(sequence.replace('T', "U")),
922 "TRANSLATE" | "CODONS" | "PROTEIN" => {
923 let frame = args[1].parse::<i32>().expect("validated frame");
924 self.translation(command, &name, frame)?;
925 }
926 "ORFS" => {
927 let minimum = args[1].parse::<usize>().expect("validated minimum");
928 self.orfs(&sequence, minimum);
929 }
930 "GC" => {
931 let fields = BASES.iter().map(|&base| {
932 format!("{}={}", base as char,
933 sequence.bytes().filter(|&c| c == base).count())
934 }).collect::<Vec<_>>();
935 self.emit(format!("{} GC={}%", fields.join(" "), percentage(gc(&sequence), n)));
936 }
937 "GCWINDOW" => {
938 let width = args[1].parse::<usize>().expect("validated width");
939 let step = args[2].parse::<usize>().expect("validated step");
940 if width > n {
941 return Err("window too large".to_owned());
942 }
943 for i in (0..=n-width).step_by(step) {
944 self.emit(format!("{}-{} {}%",
945 i+1, i+width, percentage(gc(&sequence[i..i+width]), width)));
946 }
947 }
948 "SKEW" => {
949 let mut values = vec![0_i32];
950 for base in sequence.bytes() {
951 values.push(values.last().expect("initial zero") +
952 i32::from(base == b'G') - i32::from(base == b'C'));
953 }
954 for (label, extreme) in [
955 ("min", *values.iter().min().expect("nonempty")),
956 ("max", *values.iter().max().expect("nonempty")),
957 ] {
958 let positions = values.iter().enumerate()
959 .filter(|&(_, value)| *value == extreme)
960 .map(|(position, _)| position).collect::<Vec<_>>();
961 self.emit(format!("{label} {extreme} at {}", list_text(&positions)));
962 }
963 }
964 "KMERS" => {
965 let width = args[1].parse::<usize>().expect("validated k");
966 let limit = args[2].parse::<usize>().expect("validated limit");
967 if width > n {
968 return Err("k too large".to_owned());
969 }
970 let mut counts = BTreeMap::new();
971 for i in 0..=n-width {
972 *counts.entry(sequence[i..i+width].to_owned()).or_insert(0) += 1;
973 }
974 self.emit(format!("{} windows {} distinct", n-width+1, counts.len()));
975 for word in frequency_order(&counts).into_iter().take(limit) {
976 self.emit(format!("{word} {}", counts[&word]));
977 }
978 }
979 "FIND" => {
980 self.emit(format!("forward {}", list_text(&occurrences(&sequence, args[1]))));
981 self.emit(format!("reverse {}", list_text(&occurrences(&sequence, &revcomp(args[1])))));
982 }
983 "RESTRICT" => self.restriction(&sequence, &args[1..]),
984 "PALINDROMES" => {
985 let minimum = args[1].parse::<usize>().expect("validated min");
986 let maximum = args[2].parse::<usize>().expect("validated max");
987 if minimum > maximum {
988 return Err("invalid range".to_owned());
989 }
990 let mut found = false;
991 for i in 0..n {
992 for width in minimum..=maximum.min(n-i) {
993 let part = &sequence[i..i+width];
994 if part == revcomp(part) {
995 self.emit(format!("{} {width} {part}", i+1));
996 found = true;
997 }
998 }
999 }
1000 if !found {
1001 self.emit("none");
1002 }
1003 }
1004 "REPEATS" => {
1005 let min = args[1].parse::<usize>().expect("validated min");
1006 let max = args[2].parse::<usize>().expect("validated max");
1007 let count = args[3].parse::<usize>().expect("validated count");
1008 self.repeats(&sequence, min, max, count)?;
1009 }
1010 "HAMMING" => {
1011 let other = &self.sequences[args[1]];
1012 if other.len() != n {
1013 return Err("length mismatch".to_owned());
1014 }
1015 let mut distance = 0;
1016 let mut transitions = 0;
1017 for (&a, &b) in sequence.as_bytes().iter().zip(other.bytes().collect::<Vec<_>>().iter()) {
1018 if a != b {
1019 distance += 1;
1020 transitions += i32::from(transition(a, b));
1021 }
1022 }
1023 self.emit(format!(
1024 "distance {distance} transitions {transitions} transversions {}",
1025 distance-transitions
1026 ));
1027 }
1028 "ALIGN" => {
1029 let match_score = args[3].parse::<i32>().expect("validated match");
1030 let mismatch = args[4].parse::<i32>().expect("validated mismatch");
1031 let gap = args[5].parse::<i32>().expect("validated gap");
1032 let other = self.sequences[args[1]].clone();
1033 self.alignment(&sequence, &other, args[2], match_score, mismatch, gap);
1034 }
1035 "CONSENSUS" => self.consensus(args)?,
1036 "TM" => self.emit(format!(
1037 "Tm {} {}", fixed(temperature(&sequence), 1),
1038 if n < 14 { "wallace" } else { "gc" }
1039 )),
1040 "MASS" => {
1041 let single = strand_mass(&sequence);
1042 self.emit(format!(
1043 "ss {} ds {}", fixed(single, 3),
1044 fixed(single + strand_mass(&revcomp(&sequence)), 3)
1045 ));
1046 }
1047 "PRIMER" => self.primer(&sequence),
1048 "FOLD" => {
1049 let loop_size = args[1].parse::<usize>().expect("validated loop");
1050 self.folding(&sequence, loop_size)?;
1051 }
1052 "PACK" => {
1053 let mut codes: Vec<usize> = sequence.bytes()
1054 .map(|base| BASES.iter().position(|&c| c == base).expect("DNA")).collect();
1055 codes.resize(codes.len().div_ceil(4) * 4, 0);
1056 let hex = codes.chunks(4).map(|group| format!("{:02X}",
1057 64*group[0] + 16*group[1] + 4*group[2] + group[3]))
1058 .collect::<String>();
1059 self.emit(format!("{n} {hex}"));
1060 }
1061 _ => unreachable!("validated command"),
1062 }
1063 Ok(())
1064 }
1065
1066 fn execute(&mut self, line: &str) {
1067 let words: Vec<&str> = line.split(' ').collect();
1068 let command = words[0];
1069 let args = &words[1..];
1070 let result = self.validate(command, args)
1071 .and_then(|_| self.command(command, args));
1072 if let Err(error) = result {
1073 self.emit(format!("ERROR {error}"));
1074 }
1075 }
1076}
