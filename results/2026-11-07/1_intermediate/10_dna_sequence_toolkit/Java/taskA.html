121 static int overlap(String a, String b) {
122 String joined=b+"#"+a;
123 int[] prefix=new int[joined.length()];
124 for (int i=1;i<joined.length();i++) {
125 int j=prefix[i-1];
126 while (j>0 && joined.charAt(i)!=joined.charAt(j)) j=prefix[j-1];
127 if (joined.charAt(i)==joined.charAt(j)) j++;
128 prefix[i]=j;
129 }
130 int result=prefix[joined.length()-1], limit=Math.min(a.length(),b.length());
131 while (result>=limit) result=prefix[result-1];
132 return result;

173 void validate(String command, String[] a, int[] nums, String[] sequences) {
174 error(!schemas.containsKey(command),"unknown command "+command);
175 int[] count=arities.get(command);
176 error(a.length<count[0] || a.length>count[1],"wrong argument count");
177 String[] schema=schemas.get(command);
178 Set<String> enzymes=new HashSet<>();
179 for (int i=0;i<a.length;i++) {
180 String kind=schema[Math.min(i,schema.length-1)], value=a[i];
181 switch (kind) {
182 case "NAME":
183 error(!registry.containsKey(value),"unknown sequence "+value);
184 sequences[i]=registry.get(value);
185 break;
186 case "NEW":
187 error(!value.matches("[a-z][a-z0-9]{0,9}"),"invalid name "+value);
188 break;
189 case "SEQ":
190 for (int j=0;j<value.length();j++) error(BASES.indexOf(value.charAt(j))<0,"invalid base "+value.charAt(j)+" at "+(j+1));
191 break;
192 case "BASE":
193 error(value.length()!=1 || BASES.indexOf(value.charAt(0))<0,"invalid base "+value);
194 break;
195 case "MOTIF":
196 error(!value.matches("[ACGTRYSWKMBDHVN]{1,30}"),"invalid motif "+value);
197 break;
198 case "ENZYME":
199 error(!Arrays.asList(ENZYMES).contains(value),"unknown enzyme "+value);
200 error(!enzymes.add(value),"duplicate enzyme "+value);
201 break;
202 case "MODE":
203 error(!value.equals("GLOBAL") && !value.equals("LOCAL"),"invalid mode "+value);
204 break;
205 case "HEX":
206 error(value.length()!=2*((nums[1]+3)/4) || !value.matches("[0-9A-F]+"),"invalid hex");
207 break;
208 case "FRAME":

210 error(nums[i]==0,"invalid number "+value);
211 break;
212 case "POS":

216 String[] bounds=kind.split(":");

218 }

221 void align(String a, String b, boolean local, int match, int mismatch, int gap) {
222 int m=a.length(),n=b.length();
223 int[][] h=new int[m+1][n+1];
224 if (!local) {
225 for (int i=0;i<=m;i++) h[i][0]=i*gap;
226 for (int j=0;j<=n;j++) h[0][j]=j*gap;
227 }
228 int best=0,bi=0,bj=0;
229 for (int i=1;i<=m;i++) {
230 for (int j=1;j<=n;j++) {
231 int value=Math.max(h[i-1][j-1]+(a.charAt(i-1)==b.charAt(j-1) ? match : mismatch),Math.max(h[i-1][j]+gap,h[i][j-1]+gap));
232 h[i][j]=local ? Math.max(0,value) : value;
233 if (local && h[i][j]>best) {
234 best=h[i][j];
235 bi=i;
236 bj=j;
237 }
238 }
239 }
240 if (!local) {
241 bi=m;
242 bj=n;
243 best=h[m][n];
244 }
245 if (local && best==0) {
246 output.add("score 0");
247 return;
248 }
249 int i=bi,j=bj;
250 StringBuilder x=new StringBuilder(),y=new StringBuilder();
251 while (local ? h[i][j]!=0 : i!=0 || j!=0) {
252 if (i>0 && j>0 && h[i][j]==h[i-1][j-1]+(a.charAt(i-1)==b.charAt(j-1) ? match : mismatch)) {
253 x.append(a.charAt(--i));
254 y.append(b.charAt(--j));
255 } else if (i>0 && h[i][j]==h[i-1][j]+gap) {
256 x.append(a.charAt(--i));
257 y.append('-');
258 } else {
259 x.append('-');
260 y.append(b.charAt(--j));
261 }
262 }
263 String left=x.reverse().toString(),right=y.reverse().toString();
264 StringBuilder middle=new StringBuilder();
265 int identity=0,gaps=0;
266 for (int k=0;k<left.length();k++) {
267 char c=left.charAt(k)=='-' || right.charAt(k)=='-' ? '-' : left.charAt(k)==right.charAt(k) ? '|' : '*';
268 middle.append(c);
269 if (c=='|') identity++;
270 if (c=='-') gaps++;
271 }
272 output.add("score "+best+" a "+(i+1)+"-"+bi+" b "+(j+1)+"-"+bj+" identity "+identity+"/"+left.length()+" gaps "+gaps);
273 output.add(left);
274 output.add(middle.toString());
275 output.add(right);
276 }

279 void fold(String s, int loop) {
280 int n=s.length();
281 error(n>100,"too long");
282 int[][] f=new int[n][n];
283 for (int length=1;length<=n;length++) {
284 for (int i=0;i+length<=n;i++) {
285 int j=i+length-1;
286 f[i][j]=value(f,i,j-1);
287 for (int k=i;k<=j-loop-1;k++) if (pair(s.charAt(k),s.charAt(j))) f[i][j]=Math.max(f[i][j],value(f,i,k-1)+1+value(f,k+1,j-1));
288 }
289 }
290 char[] structure=new char[n];
291 Arrays.fill(structure,'.');
292 Deque<int[]> intervals=new ArrayDeque<>();
293 intervals.push(new int[]{0,n-1});
294 while (!intervals.isEmpty()) {
295 int[] interval=intervals.pop();
296 int i=interval[0],j=interval[1];
297 if (value(f,i,j)==0) continue;
298 if (f[i][j]==value(f,i,j-1)) intervals.push(new int[]{i,j-1});
299 else {
300 for (int k=i;k<=j-loop-1;k++) {
301 if (pair(s.charAt(k),s.charAt(j)) && value(f,i,k-1)+1+value(f,k+1,j-1)==f[i][j]) {
302 structure[k]='(';
303 structure[j]=')';
304 intervals.push(new int[]{i,k-1});
305 intervals.push(new int[]{k+1,j-1});
306 break;
307 }
308 }
309 }
310 }
311 output.add(new String(structure));
312 output.add("pairs "+f[0][n-1]);
313 }

332 void assemble(String name, int minimum, String[] sequences) {
333 List<String> unique=new ArrayList<>(new LinkedHashSet<>(Arrays.asList(sequences).subList(2,sequences.length)));
334 List<String> fragments=new ArrayList<>();
335 for (String s:unique) if (unique.stream().noneMatch(t -> !t.equals(s) && t.contains(s))) fragments.add(s);
336 while (fragments.size()>=2) {
337 int best=-1,bx=0,by=0;
338 for (int x=0;x<fragments.size();x++) {
339 for (int y=0;y<fragments.size();y++) {
340 if (x==y) continue;
341 String a=fragments.get(x),b=fragments.get(y);
342 int overlap=overlap(a,b);
343 if (overlap>best) {
344 best=overlap;
345 bx=x;
346 by=y;
347 }
348 }
349 }
350 if (best<minimum) break;
351 String merged=fragments.get(bx)+fragments.get(by).substring(best);
352 fragments.set(bx,merged);
353 fragments.remove(by);
354 int keep=bx-(by<bx ? 1 : 0);
355 for (int i=fragments.size()-1;i>=0;i--) {
356 if (i!=keep && merged.contains(fragments.get(i))) {
357 fragments.remove(i);
358 if (i<keep) keep--;
359 }
360 }
361 }
362 String longest=Collections.max(fragments,Comparator.comparingInt(String::length));
363 error(longest.length()>500,"too long");
364 output.add("contigs "+fragments.size());
365 for (String s:fragments) output.add(s.length()+" "+s);
366 store(name,longest);
367 }
368 void execute(String line) {
369 String[] t=line.split(" "),a=Arrays.copyOfRange(t,1,t.length);
370 int[] nums=new int[a.length];
371 String[] sequences=new String[a.length];
372 String command=t[0];
373 validate(command,a,nums,sequences);
374 String s=a.length==0 ? "" : sequences[0];
375 int n=s==null ? 0 : s.length();
376 switch (command) {
377 case "SEQ":
378 store(a[0],a[1]);
379 break;
380 case "SLICE": {
381 int left=Math.min(nums[1],nums[2]),right=Math.max(nums[1],nums[2]);
382 String part=s.substring(left-1,right);
383 store(a[3],nums[1]<=nums[2] ? part : revcomp(part));
384 break;
385 }
386 case "JOIN":
387 store(a[0],String.join("",Arrays.asList(sequences).subList(1,sequences.length)));
388 break;
389 case "MUTATE": {
390 char old=s.charAt(nums[1]-1),next=a[2].charAt(0);
391 error(old==next,"no change");
392 registry.put(a[0],s.substring(0,nums[1]-1)+next+s.substring(nums[1]));
393 output.add(a[0]+" "+old+nums[1]+next+" "+(transition(old,next) ? "transition" : "transversion"));
394 break;
395 }
396 case "DEL":
397 registry.remove(a[0]);
398 output.add("deleted "+a[0]);
399 break;
400 case "LIST":
401 if (registry.isEmpty()) output.add("no sequences");
402 else for (String name:registry.keySet()) output.add(name+" "+registry.get(name).length()+" nt");
403 break;
404 case "REVCOMP":
405 output.add(revcomp(s));
406 break;
407 case "COMPLEMENT":
408 output.add(complement(s));
409 break;
410 case "REVERSE":
411 output.add(reverse(s));
412 break;
413 case "TRANSCRIBE":
414 output.add(s.replace('T','U'));
415 break;
416 case "TRANSLATE": {
417 String protein=translation(codons(s,nums[1]));
418 output.add(protein.isEmpty() ? "none" : protein);
419 break;
420 }
421 case "CODONS": {
422 Map<String,Integer> counts=counts(codons(s,nums[1]));
423 if (counts.isEmpty()) output.add("none");
424 for (String c:frequency(counts)) output.add(c+" "+GENETIC.get(c)+" "+counts.get(c));
425 break;
426 }
427 case "PROTEIN": {
428 String protein=translation(codons(s,nums[1])).split("\\*",-1)[0];
429 error(protein.isEmpty(),"empty protein");
430 int mass=18015+protein.chars().map(c -> RESIDUES[AMINO.indexOf(c)]).sum();
431 long charge=protein.chars().filter(c -> "KR".indexOf(c)>=0).count()
432 -protein.chars().filter(c -> "DE".indexOf(c)>=0).count();
433 Map<String,Integer> counts=counts(protein.chars().mapToObj(Character::toString).toList());
434 output.add(protein);
435 output.add("length "+protein.length()+" mass "+milli(mass)+" charge "+(charge>0 ? "+" : "")+charge);
436 output.add(String.join(" ",frequency(counts).stream().map(c -> c+"="+counts.get(c)).toList()));
437 break;
438 }
439 case "ORFS": {
440 List<Orf> found=new ArrayList<>();
441 for (int frame:new int[]{1,2,3,-1,-2,-3}) {
442 List<String> codons=codons(s,frame);
443 int start=-1;
444 for (int i=0;i<codons.size();i++) {
445 if (start<0 && codons.get(i).equals("ATG")) start=i;
446 else if (start>=0 && GENETIC.get(codons.get(i))=='*') {
447 if (i-start>=nums[1]) {
448 int left=Math.abs(frame)-1+3*start,right=Math.abs(frame)-1+3*(i+1)-1;
449 found.add(new Orf(frame,frame>0 ? left+1 : n-right,frame>0 ? right+1 : n-left,translation(codons.subList(start,i))));
450 }
451 start=-1;
452 }
453 }
454 }
455 found.sort(Comparator.comparingInt(Orf::left));
456 if (found.isEmpty()) output.add("none");
457 for (Orf o:found) output.add((o.frame()>0 ? "+" : "")+o.frame()+" "+o.left()+" "+o.right()+" "+o.protein().length()+" "+o.protein());
458 break;
459 }
460 case "GC": {
461 String lineOut="";
462 for (char c:BASES.toCharArray()) lineOut+=(lineOut.isEmpty() ? "" : " ")+c+"="+s.chars().filter(x -> x==c).count();
463 output.add(lineOut+" GC="+pct(gc(s),n)+"%");
464 break;
465 }
466 case "GCWINDOW":
467 error(nums[1]>n,"window too large");
468 for (int i=0;i+nums[1]<=n;i+=nums[2]) output.add((i+1)+"-"+(i+nums[1])+" "+pct(gc(s.substring(i,i+nums[1])),nums[1])+"%");
469 break;
470 case "SKEW": {
471 int[] skew=new int[n+1];
472 for (int i=0;i<n;i++) skew[i+1]=skew[i]+(s.charAt(i)=='G' ? 1 : s.charAt(i)=='C' ? -1 : 0);
473 for (boolean minimum:new boolean[]{true,false}) {
474 int v=minimum ? Arrays.stream(skew).min().orElse(0) : Arrays.stream(skew).max().orElse(0);
475 List<Integer> at=IntStream.rangeClosed(0,n).filter(i -> skew[i]==v).boxed().toList();
476 output.add((minimum ? "min " : "max ")+v+" at "+list(at));
477 }
478 break;
479 }
480 case "KMERS": {
481 error(nums[1]>n,"k too large");
482 List<String> words=IntStream.rangeClosed(0,n-nums[1])
483 .mapToObj(i -> s.substring(i,i+nums[1])).toList();
484 Map<String,Integer> counts=counts(words);
485 output.add(words.size()+" windows "+counts.size()+" distinct");
486 List<String> order=frequency(counts);
487 for (int i=0;i<Math.min(nums[2],order.size());i++) output.add(order.get(i)+" "+counts.get(order.get(i)));
488 break;
489 }
490 case "FIND":
491 output.add("forward "+list(occurrences(s,a[1])));
492 output.add("reverse "+list(occurrences(s,revcomp(a[1]))));
493 break;
494 case "RESTRICT": {
495 TreeSet<Integer> cuts=new TreeSet<>();
496 for (int i=1;i<a.length;i++) {
497 int e=Arrays.asList(ENZYMES).indexOf(a[i]);
498 List<Integer> found=new ArrayList<>();
499 for (int p:occurrences(s,SITES[e])) {
500 int c=p+CUTS[e]-1;
501 if (c>=1 && c<n) {
502 found.add(c);
503 cuts.add(c);
504 }
505 }
506 output.add(a[i]+" "+list(found));
507 }
508 List<String> fragments=new ArrayList<>();
509 List<Integer> sizes=new ArrayList<>(),ends=new ArrayList<>(cuts);
510 ends.add(n);
511 int previous=0;
512 for (int end:ends) {
513 fragments.add((previous+1)+"-"+end);
514 sizes.add(end-previous);
515 previous=end;
516 }
517 sizes.sort(Comparator.reverseOrder());
518 output.add("fragments "+list(fragments));
519 output.add("sizes "+list(sizes));
520 break;
521 }
522 case "PALINDROMES": {
523 error(nums[1]>nums[2],"invalid range");
524 int count=0;
525 for (int i=0;i<n;i++) {
526 for (int length=nums[1];length<=nums[2] && i+length<=n;length++) {
527 String part=s.substring(i,i+length);
528 if (part.equals(revcomp(part))) {
529 output.add((i+1)+" "+length+" "+part);
530 count++;
531 }
532 }
533 }
534 if (count==0) output.add("none");
535 break;
536 }
537 case "REPEATS": {
538 error(nums[1]>nums[2],"invalid range");
539 List<Repeat> found=new ArrayList<>();
540 for (int u=nums[1];u<=nums[2];u++) {
541 int i=0;
542 while (i+u<=n) {
543 String unit=s.substring(i,i+u);
544 int count=repeatCount(s,i,u);
545 if (primitive(unit) && count>=nums[3]) {
546 found.add(new Repeat(i+1,i+count*u,u,unit,count));
547 i+=count*u;
548 } else i++;
549 }
550 }
551 found.sort(Comparator.comparingInt(Repeat::start).thenComparingInt(Repeat::unitLength));
552 if (found.isEmpty()) output.add("none");
553 for (Repeat r:found) output.add(r.start()+"-"+r.end()+" "+r.unit()+"x"+r.count());
554 break;
555 }
556 case "HAMMING": {
557 String b=sequences[1];
558 error(n!=b.length(),"length mismatch");
559 int distance=0,transitions=0;
560 for (int i=0;i<n;i++) {
561 if (s.charAt(i)!=b.charAt(i)) {
562 distance++;
563 if (transition(s.charAt(i),b.charAt(i))) transitions++;
564 }
565 }
566 output.add("distance "+distance+" transitions "+transitions+" transversions "+(distance-transitions));
567 break;
568 }
569 case "ALIGN":
570 align(s,sequences[1],a[2].equals("LOCAL"),nums[3],nums[4],nums[5]);
571 break;
572 case "CONSENSUS": {
573 for (String sequence:sequences) error(sequence.length()!=n,"length mismatch");
574 int[][] profile=new int[4][n];
575 StringBuilder consensus=new StringBuilder(),iupac=new StringBuilder();
576 for (int i=0;i<n;i++) {
577 for (String sequence:sequences) profile[BASES.indexOf(sequence.charAt(i))][i]++;
578 int maximum=0,mask=0;
579 for (int b=0;b<4;b++) maximum=Math.max(maximum,profile[b][i]);
580 boolean chosen=false;
581 for (int b=0;b<4;b++) {
582 if (profile[b][i]==maximum) {
583 mask|=1<<b;
584 if (!chosen) {
585 consensus.append(BASES.charAt(b));
586 chosen=true;
587 }
588 }
589 }
590 for (int j=0;j<MASKS.length;j++) if (MASKS[j]==mask) iupac.append(SYMBOLS.charAt(j));
591 }
592 for (int b=0;b<4;b++) output.add(BASES.charAt(b)+" "+list(Arrays.stream(profile[b]).boxed().toList()));
593 output.add("consensus "+consensus);
594 output.add("iupac "+iupac);
595 break;
596 }
597 case "TM":
598 output.add("Tm "+fixed(tm(s),10,1)+" "+(n<14 ? "wallace" : "gc"));
599 break;
600 case "MASS":
601 output.add("ss "+milli(mass(s))+" ds "+milli(mass(s)+mass(revcomp(s))));
602 break;
603 case "PRIMER":
604 primer(s);
605 break;
606 case "FOLD":
607 fold(s,nums[1]);
608 break;
609 case "ASSEMBLE":
610 assemble(a[0],nums[1],sequences);
611 break;
612 case "PACK": {
613 StringBuilder hex=new StringBuilder();
614 for (int i=0;i<n;i+=4) {
615 int value=0;
616 for (int j=0;j<4;j++) value=4*value+(i+j<n ? BASES.indexOf(s.charAt(i+j)) : 0);
617 hex.append(HEX_FORMAT.toHexDigits((byte)value));
618 }
619 output.add(n+" "+hex);
620 break;
621 }
622 case "UNPACK": {
623 StringBuilder decoded=new StringBuilder();
624 for (byte packed:HEX_FORMAT.parseHex(a[2])) {
625 int value=Byte.toUnsignedInt(packed);
626 for (int shift:new int[]{6,4,2,0}) decoded.append(BASES.charAt(value>>shift&3));
627 }
628 for (int i=nums[1];i<decoded.length();i++) error(decoded.charAt(i)!='A',"invalid padding");
629 store(a[0],decoded.substring(0,nums[1]));
630 break;
631 }
632 }
633 }
634 List<String> process(String line) {
635 output=new ArrayList<>();
636 try { execute(line); }
637 catch (Fault fault) {
638 output.clear();
639 output.add("ERROR "+fault.getMessage());
640 }
641 return output;
642 }
643 }
644 public static void main(String[] args) throws IOException {
645 BufferedReader input=new BufferedReader(new InputStreamReader(System.in));
646 PrintWriter output=new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));
647 Toolkit toolkit=new Toolkit();
648 int n=Integer.parseInt(input.readLine());
649 for (int i=0;i<n;i++) {
650 for (String line:toolkit.process(input.readLine())) output.println(line);
651 }
652 output.flush();
653 }
