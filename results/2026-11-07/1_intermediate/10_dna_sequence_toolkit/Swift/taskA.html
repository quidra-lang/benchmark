63func complement(_ s: String) -> String { String(s.map { complements[Array(symbols).firstIndex(of:$0)!] }) }
64func revcomp(_ s: String) -> String { String(complement(s).reversed()) }
65func complementBase(_ c: Character) -> Character { bases[3-Array(bases).firstIndex(of:c)!] }
66func gc(_ s: String) -> Int { s.utf8.filter { $0 == 71 || $0 == 67 }.count }

70func pct(_ a: Int, _ b: Int) -> String { fixed(10000*a/b,100,2) }

73func codons(_ s: String, _ frame: Int) -> [String] {
74 let strand = frame > 0 ? s : revcomp(s)
75 var result: [String] = []
76 var i = abs(frame)-1
77 while i+3 <= s.count {
78 result.append(strand.part(i,i+3))
79 i += 3
80 }
81 return result

87func frequency(_ counts: [String:Int]) -> [String] { counts.keys.sorted { counts[$0] == counts[$1] ? $0 < $1 : counts[$0]! > counts[$1]! } }
88func occurrences(_ s: String, _ motif: String) -> [Int] {
89 let sequence = Array(s)
90 let pattern = Array(motif).map { masks[Array(symbols).firstIndex(of:$0)!] }
91 if pattern.count > sequence.count { return [] }
92 let codes = sequence.map { 1 << Array(bases).firstIndex(of:$0)! }
93 return (0...(sequence.count-pattern.count)).filter { i in
94 pattern.indices.allSatisfy { pattern[$0]&codes[i+$0] != 0 }
95 }.map { $0+1 }
96}
97func transition(_ a: Character, _ b: Character) -> Bool {
98 a != b && ("AG".contains(a) && "AG".contains(b) || "CT".contains(a) && "CT".contains(b))
99}
100func repeatCount(_ s: String, _ i: Int, _ u: Int) -> Int {
101 let unit = s.part(i,i+u)
102 var count = 1
103 while i+(count+1)*u <= s.count && s.part(i+count*u,i+(count+1)*u) == unit { count += 1 }
104 return count
105}
106func primitive(_ s: String) -> Bool {
107 (1..<s.count).allSatisfy { s.count%$0 != 0 || String(repeating:s.part(0,$0),count:s.count/$0) != s }
108}
109func selfcomp(_ s: String) -> Int {
110 let a = Array(s)
111 var maximum = 0
112 if a.count > 1 {
113 for center in 1..<a.count {
114 var left = center-1
115 var right = center
116 while left >= 0 && right < a.count && complementBase(a[left]) == a[right] {
117 maximum = max(maximum,right-left+1)
118 left -= 1
119 right += 1
120 }
121 }
122 }
123 return maximum
124}
125func tm(_ s: String) -> Int {
126 let n = s.count
127 let g = gc(s)
128 return n < 14 ? 10*(2*(n-g)+4*g) : (649*n+410*g-6724)/n
129}
130func mass(_ s: String) -> Int { s.reduce(0) { $0+baseMass[Array(bases).firstIndex(of:$1)!] }-(s.count-1)*18015 }
131func overlap(_ a: String, _ b: String) -> Int {
132 let joined = Array((b+"#"+a).utf8)
133 var prefix = Array(repeating:0,count:joined.count)
134 for i in 1..<joined.count {
135 var j = prefix[i-1]
136 while j > 0 && joined[i] != joined[j] { j = prefix[j-1] }
137 if joined[i] == joined[j] { j += 1 }
138 prefix[i] = j
139 }
140 var result = prefix.last!
141 let limit = min(a.count,b.count)
142 while result >= limit { result = prefix[result-1] }
143 return result
144}

189 func store(_ name: String, _ sequence: String) throws {
190 try error(sequence.count > 500,"too long")
191 let replacing = registry[name] != nil
192 registry[name] = sequence
193 output.append((replacing ? "replaced " : "stored ")+"\(name) \(sequence.count) nt")
194 }
195 func validate(_ command: String, _ a: [String], _ nums: inout [Int], _ sequences: inout [String]) throws {
196 guard let schema = schemas[command] else { throw Fault.reason("unknown command "+command) }
197 try error(!arities[command]!.contains(a.count),"wrong argument count")
198 var seen = Set<String>()
199 for (i,value) in a.enumerated() {
200 let kind = schema[min(i,schema.count-1)]
201 switch kind {
202 case "NAME":
203 guard let sequence = registry[value] else { throw Fault.reason("unknown sequence "+value) }
204 sequences[i] = sequence
205 case "NEW": try error(!matches(value,"[a-z][a-z0-9]{0,9}"),"invalid name "+value)
206 case "SEQ":
207 for (j,c) in value.enumerated() { try error(!bases.contains(c),"invalid base \(c) at \(j+1)") }
208 case "BASE": try error(value.count != 1 || !bases.contains(value.first!),"invalid base "+value)
209 case "MOTIF": try error(!matches(value,"[ACGTRYSWKMBDHVN]{1,30}"),"invalid motif "+value)
210 case "ENZYME":
211 try error(!enzymes.contains(value),"unknown enzyme "+value)
212 try error(!seen.insert(value).inserted,"duplicate enzyme "+value)
213 case "MODE": try error(value != "GLOBAL" && value != "LOCAL","invalid mode "+value)
214 case "HEX": try error(value.count != 2*((nums[1]+3)/4) || !matches(value,"[0-9A-F]+"),"invalid hex")
215 case "FRAME":
216 nums[i] = try number(value,-3,3)
217 try error(nums[i] == 0,"invalid number "+value)
218 case "POS": nums[i] = try number(value,1,sequences[0].count)
219 default:
220 let bounds = kind.split(separator:":")
221 nums[i] = try number(value,Int(bounds[0])!,Int(bounds[1])!)
222 }
223 }
224 }
225 func align(_ a: String, _ b: String, _ local: Bool, _ match: Int, _ mismatch: Int, _ gap: Int) {
226 let x = Array(a)
227 let y = Array(b)
228 let m = x.count
229 let n = y.count
230 var h = Array(repeating:Array(repeating:0,count:n+1),count:m+1)
231 if !local {
232 for i in 0...m { h[i][0] = i*gap }
233 for j in 0...n { h[0][j] = j*gap }
234 }
235 var best = 0
236 var bi = 0
237 var bj = 0
238 for i in 1...m {
239 for j in 1...n {
240 let value = max(h[i-1][j-1]+(x[i-1] == y[j-1] ? match : mismatch),max(h[i-1][j]+gap,h[i][j-1]+gap))
241 h[i][j] = local ? max(0,value) : value
242 if local && h[i][j] > best {
243 best = h[i][j]
244 bi = i
245 bj = j
246 }
247 }
248 }
249 if !local {
250 bi = m
251 bj = n
252 best = h[m][n]
253 }
254 if local && best == 0 {
255 output.append("score 0")
256 return
257 }
258 var i = bi
259 var j = bj
260 var left: [Character] = []
261 var right: [Character] = []
262 while (local ? h[i][j] != 0 : i != 0 || j != 0) {
263 if i > 0 && j > 0 && h[i][j] == h[i-1][j-1]+(x[i-1] == y[j-1] ? match : mismatch) {
264 i -= 1
265 j -= 1
266 left.append(x[i])
267 right.append(y[j])
268 } else if i > 0 && h[i][j] == h[i-1][j]+gap {
269 i -= 1
270 left.append(x[i])
271 right.append("-")
272 } else {
273 j -= 1
274 left.append("-")
275 right.append(y[j])
276 }
277 }
278 left.reverse()
279 right.reverse()
280 let middle: [Character] = left.indices.map { k in left[k] == "-" || right[k] == "-" ? "-" : left[k] == right[k] ? "|" : "*" }
281 let identity = middle.filter { $0 == "|" }.count
282 let gaps = middle.filter { $0 == "-" }.count
283 output.append("score \(best) a \(i+1)-\(bi) b \(j+1)-\(bj) identity \(identity)/\(left.count) gaps \(gaps)")
284 output.append(String(left))
285 output.append(String(middle))
286 output.append(String(right))
287 }
288 func pair(_ a: Character, _ b: Character) -> Bool { complementBase(a) == b || a == "G" && b == "T" || a == "T" && b == "G" }
289 func value(_ f: [[Int]], _ i: Int, _ j: Int) -> Int { i > j ? 0 : f[i][j] }
290 func fold(_ s: String, _ loop: Int) throws {
291 let a = Array(s)
292 let n = a.count
293 try error(n > 100,"too long")
294 var f = Array(repeating:Array(repeating:0,count:n),count:n)
295 for length in 1...n {
296 for i in 0...(n-length) {
297 let j = i+length-1
298 f[i][j] = value(f,i,j-1)
299 if i <= j-loop-1 {
300 for k in i...(j-loop-1) where pair(a[k],a[j]) { f[i][j] = max(f[i][j],value(f,i,k-1)+1+value(f,k+1,j-1)) }
301 }
302 }
303 }
304 var structure = Array(repeating:Character("."),count:n)
305 var intervals = [(0,n-1)]
306 while let (i,j) = intervals.popLast() {
307 if value(f,i,j) == 0 { continue }
308 if f[i][j] == value(f,i,j-1) { intervals.append((i,j-1)) }
309 else if i <= j-loop-1 {
310 for k in i...(j-loop-1) {
311 if pair(a[k],a[j]) && value(f,i,k-1)+1+value(f,k+1,j-1) == f[i][j] {
312 structure[k] = "("
313 structure[j] = ")"
314 intervals.append((i,k-1))
315 intervals.append((k+1,j-1))
316 break
317 }
318 }
319 }
320 }
321 output.append(String(structure))
322 output.append("pairs \(f[0][n-1])")
323 }
324 func primer(_ s: String) {
325 let a = Array(s)
326 let n = a.count
327 let g = gc(s)
328 let temperature = tm(s)
329 let clamp = gc(String(s.suffix(5)))
330 var run = 1
331 var bestRun = 1
332 var repeats = 0
333 if n > 1 {
334 for i in 1..<n {
335 run = a[i] == a[i-1] ? run+1 : 1
336 bestRun = max(bestRun,run)
337 }
338 for i in 0..<n-1 where a[i] != a[i+1] { repeats = max(repeats,repeatCount(s,i,2)) }
339 }
340 let selfMatch = selfcomp(s)
341 let names = ["length","gc","tm","clamp","run","repeat","selfcomp"]
342 let values = [String(n),pct(g,n),fixed(temperature,10,1),String(clamp),String(bestRun),String(repeats),String(selfMatch)]
343 let ok = [n >= 18 && n <= 25,40*n <= 100*g && 100*g <= 60*n,temperature >= 520 && temperature <= 620,clamp >= 1 && clamp <= 3,bestRun <= 4,repeats <= 4,selfMatch <= 6]
344 var failed: [String] = []
345 for i in 0..<7 {
346 output.append(names[i]+" "+values[i]+" "+(ok[i] ? "OK" : "FAIL"))
347 if !ok[i] { failed.append(names[i]) }
348 }
349 output.append(failed.isEmpty ? "PASS" : "FAIL "+list(failed))
350 }
351 func assemble(_ name: String, _ minimum: Int, _ sequences: [String]) throws {
352 var unique: [String] = []
353 for s in sequences.dropFirst(2) where !unique.contains(s) { unique.append(s) }
354 var fragments = unique.filter { s in !unique.contains { $0 != s && $0.contains(s) } }
355 while fragments.count >= 2 {
356 var best = -1
357 var bx = 0
358 var by = 0
359 for x in fragments.indices {
360 for y in fragments.indices where x != y {
361 let current = overlap(fragments[x],fragments[y])
362 if current > best {
363 best = current
364 bx = x
365 by = y
366 }
367 }
368 }
369 if best < minimum { break }
370 let merged = fragments[bx]+fragments[by].part(best,fragments[by].count)
371 fragments[bx] = merged
372 fragments.remove(at:by)
373 var keep = bx-(by < bx ? 1 : 0)
374 for i in stride(from:fragments.count-1,through:0,by:-1) {
375 if i != keep && merged.contains(fragments[i]) {
376 fragments.remove(at:i)
377 if i < keep { keep -= 1 }
378 }
379 }
380 }
381 let longest = fragments.max { $0.count < $1.count }!
382 try error(longest.count > 500,"too long")
383 output.append("contigs \(fragments.count)")
384 for s in fragments { output.append("\(s.count) \(s)") }
385 try store(name,longest)
386 }
387 func execute(_ line: String) throws {
388 let t = line.split(separator:" ").map(String.init)
389 let a = Array(t.dropFirst())
390 var nums = Array(repeating:0,count:a.count)
391 var sequences = Array(repeating:"",count:a.count)
392 let command = t[0]
393 try validate(command,a,&nums,&sequences)
394 let s = sequences.first ?? ""
395 let n = s.count
396 switch command {
397 case "SEQ": try store(a[0],a[1])
398 case "SLICE":
399 let part = s.part(min(nums[1],nums[2])-1,max(nums[1],nums[2]))
400 try store(a[3],nums[1] <= nums[2] ? part : revcomp(part))
401 case "JOIN": try store(a[0],sequences.dropFirst().joined())
402 case "MUTATE":
403 let old = s[nums[1]-1]
404 let next = a[2][0]
405 try error(old == next,"no change")
406 registry[a[0]] = s.part(0,nums[1]-1)+String(next)+s.part(nums[1],n)
407 output.append("\(a[0]) \(old)\(nums[1])\(next) "+(transition(old,next) ? "transition" : "transversion"))
408 case "DEL":
409 registry.removeValue(forKey:a[0])
410 output.append("deleted "+a[0])
411 case "LIST":
412 if registry.isEmpty { output.append("no sequences") }
413 for name in registry.keys.sorted() { output.append("\(name) \(registry[name]!.count) nt") }
414 case "REVCOMP": output.append(revcomp(s))
415 case "COMPLEMENT": output.append(complement(s))
416 case "REVERSE": output.append(String(s.reversed()))
417 case "TRANSCRIBE": output.append(s.replacingOccurrences(of:"T",with:"U"))
418 case "TRANSLATE":
419 let protein = translation(codons(s,nums[1]))
420 output.append(protein.isEmpty ? "none" : protein)
421 case "CODONS":
422 let counts = wordCounts(codons(s,nums[1]))
423 if counts.isEmpty { output.append("none") }
424 for c in frequency(counts) { output.append("\(c) \(genetic[c]!) \(counts[c]!)") }
425 case "PROTEIN":
426 let full = translation(codons(s,nums[1]))
427 let protein = String(full.prefix { $0 != "*" })
428 try error(protein.isEmpty,"empty protein")
429 let weight = 18015+protein.reduce(0) { $0+residues[Array(amino).firstIndex(of:$1)!] }
430 let charge = protein.filter { "KR".contains($0) }.count-protein.filter { "DE".contains($0) }.count
431 let counts = wordCounts(protein.map { String($0) })
432 output.append(protein)
433 output.append("length \(protein.count) mass \(milli(weight)) charge "+(charge > 0 ? "+" : "")+String(charge))
434 output.append(frequency(counts).map { "\($0)=\(counts[$0]!)" }.joined(separator:" "))
435 case "ORFS":
436 var found: [Orf] = []
437 for frame in [1,2,3,-1,-2,-3] {
438 let codonList = codons(s,frame)
439 var start = -1
440 for i in codonList.indices {
441 if start < 0 && codonList[i] == "ATG" { start = i }
442 else if start >= 0 && genetic[codonList[i]] == "*" {
443 if i-start >= nums[1] {
444 let left = abs(frame)-1+3*start
445 let right = abs(frame)-1+3*(i+1)-1
446 found.append(Orf(frame:frame,left:frame > 0 ? left+1 : n-right,right:frame > 0 ? right+1 : n-left,protein:translation(Array(codonList[start..<i]))))
447 }
448 start = -1
449 }
450 }
451 }
452 if found.isEmpty { output.append("none") }
453 for o in found.sorted(by:{ $0.left < $1.left }) { output.append((o.frame > 0 ? "+" : "")+"\(o.frame) \(o.left) \(o.right) \(o.protein.count) \(o.protein)") }
454 case "GC":
455 let composition = bases.map { base in "\(base)=\(s.filter { $0 == base }.count)" }.joined(separator:" ")
456 output.append(composition+" GC=\(pct(gc(s),n))%")
457 case "GCWINDOW":
458 try error(nums[1] > n,"window too large")
459 for i in stride(from:0,through:n-nums[1],by:nums[2]) { output.append("\(i+1)-\(i+nums[1]) \(pct(gc(s.part(i,i+nums[1])),nums[1]))%") }
460 case "SKEW":
461 var skew = [0]
462 for c in s { skew.append(skew.last!+(c == "G" ? 1 : c == "C" ? -1 : 0)) }
463 let low = skew.min()!
464 let high = skew.max()!
465 output.append("min \(low) at "+list(skew.indices.filter { skew[$0] == low }))
466 output.append("max \(high) at "+list(skew.indices.filter { skew[$0] == high }))
467 case "KMERS":
468 try error(nums[1] > n,"k too large")
469 let words = (0...(n-nums[1])).map { s.part($0,$0+nums[1]) }
470 let counts = wordCounts(words)
471 output.append("\(words.count) windows \(counts.count) distinct")
472 for word in frequency(counts).prefix(nums[2]) { output.append("\(word) \(counts[word]!)") }
473 case "FIND":
474 output.append("forward "+list(occurrences(s,a[1])))
475 output.append("reverse "+list(occurrences(s,revcomp(a[1]))))
476 case "RESTRICT":
477 var allCuts = Set<Int>()
478 for enzyme in a.dropFirst() {
479 let e = enzymes.firstIndex(of:enzyme)!
480 let found = occurrences(s,sites[e]).map { $0+cuts[e]-1 }.filter { $0 >= 1 && $0 < n }
481 allCuts.formUnion(found)
482 output.append(enzyme+" "+list(found))
483 }
484 let ends = allCuts.sorted()+[n]
485 var previous = 0
486 var fragments: [String] = []
487 var sizes: [Int] = []
488 for end in ends {
489 fragments.append("\(previous+1)-\(end)")
490 sizes.append(end-previous)
491 previous = end
492 }
493 output.append("fragments "+list(fragments))
494 output.append("sizes "+list(sizes.sorted(by:>)))
495 case "PALINDROMES":
496 try error(nums[1] > nums[2],"invalid range")
497 var count = 0
498 for i in 0..<n {
499 let last = min(nums[2],n-i)
500 if nums[1] <= last {
501 for length in nums[1]...last {
502 let part = s.part(i,i+length)
503 if part == revcomp(part) {
504 output.append("\(i+1) \(length) \(part)")
505 count += 1
506 }
507 }
508 }
509 }
510 if count == 0 { output.append("none") }
511 case "REPEATS":
512 try error(nums[1] > nums[2],"invalid range")
513 var found: [Repeat] = []
514 for u in nums[1]...nums[2] {
515 var i = 0
516 while i+u <= n {
517 let unit = s.part(i,i+u)
518 let count = repeatCount(s,i,u)
519 if primitive(unit) && count >= nums[3] {
520 found.append(Repeat(start:i+1,end:i+count*u,unitLength:u,unit:unit,count:count))
521 i += count*u
522 } else { i += 1 }
523 }
524 }
525 if found.isEmpty { output.append("none") }
526 for r in found.sorted(by:{ $0.start == $1.start ? $0.unitLength < $1.unitLength : $0.start < $1.start }) { output.append("\(r.start)-\(r.end) \(r.unit)x\(r.count)") }
527 case "HAMMING":
528 let b = sequences[1]
529 try error(n != b.count,"length mismatch")
530 let pairs = Array(zip(s,b))
531 let distance = pairs.filter { $0.0 != $0.1 }.count
532 let transitions = pairs.filter { transition($0.0,$0.1) }.count
533 output.append("distance \(distance) transitions \(transitions) transversions \(distance-transitions)")
534 case "ALIGN": align(s,sequences[1],a[2] == "LOCAL",nums[3],nums[4],nums[5])
535 case "CONSENSUS":
536 try error(sequences.contains { $0.count != n },"length mismatch")
537 let rows = sequences.map { Array($0) }
538 var profile = Array(repeating:Array(repeating:0,count:n),count:4)
539 var consensus = ""
540 var iupac = ""
541 for i in 0..<n {
542 for sequence in rows { profile[Array(bases).firstIndex(of:sequence[i])!][i] += 1 }
543 let maximum = (0..<4).map { profile[$0][i] }.max()!
544 let winners = (0..<4).filter { profile[$0][i] == maximum }
545 consensus.append(bases[winners[0]])
546 let mask = winners.reduce(0) { $0+(1 << $1) }
547 iupac.append(symbols[masks.firstIndex(of:mask)!])
548 }
549 for b in 0..<4 { output.append("\(bases[b]) "+list(profile[b])) }
550 output.append("consensus "+consensus)
551 output.append("iupac "+iupac)
552 case "TM": output.append("Tm \(fixed(tm(s),10,1)) "+(n < 14 ? "wallace" : "gc"))
553 case "MASS": output.append("ss \(milli(mass(s))) ds \(milli(mass(s)+mass(revcomp(s))))")
554 case "PRIMER": primer(s)
555 case "FOLD": try fold(s,nums[1])
556 case "ASSEMBLE": try assemble(a[0],nums[1],sequences)
557 case "PACK":
558 let sequence = Array(s)
559 var hex = ""
560 for i in stride(from:0,to:n,by:4) {
561 var value = 0
562 for j in 0..<4 { value = 4*value+(i+j < n ? Array(bases).firstIndex(of:sequence[i+j])! : 0) }
563 hex += String(format:"%02X",locale:Locale(identifier:"en_US_POSIX"),value)
564 }
565 output.append("\(n) \(hex)")
566 case "UNPACK":
567 var decoded: [Character] = []
568 for i in stride(from:0,to:a[2].count,by:2) {
569 let value = Int(a[2].part(i,i+2),radix:16)!
570 for shift in [6,4,2,0] { decoded.append(bases[(value >> shift)&3]) }
571 }
572 for i in nums[1]..<decoded.count { try error(decoded[i] != "A","invalid padding") }
573 try store(a[0],String(decoded.prefix(nums[1])))
574 default: break
575 }
576 }
577 func process(_ line: String) -> [String] {
578 output = []
579 do { try execute(line) }
580 catch Fault.reason(let reason) {
581 output = []
582 output.append("ERROR "+reason)
583 }
584 catch { fatalError("Unexpected error") }
585 return output
586 }
587}
