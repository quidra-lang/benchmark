42private class Fault(reason: String) : RuntimeException(reason)
43private fun error(condition: Boolean, reason: String) { if (condition) throw Fault(reason) }
44private fun number(s: String, low: Int, high: Int): Int {
45 error(!s.matches(Regex("-?[0-9]{1,6}")),"invalid number $s")
46 return s.toInt().also { error(it !in low..high,"invalid number $s") }

48private fun complement(s: String) = s.map { COMPLEMENTS[SYMBOLS.indexOf(it)] }.joinToString("")
49private fun revcomp(s: String) = complement(s).reversed()
50private fun complementBase(c: Char) = BASES[3-BASES.indexOf(c)]
51private fun gc(s: String) = s.count { it == 'G' || it == 'C' }
52private fun fixed(n: Int, divisor: Int, width: Int) = "${n/divisor}."+(n%divisor).toString().padStart(width,'0')
53private fun pct(a: Int, b: Int) = fixed(10000*a/b,100,2)
54private fun milli(n: Int) = fixed(n,1000,3)
55private fun list(values: Collection<*>) = if (values.isEmpty()) "none" else values.joinToString(" ")
56private fun codons(s: String, frame: Int): List<String> {
57 val strand = if (frame > 0) s else revcomp(s)
58 val result = mutableListOf<String>()
59 var i = abs(frame)-1
60 while (i+3 <= s.length) {
61 result.add(strand.substring(i,i+3))
62 i += 3
63 }
64 return result
65}
66private fun translation(codons: List<String>) = codons.map { genetic.getValue(it) }.joinToString("")
67private fun wordCounts(words: List<String>) = words.groupingBy { it }.eachCount()
68private fun frequency(counts: Map<String,Int>) = counts.keys.sortedWith(compareByDescending<String> { counts.getValue(it) }.thenBy { it })
69private fun occurrences(s: String, motif: String): List<Int> {
70 return (0..s.length-motif.length).filter { i ->
71 motif.indices.all { j -> masks[SYMBOLS.indexOf(motif[j])] and (1 shl BASES.indexOf(s[i+j])) != 0 }
72 }.map { it+1 }
73}
74private fun transition(a: Char, b: Char) = a != b && (a in "AG" && b in "AG" || a in "CT" && b in "CT")
75private fun repeatCount(s: String, i: Int, u: Int): Int {
76 val unit = s.substring(i,i+u)
77 var count = 1
78 while (i+(count+1)*u <= s.length && s.substring(i+count*u,i+(count+1)*u) == unit) count++
79 return count
80}
81private fun primitive(s: String) = (1 until s.length).none { s.length%it == 0 && s.take(it).repeat(s.length/it) == s }
82private fun selfcomp(s: String): Int {
83 var maximum = 0
84 for (center in 1 until s.length) {
85 var left = center-1
86 var right = center
87 while (left >= 0 && right < s.length && complementBase(s[left]) == s[right]) {
88 maximum = max(maximum,right-left+1)
89 left--
90 right++
91 }
92 }
93 return maximum
94}
95private fun tm(s: String): Int {
96 val n = s.length
97 val g = gc(s)
98 return if (n < 14) 10*(2*(n-g)+4*g) else (649*n+410*g-6724)/n
99}
100private fun mass(s: String) = s.sumOf { baseMass[BASES.indexOf(it)] }-(s.length-1)*18015
101private fun overlap(a: String, b: String): Int {
102 val joined = b+"#"+a
103 val prefix = IntArray(joined.length)
104 for (i in 1 until joined.length) {
105 var j = prefix[i-1]
106 while (j > 0 && joined[i] != joined[j]) j = prefix[j-1]
107 if (joined[i] == joined[j]) j++
108 prefix[i] = j
109 }
110 var result = prefix.last()
111 val limit = min(a.length,b.length)
112 while (result >= limit) result = prefix[result-1]
113 return result
114}

148 fun store(name: String, sequence: String) {
149 error(sequence.length > 500,"too long")
150 val replacing = name in registry
151 registry[name] = sequence
152 output.add((if (replacing) "replaced " else "stored ")+"$name ${sequence.length} nt")
153 }
154 fun validate(command: String, a: List<String>, nums: IntArray, sequences: Array<String>) {
155 val schema = schemas[command] ?: throw Fault("unknown command $command")
156 error(a.size !in arities.getValue(command),"wrong argument count")
157 val seen = mutableSetOf<String>()
158 for ((i,value) in a.withIndex()) {
159 when (val kind = schema[min(i,schema.lastIndex)]) {
160 "NAME" -> sequences[i] = registry[value] ?: throw Fault("unknown sequence $value")
161 "NEW" -> error(!value.matches(Regex("[a-z][a-z0-9]{0,9}")),"invalid name $value")
162 "SEQ" -> for ((j,c) in value.withIndex()) error(c !in BASES,"invalid base $c at ${j+1}")
163 "BASE" -> error(value.length != 1 || value[0] !in BASES,"invalid base $value")
164 "MOTIF" -> error(!value.matches(Regex("[ACGTRYSWKMBDHVN]{1,30}")),"invalid motif $value")
165 "ENZYME" -> {
166 error(value !in enzymes,"unknown enzyme $value")
167 error(!seen.add(value),"duplicate enzyme $value")
168 }
169 "MODE" -> error(value != "GLOBAL" && value != "LOCAL","invalid mode $value")
170 "HEX" -> error(value.length != 2*((nums[1]+3)/4) || !value.matches(Regex("[0-9A-F]+")),"invalid hex")
171 "FRAME" -> {
172 nums[i] = number(value,-3,3)
173 error(nums[i] == 0,"invalid number $value")
174 }
175 "POS" -> nums[i] = number(value,1,sequences[0].length)
176 else -> {
177 val bounds = kind.split(':')
178 nums[i] = number(value,bounds[0].toInt(),bounds[1].toInt())
179 }
180 }
181 }
182 }
183 fun align(a: String, b: String, local: Boolean, match: Int, mismatch: Int, gap: Int) {
184 val m = a.length
185 val n = b.length
186 val h = Array(m+1) { IntArray(n+1) }
187 if (!local) {
188 for (i in 0..m) h[i][0] = i*gap
189 for (j in 0..n) h[0][j] = j*gap
190 }
191 var best = 0
192 var bi = 0
193 var bj = 0
194 for (i in 1..m) {
195 for (j in 1..n) {
196 val value = max(h[i-1][j-1]+(if (a[i-1] == b[j-1]) match else mismatch),max(h[i-1][j]+gap,h[i][j-1]+gap))
197 h[i][j] = if (local) max(0,value) else value
198 if (local && h[i][j] > best) {
199 best = h[i][j]
200 bi = i
201 bj = j
202 }
203 }
204 }
205 if (!local) {
206 bi = m
207 bj = n
208 best = h[m][n]
209 }
210 if (local && best == 0) {
211 output.add("score 0")
212 return
213 }
214 var i = bi
215 var j = bj
216 val x = StringBuilder()
217 val y = StringBuilder()
218 while (if (local) h[i][j] != 0 else i != 0 || j != 0) {
219 if (i > 0 && j > 0 && h[i][j] == h[i-1][j-1]+(if (a[i-1] == b[j-1]) match else mismatch)) {
220 x.append(a[--i])
221 y.append(b[--j])
222 } else if (i > 0 && h[i][j] == h[i-1][j]+gap) {
223 x.append(a[--i])
224 y.append('-')
225 } else {
226 x.append('-')
227 y.append(b[--j])
228 }
229 }
230 val left = x.reverse().toString()
231 val right = y.reverse().toString()
232 val middle = left.indices.map { k -> if (left[k] == '-' || right[k] == '-') '-' else if (left[k] == right[k]) '|' else '*' }.joinToString("")
233 output.add("score $best a ${i+1}-$bi b ${j+1}-$bj identity ${middle.count { it == '|' }}/${left.length} gaps ${middle.count { it == '-' }}")
234 output.add(left)
235 output.add(middle)
236 output.add(right)
237 }
238 fun pair(a: Char, b: Char) = complementBase(a) == b || a == 'G' && b == 'T' || a == 'T' && b == 'G'
239 fun value(f: Array<IntArray>, i: Int, j: Int) = if (i > j) 0 else f[i][j]
240 fun fold(s: String, loop: Int) {
241 val n = s.length
242 error(n > 100,"too long")
243 val f = Array(n) { IntArray(n) }
244 for (length in 1..n) {
245 for (i in 0..n-length) {
246 val j = i+length-1
247 f[i][j] = value(f,i,j-1)
248 for (k in i..j-loop-1) {
249 if (pair(s[k],s[j])) f[i][j] = max(f[i][j],value(f,i,k-1)+1+value(f,k+1,j-1))
250 }
251 }
252 }
253 val structure = CharArray(n) { '.' }
254 val intervals = ArrayDeque<Pair<Int,Int>>()
255 intervals.addLast(0 to n-1)
256 while (intervals.isNotEmpty()) {
257 val (i,j) = intervals.removeLast()
258 if (value(f,i,j) == 0) continue
259 if (f[i][j] == value(f,i,j-1)) intervals.addLast(i to j-1)
260 else {
261 for (k in i..j-loop-1) {
262 if (pair(s[k],s[j]) && value(f,i,k-1)+1+value(f,k+1,j-1) == f[i][j]) {
263 structure[k] = '('
264 structure[j] = ')'
265 intervals.addLast(i to k-1)
266 intervals.addLast(k+1 to j-1)
267 break
268 }
269 }
270 }
271 }
272 output.add(structure.concatToString())
273 output.add("pairs ${f[0][n-1]}")
274 }
275 fun primer(s: String) {
276 val n = s.length
277 val g = gc(s)
278 val temperature = tm(s)
279 val clamp = gc(s.takeLast(5))
280 var run = 1
281 var bestRun = 1
282 var repeats = 0
283 for (i in 1 until n) {
284 run = if (s[i] == s[i-1]) run+1 else 1
285 bestRun = max(bestRun,run)
286 }
287 for (i in 0 until n-1) if (s[i] != s[i+1]) repeats = max(repeats,repeatCount(s,i,2))
288 val self = selfcomp(s)
289 val names = listOf("length","gc","tm","clamp","run","repeat","selfcomp")
290 val values = listOf(n.toString(),pct(g,n),fixed(temperature,10,1),clamp.toString(),bestRun.toString(),repeats.toString(),self.toString())
291 val ok = listOf(n in 18..25,40*n <= 100*g && 100*g <= 60*n,temperature in 520..620,clamp in 1..3,bestRun <= 4,repeats <= 4,self <= 6)
292 val failed = mutableListOf<String>()
293 for (i in 0..6) {
294 output.add("${names[i]} ${values[i]} "+(if (ok[i]) "OK" else "FAIL"))
295 if (!ok[i]) failed.add(names[i])
296 }
297 output.add(if (failed.isEmpty()) "PASS" else "FAIL "+list(failed))
298 }
299 fun assemble(name: String, minimum: Int, sequences: Array<String>) {
300 val unique = sequences.drop(2).distinct()
301 val fragments = unique.filter { s -> unique.none { it != s && it.contains(s) } }.toMutableList()
302 while (fragments.size >= 2) {
303 var best = -1
304 var bx = 0
305 var by = 0
306 for (x in fragments.indices) {
307 for (y in fragments.indices) {
308 if (x == y) continue
309 val a = fragments[x]
310 val b = fragments[y]
311 val overlap = overlap(a,b)
312 if (overlap > best) {
313 best = overlap
314 bx = x
315 by = y
316 }
317 }
318 }
319 if (best < minimum) break
320 val merged = fragments[bx]+fragments[by].drop(best)
321 fragments[bx] = merged
322 fragments.removeAt(by)
323 var keep = bx-(if (by < bx) 1 else 0)
324 for (i in fragments.lastIndex downTo 0) {
325 if (i != keep && merged.contains(fragments[i])) {
326 fragments.removeAt(i)
327 if (i < keep) keep--
328 }
329 }
330 }
331 val longest = fragments.maxBy { it.length }
332 error(longest.length > 500,"too long")
333 output.add("contigs ${fragments.size}")
334 for (s in fragments) output.add("${s.length} $s")
335 store(name,longest)
336 }
337 fun execute(line: String) {
338 val t = line.split(' ')
339 val a = t.drop(1)
340 val nums = IntArray(a.size)
341 val sequences = Array(a.size) { "" }
342 val command = t[0]
343 validate(command,a,nums,sequences)
344 val s = sequences.firstOrNull() ?: ""
345 val n = s.length
346 when (command) {
347 "SEQ" -> store(a[0],a[1])
348 "SLICE" -> {
349 val part = s.substring(min(nums[1],nums[2])-1,max(nums[1],nums[2]))
350 store(a[3],if (nums[1] <= nums[2]) part else revcomp(part))
351 }
352 "JOIN" -> store(a[0],sequences.drop(1).joinToString(""))
353 "MUTATE" -> {
354 val old = s[nums[1]-1]
355 val next = a[2][0]
356 error(old == next,"no change")
357 registry[a[0]] = s.take(nums[1]-1)+next+s.drop(nums[1])
358 output.add("${a[0]} $old${nums[1]}$next "+(if (transition(old,next)) "transition" else "transversion"))
359 }
360 "DEL" -> {
361 registry.remove(a[0])
362 output.add("deleted ${a[0]}")
363 }
364 "LIST" -> {
365 if (registry.isEmpty()) output.add("no sequences")
366 for ((name,sequence) in registry) output.add("$name ${sequence.length} nt")
367 }
368 "REVCOMP" -> output.add(revcomp(s))
369 "COMPLEMENT" -> output.add(complement(s))
370 "REVERSE" -> output.add(s.reversed())
371 "TRANSCRIBE" -> output.add(s.replace('T','U'))
372 "TRANSLATE" -> output.add(translation(codons(s,nums[1])).ifEmpty { "none" })
373 "CODONS" -> {
374 val counts = wordCounts(codons(s,nums[1]))
375 if (counts.isEmpty()) output.add("none")
376 for (c in frequency(counts)) output.add("$c ${genetic[c]} ${counts[c]}")
377 }
378 "PROTEIN" -> {
379 val protein = translation(codons(s,nums[1])).substringBefore('*')
380 error(protein.isEmpty(),"empty protein")
381 val mass = 18015+protein.sumOf { residues[AMINO.indexOf(it)] }
382 val charge = protein.count { it in "KR" }-protein.count { it in "DE" }
383 val counts = wordCounts(protein.map { it.toString() })
384 output.add(protein)
385 output.add("length ${protein.length} mass ${milli(mass)} charge "+(if (charge > 0) "+" else "")+charge)
386 output.add(frequency(counts).joinToString(" ") { "$it=${counts[it]}" })
387 }
388 "ORFS" -> {
389 val found = mutableListOf<Orf>()
390 for (frame in intArrayOf(1,2,3,-1,-2,-3)) {
391 val codons = codons(s,frame)
392 var start = -1
393 for (i in codons.indices) {
394 if (start < 0 && codons[i] == "ATG") start = i
395 else if (start >= 0 && genetic[codons[i]] == '*') {
396 if (i-start >= nums[1]) {
397 val left = abs(frame)-1+3*start
398 val right = abs(frame)-1+3*(i+1)-1
399 found.add(Orf(frame,if (frame > 0) left+1 else n-right,if (frame > 0) right+1 else n-left,translation(codons.subList(start,i))))
400 }
401 start = -1
402 }
403 }
404 }
405 if (found.isEmpty()) output.add("none")
406 for (o in found.sortedBy { it.left }) output.add((if (o.frame > 0) "+" else "")+"${o.frame} ${o.left} ${o.right} ${o.protein.length} ${o.protein}")
407 }
408 "GC" -> output.add(BASES.map { "$it=${s.count { c -> c == it }}" }.joinToString(" ")+" GC=${pct(gc(s),n)}%")
409 "GCWINDOW" -> {
410 error(nums[1] > n,"window too large")
411 for (i in 0..n-nums[1] step nums[2]) output.add("${i+1}-${i+nums[1]} ${pct(gc(s.substring(i,i+nums[1])),nums[1])}%")
412 }
413 "SKEW" -> {
414 val skew = IntArray(n+1)
415 for (i in 0 until n) skew[i+1] = skew[i]+(if (s[i] == 'G') 1 else if (s[i] == 'C') -1 else 0)
416 val low = skew.minOrNull()!!
417 val high = skew.maxOrNull()!!
418 output.add("min $low at "+list(skew.indices.filter { skew[it] == low }))
419 output.add("max $high at "+list(skew.indices.filter { skew[it] == high }))
420 }
421 "KMERS" -> {
422 error(nums[1] > n,"k too large")
423 val words = (0..n-nums[1]).map { s.substring(it,it+nums[1]) }
424 val counts = wordCounts(words)
425 output.add("${words.size} windows ${counts.size} distinct")
426 for (word in frequency(counts).take(nums[2])) output.add("$word ${counts[word]}")
427 }
428 "FIND" -> {
429 output.add("forward "+list(occurrences(s,a[1])))
430 output.add("reverse "+list(occurrences(s,revcomp(a[1]))))
431 }
432 "RESTRICT" -> {
433 val allCuts = sortedSetOf<Int>()
434 for (enzyme in a.drop(1)) {
435 val e = enzymes.indexOf(enzyme)
436 val found = occurrences(s,sites[e]).map { it+cuts[e]-1 }.filter { it in 1 until n }
437 allCuts.addAll(found)
438 output.add("$enzyme "+list(found))
439 }
440 val ends = allCuts.toList()+n
441 var previous = 0
442 val fragments = mutableListOf<String>()
443 val sizes = mutableListOf<Int>()
444 for (end in ends) {
445 fragments.add("${previous+1}-$end")
446 sizes.add(end-previous)
447 previous = end
448 }
449 output.add("fragments "+list(fragments))
450 output.add("sizes "+list(sizes.sortedDescending()))
451 }
452 "PALINDROMES" -> {
453 error(nums[1] > nums[2],"invalid range")
454 var count = 0
455 for (i in 0 until n) {
456 for (length in nums[1]..min(nums[2],n-i)) {
457 val part = s.substring(i,i+length)
458 if (part == revcomp(part)) {
459 output.add("${i+1} $length $part")
460 count++
461 }
462 }
463 }
464 if (count == 0) output.add("none")
465 }
466 "REPEATS" -> {
467 error(nums[1] > nums[2],"invalid range")
468 val found = mutableListOf<Repeat>()
469 for (u in nums[1]..nums[2]) {
470 var i = 0
471 while (i+u <= n) {
472 val unit = s.substring(i,i+u)
473 val count = repeatCount(s,i,u)
474 if (primitive(unit) && count >= nums[3]) {
475 found.add(Repeat(i+1,i+count*u,u,unit,count))
476 i += count*u
477 } else i++
478 }
479 }
480 if (found.isEmpty()) output.add("none")
481 for (r in found.sortedWith(compareBy<Repeat> { it.start }.thenBy { it.unitLength })) output.add("${r.start}-${r.end} ${r.unit}x${r.count}")
482 }
483 "HAMMING" -> {
484 val b = sequences[1]
485 error(n != b.length,"length mismatch")
486 val pairs = s.zip(b)
487 val distance = pairs.count { (a,b) -> a != b }
488 val transitions = pairs.count { (a,b) -> transition(a,b) }
489 output.add("distance $distance transitions $transitions transversions ${distance-transitions}")
490 }
491 "ALIGN" -> align(s,sequences[1],a[2] == "LOCAL",nums[3],nums[4],nums[5])
492 "CONSENSUS" -> {
493 error(sequences.any { it.length != n },"length mismatch")
494 val profile = Array(4) { IntArray(n) }
495 val consensus = StringBuilder()
496 val iupac = StringBuilder()
497 for (i in 0 until n) {
498 for (sequence in sequences) profile[BASES.indexOf(sequence[i])][i]++
499 val maximum = (0..3).maxOf { profile[it][i] }
500 val winners = (0..3).filter { profile[it][i] == maximum }
501 consensus.append(BASES[winners[0]])
502 val mask = winners.sumOf { 1 shl it }
503 iupac.append(SYMBOLS[masks.indexOf(mask)])
504 }
505 for (b in 0..3) output.add("${BASES[b]} "+list(profile[b].toList()))
506 output.add("consensus $consensus")
507 output.add("iupac $iupac")
508 }
509 "TM" -> output.add("Tm ${fixed(tm(s),10,1)} "+(if (n < 14) "wallace" else "gc"))
510 "MASS" -> output.add("ss ${milli(mass(s))} ds ${milli(mass(s)+mass(revcomp(s)))}")
511 "PRIMER" -> primer(s)
512 "FOLD" -> fold(s,nums[1])
513 "ASSEMBLE" -> assemble(a[0],nums[1],sequences)
514 "PACK" -> {
515 val hex = StringBuilder()
516 for (i in 0 until n step 4) {
517 var value = 0
518 for (j in 0..3) value = 4*value+(if (i+j < n) BASES.indexOf(s[i+j]) else 0)
519 hex.append(value.toString(16).uppercase().padStart(2,'0'))
520 }
521 output.add("$n $hex")
522 }
523 "UNPACK" -> {
524 val decoded = StringBuilder()
525 for (i in a[2].indices step 2) {
526 val value = a[2].substring(i,i+2).toInt(16)
527 for (shift in intArrayOf(6,4,2,0)) decoded.append(BASES[(value shr shift) and 3])
528 }
529 for (i in nums[1] until decoded.length) error(decoded[i] != 'A',"invalid padding")
530 store(a[0],decoded.substring(0,nums[1]))
531 }
532 }
533 }
534 fun process(line: String): List<String> {
535 output = mutableListOf()
536 try { execute(line) }
537 catch (f: Fault) {
538 output.clear()
539 output.add("ERROR ${f.message}")
540 }
541 return output
542 }
543}
