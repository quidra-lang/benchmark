70revcomp(sequence::AbstractString) = join(COMPLEMENT[base] for base in reverse(sequence))
71gc(sequence::AbstractString) = count(base -> base == 'G' || base == 'C', sequence)
72fixed(value::Int, places::Int) = string(div(value, 10^places), ".", lpad(string(mod(value, 10^places)), places, '0'))
73pct(part::Int, whole::Int) = fixed(div(10000 * part, whole), 2)
74transition(a::Char, b::Char) = a != b && string(a, b) in ("AG", "GA", "CT", "TC")
75frequency_order(table) = sort(collect(keys(table)); by=key -> (-table[key], key))

77function codons(sequence::String, frame::Int)
78 strand = frame > 0 ? sequence : revcomp(sequence)
79 triplets = String[]
80 for i in abs(frame):3:(length(strand) - 2)
81 push!(triplets, strand[i:i+2])
82 end
83 triplets
84end

86function occurrences(sequence::String, motif::String)
87 positions = Int[]
88 n, width = length(sequence), length(motif)
89 for i in 1:(n - width + 1)
90 if all(sequence[i+j-1] in IUPAC[motif[j]] for j in 1:width)
91 push!(positions, i)
92 end
93 end
94 positions
95end

97function repeat_count(sequence::String, position::Int, width::Int)
98 unit = sequence[position:position+width-1]
99 count = 0
100 while position + (count + 1) * width - 1 <= length(sequence) &&
101 sequence[position+count*width:position+(count+1)*width-1] == unit
102 count += 1
103 end
104 count
105end

107function primitive_motif(unit::String)
108 n = length(unit)
109 all(n % width != 0 || repeat(unit[1:width], div(n, width)) != unit
110 for width in 1:(n-1))
111end

113function longest_palindrome(sequence::String)
114 best = 0
115 for center in 2:length(sequence)
116 left, right = center - 1, center
117 while left >= 1 && right <= length(sequence) &&
118 COMPLEMENT[sequence[left]] == sequence[right]
119 best = max(best, right - left + 1)
120 left -= 1
121 right += 1
122 end
123 end
124 best
125end

127function temperature(sequence::String)
128 n, amount = length(sequence), gc(sequence)
129 n < 14 ? 10 * (2 * (n - amount) + 4 * amount) :
130 fld(649 * n + 410 * amount - 6724, n)
131end

146function number(text::String, minimum::Int, maximum::Int; nonzero::Bool=false)
147 occursin(r"^-?[0-9]{1,6}$", text) || fail("invalid number " * text)
148 value = parse(Int, text)
149 minimum <= value <= maximum && (!nonzero || value != 0) ||
150 fail("invalid number " * text)
151 value
152end

154function kinds(command::String, count::Int)
155 if command == "SEQ"
156 return ["new", "sequence"]
157 elseif command == "SLICE"
158 return ["name", "position", "position", "new"]
159 elseif command == "JOIN"
160 return vcat(["new"], fill("name", count - 1))
161 elseif command == "MUTATE"
162 return ["name", "position", "base"]
163 elseif command in ("TRANSLATE", "CODONS", "PROTEIN")
164 return ["name", "frame"]
165 elseif command == "ORFS"
166 return ["name", "orf"]
167 elseif command == "GCWINDOW"
168 return ["name", "window", "window"]
169 elseif command == "KMERS"
170 return ["name", "kmer", "limit"]
171 elseif command == "FIND"
172 return ["name", "motif"]
173 elseif command == "RESTRICT"
174 return vcat(["name"], fill("enzyme", count - 1))
175 elseif command == "PALINDROMES"
176 return ["name", "palindrome", "palindrome"]
177 elseif command == "REPEATS"
178 return ["name", "repeat", "repeat", "repeat_count"]
179 elseif command == "ALIGN"
180 return ["name", "name", "mode", "match", "mismatch", "gap"]
181 elseif command == "FOLD"
182 return ["name", "loop"]
183 elseif command == "ASSEMBLE"
184 return vcat(["new", "overlap"], fill("name", count - 2))
185 elseif command == "UNPACK"
186 return ["new", "length", "hex"]
187 end
188 fill("name", count)
189end

191function validate(tool::Toolkit, command::String, arguments::Vector{String})
192 haskey(ARITY, command) || fail("unknown command " * command)
193 minimum, maximum = ARITY[command]
194 minimum <= length(arguments) <= maximum || fail("wrong argument count")
195 values = Any[arguments...]
196 seen_enzymes = Set{String}()
197 for (i, (kind, text)) in enumerate(zip(kinds(command, length(arguments)), arguments))
198 if kind in ("position", "frame", "orf", "window", "kmer", "limit",
199 "palindrome", "repeat", "repeat_count", "match", "mismatch",
200 "gap", "loop", "overlap", "length")
201 bounds = if kind == "position"
202 (1, length(tool.sequences[arguments[1]]))
203 elseif kind == "frame"
204 (-3, 3)
205 elseif kind == "orf"
206 (1, 200)
207 elseif kind == "window"
208 (1, 500)
209 elseif kind == "kmer"
210 (1, 12)
211 elseif kind == "limit"
212 (1, 100)
213 elseif kind == "palindrome"
214 (2, 20)
215 elseif kind == "repeat"
216 (1, 6)
217 elseif kind == "repeat_count"
218 (2, 50)
219 elseif kind == "match"
220 (1, 20)
221 elseif kind == "mismatch"
222 (-20, 0)
223 elseif kind == "gap"
224 (-20, -1)
225 elseif kind == "loop"
226 (0, 10)
227 elseif kind == "overlap"
228 (1, 100)
229 else
230 (1, 500)
231 end
232 values[i] = number(text, bounds[1], bounds[2]; nonzero=kind == "frame")
233 elseif kind == "name"
234 haskey(tool.sequences, text) || fail("unknown sequence " * text)
235 elseif kind == "new"
236 occursin(r"^[a-z][a-z0-9]{0,9}$", text) || fail("invalid name " * text)
237 elseif kind == "sequence"
238 for (position, base) in enumerate(text)
239 base in BASES || fail("invalid base $base at $position")
240 end
241 elseif kind == "base"
242 length(text) == 1 && text[1] in BASES || fail("invalid base " * text)
243 elseif kind == "motif"
244 1 <= length(text) <= 30 && all(haskey(IUPAC, c) for c in text) ||
245 fail("invalid motif " * text)
246 elseif kind == "enzyme"
247 haskey(ENZYMES, text) || fail("unknown enzyme " * text)
248 text in seen_enzymes && fail("duplicate enzyme " * text)
249 push!(seen_enzymes, text)
250 elseif kind == "mode"
251 text in ("GLOBAL", "LOCAL") || fail("invalid mode " * text)
252 elseif kind == "hex"
253 length(text) == 2 * cld(values[2], 4) &&
254 occursin(r"^[0-9A-F]+$", text) || fail("invalid hex")
255 end
256 end
257 values
258end

267function translation!(tool::Toolkit, command::String, name::String, frame::Int)
268 triplets = codons(tool.sequences[name], frame)
269 protein = join(GENETIC_CODE[c] for c in triplets)
270 if command == "TRANSLATE"
271 emit!(tool, isempty(protein) ? "none" : protein)
272 elseif command == "CODONS"
273 table = Dict(codon => count(==(codon), triplets) for codon in unique(triplets))
274 isempty(table) && emit!(tool, "none")
275 for codon in frequency_order(table)
276 emit!(tool, codon, GENETIC_CODE[codon], table[codon])
277 end
278 else
279 protein = first(split(protein, "*"; limit=2))
280 isempty(protein) && fail("empty protein")
281 table = Dict(amino => count(==(amino), protein) for amino in unique(protein))
282 mass = sum(AMINO_MASS[c] for c in protein) + 18015
283 charge = count(c -> c in "KR", protein) - count(c -> c in "DE", protein)
284 emit!(tool, protein)
285 emit!(tool, "length", length(protein), "mass", fixed(mass, 3),
286 "charge", charge > 0 ? "+$charge" : string(charge))
287 emit!(tool, join(["$amino=$(table[amino])" for amino in frequency_order(table)], " "))
288 end
289end

291function orfs!(tool::Toolkit, sequence::String, minimum::Int)
292 rows = Tuple{Int, Int, Int, String}[]
293 for frame in (1, 2, 3, -1, -2, -3)
294 triplets = codons(sequence, frame)
295 first_codon = nothing
296 for (position, codon) in enumerate(triplets)
297 if first_codon === nothing
298 codon == "ATG" && (first_codon = position)
299 elseif GENETIC_CODE[codon] == "*"
300 if position - first_codon >= minimum
301 left = abs(frame) - 1 + 3 * (first_codon - 1)
302 right = abs(frame) - 1 + 3 * position
303 if frame < 0
304 left, right = length(sequence) - right, length(sequence) - left
305 end
306 protein = join(GENETIC_CODE[triplets[i]] for i in first_codon:(position-1))
307 push!(rows, (left + 1, right, frame, protein))
308 end
309 first_codon = nothing
310 end
311 end
312 end
313 isempty(rows) && emit!(tool, "none")
314 for (left, right, frame, protein) in sort(rows)
315 emit!(tool, @sprintf("%+d", frame), left, right, length(protein), protein)
316 end
317end

319function restrict!(tool::Toolkit, sequence::String, selected::Vector{String})
320 cuts = Set{Int}()
321 for enzyme in selected
322 motif, offset = ENZYMES[enzyme]
323 positions = [p + offset - 1 for p in occurrences(sequence, motif)
324 if 1 <= p + offset - 1 < length(sequence)]
325 emit_list!(tool, enzyme, positions)
326 union!(cuts, positions)
327 end
328 endpoints = vcat([0], sort(collect(cuts)), [length(sequence)])
329 fragments = ["$(a+1)-$b" for (a, b) in zip(endpoints[1:end-1], endpoints[2:end])]
330 sizes = sort([b - a for (a, b) in zip(endpoints[1:end-1], endpoints[2:end])]; rev=true)
331 emit_list!(tool, "fragments", fragments)
332 emit_list!(tool, "sizes", sizes)
333end

335function repeats!(tool::Toolkit, sequence::String, minimum::Int, maximum::Int, count::Int)
336 minimum <= maximum || fail("invalid range")
337 rows = Tuple{Int, Int, Int, String}[]
338 for width in minimum:maximum
339 position = 1
340 while position + width - 1 <= length(sequence)
341 unit = sequence[position:position+width-1]
342 repeated = repeat_count(sequence, position, width)
343 if primitive_motif(unit) && repeated >= count
344 push!(rows, (position, width, repeated, unit))
345 position += width * repeated
346 else
347 position += 1
348 end
349 end
350 end
351 isempty(rows) && emit!(tool, "none")
352 for (position, width, repeated, unit) in sort(rows)
353 emit!(tool, "$position-$(position + width * repeated - 1)", "$(unit)x$repeated")
354 end
355end

357function align!(tool::Toolkit, a::String, b::String, mode::String,
358 match_score::Int, mismatch::Int, gap::Int)
359 m, n = length(a), length(b)
360 local_alignment = mode == "LOCAL"
361 table = zeros(Int, m + 1, n + 1)
362 if !local_alignment
363 for i in 0:m
364 table[i+1, 1] = i * gap
365 end
366 for j in 0:n
367 table[1, j+1] = j * gap
368 end
369 end
370 best, end_i, end_j = 0, 0, 0
371 for i in 1:m, j in 1:n
372 diagonal = table[i, j] + (a[i] == b[j] ? match_score : mismatch)
373 score = max(diagonal, table[i, j+1] + gap, table[i+1, j] + gap)
374 table[i+1, j+1] = local_alignment ? max(0, score) : score
375 if local_alignment && table[i+1, j+1] > best
376 best, end_i, end_j = table[i+1, j+1], i, j
377 end
378 end
379 if !local_alignment
380 best, end_i, end_j = table[m+1, n+1], m, n
381 end
382 if local_alignment && best == 0
383 emit!(tool, "score", 0)
384 return
385 end
386 i, j = end_i, end_j
387 row_a, row_b = Char[], Char[]
388 while (local_alignment && table[i+1, j+1] != 0) ||
389 (!local_alignment && (i != 0 || j != 0))
390 if i > 0 && j > 0 && table[i+1, j+1] ==
391 table[i, j] + (a[i] == b[j] ? match_score : mismatch)
392 push!(row_a, a[i])
393 push!(row_b, b[j])
394 i -= 1
395 j -= 1
396 elseif i > 0 && table[i+1, j+1] == table[i, j+1] + gap
397 push!(row_a, a[i])
398 push!(row_b, '-')
399 i -= 1
400 else
401 push!(row_a, '-')
402 push!(row_b, b[j])
403 j -= 1
404 end
405 end
406 reverse!(row_a)
407 reverse!(row_b)
408 marks = [x == '-' || y == '-' ? '-' : x == y ? '|' : '*'
409 for (x, y) in zip(row_a, row_b)]
410 emit!(tool, "score", best, "a", "$(i+1)-$end_i", "b", "$(j+1)-$end_j",
411 "identity", "$(count(==('|'), marks))/$(length(marks))",
412 "gaps", count(==('-'), marks))
413 emit!(tool, join(row_a))
414 emit!(tool, join(marks))
415 emit!(tool, join(row_b))
416end

418function consensus!(tool::Toolkit, sequences::Vector{String})
419 n = length(first(sequences))
420 all(length(s) == n for s in sequences) || fail("length mismatch")
421 profile = Dict(base => Int[] for base in BASES)
422 consensus, ambiguous = Char[], Char[]
423 for column in 1:n
424 counts = Dict(base => count(s -> s[column] == base, sequences) for base in BASES)
425 for base in BASES
426 push!(profile[base], counts[base])
427 end
428 largest = maximum(values(counts))
429 tied = join(base for base in BASES if counts[base] == largest)
430 push!(consensus, first(tied))
431 push!(ambiguous, SYMBOL[tied])
432 end
433 for base in BASES
434 emit!(tool, base, join(profile[base], " "))
435 end
436 emit!(tool, "consensus", join(consensus))
437 emit!(tool, "iupac", join(ambiguous))
438end

440function primer!(tool::Toolkit, sequence::String)
441 n, amount = length(sequence), gc(sequence)
442 tm = temperature(sequence)
443 clamp = gc(sequence[max(1, n-4):n])
444 longest_run, current_run = 0, 0
445 previous = '\0'
446 for c in sequence
447 current_run = c == previous ? current_run + 1 : 1
448 longest_run = max(longest_run, current_run)
449 previous = c
450 end
451 repeated = 0
452 for i in 1:(n-1)
453 if sequence[i] != sequence[i+1]
454 repeated = max(repeated, repeat_count(sequence, i, 2))
455 end
456 end
457 selfcomp = longest_palindrome(sequence)
458 checks = [
459 ("length", string(n), 18 <= n <= 25),
460 ("gc", pct(amount, n), 40*n <= 100*amount <= 60*n),
461 ("tm", fixed(tm, 1), 520 <= tm <= 620),
462 ("clamp", string(clamp), 1 <= clamp <= 3),
463 ("run", string(longest_run), longest_run <= 4),
464 ("repeat", string(repeated), repeated <= 4),
465 ("selfcomp", string(selfcomp), selfcomp <= 6)
466 ]
467 failed = String[]
468 for (label, value, passed) in checks
469 emit!(tool, label, value, passed ? "OK" : "FAIL")
470 !passed && push!(failed, label)
471 end
472 if isempty(failed)
473 emit!(tool, "PASS")
474 else
475 emit!(tool, "FAIL", join(failed, " "))
476 end
477end

481function fold!(tool::Toolkit, sequence::String, loop::Int)
482 n = length(sequence)
483 n <= 100 || fail("too long")
484 table = zeros(Int, n, n)
485 value(i, j) = i > j ? 0 : table[i, j]
486 for width in 2:n, i in 1:(n-width+1)
487 j = i + width - 1
488 best = value(i, j-1)
489 for k in i:(j-loop-1)
490 if paired(sequence[k], sequence[j])
491 best = max(best, value(i, k-1) + 1 + value(k+1, j-1))
492 end
493 end
494 table[i, j] = best
495 end
496 structure = fill('.', n)
497 pending = [(1, n)]
498 while !isempty(pending)
499 i, j = pop!(pending)
500 if i > j || value(i, j) == 0
501 continue
502 end
503 if value(i, j) == value(i, j-1)
504 push!(pending, (i, j-1))
505 continue
506 end
507 for k in i:(j-loop-1)
508 if paired(sequence[k], sequence[j]) &&
509 value(i, k-1) + 1 + value(k+1, j-1) == value(i, j)
510 structure[k] = '('
511 structure[j] = ')'
512 push!(pending, (i, k-1))
513 push!(pending, (k+1, j-1))
514 break
515 end
516 end
517 end
518 emit!(tool, join(structure))
519 emit!(tool, "pairs", table[1, n])
520end

522function assemble!(tool::Toolkit, name::String, minimum::Int, sequences::Vector{String})
523 unique_sequences = unique(sequences)
524 fragments = [s for s in unique_sequences if
525 !any(s != t && occursin(s, t) for t in unique_sequences)]
526 while length(fragments) >= 2
527 best = 0
528 chosen = nothing
529 for i in eachindex(fragments), j in eachindex(fragments)
530 i == j && continue
531 x, y = fragments[i], fragments[j]
532 overlap = 0
533 for width in (min(length(x), length(y))-1):-1:1
534 if x[end-width+1:end] == y[1:width]
535 overlap = width
536 break
537 end
538 end
539 if chosen === nothing || overlap > best
540 best = overlap
541 chosen = (i, j)
542 end
543 end
544 best < minimum && break
545 i, j = chosen
546 merged = fragments[i] * fragments[j][best+1:end]
547 fragments = [k == i ? merged : fragments[k]
548 for k in eachindex(fragments)
549 if k == i || (k != j && !occursin(fragments[k], merged))]
550 end
551 longest = reduce((a, b) -> length(b) > length(a) ? b : a, fragments)
552 registration = store!(tool, name, longest)
553 emit!(tool, "contigs", length(fragments))
554 for fragment in fragments
555 emit!(tool, length(fragment), fragment)
556 end
557 emit!(tool, registration)
558end

560function run_command!(tool::Toolkit, command::String, a::Vector{Any})
561 if command == "SEQ"
562 emit!(tool, store!(tool, a[1], a[2]))
563 return
564 elseif command == "UNPACK"
565 codes = Int[]
566 for i in 1:2:length(a[3])
567 byte = parse(Int, a[3][i:i+1]; base=16)
568 append!(codes, (div(byte, 64), div(byte, 16) % 4,
569 div(byte, 4) % 4, byte % 4))
570 end
571 all(==(0), codes[a[2]+1:end]) || fail("invalid padding")
572 emit!(tool, store!(tool, a[1], join(BASES[c+1] for c in codes[1:a[2]])))
573 return
574 elseif command == "JOIN"
575 emit!(tool, store!(tool, a[1], join(tool.sequences[name] for name in a[2:end])))
576 return
577 elseif command == "ASSEMBLE"
578 assemble!(tool, a[1], a[2], [tool.sequences[name] for name in a[3:end]])
579 return
580 elseif command == "LIST"
581 names = sort(collect(keys(tool.sequences)))
582 isempty(names) && emit!(tool, "no sequences")
583 for name in names
584 emit!(tool, name, length(tool.sequences[name]), "nt")
585 end
586 return
587 end
588 name, sequence = a[1], tool.sequences[a[1]]
589 n = length(sequence)
590 if command == "SLICE"
591 first, last = a[2:3]
592 part = first <= last ? sequence[first:last] : revcomp(sequence[last:first])
593 emit!(tool, store!(tool, a[4], part))
594 elseif command == "MUTATE"
595 position, base = a[2:3]
596 old = sequence[position]
597 string(old) == base && fail("no change")
598 prefix = position > 1 ? sequence[1:position-1] : ""
599 suffix = position < n ? sequence[position+1:end] : ""
600 tool.sequences[name] = prefix * base * suffix
601 emit!(tool, name, "$old$position$base",
602 transition(old, first(base)) ? "transition" : "transversion")
603 elseif command == "DEL"
604 delete!(tool.sequences, name)
605 emit!(tool, "deleted", name)
606 elseif command in ("REVCOMP", "COMPLEMENT", "REVERSE", "TRANSCRIBE")
607 result = if command == "REVCOMP"
608 revcomp(sequence)
609 elseif command == "COMPLEMENT"
610 join(COMPLEMENT[base] for base in sequence)
611 elseif command == "REVERSE"
612 reverse(sequence)
613 else
614 replace(sequence, 'T' => 'U')
615 end
616 emit!(tool, result)
617 elseif command in ("TRANSLATE", "CODONS", "PROTEIN")
618 translation!(tool, command, name, a[2])
619 elseif command == "ORFS"
620 orfs!(tool, sequence, a[2])
621 elseif command == "GC"
622 emit!(tool, join(["$base=$(count(==(base), sequence))" for base in BASES], " "),
623 "GC=$(pct(gc(sequence), n))%")
624 elseif command == "GCWINDOW"
625 width, step = a[2:3]
626 width <= n || fail("window too large")
627 for i in 1:step:(n-width+1)
628 emit!(tool, "$i-$(i+width-1)",
629 "$(pct(gc(sequence[i:i+width-1]), width))%")
630 end
631 elseif command == "SKEW"
632 values = [0]
633 for base in sequence
634 push!(values, values[end] + (base == 'G') - (base == 'C'))
635 end
636 for (label, extreme) in (("min", minimum(values)), ("max", maximum(values)))
637 emit!(tool, label, extreme, "at",
638 join((i-1 for (i, v) in enumerate(values) if v == extreme), " "))
639 end
640 elseif command == "KMERS"
641 width, limit = a[2:3]
642 width <= n || fail("k too large")
643 table = Dict{String, Int}()
644 for i in 1:(n-width+1)
645 word = sequence[i:i+width-1]
646 table[word] = get(table, word, 0) + 1
647 end
648 emit!(tool, n-width+1, "windows", length(table), "distinct")
649 for word in frequency_order(table)[1:min(limit, length(table))]
650 emit!(tool, word, table[word])
651 end
652 elseif command == "FIND"
653 emit_list!(tool, "forward", occurrences(sequence, a[2]))
654 emit_list!(tool, "reverse", occurrences(sequence, revcomp(a[2])))
655 elseif command == "RESTRICT"
656 restrict!(tool, sequence, String[a[2:end]...])
657 elseif command == "PALINDROMES"
658 minimum, maximum = a[2:3]
659 minimum <= maximum || fail("invalid range")
660 found = false
661 for position in 1:n, width in minimum:min(maximum, n-position+1)
662 part = sequence[position:position+width-1]
663 if part == revcomp(part)
664 emit!(tool, position, width, part)
665 found = true
666 end
667 end
668 !found && emit!(tool, "none")
669 elseif command == "REPEATS"
670 repeats!(tool, sequence, a[2], a[3], a[4])
671 elseif command == "HAMMING"
672 other = tool.sequences[a[2]]
673 length(other) == n || fail("length mismatch")
674 distance = count(i -> sequence[i] != other[i], 1:n)
675 transitions = count(i -> transition(sequence[i], other[i]), 1:n)
676 emit!(tool, "distance", distance, "transitions", transitions,
677 "transversions", distance - transitions)
678 elseif command == "ALIGN"
679 align!(tool, sequence, tool.sequences[a[2]], a[3], a[4], a[5], a[6])
680 elseif command == "CONSENSUS"
681 consensus!(tool, String[tool.sequences[key] for key in a])
682 elseif command == "TM"
683 emit!(tool, "Tm", fixed(temperature(sequence), 1), n < 14 ? "wallace" : "gc")
684 elseif command == "MASS"
685 single = strand_mass(sequence)
686 emit!(tool, "ss", fixed(single, 3),
687 "ds", fixed(single + strand_mass(revcomp(sequence)), 3))
688 elseif command == "PRIMER"
689 primer!(tool, sequence)
690 elseif command == "FOLD"
691 fold!(tool, sequence, a[2])
692 elseif command == "PACK"
693 codes = [findfirst(==(base), BASES) - 1 for base in sequence]
694 append!(codes, fill(0, mod(-n, 4)))
695 encoded = join(@sprintf("%02X",
696 64*codes[i]+16*codes[i+1]+4*codes[i+2]+codes[i+3])
697 for i in 1:4:length(codes))
698 emit!(tool, n, encoded)
699 end
700end
701
702function execute!(tool::Toolkit, line::String)
703 fields = String.(split(line, ' '))
704 command = first(fields)
705 try
706 args = validate(tool, command, fields[2:end])
707 run_command!(tool, command, args)
708 catch error
709 if error isa CommandError
710 emit!(tool, "ERROR", error.message)
711 else
712 rethrow()
713 end
714 end
715end
716
717function main()
718 count = parse(Int, readline(stdin))
719 toolkit = Toolkit()
720 for _ in 1:count
721 execute!(toolkit, readline(stdin))
722 end
723 println(join(toolkit.output, "\n"))
724end
