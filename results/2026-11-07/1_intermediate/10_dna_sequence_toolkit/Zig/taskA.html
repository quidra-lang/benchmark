12fn find(xs: []const []const u8, s: []const u8) ?usize {
13 for (xs, 0..) |x, i| {
14 if (std.mem.eql(u8, x, s)) return i;
15 }
16 return null;
17}
18fn baseIndex(c: u8) usize {
19 return std.mem.findScalar(u8, bases, c).?;
20}
21fn gc(s: []const u8) i64 {
22 return @intCast(std.mem.countScalar(u8, s, 'G') + std.mem.countScalar(u8, s, 'C'));
23}
24fn transition(a: u8, b: u8) bool {
25 return a != b and ((a == 'A' or a == 'G') == (b == 'A' or b == 'G'));
26}
27fn complementBase(c: u8) u8 {
28 const mask = std.mem.findScalar(u8, symbols, c).?;
29 const comp = ((mask & 1) << 3) | ((mask & 2) << 1) | ((mask & 4) >> 1) | ((mask & 8) >> 3);
30 return symbols[comp];
31}
32fn aa(codon: []const u8) u8 {
33 const code = "FFLLSSSSYY**CC*WLLLLPPPPHHQQRRRRIIIMTTTTNNKKSSRRVVVVAAAADDEEGGGG";
34 const alphabet = "TCAG";
35 return code[16 * std.mem.findScalar(u8, alphabet, codon[0]).? + 4 * std.mem.findScalar(u8, alphabet, codon[1]).? + std.mem.findScalar(u8, alphabet, codon[2]).?];
36}
37fn repeatCount(s: []const u8, i: usize, u: usize) usize {
38 var count: usize = 1;
39 while (i + (count + 1) * u <= s.len and std.mem.eql(u8, s[i .. i + u], s[i + count * u .. i + (count + 1) * u])) count += 1;
40 return count;
41}
42fn palindrome(s: []const u8) bool {
43 if (s.len % 2 != 0) return false;
44 for (s[0 .. s.len / 2], 0..) |c, i| {
45 if (baseIndex(c) + baseIndex(s[s.len - i - 1]) != 3) return false;
46 }
47 return true;
48}
49fn tm(s: []const u8) i64 {
50 const n: i64 = @intCast(s.len);
51 const count = gc(s);
52 return if (n < 14) 10 * (2 * (n - count) + 4 * count) else @divTrunc(649 * n + 410 * count - 6724, n);
53}
54fn mass(s: []const u8) i64 {
55 const weights = [_]i64{ 331222, 307197, 347221, 322208 };
56 var result: i64 = -@as(i64, @intCast(s.len - 1)) * 18015;
57 for (s) |c| result += weights[baseIndex(c)];
58 return result;
59}

65 return a.count > b.count or (a.count == b.count and std.mem.order(u8, a.text, b.text) == .lt);

111 fn numeric(self: *Dna, s: []const u8, min: i64, max: i64, frame: bool) Error!i64 {
112 const first: usize = if (s.len > 0 and s[0] == '-') 1 else 0;
113 if (s.len <= first or s.len - first > 6) return self.fail("invalid number {s}", .{s});
114 for (s[first..]) |c| {
115 if (!std.ascii.isDigit(c)) return self.fail("invalid number {s}", .{s});
116 }
117 const n = std.fmt.parseInt(i64, s, 10) catch return self.fail("invalid number {s}", .{s});
118 if (n < min or n > max or (frame and n == 0)) return self.fail("invalid number {s}", .{s});
119 return n;
120 }
121 fn validate(self: *Dna, op: Op, a: []const []const u8) Error![10]i64 {
122 var schema: [10]Param = @splat(.{});
123 var min: usize = 1;
124 var max: usize = 1;
125 switch (op) {
126 .SEQ => {
127 min = 2;
128 max = 2;
129 schema[0].kind = .new;
130 schema[1].kind = .sequence;
131 },
132 .SLICE => {
133 min = 4;
134 max = 4;
135 schema[1].kind = .position;
136 schema[2].kind = .position;
137 schema[3].kind = .new;
138 },
139 .JOIN => {
140 min = 3;
141 max = 9;
142 schema[0].kind = .new;
143 },
144 .MUTATE => {
145 min = 3;
146 max = 3;
147 schema[1].kind = .position;
148 schema[2].kind = .base;
149 },
150 .LIST => {
151 min = 0;
152 max = 0;
153 },
154 .TRANSLATE, .CODONS, .PROTEIN => {
155 min = 2;
156 max = 2;
157 schema[1].kind = .frame;
158 },
159 .ORFS => {
160 min = 2;
161 max = 2;
162 schema[1] = numberParam(1, 200);
163 },
164 .GCWINDOW => {
165 min = 3;
166 max = 3;
167 schema[1] = numberParam(1, 500);
168 schema[2] = numberParam(1, 500);
169 },
170 .KMERS => {
171 min = 3;
172 max = 3;
173 schema[1] = numberParam(1, 12);
174 schema[2] = numberParam(1, 100);
175 },
176 .FIND => {
177 min = 2;
178 max = 2;
179 schema[1].kind = .motif;
180 },
181 .RESTRICT => {
182 min = 2;
183 max = 7;
184 for (1..7) |i| schema[i].kind = .enzyme;
185 },
186 .PALINDROMES => {
187 min = 3;
188 max = 3;
189 schema[1] = numberParam(2, 20);
190 schema[2] = numberParam(2, 20);
191 },
192 .REPEATS => {
193 min = 4;
194 max = 4;
195 schema[1] = numberParam(1, 6);
196 schema[2] = numberParam(1, 6);
197 schema[3] = numberParam(2, 50);
198 },
199 .HAMMING => {
200 min = 2;
201 max = 2;
202 },
203 .ALIGN => {
204 min = 6;
205 max = 6;
206 schema[2].kind = .mode;
207 schema[3] = numberParam(1, 20);
208 schema[4] = numberParam(-20, 0);
209 schema[5] = numberParam(-20, -1);
210 },
211 .CONSENSUS => {
212 min = 2;
213 max = 10;
214 },
215 .FOLD => {
216 min = 2;
217 max = 2;
218 schema[1] = numberParam(0, 10);
219 },
220 .ASSEMBLE => {
221 min = 4;
222 max = 10;
223 schema[0].kind = .new;
224 schema[1] = numberParam(1, 100);
225 },
226 .UNPACK => {
227 min = 3;
228 max = 3;
229 schema[0].kind = .new;
230 schema[1] = numberParam(1, 500);
231 schema[2].kind = .hex;
232 },
233 else => {},
234 }
235 if (a.len < min or a.len > max) return self.fail("wrong argument count", .{});
236 var values: [10]i64 = @splat(0);
237 for (a, 0..) |s, i| {
238 const p = schema[i];
239 switch (p.kind) {
240 .name => {
241 if (!self.registry.contains(s)) return self.fail("unknown sequence {s}", .{s});
242 },
243 .new => {
244 if (s.len < 1 or s.len > 10 or s[0] < 'a' or s[0] > 'z') return self.fail("invalid name {s}", .{s});
245 for (s[1..]) |c| {
246 if (!(std.ascii.isLower(c) or std.ascii.isDigit(c))) return self.fail("invalid name {s}", .{s});
247 }
248 },
249 .sequence => {
250 for (s, 0..) |c, j| {
251 if (std.mem.findScalar(u8, bases, c) == null) return self.fail("invalid base {c} at {d}", .{ c, j + 1 });
252 }
253 },
254 .base => {
255 if (s.len != 1 or std.mem.findScalar(u8, bases, s[0]) == null) return self.fail("invalid base {s}", .{s});
256 },
257 .position => values[i] = try self.numeric(s, 1, @intCast(self.registry.get(a[0]).?.len), false),
258 .frame => values[i] = try self.numeric(s, -3, 3, true),
259 .number => values[i] = try self.numeric(s, p.min, p.max, false),
260 .motif => {
261 if (s.len < 1 or s.len > 30) return self.fail("invalid motif {s}", .{s});
262 for (s) |c| {
263 if (c == '_' or std.mem.findScalar(u8, symbols, c) == null) return self.fail("invalid motif {s}", .{s});
264 }
265 },
266 .enzyme => {
267 if (find(&enzyme_names, s) == null) return self.fail("unknown enzyme {s}", .{s});
268 for (a[1..i]) |previous| {
269 if (std.mem.eql(u8, previous, s)) return self.fail("duplicate enzyme {s}", .{s});
270 }
271 },
272 .mode => {
273 if (!std.mem.eql(u8, s, "GLOBAL") and !std.mem.eql(u8, s, "LOCAL")) return self.fail("invalid mode {s}", .{s});
274 },
275 .hex => {
276 if (s.len != 2 * @as(usize, @intCast(@divTrunc(values[1] + 3, 4)))) return self.fail("invalid hex", .{});
277 for (s) |c| {
278 if (!(std.ascii.isDigit(c) or (c >= 'A' and c <= 'F'))) return self.fail("invalid hex", .{});
279 }
280 },
281 }
282 }
283 return values;
284 }
285 fn store(self: *Dna, name: []const u8, s: []const u8) Error!void {
286 if (s.len > 500) return self.fail("too long", .{});
287 const old = self.registry.contains(name);
288 try self.registry.put(self.gpa, name, s);
289 try self.out.print("{s} {s} {d} nt\n", .{ if (old) "replaced" else "stored", name, s.len });
290 }
291 fn codons(self: *Dna, s: []const u8, frame: i64) Error![][]const u8 {
292 const strand = if (frame < 0) try self.reverse(s, true) else s;
293 var cs: std.ArrayList([]const u8) = .empty;
294 var i: usize = @intCast(@abs(frame) - 1);
295 while (i + 2 < strand.len) : (i += 3) try cs.append(self.gpa, strand[i .. i + 3]);
296 return cs.items;
297 }
298 fn translate(self: *Dna, cs: []const []const u8) Allocator.Error![]const u8 {
299 const result = try self.gpa.alloc(u8, cs.len);
300 for (cs, 0..) |c, i| result[i] = aa(c);
301 return result;
302 }
303 fn counted(self: *Dna, items: []const []const u8) Error![]Count {
304 var map: std.StringHashMapUnmanaged(i64) = .empty;
305 for (items) |item| try map.put(self.gpa, item, (map.get(item) orelse 0) + 1);
306 var result: std.ArrayList(Count) = .empty;
307 var it = map.iterator();
308 while (it.next()) |entry| try result.append(self.gpa, .{ .text = entry.key_ptr.*, .count = entry.value_ptr.* });
309 std.mem.sort(Count, result.items, {}, Count.less);
310 return result.items;
311 }
312 fn occurrences(self: *Dna, s: []const u8, motif: []const u8) Error![]usize {
313 var result: std.ArrayList(usize) = .empty;
314 var i: usize = 0;
315 while (i + motif.len <= s.len) : (i += 1) {
316 var matched = true;
317 for (motif, 0..) |c, j| {
318 const mask = std.mem.findScalar(u8, symbols, c).?;
319 if (mask & (@as(usize, 1) << @intCast(baseIndex(s[i + j]))) == 0) {
320 matched = false;
321 break;
322 }
323 }
324 if (matched) try result.append(self.gpa, i + 1);
325 }
326 return result.items;
327 }
328 fn alignSequences(self: *Dna, a: []const u8, b: []const u8, local: bool, match_score: i64, mismatch: i64, gap: i64) Error!void {
329 const m = a.len;
330 const n = b.len;
331 const width = n + 1;
332 const dp = try self.gpa.alloc(i64, (m + 1) * width);
333 @memset(dp, 0);
334 if (!local) {
335 for (0..m + 1) |i| dp[i * width] = @as(i64, @intCast(i)) * gap;
336 for (0..n + 1) |j| dp[j] = @as(i64, @intCast(j)) * gap;
337 }
338 var score: i64 = 0;
339 var i1: usize = 0;
340 var j1: usize = 0;
341 for (1..m + 1) |i| {
342 for (1..n + 1) |j| {
343 const value = @max(dp[(i - 1) * width + j - 1] + (if (a[i - 1] == b[j - 1]) match_score else mismatch), dp[(i - 1) * width + j] + gap, dp[i * width + j - 1] + gap);
344 dp[i * width + j] = if (local) @max(0, value) else value;
345 if (local and dp[i * width + j] > score) {
346 score = dp[i * width + j];
347 i1 = i;
348 j1 = j;
349 }
350 }
351 }
352 if (!local) {
353 score = dp[m * width + n];
354 i1 = m;
355 j1 = n;
356 }
357 if (local and score == 0) {
358 try self.out.writeAll("score 0\n");
359 return;
360 }
361 var i = i1;
362 var j = j1;
363 var top: std.ArrayList(u8) = .empty;
364 var bottom: std.ArrayList(u8) = .empty;
365 while (if (local) dp[i * width + j] != 0 else i != 0 or j != 0) {
366 if (i > 0 and j > 0 and dp[i * width + j] == dp[(i - 1) * width + j - 1] + (if (a[i - 1] == b[j - 1]) match_score else mismatch)) {
367 i -= 1;
368 j -= 1;
369 try top.append(self.gpa, a[i]);
370 try bottom.append(self.gpa, b[j]);
371 } else if (i > 0 and dp[i * width + j] == dp[(i - 1) * width + j] + gap) {
372 i -= 1;
373 try top.append(self.gpa, a[i]);
374 try bottom.append(self.gpa, '-');
375 } else {
376 j -= 1;
377 try top.append(self.gpa, '-');
378 try bottom.append(self.gpa, b[j]);
379 }
380 }
381 std.mem.reverse(u8, top.items);
382 std.mem.reverse(u8, bottom.items);
383 const middle = try self.gpa.alloc(u8, top.items.len);
384 for (top.items, bottom.items, 0..) |x, y, k| {
385 middle[k] = if (x == '-' or y == '-') '-' else if (x == y) '|' else '*';
386 }
387 try self.out.print("score {d} a {d}-{d} b {d}-{d} identity {d}/{d} gaps {d}\n{s}\n{s}\n{s}\n", .{ score, i + 1, i1, j + 1, j1, std.mem.countScalar(u8, middle, '|'), top.items.len, std.mem.countScalar(u8, middle, '-'), top.items, middle, bottom.items });
388 }
389 fn fold(self: *Dna, s: []const u8, loop: usize) Error!void {
390 const n = s.len;
391 const dp = try self.gpa.create([100][100]i64);
392 dp.* = @splat(@splat(0));
393 for (2..n + 1) |length| {
394 for (0..n - length + 1) |i| {
395 const j = i + length - 1;
396 dp[i][j] = dp[i][j - 1];
397 var k = i;
398 while (k + loop < j) : (k += 1) {
399 if (canPair(s[k], s[j])) dp[i][j] = @max(dp[i][j], foldValue(dp, @intCast(i), @as(isize, @intCast(k)) - 1) + 1 + foldValue(dp, @intCast(k + 1), @as(isize, @intCast(j)) - 1));
400 }
401 }
402 }
403 const structure = try self.gpa.alloc(u8, n);
404 @memset(structure, '.');
405 var stack: std.ArrayList([2]isize) = .empty;
406 try stack.append(self.gpa, .{ 0, @intCast(n - 1) });
407 while (stack.pop()) |interval| {
408 const i = interval[0];
409 const j = interval[1];
410 const value = foldValue(dp, i, j);
411 if (value == 0) continue;
412 if (value == foldValue(dp, i, j - 1)) {
413 try stack.append(self.gpa, .{ i, j - 1 });
414 continue;
415 }
416 var k = i;
417 while (k + @as(isize, @intCast(loop)) < j) : (k += 1) {
418 if (canPair(s[@intCast(k)], s[@intCast(j)]) and foldValue(dp, i, k - 1) + 1 + foldValue(dp, k + 1, j - 1) == value) {
419 structure[@intCast(k)] = '(';
420 structure[@intCast(j)] = ')';
421 try stack.append(self.gpa, .{ i, k - 1 });
422 try stack.append(self.gpa, .{ k + 1, j - 1 });
423 break;
424 }
425 }
426 }
427 try self.out.print("{s}\npairs {d}\n", .{ structure, dp[0][n - 1] });
428 }
429 fn execute(self: *Dna, tokens: []const []const u8) Error!void {
430 const op = std.meta.stringToEnum(Op, tokens[0]) orelse return self.fail("unknown command {s}", .{tokens[0]});
431 const a = tokens[1..];
432 const v = try self.validate(op, a);
433 const name: []const u8 = if (a.len == 0) "" else a[0];
434 const s = self.registry.get(name) orelse "";
435 const n = s.len;
436 switch (op) {
437 .SEQ => try self.store(name, a[1]),
438 .SLICE => {
439 const start: usize = @intCast(v[1]);
440 const end: usize = @intCast(v[2]);
441 try self.store(a[3], if (start <= end) s[start - 1 .. end] else try self.reverse(s[end - 1 .. start], true));
442 },
443 .JOIN => {
444 var parts: [8][]const u8 = undefined;
445 for (a[1..], 0..) |id, i| parts[i] = self.registry.get(id).?;
446 try self.store(name, try std.mem.concat(self.gpa, u8, parts[0 .. a.len - 1]));
447 },
448 .MUTATE => {
449 const p: usize = @intCast(v[1] - 1);
450 const old = s[p];
451 if (old == a[2][0]) return self.fail("no change", .{});
452 const result = try self.gpa.dupe(u8, s);
453 result[p] = a[2][0];
454 try self.registry.put(self.gpa, name, result);
455 try self.out.print("{s} {c}{d}{c} {s}\n", .{ name, old, p + 1, a[2][0], if (transition(old, a[2][0])) "transition" else "transversion" });
456 },
457 .DEL => {
458 _ = self.registry.swapRemove(name);
459 try self.out.print("deleted {s}\n", .{name});
460 },
461 .LIST => {
462 const entries = try self.gpa.dupe([]const u8, self.registry.keys());
463 std.mem.sort([]const u8, entries, {}, stringLess);
464 if (entries.len == 0) try self.out.writeAll("no sequences\n");
465 for (entries) |id| try self.out.print("{s} {d} nt\n", .{ id, self.registry.get(id).?.len });
466 },
467 .REVCOMP, .REVERSE => try self.out.print("{s}\n", .{try self.reverse(s, op == .REVCOMP)}),
468 .COMPLEMENT, .TRANSCRIBE => {
469 const result = try self.gpa.dupe(u8, s);
470 for (result) |*c| c.* = if (op == .COMPLEMENT) complementBase(c.*) else if (c.* == 'T') 'U' else c.*;
471 try self.out.print("{s}\n", .{result});
472 },
473 .TRANSLATE, .CODONS, .PROTEIN => {
474 const cs = try self.codons(s, v[1]);
475 const translation = try self.translate(cs);
476 if (op == .TRANSLATE) {
477 try self.out.print("{s}\n", .{if (translation.len == 0) "none" else translation});
478 } else if (op == .CODONS) {
479 const freq = try self.counted(cs);
480 if (freq.len == 0) try self.out.writeAll("none\n");
481 for (freq) |c| try self.out.print("{s} {c} {d}\n", .{ c.text, aa(c.text), c.count });
482 } else {
483 const stop = std.mem.findScalar(u8, translation, '*') orelse translation.len;
484 const protein = translation[0..stop];
485 if (protein.len == 0) return self.fail("empty protein", .{});
486 var weight: i64 = 18015;
487 var charge: i64 = 0;
488 var letters: std.ArrayList([]const u8) = .empty;
489 for (protein, 0..) |c, i| {
490 weight += residue_masses[std.mem.findScalar(u8, amino_acids, c).?];
491 if (c == 'K' or c == 'R') charge += 1;
492 if (c == 'D' or c == 'E') charge -= 1;
493 try letters.append(self.gpa, protein[i .. i + 1]);
494 }
495 try self.out.print("{s}\nlength {d} mass {s} charge {s}{d}\n", .{ protein, protein.len, try self.milli(weight), if (charge > 0) "+" else "", charge });
496 for (try self.counted(letters.items), 0..) |c, i| {
497 if (i > 0) try self.out.writeByte(' ');

829pub fn main(init: std.process.Init) !void {
830 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
831 defer arena.deinit();
832 const gpa = arena.allocator();
833 var ib: [4096]u8 = undefined;
834 var input = std.Io.File.stdin().readerStreaming(init.io, &ib);
835 var ob: [8192]u8 = undefined;
836 var output = std.Io.File.stdout().writerStreaming(init.io, &ob);
837 var dna = Dna{ .gpa = gpa, .out = &output.interface };
838 const n = std.fmt.parseInt(usize, (try input.interface.takeDelimiter('\n')).?, 10) catch unreachable;
839 for (0..n) |_| {
840 const line = try gpa.dupe(u8, (try input.interface.takeDelimiter('\n')).?);
841 var tokens: std.ArrayList([]const u8) = .empty;
842 var it = std.mem.splitScalar(u8, line, ' ');
843 while (it.next()) |part| try tokens.append(gpa, part);
844 var draft: std.Io.Writer.Allocating = .init(gpa);
845 dna.out = &draft.writer;
846 dna.execute(tokens.items) catch |err| {
847 if (err == error.BadCommand) {
848 try output.interface.print("ERROR {s}\n", .{dna.error_message});
849 continue;
850 }
851 return err;
852 };
853 try output.interface.writeAll(draft.writer.buffered());
854 }
855 try output.interface.flush();
