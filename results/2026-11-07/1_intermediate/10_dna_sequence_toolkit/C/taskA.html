113 const char *original = "ACGTRYWSKMBVDHN";

120 size_t n = strlen(sequence), i;
121 for (i = 0; i < n; ++i) result[i] = complement(sequence[n - i - 1]);

172 size_t i;
173 for (i = 0; i < sizeof(codon_groups) / sizeof(*codon_groups); ++i) {
174 const char *list = codon_groups[i].triplets;

180 }

222static const char *argument_kind(const char *command, int i) {
223 if (strcmp(command, "SEQ") == 0) return i == 0 ? "new" : "sequence";
224 if (strcmp(command, "SLICE") == 0)
225 return i == 0 ? "name" : i == 3 ? "new" : "position";
226 if (strcmp(command, "JOIN") == 0)
227 return i == 0 ? "new" : "name";
228 if (strcmp(command, "MUTATE") == 0)
229 return i == 0 ? "name" : i == 1 ? "position" : "base";
230 if (strcmp(command, "ASSEMBLE") == 0)
231 return i == 0 ? "new" : i == 1 ? "overlap" : "name";
232 if (strcmp(command, "UNPACK") == 0)
233 return i == 0 ? "new" : i == 1 ? "length" : "hex";
234 if (strcmp(command, "RESTRICT") == 0)
235 return i == 0 ? "name" : "enzyme";
236 if (strcmp(command, "ALIGN") == 0)
237 return i < 2 ? "name" : i == 2 ? "mode" :
238 i == 3 ? "match" : i == 4 ? "mismatch" : "gap";
239 if (i == 0) return "name";
240 if (strcmp(command, "TRANSLATE") == 0 ||
241 strcmp(command, "CODONS") == 0 || strcmp(command, "PROTEIN") == 0)
242 return "frame";
243 if (strcmp(command, "ORFS") == 0) return "orf";
244 if (strcmp(command, "GCWINDOW") == 0) return "window";
245 if (strcmp(command, "KMERS") == 0) return i == 1 ? "kmer" : "limit";
246 if (strcmp(command, "FIND") == 0) return "motif";
247 if (strcmp(command, "PALINDROMES") == 0) return "palindrome";
248 if (strcmp(command, "REPEATS") == 0)
249 return i == 3 ? "repeat_count" : "repeat";
250 if (strcmp(command, "FOLD") == 0) return "loop";
251 return "name";
252}

254static int valid_number(const char *text, int minimum, int maximum, int nonzero) {
255 const char *digits = *text == '-' ? text + 1 : text;
256 int value;
257 const char *p;
258 if (strlen(digits) == 0 || strlen(digits) > 6) {
259 fail("invalid number %s", text);
260 return 0;
261 }
262 for (p = digits; *p; ++p) {
263 if (!isdigit((unsigned char)*p)) {
264 fail("invalid number %s", text);
265 return 0;
266 }
267 }
268 value = atoi(text);
269 if (value < minimum || value > maximum || (nonzero && value == 0)) {
270 fail("invalid number %s", text);
271 return 0;
272 }
273 return 1;
274}

276static int validate(const char *command, int count, char **args) {
277 size_t k;
278 int i, j, minimum, maximum, nonzero, value;
279 const Arity *arity = NULL;
280 for (k = 0; k < sizeof(arities) / sizeof(*arities); ++k)
281 if (strcmp(arities[k].command, command) == 0) {
282 arity = &arities[k];
283 break;
284 }
285 if (!arity) {
286 fail("unknown command %s", command);
287 return 0;
288 }
289 if (count < arity->minimum || count > arity->maximum) {
290 fail("wrong argument count");
291 return 0;
292 }
293 for (i = 0; i < count; ++i) {
294 const char *kind = argument_kind(command, i);
295 const char *text = args[i];
296 size_t len = strlen(text);
297 if (is_numeric_kind(kind)) {
298 minimum = 1;
299 maximum = 500;
300 nonzero = 0;
301 if (strcmp(kind, "position") == 0) maximum = (int)strlen(sequence_of(args[0]));
302 else if (strcmp(kind, "frame") == 0) minimum = -3, maximum = 3, nonzero = 1;
303 else if (strcmp(kind, "orf") == 0) maximum = 200;
304 else if (strcmp(kind, "kmer") == 0) maximum = 12;
305 else if (strcmp(kind, "limit") == 0 ||
306 strcmp(kind, "overlap") == 0) maximum = 100;
307 else if (strcmp(kind, "palindrome") == 0) minimum = 2, maximum = 20;
308 else if (strcmp(kind, "repeat") == 0) maximum = 6;
309 else if (strcmp(kind, "repeat_count") == 0) minimum = 2, maximum = 50;
310 else if (strcmp(kind, "match") == 0) maximum = 20;
311 else if (strcmp(kind, "mismatch") == 0) minimum = -20, maximum = 0;
312 else if (strcmp(kind, "gap") == 0) minimum = -20, maximum = -1;
313 else if (strcmp(kind, "loop") == 0) minimum = 0, maximum = 10;
314 if (!valid_number(text, minimum, maximum, nonzero)) return 0;
315 } else if (strcmp(kind, "name") == 0) {
316 if (exists(text) < 0) {
317 fail("unknown sequence %s", text);
318 return 0;
319 }
320 } else if (strcmp(kind, "new") == 0) {
321 if (!len || len > 10 || text[0] < 'a' || text[0] > 'z') {
322 fail("invalid name %s", text);
323 return 0;
324 }
325 for (j = 0; j < (int)len; ++j) {
326 if (!((text[j] >= 'a' && text[j] <= 'z') ||
327 (text[j] >= '0' && text[j] <= '9'))) {
328 fail("invalid name %s", text);
329 return 0;
330 }
331 }
332 } else if (strcmp(kind, "sequence") == 0) {
333 for (j = 0; j < (int)len; ++j) {
334 if (!strchr("ACGT", text[j])) {
335 fail("invalid base %c at %d", text[j], j + 1);
336 return 0;
337 }
338 }
339 } else if (strcmp(kind, "base") == 0) {
340 if (len != 1 || !strchr("ACGT", text[0])) {
341 fail("invalid base %s", text);
342 return 0;
343 }
344 } else if (strcmp(kind, "motif") == 0) {
345 if (!len || len > 30) {
346 fail("invalid motif %s", text);
347 return 0;
348 }
349 for (j = 0; j < (int)len; ++j)
350 if (!iupac(text[j])) {
351 fail("invalid motif %s", text);
352 return 0;
353 }
354 } else if (strcmp(kind, "enzyme") == 0) {
355 if (!find_enzyme(text)) {
356 fail("unknown enzyme %s", text);
357 return 0;
358 }
359 for (j = 1; j < i; ++j)
360 if (strcmp(args[j], text) == 0) {
361 fail("duplicate enzyme %s", text);
362 return 0;
363 }
364 } else if (strcmp(kind, "mode") == 0) {
365 if (strcmp(text, "GLOBAL") != 0 && strcmp(text, "LOCAL") != 0) {
366 fail("invalid mode %s", text);
367 return 0;
368 }
369 } else if (strcmp(kind, "hex") == 0) {
370 value = atoi(args[1]);
371 if (len != (size_t)(2 * ((value + 3) / 4))) {
372 fail("invalid hex");
373 return 0;
374 }
375 for (j = 0; j < (int)len; ++j) {
376 if (!isdigit((unsigned char)text[j]) ||
377 (text[j] < '0' || text[j] > '9')) {
378 if (text[j] < 'A' || text[j] > 'F') {
379 fail("invalid hex");
380 return 0;
381 }
382 }
383 }
384 }
385 }
386 return 1;
387}

389static int get_codons(const char *sequence, int frame, char triplets[167][4]) {
390 char strand[MAX_LENGTH + 1];
391 int offset, n, i, count = 0;
392 if (frame < 0) {
393 revcomp(sequence, strand);
394 frame = -frame;
395 } else strcpy(strand, sequence);
396 offset = frame - 1;
397 n = (int)strlen(strand);
398 for (i = offset; i + 2 < n; i += 3) {
399 memcpy(triplets[count], strand + i, 3);
400 triplets[count][3] = '\0';
401 ++count;
402 }
403 return count;
404}

406static int occurrences(const char *sequence, const char *motif, int *locations) {
407 int n = (int)strlen(sequence), m = (int)strlen(motif);
408 int i, j, count = 0, matched;
409 for (i = 0; i + m <= n; ++i) {
410 matched = 1;
411 for (j = 0; j < m; ++j)
412 if (!strchr(iupac(motif[j]), sequence[i + j])) {
413 matched = 0;
414 break;
415 }
416 if (matched) locations[count++] = i + 1;
417 }
418 return count;
419}

430static int primitive(const char *unit, int n) {
431 int width, i, repeated;
432 for (width = 1; width < n; ++width) {
433 if (n % width) continue;
434 repeated = 1;
435 for (i = width; i < n; ++i)
436 if (unit[i] != unit[i % width]) {
437 repeated = 0;
438 break;
439 }
440 if (repeated) return 0;
441 }
442 return 1;
443}

445static int palindrome_length(const char *sequence) {
446 int n = (int)strlen(sequence), best = 0, center, left, right;
447 for (center = 1; center < n; ++center) {
448 left = center - 1;
449 right = center;
450 while (left >= 0 && right < n &&
451 complement(sequence[left]) == sequence[right]) {
452 if (right - left + 1 > best) best = right - left + 1;
453 --left;
454 ++right;
455 }
456 }
457 return best;
458}

497static int compare_orfs(const void *left, const void *right) {
498 const Orf *a = left, *b = right;
499 if (a->left != b->left) return (a->left > b->left) - (a->left < b->left);
500 if (a->right != b->right) return (a->right > b->right) - (a->right < b->right);
501 return (a->frame > b->frame) - (a->frame < b->frame);
502}

504static void orfs(const char *sequence, int minimum) {
505 Orf result[600];
506 int found = 0, frames[] = {1, 2, 3, -1, -2, -3};
507 int i, j, position, first, n = (int)strlen(sequence);
508 char codon[167][4];
509 for (i = 0; i < 6; ++i) {
510 int frame = frames[i], offset = abs(frame) - 1;
511 int count = get_codons(sequence, frame, codon);
512 first = -1;
513 for (j = 0; j < count; ++j) {
514 if (first < 0) {
515 if (strcmp(codon[j], "ATG") == 0) first = j;
516 } else if (amino(codon[j]) == '*') {
517 if (j - first >= minimum) {
518 Orf *row = &result[found++];
519 int left = offset + 3 * first;
520 int right = offset + 3 * (j + 1);
521 if (frame < 0) {
522 position = left;
523 left = n - right;
524 right = n - position;
525 }
526 row->left = left + 1;
527 row->right = right;
528 row->frame = frame;
529 for (position = first; position < j; ++position)
530 row->protein[position - first] = amino(codon[position]);
531 row->protein[j - first] = '\0';
532 }
533 first = -1;
534 }
535 }
536 }
537 qsort(result, (size_t)found, sizeof(*result), compare_orfs);
538 if (found == 0) puts("none");
539 for (i = 0; i < found; ++i)
540 printf("%+d %d %d %zu %s\n", result[i].frame, result[i].left,
541 result[i].right, strlen(result[i].protein), result[i].protein);
542}

544static void translation(const char *sequence, const char *command, int frame) {
545 char triplets[167][4], protein[168];
546 int counts[167] = {0};
547 int count = get_codons(sequence, frame, triplets), i, j;
548 if (strcmp(command, "TRANSLATE") == 0) {
549 if (!count) {
550 puts("none");
551 return;
552 }
553 for (i = 0; i < count; ++i) putchar(amino(triplets[i]));
554 putchar('\n');
555 } else if (strcmp(command, "CODONS") == 0) {
556 int order[167], distinct = 0;
557 for (i = 0; i < count; ++i) {
558 for (j = 0; j < distinct; ++j)
559 if (strcmp(triplets[order[j]], triplets[i]) == 0) break;
560 if (j == distinct) order[distinct++] = i;
561 ++counts[order[j]];
562 }
563 for (i = 0; i < distinct; ++i)
564 for (j = i + 1; j < distinct; ++j)
565 if (counts[order[j]] > counts[order[i]] ||
566 (counts[order[j]] == counts[order[i]] &&
567 strcmp(triplets[order[j]], triplets[order[i]]) < 0)) {
568 int temporary = order[i];
569 order[i] = order[j];
570 order[j] = temporary;
571 }
572 if (!distinct) puts("none");
573 for (i = 0; i < distinct; ++i)
574 printf("%s %c %d\n", triplets[order[i]], amino(triplets[order[i]]),
575 counts[order[i]]);
576 } else {
577 int composition[256] = {0}, mass = 18015, charge = 0, length = 0;
578 char text[32];
579 for (i = 0; i < count; ++i) {
580 char aa = amino(triplets[i]);
581 if (aa == '*') break;
582 protein[length++] = aa;
583 ++composition[(unsigned char)aa];
584 mass += amino_mass(aa);
585 charge += aa == 'K' || aa == 'R';
586 charge -= aa == 'D' || aa == 'E';
587 }
588 if (!length) {
589 fail("empty protein");
590 return;
591 }
592 protein[length] = '\0';
593 puts(protein);
594 fixed(mass, 3, text);
595 printf("length %d mass %s charge ", length, text);
596 if (charge > 0) putchar('+');
597 printf("%d\n", charge);
598 {
599 const char *labels = "ACDEFGHIKLMNPQRSTVWY";
600 int first = 1;
601 for (j = 0; j < 20; ++j) {
602 int best = -1;
603 for (i = 0; i < 20; ++i) {
604 char aa = labels[i];
605 if (composition[(unsigned char)aa] <= 0) continue;
606 if (best < 0 ||
607 composition[(unsigned char)aa] >
608 composition[(unsigned char)labels[best]] ||
609 (composition[(unsigned char)aa] ==
610 composition[(unsigned char)labels[best]] && i < best))
611 best = i;
612 }
613 if (best < 0) break;
614 if (!first) putchar(' ');
615 first = 0;
616 printf("%c=%d", labels[best], composition[(unsigned char)labels[best]]);
617 composition[(unsigned char)labels[best]] = 0;
618 }
619 putchar('\n');
620 }
621 }
622}

624static void restriction(const char *sequence, int count, char **selected) {
625 int cut[MAX_LENGTH + 1] = {0}, positions[500];
626 int i, j, k, found, n = (int)strlen(sequence), previous, sizes[501], fragments = 0;
627 for (i = 0; i < count; ++i) {
628 const Enzyme *enzyme = find_enzyme(selected[i]);
629 int number = occurrences(sequence, enzyme->motif, positions);
630 int printed = 0;
631 printf("%s", selected[i]);
632 for (j = 0; j < number; ++j) {
633 int location = positions[j] + enzyme->offset - 1;
634 if (location >= 1 && location < n) {
635 printf(" %d", location);
636 cut[location] = 1;
637 printed = 1;
638 }
639 }
640 if (!printed) printf(" none");
641 putchar('\n');
642 }
643 printf("fragments");
644 previous = 0;
645 for (k = 1; k <= n; ++k) {
646 if (cut[k] || k == n) {
647 printf(" %d-%d", previous + 1, k);
648 sizes[fragments++] = k - previous;
649 previous = k;
650 }
651 }

704static int alignment_table[501][501];
705
706static void alignment(const char *a, const char *b, const char *mode,
707 int match, int mismatch, int gap) {
708 int m = (int)strlen(a), n = (int)strlen(b);
709 int local = strcmp(mode, "LOCAL") == 0;
710 int best = 0, end_i = 0, end_j = 0, i, j, score, diagonal, value;
711 char left[1001], right[1001], middle[1001];
712 int length = 0, identical = 0, gaps = 0, position;
713 memset(alignment_table, 0, sizeof(alignment_table));
714 if (!local) {
715 for (i = 0; i <= m; ++i) alignment_table[i][0] = i * gap;
716 for (j = 0; j <= n; ++j) alignment_table[0][j] = j * gap;
717 }
718 for (i = 1; i <= m; ++i) {
719 for (j = 1; j <= n; ++j) {
720 score = a[i - 1] == b[j - 1] ? match : mismatch;
721 diagonal = alignment_table[i - 1][j - 1] + score;
722 value = diagonal;
723 if (alignment_table[i - 1][j] + gap > value)
724 value = alignment_table[i - 1][j] + gap;
725 if (alignment_table[i][j - 1] + gap > value)
726 value = alignment_table[i][j - 1] + gap;
727 if (local && value < 0) value = 0;
728 alignment_table[i][j] = value;
729 if (local && value > best) {
730 best = value;
731 end_i = i;
732 end_j = j;
733 }
734 }
735 }
736 if (!local) {
737 best = alignment_table[m][n];
738 end_i = m;
739 end_j = n;
740 }
741 if (local && best == 0) {
742 puts("score 0");
743 return;
744 }
745 i = end_i;
746 j = end_j;
747 while ((local && alignment_table[i][j] != 0) ||
748 (!local && (i != 0 || j != 0))) {
749 score = i > 0 && j > 0 && a[i - 1] == b[j - 1] ? match : mismatch;
750 if (i > 0 && j > 0 &&
751 alignment_table[i][j] == alignment_table[i - 1][j - 1] + score) {
752 left[length] = a[--i];
753 right[length] = b[--j];
754 } else if (i > 0 &&
755 alignment_table[i][j] == alignment_table[i - 1][j] + gap) {
756 left[length] = a[--i];
757 right[length] = '-';
758 } else {
759 left[length] = '-';
760 right[length] = b[--j];
761 }
762 ++length;
763 }
764 for (position = 0; position < length; ++position) {
765 char x = left[position], y = right[position];
766 if (x == '-' || y == '-') {
767 middle[position] = '-';
768 ++gaps;
769 } else if (x == y) {
770 middle[position] = '|';
771 ++identical;
772 } else middle[position] = '*';
773 }
774 printf("score %d a %d-%d b %d-%d identity %d/%d gaps %d\n",
775 best, i + 1, end_i, j + 1, end_j, identical, length, gaps);
776 for (position = length - 1; position >= 0; --position) putchar(left[position]);
777 putchar('\n');
778 for (position = length - 1; position >= 0; --position) putchar(middle[position]);
779 putchar('\n');
780 for (position = length - 1; position >= 0; --position) putchar(right[position]);
781 putchar('\n');
782}

784static void consensus(int count, char **names) {
785 int n = (int)strlen(sequence_of(names[0]));
786 int profile[4][501] = {{0}};
787 int i, j, k, best, mask;
788 const char *symbols[] = {
789 "", "A", "C", "M", "G", "R", "S", "V",
790 "T", "W", "Y", "H", "K", "D", "B", "N"
791 };
792 const char *bases = "ACGT";
793 char consensus_seq[501], ambiguous[501];
794 for (i = 1; i < count; ++i) {
795 if (strlen(sequence_of(names[i])) != (size_t)n) {
796 fail("length mismatch");
797 return;
798 }
799 }
800 for (j = 0; j < n; ++j) {
801 for (i = 0; i < count; ++i) {
802 const char *base = strchr(bases, sequence_of(names[i])[j]);
803 ++profile[base - bases][j];
804 }
805 best = 0;
806 for (k = 0; k < 4; ++k)
807 if (profile[k][j] > best) best = profile[k][j];
808 mask = 0;
809 for (k = 0; k < 4; ++k)
810 if (profile[k][j] == best) mask |= 1 << k;
811 for (k = 0; k < 4; ++k) {
812 if (mask & (1 << k)) {
813 consensus_seq[j] = bases[k];
814 break;
815 }
816 }
817 ambiguous[j] = symbols[mask][0];
818 }
819 consensus_seq[n] = '\0';
820 ambiguous[n] = '\0';
821 for (k = 0; k < 4; ++k) {
822 printf("%c", bases[k]);
823 for (j = 0; j < n; ++j) printf(" %d", profile[k][j]);
824 putchar('\n');
825 }
826 printf("consensus %s\niupac %s\n", consensus_seq, ambiguous);
827}

829static void primer(const char *sequence) {
830 int n = (int)strlen(sequence), amount = gc(sequence), tm = temperature(sequence);
831 int clamp = gc(sequence + (n > 5 ? n - 5 : 0));
832 int run = 0, longest = 0, repeated = 0, selfcomp;
833 int i, candidate, passed[7];
834 const char *labels[] = {"length", "gc", "tm", "clamp", "run", "repeat", "selfcomp"};
835 char percentage_text[32], temperature_text[32], values[7][32];
836 int failures = 0;
837 for (i = 0; i < n; ++i) {
838 if (i > 0 && sequence[i] == sequence[i - 1]) ++run;
839 else run = 1;
840 if (run > longest) longest = run;
841 }
842 for (i = 0; i + 1 < n; ++i) {
843 if (sequence[i] != sequence[i + 1]) {
844 candidate = repeat_count(sequence, i, 2);
845 if (candidate > repeated) repeated = candidate;
846 }
847 }
848 selfcomp = palindrome_length(sequence);
849 percentage(amount, n, percentage_text);
850 fixed(tm, 1, temperature_text);

853 strcpy(values[2], temperature_text);
854 sprintf(values[3], "%d", clamp);
855 sprintf(values[4], "%d", longest);
856 sprintf(values[5], "%d", repeated);
857 sprintf(values[6], "%d", selfcomp);
858 passed[0] = n >= 18 && n <= 25;
859 passed[1] = 40 * n <= 100 * amount && 100 * amount <= 60 * n;
860 passed[2] = tm >= 520 && tm <= 620;
861 passed[3] = clamp >= 1 && clamp <= 3;
862 passed[4] = longest <= 4;
863 passed[5] = repeated <= 4;
864 passed[6] = selfcomp <= 6;
865 for (i = 0; i < 7; ++i) {
866 printf("%s %s %s\n", labels[i], values[i], passed[i] ? "OK" : "FAIL");
867 if (!passed[i]) ++failures;
868 }
869 if (!failures) puts("PASS");
870 else {
871 printf("FAIL");
872 for (i = 0; i < 7; ++i)
873 if (!passed[i]) printf(" %s", labels[i]);
874 putchar('\n');
875 }
876}

880static int fold_value(int i, int j) {
881 return i > j ? 0 : fold_table[i][j];

885 return (a == 'A' && b == 'T') || (a == 'T' && b == 'A') ||
886 (a == 'C' && b == 'G') || (a == 'G' && b == 'C') ||

890static void fold(const char *sequence, int loop) {
891 int n = (int)strlen(sequence), width, i, j, k, best;
892 int pending_i[201], pending_j[201], pending_count = 0;
893 char structure[101];
894 if (n > 100) {
895 fail("too long");
896 return;
897 }
898 memset(fold_table, 0, sizeof(fold_table));
899 memset(structure, '.', (size_t)n);
900 structure[n] = '\0';
901 for (width = 2; width <= n; ++width) {
902 for (i = 0; i + width <= n; ++i) {
903 j = i + width - 1;
904 best = fold_value(i, j - 1);
905 for (k = i; k < j - loop; ++k)
906 if (can_pair(sequence[k], sequence[j])) {
907 int candidate = fold_value(i, k - 1) + 1 + fold_value(k + 1, j - 1);
908 if (candidate > best) best = candidate;
909 }
910 fold_table[i][j] = best;
911 }
912 }
913 pending_i[pending_count] = 0;
914 pending_j[pending_count++] = n - 1;
915 while (pending_count > 0) {
916 --pending_count;
917 i = pending_i[pending_count];
918 j = pending_j[pending_count];
919 if (i > j || fold_value(i, j) == 0) continue;
920 if (fold_value(i, j) == fold_value(i, j - 1)) {
921 pending_i[pending_count] = i;
922 pending_j[pending_count++] = j - 1;
923 continue;
924 }
925 for (k = i; k < j - loop; ++k)
926 if (can_pair(sequence[k], sequence[j]) &&
927 fold_value(i, k - 1) + 1 + fold_value(k + 1, j - 1) ==
928 fold_value(i, j)) {
929 structure[k] = '(';
930 structure[j] = ')';
931 pending_i[pending_count] = i;
932 pending_j[pending_count++] = k - 1;
933 pending_i[pending_count] = k + 1;
934 pending_j[pending_count++] = j - 1;
935 break;
936 }
937 }
938 printf("%s\npairs %d\n", structure, fold_table[0][n - 1]);
939}

1023 puts(registration);
1024 }
