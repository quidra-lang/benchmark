45const lex = (a: string, b: string): number => a < b ? -1 : a > b ? 1 : 0;
46const revcomp = (s: string): string => [...s].reverse().map(b => COMPLEMENT[b]).join("");
47const gc = (s: string): number => [...s].filter(b => b === "G" || b === "C").length;
48const fixed = (v: number, places: number): string => Math.floor(v / 10 ** places) + "." + String(v % 10 ** places).padStart(places, "0");
49const pct = (a: number, b: number): string => fixed(Math.floor(10000 * a / b), 2);
50const transition = (a: string, b: string): boolean => ["AG", "GA", "CT", "TC"].includes(a + b);
51const temperature = (s: string): number => s.length < 14 ? 10 * (2 * (s.length - gc(s)) + 4 * gc(s)) : Math.floor((649 * s.length + 410 * gc(s) - 6724) / s.length);
52const strandMass = (s: string): number => [...s].reduce((total, b) => total + BASE_MASS[b], 0) - (s.length - 1) * 18015;

54function counts(items: Iterable<string>): Map<string, number> {
55 const table = new Map<string, number>();
56 for (const item of items) table.set(item, (table.get(item) ?? 0) + 1);
57 return table;
58}

60function frequencyOrder(table: Map<string, number>): string[] {
61 return [...table.keys()].sort((a, b) => table.get(b)! - table.get(a)! || lex(a, b));
62}

64function codons(s: string, frame: number): string[] {
65 const strand = frame > 0 ? s : revcomp(s);
66 return strand.slice(Math.abs(frame) - 1).match(/.../g) ?? [];

69function occurrences(s: string, motif: string): number[] {
70 const result: number[] = [];
71 for (let i = 0; i + motif.length <= s.length; i++) {
72 if ([...motif].every((symbol, j) => IUPAC[symbol].includes(s[i + j]))) result.push(i + 1);
73 }
74 return result;
75}

77function repeatCount(s: string, start: number, width: number): number {
78 const unit = s.slice(start, start + width);
79 let count = 0;
80 while (s.slice(start + count * width, start + (count + 1) * width) === unit) count++;
81 return count;
82}

84function primitive(unit: string): boolean {
85 for (let d = 1; d < unit.length; d++) {
86 if (unit.length % d === 0 && unit.slice(0, d).repeat(unit.length / d) === unit) return false;
87 }
88 return true;
89}

91function longestPalindrome(s: string): number {
92 let best = 0;
93 for (let center = 1; center < s.length; center++) {
94 let left = center - 1;
95 let right = center;
96 while (left >= 0 && right < s.length && COMPLEMENT[s[left]] === s[right]) {
97 best = Math.max(best, right - left + 1);
98 left--;
99 right++;
100 }
101 }
102 return best;
103}

141 validate(cmd: string, a: string[]): void {
142 if (!Object.hasOwn(ARITY, cmd)) throw new CommandError("unknown command " + cmd);
143 const [minimum, maximum] = ARITY[cmd];
144 if (a.length < minimum || a.length > maximum) throw new CommandError("wrong argument count");
145 const kinds = this.kinds(cmd, a.length);
146 const seen = new Set<string>();
147 for (let i = 0; i < a.length; i++) {
148 const text = a[i];
149 const kind = kinds[i];
150 if (Array.isArray(kind)) {
151 const upper = kind[1] === "n" ? this.registry.get(a[0])!.length : kind[1];
152 const value = Number(text);
153 if (!/^-?[0-9]{1,6}$/.test(text) || value < kind[0] || value > upper || (kind[2] && value === 0)) throw new CommandError("invalid number " + text);
154 } else if (kind === "name") {
155 if (!this.registry.has(text)) throw new CommandError("unknown sequence " + text);
156 } else if (kind === "new") {
157 if (!/^[a-z][a-z0-9]{0,9}$/.test(text)) throw new CommandError("invalid name " + text);
158 } else if (kind === "sequence") {
159 for (let p = 0; p < text.length; p++) {
160 if (!BASES.includes(text[p])) throw new CommandError("invalid base " + text[p] + " at " + (p + 1));
161 }
162 } else if (kind === "base") {
163 if (text.length !== 1 || !BASES.includes(text)) throw new CommandError("invalid base " + text);
164 } else if (kind === "motif") {
165 if (text.length < 1 || text.length > 30 || [...text].some(c => !Object.hasOwn(IUPAC, c))) throw new CommandError("invalid motif " + text);
166 } else if (kind === "enzyme") {
167 if (!Object.hasOwn(ENZYMES, text)) throw new CommandError("unknown enzyme " + text);
168 if (seen.has(text)) throw new CommandError("duplicate enzyme " + text);
169 seen.add(text);
170 } else if (kind === "mode") {
171 if (!["GLOBAL", "LOCAL"].includes(text)) throw new CommandError("invalid mode " + text);
172 } else if (kind === "hex") {
173 if (!/^[0-9A-F]+$/.test(text) || text.length !== 2 * Math.ceil(Number(a[1]) / 4)) throw new CommandError("invalid hex");
174 }
175 }

185 translation(cmd: string, s: string, frame: number): void {
186 const triplets = codons(s, frame);
187 let protein = triplets.map(c => CODE[c]).join("");
188 if (cmd === "TRANSLATE") {
189 this.emit(protein || "none");
190 } else if (cmd === "CODONS") {
191 const table = counts(triplets);
192 for (const c of frequencyOrder(table)) this.emit(c, CODE[c], table.get(c)!);
193 if (!table.size) this.emit("none");
194 } else {
195 protein = protein.split("*")[0];
196 if (!protein) throw new CommandError("empty protein");
197 const table = counts(protein);
198 const mass = [...protein].reduce((v, amino) => v + AMINO_MASS[amino], 18015);
199 const charge = [...protein].reduce((v, amino) => v + Number("KR".includes(amino)) - Number("DE".includes(amino)), 0);
200 this.emit(protein);
201 this.emit("length", protein.length, "mass", fixed(mass, 3), "charge", charge > 0 ? "+" + charge : charge);
202 this.emit(...frequencyOrder(table).map(a => a + "=" + table.get(a)));
203 }

206 orfs(s: string, minimum: number): void {
207 const results: Orf[] = [];
208 for (const frame of [1, 2, 3, -1, -2, -3]) {
209 const triplets = codons(s, frame);
210 let start: number | null = null;
211 for (let p = 0; p < triplets.length; p++) {
212 if (start === null) {
213 if (triplets[p] === "ATG") start = p;
214 } else if (CODE[triplets[p]] === "*") {
215 if (p - start >= minimum) {
216 let left = Math.abs(frame) - 1 + 3 * start;
217 let right = Math.abs(frame) - 1 + 3 * (p + 1);
218 if (frame < 0) [left, right] = [s.length - right, s.length - left];
219 const protein = triplets.slice(start, p).map(c => CODE[c]).join("");
220 results.push({left: left + 1, right, frame, protein});
221 }
222 start = null;
223 }
224 }
225 }
226 results.sort((a, b) => a.left - b.left);
227 for (const r of results) this.emit(r.frame > 0 ? "+" + r.frame : r.frame, r.left, r.right, r.protein.length, r.protein);
228 if (!results.length) this.emit("none");
229 }

231 restrict(s: string, enzymes: string[]): void {
232 const combined = new Set<number>();
233 for (const enzyme of enzymes) {
234 const [motif, offset] = ENZYMES[enzyme];
235 const cuts = occurrences(s, motif).map(p => p + offset - 1).filter(c => c >= 1 && c < s.length);
236 this.emitList(enzyme, cuts);
237 for (const c of cuts) combined.add(c);
238 }
239 const endpoints = [0, ...[...combined].sort((a, b) => a - b), s.length];
240 const fragments: string[] = [];
241 const sizes: number[] = [];
242 for (let i = 1; i < endpoints.length; i++) {
243 fragments.push((endpoints[i - 1] + 1) + "-" + endpoints[i]);
244 sizes.push(endpoints[i] - endpoints[i - 1]);
245 }
246 this.emitList("fragments", fragments);
247 this.emitList("sizes", sizes.sort((a, b) => b - a));
248 }

250 repeats(s: string, minimum: number, maximum: number, limit: number): void {
251 if (minimum > maximum) throw new CommandError("invalid range");
252 const results: Repeat[] = [];
253 for (let width = minimum; width <= maximum; width++) {
254 let position = 0;
255 while (position + width <= s.length) {
256 const unit = s.slice(position, position + width);
257 const count = repeatCount(s, position, width);
258 if (primitive(unit) && count >= limit) {
259 results.push({position, width, count, unit});
260 position += count * width;
261 } else {
262 position++;
263 }
264 }
265 }
266 results.sort((a, b) => a.position - b.position || a.width - b.width);
267 for (const r of results) this.emit((r.position + 1) + "-" + (r.position + r.width * r.count), r.unit + "x" + r.count);
268 if (!results.length) this.emit("none");
269 }

271 align(a: string, b: string, mode: string, match: number, mismatch: number, gap: number): void {
272 const m = a.length;
273 const n = b.length;
274 const local = mode === "LOCAL";
275 const table: number[][] = Array.from({length: m + 1}, () => Array(n + 1).fill(0));
276 if (!local) {
277 for (let i = 0; i <= m; i++) table[i][0] = i * gap;
278 for (let j = 0; j <= n; j++) table[0][j] = j * gap;
279 }
280 let best = 0;
281 let endI = 0;
282 let endJ = 0;
283 for (let i = 1; i <= m; i++) {
284 for (let j = 1; j <= n; j++) {
285 const diagonal = table[i - 1][j - 1] + (a[i - 1] === b[j - 1] ? match : mismatch);
286 const value = Math.max(diagonal, table[i - 1][j] + gap, table[i][j - 1] + gap);
287 table[i][j] = local ? Math.max(0, value) : value;
288 if (local && table[i][j] > best) {
289 best = table[i][j];
290 endI = i;
291 endJ = j;
292 }
293 }
294 }
295 if (!local) {
296 endI = m;
297 endJ = n;
298 best = table[m][n];
299 }
300 if (local && best === 0) {
301 this.emit("score", 0);
302 return;
303 }
304 let i = endI;
305 let j = endJ;
306 const rowA: string[] = [];
307 const rowB: string[] = [];
308 while (local ? table[i][j] !== 0 : i !== 0 || j !== 0) {
309 if (i && j && table[i][j] === table[i - 1][j - 1] + (a[i - 1] === b[j - 1] ? match : mismatch)) {
310 rowA.push(a[i - 1]);
311 rowB.push(b[j - 1]);
312 i--;
313 j--;
314 } else if (i && table[i][j] === table[i - 1][j] + gap) {
315 rowA.push(a[i - 1]);
316 rowB.push("-");
317 i--;
318 } else {
319 rowA.push("-");
320 rowB.push(b[j - 1]);
321 j--;
322 }
323 }
324 rowA.reverse();
325 rowB.reverse();
326 const middle = rowA.map((a, h) => a === "-" || rowB[h] === "-" ? "-" : a === rowB[h] ? "|" : "*").join("");
327 this.emit("score", best, "a", (i + 1) + "-" + endI, "b", (j + 1) + "-" + endJ,
328 "identity", [...middle].filter(c => c === "|").length + "/" + middle.length, "gaps", [...middle].filter(c => c === "-").length);
329 this.emit(rowA.join(""));
330 this.emit(middle);
331 this.emit(rowB.join(""));
332 }

334 consensus(sequences: string[]): void {
335 const n = sequences[0].length;
336 if (sequences.some(s => s.length !== n)) throw new CommandError("length mismatch");
337 const profile: Record<string, number[]> = Object.fromEntries([...BASES].map(b => [b, []]));
338 let consensus = "";
339 let ambiguous = "";
340 for (let i = 0; i < n; i++) {
341 const table = counts(sequences.map(s => s[i]));
342 const largest = Math.max(...table.values());
343 const tied = [...BASES].filter(b => table.get(b) === largest).join("");
344 for (const b of BASES) profile[b].push(table.get(b) ?? 0);
345 consensus += tied[0];
346 ambiguous += SYMBOL[tied];
347 }
348 for (const b of BASES) this.emit(b, ...profile[b]);
349 this.emit("consensus", consensus);
350 this.emit("iupac", ambiguous);
351 }

353 primer(s: string): void {
354 const n = s.length;
355 const amount = gc(s);
356 const tm = temperature(s);
357 const clamp = gc(s.slice(-5));
358 const longestRun = Math.max(...s.match(/(.)\1*/g)!.map(run => run.length));
359 let repeated = 0;
360 for (let i = 0; i + 1 < n; i++) {
361 if (s[i] !== s[i + 1]) repeated = Math.max(repeated, repeatCount(s, i, 2));
362 }
363 const selfcomp = longestPalindrome(s);
364 const items: [string, string | number, boolean][] = [
365 ["length", n, n >= 18 && n <= 25],
366 ["gc", pct(amount, n), 40 * n <= 100 * amount && 100 * amount <= 60 * n],
367 ["tm", fixed(tm, 1), tm >= 520 && tm <= 620],
368 ["clamp", clamp, clamp >= 1 && clamp <= 3],
369 ["run", longestRun, longestRun <= 4],
370 ["repeat", repeated, repeated <= 4],
371 ["selfcomp", selfcomp, selfcomp <= 6]
372 ];
373 const failed: string[] = [];
374 for (const [label, value, passed] of items) {
375 this.emit(label, value, passed ? "OK" : "FAIL");
376 if (!passed) failed.push(label);
377 }
378 this.emit(failed.length ? "FAIL" : "PASS", ...failed);
379 }

381 fold(s: string, loop: number): void {
382 const n = s.length;
383 if (n > 100) throw new CommandError("too long");
384 const table: number[][] = Array.from({length: n}, () => Array(n).fill(0));
385 const value = (i: number, j: number): number => i > j ? 0 : table[i][j];
386 const pair = (i: number, j: number): boolean => ["AT", "TA", "CG", "GC", "GT", "TG"].includes(s[i] + s[j]);
387 for (let width = 2; width <= n; width++) {
388 for (let i = 0; i + width <= n; i++) {
389 const j = i + width - 1;
390 let best = value(i, j - 1);
391 for (let k = i; k <= j - loop - 1; k++) {
392 if (pair(k, j)) best = Math.max(best, value(i, k - 1) + 1 + value(k + 1, j - 1));
393 }
394 table[i][j] = best;
395 }
396 }
397 const structure = Array(n).fill(".") as string[];
398 const pending: number[][] = [[0, n - 1]];
399 while (pending.length) {
400 const [i, j] = pending.pop()!;
401 if (value(i, j) === 0) continue;
402 if (value(i, j) === value(i, j - 1)) {
403 pending.push([i, j - 1]);
404 continue;
405 }
406 for (let k = i; k <= j - loop - 1; k++) {
407 if (pair(k, j) && value(i, k - 1) + 1 + value(k + 1, j - 1) === value(i, j)) {
408 structure[k] = "(";
409 structure[j] = ")";
410 pending.push([i, k - 1], [k + 1, j - 1]);
411 break;
412 }
413 }
414 }
415 this.emit(structure.join(""));
416 this.emit("pairs", table[0][n - 1]);
417 }

419 assemble(name: string, minimum: number, sequences: string[]): void {
420 const unique = [...new Set(sequences)];
421 let fragments = unique.filter(s => !unique.some(t => t !== s && t.includes(s)));
422 while (fragments.length >= 2) {
423 let best = -1;
424 let chosenI = 0;
425 let chosenJ = 1;
426 for (let i = 0; i < fragments.length; i++) {
427 for (let j = 0; j < fragments.length; j++) {
428 if (i === j) continue;
429 const x = fragments[i];
430 const y = fragments[j];
431 let overlap = 0;
432 for (let o = Math.min(x.length, y.length) - 1; o >= 1; o--) {
433 if (x.slice(-o) === y.slice(0, o)) {
434 overlap = o;
435 break;
436 }
437 }
438 if (overlap > best) {
439 best = overlap;
440 chosenI = i;
441 chosenJ = j;
442 }
443 }
444 }
445 if (best < minimum) break;
446 const merged = fragments[chosenI] + fragments[chosenJ].slice(best);
447 fragments = fragments.flatMap((s, i) => i === chosenI ? [merged] : i === chosenJ || merged.includes(s) ? [] : [s]);
448 }
449 const longest = fragments.reduce((best, s) => s.length > best.length ? s : best);
450 const registration = this.store(name, longest);
451 this.emit("contigs", fragments.length);
452 for (const s of fragments) this.emit(s.length, s);
453 this.emit(registration);
454 }

456 command(cmd: string, a: string[]): void {
457 if (cmd === "SEQ") {
458 this.emit(this.store(a[0], a[1]));
459 return;
460 }
461 if (cmd === "UNPACK") {
462 const codes: number[] = [];
463 const length = Number(a[1]);
464 for (let i = 0; i < a[2].length; i += 2) {
465 const v = Number.parseInt(a[2].slice(i, i + 2), 16);
466 codes.push(Math.floor(v / 64), Math.floor(v / 16) % 4, Math.floor(v / 4) % 4, v % 4);
467 }
468 if (codes.slice(length).some(c => c !== 0)) throw new CommandError("invalid padding");
469 this.emit(this.store(a[0], codes.slice(0, length).map(c => BASES[c]).join("")));
470 return;
471 }
472 if (cmd === "JOIN") {
473 this.emit(this.store(a[0], a.slice(1).map(name => this.registry.get(name)!).join("")));
474 return;
475 }
476 if (cmd === "ASSEMBLE") {
477 this.assemble(a[0], Number(a[1]), a.slice(2).map(name => this.registry.get(name)!));
478 return;
479 }
480 if (cmd === "LIST") {
481 for (const name of [...this.registry.keys()].sort()) this.emit(name, this.registry.get(name)!.length, "nt");
482 if (!this.registry.size) this.emit("no sequences");
483 return;
484 }
485 const name = a[0];
486 const s = this.registry.get(name)!;
487 const n = s.length;
488 switch (cmd) {
489 case "SLICE": {
490 const start = Number(a[1]);
491 const end = Number(a[2]);
492 this.emit(this.store(a[3], start <= end ? s.slice(start - 1, end) : revcomp(s.slice(end - 1, start))));
493 break;
494 }
495 case "MUTATE": {
496 const position = Number(a[1]);
497 const base = a[2];
498 const old = s[position - 1];
499 if (old === base) throw new CommandError("no change");
500 this.registry.set(name, s.slice(0, position - 1) + base + s.slice(position));
501 this.emit(name, old + position + base, transition(old, base) ? "transition" : "transversion");
502 break;
503 }
504 case "DEL":
505 this.registry.delete(name);
506 this.emit("deleted", name);
507 break;
508 case "REVCOMP":
509 this.emit(revcomp(s));
510 break;
511 case "COMPLEMENT":
512 this.emit([...s].map(b => COMPLEMENT[b]).join(""));
513 break;
514 case "REVERSE":
515 this.emit([...s].reverse().join(""));
516 break;
517 case "TRANSCRIBE":
518 this.emit(s.replaceAll("T", "U"));
519 break;
520 case "TRANSLATE":
521 case "CODONS":
522 case "PROTEIN":
523 this.translation(cmd, s, Number(a[1]));
524 break;
525 case "ORFS":
526 this.orfs(s, Number(a[1]));
527 break;
528 case "GC": {
529 const table = counts(s);
530 this.emit(...[...BASES].map(b => b + "=" + (table.get(b) ?? 0)), "GC=" + pct(gc(s), n) + "%");
531 break;
532 }
533 case "GCWINDOW": {
534 const width = Number(a[1]);
535 const step = Number(a[2]);
536 if (width > n) throw new CommandError("window too large");
537 for (let i = 0; i + width <= n; i += step) this.emit((i + 1) + "-" + (i + width), pct(gc(s.slice(i, i + width)), width) + "%");
538 break;
539 }
540 case "SKEW": {
541 const values = [0];
542 for (const b of s) values.push(values[values.length - 1] + Number(b === "G") - Number(b === "C"));
543 for (const [label, extreme] of [["min", Math.min(...values)], ["max", Math.max(...values)]] as [string, number][]) {
544 this.emit(label, extreme, "at", ...values.flatMap((v, i) => v === extreme ? [i] : []));
545 }
546 break;
547 }
548 case "KMERS": {
549 const width = Number(a[1]);
550 if (width > n) throw new CommandError("k too large");
551 const table = counts(Array.from({length: n - width + 1}, (_, i) => s.slice(i, i + width)));
552 this.emit(n - width + 1, "windows", table.size, "distinct");
553 for (const word of frequencyOrder(table).slice(0, Number(a[2]))) this.emit(word, table.get(word)!);
554 break;
555 }
556 case "FIND":
557 this.emitList("forward", occurrences(s, a[1]));
558 this.emitList("reverse", occurrences(s, revcomp(a[1])));
559 break;
560 case "RESTRICT":
561 this.restrict(s, a.slice(1));
562 break;
563 case "PALINDROMES": {
564 const minimum = Number(a[1]);
565 const maximum = Number(a[2]);
566 if (minimum > maximum) throw new CommandError("invalid range");
567 let found = false;
568 for (let i = 0; i < n; i++) {
569 for (let width = minimum; width <= maximum && i + width <= n; width++) {
570 const part = s.slice(i, i + width);
571 if (part === revcomp(part)) {
572 this.emit(i + 1, width, part);
573 found = true;
574 }
575 }
576 }
577 if (!found) this.emit("none");
578 break;
579 }
580 case "REPEATS":
581 this.repeats(s, Number(a[1]), Number(a[2]), Number(a[3]));
582 break;
583 case "HAMMING": {
584 const other = this.registry.get(a[1])!;
585 if (other.length !== n) throw new CommandError("length mismatch");
586 let distance = 0;
587 let transitions = 0;
588 for (let i = 0; i < n; i++) {
589 if (s[i] !== other[i]) distance++;
590 if (transition(s[i], other[i])) transitions++;
591 }
592 this.emit("distance", distance, "transitions", transitions, "transversions", distance - transitions);
593 break;
594 }
595 case "ALIGN":
596 this.align(s, this.registry.get(a[1])!, a[2], Number(a[3]), Number(a[4]), Number(a[5]));
597 break;
598 case "CONSENSUS":
599 this.consensus(a.map(name => this.registry.get(name)!));
600 break;
601 case "TM":
602 this.emit("Tm", fixed(temperature(s), 1), n < 14 ? "wallace" : "gc");
603 break;
604 case "MASS": {
605 const single = strandMass(s);
606 this.emit("ss", fixed(single, 3), "ds", fixed(single + strandMass(revcomp(s)), 3));
607 break;
608 }
609 case "PRIMER":
610 this.primer(s);
611 break;
612 case "FOLD":
613 this.fold(s, Number(a[1]));
614 break;
615 case "PACK": {
616 const codes = [...s].map(b => BASES.indexOf(b));
617 while (codes.length % 4) codes.push(0);
618 let encoded = "";
619 for (let i = 0; i < codes.length; i += 4) {
620 const v = 64 * codes[i] + 16 * codes[i + 1] + 4 * codes[i + 2] + codes[i + 3];
621 encoded += v.toString(16).toUpperCase().padStart(2, "0");
622 }
623 this.emit(n, encoded);
624 break;
625 }
626 }
627 }

629 execute(line: string): void {
630 const [cmd, ...args] = line.split(" ");
631 try {
632 this.validate(cmd, args);
633 this.command(cmd, args);
634 } catch (error) {
635 if (!(error instanceof CommandError)) throw error;
636 this.emit("ERROR", error.message);
637 }
638 }
