29String reverse(String s) => s.split('').reversed.join();
30String complement(String s) => s.split('').map((c) {
31 final mask = symbols.indexOf(c), comp = ((mask & 1) << 3) | ((mask & 2) << 1) | ((mask & 4) >> 1) | ((mask & 8) >> 3);

33}).join();

61List<int> occurrences(String s, String motif) {
62 final pattern = motif.split('').map((symbol) {
63 final mask = symbols.indexOf(symbol);
64 return '[${[for (var k = 0; k < 4; k++) if ((mask & (1 << k)) != 0) bases[k]].join()}]';
65 }).join();
66 return RegExp('(?=$pattern)').allMatches(s).map((match) => match.start + 1).toList();

68int repeatCount(String s, int i, int u) {
69 final unit = s.substring(i, i + u);
70 var count = 1;
71 while (i + (count + 1) * u <= s.length && s.substring(i + count * u, i + (count + 1) * u) == unit) count++;
72 return count;
73}
74int tm(String s) {
75 final n = s.length, count = gc(s);
76 return n < 14 ? 10 * (2 * (n - count) + 4 * count) : (649 * n + 410 * count - 6724) ~/ n;
77}

80 const weights = [331222, 307197, 347221, 322208];

85 List<int> validate(String op, List<String> a) {
86 final schema = schemas[op];
87 if (schema == null) fail('unknown command $op');
88 var spec = schema;
89 final n = a.length;
90 if (op == 'JOIN') {
91 if (n < 3 || n > 9) fail('wrong argument count');
92 spec = ['new', ...List.filled(n - 1, 'name')];
93 } else if (op == 'RESTRICT') {
94 if (n < 2 || n > 7) fail('wrong argument count');
95 spec = ['name', ...List.filled(n - 1, 'enzyme')];
96 } else if (op == 'CONSENSUS') {
97 if (n < 2 || n > 10) fail('wrong argument count');
98 spec = List.filled(n, 'name');
99 } else if (op == 'ASSEMBLE') {
100 if (n < 4 || n > 10) fail('wrong argument count');
101 spec = ['new', '1:100', ...List.filled(n - 2, 'name')];
102 } else if (n != spec.length) {
103 fail('wrong argument count');
104 }
105 final values = List.filled(n, 0), seen = <String>{};
106 for (var i = 0; i < n; i++) {
107 final s = a[i], kind = spec[i];
108 if (kind == 'name') {
109 if (!registry.containsKey(s)) fail('unknown sequence $s');
110 } else if (kind == 'new') {
111 if (!RegExp(r'^[a-z][a-z0-9]{0,9}$').hasMatch(s)) fail('invalid name $s');
112 } else if (kind == 'seq') {
113 for (var j = 0; j < s.length; j++) if (!bases.contains(s[j])) fail('invalid base ${s[j]} at ${j + 1}');
114 } else if (kind == 'base') {
115 if (s.length != 1 || !bases.contains(s)) fail('invalid base $s');
116 } else if (kind == 'motif') {
117 if (s.isEmpty || s.length > 30 || s.split('').any((c) => symbols.indexOf(c) < 1)) fail('invalid motif $s');
118 } else if (kind == 'enzyme') {
119 if (!enzymeNames.contains(s)) fail('unknown enzyme $s');
120 if (!seen.add(s)) fail('duplicate enzyme $s');
121 } else if (kind == 'mode') {
122 if (s != 'GLOBAL' && s != 'LOCAL') fail('invalid mode $s');
123 } else if (kind == 'hex') {
124 if (s.length != 2 * ((values[1] + 3) ~/ 4) || !RegExp(r'^[0-9A-F]+$').hasMatch(s)) fail('invalid hex');
125 } else {
126 if (!RegExp(r'^-?[0-9]{1,6}$').hasMatch(s)) fail('invalid number $s');
127 final v = int.parse(s);
128 final range = kind == 'pos' ? [1, registry[a[0]]!.length] : kind == 'frame' ? [-3, 3] : kind.split(':').map(int.parse).toList();
129 if (v < range[0] || v > range[1] || kind == 'frame' && v == 0) fail('invalid number $s');
130 values[i] = v;
131 }
132 }
133 return values;
134 }
135 void store(String name, String s, StringBuffer out) {
136 if (s.length > 500) fail('too long');
137 final verb = registry.containsKey(name) ? 'replaced' : 'stored';
138 registry[name] = s;
139 out.writeln('$verb $name ${s.length} nt');
140 }
141 void align(String a, String b, bool local, int match, int mismatch, int gap, StringBuffer out) {
142 final m = a.length, n = b.length, h = List.generate(a.length + 1, (_) => List.filled(b.length + 1, 0));
143 if (!local) {
144 for (var i = 0; i <= m; i++) h[i][0] = i * gap;
145 for (var j = 0; j <= n; j++) h[0][j] = j * gap;
146 }
147 var score = 0, i1 = 0, j1 = 0;
148 for (var i = 1; i <= m; i++) {
149 for (var j = 1; j <= n; j++) {
150 h[i][j] = math.max(math.max(h[i - 1][j - 1] + (a[i - 1] == b[j - 1] ? match : mismatch), h[i - 1][j] + gap), h[i][j - 1] + gap);
151 if (local) {
152 h[i][j] = math.max(0, h[i][j]);
153 if (h[i][j] > score) {
154 score = h[i][j];
155 i1 = i;
156 j1 = j;
157 }
158 }
159 }
160 }
161 if (!local) {
162 score = h[m][n];
163 i1 = m;
164 j1 = n;
165 }
166 if (local && score == 0) {
167 out.writeln('score 0');
168 return;
169 }
170 var i = i1, j = j1;
171 final top = <String>[], bottom = <String>[];
172 while (local ? h[i][j] != 0 : i != 0 || j != 0) {
173 if (i > 0 && j > 0 && h[i][j] == h[i - 1][j - 1] + (a[i - 1] == b[j - 1] ? match : mismatch)) {
174 top.add(a[--i]);
175 bottom.add(b[--j]);
176 } else if (i > 0 && h[i][j] == h[i - 1][j] + gap) {
177 top.add(a[--i]);
178 bottom.add('-');
179 } else {
180 top.add('-');
181 bottom.add(b[--j]);
182 }
183 }
184 final aa = top.reversed.toList(), bb = bottom.reversed.toList();
185 final middle = List.generate(aa.length, (k) => aa[k] == '-' || bb[k] == '-' ? '-' : aa[k] == bb[k] ? '|' : '*');
186 out.writeln('score $score a ${i + 1}-$i1 b ${j + 1}-$j1 identity ${middle.where((c) => c == '|').length}/${aa.length} gaps ${middle.where((c) => c == '-').length}');
187 out.writeln(aa.join());
188 out.writeln(middle.join());
189 out.writeln(bb.join());
190 }
191 void fold(String s, int loop, StringBuffer out) {
192 final n = s.length, dp = List.generate(s.length, (_) => List.filled(s.length, 0));
193 int value(int i, int j) => i > j ? 0 : dp[i][j];
194 bool pair(int i, int j) => ['AT', 'TA', 'CG', 'GC', 'GT', 'TG'].contains('${s[i]}${s[j]}');
195 for (var length = 2; length <= n; length++) {
196 for (var i = 0; i + length <= n; i++) {
197 final j = i + length - 1;
198 dp[i][j] = dp[i][j - 1];
199 for (var k = i; k <= j - loop - 1; k++) if (pair(k, j)) dp[i][j] = math.max(dp[i][j], value(i, k - 1) + 1 + value(k + 1, j - 1));
200 }
201 }
202 final structure = List.filled(n, '.'), stack = <(int, int)>[(0, n - 1)];
203 while (stack.isNotEmpty) {
204 final (i, j) = stack.removeLast();
205 if (value(i, j) == 0) continue;
206 if (value(i, j) == value(i, j - 1)) {
207 stack.add((i, j - 1));
208 continue;
209 }
210 for (var k = i; k <= j - loop - 1; k++) {
211 if (pair(k, j) && value(i, k - 1) + 1 + value(k + 1, j - 1) == value(i, j)) {
212 structure[k] = '(';
213 structure[j] = ')';
214 stack.add((i, k - 1));
215 stack.add((k + 1, j - 1));
216 break;
217 }
218 }
219 }
220 out.writeln(structure.join());
221 out.writeln('pairs ${dp[0][n - 1]}');
222 }
223 void execute(List<String> tokens, StringBuffer out) {
224 final op = tokens[0], a = tokens.sublist(1), v = validate(op, a);
225 final name = a.isEmpty ? '' : a[0], s = registry[name] ?? '', n = s.length;
226 if (op == 'SEQ') {
227 store(name, a[1], out);
228 } else if (op == 'SLICE') {
229 store(a[3], v[1] <= v[2] ? s.substring(v[1] - 1, v[2]) : revcomp(s.substring(v[2] - 1, v[1])), out);
230 } else if (op == 'JOIN') {
231 store(name, a.skip(1).map((id) => registry[id]!).join(), out);
232 } else if (op == 'MUTATE') {
233 final p = v[1] - 1, old = s[v[1] - 1];
234 if (old == a[2]) fail('no change');
235 registry[name] = s.substring(0, p) + a[2] + s.substring(p + 1);
236 out.writeln('$name $old${p + 1}${a[2]} ${transition(old, a[2]) ? 'transition' : 'transversion'}');
237 } else if (op == 'DEL') {
238 registry.remove(name);
239 out.writeln('deleted $name');
240 } else if (op == 'LIST') {
241 final names = registry.keys.toList()..sort();
242 if (names.isEmpty) out.writeln('no sequences');
243 for (final name in names) out.writeln('$name ${registry[name]!.length} nt');
244 } else if (['REVCOMP', 'COMPLEMENT', 'REVERSE', 'TRANSCRIBE'].contains(op)) {
245 out.writeln(op == 'REVCOMP' ? revcomp(s) : op == 'COMPLEMENT' ? complement(s) : op == 'REVERSE' ? reverse(s) : s.replaceAll('T', 'U'));
246 } else if (['TRANSLATE', 'CODONS', 'PROTEIN'].contains(op)) {
247 final cs = codons(s, v[1]);
248 final translation = cs.map(aa).join();
249 if (op == 'TRANSLATE') {
250 out.writeln(translation.isEmpty ? 'none' : translation);
251 } else if (op == 'CODONS') {
252 final freq = counts(cs);
253 if (freq.isEmpty) out.writeln('none');
254 for (final c in rankedCounts(freq)) out.writeln('$c ${aa(c)} ${freq[c]}');
255 } else {
256 final protein = translation.split('*').first;
257 if (protein.isEmpty) fail('empty protein');
258 var weight = 18015, charge = 0;
259 for (final c in protein.split('')) {
260 weight += residueMasses[aminoAcids.indexOf(c)];
261 if ('KR'.contains(c)) charge++;
262 if ('DE'.contains(c)) charge--;
263 }
264 final freq = counts(protein.split(''));
265 out.writeln(protein);
266 out.writeln('length ${protein.length} mass ${milli(weight)} charge ${charge > 0 ? '+' : ''}$charge');
267 out.writeln(rankedCounts(freq).map((c) => '$c=${freq[c]}').join(' '));
268 }
269 } else if (op == 'ORFS') {
270 final results = <(int, int, int, String)>[];
271 for (final f in [1, 2, 3, -1, -2, -3]) {
272 final cs = codons(s, f);
273 int? start;
274 for (var i = 0; i < cs.length; i++) {
275 if (start == null && cs[i] == 'ATG') {
276 start = i;
277 } else if (start != null && aa(cs[i]) == '*') {
278 if (i - start >= v[1]) {
279 final left = f.abs() + 3 * start, right = f.abs() + 3 * i + 2;
280 results.add((f < 0 ? n + 1 - right : left, f < 0 ? n + 1 - left : right, f, cs.sublist(start, i).map(aa).join()));
281 }
282 start = null;
283 }
284 }
285 }
286 results.sort((a, b) => a.$1.compareTo(b.$1));
287 if (results.isEmpty) out.writeln('none');
288 for (final r in results) out.writeln('${r.$3 > 0 ? '+' : ''}${r.$3} ${r.$1} ${r.$2} ${r.$4.length} ${r.$4}');
289 } else if (op == 'GC') {
290 final freq = counts(s.split(''));
291 out.writeln('${bases.split('').map((c) => '$c=${freq[c] ?? 0}').join(' ')} GC=${pct(gc(s), n)}%');
292 } else if (op == 'GCWINDOW') {
293 if (v[1] > n) fail('window too large');
294 for (var i = 0; i + v[1] <= n; i += v[2]) out.writeln('${i + 1}-${i + v[1]} ${pct(gc(s.substring(i, i + v[1])), v[1])}%');
295 } else if (op == 'SKEW') {
296 final values = [0];
297 for (final c in s.split('')) values.add(values.last + (c == 'G' ? 1 : c == 'C' ? -1 : 0));
298 final low = values.reduce(math.min), high = values.reduce(math.max);
299 out.writeln('min $low at ${List.generate(n + 1, (i) => i).where((i) => values[i] == low).join(' ')}');
300 out.writeln('max $high at ${List.generate(n + 1, (i) => i).where((i) => values[i] == high).join(' ')}');
301 } else if (op == 'KMERS') {
302 if (v[1] > n) fail('k too large');
303 final freq = counts([for (var i = 0; i + v[1] <= n; i++) s.substring(i, i + v[1])]);
304 out.writeln('${n - v[1] + 1} windows ${freq.length} distinct');
305 for (final k in rankedCounts(freq).take(v[2])) out.writeln('$k ${freq[k]}');
306 } else if (op == 'FIND') {
307 out.writeln('forward ${listText(occurrences(s, a[1]))}');
308 out.writeln('reverse ${listText(occurrences(s, revcomp(a[1])))}');
309 } else if (op == 'RESTRICT') {
310 final cuts = <int>{};
311 for (final e in a.skip(1)) {
312 final k = enzymeNames.indexOf(e);
313 final points = occurrences(s, recognition[k]).map((p) => p + cutOffsets[k] - 1).where((p) => p > 0 && p < n).toList();
314 cuts.addAll(points);
315 out.writeln('$e ${listText(points)}');
316 }
317 final positions = [0, ...(cuts.toList()..sort()), n];
318 final sizes = [for (var i = 1; i < positions.length; i++) positions[i] - positions[i - 1]]
319 ..sort((a, b) => b.compareTo(a));
320 out.writeln('fragments ${[for (var i = 1; i < positions.length; i++) '${positions[i - 1] + 1}-${positions[i]}'].join(' ')}');
321 out.writeln('sizes ${sizes.join(' ')}');
322 } else if (op == 'PALINDROMES') {
323 if (v[1] > v[2]) fail('invalid range');
324 var found = false;
325 for (var i = 0; i < n; i++) for (var len = v[1]; len <= v[2] && i + len <= n; len++) {
326 final part = s.substring(i, i + len);
327 if (part == revcomp(part)) {
328 found = true;
329 out.writeln('${i + 1} $len $part');
330 }
331 }
332 if (!found) out.writeln('none');
333 } else if (op == 'REPEATS') {
334 if (v[1] > v[2]) fail('invalid range');
335 final results = <(int, int, int, String)>[];
336 for (var u = v[1]; u <= v[2]; u++) {
337 var i = 0;
338 while (i + u <= n) {
339 final unit = s.substring(i, i + u), count = repeatCount(s, i, u);
340 var primitive = true;
341 for (var d = 1; d < u; d++) if (u % d == 0 && unit.substring(0, d) * (u ~/ d) == unit) primitive = false;
342 if (primitive && count >= v[3]) {
343 results.add((i, u, count, unit));
344 i += count * u;
345 } else {
346 i++;
347 }
348 }
349 }
350 results.sort((a, b) {
351 final c = a.$1.compareTo(b.$1);
352 return c != 0 ? c : a.$2.compareTo(b.$2);
353 });
354 if (results.isEmpty) out.writeln('none');
355 for (final r in results) out.writeln('${r.$1 + 1}-${r.$1 + r.$2 * r.$3} ${r.$4}x${r.$3}');
356 } else if (op == 'HAMMING') {
357 final b = registry[a[1]]!;
358 if (n != b.length) fail('length mismatch');
359 var distance = 0, transitions = 0;
360 for (var i = 0; i < n; i++) {
361 if (s[i] != b[i]) {
362 distance++;
363 if (transition(s[i], b[i])) transitions++;
364 }
365 }
366 out.writeln('distance $distance transitions $transitions transversions ${distance - transitions}');
367 } else if (op == 'ALIGN') {
368 align(s, registry[a[1]]!, a[2] == 'LOCAL', v[3], v[4], v[5], out);
369 } else if (op == 'CONSENSUS') {
370 final seqs = a.map((id) => registry[id]!).toList();
371 if (seqs.any((s) => s.length != n)) fail('length mismatch');
372 final profile = List.generate(4, (_) => List.filled(n, 0)), consensus = <String>[], iupac = <String>[];
373 for (var i = 0; i < n; i++) {
374 for (final seq in seqs) profile[bases.indexOf(seq[i])][i]++;
375 final maxCount = profile.map((row) => row[i]).reduce(math.max);
376 final winners = [for (var k = 0; k < 4; k++) if (profile[k][i] == maxCount) k];
377 consensus.add(bases[winners.first]);
378 iupac.add(symbols[winners.fold(0, (mask, k) => mask | (1 << k))]);
379 }
380 for (var k = 0; k < 4; k++) out.writeln('${bases[k]} ${profile[k].join(' ')}');
381 out.writeln('consensus ${consensus.join()}');
382 out.writeln('iupac ${iupac.join()}');
383 } else if (op == 'TM') {
384 out.writeln('Tm ${temperature(tm(s))} ${n < 14 ? 'wallace' : 'gc'}');
385 } else if (op == 'MASS') {
386 final single = mass(s);
387 out.writeln('ss ${milli(single)} ds ${milli(single + mass(revcomp(s)))}');
388 } else if (op == 'PRIMER') {
389 final maxRun = RegExp(r'([ACGT])\1*').allMatches(s).map((run) => run[0]!.length).reduce(math.max);
390 var repeat = 0, selfcomp = 0;
391 for (var i = 0; i + 1 < n; i++) if (s[i] != s[i + 1]) repeat = math.max(repeat, repeatCount(s, i, 2));
392 for (var center = 1; center < n; center++) {
393 var left = center - 1, right = center;
394 while (left >= 0 && right < n && bases.indexOf(s[left]) + bases.indexOf(s[right]) == 3) {
395 selfcomp = math.max(selfcomp, right - left + 1);
396 left--;
397 right++;
398 }
399 }
400 final clamp = gc(s.substring(math.max(0, n - 5))), count = gc(s), t = tm(s);
401 final names = ['length', 'gc', 'tm', 'clamp', 'run', 'repeat', 'selfcomp'];
402 final values = ['$n', pct(count, n), temperature(t), '$clamp', '$maxRun', '$repeat', '$selfcomp'];
403 final passes = [n >= 18 && n <= 25, 40 * n <= 100 * count && 100 * count <= 60 * n, t >= 520 && t <= 620, clamp >= 1 && clamp <= 3, maxRun <= 4, repeat <= 4, selfcomp <= 6];
404 for (var i = 0; i < 7; i++) out.writeln('${names[i]} ${values[i]} ${passes[i] ? 'OK' : 'FAIL'}');
405 out.writeln(passes.every((x) => x) ? 'PASS' : 'FAIL ${[for (var i = 0; i < 7; i++) if (!passes[i]) names[i]].join(' ')}');
406 } else if (op == 'FOLD') {
407 if (n > 100) fail('too long');
408 fold(s, v[1], out);
409 } else if (op == 'ASSEMBLE') {
410 var fragments = {for (final id in a.skip(2)) registry[id]!}.toList();
411 fragments = fragments.where((s) => !fragments.any((b) => s != b && b.contains(s))).toList();
412 while (fragments.length > 1) {
413 var best = 0, bx = 0, by = 0;
414 for (var x = 0; x < fragments.length; x++) for (var y = 0; y < fragments.length; y++) {
415 if (x == y) continue;
416 for (var o = math.min(fragments[x].length, fragments[y].length) - 1; o > best; o--) {
417 if (fragments[x].endsWith(fragments[y].substring(0, o))) {
418 best = o;
419 bx = x;
420 by = y;
421 break;
422 }
423 }
424 }
425 if (best < v[1]) break;
426 final merged = fragments[bx] + fragments[by].substring(best);
427 fragments[bx] = merged;
428 fragments = [for (var i = 0; i < fragments.length; i++) if (i == bx || i != by && !merged.contains(fragments[i])) fragments[i]];
429 }
430 final longest = fragments.reduce((a, b) => a.length >= b.length ? a : b);
431 if (longest.length > 500) fail('too long');
432 out.writeln('contigs ${fragments.length}');
433 for (final f in fragments) out.writeln('${f.length} $f');
434 store(name, longest, out);
435 } else if (op == 'PACK') {
436 final hex = StringBuffer();
437 for (var i = 0; i < n; i += 4) {
438 var value = 0;
439 for (var j = 0; j < 4; j++) value = value * 4 + (i + j < n ? bases.indexOf(s[i + j]) : 0);
440 hex.write(value.toRadixString(16).toUpperCase().padLeft(2, '0'));
441 }
442 out.writeln('$n $hex');
443 } else if (op == 'UNPACK') {
444 final codes = <int>[];
445 for (var i = 0; i < a[2].length; i += 2) {
446 final value = int.parse(a[2].substring(i, i + 2), radix: 16);
447 codes.addAll([value ~/ 64, value ~/ 16 % 4, value ~/ 4 % 4, value % 4]);
448 }
449 if (codes.skip(v[1]).any((c) => c != 0)) fail('invalid padding');
450 store(name, codes.take(v[1]).map((c) => bases[c]).join(), out);
451 }
452 }
453}
