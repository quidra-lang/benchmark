39static bool leap(int year) {
40 return year % 4 == 0 && (year % 100 != 0 || year % 400 == 0);
41}

43static int monthDays(int year, int month) {
44 static const int lengths[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
45 return lengths[month - 1] + (month == 2 && leap(year) ? 1 : 0);
46}

48static int parseDate(const std::string& text) {
49 int year = std::stoi(text.substr(0, 4)), month = std::stoi(text.substr(5, 2)), day = std::stoi(text.substr(8, 2));
50 if (month < 1 || month > 12 || day < 1 || day > monthDays(year, month)) fail("invalid date " + text);
51 int serial = day - 1;
52 for (int y = 2000; y < year; ++y) serial += leap(y) ? 366 : 365;
53 for (int m = 1; m < month; ++m) serial += monthDays(year, m);
54 return serial;
55}
56
57static int weekday(int date) {
58 return (date + 5) % 7;
59}
60
61static std::string dateText(int date) {
62 static const char* weekdays[] = {"Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"};
63 int year = 2000, month = 1, remaining = date;
64 while (remaining >= (leap(year) ? 366 : 365)) {
65 remaining -= leap(year) ? 366 : 365;
66 ++year;
67 }
68 while (remaining >= monthDays(year, month)) {
69 remaining -= monthDays(year, month);
70 ++month;
71 }
72 std::ostringstream text;
73 text << year << '-' << std::setfill('0') << std::setw(2) << month << '-' << std::setw(2) << remaining + 1
74 << ' ' << weekdays[weekday(date)];
75 return text.str();
76}

134 std::pair<std::vector<std::string>, std::string> topology() const {
135 std::map<std::string, int> degree;
136 for (const auto& [name, item] : tasks) degree[name] = 0;
137 for (const auto& [ends, link] : dependencies) ++degree[ends.second];
138 std::set<std::string> ready;
139 for (const auto& [name, count] : degree) {
140 if (count == 0) ready.insert(name);
141 }
142 std::vector<std::string> order;
143 while (!ready.empty()) {
144 std::string name = *ready.begin();
145 ready.erase(ready.begin());
146 order.push_back(name);
147 for (const auto& [ends, link] : dependencies) {
148 if (ends.first == name && --degree[ends.second] == 0) ready.insert(ends.second);
149 }
150 }
151 if (order.size() == tasks.size()) return {order, ""};
152 std::string current;
153 for (const auto& [name, count] : degree) {
154 if (count > 0) {
155 current = name;
156 break;
157 }
158 }
159 std::vector<std::string> backwards;
160 while (std::find(backwards.begin(), backwards.end(), current) == backwards.end()) {
161 backwards.push_back(current);
162 for (const auto& [ends, link] : dependencies) {
163 if (ends.second == current && degree.at(ends.first) > 0) {
164 current = ends.first;
165 break;
166 }
167 }
168 }
169 auto first = std::find(backwards.begin(), backwards.end(), current);
170 std::vector<std::string> cycle(first, backwards.end());
171 std::reverse(cycle.begin(), cycle.end());
172 std::rotate(cycle.begin(), std::min_element(cycle.begin(), cycle.end()), cycle.end());
173 cycle.push_back(cycle.front());
174 return {order, joined(cycle, " -> ")};
175 }

177 Schedule compute(const std::vector<std::string>& order, const std::string& changed = "", int replacement = 0) const {
178 Schedule schedule;
179 for (const std::string& name : order) {
180 Timing time;
181 const Task& item = tasks.at(name);
182 time.duration = name == changed ? replacement : item.duration();
183 time.es = item.notBefore;
184 for (const auto& [ends, link] : dependencies) {
185 if (ends.second != name) continue;
186 const Timing& previous = schedule.times.at(ends.first);
187 int bound = link.kind == "SS" ? previous.es : previous.ef;
188 bound += link.lag;
189 if (link.kind == "FF") bound -= time.duration;
190 time.es = std::max(time.es, bound);
191 }
192 time.ef = time.es + time.duration;
193 schedule.duration = std::max(schedule.duration, time.ef);
194 schedule.times[name] = time;
195 }
196 for (auto position = order.rbegin(); position != order.rend(); ++position) {
197 Timing& time = schedule.times.at(*position);
198 time.lf = schedule.duration;
199 for (const auto& [ends, link] : dependencies) {
200 if (ends.first != *position) continue;
201 const Timing& next = schedule.times.at(ends.second);
202 int bound = link.kind == "FF" ? next.lf : next.ls;
203 bound -= link.lag;
204 if (link.kind == "SS") bound += time.duration;
205 time.lf = std::min(time.lf, bound);
206 }
207 time.ls = time.lf - time.duration;
208 time.tf = time.ls - time.es;
209 time.free = schedule.duration - time.ef;
210 }
211 for (const auto& [ends, link] : dependencies) {
212 Timing& previous = schedule.times.at(ends.first);
213 const Timing& next = schedule.times.at(ends.second);
214 int margin = (link.kind == "FF" ? next.ef : next.es)
215 - (link.kind == "SS" ? previous.es : previous.ef) - link.lag;
216 previous.free = std::min(previous.free, margin);
217 }
218 return schedule;
219 }

221 static std::vector<std::string> criticalNames(const Schedule& schedule) {
222 std::vector<std::string> names;
223 for (const auto& [name, time] : schedule.times) {
224 if (time.tf == 0) names.push_back(name);
225 }
226 return names;
227 }

229 std::vector<std::vector<std::string>> criticalPaths(const Schedule& schedule) const {
230 std::map<std::string, std::vector<std::string>> next;
231 std::set<std::string> nonroots;
232 for (const std::string& name : criticalNames(schedule)) next[name] = {};
233 for (const auto& [ends, link] : dependencies) {
234 const Timing& previous = schedule.times.at(ends.first);
235 const Timing& following = schedule.times.at(ends.second);
236 int margin = (link.kind == "FF" ? following.ef : following.es)
237 - (link.kind == "SS" ? previous.es : previous.ef) - link.lag;
238 if (previous.tf == 0 && following.tf == 0 && margin == 0) {
239 next[ends.first].push_back(ends.second);
240 nonroots.insert(ends.second);
241 }
242 }
243 std::vector<std::vector<std::string>> paths;
244 std::vector<std::string> path;
245 std::function<void(const std::string&)> visit = [&](const std::string& name) {
246 path.push_back(name);
247 if (next.at(name).empty()) paths.push_back(path);
248 else for (const std::string& child : next.at(name)) visit(child);
249 path.pop_back();
250 };
251 for (const auto& [name, children] : next) {
252 if (!nonroots.count(name)) visit(name);
253 }
254 return paths;
255 }

258 std::vector<std::string> names;
259 for (const auto& [name, time] : schedule.times) names.push_back(name);
260 std::sort(names.begin(), names.end(), [&](const std::string& a, const std::string& b) {
261 return std::make_pair(schedule.times.at(a).es, a) < std::make_pair(schedule.times.at(b).es, b);
262 });
263 return names;

270 void calendar(const Schedule& schedule) const {
271 if (!startDate) fail("no start date");
272 std::vector<int> dates;
273 for (int date = *startDate; dates.size() < static_cast<std::size_t>(schedule.duration); ++date) {
274 if (working(date)) dates.push_back(date);
275 }
276 for (const std::string& name : startOrder(schedule)) {
277 const Timing& time = schedule.times.at(name);
278 std::cout << name << ' ' << dateText(dates[time.es]) << " - " << dateText(dates[time.ef - 1]) << '\n';
279 }
280 int first = dates.front(), last = dates.back();
281 std::cout << "project " << dateText(first) << " - " << dateText(last) << ", " << schedule.duration
282 << " working days, " << last - first + 1 << " calendar days\n";
283 if (deadline) {
284 int spare = 0;
285 for (int date = std::min(last, *deadline) + 1; date <= std::max(last, *deadline); ++date) spare += working(date);
286 std::cout << "deadline " << dateText(*deadline);
287 if (last <= *deadline) std::cout << " met, " << spare << " working days to spare\n";
288 else std::cout << " missed by " << spare << " working days\n";
289 }
290 std::vector<std::string> inSpan;
291 for (int date : holidays) {
292 if (first <= date && date <= last && weekday(date) < 5) inSpan.push_back(dateText(date));
293 }
294 std::cout << "holidays in span: " << (inSpan.empty() ? "none" : joined(inSpan, ", ")) << '\n';
295 }

297 std::map<std::string, std::vector<int>> usage(const Schedule& schedule) const {
298 std::map<std::string, std::vector<int>> totals;
299 for (const auto& [name, item] : tasks) {
300 std::vector<int>& days = totals[item.resource];
301 days.resize(schedule.duration, 0);
302 const Timing& time = schedule.times.at(name);
303 for (int day = time.es; day < time.ef; ++day) days[day] += item.units;
304 }
305 return totals;
306 }

308 void level(const Schedule& schedule, int fallback) const {
309 auto capacity = [&](const std::string& resource) {
310 auto found = capacities.find(resource);
311 return found == capacities.end() ? fallback : found->second;
312 };
313 for (const auto& [name, item] : tasks) {
314 int limit = capacity(item.resource);
315 if (item.units > limit) fail("capacity " + std::to_string(limit) + " too small for " + name);
316 }
317 Schedule leveled;
318 std::vector<std::string> decided;
319 std::map<std::string, std::vector<int>> totals;
320 while (decided.size() < tasks.size()) {
321 std::string best;
322 for (const auto& [name, item] : tasks) {
323 if (leveled.times.count(name)) continue;
324 bool ready = true;
325 for (const auto& [ends, link] : dependencies) {
326 if (ends.second == name && !leveled.times.count(ends.first)) ready = false;
327 }
328 if (ready && (best.empty() || schedule.times.at(name).ls < schedule.times.at(best).ls)) best = name;
329 }
330 const Task& item = tasks.at(best);
331 Timing time;
332 time.duration = item.duration();
333 time.es = item.notBefore;
334 for (const auto& [ends, link] : dependencies) {
335 if (ends.second != best) continue;
336 const Timing& previous = leveled.times.at(ends.first);
337 int bound = (link.kind == "SS" ? previous.es : previous.ef) + link.lag;
338 if (link.kind == "FF") bound -= time.duration;
339 time.es = std::max(time.es, bound);
340 }
341 std::vector<int>& days = totals[item.resource];
342 while (true) {
343 days.resize(std::max(days.size(), static_cast<std::size_t>(time.es + time.duration)), 0);
344 bool available = true;
345 for (int day = time.es; day < time.es + time.duration; ++day) {
346 if (days[day] + item.units > capacity(item.resource)) available = false;
347 }
348 if (available) break;
349 ++time.es;
350 }
351 time.ef = time.es + time.duration;
352 for (int day = time.es; day < time.ef; ++day) days[day] += item.units;
353 leveled.times[best] = time;
354 leveled.duration = std::max(leveled.duration, time.ef);
355 decided.push_back(best);
356 }
357 std::cout << "level capacity " << fallback << ": duration " << schedule.duration << " -> " << leveled.duration << '\n';
358 for (const std::string& name : decided) {
359 const Timing& time = leveled.times.at(name);
360 std::cout << name << " start " << time.es << " finish " << time.ef << " shift "
361 << time.es - schedule.times.at(name).es << '\n';
362 }
363 auto original = usage(schedule), result = usage(leveled);
364 for (const auto& [resource, days] : original) {
365 const std::vector<int>& changed = result.at(resource);
366 std::cout << "resource " << resource << " capacity " << capacity(resource) << " peak "
367 << *std::max_element(days.begin(), days.end()) << " -> " << *std::max_element(changed.begin(), changed.end()) << '\n';
368 }
369 }

371 void crash(const std::vector<std::string>& order, int budget) {
372 int initial = compute(order).duration, remaining = budget, count = 0;
373 std::string reason;
374 while (true) {
375 Schedule schedule = compute(order);
376 std::string best;
377 for (const auto& [name, item] : tasks) {
378 if (schedule.times.at(name).tf != 0 || !item.crash || item.used >= item.crash->limit) continue;
379 if (compute(order, name, item.duration() - 1).duration >= schedule.duration) continue;
380 if (best.empty() || item.crash->cost < tasks.at(best).crash->cost) best = name;
381 }
382 if (best.empty()) {
383 reason = "no candidate";
384 break;
385 }
386 Task& item = tasks.at(best);
387 if (item.crash->cost > remaining) {
388 reason = "insufficient budget";
389 break;
390 }
391 int before = item.duration();
392 ++item.used;
393 remaining -= item.crash->cost;
394 std::cout << "step " << ++count << ": " << best << ' ' << before << " -> " << item.duration()
395 << ", cost " << item.crash->cost << ", project " << compute(order).duration << '\n';
396 }
397 std::cout << "crash done: " << count << " days, cost " << budget - remaining << ", project " << initial
398 << " -> " << compute(order).duration << ", budget left " << remaining << ", " << reason << '\n';
399 }

401 void analyze(const std::vector<std::string>& words) {
402 if (tasks.empty()) fail("no tasks");
403 auto [order, cycle] = topology();
404 if (!cycle.empty()) fail("cycle " + cycle);
405 const std::string& command = words[0];
406 if (command == "WHATIF") task(words[1]);
407 if (command == "VARIANCE" && !baseline) fail("no baseline");
408 Schedule schedule = compute(order);
409 if (command == "CPM") {
410 std::size_t width = 4;
411 for (const auto& [name, item] : tasks) width = std::max(width, name.size());
412 std::cout << std::left << std::setw(static_cast<int>(width)) << "TASK" << std::right;
413 for (const char* label : {"DUR", "ES", "EF", "LS", "LF", "TF", "FF"}) std::cout << std::setw(5) << label;
414 std::cout << '\n';
415 for (const auto& [name, time] : schedule.times) {
416 std::cout << std::left << std::setw(static_cast<int>(width)) << name << std::right;
417 for (int value : {time.duration, time.es, time.ef, time.ls, time.lf, time.tf, time.free}) std::cout << std::setw(5) << value;
418 if (time.tf == 0) std::cout << " *";
419 std::cout << '\n';
420 }
421 std::cout << "project duration " << schedule.duration << '\n';
422 } else if (command == "CRITICAL" || command == "PERTSUM") {
423 std::vector<std::vector<std::string>> paths = criticalPaths(schedule);
424 if (command == "CRITICAL") std::cout << "critical paths: " << paths.size() << '\n';
425 for (const auto& path : paths) {
426 std::cout << joined(path, " -> ");
427 if (command == "PERTSUM") {
428 int expected = 0, variance = 0;
429 for (const std::string& name : path) {
430 const Task& item = tasks.at(name);
431 std::array<int, 3> estimate = item.estimate.value_or(std::array<int, 3>{item.duration(), item.duration(), item.duration()});
432 expected += estimate[0] + 4 * estimate[1] + estimate[2];
433 variance += (estimate[2] - estimate[0]) * (estimate[2] - estimate[0]);
434 }
435 std::cout << ": expected " << decimal(expected, 6) << " variance " << decimal(variance, 36);
436 }
437 std::cout << '\n';
438 }
439 } else if (command == "SLACK") {
440 std::map<int, std::vector<std::string>> floats;
441 for (const auto& [name, time] : schedule.times) floats[time.tf].push_back(name);
442 for (const auto& [amount, names] : floats) std::cout << "float " << amount << ": " << joined(names) << '\n';
443 } else if (command == "DRAG") {
444 for (const std::string& name : criticalNames(schedule)) {
445 std::cout << name << " drag " << schedule.duration - compute(order, name, 0).duration << '\n';
446 }
447 } else if (command == "STATS") {
448 std::map<std::string, int> kinds, work;
449 for (const auto& [ends, link] : dependencies) ++kinds[link.kind];
450 for (const auto& [name, item] : tasks) work[item.resource] += item.duration() * item.units;
451 int totalFloat = 0;
452 for (const auto& [name, time] : schedule.times) totalFloat += time.tf;
453 std::cout << "tasks " << tasks.size() << ", critical " << criticalNames(schedule).size() << ", dependencies "
454 << dependencies.size() << " (FS " << kinds["FS"] << ", SS " << kinds["SS"] << ", FF " << kinds["FF"] << ")\nwork ";
455 std::vector<std::string> totals;
456 for (const auto& [resource, amount] : work) totals.push_back(resource + " " + std::to_string(amount));
457 std::cout << joined(totals, ", ") << "\nfloat total " << totalFloat << ", average " << decimal(totalFloat, static_cast<int>(tasks.size()))
458 << "\nduration " << schedule.duration << ", critical share " << decimal(100 * static_cast<int>(criticalNames(schedule).size()), static_cast<int>(tasks.size())) << "%\n";
459 } else if (command == "CALENDAR") calendar(schedule);
460 else if (command == "GANTT") {
461 if (schedule.duration > 60) fail("chart too wide (" + std::to_string(schedule.duration) + " days)");
462 std::size_t width = 0;
463 for (const auto& [name, item] : tasks) width = std::max(width, name.size());
464 std::cout << std::string(width, ' ') << " |";
465 for (int day = 0; day < schedule.duration; ++day) std::cout << day % 10;
466 std::cout << "|\n";
467 for (const std::string& name : startOrder(schedule)) {
468 const Timing& time = schedule.times.at(name);
469 std::string bar(schedule.duration, ' ');
470 for (int day = 0; day < schedule.duration; ++day) {
471 if (time.es <= day && day < time.ef) bar[day] = time.tf == 0 ? '=' : '-';
472 else if (time.ef <= day && day < time.lf) bar[day] = '.';
473 }
474 std::cout << name << std::string(width - name.size(), ' ') << " |" << bar << "|\n";
475 }
476 } else if (command == "LEVEL") level(schedule, std::stoi(words[1]));
477 else if (command == "LOAD") {
478 auto totals = usage(schedule);
479 auto found = totals.find(words[1]);
480 if (found == totals.end()) fail("unknown resource " + words[1]);
481 const std::vector<int>& days = found->second;
482 std::cout << "load " << words[1] << ':';
483 for (int amount : days) std::cout << ' ' << amount;
484 auto peak = std::max_element(days.begin(), days.end());
485 std::cout << "\npeak " << *peak << " on day " << peak - days.begin() << '\n';
486 auto limit = capacities.find(words[1]);
487 if (limit != capacities.end()) {
488 int exceeded = static_cast<int>(std::count_if(days.begin(), days.end(), [&](int amount) { return amount > limit->second; }));
489 std::cout << "capacity " << limit->second << " exceeded on " << exceeded << " days\n";
490 }
491 } else if (command == "CRASH") crash(order, std::stoi(words[1]));
492 else if (command == "WHATIF") {
493 Schedule after = compute(order, words[1], std::stoi(words[2]));
494 std::cout << "whatif " << words[1] << ' ' << tasks.at(words[1]).duration() << " -> " << words[2]
495 << ": project " << schedule.duration << " -> " << after.duration << '\n';
496 for (const auto& [name, before] : schedule.times) {
497 const Timing& changed = after.times.at(name);
498 if (before.es == changed.es && before.ef == changed.ef && before.tf == changed.tf) continue;
499 std::cout << name;
500 if (before.es != changed.es) std::cout << " ES " << before.es << "->" << changed.es;
501 if (before.ef != changed.ef) std::cout << " EF " << before.ef << "->" << changed.ef;
502 if (before.tf != changed.tf) std::cout << " TF " << before.tf << "->" << changed.tf;
503 std::cout << '\n';
504 }
505 if (criticalNames(schedule) != criticalNames(after)) std::cout << "critical " << joined(criticalNames(schedule)) << " -> " << joined(criticalNames(after)) << '\n';
506 } else if (command == "BASELINE") {
507 Baseline saved{schedule.duration, {}};
508 for (const auto& [name, time] : schedule.times) saved.times[name] = {time.es, time.ef};
509 baseline = std::move(saved);
510 std::cout << "baseline saved: " << tasks.size() << " tasks, duration " << schedule.duration << '\n';
511 } else if (command == "VARIANCE") {
512 std::cout << "variance: duration " << baseline->duration << " -> " << schedule.duration << " (" << signedNumber(schedule.duration - baseline->duration) << ")\n";
513 std::set<std::string> names;
514 for (const auto& [name, time] : baseline->times) names.insert(name);
515 for (const auto& [name, time] : schedule.times) names.insert(name);
516 bool changed = false;
517 for (const std::string& name : names) {
518 auto previous = baseline->times.find(name);
519 auto current = schedule.times.find(name);
520 if (previous == baseline->times.end()) std::cout << name << " new\n";
521 else if (current == schedule.times.end()) std::cout << name << " removed\n";
522 else if (current->second.es != previous->second.first || current->second.ef != previous->second.second) {
523 std::cout << name << " start " << signedNumber(current->second.es - previous->second.first)
524 << " finish " << signedNumber(current->second.ef - previous->second.second) << '\n';
525 } else continue;
526 changed = true;
527 }
528 if (!changed) std::cout << "no task changes\n";
529 }

532public:
533 void execute(const std::string& line) {
534 std::istringstream input(line);
535 std::vector<std::string> words;
536 for (std::string word; input >> word;) words.push_back(word);
537 const std::string& command = words[0];
538 if (command == "TASK") {
539 if (tasks.count(words[1])) fail("task " + words[1] + " already exists");
540 tasks.emplace(words[1], Task{std::stoi(words[2]), words[3], std::stoi(words[4])});
541 std::cout << "task " << words[1] << " added\n";
542 } else if (command == "DEP" || command == "UNDEP") {
543 task(words[1]);
544 task(words[2]);
545 std::pair<std::string, std::string> ends{words[1], words[2]};
546 if (command == "DEP") {
547 if (words[1] == words[2]) fail("self dependency " + words[1]);
548 if (dependencies.count(ends)) fail("dependency " + words[1] + " -> " + words[2] + " already exists");
549 int lag = std::stoi(words[4]);
550 dependencies.emplace(ends, Dependency{words[3], lag});
551 std::cout << "dependency " << words[1] << " -> " << words[2] << ' ' << words[3] << signedNumber(lag) << " added\n";
552 } else {
553 if (!dependencies.erase(ends)) fail("no dependency " + words[1] + " -> " + words[2]);
554 std::cout << "dependency " << words[1] << " -> " << words[2] << " removed\n";
555 }
556 } else if (command == "REMOVE") {
557 task(words[1]);
558 int count = 0;
559 for (auto position = dependencies.begin(); position != dependencies.end();) {
560 if (position->first.first == words[1] || position->first.second == words[1]) {
561 position = dependencies.erase(position);
562 ++count;
563 } else ++position;
564 }
565 tasks.erase(words[1]);
566 std::cout << "task " << words[1] << " removed with " << count << " dependencies\n";
567 } else if (command == "NOTBEFORE") {
568 task(words[1]).notBefore = std::stoi(words[2]);
569 std::cout << "notbefore " << words[1] << ' ' << words[2] << '\n';
570 } else if (command == "CRASHCOST") {
571 Task& item = task(words[1]);
572 int limit = std::stoi(words[2]), cost = std::stoi(words[3]);
573 if (limit >= item.base) fail("crash limit too large for " + words[1]);
574 if (limit < item.used) fail("crash limit below used days for " + words[1]);
575 item.crash = Crash{limit, cost};
576 std::cout << "crash " << words[1] << " max " << limit << " cost " << cost << '\n';
577 } else if (command == "UNCRASH") {
578 Task& item = task(words[1]);
579 std::cout << "uncrash " << words[1] << " restored " << item.used << " days\n";
580 item.used = 0;
581 } else if (command == "PERT") {
582 Task& item = task(words[1]);
583 int optimistic = std::stoi(words[2]), likely = std::stoi(words[3]), pessimistic = std::stoi(words[4]);
584 if (optimistic > likely || likely > pessimistic) fail("invalid estimate");
585 item.estimate = std::array<int, 3>{optimistic, likely, pessimistic};
586 std::cout << "pert " << words[1] << " expected " << decimal(optimistic + 4 * likely + pessimistic, 6)
587 << " variance " << decimal((pessimistic - optimistic) * (pessimistic - optimistic), 36) << '\n';
588 } else if (command == "CAPACITY") {
589 capacities[words[1]] = std::stoi(words[2]);
590 std::cout << "capacity " << words[1] << ' ' << words[2] << '\n';
591 } else if (command == "START" || command == "HOLIDAY" || command == "DEADLINE") {
592 int date = parseDate(words[1]);
593 if (command == "HOLIDAY") {
594 if (!holidays.insert(date).second) fail("duplicate holiday " + words[1]);
595 std::cout << "holiday ";
596 } else if (command == "START") {
597 startDate = date;
598 std::cout << "start ";
599 } else {
600 deadline = date;
601 std::cout << "deadline ";
602 }
603 std::cout << dateText(date) << '\n';
604 } else if (command == "LIST") {
605 std::cout << "tasks " << tasks.size() << ", dependencies " << dependencies.size() << ", holidays " << holidays.size() << '\n';
606 for (const auto& [name, item] : tasks) {
607 std::vector<std::string> predecessors;
608 for (const auto& [ends, link] : dependencies) {
609 if (ends.second == name) predecessors.push_back(ends.first);
610 }
611 std::cout << name << " duration " << item.duration() << " resource " << item.resource << " units " << item.units
612 << " preds " << (predecessors.empty() ? "none" : joined(predecessors)) << '\n';
613 }
614 } else if (command == "SHOW") {
615 const Task& item = task(words[1]);
616 std::cout << "task " << words[1] << " duration " << item.duration() << " resource " << item.resource << " units " << item.units;
617 if (item.notBefore != 0) std::cout << " notbefore " << item.notBefore;
618 std::cout << '\n';
619 for (bool predecessor : {true, false}) {
620 std::vector<std::string> links;
621 for (const auto& [ends, link] : dependencies) {
622 if ((predecessor ? ends.second : ends.first) == words[1]) links.push_back((predecessor ? ends.first : ends.second) + " " + link.kind + signedNumber(link.lag));
623 }
624 std::cout << (predecessor ? "preds: " : "succs: ") << (links.empty() ? "none" : joined(links, ", ")) << '\n';
625 }
626 if (item.crash) std::cout << "crash: max " << item.crash->limit << " used " << item.used << " cost " << item.crash->cost << '\n';
627 else std::cout << "crash: none\n";
628 if (item.estimate) std::cout << "pert: " << (*item.estimate)[0] << ' ' << (*item.estimate)[1] << ' ' << (*item.estimate)[2] << '\n';
629 else std::cout << "pert: none\n";
630 auto [order, cycle] = topology();
631 if (!cycle.empty()) std::cout << "schedule: unavailable\n";
632 else {
633 Timing time = compute(order).times.at(words[1]);
634 std::cout << "schedule: ES " << time.es << " EF " << time.ef << " LS " << time.ls << " LF " << time.lf
635 << " TF " << time.tf << " FF " << time.free << (time.tf == 0 ? " critical\n" : "\n");
636 }
637 } else if (command == "IMPACT") {
638 task(words[1]);
639 std::set<std::string> reached{words[1]};
640 std::vector<std::string> pending{words[1]};
641 for (std::size_t i = 0; i < pending.size(); ++i) {
642 for (const auto& [ends, link] : dependencies) {
643 if (ends.first == pending[i] && reached.insert(ends.second).second) pending.push_back(ends.second);
644 }
645 }
646 reached.erase(words[1]);
647 std::vector<std::string> names(reached.begin(), reached.end());
648 std::cout << "impact " << words[1] << ": " << (names.empty() ? "none" : joined(names)) << '\n';
649 } else analyze(words);
650 }
651};
