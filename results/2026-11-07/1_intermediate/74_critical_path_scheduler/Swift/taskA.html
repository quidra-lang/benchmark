20func leap(_ y: Int) -> Bool {
21 y%4 == 0 && (y%100 != 0 || y%400 == 0)
22}

24func monthDays(_ y: Int,_ m: Int) -> Int {
25 m == 2 && leap(y) ? 29 : [31,28,31,30,31,30,31,31,30,31,30,31][m-1]
26}

28func date(_ text: String) throws -> Int {
29 let p = text.split(separator: "-").map {
30 Int($0)!
31 }
32 let y = p[0],m = p[1],d = p[2]
33 if m < 1 || m > 12 || d < 1 || d > monthDays(y,m) {
34 try fail("invalid date \(text)")
35 }
36 var day = 0
37 for year in 2000..<y {
38 day += leap(year) ? 366 : 365
39 }
40 if m > 1 {
41 for month in 1..<m {
42 day += monthDays(y,month)
43 }
44 }
45 return day+d-1
46}

48func dateText(_ day: Int) -> String {
49 var left = day
50 var y = 2000
51 while left >= (leap(y) ? 366 : 365) {
52 left -= leap(y) ? 366 : 365
53 y += 1
54 }
55 var m = 1
56 while left >= monthDays(y,m) {
57 left -= monthDays(y,m)
58 m += 1
59 }
60 return String(format: "%04d-%02d-%02d",y,m,left+1)+" "+["Mon","Tue","Wed","Thu","Fri","Sat","Sun"][(day+5)%7]
61}

114 func check(_ n: String) throws {
115 if tasks[n] == nil {
116 try fail("unknown task \(n)")
117 }
118 }
119 func preds(_ n: String) -> [Dep] {
120 deps.filter {
121 $0.b == n
122 }.sorted {
123 $0.a < $1.a
124 }
125 }
126 func succs(_ n: String) -> [Dep] {
127 deps.filter {
128 $0.a == n
129 }.sorted {
130 $0.b < $1.b
131 }
132 }
133 func order() -> ([String],String?) {
134 var remaining = Set(names)
135 var result = [String]()
136 while let n = remaining.sorted().first(where: {
137 t in preds(t).allSatisfy {
138 !remaining.contains($0.a)
139 }
140 }) {
141 remaining.remove(n)
142 result.append(n)
143 }
144 if remaining.isEmpty {
145 return (result,nil)
146 }
147 var walked = [remaining.sorted().first!]
148 while true {
149 let n = preds(walked.last!).first {
150 remaining.contains($0.a)
151 }!.a
152 if let q = walked.firstIndex(of: n) {
153 let cycle = [n]+Array(walked.dropFirst(q+1).reversed())
154 let k = cycle.indices.min {
155 cycle[$0] < cycle[$1]
156 }!
157 let rotated = Array(cycle.dropFirst(k))+Array(cycle.prefix(k))
158 return (result,(rotated+[rotated.first!]).joined(separator: " -> "))
159 }
160 walked.append(n)
161 }
162 }
163 func schedule(_ changed: String? = nil,_ duration: Int = 0) -> Schedule {
164 let order = order().0
165 var times = [String:Times]()
166 for n in order {
167 let t = tasks[n]!
168 let d = n == changed ? duration : t.duration
169 var es = t.notBefore
170 for e in preds(n) {
171 let p = times[e.a]!
172 es = max(es,e.kind == "FS" ? p.ef+e.lag : e.kind == "SS" ? p.es+e.lag : p.ef+e.lag-d)
173 }
174 times[n] = Times(duration: d,es: es,ef: es+d)
175 }
176 let total = times.values.map(\.ef).max()!
177 for n in order.reversed() {
178 var t = times[n]!
179 var lf = total
180 for e in succs(n) {
181 let p = times[e.b]!
182 lf = min(lf,e.kind == "FS" ? p.ls-e.lag : e.kind == "SS" ? p.ls-e.lag+t.duration : p.lf-e.lag)
183 }
184 t.lf = lf
185 t.ls = lf-t.duration
186 t.tf = t.ls-t.es
187 times[n] = t
188 }
189 let s = Schedule(times: times,total: total)
190 for n in order {
191 var t = times[n]!
192 t.ff = min(total-t.ef,succs(n).map {
193 margin($0,s)
194 }.min() ?? Int.max)
195 times[n] = t
196 }
197 return Schedule(times: times,total: total)
198 }
199 func margin(_ e: Dep,_ s: Schedule) -> Int {
200 let a = s.times[e.a]!,b = s.times[e.b]!
201 return e.kind == "FS" ? b.es-a.ef-e.lag : e.kind == "SS" ? b.es-a.es-e.lag : b.ef-a.ef-e.lag
202 }
203 func critical(_ s: Schedule) -> [String] {
204 names.filter {
205 s.times[$0]!.tf == 0
206 }
207 }
208 func paths(_ s: Schedule) -> [[String]] {
209 let links = deps.filter {
210 s.times[$0.a]!.tf == 0 && s.times[$0.b]!.tf == 0 && margin($0,s) == 0
211 }
212 var result = [[String]]()
213 func visit(_ p: [String]) {
214 let next = links.filter {
215 $0.a == p.last!
216 }.map(\.b).sorted()
217 if next.isEmpty {
218 result.append(p)
219 } else {
220 for n in next {
221 visit(p+[n])
222 }
223 }
224 }
225 for n in critical(s) where !links.contains(where: {
226 $0.b == n
227 }) {
228 visit([n])
229 }
230 return result
231 }
232 func working(_ d: Int) -> Bool {
233 (d+5)%7 < 5 && !holidays.contains(d)
234 }
235 func usage(_ resource: String,_ times: [String:(Int,Int)],_ total: Int) -> [Int] {
236 (0..<total).map {
237 day in tasks.reduce(0) {
238 sum,pair in let (n,t) = pair
239 let interval = times[n]!
240 return sum+(t.resource == resource && day >= interval.0 && day < interval.1 ? t.units : 0)
241 }
242 }
243 }
244 func command(_ w: [String]) throws -> [String] {
245 let c = w[0]
246 let a = Array(w.dropFirst())
247 func num(_ i: Int) -> Int {
248 Int(a[i])!
249 }
250 var out = [String]()
251 switch c {
252 case "TASK":
253 if tasks[a[0]] != nil {
254 try fail("task \(a[0]) already exists")
255 }
256 tasks[a[0]] = Task(base: num(1),resource: a[2],units: num(3))
257 return ["task \(a[0]) added"]
258 case "DEP","UNDEP":
259 try check(a[0])
260 try check(a[1])
261 let i = deps.firstIndex {
262 $0.a == a[0] && $0.b == a[1]
263 }
264 if c == "DEP" {
265 if a[0] == a[1] {
266 try fail("self dependency \(a[0])")
267 }
268 if i != nil {
269 try fail("dependency \(a[0]) -> \(a[1]) already exists")
270 }
271 deps.append(Dep(a: a[0],b: a[1],kind: a[2],lag: num(3)))
272 return ["dependency \(a[0]) -> \(a[1]) \(a[2])\(signed(num(3))) added"]
273 }
274 guard let i = i else {
275 try fail("no dependency \(a[0]) -> \(a[1])")
276 }
277 deps.remove(at: i)
278 return ["dependency \(a[0]) -> \(a[1]) removed"]
279 case "REMOVE":
280 try check(a[0])
281 let k = deps.filter {
282 $0.a == a[0] || $0.b == a[0]
283 }.count
284 deps.removeAll {
285 $0.a == a[0] || $0.b == a[0]
286 }
287 tasks.removeValue(forKey: a[0])
288 return ["task \(a[0]) removed with \(k) dependencies"]
289 case "NOTBEFORE":
290 try check(a[0])
291 tasks[a[0]]!.notBefore = num(1)
292 return ["notbefore \(a[0]) \(a[1])"]
293 case "CRASHCOST":
294 try check(a[0])
295 let t = tasks[a[0]]!
296 if num(1) >= t.base {
297 try fail("crash limit too large for \(a[0])")
298 }
299 if num(1) < t.used {
300 try fail("crash limit below used days for \(a[0])")
301 }
302 tasks[a[0]]!.crash = (num(1),num(2))
303 return ["crash \(a[0]) max \(a[1]) cost \(a[2])"]
304 case "UNCRASH":
305 try check(a[0])
306 out.append("uncrash \(a[0]) restored \(tasks[a[0]]!.used) days")
307 tasks[a[0]]!.used = 0
308 return out
309 case "PERT":
310 try check(a[0])
311 let o = num(1),m = num(2),p = num(3)
312 if o > m || m > p {
313 try fail("invalid estimate")
314 }
315 tasks[a[0]]!.pert = [o,m,p]
316 return ["pert \(a[0]) expected \(decimal(o+4*m+p,6)) variance \(decimal((p-o)*(p-o),36))"]
317 case "CAPACITY":
318 capacities[a[0]] = num(1)
319 return ["capacity \(a[0]) \(a[1])"]
320 case "START","HOLIDAY","DEADLINE":
321 let d = try date(a[0])
322 if c == "HOLIDAY" && holidays.contains(d) {
323 try fail("duplicate holiday \(a[0])")
324 }
325 if c == "START" {
326 start = d
327 } else if c == "DEADLINE" {
328 deadline = d
329 } else {
330 holidays.insert(d)
331 }
332 return ["\(c.lowercased()) \(dateText(d))"]
333 case "LIST":
334 out.append("tasks \(tasks.count), dependencies \(deps.count), holidays \(holidays.count)")
335 for n in names {
336 let t = tasks[n]!
337 let p = preds(n).map(\.a).joined(separator: " ")
338 out.append("\(n) duration \(t.duration) resource \(t.resource) units \(t.units) preds \(p.isEmpty ? "none" : p)")
339 }
340 return out
341 case "SHOW":
342 let n = a[0]
343 try check(n)
344 let t = tasks[n]!
345 out.append("task \(n) duration \(t.duration) resource \(t.resource) units \(t.units)"+(t.notBefore != 0 ? " notbefore \(t.notBefore)" : ""))
346 let p = preds(n).map {
347 "\($0.a) \($0.kind)\(signed($0.lag))"
348 }.joined(separator: ", ")
349 let q = succs(n).map {
350 "\($0.b) \($0.kind)\(signed($0.lag))"
351 }.joined(separator: ", ")
352 out.append("preds: "+(p.isEmpty ? "none" : p))
353 out.append("succs: "+(q.isEmpty ? "none" : q))
354 out.append(t.crash.map {
355 "crash: max \($0.0) used \(t.used) cost \($0.1)"
356 } ?? "crash: none")
357 out.append(t.pert.map {
358 "pert: "+$0.map(String.init).joined(separator: " ")
359 } ?? "pert: none")
360 if order().1 != nil {
361 out.append("schedule: unavailable")
362 } else {
363 let x = schedule().times[n]!
364 out.append("schedule: ES \(x.es) EF \(x.ef) LS \(x.ls) LF \(x.lf) TF \(x.tf) FF \(x.ff)"+(x.tf == 0 ? " critical" : ""))
365 }
366 return out
367 case "IMPACT":
368 try check(a[0])
369 var reached: Set<String> = [a[0]]
370 while true {
371 let before = reached.count
372 for e in deps where reached.contains(e.a) {
373 reached.insert(e.b)
374 }
375 if reached.count == before {
376 break
377 }
378 }
379 reached.remove(a[0])
380 return ["impact \(a[0]): "+(reached.isEmpty ? "none" : reached.sorted().joined(separator: " "))]
381 default:
382 break
383 }

385 try fail("no tasks")
386 }
387 if let cycle = order().1 {
388 try fail("cycle "+cycle)

391 case "CALENDAR":
392 if start == nil {
393 try fail("no start date")
394 }
395 case "LEVEL":

397 let t = tasks[n]!

457 case "DRAG":

459 out.append("\(n) drag \(s.total-schedule(n,0).total)")

464 $0+$1.tf
465 }
466 out.append("tasks \(tasks.count), critical \(count), dependencies \(deps.count) (FS \(deps.filter { $0.kind == "FS" }.count), SS \(deps.filter { $0.kind == "SS" }.count), FF \(deps.filter { $0.kind == "FF" }.count))")
467 out.append("work "+resources.map {
468 r in "\(r) \(tasks.values.filter { $0.resource == r }.reduce(0) { $0+$1.duration*$1.units })"
469 }.joined(separator: ", "))

484 }

591 }

594 }!

596 if cost > remaining {

599 }

603 count += 1

605 }

607 case "WHATIF":

614 }

616 out.append(n+fields.map {
617 " \($0.0) \($0.1)->\($0.2)"
618 }.joined())
619 }
620 }

622 if beforeCritical != afterCritical {
623 out.append("critical "+beforeCritical.joined(separator: " ")+" -> "+afterCritical.joined(separator: " "))
624 }
625 case "BASELINE":
626 baseline = Baseline(total: s.total,times: initialTimes)
627 out.append("baseline saved: \(tasks.count) tasks, duration \(s.total)")
628 case "VARIANCE":
629 let old = baseline!
630 out.append("variance: duration \(old.total) -> \(s.total) (\(signed(s.total-old.total)))")
631 for n in Set(old.times.keys).union(tasks.keys).sorted() {
632 if let x = old.times[n] {
633 if let y = initialTimes[n] {
634 if x.0 != y.0 || x.1 != y.1 {
635 out.append("\(n) start \(signed(y.0-x.0)) finish \(signed(y.1-x.1))")
636 }
637 } else {
638 out.append("\(n) removed")
639 }
640 } else {
641 out.append("\(n) new")
642 }
643 }
644 if out.count == 1 {
645 out.append("no task changes")
646 }
647 default:
648 break
649 }
650 return out
651 }
