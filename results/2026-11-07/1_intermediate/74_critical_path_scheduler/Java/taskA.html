83 List<String> topological() {
84 List<String> result = new ArrayList<>();
85 Set<String> remaining = new TreeSet<>(tasks.keySet());
86 while (true) {
87 String next = null;
88 for (String name : remaining) if (incoming(name).stream().noneMatch(d -> remaining.contains(d.a))) {
89 next = name;
90 break;
91 }
92 if (next == null) break;
93 result.add(next);
94 remaining.remove(next);
95 }
96 return result;
97 }

99 String cycle() {
100 TreeSet<String> remaining = new TreeSet<>(tasks.keySet());
101 remaining.removeAll(topological());
102 if (remaining.isEmpty()) return null;
103 List<String> chain = new ArrayList<>();
104 String current = remaining.first();
105 while (!chain.contains(current)) {
106 chain.add(current);
107 current = incoming(current).stream().filter(d -> remaining.contains(d.a)).findFirst().orElseThrow().a;
108 }
109 int begin = chain.indexOf(current);
110 List<String> cycle = new ArrayList<>(List.of(current));
111 for (int i = chain.size() - 1; i > begin; i--) cycle.add(chain.get(i));
112 int minimum = cycle.indexOf(Collections.min(cycle));
113 Collections.rotate(cycle, -minimum);
114 cycle.add(cycle.getFirst());
115 return String.join(" -> ", cycle);
116 }

118 Schedule schedule(String changed, int duration) {
119 Schedule s = new Schedule();
120 List<String> order = topological();
121 for (String name : order) {
122 Task t = tasks.get(name);
123 Timing x = new Timing();
124 x.duration = name.equals(changed) ? duration : t.duration();
125 x.es = t.notBefore;
126 for (Dependency d : incoming(name)) {
127 Timing a = s.times.get(d.a);
128 int bound = switch (d.kind) {
129 case "FS" -> a.ef + d.lag;
130 case "SS" -> a.es + d.lag;
131 default -> a.ef + d.lag - x.duration;
132 };
133 x.es = Math.max(x.es, bound);
134 }
135 x.ef = x.es + x.duration;
136 s.total = Math.max(s.total, x.ef);
137 s.times.put(name, x);
138 }
139 for (int i = order.size() - 1; i >= 0; i--) {
140 String name = order.get(i);
141 Timing x = s.times.get(name);
142 x.lf = s.total;
143 for (Dependency d : outgoing(name)) {
144 Timing b = s.times.get(d.b);
145 int bound = switch (d.kind) {
146 case "FS" -> b.ls - d.lag;
147 case "SS" -> b.ls - d.lag + x.duration;
148 default -> b.lf - d.lag;
149 };
150 x.lf = Math.min(x.lf, bound);
151 }
152 x.ls = x.lf - x.duration;
153 x.tf = x.ls - x.es;
154 x.free = s.total - x.ef;
155 for (Dependency d : outgoing(name)) x.free = Math.min(x.free, margin(s, d));
156 }
157 return s;
158 }
159
160 int margin(Schedule s, Dependency d) {
161 Timing a = s.times.get(d.a), b = s.times.get(d.b);
162 return (d.kind.equals("FF") ? b.ef : b.es) - (d.kind.equals("SS") ? a.es : a.ef) - d.lag;
163 }

165 List<List<String>> criticalPaths(Schedule s) {
166 List<List<String>> paths = new ArrayList<>();
167 Map<String, List<String>> successors = new TreeMap<>();
168 Set<String> hasPredecessor = new HashSet<>();
169 for (String name : s.critical()) successors.put(name, new ArrayList<>());
170 for (Dependency d : dependencies) if (successors.containsKey(d.a) && successors.containsKey(d.b) && margin(s, d) == 0) {
171 successors.get(d.a).add(d.b);
172 hasPredecessor.add(d.b);
173 }
174 for (List<String> list : successors.values()) Collections.sort(list);
175 class Walk {
176 void visit(String name, List<String> path) {
177 path.add(name);
178 if (successors.get(name).isEmpty()) paths.add(new ArrayList<>(path));
179 else for (String next : successors.get(name)) visit(next, path);
180 path.removeLast();
181 }
182 }
183 Walk walk = new Walk();
184 for (String name : successors.keySet()) if (!hasPredecessor.contains(name)) walk.visit(name, new ArrayList<>());
185 return paths;
186 }

189 try {
190 return LocalDate.of(Integer.parseInt(text.substring(0, 4)), Integer.parseInt(text.substring(5, 7)), Integer.parseInt(text.substring(8, 10)));

209 List<LocalDate> calendar(int total) {
210 List<LocalDate> days = new ArrayList<>();
211 for (LocalDate d = start; days.size() < total; d = d.plusDays(1)) if (working(d)) days.add(d);

215 int workingBetween(LocalDate a, LocalDate b) {
216 int count = 0;
217 for (LocalDate d = a.plusDays(1); !d.isAfter(b); d = d.plusDays(1)) if (working(d)) count++;
218 return count;

231 List<String> command(String line) {
232 String[] w = line.split(" ");
233 String c = w[0];
234 List<String> out = new ArrayList<>();
235 Set<String> analyses = Set.of("CPM", "CRITICAL", "SLACK", "DRAG", "STATS", "CALENDAR", "GANTT", "LEVEL", "LOAD", "PERTSUM", "CRASH", "WHATIF", "BASELINE", "VARIANCE");
236 if (analyses.contains(c)) {
237 if (tasks.isEmpty()) fail("no tasks");
238 String cycle = cycle();
239 if (cycle != null) fail("cycle " + cycle);
240 }
241 switch (c) {
242 case "TASK" -> {
243 if (tasks.containsKey(w[1])) fail("task " + w[1] + " already exists");
244 tasks.put(w[1], new Task(w[1], Integer.parseInt(w[2]), w[3], Integer.parseInt(w[4])));
245 out.add("task " + w[1] + " added");
246 }
247 case "DEP", "UNDEP" -> {
248 task(w[1]);
249 task(w[2]);
250 Dependency old = dependencies.stream().filter(d -> d.a.equals(w[1]) && d.b.equals(w[2])).findFirst().orElse(null);
251 if (c.equals("DEP")) {
252 if (w[1].equals(w[2])) fail("self dependency " + w[1]);
253 if (old != null) fail("dependency " + w[1] + " -> " + w[2] + " already exists");
254 int lag = Integer.parseInt(w[4]);
255 dependencies.add(new Dependency(w[1], w[2], w[3], lag));
256 out.add("dependency " + w[1] + " -> " + w[2] + " " + w[3] + signed(lag) + " added");
257 } else {
258 if (old == null) fail("no dependency " + w[1] + " -> " + w[2]);
259 dependencies.remove(old);
260 out.add("dependency " + w[1] + " -> " + w[2] + " removed");
261 }
262 }
263 case "REMOVE" -> {
264 task(w[1]);
265 int before = dependencies.size();
266 dependencies.removeIf(d -> d.a.equals(w[1]) || d.b.equals(w[1]));
267 tasks.remove(w[1]);
268 out.add("task " + w[1] + " removed with " + (before - dependencies.size()) + " dependencies");
269 }
270 case "NOTBEFORE", "CRASHCOST", "UNCRASH", "PERT" -> {
271 Task t = task(w[1]);
272 if (c.equals("NOTBEFORE")) {
273 t.notBefore = Integer.parseInt(w[2]);
274 out.add("notbefore " + t.name + " " + t.notBefore);
275 } else if (c.equals("CRASHCOST")) {
276 int limit = Integer.parseInt(w[2]);
277 if (limit >= t.base) fail("crash limit too large for " + t.name);
278 if (limit < t.crashed) fail("crash limit below used days for " + t.name);
279 t.crashLimit = limit;
280 t.crashCost = Integer.parseInt(w[3]);
281 out.add("crash " + t.name + " max " + limit + " cost " + t.crashCost);
282 } else if (c.equals("UNCRASH")) {
283 out.add("uncrash " + t.name + " restored " + t.crashed + " days");
284 t.crashed = 0;
285 } else {
286 int o = Integer.parseInt(w[2]), m = Integer.parseInt(w[3]), p = Integer.parseInt(w[4]);
287 if (o > m || m > p) fail("invalid estimate");
288 t.estimate = new int[]{o, m, p};
289 out.add("pert " + t.name + " expected " + decimal(o + 4L * m + p, 6) + " variance " + decimal((long)(p - o) * (p - o), 36));
290 }
291 }
292 case "CAPACITY" -> {
293 capacities.put(w[1], Integer.parseInt(w[2]));
294 out.add("capacity " + w[1] + " " + w[2]);
295 }
296 case "START", "HOLIDAY", "DEADLINE" -> {
297 LocalDate d = parseDate(w[1]);
298 if (c.equals("HOLIDAY") && holidays.contains(d)) fail("duplicate holiday " + w[1]);
299 if (c.equals("START")) start = d;
300 else if (c.equals("DEADLINE")) deadline = d;
301 else holidays.add(d);
302 out.add(c.toLowerCase(Locale.ROOT) + " " + date(d));
303 }
304 case "LIST" -> {
305 out.add("tasks " + tasks.size() + ", dependencies " + dependencies.size() + ", holidays " + holidays.size());
306 for (Task t : tasks.values()) out.add(t.name + " duration " + t.duration() + " resource " + t.resource + " units " + t.units + " preds " + names(incoming(t.name).stream().map(Dependency::a).toList()));
307 }
308 case "SHOW" -> {
309 Task t = task(w[1]);
310 out.add("task " + t.name + " duration " + t.duration() + " resource " + t.resource + " units " + t.units + (t.notBefore == 0 ? "" : " notbefore " + t.notBefore));
311 for (boolean predecessors : new boolean[]{true, false}) {
312 List<String> parts = new ArrayList<>();
313 for (Dependency d : predecessors ? incoming(t.name) : outgoing(t.name)) parts.add((predecessors ? d.a : d.b) + " " + d.kind + signed(d.lag));
314 out.add((predecessors ? "preds: " : "succs: ") + (parts.isEmpty() ? "none" : String.join(", ", parts)));
315 }
316 out.add(t.crashLimit == null ? "crash: none" : "crash: max " + t.crashLimit + " used " + t.crashed + " cost " + t.crashCost);
317 out.add(t.estimate == null ? "pert: none" : "pert: " + t.estimate[0] + " " + t.estimate[1] + " " + t.estimate[2]);
318 if (cycle() != null) out.add("schedule: unavailable");
319 else {
320 Timing x = schedule(null, 0).times.get(t.name);
321 out.add("schedule: ES " + x.es + " EF " + x.ef + " LS " + x.ls + " LF " + x.lf + " TF " + x.tf + " FF " + x.free + (x.tf == 0 ? " critical" : ""));
322 }
323 }
324 case "IMPACT" -> {
325 task(w[1]);
326 Set<String> seen = new TreeSet<>();
327 ArrayDeque<String> queue = new ArrayDeque<>(List.of(w[1]));
328 seen.add(w[1]);
329 while (!queue.isEmpty()) for (Dependency d : outgoing(queue.remove())) if (seen.add(d.b)) queue.add(d.b);
330 seen.remove(w[1]);
331 out.add("impact " + w[1] + ": " + names(seen));
332 }
333 default -> {
334 if (c.equals("CALENDAR") && start == null) fail("no start date");
335 if (c.equals("WHATIF")) task(w[1]);
336 if (c.equals("VARIANCE") && baseline == null) fail("no baseline");
337 if (c.equals("LOAD") && tasks.values().stream().noneMatch(t -> t.resource.equals(w[1]))) fail("unknown resource " + w[1]);
338 if (c.equals("LEVEL")) for (Task t : tasks.values()) {
339 int cap = capacities.getOrDefault(t.resource, Integer.parseInt(w[1]));
340 if (t.units > cap) fail("capacity " + cap + " too small for " + t.name);
341 }
342 Schedule s = schedule(null, 0);
343 switch (c) {
344 case "CPM" -> {
345 int width = Math.max(4, tasks.lastKey().length());
346 for (String name : tasks.keySet()) width = Math.max(width, name.length());
347 out.add(String.format(Locale.ROOT, "%-" + width + "s%5s%5s%5s%5s%5s%5s%5s", "TASK", "DUR", "ES", "EF", "LS", "LF", "TF", "FF"));
348 for (var e : s.times.entrySet()) {
349 Timing x = e.getValue();
350 out.add(String.format(Locale.ROOT, "%-" + width + "s%5d%5d%5d%5d%5d%5d%5d", e.getKey(), x.duration, x.es, x.ef, x.ls, x.lf, x.tf, x.free) + (x.tf == 0 ? " *" : ""));
351 }
352 out.add("project duration " + s.total);
353 }
354 case "CRITICAL", "PERTSUM" -> {
355 List<List<String>> paths = criticalPaths(s);
356 if (c.equals("CRITICAL")) out.add("critical paths: " + paths.size());
357 for (List<String> path : paths) {
358 String text = String.join(" -> ", path);
359 if (c.equals("PERTSUM")) {
360 long expected = 0, variance = 0;
361 for (String name : path) {
362 Task t = tasks.get(name);
363 int d = t.duration();
364 int[] p = t.estimate == null ? new int[]{d, d, d} : t.estimate;
365 expected += p[0] + 4L * p[1] + p[2];
366 variance += (long)(p[2] - p[0]) * (p[2] - p[0]);
367 }
368 text += ": expected " + decimal(expected, 6) + " variance " + decimal(variance, 36);
369 }
370 out.add(text);
371 }
372 }
373 case "SLACK" -> {
374 TreeMap<Integer, List<String>> groups = new TreeMap<>();
375 for (var e : s.times.entrySet()) groups.computeIfAbsent(e.getValue().tf, k -> new ArrayList<>()).add(e.getKey());
376 for (var e : groups.entrySet()) out.add("float " + e.getKey() + ": " + names(e.getValue()));
377 }
378 case "DRAG" -> {
379 for (String name : s.critical()) out.add(name + " drag " + (s.total - schedule(name, 0).total));
380 }
381 case "STATS" -> {
382 int critical = s.critical().size(), totalFloat = s.times.values().stream().mapToInt(x -> x.tf).sum();
383 out.add("tasks " + tasks.size() + ", critical " + critical + ", dependencies " + dependencies.size() + " (FS " + dependencies.stream().filter(d -> d.kind.equals("FS")).count() + ", SS " + dependencies.stream().filter(d -> d.kind.equals("SS")).count() + ", FF " + dependencies.stream().filter(d -> d.kind.equals("FF")).count() + ")");
384 TreeMap<String, Integer> work = new TreeMap<>();
385 for (Task t : tasks.values()) work.merge(t.resource, t.duration() * t.units, Integer::sum);
386 out.add("work " + String.join(", ", work.entrySet().stream().map(e -> e.getKey() + " " + e.getValue()).toList()));
387 out.add("float total " + totalFloat + ", average " + decimal(totalFloat, tasks.size()));
388 out.add("duration " + s.total + ", critical share " + decimal(100L * critical, tasks.size()) + "%");
389 }
390 case "CALENDAR" -> {
391 List<LocalDate> days = calendar(s.total);
392 for (String name : startOrder(s)) {
393 Timing x = s.times.get(name);
394 out.add(name + " " + date(days.get(x.es)) + " - " + date(days.get(x.ef - 1)));
395 }
396 LocalDate first = days.getFirst(), last = days.getLast();
397 out.add("project " + date(first) + " - " + date(last) + ", " + s.total + " working days, " + (last.toEpochDay() - first.toEpochDay() + 1) + " calendar days");
398 if (deadline != null) out.add("deadline " + date(deadline) + (!last.isAfter(deadline) ? " met, " + workingBetween(last, deadline) + " working days to spare" : " missed by " + workingBetween(deadline, last) + " working days"));
399 List<String> span = holidays.stream().filter(d -> weekday(d) && !d.isBefore(first) && !d.isAfter(last)).map(Main::date).toList();
400 out.add("holidays in span: " + (span.isEmpty() ? "none" : String.join(", ", span)));
401 }
402 case "GANTT" -> {
403 if (s.total > 60) fail("chart too wide (" + s.total + " days)");
404 int width = tasks.keySet().stream().mapToInt(String::length).max().orElseThrow();
405 StringBuilder axis = new StringBuilder(" ".repeat(width) + " |");
406 for (int k = 0; k < s.total; k++) axis.append(k % 10);
407 out.add(axis + "|");
408 for (String name : startOrder(s)) {
409 Timing x = s.times.get(name);
410 StringBuilder bar = new StringBuilder(String.format(Locale.ROOT, "%-" + width + "s |", name));
411 for (int k = 0; k < s.total; k++) bar.append(x.es <= k && k < x.ef ? (x.tf == 0 ? '=' : '-') : x.ef <= k && k < x.lf ? '.' : ' ');
412 out.add(bar + "|");
413 }
414 }
415 case "LEVEL" -> out.addAll(level(s, Integer.parseInt(w[1])));
416 case "LOAD" -> {
417 int[] load = usage(s).get(w[1]);
418 int peak = Arrays.stream(load).max().orElseThrow(), day = 0;
419 while (load[day] != peak) day++;
420 out.add("load " + w[1] + ": " + String.join(" ", Arrays.stream(load).mapToObj(Integer::toString).toList()));
421 out.add("peak " + peak + " on day " + day);
422 if (capacities.containsKey(w[1])) out.add("capacity " + capacities.get(w[1]) + " exceeded on " + Arrays.stream(load).filter(x -> x > capacities.get(w[1])).count() + " days");
423 }
424 case "CRASH" -> {
425 int remaining = Integer.parseInt(w[1]), initial = s.total, steps = 0;
426 String reason;
427 while (true) {
428 Task candidate = null;
429 for (String name : s.critical()) {
430 Task t = tasks.get(name);
431 if (t.crashLimit != null && t.crashed < t.crashLimit && schedule(name, t.duration() - 1).total < s.total && (candidate == null || t.crashCost < candidate.crashCost)) candidate = t;
432 }
433 if (candidate == null) {
434 reason = "no candidate";
435 break;
436 }
437 if (candidate.crashCost > remaining) {
438 reason = "insufficient budget";
439 break;
440 }
441 int before = candidate.duration();
442 candidate.crashed++;
443 remaining -= candidate.crashCost;
444 s = schedule(null, 0);
445 out.add("step " + (++steps) + ": " + candidate.name + " " + before + " -> " + candidate.duration() + ", cost " + candidate.crashCost + ", project " + s.total);
446 }
447 out.add("crash done: " + steps + " days, cost " + (Integer.parseInt(w[1]) - remaining) + ", project " + initial + " -> " + s.total + ", budget left " + remaining + ", " + reason);
448 }
449 case "WHATIF" -> {
450 Schedule after = schedule(w[1], Integer.parseInt(w[2]));
451 out.add("whatif " + w[1] + " " + tasks.get(w[1]).duration() + " -> " + w[2] + ": project " + s.total + " -> " + after.total);
452 for (String name : tasks.keySet()) {
453 Timing a = s.times.get(name), b = after.times.get(name);
454 String text = name + (a.es != b.es ? " ES " + a.es + "->" + b.es : "") + (a.ef != b.ef ? " EF " + a.ef + "->" + b.ef : "") + (a.tf != b.tf ? " TF " + a.tf + "->" + b.tf : "");
455 if (!text.equals(name)) out.add(text);
456 }
457 if (!s.critical().equals(after.critical())) out.add("critical " + names(s.critical()) + " -> " + names(after.critical()));
458 }
459 case "BASELINE" -> {
460 Map<String, int[]> times = new TreeMap<>();
461 for (var e : s.times.entrySet()) times.put(e.getKey(), new int[]{e.getValue().es, e.getValue().ef});
462 baseline = new Baseline(s.total, times);
463 out.add("baseline saved: " + tasks.size() + " tasks, duration " + s.total);
464 }
465 case "VARIANCE" -> {
466 out.add("variance: duration " + baseline.total + " -> " + s.total + " (" + signed(s.total - baseline.total) + ")");
467 TreeSet<String> union = new TreeSet<>(tasks.keySet());
468 union.addAll(baseline.times.keySet());
469 for (String name : union) {
470 if (!tasks.containsKey(name)) out.add(name + " removed");
471 else if (!baseline.times.containsKey(name)) out.add(name + " new");
472 else {
473 Timing x = s.times.get(name);
474 int[] old = baseline.times.get(name);
475 if (x.es != old[0] || x.ef != old[1]) out.add(name + " start " + signed(x.es - old[0]) + " finish " + signed(x.ef - old[1]));
476 }
477 }
478 if (out.size() == 1) out.add("no task changes");
479 }
480 default -> throw new IllegalStateException();
481 }
482 }
483 }
484 return out;
485 }

491 List<String> level(Schedule s, int limit) {
492 Map<String, int[]> decided = new HashMap<>();
493 TreeMap<String, Map<Integer, Integer>> loads = new TreeMap<>();
494 for (Task t : tasks.values()) loads.computeIfAbsent(t.resource, r -> new HashMap<>());
495 List<String> order = new ArrayList<>();
496 int total = 0;
497 while (decided.size() < tasks.size()) {
498 String name = tasks.keySet().stream().filter(n -> !decided.containsKey(n) && incoming(n).stream().allMatch(d -> decided.containsKey(d.a))).min(Comparator.<String>comparingInt(n -> s.times.get(n).ls).thenComparing(Comparator.naturalOrder())).orElseThrow();
499 Task t = tasks.get(name);
500 int begin = t.notBefore, duration = t.duration(), capacity = capacities.getOrDefault(t.resource, limit);
501 for (Dependency d : incoming(name)) {
502 int[] a = decided.get(d.a);
503 begin = Math.max(begin, (d.kind.equals("SS") ? a[0] : a[1]) + d.lag - (d.kind.equals("FF") ? duration : 0));
504 }
505 Map<Integer, Integer> load = loads.get(t.resource);
506 while (true) {
507 boolean fits = true;
508 for (int k = begin; k < begin + duration; k++) if (load.getOrDefault(k, 0) + t.units > capacity) {
509 fits = false;
510 break;
511 }
512 if (fits) break;
513 begin++;
514 }
515 int finish = begin + duration;
516 for (int k = begin; k < finish; k++) load.merge(k, t.units, Integer::sum);
517 decided.put(name, new int[]{begin, finish});
518 order.add(name);
519 total = Math.max(total, finish);
520 }
521 List<String> out = new ArrayList<>(List.of("level capacity " + limit + ": duration " + s.total + " -> " + total));
522 for (String name : order) {
523 int[] t = decided.get(name);
524 out.add(name + " start " + t[0] + " finish " + t[1] + " shift " + (t[0] - s.times.get(name).es));
525 }
526 TreeMap<String, int[]> original = usage(s);
527 for (String resource : loads.keySet()) out.add("resource " + resource + " capacity " + capacities.getOrDefault(resource, limit) + " peak " + Arrays.stream(original.get(resource)).max().orElseThrow() + " -> " + loads.get(resource).values().stream().mapToInt(Integer::intValue).max().orElse(0));
528 return out;
529 }
