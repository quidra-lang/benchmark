4Task = Struct.new(:base, :resource, :units, :day, :crash_limit, :cost, :used, :estimate) do
5 def duration
6 base - used
7 end
8end

10class ScheduleError < StandardError; end
11
12class Scheduler
13 ANALYSIS = %w[CPM CRITICAL SLACK DRAG STATS CALENDAR GANTT LEVEL LOAD PERTSUM CRASH WHATIF BASELINE VARIANCE].freeze
14
15 def initialize
16 @tasks = {}
17 @deps = {}
18 @capacities = {}
19 @start = nil
20 @deadline = nil
21 @holidays = Set.new
22 @baseline = nil
23 end
24
25 def require_task(name)
26 raise ScheduleError, "unknown task #{name}" unless @tasks.key?(name)
27 end
28
29 def predecessors(name)
30 @deps.keys.select { |a, b| b == name }.map(&:first).sort
31 end
32
33 def successors(name)
34 @deps.keys.select { |a, b| a == name }.map(&:last).sort
35 end
36
37 def topology
38 remaining = @tasks.keys.sort
39 order = []
40 loop do
41 name = remaining.find { |t| (predecessors(t) & remaining).empty? }
42 break unless name
43 remaining.delete(name)
44 order << name
45 end
46 return [order, nil] if remaining.empty?
47 walk = [remaining.first]
48 loop do
49 following = (predecessors(walk.last) & remaining).min
50 if (i = walk.index(following))
51 cycle = [walk[i]] + walk[(i + 1)..].reverse
52 cycle = cycle.rotate(cycle.index(cycle.min))
53 return [order, (cycle + [cycle.first]).join(' -> ')]
54 end
55 walk << following
56 end
57 end
58
59 def compute(overrides = {})
60 order, = topology
61 result = {}
62 order.each do |name|
63 task = @tasks[name]
64 duration = overrides.fetch(name, task.duration)
65 start = predecessors(name).map do |a|
66 kind, lag = @deps[[a, name]]
67 case kind
68 when 'FS' then result[a][:EF] + lag
69 when 'SS' then result[a][:ES] + lag
70 else result[a][:EF] + lag - duration
71 end
72 end.push(task.day).max
73 result[name] = { DUR: duration, ES: start, EF: start + duration }
74 end
75 total = result.values.map { |v| v[:EF] }.max
76 order.reverse_each do |name|
77 value = result[name]
78 finish = successors(name).map do |b|
79 kind, lag = @deps[[name, b]]
80 case kind
81 when 'FS' then result[b][:LS] - lag
82 when 'SS' then result[b][:LS] - lag + value[:DUR]
83 else result[b][:LF] - lag
84 end
85 end.push(total).min
86 value[:LF] = finish
87 value[:LS] = finish - value[:DUR]
88 value[:TF] = value[:LS] - value[:ES]
89 end
90 @tasks.each_key do |name|
91 value = result[name]
92 value[:FF] = successors(name).map { |b| margin(name, b, result) }.push(total - value[:EF]).min
93 end
94 [result, total]
95 end
96
97 def margin(a, b, schedule)
98 kind, lag = @deps[[a, b]]
99 case kind
100 when 'FS' then schedule[b][:ES] - schedule[a][:EF] - lag
101 when 'SS' then schedule[b][:ES] - schedule[a][:ES] - lag
102 else schedule[b][:EF] - schedule[a][:EF] - lag
103 end
104 end
105
106 def critical(schedule)
107 @tasks.keys.sort.select { |name| schedule[name][:TF].zero? }
108 end
109
110 def critical_paths(schedule)
111 selected = critical(schedule)
112 edges = @deps.keys.select { |a, b| selected.include?(a) && selected.include?(b) && margin(a, b, schedule).zero? }
113 roots = selected.reject { |name| edges.any? { |_, b| b == name } }
114 paths = []
115 visit = lambda do |path|
116 next_tasks = edges.select { |a, _| a == path.last }.map(&:last).sort
117 if next_tasks.empty?
118 paths << path
119 else
120 next_tasks.each { |name| visit.call(path + [name]) }
121 end
122 end
123 roots.each { |name| visit.call([name]) }
124 paths
125 end

127 def signed(value)
128 format('%+d', value)
129 end

131 def decimal(numerator, denominator)
132 hundredths = numerator * 100 / denominator
133 format('%d.%02d', hundredths / 100, hundredths % 100)
134 end

136 def date_value(text)
137 year, month, day = text.split('-').map(&:to_i)
138 raise ScheduleError, "invalid date #{text}" unless Date.valid_date?(year, month, day)
139 Date.new(year, month, day)
140 end
141
142 def date_text(date)
143 date.strftime('%Y-%m-%d') + ' ' + %w[Sun Mon Tue Wed Thu Fri Sat][date.wday]
144 end
145
146 def working?(date)
147 !date.saturday? && !date.sunday? && !@holidays.include?(date)
148 end
149
150 def calendar(schedule, total)
151 dates = []
152 current = @start
153 while dates.length < total
154 dates << current if working?(current)
155 current += 1
156 end
157 @tasks.keys.sort_by { |name| [schedule[name][:ES], name] }.each do |name|
158 puts "#{name} #{date_text(dates[schedule[name][:ES]])} - #{date_text(dates[schedule[name][:EF] - 1])}"
159 end
160 first = dates.first
161 finish = dates.last
162 puts "project #{date_text(first)} - #{date_text(finish)}, #{total} working days, #{(finish - first).to_i + 1} calendar days"
163 if @deadline
164 days = 0
165 lower, upper = [finish, @deadline].minmax
166 current = lower + 1
167 while current <= upper
168 days += 1 if working?(current)
169 current += 1
170 end
171 description = finish <= @deadline ? "met, #{days} working days to spare" : "missed by #{days} working days"
172 puts "deadline #{date_text(@deadline)} #{description}"
173 end
174 holidays = @holidays.to_a.sort.select { |date| date >= first && date <= finish && !date.saturday? && !date.sunday? }
175 puts "holidays in span: #{holidays.empty? ? 'none' : holidays.map { |date| date_text(date) }.join(', ')}"
176 end

178 def usage(schedule, resource, start_key = :ES, finish_key = :EF)
179 daily = Hash.new(0)
180 @tasks.each do |name, task|
181 next unless task.resource == resource
182 (schedule[name][start_key]...schedule[name][finish_key]).each { |day| daily[day] += task.units }
183 end
184 daily
185 end

187 def level(limit, schedule, total)
188 capacity = ->(r) { @capacities.fetch(r, limit) }
189 @tasks.keys.sort.each do |name|
190 task = @tasks[name]
191 raise ScheduleError, "capacity #{capacity.call(task.resource)} too small for #{name}" if task.units > capacity.call(task.resource)
192 end
193 decided = {}
194 order = []
195 daily = Hash.new { |hash, key| hash[key] = Hash.new(0) }
196 until decided.length == @tasks.length
197 name = (@tasks.keys - decided.keys).select { |t| predecessors(t).all? { |a| decided.key?(a) } }.min_by { |t| [schedule[t][:LS], t] }
198 task = @tasks[name]
199 start = predecessors(name).map do |a|
200 kind, lag = @deps[[a, name]]
201 case kind
202 when 'FS' then decided[a][:F] + lag
203 when 'SS' then decided[a][:S] + lag
204 else decided[a][:F] + lag - task.duration
205 end
206 end.push(task.day).max
207 while (start...(start + task.duration)).any? { |day| daily[task.resource][day] + task.units > capacity.call(task.resource) }
208 start += 1
209 end
210 decided[name] = { S: start, F: start + task.duration }
211 (start...(start + task.duration)).each { |day| daily[task.resource][day] += task.units }
212 order << name
213 end
214 puts "level capacity #{limit}: duration #{total} -> #{decided.values.map { |v| v[:F] }.max}"
215 order.each { |name| puts "#{name} start #{decided[name][:S]} finish #{decided[name][:F]} shift #{decided[name][:S] - schedule[name][:ES]}" }
216 @tasks.values.map(&:resource).uniq.sort.each do |r|
217 peak = usage(schedule, r).values.max || 0
218 leveled = daily[r].values.max || 0
219 puts "resource #{r} capacity #{capacity.call(r)} peak #{peak} -> #{leveled}"
220 end
221 end

223 def crash(budget, initial)
224 remaining = budget
225 steps = 0
226 final = initial
227 reason = nil
228 loop do
229 schedule, total = compute
230 candidates = critical(schedule).select do |name|
231 task = @tasks[name]
232 task.crash_limit && task.used < task.crash_limit && compute(name => task.duration - 1)[1] < total
233 end
234 if candidates.empty?
235 reason = 'no candidate'
236 break
237 end
238 name = candidates.min_by { |t| [@tasks[t].cost, t] }
239 task = @tasks[name]
240 if task.cost > remaining
241 reason = 'insufficient budget'
242 break
243 end
244 before = task.duration
245 task.used += 1
246 remaining -= task.cost
247 steps += 1
248 final = compute[1]
249 puts "step #{steps}: #{name} #{before} -> #{task.duration}, cost #{task.cost}, project #{final}"
250 end
251 puts "crash done: #{steps} days, cost #{budget - remaining}, project #{initial} -> #{final}, budget left #{remaining}, #{reason}"
252 end

254 def analyze(command, args)
255 raise ScheduleError, 'no tasks' if @tasks.empty?
256 _, cycle = topology
257 raise ScheduleError, "cycle #{cycle}" if cycle
258 case command
259 when 'CALENDAR'
260 raise ScheduleError, 'no start date' unless @start
261 when 'LOAD'
262 raise ScheduleError, "unknown resource #{args[0]}" unless @tasks.values.any? { |t| t.resource == args[0] }
263 when 'WHATIF'
264 require_task(args[0])
265 when 'VARIANCE'
266 raise ScheduleError, 'no baseline' unless @baseline
267 end
268 schedule, total = compute
269 names = @tasks.keys.sort
270 case command
271 when 'CPM'
272 width = [4, names.map(&:length).max].max
273 puts 'TASK'.ljust(width) + %w[DUR ES EF LS LF TF FF].map { |s| s.rjust(5) }.join
274 names.each { |name| puts name.ljust(width) + %i[DUR ES EF LS LF TF FF].map { |key| schedule[name][key].to_s.rjust(5) }.join + (schedule[name][:TF].zero? ? ' *' : '') }
275 puts "project duration #{total}"
276 when 'CRITICAL', 'PERTSUM'
277 paths = critical_paths(schedule)
278 puts "critical paths: #{paths.length}" if command == 'CRITICAL'
279 paths.each do |path|
280 notation = path.join(' -> ')
281 if command == 'CRITICAL'
282 puts notation
283 else
284 estimates = path.map { |name| @tasks[name].estimate || Array.new(3, @tasks[name].duration) }
285 expected = estimates.sum { |o, m, p| o + 4 * m + p }
286 variance = estimates.sum { |o, _, p| (p - o)**2 }
287 puts "#{notation}: expected #{decimal(expected, 6)} variance #{decimal(variance, 36)}"
288 end
289 end
290 when 'SLACK'
291 names.group_by { |name| schedule[name][:TF] }.sort.each { |float, tasks| puts "float #{float}: #{tasks.join(' ')}" }
292 when 'DRAG'
293 critical(schedule).each { |name| puts "#{name} drag #{total - compute(name => 0)[1]}" }
294 when 'STATS'
295 count = critical(schedule).length
296 kinds = %w[FS SS FF].map { |kind| @deps.values.count { |k, _| k == kind } }
297 puts "tasks #{names.length}, critical #{count}, dependencies #{@deps.length} (FS #{kinds[0]}, SS #{kinds[1]}, FF #{kinds[2]})"
298 work = @tasks.values.group_by(&:resource).sort.map { |r, tasks| "#{r} #{tasks.sum { |t| t.duration * t.units }}" }
299 puts 'work ' + work.join(', ')
300 sum = schedule.values.sum { |value| value[:TF] }
301 puts "float total #{sum}, average #{decimal(sum, names.length)}"
302 puts "duration #{total}, critical share #{decimal(100 * count, names.length)}%"
303 when 'CALENDAR'
304 calendar(schedule, total)
305 when 'GANTT'
306 raise ScheduleError, "chart too wide (#{total} days)" if total > 60
307 width = names.map(&:length).max
308 puts ' ' * width + ' |' + (0...total).map { |d| d % 10 }.join + '|'
309 names.sort_by { |name| [schedule[name][:ES], name] }.each do |name|
310 value = schedule[name]
311 bar = (0...total).map do |day|
312 if day >= value[:ES] && day < value[:EF]
313 value[:TF].zero? ? '=' : '-'
314 elsif day >= value[:EF] && day < value[:LF]
315 '.'
316 else
317 ' '
318 end
319 end.join
320 puts name.ljust(width) + ' |' + bar + '|'
321 end
322 when 'LEVEL'
323 level(args[0].to_i, schedule, total)
324 when 'LOAD'
325 resource = args[0]
326 daily = usage(schedule, resource)
327 values = (0...total).map { |day| daily[day] }
328 puts "load #{resource}: #{values.join(' ')}"
329 puts "peak #{values.max} on day #{values.index(values.max)}"
330 puts "capacity #{@capacities[resource]} exceeded on #{values.count { |u| u > @capacities[resource] }} days" if @capacities.key?(resource)
331 when 'CRASH'
332 crash(args[0].to_i, total)
333 when 'WHATIF'
334 name, duration = args
335 after, new_total = compute(name => duration.to_i)
336 puts "whatif #{name} #{@tasks[name].duration} -> #{duration}: project #{total} -> #{new_total}"
337 names.each do |t|
338 differences = %i[ES EF TF].select { |key| schedule[t][key] != after[t][key] }
339 puts t + differences.map { |key| " #{key} #{schedule[t][key]}->#{after[t][key]}" }.join unless differences.empty?
340 end
341 puts "critical #{critical(schedule).join(' ')} -> #{critical(after).join(' ')}" if critical(schedule) != critical(after)
342 when 'BASELINE'
343 @baseline = [total, names.to_h { |name| [name, [schedule[name][:ES], schedule[name][:EF]]] }]
344 puts "baseline saved: #{names.length} tasks, duration #{total}"
345 when 'VARIANCE'
346 old_total, old = @baseline
347 puts "variance: duration #{old_total} -> #{total} (#{signed(total - old_total)})"
348 lines = []
349 (names | old.keys).sort.each do |name|
350 if !@tasks.key?(name)
351 lines << "#{name} removed"
352 elsif !old.key?(name)
353 lines << "#{name} new"
354 elsif [schedule[name][:ES], schedule[name][:EF]] != old[name]
355 lines << "#{name} start #{signed(schedule[name][:ES] - old[name][0])} finish #{signed(schedule[name][:EF] - old[name][1])}"
356 end
357 end
358 puts(lines.empty? ? ['no task changes'] : lines)
359 end
360 end

362 def execute(line)
363 command, *args = line.split(' ')
364 if ANALYSIS.include?(command)
365 analyze(command, args)
366 return
367 end
368 name = args[0]
369 case command
370 when 'TASK'
371 raise ScheduleError, "task #{name} already exists" if @tasks.key?(name)
372 @tasks[name] = Task.new(args[1].to_i, args[2], args[3].to_i, 0, nil, nil, 0, nil)
373 puts "task #{name} added"
374 when 'DEP', 'UNDEP'
375 a, b = args
376 require_task(a)
377 require_task(b)
378 if command == 'DEP'
379 raise ScheduleError, "self dependency #{a}" if a == b
380 raise ScheduleError, "dependency #{a} -> #{b} already exists" if @deps.key?([a, b])
381 @deps[[a, b]] = [args[2], args[3].to_i]
382 puts "dependency #{a} -> #{b} #{args[2]}#{signed(args[3].to_i)} added"
383 else
384 raise ScheduleError, "no dependency #{a} -> #{b}" unless @deps.key?([a, b])
385 @deps.delete([a, b])
386 puts "dependency #{a} -> #{b} removed"
387 end
388 when 'CAPACITY'
389 @capacities[name] = args[1].to_i
390 puts "capacity #{name} #{args[1]}"
391 when 'START', 'HOLIDAY', 'DEADLINE'
392 date = date_value(name)
393 if command == 'START'
394 @start = date
395 elsif command == 'DEADLINE'
396 @deadline = date
397 else
398 raise ScheduleError, "duplicate holiday #{name}" if @holidays.include?(date)
399 @holidays << date
400 end
401 puts "#{command.downcase} #{date_text(date)}"
402 when 'LIST'
403 puts "tasks #{@tasks.length}, dependencies #{@deps.length}, holidays #{@holidays.length}"
404 @tasks.keys.sort.each do |t|
405 task = @tasks[t]
406 preds = predecessors(t)
407 puts "#{t} duration #{task.duration} resource #{task.resource} units #{task.units} preds #{preds.empty? ? 'none' : preds.join(' ')}"
408 end
409 else
410 require_task(name)
411 task = @tasks[name]
412 case command
413 when 'REMOVE'
414 removed = @deps.keys.select { |pair| pair.include?(name) }
415 removed.each { |key| @deps.delete(key) }
416 @tasks.delete(name)
417 puts "task #{name} removed with #{removed.length} dependencies"
418 when 'NOTBEFORE'
419 task.day = args[1].to_i
420 puts "notbefore #{name} #{args[1]}"
421 when 'CRASHCOST'
422 maximum, cost = args.drop(1).map(&:to_i)
423 raise ScheduleError, "crash limit too large for #{name}" if maximum >= task.base
424 raise ScheduleError, "crash limit below used days for #{name}" if maximum < task.used
425 task.crash_limit = maximum
426 task.cost = cost
427 puts "crash #{name} max #{maximum} cost #{cost}"
428 when 'UNCRASH'
429 puts "uncrash #{name} restored #{task.used} days"
430 task.used = 0
431 when 'PERT'
432 o, m, p = args.drop(1).map(&:to_i)
433 raise ScheduleError, 'invalid estimate' unless o <= m && m <= p
434 task.estimate = [o, m, p]
435 puts "pert #{name} expected #{decimal(o + 4 * m + p, 6)} variance #{decimal((p - o)**2, 36)}"
436 when 'SHOW'
437 puts "task #{name} duration #{task.duration} resource #{task.resource} units #{task.units}#{task.day.zero? ? '' : " notbefore #{task.day}"}"
438 preds = predecessors(name).map do |a|
439 kind, lag = @deps[[a, name]]
440 "#{a} #{kind}#{signed(lag)}"
441 end
442 succs = successors(name).map do |b|
443 kind, lag = @deps[[name, b]]
444 "#{b} #{kind}#{signed(lag)}"
445 end
446 puts "preds: #{preds.empty? ? 'none' : preds.join(', ')}"
447 puts "succs: #{succs.empty? ? 'none' : succs.join(', ')}"
448 puts(task.crash_limit ? "crash: max #{task.crash_limit} used #{task.used} cost #{task.cost}" : 'crash: none')
449 puts(task.estimate ? "pert: #{task.estimate.join(' ')}" : 'pert: none')
450 _, cycle = topology
451 if cycle
452 puts 'schedule: unavailable'
453 else
454 schedule, = compute
455 puts 'schedule: ' + %i[ES EF LS LF TF FF].map { |key| "#{key} #{schedule[name][key]}" }.join(' ') + (schedule[name][:TF].zero? ? ' critical' : '')
456 end
457 when 'IMPACT'
458 seen = [name]
459 seen.each { |a| successors(a).each { |b| seen << b unless seen.include?(b) } }
460 affected = (seen - [name]).sort
461 puts "impact #{name}: #{affected.empty? ? 'none' : affected.join(' ')}"
462 end
463 end
464 rescue ScheduleError => error
465 puts "error: #{error.message}"
466 end
467end
