82def start_bound(dep, starts, finishes, duration):
83 if dep.kind == "FS":
84 return finishes[dep.pred] + dep.lag
85 if dep.kind == "SS":
86 return starts[dep.pred] + dep.lag
87 return finishes[dep.pred] + dep.lag - duration

89
90def finish_bound(dep, latest_starts, latest_finishes, duration):
91 if dep.kind == "FS":
92 return latest_starts[dep.succ] - dep.lag
93 if dep.kind == "SS":
94 return latest_starts[dep.succ] - dep.lag + duration
95 return latest_finishes[dep.succ] - dep.lag

98def earliest_start(task, starts, finishes, duration):
99 bounds = [start_bound(dep, starts, finishes, duration) for dep in task.preds.values()]
100 return max([task.not_before] + bounds)

103class Schedule:
104 def __init__(self, tasks, order, durations):
105 self.es = {}
106 self.ef = {}
107 for name in order:
108 self.es[name] = earliest_start(tasks[name], self.es, self.ef, durations[name])
109 self.ef[name] = self.es[name] + durations[name]
110 self.length = max(self.ef.values())
111 self.ls = {}
112 self.lf = {}
113 for name in reversed(order):
114 bounds = [
115 finish_bound(dep, self.ls, self.lf, durations[name])
116 for dep in tasks[name].succs.values()
117 ]
118 self.lf[name] = min([self.length] + bounds)
119 self.ls[name] = self.lf[name] - durations[name]
120 self.total_float = {name: self.ls[name] - self.es[name] for name in order}
121 self.free_float = {}
122 for name in order:
123 margins = [self.margin(dep) for dep in tasks[name].succs.values()]
124 self.free_float[name] = min([self.length - self.ef[name]] + margins)
125 self.critical = {name for name in order if self.total_float[name] == 0}
126
127 def margin(self, dep):
128 if dep.kind == "FS":
129 return self.es[dep.succ] - self.ef[dep.pred] - dep.lag
130 if dep.kind == "SS":
131 return self.es[dep.succ] - self.es[dep.pred] - dep.lag
132 return self.ef[dep.succ] - self.ef[dep.pred] - dep.lag
133
134 def values(self, name):
135 return (
136 self.es[name],
137 self.ef[name],
138 self.ls[name],
139 self.lf[name],
140 self.total_float[name],
141 self.free_float[name],
142 )

144 def by_start(self):
145 return sorted(self.es, key=lambda name: (self.es[name], name))

203 def topological_order(self):
204 waiting = {name: len(task.preds) for name, task in self.tasks.items()}
205 ready = [name for name, count in waiting.items() if count == 0]
206 order = []
207 while ready:
208 name = ready.pop()
209 order.append(name)
210 for succ in self.tasks[name].succs:
211 waiting[succ] -= 1
212 if waiting[succ] == 0:
213 ready.append(succ)
214 return order

216 def cycle_notation(self, order):
217 remaining = set(self.tasks) - set(order)
218 walk = [min(remaining)]
219 while True:
220 pred = min(name for name in self.tasks[walk[-1]].preds if name in remaining)
221 if pred in walk:
222 break
223 walk.append(pred)
224 cycle = [pred] + walk[walk.index(pred) + 1 :][::-1]
225 first = cycle.index(min(cycle))
226 cycle = cycle[first:] + cycle[:first]
227 return " -> ".join(cycle + [cycle[0]])

229 def analysis_order(self):
230 if not self.tasks:
231 raise CommandError("no tasks")
232 order = self.topological_order()
233 if len(order) < len(self.tasks):
234 raise CommandError("cycle " + self.cycle_notation(order))
235 return order

243 def critical_paths(self, schedule):
244 links = {}
245 for name in schedule.critical:
246 links[name] = [
247 succ
248 for succ, dep in sorted(self.tasks[name].succs.items())
249 if succ in schedule.critical and schedule.margin(dep) == 0
250 ]
251 targets = {succ for following in links.values() for succ in following}
252 paths = []
253
254 def extend(path):
255 following = links[path[-1]]
256 if not following:
257 paths.append(path)
258 for succ in following:
259 extend(path + [succ])
260
261 for head in sorted(schedule.critical - targets):
262 extend([head])
263 return paths

295 def add_dependency(self, pred, succ, kind, lag):
296 first = self.task(pred)
297 second = self.task(succ)
298 if pred == succ:
299 raise CommandError(f"self dependency {pred}")
300 if succ in first.succs:
301 raise CommandError(f"dependency {pred} -> {succ} already exists")
302 dep = Dependency(pred, succ, kind, int(lag))
303 first.succs[succ] = dep
304 second.preds[pred] = dep
305 print(f"dependency {pred} -> {succ} {kind}{signed(dep.lag)} added")

513 def gantt(self):
514 schedule = self.schedule(self.analysis_order())
515 length = schedule.length
516 if length > MAX_CHART_DAYS:
517 raise CommandError(f"chart too wide ({length} days)")
518 width = max(map(len, self.tasks))
519 print(" " * width + " |" + "".join(str(day % 10) for day in range(length)) + "|")
520 for name in schedule.by_start():
521 es = schedule.es[name]
522 ef = schedule.ef[name]
523 lf = schedule.lf[name]
524 mark = "=" if name in schedule.critical else "-"
525 bar = " " * es + mark * (ef - es) + "." * (lf - ef) + " " * (length - lf)
526 print(f"{name.ljust(width)} |{bar}|")

528 def level(self, limit):
529 order = self.analysis_order()
530 capacity = {
531 task.resource: self.capacities.get(task.resource, int(limit))
532 for task in self.tasks.values()
533 }
534 for task in self.sorted_tasks():
535 if task.units > capacity[task.resource]:
536 raise CommandError(f"capacity {capacity[task.resource]} too small for {task.name}")
537 schedule = self.schedule(order)
538 starts = {}
539 finishes = {}
540 usage = defaultdict(int)
541 while len(starts) < len(self.tasks):
542 ready = [
543 task
544 for task in self.tasks.values()
545 if task.name not in starts and all(pred in starts for pred in task.preds)
546 ]
547 task = min(ready, key=lambda candidate: (schedule.ls[candidate.name], candidate.name))
548 available = capacity[task.resource]
549 start = earliest_start(task, starts, finishes, task.duration)
550 while any(
551 usage[task.resource, day] + task.units > available
552 for day in range(start, start + task.duration)
553 ):
554 start += 1
555 for day in range(start, start + task.duration):
556 usage[task.resource, day] += task.units
557 starts[task.name] = start
558 finishes[task.name] = start + task.duration
559 print(f"level capacity {limit}: duration {schedule.length} -> {max(finishes.values())}")
560 for name, start in starts.items():
561 shift = start - schedule.es[name]
562 print(f"{name} start {start} finish {finishes[name]} shift {shift}")
563 for resource in sorted(capacity):
564 planned = max(self.resource_usage(resource, schedule.es, schedule.ef).values())
565 leveled = max(self.resource_usage(resource, starts, finishes).values())
566 print(f"resource {resource} capacity {capacity[resource]} peak {planned} -> {leveled}")

583 def pert_sum(self):
584 schedule = self.schedule(self.analysis_order())
585 for path in self.critical_paths(schedule):
586 moments = [pert_moments(self.tasks[name].three_point()) for name in path]
587 expected = two_decimals(sum(mean for mean, _ in moments))
588 variance = two_decimals(sum(spread for _, spread in moments))
589 print(f"{' -> '.join(path)}: expected {expected} variance {variance}")

591 def crash(self, budget):
592 order = self.analysis_order()
593 remaining = int(budget)
594 steps = 0
595 schedule = self.schedule(order)
596 initial = schedule.length
597 while True:
598 candidates = [
599 task
600 for task in self.sorted_tasks()
601 if task.name in schedule.critical
602 and task.crash_limit is not None
603 and task.crashed < task.crash_limit
604 and self.schedule(order, task.name, task.duration - 1).length < schedule.length
605 ]
606 if not candidates:
607 reason = "no candidate"
608 break
609 task = min(candidates, key=lambda candidate: candidate.crash_cost)
610 if task.crash_cost > remaining:
611 reason = "insufficient budget"
612 break
613 previous = task.duration
614 task.crashed += 1
615 remaining -= task.crash_cost
616 steps += 1
617 schedule = self.schedule(order)
618 print(
619 f"step {steps}: {task.name} {previous} -> {task.duration}, "
620 f"cost {task.crash_cost}, project {schedule.length}"
621 )
622 print(
623 f"crash done: {steps} days, cost {int(budget) - remaining}, "
624 f"project {initial} -> {schedule.length}, budget left {remaining}, {reason}"
625 )

630 before = self.schedule(order)
631 after = self.schedule(order, name, int(duration))
632 print(
633 f"whatif {name} {task.duration} -> {duration}: "

637 old = (before.es[other], before.ef[other], before.total_float[other])
638 new = (after.es[other], after.ef[other], after.total_float[other])
639 changes = [
640 f"{label} {previous}->{current}"
641 for label, previous, current in zip(("ES", "EF", "TF"), old, new)
642 if previous != current
643 ]
644 if changes:
645 print(other + " " + " ".join(changes))
646 if before.critical != after.critical:
647 print(f"critical {name_list(before.critical)} -> {name_list(after.critical)}")

653 def variance(self):
654 order = self.analysis_order()
655 if self.baseline is None:
656 raise CommandError("no baseline")
657 schedule = self.schedule(order)
658 base = self.baseline
659 print(
660 f"variance: duration {base.length} -> {schedule.length} "
661 f"({signed(schedule.length - base.length)})"
662 )
663 lines = []
664 for name in sorted(set(base.es) | set(self.tasks)):
665 if name not in self.tasks:
666 lines.append(f"{name} removed")
667 elif name not in base.es:
668 lines.append(f"{name} new")
669 elif (base.es[name], base.ef[name]) != (schedule.es[name], schedule.ef[name]):
