4class Task {
5 final String id, resource;
6 final int base, units;
7 int notBefore = 0, used = 0;
8 (int, int)? crash;
9 (int, int, int)? pert;
10 Task(this.id, this.base, this.resource, this.units);
11 int get duration => base - used;
12}
13class Dependency {
14 final String a, b, kind;
15 final int lag;
16 Dependency(this.a, this.b, this.kind, this.lag);
17}
18class Timing {
19 final int duration, es, ef;
20 int ls = 0, lf = 0, tf = 0, ff = 0;
21 Timing(this.duration, this.es) : ef = es + duration;
22}
23class Schedule {
24 final Map<String, Timing> rows;
25 final int total;
26 Schedule(this.rows, this.total);
27}
28class Baseline {
29 final int total;
30 final Map<String, (int, int)> rows;
31 Baseline(this.total, this.rows);
32}
33class ScheduleError implements Exception {
34 final String message;
35 ScheduleError(this.message);
36}
37final tasks = <String, Task>{};
38var dependencies = <Dependency>[];
39final capacities = <String, int>{};
40final holidays = <int>{};
41final output = <String>[];
42int? startDate, deadline;
43Baseline? baseline;
44final epoch = DateTime.utc(2000, 1, 1);
45Never fail(String message) => throw ScheduleError('error: $message');
46List<String> names() => tasks.keys.toList()..sort();
47Task task(String name) {
48 final result = tasks[name];
49 if (result == null) fail('unknown task $name');
50 return result;
51}
52String signed(int value) => value >= 0 ? '+$value' : '$value';
53String decimal(int numerator, int denominator) {
54 final hundredths = numerator * 100 ~/ denominator;
55 return '${hundredths ~/ 100}.${(hundredths % 100).toString().padLeft(2, '0')}';
56}
57List<Dependency> predecessors(String name) => dependencies.where((dep) => dep.b == name).toList()
58 ..sort((a, b) => a.a.compareTo(b.a));
59List<Dependency> successors(String name) => dependencies.where((dep) => dep.a == name).toList()
60 ..sort((a, b) => a.b.compareTo(b.b));
61(List<String>, Set<String>) topological() {
62 final remaining = names().toSet(), order = <String>[];
63 while (true) {
64 String? next;
65 for (final name in remaining) {
66 if (predecessors(name).every((dep) => !remaining.contains(dep.a))) {
67 next = name;
68 break;
69 }
70 }
71 if (next == null) return (order, remaining);
72 remaining.remove(next);
73 order.add(next);
74 }
75}
76String? cycle() {
77 final remaining = topological().$2;
78 if (remaining.isEmpty) return null;
79 final trail = <String>[remaining.first];
80 while (true) {
81 final next = predecessors(trail.last).firstWhere((dep) => remaining.contains(dep.a)).a;
82 final at = trail.indexOf(next);
83 if (at >= 0) {
84 final loop = [next, ...trail.sublist(at + 1).reversed];
85 final smallest = (loop.toList()..sort()).first, first = loop.indexOf(smallest);
86 final rotated = [...loop.sublist(first), ...loop.sublist(0, first)];
87 return [...rotated, rotated.first].join(' -> ');
88 }
89 trail.add(next);
90 }
91}
92Schedule schedule([Map<String, int> overrides = const {}]) {
93 final order = topological().$1, rows = <String, Timing>{};
94 var total = 0;
95 for (final name in order) {
96 final item = tasks[name]!, d = overrides[name] ?? tasks[name]!.duration;
97 var es = item.notBefore;
98 for (final dep in predecessors(name)) {
99 final old = rows[dep.a]!;
100 es = math.max(es, (dep.kind == 'SS' ? old.es : old.ef) + dep.lag - (dep.kind == 'FF' ? d : 0));
101 }
102 rows[name] = Timing(d, es);
103 total = math.max(total, es + d);
104 }
105 for (final name in order.reversed) {
106 final row = rows[name]!;
107 var lf = total;
108 for (final dep in successors(name)) {
109 final next = rows[dep.b]!;
110 lf = math.min(lf, (dep.kind == 'FF' ? next.lf : next.ls) - dep.lag + (dep.kind == 'SS' ? row.duration : 0));
111 }
112 row.lf = lf;
113 row.ls = lf - row.duration;
114 row.tf = row.ls - row.es;
115 }
116 for (final name in order) {
117 final row = rows[name]!;
118 row.ff = total - row.ef;
119 for (final dep in successors(name)) { row.ff = math.min(row.ff, margin(dep, rows)); }
120 }
121 return Schedule(rows, total);
122}
123int margin(Dependency dep, Map<String, Timing> rows) {
124 final a = rows[dep.a]!, b = rows[dep.b]!;
125 return (dep.kind == 'FF' ? b.ef : b.es) - (dep.kind == 'SS' ? a.es : a.ef) - dep.lag;
126}
127List<String> critical(Schedule schedule) => names().where((name) => schedule.rows[name]!.tf == 0).toList();
128List<List<String>> criticalPaths(Schedule schedule) {
129 final criticalNames = critical(schedule).toSet();
130 final links = dependencies.where((dep) => criticalNames.contains(dep.a) &&
131 criticalNames.contains(dep.b) && margin(dep, schedule.rows) == 0).toList();
132 final roots = criticalNames.where((name) => !links.any((dep) => dep.b == name));
133 final result = <List<String>>[];
134 void visit(List<String> path) {
135 final next = links.where((dep) => dep.a == path.last).map((dep) => dep.b).toList()..sort();
136 if (next.isEmpty) result.add(path);
137 for (final name in next) { visit([...path, name]); }
138 }
139 for (final name in roots) { visit([name]); }
140 return result;
141}
142int parseDate(String word) {
143 final parts = word.split('-').map(int.parse).toList();
144 final value = DateTime.utc(parts[0], parts[1], parts[2]);
145 if (value.year != parts[0] || value.month != parts[1] || value.day != parts[2]) fail('invalid date $word');
146 return value.difference(epoch).inDays;
147}
148int weekday(int day) => (day + 5) % 7;
149String dateText(int day) {
150 final date = epoch.add(Duration(days: day));
151 return '${date.year.toString().padLeft(4, '0')}-${date.month.toString().padLeft(2, '0')}-${date.day.toString().padLeft(2, '0')} ${['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'][weekday(day)]}';
152}
153bool working(int day) => weekday(day) < 5 && !holidays.contains(day);
154List<int> dayDates(int count) {
155 final result = <int>[];
156 for (var day = startDate!; result.length < count; day++) {
157 if (working(day)) result.add(day);
158 }
159 return result;
160}
161int workingBetween(int a, int b) {
162 var count = 0;
163 for (var day = a + 1; day <= b; day++) {
164 if (working(day)) count++;
165 }
166 return count;
167}
168List<String> resources() => tasks.values.map((item) => item.resource).toSet().toList()..sort();
169List<int> usage(String resource, Schedule schedule) {
170 final result = List<int>.filled(schedule.total, 0);
171 for (final item in tasks.values) {
172 if (item.resource != resource) continue;
173 final row = schedule.rows[item.id]!;
174 for (var day = row.es; day < row.ef; day++) { result[day] += item.units; }
175 }
176 return result;
177}
178void level(int limit, Schedule original) {
179 int capacity(String resource) => capacities[resource] ?? limit;
180 for (final name in names()) {
181 final item = tasks[name]!;
182 if (item.units > capacity(item.resource)) fail('capacity ${capacity(item.resource)} too small for $name');
183 }
184 final undecided = names().toSet(), decided = <String, (int, int)>{};
185 final used = <String, Map<int, int>>{for (final name in resources()) name: {}}, order = <String>[];
186 while (undecided.isNotEmpty) {
187 final available = undecided.where((name) => predecessors(name).every((dep) => decided.containsKey(dep.a))).toList();
188 available.sort((a, b) {
189 final comparison = original.rows[a]!.ls.compareTo(original.rows[b]!.ls);
190 return comparison == 0 ? a.compareTo(b) : comparison;
191 });
192 final name = available.first, item = tasks[available.first]!;
193 final d = item.duration, days = used[item.resource]!;
194 var start = item.notBefore;
195 for (final dep in predecessors(name)) {
196 final old = decided[dep.a]!;
197 start = math.max(start, (dep.kind == 'SS' ? old.$1 : old.$2) + dep.lag - (dep.kind == 'FF' ? d : 0));
198 }
199 while (true) {
200 var valid = true;
201 for (var day = start; day < start + d; day++) {
202 if ((days[day] ?? 0) + item.units > capacity(item.resource)) { valid = false; break; }
203 }
204 if (valid) break;
205 start++;
206 }
207 for (var day = start; day < start + d; day++) { days[day] = (days[day] ?? 0) + item.units; }
208 decided[name] = (start, start + d);
209 undecided.remove(name);
210 order.add(name);
211 }
212 final finish = decided.values.map((row) => row.$2).reduce(math.max);
213 output.add('level capacity $limit: duration ${original.total} -> $finish');
214 for (final name in order) {
215 final row = decided[name]!;
216 output.add('$name start ${row.$1} finish ${row.$2} shift ${row.$1 - original.rows[name]!.es}');
217 }
218 for (final resource in resources()) {
219 final peak = usage(resource, original).reduce(math.max);
220 final next = used[resource]!.values.fold<int>(0, math.max);
221 output.add('resource $resource capacity ${capacity(resource)} peak $peak -> $next');
222 }
223}
224void execute(String commandLine) {
225 final words = commandLine.split(' '), command = words.first, args = words.skip(1).toList();
226 final a = args.isEmpty ? '' : args[0], b = args.length < 2 ? '' : args[1];
227 if (command == 'TASK') {
228 if (tasks.containsKey(a)) fail('task $a already exists');
229 tasks[a] = Task(a, int.parse(b), args[2], int.parse(args[3]));
230 output.add('task $a added');
231 return;
232 }
233 if (command == 'DEP' || command == 'UNDEP') {
234 task(a);
235 task(b);
236 final at = dependencies.indexWhere((dep) => dep.a == a && dep.b == b);
237 if (command == 'DEP') {
238 if (a == b) fail('self dependency $a');
239 if (at >= 0) fail('dependency $a -> $b already exists');
240 dependencies.add(Dependency(a, b, args[2], int.parse(args[3])));
241 output.add('dependency $a -> $b ${args[2]}${signed(int.parse(args[3]))} added');
242 } else {
243 if (at < 0) fail('no dependency $a -> $b');
244 dependencies.removeAt(at);
245 output.add('dependency $a -> $b removed');
246 }
247 return;
248 }
249 if (['REMOVE', 'NOTBEFORE', 'CRASHCOST', 'UNCRASH', 'PERT', 'SHOW', 'IMPACT'].contains(command)) {
250 final item = task(a);
251 if (command == 'REMOVE') {
252 final before = dependencies.length;
253 dependencies.removeWhere((dep) => dep.a == a || dep.b == a);
254 tasks.remove(a);
255 output.add('task $a removed with ${before - dependencies.length} dependencies');
256 } else if (command == 'NOTBEFORE') {
257 item.notBefore = int.parse(b);
258 output.add('notbefore $a $b');
259 } else if (command == 'CRASHCOST') {
260 if (int.parse(b) >= item.base) fail('crash limit too large for $a');
261 if (int.parse(b) < item.used) fail('crash limit below used days for $a');
262 item.crash = (int.parse(b), int.parse(args[2]));
263 output.add('crash $a max $b cost ${args[2]}');
264 } else if (command == 'UNCRASH') {
265 output.add('uncrash $a restored ${item.used} days');
266 item.used = 0;
267 } else if (command == 'PERT') {
268 final o = int.parse(args[1]), m = int.parse(args[2]), p = int.parse(args[3]);
269 if (o > m || m > p) fail('invalid estimate');
270 item.pert = (o, m, p);
271 output.add('pert $a expected ${decimal(o + 4 * m + p, 6)} variance ${decimal((p - o) * (p - o), 36)}');
272 } else if (command == 'IMPACT') {
273 final seen = <String>{a}, queue = <String>[a];
274 for (var i = 0; i < queue.length; i++) {
275 for (final dep in successors(queue[i])) {
276 if (seen.add(dep.b)) queue.add(dep.b);
277 }
278 }
279 seen.remove(a);
280 final affected = seen.toList()..sort();
281 output.add('impact $a: ${affected.isEmpty ? 'none' : affected.join(' ')}');
282 } else {
283 final preds = predecessors(a).map((dep) => '${dep.a} ${dep.kind}${signed(dep.lag)}').join(', ');
284 final succs = successors(a).map((dep) => '${dep.b} ${dep.kind}${signed(dep.lag)}').join(', ');
285 output.addAll(['task $a duration ${item.duration} resource ${item.resource} units ${item.units}${item.notBefore != 0 ? ' notbefore ${item.notBefore}' : ''}',
286 'preds: ${preds.isEmpty ? 'none' : preds}', 'succs: ${succs.isEmpty ? 'none' : succs}',
287 item.crash == null ? 'crash: none' : 'crash: max ${item.crash!.$1} used ${item.used} cost ${item.crash!.$2}',
288 item.pert == null ? 'pert: none' : 'pert: ${item.pert!.$1} ${item.pert!.$2} ${item.pert!.$3}']);
289 if (cycle() != null) {
290 output.add('schedule: unavailable');
291 } else {
292 final row = schedule().rows[a]!;
293 output.add('schedule: ES ${row.es} EF ${row.ef} LS ${row.ls} LF ${row.lf} TF ${row.tf} FF ${row.ff}${row.tf == 0 ? ' critical' : ''}');
294 }
295 }
296 return;
297 }
298 if (command == 'CAPACITY') {
299 capacities[a] = int.parse(b);
300 output.add('capacity $a $b');
301 return;
302 }
303 if (['START', 'HOLIDAY', 'DEADLINE'].contains(command)) {
304 final day = parseDate(a);
305 if (command == 'HOLIDAY' && holidays.contains(day)) fail('duplicate holiday $a');
306 if (command == 'START') { startDate = day; }
307 else if (command == 'DEADLINE') { deadline = day; }
308 else { holidays.add(day); }
309 output.add('${command.toLowerCase()} ${dateText(day)}');
310 return;
311 }
312 if (command == 'LIST') {
313 output.add('tasks ${tasks.length}, dependencies ${dependencies.length}, holidays ${holidays.length}');
314 for (final name in names()) {
315 final item = tasks[name]!, preds = predecessors(name).map((dep) => dep.a).join(' ');
316 output.add('$name duration ${item.duration} resource ${item.resource} units ${item.units} preds ${preds.isEmpty ? 'none' : preds}');
317 }
318 return;
319 }
320 if (tasks.isEmpty) fail('no tasks');
321 final foundCycle = cycle();
322 if (foundCycle != null) fail('cycle $foundCycle');
323 if (command == 'CALENDAR' && startDate == null) fail('no start date');
324 if (command == 'LOAD' && !resources().contains(a)) fail('unknown resource $a');
325 if (command == 'WHATIF') task(a);
326 if (command == 'VARIANCE' && baseline == null) fail('no baseline');
327 final current = schedule(), byStart = names();
328 byStart.sort((a, b) {
329 final comparison = current.rows[a]!.es.compareTo(current.rows[b]!.es);
330 return comparison == 0 ? a.compareTo(b) : comparison;
331 });
332 switch (command) {
333 case 'CPM':
334 final width = names().map((name) => name.length).fold<int>(4, math.max);
335 output.add('TASK'.padRight(width) + ['DUR', 'ES', 'EF', 'LS', 'LF', 'TF', 'FF'].map((word) => word.padLeft(5)).join());
336 for (final name in names()) {
337 final row = current.rows[name]!;
338 output.add(name.padRight(width) + [row.duration, row.es, row.ef, row.ls, row.lf, row.tf, row.ff]
339 .map((value) => value.toString().padLeft(5)).join() + (row.tf == 0 ? ' *' : ''));
340 }
341 output.add('project duration ${current.total}');
342 case 'CRITICAL':
343 final paths = criticalPaths(current);
344 output.add('critical paths: ${paths.length}');
345 output.addAll(paths.map((path) => path.join(' -> ')));
346 case 'SLACK':
347 final floats = current.rows.values.map((row) => row.tf).toSet().toList()..sort();
348 for (final value in floats) {
349 output.add('float $value: ${names().where((name) => current.rows[name]!.tf == value).join(' ')}');
350 }
351 case 'DRAG':
352 for (final name in critical(current)) { output.add('$name drag ${current.total - schedule({name: 0}).total}'); }
353 case 'STATS':
354 final count = critical(current).length, floats = current.rows.values.fold<int>(0, (sum, row) => sum + row.tf);
355 final work = resources().map((resource) => '$resource ${tasks.values.where((item) => item.resource == resource).fold<int>(0, (sum, item) => sum + item.duration * item.units)}').join(', ');
356 output.addAll(['tasks ${tasks.length}, critical $count, dependencies ${dependencies.length} (FS ${dependencies.where((dep) => dep.kind == 'FS').length}, SS ${dependencies.where((dep) => dep.kind == 'SS').length}, FF ${dependencies.where((dep) => dep.kind == 'FF').length})',
357 'work $work', 'float total $floats, average ${decimal(floats, tasks.length)}',
358 'duration ${current.total}, critical share ${decimal(100 * count, tasks.length)}%']);
359 case 'CALENDAR':
360 final dates = dayDates(current.total), first = dates.first, last = dates.last;
361 for (final name in byStart) {
362 final row = current.rows[name]!;
363 output.add('$name ${dateText(dates[row.es])} - ${dateText(dates[row.ef - 1])}');
364 }
365 output.add('project ${dateText(first)} - ${dateText(last)}, ${current.total} working days, ${last - first + 1} calendar days');
366 if (deadline != null) {
367 output.add(last <= deadline! ? 'deadline ${dateText(deadline!)} met, ${workingBetween(last, deadline!)} working days to spare'
368 : 'deadline ${dateText(deadline!)} missed by ${workingBetween(deadline!, last)} working days');
369 }
370 final within = holidays.where((day) => weekday(day) < 5 && day >= first && day <= last).toList()..sort();
371 output.add('holidays in span: ${within.isEmpty ? 'none' : within.map(dateText).join(', ')}');
372 case 'GANTT':
373 if (current.total > 60) fail('chart too wide (${current.total} days)');
374 final width = names().map((name) => name.length).reduce(math.max);
375 output.add(''.padRight(width) + ' |${List.generate(current.total, (i) => i % 10).join()}|');
376 for (final name in byStart) {
377 final row = current.rows[name]!;
378 final bar = List.generate(current.total, (day) => day >= row.es && day < row.ef
379 ? row.tf == 0 ? '=' : '-' : day >= row.ef && day < row.lf ? '.' : ' ').join();
380 output.add('${name.padRight(width)} |$bar|');
381 }
382 case 'LEVEL':
383 level(int.parse(a), current);
384 case 'LOAD':
385 final days = usage(a, current), peak = days.reduce(math.max);
386 output.addAll(['load $a: ${days.join(' ')}', 'peak $peak on day ${days.indexOf(peak)}']);
387 if (capacities.containsKey(a)) output.add('capacity ${capacities[a]} exceeded on ${days.where((value) => value > capacities[a]!).length} days');
388 case 'PERTSUM':
389 for (final path in criticalPaths(current)) {
390 var expected = 0, variance = 0;
391 for (final name in path) {
392 final item = tasks[name]!;
393 final (o, m, p) = item.pert ?? (item.duration, item.duration, item.duration);
394 expected += o + 4 * m + p;
395 variance += (p - o) * (p - o);
396 }
397 output.add('${path.join(' -> ')}: expected ${decimal(expected, 6)} variance ${decimal(variance, 36)}');
398 }
399 case 'CRASH':
400 var remaining = int.parse(a), count = 0, total = current.total, reason = '';
401 while (true) {
402 final now = schedule();
403 final candidates = critical(now).where((name) {
404 final item = tasks[name]!;
405 return item.crash != null && item.used < item.crash!.$1 && schedule({name: item.duration - 1}).total < now.total;
406 }).toList();
407 candidates.sort((a, b) {
408 final comparison = tasks[a]!.crash!.$2.compareTo(tasks[b]!.crash!.$2);
409 return comparison == 0 ? a.compareTo(b) : comparison;
410 });
411 if (candidates.isEmpty) { reason = 'no candidate'; break; }
412 final name = candidates.first, item = tasks[candidates.first]!;
413 final cost = item.crash!.$2;
414 if (cost > remaining) { reason = 'insufficient budget'; break; }
415 final old = item.duration;
416 item.used++;
417 remaining -= cost;
418 count++;
419 total = schedule().total;
420 output.add('step $count: $name $old -> ${item.duration}, cost $cost, project $total');
421 }
422 output.add('crash done: $count days, cost ${int.parse(a) - remaining}, project ${current.total} -> $total, budget left $remaining, $reason');
423 case 'WHATIF':
424 final after = schedule({a: int.parse(b)});
425 output.add('whatif $a ${tasks[a]!.duration} -> $b: project ${current.total} -> ${after.total}');
426 for (final name in names()) {
427 final before = current.rows[name]!, next = after.rows[name]!;
428 final changes = <String>[];
429 if (before.es != next.es) changes.add('ES ${before.es}->${next.es}');
430 if (before.ef != next.ef) changes.add('EF ${before.ef}->${next.ef}');
431 if (before.tf != next.tf) changes.add('TF ${before.tf}->${next.tf}');
432 if (changes.isNotEmpty) output.add('$name ${changes.join(' ')}');
433 }
434 final beforeCritical = critical(current).join(' '), afterCritical = critical(after).join(' ');
435 if (beforeCritical != afterCritical) output.add('critical $beforeCritical -> $afterCritical');
436 case 'BASELINE':
437 baseline = Baseline(current.total, {for (final entry in current.rows.entries) entry.key: (entry.value.es, entry.value.ef)});
438 output.add('baseline saved: ${tasks.length} tasks, duration ${current.total}');
439 case 'VARIANCE':
440 output.add('variance: duration ${baseline!.total} -> ${current.total} (${signed(current.total - baseline!.total)})');
441 final before = output.length, allNames = {...baseline!.rows.keys, ...tasks.keys}.toList()..sort();
442 for (final name in allNames) {
443 final old = baseline!.rows[name], row = current.rows[name];
444 if (row == null) { output.add('$name removed'); }
445 else if (old == null) { output.add('$name new'); }
446 else if (old.$1 != row.es || old.$2 != row.ef) { output.add('$name start ${signed(row.es - old.$1)} finish ${signed(row.ef - old.$2)}'); }
447 }
448 if (output.length == before) output.add('no task changes');
449 }
450}
451void main() {
452 final count = int.parse(stdin.readLineSync()!);
453 for (var i = 0; i < count; i++) {
454 try { execute(stdin.readLineSync()!); }
455 on ScheduleError catch (error) { output.add(error.message); }
456 }
457 if (output.isNotEmpty) stdout.write('${output.join('\n')}\n');
458}
