14function parsedate(text)

16 Date(parse(Int, text[1:4]), parse(Int, text[6:7]), parse(Int, text[9:10]))
17 catch e

21end
22datetext(d) =
23 Dates.format(d, dateformat"yyyy-mm-dd")*" "*["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"][dayofweek(

26mutable struct WorkTask
27 base::Int
28 resource::String
29 units::Int
30 notbefore::Int
31 crashlimit::Union{Nothing,Int}
32 crashcost::Int

35end
36WorkTask(d, r, u) = WorkTask(d, r, u, 0, nothing, 0, 0, nothing)
37duration(t) = t.base-t.crashed

72function gettask(p, name)
73 check(haskey(p.tasks, name), "unknown task $name")
74 p.tasks[name]
75end
76tasknames(p) = sort!(collect(keys(p.tasks)))
77preds(p, name) = sort!([dep for dep in values(p.links) if dep.b==name]; by = x->x.a)
78succs(p, name) = sort!([dep for dep in values(p.links) if dep.a==name]; by = x->x.b)
79function topo(p)
80 pending=Set(tasknames(p))
81 order=String[]
82 while !isempty(pending)
83 ready=[
84 name for
85 name in sort!(collect(pending)) if all(dep->!(dep.a in pending), preds(p, name))
86 ]
87 isempty(ready) && break
88 name=first(ready)
89 delete!(pending, name)
90 push!(order, name)
91 end
92 order
93end
94function cyclenotation(p, order)
95 remaining=setdiff(Set(tasknames(p)), Set(order))
96 walk=[minimum(remaining)]
97 while true
98 previous=first([dep.a for dep in preds(p, last(walk)) if dep.a in remaining])
99 at=findfirst(==(previous), walk)
100 if at!==nothing
101 cycle=[previous; reverse(walk[(at+1):end])]
102 firstat=argmin(cycle)
103 cycle=[cycle[firstat:end]; cycle[1:(firstat-1)]]
104 return join([cycle; first(cycle)], " -> ")
105 end
106 push!(walk, previous)
107 end
108end
109function analysisorder(p)
110 check(!isempty(p.tasks), "no tasks")
111 order=topo(p)
112 check(
113 length(order)==length(p.tasks),
114 "cycle "*(length(order)<length(p.tasks) ? cyclenotation(p, order) : ""),
115 )
116 order
117end
118startbound(dep, starts, finishes, d) =
119 dep.kind=="FS" ? finishes[dep.a]+dep.lag :
120 dep.kind=="SS" ? starts[dep.a]+dep.lag : finishes[dep.a]+dep.lag-d
121finishbound(dep, starts, finishes, d) =
122 dep.kind=="FS" ? starts[dep.b]-dep.lag :
123 dep.kind=="SS" ? starts[dep.b]-dep.lag+d : finishes[dep.b]-dep.lag
124margin(s, dep) =
125 dep.kind=="FS" ? s.es[dep.b]-s.ef[dep.a]-dep.lag :
126 dep.kind=="SS" ? s.es[dep.b]-s.es[dep.a]-dep.lag : s.ef[dep.b]-s.ef[dep.a]-dep.lag
127function schedulefields(s, name)
128 [s.es[name], s.ef[name], s.ls[name], s.lf[name], s.tf[name], s.ff[name]]
129end
130critical(s) = Set(name for (name, value) in s.tf if value==0)
131function compute_schedule(p, order, changed = nothing, replacement = 0)
132 durations=Dict(name=>duration(task) for (name, task) in p.tasks)
133 changed!==nothing && (durations[changed]=replacement)
134 es=Dict{String,Int}()
135 ef=Dict{String,Int}()
136 ls=Dict{String,Int}()
137 lf=Dict{String,Int}()
138 for name in order
139 es[name]=maximum(
140 [
141 p.tasks[name].notbefore;
142 [startbound(dep, es, ef, durations[name]) for dep in preds(p, name)]
143 ],
144 )
145 ef[name]=es[name]+durations[name]
146 end
147 projectlength=maximum(values(ef))
148 for name in reverse(order)
149 lf[name]=minimum(
150 [projectlength; [finishbound(dep, ls, lf, durations[name]) for dep in succs(p, name)]],
151 )
152 ls[name]=lf[name]-durations[name]
153 end
154 tf=Dict(name=>ls[name]-es[name] for name in order)
155 ff=Dict{String,Int}()
156 result=Schedule(durations, es, ef, ls, lf, tf, ff, projectlength)
157 for name in order
158 ff[name]=minimum([projectlength-ef[name]; [margin(result, dep) for dep in succs(p, name)]])
159 end
160 result
161end
162function criticalpaths(p, s)
163 selected=critical(s)
164 links=Dict(
165 name=>[dep.b for dep in succs(p, name) if dep.b in selected && margin(s, dep)==0] for
166 name in selected
167 )
168 targets=Set([name for following in values(links) for name in following])
169 result=Vector{String}[]
170 function visit(path)
171 following=links[last(path)]
172 if isempty(following)
173 push!(result, path)
174 else
175 for next in following
176 visit([path; next])
177 end
178 end
179 end
180 for name in sort!(collect(setdiff(selected, targets)))
181 visit([name])
182 end
183 result
184end
185function resourceusage(p, resource, starts, finishes)
186 usage=Dict{Int,Int}()
187 for (name, task) in p.tasks
188 task.resource==resource || continue
189 for day in starts[name]:(finishes[name]-1)
190 usage[day]=get(usage, day, 0)+task.units
191 end
192 end
193 usage
194end
195working(p, date) = dayofweek(date)<=5 && !(date in p.holidays)
196function workingdates(p, count)
197 result=Date[]
198 date=p.start
199 while length(result)<count
200 working(p, date) && push!(result, date)
201 date+=Day(1)
202 end
203 result
204end
205function workbetween(p, after, until)
206 total=0
207 date=after+Day(1)
208 while date<=until
209 total+=working(p, date)
210 date+=Day(1)
211 end
212 total
213end
214function calendarreport(p, s)
215 dates=workingdates(p, s.length)
216 for name in sort!(tasknames(p); by = x->(s.es[x], x))
217 println("$name $(datetext(dates[s.es[name]+1])) - $(datetext(dates[s.ef[name]]))")
218 end
219 spanstart=first(dates)
220 finish=last(dates)
221 println(
222 "project $(datetext(spanstart)) - $(datetext(finish)), $(s.length) working days, $(Dates.value(finish-spanstart)+1) calendar days",
223 )
224 if p.deadline!==nothing
225 if finish<=p.deadline
226 println(
227 "deadline $(datetext(p.deadline)) met, $(workbetween(p,finish,p.deadline)) working days to spare",
228 )
229 else
230 println(
231 "deadline $(datetext(p.deadline)) missed by $(workbetween(p,p.deadline,finish)) working days",
232 )
233 end
234 end
235 holidays=sort!([date for date in p.holidays if dayofweek(date)<=5 && spanstart<=date<=finish])
236 println("holidays in span: ", isempty(holidays) ? "none" : join(datetext.(holidays), ", "))
237end
238function levelreport(p, s, limit)
239 resources=sort!(unique([task.resource for task in values(p.tasks)]))
240 capacity=Dict(r=>get(p.capacities, r, limit) for r in resources)
241 for name in tasknames(p)
242 task=p.tasks[name]
243 check(
244 task.units<=capacity[task.resource],
245 "capacity $(capacity[task.resource]) too small for $name",
246 )
247 end
248 starts=Dict{String,Int}()
249 finishes=Dict{String,Int}()
250 usage=Dict{Tuple{String,Int},Int}()
251 order=String[]
252 while length(order)<length(p.tasks)
253 ready=[
254 name for name in tasknames(p) if
255 !haskey(starts, name) && all(dep->haskey(starts, dep.a), preds(p, name))
256 ]
257 name=ready[argmin([(s.ls[x], x) for x in ready])]
258 task=p.tasks[name]
259 d=duration(task)
260 start=maximum(
261 [task.notbefore; [startbound(dep, starts, finishes, d) for dep in preds(p, name)]],
262 )
263 while any(
264 day->get(usage, (task.resource, day), 0)+task.units>capacity[task.resource],
265 start:(start+d-1),
266 )
267 start+=1
268 end
269 starts[name]=start
270 finishes[name]=start+d
271 push!(order, name)
272 for day in start:(start+d-1)
273 key=(task.resource, day)
274 usage[key]=get(usage, key, 0)+task.units
275 end
276 end
277 println("level capacity $limit: duration $(s.length) -> $(maximum(values(finishes)))")
278 for name in order
279 println(
280 "$name start $(starts[name]) finish $(finishes[name]) shift $(starts[name]-s.es[name])",
281 )
282 end
283 for resource in resources
284 before=maximum(values(resourceusage(p, resource, s.es, s.ef)); init = 0)
285 after=maximum(values(resourceusage(p, resource, starts, finishes)); init = 0)
286 println("resource $resource capacity $(capacity[resource]) peak $before -> $after")
287 end
288end
289function crash!(p, order, budget)
290 remaining=budget
291 steps=0
292 s=compute_schedule(p, order)
293 initial=s.length
294 reason=""
295 while true
296 candidates=[
297 name for name in tasknames(p) if s.tf[name]==0 &&
298 p.tasks[name].crashlimit!==nothing &&
299 p.tasks[name].crashed<p.tasks[name].crashlimit &&
300 compute_schedule(p, order, name, duration(p.tasks[name])-1).length<s.length
301 ]
302 if isempty(candidates)
303 reason="no candidate"
304 break
305 end
306 name=candidates[argmin([(p.tasks[x].crashcost, x) for x in candidates])]
307 task=p.tasks[name]
308 if task.crashcost>remaining
309 reason="insufficient budget"
310 break
311 end
312 old=duration(task)
313 task.crashed+=1
314 remaining-=task.crashcost
315 steps+=1
316 s=compute_schedule(p, order)
317 println(
318 "step $steps: $name $old -> $(duration(task)), cost $(task.crashcost), project $(s.length)",
319 )
320 end
321 println(
322 "crash done: $steps days, cost $(budget-remaining), project $initial -> $(s.length), budget left $remaining, $reason",
323 )
324end
325function execute!(p, line)
326 words=String.(split(line, ' '))
327 command=words[1]
328 args=words[2:end]
329 if command=="TASK"
330 name=args[1]
331 check(!haskey(p.tasks, name), "task $name already exists")
332 p.tasks[name]=WorkTask(parse(Int, args[2]), args[3], parse(Int, args[4]))
333 println("task $name added")
334 elseif command in ("DEP", "UNDEP")
335 a, b=args[1:2]
336 gettask(p, a)
337 gettask(p, b)
338 key=(a, b)
339 if command=="DEP"
340 check(a!=b, "self dependency $a")
341 check(!haskey(p.links, key), "dependency $a -> $b already exists")
342 dep=Dependency(a, b, args[3], parse(Int, args[4]))
343 p.links[key]=dep
344 println("dependency $a -> $b $(dep.kind)$(signedtext(dep.lag)) added")
345 else
346 check(haskey(p.links, key), "no dependency $a -> $b")
347 delete!(p.links, key)
348 println("dependency $a -> $b removed")
349 end
350 elseif command=="REMOVE"
351 name=args[1]
352 gettask(p, name)
353 selected=[key for key in keys(p.links) if name in key]
354 for key in selected
355 delete!(p.links, key)
356 end
357 delete!(p.tasks, name)
358 println("task $name removed with $(length(selected)) dependencies")
359 elseif command in ("NOTBEFORE", "CRASHCOST", "UNCRASH", "PERT")
360 name=args[1]
361 task=gettask(p, name)
362 if command=="NOTBEFORE"
363 task.notbefore=parse(Int, args[2])
364 println("notbefore $name $(args[2])")
365 elseif command=="CRASHCOST"
366 limit=parse(Int, args[2])
367 check(limit<task.base, "crash limit too large for $name")
368 check(limit>=task.crashed, "crash limit below used days for $name")
369 task.crashlimit=limit
370 task.crashcost=parse(Int, args[3])
371 println("crash $name max $limit cost $(args[3])")
372 elseif command=="UNCRASH"
373 println("uncrash $name restored $(task.crashed) days")
374 task.crashed=0
375 else
376 o, m, pe=parse.(Int, args[2:4])
377 check(o<=m<=pe, "invalid estimate")
378 task.estimate=(o, m, pe)
379 println("pert $name expected $(decimals(o+4m+pe,6)) variance $(decimals((pe-o)^2,36))")
380 end
381 elseif command=="CAPACITY"
382 p.capacities[args[1]]=parse(Int, args[2])
383 println("capacity $(args[1]) $(args[2])")
384 elseif command in ("START", "HOLIDAY", "DEADLINE")
385 date=parsedate(args[1])
386 if command=="START"
387 p.start=date
388 elseif command=="DEADLINE"
389 p.deadline=date
390 else
391 check(!(date in p.holidays), "duplicate holiday $(args[1])")
392 push!(p.holidays, date)
393 end
394 println(lowercase(command), " ", datetext(date))
395 elseif command=="LIST"
396 println(
397 "tasks $(length(p.tasks)), dependencies $(length(p.links)), holidays $(length(p.holidays))",
398 )
399 for name in tasknames(p)
400 task=p.tasks[name]
401 previous=[dep.a for dep in preds(p, name)]
402 println(
403 "$name duration $(duration(task)) resource $(task.resource) units $(task.units) preds ",
404 isempty(previous) ? "none" : join(previous, ' '),
405 )
406 end
407 elseif command=="SHOW"
408 name=args[1]
409 task=gettask(p, name)
410 println(
411 "task $name duration $(duration(task)) resource $(task.resource) units $(task.units)",
412 task.notbefore==0 ? "" : " notbefore $(task.notbefore)",
413 )
414 for (title, links) in (("preds", preds(p, name)), ("succs", succs(p, name)))
415 entries=[
416 "$(title=="preds" ? dep.a : dep.b) $(dep.kind)$(signedtext(dep.lag))" for
417 dep in links
418 ]
419 println("$title: ", isempty(entries) ? "none" : join(entries, ", "))
420 end
421 println(
422 "crash: ",
423 task.crashlimit===nothing ? "none" :
424 "max $(task.crashlimit) used $(task.crashed) cost $(task.crashcost)",
425 )
426 println("pert: ", task.estimate===nothing ? "none" : join(task.estimate, ' '))
427 order=topo(p)
428 if length(order)<length(p.tasks)
429 println("schedule: unavailable")
430 else
431 s=compute_schedule(p, order)
432 println(
433 "schedule: ",
434 join(
435 [
436 "$label $value" for (label, value) in
437 zip(["ES", "EF", "LS", "LF", "TF", "FF"], schedulefields(s, name))
438 ],
439 ' ',
440 ),
441 s.tf[name]==0 ? " critical" : "",
442 )
443 end
444 elseif command=="IMPACT"
445 name=args[1]
446 gettask(p, name)
447 seen=Set{String}()
448 queue=[name]
449 at=1
450 while at<=length(queue)
451 current=queue[at]
452 at+=1
453 for dep in succs(p, current)
454 if !(dep.b in seen)
455 push!(seen, dep.b)
456 push!(queue, dep.b)
457 end
458 end
459 end
460 delete!(seen, name)
461 println("impact $name: ", isempty(seen) ? "none" : namelist(seen))
462 else
463 order=analysisorder(p)
464 if command=="CALENDAR"
465 check(p.start!==nothing, "no start date")
466 elseif command=="LOAD"
467 check(any(t->t.resource==args[1], values(p.tasks)), "unknown resource $(args[1])")
468 elseif command=="WHATIF"
469 gettask(p, args[1])
470 elseif command=="VARIANCE"
471 check(p.baseline!==nothing, "no baseline")
472 end
473 s=compute_schedule(p, order)
474 if command=="CPM"
475 width=max(4, maximum(length, tasknames(p)))
476 println(
477 rpad("TASK", width),
478 join(lpad.(["DUR", "ES", "EF", "LS", "LF", "TF", "FF"], 5)),
479 )
480 for name in tasknames(p)
481 println(
482 rpad(name, width),
483 join(lpad.(string.([s.d[name]; schedulefields(s, name)]), 5)),
484 s.tf[name]==0 ? " *" : "",
485 )
486 end
487 println("project duration $(s.length)")
488 elseif command=="CRITICAL"
489 paths=criticalpaths(p, s)
490 println("critical paths: $(length(paths))")
491 for path in paths
492 println(join(path, " -> "))
493 end
494 elseif command=="SLACK"
495 for f in sort!(unique(collect(values(s.tf))))
496 println("float $f: ", namelist([name for name in tasknames(p) if s.tf[name]==f]))
497 end
498 elseif command=="DRAG"
499 for name in sort!(collect(critical(s)))
500 println("$name drag $(s.length-compute_schedule(p,order,name,0).length)")
501 end
502 elseif command=="STATS"
503 n=length(p.tasks)
504 c=length(critical(s))
505 counts=[count(dep->dep.kind==kind, values(p.links)) for kind in ("FS", "SS", "FF")]
506 println(
507 "tasks $n, critical $c, dependencies $(length(p.links)) (FS $(counts[1]), SS $(counts[2]), FF $(counts[3]))",
508 )
509 resources=sort!(unique([t.resource for t in values(p.tasks)]))
510 totals=[
511 "$r $(sum(duration(t)*t.units for t in values(p.tasks) if t.resource==r))" for
512 r in resources
513 ]
514 println("work ", join(totals, ", "))
515 total=sum(values(s.tf))
516 println(
517 "float total $total, average $(decimals(total,n))\nduration $(s.length), critical share $(decimals(100c,n))%",
518 )
519 elseif command=="CALENDAR"
520 calendarreport(p, s)
521 elseif command=="GANTT"
522 check(s.length<=60, "chart too wide ($(s.length) days)")
523 width=maximum(length, tasknames(p))
524 println(repeat(" ", width), " |", join(k%10 for k in 0:(s.length-1)), "|")
525 for name in sort!(tasknames(p); by = x->(s.es[x], x))
526 bar=join(
527 s.es[name]<=k<s.ef[name] ? (s.tf[name]==0 ? '=' : '-') :
528 s.ef[name]<=k<s.lf[name] ? '.' : ' ' for k in 0:(s.length-1)
529 )
530 println(rpad(name, width), " |", bar, "|")
531 end
532 elseif command=="LEVEL"
533 levelreport(p, s, parse(Int, args[1]))
534 elseif command=="LOAD"
535 resource=args[1]
536 usage=resourceusage(p, resource, s.es, s.ef)
537 daily=[get(usage, k, 0) for k in 0:(s.length-1)]
538 peak=maximum(daily)
539 println("load $resource: ", join(daily, ' '))
540 println("peak $peak on day $(findfirst(==(peak),daily)-1)")
541 haskey(p.capacities, resource) && println(
542 "capacity $(p.capacities[resource]) exceeded on $(count(>(p.capacities[resource]),daily)) days",
543 )
544 elseif command=="PERTSUM"
545 for path in criticalpaths(p, s)
546 mean=0
547 variance=0
548 for name in path
549 task=p.tasks[name]
550 o, m, pe=task.estimate===nothing ?
551 (duration(task), duration(task), duration(task)) : task.estimate
552 mean+=o+4m+pe
553 variance+=(pe-o)^2
554 end
555 println(
556 join(path, " -> "),
557 ": expected $(decimals(mean,6)) variance $(decimals(variance,36))",
558 )
559 end
560 elseif command=="CRASH"
561 crash!(p, order, parse(Int, args[1]))
562 elseif command=="WHATIF"
563 name=args[1]
564 replacement=parse(Int, args[2])
565 after=compute_schedule(p, order, name, replacement)
566 println(
567 "whatif $name $(duration(p.tasks[name])) -> $replacement: project $(s.length) -> $(after.length)",
568 )
569 for other in tasknames(p)
570 changes=[
571 "$label $old->$new" for (label, old, new) in zip(
572 ("ES", "EF", "TF"),
573 (s.es[other], s.ef[other], s.tf[other]),
574 (after.es[other], after.ef[other], after.tf[other]),
575 ) if old!=new
576 ]
577 !isempty(changes) && println(other, " ", join(changes, ' '))
578 end
579 critical(s)!=critical(after) &&
580 println("critical ", namelist(critical(s)), " -> ", namelist(critical(after)))
581 elseif command=="BASELINE"
582 p.baseline=s
583 println("baseline saved: $(length(p.tasks)) tasks, duration $(s.length)")
584 elseif command=="VARIANCE"
585 before=p.baseline
586 println(
587 "variance: duration $(before.length) -> $(s.length) ($(signedtext(s.length-before.length)))",
588 )
589 printed=false
590 for name in sort!(collect(union(Set(keys(before.es)), Set(tasknames(p)))))
591 if !haskey(p.tasks, name)
592 println("$name removed")
593 printed=true
594 elseif !haskey(before.es, name)
595 println("$name new")
596 printed=true
597 elseif (before.es[name], before.ef[name])!=(s.es[name], s.ef[name])
598 println(
599 "$name start $(signedtext(s.es[name]-before.es[name])) finish $(signedtext(s.ef[name]-before.ef[name]))",
600 )
601 printed=true
602 end
603 end
604 !printed && println("no task changes")
605 end
606 end
607end
608function main()
609 project=Project()
610 n=parse(Int, readline())
611 for _ in 1:n
612 line=readline()
613 try
614 execute!(project, line)
615 catch e
616 e isa CommandError || rethrow()
617 println("error: $(e.message)")
618 end
619 end
620end
