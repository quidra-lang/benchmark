18fn leap(y: i64) -> bool {
19 y%4 == 0 && (y%100 != 0 || y%400 == 0)
20}
21
22fn month_days(y: i64,m: i64) -> i64 {
23 if m == 2 && leap(y) {
24 29
25 } else {
26 [31,28,31,30,31,30,31,31,30,31,30,31][m as usize-1]
27 }
28}

30fn date(text: &str) -> Result<i64,String> {
31 let p: Vec<i64> = text.split('-').map(|s| s.parse().unwrap()).collect();
32 let (y,m,d) = (p[0],p[1],p[2]);
33 if !(1..=12).contains(&m) || d < 1 || d > month_days(y,m) {
34 return Err(fail(format!("invalid date {}",text)));
35 }
36 let day: i64 = (2000..y).map(|year| if leap(year) {
37 366
38 } else {
39 365
40 }).sum::<i64>()+(1..m).map(|month| month_days(y,month)).sum::<i64>();
41 Ok(day+d-1)
42}
43
44fn date_text(day: i64) -> String {
45 let mut left = day;
46 let mut y = 2000;
47 while left >= if leap(y) {
48 366
49 } else {
50 365
51 }
52 {
53 left -= if leap(y) {
54 366
55 } else {
56 365
57 };
58 y += 1;
59 }
60 let mut m = 1;
61 while left >= month_days(y,m) {
62 left -= month_days(y,m);
63 m += 1;
64 }
65 format!("{:04}-{:02}-{:02} {}",y,m,left+1,["Mon","Tue","Wed","Thu","Fri","Sat","Sun"][(day as usize+5)%7])
66}

124impl Project {
125 fn task(&self,n: &str) -> Result<&Task,String> {
126 self.tasks.get(n).ok_or_else(|| fail(format!("unknown task {}",n)))
127 }
128 fn preds(&self,n: &str) -> Vec<&Dep> {
129 let mut p: Vec<_> = self.deps.iter().filter(|e| e.b == n).collect();
130 p.sort_by(|x,y| x.a.cmp(&y.a));
131 p
132 }
133 fn succs(&self,n: &str) -> Vec<&Dep> {
134 let mut p: Vec<_> = self.deps.iter().filter(|e| e.a == n).collect();
135 p.sort_by(|x,y| x.b.cmp(&y.b));
136 p
137 }
138 fn order(&self) -> (Vec<String>,Option<String>) {
139 let mut remaining: BTreeSet<String> = self.tasks.keys().cloned().collect();
140 let mut order = Vec::new();
141 while let Some(n) = remaining.iter().find(|n| self.preds(n).iter().all(|e| !remaining.contains(&e.a))).cloned() {
142 remaining.remove(&n);
143 order.push(n);
144 }
145 if remaining.is_empty() {
146 return (order,None);
147 }
148 let mut walked = vec![remaining.first().unwrap().clone()];
149 loop {
150 let n = self.preds(walked.last().unwrap()).into_iter().find(|e| remaining.contains(&e.a)).unwrap().a.clone();
151 if let Some(q) = walked.iter().position(|t| *t == n) {
152 let mut cycle = vec![n];
153 cycle.extend(walked[q+1..].iter().rev().cloned());
154 let k = (0..cycle.len()).min_by_key(|&i| &cycle[i]).unwrap();
155 cycle.rotate_left(k);
156 cycle.push(cycle[0].clone());
157 return (order,Some(cycle.join(" -> ")));
158 }
159 walked.push(n);
160 }
161 }
162 fn schedule(&self,changed: Option<(&str,i64)>) -> Schedule {
163 let order = self.order().0;
164 let mut times: BTreeMap<String,Times> = BTreeMap::new();
165 for n in &order {
166 let t = &self.tasks[n];
167 let d = changed.filter(|(name,_)| *name == n.as_str()).map_or(t.duration(),|(_,d)| d);
168 let mut es = t.not_before;
169 for e in self.preds(n) {
170 let p = &times[&e.a];
171 es = es.max(match e.kind.as_str() {
172 "FS" => p.ef+e.lag, "SS" => p.es+e.lag, _ => p.ef+e.lag-d
173 });
174 }
175 times.insert(n.clone(),Times {
176 duration:d,es,ef:es+d,ls:0,lf:0,tf:0,ff:0
177 });
178 }
179 let total = times.values().map(|t| t.ef).max().unwrap();
180 for n in order.iter().rev() {
181 let d = times[n].duration;
182 let mut lf = total;
183 for e in self.succs(n) {
184 let p = &times[&e.b];
185 lf = lf.min(match e.kind.as_str() {
186 "FS" => p.ls-e.lag,"SS" => p.ls-e.lag+d,_ => p.lf-e.lag
187 });
188 }
189 let t = times.get_mut(n).unwrap();
190 t.lf = lf;
191 t.ls = lf-d;
192 t.tf = t.ls-t.es;
193 }
194 let mut s = Schedule {
195 times,total
196 };
197 for n in &order {
198 let ff = self.succs(n).iter().map(|e| self.margin(e,&s)).fold(total-s.times[n].ef,i64::min);
199 s.times.get_mut(n).unwrap().ff = ff;
200 }
201 s
202 }
203 fn margin(&self,e: &Dep,s: &Schedule) -> i64 {
204 let a = &s.times[&e.a];
205 let b = &s.times[&e.b];
206 match e.kind.as_str() {
207 "FS" => b.es-a.ef-e.lag,"SS" => b.es-a.es-e.lag,_ => b.ef-a.ef-e.lag
208 }
209 }
210 fn critical(&self,s: &Schedule) -> Vec<String> {
211 self.tasks.keys().filter(|n| s.times[*n].tf == 0).cloned().collect()
212 }
213 fn paths(&self,s: &Schedule) -> Vec<Vec<String>> {
214 let links: Vec<&Dep> = self.deps.iter().filter(|e| s.times[&e.a].tf == 0 && s.times[&e.b].tf == 0 && self.margin(e,s) == 0).collect();
215 let mut result = Vec::new();
216 fn visit(p: Vec<String>,links: &[&Dep],result: &mut Vec<Vec<String>>) {
217 let mut next: Vec<String> = links.iter().filter(|e| e.a == *p.last().unwrap()).map(|e| e.b.clone()).collect();
218 next.sort();
219 if next.is_empty() {
220 result.push(p);
221 } else {
222 for n in next {
223 let mut q = p.clone();
224 q.push(n);
225 visit(q,links,result);
226 }
227 }
228 }
229 for n in self.critical(s) {
230 if !links.iter().any(|e| e.b == n) {
231 visit(vec![n],&links,&mut result);
232 }
233 }
234 result
235 }
236 fn working(&self,d: i64) -> bool {
237 (d+5)%7 < 5 && !self.holidays.contains(&d)
238 }
239 fn usage(&self,r: &str,times: &BTreeMap<String,(i64,i64)>,total: i64) -> Vec<i64> {
240 (0..total).map(|day| self.tasks.iter().filter(|(n,t)| t.resource == r && day >= times[*n].0 && day < times[*n].1).map(|(_,t)| t.units).sum()).collect()
241 }
242 fn command(&mut self,w: &[&str]) -> Answer {
243 let c = w[0];
244 let a = &w[1..];
245 let num = |i: usize| -> i64 {
246 a[i].parse().unwrap()
247 };
248 let mut out = Vec::new();
249 match c {
250 "TASK" => {
251 if self.tasks.contains_key(a[0]) {
252 return Err(fail(format!("task {} already exists",a[0])));
253 }
254 self.tasks.insert(a[0].into(),Task {
255 base:num(1),resource:a[2].into(),units:num(3),not_before:0,crash:None,used:0,pert:None
256 });
257 return Ok(vec![format!("task {} added",a[0])]);
258 }
259 "DEP"|"UNDEP" => {
260 self.task(a[0])?;
261 self.task(a[1])?;
262 let i = self.deps.iter().position(|e| e.a == a[0] && e.b == a[1]);
263 if c == "DEP" {
264 if a[0] == a[1] {
265 return Err(fail(format!("self dependency {}",a[0])));
266 }
267 if i.is_some() {
268 return Err(fail(format!("dependency {} -> {} already exists",a[0],a[1])));
269 }
270 self.deps.push(Dep {
271 a:a[0].into(),b:a[1].into(),kind:a[2].into(),lag:num(3)
272 });
273 return Ok(vec![format!("dependency {} -> {} {}{} added",a[0],a[1],a[2],signed(num(3)))]);
274 }
275 let i = i.ok_or_else(|| fail(format!("no dependency {} -> {}",a[0],a[1])))?;
276 self.deps.remove(i);
277 return Ok(vec![format!("dependency {} -> {} removed",a[0],a[1])]);
278 }
279 "REMOVE" => {
280 self.task(a[0])?;
281 let k = self.deps.iter().filter(|e| e.a == a[0] || e.b == a[0]).count();
282 self.deps.retain(|e| e.a != a[0] && e.b != a[0]);
283 self.tasks.remove(a[0]);
284 return Ok(vec![format!("task {} removed with {} dependencies",a[0],k)]);
285 }
286 "NOTBEFORE" => {
287 self.task(a[0])?;
288 self.tasks.get_mut(a[0]).unwrap().not_before = num(1);
289 return Ok(vec![format!("notbefore {} {}",a[0],a[1])]);
290 }
291 "CRASHCOST" => {
292 let t = self.task(a[0])?;
293 if num(1) >= t.base {
294 return Err(fail(format!("crash limit too large for {}",a[0])));
295 }
296 if num(1) < t.used {
297 return Err(fail(format!("crash limit below used days for {}",a[0])));
298 }
299 self.tasks.get_mut(a[0]).unwrap().crash = Some((num(1),num(2)));
300 return Ok(vec![format!("crash {} max {} cost {}",a[0],a[1],a[2])]);
301 }
302 "UNCRASH" => {
303 let used = self.task(a[0])?.used;
304 self.tasks.get_mut(a[0]).unwrap().used = 0;
305 return Ok(vec![format!("uncrash {} restored {} days",a[0],used)]);
306 }
307 "PERT" => {
308 self.task(a[0])?;
309 let (o,m,p) = (num(1),num(2),num(3));
310 if o > m || m > p {
311 return Err(fail("invalid estimate"));
312 }
313 self.tasks.get_mut(a[0]).unwrap().pert = Some([o,m,p]);
314 return Ok(vec![format!("pert {} expected {} variance {}",a[0],decimal(o+4*m+p,6),decimal((p-o)*(p-o),36))]);
315 }
316 "CAPACITY" => {
317 self.capacities.insert(a[0].into(),num(1));
318 return Ok(vec![format!("capacity {} {}",a[0],a[1])]);
319 }
320 "START"|"HOLIDAY"|"DEADLINE" => {
321 let d = date(a[0])?;
322 if c == "HOLIDAY" && self.holidays.contains(&d) {
323 return Err(fail(format!("duplicate holiday {}",a[0])));
324 }
325 match c {
326 "START" => self.start = Some(d),"DEADLINE" => self.deadline = Some(d),_ => {
327 self.holidays.insert(d);
328 }
329 }
330 return Ok(vec![format!("{} {}",c.to_lowercase(),date_text(d))]);
331 }
332 "LIST" => {
333 out.push(format!("tasks {}, dependencies {}, holidays {}",self.tasks.len(),self.deps.len(),self.holidays.len()));
334 for (n,t) in &self.tasks {
335 let p = self.preds(n).iter().map(|e| e.a.as_str()).collect::<Vec<_>>().join(" ");
336 out.push(format!("{} duration {} resource {} units {} preds {}",n,t.duration(),t.resource,t.units,if p.is_empty() {
337 "none"
338 } else {
339 &p
340 }));
341 }
342 return Ok(out);
343 }
344 "SHOW" => {
345 let n = a[0];
346 let t = self.task(n)?;
347 out.push(format!("task {} duration {} resource {} units {}{}",n,t.duration(),t.resource,t.units,if t.not_before != 0 {
348 format!(" notbefore {}",t.not_before)
349 } else {
350 String::new()
351 }));
352 for incoming in [true,false] {
353 let list = if incoming {
354 self.preds(n)
355 } else {
356 self.succs(n)
357 };
358 let p = list.iter().map(|e| format!("{} {}{}",if incoming {
359 &e.a
360 } else {
361 &e.b
362 },e.kind,signed(e.lag))).collect::<Vec<_>>().join(", ");
363 out.push(format!("{}: {}",if incoming {
364 "preds"
365 } else {
366 "succs"
367 },if p.is_empty() {
368 "none"
369 } else {
370 &p
371 }));
372 }
373 out.push(t.crash.map_or("crash: none".into(),|(m,c)| format!("crash: max {} used {} cost {}",m,t.used,c)));
374 out.push(t.pert.map_or("pert: none".into(),|p| format!("pert: {} {} {}",p[0],p[1],p[2])));
375 if self.order().1.is_some() {
376 out.push("schedule: unavailable".into());
377 } else {
378 let s = self.schedule(None);
379 let x = &s.times[n];
380 out.push(format!("schedule: ES {} EF {} LS {} LF {} TF {} FF {}{}",x.es,x.ef,x.ls,x.lf,x.tf,x.ff,if x.tf==0 {
381 " critical"
382 } else {
383 ""
384 }));
385 }
386 return Ok(out);
387 }
388 "IMPACT" => {
389 self.task(a[0])?;
390 let mut reached = BTreeSet::from([a[0].to_string()]);
391 loop {
392 let before = reached.len();
393 for e in &self.deps {
394 if reached.contains(&e.a) {
395 reached.insert(e.b.clone());
396 }
397 }
398 if reached.len() == before {
399 break;
400 }
401 }
402 reached.remove(a[0]);
403 let p = reached.into_iter().collect::<Vec<_>>().join(" ");
404 return Ok(vec![format!("impact {}: {}",a[0],if p.is_empty() {
405 "none"
406 } else {
407 &p
408 })]);
409 }
410 _ => {
411 }
412 }
413 if self.tasks.is_empty() {
414 return Err(fail("no tasks"));
415 }
416 if let Some(cycle) = self.order().1 {
417 return Err(fail(format!("cycle {}",cycle)));
418 }
419 match c {
420 "CALENDAR" if self.start.is_none() => return Err(fail("no start date")),"LEVEL" => {
421 for (n,t) in &self.tasks {
422 let cap = self.capacities.get(&t.resource).copied().unwrap_or(num(0));
423 if t.units > cap {
424 return Err(fail(format!("capacity {} too small for {}",cap,n)));
425 }
426 }
427 },"LOAD" if !self.tasks.values().any(|t| t.resource == a[0]) => return Err(fail(format!("unknown resource {}",a[0]))),"WHATIF" => {
428 self.task(a[0])?;
429 },"VARIANCE" if self.baseline.is_none() => return Err(fail("no baseline")),_ => {
430 }
431 }
432 let s = self.schedule(None);
433 let mut by_start: Vec<String> = self.tasks.keys().cloned().collect();
434 by_start.sort_by_key(|n| (s.times[n].es,n.clone()));
435 let resources: BTreeSet<String> = self.tasks.values().map(|t| t.resource.clone()).collect();
436 let initial: BTreeMap<String,(i64,i64)> = s.times.iter().map(|(n,t)| (n.clone(),(t.es,t.ef))).collect();
437 match c {
438 "CPM" => {
439 let width = 4.max(self.tasks.keys().map(|n| n.len()).max().unwrap());
440 let mut header = format!("{:<width$}","TASK");
441 for title in ["DUR","ES","EF","LS","LF","TF","FF"] {
442 header += &format!("{:>5}",title);
443 }
444 out.push(header);
445 for (n,x) in &s.times {
446 let mut row = format!("{:<width$}",n);
447 for v in [x.duration,x.es,x.ef,x.ls,x.lf,x.tf,x.ff] {
448 row += &format!("{:>5}",v);
449 }
450 if x.tf == 0 {
451 row += " *";
452 }
453 out.push(row);
454 }
455 out.push(format!("project duration {}",s.total));
456 }
457 "CRITICAL" => {
458 let paths = self.paths(&s);
459 out.push(format!("critical paths: {}",paths.len()));
460 out.extend(paths.iter().map(|p| p.join(" -> ")));
461 }
462 "SLACK" => {
463 let floats: BTreeSet<i64> = s.times.values().map(|t| t.tf).collect();
464 for f in floats {
465 out.push(format!("float {}: {}",f,s.times.iter().filter(|(_,t)| t.tf == f).map(|(n,_)| n.as_str()).collect::<Vec<_>>().join(" ")));
466 }
467 }
468 "DRAG" => for n in self.critical(&s) {
469 out.push(format!("{} drag {}",n,s.total-self.schedule(Some((&n,0))).total));
470 },
471 "STATS" => {
472 let count = self.critical(&s).len() as i64;
473 let sum: i64 = s.times.values().map(|t| t.tf).sum();
474 out.push(format!("tasks {}, critical {}, dependencies {} (FS {}, SS {}, FF {})",self.tasks.len(),count,self.deps.len(),self.deps.iter().filter(|e| e.kind == "FS").count(),self.deps.iter().filter(|e| e.kind == "SS").count(),self.deps.iter().filter(|e| e.kind == "FF").count()));
475 out.push(format!("work {}",resources.iter().map(|r| format!("{} {}",r,self.tasks.values().filter(|t| t.resource == *r).map(|t| t.duration()*t.units).sum::<i64>())).collect::<Vec<_>>().join(", ")));
476 out.push(format!("float total {}, average {}",sum,decimal(sum,self.tasks.len() as i64)));
477 out.push(format!("duration {}, critical share {}%",s.total,decimal(100*count,self.tasks.len() as i64)));
478 }
479 "CALENDAR" => {
480 let mut dates = Vec::new();
481 let mut day = self.start.unwrap();
482 while dates.len() < s.total as usize {
483 if self.working(day) {
484 dates.push(day);
485 }
486 day += 1;
487 }
488 for n in &by_start {
489 let x = &s.times[n];
490 out.push(format!("{} {} - {}",n,date_text(dates[x.es as usize]),date_text(dates[(x.ef-1) as usize])));
491 }
492 let first = dates[0];
493 let last = *dates.last().unwrap();
494 out.push(format!("project {} - {}, {} working days, {} calendar days",date_text(first),date_text(last),s.total,last-first+1));
495 if let Some(d) = self.deadline {
496 let count = (d.min(last)+1..=d.max(last)).filter(|&v| self.working(v)).count();
497 out.push(if last <= d {
498 format!("deadline {} met, {} working days to spare",date_text(d),count)
499 } else {
500 format!("deadline {} missed by {} working days",date_text(d),count)
501 });
502 }
503 let h = self.holidays.iter().filter(|&&d| d >= first && d <= last && (d+5)%7 < 5).map(|&d| date_text(d)).collect::<Vec<_>>().join(", ");
504 out.push(format!("holidays in span: {}",if h.is_empty() {
505 "none"
506 } else {
507 &h
508 }));
509 }
510 "GANTT" => {
511 if s.total > 60 {
512 return Err(fail(format!("chart too wide ({} days)",s.total)));
513 }
514 let width = self.tasks.keys().map(|n| n.len()).max().unwrap();
515 out.push(format!("{} |{}|"," ".repeat(width),(0..s.total).map(|d| (d%10).to_string()).collect::<String>()));
516 for n in &by_start {
517 let x = &s.times[n];
518 let bar: String = (0..s.total).map(|k| if k >= x.es && k < x.ef {
519 if x.tf == 0 {
520 '='
521 } else {
522 '-'
523 }
524 } else if k >= x.ef && k < x.lf {
525 '.'
526 } else {
527 ' '
528 }).collect();
529 out.push(format!("{:<width$} |{}|",n,bar));
530 }
531 }
532 "LEVEL" => {
533 let mut decided: BTreeMap<String,(i64,i64)> = BTreeMap::new();
534 let mut decision = Vec::new();
535 let mut daily: BTreeMap<(String,i64),i64> = BTreeMap::new();
536 while decided.len() < self.tasks.len() {
537 let n = self.tasks.keys().filter(|n| !decided.contains_key(*n) && self.preds(n).iter().all(|e| decided.contains_key(&e.a))).min_by_key(|n| (s.times[*n].ls,(*n).clone())).unwrap().clone();
538 let t = &self.tasks[&n];
539 let mut begin = t.not_before;
540 for e in self.preds(&n) {
541 let p = decided[&e.a];
542 begin = begin.max(match e.kind.as_str() {
543 "FS" => p.1+e.lag,"SS" => p.0+e.lag,_ => p.1+e.lag-t.duration()
544 });
545 }
546 let capacity = self.capacities.get(&t.resource).copied().unwrap_or(num(0));
547 while (begin..begin+t.duration()).any(|day| daily.get(&(t.resource.clone(),day)).copied().unwrap_or(0)+t.units > capacity) {
548 begin += 1;
549 }
550 let finish = begin+t.duration();
551 decided.insert(n.clone(),(begin,finish));
552 decision.push(n);
553 for day in begin..finish {
554 *daily.entry((t.resource.clone(),day)).or_default() += t.units;
555 }
556 }
557 let total = decided.values().map(|p| p.1).max().unwrap();
558 out.push(format!("level capacity {}: duration {} -> {}",a[0],s.total,total));
559 for n in decision {
560 let p = decided[&n];
561 out.push(format!("{} start {} finish {} shift {}",n,p.0,p.1,p.0-s.times[&n].es));
562 }
563 for r in &resources {
564 out.push(format!("resource {} capacity {} peak {} -> {}",r,self.capacities.get(r).copied().unwrap_or(num(0)),self.usage(r,&initial,s.total).iter().max().unwrap(),self.usage(r,&decided,total).iter().max().unwrap()));
565 }
566 }
567 "LOAD" => {
568 let load = self.usage(a[0],&initial,s.total);
569 let peak = *load.iter().max().unwrap();
570 out.push(format!("load {}: {}",a[0],load.iter().map(|n| n.to_string()).collect::<Vec<_>>().join(" ")));
571 out.push(format!("peak {} on day {}",peak,load.iter().position(|&x| x == peak).unwrap()));
572 if let Some(cap) = self.capacities.get(a[0]) {
573 out.push(format!("capacity {} exceeded on {} days",cap,load.iter().filter(|&x| x > cap).count()));
574 }
575 }
576 "PERTSUM" => for path in self.paths(&s) {
577 let mut expected = 0;
578 let mut variance = 0;
579 for n in &path {
580 let t = &self.tasks[n];
581 let p = t.pert.unwrap_or([t.duration();3]);
582 expected += p[0]+4*p[1]+p[2];
583 variance += (p[2]-p[0])*(p[2]-p[0]);
584 }
585 out.push(format!("{}: expected {} variance {}",path.join(" -> "),decimal(expected,6),decimal(variance,36)));
586 },
587 "CRASH" => {
588 let budget = num(0);
589 let mut remaining = budget;
590 let mut count = 0;
591 let reason;
592 loop {
593 let now = self.schedule(None);
594 let candidates: Vec<String> = self.critical(&now).into_iter().filter(|n| {
595 let t = &self.tasks[n];
596 t.crash.is_some_and(|(limit,_)| t.used < limit) && self.schedule(Some((n,t.duration()-1))).total < now.total
597 }).collect();
598 let n = match candidates.into_iter().min_by_key(|n| (self.tasks[n].crash.unwrap().1,n.clone())) {
599 Some(n) => n,None => {
600 reason = "no candidate";
601 break;
602 }
603 };
604 let cost = self.tasks[&n].crash.unwrap().1;
605 if cost > remaining {
606 reason = "insufficient budget";
607 break;
608 }
609 let before = self.tasks[&n].duration();
610 self.tasks.get_mut(&n).unwrap().used += 1;
611 remaining -= cost;
612 count += 1;
613 out.push(format!("step {}: {} {} -> {}, cost {}, project {}",count,n,before,self.tasks[&n].duration(),cost,self.schedule(None).total));
614 }
615 out.push(format!("crash done: {} days, cost {}, project {} -> {}, budget left {}, {}",count,budget-remaining,s.total,self.schedule(None).total,remaining,reason));
616 }
617 "WHATIF" => {
618 let after = self.schedule(Some((a[0],num(1))));
619 out.push(format!("whatif {} {} -> {}: project {} -> {}",a[0],self.tasks[a[0]].duration(),a[1],s.total,after.total));
620 for n in self.tasks.keys() {
621 let x = &s.times[n];
622 let y = &after.times[n];
623 let fields: String = [("ES",x.es,y.es),("EF",x.ef,y.ef),("TF",x.tf,y.tf)].iter().filter(|(_,p,q)| p != q).map(|(title,p,q)| format!(" {} {}->{}",title,p,q)).collect();
624 if !fields.is_empty() {
625 out.push(format!("{}{}",n,fields));
626 }
627 }
628 let before = self.critical(&s);
629 let after = self.critical(&after);
630 if before != after {
631 out.push(format!("critical {} -> {}",before.join(" "),after.join(" ")));
632 }
633 }
634 "BASELINE" => {
635 self.baseline = Some(Baseline {
636 total:s.total,times:initial
637 });
638 out.push(format!("baseline saved: {} tasks, duration {}",self.tasks.len(),s.total));
639 }
640 "VARIANCE" => {
641 let old = self.baseline.as_ref().unwrap();
642 out.push(format!("variance: duration {} -> {} ({})",old.total,s.total,signed(s.total-old.total)));
643 let names: BTreeSet<&String> = old.times.keys().chain(self.tasks.keys()).collect();
644 for n in names {
645 match (old.times.get(n),initial.get(n)) {
646 (None,Some(_)) => out.push(format!("{} new",n)),(Some(_),None) => out.push(format!("{} removed",n)),(Some(x),Some(y)) if x != y => out.push(format!("{} start {} finish {}",n,signed(y.0-x.0),signed(y.1-x.1))),_ => {
647 }
648 }
649 }
650 if out.len() == 1 {
651 out.push("no task changes".into());
652 }
653 }
654 _ => {
655 }
656 }
657 Ok(out)
658 }
659}
