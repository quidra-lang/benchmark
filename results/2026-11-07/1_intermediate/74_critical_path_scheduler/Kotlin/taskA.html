3private fun fail(s: String): Nothing = throw Problem("error: $s")
4
5private fun signed(n: Int) = if (n >= 0) "+$n" else "$n"

7private fun decimal(n: Int, d: Int): String {
8 val h = n.toLong()*100/d
9 return "${h/100}.${(h%100).toString().padStart(2,'0')}"
10}
11
12private fun leap(y: Int) = y%4 == 0 && (y%100 != 0 || y%400 == 0)
13
14private fun monthDays(y: Int, m: Int) = if (m == 2 && leap(y)) 29 else intArrayOf(31,28,31,30,31,30,31,31,30,31,30,31)[m-1]
15
16private fun date(text: String): Int {
17 val y = text.substring(0,4).toInt()
18 val m = text.substring(5,7).toInt()
19 val d = text.substring(8,10).toInt()
20 if (m !in 1..12 || d !in 1..monthDays(y,m)) fail("invalid date $text")
21 var day = 0
22 for (year in 2000 until y) day += if (leap(year)) 366 else 365
23 for (month in 1 until m) day += monthDays(y,month)
24 return day+d-1
25}
26
27private fun dateText(day: Int): String {
28 var left = day
29 var y = 2000
30 while (left >= if (leap(y)) 366 else 365) {
31 left -= if (leap(y)) 366 else 365
32 y++
33 }
34 var m = 1
35 while (left >= monthDays(y,m)) {
36 left -= monthDays(y,m)
37 m++
38 }
39 return "$y-${m.toString().padStart(2,'0')}-${(left+1).toString().padStart(2,'0')} ${listOf("Mon","Tue","Wed","Thu","Fri","Sat","Sun")[(day+5)%7]}"
40}

54private class Project {
55 val tasks = sortedMapOf<String,Task>()
56 val deps = mutableListOf<Dep>()
57 val holidays = sortedSetOf<Int>()
58 val capacities = sortedMapOf<String,Int>()
59 var start: Int? = null
60 var deadline: Int? = null
61 var baseline: Baseline? = null
62 fun task(n: String) = tasks[n] ?: fail("unknown task $n")
63 fun preds(n: String) = deps.filter {
64 it.b == n
65 }.sortedBy {
66 it.a
67 }
68 fun succs(n: String) = deps.filter {
69 it.a == n
70 }.sortedBy {
71 it.b
72 }
73 fun order(): Pair<List<String>,String?> {
74 val remaining = tasks.keys.toSortedSet()
75 val order = mutableListOf<String>()
76 while (true) {
77 val n = remaining.firstOrNull {
78 t -> preds(t).none {
79 it.a in remaining
80 }
81 }?: break
82 remaining.remove(n)
83 order.add(n)
84 }
85 if (remaining.isEmpty()) return order to null
86 val walked = mutableListOf(remaining.first())
87 while (true) {
88 val n = preds(walked.last()).first {
89 it.a in remaining
90 }.a
91 val q = walked.indexOf(n)
92 if (q >= 0) {
93 val cycle = listOf(n)+walked.drop(q+1).asReversed()
94 val k = cycle.indices.minBy {
95 cycle[it]
96 }
97 val rotated = cycle.drop(k)+cycle.take(k)
98 return order to (rotated+rotated.first()).joinToString(" -> ")
99 }
100 walked.add(n)
101 }
102 }
103 fun schedule(changed: String? = null,duration: Int = 0): Schedule {
104 val order = order().first
105 val times = sortedMapOf<String,Times>()
106 for (n in order) {
107 val t = task(n)
108 val d = if (n == changed) duration else t.duration
109 var es = t.notBefore
110 for (e in preds(n)) {
111 val p = times.getValue(e.a)
112 es = maxOf(es,when(e.kind) {
113 "FS" -> p.ef+e.lag
114 "SS" -> p.es+e.lag
115 else -> p.ef+e.lag-d
116 })
117 }
118 times[n] = Times(d,es,es+d)
119 }
120 val total = times.values.maxOf {
121 it.ef
122 }
123 for (n in order.asReversed()) {
124 val t = times.getValue(n)
125 var lf = total
126 for (e in succs(n)) {
127 val p = times.getValue(e.b)
128 lf = minOf(lf,when(e.kind) {
129 "FS" -> p.ls-e.lag
130 "SS" -> p.ls-e.lag+t.duration
131 else -> p.lf-e.lag
132 })
133 }
134 t.lf = lf
135 t.ls = lf-t.duration
136 t.tf = t.ls-t.es
137 }
138 val s = Schedule(times,total)
139 for (n in order) {
140 val t = times.getValue(n)
141 t.ff = minOf(total-t.ef,succs(n).minOfOrNull {
142 margin(it,s)
143 }?: Int.MAX_VALUE)
144 }
145 return s
146 }
147 fun margin(e: Dep,s: Schedule): Int {
148 val a = s.times.getValue(e.a)
149 val b = s.times.getValue(e.b)
150 return when(e.kind) {
151 "FS" -> b.es-a.ef-e.lag
152 "SS" -> b.es-a.es-e.lag
153 else -> b.ef-a.ef-e.lag
154 }
155 }
156 fun critical(s: Schedule) = tasks.keys.filter {
157 s.times.getValue(it).tf == 0
158 }
159 fun paths(s: Schedule): List<List<String>> {
160 val links = deps.filter {
161 s.times.getValue(it.a).tf == 0 && s.times.getValue(it.b).tf == 0 && margin(it,s) == 0
162 }
163 val result = mutableListOf<List<String>>()
164 fun visit(p: List<String>) {
165 val next = links.filter {
166 it.a == p.last()
167 }.map {
168 it.b
169 }.sorted()
170 if (next.isEmpty()) result.add(p) else for (n in next) visit(p+n)
171 }
172 for (n in critical(s)) if (links.none {
173 it.b == n
174 }) visit(listOf(n))
175 return result
176 }
177 fun working(d: Int) = (d+5)%7 < 5 && d !in holidays
178 fun usage(resource: String,times: Map<String,Pair<Int,Int>>,total: Int) = IntArray(total) {
179 day -> tasks.filter {
180 (n,t) -> t.resource == resource && day >= times.getValue(n).first && day < times.getValue(n).second
181 }.values.sumOf {
182 it.units
183 }
184 }
185 fun command(w: List<String>): List<String> {
186 val c = w[0]
187 val a = w.drop(1)
188 fun num(i: Int) = a[i].toInt()
189 val out = mutableListOf<String>()
190 when(c) {
191 "TASK" -> {
192 if (a[0] in tasks) fail("task ${a[0]} already exists")
193 tasks[a[0]] = Task(num(1),a[2],num(3))
194 return listOf("task ${a[0]} added")
195 }
196 "DEP","UNDEP" -> {
197 task(a[0])
198 task(a[1])
199 val i = deps.indexOfFirst {
200 it.a == a[0] && it.b == a[1]
201 }
202 if (c == "DEP") {
203 if (a[0] == a[1]) fail("self dependency ${a[0]}")
204 if (i >= 0) fail("dependency ${a[0]} -> ${a[1]} already exists")
205 deps.add(Dep(a[0],a[1],a[2],num(3)))
206 return listOf("dependency ${a[0]} -> ${a[1]} ${a[2]}${signed(num(3))} added")
207 }
208 if (i < 0) fail("no dependency ${a[0]} -> ${a[1]}")
209 deps.removeAt(i)
210 return listOf("dependency ${a[0]} -> ${a[1]} removed")
211 }
212 "REMOVE" -> {
213 task(a[0])
214 val k = deps.count {
215 it.a == a[0] || it.b == a[0]
216 }
217 deps.removeAll {
218 it.a == a[0] || it.b == a[0]
219 }
220 tasks.remove(a[0])
221 return listOf("task ${a[0]} removed with $k dependencies")
222 }
223 "NOTBEFORE" -> {
224 task(a[0]).notBefore = num(1)
225 return listOf("notbefore ${a[0]} ${a[1]}")
226 }
227 "CRASHCOST" -> {
228 val t = task(a[0])
229 if (num(1) >= t.base) fail("crash limit too large for ${a[0]}")
230 if (num(1) < t.used) fail("crash limit below used days for ${a[0]}")
231 t.crash = num(1) to num(2)
232 return listOf("crash ${a[0]} max ${a[1]} cost ${a[2]}")
233 }
234 "UNCRASH" -> {
235 val t = task(a[0])
236 out.add("uncrash ${a[0]} restored ${t.used} days")
237 t.used = 0
238 return out
239 }
240 "PERT" -> {
241 val t = task(a[0])
242 val o = num(1)
243 val m = num(2)
244 val p = num(3)
245 if (o > m || m > p) fail("invalid estimate")
246 t.pert = listOf(o,m,p)
247 return listOf("pert ${a[0]} expected ${decimal(o+4*m+p,6)} variance ${decimal((p-o)*(p-o),36)}")
248 }
249 "CAPACITY" -> {
250 capacities[a[0]] = num(1)
251 return listOf("capacity ${a[0]} ${a[1]}")
252 }
253 "START","HOLIDAY","DEADLINE" -> {
254 val d = date(a[0])
255 if (c == "HOLIDAY" && d in holidays) fail("duplicate holiday ${a[0]}")
256 when(c) {
257 "START" -> start = d
258 "DEADLINE" -> deadline = d
259 else -> holidays.add(d)
260 }
261 return listOf("${c.lowercase()} ${dateText(d)}")
262 }
263 "LIST" -> {
264 out.add("tasks ${tasks.size}, dependencies ${deps.size}, holidays ${holidays.size}")
265 for ((n,t) in tasks) out.add("$n duration ${t.duration} resource ${t.resource} units ${t.units} preds ${preds(n).joinToString(" ") { it.a }.ifEmpty { "none" }}")
266 return out
267 }
268 "SHOW" -> {
269 val n = a[0]
270 val t = task(n)
271 out.add("task $n duration ${t.duration} resource ${t.resource} units ${t.units}"+if (t.notBefore != 0) " notbefore ${t.notBefore}" else "")
272 out.add("preds: "+preds(n).joinToString(", ") {
273 "${it.a} ${it.kind}${signed(it.lag)}"
274 }.ifEmpty {
275 "none"
276 })
277 out.add("succs: "+succs(n).joinToString(", ") {
278 "${it.b} ${it.kind}${signed(it.lag)}"
279 }.ifEmpty {
280 "none"
281 })
282 out.add(t.crash?.let {
283 "crash: max ${it.first} used ${t.used} cost ${it.second}"
284 }?: "crash: none")
285 out.add(t.pert?.joinToString(" ",prefix="pert: ") ?: "pert: none")
286 if (order().second != null) out.add("schedule: unavailable") else {
287 val x = schedule().times.getValue(n)
288 out.add("schedule: ES ${x.es} EF ${x.ef} LS ${x.ls} LF ${x.lf} TF ${x.tf} FF ${x.ff}"+if (x.tf == 0) " critical" else "")
289 }
290 return out
291 }
292 "IMPACT" -> {
293 task(a[0])
294 val reached = mutableSetOf(a[0])
295 while (true) {
296 val next = deps.filter {
297 it.a in reached
298 }.map {
299 it.b
300 }
301 if (!reached.addAll(next)) break
302 }
303 reached.remove(a[0])
304 return listOf("impact ${a[0]}: "+reached.sorted().joinToString(" ").ifEmpty {
305 "none"
306 })
307 }
308 }
309 if (tasks.isEmpty()) fail("no tasks")
310 order().second?.let {
311 fail("cycle $it")
312 }
313 when(c) {
314 "CALENDAR" -> if (start == null) fail("no start date")
315 "LEVEL" -> for ((n,t) in tasks) {
316 val cap = capacities[t.resource] ?: num(0)
317 if (t.units > cap) fail("capacity $cap too small for $n")
318 }
319 "LOAD" -> if (tasks.values.none {
320 it.resource == a[0]
321 }) fail("unknown resource ${a[0]}")
322 "WHATIF" -> task(a[0])
323 "VARIANCE" -> if (baseline == null) fail("no baseline")
324 }
325 val s = schedule()
326 val byStart = tasks.keys.sortedWith(compareBy<String> {
327 s.times.getValue(it).es
328 }.thenBy {
329 it
330 })
331 val resources = tasks.values.map {
332 it.resource
333 }.toSortedSet()
334 val initialTimes = s.times.mapValues {
335 it.value.es to it.value.ef
336 }
337 when(c) {
338 "CPM" -> {
339 val width = maxOf(4,tasks.keys.maxOf {
340 it.length
341 })
342 out.add("TASK".padEnd(width)+listOf("DUR","ES","EF","LS","LF","TF","FF").joinToString("") {
343 it.padStart(5)
344 })
345 for (n in tasks.keys) {
346 val x = s.times.getValue(n)
347 out.add(n.padEnd(width)+listOf(x.duration,x.es,x.ef,x.ls,x.lf,x.tf,x.ff).joinToString("") {
348 it.toString().padStart(5)
349 }+if (x.tf == 0) " *" else "")
350 }
351 out.add("project duration ${s.total}")
352 }
353 "CRITICAL" -> {
354 val paths = paths(s)
355 out.add("critical paths: ${paths.size}")
356 out.addAll(paths.map {
357 it.joinToString(" -> ")
358 })
359 }
360 "SLACK" -> for ((f,names) in tasks.keys.groupBy {
361 s.times.getValue(it).tf
362 }.toSortedMap()) out.add("float $f: ${names.joinToString(" ")}")
363 "DRAG" -> for (n in critical(s)) out.add("$n drag ${s.total-schedule(n,0).total}")
364 "STATS" -> {
365 val critical = critical(s).size
366 val sum = s.times.values.sumOf {
367 it.tf
368 }
369 out.add("tasks ${tasks.size}, critical $critical, dependencies ${deps.size} (FS ${deps.count { it.kind == "FS" }}, SS ${deps.count { it.kind == "SS" }}, FF ${deps.count { it.kind == "FF" }})")
370 out.add("work "+resources.joinToString(", ") {
371 r -> "$r ${tasks.values.filter { it.resource == r }.sumOf { it.duration*it.units }}"
372 })
373 out.add("float total $sum, average ${decimal(sum,tasks.size)}")
374 out.add("duration ${s.total}, critical share ${decimal(100*critical,tasks.size)}%")
375 }
376 "CALENDAR" -> {
377 val dates = mutableListOf<Int>()
378 var day = start!!
379 while (dates.size < s.total) {
380 if (working(day)) dates.add(day)
381 day++
382 }
383 for (n in byStart) {
384 val x = s.times.getValue(n)
385 out.add("$n ${dateText(dates[x.es])} - ${dateText(dates[x.ef-1])}")
386 }
387 val first = dates.first()
388 val last = dates.last()
389 out.add("project ${dateText(first)} - ${dateText(last)}, ${s.total} working days, ${last-first+1} calendar days")
390 deadline?.let {
391 d -> val count = (minOf(d,last)+1..maxOf(d,last)).count {
392 working(it)
393 }
394 out.add(if (last <= d) "deadline ${dateText(d)} met, $count working days to spare" else "deadline ${dateText(d)} missed by $count working days")
395 }
396 out.add("holidays in span: "+holidays.filter {
397 it in first..last && (it+5)%7 < 5
398 }.joinToString(", ") {
399 dateText(it)
400 }.ifEmpty {
401 "none"
402 })
403 }
404 "GANTT" -> {
405 if (s.total > 60) fail("chart too wide (${s.total} days)")
406 val width = tasks.keys.maxOf {
407 it.length
408 }
409 out.add(" ".repeat(width)+" |"+(0 until s.total).joinToString("") {
410 (it%10).toString()
411 }+"|")
412 for (n in byStart) {
413 val x = s.times.getValue(n)
414 val bar = (0 until s.total).map {
415 if (it >= x.es && it < x.ef) {
416 if (x.tf == 0) '=' else '-'
417 } else if (it >= x.ef && it < x.lf) '.' else ' '
418 }.joinToString("")
419 out.add(n.padEnd(width)+" |"+bar+"|")
420 }
421 }
422 "LEVEL" -> {
423 val decided = mutableMapOf<String,Pair<Int,Int>>()
424 val decision = mutableListOf<String>()
425 val dailyUsage = mutableMapOf<Pair<String,Int>,Int>()
426 while (decided.size < tasks.size) {
427 val n = tasks.keys.filter {
428 it !in decided && preds(it).all {
429 d -> d.a in decided
430 }
431 }.minWith(compareBy<String> {
432 s.times.getValue(it).ls
433 }.thenBy {
434 it
435 })
436 val t = task(n)
437 var begin = t.notBefore
438 for (e in preds(n)) {
439 val p = decided.getValue(e.a)
440 begin = maxOf(begin,when(e.kind) {
441 "FS" -> p.second+e.lag
442 "SS" -> p.first+e.lag
443 else -> p.second+e.lag-t.duration
444 })
445 }
446 val capacity = capacities[t.resource] ?: num(0)
447 while ((begin until begin+t.duration).any {
448 (dailyUsage[t.resource to it] ?: 0)+t.units > capacity
449 }) begin++
450 val finish = begin+t.duration
451 decided[n] = begin to finish
452 decision.add(n)
453 for (d in begin until finish) dailyUsage[t.resource to d] = (dailyUsage[t.resource to d] ?: 0)+t.units
454 }
455 val total = decided.values.maxOf {
456 it.second
457 }
458 out.add("level capacity ${a[0]}: duration ${s.total} -> $total")
459 for (n in decision) {
460 val p = decided.getValue(n)
461 out.add("$n start ${p.first} finish ${p.second} shift ${p.first-s.times.getValue(n).es}")
462 }
463 for (r in resources) out.add("resource $r capacity ${capacities[r] ?: num(0)} peak ${usage(r,initialTimes,s.total).maxOrNull()!!} -> ${usage(r,decided,total).maxOrNull()!!}")
464 }
465 "LOAD" -> {
466 val load = usage(a[0],initialTimes,s.total)
467 val peak = load.maxOrNull()!!
468 out.add("load ${a[0]}: ${load.joinToString(" ")}")
469 out.add("peak $peak on day ${load.indexOf(peak)}")
470 capacities[a[0]]?.let {
471 cap -> out.add("capacity $cap exceeded on ${load.count { it > cap }} days")
472 }
473 }
474 "PERTSUM" -> for (path in paths(s)) {
475 var expected = 0
476 var variance = 0
477 for (n in path) {
478 val t = task(n)
479 val p = t.pert ?: listOf(t.duration,t.duration,t.duration)
480 expected += p[0]+4*p[1]+p[2]
481 variance += (p[2]-p[0])*(p[2]-p[0])
482 }
483 out.add("${path.joinToString(" -> ")}: expected ${decimal(expected,6)} variance ${decimal(variance,36)}")
484 }
485 "CRASH" -> {
486 val budget = num(0)
487 var remaining = budget
488 var count = 0
489 val reason: String
490 while (true) {
491 val now = schedule()
492 val candidates = critical(now).filter {
493 n -> val t = task(n)
494 t.crash != null && t.used < t.crash!!.first && schedule(n,t.duration-1).total < now.total
495 }
496 if (candidates.isEmpty()) {
497 reason = "no candidate"
498 break
499 }
500 val n = candidates.minWith(compareBy<String> {
501 task(it).crash!!.second
502 }.thenBy {
503 it
504 })
505 val t = task(n)
506 val cost = t.crash!!.second
507 if (cost > remaining) {
508 reason = "insufficient budget"
509 break
510 }
511 val before = t.duration
512 t.used++
513 remaining -= cost
514 count++
515 out.add("step $count: $n $before -> ${t.duration}, cost $cost, project ${schedule().total}")
516 }
517 out.add("crash done: $count days, cost ${budget-remaining}, project ${s.total} -> ${schedule().total}, budget left $remaining, $reason")
518 }
519 "WHATIF" -> {
520 val after = schedule(a[0],num(1))
521 out.add("whatif ${a[0]} ${task(a[0]).duration} -> ${a[1]}: project ${s.total} -> ${after.total}")
522 for (n in tasks.keys) {
523 val x = s.times.getValue(n)
524 val y = after.times.getValue(n)
525 val fields = listOf(Triple("ES",x.es,y.es),Triple("EF",x.ef,y.ef),Triple("TF",x.tf,y.tf)).filter {
526 it.second != it.third
527 }
528 if (fields.isNotEmpty()) out.add(n+fields.joinToString("") {
529 " ${it.first} ${it.second}->${it.third}"
530 })
531 }
532 val beforeCritical = critical(s)
533 val afterCritical = critical(after)
534 if (beforeCritical != afterCritical) out.add("critical ${beforeCritical.joinToString(" ")} -> ${afterCritical.joinToString(" ")}")
535 }
536 "BASELINE" -> {
537 baseline = Baseline(s.total,initialTimes)
538 out.add("baseline saved: ${tasks.size} tasks, duration ${s.total}")
539 }
540 "VARIANCE" -> {
541 val old = baseline!!
542 out.add("variance: duration ${old.total} -> ${s.total} (${signed(s.total-old.total)})")
543 for (n in (old.times.keys+tasks.keys).toSortedSet()) {
544 val x = old.times[n]
545 val y = initialTimes[n]
546 if (x == null) out.add("$n new") else if (y == null) out.add("$n removed") else if (x != y) out.add("$n start ${signed(y.first-x.first)} finish ${signed(y.second-x.second)}")
547 }
548 if (out.size == 1) out.add("no task changes")
549 }
550 }
551 return out
552 }
553}
