58int | error parse_date(string text)
59 year := int(text[:<4])
60 month := int(text[5:<7])
61 day := int(text[8:])
62 if month < 1 or month > 12
63 return error("invalid date {text}")
64 lengths := month_lengths(year)
65 if day < 1 or day > lengths[nat(month - 1)]
66 return error("invalid date {text}")
67 result := day - 1
68 for: 2000 <= y < year
69 result += if leap_year(y) then 366 else 365
70 for: 1 <= m < month
71 result += lengths[nat(m - 1)]
72 return result

74string date_text(int ordinal)
75 year := 2000
76 rest := ordinal
77 while rest >= (if leap_year(year) then 366 else 365)
78 rest -= if leap_year(year) then 366 else 365

83 rest -= lengths[month]
84 month += 1
85 weekday := ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"][(ordinal + 5) % 7]

234 Schedule compute_schedule(string[] order, changed := "", replacement := 0)
235 string[] names := task_names()
236 int[] durations := [for name := names[] then if name == changed then replacement else get_task(name).current_duration()]
237 es := array(#names, fill = 0)
238 ef := es
239 ls := es
240 lf := es
241 for name := order[]
242 index := position(names, name)
243 int start := get_task(name).not_before
244 for dep := pred_links(name)[]
245 bound := start_bound(dep, names, es, ef, durations[index])
246 if bound > start
247 start = bound
248 es[index] = start
249 ef[index] = start + durations[index]
250 project_length := maximum(ef)
251 for name := order[::-1][]
252 index := position(names, name)
253 finish := project_length
254 for dep := succ_links(name)[]
255 bound := finish_bound(dep, names, ls, lf, durations[index])
256 if bound < finish
257 finish = bound
258 lf[index] = finish
259 ls[index] = finish - durations[index]
260 tf := [for i := (nat(0):<#names)[] then ls[i] - es[i]]
261 ff := [for i := (nat(0):<#names)[] then project_length - ef[i]]
262 result := Schedule(names, durations, es, ef, ls, lf, tf, ff, project_length)
263 for: name := names[i]
264 for dep := succ_links(name)[]
265 value := result.margin(dep)
266 if value < ff[i]
267 ff[i] = value
268 result.ff = ff
269 return result
