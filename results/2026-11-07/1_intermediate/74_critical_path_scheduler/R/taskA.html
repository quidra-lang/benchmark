22topology <- function() {
23 remaining <- task_names()
24 result <- character()
25 repeat {
26 eligible <- remaining[vapply(remaining, function(t) !length(intersect(predecessors(t), remaining)), logical(1))]
27 if (!length(eligible)) break
28 name <- eligible[[1]]
29 remaining <- setdiff(remaining, name)
30 result <- c(result, name)
31 }
32 if (!length(remaining)) return(list(order = result, cycle = NULL))
33 walk <- remaining[[1]]
34 repeat {
35 following <- intersect(predecessors(tail(walk, 1L)), remaining)[[1]]
36 i <- match(following, walk)
37 if (!is.na(i)) {
38 cycle <- c(walk[[i]], rev(if (i < length(walk)) walk[(i + 1L):length(walk)] else character()))
39 first <- match(ordered(cycle)[[1]], cycle)
40 cycle <- c(cycle[first:length(cycle)], if (first > 1L) cycle[seq_len(first - 1L)] else character())
41 return(list(order = result, cycle = paste(c(cycle, cycle[[1]]), collapse = " -> ")))
42 }
43 walk <- c(walk, following)
44 }

46margin <- function(a, b, schedule) {
47 dep <- dependency(a, b)
48 switch(dep$kind, FS = schedule[[b]]$ES - schedule[[a]]$EF - dep$lag,
49 SS = schedule[[b]]$ES - schedule[[a]]$ES - dep$lag,
50 FF = schedule[[b]]$EF - schedule[[a]]$EF - dep$lag)
51}
52compute <- function(overrides = list()) {
53 order <- topology()$order
54 result <- list()
55 for (name in order) {
56 task <- state$tasks[[name]]
57 d <- overrides[[name]]
58 if (is.null(d)) d <- duration(task)
59 start <- task$day
60 for (a in predecessors(name)) {
61 dep <- dependency(a, name)
62 value <- switch(dep$kind, FS = result[[a]]$EF + dep$lag,
63 SS = result[[a]]$ES + dep$lag, FF = result[[a]]$EF + dep$lag - d)
64 start <- max(start, value)
65 }
66 result[[name]] <- list(DUR = d, ES = start, EF = start + d)
67 }
68 total <- max(vapply(result, function(v) v$EF, numeric(1)))
69 for (name in rev(order)) {
70 value <- result[[name]]
71 finish <- total
72 for (b in successors(name)) {
73 dep <- dependency(name, b)
74 candidate <- switch(dep$kind, FS = result[[b]]$LS - dep$lag,
75 SS = result[[b]]$LS - dep$lag + value$DUR, FF = result[[b]]$LF - dep$lag)
76 finish <- min(finish, candidate)
77 }
78 value$LF <- finish
79 value$LS <- finish - value$DUR
80 value$TF <- value$LS - value$ES
81 result[[name]] <- value
82 }
83 for (name in task_names()) {
84 floats <- vapply(successors(name), function(b) margin(name, b, result), numeric(1))
85 result[[name]]$FF <- min(c(total - result[[name]]$EF, floats))
86 }
87 list(schedule = result, total = total)
88}
89critical <- function(schedule) task_names()[vapply(task_names(), function(name) schedule[[name]]$TF == 0L, logical(1))]
90critical_paths <- function(schedule) {
91 selected <- critical(schedule)
92 edges <- Filter(function(d) d$a %in% selected && d$b %in% selected && margin(d$a, d$b, schedule) == 0L, state$deps)
93 roots <- selected[!vapply(selected, function(name) any(vapply(edges, function(d) d$b == name, logical(1))), logical(1))]
94 paths <- list()
95 visit <- function(path) {
96 children <- ordered(vapply(Filter(function(d) d$a == tail(path, 1L), edges), function(d) d$b, character(1)))
97 if (!length(children)) paths[[length(paths) + 1L]] <<- path else for (name in children) visit(c(path, name))
98 }
99 for (name in roots) visit(name)
100 paths
101}

107date_value <- function(text) {
108 parts <- as.integer(strsplit(text, "-", fixed = TRUE)[[1]])
109 year <- parts[[1]]
110 month <- parts[[2]]
111 day <- parts[[3]]
112 leap <- year %% 4L == 0L && (year %% 100L != 0L || year %% 400L == 0L)
113 lengths <- c(31L, if (leap) 29L else 28L, 31L, 30L, 31L, 30L, 31L, 31L, 30L, 31L, 30L, 31L)
114 if (month < 1L || month > 12L || day < 1L || day > lengths[[month]]) fail(paste("invalid date", text))
115 as.integer(as.Date(text, format = "%Y-%m-%d"))
116}
117weekday <- function(date) (date + 3L) %% 7L
118date_text <- function(date) paste(format(as.Date(date, origin = "1970-01-01"), "%Y-%m-%d"), c("Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun")[[weekday(date) + 1L]])
119working <- function(date) weekday(date) < 5L && !(date %in% state$holidays)
120calendar <- function(schedule, total) {
121 dates <- integer()
122 current <- state$start
123 while (length(dates) < total) {
124 if (working(current)) dates <- c(dates, current)
125 current <- current + 1L
126 }
127 names <- task_names()
128 starts <- vapply(names, function(name) schedule[[name]]$ES, numeric(1))
129 for (name in names[order(starts, names, method = "radix")]) {
130 say(name, " ", date_text(dates[[schedule[[name]]$ES + 1L]]), " - ", date_text(dates[[schedule[[name]]$EF]]))
131 }
132 first <- dates[[1]]
133 finish <- tail(dates, 1L)
134 say("project ", date_text(first), " - ", date_text(finish), ", ", total, " working days, ", finish - first + 1L, " calendar days")
135 if (!is.null(state$deadline)) {
136 lower <- min(finish, state$deadline)
137 upper <- max(finish, state$deadline)
138 days <- if (lower == upper) 0L else sum(vapply(seq.int(lower + 1L, upper), working, logical(1)))
139 description <- if (finish <= state$deadline) paste0("met, ", days, " working days to spare") else paste0("missed by ", days, " working days")
140 say("deadline ", date_text(state$deadline), " ", description)
141 }
142 holidays <- sort(state$holidays[state$holidays >= first & state$holidays <= finish & weekday(state$holidays) < 5L])
143 say("holidays in span: ", if (length(holidays)) paste(vapply(holidays, date_text, character(1)), collapse = ", ") else "none")
144}
145usage <- function(schedule, resource, total) {
146 daily <- rep(0L, total)
147 for (name in task_names()) {
148 task <- state$tasks[[name]]
149 if (task$resource != resource) next
150 days <- seq.int(schedule[[name]]$ES + 1L, schedule[[name]]$EF)
151 daily[days] <- daily[days] + task$units
152 }
153 daily
154}
155level <- function(limit, schedule, total) {
156 capacity <- function(r) if (is.null(state$capacities[[r]])) limit else state$capacities[[r]]
157 for (name in task_names()) {
158 task <- state$tasks[[name]]
159 if (task$units > capacity(task$resource)) fail(paste("capacity", capacity(task$resource), "too small for", name))
160 }
161 decided <- list()
162 order <- character()
163 daily <- list()
164 use <- function(r, days) {
165 if (is.null(daily[[r]])) return(rep(0L, length(days)))
166 values <- daily[[r]][days + 1L]
167 values[is.na(values)] <- 0L
168 values
169 }
170 while (length(decided) < length(state$tasks)) {
171 eligible <- setdiff(task_names(), names(decided))
172 eligible <- eligible[vapply(eligible, function(t) all(predecessors(t) %in% names(decided)), logical(1))]
173 latest <- vapply(eligible, function(t) schedule[[t]]$LS, numeric(1))
174 name <- eligible[[which.min(latest)]]
175 task <- state$tasks[[name]]
176 d <- duration(task)
177 start <- task$day
178 for (a in predecessors(name)) {
179 dep <- dependency(a, name)
180 value <- switch(dep$kind, FS = decided[[a]]$F + dep$lag,
181 SS = decided[[a]]$S + dep$lag, FF = decided[[a]]$F + dep$lag - d)
182 start <- max(start, value)
183 }
184 repeat {
185 days <- seq.int(start, start + d - 1L)
186 if (all(use(task$resource, days) + task$units <= capacity(task$resource))) break
187 start <- start + 1L
188 }
189 decided[[name]] <- list(S = start, F = start + d)
190 values <- daily[[task$resource]]
191 if (length(values) < start + d) values <- c(values, rep(0L, start + d - length(values)))
192 values[days + 1L] <- values[days + 1L] + task$units
193 daily[[task$resource]] <- values
194 order <- c(order, name)
195 }
196 finish <- max(vapply(decided, function(v) v$F, numeric(1)))
197 say("level capacity ", limit, ": duration ", total, " -> ", finish)
198 for (name in order) say(name, " start ", decided[[name]]$S, " finish ", decided[[name]]$F,
199 " shift ", decided[[name]]$S - schedule[[name]]$ES)
200 resources <- ordered(vapply(state$tasks, function(t) t$resource, character(1)))
201 for (r in resources) say("resource ", r, " capacity ", capacity(r), " peak ", max(usage(schedule, r, total)), " -> ", max(c(0L, daily[[r]])))
202}
203crash <- function(budget, initial) {
204 remaining <- budget
205 steps <- 0L
206 final <- initial
207 repeat {
208 current <- compute()
209 candidates <- critical(current$schedule)
210 candidates <- candidates[vapply(candidates, function(name) {
211 task <- state$tasks[[name]]
212 !is.null(task$crash_limit) && task$used < task$crash_limit && compute(setNames(list(duration(task) - 1L), name))$total < current$total
213 }, logical(1))]
214 if (!length(candidates)) {
215 reason <- "no candidate"
216 break
217 }
218 costs <- vapply(candidates, function(name) state$tasks[[name]]$cost, integer(1))
219 name <- candidates[[which.min(costs)]]
220 task <- state$tasks[[name]]
221 if (task$cost > remaining) {
222 reason <- "insufficient budget"
223 break
224 }
225 before <- duration(task)
226 state$tasks[[name]]$used <- task$used + 1L
227 remaining <- remaining - task$cost
228 steps <- steps + 1L
229 final <- compute()$total
230 say("step ", steps, ": ", name, " ", before, " -> ", before - 1L, ", cost ", task$cost, ", project ", final)
231 }
232 say("crash done: ", steps, " days, cost ", budget - remaining, ", project ", initial, " -> ", final,
233 ", budget left ", remaining, ", ", reason)
234}
235analyze <- function(command, args) {
236 if (!length(state$tasks)) fail("no tasks")
237 cycle <- topology()$cycle
238 if (!is.null(cycle)) fail(paste("cycle", cycle))
239 if (command == "CALENDAR" && is.null(state$start)) fail("no start date")
240 if (command == "LOAD" && !any(vapply(state$tasks, function(t) t$resource == args[[1]], logical(1)))) fail(paste("unknown resource", args[[1]]))
241 if (command == "WHATIF") require_task(args[[1]])
242 if (command == "VARIANCE" && is.null(state$baseline)) fail("no baseline")
243 result <- compute()
244 schedule <- result$schedule
245 total <- result$total
246 names <- task_names()
247 switch(command,
248 CPM = {
249 width <- max(4L, nchar(names))
250 say(sprintf(paste0("%-", width, "s"), "TASK"), paste(sprintf("%5s", c("DUR", "ES", "EF", "LS", "LF", "TF", "FF")), collapse = ""))
251 for (name in names) {
252 values <- vapply(c("DUR", "ES", "EF", "LS", "LF", "TF", "FF"), function(key) schedule[[name]][[key]], numeric(1))
253 say(sprintf(paste0("%-", width, "s"), name), paste(sprintf("%5d", as.integer(values)), collapse = ""), if (schedule[[name]]$TF == 0L) " *" else "")
254 }
255 say("project duration ", total)
256 },
257 CRITICAL = display_paths(command, schedule),
258 PERTSUM = display_paths(command, schedule),
259 SLACK = {
260 floats <- vapply(names, function(name) schedule[[name]]$TF, numeric(1))
261 for (value in sort(unique(floats))) say("float ", value, ": ", paste(names[floats == value], collapse = " "))
262 },
263 DRAG = {
264 for (name in critical(schedule)) say(name, " drag ", total - compute(setNames(list(0L), name))$total)
265 },
266 STATS = {
267 count <- length(critical(schedule))
268 kinds <- vapply(c("FS", "SS", "FF"), function(kind) sum(vapply(state$deps, function(d) d$kind == kind, logical(1))), numeric(1))
269 say("tasks ", length(names), ", critical ", count, ", dependencies ", length(state$deps),
270 " (FS ", kinds[[1]], ", SS ", kinds[[2]], ", FF ", kinds[[3]], ")")
271 resources <- ordered(vapply(state$tasks, function(t) t$resource, character(1)))
272 work <- vapply(resources, function(r) {
273 amount <- sum(vapply(state$tasks, function(t) if (t$resource == r) duration(t) * t$units else 0L, numeric(1)))
274 paste(r, amount)
275 }, character(1))
276 say("work ", paste(work, collapse = ", "))
277 sum_float <- sum(vapply(schedule, function(v) v$TF, numeric(1)))
278 say("float total ", sum_float, ", average ", decimal(sum_float, length(names)))
279 say("duration ", total, ", critical share ", decimal(100L * count, length(names)), "%")
280 },
281 CALENDAR = calendar(schedule, total),
282 GANTT = {
283 if (total > 60L) fail(paste0("chart too wide (", total, " days)"))
284 width <- max(nchar(names))
285 say(strrep(" ", width), " |", paste((seq_len(total) - 1L) %% 10L, collapse = ""), "|")
286 starts <- vapply(names, function(name) schedule[[name]]$ES, numeric(1))
287 for (name in names[order(starts, names, method = "radix")]) {
288 value <- schedule[[name]]
289 bar <- vapply(seq_len(total) - 1L, function(day) {
290 if (day >= value$ES && day < value$EF) {
291 if (value$TF == 0L) "=" else "-"
292 } else if (day >= value$EF && day < value$LF) "." else " "
293 }, character(1))
294 say(sprintf(paste0("%-", width, "s"), name), " |", paste(bar, collapse = ""), "|")
295 }
296 },
297 LEVEL = level(as.integer(args[[1]]), schedule, total),
298 LOAD = {
299 resource <- args[[1]]
300 values <- usage(schedule, resource, total)
301 say("load ", resource, ": ", paste(values, collapse = " "))
302 say("peak ", max(values), " on day ", which.max(values) - 1L)
303 if (!is.null(state$capacities[[resource]])) say("capacity ", state$capacities[[resource]], " exceeded on ", sum(values > state$capacities[[resource]]), " days")
304 },
305 CRASH = crash(as.integer(args[[1]]), total),
306 WHATIF = {
307 name <- args[[1]]
308 after <- compute(setNames(list(as.integer(args[[2]])), name))
309 say("whatif ", name, " ", duration(state$tasks[[name]]), " -> ", args[[2]], ": project ", total, " -> ", after$total)
310 for (t in names) {
311 fields <- c("ES", "EF", "TF")
312 fields <- fields[vapply(fields, function(key) schedule[[t]][[key]] != after$schedule[[t]][[key]], logical(1))]
313 if (length(fields)) say(t, paste(vapply(fields, function(key) paste0(" ", key, " ", schedule[[t]][[key]], "->", after$schedule[[t]][[key]]), character(1)), collapse = ""))
314 }
315 before_critical <- critical(schedule)
316 after_critical <- critical(after$schedule)
317 if (!identical(before_critical, after_critical)) say("critical ", paste(before_critical, collapse = " "), " -> ", paste(after_critical, collapse = " "))
318 },
319 BASELINE = {
320 saved <- lapply(names, function(name) c(schedule[[name]]$ES, schedule[[name]]$EF))
321 names(saved) <- names
322 state$baseline <- list(total = total, tasks = saved)
323 say("baseline saved: ", length(names), " tasks, duration ", total)
324 },
325 VARIANCE = {
326 old <- state$baseline
327 say("variance: duration ", old$total, " -> ", total, " (", signed(total - old$total), ")")
328 printed <- FALSE
329 for (name in ordered(c(names, names(old$tasks)))) {
330 if (is.null(state$tasks[[name]])) {
331 say(name, " removed")
332 } else if (is.null(old$tasks[[name]])) {
333 say(name, " new")
334 } else if (any(c(schedule[[name]]$ES, schedule[[name]]$EF) != old$tasks[[name]])) {
335 say(name, " start ", signed(schedule[[name]]$ES - old$tasks[[name]][[1]]), " finish ", signed(schedule[[name]]$EF - old$tasks[[name]][[2]]))
336 } else next
337 printed <- TRUE
338 }
339 if (!printed) say("no task changes")
340 }
341 )

343display_paths <- function(command, schedule) {
344 paths <- critical_paths(schedule)
345 if (command == "CRITICAL") say("critical paths: ", length(paths))
346 for (path in paths) {
347 notation <- paste(path, collapse = " -> ")
348 if (command == "CRITICAL") {
349 say(notation)
350 } else {
351 estimates <- lapply(path, function(name) {
352 task <- state$tasks[[name]]
353 if (is.null(task$estimate)) rep(duration(task), 3L) else task$estimate
354 })
355 expected <- sum(vapply(estimates, function(e) e[[1]] + 4L * e[[2]] + e[[3]], numeric(1)))
356 variance <- sum(vapply(estimates, function(e) (e[[3]] - e[[1]])^2, numeric(1)))
357 say(notation, ": expected ", decimal(expected, 6L), " variance ", decimal(variance, 36L))
358 }
359 }

361execute <- function(line) {
362 words <- strsplit(line, " ", fixed = TRUE)[[1]]
363 command <- words[[1]]
364 args <- words[-1]
365 if (command %in% analysis_commands) {
366 analyze(command, args)
367 return(invisible(NULL))
368 }
369 name <- if (length(args)) args[[1]] else NULL
370 if (command == "TASK") {
371 if (!is.null(state$tasks[[name]])) fail(paste("task", name, "already exists"))
372 state$tasks[[name]] <- list(base = as.integer(args[[2]]), resource = args[[3]], units = as.integer(args[[4]]),
373 day = 0L, crash_limit = NULL, cost = NULL, used = 0L, estimate = NULL)
374 say("task ", name, " added")
375 } else if (command %in% c("DEP", "UNDEP")) {
376 a <- name
377 b <- args[[2]]
378 require_task(a)
379 require_task(b)
380 key <- paste0(a, ":", b)
381 if (command == "DEP") {
382 if (a == b) fail(paste("self dependency", a))
383 if (!is.null(state$deps[[key]])) fail(paste("dependency", a, "->", b, "already exists"))
384 state$deps[[key]] <- list(a = a, b = b, kind = args[[3]], lag = as.integer(args[[4]]))
385 say("dependency ", a, " -> ", b, " ", args[[3]], signed(as.integer(args[[4]])), " added")
386 } else {
387 if (is.null(state$deps[[key]])) fail(paste("no dependency", a, "->", b))
388 state$deps[[key]] <- NULL
389 say("dependency ", a, " -> ", b, " removed")
390 }
391 } else if (command == "CAPACITY") {
392 state$capacities[[name]] <- as.integer(args[[2]])
393 say("capacity ", name, " ", args[[2]])
394 } else if (command %in% c("START", "HOLIDAY", "DEADLINE")) {
395 date <- date_value(name)
396 if (command == "START") state$start <- date else if (command == "DEADLINE") state$deadline <- date else {
397 if (date %in% state$holidays) fail(paste("duplicate holiday", name))
398 state$holidays <- c(state$holidays, date)
399 }
400 say(tolower(command), " ", date_text(date))
401 } else if (command == "LIST") {
402 say("tasks ", length(state$tasks), ", dependencies ", length(state$deps), ", holidays ", length(state$holidays))
403 for (name in task_names()) {
404 task <- state$tasks[[name]]
405 preds <- predecessors(name)
406 say(name, " duration ", duration(task), " resource ", task$resource, " units ", task$units,
407 " preds ", if (length(preds)) paste(preds, collapse = " ") else "none")
408 }
409 } else {
410 require_task(name)
411 task <- state$tasks[[name]]
412 switch(command,
413 REMOVE = {
414 remove <- vapply(state$deps, function(d) d$a == name || d$b == name, logical(1))
415 state$deps <- state$deps[!remove]
416 state$tasks[[name]] <- NULL
417 say("task ", name, " removed with ", sum(remove), " dependencies")
418 },
419 NOTBEFORE = {
420 state$tasks[[name]]$day <- as.integer(args[[2]])
421 say("notbefore ", name, " ", args[[2]])
422 },
423 CRASHCOST = {
424 maximum <- as.integer(args[[2]])
425 cost <- as.integer(args[[3]])
426 if (maximum >= task$base) fail(paste("crash limit too large for", name))
427 if (maximum < task$used) fail(paste("crash limit below used days for", name))
428 state$tasks[[name]]$crash_limit <- maximum
429 state$tasks[[name]]$cost <- cost
430 say("crash ", name, " max ", maximum, " cost ", cost)
431 },
432 UNCRASH = {
433 say("uncrash ", name, " restored ", task$used, " days")
434 state$tasks[[name]]$used <- 0L
435 },
436 PERT = {
437 estimate <- as.integer(args[-1])
438 if (any(diff(estimate) < 0L)) fail("invalid estimate")
439 state$tasks[[name]]$estimate <- estimate
440 say("pert ", name, " expected ", decimal(estimate[[1]] + 4L * estimate[[2]] + estimate[[3]], 6L),
441 " variance ", decimal((estimate[[3]] - estimate[[1]])^2, 36L))
442 },
443 SHOW = {
444 say("task ", name, " duration ", duration(task), " resource ", task$resource, " units ", task$units,
445 if (task$day != 0L) paste0(" notbefore ", task$day) else "")
446 preds <- vapply(predecessors(name), function(a) {
447 d <- dependency(a, name)
448 paste0(a, " ", d$kind, signed(d$lag))
449 }, character(1))
450 succs <- vapply(successors(name), function(b) {
451 d <- dependency(name, b)
452 paste0(b, " ", d$kind, signed(d$lag))
453 }, character(1))
454 say("preds: ", if (length(preds)) paste(preds, collapse = ", ") else "none")
455 say("succs: ", if (length(succs)) paste(succs, collapse = ", ") else "none")
456 say(if (is.null(task$crash_limit)) "crash: none" else paste("crash: max", task$crash_limit, "used", task$used, "cost", task$cost))
457 say(if (is.null(task$estimate)) "pert: none" else paste("pert:", paste(task$estimate, collapse = " ")))
458 if (!is.null(topology()$cycle)) {
459 say("schedule: unavailable")
460 } else {
461 value <- compute()$schedule[[name]]
462 fields <- vapply(c("ES", "EF", "LS", "LF", "TF", "FF"), function(key) paste(key, value[[key]]), character(1))
463 say("schedule: ", paste(fields, collapse = " "), if (value$TF == 0L) " critical" else "")
464 }
465 },
466 IMPACT = {
467 seen <- name
468 at <- 1L
469 while (at <= length(seen)) {
470 seen <- union(seen, successors(seen[[at]]))
471 at <- at + 1L
472 }
473 affected <- ordered(setdiff(seen, name))
474 say("impact ", name, ": ", if (length(affected)) paste(affected, collapse = " ") else "none")
475 }
476 )
477 }
