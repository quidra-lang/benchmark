68 private function topology(): array
69 {
70 $remaining = $this->names();
71 $order = [];
72 while (true) {
73 $found = null;
74 foreach ($remaining as $name) {
75 if (!array_intersect($this->predecessors($name), $remaining)) {
76 $found = $name;
77 break;
78 }
79 }
80 if ($found === null) {
81 break;
82 }
83 $remaining = array_values(array_diff($remaining, [$found]));
84 $order[] = $found;
85 }
86 if (!$remaining) {
87 return [$order, null];
88 }
89 $walk = [$remaining[0]];
90 while (true) {
91 $preds = array_values(array_intersect($this->predecessors($walk[count($walk) - 1]), $remaining));
92 $following = $preds[0];
93 $i = array_search($following, $walk, true);
94 if ($i !== false) {
95 $cycle = [$walk[$i], ...array_reverse(array_slice($walk, $i + 1))];
96 $sorted = $cycle;
97 sort($sorted, SORT_STRING);
98 $first = array_search($sorted[0], $cycle, true);
99 $cycle = [...array_slice($cycle, $first), ...array_slice($cycle, 0, $first)];
100 return [$order, implode(' -> ', [...$cycle, $cycle[0]])];
101 }
102 $walk[] = $following;
103 }
104 }

106 private function compute(array $overrides = []): array
107 {
108 [$order] = $this->topology();
109 $result = [];
110 foreach ($order as $name) {
111 $task = $this->tasks[$name];
112 $duration = $overrides[$name] ?? $task->duration();
113 $start = $task->day;
114 foreach ($this->predecessors($name) as $a) {
115 [, , $kind, $lag] = $this->deps["$a:$name"];
116 $value = match ($kind) {
117 'FS' => $result[$a]['EF'] + $lag,
118 'SS' => $result[$a]['ES'] + $lag,
119 default => $result[$a]['EF'] + $lag - $duration
120 };
121 $start = max($start, $value);
122 }
123 $result[$name] = ['DUR' => $duration, 'ES' => $start, 'EF' => $start + $duration];
124 }
125 $total = max(array_column($result, 'EF'));
126 foreach (array_reverse($order) as $name) {
127 $finish = $total;
128 foreach ($this->successors($name) as $b) {
129 [, , $kind, $lag] = $this->deps["$name:$b"];
130 $value = match ($kind) {
131 'FS' => $result[$b]['LS'] - $lag,
132 'SS' => $result[$b]['LS'] - $lag + $result[$name]['DUR'],
133 default => $result[$b]['LF'] - $lag
134 };
135 $finish = min($finish, $value);
136 }
137 $result[$name]['LF'] = $finish;
138 $result[$name]['LS'] = $finish - $result[$name]['DUR'];
139 $result[$name]['TF'] = $result[$name]['LS'] - $result[$name]['ES'];
140 }
141 foreach ($this->names() as $name) {
142 $float = $total - $result[$name]['EF'];
143 foreach ($this->successors($name) as $b) {
144 $float = min($float, $this->margin($name, $b, $result));
145 }
146 $result[$name]['FF'] = $float;
147 }
148 return [$result, $total];
149 }

151 private function margin(string $a, string $b, array $schedule): int
152 {
153 [, , $kind, $lag] = $this->deps["$a:$b"];
154 return match ($kind) {
155 'FS' => $schedule[$b]['ES'] - $schedule[$a]['EF'] - $lag,
156 'SS' => $schedule[$b]['ES'] - $schedule[$a]['ES'] - $lag,
157 default => $schedule[$b]['EF'] - $schedule[$a]['EF'] - $lag
158 };
159 }

161 private function critical(array $schedule): array
162 {
163 return array_values(array_filter($this->names(), fn($name) => $schedule[$name]['TF'] === 0));
164 }

166 private function criticalPaths(array $schedule): array
167 {
168 $selected = $this->critical($schedule);
169 $edges = [];
170 $incoming = [];
171 foreach ($this->deps as [$a, $b]) {
172 if (in_array($a, $selected, true) && in_array($b, $selected, true) && $this->margin($a, $b, $schedule) === 0) {
173 $edges[$a][] = $b;
174 $incoming[$b] = true;
175 }
176 }
177 foreach ($edges as &$children) {
178 sort($children, SORT_STRING);
179 }
180 unset($children);
181 $paths = [];
182 $visit = function (array $path) use (&$visit, &$paths, $edges): void {
183 $last = $path[count($path) - 1];
184 if (!isset($edges[$last])) {
185 $paths[] = $path;
186 } else {
187 foreach ($edges[$last] as $name) {
188 $visit([...$path, $name]);
189 }
190 }
191 };
192 foreach ($selected as $name) {
193 if (!isset($incoming[$name])) {
194 $visit([$name]);
195 }
196 }
197 return $paths;
198 }

230 private function calendar(array $schedule, int $total): void
231 {
232 $dates = [];
233 $current = $this->start;
234 while (count($dates) < $total) {
235 if ($this->working($current)) {
236 $dates[] = $current;
237 }
238 $current = $current->modify('+1 day');
239 }
240 $names = $this->names();
241 usort($names, fn($a, $b) => ($schedule[$a]['ES'] <=> $schedule[$b]['ES']) ?: strcmp($a, $b));
242 foreach ($names as $name) {
243 echo "$name ", $this->dateText($dates[$schedule[$name]['ES']]), ' - ',
244 $this->dateText($dates[$schedule[$name]['EF'] - 1]), "\n";
245 }
246 $first = $dates[0];
247 $finish = $dates[count($dates) - 1];
248 echo 'project ', $this->dateText($first), ' - ', $this->dateText($finish),
249 ", $total working days, ", (int)$first->diff($finish)->format('%a') + 1, " calendar days\n";
250 if ($this->deadline !== null) {
251 $days = 0;
252 $lower = min($finish, $this->deadline);
253 $upper = max($finish, $this->deadline);
254 for ($current = $lower->modify('+1 day'); $current <= $upper; $current = $current->modify('+1 day')) {
255 if ($this->working($current)) {
256 $days++;
257 }
258 }
259 echo 'deadline ', $this->dateText($this->deadline), $finish <= $this->deadline ?
260 " met, $days working days to spare\n" : " missed by $days working days\n";
261 }
262 $holidays = $this->holidays;
263 ksort($holidays, SORT_STRING);
264 $selected = [];
265 foreach ($holidays as $date) {
266 if ($date >= $first && $date <= $finish && (int)$date->format('N') <= 5) {
267 $selected[] = $this->dateText($date);
268 }
269 }
270 echo 'holidays in span: ', $selected ? implode(', ', $selected) : 'none', "\n";
271 }

273 private function usage(array $schedule, string $resource): array
274 {
275 $daily = [];
276 foreach ($this->tasks as $name => $task) {
277 if ($task->resource !== $resource) {
278 continue;
279 }
280 for ($day = $schedule[$name]['ES']; $day < $schedule[$name]['EF']; $day++) {
281 $daily[$day] = ($daily[$day] ?? 0) + $task->units;
282 }
283 }
284 return $daily;
285 }

287 private function level(int $limit, array $schedule, int $total): void
288 {
289 $capacity = fn($r) => $this->capacities[$r] ?? $limit;
290 foreach ($this->names() as $name) {
291 $task = $this->tasks[$name];
292 if ($task->units > $capacity($task->resource)) {
293 throw new ScheduleError('capacity ' . $capacity($task->resource) . " too small for $name");
294 }
295 }
296 $decided = [];
297 $order = [];
298 $daily = [];
299 while (count($decided) < count($this->tasks)) {
300 $eligible = [];
301 foreach ($this->names() as $name) {
302 if (!isset($decided[$name]) && !array_diff($this->predecessors($name), array_keys($decided))) {
303 $eligible[] = $name;
304 }
305 }
306 usort($eligible, fn($a, $b) => ($schedule[$a]['LS'] <=> $schedule[$b]['LS']) ?: strcmp($a, $b));
307 $name = $eligible[0];
308 $task = $this->tasks[$name];
309 $start = $task->day;
310 foreach ($this->predecessors($name) as $a) {
311 [, , $kind, $lag] = $this->deps["$a:$name"];
312 $value = match ($kind) {
313 'FS' => $decided[$a]['F'] + $lag,
314 'SS' => $decided[$a]['S'] + $lag,
315 default => $decided[$a]['F'] + $lag - $task->duration()
316 };
317 $start = max($start, $value);
318 }
319 while (true) {
320 $valid = true;
321 for ($day = $start; $day < $start + $task->duration(); $day++) {
322 if (($daily[$task->resource][$day] ?? 0) + $task->units > $capacity($task->resource)) {
323 $valid = false;
324 break;
325 }
326 }
327 if ($valid) {
328 break;
329 }
330 $start++;
331 }
332 $finish = $start + $task->duration();
333 $decided[$name] = ['S' => $start, 'F' => $finish];
334 for ($day = $start; $day < $finish; $day++) {
335 $daily[$task->resource][$day] = ($daily[$task->resource][$day] ?? 0) + $task->units;
336 }
337 $order[] = $name;
338 }
339 echo "level capacity $limit: duration $total -> ", max(array_column($decided, 'F')), "\n";
340 foreach ($order as $name) {
341 echo "$name start {$decided[$name]['S']} finish {$decided[$name]['F']} shift ",
342 $decided[$name]['S'] - $schedule[$name]['ES'], "\n";
343 }
344 $resources = array_unique(array_map(fn($t) => $t->resource, $this->tasks));
345 sort($resources, SORT_STRING);
346 foreach ($resources as $r) {
347 echo "resource $r capacity ", $capacity($r), ' peak ', max([0, ...$this->usage($schedule, $r)]),
348 ' -> ', max([0, ...($daily[$r] ?? [])]), "\n";
349 }

352 private function crash(int $budget, int $initial): void
353 {
354 $remaining = $budget;
355 $steps = 0;
356 $final = $initial;
357 while (true) {
358 [$schedule, $total] = $this->compute();
359 $candidates = [];
360 foreach ($this->critical($schedule) as $name) {
361 $task = $this->tasks[$name];
362 if ($task->crashLimit !== null && $task->used < $task->crashLimit &&
363 $this->compute([$name => $task->duration() - 1])[1] < $total) {
364 $candidates[] = $name;
365 }
366 }
367 if (!$candidates) {
368 $reason = 'no candidate';
369 break;
370 }
371 usort($candidates, fn($a, $b) => ($this->tasks[$a]->cost <=> $this->tasks[$b]->cost) ?: strcmp($a, $b));
372 $name = $candidates[0];
373 $task = $this->tasks[$name];
374 if ($task->cost > $remaining) {
375 $reason = 'insufficient budget';
376 break;
377 }
378 $before = $task->duration();
379 $task->used++;
380 $remaining -= $task->cost;
381 $steps++;
382 $final = $this->compute()[1];
383 echo "step $steps: $name $before -> ", $task->duration(), ", cost {$task->cost}, project $final\n";
384 }
385 echo "crash done: $steps days, cost ", $budget - $remaining,
386 ", project $initial -> $final, budget left $remaining, $reason\n";
387 }

389 private function analyze(string $command, array $args): void
390 {
391 if (!$this->tasks) {
392 throw new ScheduleError('no tasks');
393 }
394 [, $cycle] = $this->topology();
395 if ($cycle !== null) {
396 throw new ScheduleError("cycle $cycle");
397 }
398 if ($command === 'CALENDAR' && $this->start === null) {
399 throw new ScheduleError('no start date');
400 }
401 if ($command === 'LOAD' && !array_filter($this->tasks, fn($t) => $t->resource === $args[0])) {
402 throw new ScheduleError("unknown resource {$args[0]}");
403 }
404 if ($command === 'WHATIF') {
405 $this->requireTask($args[0]);
406 }
407 if ($command === 'VARIANCE' && $this->baseline === null) {
408 throw new ScheduleError('no baseline');
409 }
410 [$schedule, $total] = $this->compute();
411 $names = $this->names();
412 switch ($command) {
413 case 'CPM':
414 $width = max(4, max(array_map('strlen', $names)));
415 echo str_pad('TASK', $width), implode('', array_map(fn($s) => str_pad($s, 5, ' ', STR_PAD_LEFT), ['DUR', 'ES', 'EF', 'LS', 'LF', 'TF', 'FF'])), "\n";
416 foreach ($names as $name) {
417 echo str_pad($name, $width);
418 foreach (['DUR', 'ES', 'EF', 'LS', 'LF', 'TF', 'FF'] as $key) {
419 echo str_pad((string)$schedule[$name][$key], 5, ' ', STR_PAD_LEFT);
420 }
421 echo $schedule[$name]['TF'] === 0 ? " *\n" : "\n";
422 }
423 echo "project duration $total\n";
424 break;
425 case 'CRITICAL':
426 case 'PERTSUM':
427 $paths = $this->criticalPaths($schedule);
428 if ($command === 'CRITICAL') {
429 echo 'critical paths: ', count($paths), "\n";
430 }
431 foreach ($paths as $path) {
432 $notation = implode(' -> ', $path);
433 if ($command === 'CRITICAL') {
434 echo "$notation\n";
435 } else {
436 $expected = 0;
437 $variance = 0;
438 foreach ($path as $name) {
439 $task = $this->tasks[$name];
440 [$o, $m, $p] = $task->estimate ?? array_fill(0, 3, $task->duration());
441 $expected += $o + 4 * $m + $p;
442 $variance += ($p - $o) ** 2;
443 }
444 echo "$notation: expected ", $this->decimal($expected, 6), ' variance ', $this->decimal($variance, 36), "\n";
445 }
446 }
447 break;
448 case 'SLACK':
449 $groups = [];
450 foreach ($names as $name) {
451 $groups[$schedule[$name]['TF']][] = $name;
452 }
453 ksort($groups, SORT_NUMERIC);
454 foreach ($groups as $float => $tasks) {
455 echo "float $float: ", implode(' ', $tasks), "\n";
456 }
457 break;
458 case 'DRAG':
459 foreach ($this->critical($schedule) as $name) {
460 echo "$name drag ", $total - $this->compute([$name => 0])[1], "\n";
461 }
462 break;
463 case 'STATS':
464 $count = count($this->critical($schedule));
465 $kinds = ['FS' => 0, 'SS' => 0, 'FF' => 0];
466 foreach ($this->deps as [, , $kind]) {
467 $kinds[$kind]++;
468 }
469 echo 'tasks ', count($names), ", critical $count, dependencies ", count($this->deps),
470 " (FS {$kinds['FS']}, SS {$kinds['SS']}, FF {$kinds['FF']})\n";
471 $work = [];
472 foreach ($this->tasks as $task) {
473 $work[$task->resource] = ($work[$task->resource] ?? 0) + $task->duration() * $task->units;
474 }
475 ksort($work, SORT_STRING);
476 $parts = [];
477 foreach ($work as $r => $value) {
478 $parts[] = "$r $value";
479 }
480 echo 'work ', implode(', ', $parts), "\n";
481 $sum = array_sum(array_column($schedule, 'TF'));
482 echo "float total $sum, average ", $this->decimal($sum, count($names)), "\n";
483 echo "duration $total, critical share ", $this->decimal(100 * $count, count($names)), "%\n";
484 break;
485 case 'CALENDAR':
486 $this->calendar($schedule, $total);
487 break;
488 case 'GANTT':
489 if ($total > 60) {
490 throw new ScheduleError("chart too wide ($total days)");
491 }
492 $width = max(array_map('strlen', $names));
493 echo str_repeat(' ', $width), ' |';
494 for ($day = 0; $day < $total; $day++) {
495 echo $day % 10;
496 }
497 echo "|\n";
498 usort($names, fn($a, $b) => ($schedule[$a]['ES'] <=> $schedule[$b]['ES']) ?: strcmp($a, $b));
499 foreach ($names as $name) {
500 $value = $schedule[$name];
501 $bar = '';
502 for ($day = 0; $day < $total; $day++) {
503 if ($day >= $value['ES'] && $day < $value['EF']) {
504 $bar .= $value['TF'] === 0 ? '=' : '-';
505 } elseif ($day >= $value['EF'] && $day < $value['LF']) {
506 $bar .= '.';
507 } else {
508 $bar .= ' ';
509 }
510 }
511 echo str_pad($name, $width), " |$bar|\n";
512 }
513 break;
514 case 'LEVEL':
515 $this->level((int)$args[0], $schedule, $total);
516 break;
517 case 'LOAD':
518 $resource = $args[0];
519 $daily = $this->usage($schedule, $resource);
520 $values = [];
521 for ($day = 0; $day < $total; $day++) {
522 $values[] = $daily[$day] ?? 0;
523 }
524 echo "load $resource: ", implode(' ', $values), "\n";
525 $peak = max($values);
526 echo "peak $peak on day ", array_search($peak, $values, true), "\n";
527 if (isset($this->capacities[$resource])) {
528 echo "capacity {$this->capacities[$resource]} exceeded on ",
529 count(array_filter($values, fn($u) => $u > $this->capacities[$resource])), " days\n";
530 }
531 break;
532 case 'CRASH':
533 $this->crash((int)$args[0], $total);
534 break;
535 case 'WHATIF':
536 [$name, $duration] = $args;
537 [$after, $newTotal] = $this->compute([$name => (int)$duration]);
538 echo "whatif $name ", $this->tasks[$name]->duration(), " -> $duration: project $total -> $newTotal\n";
539 foreach ($names as $t) {
540 $parts = [];
541 foreach (['ES', 'EF', 'TF'] as $key) {
542 if ($schedule[$t][$key] !== $after[$t][$key]) {
543 $parts[] = "$key {$schedule[$t][$key]}->{$after[$t][$key]}";
544 }
545 }
546 if ($parts) {
547 echo "$t ", implode(' ', $parts), "\n";
548 }
549 }
550 if ($this->critical($schedule) !== $this->critical($after)) {
551 echo 'critical ', implode(' ', $this->critical($schedule)), ' -> ', implode(' ', $this->critical($after)), "\n";
552 }
553 break;
554 case 'BASELINE':
555 $saved = [];
556 foreach ($names as $name) {
557 $saved[$name] = [$schedule[$name]['ES'], $schedule[$name]['EF']];
558 }
559 $this->baseline = [$total, $saved];
560 echo 'baseline saved: ', count($names), " tasks, duration $total\n";
561 break;
562 case 'VARIANCE':
563 [$oldTotal, $old] = $this->baseline;
564 echo "variance: duration $oldTotal -> $total (", $this->signed($total - $oldTotal), ")\n";
565 $all = array_unique(array_merge($names, array_keys($old)));
566 sort($all, SORT_STRING);
567 $printed = false;
568 foreach ($all as $name) {
569 if (!isset($this->tasks[$name])) {
570 echo "$name removed\n";
571 } elseif (!isset($old[$name])) {
572 echo "$name new\n";
573 } elseif ([$schedule[$name]['ES'], $schedule[$name]['EF']] !== $old[$name]) {
574 echo "$name start ", $this->signed($schedule[$name]['ES'] - $old[$name][0]),
575 ' finish ', $this->signed($schedule[$name]['EF'] - $old[$name][1]), "\n";
576 } else {
577 continue;
578 }
579 $printed = true;
580 }
581 if (!$printed) {
582 echo "no task changes\n";
583 }
584 break;
585 }
586 }

588 public function execute(string $line): void
589 {
590 $args = explode(' ', $line);
591 $command = array_shift($args);
592 try {
593 if (in_array($command, self::ANALYSIS, true)) {
594 $this->analyze($command, $args);
595 return;
596 }
597 $name = $args[0] ?? null;
598 switch ($command) {
599 case 'TASK':
600 if (isset($this->tasks[$name])) {
601 throw new ScheduleError("task $name already exists");
602 }
603 $this->tasks[$name] = new Task((int)$args[1], $args[2], (int)$args[3]);
604 echo "task $name added\n";
605 return;
606 case 'DEP':
607 case 'UNDEP':
608 [$a, $b] = $args;
609 $this->requireTask($a);
610 $this->requireTask($b);
611 $key = "$a:$b";
612 if ($command === 'DEP') {
613 if ($a === $b) {
614 throw new ScheduleError("self dependency $a");
615 }
616 if (isset($this->deps[$key])) {
617 throw new ScheduleError("dependency $a -> $b already exists");
618 }
619 $this->deps[$key] = [$a, $b, $args[2], (int)$args[3]];
620 echo "dependency $a -> $b {$args[2]}", $this->signed((int)$args[3]), " added\n";
621 } else {
622 if (!isset($this->deps[$key])) {
623 throw new ScheduleError("no dependency $a -> $b");
624 }
625 unset($this->deps[$key]);
626 echo "dependency $a -> $b removed\n";
627 }
628 return;
629 case 'CAPACITY':
630 $this->capacities[$name] = (int)$args[1];
631 echo "capacity $name {$args[1]}\n";
632 return;
633 case 'START':
634 case 'HOLIDAY':
635 case 'DEADLINE':
636 $date = $this->dateValue($name);
637 if ($command === 'START') {
638 $this->start = $date;
639 } elseif ($command === 'DEADLINE') {
640 $this->deadline = $date;
641 } else {
642 if (isset($this->holidays[$name])) {
643 throw new ScheduleError("duplicate holiday $name");
644 }
645 $this->holidays[$name] = $date;
646 }
647 echo strtolower($command), ' ', $this->dateText($date), "\n";
648 return;
649 case 'LIST':
650 echo 'tasks ', count($this->tasks), ', dependencies ', count($this->deps), ', holidays ', count($this->holidays), "\n";
651 foreach ($this->names() as $t) {
652 $task = $this->tasks[$t];
653 $preds = $this->predecessors($t);
654 echo "$t duration ", $task->duration(), " resource {$task->resource} units {$task->units} preds ",
655 $preds ? implode(' ', $preds) : 'none', "\n";
656 }
657 return;
658 }
659 $this->requireTask($name);
660 $task = $this->tasks[$name];
661 switch ($command) {
662 case 'REMOVE':
663 $removed = 0;
664 foreach ($this->deps as $key => [$a, $b]) {
665 if ($a === $name || $b === $name) {
666 unset($this->deps[$key]);
667 $removed++;
668 }
669 }
670 unset($this->tasks[$name]);
671 echo "task $name removed with $removed dependencies\n";
672 break;
673 case 'NOTBEFORE':
674 $task->day = (int)$args[1];
675 echo "notbefore $name {$args[1]}\n";
676 break;
677 case 'CRASHCOST':
678 $maximum = (int)$args[1];
679 $cost = (int)$args[2];
680 if ($maximum >= $task->base) {
681 throw new ScheduleError("crash limit too large for $name");
682 }
683 if ($maximum < $task->used) {
684 throw new ScheduleError("crash limit below used days for $name");
685 }
686 $task->crashLimit = $maximum;
687 $task->cost = $cost;
688 echo "crash $name max $maximum cost $cost\n";
689 break;
690 case 'UNCRASH':
691 echo "uncrash $name restored {$task->used} days\n";
692 $task->used = 0;
693 break;
694 case 'PERT':
695 [$o, $m, $p] = array_map('intval', array_slice($args, 1));
696 if ($o > $m || $m > $p) {
697 throw new ScheduleError('invalid estimate');
698 }
699 $task->estimate = [$o, $m, $p];
700 echo "pert $name expected ", $this->decimal($o + 4 * $m + $p, 6),
701 ' variance ', $this->decimal(($p - $o) ** 2, 36), "\n";
702 break;
703 case 'SHOW':
704 echo "task $name duration ", $task->duration(), " resource {$task->resource} units {$task->units}",
705 $task->day === 0 ? '' : " notbefore {$task->day}", "\n";
706 $preds = [];
707 foreach ($this->predecessors($name) as $a) {
708 [, , $kind, $lag] = $this->deps["$a:$name"];
709 $preds[] = "$a $kind" . $this->signed($lag);
710 }
711 $succs = [];
712 foreach ($this->successors($name) as $b) {
713 [, , $kind, $lag] = $this->deps["$name:$b"];
714 $succs[] = "$b $kind" . $this->signed($lag);
715 }
716 echo 'preds: ', $preds ? implode(', ', $preds) : 'none', "\n";
717 echo 'succs: ', $succs ? implode(', ', $succs) : 'none', "\n";
718 echo $task->crashLimit !== null ? "crash: max {$task->crashLimit} used {$task->used} cost {$task->cost}" : 'crash: none', "\n";
719 echo $task->estimate !== null ? 'pert: ' . implode(' ', $task->estimate) : 'pert: none', "\n";
720 [, $cycle] = $this->topology();
721 if ($cycle !== null) {
722 echo "schedule: unavailable\n";
723 } else {
724 [$schedule] = $this->compute();
725 $parts = [];
726 foreach (['ES', 'EF', 'LS', 'LF', 'TF', 'FF'] as $key) {
727 $parts[] = "$key {$schedule[$name][$key]}";
728 }
729 echo 'schedule: ', implode(' ', $parts), $schedule[$name]['TF'] === 0 ? ' critical' : '', "\n";
730 }
731 break;
732 case 'IMPACT':
733 $seen = [$name];
734 for ($i = 0; $i < count($seen); $i++) {
735 foreach ($this->successors($seen[$i]) as $b) {
736 if (!in_array($b, $seen, true)) {
737 $seen[] = $b;
738 }
739 }
740 }
741 $affected = array_diff($seen, [$name]);
742 sort($affected, SORT_STRING);
743 echo "impact $name: ", $affected ? implode(' ', $affected) : 'none', "\n";
744 break;
745 }
746 } catch (ScheduleError $error) {
747 echo 'error: ', $error->getMessage(), "\n";
748 }
749 }
750}

753$count = (int)fgets(STDIN);
754for ($i = 0; $i < $count; $i++) {
755 $scheduler->execute(rtrim(fgets(STDIN), "\r\n"));
