8class WorkTask(string name, int duration, string resource, int units)
9{
10 public string Name = name, Resource = resource;
11 public int Base = duration, Units = units, NotBefore, Crashed;
12 public int? CrashLimit, CrashCost;
13 public int[]? Estimate;
14 public int Duration => Base - Crashed;

58 WorkTask Task(string name)
59 {
60 if (!tasks.TryGetValue(name, out WorkTask? t)) Fail("unknown task " + name);
61 return t!;

63 List<Dependency> Incoming(string name) => dependencies.Where(d => d.B == name).OrderBy(d => d.A, StringComparer.Ordinal).ToList();
64 List<Dependency> Outgoing(string name) => dependencies.Where(d => d.A == name).OrderBy(d => d.B, StringComparer.Ordinal).ToList();

66 List<string> Topological()
67 {
68 List<string> result = [];
69 SortedSet<string> remaining = new(tasks.Keys, StringComparer.Ordinal);
70 while (true)
71 {
72 string? next = remaining.FirstOrDefault(n => Incoming(n).All(d => !remaining.Contains(d.A)));
73 if (next == null) break;
74 result.Add(next);
75 remaining.Remove(next);
76 }
77 return result;
78 }
79
80 string? Cycle()
81 {
82 SortedSet<string> remaining = new(tasks.Keys, StringComparer.Ordinal);
83 remaining.ExceptWith(Topological());
84 if (remaining.Count == 0) return null;
85 List<string> chain = [];
86 string current = remaining.Min!;
87 while (!chain.Contains(current))
88 {
89 chain.Add(current);
90 current = Incoming(current).First(d => remaining.Contains(d.A)).A;
91 }
92 int begin = chain.IndexOf(current);
93 List<string> cycle = [current];
94 for (int i = chain.Count - 1; i > begin; i--) cycle.Add(chain[i]);
95 int minimum = cycle.IndexOf(cycle.Min(StringComparer.Ordinal)!);
96 cycle = [.. cycle.Skip(minimum), .. cycle.Take(minimum)];
97 cycle.Add(cycle[0]);
98 return string.Join(" -> ", cycle);
99 }

101 Schedule Compute(string? changed = null, int duration = 0)
102 {
103 Schedule s = new();
104 List<string> order = Topological();
105 foreach (string name in order)
106 {
107 WorkTask t = tasks[name];
108 Timing x = new() { Duration = name == changed ? duration : t.Duration, ES = t.NotBefore };
109 foreach (Dependency d in Incoming(name))
110 {
111 Timing a = s.Times[d.A];
112 int bound = d.Kind switch { "FS" => a.EF + d.Lag, "SS" => a.ES + d.Lag, _ => a.EF + d.Lag - x.Duration };
113 x.ES = Math.Max(x.ES, bound);
114 }
115 x.EF = x.ES + x.Duration;
116 s.Total = Math.Max(s.Total, x.EF);
117 s.Times[name] = x;
118 }
119 for (int i = order.Count - 1; i >= 0; i--)
120 {
121 string name = order[i];
122 Timing x = s.Times[name];
123 x.LF = s.Total;
124 foreach (Dependency d in Outgoing(name))
125 {
126 Timing b = s.Times[d.B];
127 int bound = d.Kind switch { "FS" => b.LS - d.Lag, "SS" => b.LS - d.Lag + x.Duration, _ => b.LF - d.Lag };
128 x.LF = Math.Min(x.LF, bound);
129 }
130 x.LS = x.LF - x.Duration;
131 x.TF = x.LS - x.ES;
132 x.Free = s.Total - x.EF;
133 foreach (Dependency d in Outgoing(name)) x.Free = Math.Min(x.Free, Margin(s, d));
134 }
135 return s;
136 }
137
138 int Margin(Schedule s, Dependency d)
139 {
140 Timing a = s.Times[d.A], b = s.Times[d.B];
141 return (d.Kind == "FF" ? b.EF : b.ES) - (d.Kind == "SS" ? a.ES : a.EF) - d.Lag;
142 }

144 List<List<string>> CriticalPaths(Schedule s)
145 {
146 SortedDictionary<string, List<string>> successors = new(StringComparer.Ordinal);
147 HashSet<string> hasPredecessor = new(StringComparer.Ordinal);
148 foreach (string name in s.Critical()) successors[name] = [];
149 foreach (Dependency d in dependencies) if (successors.ContainsKey(d.A) && successors.ContainsKey(d.B) && Margin(s, d) == 0)
150 {
151 successors[d.A].Add(d.B);
152 hasPredecessor.Add(d.B);
153 }
154 foreach (List<string> list in successors.Values) list.Sort(StringComparer.Ordinal);
155 List<List<string>> paths = [];
156 List<string> path = [];
157 void Visit(string name)
158 {
159 path.Add(name);
160 if (successors[name].Count == 0) paths.Add([.. path]);
161 else foreach (string next in successors[name]) Visit(next);
162 path.RemoveAt(path.Count - 1);
163 }
164 foreach (string name in successors.Keys) if (!hasPredecessor.Contains(name)) Visit(name);
165 return paths;
166 }

168 static DateOnly ParseDate(string text)
169 {
170 if (!DateOnly.TryParseExact(text, "yyyy-MM-dd", CultureInfo.InvariantCulture, DateTimeStyles.None, out DateOnly d)) Fail("invalid date " + text);
171 return d;
172 }
173
174 static string Date(DateOnly d) => d.ToString("yyyy-MM-dd ddd", CultureInfo.InvariantCulture);
175
176 static bool Weekday(DateOnly d) => d.DayOfWeek != DayOfWeek.Saturday && d.DayOfWeek != DayOfWeek.Sunday;
177 bool Working(DateOnly d) => Weekday(d) && !holidays.Contains(d);
178
179 List<DateOnly> Calendar(int total)
180 {
181 List<DateOnly> days = [];
182 for (DateOnly d = start!.Value; days.Count < total; d = d.AddDays(1)) if (Working(d)) days.Add(d);
183 return days;
184 }
185
186 int WorkingBetween(DateOnly a, DateOnly b)
187 {
188 int count = 0;
189 for (DateOnly d = a.AddDays(1); d <= b; d = d.AddDays(1)) if (Working(d)) count++;
190 return count;
191 }

193 SortedDictionary<string, int[]> Usage(Schedule s)
194 {
195 SortedDictionary<string, int[]> result = new(StringComparer.Ordinal);
196 foreach (WorkTask t in tasks.Values)
197 {
198 if (!result.TryGetValue(t.Resource, out int[]? load)) result[t.Resource] = load = new int[s.Total];
199 Timing x = s.Times[t.Name];
200 for (int k = x.ES; k < x.EF; k++) load[k] += t.Units;
201 }
202 return result;
203 }
204 List<string> StartOrder(Schedule s) => tasks.Keys.OrderBy(n => s.Times[n].ES).ThenBy(n => n, StringComparer.Ordinal).ToList();

206 List<string> Command(string line)
207 {
208 string[] w = line.Split(' ');
209 string c = w[0];
210 List<string> output = [];
211 string[] analyses = ["CPM", "CRITICAL", "SLACK", "DRAG", "STATS", "CALENDAR", "GANTT", "LEVEL", "LOAD", "PERTSUM", "CRASH", "WHATIF", "BASELINE", "VARIANCE"];
212 if (analyses.Contains(c))
213 {
214 if (tasks.Count == 0) Fail("no tasks");
215 string? cycle = Cycle();
216 if (cycle != null) Fail("cycle " + cycle);
217 }
218 switch (c)
219 {
220 case "TASK":
221 if (tasks.ContainsKey(w[1])) Fail($"task {w[1]} already exists");
222 tasks[w[1]] = new(w[1], int.Parse(w[2]), w[3], int.Parse(w[4]));
223 output.Add($"task {w[1]} added");
224 break;
225 case "DEP":
226 case "UNDEP":
227 {
228 Task(w[1]);
229 Task(w[2]);
230 Dependency? old = dependencies.FirstOrDefault(d => d.A == w[1] && d.B == w[2]);
231 if (c == "DEP")
232 {
233 if (w[1] == w[2]) Fail("self dependency " + w[1]);
234 if (old != null) Fail($"dependency {w[1]} -> {w[2]} already exists");
235 int lag = int.Parse(w[4]);
236 dependencies.Add(new(w[1], w[2], w[3], lag));
237 output.Add($"dependency {w[1]} -> {w[2]} {w[3]}{Signed(lag)} added");
238 }
239 else
240 {
241 if (old == null) Fail($"no dependency {w[1]} -> {w[2]}");
242 dependencies.Remove(old!);
243 output.Add($"dependency {w[1]} -> {w[2]} removed");
244 }
245 break;
246 }
247 case "REMOVE":
248 {
249 Task(w[1]);
250 int count = dependencies.RemoveAll(d => d.A == w[1] || d.B == w[1]);
251 tasks.Remove(w[1]);
252 output.Add($"task {w[1]} removed with {count} dependencies");
253 break;
254 }
255 case "NOTBEFORE":
256 case "CRASHCOST":
257 case "UNCRASH":
258 case "PERT":
259 {
260 WorkTask t = Task(w[1]);
261 if (c == "NOTBEFORE")
262 {
263 t.NotBefore = int.Parse(w[2]);
264 output.Add($"notbefore {t.Name} {t.NotBefore}");
265 }
266 else if (c == "CRASHCOST")
267 {
268 int limit = int.Parse(w[2]);
269 if (limit >= t.Base) Fail("crash limit too large for " + t.Name);
270 if (limit < t.Crashed) Fail("crash limit below used days for " + t.Name);
271 t.CrashLimit = limit;
272 t.CrashCost = int.Parse(w[3]);
273 output.Add($"crash {t.Name} max {limit} cost {t.CrashCost}");
274 }
275 else if (c == "UNCRASH")
276 {
277 output.Add($"uncrash {t.Name} restored {t.Crashed} days");
278 t.Crashed = 0;
279 }
280 else
281 {
282 int o = int.Parse(w[2]), m = int.Parse(w[3]), p = int.Parse(w[4]);
283 if (o > m || m > p) Fail("invalid estimate");
284 t.Estimate = [o, m, p];
285 output.Add($"pert {t.Name} expected {Fraction(o + 4L * m + p, 6)} variance {Fraction((long)(p - o) * (p - o), 36)}");
286 }
287 break;
288 }
289 case "CAPACITY":
290 capacities[w[1]] = int.Parse(w[2]);
291 output.Add($"capacity {w[1]} {w[2]}");
292 break;
293 case "START":
294 case "HOLIDAY":
295 case "DEADLINE":
296 {
297 DateOnly d = ParseDate(w[1]);
298 if (c == "HOLIDAY" && holidays.Contains(d)) Fail("duplicate holiday " + w[1]);
299 if (c == "START") start = d;
300 else if (c == "DEADLINE") deadline = d;
301 else holidays.Add(d);
302 output.Add(c.ToLowerInvariant() + " " + Date(d));
303 break;
304 }
305 case "LIST":
306 output.Add($"tasks {tasks.Count}, dependencies {dependencies.Count}, holidays {holidays.Count}");
307 foreach (WorkTask t in tasks.Values) output.Add($"{t.Name} duration {t.Duration} resource {t.Resource} units {t.Units} preds {Names(Incoming(t.Name).Select(d => d.A))}");
308 break;
309 case "SHOW":
310 {
311 WorkTask t = Task(w[1]);
312 output.Add($"task {t.Name} duration {t.Duration} resource {t.Resource} units {t.Units}" + (t.NotBefore == 0 ? "" : $" notbefore {t.NotBefore}"));
313 foreach (bool predecessors in new[] { true, false })
314 {
315 List<string> parts = (predecessors ? Incoming(t.Name) : Outgoing(t.Name)).Select(d => $"{(predecessors ? d.A : d.B)} {d.Kind}{Signed(d.Lag)}").ToList();
316 output.Add((predecessors ? "preds: " : "succs: ") + (parts.Count == 0 ? "none" : string.Join(", ", parts)));
317 }
318 output.Add(t.CrashLimit == null ? "crash: none" : $"crash: max {t.CrashLimit} used {t.Crashed} cost {t.CrashCost}");
319 output.Add(t.Estimate == null ? "pert: none" : "pert: " + string.Join(" ", t.Estimate));
320 if (Cycle() != null) output.Add("schedule: unavailable");
321 else
322 {
323 Timing x = Compute().Times[t.Name];
324 output.Add($"schedule: ES {x.ES} EF {x.EF} LS {x.LS} LF {x.LF} TF {x.TF} FF {x.Free}" + (x.TF == 0 ? " critical" : ""));
325 }
326 break;
327 }
328 case "IMPACT":
329 {
330 Task(w[1]);
331 SortedSet<string> seen = new(StringComparer.Ordinal) { w[1] };
332 Queue<string> queue = new();
333 queue.Enqueue(w[1]);
334 while (queue.Count != 0) foreach (Dependency d in Outgoing(queue.Dequeue())) if (seen.Add(d.B)) queue.Enqueue(d.B);
335 seen.Remove(w[1]);
336 output.Add($"impact {w[1]}: {Names(seen)}");
337 break;
338 }
339 default:
340 {
341 if (c == "CALENDAR" && start == null) Fail("no start date");
342 if (c == "WHATIF") Task(w[1]);
343 if (c == "VARIANCE" && baseline == null) Fail("no baseline");
344 if (c == "LOAD" && tasks.Values.All(t => t.Resource != w[1])) Fail("unknown resource " + w[1]);
345 if (c == "LEVEL") foreach (WorkTask t in tasks.Values)
346 {
347 int capacity = capacities.GetValueOrDefault(t.Resource, int.Parse(w[1]));
348 if (t.Units > capacity) Fail($"capacity {capacity} too small for {t.Name}");
349 }
350 Schedule s = Compute();
351 switch (c)
352 {
353 case "CPM":
354 {
355 int width = Math.Max(4, tasks.Keys.Max(n => n.Length));
356 output.Add("TASK".PadRight(width) + string.Concat(new[] { "DUR", "ES", "EF", "LS", "LF", "TF", "FF" }.Select(v => v.PadLeft(5))));
357 foreach ((string name, Timing x) in s.Times) output.Add(name.PadRight(width) + string.Concat(new[] { x.Duration, x.ES, x.EF, x.LS, x.LF, x.TF, x.Free }.Select(v => v.ToString().PadLeft(5))) + (x.TF == 0 ? " *" : ""));
358 output.Add("project duration " + s.Total);
359 break;
360 }
361 case "CRITICAL":
362 case "PERTSUM":
363 {
364 List<List<string>> paths = CriticalPaths(s);
365 if (c == "CRITICAL") output.Add("critical paths: " + paths.Count);
366 foreach (List<string> path in paths)
367 {
368 string text = string.Join(" -> ", path);
369 if (c == "PERTSUM")
370 {
371 long expected = 0, variance = 0;
372 foreach (string name in path)
373 {
374 WorkTask t = tasks[name];
375 int[] p = t.Estimate ?? [t.Duration, t.Duration, t.Duration];
376 expected += p[0] + 4L * p[1] + p[2];
377 variance += (long)(p[2] - p[0]) * (p[2] - p[0]);
378 }
379 text += $": expected {Fraction(expected, 6)} variance {Fraction(variance, 36)}";
380 }
381 output.Add(text);
382 }
383 break;
384 }
385 case "SLACK": foreach (var group in s.Times.GroupBy(e => e.Value.TF).OrderBy(g => g.Key)) output.Add($"float {group.Key}: {Names(group.Select(e => e.Key))}");
386 break;
387 case "DRAG":
388 foreach (string name in s.Critical()) output.Add($"{name} drag {s.Total - Compute(name, 0).Total}");
389 break;
390 case "STATS":
391 {
392 int critical = s.Critical().Count, totalFloat = s.Times.Values.Sum(x => x.TF);
393 output.Add($"tasks {tasks.Count}, critical {critical}, dependencies {dependencies.Count} (FS {dependencies.Count(d => d.Kind == "FS")}, SS {dependencies.Count(d => d.Kind == "SS")}, FF {dependencies.Count(d => d.Kind == "FF")})");
394 output.Add("work " + string.Join(", ", tasks.Values.GroupBy(t => t.Resource).OrderBy(g => g.Key, StringComparer.Ordinal).Select(g => $"{g.Key} {g.Sum(t => t.Duration * t.Units)}")));
395 output.Add($"float total {totalFloat}, average {Fraction(totalFloat, tasks.Count)}");
396 output.Add($"duration {s.Total}, critical share {Fraction(100L * critical, tasks.Count)}%");
397 break;
398 }
399 case "CALENDAR":
400 {
401 List<DateOnly> days = Calendar(s.Total);
402 foreach (string name in StartOrder(s))
403 {
404 Timing x = s.Times[name];
405 output.Add($"{name} {Date(days[x.ES])} - {Date(days[x.EF - 1])}");
406 }
407 DateOnly first = days[0], last = days[^1];
408 output.Add($"project {Date(first)} - {Date(last)}, {s.Total} working days, {last.DayNumber - first.DayNumber + 1} calendar days");
409 if (deadline.HasValue) output.Add($"deadline {Date(deadline.Value)}" + (last <= deadline.Value ? $" met, {WorkingBetween(last, deadline.Value)} working days to spare" : $" missed by {WorkingBetween(deadline.Value, last)} working days"));
410 List<string> span = holidays.Where(d => Weekday(d) && d >= first && d <= last).Select(Date).ToList();
411 output.Add("holidays in span: " + (span.Count == 0 ? "none" : string.Join(", ", span)));
412 break;
413 }
414 case "GANTT":
415 {
416 if (s.Total > 60) Fail($"chart too wide ({s.Total} days)");
417 int width = tasks.Keys.Max(n => n.Length);
418 output.Add(new string(' ', width) + " |" + string.Concat(Enumerable.Range(0, s.Total).Select(k => k % 10)) + "|");
419 foreach (string name in StartOrder(s))
420 {
421 Timing x = s.Times[name];
422 output.Add(name.PadRight(width) + " |" + new string(Enumerable.Range(0, s.Total).Select(k => x.ES <= k && k < x.EF ? x.TF == 0 ? '=' : '-' : x.EF <= k && k < x.LF ? '.' : ' ').ToArray()) + "|");
423 }
424 break;
425 }
426 case "LEVEL":
427 output.AddRange(Level(s, int.Parse(w[1])));
428 break;
429 case "LOAD":
430 {
431 int[] load = Usage(s)[w[1]];
432 int peak = load.Max();
433 output.Add($"load {w[1]}: " + string.Join(" ", load));
434 output.Add($"peak {peak} on day {Array.IndexOf(load, peak)}");
435 if (capacities.TryGetValue(w[1], out int capacity)) output.Add($"capacity {capacity} exceeded on {load.Count(x => x > capacity)} days");
436 break;
437 }
438 case "CRASH":
439 {
440 int remaining = int.Parse(w[1]), initial = s.Total, steps = 0;
441 string reason;
442 while (true)
443 {
444 WorkTask? candidate = null;
445 foreach (string name in s.Critical())
446 {
447 WorkTask t = tasks[name];
448 if (t.CrashLimit.HasValue && t.Crashed < t.CrashLimit.Value && Compute(name, t.Duration - 1).Total < s.Total && (candidate == null || t.CrashCost!.Value < candidate.CrashCost!.Value)) candidate = t;
449 }
450 if (candidate == null)
451 {
452 reason = "no candidate";
453 break;
454 }
455 if (candidate.CrashCost!.Value > remaining)
456 {
457 reason = "insufficient budget";
458 break;
459 }
460 int before = candidate.Duration;
461 candidate.Crashed++;
462 remaining -= candidate.CrashCost.Value;
463 s = Compute();
464 output.Add($"step {++steps}: {candidate.Name} {before} -> {candidate.Duration}, cost {candidate.CrashCost}, project {s.Total}");
465 }
466 output.Add($"crash done: {steps} days, cost {int.Parse(w[1]) - remaining}, project {initial} -> {s.Total}, budget left {remaining}, {reason}");
467 break;
468 }
469 case "WHATIF":
470 {
471 Schedule after = Compute(w[1], int.Parse(w[2]));
472 output.Add($"whatif {w[1]} {tasks[w[1]].Duration} -> {w[2]}: project {s.Total} -> {after.Total}");
473 foreach (string name in tasks.Keys)
474 {
475 Timing a = s.Times[name], b = after.Times[name];
476 string text = name + (a.ES != b.ES ? $" ES {a.ES}->{b.ES}" : "") + (a.EF != b.EF ? $" EF {a.EF}->{b.EF}" : "") + (a.TF != b.TF ? $" TF {a.TF}->{b.TF}" : "");
477 if (text != name) output.Add(text);
478 }
479 if (!s.Critical().SequenceEqual(after.Critical())) output.Add($"critical {Names(s.Critical())} -> {Names(after.Critical())}");
480 break;
481 }
482 case "BASELINE": baseline = new(s.Total, s.Times.ToDictionary(e => e.Key, e => (e.Value.ES, e.Value.EF), StringComparer.Ordinal));
483 output.Add($"baseline saved: {tasks.Count} tasks, duration {s.Total}");
484 break;
485 case "VARIANCE":
486 {
487 output.Add($"variance: duration {baseline!.Total} -> {s.Total} ({Signed(s.Total - baseline.Total)})");
488 foreach (string name in tasks.Keys.Union(baseline.Times.Keys).OrderBy(n => n, StringComparer.Ordinal))
489 {
490 if (!tasks.ContainsKey(name)) output.Add(name + " removed");
491 else if (!baseline.Times.TryGetValue(name, out var old)) output.Add(name + " new");
492 else
493 {
494 Timing x = s.Times[name];
495 if (x.ES != old.ES || x.EF != old.EF) output.Add($"{name} start {Signed(x.ES - old.ES)} finish {Signed(x.EF - old.EF)}");
496 }
497 }
498 if (output.Count == 1) output.Add("no task changes");
499 break;
500 }
501 }
502 break;
503 }
504 }
505 return output;
506 }

508 List<string> Level(Schedule s, int limit)
509 {
510 Dictionary<string, (int Start, int Finish)> decided = new(StringComparer.Ordinal);
511 SortedDictionary<string, Dictionary<int, int>> loads = new(StringComparer.Ordinal);
512 foreach (WorkTask t in tasks.Values) if (!loads.ContainsKey(t.Resource)) loads[t.Resource] = [];
513 List<string> order = [];
514 int total = 0;
515 while (decided.Count < tasks.Count)
516 {
517 string name = tasks.Keys.Where(n => !decided.ContainsKey(n) && Incoming(n).All(d => decided.ContainsKey(d.A))).OrderBy(n => s.Times[n].LS).ThenBy(n => n, StringComparer.Ordinal).First();
518 WorkTask t = tasks[name];
519 int begin = t.NotBefore, duration = t.Duration, capacity = capacities.GetValueOrDefault(t.Resource, limit);
520 foreach (Dependency d in Incoming(name))
521 {
522 var a = decided[d.A];
523 begin = Math.Max(begin, (d.Kind == "SS" ? a.Start : a.Finish) + d.Lag - (d.Kind == "FF" ? duration : 0));
524 }
525 Dictionary<int, int> load = loads[t.Resource];
526 while (Enumerable.Range(begin, duration).Any(k => load.GetValueOrDefault(k) + t.Units > capacity)) begin++;
527 int finish = begin + duration;
528 for (int k = begin; k < finish; k++) load[k] = load.GetValueOrDefault(k) + t.Units;
529 decided[name] = (begin, finish);
530 order.Add(name);
531 total = Math.Max(total, finish);
532 }
533 List<string> output = [$"level capacity {limit}: duration {s.Total} -> {total}"];
534 foreach (string name in order)
535 {
536 var t = decided[name];
537 output.Add($"{name} start {t.Start} finish {t.Finish} shift {t.Start - s.Times[name].ES}");
538 }
539 var original = Usage(s);
540 foreach (string resource in loads.Keys) output.Add($"resource {resource} capacity {capacities.GetValueOrDefault(resource, limit)} peak {original[resource].Max()} -> {loads[resource].Values.DefaultIfEmpty().Max()}");
541 return output;
542 }

544 static void Main()
545 {
546 CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
547 Program project = new();
548 int n = int.Parse(Console.ReadLine()!);
549 for (int i = 0; i < n; i++)
550 {
551 try
552 {
553 foreach (string line in project.Command(Console.ReadLine()!)) Console.WriteLine(line);
554 }
555 catch (Problem e)
556 {
