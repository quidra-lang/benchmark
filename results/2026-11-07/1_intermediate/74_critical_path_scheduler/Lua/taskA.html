27local function leap(year)
28 return year % 4 == 0 and year % 100 ~= 0 or year % 400 == 0
29end
30local function month_days(year, month)
31 local days = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31}
32 return month == 2 and leap(year) and 29 or days[month]
33end
34local function date_value(text)
35 local year, month, day = tonumber(text:sub(1, 4)), tonumber(text:sub(6, 7)), tonumber(text:sub(9, 10))
36 if month < 1 or month > 12 or day < 1 or day > month_days(year, month) then
37 return nil
38 end
39 local total = day - 1
40 for y = 2000, year - 1 do
41 total = total + (leap(y) and 366 or 365)
42 end
43 for m = 1, month - 1 do
44 total = total + month_days(year, m)
45 end
46 return total
47end
48local function date_text(ordinal)
49 local original, year, month = ordinal, 2000, 1
50 while ordinal >= (leap(year) and 366 or 365) do
51 ordinal = ordinal - (leap(year) and 366 or 365)
52 year = year + 1
53 end
54 while ordinal >= month_days(year, month) do
55 ordinal = ordinal - month_days(year, month)
56 month = month + 1
57 end
58 local weekdays = {"Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"}
59 return string.format("%04d-%02d-%02d %s", year, month, ordinal + 1, weekdays[(original + 5) % 7 + 1])
60end
61local function decimal(numerator, denominator)
62 local value = numerator * 100 // denominator
63 return string.format("%d.%02d", value // 100, value % 100)
64end

77 for _, dep in pairs(dependencies) do

83 return result

85local function topology()
86 local names, removed, order = keys(tasks), {}, {}
87 while true do
88 local selected
89 for _, name in ipairs(names) do
90 if not removed[name] then
91 local ready = true
92 for _, dep in ipairs(predecessors(name)) do
93 if not removed[dep.from] then
94 ready = false
95 break
96 end
97 end
98 if ready then
99 selected = name
100 break
101 end
102 end
103 end
104 if not selected then
105 break
106 end
107 removed[selected], order[#order + 1] = true, selected
108 end
109 if #order == #names then
110 return order, nil
111 end
112 local current
113 for _, name in ipairs(names) do
114 if not removed[name] then
115 current = name
116 break
117 end
118 end
119 local visited, chain = {}, {}
120 while true do
121 if visited[current] then
122 local at, cycle = visited[current], {current}
123 for i = #chain, at + 1, -1 do
124 cycle[#cycle + 1] = chain[i]
125 end
126 local first = 1
127 for i, name in ipairs(cycle) do
128 if name < cycle[first] then
129 first = i
130 end
131 end
132 local rotated = {}
133 for i = first, #cycle do
134 rotated[#rotated + 1] = cycle[i]
135 end
136 for i = 1, first - 1 do
137 rotated[#rotated + 1] = cycle[i]
138 end
139 rotated[#rotated + 1] = rotated[1]
140 return order, table.concat(rotated, " -> ")
141 end
142 chain[#chain + 1], visited[current] = current, #chain + 1
143 for _, dep in ipairs(predecessors(current)) do
144 if not removed[dep.from] then
145 current = dep.from
146 break
147 end
148 end
149 end
150end
151local function duration(name, override)
152 if override and override[name] ~= nil then
153 return override[name]
154 end
155 return tasks[name].base - tasks[name].crashed
156end
157local function margin(dep, schedule)
158 local a, b = schedule.times[dep.from], schedule.times[dep.to]
159 if dep.kind == "SS" then
160 return b.es - a.es - dep.lag
161 elseif dep.kind == "FF" then
162 return b.ef - a.ef - dep.lag
163 end
164 return b.es - a.ef - dep.lag
165end
166local function compute(order, override)
167 local result = {times = {}, total = 0}
168 for _, name in ipairs(order) do
169 local start, d = tasks[name].not_before, duration(name, override)
170 for _, dep in ipairs(predecessors(name)) do
171 local a = result.times[dep.from]
172 local value = a.ef + dep.lag
173 if dep.kind == "SS" then
174 value = a.es + dep.lag
175 elseif dep.kind == "FF" then
176 value = value - d
177 end
178 start = math.max(start, value)
179 end
180 result.times[name] = {es = start, ef = start + d}
181 result.total = math.max(result.total, start + d)
182 end
183 for i = #order, 1, -1 do
184 local name = order[i]
185 local value, d, finish = result.times[name], duration(name, override), result.total
186 for _, dep in ipairs(successors(name)) do
187 local b = result.times[dep.to]
188 local candidate = b.ls - dep.lag
189 if dep.kind == "SS" then
190 candidate = candidate + d
191 elseif dep.kind == "FF" then
192 candidate = b.lf - dep.lag
193 end
194 finish = math.min(finish, candidate)
195 end
196 value.lf, value.ls = finish, finish - d
197 value.tf, value.free = value.ls - value.es, result.total - value.ef
198 end
199 for _, name in ipairs(order) do
200 local value = result.times[name]
201 for _, dep in ipairs(successors(name)) do
202 value.free = math.min(value.free, margin(dep, result))
203 end
204 end
205 return result
206end
207local function critical(schedule)
208 local result = {}
209 for _, name in ipairs(keys(tasks)) do
210 if schedule.times[name].tf == 0 then
211 result[#result + 1] = name
212 end
213 end
214 return result
215end
216local function critical_paths(schedule)
217 local following, incoming = {}, {}
218 for _, name in ipairs(keys(tasks)) do
219 following[name] = {}
220 for _, dep in ipairs(successors(name)) do
221 if schedule.times[name].tf == 0 and schedule.times[dep.to].tf == 0 and margin(dep, schedule) == 0 then
222 following[name][#following[name] + 1], incoming[dep.to] = dep.to, true
223 end
224 end
225 end
226 local result = {}
227 local function visit(name, prefix)
228 local path = copy(prefix)
229 path[#path + 1] = name
230 if #following[name] == 0 then
231 result[#result + 1] = path
232 else
233 for _, next_name in ipairs(following[name]) do
234 visit(next_name, path)
235 end
236 end
237 end
238 for _, name in ipairs(critical(schedule)) do
239 if not incoming[name] then
240 visit(name, {})
241 end
242 end
243 return result
244end

252local function usage(schedule, resource)
253 local result = {}
254 for day = 0, schedule.total - 1 do
255 result[day + 1] = 0
256 end
257 for name, task in pairs(tasks) do
258 if task.resource == resource then
259 local value = schedule.times[name]
260 for day = value.es, value.ef - 1 do
261 result[day + 1] = result[day + 1] + task.units
262 end
263 end
264 end
265 return result
266end

268 local result = 0
269 for _, value in ipairs(values) do

274local function working_day(day)
275 return (day + 5) % 7 < 5 and not holidays[day]
276end
277local function start_order(schedule)
278 local names = keys(tasks)
279 table.sort(names, function(a, b)
280 if schedule.times[a].es ~= schedule.times[b].es then
281 return schedule.times[a].es < schedule.times[b].es
282 end
283 return a < b
284 end)
285 return names
286end
287local function calendar(schedule)
288 local days, day = {}, start_date
289 while #days < schedule.total do
290 if working_day(day) then
291 days[#days + 1] = day
292 end
293 day = day + 1
294 end
295 for _, name in ipairs(start_order(schedule)) do
296 local value = schedule.times[name]
297 emit("%s %s - %s", name, date_text(days[value.es + 1]), date_text(days[value.ef]))
298 end
299 local first, last = days[1], days[#days]
300 emit("project %s - %s, %d working days, %d calendar days", date_text(first), date_text(last), schedule.total, last - first + 1)
301 if deadline then
302 local a, b, count = math.min(last, deadline), math.max(last, deadline), 0
303 for date = a + 1, b do
304 if working_day(date) then
305 count = count + 1
306 end
307 end
308 if last <= deadline then
309 emit("deadline %s met, %d working days to spare", date_text(deadline), count)
310 else
311 emit("deadline %s missed by %d working days", date_text(deadline), count)
312 end
313 end
314 local selected, text = {}, {}
315 for date in pairs(holidays) do
316 if date >= first and date <= last and (date + 5) % 7 < 5 then
317 selected[#selected + 1] = date
318 end
319 end
320 table.sort(selected)
321 for _, date in ipairs(selected) do
322 text[#text + 1] = date_text(date)
323 end
324 emit("holidays in span: %s", #text > 0 and table.concat(text, ", ") or "none")
325end
326local function level(order, schedule, limit)
327 local function capacity(resource)
328 return capacities[resource] or limit
329 end
330 for _, name in ipairs(keys(tasks)) do
331 local task = tasks[name]
332 if task.units > capacity(task.resource) then
333 fail("capacity %d too small for %s", capacity(task.resource), name)
334 end
335 end
336 local decided, loads, sequence, total = {}, {}, {}, 0
337 while #sequence < #order do
338 local selected
339 for _, name in ipairs(keys(tasks)) do
340 if not decided[name] then
341 local ready = true
342 for _, dep in ipairs(predecessors(name)) do
343 if not decided[dep.from] then
344 ready = false
345 break
346 end
347 end
348 if ready and (not selected or schedule.times[name].ls < schedule.times[selected].ls) then
349 selected = name
350 end
351 end
352 end
353 local task = tasks[selected]
354 local start, d = task.not_before, duration(selected)
355 for _, dep in ipairs(predecessors(selected)) do
356 local a = decided[dep.from]
357 local value = a.ef + dep.lag
358 if dep.kind == "SS" then
359 value = a.es + dep.lag
360 elseif dep.kind == "FF" then
361 value = value - d
362 end
363 start = math.max(start, value)
364 end
365 local load = loads[task.resource] or {}
366 while true do
367 local valid = true
368 for day = start, start + d - 1 do
369 if (load[day + 1] or 0) + task.units > capacity(task.resource) then
370 valid = false
371 break
372 end
373 end
374 if valid then
375 break
376 end
377 start = start + 1
378 end
379 while #load < start + d do
380 load[#load + 1] = 0
381 end
382 for day = start, start + d - 1 do
383 load[day + 1] = load[day + 1] + task.units
384 end
385 loads[task.resource], decided[selected] = load, {es = start, ef = start + d}
386 sequence[#sequence + 1], total = selected, math.max(total, start + d)
387 end
388 emit("level capacity %d: duration %d -> %d", limit, schedule.total, total)
389 for _, name in ipairs(sequence) do
390 local value = decided[name]
391 emit("%s start %d finish %d shift %d", name, value.es, value.ef, value.es - schedule.times[name].es)
392 end
393 for _, resource in ipairs(resources()) do
394 emit("resource %s capacity %d peak %d -> %d", resource, capacity(resource), peak(usage(schedule, resource)), peak(loads[resource]))
395 end
396end

398local function command(w)
399 local cmd = w[1]
400 if cmd == "TASK" then
401 if tasks[w[2]] then
402 fail("task %s already exists", w[2])
403 end
404 tasks[w[2]] = {name = w[2], base = tonumber(w[3]), resource = w[4], units = tonumber(w[5]), not_before = 0, crashed = 0}
405 emit("task %s added", w[2])
406 return
407 elseif cmd == "DEP" or cmd == "UNDEP" then
408 get(w[2])
409 get(w[3])
410 local key = w[2] .. "\0" .. w[3]
411 if cmd == "DEP" then
412 if w[2] == w[3] then
413 fail("self dependency %s", w[2])
414 end
415 if dependencies[key] then
416 fail("dependency %s -> %s already exists", w[2], w[3])
417 end
418 dependencies[key] = {from = w[2], to = w[3], kind = w[4], lag = tonumber(w[5])}
419 emit("dependency %s -> %s %s%+d added", w[2], w[3], w[4], tonumber(w[5]))
420 else
421 if not dependencies[key] then
422 fail("no dependency %s -> %s", w[2], w[3])
423 end
424 dependencies[key] = nil
425 emit("dependency %s -> %s removed", w[2], w[3])
426 end
427 return
428 elseif cmd == "REMOVE" or cmd == "NOTBEFORE" or cmd == "CRASHCOST" or cmd == "UNCRASH" or
429 cmd == "PERT" or cmd == "SHOW" or cmd == "IMPACT" then
430 local task = get(w[2])
431 if cmd == "REMOVE" then
432 local count = 0
433 for key, dep in pairs(dependencies) do
434 if dep.from == task.name or dep.to == task.name then
435 dependencies[key], count = nil, count + 1
436 end
437 end
438 tasks[task.name] = nil
439 emit("task %s removed with %d dependencies", task.name, count)
440 elseif cmd == "NOTBEFORE" then
441 task.not_before = tonumber(w[3])
442 emit("notbefore %s %s", task.name, w[3])
443 elseif cmd == "CRASHCOST" then
444 local limit = tonumber(w[3])
445 if limit >= task.base then
446 fail("crash limit too large for %s", task.name)
447 end
448 if limit < task.crashed then
449 fail("crash limit below used days for %s", task.name)
450 end
451 task.crash_limit, task.cost = limit, tonumber(w[4])
452 emit("crash %s max %d cost %d", task.name, limit, task.cost)
453 elseif cmd == "UNCRASH" then
454 emit("uncrash %s restored %d days", task.name, task.crashed)
455 task.crashed = 0
456 elseif cmd == "PERT" then
457 local o, m, p = tonumber(w[3]), tonumber(w[4]), tonumber(w[5])
458 if o > m or m > p then
459 fail("invalid estimate")
460 end
461 task.estimate = {o, m, p}
462 emit("pert %s expected %s variance %s", task.name, decimal(o + 4 * m + p, 6), decimal((p - o) * (p - o), 36))
463 elseif cmd == "IMPACT" then
464 local reached, queue, at = {[task.name] = true}, {task.name}, 1
465 while at <= #queue do
466 local name = queue[at]
467 at = at + 1
468 for _, dep in ipairs(successors(name)) do
469 if not reached[dep.to] then
470 reached[dep.to], queue[#queue + 1] = true, dep.to
471 end
472 end
473 end
474 reached[task.name] = nil
475 local names = keys(reached)
476 emit("impact %s: %s", task.name, #names > 0 and table.concat(names, " ") or "none")
477 elseif cmd == "SHOW" then
478 emit("task %s duration %d resource %s units %d%s", task.name, duration(task.name), task.resource,
479 task.units, task.not_before ~= 0 and " notbefore " .. task.not_before or "")
480 for _, relation in ipairs({"preds", "succs"}) do
481 local deps = relation == "preds" and predecessors(task.name) or successors(task.name)
482 local parts = {}
483 for _, dep in ipairs(deps) do
484 parts[#parts + 1] = string.format("%s %s%+d", relation == "preds" and dep.from or dep.to, dep.kind, dep.lag)
485 end
486 emit("%s: %s", relation, #parts > 0 and table.concat(parts, ", ") or "none")
487 end
488 if task.crash_limit ~= nil then
489 emit("crash: max %d used %d cost %d", task.crash_limit, task.crashed, task.cost)
490 else
491 emit("crash: none")
492 end
493 if task.estimate then
494 emit("pert: %d %d %d", table.unpack(task.estimate))
495 else
496 emit("pert: none")
497 end
498 local order, cycle = topology()
499 if cycle then
500 emit("schedule: unavailable")
501 else
502 local value = compute(order).times[task.name]
503 emit("schedule: ES %d EF %d LS %d LF %d TF %d FF %d%s", value.es, value.ef, value.ls,
504 value.lf, value.tf, value.free, value.tf == 0 and " critical" or "")
505 end
506 end
507 return
508 elseif cmd == "CAPACITY" then
509 capacities[w[2]] = tonumber(w[3])
510 emit("capacity %s %s", w[2], w[3])
511 return
512 elseif cmd == "START" or cmd == "HOLIDAY" or cmd == "DEADLINE" then
513 local date = date_value(w[2])
514 if not date then
515 fail("invalid date %s", w[2])
516 end
517 if cmd == "HOLIDAY" then
518 if holidays[date] then
519 fail("duplicate holiday %s", w[2])
520 end
521 holidays[date] = true
522 elseif cmd == "START" then
523 start_date = date
524 else
525 deadline = date
526 end
527 emit("%s %s", cmd:lower(), date_text(date))
528 return
529 elseif cmd == "LIST" then
530 local names = keys(tasks)
531 emit("tasks %d, dependencies %d, holidays %d", #names, #keys(dependencies), #keys(holidays))
532 for _, name in ipairs(names) do
533 local task, previous = tasks[name], {}
534 for _, dep in ipairs(predecessors(name)) do
535 previous[#previous + 1] = dep.from
536 end
537 emit("%s duration %d resource %s units %d preds %s", name, duration(name), task.resource,
538 task.units, #previous > 0 and table.concat(previous, " ") or "none")
539 end
540 return
541 end
542 if not next(tasks) then
543 fail("no tasks")
544 end
545 local order, cycle = topology()
546 if cycle then
547 fail("cycle %s", cycle)
548 end
549 if cmd == "CALENDAR" and not start_date then
550 fail("no start date")
551 end
552 if cmd == "WHATIF" then
553 get(w[2])
554 end
555 if cmd == "VARIANCE" and not baseline then
556 fail("no baseline")
557 end
558 if cmd == "LOAD" then
559 local found = false
560 for _, task in pairs(tasks) do
561 if task.resource == w[2] then
562 found = true
563 end
564 end
565 if not found then
566 fail("unknown resource %s", w[2])
567 end
568 end
569 local schedule = compute(order)
570 if cmd == "CPM" then
571 local width = 4
572 for name in pairs(tasks) do
573 width = math.max(width, #name)
574 end
575 local header = string.format("%-" .. width .. "s", "TASK")
576 for _, column in ipairs({"DUR", "ES", "EF", "LS", "LF", "TF", "FF"}) do
577 header = header .. string.format("%5s", column)
578 end
579 emit("%s", header)
580 for _, name in ipairs(keys(tasks)) do
581 local value = schedule.times[name]
582 emit("%-" .. width .. "s%5d%5d%5d%5d%5d%5d%5d%s", name, duration(name), value.es,
583 value.ef, value.ls, value.lf, value.tf, value.free, value.tf == 0 and " *" or "")
584 end
585 emit("project duration %d", schedule.total)
586 elseif cmd == "CRITICAL" or cmd == "PERTSUM" then
587 local paths = critical_paths(schedule)
588 if cmd == "CRITICAL" then
589 emit("critical paths: %d", #paths)
590 end
591 for _, path in ipairs(paths) do
592 local text = table.concat(path, " -> ")
593 if cmd == "CRITICAL" then
594 emit("%s", text)
595 else
596 local expected, variance = 0, 0
597 for _, name in ipairs(path) do
598 local d = duration(name)
599 local values = tasks[name].estimate or {d, d, d}
600 expected = expected + values[1] + 4 * values[2] + values[3]
601 variance = variance + (values[3] - values[1]) * (values[3] - values[1])
602 end
603 emit("%s: expected %s variance %s", text, decimal(expected, 6), decimal(variance, 36))
604 end
605 end
606 elseif cmd == "SLACK" then
607 local groups = {}
608 for _, name in ipairs(keys(tasks)) do
609 local value = schedule.times[name].tf
610 groups[value] = groups[value] or {}
611 groups[value][#groups[value] + 1] = name
612 end
613 for _, value in ipairs(keys(groups)) do
614 emit("float %d: %s", value, table.concat(groups[value], " "))
615 end
616 elseif cmd == "DRAG" then
617 for _, name in ipairs(critical(schedule)) do
618 emit("%s drag %d", name, schedule.total - compute(order, {[name] = 0}).total)
619 end
620 elseif cmd == "STATS" then
621 local counts = {FS = 0, SS = 0, FF = 0}
622 for _, dep in pairs(dependencies) do
623 counts[dep.kind] = counts[dep.kind] + 1
624 end
625 local critical_count, names = #critical(schedule), keys(tasks)
626 emit("tasks %d, critical %d, dependencies %d (FS %d, SS %d, FF %d)", #names, critical_count,
627 #keys(dependencies), counts.FS, counts.SS, counts.FF)
628 local work = {}
629 for _, resource in ipairs(resources()) do
630 local total = 0
631 for name, task in pairs(tasks) do
632 if task.resource == resource then
633 total = total + duration(name) * task.units
634 end
635 end
636 work[#work + 1] = resource .. " " .. total
637 end
638 emit("work %s", table.concat(work, ", "))
639 local total = 0
640 for _, value in pairs(schedule.times) do
641 total = total + value.tf
642 end
643 emit("float total %d, average %s", total, decimal(total, #names))
644 emit("duration %d, critical share %s%%", schedule.total, decimal(100 * critical_count, #names))
645 elseif cmd == "CALENDAR" then
646 calendar(schedule)
647 elseif cmd == "GANTT" then
648 if schedule.total > 60 then
649 fail("chart too wide (%d days)", schedule.total)
650 end
651 local width = 0
652 for name in pairs(tasks) do
653 width = math.max(width, #name)
654 end
655 local header = string.rep(" ", width) .. " |"
656 for day = 0, schedule.total - 1 do
657 header = header .. day % 10
658 end
659 emit("%s|", header)
660 for _, name in ipairs(start_order(schedule)) do
661 local value, bar = schedule.times[name], {}
662 for day = 0, schedule.total - 1 do
663 if value.es <= day and day < value.ef then
664 bar[#bar + 1] = value.tf == 0 and "=" or "-"
665 elseif value.ef <= day and day < value.lf then
666 bar[#bar + 1] = "."
667 else
668 bar[#bar + 1] = " "
669 end
670 end
671 emit("%-" .. width .. "s |%s|", name, table.concat(bar))
672 end
673 elseif cmd == "LEVEL" then
674 level(order, schedule, tonumber(w[2]))
675 elseif cmd == "LOAD" then
676 local values = usage(schedule, w[2])
677 local highest, first = peak(values), 1
678 while values[first] ~= highest do
679 first = first + 1
680 end
681 emit("load %s: %s", w[2], table.concat(values, " "))
682 emit("peak %d on day %d", highest, first - 1)
683 if capacities[w[2]] then
684 local count = 0
685 for _, value in ipairs(values) do
686 if value > capacities[w[2]] then
687 count = count + 1
688 end
689 end
690 emit("capacity %d exceeded on %d days", capacities[w[2]], count)
691 end
692 elseif cmd == "CRASH" then
693 local initial, remaining, steps, spent, reason = schedule.total, tonumber(w[2]), 0, 0, nil
694 while true do
695 schedule = compute(order)
696 local chosen
697 for _, name in ipairs(critical(schedule)) do
698 local task = tasks[name]
699 if task.crash_limit and task.crashed < task.crash_limit and
700 compute(order, {[name] = duration(name) - 1}).total < schedule.total then
701 if not chosen or task.cost < tasks[chosen].cost then
702 chosen = name
703 end
704 end
705 end
706 if not chosen then
707 reason = "no candidate"
708 break
709 end
710 local task = tasks[chosen]
711 if task.cost > remaining then
712 reason = "insufficient budget"
713 break
714 end
715 local before = duration(chosen)
716 task.crashed, remaining, spent, steps = task.crashed + 1, remaining - task.cost, spent + task.cost, steps + 1
717 emit("step %d: %s %d -> %d, cost %d, project %d", steps, chosen, before, before - 1, task.cost, compute(order).total)
718 end
719 emit("crash done: %d days, cost %d, project %d -> %d, budget left %d, %s", steps, spent,
720 initial, schedule.total, remaining, reason)
721 elseif cmd == "WHATIF" then
722 local after = compute(order, {[w[2]] = tonumber(w[3])})
723 emit("whatif %s %d -> %s: project %d -> %d", w[2], duration(w[2]), w[3], schedule.total, after.total)
724 for _, name in ipairs(keys(tasks)) do
725 local a, b, changes = schedule.times[name], after.times[name], {}
726 for _, field in ipairs({"es", "ef", "tf"}) do
727 if a[field] ~= b[field] then
728 changes[#changes + 1] = string.format("%s %d->%d", field:upper(), a[field], b[field])
729 end
730 end
731 if #changes > 0 then
732 emit("%s %s", name, table.concat(changes, " "))
733 end
734 end
735 local before_names, after_names = table.concat(critical(schedule), " "), table.concat(critical(after), " ")
736 if before_names ~= after_names then
737 emit("critical %s -> %s", before_names, after_names)
738 end
739 elseif cmd == "BASELINE" then
740 baseline = {total = schedule.total, times = {}}
741 for name, value in pairs(schedule.times) do
742 baseline.times[name] = {es = value.es, ef = value.ef}
743 end
744 emit("baseline saved: %d tasks, duration %d", #keys(tasks), schedule.total)
745 elseif cmd == "VARIANCE" then
746 emit("variance: duration %d -> %d (%+d)", baseline.total, schedule.total, schedule.total - baseline.total)
747 local names, count = {}, 0
748 for name in pairs(baseline.times) do
749 names[name] = true
750 end
751 for name in pairs(tasks) do
752 names[name] = true
753 end
754 for _, name in ipairs(keys(names)) do
755 local a, b = baseline.times[name], schedule.times[name]
756 if not b then
757 emit("%s removed", name)
758 count = count + 1
759 elseif not a then
760 emit("%s new", name)
761 count = count + 1
762 elseif a.es ~= b.es or a.ef ~= b.ef then
763 emit("%s start %+d finish %+d", name, b.es - a.es, b.ef - a.ef)
764 count = count + 1
765 end
766 end
767 if count == 0 then
768 emit("no task changes")
769 end
770 end
771end

774for _ = 1, count do
775 local words = {}
776 for word in io.read("*l"):gmatch("[^ ]+") do
777 words[#words + 1] = word
778 end
779 local ok, message = pcall(command, words)
780 if not ok then
781 emit("error: %s", message)
782 end
783end
