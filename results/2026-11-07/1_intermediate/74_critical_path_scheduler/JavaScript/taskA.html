3const tasks = new Map();
4let dependencies = [];
5const capacities = new Map();
6const holidays = new Set();
7const output = [];
8let startDate = null, deadline = null, baseline = null;
9const epoch = Date.UTC(2000, 0, 1);
10const dayMilliseconds = 86400000;
11function fail(message) { throw new Error('error: ' + message); }
12function compare(a, b) { return a < b ? -1 : a > b ? 1 : 0; }
13function names() { return [...tasks.keys()].sort(); }
14function task(name) {
15 const result = tasks.get(name);
16 if (result === undefined) fail('unknown task ' + name);
17 return result;
18}
19function duration(item) { return item.base - item.used; }
20function signed(value) { return value >= 0 ? '+' + value : String(value); }
21function decimal(numerator, denominator) {
22 const hundredths = Math.floor(numerator * 100 / denominator);
23 return Math.floor(hundredths / 100) + '.' + String(hundredths % 100).padStart(2, '0');
24}
25function predecessors(name) { return dependencies.filter(dep => dep.b === name).sort((a, b) => compare(a.a, b.a)); }
26function successors(name) { return dependencies.filter(dep => dep.a === name).sort((a, b) => compare(a.b, b.b)); }
27function topological() {
28 const remaining = new Set(names()), order = [];
29 while (true) {
30 const next = [...remaining].find(name => predecessors(name).every(dep => !remaining.has(dep.a)));
31 if (next === undefined) return { order, remaining };
32 remaining.delete(next); order.push(next);
33 }
34}
35function cycle() {
36 const { remaining } = topological();
37 if (!remaining.size) return null;
38 const trail = [[...remaining][0]];
39 while (true) {
40 const next = predecessors(trail[trail.length - 1]).find(dep => remaining.has(dep.a)).a;
41 const at = trail.indexOf(next);
42 if (at >= 0) {
43 const loop = [next, ...trail.slice(at + 1).reverse()];
44 const first = loop.indexOf([...loop].sort()[0]), rotated = [...loop.slice(first), ...loop.slice(0, first)];
45 return [...rotated, rotated[0]].join(' -> ');
46 }
47 trail.push(next);
48 }
49}
50function schedule(overrides = new Map()) {
51 const { order } = topological(), rows = new Map();
52 let total = 0;
53 for (const name of order) {
54 const item = tasks.get(name), d = overrides.get(name) ?? duration(item);
55 let es = item.notBefore;
56 for (const dep of predecessors(name)) {
57 const old = rows.get(dep.a);
58 es = Math.max(es, (dep.kind === 'SS' ? old.es : old.ef) + dep.lag - (dep.kind === 'FF' ? d : 0));
59 }
60 rows.set(name, { duration: d, es, ef: es + d, ls: 0, lf: 0, tf: 0, ff: 0 });
61 total = Math.max(total, es + d);
62 }
63 for (const name of [...order].reverse()) {
64 const row = rows.get(name);
65 let lf = total;
66 for (const dep of successors(name)) {
67 const next = rows.get(dep.b);
68 lf = Math.min(lf, (dep.kind === 'FF' ? next.lf : next.ls) - dep.lag + (dep.kind === 'SS' ? row.duration : 0));
69 }
70 row.lf = lf; row.ls = lf - row.duration; row.tf = row.ls - row.es;
71 }
72 for (const name of order) {
73 const row = rows.get(name);
74 row.ff = total - row.ef;
75 for (const dep of successors(name)) row.ff = Math.min(row.ff, margin(dep, rows));
76 }
77 return { rows, total };
78}
79function margin(dep, rows) {
80 const a = rows.get(dep.a), b = rows.get(dep.b);
81 return (dep.kind === 'FF' ? b.ef : b.es) - (dep.kind === 'SS' ? a.es : a.ef) - dep.lag;
82}
83function critical(schedule) { return names().filter(name => schedule.rows.get(name).tf === 0); }
84function criticalPaths(schedule) {
85 const criticalNames = new Set(critical(schedule));
86 const links = dependencies.filter(dep => criticalNames.has(dep.a) && criticalNames.has(dep.b) && margin(dep, schedule.rows) === 0);
87 const roots = [...criticalNames].filter(name => !links.some(dep => dep.b === name)), result = [];
88 function visit(path) {
89 const next = links.filter(dep => dep.a === path[path.length - 1]).map(dep => dep.b).sort();
90 if (!next.length) result.push(path);
91 for (const name of next) visit([...path, name]);
92 }
93 for (const name of roots) visit([name]);
94 return result;
95}
96function parseDate(word) {
97 const [year, month, day] = word.split('-').map(Number), value = new Date(Date.UTC(year, month - 1, day));
98 if (value.getUTCFullYear() !== year || value.getUTCMonth() !== month - 1 || value.getUTCDate() !== day) fail('invalid date ' + word);
99 return (value.getTime() - epoch) / dayMilliseconds;
100}
101function weekday(day) { return (day + 5) % 7; }
102function dateText(day) {
103 const date = new Date(epoch + day * dayMilliseconds);
104 return date.getUTCFullYear().toString().padStart(4, '0') + '-' + String(date.getUTCMonth() + 1).padStart(2, '0') + '-' + String(date.getUTCDate()).padStart(2, '0') + ' ' + ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'][weekday(day)];
105}
106function working(day) { return weekday(day) < 5 && !holidays.has(day); }
107function dayDates(count) {
108 const result = [];
109 for (let day = startDate; result.length < count; day++) if (working(day)) result.push(day);
110 return result;
111}
112function workingBetween(a, b) {
113 let count = 0;
114 for (let day = a + 1; day <= b; day++) if (working(day)) count++;
115 return count;
116}
117function resources() { return [...new Set([...tasks.values()].map(item => item.resource))].sort(); }
118function usage(resource, schedule) {
119 const result = Array(schedule.total).fill(0);
120 for (const item of tasks.values()) if (item.resource === resource) {
121 const row = schedule.rows.get(item.id);
122 for (let day = row.es; day < row.ef; day++) result[day] += item.units;
123 }
124 return result;
125}
126function level(limit, original) {
127 const capacity = resource => capacities.get(resource) ?? limit;
128 for (const name of names()) if (tasks.get(name).units > capacity(tasks.get(name).resource)) fail(`capacity ${capacity(tasks.get(name).resource)} too small for ${name}`);
129 const undecided = new Set(names()), decided = new Map(), used = new Map(resources().map(name => [name, []])), order = [];
130 while (undecided.size) {
131 const available = [...undecided].filter(name => predecessors(name).every(dep => decided.has(dep.a)));
132 available.sort((a, b) => original.rows.get(a).ls - original.rows.get(b).ls || compare(a, b));
133 const name = available[0], item = tasks.get(name), d = duration(item), days = used.get(item.resource);
134 let start = item.notBefore;
135 for (const dep of predecessors(name)) {
136 const old = decided.get(dep.a);
137 start = Math.max(start, (dep.kind === 'SS' ? old.es : old.ef) + dep.lag - (dep.kind === 'FF' ? d : 0));
138 }
139 while (true) {
140 let valid = true;
141 for (let day = start; day < start + d; day++) if ((days[day] ?? 0) + item.units > capacity(item.resource)) { valid = false; break; }
142 if (valid) break;
143 start++;
144 }
145 for (let day = start; day < start + d; day++) days[day] = (days[day] ?? 0) + item.units;
146 decided.set(name, { es: start, ef: start + d }); undecided.delete(name); order.push(name);
147 }
148 const finish = Math.max(...[...decided.values()].map(row => row.ef));
149 output.push(`level capacity ${limit}: duration ${original.total} -> ${finish}`);
150 for (const name of order) { const row = decided.get(name); output.push(`${name} start ${row.es} finish ${row.ef} shift ${row.es - original.rows.get(name).es}`); }
151 for (const resource of resources()) output.push(`resource ${resource} capacity ${capacity(resource)} peak ${Math.max(...usage(resource, original))} -> ${used.get(resource).reduce((peak, value) => Math.max(peak, value), 0)}`);
152}
153function execute(commandLine) {
154 const [command, ...args] = commandLine.split(' '), a = args[0], b = args[1];
155 if (command === 'TASK') {
156 if (tasks.has(a)) fail(`task ${a} already exists`);
157 tasks.set(a, { id: a, base: Number(b), resource: args[2], units: Number(args[3]), notBefore: 0, used: 0, crash: null, pert: null });
158 output.push(`task ${a} added`); return;
159 }
160 if (command === 'DEP' || command === 'UNDEP') {
161 task(a); task(b);
162 const at = dependencies.findIndex(dep => dep.a === a && dep.b === b);
163 if (command === 'DEP') {
164 if (a === b) fail('self dependency ' + a);
165 if (at >= 0) fail(`dependency ${a} -> ${b} already exists`);
166 dependencies.push({ a, b, kind: args[2], lag: Number(args[3]) });
167 output.push(`dependency ${a} -> ${b} ${args[2]}${signed(Number(args[3]))} added`);
168 } else {
169 if (at < 0) fail(`no dependency ${a} -> ${b}`);
170 dependencies.splice(at, 1); output.push(`dependency ${a} -> ${b} removed`);
171 }
172 return;
173 }
174 if (['REMOVE', 'NOTBEFORE', 'CRASHCOST', 'UNCRASH', 'PERT', 'SHOW', 'IMPACT'].includes(command)) {
175 const item = task(a);
176 if (command === 'REMOVE') {
177 const before = dependencies.length;
178 dependencies = dependencies.filter(dep => dep.a !== a && dep.b !== a); tasks.delete(a);
179 output.push(`task ${a} removed with ${before - dependencies.length} dependencies`);
180 } else if (command === 'NOTBEFORE') { item.notBefore = Number(b); output.push(`notbefore ${a} ${b}`); }
181 else if (command === 'CRASHCOST') {
182 if (Number(b) >= item.base) fail('crash limit too large for ' + a);
183 if (Number(b) < item.used) fail('crash limit below used days for ' + a);
184 item.crash = { limit: Number(b), cost: Number(args[2]) }; output.push(`crash ${a} max ${b} cost ${args[2]}`);
185 } else if (command === 'UNCRASH') { output.push(`uncrash ${a} restored ${item.used} days`); item.used = 0; }
186 else if (command === 'PERT') {
187 const [o, m, p] = args.slice(1).map(Number);
188 if (o > m || m > p) fail('invalid estimate');
189 item.pert = [o, m, p]; output.push(`pert ${a} expected ${decimal(o + 4 * m + p, 6)} variance ${decimal((p - o) ** 2, 36)}`);
190 } else if (command === 'IMPACT') {
191 const seen = new Set([a]), queue = [a];
192 for (let i = 0; i < queue.length; i++) for (const dep of successors(queue[i])) if (!seen.has(dep.b)) { seen.add(dep.b); queue.push(dep.b); }
193 seen.delete(a); output.push(`impact ${a}: ${[...seen].sort().join(' ') || 'none'}`);
194 } else {
195 output.push(`task ${a} duration ${duration(item)} resource ${item.resource} units ${item.units}${item.notBefore ? ` notbefore ${item.notBefore}` : ''}`,
196 'preds: ' + (predecessors(a).map(dep => `${dep.a} ${dep.kind}${signed(dep.lag)}`).join(', ') || 'none'),
197 'succs: ' + (successors(a).map(dep => `${dep.b} ${dep.kind}${signed(dep.lag)}`).join(', ') || 'none'),
198 item.crash ? `crash: max ${item.crash.limit} used ${item.used} cost ${item.crash.cost}` : 'crash: none',
199 item.pert ? `pert: ${item.pert.join(' ')}` : 'pert: none');
200 if (cycle() !== null) output.push('schedule: unavailable');
201 else { const row = schedule().rows.get(a); output.push(`schedule: ES ${row.es} EF ${row.ef} LS ${row.ls} LF ${row.lf} TF ${row.tf} FF ${row.ff}${row.tf === 0 ? ' critical' : ''}`); }
202 }
203 return;
204 }
205 if (command === 'CAPACITY') { capacities.set(a, Number(b)); output.push(`capacity ${a} ${b}`); return; }
206 if (['START', 'HOLIDAY', 'DEADLINE'].includes(command)) {
207 const day = parseDate(a);
208 if (command === 'HOLIDAY' && holidays.has(day)) fail('duplicate holiday ' + a);
209 if (command === 'START') startDate = day;
210 else if (command === 'DEADLINE') deadline = day;
211 else holidays.add(day);
212 output.push(command.toLowerCase() + ' ' + dateText(day)); return;
213 }
214 if (command === 'LIST') {
215 output.push(`tasks ${tasks.size}, dependencies ${dependencies.length}, holidays ${holidays.size}`);
216 for (const name of names()) { const item = tasks.get(name); output.push(`${name} duration ${duration(item)} resource ${item.resource} units ${item.units} preds ${predecessors(name).map(dep => dep.a).join(' ') || 'none'}`); }
217 return;
218 }
219 if (!tasks.size) fail('no tasks');
220 const foundCycle = cycle();
221 if (foundCycle !== null) fail('cycle ' + foundCycle);
222 if (command === 'CALENDAR' && startDate === null) fail('no start date');
223 if (command === 'LOAD' && !resources().includes(a)) fail('unknown resource ' + a);
224 if (command === 'WHATIF') task(a);
225 if (command === 'VARIANCE' && baseline === null) fail('no baseline');
226 const current = schedule();
227 const byStart = names().sort((a, b) => current.rows.get(a).es - current.rows.get(b).es || compare(a, b));
228 switch (command) {
229 case 'CPM': {
230 const width = Math.max(4, ...names().map(name => name.length));
231 output.push('TASK'.padEnd(width) + ['DUR', 'ES', 'EF', 'LS', 'LF', 'TF', 'FF'].map(word => word.padStart(5)).join(''));
232 for (const name of names()) {
233 const row = current.rows.get(name);
234 output.push(name.padEnd(width) + [row.duration, row.es, row.ef, row.ls, row.lf, row.tf, row.ff].map(value => String(value).padStart(5)).join('') + (row.tf === 0 ? ' *' : ''));
235 }
236 output.push(`project duration ${current.total}`); break;
237 }
238 case 'CRITICAL': {
239 const paths = criticalPaths(current);
240 output.push(`critical paths: ${paths.length}`, ...paths.map(path => path.join(' -> '))); break;
241 }
242 case 'SLACK':
243 for (const value of [...new Set([...current.rows.values()].map(row => row.tf))].sort((a, b) => a - b)) output.push(`float ${value}: ${names().filter(name => current.rows.get(name).tf === value).join(' ')}`);
244 break;
245 case 'DRAG':
246 for (const name of critical(current)) output.push(`${name} drag ${current.total - schedule(new Map([[name, 0]])).total}`);
247 break;
248 case 'STATS': {
249 const count = critical(current).length, floats = [...current.rows.values()].reduce((sum, row) => sum + row.tf, 0);
250 output.push(`tasks ${tasks.size}, critical ${count}, dependencies ${dependencies.length} (FS ${dependencies.filter(dep => dep.kind === 'FS').length}, SS ${dependencies.filter(dep => dep.kind === 'SS').length}, FF ${dependencies.filter(dep => dep.kind === 'FF').length})`,
251 'work ' + resources().map(resource => `${resource} ${[...tasks.values()].filter(item => item.resource === resource).reduce((sum, item) => sum + duration(item) * item.units, 0)}`).join(', '),
252 `float total ${floats}, average ${decimal(floats, tasks.size)}`, `duration ${current.total}, critical share ${decimal(100 * count, tasks.size)}%`);
253 break;
254 }
255 case 'CALENDAR': {
256 const dates = dayDates(current.total), first = dates[0], last = dates[dates.length - 1];
257 for (const name of byStart) { const row = current.rows.get(name); output.push(`${name} ${dateText(dates[row.es])} - ${dateText(dates[row.ef - 1])}`); }
258 output.push(`project ${dateText(first)} - ${dateText(last)}, ${current.total} working days, ${last - first + 1} calendar days`);
259 if (deadline !== null) output.push(last <= deadline ? `deadline ${dateText(deadline)} met, ${workingBetween(last, deadline)} working days to spare` : `deadline ${dateText(deadline)} missed by ${workingBetween(deadline, last)} working days`);
260 const within = [...holidays].filter(day => weekday(day) < 5 && day >= first && day <= last).sort((a, b) => a - b);
261 output.push('holidays in span: ' + (within.map(dateText).join(', ') || 'none')); break;
262 }
263 case 'GANTT': {
264 if (current.total > 60) fail(`chart too wide (${current.total} days)`);
265 const width = Math.max(...names().map(name => name.length));
266 output.push(' '.repeat(width) + ' |' + Array.from({ length: current.total }, (_, i) => i % 10).join('') + '|');
267 for (const name of byStart) {
268 const row = current.rows.get(name), bar = Array.from({ length: current.total }, (_, day) => day >= row.es && day < row.ef ? row.tf === 0 ? '=' : '-' : day >= row.ef && day < row.lf ? '.' : ' ').join('');
269 output.push(name.padEnd(width) + ' |' + bar + '|');
270 }
271 break;
272 }
273 case 'LEVEL': level(Number(a), current); break;
274 case 'LOAD': {
275 const days = usage(a, current), peak = Math.max(...days);
276 output.push(`load ${a}: ${days.join(' ')}`, `peak ${peak} on day ${days.indexOf(peak)}`);
277 if (capacities.has(a)) output.push(`capacity ${capacities.get(a)} exceeded on ${days.filter(value => value > capacities.get(a)).length} days`);
278 break;
279 }
280 case 'PERTSUM':
281 for (const path of criticalPaths(current)) {
282 let expected = 0, variance = 0;
283 for (const name of path) { const item = tasks.get(name), d = duration(item), [o, m, p] = item.pert ?? [d, d, d]; expected += o + 4 * m + p; variance += (p - o) ** 2; }
284 output.push(`${path.join(' -> ')}: expected ${decimal(expected, 6)} variance ${decimal(variance, 36)}`);
285 }
286 break;
287 case 'CRASH': {
288 let remaining = Number(a), count = 0, total = current.total, reason;
289 while (true) {
290 const now = schedule();
291 const candidates = critical(now).filter(name => { const item = tasks.get(name); return item.crash !== null && item.used < item.crash.limit && schedule(new Map([[name, duration(item) - 1]])).total < now.total; });
292 candidates.sort((a, b) => tasks.get(a).crash.cost - tasks.get(b).crash.cost || compare(a, b));
293 if (!candidates.length) { reason = 'no candidate'; break; }
294 const name = candidates[0], item = tasks.get(name), cost = item.crash.cost;
295 if (cost > remaining) { reason = 'insufficient budget'; break; }
296 const old = duration(item); item.used++; remaining -= cost; count++; total = schedule().total;
297 output.push(`step ${count}: ${name} ${old} -> ${duration(item)}, cost ${cost}, project ${total}`);
298 }
299 output.push(`crash done: ${count} days, cost ${Number(a) - remaining}, project ${current.total} -> ${total}, budget left ${remaining}, ${reason}`); break;
300 }
301 case 'WHATIF': {
302 const after = schedule(new Map([[a, Number(b)]]));
303 output.push(`whatif ${a} ${duration(tasks.get(a))} -> ${b}: project ${current.total} -> ${after.total}`);
304 for (const name of names()) {
305 const before = current.rows.get(name), next = after.rows.get(name), fields = ['es', 'ef', 'tf'].filter(key => before[key] !== next[key]);
306 if (fields.length) output.push(name + fields.map(key => ` ${key.toUpperCase()} ${before[key]}->${next[key]}`).join(''));
307 }
308 const beforeCritical = critical(current).join(' '), afterCritical = critical(after).join(' ');
309 if (beforeCritical !== afterCritical) output.push(`critical ${beforeCritical} -> ${afterCritical}`);
310 break;
311 }
312 case 'BASELINE':
313 baseline = { total: current.total, rows: new Map([...current.rows].map(([name, row]) => [name, { es: row.es, ef: row.ef }])) };
314 output.push(`baseline saved: ${tasks.size} tasks, duration ${current.total}`); break;
315 case 'VARIANCE': {
316 output.push(`variance: duration ${baseline.total} -> ${current.total} (${signed(current.total - baseline.total)})`);
317 const before = output.length;
318 for (const name of [...new Set([...baseline.rows.keys(), ...tasks.keys()])].sort()) {
319 const old = baseline.rows.get(name), row = current.rows.get(name);
320 if (row === undefined) output.push(name + ' removed');
321 else if (old === undefined) output.push(name + ' new');
322 else if (old.es !== row.es || old.ef !== row.ef) output.push(`${name} start ${signed(row.es - old.es)} finish ${signed(row.ef - old.ef)}`);
323 }
324 if (output.length === before) output.push('no task changes');
325 break;
326 }
327 }
328}
329const input = fs.readFileSync(0, 'utf8').split(/\r?\n/), count = Number(input[0]);
330for (const command of input.slice(1, count + 1)) {
331 try { execute(command); } catch (error) { output.push(error.message); }
332}
333if (output.length) process.stdout.write(output.join('\n') + '\n');
