3interface Task { id: string; base: number; resource: string; units: number; notBefore: number; used: number; crash: { limit: number; cost: number } | null; pert: [number, number, number] | null }
4interface Dependency { a: string; b: string; kind: string; lag: number }
5interface Timing { duration: number; es: number; ef: number; ls: number; lf: number; tf: number; ff: number }
6interface StartFinish { es: number; ef: number }
7interface Schedule { rows: Map<string, Timing>; total: number }
8interface Baseline { rows: Map<string, StartFinish>; total: number }

10const tasks = new Map<string, Task>();
11let dependencies: Dependency[] = [];
12const capacities = new Map<string, number>();
13const holidays = new Set<number>();
14const output: string[] = [];
15let startDate: number | null = null, deadline: number | null = null, baseline: Baseline | null = null;
16const epoch = Date.UTC(2000, 0, 1);
17const dayMilliseconds = 86400000;
18function fail(message: string): never { throw new Error('error: ' + message); }
19function compare(a: string, b: string): number { return a < b ? -1 : a > b ? 1 : 0; }
20function names(): string[] { return [...tasks.keys()].sort(); }
21function task(name: string): Task {
22 const result = tasks.get(name);
23 if (result === undefined) fail('unknown task ' + name);
24 return result;
25}
26function duration(item: Task): number { return item.base - item.used; }
27function signed(value: number): string { return value >= 0 ? '+' + value : String(value); }
28function decimal(numerator: number, denominator: number): string {
29 const hundredths = Math.floor(numerator * 100 / denominator);
30 return Math.floor(hundredths / 100) + '.' + String(hundredths % 100).padStart(2, '0');
31}
32function predecessors(name: string): Dependency[] { return dependencies.filter(dep => dep.b === name).sort((a, b) => compare(a.a, b.a)); }
33function successors(name: string): Dependency[] { return dependencies.filter(dep => dep.a === name).sort((a, b) => compare(a.b, b.b)); }
34function topological(): { order: string[]; remaining: Set<string> } {
35 const remaining = new Set(names()), order: string[] = [];
36 while (true) {
37 const next = [...remaining].find(name => predecessors(name).every(dep => !remaining.has(dep.a)));
38 if (next === undefined) return { order, remaining };
39 remaining.delete(next); order.push(next);
40 }
41}
42function cycle(): string | null {
43 const { remaining } = topological();
44 if (!remaining.size) return null;
45 const trail = [[...remaining][0]];
46 while (true) {
47 const next = predecessors(trail[trail.length - 1]).find(dep => remaining.has(dep.a))!.a;
48 const at = trail.indexOf(next);
49 if (at >= 0) {
50 const loop = [next, ...trail.slice(at + 1).reverse()];
51 const first = loop.indexOf([...loop].sort()[0]), rotated = [...loop.slice(first), ...loop.slice(0, first)];
52 return [...rotated, rotated[0]].join(' -> ');
53 }
54 trail.push(next);
55 }
56}
57function schedule(overrides: Map<string, number> = new Map()): Schedule {
58 const { order } = topological(), rows = new Map<string, Timing>();
59 let total = 0;
60 for (const name of order) {
61 const item = tasks.get(name)!, d = overrides.get(name) ?? duration(item);
62 let es = item.notBefore;
63 for (const dep of predecessors(name)) {
64 const old = rows.get(dep.a)!;
65 es = Math.max(es, (dep.kind === 'SS' ? old.es : old.ef) + dep.lag - (dep.kind === 'FF' ? d : 0));
66 }
67 rows.set(name, { duration: d, es, ef: es + d, ls: 0, lf: 0, tf: 0, ff: 0 });
68 total = Math.max(total, es + d);
69 }
70 for (const name of [...order].reverse()) {
71 const row = rows.get(name)!;
72 let lf = total;
73 for (const dep of successors(name)) {
74 const next = rows.get(dep.b)!;
75 lf = Math.min(lf, (dep.kind === 'FF' ? next.lf : next.ls) - dep.lag + (dep.kind === 'SS' ? row.duration : 0));
76 }
77 row.lf = lf; row.ls = lf - row.duration; row.tf = row.ls - row.es;
78 }
79 for (const name of order) {
80 const row = rows.get(name)!;
81 row.ff = total - row.ef;
82 for (const dep of successors(name)) row.ff = Math.min(row.ff, margin(dep, rows));
83 }
84 return { rows, total };
85}
86function margin(dep: Dependency, rows: Map<string, Timing>): number {
87 const a = rows.get(dep.a)!, b = rows.get(dep.b)!;
88 return (dep.kind === 'FF' ? b.ef : b.es) - (dep.kind === 'SS' ? a.es : a.ef) - dep.lag;
89}
90function critical(schedule: Schedule): string[] { return names().filter(name => schedule.rows.get(name)!.tf === 0); }
91function criticalPaths(schedule: Schedule): string[][] {
92 const criticalNames = new Set(critical(schedule));
93 const links = dependencies.filter(dep => criticalNames.has(dep.a) && criticalNames.has(dep.b) && margin(dep, schedule.rows) === 0);
94 const roots = [...criticalNames].filter(name => !links.some(dep => dep.b === name)), result: string[][] = [];
95 function visit(path: string[]): void {
96 const next = links.filter(dep => dep.a === path[path.length - 1]).map(dep => dep.b).sort();
97 if (!next.length) result.push(path);
98 for (const name of next) visit([...path, name]);
99 }
100 for (const name of roots) visit([name]);
101 return result;
102}
103function parseDate(word: string): number {
104 const [year, month, day] = word.split('-').map(Number), value = new Date(Date.UTC(year, month - 1, day));
105 if (value.getUTCFullYear() !== year || value.getUTCMonth() !== month - 1 || value.getUTCDate() !== day) fail('invalid date ' + word);
106 return (value.getTime() - epoch) / dayMilliseconds;
107}
108function weekday(day: number): number { return (day + 5) % 7; }
109function dateText(day: number): string {
110 const date = new Date(epoch + day * dayMilliseconds);
111 return date.getUTCFullYear().toString().padStart(4, '0') + '-' + String(date.getUTCMonth() + 1).padStart(2, '0') + '-' + String(date.getUTCDate()).padStart(2, '0') + ' ' + ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'][weekday(day)];
112}
113function working(day: number): boolean { return weekday(day) < 5 && !holidays.has(day); }
114function dayDates(count: number): number[] {
115 const result: number[] = [];
116 for (let day = startDate!; result.length < count; day++) if (working(day)) result.push(day);
117 return result;
118}
119function workingBetween(a: number, b: number): number {
120 let count = 0;
121 for (let day = a + 1; day <= b; day++) if (working(day)) count++;
122 return count;
123}
124function resources(): string[] { return [...new Set([...tasks.values()].map(item => item.resource))].sort(); }
125function usage(resource: string, schedule: Schedule): number[] {
126 const result = Array<number>(schedule.total).fill(0);
127 for (const item of tasks.values()) if (item.resource === resource) {
128 const row = schedule.rows.get(item.id)!;
129 for (let day = row.es; day < row.ef; day++) result[day] += item.units;
130 }
131 return result;
132}
133function level(limit: number, original: Schedule): void {
134 const capacity = (resource: string): number => capacities.get(resource) ?? limit;
135 for (const name of names()) if (tasks.get(name)!.units > capacity(tasks.get(name)!.resource)) fail(`capacity ${capacity(tasks.get(name)!.resource)} too small for ${name}`);
136 const undecided = new Set(names()), decided = new Map<string, StartFinish>(), used = new Map<string, number[]>(resources().map(name => [name, []])), order: string[] = [];
137 while (undecided.size) {
138 const available = [...undecided].filter(name => predecessors(name).every(dep => decided.has(dep.a)));
139 available.sort((a, b) => original.rows.get(a)!.ls - original.rows.get(b)!.ls || compare(a, b));
140 const name = available[0], item = tasks.get(name)!, d = duration(item), days = used.get(item.resource)!;
141 let start = item.notBefore;
142 for (const dep of predecessors(name)) {
143 const old = decided.get(dep.a)!;
144 start = Math.max(start, (dep.kind === 'SS' ? old.es : old.ef) + dep.lag - (dep.kind === 'FF' ? d : 0));
145 }
146 while (true) {
147 let valid = true;
148 for (let day = start; day < start + d; day++) if ((days[day] ?? 0) + item.units > capacity(item.resource)) { valid = false; break; }
149 if (valid) break;
150 start++;
151 }
152 for (let day = start; day < start + d; day++) days[day] = (days[day] ?? 0) + item.units;
153 decided.set(name, { es: start, ef: start + d }); undecided.delete(name); order.push(name);
154 }
155 const finish = Math.max(...[...decided.values()].map(row => row.ef));
156 output.push(`level capacity ${limit}: duration ${original.total} -> ${finish}`);
157 for (const name of order) { const row = decided.get(name)!; output.push(`${name} start ${row.es} finish ${row.ef} shift ${row.es - original.rows.get(name)!.es}`); }
158 for (const resource of resources()) output.push(`resource ${resource} capacity ${capacity(resource)} peak ${Math.max(...usage(resource, original))} -> ${used.get(resource)!.reduce((peak, value) => Math.max(peak, value), 0)}`);
159}
160function execute(commandLine: string): void {
161 const [command, ...args] = commandLine.split(' '), a = args[0], b = args[1];
162 if (command === 'TASK') {
163 if (tasks.has(a)) fail(`task ${a} already exists`);
164 tasks.set(a, { id: a, base: Number(b), resource: args[2], units: Number(args[3]), notBefore: 0, used: 0, crash: null, pert: null });
165 output.push(`task ${a} added`); return;
166 }
167 if (command === 'DEP' || command === 'UNDEP') {
168 task(a); task(b);
169 const at = dependencies.findIndex(dep => dep.a === a && dep.b === b);
170 if (command === 'DEP') {
171 if (a === b) fail('self dependency ' + a);
172 if (at >= 0) fail(`dependency ${a} -> ${b} already exists`);
173 dependencies.push({ a, b, kind: args[2], lag: Number(args[3]) });
174 output.push(`dependency ${a} -> ${b} ${args[2]}${signed(Number(args[3]))} added`);
175 } else {
176 if (at < 0) fail(`no dependency ${a} -> ${b}`);
177 dependencies.splice(at, 1); output.push(`dependency ${a} -> ${b} removed`);
178 }
179 return;
180 }
181 if (['REMOVE', 'NOTBEFORE', 'CRASHCOST', 'UNCRASH', 'PERT', 'SHOW', 'IMPACT'].includes(command)) {
182 const item = task(a);
183 if (command === 'REMOVE') {
184 const before = dependencies.length;
185 dependencies = dependencies.filter(dep => dep.a !== a && dep.b !== a); tasks.delete(a);
186 output.push(`task ${a} removed with ${before - dependencies.length} dependencies`);
187 } else if (command === 'NOTBEFORE') { item.notBefore = Number(b); output.push(`notbefore ${a} ${b}`); }
188 else if (command === 'CRASHCOST') {
189 if (Number(b) >= item.base) fail('crash limit too large for ' + a);
190 if (Number(b) < item.used) fail('crash limit below used days for ' + a);
191 item.crash = { limit: Number(b), cost: Number(args[2]) }; output.push(`crash ${a} max ${b} cost ${args[2]}`);
192 } else if (command === 'UNCRASH') { output.push(`uncrash ${a} restored ${item.used} days`); item.used = 0; }
193 else if (command === 'PERT') {
194 const [o, m, p] = args.slice(1).map(Number);
195 if (o > m || m > p) fail('invalid estimate');
196 item.pert = [o, m, p]; output.push(`pert ${a} expected ${decimal(o + 4 * m + p, 6)} variance ${decimal((p - o) ** 2, 36)}`);
197 } else if (command === 'IMPACT') {
198 const seen = new Set([a]), queue = [a];
199 for (let i = 0; i < queue.length; i++) for (const dep of successors(queue[i])) if (!seen.has(dep.b)) { seen.add(dep.b); queue.push(dep.b); }
200 seen.delete(a); output.push(`impact ${a}: ${[...seen].sort().join(' ') || 'none'}`);
201 } else {
202 output.push(`task ${a} duration ${duration(item)} resource ${item.resource} units ${item.units}${item.notBefore ? ` notbefore ${item.notBefore}` : ''}`,
203 'preds: ' + (predecessors(a).map(dep => `${dep.a} ${dep.kind}${signed(dep.lag)}`).join(', ') || 'none'),
204 'succs: ' + (successors(a).map(dep => `${dep.b} ${dep.kind}${signed(dep.lag)}`).join(', ') || 'none'),
205 item.crash ? `crash: max ${item.crash.limit} used ${item.used} cost ${item.crash.cost}` : 'crash: none',
206 item.pert ? `pert: ${item.pert.join(' ')}` : 'pert: none');
207 if (cycle() !== null) output.push('schedule: unavailable');
208 else { const row = schedule().rows.get(a)!; output.push(`schedule: ES ${row.es} EF ${row.ef} LS ${row.ls} LF ${row.lf} TF ${row.tf} FF ${row.ff}${row.tf === 0 ? ' critical' : ''}`); }
209 }
210 return;
211 }
212 if (command === 'CAPACITY') { capacities.set(a, Number(b)); output.push(`capacity ${a} ${b}`); return; }
213 if (['START', 'HOLIDAY', 'DEADLINE'].includes(command)) {
214 const day = parseDate(a);
215 if (command === 'HOLIDAY' && holidays.has(day)) fail('duplicate holiday ' + a);
216 if (command === 'START') startDate = day;
217 else if (command === 'DEADLINE') deadline = day;
218 else holidays.add(day);
219 output.push(command.toLowerCase() + ' ' + dateText(day)); return;
220 }
221 if (command === 'LIST') {
222 output.push(`tasks ${tasks.size}, dependencies ${dependencies.length}, holidays ${holidays.size}`);
223 for (const name of names()) { const item = tasks.get(name)!; output.push(`${name} duration ${duration(item)} resource ${item.resource} units ${item.units} preds ${predecessors(name).map(dep => dep.a).join(' ') || 'none'}`); }
224 return;
225 }
226 if (!tasks.size) fail('no tasks');
227 const foundCycle = cycle();
228 if (foundCycle !== null) fail('cycle ' + foundCycle);
229 if (command === 'CALENDAR' && startDate === null) fail('no start date');
230 if (command === 'LOAD' && !resources().includes(a)) fail('unknown resource ' + a);
231 if (command === 'WHATIF') task(a);
232 if (command === 'VARIANCE' && baseline === null) fail('no baseline');
233 const current = schedule();
234 const byStart = names().sort((a, b) => current.rows.get(a)!.es - current.rows.get(b)!.es || compare(a, b));
235 switch (command) {
236 case 'CPM': {
237 const width = Math.max(4, ...names().map(name => name.length));
238 output.push('TASK'.padEnd(width) + ['DUR', 'ES', 'EF', 'LS', 'LF', 'TF', 'FF'].map(word => word.padStart(5)).join(''));
239 for (const name of names()) {
240 const row = current.rows.get(name)!;
241 output.push(name.padEnd(width) + [row.duration, row.es, row.ef, row.ls, row.lf, row.tf, row.ff].map(value => String(value).padStart(5)).join('') + (row.tf === 0 ? ' *' : ''));
242 }
243 output.push(`project duration ${current.total}`); break;
244 }
245 case 'CRITICAL': {
246 const paths = criticalPaths(current);
247 output.push(`critical paths: ${paths.length}`, ...paths.map(path => path.join(' -> '))); break;
248 }
249 case 'SLACK':
250 for (const value of [...new Set([...current.rows.values()].map(row => row.tf))].sort((a, b) => a - b)) output.push(`float ${value}: ${names().filter(name => current.rows.get(name)!.tf === value).join(' ')}`);
251 break;
252 case 'DRAG':
253 for (const name of critical(current)) output.push(`${name} drag ${current.total - schedule(new Map([[name, 0]])).total}`);
254 break;
255 case 'STATS': {
256 const count = critical(current).length, floats = [...current.rows.values()].reduce((sum, row) => sum + row.tf, 0);
257 output.push(`tasks ${tasks.size}, critical ${count}, dependencies ${dependencies.length} (FS ${dependencies.filter(dep => dep.kind === 'FS').length}, SS ${dependencies.filter(dep => dep.kind === 'SS').length}, FF ${dependencies.filter(dep => dep.kind === 'FF').length})`,
258 'work ' + resources().map(resource => `${resource} ${[...tasks.values()].filter(item => item.resource === resource).reduce((sum, item) => sum + duration(item) * item.units, 0)}`).join(', '),
259 `float total ${floats}, average ${decimal(floats, tasks.size)}`, `duration ${current.total}, critical share ${decimal(100 * count, tasks.size)}%`);
260 break;
261 }
262 case 'CALENDAR': {
263 const dates = dayDates(current.total), first = dates[0], last = dates[dates.length - 1];
264 for (const name of byStart) { const row = current.rows.get(name)!; output.push(`${name} ${dateText(dates[row.es])} - ${dateText(dates[row.ef - 1])}`); }
265 output.push(`project ${dateText(first)} - ${dateText(last)}, ${current.total} working days, ${last - first + 1} calendar days`);
266 if (deadline !== null) output.push(last <= deadline ? `deadline ${dateText(deadline)} met, ${workingBetween(last, deadline)} working days to spare` : `deadline ${dateText(deadline)} missed by ${workingBetween(deadline, last)} working days`);
267 const within = [...holidays].filter(day => weekday(day) < 5 && day >= first && day <= last).sort((a, b) => a - b);
268 output.push('holidays in span: ' + (within.map(dateText).join(', ') || 'none')); break;
269 }
270 case 'GANTT': {
271 if (current.total > 60) fail(`chart too wide (${current.total} days)`);
272 const width = Math.max(...names().map(name => name.length));
273 output.push(' '.repeat(width) + ' |' + Array.from({ length: current.total }, (_, i) => i % 10).join('') + '|');
274 for (const name of byStart) {
275 const row = current.rows.get(name)!, bar = Array.from({ length: current.total }, (_, day) => day >= row.es && day < row.ef ? row.tf === 0 ? '=' : '-' : day >= row.ef && day < row.lf ? '.' : ' ').join('');
276 output.push(name.padEnd(width) + ' |' + bar + '|');
277 }
278 break;
279 }
280 case 'LEVEL': level(Number(a), current); break;
281 case 'LOAD': {
282 const days = usage(a, current), peak = Math.max(...days);
283 output.push(`load ${a}: ${days.join(' ')}`, `peak ${peak} on day ${days.indexOf(peak)}`);
284 if (capacities.has(a)) output.push(`capacity ${capacities.get(a)!} exceeded on ${days.filter(value => value > capacities.get(a)!).length} days`);
285 break;
286 }
287 case 'PERTSUM':
288 for (const path of criticalPaths(current)) {
289 let expected = 0, variance = 0;
290 for (const name of path) { const item = tasks.get(name)!, d = duration(item), [o, m, p] = item.pert ?? [d, d, d]; expected += o + 4 * m + p; variance += (p - o) ** 2; }
291 output.push(`${path.join(' -> ')}: expected ${decimal(expected, 6)} variance ${decimal(variance, 36)}`);
292 }
293 break;
294 case 'CRASH': {
295 let remaining = Number(a), count = 0, total = current.total, reason = '';
296 while (true) {
297 const now = schedule();
298 const candidates = critical(now).filter(name => { const item = tasks.get(name)!; return item.crash !== null && item.used < item.crash.limit && schedule(new Map([[name, duration(item) - 1]])).total < now.total; });
299 candidates.sort((a, b) => tasks.get(a)!.crash!.cost - tasks.get(b)!.crash!.cost || compare(a, b));
300 if (!candidates.length) { reason = 'no candidate'; break; }
301 const name = candidates[0], item = tasks.get(name)!, cost = item.crash!.cost;
302 if (cost > remaining) { reason = 'insufficient budget'; break; }
303 const old = duration(item); item.used++; remaining -= cost; count++; total = schedule().total;
304 output.push(`step ${count}: ${name} ${old} -> ${duration(item)}, cost ${cost}, project ${total}`);
305 }
306 output.push(`crash done: ${count} days, cost ${Number(a) - remaining}, project ${current.total} -> ${total}, budget left ${remaining}, ${reason}`); break;
307 }
308 case 'WHATIF': {
309 const after = schedule(new Map([[a, Number(b)]]));
310 output.push(`whatif ${a} ${duration(tasks.get(a)!)} -> ${b}: project ${current.total} -> ${after.total}`);
311 for (const name of names()) {
312 const before = current.rows.get(name)!, next = after.rows.get(name)!, fields = (['es', 'ef', 'tf'] as const).filter(key => before[key] !== next[key]);
313 if (fields.length) output.push(name + fields.map(key => ` ${key.toUpperCase()} ${before[key]}->${next[key]}`).join(''));
314 }
315 const beforeCritical = critical(current).join(' '), afterCritical = critical(after).join(' ');
316 if (beforeCritical !== afterCritical) output.push(`critical ${beforeCritical} -> ${afterCritical}`);
317 break;
318 }
319 case 'BASELINE':
320 baseline = { total: current.total, rows: new Map<string, StartFinish>([...current.rows].map(([name, row]) => [name, { es: row.es, ef: row.ef }])) };
321 output.push(`baseline saved: ${tasks.size} tasks, duration ${current.total}`); break;
322 case 'VARIANCE': {
323 output.push(`variance: duration ${baseline!.total} -> ${current.total} (${signed(current.total - baseline!.total)})`);
324 const before = output.length;
325 for (const name of [...new Set([...baseline!.rows.keys(), ...tasks.keys()])].sort()) {
326 const old = baseline!.rows.get(name), row = current.rows.get(name);
327 if (row === undefined) output.push(name + ' removed');
328 else if (old === undefined) output.push(name + ' new');
329 else if (old.es !== row.es || old.ef !== row.ef) output.push(`${name} start ${signed(row.es - old.es)} finish ${signed(row.ef - old.ef)}`);
330 }
331 if (output.length === before) output.push('no task changes');
332 break;
333 }
334 }
335}
336const input = fs.readFileSync(0, 'utf8').split(/\r?\n/), count = Number(input[0]);
337for (const command of input.slice(1, count + 1)) {
338 try { execute(command); } catch (error) { output.push((error as Error).message); }
339}
340if (output.length) process.stdout.write(output.join('\n') + '\n');
