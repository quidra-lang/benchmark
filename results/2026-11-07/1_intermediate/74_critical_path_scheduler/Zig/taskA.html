18const Builder = struct {
19 gpa: Allocator,
20 bytes: std.ArrayList(u8) = .empty,
21 fn add(self: *Builder, text: Text) Allocator.Error!void {
22 try self.bytes.appendSlice(self.gpa, text);
23 }
24 fn print(self: *Builder, comptime pattern: Text, args: anytype) Allocator.Error!void {
25 try self.add(try std.fmt.allocPrint(self.gpa, pattern, args));
26 }
27 fn line(self: *Builder, comptime pattern: Text, args: anytype) Allocator.Error!void {

31fn signed(gpa: Allocator, value: Int) Allocator.Error!Text {
32 return if (value >= 0) std.fmt.allocPrint(gpa, "+{d}", .{value}) else std.fmt.allocPrint(gpa, "{d}", .{value});
33}
34fn decimal(gpa: Allocator, numerator: Int, denominator: Int) Allocator.Error!Text {
35 const value = @divTrunc(numerator * 100, denominator);
36 return std.fmt.allocPrint(gpa, "{d}.{d:0>2}", .{ @divTrunc(value, 100), @mod(value, 100) });
37}
38fn textLess(_: void, a: Text, b: Text) bool {
39 return std.mem.order(u8, a, b) == .lt;
40}
41fn keys(gpa: Allocator, map: anytype) Allocator.Error![]Text {
42 var result: std.ArrayList(Text) = .empty;
43 var iterator = map.keyIterator();
44 while (iterator.next()) |key| try result.append(gpa, key.*);
45 std.mem.sort(Text, result.items, {}, textLess);

48fn join(gpa: Allocator, parts: []const Text, separator: Text) Allocator.Error!Text {
49 var out: Builder = .{ .gpa = gpa };
50 for (parts, 0..) |part, i| {
51 if (i > 0) try out.add(separator);
52 try out.add(part);
53 }
54 return out.bytes.items;
55}
56fn leap(year: Int) bool {
57 return @mod(year, 4) == 0 and @mod(year, 100) != 0 or @mod(year, 400) == 0;
58}
59fn monthDays(year: Int, month: Int) Int {
60 const days = [_]Int{ 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31 };
61 return if (month == 2 and leap(year)) 29 else days[@intCast(month - 1)];
62}
63fn dateValue(text: Text) ?Int {
64 const year = number(text[0..4]);
65 const month = number(text[5..7]);
66 const day = number(text[8..10]);
67 if (month < 1 or month > 12 or day < 1 or day > monthDays(year, month)) return null;
68 var total = day - 1;
69 var y: Int = 2000;
70 while (y < year) : (y += 1) total += if (leap(y)) @as(Int, 366) else 365;
71 var m: Int = 1;
72 while (m < month) : (m += 1) total += monthDays(year, m);
73 return total;
74}
75fn dateText(gpa: Allocator, ordinal: Int) Allocator.Error!Text {
76 var remaining = ordinal;
77 var year: Int = 2000;
78 var month: Int = 1;
79 while (true) {
80 const count: Int = if (leap(year)) 366 else 365;
81 if (remaining < count) break;
82 remaining -= count;
83 year += 1;
84 }
85 while (remaining >= monthDays(year, month)) {
86 remaining -= monthDays(year, month);
87 month += 1;
88 }
89 const weekdays = [_]Text{ "Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun" };
90 return std.fmt.allocPrint(gpa, "{d:0>4}-{d:0>2}-{d:0>2} {s}", .{ year, month, remaining + 1, weekdays[@intCast(@mod(ordinal + 5, 7))] });

107const Dependency = struct { from: *Task, to: *Task, kind: Text, lag: Int };
108const Timing = struct { es: Int = 0, ef: Int = 0, ls: Int = 0, lf: Int = 0, tf: Int = 0, free: Int = 0 };
109const Schedule = struct { times: [30]Timing = @splat(.{}), total: Int = 0 };

112fn taskLess(_: void, a: *Task, b: *Task) bool {
113 return std.mem.order(u8, a.name, b.name) == .lt;
114}
115fn margin(dep: Dependency, schedule: Schedule) Int {
116 const a = schedule.times[dep.from.slot];
117 const b = schedule.times[dep.to.slot];
118 if (equal(dep.kind, "SS")) return b.es - a.es - dep.lag;
119 if (equal(dep.kind, "FF")) return b.ef - a.ef - dep.lag;
120 return b.es - a.ef - dep.lag;
121}

136 var result: std.ArrayList(*Task) = .empty;
137 var iterator = self.tasks.valueIterator();
138 while (iterator.next()) |task| try result.append(gpa, task.*);

142 fn relations(self: *const Project, gpa: Allocator, task: *Task, predecessors: bool) Allocator.Error![]Dependency {
143 var result: std.ArrayList(Dependency) = .empty;
144 for (self.deps.items) |dep| {
145 if ((predecessors and dep.to == task) or (!predecessors and dep.from == task)) try result.append(gpa, dep);
146 }
147 const Less = struct {
148 fn less(before: bool, a: Dependency, b: Dependency) bool {
149 return std.mem.order(u8, if (before) a.from.name else a.to.name, if (before) b.from.name else b.to.name) == .lt;
150 }
151 };
152 std.mem.sort(Dependency, result.items, predecessors, Less.less);
153 return result.items;
154 }
155 fn topology(self: *const Project, gpa: Allocator) Allocator.Error!Topology {
156 const names = try self.sorted(gpa);
157 var removed: [30]bool = @splat(false);
158 var order: std.ArrayList(*Task) = .empty;
159 while (true) {
160 var selected: ?*Task = null;
161 for (names) |task| {
162 if (removed[task.slot]) continue;
163 var ready = true;
164 for (self.deps.items) |dep| {
165 if (dep.to == task and !removed[dep.from.slot]) {
166 ready = false;
167 break;
168 }
169 }
170 if (ready) {
171 selected = task;
172 break;
173 }
174 }
175 const task = selected orelse break;
176 removed[task.slot] = true;
177 try order.append(gpa, task);
178 }
179 if (order.items.len == names.len) return .{ .order = order.items };
180 var current: *Task = undefined;
181 for (names) |task| if (!removed[task.slot]) {
182 current = task;
183 break;
184 };
185 var visited: [30]?usize = @splat(null);
186 var chain: std.ArrayList(*Task) = .empty;
187 while (true) {
188 if (visited[current.slot]) |at| {
189 var cycle: std.ArrayList(*Task) = .empty;
190 try cycle.append(gpa, current);
191 var i = chain.items.len;
192 while (i > at + 1) {
193 i -= 1;
194 try cycle.append(gpa, chain.items[i]);
195 }
196 var first: usize = 0;
197 for (cycle.items, 0..) |task, j| if (taskLess({}, task, cycle.items[first])) {
198 first = j;
199 };
200 var text: std.ArrayList(Text) = .empty;
201 for (cycle.items[first..]) |task| try text.append(gpa, task.name);
202 for (cycle.items[0..first]) |task| try text.append(gpa, task.name);
203 try text.append(gpa, text.items[0]);
204 return .{ .order = order.items, .cycle = try join(gpa, text.items, " -> ") };
205 }
206 visited[current.slot] = chain.items.len;
207 try chain.append(gpa, current);
208 var predecessor: ?*Task = null;
209 for (self.deps.items) |dep| {
210 if (dep.to == current and !removed[dep.from.slot] and
211 (predecessor == null or taskLess({}, dep.from, predecessor.?))) predecessor = dep.from;
212 }
213 current = predecessor.?;
214 }
215 }
216 fn compute(self: *const Project, order: []const *Task, changed: ?*Task, replacement: Int) Schedule {
217 var result: Schedule = .{};
218 for (order) |task| {
219 const d = if (task == changed) replacement else task.duration();
220 var earliest = task.not_before;
221 for (self.deps.items) |dep| {
222 if (dep.to != task) continue;
223 const a = result.times[dep.from.slot];
224 const value = if (equal(dep.kind, "SS")) a.es + dep.lag else if (equal(dep.kind, "FF")) a.ef + dep.lag - d else a.ef + dep.lag;
225 earliest = @max(earliest, value);
226 }
227 result.times[task.slot] = .{ .es = earliest, .ef = earliest + d };
228 result.total = @max(result.total, earliest + d);
229 }
230 var i = order.len;
231 while (i > 0) {
232 i -= 1;
233 const task = order[i];
234 const d = if (task == changed) replacement else task.duration();
235 const value = &result.times[task.slot];
236 var finish = result.total;
237 for (self.deps.items) |dep| {
238 if (dep.from != task) continue;
239 const b = result.times[dep.to.slot];
240 const candidate = if (equal(dep.kind, "SS")) b.ls - dep.lag + d else if (equal(dep.kind, "FF")) b.lf - dep.lag else b.ls - dep.lag;
241 finish = @min(finish, candidate);
242 }
243 value.lf = finish;
244 value.ls = finish - d;
245 value.tf = value.ls - value.es;
246 value.free = result.total - value.ef;
247 }
248 for (order) |task| {
249 for (self.deps.items) |dep| {
250 if (dep.from == task) result.times[task.slot].free = @min(result.times[task.slot].free, margin(dep, result));
251 }
252 }
253 return result;
254 }
255 fn critical(self: *const Project, gpa: Allocator, schedule: Schedule) Allocator.Error![]*Task {
256 var result: std.ArrayList(*Task) = .empty;
257 for (try self.sorted(gpa)) |task| if (schedule.times[task.slot].tf == 0) {
258 try result.append(gpa, task);
259 };
260 return result.items;
261 }
262 fn visitPath(self: *const Project, gpa: Allocator, schedule: Schedule, task: *Task, prefix: []const *Task, result: *std.ArrayList([]*Task)) Allocator.Error!void {
263 var path: std.ArrayList(*Task) = .empty;
264 try path.appendSlice(gpa, prefix);
265 try path.append(gpa, task);
266 var children: std.ArrayList(*Task) = .empty;
267 for (self.deps.items) |dep| {
268 if (dep.from == task and schedule.times[dep.to.slot].tf == 0 and margin(dep, schedule) == 0) try children.append(gpa, dep.to);
269 }
270 std.mem.sort(*Task, children.items, {}, taskLess);
271 if (children.items.len == 0) try result.append(gpa, path.items) else {
272 for (children.items) |child| try self.visitPath(gpa, schedule, child, path.items, result);
273 }
274 }
275 fn criticalPaths(self: *const Project, gpa: Allocator, schedule: Schedule) Allocator.Error![][]*Task {
276 var result: std.ArrayList([]*Task) = .empty;
277 for (try self.critical(gpa, schedule)) |task| {
278 var incoming = false;
279 for (self.deps.items) |dep| {
280 if (dep.to == task and schedule.times[dep.from.slot].tf == 0 and margin(dep, schedule) == 0) {
281 incoming = true;
282 break;
283 }
284 }
285 if (!incoming) try self.visitPath(gpa, schedule, task, &.{}, &result);
286 }
287 return result.items;
288 }

295 fn usage(self: *const Project, gpa: Allocator, schedule: Schedule, resource: Text) Allocator.Error![]Int {
296 const result = try gpa.alloc(Int, @intCast(schedule.total));
297 @memset(result, 0);
298 var iterator = self.tasks.valueIterator();
299 while (iterator.next()) |task| {
300 if (equal(task.*.resource, resource)) {
301 const value = schedule.times[task.*.slot];
302 for (@as(usize, @intCast(value.es))..@as(usize, @intCast(value.ef))) |day| result[day] += task.*.units;
303 }
304 }
305 return result;
306 }
307 fn workingDay(self: *const Project, day: Int) bool {
308 return @mod(day + 5, 7) < 5 and !self.holidays.contains(day);
309 }
310 fn startOrder(self: *const Project, gpa: Allocator, schedule: Schedule) Allocator.Error![]*Task {
311 const result = try self.sorted(gpa);
312 const Less = struct {
313 fn less(s: Schedule, a: *Task, b: *Task) bool {
314 const x = s.times[a.slot].es;
315 const y = s.times[b.slot].es;
316 return if (x != y) x < y else taskLess({}, a, b);
317 }
318 };
319 std.mem.sort(*Task, result, schedule, Less.less);
320 return result;
321 }
322 fn calendar(self: *Project, scratch: Allocator, schedule: Schedule) Allocator.Error!void {
323 var days: std.ArrayList(Int) = .empty;
324 var day = self.start.?;
325 while (days.items.len < @as(usize, @intCast(schedule.total))) : (day += 1) {
326 if (self.workingDay(day)) try days.append(scratch, day);
327 }
328 for (try self.startOrder(scratch, schedule)) |task| {
329 const value = schedule.times[task.slot];
330 try self.out.line("{s} {s} - {s}", .{ task.name, try dateText(scratch, days.items[@intCast(value.es)]), try dateText(scratch, days.items[@intCast(value.ef - 1)]) });
331 }
332 const first = days.items[0];
333 const last = days.items[days.items.len - 1];
334 try self.out.line("project {s} - {s}, {d} working days, {d} calendar days", .{ try dateText(scratch, first), try dateText(scratch, last), schedule.total, last - first + 1 });
335 if (self.deadline) |date| {
336 var count: usize = 0;
337 day = @min(date, last) + 1;
338 while (day <= @max(date, last)) : (day += 1) if (self.workingDay(day)) {
339 count += 1;
340 };
341 if (last <= date) try self.out.line("deadline {s} met, {d} working days to spare", .{ try dateText(scratch, date), count }) else try self.out.line("deadline {s} missed by {d} working days", .{ try dateText(scratch, date), count });
342 }
343 var selected: std.ArrayList(Int) = .empty;
344 var iterator = self.holidays.keyIterator();
345 while (iterator.next()) |date| {
346 if (date.* >= first and date.* <= last and @mod(date.* + 5, 7) < 5) try selected.append(scratch, date.*);
347 }
348 const Less = struct {
349 fn less(_: void, a: Int, b: Int) bool {
350 return a < b;
351 }
352 };
353 std.mem.sort(Int, selected.items, {}, Less.less);
354 try self.out.add("holidays in span: ");
355 if (selected.items.len == 0) try self.out.add("none");
356 for (selected.items, 0..) |date, i| {
357 if (i > 0) try self.out.add(", ");
358 try self.out.add(try dateText(scratch, date));
359 }
360 try self.out.add("\n");
361 }

365 fn level(self: *Project, scratch: Allocator, order: []const *Task, schedule: Schedule, limit: Int) Error!void {
366 const names = try self.sorted(scratch);
367 for (names) |task| {
368 const cap = self.capacity(task.resource, limit);
369 if (task.units > cap) return bad(self.gpa, "capacity {d} too small for {s}", .{ cap, task.name });
370 }
371 var decided: [30]?Timing = @splat(null);
372 var loads: std.StringHashMapUnmanaged(std.ArrayList(Int)) = .empty;
373 var sequence: std.ArrayList(*Task) = .empty;
374 var total: Int = 0;
375 while (sequence.items.len < order.len) {
376 var selected: ?*Task = null;
377 for (names) |task| {
378 if (decided[task.slot] != null) continue;
379 var ready = true;
380 for (self.deps.items) |dep| if (dep.to == task and decided[dep.from.slot] == null) {
381 ready = false;
382 break;
383 };
384 if (ready and (selected == null or schedule.times[task.slot].ls < schedule.times[selected.?.slot].ls)) selected = task;
385 }
386 const task = selected.?;
387 const d = task.duration();
388 var earliest = task.not_before;
389 for (self.deps.items) |dep| {
390 if (dep.to != task) continue;
391 const a = decided[dep.from.slot].?;
392 const value = if (equal(dep.kind, "SS")) a.es + dep.lag else if (equal(dep.kind, "FF")) a.ef + dep.lag - d else a.ef + dep.lag;
393 earliest = @max(earliest, value);
394 }
395 var load = loads.get(task.resource) orelse std.ArrayList(Int).empty;
396 while (true) {
397 var valid = true;
398 var day = earliest;
399 while (day < earliest + d) : (day += 1) {
400 const used = if (@as(usize, @intCast(day)) < load.items.len) load.items[@intCast(day)] else 0;
401 if (used + task.units > self.capacity(task.resource, limit)) {
402 valid = false;
403 break;
404 }
405 }
406 if (valid) break;
407 earliest += 1;
408 }
409 while (load.items.len < @as(usize, @intCast(earliest + d))) try load.append(scratch, 0);
410 for (@as(usize, @intCast(earliest))..@as(usize, @intCast(earliest + d))) |day| load.items[day] += task.units;
411 try loads.put(scratch, task.resource, load);
412 decided[task.slot] = .{ .es = earliest, .ef = earliest + d };
413 try sequence.append(scratch, task);
414 total = @max(total, earliest + d);
415 }
416 try self.out.line("level capacity {d}: duration {d} -> {d}", .{ limit, schedule.total, total });
417 for (sequence.items) |task| {
418 const value = decided[task.slot].?;
419 try self.out.line("{s} start {d} finish {d} shift {d}", .{ task.name, value.es, value.ef, value.es - schedule.times[task.slot].es });
420 }
421 for (try self.resources(scratch)) |resource| try self.out.line("resource {s} capacity {d} peak {d} -> {d}", .{ resource, self.capacity(resource, limit), peak(try self.usage(scratch, schedule, resource)), peak(loads.get(resource).?.items) });
422 }
423 fn command(self: *Project, scratch: Allocator, w: []const Text) Error!void {
424 const cmd = w[0];
425 if (equal(cmd, "TASK")) {
426 if (self.tasks.contains(w[1])) return bad(self.gpa, "task {s} already exists", .{w[1]});
427 var occupied: [30]bool = @splat(false);
428 var iterator = self.tasks.valueIterator();
429 while (iterator.next()) |task| occupied[task.*.slot] = true;
430 const slot = std.mem.findScalar(bool, &occupied, false).?;
431 const task = try self.gpa.create(Task);
432 task.* = .{ .name = w[1], .slot = slot, .base = number(w[2]), .resource = w[3], .units = number(w[4]) };
433 try self.tasks.put(self.gpa, task.name, task);
434 try self.out.line("task {s} added", .{task.name});
435 return;
436 } else if (equal(cmd, "DEP") or equal(cmd, "UNDEP")) {
437 const a = try self.get(w[1]);
438 const b = try self.get(w[2]);
439 var existing: ?usize = null;
440 for (self.deps.items, 0..) |dep, i| if (dep.from == a and dep.to == b) {
441 existing = i;
442 break;
443 };
444 if (equal(cmd, "DEP")) {
445 if (a == b) return bad(self.gpa, "self dependency {s}", .{a.name});
446 if (existing != null) return bad(self.gpa, "dependency {s} -> {s} already exists", .{ a.name, b.name });
447 try self.deps.append(self.gpa, .{ .from = a, .to = b, .kind = w[3], .lag = number(w[4]) });
448 try self.out.line("dependency {s} -> {s} {s}{s} added", .{ a.name, b.name, w[3], try signed(scratch, number(w[4])) });
449 } else {
450 const at = existing orelse return bad(self.gpa, "no dependency {s} -> {s}", .{ a.name, b.name });
451 _ = self.deps.orderedRemove(at);
452 try self.out.line("dependency {s} -> {s} removed", .{ a.name, b.name });
453 }
454 return;
455 } else if (equal(cmd, "REMOVE") or equal(cmd, "NOTBEFORE") or equal(cmd, "CRASHCOST") or equal(cmd, "UNCRASH") or
456 equal(cmd, "PERT") or equal(cmd, "SHOW") or equal(cmd, "IMPACT"))
457 {
458 const task = try self.get(w[1]);
459 if (equal(cmd, "REMOVE")) {
460 var count: usize = 0;
461 var i: usize = 0;
462 while (i < self.deps.items.len) {
463 const dep = self.deps.items[i];
464 if (dep.from == task or dep.to == task) {
465 _ = self.deps.orderedRemove(i);
466 count += 1;
467 } else i += 1;
468 }
469 _ = self.tasks.remove(task.name);
470 try self.out.line("task {s} removed with {d} dependencies", .{ task.name, count });
471 } else if (equal(cmd, "NOTBEFORE")) {
472 task.not_before = number(w[2]);
473 try self.out.line("notbefore {s} {s}", .{ task.name, w[2] });
474 } else if (equal(cmd, "CRASHCOST")) {
475 const limit = number(w[2]);
476 if (limit >= task.base) return bad(self.gpa, "crash limit too large for {s}", .{task.name});
477 if (limit < task.crashed) return bad(self.gpa, "crash limit below used days for {s}", .{task.name});
478 task.crash = .{ .limit = limit, .cost = number(w[3]) };
479 try self.out.line("crash {s} max {d} cost {d}", .{ task.name, limit, task.crash.?.cost });
480 } else if (equal(cmd, "UNCRASH")) {
481 try self.out.line("uncrash {s} restored {d} days", .{ task.name, task.crashed });
482 task.crashed = 0;
483 } else if (equal(cmd, "PERT")) {
484 const o = number(w[2]);
485 const m = number(w[3]);
486 const p = number(w[4]);
487 if (o > m or m > p) return bad(self.gpa, "invalid estimate", .{});
488 task.estimate = .{ o, m, p };
489 try self.out.line("pert {s} expected {s} variance {s}", .{ task.name, try decimal(scratch, o + 4 * m + p, 6), try decimal(scratch, (p - o) * (p - o), 36) });
490 } else if (equal(cmd, "IMPACT")) {
491 var reached: [30]bool = @splat(false);
492 reached[task.slot] = true;
493 var queue: std.ArrayList(*Task) = .empty;
494 try queue.append(scratch, task);
495 var at: usize = 0;
496 while (at < queue.items.len) : (at += 1) {
497 for (self.deps.items) |dep| {
498 if (dep.from == queue.items[at] and !reached[dep.to.slot]) {
499 reached[dep.to.slot] = true;
500 try queue.append(scratch, dep.to);
501 }
502 }
503 }
504 reached[task.slot] = false;
505 try self.out.print("impact {s}:", .{task.name});
506 var count: usize = 0;
507 for (try self.sorted(scratch)) |other| if (reached[other.slot]) {
508 try self.out.print(" {s}", .{other.name});
509 count += 1;
510 };
511 if (count == 0) try self.out.add(" none");
512 try self.out.add("\n");
513 } else {
514 try self.out.print("task {s} duration {d} resource {s} units {d}", .{ task.name, task.duration(), task.resource, task.units });
515 if (task.not_before != 0) try self.out.print(" notbefore {d}", .{task.not_before});
516 try self.out.add("\n");
517 for ([_]bool{ true, false }) |before| {
518 const deps = try self.relations(scratch, task, before);
519 try self.out.print("{s}: ", .{if (before) @as(Text, "preds") else "succs"});
520 if (deps.len == 0) try self.out.add("none");
521 for (deps, 0..) |dep, i| {
522 if (i > 0) try self.out.add(", ");
523 try self.out.print("{s} {s}{s}", .{ if (before) dep.from.name else dep.to.name, dep.kind, try signed(scratch, dep.lag) });
524 }
525 try self.out.add("\n");
526 }
527 if (task.crash) |data| try self.out.line("crash: max {d} used {d} cost {d}", .{ data.limit, task.crashed, data.cost }) else try self.out.add("crash: none\n");
528 if (task.estimate) |values| try self.out.line("pert: {d} {d} {d}", .{ values[0], values[1], values[2] }) else try self.out.add("pert: none\n");
529 const topo = try self.topology(scratch);
530 if (topo.cycle != null) try self.out.add("schedule: unavailable\n") else {
531 const value = self.compute(topo.order, null, 0).times[task.slot];
532 try self.out.line("schedule: ES {d} EF {d} LS {d} LF {d} TF {d} FF {d}{s}", .{ value.es, value.ef, value.ls, value.lf, value.tf, value.free, if (value.tf == 0) @as(Text, " critical") else "" });
533 }
534 }
535 return;
536 } else if (equal(cmd, "CAPACITY")) {
537 try self.capacities.put(self.gpa, w[1], number(w[2]));
538 try self.out.line("capacity {s} {s}", .{ w[1], w[2] });
539 return;
540 } else if (equal(cmd, "START") or equal(cmd, "HOLIDAY") or equal(cmd, "DEADLINE")) {
541 const date = dateValue(w[1]) orelse return bad(self.gpa, "invalid date {s}", .{w[1]});
542 if (equal(cmd, "HOLIDAY")) {
543 if (self.holidays.contains(date)) return bad(self.gpa, "duplicate holiday {s}", .{w[1]});
544 try self.holidays.put(self.gpa, date, {});
545 } else if (equal(cmd, "START")) self.start = date else self.deadline = date;
546 const verb: Text = if (equal(cmd, "START")) "start" else if (equal(cmd, "HOLIDAY")) "holiday" else "deadline";
547 try self.out.line("{s} {s}", .{ verb, try dateText(scratch, date) });
548 return;
549 } else if (equal(cmd, "LIST")) {
550 try self.out.line("tasks {d}, dependencies {d}, holidays {d}", .{ self.tasks.count(), self.deps.items.len, self.holidays.count() });
551 for (try self.sorted(scratch)) |task| {
552 try self.out.print("{s} duration {d} resource {s} units {d} preds", .{ task.name, task.duration(), task.resource, task.units });
553 const previous = try self.relations(scratch, task, true);
554 if (previous.len == 0) try self.out.add(" none");
555 for (previous) |dep| try self.out.print(" {s}", .{dep.from.name});
556 try self.out.add("\n");
557 }
558 return;
559 }
560 if (self.tasks.count() == 0) return bad(self.gpa, "no tasks", .{});
561 const topo = try self.topology(scratch);
562 if (topo.cycle) |cycle| return bad(self.gpa, "cycle {s}", .{cycle});
563 if (equal(cmd, "CALENDAR") and self.start == null) return bad(self.gpa, "no start date", .{});
564 if (equal(cmd, "WHATIF")) _ = try self.get(w[1]);
565 if (equal(cmd, "VARIANCE") and self.baseline == null) return bad(self.gpa, "no baseline", .{});
566 if (equal(cmd, "LOAD")) {
567 var found = false;
568 for (try self.sorted(scratch)) |task| if (equal(task.resource, w[1])) {
569 found = true;
570 break;
571 };
572 if (!found) return bad(self.gpa, "unknown resource {s}", .{w[1]});
573 }
574 var schedule = self.compute(topo.order, null, 0);
575 const names = try self.sorted(scratch);
576 if (equal(cmd, "CPM")) {
577 var width: usize = 4;
578 for (names) |task| width = @max(width, task.name.len);
579 try padded(self.out, scratch, "TASK", width, false);
580 for ([_]Text{ "DUR", "ES", "EF", "LS", "LF", "TF", "FF" }) |column| try padded(self.out, scratch, column, 5, true);
581 try self.out.add("\n");
582 for (names) |task| {
583 const value = schedule.times[task.slot];
584 try padded(self.out, scratch, task.name, width, false);
585 for ([_]Int{ task.duration(), value.es, value.ef, value.ls, value.lf, value.tf, value.free }) |amount| {
586 try padded(self.out, scratch, try std.fmt.allocPrint(scratch, "{d}", .{amount}), 5, true);
587 }
588 if (value.tf == 0) try self.out.add(" *");
589 try self.out.add("\n");
590 }
591 try self.out.line("project duration {d}", .{schedule.total});
592 } else if (equal(cmd, "CRITICAL") or equal(cmd, "PERTSUM")) {
593 const paths = try self.criticalPaths(scratch, schedule);
594 if (equal(cmd, "CRITICAL")) try self.out.line("critical paths: {d}", .{paths.len});
595 for (paths) |path| {
596 const text = try taskNames(scratch, path, " -> ");
597 if (equal(cmd, "CRITICAL")) try self.out.line("{s}", .{text}) else {
598 var expected: Int = 0;
599 var variance: Int = 0;
600 for (path) |task| {
601 const d = task.duration();
602 const values = task.estimate orelse [3]Int{ d, d, d };
603 expected += values[0] + 4 * values[1] + values[2];
604 variance += (values[2] - values[0]) * (values[2] - values[0]);
605 }
606 try self.out.line("{s}: expected {s} variance {s}", .{ text, try decimal(scratch, expected, 6), try decimal(scratch, variance, 36) });
607 }
608 }
609 } else if (equal(cmd, "SLACK")) {
610 const values = try scratch.alloc(Int, names.len);
611 for (names, 0..) |task, i| values[i] = schedule.times[task.slot].tf;
612 const Less = struct {
613 fn less(_: void, a: Int, b: Int) bool {
614 return a < b;
615 }
616 };
617 std.mem.sort(Int, values, {}, Less.less);
618 for (values, 0..) |value, i| {
619 if (i > 0 and value == values[i - 1]) continue;
620 try self.out.print("float {d}:", .{value});
621 for (names) |task| if (schedule.times[task.slot].tf == value) {
622 try self.out.print(" {s}", .{task.name});
623 };
624 try self.out.add("\n");
625 }
626 } else if (equal(cmd, "DRAG")) {
627 for (try self.critical(scratch, schedule)) |task| {
628 const after = self.compute(topo.order, task, 0);
629 try self.out.line("{s} drag {d}", .{ task.name, schedule.total - after.total });
630 }
631 } else if (equal(cmd, "STATS")) {
632 var counts = [_]usize{ 0, 0, 0 };
633 for (self.deps.items) |dep| counts[if (equal(dep.kind, "FS")) @as(usize, 0) else if (equal(dep.kind, "SS")) 1 else 2] += 1;
634 const count = (try self.critical(scratch, schedule)).len;
635 try self.out.line("tasks {d}, critical {d}, dependencies {d} (FS {d}, SS {d}, FF {d})", .{ names.len, count, self.deps.items.len, counts[0], counts[1], counts[2] });
636 try self.out.add("work ");
637 for (try self.resources(scratch), 0..) |resource, i| {
638 var work: Int = 0;
639 for (names) |task| if (equal(task.resource, resource)) {
640 work += task.duration() * task.units;
641 };
642 if (i > 0) try self.out.add(", ");
643 try self.out.print("{s} {d}", .{ resource, work });
644 }
645 var total: Int = 0;
646 for (names) |task| total += schedule.times[task.slot].tf;
647 try self.out.line("\nfloat total {d}, average {s}", .{ total, try decimal(scratch, total, @intCast(names.len)) });
648 try self.out.line("duration {d}, critical share {s}%", .{ schedule.total, try decimal(scratch, @as(Int, @intCast(count)) * 100, @intCast(names.len)) });
649 } else if (equal(cmd, "CALENDAR")) {
650 try self.calendar(scratch, schedule);
651 } else if (equal(cmd, "GANTT")) {
652 if (schedule.total > 60) return bad(self.gpa, "chart too wide ({d} days)", .{schedule.total});
653 var width: usize = 0;
654 for (names) |task| width = @max(width, task.name.len);
655 for (0..width) |_| try self.out.add(" ");
656 try self.out.add(" |");
657 for (0..@as(usize, @intCast(schedule.total))) |day| try self.out.print("{d}", .{day % 10});
658 try self.out.add("|\n");
659 for (try self.startOrder(scratch, schedule)) |task| {
660 const value = schedule.times[task.slot];
661 try padded(self.out, scratch, task.name, width, false);
662 try self.out.add(" |");
663 var day: Int = 0;
664 while (day < schedule.total) : (day += 1) {
665 const symbol: Text = if (value.es <= day and day < value.ef) (if (value.tf == 0) "=" else "-") else if (value.ef <= day and day < value.lf) "." else " ";
666 try self.out.add(symbol);
667 }
668 try self.out.add("|\n");
669 }
670 } else if (equal(cmd, "LEVEL")) {
671 try self.level(scratch, topo.order, schedule, number(w[1]));
672 } else if (equal(cmd, "LOAD")) {
673 const values = try self.usage(scratch, schedule, w[1]);
674 const highest = peak(values);
675 const first = std.mem.findScalar(Int, values, highest).?;
676 try self.out.print("load {s}:", .{w[1]});
677 for (values) |value| try self.out.print(" {d}", .{value});
678 try self.out.line("\npeak {d} on day {d}", .{ highest, first });
679 if (self.capacities.get(w[1])) |cap| {
680 var count: usize = 0;
681 for (values) |value| if (value > cap) {
682 count += 1;
683 };
684 try self.out.line("capacity {d} exceeded on {d} days", .{ cap, count });
685 }
686 } else if (equal(cmd, "CRASH")) {
687 const initial = schedule.total;
688 var remaining = number(w[1]);
689 var steps: usize = 0;
690 var spent: Int = 0;
691 var reason: Text = "";
692 while (true) {
693 schedule = self.compute(topo.order, null, 0);
694 var chosen: ?*Task = null;
695 for (names) |task| {
696 if (schedule.times[task.slot].tf != 0) continue;
697 if (task.crash) |data| {
698 if (task.crashed < data.limit and self.compute(topo.order, task, task.duration() - 1).total < schedule.total and
699 (chosen == null or data.cost < chosen.?.crash.?.cost)) chosen = task;
700 }
701 }
702 const task = chosen orelse {
703 reason = "no candidate";
704 break;
705 };
706 const cost = task.crash.?.cost;
707 if (cost > remaining) {
708 reason = "insufficient budget";
709 break;
710 }
711 const before = task.duration();
712 task.crashed += 1;
713 remaining -= cost;
714 spent += cost;
715 steps += 1;
716 try self.out.line("step {d}: {s} {d} -> {d}, cost {d}, project {d}", .{ steps, task.name, before, before - 1, cost, self.compute(topo.order, null, 0).total });
717 }
718 try self.out.line("crash done: {d} days, cost {d}, project {d} -> {d}, budget left {d}, {s}", .{ steps, spent, initial, schedule.total, remaining, reason });
719 } else if (equal(cmd, "WHATIF")) {
720 const changed = try self.get(w[1]);
721 const after = self.compute(topo.order, changed, number(w[2]));
722 try self.out.line("whatif {s} {d} -> {s}: project {d} -> {d}", .{ changed.name, changed.duration(), w[2], schedule.total, after.total });
723 for (names) |task| {
724 const a = schedule.times[task.slot];
725 const b = after.times[task.slot];
726 if (a.es == b.es and a.ef == b.ef and a.tf == b.tf) continue;
727 try self.out.add(task.name);
728 if (a.es != b.es) try self.out.print(" ES {d}->{d}", .{ a.es, b.es });
729 if (a.ef != b.ef) try self.out.print(" EF {d}->{d}", .{ a.ef, b.ef });
730 if (a.tf != b.tf) try self.out.print(" TF {d}->{d}", .{ a.tf, b.tf });
731 try self.out.add("\n");
732 }
733 const before_names = try taskNames(scratch, try self.critical(scratch, schedule), " ");
734 const after_names = try taskNames(scratch, try self.critical(scratch, after), " ");
735 if (!equal(before_names, after_names)) try self.out.line("critical {s} -> {s}", .{ before_names, after_names });
736 } else if (equal(cmd, "BASELINE")) {
737 var saved: Baseline = .{ .total = schedule.total };
738 for (names) |task| {
739 const value = schedule.times[task.slot];
740 try saved.times.put(self.gpa, task.name, .{ value.es, value.ef });
741 }
742 self.baseline = saved;
743 try self.out.line("baseline saved: {d} tasks, duration {d}", .{ names.len, schedule.total });
744 } else if (equal(cmd, "VARIANCE")) {
745 const old = self.baseline.?;
746 try self.out.line("variance: duration {d} -> {d} ({s})", .{ old.total, schedule.total, try signed(scratch, schedule.total - old.total) });
747 var all: std.StringHashMapUnmanaged(void) = .empty;
748 for (try keys(scratch, old.times)) |name| try all.put(scratch, name, {});
749 for (names) |task| try all.put(scratch, task.name, {});
750 var count: usize = 0;
751 for (try keys(scratch, all)) |name| {
752 const before = old.times.get(name);
753 if (self.tasks.get(name)) |task| {
754 const after = schedule.times[task.slot];
755 if (before) |value| {
756 if (value[0] == after.es and value[1] == after.ef) continue;
757 try self.out.line("{s} start {s} finish {s}", .{ name, try signed(scratch, after.es - value[0]), try signed(scratch, after.ef - value[1]) });
758 } else try self.out.line("{s} new", .{name});
759 } else try self.out.line("{s} removed", .{name});
760 count += 1;
761 }
762 if (count == 0) try self.out.add("no task changes\n");
763 }
764 }
765};
766
767fn peak(values: []const Int) Int {
768 var result: Int = 0;
769 for (values) |value| result = @max(result, value);
770 return result;
771}
772fn padded(out: *Builder, scratch: Allocator, text: Text, width: usize, left: bool) Allocator.Error!void {
773 _ = scratch;
774 const count = width - @min(width, text.len);
775 if (left) for (0..count) |_| {
776 try out.add(" ");
777 };
778 try out.add(text);
779 if (!left) for (0..count) |_| {
780 try out.add(" ");
781 };
782}
783fn taskNames(gpa: Allocator, tasks: []const *Task, separator: Text) Allocator.Error!Text {
784 var parts: std.ArrayList(Text) = .empty;
785 for (tasks) |task| try parts.append(gpa, task.name);
786 return join(gpa, parts.items, separator);
787}
788pub fn main(init: std.process.Init) !void {
789 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
790 defer arena.deinit();
791 const gpa = arena.allocator();
792 var input_buffer: [8192]u8 = undefined;
793 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
794 const text = try input.interface.allocRemaining(gpa, .unlimited);
795 var lines = std.mem.splitScalar(u8, text, '\n');
796 const count: usize = @intCast(number(lines.next().?));
797 var out: Builder = .{ .gpa = gpa };
798 var project: Project = .{ .gpa = gpa, .out = &out };
799 for (0..count) |_| {
800 var scratch_arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
801 defer scratch_arena.deinit();
802 const scratch = scratch_arena.allocator();
803 var words: std.ArrayList(Text) = .empty;
804 var iterator = std.mem.splitScalar(u8, lines.next().?, ' ');
805 while (iterator.next()) |word| try words.append(scratch, word);
806 project.command(scratch, words.items) catch |err| switch (err) {
807 error.Invalid => try out.line("error: {s}", .{failure}),
808 else => return err,
809 };
810 }
811 var output_buffer: [8192]u8 = undefined;
812 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
813 try output.interface.writeAll(out.bytes.items);
814 try output.interface.flush();
815}
