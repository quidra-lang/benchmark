116static int topological(int *order, bool *remaining) {
117 for (int i = 0; i < task_count; i++) remaining[i] = true;
118 int count = 0;
119 for (;;) {
120 int next = -1;
121 for (int v = 0; v < task_count; v++) if (remaining[v]) {
122 bool ready = true;
123 for (int i = 0; i < dep_count; i++) if (dependencies[i].b == tasks[v] && remaining[index_of(dependencies[i].a)]) {
124 ready = false;
125 break;
126 }
127 if (ready) {
128 next = v;
129 break;
130 }
131 }
132 if (next < 0) break;
133 order[count++] = next;
134 remaining[next] = false;
135 }
136 return count;
137}

139static bool cycle(char *text) {
140 int order[MAX_TASKS];
141 bool remaining[MAX_TASKS];
142 if (topological(order, remaining) == task_count) return false;
143 int current = 0;
144 while (!remaining[current]) current++;
145 int chain[MAX_TASKS], count = 0, repeated;
146 for (;;) {
147 repeated = -1;
148 for (int i = 0; i < count; i++) if (chain[i] == current) {
149 repeated = i;
150 break;
151 }
152 if (repeated >= 0) break;
153 chain[count++] = current;
154 for (int u = 0; u < task_count; u++) if (remaining[u] && dependency(tasks[u], tasks[current]) >= 0) {
155 current = u;
156 break;
157 }
158 }
159 int vertices[MAX_TASKS], length = 0, first = 0;
160 vertices[length++] = current;
161 for (int i = count - 1; i > repeated; i--) vertices[length++] = chain[i];
162 for (int i = 1; i < length; i++) if (vertices[i] < vertices[first]) first = i;
163 text[0] = '\0';
164 for (int i = 0; i <= length; i++) {
165 if (i) strcat(text, " -> ");
166 strcat(text, tasks[vertices[(first + i) % length]]->name);
167 }
168 return true;
169}

171static int margin(Schedule *s, Dependency *d) {
172 Timing *a = &s->times[index_of(d->a)], *b = &s->times[index_of(d->b)];
173 return (d->kind == 2 ? b->ef : b->es) - (d->kind == 1 ? a->es : a->ef) - d->lag;
174}

176static Schedule compute(Task *changed, int changed_duration) {
177 Schedule s = {0};
178 int order[MAX_TASKS];
179 bool remaining[MAX_TASKS];
180 topological(order, remaining);
181 for (int k = 0; k < task_count; k++) {
182 int v = order[k];
183 Task *t = tasks[v];
184 Timing *x = &s.times[v];
185 x->duration = t == changed ? changed_duration : duration(t);
186 x->es = t->not_before;
187 for (int i = 0; i < dep_count; i++) if (dependencies[i].b == t) {
188 Dependency *d = &dependencies[i];
189 Timing *a = &s.times[index_of(d->a)];
190 int bound = (d->kind == 1 ? a->es : a->ef) + d->lag - (d->kind == 2 ? x->duration : 0);
191 x->es = maximum(x->es, bound);
192 }
193 x->ef = x->es + x->duration;
194 s.total = maximum(s.total, x->ef);
195 }
196 for (int k = task_count - 1; k >= 0; k--) {
197 int v = order[k];
198 Timing *x = &s.times[v];
199 x->lf = s.total;
200 for (int i = 0; i < dep_count; i++) if (dependencies[i].a == tasks[v]) {
201 Dependency *d = &dependencies[i];
202 Timing *b = &s.times[index_of(d->b)];
203 int bound = (d->kind == 2 ? b->lf : b->ls) - d->lag + (d->kind == 1 ? x->duration : 0);
204 x->lf = minimum(x->lf, bound);
205 }
206 x->ls = x->lf - x->duration;
207 x->tf = x->ls - x->es;
208 x->free = s.total - x->ef;
209 for (int i = 0; i < dep_count; i++) if (dependencies[i].a == tasks[v]) x->free = minimum(x->free, margin(&s, &dependencies[i]));
210 }
211 return s;
212}

214static bool critical_dependency(Schedule *s, int a, int b) {
215 int i = dependency(tasks[a], tasks[b]);
216 return i >= 0 && s->times[a].tf == 0 && s->times[b].tf == 0 && margin(s, &dependencies[i]) == 0;

219static void path_visit(Schedule *s, Paths *paths, int *path, int length, int v) {
220 path[length++] = v;
221 bool leaf = true;
222 for (int w = 0; w < task_count; w++) if (critical_dependency(s, v, w)) {
223 leaf = false;
224 path_visit(s, paths, path, length, w);
225 }
226 if (leaf) {
227 paths->lengths[paths->count] = length;
228 memcpy(paths->vertices[paths->count++], path, (size_t)length * sizeof *path);
229 }
230}

232static Paths critical_paths(Schedule *s) {
233 Paths result = {0};
234 int path[MAX_TASKS];
235 for (int v = 0; v < task_count; v++) if (s->times[v].tf == 0) {
236 bool root = true;
237 for (int u = 0; u < task_count; u++) if (critical_dependency(s, u, v)) root = false;
238 if (root) path_visit(s, &result, path, 0, v);
239 }
240 return result;
241}

262static int month_days(int year, int month) {
263 const int lengths[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
264 return lengths[month - 1] + (month == 2 && leap(year));

267static int parse_date(const char *text) {
268 int year, month, day;
269 sscanf(text, "%d-%d-%d", &year, &month, &day);
270 if (month < 1 || month > 12 || day < 1 || day > month_days(year, month)) fail("invalid date %s", text);
271 int serial = day - 1;
272 for (int y = 2000; y < year; y++) serial += leap(y) ? 366 : 365;
273 for (int m = 1; m < month; m++) serial += month_days(year, m);
274 return serial;
275}

277static void print_date(int serial) {
278 const char *weekdays[] = {"Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"};
279 int year = 2000, month = 1, day = serial;
280 while (day >= (leap(year) ? 366 : 365)) {
281 day -= leap(year) ? 366 : 365;
282 year++;
283 }
284 while (day >= month_days(year, month)) {
285 day -= month_days(year, month);
286 month++;
287 }
288 printf("%04d-%02d-%02d %s", year, month, day + 1, weekdays[(serial + 5) % 7]);
289}

340static void start_order(Schedule *s, int *order) {
341 for (int i = 0; i < task_count; i++) order[i] = i;
342 for (int i = 1; i < task_count; i++) {
343 int v = order[i], j = i;
344 while (j && (s->times[order[j - 1]].es > s->times[v].es || (s->times[order[j - 1]].es == s->times[v].es && order[j - 1] > v))) {
345 order[j] = order[j - 1];
346 j--;
347 }
348 order[j] = v;
349 }

352static void ensure_load(Load *load, int length) {
353 if (load->length >= length) return;
354 int size = maximum(length, maximum(16, load->length * 2));
355 int *days = realloc(load->days, (size_t)size * sizeof *days);
356 if (!days) exit(EXIT_FAILURE);
357 memset(days + load->length, 0, (size_t)(size - load->length) * sizeof *days);
358 load->days = days;
359 load->length = size;
360}

362static void level(Schedule *s, int limit) {
363 char *names[MAX_TASKS];
364 int resource_count = resources(names);
365 Load loads[MAX_TASKS] = {0};
366 for (int i = 0; i < resource_count; i++) loads[i].resource = names[i];
367 bool decided[MAX_TASKS] = {false};
368 int starts[MAX_TASKS], finishes[MAX_TASKS], order[MAX_TASKS], total = 0;
369 for (int step = 0; step < task_count; step++) {
370 int v = -1;
371 for (int i = 0; i < task_count; i++) if (!decided[i]) {
372 bool ready = true;
373 for (int j = 0; j < dep_count; j++) if (dependencies[j].b == tasks[i] && !decided[index_of(dependencies[j].a)]) ready = false;
374 if (ready && (v < 0 || s->times[i].ls < s->times[v].ls)) v = i;
375 }
376 Task *t = tasks[v];
377 int begin = t->not_before, d = duration(t), cap = capacity(t->resource, limit);
378 for (int i = 0; i < dep_count; i++) if (dependencies[i].b == t) {
379 Dependency *dep = &dependencies[i];
380 int a = index_of(dep->a);
381 begin = maximum(begin, (dep->kind == 1 ? starts[a] : finishes[a]) + dep->lag - (dep->kind == 2 ? d : 0));
382 }
383 int r = 0;
384 while (strcmp(names[r], t->resource)) r++;
385 Load *load = &loads[r];
386 for (;;) {
387 ensure_load(load, begin + d);
388 bool fits = true;
389 for (int k = begin; k < begin + d; k++) if (load->days[k] + t->units > cap) {
390 fits = false;
391 break;
392 }
393 if (fits) break;
394 begin++;
395 }
396 starts[v] = begin;
397 finishes[v] = begin + d;
398 for (int k = begin; k < finishes[v]; k++) load->days[k] += t->units;
399 order[step] = v;
400 decided[v] = true;
401 total = maximum(total, finishes[v]);
402 }
403 printf("level capacity %d: duration %d -> %d\n", limit, s->total, total);
404 for (int i = 0; i < task_count; i++) {
405 int v = order[i];
406 printf("%s start %d finish %d shift %d\n", tasks[v]->name, starts[v], finishes[v], starts[v] - s->times[v].es);
407 }
408 for (int r = 0; r < resource_count; r++) {
409 int *original = usage(s, names[r]);
410 printf("resource %s capacity %d peak %d -> %d\n", names[r], capacity(names[r], limit), peak(original, s->total), peak(loads[r].days, loads[r].length));
411 free(original);
412 free(loads[r].days);
413 }
414}

416static void analysis(const char *c, char **w) {
417 if (!task_count) fail("no tasks");
418 char cycle_text[(MAX_TASKS + 1) * 13];
419 if (cycle(cycle_text)) fail("cycle %s", cycle_text);
420 if (!strcmp(c, "CALENDAR") && start_date < 0) fail("no start date");
421 if (!strcmp(c, "WHATIF")) task(w[1]);
422 if (!strcmp(c, "VARIANCE") && !has_baseline) fail("no baseline");
423 if (!strcmp(c, "LOAD")) {
424 bool found = false;
425 for (int i = 0; i < task_count; i++) if (!strcmp(tasks[i]->resource, w[1])) found = true;
426 if (!found) fail("unknown resource %s", w[1]);
427 }
428 if (!strcmp(c, "LEVEL")) for (int i = 0; i < task_count; i++) {
429 int cap = capacity(tasks[i]->resource, atoi(w[1]));
430 if (tasks[i]->units > cap) fail("capacity %d too small for %s", cap, tasks[i]->name);
431 }
432 Schedule s = compute(NULL, 0);
433 if (!strcmp(c, "CPM")) {
434 int width = 4;
435 for (int i = 0; i < task_count; i++) width = maximum(width, (int)strlen(tasks[i]->name));
436 printf("%-*s%5s%5s%5s%5s%5s%5s%5s\n", width, "TASK", "DUR", "ES", "EF", "LS", "LF", "TF", "FF");
437 for (int i = 0; i < task_count; i++) {
438 Timing *x = &s.times[i];
439 printf("%-*s%5d%5d%5d%5d%5d%5d%5d%s\n", width, tasks[i]->name, x->duration, x->es, x->ef, x->ls, x->lf, x->tf, x->free, x->tf == 0 ? " *" : "");
440 }
441 printf("project duration %d\n", s.total);
442 } else if (!strcmp(c, "CRITICAL") || !strcmp(c, "PERTSUM")) {
443 Paths paths = critical_paths(&s);
444 if (!strcmp(c, "CRITICAL")) printf("critical paths: %d\n", paths.count);
445 for (int i = 0; i < paths.count; i++) {
446 long long expected = 0, variance = 0;
447 for (int j = 0; j < paths.lengths[i]; j++) {
448 Task *t = tasks[paths.vertices[i][j]];
449 if (j) fputs(" -> ", stdout);
450 fputs(t->name, stdout);

452 if (t->has_estimate) {
453 o = t->estimate[0];
454 m = t->estimate[1];
455 p = t->estimate[2];
456 }
457 expected += o + 4LL * m + p;
458 variance += (long long)(p - o) * (p - o);
459 }
460 if (!strcmp(c, "PERTSUM")) {
461 fputs(": expected ", stdout);
462 fraction(expected, 6);
463 fputs(" variance ", stdout);
464 fraction(variance, 36);
465 }
466 putchar('\n');
467 }
468 } else if (!strcmp(c, "SLACK")) {
469 int floats[MAX_TASKS], count = 0;
470 for (int i = 0; i < task_count; i++) {
471 bool found = false;
472 for (int j = 0; j < count; j++) if (floats[j] == s.times[i].tf) found = true;
473 if (!found) floats[count++] = s.times[i].tf;
474 }
475 qsort(floats, (size_t)count, sizeof *floats, int_compare);
476 for (int i = 0; i < count; i++) {
477 bool included[MAX_TASKS] = {false};
478 for (int j = 0; j < task_count; j++) included[j] = s.times[j].tf == floats[i];
479 printf("float %d: ", floats[i]);
480 name_list(included);
481 putchar('\n');
482 }
483 } else if (!strcmp(c, "DRAG")) {
484 for (int i = 0; i < task_count; i++) if (s.times[i].tf == 0) {
485 Schedule after = compute(tasks[i], 0);
486 printf("%s drag %d\n", tasks[i]->name, s.total - after.total);
487 }
488 } else if (!strcmp(c, "STATS")) {
489 int critical = 0, total_float = 0, counts[3] = {0};
490 for (int i = 0; i < task_count; i++) {
491 critical += s.times[i].tf == 0;
492 total_float += s.times[i].tf;
493 }
494 for (int i = 0; i < dep_count; i++) counts[dependencies[i].kind]++;
495 printf("tasks %d, critical %d, dependencies %d (FS %d, SS %d, FF %d)\nwork ", task_count, critical, dep_count, counts[0], counts[1], counts[2]);
496 char *names[MAX_TASKS];
497 int count = resources(names);
498 for (int r = 0; r < count; r++) {
499 int work = 0;
500 for (int i = 0; i < task_count; i++) if (!strcmp(tasks[i]->resource, names[r])) work += duration(tasks[i]) * tasks[i]->units;
501 if (r) fputs(", ", stdout);
502 printf("%s %d", names[r], work);
503 }

507 fraction(100LL * critical, task_count);
508 puts("%");
509 } else if (!strcmp(c, "CALENDAR")) {
510 int *days = allocate((size_t)s.total * sizeof *days), count = 0;
511 for (int day = start_date; count < s.total; day++) if (working(day)) days[count++] = day;
512 int order[MAX_TASKS];

514 for (int i = 0; i < task_count; i++) {

524 print_date(first);
525 fputs(" - ", stdout);
526 print_date(last);
527 printf(", %d working days, %d calendar days\n", s.total, last - first + 1);
528 if (deadline >= 0) {
529 fputs("deadline ", stdout);
530 print_date(deadline);
531 if (last <= deadline) printf(" met, %d working days to spare\n", working_between(last, deadline));
532 else printf(" missed by %d working days\n", working_between(deadline, last));
533 }
534 fputs("holidays in span: ", stdout);
535 bool any = false;
536 for (int i = 0; i < holiday_count; i++) if (weekday(holidays[i]) && holidays[i] >= first && holidays[i] <= last) {
537 if (any) fputs(", ", stdout);
538 print_date(holidays[i]);
539 any = true;
540 }
541 if (!any) fputs("none", stdout);
542 putchar('\n');
543 free(days);
544 } else if (!strcmp(c, "GANTT")) {
545 if (s.total > 60) fail("chart too wide (%d days)", s.total);
546 int width = 0, order[MAX_TASKS];
547 for (int i = 0; i < task_count; i++) width = maximum(width, (int)strlen(tasks[i]->name));
548 printf("%*s |", width, "");
549 for (int day = 0; day < s.total; day++) printf("%d", day % 10);
550 puts("|");
551 start_order(&s, order);
552 for (int i = 0; i < task_count; i++) {
553 int v = order[i];
554 Timing *x = &s.times[v];
555 printf("%-*s |", width, tasks[v]->name);
556 for (int day = 0; day < s.total; day++) putchar(x->es <= day && day < x->ef ? (x->tf == 0 ? '=' : '-') : x->ef <= day && day < x->lf ? '.' : ' ');
557 puts("|");
558 }
559 } else if (!strcmp(c, "LEVEL")) level(&s, atoi(w[1]));
560 else if (!strcmp(c, "LOAD")) {
561 int *load = usage(&s, w[1]), highest = peak(load, s.total), day = 0;
562 printf("load %s:", w[1]);
563 for (int i = 0; i < s.total; i++) printf(" %d", load[i]);
564 while (load[day] != highest) day++;
565 printf("\npeak %d on day %d\n", highest, day);
566 int cap_index = capacity_index(w[1]);
567 if (cap_index >= 0) {
568 int exceeded = 0, cap = capacities[cap_index].limit;
569 for (int i = 0; i < s.total; i++) exceeded += load[i] > cap;
570 printf("capacity %d exceeded on %d days\n", cap, exceeded);
571 }
572 free(load);
573 } else if (!strcmp(c, "CRASH")) {
574 int budget = atoi(w[1]), remaining = budget, initial = s.total, steps = 0;
575 const char *reason;
576 for (;;) {
577 Task *candidate = NULL;
578 for (int i = 0; i < task_count; i++) {
579 Task *t = tasks[i];
580 if (s.times[i].tf != 0 || !t->has_crash || t->crashed >= t->crash_limit) continue;
581 Schedule after = compute(t, duration(t) - 1);
582 if (after.total < s.total && (!candidate || t->crash_cost < candidate->crash_cost)) candidate = t;
583 }
584 if (!candidate) {
585 reason = "no candidate";
586 break;
587 }
588 if (candidate->crash_cost > remaining) {
589 reason = "insufficient budget";
590 break;
591 }
592 int before = duration(candidate);
593 candidate->crashed++;
594 remaining -= candidate->crash_cost;
595 s = compute(NULL, 0);
596 printf("step %d: %s %d -> %d, cost %d, project %d\n", ++steps, candidate->name, before, duration(candidate), candidate->crash_cost, s.total);
597 }
598 printf("crash done: %d days, cost %d, project %d -> %d, budget left %d, %s\n", steps, budget - remaining, initial, s.total, remaining, reason);
599 } else if (!strcmp(c, "WHATIF")) {
600 Task *t = task(w[1]);
601 Schedule after = compute(t, atoi(w[2]));
602 printf("whatif %s %d -> %s: project %d -> %d\n", t->name, duration(t), w[2], s.total, after.total);
603 bool changed_critical = false;
604 for (int i = 0; i < task_count; i++) {
605 Timing *a = &s.times[i], *b = &after.times[i];
606 changed_critical |= (a->tf == 0) != (b->tf == 0);
607 if (a->es == b->es && a->ef == b->ef && a->tf == b->tf) continue;
608 fputs(tasks[i]->name, stdout);
609 if (a->es != b->es) printf(" ES %d->%d", a->es, b->es);
610 if (a->ef != b->ef) printf(" EF %d->%d", a->ef, b->ef);
611 if (a->tf != b->tf) printf(" TF %d->%d", a->tf, b->tf);
612 putchar('\n');
613 }
614 if (changed_critical) {
615 bool before[MAX_TASKS], later[MAX_TASKS];
616 for (int i = 0; i < task_count; i++) {
617 before[i] = s.times[i].tf == 0;
618 later[i] = after.times[i].tf == 0;
619 }
620 fputs("critical ", stdout);
621 name_list(before);
622 fputs(" -> ", stdout);
623 name_list(later);
624 putchar('\n');
625 }
626 } else if (!strcmp(c, "BASELINE")) {
627 baseline_count = task_count;
628 baseline_total = s.total;
629 has_baseline = true;
630 for (int i = 0; i < task_count; i++) {
631 strcpy(baseline[i].name, tasks[i]->name);
632 baseline[i].es = s.times[i].es;
633 baseline[i].ef = s.times[i].ef;
634 }
635 printf("baseline saved: %d tasks, duration %d\n", task_count, s.total);
636 } else if (!strcmp(c, "VARIANCE")) {
637 printf("variance: duration %d -> %d (%+d)\n", baseline_total, s.total, s.total - baseline_total);
638 int a = 0, b = 0;
639 bool changed = false;
640 while (a < task_count || b < baseline_count) {
641 int order = a == task_count ? 1 : b == baseline_count ? -1 : strcmp(tasks[a]->name, baseline[b].name);
642 if (order < 0) {
643 printf("%s new\n", tasks[a++]->name);
644 changed = true;
645 } else if (order > 0) {
646 printf("%s removed\n", baseline[b++].name);
647 changed = true;
648 } else {
649 Timing *x = &s.times[a];
650 BaselineTask *old = &baseline[b];
651 if (x->es != old->es || x->ef != old->ef) {
652 printf("%s start %+d finish %+d\n", tasks[a]->name, x->es - old->es, x->ef - old->ef);
653 changed = true;
654 }
655 a++;
656 b++;
657 }
658 }
659 if (!changed) puts("no task changes");
660 }
661}

663static void command(char *line) {
664 char *w[6];
665 int count = 0;
666 for (char *p = strtok(line, " "); p; p = strtok(NULL, " ")) w[count++] = p;
667 const char *c = w[0];
668 if (!strcmp(c, "TASK")) {
669 if (find_task(w[1])) fail("task %s already exists", w[1]);
670 Task *t = allocate(sizeof *t);
671 strcpy(t->name, w[1]);
672 strcpy(t->resource, w[3]);
673 t->base = atoi(w[2]);
674 t->units = atoi(w[4]);
675 tasks[task_count++] = t;
676 qsort(tasks, (size_t)task_count, sizeof *tasks, task_compare);
677 printf("task %s added\n", t->name);
678 } else if (!strcmp(c, "DEP") || !strcmp(c, "UNDEP")) {
679 Task *a = task(w[1]), *b = task(w[2]);
680 int i = dependency(a, b);
681 if (!strcmp(c, "DEP")) {
682 if (a == b) fail("self dependency %s", a->name);
683 if (i >= 0) fail("dependency %s -> %s already exists", a->name, b->name);
684 int kind = !strcmp(w[3], "FS") ? 0 : !strcmp(w[3], "SS") ? 1 : 2, lag = atoi(w[4]);
685 dependencies[dep_count++] = (Dependency){a, b, kind, lag};
686 printf("dependency %s -> %s %s%+d added\n", a->name, b->name, kinds[kind], lag);
687 } else {
688 if (i < 0) fail("no dependency %s -> %s", a->name, b->name);
689 memmove(dependencies + i, dependencies + i + 1, (size_t)(dep_count - i - 1) * sizeof *dependencies);
690 dep_count--;
691 printf("dependency %s -> %s removed\n", a->name, b->name);
692 }
693 } else if (!strcmp(c, "REMOVE")) {
694 Task *t = task(w[1]);
695 int removed = 0;
696 for (int i = dep_count - 1; i >= 0; i--) if (dependencies[i].a == t || dependencies[i].b == t) {
697 memmove(dependencies + i, dependencies + i + 1, (size_t)(dep_count - i - 1) * sizeof *dependencies);
698 dep_count--;
699 removed++;
700 }
701 int i = index_of(t);
702 memmove(tasks + i, tasks + i + 1, (size_t)(task_count - i - 1) * sizeof *tasks);
703 task_count--;
704 printf("task %s removed with %d dependencies\n", t->name, removed);
705 free(t);
706 } else if (!strcmp(c, "NOTBEFORE") || !strcmp(c, "CRASHCOST") || !strcmp(c, "UNCRASH") || !strcmp(c, "PERT")) {
707 Task *t = task(w[1]);
708 if (!strcmp(c, "NOTBEFORE")) {
709 t->not_before = atoi(w[2]);
710 printf("notbefore %s %d\n", t->name, t->not_before);
711 } else if (!strcmp(c, "CRASHCOST")) {
712 int limit = atoi(w[2]);
713 if (limit >= t->base) fail("crash limit too large for %s", t->name);
714 if (limit < t->crashed) fail("crash limit below used days for %s", t->name);
715 t->has_crash = true;
716 t->crash_limit = limit;
717 t->crash_cost = atoi(w[3]);
718 printf("crash %s max %d cost %d\n", t->name, limit, t->crash_cost);
719 } else if (!strcmp(c, "UNCRASH")) {
720 printf("uncrash %s restored %d days\n", t->name, t->crashed);
721 t->crashed = 0;
722 } else {
723 int o = atoi(w[2]), m = atoi(w[3]), p = atoi(w[4]);
724 if (o > m || m > p) fail("invalid estimate");
725 t->has_estimate = true;
726 t->estimate[0] = o;
727 t->estimate[1] = m;
728 t->estimate[2] = p;
729 printf("pert %s expected ", t->name);
730 fraction(o + 4LL * m + p, 6);
731 fputs(" variance ", stdout);
732 fraction((long long)(p - o) * (p - o), 36);
733 putchar('\n');
734 }
735 } else if (!strcmp(c, "CAPACITY")) {
736 int i = capacity_index(w[1]);
737 if (i < 0) {
738 Capacity *p = realloc(capacities, (size_t)(capacity_count + 1) * sizeof *p);
739 if (!p) exit(EXIT_FAILURE);
740 capacities = p;
741 i = capacity_count++;
742 strcpy(capacities[i].resource, w[1]);
743 }
744 capacities[i].limit = atoi(w[2]);
745 printf("capacity %s %s\n", w[1], w[2]);
746 } else if (!strcmp(c, "START") || !strcmp(c, "HOLIDAY") || !strcmp(c, "DEADLINE")) {
747 int date = parse_date(w[1]);
748 if (!strcmp(c, "HOLIDAY")) {
749 for (int i = 0; i < holiday_count; i++) if (holidays[i] == date) fail("duplicate holiday %s", w[1]);
750 holidays[holiday_count++] = date;
751 qsort(holidays, (size_t)holiday_count, sizeof *holidays, int_compare);
752 fputs("holiday ", stdout);
753 } else if (!strcmp(c, "START")) {
754 start_date = date;
755 fputs("start ", stdout);
756 } else {
757 deadline = date;
758 fputs("deadline ", stdout);
759 }
760 print_date(date);
761 putchar('\n');
762 } else if (!strcmp(c, "LIST")) {
763 printf("tasks %d, dependencies %d, holidays %d\n", task_count, dep_count, holiday_count);
764 for (int i = 0; i < task_count; i++) {
765 Task *t = tasks[i];
766 bool predecessors[MAX_TASKS] = {false};
767 for (int j = 0; j < task_count; j++) predecessors[j] = dependency(tasks[j], t) >= 0;
768 printf("%s duration %d resource %s units %d preds ", t->name, duration(t), t->resource, t->units);
769 name_list(predecessors);
770 putchar('\n');
771 }
772 } else if (!strcmp(c, "SHOW")) {
773 Task *t = task(w[1]);
774 printf("task %s duration %d resource %s units %d", t->name, duration(t), t->resource, t->units);
775 if (t->not_before) printf(" notbefore %d", t->not_before);
776 putchar('\n');
777 for (int predecessors = 1; predecessors >= 0; predecessors--) {
778 fputs(predecessors ? "preds: " : "succs: ", stdout);
779 bool any = false;
780 for (int j = 0; j < task_count; j++) {
781 int i = predecessors ? dependency(tasks[j], t) : dependency(t, tasks[j]);
782 if (i < 0) continue;
783 if (any) fputs(", ", stdout);
784 printf("%s %s%+d", tasks[j]->name, kinds[dependencies[i].kind], dependencies[i].lag);
785 any = true;
786 }
787 if (!any) fputs("none", stdout);
788 putchar('\n');
789 }
790 if (t->has_crash) printf("crash: max %d used %d cost %d\n", t->crash_limit, t->crashed, t->crash_cost);
791 else puts("crash: none");
792 if (t->has_estimate) printf("pert: %d %d %d\n", t->estimate[0], t->estimate[1], t->estimate[2]);
793 else puts("pert: none");
794 char cycle_text[(MAX_TASKS + 1) * 13];
795 if (cycle(cycle_text)) puts("schedule: unavailable");
796 else {
797 Schedule s = compute(NULL, 0);
798 Timing *x = &s.times[index_of(t)];
799 printf("schedule: ES %d EF %d LS %d LF %d TF %d FF %d%s\n", x->es, x->ef, x->ls, x->lf, x->tf, x->free, x->tf == 0 ? " critical" : "");
800 }
801 } else if (!strcmp(c, "IMPACT")) {
802 int v = index_of(task(w[1])), queue[MAX_TASKS], head = 0, tail = 0;
803 bool seen[MAX_TASKS] = {false};
804 seen[v] = true;
805 queue[tail++] = v;
806 while (head < tail) {
807 int u = queue[head++];
808 for (int j = 0; j < task_count; j++) if (!seen[j] && dependency(tasks[u], tasks[j]) >= 0) {
809 seen[j] = true;
810 queue[tail++] = j;
811 }
812 }
813 seen[v] = false;
814 printf("impact %s: ", w[1]);
815 name_list(seen);
816 putchar('\n');
817 } else analysis(c, w);
818}

820static char *read_line(void) {
821 size_t length = 0, capacity = 128;
822 char *line = allocate(capacity);
823 int ch;
824 while ((ch = getchar()) != EOF && ch != '\n') {
825 if (length + 1 == capacity) {
826 capacity *= 2;
827 char *p = realloc(line, capacity);
828 if (!p) exit(EXIT_FAILURE);
829 line = p;
830 }
831 line[length++] = (char)ch;
832 }
833 line[length] = '\0';
834 return line;
835}
836int main(void) {
837 char *line = read_line();
838 int n = atoi(line);
839 free(line);
840 for (int i = 0; i < n; i++) {
841 line = read_line();
842 if (setjmp(command_error) == 0) command(line);
843 free(line);
844 }
845 for (int i = 0; i < task_count; i++) free(tasks[i]);
846 free(capacities);
847 return 0;
848}
