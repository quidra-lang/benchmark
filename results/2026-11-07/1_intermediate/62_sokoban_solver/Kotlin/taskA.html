75fun decode(text: String, allowed: String): String? {
76 val result = StringBuilder()
77 var index = 0
78 while (index < text.length) {
79 val start = index
80 while (index < text.length && text[index].isDigit()) {
81 index++
82 }
83 val count = text.substring(start, index)
84 if (count.length > 2 || count.startsWith("0") || index == text.length) {
85 return null
86 }
87 if (text[index] !in allowed) {
88 return null
89 }
90 result.append(text[index].toString().repeat(if (count.isEmpty()) 1 else count.toInt()))
91 index++
92 }
93 return result.toString()
94}

96fun encode(text: String): String {
97 val result = StringBuilder()
98 var start = 0
99 while (start < text.length) {
100 var end = start
101 while (end < text.length && text[end] == text[start]) {
102 end++
103 }
104 val run = end - start
105 if (run > 1) {
106 result.append(run)
107 }
108 result.append(text[start])
109 start = end
110 }
111 return result.toString()
112}

148fun findInterior(height: Int, width: Int, walls: BooleanArray, start: Int): BooleanArray {
149 val interior = BooleanArray(height * width)
150 interior[start] = true
151 val stack = ArrayDeque(listOf(start))
152 while (stack.isNotEmpty()) {
153 val cell = stack.removeLast()
154 val row = cell / width
155 val column = cell % width
156 for ((dRow, dColumn) in DELTAS) {
157 val nextRow = row + dRow
158 val nextColumn = column + dColumn
159 if (nextRow in 0 until height && nextColumn in 0 until width) {
160 val neighbor = nextRow * width + nextColumn
161 if (!walls[neighbor] && !interior[neighbor]) {
162 interior[neighbor] = true
163 stack.addLast(neighbor)
164 }
165 }
166 }
167 }
168 return interior
169}

199 fun invalidity(players: Int, boxes: List<Int>): String? {
200 if (players != 1) {
201 return "players $players"
202 }
203 if (boxes.isEmpty()) {
204 return "no boxes"
205 }
206 if (boxes.size != goalCells.size) {
207 return "boxes ${boxes.size} goals ${goalCells.size}"
208 }
209 for (cell in 0 until size) {
210 if (interior[cell] && onBorder(cell)) {
211 return "open ${cellText(cell)}"
212 }
213 }
214 for (cell in 0 until size) {
215 if ((goals[cell] || cell in boxes) && !interior[cell]) {
216 return "outside ${cellText(cell)}"
217 }
218 }
219 return null

222 fun analyze() {
223 distances = goalCells.map { pushDistances(it) }
224 dead = BooleanArray(size) { interior[it] && goalDistance(it) == null }

227 private fun pushDistances(goal: Int): Array<Int?> {
228 val distance = arrayOfNulls<Int>(size)
229 distance[goal] = 0
230 val queue = ArrayDeque(listOf(goal))
231 while (queue.isNotEmpty()) {
232 val cell = queue.removeFirst()
233 for (offset in offsets) {
234 val box = cell - offset
235 if (interior[box] && interior[box - offset] && distance[box] == null) {
236 distance[box] = distance[cell]!! + 1
237 queue.addLast(box)
238 }
239 }
240 }
241 return distance
242 }

244 fun goalDistance(cell: Int): Int? = distances.mapNotNull { it[cell] }.minOrNull()

250 fun reachable(player: Int, boxes: List<Int>): BooleanArray {
251 val seen = BooleanArray(size)
252 seen[player] = true
253 val stack = ArrayDeque(listOf(player))
254 while (stack.isNotEmpty()) {
255 val cell = stack.removeLast()
256 for (offset in offsets) {
257 val neighbor = cell + offset
258 if (interior[neighbor] && !seen[neighbor] && neighbor !in boxes) {
259 seen[neighbor] = true
260 stack.addLast(neighbor)
261 }
262 }
263 }
264 return seen
265 }

269 fun walkPath(start: Int, target: Int, boxes: List<Int>): String {
270 val cameFrom = arrayOfNulls<Pair<Int, Char>>(size)
271 val queue = ArrayDeque(listOf(start))
272 while (queue.isNotEmpty()) {
273 val cell = queue.removeFirst()
274 for (direction in DIRECTIONS) {
275 val neighbor = cell + offsetOf(direction)
276 if (interior[neighbor] && neighbor !in boxes && neighbor != start &&
277 cameFrom[neighbor] == null
278 ) {
279 cameFrom[neighbor] = cell to direction
280 queue.addLast(neighbor)
281 }
282 }
283 }
284 val directions = StringBuilder()
285 var cell = target
286 while (cell != start) {
287 val (previous, direction) = cameFrom[cell]!!
288 directions.append(direction)
289 cell = previous
290 }
291 return directions.reverse().toString()
292 }

294 fun step(position: Position, direction: Char): Pair<Position, Char> {
295 val offset = offsetOf(direction)
296 val target = position.player + offset
297 if (walls[target]) {
298 throw StepException("wall")
299 }
300 if (target !in position.boxes) {
301 return Position(target, position.boxes) to direction
302 }
303 val beyond = target + offset
304 if (walls[beyond] || beyond in position.boxes) {
305 throw StepException("blocked")
306 }
307 val boxes = moveBox(position.boxes, target, beyond)
308 return Position(target, boxes) to direction.uppercaseChar()
309 }

311 fun pushable(box: Int, direction: Char, boxes: List<Int>, reachable: BooleanArray): Boolean {
312 val offset = offsetOf(direction)
313 val destination = box + offset
314 return reachable[box - offset] && interior[destination] && destination !in boxes
315 }

317 private fun sideBlocked(
318 box: Int,
319 sides: List<Int>,
320 boxes: List<Int>,
321 assumed: List<Int>,
322 ): Boolean {
323 if (sides.any { walls[it] || it in assumed }) {
324 return true

326 if (sides.all { dead[it] }) {
327 return true

336 return sideBlocked(box, horizontal, boxes, assumed) &&

365 private fun mapCell(kind: String, cell: Int): Int {
366 val row = cell / width
367 val column = cell % width
368 val lastRow = height - 1
369 val lastColumn = width - 1
370 return when (kind) {
371 "ROT90" -> column * height + lastRow - row
372 "ROT180" -> (lastRow - row) * width + lastColumn - column
373 "ROT270" -> (lastColumn - column) * height + row
374 "FLIPH" -> row * width + lastColumn - column
375 else -> (lastRow - row) * width + column

382 fun transformed(kind: String): Level {
383 val rotated = kind == "ROT90" || kind == "ROT270"
384 val movedWalls = BooleanArray(size)
385 val movedGoals = BooleanArray(size)
386 val movedInterior = BooleanArray(size)
387 for (cell in 0 until size) {
388 val moved = mapCell(kind, cell)
389 movedWalls[moved] = walls[cell]
390 movedGoals[moved] = goals[cell]
391 movedInterior[moved] = interior[cell]
392 }
393 val level = if (rotated) {
394 Level(width, height, movedWalls, movedGoals, movedInterior)
395 } else {
396 Level(height, width, movedWalls, movedGoals, movedInterior)
397 }
398 level.analyze()
399 return level
400 }

403fun buildLevel(rows: List<String>): Pair<Level, Position> {
404 val height = rows.size
405 val width = rows[0].length
406 val cells = rows.joinToString("")
407 val walls = BooleanArray(cells.length) { cells[it] == '#' }
408 val goals = BooleanArray(cells.length) { cells[it] in ".*+" }
409 val players = cells.indices.filter { cells[it] in "@+" }
410 val boxes = cells.indices.filter { cells[it] in "\$*" }
411 val interior = if (players.size == 1) {
412 findInterior(height, width, walls, players[0])
413 } else {
414 BooleanArray(cells.length)
415 }
416 val level = Level(height, width, walls, goals, interior)
417 level.reason = level.invalidity(players.size, boxes)
418 if (level.reason == null) {
419 level.analyze()
420 }
421 return level to Position(players.firstOrNull() ?: -1, boxes)
422}

424fun solutionPushes(nodes: List<SearchNode>): List<Push> =

430fun solutionNotation(level: Level, start: Position, pushes: List<Push>): String {
431 var position = start
432 val notation = StringBuilder()
433 for ((box, direction) in pushes) {
434 val offset = level.offsetOf(direction)
435 notation.append(level.walkPath(position.player, box - offset, position.boxes))
436 notation.append(direction.uppercaseChar())
437 position = Position(box, moveBox(position.boxes, box, box + offset))
438 }
439 return notation.toString()
440}

442fun pushSearch(
443 level: Level,
444 start: Position,
445 fixed: List<Int>,
446 target: Int?,
447 pruning: Boolean,
448 limit: Int?,
449): SearchResult {
450 val first = SearchNode(start.boxes, level.representative(start.player, start.boxes), null, null)
451 val nodes = mutableListOf(first)
452 val known = hashSetOf(first.boxes to first.representative)
453 var expanded = 0
454 while (expanded < nodes.size) {
455 if (expanded == limit) {
456 return SearchResult("limit", expanded, nodes.size)
457 }
458 val node = nodes[expanded]
459 expanded++
460 val reachable = level.reachable(node.representative, node.boxes)
461 for (box in node.boxes) {
462 if (box in fixed) {
463 continue
464 }
465 for (direction in DIRECTIONS) {
466 if (!level.pushable(box, direction, node.boxes, reachable)) {
467 continue
468 }
469 val destination = box + level.offsetOf(direction)
470 val boxes = moveBox(node.boxes, box, destination)
471 if (pruning && (level.dead[destination] || level.hasFrozenBox(boxes))) {
472 continue
473 }
474 val representative = level.representative(box, boxes)
475 if (!known.add(boxes to representative)) {
476 continue
477 }
478 nodes.add(SearchNode(boxes, representative, expanded - 1, Push(box, direction)))
479 val reached = if (target == null) level.solved(boxes) else destination == target
480 if (reached) {
481 val notation = solutionNotation(level, start, solutionPushes(nodes))
482 return SearchResult("found", expanded, nodes.size, notation)
483 }
484 }
485 }
486 }
487 return SearchResult("no solution", expanded, nodes.size)
488}

664 if (history.isEmpty()) {
665 throw CommandException("nothing to undo")
666 }
667 val count = minOf(requested, history.length)
668 repeat(count) {
669 val letter = history.last()

672 val offset = level.offsetOf(direction)
673 val player = current.player

677 current.boxes

680 redoList.append(direction)
681 }

683 }

747 private fun bound() {
748 val boxes = current.boxes
749 var best: Assignment? = null
750 for (order in permutations(level.goalCells.indices.toList())) {
751 val pushes = boxes.zip(order) { box, goal -> level.distances[goal][box] }
752 if (null in pushes) {
753 continue
754 }
755 val assignment = Assignment(order, pushes.filterNotNull())
756 if (best == null || assignment.total < best.total) {
757 best = assignment
758 }
759 }
760 if (best == null) {
761 emit("bound none")
762 return
763 }
764 emit("bound ${best.total}")
765 for (index in boxes.indices) {
766 val goal = level.goalCells[best.order[index]]
767 emit(
768 "box ${level.cellText(boxes[index])} goal ${level.cellText(goal)} " +
769 "pushes ${best.pushes[index]}",
770 )
771 }
772 }

844 private fun checkValues(command: Command, arguments: List<String>) {
845 val cellCount = when (command) {
846 Command.PATH, Command.PUSH -> 1
847 Command.BRING -> 2
848 else -> 0
849 }
850 for (index in 0 until 2 * cellCount step 2) {
851 if (!level.contains(arguments[index].toInt(), arguments[index + 1].toInt())) {
852 throw CommandException("bad value")
853 }
854 }
855 if ((command == Command.UNDO || command == Command.REDO) && arguments[0].toInt() < 1) {
856 throw CommandException("bad value")
857 }
858 if ((command == Command.SOLVE || command == Command.HINT) &&
859 arguments[0].toInt() !in 1..MAX_LIMIT
860 ) {
861 throw CommandException("bad value")
862 }
863 }

865 private fun runCommand(name: String, arguments: List<String>) {
866 val command = Command.entries.find { it.name == name }
867 ?: throw CommandException("unknown command $name")
868 val kinds = command.kinds
869 if (arguments.size != kinds.size ||
870 !kinds.zip(arguments).all { (kind, text) -> validArgument(kind, text) }
871 ) {
872 throw CommandException("bad arguments")
873 }
874 if (command != Command.IMPORT && command != Command.VALID) {
875 if (level.reason != null) {
876 throw CommandException("invalid level")
877 }
878 checkValues(command, arguments)
879 }
880 when (command) {
881 Command.IMPORT -> importLevel(arguments[0])
882 Command.VALID -> valid()
883 Command.INFO -> info()
884 Command.SHOW -> show()
885 Command.EXPORT -> exportLevel()
886 Command.MOVE -> move(arguments[0])
887 Command.PUSH -> push(cellArgument(arguments, 0), arguments[2][0])
888 Command.BRING -> bring(cellArgument(arguments, 0), cellArgument(arguments, 2))
889 Command.PATH -> path(cellArgument(arguments, 0))
890 Command.UNDO -> undo(arguments[0].toInt())
891 Command.REDO -> redo(arguments[0].toInt())
892 Command.RESET -> reset()
893 Command.STATUS -> status()
894 Command.HISTORY -> showHistory()
895 Command.BOXES -> showBoxes()
896 Command.REACH -> reach()
897 Command.DEADSQUARES -> deadSquares()
898 Command.DEADLOCK -> deadlock()
899 Command.BOUND -> bound()
900 Command.SAVE -> save(arguments[0])
901 Command.RESTORE -> restore(arguments[0])
902 Command.CHECK -> check(arguments[0])
903 Command.SOLVE -> solve(arguments[0].toInt())
904 Command.HINT -> hint(arguments[0].toInt())
905 Command.PLAY -> play()
906 Command.TRANSFORM -> transform(arguments[0])
907 }
908 }
