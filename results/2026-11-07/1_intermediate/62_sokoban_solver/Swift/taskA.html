5func string(_ bytes: [UInt8]) -> String {
6 String(decoding: bytes, as: UTF8.self)
7}
8func lower(_ c: UInt8) -> UInt8 {
9 c >= 65 && c <= 90 ? c + 32 : c
10}
11func upper(_ c: UInt8) -> UInt8 {
12 c >= 97 && c <= 122 ? c - 32 : c
13}
14func isPush(_ c: UInt8) -> Bool {
15 c >= 65 && c <= 90
16}
17func pushCount(_ letters: [UInt8]) -> Int {
18 letters.filter(isPush).count
19}
20func count(_ flags: [Bool]) -> Int {
21 flags.filter { $0 }.count
22}
23func moveBox(_ boxes: [Int], _ from: Int, _ to: Int) -> [Int] {
24 boxes.map { $0 == from ? to : $0 }.sorted()
25}
26func decode(_ word: String, allowed: [UInt8]) -> [UInt8]? {
27 let bytes = Array(word.utf8)
28 var result: [UInt8] = [], i = 0
29 while i < bytes.count {
30 let start = i
31 var n = 0
32 while i < bytes.count && bytes[i] >= 48 && bytes[i] <= 57 {
33 if i - start < 2 {
34 n = n * 10 + Int(bytes[i] - 48)
35 }
36 i += 1
37 }
38 if i - start > 2 || (i > start && bytes[start] == 48) || i == bytes.count {
39 return nil
40 }
41 guard allowed.contains(bytes[i]) else {
42 return nil
43 }
44 result += Array(repeating: bytes[i], count: i == start ? 1 : n)
45 i += 1
46 }
47 return result
48}
49func encode(_ row: String) -> String {
50 let bytes = Array(row.utf8)
51 var result = "", i = 0
52 while i < bytes.count {
53 var j = i + 1
54 while j < bytes.count && bytes[j] == bytes[i] {
55 j += 1
56 }
57 if j - i > 1 {
58 result += String(j - i)
59 }
60 result += string([bytes[i]])
61 i = j
62 }
63 return result
64}
65func turnMap(_ kind: String) -> [UInt8]? {
66 switch kind {
67 case "ROT90":
68 return Array("rlud".utf8)
69 case "ROT180":
70 return Array("durl".utf8)
71 case "ROT270":
72 return Array("lrdu".utf8)
73 case "FLIPH":
74 return Array("udrl".utf8)
75 case "FLIPV":
76 return Array("dulr".utf8)
77 default:
78 return nil
79 }

81func turn(_ letters: [UInt8], _ kind: String) -> [UInt8] {
82 let map = turnMap(kind)!
83 return letters.map { c in
84 let d = map[directions.firstIndex(of: lower(c))!]
85 return isPush(c) ? upper(d) : d
86 }

88func schema(_ command: String) -> String? {
89 switch command {
90 case "IMPORT":
91 return "t"
92 case "MOVE", "CHECK":
93 return "s"
94 case "PUSH":
95 return "nnd"
96 case "BRING":
97 return "nnnn"
98 case "PATH":
99 return "nn"
100 case "UNDO", "REDO", "SOLVE", "HINT":
101 return "n"
102 case "SAVE", "RESTORE":
103 return "a"
104 case "TRANSFORM":
105 return "f"
106 case "VALID", "INFO", "SHOW", "EXPORT", "RESET", "STATUS", "HISTORY", "BOXES", "REACH", "DEADSQUARES", "DEADLOCK",
107 "BOUND",
108 "PLAY":
109 return ""
110 default:
111 return nil
112 }

114func validArgument(_ kind: Character, _ word: String) -> Bool {
115 let b = Array(word.utf8)
116 switch kind {
117 case "n":
118 return !b.isEmpty && b.count <= 6 && b.allSatisfy { $0 >= 48 && $0 <= 57 } && (b.count == 1 || b[0] != 48)
119 case "s":
120 return decode(word, allowed: Array("udlrUDLR".utf8)) != nil
121 case "d":
122 return b.count == 1 && directions.contains(b[0])
123 case "a":
124 return !b.isEmpty && b.count <= 8 && b.allSatisfy { $0 >= 97 && $0 <= 122 }
125 case "f":
126 return turnMap(word) != nil
127 default:
128 return true
129 }

141struct Level {
142 var h: Int, w: Int
143 var walls: [Bool], goals: [Bool], inside: [Bool], dead: [Bool]
144 var goalCells: [Int], distances: [[Int]] = []
145 var reason: String? = nil
146 var size: Int {
147 h * w
148 }
149 var offsets: [Int] {
150 [-w, w, -1, 1]
151 }
152
153 func offset(_ d: UInt8) -> Int {
154 offsets[directions.firstIndex(of: lower(d))!]
155 }
156
157 func interior(_ p: Int) -> Bool {
158 p >= 0 && p < size && inside[p]
159 }
160
161 func cell(_ p: Int) -> String {
162 "\(p / w + 1) \(p % w + 1)"
163 }
164 static func build(_ rows: [String]) -> (Level, Position) {
165 let h = rows.count, w = rows[0].utf8.count
166 let bytes = rows.flatMap { Array($0.utf8) }
167 let walls = bytes.map { $0 == 35 }
168 let goals = bytes.map { [UInt8(46), 42, 43].contains($0) }
169 let players = bytes.indices.filter { bytes[$0] == 64 || bytes[$0] == 43 }
170 let boxes = bytes.indices.filter { bytes[$0] == 36 || bytes[$0] == 42 }
171 let goalCells = bytes.indices.filter { goals[$0] }
172 var level = Level(h: h, w: w, walls: walls, goals: goals,
173 inside: Array(repeating: false, count: h * w),
174 dead: Array(repeating: false, count: h * w), goalCells: goalCells)
175 if players.count == 1 {
176 var queue = [players[0]], head = 0
177 level.inside[players[0]] = true
178 while head < queue.count {
179 let p = queue[head], r = p / w, c = p % w
180 head += 1
181 for (dr, dc) in [(-1, 0), (1, 0), (0, -1), (0, 1)] {
182 let nr = r + dr, nc = c + dc
183 if nr >= 0 && nr < h && nc >= 0 && nc < w {
184 let q = nr * w + nc
185 if !walls[q] && !level.inside[q] {
186 level.inside[q] = true
187 queue.append(q)
188 }
189 }
190 }
191 }
192 }
193 if players.count != 1 {
194 level.reason = "players \(players.count)"
195 } else if boxes.isEmpty {
196 level.reason = "no boxes"
197 } else if boxes.count != goalCells.count {
198 level.reason = "boxes \(boxes.count) goals \(goalCells.count)"
199 } else if let p = (0..<h * w)
200 .first(where: { level.inside[$0] && ($0 / w == 0 || $0 / w == h - 1 || $0 % w == 0 || $0 % w == w - 1) })
201 {
202 level.reason = "open \(level.cell(p))"
203 } else if let p = (0..<h * w)
204 .first(where: { (goals[$0] || boxes.contains($0)) && !level.inside[$0] })
205 {
206 level.reason = "outside \(level.cell(p))"
207 }
208 if level.reason == nil {
209 level.analyze()
210 }
211 return (level, Position(player: players.first ?? 0, boxes: boxes))
212 }
213
214 mutating func analyze() {
215 goalCells = (0..<size).filter { goals[$0] }
216 distances = goalCells.map { goal in
217 var dist = Array(repeating: -1, count: size), queue = [goal], head = 0
218 dist[goal] = 0
219 while head < queue.count {
220 let q = queue[head]
221 head += 1
222 for delta in offsets {
223 let b = q - delta
224 if interior(b) && interior(b - delta) && dist[b] < 0 {
225 dist[b] = dist[q] + 1
226 queue.append(b)
227 }
228 }
229 }
230 return dist
231 }
232 dead = (0..<size).map { p in inside[p] && distances.allSatisfy { $0[p] < 0 } }
233 }
234
235 func reach(_ player: Int, _ boxes: [Int]) -> [Bool] {
236 var seen = Array(repeating: false, count: size), queue = [player], head = 0
237 seen[player] = true
238 while head < queue.count {
239 let q = queue[head]
240 head += 1
241 for delta in offsets {
242 let p = q + delta
243 if interior(p) && !seen[p] && !boxes.contains(p) {
244 seen[p] = true
245 queue.append(p)
246 }
247 }
248 }
249 return seen
250 }
251
252 func representative(_ player: Int, _ boxes: [Int]) -> Int {
253 reach(player, boxes).firstIndex(of: true)!
254 }
255
256 func path(_ player: Int, _ target: Int, _ boxes: [Int]) -> [UInt8] {
257 var from = Array<(Int, UInt8)?>(repeating: nil, count: size), queue = [player], head = 0
258 while head < queue.count {
259 let q = queue[head]
260 head += 1
261 for d in 0..<4 {
262 let p = q + offsets[d]
263 if interior(p) && p != player && !boxes.contains(p) && from[p] == nil {
264 from[p] = (q, directions[d])
265 queue.append(p)
266 }
267 }
268 }
269 var result: [UInt8] = [], p = target
270 while p != player {
271 let (q, d) = from[p]!
272 result.append(d)
273 p = q
274 }
275 return Array(result.reversed())
276 }

278 func step(_ pos: Position, _ direction: UInt8) -> (Position, UInt8, String?) {
279 let d = lower(direction), delta = offset(d), p = pos.player + delta
280 if walls[p] {
281 return (pos, d, "wall")
282 }
283 if pos.boxes.contains(p) {
284 let dest = p + delta
285 if walls[dest] || pos.boxes.contains(dest) {
286 return (pos, d, "blocked")
287 }
288 return (Position(player: p, boxes: moveBox(pos.boxes, p, dest)), upper(d), nil)
289 }
290 return (Position(player: p, boxes: pos.boxes), d, nil)
291 }

293 func pushable(_ b: Int, _ d: Int, _ boxes: [Int], _ reachable: [Bool]) -> Bool {
294 let delta = offsets[d]
295 return reachable[b - delta] && interior(b + delta) && !boxes.contains(b + delta)
296 }

298 func frozen(_ b: Int, _ boxes: [Int], _ assumed: Set<Int> = []) -> Bool {
299 func axis(_ sides: [Int]) -> Bool {
300 if sides.contains(where: { walls[$0] || assumed.contains($0) }) || sides
301 .allSatisfy({ dead[$0] })
302 {
303 return true
304 }
305 let extended = assumed.union([b])
306 return sides.contains { boxes.contains($0) && frozen($0, boxes, extended) }
307 }
308 return axis([b - 1, b + 1]) && axis([b - w, b + w])
309 }

311 func onGoals(_ boxes: [Int]) -> Int {
312 boxes.filter { goals[$0] }.count
313 }
314
315 func solved(_ boxes: [Int]) -> Bool {
316 onGoals(boxes) == boxes.count
317 }
318
319 func render(_ pos: Position, deadMap: Bool = false) -> [String] {
320 let bytes: [UInt8] = (0..<size).map { p in
321 if walls[p] {
322 return 35
323 }
324 if deadMap {
325 return goals[p] ? 46 : (dead[p] ? 120 : 45)
326 }
327 if p == pos.player {
328 return goals[p] ? 43 : 64
329 }
330 if pos.boxes.contains(p) {
331 return goals[p] ? 42 : 36
332 }
333 return goals[p] ? 46 : 45
334 }
335 return (0..<h).map { string(Array(bytes[$0 * w..<($0 + 1) * w])) }
336 }

338 func mapCell(_ p: Int, _ kind: String) -> Int {
339 let r = p / w, c = p % w
340 switch kind {
341 case "ROT90":
342 return c * h + h - 1 - r
343 case "ROT180":
344 return (h - 1 - r) * w + w - 1 - c
345 case "ROT270":
346 return (w - 1 - c) * h + r
347 case "FLIPH":
348 return r * w + w - 1 - c
349 default:
350 return (h - 1 - r) * w + c
351 }
352 }
353
354 func mapPosition(_ pos: Position, _ kind: String) -> Position {
355 Position(player: mapCell(pos.player, kind), boxes: pos.boxes.map { mapCell($0, kind) }.sorted())
356 }

358 func transformed(_ kind: String) -> Level {
359 var result = self
360 if kind == "ROT90" || kind == "ROT270" {
361 result.h = w
362 result.w = h
363 }
364 for p in 0..<size {
365 let q = mapCell(p, kind)
366 result.walls[q] = walls[p]
367 result.goals[q] = goals[p]
368 result.inside[q] = inside[p]
369 }
370 result.analyze()
371 return result
372 }

374struct SearchState: Hashable {
375 let boxes: [Int]
376 let representative: Int
377}
378struct SearchNode {
379 let state: SearchState
380 let parent: Int?
381 let box: Int
382 let direction: Int
383}
384struct SearchResult {
385 let outcome: String
386 let expanded: Int
387 let states: Int
388 var notation: [UInt8] = []
389}
390func search(_ level: Level, _ start: Position, fixed: [Int] = [], target: Int? = nil, pruning: Bool,
391 limit: Int?) -> SearchResult
392{
393 let first = SearchState(boxes: start.boxes, representative: level.representative(start.player, start.boxes))
394 var nodes = [SearchNode(state: first, parent: nil, box: 0, direction: 0)]
395 var known: Set<SearchState> = [first], expanded = 0
396 while expanded < nodes.count {
397 if let limit,
398 expanded == limit
399 {
400 return SearchResult(outcome: "limit", expanded: expanded, states: nodes.count)
401 }
402 let parent = expanded, state = nodes[parent].state
403 expanded += 1
404 let reachable = level.reach(state.representative, state.boxes)
405 for b in state.boxes where !fixed.contains(b) {
406 for d in 0..<4 {
407 if !level.pushable(b, d, state.boxes, reachable) {
408 continue
409 }
410 let dest = b + level.offsets[d], boxes = moveBox(state.boxes, b, dest)
411 if pruning &&
412 (level.dead[dest] || boxes.contains { !level.goals[$0] && level.frozen($0, boxes) })
413 {
414 continue
415 }
416 let next = SearchState(boxes: boxes, representative: level.representative(b, boxes))
417 if !known.insert(next).inserted {
418 continue
419 }
420 nodes.append(SearchNode(state: next, parent: parent, box: b, direction: d))
421 let found = target.map { dest == $0 } ?? level.solved(boxes)
422 if found {
423 var pushes: [(Int, Int)] = [], i = nodes.count - 1
424 while let p = nodes[i].parent {
425 pushes.append((nodes[i].box, nodes[i].direction))
426 i = p
427 }
428 var pos = start, notation: [UInt8] = []
429 for (box, direction) in pushes.reversed() {
430 let delta = level.offsets[direction]
431 notation += level.path(pos.player, box - delta, pos.boxes)
432 notation.append(upper(directions[direction]))
433 pos = Position(player: box, boxes: moveBox(pos.boxes, box, box + delta))
434 }
435 return SearchResult(outcome: "found", expanded: expanded, states: nodes.count, notation: notation)
436 }
437 }
438 }
439 }
440 return SearchResult(outcome: "no solution", expanded: expanded, states: nodes.count)
441}

483 func simulate(_ steps: [UInt8]) -> (Position, [UInt8], String?) {
484 var pos = current, letters: [UInt8] = []
485 for d in steps {
486 let (next, letter, error) = level.step(pos, d)
487 if let error {
488 return (pos, letters, "step \(letters.count + 1) \(error)")
489 }
490 pos = next
491 letters.append(letter)
492 }
493 return (pos, letters, nil)
494 }
495 @discardableResult func take(_ steps: [UInt8]) -> ([UInt8], String?) {
496 let (pos, letters, error) = simulate(steps)
497 current = pos
498 history += letters
499 if !letters.isEmpty {
500 redo = []
501 }
502 return (letters, error)
503 }

505 func cellArgument(_ a: [String], _ i: Int) -> Int {
506 (Int(a[i])! - 1) * level.w + Int(a[i + 1])! - 1
507 }

509 func assignment() -> (Int, [Int])? {
510 var best: (Int, [Int])? = nil
511 func visit(_ order: [Int], _ used: Set<Int>, _ total: Int) {
512 let i = order.count
513 if i == current.boxes.count {
514 if best == nil || total < best!.0 {
515 best = (total, order)
516 }
517 return
518 }
519 for g in level.goalCells.indices where !used.contains(g) {
520 let d = level.distances[g][current.boxes[i]]
521 if d >= 0 {
522 visit(order + [g], used.union([g]), total + d)
523 }
524 }
525 }
526 visit([], [], 0)
527 return best
528 }

530 func searchSolution(_ limit: Int) -> (String, [UInt8]?) {
531 if level.solved(current.boxes) {
532 return ("already solved", nil)
533 }
534 let result = search(level, current, pruning: true, limit: limit)
535 let counts = "expanded \(result.expanded) states \(result.states)"
536 if result.outcome != "found" {
537 return ("\(result.outcome) \(counts)", nil)
538 }
539 return (
540 "solution pushes \(pushCount(result.notation)) moves \(result.notation.count) \(counts)",
541 result.notation
542 )
543 }

545 func execute(_ words: [String], _ rows: [String]) throws {
546 let name = words[0], a = Array(words.dropFirst())
547 if name == "LOAD" {
548 let w = Int(a[1])!
549 if let i = rows
550 .firstIndex(where: { $0.utf8.count != w || !$0.utf8.allSatisfy { levelCharacters.contains($0) } })
551 {
552 throw CommandError("bad row \(i + 1)")
553 }
554 load(rows)
555 emit("loaded \(rows.count)x\(w)")
556 return
557 }
558 guard let kinds = schema(name) else {
559 throw CommandError("unknown command \(name)")
560 }
561 guard a.count == kinds.count && zip(kinds, a).allSatisfy({ validArgument($0.0, $0.1) })
562 else {
563 throw CommandError("bad arguments")
564 }
565 if name == "IMPORT" {
566 let parts = a[0].components(separatedBy: "|")
567 var imported: [String] = []
568 for part in parts {
569 guard let decoded = decode(part, allowed: levelCharacters) else {
570 throw CommandError("bad level")
571 }
572 imported.append(string(decoded))
573 }
574 let w = imported[0].utf8.count
575 guard imported.count <= 10 && w >= 1 && w <= 10 && imported.allSatisfy({ $0.utf8.count == w })
576 else {
577 throw CommandError("bad level")
578 }
579 load(imported)
580 emit("loaded \(imported.count)x\(w)")
581 return
582 }
583 if name == "VALID" {
584 emit(level.reason.map { "invalid \($0)" } ?? "valid")
585 return
586 }
587 guard level.reason == nil else {
588 throw CommandError("invalid level")
589 }
590 let cells = name == "BRING" ? 2 : (["PATH", "PUSH"].contains(name) ? 1 : 0)
591 for i in 0..<cells {
592 let r = Int(a[2 * i])!, c = Int(a[2 * i + 1])!
593 guard r >= 1 && r <= level.h && c >= 1 && c <= level.w else {
594 throw CommandError("bad value")
595 }
596 }
597 if ["UNDO", "REDO", "SOLVE", "HINT"].contains(name) {
598 let n = Int(a[0])!
599 guard n >= 1 && (!["SOLVE", "HINT"].contains(name) || n <= 100000) else {
600 throw CommandError("bad value")
601 }
602 }
603 switch name {
604 case "INFO":
605 emit(
606 "size \(level.h)x\(level.w) walls \(count(level.walls)) cells \(count(level.inside)) boxes \(current.boxes.count) dead \(count(level.dead))"
607 )
608 case "SHOW":
609 output += level.render(current)
610 case "EXPORT":
611 emit("level " + level.render(current).map(encode).joined(separator: "|"))
612 case "MOVE":
613 let (letters, error) = take(decode(a[0], allowed: Array("udlrUDLR".utf8))!)
614 if !letters.isEmpty {
615 emit("moved \(string(letters))")
616 }
617 if let error {
618 throw CommandError(error)
619 }
620 case "PUSH":
621 let b = cellArgument(a, 0), d = Array(a[2].utf8)[0], delta = level.offset(d)
622 guard current.boxes.contains(b) else {
623 throw CommandError("no box")
624 }
625 guard level.interior(b + delta) && !current.boxes.contains(b + delta) else {
626 throw CommandError("blocked")
627 }
628 guard level.reach(current.player, current.boxes)[b - delta] else {
629 throw CommandError("unreachable")
630 }
631 let path = level.path(current.player, b - delta, current.boxes) + [d]
632 emit("pushed \(string(take(path).0))")
633 case "BRING":
634 let b = cellArgument(a, 0), target = cellArgument(a, 2)
635 guard current.boxes.contains(b) else {
636 throw CommandError("no box")
637 }
638 if b == target {
639 emit("brought 0")
640 break
641 }
642 let fixed = current.boxes.filter { $0 != b }
643 let result = search(level, current, fixed: fixed, target: target, pruning: false, limit: nil)
644 guard result.outcome == "found" else {
645 throw CommandError("no route")
646 }
647 take(result.notation)
648 emit("brought \(pushCount(result.notation)) \(string(result.notation))")
649 case "PATH":
650 let target = cellArgument(a, 0)
651 guard level.reach(current.player, current.boxes)[target] else {
652 throw CommandError("unreachable")
653 }
654 let path = level.path(current.player, target, current.boxes), letters = take(path).0
655 emit(letters.isEmpty ? "path 0" : "path \(letters.count) \(string(letters))")
656 case "UNDO":
657 guard !history.isEmpty else {
658 throw CommandError("nothing to undo")
659 }
660 let n = min(Int(a[0])!, history.count)
661 for _ in 0..<n {
662 let letter = history.removeLast(), d = lower(letter), delta = level.offset(d), p = current.player
663 if isPush(letter) {
664 current.boxes = moveBox(current.boxes, p + delta, p)
665 }
666 current.player = p - delta
667 redo.append(d)
668 }
669 emit("undone \(n)")
670 case "REDO":
671 guard !redo.isEmpty else {
672 throw CommandError("nothing to redo")
673 }
674 let n = min(Int(a[0])!, redo.count)
675 for _ in 0..<n {
676 let d = redo.removeLast()
677 let (pos, letter, _) = level.step(current, d)
678 current = pos
679 history.append(letter)
680 }
681 emit("redone \(n)")
682 case "RESET":
683 current = initial
684 history = []
685 redo = []
686 emit("reset")
687 case "STATUS":
688 emit("moves \(history.count) pushes \(pushCount(history)) \(progress(current.boxes))")
689 case "HISTORY":
690 emit(history.isEmpty ? "history 0" : "history \(history.count) \(string(history))")
691 case "BOXES":
692 emit("boxes \(current.boxes.count)")
693 let reachable = level.reach(current.player, current.boxes)
694 for b in current.boxes {
695 let distance = level.distances.map { $0[b] }.filter { $0 >= 0 }.min().map(String.init) ?? "dead"
696 let dirs = (0..<4).filter { level.pushable(b, $0, current.boxes, reachable) }.map { directions[$0] }
697 emit("box \(level.cell(b)) \(distance) \(dirs.isEmpty ? "-" : string(dirs))")
698 }
699 case "REACH":
700 let reachable = level.reach(current.player, current.boxes)
701 emit("reach \(count(reachable)) first \(level.cell(reachable.firstIndex(of: true)!))")
702 case "DEADSQUARES":
703 emit("dead \(count(level.dead))")
704 output += level.render(current, deadMap: true)
705 case "DEADLOCK":
706 if let b = current.boxes.first(where: { level.dead[$0] }) { emit("deadlock dead \(level.cell(b))") }
707 else if let b = current.boxes
708 .first(where: { !level.goals[$0] && level.frozen($0, current.boxes) })
709 {
710 emit("deadlock frozen \(level.cell(b))")
711 } else { emit("no deadlock") }
712 case "BOUND":
713 if let (total, order) = assignment() {
714 emit("bound \(total)")
715 for (i, g) in order.enumerated() {
716 let b = current.boxes[i]
717 emit("box \(level.cell(b)) goal \(level.cell(level.goalCells[g])) pushes \(level.distances[g][b])")
718 }
719 } else {
720 emit("bound none")
721 }
722 case "SAVE":
723 snapshots[a[0]] = Snapshot(position: current, history: history)
724 emit("saved \(a[0]) moves \(history.count)")
725 case "RESTORE":
726 guard let snapshot = snapshots[a[0]] else {
727 throw CommandError("no snapshot \(a[0])")
728 }
729 current = snapshot.position
730 history = snapshot.history
731 redo = []
732 emit("restored \(a[0]) moves \(history.count)")
733 case "CHECK":
734 let (pos, letters, error) = simulate(decode(a[0], allowed: Array("udlrUDLR".utf8))!)
735 if let error {
736 emit("check fail \(error)")
737 } else {
738 emit("check ok moves \(letters.count) pushes \(pushCount(letters)) \(progress(pos.boxes))")
739 }
740 case "SOLVE", "HINT":
741 let (summary, notation) = searchSolution(Int(a[0])!)
742 if name == "SOLVE" {
743 solution = notation.map { Solution(position: current, notation: $0) }
744 emit(summary)
745 if let notation {
746 emit(string(notation))
747 }
748 } else if let notation {
749 let end = notation.firstIndex(where: isPush)! + 1, part = Array(notation[..<end])
750 take(part)
751 emit("hint \(string(part))")
752 } else {
753 emit(summary)
754 }
755 case "PLAY":
756 guard let solution, solution.position == current else {
757 throw CommandError("no solution")
758 }
759 take(solution.notation)
760 emit("played pushes \(pushCount(solution.notation)) moves \(solution.notation.count)")
761 case "TRANSFORM":
762 let kind = a[0], old = level
763 level = old.transformed(kind)
764 initial = old.mapPosition(initial, kind)
765 current = old.mapPosition(current, kind)
766 history = turn(history, kind)
767 redo = turn(redo, kind)
768 snapshots = snapshots.mapValues { Snapshot(
769 position: old.mapPosition($0.position, kind),
770 history: turn($0.history, kind)
771 ) }
772 if let stored = solution {
773 solution = Solution(
774 position: old.mapPosition(stored.position, kind),
775 notation: turn(stored.notation, kind)
776 )
777 }
778 emit("transformed \(level.h)x\(level.w)")
779 default:
780 break
781 }
782 }
783}
784let height = Int(readLine()!.split(separator: " ")[0])!
785let initialRows = (0..<height).map { _ in readLine()! }
786let game = Game(initialRows)
787let commandCount = Int(readLine()!)!
788for _ in 0..<commandCount {
789 let words = readLine()!.components(separatedBy: " ")
790 let rows = words[0] == "LOAD" ? (0..<Int(words[1])!).map { _ in readLine()! } : []
791 do { try game.execute(words, rows) }
792 catch let error as CommandError { game.emit("error: \(error.reason)") }
793 catch { fatalError("Unexpected error") }
794}
795print(game.output.joined(separator: "\n"))
