56def decode(text, allowed):
57 pieces = []
58 index = 0
59 while index < len(text):
60 start = index
61 while index < len(text) and text[index] in DIGITS:
62 index += 1
63 count = text[start:index]
64 if len(count) > 2 or count.startswith("0") or index == len(text):
65 return None
66 if text[index] not in allowed:
67 return None
68 pieces.append(text[index] * (int(count) if count else 1))
69 index += 1
70 return "".join(pieces)

73def encode(text):
74 pieces = []
75 start = 0
76 while start < len(text):
77 end = start
78 while end < len(text) and text[end] == text[start]:
79 end += 1
80 run = end - start
81 pieces.append((str(run) if run > 1 else "") + text[start])
82 start = end
83 return "".join(pieces)

126def find_interior(height, width, walls, start):
127 interior = [False] * (height * width)
128 interior[start] = True
129 stack = [start]
130 while stack:
131 cell = stack.pop()
132 row, column = divmod(cell, width)
133 for d_row, d_column in DELTAS.values():
134 next_row, next_column = row + d_row, column + d_column
135 if 0 <= next_row < height and 0 <= next_column < width:
136 neighbor = next_row * width + next_column
137 if not walls[neighbor] and not interior[neighbor]:
138 interior[neighbor] = True
139 stack.append(neighbor)
140 return interior

160class Level:
161 def __init__(self, height, width, walls, goals, interior):
162 self.height = height
163 self.width = width
164 self.size = height * width
165 self.walls = walls
166 self.goals = goals
167 self.interior = interior
168 self.offsets = {
169 direction: d_row * width + d_column
170 for direction, (d_row, d_column) in DELTAS.items()
171 }
172 self.goal_cells = [cell for cell in range(self.size) if goals[cell]]
173 self.reason = None
174 self.distances = []
175 self.dead = [False] * self.size
176
177 def cell_text(self, cell):
178 row, column = divmod(cell, self.width)
179 return f"{row + 1} {column + 1}"
180
181 def cell_at(self, row, column):
182 return (row - 1) * self.width + column - 1
183
184 def contains(self, row, column):
185 return 1 <= row <= self.height and 1 <= column <= self.width
186
187 def on_border(self, cell):
188 row, column = divmod(cell, self.width)
189 return row in (0, self.height - 1) or column in (0, self.width - 1)
190
191 def invalidity(self, players, boxes):
192 if players != 1:
193 return f"players {players}"
194 if not boxes:
195 return "no boxes"
196 if len(boxes) != len(self.goal_cells):
197 return f"boxes {len(boxes)} goals {len(self.goal_cells)}"
198 for cell in range(self.size):
199 if self.interior[cell] and self.on_border(cell):
200 return "open " + self.cell_text(cell)
201 for cell in range(self.size):
202 if (self.goals[cell] or cell in boxes) and not self.interior[cell]:
203 return "outside " + self.cell_text(cell)
204 return None

206 def analyze(self):
207 self.distances = [self.push_distances(goal) for goal in self.goal_cells]
208 self.dead = [
209 self.interior[cell]
210 and all(distance[cell] is None for distance in self.distances)
211 for cell in range(self.size)
212 ]
213
214 def push_distances(self, goal):
215 distance = [None] * self.size
216 distance[goal] = 0
217 queue = deque([goal])
218 while queue:
219 cell = queue.popleft()
220 for direction in DIRECTIONS:
221 offset = self.offsets[direction]
222 box = cell - offset
223 if (
224 self.interior[box]
225 and self.interior[box - offset]
226 and distance[box] is None
227 ):
228 distance[box] = distance[cell] + 1
229 queue.append(box)
230 return distance
231
232 def goal_distance(self, cell):
233 defined = [
234 distance[cell] for distance in self.distances if distance[cell] is not None
235 ]
236 return min(defined) if defined else None
237
238 def boxes_on_goals(self, boxes):
239 return sum(self.goals[box] for box in boxes)
240
241 def solved(self, boxes):
242 return self.boxes_on_goals(boxes) == len(boxes)
243
244 def reachable(self, player, boxes):
245 seen = [False] * self.size
246 seen[player] = True
247 stack = [player]
248 while stack:
249 cell = stack.pop()
250 for offset in self.offsets.values():
251 neighbor = cell + offset
252 if self.interior[neighbor] and not seen[neighbor] and neighbor not in boxes:
253 seen[neighbor] = True
254 stack.append(neighbor)
255 return seen
256
257 def representative(self, player, boxes):
258 return self.reachable(player, boxes).index(True)
259
260 def walk_path(self, start, target, boxes):
261 came_from = [None] * self.size
262 queue = deque([start])
263 while queue:
264 cell = queue.popleft()
265 for direction in DIRECTIONS:
266 neighbor = cell + self.offsets[direction]
267 if (
268 self.interior[neighbor]
269 and neighbor not in boxes
270 and neighbor != start
271 and came_from[neighbor] is None
272 ):
273 came_from[neighbor] = (cell, direction)
274 queue.append(neighbor)
275 directions = []
276 cell = target
277 while cell != start:
278 cell, direction = came_from[cell]
279 directions.append(direction)
280 return "".join(reversed(directions))

282 def step(self, position, direction):
283 offset = self.offsets[direction]
284 target = position.player + offset
285 if self.walls[target]:
286 raise StepFailure("wall")
287 if target not in position.boxes:
288 return Position(target, position.boxes), direction
289 beyond = target + offset
290 if self.walls[beyond] or beyond in position.boxes:
291 raise StepFailure("blocked")
292 return Position(target, move_box(position.boxes, target, beyond)), direction.upper()
293
294 def pushable(self, box, direction, boxes, reachable):
295 offset = self.offsets[direction]
296 destination = box + offset
297 return (
298 reachable[box - offset]
299 and self.interior[destination]
300 and destination not in boxes
301 )
302
303 def side_blocked(self, box, sides, boxes, assumed):
304 if any(self.walls[side] or side in assumed for side in sides):
305 return True
306 if all(self.dead[side] for side in sides):
307 return True
308 extended = assumed + (box,)
309 return any(side in boxes and self.frozen(side, boxes, extended) for side in sides)
310
311 def frozen(self, box, boxes, assumed):
312 horizontal = (box + self.offsets["l"], box + self.offsets["r"])
313 vertical = (box + self.offsets["u"], box + self.offsets["d"])
314 return (
315 self.side_blocked(box, horizontal, boxes, assumed)
316 and self.side_blocked(box, vertical, boxes, assumed)
317 )
318
319 def has_frozen_box(self, boxes):
320 return any(not self.goals[box] and self.frozen(box, boxes, ()) for box in boxes)

322 def split_rows(self, symbols):

340 def dead_symbol(self, cell):
341 if self.walls[cell]:
342 return "#"
343 if self.goals[cell]:
344 return "."

347 def dead_map(self):
348 return self.split_rows([self.dead_symbol(cell) for cell in range(self.size)])

350 def map_cell(self, kind, cell):
351 row, column = divmod(cell, self.width)
352 last_row, last_column = self.height - 1, self.width - 1
353 if kind == "ROT90":
354 return column * self.height + last_row - row
355 if kind == "ROT180":
356 return (last_row - row) * self.width + last_column - column
357 if kind == "ROT270":
358 return (last_column - column) * self.height + row
359 if kind == "FLIPH":
360 return row * self.width + last_column - column
361 return (last_row - row) * self.width + column
362
363 def map_position(self, kind, position):
364 boxes = tuple(sorted(self.map_cell(kind, box) for box in position.boxes))
365 return Position(self.map_cell(kind, position.player), boxes)
366
367 def transformed(self, kind):
368 if kind in ("ROT90", "ROT270"):
369 height, width = self.width, self.height
370 else:
371 height, width = self.height, self.width
372 walls = [False] * self.size
373 goals = [False] * self.size
374 interior = [False] * self.size
375 for cell in range(self.size):
376 moved = self.map_cell(kind, cell)
377 walls[moved] = self.walls[cell]
378 goals[moved] = self.goals[cell]
379 interior[moved] = self.interior[cell]
380 level = Level(height, width, walls, goals, interior)
381 level.analyze()
382 return level

385def build_level(rows):
386 height, width = len(rows), len(rows[0])
387 cells = "".join(rows)
388 walls = [symbol == "#" for symbol in cells]
389 goals = [symbol in ".*+" for symbol in cells]
390 players = [cell for cell, symbol in enumerate(cells) if symbol in "@+"]
391 boxes = tuple(cell for cell, symbol in enumerate(cells) if symbol in "$*")
392 if len(players) == 1:
393 interior = find_interior(height, width, walls, players[0])
394 else:
395 interior = [False] * len(cells)
396 level = Level(height, width, walls, goals, interior)
397 level.reason = level.invalidity(len(players), boxes)
398 if level.reason is None:
399 level.analyze()
400 player = players[0] if players else None
401 return level, Position(player, boxes)

420def solution_pushes(nodes):
421 pushes = []
422 node = nodes[-1]
423 while node.parent is not None:
424 pushes.append(node.push)
425 node = nodes[node.parent]
426 pushes.reverse()
427 return pushes

430def solution_notation(level, position, pushes):
431 pieces = []
432 for box, direction in pushes:
433 offset = level.offsets[direction]
434 pieces.append(level.walk_path(position.player, box - offset, position.boxes))
435 pieces.append(direction.upper())
436 position = Position(box, move_box(position.boxes, box, box + offset))
437 return "".join(pieces)

440def push_search(level, start, fixed, target, pruning, limit):
441 first = SearchNode(
442 start.boxes, level.representative(start.player, start.boxes), None, None
443 )
444 nodes = [first]
445 known = {(first.boxes, first.representative)}
446 expanded = 0
447 while expanded < len(nodes):
448 if expanded == limit:
449 return SearchResult("limit", expanded, len(nodes))
450 node = nodes[expanded]
451 expanded += 1
452 reachable = level.reachable(node.representative, node.boxes)
453 for box in node.boxes:
454 if box in fixed:
455 continue
456 for direction in DIRECTIONS:
457 if not level.pushable(box, direction, node.boxes, reachable):
458 continue
459 destination = box + level.offsets[direction]
460 boxes = move_box(node.boxes, box, destination)
461 if pruning and (level.dead[destination] or level.has_frozen_box(boxes)):
462 continue
463 state = (boxes, level.representative(box, boxes))
464 if state in known:
465 continue
466 known.add(state)
467 nodes.append(SearchNode(boxes, state[1], expanded - 1, (box, direction)))
468 reached = level.solved(boxes) if target is None else destination == target
469 if reached:
470 notation = solution_notation(level, start, solution_pushes(nodes))
471 return SearchResult("found", expanded, len(nodes), notation)
472 return SearchResult("no solution", expanded, len(nodes))

522 def simulate(self, directions):
523 position = self.current
524 letters = []
525 try:
526 for direction in directions:
527 position, letter = self.level.step(position, direction)
528 letters.append(letter)
529 except StepFailure as failure:

547 def search_solution(self, limit):

604 def push(self, arguments):
605 level, position = self.level, self.current
606 box = self.cell_argument(arguments, 0)
607 direction = arguments[2]
608 offset = level.offsets[direction]
609 if box not in position.boxes:
610 raise CommandError("no box")
611 destination = box + offset
612 if not level.interior[destination] or destination in position.boxes:
613 raise CommandError("blocked")
614 if not level.reachable(position.player, position.boxes)[box - offset]:
615 raise CommandError("unreachable")
616 path = level.walk_path(position.player, box - offset, position.boxes)
617 letters, _ = self.take_steps(path + direction)
618 self.emit("pushed", "".join(letters))

620 def bring(self, arguments):
621 position = self.current
622 source = self.cell_argument(arguments, 0)
623 target = self.cell_argument(arguments, 2)
624 if source not in position.boxes:
625 raise CommandError("no box")
626 if source == target:
627 self.emit("brought", 0)
628 return
629 fixed = tuple(box for box in position.boxes if box != source)
630 result = push_search(self.level, position, fixed, target, False, None)
631 if result.outcome != "found":
632 raise CommandError("no route")
633 self.take_steps(result.notation.lower())
634 self.emit("brought", push_count(result.notation), result.notation)

644 def undo(self, arguments):
645 if not self.history:
646 raise CommandError("nothing to undo")
647 count = min(int(arguments[0]), len(self.history))
648 for _ in range(count):
649 letter = self.history.pop()
650 direction = letter.lower()
651 offset = self.level.offsets[direction]
652 player = self.current.player
653 boxes = self.current.boxes
654 if letter.isupper():
655 boxes = move_box(boxes, player + offset, player)
656 self.current = Position(player - offset, boxes)
657 self.redo_list.append(direction)
658 self.emit("undone", count)

665 direction = self.redo_list.pop()

667 self.history.append(letter)

711 def deadlock(self, arguments):
712 level, boxes = self.level, self.current.boxes
713 for box in boxes:
714 if level.dead[box]:
715 self.emit("deadlock", "dead", level.cell_text(box))
716 return
717 for box in boxes:
718 if not level.goals[box] and level.frozen(box, boxes, ()):
719 self.emit("deadlock", "frozen", level.cell_text(box))
720 return
721 self.emit("no deadlock")

723 def bound(self, arguments):
724 level, boxes = self.level, self.current.boxes
725 best = None
726 for order in permutations(range(len(level.goal_cells))):
727 pushes = [level.distances[goal][box] for box, goal in zip(boxes, order)]
728 if None in pushes:
729 continue
730 if best is None or sum(pushes) < sum(best[1]):
731 best = (order, pushes)
732 if best is None:
733 self.emit("bound", "none")
734 return
735 order, pushes = best
736 self.emit("bound", sum(pushes))
737 for box, goal, count in zip(boxes, order, pushes):
738 self.emit(
739 "box", level.cell_text(box), "goal", level.cell_text(level.goal_cells[goal]),
740 "pushes", count,
741 )

743 def save(self, arguments):
744 name = arguments[0]
745 self.snapshots[name] = (self.current, list(self.history))
746 self.emit("saved", name, "moves", len(self.history))
747
748 def restore(self, arguments):
749 name = arguments[0]
750 if name not in self.snapshots:
751 raise CommandError(f"no snapshot {name}")
752 position, history = self.snapshots[name]
753 self.current = position
754 self.history = list(history)
755 self.redo_list = []
756 self.emit("restored", name, "moves", len(self.history))

758 def check(self, arguments):
759 position, letters, failure = self.simulate(step_sequence(arguments[0]))
760 if failure is not None:
761 self.emit("check", "fail", failure)
762 return
763 self.emit(
764 "check", "ok", "moves", len(letters), "pushes", push_count(letters),
765 *self.progress(position.boxes),
766 )

768 def solve(self, arguments):
769 summary, notation = self.search_solution(int(arguments[0]))
770 self.emit(summary)
771 self.solution = None
772 if notation is not None:
773 self.emit(notation)
774 self.solution = (notation, self.current)
775
776 def hint(self, arguments):
777 summary, notation = self.search_solution(int(arguments[0]))
778 if notation is None:
779 self.emit(summary)
780 return
781 end = next(index for index, letter in enumerate(notation) if letter.isupper())
782 part = notation[:end + 1]
783 self.take_steps(part.lower())
784 self.emit("hint", part)
785
786 def play(self, arguments):
787 if self.solution is None or not self.solution[1].same(self.current):
788 raise CommandError("no solution")
789 notation = self.solution[0]
790 self.take_steps(notation.lower())
791 self.emit("played", "pushes", push_count(notation), "moves", len(notation))

793 def transform(self, arguments):
794 kind = arguments[0]
795 old = self.level
796 self.level = old.transformed(kind)
797 self.initial = old.map_position(kind, self.initial)
798 self.current = old.map_position(kind, self.current)
799 self.history = turn_letters(self.history, kind)
800 self.redo_list = turn_letters(self.redo_list, kind)
801 self.snapshots = {
802 name: (old.map_position(kind, position), turn_letters(history, kind))
803 for name, (position, history) in self.snapshots.items()
804 }
805 if self.solution is not None:
806 notation, position = self.solution
807 self.solution = (
808 "".join(turn_letters(notation, kind)), old.map_position(kind, position)
809 )
810 self.emit("transformed", f"{self.level.height}x{self.level.width}")

812 def check_values(self, name, arguments):
813 for index in range(0, 2 * CELL_COUNTS.get(name, 0), 2):
814 if not self.level.contains(int(arguments[index]), int(arguments[index + 1])):
815 raise CommandError("bad value")
816 if name in ("UNDO", "REDO") and int(arguments[0]) < 1:
817 raise CommandError("bad value")
818 if name in ("SOLVE", "HINT") and not 1 <= int(arguments[0]) <= MAX_LIMIT:
819 raise CommandError("bad value")
820
821 def run_command(self, name, arguments):
822 if name not in COMMANDS:
823 raise CommandError(f"unknown command {name}")
824 kinds = COMMANDS[name]
825 if len(arguments) != len(kinds) or not all(map(valid_argument, kinds, arguments)):
826 raise CommandError("bad arguments")
827 if name not in LEVEL_FREE_COMMANDS:
828 if self.level.reason is not None:
829 raise CommandError("invalid level")
830 self.check_values(name, arguments)
831 self.handlers[name](arguments)
832
833 def execute(self, words, rows):
834 try:
835 if words[0] == "LOAD":
836 self.load_rows(int(words[2]), rows)
837 else:
838 self.run_command(words[0], words[1:])
839 except CommandError as error:
840 self.emit("error:", error)
