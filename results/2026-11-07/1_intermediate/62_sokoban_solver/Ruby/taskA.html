2DIRECTIONS = %w[u d l r].freeze
3TURNS = { 'ROT90' => 'rlud', 'ROT180' => 'durl', 'ROT270' => 'lrdu',

5LEVEL_CHARACTERS = '#-.$*@+'
6STEP_CHARACTERS = 'udlrUDLR'
7COMMANDS = {

9 'MOVE' => [:steps], 'PUSH' => %i[number number direction],
10 'BRING' => %i[number number number number], 'PATH' => %i[number number],
11 'UNDO' => [:number], 'REDO' => [:number], 'RESET' => [], 'STATUS' => [],
12 'HISTORY' => [], 'BOXES' => [], 'REACH' => [], 'DEADSQUARES' => [],
13 'DEADLOCK' => [], 'BOUND' => [], 'SAVE' => [:name], 'RESTORE' => [:name],

24def decode(text, allowed)
25 result = +''
26 index = 0
27 while index < text.size
28 start = index
29 index += 1 while index < text.size && text[index].match?(/[0-9]/)
30 count = text[start...index]
31 return nil if count.size > 2 || count.start_with?('0') || index == text.size
32 return nil unless allowed.include?(text[index])
33
34 result << text[index] * (count.empty? ? 1 : count.to_i)
35 index += 1
36 end
37 result
38end

40def encode(text)
41 text.chars.chunk(&:itself).map { |letter, run| "#{run.size > 1 ? run.size : ''}#{letter}" }.join
42end

44def valid_argument?(kind, text)
45 case kind
46 when :number then /\A(?:0|[1-9][0-9]{0,5})\z/.match?(text)
47 when :steps then !decode(text, STEP_CHARACTERS).nil?
48 when :direction then DIRECTIONS.include?(text)
49 when :name then /\A[a-z]{1,8}\z/.match?(text)
50 when :transform then TURNS.key?(text)
51 else true
52 end
53end

55def push_count(letters) = letters.count('UDLR')
56def moved_boxes(boxes, source, destination) = boxes.map { |box| box == source ? destination : box }.sort

58def turn_letters(letters, kind)
59 letters.tr('udlrUDLR', TURNS.fetch(kind) + TURNS.fetch(kind).upcase)
60end

62class Level
63 attr_reader :height, :width, :walls, :goals, :interior, :goal_cells, :offsets, :distances, :dead, :reason
64
65 def initialize(rows)
66 @height, @width = rows.size, rows.first.size
67 cells = rows.join.chars
68 @walls = cells.map { |letter| letter == '#' }
69 @goals = cells.map { |letter| '.*+'.include?(letter) }
70 @goal_cells = cells.each_index.select { |cell| @goals[cell] }
71 players = cells.each_index.select { |cell| '@+'.include?(cells[cell]) }
72 boxes = cells.each_index.select { |cell| '$*'.include?(cells[cell]) }
73 @offsets = { 'u' => -@width, 'd' => @width, 'l' => -1, 'r' => 1 }
74 @interior = Array.new(cells.size, false)
75 if players.size == 1
76 @interior[players.first] = true
77 stack = [players.first]
78 until stack.empty?
79 row, column = stack.pop.divmod(@width)
80 [[row - 1, column], [row + 1, column], [row, column - 1], [row, column + 1]].each do |r, c|
81 next unless r.between?(0, @height - 1) && c.between?(0, @width - 1)
82
83 cell = r * @width + c
84 next if @walls[cell] || @interior[cell]
85
86 @interior[cell] = true
87 stack << cell
88 end
89 end
90 end
91 @reason = invalidity(players.size, boxes)
92 analyze if @reason.nil?
93 @loaded_position = Position.new(players.first, boxes)
94 end
95
96 def loaded_position = @loaded_position
97 def size = @height * @width
98 def cell_text(cell) = "#{cell / @width + 1} #{cell % @width + 1}"
99 def cell_at(row, column) = (row - 1) * @width + column - 1
100 def contains?(row, column) = row.between?(1, @height) && column.between?(1, @width)
101 def boxes_on_goals(boxes) = boxes.count { |box| @goals[box] }
102 def solved?(boxes) = boxes.all? { |box| @goals[box] }
103
104 def invalidity(players, boxes)
105 return "players #{players}" unless players == 1
106 return 'no boxes' if boxes.empty?
107 return "boxes #{boxes.size} goals #{@goal_cells.size}" unless boxes.size == @goal_cells.size
108
109 size.times do |cell|
110 row, column = cell.divmod(@width)
111 if @interior[cell] && (row.zero? || row == @height - 1 || column.zero? || column == @width - 1)
112 return "open #{cell_text(cell)}"
113 end
114 end
115 size.times do |cell|
116 return "outside #{cell_text(cell)}" if (@goals[cell] || boxes.include?(cell)) && !@interior[cell]
117 end
118 nil
119 end
120
121 def analyze
122 @goal_cells = (0...size).select { |cell| @goals[cell] }
123 @distances = @goal_cells.map do |goal|
124 distance = Array.new(size)
125 distance[goal] = 0
126 queue = [goal]
127 head = 0
128 while head < queue.size
129 cell = queue[head]
130 head += 1
131 @offsets.each_value do |offset|
132 box = cell - offset
133 next unless @interior[box] && @interior[box - offset] && distance[box].nil?
134
135 distance[box] = distance[cell] + 1
136 queue << box
137 end
138 end
139 distance
140 end
141 @dead = (0...size).map { |cell| @interior[cell] && @distances.all? { |distance| distance[cell].nil? } }
142 end
143
144 def reachable(player, boxes)
145 seen = Array.new(size, false)
146 seen[player] = true
147 queue = [player]
148 head = 0
149 while head < queue.size
150 cell = queue[head]
151 head += 1
152 @offsets.each_value do |offset|
153 neighbor = cell + offset
154 next unless @interior[neighbor] && !seen[neighbor] && !boxes.include?(neighbor)
155
156 seen[neighbor] = true
157 queue << neighbor
158 end
159 end
160 seen
161 end
162
163 def representative(player, boxes) = reachable(player, boxes).index(true)
164
165 def walk_path(start, target, boxes)
166 parents = Array.new(size)
167 queue = [start]
168 head = 0
169 while head < queue.size
170 cell = queue[head]
171 head += 1
172 @offsets.each do |direction, offset|
173 neighbor = cell + offset
174 next unless @interior[neighbor] && neighbor != start && parents[neighbor].nil? && !boxes.include?(neighbor)
175
176 parents[neighbor] = [cell, direction]
177 queue << neighbor
178 end
179 end
180 path = +''
181 until target == start
182 target, direction = parents[target]
183 path << direction
184 end
185 path.reverse
186 end
187
188 def step(position, direction)
189 offset = @offsets.fetch(direction)
190 target = position.player + offset
191 raise StepFailure, 'wall' if @walls[target]
192 return [Position.new(target, position.boxes), direction] unless position.boxes.include?(target)
193
194 beyond = target + offset
195 raise StepFailure, 'blocked' if @walls[beyond] || position.boxes.include?(beyond)
196
197 [Position.new(target, moved_boxes(position.boxes, target, beyond)), direction.upcase]
198 end
199
200 def pushable?(box, direction, boxes, reach)
201 offset = @offsets.fetch(direction)
202 reach[box - offset] && @interior[box + offset] && !boxes.include?(box + offset)
203 end
204
205 def side_blocked?(box, sides, boxes, assumed)
206 return true if sides.any? { |side| @walls[side] || assumed.include?(side) }
207 return true if sides.all? { |side| @dead[side] }
208
209 sides.any? { |side| boxes.include?(side) && frozen?(side, boxes, assumed + [box]) }
210 end
211
212 def frozen?(box, boxes, assumed = [])
213 side_blocked?(box, [box - 1, box + 1], boxes, assumed) &&
214 side_blocked?(box, [box - @width, box + @width], boxes, assumed)
215 end
216
217 def frozen_box?(boxes) = boxes.any? { |box| !@goals[box] && frozen?(box, boxes) }
218
219 def render(position = nil)
220 symbols = (0...size).map do |cell|
221 if @walls[cell] then '#'
222 elsif position.nil?
223 @goals[cell] ? '.' : (@dead[cell] ? 'x' : '-')
224 elsif cell == position.player then @goals[cell] ? '+' : '@'
225 elsif position.boxes.include?(cell) then @goals[cell] ? '*' : '$'
226 else @goals[cell] ? '.' : '-'
227 end
228 end
229 symbols.each_slice(@width).map(&:join)
230 end
231
232 def map_cell(kind, cell)
233 row, column = cell.divmod(@width)
234 case kind
235 when 'ROT90' then column * @height + @height - 1 - row
236 when 'ROT180' then (@height - 1 - row) * @width + @width - 1 - column
237 when 'ROT270' then (@width - 1 - column) * @height + row
238 when 'FLIPH' then row * @width + @width - 1 - column
239 else (@height - 1 - row) * @width + column
240 end
241 end
242
243 def map_position(kind, position)
244 Position.new(map_cell(kind, position.player), position.boxes.map { |box| map_cell(kind, box) }.sort)
245 end
246
247 def transformed(kind)
248 width = %w[ROT90 ROT270].include?(kind) ? @height : @width
249 symbols = Array.new(size)
250 render(@loaded_position).join.chars.each_with_index { |letter, cell| symbols[map_cell(kind, cell)] = letter }
251 Level.new(symbols.each_slice(width).map(&:join))
252 end
253end

255def push_search(level, start, fixed: [], target: nil, pruning: true, limit: nil)
256 first = SearchNode.new(start.boxes, level.representative(start.player, start.boxes), nil, nil)
257 nodes = [first]
258 known = { [first.boxes, first.representative] => true }
259 expanded = 0
260 while expanded < nodes.size
261 return SearchResult.new('limit', expanded, nodes.size, nil) if expanded == limit
262
263 parent = expanded
264 node = nodes[expanded]
265 expanded += 1
266 reach = level.reachable(node.representative, node.boxes)
267 node.boxes.each do |box|
268 next if fixed.include?(box)
269
270 DIRECTIONS.each do |direction|
271 next unless level.pushable?(box, direction, node.boxes, reach)
272
273 destination = box + level.offsets.fetch(direction)
274 boxes = moved_boxes(node.boxes, box, destination)
275 next if pruning && (level.dead[destination] || level.frozen_box?(boxes))
276
277 representative = level.representative(box, boxes)
278 key = [boxes, representative]
279 next if known.key?(key)
280
281 known[key] = true
282 nodes << SearchNode.new(boxes, representative, parent, [box, direction])
283 next unless (target.nil? ? level.solved?(boxes) : destination == target)
284
285 pushes = []
286 cursor = nodes.last
287 until cursor.parent.nil?
288 pushes << cursor.push
289 cursor = nodes[cursor.parent]
290 end
291 position = start
292 notation = +''
293 pushes.reverse_each do |source, step|
294 offset = level.offsets.fetch(step)
295 notation << level.walk_path(position.player, source - offset, position.boxes) << step.upcase
296 position = Position.new(source, moved_boxes(position.boxes, source, source + offset))
297 end
298 return SearchResult.new('found', expanded, nodes.size, notation)
299 end
300 end
301 end
302 SearchResult.new('no solution', expanded, nodes.size, nil)
303end

305class Sokoban
306 def initialize(rows) = load(rows)
307
308 def load(rows)
309 @level = Level.new(rows)
310 @initial = @current = @level.loaded_position
311 @history = +''
312 @redo = +''
313 @snapshots = {}
314 @solution = nil
315 end
316
317 def emit(*parts) = puts(parts.join(' '))
318 def cell(arguments, index = 0) = @level.cell_at(arguments[index].to_i, arguments[index + 1].to_i)
319
320 def simulate(directions)
321 position = @current
322 letters = +''
323 directions.each_char do |direction|
324 position, letter = @level.step(position, direction)
325 letters << letter
326 rescue StepFailure => error
327 return [position, letters, "step #{letters.size + 1} #{error.message}"]
328 end
329 [position, letters, nil]
330 end
331
332 def take_steps(directions)
333 @current, letters, failure = simulate(directions)
334 @history << letters
335 @redo.clear unless letters.empty?
336 [letters, failure]
337 end
338
339 def progress(boxes)
340 count = @level.boxes_on_goals(boxes)
341 "goals #{count}/#{boxes.size}#{count == boxes.size ? ' solved' : ''}"
342 end
343
344 def counted(label, letters)
345 letters.empty? ? emit(label, 0) : emit(label, letters.size, letters)
346 end
347
348 def search_solution(limit)
349 return ['already solved', nil] if @level.solved?(@current.boxes)
350
351 result = push_search(@level, @current, limit: limit)
352 counts = "expanded #{result.expanded} states #{result.states}"
353 return ["#{result.outcome} #{counts}", nil] unless result.outcome == 'found'
354
355 notation = result.notation
356 ["solution pushes #{push_count(notation)} moves #{notation.size} #{counts}", notation]
357 end

359 def execute(words, rows = [])
360 name, *arguments = words
361 if name == 'LOAD'
362 rows.each_with_index do |row, index|
363 raise CommandError, "bad row #{index + 1}" unless row.size == arguments[1].to_i && row.chars.all? { |letter| LEVEL_CHARACTERS.include?(letter) }
364 end
365 load(rows)
366 emit('loaded', "#{@level.height}x#{@level.width}")
367 return
368 end
369 raise CommandError, "unknown command #{name}" unless COMMANDS.key?(name)
370
371 kinds = COMMANDS.fetch(name)
372 unless arguments.size == kinds.size && kinds.zip(arguments).all? { |kind, argument| valid_argument?(kind, argument) }
373 raise CommandError, 'bad arguments'
374 end
375 unless %w[IMPORT VALID].include?(name)
376 raise CommandError, 'invalid level' unless @level.reason.nil?
377
378 ({ 'PATH' => 1, 'PUSH' => 1, 'BRING' => 2 }[name] || 0).times do |index|
379 raise CommandError, 'bad value' unless @level.contains?(arguments[index * 2].to_i, arguments[index * 2 + 1].to_i)
380 end
381 raise CommandError, 'bad value' if %w[UNDO REDO].include?(name) && arguments[0].to_i.zero?
382 raise CommandError, 'bad value' if %w[SOLVE HINT].include?(name) && !arguments[0].to_i.between?(1, 100_000)
383 end
384 process(name, arguments)
385 rescue CommandError => error
386 emit('error:', error.message)
387 end
388
389 def process(name, arguments)
390 level = @level
391 position = @current
392 case name
393 when 'IMPORT'
394 rows = arguments[0].split('|', -1).map { |part| decode(part, LEVEL_CHARACTERS) }
395 if rows.any?(&:nil?) || rows.size > 10 || !rows.first.size.between?(1, 10) || rows.any? { |row| row.size != rows.first.size }
396 raise CommandError, 'bad level'
397 end
398 load(rows)
399 emit('loaded', "#{@level.height}x#{@level.width}")
400 when 'VALID' then emit(level.reason.nil? ? 'valid' : "invalid #{level.reason}")
401 when 'INFO'
402 emit('size', "#{level.height}x#{level.width}", 'walls', level.walls.count(true), 'cells', level.interior.count(true), 'boxes', position.boxes.size, 'dead', level.dead.count(true))
403 when 'SHOW' then level.render(position).each { |row| emit(row) }
404 when 'EXPORT' then emit('level', level.render(position).map { |row| encode(row) }.join('|'))
405 when 'MOVE'
406 letters, failure = take_steps(decode(arguments[0], STEP_CHARACTERS).downcase)
407 emit('moved', letters) unless letters.empty?
408 raise CommandError, failure unless failure.nil?
409 when 'PUSH'
410 box, direction = cell(arguments), arguments[2]
411 offset = level.offsets.fetch(direction)
412 raise CommandError, 'no box' unless position.boxes.include?(box)
413 raise CommandError, 'blocked' unless level.interior[box + offset] && !position.boxes.include?(box + offset)
414 raise CommandError, 'unreachable' unless level.reachable(position.player, position.boxes)[box - offset]
415
416 letters, = take_steps(level.walk_path(position.player, box - offset, position.boxes) + direction)
417 emit('pushed', letters)
418 when 'BRING'
419 source, target = cell(arguments), cell(arguments, 2)
420 raise CommandError, 'no box' unless position.boxes.include?(source)
421 if source == target
422 emit('brought', 0)
423 else
424 result = push_search(level, position, fixed: position.boxes - [source], target: target, pruning: false)
425 raise CommandError, 'no route' unless result.outcome == 'found'
426
427 take_steps(result.notation.downcase)
428 emit('brought', push_count(result.notation), result.notation)
429 end
430 when 'PATH'
431 target = cell(arguments)
432 raise CommandError, 'unreachable' unless level.reachable(position.player, position.boxes)[target]
433
434 letters, = take_steps(level.walk_path(position.player, target, position.boxes))
435 counted('path', letters)
436 when 'UNDO'
437 raise CommandError, 'nothing to undo' if @history.empty?
438
439 count = [arguments[0].to_i, @history.size].min
440 count.times do
441 letter = @history.slice!(-1)
442 direction = letter.downcase
443 offset = level.offsets.fetch(direction)
444 boxes = position.boxes
445 boxes = moved_boxes(boxes, position.player + offset, position.player) if letter != direction
446 position = Position.new(position.player - offset, boxes)
447 @redo << direction
448 end
449 @current = position
450 emit('undone', count)
451 when 'REDO'
452 raise CommandError, 'nothing to redo' if @redo.empty?
453
454 count = [arguments[0].to_i, @redo.size].min
455 count.times do
456 @current, letter = level.step(@current, @redo.slice!(-1))
457 @history << letter
458 end
459 emit('redone', count)
460 when 'RESET'
461 @current = @initial
462 @history.clear
463 @redo.clear
464 emit('reset')
465 when 'STATUS' then emit('moves', @history.size, 'pushes', push_count(@history), progress(position.boxes))
466 when 'HISTORY' then counted('history', @history)
467 when 'BOXES'
468 reach = level.reachable(position.player, position.boxes)
469 emit('boxes', position.boxes.size)
470 position.boxes.each do |box|
471 distance = level.distances.filter_map { |distances| distances[box] }.min
472 directions = DIRECTIONS.select { |direction| level.pushable?(box, direction, position.boxes, reach) }.join
473 emit('box', level.cell_text(box), distance.nil? ? 'dead' : distance, directions.empty? ? '-' : directions)
474 end
475 when 'REACH'
476 reach = level.reachable(position.player, position.boxes)
477 emit('reach', reach.count(true), 'first', level.cell_text(reach.index(true)))
478 when 'DEADSQUARES'
479 emit('dead', level.dead.count(true))
480 level.render.each { |row| emit(row) }
481 when 'DEADLOCK'
482 box = position.boxes.find { |candidate| level.dead[candidate] }
483 if box
484 emit('deadlock dead', level.cell_text(box))
485 elsif (box = position.boxes.find { |candidate| !level.goals[candidate] && level.frozen?(candidate, position.boxes) })
486 emit('deadlock frozen', level.cell_text(box))
487 else emit('no deadlock')
488 end
489 when 'BOUND' then bound
490 when 'SAVE'
491 @snapshots[arguments[0]] = [position, @history.dup]
492 emit('saved', arguments[0], 'moves', @history.size)
493 when 'RESTORE'
494 snapshot = @snapshots[arguments[0]]
495 raise CommandError, "no snapshot #{arguments[0]}" if snapshot.nil?
496
497 @current, history = snapshot
498 @history = history.dup
499 @redo.clear
500 emit('restored', arguments[0], 'moves', @history.size)
501 when 'CHECK'
502 after, letters, failure = simulate(decode(arguments[0], STEP_CHARACTERS).downcase)
503 if failure then emit('check fail', failure)
504 else emit('check ok moves', letters.size, 'pushes', push_count(letters), progress(after.boxes))
505 end
506 when 'SOLVE', 'HINT'
507 summary, notation = search_solution(arguments[0].to_i)
508 if name == 'SOLVE'
509 emit(summary)
510 @solution = notation.nil? ? nil : [notation, position]
511 emit(notation) unless notation.nil?
512 elsif notation.nil? then emit(summary)
513 else
514 part = notation[/\A[udlr]*[UDLR]/]
515 take_steps(part.downcase)
516 emit('hint', part)
517 end
518 when 'PLAY'
519 raise CommandError, 'no solution' if @solution.nil? || @solution[1] != position
520
521 notation = @solution[0]
522 take_steps(notation.downcase)
523 emit('played pushes', push_count(notation), 'moves', notation.size)
524 when 'TRANSFORM'
525 kind = arguments[0]
526 @initial = level.map_position(kind, @initial)
527 @current = level.map_position(kind, position)
528 @history = turn_letters(@history, kind)
529 @redo = turn_letters(@redo, kind)
530 @snapshots.transform_values! { |saved, history| [level.map_position(kind, saved), turn_letters(history, kind)] }
531 @solution = [turn_letters(@solution[0], kind), level.map_position(kind, @solution[1])] unless @solution.nil?
532 @level = level.transformed(kind)
533 emit('transformed', "#{@level.height}x#{@level.width}")
534 end
535 end

537 def bound
538 boxes = @current.boxes
539 best = nil
540 (0...boxes.size).to_a.permutation do |order|
541 distances = boxes.zip(order).map { |box, goal| @level.distances[goal][box] }
542 next if distances.any?(&:nil?)
543 next if best && distances.sum >= best[1].sum
544
545 best = [order, distances]
546 end
547 if best.nil?
548 emit('bound none')
549 return
550 end
551 order, distances = best
552 emit('bound', distances.sum)
553 boxes.each_with_index do |box, index|
554 emit('box', @level.cell_text(box), 'goal', @level.cell_text(@level.goal_cells[order[index]]), 'pushes', distances[index])
555 end
556 end
557end

560rows = Array.new(height) { STDIN.readline.chomp }
561game = Sokoban.new(rows)
562STDIN.readline.to_i.times do
563 words = STDIN.readline.chomp.split(' ')
564 rows = words[0] == 'LOAD' ? Array.new(words[1].to_i) { STDIN.readline.chomp } : []
565 game.execute(words, rows)
