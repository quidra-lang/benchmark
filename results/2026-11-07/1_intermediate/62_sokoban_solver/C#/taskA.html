25sealed class Level
26{
27 public int Height, Width;
28 public int Size => Height * Width;
29 public bool[] Walls, Goals, Interior, Dead;
30 public int[] GoalCells, Offsets;
31 public int[][] Distances = Array.Empty<int[]>();
32 public string? Reason;
33 public Level(int height, int width, bool[] walls, bool[] goals, bool[] interior)
34 {
35 Height = height;
36 Width = width;
37 Walls = walls;
38 Goals = goals;
39 Interior = interior;
40 Offsets = new[] { -width, width, -1, 1 };
41 GoalCells = Enumerable.Range(0, Size).Where(c => goals[c]).ToArray();
42 Dead = new bool[Size];
43 }
44 public string Cell(int c) => $"{c / Width + 1} {c % Width + 1}";
45 public int At(int row, int column) => (row - 1) * Width + column - 1;
46 public bool Contains(int row, int column) => row >= 1 && row <= Height
47 && column >= 1 && column <= Width;
48 public bool Inside(int c) => c >= 0 && c < Size && Interior[c];
49 public int Neighbor(int c, int d)
50 {
51 int row = c / Width, col = c % Width;
52 if (d == 0)
53 row--;
54 else if (d == 1)
55 row++;
56 else if (d == 2)
57 col--;
58 else
59 col++;
60 return row < 0 || row >= Height || col < 0 || col >= Width ? -1 : row * Width + col;
61 }
62 public void Analyze()
63 {
64 Distances = new int [GoalCells.Length][];
65 for (int g = 0; g < GoalCells.Length; g++)
66 {
67 int[] distance = Enumerable.Repeat(-1, Size).ToArray();
68 var queue = new Queue<int>();
69 queue.Enqueue(GoalCells[g]);
70 distance[GoalCells[g]] = 0;
71 while (queue.Count > 0)
72 {
73 int q = queue.Dequeue();
74 foreach (int offset in Offsets)
75 {
76 int box = q - offset, stand = box - offset;
77 if (Inside(box) && Inside(stand) && distance[box] < 0)
78 {
79 distance[box] = distance[q] + 1;
80 queue.Enqueue(box);
81 }
82 }
83 }
84 Distances[g] = distance;
85 }
86 for (int c = 0; c < Size; c++)
87 Dead[c] = Interior[c] && Distances.All(distance => distance[c] < 0);
88 }
89 public bool[] Reach(Position p)
90 {
91 bool[] seen = new bool[Size];
92 seen[p.Player] = true;
93 var queue = new Queue<int>();
94 queue.Enqueue(p.Player);
95 while (queue.Count > 0)
96 {
97 int c = queue.Dequeue();
98 for (int d = 0; d < 4; d++)
99 {
100 int n = Neighbor(c, d);
101 if (Inside(n) && !seen[n] && !p.Boxes.Contains(n))
102 {
103 seen[n] = true;
104 queue.Enqueue(n);
105 }
106 }
107 }
108 return seen;
109 }
110 public int Representative(Position p) => Array.IndexOf(Reach(p), true);
111 public string Walk(Position p, int target)
112 {
113 int[] previous = Enumerable.Repeat(-1, Size).ToArray(), arrived = new int[Size];
114 var queue = new Queue<int>();
115 queue.Enqueue(p.Player);
116 previous[p.Player] = p.Player;
117 while (queue.Count > 0)
118 {
119 int c = queue.Dequeue();
120 for (int d = 0; d < 4; d++)
121 {
122 int n = Neighbor(c, d);
123 if (Inside(n) && !p.Boxes.Contains(n) && previous[n] < 0)
124 {
125 previous[n] = c;
126 arrived[n] = d;
127 queue.Enqueue(n);
128 }
129 }
130 }
131 var reverse = new List<char>();
132 for (int c = target; c != p.Player; c = previous[c])
133 reverse.Add(Sokoban.Directions[arrived[c]]);
134 reverse.Reverse();
135 return new string(reverse.ToArray());
136 }
137 public (Position Position, char Letter) Step(Position p, char direction)
138 {
139 int d = Sokoban.Directions.IndexOf(char.ToLowerInvariant(direction)), offset = Offsets[d];
140 int target = p.Player + offset;
141 if (Walls[target])
142 throw new StepFailure("wall");
143 char letter = Sokoban.Directions[d];
144 if (p.Boxes.Contains(target))
145 {
146 int beyond = target + offset;
147 if (Walls[beyond] || p.Boxes.Contains(beyond))
148 throw new StepFailure("blocked");
149 p = p.MoveBox(target, beyond);
150 letter = char.ToUpperInvariant(letter);
151 }
152 return (p with { Player = target }, letter);
153 }
154 public int OnGoals(Position p) => p.Boxes.Count(b => Goals[b]);
155 public bool Solved(Position p) => OnGoals(p) == p.Boxes.Length;
156 public bool Pushable(Position p, int box, int d, bool[] reachable)
157 {
158 int stand = box - Offsets[d], target = box + Offsets[d];
159 return Inside(stand) && reachable[stand] && Inside(target) && !p.Boxes.Contains(target);
160 }
161 bool AxisBlocked(Position p, int box, int a, int b, int assumed)
162 {
163 int ia = Array.IndexOf(p.Boxes, a), ib = Array.IndexOf(p.Boxes, b);
164 if (Walls[a] || Walls[b] || (ia >= 0 && (assumed & (1 << ia)) != 0) ||
165 (ib >= 0 && (assumed & (1 << ib)) != 0))
166 return true;
167 if (Dead[a] && Dead[b])
168 return true;
169 int more = assumed | (1 << Array.IndexOf(p.Boxes, box));
170 return (ia >= 0 && Frozen(p, a, more)) || (ib >= 0 && Frozen(p, b, more));
171 }
172 public bool Frozen(Position p, int box,
173 int assumed = 0) => AxisBlocked(p, box, box - 1, box + 1, assumed) &&
174 AxisBlocked(p, box, box - Width, box + Width, assumed);
175 public bool HasFrozen(Position p) => p.Boxes.Any(b => !Goals[b] && Frozen(p, b));
176 public char Symbol(Position p, int c) => Walls[c] ? '#'
177 : c == p.Player ? (Goals[c] ? '+' : '@')
178 : p.Boxes.Contains(c) ? (Goals[c] ? '*' : '$')
179 : Goals[c] ? '.'
180 : '-';
181 public IEnumerable<string> Render(Position p, bool deadMap = false)
182 {
183 for (int r = 0; r < Height; r++)
184 {
185 char[] row = new char[Width];
186 for (int c = 0; c < Width; c++)
187 {
188 int index = r * Width + c;
189 row[c] = deadMap ? (Walls[index] ? '#'
190 : Goals[index] ? '.'
191 : Dead[index] ? 'x'
192 : '-')
193 : Symbol(p, index);
194 }
195 yield return new string(row);
196 }
197 }
198 public int MapCell(int c, int transform)
199 {
200 int r = c / Width, col = c % Width;
201 return transform switch {
202 0 => col * Height + Height - 1 - r,
203 1 => (Height - 1 - r) * Width + Width - 1 - col,
204 2 => (Width - 1 - col) * Height + r,
205 3 => r * Width + Width - 1 - col,
206 _ => (Height - 1 - r) * Width + col
207 };
208 }
209 public Position MapPosition(Position p, int transform) =>
210 new(MapCell(p.Player, transform),
211 p.Boxes.Select(b => MapCell(b, transform)).OrderBy(b => b).ToArray());
212 public Level Transform(int transform)
213 {
214 bool[] walls = new bool[Size], goals = new bool[Size], interior = new bool[Size];
215 for (int c = 0; c < Size; c++)
216 {
217 int next = MapCell(c, transform);
218 walls[next] = Walls[c];
219 goals[next] = Goals[c];
220 interior[next] = Interior[c];
221 }
222 bool rotate = transform == 0 || transform == 2;
223 var level =
224 new Level(rotate ? Width : Height, rotate ? Height : Width, walls, goals, interior);
225 level.Analyze();
226 return level;
227 }
228}

230sealed class Sokoban
231{
232 public const string Directions = "udlr";
233 const string Symbols = "#-.$*@+";
234 static readonly string[] Turns = { "ROT90", "ROT180", "ROT270", "FLIPH", "FLIPV" };
235 static readonly string[] DirectionMaps = { "rlud", "durl", "lrdu", "udrl", "dulr" };
236 static readonly Dictionary<string, string> Formats =
237 new() { ["IMPORT"] = "x", ["VALID"] = "", ["INFO"] = "", ["SHOW"] = "",
238 ["EXPORT"] = "", ["MOVE"] = "s", ["PUSH"] = "nnd", ["BRING"] = "nnnn",
239 ["PATH"] = "nn", ["UNDO"] = "n", ["REDO"] = "n", ["RESET"] = "",
240 ["STATUS"] = "", ["HISTORY"] = "", ["BOXES"] = "", ["REACH"] = "",
241 ["DEADSQUARES"] = "", ["DEADLOCK"] = "", ["BOUND"] = "", ["SAVE"] = "a",
242 ["RESTORE"] = "a", ["CHECK"] = "s", ["SOLVE"] = "n", ["HINT"] = "n",
243 ["PLAY"] = "", ["TRANSFORM"] = "t" };
244 Level level = null!;
245 Position initial = null!, current = null!;
246 StringBuilder history = new(), redo = new();
247 readonly Dictionary<string, Snapshot> snapshots = new();
248 StoredSolution? solution;
249 public Sokoban(string[] rows) => Load(rows);
250 static string? Decode(string word, string allowed)
251 {
252 var output = new StringBuilder();
253 for (int i = 0; i < word.Length;)
254 {
255 int start = i, count = 0;
256 while (i < word.Length && word[i] >= '0' && word[i] <= '9')
257 {
258 if (i - start < 3)
259 count = count * 10 + word[i] - '0';
260 i++;
261 }
262 int digits = i - start;
263 if (digits > 2 || (digits > 0 && word[start] == '0') || i == word.Length ||
264 !allowed.Contains(word[i]))
265 return null;
266 output.Append(word[i++], digits == 0 ? 1 : count);
267 }
268 return output.ToString();
269 }
270 static string Encode(string row)
271 {
272 var output = new StringBuilder();
273 for (int i = 0; i < row.Length;)
274 {
275 int end = i + 1;
276 while (end < row.Length && row[end] == row[i])
277 end++;
278 if (end - i > 1)
279 output.Append(end - i);
280 output.Append(row[i]);
281 i = end;
282 }
283 return output.ToString();
284 }
285 static bool Argument(char kind, string s) => kind switch {
286 'n' => s.Length >= 1 && s.Length <= 6 && (s.Length == 1 || s[0] != '0') &&
287 s.All(c => c >= '0' && c <= '9'),
288 's' => Decode(s, "udlrUDLR") != null,
289 'd' => s.Length == 1 && Directions.Contains(s[0]),
290 'a' => s.Length >= 1 && s.Length <= 8 && s.All(c => c >= 'a' && c <= 'z'),
291 't' => Turns.Contains(s),
292 _ => true
293 };
294 static int Pushes(string s) => s.Count(c => c >= 'A' && c <= 'Z');
295 static string Counted(string label, string s) => $"{label} {s.Length}" +
296 (s.Length > 0 ? $" {s}" : "");
297 static void Emit(string line) => Console.WriteLine(line);
298 static void Require(bool condition, string message)
299 {
300 if (!condition)
301 throw new CommandError(message);
302 }
303 string Progress(Position p) => $"goals {level.OnGoals(p)}/{p.Boxes.Length}" +
304 (level.Solved(p) ? " solved" : "");
305 void Load(string[] rows)
306 {
307 int h = rows.Length, w = rows[0].Length, players = 0, player = -1;
308 bool[] walls = new bool[h * w], goals = new bool[h * w], interior = new bool[h * w];
309 var boxes = new List<int>();
310 for (int c = 0; c < h * w; c++)
311 {
312 char ch = rows[c / w][c % w];
313 walls[c] = ch == '#';
314 goals[c] = ".*+".Contains(ch);
315 if (ch == '@' || ch == '+')
316 {
317 players++;
318 player = c;
319 }
320 if (ch == '$' || ch == '*')
321 boxes.Add(c);
322 }
323 level = new Level(h, w, walls, goals, interior);
324 initial = current = new Position(player, boxes.ToArray());
325 if (players == 1)
326 {
327 var queue = new Queue<int>();
328 queue.Enqueue(player);
329 interior[player] = true;
330 while (queue.Count > 0)
331 {
332 int c = queue.Dequeue();
333 for (int d = 0; d < 4; d++)
334 {
335 int n = level.Neighbor(c, d);
336 if (n >= 0 && !walls[n] && !interior[n])
337 {
338 interior[n] = true;
339 queue.Enqueue(n);
340 }
341 }
342 }
343 }
344 if (players != 1)
345 level.Reason = $"players {players}";
346 else if (boxes.Count == 0)
347 level.Reason = "no boxes";
348 else if (boxes.Count != level.GoalCells.Length)
349 level.Reason = $"boxes {boxes.Count} goals {level.GoalCells.Length}";
350 else
351 {
352 for (int c = 0; c < h * w; c++)
353 if (interior[c] && (c / w == 0 || c / w == h - 1 || c % w == 0 || c % w == w - 1))
354 {
355 level.Reason = "open " + level.Cell(c);
356 break;
357 }
358 if (level.Reason == null)
359 for (int c = 0; c < h * w; c++)
360 if ((goals[c] || boxes.Contains(c)) && !interior[c])
361 {
362 level.Reason = "outside " + level.Cell(c);
363 break;
364 }
365 }
366 if (level.Reason == null)
367 level.Analyze();
368 history.Clear();
369 redo.Clear();
370 snapshots.Clear();
371 solution = null;
372 }
373 (Position Position, string Letters, string? Failure) Simulate(string sequence)
374 {
375 Position p = current;
376 var letters = new StringBuilder();
377 try
378 {
379 foreach (char direction in sequence)
380 {
381 var next = level.Step(p, direction);
382 p = next.Position;
383 letters.Append(next.Letter);
384 }
385 }
386 catch (StepFailure failure)
387 {
388 return (p, letters.ToString(), $"step {letters.Length + 1} {failure.Message}");
389 }
390 return (p, letters.ToString(), null);
391 }
392 (string Letters, string? Failure) Execute(string sequence)
393 {
394 var result = Simulate(sequence);
395 current = result.Position;
396 history.Append(result.Letters);
397 if (result.Letters.Length > 0)
398 redo.Clear();
399 return (result.Letters, result.Failure);
400 }
401 static ulong Key(Position p)
402 {
403 ulong key = (ulong)(p.Player + 1);
404 foreach (int b in p.Boxes)
405 key = (key << 7) | (ulong)(b + 1);
406 return key;
407 }
408 SearchResult Search(int source, int target, bool prune, int? limit)
409 {
410 Position first = current with { Player = level.Representative(current) };
411 var nodes = new List<SearchNode> { new(first, -1, 0, 0) };
412 var known = new HashSet<ulong> { Key(first) };
413 var fixedBoxes =
414 source < 0 ? new HashSet<int>() : current.Boxes.Where(b => b != source).ToHashSet();
415 int expanded = 0;
416 while (expanded < nodes.Count)
417 {
418 if (expanded == limit)
419 return new("limit", expanded, nodes.Count);
420 int parent = expanded++;
421 SearchNode node = nodes[parent];
422 bool[] reachable = level.Reach(node.Position);
423 foreach (int box in node.Position.Boxes)
424 {
425 if (fixedBoxes.Contains(box))
426 continue;
427 for (int d = 0; d < 4; d++)
428 {
429 if (!level.Pushable(node.Position, box, d, reachable))
430 continue;
431 int destination = box + level.Offsets[d];
432 Position next = node.Position.MoveBox(box, destination) with { Player = box };
433 if (prune && (level.Dead[destination] || level.HasFrozen(next)))
434 continue;
435 next = next with { Player = level.Representative(next) };
436 if (!known.Add(Key(next)))
437 continue;
438 nodes.Add(new(next, parent, box, d));
439 if (source >= 0 ? destination == target : level.Solved(next))
440 {
441 var pushes = new List<SearchNode>();
442 for (int n = nodes.Count - 1; nodes[n].Parent >= 0; n = nodes[n].Parent)
443 pushes.Add(nodes[n]);
444 pushes.Reverse();
445 Position p = current;
446 var notation = new StringBuilder();
447 foreach (SearchNode push in pushes)
448 {
449 int offset = level.Offsets[push.Direction];
450 notation.Append(level.Walk(p, push.Box - offset));
451 notation.Append(char.ToUpperInvariant(Directions[push.Direction]));
452 p = p.MoveBox(push.Box, push.Box + offset) with { Player = push.Box };
453 }
454 return new("found", expanded, nodes.Count, notation.ToString());
455 }
456 }
457 }
458 }
459 return new("no solution", expanded, nodes.Count);
460 }
461 void Bound()
462 {
463 int[] order = new int[current.Boxes.Length];
464 int[]? best = null;
465 int minimum = int.MaxValue;
466 void Visit(int depth, int used, int total)
467 {
468 if (depth == order.Length)
469 {
470 if (total < minimum)
471 {
472 minimum = total;
473 best = (int[])order.Clone();
474 }
475 return;
476 }
477 for (int g = 0; g < level.GoalCells.Length; g++)
478 {
479 int distance = level.Distances[g][current.Boxes[depth]];
480 if ((used & (1 << g)) != 0 || distance < 0)
481 continue;
482 order[depth] = g;
483 Visit(depth + 1, used | (1 << g), total + distance);
484 }
485 }
486 Visit(0, 0, 0);
487 if (best == null)
488 {
489 Emit("bound none");
490 return;
491 }
492 Emit($"bound {minimum}");
493 for (int i = 0; i < best.Length; i++)
494 Emit(
495 $"box {level.Cell(current.Boxes[i])} goal {level.Cell(level.GoalCells[best[i]])} pushes {level.Distances[best[i]][current.Boxes[i]]}");
496 }
497 static string TurnLetters(string letters, int transform) => new(
498 letters
499 .Select(c =>
500 {
501 char d =
502 DirectionMaps[transform][Directions.IndexOf(char.ToLowerInvariant(c))];
503 return char.IsUpper(c) ? char.ToUpperInvariant(d) : d;
504 })
505 .ToArray());
506 void Transform(int transform)
507 {
508 Level old = level;
509 level = old.Transform(transform);
510 initial = old.MapPosition(initial, transform);
511 current = old.MapPosition(current, transform);
512 history = new StringBuilder(TurnLetters(history.ToString(), transform));
513 redo = new StringBuilder(TurnLetters(redo.ToString(), transform));
514 foreach (string name in snapshots.Keys.ToArray())
515 {
516 Snapshot saved = snapshots[name];
517 snapshots[name] = new(old.MapPosition(saved.Position, transform),
518 TurnLetters(saved.History, transform));
519 }
520 if (solution != null)
521 solution = new(old.MapPosition(solution.Position, transform),
522 TurnLetters(solution.Notation, transform));
523 Emit($"transformed {level.Height}x{level.Width}");
524 }
525 public void Process(string[] words, string[]? rows = null)

527 try
528 {

531 catch (CommandError error)

536 void Run(string[] words, string[]? rows)
537 {
538 string command = words[0];
539 string[] args = words.Skip(1).ToArray();
540 if (command == "LOAD")
541 {
542 int width = int.Parse(args[1]);
543 for (int i = 0; i < rows!.Length; i++)
544 Require(rows[i].Length == width && rows[i].All(Symbols.Contains), $"bad row {i+1}");
545 Load(rows);
546 Emit($"loaded {rows.Length}x{width}");
547 return;
548 }
549 Require(Formats.ContainsKey(command), "unknown command " + command);
550 string format = Formats[command];
551 Require(args.Length == format.Length, "bad arguments");
552 for (int i = 0; i < args.Length; i++)
553 Require(Argument(format[i], args[i]), "bad arguments");
554 if (command == "IMPORT")
555 {
556 string?[] decoded = args[0].Split('|').Select(part => Decode(part, Symbols)).ToArray();
557 Require(decoded.Length <= 10 && decoded.All(row => row != null), "bad level");
558 int width = decoded[0]!.Length;
559 Require(width >= 1 && width <= 10 && decoded.All(row => row!.Length == width),
560 "bad level");
561 Load(decoded.Select(row => row!).ToArray());
562 Emit($"loaded {decoded.Length}x{width}");
563 return;
564 }
565 if (command == "VALID")
566 {
567 Emit(level.Reason == null ? "valid" : "invalid " + level.Reason);
568 return;
569 }
570 Require(level.Reason == null, "invalid level");
571 int cellCount = command == "BRING" ? 2 : command == "PATH" || command == "PUSH" ? 1 : 0;
572 for (int i = 0; i < 2 * cellCount; i += 2)
573 Require(level.Contains(int.Parse(args[i]), int.Parse(args[i + 1])), "bad value");
574 if (command == "UNDO" || command == "REDO")
575 Require(int.Parse(args[0]) >= 1, "bad value");
576 if (command == "SOLVE" || command == "HINT")
577 Require(int.Parse(args[0]) >= 1 && int.Parse(args[0]) <= 100000, "bad value");
578 int box = cellCount > 0 ? level.At(int.Parse(args[0]), int.Parse(args[1])) : 0;
579 switch (command)
580 {
581 case "INFO":
582 Emit(
583 $"size {level.Height}x{level.Width} walls {level.Walls.Count(x=>x)} cells {level.Interior.Count(x=>x)} boxes {current.Boxes.Length} dead {level.Dead.Count(x=>x)}");
584 break;
585 case "SHOW":
586 foreach (string row in level.Render(current))
587 Emit(row);
588 break;
589 case "EXPORT":
590 Emit("level " + string.Join("|", level.Render(current).Select(Encode)));
591 break;
592 case "MOVE": {
593 var result = Execute(Decode(args[0], "udlrUDLR")!);
594 if (result.Letters.Length > 0)
595 Emit("moved " + result.Letters);
596 if (result.Failure != null)
597 throw new CommandError(result.Failure);
598 break;
599 }
600 case "CHECK": {
601 var result = Simulate(Decode(args[0], "udlrUDLR")!);
602 Emit(
603 result.Failure != null
604 ? "check fail " + result.Failure
605 : $"check ok moves {result.Letters.Length} pushes {Pushes(result.Letters)} {Progress(result.Position)}");
606 break;
607 }
608 case "PUSH": {
609 int d = Directions.IndexOf(args[2][0]), offset = level.Offsets[d];
610 Require(current.Boxes.Contains(box), "no box");
611 Require(level.Inside(box + offset) && !current.Boxes.Contains(box + offset), "blocked");
612 bool[] reachable = level.Reach(current);
613 Require(level.Inside(box - offset) && reachable[box - offset], "unreachable");
614 Emit("pushed " + Execute(level.Walk(current, box - offset) + Directions[d]).Letters);
615 break;
616 }
617 case "BRING": {
618 int target = level.At(int.Parse(args[2]), int.Parse(args[3]));
619 Require(current.Boxes.Contains(box), "no box");
620 if (box == target)
621 {
622 Emit("brought 0");
623 break;
624 }
625 SearchResult result = Search(box, target, false, null);
626 Require(result.Outcome == "found", "no route");
627 Execute(result.Notation);
628 Emit($"brought {Pushes(result.Notation)} {result.Notation}");
629 break;
630 }
631 case "PATH":
632 Require(level.Reach(current)[box], "unreachable");
633 Emit(Counted("path", Execute(level.Walk(current, box)).Letters));
634 break;
635 case "UNDO": {
636 Require(history.Length > 0, "nothing to undo");
637 int count = Math.Min(int.Parse(args[0]), history.Length);
638 for (int i = 0; i < count; i++)
639 {
640 char letter = history[history.Length - 1],
641 direction = char.ToLowerInvariant(letter);
642 history.Length--;
643 int offset = level.Offsets[Directions.IndexOf(direction)];
644 if (char.IsUpper(letter))
645 current = current.MoveBox(current.Player + offset, current.Player);
646 current = current with { Player = current.Player - offset };
647 redo.Append(direction);
648 }
649 Emit($"undone {count}");
650 break;
651 }
652 case "REDO": {
653 Require(redo.Length > 0, "nothing to redo");
654 int count = Math.Min(int.Parse(args[0]), redo.Length);
655 for (int i = 0; i < count; i++)
656 {
657 char direction = redo[redo.Length - 1];
658 redo.Length--;
659 var next = level.Step(current, direction);
660 current = next.Position;
661 history.Append(next.Letter);
662 }
663 Emit($"redone {count}");
664 break;
665 }
666 case "RESET":
667 current = initial;
668 history.Clear();
669 redo.Clear();
670 Emit("reset");
671 break;
672 case "STATUS":
673 Emit($"moves {history.Length} pushes {Pushes(history.ToString())} {Progress(current)}");
674 break;
675 case "HISTORY":
676 Emit(Counted("history", history.ToString()));
677 break;
678 case "BOXES": {
679 bool[] reachable = level.Reach(current);
680 Emit($"boxes {current.Boxes.Length}");
681 foreach (int b in current.Boxes)
682 {
683 int[] distances = level.Distances.Select(d => d[b]).Where(d => d >= 0).ToArray();
684 string possible = new(Enumerable.Range(0, 4)
685 .Where(d => level.Pushable(current, b, d, reachable))
686 .Select(d => Directions[d])
687 .ToArray());
688 Emit(
689 $"box {level.Cell(b)} {(distances.Length > 0 ? distances.Min().ToString() : "dead")} {(possible.Length > 0 ? possible : "-")}");
690 }
691 break;
692 }
693 case "REACH": {
694 bool[] seen = level.Reach(current);
695 Emit($"reach {seen.Count(x=>x)} first {level.Cell(Array.IndexOf(seen,true))}");
696 break;
697 }
698 case "DEADSQUARES":
699 Emit($"dead {level.Dead.Count(x=>x)}");
700 foreach (string row in level.Render(current, true))
701 Emit(row);
702 break;
703 case "DEADLOCK": {
704 int dead = -1, frozen = -1;
705 foreach (int b in current.Boxes)
706 if (level.Dead[b])
707 {
708 dead = b;
709 break;
710 }
711 if (dead < 0)
712 foreach (int b in current.Boxes)
713 if (!level.Goals[b] && level.Frozen(current, b))
714 {
715 frozen = b;
716 break;
717 }
718 if (dead >= 0)
719 Emit("deadlock dead " + level.Cell(dead));
720 else if (frozen >= 0)
721 Emit("deadlock frozen " + level.Cell(frozen));
722 else
723 Emit("no deadlock");
724 break;
725 }
726 case "BOUND":
727 Bound();
728 break;
729 case "SAVE":
730 snapshots[args[0]] = new(current, history.ToString());
731 Emit($"saved {args[0]} moves {history.Length}");
732 break;
733 case "RESTORE":
734 Require(snapshots.ContainsKey(args[0]), "no snapshot " + args[0]);
735 current = snapshots[args[0]].Position;
736 history = new StringBuilder(snapshots[args[0]].History);
737 redo.Clear();
738 Emit($"restored {args[0]} moves {history.Length}");
739 break;
740 case "SOLVE":
741 case "HINT": {
742 if (command == "SOLVE")
743 solution = null;
744 if (level.Solved(current))
745 {
746 Emit("already solved");
747 break;
748 }
749 SearchResult result = Search(-1, -1, true, int.Parse(args[0]));
750 if (result.Outcome != "found")
751 {
752 Emit($"{result.Outcome} expanded {result.Expanded} states {result.States}");
753 break;
754 }
755 if (command == "SOLVE")
756 {
757 Emit(
758 $"solution pushes {Pushes(result.Notation)} moves {result.Notation.Length} expanded {result.Expanded} states {result.States}");
759 Emit(result.Notation);
760 solution = new(current, result.Notation);
761 }
762 else
763 {
764 int end = 0;
765 while (!char.IsUpper(result.Notation[end]))
766 end++;
767 string part = result.Notation[..(end + 1)];
768 Execute(part);
769 Emit("hint " + part);
770 }
771 break;
772 }
773 case "PLAY":
774 Require(solution != null && solution.Position.Same(current), "no solution");
775 Execute(solution!.Notation);
776 Emit($"played pushes {Pushes(solution.Notation)} moves {solution.Notation.Length}");
777 break;
778 case "TRANSFORM":
779 Transform(Array.IndexOf(Turns, args[0]));
780 break;
781 }
782 }
783}

787 {
788 var rows = new string[count];
789 for (int i = 0; i < count; i++)
790 rows[i] = Console.ReadLine()!;
791 return rows;
792 }

794 {
795 string? first = Console.ReadLine();
796 if (first == null)
797 return;
798 int height = int.Parse(first.Split(' ')[0]);
799 var game = new Sokoban(ReadRows(height));
800 int count = int.Parse(Console.ReadLine()!);
801 for (int i = 0; i < count; i++)
802 {
803 string[] words = Console.ReadLine()!.Split(' ');
804 string[]? rows = words[0] == "LOAD" ? ReadRows(int.Parse(words[1])) : null;
805 game.Process(words, rows);
806 }
