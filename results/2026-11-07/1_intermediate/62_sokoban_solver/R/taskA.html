66decode <- function(text, allowed) {
67 pieces <- character(0)
68 index <- 1
69 while (index <= nchar(text)) {
70 start <- index
71 while (index <= nchar(text) && substr(text, index, index) %in% DIGITS) {
72 index <- index + 1
73 }
74 count <- substr(text, start, index - 1)
75 if (nchar(count) > 2 || startsWith(count, "0") || index > nchar(text)) {
76 return(NULL)
77 }
78 symbol <- substr(text, index, index)
79 if (!(symbol %in% allowed)) {
80 return(NULL)
81 }
82 pieces <- c(pieces, strrep(symbol, if (count == "") 1 else as.integer(count)))
83 index <- index + 1
84 }
85 paste(pieces, collapse = "")
86}

153find_interior <- function(height, width, walls, start) {
154 interior <- logical(height * width)
155 interior[start] <- TRUE
156 stack <- start
157 while (length(stack) > 0) {
158 cell <- stack[length(stack)]
159 stack <- stack[-length(stack)]
160 row <- (cell - 1) %/% width
161 column <- (cell - 1) %% width
162 for (delta in DELTAS) {
163 next_row <- row + delta[1]
164 next_column <- column + delta[2]
165 if (next_row >= 0 && next_row < height && next_column >= 0 && next_column < width) {
166 neighbor <- next_row * width + next_column + 1
167 if (!walls[neighbor] && !interior[neighbor]) {
168 interior[neighbor] <- TRUE
169 stack <- c(stack, neighbor)
170 }
171 }
172 }
173 }
174 interior
175}

219invalidity <- function(level, players, boxes) {
220 if (players != 1) {
221 return(paste("players", players))
222 }
223 if (length(boxes) == 0) {
224 return("no boxes")
225 }
226 if (length(boxes) != length(level$goal_cells)) {
227 return(paste("boxes", length(boxes), "goals", length(level$goal_cells)))
228 }
229 cells <- seq_len(level$size)
230 open_cells <- which(level$interior & on_border(level, cells))
231 if (length(open_cells) > 0) {
232 return(paste("open", cell_text(level, open_cells[1])))
233 }
234 outside_cells <- which((level$goals | cells %in% boxes) & !level$interior)
235 if (length(outside_cells) > 0) {
236 return(paste("outside", cell_text(level, outside_cells[1])))
237 }
238 NULL
239}

249push_distances <- function(level, goal) {
250 distance <- rep(NA_integer_, level$size)
251 distance[goal] <- 0L
252 queue <- goal
253 while (length(queue) > 0) {
254 cell <- queue[1]
255 queue <- queue[-1]
256 for (direction in DIRECTIONS) {
257 offset <- level$offsets[[direction]]
258 box <- cell - offset
259 if (level$interior[box] && level$interior[box - offset] && is.na(distance[box])) {
260 distance[box] <- distance[cell] + 1L
261 queue <- c(queue, box)
262 }
263 }
264 }
265 distance
266}

303walk_path <- function(level, start, target, boxes) {
304 came_from <- rep(NA_integer_, level$size)
305 came_by <- rep(NA_character_, level$size)
306 queue <- start
307 while (length(queue) > 0) {
308 cell <- queue[1]
309 queue <- queue[-1]
310 for (direction in DIRECTIONS) {
311 neighbor <- cell + level$offsets[[direction]]
312 if (level$interior[neighbor] && !(neighbor %in% boxes) && neighbor != start &&
313 is.na(came_from[neighbor])) {
314 came_from[neighbor] <- cell
315 came_by[neighbor] <- direction
316 queue <- c(queue, neighbor)
317 }
318 }
319 }
320 directions <- character(0)
321 cell <- target
322 while (cell != start) {
323 directions <- c(came_by[cell], directions)
324 cell <- came_from[cell]
325 }
326 directions
327}

329take_step <- function(level, position, direction) {
330 offset <- level$offsets[[direction]]
331 target <- position$player + offset
332 if (level$walls[target]) {
333 step_failure("wall")
334 }
335 if (!(target %in% position$boxes)) {
336 return(list(position = new_position(target, position$boxes), letter = direction))
337 }
338 beyond <- target + offset
339 if (level$walls[beyond] || beyond %in% position$boxes) {
340 step_failure("blocked")
341 }
342 boxes <- move_box(position$boxes, target, beyond)
343 list(position = new_position(target, boxes), letter = toupper(direction))
344}

346pushable <- function(level, box, direction, boxes, reachable_cells) {
347 offset <- level$offsets[[direction]]
348 destination <- box + offset
349 reachable_cells[box - offset] && level$interior[destination] && !(destination %in% boxes)
350}

352side_blocked <- function(level, box, sides, boxes, assumed) {
353 if (any(level$walls[sides] | sides %in% assumed)) {
354 return(TRUE)
355 }
356 if (all(level$dead[sides])) {
357 return(TRUE)
358 }
359 extended <- c(assumed, box)
360 for (side in sides) {
361 if (side %in% boxes && frozen(level, side, boxes, extended)) {
362 return(TRUE)
363 }
364 }
365 FALSE
366}

368frozen <- function(level, box, boxes, assumed = integer(0)) {
369 horizontal <- box + level$offsets[c("l", "r")]
370 vertical <- box + level$offsets[c("u", "d")]
371 side_blocked(level, box, horizontal, boxes, assumed) &&
372 side_blocked(level, box, vertical, boxes, assumed)
373}

407map_cell <- function(level, kind, cell) {
408 row <- (cell - 1) %/% level$width
409 column <- (cell - 1) %% level$width
410 last_row <- level$height - 1
411 last_column <- level$width - 1
412 moved <- switch(kind,
413 ROT90 = column * level$height + last_row - row,
414 ROT180 = (last_row - row) * level$width + last_column - column,
415 ROT270 = (last_column - column) * level$height + row,

417 FLIPV = (last_row - row) * level$width + column
418 )
419 moved + 1
420}

441build_level <- function(rows) {
442 height <- length(rows)
443 width <- nchar(rows[1])
444 cells <- unlist(strsplit(rows, ""))
445 walls <- cells == "#"
446 goals <- cells %in% c(".", "*", "+")
447 players <- which(cells %in% c("@", "+"))
448 boxes <- which(cells %in% c("$", "*"))
449 interior <- if (length(players) == 1) {
450 find_interior(height, width, walls, players)
451 } else {
452 logical(length(cells))
453 }
454 level <- new_level(height, width, walls, goals, interior)
455 level$reason <- invalidity(level, length(players), boxes)
456 if (is.null(level$reason)) {
457 level <- analyze(level)
458 }
459 player <- if (length(players) > 0) players[1] else NA_integer_
460 list(level = level, position = new_position(player, boxes))
461}

489push_search <- function(level, start, fixed, target, pruning, limit) {
490 root <- list(
491 boxes = start$boxes,
492 representative = representative(level, start$player, start$boxes),
493 parent = NA,
494 push = NULL
495 )
496 nodes <- list(root)
497 known <- new.env(hash = TRUE)
498 known[[state_key(root$boxes, root$representative)]] <- TRUE
499 expanded <- 0L
500 while (expanded < length(nodes)) {
501 if (!is.null(limit) && expanded == limit) {
502 return(list(outcome = "limit", expanded = expanded, states = length(nodes)))
503 }
504 expanded <- expanded + 1L
505 node <- nodes[[expanded]]
506 reachable_cells <- reachable(level, node$representative, node$boxes)
507 for (box in node$boxes) {
508 if (box %in% fixed) {
509 next
510 }
511 for (direction in DIRECTIONS) {
512 if (!pushable(level, box, direction, node$boxes, reachable_cells)) {
513 next
514 }
515 destination <- box + level$offsets[[direction]]
516 boxes <- move_box(node$boxes, box, destination)
517 if (pruning && (level$dead[destination] || has_frozen_box(level, boxes))) {
518 next
519 }
520 first_cell <- representative(level, box, boxes)
521 key <- state_key(boxes, first_cell)
522 if (!is.null(known[[key]])) {
523 next
524 }
525 known[[key]] <- TRUE
526 nodes[[length(nodes) + 1]] <- list(
527 boxes = boxes,
528 representative = first_cell,
529 parent = expanded,
530 push = list(box = box, direction = direction)
531 )
532 reached <- if (is.null(target)) solved(level, boxes) else destination == target
533 if (reached) {
534 notation <- solution_notation(level, start, solution_pushes(nodes))
535 return(list(
536 outcome = "found", expanded = expanded, states = length(nodes),
537 notation = notation
538 ))
539 }
540 }
541 }
542 }
543 list(outcome = "no solution", expanded = expanded, states = length(nodes))
544}

731command_undo <- function(game, arguments) {

733 command_error("nothing to undo")
734 }
735 count <- min(as.integer(arguments[1]), length(game$history))
736 for (index in seq_len(count)) {
737 letter <- game$history[length(game$history)]
738 game$history <- game$history[-length(game$history)]
739 direction <- tolower(letter)
740 offset <- game$level$offsets[[direction]]
741 player <- game$current$player
742 boxes <- game$current$boxes
743 if (is_push(letter)) {
744 boxes <- move_box(boxes, player + offset, player)
745 }
746 game$current <- new_position(player - offset, boxes)
747 game$redo_list <- c(game$redo_list, direction)
748 }
749 emit("undone", count)
750}

831command_bound <- function(game, arguments) {
832 level <- game$level
833 boxes <- game$current$boxes
834 best <- NULL
835 for (order in permutations(seq_along(level$goal_cells))) {
836 pushes <- mapply(function(box, goal) level$distances[[goal]][box], boxes, order)
837 if (anyNA(pushes)) {
838 next
839 }
840 if (is.null(best) || sum(pushes) < sum(best$pushes)) {
841 best <- list(order = order, pushes = pushes)
842 }
843 }
844 if (is.null(best)) {
845 emit("bound none")
846 return()
847 }
848 emit("bound", sum(best$pushes))
849 for (index in seq_along(boxes)) {
850 goal <- level$goal_cells[best$order[index]]
851 emit(
852 "box", cell_text(level, boxes[index]), "goal", cell_text(level, goal),
853 "pushes", best$pushes[index]
854 )
855 }
856}

974check_values <- function(game, name, arguments) {
975 cells <- if (name %in% names(CELL_COUNTS)) CELL_COUNTS[[name]] else 0
976 for (index in seq_len(cells) * 2 - 1) {
977 row <- as.integer(arguments[index])
978 column <- as.integer(arguments[index + 1])
979 if (!contains_cell(game$level, row, column)) {
980 command_error("bad value")
981 }
982 }
983 if (name %in% c("UNDO", "REDO") && as.integer(arguments[1]) < 1) {
984 command_error("bad value")
985 }
986 if (name %in% c("SOLVE", "HINT")) {

994run_command <- function(game, name, arguments) {
995 if (!(name %in% names(COMMANDS))) {
996 command_error(paste("unknown command", name))
997 }
998 kinds <- COMMANDS[[name]]
999 well_formed <- length(arguments) == length(kinds) && all(vapply(
1000 seq_along(kinds),
1001 function(index) valid_argument(kinds[index], arguments[index]),
1002 logical(1)
1003 ))
1004 if (!well_formed) {
1005 command_error("bad arguments")
1006 }
1007 if (!(name %in% LEVEL_FREE_COMMANDS)) {
1008 if (!is.null(game$level$reason)) {
1009 command_error("invalid level")
1010 }
1011 check_values(game, name, arguments)
1012 }
1013 HANDLERS[[name]](game, arguments)
1014}
