26function decode(text, allowed) {
27 let result = '';
28 for (let index = 0; index < text.length;) {
29 const start = index;
30 while (index < text.length && text[index] >= '0' && text[index] <= '9') index++;
31 const count = text.slice(start, index);
32 if (count.length > 2 || count.startsWith('0') || index === text.length) return null;
33 const symbol = text[index++];
34 if (!allowed.includes(symbol)) return null;
35 result += symbol.repeat(count ? Number(count) : 1);
36 }
37 return result;
38}

40function encode(text) {
41 let result = '';
42 for (let start = 0; start < text.length;) {
43 let end = start + 1;
44 while (end < text.length && text[end] === text[start]) end++;
45 const count = end - start;
46 result += (count > 1 ? String(count) : '') + text[start];
47 start = end;
48 }
49 return result;
50}

52function validArgument(kind, text) {
53 switch (kind) {
54 case 'number': return /^(0|[1-9][0-9]{0,5})$/.test(text);
55 case 'steps': return decode(text, STEP_LETTERS) !== null;
56 case 'direction': return DIRECTIONS.includes(text);
57 case 'name': return /^[a-z]{1,8}$/.test(text);
58 case 'transform': return TURNS.has(text);
59 default: return true;
60 }
61}

63function directionOf(letter) { return letter.toLowerCase(); }
64function pushCount(letters) { return [...letters].filter(letter => letter !== letter.toLowerCase()).length; }
65function moveBox(boxes, source, destination) {
66 return boxes.map(box => box === source ? destination : box).sort((a, b) => a - b);
67}
68function samePosition(a, b) {
69 return a.player === b.player && a.boxes.length === b.boxes.length
70 && a.boxes.every((box, index) => box === b.boxes[index]);
71}
72function turnLetters(letters, kind) {
73 const turn = TURNS.get(kind);
74 return [...letters].map(letter => {
75 const direction = turn[DIRECTIONS.indexOf(directionOf(letter))];
76 return letter === letter.toLowerCase() ? direction : direction.toUpperCase();
77 }).join('');
78}
79function counted(label, letters) {
80 return letters.length ? `${label} ${letters.length} ${letters}` : `${label} 0`;

84function findInterior(height, width, walls, start) {
85 const seen = Array(height * width).fill(false);
86 const queue = [start];
87 seen[start] = true;
88 for (let head = 0; head < queue.length; head++) {
89 const cell = queue[head];
90 const row = Math.floor(cell / width), column = cell % width;
91 for (const [dr, dc] of DELTAS) {
92 const r = row + dr, c = column + dc;
93 if (r < 0 || r >= height || c < 0 || c >= width) continue;
94 const neighbor = r * width + c;
95 if (!walls[neighbor] && !seen[neighbor]) {
96 seen[neighbor] = true;
97 queue.push(neighbor);
98 }
99 }
100 }
101 return seen;
102}

104class Level {
105 constructor(height, width, walls, goals, interior) {
106 this.height = height;
107 this.width = width;
108 this.size = height * width;
109 this.walls = walls;
110 this.goals = goals;
111 this.interior = interior;
112 this.offsets = [-width, width, -1, 1];
113 this.goalCells = Array.from({length: this.size}, (_, cell) => cell).filter(cell => goals[cell]);
114 this.reason = null;
115 this.distances = [];
116 this.dead = Array(this.size).fill(false);
117 }
118
119 offsetOf(direction) { return this.offsets[DIRECTIONS.indexOf(direction)]; }
120 cellText(cell) { return `${Math.floor(cell / this.width) + 1} ${cell % this.width + 1}`; }
121 cellAt(row, column) { return (row - 1) * this.width + column - 1; }
122 contains(row, column) { return row >= 1 && row <= this.height && column >= 1 && column <= this.width; }
123 onBorder(cell) {
124 const row = Math.floor(cell / this.width), column = cell % this.width;
125 return row === 0 || row === this.height - 1 || column === 0 || column === this.width - 1;
126 }
127
128 invalidity(players, boxes) {
129 if (players !== 1) return `players ${players}`;
130 if (!boxes.length) return 'no boxes';
131 if (boxes.length !== this.goalCells.length) return `boxes ${boxes.length} goals ${this.goalCells.length}`;
132 for (let cell = 0; cell < this.size; cell++) {
133 if (this.interior[cell] && this.onBorder(cell)) return `open ${this.cellText(cell)}`;
134 }
135 for (let cell = 0; cell < this.size; cell++) {
136 if ((this.goals[cell] || boxes.includes(cell)) && !this.interior[cell]) return `outside ${this.cellText(cell)}`;
137 }
138 return null;
139 }
140
141 analyze() {
142 this.distances = this.goalCells.map(goal => this.pushDistances(goal));
143 this.dead = this.interior.map((inside, cell) => inside && this.distances.every(distance => distance[cell] === -1));
144 }
145
146 pushDistances(goal) {
147 const distance = Array(this.size).fill(-1);
148 const queue = [goal];
149 distance[goal] = 0;
150 for (let head = 0; head < queue.length; head++) {
151 const cell = queue[head];
152 for (const offset of this.offsets) {
153 const box = cell - offset;
154 if (this.interior[box] && this.interior[box - offset] && distance[box] === -1) {
155 distance[box] = distance[cell] + 1;
156 queue.push(box);
157 }
158 }
159 }
160 return distance;
161 }
162
163 goalDistance(cell) {
164 const defined = this.distances.map(distance => distance[cell]).filter(distance => distance >= 0);
165 return defined.length ? Math.min(...defined) : null;
166 }
167 boxesOnGoals(boxes) { return boxes.filter(box => this.goals[box]).length; }
168 solved(boxes) { return this.boxesOnGoals(boxes) === boxes.length; }
169
170 reachable(player, boxes) {
171 const seen = Array(this.size).fill(false);
172 const queue = [player];
173 seen[player] = true;
174 for (let head = 0; head < queue.length; head++) {
175 for (const offset of this.offsets) {
176 const neighbor = queue[head] + offset;
177 if (this.interior[neighbor] && !seen[neighbor] && !boxes.includes(neighbor)) {
178 seen[neighbor] = true;
179 queue.push(neighbor);
180 }
181 }
182 }
183 return seen;
184 }
185 representative(player, boxes) { return this.reachable(player, boxes).indexOf(true); }
186
187 walkPath(start, target, boxes) {
188 const cameFrom = Array(this.size).fill(null);
189 const queue = [start];
190 for (let head = 0; head < queue.length; head++) {
191 const cell = queue[head];
192 for (const direction of DIRECTIONS) {
193 const neighbor = cell + this.offsetOf(direction);
194 if (this.interior[neighbor] && !boxes.includes(neighbor)
195 && neighbor !== start && cameFrom[neighbor] === null) {
196 cameFrom[neighbor] = {cell, direction};
197 queue.push(neighbor);
198 }
199 }
200 }
201 const path = [];
202 for (let cell = target; cell !== start;) {
203 const from = cameFrom[cell];
204 path.push(from.direction);
205 cell = from.cell;
206 }
207 return path.reverse().join('');
208 }
209
210 step(position, direction) {
211 const offset = this.offsetOf(direction);
212 const target = position.player + offset;
213 if (this.walls[target]) throw new StepFailure('wall');
214 if (!position.boxes.includes(target)) {
215 return {position: {player: target, boxes: position.boxes}, letter: direction};
216 }
217 const beyond = target + offset;
218 if (this.walls[beyond] || position.boxes.includes(beyond)) throw new StepFailure('blocked');
219 return {position: {player: target, boxes: moveBox(position.boxes, target, beyond)}, letter: direction.toUpperCase()};
220 }
221
222 pushable(box, direction, boxes, reachable) {
223 const offset = this.offsetOf(direction), destination = box + offset;
224 return !!reachable[box - offset] && !!this.interior[destination] && !boxes.includes(destination);
225 }
226 sideBlocked(box, sides, boxes, assumed) {
227 if (sides.some(side => this.walls[side] || assumed.includes(side))) return true;
228 if (sides.every(side => this.dead[side])) return true;
229 const extended = [...assumed, box];
230 return sides.some(side => boxes.includes(side) && this.frozen(side, boxes, extended));
231 }
232 frozen(box, boxes, assumed = []) {
233 return this.sideBlocked(box, [box - 1, box + 1], boxes, assumed)
234 && this.sideBlocked(box, [box - this.width, box + this.width], boxes, assumed);
235 }
236 hasFrozenBox(boxes) { return boxes.some(box => !this.goals[box] && this.frozen(box, boxes)); }
237
238 splitRows(symbols) {
239 const rows = [];
240 for (let start = 0; start < this.size; start += this.width) rows.push(symbols.slice(start, start + this.width));
241 return rows;
242 }
243 render(position) {
244 let symbols = '';
245 for (let cell = 0; cell < this.size; cell++) {
246 if (this.walls[cell]) symbols += '#';
247 else if (cell === position.player) symbols += this.goals[cell] ? '+' : '@';
248 else if (position.boxes.includes(cell)) symbols += this.goals[cell] ? '*' : '$';
249 else symbols += this.goals[cell] ? '.' : '-';
250 }
251 return this.splitRows(symbols);
252 }
253 deadMap() {
254 let symbols = '';
255 for (let cell = 0; cell < this.size; cell++) {
256 symbols += this.walls[cell] ? '#' : this.goals[cell] ? '.' : this.dead[cell] ? 'x' : '-';
257 }
258 return this.splitRows(symbols);
259 }
260
261 mapCell(kind, cell) {
262 const row = Math.floor(cell / this.width), column = cell % this.width;
263 switch (kind) {
264 case 'ROT90': return column * this.height + this.height - 1 - row;
265 case 'ROT180': return (this.height - 1 - row) * this.width + this.width - 1 - column;
266 case 'ROT270': return (this.width - 1 - column) * this.height + row;
267 case 'FLIPH': return row * this.width + this.width - 1 - column;
268 default: return (this.height - 1 - row) * this.width + column;
269 }
270 }
271 mapPosition(kind, position) {
272 return {player: this.mapCell(kind, position.player), boxes: position.boxes.map(box => this.mapCell(kind, box)).sort((a, b) => a - b)};
273 }
274 transformed(kind) {
275 const rotated = kind === 'ROT90' || kind === 'ROT270';
276 const walls = Array(this.size).fill(false), goals = Array(this.size).fill(false), interior = Array(this.size).fill(false);
277 for (let cell = 0; cell < this.size; cell++) {
278 const moved = this.mapCell(kind, cell);
279 walls[moved] = this.walls[cell];
280 goals[moved] = this.goals[cell];
281 interior[moved] = this.interior[cell];
282 }
283 const level = new Level(rotated ? this.width : this.height, rotated ? this.height : this.width, walls, goals, interior);
284 level.analyze();
285 return level;
286 }
287}

289function buildLevel(rows) {
290 const height = rows.length, width = rows[0].length;
291 const cells = [...rows.join('')];
292 const walls = cells.map(symbol => symbol === '#'), goals = cells.map(symbol => '.*+'.includes(symbol));
293 const players = [], boxes = [];
294 cells.forEach((symbol, cell) => {
295 if ('@+'.includes(symbol)) players.push(cell);
296 if ('$*'.includes(symbol)) boxes.push(cell);
297 });
298 const interior = players.length === 1 ? findInterior(height, width, walls, players[0]) : Array(cells.length).fill(false);
299 const level = new Level(height, width, walls, goals, interior);
300 level.reason = level.invalidity(players.length, boxes);
301 if (level.reason === null) level.analyze();
302 return {level, position: {player: players.length ? players[0] : -1, boxes}};
303}

305function stateKey(boxes, representative) { return `${boxes.join(',')}:${representative}`; }
306function solutionNotation(level, start, nodes) {
307 const pushes = [];
308 for (let node = nodes[nodes.length - 1]; node.parent !== -1; node = nodes[node.parent]) pushes.push(node.push);
309 let position = start, notation = '';
310 for (const {box, direction} of pushes.reverse()) {
311 const offset = level.offsetOf(direction);
312 notation += level.walkPath(position.player, box - offset, position.boxes) + direction.toUpperCase();
313 position = {player: box, boxes: moveBox(position.boxes, box, box + offset)};
314 }
315 return notation;
316}

318function pushSearch(level, start, fixed, target, pruning, limit) {
319 const representative = level.representative(start.player, start.boxes);
320 const nodes = [{boxes: start.boxes, representative, parent: -1, push: null}];
321 const known = new Set([stateKey(start.boxes, representative)]);
322 let expanded = 0;
323 while (expanded < nodes.length) {
324 if (limit !== null && expanded === limit) return {outcome: 'limit', expanded, states: nodes.length, notation: ''};
325 const parent = expanded, node = nodes[expanded++];
326 const reachable = level.reachable(node.representative, node.boxes);
327 for (const box of node.boxes) {
328 if (fixed.includes(box)) continue;
329 for (const direction of DIRECTIONS) {
330 if (!level.pushable(box, direction, node.boxes, reachable)) continue;
331 const destination = box + level.offsetOf(direction);
332 const boxes = moveBox(node.boxes, box, destination);
333 if (pruning && (level.dead[destination] || level.hasFrozenBox(boxes))) continue;
334 const nextRepresentative = level.representative(box, boxes);
335 const key = stateKey(boxes, nextRepresentative);
336 if (known.has(key)) continue;
337 known.add(key);
338 nodes.push({boxes, representative: nextRepresentative, parent, push: {box, direction}});
339 if (target === null ? level.solved(boxes) : destination === target) {
340 return {outcome: 'found', expanded, states: nodes.length, notation: solutionNotation(level, start, nodes)};
341 }
342 }
343 }
344 }
345 return {outcome: 'no solution', expanded, states: nodes.length, notation: ''};
346}

348class Sokoban {
349 constructor(rows) {
350 this.output = [];
351 this.load(rows);
352 }
353 emit(...parts) { this.output.push(parts.join(' ')); }
354 load(rows) {
355 const loaded = buildLevel(rows);
356 this.level = loaded.level;
357 this.initial = loaded.position;
358 this.current = this.initial;
359 this.history = '';
360 this.redoList = '';
361 this.snapshots = new Map();
362 this.solution = null;
363 }
364 cellArgument(args, index) { return this.level.cellAt(Number(args[index]), Number(args[index + 1])); }
365
366 simulate(directions) {
367 let position = this.current, letters = '';
368 for (const letter of directions) {
369 try {
370 const step = this.level.step(position, directionOf(letter));
371 position = step.position;
372 letters += step.letter;
373 } catch (error) {
374 if (!(error instanceof StepFailure)) throw error;
375 return {position, letters, failure: `step ${letters.length + 1} ${error.message}`};
376 }
377 }
378 return {position, letters, failure: null};
379 }
380 takeSteps(directions) {
381 const result = this.simulate(directions);
382 this.current = result.position;
383 this.history += result.letters;
384 if (result.letters.length) this.redoList = '';
385 return result;
386 }
387 progress(boxes) {
388 const count = this.level.boxesOnGoals(boxes);
389 return `goals ${count}/${boxes.length}` + (count === boxes.length ? ' solved' : '');
390 }
391 searchSolution(limit) {
392 if (this.level.solved(this.current.boxes)) return {text: 'already solved', notation: null};
393 const result = pushSearch(this.level, this.current, [], null, true, limit);
394 const counts = `expanded ${result.expanded} states ${result.states}`;
395 if (result.outcome !== 'found') return {text: `${result.outcome} ${counts}`, notation: null};
396 const notation = result.notation;
397 return {text: `solution pushes ${pushCount(notation)} moves ${notation.length} ${counts}`, notation};
398 }
399
400 loadRows(width, rows) {
401 const bad = rows.findIndex(row => row.length !== width || [...row].some(symbol => !LEVEL_CHARACTERS.includes(symbol)));
402 if (bad !== -1) throw new CommandError(`bad row ${bad + 1}`);
403 this.load(rows);
404 this.emit('loaded', `${rows.length}x${width}`);
405 }
406 importLevel(text) {
407 const rows = [];
408 for (const part of text.split('|')) {
409 const row = decode(part, LEVEL_CHARACTERS);
410 if (row === null) throw new CommandError('bad level');
411 rows.push(row);
412 }
413 const width = rows[0].length;
414 if (rows.length > 10 || width < 1 || width > 10 || rows.some(row => row.length !== width)) throw new CommandError('bad level');
415 this.load(rows);
416 this.emit('loaded', `${rows.length}x${width}`);
417 }
418 move(word) {
419 const result = this.takeSteps(decode(word, STEP_LETTERS));
420 if (result.letters.length) this.emit('moved', result.letters);
421 if (result.failure !== null) throw new CommandError(result.failure);
422 }
423 push(box, direction) {
424 const level = this.level, boxes = this.current.boxes;
425 if (!boxes.includes(box)) throw new CommandError('no box');
426 const offset = level.offsetOf(direction), destination = box + offset;
427 if (!level.interior[destination] || boxes.includes(destination)) throw new CommandError('blocked');
428 if (!level.reachable(this.current.player, boxes)[box - offset]) throw new CommandError('unreachable');
429 const path = level.walkPath(this.current.player, box - offset, boxes);
430 this.emit('pushed', this.takeSteps(path + direction).letters);
431 }
432 bring(source, target) {
433 const boxes = this.current.boxes;
434 if (!boxes.includes(source)) throw new CommandError('no box');
435 if (source === target) { this.emit('brought', 0); return; }
436 const result = pushSearch(this.level, this.current, boxes.filter(box => box !== source), target, false, null);
437 if (result.outcome !== 'found') throw new CommandError('no route');
438 this.takeSteps(result.notation);
439 this.emit('brought', pushCount(result.notation), result.notation);
440 }
441 path(target) {
442 if (!this.level.reachable(this.current.player, this.current.boxes)[target]) throw new CommandError('unreachable');
443 const path = this.level.walkPath(this.current.player, target, this.current.boxes);
444 this.emit(counted('path', this.takeSteps(path).letters));
445 }
446 undo(requested) {
447 if (!this.history.length) throw new CommandError('nothing to undo');
448 const count = Math.min(requested, this.history.length);
449 for (let index = 0; index < count; index++) {
450 const letter = this.history[this.history.length - 1], direction = directionOf(letter);
451 this.history = this.history.slice(0, -1);
452 const offset = this.level.offsetOf(direction), player = this.current.player;
453 const boxes = letter === direction ? this.current.boxes : moveBox(this.current.boxes, player + offset, player);
454 this.current = {player: player - offset, boxes};
455 this.redoList += direction;
456 }
457 this.emit('undone', count);
458 }
459 redo(requested) {
460 if (!this.redoList.length) throw new CommandError('nothing to redo');
461 const count = Math.min(requested, this.redoList.length);
462 for (let index = 0; index < count; index++) {
463 const direction = directionOf(this.redoList[this.redoList.length - 1]);
464 this.redoList = this.redoList.slice(0, -1);
465 const step = this.level.step(this.current, direction);
466 this.current = step.position;
467 this.history += step.letter;
468 }
469 this.emit('redone', count);
470 }
471 showBoxes() {
472 const boxes = this.current.boxes, level = this.level;
473 const reachable = level.reachable(this.current.player, boxes);
474 this.emit('boxes', boxes.length);
475 for (const box of boxes) {
476 const distance = level.goalDistance(box);
477 const directions = DIRECTIONS.filter(direction => level.pushable(box, direction, boxes, reachable)).join('');
478 this.emit('box', level.cellText(box), distance === null ? 'dead' : distance, directions || '-');
479 }
480 }
481 deadlock() {
482 const level = this.level, boxes = this.current.boxes;
483 const dead = boxes.find(box => level.dead[box]);
484 if (dead !== undefined) { this.emit('deadlock dead', level.cellText(dead)); return; }
485 const frozen = boxes.find(box => !level.goals[box] && level.frozen(box, boxes));
486 if (frozen !== undefined) { this.emit('deadlock frozen', level.cellText(frozen)); return; }
487 this.emit('no deadlock');
488 }
489 bound() {
490 const boxes = this.current.boxes, level = this.level;
491 let best = null;
492 const assign = (order, pushes, used, total) => {
493 if (order.length === boxes.length) {
494 if (best === null || total < best.total) best = {order, pushes, total};
495 return;
496 }
497 for (let goal = 0; goal < level.goalCells.length; goal++) {
498 const distance = level.distances[goal][boxes[order.length]];
499 if (!used.has(goal) && distance >= 0) {
500 assign([...order, goal], [...pushes, distance], new Set([...used, goal]), total + distance);
501 }
502 }
503 };
504 assign([], [], new Set(), 0);
505 if (best === null) { this.emit('bound none'); return; }
506 this.emit('bound', best.total);
507 boxes.forEach((box, index) => this.emit('box', level.cellText(box), 'goal', level.cellText(level.goalCells[best.order[index]]), 'pushes', best.pushes[index]));
508 }
509 save(name) {
510 this.snapshots.set(name, {position: this.current, history: this.history});
511 this.emit('saved', name, 'moves', this.history.length);
512 }
513 restore(name) {
514 const snapshot = this.snapshots.get(name);
515 if (snapshot === undefined) throw new CommandError(`no snapshot ${name}`);
516 this.current = snapshot.position;
517 this.history = snapshot.history;
518 this.redoList = '';
519 this.emit('restored', name, 'moves', this.history.length);
520 }
521 check(word) {
522 const result = this.simulate(decode(word, STEP_LETTERS));
523 if (result.failure !== null) this.emit('check fail', result.failure);
524 else this.emit('check ok moves', result.letters.length, 'pushes', pushCount(result.letters), this.progress(result.position.boxes));
525 }
526 solve(limit) {
527 const summary = this.searchSolution(limit);
528 this.emit(summary.text);
529 this.solution = null;
530 if (summary.notation !== null) {
531 this.emit(summary.notation);
532 this.solution = {notation: summary.notation, position: this.current};
533 }
534 }
535 hint(limit) {
536 const summary = this.searchSolution(limit);
537 if (summary.notation === null) { this.emit(summary.text); return; }
538 const end = [...summary.notation].findIndex(letter => letter !== letter.toLowerCase());
539 const part = summary.notation.slice(0, end + 1);
540 this.takeSteps(part);
541 this.emit('hint', part);
542 }
543 play() {
544 if (this.solution === null || !samePosition(this.solution.position, this.current)) throw new CommandError('no solution');
545 const notation = this.solution.notation;
546 this.takeSteps(notation);
547 this.emit('played pushes', pushCount(notation), 'moves', notation.length);
548 }
549 transform(kind) {
550 const old = this.level;
551 this.level = old.transformed(kind);
552 this.initial = old.mapPosition(kind, this.initial);
553 this.current = old.mapPosition(kind, this.current);
554 this.history = turnLetters(this.history, kind);
555 this.redoList = turnLetters(this.redoList, kind);
556 for (const [name, snapshot] of this.snapshots) {
557 this.snapshots.set(name, {position: old.mapPosition(kind, snapshot.position), history: turnLetters(snapshot.history, kind)});
558 }
559 if (this.solution !== null) this.solution = {notation: turnLetters(this.solution.notation, kind), position: old.mapPosition(kind, this.solution.position)};
560 this.emit('transformed', `${this.level.height}x${this.level.width}`);
561 }

563 checkValues(name, args) {
564 const cells = name === 'BRING' ? 2 : name === 'PATH' || name === 'PUSH' ? 1 : 0;
565 for (let index = 0; index < cells * 2; index += 2) {
566 if (!this.level.contains(Number(args[index]), Number(args[index + 1]))) throw new CommandError('bad value');
567 }
568 if ((name === 'UNDO' || name === 'REDO') && Number(args[0]) < 1) throw new CommandError('bad value');
569 if ((name === 'SOLVE' || name === 'HINT') && (Number(args[0]) < 1 || Number(args[0]) > 100000)) throw new CommandError('bad value');
570 }
571 runCommand(name, args) {
572 const kinds = COMMANDS.get(name);
573 if (kinds === undefined) throw new CommandError(`unknown command ${name}`);
574 if (args.length !== kinds.length || kinds.some((kind, index) => !validArgument(kind, args[index]))) throw new CommandError('bad arguments');
575 if (name === 'IMPORT') { this.importLevel(args[0]); return; }
576 if (name === 'VALID') { this.emit(this.level.reason === null ? 'valid' : `invalid ${this.level.reason}`); return; }
577 if (this.level.reason !== null) throw new CommandError('invalid level');
578 this.checkValues(name, args);
579 const level = this.level;
580 switch (name) {
581 case 'INFO': this.emit('size', `${level.height}x${level.width}`, 'walls', countTrue(level.walls), 'cells', countTrue(level.interior), 'boxes', this.current.boxes.length, 'dead', countTrue(level.dead)); break;
582 case 'SHOW': level.render(this.current).forEach(row => this.emit(row)); break;
583 case 'EXPORT': this.emit('level', level.render(this.current).map(encode).join('|')); break;
584 case 'MOVE': this.move(args[0]); break;
585 case 'PUSH': this.push(this.cellArgument(args, 0), directionOf(args[2])); break;
586 case 'BRING': this.bring(this.cellArgument(args, 0), this.cellArgument(args, 2)); break;
587 case 'PATH': this.path(this.cellArgument(args, 0)); break;
588 case 'UNDO': this.undo(Number(args[0])); break;
589 case 'REDO': this.redo(Number(args[0])); break;
590 case 'RESET': this.current = this.initial; this.history = ''; this.redoList = ''; this.emit('reset'); break;
591 case 'STATUS': this.emit('moves', this.history.length, 'pushes', pushCount(this.history), this.progress(this.current.boxes)); break;
592 case 'HISTORY': this.emit(counted('history', this.history)); break;
593 case 'BOXES': this.showBoxes(); break;
594 case 'REACH': {
595 const reach = level.reachable(this.current.player, this.current.boxes);
596 this.emit('reach', countTrue(reach), 'first', level.cellText(reach.indexOf(true)));
597 break;
598 }
599 case 'DEADSQUARES': this.emit('dead', countTrue(level.dead)); level.deadMap().forEach(row => this.emit(row)); break;
600 case 'DEADLOCK': this.deadlock(); break;
601 case 'BOUND': this.bound(); break;
602 case 'SAVE': this.save(args[0]); break;
603 case 'RESTORE': this.restore(args[0]); break;
604 case 'CHECK': this.check(args[0]); break;
605 case 'SOLVE': this.solve(Number(args[0])); break;
606 case 'HINT': this.hint(Number(args[0])); break;
607 case 'PLAY': this.play(); break;
608 case 'TRANSFORM': this.transform(args[0]); break;
609 }
610 }
611 execute(words, rows) {
612 try {
613 if (words[0] === 'LOAD') this.loadRows(Number(words[2]), rows);
614 else this.runCommand(words[0], words.slice(1));
615 } catch (error) {
616 if (!(error instanceof CommandError)) throw error;
617 this.emit('error:', error.message);
618 }
619 }
