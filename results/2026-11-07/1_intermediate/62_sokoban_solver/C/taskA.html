38static void reserve(String *s, size_t size) {
39 if (size + 1 > s->cap) {

41 while (cap < size + 1)
42 cap *= 2;
43 char *p = realloc(s->data, cap);
44 if (!p)
45 exit(EXIT_FAILURE);
46 s->data = p;
47 s->cap = cap;
48 }
49}

83static void sort_boxes(Position *p) {
84 for (int i = 1; i < p->count; i++) {
85 int b = p->boxes[i], j = i;
86 while (j && p->boxes[j - 1] > b) {

90 p->boxes[j] = b;

105static int neighbor(const Level *l, int c, int d) {
106 int r = c / l->w, col = c % l->w;
107 static const int dr[] = {-1, 1, 0, 0}, dc[] = {0, 0, -1, 1};
108 r += dr[d];
109 col += dc[d];
110 return r < 0 || r >= l->h || col < 0 || col >= l->w ? -1 : r * l->w + col;
111}
112static void reach(const Level *l, Position p, unsigned char seen[100]) {
113 int queue[100], head = 0, tail = 0;
114 memset(seen, 0, 100);
115 seen[p.player] = 1;
116 queue[tail++] = p.player;
117 while (head < tail) {
118 int c = queue[head++];
119 for (int d = 0; d < 4; d++) {
120 int n = neighbor(l, c, d);
121 if (inside(l, n) && !seen[n] && box_index(p, n) < 0) {
122 seen[n] = 1;
123 queue[tail++] = n;
124 }
125 }
126 }
127}
128static int representative(const Level *l, Position p) {
129 unsigned char seen[100];
130 reach(l, p, seen);
131 for (int i = 0; i < l->size; i++)
132 if (seen[i])
133 return i;
134 return -1;
135}
136static String walk(const Level *l, Position p, int target) {
137 int previous[100], arrived[100], queue[100], head = 0, tail = 0;
138 for (int i = 0; i < 100; i++)
139 previous[i] = -1;
140 queue[tail++] = p.player;
141 previous[p.player] = p.player;
142 while (head < tail) {
143 int c = queue[head++];
144 for (int d = 0; d < 4; d++) {
145 int n = neighbor(l, c, d);
146 if (inside(l, n) && box_index(p, n) < 0 && previous[n] < 0) {
147 previous[n] = c;
148 arrived[n] = d;
149 queue[tail++] = n;
150 }
151 }
152 }
153 String path = {0};
154 for (int c = target; c != p.player; c = previous[c])
155 append_char(&path, directions[arrived[c]]);
156 for (size_t i = 0; i < path.len / 2; i++) {
157 char c = path.data[i];
158 path.data[i] = path.data[path.len - 1 - i];
159 path.data[path.len - 1 - i] = c;
160 }
161 return path;
162}
163static void analyze(Level *l) {
164 l->goal_count = 0;
165 for (int c = 0; c < l->size; c++)
166 if (l->goal[c])
167 l->goals[l->goal_count++] = c;
168 for (int g = 0; g < l->goal_count; g++) {
169 int queue[100], head = 0, tail = 0;
170 for (int c = 0; c < 100; c++)
171 l->distance[g][c] = -1;
172 int start = l->goals[g];
173 l->distance[g][start] = 0;
174 queue[tail++] = start;
175 while (head < tail) {
176 int c = queue[head++];
177 for (int d = 0; d < 4; d++) {
178 int b = c - l->offsets[d], stand = b - l->offsets[d];
179 if (inside(l, b) && inside(l, stand) && l->distance[g][b] < 0) {
180 l->distance[g][b] = l->distance[g][c] + 1;
181 queue[tail++] = b;
182 }
183 }
184 }
185 }
186 for (int c = 0; c < l->size; c++) {
187 l->dead[c] = l->interior[c];
188 for (int g = 0; g < l->goal_count; g++)
189 if (l->distance[g][c] >= 0)
190 l->dead[c] = 0;
191 }
192}
193static void offsets(Level *l) {
194 l->offsets[0] = -l->w;
195 l->offsets[1] = l->w;
196 l->offsets[2] = -1;
197 l->offsets[3] = 1;
198}
199static void load(Game *game, char rows[10][101], int h, int w) {
200 Level l = {0};
201 Position p = {.player = -1};
202 int players = 0;
203 l.h = h;
204 l.w = w;
205 l.size = h * w;
206 offsets(&l);
207 for (int c = 0; c < l.size; c++) {
208 char ch = rows[c / w][c % w];
209 l.walls[c] = ch == '#';
210 l.goal[c] = strchr(".*+", ch) != NULL;
211 if (ch == '@' || ch == '+') {
212 p.player = c;
213 players++;
214 }
215 if (ch == '$' || ch == '*')
216 p.boxes[p.count++] = c;
217 if (l.goal[c])
218 l.goals[l.goal_count++] = c;
219 }
220 if (players == 1) {
221 int queue[100], head = 0, tail = 0;
222 queue[tail++] = p.player;
223 l.interior[p.player] = 1;
224 while (head < tail) {
225 int c = queue[head++];
226 for (int d = 0; d < 4; d++) {
227 int n = neighbor(&l, c, d);
228 if (n >= 0 && !l.walls[n] && !l.interior[n]) {
229 l.interior[n] = 1;
230 queue[tail++] = n;
231 }
232 }
233 }
234 }
235 if (players != 1)
236 snprintf(l.reason, sizeof l.reason, "players %d", players);
237 else if (!p.count)
238 strcpy(l.reason, "no boxes");
239 else if (p.count != l.goal_count)
240 snprintf(l.reason, sizeof l.reason, "boxes %d goals %d", p.count, l.goal_count);
241 else {
242 for (int c = 0; c < l.size; c++)
243 if (l.interior[c] && (c / w == 0 || c / w == h - 1 || c % w == 0 || c % w == w - 1)) {
244 snprintf(l.reason, sizeof l.reason, "open %d %d", c / w + 1, c % w + 1);
245 break;
246 }
247 if (!l.reason[0])
248 for (int c = 0; c < l.size; c++)
249 if ((l.goal[c] || box_index(p, c) >= 0) && !l.interior[c]) {
250 snprintf(l.reason, sizeof l.reason, "outside %d %d", c / w + 1, c % w + 1);
251 break;
252 }
253 }
254 if (!l.reason[0])
255 analyze(&l);
256 for (int i = 0; i < game->snapshot_count; i++)
257 dispose(&game->snapshots[i].history);
258 game->snapshot_count = 0;
259 game->has_solution = 0;
260 clear(&game->solution);
261 clear(&game->history);
262 clear(&game->redo);
263 game->level = l;
264 game->initial = game->current = p;
265}
266static int decode(const char *word, const char *allowed, String *out) {
267 for (size_t i = 0; word[i];) {
268 int count = 0, digits = 0;
269 size_t start = i;
270 while (word[i] >= '0' && word[i] <= '9') {
271 if (digits < 3)
272 count = count * 10 + word[i] - '0';
273 digits++;
274 i++;
275 }
276 if (digits > 2 || (digits && word[start] == '0') || !word[i] || !strchr(allowed, word[i]))
277 return 0;
278 if (!digits)
279 count = 1;
280 for (int j = 0; j < count; j++)
281 append_char(out, word[i]);
282 i++;
283 }
284 return 1;
285}

311static int step(const Level *l, Position *p, char direction, char *letter) {
312 int d = (int)(strchr(directions, tolower((unsigned char)direction)) - directions),
313 off = l->offsets[d];
314 int t = p->player + off;
315 if (l->walls[t])
316 return 1;
317 *letter = directions[d];
318 if (box_index(*p, t) >= 0) {
319 int beyond = t + off;
320 if (l->walls[beyond] || box_index(*p, beyond) >= 0)
321 return 2;
322 *p = move_box(*p, t, beyond);
323 *letter = (char)toupper((unsigned char)*letter);
324 }
325 p->player = t;
326 return 0;
327}
328static int execute(Game *g, const char *sequence, String *letters, int commit) {
329 Position p = g->current;
330 int failure = 0;
331 for (size_t i = 0; sequence[i]; i++) {
332 char letter;
333 failure = step(&g->level, &p, sequence[i], &letter);
334 if (failure)
335 break;
336 append_char(letters, letter);
337 }
338 if (commit) {
339 g->current = p;
340 append(&g->history, text(letters));
341 if (letters->len)
342 clear(&g->redo);
343 }
344 return failure;
345}

347static int axis_blocked(const Level *l, Position p, int b, int a, int z, unsigned assumed) {
348 int ia = box_index(p, a), iz = box_index(p, z);
349 if (l->walls[a] || l->walls[z] || (ia >= 0 && (assumed & (1u << ia))) ||
350 (iz >= 0 && (assumed & (1u << iz))))
351 return 1;
352 if (l->dead[a] && l->dead[z])
353 return 1;
354 unsigned more = assumed | (1u << box_index(p, b));
355 return (ia >= 0 && frozen(l, p, a, more)) || (iz >= 0 && frozen(l, p, z, more));
356}
357static int frozen(const Level *l, Position p, int b, unsigned assumed) {
358 return axis_blocked(l, p, b, b - 1, b + 1, assumed) &&
359 axis_blocked(l, p, b, b - l->w, b + l->w, assumed);
360}
361static int has_frozen(const Level *l, Position p) {
362 for (int i = 0; i < p.count; i++)
363 if (!l->goal[p.boxes[i]] && frozen(l, p, p.boxes[i], 0))
364 return 1;
365 return 0;
366}

386static uint64_t state_key(Position p) {
387 uint64_t key = (uint64_t)p.player + 1;
388 for (int i = 0; i < p.count; i++)
389 key = (key << 7) | (uint64_t)(p.boxes[i] + 1);
390 return key;
391}
392static size_t hash_key(uint64_t key) {
393 key ^= key >> 30;
394 key *= UINT64_C(0xbf58476d1ce4e5b9);
395 key ^= key >> 27;
396 key *= UINT64_C(0x94d049bb133111eb);
397 return (size_t)(key ^ (key >> 31));
398}
399static int remember(Known *k, uint64_t key) {
400 if (!k->cap || (k->count + 1) * 10 > k->cap * 7) {
401 size_t old_cap = k->cap;
402 uint64_t *old = k->keys;
403 k->cap = k->cap ? k->cap * 2 : 1024;
404 k->keys = calloc(k->cap, sizeof(uint64_t));
405 if (!k->keys)
406 exit(EXIT_FAILURE);
407 for (size_t i = 0; i < old_cap; i++)
408 if (old[i]) {
409 size_t slot = hash_key(old[i]) & (k->cap - 1);
410 while (k->keys[slot])
411 slot = (slot + 1) & (k->cap - 1);
412 k->keys[slot] = old[i];
413 }
414 free(old);
415 }
416 size_t slot = hash_key(key) & (k->cap - 1);
417 while (k->keys[slot]) {
418 if (k->keys[slot] == key)
419 return 0;
420 slot = (slot + 1) & (k->cap - 1);
421 }
422 k->keys[slot] = key;
423 k->count++;
424 return 1;
425}
426static Search search(const Level *l, Position start, int source, int target, int prune,
427 size_t limit) {
428 size_t capacity = 1024, count = 1, expanded = 0, result_index = 0;
429 Node *nodes = allocate(capacity * sizeof(Node));
430 Known known = {0};
431 Search result = {0};
432 Position first = start;
433 first.player = representative(l, first);
434 nodes[0] = (Node){first, SIZE_MAX, 0, 0};
435 remember(&known, state_key(first));
436 while (expanded < count) {
437 if (limit && expanded == limit) {
438 result.outcome = 2;
439 break;
440 }
441 size_t parent = expanded++;
442 Node node = nodes[parent];
443 unsigned char seen[100];
444 reach(l, node.p, seen);
445 for (int i = 0; i < node.p.count; i++) {
446 int b = node.p.boxes[i], fixed = 0;
447 if (source >= 0)
448 for (int j = 0; j < start.count; j++)
449 if (start.boxes[j] != source && start.boxes[j] == b)
450 fixed = 1;
451 if (fixed)
452 continue;
453 for (int d = 0; d < 4; d++) {
454 if (!pushable(l, node.p, b, d, seen))
455 continue;
456 int destination = b + l->offsets[d];
457 Position next = move_box(node.p, b, destination);
458 if (prune && (l->dead[destination] || has_frozen(l, next)))
459 continue;
460 next.player = b;
461 next.player = representative(l, next);
462 if (!remember(&known, state_key(next)))
463 continue;
464 if (count == capacity) {
465 capacity *= 2;
466 Node *grown = realloc(nodes, capacity * sizeof(Node));
467 if (!grown)
468 exit(EXIT_FAILURE);
469 nodes = grown;
470 }
471 nodes[count] = (Node){next, parent, b, d};
472 result_index = count++;
473 if ((source >= 0 && destination == target) ||
474 (source < 0 && goals_on(l, next) == next.count)) {
475 result.outcome = 1;
476 goto finished;
477 }
478 }
479 }
480 }
481finished:
482 result.expanded = expanded;
483 result.states = count;
484 if (result.outcome == 1) {
485 size_t length = 0;
486 for (size_t n = result_index; nodes[n].parent != SIZE_MAX; n = nodes[n].parent)
487 length++;
488 size_t *chain = allocate(length * sizeof(size_t)), n = result_index;
489 for (size_t i = length; i > 0; i--) {
490 chain[i - 1] = n;
491 n = nodes[n].parent;
492 }
493 Position p = start;
494 for (size_t i = 0; i < length; i++) {
495 Node push = nodes[chain[i]];
496 int off = l->offsets[push.direction];
497 String path = walk(l, p, push.box - off);
498 append(&result.notation, text(&path));
499 dispose(&path);
500 append_char(&result.notation, (char)toupper((unsigned char)directions[push.direction]));
501 p = move_box(p, push.box, push.box + off);
502 p.player = push.box;
503 }
504 free(chain);
505 }
506 free(nodes);
507 free(known.keys);
508 return result;
509}

536static int map_cell(const Level *l, int c, int t) {
537 int r = c / l->w, col = c % l->w;
538 switch (t) {
539 case 0:
540 return col * l->h + l->h - 1 - r;
541 case 1:
542 return (l->h - 1 - r) * l->w + l->w - 1 - col;
543 case 2:
544 return (l->w - 1 - col) * l->h + r;
545 case 3:
546 return r * l->w + l->w - 1 - col;
547 default:
548 return (l->h - 1 - r) * l->w + col;
549 }
550}

555 sort_boxes(&p);
556 return p;
557}

565static void transform(Game *g, int t) {
566 Level old = g->level, next = {0};
567 next.h = (t == 0 || t == 2) ? old.w : old.h;
568 next.w = (t == 0 || t == 2) ? old.h : old.w;
569 next.size = old.size;
570 offsets(&next);
571 for (int c = 0; c < old.size; c++) {
572 int n = map_cell(&old, c, t);
573 next.walls[n] = old.walls[c];
574 next.goal[n] = old.goal[c];
575 next.interior[n] = old.interior[c];
576 }
577 analyze(&next);
578 g->level = next;
579 g->initial = map_position(&old, g->initial, t);
580 g->current = map_position(&old, g->current, t);
581 turn_letters(&g->history, t);
582 turn_letters(&g->redo, t);
583 for (int i = 0; i < g->snapshot_count; i++) {
584 g->snapshots[i].position = map_position(&old, g->snapshots[i].position, t);
585 turn_letters(&g->snapshots[i].history, t);
586 }
587 if (g->has_solution) {
588 g->solution_position = map_position(&old, g->solution_position, t);
589 turn_letters(&g->solution, t);
590 }
591 printf("transformed %dx%d\n", next.h, next.w);
592}
593static void bound_visit(const Level *l, Position p, int depth, unsigned used, int total,
594 int order[4], int best[4], int *minimum) {
595 if (depth == p.count) {
596 if (*minimum < 0 || total < *minimum) {
597 *minimum = total;
598 memcpy(best, order, (size_t)p.count * sizeof(int));
599 }
600 return;
601 }
602 for (int g = 0; g < l->goal_count; g++)
603 if (!(used & (1u << g)) && l->distance[g][p.boxes[depth]] >= 0) {
604 order[depth] = g;
605 bound_visit(l, p, depth + 1, used | (1u << g), total + l->distance[g][p.boxes[depth]],
606 order, best, minimum);
607 }
608}

651 "", "", "", "", "", "", "a", "a", "s", "n", "n", "", "t"};
652static int valid_argument(char kind, const char *s) {
653 if (kind == 'n')
654 return number_word(s);
655 if (kind == 'd')
656 return strlen(s) == 1 && strchr(directions, s[0]);
657 if (kind == 't')
658 return transform_index(s) >= 0;
659 if (kind == 'a') {
660 size_t n = strlen(s);
661 if (n < 1 || n > 8)
662 return 0;
663 for (size_t i = 0; i < n; i++)
664 if (s[i] < 'a' || s[i] > 'z')
665 return 0;
666 }
667 if (kind == 's') {
668 String out = {0};
669 int ok = decode(s, "udlrUDLR", &out);
670 dispose(&out);
671 return ok;
672 }
673 return 1;
674}
675static void command(Game *g, char **words, int count) {
676 int cmd = -1;
677 for (int i = 0; i < 26; i++)
678 if (!strcmp(words[0], commands[i]))
679 cmd = i;
680 if (cmd < 0) {
681 printf("error: unknown command %s\n", words[0]);
682 return;
683 }
684 if ((size_t)(count - 1) != strlen(formats[cmd])) {
685 error("bad arguments");
686 return;
687 }
688 for (int i = 1; i < count; i++)
689 if (!valid_argument(formats[cmd][i - 1], words[i])) {
690 error("bad arguments");
691 return;
692 }
693 Level *l = &g->level;
694 if (cmd == CMD_IMPORT) {
695 char rows[10][101];
696 int h = 0, w = -1;
697 const char *start = words[1];
698 while (1) {
699 const char *end = strchr(start, '|');
700 size_t n = end ? (size_t)(end - start) : strlen(start);
701 char part[101];
702 memcpy(part, start, n);
703 part[n] = 0;
704 String row = {0};
705 if (h == 10 || !decode(part, symbols, &row) || row.len < 1 || row.len > 10 ||
706 (w >= 0 && (size_t)w != row.len)) {
707 dispose(&row);
708 error("bad level");
709 return;
710 }
711 w = (int)row.len;
712 strcpy(rows[h++], text(&row));
713 dispose(&row);
714 if (!end)
715 break;
716 start = end + 1;
717 }
718 load(g, rows, h, w);
719 printf("loaded %dx%d\n", h, w);
720 return;
721 }
722 if (cmd == CMD_VALID) {
723 if (l->reason[0])
724 printf("invalid %s\n", l->reason);
725 else
726 puts("valid");
727 return;
728 }
729 if (l->reason[0]) {
730 error("invalid level");
731 return;
732 }
733 int cell_count = cmd == CMD_BRING ? 2 : (cmd == CMD_PUSH || cmd == CMD_PATH ? 1 : 0);
734 for (int i = 0; i < cell_count; i++) {
735 int r = atoi(words[1 + 2 * i]), c = atoi(words[2 + 2 * i]);
736 if (r < 1 || r > l->h || c < 1 || c > l->w) {
737 error("bad value");
738 return;
739 }
740 }
741 if ((cmd == CMD_UNDO || cmd == CMD_REDO) && atoi(words[1]) < 1) {
742 error("bad value");
743 return;
744 }
745 if ((cmd == CMD_SOLVE || cmd == CMD_HINT) && (atoi(words[1]) < 1 || atoi(words[1]) > 100000)) {
746 error("bad value");
747 return;
748 }
749 int b = cell_count ? (atoi(words[1]) - 1) * l->w + atoi(words[2]) - 1 : 0;
750 switch (cmd) {
751 case CMD_INFO:
752 printf("size %dx%d walls %d cells %d boxes %d dead %d\n", l->h, l->w,
753 count_flags(l->walls, l->size), count_flags(l->interior, l->size), g->current.count,
754 count_flags(l->dead, l->size));
755 break;
756 case CMD_SHOW:
757 for (int c = 0; c < l->size; c++) {
758 putchar(symbol(l, g->current, c));
759 if (c % l->w == l->w - 1)
760 putchar('\n');
761 }
762 break;
763 case CMD_EXPORT:
764 fputs("level ", stdout);
765 for (int r = 0; r < l->h; r++) {
766 if (r)
767 putchar('|');
768 for (int c = 0; c < l->w;) {
769 char ch = symbol(l, g->current, r * l->w + c);
770 int n = 1;
771 while (c + n < l->w && symbol(l, g->current, r * l->w + c + n) == ch)
772 n++;
773 if (n > 1)
774 printf("%d", n);
775 putchar(ch);
776 c += n;
777 }
778 }
779 putchar('\n');
780 break;
781 case CMD_MOVE:
782 case CMD_CHECK: {
783 String sequence = {0}, letters = {0};
784 decode(words[1], "udlrUDLR", &sequence);
785 int failure = execute(g, text(&sequence), &letters, cmd == CMD_MOVE);
786 if (cmd == CMD_MOVE) {
787 if (letters.len)
788 printf("moved %s\n", text(&letters));
789 if (failure)
790 printf("error: step %zu %s\n", letters.len + 1, failure == 1 ? "wall" : "blocked");
791 } else if (failure)
792 printf("check fail step %zu %s\n", letters.len + 1, failure == 1 ? "wall" : "blocked");
793 else {
794 Position p = g->current;
795 for (size_t i = 0; i < sequence.len; i++) {
796 char letter;
797 step(l, &p, sequence.data[i], &letter);
798 }
799 printf("check ok moves %zu pushes %d ", letters.len, pushes(text(&letters)));
800 progress(l, p);
801 putchar('\n');
802 }
803 dispose(&sequence);
804 dispose(&letters);
805 break;
806 }
807 case CMD_PUSH: {
808 int d = (int)(strchr(directions, words[3][0]) - directions), off = l->offsets[d];
809 unsigned char seen[100];
810 if (box_index(g->current, b) < 0) {
811 error("no box");
812 break;
813 }
814 if (!inside(l, b + off) || box_index(g->current, b + off) >= 0) {
815 error("blocked");
816 break;
817 }
818 reach(l, g->current, seen);
819 if (!inside(l, b - off) || !seen[b - off]) {
820 error("unreachable");
821 break;
822 }
823 String path = walk(l, g->current, b - off), letters = {0};
824 append_char(&path, directions[d]);
825 execute(g, text(&path), &letters, 1);
826 printf("pushed %s\n", text(&letters));
827 dispose(&path);
828 dispose(&letters);
829 break;
830 }
831 case CMD_BRING: {
832 int target = (atoi(words[3]) - 1) * l->w + atoi(words[4]) - 1;
833 if (box_index(g->current, b) < 0) {
834 error("no box");
835 break;
836 }
837 if (b == target) {
838 puts("brought 0");
839 break;
840 }
841 Search s = search(l, g->current, b, target, 0, 0);
842 if (s.outcome != 1)
843 error("no route");
844 else {
845 String letters = {0};
846 execute(g, text(&s.notation), &letters, 1);
847 printf("brought %d %s\n", pushes(text(&s.notation)), text(&s.notation));
848 dispose(&letters);
849 }
850 dispose(&s.notation);
851 break;
852 }
853 case CMD_PATH: {
854 unsigned char seen[100];
855 reach(l, g->current, seen);
856 if (!seen[b]) {
857 error("unreachable");
858 break;
859 }
860 String path = walk(l, g->current, b), letters = {0};
861 execute(g, text(&path), &letters, 1);
862 counted("path", &letters);
863 dispose(&path);
864 dispose(&letters);
865 break;
866 }
867 case CMD_UNDO:
868 case CMD_REDO: {
869 String *from = cmd == CMD_UNDO ? &g->history : &g->redo;
870 if (!from->len) {
871 error(cmd == CMD_UNDO ? "nothing to undo" : "nothing to redo");
872 break;
873 }
874 size_t k = (size_t)atoi(words[1]);
875 if (k > from->len)
876 k = from->len;
877 for (size_t i = 0; i < k; i++) {
878 char ch = from->data[--from->len];
879 from->data[from->len] = 0;
880 if (cmd == CMD_UNDO) {
881 char dir = (char)tolower((unsigned char)ch);
882 int off = l->offsets[strchr(directions, dir) - directions];
883 if (isupper((unsigned char)ch))
884 g->current = move_box(g->current, g->current.player + off, g->current.player);
885 g->current.player -= off;
886 append_char(&g->redo, dir);
887 } else {
888 char letter;
889 step(l, &g->current, ch, &letter);
890 append_char(&g->history, letter);
891 }
892 }
893 printf("%s %zu\n", cmd == CMD_UNDO ? "undone" : "redone", k);
894 break;
895 }
896 case CMD_RESET:
897 g->current = g->initial;
898 clear(&g->history);
899 clear(&g->redo);
900 puts("reset");
901 break;
902 case CMD_STATUS:
903 printf("moves %zu pushes %d ", g->history.len, pushes(text(&g->history)));
904 progress(l, g->current);
905 putchar('\n');
906 break;
907 case CMD_HISTORY:
908 counted("history", &g->history);
909 break;
910 case CMD_BOXES: {
911 unsigned char seen[100];
912 reach(l, g->current, seen);
913 printf("boxes %d\n", g->current.count);
914 for (int i = 0; i < g->current.count; i++) {
915 int box = g->current.boxes[i], best = -1;
916 char possible[5];
917 int n = 0;
918 for (int j = 0; j < l->goal_count; j++)
919 if (l->distance[j][box] >= 0 && (best < 0 || l->distance[j][box] < best))
920 best = l->distance[j][box];
921 for (int d = 0; d < 4; d++)
922 if (pushable(l, g->current, box, d, seen))
923 possible[n++] = directions[d];
924 possible[n] = 0;
925 fputs("box ", stdout);
926 cell(l, box);
927 if (best < 0)
928 fputs(" dead", stdout);
929 else
930 printf(" %d", best);
931 printf(" %s\n", n ? possible : "-");
932 }
933 break;
934 }
935 case CMD_REACH: {
936 unsigned char seen[100];
937 reach(l, g->current, seen);
938 printf("reach %d first ", count_flags(seen, l->size));
939 for (int c = 0; c < l->size; c++)
940 if (seen[c]) {
941 cell(l, c);
942 break;
943 }
944 putchar('\n');
945 break;
946 }
947 case CMD_DEADSQUARES:
948 printf("dead %d\n", count_flags(l->dead, l->size));
949 for (int c = 0; c < l->size; c++) {
950 putchar(l->walls[c] ? '#' : l->goal[c] ? '.' : l->dead[c] ? 'x' : '-');
951 if (c % l->w == l->w - 1)
952 putchar('\n');
953 }
954 break;
955 case CMD_DEADLOCK: {
956 int found = -1;
957 const char *why = "dead";
958 for (int i = 0; i < g->current.count; i++)
959 if (l->dead[g->current.boxes[i]]) {
960 found = g->current.boxes[i];
961 break;
962 }
963 if (found < 0) {
964 why = "frozen";
965 for (int i = 0; i < g->current.count; i++)
966 if (!l->goal[g->current.boxes[i]] &&
967 frozen(l, g->current, g->current.boxes[i], 0)) {
968 found = g->current.boxes[i];
969 break;
970 }
971 }
972 if (found < 0)
973 puts("no deadlock");
974 else {
975 printf("deadlock %s ", why);
976 cell(l, found);
977 putchar('\n');
978 }
979 break;
980 }
981 case CMD_BOUND: {
982 int order[4], best[4], minimum = -1;
983 bound_visit(l, g->current, 0, 0, 0, order, best, &minimum);
984 if (minimum < 0) {
985 puts("bound none");
986 break;
987 }
988 printf("bound %d\n", minimum);
989 for (int i = 0; i < g->current.count; i++) {
990 int box = g->current.boxes[i];
991 fputs("box ", stdout);
992 cell(l, box);
993 fputs(" goal ", stdout);
994 cell(l, l->goals[best[i]]);
995 printf(" pushes %d\n", l->distance[best[i]][box]);
996 }
997 break;
998 }
999 case CMD_SAVE:
1000 case CMD_RESTORE: {
1001 int index = -1;
1002 for (int i = 0; i < g->snapshot_count; i++)
1003 if (!strcmp(g->snapshots[i].name, words[1]))
1004 index = i;
1005 if (cmd == CMD_SAVE) {
1006 if (index < 0) {
1007 index = g->snapshot_count++;
1008 strcpy(g->snapshots[index].name, words[1]);
1009 }
1010 g->snapshots[index].position = g->current;
1011 assign(&g->snapshots[index].history, text(&g->history));
1012 printf("saved %s moves %zu\n", words[1], g->history.len);
1013 } else if (index < 0)
1014 printf("error: no snapshot %s\n", words[1]);
1015 else {
1016 g->current = g->snapshots[index].position;
1017 assign(&g->history, text(&g->snapshots[index].history));
1018 clear(&g->redo);
1019 printf("restored %s moves %zu\n", words[1], g->history.len);
1020 }
1021 break;
1022 }
1023 case CMD_SOLVE:
1024 case CMD_HINT: {
1025 if (cmd == CMD_SOLVE) {
1026 g->has_solution = 0;
1027 clear(&g->solution);
1028 }
1029 if (goals_on(l, g->current) == g->current.count) {
1030 puts("already solved");
1031 break;
1032 }
1033 Search s = search(l, g->current, -1, -1, 1, (size_t)atoi(words[1]));
1034 if (s.outcome != 1)
1035 printf("%s expanded %zu states %zu\n", s.outcome == 2 ? "limit" : "no solution",
1036 s.expanded, s.states);
1037 else if (cmd == CMD_SOLVE) {
1038 printf("solution pushes %d moves %zu expanded %zu states %zu\n%s\n",
1039 pushes(text(&s.notation)), s.notation.len, s.expanded, s.states,
1040 text(&s.notation));
1041 assign(&g->solution, text(&s.notation));
1042 g->solution_position = g->current;
1043 g->has_solution = 1;
1044 } else {
1045 size_t i = 0;
1046 while (!isupper((unsigned char)s.notation.data[i]))
1047 i++;
1048 s.notation.data[i + 1] = 0;
1049 String letters = {0};
1050 execute(g, text(&s.notation), &letters, 1);
1051 printf("hint %s\n", text(&s.notation));
1052 dispose(&letters);
1053 }
1054 dispose(&s.notation);
1055 break;
1056 }
1057 case CMD_PLAY:
1058 if (!g->has_solution || !same(g->current, g->solution_position))
1059 error("no solution");
1060 else {
1061 String letters = {0};
1062 execute(g, text(&g->solution), &letters, 1);
1063 printf("played pushes %d moves %zu\n", pushes(text(&g->solution)), g->solution.len);
1064 dispose(&letters);
1065 }
1066 break;
1067 case CMD_TRANSFORM:
1068 transform(g, transform_index(words[1]));
1069 break;
1070 }
1071}
1072static int read_line(String *line) {
1073 clear(line);
1074 int ch;
1075 while ((ch = getchar()) != EOF && ch != '\n')
1076 append_char(line, (char)ch);
1077 if (line->len && line->data[line->len - 1] == '\r')
1078 line->data[--line->len] = 0;
1079 return ch != EOF || line->len > 0;
1080}
1081static void read_row(char row[101]) {
1082 String line = {0};
1083 read_line(&line);
1084 strcpy(row, text(&line));
1085 dispose(&line);
1086}
1087int main(void) {
1088 String line = {0};
1089 char rows[10][101];
1090 int h, w;
1091 if (!read_line(&line) || sscanf(text(&line), "%d %d", &h, &w) != 2) {
1092 dispose(&line);
1093 return 0;
1094 }
1095 for (int i = 0; i < h; i++)
1096 read_row(rows[i]);
1097 Game *g = calloc(1, sizeof(Game));
1098 if (!g)
1099 return EXIT_FAILURE;
1100 load(g, rows, h, w);
1101 read_line(&line);
1102 int count = atoi(text(&line));
1103 for (int n = 0; n < count; n++) {
1104 if (!read_line(&line))
1105 break;
1106 char **words = allocate((line.len + 1) * sizeof(char *));
1107 int word_count = 0;
1108 for (char *word = strtok(line.data, " "); word; word = strtok(NULL, " "))
1109 words[word_count++] = word;
1110 if (!strcmp(words[0], "LOAD")) {
1111 h = atoi(words[1]);
1112 w = atoi(words[2]);
1113 int bad = 0;
1114 for (int i = 0; i < h; i++) {
1115 read_row(rows[i]);
1116 if (!bad && (strlen(rows[i]) != (size_t)w || strspn(rows[i], symbols) != (size_t)w))
1117 bad = i + 1;
1118 }
1119 if (bad)
1120 printf("error: bad row %d\n", bad);
1121 else {
1122 load(g, rows, h, w);
1123 printf("loaded %dx%d\n", h, w);
1124 }
1125 } else
1126 command(g, words, word_count);
1127 free(words);
1128 }
1129 for (int i = 0; i < g->snapshot_count; i++)
1130 dispose(&g->snapshots[i].history);
1131 dispose(&g->history);
1132 dispose(&g->redo);
1133 dispose(&g->solution);
1134 dispose(&line);
1135 free(g);
1136 return 0;
1137}
