17local function split(text, separator)
18 local parts, start = {}, 1
19 while true do
20 local stop = text:find(separator, start, true)
21 parts[#parts + 1] = text:sub(start, stop and stop - 1 or #text)
22 if not stop then return parts end
23 start = stop + 1
24 end
25end

52local function decode(text, allowed)
53 local parts, index = {}, 1
54 while index <= #text do
55 local start = index
56 while index <= #text and text:sub(index, index):match("%d") do index = index + 1 end
57 local digits = text:sub(start, index - 1)
58 if #digits > 2 or digits:sub(1, 1) == "0" or index > #text then return nil end
59 local letter = text:sub(index, index)
60 if not allowed:find(letter, 1, true) then return nil end
61 parts[#parts + 1] = letter:rep(digits == "" and 1 or tonumber(digits))
62 index = index + 1
63 end
64 return table.concat(parts)
65end
66
67local function encode(text)
68 local parts, index = {}, 1
69 while index <= #text do
70 local stop = index + 1
71 while stop <= #text and text:sub(stop, stop) == text:sub(index, index) do stop = stop + 1 end
72 local run = stop - index
73 parts[#parts + 1] = (run > 1 and tostring(run) or "") .. text:sub(index, index)
74 index = stop
75 end
76 return table.concat(parts)
77end

106function Level.new(height, width, walls, goals, interior)
107 local self = setmetatable({
108 height = height, width = width, size = height * width, walls = walls, goals = goals,
109 interior = interior, offsets = { u = -width, d = width, l = -1, r = 1 },
110 goal_cells = {}, distances = {}, dead = {},
111 }, Level)
112 for cell = 1, self.size do
113 if goals[cell] then self.goal_cells[#self.goal_cells + 1] = cell end
114 end
115 return self
116end
117
118function Level:cell_text(cell)
119 return ((cell - 1) // self.width + 1) .. " " .. ((cell - 1) % self.width + 1)
120end
121
122function Level:analyze()
123 self.distances, self.dead = {}, {}
124 for index, goal in ipairs(self.goal_cells) do
125 local distance, queue = { [goal] = 0 }, { goal }
126 local head = 1
127 while head <= #queue do
128 local cell = queue[head]
129 head = head + 1
130 for _, direction in ipairs(DIRECTIONS) do
131 local offset = self.offsets[direction]
132 local box = cell - offset
133 if self.interior[box] and self.interior[box - offset] and distance[box] == nil then
134 distance[box] = distance[cell] + 1
135 queue[#queue + 1] = box
136 end
137 end
138 end
139 self.distances[index] = distance
140 end
141 for cell = 1, self.size do
142 if self.interior[cell] and self:goal_distance(cell) == nil then self.dead[cell] = true end
143 end
144end
145
146function Level:goal_distance(cell)
147 local best
148 for _, distance in ipairs(self.distances) do
149 if distance[cell] ~= nil and (best == nil or distance[cell] < best) then best = distance[cell] end
150 end
151 return best
152end

154function Level:on_goals(boxes)

156 for _, box in ipairs(boxes) do
157 if self.goals[box] then result = result + 1 end

162function Level:solved(boxes)

166function Level:reachable(player, boxes)
167 local seen, queue, head = { [player] = true }, { player }, 1
168 while head <= #queue do
169 local cell = queue[head]
170 head = head + 1
171 for _, direction in ipairs(DIRECTIONS) do
172 local neighbor = cell + self.offsets[direction]
173 if self.interior[neighbor] and not seen[neighbor] and not contains(boxes, neighbor) then
174 seen[neighbor] = true
175 queue[#queue + 1] = neighbor
176 end
177 end
178 end
179 return seen
180end

182function Level:representative(player, boxes)
183 return first(self:reachable(player, boxes), self.size)

186function Level:walk_path(start, target, boxes)
187 local came, queue, head = { [start] = true }, { start }, 1
188 while head <= #queue do
189 local cell = queue[head]
190 head = head + 1
191 for _, direction in ipairs(DIRECTIONS) do
192 local neighbor = cell + self.offsets[direction]
193 if self.interior[neighbor] and not contains(boxes, neighbor) and not came[neighbor] then
194 came[neighbor] = { cell, direction }
195 queue[#queue + 1] = neighbor
196 end
197 end
198 end
199 local reversed = {}
200 while target ~= start do
201 local entry = came[target]
202 reversed[#reversed + 1] = entry[2]
203 target = entry[1]
204 end
205 local path = {}
206 for index = #reversed, 1, -1 do path[#path + 1] = reversed[index] end
207 return table.concat(path)
208end
209
210function Level:step(position, direction)
211 local offset = self.offsets[direction]
212 local target = position.player + offset
213 if self.walls[target] then return nil, "wall" end
214 if not contains(position.boxes, target) then
215 return { player = target, boxes = position.boxes }, direction
216 end
217 local beyond = target + offset
218 if self.walls[beyond] or contains(position.boxes, beyond) then return nil, "blocked" end
219 return { player = target, boxes = move_box(position.boxes, target, beyond) }, direction:upper()
220end
221
222function Level:pushable(box, direction, boxes, reachable)
223 local offset = self.offsets[direction]
224 return reachable[box - offset] and self.interior[box + offset] and not contains(boxes, box + offset)
225end
226
227function Level:side_blocked(box, a, b, boxes, assumed)
228 if self.walls[a] or self.walls[b] or assumed[a] or assumed[b] then return true end
229 if self.dead[a] and self.dead[b] then return true end
230 local extended = {}
231 for cell in pairs(assumed) do extended[cell] = true end
232 extended[box] = true
233 return (contains(boxes, a) and self:frozen(a, boxes, extended))
234 or (contains(boxes, b) and self:frozen(b, boxes, extended))
235end
236
237function Level:frozen(box, boxes, assumed)
238 return self:side_blocked(box, box - 1, box + 1, boxes, assumed)
239 and self:side_blocked(box, box - self.width, box + self.width, boxes, assumed)
240end
241
242function Level:has_frozen_box(boxes)
243 for _, box in ipairs(boxes) do
244 if not self.goals[box] and self:frozen(box, boxes, {}) then return true end
245 end
246 return false
247end

249function Level:rows(position, dead_map)
250 local rows = {}
251 for row = 1, self.height do
252 local symbols = {}
253 for column = 1, self.width do
254 local cell = (row - 1) * self.width + column
255 local symbol
256 if self.walls[cell] then symbol = "#"
257 elseif dead_map then symbol = self.goals[cell] and "." or (self.dead[cell] and "x" or "-")
258 elseif cell == position.player then symbol = self.goals[cell] and "+" or "@"
259 elseif contains(position.boxes, cell) then symbol = self.goals[cell] and "*" or "$"
260 else symbol = self.goals[cell] and "." or "-" end
261 symbols[column] = symbol
262 end
263 rows[row] = table.concat(symbols)
264 end
265 return rows
266end

268function Level:map_cell(kind, cell)
269 local row, column = (cell - 1) // self.width, (cell - 1) % self.width
270 if kind == "ROT90" then return column * self.height + self.height - row end
271 if kind == "ROT180" then return (self.height - 1 - row) * self.width + self.width - column end
272 if kind == "ROT270" then return (self.width - 1 - column) * self.height + row + 1 end
273 if kind == "FLIPH" then return row * self.width + self.width - column end
274 return (self.height - 1 - row) * self.width + column + 1
275end

277function Level:map_position(kind, position)
278 local boxes = {}
279 for index, box in ipairs(position.boxes) do boxes[index] = self:map_cell(kind, box) end
280 table.sort(boxes)
281 return { player = self:map_cell(kind, position.player), boxes = boxes }
282end

284local function build_level(rows)
285 local height, width = #rows, #rows[1]
286 local walls, goals, boxes, players = {}, {}, {}, {}
287 for row, text in ipairs(rows) do
288 for column = 1, width do
289 local cell, symbol = (row - 1) * width + column, text:sub(column, column)
290 if symbol == "#" then walls[cell] = true end
291 if symbol == "." or symbol == "*" or symbol == "+" then goals[cell] = true end
292 if symbol == "$" or symbol == "*" then boxes[#boxes + 1] = cell end
293 if symbol == "@" or symbol == "+" then players[#players + 1] = cell end
294 end
295 end
296 local interior = {}
297 if #players == 1 then
298 local queue, head = { players[1] }, 1
299 interior[players[1]] = true
300 while head <= #queue do
301 local cell = queue[head]
302 head = head + 1
303 local row, column = (cell - 1) // width, (cell - 1) % width
304 for _, delta in ipairs({ { -1, 0 }, { 1, 0 }, { 0, -1 }, { 0, 1 } }) do
305 local r, c = row + delta[1], column + delta[2]
306 local neighbor = r * width + c + 1
307 if r >= 0 and r < height and c >= 0 and c < width and not walls[neighbor] and not interior[neighbor] then
308 interior[neighbor] = true
309 queue[#queue + 1] = neighbor
310 end
311 end
312 end
313 end
314 local level = Level.new(height, width, walls, goals, interior)
315 if #players ~= 1 then level.reason = "players " .. #players
316 elseif #boxes == 0 then level.reason = "no boxes"
317 elseif #boxes ~= #level.goal_cells then level.reason = "boxes " .. #boxes .. " goals " .. #level.goal_cells
318 else
319 for cell = 1, level.size do
320 local row, column = (cell - 1) // width, (cell - 1) % width
321 if interior[cell] and (row == 0 or row == height - 1 or column == 0 or column == width - 1) then
322 level.reason = "open " .. level:cell_text(cell)
323 break
324 end
325 end
326 if not level.reason then
327 for cell = 1, level.size do
328 if (goals[cell] or contains(boxes, cell)) and not interior[cell] then
329 level.reason = "outside " .. level:cell_text(cell)
330 break
331 end
332 end
333 end
334 end
335 if not level.reason then level:analyze() end
336 return level, { player = players[1], boxes = boxes }
337end

339local function state_key(boxes, representative)
340 return table.concat(boxes, ",") .. ":" .. representative
341end

343local function push_search(level, start, fixed, target, pruning, limit)
344 local representative = level:representative(start.player, start.boxes)
345 local nodes = { { boxes = start.boxes, representative = representative } }
346 local known = { [state_key(start.boxes, representative)] = true }
347 local expanded = 0
348 while expanded < #nodes do
349 if expanded == limit then return { outcome = "limit", expanded = expanded, states = #nodes } end
350 local parent = expanded + 1
351 local node = nodes[parent]
352 expanded = expanded + 1
353 local reachable = level:reachable(node.representative, node.boxes)
354 for _, box in ipairs(node.boxes) do
355 if not contains(fixed, box) then
356 for _, direction in ipairs(DIRECTIONS) do
357 if level:pushable(box, direction, node.boxes, reachable) then
358 local destination = box + level.offsets[direction]
359 local boxes = move_box(node.boxes, box, destination)
360 if not pruning or (not level.dead[destination] and not level:has_frozen_box(boxes)) then
361 local next_rep = level:representative(box, boxes)
362 local key = state_key(boxes, next_rep)
363 if not known[key] then
364 known[key] = true
365 nodes[#nodes + 1] = { boxes = boxes, representative = next_rep, parent = parent, box = box, direction = direction }
366 if (target and destination == target) or (not target and level:solved(boxes)) then
367 local reversed, index = {}, #nodes
368 while nodes[index].parent do
369 reversed[#reversed + 1] = nodes[index]
370 index = nodes[index].parent
371 end
372 local position, pieces = start, {}
373 for i = #reversed, 1, -1 do
374 local push = reversed[i]
375 local offset = level.offsets[push.direction]
376 pieces[#pieces + 1] = level:walk_path(position.player, push.box - offset, position.boxes)
377 pieces[#pieces + 1] = push.direction:upper()
378 position = { player = push.box, boxes = move_box(position.boxes, push.box, push.box + offset) }
379 end
380 return { outcome = "found", expanded = expanded, states = #nodes, notation = table.concat(pieces) }
381 end
382 end
383 end
384 end
385 end
386 end
387 end
388 end
389 return { outcome = "no solution", expanded = expanded, states = #nodes }
390end

401function Game:load(rows)
402 self.level, self.initial = build_level(rows)
403 self.current = self.initial
404 self.history, self.redo_list, self.snapshots, self.solution = "", "", {}, nil
405end

407function Game:simulate(directions)
408 local position, letters = self.current, {}
409 for index = 1, #directions do
410 local next_position, letter = self.level:step(position, directions:sub(index, index):lower())
411 if not next_position then
412 return position, table.concat(letters), "step " .. index .. " " .. letter
413 end
414 position, letters[#letters + 1] = next_position, letter
415 end
416 return position, table.concat(letters)
417end
418
419function Game:take_steps(directions)
420 local position, letters, failure = self:simulate(directions)
421 self.current, self.history = position, self.history .. letters
422 if #letters > 0 then self.redo_list = "" end
423 return letters, failure
424end

426function Game:cell(arguments, index)
427 return (tonumber(arguments[index]) - 1) * self.level.width + tonumber(arguments[index + 1])
428end

430function Game:progress(boxes)
431 local goals = self.level:on_goals(boxes)
432 return "goals " .. goals .. "/" .. #boxes .. (goals == #boxes and " solved" or "")
433end
434
435function Game:search_solution(limit)
436 if self.level:solved(self.current.boxes) then return "already solved" end
437 local result = push_search(self.level, self.current, {}, nil, true, limit)
438 local counts = "expanded " .. result.expanded .. " states " .. result.states
439 if result.outcome ~= "found" then return result.outcome .. " " .. counts end
440 local notation = result.notation
441 return "solution pushes " .. push_count(notation) .. " moves " .. #notation .. " " .. counts, notation
442end

456function Game:process(name, a)
457 local level, position = self.level, self.current
458 if name == "IMPORT" then
459 local rows = {}
460 for _, part in ipairs(split(a[1], "|")) do
461 local row = decode(part, LEVEL_CHARACTERS)
462 if not row then fail("bad level") end
463 rows[#rows + 1] = row
464 end
465 if #rows > 10 or #rows[1] < 1 or #rows[1] > 10 then fail("bad level") end
466 for _, row in ipairs(rows) do if #row ~= #rows[1] then fail("bad level") end end
467 self:load(rows)
468 self:emit("loaded", #rows .. "x" .. #rows[1])
469 elseif name == "VALID" then self:emit(level.reason and "invalid " .. level.reason or "valid")
470 elseif name == "INFO" then
471 self:emit("size", level.height .. "x" .. level.width, "walls", count(level.walls), "cells", count(level.interior), "boxes", #position.boxes, "dead", count(level.dead))
472 elseif name == "SHOW" then
473 for _, row in ipairs(level:rows(position)) do self:emit(row) end
474 elseif name == "EXPORT" then
475 local rows = level:rows(position)
476 for index, row in ipairs(rows) do rows[index] = encode(row) end
477 self:emit("level", table.concat(rows, "|"))
478 elseif name == "MOVE" then
479 local letters, failure = self:take_steps(decode(a[1], STEP_CHARACTERS))
480 if #letters > 0 then self:emit("moved", letters) end
481 if failure then fail(failure) end
482 elseif name == "PUSH" then
483 local box, direction = self:cell(a, 1), a[3]
484 local offset = level.offsets[direction]
485 if not contains(position.boxes, box) then fail("no box") end
486 if not level.interior[box + offset] or contains(position.boxes, box + offset) then fail("blocked") end
487 if not level:reachable(position.player, position.boxes)[box - offset] then fail("unreachable") end
488 local letters = self:take_steps(level:walk_path(position.player, box - offset, position.boxes) .. direction)
489 self:emit("pushed", letters)
490 elseif name == "BRING" then
491 local source, target = self:cell(a, 1), self:cell(a, 3)
492 if not contains(position.boxes, source) then fail("no box") end
493 if source == target then self:emit("brought", 0); return end
494 local fixed = {}
495 for _, box in ipairs(position.boxes) do if box ~= source then fixed[#fixed + 1] = box end end
496 local result = push_search(level, position, fixed, target, false)
497 if result.outcome ~= "found" then fail("no route") end
498 self:take_steps(result.notation)
499 self:emit("brought", push_count(result.notation), result.notation)
500 elseif name == "PATH" then
501 local target = self:cell(a, 1)
502 if not level:reachable(position.player, position.boxes)[target] then fail("unreachable") end
503 local letters = self:take_steps(level:walk_path(position.player, target, position.boxes))
504 self:emit(counted("path", letters))
505 elseif name == "UNDO" then
506 if #self.history == 0 then fail("nothing to undo") end
507 local n = math.min(tonumber(a[1]), #self.history)
508 for _ = 1, n do
509 local letter = self.history:sub(-1)
510 self.history = self.history:sub(1, -2)
511 local direction, player = letter:lower(), self.current.player
512 local offset, boxes = level.offsets[direction], self.current.boxes
513 if letter ~= direction then boxes = move_box(boxes, player + offset, player) end
514 self.current = { player = player - offset, boxes = boxes }
515 self.redo_list = self.redo_list .. direction
516 end
517 self:emit("undone", n)
518 elseif name == "REDO" then
519 if #self.redo_list == 0 then fail("nothing to redo") end
520 local n = math.min(tonumber(a[1]), #self.redo_list)
521 for _ = 1, n do
522 local direction = self.redo_list:sub(-1)
523 self.redo_list = self.redo_list:sub(1, -2)
524 local next_position, letter = level:step(self.current, direction)
525 self.current, self.history = next_position, self.history .. letter
526 end
527 self:emit("redone", n)
528 elseif name == "RESET" then
529 self.current, self.history, self.redo_list = self.initial, "", ""
530 self:emit("reset")
531 elseif name == "STATUS" then
532 self:emit("moves", #self.history, "pushes", push_count(self.history), self:progress(position.boxes))
533 elseif name == "HISTORY" then self:emit(counted("history", self.history))
534 elseif name == "BOXES" then
535 local reachable = level:reachable(position.player, position.boxes)
536 self:emit("boxes", #position.boxes)
537 for _, box in ipairs(position.boxes) do
538 local directions = ""
539 for _, direction in ipairs(DIRECTIONS) do
540 if level:pushable(box, direction, position.boxes, reachable) then directions = directions .. direction end
541 end
542 local distance = level:goal_distance(box)
543 self:emit("box", level:cell_text(box), distance == nil and "dead" or distance, directions == "" and "-" or directions)
544 end
545 elseif name == "REACH" then
546 local reachable = level:reachable(position.player, position.boxes)
547 self:emit("reach", count(reachable), "first", level:cell_text(first(reachable, level.size)))
548 elseif name == "DEADSQUARES" then
549 self:emit("dead", count(level.dead))
550 for _, row in ipairs(level:rows(position, true)) do self:emit(row) end
551 elseif name == "DEADLOCK" then
552 for _, box in ipairs(position.boxes) do
553 if level.dead[box] then self:emit("deadlock dead", level:cell_text(box)); return end
554 end
555 for _, box in ipairs(position.boxes) do
556 if not level.goals[box] and level:frozen(box, position.boxes, {}) then
557 self:emit("deadlock frozen", level:cell_text(box)); return
558 end
559 end
560 self:emit("no deadlock")
561 elseif name == "BOUND" then
562 local best, best_total, order, used = nil, nil, {}, {}
563 local function assign(index, total)
564 if index > #position.boxes then
565 if best_total == nil or total < best_total then best, best_total = copy(order), total end
566 return
567 end
568 for goal = 1, #level.goal_cells do
569 local distance = level.distances[goal][position.boxes[index]]
570 if not used[goal] and distance ~= nil then
571 used[goal], order[index] = true, goal
572 assign(index + 1, total + distance)
573 used[goal] = nil
574 end
575 end
576 end
577 assign(1, 0)
578 if not best then self:emit("bound none"); return end
579 self:emit("bound", best_total)
580 for index, box in ipairs(position.boxes) do
581 local goal = best[index]
582 self:emit("box", level:cell_text(box), "goal", level:cell_text(level.goal_cells[goal]), "pushes", level.distances[goal][box])
583 end
584 elseif name == "SAVE" then
585 self.snapshots[a[1]] = { position = position, history = self.history }
586 self:emit("saved", a[1], "moves", #self.history)
587 elseif name == "RESTORE" then
588 local snapshot = self.snapshots[a[1]]
589 if not snapshot then fail("no snapshot " .. a[1]) end
590 self.current, self.history, self.redo_list = snapshot.position, snapshot.history, ""
591 self:emit("restored", a[1], "moves", #self.history)
592 elseif name == "CHECK" then
593 local result, letters, failure = self:simulate(decode(a[1], STEP_CHARACTERS))
594 if failure then self:emit("check fail", failure)
595 else self:emit("check ok moves", #letters, "pushes", push_count(letters), self:progress(result.boxes)) end
596 elseif name == "SOLVE" or name == "HINT" then
597 local summary, notation = self:search_solution(tonumber(a[1]))
598 if name == "SOLVE" then
599 self:emit(summary)
600 self.solution = notation and { notation = notation, position = position } or nil
601 if notation then self:emit(notation) end
602 elseif not notation then self:emit(summary)
603 else
604 local first_push = notation:find("[UDLR]")
605 local part = notation:sub(1, first_push)
606 self:take_steps(part)
607 self:emit("hint", part)
608 end
609 elseif name == "PLAY" then
610 if not self.solution or not same_position(position, self.solution.position) then fail("no solution") end
611 local notation = self.solution.notation
612 self:take_steps(notation)
613 self:emit("played pushes", push_count(notation), "moves", #notation)
614 elseif name == "TRANSFORM" then
615 local kind, walls, goals, interior = a[1], {}, {}, {}
616 for cell = 1, level.size do
617 local moved = level:map_cell(kind, cell)
618 walls[moved], goals[moved], interior[moved] = level.walls[cell], level.goals[cell], level.interior[cell]
619 end
620 local rotated = kind == "ROT90" or kind == "ROT270"
621 self.level = Level.new(rotated and level.width or level.height, rotated and level.height or level.width, walls, goals, interior)
622 self.level:analyze()
623 self.initial, self.current = level:map_position(kind, self.initial), level:map_position(kind, position)
624 self.history, self.redo_list = turn_letters(self.history, kind), turn_letters(self.redo_list, kind)
625 for _, snapshot in pairs(self.snapshots) do
626 snapshot.position, snapshot.history = level:map_position(kind, snapshot.position), turn_letters(snapshot.history, kind)
627 end
628 if self.solution then
629 self.solution.position, self.solution.notation = level:map_position(kind, self.solution.position), turn_letters(self.solution.notation, kind)
630 end
631 self:emit("transformed", self.level.height .. "x" .. self.level.width)
632 end
633end

635local function valid_argument(kind, text)
636 if kind == "number" then
637 return #text <= 6 and text:match("^%d+$") ~= nil and (#text == 1 or text:sub(1, 1) ~= "0")
638 elseif kind == "steps" then return decode(text, STEP_CHARACTERS) ~= nil
639 elseif kind == "direction" then return #text == 1 and ("udlr"):find(text, 1, true) ~= nil
640 elseif kind == "name" then return #text <= 8 and text:match("^[a-z]+$") ~= nil
641 elseif kind == "transform" then return TURNS[text] ~= nil end
642 return true
643end

645function Game:execute(words, rows)
646 local ok, problem = pcall(function()
647 local name = words[1]
648 if name == "LOAD" then
649 local width = tonumber(words[3])
650 for index, row in ipairs(rows) do
651 if #row ~= width or row:find("[^#%-%.$*@+]") then fail("bad row " .. index) end
652 end
653 self:load(rows)
654 self:emit("loaded", #rows .. "x" .. width)
655 return
656 end
657 local kinds = COMMANDS[name]
658 if not kinds then fail("unknown command " .. name) end
659 if #words ~= #kinds + 1 then fail("bad arguments") end
660 local arguments = {}
661 for index, kind in ipairs(kinds) do
662 arguments[index] = words[index + 1]
663 if not valid_argument(kind, arguments[index]) then fail("bad arguments") end
664 end
665 if name ~= "IMPORT" and name ~= "VALID" then
666 if self.level.reason then fail("invalid level") end
667 local cells = name == "BRING" and 2 or ((name == "PUSH" or name == "PATH") and 1 or 0)
668 for index = 1, cells * 2, 2 do
669 local row, column = tonumber(arguments[index]), tonumber(arguments[index + 1])
670 if row < 1 or row > self.level.height or column < 1 or column > self.level.width then fail("bad value") end
671 end
672 if (name == "UNDO" or name == "REDO") and tonumber(arguments[1]) < 1 then fail("bad value") end
673 if (name == "SOLVE" or name == "HINT") and (tonumber(arguments[1]) < 1 or tonumber(arguments[1]) > 100000) then fail("bad value") end
674 end
675 self:process(name, arguments)
676 end)
677 if not ok then
678 if type(problem) ~= "table" then error(problem, 0) end
679 self:emit("error:", problem.message)
680 end
681end

683local function main()
684 local header = split(io.read("l"), " ")
685 local rows = {}
686 for index = 1, tonumber(header[1]) do rows[index] = io.read("l") end
687 local game = setmetatable({ output = {} }, Game)
688 game:load(rows)
689 local commands = tonumber(io.read("l"))
690 for _ = 1, commands do
691 local words, loaded = split(io.read("l"), " "), {}
692 if words[1] == "LOAD" then
693 for index = 1, tonumber(words[2]) do loaded[index] = io.read("l") end
694 end
695 game:execute(words, loaded)
696 end
697 io.write(table.concat(game.output, "\n"), "\n")
698end
