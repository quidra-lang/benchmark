51function decode(string $text, string $allowed): ?string
52{
53 $result = '';
54 for ($index = 0; $index < strlen($text); $index++) {
55 $start = $index;
56 while ($index < strlen($text) && ctype_digit($text[$index])) {
57 $index++;
58 }
59 $count = substr($text, $start, $index - $start);
60 if (strlen($count) > 2 || str_starts_with($count, '0') || $index === strlen($text)) {
61 return null;
62 }
63 if (!str_contains($allowed, $text[$index])) {
64 return null;
65 }
66 $result .= str_repeat($text[$index], $count === '' ? 1 : (int) $count);
67 }
68 return $result;
69}

71function encode(string $text): string
72{
73 $result = '';
74 for ($start = 0; $start < strlen($text); $start = $end) {
75 $end = $start + 1;
76 while ($end < strlen($text) && $text[$end] === $text[$start]) {
77 $end++;
78 }
79 $count = $end - $start;
80 $result .= ($count > 1 ? (string) $count : '') . $text[$start];
81 }
82 return $result;
83}

85function validArgument(string $kind, string $text): bool
86{
87 return match ($kind) {
88 'number' => preg_match('/\A(?:0|[1-9][0-9]{0,5})\z/', $text) === 1,
89 'steps' => decode($text, STEP_CHARACTERS) !== null,
90 'direction' => in_array($text, DIRECTIONS, true),
91 'name' => preg_match('/\A[a-z]{1,8}\z/', $text) === 1,
92 'transform' => array_key_exists($text, TURNS),
93 default => true,
94 };
95}

114final class Level
115{
116 public int $height;
117 public int $width;
118 public array $walls;
119 public array $goals;
120 public array $interior;
121 public array $goalCells;
122 public array $offsets;
123 public array $distances = [];
124 public array $dead = [];
125 public ?string $reason;
126 public Position $loadedPosition;
127
128 public function __construct(array $rows)
129 {
130 $this->height = count($rows);
131 $this->width = strlen($rows[0]);
132 $cells = str_split(implode('', $rows));
133 $this->walls = array_map(fn(string $letter): bool => $letter === '#', $cells);
134 $this->goals = array_map(fn(string $letter): bool => str_contains('.*+', $letter), $cells);
135 $this->goalCells = array_keys(array_filter($this->goals));
136 $players = array_keys(array_filter($cells, fn(string $letter): bool => str_contains('@+', $letter)));
137 $boxes = array_keys(array_filter($cells, fn(string $letter): bool => str_contains('$*', $letter)));
138 $this->offsets = ['u' => -$this->width, 'd' => $this->width, 'l' => -1, 'r' => 1];
139 $this->interior = array_fill(0, $this->size(), false);
140 if (count($players) === 1) {
141 $this->interior[$players[0]] = true;
142 $stack = [$players[0]];
143 while ($stack !== []) {
144 $cell = array_pop($stack);
145 $row = intdiv($cell, $this->width);
146 $column = $cell % $this->width;
147 foreach ([[$row - 1, $column], [$row + 1, $column], [$row, $column - 1], [$row, $column + 1]] as [$r, $c]) {
148 if ($r < 0 || $r >= $this->height || $c < 0 || $c >= $this->width) {
149 continue;
150 }
151 $neighbor = $r * $this->width + $c;
152 if (!$this->walls[$neighbor] && !$this->interior[$neighbor]) {
153 $this->interior[$neighbor] = true;
154 $stack[] = $neighbor;
155 }
156 }
157 }
158 }
159 $this->reason = $this->invalidity(count($players), $boxes);
160 if ($this->reason === null) {
161 $this->analyze();
162 }
163 $this->loadedPosition = new Position($players[0] ?? null, $boxes);
164 }
165

184 }

196 private function invalidity(int $players, array $boxes): ?string
197 {
198 if ($players !== 1) {
199 return "players $players";
200 }
201 if ($boxes === []) {
202 return 'no boxes';
203 }
204 if (count($boxes) !== count($this->goalCells)) {
205 return 'boxes ' . count($boxes) . ' goals ' . count($this->goalCells);
206 }
207 for ($cell = 0; $cell < $this->size(); $cell++) {
208 $row = intdiv($cell, $this->width);
209 $column = $cell % $this->width;
210 if ($this->interior[$cell] && ($row === 0 || $row === $this->height - 1 || $column === 0 || $column === $this->width - 1)) {
211 return 'open ' . $this->cellText($cell);
212 }
213 }
214 for ($cell = 0; $cell < $this->size(); $cell++) {
215 if (($this->goals[$cell] || in_array($cell, $boxes, true)) && !$this->interior[$cell]) {
216 return 'outside ' . $this->cellText($cell);
217 }
218 }
219 return null;
220 }
221
222 private function analyze(): void
223 {
224 foreach ($this->goalCells as $goal) {
225 $distance = array_fill(0, $this->size(), null);
226 $distance[$goal] = 0;
227 $queue = [$goal];
228 for ($head = 0; $head < count($queue); $head++) {
229 $cell = $queue[$head];
230 foreach ($this->offsets as $offset) {
231 $box = $cell - $offset;
232 if ($this->interior[$box] && $this->interior[$box - $offset] && $distance[$box] === null) {
233 $distance[$box] = $distance[$cell] + 1;
234 $queue[] = $box;
235 }
236 }
237 }
238 $this->distances[] = $distance;
239 }
240 $this->dead = array_fill(0, $this->size(), false);
241 for ($cell = 0; $cell < $this->size(); $cell++) {
242 $defined = array_filter($this->distances, fn(array $distance): bool => $distance[$cell] !== null);
243 $this->dead[$cell] = $this->interior[$cell] && $defined === [];
244 }
245 }
246
247 public function reachable(int $player, array $boxes): array
248 {
249 $seen = array_fill(0, $this->size(), false);
250 $seen[$player] = true;
251 $queue = [$player];
252 for ($head = 0; $head < count($queue); $head++) {
253 foreach ($this->offsets as $offset) {
254 $neighbor = $queue[$head] + $offset;
255 if ($this->interior[$neighbor] && !$seen[$neighbor] && !in_array($neighbor, $boxes, true)) {
256 $seen[$neighbor] = true;
257 $queue[] = $neighbor;
258 }
259 }
260 }
261 return $seen;
262 }
263
264 public function representative(int $player, array $boxes): int
265 {
266 return array_search(true, $this->reachable($player, $boxes), true);
267 }

269 public function walkPath(int $start, int $target, array $boxes): string
270 {
271 $parents = array_fill(0, $this->size(), null);
272 $queue = [$start];
273 for ($head = 0; $head < count($queue); $head++) {
274 $cell = $queue[$head];
275 foreach ($this->offsets as $direction => $offset) {
276 $neighbor = $cell + $offset;
277 if ($this->interior[$neighbor] && $neighbor !== $start && $parents[$neighbor] === null && !in_array($neighbor, $boxes, true)) {
278 $parents[$neighbor] = [$cell, $direction];
279 $queue[] = $neighbor;
280 }
281 }
282 }
283 $path = '';
284 while ($target !== $start) {
285 [$target, $direction] = $parents[$target];
286 $path .= $direction;
287 }
288 return strrev($path);
289 }

291 public function step(Position $position, string $direction): array
292 {
293 $offset = $this->offsets[$direction];
294 $target = $position->player + $offset;
295 if ($this->walls[$target]) {
296 throw new StepFailure('wall');
297 }
298 if (!in_array($target, $position->boxes, true)) {
299 return [new Position($target, $position->boxes), $direction];
300 }
301 $beyond = $target + $offset;
302 if ($this->walls[$beyond] || in_array($beyond, $position->boxes, true)) {
303 throw new StepFailure('blocked');
304 }
305 return [new Position($target, movedBoxes($position->boxes, $target, $beyond)), strtoupper($direction)];
306 }

308 public function pushable(int $box, string $direction, array $boxes, array $reach): bool
309 {
310 $offset = $this->offsets[$direction];
311 return $reach[$box - $offset] && $this->interior[$box + $offset] && !in_array($box + $offset, $boxes, true);
312 }

314 private function sideBlocked(int $box, array $sides, array $boxes, array $assumed): bool
315 {
316 foreach ($sides as $side) {
317 if ($this->walls[$side] || in_array($side, $assumed, true)) {
318 return true;
319 }
320 }
321 if ($this->dead[$sides[0]] && $this->dead[$sides[1]]) {
322 return true;
323 }
324 $assumed[] = $box;
325 foreach ($sides as $side) {
326 if (in_array($side, $boxes, true) && $this->frozen($side, $boxes, $assumed)) {
327 return true;
328 }
329 }
330 return false;
331 }
332
333 public function frozen(int $box, array $boxes, array $assumed = []): bool
334 {
335 return $this->sideBlocked($box, [$box - 1, $box + 1], $boxes, $assumed)
336 && $this->sideBlocked($box, [$box - $this->width, $box + $this->width], $boxes, $assumed);
337 }
338
339 public function hasFrozenBox(array $boxes): bool
340 {
341 foreach ($boxes as $box) {
342 if (!$this->goals[$box] && $this->frozen($box, $boxes)) {
343 return true;
344 }
345 }
346 return false;
347 }

349 public function render(?Position $position = null): array
350 {
351 $symbols = '';
352 for ($cell = 0; $cell < $this->size(); $cell++) {
353 $symbols .= match (true) {
354 $this->walls[$cell] => '#',
355 $position === null => $this->goals[$cell] ? '.' : ($this->dead[$cell] ? 'x' : '-'),
356 $cell === $position->player => $this->goals[$cell] ? '+' : '@',
357 in_array($cell, $position->boxes, true) => $this->goals[$cell] ? '*' : '$',
358 default => $this->goals[$cell] ? '.' : '-',
359 };
360 }
361 return str_split($symbols, $this->width);
362 }
363
364 public function mapCell(string $kind, int $cell): int
365 {
366 $row = intdiv($cell, $this->width);
367 $column = $cell % $this->width;
368 return match ($kind) {
369 'ROT90' => $column * $this->height + $this->height - 1 - $row,
370 'ROT180' => ($this->height - 1 - $row) * $this->width + $this->width - 1 - $column,
371 'ROT270' => ($this->width - 1 - $column) * $this->height + $row,
372 'FLIPH' => $row * $this->width + $this->width - 1 - $column,
373 default => ($this->height - 1 - $row) * $this->width + $column,
374 };
375 }
376
377 public function mapPosition(string $kind, Position $position): Position
378 {
379 $boxes = array_map(fn(int $box): int => $this->mapCell($kind, $box), $position->boxes);
380 sort($boxes, SORT_NUMERIC);
381 return new Position($this->mapCell($kind, $position->player), $boxes);
382 }
383
384 public function transformed(string $kind): self
385 {
386 $width = in_array($kind, ['ROT90', 'ROT270'], true) ? $this->height : $this->width;
387 $symbols = array_fill(0, $this->size(), '');
388 foreach (str_split(implode('', $this->render($this->loadedPosition))) as $cell => $letter) {
389 $symbols[$this->mapCell($kind, $cell)] = $letter;
390 }
391 return new self(str_split(implode('', $symbols), $width));
392 }

395function pushSearch(Level $level, Position $start, array $fixed = [], ?int $target = null, bool $pruning = true, ?int $limit = null): SearchResult
396{
397 $first = new SearchNode($start->boxes, $level->representative($start->player, $start->boxes), null, null);
398 $nodes = [$first];
399 $key = static fn(array $boxes, int $representative): string => implode(',', $boxes) . ':' . $representative;
400 $known = [$key($first->boxes, $first->representative) => true];
401 $expanded = 0;
402 while ($expanded < count($nodes)) {
403 if ($expanded === $limit) {
404 return new SearchResult('limit', $expanded, count($nodes));
405 }
406 $parent = $expanded;
407 $node = $nodes[$expanded++];
408 $reach = $level->reachable($node->representative, $node->boxes);
409 foreach ($node->boxes as $box) {
410 if (in_array($box, $fixed, true)) {
411 continue;
412 }
413 foreach (DIRECTIONS as $direction) {
414 if (!$level->pushable($box, $direction, $node->boxes, $reach)) {
415 continue;
416 }
417 $destination = $box + $level->offsets[$direction];
418 $boxes = movedBoxes($node->boxes, $box, $destination);
419 if ($pruning && ($level->dead[$destination] || $level->hasFrozenBox($boxes))) {
420 continue;
421 }
422 $representative = $level->representative($box, $boxes);
423 $state = $key($boxes, $representative);
424 if (isset($known[$state])) {
425 continue;
426 }
427 $known[$state] = true;
428 $nodes[] = new SearchNode($boxes, $representative, $parent, [$box, $direction]);
429 if (!($target === null ? $level->solved($boxes) : $destination === $target)) {
430 continue;
431 }
432 $pushes = [];
433 $cursor = $nodes[array_key_last($nodes)];
434 while ($cursor->parent !== null) {
435 $pushes[] = $cursor->push;
436 $cursor = $nodes[$cursor->parent];
437 }
438 $position = $start;
439 $notation = '';
440 foreach (array_reverse($pushes) as [$source, $step]) {
441 $offset = $level->offsets[$step];
442 $notation .= $level->walkPath($position->player, $source - $offset, $position->boxes) . strtoupper($step);
443 $position = new Position($source, movedBoxes($position->boxes, $source, $source + $offset));
444 }
445 return new SearchResult('found', $expanded, count($nodes), $notation);
446 }
447 }
448 }
449 return new SearchResult('no solution', $expanded, count($nodes));
450}

452final class Sokoban
453{
454 private Level $level;
455 private Position $initial;
456 private Position $current;
457 private string $history = '';
458 private string $redo = '';
459 private array $snapshots = [];
460 private ?array $solution = null;
461
462 public function __construct(array $rows)

464 $this->load($rows);

467 private function load(array $rows): void
468 {
469 $this->level = new Level($rows);
470 $this->initial = $this->current = $this->level->loadedPosition;
471 $this->history = $this->redo = '';
472 $this->snapshots = [];
473 $this->solution = null;
474 }

478 echo implode(' ', $parts), "\n";

480
481 private function cell(array $arguments, int $index = 0): int
482 {
483 return $this->level->cellAt((int) $arguments[$index], (int) $arguments[$index + 1]);

486 private function simulate(string $directions): array
487 {
488 $position = $this->current;
489 $letters = '';
490 try {
491 foreach (str_split($directions) as $direction) {
492 [$position, $letter] = $this->level->step($position, $direction);
493 $letters .= $letter;
494 }
495 } catch (StepFailure $error) {
496 return [$position, $letters, 'step ' . (strlen($letters) + 1) . ' ' . $error->getMessage()];
497 }
498 return [$position, $letters, null];
499 }

502 {
503 [$this->current, $letters, $failure] = $this->simulate($directions);
504 $this->history .= $letters;
505 if ($letters !== '') {
506 $this->redo = '';
507 }

509 }

513 $count = $this->level->boxesOnGoals($boxes);
514 return "goals $count/" . count($boxes) . ($count === count($boxes) ? ' solved' : '');
515 }

526 private function searchSolution(int $limit): array
527 {
528 if ($this->level->solved($this->current->boxes)) {
529 return ['already solved', null];
530 }
531 $result = pushSearch($this->level, $this->current, limit: $limit);
532 $counts = "expanded {$result->expanded} states {$result->states}";
533 if ($result->outcome !== 'found') {
534 return ["{$result->outcome} $counts", null];
535 }
536 $notation = $result->notation;
537 return ['solution pushes ' . pushCount($notation) . ' moves ' . strlen($notation) . " $counts", $notation];
538 }

540 public function execute(array $words, array $rows = []): void
541 {
542 $name = $words[0];
543 $arguments = array_slice($words, 1);
544 try {
545 if ($name === 'LOAD') {
546 foreach ($rows as $index => $row) {
547 if (strlen($row) !== (int) $arguments[1] || strspn($row, LEVEL_CHARACTERS) !== strlen($row)) {
548 throw new CommandError('bad row ' . ($index + 1));
549 }
550 }
551 $this->load($rows);
552 $this->emit('loaded', "{$this->level->height}x{$this->level->width}");
553 return;
554 }
555 if (!array_key_exists($name, COMMANDS)) {
556 throw new CommandError("unknown command $name");
557 }
558 $kinds = COMMANDS[$name];
559 if (count($arguments) !== count($kinds)) {
560 throw new CommandError('bad arguments');
561 }
562 foreach ($kinds as $index => $kind) {
563 if (!validArgument($kind, $arguments[$index])) {
564 throw new CommandError('bad arguments');
565 }
566 }
567 if (!in_array($name, ['IMPORT', 'VALID'], true)) {
568 if ($this->level->reason !== null) {
569 throw new CommandError('invalid level');
570 }
571 $cells = ['PATH' => 1, 'PUSH' => 1, 'BRING' => 2][$name] ?? 0;
572 for ($index = 0; $index < 2 * $cells; $index += 2) {
573 if (!$this->level->contains((int) $arguments[$index], (int) $arguments[$index + 1])) {
574 throw new CommandError('bad value');
575 }
576 }
577 if (in_array($name, ['UNDO', 'REDO'], true) && (int) $arguments[0] < 1) {
578 throw new CommandError('bad value');
579 }
580 if (in_array($name, ['SOLVE', 'HINT'], true) && ((int) $arguments[0] < 1 || (int) $arguments[0] > 100000)) {
581 throw new CommandError('bad value');
582 }
583 }
584 $this->process($name, $arguments);
585 } catch (CommandError $error) {
586 $this->emit('error:', $error->getMessage());
587 }
588 }
589
590 private function process(string $name, array $arguments): void
591 {
592 $level = $this->level;
593 $position = $this->current;
594 switch ($name) {
595 case 'IMPORT':
596 $rows = array_map(fn(string $part): ?string => decode($part, LEVEL_CHARACTERS), explode('|', $arguments[0]));
597 if (in_array(null, $rows, true) || count($rows) > 10) {
598 throw new CommandError('bad level');
599 }
600 $width = strlen($rows[0]);
601 if ($width < 1 || $width > 10 || array_filter($rows, fn(string $row): bool => strlen($row) !== $width) !== []) {
602 throw new CommandError('bad level');
603 }
604 $this->load($rows);
605 $this->emit('loaded', "{$this->level->height}x{$this->level->width}");
606 break;
607 case 'VALID':
608 $this->emit($level->reason === null ? 'valid' : "invalid {$level->reason}");
609 break;
610 case 'INFO':
611 $this->emit('size', "{$level->height}x{$level->width}", 'walls', array_sum($level->walls), 'cells', array_sum($level->interior), 'boxes', count($position->boxes), 'dead', array_sum($level->dead));
612 break;
613 case 'SHOW':
614 foreach ($level->render($position) as $row) {
615 $this->emit($row);
616 }
617 break;
618 case 'EXPORT':
619 $this->emit('level', implode('|', array_map(encode(...), $level->render($position))));
620 break;
621 case 'MOVE':
622 [$letters, $failure] = $this->takeSteps(strtolower(decode($arguments[0], STEP_CHARACTERS)));
623 if ($letters !== '') {
624 $this->emit('moved', $letters);
625 }
626 if ($failure !== null) {
627 throw new CommandError($failure);
628 }
629 break;
630 case 'PUSH':
631 $box = $this->cell($arguments);
632 $direction = $arguments[2];
633 $offset = $level->offsets[$direction];
634 if (!in_array($box, $position->boxes, true)) {
635 throw new CommandError('no box');
636 }
637 if (!$level->interior[$box + $offset] || in_array($box + $offset, $position->boxes, true)) {
638 throw new CommandError('blocked');
639 }
640 if (!$level->reachable($position->player, $position->boxes)[$box - $offset]) {
641 throw new CommandError('unreachable');
642 }
643 [$letters] = $this->takeSteps($level->walkPath($position->player, $box - $offset, $position->boxes) . $direction);
644 $this->emit('pushed', $letters);
645 break;
646 case 'BRING':
647 $source = $this->cell($arguments);
648 $target = $this->cell($arguments, 2);
649 if (!in_array($source, $position->boxes, true)) {
650 throw new CommandError('no box');
651 }
652 if ($source === $target) {
653 $this->emit('brought', 0);
654 break;
655 }
656 $fixed = array_values(array_diff($position->boxes, [$source]));
657 $result = pushSearch($level, $position, $fixed, $target, false);
658 if ($result->outcome !== 'found') {
659 throw new CommandError('no route');
660 }
661 $this->takeSteps(strtolower($result->notation));
662 $this->emit('brought', pushCount($result->notation), $result->notation);
663 break;
664 case 'PATH':
665 $target = $this->cell($arguments);
666 if (!$level->reachable($position->player, $position->boxes)[$target]) {
667 throw new CommandError('unreachable');
668 }
669 [$letters] = $this->takeSteps($level->walkPath($position->player, $target, $position->boxes));
670 $this->counted('path', $letters);
671 break;
672 case 'UNDO':
673 if ($this->history === '') {
674 throw new CommandError('nothing to undo');
675 }
676 $count = min((int) $arguments[0], strlen($this->history));
677 for ($index = 0; $index < $count; $index++) {
678 $letter = substr($this->history, -1);
679 $this->history = substr($this->history, 0, -1);
680 $direction = strtolower($letter);
681 $offset = $level->offsets[$direction];
682 $boxes = $position->boxes;
683 if ($letter !== $direction) {
684 $boxes = movedBoxes($boxes, $position->player + $offset, $position->player);
685 }
686 $position = new Position($position->player - $offset, $boxes);
687 $this->redo .= $direction;
688 }
689 $this->current = $position;
690 $this->emit('undone', $count);
691 break;
692 case 'REDO':
693 if ($this->redo === '') {
694 throw new CommandError('nothing to redo');
695 }
696 $count = min((int) $arguments[0], strlen($this->redo));
697 for ($index = 0; $index < $count; $index++) {
698 $direction = substr($this->redo, -1);
699 $this->redo = substr($this->redo, 0, -1);
700 [$this->current, $letter] = $level->step($this->current, $direction);
701 $this->history .= $letter;
702 }
703 $this->emit('redone', $count);
704 break;
705 case 'RESET':
706 $this->current = $this->initial;
707 $this->history = $this->redo = '';
708 $this->emit('reset');
709 break;
710 case 'STATUS':
711 $this->emit('moves', strlen($this->history), 'pushes', pushCount($this->history), $this->progress($position->boxes));
712 break;
713 case 'HISTORY':
714 $this->counted('history', $this->history);
715 break;
716 case 'BOXES':
717 $reach = $level->reachable($position->player, $position->boxes);
718 $this->emit('boxes', count($position->boxes));
719 foreach ($position->boxes as $box) {
720 $distances = array_filter(array_column($level->distances, $box), fn(?int $distance): bool => $distance !== null);
721 $directions = implode('', array_filter(DIRECTIONS, fn(string $direction): bool => $level->pushable($box, $direction, $position->boxes, $reach)));
722 $this->emit('box', $level->cellText($box), $distances === [] ? 'dead' : min($distances), $directions === '' ? '-' : $directions);
723 }
724 break;
725 case 'REACH':
726 $reach = $level->reachable($position->player, $position->boxes);
727 $this->emit('reach', array_sum($reach), 'first', $level->cellText(array_search(true, $reach, true)));
728 break;
729 case 'DEADSQUARES':
730 $this->emit('dead', array_sum($level->dead));
731 foreach ($level->render() as $row) {
732 $this->emit($row);
733 }
734 break;
735 case 'DEADLOCK':
736 foreach ($position->boxes as $box) {
737 if ($level->dead[$box]) {
738 $this->emit('deadlock dead', $level->cellText($box));
739 return;
740 }
741 }
742 foreach ($position->boxes as $box) {
743 if (!$level->goals[$box] && $level->frozen($box, $position->boxes)) {
744 $this->emit('deadlock frozen', $level->cellText($box));
745 return;
746 }
747 }
748 $this->emit('no deadlock');
749 break;
750 case 'BOUND':
751 $this->bound();
752 break;
753 case 'SAVE':
754 $this->snapshots[$arguments[0]] = [$position, $this->history];
755 $this->emit('saved', $arguments[0], 'moves', strlen($this->history));
756 break;
757 case 'RESTORE':
758 if (!isset($this->snapshots[$arguments[0]])) {
759 throw new CommandError("no snapshot {$arguments[0]}");
760 }
761 [$this->current, $this->history] = $this->snapshots[$arguments[0]];
762 $this->redo = '';
763 $this->emit('restored', $arguments[0], 'moves', strlen($this->history));
764 break;
765 case 'CHECK':
766 [$after, $letters, $failure] = $this->simulate(strtolower(decode($arguments[0], STEP_CHARACTERS)));
767 if ($failure !== null) {
768 $this->emit('check fail', $failure);
769 } else {
770 $this->emit('check ok moves', strlen($letters), 'pushes', pushCount($letters), $this->progress($after->boxes));
771 }
772 break;
773 case 'SOLVE':
774 case 'HINT':
775 [$summary, $notation] = $this->searchSolution((int) $arguments[0]);
776 if ($name === 'SOLVE') {
777 $this->emit($summary);
778 $this->solution = $notation === null ? null : [$notation, $position];
779 if ($notation !== null) {
780 $this->emit($notation);
781 }
782 } elseif ($notation === null) {
783 $this->emit($summary);
784 } else {
785 preg_match('/\A[udlr]*[UDLR]/', $notation, $match);
786 $this->takeSteps(strtolower($match[0]));
787 $this->emit('hint', $match[0]);
788 }
789 break;
790 case 'PLAY':
791 if ($this->solution === null || $this->solution[1] != $position) {
792 throw new CommandError('no solution');
793 }
794 $notation = $this->solution[0];
795 $this->takeSteps(strtolower($notation));
796 $this->emit('played pushes', pushCount($notation), 'moves', strlen($notation));
797 break;
798 case 'TRANSFORM':
799 $kind = $arguments[0];
800 $this->initial = $level->mapPosition($kind, $this->initial);
801 $this->current = $level->mapPosition($kind, $position);
802 $this->history = turnLetters($this->history, $kind);
803 $this->redo = turnLetters($this->redo, $kind);
804 $this->snapshots = array_map(
805 fn(array $snapshot): array => [$level->mapPosition($kind, $snapshot[0]), turnLetters($snapshot[1], $kind)],
806 $this->snapshots,
807 );
808 if ($this->solution !== null) {
809 $this->solution = [turnLetters($this->solution[0], $kind), $level->mapPosition($kind, $this->solution[1])];
810 }
811 $this->level = $level->transformed($kind);
812 $this->emit('transformed', "{$this->level->height}x{$this->level->width}");
813 break;
814 }
815 }

817 private function bound(): void
818 {
819 $boxes = $this->current->boxes;
820 $best = null;
821 $visit = function (array $order, array $distances, int $total) use (&$visit, &$best, $boxes): void {
822 $index = count($order);
823 if ($index === count($boxes)) {
824 if ($best === null || $total < $best[0]) {
825 $best = [$total, $order, $distances];
826 }
827 return;
828 }
829 foreach ($this->level->goalCells as $goal => $_cell) {
830 if (in_array($goal, $order, true)) {
831 continue;
832 }
833 $distance = $this->level->distances[$goal][$boxes[$index]];
834 if ($distance !== null) {
835 $visit([...$order, $goal], [...$distances, $distance], $total + $distance);
836 }
837 }
838 };
839 $visit([], [], 0);
840 if ($best === null) {
841 $this->emit('bound none');
842 return;
843 }
844 [$total, $order, $distances] = $best;
845 $this->emit('bound', $total);
846 foreach ($boxes as $index => $box) {
847 $this->emit('box', $this->level->cellText($box), 'goal', $this->level->cellText($this->level->goalCells[$order[$index]]), 'pushes', $distances[$index]);
848 }
849 }
850}
